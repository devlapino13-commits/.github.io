CREATE PROCEDURE HP_CHECK_DATA_COMPLETE(
    OBJ_KIND VARCHAR(20),
    OBJ_ID DECIMAL(18,0))
RETURNS (
    RES_CLIENT_FIO VARCHAR(255),
    RES_TXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_IS_CRITICAL INTEGER)
AS
declare variable c_id decimal(18,0);
declare variable object_kind integer;
declare variable c_verify_status integer;
declare variable c_citizen integer;
declare variable c_doc_type integer;
declare variable c_pass_no varchar(100);
declare variable c_pass_date date;
declare variable c_pass_date_expire date;
declare variable citizen_code varchar(100);
declare variable db_id integer;
declare variable adl_id decimal(18,0);
declare variable pl_id decimal(18,0);
declare variable country_name varchar(20);
declare variable ma_id integer;
begin
-- Процедура проверяет полноту данных клиента
  res_txt = '';

  :db_id = f_get_db_id();
  if (db_id = 81001) then
    country_name = 'РФ';
  if (db_id in (39801, 41701)) then
    country_name = 'КР';
  if (db_id = 94401) then
    country_name = 'АЗ';

  ma_id = 0;
  if (obj_kind <> '') then
    object_kind = f_objkind(obj_kind, 'id');

  if (object_kind = 11) then
    ma_id = cast(obj_id as integer);

  res_is_critical = 0;

-- Из переданного параметра получаем список клиентов
  for select mac_from_c_id, c_name_fio
      from miniappclient, miniapp, clients
      where mac_from_ma_id = ma_id and
            ma_id = :ma_id and
            :object_kind = 11 -- Miniapp
            and
            mac_is_poruch = 0 and
            ma_date_end >= '31.12.2999' and
            mac_from_c_id = c_id
      union
      select kc_from_c_id, c_name_fio
      from knclient, clients
      where kc_from_k_id = :obj_id and
            :object_kind = 9 -- Kontract
            and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id
      union
      select c_id, c_name_fio
      from clients
      where c_id = :obj_id and
            :object_kind = 0 -- Clients
      into :c_id, res_client_fio
  do
  begin
    res_txt = '';

-- Получаем данные клиента
    for select coalesce(c_verify_status, 0), coalesce(c_citizen, 0), coalesce(c_doc_type, 0), coalesce(c_pass_no, ''),
               c_pass_date, c_pass_date_expire
        from clients
        where c_id = :c_id
        into c_verify_status, c_citizen, c_doc_type, c_pass_no, c_pass_date, c_pass_date_expire
    do
    begin
      citizen_code = '';
      if (coalesce(c_citizen, 0) <> 0) then
      begin
        citizen_code = (select rd_code
                        from refdata
                        where rd_id = :c_citizen); -- Код гражданства
      end

-- Проверка статуса верификации
      if (c_verify_status = -2) then
      begin
        res_txt = 'Верификация: Данные не верные.';
        res_is_critical = 2;
        suspend;
      end
      if (c_verify_status = 0) then
      begin
        res_txt = 'Верификация: Данные не полные.';
        res_is_critical = 2;
        suspend;
      end
      if (c_verify_status = 1) then
      begin
        res_txt = 'Верификация: В ожидании.';
        res_is_critical = 2;
        suspend;
      end

-- Проверка гражданства
      if (c_citizen = 0) then
      begin
        res_txt = 'Гражданство не указано';
        res_is_critical = 2;
        suspend;
      end

-- Проверка наличия типа документа
      if (c_doc_type = 0) then
      begin
        res_txt = 'Тип удостоверения личности не указан';
        res_is_critical = 2;
        suspend;
      end

-- Проверка номера документа
      if (c_pass_no = '') then
      begin
        res_txt = 'Номер паспорта не заполнен';
        res_is_critical = 2;
        suspend;
      end

-- Проверка даты выдачи удостоверения личности
      if (c_pass_date is null) then
      begin
        res_txt = 'Дата выдачи паспорта не заполнена';
        res_is_critical = 2;
        suspend;
      end

-- Проверка срока действия удостоверения личности
      if (not(current_date between c_pass_date and coalesce(c_pass_date_expire, '31.12.2999'))) then
      begin
        res_txt = 'Срок паспорта не действителен';
        res_is_critical = 1;
        suspend;
      end

-- Проверка кредитования по гражданству РФ
      if (db_id = 81001 and
          citizen_code not in ('Citizen.RU', 'Citizen.KG', 'Citizen.TJ', 'Citizen.UZ')) then
      begin
        res_txt = 'Кредитование возможно только граждан: РФ, КР, ТЖ и УЗ';
        res_is_critical = 1;
        suspend;
      end

-- Проверка кредитования по гражданству КР
      if (db_id in (39801, 41701) and
          citizen_code not in ('Citizen.KG')) then
      begin
        res_txt = 'Кредитование возможно только граждан КР';
        res_is_critical = 1;
        suspend;
      end

-- Проверка кредитования по гражданству АЗ
      if (db_id = 94401 and
          citizen_code not in ('Citizen.AZ')) then
      begin
        res_txt = 'Кредитование возможно только граждан АЗ';
        res_is_critical = 1;
        suspend;
      end

-- Проверка подтвержденного номера телефона
      if ((select cont_value
           from hp_contdata_get_cell_main(0, :c_id)) = '') then
      begin
        res_txt = 'Нет подтвержденного номера телефона';
        res_is_critical = 1;
        suspend;
      end

-- Проверка юридического адреса
      adl_id = 0;
      pl_id = 0;
      select adc_from_adl_id, adl_from_pl_id
      from addrclient, addrlist
      where adc_from_c_id = :c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id
      into :adl_id, pl_id;
      if (adl_id is null) then
        adl_id = 0;
      if (pl_id is null) then
        pl_id = 0;

      if (mod(adl_id, 1000000000) = 0) then
      begin
        res_txt = 'Юридически адрес не заполнен';
        res_is_critical = 1;
        suspend;
      end
      else
      if (db_id not in (39801, 41701)) then
      if (left(pl_id, 3) <> left(db_id, 3)) then -- Проверка выбран ли населенный пункт из нужной страны
      begin
        res_txt = 'В юридическом адресе населенный пункт не из ' || country_name;
        res_is_critical = 1;
        suspend;
      end

-- Проверка фактического адреса (место проживания)
      adl_id = 0;
      pl_id = 0;
      select adc_from_adl_id, adl_from_pl_id
      from addrclient, addrlist
      where adc_from_c_id = :c_id and
            adc_address_type = 2 and
            adc_from_adl_id = adl_id
      into :adl_id, pl_id;
      if (adl_id is null) then
        adl_id = 0;
      if (pl_id is null) then
        pl_id = 0;

      if (mod(adl_id, 1000000000) = 0) then
      begin
        res_txt = 'Адрес проживания не заполнен';
        res_is_critical = 1;
        suspend;
      end
      else
      if (left(pl_id, 3) <> left(db_id, 3)) then -- Проверка выбран ли населенный пункт из нужной страны
      begin
        res_txt = 'В адресе проживания населенный пункт не из ' || country_name;
        res_is_critical = 1;
        suspend;
      end

-- Если не гражданин текущей страны, проверяем наличие адреса проживания на родине
      if ((citizen_code <> 'Citizen.RU' and
          db_id = 81001) or (citizen_code <> 'Citizen.KG' and
          db_id in (39801, 41701)) or (citizen_code <> 'Citizen.AZ' and
          db_id = 94401)) then
      begin
        adl_id = 0;
        pl_id = 0;
        select adc_from_adl_id, adl_from_pl_id
        from addrclient, addrlist
        where adc_from_c_id = :c_id and
              adc_address_type = 5 and
              adc_from_adl_id = adl_id
        into :adl_id, pl_id;
        if (adl_id is null) then
          adl_id = 0;
        if (pl_id is null) then
          pl_id = 0;

        if (mod(adl_id, 1000000000) = 0) then
        begin
          res_txt = 'Адрес проживания на родине не заполнен';
          res_is_critical = 1;
          suspend;
        end
        else
        if (left(pl_id, 3) = left(db_id, 3)) then -- Адрес на родине должен быть не из текущей страны
        begin
          res_txt = 'Адрес на родине указан из ' || country_name;
          res_is_critical = 1;
          suspend;
        end
      end
    end
  end
end
