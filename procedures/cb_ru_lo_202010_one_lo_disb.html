      if (aarep_fact_all_float is null) then
      aarep_fact_all_float = 0;
    if (aarep_month_plan_os_float is null) then
      aarep_month_plan_os_float = 0;
    if (aarep_month_plan_perc_float is null) then
      aarep_month_plan_perc_float = 0;
    if (aarep_fact_other_float is null) then
      aarep_fact_other_float = 0;


    aarep_returnment_koef = 1.0; --конэфициент факт делим на план
    if (aarep_plan_all <> 0) then
      aarep_returnment_koef = round(aarep_fact_planned / aarep_plan_all, 4);
    else
      aarep_returnment_koef = 0.0;
    if (aarep_returnment_koef > 1.0) then
      aarep_returnment_koef = 1.0;

    if (rep_date_end < '01.05.2019') then
    begin
      aarep_bonus_plus = round((aarep_fact_all / 1000.0) * koef_bonus, 0);
      aarep_bonus_ideal = round(((aarep_month_plan_os + aarep_month_plan_perc + aarep_fact_other) / 1000.0) * koef_bonus, 0);
    end else
    if (:rep_date_end < '01.09.2020') then
    begin
      aarep_fact_all = aarep_fact_all - aarep_fact_all_zalog;
      aarep_bonus_plus = round((aarep_fact_all / 1000.0) * koef_bonus, 0) + round((aarep_fact_all_zalog / 1000.0) * koef_bonus_zalog, 0); --считаем отдельно прочее с залогом
      aarep_month_plan_os = aarep_month_plan_os - aarep_month_plan_os_zalog;
      aarep_month_plan_perc = aarep_month_plan_perc - aarep_month_plan_perc_zalog;
      aarep_fact_other = aarep_fact_other - aarep_fact_other_zalog;
      aarep_bonus_ideal = round(((aarep_month_plan_os + aarep_month_plan_perc + aarep_fact_other) / 1000.0) * koef_bonus, 0) + round(((aarep_month_plan_os_zalog + aarep_month_plan_perc_zalog + aarep_fact_other_zalog) / 1000.0) * koef_bonus_zalog, 0);
    end else
    begin
      aarep_fact_all = aarep_fact_all - aarep_fact_all_zalog - aarep_fact_all_float;
      aarep_bonus_plus = round((aarep_fact_all / 1000.0) * koef_bonus, 0) + round((aarep_fact_all_zalog / 1000.0) * koef_bonus_zalog, 0) + + round((aarep_fact_all_float / 1000.0) * koef_bonus_float, 0); --считаем отдельно прочее с залогом
      aarep_month_plan_os = aarep_month_plan_os - aarep_month_plan_os_zalog - aarep_month_plan_os_float;
      aarep_month_plan_perc = aarep_month_plan_perc - aarep_month_plan_perc_zalog - aarep_month_plan_perc_float;
      aarep_fact_other = aarep_fact_other - aarep_fact_other_zalog - aarep_fact_other_float;
      aarep_bonus_ideal = round(((aarep_month_plan_os + aarep_month_plan_perc + aarep_fact_other) / 1000.0) * koef_bonus, 0) + round(((aarep_month_plan_os_zalog + aarep_month_plan_perc_zalog + aarep_fact_other_zalog) / 1000.0) * koef_bonus_zalog, 0) + round(((aarep_month_plan_os_float + aarep_month_plan_perc_float + aarep_fact_other_float) / 1000.0) * koef_bonus_float, 0);
    end

    if (aarep_plan_all > aarep_fact_planned) then
      aarep_bonus_minus = round(((aarep_plan_all - aarep_fact_planned) / 1000.0) * koef_penalty, 0);
    else
      aarep_bonus_minus = 0;

    select jk_system_code
    from hr_pstatus h, joblist, jobkind
    where :rep_date_end between ps_date_begin and ps_date_end and
          h.ps_from_c_id = :c_id and
          ps_is_working = 1 and
          ps_status = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id
    into :jk_system_code;

    select lo_stage_month_full
    from hp_get_lo_stage_full(:c_id, :rep_date_end, :jk_system_code)
    into :lo_stage_month_full;

    if (coalesce(lo_stage_month_full, -1) < 0) then
      lo_stage_month_full = 0;

    aarep_bonus_minus_disb = 0;
    kol_disb_kont = 0;
    scd_work_hours = 0;

    sfb_fix_bonus = 0;
    sfb_total_work = 0;
    sfb_fact_work = 0;

    select coalesce(sfb_fix_bonus, 0), coalesce(sfb_total_work, 0), coalesce(sfb_fact_work, 0)
    from sa_fixbonus s
    where sfb_from_c_id = :c_id and
          sfb_period = :rep_period_end
    into :sfb_fix_bonus, sfb_total_work, sfb_fact_work;

    if (sfb_total_work > 0) then
      sfb_fix_bonus = round((sfb_fix_bonus / sfb_total_work) * sfb_fact_work, 2);  --Фикс.бонус к начислению
    else
    if (sfb_total_work = 0) then
      sfb_fix_bonus = 0.00;

    select count(ks.ks_from_k_id)
    from kontstate ks
    where ks.ks_ol_is_delete = 0 and
          ks.ks_state = 1 and-- говорит о том что кредит активный
          ks.ks_date_begin between :rep_date_begin and :rep_date_end and
          ks.ks_kc_id = :c_id
    into :kol_disb_kont;

    if (lo_stage_month_full>4) then
    begin
    if (sfb_fact_work > 0) then
      aarep_bonus_minus_disb = iif(kol_disb_kont / sfb_fact_work >= 1, 0, 3000);
    end
    if (sfb_fact_work = 0) then
      aarep_bonus_minus_disb = 0;

    aarep_bonus = 0;
    aarep_bonus_total_fact = 0;
    aarep_bonus_total_to_pay = 0;
    aarep_bonus_constant = 5747;
    aarep_bonus_total = aarep_bonus_plus - aarep_bonus_minus;
    aarep_bonus_total_fact = aarep_bonus_total - aarep_oklad - aarep_bonus_constant;
    aarep_bonus = aarep_bonus_total - aarep_oklad - aarep_bonus_constant;
    if (aarep_bonus < 0) then
      aarep_bonus =0;
    --    штраф по заявкам
    lp_amount = null;
    select coalesce(sum(lp_amount),0)
    from lo_penalty
    where lp_is_active = 1 and
          lp_date between :rep_date_begin and :rep_date_end and
          lp_from_c_id = :c_id
    into lp_amount;
    aarep_bonus_total_to_pay = aarep_bonus + sfb_fix_bonus-lp_amount-aarep_bonus_minus_disb;
    if (aarep_bonus_total_to_pay < 0) then
      aarep_bonus_total_to_pay = 0;

    aarep_day_bonus_plus = aarep_bonus_plus;
    aarep_day_bonus_minus = aarep_bonus_minus;
    aarep_day_bonus_total = aarep_bonus_total;
    aarep_day_returnment_koef = aarep_returnment_koef;

    aarep1_plan_all = 0;
    aarep1_fact_all = 0;
    aarep1_fact_planned = 0;
    aarep1_fact_other = 0;

    --    стаж СВЗ
    lo_stage_month = 0;
    select lo_stage_month
    from hp_get_lo_stage(:c_id, :rep_date_end)
    into lo_stage_month;

    if ((extract(day from :rep_date_end) > 1) and
        (need_one_day = 1)) then
    begin
      select sum(aarep_plan_all), sum(aarep_fact_all), sum(aarep_fact_planned), sum(aarep_fact_other),
             sum(iif(pr_code containing '.zalog_car.', aarep_fact_all, 0)),
             sum(iif(pr_code containing '.float_percent.' and k_disb_date >= '01.09.2020', aarep_fact_all, 0))

      from cb_ru_lo_202010_one_tmp(:reg_id, :c_id, :rep_date_begin, :rep_date_end - 1)
      into :aarep1_plan_all, :aarep1_fact_all, :aarep1_fact_planned, :aarep1_fact_other, aarep1_fact_all_zalog, aarep1_fact_all_float;

      if (aarep1_plan_all is null) then
        aarep1_plan_all = 0;
      if (aarep1_fact_all is null) then
        aarep1_fact_all = 0;
      if (aarep1_fact_planned is null) then
        aarep1_fact_planned = 0;
      if (aarep1_fact_other is null) then
        aarep1_fact_other = 0;
      if (aarep1_fact_all_zalog is null) then
        aarep1_fact_all_zalog = 0;
      if (aarep1_fact_all_float is null) then
        aarep1_fact_all_float = 0;


      aarep_day_returnment_koef = 1.0;
      if (aarep_plan_all - aarep1_plan_all <> 0) then
        aarep_day_returnment_koef = round((aarep_fact_planned - aarep1_fact_planned) / (aarep_plan_all - aarep1_plan_all), 4);
      else
        aarep_day_returnment_koef = 0.0;
      if (aarep_day_returnment_koef > 1.0) then
        aarep_day_returnment_koef = 1.0;
      if (aarep_day_returnment_koef < 0) then
        aarep_day_returnment_koef = 1.0;

      if (rep_date_end < '01.05.2019') then
      begin
        aarep_day_bonus_plus = round((aarep1_fact_all / 1000.0) * koef_bonus, 0);
      end else
      if (:rep_date_end <= '01.09.2020') then
      begin
        aarep1_fact_all = aarep1_fact_all - aarep1_fact_all_zalog;
        aarep_day_bonus_plus = round((aarep1_fact_all / 1000.0) * koef_bonus, 0) + round((aarep1_fact_all_zalog / 1000.0) * koef_bonus_zalog, 0);
      end else
      begin
        aarep1_fact_all = aarep1_fact_all - aarep1_fact_all_zalog - aarep1_fact_all_float;
        aarep_day_bonus_plus = round((aarep1_fact_all / 1000.0) * koef_bonus, 0) + round((aarep1_fact_all_zalog / 1000.0) * koef_bonus_zalog, 0) + round((aarep1_fact_all_float / 1000.0) * koef_bonus_float, 0);
      end

      if (aarep1_plan_all > aarep1_fact_planned) then
        aarep_day_bonus_minus = round(((aarep1_plan_all - aarep1_fact_planned) / 1000.0) * koef_penalty, 0);
      else
        aarep_day_bonus_minus = 0;
      aarep_day_bonus_total = aarep_day_bonus_plus - aarep_day_bonus_minus;

      aarep_day_bonus_plus = aarep_bonus_plus - aarep_day_bonus_plus;
      aarep_day_bonus_minus = aarep_bonus_minus - aarep_day_bonus_minus;
      aarep_day_bonus_total = aarep_bonus_total - aarep_day_bonus_total;
    end
    if (rep_c_id <> 0) then
      suspend;
    else
    begin
      if ((aarep_port_balance_all <> 0) or (aarep_bonus_total <> 0)) then
        suspend;
    end
  end
end
