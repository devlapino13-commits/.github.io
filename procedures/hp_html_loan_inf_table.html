CREATE PROCEDURE HP_HTML_LOAN_INF_TABLE(
    K_ID D_ID,
    MA_ID D_ID,
    CONFIG D_STR_CODE,
    K_DISB_DATE D_DATE,
    LANG D_STR_LANG)
RETURNS (
    RES_HTML D_STR_8100)
AS
declare variable title_text d_str_255;
declare variable tmp_id d_id;
declare variable c_name_fio d_str_100;
declare variable res_link d_str_2048;
begin
  title_text = (case config
                  when 'extinfo_animals_check' then 'Информация о количестве скота'
                  when 'extinfo_sal_info' then 'Информация о заработной плате'
                  when 'extinfo_rudb_check' then 'Информация о проверке в РФ БД'
                  when 'extinfo_job' then 'Информация о работе'
                  when 'extinfo_pension' then 'Информация о пенсии'
                  when 'extinfo_inn_entity' then 'Информация о проверке ИНН для юр.лица'
                  when 'extinfo_address_fact' then 'Информация о фактического адреса'
                  else ''
                end);

  res_html = blob_append(res_html, f_format('<h2>%1</h2>', f_lang(lang, :title_text)) || '<table width="100%"><col width="100%">');
  for select c_id, c_name_fio
      from knclient, clients
      where kc_from_k_id = :k_id and
            kc_from_c_id = c_id and
            kc_amount_approved > 0
      union
      select c_id, c_name_fio
      from miniappclient, clients
      where mac_from_ma_id = :ma_id and
            mac_from_c_id = c_id and
            mac_amount > 0
      order by 2
      into tmp_id, c_name_fio
  do
  begin
    res_link = f_uri_html(c_name_fio, :config, f_lang(lang, :title_text), 'clients', tmp_id, k_disb_date, k_disb_date);
    res_html = blob_append(res_html, '<tr><td>' || res_link || '</td></tr>');
  end
  res_html = blob_append(res_html, '</table>');
  suspend;
end
