CREATE PROCEDURE HP_CORRECT_KCP_BEGIN_SALDO
RETURNS (
    UPDATED INTEGER)
AS
declare variable k_id decimal(18,0);
declare variable perc_saldo_end decimal(18,4) = 0;
declare variable kommis_saldo_end decimal(18,4) = 0;
declare variable perc_saldo_prior decimal(18,4) = 0;
declare variable kommis_saldo_prior decimal(18,4) = 0;
declare variable k_id_prior decimal(18,0);
declare variable kcp_perc_begin_saldo decimal(18,4) = 0;
declare variable kcp_kommis_begin_saldo decimal(18,4) = 0;
declare variable k_period integer;
declare variable kcp_from_c_id id_type;
begin
  perc_saldo_prior = 0;
  kommis_saldo_prior = 0;
  k_id_prior = 0;
  updated = 0;
  for select kcp_period, kcp_from_k_id, kcp_perc_saldo, kcp_kommis_saldo, kcp_perc_begin_saldo, kcp_kommis_begin_saldo,
             kcp_from_c_id
      from kontcalcperc
      order by kcp_from_k_id, kcp_period
      into :k_period, k_id, :perc_saldo_end, :kommis_saldo_end, :kcp_perc_begin_saldo, :kcp_kommis_begin_saldo, kcp_from_c_id
--  as cursor cur
  do
  begin
    if (k_id = k_id_prior) then
    begin
      if ((kcp_perc_begin_saldo <> perc_saldo_prior) or (kcp_kommis_begin_saldo <> kommis_saldo_prior)) then
      begin
        update kontcalcperc
        set kcp_perc_begin_saldo = :perc_saldo_prior,
            kcp_kommis_begin_saldo = :kommis_saldo_prior
        where kcp_period = :k_period and
              kcp_from_k_id = :k_id and
              kcp_from_c_id = :kcp_from_c_id;
--        where current of cur;
        updated = updated + 1;
      end
    end
    else
    begin
      update kontcalcperc
      set kcp_perc_begin_saldo = 0,
          kcp_kommis_begin_saldo = 0
      where kcp_period = :k_period and
            kcp_from_k_id = :k_id and
            kcp_from_c_id = :kcp_from_c_id;
--      where current of cur;
    end
    perc_saldo_prior = :perc_saldo_end;
    kommis_saldo_prior = :kommis_saldo_end;
    k_id_prior = k_id;
  end
  suspend;
end
