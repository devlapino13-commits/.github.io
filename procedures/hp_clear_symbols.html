CREATE PROCEDURE HP_CLEAR_SYMBOLS(
    TXT VARCHAR(1024))
RETURNS (
    RES_TXT VARCHAR(1024))
AS
begin
  if (txt is null) then
    txt = '';
  res_txt = txt;
  res_txt = replace(res_txt, '-', ' ');
  res_txt = replace(res_txt, '_', ' ');
  res_txt = replace(res_txt, '"', ' ');
  res_txt = replace(res_txt, '(', ' ');
  res_txt = replace(res_txt, ')', ' ');
  res_txt = replace(res_txt, '.', ' ');
  res_txt = replace(res_txt, ',', ' ');
  res_txt = replace(res_txt, '''', ' ');
  res_txt = replace(res_txt, '''', ' ');
  res_txt = f_replace_all(res_txt, '  ', ' ');
  res_txt = trim(res_txt);
  suspend;
end
