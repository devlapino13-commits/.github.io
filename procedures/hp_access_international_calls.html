CREATE PROCEDURE HP_ACCESS_INTERNATIONAL_CALLS(
    REP_DATE_END DATE)
RETURNS (
    OKS_REGION_ID ID_TYPE,
    FIRSTDAY_CMONTH DATE,
    PAR7_PMONTH PERC_TYPE,
    LDAY_CMONTH DATE,
    PAR7_CMONTH PERC_TYPE,
    REG_NAME VARCHAR(50),
    TEXT_MESSAGE VARCHAR(10))
AS
begin
  firstday_cmonth = (select cast('01.' || cast(extract(month from cast(:rep_date_end as date)) as varchar(2)) || '.' || cast(extract(year from cast(:rep_date_end as date)) as varchar(4)) as date)
                     from rdb$database); -- Начало месяца

  lday_cmonth = /*cast(select cast('01.'||cast(extract(month from dateadd(month, 1, cast(:REP_DATE_end as date))) as varchar(2))||'.'||
                    cast(extract(year from cast(:REP_DATE_end as date)) as varchar(4)) as date) - 1
                 from rdb$database) as date;  -- последняя дата текущего месяца */
  rep_date_end;

  for select oks_region_id, cast(delinq_balance_7 / balance_b * 100 as curr_type)
      from (select oks_region_id,
                   sum(iif(cast(:firstday_cmonth as date) - oks_delinq_date > 7, oks_balance, 0)) as delinq_balance_7,
                   sum(oks_balance) as balance_b
            from oksummary
            where :firstday_cmonth between oks_date_begin and oks_date_end and
                  oks_balance > 0 and
                  oks_close_state = 0
            group by oks_region_id
            having sum(oks_balance) - sum(iif(cast(:firstday_cmonth as date) - (select k_disb_date
                                                                                from kontract
                                                                                where k_id = oks_from_k_id) < 30, oks_balance, 0)) > 0) a
      into oks_region_id, :par7_pmonth
  do
  begin
    select cast(delinq_balance_7 / balance_b * 100 as curr_type)
    from (select sum(iif(cast(:lday_cmonth as date) - oks_delinq_date > 7, oks_balance, 0)) as delinq_balance_7,
                 sum(oks_balance) as balance_b
          from oksummary
          where :lday_cmonth between oks_date_begin and oks_date_end and
                oks_balance > 0 and
                oks_close_state = 0 and
                oks_region_id = :oks_region_id having sum(oks_balance) - sum(iif(cast(:lday_cmonth as date) - (select k_disb_date
                                                                                                               from kontract
                                                                                                               where k_id = oks_from_k_id) < 30, oks_balance, 0)) > 0) a
    into :par7_cmonth;

    par7_pmonth = round(par7_pmonth, 1);
    par7_cmonth = round(par7_cmonth, 1);

    if (par7_cmonth >= par7_pmonth and
        par7_cmonth > 7) then
      text_message = 'блокировка';

    if (par7_cmonth > 7) then
      text_message = 'блокировка';

    if (par7_cmonth < par7_pmonth) then
      text_message = 'пополнение';

    if (par7_cmonth <= 7) then
      text_message = 'пополнение';

    select reg_name
    from region
    where reg_unicode = :oks_region_id
    into reg_name;

    suspend;
  end
end
