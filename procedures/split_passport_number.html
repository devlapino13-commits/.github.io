CREATE PROCEDURE SPLIT_PASSPORT_NUMBER(
    PK_ID D_INTEGER,
    PASS_NO D_STR_CODE)
RETURNS (
    PASS_SERIA D_STR_10,
    PASS_NUMBER D_STR_CODE)
AS
declare variable DRM_CODE D_STR_255;
begin
    -- RFCKG-3780 Разбивка на серию и номер паспорта
    pass_no = replace(pass_no, ' ', '');
    pass_seria = '';
    pass_number = '';

    select drm.drm_code
    from docrecogmain drm
    inner join passkind pk on pk.pk_from_drm_id = drm.drm_id
    where pk.pk_id = :pk_id
    rows 1
    into drm_code;

    if(drm_code = 'rus.passport.national') then --внутрен. паспорт рф (rus.passport.national), 4 первых символа серия, далее убирается пробел и все чо за ним - это номер
    begin
        pass_seria = left(pass_no, 4);
        pass_seria =  left(pass_seria, 2) || ' ' || right(pass_seria, 2);
        pass_number = substring(pass_no from 5);
    end
    else if (drm_code = 'd_90') then --паспорт тадж. (d_90), 1 первый символ серия, далее убирается пробел и все чо за ним - это номер
    begin
        pass_seria = left(pass_no, 1);
        pass_number = substring(pass_no from 2);
    end
    else if (drm_code = 'tjk.passport.type1') then --загран. паспорт тадж. (tjk.passport.type1), серия отсутствует. все 9 символов относятся к номеру
        pass_number = pass_no;
    else  -- остальные 2 первых символа, остальное номер
    begin
        pass_seria = left(pass_no, 2);
        pass_number = substring(pass_no from 3);
    end
    suspend;
end
