CREATE PROCEDURE AAREP_REQSIGN_PRINT(
    REPORT_RM_ID DECIMAL(18,0))
RETURNS (
    RS_FROM_RM_ID DECIMAL(18,0),
    RS_NUMBER INTEGER,
    RS_USER VARCHAR(100),
    RS_FROM_UN_ID DECIMAL(18,0),
    RS_DATETIME TIMESTAMP,
    RS_STATUS_TEXT VARCHAR(20))
AS
declare variable rs_status integer;
declare variable rs_c varchar(100);
declare variable rs_st integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  rs_c = '';
  rs_st = -5;
  for select rs_from_rm_id, rs_number, rs_status, c_name_fio, rs_from_un_id, rs_datetime
      from reqsign, clients
      where rs_from_rm_id = :report_rm_id and
            rs_from_c_id = c_id
      order by rs_datetime desc
      into :rs_from_rm_id, :rs_number, :rs_status, :rs_user, :rs_from_un_id, :rs_datetime
  do
  begin
    if ((rs_c <> rs_user) or (rs_status <> rs_st)) then
    begin
      if (rs_status = -1) then
        rs_status_text = 'Аннулирован';
      if (rs_status = 0) then
        rs_status_text = 'В процессе';
      if (rs_status = 1) then
        rs_status_text = 'Утвержден';
      if (rs_status = 2) then
        rs_status_text = 'Отказано';
      if (rs_status = 3) then
        rs_status_text = 'Исполнен';
      if (rs_status = 4) then
        rs_status_text = 'Проверен';
      if (rs_status = 5) then
        rs_status_text = 'Согласован';
      suspend;
    end
    rs_c = rs_user;
    rs_st = rs_status;
  end
end
