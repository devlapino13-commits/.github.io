CREATE PROCEDURE AAREP_POS_TMP(
    REP_TYPE D_STR_100,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    TRM_TYPE D_STR_10,
    TRM_IPC_ID D_STR_20,
    TRM_INVENT_STR VARCHAR(50),
    TRM_SERIAL_NO D_STR_255,
    ITO_DATE_BEGIN D_DATE,
    TRM_SIMCARD_NO D_STR_20,
    ITO_OBJECT_KIND D_INTEGER,
    ITO_OBJECT_ID D_ID,
    ITO_TRM_ID D_INTEGER,
    ITO_STATUS SMALLINT)
AS
declare variable sql d_str_8100;
begin
    sql = '
        select iif (ipt.trm_type = 1, ''ТСП'', ''ПВН''),
                  ipt.trm_ipc_id,
                  ipt.trm_invent_str,
                  ipt.trm_serial_no,
                  ipt.trm_simcard_no,
                  ito.ito_object_kind,
                  ito.ito_object_id,
                  ito.ito_date_begin,
                  ito.ito_trm_id,
                  ito.ito_status
        from ipc_terminals ipt
        inner join ipc_terminal_ownership ito on ito.ito_trm_id = ipt.trm_id
        where ';

    if (rep_type = 'in_process') then
        sql = sql || ' :rep_date_end between ito.ito_date_begin
                       and coalesce(ito.ito_date_end, ''31.12.9999'')
                       and ito.ito_status in (0, 1)
                       and :rep_date_begin is null';

    else if (rep_type = 'active') then
        sql = sql || ' :rep_date_end between ito.ito_date_begin
                       and coalesce(ito.ito_date_end, ''31.12.9999'')
                       and ito.ito_status = 3
                       and :rep_date_begin is null';

    else if (rep_type = 'installed') then
        sql = sql || ' ito.ito_date_begin between :rep_date_begin
                       and :rep_date_end
                       and ito.ito_status = 0';

    else if (rep_type = 'closed') then
        sql = sql || ' ito.ito_date_end  between :rep_date_begin
                       and :rep_date_end
                       and ito.ito_status = -1';
    for execute statement(sql)
    (rep_date_begin := rep_date_begin,
     rep_date_end := rep_date_end)
    into
        :trm_type,
        :trm_ipc_id,
        :trm_invent_str,
        :trm_serial_no,
        :trm_simcard_no,
        :ito_object_kind,
        :ito_object_id,
        :ito_date_begin,
        :ito_trm_id,
        :ito_status
    do
        suspend;
end
