CREATE PROCEDURE HP_EDOC_CLIENT_CONTINFO(
    EDOC_KIND D_STR_CODE,
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    CONT_INFO D_STR_255,
    MSG_TYPE D_INTEGER,
    EDC_NAME D_STR_255,
    CDA_ID D_ID)
AS
declare variable EDOC_ID D_ID;
declare variable EDC_STATE D_SMALLINT;
declare variable EDC_USE_LAST_CONTDATA D_SMALLINT;
declare variable CONT_KIND D_STR_CODE;
declare variable DB_ID D_INTEGER;
begin
  -- Возвращает контактную информацию участника эл.документа

  edoc_id = f_edoc_kind(edoc_kind, 'id');
  edoc_kind = f_edoc_kind(edoc_kind, 'code');

  res_text = '';
  cont_info = '';
  msg_type = 3; -- quick sms
  cda_id = null;

  select edc_state, edc_name, edc_use_last_contdata
  from edoc_cfg
  where edc_id = :edoc_id
  into edc_state, edc_name, edc_use_last_contdata;

  if (edc_state = 0 or
      c_id is null) then
  begin
    suspend;
    exit;
  end

  db_id = f_get_db_id();

  -- если договор АСП и клиент активный, нужен подтвержденный номер
  if (db_id = 81001 and edoc_kind = 'asp.agreement' and
      (exists(select *
              from clientstate
              where cs_from_c_id = :c_id and
                    cs_client_state = 1))) then
    edc_use_last_contdata = 0;

  cont_info = f_cont_info('clients', c_id, 'cell_main', trim(iif(edc_use_last_contdata = 1, 'last', 'ok')));
  cont_kind = 'cell';

  if (cont_info is null) then
    cont_info = '';

  if ((db_id in (39801, 41701, 81001) and char_length(cont_info) = 11 and cont_info starting '79') or
      (db_id = 41701 and char_length(cont_info) = 12 and cont_info starting '996') or
      (db_id = 39801 and char_length(cont_info) = 11 and cont_info starting '7') or
      (db_id = 94401 and char_length(cont_info) = 12 and cont_info starting '994')) then
      cda_id = f_contdata_id(c_id, cont_kind, cont_info, 1, current_date, 'add', :un_id);
 else
  begin
    if (edc_use_last_contdata = 1) then
      res_text = 'Нет основного номера телефона';
    else
      res_text = 'Нет подтвержденного номера телефона';
  end

  suspend;
end
