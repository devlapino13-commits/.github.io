CREATE PROCEDURE AAREP_TAX_NSP_REPORT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    TAX_KIND VARCHAR(50),
    TAX_BASE DECIMAL(18,4),
    TAX_KOEF DECIMAL(15,7),
    TAX_AMOUNT DECIMAL(18,4),
    TAX_REGION VARCHAR(100),
    TAX_BRANCH VARCHAR(100),
    TAX_REGION_ID INTEGER,
    TAX_BRANCH_ID INTEGER)
AS
declare variable br_reg_code varchar(20);
begin
  for select tax_kind, sum(tax_base), tax_koef, br.reg_code, br.reg_name, br.reg_unicode,
             reg.reg_code || ' ' || reg.reg_name, tax_region_id
      from aarep_tax_nsp_temp(:report_region_id, :report_date_begin, :report_date_end), region reg, region br
      where tax_region_id = reg.reg_unicode and
            reg.reg_head_office = br.reg_unicode
      group by tax_kind, tax_koef, tax_region_id, br.reg_code, br.reg_name, br.reg_unicode, reg.reg_code, reg.reg_name
      order by tax_koef, br.reg_code, reg.reg_code, tax_region_id
      into :tax_kind, :tax_base, :tax_koef, :br_reg_code, :tax_branch, :tax_branch_id, :tax_region, :tax_region_id
  do
  begin
    tax_amount = :tax_base * :tax_koef;
    tax_amount = cast(:tax_amount / 100 as decimal(15,2));
    suspend;
  end
end
