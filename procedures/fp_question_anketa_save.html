CREATE PROCEDURE FP_QUESTION_ANKETA_SAVE(
    KC_ID ID_TYPE,
    QU_ID INTEGER,
    AN_ID INTEGER,
    AN_OTHER STR_255)
AS
declare variable k_id id_type;
declare variable c_id id_type;
begin
  if (kc_id > 0) then begin
    select
      kc_from_k_id,
      kc_from_c_id
    from knclient
    where kc_id = :kc_id
    into
      k_id,
      c_id;
  end else begin
    select
      z_from_k_id,
      z_from_c_id
    from zalog
    where z_id = - :kc_id
    into
      k_id,
      c_id;
  end

  if (an_id = 1) then
  begin
    update or insert into fp_question_anketa (fqa_from_kc_id, fqa_from_k_id, fqa_from_c_id, fqa_qu_id, fqa_an_id, fqa_an_other)
    values (:kc_id, :k_id, :c_id, :qu_id, :an_id, :an_other)
    matching (fqa_from_k_id, fqa_from_c_id, fqa_qu_id);
  end
  else
  begin
    delete from fp_question_anketa
    where fqa_from_k_id = :k_id and
      fqa_from_c_id = :c_id and
      fqa_qu_id = :qu_id;
  end
end
