CREATE PROCEDURE FM_USER_RESET_PWD(
    UN_NAME VARCHAR(30),
    UN_PASSWORD_RESET VARCHAR(255))
RETURNS (
    PHONE_NO VARCHAR(100),
    ERR_TEXT VARCHAR(255))
AS
declare variable c_id decimal(18,0);
declare variable db_id integer;
declare variable un_password_change_bio d_smallint;
declare variable un_id d_id;
declare variable br_result d_smallint;
begin
  -- Процедура заполняет поле un_password_reset, а непосредственно сам сброс
  -- пароля, с запросом его смены и очисткой android_id производится в
  -- процедуре логина hp_check_user_password, в случае их совпадения
  -- Если операция прошла успешно, то err_text пустой

  :db_id = f_get_db_id();
  -- Сначала проверяем, что смена пароля разрешена
  c_id = null;
  err_text = '';
  phone_no = '';
  select un_id, un_from_c_id, un_password_change_bio
  from username
  where un_name = :un_name rows 1
  into un_id, c_id, un_password_change_bio;

  if (c_id is null) then
    err_text = case db_id
                 when 94401 then 'Istifadechi tapilmadi.'
                 else 'Пользователь не найден.'
               end;
  else
  begin
    phone_no = '';

    select cont_value
    from hp_contdata_get_cell_main(0, :c_id)
    into phone_no;

    if (phone_no = '') then
      err_text = case db_id
                   when 94401 then 'Tesdiqlenen mobil nomre yoxdur, texniki destek shobesine muraciyet edin'
                   else 'Отсутствует подтвержденный № телефона, обратитесь в тех. поддержку.'
                 end;
  end

  if (un_password_change_bio = 1) then
  begin
    select br.br_result
    from bio_results br
    where br.br_from_un_id = :un_id
    order by br.br_datetime desc
    rows 1
    into br_result;

    if (br_result = 0) then
      err_text = 'Вы не прошли биометрическую проверку, попробуйте ещё раз.';
  end

  if (err_text <> '') then
  begin
    suspend;
    exit;
  end

  update username
  set un_password_reset = :un_password_reset
  where un_name = :un_name;
  suspend;
end
