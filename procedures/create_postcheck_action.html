CREATE PROCEDURE CREATE_POSTCHECK_ACTION(
    K_ID D_ID NOT NULL,
    REG_ID D_INTEGER NOT NULL)
RETURNS (
    ACL_ID D_ID)
AS
declare variable rd_id d_integer;
declare variable rd_id_check d_integer;
declare variable un_id id_type;
declare variable object_kind d_integer;
declare variable c_id d_id;
declare variable acl_id1 d_id;
declare variable db_id d_integer;
declare variable action_code d_str_100;
begin
  db_id = f_get_db_id();
  if (db_id not in (39801, 41701)) then
    exit;

  if (exists(select *
             from expressapp exa
             where exa.exa_from_k_id = :k_id)) then
    exit;

  if (exists(select *
             from kontract kt
             inner join product pr on pr.pr_id = kt.k_from_pr_id
             where kt.k_id = :k_id and
                   (pr.pr_code containing '.express_similar.' or
                    pr.pr_code containing '.fitness.'))) then
    exit;

  --rd_id = f_get_rd_id('ActionGroup.Check_Loan_Data');
  select rd_id
  from hp_get_check_loan_data(:k_id)
  into:rd_id;
  object_kind = f_objkind('Kontract', 'id');
  un_id = 1000000000000;

  -- Задача на пост-проверку(при выдаче займа) формируется если:
  -- по займу не было сертификации
  if (not exists(select *
                 from actionlist
                 where acl_object_kind = :object_kind and
                       acl_object_id = :k_id and
                       acl_from_rd_id = :rd_id) or
      -- была авто-сертификация роботом
      exists(select *
             from actionlist
             where acl_object_kind = :object_kind and
                   acl_object_id = :k_id and
                   acl_from_rd_id = :rd_id and
                   acl_respons_c_id = :un_id and
                   acl_status = 2)) then
  begin
    action_code = iif(db_id = 41701 and
      exists(select * from kontract where k_id = :k_id and k_by_rating = 1),
        'ActionGroup.Check_Loan_Data_PostCheckR',
        'ActionGroup.Check_Loan_Data_PostCheck');
    acl_id = (select res_acl_id
              from hp_actionlist_insert(null, :reg_id, 'Kontract', :k_id, :action_code, 0, :un_id));

    for select kc.kc_from_c_id
        from knclient kc
        where kc.kc_from_k_id = :k_id and
              kc.kc_amount_approved > 0
        into :c_id
    do
    begin
      rd_id_check = f_get_rd_id('ActionGroup.Check_Client_Data');

      if (exists(select *
                 from actionlist acl
                 where acl.acl_object_kind = f_objkind('Clients', 'id')
                       and acl.acl_object_id = :c_id
                       and acl.acl_from_rd_id = :rd_id_check
                       and acl.acl_parent_id = :acl_id)) then
          continue;

      acl_id1 = (select res_acl_id
                 from hp_actionlist_insert(:acl_id, :reg_id, 'Clients', :c_id, 'ActionGroup.Check_Client_Data', 0, :un_id));
    end
    suspend;
  end
end
