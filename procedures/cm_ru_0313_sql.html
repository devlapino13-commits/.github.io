CREATE PROCEDURE CM_RU_0313_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable disb_date d_date;
declare variable cli_balance curr_type;
declare variable amount curr_type;
declare variable delinq_balance d_currency;
declare variable office_id d_integer;
begin
  res_error_is_critical = 2;

  for select mac_from_c_id, ma_date_oper, ma_office_id, mac_amount
      from miniapp
      inner join miniappclient on mac_from_ma_id = ma_id
      inner join product on pr_id = ma_from_pr_id
      where ma_id = :ma_id and
            ma_is_active = 1 and
            ma_date_end = '31.12.9999' and
            pr_code containing '.b_refinance.'
      into c_id, disb_date, office_id, amount
  do
  begin
    if (office_id in (1490600, 1050600)) then
    begin
      res_error_text = 'В текущем офисе выдача запрещена';
      suspend;
      exit;
    end

    if (disb_date > '31.07.2026') then
    begin 
      res_error_text = 'Заем не может быть выдан позднее 31 июля';
      suspend;
      exit;
    end

    --заем может быть выдан только 1 раз за акцию
    if (exists
       (select 1
        from kontstate, knclient, kontract, product
        where ks_ol_is_delete = 0 and
              ks_state >= 1 and
              ks_from_k_id = k_id and
              k_from_pr_id = pr_id and
              ks_from_k_id = kc_from_k_id and
              :disb_date between ks_date_begin and ks_date_end and
              kc_amount_approved > 0 and
              kc_from_c_id = :c_id and
              pr_code containing '.b_refinance.')) then
    begin
      res_error_text = f_format('Заем может быть выдан не более 1 раза');
      suspend;
    end

    --Заем может быть выдан при условии, что после его выдачи общая задолженность клиента по основному долгу не превысит 500 000 руб.
    cli_balance = 0;
    delinq_balance = 0;
    select sum(oks_balance), sum(iif(:disb_date > oks_delinq_date, oks_delinq_os + oks_delinq_perc, 0))
    from knclient, oksummary
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = oks_from_k_id and
          oks_balance > 0 and
          oks_close_state = 0 and
          :disb_date between oks_date_begin and oks_date_end
    into cli_balance, delinq_balance;
    cli_balance = coalesce(cli_balance, 0);
    delinq_balance = coalesce(delinq_balance, 0);

    if (cli_balance + amount > 500000) then
    begin
      res_error_text = 'Общий баланс всех займов превысит 500 000 р.';
      suspend;
    end

    if (delinq_balance > 0) then
    begin
      res_error_text = 'У клиента имеется активная просрочка';
      suspend;
    end
  end
end
