CREATE PROCEDURE MKR_FILL_POKET_AKB(
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_CATEG SMALLINT)
RETURNS (
    POCKET_ID D_ID)
AS
begin
  -- создаю кредитный пакет ------------------->
  if (report_categ is null) then
    report_categ = 0;

  pocket_id = gen_id(gen_mkrpoket_id, 1);

  insert into mkr_poket (mp_id, mp_create_time, mp_date_begin, mp_date_end, mp_type, mp_categ)
    values (:pocket_id, current_timestamp, :report_date_begin, :report_date_end, 1, :report_categ);
  ---------------------------------------------<

  if (report_categ in (0, 1, 2)) then -- если тип пакета "Общий", "Выданные", "Закрытые"
  begin
    insert into mkr_credit (mc_type, mc_from_mp_id, mc_id, mc_from_c_id, mc_name, mc_placeofbirth, mc_countrycode,
                            mc_pincode, mc_bankruptcystatus, mc_dateofbirth, mc_from_k_id, mc_accountno, mc_credittype,
                            mc_creditlineamount, mc_purposeofcredit, mc_currencyofcredit, mc_initialamountofcredit,
                            mc_disoutamountofcredit, mc_annualinterestrate, mc_dateofgrant, mc_duetimefirstcontract,
                            mc_duetimelastcontract, mc_lastpaymentdate, mc_monthlypaymentamount,
                            mc_daysmainsumisoverdue, mc_daysinterestisoverdue, mc_creditperiodinmonths,
                            mc_oiaforrepperiod, mc_oiaforlastthreemonths, mc_numberofprolongs, mc_creditclasscode,
                            mc_creditstatuscode, mc_provision)
    select cr_type, :pocket_id, id, knc_id, name, placeofbirth, countrycode, pincode, bankruptcystatus, dateofbirth,
           k_id, accountno, credittype, creditlineamount, purposeofcredit, currencyofcredit, initialamountofcredit,
           disoutamountofcredit, annualinterestrate, dateofgrant, duetimefirstcontract, duetimelastcontract,
           lastpaymentdate, monthlypaymentamount, daysmainsumisoverdue, daysinterestisoverdue, creditperiodinmonths,
           oiaforrepperiod, oiaforlastthreemonths, numberofprolongs, creditclasscode, creditstatuscode, provision
    from mkr_prepare_credlist(:report_date_begin, :report_date_end, :report_categ);
  end
  else
  if (report_categ = -1) then -- если тип пакета "Внесистмные кредиты"
  begin
    insert into mkr_credit (mc_type, mc_from_mp_id, mc_id, mc_from_c_id, mc_name, mc_placeofbirth, mc_countrycode,
                            mc_pincode, mc_bankruptcystatus, mc_dateofbirth, mc_from_k_id, mc_accountno, mc_credittype,
                            mc_creditlineamount, mc_purposeofcredit, mc_currencyofcredit, mc_initialamountofcredit,
                            mc_disoutamountofcredit, mc_annualinterestrate, mc_dateofgrant, mc_duetimefirstcontract,
                            mc_duetimelastcontract, mc_lastpaymentdate, mc_monthlypaymentamount,
                            mc_daysmainsumisoverdue, mc_daysinterestisoverdue, mc_creditperiodinmonths,
                            mc_oiaforrepperiod, mc_oiaforlastthreemonths, mc_numberofprolongs, mc_creditclasscode,
                            mc_creditstatuscode, mc_provision)
    select cr_type, :pocket_id, id, knc_id, name, placeofbirth, countrycode, pincode, bankruptcystatus, dateofbirth,
           k_id, accountno, credittype, creditlineamount, purposeofcredit, currencyofcredit, initialamountofcredit,
           disoutamountofcredit, annualinterestrate, dateofgrant, duetimefirstcontract, duetimelastcontract,
           lastpaymentdate, monthlypaymentamount, daysmainsumisoverdue, daysinterestisoverdue, creditperiodinmonths,
           oiaforrepperiod, oiaforlastthreemonths, numberofprolongs, creditclasscode, creditstatuscode, provision
    from mkr_prepare_operkroff_credlist(:report_date_begin, :report_date_end);
  end
end
