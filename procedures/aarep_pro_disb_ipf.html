CREATE PROCEDURE AAREP_PRO_DISB_IPF(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    BT_NAME VARCHAR(100),
    KC_AMOUNT CURR_TYPE,
    KC_AMOUNT_1 CURR_TYPE,
    KC_AMOUNT_1_2 CURR_TYPE,
    KC_AMOUNT_3_6 CURR_TYPE,
    KC_AMOUNT_6_12 CURR_TYPE,
    KC_AMOUNT_12_24 CURR_TYPE,
    KC_AMOUNT_24_36 CURR_TYPE,
    KC_AMOUNT_36_60 CURR_TYPE,
    KC_AMOUNT_60_ CURR_TYPE,
    AVG_PERCENT CURR_TYPE,
    KOL_KRED INTEGER,
    AVG_PERCENT_EFFECT CURR_TYPE)
AS
begin
  for with tabl1
      as (select k_id, round(k_percent_per_day * 365, 2) k_percent, k_percent_effect, k_period
          from kontstate, kontract, product
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_begin between :rep_date_begin and :rep_date_end and
                ks_from_k_id = k_id and
                k_from_pr_id = pr_id and
                pr_code containing '.ipf.'),
      tabl2
      as (select kc_from_c_id, kc_amount_approved kc_amount, k_percent, k_percent_effect,
                 case
                   when bt_name = 'Животноводство' then 'Сельское хозяйство'
                   when bt_name = 'Производство' then 'Промышленность'
                   when bt_name in ('Другое', 'Физические лица Зарплата') then 'Физические лица(потреб. кредиты) в т.ч факторинговые операции'
                   else bt_name
                 end as bt_name,
                 k_period
          from tabl1, knclient, bussubtype, bustype
          where k_id = kc_from_k_id and
                kc_amount_approved > 0 and
                kc_busuness_subtype = bs_id and
                bs_from_bt_id = bt_id)
      select bt_name, sum(kc_amount) kc_amount, sum(iif(k_period <= 1, kc_amount, 0)) kc_amount_1,
             sum(iif(k_period > 1 and k_period <= 3, kc_amount, 0)) kc_amount_1_2,
             sum(iif(k_period > 3 and k_period <= 6, kc_amount, 0)) kc_amount_3_6,
             sum(iif(k_period > 6 and k_period <= 12, kc_amount, 0)) kc_amount_6_12,
             sum(iif(k_period > 12 and k_period <= 24, kc_amount, 0)) kc_amount_12_24,
             sum(iif(k_period > 24 and k_period <= 36, kc_amount, 0)) kc_amount_24_36,
             sum(iif(k_period > 36 and k_period <= 60, kc_amount, 0)) kc_amount_36_60,
             sum(iif(k_period > 60, kc_amount, 0)) kc_amount_60_, avg(k_percent) avg_percent, count(*) kol_kred,
             avg(k_percent_effect) avg_percent_effect
      from tabl2
      group by 1
      into bt_name, kc_amount, kc_amount_1, kc_amount_1_2, kc_amount_3_6, kc_amount_6_12, kc_amount_12_24,
           kc_amount_24_36, kc_amount_36_60, kc_amount_60_, avg_percent, kol_kred, avg_percent_effect
  do
  begin

    suspend;
  end
end
