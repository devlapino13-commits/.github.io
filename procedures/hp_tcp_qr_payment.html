CREATE PROCEDURE HP_TCP_QR_PAYMENT(
    JSON D_STR_8100)
AS
declare variable created_time timestamp;
declare variable created_date date;
declare variable cs_id d_id;
declare variable amount decimal(15,2);
declare variable cs_phone d_str_100;
declare variable mcl_id d_integer;
declare variable check_number d_str_100;
declare variable rd_id_tsp d_integer;
declare variable rd_id_tsp_and_ipf d_integer;
declare variable ag_id d_integer;
begin
  created_time = cast(replace(left(f_json_str_get_value(json, 'created', null), position('.', f_json_str_get_value(json, 'created', null)) - 1), 'T', ' ') as timestamp);
  created_date = cast(created_time as date);
  amount = cast(f_json_str_get_value(:json, 'amount', null) as d_currency) / 100;
  check_number = coalesce(f_json_str_get_value(:json, 'check_number', null), '');

  select cs.cs_id, cs.cs_phone
  from clientshops cs
  where cs.cs_ipc_id = f_json_str_get_value(:json, 'mid', null)
  into :cs_id, :cs_phone;

  rd_id_tsp = f_get_rd_id('AgreementKinds.tsp');
  rd_id_tsp_and_ipf = f_get_rd_id('AgreementKinds.tsp_and_ipf');

  select ag.ag_id
  from clientshops cs
  inner join clients cl on cs.cs_from_c_id = cl.c_id
  inner join agreements ag on ag.ag_object_id = cl.c_id
  where cs.cs_id = :cs_id
  and ag.ag_object_kind = 0
  and cast(:created_time as date) between ag.ag_date_begin and coalesce(ag.ag_date_end, '31.12.9999')
  and ag.ag_status = 1
  and ag.ag_from_rd_id in (:rd_id_tsp, :rd_id_tsp_and_ipf)
  rows 1
  into :ag_id;

  insert into ipc_qr_payments (iqp_transaction, iqp_from_cs_id, iqp_ipc_state, iqp_amount, iqp_fee, iqp_check_number, iqp_ipc_datetime,
                               iqp_state, iqp_date, iqp_datetime, iqp_request_json, iqp_from_ag_id)
  values (
    f_json_str_get_value(:json, 'transaction_id', null),
    :cs_id,
    f_json_str_get_value(:json, 'state', null),
    :amount,
    cast(f_json_str_get_value(:json, 'fee', null) as d_currency) / 100,
    :check_number,
    :created_time,
    0,
    :created_date,
    current_timestamp,
    :json,
    :ag_id);

  for select ml.mcl_id
  from mc_login ml where ml.mcl_login = :cs_phone
  into :mcl_id
  do
  begin
    execute procedure mc_send_push('mc_login', :mcl_id, 'Оплата ч/з QR-код Элькарт', 'Чек №: '||:check_number||'. Оплата на сумму: '||:amount||' сом', 30);
  end
end
