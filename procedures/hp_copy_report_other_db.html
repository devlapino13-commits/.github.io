CREATE PROCEDURE HP_COPY_REPORT_OTHER_DB(
    REPORT_ID INTEGER)
RETURNS (
    REPORT_MEMO VARCHAR(512))
AS
declare variable mss_id integer;
declare variable mss_db_connect varchar(255);
declare variable mss_user_connect varchar(50);
declare variable mss_pass varchar(50);
declare variable mss_user_save decimal(18,0);
declare variable mss_office_save integer;
declare variable sql_text varchar(2048);
declare variable rf_number integer;
declare variable rf_field_name varchar(50);
declare variable rf_name_rus varchar(50);
declare variable rf_hint_rus varchar(256);
declare variable rf_name_eng varchar(50);
declare variable rf_hint_eng varchar(256);
declare variable rf_type smallint;
declare variable rf_width smallint;
declare variable rf_can_group smallint;
declare variable rf_agregate smallint;
declare variable rf_lookup_sql varchar(256);
declare variable rt_number integer;
declare variable rt_name varchar(256);
declare variable rt_name_eng varchar(256);
declare variable rt_user_id decimal(18,0);
declare variable rt_date_created timestamp;
declare variable rt_for_all smallint;
declare variable rt_default smallint;
declare variable rt_header varchar(1024);
declare variable rt_footer varchar(1024);
declare variable rt_autowidth smallint;
declare variable rt_page_orientation smallint;
declare variable rtf_from_rt_number integer;
declare variable rtf_from_rf_number integer;
declare variable rtf_order_no smallint;
declare variable rtf_group_number smallint;
declare variable rtf_agr_function smallint;
declare variable rtf_width smallint;
declare variable rtf_sort_index smallint;
declare variable rtf_sort_order smallint;
declare variable rl_id integer;
declare variable parent integer;
declare variable lb integer;
declare variable rb integer;
declare variable rl_name_rus varchar(255);
declare variable rl_hint_rus varchar(1000);
declare variable rl_name_eng varchar(255);
declare variable rl_hint_eng varchar(1000);
declare variable rl_type smallint;
declare variable rl_hotkey varchar(20);
declare variable rl_data_rus blob sub_type 0 segment size 80;
declare variable rl_data_eng blob sub_type 0 segment size 80;
declare variable rl_is_ready smallint;
declare variable rl_user_id decimal(18,0);
declare variable rl_date_kind smallint;
declare variable rl_report_kind integer;
declare variable rl_sql varchar(1024);
declare variable rl_filter blob sub_type 0 segment size 80;
declare variable rl_order_no integer;
declare variable rl_id_new integer;
begin
  for select mss_id, mss_db_connect, mss_user_connect, mss_pass, mss_user_save, mss_office_save
      from morsynsettings
      into :mss_id, :mss_db_connect, :mss_user_connect, :mss_pass, :mss_user_save, :mss_office_save
  do
  begin
--  Копирование REPORTLIST
    sql_text = 'select RL_ID, PARENT, LB, RB, RL_NAME_RUS, RL_HINT_RUS, RL_NAME_ENG, RL_HINT_ENG, RL_TYPE, RL_HOTKEY, RL_DATA_RUS,
           RL_DATA_ENG, RL_IS_READY, RL_USER_ID, RL_DATE_KIND, RL_REPORT_KIND, RL_SQL, RL_FILTER, RL_ORDER_NO
    from REPORTLIST
    where rl_id = ' || :report_id;
    rl_id = null;
    execute statement :sql_text
        on external :mss_db_connect
        as user :mss_user_connect password :mss_pass
        into :rl_id, :parent, :lb, :rb, :rl_name_rus, :rl_hint_rus, :rl_name_eng, :rl_hint_eng, :rl_type, :rl_hotkey,
        :rl_data_rus, :rl_data_eng, :rl_is_ready, :rl_user_id, :rl_date_kind, :rl_report_kind, :rl_sql, :rl_filter,
        :rl_order_no;
    if (rl_id is not null) then
    begin
      if (not exists(select *
                     from reportlist
                     where rl_id = :parent)) then
      begin
        sql_text = '
        select RL_NAME_RUS from REPORTLIST
        where rl_id =' || :parent;
        execute statement :sql_text
            on external :mss_db_connect
            as user :mss_user_connect password :mss_pass
            into :rl_name_rus;
        report_memo = 'В текущей базе не имеется раздел отчетов "' || :rl_name_rus || '", прошу создать и заново повторить операцию.';
        suspend;
        exit;
      end

      rl_id_new = coalesce((select max(rl_id)
                            from reportlist), 0) + 1;

      insert into reportlist (rl_id, parent, lb, rb, rl_name_rus, rl_hint_rus, rl_name_eng, rl_hint_eng, rl_type,
                              rl_hotkey, rl_data_rus, rl_data_eng, rl_is_ready, rl_user_id, rl_date_kind,
                              rl_report_kind, rl_sql, rl_filter, rl_order_no)
      values (:rl_id_new, :parent, :lb, :rb, :rl_name_rus, :rl_hint_rus, :rl_name_eng, :rl_hint_eng, :rl_type,
              :rl_hotkey, :rl_data_rus, :rl_data_eng, :rl_is_ready, :mss_user_save, :rl_date_kind, :rl_report_kind,
              :rl_sql, :rl_filter, :rl_order_no);
--    Копирование REPFIELDS
      sql_text = '
      select RF_NUMBER, RF_FIELD_NAME, RF_NAME_RUS, RF_HINT_RUS, RF_NAME_ENG, RF_HINT_ENG, RF_TYPE, RF_WIDTH,
             RF_CAN_GROUP, RF_AGREGATE, RF_LOOKUP_SQL
      from REPFIELDS
      where RF_FROM_RL_ID =' || :report_id;

      for execute statement :sql_text
              on external :mss_db_connect
              as user :mss_user_connect password :mss_pass
              into :rf_number, :rf_field_name, :rf_name_rus, :rf_hint_rus, :rf_name_eng, :rf_hint_eng, :rf_type,
              :rf_width, :rf_can_group, :rf_agregate, :rf_lookup_sql
      do
      begin
        insert into repfields (rf_from_rl_id, rf_number, rf_field_name, rf_name_rus, rf_hint_rus, rf_name_eng,
                               rf_hint_eng, rf_type, rf_width, rf_can_group, rf_agregate, rf_lookup_sql)
        values (:rl_id_new, :rf_number, :rf_field_name, :rf_name_rus, :rf_hint_rus, :rf_name_eng, :rf_hint_eng,
                :rf_type, :rf_width, :rf_can_group, :rf_agregate, :rf_lookup_sql);
      end
--    Копирование REPTEMPLATES
      sql_text = '
      select RT_NUMBER, RT_NAME, RT_NAME_ENG, RT_USER_ID, RT_DATE_CREATED, RT_FOR_ALL, RT_DEFAULT, RT_HEADER,
             RT_FOOTER, RT_AUTOWIDTH, RT_PAGE_ORIENTATION
      from REPTEMPLATES
      where RT_FROM_RL_ID =' || :report_id;
      for execute statement :sql_text
              on external :mss_db_connect
              as user :mss_user_connect password :mss_pass
              into :rt_number, :rt_name, :rt_name_eng, :rt_user_id, :rt_date_created, :rt_for_all, :rt_default,
              :rt_header, :rt_footer, :rt_autowidth, :rt_page_orientation
      do
      begin
        insert into reptemplates (rt_from_rl_id, rt_number, rt_name, rt_name_eng, rt_user_id, rt_date_created,
                                  rt_for_all, rt_default, rt_header, rt_footer, rt_autowidth, rt_page_orientation)
        values (:rl_id_new, :rt_number, :rt_name, :rt_name_eng, :mss_user_save, :rt_date_created, :rt_for_all,
                :rt_default, :rt_header, :rt_footer, :rt_autowidth, :rt_page_orientation);
      end

--    Копирование REPTEMPFIELDS
      sql_text = '
      select RTF_FROM_RT_NUMBER, RTF_FROM_RF_NUMBER, RTF_ORDER_NO, RTF_GROUP_NUMBER, RTF_AGR_FUNCTION,
             RTF_WIDTH, RTF_SORT_INDEX, RTF_SORT_ORDER
      from REPTEMPFIELDS
      where RTF_FROM_RL_ID =' || :report_id;
      for execute statement :sql_text
              on external :mss_db_connect
              as user :mss_user_connect password :mss_pass
              into :rtf_from_rt_number, :rtf_from_rf_number, :rtf_order_no, :rtf_group_number, :rtf_agr_function,
              :rtf_width, :rtf_sort_index, :rtf_sort_order
      do
      begin
        insert into reptempfields (rtf_from_rl_id, rtf_from_rt_number, rtf_from_rf_number, rtf_order_no,
                                   rtf_group_number, rtf_agr_function, rtf_width, rtf_sort_index, rtf_sort_order)
        values (:rl_id_new, :rtf_from_rt_number, :rtf_from_rf_number, :rtf_order_no, :rtf_group_number,
                :rtf_agr_function, :rtf_width, :rtf_sort_index, :rtf_sort_order);
      end
      report_memo = 'Отчет "' || :rl_name_rus || '" успешно скопирован!';
    end
    else
    begin
      report_memo = 'Отчет с ID - ' || :report_id || ' не существует, введите правильный ID отчета!';
    end

    suspend;
  end
end
