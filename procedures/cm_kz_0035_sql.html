CREATE PROCEDURE CM_KZ_0035_SQL(
    MA_ID D_ID,
    CHECK_KIND D_STR_255,
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable MA_AMOUNT CURR_TYPE;
declare variable MA_DATE D_DATE;
declare variable PKB_HISTORY_COUNT D_INTEGER;
declare variable CV_VALUE STR_100;
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable EXISTING_CNT D_INTEGER;
declare variable TERMINATED_CNT D_INTEGER;
declare variable DOC_TIME_CHIP timestamp;
declare variable DOC_TIME_FOTO timestamp;
declare variable IS_REQUIRED D_SMALLINT;
begin
  res_info_text = 'Биометрия';
  res_error_text = '';
  res_error_is_critical = 0;
  is_required = 0;

  select mac.mac_amount, mac.mac_from_c_id, ma.ma_date_oper
  from miniapp ma
  inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
  where ma.ma_id = :ma_id and
        mac.mac_amount > 0 and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.9999'
  rows 1
  into :ma_amount, :c_id, :ma_date;

  select cast(cv.cv_value as decimal(18,2))
  from const_var cv
  where cv.cv_code = 'kz_freeze_start_amount'
  into :cv_value;

  if (coalesce(:ma_amount, 0) > coalesce(:cv_value, 0)) then
  begin
    if (exists(select 1 from kontstate ks 
               inner join knclient kc on ks.ks_from_k_id = kc.kc_from_k_id
               where ks.ks_ol_is_delete = 0 and
                     ks.ks_state = 1 and
                     kc.kc_amount_approved > 0 and
                     kc.kc_from_c_id = :c_id and
                     ks.ks_date_begin < :ma_date)) then
      exit;

    pkb_history_count = -1;
    select (prp.prp_active_contract_count + prp.prp_te_cr_history_count)
    from pkb_report_parsed prp
    where prp.prp_from_ma_id = :ma_id
    order by prp.prp_from_eil_id desc
    rows 1
    into :pkb_history_count;

    if (:pkb_history_count = -1) then
    begin
      select eil.eil_from_bls_id
      from extinfolist eil
      where eil.eil_object_kind = 11 and
            eil.eil_object_id = :ma_id and
            eil.eil_status = 2 and
            eil.eil_from_eic_id = 1
      order by eil.eil_status_datetime desc
      rows 1
      into :bls_id;

      if (:bls_id is not null) then
      begin
        select cast(blx_blob as d_blob_text)
        from hp_get_blobex_blob(:bls_id)
        into :json_text;
        if (:json_text is not null) then
        begin
          existing_cnt = f_json_blob_get_count(:json_text, 'report.root.existingcontracts.contract');
          terminated_cnt = f_json_blob_get_count(:json_text, 'report.root.terminatedcontracts.contract');
          pkb_history_count = iif(existing_cnt < 0, 0, existing_cnt) + iif(terminated_cnt < 0, 0, terminated_cnt);
        end
      end
    end
    
    if (:pkb_history_count = 0) then
    is_required = 1;

    if (is_required = 0)
    then exit;

    doc_time_chip = null;
    select first 1 cd.cds_create_datetime
    from clientdocs cd
    inner join passkind pk on pk.pk_id = cd.cds_from_pk_id
    inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
    where cd.cds_from_c_id = :c_id and
          pdt.pdt_code = 'info_with_chip' and
          cd.cds_is_delete = 0
    order by cd.cds_create_datetime desc
    into :doc_time_chip;
    
    if (doc_time_chip is null) then
    begin
      res_error_text = 'Отсутствует документ "Согласие на биометрию". Сформируйте и загрузите его.';
      res_error_is_critical = 2;
      suspend;
    end
    
    doc_time_foto = null;
    select first 1 cd.cds_create_datetime
    from clientdocs cd
    inner join passkind pk on pk.pk_id = cd.cds_from_pk_id
    inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
    where cd.cds_from_c_id = :c_id and
          pdt.pdt_code = 'foto_on_date_disb' and
          cd.cds_is_delete = 0
    order by cd.cds_create_datetime desc
    into :doc_time_foto;
    
    if (doc_time_foto is null) then
    begin
      res_error_text = 'Загрузите фото клиента на дату выдачи.';
      res_error_is_critical = 2;
      suspend; exit;
    end
    
    if (cast(doc_time_chip as date) <> current_date) then
    begin
      res_error_text = 'Документ "Согласие на биометрию" устарел. Загрузите новое фото согласия за сегодня.';
      res_error_is_critical = 2;
      suspend; exit;
    end
    
    if (cast(doc_time_foto as date) <> current_date) then
    begin
      res_error_text = 'Фото клиента на выдачу устарело. Сделайте новое фото за сегодня.';
      res_error_is_critical = 2;
      suspend; exit;
    end
    suspend;
  end
end
