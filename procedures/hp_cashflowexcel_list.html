CREATE PROCEDURE HP_CASHFLOWEXCEL_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    CFX_ID D_ID,
    CFX_DATE D_DATE,
    CFX_DOC_KIND D_STR_20,
    CFX_BANK_BIK D_STR_20,
    CFX_DOC_NO D_STR_20,
    CFX_OPER_KIND D_STR_20,
    CFX_CODE D_STR_20,
    CFX_RESIDENCY D_STR_20,
    CFX_CLIENT_TYPE D_STR_20,
    CFX_CLIENT_NAME D_STR_255,
    CFX_CLIENT_COUNTRY D_STR_20,
    CFX_ACC_DT D_STR_20,
    CFX_ACC_KT D_STR_20,
    CFX_SUMMA D_CURRENCY,
    CFX_CURRENCY D_STR_10,
    CFX_CLIENT_BIRTH_DATE D_STR_20,
    CFX_LOAN_ID D_STR_20,
    CFX_KONTRACT_NO D_STR_20,
    CFX_CLIENT_ACC D_STR_20,
    CFX_OL_ID D_STR_20,
    CFX_OL_MEMO D_STR_LONG,
    CFX_NEED_RECALC D_BOOL,
    CFX_RES_SIGN D_SMALLINT,
    CFX_RES_NON_RESIDENT D_BOOL,
    CFX_RES_OL_ID D_ID,
    CFX_RES_OL_OB_NUMBER D_INTEGER,
    CFX_RES_KOB_ID D_ID,
    CFX_RES_CLIENT_ID D_ID,
    CFX_RES_ACCBANK_ID D_ID,
    CFX_RES_STATUS D_SMALLINT,
    CFX_RES_CHECK D_SMALLINT,
    KOB_ACC_DT_CODE D_STR_CODE,
    KOB_ACC_KT_CODE D_STR_CODE)
AS
begin
  for select cfx_id,
             cfx_date,
             cfx_doc_kind,
             cfx_bank_bik,
             cfx_doc_no,
             cfx_oper_kind,
             cfx_code,
             cfx_residency,
             cfx_client_type,
             cfx_client_name,
             cfx_client_country,
             cfx_acc_dt,
             cfx_acc_kt,
             cfx_summa,
             cfx_currency,
             cfx_client_birth_date,
             cfx_loan_id,
             cfx_kontract_no,
             cfx_client_acc,
             cfx_ol_id,
             cfx_ol_memo,
             cfx_need_recalc,
             cfx_res_sign,
             cfx_res_non_resident,
             cfx_res_ol_id,
             cfx_res_ol_ob_number,
             cfx_res_kob_id,
             cfx_res_client_id,
             cfx_res_accbank_id,
             cfx_res_status,
             cfx_res_check,

             a_dt.acc_add_code,
             a_kt.acc_add_code
      from cashflowexcel
           left join kn_operbook on (cfx_res_kob_id = kob_id)
                left join ref_account a_dt on (a_dt.acc_id = kob_acc_dt)
                left join ref_account a_kt on (a_kt.acc_id = kob_acc_kt)
      where cfx_date between :rep_date_begin and :rep_date_end
      into :cfx_id,
           :cfx_date,
           :cfx_doc_kind,
           :cfx_bank_bik,
           :cfx_doc_no,
           :cfx_oper_kind,
           :cfx_code,
           :cfx_residency,
           :cfx_client_type,
           :cfx_client_name,
           :cfx_client_country,
           :cfx_acc_dt,
           :cfx_acc_kt,
           :cfx_summa,
           :cfx_currency,
           :cfx_client_birth_date,
           :cfx_loan_id,
           :cfx_kontract_no,
           :cfx_client_acc,
           :cfx_ol_id,
           :cfx_ol_memo,
           :cfx_need_recalc,
           :cfx_res_sign,
           :cfx_res_non_resident,
           :cfx_res_ol_id,
           :cfx_res_ol_ob_number,
           :cfx_res_kob_id,
           :cfx_res_client_id,
           :cfx_res_accbank_id,
           :cfx_res_status,
           :cfx_res_check,

           :kob_acc_dt_code,
           :kob_acc_kt_code
  do
  begin
    suspend;
  end
end
