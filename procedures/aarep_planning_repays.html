CREATE PROCEDURE AAREP_PLANNING_REPAYS(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_INCLUDE_FIRST_DAY INTEGER)
RETURNS (
    AAREP_REGION_ID INTEGER,
    AAREP_DATE_ACTION DATE,
    AAREP_PLAN_OS DECIMAL(18,4),
    AAREP_PLAN_PERC DECIMAL(18,4))
AS
declare variable curr_code varchar(3);
declare variable cr_rate decimal(18,4);
declare variable dk_id decimal(18,0);
declare variable dd_tmp date;
begin
  report_include_first_day = 1 - report_include_first_day;
  dd_tmp = :report_date_begin - :report_include_first_day;

  for select kk_id, kk_region_id, kk_curr_rate
      from aarep_active_kont_id_in_period(:report_region_id, null, :dd_tmp, 0, 0, 1, 0, 0)
      into :dk_id, :aarep_region_id, :cr_rate
  do
  begin
    for select ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_ol_date <= :dd_tmp and
              ok_sh_date between :dd_tmp and :report_date_end and
              ok_type < 7 and
              ok_ol_is_active = 1 and
              ok_from_k_id = :dk_id and
              ok_from_c_id = 0
        group by ok_sh_date
        having(sum(ok_os + ok_sh_os) <> 0) or (sum(ok_perc + ok_sh_perc) <> 0)
        into :aarep_date_action, :aarep_plan_os, :aarep_plan_perc
    do
    begin
      aarep_plan_os = aarep_plan_os * cr_rate;
      aarep_plan_perc = aarep_plan_perc * cr_rate;
      suspend;
    end
  end

end
