CREATE PROCEDURE HP_CALC_EPS_FLOAT_SCHEDULE(
    K_ID D_ID,
    LEVEL_NO D_INTEGER)
RETURNS (
    CALC_EFF PERC_TYPE)
AS
declare variable sh_id d_integer;
declare variable sh_date d_date;
declare variable sh_all d_currency;
declare variable sh_date_prev d_date;
declare variable sh_days d_integer;
declare variable is_mbf_card d_smallint;
begin
    delete from temp_schedule;

    select iif(pr.pr_mbulak_cards_kind = 0, 0, 1)
    from product pr
    inner join kontract kont on kont.k_from_pr_id = pr.pr_id
    where kont.k_id = :k_id
    into :is_mbf_card;

    for with sh as
    (
       select sh_id, sh_date, sh_os, sh_perc, sh_kommis
       from print_schedule_float_kg_tmp(:k_id, :level_no)
       union
       select sh_id, sh_date, sh_os, sh_perc, iif(:is_mbf_card = 0, sh_kommis, 0)
       from schedule sh
       where sh.sh_from_k_id = :k_id
       and sh_id = 0
    )
    select 0, min(sh_date), -sum(sh_os - sh_kommis)
    from sh
    union
    select sh_id, sh_date,
           (sh_os + sh_perc)
    from sh
    where sh_id > 0
    into :sh_id, :sh_date, :sh_all
    do
    begin
      if (sh_date_prev is null) then
        sh_days = 0;
      else
        sh_days = sh_date - sh_date_prev;
  
      insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
      values (:sh_id, :sh_date, :sh_days, :sh_all);
      sh_date_prev = sh_date;
    end

    calc_eff =  (select calc_eff from hp_effect_rate_kg(:k_id));

    suspend;
end
