CREATE PROCEDURE HP_INFO_FIXBEM(
    KONT_NO D_STR_20)
RETURNS (
    K_ID D_ID,
    K_DISB_DATE D_DATE,
    K_KONTRACT_NO D_STR_20,
    IS_IPF D_INTEGER,
    KC_ID D_ID,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    MA_ID D_INTEGER,
    MAC_ID D_INTEGER,
    KT_RD_ID D_INTEGER,
    KT_RD_NAME D_STR_255,
    KSB_BS_ID D_INTEGER,
    KSB_BS_NAME D_STR_CODE,
    KB_BT_ID D_INTEGER,
    KB_BT_NAME D_STR_CODE,
    MT_RD_ID D_ID,
    MT_RD_NAME D_STR_255,
    MSB_BS_ID D_INTEGER,
    MSB_BS_NAME D_STR_CODE,
    MB_BT_ID D_INTEGER,
    MB_BT_NAME D_STR_CODE,
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID,
    BLN_ID D_INTEGER,
    DESCR_DET D_STR_255,
    ACL_ID D_ID,
    ACL_ID1 D_ID,
    ON_DATE D_DATE,
    ACTUAL_USE D_STR_CODE,
    SPF_COMMENT D_STR_255)
AS
declare variable rd_id d_integer;
declare variable rd_id1 d_integer;
begin
  -- Инфо для корректировки перед закрытием месяца
  if (kont_no similar to '[-]?[[:DIGIT:]]+') then
    k_id = (select k_id from kontract where k_id = :kont_no);
  else
    k_id = (select k_id from kontract where k_kontract_no = :kont_no);

  rd_id = f_get_rd_id('Anketa.TargetUseLoan');
  rd_id1 = f_get_rd_id('Anketa.Main');
  bln_id = 1638;

  for
    select
      k_id,
      k_disb_date,
      k_kontract_no,
      iif(pr_code containing '.ipf.', 1 ,0),
      kc_id,
      c_id,
      c_name_fio,
      ma_id,
      mac_id,
      kt.rd_id,
      kt.rd_name,
      ksb.bs_id,
      ksb.bs_name,
      kb.bt_id,
      kb.bt_name,
      mt.rd_id,
      mt.rd_name,
      msb.bs_id,
      msb.bs_name,
      mb.bt_id,
      mb.bt_name
    from kontract
      inner join product
        on pr_id = k_from_pr_id
      inner join knclient
        on kc_from_k_id = k_id
          and kc_amount_approved > 0
      inner join clients
        on c_id = kc_from_c_id
    inner join miniapp
      on ma_id = k_from_ma_app_id
        and ma_date_end = '31.12.9999'
    inner join miniappclient
      on mac_from_ma_id = ma_id
        and mac_from_c_id = c_id
    left join refdata as kt
      on kt.rd_id = kc_loan_target
    left join bussubtype as ksb
      on ksb.bs_id = kc_busuness_subtype
    left join bustype as kb
      on kb.bt_id = ksb.bs_from_bt_id
    left join refdata as mt
      on mt.rd_id = mac_loan_target
    left join bussubtype as msb
      on msb.bs_id = mac_business_subtype
    left join bustype as mb
      on mb.bt_id = msb.bs_from_bt_id
    where k_id = :k_id
  into
    :k_id,
    :k_disb_date,
    :k_kontract_no,
    :is_ipf,
    :kc_id,
    :c_id,
    :c_name_fio,
    :ma_id,
    :mac_id,
    :kt_rd_id,
    :kt_rd_name,
    :ksb_bs_id,
    :ksb_bs_name,
    :kb_bt_id,
    :kb_bt_name,
    :mt_rd_id,
    :mt_rd_name,
    :msb_bs_id,
    :msb_bs_name,
    :mb_bt_id,
    :mb_bt_name
  do
  begin
    acl_id = null;
    on_date = null;

    if (k_disb_date >= '03.03.2022') then
    begin
      obj_kind = 'KnClient';
      obj_id = kc_id;

      acl_id = (
        select acl_id
        from actionlist
        where acl_object_kind = f_objkind(:obj_kind, 'id')
         and acl_object_id = :obj_id
         and acl_from_rd_id = :rd_id
        order by acl_id desc
        rows 1);
      on_date = current_date;
    end else begin
      obj_kind = 'Clients';
      obj_id = c_id;

      if (k_disb_date < '14.02.2022')
      then
        on_date = k_disb_date +
          iif(k_disb_date >= '22.12.2021' and is_ipf = 1, 1, 8);
      else
        on_date = k_disb_date + 1;

      select
        acl_id,
        acl_last_datetime
      from actionlist
      where acl_object_kind = f_objkind(:obj_kind, 'id')
        and acl_object_id = :obj_id
        and acl_from_rd_id = :rd_id
        and acl_date = :on_date
        --and acl_status = 2
      order by acl_id desc
      rows 1
      into
        :acl_id,
        :on_date;
    end


    acl_id1 = (
      select
        acl_id
      from actionlist
    where acl_object_kind = 0
      and acl_object_id = :c_id
      and acl_from_rd_id = :rd_id1
      and acl_date <= :k_disb_date
    order by acl_id desc
    rows 1);
    descr_det = (
      select
        coalesce(res_str_values, '')
      from hp_objectprops_anketa_get('Clients', :c_id, -1, 'AnketaQuestions.descr_of_loan', :k_disb_date, null));

    if (acl_id > 0) then
    begin
      actual_use = (
        select
          coalesce(res_str_values, '')
      from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'AnketaQuestions.LoanUseFact', :on_date, null));

      spf_comment = (
        select
          coalesce(res_str_values, '')
        from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'Monitoring.Memo', :on_date, null));
    end

    suspend;
  end
end
