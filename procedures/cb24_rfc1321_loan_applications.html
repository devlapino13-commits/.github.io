CREATE PROCEDURE CB24_RFC1321_LOAN_APPLICATIONS(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    APPLICATION_ID D_INTEGER,
    APPLICATION_UID D_STR_CODE,
    APPLICATION_DATE D_DATE,
    APPLICATION_SUM D_CURRENCY,
    APPLICATION_TYPE D_STR_20,
    APPLICATION_STATUS D_STR_20,
    APPLICATION_DECISION D_DATE,
    APPLICATION_REJECT_REASON D_STR_255,
    SUBJECT_TYPE D_STR_20,
    SUBJECT_PRIVATE_FIO D_STR_100,
    SUBJECT_PRIVATE_BIRTH_DATE D_DATE,
    SUBJECT_PRIVATE_COUNTRY_CODE D_STR_20,
    SUBJECT_PRIVATE_DOC_SERIAL D_STR_20,
    SUBJECT_PRIVATE_DOC_NUMBER D_STR_20,
    RES_KONTRACT_NO D_STR_20,
    RES_C_ID D_ID,
    RES_KC_ID D_ID,
    RES_DISB_DATE D_DATE,
    RES_DOC_TYPE D_STR_100,
    RES_DOC_NO D_STR_CODE,
    RES_DOC_ID D_ID,
    RES_SIGN_DATE D_DATE)
AS
declare variable ma_id d_integer;
declare variable ma_number d_integer;
declare variable ma_is_online d_smallint;
declare variable k_id d_id;
declare variable k_status d_smallint;
declare variable mac_id d_integer;
declare variable ma_date_oper d_date;
declare variable ma_id_prv d_integer;
declare variable ma_is_active d_smallint;
declare variable ma_date_save d_datetime;
declare variable edc_id d_integer;
declare variable obj_kind d_integer;
declare variable app_signed d_bool;
declare variable kc_disb_passport_id d_integer;
declare variable pk_id d_integer;
declare variable edl_id d_id;
begin
  subject_type = 'ФЛ';

  select edc_id, edc_object_kind
  from edoc_cfg
  where edc_code = 'loan.application'
  into edc_id, obj_kind;

  ma_id_prv = 0;
  for select ma_id, ma_number, ma_date_app, ma_is_active, ma_is_online, ma_date_oper, ma_date_save
  from miniapp
       inner join region_list(:rep_region_id, 1, 3, 1) on (ma_office_id + 0 = id_list)
  where (ma_date_app between :rep_date_begin and :rep_date_end) or
        (ma_date_oper between :rep_date_begin and :rep_date_end)
  order by ma_id desc, ma_number desc
  into ma_id, ma_number, application_date, ma_is_active, ma_is_online, ma_date_oper, ma_date_save
  do
  begin
    if (ma_id = ma_id_prv) then
      Continue;

    app_signed = 1;
    edl_id = null;
    select edl_id
    from edoc_list
    where edl_from_edc_id = :edc_id and
          edl_object_kind = :obj_kind and  -- miniapp
          edl_object_id = :ma_id and
          edl_status = 1
    order by edl_id desc
    rows 1
    into edl_id;

    if (edl_id is null) then
    begin
      app_signed = 0;
      res_sign_date = null;
    end else
    begin
      select max(cast(eds_status_datetime as date))
      from edoc_sign
      where eds_from_edl_id = :edl_id and
            eds_status = 1
      into res_sign_date;
    end

    ma_id_prv = ma_id;

    application_id = ma_id;

    if (ma_number > 0) then
    begin
      select ma_date_app
      from miniapp
      where ma_id = :ma_id and
            ma_number = 0
      into application_date;
    end

    k_id = null;
    k_status = 0;
    application_decision = null;
    application_reject_reason = '';
    application_status = '';
    res_disb_date = null;
    res_kontract_no = '';
    res_kc_id = null;
    res_doc_type = '';
    res_doc_no = '';
    subject_private_country_code = '';
    subject_private_doc_serial = '';
    subject_private_doc_number = '';

    select distinct k_id, k_status, k_kontract_no, k_is_online, k_disb_date
    from kontract, knclient
    where k_from_ma_app_id = :ma_id and
          k_id = kc_from_k_id and
          kc_amount_approved > 0
    order by k_status desc, k_id desc
    rows 1
    into k_id, k_status, res_kontract_no, ma_is_online, ma_date_oper;

    if (not((coalesce(res_sign_date, application_date) between :rep_date_begin and :rep_date_end) or
            (ma_date_oper between :rep_date_begin and :rep_date_end))) then
      Continue;

    if (row_count = 0) then
    begin
      select krl_datetime, kr_text
      from kontreslist, kontres
      where krl_from_kr_id = kr_id and
            krl_ma_id = :ma_id
      order by krl_datetime desc
      rows 1
      into :application_decision, :application_reject_reason;

      if (row_count = 1) then
        application_status = 'Отклонено';
      else
      begin
        if (ma_is_active = 1) then
          application_status = 'РешениеНеПринято';
        else
        begin
          application_status = 'Отклонено';
          application_decision = ma_date_save;
          application_reject_reason = 'Отказ по инициативе клиента';
        end
      end
    end else
    begin
      select ks_date_begin
      from kontstate
      where ks_from_k_id = :k_id and
            ks_state = 1 and
            ks_ol_is_delete = 0
      into res_disb_date;

      if (res_disb_date is not null) then
      begin
        application_reject_reason = '';
        application_status = 'Одобрено';
        ma_date_oper = res_disb_date;
      end else
      begin
        if (k_status < 0) then
        begin
          select krl_datetime, kr_text
          from kontreslist, kontres
          where krl_from_kr_id = kr_id and
                krl_from_k_id = :k_id
          order by krl_datetime desc
          rows 1
          into :application_decision, :application_reject_reason;

          if (row_count = 0) then
          begin
            select krl_datetime, kr_text
            from kontreslist, kontres
            where krl_from_kr_id = kr_id and
                  krl_ma_id = :ma_id
            order by krl_datetime desc
            rows 1
            into :application_decision, :application_reject_reason;
    
            if (row_count = 0 and ma_is_active = -1) then
            begin
              application_decision = ma_date_save;
              application_reject_reason = 'Отказ по инициативе клиента';
            end
          end
    
          application_status = 'Отклонено';
        end
        else
        begin
          application_reject_reason = '';
  
          if (k_status = 0) then
            application_status = 'РешениеНеПринято';
          if (k_status = 1) then
            application_status = 'Одобрено';
        end
      end

      if (application_decision is null) then
      begin
        select kl_datetime
        from kontlog
        where kl_from_k_id = :k_id and
              kl_state = :k_status
        order by kl_datetime desc
        rows 1
        into application_decision;
      end
    end

    if (application_reject_reason <> '' and
        (not((application_date between :rep_date_begin and :rep_date_end) or
             (application_decision between :rep_date_begin and :rep_date_end)))) then
      Continue;

    for select mac_id, mac_from_c_id, mac_amount, c_name_fio, c_birth_date
    from miniappclient, clients
    where mac_from_ma_id = :ma_id and
          mac_amount > 0 and
          mac_from_c_id = c_id
    into :mac_id, res_c_id, application_sum, subject_private_fio, subject_private_birth_date
    do
    begin
      if (ma_is_online = 1) then
        application_type = 'онлайн';
      else
        application_type = 'очно';

      res_kc_id = null;
      subject_private_country_code = '';
      subject_private_doc_serial = '';
      subject_private_doc_number = '';
      subject_private_country_code = '';
      res_doc_id = null;
      res_doc_type = '';
      res_doc_no = '';
      pk_id = null;

      kc_disb_passport_id = null;

      if (k_id is not null) then
      begin
        select kc_id, kc_disb_passport_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_from_c_id= :res_c_id
        into res_kc_id, kc_disb_passport_id;

        if (res_kc_id is not null) then
          application_uid = f_uuid_cbr(f_uuid('knclient', res_kc_id));

        if (kc_disb_passport_id is not null) then
        begin
          res_doc_id = kc_disb_passport_id;

          select cds_doc_no, pk_id, pk_name, left(pk_country, 2)
          from clientdocs, passkind
          where cds_id = :res_doc_id and
                cds_from_pk_id = pk_id
          into res_doc_no, pk_id, res_doc_type, subject_private_country_code;

          select res_doc_seria, res_doc_number
          from hp_passport_serial_num(:res_doc_no, :pk_id)
          into subject_private_doc_serial, subject_private_doc_number;
        end
      end else
      begin
        application_uid = f_uuid_cbr(f_uuid('miniappclient', mac_id));

        ma_date_oper = cast(ma_date_oper as date);
  
        select res_doc_id, res_doc_full_no, res_doc_seria, res_doc_number, res_country, res_pk_name
        from hp_passport_to_date(:res_c_id, :ma_date_oper)
        into :res_doc_id, :res_doc_no, :subject_private_doc_serial, :subject_private_doc_number, :subject_private_country_code, :res_doc_type;
      end

      if (app_signed = 0) then
        application_uid = '';

      suspend;
    end
  end
end
