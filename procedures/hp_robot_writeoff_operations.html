CREATE PROCEDURE HP_ROBOT_WRITEOFF_OPERATIONS(
    REP_DATE D_DATE = current_date)
AS
declare variable k_id d_id;
declare variable kontract_no d_str_100;
declare variable close_date d_date;
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable sh_calc_perc curr_type;
declare variable zad_fine curr_type;
declare variable zad_other curr_type;
declare variable writeoff_reason d_integer;
declare variable ol_id id_type;
declare variable k_last_reg_id d_integer;
declare variable ol_repay_amount curr_type;
declare variable error_text str_255;
declare variable object_kind d_integer;
begin
  writeoff_reason = f_get_rd_id('Reason_WriteOff.ignore_balance'); --новая причина списания
  object_kind = f_objkind('kontract', 'id');

  for select k_id, k_kontract_no, oks_date_begin, k_last_reg_id
      from oksummary oks
      inner join kontract k on oks.oks_from_k_id = k.k_id
      where :rep_date between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_close_state = 0 and
            oks.oks_balance = 0
      into :k_id, :kontract_no, :close_date, k_last_reg_id
  do
  begin
    zad_os = null;
    sh_calc_perc = null;
    zad_fine = null;
    zad_perc = null;

    select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine), sum(zad_perc)
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    into zad_os, sh_calc_perc, zad_fine, zad_perc;  

    zad_os = coalesce(zad_os, 0);
    sh_calc_perc = coalesce(sh_calc_perc, 0);
    zad_fine = coalesce(zad_fine, 0);
    zad_perc = coalesce(zad_perc, 0);

    zad_other = null;

    select sum(ox_amount_plus + ox_amount_minus)
    from operextra o
    where o.ox_object_id = :k_id and
          o.ox_object_kind = :object_kind and
          o.ox_ol_is_active = 1
    into :zad_other;

    zad_other = coalesce(zad_other, 0);

    -- если долг только по % в размере не более 10 сом включительно
    if (((sh_calc_perc <= 10) or (zad_perc <= 10)) and
        (zad_other <= 0) and (zad_os <= 0) and (zad_fine <= 0)) then
    begin
      select ol_id, ol_repay_amount, error_text
      from hp_do_repayment(:k_id, 0, :rep_date, -777, 1, 1, 1, 1, 1, 1, 1, :k_last_reg_id, 3, null, 0, 1000000000000, null, 'Протокол ГКК № 209 от 19.11.2021 года', null, :writeoff_reason, -777)
      into :ol_id, :ol_repay_amount, error_text;
    end
  end
end
