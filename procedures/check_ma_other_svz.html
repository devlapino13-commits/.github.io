CREATE PROCEDURE CHECK_MA_OTHER_SVZ(
    LOANMINIAPPID INTEGER,
    MAC_FROM_C_ID DECIMAL(18,0))
RETURNS (
    MA_ID_TMP INTEGER,
    KF_NAME VARCHAR(100),
    REG_NAME VARCHAR(100),
    KF_OFFICER VARCHAR(100),
    KF_OFFICER_ID DECIMAL(18,0),
    REG_ID INTEGER)
AS
begin
-- проверка на наличие заявки у других СВЗ
  for select ma_id, kal_new_kf_name, reg_code || ' ' || reg_name, c_name_fio, kal_new_kf_officer, reg_unicode
      from miniapp, miniappclient, knaktlist, region, clients
      where knaktlist.is_delete = 0 and
            kal_date_end >= '31.12.2999' and
            kal_new_reg_id = reg_unicode and
            kal_new_kf_officer = c_id and
            kal_from_kf_id = ma_from_kf_id and
            ma_id <> :loanminiappid and
            ma_is_active = 1 and
            ma_id = mac_from_ma_id and
            ma_date_app >= (current_date - 30) and
            ma_date_end >= '31.12.2999' and
            mac_amount > 0 and
            mac_from_c_id = :mac_from_c_id
      into :ma_id_tmp, :kf_name, :reg_name, kf_officer, kf_officer_id, reg_id
  do
  begin
    suspend;
  end
end
