CREATE PROCEDURE ADMIN_CREATE
RETURNS (
    K_SUMMA_ALL DECIMAL(18,4))
AS
declare variable ok_from_ol_id decimal(18,0);
declare variable ok_number integer;
declare variable ok_from_k_id decimal(18,0);
declare variable ok_sh_id integer;
declare variable ok_sh_date date;
declare variable ok_type integer;
declare variable ok_os decimal(18,4);
declare variable ok_perc decimal(18,4);
declare variable ok_fine decimal(18,4);
declare variable ok_pay_type integer;
declare variable ok_kommis decimal(18,4);
declare variable ok_sh_os decimal(18,4);
declare variable ok_sh_perc decimal(18,4);
declare variable k_id decimal(18,0);
declare variable k_summa decimal(18,4);
declare variable last_no integer;
begin
  k_summa_all = 0;
  for select distinct ok_from_k_id
      from operkred, kontract, product, knclient
      where ok_type = 8 and
            ok_from_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_loan_type = 0 and
            kc_from_k_id = k_id and
            ok_ol_is_active = 1 and
            ok_from_c_id = 0
      group by ok_from_k_id
      having count(kc_from_c_id) > 2
      into :k_id
  do
  begin
    select sum(ok_os)
    from operkred
    where ok_from_k_id = :k_id and
          ok_from_c_id = 0 and
          ((ok_type = 1) or
          (ok_type = 8))
    into :k_summa;
    k_summa_all = k_summa_all + k_summa;

    for select ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
               ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc
        from operkred
        where ok_type = 8 and
              ok_from_c_id = 0 and
              ok_from_k_id = :k_id
        into :ok_from_ol_id, :ok_number, :ok_from_k_id, :ok_sh_id, :ok_sh_date, :ok_type, :ok_os, :ok_perc, :ok_fine,
             :ok_pay_type, :ok_kommis, :ok_sh_os, :ok_sh_perc
    do
    begin
      if (:last_no is null) then
        last_no = 0;

      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc)
      values (:ok_from_ol_id, -1, :ok_from_k_id, :ok_sh_id, :ok_sh_date, 1, 0, 0, 0, 2, :k_summa * 0.015, 0, 0);

      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc)
      values (:ok_from_ol_id, -1, :ok_from_k_id, :ok_sh_id, :ok_sh_date, 4, 0, 0, 0, 0, :k_summa * (-0.015), 0, 0);
    end
  end
end
