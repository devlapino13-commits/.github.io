CREATE PROCEDURE FM_ADDR_FIND_EDIT(
    PARENT_ID DECIMAL(18,0),
    OBJECT_KIND INTEGER,
    OBJECT_ID DECIMAL(18,0),
    FIND_STR VARCHAR(255),
    FIND_KIND VARCHAR(100),
    UN_ID DECIMAL(18,0))
RETURNS (
    PL_ID DECIMAL(18,0),
    PL_NAME VARCHAR(255),
    PL_REGION_NAME VARCHAR(255),
    ADL_ID DECIMAL(18,0),
    RES_KIND VARCHAR(100),
    PL_POST_INDEX VARCHAR(20),
    PL_FAVORITE INTEGER,
    ERR_TEXT VARCHAR(255))
AS
declare variable PL_NAME_KOMITET varchar(255);
declare variable FIND_PLACE varchar(255);
declare variable KOL integer;
declare variable PL_CAN_SELECT integer;
declare variable COUNTRY_COUNT integer;
declare variable CODE varchar(10);
declare variable KOL_TMP integer;
declare variable DB_ID decimal(18,0);
declare variable ADDRESS_TYPE integer;
declare variable ADL_ID_TMP decimal(18,0);
declare variable IS_ACTIVE_CLIENT smallint;
declare variable HAS_MINIAPP smallint;
declare variable OIVO_ERR_TEXT D_STR_255;
begin
  --  2  Street             завершение работы
  --  1  City               завершение работы
  --  0  Region
  --     Favorite
  --  -1 Country         X
  --  -2 Add_street         завершение работы
  --     No_street          завершение работы
  --  -3 Address         X  завершение работы
  --  -4 Address_type    X
  --  -5 start           X

  -- Процедура возвращает список населенный пунктов и улиц в зависимости от входящего параметра
  :db_id = f_get_db_id();
  pl_favorite = 0;
  err_text = '';
  oivo_err_text = '';

  if (coalesce(find_kind, '') = '') then
    exit;
  if (object_kind is null) then
    exit;
  if (object_id is null) then
    object_id = 0;
  if (find_str is null) then
    find_str = '';
  pl_post_index = '';

  find_place = '';
  pl_name_komitet = '';
  adl_id = 0;

  find_str = trim(find_str);
  find_str = f_replace_all(find_str, '  ', ' ');
  kol = 0;

  -- Если редактируется НЕ адрес клиента, начинаем со страны
  if (object_kind <> 0 and
      find_kind in ('start', 'address_type', 'address')) then
    find_kind = 'country';

  if (find_kind = 'country' and
      parent_id <= 5) then
  begin
    if (parent_id in (1, 2, 3)) then
    begin

      select pl_id
      from places
      where parent is null and
            pl_id starting left(:db_id, 3)
      into parent_id;

      country_count = 1;
    end
    else
    begin
      select count(*)
      from places
      where parent is null
      into :country_count;
      if (country_count is null) then
        country_count = 0;

      if (country_count = 1) then -- если более одного, даем список на выбор
      begin
        select pl_id
        from places
        where parent is null
        into parent_id;
      end
    end
  end
  if (find_kind = 'country' and
      coalesce(parent_id, 0) > 5) then
    find_kind = 'favorite';
  if (find_str <> '' and
      find_kind = 'favorite') then
    find_kind = 'region';

  if (find_str <> '') then
    find_place = :find_str;

  -- если поиск только начинается, выдаем список типов адресов
  -- данный кусок работает только для Object_Kind=0 (физ.лицо)
  if (find_kind = 'start') then
  begin
    if (db_id = 81001 and
        exists(select *
               from hp_check_oivo_assignment(:object_id, :un_id)
               where is_oivo = 1 and
                     res_id <> 1)) then
      oivo_err_text = 'Изменить не удалось, так как клиент закреплен не за Вами';

    pl_favorite = 0;

    res_kind = 'address_type';
    pl_id = 1;
    pl_name = 'Юридический адрес';

    select adc_from_adl_id, adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :object_id and
          adc_address_type = 1 and
          adc_from_adl_id = adl_id
    into :adl_id, :pl_region_name;
    if (right(adl_id, 9) = '000000000' or coalesce(adl_id, 0) = 0) then
    begin
      pl_region_name = 'Нажмите, чтоб заполнить место прописки';
    end
    else
      pl_region_name = pl_region_name || ascii_char(13) || ascii_char(10) || 'Нажмите, чтоб изменить этот адрес';

    adl_id = 0;

    select is_active_client
    from fm_client_status(:un_id, :object_id)
    into is_active_client;

    select has_miniapp
    from fm_client_has_miniapp(:un_id, :object_id)
    into :has_miniapp;

    if (has_miniapp = 0 and
        is_active_client = 1) then
      err_text = 'Редактирование юридического адреса активного клиента запрещено.';

    if (err_text = '') then
      err_text = oivo_err_text;

    suspend;
    err_text = '';

    pl_id = 2;
    pl_name = 'Фактический адрес';

    select adc_from_adl_id, adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :object_id and
          adc_address_type = 2 and
          adc_from_adl_id = adl_id
    into :adl_id, :pl_region_name;
    if (right(adl_id, 9) = '000000000' or coalesce(adl_id, 0) = 0) then
    begin
      pl_region_name = 'Нажмите, чтобы заполнить адрес проживания';
    end
    else
      pl_region_name = pl_region_name || ascii_char(13) || ascii_char(10) || 'Нажмите, чтобы изменить этот адрес';
    adl_id = 0;

    err_text = :oivo_err_text;
    suspend;

    pl_id = 3;
    pl_name = 'Место работы';

    select adc_from_adl_id, adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :object_id and
          adc_address_type = 3 and
          adc_from_adl_id = adl_id
    into :adl_id, :pl_region_name;
    if (right(adl_id, 9) = '000000000' or coalesce(adl_id, 0) = 0) then
    begin
      pl_region_name = 'Нажмите, чтобы заполнить место работы/бизнеса';
    end
    else
      pl_region_name = pl_region_name || ascii_char(13) || ascii_char(10) || 'Нажмите, чтобы изменить этот адрес';
    adl_id = 0;

    err_text = :oivo_err_text;
    suspend;
    if (db_id not in (39801, 41701)) then
    begin
      pl_id = 4;
      pl_name = 'Юридический адрес на родине';

      select adc_from_adl_id, adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :object_id and
            adc_address_type = 4 and
            adc_from_adl_id = adl_id
      into :adl_id, :pl_region_name;
      if (right(adl_id, 9) = '000000000' or coalesce(adl_id, 0) = 0) then
      begin
        pl_region_name = 'Нажмите, чтобы заполнить место прописки на родине';
      end
      else
        pl_region_name = pl_region_name || ascii_char(13) || ascii_char(10) || 'Нажмите, чтобы изменить этот адрес';
      adl_id = 0;

      err_text = :oivo_err_text;
      suspend;

      pl_id = 5;
      pl_name = 'Фактический адрес на родине';

      select adc_from_adl_id, adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :object_id and
            adc_address_type = 5 and
            adc_from_adl_id = adl_id
      into :adl_id, :pl_region_name;
      if (right(adl_id, 9) = '000000000' or coalesce(adl_id, 0) = 0) then
      begin
        pl_region_name = 'Нажмите, чтобы заполнить место жительства на родине';
      end
      else
        pl_region_name = pl_region_name || ascii_char(13) || ascii_char(10) || 'Нажмите, чтобы изменить этот адрес';
      adl_id = 0;

      err_text = :oivo_err_text;
      suspend;
      exit;
    end
  end
  adl_id = 0;
  -- если выбран тип адреса, далее показываем список существующих адресов.
  if (find_kind = 'address_type') then
  begin
    pl_favorite = 0;
    kol = 0;
    res_kind = 'address';
    pl_name = '';
    adl_id_tmp = 0;

    for select distinct adc_from_adl_id, adl_address_str, adl_from_pl_id, adc_address_type
        from addrclient, addrlist
        where adc_from_c_id = :object_id and
              adc_from_adl_id = adl_id and
              adc_address_type = :parent_id and
              (right(adl_id, 9) <> '000000000' and
              coalesce(adl_id, 0) <> 0)
        into :adl_id, :pl_region_name, pl_id, address_type
    do
    begin
      kol = kol + 1;
      select pl_name
      from places
      where pl_id = :pl_id
      into pl_name;
      pl_name = pl_name || ' - Текущий адрес';
      pl_id = address_type;
      adl_id_tmp = adl_id;
      suspend;
    end

    for select adc_from_adl_id, adl_address_str, adl_from_pl_id, min(adc_address_type)
        from addrclient, addrlist
        where adc_from_c_id = :object_id and
              adc_from_adl_id = adl_id and
              adc_address_type <= iif(:parent_id <= 3, 3, 5) and
              adc_address_type <> :parent_id and
              adc_from_adl_id <> :adl_id_tmp and
              (right(adl_id, 9) <> '000000000' and
              coalesce(adl_id, 0) <> 0)
        group by adc_from_adl_id, adl_address_str, adl_from_pl_id
        into :adl_id, :pl_region_name, pl_id, address_type
    do
    begin
      kol = kol + 1;
      select pl_name
      from places
      where pl_id = :pl_id
      into pl_name;
      pl_id = address_type;
      suspend;
    end
    adl_id = 0;
    if (kol > 0) then
    begin
      -- в конце списка добавляем возможность добавления нового адреса
      res_kind = 'country';
      pl_id = parent_id;
      pl_name = 'Новый адрес';
      pl_region_name = 'Нажмите, чтобы добавить новый адрес...';
      suspend;

      exit;
    end
    else
    begin
      -- Если ранее адреса не заполнены, сразу продолжаем поиск
      find_kind = 'country';
    end
  end
  adl_id = 0;
  pl_region_name = '';
  pl_id = 0;

  if (find_kind = 'country') then
  begin
    -- смотрим количество стран в базе
    pl_favorite = 0;
    select count(*)
    from places
    where parent is null
    into :country_count;
    if (country_count is null) then
      country_count = 0;

    if (country_count > 1) then -- если более одного, даем список на выбор
    begin
      -- Если в базе стран более одного, даем на выбор
      res_kind = 'country';
      for select pl_id, pl_name
          from places
          where parent is null
          into :pl_id, pl_name
      do
        suspend;
      exit;
    end
    else
    begin -- иначе берем код страны и продолжаем поиск
      -- Если только одна страна, продолжаем поиск
      select pl_id
      from places
      where parent is null
      into parent_id;
      find_kind = 'favorite';
    end
  end

  if (find_kind = 'favorite') then
  begin
    -- показываем избранные населенные пункты
    kol = 0;
    code = left(parent_id, 3);
    for with tab
        as (select ups_from_pl_id as pl_id, ups_access_kolich as kol_tmp
            from userplacestat
            where ups_from_un_id = :un_id and
                  ups_from_pl_id starting :code
            union
            select adl_from_pl_id as pl_id, 1000 as kol_tmp
            from addrclient, addrlist
            where adc_from_c_id = :object_id and
                  :object_kind = 0 and
                  adc_from_adl_id = adl_id and
                  right(adl_from_pl_id, 9) <> '000000000' and
                  right(adl_from_pl_id, 9) <> '000000001' and
                  adl_from_pl_id starting :code
            union
            select adl_from_pl_id as pl_id, 1000 as kol_tmp
            from contacts, addrlist
            where cc_from_c_id = :object_id and
                  :object_kind = 0 and
                  cc_from_adl_id = adl_id and
                  right(adl_from_pl_id, 9) <> '000000000' and
                  right(adl_from_pl_id, 9) <> '000000001' and
                  adl_from_pl_id starting :code
                  --            union
                  --            select adl_from_pl_id as pl_id, 1000 as kol_tmp
                  --            from contacts c1, contacts c2, addrlist
                  --            where c1.cc_id = :object_id and
                  --                  :object_kind = 13 and
                  --                  c2.cc_from_c_id = c1.cc_from_c_id and
                  --                  c2.cc_from_adl_id = adl_id and
                  --                  right(adl_from_pl_id, 9) <> '000000000' and
                  --                  right(adl_from_pl_id, 9) <> '000000001' and
                  --                  adl_from_pl_id starting :code
        )
        select pl_id, max(kol_tmp)
        from tab
        group by pl_id
        order by 2 desc rows 100
        into pl_id, kol_tmp
    do
    begin
      kol = kol + 1;
      select
             case plt_short_name_side
               when -1 then plt_short_name || ' ' || pl_name
               when 0 then pl_name
               when 1 then pl_name || ' ' || plt_short_name
             end,
             pl_name_komitet, plt_can_select, pl_post_index
      from places, placetype
      where pl_id = :pl_id and
            pl_status = plt_id and
            plt_can_select <> 2
      into pl_name, pl_name_komitet, pl_can_select, pl_post_index;

      select place_full_region
      from aarep_places_link(:pl_id)
      into :pl_region_name;

      if (pl_can_select = 0) then
        res_kind = 'region';
      if (pl_can_select = 1) then
        res_kind = 'city';

      pl_favorite = 0;
      if (pl_can_select = 1) then
      begin
        if (kol_tmp is null) then
          kol_tmp = 0;
        if (kol_tmp >= 1000000) then
          pl_favorite = 1;
        else
          pl_favorite = -1;
      end
      suspend;
    end
    if (kol > 0) then
      exit;
    else
      find_kind = 'region'; -- Если избранных нет, продолжаем поиск
  end

  if (find_kind in ('region' /*, 'city'*/ )) then
  begin
    pl_favorite = 0;
    -- поиск по тексу
    pl_post_index = '';

    for select pl_id, full_text, pl_name, pl_name_komitet, pl_can_select, pl_post_index
        from hp_find_places(:parent_id, :find_place, '')
        into :pl_id, :pl_region_name, :pl_name, :pl_name_komitet, :pl_can_select, pl_post_index
    do
    begin
      if (pl_name_komitet is null) then
        pl_name_komitet = '';
      if (pl_region_name = '') then
        pl_region_name = pl_name;
      if (pl_can_select = 0) then
        res_kind = 'region';
      if (pl_can_select = 1) then
        res_kind = 'city';
      if (res_kind in ('region')) then
      begin
        if (pl_name_komitet <> '' and
            position(pl_name_komitet, pl_name) <= 0) then
          pl_name = pl_name || ' (' || pl_name_komitet || ')';
      end

      pl_favorite = 0;
      if (pl_can_select = 1) then
      begin
        pl_favorite = -1;
        if (exists(select *
                   from userplacestat
                   where ups_from_un_id = :un_id and
                         ups_from_pl_id = :pl_id and
                         ups_access_kolich >= 1000000)) then
          pl_favorite = 1;
      end

      suspend;
    end
  end
end
