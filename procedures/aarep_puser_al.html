CREATE PROCEDURE AAREP_PUSER_AL(
    REG_ID INTEGER)
RETURNS (
    UA_CODE VARCHAR(20),
    AL_NAME VARCHAR(200),
    R_LIST INTEGER,
    AL_FROM_BL_ID VARCHAR(10),
    AL_CURRENCY VARCHAR(3),
    AL_ACCOUNT VARCHAR(20),
    AL_DATE_CLOSE DATE)
AS
declare variable id_list integer;
begin
  for select id_list
      from region_list(:reg_id, 1, 3, 1)
      into :id_list
  do
  begin
    for select al_id, al_name, al_from_reg_id, al_from_bl_id, al_currency, al_account, al_date_close
        from acc_list
        where (al_date_close is null or
              al_date_close > current_date) and
              al_from_reg_id = :id_list
        into ua_code, al_name, r_list, al_from_bl_id, al_currency, al_account, al_date_close
    do
    begin
      suspend;
    end
  end
end
