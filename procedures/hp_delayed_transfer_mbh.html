CREATE PROCEDURE HP_DELAYED_TRANSFER_MBH(
    OL_STR D_STR_100)
AS
declare variable ol_parent_id d_id;
declare variable ol_id d_id;
declare variable tmp_ol_id d_id;
declare variable doc_no d_integer;
declare variable doc_date d_date;
declare variable memo d_str_long;
declare variable un_id d_id;

declare variable ka_id d_id;
declare variable ka_date d_date;
declare variable ka_date_prev d_date;
declare variable kcp_period d_integer;
declare variable offices_id d_integer;
declare variable officer_id d_integer;
declare variable delayed_perc d_currency;

declare variable reg_id d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable line_no d_integer;
begin
  ol_parent_id = (select first 1 res_str from hp_split_str(:ol_str, '-'));

  select
    ka_id,
    ka_date,
    ol_doc_no,
    ol_date,
    ol_memo,
    ol_user_id
  from knakt
    inner join operlist
      on ol_id = ka_from_ol_id
  where ka_from_ol_id = :ol_parent_id
  into
    :ka_id,
    :ka_date,
    :doc_no,
    :doc_date,
    :memo,
    :un_id;
  ka_date_prev = ka_date - extract(day from ka_date);
  kcp_period = 100 * extract(year from ka_date_prev) + extract(month from ka_date_prev);

  for
    select
      kal_prior_reg_id,
      kal_new_reg_id,
      sum(kcp_delayed_perc_end)
    from knaktlist
      inner join kontract
        on k_from_kf_id = kal_from_kf_id
      inner join product
        on pr_id = k_from_pr_id
          and pr_code containing '.murabaha.'
      inner join kontcalcperc
        on kcp_from_k_id = k_id
          and kcp_period = :kcp_period
    where kal_from_ka_id = :ka_id
    group by kal_prior_reg_id, kal_new_reg_id
  into
    :offices_id,
    :officer_id,
    :delayed_perc
  do
  if (offices_id <> officer_id
    and delayed_perc > 0) then
  begin
    acc_dt  = (
      select
        al_id
      from acc_list
      where al_from_reg_id = :offices_id
        and al_currency = '417'
        and al_account = '21114/1');
    acc_kt  = (
      select
        al_id
      from acc_list
      where al_from_reg_id = :officer_id
        and al_currency = '417'
        and al_account = '21114/1');

    for
      select
        res_reg_id,
        res_acc_dt,
        res_acc_kt,
        res_line_no
      from aarep_loronostro_link(:offices_id, :acc_dt, :acc_kt)
    into
      :reg_id,
      :acc_dt,
      :acc_kt,
      :line_no
    do
    begin
      if (line_no = 1) then
        tmp_ol_id = ol_parent_id;

      insert into operlist (
        ol_parent_id,
        ol_from_reg_id,
        ol_oper_type,
        ol_oper_subtype,
        ol_date,
        ol_doc_no,
        ol_doc_kind,
        ol_memo,
        ol_user_id
      )
      values (
        :tmp_ol_id,
        :reg_id,
        2,
        22,
        :doc_date,
        :doc_no,
        9,
        :memo,
        :un_id
      )
      returning ol_id
      into :ol_id;

      if (line_no = 1) then
        tmp_ol_id = ol_id;

      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id,
        '417',
        1,
        :acc_dt,
        :acc_kt,
        :delayed_perc,
        :delayed_perc
        );
    end
  end
end
