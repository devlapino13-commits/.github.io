                         group by okf.oko_from_k_id
                         having sum(okf.oko_os + okf.oko_perc + okf.oko_sh_os + okf.oko_sh_perc) > 0)) then

          -- прохожусь по внесистемным займам клиента
          for select distinct okf.oko_from_k_id
              from operkroff okf
              where okf.oko_ol_is_active = 1 and
                    okf.oko_from_c_id = :c_id
              into :okf_k_id
          do
          begin
            -- если есть задолженность по кредиту на внесистемном учете, принимаем платеж
            if (exists(select okf.oko_from_k_id
                       from operkroff okf
                       where okf.oko_ol_is_active = 1 and
                             okf.oko_from_k_id = :okf_k_id
                       group by okf.oko_from_k_id
                       having sum(okf.oko_os + okf.oko_perc + okf.oko_sh_os + okf.oko_sh_perc) > 0)) then
              okf_debt = 1;
          end
        end

        if (okf_debt = 0) then
        begin
          err_no = 60;
          err_text = 'Ошибка. Обратитесь в офис Компании'; -- У клиента нет активных кредитов
        end
      end
      else
      if (t = 1) then
      begin
        err_no = 60;
        err_text = 'Погашение в день выдачи запрещено';
      end
      else -- t=2 есть подходящие для погашения кредиты
      begin
        select cl.c_name_fio
        from clients cl
        where cl.c_id = :c_id
        into c_name_fio;
      end
    end

    if (err_no = 0) then
    begin
      select err_no, err_text
      from hp_extsys_check_unearned_pay(:project, :esm_id, :cl_birthdate || ':' || :cl_uniq_acc, :c_id, :k_id)
      into :err_no, :err_text;
    end

    /*В случае ошибки, сохранение данных и отправка отчета об ошибке*/
    if (err_no <> 0) then
    begin
      update extsysmain
      set esm_result_id = :err_no,
          esm_status = -1
      where esm_id = :esm_id;

      result = 'ans_code=' || iif(err_no < 10, '0' || err_no, err_no) || '&ansid=' || err_text || '&message=' || err_text;

      result = replace(result, ' ', '_');
      result = replace(result, '-', '=');

      insert into extsysanswer (esa_id, esa_from_esm_id, esa_message, esa_result_id, esa_status)
      values (-1, :esm_id, :result, :err_no, -1);
      suspend;
      exit;
    end

    -- Все проверки пройдены, заполняем оставшиеся поля
    update extsysmain
    set esm_object_id = :c_id,
        esm_object_kind = 0,
        esm_result_id = 0,
        esm_status = 0
    where esm_id = :esm_id;

    if (command = 'check') then
      result = 'ans_code=00&ansid=' || c_name_fio || '&message=OK';
    else
    if (command = 'pay') then
      result = 'ans_code=00&ansid=' || c_name_fio || '; сумма=' || round(cl_amount, 2) || '&message=OK' || '; auth_code=' || txn_id;

    result = replace(result, ' ', '_');
    result = replace(result, '-', '=');

    insert into extsysanswer (esa_id, esa_from_esm_id, esa_message, esa_result_id, esa_status)
    values (-1, :esm_id, :result, :err_no, 0);

    suspend;
  end -- < if (project = 'eleksnet') then

  /* -------------------------------- Moneta.ru ------------------------------*/
  if (project in ('moneta', 'tinkoff', 'processing_kz', 'Wirebank', 'CaRuSell')) then
  begin
    err_no = 200; -- Заказ оплачен (если не будет ошибок и запрос - pay)
    err_text = '';
    c_id = null;
    is_test = 0;
    command = 'pay';
    currency_code = '';
    k_id = null;

    /* Получаем значения главных полей, и обновляем extsysmain */
    for select field_code, field_value
        from hp_extsys_parcer(:params, '')
        into :field_code, :field_value
    do
    begin
      if (field_code = 'mnt_command') then
        command = field_value;
      if (field_code = 'mnt_subscriber_id') then
        c_id = field_value;
      if (field_code = 'mnt_test_mode') then
        is_test = field_value;
      if (field_code = 'mnt_currency_code') then
        currency_code = field_value;
      if (field_code = 'k_id') then
        k_id = field_value;
    end

    update extsysmain
    set esm_account = :c_id,
        esm_action = lower(:command),
        esm_ext_datetime = current_timestamp
    where esm_id = :esm_id;

    if (c_id is null) then
    begin
      err_no = 500;
      err_text = 'Не указан клиент, оплата не может быть произведена';
    end
    else
    begin
      update extsysmain
      set esm_object_kind = 0,
          esm_object_id = :c_id
      where esm_id = :esm_id;
    end

    if (cl_amount <= 0) then
    begin
      err_no = 500;
      err_text = 'Не указана сумма погашения кредита';
    end

    if ((currency_code <> 'rub') and (project <> 'processing_kz')) then
    begin
      err_no = 500;
      err_text = 'Неверная валюта платежа';
    end

    /* Проверка наличия у клиента активных кредитов*/
    if ((err_no = 200) and
        (command = 'check')) then
    begin
      tmp_timestamp = current_timestamp;

      -- t=0 нет активных кредитов,
      -- t=1 текущая дата равна дате выдачи (погашение запрещено),
      -- t=2 есть подходящие для погашения кредиты
      t = 0;
      for select ks_date_begin
          from knclient, kontstate ks
          where kc_from_c_id = :c_id and
                kc_from_k_id = ks_from_k_id and
                ks_ol_is_delete = 0 and
                ks_state = 1 and
                cast(:tmp_timestamp as date) between ks_date_begin and ks_date_end and
                ((ks.ks_from_k_id = :k_id) or
                (:k_id is null)) and
                kc_amount_approved > 0
          into tmp_date
      do
      begin
        if (tmp_date <> cast(tmp_timestamp as date)) then
          t = 2;
        else
        if (t <> 2) then
        begin
          t = 1;
          if (db_id in (81001, 39801) and f_get_const('repayment_on_disb_date') = 1) then  -- RFCKZ-25 для КЗ и РФ разрешаем погашение в день выдачи займа
            t = 2;
        end
      end

      if (t = 0) then
      begin
        okf_debt = 0;

        if (db_id in (81001, 41701)) then
        begin
          -- прохожусь по внесистемным займам клиента
          for select distinct okf.oko_from_k_id
              from operkroff okf
              where okf.oko_ol_is_active = 1 and
                    okf.oko_from_c_id = :c_id and
                    ((okf.oko_from_k_id = :k_id) or
                    (:k_id is null))
              into :okf_k_id
          do
          begin
            -- если есть задолженность по кредиту на внесистемном учете, принимаем платеж
            if (exists(select okf.oko_from_k_id
                       from operkroff okf
                       where okf.oko_ol_is_active = 1 and
                             okf.oko_from_k_id = :okf_k_id
                       group by okf.oko_from_k_id
                       having sum(okf.oko_os + okf.oko_perc + okf.oko_sh_os + okf.oko_sh_perc) > 0)) then
              okf_debt = 1;
          end
        end
        if (okf_debt = 0) then
        begin
          err_no = 500;
          err_text = 'Ошибка. Обратитесь в офис Компании'; -- У клиента нет активных кредитов
        end
      end
      else
      if (t = 1) then
      begin
        err_no = 500;
        err_text = 'Ошибка 505'; -- RFCRU-4080
      end
      else -- t=2 есть подходящие для погашения кредиты
      begin
        select cl.c_name_fio
        from clients cl
        where cl.c_id = :c_id
        into c_name_fio;
      end
    end

    if (err_text = '') then
    begin
      select err_no, err_text
      from hp_extsys_check_unearned_pay(:project, :esm_id, :c_id, :c_id, :k_id)
      into :err_no, :err_text;

    end

    mnt_description = err_text;
    if ((err_text = '') and
        (command = 'pay')) then
    begin
      if (k_id is null) then
      begin
        select oks_from_k_id
        from oksummary, knclient
        where oks_from_k_id = kc_from_k_id and
              kc_from_c_id = :c_id and
              kc_amount_approved > 0 and
              oks_date_end >= '31.12.2999' and
              oks_close_state = 0
        order by oks_delinq_date, oks_balance
        rows 1
        into :esm_k_id;
      end
      else
        esm_k_id = :k_id;

      if (esm_k_id is null) then
      begin
        select k_id
        from (select okf.oko_from_k_id as k_id, max(okf.oko_ol_date) as ol_date
              from operkroff okf
              inner join kontstate ks on ks.ks_from_k_id = okf.oko_from_k_id
              inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
              where okf.oko_ol_is_active = 1 and
                    ks.ks_state = 2 and
                    ks.ks_ol_is_delete = 0 and
                    current_date between ks.ks_date_begin and ks.ks_date_end and
                    kncl.kc_amount_approved > 0 and
                    kncl.kc_from_c_id = :c_id
              group by okf.oko_from_k_id
              having sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc) > 0) a
        order by ol_date
        rows 1
        into :esm_k_id;
      end

      if (esm_k_id is not null) then
      begin
        select kont.k_kontract_no, kont.k_kolich_client_approved
        from kontract kont
        where kont.k_id = :esm_k_id
        into :k_kontract_no, :k_kolich_client_approved;

        fio = 'Групповой займ';
        if (k_kolich_client_approved = 1) then
        begin
          select cl.c_name_fio
          from clients cl
          where cl.c_id = :c_id
          into :fio;
        end
      end
      mnt_description = 'Погашение займа по договору №' || :k_kontract_no || ', ' || fio || ', без НДС';
    end

    if (command = 'check' and
        err_no = 200) then
      err_no = 402; -- Заказ готов к оплате

    -- Вычисляем статус
    t = iif(err_no in (402, 200), iif(is_test = 1, -2, 0), -1);

    update extsysmain em
    set em.esm_result_id = :err_no,
        em.esm_status = :t,
        em.esm_from_k_id = :k_id
    where esm_id = :esm_id;


    if (db_id = 81001) then
    begin
      select iif(:project = 'Wirebank',
          iif(pr.pr_code containing '.oivo_pack.', f_get_const('wirebank_oivo_acc'), f_get_const('wirebank_acc')),
          iif(pr.pr_code containing '.oivo_pack.', f_get_const('moneta_oivo_acc'), f_get_const('moneta_acc')))
      from kontract kont
      inner join product pr on pr.pr_id = kont.k_from_pr_id
      where kont.k_id = :k_id
      into mnt_id;
    end
    else if (db_id = 39801) then
        mnt_id = f_get_const('processing_kz_acc');
    else
        mnt_id = f_get_const('moneta_acc');

    result = '<?xml version="1.0" encoding="UTF-8"?>' || '<MNT_RESPONSE>' || ' <MNT_ID>' || mnt_id || '</MNT_ID>' || ' <MNT_TRANSACTION_ID>' || txn_id || '</MNT_TRANSACTION_ID>' || ' <MNT_RESULT_CODE>' || err_no || '</MNT_RESULT_CODE>' || ' <MNT_AMOUNT>' || coalesce(cast(cl_amount as decimal(18,2)), '') || '</MNT_AMOUNT>' || -- md5 от подписи возьмем в сервисе, там же в хвост строки доклеивается код проверки целостности данных
    ' <MNT_SIGNATURE>' || err_no || mnt_id || txn_id || '</MNT_SIGNATURE>' || ' <MNT_DESCRIPTION>' || mnt_description || '</MNT_DESCRIPTION>' || '</MNT_RESPONSE>';

    insert into extsysanswer (esa_id, esa_from_esm_id, esa_message, esa_result_id, esa_status)
    values (-1, :esm_id, :result, :err_no, :t);

    suspend;
  end -- < if (project = 'moneta') then
