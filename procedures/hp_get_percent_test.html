CREATE PROCEDURE HP_GET_PERCENT_TEST(
    LOANPRODUCTID INTEGER,
    LOANDISBDATE DATE,
    CLIENT_COUNT INTEGER,
    ALL_CL_AMOUNT CURR_TYPE,
    LOANPERIOD INTEGER,
    MIN_CL_AMOUNT CURR_TYPE,
    MAX_CL_AMOUNT CURR_TYPE,
    C_ID_LIST VARCHAR(255),
    ADD_MONTH SMALLINT = 0,
    REFUND SMALLINT = 0,
    DISB_KIND D_INTEGER = 0)
RETURNS (
    PERC_MIN PERC_TYPE,
    PERC_MAX PERC_TYPE,
    PERC_DEFAULT PERC_TYPE,
    KOMMIS_PERC_MIN PERC_TYPE,
    KOMMIS_PERC_MAX PERC_TYPE,
    KOMMIS_PERC_DEFAULT PERC_TYPE,
    PERIOD_MIN INTEGER,
    PERIOD_MAX INTEGER,
    PERIOD_DEFAULT INTEGER,
    IS_CORRECT INTEGER,
    ADD_INFO VARCHAR(255))
AS
declare variable db_id integer;
declare variable prl_percent_min curr_type;
declare variable prl_percent_max curr_type;
declare variable prl_period_min integer;
declare variable prl_period_max integer;
declare variable prl_kommis_perc_min curr_type;
declare variable prl_kommis_perc_max curr_type;
declare variable percent_lgota perc_type;
declare variable res_kind integer;
declare variable re_k_id id_type;
declare variable re_percent_per_day perc_type;
declare variable max_delinq_day integer;
declare variable pr_code varchar(100);
declare variable re_kommis_perc perc_type;
declare variable re_pr_code varchar(100);
declare variable re_percent_year perc_type;
declare variable sql_text varchar(1024);
declare variable c_id id_type;
declare variable rep_c_id_list varchar(255);
declare variable kol_cli_is_kib smallint;
declare variable kol_cli smallint;
declare variable is_new_client smallint;
declare variable is_kibhistory smallint;
declare variable kol_uchitel smallint;
declare variable pr_percent_for_days integer;
declare variable cell_tmp varchar(20);
declare variable ks_date_end date;
declare variable kol_cli_new integer;
declare variable kol_cli_old integer;
declare variable is_percent_correct integer;
declare variable re_disb_date date;
declare variable re_close_date date;
declare variable not_repeated smallint;
declare variable re_add_info varchar(255);
declare variable group_status integer;
declare variable c_curr_status integer;
declare variable pr_product_kind smallint;
declare variable c_id_oskred id_type;
declare variable kc_circle integer;
declare variable kc_max_circle integer;
declare variable main_kont_percent perc_type;
declare variable re_k_balance d_currency;
declare variable active_kont_percent d_percent;
declare variable active_pr_id d_integer;
declare variable same_product d_smallint;
declare variable active_kont_percent_tmp d_percent;
declare variable active_pr_id_tmp d_integer;
declare variable group_ball d_integer;
declare variable c_curr_ball d_integer;
declare variable c_stage_month d_currency;
declare variable score_rate d_integer;
declare variable avg_score d_integer;
declare variable client_count_for_score d_integer;
declare variable koef decimal(18,4);
declare variable cal_card_id d_id;
declare variable k_percent integer;
declare variable last_k_id d_id;
declare variable prl_id d_integer;
begin
  db_id = f_get_db_id();
  main_kont_percent = 100;
  koef = 0;

  loandisbdate = coalesce(:loandisbdate, current_date);
  is_correct = 0;
  add_info = '';
  if (db_id = 81001) then
  begin
    perc_min = null;
    perc_max = null;
    kommis_perc_min = null;
    kommis_perc_max = null;
    period_min = null;
    period_max = null;

    for select coalesce(prl_percent_min, 0), coalesce(prl_percent_max, 100), coalesce(prl_repay_min, 1),
               coalesce(prl_repay_max, 120), coalesce(prl_admin_perc_min, 0), coalesce(prl_admin_perc_max, 100)
        from aarep_product_logic_list(:loanproductid, :loandisbdate, :client_count, -1, :all_cl_amount, -1, :loanperiod, -1, -1, -1, :min_cl_amount, :max_cl_amount, -1)
        into :prl_percent_min, prl_percent_max, prl_period_min, prl_period_max, :prl_kommis_perc_min,
             :prl_kommis_perc_max
    do
    begin
      is_correct = 1;
      if (perc_min is null) then
        perc_min = prl_percent_min;
      if (perc_max is null) then
        perc_max = prl_percent_max;
      if (perc_min > prl_percent_min) then
        perc_min = prl_percent_min;
      if (perc_max < prl_percent_max) then
        perc_max = prl_percent_max;

      if (kommis_perc_min is null) then
        kommis_perc_min = prl_kommis_perc_min;
      if (kommis_perc_max is null) then
        kommis_perc_max = prl_kommis_perc_max;
      if (kommis_perc_min > prl_kommis_perc_min) then
        kommis_perc_min = prl_kommis_perc_min;
      if (kommis_perc_max < prl_kommis_perc_max) then
        kommis_perc_max = prl_kommis_perc_max;

      if (period_min is null) then
        period_min = prl_period_min;
      if (period_max is null) then
        period_max = prl_period_max;
      if (period_min > prl_period_min) then
        period_min = prl_period_min;
      if (period_max < prl_period_max) then
        period_max = prl_period_max;

    end

    if (perc_min is null) then
    begin
      perc_min = 0.0;
      perc_max = 100.0;
    end
    perc_default = perc_max;

    if (kommis_perc_min is null) then
    begin
      kommis_perc_min = 0.0;
      kommis_perc_max = 100.0;
    end
    kommis_perc_default = kommis_perc_max;

    if (period_min is null) then
    begin
      period_min = 1;
      period_max = 12;
    end

    period_default = period_max;
    if (12 between period_min and period_max) then
      period_default = 12;
    if (loanperiod > 0) then
      period_default = loanperiod;

    res_kind = 0;
    percent_lgota = 0.0;
    not_repeated = 0; --------------- нужно сделать проверку, если предыдущий цикл брал по по новым ставкам т.е. завышенным
    --------------- тогда на эти проверки не нужно проверять т.е. заранее нужно сделать not_repeated = 1
    if (perc_min > perc_max) then
      perc_min = perc_max;

    if (coalesce(c_id_list, '') <> '' and
        loanproductid > 0) then
    begin
      select pr_code, pr_percent_for_days, pr_product_kind
      from product
      where pr_id = :loanproductid
      into pr_code, pr_percent_for_days, pr_product_kind;

      if (loandisbdate <= '15.10.2017') then
      begin
        if ((pr_code containing 'barakat' or pr_code containing 'family') and
            pr_code not containing '.ACT.' and
            pr_code not containing '.KAZAN.') then
        begin
          res_kind = 1;
          --      проверка клиентов, попадает под акцией по возвращению клиентов
          kol_cli = 0;
          kol_uchitel = 0; -- данную переменную использовал чтобы не создавать новую переменную, данная переменная при работе РФ нигде не учавствовал
          if (coalesce(c_id_list, '') <> '') then
          begin
            rep_c_id_list = replace(c_id_list, ';', ',');
            sql_text = 'select c_id from clients
                   where c_id in (' || :rep_c_id_list || ')';
            for execute statement sql_text
                    into c_id
            do
            begin
              kol_cli = kol_cli + 1;
              cell_tmp = '';
              select cont_value
              from hp_contdata_get_cell_valid(0, :c_id)
              into :cell_tmp;
              if (cell_tmp <> '') then
              begin
                if (exists(select *
                           from listcellno
                           where lc_cell_no = :cell_tmp)) then
                begin
                  kol_uchitel = kol_uchitel + 1;
                end
              end
              --            Реализация по новым условиям, выданные от 7 июня 2017г.
              if (not_repeated = 0) then
              begin
                --              проверка, у клиенты был ли кредит выданный до 07.06.2017
                if (exists(select *
                           from knclient, kontstate
                           where kc_from_c_id = :c_id and
                                 kc_amount_approved > 0 and
                                 kc_from_k_id = ks_from_k_id and
                                 ks_ol_is_delete = 0 and
                                 ks_state = 1 and
                                 ks_date_begin < '07.06.2017')) then
                begin
                  --                проверка предыдущего цикла клиента
                  re_k_id = null;
                  re_disb_date = null;
                  re_close_date = null;
                  select k_id, ks_date_begin, ks_date_end, k_add_info
                  from knclient, kontstate, kontract, product
                  where kc_from_c_id = :c_id and
                        kc_amount_approved > 0 and
                        kc_from_k_id = ks_from_k_id and
                        ks_ol_is_delete = 0 and
                        ks_state = 1 and
                        ks_date_begin < :loandisbdate and
                        ks_from_k_id = k_id and
                        k_from_pr_id = pr_id and
                        pr_product_kind <> 2
                  order by ks_date_begin desc
                  rows 1
                  into re_k_id, re_disb_date, re_close_date, re_add_info;
                  max_delinq_day = 100;
                  if (re_k_id is not null) then
                  begin
                    if (loandisbdate - 11 > re_close_date) then
                      not_repeated = 1;
                    else


                    -- если один раз по новому продукту брал то в след. раз не может брать по старым продуктам
                    if (re_add_info = 'new_product') then
                      not_repeated = 1;
                    else
                    begin
                      max_delinq_day = null;
                      select max(ok_ol_date - ok_sh_date)
                      from operkred
                      where ok_ol_is_active = 1 and
                            ok_from_k_id = :re_k_id and
                            ok_type in (4, 5) and
                            ok_os + ok_perc < 0 and
                            ok_ol_date > ok_sh_date
                      into max_delinq_day;
                      max_delinq_day = coalesce(max_delinq_day, 0);
                    end
                  end
                  else
                  begin
                    not_repeated = 1;
                  end

                  if (not_repeated = 0) then
                  begin
                    --                  проверка статуса клиента
                    if ((not exists(select *
                                    from hp_kontstate_client_status(:c_id, :loandisbdate)
                                    where c_curr_bonus_status > 0)) and
                        max_delinq_day > 7) then
                    begin
                      not_repeated = 1;
                    end
                  end
                end
                else
                begin
                  not_repeated = 1;
                end
              end
            end
          end
          if (kol_uchitel = 0) then
            kol_uchitel = -1;

          if (kol_cli = kol_uchitel) then -- значить попадает под акцией
          begin
            percent_lgota = 1;
          end
          else
          begin
            select c_curr_bonus_status
            from hp_kontstate_loan_status(0, 0, 0, :loandisbdate, :c_id_list)
            into :percent_lgota;
            if (percent_lgota is null) then
              percent_lgota = 0.0;
            percent_lgota = percent_lgota * 0.2;

            if (not_repeated = 1) then
            begin
              add_info = 'new_product';
              if (pr_code containing '.BARAKAT.') then
              begin
                if (loanperiod <= 6) then
                begin
                  perc_max = 8.9;
                  kommis_perc_max = 1.5;
                end
                else
                begin
                  perc_max = 9.4;
                  kommis_perc_max = 2.0;
                end
              end
              else
              if (pr_code containing '.BARAKAT PLUS.') then
              begin
                if (loanperiod <= 6) then
                begin
                  perc_max = 8.2;
                  kommis_perc_max = 1.0;
                end
                else
                begin
                  perc_max = 8.7;
                  kommis_perc_max = 1.5;
                end
              end
              else
              if (pr_code containing '.FAMILY.') then
              begin
                if (loanperiod <= 6) then
                begin
                  perc_max = 7.5;
                  kommis_perc_max = 0.0;
                end
                else
                begin
                  perc_max = 8.0;
                  kommis_perc_max = 1.0;
                end
              end
              else
              if (pr_code containing '.FAMILY PLUS.') then
              begin
                if (loanperiod <= 6) then
                begin
                  perc_max = 8.5;
                  kommis_perc_max = 1.0;
                end
                else
                begin
                  perc_max = 9.0;
                  kommis_perc_max = 1.5;
                end
              end
              kommis_perc_default = kommis_perc_max;
              kommis_perc_min = kommis_perc_default;
            end
          end
        end
        else
        if (pr_code containing '.ACT.') then
        begin
          select re_k_id, re_percent_per_day
          from hp_get_last_circle(0, 0, :c_id_list, :loandisbdate)
          into re_k_id, re_percent_per_day;

          if (re_k_id is not null) then
          begin
            perc_default = round(re_percent_per_day * pr_percent_for_days, 1) - 1;

            if (perc_min < perc_default) then
              perc_min = perc_default;
            else
              perc_default = perc_min;
            if (perc_max > perc_default) then
              perc_max = perc_default;
            else
            if (perc_default > perc_max) then
              perc_default = perc_max;

            kommis_perc_min = 0;
            kommis_perc_max = 0;
            kommis_perc_default = 0;
          end
        end
        else
        if (pr_code containing '.KAZAN.') then
        begin
          select re_k_id, re_percent_per_day
          from hp_get_last_circle(0, 0, :c_id_list, :loandisbdate)
          into re_k_id, re_percent_per_day;
          re_pr_code = '';
          if (re_k_id is not null) then
          begin
            --          проверяем предыдущий цикл, для выяснение что кредит перекредитуется или нет
            ks_date_end = null;
            select ks_date_end
            from kontstate
            where ks_ol_is_delete = 0 and
                  ks_state = 1 and
                  ks_from_k_id = :re_k_id
            into ks_date_end;
            ks_date_end = coalesce(ks_date_end, '01.01.1990');

            if (ks_date_end >= loandisbdate - 1) then -- перекредитуется
            begin
              kommis_perc_min = 0;
              kommis_perc_max = 0;
              kommis_perc_default = 0;
            end

            select pr_code
            from kontract, product
            where k_id = :re_k_id and
                  k_from_pr_id = pr_id
            into re_pr_code;
          end
          --        если предыдущий цикл был акция для казани то в тех же условиях должен брать кредит
          if (re_pr_code containing '.KAZAN.') then
          begin
            perc_default = round(re_percent_per_day * pr_percent_for_days, 1);
          end
          else
          begin
            --          perc_default = round(re_percent_per_day * pr_percent_for_days,1) - 1;
            res_kind = 1;
            select c_curr_bonus_status
            from hp_kontstate_loan_status(0, 0, 0, :loandisbdate, :c_id_list)
            into :percent_lgota;
            if (percent_lgota is null) then
              percent_lgota = 0.0;
            percent_lgota = percent_lgota * 0.2 + 1;

          end
        end
      end
      else ------- Разработка по новым статусам, который дествует с 16 октября 2017 года
      if (pr_code containing 'barakat' or pr_code containing 'family' or pr_product_kind = 2 or pr_code containing 'blago' or pr_code containing 'uspeh' or
      pr_code containing '.zalog_car.' or pr_code containing '.education.' or pr_code containing '.food.' or pr_code containing '.reliable_client.' or
      pr_code containing '.reliable_client_plus.' or pr_code containing '.reliable_emp.' or pr_code containing '.express_online.' or pr_code containing '.zalog.' or
      pr_code containing 'selfemployed' or pr_code containing '.universal.' or pr_code containing '.oivo.') then
      begin
        group_status = 5;
        group_ball = 100;
        c_curr_status = null;
        c_curr_ball = null;

        rep_c_id_list = replace(c_id_list, ';', ',');
        sql_text = 'select c_id from clients
           where c_id in (' || :rep_c_id_list || ')';
        for execute statement sql_text
                into c_id
        do
        begin
          if ((pr_product_kind <> 2 or pr_code containing '.express_online.') and
              pr_product_kind <> -1) then
          begin
            --          если в группе несколько участвников то берем статус/балл самого минимального пункт 3.2.
            c_curr_status = null;
            select c_curr_status
            from hp_get_client_status(:c_id, :loandisbdate)
            into c_curr_status;
            if (c_curr_status is null) then
              c_curr_status = 0;
            if (c_curr_status < group_status) then
              group_status = c_curr_status;

            c_curr_ball = null;
            c_curr_ball = f_point(:c_id, :loandisbdate);

            if (c_curr_ball is null) then
              c_curr_ball = 0;

            if (c_curr_ball < group_ball) then
              group_ball = c_curr_ball;
          end
          else
          if (pr_product_kind = 2) then
          begin
            if (loandisbdate < '01.11.2020') then
            begin
              --          если параллельный кредит то вычисляем основной активный заем
              pr_code = null;
              re_k_id = null;
              select k_id, pr_code, kont.k_percent
              from kontstate, kontract kont, product, knclient
              where ks_ol_is_delete = 0 and
                    ks_state = 1 and
                    :loandisbdate between ks_date_begin and ks_date_end and
                    ks_from_k_id = k_id and
                    k_from_pr_id = pr_id and
                    pr_product_kind <> 2 and
                    k_id = kc_from_k_id and
                    kc_amount_approved > 0 and
                    kc_from_c_id = :c_id
              order by ks_date_begin desc
              rows 1
              into re_k_id, pr_code, main_kont_percent;
              pr_product_kind = -1; -- сделано чтобы второй раз не обрабатывал данный скрипт, если в параллельном 2 клиента то у них предыдущий цикл один и тот же контракт
              if (pr_code is null) then
                perc_max = 100; -- это сделано для того чтобы если основного кредита не будет то процент показывал 100%

              for select kc_from_c_id
                  from knclient
                  where kc_from_k_id = :re_k_id and
                        kc_amount_approved > 0
                  into c_id_oskred
              do
              begin
                --            если в группе несколько участвников то берем статус самого минимального пункт 3.2.
                c_curr_status = null;
                select c_last_status
                from hp_get_client_status(:c_id_oskred, :loandisbdate)
                into c_curr_status;
                if (c_curr_status is null) then
                  c_curr_status = 0;
                if (c_curr_status < group_status) then
                  group_status = c_curr_status;
              end
            end
            else
            if (pr_code containing '.LOYAL.') then --параллельный
            begin
              pr_code = '.reliable_client.'; --  на момент получения займа «Параллельный» процентная ставка определяется по ставке продукта «Надежный клиент» согласно накопленным баллам
              c_curr_ball = null;
              c_curr_ball = f_point(:c_id, :loandisbdate);

              if (c_curr_ball is null) then
                c_curr_ball = 0;

              if (c_curr_ball < group_ball) then
                group_ball = c_curr_ball;
            end
          end
        end

        if (c_curr_status is null) then
          group_status = 0;

        percent_lgota = case group_status
                          when 5 then 1.00
                          when 4 then 0.75
                          when 3 then 0.75
                          when 2 then 0.50
                          when 1 then 0.25
                          else 0
                        end;

        if (current_date < '01.01.2021') then
        begin
          add_info = case group_status
                       when 5 then 'Статус: *****'
                       when 4 then 'Статус: ****'
                       when 3 then 'Статус: ***'
                       when 2 then 'Статус: **'
                       when 1 then 'Статус: *'
                       else 'Без статуса'
                     end;
        end
        else
        begin
          add_info = 'Балл:' || group_ball;
        end

        if (pr_code containing '.food.') then
        begin
          if (loanperiod < 10) then
          begin
            perc_max = 9.4;
            kommis_perc_max = 0;
          end
          else
          begin
            perc_max = 9.9;
            kommis_perc_max = 0;
          end
        end
        else
        if (pr_code containing '.shans plus.') then
        begin
          perc_max = 9.9;
          kommis_perc_max = 0;
        end
        else
        if (pr_code containing '.status_plus.') then
        begin
          perc_max = 7.9;
          kommis_perc_max = 0;
        end
        else
        if (pr_code containing '.BARAKAT.' or pr_code containing '.blago.') then
        begin
          if (loanperiod <= 6) then
          begin
            perc_max = 8.9;
            kommis_perc_max = 1.5;
          end
          else
          begin
            perc_max = 9.4;
            kommis_perc_max = 2.0;
          end

        perc_min = prl_percent_min;
      if (perc_max is null) then
        perc_max = prl_percent_max;
      if (perc_min > prl_percent_min) then
        perc_min = prl_percent_min;
      if (perc_max < prl_percent_max) then
        perc_max = prl_percent_max;

      if (kommis_perc_min is null) then
        kommis_perc_min = prl_kommis_perc_min;
      if (kommis_perc_max is null) then
        kommis_perc_max = prl_kommis_perc_max;
      if (kommis_perc_min > prl_kommis_perc_min) then
        kommis_perc_min = prl_kommis_perc_min;
      if (kommis_perc_max < prl_kommis_perc_max) then
        kommis_perc_max = prl_kommis_perc_max;

      if (period_min is null) then
        period_min = prl_period_min;
      if (period_max is null) then
        period_max = prl_period_max;
      if (period_min > prl_period_min) then
        period_min = prl_period_min;
      if (period_max < prl_period_max) then
        period_max = prl_period_max;

    end

    if (perc_min is null) then
    begin
      perc_min = 0.0;
      perc_max = 100.0;
    end
    perc_default = perc_max;

    if (kommis_perc_min is null) then
    begin
      kommis_perc_min = 0.0;
      kommis_perc_max = 100.0;
    end
    kommis_perc_default = kommis_perc_max;

    if (period_min is null) then
    begin
      period_min = 1;
      period_max = 18;
    end

    period_default = period_max;
    if (18 between period_min and period_max) then
      period_default = 18;
    if (loanperiod > 0) then
      period_default = loanperiod;

    select pr_code
    from product
    where pr_id = :loanproductid
    into pr_code;

    if (pr_code containing '.score_rate.' and coalesce(c_id_list, '') <> '') then
    begin
        sql_text = 'select c_id from clients where c_id in ('||replace(:c_id_list, ';', ',')||')';
        for execute statement sql_text
        into c_id
        do
        begin
            score_rate = f_point(:c_id, :loandisbdate);

            select k_id,
                   k_percent
            from kontract
            inner join knclient on kc_from_k_id = k_id
            inner join kontstate on ks_from_k_id = kc_from_k_id
            where kc_from_c_id = :c_id and
                  kc_amount_approved > 0 and
                  ks_ol_is_delete = 0 and
                  ks_state = 1 and
                  ks_date_begin <= :loandisbdate
            order by ks_date_begin desc rows 1
            into last_k_id, -- предыдущий займ
                 k_percent; -- % ставка предыдущего займа
    
            max_delinq_day = null;
            select max(ok_ol_date - ok_sh_date)
            from operkred
            where ok_ol_is_active = 1 and
                  ok_from_k_id = :last_k_id and
                  ok_type in (4, 5) and
                  ok_os + ok_perc < 0 and
                  ok_ol_date > ok_sh_date
            into max_delinq_day; -- максимальная просрочка предыдущего займа
            max_delinq_day = coalesce(max_delinq_day, 0);

            -- Если в предыдущем цикле клиент кредитовался по 23% или 24% ставке и не допустил ни дня просрочки,
            -- то и в новом цикле он кредитуется по той же % ставке
            if (k_percent in (23, 24) and max_delinq_day = 0) then
              perc_default = k_percent;

            else
            begin
                perc_default =
                case
                    when score_rate between 0 and 2 then 43.5
                    when score_rate between 3 and 5 then 40.5
                    when score_rate between 6 and 11 then 36
                    when score_rate between 12 and 17 then 34
                    when score_rate between 18 and 23 then 32
                    when score_rate between 24 and 29 then 30
                    when score_rate between 30 and 35 then 28
                    when score_rate between 36 and 41 then 26
                    when score_rate between 42 and 47 then 25
                    when score_rate >= 48 then 24.5
                    else 0
                end;
            
                -- RFCAZ-392
                if (pr_code containing ('.test_product.')) then
                    period_default =
                    case
                        when score_rate between 0 and 2 then 18
                        when score_rate between 3 and 5 then 18
                        when score_rate between 6 and 23 then 24
                        when score_rate >= 24 then 36
                        else 0
                    end;
                else
                    period_default =
                    case
                        when score_rate between 0 and 2 then 18
                        when score_rate between 3 and 5 then 18
                        when score_rate >= 6 then 24
                        else 0
                    end;    
            end
        end

        if (perc_min < perc_default) then
            perc_min = perc_default;
        else
            perc_default = perc_min;

        if (perc_max > perc_default) then
            perc_max = perc_default;
        else if (perc_default > perc_max) then
            perc_default = perc_max;

        kommis_perc_min = 2;
        kommis_perc_max = 3;
    end
    suspend;
  end
