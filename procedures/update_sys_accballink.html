CREATE PROCEDURE UPDATE_SYS_ACCBALLINK(
    NEW_SABL_SAB_ID D_ID,
    NEW_SABL_TYPE D_ID,
    NEW_SABL_RESIDENT_AB_ID D_ID,
    NEW_SABL_NON_RESIDENT_AB_ID D_ID,
    NEW_SABL_RESIDENT_INX_ID D_ID,
    NEW_SABL_NON_RESIDENT_INX_ID D_ID)
RETURNS (
    SABL_SAB_ID D_ID,
    SABL_TYPE D_ID,
    SABL_RESIDENT_AB_ID D_ID,
    SABL_NON_RESIDENT_AB_ID D_ID,
    SABL_RESIDENT_INX_ID D_ID,
    SABL_NON_RESIDENT_INX_ID D_ID)
AS
begin
  sabl_sab_id = new_sabl_sab_id;
  sabl_type = new_sabl_type;

  if (new_sabl_resident_ab_id is null and
      new_sabl_non_resident_ab_id is null and
      new_sabl_resident_inx_id is null and
      new_sabl_non_resident_inx_id is null) then
  begin
    delete from sys_accballink
    where (sabl_sab_id = :sabl_sab_id) and
          (sabl_type = :sabl_type);
  end
  else
  begin
    update or insert into sys_accballink (sabl_sab_id, sabl_type, sabl_resident_ab_id, sabl_non_resident_ab_id,
                                          sabl_resident_inx_id, sabl_non_resident_inx_id)
    values (:new_sabl_sab_id, :new_sabl_type, :new_sabl_resident_ab_id, :new_sabl_non_resident_ab_id,
            :new_sabl_resident_inx_id, :new_sabl_non_resident_inx_id)
    matching (sabl_sab_id, sabl_type)
    returning sabl_resident_ab_id, sabl_non_resident_ab_id, sabl_resident_inx_id, sabl_non_resident_inx_id
    into sabl_resident_ab_id, sabl_non_resident_ab_id, sabl_resident_inx_id, sabl_non_resident_inx_id;
  end
  suspend;
end
