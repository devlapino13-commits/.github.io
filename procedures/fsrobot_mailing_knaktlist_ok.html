CREATE PROCEDURE FSROBOT_MAILING_KNAKTLIST_OK(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable kal_new_kf_officer id_type;
declare variable kal_new_kf_name str_100;
declare variable kal_date date;
declare variable kal_prior_lo_id id_type;
declare variable k_id id_type;
declare variable kal_date_str str_100;
declare variable new_officer_fio_full str_100;
declare variable prior_officer_fio_full str_100;
declare variable kc_closed_ol_id id_type;
declare variable fname str_100;
declare variable lname str_100;
declare variable kk integer;
declare variable firstname str_100;
declare variable lastname str_100;
declare variable new_officer_fio_short str_100;
declare variable prior_officer_fio_short str_100;
declare variable c_cell_no str_100;
declare variable counter smallint;
declare variable cell_no str_20;
begin
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID  - пользователь

-- СМС при передачи группы
  if (user_fio is null) then
    user_fio = '';
  res_maillist = :rep_email;

  counter = 0;

  for select kal.kal_new_kf_officer, kal.kal_new_kf_name, kal.kal_date, kal.kal_prior_lo_id, kont.k_id
      from knaktlist kal
      inner join kontract kont on kal.kal_from_kf_id = kont.k_from_kf_id
      inner join kontstate ks on kont.k_id = ks.ks_from_k_id
      where kal.is_delete = 0 and
            kal.kal_from_ka_id = :rep_params and
            current_date between kal.kal_date and kal.kal_date_end and
            current_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_state = 1 -- только активные контракты
            and
            ks.ks_ol_is_delete = 0
      into :kal_new_kf_officer, :kal_new_kf_name, :kal_date, :kal_prior_lo_id, :k_id
  do
  begin
    if (kal_new_kf_officer <> kal_prior_lo_id) then
    begin
      if (counter = 0) then
      begin
        counter = 1;
        kal_date_str = right('0' || extract(day from kal_date), 2) || '.' || right('0' || extract(month from kal_date), 2) || '.' || extract(year from kal_date);

------- КС, которому передали группу ------------- >>
        select cl.c_name_fio
        from clients cl
        where cl.c_id = :kal_new_kf_officer
        into :new_officer_fio_full;

        cell_no = null;
        select cont_value
        from hp_contdata_get_cell_main(0, :kal_new_kf_officer)
        into :cell_no;

        fname = '';
        lname = '';
        kk = position(' ', :new_officer_fio_full) - 1;
        if (kk > 0) then
        begin
          lname = substring(:new_officer_fio_full from 1 for kk);
          new_officer_fio_full = substring(:new_officer_fio_full from kk + 2 for 255);
          kk = position(' ', new_officer_fio_full) - 1;
          if (kk > 0) then
          begin
            fname = substring(:new_officer_fio_full from 1 for kk);
            new_officer_fio_full = substring(:new_officer_fio_full from kk + 2 for 255);
          end
          else
            fname = :new_officer_fio_full;
        end
        firstname = trim(fname);
        lastname = trim(lname);

--if (character_length(firstname)>1) then firstname = substring(firstname from 1 for 1)||'.';
        new_officer_fio_short = coalesce(lastname || ' ' || firstname, '');
--------- <<

------- КС, от которого передали группу  ------------- >>
        select cl.c_name_fio
        from clients cl
        where cl.c_id = :kal_prior_lo_id
        into :prior_officer_fio_full;

        fname = '';
        lname = '';
        kk = position(' ', :prior_officer_fio_full) - 1;
        if (kk > 0) then
        begin
          lname = substring(:prior_officer_fio_full from 1 for kk);
          prior_officer_fio_full = substring(:prior_officer_fio_full from kk + 2 for 255);
          kk = position(' ', prior_officer_fio_full) - 1;
          if (kk > 0) then
          begin
            fname = substring(:prior_officer_fio_full from 1 for kk);
            prior_officer_fio_full = substring(:prior_officer_fio_full from kk + 2 for 255);
          end
          else
            fname = :prior_officer_fio_full;
        end
        firstname = trim(fname);
        lastname = trim(lname);

--if (character_length(firstname)>1) then firstname = substring(firstname from 1 for 1)||'.';
        prior_officer_fio_short = coalesce(lastname || ' ' || firstname, '');
--------  <<

-- Первый СМС, о транзакции
        res_body = '';
        res_subject = 'Передача группы другому КС';
        res_body = 'C' || kal_date_str || 'г.Кредитный Служащий ' || prior_officer_fio_short || ' не работает.Ваш новый Кред.Служ. ' || new_officer_fio_short || ',' || cell_no;
      end

-- вытаскиваю всех участников группы
      for select c_id,
                 (select cont_value
                  from hp_contdata_get_cell_main(0, cl.c_id)), iif(kncl.kc_closed_ol_id is null, null,
                 (select ol.ol_id
                  from operlist ol
                  where ol.ol_id = kncl.kc_closed_ol_id and
                        ol.is_delete = 0))
          from knclient kncl, clients cl
          where kncl.kc_from_k_id = :k_id and
                cl.c_id = kncl.kc_from_c_id and
                kncl.kc_amount_approved > 0
          into :res_c_id, :c_cell_no, :kc_closed_ol_id
      do
      begin
        if (kc_closed_ol_id is null and
            c_cell_no <> '') then
        begin
          res_maillist = :c_cell_no;
          suspend;
        end
      end
    end
  end
end
