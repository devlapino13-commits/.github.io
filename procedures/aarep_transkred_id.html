CREATE PROCEDURE AAREP_TRANSKRED_ID(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    TRANS_KONTRACT_ID ID_TYPE,
    TRANS_OPERLIST_ID ID_TYPE,
    TRANS_DATE DATE,
    TRANS_TYPE INTEGER,
    TRANS_PAY_TYPE INTEGER,
    TRANS_DISB_OS CURR_TYPE,
    TRANS_OS CURR_TYPE,
    TRANS_PERC CURR_TYPE,
    TRANS_FINE CURR_TYPE,
    TRANS_KOMMIS CURR_TYPE,
    TRANS_KOMMIS_ADV CURR_TYPE,
    TRANS_REGION_ID INTEGER,
    TRANS_DOC_NO INTEGER,
    TRANS_OTHER_CLIENT VARCHAR(255),
    TRANS_DATETIME TIMESTAMP,
    TRANS_OTHER_DAY VARCHAR(20),
    TRANS_TYPE_TEXT VARCHAR(50),
    TRANS_PAY_TYPE_TEXT VARCHAR(20),
    TRANS_PAY_ALL CURR_TYPE,
    TRANS_REFDATA_ID INTEGER,
    TRANS_KF_ID DECIMAL(18,0),
    TRANS_PLAN_DATE DATE)
AS
declare variable ol_id_prior id_type;
declare variable cr_rate curr_type;
declare variable def_curr varchar(3);
declare variable ol_date_prior date;
declare variable k_index_curr_code varchar(3);
declare variable k_id_prior id_type;
begin
  select cm_default_currency
  from company
  into def_curr;

  if (:report_currency is null) then
    report_currency = '000';

  ol_id_prior = 0;
  ol_date_prior = '01.01.1900';
  k_id_prior = 0;

  for select id_list
      from region_list(:report_region_id, 1, -1, 1)
      into trans_region_id
  do
  begin

    for select ok_from_k_id, ok_from_ol_id, ok_ol_date, ok_type, ok_pay_type, sum(iif(ok_type = 8, ok_os, 0)),
               sum(iif(ok_type <> 8, ok_os, 0)), sum(ok_perc), sum(ok_fine), sum(ok_kommis), sum(ok_kommis_adv), ok_sh_date
        from operkred
        where ok_ol_is_active = 1 and
              ok_ol_date between :report_date_begin and :report_date_end and
              ok_ol_reg_id = :trans_region_id and
              ok_type > 3 and
              ok_type <> 7 and
              ok_from_c_id = 0
        group by ok_from_k_id, ok_from_ol_id, ok_ol_date, ok_type, ok_pay_type, ok_sh_date
        order by ok_ol_date, ok_from_ol_id
        into trans_kontract_id, trans_operlist_id, trans_date, trans_type, trans_pay_type, trans_disb_os, trans_os,
             trans_perc, trans_fine, trans_kommis, trans_kommis_adv, trans_plan_date
    do
    begin
      if (report_currency = '000') then
        if (k_id_prior <> trans_kontract_id) then
        begin
          k_id_prior = trans_kontract_id;

          select k_index_curr_code, k_from_kf_id
          from kontract
          where k_id = :trans_kontract_id
          into k_index_curr_code, trans_kf_id;
        end

      if (ol_id_prior <> trans_operlist_id) then
      begin
        ol_id_prior = trans_operlist_id;
        select ol_doc_no, ol_doc_client_name, ol_datetime, ol_from_rd_id
        from operlist
        where ol_id = :trans_operlist_id
        into trans_doc_no, trans_other_client, trans_datetime, trans_refdata_id;

        if (report_currency = '000' and
            ol_date_prior <> trans_date and
            def_curr <> k_index_curr_code) then
        begin
          ol_date_prior = trans_date;

          select cr_rate
          from opercurr
          where cr_from_cu_id = :k_index_curr_code and
                :trans_date between cr_date and cr_date_end
          into cr_rate;
        end
      end

      if (def_curr <> k_index_curr_code) then
      begin
        trans_disb_os = cr_rate * trans_disb_os;
        trans_os = cr_rate * trans_os;
        trans_perc = cr_rate * trans_perc;
        trans_fine = cr_rate * trans_fine;
        trans_kommis = cr_rate * trans_kommis;
        trans_kommis_adv = cr_rate * trans_kommis_adv;
      end

      if (cast(:trans_datetime as date) <> trans_date) then
        trans_other_day = 'Другой день';
      else
        trans_other_day = '';

      if (trans_other_client is null) then
        trans_other_client = '';

      if (trans_type = 4) then
        trans_type_text = 'Погашение';
      if (trans_type = 5) then
        trans_type_text = 'Списание';
      if (trans_type = 6) then
        trans_type_text = 'Недоплата (списан.)';
      if (trans_type = 7) then
        trans_type_text = 'Начисление';
      if (trans_type = 8) then
        trans_type_text = 'Выдача';
      if (trans_type = 9) then
        trans_type_text = 'Переплата';

      trans_pay_type_text = '';
      if (trans_pay_type = 1) then
        trans_pay_type_text = 'Наличная';
      if (trans_pay_type = 2) then
        trans_pay_type_text = 'Безналичная';
      if (trans_pay_type = 3) then
        trans_pay_type_text = 'Внутренняя';

      trans_pay_all = trans_disb_os + trans_os + trans_perc + trans_fine + trans_kommis + trans_kommis_adv;

      suspend;
    end
  end
end
