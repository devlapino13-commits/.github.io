CREATE PROCEDURE UXP$UXFRBUTTON_SET_IMAGE(
    UFI_UUID D_UUID,
    IMG_UUID_CODE D_STR_CODE,
    UFI_HIDE_MENU D_BOOL = null)
AS
begin
  if (coalesce(img_uuid_code, '') = '') then
  begin
    update ux$frbutton
    set ufb_img_uuid = null,
        ufb_hide_menu = coalesce(:ufi_hide_menu, 0)
    where ufb_ufi_uuid = :ufi_uuid;

    delete from ux$frbutton
    where ufb_ufi_uuid = :ufi_uuid and
          ufb_ua_uuid is null and
          ufb_img_uuid is null;
  end else
  begin
    update or insert into ux$frbutton (ufb_ufi_uuid, ufb_img_uuid, ufb_hide_menu)
    values (:ufi_uuid, char_to_uuid(:img_uuid_code), :ufi_hide_menu)
    matching (ufb_ufi_uuid);
  end
end
