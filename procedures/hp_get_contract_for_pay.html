CREATE PROCEDURE HP_GET_CONTRACT_FOR_PAY(
    C_ID D_ID)
RETURNS (
    K_ID D_ID)
AS
declare variable current_k_id d_id;
declare variable current_k_fact_disb_date d_date;
declare variable current_pr_code d_str_100;
declare variable temp_k_id d_id;
declare variable temp_k_fact_disb_date d_date;
declare variable temp_pr_code d_str_100;
begin
  -- Возвращает id контракта с самой ранней датой выдачи из группы активных контрактов клиента с одним и тем же продуктом (RFCRU-2129)
  -- Если у клиента нет нескольких активных займов с одним и тем же продуктом, то процедура возвращает 0.
  -- Возвращаемый k_id используется для показа на сайте, терминале, КМП и т.д. в процессе внесения погашения займов.

  k_id = 0;
  temp_k_id = 0;

  -- берем все активные контракты клиента, отсортировав в группы по продуктам
  -- каждая группа может состоять из одного или более контрактов
  for select k_id,
             pr_code,
             k_fact_disb_date
      from kontract
      inner join product on pr_id = k_from_pr_id
      inner join knclient on kc_from_k_id = k_id
      inner join kontstate on ks_from_k_id = k_id
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            current_date between ks_date_begin and ks_date_end
      order by pr_code, k_fact_disb_date desc
      into current_k_id,
           current_pr_code,
           current_k_fact_disb_date
  do
  begin
    -- условие ниже обеспечит пропуск контрактов, у которых продукт уникальный в рамках активных контрактов клиента
    -- условие также обеспечит пропуск первого контракта каждой группы, т.к. temp_pr_code будет равна null (1-я итерация) или будет отличаться от продукта текущего контракта
    if (temp_pr_code = current_pr_code) then
    begin
      -- если у текущего контракта дата выдачи раньше, чем у сохраненного, то сохраняем его
      -- если сохраненная дата выдачи равна null, то сохраненного контракта нет, а текущий контракт - второй в общей выборке, сохраняем его
      if (current_k_fact_disb_date < temp_k_fact_disb_date or temp_k_fact_disb_date is null) then
      begin
        temp_k_id = current_k_id;
        temp_k_fact_disb_date = current_k_fact_disb_date;
      end
    end
    temp_pr_code = current_pr_code; -- сохраняем продукт текущего контракта
  end

  if (temp_k_id > 0) then
    k_id = temp_k_id; -- контракт с самой ранней датой выдачи из группы контрактов с одним и тем же продуктом

  suspend;
end
