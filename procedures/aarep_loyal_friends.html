CREATE PROCEDURE AAREP_LOYAL_FRIENDS(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    OFFICE STR_100,
    LO_FIO STR_100,
    PARTNER_FIO STR_100,
    DATE_START D_DATE,
    FRIENDS_COUNT D_INTEGER,
    FR_DISB D_INTEGER,
    POINTS_COUNT CURR_TYPE,
    POINTS_ALL CURR_TYPE,
    POINT_STATUS STR_20,
    BONUS_GROUP D_INTEGER)
AS
declare variable lo_id id_type;
declare variable c_id id_type;
declare variable ag_id id_type;
begin
  for select c_id, c_name_fio, reg_code || ' ' || reg_name
      from hr_pstatus
      inner join joblist on ps_from_jl_id = jl_id
      inner join jobkind on jl_kind = jk_id
      inner join clients on c_id = ps_from_c_id
      inner join region_list(:rep_region_id, 1, -1, 1) on id_list = ps_from_reg_id
      inner join region on reg_unicode = id_list
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            ps_status = 1
      into lo_id, lo_fio, office
  do
  begin
    for select distinct kc_from_c_id, c_name_fio, ag_id, ag_date_begin
        from kontstate, knclient, knaktlist, kontract, agreements, clients
        where ks_ol_is_delete = 0
        and ks_state = 1
        and :rep_date_end >= ks_date_begin
        and dateadd(month, -24, :rep_date_end) <= ks_date_end
        and ks_from_k_id = kc_from_k_id
        and kc_amount_approved > 0
        and ks_from_k_id = k_id
        and k_from_kf_id = kal_from_kf_id
        and knaktlist.is_delete = 0
        and :rep_date_end between kal_date and kal_date_end
        and ag_object_kind = 0
        and ag_object_id = kc_from_c_id
        and kal_new_kf_officer = :lo_id
        and c_id = kc_from_c_id
        and ag_status = 1
        into c_id, partner_fio, ag_id, date_start
    do
    begin
      points_count = 0;
      points_all = 0;
      point_status = 'Копить';

      bonus_group = null;
      select ppg_bonus
      from partner_program_group
      where ppg_from_c_id = :c_id
      into bonus_group;

      if (exists(select *
                 from agreement_exchange
                 where ae_from_ag_id = :ag_id and
                       ae_is_exchange = 1)) then
        point_status = 'Обменять';

      select sum(ogr_amount * ogr_sign * f_loyal(ogr_from_ol_id, :rep_date_end) + ogr_tax_fl * ogr_sign * f_loyal(ogr_from_ol_id, :rep_date_end))
      from operagr
      where ogr_from_ag_id = :ag_id and
            ogr_ol_is_active = 1
      into points_count;

      select sum(ogr_amount + ogr_tax_fl)
      from operagr
      where ogr_from_ag_id = :ag_id and
            ogr_ol_is_active = 1 and
            ogr_sign = -1
      into points_all;

      for select count(clk_link_cell_no), sum(iif(clk_status = 2, 1, 0))
          from clientlink
          where clk_from_c_id = :c_id
          into friends_count, fr_disb
      do
      begin
        suspend;
      end
    end
  end
end
