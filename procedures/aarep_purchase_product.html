CREATE PROCEDURE AAREP_PURCHASE_PRODUCT(
    OL_ID ID_TYPE)
RETURNS (
    PROD_CODE D_STR_10,
    PROD_NAME D_STR_255,
    PROD_MODEL D_STR_255,
    PROD_COUNT CURR_TYPE,
    PROD_COST CURR_TYPE,
    PROD_TOTAL CURR_TYPE,
    PROD_UNIT D_STR_20,
    KONT_NO D_STR_20)
AS
begin
  for
    select
      feg_code,
      ml_name,
      ml_model,
      kml_count,
      kml_cost,
      kml_total,
      yl_code,
      k_kontract_no
    from kontmat
      left join foreign_econom_goods_ref
        on feg_id = kml_from_feg_id
      left join matlist
        on ml_id = kml_from_ml_id
      left join yedizlist
        on yl_id = ml_from_yl_id
      inner join kontract
        on k_id = kml_object_id
    where kml_bill_id = iif(:ol_id is null, -1, :ol_id)
    order by kml_bill_line_no
  into
      :prod_code,
      :prod_name,
      :prod_model,
      :prod_count,
      :prod_cost,
      :prod_total,
      :prod_unit,
      :kont_no
  do suspend;
end
