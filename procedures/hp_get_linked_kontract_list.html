CREATE PROCEDURE HP_GET_LINKED_KONTRACT_LIST(
    K_ID ID_TYPE,
    SELF_INCLUDE INTEGER = 1)
RETURNS (
    RES_K_ID ID_TYPE)
AS
declare variable base_k_id id_type;
begin
-- возвращает список связанных договоров (до амнистирования были в одном составе)

  select k_base_k_id
  from kontract
  where k_id = :k_id
  into base_k_id;

  if (base_k_id is null) then
    exit;

  if (base_k_id <> k_id) then
  begin
    res_k_id = base_k_id;
    suspend;
  end

  if (self_include = 1) then
  begin
    res_k_id = k_id;
    suspend;
  end

  for select kd_k_new
      from kontdevides
      where kd_k_orig = :base_k_id and
            kd_k_new <> :k_id and
            kd_k_new <> :base_k_id
      into res_k_id
  do
    suspend;

end
