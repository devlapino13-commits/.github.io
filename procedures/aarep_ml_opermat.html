CREATE PROCEDURE AAREP_ML_OPERMAT(
    BILL_ID DECIMAL(18,0),
    BILL_LINE_NO INTEGER)
RETURNS (
    AMO_DATE DATE,
    AMO_NUMBER INTEGER,
    AMO_OPER_TYPE_NAME VARCHAR(50),
    AMO_WP_NAME VARCHAR(100),
    AMO_WP_STATUS_NAME VARCHAR(50),
    AMO_MO_NAME VARCHAR(100),
    AMO_WP_NAME_INFO VARCHAR(100),
    AMO_WP_STATUS_NAME_INFO VARCHAR(50),
    AMO_MO_NAME_INFO VARCHAR(100),
    AMO_KOLICH_PLUS DECIMAL(18,4),
    AMO_SUMMA_PLUS DECIMAL(18,4),
    AMO_KOLICH_MINUS DECIMAL(18,4),
    AMO_SUMMA_MINUS DECIMAL(18,4),
    AMO_MEMO VARCHAR(1024),
    AMO_TEXT VARCHAR(30))
AS
declare variable amo_oper_type integer;
declare variable amo_wp_status integer;
declare variable amo_wp_status_info integer;
declare variable om_workplace_info decimal(18,0);
declare variable om_matotvet_info decimal(18,0);
declare variable ol_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ol_id, ol_date, om_number, om_oper_type, w1.wp_name, w1.wp_status, c1.c_name_fio, om_kolich_plus,
             om_summa_plus, om_kolich_minus, om_summa_minus, ol_memo, om_workplace_info, om_matotvet_info
      from operlist, opermat, workplace w1, clients c1
      where ol_id = om_from_ol_id and
            operlist.is_delete = 0 and
            om_bill_id = :bill_id and
            om_bill_line_no = :bill_line_no and
            om_workplace_id = w1.wp_id and
            om_matotvet_id = c1.c_id and
            (om_kolich_plus > 0 or
            om_summa_plus > 0 or
            om_matotvet_info is null)
      order by ol_date, ol_id, om_number
      into :ol_id, :amo_date, :amo_number, :amo_oper_type, :amo_wp_name, :amo_wp_status, :amo_mo_name, :amo_kolich_plus,
           :amo_summa_plus, :amo_kolich_minus, :amo_summa_minus, :amo_memo, :om_workplace_info, :om_matotvet_info
  do
  begin
    amo_wp_name_info = '';
    amo_mo_name_info = '';
    amo_wp_status_info = null;
    if (om_workplace_info is not null) then
    begin
      select wp_name, wp_status
      from workplace
      where wp_id = :om_workplace_info
      into :amo_wp_name_info, :amo_wp_status_info;
    end
    if (om_matotvet_info is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :om_matotvet_info
      into :amo_mo_name_info;
    end
    amo_oper_type_name = '';
    amo_wp_status_name = '';
    amo_wp_status_name_info = '';
    if (amo_oper_type = 1) then
      amo_oper_type_name = 'Приобретение';
    if (amo_oper_type = 2) then
      amo_oper_type_name = 'Приход от произв.';
    if (amo_oper_type = 3) then
      amo_oper_type_name = 'Перемещение';
    if (amo_oper_type = 4) then
      amo_oper_type_name = 'Внутренн.списание';
    if (amo_oper_type = 5) then
      amo_oper_type_name = 'Реализация, передача';
    if (amo_oper_type = 6) then
      amo_oper_type_name = 'Модернизация';
    if (amo_oper_type = 7) then
      amo_oper_type_name = 'Декомплектация';
    if (amo_oper_type = 8) then
      amo_oper_type_name = 'Комплектация';
    if (amo_oper_type = 9) then
      amo_oper_type_name = 'Переоценка';
    if (amo_oper_type = 10) then
      amo_oper_type_name = 'Переоц.за счет износа';

    if (amo_wp_status = 5) then
      amo_wp_status_name = 'Рабочие места';
    if (amo_wp_status = 6) then
      amo_wp_status_name = 'Склад';
    if (amo_wp_status = 7) then
      amo_wp_status_name = 'Отдел';
    if (amo_wp_status = 8) then
      amo_wp_status_name = '';

    if (amo_wp_status_info = 5) then
      amo_wp_status_name_info = 'Рабочие места';
    if (amo_wp_status_info = 6) then
      amo_wp_status_name_info = 'Склад';
    if (amo_wp_status_info = 7) then
      amo_wp_status_name_info = 'Отдел';
    if (amo_wp_status_info = 8) then
      amo_wp_status_name_info = '';

    amo_text = '';
    if ((amo_mo_name_info <> '') and
        (amo_mo_name_info is not null)) then
    begin
      if ((amo_wp_name <> amo_wp_name_info) or (amo_mo_name <> amo_mo_name_info)) then
      begin
        if (amo_summa_plus > 0) then
          amo_text = 'От кого';
        else
          amo_text = 'Кому';
      end
    end
    amo_kolich_plus = :amo_kolich_plus - :amo_kolich_minus;
    amo_summa_plus = :amo_summa_plus - :amo_summa_minus;
    amo_kolich_minus = 0;
    amo_summa_minus = 0;
    suspend;
  end

end
