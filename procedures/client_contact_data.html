CREATE PROCEDURE CLIENT_CONTACT_DATA(
    C_ID ID_TYPE)
RETURNS (
    CONTACT_DATA BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable cc_total_res_id integer;
declare variable cc_client_ok_str str_100;
declare variable cc_name_ok_str str_100;
declare variable cc_status_ok_str str_100;
declare variable cc_address_ok_str str_100;
declare variable cc_checked_date_str str_100;
declare variable cc_name str_100;
declare variable cc_status str_100;
declare variable cc_type str_100;
declare variable cc_address str_255;
declare variable cc_cell_no str_255;
declare variable cc_total_res str_100;
declare variable cc_checked_date timestamp;
declare variable cc_try_kol integer;
declare variable cc_create_date timestamp;
declare variable cc_client_ok smallint;
declare variable cc_name_ok smallint;
declare variable cc_status_ok smallint;
declare variable cc_address_ok smallint;
declare variable cc_check_memo str_255;
declare variable cc_create_date_str str_100;
declare variable cc_from_adl_id id_type;
declare variable db_id integer;
begin

  :db_id = f_get_db_id();

  contact_data = '';
  for select cont.cc_name, coalesce(cont.cc_status, 'Не указан'), -- кем пиходится
             trim(iif(cont.cc_type = 1, 'Родина', trim(iif(:db_id = 81001, 'РФ', 'Местный')))),
             cont.cc_from_adl_id, cont.cc_cell_no,
             (
             case cont.cc_total_res
               when 0 then 'Еще не проверифицирован'
               when 1 then 'Информация подтвердена'
               when -1 then 'Не дозвонились'
               when -2 then 'Инофрмация не подтверждена'
             end),
             cc_total_res as cc_total_res_id, cont.cc_checked_datetime, cont.cc_try_kol, cont.cc_create_date,
             cont.cc_client_ok, cont.cc_name_ok, cont.cc_status_ok, cont.cc_address_ok,
             coalesce(cont.cc_check_memo, '')
      from contacts cont
      where cont.cc_is_old_contact = 0 and
            cont.cc_from_c_id = :c_id
      into :cc_name, :cc_status, :cc_type, :cc_from_adl_id, :cc_cell_no, :cc_total_res, :cc_total_res_id,
           :cc_checked_date, :cc_try_kol, :cc_create_date, :cc_client_ok, :cc_name_ok, :cc_status_ok, :cc_address_ok,
           :cc_check_memo
  do
  begin

    cc_client_ok_str = 'Не проверялось';
    cc_name_ok_str = 'Не проверялось';
    cc_status_ok_str = 'Не проверялось';
    cc_address_ok_str = 'Не проверялось';

    if (cc_total_res_id in (1, -2)) then
    begin
      cc_client_ok_str = iif(cc_client_ok = 0, 'Не подтверждено', 'Подтверждено');
      cc_name_ok_str = iif(cc_name_ok = 0, 'Не подтверждено', 'Подтверждено');
      cc_status_ok_str = iif(cc_status_ok = 0, 'Не подтверждено', 'Подтверждено');
      cc_address_ok_str = iif(cc_address_ok = 0, 'Не подтверждено', 'Подтверждено');
    end

    if (cc_checked_date is null) then
      cc_checked_date_str = 'Отсутствует';
    else
    begin
      cc_checked_date_str = replace(iif(extract(day from cc_checked_date) < 10, '0', '') || extract(day from cc_checked_date) || '.' || iif(extract(month from cc_checked_date) < 10, '0', '') || extract(month from cc_checked_date) || '.' || extract(year from cc_checked_date), ' ', '') || ' ' || substring(cast(cc_checked_date as str_100) from 12 for 5);
    end

    if (cc_create_date is null) then
      cc_create_date_str = 'Отсутствует';
    else
    begin
      cc_create_date_str = replace(iif(extract(day from cc_create_date) < 10, '0', '') || extract(day from cc_create_date) || '.' || iif(extract(month from cc_create_date) < 10, '0', '') || extract(month from cc_create_date) || '.' || extract(year from cc_create_date), ' ', '') || ' ' || substring(cast(cc_create_date as str_100) from 12 for 5);
    end

    cc_address = null;

    select adl.adl_address_str
    from addrlist adl
    where adl.adl_id = :cc_from_adl_id
    into :cc_address;

    contact_data = BLOB_APPEND(contact_data, 'Контактное лицо:       <b>' || coalesce(cc_name, 'Не указано') || '</b><br>' || 'Местонахождение:       <b>' || cc_type || '</b><br>' || 'Кем приходится клиенту:<b>' || cc_status || '</b><br>' || 'Номер телефона:        <b>' || coalesce(cc_cell_no, 'Не указан') || '</b><br>' || 'Адрес проживания:      <b>' || coalesce(cc_address, 'Не указан') || '</b><br>' || 'Результат обзвона:     <b>' || cc_total_res || '</b><br>' || '  Телефонный номер:    <b>' || cc_client_ok_str || '</b><br>' || '  ФИО контактного лица:<b>' || cc_name_ok_str || '</b><br>' || '  Кем приходится:      <b>' || cc_status_ok_str || '</b><br>' || '  Адрес:               <b>' || cc_address_ok_str || '</b><br>' || 'Дата/время проверки:   <b>' || cc_checked_date_str || '</b><br>' || 'Дата/время создания:   <b>' || cc_create_date_str || '</b><br>' || 'Примечание:            <b>' || coalesce(cc_check_memo, 'Не указано') || '</b><br>' || '</b><br>');
  end
  suspend;
end
