CREATE PROCEDURE AAREP_FP_OFFICE_NOTICE(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(50),
    K_DISB_DATE DATE,
    C_NAME_FIO STR_100,
    REFRESH_DATE DATE,
    REG_NAME STR_100)
AS
declare variable reg_id_old integer;
declare variable reg_id integer;
declare variable c_id id_type;
declare variable fpa_date date;
begin
  reg_id_old = 0;

-- беру всех клиентов, граждан ТЖ, которые получили заем 162 дня назад и более
  for select kal_new_reg_id, kont.k_kontract_no, ks.ks_date_begin, cl.c_name_fio, cl.c_id
      from kontract kont
      inner join kontstate ks on ks.ks_from_k_id = kont.k_id
      inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
      inner join clients cl on cl.c_id = kncl.kc_from_c_id
      inner join refdata rd on rd.rd_id = cl.c_citizen
      inner join knaktlist ka on ka.kal_from_kf_id = kont.k_from_kf_id
      inner join region_list(:rep_region_id, 1, -1, 1) rl on id_list = ka.kal_new_reg_id
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            (datediff(day, ks.ks_date_begin, :rep_date) = 162)

            and
            ka.is_delete = 0 and
            :rep_date between ka.kal_date and ka.kal_date_end

            and
            kncl.kc_amount_approved > 0 and
            rd.rd_code = 'Citizen.TJ'
      order by kal_new_reg_id, k_kontract_no, c_name_fio
      into :reg_id, :k_kontract_no, :k_disb_date, :c_name_fio, :c_id
  do
  begin
-- дата последнего обновления анкеты
    select fpa_date
    from fp_anketa_cl fpcl
    where fpcl.fpa_c_id = :c_id and
          fpcl.fpa_date <= :rep_date
    order by fpcl.fpa_date desc
    rows 1
    into :fpa_date;

    fpa_date = coalesce(fpa_date, cast('01.01.1900' as date));

    if (datediff(day, :fpa_date, :rep_date) >= 162) then -- если с момента последнего обновления анкеты прошло 162 дня
    begin
      refresh_date = dateadd(179 day to fpa_date); -- плановая дата обновления анкеты

      if (reg_id <> reg_id_old) then
      begin
        select reg.reg_name
        from region reg
        where reg.reg_unicode = :reg_id
        into :reg_name;

        reg_id_old = reg_id;
      end

      suspend;
    end
  end
end
