CREATE PROCEDURE HP_INSURANCE_COMPANY_ADD(
    IA_ID D_INTEGER,
    AG_ID D_INTEGER,
    PERCENT D_SMALLINT)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable edc_code d_str_code;
declare variable current_total d_smallint;
begin
    -- 1. базовая валидация входных данных
    if (percent not between 0 and 100) then
    begin
        res_text = 'Процент должен быть в диапазоне от 1 до 100.';
        suspend;
        exit;

    end

    -- 2. проверяем, сколько процентов уже распределено по этому договору
    -- используем coalesce на случай, если это первая компания для договора
    select coalesce(sum(ic_percent), 0)
    from insurance_company
    where ic_from_ia_id = :ia_id
    into :current_total;

    -- 3. проверяем, не будет ли превышен лимит в 100%
    if (current_total + percent > 100) then
    begin
        res_text = 'Ошибка! Суммарный процент по договору превысит 100%. доступно для распределения: ' || cast((100 - current_total) as varchar(3)) || '%';
        suspend;
        exit;

    end

    -- 4. проверяем, были ли добавлена уже данная страховая компания к распределению
    if (exists
        (
            select 1
            from insurance_company ic
            where ic.ic_from_ia_id = :ia_id
                and ic.ic_from_ag_id = :ag_id
        )
    ) then
    begin
        res_text = 'Ошибка! Выберите другую страховую компанию.';
        suspend;
        exit;

    end


    edc_code = (select edc_code from hp_insurance_company_combobox where ag_id = :ag_id);

    -- 5. вставка записи.
    insert into insurance_company (
        ic_from_ia_id,
        ic_from_ag_id,
        ic_edc_code,
        ic_percent
    ) values (
        :ia_id,
        :ag_id,
        :edc_code,
        :percent
    );

end
