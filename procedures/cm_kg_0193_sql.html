CREATE PROCEDURE CM_KG_0193_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable date_app d_date;
declare variable rd_id d_id;
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable obj_kind d_str_20;
declare variable k_no d_str_20;
declare variable c_fio d_str_100;
begin
  for select ma.ma_date_app, mac.mac_from_c_id
      from miniapp ma
      inner join miniappclient mac on ma.ma_id = mac.mac_from_ma_id and
            mac.mac_amount > 0
      inner join product pr on pr.pr_id = ma.ma_from_pr_id
      where ma.ma_is_active = 1 and
            ma.ma_id = :ma_id and
            ma.ma_date_end = '31.12.9999' and
            pr.pr_code not containing '.sugarmill.' and
            pr.pr_code not containing '.installment_goods.' and
            pr.pr_code not containing '.online_loan.'
      into :date_app, :c_id
  do
  begin
    rd_id = f_get_rd_id('Anketa.TargetUseLoan');

    for select kc.kc_id, k.k_kontract_no
        from knclient kc
        inner join kontract k on kc.kc_from_k_id = k.k_id
        inner join kontstate ks on k.k_id = ks.ks_from_k_id and
              ks.ks_state = 1 and
              ks.ks_ol_is_delete = 0
        where kc.kc_from_c_id = :c_id and
              kc.kc_amount_approved > 0 and
              :date_app between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_date_begin >= '01.01.2022'
        into :kc_id, :k_no
    do
    begin
      obj_kind = 'KnClient';

      c_fio = '';
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into :c_fio;

      if (exists(select *
                 from actionorder aco
                 where aco.aco_from_rd_id = :rd_id and
                       aco.aco_object_kind = f_objkind(:obj_kind, 'id') and
                       aco.aco_object_id = :kc_id and
                       aco.aco_status = 0 and
                       aco.aco_date <= :date_app)) then
      begin
        res_error_text = 'Имеется необработанный целевой мониторинг по клиенту: ' || :c_fio || ', Договор №' || :k_no;
        res_error_is_critical = 1;
        suspend;
      end
    end
  end
end
