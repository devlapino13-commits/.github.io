CREATE PROCEDURE MILLICARD_CREDIT_CHECK(
    ESM_ID ID_TYPE,
    ERROR_CODE STR_100,
    FULL_SOURCE D_BLOB_TEXT)
AS
declare variable ERROR_TEXT STR_255;
declare variable ESM_PROJECT varchar(20);
declare variable ESM_ACTION STR_100;
declare variable ESM_STATUS smallint;
declare variable KONTRACT_NO STR_100;
begin
  if (esm_id is null) then
    exit;

  select em.esm_project, em.esm_action, em.esm_status
  from extsysmain em
  where em.esm_id = :esm_id
  into :esm_project, :esm_action, :esm_status;

  if (esm_project not in ('millicard_disb', 'millicard_biz_disb')) then
    exception e_default 'Project not related with Millicard';

  if (esm_action <> 'checkCredit') then
    exception e_default 'Action is not an checkCredit';

  if (esm_status <> 0) then
    exception e_default 'ESM_STATUS <> 0';

  delete from extsysanswer ea
  where ea.esa_from_esm_id = :esm_id;

  error_code = trim(coalesce(error_code, ''));
  full_source = coalesce(full_source, '');
  error_text = '';
  if (error_code <> '') then
  begin
    select rd.rd_name
    from refdata rd
    where rd.rd_main_code = 'MillicardErrorsApi' and
          rd.rd_code = :error_code
    into :error_text;

    if (:error_code = 'E010002') then
    begin
        update const_var
        set cv_value = '1'
        where cv_code = 'mbf_empty_balance';
    end
    if (error_text = '') then
      error_text = error_code;

    insert into extsysanswer (esa_from_esm_id, esa_message)
    values (:esm_id, 'error: ' || :error_text);

    update extsysmain em
    set em.esm_status = -1,
        em.esm_full_source = :full_source
    where em.esm_id = :esm_id;
  end
  else
  begin
    update extsysmain em
    set em.esm_status = 1,
        em.esm_full_source = :full_source
    where em.esm_id = :esm_id;

    execute procedure millicard_create_operbook(:esm_id);
    when any do
    begin
      select kont.k_kontract_no
      from extsysmain em
      inner join kontract kont on kont.k_id = em.esm_from_k_id
      where em.esm_id = :esm_id
      into :kontract_no;

      execute procedure hp_send_quick_message('email',
          'finstart@mbfplanet.com, n.zeynalov@molbulak.com, s.morozova@mbulak.kg, e.axundov@molbulak.com, t.hajili@molbulak.com, g.gadirova@molbulak.com',
          'Не удалось сформировать бух. проводку при выдаче через Millicard',
          'Кредитный договор: ' || kontract_no, '', null, 1000000000000);
    end
  end
end
