CREATE PROCEDURE FM_LOAN_DAILY_REPAY(
    UN_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REG_ID INTEGER)
RETURNS (
    K_ID ID_TYPE,
    PLAN_DATE DATE,
    SUM_ZADOL CURR_TYPE,
    SUM_REPAY CURR_TYPE,
    GR_NAME VARCHAR(100))
AS
declare variable c_id id_type;
declare variable delinq_month_payed decimal(18,4) = 0;
declare variable delinq_payed decimal(18,4) = 0;
declare variable plan_payed decimal(18,4) = 0;
declare variable plan_zadol decimal(18,4) = 0;
declare variable advance_payed decimal(18,4) = 0;
declare variable advance_next_month_payed decimal(18,4) = 0;
declare variable count_delinq_payed integer;
declare variable count_plan_payed integer;
declare variable count_plan_zadol integer;
declare variable count_advance_payed integer;
declare variable count_advance_next_month_payed integer;
declare variable rep_month_end date;
declare variable count_delinq_month_payed integer;
declare variable rep_month_begin date;
declare variable k_disb_date date;
declare variable tax_perc decimal(18,7);
declare variable cm_database_id integer;
declare variable c_name_fio varchar(100);
begin
  :cm_database_id = f_get_db_id();

  rep_month_end = dateadd(month, 1, rep_date_end - extract(day from rep_date_end) + 1) - 1;
  rep_month_begin = rep_date_begin - extract(day from rep_date_begin) + 1;

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  if (c_id is null) then
    exit;

  if (not exists(select ps_from_reg_id
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :c_id and
                       ps_date_begin <= current_date and
                       ps_date_end >= current_date and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code in ('JobSystem.Region_KM')
                       and
                       ps_is_working = 1)) then
    reg_id = 0;

  for select res_c_id
      from hp_staff_changer_list(:un_id, 1)
      where :reg_id = 0
      union
      select res_c_id
      from hp_staff_changer_list(:un_id, -1)
      where :reg_id <> 0
      into :c_id
  do
  begin
    delinq_month_payed = 0;
    delinq_payed = 0;
    plan_payed = 0;
    plan_zadol = 0;
    advance_payed = 0;
    advance_next_month_payed = 0;

    count_delinq_month_payed = 0;
    count_delinq_payed = 0;
    count_plan_payed = 0;
    count_plan_zadol = 0;
    count_advance_payed = 0;
    count_advance_next_month_payed = 0;

    if (reg_id = 0) then
    begin
      for with tabl1 (k_id, plan_date, sum_repay)
          as (select oks_from_k_id, oks_delinq_date, 0
              from oksummary
              where oks_date_begin + 0 <= :rep_date_end and
                    oks_date_end >= :rep_date_begin and
                    oks_kc_id = :c_id and
                    oks_close_state = 0 and
                    oks_delinq_date between :rep_date_begin and :rep_date_end
              union
              select ok_from_k_id, min(ok_sh_date), -sum(ok_os + ok_perc + ok_fine)
              from knaktlist, kontract, operkred
              where kal_new_kf_officer = :c_id and
                    knaktlist.is_delete = 0 and
                    :rep_date_end between kal_date and kal_date_end and
                    kal_from_kf_id = k_from_kf_id and
                    k_id = ok_from_k_id and
                    ok_ol_is_active = 1 and
                    ok_type = 4 and
                    ok_os + ok_perc + ok_fine < 0 and
                    ok_ol_date between :rep_date_begin and :rep_date_end
              group by ok_from_k_id)

          select tabl1.k_id, k_disb_date, min(plan_date), sum(sum_repay)
          from tabl1, kontract
          where tabl1.k_id = kontract.k_id
          group by k_id, k_disb_date
          into k_id, k_disb_date, plan_date, sum_repay
      do
      begin
        sum_zadol = null;
        tax_perc = 0;
        select nsp_percent_for_int
        from hp_get_kontract_nsp_acc(0, :k_disb_date)
        into :tax_perc;

        select sum(zad_os + maxvalue(sh_calc_perc, zad_perc) + sh_calc_fine + round((maxvalue(sh_calc_perc, zad_perc) * :tax_perc) / 100.00, 2))
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        where sh_date <= :rep_date_end
        into sum_zadol;


        if (sum_zadol is null) then
          sum_zadol = 0;
        if (sum_zadol < 0) then
          sum_zadol = 0;

        if (plan_date < :rep_date_begin and
            sum_repay <> 0) then
        begin
          if (plan_date < rep_month_begin) then
          begin
            delinq_month_payed = delinq_month_payed + sum_repay;
            count_delinq_month_payed = count_delinq_month_payed + 1;
          end
          else
          begin
            delinq_payed = delinq_payed + sum_repay;
            count_delinq_payed = count_delinq_payed + 1;
          end
        end
        if (plan_date between :rep_date_begin and :rep_date_end and
            sum_zadol <> 0) then
        begin
          plan_zadol = plan_zadol + sum_zadol;
          count_plan_zadol = count_plan_zadol + 1;
        end
        if (plan_date between :rep_date_begin and :rep_date_end and
            sum_repay <> 0) then
        begin
          plan_payed = plan_payed + sum_repay;
          count_plan_payed = count_plan_payed + 1;
        end
        if (plan_date > :rep_date_end and
            sum_repay <> 0) then
        begin
          if (plan_date > rep_month_end) then
          begin
            advance_next_month_payed = advance_next_month_payed + sum_repay;
            count_advance_next_month_payed = count_advance_next_month_payed + 1;
          end
          else
          begin
            advance_payed = advance_payed + sum_repay;
            count_advance_payed = count_advance_payed + 1;
          end
        end

        gr_name = '';
        select kal_new_kf_name, c_name_fio
        from kontract k, knaktlist, clients c
        where k_id = :k_id and
              k_from_kf_id = kal_from_kf_id and
              k.k_clients_cashier_id = c.c_id and
              knaktlist.is_delete = 0 and
              current_date between kal_date and kal_date_end
        into :gr_name, c_name_fio;
        
        
         if (cm_database_id in (39801, 41701) and gr_name is not null) then
        begin
          select short_fio
          from hp_fio_clear(:c_name_fio)
          into :c_name_fio;
          gr_name = gr_name || ', ' || c_name_fio;
        end

        suspend;
      end
    end
    else
    begin
      for with tabl1 (k_id, plan_date, sum_repay)
          as (select oks_from_k_id, oks_delinq_date, 0
              from oksummary
              inner join region_list(:reg_id, 1, 3, 1) reg on oks_region_id = reg.id_list
              where oks_date_begin + 0 <= :rep_date_end and
                    oks_date_end >= :rep_date_begin and
                    oks_close_state = 0 and
                    oks_delinq_date between :rep_date_begin and :rep_date_end
              union
              select ok_from_k_id, min(ok_sh_date), -sum(ok_os + ok_perc + ok_fine)
              from operkred
              inner join region_list(:reg_id, 1, 3, 1) reg on ok_ol_reg_id = reg.id_list
              where ok_ol_is_active = 1 and
                    ok_type = 4 and
                    ok_os + ok_perc + ok_fine < 0 and
                    ok_ol_date between :rep_date_begin and :rep_date_end
              group by ok_from_k_id)
          select tabl1.k_id, k_disb_date, min(plan_date), sum(sum_repay)
          from tabl1, kontract
          where tabl1.k_id = kontract.k_id
          group by k_id, k_disb_date
          into k_id, k_disb_date, plan_date, sum_repay
      do
      begin
        sum_zadol = null;

        tax_perc = 0;
        select nsp_percent_for_int
        from hp_get_kontract_nsp_acc(0, :k_disb_date)
        into :tax_perc;

        select sum(zad_os + maxvalue(sh_calc_perc, zad_perc) + sh_calc_fine + round((maxvalue(sh_calc_perc, zad_perc) * :tax_perc) / 100.00, 2))
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        where sh_date <= :rep_date_end
        into sum_zadol;

        if (sum_zadol is null) then
          sum_zadol = 0;
        if (sum_zadol < 0) then
          sum_zadol = 0;

        if (plan_date < :rep_date_begin and
            sum_repay <> 0) then
        begin
          if (plan_date < rep_month_begin) then
          begin
            delinq_month_payed = delinq_month_payed + sum_repay;
            count_delinq_month_payed = count_delinq_month_payed + 1;
          end
          else
          begin
            delinq_payed = delinq_payed + sum_repay;
            count_delinq_payed = count_delinq_payed + 1;
          end
        end
        if (plan_date between :rep_date_begin and :rep_date_end and
            sum_zadol <> 0) then
        begin
          plan_zadol = plan_zadol + sum_zadol;
          count_plan_zadol = count_plan_zadol + 1;
        end
        if (plan_date between :rep_date_begin and :rep_date_end and
            sum_repay <> 0) then
        begin
          plan_payed = plan_payed + sum_repay;
          count_plan_payed = count_plan_payed + 1;
        end
        if (plan_date > :rep_date_end and
            sum_repay <> 0) then
        begin
          if (plan_date > rep_month_end) then
          begin
            advance_next_month_payed = advance_next_month_payed + sum_repay;
            count_advance_next_month_payed = count_advance_next_month_payed + 1;
          end
          else
          begin
            advance_payed = advance_payed + sum_repay;
            count_advance_payed = count_advance_payed + 1;
          end
        end

        gr_name = '';
        select kal_new_kf_name
        from kontract, knaktlist
        where k_id = :k_id and
              k_from_kf_id = kal_from_kf_id and
              knaktlist.is_delete = 0 and
              current_date between kal_date and kal_date_end
        into :gr_name;

        suspend;
      end
    end

    k_id = 0;
    plan_date = null;

    sum_zadol = 0;
    sum_repay = 0;
    gr_name = f_format('1.%1', iif(cm_database_id <> 94401, 'Проср. пред. месяцев', 'Keçmiş aylar üzrə gecikmələr'));
    sum_repay = delinq_month_payed;
    k_id = count_delinq_month_payed;
    suspend;

    sum_zadol = 0;
    sum_repay = 0;
    gr_name = f_format('2.%1', iif(cm_database_id <> 94401, 'Проср. текущ. месяца', 'Cari ay üzrə gecikmələr'));
    sum_repay = delinq_payed;
    k_id = count_delinq_payed;
    suspend;

    sum_zadol = 0;
    sum_repay = 0;
    gr_name = f_format('3.%1', iif(cm_database_id <> 94401, 'Плановые за период', 'Planlaşdırılan müddət üzrə'));
    sum_repay = plan_payed;
    k_id = count_plan_payed;
    suspend;

    sum_zadol = 0;
    sum_repay = 0;
    gr_name = f_format('4.%1', iif(cm_database_id <> 94401, 'Плановые за период', 'Planlaşdırılan müddət üzrə'));
    sum_zadol = plan_zadol;
    k_id = count_plan_zadol;
    suspend;

    sum_zadol = 0;
    sum_repay = 0;
    gr_name = f_format('5.%1', iif(cm_database_id <> 94401, 'Досрочно за этот месяц', 'Vaxtından əvvəl cari ay üzrə'));
    sum_repay = advance_payed;
    k_id = count_advance_payed;
    suspend;

    sum_zadol = 0;
    sum_repay = 0;
    gr_name = f_format('6.%1', iif(cm_database_id <> 94401, 'Досрочно за будущие месяцы', 'Vaxtından əvvəl gələcək aylar üzrə'));
    sum_repay = advance_next_month_payed;
    k_id = count_advance_next_month_payed;
    suspend;
  end
end
