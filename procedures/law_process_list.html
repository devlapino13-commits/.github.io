CREATE PROCEDURE LAW_PROCESS_LIST(
    LAM_ID INTEGER)
RETURNS (
    LAD_FROM_LAF_ID INTEGER,
    LAF_NAME STR_255,
    IS_CHECKED SMALLINT,
    LAD_IS_DONE SMALLINT,
    LAD_FROM_LAM_ID INTEGER,
    LAD_DATE DATE,
    LAD_TIME TIME,
    LAD_MEMO STR_255,
    LAD_FROM_UN_ID ID_TYPE,
    LAD_DATETIME TIMESTAMP,
    LAD_OFFICER_ID ID_TYPE,
    LAD_OFFICER_FIO STR_100,
    LAF_ORDER_NO INTEGER,
    LAF_FIELD_KIND INTEGER,
    LAF_FIELD_VALUES STR_255)
AS
begin
  for select b.lad_from_laf_id, laf_name, b.is_checked, b.lad_is_done, ld2.lad_from_lam_id, ld2.lad_date, ld2.lad_time,
             ld2.lad_memo, ld2.lad_from_un_id, ld2.lad_datetime, ld2.lad_officer_id, cl.c_name_fio as lad_officer_fio,
             laf_order_no, laf_field_kind, laf_field_values
      from (select laf_id as lad_from_laf_id, laf_name, laf_order_no, sum(is_checked) as is_checked, laf_field_kind,
                   laf_field_values, sum(lad_is_done) as lad_is_done

            from (select 1 as is_checked, lf.laf_id, lf.laf_name, lf.laf_order_no, lf.laf_field_kind, laf_field_values,
                         ld.lad_is_done
                  from lawdata ld
                  inner join lawfields lf on lf.laf_id = ld.lad_from_laf_id and lf.laf_is_active = 1
                  where ld.lad_from_lam_id = :lam_id
                  union all
                  select 0 as is_checked, lf.laf_id, lf.laf_name, lf.laf_order_no, laf_field_kind, lf.laf_field_values,
                         0
                  from lawfields lf
                  where lf.laf_is_active = 1) a
            group by laf_id, laf_name, laf_order_no, laf_field_kind, laf_field_values) b
      left join lawdata ld2 on ld2.lad_from_laf_id = b.lad_from_laf_id and
            ld2.lad_from_lam_id = :lam_id
      left join clients cl on cl.c_id = ld2.lad_officer_id
      order by laf_order_no
      into :lad_from_laf_id, :laf_name, :is_checked, :lad_is_done, :lad_from_lam_id, :lad_date, :lad_time, :lad_memo,
           :lad_from_un_id, :lad_datetime, :lad_officer_id, :lad_officer_fio, :laf_order_no, :laf_field_kind,
           :laf_field_values
  do
  begin
    suspend;
  end
end
