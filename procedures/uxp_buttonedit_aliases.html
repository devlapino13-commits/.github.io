CREATE PROCEDURE UXP$BUTTONEDIT_ALIASES(
    UFP_UUID D_UUID)
RETURNS (
    ID D_STR_255,
    ID_VISIBLE D_STR_255,
    TEXT D_STR_255,
    ID_UUID D_UUID)
AS
declare variable field_name d_str_code;
declare variable source_name d_str_code;
declare variable tmp_uuid d_uuid;
declare variable tmp_code d_str_code;
begin
  select uf_field_name
  from ux$frparam, ux$field
  where ufp_uuid = :ufp_uuid and
        ufp_uf_uuid = uf_uuid
  into field_name;

  for select distinct uix_fk_source
  from uxp$uxfrparam_sources(:ufp_uuid)
       join ux$index on (source_name = uix_source and uix_fields = :field_name and uix_fk_source <> '')
  into :source_name
  do
  begin
    tmp_code = source_name;

    for select us_code, us_uuid
    from uxp$uxsource_views(:tmp_code)
    into :source_name, :tmp_uuid
    do
    begin
      for select distinct uxa_uuid, uxa_fcode, uxa_fcode, uxa_name || '  - ' || Upper(:source_name)
      from ux$form, ux$frame, ux$alias, ux$code, ux$search
      where ux_us_uuid = :tmp_uuid and
            ux_uuid = ufm_ux_uuid and
            ufr_owner = ufm_uuid and
            ufr_is_mainframe = 1 and
            ux_uuid = uxa_owner and
            uxa_uuid = usr_uuid
      union
      select uxa_uuid, uxa_fcode, uxa_fcode, uxa_name || '  - ' || Upper(:source_name)
      from ux$form, ux$frame, ux$alias, ux$code, ux$search
      where ufr_us_uuid = :tmp_uuid and
            ux_uuid = ufm_ux_uuid and
            ufr_owner = ufm_uuid and
            ufr_is_mainframe = 1 and
            ux_uuid = uxa_owner and
            uxa_uuid = usr_uuid and
            ufr_us_uuid <> ux_us_uuid
      union
      select uxa_uuid, uxa_fcode, uxa_fcode, uxa_name || '  - ' || Upper(:source_name)
      from ux$form, ux$frame, ux$alias, ux$code, ux$search
      where ufr_add_table_fields = :source_name and
            ux_uuid = ufm_ux_uuid and
            ufr_owner = ufm_uuid and
            ufr_is_mainframe = 1 and
            ux_uuid = uxa_owner and
            uxa_uuid = usr_uuid
      into id_uuid, id, id_visible, text
      do
        suspend;
    end
  end
end
