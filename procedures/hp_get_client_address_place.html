CREATE PROCEDURE HP_GET_CLIENT_ADDRESS_PLACE(
    C_ID DECIMAL(18,0),
    ADDR_TYPE INTEGER)
RETURNS (
    RES_ADDRESS_PLACE VARCHAR(255))
AS
declare variable pl_id decimal(18,0);
begin
  select adl_from_pl_id
  from addrlist, addrclient
  where adc_from_c_id = :c_id and
        adc_address_type = :addr_type and
        adc_from_adl_id = adl_id
  into :pl_id;

  if (pl_id is null) then
  begin
    res_address_place = 'Не указан';
  end
  else
  begin
    res_address_place = '';
    select iif(place_full_region <> '', place_full_region, place_full)
    from aarep_places_link(:pl_id)
    into res_address_place;
  end
  suspend;
end
