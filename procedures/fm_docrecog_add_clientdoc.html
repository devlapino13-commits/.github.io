CREATE PROCEDURE FM_DOCRECOG_ADD_CLIENTDOC(
    DRL_ID D_ID,
    UN_ID D_ID)
RETURNS (
    CDS_ID D_ID,
    ERR_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable pk_id d_integer;
declare variable is_main d_bool;
declare variable doc_no d_str_code;
declare variable date_begin d_date;
declare variable date_end d_date;
declare variable authority d_str_255;
declare variable authority_code d_str_255;
declare variable bls_1 d_id;
declare variable bls_2 d_id;
declare variable for_company d_str_255;
declare variable territory d_str_255;
declare variable memo_1 d_str_255;
declare variable memo_2 d_str_255;
declare variable memo_3 d_str_255;
declare variable memo_4 d_str_255;
declare variable memo_5 d_str_255;
declare variable lang d_str_lang;
begin
  cds_id = null;
  err_text = '';
  lang = f_get_lang_code(un_id, null);

  select drl_from_c_id, drl_from_pk_id, coalesce(drl_is_main_doc, 0), drl_from_bls_1, drl_from_bls_2
  from docrecoglist
  where drl_id = :drl_id
  into c_id, pk_id, is_main, bls_1, bls_2;

  select cds_id
  from clientdocs
  where cds_from_drl_id = :drl_id
  order by cds_id desc
  rows 1
  into cds_id;

  if (c_id is null) then
  begin
    err_text = f_lang(lang, 'Клиент не указан');
    suspend;
    exit;
  end

  if (f_is_valid_date(f_docrecog_value(drl_id, 'date_begin', null), 0) = 0) then
  begin
    err_text = f_lang(lang, 'Неверная дата выдачи');
    suspend;
    exit;
  end

  if (f_is_valid_date(f_docrecog_value(drl_id, 'date_end', null)) = 0) then
  begin
    err_text = f_lang(lang, 'Неверный срок действия документа');
    suspend;
    exit;
  end

  doc_no = f_docrecog_value(drl_id, 'doc_no', '');
  date_begin = f_docrecog_value(drl_id, 'date_begin', null);
  date_end = f_docrecog_value(drl_id, 'date_end', null);
  authority = f_docrecog_value(drl_id, 'authority', '');
  authority_code = f_docrecog_value(drl_id, 'authority_code', '');
  for_company = f_docrecog_value(drl_id, 'for_company', '');
  territory = f_docrecog_value(drl_id, 'territory', '');
  memo_1 = f_docrecog_value(drl_id, 'memo_1', '');
  memo_2 = f_docrecog_value(drl_id, 'memo_2', '');
  memo_3 = f_docrecog_value(drl_id, 'memo_3', '');
  memo_4 = f_docrecog_value(drl_id, 'memo_4', '');
  memo_5 = f_docrecog_value(drl_id, 'memo_5', '');

  if (extract(year from date_begin) < 1900) then
  begin
    err_text = f_lang(lang, 'Неверная дата выдачи');
    suspend;
    exit;
  end

  if (date_end is not null and date_end < current_date and
    not exists(
      select *
      from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_AttachExpiredDoc')
      where has_access = 1)) then
  begin
    err_text = f_lang(lang, 'Срок действия документа истек');
    suspend;
    exit;
  end

  if (cds_id is null) then
  begin
    select cds_id, err_text
    from fm_insert_cldoc(:c_id, :pk_id, :un_id, :is_main, :doc_no, :date_begin,
      :date_end, :authority, :authority_code, :for_company, :territory,
      :memo_1, :memo_2, :memo_3, :memo_4, :memo_5, :bls_1, :bls_2)
    into :cds_id, :err_text;

    if (err_text = '') then
    begin
      update clientdocs
      set cds_from_drl_id = :drl_id,
          cds_doc_blob_1 = :bls_1,
          cds_doc_blob_2 = :bls_2
      where cds_id = :cds_id;
    end
  end
  else
  if (cds_id is not null) then
  begin
    if (is_main = 1) then
    begin
      -- На случай, если is_main только что установлен
      update clientdocs
      set cds_is_main_doc = 1
      where cds_id = :cds_id
      and coalesce(cds_is_main_doc, 0) <> 1;

      update clients
      set c_doc_type = :pk_id,
          c_pass_no = :doc_no,
          c_pass_date = :date_begin,
          c_pass_date_expire = :date_end,
          c_pass_who = :authority,
          c_pass_who_code = :authority_code,
          c_blob_pass1_id = :bls_1,
          c_blob_pass2_id = :bls_2,
          c_from_un_id = :un_id
      where c_id = :c_id;
    end
    else
    begin
      update clientdocs
      set cds_doc_no = :doc_no,
          cds_doc_date_begin = :date_begin,
          cds_doc_date_end = :date_end,
          cds_doc_issued_by = :authority,
          cds_doc_issued_code = :authority_code,
          cds_doc_for_company = :for_company, 
          cds_doc_territory = :territory,
          cds_memo_1 = :memo_1,
          cds_memo_2 = :memo_2,
          cds_memo_3 = :memo_3,
          cds_memo_4 = :memo_4,
          cds_memo_5 = :memo_5,
          cds_doc_blob_1 = :bls_1,
          cds_doc_blob_2 = :bls_2,
          cds_verify_status = -3,
          cds_from_un_id = :un_id
      where cds_id = :cds_id;
    end
  end

  suspend;
end
