CREATE PROCEDURE AAREP_PRETRIAL_CLAIM(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    SLD_ID D_ID,
    SLD_OUTGOING_NUMBER D_STR_CODE,
    SLD_DATETIME D_DATETIME,
    SLD_UN_NAME D_STR_100,
    OFFICE_PARENT_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    K_FACT_DISB_DATE D_DATE,
    SLC_SUM D_CURRENCY,
    CLIENT_ID D_ID,
    CLIENT_FIO D_STR_100,
    SMR_DATE D_DATE,
    SMR_STATUS_STR D_STR_CODE,
    SMR_LAWYER_FIO D_STR_100,
    EDL_STATUS_NAME D_STR_CODE,
    EDL_BLANK_STATUS D_SMALLINT,
    SLD_EDL_ID D_ID,
    SLD_EDOC_ID D_ID,
    SLD_EDS_ID D_ID)
AS
declare variable smd_id d_id;
declare variable smr_from_k_id d_id;
declare variable smr_status d_integer;
declare variable smr_from_c_id d_id;
declare variable smr_from_reg_id d_integer;
declare variable slc_from_kc_id d_id;
declare variable sld_from_un_id d_id;
declare variable office_parent_id d_integer;
declare variable edl_status d_integer;
begin
  -- реестр досудебных претензий (RFCRU-4792)

  for select sld_id,
             sld_from_smd_id,
             sld_from_un_id,
             sld_datetime,
             sld_outgoing_number,
             edl_id,
             edl_status,
             edl_blank_status
      from sud_loandocs
      inner join sud_cfgdockind on scd_id = sld_doc_kind and scd_code = 'pretrial_claim'
      inner join edoc_list on edl_object_kind = f_objkind('sud_mainreq', 'id') and edl_object_id = sld_from_smd_id and edl_from_edc_id = f_edoc_kind('pretrial_claim', 'id')
      where cast(sld_datetime as date) between :rep_date_begin and :rep_date_end
      order by sld_datetime desc
      into sld_id,
           smd_id,
           sld_from_un_id,
           sld_datetime,
           sld_outgoing_number,
           sld_edl_id,
           edl_status,
           edl_blank_status
  do
  begin
    sld_edoc_id = sld_edl_id; -- edl_id для просмотра

    edl_status_name = case
                        when edl_status is null then 'не отправлен'
                        when edl_status = -1 then 'аннулирован'
                        else 'отправлен'
                      end;

    sld_un_name = null;
    select lower(un_name)
    from username
    where un_id = :sld_from_un_id
    into sld_un_name;

    smr_date = null;
    smr_status = null;
    smr_status_str = null;
    smr_from_reg_id = null;
    smr_from_k_id = null;
    smr_from_c_id = null;
    select smr_date,
           smr_status,
           smr_from_reg_id,
           smr_from_k_id,
           smr_from_c_id
    from sud_mainreq
    where smr_id = :smd_id
    into smr_date,
         smr_status,
         smr_from_reg_id,
         smr_from_k_id,
         smr_from_c_id;

    smr_status_str = case
                       when smr_status = 0 then 'добавлен'
                       when smr_status = 1 then 'в обработке'
                       when smr_status = 2 then 'одобрен'
                       when smr_status = -1 then 'доработка'
                       when smr_status = -2 then 'отказ'
                     end;

    smr_lawyer_fio = null;
    select c_name_fio
    from clients
    where c_id = :smr_from_c_id
    into smr_lawyer_fio;

    office_name = null;
    office_parent_id = null;
    select reg_code || ' ' || reg_name,
           parent
    from region
    where reg_unicode = :smr_from_reg_id
    into office_name,
         office_parent_id;

    office_parent_name = null;
    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :office_parent_id
    into office_parent_name;

    k_kontract_no = null;
    k_fact_disb_date = null;
    select k_kontract_no,
           k_fact_disb_date
    from kontract
    where k_id = :smr_from_k_id
    into k_kontract_no,
         k_fact_disb_date;

    slc_from_kc_id = null;
    slc_sum = null;
    select slc_from_kc_id,
           slc_sum_os + slc_sum_perc
    from sud_loanclient
    where slc_from_smr_id = :smd_id
    into slc_from_kc_id,
         slc_sum;

    client_id = null;
    client_fio = null;
    select c_id,
           c_name_fio
    from clients
    inner join knclient on kc_from_c_id = c_id
    where kc_id = :slc_from_kc_id and
          kc_amount_approved > 0
    into client_id,
         client_fio;

    sld_eds_id = null;
    select eds_id
    from edoc_sign
    where eds_from_edl_id = :sld_edl_id and
          eds_from_c_id = :client_id
    into sld_eds_id;

    suspend;
  end
end
