CREATE PROCEDURE AAREP_KONTSIGNREG_LIST(
    REG_ID INTEGER,
    ONLY_ACTIVE INTEGER)
RETURNS (
    KSR_KONT_TITUL VARCHAR(1024),
    KSR_KONT_POSITION VARCHAR(255),
    KSR_OTHER_1 VARCHAR(255),
    KSR_OTHER_2 VARCHAR(255),
    KSR_OTHER_3 VARCHAR(255),
    KSR_IS_ACTIVE INTEGER,
    KSR_FIO VARCHAR(100),
    KSR_FROM_C_ID DECIMAL(18,0),
    KSR_FROM_REG_ID INTEGER,
    KSR_ID INTEGER,
    KSR_REGION_NAME VARCHAR(100),
    KSR_REGION_ADDRESS VARCHAR(255),
    KSR_REGION_COMPANY VARCHAR(255),
    KSR_CAN_APPROVE INTEGER)
AS
declare variable ksm_is_active integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select distinct ksr_id, ksr_kont_titul, ksr_kont_postion, ksr_other_1, ksr_other_2, ksr_other_3, ksr_is_active,
                      ksm_is_active, c_name_fio, ksm_from_c_id, ksr_from_reg_id, reg_code || ' ' || reg_name,
                      al.adl_address_str, reg_kont_company, ksm_can_approve
      from kontsignreg, clients, kontsignmen, region_list(:reg_id, 1, -1, 0), region, addrlist al
      where (id_list = ksr_from_reg_id) and
            ksm_id = ksr_from_ksm_id and
            ksm_from_c_id = c_id and
            reg_unicode = :reg_id and
            al.adl_id = reg_address_legal and
            ((:only_active = 0) or
            (:only_active = 1 and
            ksr_is_active = 1 and
            ksm_is_active = 1 and
            coalesce(ksr_date_begin, '01.01.1990') <= current_date and
            coalesce(ksr_date_end, '31.12.9999') >= current_date))
      order by c_name_fio
      into :ksr_id, :ksr_kont_titul, :ksr_kont_position, :ksr_other_1, :ksr_other_2, :ksr_other_3, :ksr_is_active,
           :ksm_is_active, :ksr_fio, :ksr_from_c_id, :ksr_from_reg_id, :ksr_region_name, :ksr_region_address,
           :ksr_region_company, :ksr_can_approve
  do
  begin
    ksr_is_active = :ksr_is_active * :ksm_is_active;
    suspend;
  end
end
