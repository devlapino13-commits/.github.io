CREATE PROCEDURE HP_FCM_USER_TOPIC_MAILING(
    TOPIC_NAME D_STR_255,
    PUSH_TITLE D_STR_255,
    PUSH_BODY D_STR_LONG,
    LIFETIME_DAYS D_INTEGER = 7)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable topic_found d_bool;
declare variable jso_body d_str_8100;
declare variable db_id d_id;
declare variable msg_id d_id;
begin
  -- Отправка сообщений в группу рассылок

  -- Текущие, реализованные группы рассылок (топики)
  -- ru_kmp_all, kg_kmp_all, az_kmp_all, kz_kmp_all - Все пользователи КМП (по стране)
  -- ru_oivo_all - Все пользователи OIVO КМП (РФ)
  -- ru_kmp_ad  - Все пользователи КМП (РФ), подписавшие согласие на рекламу
  -- ru_kmp_test - Тестовая рассылка программистам РФ

  -- Если требуется новая группа рассылки (топик), требуются нижеописанные действия:
  --   1. Нужно доработать hp_fcm_user_topics, чтобы она возвращала имя
  --       нового топика для целевых пользователей.
  --   2. Запустить процедуру hp_fcm_user_list_fill('clear_all,add_username,add_mc_login')
  --      Эта процедура заполняет таблицу fcm_user_list, для синхронизации в FsSvc
  --      Эту процедуру вызывает робот в 21:00, так что вызывать ее
  --      вручную стоит, если только нужно ускорить процесс.
  --   3. Дождаться когда сервис FsSvc сделает синхронизацию топиков.
  --      Сервис FsSvc вычитывает записи из fcm_user_list, делает синхронизацию
  --      топиков пользователя в Firebase, после этого удаляет запись в fcm_user_list.
  --      Таким образом если fcm_user_list пустая, значит синхронизация завершена.
  --      Эта задача находится в fsrobot_sh_main, fsm_kind = 7,
  --      работает порционно, запускается регулярно
  --   4. Доработать проверки в данной процедуре, добавив топик в список допустимых
  --      и дописав в начало название и описание топика.

  --      Важно: Добавлять топик нужно только для периодических рассылок.
  --      Для разовых рассылок проще отправлять напрямую, через mc_send_push

  res_text = '';
  topic_found = 0;
  db_id = f_get_db_id();

  -- Проверяем, что топик в списке разрешенных в КМП
  if ((db_id = 81001 and topic_name in (
         'ru_kmp_ad',
         'ru_kmp_all',
         'ru_kmp_test',
         'ru_oivo_all')) or
      (db_id in (39801, 41701) and topic_name in ('kg_kmp_all')) or
      (db_id = 94401 and topic_name in ('az_kmp_all')))
  then
      topic_found = 1;

  -- Проверяем, что топик в списке разрешенных в МП
  if (topic_found = 0 and
      exists (
        select *
        from refdata
        where rd_main_code = 'mailgroup' and
              rd_parent_id is not null and
              replace(rd_memo, '/topics/', '') = :topic_name))
  then
    topic_found = 1;

  if (topic_found = 0) then
    res_text =  'Группа рассылки не реализована:' || coalesce(topic_name, 'NULL');

  -- Производим отправку
  if (res_text = '') then
  begin
    select res_text
    from mc_fcm_data(:push_body, :push_title, 'mailmessages_push',
      'fsrobot_msgbox', null, null, null, null, null)
    into jso_body;

    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body,
      msg_lifetime_days, msg_from_un_id, msg_command_kind, msg_object_kind,
      msg_object_id, msg_status)
    values (2, '/topics/' || :topic_name, :push_title, :jso_body, :lifetime_days, 1000000000000,
      'msg_to_clients', f_objkind('clients', 'id'), 1000000000000, 0)
    returning msg_id
    into msg_id;

    res_text = 'Сообщение отправлено, msg_id:' || msg_id;
  end
  suspend;
end
