CREATE PROCEDURE AAREP_DOVLIST(
    REP_REGION INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    DL_ID INTEGER,
    REG_UNICODE INTEGER,
    PARENT INTEGER,
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    DL_DOV_NO_1 INTEGER,
    DL_DOV_NO_2 INTEGER,
    DL_DATE DATE,
    DVC_FROM_C_ID DECIMAL(18,0),
    DVC_JOBNAME VARCHAR(100),
    DVC_C_NAME_FIO VARCHAR(100),
    DL_MAIN_TEXT VARCHAR(255),
    DL_MANAGER_ID DECIMAL(18,0),
    DL_MANAGER_NAME VARCHAR(100),
    DL_MANAGER_JOBNAME VARCHAR(100),
    DL_DATE_BEGIN DATE,
    DL_DATE_END DATE,
    DL_CITY VARCHAR(255),
    DL_IS_BLOB INTEGER,
    DL_BLOB_NAME VARCHAR(255),
    DL_STATUS INTEGER,
    UN_FIO VARCHAR(50),
    DL_FROM_BLN_ID INTEGER,
    BLN_NAME VARCHAR(255),
    DL_FROM_C_ID DECIMAL(18,0),
    DL_STATUS_MEMO VARCHAR(20))
AS
declare variable reg_prior integer;
begin
  reg_prior = 0;
  for select distinct dl_id, reg_unicode, parent, reg_code || ' ' || reg_name, dl_dov_no_1, dl_dov_no_2,
                      dl_date_begin dl_date, dvc_from_c_id, dvc_cli.c_name_fio, dl_main_text, dl_manager_id,
                      manager.c_name_fio, dl_date_begin, dl_date_end, dl_city,
                      case
                        when dl_blob is not null then 1
                        else 0
                      end,
                      dl_blob_name, dl_status, un_fio, dl_from_bln_id, dl_from_c_id
      from region, region_list(:rep_region, 1, -1, 1), dovlist, dovclients, clients dvc_cli, clients manager, username
      where (reg_unicode = id_list) and
            dl_from_reg_id = reg_unicode and
            dvc_from_dl_id = dl_id and
            dvc_cli.c_id = dvc_from_c_id and
            manager.c_id = dl_manager_id and
            un_id = dl_from_un_id and
            dl_date_begin between :rep_date_begin and :rep_date_end
      order by parent, reg_unicode, dl_date_begin

      into :dl_id, :reg_unicode, :parent, :reg_name, :dl_dov_no_1, :dl_dov_no_2, :dl_date, :dvc_from_c_id,
           :dvc_c_name_fio, :dl_main_text, :dl_manager_id, :dl_manager_name, :dl_date_begin, :dl_date_end, :dl_city,
           :dl_is_blob, :dl_blob_name, :dl_status, :un_fio, :dl_from_bln_id, :dl_from_c_id

  do
  begin
    if (parent <> reg_prior) then
    begin
      select reg_name
      from region
      where reg_unicode = :parent
      into :parent_name;
      reg_prior = :parent;
    end

    select jl_name
    from hr_pstatus, clients, joblist
    where ps_from_c_id = :dvc_from_c_id and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          c_id = ps_from_c_id and
          ps_from_jl_id = jl_id rows 1
    into dvc_jobname;

    select jl_name
    from hr_pstatus, clients, joblist
    where ps_from_c_id = :dl_manager_id and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          c_id = ps_from_c_id and
          ps_from_jl_id = jl_id rows 1
    into dl_manager_jobname;

    bln_name = null;
    if ((dl_from_bln_id is not null) and
        (dl_from_bln_id > 0)) then
    begin
      select f_str_join(bln_name, bln_memo)
      from blankname
      where bln_id = :dl_from_bln_id
      into :bln_name;
    end

    dl_status_memo = null;
    if (dl_status = -1) then
      dl_status_memo = 'аннулирован';
    if (dl_status = 0) then
      dl_status_memo = 'в процессе';
    if (dl_status = 1) then
      dl_status_memo = 'подписан';

    suspend;
  end
end
