CREATE PROCEDURE AAREP_KS_STAT_CLI(
    REP_C_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    REP_REGION_NAME VARCHAR(100),
    REP_REGION_CODE VARCHAR(20),
    REP_KF_NAME VARCHAR(100),
    REP_KONTRACT_NO VARCHAR(20),
    REP_CLIENT_FIO VARCHAR(100),
    REP_BONUS_TEXT VARCHAR(100),
    REP_LO_FIO VARCHAR(100),
    REP_CELL_NO VARCHAR(20),
    REP_BONUS_LAST_TEXT VARCHAR(100))
AS
declare variable c_curr_bonus_status integer;
declare variable region_id integer;
declare variable kc_id decimal(18,0);
declare variable k_id decimal(18,0);
declare variable c_id decimal(18,0);
declare variable c_last_bonus_status integer;
begin
  select c_name_fio
  from clients
  where c_id = :rep_c_id
  into :rep_lo_fio;

  for select reg1.reg_code, reg1.reg_name, oks.oks_from_k_id, kncl.kc_from_c_id, c_name_fio,
             (select cont_value
              from hp_contdata_get_cell_main(0, kncl.kc_from_c_id))
      from oksummary oks
      inner join knclient kncl on kncl.kc_from_k_id = oks.oks_from_k_id
      inner join clients cl1 on cl1.c_id = kncl.kc_from_c_id
      inner join region reg1 on reg1.reg_unicode = oks.oks_region_id
      where :rep_date between oks.oks_date_begin and oks.oks_date_end and
            kncl.kc_amount_approved > 0 and
            oks.oks_kc_id = :rep_c_id and
            oks.oks_close_state = 0
      order by 1, 3
      into :rep_region_code, :rep_region_name, :k_id, :c_id, :rep_client_fio, :rep_cell_no
  do
  begin
    select c_curr_bonus_status, c_last_bonus_status, c_curr_bonus_status_str, c_last_bonus_status_str
    from hp_kontstate_client_status(:c_id, :rep_date)
    into :c_curr_bonus_status, :c_last_bonus_status, :rep_bonus_text, :rep_bonus_last_text;

    if (c_curr_bonus_status > 0) then
    begin
      select kal_new_kf_name, kont.k_kontract_no
      from kontract kont, knaktlist kal
      where kont.k_id = :k_id and
            k_from_kf_id = kal_from_kf_id and
            kal.is_delete = 0 and
            :rep_date between kal_date and kal_date_end
      into :rep_kf_name, :rep_kontract_no;

      suspend;
    end
  end
end
