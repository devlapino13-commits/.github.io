CREATE PROCEDURE HP_SEND_PUSH_TO_STAFF(
    C_ID DECIMAL(18,0),
    SUBJECT VARCHAR(255),
    BODY VARCHAR(1024),
    UN_ID DECIMAL(18,0),
    MSG_COMMAND_KIND VARCHAR(255) = '',
    MSG_LIFETIME_DAYS INTEGER = 7)
AS
declare variable res_body blob sub_type 1 segment size 80;
declare variable un_google_id varchar(255);
begin
  if (msg_command_kind = '') then
    msg_command_kind = 'msg_to_staff';
  if (msg_lifetime_days is null) then
    msg_lifetime_days = 7;

  for select un_google_id
      from username
      where un_from_c_id = :c_id and
            coalesce(un_google_id, '') <> ''
      into :un_google_id
  do
  begin
    select res_text
    from fm_json_data(:body, :subject, 'mailmessages_push', 'fsrobot_msgbox', null, null, null, null)
    into res_body;

    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                msg_command_kind, msg_object_kind, msg_object_id)
    values (2, :un_google_id, :subject, :res_body, :msg_lifetime_days, :un_id, :msg_command_kind, 0, :c_id);
  end
end
