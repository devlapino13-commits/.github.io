CREATE PROCEDURE AAREP_HIGH_ALTITUDE_EXP(
    REP_REG_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    REG_NAME_CODE D_STR_100,
    C_FIO D_STR_100,
    JL_NAME D_STR_100,
    HIRED_DATE D_DATE,
    C_BIRTH_DATE D_DATE,
    C_AGE D_INTEGER,
    HAE_EXP VARCHAR(50),
    REG_KOEF DECIMAL(18,2),
    PERC D_INTEGER,
    CURR_STATUS D_STR_100,
    HM_EXP VARCHAR(50))
AS
declare variable C_ID D_ID;
declare variable HM_MOUNT D_INTEGER;
declare variable CALC_EXP D_INTEGER;
declare variable REG_ID D_INTEGER;
begin
  for select ps.ps_from_c_id,
             (reg.reg_code || ' ' || reg.reg_name),
             reg.reg_unicode
      from hr_pstatus ps
      inner join region_list(:rep_reg_id, 1, 3, 1) rl on rl.id_list = cast(ps.ps_from_reg_id as integer)
      inner join region reg on reg.reg_unicode = rl.id_list and
            f_reg_mount_koef(reg.reg_unicode, :rep_date) > 1 -- RFCKG-4782
      where :rep_date between ps.ps_date_begin and ps.ps_date_end and
            ps.ps_is_working = 1 and
            ps.ps_status = 1 and
            (((ps_is_working = 0) and
            ((ps_transfer_office is null) or
            (ps_transfer_office = ps_from_reg_id))) or
            (ps_is_working = 1))
      order by rl.id_list
      into :c_id, :reg_name_code, :reg_id

  do
  begin
    -- RFCKG-4782
    reg_koef = f_reg_mount_koef(:reg_id, :rep_date);
    
    hm_mount = null;

    select ps.ps_date_begin, hm.hm_mount
    from hr_pstatus ps
    inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id and
          ho.ho_order_type = 1 and
          ho.ho_status = 1
    left join hr_mount hm on hm.hm_ps_from_ho_id = ps.ps_from_ho_id and
          hm.hm_from_ps_id = ps.ps_id
    where ps.ps_from_c_id = :c_id and
          ps.ps_status = 1 and
          ps.ps_date_begin <= :rep_date
    order by ps.ps_date_begin desc
    rows 1
    into :hired_date, :hm_mount;

    if (:hired_date is null) then
      continue;

    if (:hm_mount is null) then
      hm_exp = 'Не установлен';
    else
      hm_exp = (floor(:hm_mount / 12) || iif(floor(:hm_mount / 12) between 1 and 5, ' г. ', ' л. ') || mod(:hm_mount, 12) || ' м.');

    select c.c_name_fio, c.c_birth_date, f_age(c.c_birth_date, :rep_date), jl.jl_name
    from clients c
    inner join hr_pstatus ps on ps.ps_from_c_id = c.c_id and
          ps.ps_status = 1 and
          :rep_date between ps.ps_date_begin and ps.ps_date_end
    inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
    where c.c_id = :c_id
    order by ps.ps_date_begin desc
    into :c_fio, :c_birth_date, :c_age, :jl_name;

    curr_status = '';

    select sw.sawt_name
    from hr_orders ho
    inner join hr_voctravel hrv on hrv.hrv_from_ho_id = ho.ho_id and
          hrv.hrv_from_c_id = :c_id and
          :rep_date between hrv.hrv_date_begin and hrv.hrv_date_end_fact
    inner join sa_worktype sw on ho.ho_order_type = sw.sawt_id
    where ho.ho_status = 1
    order by ho.ho_date desc
    rows 1
    into :curr_status;

    select month_out
    from hp_high_mountain_exp_calc(:c_id, :hired_date, :rep_date)
    into :calc_exp;

    hm_mount = iif(:hm_mount is null, 0, :hm_mount) + calc_exp;
    hae_exp = (floor(:hm_mount / 12) || iif(floor(:hm_mount / 12) between 1 and 4, ' г. ', ' л. ') || mod(:hm_mount, 12) || ' м.');
    perc = 0;
    if (:c_age < 28 and (:hm_mount / 12) < 10) then
      perc = 5;
    else
    begin
      perc = floor(hm_mount / 60) * 5;
    end

    if (perc > 30 and f_get_db_id() = 41701) then
        perc = 30;

    suspend;

  end
end
