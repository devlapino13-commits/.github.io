CREATE PROCEDURE FIND_KANSORDERLIST(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_KIND INTEGER,
    REPORT_TEXT VARCHAR(255))
RETURNS (
    KSO_ID INTEGER,
    KSO_DATE DATE,
    KSO_FROM_REG_ID INTEGER,
    KSO_DATETIME TIMESTAMP,
    KSO_STATE INTEGER,
    KSO_MEMO VARCHAR(255),
    KSO_REGION_NAME VARCHAR(100),
    KSO_STATE_TEXT VARCHAR(20),
    KSO_IS_DELETE INTEGER)
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  report_text = upper(:report_text);
  if (report_text <> '') then
  begin
    for select distinct kso_id, kso_date, kso_from_reg_id, kso_datetime, kso_state, kso_memo, kso_is_delete,
                        reg_code || ' ' || reg_name
        from kansorder, region, kanslog, username, clients, kansorlist, kanslist, region_list(:report_region_id, 1, -1, 1)
        where kso_from_reg_id = reg_unicode and
              (kso_from_reg_id = id_list) and
              kso_date >= :report_date_begin and
              kso_date <= :report_date_end and
              ((:report_kind = -1) or
              (kso_state = :report_kind)) and
              (upper(kso_memo) like '%' || :report_text || '%' or
              upper(reg_name) like '%' || :report_text || '%' or
              upper(c_name_fio) like '%' || :report_text || '%' or
              upper(ksl_name) like '%' || :report_text || '%' or
              upper(ksl_memo) like '%' || :report_text || '%') and
              ksg_from_kso_id = kso_id and
              ksg_from_un_id = un_id and
              un_from_c_id = c_id and
              kol_from_kso_id = kso_id and
              kol_from_ksl_id = ksl_id and
              kso_is_delete = 0
        into :kso_id, :kso_date, :kso_from_reg_id, :kso_datetime, :kso_state, :kso_memo, :kso_is_delete,
             :kso_region_name
    do
    begin
      if (kso_state = 0) then
        kso_state_text = 'В процессе';
      if (kso_state = 1) then
        kso_state_text = 'Проверен';
      if (kso_state = 2) then
        kso_state_text = 'Согласован';
      if (kso_state = 3) then
        kso_state_text = 'Утвержден';
      if (kso_state = 4) then
        kso_state_text = 'Куплен';
      if (kso_state = 5) then
        kso_state_text = 'Отправлен';
      if (kso_state = 6) then
        kso_state_text = 'Доставлен';
      suspend;
    end
  end
  else
  begin
    for select distinct kso_id, kso_date, kso_from_reg_id, kso_datetime, kso_state, kso_memo, kso_is_delete,
                        reg_code || ' ' || reg_name
        from kansorder, region, region_list(:report_region_id, 1, -1, 1)
        where kso_from_reg_id = reg_unicode and
              (kso_from_reg_id = id_list) and
              kso_date >= :report_date_begin and
              kso_date <= :report_date_end and
              ((:report_kind = -1) or
              (kso_state = :report_kind)) and
              kso_is_delete = 0
        into :kso_id, :kso_date, :kso_from_reg_id, :kso_datetime, :kso_state, :kso_memo, :kso_is_delete,
             :kso_region_name
    do
    begin
      if (kso_state = 0) then
        kso_state_text = 'В процессе';
      if (kso_state = 1) then
        kso_state_text = 'Проверен';
      if (kso_state = 2) then
        kso_state_text = 'Согласован';
      if (kso_state = 3) then
        kso_state_text = 'Утвержден';
      if (kso_state = 4) then
        kso_state_text = 'Куплен';
      if (kso_state = 5) then
        kso_state_text = 'Отправлен';
      if (kso_state = 6) then
        kso_state_text = 'Доставлен';
      suspend;
    end
  end
end
