CREATE PROCEDURE HP_REPAY_FILL(
    K_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    REP_DATE DATE,
    IS_WRITEOFF INTEGER,
    REPAY_TO_OS DOUBLE PRECISION,
    REPAY_TO_PERC DOUBLE PRECISION,
    REPAY_TO_FINE DOUBLE PRECISION,
    REPAY_TO_KOMMIS DOUBLE PRECISION,
    SHOW_ZERO_LINE D_SMALLINT = 0)
RETURNS (
    RES_AMOUNT DECIMAL(18,4),
    S_RUNNING_TOTAL DECIMAL(18,4))
AS
declare variable s_rec_no integer;
declare variable pr_pay_pros_more integer;
declare variable pr_round_perc integer;
declare variable pr_percent_method integer;
declare variable k_disb_date date;
declare variable tax_perc decimal(18,2);
declare variable s_total_sh_all decimal(18,4) = 0;
declare variable rep_saldo decimal(18,4) = 0;
declare variable s_all decimal(18,4) = 0;
declare variable s_total_sh decimal(18,4) = 0;
declare variable cs_pros integer;
declare variable cs_pred integer;
declare variable curr_repay decimal(18,4) = 0;
declare variable disb_amount decimal(18,4) = 0;
declare variable kd_percent_per_day decimal(18,7);
declare variable repay_need_reschedule integer;
declare variable client_koef double precision;
declare variable fact_loan_amount decimal(18,4) = 0;
declare variable kc_disb_amount decimal(18,4) = 0;
declare variable kc_from_c_id decimal(18,0);
declare variable kc_amount_approved decimal(18,4);
declare variable kc_closed_ol_id decimal(18,0);
declare variable pr_is_360 integer;
declare variable client_fio varchar(100);
declare variable pr_kommis_is_percent integer;
declare variable res_kommis_amount decimal(18,4) = 0;
declare variable pr_pay_curr integer;
declare variable db_id integer;
declare variable is_closed smallint;
declare variable pr_code d_str_255;
declare variable is_extra d_smallint;
declare variable has_writeoff_debt d_smallint;
declare variable new_ipf_repay_mode d_bool;
declare variable k_is_vacated d_integer;
declare variable vac_os_sh_id d_integer;
declare variable vac_date_begin d_date;
declare variable vac_date_end d_date;
declare variable vac_max_part_limit d_currency;
begin
  delete from temp_vrepay
  where 1 = 1;

  if (c_id is null) then
    c_id = 0;

  if (show_zero_line is null) then
    show_zero_line = 0;

  db_id = f_get_db_id();

  -- Вычисляем коэффициент доли клиента
  client_koef = 1.0;
  if (c_id <> 0) then
  begin
    fact_loan_amount = 0;
    kc_disb_amount = 0;
    for select kc_from_c_id, kc_amount_approved, kc_closed_ol_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into :kc_from_c_id, :kc_amount_approved, :kc_closed_ol_id
    do
    begin
      if (kc_from_c_id = c_id) then
        kc_disb_amount = kc_amount_approved;
      if (kc_closed_ol_id is null) then
        fact_loan_amount = fact_loan_amount + kc_amount_approved;
      else
      if (not(exists(select *
                     from operlist
                     where ol_id = :kc_closed_ol_id and
                           is_delete = 0))) then
        fact_loan_amount = fact_loan_amount + kc_amount_approved;
    end
    if ((fact_loan_amount <> 0) and
        (kc_disb_amount <> 0)) then
      client_koef = kc_disb_amount / fact_loan_amount;
    else
      client_koef = 1.0;
  end

  has_writeoff_debt = 0;
  if (db_id in (39801, 41701) and ((select sum(oko.oko_os + oko.oko_sh_os + oko.oko_perc + oko.oko_sh_perc + coalesce(oko.oko_other, 0) + coalesce(oko.oko_sh_other, 0))
                         from operkroff oko
                         where oko.oko_from_k_id = :k_id and
                               oko.oko_ol_is_active = 1 and
                               oko.oko_ol_date <= :rep_date) > 0)) then
    has_writeoff_debt = 1;

  -- Заполнение необходимых параметров кредита во временной талице для вложенных процедур
  select pr_pay_curr, pr_pay_pros_more, pr_pay_pros, pr_pay_pred, k_disb_date, -pr_round_perc, pr_interest_method,
         k_amount_approved, k_is_vacated, pr_need_reschedule, pr_is_360, pr_kommis_is_percent, pr_code
  from product, kontract
  where k_id = :k_id and
        pr_id = k_from_pr_id
  into :pr_pay_curr, :pr_pay_pros_more, cs_pros, cs_pred, k_disb_date, pr_round_perc, pr_percent_method,
       disb_amount, k_is_vacated,
       repay_need_reschedule, pr_is_360, pr_kommis_is_percent, pr_code;
  if (pr_pay_pros_more is null) then
    pr_pay_pros_more = 999999;

  if (k_is_vacated is null) then
    k_is_vacated = 0;

  vac_os_sh_id = null;
  if (k_is_vacated = 1) then
  begin
    select last_os_sh_id, vac_date_begin, vac_date_end, vac_max_part_limit
    from hp_kont_vac_info(:k_id, :rep_date)
    into vac_os_sh_id, vac_date_begin, vac_date_end, vac_max_part_limit;
  end

  new_ipf_repay_mode = f_ipf_newrepmode(rep_date, k_id, is_writeoff);

  -- получаем ближайшую не нулевую ставку, чтобы расчет аннуитета не обнулился, 24.11.2019, Sanjari
  kd_percent_per_day = 0;
  select kd_percent_per_day
  from kontdata
  where kd_from_k_id = :k_id and
        kd_ol_date <= :rep_date and
        kd_date_end >= :rep_date and
        kd_ol_is_delete = 0
  into kd_percent_per_day;

  if (kd_percent_per_day = 0) then
  begin
    select kd_percent_per_day
    from kontdata
    where kd_from_k_id = :k_id and
          kd_ol_date > :rep_date and
          kd_ol_date <= kd_date_end and
          kd_ol_is_delete = 0
    order by kd_ol_date
    rows 1
    into kd_percent_per_day;
  end

  --проверка контракт закрыт или нет, ubakachiev
  is_closed = 0;
  if (exists(select *
             from kontstate
             where ks_ol_is_delete = 0 and
                   ks_state = 2 and
                   ks_from_k_id = :k_id and
                   :rep_date between ks_date_begin and ks_date_end)) then
    is_closed = 1;



  -- << -- igusev 22.09.2018 #operkroff_pay вместо закомментированного блока выше, создал скрипт:
  if (c_id = 0) then
  begin
    -- заполняем временную ОСНОВНУЮ таблицу
    insert into temp_vrepay (s_id, s_date, s_all, zad_os, zad_perc, zad_fine, calcing_perc,
                             calcing_perc_orig,
                             zad_kommis, is_grace,
                             repay_mode, k_index_curr_rate, k_stop_perc, dd_change_begin, dd_change_end, dd_change_days,
                             zad_extra, sh_prior_percent_per_day, is_extra, is_hidden_line, is_vac_perc)
    select sh_id, sh_date, s_all, iif(:is_closed = 1 and zad_os < 0 and zad_os + zad_perc <> 0, 0, zad_os) as zad_os,
           zad_perc + sh_add_perc, sh_calc_fine, iif(:pr_pay_curr = 2, sh_calc_perc, zad_perc) + sh_add_perc, -- либо накопленные либо процент по графику (наценка по ИПФ)
           sh_calc_perc + sh_add_perc,
           iif(is_extra = 0, zad_kommis, 0.0), is_grace, sh_repay_type, k_index_curr_rate, k_stop_perc, dd_change_begin,
           dd_change_end, dd_change_days, iif(is_extra = 1, zad_kommis, 0.0), sh_prior_percent_per_day, is_extra, is_hidden_line,
           iif(sh_id > :vac_os_sh_id, 1, 0)
    from aarep_make_zad_list(:k_id, :rep_date, 1, :show_zero_line)
    where (:is_writeoff = 0 or
          (:is_writeoff = 1 and
          sh_date is not null))
    order by coalesce(sh_date, '31.12.9999'), sh_id; -- Sanjari, 22.04.2020;

    if (new_ipf_repay_mode = 1) then
    begin
      update temp_vrepay
      set calcing_perc = calcing_perc_orig
      where repay_mode = 3;
    end

    if (not exists(select *
                   from temp_vrepay) and
        is_writeoff = 0 and
        has_writeoff_debt = 0) then
    begin
      insert into temp_vrepay (s_id, s_date, sh_prior_percent_per_day, is_hidden_line)
      select ok_sh_id, ok_sh_date, :kd_percent_per_day, 0
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1
      order by ok_sh_date desc, ok_sh_id desc
      rows 1;
    end

    if (has_writeoff_debt = 1) then
    begin
      delete from temp_vrepay
      where 1 = 1;

      insert into temp_vrepay (s_id, s_date, s_all, zad_os, zad_perc, zad_fine, calcing_perc, zad_kommis, is_grace,
                               repay_mode, k_index_curr_rate, k_stop_perc, dd_change_begin, dd_change_end,
                               dd_change_days, zad_extra, sh_prior_percent_per_day, is_hidden_line)
      select 999, null, sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc),
             sum(okf.oko_os + okf.oko_sh_os), sum(okf.oko_perc + okf.oko_sh_perc), sum(okf.oko_other + okf.oko_sh_other),
             sum(okf.oko_perc + okf.oko_sh_perc), 0, 0, 1, null, null, null, null, null, 0, max(:kd_percent_per_day), max(0)
      from operkroff okf
      where okf.oko_from_k_id = :k_id and
            okf.oko_ol_is_active = 1;
    end
  end
  else
  begin
    -- заполняем временную ОСНОВНУЮ таблицу
    insert into temp_vrepay (s_id, s_date, s_all, zad_os, zad_perc, zad_fine, calcing_perc, zad_kommis, is_grace,
                             repay_mode, k_index_curr_rate, k_stop_perc, dd_change_begin, dd_change_end, dd_change_days,
                             zad_extra, sh_prior_percent_per_day, is_extra, is_hidden_line, is_vac_perc)
    select sh_id, sh_date, s_all, iif(:is_closed = 1 and zad_os < 0 and zad_os + zad_perc <> 0, 0, zad_os) as zad_os,
           zad_perc + sh_add_perc, sh_calc_fine, iif(:pr_pay_curr = 2, sh_calc_perc, zad_perc) + sh_add_perc, -- либо накопленные либо процент по графику (наценка по ИПФ)
           iif(is_extra = 0, zad_kommis, 0.0), is_grace, sh_repay_type, k_index_curr_rate, k_stop_perc, dd_change_begin,
           dd_change_end, dd_change_days, iif(is_extra = 1, zad_kommis, 0.0), sh_prior_percent_per_day, is_extra, is_hidden_line,
           iif(sh_id > :vac_os_sh_id, 1, 0)
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    where sh_date is not null
    order by coalesce(sh_date, '31.12.9999'), sh_id; -- Sanjari, 22.04.2020;

    if ((is_writeoff = 0) and
        (db_id = 81001)) then
    begin
      insert into temp_vrepay (s_id, s_date, s_all, zad_os, zad_perc, zad_fine, calcing_perc, zad_kommis, is_grace,
                               repay_mode, k_index_curr_rate, k_stop_perc, dd_change_begin, dd_change_end,
                               dd_change_days, zad_extra, sh_prior_percent_per_day, is_hidden_line)
      select 999, null, sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc),
             sum(okf.oko_os + okf.oko_sh_os), sum(okf.oko_perc + okf.oko_sh_perc), 0,
             sum(okf.oko_perc + okf.oko_sh_perc), 0, 0, 1, null, null, null, null, null, 0, max(:kd_percent_per_day), max(0)
      from operkroff okf
      where okf.oko_from_c_id = :c_id and
            okf.oko_from_k_id = :k_id and
            okf.oko_ol_is_active = 1;
    end
  end
  -- << ------------------------------------------------------------------------

  -- количество записей
  select max(s_rec_no)
  from temp_vrepay
  into :s_rec_no;

  -- Вычисление комиссии за досрочное погашение
  if (is_writeoff = 0) then
  begin
    select res_kommis_amount
    from hp_repay_calc_kommis(:k_id, :c_id, :rep_date)
    into :res_kommis_amount;

    -- Если есть комиссия, вставляем сумму на последнее погашение
    if (res_kommis_amount <> 0) then
    begin
      update temp_vrepay
      set zad_kommis = :res_kommis_amount
      where s_rec_no = :s_rec_no;
    end
  end

  tax_perc = 0;
  select nsp_percent_for_int
  from hp_get_kontract_nsp_acc(0, :k_disb_date)
  into :tax_perc;

  if (has_writeoff_debt = 1) then
    tax_perc = 0;

  -- по индексированным кредитам не должно начисляться НСП, добавил Улан от 30.03.2016
  if (k_id in (109010020831300101, 109010020831500101, 109010020831700101, 109010020831900101)) then
  begin
    tax_perc = 0;
  end
  select is_extra
  from temp_vrepay
  where s_id = 0
  into is_extra;

  if (pr_code containing '.ipf.' and is_extra = 1) then
    tax_perc = 0;

  client_fio = '';
  if (c_id <> 0) then
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into :client_fio;
  end

  -- для РФ за указанный период отключаем пени
  -- если договор расторгнут и не числится за внешними коллекторами
  if ((rep_date between '16.11.2015' and '31.12.2015') or
      (rep_date between '21.01.2016' and '29.02.2016')) then
  begin
    if (f_get_db_id() = 81001) then -- только для РФ
    begin
      if (exists(select *
                 from kontract
                 where k_id = :k_id and
                       (coalesce(k_stop_perc, '31.12.9999') < :rep_date))) then -- по расторгнутым договорам
      begin
        -- если не числится у внешнего коллектора
        if (not exists(select *
                       from sb_akt
                       where sba_is_delete = 0 and
                             sba_status = 2 and
                             sba_from_sbl_id in (1, 12, 87) and
                             :rep_date between sba_date_begin and sba_date_end and
                             sba_from_k_id = :k_id)) then
        begin
          -- обнуляем пени
          update temp_vrepay
          set zad_fine = 0
          where 1 = 1;
        end
      end
    end
  end

  delete from temp_params
  where 1 = 1;

  insert into temp_params (pr_round_perc, tax_perc, pr_percent_method, pr_pay_pros_more, cs_pros, cs_pred, k_disb_date,
                           disb_amount, kd_percent_per_day, repay_need_reschedule, client_koef, pr_is_360, client_fio,
                           pr_kommis_is_percent, vac_date_begin, vac_date_end,
                           vac_last_sh_id, vac_updated_sh_id, vac_max_part_limit)
  values (:pr_round_perc, :tax_perc, :pr_percent_method, :pr_pay_pros_more, :cs_pros, :cs_pred, :k_disb_date,
          :disb_amount, :kd_percent_per_day, :repay_need_reschedule, :client_koef, :pr_is_360, :client_fio,
          :pr_kommis_is_percent, :vac_date_begin, :vac_date_end,
          :vac_os_sh_id, :vac_os_sh_id, :vac_max_part_limit);
  -- конец заполнения.

  -- Вычисляем нарастащий итог
  rep_saldo = 0;
  s_total_sh_all = 0;
  res_amount = 0;
  for select s_rec_no,
             zad_os + calcing_perc + zad_fine + zad_kommis + zad_extra + round((calcing_perc * iif(:db_id in (39801, 41701) and s_date is null, 0, :tax_perc)) / 100.00, :pr_round_perc) + round((zad_kommis * :tax_perc) / 100.00, :pr_round_perc) as f1,

             zad_os + zad_perc + zad_fine + zad_kommis + zad_extra + round((calcing_perc * iif(:db_id in (39801, 41701) and s_date is null, 0, :tax_perc)) / 100.00, :pr_round_perc) + round((zad_kommis * :tax_perc) / 100.00, :pr_round_perc) as f2,

             iif(repay_mode <= 2, zad_os * :repay_to_os + zad_perc * :repay_to_perc + zad_fine * :repay_to_fine + zad_kommis * :repay_to_kommis + zad_extra + round((calcing_perc * :repay_to_perc * iif(:db_id in (39801, 41701) and s_date is null, 0, :tax_perc)) / 100.00, :pr_round_perc) + round((zad_kommis * :repay_to_kommis * :tax_perc) / 100.00, :pr_round_perc), 0) as f3

      from temp_vrepay
      order by s_rec_no
      into :s_rec_no, :s_all, s_total_sh, :curr_repay
           --  as cursor cur
  do
  begin
    res_amount = res_amount + curr_repay;

    rep_saldo = rep_saldo + s_all;
    s_total_sh_all = s_total_sh_all + s_total_sh;
    update temp_vrepay
    set s_all = :s_all,
        s_running_total = maxvalue(:rep_saldo, :s_total_sh_all)
    where s_rec_no = :s_rec_no;
    --    where current of cur;
  end
  update temp_vrepay
  set s_running_total = :rep_saldo
  where s_running_total > :rep_saldo;

  select s_running_total
  from temp_vrepay
  where s_rec_no = :s_rec_no
  into :s_running_total;

  suspend;

  -- вычислить сумму комиссии в AAREP_MAKE_ZAD_LIST
end
