CREATE PROCEDURE FM_SEND_CLIENT_GBDFL(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_2048,
    NEED_REFRESH D_SMALLINT)
AS
declare variable C_INN D_STR_20;
declare variable PASS_WHO D_STR_255;
declare variable PASS_WHO_CODE D_STR_20;
declare variable BC_STATUS D_INTEGER;
begin
    need_refresh = 1;

    select c.c_inn, c.c_pass_who, c.c_pass_who_code
    from clients c
    where c.c_id = :c_id
    into c_inn, pass_who, pass_who_code;

    if (coalesce(c_inn, '') <> '' and
        char_length(c_inn) = 12) then
    begin

      if (not exists (select *                             -- Если не было запроса в БМГ
                      from bmg_clients bc
                      where bc.bc_from_c_id = :c_id
                        and bc.bc_is_last = 1
                        and cast(bc.bc_create_datetime as d_date) = current_date)) then
      begin
        execute procedure bmg_new_request(:c_id, :un_id, 1); -- RFCKG-4180 отправляем данные в БМГ
        res_text = 'Запрос БМГ в процессе';
        suspend;
        exit;
      end

      select bc.bc_status       --  проверка номера телефона
      from bmg_clients bc
      where bc.bc_from_c_id = :c_id
        and bc.bc_is_last = 1
      order by bc.bc_status_datetime
      rows 1
      into bc_status;

      if (bc_status <> 1) then
      begin
        res_text = (case bc_status
                      when -2 then 'Ответ от БМГ: Номер клиента не соотвествует ИНН'
                      when -3 then 'Ответ от БМГ: не получен'
                      when -4 then 'Ответ от БМГ: не получен'
                      else 'Запрос БМГ в процессе'
                   end);

        suspend;
        exit;
      end

      execute procedure hp_extinfo_insert('gbdfl_get_cli_data', f_objkind('clients', 'id'), :c_id, 1);
      res_text = 'Данные успешно отправлены в ГБДФЛ';
    end
    else
      res_text = 'Для отправки запроса в ГБДФЛ надо заполнить ИНН клиента';
    suspend;
end
