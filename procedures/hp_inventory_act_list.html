CREATE PROCEDURE HP_INVENTORY_ACT_LIST(
    UN_ID ID_TYPE)
RETURNS (
    IM_ID ID_TYPE,
    IM_DATE D_DATE,
    IM_ORDER_DATE D_DATE,
    IM_ORDER_NO D_STR_20,
    IM_DATE_INV D_DATE,
    IM_WORKPLACE_ID ID_TYPE,
    IM_STATUS D_INTEGER,
    IM_DATE_APPROVED D_DATE,
    IM_PRESIDENT ID_TYPE,
    IM_USER_ID ID_TYPE,
    IM_PRESIDENT_FIO D_STR_100,
    UN_NAME VARCHAR(30),
    IM_NO D_INTEGER,
    FWP_NAME D_STR_100,
    IM_CATEG_ID D_INTEGER,
    INVENTORY_SIGN_STATUS D_SMALLINT)
AS
begin
  for
    select
      im_id,
      im_date,
      im_order_date,
      im_order_no,
      im_date_inv,
      im_workplace_id,
      im_status,
      im_date_approved,
      im_president,
      im_user_id,
      c_name_fio,
      un_name,
      im_no,
      fwp_name,
      im_categ_id
    from full_wp_list(1000000, 1)
      inner join invmain
        on im_workplace_id = fwp_id
      inner join aarep_region_user_tmp(:un_id)
        on reg_list = fwp_region_id
      left join clients
        on c_id = im_president
      left join username
        on un_id = im_user_id
    order by im_date desc
  into
      :im_id,
      :im_date,
      :im_order_date,
      :im_order_no,
      :im_date_inv,
      :im_workplace_id,
      :im_status,
      :im_date_approved,
      :im_president,
      :im_user_id,
      :im_president_fio,
      :un_name,
      :im_no,
      :fwp_name,
      :im_categ_id
  do
  begin
    inventory_sign_status = 0;
    select iif(count(*) = sum(coalesce(imem.imem_sign_status, 0)), 1, 0)
    from invmember imem
    where imem.imem_from_im_id = :im_id
    into inventory_sign_status;
    suspend;
  end
end
