CREATE PROCEDURE HP_CHECK_RU_INN(
    C_INN D_STR_CODE,
    C_TYPE D_SMALLINT)
AS
declare variable msg_wrong_length d_str_255;
declare variable msg_wrong_char d_str_255;
declare variable msg_wrong_check d_str_255;
declare variable inn_length d_integer;
declare variable i d_integer;
declare variable c1 d_integer;
declare variable n d_integer;
declare variable c2 d_integer;
begin
  msg_wrong_length = 'Неверное количество цифр в поле ИНН: %1, должно быть %2';
  msg_wrong_char = 'Неверные символы в поле ИНН: %1, должны быть цифры';
  msg_wrong_check = 'Не совпадает контрольная сумма в поле ИНН: %1, проверьте правильность ввода.';

  inn_length = char_length(c_inn);

  -- У физ. лиц длина ИНН 12
  if (c_type = 0 and inn_length <> 12) then
    exception e_message f_format(msg_wrong_length, inn_length, 12);

  -- У юр. лиц длина ИНН 10, у ИП длина 12
  if (c_type = 1 and inn_length not in (10, 12)) then
    exception e_message f_format(msg_wrong_length, inn_length, '10 или 12');

  if (c_inn not similar to '[[:DIGIT:]]+') then
    exception e_message f_format(msg_wrong_char, c_inn);

  -- Юр. лица, проверка контрольной последней цифры
  if (inn_length = 10) then
  begin
    i = 1;
    c1 = 0;
    while (i <= 9) do
    begin
      n = substring(c_inn from i for 1);
      c1 = c1 + n *
        case i
          when 1 then 2
          when 2 then 4
          when 3 then 10
          when 4 then 3
          when 5 then 5
          when 6 then 9
          when 7 then 4
          when 8 then 6
          when 9 then 8
        end;
      i = i + 1;
    end
    c1 = mod(mod(c1, 11), 10);
    n = substring(c_inn from 10 for 1);
    if (n <> c1) then
      exception e_message f_format(msg_wrong_check, c_inn);
  end

  -- Физ. лица и ИП, проверка контрольных двух последних цифр
  if (inn_length = 12) then
  begin
    i = 1;
    c1 = 0;
    c2 = 0;
    while (i <= 10) do
    begin
      n = substring(c_inn from i for 1);
      c1 = c1 + n * decode(i,1,7,2,2,3,4,4,10,5,3,6,5,7,9,8,4,9,6,10,8);
      c2 = c2 + n * decode(i,1,3,2,7,3,2,4,4,5,10,6,3,7,5,8,9,9,4,10,6);
      i = i + 1;
    end
    c1 = mod(mod(c1, 11), 10);
    n = substring(c_inn from 11 for 1);
    if (n <> c1) then
      exception e_message f_format(msg_wrong_check, c_inn);
    c2 = mod(mod(c2 + c1 * 8, 11), 10);
    n = substring(c_inn from 12 for 1);
    if (n <> c2) then
      exception e_message f_format(msg_wrong_check, c_inn);
  end
end
