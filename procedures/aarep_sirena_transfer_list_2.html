CREATE PROCEDURE AAREP_SIRENA_TRANSFER_LIST_2(
    REP_PREOJECT STR_100,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OPER_CALLED STR_100,
    PANT_DATE_BEGIN DATE,
    PANT_DATE_END DATE,
    CC_PROJECT STR_100,
    K_KONTRACT_NO STR_100,
    DELINQ_OS CURR_TYPE,
    DELINQ_PERC CURR_TYPE,
    DELINQ_FINE CURR_TYPE,
    DELINQ_DAYS INTEGER,
    DAYS_ON_HAND INTEGER,
    PROJECT STR_100)
AS
declare variable pant_id id_type;
declare variable kf_officer id_type;
begin
  for select pantk.pant_oper_called, pantk.pant_date_begin, pantk.pant_date_end, pantk.pant_id, rd.rd_name,
             kont.k_kontract_no, kal_new_kf_officer
      from pantera_kontracts pantk
      inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
      inner join kontract kont on kont.k_id = pantk.pant_from_k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
      where ((pantk.pant_date_end between :rep_date_begin and :rep_date_end) -- чтобы учесть группы которые были заблокированы за прерыдущие периоды и азакрыты в указанный
            or
            (pantk.pant_date_begin between :rep_date_begin and :rep_date_end)) -- чтобы учесть группы заблокированные за указанный период
            and
            rd_code containing (:rep_preoject) and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into :oper_called, :pant_date_begin, :pant_date_end, :pant_id, :cc_project, :k_kontract_no, :kf_officer
  do
  begin
    delinq_os = 0;
    delinq_perc = 0;
    delinq_fine = 0;
    delinq_days = 0;

    select pt.pt_delinq_os, pt.pt_delinq_perc, pt.pt_delinq_fine, pt.pt_delinq_days
    from pantera_transfer pt
    where pt.pt_input_pant_id = :pant_id rows 1
    into :delinq_os, :delinq_perc, :delinq_fine, :delinq_days;

    if (pant_date_end = '30.12.9999') then
      days_on_hand = datediff(day from :pant_date_begin to :rep_date_end) + 1;
    else
      days_on_hand = datediff(day from :pant_date_begin to :pant_date_end) + 1;

    if (oper_called = 'IGUSEV') then
      oper_called = 'Виртуальный оператор';

    if (kf_officer = 1000000000004) then
      project = 'ВС Сирена';
    else
      project = 'СВЗ';

    suspend;
  end
end
