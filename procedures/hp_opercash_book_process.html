CREATE PROCEDURE HP_OPERCASH_BOOK_PROCESS(
    OPC_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable opc_oper_kind d_integer;
declare variable opc_from_ol_id d_id;
declare variable oper_kind d_str_code;
declare variable opc_from_k_id d_id;
declare variable cash_al_id d_str_20;
declare variable opc_manager_id d_id;
declare variable opc_office_id d_integer;
declare variable opc_date d_date;
declare variable opc_amount d_currency;
declare variable opc_from_c_id d_id;
declare variable opc_doc_no d_integer;
declare variable opc_memo d_str_long;
declare variable oper_name d_str_long;
declare variable opc_corr_al_id d_str_20;
declare variable main_ol_id d_id;
declare variable res_opc_id d_integer;
declare variable opc_corr_reg_id d_integer;
declare variable opc_sign d_smallint;
declare variable kommis_ol_id d_id;
declare variable opc_is_signed d_bool;
declare variable opc_status d_smallint;
declare variable ol_is_delete d_bool;
declare variable pr_id d_integer;
declare variable ok_sh_id d_integer;
declare variable ok_sh_date d_date;
declare variable res_main_ol_id d_id;
declare variable loan_ol_id d_id;
declare variable cac_id d_integer;
declare variable edl_id d_id;
declare variable opc_other_client d_str_255;
declare variable temp_ol_id d_id;
declare variable temp_ok_os d_currency;
declare variable temp_ok_perc d_currency;
declare variable temp_ok_tax d_currency;
declare variable temp_amount d_currency;
begin
  res_is_error = 1;
  res_text = '';

  for select opc_id, opc_date, opc_sign, opc_doc_no, opc_oper_kind, opc_from_ol_id, opc_from_k_id, opc_office_id, opc_manager_id,
         opc_amount, opc_from_c_id, opc_corr_reg_id, opc_is_signed, opc_status,
         iif(opc_is_ipf = 1, csp_ipf_al_id, csp_cls_al_id), opc_corr_al_id, opc_memo, opc_other_client, ol.is_delete
  from opercash
       inner join cashplace on opc_from_csp_id = csp_id
       left join operlist OL on opc_from_ol_id = ol_id
  where opc_id = :opc_id and            
        opc_parent_id is null
  union
  select opc_id, opc_date, opc_sign, opc_doc_no, opc_oper_kind, opc_from_ol_id, opc_from_k_id, opc_office_id, opc_manager_id,
         opc_amount, opc_from_c_id, opc_corr_reg_id, opc_is_signed, opc_status,
         iif(opc_is_ipf = 1, csp_ipf_al_id, csp_cls_al_id), opc_corr_al_id, opc_memo, opc_other_client, OL.is_delete
  from opercash
       inner join cashplace on opc_from_csp_id = csp_id
       left join operlist OL on opc_from_ol_id = ol_id
  where opc_parent_id = :opc_id
  order by 1
  into res_opc_id, opc_date, opc_sign, opc_doc_no, opc_oper_kind, opc_from_ol_id, opc_from_k_id, opc_office_id, opc_manager_id,
       opc_amount, opc_from_c_id, opc_corr_reg_id, opc_is_signed, opc_status,
       cash_al_id, opc_corr_al_id, opc_memo, opc_other_client, ol_is_delete
  do
  begin
    res_text = '';
    res_is_error = 1;
    res_main_ol_id = null;

    if (ol_is_delete is null) then
      ol_is_delete = 0;

    if (ol_is_delete = 1) then
      opc_from_ol_id = null;

    if (opc_is_signed = 0) then
    begin
      res_text = 'Документ не подтвержден менеджером, № ' || opc_doc_no || ', сумма: ' || f_numtostr(opc_amount, 0);

      rdb$set_context('USER_TRANSACTION', 'Opercash_error_text', res_text);
      exception e_message res_text;
    end

    if (opc_status <> 1) then
    begin
      res_text = 'Документ еще не подписан, № ' || opc_doc_no || ', сумма: ' || f_numtostr(opc_amount, 0);

      rdb$set_context('USER_TRANSACTION', 'Opercash_error_text', res_text);
      exception e_message res_text;
    end

    if (opc_from_ol_id is not null) then
    begin
      res_text = 'Операция уже проведена, № ' || opc_doc_no || ', сумма: ' || f_numtostr(opc_amount, 0);

      rdb$set_context('USER_TRANSACTION', 'Opercash_error_text', res_text);
      exception e_message res_text;
    end
  
    oper_kind = f_enum_code('cashoperkind', opc_oper_kind);
    oper_name = f_enum_name('cashoperkind', opc_oper_kind);
  
    res_is_error = 0;
    if (oper_kind = 'minus.loan.disb') then
    begin
      select ol_id_main, res_text
      from hp_do_disbursement(:un_id, :opc_from_k_id, :opc_office_id, 1, :cash_al_id, :opc_office_id, 1, :cash_al_id, :opc_from_c_id, :opc_manager_id)
      into :opc_from_ol_id, :res_text;
    end else
    if (oper_kind = 'minus.loan.overpay') then
    begin
      if (coalesce(opc_corr_al_id, '') = '') then
      begin
        select k_from_pr_id, k_kontract_no || ' от ' || f_datetostr(k_disb_date)
        from kontract
        where k_id = :opc_from_k_id
        into pr_id, opc_memo;

        opc_memo = 'Возврат переплаты по договору займа ' || opc_memo || ' г.';

        select res_account
        from hp_make_kredit_acclist(:pr_id, :opc_corr_reg_id, 8) -- Счет переплаты
        into :opc_corr_al_id;
      end

      select res_main_ol_id, ol_id, res_text
      from hp_opercash_other_save(:opc_date, :opc_office_id, -1, :opc_doc_no, :opc_memo, :opc_from_c_id, :opc_corr_reg_id, :opc_corr_al_id, :cash_al_id, :opc_amount, :oper_name, :un_id, :main_ol_id)
      into :res_main_ol_id, :opc_from_ol_id, :res_text;

      if (exists(select *
               from temp_minus_loan_overpay
               where k_id = :opc_from_k_id)) then -- RFCKG-5505
      begin
        select max(ok_sh_id), max(ok_sh_date), sum(ok_os), sum(ok_perc), sum(ok_tax)
        from operkred
        where ok_from_k_id = :opc_from_k_id and
              ok_ol_is_active = 1 and
              ok_type = 9
        group by ok_from_k_id
        into ok_sh_id, ok_sh_date, temp_ok_os, temp_ok_perc, temp_ok_tax;

        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                              ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                              ok_ol_is_active)
        values (:opc_from_ol_id, -1, :opc_from_k_id, :ok_sh_id, :ok_sh_date, 9, -:temp_ok_os, -:temp_ok_perc, 0,
                1, 0, 0, 0, 0, -:temp_ok_tax, :opc_date, :opc_corr_reg_id, 1);
      end
      else
      begin
        select ok_sh_id, ok_sh_date
        from operkred
        where ok_from_k_id = :opc_from_k_id and
              ok_ol_is_active = 1 and
              ok_type = 9 and
              ok_os < 0
        rows 1
        into ok_sh_id, ok_sh_date;
  
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                              ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                              ok_ol_is_active)
        values (:opc_from_ol_id, -1, :opc_from_k_id, :ok_sh_id, :ok_sh_date, 9, :opc_amount, 0, 0,
                1, 0, 0, 0, 0, 0, :opc_date, :opc_corr_reg_id, 1);
      end
    end else
    if (oper_kind in ('minus.to_kassa', 'minus.cash', 'minus.office', 'minus.to_bank', 'minus.saldo', 'minus.round')) then
    begin
      select res_main_ol_id, ol_id, res_text
      from hp_opercash_other_save(:opc_date, :opc_office_id, -1, :opc_doc_no, :opc_memo, :opc_from_c_id, :opc_corr_reg_id, :opc_corr_al_id, :cash_al_id, :opc_amount, :oper_name, :un_id, :main_ol_id)
      into :res_main_ol_id, :opc_from_ol_id, :res_text;
    end else
    if (oper_kind in ('plus.cash', 'plus.transfer', 'plus.saldo', 'plus.round', 'plus.inkassa', 'plus.from_bank', 'check.saldo', 'check.audit', 'check.close')) then
    begin
      select res_main_ol_id, ol_id, res_text
      from hp_opercash_other_save(:opc_date, :opc_office_id, :opc_sign, :opc_doc_no, :opc_memo, :opc_from_c_id, :opc_corr_reg_id, :cash_al_id, :opc_corr_al_id, :opc_amount, :oper_name, :un_id, :main_ol_id)
      into :res_main_ol_id, :opc_from_ol_id, :res_text;
    end else
    if (oper_kind = 'plus.loan.repay') then
    begin
      select ol_id, error_text
      from hp_do_repayment(:opc_from_k_id, 0, :opc_date, :opc_amount, 1, 1, 1, 1, 1, 0, 0, :opc_office_id, 1, :cash_al_id, 1, :un_id, :opc_from_c_id, :opc_other_client, :opc_manager_id, null)
      into :opc_from_ol_id, :res_text;
    end else
    if (oper_kind = 'plus.loan.cl_repay') then
    begin
      select ol_id, error_text
      from hp_do_repayment(:opc_from_k_id, :opc_from_c_id, :opc_date, :opc_amount, 1, 1, 1, 1, 1, 0, 0, :opc_office_id, 1, :cash_al_id, 1, :un_id, :opc_from_c_id, :opc_other_client, :opc_manager_id, null)
      into :opc_from_ol_id, :res_text;
    end else
    if (oper_kind = 'plus.loan.offpay') then
    begin
      select ol_id, loan_ol_id, error_text
      from hp_do_repayment(:opc_from_k_id, 0, :opc_date, :opc_amount, 1, 1, 1, 1, 1, 0, 0, :opc_office_id, 1, :cash_al_id, 1, :un_id, :opc_from_c_id, :opc_other_client, :opc_manager_id, null)
      into :opc_from_ol_id, :loan_ol_id, :res_text;

      execute procedure hp_opercash_do_operkroff(:loan_ol_id, :opc_amount);
    end else
    if (oper_kind = 'plus.loan.sud') then
    begin
      select res_main_ol_id, ol_id, res_text
      from hp_opercash_other_save(:opc_date, :opc_office_id, :opc_sign, :opc_doc_no, :opc_memo, :opc_from_c_id, :opc_corr_reg_id, :cash_al_id, :opc_corr_al_id, :opc_amount, :oper_name, :un_id, :main_ol_id)
      into :res_main_ol_id, :opc_from_ol_id, :res_text;

      if (res_text = '') then
      begin
        insert into operextra (ox_from_ol_id, ox_ol_date, ox_ol_is_active, ox_object_kind, ox_object_id, ox_from_rd_id,
                               ox_amount_plus, ox_amount_minus, ox_detail, ox_from_reg_id, ox_amount_writeoff)
        values (:opc_from_ol_id, :opc_date, 1, 9, :opc_from_k_id, f_get_rd_id('OperExtraKind.LawClientExpense'),
                0, :opc_amount, 'Оплата', :opc_office_id, 0);
      end
    end else
    if (oper_kind = 'plus.loan.partial') then
    begin
      select res_main_ol_id, ol_id, res_text
      from hp_opercash_other_save(:opc_date, :opc_office_id, :opc_sign, :opc_doc_no, :opc_memo, :opc_from_c_id, :opc_corr_reg_id, :cash_al_id, :opc_corr_al_id, :opc_amount, :oper_name, :un_id, :main_ol_id)
      into :res_main_ol_id, :opc_from_ol_id, :res_text;

      if (res_text = '') then
      begin
        insert into operkontract (ok_from_k_id, ok_from_ol_id, ok_amount, ok_operation_type)
        values (:opc_from_k_id, :opc_from_ol_id, :opc_amount, f_get_rd_id('CashTransactionIncom.PartialPrice'));
      end
    end else
    if (oper_kind = 'plus.loan.konsult') then
    begin
      select res_main_ol_id, ol_id, res_text
      from hp_opercash_other_save(:opc_date, :opc_office_id, :opc_sign, :opc_doc_no, :opc_memo, :opc_from_c_id, :opc_corr_reg_id, :cash_al_id, :opc_corr_al_id, :opc_amount, :oper_name, :un_id, :main_ol_id)
      into :res_main_ol_id, :opc_from_ol_id, :res_text;

      if (res_text = '') then
      begin
        insert into operkontract (ok_from_k_id, ok_from_ol_id, ok_amount, ok_operation_type)
        values (:opc_from_k_id, :opc_from_ol_id, :opc_amount, f_get_rd_id('CashTransactionIncom.Consultation'));

        if (exists(select *
                   from kontract
                   where k_id = :opc_from_k_id and
                         k_cons_kind > 0)) then
          execute procedure hp_opercash_do_konsult(:opc_id, :opc_from_ol_id, 'repayment'); -- доп. проводки при оплате конс. услуг в рассрочку (RFCKG-4900)
      end
    end else
    if (oper_kind = 'plus.loan.insurance') then
    begin
      select res_main_ol_id, ol_id, res_text
      from hp_opercash_other_save(:opc_date, :opc_office_id, :opc_sign, :opc_doc_no, :opc_memo, :opc_from_c_id, :opc_corr_reg_id, :cash_al_id, :opc_corr_al_id, :opc_amount, :oper_name, :un_id, :main_ol_id)
      into :res_main_ol_id, :opc_from_ol_id, :res_text;

      if (res_text = '') then
      begin
        insert into operkontract (ok_from_k_id, ok_from_ol_id, ok_amount, ok_operation_type)
        values (:opc_from_k_id, :opc_from_ol_id, :opc_amount, f_get_rd_id('CashTransactionIncom.Insurance'));

        execute procedure hp_opercash_do_insurance(:opc_id, :opc_from_ol_id); -- доп. проводки
      end
    end else
    if (oper_kind = 'plus.loan.card_kommis') then
    begin
      select cac_id
      from cardaktclient
      where cac_from_k_id = :opc_from_k_id and
            cac_from_c_id = :opc_from_c_id and
            cac_status  = f_enum_id('cardaktclient', 'issued_delayed')
      order by cac_date desc
      rows 1
      into cac_id;

      if (cac_id is null) then
        res_text = 'Не найдена заявка на перевыпуск карты';
      else
      begin
        select res_main_ol_id, ol_id, res_text
        from hp_opercash_other_save(:opc_date, :opc_office_id, :opc_sign, :opc_doc_no, :opc_memo, :opc_from_c_id, :opc_corr_reg_id, :cash_al_id, :opc_corr_al_id, :opc_amount, :oper_name, :un_id, :main_ol_id)
        into :res_main_ol_id, :opc_from_ol_id, :res_text;
      end

      if (res_text = '') then
      begin
        select edl_id
        from hp_edoc_insert('bankcard.mbulak', f_objkind('cardaktclient', 'id'), :cac_id, :un_id)
        into edl_id;
  
        update cardaktclient
        set cac_from_ol_id_com = :res_main_ol_id,
            cac_from_edl_id = :edl_id
        where cac_id = :cac_id;
      end
    end else
    if (oper_kind = 'plus.loan.kommis') then
    begin
      kommis_ol_id = null;

      select ok_from_ol_id
      from operkred, operlist
      where ok_from_k_id = :opc_from_k_id and
            ok_ol_date = :opc_date and
            ok_ol_is_active = 1 and
            ok_kommis < 0 and
            ok_type = 4 and
            ok_from_ol_id = ol_id and
            ol_parent_id = :main_ol_id
      into kommis_ol_id;

      if (kommis_ol_id is not null) then
        opc_from_ol_id = kommis_ol_id;
      else
        res_text = 'После выдачи не найдена операция по комиссии: ' || oper_kind || '  main_ol_id: ' || coalesce(main_ol_id, '<Null>');
    end else
      res_text = 'Неизвестный тип операции: ' || oper_kind;
  
    if (res_text <> '') then
      res_is_error = 1;

    if (res_is_error = 0) then
    begin
      if (opc_from_ol_id is not null) then
      begin
        if (res_text = '') then
          res_text = 'Операция успешно проведена';

        if (res_main_ol_id is null) then
          res_main_ol_id = opc_from_ol_id;

        update opercash
        set opc_from_ol_id = :res_main_ol_id
        where opc_id = :res_opc_id;
  
        if (main_ol_id is null) then
          main_ol_id = res_main_ol_id;

        update operlist
        set ol_doc_no = :opc_doc_no,
            ol_doc_no_str = :opc_doc_no,
            ol_parent_id = :main_ol_id
        where ol_id = :opc_from_ol_id;
      end else
      begin
        res_is_error = 1;
        res_text = 'Метод еще не реализован';
      end
    end

    if (res_is_error = 1) then
    begin
      rdb$set_context('USER_TRANSACTION', 'Opercash_error_text', res_text);
      exception e_message res_text;
    end
  end

  if (opc_date is null) then
  begin
    res_text = 'Операция не найдена либо она связана с другой операцией';

    rdb$set_context('USER_TRANSACTION', 'Opercash_error_text', res_text);
    exception e_message res_text;
  end

  delete from cashprocess
  where cpr_from_opc_id = :opc_id;

  suspend;
end
