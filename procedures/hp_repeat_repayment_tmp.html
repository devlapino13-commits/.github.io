CREATE PROCEDURE HP_REPEAT_REPAYMENT_TMP(
    ORP_ID D_INTEGER,
    K_ID D_ID,
    START_DATE D_DATE,
    REVERSE_DATE D_DATE,
    LOAN_REVERSE D_SMALLINT,
    BOOK_REVERSE D_SMALLINT,
    SQL_TEXT D_STR_8100,
    UN_ID D_ID NOT NULL)
AS
declare variable ol_id d_id;
declare variable ol_date d_date;
declare variable rep_amount d_currency;
declare variable off_amount d_currency;
declare variable opr_is_writeoff d_smallint;
declare variable main_ol_id d_id;
declare variable kc_id d_id;
declare variable tmp_ol_id d_id;
declare variable opr_id d_integer;
declare variable error_text d_str_255;
declare variable reverse_ol_id d_id;
declare variable is_repay d_smallint;
declare variable hq_reg_id d_integer;
declare variable calc_perc_period d_integer;
declare variable calc_perc_period_prv d_integer;
declare variable calc_perc_period_last d_integer;
declare variable max_perc_period d_integer;
declare variable db_id d_integer;
declare variable oks_date_begin d_date;
declare variable need_link d_integer;
declare variable res_text d_str_255;
declare variable main_reverse_ol_id d_id;
declare variable ol_main_memo d_str_long;
declare variable is_transfer d_bool;
declare variable min_sh_date d_date;
declare variable max_sh_date d_date;
declare variable ol_parent_id d_id;
declare variable fact_reg_id d_integer;
declare variable opr_is_linked_loans d_bool;
begin
  -- Процедура пере-принимает погашения по займу начиная с даты START_DATE
  -- Операции сторнирования будут сохранены датой REVERSE_DATE

  if (coalesce(loan_reverse, 0) = 0) then
    loan_reverse = 1;

  if (coalesce(book_reverse, 0) = 0) then
    book_reverse = 2;

  select res_text
  from hp_repeat_repayment_check(:k_id, :start_date, :reverse_date, :loan_reverse, :book_reverse, :sql_text, :un_id)
  into :res_text;

  if (coalesce(res_text, '') <> '') then
    exception e_message res_text;

  db_id = f_get_db_id();
  hq_reg_id = iif(db_id = 41701, 1090100, 1010100);

  delete from temp_operrepeat
  where opr_from_k_id = :k_id;

  -- Собираем все кредитные операции
  for with tab as (
  select ok_ol_date as ol_date, ok_from_ol_id as ol_id,
         -sum(ok_os + ok_perc + ok_fine + ok_tax + ok_kommis + ok_kommis_adv) as rep_amount,
         -sum(iif(ok_type in (5, 6), ok_os + ok_perc + ok_fine + ok_tax + ok_kommis + ok_kommis_adv, 0)) as off_amount,
         max(1) as is_repay,
         max(iif(ok_type = 2 and ol_oper_type = 1 and ol_oper_subtype = 9, 1, 0)) as is_transfer
  from operkred, operlist
  where ok_from_k_id = :k_id and
        ok_from_ol_id = ol_id and
        ok_ol_is_active = 1 and
        ok_ol_date >= :start_date and
        (ok_type in (3, 4, 5, 6, 9) or (ok_type = 2 and ol_oper_type = 1 and ol_oper_subtype = 9))
  group by 1, 2
  union all
  select oko_ol_date, oko_from_ol_id,
         -sum(oko_os + oko_perc + oko_other),
         -sum(iif(oko_type in (5, 6), oko_os + oko_perc + oko_other, 0)),
         max(1),
         max(0)
  from operkroff
  where oko_from_k_id = :k_id and
        oko_ol_is_active = 1 and
        oko_type in (4, 5, 6, 9) and
        oko_ol_date >= :start_date
  group by 1, 2
  union all
  select ox_ol_date, ox_from_ol_id,
         -sum(ox_amount_minus),
         -sum(ox_amount_writeoff),
         max(1),
         max(0)
  from operextra
  where ox_object_kind = 9 and
        ox_object_id = :k_id and
        ox_ol_is_active = 1 and
        ox_ol_date >= :start_date
  group by 1, 2
  union all
  select kob_ol_date, kob_from_ol_id,
         sum(0),
         sum(0),
         max(0),
         max(0)
  from kn_operbook, knclient
  where kc_from_k_id = :k_id and
        kc_id = kob_from_kc_id and
        kob_ol_is_active = 1 and
        kob_ol_date >= :start_date
  group by 1, 2)
  select ol_date, ol_id, sum(rep_amount), sum(off_amount), max(is_repay), max(is_transfer)
  from tab
  group by 1, 2
  order by 1, 2
  into ol_date, ol_id, rep_amount, off_amount, is_repay, is_transfer
  do
  begin
    if (rep_amount < 0 and is_repay = 1) then
    begin
      if (exists(select * from operkred
      where ok_from_ol_id = :ol_id and
            ok_type = 9 and
            ok_os > 0)) then
      Continue;
    end

    if (ol_id = 1010100017208028) then
    begin
      opr_is_writeoff = 0;
    end

    opr_is_writeoff = 0;
    if (off_amount <> 0) then
    begin
      if (rep_amount = off_amount)
      then opr_is_writeoff = 1;
      else
      begin
        rep_amount = rep_amount - off_amount;
        if (rep_amount < 0) then
          exception e_default f_format('Неверная сумма операции и списания, ol_id: %1', ol_id);
      end
    end

    fact_reg_id = null;
    ol_parent_id = null;

    if (is_repay = 1 and is_transfer = 0) then
    begin
      select ol_parent_id, iif(ol_parent_id is not null, ol_parent_id, ol_id)
      from operlist
      where ol_id = :ol_id
      into ol_parent_id, main_ol_id;

      if (not exists(select * from operloan
      where opl_from_ol_id = :main_ol_id)) then
        exception e_default f_format('Operloan record not found, ol_id: %1', ol_id);

      if (ol_parent_id is not null) then
      begin
        select ol_from_reg_id
        from operlist
        where ol_id = :ol_parent_id
        into fact_reg_id;
      end
    end

    if (is_transfer = 1) then
      opr_is_writeoff = -1;

    insert into temp_operrepeat (opr_from_k_id, opr_from_ol_id, opr_date, opr_amount,
                           opr_is_writeoff, opr_is_repay, opr_from_reg_id, opr_is_linked_loans)
    values (:k_id, :ol_id, :ol_date, :rep_amount, :opr_is_writeoff, :is_repay, :fact_reg_id, 0);
  end

  -- Меняем значение чтобы система давала редактировать закрытые периоды
  update company set cm_loan_history_action = 2
  where 1 = 1;

  execute procedure hp_kn_oksummary_kont_recalc(:k_id, :start_date);

  -- Здесь производится необходимые изменения до перепринятия погашений
  if (sql_text <> 'empty') then
    execute statement (sql_text) (id := :k_id);

  select max(kcp_period)
  from kontcalcperc
  where kcp_from_k_id = :k_id
  into max_perc_period;

  calc_perc_period = extract(year from start_date) * 100 + extract(month from start_date);

  delete from kontcalcperc
  where kcp_from_k_id = :k_id and
        kcp_period between :calc_perc_period and :max_perc_period;

  for select distinct oks_date_begin
  from kn_oksummary
  where oks_from_k_id = :k_id and
        oks_date_begin >= :start_date and
        (oks_is_fix_bal_perc <> 0 or oks_is_fix_bal_kommis <> 0 or oks_month_state = 1)
  into oks_date_begin
  do
  begin
    update kn_oksummary
    set oks_need_recalc = 1,
        oks_is_fix_bal_perc = 0,
        oks_is_fix_bal_kommis = 0,
        oks_month_state = 0
    where oks_from_k_id = :k_id and
          oks_date_begin = :oks_date_begin and
          (oks_is_fix_bal_perc <> 0 or oks_is_fix_bal_kommis <> 0 or oks_month_state = 1);
  end

  -- Удаляем все операции займу если прием из другого офиса
  for select opr_from_ol_id
  from temp_operrepeat
  where opr_from_k_id = :k_id and
        opr_from_reg_id is not null
  order by opr_id desc
  into ol_id
  do
  begin
    ol_parent_id = null;
    select ol_parent_id
    from operlist
    where ol_id = :ol_id and
          ol_parent_id is not null
    into ol_parent_id;

    if (ol_parent_id is not null) then
    begin
      update temp_operrepeat
      set opr_is_linked_loans = 1
      where opr_from_k_id = :k_id and
            opr_from_ol_id = :ol_parent_id;

      for select ol_id
      from operlist
      where ol_parent_id = :ol_parent_id and
            ol_id <> :ol_id
      into tmp_ol_id
      do
      begin
        update temp_operrepeat
        set opr_is_linked_loans = 1
        where opr_from_k_id = :k_id and
              opr_from_ol_id = :tmp_ol_id;
      end
    end
  end

  -- Все интересующиеся записи связанные с OL_ID переносим на другой временный, чтобы временно удалить их.
  for select opr_from_ol_id, opr_is_linked_loans
  from temp_operrepeat
  where opr_from_k_id = :k_id
  order by opr_id desc
  into ol_id, opr_is_linked_loans
  do
  begin
    tmp_ol_id = null;
    select result from get_kv_id(5)
    into tmp_ol_id;

    -- Новую временную операцию создаем как Мем.ордер
    insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                          ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                          ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                          ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month, ol_parent_id,
                          ol_manager_c_id, ol_from_rd_id)
    select :tmp_ol_id, ol_from_reg_id, 2, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo, ol_status,
           is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id, ol_is_mo_modified, ol_document_office,
           ol_acc_no_client, ol_acc_no_office, 11, ol_date_oper_day, ol_pocket, ol_doc_client_name,
           ol_is_client, ol_year, ol_month, null, ol_manager_c_id, ol_from_rd_id
    from operlist
    where ol_id = :ol_id;

    update operkred
    set ok_from_ol_id = :tmp_ol_id
    where ok_from_k_id = :k_id and
          ok_from_ol_id = :ol_id;

    update operkroff
    set oko_from_ol_id = :tmp_ol_id
    where oko_from_k_id = :k_id and
          oko_from_ol_id = :ol_id;

    update operextra
    set ox_from_ol_id = :tmp_ol_id
    where ox_object_kind = 9 and
          ox_object_id = :k_id and
          ox_from_ol_id = :ol_id;

    if (opr_is_linked_loans = 0) then
    begin
      update operbook
      set ob_from_ol_id = :tmp_ol_id
      where ob_from_ol_id = :ol_id;
    end

    for select kc_id
    from knclient
    where kc_from_k_id = :k_id
    into kc_id
    do
    begin
      update kn_operbook
      set kob_from_ol_id = :tmp_ol_id
      where kob_from_ol_id = :ol_id and
            kob_from_kc_id = :kc_id;
    end

    update kontstate
    set ks_from_ol_id = :tmp_ol_id
    where ks_from_k_id = :k_id and
          ks_from_ol_id = :ol_id;

    update temp_operrepeat
    set opr_temp_ol_id = :tmp_ol_id
    where opr_from_ol_id = :ol_id;

    update operlist
    set is_delete = 1
    where ol_id = :tmp_ol_id;
  end

  tmp_ol_id = null;
  calc_perc_period = null;
  calc_perc_period_prv = extract(year from start_date) * 100 + extract(month from start_date);
  calc_perc_period_last = extract(year from start_date) * 100 + extract(month from start_date);
  for select opr_id, opr_from_ol_id, opr_date, opr_amount, opr_is_writeoff, opr_is_repay, opr_temp_ol_id,
             opr_from_reg_id
  from temp_operrepeat
  where opr_from_k_id = :k_id
  order by opr_id
  into :opr_id, :ol_id, :ol_date, :rep_amount, :opr_is_writeoff, is_repay, tmp_ol_id, fact_reg_id
  do
  begin
    calc_perc_period = extract(year from ol_date) * 100 + extract(month from ol_date);

    if (calc_perc_period_prv <> calc_perc_period) then
    begin
      if (mod(calc_perc_period, 100) > 1)
      then calc_perc_period_prv = calc_perc_period - 1;
      else calc_perc_period_prv = (cast(left(calc_perc_period, 4) as integer) - 1) * 100 + 12;

      if (calc_perc_period_prv <= max_perc_period) then
      begin
        execute procedure aarep_calc_perc_fill_one(:k_id, :calc_perc_period_last, :calc_perc_period_prv);
        calc_perc_period_last = calc_perc_period;

        calc_perc_period_prv = calc_perc_period;
      end
    end

    if (opr_is_writeoff <> -1) then -- Если это не перенос погашений
    begin
      -- Запоминаем K_ID как подсказка на особую обработку hp_do_repayment
      delete from temp_operrepay
      where topr_from_k_id = :k_id;
  
      insert into temp_operrepay (topr_from_k_id, topr_from_ol_id, topr_fact_reg_id)
      values (:k_id, :ol_id, :fact_reg_id);
  
      error_text = '';
      -- Если это кредитная операция, принимаем повторно
      -- Если имеется запись в таблице temp_operrepay, процедура hp_do_repayment ведет себя по особому: Не создает новую операцию и цепляется на существующую
      if (is_repay = 1) then
      begin
        select error_text
        from hp_do_repayment(:k_id, 0, :ol_date, :rep_amount, 1, 1, 1, 1, 1, 0, :opr_is_writeoff,
             null, null, null, 1, null, null, null, null, null, 0, null)
        into error_text;
      end
  
      -- Очищаем K_ID
      delete from temp_operrepay
      where topr_from_k_id = :k_id;
  
      if (coalesce(error_text, '') <> '') then
        exception e_message 'hp_do_repayment: ' || error_text;
    end else
    begin
      -- Перенос погашений из-за выходных
      min_sh_date = null;
      max_sh_date = null;
      select min(ok_sh_date), max(ok_sh_date)
      from operkred
      where ok_from_ol_id = :tmp_ol_id and
            ok_from_k_id = :k_id and
            ok_type = 2
      into min_sh_date, max_sh_date;

      if (min_sh_date is not null) then
        execute procedure hp_operkred_dd_one_change(:k_id, :ol_id, :min_sh_date, :max_sh_date - 1, :max_sh_date);
    end
  end

  tmp_ol_id = null;

  execute procedure aarep_calc_perc_fill_one(:k_id, :calc_perc_period_last, :max_perc_period);

  -- Откатываем нормализованную дату проводок по ЕПС на дату пересчета
  update knclient
  set kc_last_ol_date = :start_date - 1
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0;

  -- Есть данные до даты сторнирования, необходимо создать главную операцию
  if (exists(select *
  from temp_operrepeat
  where opr_from_k_id = :k_id and
        opr_is_writeoff <> -1 and
        opr_date < :reverse_date)) then
  begin
    if (not exists(select * from kn_oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin = :reverse_date)) then
      execute procedure hp_kn_oksummary_kont_insert(:k_id, :reverse_date);

    ol_main_memo = '';
    select f_format('Сторнирование бух.проводок с %1 по займу %2 от %3',
                    f_datetostr(:start_date), k_kontract_no, k_kontract_date)
    from kontract
    where k_id = :k_id
    into ol_main_memo;

    main_reverse_ol_id = null;
    select kv
    from get_kv(:hq_reg_id, 5)
    into :main_reverse_ol_id;

    insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo,
                          ol_user_id, ol_oper_subtype, ol_date_oper_day)
    values (:main_reverse_ol_id, :hq_reg_id, 1, :reverse_date, 9, :ol_main_memo,
            :un_id, 13, :reverse_date);
  end

  -- Перестраиваем проводки по ЕПС
  execute procedure hp_do_kn_operbook_one_online(:k_id, current_date, null, :start_date - 1);

  for select opr_id, opr_from_ol_id, opr_date, opr_amount, opr_is_writeoff, opr_temp_ol_id, opr_is_linked_loans
  from temp_operrepeat
  where opr_from_k_id = :k_id and
        opr_date < :reverse_date and
        opr_is_repay = 1 and
        opr_is_writeoff <> -1
  order by opr_id desc
  into :opr_id, :ol_id, :ol_date, :rep_amount, :opr_is_writeoff, :tmp_ol_id, opr_is_linked_loans
  do
  begin
    --создаем новый OL_ID для сторнирования
    reverse_ol_id = null;
    select kv
    from get_kv(:hq_reg_id, 5)
    into :reverse_ol_id;

    insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                          ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                          ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                          ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client,
                          ol_parent_id, ol_manager_c_id, ol_from_rd_id)
    select :reverse_ol_id, ol_from_reg_id, ol_oper_type, :reverse_date, ol_doc_no, ol_doc_no_str, ol_doc_kind,
           'Сторнирование операции: ' || coalesce(ol_memo, ''), ol_status, is_delete,
           current_timestamp, :un_id, ol_document_client, ol_from_mo_id, ol_is_mo_modified, ol_document_office,
           ol_acc_no_client, ol_acc_no_office, ol_oper_subtype, :reverse_date, ol_pocket, ol_doc_client_name, ol_is_client,
           :main_reverse_ol_id, ol_manager_c_id, ol_from_rd_id
    from operlist
    where ol_id = :ol_id;

    update temp_operrepeat
    set opr_reverse_ol_id = :reverse_ol_id
    where opr_id = :opr_id;

    insert into reversallist (rl_new_ol_id, rl_old_ol_id, rl_user_id, rl_date, rl_ol_is_delete)
    values (:reverse_ol_id, :ol_id, :un_id, :reverse_date, 0);

    if (opr_is_linked_loans = 0) then
    begin
      insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                            ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                            ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id,
                            ob_short_acc_dt, ob_short_acc_kt)
      select :reverse_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, -ob_summa_val, -ob_summa,
             ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail, ob_subkonto_dt,
             ob_subkonto_kt, ob_from_sac_id, :reverse_date, 1, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt
      from operbook
      where ob_from_ol_id = :tmp_ol_id;

      insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                            ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                            ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id,
                            ob_short_acc_dt, ob_short_acc_kt)
      select :reverse_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa,
             ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail, ob_subkonto_dt,
             ob_subkonto_kt, ob_from_sac_id, :reverse_date, 1, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt
      from operbook
      where ob_from_ol_id = :ol_id;
  
      delete from operbook
      where ob_from_ol_id = :ol_id;
  
      update operbook
      set ob_from_ol_id = :ol_id,
          ob_ol_is_active = 1
      where ob_from_ol_id = :tmp_ol_id;
    end
  end

  if (reverse_date is not null) then
  begin
    -- Переносим все новые проводки до даты сторнирования на дату сторнирования
    for select distinct kob_from_ol_id
    from knclient, kn_operbook
    where kc_from_k_id = :k_id and
          kc_id = kob_from_kc_id and
          kob_ol_date between :start_date and (:reverse_date - 1) and
          kob_ol_is_active = 1
    into ol_id
    do
    begin
      reverse_ol_id = null;

      select opr_reverse_ol_id
      from temp_operrepeat
      where opr_from_ol_id = :ol_id
      into reverse_ol_id;

      if (reverse_ol_id is null) then
      begin
        need_link = 0;
        if (row_count > 0) then
          need_link = 1;

        --создаем новый OL_ID для сторнирования
        select kv
        from get_kv(:hq_reg_id, 5)
        into :reverse_ol_id;
  
        insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                              ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                              ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                              ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client,
                              ol_parent_id, ol_manager_c_id, ol_from_rd_id)
        select :reverse_ol_id, ol_from_reg_id, ol_oper_type, :reverse_date, ol_doc_no, ol_doc_no_str, ol_doc_kind,
               'Коррект. сторнирования: ' || coalesce(ol_memo, ''), ol_status, is_delete,
               current_timestamp, :un_id, ol_document_client, ol_from_mo_id, ol_is_mo_modified, ol_document_office,
               ol_acc_no_client, ol_acc_no_office, ol_oper_subtype, :reverse_date, ol_pocket, ol_doc_client_name, ol_is_client,
               :main_reverse_ol_id, ol_manager_c_id, ol_from_rd_id
        from operlist
        where ol_id = :ol_id;
  
        insert into reversallist (rl_new_ol_id, rl_old_ol_id, rl_user_id, rl_date, rl_ol_is_delete)
        values (:reverse_ol_id, :ol_id, :un_id, :reverse_date, 0);

        if (need_link = 1) then
        begin
          update temp_operrepeat
          set opr_reverse_ol_id = :reverse_ol_id
          where opr_from_ol_id = :ol_id;
        end
      end

      for select kc_id
      from knclient
      where kc_from_k_id = :k_id
      into kc_id
      do
      begin
        update kn_operbook
        set kob_from_ol_id = :reverse_ol_id,
            kob_ol_date = :reverse_date,
            kob_ol_is_active = 1
        where kob_from_ol_id = :ol_id and
              kob_from_kc_id = :kc_id and
              kob_ol_is_active = 1;
      end
    end


    for select opr_id, opr_from_ol_id, opr_date, opr_amount, opr_is_writeoff, opr_temp_ol_id, opr_reverse_ol_id
    from temp_operrepeat
    where opr_from_k_id = :k_id and
          opr_date < :reverse_date
    order by opr_id
    into :opr_id, :ol_id, :ol_date, :rep_amount, :opr_is_writeoff, :tmp_ol_id, reverse_ol_id
    do
    begin
      if (reverse_ol_id is null) then
      begin
        --создаем новый OL_ID для сторнирования
        select kv
        from get_kv(:hq_reg_id, 5)
        into :reverse_ol_id;

        insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                              ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                              ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                              ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client,
                              ol_parent_id, ol_manager_c_id, ol_from_rd_id)
        select :reverse_ol_id, ol_from_reg_id, ol_oper_type, :reverse_date, ol_doc_no, ol_doc_no_str, ol_doc_kind,
               'Сторнирование операции: ' || coalesce(ol_memo, ''), ol_status, is_delete,
               current_timestamp, :un_id, ol_document_client, ol_from_mo_id, ol_is_mo_modified, ol_document_office,
               ol_acc_no_client, ol_acc_no_office, ol_oper_subtype, :reverse_date, ol_pocket, ol_doc_client_name, ol_is_client,
               :main_reverse_ol_id, ol_manager_c_id, ol_from_rd_id
        from operlist
        where ol_id = :ol_id;

        update temp_operrepeat
        set opr_reverse_ol_id = :reverse_ol_id
        where opr_id = :opr_id;

        insert into reversallist (rl_new_ol_id, rl_old_ol_id, rl_user_id, rl_date, rl_ol_is_delete)
        values (:reverse_ol_id, :ol_id, :un_id, :reverse_date, 0);
      end

      for select kc_id
      from knclient
      where kc_from_k_id = :k_id
      into kc_id
      do
      begin
        insert into kn_operbook (kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                                 kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id, kob_ol_is_active,
                                 kob_detail_id, kob_client_dt, kob_client_kt, kob_from_ol_id, kob_from_kc_id)
        select kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, -kob_summa, -kob_summa_val, -kob_summa_tax_dt,
               -kob_summa_tax_kt, :reverse_date, kob_ol_reg_id, 1, kob_detail_id, kob_client_dt, kob_client_kt,
               :reverse_ol_id, kob_from_kc_id
        from kn_operbook
        where kob_from_ol_id = :tmp_ol_id and
              kob_from_kc_id = :kc_id;

        update kn_operbook
        set kob_from_ol_id = :ol_id,
            kob_ol_is_active = 1
        where kob_from_ol_id = :tmp_ol_id;
      end
    end

    for select opr_id, opr_from_ol_id, opr_date, opr_amount, opr_is_writeoff, opr_temp_ol_id,
               opr_reverse_ol_id, opr_is_linked_loans
    from temp_operrepeat
    where opr_from_k_id = :k_id and
          opr_date < :reverse_date and
          opr_is_repay = 0 and
          opr_is_writeoff <> -1
    order by opr_id desc
    into :opr_id, :ol_id, :ol_date, :rep_amount, :opr_is_writeoff, :tmp_ol_id,
         reverse_ol_id, opr_is_linked_loans
    do
    begin
      if (opr_is_linked_loans = 1) then
        Continue;

      if (reverse_ol_id is not null and exists(select * from operbook where ob_from_ol_id = :reverse_ol_id)) then
        Continue;

      --создаем новый OL_ID для сторнирования
      if (reverse_ol_id is null) then
      begin
        select kv
        from get_kv(:hq_reg_id, 5)
        into :reverse_ol_id;

        insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                              ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                              ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                              ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client,
                              ol_parent_id, ol_manager_c_id, ol_from_rd_id)
        select :reverse_ol_id, ol_from_reg_id, ol_oper_type, :reverse_date, ol_doc_no, ol_doc_no_str, ol_doc_kind,
               'Сторнирование операции: ' || coalesce(ol_memo, ''), ol_status, is_delete,
               current_timestamp, :un_id, ol_document_client, ol_from_mo_id, ol_is_mo_modified, ol_document_office,
               ol_acc_no_client, ol_acc_no_office, ol_oper_subtype, :reverse_date, ol_pocket, ol_doc_client_name, ol_is_client,
               :main_reverse_ol_id, ol_manager_c_id, ol_from_rd_id
        from operlist
        where ol_id = :ol_id;

        update temp_operrepeat
        set opr_reverse_ol_id = :reverse_ol_id
        where opr_id = :opr_id;

        insert into reversallist (rl_new_ol_id, rl_old_ol_id, rl_user_id, rl_date, rl_ol_is_delete)
        values (:reverse_ol_id, :ol_id, :un_id, :reverse_date, 0);
      end
  
      insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                            ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                            ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id,
                            ob_short_acc_dt, ob_short_acc_kt)
      select :reverse_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, -ob_summa_val, -ob_summa,
             ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail, ob_subkonto_dt,
             ob_subkonto_kt, ob_from_sac_id, :reverse_date, 1, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt
      from operbook
      where ob_from_ol_id = :tmp_ol_id;
  
      insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                            ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                            ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id,
                            ob_short_acc_dt, ob_short_acc_kt)
      select :reverse_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa,
             ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail, ob_subkonto_dt,
             ob_subkonto_kt, ob_from_sac_id, :reverse_date, 1, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt
      from operbook
      where ob_from_ol_id = :ol_id;
  
      delete from operbook
      where ob_from_ol_id = :ol_id;
  
      update operbook
      set ob_from_ol_id = :ol_id,
          ob_ol_is_active = 1
      where ob_from_ol_id = :tmp_ol_id;
    end
  end

  for select opr_id
  from temp_operrepeat
  where opr_from_k_id = :k_id
  into :opr_id
  do
  begin
    delete from temp_operrepeat
    where opr_id = :opr_id;
  end

  if (orp_id is not null and main_reverse_ol_id is not null) then
  begin
    update operrepeat
    set orp_from_ol_id = :main_reverse_ol_id
    where orp_id = :orp_id;
  end

  update company set cm_loan_history_action = 0
  where 1 = 1;
end
