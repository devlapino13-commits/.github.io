CREATE PROCEDURE CM_RU_0116_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable KC_FROM_C_ID ID_TYPE;
declare variable K_FROM_KF_ID ID_TYPE;
declare variable K_ID_OLD ID_TYPE;
declare variable K_CIRCLE integer;
begin
  /*«Семья плюс» остается только для повторных групп, чей состав не изменился, но мог уменьшится.  Пункт 2.1 Параметров*/
  select K_DISB_DATE, K_FROM_KF_ID, K_CIRCLE
  from KONTRACT, PRODUCT
  where K_ID = :K_ID and
        K_FROM_PR_ID = PR_ID and
        PR_CODE containing '.FAMILY PLUS.'
  into KF_DISB_DATE, K_FROM_KF_ID, K_CIRCLE;
  if (K_FROM_KF_ID is null) then
    exit;
  select K_ID
  from KONTRACT K, PRODUCT PR, KONTSTATE KS
  where K_FROM_KF_ID = :K_FROM_KF_ID and
        K_FROM_PR_ID = PR_ID and
        PR_CODE containing '.FAMILY PLUS.' and
        PR_CODE not containing '.PAR.' and
        K_ID = KS_FROM_K_ID and
        KS_OL_IS_DELETE = 0 and
        KS_STATE = 1 and
        KS_DATE_BEGIN < :KF_DISB_DATE  
        order by KS_DATE_BEGIN desc rows 1
  into K_ID_OLD;
  if (K_CIRCLE < 2 or K_ID_OLD is null) then
  begin
    res_error_is_critical = 1;
    res_error_text = 'Группа ранее не получала займ по данному продукту!';
    suspend;
    exit;
  end
  for select KC_FROM_C_ID, KC_AMOUNT_APPROVED
      from KNCLIENT
      where KC_FROM_K_ID = :K_ID and
            KC_AMOUNT_APPROVED > 0
      into KC_FROM_C_ID, KC_AMOUNT
  do
  begin
    if (not exists(select *
                   from KNCLIENT KC
                   where KC_FROM_K_ID = :K_ID_OLD and
                         KC_FROM_C_ID = :KC_FROM_C_ID and
                         KC_AMOUNT_APPROVED > 0)) then
    begin
      select C_NAME_FIO, C_PASS_NO
      from CLIENTS
      where C_ID = :KC_FROM_C_ID
      into res_info_text, KF_CLIENT_PASS_NO;
      res_error_is_critical = 1;
      res_error_text = 'Клиент ранее не получал займ в составе этой группы!';
      suspend;
    end
  end
end
