CREATE PROCEDURE HP_EXPRESSAPP_FILL_ALL(
    REP_DATE DATE,
    NEED_INSERT D_SMALLINT = 0,
    SHOW_ONLY_VALID D_SMALLINT = 0,
    MAX_ROW D_INTEGER = 0)
RETURNS (
    C_ID ID_TYPE,
    CHECK_DOCPRLOAN SMALLINT,
    CS_INFO_STATE SMALLINT,
    RES D_SMALLINT,
    RES_TEXT STR_100,
    IS_GOOGLE_ID D_SMALLINT)
AS
declare variable REC_COUNT D_INTEGER;
declare variable IS_ACTIVE D_SMALLINT;
declare variable LAST_K_ID ID_TYPE;
begin
  rec_count = 0;

  -- вытаскиваем всех клиентво исключая дубликатов
  for
  select kc_from_c_id,
         max(iif(ks_state = 1, 1, 0))
  from kontstate, knclient
  where ks_ol_is_delete = 0
  and :rep_date between ks_date_begin and ks_date_end
  and ks_state in (1,2)
  and ks_from_k_id = kc_from_k_id
  and kc_amount_approved > 0
  group by kc_from_c_id
  into c_id, is_active
  do
  begin
    if (exists(select * from expressapp
    where exa_from_c_id = :c_id)) then
      Continue;

    -- проверка по документам имеется ли соответствии
    if (f_docprloan_valid(:c_id,146,0,:rep_date) = -1)
    then check_docprloan = 0; -- 0 - не подходит по документам;
    else check_docprloan = 1; -- 1 - соответсвует по документам

    -- после выдачи по документам данные менялись или нет
    cs_info_state = null;
    select cs_info_state from clientstate
    where cs_from_c_id = :c_id
    into cs_info_state; -- 0 - Данные менялись; 1 - Данные не изменились

    if (show_only_valid = 1 and (check_docprloan = 0 or cs_info_state = 0)) then
      Continue;

    if (show_only_valid = -1 and (check_docprloan = 1 and cs_info_state = 1)) then
      Continue;

    --проверка GOOGLE ID
    is_google_id = 0;
    if (exists(select * from mc_login
    where mcl_from_c_id = :c_id
    and coalesce(mcl_google_id, '') <> '')) then is_google_id = 1;
    if (is_google_id = 0) then
       Continue;

    res = null;
    res_text = null;
    for select res, res_text
    from hp_expressapp_fill_one(:c_id, :rep_date, :need_insert, :is_active)
    where res = 1
    into :res, :res_text
    do
    begin
      rec_count = rec_count + 1;
      suspend;

      if (max_row > 0 and rec_count >= max_row) then
        Exit;
    end
  end
end
