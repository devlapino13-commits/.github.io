CREATE PROCEDURE AAREP_DELINQ_CLOSE(
    REP_REGION INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KS_FROM_OL_ID DECIMAL(18,0),
    KS_FROM_K_ID DECIMAL(18,0),
    CLOSE_DATE DATE,
    K_KONTRACT_NO VARCHAR(20),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_DISB_DATE DATE,
    KF_NAME VARCHAR(100),
    REG_ID INTEGER,
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    OL_DATETIME TIMESTAMP,
    DELINQ_DATE DATE,
    OK_OS DECIMAL(18,4),
    OK_PERC DECIMAL(18,4),
    OK_FINE DECIMAL(18,4),
    DELINQ_DAY INTEGER)
AS
declare variable parent integer;
begin
  delinq_day = 0;
  for select parent, ks_from_ol_id, ks_from_k_id, ks_date_begin, k_kontract_no, k_amount_approved, k_disb_date,
             kal_new_kf_name, kal_new_reg_id, reg_code || ' ' || reg_name, parent_name, ol_datetime, min(ok_sh_date),
             sum(ok_os) * -1, sum(ok_perc) * -1, sum(ok_fine) * -1

      from kontstate, knaktlist, kontract, region, region_list(:rep_region, 1, 3, 1), operkred, operlist
      where ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_ol_is_delete = 0 and
            ks_state = 2 and
            k_id = ks_from_k_id and
            k_from_kf_id = kal_from_kf_id and
            ks_date_begin between kal_date and kal_date_end and
            kal_new_reg_id = reg_unicode and
            (kal_new_reg_id = id_list) and
            ok_from_ol_id = ks_from_ol_id and
            ol_id = ks_from_ol_id and
            operlist.is_delete = 0 and
            ok_from_c_id = 0 and
            ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
      group by parent, ks_from_ol_id, ks_from_k_id, ks_date_begin, k_kontract_no, k_amount_approved, k_disb_date, kal_new_kf_name, kal_new_reg_id, reg_code, reg_name, parent_name, ol_datetime
      having sum(ok_fine) <> 0 and min(ok_sh_date) >= :rep_date_begin
      order by parent, kal_new_reg_id, ks_date_begin

      into :parent, :ks_from_ol_id, :ks_from_k_id, :close_date, :k_kontract_no, :k_amount_approved, :k_disb_date,
           :kf_name, :reg_id, :reg_name, :parent_name, :ol_datetime, :delinq_date, :ok_os, :ok_perc, :ok_fine

  do
  begin
    delinq_day = :close_date - :delinq_date;
    suspend;
  end
end
