CREATE PROCEDURE HP_CHECK_RESTRUCT_CONDITIONS(
    K_ID D_ID NOT NULL)
RETURNS (
    IS_RESTRUCTURING_ALLOWED D_BOOL)
AS
declare variable kont_has_delinq d_bool;
declare variable kont_in_legal d_bool;
declare variable kont_has_court_fee d_bool;
declare variable rep_date d_date;
begin
    -- Процедура проверяет, возможна ли реструктуризация контракта

    /*
        УСЛОВИЯ ДЛЯ БЛОКИРОВКИ РЕСТРУКТУРИЗАЦИИ (RFCKZ-705)
    
        1. По данному контракту есть активная просрочка
        2. Контракт передан в юридический отдел
        3. Компания уже понесла расходы для принудительного взыскания, то есть, по контракту есть долг по
           госпошлине
    
        Все 3 условия должны соблюдаться одновременно для блокировки возможности реструктуризации
    */

    is_restructuring_allowed = 1;
    kont_has_delinq = 0;
    kont_in_legal = 0;
    kont_has_court_fee = 0;
    rep_date = current_date;

    -- Проверяем, был ли блок на реструктуризацию снят. Если да, то проверять следующие условия нет смысла
    if (exists(select *
               from restruct_block_cancel rbc
               where rbc.rbc_from_k_id = :k_id and
                     :rep_date between rbc.rbc_date_begin and rbc.rbc_date_end)
    ) then
    begin
        is_restructuring_allowed = 1;
        suspend;
        exit;
    end

    -- УСЛОВИЕ №1. По данному контракту есть активная просрочка
    if (exists(select *
               from oksummary oks
               where oks.oks_from_k_id = :k_id and
                     :rep_date between oks.oks_date_begin and oks.oks_date_end and
                     oks.oks_close_state = 0 and
                     :rep_date - oks.oks_delinq_date > 0)
    ) then
        kont_has_delinq = 1;

    -- УСЛОВИЕ №2. Контракт передан в юридический отдел
    if (exists(select *
               from sb_akt sba
               inner join sb_list sbl on sba.sba_from_sbl_id = sbl.sbl_id
               where sba.sba_from_k_id = :k_id and
                     :rep_date between sba.sba_date_begin and sba.sba_date_end and
                     sba.sba_status = 2 and
                     sba.sba_is_delete = 0 and
                     sbl.sbl_name = 'Юридический Отдел')
    ) then
        kont_in_legal = 1;
        
    -- УСЛОВИЕ №3. Компания уже понесла расходы для принудительного взыскания, то есть, по контракту есть долг по госпошлине
    if (exists(select *
               from hp_repay_extra_saldo(:k_id, 0, :rep_date)
               where res_amount_saldo > 0)
    ) then
        kont_has_court_fee = 1;
        
    -- Если все 3 условия соблюдаются, блокируем возможность реструктуризации
    if (kont_has_delinq = 1 and kont_in_legal = 1 and kont_has_court_fee = 1) then
        is_restructuring_allowed = 0;

    suspend;
end
