CREATE PROCEDURE HP_CREDITOR_REP_DELINQ_SMS
AS
declare variable k_id d_id;
  declare variable edl_id d_id;
  declare variable res_text d_str_255;

  declare variable v_tz_offset integer;
  declare variable v_hour integer;
  declare variable v_local_hour integer;
BEGIN
  if (f_get_db_id() <> 81001) then
    exit;

  for
    select t.cr_k_id,
           t.cr_tz_offset
    from creditor_rep_list t
    where t.cr_status = 1
      and not exists (
      select 1
      from edoc_list edl
      inner join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id
      where edl.edl_object_id = t.cr_k_id
        and edl.edl_object_kind = f_objkind('kontract','id')
        and edc.edc_code = 'loan.creditor_rep')

       -- проверка сумм
    and not exists (
      select 1
      from print_kont_dopinfo(t.cr_k_id) p
      where abs(coalesce(t.cr_sum_os,0) - coalesce(p.kdi_zad_os_now,0)) > 0.01
         or abs(coalesce(t.cr_sum_perc,0) - coalesce(p.kdi_calc_perc_now,0)) > 0.01
         or abs(coalesce(t.cr_sum_all,0) - coalesce(p.kdi_sum_total_now,0)) > 0.01)
      into :k_id, :v_tz_offset
  do
  begin
    -- текущее время сервера
    v_hour = extract(hour from current_timestamp);

    -- локальное время клиента
    v_local_hour = mod((v_hour + (coalesce(v_tz_offset, 3) - 3) + 24), 24);

    --  отправляем только в "разрешённое окно"
    if (v_local_hour between 10 and 18) then
    begin
      select edl_id, res_text
      from hp_edoc_ins('loan.creditor_rep', f_objkind('kontract', 'id'), :k_id, 1000000000000)
      into :edl_id, :res_text;
    end
  end
END
