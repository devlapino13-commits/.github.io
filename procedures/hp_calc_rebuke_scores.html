CREATE PROCEDURE HP_CALC_REBUKE_SCORES(
    REP_DATE_END D_DATE,
    C_ID D_ID,
    HIST_DATETIME D_DATETIME = null)
RETURNS (
    SCORES_WAS_ADDED D_INTEGER,
    SCORES_WAS_USED D_INTEGER,
    SCORES_SUM D_INTEGER,
    ZEROED_SCORES D_INTEGER)
AS
declare variable scores_tmp d_integer;
declare variable un_id d_id;
begin
  scores_sum = 0;
  scores_was_used = 0;
  zeroed_scores = 0;

  if (hist_datetime is null) then
  begin
    select coalesce(sum(rs_scores), 0) as scores_sum
    from rebuke_scores rs
    inner join rebuke rb on rb.rb_id = rs.rs_from_rb_id and
          rb.rb_is_delete = 0 and
          rb.rb_date <= :rep_date_end
    where rs.rs_from_c_id = :c_id and
          rs.rs_from_rb_id is not null
    into :scores_was_added;

    for select rs.rs_scores, rs.rs_from_un_id
        from rebuke_scores rs
        where rs.rs_from_c_id = :c_id and
              rs.rs_from_rb_id is null and
              rs.rs_is_delete = 0 and
              rs.rs_date <= :rep_date_end
        into :scores_tmp, :un_id
    do
    begin
      if (:un_id = 1000000000000) then
        zeroed_scores = :zeroed_scores + :scores_tmp;
      scores_was_used = :scores_was_used + :scores_tmp;
    end
  end
  else
  begin
    select coalesce(sum(rs_scores), 0) as scores_sum
    from rebuke_scores rs
    inner join rebuke rb on rb.rb_id = rs.rs_from_rb_id and
          rb.rb_is_delete = 0 and
          rb.rb_date <= :rep_date_end and
          rs.rs_datetime < :hist_datetime
    where rs.rs_from_c_id = :c_id and
          rs.rs_from_rb_id is not null
    into :scores_was_added;

    for select rs.rs_scores, rs.rs_from_un_id
        from rebuke_scores rs
        where rs.rs_from_c_id = :c_id and
              rs.rs_from_rb_id is null and
              rs.rs_is_delete = 0 and
              rs.rs_date <= :rep_date_end and
              rs.rs_datetime < :hist_datetime
        into :scores_tmp, :un_id
    do
    begin
      if (:un_id = 1000000000000) then
        zeroed_scores = :zeroed_scores + :scores_tmp;
      scores_was_used = :scores_was_used + :scores_tmp;
    end
  end

  scores_sum = :scores_was_added + :scores_was_used;
  suspend;
end
