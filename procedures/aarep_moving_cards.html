CREATE PROCEDURE AAREP_MOVING_CARDS(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    CAM_ID D_INTEGER,
    CAM_DATE D_DATE,
    REGS_NAME D_STR_CODE,
    CS_NAME_FIO D_STR_100,
    REGD_NAME D_STR_CODE,
    CD_NAME_FIO D_STR_100,
    CD_PIN_NAME_FIO D_STR_100,
    KIND_AKT D_STR_100,
    FIRST_CARD_ID D_INTEGER,
    LAST_CARD_ID D_INTEGER,
    CARD_COUNT D_INTEGER)
AS
declare variable is_pincover d_bool;
begin
  for
    select
      cam_id,
      cam_date,
      regs_name,
      cs_name_fio,
      regd_name,
      cd_name_fio,
      cd_pin_name_fio,
      kind_akt
    from hp_cardaktmain_view(null, :rep_date_begin, :rep_date_end, :rep_region_id)
    where cam_status = 1
  into
    :cam_id,
    :cam_date,
    :regs_name,
    :cs_name_fio,
    :regd_name,
    :cd_name_fio,
    :cd_pin_name_fio,
    :kind_akt
  do
  begin
    for
      select
        car_first_card_id,
        car_last_card_id,
        car_card_count
      from cardaktrange
      where car_from_cam_id = :cam_id
    into
      :first_card_id,
      :last_card_id,
      :card_count
    do
    begin
      suspend;
    end
  end

  cd_pin_name_fio = null;
  kind_akt = 'выдача карты клиенту';
  for
    select
      cac_id,
      cac_date,
      card_id,
      reg_name,
      un_fio,
      c_name_fio,
      is_pincover
    from hp_cardaktclient_view(null, :rep_date_begin, :rep_date_end, :rep_region_id)
    where cac_status >= 4
  into
    :cam_id,
    :cam_date,
    :first_card_id,
    :regs_name,
    :cs_name_fio,
    :cd_name_fio,
    :is_pincover
  do
  begin
    regd_name = regs_name;
    if (is_pincover = 1) then
      select
        c_name_fio
      from cardlist
        left join cardplace
          on cpl_id = cal_last_pin_cpl_id
        left join clients
          on c_id = cpl_from_c_id
      where cal_card_id = :first_card_id
      into
        :cd_pin_name_fio;
    last_card_id = first_card_id;
    card_count = 1;
    suspend;
  end
end
