CREATE PROCEDURE HP_BKI_GET_VALUELIST4EVENT(
    MAIN_NODE D_STR_CODE,
    XML_VERSION_ID D_ID,
    JSO D_INTEGER)
RETURNS (
    BFV_ID D_ID,
    DB_FIELD D_STR_CODE,
    I D_INTEGER,
    VAR_VALUE D_STR_8100)
AS
declare variable jso_tmp d_integer;
declare variable default_value d_str_255;
declare variable forced_value d_str_255;
begin
  for with recursive tab as(
    select bfv_id, bfv_position, bki_code, bki_name, bki_db_field, bxf_xml_node, bxf_default_value, bxf_id, bfv_parent_id, bfv_forced_value
    from bki_xml_fields_ver, bki_xml_fields, bki_indicators
    where bxf_id = bfv_from_bxf_id
      and bki_code = bxf_from_bki_code
      and bfv_from_bxv_id = :xml_version_id
    order by bfv_position
  ), t1 (id, pos, code, name, db, xml, def, xml_id, jso, ind, forced_value) as(
    select bfv_id, bfv_position, bki_code, bki_name, bki_db_field, bxf_xml_node, bxf_default_value, bxf_id, :jso, cast(null as d_integer), bfv_forced_value
    from tab
    where bki_code = :main_node
    union all
    select bfv_id, bfv_position, bki_code, bki_name, bki_db_field, bxf_xml_node, bxf_default_value, bxf_id, t1.jso, ind, bfv_forced_value
    from t1, tab
    where bfv_parent_id = id
      and coalesce(bfv_forced_value, bki_db_field) is distinct from '<ARR>'
    union all
    select bfv_id, bfv_position, bki_code, bki_name, bki_db_field, bxf_xml_node, bxf_default_value, bxf_id, hp.jso, i, bfv_forced_value
    from t1, tab, hp_bki_json_get_list(json.obj_get_arr(jso, bki_code), 1) hp
    where bfv_parent_id = id
      and coalesce(bfv_forced_value, bki_db_field) = '<ARR>'
  )
  select id, db, jso, ind, def, forced_value
  from t1
  where coalesce(db, '') not starting with '<'
  into bfv_id, db_field, jso_tmp, i, default_value, forced_value
  do
  begin
    if (forced_value starting with '<' and forced_value <> '<NULL>') then
      db_field = forced_value;

    else if (coalesce(forced_value, '') <> '') then
      var_value = case
                    when forced_value = '<NULL>' then  null
                    when forced_value starting with '#' then json.obj_get_val(jso, substring(forced_value from 2))
                    else forced_value
                  end;
    else if (db_field is null) then
      var_value = default_value;
    else
      var_value = json.obj_get_val(jso_tmp, db_field);

    suspend;
  end
end
