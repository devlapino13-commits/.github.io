CREATE PROCEDURE AAREP_REPORT_ZB(
    OBEG INTEGER,
    OEND INTEGER)
RETURNS (
    KC_ID D_ID,
    C_NAME_FIO VARCHAR(300),
    REP_QUERY_DATE D_DATE,
    REP_DATE_BEGIN D_DATE,
    C_PASS_NUMBER D_STR_255,
    C_PASS_DATE D_DATE,
    C_BIRTH_DATE D_DATE,
    C_TYPE D_STR_CODE,
    C_INN D_STR_CODE,
    KC_CIRCLE D_INTEGER,
    METHOD_ISSUE D_STR_100,
    K_KONTRACT_NO D_STR_20,
    K_KONTRACT_DATE D_DATE,
    K_DISB_DATE D_DATE,
    K_AMOUNT D_CURRENCY,
    OKS_KONT_LAST_DATE D_DATE,
    K_KONT_LAST_DATE D_DATE,
    STR_61 D_STR_10,
    TYPE_PSK D_STR_10,
    CIRCLE D_STR_10,
    KC_PDN_KOEF D_CURRENCY,
    TERM_IN_DAYS D_INTEGER,
    Z_COST D_CURRENCY,
    PR_PRODUCT_KIND D_INTEGER,
    STR_29 D_CURRENCY,
    K_LOAN_KIND D_STR_CODE,
    K_DISB_PLACE D_STR_255,
    K_KREDLINE_END D_DATE,
    OKS_DATE_BEGIN D_DATE,
    BRT_STATUS_DATE D_DATE,
    WRITEOFF_DATE D_DATE,
    PR_NAME D_STR_64,
    KS_DATE_BEGIN D_DATE,
    KPL_CREDID D_STR_CODE,
    CRED_FULL_COST D_CURRENCY,
    K_RESTRUCT_DATE D_DATE,
    KOBS_DATE D_DATE,
    K_PERCENT D_CURRENCY,
    Z_SECURED D_STR_100,
    KOBS_DATE_LAST D_DATE,
    ST_75 D_DATE,
    BRT_ZALOG_FIO D_STR_100,
    C_PASS_NUMBER_NEW_Z D_STR_255,
    C_PASS_DATE_Z D_DATE,
    OKS_FIRST_DELINQ_90 D_DATE,
    LAST_DATE D_DATE,
    OKS_DELINQ_START_180 D_DATE,
    STR_ID D_INTEGER,
    PERC_STR D_STR_CODE,
    PERCENT_YEAR_R DECIMAL(15,3),
    SH_PERC D_CURRENCY,
    DATE_RES_REF D_DATE,
    ST65 D_STR_20,
    ST_73 D_INTEGER,
    KC_FINAL_INCOME D_CURRENCY,
    KC_STAT_AVG_INCOME D_CURRENCY,
    KC_ANKETA_INCOME D_CURRENCY,
    END_PDN CURR_TYPE,
    ITOGI_AMP D_CURRENCY,
    AMP_SUM_NEW D_CURRENCY)
AS
declare variable k_id bigint;
begin
  --rec_no = 0;
  for select k_id, kc_id, rep_date_begin, rep_query_date
      from rep_ru_cbank
      --   where line_no>=:obeg and line_no<=:oend
      into :k_id, :kc_id, :rep_date_begin, :rep_query_date
  do
  begin
    select c_name_fio, c_pass_number_new, c_pass_date, c_birth_date, c_type, c_inn, kc_circle, k_loan_kind,
           method_issue, k_disb_place, k_kontract_no, k_kontract_date, k_disb_date, k_amount, oks_kont_last_date,
           k_kont_last_date, pr_product_kind, k_kredline_end, oks_date_begin, ks_date_begin, writeoff_date,
           brt_status_date, pr_name, kpl_credid, k_percent, cred_full_cost, k_restruct_date, kobs_date, z_cost,
           z_secured, kobs_date_last, brt_zalog_fio, c_pass_number_new_z, c_pass_date_z, oks_first_delinq_90, last_date,
           oks_delinq_start_180, str_id, perc_str, percent_year_r, sh_perc, date_res_ref,
           itogi_amp_our_sum + itogi_amp_ext_sum, amp_sum_new, kc_stat_avg_income, kc_anketa_income, kc_final_income, end_pdn

    from aarep_centerbank_report_kc_2025(:k_id, :rep_query_date)
    into :c_name_fio, :c_pass_number, :c_pass_date, :c_birth_date, :c_type, :c_inn, :kc_circle, :k_loan_kind,
         :method_issue, :k_disb_place, :k_kontract_no, :k_kontract_date, :k_disb_date, :k_amount, oks_kont_last_date,
         :k_kont_last_date, :pr_product_kind, :k_kredline_end, :oks_date_begin, :ks_date_begin, :writeoff_date,
         :brt_status_date, :pr_name, :kpl_credid, :k_percent, :cred_full_cost, :k_restruct_date, :kobs_date, :z_cost,
         :z_secured, :kobs_date_last, :brt_zalog_fio, :c_pass_number_new_z, :c_pass_date_z, :oks_first_delinq_90,
         :last_date, :oks_delinq_start_180, :str_id, :perc_str, :percent_year_r, :sh_perc, :date_res_ref, :itogi_amp,
         :amp_sum_new, :kc_stat_avg_income, :kc_anketa_income, :kc_final_income, end_pdn;
    if (kc_circle = 1) then
      circle = 'первичный';
    else
    if (kc_circle > 1) then
      circle = 'повторный';

    select kc_pdn_koef
    from knclient_pdn
    where kc_id = :kc_id and
          kcp_kind = 0
    into kc_pdn_koef;

    term_in_days = 0;
    term_in_days = datediff(day, k_disb_date, oks_kont_last_date);
    str_61 = '';
    str_61 = case
               when k_amount <= 30000 and term_in_days <= 30 then 'А2'
               when kc_pdn_koef > 50 and kc_pdn_koef <= 80 then 'А3'
               when kc_pdn_koef > 80 then 'А4'
               else ''
             end;

    type_psk = '';
    type_psk = case
                 when term_in_days >= 0 and term_in_days <= 365 and k_amount <= 10000000 and z_cost > 0 and (pr_product_kind <> 1) then '2.1'
                 when term_in_days >= 366 and term_in_days <= 10000000 and k_amount <= 1000000 and z_cost > 0 and (pr_product_kind <> 1) then '2.1'
                 when term_in_days >= 0 and term_in_days <= 30 and k_amount <= 30000 and z_cost = 0 then '2.3.1.1'
                 when term_in_days >= 0 and term_in_days <= 30 and k_amount >= 30001 and k_amount <= 10000000 and z_cost = 0 then '2.3.1.2'
                 when term_in_days >= 31 and term_in_days <= 60 and k_amount <= 30000 and z_cost = 0 then '2.3.2.1'
                 when term_in_days >= 31 and term_in_days <= 30 and k_amount >= 30001 and k_amount <= 10000000 and z_cost = 0 then '2.3.2.2'
                 when term_in_days >= 61 and term_in_days <= 180 and k_amount <= 30000 and z_cost = 0 then '2.3.3.1'
                 when term_in_days >= 61 and term_in_days <= 180 and k_amount >= 30001 and k_amount <= 100000 and z_cost = 0 then '2.3.3.2'
                 when term_in_days >= 61 and term_in_days <= 180 and k_amount >= 100001 and k_amount <= 10000000 and z_cost = 0 then '2.3.3.3'
                 when term_in_days >= 181 and term_in_days <= 365 and k_amount <= 100000 and z_cost = 0 then '2.3.4.1'
                 when term_in_days >= 181 and term_in_days <= 365 and k_amount >= 100001 and k_amount <= 10000000 and z_cost = 0 then '2.3.4.2'
                 when term_in_days >= 365 and term_in_days <= 10000000 and k_amount <= 100000 and z_cost = 0 then '2.3.5.1'
                 when term_in_days >= 365 and term_in_days <= 10000000 and k_amount >= 100001 and k_amount <= 10000000 and z_cost = 0 then '2.3.5.2'
                 else ''
               end;

    if ((:k_disb_date >= :rep_date_begin) and
        (:k_disb_date <= :rep_query_date)) then
      str_29 = k_amount;
    ------------------------------------------
    if (brt_status_date > :ks_date_begin) then
      brt_status_date = null;
    if (brt_status_date > :writeoff_date) then
      brt_status_date = null;

    if (k_kredline_end = k_kont_last_date) then
      k_kredline_end = null;
    if (writeoff_date is not null) then
      ks_date_begin = writeoff_date;

    if (ks_date_begin > '31.12.2024') then
      ks_date_begin = null;
    if (writeoff_date > '31.12.2024') then
      writeoff_date = null;

    if ((k_restruct_date is not null) and
        (k_kredline_end is not null)) then
      last_date = k_restruct_date;
    else
    if ((k_restruct_date is null) and
        (k_kredline_end is null) and
        (oks_first_delinq_90 is not null)) then
      last_date = oks_first_delinq_90;
    else
      last_date = null;

    if (oks_date_begin > brt_status_date) then
      brt_status_date = oks_date_begin;

    st_75 = null;
    if (k_disb_date >= '01.10.2019') then
      st_75 = k_disb_date;
    -----------------------------------------------

    if ((date_res_ref <> k_disb_date) and
        (date_res_ref is not null)) then
      st65 = 'иное основание';

    -------------------------------------------------
    st_73 = 0;
    if (kc_final_income = kc_stat_avg_income) then
      st_73 = 13;
    if (kc_final_income = kc_anketa_income) then
      st_73 = 12;
    suspend;
  end

end
