CREATE PROCEDURE AAREP_DELINQ_KONTRACTS_INFO(
    REP_REGION_ID ID_TYPE,
    REP_DATE DATE,
    DELINQ_PLUS_ONE SMALLINT)
RETURNS (
    SBA_STATUS_STR D_STR_10,
    REG_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(50),
    K_KF_NAME VARCHAR(100),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED CURR_TYPE,
    BALANCE_KONTRACT CURR_TYPE,
    DELINQ_OS CURR_TYPE,
    DELINQ_CALC_PERC CURR_TYPE,
    DELINQ_SH_PERC CURR_TYPE,
    FINE_BALANCE CURR_TYPE,
    DELINQ_DATE DATE,
    DELINQ_DAYS INTEGER,
    DELINQ_DATE_OS DATE,
    DELINQ_DAYS_OS INTEGER,
    DELINQ_DATE_PERC DATE,
    DELINQ_DAYS_PERC INTEGER,
    TOTAL_BALANCE CURR_TYPE,
    K_PERCENT_PER_DAY PERC_TYPE,
    K_PERCENT_EFFECT PERC_TYPE,
    K_CASHIER_FIO VARCHAR(100),
    K_KREDLINE_END DATE,
    TYPE_PROD STR_10,
    MAIN_NUMBER D_STR_100,
    ADD_NUMBER D_STR_100,
    K_SPF_ORIGIN STR_100,
    K_SPF_IS_ACTIVE STR_10)
AS
declare variable K_ID ID_TYPE;
declare variable REG_ID ID_TYPE;
declare variable REG_ID_TMP ID_TYPE;
declare variable ZAD_OS CURR_TYPE;
declare variable SH_CALC_PERC CURR_TYPE;
declare variable ZAD_PERC CURR_TYPE;
declare variable SH_CALC_FINE CURR_TYPE;
declare variable SH_DATE date;
declare variable SB_DATE_BEGIN D_DATE;
declare variable SBA_STATUS D_INTEGER;
declare variable C_ID D_ID;
declare variable VIPS_ID D_ID;
begin
  -- для всех активных контрактов на дату, вычисляю задолженность
  for select oks_region_id, oks_from_k_id, oks_delinq_date, oks_balance
      from region_list(:rep_region_id, 1, -1, 1), oksummary oks
      where id_list = oks_region_id and
            oks_close_state = 0 and
            /*$$IBEC$$ OKS_FROM_K_ID = 109010081087400301 and $$IBEC$$*/
            :rep_date between oks_date_begin and oks_date_end and
            :rep_date >= oks_delinq_date
      order by oks_region_id
      into :reg_id, :k_id, :delinq_date, :balance_kontract
  do
  begin
    if (reg_id <> coalesce(reg_id_tmp, 0)) then
    begin
      reg_id_tmp = reg_id;
      select reg_name
      from region
      where reg_unicode = :reg_id
      into :reg_name;
    end

    k_kontract_no = null;
    k_kf_name = null;
    k_amount_approved = null;
    k_percent_per_day = null;
    k_kredline_end = null;
    k_cashier_fio = null;

    select k_kontract_no, k_disb_date, k_amount_approved, k_percent_per_day, k.k_percent_effect, k_kredline_end, --kontract_dates
           kal.kal_new_kf_name, c_name_fio, iif(pr_code containing '.IPF.', 'ИПФ', 'Классика'),c_id, k.k_fact_disb_lo_id
    from kontract k, knaktlist kal, clients, product p
    where k_id = :k_id and
          k_from_kf_id = kal_from_kf_id and
          k.k_from_pr_id = p.pr_id and
          kal.is_delete = 0 and
          :rep_date between kal_date and kal_date_end and
          k.k_clients_cashier_id = c_id
    into k_kontract_no, k_disb_date, k_amount_approved, k_percent_per_day, k_percent_effect, k_kredline_end, --
         k_kf_name, k_cashier_fio, type_prod, :c_id, :vips_id;

    k_spf_origin = 'Не найден';
    k_spf_is_active = 'Нет';

    if (vips_id is not null) then
    begin
        select c.c_name_fio from clients c
        where c.c_id = :vips_id
        into :k_spf_origin;

        if (exists(
            select 1 from hr_pstatus ps
            inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
            inner join jobkind jk on jk.jk_id = jl.jl_kind
            where ps.ps_from_c_id = :vips_id
              and ps.ps_is_working = 1
              and ps.ps_status = 1
              and jk.jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior')
              and current_date >= ps.ps_date_begin
              and (ps.ps_date_end is null or current_date <= ps.ps_date_end)
        )) then
            k_spf_is_active = 'Да';
    end

    delinq_os = 0;
    delinq_calc_perc = 0;
    delinq_sh_perc = 0;
    fine_balance = 0;
    total_balance = 0;
    delinq_date_os = null;
    delinq_date_perc = null;

    for select zad_os, sh_calc_perc, zad_perc, sh_calc_fine, sh_date
        from aarep_make_zad_list_temp(:k_id, :rep_date, 1)
        where is_extra = 0 and sh_date <= :rep_date
        into zad_os, sh_calc_perc, zad_perc, sh_calc_fine, sh_date
    do
    begin
      delinq_os = delinq_os + zad_os;
      delinq_calc_perc = delinq_calc_perc + sh_calc_perc;
      delinq_sh_perc = delinq_sh_perc + zad_perc;
      fine_balance = fine_balance + sh_calc_fine;

      --По просьбе финансистов, если в оплаты не было о графику просроч.дни равен 1(раньше было 0)
      if (delinq_plus_one = 1) then
        sh_date = sh_date - 1;
      if (sh_date <= delinq_date) then
        delinq_date = sh_date;

      if (zad_os <> 0 and
          delinq_date_os is null) then
        delinq_date_os = sh_date;

      if (zad_perc + sh_calc_perc <> 0 and
          delinq_date_perc is null) then
        delinq_date_perc = sh_date;

    end
    sba_status = 0;
    sba_status_str = 'нет';
    for select coalesce(sb.sba_status, 0)
        from sb_akt sb
        inner join sb_list sbl on sbl.sbl_id = sb.sba_from_sbl_id
        where sb.sba_from_k_id = :k_id
        into sba_status
    do
    begin

      if (sba_status <> 2) then
      begin
        sba_status_str = 'нет';
      end
      if (sba_status = 2) then
        sba_status_str = 'да';

    end
    delinq_days = rep_date - delinq_date;
    delinq_days_os = rep_date - coalesce(delinq_date_os, rep_date);
    delinq_days_perc = rep_date - coalesce(delinq_date_perc, rep_date);

    total_balance = delinq_os + delinq_calc_perc + fine_balance;


    main_number = f_cont_info('clients', :c_id, 'cell_main', 'ok');
    add_number  = f_cont_info('clients', :c_id, 'cell_add', 'ok');

    suspend;
  end
end
