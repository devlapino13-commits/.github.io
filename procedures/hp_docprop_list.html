CREATE PROCEDURE HP_DOCPROP_LIST
RETURNS (
    DP_ID D_INTEGER,
    DP_CODE D_STR_CODE,
    DP_NAME D_STR_255,
    DP_MEMO D_STR_255,
    DP_POSITION D_INTEGER,
    DP_IS_BASE D_BOOL,
    DP_IS_LGOTA D_BOOL)
AS
begin
  for select dp_id, dp_code, dp_name, dp_memo, dp_position, dp_is_base, dp_is_lgota
  from docprop
  order by dp_position, dp_id
  into :dp_id, :dp_code, :dp_name, :dp_memo, :dp_position, :dp_is_base, :dp_is_lgota
  do
    suspend;
end
