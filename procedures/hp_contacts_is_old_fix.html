CREATE PROCEDURE HP_CONTACTS_IS_OLD_FIX
AS
declare variable cc_date date;
declare variable c_id id_type;
declare variable cc_id id_type;
declare variable c_id_old id_type;
declare variable fix_need integer;
declare variable cc_checked_date date;
declare variable cc_check_result integer;
declare variable calc_date date;
begin
  c_id_old = 0;
-- беру всех контактных лиц созданных 30 дней назад
  for select cont.cc_checked_date, cont.cc_date, cont.cc_check_result, cont.cc_id, cont.cc_from_c_id
      from contacts cont
      where cont.cc_date = current_date - 30 and
            cont.cc_is_old_contact = 0 and
            cont.cc_from_c_id <> 0
      order by cc_from_c_id
      into :cc_checked_date, :cc_date, :cc_check_result, :cc_id, :c_id
  do
  begin
    calc_date = cc_date;
-- если контакт не обзвонен, за рассчетную дату беру дату создания контакта в БД.
    if ((cc_checked_date is not null) and
        (cc_check_result > 0)) then
      calc_date = cc_checked_date;

    if (datediff(day, :calc_date, current_date) >= 30) then -- если разность между расчтеной датой и текущей  = или превышает 30 дней
    begin
      if (c_id <> c_id_old) then
      begin
        c_id_old = c_id;
        fix_need = 0;
        if (not exists(select * -- проверяю наличие выданного займа с расчетной даты по текущий момент
                       from kontstate ks
                       inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
                       where kncl.kc_from_c_id = :c_id and
                             kncl.kc_amount_approved > 0 and
                             ks.ks_state = 1 and
                             ks.ks_ol_is_delete = 0 and
                             ((ks.ks_date_begin >= :calc_date) or
                             (current_date between ks.ks_date_begin and ks.ks_date_end)))) then
          fix_need = 1;
      end

      if (fix_need = 1) then
      begin
        update contacts cont
        set cont.cc_is_old_contact = 1
        where cont.cc_id = :cc_id;
      end
    end
  end
end
