CREATE PROCEDURE HP_EXTINFO_TO_SEND_EQUIFAX(
    C_ID ID_TYPE,
    OBJECT_KIND INTEGER,
    OBJECT_ID ID_TYPE)
RETURNS (
    JSON_VALUE D_BLOB_TEXT)
AS
declare variable c_name_fio str_100;
declare variable c_citizen integer;
declare variable c_birth_city str_100;
declare variable c_birth_country str_100;
declare variable c_doc_type integer;
declare variable pk_id integer;
declare variable adr_type integer;
declare variable adr_is_owner integer;
declare variable adr_addr_reg_total str_255;
declare variable adr_post_index str_100;
declare variable fl_id id_type;
declare variable lastname str_100;
declare variable firstname str_100;
declare variable middlename str_100;
declare variable gender smallint;
declare variable birthday str_100;
declare variable birthplace str_100;
declare variable doctype integer;
declare variable docno str_100;
declare variable docdate str_100;
declare variable docplace str_255;
declare variable ar_owner smallint;
declare variable ar_addr_reg_total str_255;
declare variable ar_post_index str_100;
declare variable af_owner smallint;
declare variable af_addr_reg_total str_255;
declare variable af_post_index str_100;
declare variable street_post_index str_100;
declare variable city_post_index str_100;
declare variable create_date date;
declare variable close_date date;
declare variable consent str_100;
declare variable consentdate str_100;
declare variable consentenddate str_100;
declare variable admcodeinform str_100;
declare variable ma_id integer;
begin
  if (coalesce(object_kind, 0) not in (9,11)) then
  begin
    exception e_default 'object_kind must be 11 or 9';
  end 

  if (object_kind = 9) then
  begin
    select k_kredline_end, k_from_ma_app_id
    from kontract
    where k_id = :object_id
    into close_date, ma_id;

    if (close_date is null) then
    begin
        select max(sh_date)
        from schedule sh
        where sh.sh_from_k_id = :object_id
        into :close_date;
    end
  end else
  if (object_kind = 11) then
  begin
    ma_id = object_id;
    select max(sh_date) from schedule
    where sh_from_k_id = -:object_id
    into close_date;

    if (close_date is null) then
    begin
        select max(sh_date)
        from schedule sh
        inner join kontract kont on kont.k_id = sh.sh_from_k_id
        where kont.k_from_ma_app_id = :ma_id
        into close_date;
    end
  end 
  select cast(maa.ma_date_save as date)
  from miniapp maa
  where maa.ma_id = :ma_id
  order by maa.ma_number
  rows 1
  into create_date;

  consent = 1;
  consentdate = f_datetostr(create_date);
  consentenddate = f_datetostr(close_date);
  admcodeinform = 1;

  for select cl.c_name_fio, f_datetostr(cl.c_birth_date), coalesce(cl.c_pass_no, ''),
             (case c_sex
               when 1 then 1
               when 0 then 2
               else 9
             end),
             cl.c_doc_type, c_pass_no, f_datetostr(cl.c_pass_date), coalesce(cl.c_pass_who, ''), cl.c_citizen,
             trim(coalesce(cl.c_birth_city, '')), trim(coalesce(c_birth_country, '')), cl.c_from_fl_id
      from clients cl
      where cl.c_id = :c_id
      into :c_name_fio, :birthday, :docno, :gender, :c_doc_type, :docno, :docdate, :docplace, :c_citizen, :c_birth_city,
           :c_birth_country, :fl_id
  do
  begin
    select fl.fl_name1, fl.fl_name2, fl.fl_name3
    from fiolist fl
    where fl.fl_id = :fl_id
    into :lastname, :firstname, :middlename;

    if (c_name_fio like '%УУЛУ%') then
        lastname = lastname || ' УУЛУ';

    if (c_name_fio like '%КЫЗЫ%') then
       lastname = lastname || ' КЫЗЫ';
    
    if (c_name_fio like '%UULU%') then
       lastname = lastname || ' UULU';
    
    if (c_name_fio like '%KYZY%') then
       lastname = lastname || ' KYZY';

    birthplace = c_birth_city;
    if (birthplace = '') then
      birthplace = c_birth_country;

    if (birthplace = '') then
    begin
      select ddr.rd_name
      from refdata ddr
      where ddr.rd_id = :c_citizen
      into :birthplace; --  место роджения беру исходя из гражданства
    end

    select pask.pk_id
    from passkind pask
    where pask.pk_id = :c_doc_type
    into :pk_id;

    if (c_citizen = 13) then -- Рф
    begin
      if ((pk_id = 12) or (pk_id = 13)) then
        doctype = 1; -- Паспорт гражданина РФ
      if ((pk_id <> 12) and
          (pk_id <> 13)) then
        doctype = 14; -- Неизвестно
    end
    else
      doctype = 6; -- Паспорт, выданный иностранным государством и признаваемый в Российской Федерации
    docno = replace(docno, ' ', '');

    for select adcl.adc_address_type, iif(adcl.adc_is_rent = 0, 3, 9), -- 3 - аренда, 9 - неизвестно
               adl.adl_address_str,
               iif(coalesce(adl_post_index, '') = '', '000000', adl_post_index),
               iif(coalesce(pl_street.pl_post_index, '') = '', '000000', pl_street.pl_post_index),
               iif(coalesce(pl_city.pl_post_index, '') = '', '000000', pl_city.pl_post_index)
        from addrclient adcl
        inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
        left join places pl_street on pl_street.pl_id = adl.adl_street_id
        left join places pl_city on pl_city.pl_id = adl.adl_from_pl_id
        where adcl.adc_from_c_id = :c_id and
              adcl.adc_address_type in (1, 2)
        into :adr_type, :adr_is_owner, :adr_addr_reg_total, :adr_post_index,
             :street_post_index, :city_post_index
    do
    begin
      if (adr_post_index = '000000') then
      begin
        if (street_post_index <> '') then
            adr_post_index = :street_post_index;
        else
            adr_post_index = :city_post_index;
      end


      if (adr_type = 1) then -- место регистрации
      begin
        ar_owner = adr_is_owner;
        ar_addr_reg_total = adr_addr_reg_total;
        ar_post_index = adr_post_index;
      end
      else
      begin
        af_owner = adr_is_owner;
        af_addr_reg_total = adr_addr_reg_total;
        af_post_index = adr_post_index;
      end
    end

    json_value = '{ "request": {' ||
                         '"_private": {' ||
                            f_json_make_pair('lastname', lastname, 'str') || ', ' ||
                            f_json_make_pair('firstname', firstname, 'str') || ', ' ||
                            iif(middlename <> '', f_json_make_pair('middlename', middlename, 'str') || ', ', '') ||
                            f_json_make_pair('gender', gender, 'str') || ', ' ||
                            f_json_make_pair('birthday', birthday, 'str') || ', ' ||
                            f_json_make_pair('birthplace', birthplace, 'str') || ', ' ||
                                '"doc": {' ||
                                    f_json_make_pair('doctype', doctype, 'str') || ', ' ||
                                    f_json_make_pair('docno', docno, 'str') || ', ' ||
                                    f_json_make_pair('docdate', docdate, 'str') || ', ' ||
                                    f_json_make_pair('docplace', docplace, 'str') || ' ' || '} ' ||
                                    '}, ' ||
                                '"addrReg": {' || f_json_make_pair('owner', ar_owner, 'str') || ', ' ||
                                    f_json_make_pair('index', ar_post_index, 'str') || ', ' ||
                                    f_json_make_pair('addrRegTotal', ar_addr_reg_total, 'str') || ' ' || '}, ' ||
                                '"addrFact": {' ||
                                    f_json_make_pair('owner', af_owner, 'str') || ', ' ||
                                    f_json_make_pair('index', af_post_index, 'str') || ', ' ||
                                    f_json_make_pair('addrFactTotal', af_addr_reg_total, 'str') || ' ' ||
                                    '}, ' ||
                                '"application": {' ||
                                    f_json_make_pair('consent', consent, 'str') || ', ' ||
                                    f_json_make_pair('consentdate', consentdate, 'str') || ', ' ||
                                    f_json_make_pair('consentenddate', consentenddate, 'str') || ', ' ||
                                    f_json_make_pair('admcodeInform', admcodeInform, 'str') || ' ' ||
                                    '} ' ||
                            '} ' ||
                         '} ';
    suspend;
  end
end
