CREATE PROCEDURE HP_DO_KN_OPERKRED_PEREPLAT_OFF(
    REP_DATE D_DATE)
AS
declare variable kc_id d_id;
declare variable amount_sum d_currency;
declare variable amount_tax d_currency;
declare variable sab_id d_id;
declare variable acc_id d_id;
declare variable profit_acc_id d_id;
declare variable kob_id d_id;
declare variable k_id d_id;
declare variable ok_os d_currency;
declare variable sh_id d_integer;
declare variable sh_date d_date;
declare variable ok_number d_integer;
declare variable ol_id d_id;
declare variable total_sum d_currency;
declare variable total_tax d_currency;
declare variable rec_count d_integer;
begin
  -- процедура переносит суммы переплаты на доход.
  -- список KC_ID займа предоставляет бухгалтерия
  -- После чего этот список загружается в таблицу Temp_KC_ID и запускается данная процедура.

  total_sum = 0;
  total_tax = 0;
  rec_count = 0;

  select ol_id
  from operlist
  where ol_from_reg_id = 1010100 and
        ol_date = :rep_date and
        ol_oper_type = 2 and
        ol_oper_subtype = 11 and
        ol_memo = 'Списание на доход суммы переплаты' rows 1
  into ol_id;

  update company
  set cm_loan_history_action = 2
  where 1 = 1;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Погашено.Переплата'
  into :sab_id;

  for select kc_from_k_id, kc.kc_id
      from temp_kc_id, knclient kc
      where temp_kc_id.kc_id = kc.kc_id
      into k_id, kc_id
  do
  begin
    acc_id = null;
    amount_sum = 0;
    amount_tax = 0;

    select acc_id, -kobs_bal_sum, -kobs_bal_tax
    from ref_account, kn_obsummary
    where acc_from_sab_id = :sab_id and
          acc_object_kind = 8 and
          acc_object_id = :kc_id and
          kobs_acc_id = acc_id and
          kobs_date <= :rep_date and
          kobs_date_end >= :rep_date
    order by iif(kobs_bal_sum <> 0, 0, 1)
    rows 1
    into acc_id, amount_sum, amount_tax;

    if (acc_id is null) then
      exception e_default 'Acc_ID not found for KC_ID: ' || kc_id;

    if (amount_sum = 0 and
        amount_tax = 0) then
      continue;

    if (amount_sum > 0) then
    begin
      profit_acc_id = null;
      select acc_id
      from ref_account
      where acc_code = '71801810005440610101'
      into profit_acc_id;

      if (profit_acc_id is null) then
        exception e_default 'Profit account not found: ';
    end
    else
    begin
      profit_acc_id = null;
      select acc_id
      from ref_account
      where acc_code = '71701810005240210101'
      into profit_acc_id;

      if (profit_acc_id is null) then
        exception e_default 'Profit account not found: ';
    end

    ok_os = 0;
    sh_id = null;
    sh_date = null;

    select max(ok_sh_id), max(ok_sh_date), sum(-ok_os)
    from kn_operkred
    where ok_from_kc_id = :kc_id and
          ok_type = 9 and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date
    into sh_id, sh_date, ok_os;
    if (ok_os is null) then
      ok_os = 0;

    if (sh_id is null) then
    begin
      if (sh_date is not null) then
      begin
        select ok_sh_id
        from kn_operkred
        where ok_from_kc_id = :kc_id and
              ok_sh_date = :sh_date and
              ok_sh_id is not null rows 1
        into sh_id;
      end
      if (sh_id is null) then
        exception e_default 'OK_Type = 9 not found for KC_ID: ' || kc_id;
    end

    if (ok_os = 0) then
      continue;
    -- exception e_default 'Zero amount in KN_Operkred for KC_ID: ' || kc_id;

    if (ok_os <> amount_sum) then
      continue;
    --exception e_default 'Different amount in KN_Operkred for KC_ID: ' || kc_id;

    if (ol_id is null) then
    begin
      insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_doc_no,
                            ol_doc_no_str, ol_memo, ol_parent_id, is_delete, ol_user_id)
      values (1010100, 2, 11, :rep_date, 9, 0, '', 'Списание на доход суммы переплаты',
              null, 0, 1000000000001)
      returning ol_id
      into :ol_id;
    end

    kob_id = null;
    select kob_id
    from hp_do_kn_operbook_memorder(:ol_id, :kc_id, null, null, :amount_sum, 0, :amount_tax, :amount_tax, 'Признание доходом суммы переплаты', :acc_id, :profit_acc_id, :rep_date, '810')
    into :kob_id;

    if (kob_id is not null) then
    begin
      total_sum = total_sum + amount_sum;
      total_tax = total_tax + amount_tax;
    end

    ok_number = null;
    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                          ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                          ok_ol_reg_id, ok_ol_is_active)
    values (:ol_id, -1, :k_id, :sh_id, :sh_date, 9, :ok_os, 0, 0, 3, 0, 0, 0, 0, 0, :rep_date, 1010100, 1)
    returning ok_number
    into :ok_number;

    if (ok_number is null) then
      exception e_default 'Error when insert Operkred, KC_ID: ' || kc_id;

    delete from kn_operkred
    where ok_from_ol_id = :ol_id and
          ok_number = :ok_number;

    insert into kn_operkred (ok_from_ol_id, ok_number, ok_from_kc_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                             ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                             ok_ol_reg_id, ok_ol_is_active, ok_is_actual)
    values (:ol_id, :ok_number, :kc_id, :sh_id, :sh_date, 9, :ok_os, 0, 0, 3, 0, 0, 0, 0, 0, :rep_date, 1010100, 1, 1);

    rec_count = rec_count + 1;
    --    if (rec_count > 1000) then
    --      Break;
  end

  update company
  set cm_loan_history_action = 0
  where 1 = 1;

end
