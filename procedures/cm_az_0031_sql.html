CREATE PROCEDURE CM_AZ_0031_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable C_ID ID_TYPE;
declare variable C_VERIFY_STATUS integer;
declare variable C_CELL_NO_DATE date;
declare variable C_BIRTH_DATE date;
declare variable C_DOC_TYPE integer;
declare variable C_FROM_IS_ID integer;
declare variable KOL_CLI_KNCLIENT integer;
declare variable KOL_CLI_KONTRACT integer;
begin
  KOL_CLI_KNCLIENT = 0;
  for select KC_FROM_C_ID, KC_AMOUNT_APPROVED, K_DISB_DATE, C_VERIFY_STATUS, C_NAME_FIO, C_PASS_NO, C_BIRTH_DATE,
             C_DOC_TYPE, C_FROM_IS_ID, K_KOLICH_CLIENT_APPROVED
      from KNCLIENT, KONTRACT, CLIENTS
      where KC_FROM_K_ID = :K_ID and
            KC_AMOUNT_APPROVED > 0 and
            KC_FROM_K_ID = K_ID and
            KC_FROM_C_ID = C_ID
      into C_ID, KC_AMOUNT, KF_DISB_DATE, C_VERIFY_STATUS, res_info_text, KF_CLIENT_PASS_NO, C_BIRTH_DATE, C_DOC_TYPE,
           C_FROM_IS_ID, KOL_CLI_KONTRACT
  do
  begin
    KOL_CLI_KNCLIENT = KOL_CLI_KNCLIENT + 1;
    if (:C_BIRTH_DATE is null) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Дата рождения не заполнена';
      suspend;
    end
    if (coalesce(:KF_CLIENT_PASS_NO, '') = '') then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Номер паспорта не заполнен';
      suspend;
    end
    if ((coalesce(:C_DOC_TYPE, 0) < 1) and
        res_error_text = '') then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Тип документа не указан';
      suspend;
    end
    res_error_is_critical = null;
    if (exists(select *
               from ADDRCLIENT
               where ADC_FROM_C_ID = :C_ID and
                     ADC_ADDRESS_TYPE = 1)) then
    begin
      if (exists(select *
                 from ADDRCLIENT
                 where ADC_FROM_C_ID = :C_ID and
                       ADC_ADDRESS_TYPE = 1 and
                       mod(ADC_FROM_ADL_ID, 1000000000) = 0)) then
      begin
        res_error_is_critical = 2;
        res_error_text = '1. '; --Юридический адрес не заполнен
        --        suspend;
      end
    end
    else
    begin
      res_error_is_critical = 2;
      res_error_text = '2. '; --Юридический адрес не существует
      --suspend;
    end
    if (exists(select *
               from ADDRCLIENT
               where ADC_FROM_C_ID = :C_ID and
                     ADC_ADDRESS_TYPE = 2)) then
    begin
      if (exists(select *
                 from ADDRCLIENT
                 where ADC_FROM_C_ID = :C_ID and
                       ADC_ADDRESS_TYPE = 2 and
                       mod(ADC_FROM_ADL_ID, 1000000000) = 0)) then
      begin
        res_error_is_critical = 2;
        res_error_text = '3. '; --Фактический адрес не заполнен
        --        suspend;
      end
    end
    else
    begin
      res_error_is_critical = 2;
      res_error_text = '4. '; --Фактический адрес не существует
      --suspend;
    end
    if (res_error_is_critical = 2) then
    begin
      /*
          SYSDBA
          |mrahmanov
        Для выдачи кредита клиенту обязательно указание юр. и факт. адресов.
        
        Комментарий: “В личной инф. по клиенту обязательно заполнение юр. и факт. адресов.”
        Сообщение:      “Müştərinin məlumatlarında qeydiyyat və faktiki ünvanların qeyd olunması zəruridir.”
      */
      res_error_text = res_error_text || 'Müştərinin məlumatlarında qeydiyyat və faktiki ünvanların qeyd olunması zəruridir.';
      suspend;
    end
  end
  if (KOL_CLI_KNCLIENT <> KOL_CLI_KONTRACT) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Прошу вас переоформить контракт, общее кол-во клиентов не совпададает по кол-ву договору!';
    suspend;
  end
  res_error_is_critical = null;
  for select Z.Z_FROM_C_ID
      from ZALOG Z
      where Z.Z_FROM_K_ID = :K_ID
      into C_ID
  do
  begin
    select C_NAME_FIO
    from CLIENTS
    where C_ID = :C_ID
    into res_info_text;
    if (exists(select *
               from ADDRCLIENT
               where ADC_FROM_C_ID = :C_ID and
                     ADC_ADDRESS_TYPE = 1)) then
    begin
      if (exists(select *
                 from ADDRCLIENT
                 where ADC_FROM_C_ID = :C_ID and
                       ADC_ADDRESS_TYPE = 1 and
                       mod(ADC_FROM_ADL_ID, 1000000000) = 0)) then
      begin
        res_error_is_critical = 2;
        res_error_text = '5. '; --Юридический адрес не заполнен
      end
    end
    else
    begin
      res_error_is_critical = 2;
      res_error_text = '6. '; --юридический адрес не существует
      --suspend;
    end
    if (exists(select *
               from ADDRCLIENT
               where ADC_FROM_C_ID = :C_ID and
                     ADC_ADDRESS_TYPE = 2)) then
    begin
      if (exists(select *
                 from ADDRCLIENT
                 where ADC_FROM_C_ID = :C_ID and
                       ADC_ADDRESS_TYPE = 2 and
                       mod(ADC_FROM_ADL_ID, 1000000000) = 0)) then
      begin
        res_error_is_critical = 2;
        res_error_text = '7. '; --фактический адрес не заполнен
        --        suspend;
      end
    end
    else
    begin
      res_error_is_critical = 2;
      res_error_text = '8. '; --фактический адрес не существует
      --suspend;
    end
    if (res_error_is_critical = 2) then
    begin
      /*
         В личной инф. по клиенту и поручителю обязательно заполнение юр. и факт. Адресов. Очень критично.
      */
      res_error_text = res_error_text || 'Girovqoyanın və zaminin məlumatlarında qeydiyyat və faktiki ünvanların qeyd olunması zəruridir.';
      suspend;
    end
  end
end
