CREATE PROCEDURE UXP$ACTION_GROUP_LIST(
    UA_UUID D_UUID)
RETURNS (
    EXP_VALUE D_STR_CODE,
    UXA_CODE D_STR_CODE,
    UXA_NAME D_STR_255,
    UXA_HIDE BOOLEAN)
AS
declare variable mm_uuid d_uuid;
declare variable ua_owner d_uuid;
declare variable ua_is_group d_bool;
declare variable ua_alias_group d_str_8100;
declare variable exp_alias_code d_str_code;
declare variable exp_alias_value d_str_code;
declare variable arr_count d_integer;
declare variable arr_index d_integer;
declare variable jsm d_integer;
declare variable jsa d_integer;
declare variable jso d_integer;
begin
  mm_uuid = uxf$mainmenu();

  -- если передан ua_code, возващаем одну запись
  select ua_owner, ua_is_group, ua_alias_group
  from ux$action
  where ua_uuid = :ua_uuid
  into ua_owner, ua_is_group, ua_alias_group;

  if (ua_owner = mm_uuid) then
    exit;

  if (ua_is_group = 0) then
    exit;

  arr_count = 0;
  if (coalesce(ua_alias_group, '') <> '') then
  begin
    -- Парсинг массива
    jsm = json.parse(ua_alias_group);
    jsa = json.obj_get_arr(jsm, 'actions');
    arr_count = json.count_get(jsa);
  end

  for select uxa_fcode, uxa_fcode || '   - ' || uxa_name
  from ux$action, ux$alias
  where ua_owner = :ua_owner and
        ua_is_group = 0 and
        ua_usr_id_field = '' and
        ua_uxa_uuid = uxa_uuid
  order by uxa_fcode
  into uxa_code, uxa_name
  do
  begin
    exp_value = '';
    uxa_hide = null;

    if (arr_count > 0) then
    begin
      arr_index = 0;
      while (arr_index < arr_count) do
      begin
        exp_alias_value = '';
        exp_alias_code = '';
        jso = json.arr_get_obj(jsa, arr_index);
  
        -- Получение значения из объекта
        exp_alias_value = json.obj_get_val(jso, 'v', exp_value);
        exp_alias_code = json.obj_get_val(jso, 'a', exp_alias_code);

        if (exp_alias_code = uxa_code) then
        begin
          exp_value = exp_alias_value;
          uxa_hide = json.obj_get_val(jso, 'h', False);
          Break;
        end
        arr_index = arr_index + 1;
      end
    end

    suspend;
  end

  if (jsm is not null) then
    json.destroy(jsm);
end
