CREATE PROCEDURE UXP$UI_UXMENUACTIONS(
    UA_UUID D_UUID,
    UFR_UUID D_UUID)
RETURNS (
    RES_CODE D_STR_100,
    RES_PARENT_CODE D_STR_100,
    RES_CAPTION D_STR_100,
    RES_HINT D_STR_255,
    RES_UA_UUID D_UUID,
    RES_JSONDOC_FIELD D_STR_CODE,
    RES_IMG_UUID_CODE D_STR_CODE,
    RES_INACTIVE D_BOOL,
    RES_DISABLE_ON_EDIT D_BOOL,
    RES_IS_FRAME D_BOOL,
    RES_IS_REPORT D_BOOL,
    RES_IS_COMMAND D_BOOL,
    RES_UXA_OWNER D_UUID,
    RES_DATA_LINKED D_BOOL,
    RES_SOFT_LINKED D_BOOL,
    RES_HOTKEY D_STR_CODE,
    RES_ADD_DELIMITER D_SMALLINT,
    RES_EXPR_UUID D_UUID,
    RES_IS_GROUP D_BOOL,
    RES_ALIAS_GROUP D_STR_8100,
    RES_GROUP_EXPR_UUID D_UUID,
    RES_UXA_FCODE D_STR_100,
    RES_LINKED_FRAMES D_STR_LONG,
    RES_TRIGGER_CODE D_STR_CODE,
    RES_IS_BLANK D_BOOL,
    RES_HIDE_MENU D_BOOL,
    RES_IS_INSERT D_BOOL)
AS
declare variable ux_uuid d_uuid;
declare variable fr_uuid d_uuid;
declare variable tmp_uuid d_uuid;
declare variable is_mainframe d_bool;
declare variable uxa_uuid d_uuid;
declare variable add_delim d_boolean;
begin
  if (ufr_uuid is null) then
    Exit;

  res_hide_menu = 0;

  select ua_uxa_uuid
  from ux$action
  where ua_uuid = :ua_uuid
  into uxa_uuid;

  select ufr_uuid, ufr_owner, ufr_is_mainframe
  from ux$frame
  where ufr_uuid = :ufr_uuid
  into fr_uuid, ux_uuid, is_mainframe;

  add_delim = false;
  res_is_frame = 1;
  res_is_report = 0;
  for select uf_field_name, uf_name
  from ux$frparam, ux$field
  where ufp_ufr_uuid = :fr_uuid and
        ufp_editor = 'blob_json' and
        uf_uuid = ufp_uf_uuid
  into res_jsondoc_field, res_caption
  do
  begin
    res_code = 'b_jsondoc_'  || res_jsondoc_field;
    res_jsondoc_field = res_jsondoc_field || '$json';
    add_delim = true;
    res_caption = uxf$lang(res_caption);
    suspend;
  end

  if (res_jsondoc_field <> '') then
  begin
    res_code = res_code || '_';
    res_caption = '-';
    suspend;
  end

  res_jsondoc_field = '';
  res_hint = '';
  res_ua_uuid = null;
  res_inactive = 0;
  res_disable_on_edit = 0;
  res_parent_code = '';
  res_is_frame = 0;

  if (is_mainframe = 1) then
  begin
    res_is_frame = 0;
    tmp_uuid = ux_uuid;
    for select res_code, res_parent_code, res_caption, res_ua_uuid, res_uxa_owner,
               res_inactive, res_disable_on_edit, res_hint, res_img_uuid_code, res_is_report, res_data_linked, res_soft_linked,
               res_hotkey, res_add_delimiter, res_is_command, res_is_group, res_alias_group, res_uxa_fcode, res_linked_frames,
               res_trigger_code, res_is_blank, res_is_insert
    from uxp$uxsubmenu_items(:tmp_uuid, :uxa_uuid, 1, 0)
    into :res_code, :res_parent_code, :res_caption, :res_ua_uuid, res_uxa_owner,
         :res_inactive, :res_disable_on_edit, :res_hint, :res_img_uuid_code, :res_is_report, :res_data_linked, :res_soft_linked,
         res_hotkey, :res_add_delimiter, res_is_command, res_is_group, res_alias_group, res_uxa_fcode, res_linked_frames,
         res_trigger_code, res_is_blank, res_is_insert
    do
    begin
      res_expr_uuid = null;
      res_group_expr_uuid = null;
      res_hide_menu = 0;

      if (exists(select * from ux$frbutton, ux$action
      where ufb_ua_uuid = :res_ua_uuid and
            ufb_hide_menu = 1 and
            ufb_ua_uuid = ua_uuid and
            ua_owner in (:ux_uuid, :ufr_uuid))) then
        res_hide_menu = 1;

      select exp_ux_uuid
      from ux$expression
      where exp_ux_uuid = :res_ua_uuid and
            exp_kind = 2
      into res_expr_uuid;

      select exp_ux_uuid
      from ux$expression
      where exp_ux_uuid = :res_ua_uuid and
            exp_kind = 1
      into res_group_expr_uuid;

      add_delim = true;
      res_caption = uxf$lang(res_caption);
      res_hint = uxf$lang(res_hint);
      suspend;
    end
  end

  res_trigger_code = '';
  res_is_blank = 0;

  tmp_uuid = ufr_uuid;
  res_is_frame = 1;
  for select res_code, res_parent_code, res_caption, res_ua_uuid, res_uxa_owner,
             res_inactive, res_disable_on_edit, res_hint, res_img_uuid_code, res_is_report, res_data_linked, res_soft_linked,
             res_hotkey, res_add_delimiter, res_is_command, res_is_group, res_alias_group, res_uxa_fcode, res_linked_frames,
             res_is_insert
  from uxp$uxsubmenu_items(:tmp_uuid, :uxa_uuid, 1, 0)
  into :res_code, :res_parent_code, :res_caption, :res_ua_uuid, res_uxa_owner,
       :res_inactive, :res_disable_on_edit, :res_hint, :res_img_uuid_code, :res_is_report, :res_data_linked, :res_soft_linked,
       :res_hotkey, :res_add_delimiter, res_is_command, res_is_group, res_alias_group, res_uxa_fcode, res_linked_frames,
       :res_is_insert
  do
  begin
    res_expr_uuid = null;
    res_group_expr_uuid = null;
    res_hide_menu = 0;

    if (exists(select * from ux$frbutton, ux$fritem
    where ufb_ua_uuid = :res_ua_uuid and
          ufb_hide_menu = 1 and
          ufb_ufi_uuid = ufi_uuid and
          ufi_ufr_uuid = :ufr_uuid)) then
      res_hide_menu = 1;

    select exp_ux_uuid
    from ux$expression
    where exp_ux_uuid = :res_ua_uuid and
          exp_kind = 2
    into res_expr_uuid;

    if (res_is_group = 1) then
    begin
      select exp_ux_uuid
      from ux$expression
      where exp_ux_uuid = :res_ua_uuid and
            exp_kind = 1
      into res_group_expr_uuid;

      if (res_group_expr_uuid is null or coalesce(res_alias_group, '') = '') then
        continue;
    end

    add_delim = true;
    res_caption = uxf$lang(res_caption);
    res_hint = uxf$lang(res_hint);
    suspend;
  end

  res_hint = '';
  res_ua_uuid = null;
  res_inactive = 0;
  res_disable_on_edit = 0;
  res_parent_code = '';
  res_jsondoc_field = '';
  res_expr_uuid = null;
  res_is_group = 0;
  res_alias_group = '';
  res_uxa_fcode = '';
  res_is_frame = 1;
  res_is_report = 0;

  if (uxf$docframe(ux_uuid) = fr_uuid) then
  begin
    if (is_mainframe = 1) then
      res_is_frame = 0;

    if (add_delim) then
    begin
      add_delim = false;
      res_add_delimiter = -1;
    end
    res_code = 'mdoc_RepostDocument';
    res_caption = 'Перепровести документ(ы)';
    res_caption = uxf$lang(res_caption);
    suspend;

    res_add_delimiter = 0;
    res_code = 'mdoc_ChangeDocTime';
    res_caption = 'Изменить время документа';
    res_caption = uxf$lang(res_caption);
    suspend;

    res_add_delimiter = -1;
    res_code = 'mdoc_UnpostDocument';
    res_caption = 'Отменить проведение документов';
    res_caption = uxf$lang(res_caption);
    suspend;
  end
end
