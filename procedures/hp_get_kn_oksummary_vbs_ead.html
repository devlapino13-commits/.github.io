CREATE PROCEDURE HP_GET_KN_OKSUMMARY_VBS_EAD(
    NEW_OKS_DATE_BEGIN D_DATE,
    NEW_OKS_DATE_PRV D_DATE,
    NEW_OKS_DELINQ_DATE D_DATE,
    NEW_OKS_CLOSE_STATE D_SMALLINT,
    NEW_OKS_PERCENT_ESP D_PERCENT,
    NEW_OKS_DATE_STOP_PERC D_DATE,
    NEW_OKS_BAL_ESP_TMP D_CURRENCY,
    NEW_OKS_VBS_END D_CURRENCY,
    NEW_OKS_PMT_ALL_ADV D_CURRENCY,
    NEW_OKS_PMT_OS_NORM D_CURRENCY,
    NEW_OKS_PMT_PERC_NORM D_CURRENCY,
    NEW_OKS_BAL_PERC D_CURRENCY,
    NEW_OKS_PMT_PERC_DELINQ D_CURRENCY,
    NEW_OKS_BAL_ZAD_PERC_PRV D_CURRENCY,
    NEW_OKS_BAL_ESP_NORM D_CURRENCY,
    VBS_AMOUNT_STOPPED D_CURRENCY,
    VBS_AMOUNT_FINAL D_CURRENCY,
    MAX_DATE D_DATE,
    SH_COUNT D_INTEGER,
    STR_ESP_NORM D_STR_LONG,
    STR_DATES D_STR_LONG,
    RES_RESH_DELINQ D_CURRENCY,
    SH_CHANGED D_SMALLINT = 1)
RETURNS (
    RES_OKS_BAL_ESP_NORM_TMP D_CURRENCY,
    RES_OKS_BAL_ESP_NORM D_CURRENCY,
    RES_OKS_BAL_ESP_DELINQ D_CURRENCY,
    RES_DELINQ_DATE D_DATE,
    RES_STOP_PERC D_DATE,
    RES_FUTURE_PERC D_CURRENCY,
    RES_FUTURE_VBS_4 D_CURRENCY,
    RES_EAD D_CURRENCY)
AS
declare variable norm_perc d_currency;
declare variable estimate_bal_end d_currency;
declare variable ok_sh_date d_date;
declare variable middle_no d_integer;
declare variable delinq_no d_integer;
declare variable n d_integer;
declare variable future_esp_norm d_currency;
declare variable tmp_amount d_currency;
declare variable d_days d_double;
declare variable vbs_tmp d_currency;
declare variable res_discount_koef d_double;
begin
-- Вычисляем ВБС по ЧистНЗ
  if (new_oks_date_begin is null) then
    new_oks_date_begin = new_oks_date_begin;
  if (sh_changed is null) then
    sh_changed = 1;

  if (new_oks_pmt_os_norm is null) then
    new_oks_pmt_os_norm = 0;
  if (new_oks_pmt_perc_norm is null) then
    new_oks_pmt_perc_norm = 0;
  if (new_oks_bal_esp_norm is null) then
    new_oks_bal_esp_norm = 0;

  res_oks_bal_esp_norm_tmp = 0;
  res_oks_bal_esp_norm = new_oks_bal_esp_norm;
  res_oks_bal_esp_delinq = 0;

-- res_oks_bal_esp_norm, res_oks_bal_esp_delinq

  res_oks_bal_esp_norm_tmp = -new_oks_pmt_all_adv;

-- если договор не расторгнут, считаем обычный ЧистНЗ на будущие платежи (vbs_amount_final)
-- иначе считаем все как свернуто в один платеж (vbs_amount_stopped)
  if (new_oks_date_stop_perc is null or
      cast(new_oks_date_stop_perc as date) > cast(new_oks_date_begin as date) or
      res_resh_delinq <> 0 or
      sh_changed = 1) then
    res_oks_bal_esp_norm_tmp = round(res_oks_bal_esp_norm_tmp + vbs_amount_final, 2);
  else
  begin
    norm_perc = new_oks_bal_perc - (new_oks_bal_zad_perc_prv + new_oks_pmt_perc_delinq);
    if (norm_perc < 0) then
      norm_perc = 0;
    res_oks_bal_esp_norm_tmp = round(res_oks_bal_esp_norm_tmp + vbs_amount_stopped + norm_perc, 2);
  end

  res_oks_bal_esp_norm = round(res_oks_bal_esp_norm + new_oks_pmt_os_norm + new_oks_pmt_perc_norm, 2);

  if (new_oks_bal_esp_tmp < res_oks_bal_esp_norm) then
    res_oks_bal_esp_norm = new_oks_bal_esp_tmp;

  res_oks_bal_esp_norm = res_oks_bal_esp_norm + res_resh_delinq;

  if (new_oks_date_begin = new_oks_date_prv and
      abs(new_oks_bal_esp_tmp - res_oks_bal_esp_norm) <= 10) then
    res_oks_bal_esp_norm = new_oks_bal_esp_tmp;

  if (new_oks_close_state = 2) then
  begin
    res_oks_bal_esp_norm = 0;
    res_oks_bal_esp_norm_tmp = 0;
    new_oks_bal_esp_tmp = 0;
  end

  if (new_oks_date_begin = new_oks_date_prv and
      abs(new_oks_bal_esp_tmp - res_oks_bal_esp_norm_tmp) <= 10) then
    res_oks_bal_esp_norm_tmp = new_oks_bal_esp_tmp;

  if (new_oks_close_state = 2) then
    res_oks_bal_esp_delinq = 0;
  else
  begin
    res_oks_bal_esp_delinq = new_oks_bal_esp_tmp - res_oks_bal_esp_norm;

--    if (res_oks_bal_esp_norm = 0 and
--        res_oks_bal_esp_norm_tmp = 0 and
--        res_oks_bal_esp_delinq < new_oks_vbs_end) then
--      res_oks_bal_esp_delinq = new_oks_vbs_end;
  end

  estimate_bal_end = res_oks_bal_esp_norm_tmp + res_oks_bal_esp_delinq + new_oks_pmt_all_adv;

  if (estimate_bal_end < 0 and
      estimate_bal_end >= -1 and
      new_oks_date_begin <= coalesce(new_oks_date_stop_perc, new_oks_date_begin) and
      res_oks_bal_esp_norm_tmp > estimate_bal_end) then
  begin
    res_oks_bal_esp_norm_tmp = res_oks_bal_esp_norm_tmp - estimate_bal_end;
  end
  else
  if (abs(estimate_bal_end) > 0 and
      abs(estimate_bal_end) <= 1 and
      new_oks_close_state = 1) then
    res_oks_bal_esp_norm_tmp = res_oks_bal_esp_norm_tmp - estimate_bal_end;

-- вычисляем EAD для расчета РО
  res_ead = null;
  res_delinq_date = null;
  res_stop_perc = null;
  res_future_vbs_4 = 0;
  res_future_perc = 0;

  if (str_dates <> '') then
  begin
    if (new_oks_delinq_date is not null) then
    begin
      res_delinq_date = new_oks_delinq_date;
      delinq_no = 1;
    end
    else
    begin
      if (sh_count > 3) then
      begin
        middle_no = sh_count / 2;
        if (sh_count > middle_no * 2) then
          middle_no = middle_no + 1;

        if (sh_count > 5) then
        begin
          if (sh_count > 12 and new_oks_date_begin >= '01.11.2019')
          then delinq_no = 6;
          else delinq_no = middle_no - 3;
        end
        else
          delinq_no = 1;
        if (delinq_no <= 0) then
          delinq_no = 1;
        res_delinq_date = substring(str_dates from (delinq_no - 1) * 10 + 1 for 10);
      end
      else
      begin
        delinq_no = 1;
        res_delinq_date = new_oks_date_begin;
      end
    end
    res_stop_perc = dateadd(day, 91, res_delinq_date);
    if (res_stop_perc > max_date) then
      res_stop_perc = max_date;

    if (res_stop_perc >= new_oks_date_begin) then
    begin
      if (res_delinq_date >= new_oks_date_begin and
          sh_count > 3) then
      begin
        future_esp_norm = 0;

        n = delinq_no;
        while (n <= sh_count) do
        begin
          ok_sh_date = substring(str_dates from (n - 1) * 10 + 1 for 10);
          tmp_amount = substring(str_esp_norm from (n - 1) * 20 + 1 for 20);
          future_esp_norm = future_esp_norm + tmp_amount / power(1.0 + cast(new_oks_percent_esp as double precision) / 100.0, cast(datediff(day, res_delinq_date, ok_sh_date) as double precision) / 365.0);
          n = n + 1;
        end
        res_future_vbs_4 = round(future_esp_norm, 2) + res_oks_bal_esp_delinq;
      end
      else
      begin
        if (sh_changed = 1) then
          res_future_vbs_4 = res_oks_bal_esp_norm_tmp + res_oks_bal_esp_delinq + new_oks_pmt_all_adv;
        else
          res_future_vbs_4 = new_oks_bal_esp_tmp + new_oks_pmt_all_adv;
      end

      d_days = datediff(day, new_oks_date_begin, res_stop_perc);
      res_discount_koef = 1 / power(1 + new_oks_percent_esp / 100.0, d_days / 365.0);

      d_days = datediff(day, maxvalue(res_delinq_date, new_oks_date_begin), res_stop_perc);
      if (d_days <> 0) then
        res_future_perc = round(res_future_vbs_4 * (power(1 + new_oks_percent_esp / 100.0, d_days / 365.0) - 1.0), 2);
      else
        res_future_perc = 0;

      res_ead = round((res_future_vbs_4 + res_future_perc) * res_discount_koef, 2);
      if (abs(res_ead - res_future_vbs_4) > 0 and
          abs(res_ead - res_future_vbs_4) <= 0.02) then
        res_ead = res_future_vbs_4;

      if (sh_changed = 1) then
        vbs_tmp = res_oks_bal_esp_norm_tmp + res_oks_bal_esp_delinq + new_oks_pmt_all_adv;
      else
        vbs_tmp = new_oks_bal_esp_tmp + new_oks_pmt_all_adv;

      if (abs(res_ead - vbs_tmp) <= 0.02) then
        res_ead = vbs_tmp;

      if (res_ead > vbs_tmp) then
      begin
        if (res_discount_koef <> 0) then
          res_future_vbs_4 = round(vbs_tmp / ((1 + (power(1 + new_oks_percent_esp / 100.0, d_days / 365.0) - 1.0)) * res_discount_koef), 2);

        d_days = datediff(day, maxvalue(res_delinq_date, new_oks_date_begin), res_stop_perc);
        if (d_days <> 0) then
          res_future_perc = round(res_future_vbs_4 * (power(1 + new_oks_percent_esp / 100.0, d_days / 365.0) - 1.0), 2);
        else
          res_future_perc = 0;

        res_ead = round((res_future_vbs_4 + res_future_perc) * res_discount_koef, 2);
        if (abs(res_ead - res_future_vbs_4) > 0 and
            abs(res_ead - res_future_vbs_4) <= 0.02) then
          res_ead = res_future_vbs_4;

        if (abs(res_ead - vbs_tmp) <= 0.02) then
          res_ead = vbs_tmp;
      end
    end
  end

  if (res_ead is null) then
  begin
    if (sh_changed = 1) then
      res_future_vbs_4 = res_oks_bal_esp_norm_tmp + res_oks_bal_esp_delinq + new_oks_pmt_all_adv;
    else
      res_future_vbs_4 = res_oks_bal_esp_norm + res_oks_bal_esp_delinq;
    res_future_perc = 0;
    res_ead = res_future_vbs_4;
    res_discount_koef = 1;
    res_delinq_date = new_oks_delinq_date;
    res_stop_perc = new_oks_date_stop_perc;
  end

  suspend;
end
