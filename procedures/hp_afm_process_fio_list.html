CREATE PROCEDURE HP_AFM_PROCESS_FIO_LIST(
    C_ID D_ID NOT NULL,
    MA_ID D_INTEGER NOT NULL)
AS
declare variable lo_c_id d_id; /* c_id кредитного специалиста клиента, которому будет оптравлен пуш */
declare variable push_body d_str_long;
declare variable afm_list d_str_long; /* Список категорий, в которых клиент был найден */
declare variable c_name_fio d_str_255; /* ФИО клиента */
begin
  select categories
  from hp_afm_get_client_categories(:c_id,
                                    ', ',
                                    f_format('%1,%2,%3', f_enum_id('afm_status', 'acting'), f_enum_id('afm_status', 'listed'), f_enum_id('afm_status', 'included')),
                                    '1,4,5,6,7,8,9,10')
  into :afm_list;

  if (coalesce(:afm_list, '') = '') then
    exit; -- Если клиент не был найден ни в одной из интересующих категорий, ничего не делаем

  -- Поиск кредитного специалиста клиента
  select
      kal.kal_new_kf_officer
  from knaktlist kal
  where kal.kal_from_kf_id = (select first 1
                                  ma.ma_from_kf_id
                              from miniapp ma
                              where ma.ma_id = :ma_id and
                                    ma.ma_date_end = '31.12.9999'
                              order by ma.ma_number desc) and
        current_date between kal.kal_date and kal.kal_date_end and
        kal.is_delete = 0
  into :lo_c_id;

  if (coalesce(:lo_c_id, 0) <> 0) then
  begin
    push_body = 'Напоминаем: %1 находится в списке повышенного риска. Причина: %2. Решение о выдачи за вами';

    select c.c_name_fio from clients c
    where c.c_id = :c_id
    into :c_name_fio;

    push_body = left(f_format(:push_body, :c_name_fio, :afm_list), 1024);

    -- Отправка PUSH уведомления кредитному специалисту клиента
    f_send_push('username',
                (select first 1 un_id from username where un_from_c_id = :lo_c_id),
                f_format('%1 находится в списке повышенного риска.', :c_name_fio),
                :push_body,
                '');
  end


end
