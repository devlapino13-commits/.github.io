CREATE PROCEDURE OKT_SIRENA_LOAN_INFO(
    KONTRACT_NO D_STR_100)
RETURNS (
    BLOB_TXT D_BLOB_TEXT)
AS
declare variable kf_id id_type;
declare variable ok_last_payed d_date;
declare variable k_disb_date_str d_str_100;
declare variable last_pay_date d_date;
declare variable last_pay_date_str d_str_100;
declare variable sh_date d_date;
declare variable zad_summ_temp curr_type;
declare variable zad_summ curr_type;
declare variable next_pay_date d_date;
declare variable next_pay_date_str d_str_100;
declare variable k_kolich_client_approved d_integer;
declare variable k_disb_date d_date;
declare variable k_amount_approved curr_type;
declare variable kc_amount_approved curr_type;
declare variable k_period d_integer;
declare variable k_kredline_end d_date;
declare variable reg_name d_str_100;
declare variable kf_name d_str_100;
declare variable ok_ol_date d_date;
declare variable ok_sum_total curr_type;
declare variable ok_last_payed_str d_str_100;
declare variable ok_last_payed_sum curr_type;
declare variable kred_full_balance curr_type;
declare variable qty_left d_integer;
declare variable delinq_days d_integer;
declare variable i d_integer;
declare variable ok_ol_date_str str_100;
declare variable k_id d_id;
begin
  blob_txt = '';
  for select k.k_id,
             k.k_kolich_client_approved,
             k.k_disb_date,
             k.k_amount_approved,
             kc.kc_amount_approved,
             k.k_period,
             k.k_kredline_end,
             reg.reg_name,
             k.k_from_kf_id
      from kontract k
      inner join region reg on reg.reg_unicode = k.k_last_reg_id
      inner join knclient kc on kc.kc_from_k_id = k.k_id
      where k.k_kontract_no = :kontract_no
      into :k_id,
           :k_kolich_client_approved,
           :k_disb_date,
           :k_amount_approved,
           :kc_amount_approved,
           :k_period,
           :k_kredline_end,
           :reg_name,
           :kf_id
  do
  begin
    select kal.kal_new_kf_name
    from knaktlist kal
    where kal.kal_from_kf_id = :kf_id and
          current_date between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into :kf_name;

    delinq_days = 0;

    select current_date - oks.oks_delinq_date
    from oksummary oks
    where oks.oks_from_k_id = :k_id and
          current_date between oks.oks_date_begin and oks.oks_date_end
    into :delinq_days;

    delinq_days = coalesce(delinq_days, 0);
    kred_full_balance = 0; -- сумма полного долга по заему
    qty_left = 0; -- осталось погашений

    for select coalesce(sh_date, cast('01.01.1900' as date)), sum(zad_os + sh_calc_perc + sh_calc_fine)
        from aarep_make_zad_list(:k_id, current_date, 1)
        group by sh_date
        order by sh_date
        into :sh_date,
             :zad_summ_temp
    do
    begin
      if (sh_date < current_date) then
      begin
        zad_summ = zad_summ + zad_summ_temp;
      end
      else
      begin
        if (next_pay_date is null) then
          next_pay_date = sh_date;
      end
      if (sh_date <> '01.01.1900') then
        last_pay_date = sh_date;
      kred_full_balance = kred_full_balance + zad_summ_temp;
      qty_left = qty_left + 1;
    end
    if (next_pay_date is null) then
      next_pay_date_str = 'отсутсвует';
    else
      next_pay_date_str = f_datetostr(next_pay_date);

    select ol_date, sum(payed_sum)
    from (select ok.ok_ol_date as ol_date,
               -(ok.ok_os + ok.ok_perc + ok.ok_fine) as payed_sum
          from operkred ok
          where ok.ok_from_k_id = :k_id and
                ok.ok_type = 4 and
                ok.ok_ol_is_active = 1 and
                ok.ok_from_c_id = 0
          union all
          select oko_ol_date as ol_date, -(okf.oko_os + okf.oko_perc + okf.oko_other)
          from operkroff okf
          where okf.oko_ol_is_active = 1 and
                okf.oko_from_k_id = :k_id and
                okf.oko_type = 4
          union all
          select oe.ox_ol_date as ol_date, oe.ox_amount_minus
          from operextra oe
          where oe.ox_object_kind = 9 and
                oe.ox_object_id = :k_id and
                oe.ox_from_rd_id = f_get_rd_id('operextrakind.lawclientexpense') and
                oe.ox_ol_is_active = 1) a
    group by ol_date
    having sum(payed_sum) > 0
    order by ol_date desc
    rows 1
    into :ok_last_payed,
         :ok_last_payed_sum;

    k_disb_date_str = f_datetostr(k_disb_date);
    if (ok_last_payed is not null) then
      ok_last_payed_str = f_datetostr(ok_last_payed);
    else
      ok_last_payed_str = 'отсутствует';
    if (last_pay_date is not null) then
      last_pay_date_str = f_datetostr(last_pay_date);
    else
      last_pay_date_str = 'отсутствует';
    last_pay_date_str = coalesce(last_pay_date_str, 'договор закрыт');
    ok_last_payed_str = coalesce(ok_last_payed_str, '');
    ok_last_payed_sum = round(coalesce(ok_last_payed_sum, 0));


    blob_txt = BLOB_APPEND(blob_txt, '<br><b>Общая информация по кредиту</b><br>');
    blob_txt = BLOB_APPEND(blob_txt, '№ группы..................................................................' || coalesce(kf_name, '') || '<br>');
    blob_txt = BLOB_APPEND(blob_txt, '№ контракта..............................................................' || coalesce(kontract_no, '') || '<br>');
    blob_txt = BLOB_APPEND(blob_txt, 'офис..........................................................................' || coalesce(reg_name, '') || '<br>');
    blob_txt = BLOB_APPEND(blob_txt, 'участников в группе................................................' || coalesce(k_kolich_client_approved, 0) || '<br>');
    blob_txt = BLOB_APPEND(blob_txt, 'дата выдачи..............................................................' || k_disb_date_str || '<br>');
    blob_txt = BLOB_APPEND(blob_txt, 'сумма займа (всей группы)....................................' || round(k_amount_approved) || ' <br>');
    blob_txt = BLOB_APPEND(blob_txt, 'собственная сумма заемщика................................' || round(kc_amount_approved) || ' <br>');
    blob_txt = BLOB_APPEND(blob_txt, 'срок займа................................................................' || coalesce(k_period, 0) || ' мес. <br>');
    blob_txt = BLOB_APPEND(blob_txt, 'дата закрытия кредита............................................' || f_datetostr(k_kredline_end) || '<br>');
    if (next_pay_date is not null) then
      blob_txt = BLOB_APPEND(blob_txt, 'дата ближайшего платежа......................................' || coalesce(next_pay_date_str, '') || '<br>');
    blob_txt = BLOB_APPEND(blob_txt, '<br><b>осталось погасить</b><br>');
    blob_txt = BLOB_APPEND(blob_txt, 'осталось погасить по заему (всей группе)...........' || coalesce(round(kred_full_balance), 0) || ' <br>');
    blob_txt = BLOB_APPEND(blob_txt, 'осталось погашений...............................................' || coalesce(qty_left, 0) || '<br>');
    blob_txt = BLOB_APPEND(blob_txt, '<br><b>последний раз платили</b><br>');
    blob_txt = BLOB_APPEND(blob_txt, 'дата платежа............................................................' || coalesce(ok_last_payed_str, '') || '<br>');
    blob_txt = BLOB_APPEND(blob_txt, 'сумма платежа.........................................................' || coalesce(ok_last_payed_sum, 0) || ' <br>');
    blob_txt = BLOB_APPEND(blob_txt, '<br><b>активная просрочка</b><br>');
    blob_txt = BLOB_APPEND(blob_txt, 'просроченная сумма (всей группы)......................' || coalesce(round(zad_summ), 0) || ' <br>');
    blob_txt = BLOB_APPEND(blob_txt, 'дней просрчоки ......................................................' || coalesce(delinq_days, 0) || ' дней<br>');
    blob_txt = BLOB_APPEND(blob_txt, '<br><b>ИСТОРИЯ ПЛАТЕЖЕЙ</b><br>');
    i = 0;
    for select ok.ok_ol_date,
          -sum(ok.ok_os + ok.ok_perc + ok.ok_fine)
        from operkred ok
        where ok.ok_ol_is_active = 1 and
              ok.ok_from_k_id = :k_id and
              ok.ok_type = 4
        group by ok_ol_date
        having(-sum(ok.ok_os + ok.ok_perc + ok.ok_fine)) > 0
        order by ok.ok_ol_date desc
        into :ok_ol_date,
             :ok_sum_total
    do
    begin
      i = i + 1;
      ok_ol_date_str = replace(iif(extract(day from ok_ol_date) < 10, '0', '') || extract(day from ok_ol_date) || '.' || iif(extract(month from ok_ol_date) < 10, '0', '') || extract(month from ok_ol_date) || '.' || extract(year from ok_ol_date), ' ', '');
      blob_txt = BLOB_APPEND(blob_txt, '№ ' || cast(i as varchar(10)) || ' Дата платежа: ' || coalesce(ok_ol_date_str, '') || '; Сумма: ' || coalesce(round(ok_sum_total), 0) || ' <br>');
    end
    if (i = 0) then
      blob_txt = BLOB_APPEND(blob_txt, 'Ни разу не платили!');
    suspend;
  end
end
