CREATE PROCEDURE AAREP_SSVZ_WEEK(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    KF_OFFICER VARCHAR(100),
    WEEK_1_KOL_DISB_KRED INTEGER,
    WEEK_2_KOL_DISB_KRED INTEGER,
    WEEK_3_KOL_DISB_KRED INTEGER,
    WEEK_4_KOL_DISB_KRED INTEGER,
    KOL_DISB_PARALEL INTEGER,
    DOLYA_UDERJ CURR_TYPE,
    WEEK_1_PAR7 CURR_TYPE,
    WEEK_2_PAR7 CURR_TYPE,
    WEEK_3_PAR7 CURR_TYPE,
    WEEK_4_PAR7 CURR_TYPE,
    KOL_DELINQ_GR INTEGER,
    REP_BALANCE_KAT_1 CURR_TYPE,
    REP_BALANCE_KAT_2 CURR_TYPE,
    KOL_CLI_KAT_1 INTEGER,
    KOL_CLI_KAT_2 INTEGER,
    PRIROST_CLI_KAT_1 INTEGER)
AS
declare variable i smallint;
declare variable rep_date date;
declare variable rep_date_begin date;
declare variable svz_id id_type;
declare variable par7 curr_type;
declare variable kol_disb_kred integer;
declare variable rep_balance curr_type;
declare variable delinq_balance_7 curr_type;
declare variable kol_cli_kat_1_last integer;
declare variable closed_kolich_gr integer;
declare variable disb_kolich_gr integer;
begin
  for
-- Вытаскивает всех СВЗ кроме декретников
      select ps_from_c_id
      from jobkind, joblist, hr_pstatus, region_list(:rep_region_id, 1, 3, 1)
      where jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            jk_id = jl_kind and
            jl_id = ps_from_jl_id and
            ps_status = 1 and
            ps_is_working = 1 and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = id_list
-- Исключаем декретников
            and
            not coalesce((select sawt_id
                          from hr_voctravel, hr_orders, sa_worktype
                          where hrv_from_ho_id = ho_id and
                                ho_order_type = sawt_id and
                                hrv_from_c_id = ps_from_c_id and
                                :rep_date_end between hrv_date_begin and hrv_date_end_fact
                          order by hrv_date_begin desc
                          rows 1), 0) in (16, 17, 18)
      into svz_id
  do
  begin
    rep_date = :rep_date_end;
    i = 0;
    while (i < 4) do
    begin
      rep_date = rep_date - i * 7;
      rep_date_begin = rep_date - 6; -- начало недели
      i = i + 1;
--    Кол-во выдач за неделю
      kol_disb_kred = 0;
      select sum(iif((select pr_product_kind
                      from product
                      where pr_id = k_from_pr_id) <> 2, k_kolich_client_approved, 0))
      from kontstate, kontract
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            cast(ks_date_begin as date) between :rep_date_begin and :rep_date and
            ks_kc_id = :svz_id and
            ks_from_k_id = k_id
      into kol_disb_kred;
      if (kol_disb_kred is null) then
        kol_disb_kred = 0;
--    Пар
      rep_balance = null;
      delinq_balance_7 = null;
      select sum(oks_balance), sum(iif(:rep_date_end - 7 >= oks_delinq_date, oks_balance, 0))
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_kc_id = :svz_id and
            oks_close_state = 0 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
      into rep_balance, delinq_balance_7;
      if (rep_balance is null) then
        rep_balance = 0;
      if (delinq_balance_7 is null) then
        delinq_balance_7 = 0;
      if (rep_balance > 0) then
      begin
        par7 = delinq_balance_7 / rep_balance;
      end
      else
      begin
        par7 = 0;
      end

      if (i = 1) then
      begin
        week_1_kol_disb_kred = kol_disb_kred;
        week_1_par7 = par7;
      end
      if (i = 2) then
      begin
        week_2_kol_disb_kred = kol_disb_kred;
        week_2_par7 = par7;
      end
      if (i = 3) then
      begin
        week_3_kol_disb_kred = kol_disb_kred;
        week_3_par7 = par7;
      end
      if (i = 4) then
      begin
        week_4_kol_disb_kred = kol_disb_kred;
        week_4_par7 = par7;
      end
    end
--    Кол-во выдач за неделю, параллельные
    kol_disb_paralel = 0;
    select sum(iif((select pr_product_kind
                    from product
                    where pr_id = k_from_pr_id) = 2, k_kolich_client_approved, 0))
    from kontstate, kontract
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_end - 6 and :rep_date_end and
          ks_kc_id = :svz_id and
          ks_from_k_id = k_id
    into kol_disb_paralel;
    if (kol_disb_paralel is null) then
      kol_disb_paralel = 0;
--  Кол-во закрытых групп за текущий месяц
    closed_kolich_gr = null;
    select count(k_id)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 2 and
          cast(ks_date_begin as date) between :rep_date_end - 6 and :rep_date_end and
          ks_kc_id = :svz_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2
    into closed_kolich_gr;
    if (closed_kolich_gr is null) then
      closed_kolich_gr = 0;
--  Кол-во выданных групп для расчета % удержание групп
    disb_kolich_gr = null;
    select count(k_id)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_end - 6 and :rep_date_end and
          ks_kc_id = :svz_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 and
          k_circle > 1 -- для расчете удержание будем брать только повтоно выданные
    into disb_kolich_gr;
    if (disb_kolich_gr is null) then
      disb_kolich_gr = 0;
    if (closed_kolich_gr <> 0) then
      dolya_uderj = cast(disb_kolich_gr as double precision) / closed_kolich_gr;
    else
      dolya_uderj = 0;

--    Кол-во просроченных групп
    kol_delinq_gr = null;
    select count(oks_from_k_id)
    from oksummary, kontract, product
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :svz_id and
          oks_close_state = 0 and
          :rep_date_end - 30 >= oks_delinq_date and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2) and
          oks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2
    into kol_delinq_gr;
    if (kol_delinq_gr is null) then
      kol_delinq_gr = 0;

--    Кредитный портфель категории 1 и 2
    rep_balance_kat_1 = null;
    rep_balance_kat_2 = null;
    select sum(iif((select aa_category_id
                    from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 1, oks_balance, 0)), sum(iif((select aa_category_id
                                                                                                                                                  from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 2, oks_balance, 0))
    from oksummary
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :svz_id and
          oks_close_state = 0
    into rep_balance_kat_1, rep_balance_kat_2;
    if (rep_balance_kat_1 is null) then
      rep_balance_kat_1 = 0;
    if (rep_balance_kat_2 is null) then
      rep_balance_kat_2 = 0;
--    Активные клиенты категории 1 и 2
    kol_cli_kat_1 = null;
    kol_cli_kat_2 = null;
    select sum(iif((select aa_category_id
                    from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) = 1, 1, 0)), sum(iif((select aa_category_id
                                                                                                                                         from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) = 2, 1, 0))
    from aarep_cli_active_list(:rep_date_end)
    where ac_kc_id = :svz_id and
          ac_is_client = 1
    into kol_cli_kat_1, kol_cli_kat_2;
    if (kol_cli_kat_1 is null) then
      kol_cli_kat_1 = 0;
    if (kol_cli_kat_2 is null) then
      kol_cli_kat_2 = 0;
--    Активные клиенты категории 1 на начало недели для вычисления прироста
    kol_cli_kat_1_last = null;
    select sum(ac_is_client)
    from aarep_cli_active_list(:rep_date_end - 7)
    where ac_kc_id = :svz_id and
          ac_is_client = 1 and
          (select aa_category_id
           from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) = 1
    into kol_cli_kat_1_last;
    if (kol_cli_kat_1_last is null) then
      kol_cli_kat_1_last = 0;
    prirost_cli_kat_1 = kol_cli_kat_1 - kol_cli_kat_1_last;
--    ФИО СВЗ
    kf_officer = (select c_name_fio
                  from clients
                  where c_id = :svz_id);

    suspend;
  end
end
