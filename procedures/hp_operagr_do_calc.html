CREATE PROCEDURE HP_OPERAGR_DO_CALC(
    AG_RD_CODE D_STR_255 NOT NULL,
    OBJ_KIND D_STR_CODE NOT NULL,
    OBJ_ID D_ID NOT NULL)
AS
declare variable rd_id d_integer;
declare variable rd_code d_str_255;
declare variable obj_kind_str d_str_code;
declare variable cell_main d_str_code;
declare variable cell_add d_str_code;
declare variable disb_date d_date;
declare variable c_id d_id;
declare variable ol_id d_id;
declare variable calc_amount d_currency;
declare variable ag_id d_integer;
declare variable clk_id d_id;
declare variable last_close_date d_date;
declare variable cli_id d_id;
begin
  -- Процедура возвращать Agreements.AG_ID

  if (ag_rd_code similar to '[-]?[[:DIGIT:]]+') then
  begin
    select rd_id, rd_code
    from refdata
    where rd_id = :ag_rd_code
    into rd_id, rd_code;
  end else
  begin
    select rd_id, rd_code
    from refdata
    where rd_code = :ag_rd_code
    into rd_id, rd_code;
  end

  if (rd_id is null) then Exit;

  obj_kind_str = f_objkind(obj_kind, 'name');
  obj_kind = f_objkind(obj_kind, 'id');

  if (rd_code = 'AgreementKinds.loyal_agreement') then -- программа лояльности
  begin
    if (obj_kind_str <> 'kontract') then
      exception e_default 'obj_kind expected Kontract, actually: ' || obj_kind_str;

    -- получаем телефонные номера клиента, получивщий займ
    for select ks_from_ol_id, ks_date_begin,
               f_cont_info('clients', kc_from_c_id, 'cell_main', 'ok'),
               f_cont_info('clients', kc_from_c_id, 'cell_add', 'last')
    from knclient, kontstate
    where ks_from_k_id = :obj_id and
          ks_state = 1 and
          ks_ol_is_delete = 0 and
          kc_from_k_id = ks_from_k_id and
          kc_amount_approved >= 10000
    into ol_id, disb_date, cell_main, cell_add
    do
    begin
      ag_id = null;
      if (cell_main <> '' and cell_add = cell_main) then cell_add = '';

      -- Ищем клиента, кто предоставлял эти номера как номера своего друга.
      -- Номер друга должен быть предоставлен не позже дня выдачи

      -- поиск по основному номеру
      c_id = null;
      if (cell_main <> '') then
      begin
        select clk_id, clk_from_c_id
        from clientlink
        where clk_link_cell_no = :cell_main and
              clk_add_datetime between dateadd(day, -90, :disb_date) and (:disb_date + 1)
        order by clk_id desc
        rows 1
        into clk_id, c_id;
      end

      -- поиск по доп.номеру
      if (cell_add <> '' and c_id is null) then
      begin
        select clk_id, clk_from_c_id
        from clientlink
        where clk_link_cell_no = :cell_add and
              clk_add_datetime between dateadd(day, -90, :disb_date) and (:disb_date + 1)
        order by clk_id desc
        rows 1
        into clk_id, c_id;
      end

      -- находит ID клиета чтобы проверить последний закрытый контракт
      select kc_from_c_id from knclient
      where kc_from_k_id = :obj_id
      and kc_amount_approved > 0
      rows 1
      into cli_id;

      -- проверка на закрытие последнего контракта должно пройти более 6ти месяцев
      if (c_id is not null) then
      begin
        last_close_date = null;
        select max(ks_date_end) from knclient, kontstate
        where kc_from_c_id = :cli_id
        and kc_amount_approved > 0
        and kc_from_k_id = ks_from_k_id
        and ks_ol_is_delete = 0
        and ks_state = 1
        and ks_from_ol_id <> :ol_id
        into last_close_date;

        if (last_close_date is not null) then
        begin
          if (last_close_date >= dateadd(month, -6, disb_date) - 1) then
          begin
            c_id = null;
          end
        end
      end 

      -- если нашли клиента-агента, ищем его договор вступления в программу лояльности
      -- договор должен быть подтвержденным и действующим на дату выдачи
      if (c_id is not null) then
      begin
        select ag_id
        from agreements
        where ag_from_rd_id = :rd_id and
              ag_object_kind = f_objkind('clients', 'id') and
              ag_object_id = :c_id and
              ag_status = 1 and
              :disb_date between ag_date_begin and coalesce(ag_date_end, :disb_date)
        order by ag_id desc
        rows 1
        into ag_id;

        if (ag_id is not null) then
        begin
          if (not exists(
          select * from operagr
          where ogr_from_ag_id = :ag_id and
                ogr_from_rd_id = :rd_id and
                ogr_ol_is_active = 1 and
                ogr_object_kind = f_objkind('clientlink', 'id') and
                ogr_object_id = :clk_id)) then
          begin
            if (not exists(
            select * from operagr
            where ogr_from_ol_id = :ol_id and
                  ogr_ol_is_active = 1 and
                  ogr_from_ag_id = :ag_id and
                  ogr_from_rd_id = :rd_id)) then
            begin
              if (exists(select *
                         from partner_program_group
                         where ppg_bonus = 600 and
                               ppg_from_c_id = :c_id)) then
                calc_amount = 600.00 * 2;
              else
                calc_amount = 600.00;

              -- RFCRU-6095: изменение баллов с 429 на 600

              insert into operagr (ogr_from_ol_id, ogr_from_ag_id, ogr_from_rd_id, ogr_sign, ogr_amount, ogr_tax_fl,
                                   ogr_object_kind, ogr_object_id, ogr_ol_date, ogr_ol_is_active)
              values (:ol_id, :ag_id, :rd_id, 1, :calc_amount, 0, f_objkind('clientlink', 'id'), :clk_id,  :disb_date, 1);
            end
          end
        end
      end
    end
  end
end
