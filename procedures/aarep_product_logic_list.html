CREATE PROCEDURE AAREP_PRODUCT_LOGIC_LIST(
    CURR_PR_ID INTEGER,
    DISB_DATE DATE,
    CLIENT_KOLICH INTEGER,
    LOAN_CIRCLE INTEGER,
    LOAN_AMOUNT DECIMAL(18,4),
    LOAN_PERCENT DECIMAL(18,7),
    LOAN_REPAY INTEGER,
    LOAN_ADMIN_PERC DECIMAL(18,7),
    LOAN_MIN_MAX_DIFF DECIMAL(18,4),
    LOAN_MALE_PERCENT DECIMAL(18,7),
    LOAN_MIN_AMOUNT DECIMAL(18,4),
    LOAN_MAX_AMOUNT DECIMAL(18,4),
    INCLUDE_CURR_PR INTEGER)
RETURNS (
    PR_ID INTEGER,
    PRL_ID INTEGER,
    PR_DATE_BEGIN DATE,
    PR_NAME VARCHAR(50),
    PR_MEMO VARCHAR(255),
    PRL_DATE_BEGIN DATE,
    PRL_DATE_END DATE,
    PRL_CIRCLE_MIN INTEGER,
    PRL_CIRCLE_MAX INTEGER,
    PRL_KOLICH_MIN INTEGER,
    PRL_KOLICH_MAX INTEGER,
    PRL_AMOUNT_GR_MIN DECIMAL(18,4),
    PRL_AMOUNT_GR_MAX DECIMAL(18,4),
    PRL_AMOUNT_CL_MIN DECIMAL(18,4),
    PRL_AMOUNT_CL_MAX DECIMAL(18,4),
    PRL_REPAY_MIN INTEGER,
    PRL_REPAY_MAX INTEGER,
    PRL_PERCENT_MIN DECIMAL(18,7),
    PRL_PERCENT_MAX DECIMAL(18,7),
    PRL_ADMIN_PERC_MIN DECIMAL(18,7),
    PRL_ADMIN_PERC_MAX DECIMAL(18,7),
    PRL_MIN_MAX_DIFF DECIMAL(18,4),
    PRL_MALE_MAX_PERC DECIMAL(18,7),
    PR_CATEGORY_PERC INTEGER,
    PR_CURRENCY VARCHAR(3),
    PR_INDEX_CURR_CODE VARCHAR(3),
    PR_DATE_END DATE)
AS
declare variable kol integer;
begin
  pr_memo = '';
  if (client_kolich is null) then
    client_kolich = -1;
  if (loan_circle is null) then
    loan_circle = -1;
  if (loan_amount is null) then
    loan_amount = -1;
  if (loan_min_amount is null) then
    loan_min_amount = -1;
  if (loan_max_amount is null) then
    loan_max_amount = -1;
  if (loan_repay is null) then
    loan_repay = -1;
  if (loan_percent is null) then
    loan_percent = -1;
  if (loan_admin_perc is null) then
    loan_admin_perc = -1;
  if (loan_min_max_diff is null) then
    loan_min_max_diff = -1;
  if (loan_male_percent is null) then
    loan_male_percent = -1;

  for select distinct pr_id, pr_date_begin, pr_date_end, pr_name, pr_category_perc, pr_currency, pr_index_curr_code
      from product
      where pr_id = :curr_pr_id
      into :pr_id, :pr_date_begin, :pr_date_end, :pr_name, :pr_category_perc, :pr_currency, :pr_index_curr_code
  do
  begin
    kol = 0;
    prl_id = 0;
    prl_date_begin = null;
    prl_date_end = null;
    prl_circle_min = null;
    prl_circle_max = null;
    prl_kolich_min = null;
    prl_kolich_max = null;
    prl_amount_gr_min = null;
    prl_amount_gr_max = null;
    prl_amount_cl_min = null;
    prl_amount_cl_max = null;
    prl_repay_min = null;
    prl_repay_max = null;
    prl_percent_min = null;
    prl_percent_max = null;
    prl_admin_perc_min = null;
    prl_admin_perc_max = null;
    prl_min_max_diff = null;
    prl_male_max_perc = null;
    if (exists(select *
               from prodlogic
               where prl_from_pr_id = :pr_id)) then
    begin
      for select distinct prl_id, prl_date_begin, prl_date_end, prl_circle_min, prl_circle_max, prl_kolich_min,
                          prl_kolich_max, prl_amount_gr_min, prl_amount_gr_max, prl_amount_cl_min, prl_amount_cl_max,
                          prl_repay_min, prl_repay_max, prl_percent_min, prl_percent_max, prl_admin_perc_min,
                          prl_admin_perc_max, prl_min_max_diff, prl_male_max_perc
          from prodlogic
          where prl_from_pr_id = :pr_id and
                ((prl_date_begin <= :disb_date and
                prl_date_end >= :disb_date and
                ((:client_kolich = -1) or
                (:client_kolich <> -1 and
                :client_kolich >= coalesce(prl_kolich_min, 0) and
                :client_kolich <= coalesce(prl_kolich_max, 99999999999))) and
                ((:loan_circle = -1) or
                (:loan_circle <> -1 and
                :loan_circle >= coalesce(prl_circle_min, 0) and
                :loan_circle <= coalesce(prl_circle_max, 99999999999))) and
                ((:loan_amount = -1) or
                (:loan_amount <> -1 and
                :loan_amount >= coalesce(prl_amount_gr_min, 0) and
                :loan_amount <= coalesce(prl_amount_gr_max, 99999999999))) and
                ((:loan_min_amount = -1) or
                (:loan_min_amount <> -1 and
                :loan_min_amount >= coalesce(prl_amount_cl_min, 0) and
                :loan_min_amount <= coalesce(prl_amount_cl_max, 99999999999))) and
                ((:loan_max_amount = -1) or
                (:loan_max_amount <> -1 and
                :loan_max_amount >= coalesce(prl_amount_cl_min, 0) and
                :loan_max_amount <= coalesce(prl_amount_cl_max, 99999999999))) and
                ((:loan_percent = -1) or
                (:loan_percent <> -1 and
                :loan_percent >= coalesce(prl_percent_min, 0) and
                :loan_percent <= coalesce(prl_percent_max, 99999999999))) and
                ((:loan_repay = -1) or
                (:loan_repay <> -1 and
                :loan_repay >= coalesce(prl_repay_min, 0) and
                :loan_repay <= coalesce(prl_repay_max, 99999999999))) and
                ((:loan_admin_perc = -1) or
                (:loan_admin_perc <> -1 and
                :loan_admin_perc >= coalesce(prl_admin_perc_min, 0) and
                :loan_admin_perc <= coalesce(prl_admin_perc_max, 99999999999))) and
                ((:loan_min_max_diff = -1) or
                (:loan_min_max_diff <> -1 and
                :loan_min_max_diff <= coalesce(prl_min_max_diff, 99999999999))) and
                ((:loan_male_percent = -1) or
                (:loan_male_percent <> -1 and
                :loan_male_percent <= coalesce(prl_male_max_perc, 99999999999)))))

          into :prl_id, :prl_date_begin, :prl_date_end, :prl_circle_min, :prl_circle_max, :prl_kolich_min,
               :prl_kolich_max, :prl_amount_gr_min, :prl_amount_gr_max, :prl_amount_cl_min, :prl_amount_cl_max,
               :prl_repay_min, :prl_repay_max, :prl_percent_min, :prl_percent_max, :prl_admin_perc_min,
               :prl_admin_perc_max, :prl_min_max_diff, :prl_male_max_perc
      do
      begin
        kol = kol + 1;
        suspend;
      end
    end
    else
      suspend;
  end
end
