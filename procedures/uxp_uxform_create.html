CREATE PROCEDURE UXP$UXFORM_CREATE(
    CREATE_KIND D_STR_CODE,
    UX_OWNER D_UUID,
    FR_CODE D_STR_CODE,
    FR_NAME D_STR_100,
    FR_SOURCE D_STR_8100,
    FR_JSON_DATA D_BLOB_TEXT = null,
    UX_OWNER_CODE D_STR_CODE = null)
RETURNS (
    RES_FR_UUID D_UUID,
    RES_FR_UUID_STR D_STR_CODE,
    RES_UX_UUID D_UUID)
AS
declare variable us_uuid d_uuid;
declare variable is_grid d_bool;
declare variable ufm_uuid d_uuid;
declare variable ugr_uuid d_uuid;
declare variable ux_uuid d_uuid;
declare variable urt_kind d_smallint;
declare variable res_source_code d_str_code;
declare variable res_kind d_str_code;
declare variable res_sql_from d_str_long;
declare variable next_num d_integer;
declare variable is_import d_bool;
declare variable res_fields d_smallint;
begin
  if (coalesce(ux_owner, '') = '') then
    ux_owner = uxf$uniq(ux_owner_code, 0);

  if (fr_code is null) then
    fr_code = '';

  is_import = 0;
  is_grid = 1;
  if (create_kind = 'uxform_one') then
  begin
    is_grid = 0;
    create_kind = 'uxform';
  end
  else
  if (create_kind = 'uxcommand') then
  begin
    is_grid = 0;
    if (position(' ', fr_source) > 0) then
      exception e_message ('В команде источником может быть только процедура');
  end else
  if (create_kind = 'uxreport_import') then
  begin
    is_import = 1;
    create_kind = 'uxreport';
  end

  if (coalesce(fr_json_data, '') <> '') then
    execute procedure uxp$uxsqlfields_fill(:fr_json_data);

  if (ux_owner is null and create_kind = 'uxreport') then
  begin
    if (position(' ', fr_source) > 0 or position('(', fr_source) > 0) then
    begin
      res_source_code = '';

      select res_kind, res_source_code, res_sql_from
      from uxp$sql_parcer(:fr_source)
      into :res_kind, :res_source_code, :res_sql_from;

      if (res_kind = 'sql') then
      begin
        select source_code
        from uxp$sql_source_parcer(:res_sql_from)
        rows 1
        into :res_source_code;
      end
    end else
      res_source_code = fr_source;

    if (res_source_code <> '') then
      execute procedure uxp$uxsource_create(:res_source_code, '', null, 0, '');
  end

  fr_code = trim(fr_code);

  if (exists(select * from ux$code
  where ux_scode = :fr_code and
        ux_us_uuid = :ux_owner)) then
  begin
    if (fr_code <> '') then
      exception e_message 'Кодовое название занято, придумайте новый код или оставьте пустым.';
    begin
      -- Находим первый свободный номер
      next_num = 0;
      while (fr_code = '') do
      begin
        next_num = next_num + 1;
        if (not exists(select * from ux$code
        where ux_scode = :next_num and
              ux_us_uuid = :ux_owner)) then
          fr_code = next_num;
      end
    end
  end

  select ux_uuid, ufm_uuid
  from uxp$uxcode_insert(:ux_owner, :create_kind, :fr_code, :fr_name)
  into :ux_uuid, :ufm_uuid;

  res_ux_uuid = ux_uuid;

  -- Создаем фрейм
  select res_fr_uuid, res_fr_uuid_str, us_uuid
  from uxp$uxframe_add(:ufm_uuid, '', :fr_name, :is_grid, :fr_source)
  into :res_fr_uuid, :res_fr_uuid_str, us_uuid;

  if (create_kind = 'uxreport') then
  begin
    urt_kind = 0; -- Табличный отчет

    insert into ux$report (urt_uuid, urt_kind)
    values (:ufm_uuid, :urt_kind);

    if (urt_kind = 0 and is_import = 0) then
    begin
      insert into ux$grid (ugr_ufr_uuid, ugr_is_private)
      values (:res_fr_uuid, 0)
      returning ugr_uuid
      into :ugr_uuid;

      insert into ux$gritem (ugi_ugr_uuid, ugi_uf_uuid, ugi_position)
      select :ugr_uuid, uf_uuid, null
      from ux$field
      where uf_owner = :us_uuid and
            uf_is_param = 0
      order by uf_position;
    end

    res_fr_uuid = ufm_uuid;
    res_fr_uuid_str = uuid_to_char(ufm_uuid);
  end else
  if (create_kind = 'uxcommand') then
  begin
    select count(*)
    from ux$field
    where uf_owner = :us_uuid and
          uf_is_param = 0 and
          uf_field_name in ('res_is_error', 'res_text')
    into res_fields;

    if (res_fields = 2) then
    begin
      update ux$command
      set cmd_message_field = 'new.res_text'
      where cmd_uuid = :ufm_uuid;

      insert into ux$expression (exp_ux_uuid, exp_kind, exp_expression, exp_calc_expr, exp_calc_ufr_uuid, exp_message, exp_is_sql, exp_action, exp_is_bool)
      values (:ufm_uuid, 0, 'new.res_is_error = 0', ':new$res_is_error = 0', NULL, 'new.res_text', 0, 0, 1);
    end
  end

  suspend;
end
