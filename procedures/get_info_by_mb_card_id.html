CREATE PROCEDURE GET_INFO_BY_MB_CARD_ID(
    CARD_ID D_INTEGER,
    ON_DATE D_DATE = current_date)
RETURNS (
    CCA_ID D_INTEGER,
    CCA_CODE D_STR_CODE,
    K_ID D_ID,
    C_ID D_ID)
AS
begin
  select cca_id,
         cca_code
  from cardlist cal
  inner join cardclientacc  caa on cal.cal_from_cca_id = caa.cca_id
  where cal.cal_card_id = :card_id
  into :cca_id,
       :cca_code;

  select cll_from_k_id,
         cln.cll_from_c_id
  from cardlistlnk cln
  where cln.cll_card_id = :card_id
  and :on_date between cll_date_begin and cll_date_end
  into :k_id,
       :c_id;

  suspend;
end
