CREATE PROCEDURE AAREP_CLI_STOPPERC(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_STOP_PERC DATE,
    K_AMOUNT_APPROVED CURR_TYPE,
    REG_NAME VARCHAR(50),
    REG_STAT_REP_REGION VARCHAR(255),
    KNC_NAME_FIO VARCHAR(100),
    KNC_BIRTH_DATE DATE,
    KNC_AMOUNT_APPROVED CURR_TYPE,
    KNC_CIRCLE INTEGER,
    KNC_BUSINESS_TYPE VARCHAR(50),
    KNC_BUSINESS_SUBTYPE VARCHAR(50),
    KNC_REP_ACTIVE_BALANCE CURR_TYPE,
    KD_PERCENT PERC_TYPE,
    TYPE_FINANCING VARCHAR(20),
    NO_ACCRUAL_STATUS SMALLINT,
    K_DISB_KINDD D_STR_100,
    K_BY_RATING D_STR_CODE,
    C_INN D_STR_20)
AS
declare variable DEF_CURR varchar(3);
declare variable K_INDEX_CURR_CODE varchar(3);
declare variable K_INDEX_CURR_RATE CURR_TYPE;
declare variable REP_ACTIVE_BALANCE CURR_TYPE;
declare variable CR_RATE CURR_TYPE;
declare variable K_ID decimal(18,0);
declare variable PR_ID integer;
declare variable PR_ID_PRIOR integer;
declare variable PR_CODE varchar(100);
declare variable K_DISB_KIND D_INTEGER;
begin
  select cm_default_currency
  from company
  into :def_curr;

  pr_id_prior = 0;
  for select k_id, k_kontract_no, k_disb_date, iif(pr_code containing '.ipf.',
             (select rep_delinq_begin + 90
              from aarep_kont_delinq_status(k_id, :rep_date_end)
              where rep_delinq_days >= 90), k_stop_perc), k_amount_approved, reg_code || ' ' || reg_name,
             reg_stat_rep_region, k_index_curr_code, k_index_curr_rate, k_from_pr_id, k_disb_kind,
             iif(k_by_rating = 1, 'По кредитному рейтингу', 'Стандартная выдача')
      from kontstate, kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1), region, product
      where ks_ol_is_delete = 0 and
            k_from_pr_id = pr_id and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            iif(pr_code containing '.ipf.', iif(((pr_code containing '.murabaha.' or
            pr_code containing '.idjara.') and
            pr_name not containing 'Сервис') and
            (select rep_delinq_begin + 90
             from aarep_kont_delinq_status(k_id, :rep_date_end)
             where rep_delinq_days >= 90) is not null, true, false), (k_stop_perc is not null)) and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            kal_new_reg_id = id_list and
            reg_unicode = id_list

      into k_id, k_kontract_no, k_disb_date, k_stop_perc, k_amount_approved, reg_name, reg_stat_rep_region,
           k_index_curr_code, k_index_curr_rate, pr_id, k_disb_kind, k_by_rating
  do
  begin
    if (pr_id_prior <> pr_id) then
    begin
      pr_id_prior = pr_id;

      select pr_code
      from product
      where pr_id = :pr_id
      into pr_code;

      if (pr_code containing '.ipf.') then
        type_financing = 'ИПФ';
      else
        type_financing = 'Классика';
    end
    k_disb_kindd = case :k_disb_kind
                     when 6 then 'На карту Мбулак'
                     when 5 then 'На карту клиента'
                     else 'Касса'
                   end;
    rep_active_balance = 0;
    select oks_balance
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_close_state = 0
    into :rep_active_balance;

    if (def_curr <> k_index_curr_code) then
    begin
      select cr_rate
      from opercurr
      where cr_from_cu_id = :k_index_curr_code and
            :rep_date_end between cr_date and cr_date_end
      into cr_rate;
      if (cr_rate is null) then
        cr_rate = 0;
      if (:cr_rate >= :k_index_curr_rate) then
        rep_active_balance = :rep_active_balance * :cr_rate;
      else
        rep_active_balance = :rep_active_balance * :k_index_curr_rate;
    end

    for select c_name_fio, c_birth_date, kc_amount_approved, kc_circle, bt_name, bs_name, c_inn
        from knclient, clients, bustype, bussubtype
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = c_id and
              kc_busuness_subtype = bs_id and
              bs_from_bt_id = bt_id
        into knc_name_fio, knc_birth_date, knc_amount_approved, knc_circle, knc_business_type, knc_business_subtype, c_inn
    do
    begin
      knc_rep_active_balance = cast(knc_amount_approved as double precision) / k_amount_approved * rep_active_balance;

      kd_percent = 0;
      select kd.kd_percent
      from kontdata kd
      where kd_from_k_id = :k_id and
            kd_ol_is_delete = 0 and
            :rep_date_end between kd_ol_date and kd_date_end
      into kd_percent;

      no_accrual_status = 0;
      select iif(k_stop_perc <= :rep_date_end, 1, 0)
      from kontract
      where k_id = :k_id
      into no_accrual_status;

      if (k_stop_perc <= :rep_date_end) then
        suspend;
    end
  end
end
