CREATE PROCEDURE HP_HR_ACC_SIGN_VIEW(
    BEGIN_DATE D_DATE,
    END_DATE D_DATE,
    UN_ID D_ID = null)
RETURNS (
    HAS_ID D_ID,
    HO_ID D_ID,
    HO_ORDER_NO D_STR_20,
    HO_DATE D_DATE,
    HO_REGION D_STR_100,
    HO_OFFICE D_STR_100,
    EMP_FIO D_STR_100,
    EMP_POST D_STR_255,
    SAWS_NAME D_STR_255,
    HO_HELP_AMOUNT D_INTEGER,
    HO_ACCOUNTANT_FIO D_STR_100,
    HAS_IS_SIGN D_BOOL)
AS
declare variable ho_from_reg_id d_integer;
declare variable ho_order_subtype d_integer;
declare variable ho_accountant_id d_id;
declare variable parent_reg_id d_integer;
declare variable c_id d_id;
begin
  -- возвращает список сотрдуников, которым в заданном периоде оформлены приказы на мат. помощь (RFCKG-4382)
  -- записи фильтруются по пользователю, вызвавшем данную процедуру (входящий параметр un_id)
  -- пользователь видит только тех сотрудников, по которым он указан как бухгалтер-подписант

  for select ho_id,
             ho_order_no,
             ho_date,
             ho_from_reg_id,
             ho_order_subtype,
             ho_help_amount,
             ho_accountant_id,
             has_id,
             has_is_sign
      from hr_orders
      inner join hr_accountant_sign on has_ho_id = ho_id
      inner join username on un_from_c_id = has_accountant_id
      where un_id = :un_id and
            ho_date between :begin_date and :end_date and
            ho_status = 1 and
            ho_help_amount > 0 and
            has_is_delete = 0
      order by ho_date, ho_order_no
      into ho_id,
           ho_order_no,
           ho_date,
           ho_from_reg_id,
           ho_order_subtype,
           ho_help_amount,
           ho_accountant_id,
           has_id,
           has_is_sign
  do
  begin
    parent_reg_id = null;
    ho_office = null;
    ho_region = null;
    saws_name = null;
    ho_accountant_fio = null;

    select parent,
           reg_code || ' ' || reg_name
    from region
    where reg_unicode = :ho_from_reg_id
    into parent_reg_id,
         ho_office;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :parent_reg_id
    into ho_region;

    ho_region = replace(ho_region, ' р.', '');

    select saws_name
    from sa_worksubtype
    where saws_id = :ho_order_subtype
    into saws_name;

    select c_name_fio
    from clients
    where c_id = :ho_accountant_id
    into ho_accountant_fio;

    for select hrv_from_c_id
        from hr_voctravel
        where hrv_status = 1 and
              hrv_from_ho_id = :ho_id
        into c_id
    do
    begin
      emp_fio = null;
      emp_post = null;

      select c_name_fio
      from clients
      where c_id = :c_id
      into emp_fio;

      select jl_name
      from hr_pstatus
      inner join joblist on jl_id = ps_from_jl_id
      where ps_from_c_id = :c_id and
            ps_status = 1 and
            ps_is_working = 1 and
            :ho_date between ps_date_begin and ps_date_end
      into emp_post;

      suspend;
    end
  end
end
