CREATE PROCEDURE OKT_CC_CHECK_KNCL_LIST(
    REPORT_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    KF_NAME STR_100,
    KONTRACT_NO VARCHAR(20),
    KONT_TYPE SMALLINT,
    OPER STR_100,
    CHECK_TYPE SMALLINT)
RETURNS (
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    REG_NAME STR_100,
    K_KONTRACT_NO STR_100,
    C_NAME_FIO STR_255,
    DISB_DATE DATE,
    KC_AMOUNT_APPROVED CURR_TYPE,
    IS_CHECKED SMALLINT,
    CK_IS_BLOCKED SMALLINT,
    CK_BLOCKED_OPER STR_100,
    CK_CHECKED_OPER STR_100,
    CELL_NO VARCHAR(20),
    CK_STATUS SMALLINT,
    IS_LIE SMALLINT,
    CK_CALL_QTY SMALLINT,
    CK_CHECKED_DATETIME TIMESTAMP,
    CK_CHECKED_DATE DATE,
    NATION STR_100,
    NATION_ID INTEGER,
    CK_MEMO VARCHAR(1000),
    DELINQ_DAY INTEGER,
    CK_PLEASED_STATUS SMALLINT,
    CITIZEN STR_100,
    KAL_NEW_KF_NAME STR_100)
AS
declare variable ck_blocked_datetime timestamp;
declare variable k_id_temp id_type;
declare variable c_citizen integer;
declare variable prior_reg_id integer;
declare variable reg_id integer;
declare variable kf_id_basic id_type;
declare variable k_id_basic id_type;
declare variable kf_id id_type;
declare variable is_ok smallint;
begin
  k_id_temp = 0;
  prior_reg_id = 0;

  kf_name = coalesce(kf_name, '');
  kontract_no = coalesce(kontract_no, '');

  kf_id_basic = 0;

  if (kf_name <> '') then
  begin
    select kal.kal_from_kf_id
    from knaktlist kal
    where kal.kal_new_kf_name = :kf_name and
          :rep_date_end between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into :kf_id_basic;
  end

  if ((kf_name <> '') and
      (kf_id_basic = 0)) then
    kf_id_basic = -1;

  k_id_basic = 0;
  if (kontract_no <> '') then
  begin
    select kont.k_id
    from kontract kont
    where kont.k_kontract_no = :kontract_no
    into :k_id_basic;
  end

  if ((kontract_no <> '') and
      (k_id_basic = 0)) then
    k_id_basic = -1;

  for select k_fact_disb_reg_id, k_id, kont.k_kontract_no, ks_date_begin, kont.k_from_kf_id
      from kontstate, kontract kont, region_list(:report_region_id, 1, 3, 1)
      where k_id = ks_from_k_id and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_state = :kont_type and
            ks_region_id = id_list and
            id_list = kont.k_fact_disb_reg_id and
            ks_ol_is_delete = 0 and
            ((k_id = :k_id_basic) or
            (:k_id_basic = 0))
      order by :reg_id, k_id
      into :reg_id, :k_id, :k_kontract_no, :disb_date, :kf_id
  do
  begin
    is_ok = 0;

    select kal.kal_new_kf_name
    from knaktlist kal
    where kal.kal_from_kf_id = :kf_id and
          :rep_date_end between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into :kal_new_kf_name;

    if ((kf_id_basic = kf_id) or (kf_id_basic = 0)) then
    begin
      is_ok = 1;
    end

    if (is_ok = 1) then
    begin
      if (prior_reg_id <> reg_id) then
      begin
        prior_reg_id = reg_id;

        select reg_name
        from region reg
        where reg.reg_unicode = :reg_id
        into :reg_name;
      end

      for select cl.c_id, cl.c_name_fio,
                 f_cont_info('clients', kncl.kc_from_c_id, 'cell_main', 'last'),
                 cl.c_citizen, nac.rd_name, nac.rd_id, kc_amount_approved
          from knclient kncl
          inner join clients cl on cl.c_id = kncl.kc_from_c_id
          inner join refdata nac on nac.rd_id = cl.c_nation
          where kncl.kc_amount_approved > 0 and
                kncl.kc_from_k_id = :k_id
          into :c_id, :c_name_fio, :cell_no, :c_citizen, :nation, :nation_id, :kc_amount_approved
      do
      begin
        is_checked = 0;
        ck_is_blocked = 0;
        ck_blocked_datetime = null;
        ck_blocked_oper = null;
        ck_checked_oper = null;
        ck_status = 0;
        ck_call_qty = 0;
        ck_checked_datetime = null;
        ck_checked_date = null;
        ck_memo = '';
        ck_pleased_status = 0;

        citizen = '';

        select rd.rd_name
        from refdata rd
        where rd.rd_id = :c_citizen
        into :citizen;

        select ck_status, ck_is_blocked, ck_blocked_datetime, ck_blocked_oper, ck_checked_oper, ck_call_qty,
               ck_checked_datetime, ck_checked_date, ck_memo, ck_pleased_status
        from cc_check_knclient kncl
        where ck_from_k_id = :k_id and
              kncl.ck_from_c_id = :c_id and
              kncl.ck_type = :kont_type rows 1
        into :ck_status, :ck_is_blocked, :ck_blocked_datetime, :ck_blocked_oper, :ck_checked_oper, :ck_call_qty,
             :ck_checked_datetime, :ck_checked_date, :ck_memo, :ck_pleased_status;

        if ((ck_blocked_oper <> oper) and
            (ck_is_blocked = 1)) then
        begin
          if (abs(datediff(minute from :ck_blocked_datetime to current_timestamp)) > 15) then
            ck_is_blocked = 0;
        end

        if (ck_blocked_oper = oper) then
        begin
          ck_is_blocked = 0;
        end

        ck_call_qty = coalesce(ck_call_qty, 0);

        is_lie = 0;
        if (ck_status = 2) then
          is_lie = 1;
        if (ck_status > 0) then
          is_checked = 1;

        if (k_id_temp <> k_id) then
        begin
          delinq_day = 0;

          select max(ok_ol_date - ok_sh_date)
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_k_id = :k_id and
                ok_type in (4, 5, 9) and
                ok_ol_date > ok_sh_date and
                ok_ol_date <= :rep_date_end and
                ok_from_c_id = 0 and
                ok_os + ok_perc <> 0
          into :delinq_day;
          if (delinq_day is null) then
            delinq_day = 0;

          k_id_temp = k_id;
        end

        if ((check_type = 1) and
            (ck_status = 0)) then
        begin
          if (kont_type = 2) then
          begin
            if (delinq_day < 7) then
              suspend;
          end
          else
            suspend;
        end

        if ((check_type = 2) and
            (ck_status > 0)) then
        begin
          suspend;
        end

        if (check_type = 3) then
        begin
          if (kont_type = 2) then
          begin
            if (delinq_day < 7) then
              suspend;
          end
          else
            suspend;
        end
      end
    end
  end
end
