CREATE PROCEDURE AAREP_GSFR_REL_ANNUL_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    CLIENT_FIO D_STR_100,
    REL_FIO D_STR_100,
    GRAL_DATE D_DATE,
    UN_NAME D_STR_CODE)
AS
begin
  -- данные отчета "Аннулированные родственные связи клиентов ГСФР" (RFCKG-2614)

  for select client_fio,
             rel_fio,
             gral_date,
             un_name
      from hp_gsfr_rel
      where gral_date between :rep_date_begin and :rep_date_end
      into client_fio,
           rel_fio,
           gral_date,
           un_name
  do
    suspend;

end
