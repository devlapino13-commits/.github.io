CREATE PROCEDURE AAREP_LOAN_DISB_LIST_TMP(
    UN_ID DECIMAL(18,0),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REG_ID INTEGER = -1)
RETURNS (
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    RES_DATE DATE,
    RES_APP_AMOUNT DECIMAL(18,4),
    RES_DISB_AMOUNT DECIMAL(18,4),
    RES_MEMO VARCHAR(255),
    RES_KIND VARCHAR(100),
    RES_LO_ID DECIMAL(18,0),
    K_CLIENT_COUNT INTEGER,
    K_CLIENT D_ID,
    K_DISB_KIND D_SMALLINT)
AS
declare variable k_id decimal(18,0);
declare variable kf_id decimal(18,0);
declare variable ma_id integer;
declare variable ks_state integer;
declare variable amount decimal(18,4) = 0;
declare variable k_status integer;
declare variable rep_plan_disb_date date;
declare variable ma_is_active integer;
declare variable need_suspend integer;
declare variable res_object_status integer;
declare variable kf_name varchar(100);
declare variable lo_id decimal(18,0);
declare variable jk_system_code varchar(100);
declare variable cm_database_id integer;
declare variable kc_from_c_id d_id;
declare variable rd_code d_str_100;
begin
  :cm_database_id = f_get_db_id();

  un_id = (select un_id
           from hp_test_un_id(:un_id));

  select un_from_c_id
  from username
  where un_id = :un_id
  into :lo_id;

  if (lo_id is null) then
    exit;

  select iif(:reg_id = -1, ps_from_reg_id, :reg_id), jk_system_code
  from hr_pstatus, joblist, jobkind
  where ps_date_begin <= :rep_date_end and
        ps_date_end >= :rep_date_end and
        ps_from_c_id = :lo_id and
        ps_status = 1 and
        ps_is_working = 1 and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id
  into :reg_id, jk_system_code;

  if (not(jk_system_code in ('JobSystem.LO_Partner', 'JobSystem.LO_Standart', 'JobSystem.Region_KM', 'JobSystem.LO_Senior', 'JobSystem.Specialist_KOBD', 'JobSystem.Specialist_Expert'))) then
    reg_id = null;

  if (reg_id is null) then
    exit;

  if (jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Specialist_KOBD')) then
    lo_id = 0;

  for select res_c_id
      from hp_staff_changer_list(:un_id, 1)
      where :lo_id <> 0
      union
      select 0
      from company
      where :lo_id = 0
      into :lo_id
  do
  begin
    obj_kind = 'Kontract';
    res_kind = iif(:cm_database_id <> 94401, 'выдан', 'verilib');
        -- Фактически выданные кредиты
    for select k_id, k_from_kf_id, ks_date_begin, k_status, kc_amount_approved, k_kolich_client_approved,
               k_from_ma_app_id, ks_kfolder_no, ks_kc_id,
               kc.kc_from_c_id || '' || iif (pr.pr_code containing '.IPF.', '1', '0'), k_disb_kind, kc_from_c_id
        from kontstate, kontract, knclient kc, product pr, knaktlist kal, region_list(:reg_id, 1, 3, 1) reg
        where ks_from_k_id = k_id and
              ks_from_k_id = kc.kc_from_k_id and
              k_from_pr_id = pr.pr_id and
              kal.kal_from_kf_id = k_from_kf_id and
              :rep_date_end between kal.kal_date and kal.kal_date_end and
              kal.is_delete = 0 and
              kc.kc_amount_applicate > 0 and
              ks_state = 1 and
              ks_ol_is_delete = 0 and
              ks_date_begin between :rep_date_begin and :rep_date_end and
              ks_region_id = reg.id_list and
              (:lo_id = 0 or
              kal_new_kf_officer = :lo_id) and
              ks_date_end > :rep_date_end
        union -- кредиты, переданные в другой офис для выдачи (RFCKG-2179)
        select k_id, k_from_kf_id, ks_date_begin, k_status, kc_amount_approved, k_kolich_client_approved,
               k_from_ma_app_id, ks_kfolder_no, ks_kc_id,
               kc.kc_from_c_id || '' || iif (pr.pr_code containing '.IPF.', '1', '0'), k_disb_kind, kc_from_c_id
        from kontract
        inner join kontstate on ks_from_k_id = k_id and ks_ol_is_delete = 0
        inner join knclient kc on kc.kc_from_k_id = k_id and kc.kc_amount_applicate > 0
        inner join product pr on pr.pr_id = k_from_pr_id
        inner join miniapp on ma_id = k_from_ma_app_id and ma_is_active = 1
        inner join username on un_id = ma_from_un_id -- RFCKG-3970
        inner join region_list(:reg_id, 1, 3, 1) reg on ma_office_id = reg.id_list
        where :cm_database_id = 41701 and
              ks_state = 1 and -- выданные
              ks_date_begin between :rep_date_begin and :rep_date_end and
              ma_office_id <> ks_region_id and -- офис заявки и офис выдачи разные
              ma_number = 0 and
              un_from_c_id = :lo_id -- КС, за которым создана заявка (т.к. ma_number = 0)

        into :k_id, :kf_id, :res_date, :k_status, :amount, :k_client_count, ma_id, :kf_name, :res_lo_id,
             :k_client, :k_disb_kind, :kc_from_c_id
    do
    begin
      obj_id = k_id;
      res_app_amount = 0;
      res_disb_amount = amount;
      rep_plan_disb_date = null;
      if (coalesce(ma_id, 0) <> 0) then
      begin
        select ma_date_oper
        from miniapp
        where ma_date_app <= :rep_date_end and
              ma_date_end >= '31.12.2999' and
              ma_id = :ma_id and
              ma_is_active = 1
        order by ma_date_app desc
        rows 1
        into :rep_plan_disb_date;

        select sum(mac_amount)
        from miniappclient
        where mac_from_ma_id = :ma_id and
              mac_from_c_id = :kc_from_c_id
        into :res_app_amount;
      end
      res_memo = iif(cm_database_id <> 94401, 'выдан в срок', 'plan üzrə verilib');
      if (coalesce(rep_plan_disb_date, '01.01.1990') <> res_date) then
      begin
        if (rep_plan_disb_date is not null) then
          res_memo = iif(cm_database_id <> 94401, 'по плану', 'plan ilə') || ' ' || extract(day from rep_plan_disb_date) || ' ' ||(case
                                                                                                                                             when extract(month from rep_plan_disb_date) = 1 then iif(cm_database_id <> 94401, 'янв.', 'yanv')
                                                                                                                                             when extract(month from rep_plan_disb_date) = 2 then iif(cm_database_id <> 94401, 'фев.', 'fevr')
                                                                                                                                             when extract(month from rep_plan_disb_date) = 3 then iif(cm_database_id <> 94401, 'март', 'mart')
                                                                                                                                             when extract(month from rep_plan_disb_date) = 4 then iif(cm_database_id <> 94401, 'апр.', 'aprl')
                                                                                                                                             when extract(month from rep_plan_disb_date) = 5 then iif(cm_database_id <> 94401, 'май', 'may')
                                                                                                                                             when extract(month from rep_plan_disb_date) = 6 then iif(cm_database_id <> 94401, 'июнь', 'iyun')
                                                                                                                                             when extract(month from rep_plan_disb_date) = 7 then iif(cm_database_id <> 94401, 'июль', 'iyul')
                                                                                                                                             when extract(month from rep_plan_disb_date) = 8 then iif(cm_database_id <> 94401, 'авг.', 'avqs')
                                                                                                                                             when extract(month from rep_plan_disb_date) = 9 then iif(cm_database_id <> 94401, 'сен.', 'sent')
                                                                                                                                             when extract(month from rep_plan_disb_date) = 10 then iif(cm_database_id <> 94401, 'окт.', 'oktb')
                                                                                                                                             when extract(month from rep_plan_disb_date) = 11 then iif(cm_database_id <> 94401, 'ноя.', 'noyb')
                                                                                                                                             when extract(month from rep_plan_disb_date) = 12 then iif(cm_database_id <> 94401, 'дек.', 'dekb')
                                                                                                                                           end);
        else
          res_memo = iif(cm_database_id <> 94401, 'выдан без заявки', 'ərizəsiz verilən');
      end
      res_memo = kf_name || ': ' || res_memo;
      suspend;
    end

-- Все заявки
    ks_state = 0;
    for select ma_id, ma_from_kf_id, ma_date_oper, ma_is_active, mac_amount, ma_kolich_client, kal_new_kf_name,
               kal_new_kf_officer,
               mac.mac_from_c_id || '' || iif (pr.pr_code containing '.IPF.', '1', '0'), ma_disb_kind
        from miniapp, knaktlist, miniappclient mac, product pr, region_list(:reg_id, 1, 3, 1) reg
        where ma_date_app <= :rep_date_end and
              ma_date_end >= :rep_date_end and
              ma_date_oper between :rep_date_begin and :rep_date_end and
              ma_from_kf_id = kal_from_kf_id and
              mac.mac_from_ma_id = ma_id and
              ma_from_pr_id = pr.pr_id and
              mac.mac_amount > 0 and
              kal_date <= :rep_date_end and
              kal_date_end >= :rep_date_end and
              knaktlist.is_delete = 0 and
              ma_office_id = reg.id_list and
              (:lo_id = 0 or
              kal_new_kf_officer = :lo_id) and
              ma_is_active = 1
        into :ma_id, :kf_id, :res_date, :ma_is_active, :amount, :k_client_count, kf_name, res_lo_id,
             :k_client, k_disb_kind
    do
    begin
      need_suspend = 1;
      res_kind = iif(cm_database_id <> 94401, 'в процессе', 'prossesdə');

-- исключаем заявку если уже выдан (по ID заявки)
      if (exists(select *
                 from kontract, kontstate
                 where k_from_ma_app_id = :ma_id and
                       ks_from_k_id = k_id and
                       ks_ol_is_delete = 0 and
                       ks_state = 1 and
                       ks_date_end >= '31.12.2999')) then
        need_suspend = 0;

-- исключаем если выдан (по номер папки)
      if (exists(select *
                 from kontract, kontstate
                 where k_from_kf_id = :kf_id and
                       ks_from_k_id = k_id and
                       ks_ol_is_delete = 0 and
                       ks_date_begin = :res_date and
                       ks_state = 1 and
                       ks_date_end >= '31.12.2999')) then
        need_suspend = 0;

      if (not(res_date between :rep_date_begin and :rep_date_end)) then
        need_suspend = 0;

      if (need_suspend = 1) then
      begin
        obj_kind = 'Miniapp';
        obj_id = ma_id;
        res_app_amount = amount;
        res_disb_amount = 0;
        k_status = 0;

        res_memo = iif(cm_database_id <> 94401, 'Заявка', 'Ərizə');
        k_id = null;
        select k_id, k_status, kc_amount_approved, k_kolich_client_approved,
               kc.kc_from_c_id || '' || iif (pr.pr_code containing '.IPF.', '1', '0'), k_disb_kind
        from kontract k, kontstate, knclient kc, product pr
        where k_from_ma_app_id = :ma_id and
              k.k_from_pr_id = pr.pr_id and
              ks_from_k_id = k_id and
              kc.kc_from_k_id = k_id and
              kc.kc_amount_approved > 0 and
              ks_ol_is_delete = 0 and
              ks_state <> 1 and
              ks_date_end >= '31.12.2999' rows 1
        into :k_id, k_status, :amount, :k_client_count,
              :k_client, k_disb_kind;
        if (k_id is not null) then
        begin
          obj_kind = 'Kontract';
          obj_id = k_id;
          res_app_amount = amount;
          res_memo = '';
          if (k_status = -1) then
          begin
            res_memo = iif(cm_database_id <> 94401, 'Отказано', 'İmtina olunub');
            res_kind = iif(cm_database_id <> 94401, 'отказано', 'imtina olunub');
          end
          else
          begin
            res_kind = iif(cm_database_id <> 94401, 'в процессе', 'prossesdə');
            res_object_status = null;


            select rd_code
            from hp_get_check_loan_data(:k_id)
            into :rd_code;

            select ovs_add_value
            from hp_objectprops_get('Kontract', :k_id, -1, :rd_code, null)
            into :res_object_status;

            /*select ovs_add_value
            from hp_objectprops_get('Kontract', :k_id, -1, 'ActionGroup.Check_Loan_Data', null)
            into :res_object_status;*/
            if (res_object_status is null) then
            begin
              res_memo = iif(cm_database_id <> 94401, 'в процессе', 'prossesdə');
            end
            else
            begin
              res_memo = (case res_object_status
                            when -2 then f_format('КК: %1', iif(cm_database_id <> 94401, 'Отказано', 'İmtina olunub'))
                            when -1 then f_format('КК: %1', iif(cm_database_id <> 94401, 'Доработка', 'Tamamlama'))
                            when 0 then f_format('КК: %1', iif(cm_database_id <> 94401, 'В очереди', 'Növbədədir'))
                            when 1 then f_format('КК: %1', iif(cm_database_id <> 94401, 'Рассматр-ся', 'Baxılır'))
                            when 2 then f_format('КК: %1', iif(cm_database_id <> 94401, 'Подтверждено', 'Təsdiq olunub'))
                          end);
            end
          end
        end
        res_memo = kf_name || ': ' || res_memo;

        suspend;
      end
    end
  end
end
