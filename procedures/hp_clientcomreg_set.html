CREATE PROCEDURE HP_CLIENTCOMREG_SET(
    C_ID D_ID NOT NULL,
    UN_ID D_ID,
    ACTION_KIND D_STR_CODE NOT NULL,
    LIMIT_38 D_BOOL,
    LIMIT_152 D_BOOL,
    LIMIT_230 D_BOOL,
    REASON D_STR_CODE = 'manual')
AS
declare variable ccr_id d_integer;
declare variable ccl_action_kind d_smallint;
declare variable ccl_reason_kind d_smallint;
declare variable ccr_limit_38 d_bool;
declare variable ccr_limit_152 d_bool;
declare variable ccr_limit_230 d_bool;
begin
  if (action_kind not in ('add', 'remove')) then
    exception e_default 'Incorrect action kind:' || coalesce (action_kind, 'NULL');


  limit_38 = coalesce(limit_38, 0);
  limit_152 = coalesce(limit_152, 0);
  limit_230 = coalesce(limit_230, 0);

  -- Если ничего менять не нужно, выходим
  if (limit_38 = 0 and limit_152 = 0 and limit_230 = 0) then
    exit;

  select ccr_id
  from clientcomreg
  where ccr_from_c_id = :c_id
  into ccr_id;

  if (ccr_id is null) then
  begin
    -- Пока просто вставляем, далее скорректируем ограничения
    insert into clientcomreg (
      ccr_from_c_id, ccr_limit_38, ccr_limit_152, ccr_limit_230)
    values (:c_id, 0, 0, 0)
    returning ccr_id
    into :ccr_id;
  end

  -- Запоминаем сохраненные в БД значения для проверки необходимости обновления
  select ccr_limit_38, ccr_limit_152, ccr_limit_230
  from clientcomreg
  where ccr_id = :ccr_id
  into ccr_limit_38, ccr_limit_152, ccr_limit_230;

  if (action_kind = 'add') then
    ccl_action_kind = f_enum_id('clientcom_action', 'enabled'); -- 1
  else
    ccl_action_kind = f_enum_id('clientcom_action', 'disabled'); -- 0

  ccl_reason_kind = f_enum_id('clientcom_reason', reason);

  -- Обновляем реестр, ccl_action_kind содержит 1 или 0, в зависимости от action_kind
  update clientcomreg
  set
    ccr_limit_38 = iif(:limit_38 = 1, :ccl_action_kind, ccr_limit_38),
    ccr_limit_152 = iif(:limit_152 = 1, :ccl_action_kind, ccr_limit_152),
    ccr_limit_230 = iif(:limit_230 = 1, :ccl_action_kind, ccr_limit_230)
  where ccr_id = :ccr_id;

  -- Далее пишем логи, если устанавливаемое значение отличается от текущего
  if (limit_38 = 1 and ccr_limit_38 <> ccl_action_kind) then
    insert into clientcomlog (ccl_from_ccr_id, ccl_limit_kind, ccl_action_kind, ccl_reason_kind, ccl_from_un_id)
    values (:ccr_id, f_enum_id('clientcom_limit', 'limit_38'), :ccl_action_kind, :ccl_reason_kind, :un_id);

  if (limit_152 = 1 and ccr_limit_152 <> ccl_action_kind) then
    insert into clientcomlog (ccl_from_ccr_id, ccl_limit_kind, ccl_action_kind, ccl_reason_kind, ccl_from_un_id)
    values (:ccr_id, f_enum_id('clientcom_limit', 'limit_152'), :ccl_action_kind, :ccl_reason_kind, :un_id);

  if (limit_230 = 1 and ccr_limit_230 <> ccl_action_kind) then
    insert into clientcomlog (ccl_from_ccr_id, ccl_limit_kind, ccl_action_kind, ccl_reason_kind, ccl_from_un_id)
    values (:ccr_id, f_enum_id('clientcom_limit', 'limit_230'), :ccl_action_kind, :ccl_reason_kind, :un_id);
end
