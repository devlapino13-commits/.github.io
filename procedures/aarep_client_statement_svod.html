CREATE PROCEDURE AAREP_CLIENT_STATEMENT_SVOD(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_BALANCE_ID VARCHAR(20),
    REPORT_ACCOUNT_NO INTEGER,
    REPORT_CLIENT_ID D_ID,
    REP_BEGIN_DT DECIMAL(18,2),
    REP_BEGIN_KT DECIMAL(18,2),
    REP_END_DT DECIMAL(18,2),
    REP_END_KT DECIMAL(18,2),
    UN_ID D_ID)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    C_ADDRESS_YUR D_STR_255,
    C_ADDRESS_FACT D_STR_255,
    LAST_CELL_MAIN STR_20,
    DATE_BEGIN DATE,
    DATE_END DATE,
    KML_BANK_NO VARCHAR(50),
    KML_BANK_BIK ID_TYPE,
    KML_BANK_NAME VARCHAR(255),
    OB_DT_ALL DECIMAL(18,4),
    OB_KT_ALL DECIMAL(18,4),
    SALDO_BEGIN_DT DECIMAL(18,4),
    SALDO_BEGIN_KT DECIMAL(18,4),
    SALDO_END_DT DECIMAL(18,4),
    SALDO_END_KT DECIMAL(18,4),
    SS D_STR_255,
    COMPANY_NAME D_STR_100,
    RI_ID D_ID)
AS
declare variable adl_address_str d_str_255;
declare variable adc_address_type d_smallint;
declare variable curr_str varchar(1024);
begin
  saldo_begin_dt = coalesce(:rep_begin_dt, 0);
  saldo_begin_kt = coalesce(:rep_begin_kt, 0);
  saldo_end_dt = coalesce(:rep_end_dt, 0);
  saldo_end_kt = coalesce(:rep_end_kt, 0);

  c_name_fio = '';
  c_inn = '';
  select cl.c_name_fio, cl.c_inn
  from clients cl
  where cl.c_id = :report_client_id
  into :c_name_fio, :c_inn;

  last_cell_main = f_cont_info(0, :report_client_id, 'cell_main', 'ok');

  kml_bank_no = '';
  kml_bank_bik = 0;
  kml_bank_name = '';

  select bc.bc_account_no, bc.bc_from_b_id, b_name
  from bankclient bc, banks
  where bc.bc_from_c_id = :report_client_id and
        bc.bc_is_active = 1 and
        bc.bc_from_b_id = b_id rows 1
  into kml_bank_no, kml_bank_bik, kml_bank_name;

  c_address_yur = '';
  c_address_fact = '';
  for select adl.adl_address_str, adc.adc_address_type
      from addrclient adc
      inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
      where adc_from_c_id = :report_client_id and
            adc.adc_address_type in (1, 2)
      into :adl_address_str, adc_address_type
  do
  begin
    if (adc_address_type = 1) then
      c_address_yur = :adl_address_str;
    if (adc_address_type = 2) then
      c_address_fact = :adl_address_str;
  end

  select cm_name
  from company
  into :company_name;

  ss = '';
  if (rep_end_dt = rep_end_kt) then
    ss = 'На ' || f_datetostr(report_date_end) || ' год между ' || company_name || ' и ' || c_name_fio || ' нет задолженности';
  else
  begin
    if (rep_end_dt > 0) then
    begin
      select curr_str
      from fhp_currencystr(:rep_end_dt, 2)
      into :curr_str;

      ss = 'На ' || f_datetostr(report_date_end) || ' год между ' || company_name || ' и ' || c_name_fio || ' задолженность составляет ' || cast(rep_end_dt as  d_str_20) || ' (' || curr_str || ') в пользу ' || company_name;
    end
    else
    begin
      select curr_str
      from fhp_currencystr(:rep_end_kt, 2)
      into :curr_str;

      ss = 'На ' || f_datetostr(report_date_end) || ' год между ' || company_name || ' и ' || c_name_fio || ' задолженность составляет ' || cast(rep_end_kt as  d_str_20) || ' (' || curr_str || ') в пользу ' || c_name_fio;
    end
  end

  date_begin = :report_date_begin;
  date_end = :report_date_end;

  select sum(sta_ob_dt), sum(sta_ob_kt)
  from aarep_client_statement(:report_region_id, :report_date_begin, :report_date_end, :report_balance_id, :report_account_no, :report_client_id)
  into :ob_dt_all, ob_kt_all;



  suspend;
end
