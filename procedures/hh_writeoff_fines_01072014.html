CREATE PROCEDURE HH_WRITEOFF_FINES_01072014
RETURNS (
    K_ID ID_TYPE,
    OL_ID ID_TYPE,
    OL_DATE DATE,
    OK_NUMBER SMALLINT,
    SH_ID SMALLINT,
    SH_DATE DATE,
    SH_CALC_FINE CURR_TYPE,
    OK_OL_REG_ID INTEGER,
    K_KONTRACT_NO VARCHAR(20),
    REG_NAME VARCHAR(100))
AS
declare variable rep_date_begin date;
declare variable rep_date_end date;
declare variable k integer;
begin
-- Проверять всех контрактов выданные с 1 июля 2014г. и погашались пени,
-- если после выплаты всех сумм включая пени(пеня обязательно) закрылась просрочка,
--  но после перенастройки продукта выявились пени то их нужно списать с тем платежем в котором они погашались

  rep_date_begin = '01.07.2014';
  rep_date_end = current_date;
  for select ok_from_k_id, ok_from_ol_id, ok_ol_date, ok_ol_reg_id, k_kontract_no
      from operkred, kontract
      where ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_type in (4, 5, 6) and
            ok_from_k_id = k_id and
            ok_from_c_id = 0 and
            k_disb_date >= '01.07.2014' -- брать только выданные с 1 июля 2014 года
            and
            ok_fine < 0 -- если погашал пенью то их проверяем
      group by ok_from_k_id, ok_from_ol_id, ok_ol_date, ok_ol_reg_id, k_kontract_no
      having sum(ok_os + ok_perc + ok_fine) < 0
      into k_id, ol_id, ol_date, ok_ol_reg_id, k_kontract_no
  do
  begin
--  проверка на дату погашения
    k = 0;
    ok_number = 0;
    if (not exists(select *
                   from oksummary
                   where :ol_date between oks_date_begin and oks_date_end and
                         :ol_date > oks_delinq_date and
                         oks_from_k_id = :k_id and
                         oks_close_state = 0)) then -- после погашения не имеется просрчока то, по ним если есть пеня то будет списываться с тем же платежем
      for select sh_id, sh_date, sh_calc_fine
          from aarep_make_zad_list(:k_id, :ol_date, 0)
          where sh_calc_fine > 0 and
                sh_date is not null
          into sh_id, sh_date, sh_calc_fine
      do
      begin
        if (ok_number = 0) then
        begin
          ok_number = (select max(ok_number)
                       from operkred
                       where ok_from_ol_id = :ol_id and
                             ok_from_c_id = 0);
        end

        k = k + 1;
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                              ok_ol_reg_id, ok_ol_is_active)
        values (:ol_id, :ok_number + :k, :k_id, :sh_id, :sh_date, 5, 0, 0, -:sh_calc_fine, 3, 0, 0, 0, 0, 0, :ol_date,
                :ok_ol_reg_id, 1);

--    Данные для результата
        reg_name = (select reg_code || ' ' || reg_name
                    from region
                    where reg_unicode = :ok_ol_reg_id);

        suspend;
      end
  end

end
