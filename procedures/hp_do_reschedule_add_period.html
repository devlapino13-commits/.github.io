CREATE PROCEDURE HP_DO_RESCHEDULE_ADD_PERIOD(
    ADD_MONTH_COUNT D_INTEGER,
    ADD_MONTH_FIRST D_INTEGER,
    NEW_PAY_DAY D_INTEGER = 0)
AS
declare variable kol d_integer;
declare variable sh_date d_date;
declare variable sh_prior_percent_per_day d_percent;
declare variable sh_date_orig d_date;
declare variable isweekend d_smallint;
declare variable sh_id d_integer;
declare variable sh_date_min d_date;
declare variable sh_date_max d_date;
declare variable sh_rec_min d_integer;
declare variable sh_rec_max d_integer;
begin
  select sh_id, sh_date, sh_prior_percent_per_day
  from temp_sh
  order by sh_id desc
  rows 1
  into sh_id, sh_date, sh_prior_percent_per_day;

  begin
    sh_date = right('0' || new_pay_day, 2) || '.' || right('0' || extract(month from sh_date), 2) || '.' || extract(year from sh_date);
    when any do
    begin
    end
  end

  kol = 0;
  sh_date_orig = sh_date;
  while (kol < add_month_count) do
  begin
    kol = kol + 1;
    sh_id = sh_id + 1;
    sh_date = dateadd(month, kol, sh_date_orig);

    if (extract(weekday from sh_date) = 0)
    then sh_date = sh_date + 1; -- воскресенье переносим на ПН
    else
      if (extract(weekday from sh_date) = 6)
      then sh_date = sh_date + 2; -- субботу переносим на ПН

    isweekend = 1;
    while (isweekend = 1) do
    begin
      isweekend = 0;

      -- Проверяем на праздники
      if (exists(select *
                 from sa_calendar
                 where scal_month = extract(month from :sh_date) and
                       scal_day = extract(day from :sh_date) and
                       scal_active = 1)) then
        isweekend = 1;

      if (isweekend = 0) then
      begin
        -- Проверяем на календарь рабочих дней
        if (exists(select *
                   from sa_calendays
                   where scd_work_hours = 0 and
                         scd_date = :sh_date)) then
          isweekend = 1;
      end

      if (isweekend = 1) then
        sh_date = dateadd(day, 1, sh_date);
    end

    insert into temp_sh (sh_id, sh_date, sh_repay_mode, sh_prior_percent_per_day)
    values (:sh_id, :sh_date, 3, :sh_prior_percent_per_day);
  end

  if (add_month_first > 0) then
  begin
    if (f_get_db_id() = 81001) then
    begin
      sh_rec_min = 0;
      sh_rec_max = 0;
  
      with tab as
      (select sh_rec_id
      from temp_sh
      where sh_repay_mode in (2, 3) and
            (sh_zad_os <> 0 or sh_zad_perc <> 0 or sh_date > :sh_date_orig)
      rows :add_month_first)
      select min(sh_rec_id), max(sh_rec_id)
      from tab
      into sh_rec_min, sh_rec_max;
  
      update temp_sh
      set sh_is_grace_os = 1,
          sh_is_grace_perc = 1
      where sh_rec_id between :sh_rec_min and :sh_rec_max;
    end else
    begin
      sh_date_min = null;
      sh_date_max = null;
  
      with tab as
      (select distinct sh_date
      from temp_sh
      where sh_repay_mode in (2, 3) and
            (sh_zad_os <> 0 or sh_zad_perc <> 0 or sh_date > :sh_date_orig)
      rows :add_month_first)
      select min(sh_date), max(sh_date)
      from tab
      into sh_date_min, sh_date_max;
  
      update temp_sh
      set sh_is_grace_os = 1,
          sh_is_grace_perc = 1
      where sh_date between :sh_date_min and :sh_date_max;
    end
  end
end
