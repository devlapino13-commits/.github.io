Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$BONDLIST(
    HIDE_FRAC D_BOOL = 1)
RETURNS (
    BNL_ID D_CURRENCY,
    BNL_TEXT D_STR_CODE)
AS
declare variable bond_digit_str d_str_code;
declare variable bond_frac_str d_str_code;
begin
  bond_digit_str = uxf$const('currency_name');
  bond_frac_str = uxf$const('currency_cent');

  for select bnd_id
  from ux$bonds
  where bnd_id >= :hide_frac
  order by bnd_id desc
  into :bnl_id
  do
  begin
    if (bnl_id >= 1) then
      bnl_text = uxf$numtostr(round(bnl_id), 0) || ' ' || bond_digit_str;
    else
      bnl_text = uxf$numtostr(mod(round(bnl_id * 100.00), 100), 0) || ' ' || bond_frac_str;

    suspend;
  end
end
