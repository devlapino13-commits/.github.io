CREATE PROCEDURE HP_EXTINFO_PASS_STATUS_PROCESS(
    EIL_ID D_ID)
AS
declare variable RD_ID D_ID;
declare variable C_ID D_ID;
declare variable EIC_CODE D_STR_20;
declare variable BLS_ID D_ID;
declare variable RES_TEXT D_STR_255;
declare variable GC_STATUS D_INTEGER;
declare variable RD_MARITAL_STATUS D_INTEGER;
begin
    rd_id = 0;
    c_id = 0;
    bls_id = 0;
    eic_code = '';
    res_text = '';

    select
        eic.eic_code,
        eil.eil_from_c_id,
        eil.eil_from_bls_id
    from extinfolist eil
    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
        and eic.eic_code in ('passport_status', 'marital_status')
    where eil.eil_id = :eil_id
        and eil.eil_status = 2
        and eil.eil_from_bls_id is not null
    into
        eic_code,
        c_id,
        bls_id;

    if (coalesce(c_id, 0) = 0) then
        exit;

    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into res_text;

    if (res_text is null or res_text = '') then
        select json_text
        from temp_grs
        where bls_id = :bls_id
        into res_text;

    res_text = replace(res_text, '"', '');
    if (eic_code = 'passport_status') then
    begin
        res_text = lower(res_text);
        if (res_text = 'valid') then
        begin
            select gc.gc_status
            from grs_clients gc
            where gc.gc_from_c_id = :c_id and
                  gc.gc_is_last = 1
            into gc_status;

            update grs_clients
            set gc_is_last = 0
            where gc_from_c_id = :c_id
                and gc_is_last = 1;
            
            insert into grs_clients (gc_from_c_id, gc_from_eil_id, gc_status, gc_status_datetime, gc_is_last)
            values (:c_id, :eil_id, :gc_status, current_timestamp, 1);
        end
        else
        begin
            update grs_clients
            set gc_is_last = 0
            where gc_from_c_id = :c_id
                and gc_is_last = 1;
            
            insert into grs_clients (gc_from_c_id, gc_from_eil_id, gc_status, gc_status_datetime, gc_is_last)
            values (:c_id, :eil_id, -4, current_timestamp, 1);
            
            update clients c
            set c.c_verify_status = -2
            where c.c_id = :c_id;
        end
    end
    else
    begin
        rd_id = f_get_rd_id('AnketaQuestions.Family_tunduk');
        rd_marital_status =
            case :res_text
                when 'Состоит в браке' then f_get_rd_id('family.married')
                when 'В браке не состоял(а)' then f_get_rd_id('family.unmarried')
                when 'Разведен(а)' then f_get_rd_id('family.divorced')
                when 'Вдовец/вдова' then f_get_rd_id('family.widower')
            end;

        execute procedure hp_objectprops_set_with_rd('clients', :c_id, -1, :rd_id, current_date, '', 0, :rd_marital_status, 1000000000000);
    end
end
