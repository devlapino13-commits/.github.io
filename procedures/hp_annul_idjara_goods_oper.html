CREATE PROCEDURE HP_ANNUL_IDJARA_GOODS_OPER(
    OBJ_ID D_ID)
AS
declare variable k_id d_id;
declare variable ag_id d_id;
declare variable ol_id d_id;
begin
  select kma.kma_from_k_id, kma.kma_from_ag_id
  from kontmatag kma
  where kma.kma_id = :obj_id
  into k_id, ag_id;

  for select kml.kml_bill_id
      from kontmat kml
      where kml.kml_object_kind = 9 and
            kml.kml_object_id = :k_id and
            kml.kml_from_ag_id = :ag_id
      into ol_id
  do
  begin
    update operlist ol
    set ol.is_delete = 1
    where ol.ol_id = :ol_id;
  end 
end
