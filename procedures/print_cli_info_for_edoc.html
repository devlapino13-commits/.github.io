CREATE PROCEDURE PRINT_CLI_INFO_FOR_EDOC(
    OBJ_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_PASS_NO D_STR_100,
    C_PASS_WHO D_STR_255,
    C_PASS_DATE D_DATE,
    C_ADDRESS_YUR D_STR_255,
    LAST_CELL_MAIN D_STR_100,
    C_EMAIL D_STR_100,
    QR_CODE D_STR_LONG,
    REG_NAME D_STR_100,
    CONSULT_LINK D_STR_255,
    PRICE D_CURRENCY,
    CONS_LEVEL D_INTEGER,
    LEVEL_NAME D_STR_100,
    ACCESS_TIME D_STR_100,
    LEVEL_NAME_KG D_STR_100,
    ACCESS_TIME_KG D_STR_100)
AS
declare variable KC_ID D_ID;
declare variable C_ID D_ID;
declare variable EDL_ID D_ID;
declare variable DOC_DATE D_DATE;
declare variable object_kind  d_str_100;
begin
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);

    select cast(edl.edl_create_datetime as date),
           f_objkind(edl.edl_object_kind, 'code')
    from edoc_list edl where edl.edl_id = :edl_id
    into :doc_date,
         :object_kind;
  end

  if (doc_date is null) then
    doc_date = current_date;

  if (object_kind = 'knclient') then       -- RFCKG-4363
  begin
    kc_id = obj_id;

    select kc.kc_from_c_id
    from knclient kc
    inner join kontract kont on kont.k_id = kc.kc_from_k_id
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    where kc.kc_id = :kc_id
    and kont.k_status = 1
    and pr.pr_code containing '.consultation.'
    and kont.k_disb_date  >= :doc_date
    order by k_disb_date
    rows 1
    into :c_id;
  end
  else
  begin
    c_id = obj_id;

    select kncl.kc_id
    from knclient kncl
    inner join kontract kont on kont.k_id = kncl.kc_from_k_id
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    where kncl.kc_from_c_id = :c_id
    and kont.k_status = 1
    and pr.pr_code containing '.consultation.'
    and kont.k_disb_date  >= :doc_date
    order by k_disb_date
    rows 1
    into :kc_id;
  end

  qr_code = f_edoc_qr_by_edl(edl_id, c_id);

  select kc_from_c_id, reg_name, 'https://consult.mbulak.kg/?uid=' || f_uuid('knclient', kc_id) as consult_link
  from knclient kc
  inner join kontract k on kc_from_k_id = k_id
  inner join region r on k_last_reg_id = reg_unicode
  where kc_id = :kc_id
  into c_id, reg_name, consult_link;

  select c_name_fio, c.c_pass_no, c.c_pass_who, c_pass_date
  from clients c
  where c_id = :c_id
  into :c_name_fio, :c_pass_no, :c_pass_who, :c_pass_date;

  select adc_address_str
  from hp_addrclient(:c_id)
  where adc_address_type = 1
  into :c_address_yur;

  last_cell_main = f_cont_info(0, :c_id, 'cell_main', 'ok');
  c_email = f_cont_info(0, :c_id, 'email', 'ok');

  select price, cons_level, level_name, access_time_str, level_name_kg, cast(access_time as d_str_code) || ' ай' as access_time
  from clt_get_price_for_consultation(:kc_id)
  into price, cons_level, level_name, access_time, level_name_kg, access_time_kg;

  suspend;
end
