CREATE PROCEDURE UXP$UXFRPARAM_SEARCH_UPDATE(
    UFP_UFR_UUID D_UUID,
    UFP_UF_UUID D_UUID,
    UA_USR_TEXT_FIELD D_STR_CODE,
    USR_CODE D_STR_100,
    IS_DIALOG D_BOOL,
    UA_NAME D_STR_100)
AS
declare variable ube_frame_id_field d_str_code;
begin
  if (trim(coalesce(ufp_ufr_uuid, '')) = '') then
    exception e_default f_format('Frame UUID is empty');

  if (trim(coalesce(ufp_uf_uuid, '')) = '') then
    exception e_default f_format('Field UUID is empty');

  ube_frame_id_field = uxf$code(ufp_uf_uuid, 'code');

  if (not exists(select * from ux$search
  where usr_uuid = uxf$uniq(:usr_code))) then
    exception e_default f_format('Поисковик не найден: %1', usr_code);

  delete from ux$action
  where ua_owner = :ufp_ufr_uuid and
        ua_usr_id_field = :ube_frame_id_field;

  update or insert into ux$action (ua_owner, ua_uxa_fcode, ua_name, ua_usr_id_field, ua_usr_text_field)
  values (:ufp_ufr_uuid, :usr_code, :ua_name, :ube_frame_id_field, :ua_usr_text_field)
  matching (ua_owner, ua_uxa_fcode, ua_usr_id_field);

  if (is_dialog = 0) then
  begin
    update ux$frparam
    set ufp_editor = 'buttonedit'
    where ufp_ufr_uuid = :ufp_ufr_uuid and
          ufp_uf_uuid = :ufp_uf_uuid and
          ufp_editor <> 'buttonedit';
  end else
  begin
    update ux$frparam
    set ufp_editor = 'dialogedit'
    where ufp_ufr_uuid = :ufp_ufr_uuid and
          ufp_uf_uuid = :ufp_uf_uuid and
          ufp_editor <> 'dialogedit';
  end
end
