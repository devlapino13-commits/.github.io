CREATE PROCEDURE AAREP_OBORSALDO_OPEREXTRA(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ACC_BALANCE_NO VARCHAR(20),
    ACCOUNT VARCHAR(20),
    REG_NAME VARCHAR(100),
    K_ID ID_TYPE,
    K_KONTRACT_NO VARCHAR(20),
    K_AMOUNT CURR_TYPE,
    BEG_DT CURR_TYPE,
    BEG_KT CURR_TYPE,
    OB_DT CURR_TYPE,
    OB_KT CURR_TYPE,
    END_DT CURR_TYPE,
    END_KT CURR_TYPE,
    KOL_CLI INTEGER)
AS
declare variable id_list integer;
declare variable reg_code varchar(10);
declare variable k_from_pr_id integer;
declare variable acc_no integer;
declare variable k_index_curr_code varchar(3);
declare variable pr_id_prior integer;
declare variable ks_date_begin date;
declare variable oek_rd_id integer;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  pr_id_prior = 0;
  beg_dt = 0;
  beg_kt = 0;
  end_dt = 0;
  end_kt = 0;

  acc_balance_no = '76-02';
  acc_no = 5;

  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into id_list
  do
  begin
    select reg_code, reg_name
    from region
    where reg_unicode = :id_list
    into reg_code, reg_name;
    reg_name = reg_code || ' ' || reg_name;

    for with tabl1
        as (select k_id, k_kontract_no, k_amount_approved, k_index_curr_code, k_kolich_client_approved,

                   iif(ox_ol_date < :rep_date_begin, ox_amount_plus - ox_amount_minus - ox_amount_writeoff, 0) as beg_dt,
                   iif(ox_ol_date between :rep_date_begin and :rep_date_end, ox_amount_plus, 0) as ob_dt,
                   iif(ox_ol_date between :rep_date_begin and :rep_date_end, ox_amount_minus + ox_amount_writeoff, 0) as ob_kt
            from operextra
            inner join kontract k on ox_object_id = k_id
            inner join knaktlist kal on (k_from_kf_id = kal_from_kf_id and
                  kal.is_delete = 0 and
                  :rep_date_end between kal_date and kal_date_end and
                  kal.kal_new_reg_id = :id_list)
            where ox_ol_is_active = 1 and
                  ox_object_kind = 9 and
                  ox_from_rd_id = :oek_rd_id and
                  ox_ol_date <= :rep_date_end)
        select k_id, k_kontract_no, k_amount_approved, k_index_curr_code, k_kolich_client_approved, sum(beg_dt),
               sum(ob_dt), sum(ob_kt)
        from tabl1
        group by k_id, k_kontract_no, k_amount_approved, k_index_curr_code, k_kolich_client_approved
        having sum(beg_dt) <> 0 or sum(ob_dt) <> 0 or sum(ob_kt) <> 0
        into k_id, k_kontract_no, k_amount, k_index_curr_code, kol_cli, beg_dt, ob_dt, ob_kt
    do
    begin
      end_dt = beg_dt + ob_dt - ob_kt;

      account = reg_code || k_index_curr_code || acc_balance_no || '00000' || acc_no;

      suspend;
    end
  end
end
