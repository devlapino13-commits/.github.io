CREATE PROCEDURE PRINT_CLIENT_ZADOL_ON_DATE(
    PRINT_K_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    CZ_OS DECIMAL(18,4),
    CZ_PERC DECIMAL(18,4),
    CZ_FINE DECIMAL(18,4),
    CZ_ALL DECIMAL(18,4),
    CZ_DATE DATE,
    CZ_OS_PERC DECIMAL(18,4),
    CZ_DEL_DAYS INTEGER,
    CZ_FROM_C_ID DECIMAL(18,0),
    CZ_NAME_FIO VARCHAR(100),
    CZ_AMOUNT DECIMAL(18,4),
    CZ_CL_OS DECIMAL(18,4),
    CZ_CL_PERC DECIMAL(18,4),
    CZ_CL_FINE DECIMAL(18,4),
    CZ_CL_ALL DECIMAL(18,4),
    CZ_CL_OS_PERC DECIMAL(18,4),
    CZ_CL_TAX DECIMAL(18,4),
    CZ_CL_ALL_TAX DECIMAL(18,4),
    CZ_OS_SP DECIMAL(18,4),
    CZ_ALL_SP DECIMAL(18,4),
    CZ_CL_OS_SP DECIMAL(18,4),
    CZ_CL_ALL_SP DECIMAL(18,4))
AS
declare variable sh_date date; /* CZ_CL_OS_PERC */
declare variable sh_calc_fine decimal(18,4);
declare variable sh_calc_perc decimal(18,4);
declare variable zad_os decimal(18,4);
declare variable sh_min_date date;
declare variable kont_amount decimal(18,4);
declare variable st_staff integer;
declare variable st_client integer;
declare variable sum_os decimal(18,4) = 0;
declare variable sum_perc decimal(18,4) = 0;
declare variable sum_fine decimal(18,4) = 0;
declare variable sum_loan decimal(18,4) = 0;
declare variable k_disb_date date;
declare variable k_nsp_percent decimal(18,7);
declare variable sh_perc_cur decimal(18,4);
declare variable sh_kommis_cur decimal(18,4);
declare variable sh_perc_temp decimal(18,4);
declare variable sh_kommis_temp decimal(18,4);
declare variable initial_tax_cur decimal(18,4);
declare variable payed_tax decimal(18,4);
declare variable kz_tax_cur decimal(18,4);
declare variable sum_tax decimal(18,4);
declare variable k_stop_perc date;
declare variable sum_os_sp decimal(15,2);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  cz_os = 0;
  cz_os_sp = 0;
  cz_perc = 0;
  cz_fine = 0;
  cz_all = 0;
  cz_all_sp = 0;
  sh_min_date = null;

  select k_amount_approved, k_disb_date, kont.k_stop_perc
  from kontract kont
  where k_id = :print_k_id
  into :kont_amount, :k_disb_date, :k_stop_perc;

  for select sh_date, sh_calc_fine, sh_calc_perc, zad_os
      from aarep_make_zad_list(:print_k_id, :rep_date, 1)
      where sh_date is not null
      into :sh_date, :sh_calc_fine, :sh_calc_perc, :zad_os
  do
  begin
    if ((sh_min_date is null) and
        (sh_date < :rep_date)) then
    begin
      if ((zad_os <> 0) or (sh_calc_perc <> 0) or (sh_calc_fine <> 0)) then
        sh_min_date = :sh_date;
    end
    if (sh_date <= :rep_date) then
    begin
      cz_os = cz_os + zad_os;
      cz_fine = cz_fine + sh_calc_fine;
      cz_perc = cz_perc + sh_calc_perc;
    end

    if (k_stop_perc is not null) then
      cz_os_sp = cz_os_sp + zad_os;
    else
      cz_os_sp = cz_os;
  end

  cz_all = cz_os + cz_perc + cz_fine;
  cz_os_perc = cz_os + cz_perc;

  if (k_stop_perc is not null) then
    cz_all_sp = cz_os_sp + cz_perc + cz_fine;
  else
    cz_all_sp = cz_all;

  cz_date = :rep_date;
  if (sh_min_date is not null) then
    cz_del_days = (rep_date - sh_min_date);
  else
    cz_del_days = 0;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end

  sh_perc_cur = 0;
  sh_kommis_cur = 0;

  for select coalesce(sh_perc, 0), coalesce(sh_kommis, 0)
      from schedule
      where sh_date < :rep_date and
            sh_from_k_id = :print_k_id
      into :sh_perc_temp, :sh_kommis_temp
  do
  begin
    sh_perc_cur = sh_perc_cur + sh_perc_temp;
    sh_kommis_cur = sh_kommis_cur + sh_kommis_temp;
  end
  initial_tax_cur = cast(((sh_kommis_cur + sh_perc_cur) * k_nsp_percent / 100) as decimal(18,2));
  select sum(ok.ok_tax)
  from operkred ok
  where ok.ok_from_k_id = :print_k_id and
        ok.ok_ol_is_active = 1 and
        ok_from_c_id = 0
  into :payed_tax;

  payed_tax = coalesce(payed_tax, 0);
  kz_tax_cur = initial_tax_cur + payed_tax;

  sum_os = 0;
  sum_perc = 0;
  sum_fine = 0;
  sum_loan = 0;
  sum_tax = 0;
  sum_os_sp = 0;

  for select c_name_fio, kc_amount_approved, kc_from_c_id
      from clients, knclient
      where kc_from_k_id = :print_k_id and
            kc_from_c_id = c_id and
            kc_amount_approved is not null and
            kc_amount_approved <> 0
      into :cz_name_fio, :cz_amount, cz_from_c_id
  do
  begin
    sum_loan = sum_loan + cz_amount;
    if (sum_loan < kont_amount) then
    begin
      cz_cl_os = cast(cz_os as decimal(15,2)) * cz_amount; -- Было cz_cl_os= cz_os as * CZ_AMOUNT;
      cz_cl_os = cast(cz_cl_os / kont_amount as decimal(15,2));
      sum_os = sum_os + cz_cl_os;

      if (k_stop_perc is not null) then
      begin
        cz_cl_os_sp = cast(cz_os_sp as decimal(15,2)) * cz_amount; -- Было cz_cl_os= cz_os as * CZ_AMOUNT;
        cz_cl_os_sp = cast(cz_cl_os_sp / kont_amount as decimal(15,2));
        sum_os_sp = sum_os_sp + cz_cl_os_sp;
      end
      else
      begin
        cz_cl_os_sp = cz_cl_os;
        sum_os_sp = sum_os;
      end

      cz_cl_perc = cz_perc * cz_amount;
      cz_cl_perc = cast(cz_cl_perc / kont_amount as decimal(15,2));
      sum_perc = sum_perc + cz_cl_perc;

      cz_cl_fine = cz_fine * cz_amount;
      cz_cl_fine = cast(cz_cl_fine / kont_amount as decimal(15,2));
      sum_fine = sum_fine + cz_cl_fine;

      cz_cl_tax = kz_tax_cur * cz_amount;
      cz_cl_tax = cast(cz_cl_tax / kont_amount as decimal(15,2));
      sum_tax = sum_tax + cz_cl_tax;
    end
    else
    begin
      cz_cl_os = cz_os - sum_os;

      if (k_stop_perc is not null) then
        cz_cl_os_sp = cz_os_sp - sum_os_sp;
      else
        cz_cl_os_sp = cz_cl_os;

      cz_cl_perc = cz_perc - sum_perc;
      cz_cl_fine = cz_fine - sum_fine;
      cz_cl_tax = kz_tax_cur - sum_tax;
    end
    cz_cl_all = cz_cl_os + cz_cl_perc + cz_cl_fine;

    if (k_stop_perc is not null) then
      cz_cl_all_sp = cz_cl_os_sp + cz_cl_perc + cz_cl_fine;
    else
      cz_cl_all_sp = cz_cl_all;

    cz_cl_os_perc = cz_cl_os + cz_cl_perc;
    cz_cl_all_tax = cz_cl_os + cz_cl_perc + cz_cl_fine + cz_cl_tax;
    suspend;
  end

end
