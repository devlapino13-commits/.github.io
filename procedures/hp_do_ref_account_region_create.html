CREATE PROCEDURE HP_DO_REF_ACCOUNT_REGION_CREATE(
    REGION_ID INTEGER)
AS
declare variable sab_full_code d_str_255;
declare variable acc_id id_type;
declare variable ol_date date;
declare variable reg_id d_integer;
begin
  ol_date = '31.12.2017';

  for select reg_unicode
      from region
      where reg_status = 3 and
            coalesce(reg_date_close, '31.12.2999') >= :ol_date and
            (:region_id = 0 or
            reg_unicode = :region_id)
      into reg_id
  do
  begin
    for select sab_full_code
        from sys_accbalance
        where (sab_full_code starting 'деньги.нал.' or
              sab_full_code starting 'микрозаймы.') and
              sab_create_kind <> 0 and
              sab_acc_mask like '%.rrrr' and
              sab_acc_mask not containing 'A'
        into sab_full_code
    do
    begin
      acc_id = null;

      select acc_id
      from get_ref_account_office(:sab_full_code, '', :reg_id, '810', -1, 0, 3, :ol_date)
      into acc_id;

      if (acc_id is not null) then
      begin
        if (sab_full_code = 'Деньги.Нал.Касса') then
        begin
          update acc_list
          set al_from_acc_id = :acc_id
          where al_from_reg_id = :reg_id and
                al_from_bg_id = '50-01' and
                al_currency = '810' rows 1;
        end

        if (sab_full_code = 'Деньги.Нал.вПути') then
        begin
          update acc_list
          set al_from_acc_id = :acc_id
          where al_from_reg_id = :reg_id and
                al_from_bg_id = '57-01' rows 1;
        end
      end

      if (sab_full_code = 'Деньги.Нал.Касса') then
      begin
        acc_id = null;

        select acc_id
        from get_ref_account_office(:sab_full_code, '', :reg_id, '840', -1, 0, 3, :ol_date)
        into acc_id;

        if (acc_id is not null) then
        begin
          update acc_list
          set al_from_acc_id = :acc_id
          where al_from_reg_id = :reg_id and
                al_from_bg_id = '50-11' and
                al_currency = '840' rows 1;
        end

        acc_id = null;

        select acc_id
        from get_ref_account_office(:sab_full_code, '', :reg_id, '978', -1, 0, 3, :ol_date)
        into acc_id;

        if (acc_id is not null) then
        begin
          update acc_list
          set al_from_acc_id = :acc_id
          where al_from_reg_id = :reg_id and
                al_from_bg_id = '50-11' and
                al_currency = '978' rows 1;
        end
      end

      if (sab_full_code starting 'микрозаймы.') then
      begin
        select acc_id
        from get_ref_account_office(:sab_full_code, '', :reg_id, '810', -1, 1, 3, :ol_date)
        into acc_id;
      end

    end
  end
end
