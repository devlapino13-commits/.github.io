CREATE PROCEDURE CM_RU_0138_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable oko_sum decimal(15,2);
declare variable oko_k_id id_type;
begin
  if (k_id in (101010046269901101, 101010046269901201)) then
    exit;

  for select distinct CLNT.C_NAME_FIO, OKO.OKO_FROM_K_ID
      from KNCLIENT KNCL
      inner join CLIENTS CLNT on CLNT.C_ID = KNCL.KC_FROM_C_ID
      inner join OPERKROFF OKO on OKO.OKO_FROM_C_ID = KNCL.KC_FROM_C_ID
      where KNCL.KC_FROM_K_ID = :K_ID and
            OKO.OKO_OL_IS_ACTIVE = 1 and
            KNCL.KC_AMOUNT_APPROVED > 0
      into :res_info_text, :OKO_K_ID
  do
  begin
    OKO_SUM = null;
    select sum(OKO.OKO_OS + OKO.OKO_PERC + OKO.OKO_OTHER --
           + OKO.OKO_SH_OS + OKO.OKO_SH_PERC + OKO.OKO_SH_OTHER) OKO_ZAD
    from OPERKROFF OKO
    where OKO.OKO_FROM_K_ID = :OKO_K_ID
    and oko.oko_ol_is_active = 1
    into :OKO_SUM;
    OKO_SUM = coalesce(OKO_SUM, 0);
    if (:OKO_SUM > 0) then
    begin
      res_error_text =
      'У клиента имеется задолженность на внесистемном учете. Сумма долга: ' || :OKO_SUM;
      res_error_is_critical = 1;
      suspend;
    end
  end
end
