CREATE PROCEDURE HP_BKI_INDICATOR_LIST(
    PARENT_BKI_CODE D_STR_CODE,
    XML_VERSION_ID D_INTEGER)
RETURNS (
    BFV_ID D_ID,
    PARENT_ID D_ID,
    DB_FIELD D_STR_CODE,
    XML_NODE D_STR_CODE,
    BKI_CODE D_STR_CODE,
    BKI_NAME D_STR_255)
AS
begin
  for with recursive tab as(
    select bfv_id, bki_code, bki_name, bki_db_field, bfv_parent_id, bxf_xml_node
    from bki_xml_fields_ver, bki_xml_fields, bki_indicators
    where bxf_id = bfv_from_bxf_id
      and bki_code = bxf_from_bki_code
      and bfv_from_bxv_id = :xml_version_id
    order by bfv_position
  ), t1 (id, code, name, db_field, parent_id, xml) as(
    select bfv_id, bki_code, bki_name, bki_db_field, bfv_parent_id, bxf_xml_node
    from tab
    where bki_code = :parent_bki_code
    union all
    select bfv_id, bki_code, bki_name, bki_db_field, bfv_parent_id, bxf_xml_node
    from t1, tab
    where bfv_parent_id = id
  )
  select id, parent_id, code, name, db_field, xml
  from t1
  into bfv_id, parent_id, bki_code, bki_name, db_field, xml_node
  do
    suspend;
end
