CREATE PROCEDURE HP_OVERPAY_FIX_OPERBOOK(
    REP_DATE D_DATE = null)
RETURNS (
    RES_TXT D_STR_CODE,
    K_ID D_ID,
    OPL_DATE D_DATE,
    REPAY_OL_ID_MAIN D_ID,
    REPAY_OL_ID D_ID,
    OPL_SUM D_CURRENCY,
    KOB_SUMMA D_CURRENCY,
    NEW_OL_ID_MAIN D_ID,
    NEW_OL_ID D_ID,
    REG_ID_PAY D_ID,
    REG_ID_REPAY D_ID,
    OL_MEMO D_STR_255)
AS
declare variable kc_id d_id;
declare variable new_kob_id d_id;
declare variable obd_detail d_str_255;
declare variable bank_acc_id d_id;
declare variable acc_id d_id;
declare variable kob_currency d_str_10;
begin
  for select opl_from_k_id, opl_date, opl_sum
  from overpay_main, overpay_loans
  where opl_from_opm_id = opm_id
    and opm_status = 3
    and opl_type = 1
    and opl_date = coalesce(:rep_date, opl_date)
--  rows 1
  into k_id, opl_date, opl_sum
  do
  begin
    kc_id = (select kc_id from knclient where kc_from_k_id = :k_id);
    res_txt = null;

    for select
      kob_from_ol_id,   -- ID операции погашения (переплаты)
      kob_acc_dt,   -- Дт счет (Счет учета денег/ Дебет)
      kob_acc_kt,   -- КТ счет (Счет учета переплаты / Кредит)
      kob_summa,    -- Сумма
      obd_detail,   -- Детали проводки
      kob_currency
    from kn_operbook, ref_account kt, kn_obdetail
    where kob_acc_kt = kt.acc_id
      and obd_id = kob_detail_id
      and kt.acc_from_sab_id = 53--:sab_id
      and kob_from_kc_id = :kc_id
      and kob_ol_is_active = 1
      and kob_ol_date = :opl_date
    into repay_ol_id_main, bank_acc_id, acc_id, kob_summa, obd_detail, kob_currency
    do
    begin
      res_txt = null; ol_memo = null; new_kob_id = null;
      reg_id_pay = null; reg_id_repay = null;
      new_ol_id = null; new_ol_id_main = null;

      select rl_new_ol_id
      from reversallist, operlist
      where ol_id = rl_old_ol_id
        and ol_parent_id = :repay_ol_id_main
      into new_ol_id;

      if (new_ol_id is null) then
        res_txt = 'Not found new_ol_id';
      else
      begin
        select ol_parent_id, ol_memo
        from operlist
        where ol_id = :new_ol_id
        into new_ol_id_main, ol_memo;
  
        -- основное погашение
        select ol_from_reg_id
        from operbook, operlist
        where ol_id = ob_from_ol_id
          and ol_id = :repay_ol_id_main    --1010100024528074
          and ob_ol_is_active = 1
        into reg_id_pay;
  
        -- переплата
        select ol_from_reg_id, ol_id
        from operbook, operlist
        where ol_id = ob_from_ol_id
          and ol_parent_id = :repay_ol_id_main    --1010100024528074
          and ob_ol_is_active = 1
          and ob_detail = 'Переплата займа'
        into reg_id_repay, repay_ol_id;

        -- operbook не создался
        if (not exists(select 1
                      from operbook
                      where ob_from_ol_id = :new_ol_id
                        and ob_ol_is_active = 1))
        then
        begin
          insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val, ob_summa, ob_detail)
          select :new_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, -:kob_summa, -:kob_summa, :ol_memo
          from operbook
          where ob_from_ol_id = :repay_ol_id
            and ob_detail = 'Переплата займа'
          rows 1;
        end

        -- если разные офисы на основном погашении и на переплате, то надо строить две проводки
        if (reg_id_pay <> reg_id_repay) then
        begin
--  delete from operbook where ob_from_ol_id = :new_ol_id_main;

          if (not exists(select 1
                        from operbook
                        where ob_from_ol_id = :new_ol_id_main
                          and ob_ol_is_active = 1))
          then
          begin
            insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val, ob_summa, ob_detail)
            select :new_ol_id_main, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, -:kob_summa, -:kob_summa, :ol_memo
            from operbook
            where ob_from_ol_id = :repay_ol_id_main
            rows 1;
          end
        end
      end

      -- ЕПС  Пересчет ЕПС сносит сторно проводки
      if (not exists(select 1 from kn_operbook where kob_from_ol_id = :new_ol_id_main)) then  -- у родителя !!! нет операций
      begin
        select kob_id from hp_do_kn_operbook_memorder(:new_ol_id_main, :kc_id, null, null, -:kob_summa, 0,
              -:kob_summa, -:kob_summa, :obd_detail, :bank_acc_id, :acc_id, :rep_date, :kob_currency)
        into new_kob_id;

        if (new_kob_id is null) then
          res_txt = 'new_kob_id is null';
      end

      suspend;
    end

    if (kc_id is null) then
    begin
      repay_ol_id_main = null; kob_summa = null;
      res_txt = 'Not found kn_operbook';
      suspend;
    end
  end
end
