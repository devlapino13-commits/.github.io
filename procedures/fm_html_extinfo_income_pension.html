CREATE PROCEDURE FM_HTML_EXTINFO_INCOME_PENSION(
    EIL_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable LANG D_STR_LANG;
declare variable C_ID D_ID;
declare variable EIL_DATE D_DATE;
declare variable C_NAME_FIO D_STR_255;
declare variable CALC_STATUS D_STR_2048;
declare variable JSON_DOCUMENT D_BLOB_TEXT;
declare variable EIC_CODE D_STR_CODE;
declare variable PAYMENT_AVERAGE D_CURRENCY;
begin
  -- RFCKG-4255 Справки о доходах (ЗП, пенсия)
  -- отображение полученных справок от сервиса 
  lang = f_get_lang_code(un_id, null);

  for select eil_date, res_blob, eic_code, eil_from_c_id
  from extinfolist eil, extinfocfg eic, fm_blob_get(eil_from_bls_id)
  where eil_id = :eil_id and
        eil_from_eic_id = eic_id and
        eil_is_json = 1 and
        eil_status = 2
  rows 1
  into eil_date, :json_document, eic_code, c_id
  do
  begin
    c_name_fio = f_lang(lang, 'Клиент не найден');

    if (coalesce(:json_document, 0) = 0) then
      calc_status = f_lang(lang, 'Информация о пенсии не получена');
  
    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;
    res_html = (select res_style
                from fm_html_styles) || f_format('<body><h2>%1</h2>%2<br>', c_name_fio, calc_status);
  
    res_html = BLOB_APPEND(res_html, f_format('<h1>%1</h1>', 'Информация о пенсии'));

    payment_average = cast(f_json_str_get_value(:json_document, 'data.baseResponse.paymentAverage', 0) as d_currency);

    res_html = BLOB_APPEND(res_html, '<table width="100%"><col width="50%"><col width="50%">');
    res_html = BLOB_APPEND(res_html, f_format('<thead><tr><th scope="col">%1</th><th scope="col">%2</th></tr></thead>', 'Дата получения ответа Компанией', 'Средняя сумма выплат'));

    res_html = blob_append(res_html, f_html_table_row(f_datetostr(:eil_date),
                                                      round(payment_average, 2),
                                                      '',
                                                      -1, -1, -1, -1, -1, 2, ''));
    res_html = BLOB_APPEND(res_html, '</table>');
  end

  res_html = BLOB_APPEND(res_html, '</body>');
  suspend;
end
