CREATE PROCEDURE HP_RF_CREDIT_HISTORY_INFO(
    JSON_STR D_STR_8100)
RETURNS (
    RES_TEXT D_STR_8100)
AS
declare variable json_str_tmp d_str_8100;
declare variable fio d_str_100;
declare variable birth_date d_date;
declare variable inn d_str_20;
declare variable pass_no varchar(30);
declare variable c_id d_id;
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable debt d_currency;
declare variable delinq_days d_integer;
declare variable c_name_fio d_str_100;
declare variable c_birth_date d_date;
declare variable c_pass_no varchar(30);
declare variable c_inn d_str_20;
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable is_oriv d_smallint;
declare variable max_delinq_days d_integer;
declare variable total_debt d_currency;
begin
  res_text = '';
  total_debt = 0;
  json_str_tmp = f_json_str_get_value(:json_str, 'json', '');
  fio = f_json_str_get_value(:json_str_tmp, 'FIO', '');
  birth_date = f_json_str_get_value(:json_str_tmp, 'BD', '');
  inn = f_json_str_get_value(:json_str_tmp, 'INN', '');
  pass_no = f_json_str_get_value(:json_str_tmp, 'PASS_NO', '');
  c_name_fio = '';

  fio = trim(fio);
  if (fio = '') then
    fio = null;
  inn = replace(inn, ' ', '');
  if (inn = '') then
    inn = null;
  pass_no = replace(pass_no, ' ', '');
  if (pass_no = '') then
    pass_no = null;

  select c_id, c_name_fio, c_birth_date, c_pass_no, c_inn
  from clients cl1
  where (cl1.c_name_fio = :fio and
        cl1.c_birth_date = :birth_date) or (cl1.c_inn = :inn) and
        cl1.c_id > 0
  union
  select c_id, c_name_fio, c_birth_date, c_pass_no, c_inn
  from clients cl2
  inner join clientdocs cds on cds.cds_from_c_id = c_id
  where cds.cds_index_doc_no = :pass_no and
        cl2.c_id > 0 and
        cl2.c_birth_date = :birth_date
  into c_id, c_name_fio, c_birth_date, c_pass_no, c_inn;

  if (c_id is not null) then
  begin
    for select kt.k_id, kt.k_kontract_no, kt.k_disb_date
        from knclient kc
        inner join kontract kt on kt.k_id = kc.kc_from_k_id
        where kc.kc_from_c_id = :c_id and
              kc.kc_amount_approved > 0
        into k_id, k_kontract_no, k_disb_date
    do
    begin
      debt = 0;
      zad_os = 0;
      zad_perc = 0;

      select sum(zad_os), sum(zad_perc)
      from aarep_make_zad_list(:k_id, current_date, 1)
      into zad_os, zad_perc;

      debt = coalesce(zad_os, 0) + coalesce(zad_perc, 0);

      if (debt < 0) then
        debt = 0;

      delinq_days = 0;

      select iif(current_date - oks.oks_delinq_date < 0, 0, current_date - oks.oks_delinq_date)
      from oksummary oks
      where current_date between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_from_k_id = :k_id and
            oks.oks_close_state = 0
      into delinq_days;

      select coalesce(max(ok.ok_ol_date - ok_sh_date), 0)
      from operkred ok
      where ok.ok_ol_is_active = 1 and
            ok.ok_ol_date <= current_date and
            ok.ok_ol_date > ok_sh_date and
            ok_type < 7 and
            ok_from_k_id = :k_id and
            ok_os + ok_perc < 0 and
            ok_from_c_id = 0
      into max_delinq_days;

      is_oriv = 0;

      if ((select sum(oko.oko_os) + sum(oko.oko_perc) + sum(oko.oko_sh_os) + sum(oko.oko_sh_perc)
          from operkroff oko
          where oko.oko_from_c_id = :c_id and
                oko.oko_from_k_id = :k_id and
                oko.oko_ol_date <= current_date and
                oko.oko_ol_is_active = 1 and
                oko.oko_type <= 4) > 0) then
        is_oriv = 1;

      total_debt = coalesce(total_debt, 0) + coalesce(:debt, 0);
    end
  end

  if (coalesce(total_debt, 0) > 0) then
    res_text = trim(cast(round(total_debt, 2) as varchar(50))); -- RFCKG-5062

  suspend;
end
