CREATE PROCEDURE CM_KZ_0043_SQL(
    K_ID D_ID,
    CHECK_KIND D_STR_255,
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable ma_id d_id;
declare variable c_id d_id;
declare variable mac_id d_id;
declare variable kdn d_currency;
begin
  -- RFCKZ-871
  -- Утверждение договора должно быть доступно только после успешного ответа от сервиса и сохранения значений КДН и КДД.
  select k.k_from_ma_app_id
  from kontract k
  where k.k_id = :k_id
  into ma_id;

  for select mac.mac_from_c_id,
         mac.mac_id
  from miniappclient mac
  where mac.mac_from_ma_id = :ma_id and
        mac.mac_amount > 0
  into :c_id, :mac_id
  do
  begin
    if (not exists (select *
                    from extinfolist eil
                    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
                    where eic.eic_code = 'kdn_score' and
                          eil.eil_object_kind = f_objkind('miniappclient', 'id') and
                          eil.eil_object_id = :mac_id and
                          eil.eil_status = 2)) then
     begin
       res_error_text = 'КДН не рассчитан';
       res_error_is_critical = 2;
       suspend;
       exit;
     end

    if (not exists(select *
                   from cfp_anketa_kdn cakdn
                   where cakdn.cakdn_from_c_id = :c_id and
                         cakdn.cakdn_from_mac_id = :mac_id)) then
    begin
      res_error_text = 'КДН не рассчитан';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
  end
end
