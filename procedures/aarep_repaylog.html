CREATE PROCEDURE AAREP_REPAYLOG
RETURNS (
    RPL_DATETIME TIMESTAMP,
    OPER_DATE DATE,
    STAFF_FIO VARCHAR(100),
    STAFF_REGION VARCHAR(100),
    STAFF_JOBNAME VARCHAR(100),
    STAFF_IS_WORKING VARCHAR(100),
    STAFF_BRANCH VARCHAR(100),
    LOAN_FOLDER_NAME VARCHAR(100),
    LOAN_REGION VARCHAR(100),
    LOAN_BRANCH VARCHAR(100))
AS
declare variable rpl_from_k_id decimal(18,0);
declare variable rpl_from_un_id decimal(18,0);
declare variable staff_id decimal(18,0);
declare variable loan_reg_id integer;
declare variable branch_id integer;
begin
  for select rpl_from_k_id, rpl_from_un_id, rpl_datetime
      from repaylog
      into :rpl_from_k_id, :rpl_from_un_id, :rpl_datetime
  do
  begin
    oper_date = cast(rpl_datetime as date);
    if (not(exists(select *
                   from operkred
                   where ok_ol_date = :oper_date and
                         ok_ol_is_active = 1 and
                         ok_from_k_id = :rpl_from_k_id))) then
    begin
      staff_id = null;
      select un_from_c_id
      from username
      where un_id = :rpl_from_un_id
      into :staff_id;

      staff_fio = '';
      staff_region = '';
      staff_jobname = '';
      staff_is_working = '';
      branch_id = 0;
      select c_name_fio, reg_code || ' ' || reg_name, jl_name,
             iif(ps_is_working = 1, 'Работает', 'Уволен'), region.parent
      from clients, hr_pstatus, region, joblist
      where ps_from_c_id = :staff_id and
            ps_date_begin <= :oper_date and
            ps_date_end >= :oper_date and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id and
            ps_from_c_id = c_id
      into :staff_fio, staff_region, staff_jobname, staff_is_working, branch_id;

      select reg_name
      from region
      where reg_unicode = :branch_id
      into :staff_branch;

      loan_reg_id = 0;
      loan_folder_name = '';
      loan_region = '';
      loan_branch = '';

      select kal_new_reg_id, kal_new_kf_name
      from knaktlist, kontract
      where k_id = :rpl_from_k_id and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= :oper_date and
            kal_date_end >= :oper_date and
            knaktlist.is_delete = 0
      into loan_reg_id, loan_folder_name;

      select r1.reg_code || ' ' || r1.reg_name, r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :loan_reg_id and
            r1.parent = r2.reg_unicode
      into loan_region, loan_branch;

      if (loan_region <> staff_region) then
        suspend;
    end
  end
end
