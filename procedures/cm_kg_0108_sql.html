CREATE PROCEDURE CM_KG_0108_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KF_DISB_DATE D_DATE;
declare variable KH_UNPAIDAMOUNT_ALL D_CURRENCY; /* Неоплаченная сумма. */
declare variable K_FROM_MA_APP_ID D_ID;
declare variable KC_FROM_C_ID D_ID;
declare variable MBF_DEPT D_CURRENCY;
declare variable K_ID_ACTIVE D_ID;
declare variable MBF_DEPT_TMP D_CURRENCY;
declare variable OC_RATE D_CURRENCY;
declare variable CUR varchar(3);
declare variable KH_UNPAIDAMOUNT_ALL_TMP D_CURRENCY;
declare variable KH_CURRENCY varchar(50);
declare variable IS_DR D_SMALLINT;
declare variable C_ID D_ID;
declare variable KH_UNPAIDAMOUNT_SUPRUG_TMP D_CURRENCY;
declare variable KH_UNPAIDAMOUNT_SUPRUG D_CURRENCY;
declare variable CREL_OTHER_C_ID D_ID;
declare variable MBF_DEPT_SUPRUG_TMP D_CURRENCY;
declare variable MBF_DEPT_SUPRUG D_CURRENCY;
declare variable PR_CODE D_STR_100;
declare variable SCORE_RATE D_INTEGER;
begin
  if (k_id = 109010278901200101) then  exit;

  is_dr = 0;
  select k_from_ma_app_id, k.k_disb_date, pr.pr_code
  from kontract k
  inner join product pr on pr.pr_id = k.k_from_pr_id
  where k_id = :k_id
  into :k_from_ma_app_id, kf_disb_date, pr_code; --Ссылка на кредитную заявку miniapp

  if (pr_code containing '.express_similar.') then
    exit; -- Для акции "Жаркое лето" JIRA BOARD-5648

  if (pr_code containing '.sugarmill.') then
    exit;

  res_error_is_critical = 1; -- уровень критичности 1
  for select kc_from_c_id
      from knclient kc
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into :kc_from_c_id
  do
  begin
    res_info_text = null;
    select c_name_fio
    from clients c
    where c_id = :kc_from_c_id
    into res_info_text;

    --Неоплаченные остатки
    kh_unpaidamount_all = 0;
    for select sum(coalesce(kh_unpaidamount, 0) + coalesce(kh_unpaidpercenamount, 0)) kh_unpaidamount_all, kh_currency
        from hp_kibminiapp_last(:kc_from_c_id, :k_from_ma_app_id)
        inner join kibhistory on kh_from_kbm_id = kbm_id
        where kh_creditstatus not like '%Отказано%' and
              kh_creditstatus not like '%Рассмотрение заявки%' and
              kh_creditstatus not like 'Погашен%' and
              kh_creditstatus not like 'FULLY_PAID%' and
              :kf_disb_date between kh_dateaward and kh_datestop and -- Нельзя учитывать сумму кредита, отраженную в отчете КИБ, но закрытого у нас на момент утверждения кредита.
              kh_is_debtor = 1 and -- не поручитель
              kh_is_our = 0 -- не МБФ
        group by kh_currency
        into kh_unpaidamount_all_tmp, -- сумма долга в других ФКУ
             kh_currency
    do
    begin
      oc_rate = 1;
      cur = '417';
      if (kh_currency in ('EUR', 'Евро')) then
        cur = '978';
      else
      if (kh_currency in ('RUB', 'Российский рубл')) then
        cur = '810';
      else
      if (kh_currency in ('USD', 'Доллар США')) then
        cur = '840';
      if (cur <> 417) then -- если не  Сомы, тогда получаем курс и умножаем на него
      begin
        select oc.cr_rate
        from opercurr oc
        where current_date between oc.cr_date and oc.cr_date_end and
              oc.cr_from_cu_id = :cur
        into :oc_rate;
        kh_unpaidamount_all_tmp = kh_unpaidamount_all_tmp * oc_rate;
      end
      kh_unpaidamount_all = kh_unpaidamount_all + kh_unpaidamount_all_tmp;
    end
    -- получаю активный долг в МБФ -- >
    mbf_dept = 0;
    k_id_active = null;

    for select ks.ks_from_k_id
        from kontstate ks
        inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
        where ks.ks_state = 1 and
              :kf_disb_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              kncl.kc_from_c_id = :kc_from_c_id and
              kncl.kc_amount_approved > 0 and -- Нельзя учитывать активные кредиты по которому клиент выступает поручителем
              ks.ks_from_k_id <> :k_id
        into :k_id_active
    do
    begin
      mbf_dept_tmp = 0;
      select sum(zad_os + sh_calc_perc)
      from aarep_make_zad_list(:k_id_active, :kf_disb_date, 1)
      into :mbf_dept_tmp;
      mbf_dept_tmp = coalesce(mbf_dept_tmp, 0);
      mbf_dept = mbf_dept + mbf_dept_tmp;
    end
    --- <
    -- Вычисляю сумму долга по кибу + в мбф по супругу клиента
    crel_other_c_id = null;

    select rel_other_c_id
    from hp_client_relation_tmp(:kc_from_c_id, 1)
    where rel_code in (trim('1'), trim('2')) and rel_is_active = 1
    into crel_other_c_id;

    if (crel_other_c_id is not null) then
    begin
      kh_unpaidamount_suprug = 0;
      for select sum(coalesce(kh_unpaidamount, 0) + coalesce(kh_unpaidpercenamount, 0)) kh_unpaidamount_all, kh_currency
          from hp_kibminiapp_last(:crel_other_c_id, :k_from_ma_app_id)
          inner join kibhistory on kh_from_kbm_id = kbm_id
          where kh_creditstatus not like '%Отказано%' and
                kh_creditstatus not like '%Рассмотрение заявки%' and
                kh_creditstatus not like 'Погашен%' and
                kh_creditstatus not like 'FULLY_PAID%' and
                :kf_disb_date between kh_dateaward and kh_datestop and -- Нельзя учитывать сумму кредита, отраженную в отчете КИБ, но закрытого у нас на момент утверждения кредита.
                kh_is_debtor = 1 and  --заёмщик (не поручитель)
                kh_is_our = 0 -- не МБФ
          group by kh_currency
          into kh_unpaidamount_suprug_tmp, -- сумма долга в других ФКУ
               kh_currency
      do
      begin
        kh_unpaidamount_suprug_tmp = coalesce(kh_unpaidamount_suprug_tmp, 0);
        oc_rate = 1;
        cur = '417';
        if (kh_currency in ('EUR', 'Евро')) then
          cur = '978';
        else
        if (kh_currency in ('RUB', 'Российский рубл')) then
          cur = '810';
        else
        if (kh_currency in ('USD', 'Доллар США')) then
          cur = '840';
        if (cur <> 417) then -- если не  Сомы, тогда получаем курс и умножаем на него
        begin
          select oc.cr_rate
          from opercurr oc
          where current_date between oc.cr_date and oc.cr_date_end and
                oc.cr_from_cu_id = :cur
          into :oc_rate;
          kh_unpaidamount_suprug_tmp = kh_unpaidamount_suprug_tmp * oc_rate;
        end
        kh_unpaidamount_suprug = kh_unpaidamount_suprug + kh_unpaidamount_suprug_tmp;
      end
      -- получаю активный долг в МБФ -- >
      mbf_dept_suprug = 0;
      k_id_active = null;
      for select ks.ks_from_k_id
          from kontstate ks
          inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
          where ks.ks_state = 1 and
                :kf_disb_date between ks.ks_date_begin and ks.ks_date_end and
                ks.ks_ol_is_delete = 0 and
                kncl.kc_from_c_id = :crel_other_c_id and
                ks.ks_from_k_id <> :k_id
          into :k_id_active
      do
      begin
        mbf_dept_suprug_tmp = 0;
        select sum(zad_os + sh_calc_perc)
        from aarep_make_zad_list(:k_id_active, :kf_disb_date, 1)
        into :mbf_dept_suprug_tmp;
        mbf_dept_suprug_tmp = coalesce(mbf_dept_suprug_tmp, 0);
        mbf_dept_suprug = mbf_dept_suprug + mbf_dept_suprug_tmp;
      end
    end
    kh_unpaidamount_suprug = coalesce(kh_unpaidamount_suprug, 0);
    mbf_dept_suprug = coalesce(mbf_dept_suprug, 0);
    --<
    -- RFCKG-3788
    if (kh_unpaidamount_all + kh_unpaidamount_suprug + mbf_dept + mbf_dept_suprug > 700000 and
        k_id <> 109010102314600101) then --сумма с др.ФКУ + сумма МБФ
    begin
      res_error_text = 'Совокупный баланс задолженности в других ФКУ (без учета нового займа в МБФ) 700 000 и более';
      -- по транзакции определяю пользователя

      select un.un_from_c_id
      from username un
      where un.un_id = :un_id
      into c_id;

      if (un_id = 1090100001337) then -- если Шавкат
        is_dr = 1;
      else
      if ((is_dr = 0) and
          (exists(select *
                  from hr_pstatus hps
                  inner join joblist jl on jl.jl_id = ps_from_jl_id
                  inner join jobkind jk on jk.jk_id = jl.jl_kind
                  where current_date between ps_date_begin and ps_date_end and
                        ps_is_working = 1 and
                        jk.jk_system_code in ('JobSystem.Region_Director', 'JobSystem.Director_Executive') and
                        hps.ps_from_c_id = :c_id and
                        hps.ps_status = 1))) then
        is_dr = 1;

      if (is_dr = 0) then
      begin
          --> находим расчеты риска из КИБа (JIRA RFCKG-636)
          select kbm_score_rate
          from hp_kibminiapp_last(:kc_from_c_id, :k_from_ma_app_id)
          into score_rate;
          --<--                                               score_rate = 0 - RFCKG-1009
          if (:score_rate >= 599 or :score_rate = 0) then--если расчет риска между средним риском и очень низким и если статус риска раввен 999 то МО может утвердить договор и заменяющий МО
          begin
            if (exists(select *
                       from hp_permission_access_check(:un_id, 'Rolename', 24, 'kred_manager')
                       where has_access = 1)) then
              is_dr = 1;
          end
      end

      if (is_dr = 1) then
        res_error_is_critical = 0;
      suspend;
    end
  end
end
