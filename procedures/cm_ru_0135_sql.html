CREATE PROCEDURE CM_RU_0135_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable C_ID ID_TYPE;
declare variable MA_FROM_PR_ID integer;
declare variable PRL_AMOUNT_GR_MIN CURR_TYPE;
declare variable PRL_AMOUNT_GR_MAX CURR_TYPE;
declare variable PRL_AMOUNT_CL_MIN CURR_TYPE;
declare variable PRL_AMOUNT_CL_MAX CURR_TYPE;
declare variable GR_AMOUNT CURR_TYPE;
declare variable MA_PERCENT PERC_TYPE;
declare variable MA_KOLICH_REPAY integer;
declare variable CL_QTY integer;
declare variable PRL_KOLICH_MIN integer;
declare variable PRL_KOLICH_MAX integer;
begin
  select MAC.MA_FROM_PR_ID, MAC.MA_DATE_OPER, MAC.MA_PERCENT, MAC.MA_KOLICH_REPAY
  from MINIAPP MAC
  where MAC.MA_ID = :MA_ID and
        current_date between MAC.MA_DATE_APP and MAC.MA_DATE_END and
        MAC.MA_IS_ACTIVE = 1
  into MA_FROM_PR_ID, :KF_DISB_DATE, MA_PERCENT, MA_KOLICH_REPAY;
  select min(PRL_KOLICH_MIN) PRL_KOLICH_MIN, max(PRL_KOLICH_MAX) PRL_KOLICH_MAX, --cl_qty
         min(PRL_AMOUNT_GR_MIN) PRL_AMOUNT_GR_MIN, max(PRL_AMOUNT_GR_MAX) PRL_AMOUNT_GR_MAX, --gr
         min(PRL_AMOUNT_CL_MIN) PRL_AMOUNT_CL_MIN, max(PRL_AMOUNT_CL_MAX) PRL_AMOUNT_CL_MAX --cl
  from PRODUCT PR, PRODLOGIC PL
  where PR.PR_ID = :MA_FROM_PR_ID and
        PR.PR_ID = PL.PRL_FROM_PR_ID and
        :KF_DISB_DATE between PL.PRL_DATE_BEGIN and PL.PRL_DATE_END /*and
        :MA_PERCENT between PL.PRL_PERCENT_MIN and PL.PRL_PERCENT_MAX and
        :MA_KOLICH_REPAY between PL.PRL_REPAY_MIN and PL.PRL_REPAY_MAX */
  into PRL_KOLICH_MIN, PRL_KOLICH_MAX, --cl_qty
       PRL_AMOUNT_GR_MIN, PRL_AMOUNT_GR_MAX, --gr
       PRL_AMOUNT_CL_MIN, PRL_AMOUNT_CL_MAX; --cl;
  for select MACL.MAC_FROM_C_ID, MAC_AMOUNT
      from MINIAPPCLIENT MACL
      where MACL.MAC_FROM_MA_ID = :MA_ID and
            MACL.MAC_AMOUNT > 0
      into :C_ID, KC_AMOUNT
  do
  begin
    GR_AMOUNT = coalesce(GR_AMOUNT, 0) + KC_AMOUNT;
    CL_QTY = coalesce(CL_QTY, 0) + 1;
    if (KC_AMOUNT not between PRL_AMOUNT_CL_MIN and PRL_AMOUNT_CL_MAX) then
    begin
      select C_NAME_FIO
      from CLIENTS
      where C_ID = :C_ID
      into res_info_text;
      if (KC_AMOUNT < PRL_AMOUNT_CL_MIN) then
        res_error_text = 'Сумма займа не может быть менее:  ' || PRL_AMOUNT_CL_MIN;
      else
        res_error_text = 'Сумма займа не может быть более: ' || PRL_AMOUNT_CL_MAX;
      res_error_is_critical = 1;
      suspend;
    end
  end
  KC_AMOUNT = 0;
  res_info_text = '';
  if (CL_QTY not between PRL_KOLICH_MIN and PRL_KOLICH_MAX) then
  begin
    if (CL_QTY < PRL_KOLICH_MIN) then
      res_error_text = 'Количество клиентов должен быть от : ' || PRL_KOLICH_MIN;
    else
      res_error_text = 'Количество клиентов должен быть до : ' || PRL_KOLICH_MAX;
    res_error_is_critical = 1;
    suspend;
  end
  if (CL_QTY > 1 and
      GR_AMOUNT not between PRL_AMOUNT_GR_MIN and PRL_AMOUNT_GR_MAX) then
  begin
    if (GR_AMOUNT < PRL_AMOUNT_GR_MIN) then
      res_error_text = 'Сумма займа не может быть менее:  ' || PRL_AMOUNT_GR_MIN;
    else
      res_error_text = 'Сумма займа не может быть более:  ' || PRL_AMOUNT_GR_MAX;
    res_error_is_critical = 1;
    suspend;
  end
end
