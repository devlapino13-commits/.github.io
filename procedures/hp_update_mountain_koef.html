CREATE PROCEDURE HP_UPDATE_MOUNTAIN_KOEF(
    REP_DATE D_DATE,
    REP_REG D_INTEGER,
    UN_ID D_ID,
    MOUNTAIN_KOEF D_CURRENCY)
RETURNS (
    RES_TEXT D_STR_255,
    IS_ERROR D_SMALLINT)
AS
declare variable rmk_date_begin d_date;
begin
  -- RFCKG-5508 Изменения коэффициента высокогорья И Дата действия коэффициента высокогорья

  res_text = '';
  is_error = 0;

  if (res_text = '') then
  begin
    select rmk.rmk_date_begin
    from regmountkoefs rmk
    where rmk_from_reg_id = :rep_reg
      and rmk_date_end = '31.12.9999'
    into rmk_date_begin;

    if (rmk_date_begin < :rep_date) then
    begin
      update regmountkoefs
         set rmk_date_end = :rep_date
      where  rmk_from_reg_id = :rep_reg
        and  rmk_date_end = '31.12.9999';

       insert into regmountkoefs (rmk_from_reg_id, rmk_date_begin, rmk_date_end, rmk_mount_koef, rmk_from_un_id)
       values (:rep_reg, :rep_date, '31.12.9999', coalesce(:mountain_koef, 1), :un_id);
     end
     else
     begin
       update regmountkoefs
          set rmk_date_begin = :rep_date,
              rmk_from_un_id = :un_id,
              rmk_mount_koef = :mountain_koef
       where  rmk_from_reg_id = :rep_reg
         and  rmk_date_end = '31.12.9999';
     end

     update region reg
        set reg.reg_mountain_koef = :mountain_koef
     where  reg.reg_unicode = :rep_reg;
    res_text = 'Коэффицент успешно изменен';
  end

  suspend;
end
