CREATE PROCEDURE CB_KG_TEST_2012_12(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    CB_PERSONAL_ID DECIMAL(18,0),
    CB_REGION_ID INTEGER,
    CB_PERSONAL_FIO VARCHAR(100),
    CB_POSITION_NAME VARCHAR(100),
    CB_JOBKIND_NAME VARCHAR(100),
    CB_IS_WORKING_STR VARCHAR(20),
    CB_BALANCE_END CURR_TYPE,
    CB_KOLICH_CL_END INTEGER,
    CB_TOTAL_BONUS CURR_TYPE,
    CB_REGION_NAME VARCHAR(100),
    CB_CORRECT_CALC DECIMAL(18,4),
    CB_NEW_FIELD DECIMAL(18,4))
AS
declare variable ps_is_working integer;
declare variable rep_date_begin date;
begin
  cb_balance_end = 0;
  cb_kolich_cl_end = 0;
  cb_total_bonus = 0;
  for select distinct ps_from_c_id, ps_from_reg_id, c_name_fio, jl_name, jk_name, ps_is_working,
                      reg_code || ' ' || reg_name
      from hr_pstatus, clients, joblist, jobkind, region_list(:rep_region_id, 1, -1, 1), region
      where ps_from_c_id = c_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            ps_from_reg_id = id_list and
            ps_from_reg_id = reg_unicode and
            ps_is_working = 1
      into :cb_personal_id, :cb_region_id, :cb_personal_fio, :cb_position_name, :cb_jobkind_name, :ps_is_working,
           :cb_region_name
  do
  begin
    if (ps_is_working = 1) then
      cb_is_working_str = 'Работает';
    else
      cb_is_working_str = 'Уволен';

    cb_total_bonus = mod(cb_personal_id, 1000) * 10;
    cb_correct_calc = cb_total_bonus;
    cb_new_field = cb_total_bonus / 2;
    suspend;
  end
end
