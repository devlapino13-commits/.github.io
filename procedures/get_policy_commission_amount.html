CREATE PROCEDURE GET_POLICY_COMMISSION_AMOUNT(
    K_ID D_ID)
RETURNS (
    COMMISSION_AMOUNT D_CURRENCY,
    MBULAK_SHARE D_CURRENCY,
    POLICY_COMPANY_SHARE D_CURRENCY,
    TAX D_CURRENCY)
AS
declare variable N_VALUE D_INTEGER;
declare variable K_AMOUNT_APPROVED D_CURRENCY;
declare variable PR_CODE D_STR_255;
declare variable K_DISB_DATE D_DATE;
begin
    -- RFCKG-3652
    select
        k.k_amount_approved, -- сумма страхования (в контексте страхования конечно же)
        k.k_kredline_end - k.k_disb_date,
        pr.pr_code,
        k.k_disb_date
    from kontract k
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where k.k_id = :k_id
    into
        :k_amount_approved,
        :n_value,
        :pr_code,
        :k_disb_date;

    select commission_amount,
           mbulak_share,
           policy_company_share,
           tax
    from calc_insurance_amount(:k_amount_approved, :n_value, :pr_code, :k_disb_date, :k_id)   -- RFCKG-4803
    into :commission_amount, mbulak_share, policy_company_share, tax;

    suspend;
end
