CREATE PROCEDURE UXP$DEMO_PLACE_LIST(
    REP_REGION_ID D_INTEGER)
RETURNS (
    PLC_ID D_INTEGER,
    PLC_PARENT_ID D_INTEGER,
    PLC_FROM_REG_ID D_INTEGER,
    PLC_TYPE D_SMALLINT,
    PLC_NAME D_STR_100,
    PLC_MEMO D_STR_255)
AS
begin
  for select plc_id,
             plc_parent_id,
             plc_from_reg_id,
             plc_type,
             plc_name,
             plc_memo
      from ux$demo_place
      where plc_from_reg_id = :rep_region_id
      into :plc_id,
           :plc_parent_id,
           :plc_from_reg_id,
           :plc_type,
           :plc_name,
           :plc_memo
  do
  begin
    suspend;
  end
end
