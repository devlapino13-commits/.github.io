CREATE PROCEDURE FM_SEND_SAL_CODE(
    C_ID D_ID,
    TEXT_CODE D_STR_10,
    UN_ID D_ID)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable full_source d_blob_text;
begin
  -- Процедура создания записи в таблице extsysmain для отправки код подтверждения. (RFCKG-3224)

    if (c_id is null or not exists (select * from clients c where c.c_id = :c_id))
    then
    begin
        err_text = 'Клиент не найден. Неверный c_id: ' || :c_id;
        suspend;
        exit;
    end

    if (text_code = ''
        or text_code is null
        or char_length(text_code) <> 6
        or not (text_code similar to '[0-9]{6}')) -- соответствует ли text_code числовому формату
    then
    begin
        err_text = 'Неверный Код';
        suspend;
        exit;
    end

   select ea.esa_message
   from extsysmain esm
   inner join extsysanswer ea on ea.esa_from_esm_id = esm.esm_id
   where esm.esm_project = 'salary_info'
    and esm.esm_action ='req_permission'
    and esm.esm_object_kind = f_objkind('clients', 'id')
    and esm.esm_status = 1
    and esm.esm_object_id = :c_id
   order by esm.esm_datetime desc
   rows 1
   into :full_source;

   if (full_source is null)
   then
   begin
      err_text = 'Запрос не найден';
      suspend;
      exit;
   end

   full_source = f_json_str_set_value(full_source, 'code', text_code);

   insert into extsysmain (esm_project, esm_action, esm_date,  esm_object_kind,
                           esm_object_id, esm_status, esm_full_source)
   values ('salary_info', 'send_code', current_date, f_objkind('clients', 'id'), :c_id,
           0, :full_source);
end
