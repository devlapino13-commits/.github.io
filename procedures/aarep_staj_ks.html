CREATE PROCEDURE AAREP_STAJ_KS(
    C_ID DECIMAL(18,0),
    REPORT_DATE_END DATE)
RETURNS (
    PS_DATE_BEGIN DATE,
    C_STAGE INTEGER,
    C_STAGE_YEAR INTEGER)
AS
begin
  for select min(ps_date_begin)
      from hr_pstatus, joblist, jobkind
      where ps_from_c_id = :c_id and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')
      into :ps_date_begin

  do
  begin
    if (ps_date_begin is not null) then
      c_stage = datediff(month, ps_date_begin, report_date_end);
    else
      c_stage = 0;

    c_stage_year = cast((c_stage / 12) as integer);
    suspend;
  end
end
