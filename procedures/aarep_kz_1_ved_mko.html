CREATE PROCEDURE AAREP_KZ_1_VED_MKO(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    DISB_SUM D_CURRENCY,
    DISB_KOL_KRED D_INTEGER,
    DISB_KOL_CLI D_INTEGER,
    AVG_PERCENT D_CURRENCY,
    KC_BALANCE D_CURRENCY,
    KOL_KRED D_INTEGER,
    KOL_CLI D_INTEGER,
    PERIOD_STR D_STR_255)
AS
declare variable RD_ID D_INTEGER;
begin
  select rd_id from refdata
  where rd_code = 'Loan_Target.Consumer'
  into rd_id;

  -- за период
  disb_sum = null;
  disb_kol_kred = null;
  disb_kol_cli = null;
  select sum(kc_amount_approved),
         count(*),
         sum((select res_just_client_count
           from hp_get_just_client_in_loan(ks_date_begin, kc_from_k_id, k_disb_date))),
         sum(cast(k_percent as double precision) * kc_amount_approved) / sum(kc_amount_approved)
  from knclient, kontstate, clients, region_list(:rep_region_id,1,3,1), kontract
  where kc_loan_target = :rd_id
  and kc_amount_approved > 0
  and kc_from_k_id = ks_from_k_id
  and ks_region_id = id_list
  and ks_ol_is_delete = 0
  and ks_state = 1
  and ks_date_begin between :rep_date_begin and :rep_date_end
  and kc_from_c_id = c_id
  and c_type = 0 -- только физические лица
  and ks_from_k_id = k_id
  into disb_sum, disb_kol_kred, disb_kol_cli, avg_percent;
  disb_sum = disb_sum / 1000; -- в тыс. тенге
  -- на дату
  kc_balance = null;
  kol_kred = null;
  kol_cli = null;
  select  sum(cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance),
          count(*),
          sum((select res_just_client_count
           from hp_get_just_client_in_loan(:rep_date_end, kc_from_k_id, k_disb_date)))
  from oksummary, knclient, kontract
  where :rep_date_end between oks_date_begin and oks_date_end
  and oks_close_state = 0
  and oks_balance > 0
  and oks_from_k_id = kc_from_k_id
  and kc_amount_approved > 0
  and kc_loan_target = :rd_id
  and kc_from_k_id = k_id
  into kc_balance, kol_kred, kol_cli;
  kc_balance = kc_balance / 1000;  -- в тыс. тенге

  period_str = 'Отчетный период: по состоянию c '||f_datetostr(:rep_date_begin)||' по '||f_datetostr(:rep_date_end);

  suspend;
end
