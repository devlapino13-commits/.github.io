CREATE PROCEDURE HP_PDN_GET_LIMIT(
    KC_ID D_ID)
RETURNS (
    PDN_LIMIT D_CURRENCY)
AS
declare variable CLIENT_POINT D_INTEGER;
declare variable C_ID D_ID;
declare variable K_ID D_ID;
declare variable PR_CODE STR_100;
declare variable K_DISB_DATE D_DATE;
begin
  -- процедура возвращает лимит ПДН для клиента по контракту
  select kc_from_c_id, kc_from_k_id
  from knclient
  where kc_id = :kc_id and
        kc_amount_approved > 0
  into c_id, k_id;

  select pr_code, k_disb_date
  from kontract
  inner join product on pr_id = k_from_pr_id
  where k_id = :k_id
  into pr_code, k_disb_date;

  client_point = f_point_get(9, k_id, c_id, k_disb_date);
  pdn_limit = 50;

  if (client_point >= 7) then
    pdn_limit = 80;

  if (pr_code containing '.oivo_pack.') then
  begin
    pdn_limit = 80;

    if (client_point >= 61) then
      pdn_limit = 0;
  end

  if (pr_code containing '.plus_100.' or pr_code containing '.plus_100_b.' )
    then pdn_limit = 80;


--  if (client_point >= 61) then
--    pdn_limit = 0;

  if (pr_code containing '.zalog.' or pr_code containing '.ZALOG_CAR.') then
    pdn_limit = 50;

  suspend;
end
