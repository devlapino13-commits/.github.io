      pre_list = :pre_list || '20.';

        pre_list not containing '.21.') then
      pre_list = :pre_list || '21.';
    if (:sat_day_22 = 0 and
        pre_list not containing '.22.') then
      pre_list = :pre_list || '22.';
    if (:sat_day_23 = 0 and
        pre_list not containing '.23.') then
      pre_list = :pre_list || '23.';

        pre_list not containing '.24.') then
      pre_list = :pre_list || '24.';
