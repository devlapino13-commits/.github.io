CREATE PROCEDURE HP_PRODUCTBLANK_BLANK(
    REP_DATE D_DATE,
    SHOW_ONLY_ACTIVE D_SMALLINT,
    LINK_KIND D_SMALLINT,
    IN_PR_ID D_INTEGER)
RETURNS (
    BLN_ID D_INTEGER,
    BLN_NAME D_STR_255,
    BLN_TYPE D_STR_255,
    BLN_LANG D_STR_LANG,
    BLN_DATE_END D_DATE,
    RES_CHECKED D_SMALLINT,
    PR_ID D_INTEGER)
AS
declare variable bln_memo d_str_255;
declare variable bln_is_active d_smallint;
begin
  pr_id = in_pr_id;

  for select bln_id, f_str_join(bln_name, bln_memo, '/'), bln_memo, bln_lang, blt_name
  from blankname, blanktype
  where bln_kind in (1, 11) and
        bln_from_blt_id = blt_id
  into bln_id, bln_name, bln_memo, bln_lang, bln_type
  do
  begin
    bln_memo = trim(coalesce(bln_memo, ''));

    bln_is_active = 0;
    if (exists(select *
               from blankblob
               where blb_from_bln_id = :bln_id and
                     :rep_date between coalesce(blb_date_begin, '01.01.1990') and coalesce(blb_date_end, '31.12.9999') and
                     (blb_blob is not null or
                     blb_frx_blob is not null))) then
      bln_is_active = 1;

    bln_date_end = null;

    if (show_only_active = 0 and bln_is_active = 0) then
    begin
      select max(blb_date_end)
      from blankblob
      where blb_from_bln_id = :bln_id and
            (blb_blob is not null or blb_frx_blob is not null)
      into bln_date_end;
    end

    if (show_only_active = 0 or bln_is_active = 1) then
    begin
      res_checked = 1;

      if (link_kind >= 0) then
      begin
        res_checked = 0;

        if (exists(select *
        from productblank
        where prb_from_pr_id = :pr_id and
              prb_from_bln_id = :bln_id and
              :rep_date between prb_date_begin and coalesce(prb_date_end, :rep_date)))
        then res_checked = 1;
      end

      if (link_kind = 0 or res_checked = 1) then
        suspend;
    end
  end
end
