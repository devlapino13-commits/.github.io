CREATE PROCEDURE MC_LOANS_FOR_REF(
    MCL_ID D_ID)
RETURNS (
    K_ID D_ID,
    K_KONTRACT_NO STR_20,
    K_DISB_DATE D_DATE,
    IS_ACTIVE D_BOOL)
AS
declare variable DATE_END D_DATE;
declare variable C_ID D_ID;
declare variable SUM_ZADOL D_CURRENCY;
declare variable KS_STATE D_SMALLINT;
declare variable SUM_ZAD D_CURRENCY;
declare variable MCL_PROJECT D_STR_20;
begin
    date_end = current_date;
    if (coalesce(mcl_id, 0) <= 0) then
        exception e_default 'parametr "mcl_id" has invalid value';

    select
        mcl.mcl_from_c_id,
        f_enum('mcl_project', mcl.mcl_project, 'code', 0)
    from mc_login mcl
    where mcl.mcl_id = :mcl_id
    into
        :c_id,
        :mcl_project;

    for select
        k.k_id,
        k.k_kontract_no,
        ks.ks_state,
        k.k_disb_date
    from kontract k
    inner join knclient kc on kc.kc_from_k_id = k.k_id
        and kc.kc_from_c_id = :c_id
        and kc.kc_amount_approved > 0
    inner join kontstate ks on ks.ks_from_k_id = k.k_id
        and ks.ks_ol_is_delete = 0
        and ks.ks_state in (1,2)
        and :date_end between ks.ks_date_begin and ks.ks_date_end
    inner join product pr on pr.pr_id = k.k_from_pr_id
        and iif(:mcl_project = 'oivo', pr.pr_code containing '.oivo_pack.', pr.pr_code not containing '.oivo_pack.')
    order by k.k_disb_date desc
    into
        k_id,
        k_kontract_no,
        ks_state,
        k_disb_date
    do
    begin
        if (ks_state = 1) then is_active = 1;
        else is_active = 0;

    -- проверка внесистемного учета, если имеется задолженность на внесистемке то статуст буде активным, так как по нему не должно печататься справка
        if (is_active = 0) then
        begin
            sum_zad = 0;
            select sum(oko_os + oko_sh_os + oko_other + oko_perc + oko_sh_perc + oko_sh_other)
            from operkroff
            where oko_from_k_id = :k_id
            and oko_ol_is_active = 1
            and oko_ol_date <= :date_end
            into sum_zad;
            sum_zad = coalesce(sum_zad, 0);
            if (sum_zad > 0) then is_active = 1;
        end
        suspend;
    end 
end
