CREATE PROCEDURE MC_NEW_PHONE(
    MCL_ID ID_TYPE NOT NULL,
    PHONE D_STR_20 NOT NULL,
    ACTION_CODE D_STR_20 NOT NULL,
    SMS_CODE D_STR_10)
RETURNS (
    RES_CODE D_STR_20,
    ERROR_TXT D_STR_255,
    IS_SYSTEM_ERROR D_BOOL)
AS
declare variable max_check_cnt d_smallint;
declare variable mcl_login d_str_code;
declare variable mcc_login d_str_code;
declare variable mcc_pin_code d_integer;
declare variable check_code_cnt d_integer;
declare variable db_id d_integer;
declare variable mcc_id d_id;
declare variable c_id d_id;
declare variable clt_id d_integer;
declare variable old_phone d_str_20;
declare variable mcl_project d_smallint;
declare variable mcl_project_txt d_str_code;
declare variable msg_object_kind d_integer;
declare variable msg_object_id d_id;
declare variable msg_sender d_str_code;
declare variable mcc_check_kind d_integer;
declare variable test_sms_code d_integer;
declare variable phone_verify_lookback_days d_integer;
declare variable check_quota_cfg_code d_str_code;
declare variable send_quota_cfg_code d_str_code;
declare variable phone_change_attempts_cfg_code d_str_code;
declare variable db_is_actual d_integer;
declare variable sms_code_ttl_minutes d_integer;
declare variable sms_code_last_send_time d_datetime;
declare variable quota_code d_str_code;
declare variable quota_exceeded d_bool;
declare variable quota_interval_parsed d_integer;
declare variable time_left_str d_str_100;
begin
    db_id = f_get_db_id();
    db_is_actual = coalesce((select 1 from hp_db_is_actual), 0);

    res_code = 'success';
    error_txt = '';
    is_system_error = 1;

    sms_code = coalesce(sms_code, '');
    msg_sender = null;
    mcc_check_kind = 1; -- Смена номера телефона

    check_code_cnt = 0;
    max_check_cnt = 3;
    test_sms_code = 7777; -- SMS-код для тестового режима
    phone_verify_lookback_days = 90;
    sms_code_ttl_minutes = 15; -- Код живет 15 минут, потом использовать его нельзя

    if (action_code not in ('send_code', 'check_code', 'send_code_no_access', 'check_code_no_access')) then
    begin
        error_txt = 'Тип операции не определен';
        suspend;
        exit;
    end

    select first 1
        mcl_login,
        mhl.mhl_from_mcc_id,
        mcl.mcl_from_c_id,
        mcl.mcl_project,
        f_enum_code('mcl_project', mcl.mcl_project)
    from mc_login mcl
    left join mc_check_link mhl on mhl.mhl_from_mcl_id = mcl.mcl_id
    left join mc_check mc on mc.mcc_id = mhl.mhl_from_mcc_id and
                             mcc_check_kind = :mcc_check_kind
    where mcl_id = :mcl_id and
          mcl_is_delete = 0
    order by mcc_last_send_time desc
    into
        :mcl_login,
        :mcc_id,
        :c_id,
        :mcl_project,
        :mcl_project_txt;

    if (action_code in ('send_code', 'send_code_no_access')) then
    begin
        send_quota_cfg_code =
            case mcl_project_txt
                when 'kmp' then 'mobile.clients.phone_change.send_code'
                when 'oivo' then 'mobile.clients.oivo.phone_change.send_code'
                else ''
            end;

        phone_change_attempts_cfg_code =
            case mcl_project_txt
                when 'kmp' then 'mobile.clients.phone_change'
                when 'oivo' then 'mobile.clients.oivo.phone_change'
                else ''
            end;

        if (coalesce(send_quota_cfg_code, '') = '' or coalesce(phone_change_attempts_cfg_code, '') = '') then
            error_txt = 'Произошла ошибка. Обратитесь к тех. поддержке';

        if (error_txt = '' and coalesce(:c_id, 0) = 0) then
            error_txt = 'Смена номера телефона невозможна';

        if (error_txt = '' and coalesce(phone, '') = '') then
            error_txt = 'Номер телефона не указан';

        -- RFCRU-5785
        -- Проверка номера телефона по маске
        if (error_txt = '' and
            f_phone_validate_mask(:phone, :db_id) = 0
        ) then
            error_txt = 'Неверный формат номера телефона';

        old_phone = coalesce(f_cont_info('clients', :c_id, 'cell', 'ok'), '');

        if (error_txt = '' and old_phone = phone) then
            error_txt = 'Введенный номер совпадает со старым. Пожалуйста, введите другой номер телефона';

        -- Специфические проверки, только для РФ
        if (error_txt = '' and db_id = 81001) then
        begin
            -- Запрет действующим сотрудникам в РФ менять номер телефона через КМП
            -- Данная проверка уже есть в mc_ui_visibility, но здесь продублирована для надежности
            if (exists(select *
                       from hr_pstatus
                       where ps_from_c_id = :c_id and
                             current_date between ps_date_begin and ps_date_end and
                             ps_is_working = 1
            
                )) then
            begin
                error_txt = 'Невозможно сменить номер телефона через КМП, т.к. вы являетесь действующим сотрудником компании';
            end

            clt_id = f_cont_id(:phone, 'null');

            if (error_txt = '' and clt_id is not null) then
            begin
                if (exists(select *
                           from contdata cd
                           where cda_from_clt_id = :clt_id and
                                 cda_is_delete = 0 and
                                 cda_from_c_id <> 0 and
                                 cda_from_c_id <> :c_id and
                                 cd.cda_is_main = 1 and
                                 cd.cda_status = 1 and
                                 cd.cda_date >= current_date - :phone_verify_lookback_days)) then
      
                begin
                    error_txt =
                        case action_code
                            when 'send_code_no_access' then 'Данный номер телефона был верифицирован ранее другим клиентом, обратитесь к своему специалисту'
                            else 'К сожалению, данный номер привязан к другому клиенту. Попробуйте ввести другой номер'
                        end;

                end
            end
        end

        if (error_txt = '') then
        begin
            select quota_exceeded, quota_code
            from hp_msgquota_process(:send_quota_cfg_code, :mcl_login)
            into :quota_exceeded, :quota_code;

            if (quota_exceeded = 1) then
            begin
                is_system_error = 0;

                if (quota_code is not null) then
                begin
                    quota_interval_parsed = cast(substring(:quota_code from position('t' in :quota_code) + 1 for position('c' in :quota_code) - position('t' in :quota_code) - 1) as d_integer);

                    select time_left
                    from hp_get_time_left_formatted(:quota_interval_parsed, :mcl_login, :send_quota_cfg_code)
                    into :time_left_str;

                    error_txt = f_format('Превышен лимит отправки СМС. Пожалуйста, повторите попытку через %1', :time_left_str);
                end
                else
                    error_txt = 'Пожалуйста, подождите немного прежде чем запросить повторную отправку кода';
            end

        end

        --  RFCRU-4650
        if (db_id = 81001 and
            error_txt = '' and
            action_code = 'send_code_no_access' and
            exists(select *
                   from hp_msgquota_process(:phone_change_attempts_cfg_code, :mcl_login)
                   where quota_exceeded = 1)) then
        begin
            is_system_error = 0;
            error_txt = 'Превышено количество попыток смены номера телефона. Попробуйте позже';
        end

        -- Если есть ошибка выводим текст и выходим из процедуры
        if (coalesce(error_txt, '') <> '') then
        begin
            res_code = 'error';
            suspend;
            exit;
        end


        if (db_is_actual = 1) then
            sms_code = cast(trunc(rand() * 9000 + 1000) as integer);
        else
            sms_code = test_sms_code;
    
        select res_object_kind, res_object_id
        from hp_contdata_find_object(:phone, 'cell')
        rows 1
        into :msg_object_kind, :msg_object_id;
    
        if (db_id = 81001) then
            msg_sender = trim(iif (:mcl_project = 2, 'oivo', 'mbulak'));
    
        f_send_sms(:phone,
                   iif(db_id <> 94401, 'Ваш код для подтверждения: ', 'M Bulak: Tesdiq uchun kod: ') || :sms_code,
                   coalesce(:msg_object_kind, 0), coalesce(:msg_object_id, 0), 'quick sms', 365, 1000000000000, 1, :msg_sender
                   );
    
        update or insert into mc_check (mcc_login, mcc_pin_code, mcc_check_kind, mcc_project, mcc_last_send_time)
        values (:mcl_login, :sms_code, :mcc_check_kind, :mcl_project, current_timestamp)
        matching (mcc_login, mcc_check_kind, mcc_project)
        returning mcc_id
        into :mcc_id;
    
        if (not exists(select *
                       from mc_check_link mhl
                       where mhl.mhl_from_mcc_id = :mcc_id and
                             mhl.mhl_from_mcl_id = :mcl_id)) then
          insert into mc_check_link (mhl_from_mcc_id, mhl_from_mcl_id)
          values (:mcc_id, :mcl_id);
    end

    ---------------------------------------------------------------------------------------------------------------------

    -- Проверка введенного кода
    if (action_code in ('check_code', 'check_code_no_access')) then
    begin
        check_quota_cfg_code =
            case
                when action_code = 'check_code' and mcl_project_txt = 'kmp' then 'mobile.clients.phone_change.check_code'
                when action_code = 'check_code' and mcl_project_txt = 'oivo' then 'mobile.clients.oivo.phone_change.check_code'
                when action_code = 'check_code_no_access' and mcl_project_txt = 'kmp' then 'mobile.clients.phone_change.check_code_no_acc'
                when action_code = 'check_code_no_access' and mcl_project_txt = 'oivo' then 'mobile.clients.oivo.phone_change.check_code_no_acc'
                else ''
            end;

        if (coalesce(check_quota_cfg_code, '') = '') then
            error_txt = 'Произошла ошибка. Обратитесь к тех. поддержке';

        select first 1
            mcc_pin_code,
            mcc_login,
            mcc_last_send_time
        from mc_check mc
        inner join mc_check_link mh on mh.mhl_from_mcc_id = mc.mcc_id
        where mh.mhl_from_mcl_id = :mcl_id and
              mcc_check_kind = :mcc_check_kind and
              mcc_project = :mcl_project
        order by mcc_last_send_time desc
        into
            :mcc_pin_code,
            :mcc_login,
            :sms_code_last_send_time;
        
        if (error_txt = '' and coalesce(sms_code, '') = '') then
            error_txt = 'Код не заполнен';

        if (error_txt = '' and db_id not in (81001, 39801) and :mcc_login <> :mcl_login) then
            error_txt = 'Номер телефона не совпадает';

        if (error_txt = '' and datediff(minute, :sms_code_last_send_time, current_timestamp) >= sms_code_ttl_minutes) then
            error_txt = 'Отправленный вам код подтверждения уже не актуален. Пожалуйста, запросите новый';

        if (error_txt = '') then
        begin
            if (exists(select *
                       from hp_msgquota_process(:check_quota_cfg_code, :mcl_login)
                       where quota_exceeded = 1)) then
            begin
                error_txt =
                    case action_code
                        when 'check_code_no_access' then 'Превышено количество попыток смены номера телефона. Повторите попытку через 12 часов'
                        else 'Превышен лимит проверок кода. Пожалуйста, повторите попытку позже'
                    end;
            end

            -- Если квота на проверку не превышена, проверяем введеный смс-код
            if (error_txt = '' and mcc_pin_code <> sms_code) then
            begin
                select count(mql.mql_id)
                from msgquotalist mql
                inner join msgquotacfg mqc on mql.mql_from_mqc_id = mqc.mqc_id
                where
                    mqc.mqc_code = :check_quota_cfg_code and
                    mql.mql_cont_code = :mcl_login and
                    (
                        (:action_code = 'check_code' and datediff(minute, mql.mql_datetime, current_timestamp) <= 15)
                        or
                        (:action_code = 'check_code_no_access' and datediff(hour, mql.mql_datetime, current_timestamp) <= 12)
                    )
                into :check_code_cnt;
                
                is_system_error = 0;

                if (check_code_cnt >= max_check_cnt) then
                begin
                    error_txt =
                        case action_code
                            when 'check_code_no_access' then 'Превышено количество попыток смены номера телефона. Повторите попытку через 12 часов'
                            else 'Превышен лимит проверок кода. Пожалуйста, повторите попытку позже'
                        end;
                end
                else
                    error_txt = f_format('Осталось попыток: %1', :max_check_cnt - :check_code_cnt);
            end
        end
    
        -- Если есть ошибка выводим текст и выходим из процедуры
        if (coalesce(error_txt, '') <> '') then
        begin
            res_code = 'error';
            suspend;
            exit;
        end
    
        if (db_id = 81001) then
        begin
            -- Устанавливаем клиенту новый номер телефона (НЕ ПОДТВЕРЖДЕННЫЙ)
            execute procedure hp_continfo_force_update(:c_id, f_get_rd_id('ContactInfo.Cell_Main'), :phone, 0, 1000000000000, current_timestamp);
            
            if (action_code = 'check_code_no_access') then
            begin
                insert into mc_phone_change (mpc_from_mcl_id, mpc_new_phone, mpc_status)
                values (:mcl_id, :phone, f_enum_id('mc_phone_change', 'not_yet_created'));
            end
                
            if (action_code = 'check_code') then
            begin
                select res_text
                from hp_edoc_insert('cell.application.kmp', f_objkind('clients', 'id'), :c_id, 1000000000000)
                into :error_txt;

                -- Если при формировании документа произошла ошибка, выводим клиенту не системное сообщение, а обобщенное
                if (coalesce(error_txt, '') <> '') then
                begin
                    error_txt = 'При формировании заявления на смену номера произошла ошибка. Повторите попытку позже или обратитесь к тех. поддержке';
                    res_code = 'error';
                end
            end

        end
    end

    if (res_code = 'success') then
        is_system_error = 0;

    suspend;
end
