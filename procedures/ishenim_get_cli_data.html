CREATE PROCEDURE ISHENIM_GET_CLI_DATA(
    POKET_ID INTEGER,
    K_ID ID_TYPE)
RETURNS (
    C_TYPE D_SMALLINT,
    C_ID ID_TYPE,
    CITIZENSHIP VARCHAR(10),
    TRAVELPASSPORTISSUERCOUNTRY VARCHAR(10),
    GENDER VARCHAR(10),
    DATEOFBIRTH VARCHAR(10),
    PLACEOFBIRTH STR_100,
    MARITALSTATUS STR_100,
    SOCCODE VARCHAR(20),
    CUSTOMERCODE ID_TYPE,
    PRESENTSURNAME STR_100,
    FIRSTNAME STR_100,
    PATRONYMIC STR_100,
    FULLNAME STR_100,
    CLASSIFICATIONOFINDIVIDUAL STR_100,
    MAINADDRESS STR_255,
    ROLEOFCUSTOMER STR_100,
    DEPENDENTCOUNT INTEGER,
    INTERNALPASSPORT STR_100,
    INTERNALPASSPORTISSUEDATE DATE,
    INTERNAPASSPORTEXPIRATIONDATE DATE,
    INTERNALPASSPORTISSUEAUTHORITY STR_255,
    INTERNALPASSPORTISSUERCOUNTRY VARCHAR(2),
    TOTALAMOUNT CURR_TYPE,
    CONTRACTSTATUS STR_100,
    PRINCIPALOUTSTANDINGAMOUNT CURR_TYPE,
    COMPANYNAME D_STR_100,
    ESTABLISHMENTDATE D_DATE,
    REGISTRATIONCOUNTRY D_STR_10,
    INN VARCHAR(20))
AS
declare variable fname str_100;
declare variable lname str_100;
declare variable kk integer;
declare variable c_name_fio str_100;
declare variable rep_date date;
declare variable oks_balance curr_type;
declare variable k_amount_approved curr_type;
declare variable cli_qty integer;
declare variable kc_closed_ol_id id_type;
declare variable oks_balance_calc curr_type;
declare variable om_perc perc_type;
declare variable k_fact_closed_ol_id id_type;
declare variable kont_contractstatus str_100;
declare variable k_kolich_client_approved integer;
begin
  --citizenship = 'KG';
  --travelpassportissuercountry = 'KG';

  select cp.cp_date_end
  from cib_pocket cp
  where cp.cp_id = :poket_id
  into :rep_date;

  select oks.oks_balance
  from oksummary oks
  where oks.oks_from_k_id = :k_id and
        :rep_date between oks.oks_date_begin and oks.oks_date_end and
        oks_close_state = 0 
  into :oks_balance;

  select kont.k_amount_approved, kont.k_kolich_client_approved, ol.ol_id
  from kontract kont
  left join operlist ol on ol.ol_id = kont.k_fact_closed_ol_id and
        ol.is_delete = 0
  where kont.k_id = :k_id
  into :k_amount_approved, k_kolich_client_approved, k_fact_closed_ol_id;

  cli_qty = k_kolich_client_approved;

  if (k_fact_closed_ol_id is null) then
    kont_contractstatus = 'GrantedAndActivated';
  else
  begin
    kont_contractstatus = 'Settled';
    if (exists(select *
               from operkred ok
               where ok.ok_from_ol_id = :k_fact_closed_ol_id and
                     ok.ok_from_k_id = :k_id and
                     ok.ok_type = 5 and
                     ok.ok_os < 0 and
                     ok_from_c_id = 0)) then
      kont_contractstatus = 'WrittenOff';
  end
  oks_balance_calc = oks_balance;

  -- Клиенты ------->
  for select cl.c_id,
             c_name_fio,
             trim(iif(c_sex = 1, 'Male', 'Female')),
             (select result_date_str from cib_date_to_str(cl.c_birth_date)) as dateofbirth, coalesce(cl.c_birth_city, ''),
             (case cl.c_family
                   when 14 then 'Single'
                   when 15 then 'Married'
                   when 16 then 'Divorced'
                   when 17 then 'Divorced'
                   else 'Other'
             end),
             cl.c_inn,
             coalesce(cl.c_kol_child, 0),
             cl.c_pass_no,
             cl.c_pass_date,
             cl.c_pass_date_expire,
             cl.c_pass_who,
             kncl.kc_amount_approved,
             ol.ol_id,
             cl.c_type, -- Тип клиента: 0 - физ лицо, 1 - юр. лицоб
             (case rd.rd_code when 'Citizen.KG' then 'KG'
                              when 'Citizen.UZ' then 'UZ'
                              when 'Citizen.TJ' then 'TJ'
                              when 'Citizen.RU' then 'RU'
             end)
      from knclient kncl
      inner join clients cl on cl.c_id = kncl.kc_from_c_id
      inner join refdata rd on rd.rd_id = cl.c_citizen
      left join operlist ol on ol.ol_id = kc_closed_ol_id and ol.is_delete = 0
      where kncl.kc_amount_approved > 0 and
            kncl.kc_from_k_id = :k_id
      order by ol_id desc
      into :c_id, :c_name_fio, :gender, :dateofbirth, :placeofbirth, :maritalstatus, :soccode, :dependentcount,
           :internalpassport, :internalpassportissuedate, :internapassportexpirationdate,
           :internalpassportissueauthority, :totalamount, :kc_closed_ol_id, :c_type,
           :citizenship
  do
  begin
    contractstatus = kont_contractstatus;
    travelpassportissuercountry = citizenship;
    internalpassportissuercountry = citizenship;

    if (kc_closed_ol_id is not null) then
    begin
      if (exists(select *
                 from operkred ok
                 where ok.ok_from_ol_id = :kc_closed_ol_id and
                       ok.ok_from_k_id = :k_id and
                       ok.ok_type = 5 and
                       ok.ok_os < 0)) then
        contractstatus = 'WrittenOff';
      else
        contractstatus = 'Settled';
    end
    if (contractstatus = 'GrantedAndActivated') then
    begin
      if (cli_qty = 1) then
        principaloutstandingamount = oks_balance_calc;
      else
      begin
        om_perc = totalamount * 100 / k_amount_approved; -- доля уч-ка в процентном соотношении к доли группы
        principaloutstandingamount = round(oks_balance * om_perc / 100);
        if (totalamount < principaloutstandingamount) then
          principaloutstandingamount = totalamount;
        oks_balance_calc = oks_balance_calc - principaloutstandingamount;
      end
    end
    else
      principaloutstandingamount = 0;
    cli_qty = cli_qty - 1;

    if ((internapassportexpirationdate is null) and
        (internalpassportissuedate is not null)) then
      internapassportexpirationdate = dateadd(year, 10, internalpassportissuedate);

    customercode = c_id;
    fullname = c_name_fio;

    if (placeofbirth = '') then
      placeofbirth = 'Кыргызстан';

    fname = '';
    lname = '';
    patronymic = '';

    if (c_name_fio containing ('кызы ')) then
    begin
      kk = position('кызы ', c_name_fio);
      lname = substring(c_name_fio from 1 for kk + 3);
      fname = substring(c_name_fio from kk + 5);
    end
    else
    if (c_name_fio containing ('уулу ')) then
    begin
      kk = position('уулу ', c_name_fio);
      lname = substring(c_name_fio from 1 for kk + 3);
      fname = substring(c_name_fio from kk + 5);
    end
    else
    begin
      kk = position(' ', :c_name_fio) - 1;
      if (kk > 0) then
      begin
        lname = substring(:c_name_fio from 1 for kk);
        c_name_fio = substring(:c_name_fio from kk + 2 for 255);
        kk = position(' ', c_name_fio) - 1;
        if (kk > 0) then
        begin
          fname = substring(:c_name_fio from 1 for kk);
          c_name_fio = substring(:c_name_fio from kk + 2 for 255);
          patronymic = :c_name_fio;
        end
        else
          fname = :c_name_fio;
      end
    end

    presentsurname = lname;
    firstname = fname;
    classificationofindividual = 'Individual';

    select adl.adl_address_str
    from addrlist adl
    inner join addrclient adcl on adl.adl_id = adcl.adc_from_adl_id
    where adcl.adc_from_c_id = :c_id and
          adcl.adc_address_type = 1
    into :mainaddress;

    roleofcustomer = 'MainDebtor';

    -------------- Если клиент юр лицо ------ >
    companyname = '';
    establishmentdate = null;
    registrationcountry = '';
    inn = '';

    if (c_type = 1) then
    begin
        companyname = fullname;
        establishmentdate = dateofbirth;
        registrationcountry = citizenship;
        inn = soccode;
    end
    ----------------------------------------- <

    suspend;
  end
  -- < -----------------
  -- Поручители ------->
  for select cl.c_id, cl.c_name_fio, trim(iif(c_sex = 1, 'Male', 'Female')),
               (select result_date_str
                from cib_date_to_str(cl.c_birth_date)) as dateofbirth, coalesce(cl.c_birth_city, ''),
               (
               case cl.c_family
                 when 14 then 'Single'
                 when 15 then 'Married'
                 when 16 then 'Divorced'
                 when 17 then 'Divorced'
                 else 'Other'
               end),
               cl.c_inn, coalesce(cl.c_kol_child, 0), cl.c_pass_no, cl.c_pass_date, cl.c_pass_date_expire,
               cl.c_pass_who,
               cl.c_type,
               (case rd.rd_code when 'Citizen.KG' then 'KG'
                                when 'Citizen.UZ' then 'UZ'
                                when 'Citizen.TJ' then 'TJ'
                                when 'Citizen.RU' then 'RU'
               end)
        from zalog zal
        inner join clients cl on cl.c_id = zal.z_from_c_id
        inner join refdata rd on rd.rd_id = cl.c_citizen
        where zal.z_from_k_id = :k_id and
              zal.z_from_c_id is not null and
              zal.z_type = 0 -- 0 - поручитель, 1 - залог
        into :c_id, :c_name_fio, :gender, :dateofbirth, :placeofbirth, :maritalstatus, :soccode, :dependentcount,
             :internalpassport, :internalpassportissuedate, :internapassportexpirationdate,
             :internalpassportissueauthority, :c_type, :citizenship     
  do
  begin
      if ((internapassportexpirationdate is null) and
          (internalpassportissuedate is not null)) then
        internapassportexpirationdate = dateadd(year, 10, internalpassportissuedate);

      --internalpassportissuercountry = 'KG';
      travelpassportissuercountry = citizenship;
      internalpassportissuercountry = citizenship;

      customercode = c_id;
      fullname = c_name_fio;

      if (placeofbirth = '') then
        placeofbirth = 'Кыргызстан';

      fname = '';
      lname = '';
      patronymic = '';

      if (c_name_fio containing ('кызы ')) then
      begin
        kk = position('кызы ', c_name_fio);
        lname = substring(c_name_fio from 1 for kk + 3);
        fname = substring(c_name_fio from kk + 5);
      end
      else
      if (c_name_fio containing ('уулу ')) then
      begin
        kk = position('уулу ', c_name_fio);
        lname = substring(c_name_fio from 1 for kk + 3);
        fname = substring(c_name_fio from kk + 5);
      end
      else
      begin
        kk = position(' ', :c_name_fio) - 1;
        if (kk > 0) then
        begin
          lname = substring(:c_name_fio from 1 for kk);
          c_name_fio = substring(:c_name_fio from kk + 2 for 255);
          kk = position(' ', c_name_fio) - 1;
          if (kk > 0) then
          begin
            fname = substring(:c_name_fio from 1 for kk);
            c_name_fio = substring(:c_name_fio from kk + 2 for 255);
            patronymic = :c_name_fio;
          end
          else
            fname = :c_name_fio;
        end
      end

      presentsurname = lname;
      firstname = fname;
      classificationofindividual = 'Individual';

      select adl.adl_address_str
      from addrlist adl
      inner join addrclient adcl on adl.adl_id = adcl.adc_from_adl_id
      where adcl.adc_from_c_id = :c_id and
            adcl.adc_address_type = 1
      into :mainaddress;

      roleofcustomer = 'Guarantor';
      contractstatus = kont_contractstatus;
      principaloutstandingamount = 0;
      totalamount = 0;

      -------------- Если поручитель юр лицо ------ >
      companyname = '';
      establishmentdate = null;
      registrationcountry = '';
      inn = '';

      if (c_type = 1) then
      begin
        companyname = c_name_fio;
        establishmentdate = dateofbirth;
        registrationcountry = citizenship;
        inn = soccode;
      end
      ----------------------------------------- <
      suspend;
  end
end
