CREATE PROCEDURE HP_JIRA_ISSUES_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    SHOW_ALL D_BOOL,
    SHOW_KPI D_BOOL)
RETURNS (
    JRD_LOGIN D_STR_CODE,
    JRD_FIO D_STR_100,
    JRD_COUNTRY D_STR_10,
    JRD_EMAIL D_STR_100,
    JRD_LEVEL D_SMALLINT,
    JRD_IS_MANAGER D_BOOL,
    JRD_IS_DEV D_BOOL,
    JRS_TEXT D_STR_CODE,
    JRS_CODE D_STR_CODE,
    JRS_RESULT D_SMALLINT,
    JRI_ISSUE_ID D_INTEGER,
    JRI_DEV_ID D_ID,
    JRI_JIRA_CODE D_STR_CODE,
    JRI_TITLE D_STR_LONG,
    JRI_PROJECT D_STR_CODE,
    JRI_LEVEL D_SMALLINT,
    JRI_ESTIMATE_SECS D_ID,
    JRI_ESTIMATE_PERIOD D_STR_CODE,
    JRI_CREATED D_DATETIME,
    JRI_FINISHED D_DATETIME,
    JRI_CLOSED D_DATETIME,
    JRI_STATE_ID D_SMALLINT,
    JRI_RECALC D_BOOL,
    JRI_KPI_PERIOD D_INTEGER,
    JRI_CHECK_STATUS D_SMALLINT,
    JRI_CHECK_MEMO D_STR_255,
    JRT_TEXT D_STR_CODE,
    JRT_KPI_KIND D_SMALLINT,
    JRI_WORK_STARTED D_DATETIME,
    JRI_WORKED_TIME D_STR_CODE,
    JRI_ESTIMATE_TIME D_STR_CODE,
    JRI_LEFT_TIME D_STR_CODE,
    JRI_WARNING D_SMALLINT,
    GMT_TIMEZONE D_STR_10,
    JRI_JIRA_WEBLINK D_STR_LONG,
    JRI_FILL_STATE D_SMALLINT,
    JRI_CATEGORY D_STR_255,
    JRI_SOFTWARE D_STR_255,
    JRI_JOB_KIND_STR D_STR_CODE,
    JRI_JOB_KIND D_INTEGER,
    JRI_CATEGORY_ID D_INTEGER,
    JRI_SOFTWARE_ID D_INTEGER,
    JRI_WORKED_SECOND D_ID)
AS
declare variable left_second d_integer;
declare variable jri_kpi_timezone d_smallint;
declare variable dt_begin d_datetime;
declare variable dt_end d_datetime;
declare variable un_login d_str_code;
declare variable jri_job_kind_prv d_integer;
declare variable jrd_db_login d_str_code;
begin
  dt_begin = rep_date_begin;
  dt_end = rep_date_end + 1;

  if (show_all is null) then
    show_all = 0;

  if (dt_begin is null or show_all = 1) then
    dt_begin = '01.01.2000';

  if (dt_end is null or show_all = 1) then
    dt_end = current_timestamp;

  select un_name
  from username
  where un_id = uxf$user()
  into un_login;

  if (show_kpi is null) then
    show_kpi = 1;

  for select jri_issue_id,
             jri_dev_id,
             jri_jira_code,
             jri_title,
             jri_project,
             jri_level,
             jri_estimate_secs,
             jri_estimate_period,
             jri_created,
             jri_finished,
             jri_closed,
             jri_state_id,
             jri_recalc,
             jri_kpi_period,
             jri_check_status,
             jri_check_memo,
             jri_kpi_timezone,

             jrd_login,
             jrd_db_login,
             jrd_fio, 
             jrd_country, 
             jrd_email, 
             jrd_level, 
             jrd_is_manager, 
             jrd_is_dev,

             jrs_text,
             jrs_code,
             jrs_result,

             jrt_text,
             jrt_kpi_kind,

             jrc_full_code,
             jsw_full_code,
             jri_job_kind,
             jri_category_id,
             jri_software_id
      from jira_issues, jira_dev, jira_states, jira_types, jira_category, jira_soft
      where jri_dev_id = jrd_id and
            jri_state_id = jrs_id and
            jri_type_id = jrt_id and
            jri_category_id = jrc_id and
            jri_software_id = jsw_id and
            jrt_kpi_kind >= :show_kpi and
            (jri_created between :dt_begin and :dt_end or
            (jri_finished between :dt_begin and :dt_end or jri_finished is null))
      into :jri_issue_id,
           :jri_dev_id,
           :jri_jira_code,
           :jri_title,
           :jri_project,
           :jri_level,
           :jri_estimate_secs,
           :jri_estimate_period,
           :jri_created,
           :jri_finished,
           :jri_closed,
           :jri_state_id,
           :jri_recalc,
           :jri_kpi_period,
           :jri_check_status,
           :jri_check_memo,
           :jri_kpi_timezone,

           :jrd_login,
           :jrd_db_login,
           :jrd_fio,
           :jrd_country,
           :jrd_email,
           :jrd_level,
           :jrd_is_manager,
           :jrd_is_dev,

           :jrs_text,
           :jrs_code,
           :jrs_result,

           :jrt_text,
           :jrt_kpi_kind,

           :jri_category,
           :jri_software,
           :jri_job_kind,
           :jri_category_id,
           :jri_software_id
  do
  begin
    -- Кроме руководителей остальные видят только свои задачи
    if (un_login not in ('igusev', 'aschegolskiy', 'sislamov', 'eabdykulova', 'mmirzaitova')) then
    begin
      if (un_login is distinct from jrd_db_login) then
        Continue;
    end

    if (jri_job_kind is distinct from jri_job_kind_prv) then
    begin
      jri_job_kind_prv = jri_job_kind;
      jri_job_kind_str = f_enum_name('jira_job_kind', jri_job_kind, 0);
    end

    jri_fill_state = 0;
    if (jri_category = '' or jri_software = '' or jri_job_kind = 0) then
      jri_fill_state = 1;
    else
    if (jri_category = '?' or jri_software = '?') then
      jri_fill_state = 2;

    jri_estimate_time = jri_estimate_period;

    jri_jira_weblink = '';
    if (jri_jira_code <> '') then
      jri_jira_weblink = 'https://jira.mbulak.kg/browse/' || jri_jira_code;

    jri_work_started = null;
    jri_worked_second = 0;

    if (jri_kpi_timezone = 0) then
      gmt_timezone = 'GMT БД';
    else
    if (jri_kpi_timezone > 0) then
      gmt_timezone = 'GMT+' || jri_kpi_timezone;
    else
      gmt_timezone = 'GMT' || jri_kpi_timezone;

    select min(min_datetime), sum(second_total)
    from hp_jira_log_detail(:jri_issue_id, null, null)
    where jrs_code = 'work'
    into jri_work_started, jri_worked_second;

    if (jri_worked_second is null) then
      jri_worked_second = 0;

    jri_worked_time = f_jira_period(jri_worked_second);

    jri_warning = 0;
    left_second = 0;
    jri_left_time = '-';

    if (jri_estimate_secs > 0) then
    begin
      left_second = jri_estimate_secs - jri_worked_second;
      jri_left_time = f_jira_period(abs(left_second));
      if (left_second < 0) then
      begin
        jri_left_time = '- ' || jri_left_time;
        jri_warning = 2; -- Просрочен
      end else
      if (jri_worked_second > jri_estimate_secs * 0.9) then
        jri_warning = 1; -- Осталось меньше 10%
    end else
      jri_warning = -1; -- Время не указано

    suspend;
  end
end
