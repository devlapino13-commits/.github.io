CREATE PROCEDURE PRINT_PENSION(
    K_ID D_ID)
RETURNS (
    PENSION_AMOUNT D_CURRENCY,
    PENSION_TYPE_DESCRIPTION D_STR_255,
    PENSION_GROUP_DESCRIPTION D_STR_255)
AS
declare variable COUNTER D_INTEGER;
declare variable PENSION_COUNTER D_INTEGER;
declare variable C_ID D_ID;
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_BLOB_TEXT;
begin
  -- На момент написания в АЗ нет групповых кредитов
  select kc.kc_from_c_id
  from knclient kc
  where kc.kc_from_k_id = :k_id
  into :c_id;

  select eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id and
        eic.eic_code = 'pension'
  where eil.eil_from_c_id = :c_id and
        eil.eil_status = 2
  order by eil.eil_id desc
  rows 1
  into :bls_id;

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into :json_text;

  pension_counter = f_json_str_get_count(:json_text, 'Pension');
  counter = 0;

  while (pension_counter > counter) do
  begin
    pension_type_description = f_json_str_get_value(:json_text, 'Pension[' || cast(counter as varchar(25)) || '].Type.Description', '');
    pension_group_description = f_json_str_get_value(:json_text, 'Pension[' || cast(counter as varchar(25)) || '].Group.Description', '');
    pension_amount = f_json_str_get_value(:json_text, 'Pension[' || cast(counter as varchar(25)) || '].Amount', '');
    counter = counter + 1;
    suspend;
  end
end
