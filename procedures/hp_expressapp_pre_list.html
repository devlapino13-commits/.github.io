CREATE PROCEDURE HP_EXPRESSAPP_PRE_LIST(
    EXP_ID D_INTEGER)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    RECOM_SUM D_CURRENCY,
    RECOM_PERIOD D_INTEGER,
    DISB_DATE D_DATE,
    POINT D_INTEGER,
    KS_FIO D_STR_100)
AS
declare variable last_amount_approved d_integer;
declare variable k_id d_id;
declare variable sum_active_dolg d_currency;
declare variable active_k_id d_id;
declare variable active_dolg d_currency;
declare variable c_id_spouse d_id;
declare variable sum_active_dolg_spouse d_currency;
declare variable active_k_id_spouse d_id;
declare variable active_dolg_spouse d_currency;
declare variable rep_region_id d_integer;
declare variable rep_date_begin d_date;
declare variable rep_date_end d_date;
declare variable balance d_currency;
declare variable k_from_kf_id id_type;
begin
  select
    exp.exp_from_reg_id,
    exp.exp_date_begin,
    exp.exp_date_end
  from expresspkg exp
  where exp.exp_id = :exp_id
  into
    :rep_region_id,
    :rep_date_begin,
    :rep_date_end;

  for
    select
      kc.kc_from_c_id,
      k.k_id,
      kc.kc_amount_approved,
      k.k_period,
      ks.ks_date_begin,
      k.k_from_kf_id
    from kontstate ks
      inner join kontract k
        on k.k_id = ks.ks_from_k_id
      inner join product pr
        on pr.pr_id = k.k_from_pr_id
          and pr.pr_code not containing '.IPF.'
      inner join knclient kc
        on kc.kc_from_k_id = k.k_id
          and kc_amount_approved > 0
      inner join region_list(:rep_region_id, 1, 3, 1) t
        on t.id_list = ks.ks_region_id + 0
    where ks.ks_date_begin between :rep_date_begin and :rep_date_end
      and ks.ks_state = 2
      and ks.ks_ol_is_delete = 0
      and not exists(
        select *
        from expressapp exa
        where exa.exa_from_c_id = kc.kc_from_c_id
          and exa.exa_from_exp_id > 1
          and exa.exa_final_status in (0, 1))
  into
    :c_id,
    :k_id,
    :last_amount_approved,
    :recom_period,
    :disb_date,
    :k_from_kf_id
  do
  begin
    if (not exists(
      select *
      from knclient kc
        inner join kontstate ks
          on ks.ks_from_k_id = kc.kc_from_k_id
            and current_date between ks.ks_date_begin and ks.ks_date_end
            and ks.ks_state = 1
            and ks.ks_ol_is_delete = 0
        inner join kontract k
          on k.k_id = kc.kc_from_k_id
            and k.k_id <> :k_id
        inner join product pr         -- Убрал условие, письмо от Азамата М. 26.05.2022
          on pr.pr_id = k.k_from_pr_id
            and not pr.pr_code containing '.IPF.'
      where kc.kc_from_c_id = :c_id
        and kc_amount_approved > 0)) then
    begin
      point = f_point(:c_id, current_date);

      if (point < 10) then continue;   -- Условие добавлено по просьбе Бактыгуль Дж. 26.05.2022

      balance = 0;
      recom_sum = 0;

      ks_fio = null;
      select cl.c_name_fio
      from knaktlist kal
      inner join clients cl on cl.c_id = kal.kal_new_kf_officer
      where kal.kal_from_kf_id = :k_from_kf_id and
            kal.is_delete = 0 and
            current_date between kal.kal_date and kal.kal_date_end
      into ks_fio;

      c_name_fio = (
        select
          c.c_name_fio
        from clients c
        where c.c_id = :c_id);

      sum_active_dolg = 0;
      for
        select
          ks.ks_from_k_id
        from kontstate ks
          inner join knclient kc
            on kc.kc_from_k_id = ks.ks_from_k_id
              and kc.kc_amount_approved > 0
        where kc.kc_from_c_id = :c_id
          and :rep_date_end between ks.ks_date_begin and ks.ks_date_end
          and ks.ks_state = 1
          and ks.ks_ol_is_delete = 0
      into
        :active_k_id
      do
      begin
        active_dolg = coalesce((
          select
            sum(ok.ok_os + ok_sh_os)
          from operkred ok
          where ok.ok_from_k_id = :active_k_id
            and ok_from_c_id = 0
            and ok.ok_type < 7
            and ok.ok_ol_is_active = 1), 0);
        sum_active_dolg = sum_active_dolg + active_dolg;
      end

      c_id_spouse = (
        select
          crel.crel_other_c_id
        from clientrelation crel
        where crel.crel_from_c_id = :c_id
          and crel.crel_is_active = 1
          and crel.crel_from_rel_id in (1, 2)); -- супруг/супруга

      sum_active_dolg_spouse = 0;
      if (c_id_spouse is not null) then
      begin
        for
          select ks.ks_from_k_id
          from kontstate ks
            inner join knclient kc
              on kc.kc_from_k_id = ks.ks_from_k_id
                and kc.kc_amount_approved > 0
          where kc.kc_from_c_id = :c_id_spouse
            and :rep_date_end between ks.ks_date_begin and ks.ks_date_end
            and ks.ks_state = 1
            and ks.ks_ol_is_delete = 0
        into
          :active_k_id_spouse
        do
        begin
          active_dolg_spouse = coalesce((
            select
              sum(ok.ok_os + ok_sh_os)
            from operkred ok
            where ok.ok_from_k_id = :active_k_id_spouse
              and ok_from_c_id = 0
              and ok.ok_type < 7
              and ok.ok_ol_is_active = 1), 0);
          sum_active_dolg_spouse = sum_active_dolg_spouse + active_dolg_spouse;
        end
      end

      balance = 300000 - (sum_active_dolg + sum_active_dolg_spouse);
      recom_sum = iif(balance > :last_amount_approved, :last_amount_approved, balance);
      if (recom_sum > 100000) then   --
        recom_sum = 100000;

      suspend;
    end
  end
end
