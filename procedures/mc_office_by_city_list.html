CREATE PROCEDURE MC_OFFICE_BY_CITY_LIST(
    CITY_ID ID_TYPE,
    MCL_ID ID_TYPE)
RETURNS (
    ID ID_TYPE,
    NAME STR_255,
    LONGITUDE PERC_TYPE,
    LATITUDE PERC_TYPE,
    IS_DEFAULT D_SMALLINT)
AS
begin
  city_id = coalesce(city_id, 0);
  is_default = 0;

  for select reg.reg_unicode, reg.reg_name, adl.adl_long, adl.adl_lat
  from places pl
  inner join addrlist adl on adl.adl_from_pl_id = pl.pl_id
  inner join region reg on reg.reg_address_fact = adl.adl_id
  where pl.pl_id = :city_id and
        reg.reg_date_close is null and
        reg.is_delete = 0 and
        reg.reg_status = 3
  order by pl.pl_name
  into id, name, longitude, latitude
  do
  begin
    if (id not in (1012100, 1012300, 1010100, 1012700)) then
        suspend;
  end

  /* в данной процедуре нужно добавить условие при котором is_default = 1

    По умолчанию is_default  =0. Чз mcl_id получаю c_id и если клиента был активный или закрытый заем в течении 3 месяцев по отношению к текущей дате в рамках переданного города city_id, тогда получаю самый последний заем (ближайший к текущей дате), из займа получаю офис, офис и для этой записи is_default = 1.

    Для Тилека is_default  = 1 значит, что в контроле со списком офисов, офис с is_default = 1 должен быть автоматом выставлен.
    Записей с is_default  = 1 не может быть более 1-ой, но может быть что записей с is_default  = 1 нет вообще.
    Если нет ни одной записи с is_default  = 1, для Тилека это значит, что предопределённого офиса нет, и клиент должен сам руками выбрать гофис из списка

   */
end
