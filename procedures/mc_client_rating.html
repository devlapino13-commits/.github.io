CREATE PROCEDURE MC_CLIENT_RATING(
    MCL_ID D_ID)
RETURNS (
    SCORE_STATUS D_SMALLINT,
    SCORE_STATUS_CLASSIC D_SMALLINT,
    SCORE_STATUS_IPF D_SMALLINT,
    SCORE D_INTEGER,
    SCORE_CLASSIC D_INTEGER,
    SCORE_IPF D_INTEGER,
    NEXT_LOAN_AMOUNT D_INTEGER,
    SAVING_AMOUNT D_INTEGER,
    AVAILABLE_AMOUNT D_INTEGER,
    CURRENT_PERC_RATE DECIMAL(2,1),
    NEXT_PERC_RATE DECIMAL(2,1),
    PR_ID D_INTEGER)
AS
declare variable c_id id_type;
declare variable monthly_payment_next_score curr_type;
declare variable monthly_payment_0_score curr_type;
declare variable zero_perc_rate decimal(2,1);
declare variable activa_balance curr_type;
declare variable db_id d_integer;
declare variable calc_ipf d_integer;
declare variable calc_cls d_integer;
begin
  db_id = f_get_db_id();
  activa_balance = 0;

  mcl_id = coalesce(mcl_id, -777);
  if (mcl_id <= 0) then
    exception e_default '"mcl_id" is not valid';

  saving_amount = 0;
  score_status_classic = 0;
  score_classic = 0;
  score_status_ipf = 0;
  score_ipf = 0;


  select mc.mcl_from_c_id
  from mc_login mc
  where mc.mcl_id = :mcl_id
  into :c_id;

  -- посчитать балл клиента
  score = f_point(:c_id, current_date);

  if (c_id is null) then
    score_status = 0; --новый клиент
  else
  begin
    if (score > 0) then
      score_status = 1; --активный клиент
    else
    begin
      if (score = 0) then
      begin
        if (exists(select *
                   from clientstate cst
                   where cst.cs_from_c_id = :c_id and
                         cs_client_circle > 0)) then
        begin
          score_status = -1; --потерял баллы
        end
        else
          score_status = 0;
      end
    end

    select sum(oks.oks_balance)
    from oksummary oks
    inner join knclient kncl on oks.oks_from_k_id = kncl.kc_from_k_id
    where oks.oks_close_state = 0 and
          current_date between oks.oks_date_begin and oks.oks_date_end and
          kncl.kc_amount_approved > 0 and
          kncl.kc_from_c_id = :c_id
    into :activa_balance;

    activa_balance = coalesce(activa_balance, 0);
  end

  if (db_id = 81001) then
  begin
    select pr.pr_id
    from product pr
    where pr.pr_code like '%.reliable_client.%' and
          pr.is_delete = 0 and
          current_date between pr.pr_date_begin and pr.pr_date_end
    into :pr_id;
  end

  if (db_id in (39801, 41701)) then
  begin
    select pr.pr_id
    from product pr
    where pr.pr_code like '%.consultation.%' rows 1
    into :pr_id;
  end

  if (db_id = 94401) then
  begin
    select pr.pr_id
    from product pr
    where pr.pr_code like '%.score_rate.%'
    order by pr.pr_id desc
    rows 1
    into :pr_id;
  end

  select cast(percent as decimal(2,1)) -- добавить по гк
  from perc_amount_interval_by_point(:pr_id, 0)
  into :zero_perc_rate;

  select cast(percent as decimal(2,1)), amount_max
  from perc_amount_interval_by_point(:pr_id, :score)
  into :current_perc_rate, available_amount;

  available_amount = available_amount - activa_balance;
  if (available_amount < 0) then
    available_amount = 0;

  select cast(percent as decimal(2,1)), amount_max
  from perc_amount_interval_by_point(:pr_id, :score + 3)
  into :next_perc_rate, next_loan_amount;

  next_loan_amount = next_loan_amount - activa_balance;
  if (next_loan_amount < 0) then
    next_loan_amount = 0;

  -- определяем сумму экономии из расчета что срок займа = 12 месяцев
  monthly_payment_next_score = 0;
  monthly_payment_0_score = 0;

  -- рассчитываею ежемесячный палатеж по следующей % ставке -- >
  execute procedure hp_make_schedule_date(:next_loan_amount, 12, 0, current_date, :pr_id, 0, 0, '', 1, 0);
  execute procedure hp_make_schedule_perc(:next_loan_amount, 12, :pr_id, :next_perc_rate, 0, 0, 0);

  select sh_os + sh_perc
  from temp_schedule
  where sh_id = 1
  into monthly_payment_next_score;

  delete from temp_schedule;
  ------------------------------------------------------------ <
  -- рассчитываею ежемесячный палатеж по следующей % ставке -- >
  execute procedure hp_make_schedule_date(:next_loan_amount, 12, 0, current_date, :pr_id, 0, 0, '', 1, 0);
  execute procedure hp_make_schedule_perc(:next_loan_amount, 12, :pr_id, :zero_perc_rate, 0, 0, 0);

  select sh_os + sh_perc
  from temp_schedule
  where sh_id = 1
  into monthly_payment_0_score;

  delete from temp_schedule;
  ------------------------------------------------------------ <

  saving_amount = round(monthly_payment_0_score - monthly_payment_next_score) * 12;

  if (db_id in (39801, 41701)) then
  begin
    -- получаем баллы по классике
    score_classic = f_point(:c_id, current_date, 1);

    if (c_id is null) then
      score_status_classic = 0; --новый клиент

    if (score_classic > 0) then
      score_status_classic = 1; --активный клиент
    else
    begin
      if (score_classic = 0) then
      begin
        if (exists(select *
                   from clientstate cst
                   where cst.cs_from_c_id = :c_id and
                         cs_client_circle > 0)) then
        begin
          score_status_classic = -1; --потерял баллы
        end

        select sum(p.pts_saldo_cls)
        from pointsaldo p
        where pts_from_c_id = :c_id
        into :calc_cls;

        if (calc_cls is null or calc_cls = 0) then
        begin
          score_status_classic = 1;
        end
      end
      end
      -- получаем баллы по ипф
      score_ipf = f_point(:c_id, current_date, 2);

      if (c_id is null) then
        score_status_ipf = 0; --новый клиент

      if (score_ipf > 0) then
        score_status_ipf = 1; --активный клиент

      if (score_ipf = 0) then
      begin
        select sum(p.pts_saldo_ipf)
        from pointsaldo p
        where pts_from_c_id = :c_id
        into :calc_ipf;

        if (calc_ipf > 0) then
        begin
          score_status_ipf = -1; --потерял баллы
        end
        else
          score_status_ipf = 0;
      end
    end
  suspend;
end
