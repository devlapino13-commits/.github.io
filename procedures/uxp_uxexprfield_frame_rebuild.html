CREATE PROCEDURE UXP$UXEXPRFIELD_FRAME_REBUILD(
    UFR_UUID D_UUID)
AS
declare variable exp_ux_uuid d_uuid;
declare variable exp_kind d_integer;
begin
  for select exp_ux_uuid, exp_kind
  from ux$expression
  where exp_ux_uuid = :ufr_uuid
  into exp_ux_uuid, exp_kind
  do
    execute procedure uxp$uxexprfield_update(:exp_ux_uuid, :exp_kind);

  for select exp_ux_uuid, exp_kind
  from ux$expression, ux$action, ux$frame
  where ufr_uuid = :ufr_uuid and
        ua_owner = ufr_owner and
        ua_uuid = exp_ux_uuid
  into exp_ux_uuid, exp_kind
  do
    execute procedure uxp$uxexprfield_update(:exp_ux_uuid, :exp_kind);

  for select exp_ux_uuid, exp_kind
  from ux$expression, ux$action
  where ua_owner = :ufr_uuid and
        ua_uuid = exp_ux_uuid
  into exp_ux_uuid, exp_kind
  do
    execute procedure uxp$uxexprfield_update(:exp_ux_uuid, :exp_kind);


  for select exp_ux_uuid, exp_kind
  from ux$expression, ux$frparam
  where exp_ux_uuid = ufp_uuid and
        ufp_ufr_uuid = :ufr_uuid
  into exp_ux_uuid, exp_kind
  do
    execute procedure uxp$uxexprfield_update(:exp_ux_uuid, :exp_kind);

  for select exp_ux_uuid, exp_kind
  from ux$expression, ux$fritem
  where exp_ux_uuid = ufi_uuid and
        ufi_owner = :ufr_uuid
  into exp_ux_uuid, exp_kind
  do
    execute procedure uxp$uxexprfield_update(:exp_ux_uuid, :exp_kind);

  for select exp_ux_uuid, exp_kind
  from ux$expression, ux$grid
  where ugr_ufr_uuid = :ufr_uuid and
        ugr_uuid = exp_ux_uuid
  into exp_ux_uuid, exp_kind
  do
    execute procedure uxp$uxexprfield_update(:exp_ux_uuid, :exp_kind);

  for select exp_ux_uuid, exp_kind
  from ux$expression, ux$grid, ux$gritem
  where ugr_ufr_uuid = :ufr_uuid and
        ugr_uuid = ugi_ugr_uuid and
        ugi_uuid = exp_ux_uuid
  into exp_ux_uuid, exp_kind
  do
    execute procedure uxp$uxexprfield_update(:exp_ux_uuid, :exp_kind);
end
