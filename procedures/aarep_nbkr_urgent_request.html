CREATE PROCEDURE AAREP_NBKR_URGENT_REQUEST(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    K_DISB_DATE D_DATE,
    KC_AMOUNT_APPROVED DECIMAL(18,2),
    OKS_BALANCE DECIMAL(18,2),
    K_KONTRACT_NO D_STR_20,
    KAL_NEW_KF_NAME D_STR_20,
    K_PERCENT D_PERCENT,
    PR_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    REGION_NAME D_STR_100,
    CREDIT_TYPE D_STR_10,
    C_INN D_STR_20,
    SEX D_STR_10,
    C_NAME_FIO D_STR_100,
    BT_NAME D_STR_64,
    JUR_ADDRESS D_STR_255,
    FACT_ADDRESS D_STR_255,
    AGE D_INTEGER,
    WORK_STATUS_ANSWER D_STR_LONG,
    WORK_BUSINESS_KIND_ANSWER D_STR_LONG,
    WORK_JOB_NAME_ANSWER D_STR_LONG,
    WORK_STAJ_ANSWER D_STR_LONG,
    IND_ENTREPRENEUR_ANSWER D_STR_LONG,
    C_PFT_RISK_KOEF D_STR_20,
    C_PFT_RISK_MEMO D_STR_LONG)
AS
declare variable old_reg_id d_integer;
declare variable object_kind d_integer;
declare variable subject_kind d_integer;
declare variable work_status d_integer;
declare variable work_business_kind d_integer;
declare variable work_job_name d_integer;
declare variable work_staj d_integer;
declare variable ind_entrepreneur d_integer;
declare variable reg_id d_integer;
declare variable k_id d_id;
declare variable kf_id d_id;
declare variable pr_code d_str_100;
declare variable c_id d_id;
declare variable kc_business_type d_integer;
declare variable c_sex d_smallint;
declare variable c_birth_date d_date;
declare variable oks_balance_tmp decimal(18,2);
declare variable k_amount_approved decimal(18,2);
declare variable c_dolya d_currency;
begin
  old_reg_id = 0;
  object_kind = f_objkind('clients', 'id');
  subject_kind = f_objkind('refdata', 'id');
  work_status = f_get_rd_id('work_status');
  work_business_kind = f_get_rd_id('work_business_kind');
  work_job_name = f_get_rd_id('work_job_name');
  work_staj = f_get_rd_id('AnketaQuestions.work_staj');
  ind_entrepreneur = f_get_rd_id('Ind_Entrepreneur');

  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into reg_id
  do
  begin
    for select ks.ks_from_k_id
        from kontstate ks
        where :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_state = 1 and
              ks.ks_ol_is_delete = 0 and
              ks.ks_region_id = :reg_id
        into k_id
    do
    begin
      if (old_reg_id <> reg_id) then
      begin
        office_name = null;
        region_name = null;

        select rg.reg_name, rg2.reg_name
        from region rg
        inner join region rg2 on rg2.reg_unicode = rg.parent
        where rg.reg_unicode = :reg_id
        into office_name, region_name;

        old_reg_id = reg_id;
      end

      k_disb_date = null;
      k_kontract_no = null;
      k_percent = null;
      kf_id = null;
      pr_name = null;
      pr_code = null;

      select kt.k_disb_date, kt.k_kontract_no, kt.k_percent, kt.k_from_kf_id, pr.pr_name, pr.pr_code, kt.k_amount_approved
      from kontract kt
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      where kt.k_id = :k_id
      into k_disb_date, k_kontract_no, k_percent, kf_id, pr_name, pr_code, k_amount_approved;

      credit_type = iif(pr_code containing '.IPF.', 'ИПФ', 'Классика');

      kal_new_kf_name = null;
      select kal.kal_new_kf_name
      from knaktlist kal
      where kal.kal_from_kf_id = :kf_id and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into kal_new_kf_name;

      for select kc.kc_from_c_id, kc.kc_business_type, kc.kc_amount_approved
          from knclient kc
          where kc.kc_from_k_id = :k_id and
                kc.kc_amount_approved > 0
          into c_id, kc_business_type, kc_amount_approved
      do
      begin
        c_dolya = kc_amount_approved / k_amount_approved;

        c_name_fio = null;
        c_inn = null;
        c_sex = null;
        c_birth_date = null;
        c_pft_risk_koef = null;
        c_pft_risk_memo = null;

        select cl.c_name_fio, cl.c_inn, cl.c_sex, cl.c_birth_date,
              (case c_pft_risk_koef when 0 then 'Низкий'
                                    when 1 then 'Средний'
                                    when 2 then 'Высокий'
                                    else 'Не указан'
                                    end),
              cl.c_pft_risk_memo
        from clients cl
        where cl.c_id = :c_id
        into c_name_fio, c_inn, c_sex, c_birth_date, c_pft_risk_koef, c_pft_risk_memo;

        age = f_age(c_birth_date, :rep_date_end);

        sex = iif(c_sex = 1, 'Мужской', 'Женский');

        bt_name = null;

        select bt.bt_name
        from bustype bt
        where bt.bt_id = :kc_business_type
        into bt_name;

        jur_address = null;

        select adl.adl_address_str
        from addrclient adc
        inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
        where adc.adc_from_c_id = :c_id and
              adc.adc_address_type = 1 -- Юр адрес
        into jur_address;

        fact_address = null;

        select adl.adl_address_str
        from addrclient adc
        inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
        where adc.adc_from_c_id = :c_id and
              adc.adc_address_type = 2 -- Факт адрес
        into fact_address;

        work_status_answer = null;
        select rd.rd_name
        from objectprops opr
        inner join objectvalres ovr on ovr.ovr_from_opr_id = opr.opr_id
        inner join objectvalues ovs on ovs.ovs_from_opr_id = opr.opr_id
        inner join refdata rd on rd.rd_id = ovr.ovr_result_rd_id
        where opr.opr_object_kind = :object_kind and
              opr.opr_subject_kind = :subject_kind and
              opr.opr_object_id = :c_id and
              opr.opr_subject_id = :work_status and
              :k_disb_date between ovs.ovs_date_begin and ovs.ovs_date_end
        order by ovr.ovr_from_ovs_id desc
        rows 1
        into work_status_answer;

        work_business_kind_answer = null;
        select rd.rd_name
        from objectprops opr
        inner join objectvalres ovr on ovr.ovr_from_opr_id = opr.opr_id
        inner join objectvalues ovs on ovs.ovs_from_opr_id = opr.opr_id
        inner join refdata rd on rd.rd_id = ovr.ovr_result_rd_id
        where opr.opr_object_kind = :object_kind and
              opr.opr_subject_kind = :subject_kind and
              opr.opr_object_id = :c_id and
              opr.opr_subject_id = :work_business_kind and
              :k_disb_date between ovs.ovs_date_begin and ovs.ovs_date_end
        order by ovr.ovr_from_ovs_id desc
        rows 1
        into work_business_kind_answer;

        work_job_name_answer = null;
        select rd.rd_name || iif(ovr.ovr_value <> '', ': ' || trim(ovr.ovr_value), trim(''))
        from objectprops opr
        inner join objectvalres ovr on ovr.ovr_from_opr_id = opr.opr_id
        inner join objectvalues ovs on ovs.ovs_from_opr_id = opr.opr_id
        inner join refdata rd on rd.rd_id = ovr.ovr_result_rd_id
        where opr.opr_object_kind = :object_kind and
              opr.opr_subject_kind = :subject_kind and
              opr.opr_object_id = :c_id and
              opr.opr_subject_id = :work_job_name and
              :k_disb_date between ovs.ovs_date_begin and ovs.ovs_date_end
        order by ovr.ovr_from_ovs_id desc
        rows 1
        into work_job_name_answer;

        work_staj_answer = null;
        select rd.rd_name
        from objectprops opr
        inner join objectvalres ovr on ovr.ovr_from_opr_id = opr.opr_id
        inner join objectvalues ovs on ovs.ovs_from_opr_id = opr.opr_id
        inner join refdata rd on rd.rd_id = ovr.ovr_result_rd_id
        where opr.opr_object_kind = :object_kind and
              opr.opr_subject_kind = :subject_kind and
              opr.opr_object_id = :c_id and
              opr.opr_subject_id = :work_staj and
              :k_disb_date between ovs.ovs_date_begin and ovs.ovs_date_end
        order by ovr.ovr_from_ovs_id desc
        rows 1
        into work_staj_answer;

        ind_entrepreneur_answer = null;
        select rd.rd_name
        from objectprops opr
        inner join objectvalres ovr on ovr.ovr_from_opr_id = opr.opr_id
        inner join objectvalues ovs on ovs.ovs_from_opr_id = opr.opr_id
        inner join refdata rd on rd.rd_id = ovr.ovr_result_rd_id
        where opr.opr_object_kind = :object_kind and
              opr.opr_subject_kind = :subject_kind and
              opr.opr_object_id = :c_id and
              opr.opr_subject_id = :ind_entrepreneur and
              :k_disb_date between ovs.ovs_date_begin and ovs.ovs_date_end
        order by ovr.ovr_from_ovs_id desc
        rows 1
        into ind_entrepreneur_answer;

        oks_balance = 0;


        select oks.oks_balance
        from oksummary oks
        where oks.oks_from_k_id = :k_id and
              :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
              oks.oks_close_state = 0
        into oks_balance;

        oks_balance = oks_balance * c_dolya;

        suspend;
      end
    end
  end
end
