CREATE PROCEDURE HP_CARD_INFO(
    ID D_STR_20,
    KIND D_SMALLINT = 0,
    DTBEG D_DATE = null,
    DTEND D_DATE = null,
    ONLY_ISSUED D_BOOL = 1)
RETURNS (
    CARD_ID D_INTEGER,
    CCA_ID D_INTEGER,
    K_ID D_ID,
    K_NO D_STR_20,
    C_ID D_ID,
    C_NAME D_STR_100,
    DT_OPEN D_DATE,
    AMOUNT_BEG D_CURRENCY,
    AMOUNT_DT D_CURRENCY,
    AMOUNT_KT D_CURRENCY,
    AMOUNT_END D_CURRENCY)
AS
declare variable acc d_str_10;
declare variable dtbuff d_date;
declare variable k_amount d_currency;
declare variable status d_integer;
declare variable p d_integer;
declare variable specific_loan boolean;
begin
  dtbeg = coalesce(dtbeg, '01.01.1900');
  dtend = coalesce(dtend, '31.12.9999');
  status = f_enum_id('cardlist_status', 'transfer_to_client');

  p = position('-', id);
  specific_loan = false;
  if (p > 0) then
  begin
    if (p = 1) then
    begin
      id = -cast(id as d_id);
      specific_loan = true;
    end else
      id = (select k_id from kontract where k_kontract_no = :id);
    kind = 2;
  end

  -- № карты
  if (kind = 0) then
    select
      cal_card_id,
      cal_from_cca_id
    from cardlist
    where cal_card_id = :id
      and cal_status = :status
    into
      :card_id,
      :cca_id;
  -- № картсчета
  if (kind = 1) then
    select
      cal_card_id,
      cal_from_cca_id
    from cardlist
    where cal_from_cca_id = :id
      and cal_status = :status
    into
      :card_id,
      :cca_id;
  -- № контракта
  if (kind = 2) then
    select
      cal_card_id,
      cal_from_cca_id
    from kontract
      inner join cardlistloan
        on clo_id = (
          select first 1 clo_id
          from cardlistloan
          where clo_from_k_id = k_id
          order by clo_date desc)
      inner join cardlist
        on cal_card_id = clo_card_id
          and cal_status = :status
    where k_id = :id
    into
      :card_id,
      :cca_id;

  select
    k_id,
    k_kontract_no,
    k_amount_approved,
    c_id,
    c_name_fio,
    cll_date_begin
  from cardlistlnk
    inner join kontract
      on k_id = cll_from_k_id
    inner join clients
      on c_id = cll_from_c_id
  where cll_card_id = :card_id
  into
    :k_id,
    :k_no,
    :k_amount,
    :c_id,
    :c_name,
    :dt_open;

  acc = '20001/1';
  with cards as
  (
    select distinct clo2.clo_card_id as card_id
    from cardlistloan as clo1
      inner join cardlistloan as clo2
        on clo2.clo_from_k_id = clo1.clo_from_k_id
    where clo1.clo_card_id = :card_id
  )
  select
    sum(iif(ol_date < :dtbeg, iif(ob_short_acc_kt = :acc, ob_summa, 0) - iif(ob_short_acc_dt = :acc, ob_summa, 0), 0)),
    sum(iif(ol_date between :dtbeg and :dtend and ob_short_acc_dt = :acc, ob_summa, 0)),
    sum(iif(ol_date between :dtbeg and :dtend and ob_short_acc_kt = :acc, ob_summa, 0)),
    sum(iif(ol_date <= :dtend, iif(ob_short_acc_kt = :acc, ob_summa, 0) - iif(ob_short_acc_dt = :acc, ob_summa, 0), 0))
  from operbook
    inner join (
      select distinct ol_id as olo_id
        from cardlistloan
          inner join operkred
            on ok_from_k_id = clo_from_k_id
              and (not :specific_loan or ok_from_k_id = :id)
          inner join operlist
            on ol_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = ok_from_ol_id)
        where clo_card_id = :card_id
      union
      select distinct ol_id as olo_id
        from cardlistloan
          inner join operkroff
            on oko_from_k_id = clo_from_k_id
              and (not :specific_loan or oko_from_k_id = :id)
          inner join operlist
            on ol_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = oko_from_ol_id)
        where clo_card_id = :card_id
      union
      select ol_id as olo_id
        from pvnopers
          inner join cards
            on card_id = po_card_id
          inner join operlist
            on ol_parent_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = po_from_ol_id )
              or ol_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = po_from_ol_id )
        where po_is_mb_card = 1
          and po_from_ol_id > 0
          and po_is_delete = 0
          and (not :specific_loan or po_from_k_id = :id)
      union
      select ol_id as olo_id
        from ipc_exd_detail
          inner join cards
            on card_id = ied_from_card_id
          inner join operlist
            on ied_from_ol_id in (ol_id, ol_parent_id)
        where ied_from_ol_id > 0
          and (not :specific_loan or ied_from_k_id = :id)
      --
      union
      select
        ol_id as olo_id
      from cardaktclient
        inner join refdata
          on rd_id = cac_from_rd_id
            and rd_code in ('ReceivCard_LossTheft', 'ReceivCard_Spoilage')
        inner join cards
          on card_id = cac_card_id
        inner join operlist
          on ol_id = cac_from_ol_id_com
      where cac_status >= 4
        and (not :specific_loan or cac_from_k_id = :id)
      union
      select
        iif(:card_id = 618870, 1090100070021223,
          iif(:card_id = 647842, 1090100069992968, 1090100070021033)) as olo_id
      from rdb$database
      where :card_id in (618870, 647842, 669459)
      ) as a
    on a.olo_id = ob_from_ol_id
    inner join operlist
      on ol_id = ob_from_ol_id
        and is_delete = 0
  where ob_short_acc_dt = :acc or ob_short_acc_kt = :acc
  into
    :amount_beg,
    :amount_dt,
    :amount_kt,
    :amount_end;

  if (card_id in (2028, 2030, 2043, 2015)) then
  begin
    dtbuff = iif(card_id in (2028, 2030), '23.04.2021', '28.04.2021');
    amount_beg = 0;
    amount_kt = iif(dtbuff between dtbeg and dtend, k_amount, 0);
    amount_dt = amount_kt;
    amount_end = amount_beg + amount_kt - amount_dt;
  end

  if (only_issued = 0 or row_count = 1)
    then suspend;
end
