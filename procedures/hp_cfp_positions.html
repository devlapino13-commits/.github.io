CREATE PROCEDURE HP_CFP_POSITIONS(
    ID D_INTEGER = null)
RETURNS (
    CP_ID D_INTEGER,
    CP_PARENET_ID D_INTEGER,
    CP_INCOME_PREFIX_RU D_STR_255,
    CP_INCOME_PREFIX_KG D_STR_255,
    CP_NAME_RU D_STR_100,
    CP_NAME_KG D_STR_100,
    HTML_TXT D_BLOB_TEXT,
    CP_TYPE D_INTEGER,
    CP_IS_DELETE D_BOOL,
    CP_FROM_UN_ID D_ID,
    CP_DATETIME D_DATETIME)
AS
begin
  for
    select
      cp_id,
      cp_parenet_id,
      cp_income_prefix_ru,
      cp_income_prefix_kg,
      cp_name_ru,
      cp_name_kg,
      cp_type,
      cp_is_delete,
      cp_from_un_id,
      cp_datetime
    from cfp_positions
    where cp_id = coalesce(:id, cp_id)
  into
    :cp_id,
    :cp_parenet_id,
    :cp_income_prefix_ru,
    :cp_income_prefix_kg,
    :cp_name_ru,
    :cp_name_kg,
    :cp_type,
    :cp_is_delete,
    :cp_from_un_id,
    :cp_datetime
  do
  begin
    html_txt = (select html_txt from hp_cfp_positions_html(:cp_id));
    suspend;
  end
end
