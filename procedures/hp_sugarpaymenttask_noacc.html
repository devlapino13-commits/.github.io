CREATE PROCEDURE HP_SUGARPAYMENTTASK_NOACC(
    K_ID D_ID,
    IS_SUGAR BOOLEAN = true)
AS
declare variable spt_id d_integer;
declare variable reg_id d_integer;
declare variable reg_go_id d_integer;
declare variable acc d_str_20;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable line_no d_integer;
declare variable curr_code d_str_lang;
declare variable parent_ol_id d_id;
declare variable ol_id d_id;
declare variable c_id d_id;
declare variable seller_id d_id;
declare variable un_id d_id;
declare variable amount d_currency;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable c_name_fio d_str_100;
begin
  reg_go_id = 1090100;
  acc = '21199/199';
  curr_code = '417';

  for
    -- сахзавод
    select
      spt_id,
      spt_reg_id,
      spt_c_id,
      spt_seller_id,
      spt_created_un_id,
      spt_amount,
      k_kontract_no,
      k_disb_date,
      k_fact_disb_ol_id,
      c_name_fio
    from kontract
      inner join sugarpaymenttask
        on spt_k_id = k_id
      inner join clients
        on c_id = spt_c_id
    where :is_sugar and k_id = :k_id
    union
    -- товары в рассрочку
    select
      ol_doc_no,
      k_last_reg_id,
      c_id,
      cs_from_c_id,
      ol_user_id,
      mp_amount * (1 - cs_discount / 100),
      k_kontract_no,
      k_disb_date,
      k_fact_disb_ol_id,
      c_name_fio
    from kontract
      inner join mc_purchase
        on mp_from_ma_id = k_from_ma_app_id
      inner join clientshops
        on cs_id = mp_from_cs_id
          and cs_from_rd_id = f_get_rd_id('ClientShops.outlets')
          and cs_is_delete = 0
      inner join knclient
        on kc_from_k_id = k_id
      inner join clients
        on c_id = kc_from_c_id
      inner join operlist
        on ol_id = k_fact_disb_ol_id
    where not :is_sugar and k_id = :k_id
  into
    :spt_id,
    :reg_id,
    :c_id,
    :seller_id,
    :un_id,
    :amount,
    :k_kontract_no,
    :k_disb_date,
    :parent_ol_id,
    :c_name_fio
  do
  begin
    acc_dt = (
      select al_id from acc_list
      where al_from_reg_id = :reg_id
        and al_currency = :curr_code
        and al_account = :acc);
    acc_kt = (
      select al_id from acc_list
      where al_from_reg_id = :reg_go_id
        and al_currency = :curr_code
        and al_account = :acc);
    for
      select
        res_reg_id,
        res_acc_dt,
        res_acc_kt,
        res_line_no
      from aarep_loronostro_link(:reg_id, :acc_dt, :acc_kt)
    into
      :reg_id,
      :acc_dt,
      :acc_kt,
      :line_no
    do
    begin
      insert into operlist (
        ol_parent_id,
        ol_from_reg_id,
        ol_oper_type,
        ol_oper_subtype,
        ol_date,
        ol_doc_no,
        ol_doc_kind,
        ol_memo,
        ol_document_client,
        ol_user_id
      )
      values (
        :parent_ol_id,
        :reg_id,
        2,
        11,
        current_date,
        :spt_id,
        9,
        f_format('К Оплате согл.поручения № %1 от %2, %3',
          :k_kontract_no, f_datetostr(:k_disb_date), :c_name_fio),
        :seller_id,
        :un_id)
      returning ol_id
      into :ol_id;

      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_summa,
        ob_summa_val,
        ob_client_dt,
        ob_client_kt)
      values (
        :ol_id,
        :curr_code,
        1,
        :acc_dt,
        :acc_kt,
        :amount,
        :amount,
        iif(:reg_id = :reg_go_id, null, :seller_id),
        iif(:reg_id = :reg_go_id, :seller_id, null));

      if (line_no = 1) then
        parent_ol_id = ol_id;
    end
  end
end
