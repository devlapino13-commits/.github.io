CREATE PROCEDURE HP_GET_INVENT_NO_DATA(
    INV_NO VARCHAR(50))
RETURNS (
    REP_ML_NAME VARCHAR(100),
    REP_PERSONAL_FIO VARCHAR(100),
    REP_OFFICE VARCHAR(100),
    REP_BRANCH VARCHAR(100),
    REP_JOBSTATUS VARCHAR(100),
    REP_PHONE VARCHAR(255))
AS
declare variable ob_bill_id decimal(18,0);
declare variable ob_bill_line_no integer;
declare variable om_matotvet_id decimal(18,0);
declare variable kolich integer;
declare variable c_phone varchar(20);
begin
  ob_bill_id = null;
  select ob_bill_id, ob_bill_line_no, ml_name
  from operbill, matlist
  where ob_invent_str = :inv_no and
        ob_is_delete = 0 and
        ob_date_begin <= current_date and
        ob_date_end >= current_date and
        ob_from_ml_id = ml_id rows 1
  into :ob_bill_id, :ob_bill_line_no, :rep_ml_name;
  if (ob_bill_id is not null) then
  begin
    om_matotvet_id = null;
    rep_personal_fio = 'Владелец не определен';

    select om_matotvet_id, sum(om_kolich_plus - om_kolich_minus)
    from operlist, opermat
    where om_from_ol_id = ol_id and
          operlist.is_delete = 0 and
          om_bill_id = :ob_bill_id and
          om_bill_line_no = :ob_bill_line_no
    group by om_matotvet_id
    having sum(om_kolich_plus - om_kolich_minus) > 0
    order by om_matotvet_id
    rows 1
    into :om_matotvet_id, :kolich;
    if (om_matotvet_id is not null) then
    begin
      select c_name_fio,
             (select cont_value
              from hp_contdata_get_cell_valid(0, :om_matotvet_id))
      from clients
      where c_id = :om_matotvet_id
      into :rep_personal_fio, :c_phone;
      if ((c_phone is not null) and
          (c_phone <> '')) then
      begin
        rep_phone = :c_phone;
      end
      else
      begin
        select cal_cell_no
        from cellaktlist
        where cal_from_c_id = :om_matotvet_id and
              cal_date_begin <= current_date and
              cal_date_end >= current_date and
              cal_is_contact = 1 rows 1
        into :c_phone;
        if (c_phone is not null) then
        begin
          rep_phone = :c_phone;
        end
        else
          rep_phone = 'Тел.№ не определен';
      end
      select r1.reg_name, r2.reg_name, jl_name
      from region r1, region r2, hr_pstatus, joblist
      where ps_from_c_id = :om_matotvet_id and
            ps_date_begin <= current_date and
            ps_date_end >= current_date and
            ps_from_reg_id = r1.reg_unicode and
            r1.parent = r2.reg_unicode and
            ps_from_jl_id = jl_id rows 1
      into :rep_office, :rep_branch, :rep_jobstatus;
    end
  end
  else
    rep_ml_name = 'Не определен';
  suspend;
end
