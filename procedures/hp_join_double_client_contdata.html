CREATE PROCEDURE HP_JOIN_DOUBLE_CLIENT_CONTDATA(
    C_STAY D_ID,
    C_REMOVE D_ID)
AS
declare variable cda_id_remove d_id;
declare variable cda_id_stay d_id;
declare variable cda_cont_kind d_integer;
declare variable cda_from_clt_id d_integer;
declare variable cda_date_remove d_date;
declare variable cda_is_main d_smallint;
declare variable cda_is_delete_remove d_smallint;
declare variable cda_is_delete_stay d_smallint;
declare variable cda_status_remove d_smallint;
declare variable cda_date_stay d_date;
declare variable un_id d_id;
declare variable cda_status_stay d_smallint;
declare variable un_name d_str_code;
begin
  -- Процедура переносит контактные данные удаляемого клиента на нового клиента

  un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN');

  select un_id
  from username
  where un_name = :un_name
  into un_id;

  -- Предварительный цикл. Поиск и удаление старых (по дате обновления) контактных данных.
  -- Сравниваем все неудаленные, одинаковые по типу контакты, но отличающиеся значениями
  for select cda_id, cda_cont_kind, cda_from_clt_id, cda_date, cda_is_main, cda_status
  from contdata
  where cda_from_c_id = :c_remove and
        cda_is_delete = 0
  into cda_id_remove, cda_cont_kind, cda_from_clt_id, cda_date_remove, cda_is_main, cda_status_remove
  do
  begin
    cda_id_stay = null;
    for select cda_id, cda_date, cda_status
    from contdata
    where cda_from_c_id = :c_stay and
          cda_cont_kind = :cda_cont_kind and
          cda_is_main = :cda_is_main and
          cda_from_clt_id <> :cda_from_clt_id and
          cda_is_delete = 0
    into cda_id_stay, cda_date_stay, cda_status_stay
    do
    begin
      -- Устанавливаем признак удаления более старым контактам с совпадающим или неподтвержденным статусом
      if (cda_id_stay is not null) then
      begin
        if (cda_date_stay >= cda_date_remove) then
        begin
          if (cda_status_stay >= cda_status_remove) then
            execute procedure hp_contdata_delete(:cda_id_remove, 'joined', null, :un_id);
        end else
        begin
          if (cda_status_remove >= cda_status_stay) then
            execute procedure hp_contdata_delete(:cda_id_stay, 'joined', null, :un_id);
        end
      end
    end
  end

  -- Основной цикл. Обрабатываются совпадающие по типу и значению контактные данные
  -- Если нет совпадений, заменяем ссылку на клиента, через cda_from_c_id
  -- Если совпадения есть, объединяем логи, переносим ссылки на cda_id
  for select cda_id, cda_cont_kind, cda_from_clt_id, cda_date, cda_is_main,
             cda_is_delete, cda_status, cda_from_un_id
  from contdata
  where cda_from_c_id = :c_remove
  into cda_id_remove, cda_cont_kind, cda_from_clt_id, cda_date_remove, cda_is_main,
        cda_is_delete_remove, cda_status_remove, un_id
  do
  begin
    cda_id_stay = null;
    select cda_id, cda_is_delete, cda_status, cda_date
    from contdata
    where cda_from_c_id = :c_stay and
          cda_cont_kind = :cda_cont_kind and
          cda_from_clt_id = :cda_from_clt_id and
          cda_is_main = :cda_is_main
    into cda_id_stay, cda_is_delete_stay, cda_status_stay, cda_date_stay;

    if (cda_id_stay is null) then
    begin
      update contdata
      set cda_from_c_id = :c_stay
      where cda_id = :cda_id_remove;
    end else
    begin
      update contlog set clg_from_cda_id = :cda_id_stay
      where clg_from_cda_id = :cda_id_remove;

      if (cda_is_delete_stay = 1 and
          cda_is_delete_remove = 0) then
      begin
        -- Восстанавливаем контактные данные у "оставляемого" клиента
        -- Статус верификации восстановим позже
        delete from contdelete
        where cdl_from_cda_id = :cda_id_stay;
      end

      -- Копируем/переносим ссылки с "удаляемых" на "оставляемые" контактные данные
      insert into contveri (cvr_from_cda_id, cvr_hash_pin, cvr_status, cvr_sign_kind, cvr_from_un_id, cvr_datetime)
      select :cda_id_stay, cvr_hash_pin, cvr_status, cvr_sign_kind, cvr_from_un_id, cvr_datetime
      from contveri
      where cvr_from_cda_id = :cda_id_remove
      order by cvr_id;

      update edoc_msg
      set edm_from_cda_id = :cda_id_stay
      where edm_from_cda_id = :cda_id_remove;

      update contdata
      set cda_date = maxvalue(:cda_date_stay, :cda_date_remove)
      where cda_id = :cda_id_stay;

      -- Если на оставляемые контактные данные есть ссылка в contdelete
      -- Тогда ссылки на удаляемые контактные данные чистим
      -- Иначе перепривязываем на оставляемые контактные данные
      -- Update'ы ниже требуются, для исключения случайного восстановления
      -- удаленных контактных данных, если в contdelete.base_cda_id будет
      -- ссылка на удаляемую contdata
      if (exists(
        select count(*) from contdelete
        where cdl_base_cda_id = :cda_id_stay))
      then
        update contdelete
        set cdl_base_cda_id = null
        where cdl_base_cda_id = :cda_id_remove;
      else
        update contdelete
        set cdl_base_cda_id = :cda_id_stay
        where cdl_base_cda_id = :cda_id_remove;

      delete from contdata
      where cda_id = :cda_id_remove;

      -- Восстанавливаем статус верификации, т.к. старые контактные данные были подтверждены
      if (cda_status_remove = 1 and
          cda_is_delete_remove = 0 and
          cda_status_stay = 0) then
        execute procedure hp_contveri_set_ok(:cda_id_stay, :un_id, minvalue(:cda_date_stay, :cda_date_remove));
    end
  end
end
