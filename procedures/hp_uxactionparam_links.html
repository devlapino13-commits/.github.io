CREATE PROCEDURE HP_UXACTIONPARAM_LINKS(
    UA_CODE D_STR_100)
RETURNS (
    ID D_STR_100,
    TEXT D_STR_255)
AS
declare variable fr_code d_str_code;
declare variable ua_ux_code d_str_code;
declare variable ufp_is_param d_integer;
declare variable ufd_source d_str_code;
declare variable source_count d_integer;
declare variable ufp_name d_str_code;
begin
  id = null;
  text = '';
  suspend;

  if (coalesce(ua_code, '') = '') then
    exit;

  select ua_ux_code
  from hp_get_uxalias(:ua_code)
  into :ua_ux_code;

  select count(*) from uxcodeframe
  where ucf_ux_code = :ua_ux_code
  into source_count;

  if (source_count > 0) then
  begin
    for select ucf_ufr_code, ufd_source
    from uxcodeframe, uxfrdata
    where ucf_ux_code = :ua_ux_code and
          ucf_ufr_code = ufd_ufr_code and
          ufd_kind = 0
    into fr_code, ufd_source
    do
    begin
      for select ufp_code, ufp_origin_name, ufp_name, ufp_is_param
      from uxfrparam
      where ufp_ufr_code = :fr_code and
            ufp_is_delete = 0 and
            ufp_in_use = 1
      order by ufp_is_param desc, ufp_index
      into id, text, ufp_name, ufp_is_param
      do
      begin
        if (text <> 'ufp_name') then
          text = text || '   - ' || ufp_name;
  
        if (ufp_is_param = 1) then
          text = text || ', параметр';
  
        if (source_count > 1) then
          text = text || ' (' || ufd_source || ')';

        suspend;
      end
    end
  end else
  begin
    -- Это Frame, действия открытия поисковика

    for select ufp_code, ufp_origin_name, ufp_name, ufd_source, ufp_is_param
    from uxfrparam, uxfrdata
    where ufp_ufr_code = :ua_ux_code and
          ufp_is_delete = 0 and
          ufp_in_use = 1 and
          ufp_ufr_code = ufd_ufr_code and
          ufd_kind = 0
    order by ufp_is_param desc, ufp_index
    into id, text, ufp_name, ufd_source, ufp_is_param
    do
    begin
      if (text <> ufp_name) then
        text = text || '   - ' || ufp_name;

      if (ufp_is_param = 1) then
        text = text || ', параметр';

      if (source_count > 1) then
        text = text || ' (' || ufd_source || ')';

      suspend;
    end
  end
end
