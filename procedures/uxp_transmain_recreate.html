CREATE PROCEDURE UXP$TRANSMAIN_RECREATE
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable text d_str_long;
declare variable res_count d_integer;
begin
  res_count = 0;
  res_is_error = 0;

  for select distinct ufi_name
  from ux$fritem
  where coalesce(ufi_name, '') <> ''
  union
  select ufi_hint
  from ux$fritem
  where coalesce(ufi_hint, '') <> ''
  union
  select ugi_caption
  from ux$gritem
  where coalesce(ugi_caption, '') <> ''
  union
  select ugi_hint
  from ux$gritem
  where coalesce(ugi_hint, '') <> ''
  union
  select uf_name
  from ux$field
  where coalesce(uf_name, '') <> ''
  union
  select ufl_name
  from ux$filter
  where coalesce(ufl_name, '') <> ''
  union
  select ufl_memo
  from ux$filter
  where coalesce(ufl_memo, '') <> ''
  union
  select uma_message
  from ux$frmask
  where coalesce(uma_message, '') <> ''
  union
  select ugr_name
  from ux$grid
  where coalesce(ugr_name, '') <> ''
  union
  select ugr_memo
  from ux$grid
  where coalesce(ugr_memo, '') <> ''
  union
  select usm_caption
  from ux$submenu
  union
  select ml_caption
  from menulist
  where coalesce(ml_caption, '') <> ''
  union
  select ml_description
  from menulist
  where coalesce(ml_description, '') <> ''
  union
  select ua_name
  from ux$action
  where coalesce(ua_name, '') <> ''
  union
  select ua_hint
  from ux$action
  where coalesce(ua_hint, '') <> ''
  union
  select ufr_name
  from ux$frame
  where coalesce(ufr_name, '') <> ''
  union
  select ux_name
  from ux$code
  where coalesce(ux_name, '') <> ''
  union
  select uxa_name
  from ux$alias
  where coalesce(uxa_name, '') <> ''
  union
  select ek_name
  from enum_kind
  union
  select el_name
  from enum_list
  union
  select ucl_text
  from ux$combolist
  where coalesce(ucl_text, '') <> ''
  into text
  do
  begin
    if (trim(coalesce(text, '')) = '') then
      continue;

    if (char_length(text) > 255) then
      continue;

    if (not uxf$is_rus(text)) then
      continue;

    if (not exists(select * from ux$transmain
                   where trm_orig_text = :text)) then
    begin
      res_count = res_count + 1;
      insert into ux$transmain (trm_orig_text)
      values (:text);
    end
  end

  if (res_count > 0) then
    res_text = 'Найдены и добавлены новые тексты: ' || res_count;
  else
    res_text = 'Нет новых текстов для перевода';

  suspend;
end
