CREATE PROCEDURE HP_EXTINFOLIST_ADD_BLS(
    LINE_ID D_INTEGER,
    EIL_ID D_ID,
    EIL_FROM_C_ID D_ID,
    EIL_FROM_EIC_ID D_ID,
    EIL_DATE D_DATE,
    EIL_OBJECT_KIND D_INTEGER,
    EIL_OBJECT_ID D_ID,
    EIL_EXT_ID D_STR_CODE,
    EIL_IS_JSON D_BOOL,
    EIL_FROM_BLS_ID D_ID,
    EIL_STATUS D_SMALLINT,
    UN_ID D_ID)
AS
begin
  if (coalesce(eil_id, 0) <> 0) then
  begin
    update extinfolist
    set eil_from_bls_id = :eil_from_bls_id
    where eil_id = :eil_id;
  end
  else
    insert into extinfolist (
        eil_id,
        eil_from_c_id,
        eil_from_eic_id,
        eil_date,
        eil_object_kind,
        eil_object_id,
        eil_ext_id,
        eil_is_json,
        eil_from_bls_id,
        eil_status,
        eil_from_un_id)
    values (
        :eil_id,
        :eil_from_c_id,
        :eil_from_eic_id,
        :eil_date,
        :eil_object_kind,
        :eil_object_id,
        :eil_ext_id,
        :eil_is_json,
        :eil_from_bls_id,
        :eil_status,
        :un_id);
end
