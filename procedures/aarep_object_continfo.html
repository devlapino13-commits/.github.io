CREATE PROCEDURE AAREP_OBJECT_CONTINFO(
    REP_LIST_KIND VARCHAR(100),
    REP_OBJECT_ID DECIMAL(18,0),
    REP_IS_SECURED INTEGER,
    REP_FILTER VARCHAR(100),
    UN_ID DECIMAL(18,0),
    IS_MOBILE D_BOOL = 0)
RETURNS (
    RD_ID INTEGER,
    RD_CODE VARCHAR(255),
    RD_STR_ID VARCHAR(255),
    RD_NAME VARCHAR(255),
    RD_MEMO VARCHAR(255),
    RES_OBJECT_KIND INTEGER,
    RES_OBJECT_ID DECIMAL(18,0),
    RES_OBJECT_NAME VARCHAR(100),
    RES_VALUE_STATUS VARCHAR(100),
    RES_VALUE VARCHAR(100),
    RES_VALUE_ORIG VARCHAR(100),
    RES_VALUE_SHOW VARCHAR(100),
    RES_EDITMASK VARCHAR(255),
    RES_DATETIME TIMESTAMP,
    RES_STATUS_ID INTEGER,
    RES_EXAMPLE VARCHAR(255),
    RES_COMBO_MASK VARCHAR(255),
    RD_TAG INTEGER,
    RES_VERIFY_STATUS INTEGER,
    RES_IS_CRITICAL INTEGER,
    RES_NEED_VERIFY INTEGER,
    RES_JUST_EDITED INTEGER,
    RES_CLIENT_SEX INTEGER,
    RES_IS_EMPTY INTEGER,
    RES_FIELD_1 VARCHAR(255),
    RES_FIELD_2 VARCHAR(255),
    RES_FIELD_3 VARCHAR(255))
AS
declare variable obj_kind integer;
declare variable obj_id decimal(18,0);
declare variable need_suspend integer;
declare variable new_cont d_str_code;
declare variable cont_kind d_str_code;
declare variable cont_type d_str_code;
declare variable orig_name d_str_100;
declare variable is_main d_smallint;
declare variable new_datetime d_datetime;
declare variable db_id d_integer;
declare variable rd_mcell_id d_integer;
declare variable access_mcell boolean;
begin
  -- Процедура выдает список контактных данных и их состояние
  -- входящие параметры могут быть: Kontract, Clients, Contacts

  -- REP_LIST_KIND:
  -- 0 или Clients  - Физ.лицо
  -- 9 или Kontract - Клиенты в договоре
  -- Staff          - Сотрудники
  -- и т.д. подробное описание в AAREP_OBJECT_CONTINFO_TMP

  --  rep_filter может быть
  -- FULL         полный список контактных данных
  -- VERIFY       требующиеся верификации
  -- ONLY_FILLED  только заполненные
  db_id = f_get_db_id();
  rd_mcell_id = f_get_rd_id('ContactInfo.Cell_Main');
  access_mcell = exists(
    select *
    from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.ChangeMainCell')
    where has_access = 1);

  if (rep_filter is null) then
    rep_filter = 'full';
  if (rep_filter = '') then
    rep_filter = 'full';
  if (rep_is_secured is null) then
    rep_is_secured = 0;

  if (rep_filter in ('verify', 'staff_critical', 'client_critical')) then
    rep_is_secured = 0;

  for select res_kind, res_id, res_field_1, res_field_2, res_field_3
      from aarep_object_continfo_tmp(:rep_list_kind, :rep_object_id)
      into :obj_kind, :obj_id, res_field_1, res_field_2, res_field_3
  do
  begin
    -- rd_tag здесь цифра означает срок устарения верификации в днях
    for select rd_id
        from refdata
        where rd_is_active = 1 and
              rd_main_code = 'ContactInfo' and
              rd_code <> rd_main_code
        order by rd_orderno, rd_id
        into rd_id
    do
    begin
      if (db_id in (39801, 41701) and rd_id = rd_mcell_id and is_mobile = 0 and not access_mcell) then
        continue;

      for select rd_code, rd_str_id, rd_name, rd_memo, res_object_kind, res_object_id, res_object_name,
                 res_value_status, res_value, res_value_orig, res_value_show, res_editmask, res_datetime, res_status_id,
                 res_example, res_combo_mask, rd_tag, res_verify_status, res_is_critical, res_need_verify,
                 res_just_edited, res_client_sex, res_is_empty
          from aarep_object_one_continfo(:obj_kind, :obj_id, :rd_id, :rep_is_secured, :un_id)
          into :rd_code, :rd_str_id, :rd_name, :rd_memo, :res_object_kind, :res_object_id, :res_object_name,
               :res_value_status, :res_value, :res_value_orig, :res_value_show, :res_editmask, :res_datetime,
               :res_status_id, :res_example, :res_combo_mask, :rd_tag, :res_verify_status, :res_is_critical,
               :res_need_verify, res_just_edited, res_client_sex, res_is_empty
      do
      begin
        need_suspend = 0;
        if (rep_filter = 'full') then
          need_suspend = 1;

        if (rep_filter = 'only_filled') then
        begin
          if (res_value_orig <> '' or
              res_is_critical = 1) then
            need_suspend = 1;
        end
        if (rep_filter = 'verify') then
        begin
          if (rd_code = 'contactinfo.cell_main' or
              rd_code = 'contactinfo.cell_add' or
              rd_code = 'contactinfo.email') then
            need_suspend = 1;
        end
        if (rep_filter = 'staff_critical' and
            res_verify_status <> 2) then
        begin
          if (rd_code = 'contactinfo.cell_main' or
              rd_code = 'contactinfo.email') then
            need_suspend = 1;
        end
        if (rep_filter = 'client_critical' and
            res_verify_status <> 2) then
        begin
          if (rd_code = 'contactinfo.cell_main') then
            need_suspend = 1;
        end

        if (need_suspend = 1) then
        begin
          if (res_value <> '' and res_need_verify = 1 and res_verify_status <> 2 and res_object_kind = 0) then
          begin
            is_main = -1;
            cont_type = 'any';
            cont_kind = replace(rd_code, 'contactinfo.', '');
            if (cont_kind = 'cell_main') then
            begin
              cont_kind = 'cell';
              cont_type = 'main';
              is_main = 1;
            end else
            if (cont_kind = 'cell_add') then
            begin
              cont_kind = 'cell';
              cont_type = 'add';
              is_main = 0;
            end

            new_cont = '';
            new_datetime = null;

            select f_cont_code(cda_from_clt_id), clg_datetime
            from contdata, contlog
            where cda_from_c_id = :res_object_id and
                  cda_cont_kind = f_enum('cont_kind', :cont_kind, 'id') and
                  (:is_main = -1 or
                  cda_is_main = :is_main) and
                  cda_id = clg_from_cda_id and
                  cda_is_delete = 0 and
                  cda_status = 1
            order by clg_id desc
            rows 1
            into :new_cont, :new_datetime;

            if (new_cont <> '') then
            begin
              orig_name = rd_name;
              rd_name = 'Новый: ' || rd_name;
            end
          end

          suspend;

          if (res_value <> '' and new_cont <> '' and res_need_verify = 1 and res_verify_status <> 2 and res_object_kind = 0) then
          begin
            rd_name = orig_name;
            new_cont = f_cont(res_object_id, cont_kind, 'ok', cont_type);
            res_value = new_cont;
            res_value_orig = new_cont;
            res_value_show = new_cont;
            res_value_status = 'Подтвержден';
            res_verify_status = 2;
            res_status_id = 2;
            res_just_edited = 0;
            res_datetime = new_datetime;

            if (rep_is_secured = 1) then
              res_value_show = f_cont_format(res_value, cont_kind, 'secured');

            suspend;
          end
        end
      end
    end
  end
end
