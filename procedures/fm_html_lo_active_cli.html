CREATE PROCEDURE FM_HTML_LO_ACTIVE_CLI(
    UN_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable C_ID D_ID;
declare variable CLI_FIO D_STR_100;
declare variable PR_NAME D_STR_100;
declare variable CLI_BALANCE D_CURRENCY;
declare variable PLAN_CLOSE_DATE D_DATE;
declare variable PR_BALANCE D_CURRENCY;
declare variable PREVIOUS_C_ID D_ID;
declare variable DB_ID integer;
declare variable LO_ID decimal(18,0);
declare variable LO_FIO varchar(100);
declare variable LINE_STR varchar(1024);
begin
  -- Список клиентов по продуктам

  db_id = f_get_db_id();

  res_html = (select res_style from fm_html_styles) || '<body>';

  select un_from_c_id, c_name_fio
  from clients, username
  where un_id = :un_id and
        un_from_c_id = c_id
  into :lo_id, :lo_fio;
  if (lo_fio is null) then
    lo_fio = '';

  line_str = f_format('<h1 class="green_text">%1</h1>', iif(db_id <> 94401, 'Активные клиенты (кр.продукт, баланс)', 'Активные клиенты (кр.продукт, баланс)'));
  res_html = BLOB_APPEND(res_html, line_str);

  res_html = BLOB_APPEND(res_html, '<br>');
  line_str = iif(db_id <> 94401, 'Дата', 'Tarix') || ': ' || iif(rep_date is not null, f_datetostr(rep_date), iif(db_id <> 94401, 'Не указана', 'Qeyd olunmayıb'));
  line_str = '<font color="black">' || :line_str || '</font>';
  res_html = BLOB_APPEND(res_html, line_str || '<br>');
  res_html = BLOB_APPEND(res_html, lo_fio || '<br><br>');
  if (lo_fio = '') then
    exit;

  res_html = BLOB_APPEND(res_html, '<table width="100%"><col width="50%"><col width="25%"><col width="25%"><tr>' ||
    f_format('<th>%1</th>', iif(db_id <> 94401, 'Кр.продукт', 'Кр.продукт')) ||
    f_format('<th>%1</th>', iif(db_id <> 94401, 'Актив. баланс', 'Актив. баланс')) ||
    f_format('<th>%1</th>', iif(db_id <> 94401, 'Дата окончение', 'Дата окончение')) || ' </th></tr>');

  previous_c_id = 0;
  for
  select c_id, cli_fio, pr_name, cli_balance, plan_close_date, pr_balance
  from aarep_cli_active_product(:lo_id, :rep_date)
  order by cli_fio, pr_name
  into c_id, cli_fio, pr_name, cli_balance, plan_close_date, pr_balance
  do
  begin
    if (previous_c_id <> c_id) then
    begin
      previous_c_id = c_id;
      res_html = BLOB_APPEND(res_html,f_html_table_row(f_format('<b>%1</b>',
      f_uri_open(cli_fio, 'client', 'Clients', c_id, cli_fio, '', un_id)||' (Балл-'||f_point(:c_id, current_date)||')'
      ), f_format('<b>%1</b>', round(cli_balance,0)), '', '', -1, 1, 1, 1,3));
    end 
    res_html = BLOB_APPEND(res_html,f_html_table_row(:pr_name, round(pr_balance,0), f_datetostr(:plan_close_date), '', -1, 1, 1, 1,3));

  end
  res_html = BLOB_APPEND(res_html,'</table></body>');

  suspend;
end
