CREATE PROCEDURE AAREP_MINIAPP_TEMP_ADMIN(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_INCLUDE_FIRST_DAY INTEGER)
RETURNS (
    MA_DATE_OPER DATE,
    MA_SUMMA_PLUS_OS DECIMAL(18,4),
    MA_SUMMA_PLUS_PERC DECIMAL(18,4),
    MA_SUMMA_PLUS_ADMIN DECIMAL(18,4),
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_ADMIN_PLUS DECIMAL(18,4),
    MA_ADMIN_MINUS DECIMAL(18,4),
    MA_TRANSFER_PLUS DECIMAL(18,4),
    MA_TRANSFER_MINUS DECIMAL(18,4),
    MA_SUMMA_MINUS_REPEAT DECIMAL(18,4),
    MA_SUMMA_MINUS_NEW DECIMAL(18,4),
    MA_SUMMA_MINIAPPNEW DECIMAL(18,4),
    MA_ADMIN_MINUS_PROCESS DECIMAL(18,4),
    MA_REGION_ID INTEGER)
AS
declare variable ma_is_mini_app integer;
declare variable rm_status integer;
declare variable rm_id decimal(18,0);
declare variable rs date;
begin
  rs = cast(rep_date_begin as date);
  ma_summa_minus_repeat = 0;
  ma_summa_minus_new = 0;
  ma_summa_plus_os = 0;
  ma_summa_plus_perc = 0;
  ma_summa_plus_admin = 0;
  ma_transfer_plus = 0;
  ma_transfer_minus = 0;
  ma_admin_plus = 0;
  ma_admin_minus = 0;
  ma_summa_miniappnew = 0;
  ma_admin_minus_process = 0;
  ma_summa_minus = 0;

  for select id_list
      from region_orig_list(:rep_region_id, 1, 3, 1)
      order by id_list
      into :ma_region_id
  do
  begin
    ma_date_oper = (:rep_date_begin + :rep_include_first_day);
    while (ma_date_oper <= :rep_date_end) do
    begin
      suspend;
      ma_date_oper = ma_date_oper + 1;
    end
  end

  /* Админ расходы, трансферы */
  for select rm_from_reg_id, rm_id, rl_date, rl_summa_plus, rl_summa_minus, rm_from_rk_id, rm_status
      from reqlist, reqmain, region_list(:rep_region_id, 1, -1, 1)
      where rl_from_rm_id = rm_id and
            rm_date between(:rep_date_begin + :rep_include_first_day) and '31.12.9999' and
            rl_date between :rep_date_begin and :rep_date_end and
            rm_from_reg_id = id_list
      into :ma_region_id, :rm_id, :ma_date_oper, :ma_admin_plus, :ma_admin_minus, :ma_is_mini_app, :rm_status
  do
  begin
    rm_status = -2;
    select rs_status
    from reqsign
    where rs_from_rm_id = :rm_id and
          rs_datetime < (:rep_date_begin + :rep_include_first_day)
    order by rs_from_rm_id desc, rs_number desc
    rows 1
    into :rm_status;
    if (rm_status is null) then
      rm_status = -2;
    if ((rm_status = 0) or (rm_status = 1) or (rm_status = 4) or (rm_status = 5)) then
    begin
      ma_admin_minus_process = 0;
      ma_transfer_plus = 0;
      ma_transfer_minus = 0;
      if (ma_is_mini_app = 0) then
      begin
        ma_transfer_plus = :ma_admin_plus;
        ma_transfer_minus = :ma_admin_minus;
        ma_admin_plus = 0;
        ma_admin_minus = 0;
      end
      if ((rm_status = 0) or (rm_status = 4) or (rm_status = 5)) then
      begin
        ma_admin_minus_process = ma_admin_minus;
        ma_admin_minus = 0;
      end
      if (ma_date_oper < :rep_date_begin) then
        ma_date_oper = :rep_date_begin;
      suspend;
    end
  end
  ma_admin_minus_process = 0;

  /* Корреспондирующая часть трансферов */
  for select rl_corr_office, rm_id, rl_date, rl_summa_plus, rl_summa_minus, rm_from_rk_id
      from reqlist, reqmain, region_list(:rep_region_id, 1, -1, 1)
      where rl_from_rm_id = rm_id and
            rm_date between(:rep_date_begin + :rep_include_first_day) and '31.12.9999' and
            rl_date between :rep_date_begin and :rep_date_end and
            rl_corr_office = id_list and
            rm_from_rk_id = 0
      into :ma_region_id, :rm_id, :ma_date_oper, :ma_admin_minus, :ma_admin_plus, :ma_is_mini_app
  do
  begin
    rm_status = -2;
    select distinct rs_status
    from reqsign
    where rs_from_rm_id = :rm_id and
          rs_datetime < (:rep_date_begin + :rep_include_first_day)
    order by rs_from_rm_id desc, rs_number desc
    rows 1
    into :rm_status;
    if (rm_status is null) then
      rm_status = -2;
    if ((rm_status = 0) or (rm_status = 1) or (rm_status = 4) or (rm_status = 5)) then
    begin
      ma_transfer_plus = 0;
      ma_transfer_minus = 0;
      if (ma_is_mini_app = 0) then
      begin
        ma_transfer_plus = :ma_admin_plus;
        ma_transfer_minus = :ma_admin_minus;
        ma_admin_plus = 0;
        ma_admin_minus = 0;
      end
      if (ma_date_oper < :rep_date_begin) then
        ma_date_oper = :rep_date_begin;
      suspend;
    end
  end
end
