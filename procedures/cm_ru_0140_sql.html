CREATE PROCEDURE CM_RU_0140_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kf_disb_date d_date;
declare variable MA_FROM_KF_ID ID_TYPE;
declare variable REG_ID integer;
declare variable PR_CODE STR_100;
begin
  /*
    MRAKHMANOV
    |доступ к «Шанс» для ОП 28-03 "Ак Барс" (Казань)
Если есть возможность установить блок на другие продукты, прошу установить данный блок, что бы они кроме продукта
Шанс не смогли выдавать по другим продуктам.
*/
  for select K_FROM_KF_ID, K_DISB_DATE, PR_CODE
      from KONTRACT, PRODUCT
      where K_ID = :K_ID and
            K_FROM_PR_ID = PR_ID
      into MA_FROM_KF_ID, KF_DISB_DATE, PR_CODE
  do
  begin
    --  Вычисляем офис папки, на дату выдачи
    select KAL_NEW_REG_ID
    from KNAKTLIST
    where KAL_FROM_KF_ID = :MA_FROM_KF_ID and
          KNAKTLIST.IS_DELETE = 0 and
          :KF_DISB_DATE between KAL_DATE and KAL_DATE_END
    into REG_ID;
    if (REG_ID = 1280300 and
        PR_CODE not containing '.SHANS.') then
    begin
      res_error_text = 'Выбранный продукт не активен в вашем офисе, выберите другой продукт.';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
  end
end
