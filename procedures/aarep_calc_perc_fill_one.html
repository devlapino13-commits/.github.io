CREATE PROCEDURE AAREP_CALC_PERC_FILL_ONE(
    K_ID D_ID,
    FIRST_PERIOD D_INTEGER,
    LAST_PERIOD D_INTEGER)
AS
declare variable db_id integer;
declare variable k_kredline_end date;
declare variable k_zero_balance_date date;
declare variable rep_date_begin date;
declare variable rep_date_end date;
declare variable reg_id integer;
declare variable rep_period integer;
declare variable rep_period_prior integer;
declare variable aarep_kommis_begin decimal(18,4) = 0;
declare variable aarep_kommis_end decimal(18,4) = 0;
declare variable aarep_kommis_period decimal(18,4) = 0;
declare variable aarep_kommis_payed decimal(18,4) = 0;
declare variable reg_id_prior integer;
declare variable perc_repay decimal(18,4) = 0;
declare variable perc_writeoff decimal(18,4) = 0;
declare variable kommis_repay decimal(18,4) = 0;
declare variable tmp_period integer;
declare variable sa_account varchar(20);
declare variable delta integer;
declare variable pr_id integer;
declare variable pr_id_prior integer;
declare variable aarep_calc_method integer;
declare variable aarep_k_id decimal(18,0);
declare variable aarep_perc_payed decimal(18,4);
declare variable aarep_perc_writeoff decimal(18,4);
declare variable aarep_perc_end decimal(18,4);
declare variable aarep_percent_per_day decimal(18,7);
declare variable k_stop_perc date;
declare variable aarep_out_perc_end decimal(18,4) = 0;
declare variable aarep_out_perc_begin decimal(18,4) = 0;
declare variable aarep_out_perc_calced decimal(18,4) = 0;
declare variable loan_kc_list d_str_255;
declare variable loan_amount_list d_str_255;
declare variable loan_kommis_list d_str_255;
declare variable kcp_from_c_id id_type;
declare variable report_year integer;
declare variable report_month integer;
begin
  db_id = f_get_db_id();
  pr_id_prior = 0;

  if (first_period > last_period) then
    exit;

  -- очистка временной таблицы
  delete from temp_kontcalcperc
  where 1 = 1;

  report_year = left(first_period, 4);
  report_month = mod(first_period, 100);

  delete from kontcalcperc
  where kcp_from_k_id = :k_id and
        kcp_period between :first_period and :last_period;

  if (report_month > 1) then
    rep_period_prior = :report_year * 100 + :report_month - 1;
  else
    rep_period_prior = (:report_year - 1) * 100 + 12;

  rep_period = first_period;

  while (rep_period <= last_period) do
  begin
    if (report_month < 10) then
      rep_date_begin = '01.0' || :report_month || '.' || :report_year;
    else
      rep_date_begin = '01.' || :report_month || '.' || :report_year;
    rep_date_end = dateadd(month, 1, :rep_date_begin) - 1;
  
    if (rep_date_end > current_date) then
      rep_date_end = current_date;
  
    -- заполнение остатка на начало периода
    insert into temp_kontcalcperc (kcp_from_k_id, kcp_from_reg_id, kcp_perc_saldo_begin, kcp_perc_saldo_end,
                                   kcp_out_perc_saldo_begin, kcp_out_perc_saldo_end, kcp_kommis_saldo_begin,
                                   kcp_kommis_saldo_end, kcp_last_reg_id, kcp_delayed_perc)
    select kcp.kcp_from_k_id, kal_new_reg_id, kcp.kcp_perc_saldo, kcp.kcp_perc_saldo, kcp.kcp_out_perc_saldo,
           kcp.kcp_out_perc_saldo, kcp.kcp_kommis_saldo, kcp.kcp_kommis_saldo, kcp.kcp_from_reg_id, kcp_delayed_perc_end
    from kontcalcperc kcp, knaktlist, kontract
    where kcp.kcp_period = :rep_period_prior and
          kcp.kcp_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          kal_date <= :rep_date_end and
          kal_date_end >= :rep_date_end and
          k_id = :k_id;
  
    if (exists(select *
        from kontstate
        where ks_date_begin between :rep_date_begin and :rep_date_end and
              ks_ol_is_delete = 0 and
              ks_state = 2 and
              ks_from_k_id = :k_id)) then
    begin
      update temp_kontcalcperc
      set kcp_perc_saldo_end = 0,
          kcp_out_perc_saldo_end = 0,
          kcp_kommis_saldo_end = 0
      where kcp_from_k_id = :k_id and
            (kcp_perc_saldo_end <> 0 or
            (kcp_out_perc_saldo_end <> 0) or
            kcp_kommis_saldo_end <> 0);
    end
  
    -- список кредитов по которым нужно начислить проценты
    for select k_id, kal_new_reg_id, k_percent_per_day, k_stop_perc, k_kredline_end, k_zero_balance_date, k_from_pr_id
        from kontract, knaktlist
        where k_id = :k_id and
              knaktlist.is_delete = 0 and
              kal_from_kf_id = k_from_kf_id and
              kal_date <= :rep_date_end and
              kal_date_end >= :rep_date_end
        into :aarep_k_id, :reg_id, :aarep_percent_per_day, :k_stop_perc, :k_kredline_end, :k_zero_balance_date, :pr_id
    do
    begin
      -- амортизация доп.процента
      select res_amount_begin, res_amount_end, res_amount_period, res_kommis_payed
      from aarep_calc_amort_kommis_temp(:aarep_k_id, :rep_date_begin, :rep_date_end, :db_id)
      into :aarep_kommis_begin, :aarep_kommis_end, :aarep_kommis_period, :aarep_kommis_payed;
  
      if (pr_id <> pr_id_prior) then
      begin
        select pr_interest_method
        from product
        where pr_id = :pr_id
        into :aarep_calc_method;
      end

      -- если предыдущий месяц заполнен, дополняем данные на конец периода
      if ((report_year * 100 + report_month) <= 201303) then
      begin
        select result_payed, result_writeoff, result_perc_end, 0
        from aarep_calc_perc_one_201302(:aarep_k_id, :rep_date_begin, :rep_date_end, :aarep_percent_per_day, :aarep_calc_method, 0, 2, :k_stop_perc)
        into :aarep_perc_payed, :aarep_perc_writeoff, :aarep_perc_end, :aarep_out_perc_end;
      end else
      begin
        if (f_new_perc_active(rep_date_end) = 1) then
        begin
          select result_payed, result_writeoff, result_perc_end, result_out_perc_end
          from aarep_calc_perc_one_saldo(:aarep_k_id, :rep_date_begin, :rep_date_end)
          into :aarep_perc_payed, :aarep_perc_writeoff, :aarep_perc_end, :aarep_out_perc_end;
        end else
        begin
          select result_payed, result_writeoff, result_perc_end, result_out_perc_end
          from aarep_calc_perc_one(:aarep_k_id, :rep_date_begin, :rep_date_end, 2, :db_id, :k_kredline_end, :k_stop_perc, :k_zero_balance_date)
          into :aarep_perc_payed, :aarep_perc_writeoff, :aarep_perc_end, :aarep_out_perc_end;
        end
      end

      update or insert into temp_kontcalcperc (kcp_from_k_id, kcp_from_reg_id, kcp_perc_payed, kcp_perc_writeoff,
                                               kcp_perc_saldo_end, kcp_out_perc_saldo_end, kcp_kommis_payed,
                                               kcp_kommis_saldo_end)
      values (:aarep_k_id, :reg_id, :aarep_perc_payed, :aarep_perc_writeoff, :aarep_perc_end, :aarep_out_perc_end,
              :aarep_kommis_payed, :aarep_kommis_end)
      matching (kcp_from_k_id);
    end
  
    -- временная таблица заполнена, переносим в боевую таблицу.
    insert into kontcalcperc (kcp_period, kcp_from_k_id, kcp_from_c_id, kcp_from_reg_id, kcp_perc_calced,
                              kcp_perc_saldo, kcp_perc_payed, kcp_perc_writeoff, kcp_perc_begin_saldo,
                              kcp_transfer_reg_id, kcp_kommis_calced, kcp_kommis_saldo, kcp_kommis_payed,
                              kcp_kommis_begin_saldo, kcp_out_perc_calced, kcp_out_perc_saldo, kcp_out_perc_begin_saldo,
                              kcp_delayed_perc_begin)
    select :rep_period, kcp_from_k_id, 0, kcp_from_reg_id, kcp_perc_calc, kcp_perc_saldo_end, kcp_perc_payed,
           kcp_perc_writeoff, kcp_perc_saldo_begin, kcp_last_reg_id, kcp_kommis_calc, kcp_kommis_saldo_end,
           kcp_kommis_payed, kcp_kommis_saldo_begin, kcp_out_perc_calc, kcp_out_perc_saldo_end,
           kcp_out_perc_saldo_begin, kcp_delayed_perc
    from temp_kontcalcperc;

    delete from temp_kontcalcperc
    where 1 = 1;
  
    for select kcp_period, kcp_from_k_id, kcp_from_reg_id, kcp_transfer_reg_id, kcp_from_c_id
        from kontcalcperc
        where kcp_from_reg_id <> kcp_transfer_reg_id and
              kcp_period = :rep_period and
              kcp_from_k_id = :k_id
        into :tmp_period, :aarep_k_id, reg_id, reg_id_prior, kcp_from_c_id
    do
    begin
      perc_repay = 0;
      perc_writeoff = 0;
      kommis_repay = 0;
      select sum(iif(ok_type = 4, -ok_perc, 0)), sum(iif(ok_type in (5, 6), -ok_perc, 0)),
             sum(iif(ok_type = 4, -ok_kommis, 0))
      from operkred
      where ok_from_k_id = :aarep_k_id and
            ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_ol_reg_id = :reg_id_prior and
            ok_from_c_id = 0
      into :perc_repay, :perc_writeoff, kommis_repay;
      if (perc_repay is null) then
        perc_repay = 0;
      if (perc_writeoff is null) then
        perc_writeoff = 0;
      if (kommis_repay is null) then
        kommis_repay = 0;
      -- проценты и комиссия, выплаченная или списанная в предыдущем офисе определены,
      -- теперь нужно закинуть в kontcalc_perc
      if ((perc_repay <> 0) or
          (perc_writeoff <> 0) or
          (kommis_repay <> 0)) then
      begin
        update kontcalcperc
        set kcp_perc_transfer_payed = :perc_repay,
            kcp_perc_transfer_writeoff = :perc_writeoff,
            kcp_kommis_transfer_payed = :kommis_repay
        where kcp_period = :tmp_period and
              kcp_from_k_id = :aarep_k_id and
              kcp_from_c_id = :kcp_from_c_id;
      end
    end
  
    -- закрепление бух.счета по кредитам
    for select kcp_from_k_id, sa_account
        from kontcalcperc, kontract, standacc
        where kcp_from_k_id = k_id and
              k_from_pr_id = sa_from_pr_id and
              kcp_from_reg_id = sa_from_reg_id and
              sa_from_f_id = k_from_f_id and
              sa_account_type = 2 and
              kcp_period = :rep_period and
              kcp_from_k_id = :k_id
        into :aarep_k_id, sa_account
    do
    begin
      update kontcalcperc
      set kcp_perc_account = :sa_account
      where kcp_from_k_id = :aarep_k_id and
            kcp_period = :rep_period;
    end
  
    for select kcp_from_k_id, sa_account
        from kontcalcperc, kontract, standacc
        where kcp_from_k_id = k_id and
              k_from_pr_id = sa_from_pr_id and
              kcp_from_reg_id = sa_from_reg_id and
              sa_from_f_id = k_from_f_id and
              sa_account_type = 6 and
              kcp_period = :rep_period and
              kcp_from_k_id = :k_id
        into :aarep_k_id, sa_account
    do
    begin
      update kontcalcperc
      set kcp_kommis_account = :sa_account
      where kcp_from_k_id = :aarep_k_id and
            kcp_period = :rep_period;
    end
  
    for select kcp_from_k_id
        from kontcalcperc
        where kcp_period = :rep_period and
              kcp_from_k_id = :k_id and
              kcp_perc_begin_saldo = 0 and
              kcp_perc_calced = 0 and
              kcp_perc_payed = 0 and
              kcp_perc_writeoff = 0 and
              kcp_perc_saldo = 0
  
              and
              kcp_out_perc_begin_saldo = 0 and
              kcp_out_perc_calced = 0 and
              kcp_out_perc_saldo = 0
  
              and
              kcp_perc_transfer_payed = 0 and
              kcp_perc_transfer_writeoff = 0 and
              kcp_kommis_begin_saldo = 0 and
              kcp_kommis_calced = 0 and
              kcp_kommis_payed = 0 and
              kcp_kommis_writeoff = 0 and
              kcp_kommis_saldo = 0 and
              kcp_kommis_transfer_payed = 0 and
              kcp_kommis_transfer_writeoff = 0
        into :aarep_k_id
    do
    begin
      if (exists(select *
                 from kontstate
                 where ks_from_k_id = :aarep_k_id and
                       ks_ol_is_delete = 0 and
                       ks_state = 2 and
                       ks_date_begin < :rep_date_begin)) then
      begin
        delete from kontcalcperc
        where kcp_period = :rep_period and
              kcp_from_k_id = :aarep_k_id;
      end
    end
  
    for select kcp_period, kcp_from_k_id,
               kcp_period - (extract(year from ks_date_begin) * 100 + extract(month from ks_date_begin))
        from kontcalcperc, kontstate
        where kcp_from_k_id = ks_from_k_id and
              (extract(year from ks_date_begin) * 100 + extract(month from ks_date_begin)) < kcp_period and
              ks_state = 2 and
              ks_ol_is_delete = 0 and
              kcp_loan_is_closed = 0 and
              kcp_period = :rep_period and
              kcp_from_k_id = :k_id
        into :tmp_period, :aarep_k_id, :delta
    do
    begin
      update kontcalcperc
      set kcp_loan_is_closed = :delta
      where kcp_period = :tmp_period and
            kcp_from_k_id = :aarep_k_id;
    end
  
    for select kcp_from_k_id
        from kontcalcperc
        where kcp_period = :rep_period and
              kcp_from_k_id = :k_id
        into :aarep_k_id
    do
    begin
      -- Деление на участников
      loan_kc_list = '';
      loan_amount_list = '';
      loan_kommis_list = '';
  
      select list(kc_id), list(kc_amount_approved),
             list(iif(kc_amount_approved > 0, coalesce(kc_amount_approved * k_kommis_perc, 0) - coalesce(kc_kommis_lgota, 0), 0))
      from knclient, kontract
      where kc_from_k_id = :aarep_k_id and
            kc_amount_approved > 0 and
            k_id = kc_from_k_id
      into loan_kc_list, loan_amount_list, loan_kommis_list;
  
      insert into kn_kontcalcperc (kcp_period, kcp_from_kc_id, kcp_from_k_id, kcp_from_reg_id, kcp_perc_saldo,
                                   kcp_transfer_reg_id, kcp_perc_calced, kcp_perc_payed, kcp_perc_writeoff,
                                   kcp_perc_begin_saldo, kcp_perc_transfer_payed, kcp_perc_transfer_writeoff,
                                   kcp_kommis_calced, kcp_kommis_payed, kcp_kommis_writeoff, kcp_kommis_saldo,
                                   kcp_kommis_begin_saldo, kcp_kommis_transfer_payed, kcp_kommis_transfer_writeoff,
                                   kcp_perc_account, kcp_kommis_account, kcp_loan_is_closed, kcp_out_perc_begin_saldo,
                                   kcp_out_perc_calced, kcp_out_perc_saldo)
      select :rep_period, kc_id, :aarep_k_id, cl_kcp_from_reg_id, cl_kcp_perc_saldo, cl_kcp_transfer_reg_id,
             cl_kcp_perc_calced, cl_kcp_perc_payed, cl_kcp_perc_writeoff, cl_kcp_perc_begin_saldo,
             cl_kcp_perc_transfer_payed, cl_kcp_perc_transfer_writeoff, cl_kcp_kommis_calced, cl_kcp_kommis_payed,
             cl_kcp_kommis_writeoff, cl_kcp_kommis_saldo, cl_kcp_kommis_begin_saldo, cl_kcp_kommis_transfer_payed,
             cl_kcp_kommis_transfer_writeoff, cl_kcp_perc_account, cl_kcp_kommis_account, cl_kcp_loan_is_closed,
             cl_kcp_out_perc_begin_saldo, cl_kcp_out_perc_calced, cl_kcp_out_perc_saldo
      from hp_devide_kn_kontcalcperc(:aarep_k_id, :rep_period, :loan_kc_list, :loan_amount_list, :loan_kommis_list);
    end

    if (report_month < 12)
    then report_month = report_month + 1;
    else
    begin
      report_year = report_year + 1;
      report_month = 1;
    end
    rep_period_prior = rep_period;
    rep_period = (:report_year * 100 + :report_month);
  end
end
