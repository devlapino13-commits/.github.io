CREATE PROCEDURE HP_EDOC_FREELANCER_CONTRACT(
    SD_ID D_INTEGER,
    UN_ID D_ID,
    SIGNER_C_ID D_ID,
    EDC_CODE D_STR_CODE = '' /* RFCKG-3484 */)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable edoc_obj d_str_code;
declare variable category d_str_255;
begin
  res_text = '';

  -- после изменеий Сергея удалить кусок кода, т.к. он будет бесполезным
  select sd.sd_category
  from servicedog sd
  where sd.sd_id = :sd_id
  into :category;

  if (:category containing 'стажер') then
    edoc_obj = 'probationer_kontract';
  else if (:category containing 'кочегар') then
    edoc_obj = 'kochegar_kontract'; -- RFCKG-3233
  else
    edoc_obj = 'freelancer_kontract';

  if (signer_c_id = 0) then
  begin
    res_text = 'Необходимо указать подписанта.';
    suspend;
    exit;
  end

  if (edc_code = '') then
    edc_code = edoc_obj;

--  RFCKG-5435
--  if (f_edoc_status(:edc_code, :sd_id, current_date) = 'signed') then
--  begin
--    res_text = 'Договор на оказание услуг уже подписан';
--    suspend;
--    exit;
--  end
  -----------------------------------------------------------------------

  select sd.sd_from_c_id
  from servicedog sd
  where sd.sd_id = :sd_id
  into :c_id;

  if (not exists(select *
                 from clientdocs cds
                 inner join passkind pk on cds.cds_from_pk_id = pk.pk_id and
                       pk.pk_is_delete = 0
                 inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id and
                       pdt.pdt_is_main_doc = 1
                 where cds.cds_from_c_id = :c_id and
                       current_date between coalesce(cds.cds_doc_date_begin, '01.01.1900') and coalesce(cds.cds_doc_date_end, '31.12.2999') and
                       cds.cds_is_delete = 0)) then
  begin
    res_text = 'Нет действующего паспорта.';
    suspend;
    exit;
  end

  if (not exists(select *
                 from hr_pstatus ps
                 where ps.ps_from_c_id = :c_id and
                       ps.ps_is_working = 1 and
                       current_date between ps.ps_date_begin and ps.ps_date_end)) then
  begin
    if (f_asp_signed(:c_id, current_date) <> 1) then
      res_text = 'Не определен актуальный прикрепленный договор ЭЦП.';
  end

  if (res_text = '') then
    -- RFCKG-3484
    select edl_id, res_text
    from hp_edoc_insert(:edc_code, f_objkind(:edc_code, 'id'), :sd_id, :un_id)
    into :edl_id, :res_text;

  if (:edl_id is null) then
    edl_id = 0;

  if (res_text = '' and
      :edl_id > 0) then
    update edoc_list
    set edl_manager_c_id = :signer_c_id
    where edl_id = :edl_id;

  suspend;
end
