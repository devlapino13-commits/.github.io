Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_UUIDLIST_ACTION(
    ACTION_KIND D_STR_CODE NOT NULL,
    OLD_OBJ_KIND D_STR_CODE,
    OLD_OBJ_ID D_ID,
    NEW_OBJ_KIND D_STR_CODE,
    NEW_OBJ_ID D_ID)
AS
begin
  -- Общая процедура для редактирования таблицы UUIDLIST

  if (action_kind not in ('delete', 'insert', 'update')) then
    exception e_default 'Unknown action_kind: ' || action_kind;

  if (old_obj_kind is not null) then
    old_obj_kind = f_objkind(:old_obj_kind, 'id');

  if (new_obj_kind is not null) then
    new_obj_kind = f_objkind(:new_obj_kind, 'id');

  if (action_kind = 'insert') then
  begin
    if (new_obj_id is null) then
      exception e_default 'new_obj_id is null on inserting';

    insert into uuidlist (udl_object_kind, udl_object_id)
    values (:new_obj_kind, :new_obj_id);
  end
  if (action_kind = 'update') then
  begin
    if (old_obj_id is null) then
      exception e_default 'old_obj_id is null on updating';

    if (new_obj_id is null) then
      exception e_default 'new_obj_id is null on updating';

    update uuidlist
    set udl_object_id = :new_obj_id,
        udl_object_kind = :new_obj_kind
    where udl_object_kind = :old_obj_kind and
          udl_object_id = :old_obj_id;
  end
  if (action_kind = 'delete') then
  begin
    if (old_obj_id is null) then
      exception e_default 'old_obj_id is null on deleting';

    delete from uuidlist
    where udl_object_kind = :old_obj_kind and
          udl_object_id = :old_obj_id;
  end
end
