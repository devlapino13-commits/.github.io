CREATE PROCEDURE HP_STRAIDER_BANKDETAIL_SET(
    C_ID D_ID,
    B_ID D_STR_20,
    ACC_NO D_STR_CODE)
AS
begin
  if (b_id is null or b_id not similar to '[0-9]{9}') then
    exception e_message 'БИК должен содержать 9-ти значный цифровой код.';
  if (acc_no is null or acc_no not similar to '[0-9]{20}') then
    exception e_message 'Номер счета должен содержать 20-ти значный цифровой код.';

  update bankclient set
    bc_is_main = 0,
    bc_is_active = 0
  where bc_from_c_id = :c_id;

  insert into bankclient(
    bc_from_c_id,
    bc_from_b_id,
    bc_account_no,
    bc_is_main,
    bc_is_active)
  values(
    :c_id,
    :b_id,
    :acc_no,
    1,
    1);
end
