CREATE PROCEDURE HP_MENULIST_DELETE(
    ML_ID D_INTEGER)
AS
declare variable ml_kind d_integer;
begin
  select ml_kind
  from menulist
  where ml_id = :ml_id
  into ml_kind;

  if (ml_kind = 0) then
  begin
    if (exists(select * from ux$menulist
    where uml_ml_id = :ml_id)) then
    begin
      if (exists(select * from menulist
      where ml_parent_id = :ml_id)) then
      begin
        update menulist
        set ml_kind = 1
        where ml_id = :ml_id;
      end else
      begin
        update menulist
        set ml_kind = 2
        where ml_id = :ml_id;
      end
    end else
    begin
      delete from menulist
      where ml_id = :ml_id;
    end
  end else
    exception e_message ('Пункты меню Ux-модуля удаляются через конфигуратор');
end
