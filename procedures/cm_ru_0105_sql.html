CREATE PROCEDURE CM_RU_0105_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare variable k_kolich_repay integer;
declare variable kc_from_c_id id_type;
declare variable k_kredline_end date;
declare variable k_disb_date date;
declare variable c_citizen integer;
declare variable rd_code varchar(255);
declare variable cds_doc_date_end date;
declare variable pdt_code varchar(100);
declare variable c_curr_status integer;
begin
  k_kolich_repay = 0;
  select k_kolich_repay,
         k_kredline_end,
         k_disb_date
  from kontract
  where k_id = :k_id
  into k_kolich_repay,
       k_kredline_end,
       k_disb_date;
  for
  select kc_from_c_id
  from knclient
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  into kc_from_c_id
  do
  begin
    select c_curr_status from hp_get_client_status(:kc_from_c_id,iif(:k_disb_date > current_date, current_date, :k_disb_date))
    into c_curr_status;
    select c_name_fio, c_citizen
    from clients
    where c_id = :kc_from_c_id
    into res_info_text, c_citizen;
    select rd_code
--         , rd_name
    from refdata
    where rd_id = :c_citizen
    into rd_code
--       , rd_name
       ;

    pdt_code = null;
    select pdt_code, cds_doc_date_end
    from CLIENTDOCS, PASSKIND, PASSDOCTYPE
    where CDS_FROM_C_ID = :kc_from_c_id
    and cds_from_pk_id = pk_id
    and pk_from_pdt_id = pdt_id
    and cds_is_delete = 0
    and pdt_code in ('passport_local', 'passport_card', 'passport_trans')
    and cds_doc_date_begin <= :k_disb_date
    and coalesce(cds_doc_blob_1, 0) > 0
    order by cds_doc_date_end desc
    rows 1
    into pdt_code, cds_doc_date_end;
    cds_doc_date_end = coalesce(cds_doc_date_end,'01.01.1990');
    if (pdt_code is null
        or (pdt_code <> 'passport_local' and cds_doc_date_end < k_disb_date)
        ) then
    begin
      res_error_text = 'Паспорт не действительный, проверьте срок паспорта';
      res_error_is_critical = 1;
      suspend;
      exit;
    end 
--    k_kolich_repay = 8;
--    cds_doc_date_end = '22.02.2017';
                   /*
--    3.1.1.    срок действия паспорта иностранного гражданина,
--    который обратился за микрозаймом на срок 3 - 5 месяцев, должен превышать срок микрозайма не менее чем на 15 календарных дней; 
    if (rd_code in ('Citizen.KG', 'Citizen.UZ', 'Citizen.TJ') and c_curr_status < 4) then
    begin
      if (k_kolich_repay between 3 and 5) then
      begin
        if (not(k_kredline_end + 15 <= cds_doc_date_end)) then
        begin
          res_error_text = 'Дата окончания срока паспорта не соответствует'; -- к политике(пункт 3.1.1)
          res_error_is_critical = 1;
          suspend;
        end
      end else
      if (not(dateadd(month,6,k_disb_date) <= cds_doc_date_end)) then
      begin
        res_error_text = 'До конца срока действия паспорта должно оставаться не менее 6-ти месяцев';
        res_error_is_critical = 1;
        suspend;
      end
    end     */
  end
end
