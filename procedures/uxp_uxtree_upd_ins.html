Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXTREE_UPD_INS(
    NEW_UTR_UUID TYPE OF D_UUID,
    NEW_UTR_PARENT_UUID TYPE OF D_UUID,
    NEW_UTR_CODE TYPE OF D_STR_CODE,
    NEW_UTR_NAME TYPE OF D_STR_100,
    NEW_UTR_MEMO D_STR_LONG,
    NEW_UTR_POSITION TYPE OF D_INTEGER)
RETURNS (
    UTR_UUID D_UUID,
    UTR_PARENT_UUID D_UUID,
    UTR_CODE D_STR_CODE,
    UTR_NAME D_STR_100,
    UTR_MEMO D_STR_LONG,
    UTR_POSITION D_INTEGER)
AS
begin
  if (new_utr_uuid is null) then
  begin
    insert into ux$tree (utr_uuid, utr_parent_uuid, utr_code, utr_name, utr_position)
    values (:new_utr_uuid, :new_utr_parent_uuid, :new_utr_code, :new_utr_name, :new_utr_position)
    returning utr_uuid, utr_parent_uuid, utr_code, utr_name, utr_position
    into utr_uuid, utr_parent_uuid, utr_code, utr_name, utr_position;
  end else
  begin
    update ux$tree
    set utr_parent_uuid = :new_utr_parent_uuid,
        utr_code = :new_utr_code,
        utr_name = :new_utr_name,
        utr_position = :new_utr_position
    where utr_uuid = :new_utr_uuid
    returning utr_uuid, utr_parent_uuid, utr_code, utr_name, utr_position
    into utr_uuid, utr_parent_uuid, utr_code, utr_name, utr_position;
  end

  utr_memo = trim(coalesce(new_utr_memo, ''));

  update or insert into ux$memo (umo_uuid, umo_memo)
  values (:utr_uuid, :utr_memo)
  matching (umo_uuid);

  suspend;
end
