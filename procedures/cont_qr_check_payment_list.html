CREATE PROCEDURE CONT_QR_CHECK_PAYMENT_LIST
RETURNS (
    KPT_ID D_INTEGER,
    TRANS_ID D_INTEGER,
    AMOUNT DECIMAL(15,2),
    QR_JSON D_STR_8100,
    TRANSACTION_ANSWER_JSON D_BLOB_TEXT)
AS
declare variable esm_id d_id;
begin
  for
    select
      kpt.kpt_id,
      kpt.kpt_amount,
      kpt.kpt_from_esm_id,
      kptq.kpt_qr_json
    from kontpaytasks kpt
    inner join kontpaytasks_qr kptq on kpt.kpt_id = kptq.kptq_from_kpt_id
    where kpt.kpt_project = 'qr'
      and kpt.kpt_status = 1
      and kpt.kpt_error_text = ''
    union
    select
      kpt_id,
      kpt_amount,
      kpt_from_esm_id,
      kqa_qr_json
    from kontpaytasks
      inner join kontpaytasks_qr_add
        on kqa_kpt_id = kpt_id
    where kpt_status = 1
      and kpt_error_text = ''
    into
      :kpt_id,
      :amount,
      :esm_id,
      :qr_json
  do
  begin
    trans_id = :kpt_id;

    transaction_answer_json = (
        select ea.esa_message
        from extsysanswer ea
        where ea.esa_from_esm_id = :esm_id
        order by ea.esa_id desc
        rows 1);

    transaction_answer_json = coalesce(transaction_answer_json, '');

    suspend;
  end
end
