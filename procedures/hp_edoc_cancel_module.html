CREATE PROCEDURE HP_EDOC_CANCEL_MODULE(
    EDC_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_255,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable EDOC_KIND D_STR_CODE;
declare variable OBJ_KIND D_INTEGER;
declare variable OBJ_ID D_ID;
begin
-- RFCRU-3890 Процедура команда выполнить для модуля "Аннулировать электронный документ"
  res_info_text = '';
  res_error_text = '';

  if (coalesce(edc_id, 0) = 0) then
    res_error_text = 'Укажите id документа';

  if (not exists(select *
                 from edoc_list edl
                 where edl.edl_id = :edc_id
                 and edl.edl_status in (0, 1))) then
    res_error_text = 'Документ не найден или уже аннулирован';

  if (res_error_text = '') then
  begin
    select edc.edc_code,
           edl.edl_object_kind,
           edl.edl_object_id
    from edoc_list edl
    inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
    where edl.edl_id = :edc_id
    rows 1
    into edoc_kind,
         obj_kind,
         obj_id;

    insert into edoc_cancel_log (ecl_from_eil_id, ecl_from_un_id)
    values (:edc_id, :un_id);

    execute procedure hp_edoc_cancel(:edoc_kind, :obj_kind, :obj_id);
    res_info_text = 'Документ аннулирован';
  end

  suspend;
end
