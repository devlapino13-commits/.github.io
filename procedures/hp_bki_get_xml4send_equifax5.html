CREATE PROCEDURE HP_BKI_GET_XML4SEND_EQUIFAX5(
    XML_VERSION_ID D_INTEGER,
    OBJ_KIND D_SMALLINT,
    OBJ_ID D_ID,
    NODE D_STR_255,
    REC_NO D_INTEGER,
    PARENT_ID D_ID = null,
    ARR_NUMBER D_INTEGER = null)
RETURNS (
    RES_XML D_STR_8100)
AS
declare variable xml_field d_str_255;
declare variable db_field d_str_100;
declare variable bfv_id d_id;
declare variable field_val d_str_8100;
declare variable field_val_tmp d_str_8100;
declare variable attr d_str_8100;
declare variable is_empty d_boolean;
declare variable forced_value d_str_255;
declare variable jso d_integer;
declare variable bkc_add d_str_long;
declare variable event_date d_date;
begin
  res_xml = '';
  attr = '';
  is_empty = true;

  if (parent_id is null) then
  begin
    select bfv_id, bxf_xml_node, bki_db_field
    from bki_xml_fields_ver, bki_xml_fields, bki_indicators
    where bxf_id = bfv_from_bxf_id
      and bki_code = bxf_from_bki_code
      and bfv_from_bxv_id = :xml_version_id
      and bki_code = :node
    into bfv_id, xml_field, db_field;

    if (db_field is distinct from '<OBJ>') then
      exception e_default 'Wrong node: ' || coalesce(node, '<NULL>');

    res_xml = (select res_xml from hp_bki_get_xml4send_equifax5(:xml_version_id, :obj_kind, :obj_id, :xml_field, :rec_no, :bfv_id));
    suspend;
  end
  else
  begin
    for select bfv_id, bxf_xml_node, bki_db_field, bfv_forced_value
    from bki_xml_fields_ver, bki_xml_fields, bki_indicators
    where bxf_id = bfv_from_bxf_id
      and bki_code = bxf_from_bki_code
      and bfv_from_bxv_id = :xml_version_id
      and bfv_parent_id = :parent_id
    order by bfv_position
    into bfv_id, xml_field, db_field, forced_value
    do
    begin
      if (forced_value starting with '<' and forced_value <> '<NULL>') then
        db_field = forced_value;

      if (db_field = '<OBJ>') then
        field_val = (select res_xml from hp_bki_get_xml4send_equifax5(:xml_version_id, :obj_kind, :obj_id, :xml_field, :rec_no, :bfv_id, :arr_number));
      else if (db_field = '<ARR>') then
      begin
        field_val = '';
  
        for with recursive t1 (id) as(
          select bfv_id
          from bki_xml_fields_ver bfv
          where bfv_parent_id = :bfv_id
          union all
          select bfv_id
          from t1, bki_xml_fields_ver
          where bfv_parent_id = id
        )
        select distinct bsd_number
        from t1, bki_send_data_oper, bki_xml_fields_ver
        where bsd_from_bfv_id = bfv_id
          and bsd_obj_kind = :obj_kind
          and bsd_obj_id = :obj_id
          and id in (bfv_parent_id, bfv_id)
        union
        select distinct bsd_number
        from t1, bki_send_data, bki_xml_fields_ver
        where bsd_from_bfv_id = bfv_id
          and bsd_obj_kind = :obj_kind
          and bsd_obj_id = :obj_id
          and id in (bfv_parent_id, bfv_id)
        order by 1
        into arr_number
        do
          select :field_val || res_xml
          from hp_bki_get_xml4send_equifax5(:xml_version_id, :obj_kind, :obj_id, :xml_field, :rec_no, :bfv_id, :arr_number)
          into field_val;

        arr_number = null;
      end
      else
      begin
        field_val = null;

        if (db_field = 'recnumber') then
          field_val = rec_no;
        else if (db_field = 'recnumber') then
          field_val = rec_no;
        else
          select f_bki_data_text(bsd_from_bkd_id)
          from bki_send_data_oper
          where bsd_from_bfv_id = :bfv_id
            and bsd_obj_kind = :obj_kind
            and bsd_obj_id = :obj_id
            and bsd_number is not distinct from :arr_number
          union
          select f_bki_data_text(bsd_from_bkd_id)
          from bki_send_data
          where bsd_from_bfv_id = :bfv_id
            and bsd_obj_kind = :obj_kind
            and bsd_obj_id = :obj_id
            and bsd_number is not distinct from :arr_number
          into field_val;
  
        if (coalesce(field_val, '') <> '') then
        begin
          field_val = replace(field_val, '<', 'lt;');
          field_val = replace(field_val, '>', 'gt;');

          if (xml_field starting with '@' ) then
          begin
            attr = attr || f_format(' %1="%2"', substring(xml_field from 2), field_val);
            field_val = null;
          end
          else
          begin
            if (xml_field containing '<X>') then
              field_val = f_format('<%1></%1>', replace(xml_field, '<X>', field_val));
            else if (db_field starting with '#arr#') then
            begin
              field_val_tmp = field_val;

              select list(f_format('<%1>%2</%1>', :xml_field, res_str), '')
              from hp_split_str(:field_val_tmp, ',')
              into field_val;
            end
            else
              field_val = f_format('<%1>%2</%1>', xml_field, field_val);
          end
        end
      end

      if (coalesce(field_val, '') <> '') then
      begin
        is_empty = false;
        res_xml = res_xml || field_val;
      end
    end

    if (node = 'FL_Event_3_4') then
    begin
      select bev_event_date, bkc_add
      from bki_events, bki_clients
      where bkc_id = bev_from_bkc_id
        and bev_id = :obj_id
      into event_date, bkc_add;

      jso = iif(coalesce(bkc_add, '') = '', json.obj_create(), json.parse(bkc_add));

      if (xml_version_id < 3) then
        attr = f_format(' operationCode="C.2" orderNum="%1" eventDate="%2" changeReason="%3"', rec_no, event_date, json.obj_get_val(jso, 'change_reason', '1'));
      else
        attr = f_format(' operationCode="C.2" orderNum="%1" eventDateTime="%2T00:00:00+03:00" changeReason="%3"', rec_no, event_date, json.obj_get_val(jso, 'change_reason', '1'));
    end

    if (coalesce(attr, '') <> '') then
      is_empty = false;

    res_xml = iif(is_empty, null, '<' || node || attr || '>' || res_xml || '</' || node || '>');
    suspend;
  end
end
