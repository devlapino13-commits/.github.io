CREATE PROCEDURE CM_KG_0213_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id id_type;
declare variable mac_amount d_integer;
declare variable ma_date_oper d_date;
declare variable sum_kc_amount d_integer;
declare variable kc_amount_approved_v d_integer;
declare variable kc_amount_approved_p d_integer;
declare variable res_str_values d_str_100;
declare variable crel_other_c_id d_id;
declare variable k_disb_date d_date;
declare variable ma_id d_id;
declare variable k_from_ma_app_id d_id;
begin
  sum_kc_amount = 0;
  for select kc.kc_from_c_id, kc.kc_amount_approved, kt.k_disb_date
      from kontstate ks
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
      inner join kontract kt on kt.k_id = ks.ks_from_k_id
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      where kt.k_id = :k_id and
            current_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_state = 0 and
            ks.ks_ol_is_delete = 0 and
            kc.kc_amount_approved > 0 and
            pr.pr_code containing '.IPF.' and
            pr.pr_code not containing '.sugarmill.' and
            pr.pr_code not containing '.installment_goods.'
      into :c_id, :kc_amount_approved_v, :k_disb_date
  do
  begin
    kc_amount_approved_p = 0;

    select kc.kc_amount_approved
    from kontstate ks
    inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
    inner join kontract kt on kt.k_id = ks.ks_from_k_id
    inner join product pr on pr.pr_id = kt.k_from_pr_id
    where kc.kc_from_c_id = :c_id and
          current_date between ks.ks_date_begin and ks.ks_date_end and
          ks.ks_state = 1 and
          ks.ks_ol_is_delete = 0 and
          kc.kc_amount_approved > 0 and
          kt.k_disb_date = :k_disb_date  and
          pr.pr_code containing '.IPF.'
    into :kc_amount_approved_p;

    sum_kc_amount = kc_amount_approved_v + coalesce(kc_amount_approved_p, 0);

    if (sum_kc_amount > 49999) then
    begin
      res_str_values = '';

      select rd.rd_name
      from clients cs
      inner join refdata rd on rd.rd_id = cs.c_family
      where cs.c_id = :c_id
      into res_str_values;

      select crel_other_c_id
      from hp_client_relation_list(:c_id, 0)
      where crel_from_rel_id in ('1', '2') rows 1
      into :crel_other_c_id;

      if ((res_str_values containing 'Женат / замужем' and crel_other_c_id is null) or
          (res_str_values is null and crel_other_c_id is not null) or
          (res_str_values containing 'Женат / замужем' and crel_other_c_id is not null)) then
      begin
        if (not exists(select *
                       from edoc_list el
                       inner join edoc_cfg edc on edc.edc_id = el.edl_from_edc_id
                       inner join edoc_frx edf on edf.edf_from_edc_id = edc.edc_id
                       where el.edl_object_id = :k_id and
                             el.edl_object_kind = 9 and
                             el.edl_status = 1 and
                             edf.edf_from_blt_id = 6)) then
        begin
          res_error_text = 'Для оформления заявки предоставьте пожалуйста согласие супруга';
          res_error_is_critical = 1;
          suspend;
        end
      end
    end
  end
end
