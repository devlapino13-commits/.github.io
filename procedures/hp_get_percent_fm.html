CREATE PROCEDURE HP_GET_PERCENT_FM(
    CURR_PR_ID INTEGER,
    CURR_STAUS INTEGER,
    DISB_DATE DATE,
    LOAN_AMOUNT CURR_TYPE,
    LOAN_REPAY INTEGER)
RETURNS (
    PERC_DEFAULT PERC_TYPE)
AS
declare variable prl_percent_min perc_type;
declare variable prl_percent_max perc_type;
declare variable perc_min perc_type;
declare variable perc_max perc_type;
declare variable percent_lgota perc_type;
declare variable pr_code varchar(100);
declare variable pr_product_kind integer;
begin
  perc_min = null;
  perc_max = null;

  for select coalesce(prl_percent_min, 0), coalesce(prl_percent_max, 100)
      from aarep_product_logic_list(:curr_pr_id, :disb_date, -1, -1, :loan_amount, -1, :loan_repay, -1, -1, -1, -1, -1, -1)
      into :prl_percent_min, prl_percent_max
  do
  begin
    if (perc_min is null) then
      perc_min = prl_percent_min;
    if (perc_max is null) then
      perc_max = prl_percent_max;
    if (perc_min > prl_percent_min) then
      perc_min = prl_percent_min;
    if (perc_max < prl_percent_max) then
      perc_max = prl_percent_max;
  end

  if (perc_min is null) then
  begin
    perc_min = 0.0;
    perc_max = 100.0;
  end

  perc_default = perc_max;

  select pr_code, pr_product_kind
  from product
  where pr_id = :curr_pr_id
  into pr_code, pr_product_kind;

  percent_lgota = 0;
  if (pr_code containing 'barakat'
      or pr_code containing 'family'
      or pr_product_kind = 2
      or pr_code containing 'blago'
      or pr_code containing 'uspeh'
      or pr_code containing '.zalog_car.') then
  begin
    percent_lgota = case curr_staus
                      when 5 then 1.00
                      when 4 then 0.75
                      when 3 then 0.75
                      when 2 then 0.50
                      when 1 then 0.25
                      else 0
                    end;
    if (pr_code containing '.barakat.' or pr_code containing '.blago.') then
    begin
      if (loan_repay <= 6) then
      begin
        perc_max = 8.9;
--        kommis_perc_max = 1.5;
      end
      else
      begin
        perc_max = 9.4;
--        kommis_perc_max = 2.0;
      end
    end
    else
    if (pr_code containing '.barakat plus.' or pr_code containing '.uspeh.') then
    begin
      if (loan_repay <= 6) then
      begin
        perc_max = 8.2;
--        kommis_perc_max = 1.0;
      end
      else
      begin
        perc_max = 8.7;
--        kommis_perc_max = 1.5;
      end
    end
    else
    if (pr_code containing '.family.') then
    begin
      if (loan_repay <= 6) then
      begin
        perc_max = 7.5;
--        kommis_perc_max = 0.0;
      end
      else
      begin
        perc_max = 8.0;
--        kommis_perc_max = 1.0;
      end
    end
    else
    if (pr_code containing '.family plus.') then
    begin
      if (loan_repay <= 6) then
      begin
        perc_max = 8.5;
--        kommis_perc_max = 1.0;
      end
      else
      begin
        perc_max = 9.0;
--        kommis_perc_max = 1.5;
      end
    end
    else
    if (pr_code containing '.zalog_car.') then
    begin
      perc_max = 4.0;
--      kommis_perc_max = 0;

      if (curr_staus >= 2) then percent_lgota = 0.4;
      else
      if (curr_staus = 1) then percent_lgota = 0.25;
      else percent_lgota = 0;
    end
    perc_default = perc_max - percent_lgota;
    if (perc_min < perc_default) then
      perc_min = perc_default;
    else
      perc_default = perc_min;
    if (perc_max > perc_default) then
      perc_max = perc_default;
  end
  suspend;
end
