CREATE PROCEDURE UXP$UI_UXEVENTS(
    UX_UUID D_UUID,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    EVT_SENDER_UUID D_UUID,
    EVT_EVENT_KIND D_INTEGER,
    EVT_TARGET_UUID D_UUID,
    EVT_HANDLER_KIND D_INTEGER,
    EVT_EXPRESSION D_STR_8100,
    EVT_SENDER_FIELD_FRAME D_UUID,
    EVT_SENDER_FIELD_NAME D_STR_CODE,
    EVT_SENDER_FIELD_IS_PARAM D_BOOL,
    EVT_TARGET_FIELD_FRAME D_UUID,
    EVT_TARGET_FIELD_NAME D_STR_CODE,
    EVT_TARGET_FIELD_IS_PARAM D_BOOL,
    EVT_LINK_KIND D_SMALLINT,
    EVT_FIELD_TYPE D_SMALLINT,
    EVT_EXPR_IS_SQL D_BOOL,
    EVT_MESSAGE D_STR_LONG)
AS
declare variable ux_kind d_str_code;
declare variable ufr_uuid d_uuid;
declare variable evt_uuid d_uuid;
declare variable exp_calc_ufr_uuid d_uuid;
declare variable event_code d_str_code;
declare variable need_afterscroll d_bool;
declare variable afterscroll_event_id d_integer;
declare variable onchange_event_id d_integer;
declare variable evt_expr_frame d_uuid;
declare variable uf_uuid d_uuid;
declare variable ufp_uuid d_uuid;
declare variable beforepost_event_id d_integer;
declare variable ufp_not_null d_bool;
declare variable uf_field_name d_str_code;
declare variable uf_name d_str_100;
declare variable afteropen_event_id d_integer;
declare variable can_edit d_bool;
declare variable ufp_parent d_uuid;
declare variable tmp_name d_str_code;
declare variable orig_field_name d_str_code;
declare variable ufr_is_mainframe d_bool;
declare variable evt_handler_kind_tmp d_smallint;
declare variable exp_kind d_smallint;
declare variable ufi_control d_str_code;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  ux_kind = uxf$code(ux_uuid, 'kind');
  evt_message = '';

  select event_id
  from uxp$uxevent_kinds('On')
  where event_code = 'OnAfterScroll'
  into afterscroll_event_id;

  select event_id
  from uxp$uxevent_kinds('On')
  where event_code = 'OnChange'
  into onchange_event_id;

  select event_id
  from uxp$uxevent_kinds('On')
  where event_code = 'OnBeforePost'
  into beforepost_event_id;

  select event_id
  from uxp$uxevent_kinds('On')
  where event_code = 'OnAfterOpen'
  into afteropen_event_id;

  if (ux_kind <> 'uxframe') then
  begin
    for select ufr_uuid, iif(ufr_can_update = 1 or ufr_can_insert = 1, 1, 0), ufr_is_mainframe
    from ux$frame
    where ufr_owner = :ux_uuid
    into ufr_uuid, can_edit, ufr_is_mainframe
    do
    begin
      evt_target_field_frame = ufr_uuid;

      for select event_id, event_code
      from uxp$uxevent_kinds('Do')
      into evt_handler_kind_tmp, event_code
      do
      begin
        evt_handler_kind = evt_handler_kind_tmp;

        evt_sender_field_is_param = 0;
        evt_target_field_is_param = 0;
        evt_expr_frame = ufr_uuid;
        evt_link_kind = 0;
        evt_field_type = 0;
        evt_event_kind = onchange_event_id;

        if (event_code in ('DoChange', 'SetVisible', 'SetEnabled', 'SetHilighted', 'SetColor')) then
        begin
          if (ufr_is_mainframe = 1) then
          begin
            for select ua_uuid
            from ux$action
            where ua_owner = :ux_uuid
            into :evt_sender_uuid
            do
            begin
              evt_handler_kind = evt_handler_kind_tmp;

              evt_target_uuid = evt_sender_uuid;
              evt_sender_field_frame = ufr_uuid;
              evt_sender_field_name = '';
              evt_sender_field_is_param = 0;
              evt_target_field_name = '';
              evt_target_field_is_param = 0;
              evt_link_kind = 0;
              evt_field_type = 0;

              if (event_code <> 'DoChange') then
              begin
                for select exp_ux_uuid, exp_calc_expr, exp_calc_ufr_uuid, exp_is_sql
                from ux$expression
                where exp_ux_uuid = :evt_sender_uuid and
                      exp_kind in (2, 3) and
                      coalesce(exp_calc_expr, '') <> '' and
                      exp_action = :evt_handler_kind
                into evt_target_uuid, evt_expression, evt_target_field_frame, evt_expr_is_sql
                do
                begin
                  if (evt_target_field_frame is null) then
                    evt_target_field_frame = ufr_uuid;
  
                  evt_expression = replace(evt_expression, ':new$', ':');
                  if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
                    evt_expression = 'select ' || evt_expression || ' from MemoryTable';
  
                  if (evt_handler_kind = 2) then -- DoVisible
                  begin
                    if (exists(select * from ux$frbutton, ux$fritem
                               where ufb_ua_uuid = :evt_sender_uuid and
                                     ufb_ufi_uuid = ufi_uuid and
                                     ufi_hidden = 0)) then
                      evt_handler_kind = 3; -- DoEnabled
                  end

                  suspend;
                end
              end else
              begin
                for select exp_ux_uuid, exp_calc_expr, exp_calc_ufr_uuid, exp_is_sql
                from ux$expression
                where exp_ux_uuid = :evt_sender_uuid and
                      exp_kind = 1 and
                      exp_calc_kind >= 0 and
                      coalesce(exp_calc_expr, '') <> '' and
                      exp_action = :evt_handler_kind
                into evt_target_uuid, evt_expression, evt_target_field_frame, evt_expr_is_sql
                do
                begin
                  if (evt_target_field_frame is null) then
                    evt_target_field_frame = ufr_uuid;
  
                  evt_expression = replace(evt_expression, ':new$', ':');
                  if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
                    evt_expression = 'select ' || evt_expression || ' from MemoryTable';
  
                  suspend;
                end
              end
            end
          end

          for select ua_uuid
          from ux$action
          where ua_owner = :ufr_uuid
          into :evt_sender_uuid
          do
          begin
            evt_handler_kind = evt_handler_kind_tmp;
            evt_target_uuid = evt_sender_uuid;
            evt_sender_field_frame = ufr_uuid;
            evt_sender_field_name = '';
            evt_sender_field_is_param = 0;
            evt_target_field_name = '';
            evt_target_field_is_param = 0;
            evt_link_kind = 0;
            evt_field_type = 0;

            if (event_code <> 'DoChange') then
            begin
              for select exp_ux_uuid, exp_calc_expr, exp_calc_ufr_uuid, exp_is_sql
              from ux$expression
              where exp_ux_uuid = :evt_sender_uuid and
                    exp_kind in (2, 3) and
                    coalesce(exp_calc_expr, '') <> '' and
                    exp_action = :evt_handler_kind
              into evt_target_uuid, evt_expression, evt_target_field_frame, evt_expr_is_sql
              do
              begin
                if (evt_target_field_frame is null) then
                  evt_target_field_frame = ufr_uuid;
  
                evt_expression = replace(evt_expression, ':new$', ':');
                if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
                  evt_expression = 'select ' || evt_expression || ' from MemoryTable';
  
                if (evt_handler_kind = 2) then -- DoVisible
                begin
                  if (exists(select * from ux$frbutton, ux$fritem
                             where ufb_ua_uuid = :evt_sender_uuid and
                                   ufb_ufi_uuid = ufi_uuid and
                                   ufi_hidden = 0)) then
                    evt_handler_kind = 3; -- DoEnabled
                end

                suspend;
              end
            end else
            begin
              for select exp_ux_uuid, exp_calc_expr, exp_calc_ufr_uuid, exp_is_sql
              from ux$expression
              where exp_ux_uuid = :evt_sender_uuid and
                    exp_kind = 1 and
                    exp_calc_kind >= 0 and
                    coalesce(exp_calc_expr, '') <> '' and
                    exp_action = :evt_handler_kind
              into evt_target_uuid, evt_expression, evt_target_field_frame, evt_expr_is_sql
              do
              begin
                if (evt_target_field_frame is null) then
                  evt_target_field_frame = ufr_uuid;
  
                evt_expression = replace(evt_expression, ':new$', ':');
                if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
                  evt_expression = 'select ' || evt_expression || ' from MemoryTable';
  
                suspend;
              end
            end
          end
        end

        evt_event_kind = onchange_event_id;
        evt_handler_kind = evt_handler_kind_tmp;

        if (event_code = 'DoChange') then
        begin
          for select ufp_uuid, ufp_parent, uf_uuid, uf_field_name, uf_type, uf_is_param
          from ux$frparam, ux$field
          where ufp_ufr_uuid = :ufr_uuid and
                ufp_uf_uuid = uf_uuid
          order by uf_is_param, uf_position
          into :ufp_uuid, :ufp_parent, :uf_uuid, :evt_target_field_name, :evt_field_type, :evt_target_field_is_param
          do
          begin
            if (can_edit = 0 and evt_target_field_is_param = 0 and evt_field_type = 0) then
              Continue;

            orig_field_name = evt_target_field_name;

            tmp_name = '';
            if (ufp_parent is not null) then
            begin
              select uf_field_name
              from ux$frparam, ux$field
              where ufp_uuid = :ufp_parent and
                    ufp_uf_uuid = uf_uuid
              into tmp_name;

              if (tmp_name <> '') then
                evt_target_field_name = evt_target_field_name || '$' || tmp_name;
            end

            evt_target_uuid = null;

            select exp_ux_uuid, exp_calc_expr, exp_calc_ufr_uuid, exp_is_sql
            from ux$expression
            where exp_ux_uuid = :uf_uuid and
                  exp_kind = 1 and
                  exp_calc_kind >= 0
            into evt_target_uuid, evt_expression, exp_calc_ufr_uuid, evt_expr_is_sql;

            if (row_count = 0) then
            begin
              select exp_ux_uuid, exp_calc_expr, exp_calc_ufr_uuid, exp_is_sql
              from ux$expression
              where exp_ux_uuid = :ufp_uuid and
                    exp_kind = 1 and
                    exp_calc_kind >= 0
              into evt_target_uuid, evt_expression, exp_calc_ufr_uuid, evt_expr_is_sql;
            end

            if (evt_target_uuid is null and ufp_parent is not null) then
            begin
              for select ufp_uuid, uf_field_name, uf_is_param
              from ux$frparam, ux$field
              where ufp_ufr_uuid = :ufr_uuid and
                    ufp_uf_uuid = uf_uuid and
                    ufp_parent is null and
                    uf_field_name = :orig_field_name and
                    uf_is_param = 1
              into :evt_sender_uuid, :evt_sender_field_name, :evt_sender_field_is_param
              do
              begin
                evt_target_uuid = ufp_uuid;
                evt_expression = 'select :' || orig_field_name || ' from MemoryTable';
                exp_calc_ufr_uuid = ufr_uuid;
                evt_expr_is_sql = 0;

                evt_expr_frame = ufr_uuid;
                evt_sender_field_frame = ufr_uuid;
                evt_link_kind = 0;

                suspend;
              end

              if (evt_target_uuid is not null) then
                Continue;
            end

            if (evt_target_uuid is null) then
              Continue;

            evt_sender_field_frame = coalesce(exp_calc_ufr_uuid, ufr_uuid);
            evt_expr_frame = coalesce(exp_calc_ufr_uuid, ufr_uuid);
            evt_link_kind = 0;

            if (evt_expression <> '') then
            begin
              evt_sender_uuid = null;

              evt_expression = replace(evt_expression, ':new$', ':');
              if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
                evt_expression = 'select ' || evt_expression || ' from MemoryTable';

              for select ufp_uuid, uf_field_name, uf_is_param
              from ux$exprfield, ux$expression, ux$frparam, ux$field
              where exf_ux_uuid = :evt_target_uuid and
                    exf_kind = 1 and
                    exp_ux_uuid = exf_ux_uuid and
                    exp_kind = exf_kind and
                    exp_calc_kind in (0, 1) and
                    exf_uf_uuid = uf_uuid and
                    ufp_ufr_uuid = :evt_expr_frame and
                    ufp_uf_uuid = uf_uuid
              into :evt_sender_uuid, :evt_sender_field_name, :evt_sender_field_is_param
              do
                suspend;

              if (evt_sender_uuid is null) then
              begin
                evt_sender_uuid = ufr_uuid;
                evt_sender_field_name = '';
                evt_sender_field_is_param = 0;
                suspend;
              end

              if (exists(select * from ux$exprfield, ux$expression, ux$frparam, ux$field
              where exf_ux_uuid = :evt_target_uuid and
                    exf_kind = 1 and
                    exp_ux_uuid = exf_ux_uuid and
                    exp_kind = exf_kind and
                    exp_calc_kind = 2 and
                    exf_uf_uuid = uf_uuid and
                    ufp_ufr_uuid = :evt_expr_frame and
                    ufp_uf_uuid = uf_uuid)) then
              begin
                evt_sender_uuid = ufr_uuid;
                evt_sender_field_name = '';
                evt_sender_field_is_param = 0;
                suspend;
              end
            end
          end
        end -- конец event_code = 'DoChange'

        evt_message = '';

        if (event_code = 'DoCheck') then
        begin
          evt_event_kind = beforepost_event_id;
          evt_sender_uuid = ufr_uuid;
          evt_target_uuid = ufr_uuid;
          exp_calc_ufr_uuid = null;
          evt_sender_field_frame = ufr_uuid;
          evt_target_field_frame = ufr_uuid;
          evt_expr_frame = ufr_uuid;

          evt_sender_field_name = '';
          evt_sender_field_is_param = 0;
          evt_field_type = 0;
          evt_target_field_name = '';
          evt_target_field_is_param = 0;

          evt_link_kind = 0;

          for select ufp_uuid, uf_uuid, ufp_not_null, uf_field_name, uf_name
          from ux$frparam, ux$field
          where ufp_ufr_uuid = :ufr_uuid and
                ufp_uf_uuid = uf_uuid
          into :ufp_uuid, :uf_uuid, ufp_not_null, uf_field_name, uf_name
          do
          begin
            if (ufp_not_null = 1) then
            begin
              evt_expr_is_sql = 0;
              evt_expression = 'select case when :%1 is not null and char_length(:%1) > 0 then 1 else 0 end from MemoryTable';

              evt_expression = f_format(evt_expression, uf_field_name);
              evt_message = uxf$lang('Поле "%1" не заполнено');
              evt_message = f_format(evt_message, uf_name);
              suspend;
            end

            evt_expression = '';

            for select exp_calc_expr, exp_is_sql, exp_message
            from ux$expression
            where exp_ux_uuid = :uf_uuid and
                  exp_kind = 0 and
                  exp_calc_expr <> ''
            into evt_expression, evt_expr_is_sql, evt_message
            do
            begin
              evt_expression = replace(evt_expression, ':new$', ':');
              if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
                evt_expression = 'select ' || evt_expression || ' from MemoryTable';

              suspend;
            end

            for select exp_calc_expr, exp_is_sql, exp_message
            from ux$expression
            where exp_ux_uuid = :ufp_uuid and
                  exp_kind = 0 and
                  exp_calc_expr <> ''
            into evt_expression, evt_expr_is_sql, evt_message
            do
            begin
              evt_expression = replace(evt_expression, ':new$', ':');
              if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
                evt_expression = 'select ' || evt_expression || ' from MemoryTable';

              suspend;
            end
          end

          for select exp_calc_expr, exp_is_sql, exp_message
          from ux$expression
          where exp_ux_uuid = :ufr_uuid and
                exp_kind = 0 and
                exp_calc_expr <> ''
          into evt_expression, evt_expr_is_sql, evt_message
          do
          begin
            evt_expression = replace(evt_expression, ':new$', ':');
            if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
              evt_expression = 'select ' || evt_expression || ' from MemoryTable';

            suspend;
          end

          if (ux_kind = 'uxcommand') then
          begin
            evt_event_kind = afteropen_event_id;
            evt_sender_uuid = ufr_uuid;
            evt_target_uuid = ufr_uuid;
            exp_calc_ufr_uuid = null;
            evt_sender_field_frame = ufr_uuid;
            evt_target_field_frame = ufr_uuid;
            evt_expr_frame = ufr_uuid;
  
            evt_sender_field_name = '';
            evt_sender_field_is_param = 0;
            evt_field_type = 0;
            evt_target_field_name = '';
            evt_target_field_is_param = 0;
  
            evt_link_kind = 0;

            for select exp_calc_expr, exp_is_sql, exp_message
            from ux$expression
            where exp_ux_uuid = :ux_uuid and
                  exp_kind = 0 and
                  exp_calc_expr <> ''
            into evt_expression, evt_expr_is_sql, evt_message
            do
            begin
              evt_expression = replace(evt_expression, ':new$', ':');
              if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
                evt_expression = 'select ' || evt_expression || ' from MemoryTable';

              suspend;
            end
          end
        end

        evt_message = '';

        if (event_code = 'DoShowMessage' and ux_kind = 'uxcommand') then
        begin
          evt_event_kind = afteropen_event_id;
          evt_sender_uuid = ufr_uuid;
          evt_target_uuid = ufr_uuid;
          exp_calc_ufr_uuid = null;
          evt_sender_field_frame = ufr_uuid;
          evt_target_field_frame = ufr_uuid;
          evt_expr_frame = ufr_uuid;

          evt_sender_field_name = '';
          evt_sender_field_is_param = 0;
          evt_field_type = 0;
          evt_target_field_name = '';
          evt_target_field_is_param = 0;

          evt_link_kind = 0;

          for select cmd_message_field
          from ux$command
          where cmd_uuid = :ux_uuid and
                cmd_message_field <> ''
          into evt_message
          do
          begin
            evt_expression = '';
            select exp_calc_expr, exp_is_sql
            from ux$expression
            where exp_ux_uuid = :ux_uuid and
                  exp_kind = 0 and
                  exp_calc_expr <> ''
            into evt_expression, evt_expr_is_sql;

            if (row_count = 0) then
            begin
              evt_expr_is_sql = 0;
              evt_expression = '1 = 1';
            end

            evt_expression = replace(evt_expression, ':new$', ':');
            if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
              evt_expression = 'select ' || evt_expression || ' from MemoryTable';

            suspend;
          end
        end

        evt_message = '';

        evt_sender_field_name = '';
        evt_sender_field_is_param = 0;

        evt_target_field_is_param = 0;
        evt_field_type = 0;
        for select ufi_uuid, ufi_control, exp_calc_expr, exp_calc_ufr_uuid, exp_is_sql, exp_kind
        from ux$fritem, ux$expression
        where ufi_owner in (:ufr_uuid, :ux_uuid) and
              exp_ux_uuid = ufi_uuid and
              exp_kind in (1, 2, 3) and
              exp_action = :evt_handler_kind
        into :evt_target_uuid, :ufi_control, :evt_expression, :exp_calc_ufr_uuid, evt_expr_is_sql, exp_kind
        do
        begin
          evt_target_field_name = '';

          if (ufi_control = 'expressbtn') then
            evt_target_field_name = 'btn$express';

          evt_expr_frame = coalesce(exp_calc_ufr_uuid, ufr_uuid);
          evt_event_kind = onchange_event_id;
          need_afterscroll = 0;

          evt_sender_field_frame = coalesce(exp_calc_ufr_uuid, ufr_uuid);
          evt_link_kind = 0;

          if (evt_expression <> '') then
          begin
            evt_expression = replace(evt_expression, ':new$', ':');
            if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
              evt_expression = 'select ' || evt_expression || ' from MemoryTable';

            for select ufp_uuid, uf_field_name, uf_is_param
            from ux$exprfield, ux$frparam, ux$field
            where exf_ux_uuid = :evt_target_uuid and
                  exf_kind = :exp_kind and
                  exf_uf_uuid = uf_uuid and
                  ufp_ufr_uuid = :evt_expr_frame and
                  ufp_uf_uuid = uf_uuid
            into :evt_sender_uuid, :evt_sender_field_name, :evt_sender_field_is_param
            do
            begin
              suspend;

              if (event_code in ('SetVisible', 'SetEnabled', 'SetHilighted', 'SetColor')) then
                need_afterscroll = 1;
            end

            if (row_count = 0) then
            begin
              suspend;

              if (event_code in ('SetVisible', 'SetEnabled', 'SetHilighted', 'SetColor')) then
                need_afterscroll = 1;
            end

            if (need_afterscroll = 1) then
            begin
              evt_sender_field_frame = coalesce(exp_calc_ufr_uuid, ufr_uuid);
              evt_event_kind = afterscroll_event_id;
              evt_sender_uuid = evt_expr_frame;
              evt_sender_field_name = '';
              evt_sender_field_is_param = 0;

              suspend;
            end
          end
        end
        evt_message = '';

        evt_target_field_is_param = 0;
        evt_field_type = 0;
        evt_target_field_name = '';
        for select ugi_uuid
        from ux$grid, ux$gritem
        where ugr_ufr_uuid = :ufr_uuid and
              ugr_uuid = ugi_ugr_uuid
        into :evt_target_uuid
        do
        begin
          for select exp_calc_expr, exp_calc_ufr_uuid, exp_is_sql, exp_kind
          from ux$expression
          where exp_ux_uuid = :evt_target_uuid and
                exp_kind in (2, 3) and
                exp_action = :evt_handler_kind
          into :evt_expression, :exp_calc_ufr_uuid, evt_expr_is_sql, exp_kind
          do
          begin
            evt_expr_frame = coalesce(exp_calc_ufr_uuid, ufr_uuid);
            evt_event_kind = onchange_event_id;
            need_afterscroll = 0;
  
            evt_sender_field_frame = coalesce(exp_calc_ufr_uuid, ufr_uuid);
            evt_link_kind = 0;
  
            if (evt_expression <> '') then
            begin
              evt_expression = replace(evt_expression, ':new$', ':');
              if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
                evt_expression = 'select ' || evt_expression || ' from MemoryTable';
  
              for select ufp_uuid, uf_field_name, uf_is_param
              from ux$exprfield, ux$frparam, ux$field
              where exf_ux_uuid = :evt_target_uuid and
                    exf_kind = :exp_kind and
                    exf_uf_uuid = uf_uuid and
                    ufp_ufr_uuid = :evt_expr_frame and
                    ufp_uf_uuid = uf_uuid
              into :evt_sender_uuid, :evt_sender_field_name, :evt_sender_field_is_param
              do
              begin
                suspend;
  
                if (event_code in ('SetVisible', 'SetEnabled', 'SetHilighted', 'SetColor')) then
                  need_afterscroll = 1;
              end
  
              if (need_afterscroll = 1) then
              begin
                evt_event_kind = afterscroll_event_id;
                evt_sender_uuid = evt_expr_frame;
                evt_sender_field_name = '';
                evt_sender_field_is_param = 0;
  
                suspend;
              end
            end
          end
        end
      end
    end
  end

  evt_expr_is_sql = 0;
  evt_field_type = 0;
  for select distinct t.evt_uuid, t.evt_sender_uuid, t.evt_event_kind, t.evt_target_uuid, t.evt_handler_kind
  from ux$event t, uxp$uxevent_list_tmp(:ux_uuid) p
  where t.evt_uuid = p.evt_uuid
  order by t.evt_sender_uuid, t.evt_position
  into :evt_uuid, :evt_sender_uuid, :evt_event_kind, :evt_target_uuid, :evt_handler_kind
  do
  begin
    evt_expr_frame = null;
    evt_expression = '';

    ux_kind = uxf$code(evt_target_uuid, 'kind');
    evt_link_kind = 0;

    if (ux_kind = 'uxevent') then
      evt_link_kind = 1; -- Связанный обработчик

    if (evt_link_kind = 1) then
    begin
      select evt_target_uuid, evt_handler_kind, exp_calc_ufr_uuid, exp_calc_expr, exp_is_sql
      from ux$event, ux$expression
      where evt_uuid = :evt_target_uuid and
            evt_uuid = exp_ux_uuid and
            exp_kind = 1 and
            exp_calc_kind >= 0
      into evt_target_uuid, evt_handler_kind, evt_expr_frame, evt_expression, evt_expr_is_sql;

      ux_kind = uxf$code(evt_target_uuid, 'kind');
    end else
    begin
      select exp_calc_ufr_uuid, exp_calc_expr, exp_is_sql
      from ux$expression
      where exp_ux_uuid = :evt_uuid
      into evt_expr_frame, evt_expression, evt_expr_is_sql;
    end

    evt_expression = replace(evt_expression, ':new$', ':');

    if (evt_expr_is_sql = 0 and evt_expression not starting 'select ') then
      evt_expression = 'select ' || evt_expression || ' from MemoryTable';

    evt_sender_field_frame = null;
    evt_sender_field_name = '';
    evt_sender_field_is_param = 0;

    evt_target_field_frame = null;
    evt_target_field_name = '';
    evt_target_field_is_param = 0;

    if (ux_kind = 'uxframe') then
      evt_target_field_frame = evt_target_uuid;
    else
    if (ux_kind = 'uxfrparam') then
    begin
      select ufp_ufr_uuid, ufp_parent, uf_field_name, uf_is_param
      from ux$frparam, ux$field
      where ufp_uuid = :evt_target_uuid and
            ufp_uf_uuid = uf_uuid
      into evt_target_field_frame, ufp_parent, evt_target_field_name, evt_target_field_is_param;

      tmp_name = '';
      if (ufp_parent is not null) then
      begin
        select uf_field_name
        from ux$frparam, ux$field
        where ufp_uuid = :ufp_parent and
              ufp_uf_uuid = uf_uuid
        into tmp_name;

        if (tmp_name <> '') then
          evt_target_field_name = evt_target_field_name || '$' || tmp_name;
      end
    end else
    if (ux_kind = 'uxitem') then
    begin
      select ufi_ufr_uuid, uf_is_param
      from ux$fritem left join ux$field on (ufi_uf_uuid = uf_uuid)
      where ufi_uuid = :evt_target_uuid
      into evt_target_field_frame, evt_target_field_is_param;
    end

    ux_kind = uxf$code(evt_sender_uuid, 'kind');

    if (ux_kind = 'uxframe') then
      evt_sender_field_frame = evt_sender_uuid;
    else
    if (ux_kind = 'uxfrparam') then
    begin
      select ufp_ufr_uuid, uf_field_name, uf_is_param
      from ux$frparam, ux$field
      where ufp_uuid = :evt_sender_uuid and
            ufp_uf_uuid = uf_uuid
      into evt_sender_field_frame, evt_sender_field_name, evt_sender_field_is_param;
    end else
    if (ux_kind = 'uxitem') then
    begin
      select ufi_ufr_uuid, uf_is_param
      from ux$fritem left join ux$field on (ufi_uf_uuid = uf_uuid)
      where ufi_uuid = :evt_sender_uuid
      into evt_sender_field_frame, evt_sender_field_is_param;
    end

    suspend;
  end
end
