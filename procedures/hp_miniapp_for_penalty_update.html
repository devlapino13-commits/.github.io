CREATE PROCEDURE HP_MINIAPP_FOR_PENALTY_UPDATE(
    OBJECT_KIND STR_100,
    OBJECT_ID ID_TYPE)
AS
declare variable ma_id id_type;
declare variable ma_date_save timestamp;
declare variable ma_from_kf_id id_type;
declare variable ma_state integer;
declare variable ma_date_end date;
declare variable state_datetime timestamp;
declare variable k_id id_type;
declare variable k_from_kf_id id_type;
declare variable k_status integer;
declare variable rd_id integer;
declare variable acl_id id_type;
declare variable db_id integer;
declare variable kf_id id_type;
declare variable lo_id id_type;
declare variable mfp_state smallint;
declare variable pr_code str_100;
begin

  db_id = f_get_db_id();
  if (db_id <> 81001) then
    exit;

-- получаю id заявки
  if (object_kind = 'kontract') then
  begin
    select kont.k_from_ma_app_id, k_from_kf_id, k_status, pr_code
    from kontract kont
      inner join product
        on pr_id = k_from_pr_id
    where kont.k_id = :object_id rows 1
    into :ma_id, :k_from_kf_id, :k_status, :pr_code;

    k_id = :object_id;
  end
  else
  begin
    ma_id = object_id;

    select kont.k_id, kont.k_from_kf_id, kont.k_status, pr_code
    from kontract kont
      inner join product
        on pr_id = k_from_pr_id
    where kont.k_from_ma_app_id = :ma_id
    rows 1
    into :k_id, :k_from_kf_id, :k_status, :pr_code;

    if (pr_code is null) then
    begin
        select pr_code
        from miniapp maa
        inner join product pr on pr.pr_id = maa.ma_from_pr_id
        where maa.ma_date_end = '31.12.9999' and
              maa.ma_id = :ma_id
        into :pr_code;
    end
  end
  --S.K.
  if (pr_code containing '.ZALOG_CAR.') then begin
    update miniapp_for_penalty set mfp_state = -2 where mfp_ma_id = :ma_id;
    exit;
  end

  mfp_state = null;

-- если запись в miniapp_for_penalty отсутствует, добавляю ее с первичным статусом
  select t.mfp_state
  from miniapp_for_penalty t
  where t.mfp_ma_id = :ma_id
  into :mfp_state;

  if (mfp_state is null) then -- заявки еще нет
  begin
    select ma.ma_date_save, ma.ma_from_kf_id, iif(ma.ma_is_active = 1, 0, -1),
           coalesce(ma_date_end, cast('01.01.1900' as date))
    from miniapp ma
    where ma.ma_id = :ma_id
    order by ma.ma_id, ma.ma_number
    rows 1
    into :ma_date_save, :ma_from_kf_id, :ma_state, ma_date_end;

    kf_id = ma_from_kf_id;
    if (k_from_kf_id is not null) then
    begin
      if (k_from_kf_id <> ma_from_kf_id) then
        kf_id = k_from_kf_id;
    end

    select kal.kal_new_kf_officer
    from knaktlist kal
    where is_delete = 0 and
          kal_from_kf_id = :kf_id and
          kal_date_end >= cast('01.01.2999' as date) and
          kal_date < kal_date_end rows 1
    into lo_id;

    insert into miniapp_for_penalty (mfp_ma_id, mfp_date_save, mfp_from_kf_id, mfp_state, mfp_state_datetime,
                                     mfk_from_k_id, mfp_current_lo_id)
    values (:ma_id, :ma_date_save, :kf_id, :ma_state, :ma_date_save, :k_id, :lo_id);
  end
  else
  begin
    if (mfp_state <> 0) then -- если уже присвоен конечный статус
      exit;
  end

  ma_date_end = coalesce(ma_date_end, cast('01.01.1900' as date));

  if (ma_date_end <> cast('31.12.9999' as date)) then
  begin
-- проверяю заявку по последнему ее состоянию, на предмет аннулирования
    select maa.ma_date_save, iif(maa.ma_is_active = 1, 0, -1), maa.ma_from_kf_id
    from miniapp maa
    where maa.ma_id = :ma_id and
          maa.ma_date_end = '31.12.9999' rows 1
    into :state_datetime, ma_state, ma_from_kf_id;

    kf_id = ma_from_kf_id;
    if (k_from_kf_id is not null) then
    begin
      if (k_from_kf_id <> ma_from_kf_id) then
        kf_id = k_from_kf_id;
    end

    select kal.kal_new_kf_officer
    from knaktlist kal
    where is_delete = 0 and
          kal_from_kf_id = :kf_id and
          kal_date_end >= cast('01.01.2999' as date) and
          kal_date < kal_date_end rows 1
    into lo_id;
  end
  else
  begin
    state_datetime = ma_date_save;
  end

  if (ma_state = -1) then
  begin
    if (ma_date_end <> cast('31.12.9999' as date)) then
    begin
      update miniapp_for_penalty
      set mfp_from_kf_id = :kf_id,
          mfk_from_k_id = :k_id,
          mfp_state = -1,
          mfp_state_datetime = :state_datetime,
          mfp_current_lo_id = :lo_id
      where (mfp_ma_id = :ma_id);

      if (ma_state <> 0) then
        exit;
    end
  end
  else
  begin
    if (k_id is not null) then
    begin
      if (k_status < 0) then
      begin
        update miniapp_for_penalty
        set mfp_from_kf_id = :kf_id,
            mfk_from_k_id = :k_id,
            mfp_state = -1,
            mfp_state_datetime = null,
            mfp_current_lo_id = :lo_id
        where (mfp_ma_id = :ma_id);

        exit;
      end
      else
      begin
        select rd_id
        from hp_get_check_loan_data(:k_id)
        into:rd_id;

        /*select rd.rd_id
        from refdata rd
        where rd.rd_code = 'actiongroup.check_loan_data'
        into :rd_id;*/

        select al.acl_id, al.acl_datetime
        from actionlist al
        where al.acl_object_kind = 9 and
              al.acl_object_id = :k_id and
              al.acl_from_rd_id = :rd_id
        order by al.acl_id
        rows 1
        into :acl_id, :state_datetime;

        if (acl_id is not null) then
        begin
          update miniapp_for_penalty
          set mfp_from_kf_id = :k_from_kf_id,
              mfk_from_k_id = :k_id,
              mfp_state = 1,
              mfp_state_datetime = :state_datetime,
              mfp_current_lo_id = :lo_id
          where (mfp_ma_id = :ma_id);

          exit;
        end
      end
    end
  end
end
