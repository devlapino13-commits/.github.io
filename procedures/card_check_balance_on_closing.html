CREATE PROCEDURE CARD_CHECK_BALANCE_ON_CLOSING(
    CS_ID D_INTEGER)
AS
begin
  if (exists(
    select *
    from cardsync
      inner join kontract
        on k_id = cs_object_id
          and k_fact_closed_ol_id > 0
    where cs_id = :cs_id
      and cs_from_csc_id = f_get_cardsynccfg_id('payment_adjunction')
      and cs_status = 2))
  then
    insert into cardsync(
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_request_json,
      cs_office_id,
      cs_from_k_id,
      cs_from_c_id,
      cs_from_un_id
      )
    select
      f_get_cardsynccfg_id('check_balance_on_closing'),
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      f_format('{"card_id": %1}', cs_card_id),
      cs_office_id,
      cs_from_k_id,
      cs_from_c_id,
      cs_from_un_id
    from cardsync
    where cs_id = :cs_id;
end
