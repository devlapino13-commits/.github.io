CREATE PROCEDURE AAREP_IPF_LOANS(
    REP_REG_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    REG D_STR_100,
    OFFICE D_STR_100,
    C_FIO D_STR_100,
    K_CIRCLE D_SMALLINT,
    SPF_FIO D_STR_100,
    K_FOLDER D_STR_20,
    K_NO D_STR_20,
    DISB_DATE D_DATE)
AS
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable spf_id d_id;
declare variable kf_id d_id;
begin
for select parent_name, k.k_circle, cl.c_name_fio, k.k_kontract_no, k.k_disb_date, ks.ks_region_id, k.k_from_kf_id, cl.c_id
    from kontract k
    inner join kontstate ks on k.k_id = ks.ks_from_k_id
               and :rep_date_end between ks.ks_date_begin and ks.ks_date_end
               and ks.ks_state = 1
               and ks.ks_ol_is_delete = 0
    inner join product pr on k.k_from_pr_id = pr.pr_id
               and pr.pr_code containing '.IPF.'
    inner join knclient kc on k.k_id = kc.kc_from_k_id
               and kc.kc_amount_approved > 0
    inner join clients cl on kc.kc_from_c_id = cl.c_id
    inner join region_list(:rep_reg_id, 1, 3, 1) on id_list = ks.ks_region_id
    into :reg, :k_circle, :c_fio, :k_no, :disb_date, :reg_id, :kf_id, :c_id
    do
    begin
        if (exists(select *
                   from kontract k
                   inner join knclient kc on k.k_id = kc.kc_from_k_id
                   inner join product pr on k.k_from_pr_id = pr.pr_id
                   where kc.kc_from_c_id = :c_id
                         and pr.pr_code not containing '.ipf.')) then continue;
        office = '';
        select reg.reg_name
        from region reg
        where reg.reg_unicode = :reg_id
        into :office;

        spf_id = null;
        k_folder = '';
        select kal.kal_new_kf_officer, kal.kal_new_kf_name
        from knaktlist kal
        where :rep_date_end between kal.kal_date and kal.kal_date_end
              and kal.kal_from_kf_id = :kf_id
              and kal.is_delete = 0
        into :spf_id, :k_folder;

        spf_fio = '';
        select cl.c_name_fio
        from clients cl
        where cl.c_id = :spf_id
        into :spf_fio;
    suspend;
    end
end
