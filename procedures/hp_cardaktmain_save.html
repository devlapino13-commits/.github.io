CREATE PROCEDURE HP_CARDAKTMAIN_SAVE(
    CAM_ID D_INTEGER NOT NULL,
    CAM_DATE D_DATE NOT NULL,
    CARD_LIST D_STR_8100 NOT NULL,
    CARD_COUNT D_INTEGER NOT NULL,
    SRC_CPL_ID D_INTEGER NOT NULL,
    DST_REG_ID D_INTEGER NOT NULL,
    DST_C_ID D_ID NOT NULL,
    DST_C_PIN_ID D_ID,
    DST_IS_TRANSFER D_BOOL NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_CAM_ID D_INTEGER,
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable dst_cpl_id d_integer;
declare variable dst_pin_cpl_id d_integer;
begin
  -- Процедура для сохранения акта приема-передачи банковских карт
  -- Если CAM_ID <> 0,
  --   значит это редактирование еще не подписанного акта
  --   иначе это новый акт

  -- CARD_LIST - массив номеров карт имеет JSON формат:
  -- Пример: {"cards":[{"b":1001,"e":2001},{"b":2501,"e":3001}]}
  -- Буквы "b" и "e" соответствуют словам begin и end

  res_cam_id = 0;
  res_is_error = 0;
  res_text = '';

  if (res_text = '' and trim(card_list) = '') then
    res_text = 'Список карт не указан';

  if (res_text = '' and card_count = 0) then
    res_text = 'Количество карт не указано';

  if (res_text = '' and cam_id <> 0 and exists(select * from cardaktmain
                      where cam_id = :cam_id and
                            cam_status = 1)) then
    res_text = 'Акт уже подписан, редактирование запрещается';

  if (res_text <> '') then
  begin
    res_is_error = 1;
    suspend;
    exit;
  end

  res_text = 'Акт успешно сохранен. Получателю необходимо подписать онлайн документ';
  select cpl_id
  from hp_get_cardplace_id(:dst_reg_id, :dst_c_id, :dst_is_transfer)
  into :dst_cpl_id;

  if (dst_c_pin_id > 0) then
    select cpl_id
    from hp_get_cardplace_id(:dst_reg_id, :dst_c_pin_id, :dst_is_transfer)
    into :dst_pin_cpl_id;

  if (cam_id = 0) then
  begin
    insert into cardaktmain (cam_date, cam_card_list, cam_card_count, cam_src_cpl_id, cam_dst_cpl_id, cam_dst_pin_cpl_id, cam_from_un_id)
    values (:cam_date, :card_list, :card_count, :src_cpl_id, :dst_cpl_id, :dst_pin_cpl_id, :un_id)
    returning cam_id
    into :res_cam_id;
  end
  else
  begin
    res_cam_id = cam_id;

    update cardaktmain
    set cam_date = :cam_date,
        cam_card_list = :card_list,
        cam_card_count = :card_count,
        cam_src_cpl_id = :src_cpl_id,
        cam_dst_cpl_id = :dst_cpl_id,
        cam_dst_pin_cpl_id = :dst_pin_cpl_id,
        cam_from_un_id = :un_id
    where (cam_id = :cam_id);
  end

  suspend;
end
