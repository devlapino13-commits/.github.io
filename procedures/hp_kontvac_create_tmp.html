CREATE PROCEDURE HP_KONTVAC_CREATE_TMP(
    K_ID D_ID,
    DATE_BEGIN D_DATE,
    KV_PERIOD D_INTEGER,
    KV_PERCENT_365 D_PERCENT,
    KV_NEW_AMOUNT D_CURRENCY,
    UN_ID D_ID)
RETURNS (
    OL_ID D_ID,
    LAST_SH_DATE D_DATE,
    RES_TEXT D_STR_255)
AS
declare variable REG_ID D_INTEGER;
declare variable OL_MEMO D_STR_LONG;
declare variable DB_ID D_INTEGER;
declare variable K_KREDLINE_END D_DATE;
declare variable K_STOP_PERC D_DATE;
declare variable PERC_SALDO D_CURRENCY;
declare variable DELINQ_OS D_CURRENCY;
declare variable NEW_PERC D_CURRENCY;
declare variable OKS_BALANCE D_CURRENCY;
declare variable PERC_BALANCE D_CURRENCY;
declare variable DATE_END D_DATE; /* Дата окончания кредитного каникула */
declare variable K_REPAY_DAY D_INTEGER;
declare variable NEW_DATE D_DATE; /* Дата начала старого графика после каникул */
declare variable SH_DATE D_DATE;
declare variable SH_ID D_INTEGER;
declare variable SH_OS D_CURRENCY;
declare variable SH_PERC D_CURRENCY;
declare variable NEW_SH_DATE D_DATE;
declare variable PRV_SH_DATE D_DATE;
declare variable OK_NUMBER D_INTEGER;
declare variable KD_PERCENT D_PERCENT;
declare variable K_PERCENT_FOR_DAYS D_INTEGER;
declare variable ADD_PERIOD D_CURRENCY;
declare variable NEW_LINE D_INTEGER;
declare variable ORIG_BALANCE D_CURRENCY;
declare variable TOTAL_OS D_CURRENCY;
declare variable TOTAL_PERC D_CURRENCY;
begin
  res_text = '';

  ol_memo = '';

  select k_kontract_no || ' от ' || f_datetostr(k_disb_date),
         k_stop_perc, k_kredline_end, k_percent_for_days,
         iif(k_repay_day > 0, k_repay_day, extract(day from k_disb_date))
  from kontract
  where k_id = :k_id
  into ol_memo, k_stop_perc, k_kredline_end, k_percent_for_days, k_repay_day;

  if (row_count = 0) then
    exception e_default 'Kontract info not found';

  if (k_stop_perc is not null) then
  begin
    res_text = 'Договор расторгнут, запрет на кредитные каникулы';
    suspend;
    exit;
  end

  delete from temp_operkred3
  where ok_from_k_id = :k_id;

  insert into temp_operkred3 (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                        ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                        ok_ol_is_active)
  select ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine, ok_pay_type,
         ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active
  from operkred
  where ok_from_k_id = :k_id
  and ok_ol_is_active = 1;

  if (exists(select * from temp_operkred3
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_ol_date > :date_begin)) then
  begin
    res_text = 'По договору имеются записи более поздней датой';
    suspend;
    exit;
  end

  oks_balance = 0;

  select oks_region_id, oks_balance
  from oksummary
  where oks_from_k_id = :k_id and
        oks_date_begin <= :date_begin and
        oks_close_state = 0
  order by oks_from_k_id desc, oks_date_begin desc
  rows 1
  into reg_id, oks_balance;

  if (row_count = 0) then
  begin
    res_text = 'Не найден активный договор';
    suspend;
    exit;
  end

  ol_memo = 'Предоставление кредитных каникул по займу ' || ol_memo;

  /*
  insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_user_id, ol_oper_subtype)
  values (:reg_id, 1, :date_begin, 9, :ol_memo, :un_id, 14)
  returning (ol_id)
  into ol_id;
  */
  ol_id = 1;

  db_id = f_get_db_id();


  perc_saldo = 0;
  if (f_new_perc_active(date_begin) = 1) then
  begin
    select result_perc_end
    from aarep_calc_perc_one_saldo(:k_id, :date_begin, :date_begin)
    into :perc_saldo;
  end else
  begin
    select result_perc_end
    from aarep_calc_perc_one(:k_id, :date_begin, :date_begin, 2, :db_id, :k_kredline_end, null, null)
    into perc_saldo;
  end


  date_end = dateadd(month, kv_period, date_begin);

  kd_percent = kv_percent_365;

  if (k_percent_for_days < 360) then
    kd_percent = round(cast(kd_percent as double precision) / 365 * k_percent_for_days, 7);


  -- по кредитам с кредитным каникулом больше не повышаем ставку
  /*
  insert into kontdata (kd_from_k_id, kd_from_ol_id, kd_ol_date, kd_ol_is_delete, kd_percent,
                        kd_percent_per_day, kd_percent_effect, kd_percent_effect_fifd, kd_memo,
                        kd_fine_percent, kd_float_perc_level, kd_vac_status)
  select kd_from_k_id, :ol_id, :date_begin, 0, :kd_percent, round(cast(:kv_percent_365 as double precision) / 365, 7),
         kd_percent_effect, kd_percent_effect_fifd, 'Начало кредитного каникула', kd_fine_percent, 999, 1
  from kontdata
  where kd_from_k_id = :k_id and
        kd_ol_date <= :date_begin and
        kd_ol_is_delete = 0
  order by kd_ol_date desc
  rows 1;

  insert into kontdata (kd_from_k_id, kd_from_ol_id, kd_ol_date, kd_ol_is_delete, kd_percent,
                        kd_percent_per_day, kd_percent_effect, kd_percent_effect_fifd, kd_memo,
                        kd_fine_percent, kd_float_perc_level, kd_vac_status)
  select kd_from_k_id, :ol_id, :date_end, 0, kd_percent, kd_percent_per_day,
         kd_percent_effect, kd_percent_effect_fifd, 'Завершение кредитного каникула', kd_fine_percent, 999, 2
  from kontdata
  where kd_from_k_id = :k_id and
        kd_ol_date < :date_begin and
        kd_ol_is_delete = 0
  order by kd_ol_date desc
  rows 1;
  */

  new_date = null;
  prv_sh_date = null;
  delinq_os = 0;
  perc_balance = 0;
  orig_balance = 0;

  for select ok_sh_date, ok_sh_id, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
  from temp_operkred3
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type <= 7
  group by ok_sh_date, ok_sh_id
  having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
  into sh_date, sh_id, sh_os, sh_perc
  do
  begin
    insert into temp_operkred3
    (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
     ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active)
    values (:ol_id, -1, :k_id, :sh_id, :sh_date, 2, 0, 0, 0,
            3, 0, -:sh_os, -:sh_perc, 0, 0, :date_begin, :reg_id, 1);

    if (sh_date < date_begin) then
    begin
      delinq_os = delinq_os + sh_os;
      Continue;
    end

    if (perc_balance = 0) then
    begin
      perc_balance = oks_balance - delinq_os; -- База для начисления процентов
      orig_balance = perc_balance;
    end

    if (new_date is null) then
    begin
      prv_sh_date = date_end;

      new_date = sh_date;
      if (extract(day from new_date) <> k_repay_day) then
        new_date = k_repay_day || f_datetostr(new_date, '.mm.yyyy');

      new_date = dateadd(month, kv_period, new_date);

      if (new_date <= date_end) then
        new_date = dateadd(month, 1, new_date);
    end else
      new_date = dateadd(month, 1, new_date);

    new_sh_date = new_date;
    new_sh_date = f_date_work_day(new_sh_date);

    sh_os = 0;
    sh_perc = 0;

    sh_perc = cast(perc_balance as double precision) * kv_percent_365;
    sh_perc = cast(sh_perc as double precision) / 365.0000;
    sh_perc = cast(sh_perc as double precision) * datediff(day, prv_sh_date, new_sh_date);
    sh_perc = round(cast(sh_perc as double precision) / 100.0000);

    if (kv_new_amount <= sh_perc) then
    begin
      res_text = 'Cумма ежемесячного платежа меньше чем сумма процента';
      suspend;
      exit;
    end

    sh_os = kv_new_amount - sh_perc;

    if (sh_os > perc_balance) then
      sh_os = perc_balance;

    if (abs(perc_balance - sh_os) <= 10) then
      sh_os = perc_balance;

    insert into temp_operkred3
    (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
     ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active)
    values (:ol_id, -1, :k_id, :sh_id, :new_sh_date, 2, 0, 0, 0,
            3, 0, :sh_os, :sh_perc, 0, 0, :date_begin, :reg_id, 1)
    returning (ok_number)
    into ok_number;

    perc_balance = perc_balance - sh_os;

    if (perc_balance <= 0) then
      Break;

    prv_sh_date = new_sh_date;
  end

  last_sh_date = new_date;

  new_perc = 0;
  new_perc = cast(orig_balance as double precision) * kv_percent_365;
  new_perc = cast(new_perc as double precision) / 365.0000;
  new_perc = cast(new_perc as double precision) * datediff(day, date_begin, date_end);
  new_perc = round(cast(new_perc as double precision) / 100.0000);

  /*
  insert into opervac (ov_from_ol_id, ov_from_k_id, ov_type, ov_os, ov_perc)
  values (:ol_id, :k_id, 1, :delinq_os, :perc_saldo);

  insert into opervac (ov_from_ol_id, ov_from_k_id, ov_type, ov_os, ov_perc)
  values (:ol_id, :k_id, 2, 0, :new_perc);

  insert into operrest (or_from_ol_id, or_from_k_id, or_kind, or_ol_date, or_ol_is_active, or_from_rd_id, or_from_reg_id, or_date_prot)
  values (:ol_id, :k_id, 4, :date_begin, 1, 0, :reg_id, :date_begin);
  */

  if (perc_balance > 0) then
  begin
    update temp_operkred3
    set ok_sh_os = ok_sh_os + :perc_balance
    where ok_from_ol_id = :ol_id and
          ok_number = :ok_number;

    perc_balance = 0;
  end

  if (delinq_os = 0) then
  begin
    update temp_operkred3
    set ok_sh_os = ok_sh_os - 1
    where ok_from_ol_id = :ol_id and
          ok_number = :ok_number;

    delinq_os = 1;
  end


  add_period = round((delinq_os + perc_saldo + new_perc) / kv_new_amount, 2);

  if (add_period <> round(add_period)) then
    add_period = trunc(add_period) + 1;

  total_os = delinq_os;
  total_perc = perc_saldo + new_perc;

  new_line = 0;
  while (new_line < add_period) do
  begin
    new_line = new_line + 1;

    new_sh_date = dateadd(month, new_line, last_sh_date);
    new_sh_date = f_date_work_day(new_sh_date);

    sh_os = 0;
    sh_perc = 0;

    if (new_line < add_period) then
    begin
      if (delinq_os > 1) then
        sh_os = round(delinq_os / add_period);
  
      sh_perc = kv_new_amount - sh_os;

      if (sh_perc > total_perc) then
        sh_perc = total_perc;

      total_os = total_os - sh_os;
      total_perc = total_perc - sh_perc;
    end else
    begin
      sh_os = total_os;
      sh_perc = total_perc;
    end

    insert into temp_operkred3
    (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
     ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active)
    values (:ol_id, -1, :k_id, :sh_id + :new_line, :new_sh_date, 2, 0, 0, 0,
            3, 0, :sh_os, :sh_perc, 0, 0, :date_begin, :reg_id, 1);

  end

  suspend;
end
