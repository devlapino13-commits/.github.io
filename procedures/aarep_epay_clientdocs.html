CREATE PROCEDURE AAREP_EPAY_CLIENTDOCS(
    C_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    CDS_FROM_C_ID D_ID,
    CDS_ID D_INTEGER,
    CDS_FROM_PK_ID D_INTEGER,
    CDS_DOC_NO D_STR_CODE,
    CDS_DOC_DATE_BEGIN D_DATE,
    CDS_DOC_DATE_END D_DATE,
    CDS_DOC_ISSUED_BY D_STR_255,
    CDS_DOC_BLOB_1 D_ID,
    CDS_DOC_BLOB_2 D_ID,
    CDS_IS_MAIN_DOC D_SMALLINT,
    CDS_DATETIME D_DATETIME,
    CDS_CREATE_DATETIME D_DATETIME,
    PK_NAME D_STR_255,
    CDS_KK_DATETIME D_DATETIME,
    CDS_HAS_BLOB_1 D_SMALLINT,
    CDS_HAS_BLOB_2 D_SMALLINT,
    PDT_CODE TYPE OF COLUMN PASSDOCTYPE.PDT_CODE)
AS
begin
  for select cds_from_c_id,
             cds_id,
             cds_from_pk_id,
             cds_doc_no,
             cds_doc_date_begin,
             cds_doc_date_end,
             cds_doc_issued_by,
             cds_doc_blob_1,
             cds_doc_blob_2,
             cds_is_main_doc,
             cds_datetime,
             cds_create_datetime,
             pk_name,
             pdt_code
      from clientdocs, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk_id and
            cds_is_delete = 0 and
            pk_from_pdt_id = pdt_id and
            :rep_date between cds_doc_date_begin and coalesce(cds_doc_date_end, :rep_date) and
            (pdt_is_main_doc = 1 or
            pdt_code in ('magration_card', 'job_agreement', 'job_agreement_2', 'work_patent', 'kvit_patent',
                         'registration_vnj', 'registration_rvp', 'work_permit',
                         'registration_fiz', 'registration_yur', 'work_permit_WTO'))
      order by cds_from_pk_id, cds_doc_date_begin desc
      into :cds_from_c_id,
           :cds_id,
           :cds_from_pk_id,
           :cds_doc_no,
           :cds_doc_date_begin,
           :cds_doc_date_end,
           :cds_doc_issued_by,
           :cds_doc_blob_1,
           :cds_doc_blob_2,
           :cds_is_main_doc,
           :cds_datetime,
           :cds_create_datetime,
           :pk_name,
           :pdt_code
  do
  begin
    cds_kk_datetime = null;




    select cdk_datetime
    from clientdocs_kk_verify
    where cdk_cds_id = :cds_id
    order by cdk_datetime desc
    rows 1
    into :cds_kk_datetime;

    cds_has_blob_1 = iif(cds_doc_blob_1 > 0, 1, 0);
    cds_has_blob_2 = iif(cds_doc_blob_1 > 0, 1, 0);

    suspend;
  end
end
