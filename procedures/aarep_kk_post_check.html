CREATE PROCEDURE AAREP_KK_POST_CHECK(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ACL_LAST_DATETIME TIMESTAMP,
    KPC_CHECK_DATE DATE,
    KPC_BLOCK_DATETIME TIMESTAMP,
    KPC_CHECK_DATETIME TIMESTAMP,
    UNDERWRITER_FIO STR_100,
    KFOLDER_NAME STR_100,
    GROUP_CLIENTS_FIO VARCHAR(1000),
    KPC_SOLUTION STR_100,
    KPC_MEMO VARCHAR(2048),
    K_AMOUNT_APPROVED CURR_TYPE,
    KOLICH_CLIENT_APPROVED INTEGER,
    PR_NAME STR_100,
    K_KONTRACT_NO STR_100,
    KONTRACT_STATUS INTEGER,
    REG_NAME STR_100,
    KF_OFFICER STR_100,
    K_APPROVED_MANAGER STR_100,
    CHECK_MINUTES INTEGER,
    K_DISB_DATE DATE,
    K_FACT_DISB_LO_FIO STR_100,
    MARK VARCHAR(5))
AS
declare variable kpc_checked_un_id id_type;
declare variable k_id id_type;
declare variable reg_id integer;
declare variable pr_id integer;
declare variable client_fio str_100;
declare variable kpc_solution_id integer;
declare variable kf_officer_id id_type;
declare variable k_fact_disb_lo_id id_type;
declare variable k_approved_manager_id id_type;
declare variable rd_id_loan integer;
declare variable rd_id_loan_sp d_integer;
declare variable is_blocked integer;
declare variable reg_id_tmp id_type;
declare variable acl_id id_type;
begin
  select rd_id
  from refdata
  where rd_code = 'ActionGroup.Check_Loan_Data' and
        rd_is_active = 1
  into :rd_id_loan;

  select rd_id
  from refdata
  where rd_code = 'ActionGroup.Check_Loan_Data_Special' and
        rd_is_active = 1
  into :rd_id_loan_sp;

  if (rd_id_loan is null or rd_id_loan_sp is null) then
    exit;

  for select acl_from_reg_id, kpc.kpc_checked_un_id, kpc.kpc_check_datetime, kpc.kpc_check_date, kpc.kpc_block_datetime,
             kpc.kpc_memo, kpc.kpc_solution, al.acl_object_id, iif(kpc.kpc_check_date is null, 1, 0) as is_blocked,
             acl_last_datetime, kpc_from_acl_id
      from kk_post_check kpc, region_list(:rep_region_id, 1, -1, 1), actionlist al
      where al.acl_object_kind = 9 and
            al.acl_from_rd_id in (:rd_id_loan, :rd_id_loan_sp) and
            kpc.kpc_action_date between :rep_date_begin and :rep_date_end and
            al.acl_id = kpc.kpc_from_acl_id and
            acl_from_reg_id = id_list
      into :reg_id, :kpc_checked_un_id, :kpc_check_datetime, :kpc_check_date, :kpc_block_datetime, :kpc_memo,
           :kpc_solution_id, :k_id, :is_blocked, :acl_last_datetime, :acl_id
  do
  begin
    mark = iif(exists(select *
                      from actionmark
                      where atm_from_acl_id = :acl_id), ' + ', '');

    if (coalesce(reg_id_tmp, 0) <> reg_id) then
    begin
      reg_id_tmp = reg_id;
      select reg.reg_code || ' ' || reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :reg_name;
    end

    select kont.k_kontract_no, kont.k_from_pr_id, kont.k_kolich_client_approved, kont.k_approved_manager,
           kont.k_amount_approved, kont.k_disb_date, kont.k_fact_disb_lo_id, kal.kal_new_kf_officer,
           kal.kal_new_kf_name, kal_new_reg_id
    from kontract kont
    inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
    where kont.k_id = :k_id and
          :rep_date_end between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into :k_kontract_no, :pr_id, :kolich_client_approved, :k_approved_manager_id, :k_amount_approved, :k_disb_date,
         :k_fact_disb_lo_id, :kf_officer_id, :kfolder_name, :reg_id;

    select pr.pr_name
    from product pr
    where pr.pr_id = :pr_id
    into :pr_name;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :kf_officer_id
    into :kf_officer;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :k_fact_disb_lo_id
    into :k_fact_disb_lo_fio;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :k_approved_manager_id
    into :k_approved_manager;

    select k_last_status
    from hp_get_kontract_status(:k_id, :k_disb_date)
    into :kontract_status;

    group_clients_fio = '';
    for select cl.c_name_fio
        from knclient kncl
        inner join clients cl on cl.c_id = kncl.kc_from_c_id
        where kncl.kc_from_k_id = :k_id and
              kncl.kc_amount_approved > 0
        into :client_fio
    do
    begin
      group_clients_fio = group_clients_fio || '' || client_fio || '; ';
    end
    group_clients_fio = substring(group_clients_fio from 1 for char_length(group_clients_fio) - 2);

    select cl.c_name_fio
    from username un
    inner join clients cl on un.un_from_c_id = cl.c_id
    where un.un_id = :kpc_checked_un_id
    into :underwriter_fio;

    kpc_solution = 'В процессе';
    check_minutes = null;
    if (is_blocked = 0) then
    begin
      check_minutes = datediff(minute, kpc_block_datetime, kpc_check_datetime);
      select rd.rd_name
      from refdata rd
      where rd.rd_id = :kpc_solution_id
      into :kpc_solution;
    end
    suspend;
  end
end
