CREATE PROCEDURE UXP$UXCODE_INFO_ACTIONS(
    IN_UFM_CODE D_STR_CODE,
    IN_UXA_CODE D_STR_CODE)
RETURNS (
    FIA_INDEX D_INTEGER,
    FIA_ID D_STR_100,
    FIA_PARENT_ID D_STR_100,
    FIA_CHECKED D_BOOL,
    FIA_CAPTION D_STR_100,
    FIA_TYPE D_STR_CODE,
    FIA_UXA_FCODE D_STR_100,
    FIA_UXA_NAME D_STR_100,
    FIA_UX_FCODE D_STR_100,
    FIA_UX_NAME D_STR_100,
    FIA_UX_KIND D_STR_CODE,
    FIA_ID_UUID D_UUID,
    FIA_UFM_UUID D_UUID,
    FIA_UFM_UUID_CODE D_STR_CODE,
    FIA_SOURCES D_STR_4096)
AS
declare variable tmp_uuid d_uuid;
declare variable ua_usr_id_field d_str_code;
declare variable tmp_caption d_str_100;
declare variable uxa_owner d_uuid;
declare variable ufm_uuid d_uuid;
declare variable uxa_uuid d_uuid;
begin
  ufm_uuid = null;
  uxa_uuid = null;

  if (coalesce(in_ufm_code, '') <> '') then
    ufm_uuid = char_to_uuid(in_ufm_code);

  if (coalesce(in_uxa_code, '') <> '') then
    uxa_uuid = char_to_uuid(in_uxa_code);

  if (ufm_uuid is null) then
    exit;

  fia_index = 0;
  fia_parent_id = null;
  fia_type = 'Форма';
  fia_sources = '';

  fia_uxa_fcode = '';
  fia_uxa_name = '';
  fia_checked = null;
  fia_ufm_uuid = ufm_uuid;
  fia_ufm_uuid_code = uuid_to_char(fia_ufm_uuid);

  for select ufm_uuid,
             uuid_to_char(ufm_uuid),
             ux_name || '  (' || trim(iif(ux_release_uuid = ufm_uuid, 'релиз',
                                          iif(ux_draft_uuid = ufm_uuid, 'драфт', 'версия'))) || ')',
             ux_fcode,
             ux_name,
             uxf$kind(uu_kind, 'caption')
  from ux$form, ux$code, ux$uniq
  where ufm_uuid = :ufm_uuid and
        ufm_ux_uuid = ux_uuid and
        uu_id = ux_uuid
  into fia_id_uuid, fia_id, fia_caption, fia_ux_fcode,  fia_ux_name, fia_ux_kind
  do
  begin
    fia_sources = '';
    select list(us_code)
    from ux$frame, ux$source
    where ufr_owner = :fia_ufm_uuid and
          ufr_us_uuid = us_uuid
    into fia_sources;

    suspend;
  end

  if (row_count = 0) then
    exit;

  fia_parent_id = fia_id;
  fia_type = 'Действия';

  for select ua_uuid,
             uuid_to_char(ua_uuid),
             ua_name,
             ux_fcode,
             ux_name,
             uxf$kind(uu_kind, 'caption'),
             uxa_fcode,
             uxa_name,
             uxa_owner
  from ux$action, ux$alias, ux$code, ux$uniq
  where ua_owner = :ufm_uuid and
        ua_uxa_uuid = uxa_uuid and
        uxa_owner = ux_uuid and
        uu_id = ux_uuid
  order by uxa_fcode
  into fia_id_uuid, fia_id, fia_caption, fia_ux_fcode,  fia_ux_name, fia_ux_kind, fia_uxa_fcode, fia_uxa_name, uxa_owner
  do
  begin
    fia_ufm_uuid = uxf$ver(uxa_owner);
    fia_ufm_uuid_code = uuid_to_char(fia_ufm_uuid);
    fia_checked = 0;

    if (uxa_uuid is not null) then
    begin
      if (exists(select * from ux$aliasaction
                 where uaa_uxa_uuid = :uxa_uuid and
                       uaa_ua_uuid = :fia_id_uuid)) then
        fia_checked = 1;
    end

    fia_sources = '';
    select list(us_code)
    from ux$frame, ux$source
    where ufr_owner = :fia_ufm_uuid and
          ufr_us_uuid = us_uuid
    into fia_sources;

    fia_index = fia_index + 1;
    suspend;
  end

  for select fr_uuid,
             uuid_to_char(fr_uuid),
             iif(master_fr_uuid is null, uuid_to_char(:ufm_uuid), uuid_to_char(master_fr_uuid)),
             ufr_name,
             ux_fcode,
             ux_name,
             uxf$kind(uu_kind, 'caption')
  from ux$frame, ux$form, ux$code, ux$uniq, uxp$uxframe_tree(:ufm_uuid, null, '')
  where fr_uuid = ufr_uuid and
        ufr_owner = ufm_uuid and
        ufm_ux_uuid = ux_uuid and
        ux_uuid = uu_id
  into :fia_id_uuid, :fia_id, :fia_parent_id, :fia_caption, :fia_ux_fcode,  :fia_ux_name, :fia_ux_kind
  do
  begin
    fia_uxa_fcode = '';
    fia_uxa_name = '';
    fia_ufm_uuid = fia_id_uuid;
    fia_ufm_uuid_code = uuid_to_char(fia_ufm_uuid);
    fia_type = 'Фрейм';

    fia_sources = '';
    select list(us_code)
    from ux$frame, ux$source
    where ufr_uuid = :fia_ufm_uuid and
          ufr_us_uuid = us_uuid
    into fia_sources;

    fia_checked = null;
    fia_index = fia_index + 1;
    suspend;

    fia_parent_id = fia_id;
    tmp_uuid = fia_id_uuid;

    for select ua_uuid,
               uuid_to_char(ua_uuid),
               ua_name,
               ux_fcode,
               ux_name,
               uxf$kind(uu_kind, 'caption'),
               uxa_fcode,
               uxa_name,
               ua_usr_id_field,
               uxa_owner
    from ux$action, ux$alias, ux$code, ux$uniq
    where ua_owner = :tmp_uuid and
          ua_uxa_uuid = uxa_uuid and
          uxa_owner = ux_uuid and
          uu_id = ux_uuid and
          coalesce(ua_usr_id_field, '') <> ''
    into fia_id_uuid, fia_id, fia_caption, fia_ux_fcode,  fia_ux_name, fia_ux_kind,
         fia_uxa_fcode, fia_uxa_name, ua_usr_id_field, uxa_owner
    do
    begin
      fia_type = 'Поисковик';
      fia_caption = '   ' || fia_caption;
      fia_checked = 0;

      fia_ufm_uuid = uxf$ver(uxa_owner);
      fia_ufm_uuid_code = uuid_to_char(fia_ufm_uuid);

      tmp_caption = '';
      select uf_name
      from ux$frparam, ux$field
      where ufp_ufr_uuid = :tmp_uuid and
            ufp_uf_uuid = uf_uuid and
            uf_field_name = :ua_usr_id_field
      into tmp_caption;

      if (row_count = 1) then
      begin
        fia_caption = '   ' || tmp_caption;

        for select ufi_name
        from ux$fritem, ux$field
        where ufi_ufr_uuid = :tmp_uuid and
              ufi_uf_uuid = uf_uuid and
              uf_field_name = :ua_usr_id_field
        into tmp_caption
        do
        begin
          fia_checked = 1;
          fia_caption = '   ' || tmp_caption;
        end

        if (fia_checked = 0) then
        begin
          for select ugi_caption
          from ux$grid, ux$gritem, ux$field
          where ugr_ufr_uuid = :tmp_uuid and
                ugi_ugr_uuid = ugr_uuid and
                ugi_uf_uuid = uf_uuid and
                uf_field_name = :ua_usr_id_field
          into tmp_caption
          do
          begin
            fia_checked = 1;
            fia_caption = '   ' || tmp_caption;
          end
        end
      end

      fia_sources = '';
      select list(us_code)
      from ux$frame, ux$source
      where ufr_owner = :fia_ufm_uuid and
            ufr_us_uuid = us_uuid
      into fia_sources;

      fia_index = fia_index + 1;
      suspend;
    end

    for select ua_uuid,
               uuid_to_char(ua_uuid),
               ua_name,
               ux_fcode,
               ux_name,
               uxf$kind(uu_kind, 'caption'),
               uxa_fcode,
               uxa_name,
               uxa_owner
    from ux$action, ux$alias, ux$code, ux$uniq
    where ua_owner = :tmp_uuid and
          ua_uxa_uuid = uxa_uuid and
          uxa_owner = ux_uuid and
          uu_id = ux_uuid and
          coalesce(ua_usr_id_field, '') = ''
    order by uxa_fcode
    into fia_id_uuid, fia_id, fia_caption, fia_ux_fcode,  fia_ux_name, fia_ux_kind,
         fia_uxa_fcode, fia_uxa_name, uxa_owner
    do
    begin
      fia_type = 'Действия';
      fia_ufm_uuid = uxf$ver(uxa_owner);
      fia_ufm_uuid_code = uuid_to_char(fia_ufm_uuid);

      fia_checked = 0;
  
      if (uxa_uuid is not null) then
      begin
        if (exists(select * from ux$aliasaction
                   where uaa_uxa_uuid = :uxa_uuid and
                         uaa_ua_uuid = :fia_id_uuid)) then
          fia_checked = 1;
      end

      fia_sources = '';
      select list(us_code)
      from ux$frame, ux$source
      where ufr_owner = :fia_ufm_uuid and
            ufr_us_uuid = us_uuid
      into fia_sources;

      fia_index = fia_index + 1;
      suspend;
    end
  end
end
