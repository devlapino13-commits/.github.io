  -- RFCRU-4941
  select pl.pl_add_code
  from addrlist adl

  where adl.adl_id = :pte_adl_id_1
  into res_okato_for_pte_adl_id_1;

  select pl.pl_add_code
  from addrlist adl
  inner join places pl on pl.pl_id = adl.adl_from_pl_id

  where adl.adl_id = :c_adl_id_2
  into res_okato_for_c_adl_id_2;

  select cds_id,
         left(pk_name || ', годен до ' || f_datetostr(cds_doc_date_end) || ', добавлен ' || f_datetimetostr(cds_create_datetime), 100)
  from clientdocs
  inner join passkind on pk_id = cds_from_pk_id
  inner join passdoctype on pdt_id = pk_from_pdt_id
  where cds_from_c_id = :in_c_id and
        pdt_code in ('registration_fiz', 'registration_permanent')
  order by iif(cds_doc_date_end < current_date or cds_is_delete = 1, 1, 0), cds_create_datetime desc
  rows 1
  into pte_cds_id, -- документ регистрации (RFCRU-5061)
       pte_reg_info;

  select atr_from_rd_id
  from addresstype_registration
  where atr_from_cds_id = :pte_cds_id
  order by atr_datetime desc
  rows 1
  into pte_reg_type; -- код адреса регистрации (RFCRU-5407)

  suspend;
