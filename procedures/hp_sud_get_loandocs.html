CREATE PROCEDURE HP_SUD_GET_LOANDOCS(
    PARAM_JSON D_STR_255,
    REQUEST_ID D_ID,
    SHOW_DELETED D_BOOL,
    UN_ID D_ID)
RETURNS (
    REC_NO D_INTEGER,
    REC_NAME D_STR_100,
    DOC_KIND D_INTEGER,
    DOC_NAME D_STR_CODE,
    SMD_ID D_ID,
    SLC_ID D_ID,
    SLD_ID D_ID,
    SLD_STATUS D_INTEGER,
    SLD_STATUS_STR D_STR_CODE,
    SLD_KIND D_INTEGER,
    SLD_KIND_STR D_STR_CODE,
    BLS_COUNT D_INTEGER,
    IS_REQUIRED D_BOOL,
    K_ID D_ID)
AS
declare variable k_kontract_no d_str_code;
declare variable scd_required d_smallint;
declare variable param d_str_10;
declare variable stage_code d_str_100;
begin
  -- процедура возвращает список документов по запросу на подачу в суд

  if (show_deleted is null) then
    show_deleted = 0;

  rec_no = 0;
  smd_id = request_id;

  param = f_json_str_get_value(param_json, 'param', '|A0|');
  stage_code = f_json_str_get_value(param_json, 'stage_code', '|A0|');

  select k_kontract_no, k_id
  from sud_mainreq, kontract
  where smr_id = :request_id and
        smr_from_k_id = k_id
  into k_kontract_no, k_id;

  for select sld_id, sld_kind, iif(sld_inactive = 0, sld_status, -99), scd_id, scd_name, sld_from_slc_id,
            maxvalue(scd_on_request, scd_on_client)
  from sud_loandocs, sud_cfgdockind
  where sld_doc_kind = scd_id and
        sld_from_smd_id = :request_id and
        (:show_deleted = 1 or sld_inactive = 0)
  order by scd_sort, scd_id
  into :sld_id, :sld_kind, :sld_status, doc_kind, doc_name, slc_id, scd_required
  do
  begin
    is_required = 1;
    -- фильтр по параметру
    -- D3 - Ожидается прикрепления документа от офиса, D4 - Ожидается прикрепления документа от архива
    if ((param = 'D3' and sld_kind <> 1 and sld_status < 1)
      or (param = 'D4' and sld_kind <> 2 and sld_status < 1)
    )then
      continue;

    if (stage_code containing '|D' and scd_required < 2) then
      is_required = 0;

    sld_status_str = '-';
    sld_kind_str = '';
    bls_count = 0;

    if (slc_id is null)
    then rec_name = k_kontract_no;
    else
    begin
      select c_name_fio
      from sud_loanclient, knclient, clients
      where slc_id = :slc_id and
            slc_from_kc_id = kc_id and
            kc_from_c_id = c_id
      into rec_name;
    end

    sld_status_str = case sld_status
      when -2 then 'Отказан'
      when -1 then 'На доработку'
      when 0 then 'Создан'
      when 1 then 'На проверке'
      when 2 then 'Одобрен'
      when -99 then 'Удален'
    end;

    sld_kind_str = case sld_kind
      when 0 then 'БД'
      when 1 then 'Запрос в офис'
      when 2 then 'Запрос в архив'
      when -1 then 'В офисе не найден'
      when -2 then 'В архиве не найден'
    end;

    select count(*)
    from sud_loanbls
    where slb_from_sld_id = :sld_id
      and slb_is_delete = 0
    into bls_count;

    if (bls_count = 0) then
      bls_count = null;

    rec_no = rec_no + 1;
    suspend;
  end
end
