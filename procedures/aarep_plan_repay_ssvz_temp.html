CREATE PROCEDURE AAREP_PLAN_REPAY_SSVZ_TEMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_C_ID ID_TYPE = 0)
RETURNS (
    REP_DATE DATE,
    OKS_KC_ID ID_TYPE,
    OKS_FROM_K_ID ID_TYPE,
    OK_SH_OS CURR_TYPE,
    OK_SH_PERC CURR_TYPE,
    DELINQ_ZAD_OS CURR_TYPE,
    DELINQ_ZAD_PERC CURR_TYPE,
    DELINQ_ZAD_FINE CURR_TYPE,
    DELINQ_ZAD_TOTAL CURR_TYPE,
    DELINQ_DAY INTEGER,
    IS_DELINQ SMALLINT,
    AA_CATEGORY_ID SMALLINT)
AS
declare variable SH_CALC_PERC CURR_TYPE;
begin
  if (not exists(select * from loanmgrlist
      where lml_from_c_id = :rep_c_id
      and :rep_date_end between lml_date_begin and lml_date_end)) then rep_c_id = 0;

  for select distinct oks_kc_id
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_kc_id <> 1000000000003 and
            oks_close_state = 0 and
            ((:rep_c_id = 0 and oks_region_id = :rep_region_id)
             or (:rep_c_id <> 0 and :rep_c_id = f_lo_mgr_c_id(f_lo_mgr_id(oks_from_k_id, :rep_date_end, oks_kc_id,oks_region_id))))
      into oks_kc_id
  do
  begin
    ok_sh_os = null;
    ok_sh_perc = null;
    rep_date = null;
--    Просроченные кредиты
    for select oks_from_k_id, :rep_date_end - oks_delinq_date,
               (select aa_category_id
                from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null))
        from oksummary
        where :rep_date_end between oks_date_begin and oks_date_end and
              oks_kc_id = :oks_kc_id and
              oks_region_id = :rep_region_id and
              oks_close_state = 0 and
              oks_delinq_date < :rep_date_end and
              (select aa_category_id
               from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
        into oks_from_k_id, delinq_day, aa_category_id
    do
    begin
      delinq_zad_os = 0;
      delinq_zad_perc = 0;
      delinq_zad_fine = 0;
      is_delinq = 0;
      select sum(zad_os), sum(zad_perc), sum(sh_calc_fine), sum(sh_calc_perc)
      from aarep_make_zad_list(:oks_from_k_id, :rep_date_end, 1)
      where sh_repay_type = 1 and
            sh_date is not null
      into delinq_zad_os, delinq_zad_perc, delinq_zad_fine, sh_calc_perc;
      if (delinq_zad_os is null) then
        delinq_zad_os = 0;
      if (delinq_zad_perc is null) then
        delinq_zad_perc = 0;
      if (delinq_zad_fine is null) then
        delinq_zad_fine = 0;
      if (delinq_zad_perc < sh_calc_perc) then
        delinq_zad_perc = sh_calc_perc;
      delinq_zad_total = delinq_zad_os + delinq_zad_perc + delinq_zad_fine;
      if (delinq_zad_total > 0) then
      begin
        is_delinq = 1;
        suspend;
      end
    end
    delinq_day = null;
    delinq_zad_os = 0;
    delinq_zad_perc = 0;
    delinq_zad_fine = 0;
    delinq_zad_total = 0;
--      План погашения
    is_delinq = 0;
    rep_date = :rep_date_end;
    while (rep_date <= rep_date_end + 7) do
    begin
      oks_from_k_id = null;
      ok_sh_os = null;
      ok_sh_perc = null;

      for select oks_from_k_id
          from oksummary
          where :rep_date_end between oks_date_begin and oks_date_end and
                oks_kc_id = :oks_kc_id and
                oks_region_id = :rep_region_id and
                oks_close_state = 0 and
                (select aa_category_id
                 from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
          into oks_from_k_id
      do
      begin
        for select sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
            from operkred
            where ok_from_k_id = :oks_from_k_id and
                  ok_ol_is_active = 1 and
                  ok_ol_date <= :rep_date_end and
                  ok_type not in (7, 8) and
                  ok_sh_date = :rep_date and
                  ok_from_c_id = 0 having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0
            into ok_sh_os, ok_sh_perc
        do
        begin
          suspend;
        end
      end
      if (ok_sh_os is null) then
      begin
        oks_from_k_id = null;
        suspend;
      end
      rep_date = rep_date + 1;
    end
  end

end
