CREATE PROCEDURE HP_SAVE_RES_ISHENIM(
    C_ID D_ID,
    MA_ID D_ID,
    RESULT D_BLOB_TEXT,
    STATUS D_STR_CODE,
    REQUEST_TEXT D_STR_8100 = null,
    ANSWER_TEXT D_STR_8100 = null)
AS
declare variable eil_id d_id;
begin
  eil_id = null;
  if (exists(select *
             from extinfolist eil
             where eil.eil_from_c_id = :c_id and
                   eil.eil_from_eic_id = 1 and
                   eil.eil_status = 0 and
                   eil.eil_date = current_date)) then
  begin
    select eil.eil_id
    from extinfolist eil
    where eil.eil_from_c_id = :c_id and
          eil.eil_from_eic_id = 1 and
          eil.eil_status = 0 and
          eil.eil_object_kind = f_objkind('miniapp', 'id') and
          eil.eil_object_id = :ma_id and
          eil.eil_date = current_date
    order by eil.eil_id desc
    rows 1
    into eil_id;
  end
  else
  begin
    insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_is_json,
                             eil_from_eip_id, eil_from_un_id)
    values (:c_id, 1, current_date, f_objkind('miniapp', 'id'), :ma_id, 1, null, 1000000000000)
    returning eil_id
    into :eil_id;
  end
  execute procedure hp_extinfo_set_status(:eil_id, :status, '', :request_text, :answer_text);
  if (coalesce(:result,  '') <> '') then
    execute procedure hp_extinfo_set_result(:eil_id, :result);
end
