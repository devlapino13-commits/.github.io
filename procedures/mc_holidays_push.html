CREATE PROCEDURE MC_HOLIDAYS_PUSH
AS
declare variable K_ID D_ID;
declare variable MCL_ID D_ID;
declare variable MCL_LANGUAGE_CODE D_STR_10;
declare variable SUBJECT D_STR_255;
declare variable MSG_BODY D_STR_LONG;
begin
    if (current_date = '30.04.2026') then
    begin
        for select
            mcl.mcl_id,
            mcl.mcl_lang_code,
            kc.kc_from_k_id
        from mc_login mcl
        inner join knclient kc on kc.kc_from_c_id = mcl.mcl_from_c_id and kc.kc_amount_approved > 0
        inner join kontstate ks on kc.kc_from_k_id = ks.ks_from_k_id
        where mcl.mcl_is_delete = 0
            and ks.ks_ol_is_delete = 0
            and ks.ks_state = 1
            and '01.05.2026' between ks.ks_date_begin and ks.ks_date_end
        into
            :mcl_id,
            :mcl_language_code,
            :k_id
        do
        begin
            -- погашения с 1 по 10 мая
            if (exists
                (
                    select *
                    from operkred ok
                    where ok.ok_from_k_id = :k_id
                        and ok.ok_sh_date between cast('01.05.2026' as date) and cast('10.05.2026' as date)
                        and ok.ok_type < 7
                        and ok.ok_os <> 0
                        and ok.ok_ol_is_active = 1
                )
            ) then
            begin
                if (:mcl_language_code = 'ru') then
                begin
                    :subject = 'Уважаемый клиент,  с праздником!';
                    :msg_body = 'В период праздничных выходных дней рекомендуем произвести плановые платежи по контрактам через доступные платежные системы до 11 мая 2026 года (включительно).';
                end

                if (:mcl_language_code = 'kg') then
                begin
                    :subject = 'Урматтуу кардар, майрамыңыз менен!';
                    :msg_body = 'Майрамдык эс алуу күнд�©рүнд�© келишимдер боюнча пландалган т�©л�©мд�©рдү 2026-жылдын 11-майына чейин жеткиликтүү т�©л�©м системалары аркылуу жүргүзс�©ңүз болот.';
                end

                execute procedure mc_send_push('mc_login', :mcl_id, :subject, :msg_body, -1);
            end
        end
    end
end
