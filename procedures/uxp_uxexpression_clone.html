Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXEXPRESSION_CLONE(
    UX_UUID D_UUID,
    NEW_UX_UUID D_UUID)
AS
begin
  delete from ux$expression
  where exp_ux_uuid = :new_ux_uuid;

  for select exp_ux_uuid, exp_kind, exp_expression, exp_message, exp_is_sql, exp_action, exp_position, exp_is_bool, exp_calc_kind
  from ux$expression
  where exp_ux_uuid = :ux_uuid
  as cursor exp
  do
  begin
    update or insert into ux$expression (exp_ux_uuid, exp_kind, exp_expression, exp_message,
             exp_is_sql, exp_action, exp_position, exp_is_bool, exp_calc_kind)
    values (:new_ux_uuid, exp.exp_kind, exp.exp_expression, exp.exp_message,
             exp.exp_is_sql, exp.exp_action, exp.exp_position, exp.exp_is_bool, exp.exp_calc_kind)
    matching (exp_ux_uuid, exp_kind);

    for select exf_ux_uuid, exf_kind, exf_uf_uuid
    from ux$exprfield
    where exf_ux_uuid = exp.exp_ux_uuid and
          exf_kind = exp.exp_kind
    as cursor exf
    do
    begin
      update or insert into ux$exprfield (exf_ux_uuid, exf_kind, exf_uf_uuid)
      values (:new_ux_uuid, exf.exf_kind, exf.exf_uf_uuid)
      matching (exf_ux_uuid, exf_kind, exf_uf_uuid);
    end
  end
end
