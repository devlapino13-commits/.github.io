CREATE PROCEDURE CC_REGION_LIST
RETURNS (
    REG_UNICODE INTEGER,
    REG_NAME STR_100,
    REG_ADDRESS_FACT ID_TYPE,
    ADL_ADDRESS_STR STR_255,
    CITY_ID ID_TYPE,
    SMS_TEXT STR_255)
AS
begin
  for select reg.reg_unicode, reg.reg_name, reg.reg_address_fact, adl_address_str, adl_from_pl_id, org.or_sms_text
      from region reg
      inner join addrlist adl on adl.adl_id = reg_address_fact
      left join okt_region  org on org.or_from_reg_id = reg.reg_unicode
      where reg.reg_status = 3 and
            reg.reg_date_close is null and
            reg.reg_unicode <> 1010100 -- Москва ГО
      order by reg_name
      into :reg_unicode, :reg_name, :reg_address_fact, :adl_address_str, :city_id, :sms_text
  do
  begin
    if (reg_name not in ('0121', '0123')) then
        suspend;
  end
end
