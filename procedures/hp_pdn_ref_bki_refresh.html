CREATE PROCEDURE HP_PDN_REF_BKI_REFRESH(
    UN_ID D_ID NOT NULL)
AS
declare variable c_id d_id;
declare variable ma_id d_integer;
declare variable k_id d_id;
declare variable k_fact_disb_date d_date;
declare variable k_fact_closed_date d_date;
declare variable pdn_status_id d_smallint;
declare variable ref_cnt_list d_str_20;
declare variable ref_cnt_active d_smallint;
declare variable ref_our_cnt d_smallint;
declare variable ref_our_active d_str_long;
declare variable ref_our_cnt_active d_smallint;
declare variable ref_ext d_str_long;
declare variable ref_ext_active d_str_long;
declare variable ref_ext_cnt d_smallint;
declare variable ref_ext_cnt_active d_smallint;
declare variable ssp_date d_date;
declare variable ssp_note d_str_255;
declare variable ssp_note_id d_smallint;
declare variable ki_date d_date;
declare variable ki_note d_str_255;
declare variable ki_note_id d_smallint;
declare variable txt d_str_255;
declare variable cnt d_integer;
begin
  -- ПДН - расчет в связи с неисполнением цели по рефинансированию (RFCRU-4309)
  -- процедура в таблице pdn_ref_recalc обновляет поля ССП/КИ и статус ПДН
  -- это нужно сделать через некоторое время после отправки запросов ССП/КИ

  for select prr_k_id,
             prr_k_fact_disb_date,
             prr_k_fact_closed_date,
             prr_pdn_status_id,
             prr_ma_id,
             prr_c_id
      from pdn_ref_recalc
      where prr_is_selected = 1
      into k_id,
           k_fact_disb_date,
           k_fact_closed_date,
           pdn_status_id,
           ma_id,
           c_id
  do
  begin
    ssp_date = null;
    ssp_note = null;
    ssp_note_id = null;

    ki_date = null;
    ki_note = null;
    ki_note_id = null;

    begin
      in autonomous transaction do
        select trim(ssp_result) || ' ' || f_datetimetostr(ssp_datetime),
               ssp_result_id,
               trim(ki_result) || ' ' || f_datetimetostr(ki_datetime),
               ki_result_id
        from hp_pdn_get_request_status(:c_id, :k_fact_disb_date + 1)
        into ssp_note, -- комментарий относительно последнего запроса ССП после выдачи займа
             ssp_note_id,
             ki_note, -- комментарий относительно последнего запроса КИ после выдачи займа
             ki_note_id;

      when any do
      begin
        txt = 'При получении результатов запросов ССП/КИ произошла ошибка (hp_pdn_ref_bki_refresh)';
        txt = txt || ascii_char(10) || 'c_id ' || c_id || ', k_fact_disb_date ' || f_datetostr(k_fact_disb_date);
        exception e_message txt;
      end
    end

    select max(eil_date)
    from extinfolist
    inner join extinfocfg on eic_id = eil_from_eic_id
    where eic_code = 'bki_ssp_40' and -- запрос ССП
          eil_status = 2 and -- запрос успешный
          eil_from_c_id = :c_id and
          eil_date > :k_fact_disb_date
    into ssp_date; -- дата последнего успешного запроса ССП после выдачи

    select max(eil_date)
    from extinfolist
    inner join extinfocfg on eic_id = eil_from_eic_id
    where eic_code = 'equifax_40' and -- запрос КИ
          eil_status = 2 and -- запрос успешный
          eil_from_c_id = :c_id and
          eil_date > :k_fact_disb_date
    into ki_date; -- дата последнего успешного запроса КИ после выдачи

    if (ssp_note is not null) then
      update pdn_ref_recalc
      set prr_ssp_date = :ssp_date,
          prr_ssp_note = :ssp_note,
          prr_ssp_note_id = :ssp_note_id
      where prr_k_id = :k_id;

    if (ki_note is not null) then
      update pdn_ref_recalc
      set prr_ki_date = :ki_date,
          prr_ki_note = :ki_note,
          prr_ki_note_id = :ki_note_id
      where prr_k_id = :k_id;

    -- обновляем статус ПДН
    if (pdn_status_id in (4, 7)) then
    begin
      ref_our_active = null;
      ref_ext_active = null;
      ref_ext = null;

      ref_our_cnt = null;
      ref_our_cnt_active = null;
      ref_ext_cnt = null;
      ref_ext_cnt_active = null;

      ref_cnt_list = null;
      ref_cnt_active = null;

      select count(*)
      from kontrefinance
      left join kontract on k_id = kr_from_k_id
      where kr_from_ma_id = :ma_id and
            k_id is not null
      into ref_our_cnt; -- внутренние рефы всего

      select list(k_kontract_no, ', '),
             count(*)
      from kontrefinance
      inner join kontract on k_id = kr_from_k_id
      inner join oksummary on oks_from_k_id = k_id
      where kr_from_ma_id = :ma_id and
            current_date between oks_date_begin and oks_date_end
      into ref_our_active,
           ref_our_cnt_active; -- внутренние рефы активные

      select list(kr_loan_uuid, ', '),
             count(*)
      from kontrefinance
      where kr_from_ma_id = :ma_id and
            kr_from_k_id is null
      into ref_ext,
           ref_ext_cnt; -- внешние рефы всего

      if (ssp_date is null) then -- нет успешного запроса ССП после выдачи займа
      begin
        ref_ext_active = ref_ext; -- внешние рефы активные повторяются из колонки внешние рефы всего
        ref_ext_cnt_active = ref_ext_cnt;
      end
      else
      begin
        in autonomous transaction do
          select list(kr_loan_uuid, ', '),
                 count(*)
          from kontrefinance
          inner join hp_bki_get_list_for_refinace(:c_id, :k_fact_disb_date + 1, 1) on uuid = kr_loan_uuid
          where kr_from_ma_id = :ma_id and
                kr_from_k_id is null
          into ref_ext_active,
               ref_ext_cnt_active; -- внешние рефы активные
      end

      ref_cnt_active = ref_our_cnt_active + ref_ext_cnt_active;
      ref_cnt_list = ref_our_cnt || '/' || ref_our_cnt_active || ' - ' || ref_ext_cnt || '/' || ref_ext_cnt_active; -- кол-во рефов (внутренних-внешних)

      if (k_fact_closed_date is null and ssp_date >= current_date - 1 and ref_cnt_active = 0) then
        update pdn_ref_recalc
        set prr_pdn_status = 'Рефы закрыты',
            prr_pdn_status_id = 3,
            prr_ref_cnt_list = :ref_cnt_list,
            prr_ref_our_active = :ref_our_active,
            prr_ref_our_cnt_active = :ref_our_cnt_active,
            prr_ref_ext_active = :ref_ext_active,
            prr_ref_ext_cnt_active = :ref_ext_cnt_active
        where prr_k_id = :k_id;

      if (k_fact_closed_date is null and ref_cnt_active > 0 and ssp_date >= current_date - 1 and ki_date >= current_date - 1 and ki_note_id = 5) then
        update pdn_ref_recalc
        set prr_pdn_status = 'Кандидат: клиент не имеет КИ',
            prr_pdn_status_id = 6,
            prr_ref_cnt_list = :ref_cnt_list,
            prr_ref_our_active = :ref_our_active,
            prr_ref_our_cnt_active = :ref_our_cnt_active,
            prr_ref_ext_active = :ref_ext_active,
            prr_ref_ext_cnt_active = :ref_ext_cnt_active
        where prr_k_id = :k_id;

      if (k_fact_closed_date is null and ref_cnt_active > 0 and ssp_date >= current_date - 1 and ki_date >= current_date - 1 and ki_note_id = 9) then
        update pdn_ref_recalc
        set prr_pdn_status = 'Кандидат: КИ получена',
            prr_pdn_status_id = 5,
            prr_ref_cnt_list = :ref_cnt_list,
            prr_ref_our_active = :ref_our_active,
            prr_ref_our_cnt_active = :ref_our_cnt_active,
            prr_ref_ext_active = :ref_ext_active,
            prr_ref_ext_cnt_active = :ref_ext_cnt_active
        where prr_k_id = :k_id;
    end
  end
end
