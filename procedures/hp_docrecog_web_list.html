CREATE PROCEDURE HP_DOCRECOG_WEB_LIST(
    UUID D_STR_CODE)
RETURNS (
    RES_JSON D_STR_8100)
AS
declare variable err_text d_str_255;
declare variable c_id d_id;
declare variable c_citizen integer;
declare variable pk_id d_integer;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable obj_code d_str_code;
declare variable drm_name varchar(255);
declare variable drm_code varchar(255);
declare variable drm_api_code varchar(255);
declare variable drm_is_manual smallint;
declare variable drm_recog_2 d_smallint;
declare variable drm_has_page1 d_smallint;
declare variable drm_has_page2 d_smallint;
declare variable json_list d_str_8100;
declare variable json_array d_str_255;
declare variable need_recog d_smallint;
declare variable lang d_str_lang;
declare variable doc_no d_str_code;
declare variable need_date_begin d_smallint;
declare variable need_date_end d_smallint;
declare variable doc_date_begin d_date;
declare variable doc_date_end d_date;
declare variable doc_datetime d_datetime;
declare variable doc_detail1 d_str_255;
declare variable doc_detail2 d_str_255;
declare variable need_datetime d_smallint;
declare variable tmp_int d_integer;
declare variable can_attach d_smallint;
declare variable drm_id d_integer;
begin
  err_text = '';
  json_list = '';

  select obj_kind, obj_id
  from hp_uuid(:uuid)
  into obj_kind, obj_id;

  lang = f_get_lang_code(null, null);

  if (err_text = '' and obj_id is null)
  then err_text = f_lang(lang, 'Неверная ссылка');
  else obj_code = f_objkind(obj_kind, 'name');

  if (obj_code = 'edoc_sign') then
  begin
    select eds_from_c_id from edoc_sign
    where eds_id = :obj_id
    into c_id;

    if (not exists(select * from clients where c_id = :c_id)) then
      err_text = 'Клиент не найден';

    if (err_text = '') then
    begin
      select c_citizen
      from clients
      where c_id = :c_id
      into c_citizen;
      c_citizen = coalesce(c_citizen, 0);

      for select drm_id, pk_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, pk_id
      from docrecogmain, passkind, passdoctype, hp_citizen_passkind(:c_citizen)
      where drm_inactive = 0
        and drm_show_to_client = 1
        and res_pk_id = pk_id
        and pk_from_drm_id = drm_id
        and pk_id > 0
        and pk_from_pdt_id = pdt_id
      order by pdt_is_main_doc desc
      into drm_id, drm_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, pk_id
      do
      begin
        -- Сначала проверяем что документ не находится на оцифровке
        can_attach = 1;
        if (exists(
            select * from docrecoglist, actionorder
            where drl_from_drm_id = :drm_id
              and drl_from_c_id = :c_id
              and drl_status = 0
              and drl_from_acl_id = aco_id
              and aco_from_rd_id in (f_get_rd_id('ActionGroup.DocRecog_Client'), f_get_rd_id('ActionGroup.DocRecog_Client_OiVO'))
              and aco_status in (0, 1))) then
        begin
         can_attach = 0;
         doc_detail1 = f_lang(lang, 'Документ находится на стадии обработки');
         doc_detail2 = '';
        end
        else
        begin
          select cds_doc_no, cds_doc_date_begin, cds_doc_date_end,
            pdt_need_date_begin, pdt_need_date_end,
            iif(coalesce(cds_doc_blob_1,0) > 0 or coalesce(cds_doc_blob_2,0) > 0, 1, 0),
            cds_create_datetime
          from clientdocs, passkind, passdoctype
          where cds_from_c_id = :c_id
            and pk_id = :pk_id
            and cds_is_delete = 0
            and cds_from_pk_id = pk_id
            and pk_from_pdt_id = pdt_id
          order by cds_doc_date_begin desc, cds_create_datetime desc
          rows 1
          into doc_no, doc_date_begin, doc_date_end, need_date_begin, need_date_end,
            need_datetime, doc_datetime;
  
          doc_detail1 = f_lang(lang, 'Документ отсутствует. Вы можете его закрепить.');
          doc_detail2 = '';
  
          if (row_count > 0) then
          begin
            doc_no = coalesce(doc_no, '');
            if (doc_no = '') then doc_no = f_lang(lang, 'б/н');
            tmp_int = char_length(doc_no);
  
            if (tmp_int >= 5) then -- Маскируем номер документа, для длинных
              doc_no = left(doc_no, tmp_int - 5) || '***' || right(doc_no, 2);
      
            if (need_date_begin = 1) then
              if (need_date_end = 1) then
                doc_detail1 = f_format(f_lang(lang,'№ %1, выдан: %2, действует до: %3'), doc_no, f_datetostr(doc_date_begin), f_datetostr(doc_date_end));
              else
                doc_detail1 = f_format(f_lang(lang,'№ %1, выдан: %2'), doc_no, f_datetostr(doc_date_begin));
            else
              doc_detail1 = f_format('№ %1', doc_no);
      
            if (need_datetime = 1) then
              doc_detail2 = f_format('%1: %2', f_lang(lang, 'Добавлен'), f_datetimetostr(doc_datetime));
          end

        select drm_has_page1, drm_has_page2
        from hp_get_drm_has_pages(:pk_id)
        into :drm_has_page1, :drm_has_page2;
        end

        json_list = json_list || iif(json_list <> '', ',', '') ||
          '{' ||
           f_json_make_pair('code', drm_code, 'str') || ',' ||
           f_json_make_pair('name', drm_name, 'str') || ',' ||
           f_json_make_pair('api_code', drm_api_code, 'str') || ',' ||
           f_json_make_pair('attachable', can_attach, 'num') || ',';

           -- Массив details
           json_array = '';
           if (doc_detail1 <> '') then
             json_array = f_format('"%1"', doc_detail1);

           if (doc_detail2 <> '') then
             json_array = json_array || iif(json_array <> '', ',', '') || f_format('"%1"', doc_detail2);
           json_list = json_list || '"details":[' || json_array || '],';

           json_array = '';
           if (drm_has_page1 = 1) then
           begin
             need_recog = iif(drm_is_manual = 0, 1, 0);
             json_array =
               '{' ||
                 f_json_make_pair('page', 1, 'num') || ',' ||
                 f_json_make_pair('need_recog', need_recog, 'num') ||
                '}';
           end
           if (drm_has_page2 = 1) then
           begin
             need_recog = iif(drm_is_manual = 0 and drm_recog_2 = 1, 1, 0);
             json_array = json_array || iif(json_array <> '', ',', '') ||
               '{' ||
                 f_json_make_pair('page', 2, 'num') || ',' ||
                 f_json_make_pair('need_recog', need_recog, 'num') ||
                '}';
           end
           json_list = json_list || '"pages":[' || json_array || ']}';
      end
    end
  end

  if (err_text <> '') then
    res_json = '{' || f_json_make_pair('err_text', err_text, 'str') || '}';
  else
  res_json =
    '{' ||
     f_json_make_pair('c_id', c_id, 'num') || ',' ||
     '"doc_list":['||json_list||']}';

  Suspend;
end
