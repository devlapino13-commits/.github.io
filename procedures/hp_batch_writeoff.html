CREATE PROCEDURE HP_BATCH_WRITEOFF(
    REP_DATE DATE,
    WRITEOFF_REASON INTEGER = null,
    MEMO STR_255 = null)
AS
declare variable un_id id_type;
declare variable reg_id integer;
declare variable k_id id_type;
declare variable ok_ol_date date;
declare variable ol_id id_type;
declare variable ol_repay_amount curr_type;
declare variable error_text str_255;
declare variable writeoff_operextra integer;
declare variable is_ok integer;
declare variable db_id integer;
declare variable writeoff_protocol d_str_100;
begin
  un_id = (select un.un_id
           from username un
           where un.un_name = 'IGUSEV');
  db_id = f_get_db_id();
  memo = trim(coalesce(memo, ''));

  /*
  if (writeoff_reason is null) then  -- если причина списания не указана
  begin
      if (db_id = 81001) then
        writeoff_reason = f_get_rd_id('Reason_WriteOff.Delinq360');
      else if (db_id = 94401) then
        writeoff_reason = f_get_rd_id('Reason_WriteOff.OldWriteOff');
  end */

  if ((memo = '') and
      (db_id in (81001, 94401))) then
    memo = 'Списание займа по истечению просрочки более 360 дней';

  -- беру список кредитов на списание
  for select tt.k_id, tt.region_id, tt.writeoff_operextra, tt.writeoff_reason_id, tt.writeoff_protocol
      from kont_for_writeoff tt
      where tt.writeoff_date = :rep_date and
            tt.is_ok = 0 rows 1
      into :k_id, :reg_id, :writeoff_operextra, :writeoff_reason, :writeoff_protocol
  do
  begin
    if (db_id in (39801, 41701) and
        :writeoff_protocol is not null and
        :memo = '') then
      :memo = :writeoff_protocol;

    is_ok = 1; -- все ОК

    if (exists(select *
               from kontstate ks
               where ks.ks_state = 2 and
                     ks.ks_ol_is_delete = 0 and
                     :rep_date between ks.ks_date_begin and ks.ks_date_end and
                     ks.ks_from_k_id = :k_id)) then
      is_ok = -1; -- не активный

    if (is_ok = 1) then
    begin
      if (exists(select *
                 from kontstate ks
                 where ks.ks_state = 2 and
                       ks.ks_ol_is_delete = 0 and
                       ks.ks_date_begin >= :rep_date and
                       ks.ks_from_k_id = :k_id)) then
        is_ok = -2; -- не активный
    end
    /*if (is_ok = 1) then
        begin

            if (exists (select *
                        from oksummary oks
                        where oks.oks_from_k_id = :k_id
                        and :rep_date between oks.oks_date_begin and oks.oks_date_end
                        and :rep_date - oks.oks_delinq_date < 361)) then
                is_ok = -3; -- просрочка меньше 361 день.
         end */
    if (is_ok = 1) then
    begin
      select max(ok.ok_ol_date)
      from operkred ok
      where ok.ok_from_k_id = :k_id and
            ok.ok_ol_is_active = 1
      into :ok_ol_date;

      -- определяю, какой датой сажать операцию
      --oper_date = iif(ok_ol_date > rep_date, ok_ol_date, rep_date);
      if (ok_ol_date > rep_date) then
        is_ok = -4; -- есть операция более поздней датой
    end
    if (is_ok = 1) then
    begin
      for select ol_id, ol_repay_amount, error_text
          from hp_do_repayment(:k_id, 0, :rep_date, -777, 1, 1, 1, 1, 1, 1, 1, :reg_id, 3, null, 0, :un_id, null, :memo, null, :writeoff_reason, -777, :writeoff_operextra)
          into :ol_id, :ol_repay_amount, error_text
      do
      begin
      end
    end

    update kont_for_writeoff tt
    set tt.is_ok = :is_ok
    where tt.k_id = :k_id and
          tt.writeoff_date = :rep_date; -- RFCKG-4135
  end
end
