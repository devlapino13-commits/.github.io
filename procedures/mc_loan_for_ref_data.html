CREATE PROCEDURE MC_LOAN_FOR_REF_DATA(
    MCL_ID D_ID,
    K_ID D_ID)
RETURNS (
    K_DISB_AMOUNT D_CURRENCY,
    K_DISB_DATE D_DATE,
    K_CLOSED_DATE D_DATE,
    EDOC_LOAN_DEBT_VISIBLE D_BOOL,
    EDOC_LOAN_DEBT_EDS_ID D_ID,
    EDOC_LOAN_DEBT_TXT D_STR_255,
    EDOC_LOAN_DEBT_STATUS D_SMALLINT,
    ADDITIONAL_INFO D_STR_255)
AS
declare variable C_ID D_ID;
declare variable EDL_BLANK_DATE D_DATE;
declare variable K_KONTRACT_NO D_STR_20;
declare variable EDOC_LOAN_DEBT_EDL_ID D_ID;
begin
--  date_end = current_date;
  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  select k_amount_approved, k_disb_date, k_fact_closed_date, k_kontract_no
  from kontract
  where k_id = :k_id
  into k_disb_amount, k_disb_date, k_closed_date, k_kontract_no;

  edoc_loan_debt_edl_id = null;
  edoc_loan_debt_eds_id = null;
  select edl_id, edl_blank_date
  from edoc_list, edoc_cfg
  where edl_from_edc_id = edc_id and
        edl_status >= 0 and
        edl_blank_status = 1 and
        edc_code = 'loan.debt' and
        edl_object_kind = 9 and
        edl_object_id = :k_id
   order by edl_create_datetime desc
   rows 1
   into edoc_loan_debt_edl_id, edl_blank_date;

  edoc_loan_debt_visible = 0;
  edoc_loan_debt_txt = '';
  if (edoc_loan_debt_edl_id is not null) then
  begin
    edoc_loan_debt_visible = 1;
    edoc_loan_debt_txt = 'Справка о полном исполнении договора займа №'||k_kontract_no||' от'||f_datetostr(edl_blank_date);
    edoc_loan_debt_eds_id = -edoc_loan_debt_edl_id;
    edoc_loan_debt_status = 1;
  end 

  additional_info = '';

  suspend;
end
