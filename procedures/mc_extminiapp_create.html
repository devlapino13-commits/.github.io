CREATE PROCEDURE MC_EXTMINIAPP_CREATE(
    MCL_ID D_ID,
    AMOUNT D_CURRENCY,
    PERIOD D_INTEGER,
    OFFICE_ID D_INTEGER,
    PR_ID D_INTEGER = null)
RETURNS (
    ERROR_TEXT D_STR_255)
AS
declare variable c_id id_type;
declare variable delinq_days d_integer;
declare variable last_office_id id_type;
declare variable db_id id_type;
begin
    -- Регистрация новой онлайн-заявки через КМП
    db_id = f_get_db_id();
    error_text = '';

    mcl_id = coalesce(mcl_id, 0);
    if (mcl_id <= 0) then
        exception e_default 'parametr "mcl_id" has invalid value';

    amount = coalesce(amount, 0);
    if (amount <= 0) then
        exception e_default 'parametr "amount" has invalid value';

    period = coalesce(period, 0);
    if (period <= 0) then
        exception e_default 'parametr "period" has invalid value';

    office_id = coalesce(office_id, 0);
    if (office_id <= 0) then
        exception e_default 'parametr "office_id" has invalid value';


    select mcl.mcl_from_c_id
    from mc_login mcl
    where mcl.mcl_id = :mcl_id
    into :c_id;

    --по запросу от Зары добавили проверку на уровне подачи заявки. jira4823
    select coalesce(ma_result_txt, '')
    from mc_chec_extminiapp(:c_id, :mcl_id)
    into :error_text;

    -- RFCRU-5310
    if (:office_id = 1031600 and :db_id = 81001) then
        error_text = 'В офисе Всеволожск временно нельзя создать заявку';

    if (error_text = '') then
    begin
        if (db_id in (39801, 41701)) then
        begin
            last_office_id = 0;
            select first 1 ks.ks_region_id
            from kontstate ks
            inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
            where kc.kc_from_c_id = :c_id
                and ks.ks_ol_is_delete = 0
                and ks.ks_state in (1, 2)
                and current_date between ks.ks_date_begin and ks.ks_date_end
            order by ks.ks_date_begin desc
            into :last_office_id;
        
            if (coalesce(last_office_id,0) <> 0) then
                office_id = last_office_id;
        end

        insert into extminiapp (ema_from_c_id, ema_amount, ema_period, ema_office_id, ema_status, ema_from_tkt_id, ema_result_rd_id,
            ema_from_mcl_id, ema_from_un_id, ema_from_pr_id)
        values (:c_id, :amount, :period, :office_id, 0, null, 0, :mcl_id, 0, :pr_id);

    end

    suspend;
end
