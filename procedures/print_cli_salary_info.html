CREATE PROCEDURE PRINT_CLI_SALARY_INFO(
    C_ID D_ID)
RETURNS (
    EIL_DATE D_DATE,
    C_NAME_FIO D_STR_255,
    C_INN D_STR_20,
    PAYER D_STR_255,
    NUM_SF D_STR_255,
    PAYER_INN D_STR_255,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    SALARY D_INTEGER,
    MESSAGE_TEXT D_STR_255)
AS
declare variable json d_blob_text;
declare variable counter d_integer;
declare variable json_one d_blob_text;
declare variable json_count d_integer;
declare variable bls_id d_id;
begin
  message_text = '';
  select eil.eil_date, eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id and
        eic.eic_code = 'get_salary_info'
  where eil.eil_from_c_id = :c_id and
        eil.eil_is_json = 1 and
        eil.eil_status = 2
  order by eil.eil_id desc
  rows 1
  into eil_date, bls_id;

  select c_name_fio, c_inn
  from clients
  where c_id = :c_id
  into c_name_fio, c_inn;

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into :json;

  json_count = f_json_blob_get_count(:json, 'WorkPeriods.WorkPeriodWithSumDto');
  counter = 0;
  while (json_count > counter) do
  begin
    json_one = f_json_blob_get_value(:json, f_format('WorkPeriods.WorkPeriodWithSumDto[%1]', :counter), '');
    payer = f_json_str_get_value(:json_one, 'Payer', '');
    num_sf = f_json_str_get_value(:json_one, 'NumSF', '');
    payer_inn= f_json_str_get_value(:json_one, 'INN', '');
    date_begin = cast(f_json_str_get_value(:json_one, 'DateBegin', '31.12.9999') as date);
    date_end = cast(f_json_str_get_value(:json_one, 'DateEnd', '31.12.9999') as date);
    salary = f_json_str_get_value(:json_one, 'Salary', 0);
    counter = counter + 1;
    suspend;
  end
end
