CREATE PROCEDURE EPASS_REQ_MISSED_LIST
RETURNS (
    ER_ID D_INTEGER,
    ER_INN D_STR_20,
    ER_PASSWORD D_STR_20)
AS
declare variable er_timestamp d_datetime;
declare variable lifetime d_integer;
declare variable er_status d_smallint;
begin
  lifetime = f_get_const('epass_pass_lifetime');
  for select er.er_id, er.er_timestamp, er.er_inn, er.er_password, er.er_status
      from epass_req er
      where er.er_status in (0, 1)
      into :er_id, :er_timestamp, :er_inn, :er_password, :er_status

  do
  begin
    if (:er_status <> 1 and
        datediff(second, :er_timestamp, current_timestamp) < lifetime) then
    begin
      update epass_req
      set er_status = 1
      where (er_id = :er_id);
      suspend;
    end
    else
      update epass_req
      set er_status = -1
      where (er_id = :er_id);
  end
end
