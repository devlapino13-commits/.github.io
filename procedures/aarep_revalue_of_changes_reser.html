CREATE PROCEDURE AAREP_REVALUE_OF_CHANGES_RESER(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO STR_20,
    K_DISB_DATE DATE,
    K_KREDLINE_END DATE,
    RO_1_BEGIN CURR_TYPE,
    RO_2_BEGIN CURR_TYPE,
    RO_3_BEGIN CURR_TYPE,
    VBS_BEGIN CURR_TYPE,
    DELINQ_DAY_BEGIN INTEGER,
    RO_1_END CURR_TYPE,
    RO_2_END CURR_TYPE,
    RO_3_END CURR_TYPE,
    VBS_END CURR_TYPE,
    DELINQ_DAY_END INTEGER,
    GOTO_RO1 CURR_TYPE,
    GOTO_RO1_OF_RO2 CURR_TYPE,
    GOTO_RO2_OF_RO1 CURR_TYPE,
    GOTO_RO2 CURR_TYPE,
    GOTO_RO3_OF_RO1 CURR_TYPE,
    GOTO_RO3_OF_RO2 CURR_TYPE,
    GOTO_RO3 CURR_TYPE,
    RECOVERY_RO1 CURR_TYPE,
    RECOVERY_RO2 CURR_TYPE,
    RECOVERY_RO3 CURR_TYPE,
    INCREASE_RO1 CURR_TYPE,
    INCREASE_RO2 CURR_TYPE,
    INCREASE_RO3 CURR_TYPE,
    NEW_RO1 CURR_TYPE,
    RETIREMENT_RO3 CURR_TYPE,
    KC_ID_STR D_STR_CODE,
    DISPOSAL_RO1 CURR_TYPE,
    DISPOSAL_RO2 CURR_TYPE,
    DISPOSAL_RO3 CURR_TYPE,
    CLIENT_TYPE D_STR_20)
AS
declare variable K_ID_PRIOR ID_TYPE;
declare variable OKS_ECL_VBS decimal(18,4);
declare variable OKS_ECL_ACC_KIND smallint;
declare variable FULL_CODE_DT D_STR_255;
declare variable FULL_CODE_KT D_STR_255;
declare variable RES_AMOUNT CURR_TYPE;
declare variable REP_DATE_BEGIN_YESTERDAY date;
declare variable ACC_ADD_CODE_DT D_STR_CODE;
declare variable ACC_ADD_CODE_KT D_STR_CODE;
declare variable KOB_FROM_KC_ID D_ID;
declare variable KOB_ACC_DT D_STR_CODE;
declare variable KOB_ACC_KT D_STR_CODE;
declare variable KOB_SUMMA CURR_TYPE;
declare variable KOB_ACC_DT_PRIOR D_STR_CODE;
declare variable KOB_ACC_KT_PRIOR D_STR_CODE;
declare variable KC_ID decimal(18,0);
declare variable K_ID decimal(18,0);
declare variable OKS_CLOSE_STATE smallint;
declare variable PR_CODE D_STR_100;
declare variable KC_CLIENT_TYPE D_INTEGER;
declare variable KC_IS_MSP D_SMALLINT;
begin
  kob_acc_dt_prior = 0;
  kob_acc_kt_prior = 0;

  for
  select kob_from_kc_id, kob_acc_dt, kob_acc_kt, sum(kob_summa)
  from kn_operbook
  where kob_ol_is_active = 1
  and kob_ol_date between :rep_date_begin and :rep_date_end
--  and kob_from_kc_id in (528485, 528486)   ---------------------------------
  group by kob_from_kc_id, kob_acc_dt, kob_acc_kt
  into kob_from_kc_id, kob_acc_dt, kob_acc_kt, kob_summa
  do
  begin
    if (kob_acc_dt_prior <> kob_acc_dt) then
    begin
      kob_acc_dt_prior = kob_acc_dt;
      select acc_add_code from ref_account
      where acc_id = :kob_acc_dt
      into acc_add_code_dt;
    end
    if (kob_acc_kt_prior <> kob_acc_kt) then
    begin
      kob_acc_kt_prior = kob_acc_kt;
      select acc_add_code from ref_account
      where acc_id = :kob_acc_kt
      into acc_add_code_kt;
    end

    insert into temp_kc_obor (kc_id, summa, acc_add_code_dt, acc_add_code_kt)
    values (:kob_from_kc_id, :kob_summa, :acc_add_code_dt, :acc_add_code_kt);
  end

  k_id_prior = 0;
  rep_date_begin_yesterday = rep_date_begin - 1;
  for
  select distinct kc_id, ks_from_k_id, kc_client_type, kc_is_msp
  from kontstate, knclient
  where ks_ol_is_delete = 0
  and ks_state = 1
--  and ks_date_end >= :rep_date_begin
  and ks_date_end >= :rep_date_begin_yesterday
  and ks_date_begin <= :rep_date_end
  and ks_from_k_id = kc_from_k_id
  and kc_amount_approved > 0
--  and kc_id in (528485, 528486) ---------------------------------------
  into kc_id, k_id, kc_client_type, kc_is_msp
  do
  begin
    kc_id_str = kc_id;
    if (k_id_prior <> k_id) then
    begin
      k_id_prior = k_id;
      select k_kontract_no, k_disb_date, k_kredline_end, p.pr_code
      from kontract k, product p
      where k_id = :k_id and
      p.pr_id = k.k_from_pr_id
      into k_kontract_no, k_disb_date, k_kredline_end, pr_code;
    end

    if (pr_code containing '.selfemployed.') then
    begin
      client_type = 'СЗ';
    end else
    begin
      kc_is_msp = coalesce(kc_is_msp, 0);

      if (kc_client_type = 1) then
      begin
        client_type = 'ЮЛ';
      end else
      if (kc_client_type = 2) then
      begin
        if (kc_is_msp = 1) then client_type = 'ИП-МСП';
          else client_type = 'ИП-неМСП';
      end else
      begin
        client_type = 'ФЛ';
      end
    end


--  Расчет РО на начало периода по корзинам
    oks_ecl_vbs = 0;
    oks_ecl_acc_kind = 0;
    vbs_begin = 0;
    delinq_day_begin = 0;

    select oks_ecl_vbs,
           oks_ecl_acc_kind,
           oks_bal_esp_end,
           oks_date_begin - oks_delinq_date
    from kn_oksummary
    where oks_from_kc_id = :kc_id
    and oks_date_begin = :rep_date_begin_yesterday
    into oks_ecl_vbs, oks_ecl_acc_kind, vbs_begin, delinq_day_begin;

    if (delinq_day_begin is null) then delinq_day_begin = 0;
    if (delinq_day_begin < 0) then delinq_day_begin = 0;

    ro_1_begin = 0;
    ro_2_begin = 0;
    ro_3_begin = 0;

    if (oks_ecl_acc_kind = 1) then ro_1_begin = oks_ecl_vbs;
    if (oks_ecl_acc_kind = 2) then ro_2_begin = oks_ecl_vbs;
    if (oks_ecl_acc_kind = 3) then ro_3_begin = oks_ecl_vbs;

--  Расчет РО на конец периода по корзинам
    oks_ecl_vbs = 0;
    oks_ecl_acc_kind = 0;
    vbs_end = 0;
    delinq_day_end = 0;

    select oks_ecl_vbs,
           oks_ecl_acc_kind,
           oks_bal_esp_end,
           oks_date_end - oks_delinq_date,
           oks_close_state
    from kn_oksummary
    where oks_from_kc_id = :kc_id
    and oks_date_begin = :rep_date_end
    into oks_ecl_vbs, oks_ecl_acc_kind, vbs_end, delinq_day_end, oks_close_state;

    if (oks_close_state = 1) then
    begin
--    письмо от e.grabovskaya@molbulak.com  18.09.2019
--    1) Займы, погашенные в дату окончания периода отчета имеют в отчете остатки по РО и ВБС, а должны показывать нули
      oks_ecl_vbs = 0;
      vbs_end = 0;
    end 

    if (delinq_day_end is null) then delinq_day_end = 0;
    if (delinq_day_end < 0) then delinq_day_end = 0;

    ro_1_end = 0;
    ro_2_end = 0;
    ro_3_end = 0;

    if (oks_ecl_acc_kind = 1) then ro_1_end = oks_ecl_vbs;
    if (oks_ecl_acc_kind = 2) then ro_2_end = oks_ecl_vbs;
    if (oks_ecl_acc_kind = 3) then ro_3_end = oks_ecl_vbs;

    goto_ro1 = 0;
    goto_ro2_of_ro1 = 0;
    goto_ro3_of_ro1 = 0;
    goto_ro3_of_ro2 = 0;
    recovery_ro1 = 0;
    recovery_ro2 = 0;
    recovery_ro3 = 0;
    increase_ro1 = 0;
    increase_ro2 = 0;
    increase_ro3 = 0;
    retirement_ro3 = 0;

    disposal_ro1 = 0;
    disposal_ro2 = 0;
    disposal_ro3 = 0;

    for
    select acc_add_code_dt,
           acc_add_code_kt,
           summa
    from temp_kc_obor
    where kc_id = :kc_id
    into acc_add_code_dt,
         acc_add_code_kt,
         res_amount
    do
    begin
--    Переход за период в Корзину 1
      if (acc_add_code_dt like '4__10 (2%)' and acc_add_code_kt like '4__10 (1%)') then goto_ro1 = goto_ro1 + res_amount;
--    Переход за период в Корзину 2
      if (acc_add_code_dt like '4__10 (1%)' and acc_add_code_kt like '4__10 (2%)') then goto_ro2_of_ro1 = goto_ro2_of_ro1 - res_amount;
--    Переход за период в Корзину 3
      if (acc_add_code_dt like '4__10 (1%)' and acc_add_code_kt like '4__10 (3%)') then goto_ro3_of_ro1 = goto_ro3_of_ro1 - res_amount;
      if (acc_add_code_dt like '4__10 (2%)' and acc_add_code_kt like '4__10 (3%)') then goto_ro3_of_ro2 = goto_ro3_of_ro2 - res_amount;
--    Восстановление РО в связи с погашением
      if (acc_add_code_dt like '4__10 (1%)' and acc_add_code_kt like '71201 (3812%)') then recovery_ro1 = recovery_ro1 - res_amount;
      if (acc_add_code_dt like '4__10 (2%)' and acc_add_code_kt like '71201 (3812%)') then recovery_ro2 = recovery_ro2 - res_amount;
      if (acc_add_code_dt like '4__10 (3%)' and acc_add_code_kt like '71201 (3812%)') then recovery_ro3 = recovery_ro3 - res_amount;
--    Увеличение РО внутри корзины
      if (acc_add_code_dt like '71202 (4812%)' and acc_add_code_kt like '4__10 (1%)') then increase_ro1 = increase_ro1 + res_amount;
      if (acc_add_code_dt like '71202 (4812%)' and acc_add_code_kt like '4__10 (2%)') then increase_ro2 = increase_ro2 + res_amount;
      if (acc_add_code_dt like '71202 (4812%)' and acc_add_code_kt like '4__10 (3%)') then increase_ro3 = increase_ro3 + res_amount;
--    Выбытие РО ввиду списания / новое название столбца 'Выбытие РО от списания'
      if (acc_add_code_dt like '4__10 (3%)' and
          (acc_add_code_kt like '61217 (1%)' or
           acc_add_code_kt like '4__01%' or
           acc_add_code_kt like '4__02%'
          )) then retirement_ro3 = retirement_ro3 - res_amount;
      if (acc_add_code_kt like '4__10 (3%)' and
          acc_add_code_dt like '61217 (1%)') then retirement_ro3 = retirement_ro3 + res_amount;

--    Выбытие РО в связи с погашением
--    Дт                        Кт                        С каким знаком отображается сумма проводки в отчете
--    4хх10.810.00.1х.хххххххх  61217.810.00.0х.хххххххх  « - »
--    61217.810.00.0х.хххххххх  4хх10.810.00.1х.хххххххх  « + »
      if (acc_add_code_dt like '4__10 (1%)' and  acc_add_code_kt like '61217 (0%)') then disposal_ro1 = disposal_ro1 - res_amount;
      if (acc_add_code_kt like '4__10 (1%)' and  acc_add_code_dt like '61217 (0%)') then disposal_ro1 = disposal_ro1 + res_amount;
--    Дт                        Кт                        С каким знаком отображается сумма проводки в отчете
--    4хх10.810.00.2х.хххххххх  61217.810.00.0х.хххххххх  « - »
--    61217.810.00.0х.хххххххх  4хх10.810.00.2х.хххххххх  « + »
      if (acc_add_code_dt like '4__10 (2%)' and  acc_add_code_kt like '61217 (0%)') then disposal_ro2 = disposal_ro2 - res_amount;
      if (acc_add_code_kt like '4__10 (2%)' and  acc_add_code_dt like '61217 (0%)') then disposal_ro2 = disposal_ro2 + res_amount;
--    Дт                        Кт                        С каким знаком отображается сумма проводки в отчете
--    4хх10.810.00.3х.хххххххх  61217.810.00.0х.хххххххх  « - »
--    61217.810.00.0х.хххххххх  4хх10.810.00.3х.хххххххх  « + »
      if (acc_add_code_dt like '4__10 (3%)' and  acc_add_code_kt like '61217 (0%)') then disposal_ro3 = disposal_ro3 - res_amount;
      if (acc_add_code_kt like '4__10 (3%)' and  acc_add_code_dt like '61217 (0%)') then disposal_ro3 = disposal_ro3 + res_amount;
    end
    goto_ro1_of_ro2 = -goto_ro1;
    goto_ro2 = -goto_ro2_of_ro1;
    goto_ro3 = -(goto_ro3_of_ro1 + goto_ro3_of_ro2);
--  если остаток на 4хх10(1х) на начало периода отчета >0
    new_ro1 = 0;
    if (k_disb_date >= rep_date_begin) then
    begin
      new_ro1 = increase_ro1;
      increase_ro1 = 0;
    end 

    suspend;
  end
end
