CREATE PROCEDURE FM_HTML_LOYAL_CLIENT(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable str varchar(1024);
declare variable c_name_fio varchar(100);
declare variable db_id integer;
declare variable cell_no d_str_code;
declare variable ag_date_begin d_date;
declare variable sum_calc d_currency;
declare variable sum_repay d_currency;
declare variable sum_tax d_currency;
declare variable sum_balance d_currency;
declare variable str_orig d_str_long;
declare variable str2 d_str_255;
declare variable str3 d_str_255;
declare variable c_id d_id;
declare variable ogr_id d_id;
declare variable ogr_ol_date d_date;
declare variable clk_id d_id;
declare variable clk_status d_integer;
declare variable ogr_amount d_currency;
declare variable sum_writeoff d_currency;
begin
  if (obj_kind is null) then
    exception e_default 'obj_kind is null';

  if (f_objkind(obj_kind, 'code') not in ('clients', 'agreements')) then
    exception e_default 'Incorrect obj_kind: ' || obj_kind;

  db_id = f_get_db_id();

  if (f_objkind(obj_kind, 'code') = 'clients') then
  begin
    c_id = obj_id;
    obj_id = null;
    select ag_id, ag_date_begin
    from agreements
    where ag_object_kind = f_objkind(:obj_kind, 'id') and
          ag_object_id = :c_id and
          ag_status <> -1
    order by ag_id desc
    rows 1
    into obj_id, ag_date_begin;

    obj_kind = f_objkind('agreements', 'id');
  end
  else
  begin
    select ag_object_id, ag_date_begin
    from agreements
    where ag_id = :obj_id
    into c_id, ag_date_begin;

    if (c_id is null) then
      exception e_default 'obj_id not found in agreements: ' || obj_id;
  end

  select c_name_fio, f_cont_info('clients', c_id, 'cell_main', 'last')
  from clients
  where c_id = :c_id
  into c_name_fio, cell_no;

  res_text = (select res_style
              from fm_html_styles) || '<body>';

  res_text = BLOB_APPEND(res_text, f_format('<h1 class="green_text">%1<br>%2</h1>',
                                  iif(db_id <> 94401, 'Детали по программе лояльности',
                                                      'Sadiqlik proqramı haqqında məlumat'), c_name_fio));

  res_text = BLOB_APPEND(res_text, f_format('%1: %2', iif(db_id <> 94401, 'Телефон', 'Tel.'),
                                  iif(cell_no <> '', f_uri_phone(cell_no, 0, c_id), '---')));
  res_text = BLOB_APPEND(res_text, '<br>');

  if (ag_date_begin is not null)
  then str = f_datetostr(ag_date_begin);
  else str = iif(db_id <> 94401, 'Нет соглашения', 'Razılaşma yoxdur');

  res_text = BLOB_APPEND(res_text, f_format('%1: <b>%2</b>',
                         iif(db_id <> 94401, 'Дата вступления в программу', 'Proqrama qoşulma tarixi'),
                         str));

  res_text = BLOB_APPEND(res_text, '<br><br>');
  res_text = BLOB_APPEND(res_text, iif(db_id <> 94401, 'Баллы с начала вступления', 'Girişin əvvəlindən yığılmış ballar') || ':<br>');

  sum_calc = 0;
  sum_repay = 0;
  sum_tax = 0;
  sum_balance = 0;
  sum_writeoff = 0;

  select sum(iif(ogr_sign = 1, ogr_amount, 0.0)),
         sum(iif(ogr_sign = -1 and ogr_object_id is null, ogr_amount, 0)), -- выплаченные баллы за все периоды
         sum(iif(ogr_sign = -1 and ogr_object_id is not null, ogr_amount + ogr_tax_fl, 0)), -- списано баллы за все периоды
         sum(iif(ogr_sign = -1 and ogr_object_id is null, ogr_tax_fl, 0.0)),
         sum(ogr_amount * ogr_sign * f_loyal(ogr_from_ol_id, current_date) + ogr_tax_fl * ogr_sign * f_loyal(ogr_from_ol_id, current_date))
  from operagr
  where ogr_from_ag_id = :obj_id and
        ogr_ol_is_active = 1
  into sum_calc, sum_repay, sum_writeoff, sum_tax, sum_balance;

  if (sum_calc is null) then sum_calc = 0;
  if (sum_repay is null) then sum_repay = 0;
  if (sum_writeoff is null) then sum_writeoff = 0;
  if (sum_tax is null) then sum_tax = 0;
  if (sum_balance is null) then sum_balance = 0;

  res_text = blob_append(res_text, f_format('%1: <b>%2</b>',
                         iif(db_id <> 94401, 'Начислены', 'Hesablanıb'),
                         round(sum_calc, 0)) || '<br>');

  res_text = blob_append(res_text, f_format('%1: <b>%2</b>',
                         iif(db_id <> 94401, 'Выплачены', 'Ödənişlidir'),
                         round(sum_repay, 0)) || '<br>');


  res_text = blob_append(res_text, f_format('%1: <b>%2</b>',
                         iif(db_id <> 94401, 'Удержан НДФЛ', 'Tutulan vergi'),
                         round(sum_tax, 0)) || '<br>');

  res_text = blob_append(res_text, f_format('%1: <b>%2</b>',
                         iif(db_id <> 94401, 'Списаны баллы', 'Ballar silindi'),
                         round(sum_writeoff, 0)) || '<br>');

  res_text = blob_append(res_text, f_format('%1: <b>%2</b>',
                         iif(db_id <> 94401, 'Остаток', 'Qalanı'),
                         round(sum_balance, 0)) || '<br>');

  sum_calc = 0;
  sum_repay = 0;
  sum_tax = 0;
  sum_balance = 0;
  sum_writeoff = 0;
  ogr_id = null;

  select ogr_id, ogr_ol_date
  from operagr
  where ogr_from_ag_id = :obj_id and
        ogr_ol_is_active = 1 and
        ogr_sign = -1
  order by ogr_id desc
  rows 1
  into ogr_id, ogr_ol_date;

  if (ogr_id is not null) then
  begin
    select sum(iif(ogr_sign = 1, ogr_amount, 0.0)),
           sum(iif(ogr_sign = -1 and ogr_object_id is null, ogr_amount, 0)), -- выплаченные баллы за все периоды
           sum(iif(ogr_sign = -1 and ogr_object_id is not null, ogr_amount + ogr_tax_fl, 0)), -- списано баллы за все периоды
           sum(iif(ogr_sign = -1 and ogr_object_id is null, ogr_tax_fl, 0.0))
    from operagr
    where ogr_from_ag_id = :obj_id and
          ogr_ol_is_active = 1 and
          ogr_ol_date >= :ogr_ol_date and
          ((ogr_ol_date = :ogr_ol_date and ogr_id > :ogr_id) or
            ogr_ol_date > :ogr_ol_date)
    into sum_calc, sum_repay, sum_writeoff, sum_tax;

    if (sum_calc is null) then sum_calc = 0;
    if (sum_repay is null) then sum_repay = 0;
    if (sum_writeoff is null) then sum_writeoff = 0;
    if (sum_tax is null) then sum_tax = 0;

    res_text = BLOB_APPEND(res_text, iif(db_id <> 94401, 'Баллы с последней выплаты', 'Son ödəmədən sonra ballar'));
    res_text = BLOB_APPEND(res_text, '<br>');

    res_text = BLOB_APPEND(res_text, f_format('%1: <b>%2</b>',
                           iif(db_id <> 94401, 'Начислены', 'Hesablanıb'),
                           round(sum_calc, 0)) || '<br>');
  
    res_text = BLOB_APPEND(res_text, f_format('%1: <b>%2</b>',
                           iif(db_id <> 94401, 'Выплачены', 'Ödənişlidir'),
                           round(sum_repay, 0)) || '<br>');
  
  
    res_text = BLOB_APPEND(res_text, f_format('%1: <b>%2</b>',
                           iif(db_id <> 94401, 'Удержан НДФЛ', 'Tutulan vergi'),
                           round(sum_tax, 0)) || '<br>');

    res_text = blob_append(res_text, f_format('%1: <b>%2</b>',
                           iif(db_id <> 94401, 'Списаны баллы', 'Ballar silindi'),
                           round(sum_writeoff, 0)) || '<br>');
  end

  res_text = BLOB_APPEND(res_text, '<br>');
  res_text = BLOB_APPEND(res_text, f_format('<h2>%1</h2>', iif(db_id <> 94401, 'Друзья клиента', 'Müştəri dostlar')));

  -- список друзей
  res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="50%">
       <col width="30%">
       <col width="20%">
       <tr>' || f_format('<th>%1</th><th>%2</th><th>%3</th>',
                         iif(db_id <> 94401, 'Имя', 'Adı'),
                         iif(db_id <> 94401, 'Статус', 'Status'),
                         iif(db_id <> 94401, 'Балл', 'Bal')));

  str_orig = '<tr>' ||
             '<td align="left">%1</td>' ||
             '<td align="left">%2</td>' ||
             '<td align="right">%3</td>' ||
             '</tr>';

  for select clk_id,
             clk_link_name || '<br>' || f_cont_format(clk_link_cell_no, 'cell', 'href'),
             clk_status,
             trim(
             case clk_status
                    when -2 then 'отказ' -- хмурое лицо, отказался
                    when -1 then 'перезвон' -- флажок, перезвонить
                    when 0 then 'новый' -- телефон, новый контакт
                    when 1 then 'интерес' -- улыбка, заинтересован
                    when 2 then 'выдан' -- галочка, выдан
             end)
  from clientlink
  where clk_from_c_id = :c_id
  order by clk_id
  into :clk_id, :str, :clk_status, str2
  do
  begin
    str2 = (case clk_status
                    when -2 then '&#9785;' -- хмурое лицо, отказался
                    when -1 then '&#9872;' -- флажок, перезвонить
                    when 0 then '&#9990;' -- телефон, новый контакт
                    when 1 then '&#9786;' -- улыбка, заинтересован
                    when 2 then '&#10004;' -- галочка, выдан
             end) || ' ' || str2;

    ogr_ol_date = null;
    ogr_amount = 0;
    select sum(ogr_amount * ogr_sign * f_loyal(ogr_from_ol_id, current_date)), min(ogr_ol_date)
    from operagr
    where ogr_object_kind = f_objkind('clientlink', 'id') and
          ogr_object_id = :clk_id and
          ogr_ol_is_active = 1
    into ogr_amount, ogr_ol_date;

    if (ogr_ol_date is not null) then
      str2 = str2 || ', ' || f_datetostr(ogr_ol_date);

    str3 = round(ogr_amount);

    res_text = BLOB_APPEND(res_text, f_format(str_orig, str, str2, str3));
  end
  res_text = BLOB_APPEND(res_text, '</table>');

  suspend;
end
