CREATE PROCEDURE HP_DO_KN_OPERBOOK_ONE_ONLINE(
    K_ID ID_TYPE,
    REP_DATE D_DATE,
    LOAN_OL_ID D_ID,
    RECALC_DATE D_DATE = null)
AS
declare variable ol_id id_type;
declare variable ol_date date;
declare variable ol_date_prv date;
declare variable kc_id id_type;
declare variable ks_date_end date;
declare variable kc_last_ol_date date;
declare variable oek_rd_id integer;
declare variable ol_oper_subtype d_integer;
declare variable kob_id d_id;
declare variable need_delete_and_exit d_smallint;
declare variable cm_loan_history_action d_integer;
declare variable acc_id d_id;
declare variable need_add d_bool;
begin
  if (rep_date is null) then
    rep_date = current_date;

  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  if (k_id < 10000000000) then
  begin
    select kc_from_k_id
    from knclient
    where kc_id = :k_id
    into k_id;

    if (row_count = 0) then
      exception e_default 'k_id not found: ' || k_id;
  end

  -- займ сторнирован 23.06.2020 и не нужно трогать
  if (k_id = 101010044606900101) then
    exit;

  -- Если имеются проводки по сторнированию из-за Фрода, отменяем перестроение проводок
  if (exists(select *
  from operfraud, operlist ol
  where ofr_from_k_id = :k_id and
        ofr_storno_date >= :recalc_date and
        ofr_storno_ol_id = ol.ol_id and
        ol.is_delete = 0)) then
    Exit;

  select cm_loan_history_action
  from company
  into cm_loan_history_action;

  -- если дата не передана, обрабатываем до последней даты (плановая, конец месяца либо факт.опер)
  need_delete_and_exit = 0;
  select oks_date_begin
  from kn_oksummary
  where oks_from_k_id = :k_id and
        oks_date_begin <= maxvalue(current_date, :rep_date)
  order by oks_date_begin desc
  rows 1
  into rep_date;
  if (row_count = 0) then
    need_delete_and_exit = 1;

  if (rep_date <= cast('28.02.2019' as date)) then
    exception e_default 'Feb.2019 is closed';

  if (need_delete_and_exit = 0) then
  begin
    select min(kc_last_ol_date)
    from knclient
    where kc_from_k_id = :k_id and
          kc_last_ol_date is not null and
          kc_amount_approved > 0
    into kc_last_ol_date;
  
    if (kc_last_ol_date is null) then
      kc_last_ol_date = cast('28.02.2019' as date);
    else
      if (kc_last_ol_date < cast(f_get_const('kn_operbook_close_date') as date) and
          cm_loan_history_action = 0) then
        kc_last_ol_date = cast(f_get_const('kn_operbook_close_date') as date);

    if (kc_last_ol_date <= cast('28.02.2019' as date)) then
      kc_last_ol_date = cast('28.02.2019' as date);
  
    -- Если принимается погашение задним числом, требуется перестройка всех проводок начиная с даты погашения
    if (rep_date < current_date and loan_ol_id is not null) then
    begin
      loan_ol_id = null;
      recalc_date = rep_date - 1;
    end

    if (rep_date = kc_last_ol_date and
        recalc_date is null) then
    begin
      if (loan_ol_id is not null) then
      begin
        execute procedure hp_do_kn_operbook_one_loan_date(:k_id, :rep_date, null, :loan_ol_id);
  
        -- проводки по переносу сальдо из-за смены резидентства
        if (rep_date = f_date_month_end(rep_date)) then
          execute procedure hp_do_kn_resident_update(:k_id, :rep_date, 13, 39, 1000000000001);
      end
  
      exit; -- обработали конкретный loan_ol_id и завершаем работу
    end
  end else
  begin
    kc_last_ol_date = rep_date;
    if (not exists(
                  select *
                  from kn_oksummary
                  where oks_from_k_id = :k_id)) then
    begin
      for select kc_id
      from knclient
      where kc_from_k_id = :k_id
      into kc_id
      do
      begin
        delete from kn_operbook
        where kob_from_kc_id = :kc_id;

        delete from kn_resident
        where knr_from_kc_id = :kc_id and
              knt_from_ol_id is not null;
      end
    end

    Exit; -- Нет движений по аналитике, заверщаем работу
  end

  if (recalc_date is not null and
      recalc_date < kc_last_ol_date) then
    kc_last_ol_date = recalc_date;

  -- начиная с марта 2019 года создаем единый OL_ID
  for select distinct kob_from_ol_id, kob_from_kc_id, ol_oper_subtype
      from kn_operbook, knclient, operlist
      where kc_from_k_id = :k_id and
            kc_id = kob_from_kc_id and
            kob_from_ol_id = ol_id and
            kob_ol_date > :kc_last_ol_date and
            kob_ol_is_active = 1 and
            ol_oper_type = 2 and
            ol_oper_subtype in (31, 32, 33, 34)
      into ol_id, kc_id, ol_oper_subtype
  do
  begin
    if (ol_oper_subtype = 31) then
    begin
      -- проводки по смене резиденства
      delete from operlist
      where ol_id = :ol_id;
    end
    else
    begin
      -- начисления: процент, резерв и корректировка ВБС
      delete from kn_operbook
      where kob_from_ol_id = :ol_id and
            kob_from_kc_id = :kc_id;
    end
  end

  for select kc_id
      from knclient
      where kc_from_k_id = :k_id
      into :kc_id
  do
  begin
    for select kob_id
        from kn_operbook, operlist
        where kob_from_ol_id = ol_id and
              not(ol_oper_type = 2 and
              ol_oper_subtype in (32, 33, 34)) and
              kob_from_kc_id = :kc_id and
              kob_ol_date > :kc_last_ol_date and
              kob_ol_is_active = 1
        order by kob_from_ol_id desc, kob_id desc
        into kob_id
    do
    begin
      delete from kn_operbook
      where kob_id = :kob_id;
    end

    delete from kn_resident
    where knr_from_kc_id = :kc_id and
          knr_date_begin > :kc_last_ol_date;
  end

  ol_date_prv = kc_last_ol_date;

  select ks_date_end
  from kontstate
  where ks_from_k_id = :k_id and
        ks_ol_is_delete = 0 and
        ks_state = 1 rows 1
  into :ks_date_end;

  for select distinct ok_ol_date
      from operkred
      where ok_from_k_id = :k_id and
            ok_from_c_id = 0 and
            (ok_os <> 0 or
            ok_perc <> 0 or
            ok_fine <> 0 or
            ok_kommis <> 0 or
            ok_kommis_adv <> 0 or
            ok_sh_os <> 0 or
            ok_sh_perc <> 0) and
            ok_type in (2, 3, 4, 5, 6, 8, 9) and
            ok_ol_date between(:kc_last_ol_date + 1) and :rep_date
      union
      select distinct oko_ol_date
      from operkroff
      where oko_from_k_id = :k_id and
            (oko_os <> 0 or
            oko_perc <> 0 or
            oko_sh_os <> 0 or
            oko_sh_perc <> 0) and
            oko_type in (4, 5) and
            oko_ol_date between(:kc_last_ol_date + 1) and :rep_date
      union
      select distinct ox_ol_date
      from operextra
      where ox_from_rd_id = :oek_rd_id and
            ox_object_id = :k_id and
            ox_object_kind = 9 and
            (ox_amount_plus <> 0 or
            ox_amount_minus <> 0 or
            ox_amount_writeoff <> 0) and
            ox_ol_date between(:kc_last_ol_date + 1) and :rep_date
      union
      select distinct oks_date_begin
      from kn_oksummary
      where oks_from_k_id = :k_id and
            (oks_is_month = 1 or
            oks_is_sh = 1) and
            oks_date_begin between(:kc_last_ol_date + 1) and :rep_date and
            oks_close_state <> 2
      into :ol_date
  do
  begin
    if (not exists(select * from kn_oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin = :ol_date)) then
    begin
      need_add = 0;

      if (exists(select * from operkred
      where ok_from_k_id = :k_id and
            ok_ol_date = :ol_date and
            ok_ol_is_active = 1)) then
        need_add = 1;

      if (need_add = 0 and exists(select * from operkroff
      where oko_from_k_id = :k_id and
            oko_ol_date = :ol_date and
            oko_ol_is_active = 1)) then
        need_add = 1;

      if (need_add = 0 and exists(select * from operextra
      where ox_object_kind = 9 and
            ox_object_id = :k_id and
            ox_ol_date = :ol_date and
            ox_ol_is_active = 1)) then
        need_add = 1;

      if (need_add = 1) then
        execute procedure hp_kn_oksummary_kont_insert(:k_id, :ol_date, 1);
    end

    execute procedure hp_do_kn_operbook_one_loan_date(:k_id, :ol_date, :ol_date_prv, null);

    -- проводки по переносу сальдо из-за смены резидентства
    if (ol_date = f_date_month_end(ol_date)) then
      execute procedure hp_do_kn_resident_update(:k_id, :ol_date, 13, 39, 1000000000001);

    ol_date_prv = ol_date;
  end

  if (rep_date <> ol_date or
      ol_date is null) then
  begin
    if ((ks_date_end > cast('31.12.2999' as date)) or
        (ks_date_end < cast('31.12.2999' as date) and
        ks_date_end >= rep_date)) then
      execute procedure hp_do_kn_operbook_one_loan_date(:k_id, :rep_date, :ol_date_prv, null);
  end

  for select kc_id
  from knclient
  where kc_from_k_id = :k_id
  into kc_id
  do
  begin
    for select acc_id
    from ref_account
    where acc_object_kind = f_objkind('knclient', 'id') and
          acc_object_id = :kc_id and
          acc_pair_acc_id is not null and
          acc_type = 0 and
          acc_code not starting '603'
    into acc_id
    do
    begin
      execute procedure hp_do_kn_operbook_fix_one_pair(:acc_id, :kc_last_ol_date + 1);
      when any do
      begin
      end
    end
  end

  -- устанавливаем последнюю успешную дату операции
  update knclient
  set kc_last_ol_date = :rep_date
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0;
end
