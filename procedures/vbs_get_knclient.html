CREATE PROCEDURE VBS_GET_KNCLIENT(
    K_ID ID_TYPE)
RETURNS (
    KC_ID ID_TYPE,
    KC_NAME D_STR_255,
    KC_NON_RESIDENT D_SMALLINT,
    KC_AMOUNT CURR_TYPE,
    K_KONTRACT_NO VARCHAR(20))
AS
declare variable c_id id_type;
declare variable doc_vnj_id integer;
declare variable citizen_ru_id integer;
declare variable k_disb_date date;
begin

  kc_id = 0;
  kc_name = 'Группа';
  kc_non_resident = 0;
  select k_amount_approved
  from kontract
  where k_id = :k_id
  into kc_amount;
  suspend;

  doc_vnj_id = null;
  select pk_id
  from passkind, passdoctype
  where pdt_id = pk_from_pdt_id and
        pk_is_delete = 0 and
        pdt_code = 'registration_vnj' rows 1
  into :doc_vnj_id;

  citizen_ru_id = null;
  select rd_id
  from refdata
  where rd_code = 'Citizen.RU'
  into :citizen_ru_id;

  for select kc_id, c_id, c_name_fio, kc_amount_approved, k_disb_date,k_kontract_no
      from knclient, clients, kontract
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id and
            kc_from_k_id = k_id
      into kc_id, c_id, kc_name, kc_amount, k_disb_date,k_kontract_no
  do
  begin
    k_disb_date = current_date;

    select res_non_resident
    from hp_get_client_type(:c_id, :citizen_ru_id, :doc_vnj_id, :k_disb_date)
    into :kc_non_resident;

    suspend;
  end
end
