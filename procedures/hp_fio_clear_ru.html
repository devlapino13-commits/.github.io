CREATE PROCEDURE HP_FIO_CLEAR_RU(
    C_NAME_FIO D_STR_255)
RETURNS (
    NAME_LAST D_STR_255,
    NAME_FIRST D_STR_255,
    NAME_MIDDLE D_STR_255)
AS
declare variable str_tmp d_str_code;
begin
  if (c_name_fio like '% ууулу %') then str_tmp = ' ууулу';
  if (c_name_fio like '% uuulu %') then str_tmp = ' uuulu';

  if (str_tmp is not null) then
    c_name_fio = replace(c_name_fio, str_tmp, '');

  select lower(txt1), lower(txt2), lower(txt3)
  from hp_fio_clear(:c_name_fio)
  into :name_last, :name_first, :name_middle;

  name_first = coalesce(name_first, ''); -- Имя
  name_last = coalesce(name_last, ''); -- Фамилия
  name_middle = coalesce(name_middle, ''); -- Отчество

  if (c_name_fio like '% уулу %') then name_last = name_last || ' уулу';
  if (c_name_fio like '% кызы %') then name_last = name_last || ' кызы';
  if (c_name_fio like '% uulu %') then name_last = name_last || ' uulu';
  if (c_name_fio like '% kyzy %') then name_last = name_last || ' kyzy';
  if (c_name_fio like '% угли %') then name_last = name_last || ' угли';
  if (c_name_fio like '% кизи %') then name_last = name_last || ' кизи';
  if (c_name_fio like '% qizi %') then name_last = name_last || ' qizi';
  if (c_name_fio like '% oglu %') then name_last = name_last || ' oglu';
  if (c_name_fio like '% ugli %') then name_last = name_last || ' ugli';

  if (c_name_fio like '% оглы') then name_middle = name_middle || ' оглы';
  if (c_name_fio like '% оглу') then name_middle = name_middle || ' оглу';
  if (c_name_fio like '% оглуу') then name_middle = name_middle || ' оглуу';
  if (c_name_fio like '% угли') then name_middle = name_middle || ' угли';
  if (c_name_fio like '% кизи') then name_middle = name_middle || ' кизи';
  if (c_name_fio like '% уулу') then name_middle = name_middle || ' уулу';
  if (c_name_fio like '% кызы') then name_middle = name_middle || ' кызы';
  if (c_name_fio like '% кизы') then name_middle = name_middle || ' кизы';
  if (c_name_fio like '% кызи') then name_middle = name_middle || ' кызи';
  if (c_name_fio like '%-кызы') then name_middle = name_middle || '-кызы';

  if (str_tmp is not null) then
    name_last = name_last || str_tmp;

  name_first = upper(name_first);
  name_last = upper(name_last);
  name_middle = upper(name_middle);

  name_first = replace(name_first, ascii_char(10), '');
  name_first = replace(name_first, ascii_char(13), '');

  if (char_length(name_first) = 1) then
    name_first = name_first || ' ';
  if (char_length(name_last) = 1) then
    name_last = name_last || ' ';
  if (char_length(name_middle) = 1) then
    name_middle = name_middle || ' ';

  if (char_length(name_first) < 2) then
    name_first = null;
  if (char_length(name_last) < 2) then
    name_last = null;
  if (char_length(name_middle) < 2) then
    name_middle = null;

  suspend;
end
