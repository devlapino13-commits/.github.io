CREATE PROCEDURE AAREP_FAMILY_STATUS_DIFF(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    CLIENT_FULLNAME D_STR_100,
    CLIENT_INN D_STR_20,
    KONTRACT_NO D_STR_20,
    OFFICE_NAME D_STR_100,
    FAMILY_STATUS_PASSPORT D_STR_64,
    FAMILY_STATUS_ZAGS D_STR_64,
    FAMILY_STATUS_FACT D_STR_64,
    CERTIFICATE_SPOUSE_DEATH D_STR_10,
    CERTIFICATE_MARRIAGE D_STR_10,
    CERTIFICATE_DIVORCE D_STR_10,
    JUDICIAL_ACT D_STR_10,
    LETTER_OF_ATTORNEY D_STR_10,
    AGREEMENT D_STR_10)
AS
declare variable C_ID ID_TYPE;
declare variable PDT_CODE D_STR_100;
declare variable OFFICE_ID ID_TYPE;
begin
  -- процедура была создана для исполнения задачи RFCKG-1546
  office_id = 0;

  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
  into
    :office_id
  do
  begin
    for select kc.kc_from_c_id,
               kn.k_kontract_no,
               rg.reg_name
        from kontract kn
        inner join kontstate ks on ks.ks_from_k_id = kn.k_id
        inner join knclient kc on kc.kc_from_k_id = kn.k_id
        inner join region rg on rg.reg_unicode = kn.k_fact_disb_reg_id
        where ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1 and
              ks.ks_date_begin = :rep_date_end and
              ks.ks_region_id = :office_id and
              kc.kc_amount_approved >= 50000
    into :c_id,
         :kontract_no,
         :office_name
    do
    begin
      client_fullname = '';
      client_inn = '';
      family_status_passport = '';
      select cl.c_name_fio,
             cl.c_inn,
             rd.rd_name
      from clients cl
      left join refdata rd on rd.rd_id = cl.c_family
      where cl.c_id = :c_id
      into :client_fullname,
           :client_inn,
           :family_status_passport;

      family_status_passport = coalesce(family_status_passport, '');

      family_status_zags = 'Не заполнен';
      select ovs_value
      from hp_objectprops_get(0, :c_id, -1, 'AnketaQuestions.Family_tunduk', :rep_date_end)
      into :family_status_zags;

      family_status_fact = 'Не заполнен';
      select ovs_value
      from hp_objectprops_get(0, :c_id, -1, 'AnketaQuestions.Family_fact', :rep_date_end)
      into :family_status_fact;

      pdt_code = '';
      certificate_spouse_death = '';
      certificate_marriage = '';
      certificate_divorce = '';
      judicial_act = '';
      letter_of_attorney = '';
      agreement = '';
      for select distinct pdt.pdt_code
          from clientdocs cds
          inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
          inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
          where pdt.pdt_code in ('certificate_death',
                                 'certificate_marriage',
                                 'certificate_annulment',
                                 'judicial_act',
                                 'power_of_attorney',
                                 'consent_to_a_loan') and
                cds.cds_from_c_id = :c_id and
                cds.cds_is_delete = 0 and
                :rep_date_end between cds.cds_doc_date_begin and coalesce(cds.cds_doc_date_end, '31.12.9999') and
                pk.pk_is_delete = 0
      into :pdt_code
      do
      begin
        if (pdt_code = 'certificate_death') then
          certificate_spouse_death = 'Да';
        if (pdt_code = 'certificate_marriage') then
          certificate_marriage = 'Да';
        if (pdt_code = 'certificate_annulment') then
          certificate_divorce = 'Да';
        if (pdt_code = 'judicial_act') then
          judicial_act = 'Да';
        if (pdt_code = 'power_of_attorney') then
          letter_of_attorney = 'Да';
        if (pdt_code = 'consent_to_a_loan') then
          agreement = 'Да';
      end

      if (family_status_passport <> family_status_zags) then
        suspend;
    end
  end
end
