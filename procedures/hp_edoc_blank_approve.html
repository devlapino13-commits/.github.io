CREATE PROCEDURE HP_EDOC_BLANK_APPROVE(
    EDL_ID D_ID NOT NULL,
    APPROVE_KIND D_BOOL NOT NULL = 0)
AS
declare variable edl_status d_smallint; /* Статус документа */
declare variable bls_id d_id;
declare variable edf_print_name d_str_255;
declare variable edb_id d_id;
declare variable edb_blob d_blob_binary;
begin
  -- Процедура утверждает документ, т.е.:
  -- Обнуляет флаг утверждения (edb_need_approve = 0)
  -- В зависимости от approve_kind
  --   0 - Отправляет на переподготовку сервисом (edb_status = 0)
  --   1 - Переносит блобы в blobstore, без переподготовки сервисом

  select edl_status
  from edoc_list
  where edl_id = :edl_id
  into edl_status;

  if (edl_status <> 1 or approve_kind not in(0, 1)) then
    exit;

  if (approve_kind = 0) then
  begin
    -- Переподготовка через сервис
    update edoc_blank
    set edb_need_approve = 0,
        edb_status = 0
    where
        edb_from_edl_id = :edl_id and
        edb_need_approve = 1;
  end
  else
  begin
    -- Иначе переносим бланк в blobstore
    for select edb_id, edb_blob, edf_print_name
    from edoc_blank, edoc_frx
    where edb_from_edl_id = :edl_id and
          edb_from_edf_id = edf_id and
          edb_blob is not null and
          edb_need_approve = 1 and
          coalesce(edb_from_bls_id, 0) = 0
    into edb_id, edb_blob, edf_print_name
    do
    begin
      execute procedure hp_insert_into_blobstore ('edoc_' || :edb_id || '.pdf', :edf_print_name, :edb_blob,
        1000000000000, 2, 0)
      returning_values :bls_id;

      update edoc_blank
      set edb_from_bls_id = :bls_id,
          edb_blob = null
      where edb_id = :edb_id;
    end
  end
end
