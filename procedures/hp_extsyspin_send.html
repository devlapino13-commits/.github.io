CREATE PROCEDURE HP_EXTSYSPIN_SEND(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    CONT_KIND D_STR_CODE,
    CONT_VALUE D_STR_CODE)
AS
declare variable CLT_ID D_INTEGER;
declare variable PIN D_STR_CODE;
declare variable MSG_BODY D_STR_LONG;
declare variable CONT_CODE D_STR_CODE;
begin
  -- Процедура отправляет пин-код для подтверждения телефона на веб сайте.
  -- Проверка правильности пин-кода осуществляется через процедуру HP_EXTSYSPIN_VERIFY
  -- В будущем можно реализовать и для email

  obj_kind = f_objkind(obj_kind, 'id');

  cont_code = f_enum('cont_kind', cont_kind, 'code', 1);
  cont_kind = f_enum('cont_kind', cont_kind, 'id', 1);

  if (cont_code <> 'cell') then
    exception e_default 'Incorrect cont_kind: ' || cont_code;
  if (cont_value not similar to '(79|992|996|998|994)[[:DIGIT:]]{9}') then
    exception e_default 'Incorrect cont_value: ' || cont_value;

  clt_id = f_cont_id(cont_value, 'add');
  pin = right('0000' || cast(trunc(rand() * 100000) as integer), 4);

  insert into extsyspin (esp_object_kind, esp_object_id, esp_cont_kind, esp_from_clt_id, esp_pin, esp_status)
  values (:obj_kind, :obj_id, :cont_kind, :clt_id, f_coder_pass(:pin), 0);

  --msg_body = f_format(f_lang('', iif(f_get_db_id() = 81001, 'М Булак', 'М Булак')||': Ваш код для подтверждения: %1'), Pin);
  msg_body = f_format(f_lang('', iif(f_get_db_id() = 81001, '', '') || 'Ваш код для подтверждения: %1'), pin);     ---RFCRU-4257

  insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                              msg_command_kind)
  values (3, :cont_value, '', :msg_body, 1, 1000000000000, '');
end
