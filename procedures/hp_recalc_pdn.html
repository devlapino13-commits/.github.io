CREATE PROCEDURE HP_RECALC_PDN
AS
declare variable mac_id d_id;
declare variable kpd_err d_str_255;
declare variable monthly_income decimal(15,2);
declare variable extra_amount decimal(15,2);
declare variable repayment_obligations decimal(15,2);
declare variable amount_monthly_payments decimal(15,2);
declare variable pdn decimal(15,2);
declare variable ma_id d_integer;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable c_id d_id;
declare variable ca_id d_id;
declare variable acl_id d_id;
begin
  for select ca.ca_id,
             ca.ca_avr_income,
             iif(f_objkind(ca.ca_object_kind, 'code') = 'miniappclient', ca.ca_object_id, 0),
             ca.ca_from_acl_id
  from cfp_anketa ca
  where coalesce(ca.ca_from_acl_id, 0) <> 0 and
        ca.ca_date = current_date and
        ca.ca_datetime between dateadd(-10 minute to current_timestamp) and current_timestamp
  into ca_id,
       monthly_income,
       mac_id,
       acl_id
  do
  begin

    if (coalesce(mac_id, 0) = 0) then
      continue;

      -- Если КПД уже посчитан
    if (exists(select *
               from cfp_anketa_kpd cak
               where cak.cak_from_ca_id = :ca_id)) then
      continue;

    select kpd_err
    from save_cfp_kpd(:mac_id, :acl_id)
    into :kpd_err;

    if (coalesce(kpd_err, '') = '') then
    begin
      select mac.mac_from_ma_id,
             mac.mac_from_c_id
      from miniappclient mac
      where mac.mac_id = :mac_id
      into ma_id,
           c_id;

      select k.k_kontract_no,
             k.k_disb_date
      from kontract k
      where k.k_from_ma_app_id = :ma_id
      into k_kontract_no,
           k_disb_date;

      select extra_amount, repayment_obligations, amount_monthly_payments
      from get_cfp_kpd(:mac_id)
      into :extra_amount, :repayment_obligations, :amount_monthly_payments;

      pdn = (repayment_obligations + amount_monthly_payments)/ (monthly_income + extra_amount) * 100;

      if (:pdn > 60) then
      begin
        execute procedure mc_send_push('clients', :c_id, 'Новый показатель долговой нагрузки', f_format('Уважаемый клиент, Ваш показатель долговой нагрузки по контракту № %1 от %2 превышает 60%', :k_kontract_no, f_datetostr(:k_disb_date)));
      end
    end
  end
end
