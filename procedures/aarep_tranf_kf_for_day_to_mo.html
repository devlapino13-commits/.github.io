CREATE PROCEDURE AAREP_TRANF_KF_FOR_DAY_TO_MO(
    REGION_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    NEW_REG_NAME D_STR_100,
    PRIOR_LO_FIO D_STR_100,
    NEW_KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    C_FIO D_STR_100,
    K_DISB_DATE D_DATE,
    SUM_OS D_CURRENCY,
    SUM_CALC_PERC D_CURRENCY,
    SUM_CALC_FINE D_CURRENCY,
    K_DELINQ_DAYS D_SMALLINT,
    UN_FIO D_STR_100,
    JL_NAME D_STR_100)
AS
declare variable k_id d_id;
declare variable ka_id d_id;
declare variable ka_id_tmp d_id;
declare variable kf_id d_id;
declare variable new_reg_id d_integer;
declare variable new_reg_id_tmp d_integer;
declare variable prior_lo_id d_id;
declare variable prior_lo_id_tmp d_id;
declare variable min_sh_date d_date;
begin
  new_reg_id_tmp = 0;
  prior_lo_id_tmp = 0;
  ka_id_tmp = 0;
  /* Для рассылки по переданным папкам на адрес старшего РМ
    В письме будет отчет с полями
  >>  Офис, от кого передан, номер папки,
      номер контракта, ФИО клиента, Дата выдачи,
      Баланс по ОС, накопленные %%, накопленные пении, кол.просроченных дней
      ФИО кто сформировал передачу (т.е. тут должно быть ФИО МО),
      должность создателя записи ( по идее должность всегда = менеджер офиса)
  */
  for
  --Переданные папки за один день
      select kal.kal_from_ka_id, kal.kal_from_kf_id, kal.kal_new_reg_id, kal.kal_prior_lo_id, kal.kal_new_kf_name
      from region_list(:region_id, 1, -1, 1)
      inner join knaktlist kal on id_list = kal_new_reg_id and
            kal.is_delete = 0 and
            :rep_date = kal_date and
            kal.kal_new_kf_officer = 1000000000100
      order by kal_new_reg_id, kal_prior_lo_id, kal_from_ka_id
      into ka_id, kf_id, new_reg_id, prior_lo_id, new_kf_name
  do
  begin
    if (ka_id <> ka_id_tmp) then
    begin
      ka_id_tmp = ka_id;
      select un.un_fio, coalesce(jl.jl_name, 'Должность не определена')
      from knakt
      inner join operlist ol on knakt.ka_from_ol_id = ol_id
      inner join username un on un.un_id = ol.ol_user_id
      left join hr_pstatus hp on hp.ps_from_c_id = un.un_from_c_id and
            :rep_date between hp.ps_date_begin and hp.ps_date_end
      left join joblist jl on hp.ps_from_jl_id = jl.jl_id
      where knakt.ka_id = :ka_id
      into un_fio, jl_name;
    end

    if (new_reg_id <> new_reg_id_tmp) then
    begin
      new_reg_id_tmp = new_reg_id;

      select r.reg_name
      from region r
      where r.reg_unicode = :new_reg_id
      into new_reg_name;
    end

    if (prior_lo_id <> prior_lo_id_tmp) then
    begin
      prior_lo_id_tmp = prior_lo_id;
      select c_name_fio
      from clients c
      where c.c_id = :prior_lo_id
      into prior_lo_fio;
    end

    k_delinq_days = 0;
    min_sh_date = null;
    sum_os = null;
    sum_calc_perc = null;
    sum_calc_fine = null;
    for select k_id, k.k_kontract_no, k_disb_date, --
               min(sh_date) as min_sh_date, --
               sum(zad_os) sum_os, --
               sum(sh_calc_perc) sum_calc_perc, --
               sum(sh_calc_fine) sum_calc_fine --
        from kontract k
        inner join aarep_make_zad_list(k_id, :rep_date, 1) on sh_date is not null
        where k.k_from_kf_id = :kf_id
        group by k_id, k_kontract_no, k_disb_date
        into :k_id, :k_kontract_no, :k_disb_date, :min_sh_date, :sum_os, :sum_calc_perc, sum_calc_fine
    do
    begin
      k_delinq_days = rep_date - min_sh_date;

      if (k_delinq_days < 0) then
        k_delinq_days = 0;

      for select c_name_fio
          from knclient kc
          inner join clients c on kc_from_c_id = c_id
          where kc_from_k_id = :k_id and
                kc_amount_approved > 0
          into c_fio
      do
      begin
        suspend;
      end
    end
  end

  if (new_reg_name is null) then
  begin
    new_reg_name = 'На сегодня передач папок виртуальному сотруднику не осуществлялось';
    suspend;
  end
end
