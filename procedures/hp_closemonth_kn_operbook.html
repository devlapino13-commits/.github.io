CREATE PROCEDURE HP_CLOSEMONTH_KN_OPERBOOK(
    KONT_NO D_STR_CODE)
RETURNS (
    KC_ID D_ID,
    KOB_ID D_ID,
    KOB_ACC_CODE_DT D_STR_CODE,
    KOB_ACC_CODE_KT D_STR_CODE,
    KOB_ACC_DT D_ID,
    KOB_ACC_KT D_ID,
    KOB_SUMMA CURR_TYPE,
    KOB_SUMMA_VAL CURR_TYPE,
    KOB_SUMMA_TAX_DT CURR_TYPE,
    KOB_SUMMA_TAX_KT CURR_TYPE,
    KOB_OL_DATE D_DATE,
    KOB_DETAIL D_STR_LONG,
    KOB_FROM_OL_ID D_STR_CODE,
    ACC_DT_NAME D_STR_255,
    ACC_KT_NAME D_STR_255,
    SYS_DT_CODE D_STR_255,
    SYS_KT_CODE D_STR_255,
    OL_MEMO D_STR_LONG)
AS
declare variable k_id d_id;
begin
  if (kont_no containing '-') then
  begin
    select k_id
    from kontract
    where k_kontract_no = :kont_no
    into k_id;
  end else k_id = kont_no;

  for select kc_id
  from knclient
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into kc_id
  do
  begin
    for select kob_id, f_fmt_acc(kob_acc_dt), f_fmt_acc(kob_acc_kt),
               kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val, kob_summa_tax_dt,
               kob_summa_tax_kt, kob_ol_date, obd_detail, kob_from_ol_id, ol_memo,
               a_dt.acc_name, a_kt.acc_name,
               s_dt.sab_full_code, s_kt.sab_full_code
    from kn_operbook, kn_obdetail, ref_account a_dt, ref_account a_kt, sys_accbalance s_dt, sys_accbalance s_kt, operlist
    where kob_from_kc_id = :kc_id and
          kob_ol_is_active = 1 and
          kob_detail_id = obd_id and
          kob_acc_dt = a_dt.acc_id and
          a_dt.acc_from_sab_id = s_dt.sab_id and
          kob_acc_kt = a_kt.acc_id and
          a_kt.acc_from_sab_id = s_kt.sab_id and
          kob_from_ol_id = ol_id
    order by kob_ol_date, kob_id
    into :kob_id, :kob_acc_code_dt, :kob_acc_code_kt,
         :kob_acc_dt, :kob_acc_kt, :kob_summa, :kob_summa_val,
         :kob_summa_tax_dt, :kob_summa_tax_kt, :kob_ol_date, :kob_detail, :kob_from_ol_id, :ol_memo,
         :acc_dt_name, :acc_kt_name, :sys_dt_code, :sys_kt_code
    do
      suspend;
  end
end
