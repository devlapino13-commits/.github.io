CREATE PROCEDURE HP_EXTINFO_PKB_AVLBL_PROCESS(
    EIL_ID D_ID)
AS
declare variable EIL_OBJECT_ID D_ID;
declare variable EIL_OBJECT_KIND D_INTEGER;
declare variable EIL_FROM_C_ID D_ID;
declare variable EIL_FROM_UN_ID D_ID;
declare variable HAS_CREDIT_HISTORY D_SMALLINT;
begin
    select
        eil.eil_object_id,
        eil.eil_object_kind,
        eil.eil_from_c_id,
        eil.eil_from_un_id
    from extinfolist eil
    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
        and eic.eic_code = 'pkb_available_reports'
    where eil.eil_id = :eil_id
        and eil.eil_status = 2
        and eil.eil_from_bls_id is not null
    into
        :eil_object_id,
        :eil_object_kind,
        :eil_from_c_id,
        :eil_from_un_id;

    if (coalesce(:eil_object_id, 0) = 0) then
        exit;

    select priority_id
    from hp_get_priority_rep_id(:eil_object_kind, :eil_object_id)
    into :has_credit_history;

    -- если has_credit_history -1, то считаем, что у клиента нет кредитной истории
    if (has_credit_history = -1) then
    begin
        update or insert into negativ_kib_history (nkh_from_c_id, nkh_is_gambler, nkh_is_overdue, nkh_is_bankrot,
                                                   nkh_last, nkh_from_un_id, nkh_from_eil_id, nkh_is_overdue_90)
        values (:eil_from_c_id, 0, 0, 0, 1, :eil_from_un_id, :eil_id, 0)
        matching (nkh_from_c_id, nkh_last);

        update or insert into kib_job_info (kji_from_c_id, kji_from_ma_id, kji_job_title, kji_from_eil_id, kji_create_datetime)
        values (:eil_from_c_id, :eil_object_id, 'Самозанятый', :eil_id, current_timestamp)
        matching (kji_from_c_id, kji_from_ma_id);

        -- RFCKZ-278
        --execute procedure hp_extinfo_insert_afm(:eil_from_c_id, :eil_object_id);  RFCKZ-389
    end
end
