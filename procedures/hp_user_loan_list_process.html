CREATE PROCEDURE HP_USER_LOAN_LIST_PROCESS(
    REP_DATE_END D_DATE NOT NULL,
    ONLY_OWN_LOANS D_BOOL NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_K_ID D_ID,
    RES_DISB_DATE D_DATE,
    RES_IS_OWN D_BOOL,
    RES_MA_ID D_INTEGER,
    RES_KF_NAME D_STR_100,
    RES_LOAN_STATUS D_STR_100,
    RES_MA_PLAN_DISB_TIME D_DATETIME,
    RES_PR_ID D_INTEGER)
AS
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable lml_id d_integer;
declare variable jk_system_code d_str_code;
declare variable rep_region_id d_integer;
declare variable need_region_suspend d_bool;
declare variable orig_c_id d_id;
declare variable ks_state d_integer;
declare variable k_status d_integer;
declare variable kf_id d_id;
declare variable ma_id_tmp d_integer;
declare variable res_object_status d_integer;
declare variable rd_code d_str_100;
begin
  select un_from_c_id
  from username
  where un_id = :un_id
  into orig_c_id;

  for select res_c_id
  from hp_staff_changer_list(:un_id, iif(:only_own_loans = 1, -1, 1))
  into :c_id
  do
  begin
    res_is_own = 0;
    if (c_id = orig_c_id) then
      res_is_own = 1;

    need_region_suspend = 1;

    select ps_from_reg_id, jk_system_code
    from hr_pstatus, joblist, jobkind
    where ps_from_c_id = :c_id and
          ps_is_working = 1 and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id
    into rep_region_id, jk_system_code;
  
    if (row_count = 0) then
      Continue;

    if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner', 'JobSystem.Specialist_Expert')) then
    begin
      -- Кредитный специалист
      res_ma_id = 0;
      ks_state = 0;
      res_k_id = 0;
      res_loan_status = '';
      res_pr_id = 0;
      res_kf_name = '';
      res_ma_plan_disb_time = null;

      for select ks_from_k_id, ks_state, k_status, k_from_kf_id, k_disb_date, k_from_ma_app_id,
                 k_from_pr_id
          from kontstate, kontract, knaktlist
          where ks_ol_is_delete = 0 and
                k_status <> -2 and
                ks_date_end >= :rep_date_end and
                ((ks_state = 0 and
                k_status >= 0 and
                ks_date_begin between(:rep_date_end - 20) and :rep_date_end) -- Договора в процессе
                or
                (ks_state = 0 and
                k_status = -1 and
                ks_date_begin between(:rep_date_end - 1) and :rep_date_end) -- Отказанные договора в процессе
                or
                (ks_state = 1 and
                ks_date_begin = :rep_date_end) -- выданные за сегодня
                ) and
                ks_from_k_id = k_id and
                knaktlist.is_delete = 0 and
                kal_date <= :rep_date_end and
                kal_date_end >= :rep_date_end and
                kal_from_kf_id = k_from_kf_id and
                kal_new_kf_officer = :c_id
          order by ks_date_begin desc
          into :res_k_id, ks_state, k_status, kf_id, res_disb_date, ma_id_tmp, :res_pr_id
      do
      begin
        res_ma_plan_disb_time = null;
        select ma_plan_disb_time
        from miniapp
        where ma_id = :ma_id_tmp and
              ma_date_end >= :rep_date_end and
              ma_date_app <= :rep_date_end
        into :res_ma_plan_disb_time;
  
        res_loan_status = '';
        if (ks_state = 1) then
          res_loan_status = 'Выдан';
        else
        begin
          if (k_status = -1) then
            res_loan_status = 'Отказано';
          if (k_status = 0) then
          begin
            res_loan_status = 'В процессе';
            res_object_status = null;

            select rd_code
            from hp_get_check_loan_data(:res_k_id)
            into:rd_code;

            select ovs_add_value
            from hp_objectprops_get('Kontract', :res_k_id, -1, :rd_code, null)
            into :res_object_status;
            if (res_object_status is not null) then
            begin
              res_loan_status = (case res_object_status
                               when -2 then 'КК: Отказан'
                               when -1 then 'КК: Доработка'
                               when 0 then 'КК: В очереди'
                               when 1 then 'КК: На рассмотр.'
                               when 2 then 'КК: Подтвержден'
                             end);
            end
          end
          if (k_status = 1) then
          begin
            res_loan_status = 'Утвержден';
          end
        end
        res_kf_name = '';
        select kal_new_kf_name
        from knaktlist
        where kal_date_end >= :rep_date_end and
              knaktlist.is_delete = 0 and
              kal_from_kf_id = :kf_id
        into :res_kf_name;
  
        suspend;
      end
  
      res_ma_id = 0;
      ks_state = 0;
      res_k_id = 0;
      res_loan_status = '';
      res_pr_id = 0;
      res_kf_name = '';
      res_ma_plan_disb_time = null;

      for select ma_id, ma_is_active, ma_from_kf_id, kal_new_kf_name, ma_date_oper, ma_plan_disb_time,
                 ma_from_pr_id
          from miniapp, knaktlist
          where ma_date_end >= :rep_date_end and
                ((ma_is_active = 1 and
                ma_date_app between(:rep_date_end - 20) and :rep_date_end) -- Действующий заявки
                or
                (ma_is_active <= 0 and
                ma_date_app between(:rep_date_end - 1) and :rep_date_end) -- Аннулированные
                ) and
                kal_new_kf_officer = :c_id and
                ma_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                kal_date <= :rep_date_end and
                kal_date_end >= :rep_date_end
          into :res_ma_id, k_status, kf_id, res_kf_name, :res_disb_date, res_ma_plan_disb_time, :res_pr_id
      do
      begin
        if (not(exists(select *
                       from kontract
                       where k_from_ma_app_id = :res_ma_id and
                             k_status >= -1))) then
        begin
          res_loan_status = '';
          if (k_status = -1) then
            res_loan_status = 'Отказана';
          if (k_status = 0) then
            res_loan_status = 'Аннулирована';
          if (k_status = 1) then
            res_loan_status = 'Заявка';
          suspend;
        end
      end

      need_region_suspend = 0;
    end else
    begin
      lml_id = null;

      select lml_id
      from loanmgrlist
      where lml_from_c_id = :c_id and
            lml_date_begin <= :rep_date_end and
            lml_date_end >= :rep_date_end
      into lml_id;
  
      if (lml_id is not null) then
      begin
        -- Менеджер мега-офиса, получаем список подчиненных специалистов
        for select ke_from_p_id, ke_from_reg_id
        from krexpert
        where ke_from_lml_id = :lml_id and
              ke_date <= :rep_date_end and
              ke_date_end >= :rep_date_end
        into c_id, reg_id
        do
        begin
          res_ma_id = 0;
          ks_state = 0;
          res_k_id = 0;
          res_loan_status = '';
          res_pr_id = 0;
          res_kf_name = '';
          res_ma_plan_disb_time = null;

          for select ks_from_k_id, ks_state, k_status, k_from_kf_id, k_disb_date, k_from_ma_app_id,
                     k_from_pr_id
              from kontstate, kontract, knaktlist
              where ks_ol_is_delete = 0 and
                    k_status <> -2 and
                    ks_date_end >= :rep_date_end and
                    ((ks_state = 0 and
                    k_status >= 0 and
                    ks_date_begin between(:rep_date_end - 20) and :rep_date_end) -- Договора в процессе
                    or
                    (ks_state = 0 and
                    k_status = -1 and
                    ks_date_begin between(:rep_date_end - 1) and :rep_date_end) -- Отказанные договора в процессе
                    or
                    (ks_state = 1 and
                    ks_date_begin = :rep_date_end) -- выданные за сегодня
                    ) and
                    ks_from_k_id = k_id and
                    knaktlist.is_delete = 0 and
                    kal_date <= :rep_date_end and
                    kal_date_end >= :rep_date_end and
                    kal_from_kf_id = k_from_kf_id and
                    kal_new_kf_officer = :c_id and
                    kal_new_reg_id = :reg_id
              order by ks_date_begin desc
              into :res_k_id, ks_state, k_status, kf_id, res_disb_date, ma_id_tmp, :res_pr_id
          do
          begin
            res_ma_plan_disb_time = null;
            select ma_plan_disb_time
            from miniapp
            where ma_id = :ma_id_tmp and
                  ma_date_end >= :rep_date_end and
                  ma_date_app <= :rep_date_end
            into :res_ma_plan_disb_time;
      
            res_loan_status = '';
            if (ks_state = 1) then
              res_loan_status = 'Выдан';
            else
            begin
              if (k_status = -1) then
                res_loan_status = 'Отказано';
              if (k_status = 0) then
              begin
                res_loan_status = 'В процессе';
                res_object_status = null;

                select rd_code
                from hp_get_check_loan_data(:res_k_id)
                into:rd_code;
      
                select ovs_add_value
                from hp_objectprops_get('Kontract', :res_k_id, -1, :rd_code, null)
                into :res_object_status;
                if (res_object_status is not null) then
                begin
                  res_loan_status = (case res_object_status
                                   when -2 then 'КК: Отказан'
                                   when -1 then 'КК: Доработка'
                                   when 0 then 'КК: В очереди'
                                   when 1 then 'КК: На рассмотр.'
                                   when 2 then 'КК: Подтвержден'
                                 end);
                end
              end
              if (k_status = 1) then
              begin
                res_loan_status = 'Утвержден';
              end
            end
            res_kf_name = '';
            select kal_new_kf_name
            from knaktlist
            where kal_date_end >= :rep_date_end and
                  knaktlist.is_delete = 0 and
                  kal_from_kf_id = :kf_id
            into :res_kf_name;
      
            suspend;
          end
      
          res_ma_id = 0;
          ks_state = 0;
          res_k_id = 0;
          res_loan_status = '';
          res_pr_id = 0;
          res_kf_name = '';
          res_ma_plan_disb_time = null;
          for select ma_id, ma_is_active, ma_from_kf_id, kal_new_kf_name, ma_date_oper, ma_plan_disb_time,
                     ma_from_pr_id
              from miniapp, knaktlist
              where ma_date_end >= :rep_date_end and
                    ((ma_is_active = 1 and
                    ma_date_app between(:rep_date_end - 20) and :rep_date_end) -- Действующий заявки
                    or
                    (ma_is_active <= 0 and
                    ma_date_app between(:rep_date_end - 1) and :rep_date_end) -- Аннулированные
                    ) and
                    kal_new_kf_officer = :c_id and
                    kal_new_reg_id = :reg_id and
                    ma_from_kf_id = kal_from_kf_id and
                    knaktlist.is_delete = 0 and
                    kal_date <= :rep_date_end and
                    kal_date_end >= :rep_date_end
              into :res_ma_id, k_status, kf_id, res_kf_name, :res_disb_date, res_ma_plan_disb_time, :res_pr_id
          do
          begin
            if (not(exists(select *
                           from kontract
                           where k_from_ma_app_id = :res_ma_id and
                                 k_status >= -1))) then
            begin
              res_loan_status = '';
              if (k_status = -1) then
                res_loan_status = 'Отказана';
              if (k_status = 0) then
                res_loan_status = 'Аннулирована';
              if (k_status = 1) then
                res_loan_status = 'Заявка';
              suspend;
            end
          end
        end

        need_region_suspend = 0;
      end else
      begin
        -- Если не сотрудник фронт-офиса, завершаем работу
        if (jk_system_code not in ('JobSystem.Region_Director', 'JobSystem.Region_KM', 'JobSystem.Specialist_SB')) then
          need_region_suspend = 0;
      end
    end

    if (need_region_suspend = 1) then
    begin
      -- Список займов в офисе.
      for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into reg_id
      do
      begin
        res_ma_id = 0;
        ks_state = 0;
        res_k_id = 0;
        res_loan_status = '';
        res_pr_id = 0;
        res_kf_name = '';
        res_ma_plan_disb_time = null;

        for select ks_from_k_id, ks_state, k_status, k_from_kf_id, k_disb_date, k_from_ma_app_id,
                   k_from_pr_id
            from kontstate, kontract, knaktlist
            where ks_ol_is_delete = 0 and
                  k_status <> -2 and
                  ks_date_end >= :rep_date_end and
                  ((ks_state = 0 and
                  k_status >= 0 and
                  ks_date_begin between(:rep_date_end - 20) and :rep_date_end) -- Договора в процессе
                  or
                  (ks_state = 0 and
                  k_status = -1 and
                  ks_date_begin between(:rep_date_end - 1) and :rep_date_end) -- Отказанные договора в процессе
                  or
                  (ks_state = 1 and
                  ks_date_begin = :rep_date_end) -- выданные за сегодня
                  ) and
                  ks_from_k_id = k_id and
                  knaktlist.is_delete = 0 and
                  kal_date <= :rep_date_end and
                  kal_date_end >= :rep_date_end and
                  kal_from_kf_id = k_from_kf_id and
                  kal_new_reg_id = :reg_id
            order by ks_date_begin desc
            into :res_k_id, ks_state, k_status, kf_id, res_disb_date, ma_id_tmp, :res_pr_id
        do
        begin
          res_ma_plan_disb_time = null;
          select ma_plan_disb_time
          from miniapp
          where ma_id = :ma_id_tmp and
                ma_date_end >= :rep_date_end and
                ma_date_app <= :rep_date_end
          into :res_ma_plan_disb_time;
    
          res_loan_status = '';
          if (ks_state = 1) then
            res_loan_status = 'Выдан';
          else
          begin
            if (k_status = -1) then
              res_loan_status = 'Отказано';
            if (k_status = 0) then
            begin
              res_loan_status = 'В процессе';
              res_object_status = null;

              select rd_code
              from hp_get_check_loan_data(:res_k_id)
              into:rd_code;
    
              select ovs_add_value
              from hp_objectprops_get('Kontract', :res_k_id, -1, :rd_code, null)
              into :res_object_status;
              if (res_object_status is not null) then
              begin
                res_loan_status = (case res_object_status
                                 when -2 then 'КК: Отказан'
                                 when -1 then 'КК: Доработка'
                                 when 0 then 'КК: В очереди'
                                 when 1 then 'КК: На рассмотр.'
                                 when 2 then 'КК: Подтвержден'
                               end);
              end
            end
            if (k_status = 1) then
            begin
              res_loan_status = 'Утвержден';
            end
          end
          res_kf_name = '';
          select kal_new_kf_name
          from knaktlist
          where kal_date_end >= :rep_date_end and
                knaktlist.is_delete = 0 and
                kal_from_kf_id = :kf_id
          into :res_kf_name;
    
          suspend;
        end
    
        res_ma_id = 0;
        ks_state = 0;
        res_k_id = 0;
        res_loan_status = '';
        res_pr_id = 0;
        res_kf_name = '';
        res_ma_plan_disb_time = null;

        for select ma_id, ma_is_active, ma_from_kf_id, kal_new_kf_name, ma_date_oper, ma_plan_disb_time,
                   ma_from_pr_id
            from miniapp, knaktlist
            where ma_date_end >= :rep_date_end and
                  ((ma_is_active = 1 and
                  ma_date_app between(:rep_date_end - 20) and :rep_date_end) -- Действующий заявки
                  or
                  (ma_is_active <= 0 and
                  ma_date_app between(:rep_date_end - 1) and :rep_date_end) -- Аннулированные
                  ) and
                  kal_new_reg_id = :reg_id and
                  ma_from_kf_id = kal_from_kf_id and
                  knaktlist.is_delete = 0 and
                  kal_date <= :rep_date_end and
                  kal_date_end >= :rep_date_end
            into :res_ma_id, k_status, kf_id, res_kf_name, :res_disb_date, res_ma_plan_disb_time, :res_pr_id
        do
        begin
          if (not(exists(select *
                         from kontract
                         where k_from_ma_app_id = :res_ma_id and
                               k_status >= -1))) then
          begin
            res_loan_status = '';
            if (k_status = -1) then
              res_loan_status = 'Отказана';
            if (k_status = 0) then
              res_loan_status = 'Аннулирована';
            if (k_status = 1) then
              res_loan_status = 'Заявка';
            suspend;
          end
        end
      end
    end
  end
