CREATE PROCEDURE AAREP_RATING_OFFICE_PRIROST_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    OKRUG_NAME VARCHAR(100),
    PRIROST_PORTFEL CURR_TYPE,
    PAR1 CURR_TYPE,
    PAR7 CURR_TYPE,
    PAR1_BALL CURR_TYPE,
    PAR7_BALL CURR_TYPE,
    PRIROST_CLI CURR_TYPE,
    PRIROST_PORTFEL_BALL CURR_TYPE,
    BALANCE_BEGIN DECIMAL(18,4),
    BALANCE_END DECIMAL(18,4),
    TOTAL_RANK CURR_TYPE)
AS
declare variable rep_date_begin date;
declare variable oks_region_id integer;
declare variable oks_balance decimal(18,4);
declare variable oks_delinq_balanse7 decimal(18,4);
declare variable kol_cli_begin integer;
declare variable oks_region_id_prior integer;
declare variable oks_delinq_balanse1 curr_type;
declare variable kol_svz integer;
declare variable kol_cli_end integer;
begin
  rep_date_begin = rep_date_end - 7;
--  delete from TEMP_KC_REG_INFO;
-- на начало
-- на начало
  insert into temp_kc_reg_info (kc_region_id, kc_kol_cli, kc_balance, kc_kol_cli_sks)
  select ac_region_id, sum(ac_is_client), sum(ac_cli_balance), 0 -- KC_KOL_CLI_SKS - 0 - на дату начало; 1 - на дату конец
  from aarep_cli_active_list(:rep_date_begin)
  where (select aa_category_id
         from aarep_get_category_id(ac_kontract_id, :rep_date_begin, null, null, null, null, null, null)) in (1, 2) and
        not exists(select *
                   from sb_akt, sb_list
                   where sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_status = 2 and
                         :rep_date_begin between sba_date_begin and sba_date_end and
                         sba_from_k_id = ac_kontract_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1 -- находящийся по внешним коллектором не будет считатся
        )
  group by ac_region_id;

-- на конец
  insert into temp_kc_reg_info (kc_region_id, kc_kol_cli, kc_balance, kc_kol_cli_sks)
  select ac_region_id, sum(ac_is_client), sum(ac_cli_balance), 1 -- KC_KOL_CLI_SKS - 0 - на дату начало; 1 - на дату конец
  from aarep_cli_active_list(:rep_date_end)
  where (select aa_category_id
         from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2) and
        not exists(select *
                   from sb_akt, sb_list
                   where sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_status = 2 and
                         :rep_date_end between sba_date_begin and sba_date_end and
                         sba_from_k_id = ac_kontract_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1 -- находящийся по внешним коллектором не будет считатся
        ) and
        (exists(select *
                from knaktlist -- если не было передано из друго офиса
                where knaktlist.is_delete = 0 -- т.е. офис на конец предыдущего месяца не поменялся
                      and
                      :rep_date_begin between kal_date and kal_date_end and
                      kal_from_kf_id = ac_kf_id and
                      kal_new_reg_id = ac_region_id) or
        ac_disb_date between :rep_date_begin + 1 and :rep_date_end) -- или в текущем периоде было выдача
  group by ac_region_id;

--  select * from TEMP_KC_REG_INFO

  oks_region_id_prior = 0;
  for select oks_region_id, sum(oks_balance),
             sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 0, oks_balance, 0)),
             sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 7, oks_balance, 0))
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_close_state = 0 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2) and
            oks_region_id = id_list
      group by oks_region_id
      into oks_region_id, oks_balance, oks_delinq_balanse1, oks_delinq_balanse7
  do
  begin
    if (oks_region_id_prior <> oks_region_id) then
    begin
      oks_region_id_prior = oks_region_id;
      select r3.reg_code || ' ' || r3.reg_name,
--             r2.reg_code||' '||r2.reg_name,
             r1.reg_code || ' ' || r1.reg_name
      from region r1, region r2, region r3
      where r1.reg_unicode = :oks_region_id and
            r2.reg_unicode = r1.parent and
            r3.reg_unicode = r2.parent
      order by r3.reg_code, r2.reg_code, r1.reg_code
      into okrug_name,
--           branch_name,
           reg_name;
    end

--  Расчет кол-во СВЗ
    kol_svz = null;
    select count(*)
    from hr_pstatus, joblist, jobkind
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          ps_status = 1 and
          ps_from_reg_id = :oks_region_id and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
          coalesce((select sawt_id
                    from hr_voctravel, hr_orders, sa_worktype
                    where hrv_from_ho_id = ho_id and
                          ho_order_type = sawt_id and
                          hrv_from_c_id = ps_from_c_id and
                          :rep_date_end between hrv_date_begin and hrv_date_end_fact
                    order by hrv_date_begin desc
                    rows 1), 0) not in (16, 17, 18) -- Декретников не включает
    into kol_svz;
    if (kol_svz is null) then
      kol_svz = 0;

    if (oks_delinq_balanse1 = 0) then
    begin
      par1 = 0;
    end
    else
    begin
      par1 = oks_delinq_balanse1 / oks_balance * 100;
    end
    if (oks_delinq_balanse7 = 0) then
    begin
      par7 = 0;
    end
    else
    begin
      par7 = oks_delinq_balanse7 / oks_balance * 100;
    end

-- клиенты на начало
    kol_cli_begin = null;
    balance_begin = null;
    select kc_kol_cli, kc_balance
    from temp_kc_reg_info
    where kc_region_id = :oks_region_id and
          kc_kol_cli_sks = 0 -- выборка начало
    into kol_cli_begin, balance_begin;
    if (kol_cli_begin is null) then
      kol_cli_begin = 0;
    if (balance_begin is null) then
      balance_begin = 0;

-- клиенты на конец
    kol_cli_end = null;
    balance_end = 0;
    select kc_kol_cli, kc_balance
    from temp_kc_reg_info
    where kc_region_id = :oks_region_id and
          kc_kol_cli_sks = 1 -- выборка конец
    into kol_cli_end, balance_end;
    if (kol_cli_end is null) then
      kol_cli_end = 0;
    if (balance_end is null) then
      balance_end = 0;

    prirost_cli = kol_cli_end - kol_cli_begin;
    prirost_portfel = balance_end - balance_begin;

    par1_ball = (10 - par1) * 20 / 100;
    if (par1_ball < 0) then
      par1_ball = 0;
    par7_ball = (7 - par7) * 20 / 100;

    prirost_portfel_ball = (balance_end - balance_begin) / balance_begin * 50;

    total_rank = par1_ball + par7_ball + prirost_cli + prirost_portfel_ball;

    suspend;
  end
end
