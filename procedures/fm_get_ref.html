CREATE PROCEDURE FM_GET_REF(
    REF_CODE VARCHAR(255))
RETURNS (
    RD_ID INTEGER,
    RD_CODE VARCHAR(255),
    RD_NAME VARCHAR(255))
AS
declare variable tmp_rd_code varchar(255);
declare variable db_id integer;
begin

  db_id = f_get_db_id();

  if (ref_code = 'citizen') then
  begin
    -- На первом месте страна БД
    tmp_rd_code = case db_id
                    when 41701 then 'citizen.kg'
                    when 39801 then 'citizen.kz'
                    when 94401 then 'citizen.az'
                    else 'citizen.ru'
                  end;

    for select rd_id, replace(rd_code, 'citizen.', ''), rd_name
        from refdata
        where rd_code = :tmp_rd_code and
              rd_is_active = 1
        union all
        select rd_id, replace(rd_code, 'citizen.', ''), rd_name
        from refdata
        where rd_main_code = 'citizen' and
              rd_is_active = 1 and
              rd_code <> rd_main_code and
              coalesce(rd_str_id, '') <> '' and
              rd_code <> :tmp_rd_code
        into :rd_id, :rd_code, :rd_name
    do
    begin
      suspend;
    end
    exit;
  end

  if (ref_code = 'nation') then
  begin
    for select rd_id, rd_name
        from refdata
        where rd_main_code = 'nation' and
              rd_is_active = 1 and
              rd_code <> rd_main_code
        into :rd_id, :rd_name
    do
      suspend;
    exit;
  end

  if (ref_code = 'relation') then
  begin
    rd_id = null;
    rd_code = null;
    for select rd_name
        from refdata
        where rd_main_code = :ref_code and
              rd_is_active = 1 and
              rd_code <> rd_main_code and
              coalesce(rd_str_id, '') <> ''
        order by rd_orderno
        into :rd_name
    do
    begin
      suspend;
    end
    rd_name = 'Коллега';
    suspend;
    rd_name = 'Друг';
    suspend;
    rd_name = 'Знакомый';
    suspend;

    exit;
  end
  if (ref_code = 'language_call_center') then
  begin

    rd_code = null;
    for select rd_id, rd_name
        from refdata
        where rd_main_code = 'Language' and
              rd_is_active = 1 and
              rd_code in ('Language.RU', 'Language.UZ', 'Language.KG', 'Language.TJ', 'Language.AZ')
        order by case
                   when :db_id = 81001 and rd_code = 'Language.RU' then -999
                   when :db_id in (39801, 41701) and rd_code = 'Language.KG' then -999
                   when :db_id = 94401 and rd_code = 'Language.AZ' then -999
                   else rd_orderno
                 end
        into :rd_id, :rd_name
    do
    begin
      suspend;
    end
    exit;
  end

  if (ref_code = 'reltype') then
  begin
    rd_code = '';
    for select rel_id, rel_name
        from relationtype
        where rel_can_select = 1 and
              char_length(rel_id) = 1
        into rd_id, rd_name
    do
      suspend;
    exit;
  end

  if (ref_code = 'qr_reason') then
  begin
    rd_id = 1;
    rd_name = 'У клиента кнопочный телефон';
    suspend;

    rd_id = 2;
    rd_name = 'У клиента не Android телефон';
    suspend;

    rd_id = 3;
    rd_name = 'Клиент не может установить моб. приложение';
    suspend;

    rd_id = 4;
    rd_name = 'Клиент не хочет установить моб. приложение';
    suspend;
  end
end
