CREATE PROCEDURE UXP$UXEXPRESSION_PARCE(
    EXP_UX_UUID D_UUID,
    EXP_EXPRESSION D_STR_LONG,
    EXP_IS_SQL D_BOOL,
    EXP_UFR_UUID D_UUID)
RETURNS (
    NEW_EXPRESSION D_STR_LONG,
    NEW_CALC_EXPR D_STR_LONG,
    NEW_CHECK_EXPR D_STR_LONG)
AS
declare variable us_uuid d_uuid;
declare variable uf_field_name d_str_code;
declare variable uf_is_param d_bool;
declare variable txt d_str_code;
declare variable uf_domain d_str_code;
declare variable uf_uuid_code d_str_code;
declare variable p d_integer;
begin
  new_expression = trim(coalesce(exp_expression, ''));

  p = position(' as  ', new_expression);
  while (p > 0) do
  begin
    new_expression = replace(new_expression, ' as  ', ' as ');
    p = position(' as  ', new_expression);
  end
  new_expression = replace(new_expression, ' as ', ' as•');

  new_calc_expr = new_expression;
  new_check_expr = new_expression;

  if (new_expression <> '') then
  begin
    select us_uuid
    from uxp$uxframe_source(:exp_ux_uuid, :exp_ufr_uuid)
    into us_uuid;

    if (us_uuid is null) then
      exception e_default 'Source not found for : ' || coalesce(exp_ux_uuid, '<Null>');

    for select '[' || uuid_to_char(uf_uuid) || ']', uf_field_name
    from ux$field
    where uf_owner = :us_uuid
    order by char_length(uf_field_name) desc
    into uf_uuid_code, uf_field_name
    do
    begin
      new_expression = replace(new_expression, uf_field_name, uf_uuid_code);
      new_expression = replace(new_expression, '•'||uf_uuid_code, '•'||uf_field_name);
    end

    for select '[' || uuid_to_char(uf_uuid) || ']', uf_is_param
    from ux$field
    where uf_owner = :us_uuid
    order by char_length(uf_field_name) desc
    into uf_field_name, uf_is_param
    do
    begin
      if (uf_is_param = 1) then
      begin
        txt = 'old.' || uf_field_name;
        if (position(txt, new_expression) > 0) then
          new_expression = replace(new_expression, txt, ':' || uf_field_name);

        txt = 'new.' || uf_field_name;
        if (position(txt, new_expression) > 0) then
          new_expression = replace(new_expression, txt, ':' || uf_field_name);

        if (exp_is_sql = 0) then
        begin
          if (position(uf_field_name, new_expression) > 0 and
              position('new.' || uf_field_name, new_expression) = 0 and
              position('old.' || uf_field_name, new_expression) = 0 and
              position(':' || uf_field_name, new_expression) = 0) then
            new_expression = replace(new_expression, uf_field_name, ':' || uf_field_name);
        end
      end else
      begin
        if (position(':' || uf_field_name, new_expression) > 0) then
          new_expression = replace(new_expression, ':' || uf_field_name, 'new.' || uf_field_name);

        if (exp_is_sql = 0) then
        begin
          if (position(uf_field_name, new_expression) > 0 and
              position('new.' || uf_field_name, new_expression) = 0 and
              position('old.' || uf_field_name, new_expression) = 0 and
              position(':' || uf_field_name, new_expression) = 0) then
            new_expression = replace(new_expression, uf_field_name, 'new.' || uf_field_name);
        end
      end
    end

    new_calc_expr = new_expression;
    new_check_expr = new_expression;

    for select '[' || uuid_to_char(uf_uuid) || ']', uf_is_param, trim(iif(:exp_is_sql = 1, uf_domain, uf_data_type))
    from ux$field
    where uf_owner = :us_uuid
    order by char_length(uf_field_name) desc
    into uf_field_name, uf_is_param, uf_domain
    do
    begin
      txt = 'old.' || uf_field_name;
      if (position(txt, new_expression) > 0) then
      begin
        new_calc_expr = replace(new_calc_expr, txt, ':old$' || uf_field_name);
        new_check_expr = replace(new_check_expr, txt, ':' || uf_field_name);
      end

      txt = 'new.' || uf_field_name;
      if (position(txt, new_expression) > 0) then
      begin
        new_calc_expr = replace(new_calc_expr, txt, ':new$' || uf_field_name);
        new_check_expr = replace(new_check_expr, txt, ':' || uf_field_name);
      end

      new_check_expr = replace(new_check_expr, ':' || uf_field_name, 'cast(:' || uf_field_name || ' as ' || uf_domain || ')');
    end

    for select '[' || uuid_to_char(uf_uuid) || ']', uf_field_name
    from ux$field
    where uf_owner = :us_uuid
    order by char_length(uf_field_name) desc
    into uf_uuid_code, uf_field_name
    do
    begin
      new_expression = replace(new_expression, uf_uuid_code, uf_field_name);
      new_calc_expr = replace(new_calc_expr, uf_uuid_code, uf_field_name);
      new_check_expr = replace(new_check_expr, uf_uuid_code, uf_field_name);
    end

    if (not(new_check_expr starting 'select ')) then
    begin
      new_check_expr = 'select ' || new_check_expr;

      if (exp_is_sql = 1) then
        new_check_expr = new_check_expr || ' from rdb$database';
      else
        new_check_expr = new_check_expr || ' from MemoryTable';
    end
  end

  new_expression = replace(new_expression, ' as•', ' as ');
  new_check_expr = replace(new_check_expr, ' as•', ' as ');
  new_calc_expr = replace(new_calc_expr, ' as•', ' as ');

  suspend;
end
