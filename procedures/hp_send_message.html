CREATE PROCEDURE HP_SEND_MESSAGE(
    SEND_KIND D_STR_CODE,
    C_ID D_ID NOT NULL,
    MAIL_TO D_STR_255,
    MAIL_SUBJECT D_STR_255,
    MAIL_BODY D_STR_8100,
    COMMAND_KIND D_STR_255,
    LIFETIME_DAYS D_INTEGER,
    UN_ID D_ID)
AS
declare variable msg_type integer;
begin
  if (lifetime_days is null) then
    lifetime_days = 7;

  if (send_kind not in ('email', 'sms')) then
    exit;

  if (lifetime_days is null) then
    lifetime_days = 7;

  if (send_kind = 'email') then
    msg_type = 0;
  if (send_kind = 'sms') then
    msg_type = 1;

  insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body,
                              msg_lifetime_days, msg_from_un_id,
                              msg_command_kind, msg_object_kind, msg_object_id)
  values (:msg_type, :mail_to, :mail_subject, :mail_body, :lifetime_days,
          :un_id, :command_kind, 0, :c_id);
end
