CREATE PROCEDURE UXP$UXGRID_COPY_COLUMNS(
    SRC_GRID_UUID D_UUID,
    DST_GRID_UUID D_UUID)
AS
declare variable uf_uuid d_uuid;
declare variable us_uuid d_uuid;
declare variable clone_count d_integer;
declare variable ufr_add_table_fields d_str_code;
declare variable main_us_uuid d_uuid;
begin
  clone_count = 0;

  delete from ux$band
  where bnd_grid_uuid = :dst_grid_uuid;

  delete from ux$gritem
  where ugi_ugr_uuid = :dst_grid_uuid
  order by ugi_position desc;

  select ufr_us_uuid, ufr_add_table_fields
  from ux$grid, ux$frame
  where ugr_uuid = :dst_grid_uuid and
        ugr_ufr_uuid = ufr_uuid
  into us_uuid, ufr_add_table_fields;

  if (ufr_add_table_fields <> '') then
  begin
    select us_uuid
    from ux$source
    where us_code = :ufr_add_table_fields
    into main_us_uuid;
  end

  if (exists(select * from ux$grid
             where ugr_uuid = :src_grid_uuid)) then
  begin
    insert into ux$band (bnd_grid_uuid, bnd_id, bnd_parent_id, bnd_index, bnd_childs, bnd_position, bnd_caption, bnd_hidden,
                         bnd_hold_own, bnd_width, bnd_fixed_kind)
    select :dst_grid_uuid, bnd_id, bnd_parent_id, bnd_index, bnd_childs, bnd_position, bnd_caption, bnd_hidden, bnd_hold_own,
           bnd_width, bnd_fixed_kind
    from ux$band
    where bnd_grid_uuid = :src_grid_uuid
    order by bnd_id;
  
    for select ugi_uuid, ugi_urt_uuid, ugi_ugr_uuid, ugi_uf_uuid, ugi_position, ugi_width, ugi_sizable, ugi_order_index,
               ugi_group_index, ugi_fixed_kind, ugi_summary_kind, ugi_bnd_id, ugi_caption, ugi_hint, ugi_hidden,
               ugi_readonly, ugi_header_align, ugi_value_align,
               uf_field_name
    from ux$gritem, ux$field
    where ugi_ugr_uuid = :src_grid_uuid and
          ugi_uf_uuid = uf_uuid
    order by ugi_position
    as cursor cur
    do
    begin
      uf_uuid = null;
  
      select uf_uuid
      from ux$field
      where uf_owner = :us_uuid and
            uf_field_name = cur.uf_field_name
      into uf_uuid;
  
      if (uf_uuid is null and main_us_uuid is not null) then
      begin
        select uf_uuid
        from ux$field
        where uf_owner = :main_us_uuid and
              uf_field_name = cur.uf_field_name
        into uf_uuid;

        Continue;
      end

      if (uf_uuid is null) then
        Continue;
  
      insert into ux$gritem (ugi_ugr_uuid, ugi_uf_uuid, ugi_position, ugi_width, ugi_sizable,
                             ugi_order_index, ugi_group_index, ugi_fixed_kind, ugi_summary_kind, ugi_bnd_id, ugi_caption,
                             ugi_hint, ugi_hidden, ugi_readonly, ugi_header_align, ugi_value_align)
      values (:dst_grid_uuid, :uf_uuid, cur.ugi_position, cur.ugi_width, cur.ugi_sizable,
              cur.ugi_order_index, cur.ugi_group_index, cur.ugi_fixed_kind, cur.ugi_summary_kind, cur.ugi_bnd_id, cur.ugi_caption, cur.ugi_hint,
              cur.ugi_hidden, cur.ugi_readonly, cur.ugi_header_align, cur.ugi_value_align);
  
      clone_count = clone_count + 1;
    end
  end else
  if (exists(select * from ux$unititem
             where uni_uuid = :src_grid_uuid)) then
  begin
    for select uni_position, uni_width, uni_sizable, uni_order_index,
               uni_group_index, uni_fixed_kind, uni_summary_kind,
               uni_caption, uni_hint, uni_hidden,
               uni_readonly, uni_fieldname
    from ux$unititem
    where uni_parent_uuid = :src_grid_uuid
    order by uni_position
    as cursor cur
    do
    begin
      uf_uuid = null;
  
      select uf_uuid
      from ux$field
      where uf_owner = :us_uuid and
            uf_field_name = cur.uni_fieldname
      into uf_uuid;
  
      if (uf_uuid is null and main_us_uuid is not null) then
      begin
        select uf_uuid
        from ux$field
        where uf_owner = :main_us_uuid and
              uf_field_name = cur.uni_fieldname
        into uf_uuid;

        Continue;
      end

      if (uf_uuid is null) then
        Continue;

      insert into ux$gritem (ugi_ugr_uuid, ugi_uf_uuid, ugi_position, ugi_width, ugi_sizable,
                             ugi_order_index, ugi_group_index, ugi_fixed_kind, ugi_summary_kind, ugi_caption,
                             ugi_hint, ugi_hidden, ugi_readonly)
      values (:dst_grid_uuid, :uf_uuid, cur.uni_position, cur.uni_width, cur.uni_sizable,
              cur.uni_order_index, cur.uni_group_index, cur.uni_fixed_kind, cur.uni_summary_kind, cur.uni_caption, cur.uni_hint,
              cur.uni_hidden, cur.uni_readonly);
  
      clone_count = clone_count + 1;
    end
  end

  if (clone_count = 0) then
    exception e_message 'Совпадения по полям не найдены';
end
