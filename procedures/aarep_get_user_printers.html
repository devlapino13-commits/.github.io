CREATE PROCEDURE AAREP_GET_USER_PRINTERS(
    UN_ID DECIMAL(18,0),
    UN_IP_ADDRESS VARCHAR(100))
RETURNS (
    RP_ID INTEGER,
    RP_PRINTER_CODE VARCHAR(255),
    RP_PRINTER_NAME VARCHAR(255),
    RP_IS_DEFAULT INTEGER)
AS
declare variable is_found integer;
declare variable i integer;
declare variable found_dot integer;
declare variable ip_mask varchar(100);
declare variable reg_id integer;
begin
  if (un_ip_address is null) then
    un_ip_address = '';
  is_found = 0;

  if (un_ip_address <> '') then
  begin
    i = char_length(un_ip_address);
    found_dot = 0;
    while (i > 0 and found_dot = 0) do
    begin
      if (substring(un_ip_address from i for 1) = '.') then
        found_dot = i;
      i = i - 1;
    end
    if (found_dot > 0) then
    begin
      ip_mask = left(un_ip_address, found_dot - 1);
      for select distinct rad_from_reg_id
          from regipaddr
          where rad_ip_address starting :ip_mask
          into reg_id
      do
      begin
        is_found = 1;
        for select rp_id, rp_printer_code, rp_printer_name || ' (' || reg_code || ')'
            from regprinters, region
            where rp_from_reg_id = :reg_id and
                  (rp_printer_status = 'online' or
                  rp_printer_status = 'active') and
                  reg_unicode = :reg_id
            into :rp_id, :rp_printer_code, rp_printer_name
        do
        begin
          rp_is_default = 0;
          if (exists(select *
                     from regprintusers
                     where rpu_from_un_id = :un_id and
                           rpu_from_reg_id = :reg_id and
                           rpu_from_rp_id = :rp_id)) then
            rp_is_default = 1;

          suspend;
        end
      end
    end
  end
  if (is_found = 0 and
      coalesce(un_id, 0) <> 0) then
  begin
    for select ps_from_reg_id
        from username, hr_pstatus
        where un_id = :un_id and
              ps_from_c_id = un_from_c_id and
              ps_date_begin <= current_date and
              ps_date_end >= current_date and
              ps_status = 1 and
              ps_is_working = 1
        into :reg_id
    do
    begin
      for select rp_id, rp_printer_code, rp_printer_name || ' (' || reg_code || ')'
          from regprinters, region
          where rp_from_reg_id = :reg_id and
                reg_unicode = :reg_id
          into :rp_id, :rp_printer_code, rp_printer_name
      do
      begin
        rp_is_default = 0;
        if (exists(select *
                   from regprintusers
                   where rpu_from_un_id = :un_id and
                         rpu_from_reg_id = :reg_id and
                         rpu_from_rp_id = :rp_id)) then
          rp_is_default = 1;
        else
          rp_is_default = 0;

        suspend;
      end
    end
  end
end
