CREATE PROCEDURE HP_DO_EXPRESSAPP_ACTION_DISB(
    EXA_ID D_INTEGER)
RETURNS (
    WAIT_STATUS D_SMALLINT)
AS
declare variable exa_from_k_id id_type;
declare variable pr_code d_str_100;
declare variable ma_id d_id;
declare variable un_id d_id;
declare variable c_id d_id;
declare variable manager_id d_id;
declare variable c_name_fio d_str_100;
declare variable k_kontract_no d_str_20;
declare variable db_id d_id;
declare variable phone d_str_20;
declare variable sms_text d_str_long;
declare variable res_text d_str_255;
declare variable crel_other_c_id d_id;
declare variable k_amount_approved d_currency;
begin
  -- Отправка контракта на выдачу
  db_id = f_get_db_id();
  wait_status = 0;

  select exa_from_k_id, pr.pr_code, exa.exa_from_ma_id, exa.exa_from_un_id, exa.exa_from_c_id
  from expressapp exa
  inner join product pr on pr.pr_id = exa.exa_from_pr_id
  where exa.exa_id = :exa_id
  into exa_from_k_id, pr_code, ma_id, un_id, c_id;

  if (exa_from_k_id is null) then
    wait_status = -1;

  if (db_id = 81001) then
  begin
    if (wait_status = 0) then
    begin
      execute procedure hp_kontcarddisb_sync(:exa_from_k_id);
  
      if (not exists(select * from kontcarddisb
      where kkd_from_k_id = :exa_from_k_id)) then
        wait_status = -1;
    end
  end

  if (db_id in (39801, 41701) and pr_code containing '.sugarmill.') then
  begin
    select ma.ma_manager_id
    from miniapp ma
    where ma.ma_id = :ma_id and
          ma.ma_is_active = 1 and
          ma.ma_date_end >= '31.12.2999'
    into manager_id;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into c_name_fio;

    select kt.k_kontract_no, kt.k_amount_approved
    from kontract kt
    where kt.k_id = :exa_from_k_id
    into k_kontract_no, k_amount_approved;

    phone = f_cont_info('clients', :c_id, 'cell_main', 'ok');

    if (k_amount_approved >= 50000) then
    begin
      execute procedure hp_send_push_to_staff(:manager_id,
                                              'Выдача займа по товарам в рассрочку',
                                              'Был выдан займ клиенту: ' || :c_name_fio || '. Номер договора: ' || :k_kontract_no,
                                              :un_id);

      select crl.crel_other_c_id
      from clientrelation crl
      where crl.crel_from_c_id = :c_id and
            crl.crel_from_rel_id in ('1', '2') and
            crl.crel_is_active = 1
      into crel_other_c_id;

      if (coalesce(crel_other_c_id, 0) > 0 and f_asp_signed(:crel_other_c_id, current_date) = 1) then
      begin
        select res_text
        from hp_edoc_insert('loan.spouses_consent', f_objkind('miniapp', 'id'), :ma_id, :un_id)
        into res_text;
      end
    end

    sms_text = 'Пройдите по ссылке в мобильное приложение МБулак, где размещены подписанные документы по кредиту https://play.google.com/store/apps/details?id=com.mbf.fsclient_android';
    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                msg_command_kind, msg_object_kind, msg_object_id)
    values (3, :phone, '', :sms_text, 7, 1000000000000, 'sms', 0, :c_id);

  end 

  suspend;
end
