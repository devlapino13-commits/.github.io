CREATE PROCEDURE FM_HTML_CLIENT_REASON_WRITEOFF(
    C_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable reason_writeoff d_str_255;
declare variable kontract_no d_str_20;
declare variable is_vu d_smallint;
declare variable k_id d_id;
declare variable rd_id d_id;
declare variable writeoff blob sub_type 1 segment size 80;
begin
  res_text = (select res_style
              from fm_html_styles) || '<body>';

  res_text = BLOB_APPEND(res_text, f_format('<br><b>%1 %2</b><br><br>', 'Списания клиента', ''));

  writeoff = '';

  for select oks.oks_from_k_id, k.k_kontract_no
      from oksummary oks
      inner join kontract k on oks.oks_from_k_id = k.k_id
      inner join knclient kc on oks.oks_from_k_id = kc.kc_from_k_id and
            kc.kc_amount_approved > 0 and
            kc.kc_from_c_id = :c_id
      where oks.oks_close_state in (0, 1) and
            :rep_date between oks.oks_date_begin and oks.oks_date_end
      into :k_id, :kontract_no
  do
  begin
    if (exists(select *
               from operkroff oko
               where oko.oko_from_k_id = :k_id and
                     oko.oko_from_c_id = :c_id and
                     oko.oko_ol_is_active = 1)) then
    begin
      for select ol.ol_from_rd_id
          from operkroff oko
          inner join operlist ol on oko.oko_from_ol_id = ol.ol_id
          where oko.oko_from_k_id = :k_id and
                oko.oko_from_c_id = :c_id and
                oko.oko_ol_is_active = 1 and
                oko.oko_ol_date <= :rep_date
          order by oko.oko_ol_date desc
          into :rd_id
      do
      begin
        if (rd_id is null) then continue;
      end

      reason_writeoff = '';

      select rd.rd_name
      from refdata rd
      where rd.rd_id = :rd_id
      into :reason_writeoff;

      if (reason_writeoff containing 'Не заполнен') then reason_writeoff = 'Внесистемный учет';

      writeoff = BLOB_APPEND(writeoff, '<tr>' || '<td align="center">' || reason_writeoff || '</td>' || --
      '<td align="center">' || kontract_no || '</td>' || '</tr>');
    end
  end

  res_text = BLOB_APPEND(res_text, '<table width="100%">
               <col width="50%">
               <col width="50%">
               <tr>
               <th>' || 'Причина списания' || '</th>' || '<th>' || 'Номер договора' || '</th>' || '</tr>', writeoff);

  res_text = BLOB_APPEND(res_text, '</table>');

  if (reason_writeoff is null) then
  begin
    res_text = BLOB_APPEND(res_text, '');
  end
  suspend;
end
