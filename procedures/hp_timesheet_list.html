CREATE PROCEDURE HP_TIMESHEET_LIST(
    REG_ID D_INTEGER,
    MONTH_INP D_INTEGER,
    YEAR_INP D_INTEGER,
    TAG D_INTEGER = 0)
RETURNS (
    FIO D_STR_100,
    JL_SHORT D_STR_100,
    SAT_DAY_1 D_STR_10,
    SAT_DAY_2 D_STR_10,
    SAT_DAY_3 D_STR_10,
    SAT_DAY_4 D_STR_10,
    SAT_DAY_5 D_STR_10,
    SAT_DAY_6 D_STR_10,
    SAT_DAY_7 D_STR_10,
    SAT_DAY_8 D_STR_10,
    SAT_DAY_9 D_STR_10,
    SAT_DAY_10 D_STR_10,
    SAT_DAY_11 D_STR_10,
    SAT_DAY_12 D_STR_10,
    SAT_DAY_13 D_STR_10,
    SAT_DAY_14 D_STR_10,
    SAT_DAY_15 D_STR_10,
    SAT_DAY_16 D_STR_10,
    SAT_DAY_17 D_STR_10,
    SAT_DAY_18 D_STR_10,
    SAT_DAY_19 D_STR_10,
    SAT_DAY_20 D_STR_10,
    SAT_DAY_21 D_STR_10,
    SAT_DAY_22 D_STR_10,
    SAT_DAY_23 D_STR_10,
    SAT_DAY_24 D_STR_10,
    SAT_DAY_25 D_STR_10,
    SAT_DAY_26 D_STR_10,
    SAT_DAY_27 D_STR_10,
    SAT_DAY_28 D_STR_10,
    SAT_DAY_29 D_STR_10,
    SAT_DAY_30 D_STR_10,
    SAT_DAY_31 D_STR_10,
    C_ID D_ID,
    IS_READY D_SMALLINT,
    SAT_DAYS_TOTAL D_CURRENCY,
    SAT_DAYS_TOTAL_OVERWORK D_CURRENCY)
AS
declare variable date_begin_prev d_date;
declare variable date_begin d_date;
declare variable date_next_month d_date;
declare variable date_end d_date;
declare variable sat_year_month d_integer;
declare variable month_prev d_integer;
declare variable year_prev d_integer;
declare variable calc_overwork boolean;
declare variable sat_days_total_prev d_currency;
declare variable norma_working_days_tm d_integer;
declare variable db_id d_integer;
declare variable ps_date_begin d_date;
declare variable ps_is_working d_smallint;
declare variable tmp_date_integer d_integer;
begin
  db_id = f_get_db_id();
  date_begin = '01.' || month_inp || '.' || year_inp;
  date_next_month = '01.' ||(iif(month_inp <> 12, month_inp, 0) + 1) || '.' || year_inp;
  date_end = extract(day from (date_next_month - 1)) || '.' || month_inp || '.' || year_inp;
  sat_year_month = 100 * year_inp + month_inp;

  -- расчёт по чётным месяцам
  calc_overwork = mod(month_inp, 2) = 0;
  norma_working_days_tm = 0;
  if (calc_overwork) then
  begin
    date_begin_prev = dateadd(month, -1, date_begin);
    month_prev = extract(month from date_begin_prev);
    year_prev = extract(year from date_begin_prev);
    -- норма рабочих дней за 2 мес. (январь-февраль, март-апрель ... etc.)
    norma_working_days_tm =
      coalesce((select norma_working_days from hp_timesheet_pre_set(:reg_id, :year_inp, :month_inp)), 0) +
      coalesce((select norma_working_days from hp_timesheet_pre_set(:reg_id, :year_prev, :month_prev)), 0);
  end

  for select distinct c.c_name_fio,
                      case ps.ps_from_jl_id
                        when 12 then 'Водитель'
                        when 36 then 'КОБД'
                        when 39 then 'Тех. раб.'
                        when 40 then 'Сис. админ.'
                        when 85 then 'Выезд. кас.'
                        when 213 then 'РМ'
                        when 225 then 'МО'
                        when 226 then 'ССпФ'
                        when 227 then 'СпФ'
                      else jl_name
                      end,
                      c.c_id,
                      ps.ps_date_begin,
                      ps.ps_is_working
      from hr_pstatus ps
      inner join clients c on ps.ps_from_c_id = c.c_id
      inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id
      inner join sa_worktype sawt on sawt.sawt_id = ho.ho_order_type
      left join joblist on jl_id = ps.ps_from_jl_id
      left join sa_tabelo sat on sat.sat_from_c_id = c.c_id and
            sat.sat_year_month = :sat_year_month and
            sat.sat_office_id = :reg_id
      where ps.ps_from_reg_id = :reg_id and
            (ps.ps_is_working = 1 or
             iif(:db_id in (39801, 41701), (ps.ps_is_working = 0 and sawt.sawt_id = 6 and ps.ps_date_begin between :date_begin and :date_end), true)) and --JIRA 1335, отображать данные об уволенных в текущем месяце
            :date_end between ps.ps_date_begin and ps.ps_date_end
            --(:date_begin <= ps.ps_date_end and :date_end >= ps.ps_date_begin)
      into :fio, :jl_short, :c_id, ps_date_begin, ps_is_working
  do
  begin
    select f_work_type_to_symbol(type_sat_day_1, res_sat_day_1, :tag), f_work_type_to_symbol(type_sat_day_2, res_sat_day_2, :tag),
           f_work_type_to_symbol(type_sat_day_3, res_sat_day_3, :tag), f_work_type_to_symbol(type_sat_day_4, res_sat_day_4, :tag),
           f_work_type_to_symbol(type_sat_day_5, res_sat_day_5, :tag), f_work_type_to_symbol(type_sat_day_6, res_sat_day_6, :tag),
           f_work_type_to_symbol(type_sat_day_7, res_sat_day_7, :tag), f_work_type_to_symbol(type_sat_day_8, res_sat_day_8, :tag),
           f_work_type_to_symbol(type_sat_day_9, res_sat_day_9, :tag), f_work_type_to_symbol(type_sat_day_10, res_sat_day_10, :tag),
           f_work_type_to_symbol(type_sat_day_11, res_sat_day_11, :tag),
           f_work_type_to_symbol(type_sat_day_12, res_sat_day_12, :tag),
           f_work_type_to_symbol(type_sat_day_13, res_sat_day_13, :tag),
           f_work_type_to_symbol(type_sat_day_14, res_sat_day_14, :tag),
           f_work_type_to_symbol(type_sat_day_15, res_sat_day_15, :tag),
           f_work_type_to_symbol(type_sat_day_16, res_sat_day_16, :tag),
           f_work_type_to_symbol(type_sat_day_17, res_sat_day_17, :tag),
           f_work_type_to_symbol(type_sat_day_18, res_sat_day_18, :tag),
           f_work_type_to_symbol(type_sat_day_19, res_sat_day_19, :tag),
           f_work_type_to_symbol(type_sat_day_20, res_sat_day_20, :tag),
           f_work_type_to_symbol(type_sat_day_21, res_sat_day_21, :tag),
           f_work_type_to_symbol(type_sat_day_22, res_sat_day_22, :tag),
           f_work_type_to_symbol(type_sat_day_23, res_sat_day_23, :tag),
           f_work_type_to_symbol(type_sat_day_24, res_sat_day_24, :tag),
           f_work_type_to_symbol(type_sat_day_25, res_sat_day_25, :tag),
           f_work_type_to_symbol(type_sat_day_26, res_sat_day_26, :tag),
           f_work_type_to_symbol(type_sat_day_27, res_sat_day_27, :tag),
           f_work_type_to_symbol(type_sat_day_28, res_sat_day_28, :tag),
           f_work_type_to_symbol(type_sat_day_29, res_sat_day_29, :tag),
           f_work_type_to_symbol(type_sat_day_30, res_sat_day_30, :tag),
           f_work_type_to_symbol(type_sat_day_31, res_sat_day_31, :tag),
           is_ready,
           total_days
    from hp_timesheet_list_tmp(:reg_id, :c_id, :sat_year_month)
    into :sat_day_1, :sat_day_2, :sat_day_3, :sat_day_4, :sat_day_5, :sat_day_6, :sat_day_7, :sat_day_8, :sat_day_9,
         :sat_day_10, :sat_day_11, :sat_day_12, :sat_day_13, :sat_day_14, :sat_day_15, :sat_day_16, :sat_day_17,
         :sat_day_18, :sat_day_19, :sat_day_20, :sat_day_21, :sat_day_22, :sat_day_23, :sat_day_24, :sat_day_25,
         :sat_day_26, :sat_day_27, :sat_day_28, :sat_day_29, :sat_day_30, :sat_day_31, :is_ready, :sat_days_total;

    if (db_id in (39801, 41701) and ps_is_working = 0) then
    begin
      tmp_date_integer = extract(year from ps_date_begin) * 10000 + extract(month from ps_date_begin) * 100 + extract(day from ps_date_begin);
      sat_day_1 = iif(year_inp * 10000 + month_inp * 100 + 1 >= tmp_date_integer, '-', :sat_day_1);
      sat_day_2 = iif(year_inp * 10000 + month_inp * 100 + 2 >= tmp_date_integer, '-', :sat_day_2);
      sat_day_3 = iif(year_inp * 10000 + month_inp * 100 + 3 >= tmp_date_integer, '-', :sat_day_3);
      sat_day_4 = iif(year_inp * 10000 + month_inp * 100 + 4 >= tmp_date_integer, '-', :sat_day_4);
      sat_day_5 = iif(year_inp * 10000 + month_inp * 100 + 5 >= tmp_date_integer, '-', :sat_day_5);
      sat_day_6 = iif(year_inp * 10000 + month_inp * 100 + 6 >= tmp_date_integer, '-', :sat_day_6);
      sat_day_7 = iif(year_inp * 10000 + month_inp * 100 + 7 >= tmp_date_integer, '-', :sat_day_7);
      sat_day_8 = iif(year_inp * 10000 + month_inp * 100 + 8 >= tmp_date_integer, '-', :sat_day_8);
      sat_day_9 = iif(year_inp * 10000 + month_inp * 100 + 9 >= tmp_date_integer, '-', :sat_day_9);
      sat_day_10 = iif(year_inp * 10000 + month_inp * 100 + 10 >= tmp_date_integer, '-', :sat_day_10);
      sat_day_11 = iif(year_inp * 10000 + month_inp * 100 + 11 >= tmp_date_integer, '-', :sat_day_11);
      sat_day_12 = iif(year_inp * 10000 + month_inp * 100 + 12 >= tmp_date_integer, '-', :sat_day_12);
      sat_day_13 = iif(year_inp * 10000 + month_inp * 100 + 13 >= tmp_date_integer, '-', :sat_day_13);
      sat_day_14 = iif(year_inp * 10000 + month_inp * 100 + 14 >= tmp_date_integer, '-', :sat_day_14);
      sat_day_15 = iif(year_inp * 10000 + month_inp * 100 + 15 >= tmp_date_integer, '-', :sat_day_15);
      sat_day_16 = iif(year_inp * 10000 + month_inp * 100 + 16 >= tmp_date_integer, '-', :sat_day_16);
      sat_day_17 = iif(year_inp * 10000 + month_inp * 100 + 17 >= tmp_date_integer, '-', :sat_day_17);
      sat_day_18 = iif(year_inp * 10000 + month_inp * 100 + 18 >= tmp_date_integer, '-', :sat_day_18);
      sat_day_19 = iif(year_inp * 10000 + month_inp * 100 + 19 >= tmp_date_integer, '-', :sat_day_19);
      sat_day_20 = iif(year_inp * 10000 + month_inp * 100 + 20 >= tmp_date_integer, '-', :sat_day_20);
      sat_day_21 = iif(year_inp * 10000 + month_inp * 100 + 21 >= tmp_date_integer, '-', :sat_day_21);
      sat_day_22 = iif(year_inp * 10000 + month_inp * 100 + 22 >= tmp_date_integer, '-', :sat_day_22);
      sat_day_23 = iif(year_inp * 10000 + month_inp * 100 + 23 >= tmp_date_integer, '-', :sat_day_23);
      sat_day_24 = iif(year_inp * 10000 + month_inp * 100 + 24 >= tmp_date_integer, '-', :sat_day_24);
      sat_day_25 = iif(year_inp * 10000 + month_inp * 100 + 25 >= tmp_date_integer, '-', :sat_day_25);
      sat_day_26 = iif(year_inp * 10000 + month_inp * 100 + 26 >= tmp_date_integer, '-', :sat_day_26);
      sat_day_27 = iif(year_inp * 10000 + month_inp * 100 + 27 >= tmp_date_integer, '-', :sat_day_27);
      sat_day_28 = iif(year_inp * 10000 + month_inp * 100 + 28 >= tmp_date_integer, '-', :sat_day_28);
      sat_day_29 = iif(year_inp * 10000 + month_inp * 100 + 29 >= tmp_date_integer, '-', :sat_day_29);
      sat_day_30 = iif(year_inp * 10000 + month_inp * 100 + 30 >= tmp_date_integer, '-', :sat_day_30);
      sat_day_31 = iif(year_inp * 10000 + month_inp * 100 + 31 >= tmp_date_integer, '-', :sat_day_31);
    end

    -- переработка в днях за 2 мес.
    sat_days_total_overwork = 0;
    if (calc_overwork) then
    begin
      sat_days_total_prev = coalesce((
        select
          total_days
        from hp_timesheet_list_tmp(:reg_id, :c_id, :sat_year_month - 1)), 0);
      sat_days_total_overwork = sat_days_total + sat_days_total_prev - norma_working_days_tm;
      if (sat_days_total_overwork < 0) then
        sat_days_total_overwork = 0;
    end

    suspend;
  end
end
