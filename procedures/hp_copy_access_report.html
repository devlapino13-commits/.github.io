CREATE PROCEDURE HP_COPY_ACCESS_REPORT(
    RL_ID_ORIG INTEGER,
    RL_ID_NEW INTEGER)
RETURNS (
    RL_MEMO VARCHAR(255))
AS
declare variable uniqtab_rolename integer;
declare variable uniqtab_menulist integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;
  select ukt_id
  from hp_get_uniqtable_id('Menulist')
  into :uniqtab_menulist;

  if (exists(select *
             from reportlist
             where rl_id = :rl_id_orig)) then
  begin
    if (exists(select *
               from reportlist
               where rl_id = :rl_id_new)) then
    begin
      insert into permissions (pm_id, pm_object_kind, pm_object_id, pm_subject_kind, pm_subject_id, pm_date_begin,
                               pm_date_end, pm_access_kind, pm_is_delete, pm_from_un_id)
      select -1, pm_object_kind, pm_object_id, pm_subject_kind, :rl_id_new, pm_date_begin, pm_date_end, pm_access_kind,
             pm_is_delete, pm_from_un_id
      from permissions
      where pm_subject_kind = :uniqtab_menulist and
            pm_subject_id = :rl_id_orig and
            pm_is_delete = 0 and
            pm_object_kind = :uniqtab_rolename -- Чтоб не копировать исключения по пользователю
            and
            (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
            (coalesce(pm_date_end, '31.12.2999') >= current_date);
      rl_memo = 'Права доступов отчета успешно скопированы!';
    end
    else
    begin
      rl_memo = 'ID нового отчета не правильно указан!';
    end
  end
  else
  begin
    rl_memo = 'ID основного отчета не правильно указан!';
  end
  suspend;
end
