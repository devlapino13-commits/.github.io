CREATE PROCEDURE HP_KONT_FOR_RESTRUCT_KZ_CMD(
    K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable res_is_error d_smallint;
declare variable db_id d_integer;
declare variable is_restruct d_smallint;
declare variable restruct_date d_date;
begin
  db_id = f_get_db_id();
  res_text = '';
  rep_date = coalesce(:rep_date, current_date);

  select is_restruct,
         restruct_date
  from aarep_get_restructuring(:k_id, :rep_date, :db_id)
  into :is_restruct,
       :restruct_date;

  if (coalesce(is_restruct, 0) = 0) then
  begin
    select hdr.res_is_error,
           hdr.res_text
    from hp_do_reschedule_kz(:k_id, :rep_date) hdr
    into :res_is_error,
         :res_text;
  end
  else
    res_text = 'Контракт: ' || 'был реструтуризован ' || :restruct_date ||'.';

  suspend;
end
