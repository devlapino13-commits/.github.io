CREATE PROCEDURE HP_GKB_FRAUD_ROBOT
AS
declare variable MA_ID D_ID;
declare variable C_ID D_ID;
declare variable MAC_ID D_ID;
declare variable EIC_ID D_ID;
declare variable OK_MINIAPP D_INTEGER;
declare variable REASON D_STR_255;
declare variable IS_FINAL D_SMALLINT;
declare variable SC_STATUS D_SMALLINT;
declare variable SC_ANSWER_DATE D_DATE;
declare variable SC_QUERY_DT D_DATETIME;
declare variable SC_MAX_DELAY D_INTEGER;
declare variable AF_CRITICAL D_INTEGER;
declare variable AF_TEXT D_STR_255;
declare variable AF_FOUND D_SMALLINT;
declare variable MIL_STATUS D_SMALLINT;
declare variable MIL_RECEIVED D_SMALLINT;
declare variable IS_OVERDUE D_SMALLINT;
declare variable IS_BANKROT D_SMALLINT;
declare variable PKB_EIL_ID D_ID;
declare variable PKB_EIL_STATUS D_SMALLINT;
declare variable PKB_SKIP D_SMALLINT;
declare variable AFM_STATUS D_SMALLINT;
declare variable AFM_TEXT D_STR_255;
declare variable KDN_IS_GAMBLER D_SMALLINT;
declare variable KDN_MAX D_CURRENCY;
declare variable KDN_VALUE D_CURRENCY;
declare variable KDN_IS_SUSN D_SMALLINT;   -- RFCKZ-996
begin
  -- RFCKZ-877: робот отправки запросов Fraud Score ГКБ
  if (f_get_db_id() <> 39801) then
    exit;

  if (coalesce(f_get_const('gkb_fraud_is_active'), '0') <> '1') then
    exit;

  ok_miniapp = f_objkind('miniapp', 'id');
  sc_max_delay = cast(f_get_const('stop_credit_service_max_delay_seconds') as d_integer);

  eic_id = (select eic.eic_id
            from extinfocfg eic
            where eic.eic_code = 'gkb_pay_checks_core');

  if (eic_id is null) then
    exit;

  for select ma.ma_id, mac.mac_from_c_id, mac.mac_id
      from miniapp ma
      inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id and
            mac.mac_amount > 0
      where ma.ma_date_app = current_date and
            ma.ma_is_active = 1 and
            ma.ma_date_end = '31.12.9999' and
            -- договор ещё не создан
            not exists(select 1
                       from kontract k
                       where k.k_from_ma_app_id = ma.ma_id) and
            -- запрос paycheck по этой заявке ещё не отправлялся
            not exists(select 1
                       from extinfolist eil
                       where eil.eil_from_eic_id = :eic_id and
                             eil.eil_object_kind = :ok_miniapp and
                             eil.eil_object_id = ma.ma_id)
      into :ma_id, :c_id, :mac_id
  do
  begin
    reason = '';
    is_final = 0;

    /*--------------------------------------------------------------------*/
    /* 1. Стоп-кредит — зеркалит CM_KZ_0003.                               */
    /*    При -2 (после таймаута) и -3 приложенный документ egov снимает    */
    /*    блокировку, поэтому проверяем f_is_stopcred_doc_loaded.          */
    /*--------------------------------------------------------------------*/
    sc_status = null;
    sc_answer_date = null;
    sc_query_dt = null;
    select status, answer_date, query_datetime
    from hp_stop_credit_query_status(:ok_miniapp, :ma_id)
    into :sc_status, :sc_answer_date, :sc_query_dt;

    if (sc_status = 1 and f_is_stopcred_actual(:sc_answer_date) = 1) then
    begin
      reason = 'Стоп-кредит: у клиента действует стоп-кредит';
      is_final = 1;
    end
    else if (sc_status in (0, 1) and f_is_stopcred_actual(:sc_answer_date) = 0) then
      reason = 'Стоп-кредит: полученный статус уже не актуален, нужен новый запрос';
    else if (sc_status = -1) then
      reason = 'Стоп-кредит: запрос ещё не осуществлялся';
    else if (sc_status = -2) then
    begin
      if (datediff(second, sc_query_dt, current_timestamp) >= sc_max_delay) then
      begin
        if (f_is_stopcred_doc_loaded(:ok_miniapp, :ma_id) = 0) then
          reason = 'Стоп-кредит: сервис не отвечает, приложите документ или дождитесь ответа';
      end
      else
        reason = 'Стоп-кредит: ответ от сервиса ещё не получен';
    end
    else if (sc_status = -3) then
    begin
      if (f_is_stopcred_doc_loaded(:ok_miniapp, :ma_id) = 0) then
        reason = 'Стоп-кредит: сервис ответил с ошибкой, приложите документ';
    end

    /*--------------------------------------------------------------------*/
    /* 2. Антифрод — оба пути: API-режим и внутренние списки.              */
    /*    При выключенном API hp_antifraud_client_status отдаёт «чисто»,   */
    /*    поэтому списки проверяем отдельно (как CM_KZ_0019).              */
    /*--------------------------------------------------------------------*/
    if (reason = '') then
    begin
      af_critical = null;
      af_text = null;
      af_found = null;
      select res_error_is_critical, res_error_text, client_was_found
      from hp_antifraud_client_status(:c_id)
      into :af_critical, :af_text, :af_found;

      if (coalesce(af_critical, 0) <> 0) then
      begin
        reason = 'Антифрод: ' || coalesce(af_text, 'проверка не пройдена');
        is_final = iif(coalesce(af_found, 0) = 1, 1, 0);
      end
    end

    if (reason = '') then
    begin
      if (exists(select 1 from hp_check_client_fio_ft(:c_id, 1))) then
      begin
        reason = 'Антифрод: клиент найден в гос. списках';
        is_final = 1;
      end
    end

    /*--------------------------------------------------------------------*/
    /* 3. Призывник — зеркалит CM_KZ_0001.                                 */
    /*    ВАЖНО: сравнение без coalesce. Если записи в military_check_query */
    /*    нет (mcq_received = null) — проверка не применима (например,     */
    /*    клиент женского пола), блок её пропускает, робот тоже.           */
    /*--------------------------------------------------------------------*/
    if (reason = '') then
    begin
      mil_status = null;
      mil_received = null;
      select status, mcq_received
      from hp_military_query_status(:c_id)
      into :mil_status, :mil_received;

      if (mil_received = 0 or mil_status in (-2, -3)) then
        reason = 'Призывник: ответ от сервиса ещё не получен';
      else if (mil_status = 1) then
      begin
        reason = 'Призывник: клиент призван на срочную воинскую службу';
        is_final = 1;
      end
    end

    /*--------------------------------------------------------------------*/
    /* 4. Просрочка и банкрот — зеркалит CM_KZ_0010.                        */
    /*    Если по клиенту есть успешный pkb_available_reports и при этом    */
    /*    priority_id = -1, значит кредитной истории нет — проверка         */
    /*    не применима, блок её пропускает.                                */
    /*    Запись в negativ_kib_history ищется ПО КЛИЕНТУ (nkh_last = 1),    */
    /*    без привязки к заявке — как в блоке.                             */
    /*--------------------------------------------------------------------*/
    if (reason = '') then
    begin
      pkb_skip = 0;

      if (exists(select 1
                 from extinfolist eil
                 inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                 where eic.eic_code = 'pkb_available_reports' and
                       eil.eil_from_c_id = :c_id and
                       eil.eil_status = 2)) then
      begin
        if ((select priority_id
             from hp_get_priority_rep_id(:ok_miniapp, :ma_id)) = -1) then
          pkb_skip = 1;   -- у клиента отсутствует кредитная история
      end

      if (pkb_skip = 0) then
      begin
        is_overdue = null;
        is_bankrot = null;
        pkb_eil_id = null;
        pkb_eil_status = null;

        select nkh.nkh_is_overdue, nkh.nkh_is_bankrot,
               nkh.nkh_from_eil_id, eil.eil_status
        from negativ_kib_history nkh
        inner join extinfolist eil on eil.eil_id = nkh.nkh_from_eil_id
        where nkh.nkh_from_c_id = :c_id and
              nkh.nkh_last = 1
        rows 1
        into :is_overdue, :is_bankrot, :pkb_eil_id, :pkb_eil_status;

        if (coalesce(pkb_eil_id, 0) = 0) then
          reason = 'ПКБ: запрос на негативную историю клиента не отправлен';
        else if (pkb_eil_status = -2) then
          reason = 'ПКБ: при запросе негативной истории возникла ошибка';
        else if (pkb_eil_status < 2) then
          reason = 'ПКБ: запрос на негативную историю ещё не обработан';
        else if (is_bankrot = 1) then
        begin
          reason = 'ПКБ: клиент проходит процедуру банкротства';
          is_final = 1;
        end
        else if (is_overdue = 1) then
        begin
          reason = 'ПКБ: у клиента есть задолженность 30 дней и более';
          is_final = 1;
        end
      end
    end

    /*--------------------------------------------------------------------*/
    /* 5. ПОД/ФТ (АФМ) — зеркалит CM_KZ_0012.                              */
    /*    Пропускаем только status = 1 (not found).                        */
    /*    Статусы >= 2 — клиент найден в списках, запрос не отправляем.    */
    /*--------------------------------------------------------------------*/
    /*if (reason = '') then
    begin
      afm_status = null;
      afm_text = null;
      select status, res_text
      from hp_afm_get_status('afm', :c_id, :ma_id)
      into :afm_status, :afm_text;

      if (coalesce(afm_status, -1) <> 1) then
      begin
        reason = 'ПОД/ФТ: ' || coalesce(afm_text, 'проверка не пройдена');
        is_final = iif(coalesce(afm_status, -1) >= 2, 1, 0);
      end
    end */

    /*--------------------------------------------------------------------*/
    /* 6. КДН — зеркалит CM_KZ_0015.                                       */
    /*    Предельное значение зависит от признака игромана из негативной   */
    /*    истории ПКБ: iif(is_gambler = 0, 0.5, 0.25).                     */
    /*    ВНИМАНИЕ: если записи в negativ_kib_history нет, is_gambler=null, */
    /*    и iif(null, ...) возвращает ветку else, то есть предел 0.25.     */
    /*    Так работает блок — робот повторяет это поведение.               */
    /*--------------------------------------------------------------------*/
    if (reason = '') then
    begin
      if (not exists(select 1
                     from cfp_anketa_kdn cakdn
                     where cakdn.cakdn_from_c_id = :c_id and
                           cakdn.cakdn_from_mac_id = :mac_id)) then
        reason = 'КДН не рассчитан';
      else
      begin
        kdn_is_gambler = null;
        select nkh.nkh_is_gambler
        from negativ_kib_history nkh
        where nkh.nkh_from_c_id = :c_id and
              nkh.nkh_last = 1
        rows 1
        into :kdn_is_gambler;

        kdn_is_susn = 0;                           -- RFCKZ-996
        select is_susn
        from hp_susn_get_status(:c_id, :ma_id)
        into :kdn_is_susn;

        kdn_max = iif(kdn_is_gambler = 0 and kdn_is_susn = 0, 0.5, 0.25);

        kdn_value = null;
        select cakdn.cakdn_kdn
        from cfp_anketa_kdn cakdn
        where cakdn.cakdn_from_c_id = :c_id and
              cakdn.cakdn_from_mac_id = :mac_id
        rows 1
        into :kdn_value;

        if (kdn_value > kdn_max) then
        begin
          reason = f_format('КДН клиента %1 больше предельного значения %2',
                            round(kdn_value, 2), round(kdn_max, 2));
          is_final = 1;
        end
      end
    end

    /*--------------------------------------------------------------------*/
    /* Результат по заявке                                                */
    /*--------------------------------------------------------------------*/
    if (reason = '') then
    begin
      -- Все стоп-факторы пройдены — отправляем запрос.
      -- Вставляем в extinfolist напрямую (как HP_EXTINFO_INSERT_ANTIFRAUD),
      -- чтобы гарантировать одну запись на заявку и обойти дедупликацию
      -- по клиенту и дате внутри HP_EXTINFO_INSERT.
      insert into extinfolist (eil_from_eic_id, eil_from_c_id, eil_object_kind,
                               eil_object_id, eil_is_json, eil_from_un_id)
      values (:eic_id, :c_id, :ok_miniapp, :ma_id, 1, 1000000000000);

      post_event 'gkb_pay_checks_core';

      delete from gkb_fraud_skip
      where gfs_from_ma_id = :ma_id;
    end
    else
    begin
      update or insert into gkb_fraud_skip
        (gfs_from_ma_id, gfs_reason, gfs_is_final, gfs_datetime)
      values (:ma_id, :reason, :is_final, current_timestamp)
      matching (gfs_from_ma_id);
    end

    -- Ошибка по одной заявке не должна останавливать обработку остальных
    when any do
    begin
      update or insert into gkb_fraud_skip
        (gfs_from_ma_id, gfs_reason, gfs_is_final, gfs_datetime)
      values (:ma_id, 'Ошибка при проверке стоп-факторов, обратитесь в тех. поддержку', 0, current_timestamp)
      matching (gfs_from_ma_id);
    end
  end
end
