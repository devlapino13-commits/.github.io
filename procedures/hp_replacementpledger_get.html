CREATE PROCEDURE HP_REPLACEMENTPLEDGER_GET(
    K_ID ID_TYPE)
RETURNS (
    Z_ID D_INTEGER,
    Z_FROM_K_ID ID_TYPE,
    Z_FROM_C_ID ID_TYPE,
    C_NAME_FIO D_STR_100,
    Z_TYPE D_INTEGER,
    Z_SUBTYPE D_INTEGER,
    Z_NAME D_STR_100,
    Z_FROM_FL_ID ID_TYPE,
    Z_NAME_FIO D_STR_255,
    Z_FROM_ADL_ID ID_TYPE,
    Z_ADDRESS D_STR_255,
    Z_PHONE D_STR_100,
    Z_DOCUMENT D_STR_255,
    Z_MEMO1 VARCHAR(1024),
    Z_MEMO2 D_STR_2048,
    Z_REGISTRY_NO VARCHAR(50),
    Z_REGISTRY_DATE D_DATE,
    Z_REGISTRY_AGENCY VARCHAR(80),
    Z_COST_MARKET CURR_TYPE,
    Z_COST_LIKVID CURR_TYPE,
    Z_COST_KONTRACT CURR_TYPE,
    Z_DATE_BEGIN D_DATE,
    Z_DATE_END D_DATE)
AS
begin
  for
    select
      z_id,
      z_from_k_id,
      z_from_c_id,
      c_name_fio,
      z_type,
      z_subtype,
      z_name,
      z_from_fl_id,
      z_name_fio,
      z_from_adl_id,
      adl_address_str,
      z_phone,
      z_document,
      z_memo1,
      z_memo2,
      z_registry_no,
      z_registry_date,
      z_registry_agency,
      z_cost_market,
      z_cost_likvid,
      z_cost_kontract,
      z_date_begin,
      z_date_end
    from zalog
      inner join kontstate
        on ks_from_k_id = z_from_k_id
          and current_date between ks_date_begin and ks_date_end
          and ks_ol_is_delete = 0
          and ks_state = 1
      inner join clients
        on c_id = z_from_c_id
      left join addrlist
        on adl_id = z_from_adl_id
    where z_from_k_id = :k_id
      and z_subtype = 4
      --and z_date_end is null
    order by z_id desc
  into
    :z_id,
    :z_from_k_id,
    :z_from_c_id,
    :c_name_fio,
    :z_type,
    :z_subtype,
    :z_name,
    :z_from_fl_id,
    :z_name_fio,
    :z_from_adl_id,
    :z_address,
    :z_phone,
    :z_document,
    :z_memo1,
    :z_memo2,
    :z_registry_no,
    :z_registry_date,
    :z_registry_agency,
    :z_cost_market,
    :z_cost_likvid,
    :z_cost_kontract,
    :z_date_begin,
    :z_date_end
  do
  begin
    suspend;
  end
end
