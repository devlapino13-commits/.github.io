CREATE PROCEDURE UXP$UXALIAS_LINKS(
    MAIN_UXA_UUID D_UUID,
    UXA_UUID D_UUID = null,
    LAST_PATH D_STR_8100 = null,
    UXA_UUID_CODE D_STR_CODE = null)
RETURNS (
    RES_ID D_INTEGER,
    RES_CODE D_STR_CODE,
    RES_PARENT_CODE D_STR_CODE,
    RES_KIND D_STR_CODE,
    RES_CAPTION D_STR_100,
    RES_IMAGE D_INTEGER,
    RES_IS_ALIAS D_BOOL,
    RES_PATH D_STR_8100,
    RES_INACTIVE D_BOOL,
    RES_UA_UUID D_UUID)
AS
declare variable ml_id d_id;
declare variable tmp_alias d_uuid;
declare variable tmp_path d_str_8100;
declare variable uxa_name d_str_100;
declare variable uxa_ml_id d_id;
declare variable uxa_inactive d_bool;
declare variable mm_uuid d_uuid;
declare variable uxa_image_id d_integer;
declare variable uxa_suspended d_bool;
begin
  if (coalesce(main_uxa_uuid, '') = '') then
    main_uxa_uuid = uxf$uniq(uxa_uuid_code);

  mm_uuid = uxf$mainmenu();
  uxa_suspended = 0;

  res_is_alias = 0;
  res_path = coalesce(last_path, '');
  if (uxa_uuid is null) then
  begin
    res_is_alias = 1;
    uxa_uuid = main_uxa_uuid;
    res_path = res_path || '[' || uuid_to_char(main_uxa_uuid) || ']';
  end

  res_id = 0;
  res_code = uuid_to_char(uxa_uuid);
  res_kind = 'uxalias';

  select uxa_name, uxa_fs_inactive
  from ux$alias
  where uxa_uuid = :uxa_uuid
  into uxa_name, uxa_inactive;

  res_caption = uxa_name;
  res_inactive = uxa_inactive;

  select ua_uuid
  from ux$action
  where ua_owner = :mm_uuid and
        ua_uxa_uuid = :uxa_uuid
  rows 1
  into res_ua_uuid;

  if (res_ua_uuid is not null) then
  begin
    select item_id
    from uxp$actionmenu(:res_ua_uuid)
    into uxa_ml_id;
  end

  ml_id = uxa_ml_id;

  if (res_is_alias = 1) then
    uxa_image_id = 19; -- Форма
  else
    uxa_image_id = 20; -- Серая форма

  res_kind = 'menuitem';
  if (ml_id is not null) then
  begin
    res_image = uxa_image_id;
    res_parent_code = ml_id;
    uxa_suspended = 1;

    suspend;
    res_is_alias = 0;

    while (ml_id is not null) do
    begin
      res_code = ml_id;
  
      select ml_parent_id, ml_caption
      from menulist
      where ml_id = :ml_id
      into ml_id, res_caption;
  
      res_parent_code = ml_id;
  
      if (res_path not containing '[' || res_code || ']') then
      begin
        res_image = 36; -- Желтая папка
        res_path = res_path || '[' || res_code || ']';
        res_id = res_id + 1;
        suspend;
      end
  
      res_kind = 'mainmenu';
    end
  end

  res_ua_uuid = null;
  res_code = uuid_to_char(uxa_uuid);
  res_kind = 'uxalias';

  res_caption = uxa_name;
  res_inactive = uxa_inactive;
  ml_id = uxa_ml_id;
  res_image = uxa_image_id;

  tmp_alias = uxa_uuid;
  for select uuid_to_char(uaa_uxa_uuid), max(ua_uuid)
  from ux$action, ux$aliasaction, ux$alias
  where ua_uxa_uuid = :tmp_alias and
        ua_uuid = uaa_ua_uuid and
        uaa_uxa_uuid = uxa_uuid
  group by 1
  into :res_parent_code, res_ua_uuid
  do
  begin
    res_kind = 'uxalias';
    res_id = res_id + 1;
    uxa_suspended = 1;

    if (res_is_alias = 1) then
      res_ua_uuid = null;

    suspend;
  end

  tmp_alias = uxa_uuid;
  for select distinct uaa_uxa_uuid
  from ux$action, ux$aliasaction, ux$alias
  where ua_uxa_uuid = :tmp_alias and
        ua_uuid = uaa_ua_uuid and
        uaa_uxa_uuid = uxa_uuid
  into :uxa_uuid
  do
  begin
    if (res_path containing '[' || uuid_to_char(uxa_uuid) || ']') then
      Continue;

    res_path = res_path || '[' || uuid_to_char(uxa_uuid) || ']';
    tmp_path = res_path;

    for select res_code, res_parent_code, res_kind, res_caption, res_image,
               res_is_alias, res_path, res_inactive, res_ua_uuid
    from uxp$uxalias_links(:main_uxa_uuid, :uxa_uuid, :tmp_path)
    into :res_code, :res_parent_code, :res_kind, :res_caption,
         :res_image, :res_is_alias, :res_path, :res_inactive, res_ua_uuid
    do
    begin
      res_id = res_id + 1;
      suspend;
    end
  end

  res_ua_uuid = null;
  if (uxa_suspended = 0) then
  begin
    res_id = res_id + 1;

    if (res_is_alias = 0) then
    begin
      select ua_uuid
      from ux$action
      where ua_owner = :mm_uuid and
            ua_uxa_uuid = :uxa_uuid
      into res_ua_uuid;
    end

    suspend;
  end
end
