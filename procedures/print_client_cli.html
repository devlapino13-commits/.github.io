    ADD_PARAMS D_STR_LONG = null)

    ANK_FILLED_FIRST_TIME_TXT STR_100,
    ANK_UPDATE_DATA_TXT STR_100,
    C_NAME_FIO STR_100,
    RES_NON_RESIDENT INTEGER,
    C_BIRTH_DATE DATE,
    C_BIRTH_CITY STR_100,
    C_NATION_STR STR_20,
    CL_SEX SMALLINT,
    C_CITIZEN_STR STR_20,
    C_FAMILY_STR D_STR_100,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_INN VARCHAR(20),
    C_ADDRESS_YUR STR_255,
    C_ADDRESS_FACT STR_255,
    LAST_CELL_MAIN STR_20,
    C_EMAIL VARCHAR(100),
    C_IS_PUBLIC INTEGER,
    C_IS_BENEFIC_OWNER INTEGER,
    REGISTR_TYPE STR_100,
    REGISTR_DOC_NO STR_100,
    REGISTR_DATE_BEGIN DATE,
    REGISTR_DATE_END DATE,
    REGISTR_ISSUED_BY STR_255,
    REGISTR_MEMO_1 STR_255,
    REGISTR_MEMO_2 STR_255,
    REGISTR_ADRESS STR_255,
    REGISTR_IIN STR_255,
    PANT_TYPE STR_100,
    PANT_DOC_NO STR_100,
    PANT_DATE_BEGIN DATE,
    PANT_DATE_END DATE,
    PANT_ISSUED_BY STR_255,
    PANT_MEMO_1 STR_255,
    PANT_MEMO_2 STR_255,
    MAC_LOAN_TARGET_ORIGINAL STR_255,
    C_PFT_RISK_KOEF INTEGER,
    C_PFT_RISK_MEMO VARCHAR(1024),
    PRINT_DATE DATE,
    CREATE_DATE DATE,
    LO_FIO STR_100,
    LO_ID ID_TYPE,
    LO_JL_NAME STR_100,
    MALE_TXT STR_10,
    FEMALE_TXT STR_20,
    ANK_HIGH_RISK_TXT STR_20,
    ANK_LOW_RISK_TXT STR_20,
    C_ORG_TYPE_LONG STR_255,
    C_NAME_FULL STR_255,
    C_ALTERNAT_FIO STR_100,
    C_FORM_STATUS_SHORT STR_20,
    C_OKPO STR_20,
    C_DATETIME TIMESTAMP,
    QR_CODE D_STR_LONG,
    MAC_LOAN_TARGET STR_255,
    KC_LOAN_TARGET_ORIGINAL STR_255,
    C_AUTHORIZED_CAPITAL D_STR_255,
    C_BRANCHES D_STR_255,
    C_DOC_INFORM D_STR_255,
    C_GOALS D_STR_255,
    C_PASS_WHO_CODE D_STR_255,
    REG_NAME VARCHAR(50),
    PL_NAME D_STR_100,
    PANT_INFOLIC1 D_STR_2048,
    PANT_INFOLIC2 D_STR_2048,
    PANT_INFOLIC3 D_STR_2048,
    KML_BANK_NAME VARCHAR(255),
    KML_BANK_NO VARCHAR(50),
    KML_BANK_BIK ID_TYPE,
    KML_POLIS_TYPE STR_100,
    KML_POLIS_DOC_NO STR_100,
    KML_POLIS_DATE_BEGIN DATE,
    KML_POLIS_DATE_END DATE,
    KML_POLIS_ISSUED_BY STR_255,
    HR_REG_NAME VARCHAR(50),
    VER_FIO VARCHAR(255),

    KSR_KONT_TITUL VARCHAR(1024),
    EDL_CREATE_DATETIME D_DATETIME,
    EDL_ID D_ID,
    EDL_CREATE_DATE D_DATE,
    PLACE_NAME D_STR_255,
    LAST_YEAR_DATE D_STR_100,
    DOC_NO_LEGAL D_STR_100,
    DOC_DATE_LEGAL D_DATE,
    DOC_ISSUED_BY_LEGAL D_STR_255,
    PK_NAME D_STR_255,
    EDL_BLANK_DATE D_DATE,
    C_TYPE SMALLINT,
    PANT_MEMO3 D_STR_255,
    C_DOC_TYPE_STR D_STR_100)
AS
declare variable db_id d_integer;
declare variable kf_id id_type;
declare variable rd_id id_type;
declare variable address_type smallint;
declare variable c_address str_255;
declare variable tmp_id d_id;
declare variable bs_name str_100;
declare variable mac_busuness_subtype integer;
declare variable c_from_un_id id_type;
declare variable un_from_c_id id_type;
declare variable ver_c_id id_type;
declare variable rd_id_1 id_type;
declare variable c_doc_type d_integer;
begin
  db_id = f_get_db_id();

  edl_id = iif(add_params starting '{', f_json_str_get_value(add_params, 'edl_id', null),add_params);

  male_txt = 'муж';
  female_txt = 'жен';
  ank_high_risk_txt = 'высокий';

  if (not exists(select * from clients
                 where c_id = :c_id)) then
  begin
    tmp_id = f_json_str_get_value(add_params, 'c_id', null);
    if (tmp_id is not null) then
      c_id = tmp_id;
  end

  ank_low_risk_txt = 'низкий';
  last_cell_main = f_cont_info(0, :c_id, 'cell_main', 'ok');

  select rd_id
  from refdata
  where rd_code = 'agreementkinds.ipf_sellers'
  into :rd_id;

  select rd_id
  from refdata
  where rd_code = 'agreementkinds.tsp_and_ipf'
  into :rd_id_1;

  first_time = 0;
  if (exists(select *
             from kontmat kml, agreements ag, kontstate ks
             where kml_from_ag_id = ag_id and
                   kml_object_kind = 9 and
                   kml_from_ml_id is not null and
                   ag_object_kind = 0 and
                   ag.ag_from_rd_id in (:rd_id, :rd_id_1) and
                   ag_object_id = :c_id and
                   ks.ks_date_begin < current_date and
                   ks.ks_from_k_id = kml.kml_object_id and
                   current_date between ks.ks_date_begin and ks.ks_date_end and
                   ks.ks_ol_is_delete = 0 and
                   ks.ks_state = 1
             order by kml_id)) then
    first_time = 1;

  ank_filled_first_time_txt = 'заполняется впервые';
  ank_update_data_txt = 'обновление анкетных данных';
  print_date = current_date;

  select cl.c_name_fio, cl.c_sex, cl.c_pass_no, cl.c_pass_date, cl.c_pass_who,
         cl.c_pass_date_expire, cl.c_inn, cl.c_email, cl.c_pft_risk_koef, cl.c_pft_risk_memo, nat.rd_name, city.rd_name,
         family.rd_name, cl.c_birth_date, cl.c_birth_city, cl.c_type, cl.c_org_type_long, cl.c_name_full,
         cl.c_alternat_fio, cl.c_form_status_short, cl.c_okpo, c_datetime,
         c_authorized_capital, c_branches, c_doc_inform, c_goals, c_is_benefic_owner,c_pass_who_code, c_doc_type
  from clients cl, refdata nat, refdata city, refdata family
  where cl.c_id = :c_id and
        nat.rd_id = cl.c_nation and
        city.rd_id = cl.c_citizen and
        family.rd_id = cl.c_family
  into :c_name_fio, cl_sex, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire,
       c_inn, c_email, c_pft_risk_koef, c_pft_risk_memo, c_nation_str, c_citizen_str, c_family_str, c_birth_date,
       c_birth_city, c_type, c_org_type_long, c_name_full, c_alternat_fio, c_form_status_short, c_okpo, c_datetime,
       c_authorized_capital, c_branches, c_doc_inform, c_goals, c_is_benefic_owner,c_pass_who_code, c_doc_type;

  -- тип документа rfckg-4225
  select pk_name
  from passkind
  where pk_id = :c_doc_type
  into c_doc_type_str;

  -- использовать данные из perstemp, если нет измененных персональных данных, то тогда из clients
  if (f_tmp_const('use_perstemp') = '1') then
  begin
    select pte_name_fio, pte_sex, pte_pass_no, pte_pass_date, pte_pass_who,
           pte_pass_date_expire, pte_inn, pte_birth_date, pte_birth_city,
           pte_alternat_fio, pte_pass_who_code
    from perstemp
    where pte_from_c_id = :c_id and
          pte_data_state = 2
    into c_name_fio, cl_sex, c_pass_no, c_pass_date, c_pass_who,
         c_pass_date_expire, c_inn, c_birth_date, c_birth_city,
         c_alternat_fio, c_pass_who_code;
  end

  -- board-5325
  -- если последнее изменение данных клиента были сделаны верификатором (пользователь без кадрового приказа) отображать информацию до верификатора
  with q as (
    select cll_from_un_id, cll_datetime
    from clientlog cll
    where cll.cll_from_c_id = :c_id
    union all
    select cl.c_from_un_id, cl.c_datetime
    from clients cl
    where cl.c_id = :c_id)
        select cll_from_un_id
        from q
        inner join username un on un.un_id = cll_from_un_id
        left join hr_pstatus ps on ps.ps_from_c_id = un.un_from_c_id
        where ps.ps_id is not null and
              ps.ps_is_working = 1 and
              ps.ps_is_on_staff = 1 and
              cast(cll_datetime as date) between ps.ps_date_begin and ps.ps_date_end
        order by cll_datetime desc
        rows 1
        into :c_from_un_id;
  -- board-5325 конец

  select bc.bc_account_no, bc.bc_from_b_id, b_name
  from bankclient bc, banks
  where bc.bc_from_c_id = :c_id and
        bc.bc_is_active = 1 and
        bc.bc_from_b_id = b_id rows 1
  into kml_bank_no, kml_bank_bik, kml_bank_name;

  kml_polis_type = null;
  kml_polis_doc_no = null;
  kml_polis_date_begin = null;
  kml_polis_date_end = null;
  kml_polis_issued_by = null;

  select pk.pk_name, cd.cds_doc_no, cd.cds_doc_date_begin, cd.cds_doc_date_end, cd.cds_doc_issued_by
  from clientdocs cd
  inner join passkind pk on cd.cds_from_pk_id = pk_id
  inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
  where cds_from_c_id = :c_id and
        current_date between cds_doc_date_begin and cds_doc_date_end and
        cds_is_delete = 0 and
        pdt.pdt_code = 'soc_polis' rows 1
  into :kml_polis_type, :kml_polis_doc_no, :kml_polis_date_begin, :kml_polis_date_end, :kml_polis_issued_by;

  un_from_c_id = null;
  select u.un_from_c_id
  from username u
  where u.un_id = :c_from_un_id
  into :un_from_c_id;

  hr_reg_name = null;
  select reg.reg_name
  from hr_pstatus hrps
  inner join region reg on reg.reg_unicode = ps_from_reg_id
  where hrps.ps_from_c_id = :un_from_c_id and
        hrps.ps_is_working = 1 and
        current_date between hrps.ps_date_begin and hrps.ps_date_end
  into :hr_reg_name;
  c_is_public = 0;
  -- бенефициарный владелец (для юриков)
  if (c_type = 1) then
    c_is_benefic_owner =
      iif(exists(
            select *
            from clientbenefic
            where cb_from_c_id = :c_id
              and cb_is_owner = 1), 1, 0);

  -- публичное должностное лицо
  if (exists(select *
             from clients c
             inner join clientbenefic cb on c.c_id = cb.cb_ben_from_c_id
             where cb.cb_from_c_id = :c_id and
                   coalesce(c.c_is_public, 0) = 1)) then
    c_is_public = 1;

  if (exists(select *
             from clients c
             inner join public_persons pp on pp.pp_from_c_id = c.c_id and
                   pp.pp_is_delete = 0
             where c.c_id = :c_id and
                   coalesce(c.c_is_public, 0) = 1)) then
    c_is_public = 1;

  c_address_yur = null;
  c_address_fact = null;
  select res_non_resident
  from hp_get_client_type(:c_id, null, null, current_date)
  into :res_non_resident;
  for select adc_address_type, adc_address_str
      from hp_addrclient(:c_id)
      into :address_type, :c_address
  do
  begin
    if (address_type = 1) then
    begin
      select adl_address_str
      from perstemp, addrlist
      where pte_from_c_id = :c_id and
            pte_data_state = 2 and
            pte_adl_id_1 = adl_id
      into c_address;

      c_address_yur = :c_address;
    end
    if (address_type = 2) then
    begin
      select adl_address_str
      from perstemp, addrlist
      where pte_from_c_id = :c_id and
            pte_data_state = 2 and
            pte_adl_id_2 = adl_id
      into c_address;

      c_address_fact = :c_address;
    end
  end

  if (c_name_fio starting 'ип ') then
    mac_loan_target_original = 'ипф';
  else
    mac_loan_target_original = 'получение кредита, финансирования';

  pant_type = null;
  pant_doc_no = null;
  pant_date_begin = null;
  pant_date_end = null;
  pant_issued_by = null;
  pant_memo_1 = '';
  pant_memo_2 = '';

  pant_infolic1 = '';
  pant_infolic2 = '';
  pant_infolic3 = '';
  for
    select pk.pk_name, cd.cds_doc_no, cd.cds_doc_date_begin, cd.cds_doc_date_end, cd.cds_doc_issued_by, cd.cds_memo_1,
           cd.cds_memo_2 ,cd.cds_memo_3
    from clientdocs cd
    inner join passkind pk on cd.cds_from_pk_id = pk_id
    inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
    where cds_from_c_id = :c_id and
          current_date between cds_doc_date_begin and cds_doc_date_end and
          cds_is_delete = 0 and
          pdt.pdt_code in ('work_patent', 'license')
    order by cds_doc_date_end desc
    into :pant_type, :pant_doc_no, :pant_date_begin, :pant_date_end, :pant_issued_by, :pant_memo_1, :pant_memo_2 ,:pant_memo3
  do
  begin
    pant_infolic1 = pant_infolic1 || iif(pant_infolic1 = '', '', ascii_char(10)) ||
      pant_doc_no || ', ' || pant_type || ', ' || f_datetostr(pant_date_begin) || ', ' || pant_issued_by || ', ' || f_datetostr(pant_date_end);
    pant_infolic2 = pant_infolic2 || iif(pant_infolic2 = '', '', ascii_char(10)) || :pant_memo_1;
    pant_infolic3 = pant_infolic3 || iif(pant_infolic3 = '', '', ascii_char(10)) || 'добровольного патента, №' || pant_doc_no || ' от ' || f_datetostr(pant_date_begin) || ' года';
  end

  registr_type = null;
  registr_doc_no = null;
  registr_date_begin = null;
  registr_date_end = null;
  registr_issued_by = null;
  registr_memo_1 = '';
  registr_memo_2 = '';
  registr_adress = ''; -- rfckg-4264
  registr_iin = ''; -- rfckg-4264

  select
    pk.pk_name,
    cd.cds_doc_no,
    cd.cds_doc_date_begin,
    cd.cds_doc_date_end,
    cd.cds_doc_issued_by,
    iif(pdt.pdt_code = 'certificate_ip' and :db_id in (39801, 41701),
      f_docrecog_value(cd.cds_from_drl_id, 'memo1'),
        cd.cds_memo_1),
    cd.cds_memo_2,
    iif(pdt.pdt_code = 'certificate_ip' and :db_id in (39801, 41701), f_docrecog_value(cd.cds_from_drl_id, 'memo_3'), ''), -- rfckg-4264
    iif(pdt.pdt_code = 'certificate_ip' and :db_id in (39801, 41701), f_docrecog_value(cd.cds_from_drl_id, 'memo_4'), '')  -- rfckg-4264
  from clientdocs cd
  inner join passkind pk on cd.cds_from_pk_id = pk_id
  inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
  where cds_from_c_id = :c_id and
        current_date between cds_doc_date_begin and cds_doc_date_end and
        cds_is_delete = 0 and
        (pdt.pdt_code = 'certificate_chp' or
        pdt.pdt_code = 'certificate_ip') rows 1
  into :registr_type, :registr_doc_no, :registr_date_begin, :registr_date_end, :registr_issued_by, :registr_memo_1,
       :registr_memo_2, :registr_iin, :registr_adress;

  create_date = null;
  lo_jl_name = '';
  lo_fio = '';
  select cast(m.ma_date_save as date),
         m.ma_from_kf_id,
         ml.mac_business_subtype,
         rd.rd_name
  from edoc_list edl
  inner join miniapp m  on m.ma_id = edl.edl_object_id
  inner join miniappclient ml on ml.mac_from_ma_id = m.ma_id
  left join refdata rd  on rd.rd_id = ml.mac_loan_target
  where edl.edl_id = :edl_id and
        ml.mac_from_c_id = :c_id and
        ml.mac_amount > 0 and
        m.ma_date_end >= '31.12.2999'
  order by m.ma_date_save desc
  rows 1
  into :create_date, :kf_id, :mac_busuness_subtype, mac_loan_target;

  
  if (db_id in (39801, 41701) and exists(select *
                               from public_persons pp
                               where pp.pp_from_c_id = :c_id and
                                     :create_date between pp.pp_date_begin and pp.pp_date_end and
                                     pp.pp_is_delete = 0)) then
    c_is_public = 1;

  bs_name = '';
  if (mac_busuness_subtype is not null) then
  begin
      select ', ' || bst.bs_name
      from bussubtype bst
      where bst.bs_id = :mac_busuness_subtype
      into :bs_name;
      bs_name = coalesce(bs_name, '');
  end

  kc_loan_target_original = mac_loan_target;


  select cl.c_name_fio, cl.c_id
  from knaktlist, clients cl
  where kal_from_kf_id = :kf_id and
        knaktlist.is_delete = 0 and
        kal_date <= current_date and
        kal_date_end >= current_date and
        kal_new_kf_officer = c_id
  into :lo_fio, lo_id;

  select jl_name
  from joblist, hr_pstatus
  where ps_from_jl_id = jl_id and
        ps_from_c_id = :lo_id and
        ps_is_working = 1 and
        current_date between ps_date_begin and ps_date_end
  into lo_jl_name;

  reg_name = '';
  pl_name = '';

  -- qr-код электронной подписи клиента. текст пустой если не подписан
  if (edl_id is not null) then
   begin
   qr_code = f_edoc_qr_by_edl(edl_id, c_id);

   select edl_blank_date
   from edoc_list
   where edl_id = :edl_id
   into edl_blank_date;
   end

  edl_create_date = null;

  if (db_id in (39801, 41701) and edl_id is not null) then
  begin
    -- фио отправившего ссылку, дата подписания
    select c_id, c_name_fio, cast(eds_status_datetime as date), edl_create_datetime, cast(edl_create_datetime as date)
    from edoc_list, edoc_sign, username, clients
    where edl_id = :edl_id and
          eds_from_edl_id = edl_id and
          eds_from_c_id = :c_id and
          edl_create_un_id = un_id and
          un_from_c_id = c_id
    into ver_c_id, ver_fio, sign_date, edl_create_datetime, edl_create_date;

    last_year_date = '';
    last_year_date = '31.12.' || extract(year from :edl_create_date);

    -- должность отправившего ссылку
    select ksr_kont_titul
    from kontsignmen, kontsignreg
    where ksm_id = ksr_from_ksm_id and
          ksm_from_c_id = :ver_c_id and
          ksm_is_active = 1 and
          ksr_is_active = 1 and
          coalesce(ksr_date_begin, '01.01.1990') <= :edl_create_datetime and
          coalesce(ksr_date_end, '31.12.9999') >= :edl_create_datetime
    order by ksr_id desc
    rows 1
    into ksr_kont_titul;
  end

  -- board-4903
  select plt.plt_short_name || pl.pl_name
  from hr_pstatus hrps
  inner join region reg on reg.reg_unicode = ps_from_reg_id
  inner join addrlist adl on adl.adl_id = reg.reg_address_fact
  inner join places pl on pl.pl_id = adl.adl_from_pl_id
  inner join placetype plt on plt.plt_id = pl.pl_status
  where hrps.ps_from_c_id = :c_id and
        hrps.ps_is_working = 1 and
        current_date between hrps.ps_date_begin and hrps.ps_date_end   
  into :place_name;

  if (edl_id is not null) then
  begin
    -- регион
    select reg_name, pl_name
    from edoc_list
      inner join region on reg_unicode = edl_from_reg_id
      inner join addrlist on adl_id = reg_address_fact
      inner join places pl on pl_id = adl_from_pl_id
    where edl_id = :edl_id
    into reg_name, pl_name;
  end

  ---по просьбе алмаз байке  jira-5100
  if (db_id in (39801, 41701) and c_type = 1) then
  begin
      select cd.cds_doc_no,cd.cds_doc_date_begin,cd.cds_doc_issued_by,pk.pk_name
      from clientdocs cd
      inner join passkind pk on cd.cds_from_pk_id = pk_id
      inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cds_from_c_id =:c_id
      and current_date between cd.cds_doc_date_begin and coalesce(cd.cds_doc_date_end,'31.12.9999')
      and cds_is_delete = 0
      and pdt.pdt_code containing 'certificate_yur'
      into doc_no_legal, doc_date_legal, doc_issued_by_legal, pk_name;
  end

  suspend;
end
