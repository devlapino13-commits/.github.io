CREATE PROCEDURE CALC_UNPAID_HOLIDAYS(
    REP_CLIENT_ID DECIMAL(18,0),
    REP_YEAR INTEGER,
    REP_MONTH INTEGER,
    SAT_CONS DECIMAL(18,4),
    STAGE_MONTH SMALLINT)
RETURNS (
    DAYS_QTY INTEGER,
    UNPAID_HOLIDAY DECIMAL(18,4))
AS
declare variable st_year integer;
declare variable st_month integer;
declare variable sawt_name varchar(255);
declare variable sat_days varchar(255);
declare variable sat_days_buf varchar(1000);
declare variable sat_days_res varchar(1000);
declare variable sat_days_first_buf varchar(1000);
declare variable pos_1 integer;
declare variable pos_2 integer;
declare variable st_month_buf integer;
declare variable i integer;
declare variable st_month_days integer;
declare variable st_month_days_res decimal(18,4);
declare variable st_year_month integer;
declare variable db_id d_integer;
begin
  /* Created by IGUSEV 14.11.12 Рассчет числа дней непрерывного неоплачиваемого отпуска*/
  db_id = f_get_db_id();
  st_month_buf = 0;
  sat_days_buf = '';
  st_month_days_res = 0;
  i = 0;

  for select sawt.sawt_name, st.sat_year_month,
             cast(st.sat_day_1 as varchar(2)) || cast(st.sat_day_2 as varchar(2)) || cast(st.sat_day_3 as varchar(2)) || cast(st.sat_day_4 as varchar(2)) || cast(st.sat_day_5 as varchar(2)) || cast(st.sat_day_6 as varchar(2)) || cast(st.sat_day_7 as varchar(2)) || cast(st.sat_day_8 as varchar(2)) || cast(st.sat_day_9 as varchar(2)) || cast(st.sat_day_10 as varchar(2)) || cast(st.sat_day_11 as varchar(2)) || cast(st.sat_day_12 as varchar(2)) || cast(st.sat_day_13 as varchar(2)) || cast(st.sat_day_14 as varchar(2)) || cast(st.sat_day_15 as varchar(2)) || cast(st.sat_day_16 as varchar(2)) || cast(st.sat_day_17 as varchar(2)) || cast(st.sat_day_18 as varchar(2)) || cast(st.sat_day_19 as varchar(2)) || cast(st.sat_day_20 as varchar(2)) || cast(st.sat_day_21 as varchar(2)) || cast(st.sat_day_22 as varchar(2)) || cast(st.sat_day_23 as varchar(2)) || cast(st.sat_day_24 as varchar(2)) || cast(st.sat_day_25 as varchar(2)) || cast(st.sat_day_26 as varchar(2)) || cast(st.sat_day_27 as varchar(2)) || cast(st.sat_day_28 as varchar(2)) || cast(st.sat_day_29 as varchar(2)) || cast(st.sat_day_30 as varchar(2)) || cast(st.sat_day_31 as varchar(2)) as sat_days,
             (
             case mod(st.sat_year_month, 100)
               when 1 then 31
               when 2 then 28
               when 3 then 31
               when 4 then 30
               when 5 then 31
               when 6 then 30
               when 7 then 31
               when 8 then 31
               when 9 then 30
               when 10 then 31
               when 11 then 30
               when 12 then 31
             end) as sat_month
      from sa_tabel st
      inner join sa_worktype sawt on sawt_id = sat_from_sawt_id
      where st.sat_from_c_id = :rep_client_id and
            ((sawt.sawt_id in (8, 10)) or /*неоплачиваем отпуск, неоплач. больничный отпуск*/
             (:db_id in (39801, 41701) and sawt.sawt_id in (25, 26))) -- RFCKG-544 Добавил отстранение от работы и прогул
            and
            st.sat_year_month between(:rep_year * 100 + 1) and (:rep_year * 100 + :rep_month) and
            st.sat_year_month >= (:rep_year * 100 + :stage_month)
      order by st.sat_year_month
      into :sawt_name, :st_year_month, :sat_days, :st_month_days
  do
  begin
    st_month_days_res = st_month_days_res + st_month_days;
    if (coalesce(mod(st_year_month, 100), 0) = (coalesce(st_month_buf, 0) + 1)) then
      sat_days_buf = sat_days_buf || sat_days;
    else
      sat_days_buf =   sat_days_buf || '0' ||  sat_days;
    st_month_buf = mod(st_year_month, 100);
    i = i + 1;
  end
  /*st_month_days_res - среднее число месяцев в году по которым брали неоплач. отпуска*/
  if (i > 0) then
    st_month_days_res = st_month_days_res / i;
  else
    st_month_days_res = 0;

  sat_days_first_buf = sat_days_buf;
  sat_days_res = '';
  sat_days_buf = sat_days_buf || '0';
  i = 0;
  while (i = 0) do
  begin
    pos_1 = position('888888888888888' in sat_days_buf);
    if (pos_1 > 0) then
    begin
      pos_2 = position('0', sat_days_buf, pos_1);
      sat_days_res = sat_days_res || substring(sat_days_buf from pos_1 for pos_2 - pos_1);
      sat_days_buf = substring(sat_days_buf from pos_2);
    end
    else
    begin
      i = 1;
    end
  end
  days_qty = char_length(sat_days_res);
  unpaid_holiday = 0;
  if (st_month_days_res > 0) then
    unpaid_holiday = days_qty * sat_cons / st_month_days_res;
  else
    unpaid_holiday = 0;
  suspend;
end
