CREATE PROCEDURE HP_UNDERWRITINGCERTIF_INS_UPD(
    C_ID D_ID,
    UCF_FROM_REG_ID D_INTEGER,
    UCF_DATE_BEGIN D_DATE,
    UCF_DATE_END D_DATE,
    UCF_ACTIVE D_SMALLINT,
    OLD_UCF_ID D_INTEGER,
    UN_ID D_ID)
AS
declare variable reg_id d_integer;
declare variable ucfs_id d_integer;
declare variable ucf_system d_smallint;
begin
  ucf_system = f_enum_id('access_sytem', 'undervriter_certif');

  --update/insert region
  if (:c_id = 0) then
  begin
    if (exists(select *
               from underwritingcertif ucf
               where ucf.ucf_from_reg_id = :ucf_from_reg_id
                 and ucf.ucf_system = :ucf_system)) then
      update underwritingcertif
      set ucf_from_reg_id = :ucf_from_reg_id,
          ucf_date_begin = :ucf_date_begin,
          ucf_date_end = :ucf_date_end,
          ucf_active = :ucf_active,
          ucf_from_un_id = :un_id
      where ucf_id = :old_ucf_id
        and ucf_system = :ucf_system;
    else
      insert into underwritingcertif (ucf_from_reg_id, ucf_date_begin, ucf_date_end, ucf_active, ucf_from_un_id, ucf_system)
      values (:ucf_from_reg_id, :ucf_date_begin, :ucf_date_end, :ucf_active, :un_id, :ucf_system);
  end
  else
  begin
    --update/insert user
    select ucfs.ucfs_from_reg_id, ucfs.ucfs_id
    from underwritingcertifspf ucfs
    where ucfs.ucfs_from_c_id = :c_id
    into :reg_id, ucfs_id;

    if (:ucfs_id is null) then
      insert into underwritingcertifspf (ucfs_from_c_id, ucfs_from_reg_id, ucfs_date_begin,
                                         ucfs_date_end, ucfs_active, ucfs_from_un_id)
      values (:c_id, :ucf_from_reg_id, :ucf_date_begin, :ucf_date_end, :ucf_active,
              :un_id);
    else
      update underwritingcertifspf
      set ucfs_from_c_id = :c_id,
          ucfs_date_begin = :ucf_date_begin,
          ucfs_date_end = :ucf_date_end,
          ucfs_active = :ucf_active,
          ucfs_from_un_id = :un_id
      where (ucfs_id = :ucfs_id);
  end
end
