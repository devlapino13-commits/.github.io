CREATE PROCEDURE AAREP_BLOBSTORE_SHOW(
    BLOB_ID DECIMAL(18,0))
RETURNS (
    BLS_ID DECIMAL(18,0),
    BLS_FILENAME VARCHAR(100),
    BLS_CAPTION VARCHAR(100),
    BLS_BLOB_SIZE FLOAT,
    BLS_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    BLS_KIND D_STR_100)
AS
declare variable bls_source_system d_integer;
declare variable bls_kind_id d_integer;
declare variable bls_datetime d_datetime;
begin
  bls_blob = null;
  if (blob_id is null) then
    blob_id = 0;

  bls_kind = '';
  if (blob_id > 0) then
  begin
    select bls_filename, bls_caption, bls_blob_size, bls_datetime, f_blob_kind(bls_source_system, bls_kind)
    from blobstore
    where bls_id = :blob_id
    into :bls_filename, :bls_caption, :bls_blob_size, bls_datetime, bls_kind;

    if (bls_datetime is not null) then
      bls_kind = bls_kind || ' ' || f_datetimetostr(bls_datetime);

    select blx_blob
    from hp_get_blobex_blob(:blob_id)
    into :bls_blob;
    bls_id = :blob_id;
  end
  else
  begin
    bls_caption = 'Нет документа';
    bls_blob_size = 0;
  end
  suspend;

end
