CREATE PROCEDURE AAREP_CALC_VOCATION_BASE(
    PERSONAL_ID DECIMAL(18,0),
    SALARY_YEAR INTEGER,
    SALARY_MONTH INTEGER)
RETURNS (
    TOTAL_CALC DECIMAL(18,4),
    TOTAL_HOUR DECIMAL(15,2),
    TOTAL_WORK_HOUR DECIMAL(15,2),
    TOTAL_CALC_ZP DECIMAL(18,4))
AS
declare variable monthly_voc decimal(18,4);
declare variable ye integer;
declare variable mo integer;
declare variable voc_days decimal(18,4);
declare variable dd date;
declare variable days integer;
declare variable ol_year integer;
declare variable ol_month integer;
declare variable amount decimal(18,4);
declare variable salary_year3 integer;
declare variable salary_month3 integer;
declare variable salary_year1 integer;
declare variable salary_month1 integer;
declare variable ol_year_month integer;
declare variable d1 d_date;
declare variable d2 d_date;
declare variable tmp_days d_integer;
declare variable period d_integer;
declare variable holidays d_integer;
begin
  if (salary_month <= 3) then
  begin
    salary_month3 = salary_month + 12 - 3;
    salary_year3 = salary_year - 1;
  end
  else
  begin
    salary_month3 = salary_month - 3;
    salary_year3 = salary_year;
  end

  if (salary_month <= 1) then
  begin
    salary_month1 = salary_month + 12 - 1;
    salary_year1 = salary_year - 1;
  end
  else
  begin
    salary_month1 = salary_month - 1;
    salary_year1 = salary_year;
  end

  total_calc = 0;
  total_hour = 0;
  monthly_voc = 0;
  dd = cast(('01.' || cast(:salary_month as varchar(2)) || '.' || cast(:salary_year as varchar(4))) as date);

  select scv_value
  from sa_consvalue
  where scv_from_scn_id = 5 /* Отпускные в месяц */
        and
        scv_date_begin <= :dd
  into :monthly_voc;

  for select osm_year_month, sum(osd_value)
      from opersalmain, opersaldetail, sa_config
      where osm_ol_is_active = 1 and
            osm_year_month between(:salary_year3 * 100 + :salary_month3) and (:salary_year1 * 100 + :salary_month1) and
            osd_from_osm_id = osm_id and
            osm_from_c_id = :personal_id and
            osd_from_sac_id = sac_id and
            sac_base_for_vocation = 3
      group by osm_year_month
      order by osm_year_month
      into :ol_year_month, amount
  do
  begin
    if ((amount <> 0) and
        (amount is not null)) then
    begin
      ol_month = mod(ol_year_month, 100);
      ol_year = (ol_year_month - ol_month) / 100;
      total_calc = :total_calc + :amount;
      voc_days = 0;

      select res_value
      from hp_get_opersal_sacvalue(:personal_id, :ol_year_month, :ol_year_month, null, null, ' where SAC_BASE_FOR_VOCATION=2')
      into :voc_days;

      if (voc_days is null) then
        voc_days = 0;
      voc_days = cast(voc_days / 8 as decimal(15,2));

      if (voc_days = 0) then
        total_hour = :total_hour + :monthly_voc;
      else
      begin
        if (ol_month = 12) then
        begin
          mo = 1;
          ye = :ol_year + 1;
        end
        else
        begin
          mo = ol_month + 1;
          ye = :ol_year;
        end
        dd = cast(('01.' || cast(:mo as varchar(2)) || '.' || cast(:ye as varchar(4))) as date) - 1;
        days = extract(day from :dd);
        total_hour = :total_hour + (:days - :voc_days);
      end
    end

  end

  total_work_hour = 0;
  total_calc_zp = 0;

  select res_value
  from hp_get_opersal_sacvalue(:personal_id, :salary_year3 * 100 + :salary_month3, :salary_year1 * 100 + :salary_month1, null, null, ' where SAC_KIND in (3,4)')
  into :amount;
  total_work_hour = :amount / 8;

  select res_value
  from hp_get_opersal_sacvalue(:personal_id, :salary_year3 * 100 + :salary_month3, :salary_year1 * 100 + :salary_month1, null, null, ' where SAC_SIGN=1')
  into :amount;
  total_calc_zp = :amount;

  if (f_get_db_id() in (39801, 41701)) then
  begin
    period = 0;
    ol_year = salary_year3;
    ol_month = salary_month3;

    total_hour = 0;

    while (period < 3) do
    begin
      voc_days = 0;
      tmp_days = 0;
      holidays = 0;

      d1 = cast('01.' || ol_month || '.' || ol_year as date);
      d2 = f_date_month_end(d1);

      if ((salary_year * 100 + salary_month) <= 202505 or f_get_db_id() <> 41701) then
      begin
        select sum(iif(extract(weekday from scd_date) not in (0, 6) and scd_work_hours = 0, 1,
                   iif(extract(weekday from scd_date) in (0, 6) and scd_work_hours = 8, -1, 0)))
        from sa_calendays
        where scd_date between :d1 and :d2
        into holidays;
      end else
      begin
        select hld_holidays
        from sa_holidays
        where hld_year = :ol_year and
              hld_month = :ol_month
        into holidays;
      end

      if (holidays is null) then
        holidays = 0;

      select sum(sat_days_total)
      from sa_tabel
      where sat_from_c_id = :personal_id and
            sat_year_month = (:ol_year * 100 + :ol_month) and
            sat_from_sawt_id in (7, 8, 9)
      into tmp_days;

      if (tmp_days is null) then
        tmp_days = 0;

      total_hour = total_hour + monthly_voc - tmp_days - holidays;

      ol_month = ol_month + 1;
      if (ol_month > 12) then
      begin
        ol_month = 1;
        ol_year = ol_year + 1;
      end

      period = period + 1;
    end
  end

  suspend;
end
