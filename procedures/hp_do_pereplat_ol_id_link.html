CREATE PROCEDURE HP_DO_PEREPLAT_OL_ID_LINK(
    OL_ID ID_TYPE,
    K_KONTRACT_NO D_STR_CODE,
    OPER_KIND INTEGER,
    DT_ACCOUNT D_STR_CODE,
    AMOUNT CURR_TYPE)
AS
declare variable k_id id_type;
declare variable sh_id integer;
declare variable sh_date date;
declare variable pay_type integer;
begin
  select k_id
  from kontract
  where k_kontract_no = :k_kontract_no
  into :k_id;

  if (k_id is null) then
    exception e_default 'K_ID not found: ' || coalesce(k_kontract_no, '<null>');

  if (not exists(select *
                 from operlist
                 where ol_id = :ol_id)) then
    exception e_default 'OL_ID not found: ' || coalesce(ol_id, '<null>');

  select ok_sh_id, ok_sh_date
  from operkred
  where ok_ol_is_active = 1 and
        ok_from_k_id = :k_id
  order by ok_sh_date desc, ok_sh_id desc
  rows 1
  into :sh_id, :sh_date;

  if (oper_kind = -1) then
    amount = -abs(amount);

  pay_type = 3;
  if (dt_account starting '50-01') then
    pay_type = 1;
  if (dt_account starting '51-01') then
    pay_type = 2;

  insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_pay_type)
  values (:ol_id, -1, :k_id, :sh_id, :sh_date, 9, :amount, :pay_type);
end
