CREATE PROCEDURE HP_OVERPAY_TRANSFER_PAY(
    REPAY_K_ID D_ID,
    REPAY_DATE D_DATE,
    ACTIVE_K_ID D_ID,
    SUMM D_CURRENCY,
    UN_ID D_ID,
    REP_DATE D_DATE = null)
RETURNS (
    ERROR_TEXT D_STR_255)
AS
declare variable kc_id d_id;
declare variable repay_ol_id d_id;
declare variable repay_ol_id_main d_id;
declare variable ol_id d_id;
declare variable sab_id d_integer;
declare variable bank_acc_id d_id;
declare variable bank_al_id d_str_20;
declare variable kob_summa d_currency;
declare variable c_id d_id;
declare variable kob_id d_id;
declare variable ok_number d_integer;
declare variable sh_id d_integer;
declare variable sh_date d_date;
declare variable acc_id d_id;
declare variable amount_sum d_currency;
declare variable storno_summ d_currency;
declare variable obd_detail d_str_255;
declare variable head_office_id d_integer;
declare variable repay_doc_client_name_reason d_str_255;
declare variable repay_kontract_no d_str_code;
declare variable active_kontract_no d_str_code;
declare variable kob_currency d_str_10;
declare variable ol_memo d_str_255;
declare variable main_ol_id d_id;
declare variable reg_id_pay d_id;
declare variable reg_id_repay d_id;
begin
  if (rep_date is null) then
    rep_date = current_date;

  head_office_id = f_hq_id(f_get_db_id());
  repay_kontract_no = (select k_kontract_no from kontract where k_id = :repay_k_id);
  active_kontract_no = (select k_kontract_no from kontract where k_id = :active_k_id);
  ol_memo = f_format('Сторно переплаты (перенос с %1 на %2)', repay_kontract_no, active_kontract_no);
  repay_doc_client_name_reason = 'Погашение за счет переплаты';

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Погашено.Переплата'
  into :sab_id;

  for select kc_id, kc_from_c_id
  from knclient
  where kc_from_k_id = :repay_k_id
    and kc_amount_approved > 0
  into kc_id, c_id
  do
  begin
    amount_sum = 0;

    for select
      kob_from_ol_id,   -- ID операции погашения (переплаты)
      kob_acc_dt,   -- Дт счет (Счет учета денег/ Дебет)
      kob_acc_kt,   -- КТ счет (Счет учета переплаты / Кредит)
      kob_summa,    -- Сумма
      obd_detail,   -- Детали проводки
      kob_currency
    from kn_operbook, ref_account kt, kn_obdetail
    where kob_acc_kt = kt.acc_id
      and obd_id = kob_detail_id
      and kt.acc_from_sab_id = :sab_id
      and kob_from_kc_id = :kc_id
      and kob_ol_is_active = 1
      and kob_ol_date = :repay_date
    into repay_ol_id_main, bank_acc_id, acc_id, kob_summa, obd_detail, kob_currency
    do
    begin
      -- Сумма уже сторнированная
      select coalesce(sum(-kob_summa), 0)
      from operlist, reversallist, kn_operbook
      where kob_from_ol_id = rl_new_ol_id
        and rl_old_ol_id = ol_id
        and ol_parent_id = :repay_ol_id_main
      into storno_summ;

      kob_summa = kob_summa - storno_summ;

      if (kob_summa <= 0) then
        continue;

      if (summ <= 0) then
        exit;

      select al_id
      from acc_list
      where al_from_acc_id = :bank_acc_id
      into bank_al_id;

      amount_sum = minvalue(summ, kob_summa);
      error_text = '';

      repay_ol_id = (select ol_id from operlist where ol_parent_id = :repay_ol_id_main and ol_is_client = 1);

      if (repay_ol_id is null) then
        repay_ol_id = repay_ol_id_main;

      --- сторнирование переплаты
      insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                            ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                            ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                            ol_pocket, ol_doc_client_name, ol_is_client, ol_manager_c_id, ol_from_rd_id)
      select ol_from_reg_id, ol_oper_type, :rep_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, :ol_memo, ol_status,
             is_delete, null, 1000000000000, ol_document_client, ol_from_mo_id, ol_is_mo_modified, ol_document_office,
             ol_acc_no_client, ol_acc_no_office, ol_oper_subtype, ol_pocket, ol_doc_client_name,
             ol_is_client, ol_manager_c_id, ol_from_rd_id
      from operlist
      where ol_id = :repay_ol_id_main
      returning ol_id
      into main_ol_id;

      insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                            ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                            ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                            ol_pocket, ol_doc_client_name, ol_is_client, ol_manager_c_id, ol_from_rd_id, ol_parent_id)
      select ol_from_reg_id, ol_oper_type, :rep_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, :ol_memo, ol_status,
             is_delete, null, 1000000000000, ol_document_client, ol_from_mo_id, ol_is_mo_modified, ol_document_office,
             ol_acc_no_client, ol_acc_no_office, ol_oper_subtype, ol_pocket, ol_doc_client_name,
             ol_is_client, ol_manager_c_id, ol_from_rd_id, :main_ol_id
      from operlist
      where ol_id = :repay_ol_id
      returning ol_id
      into :ol_id;

      update operloan
      set opl_acc_kind = 2,
          opl_from_al_id = :bank_al_id,
          opl_from_acc_id = :bank_acc_id
      where opl_from_ol_id = :ol_id;

      insert into reversallist (rl_id, rl_new_ol_id, rl_old_ol_id, rl_user_id, rl_date, rl_ol_is_delete)
      values (null, :ol_id, :repay_ol_id, :un_id, :rep_date, 0);

      reg_id_pay = (select ol_from_reg_id from operlist where ol_id = :repay_ol_id_main);
      reg_id_repay = (select ol_from_reg_id from operlist where ol_id = :repay_ol_id);

      if (reg_id_pay is null or reg_id_repay is null) then
        exception e_default 'Not found reg_id_repay or reg_id_pay';

      insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val, ob_summa, ob_detail)
      select :ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, -:amount_sum, -:amount_sum, :ol_memo
      from operbook
      where ob_from_ol_id = :repay_ol_id
        and ob_detail = 'Переплата займа'
      rows 1;

      if (reg_id_repay <> reg_id_pay) then
        insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val, ob_summa, ob_detail)
        select :main_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, -:amount_sum, -:amount_sum, :ol_memo
        from operbook
        where ob_from_ol_id = :repay_ol_id_main
        rows 1;

      kob_id = null;
      select kob_id
      from hp_do_kn_operbook_memorder(iif(:reg_id_repay <> :reg_id_pay, :main_ol_id, :ol_id), :kc_id, null, null, -:amount_sum, 0, -:amount_sum, -:amount_sum, :obd_detail, :bank_acc_id, :acc_id, :rep_date, :kob_currency)
      into :kob_id;
    
      if (kob_id is null) then
        exception e_default 'kob_id is null';

      select max(iif(ok_sh_id is not null, ok_sh_id, 1)), max(ok_sh_date)
      from kn_operkred
      where ok_from_kc_id = :kc_id
        and ok_ol_date <= :rep_date
        and ok_ol_is_active = 1
        and ok_type = 9
      into sh_id, sh_date;
    
      ok_number = null;
      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                            ok_ol_reg_id, ok_ol_is_active)
      values (:ol_id, -1, :repay_k_id, :sh_id, :sh_date, 9, :amount_sum, 0, 0, 3, 0, 0, 0, 0, 0, :rep_date, 1010100, 1)
      returning ok_number
      into :ok_number;
    
      if (ok_number is null) then
        exception e_default 'Error when insert Operkred, KC_ID: ' || kc_id;
    
      execute procedure hp_kn_oksummary_recalc(:kc_id, :rep_date);
      summ = summ - amount_sum; -- остаток

      -- посадим платеж
      begin
        select ol_id, error_text
        from hp_do_repayment(:active_k_id, 0, :rep_date, :amount_sum, 1, 1, 1, 1, 1, 0, 0,
            :head_office_id, 3, :bank_al_id, 0, :un_id, :c_id, :repay_doc_client_name_reason, null, 0, 0)
        into :main_ol_id, :error_text;

        when any do
        begin
          error_text = f_format('Ошибка оформления платежа %1 на договор %2', amount_sum,  active_kontract_no);
          error_text = error_text || ascii_char(10) ||
            f_format('hp_do_repayment(%1, 0, ''%2'', %3, 1, 1, 1, 1, 1, 0, 0,', :active_k_id, f_datetostr(:rep_date), :amount_sum)||
            f_format('%1, 3, %2, 0, %3, %4, ''%5'', null, 0, 0)', :head_office_id, :bank_al_id, :un_id, :c_id, :repay_doc_client_name_reason);
        end
      end
    end
  end

  if (summ > 0) then
    exception e_default 'summ > 0';

  suspend;
end
