CREATE PROCEDURE AAREP_KONTCALCPERC_CHECK_TMP
RETURNS (
    REP_DD DATE,
    REP_REGION_ID INTEGER,
    REP_REGION_NAME VARCHAR(100),
    REP_ACCOUNT VARCHAR(20),
    REP_KONT_SALDO DECIMAL(18,4),
    REP_BALANCE_SALDO DECIMAL(18,4),
    REP_LOAN_COUNT INTEGER)
AS
declare variable period_min integer;
declare variable period_max integer;
declare variable rep_period integer;
begin
  select min(kcp_period), max(kcp_period)
  from kontcalcperc
  into :period_min, :period_max;

  if (period_min is null) then
    period_min = 0;
  if (period_max is null) then
    period_max = 0;
  if (period_min <> 0) then
  begin
    rep_period = period_min;
    while (rep_period <= :period_max) do
    begin
      if (mod(rep_period, 100) < 12) then
      begin
        rep_dd = '01.' || mod(rep_period + 1, 100) || '.' || round(rep_period / 100);
        rep_dd = rep_dd - 1;
      end
      else
        rep_dd = '31.12.' || round(rep_period / 100);
      rep_loan_count = 0;
      rep_kont_saldo = 0;
      for select al_id, al_from_reg_id, reg_code || ' ' || reg_name
          from acc_list, region
          where al_from_bg_id = '76-09' and
                al_from_reg_id = reg_unicode
          into :rep_account, rep_region_id, rep_region_name
      do
      begin
        rep_balance_saldo = f_al_bal(rep_account, rep_dd);
        suspend;
      end

      rep_balance_saldo = 0;
      for select distinct kcp_perc_account, kcp_from_reg_id, reg_code || ' ' || reg_name
          from kontcalcperc, region
          where kcp_period = :rep_period and
                kcp_from_reg_id = reg_unicode
--      and kcp_loan_is_closed=0
          into :rep_account, rep_region_id, rep_region_name
      do
      begin
        rep_kont_saldo = 0;
        rep_loan_count = 0;
        select sum(kcp_perc_saldo), sum(iif(kcp_perc_saldo > 0, 1, 0))
        from kontcalcperc
        where kcp_period = :rep_period and
              kcp_perc_account = :rep_account and
              kcp_from_reg_id = :rep_region_id
--        and kcp_loan_is_closed=0
        into :rep_kont_saldo, :rep_loan_count;
        if (rep_kont_saldo is null) then
          rep_kont_saldo = 0;
        if (rep_loan_count is null) then
          rep_loan_count = 0;
        suspend;
      end

      if (mod(rep_period, 100) < 12) then
        rep_period = rep_period + 1;
      else
        rep_period = (round(rep_period / 100) + 1) * 100 + 1;
    end
  end
end
