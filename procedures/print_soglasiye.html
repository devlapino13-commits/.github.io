CREATE PROCEDURE PRINT_SOGLASIYE(
    MA_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    S_C_NAME_FIO D_STR_100,
    S_C_BIRTH_DATE D_DATE,
    S_C_PASS_NO D_STR_20,
    S_C_PASS_DATE D_DATE,
    S_C_PASS_WHO D_STR_255,
    S_CELL D_STR_20,
    S_REL_NAME D_STR_10,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    C_PASS_NO D_STR_20,
    C_PASS_DATE D_DATE,
    C_PASS_WHO D_STR_100,
    IS_INDIV D_SMALLINT,
    LOAN_AMOUNT_CL CURR_TYPE,
    LOAN_AMOUNT CURR_TYPE,
    QR_CODE D_STR_LONG,
    REG_NAME D_STR_100,
    SUM_OS_AND_PERC CURR_TYPE,
    PR_CODE D_STR_100,
    EDS_STATUS_DATE D_DATE)
AS
declare variable C_ID D_ID; /* ADD_PARAMS */
declare variable I D_SMALLINT;
declare variable MAC_AMOUNT CURR_TYPE;
declare variable CREL_OTHER_C_ID D_ID;
declare variable REL_NAME STR_20;
declare variable EDL_ID D_ID;
declare variable MAC_FROM_C_ID D_ID;
declare variable MA_OFFICE_ID D_ID;
declare variable K_ID D_ID;
declare variable SUM_OS_TMP CURR_TYPE;
declare variable SUM_PERC_TMP CURR_TYPE;
declare variable NSP_TMP CURR_TYPE;
begin
  eds_status_date = current_date;
  /*
    процедура отвечает за возвращение данные для печатного бланка "лицо дающее согласие" в КГ

    ma_id - id кредитной заявки
    в add_params ожидается json вида
    {
        "c_id": "1090100693989"
    }
    где c_id - c_id лица дающего согласие вказанного в заявке (параметр ma_id)

    Пример для теста
    ma_id = 1621848;
    add_params = '{"c_id": "1090100693989"}';
  */
  /*  -- для теста
  ma_id = 1621848;
  add_params = '{"c_id": "1090100693989"}';  */

  if (ma_id is null) then
    exception e_default 'ma_id = null';

  add_params = coalesce(add_params, '');
  if (add_params = '') then
    exit;

  c_id = f_json_str_get_value(add_params, 'c_id', '0');
  c_id = coalesce(c_id, 0);
  if (c_id = 0) then
    exit;

  edl_id = f_json_str_get_value(add_params, 'edl_id', null); -- берем id  электронного документа
  qr_code = f_edoc_qr_by_edl(edl_id, c_id); -- формируем QR code

  -- RFCKG-3737
  select cast(eds.eds_status_datetime as date)
  from edoc_sign eds
  where eds.eds_from_edl_id = :edl_id and
        eds.eds_status = 1 and
        eds.eds_from_c_id = :c_id
  into :eds_status_date;


  select cl.c_name_fio, c_birth_date, c_pass_no, cl.c_pass_date, cl.c_pass_who
  from clients cl
  where c_id = :c_id
  into :s_c_name_fio, :s_c_birth_date, :s_c_pass_no, :s_c_pass_date, :s_c_pass_who; --лицо дающее согласие

  s_cell = f_cont_info('clients', :c_id, 'cell_main', 'last'); --находим контактные данные лица дающего согласие

  select crel_other_c_id, rel_name
  from hp_client_relation_list(:c_id, 0)
  where crel_from_rel_id in ('1', '2') rows 1 --находим супруга/супругу(клиента), лица дающего согласие
  into :crel_other_c_id, rel_name;

  select cl.c_name_fio, c_birth_date, c_pass_no, cl.c_pass_date, cl.c_pass_who
  from miniappclient mnc, clients cl
  where mac_from_ma_id = :ma_id and
        c_id = :crel_other_c_id and
        mnc.mac_amount > 0 rows 1
  into :c_name_fio, :c_birth_date, :c_pass_no, :c_pass_date, :c_pass_who; --ищем клиента который является супругом лица дающего согласия

  if (c_name_fio is null) then
    exit; --если нет связи то выходим из hp
  s_rel_name = null;
  if (rel_name = 'супруг') then
    s_rel_name = 'жена';
  if (rel_name = 'супруга') then
    s_rel_name = 'муж';

  loan_amount_cl = 0;
  loan_amount = 0;
  mac_amount = 0;
  i = 0;
  is_indiv = null;
  for select mcl.mac_amount, mcl.mac_from_c_id,m.ma_office_id, pr.pr_code
      from miniapp m, miniappclient mcl, product pr
      where mcl.mac_from_ma_id = m.ma_id and
            m.ma_id = :ma_id and
            mcl.mac_amount > 0 and
            current_date between m.ma_date_app and m.ma_date_end and
            pr.pr_id = m.ma_from_pr_id
      into :mac_amount, :mac_from_c_id, :ma_office_id, :pr_code
  do
  begin
    i = i + 1;
    loan_amount = loan_amount + mac_amount;
    if (:mac_from_c_id = :crel_other_c_id) then
      loan_amount_cl = mac_amount;

  end
  select r.reg_name
  from region r
  where r.reg_unicode = :ma_office_id
  into:reg_name;

  if (i = 1) then
    is_indiv = 1;
  if (i > 1) then
    is_indiv = 0;
  -- добавление полей pr_code и sum_os_and_perc задача на jiro 1912 от 08,05,2020
  sum_os_and_perc = 0;

  for select sh.sh_os,
             sh.sh_perc,
             round(sh.sh_perc * 2/100, 0)
  from schedule sh
  where sh.sh_from_k_id = -:ma_id
  into :sum_os_tmp, sum_perc_tmp, :nsp_tmp
  do
  begin
    sum_os_and_perc = sum_os_and_perc + sum_os_tmp + sum_perc_tmp + nsp_tmp;
  end

  if (sum_os_and_perc = 0) then
  begin
    select kon.k_id
    from kontract kon
    where kon.k_from_ma_app_id = :ma_id
    order by k_id desc
    rows 1
    into :k_id;

    for select sh.sh_os,
               sh.sh_perc,
               round(sh.sh_perc * 2/100, 0)
    from schedule sh
    where sh.sh_from_k_id = :k_id
    into :sum_os_tmp, sum_perc_tmp, :nsp_tmp
    do
    begin
        sum_os_and_perc = sum_os_and_perc + sum_os_tmp + sum_perc_tmp + nsp_tmp;
    end
  end

  sum_os_and_perc = coalesce(sum_os_and_perc, 0);

  suspend;
end
