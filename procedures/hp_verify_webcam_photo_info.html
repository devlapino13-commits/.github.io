CREATE PROCEDURE HP_VERIFY_WEBCAM_PHOTO_INFO(
    K_ID D_ID)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    CARD_ID D_INTEGER,
    REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    VID_TYPE D_BOOL,
    STATUS D_INTEGER,
    ACL_STATUS D_INTEGER,
    COMPRESS_QUALITY D_INTEGER)
AS
declare variable acl_id d_id;
begin
  compress_quality = 80;

  select
    c_id,
    c_name_fio,
    k_kontract_no,
    k_last_reg_id,
    reg_name
  from knclient
    inner join clients
      on c_id = kc_from_c_id
    inner join kontract
      on k_id = kc_from_k_id
    inner join region
      on reg_unicode = k_last_reg_id
  where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    and kc_status = 2
  into
    :c_id,
    :c_name_fio,
    :k_kontract_no,
    :reg_id,
    :reg_name;

  select
    vwp_acl_id,
    vwp_card_id,
    vwp_type,
    vwp_status
  from verify_webcam_photo
  where vwp_k_id = :k_id
  order by vwp_id desc
  rows 1
  into
    :acl_id,
    :card_id,
    :vid_type,
    :status;

  if (card_id > 0 or status is null) then
    status = -1;
  if (acl_id > 0) then
    acl_status = (
      select
        acl_status
      from actionlist
      where acl_id = :acl_id);

  suspend;
end
