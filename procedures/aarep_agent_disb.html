CREATE PROCEDURE AAREP_AGENT_DISB(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REP_AGENT_C_ID DECIMAL(18,0),
    REP_CLIENT_C_ID DECIMAL(18,0),
    REP_LO_C_ID DECIMAL(18,0),
    REP_REGION_NAME VARCHAR(100),
    REP_AGENT_FIO VARCHAR(100),
    REP_CLIENTS_FIO VARCHAR(100),
    REP_DISB_DATE DATE,
    REP_KONTRACT_NO VARCHAR(20),
    VYD_SVZ VARCHAR(100),
    REP_KOLICH INTEGER,
    REP_CLIENT_DISB_AMOUNT DECIMAL(18,4),
    DATE_REGISTER DATE,
    REP_AGENT_KIND VARCHAR(100),
    PRODUCT_NAME VARCHAR(255),
    REP_PAR7_BALANCE DECIMAL(18,4),
    AGENT_HAVE_ACTIVE_KONT INTEGER,
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    AGENT_K_PERIOD INTEGER,
    AGENT_KONT_BALANCE_PAR7 DECIMAL(18,4),
    AGENT_MAX_DELINQ_DAY INTEGER,
    AGENT_FIRST_KONT_DATE DATE,
    AGENT_NATION STR_100,
    AGENT_FAMILY STR_100,
    AGENT_KOL_CHILD INTEGER,
    AGENT_EDUCATION STR_100,
    AGENT_GENDER VARCHAR(1),
    AGENT_C_BIRTH_DATE DATE,
    AGENT_BIRTH_CITY STR_100,
    AGENT_AGE INTEGER,
    AGENT_DATE_ARRIVAL DATE,
    CLIENT_NATION STR_100,
    CLIENT_FAMILY STR_100,
    CLIENT_KOL_CHILD INTEGER,
    CLIENT_EDUCATION STR_100,
    CLIENT_GENDER VARCHAR(1),
    CLIENT_C_BIRTH_DATE DATE,
    CLIENT_BIRTH_CITY STR_100,
    CLIENT_AGE INTEGER,
    CLIENT_DATE_ARRIVAL DATE,
    SVZ_NATION STR_100,
    SVZ_FAMILY STR_100,
    SVZ_KOL_CHILD INTEGER,
    SVZ_EDUCATION STR_100,
    SVZ_GENDER VARCHAR(1),
    SVZ_BIRTH_DATE DATE,
    SVZ_BIRTH_CITY STR_100,
    SVZ_AGE INTEGER,
    SVZ_DATE_ARRIVAL DATE,
    REP_SVZ_PAR7 CURR_TYPE,
    REP_SVZ_EXPERIENCE INTEGER,
    K_ID ID_TYPE)
AS
declare variable rep_k_id decimal(18,0);
declare variable sms_date timestamp;
declare variable sds_id integer;
declare variable k_amount_approved decimal(18,4) = 0;
declare variable rep_agent_staff varchar(100);
declare variable delinq_day_repayment integer;
declare variable oks_delinq_day integer;
declare variable dolia perc_type;
declare variable agent_nation_id integer;
declare variable agent_family_id integer;
declare variable agent_education_id integer;
declare variable client_nation_id integer;
declare variable client_family_id integer;
declare variable client_education_id integer;
declare variable svz_nation_id integer;
declare variable svz_family_id integer;
declare variable svz_education_id integer;
declare variable k_disb_svz_balance curr_type;
declare variable k_disb_svz_balance_par curr_type;
declare variable c_worked_years integer;
declare variable c_worked_months integer;
declare variable c_worked_days integer;
declare variable date_begin date;
declare variable date_end date;
declare variable c_worked_years_buf integer;
declare variable c_worked_months_buf integer;
declare variable c_worked_days_buf integer;
declare variable svz_hired_date date;
declare variable rep_agent_c_id_old id_type;
declare variable rep_client_c_id_old id_type;
declare variable rep_lo_c_id_old id_type;
declare variable hired_date date;
begin
  /* Процедура для РФ. Расчет бонуса агентам, которые раздают купоны из чековых книжек */

  rep_kolich = 1;
  for select distinct kc_from_k_id, min(sds_create_datetime), min(sds_id)
      from smsdialogsend, knclient
      where sds_action = 'Kupon_register' and
            (sds_cell_no = (select cont_value
                            from hp_contdata_get_cell_valid(0, kc_from_c_id))) and
            (sds_text like 'Ваш%' or
            sds_text like 'Поздравляем! %') and
            sds_from_c_id is not null and
            kc_amount_approved > 0
      group by kc_from_k_id
      into :rep_k_id, :sms_date, :sds_id
  do
  begin
    rep_agent_fio = '';
    rep_agent_kind = '';
    rep_par7_balance = 0;
    select c_id, c_name_fio
    from smsdialogsend, clients
    where sds_id = :sds_id and
          sds_from_c_id = c_id
    into :rep_agent_c_id, :rep_agent_fio;
    if (rep_agent_fio <> '') then
    begin
      date_register = null;
      with tab
      as (select ks_date_begin as date_min
          from kontstate, knclient
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_from_k_id = kc_from_k_id and
                kc_from_c_id = :rep_agent_c_id and
                kc_amount_approved > 0
          union
          select cast(sdr_create_datetime as date) as date_min
          from smsdialogreciever
          where sdr_from_c_id = :rep_agent_c_id
          union
          select ps_date_begin as date_min
          from hr_pstatus
          where ps_from_c_id = :rep_agent_c_id and
                ps_status = 1)
      select min(date_min)
      from tab
      into :date_register;

      if (exists(select *
                 from kontstate, knclient
                 where ks_ol_is_delete = 0 and
                       ks_state = 1 and
                       ks_from_k_id = kc_from_k_id and
                       kc_from_c_id = :rep_agent_c_id and
                       kc_amount_approved > 0 and
                       ks_date_begin = :date_register)) then
        rep_agent_kind = 'Агент был клиентом';
      else
      if (exists(select *
                 from hr_pstatus
                 where ps_from_c_id = :rep_agent_c_id and
                       ps_status = 1 and
                       ps_date_begin = :date_register)) then
        rep_agent_kind = 'Агент был сотрудником';
      else
      if (exists(select *
                 from smsdialogreciever
                 where sdr_from_c_id = :rep_agent_c_id and
                       cast(sdr_create_datetime as date) = :date_register)) then
        rep_agent_kind = 'Начинал как агент';

      rep_agent_staff = '';
      if (exists(select *
                 from clientrelation, hr_pstatus
                 where crel_other_c_id = :rep_agent_c_id and
                       crel_from_c_id = ps_from_c_id and
                       ps_status = 1)) then
        rep_agent_staff = 'Агент является родственников сотрудника';

      rep_lo_c_id_old = 0;
      rep_client_c_id_old = 0;
      rep_agent_c_id_old = 0;

      for select distinct reg_code || ' ' || reg_name, ks_date_begin, ks_from_k_id, k_kontract_no,
                          c1.c_name_fio as rep_clients_fio, c1.c_id, c2.c_name_fio as vyd_svz, c2.c_id,
                          kc_amount_approved, pr_name, k_amount_approved, c2.c_hired_date
          from knclient, kontstate ks, region, kontract kont, clients c1, clients c2, product pr
          where kc_from_k_id = :rep_k_id and
                kc_amount_approved > 0 and
                kc_from_c_id = c1.c_id and
                ks.ks_kc_id = c2.c_id and
                kc_from_k_id = ks_from_k_id and
                ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_begin between :rep_date_begin and :rep_date_end and
                ks_date_begin between(:sms_date - 1) and (:sms_date + 30) and
                ks_region_id = reg_unicode and
                ks_from_k_id = k_id and
                (kc_kommis_lgota <> 0.0 or
                :rep_date_begin >= '01.10.2015') and
                pr.pr_id = kont.k_from_pr_id
          into :rep_region_name, :rep_disb_date, :rep_k_id, :rep_kontract_no, :rep_clients_fio, :rep_client_c_id,
               :vyd_svz, :rep_lo_c_id, :rep_client_disb_amount, :product_name, :k_amount_approved, :hired_date
      do
      begin
        select oks_balance
        from oksummary
        where oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end and
              oks_from_k_id = :rep_k_id and
              datediff(day, oks_delinq_date, :rep_date_end) > 7
        into :rep_par7_balance;
        if (rep_par7_balance > 0) then
        begin
          rep_par7_balance = cast(rep_par7_balance as double precision) / k_amount_approved;
          rep_par7_balance = cast(rep_par7_balance as double precision) * rep_client_disb_amount;
        end

-- igusev ---------- 16.11.2015

        /*
            Под личным характеристиками берите поля:
            пол, возраст, национальность, гражданство, сем.положение, кол-во детей, образование, область откуда приехали (берите место рождение C_BIRTH_CITY), срок пребывания в России (минимальная дата из миграционки, если есть)
        */
-- СВЗ --------------------------------------------------------------->>

        rep_svz_par7 = 0;

-- ПАР7 СВЗ на момент выдачи этого кредита.
        select sum(oks_balance) balance,
               sum(iif(cast(:rep_disb_date as date) - oks_delinq_date > 7, oks_balance, 0)) balance_par,
               sum(iif(cast(:rep_disb_date as date) - oks_delinq_date > 7, oks_balance, 0)) / sum(oks_balance) * 100 as par
        from oksummary oks
        where :rep_disb_date between oks.oks_date_begin and oks_date_end and
              oks_close_state = 0 and
              oks_kc_id = :rep_lo_c_id -- <<
        into :k_disb_svz_balance, :k_disb_svz_balance_par, :rep_svz_par7;

-- >> --- Рассчитываю Стаж в месяцах СВЗ на момент выдачи ----------
        rep_svz_experience = 0;

        c_worked_years = 0;
        c_worked_months = 0;
        c_worked_days = 0;
--Приемы на работу
        for select ps_date_begin
            from hr_pstatus, hr_orders ho
            where ps_status = 1 and
                  ps_from_ho_id = ho_id and
                  ho.ho_status = 1 and
                  ps_date_begin < ps_date_end and
                  ho_order_type = 1 and
                  ps_from_c_id = :rep_lo_c_id and
                  ps_date_begin <= :rep_disb_date -- <<
            order by ps_date_begin
            into :date_begin
        do
        begin
          date_end = null;
--Увольнения с работы
          select ps_date_begin
          from hr_pstatus, hr_orders ho
          where ps_status = 1 and
                ps_from_ho_id = ho_id and
                ho.ho_status = 1 and
                ps_date_begin < ps_date_end and
                ho_order_type = 6 and
                ho.ho_date >= :date_begin and
                ps_from_c_id = :rep_lo_c_id and
                ps_date_begin <= :rep_disb_date -- <<
          order by ho.ho_date
          rows 1
          into :date_end;

          if (:date_end is null) then
            date_end = rep_disb_date;

          begin
            if (:date_begin >= :date_end) then
            begin
              c_worked_years_buf = 0;
              c_worked_months_buf = 0;
              c_worked_days_buf = 0;
            end
            else
            begin
-- вычислили разницу в днях
              c_worked_years_buf = datediff(year, :date_begin, :date_end);

-- увеличили на разницу в годах, если больше, то уменьшаем на 1 год
              date_begin = dateadd(year, :c_worked_years_buf, :date_begin);
              if (:date_begin > :date_end) then
              begin
                c_worked_years_buf = :c_worked_years_buf - 1;
                date_begin = dateadd(year, -1, :date_begin);
              end

-- вычислили разницу в месяцах
              c_worked_months_buf = datediff(month, :date_begin, :date_end);

-- увеличили на разницу в месяцах, если больше, то уменьшаем на 1 месяц
              date_begin = dateadd(month, :c_worked_months_buf, :date_begin);
              if (:date_begin > :date_end) then
              begin
                c_worked_months_buf = :c_worked_months_buf - 1;
                date_begin = dateadd(month, -1, :date_begin);
              end
-- вычислили разницу в днях
              c_worked_days_buf = datediff(day, :date_begin, :date_end);

-- увеличили на разницу в днях, если больше, то уменьшаем на 1 день
              date_begin = dateadd(day, :c_worked_days_buf, :date_begin);
              if (:date_begin > :date_end) then
              begin
                c_worked_days_buf = :c_worked_days_buf - 1;
                date_begin = dateadd(day, -1, :date_begin);
              end

              c_worked_years = c_worked_years + c_worked_years_buf;
              c_worked_months = c_worked_months + c_worked_months_buf;
              c_worked_days = c_worked_days + c_worked_days_buf;
            end
          end
        end
        if (c_worked_days > 31) then
        begin
          c_worked_months = c_worked_months + c_worked_days / 31;
        end
        if (c_worked_months > 12) then
        begin
          c_worked_years = c_worked_years + c_worked_months / 12;
          c_worked_months = mod(c_worked_months, 12);
        end

        rep_svz_experience = c_worked_years * 12 + c_worked_months;

-- если не удалось выявить стаж исходя из кадровых приказов, исхожу из Даты регистрации сотрудника
        if (rep_svz_experience = 0) then
        begin
          if (svz_hired_date is not null) then
          begin
            if (svz_hired_date <= rep_disb_date) then
              rep_svz_experience = datediff(month, svz_hired_date, rep_disb_date);
          end
        end

        if (rep_lo_c_id <> rep_lo_c_id_old) then
        begin
          rep_lo_c_id_old = rep_lo_c_id;
-- Личностные характеристики ---------------------------------------
          svz_nation = '';
          svz_family = '';
          svz_kol_child = 0;
          svz_education = '';

          select iif(cl.c_sex = 1, 'М', 'Ж'), cl.c_birth_date, cl.c_nation, cl.c_family, c_kol_child, cl.c_education,
                 cl.c_birth_city
          from clients cl
          where cl.c_id = :rep_lo_c_id
          into :svz_gender, :svz_birth_date, :svz_nation_id, :svz_family_id, :svz_kol_child, :svz_education_id,
               :svz_birth_city;

          if (svz_birth_date is not null) then
          begin
            svz_age = trunc(datediff(month, :svz_birth_date, :rep_date_end) / 12);
          end

          select rd.rd_name
          from refdata rd
          where rd.rd_id = :svz_nation_id
          into :svz_nation;

          select rd.rd_name
          from refdata rd
          where rd.rd_id = :svz_family_id
          into :svz_family;

          select rd.rd_name
          from refdata rd
          where rd.rd_id = :svz_education_id
          into :svz_education;

          svz_date_arrival = null;
        end
        select cds.cds_doc_date_begin
        from clientdocs cds, passkind, passdoctype
        where cds_from_c_id = :rep_lo_c_id and
              cds_from_pk_id = pk_id and
              pk_from_pdt_id = pdt_id and
              cds_is_delete = 0
--and :rep_date_end between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
              and
              cds_doc_date_begin <= :rep_disb_date and
              pdt_code = 'magration_card'
        order by cds_doc_date_begin
        rows 1
        into :svz_date_arrival;
-- <<---------------------------------------------------------------
-- << ------------------------------------------------------------------

-- АГЕНТ ------------------------------------------------------------->>
        agent_have_active_kont = 0; --Есть ли у агента действующий кредит
        k_id = null;
        kc_amount_approved = 0;
        agent_k_period = 0;
        agent_kont_balance_par7 = 0;
        agent_max_delinq_day = 0; -- Максимальный срок задержки кредита
        delinq_day_repayment = 0;
        oks_delinq_day = 0;

-- активный заем Агента
        select ks.ks_from_k_id, kncl.kc_amount_approved
        from kontstate ks
        inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
        where ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1 and
              :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
              kncl.kc_amount_approved > 0 and
              kncl.kc_from_c_id = :rep_agent_c_id
        order by ks.ks_date_begin
        rows 1
        into :k_id, :kc_amount_approved;

        if (k_id is not null) then
        begin
          agent_have_active_kont = 1;

          select round(kont.k_period), kont.k_amount_approved
          from kontract kont
          where kont.k_id = :k_id
          into :agent_k_period, k_amount_approved; -- Характеристики действующего кредита агента, срок

          select oks_balance
          from oksummary oks
          where :rep_date_end between oks.oks_date_begin and oks_date_end and
                cast(:rep_date_end as date) - oks_delinq_date > 7 and
                oks_close_state = 0 and
                oks.oks_from_k_id = :k_id
          into :agent_kont_balance_par7;

          if (agent_kont_balance_par7 > 0) then
          begin
            dolia = kc_amount_approved * 100 / k_amount_approved;
            agent_kont_balance_par7 = agent_kont_balance_par7 * dolia / 100;
          end

---------
          select max(ok.ok_ol_date - ok_sh_date)
          from operkred ok
          where ok.ok_ol_is_active = 1 and
                ok.ok_ol_date <= :rep_date_end and
                ok.ok_ol_date > ok_sh_date and
                ok_type < 7 and
                ok_from_k_id = :k_id and
                ok_os + ok_perc < 0 and
                ok_from_c_id = 0
          into delinq_day_repayment;

          select (cast(:rep_date_end as date) - oks_delinq_date) oks_delinq_day
          from oksummary oks
          where oks.oks_from_k_id = :k_id and
                oks_close_state = 0 and
                :rep_date_end between oks_date_begin and oks_date_end and
                cast(:rep_date_end as date) - oks_delinq_date >= 0
          into oks_delinq_day;

          if (delinq_day_repayment is null) then
            delinq_day_repayment = 0;
          if (oks_delinq_day is null) then
            oks_delinq_day = 0;

          if (oks_delinq_day > delinq_day_repayment) then
            agent_max_delinq_day = oks_delinq_day;
          else
            agent_max_delinq_day = delinq_day_repayment;
        end

        agent_first_kont_date = null;

        select ks.ks_date_begin
        from kontstate ks
        inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
        where ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1 and
              kncl.kc_amount_approved > 0 and
              kncl.kc_from_c_id = :rep_agent_c_id
        order by ks.ks_date_begin
        rows 1
        into :agent_first_kont_date;

        if (rep_agent_c_id <> rep_agent_c_id_old) then
        begin
          rep_agent_c_id_old = rep_agent_c_id;
-- Личностные характеристики ------------------------------------->>
          agent_nation = '';
          agent_family = '';
          agent_kol_child = 0;
          agent_education = '';

          select iif(cl.c_sex = 1, 'М', 'Ж'), cl.c_birth_date, cl.c_nation, cl.c_family, c_kol_child, cl.c_education,
                 cl.c_birth_city
          from clients cl
          where cl.c_id = :rep_agent_c_id
          into :agent_gender, :agent_c_birth_date, :agent_nation_id, :agent_family_id, :agent_kol_child,
               :agent_education_id, :agent_birth_city;

          if (agent_c_birth_date is not null) then
          begin
