CREATE PROCEDURE HP_KLADR_GET_LOG(
    IN_KSM_ID D_ID)
RETURNS (
    PLACE_NAME D_STR_LONG,
    KSL_COMM D_STR_LONG)
AS
declare variable pl_id d_id;
begin
  for select ksl_from_pl_id, ksl_comm
  from kladr_sync_log
  where ksl_from_ksm_id = :in_ksm_id
  into pl_id, ksl_comm
  do
  begin
    place_name = null;

    with recursive t1 (lvl, parent_id, name) as(
      select 0, parent, pl_name || ' ' || plt_short_name
      from places, placetype
      where plt_id = pl_status
        and pl_id = :pl_id
    union all
      select lvl + 1, parent, pl_name || ' ' || plt_short_name || ' / ' || name
      from t1, places, placetype
      where pl_id = parent_id
        and plt_id = pl_status
    )
    select name
    from t1
    order by lvl desc
    rows 1
    into place_name;

    suspend;
  end
end
