CREATE PROCEDURE HP_PERSINFO_DO_RESTORE(
    C_ID D_ID,
    JUST_CLEAR D_BOOL = 0)
AS
declare variable pif_key_id d_id;
declare variable pif_uuid d_uuid;
declare variable pif_indexed d_bool;
declare variable pif_value d_str_255;
declare variable pif_signed d_bool;
declare variable pif_usr_id d_id;
declare variable pif_datetime d_datetime;
declare variable pin_id d_id;
declare variable pda_id d_id;
declare variable pva_id d_id;
declare variable edl_id d_id;
begin
  rdb$set_context('USER_TRANSACTION', 'PERSINFO_ALL_RESTORING', 1);

  if (just_clear is null) then
    just_clear = 0;

  for select pin_id
  from persinfo
  where pin_object_kind = 0 and
        pin_object_id = :c_id
  into pin_id
  do
  begin
    for select pda_id
    from persdata
    where pda_from_pin_id = :pin_id
    into pda_id
    do
    begin
      for select pva_id
      from persvalue
      where pva_from_pda_id = :pda_id
      into pva_id
      do
      begin
        delete from persvalog
        where pvl_value_id = :pva_id;

        delete from persvalidx
        where pvx_pva_id = :pva_id;

        delete from persvalstr
        where pvs_pva_id = :pva_id;

        delete from persedoc
        where ped_pva_id = :pva_id;
      end

      delete from persvalue
      where pva_from_pda_id = :pda_id;
    end

    delete from persdata
    where pda_from_pin_id = :pin_id;
  end

  delete from persinfo
  where pin_object_kind = 0 and
        pin_object_id = :c_id;

  rdb$set_context('USER_TRANSACTION', 'PERSINFO_ALL_RESTORING', null);

  if (just_clear = 1) then
    exit;

  for select pif_key_id, pif_uuid, pif_indexed, pif_value, pif_signed, pif_usr_id,  pif_datetime
  from hp_persinfo_restore_tmp(:c_id)
  order by pif_uuid, pif_actial_data, pif_datetime
  into :pif_key_id, :pif_uuid, :pif_indexed, :pif_value, :pif_signed, :pif_usr_id, :pif_datetime
  do
    execute procedure hp_persvalue_restore(:pif_uuid, :pif_indexed, :pif_key_id, :pif_value, :pif_signed, :pif_datetime, :pif_usr_id);

  for select eds_from_edl_id
  from edoc_sign, edoc_list, edoc_cfg
  where eds_from_c_id = :c_id and
        eds_status = 1 and
        eds_from_edl_id = edl_id and
        edl_status = 1 and
        edl_from_edc_id = edc_id and
        coalesce(edc_pers_info, '') <> ''
  order by edl_create_datetime, eds_id
  into edl_id
  do
    execute procedure hp_persedoc_sign(:edl_id);
end
