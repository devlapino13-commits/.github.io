CREATE PROCEDURE HP_PDN_GET_REPLACE_UX
RETURNS (
    PRP_ID D_ID,
    KC_ID D_ID,
    KONTRACT_NO D_STR_CODE,
    PL_ID D_ID,
    PRP_DATE D_DATE,
    REASON D_STR_CODE,
    PRP_CREATE_DATETIME D_DATETIME,
    PRP_DATETIME D_DATETIME,
    K_ID D_ID,
    PL_NAME D_STR_255)
AS
begin
  for select prp_id, prp_kc_id, prp_date, prp_from_pl_id, prp_datetime, prp_create_datetime, pl_name
  from pdn_replace_places, hp_pdn_get_places
  where prp_from_pl_id = pl_id
  into prp_id, kc_id, prp_date, pl_id, prp_datetime, prp_create_datetime, pl_name
  do
  begin
    kontract_no = '';

    select k_kontract_no, k_id
    from knclient, kontract
    where kc_id = :kc_id
      and k_id = kc_from_k_id
    into kontract_no, k_id;

    select reason
    from hp_pdn_get_replace_info_ux(:prp_id, :k_id, :prp_date)
    into reason;

    suspend;
  end
end
