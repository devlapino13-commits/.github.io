CREATE PROCEDURE HP_LOAN_TARGET_CHECK(
    PARENT_ACL_ID ID_TYPE,
    OBJECT_KIND INTEGER,
    OBJECT_ID ID_TYPE,
    RES_RD_ID INTEGER,
    UN_ID ID_TYPE)
AS
declare variable fact_loan_target str_100;
declare variable mac_loan_target str_100;
declare variable is_ok integer;
declare variable res_acl_id id_type;
declare variable acl_respons_c_id id_type;
declare variable acl_from_reg_id integer;
begin
  if (object_kind = 0) then -- Client
  begin
    fact_loan_target = null;

    select rd.rd_code
    from refdata rd
    where rd.rd_id = :res_rd_id
    into :fact_loan_target;

    fact_loan_target = coalesce(fact_loan_target, '');

    mac_loan_target = null;

    for select rd.rd_code
        from kontract kont
        inner join operlist ol on kont.k_fact_disb_ol_id = ol.ol_id
        inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
        inner join refdata rd on rd.rd_id = kncl.kc_loan_target
        where ol.is_delete = 0 and
              kncl.kc_from_c_id = :object_id and
              kncl.kc_amount_approved > 0
        order by kont.k_fact_disb_date desc
        rows 1
        into :mac_loan_target
    do
    begin
      mac_loan_target = coalesce(mac_loan_target, '');

      is_ok = 0; -- 0 - не целевое использование кредита, 1 - целевое

      if ((mac_loan_target in ('Loan_Target.BusinessCreation', 'Loan_Target.BusinessGrowth')) and
          (mac_loan_target = fact_loan_target)) then
        is_ok = 1;
      if (((mac_loan_target = 'Loan_Target.BusinessCreation') and
          (fact_loan_target = 'Loan_Target.BusinessGrowth')) or ((mac_loan_target = 'Loan_Target.BusinessGrowth') and
          (fact_loan_target = 'Loan_Target.BusinessCreation'))) then
        is_ok = 1;
      if ((mac_loan_target = 'Loan_Target.Consumer') and
          (fact_loan_target <> '')) then
        is_ok = 1;

      if (is_ok = 0) then
      begin
        select al2.acl_respons_c_id, al2.acl_from_reg_id
        from actionlist al2
        where al2.acl_id = :parent_acl_id
        into :acl_respons_c_id, :acl_from_reg_id;

        execute procedure hp_actionlist_insert(:parent_acl_id, :acl_from_reg_id, 'Clients', :object_id,
            'Anketa.ClientAnaliz', 0, :un_id)
            returning_values :res_acl_id;

        update actionlist al
        set al.acl_respons_c_id = :acl_respons_c_id
        where al.acl_id = :res_acl_id;
      end
    end
  end
end
