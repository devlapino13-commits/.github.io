CREATE PROCEDURE AAREP_EXTERNAL_CALLS_STATISTICS(
    D1 DATE,
    D2 DATE)
RETURNS (
    CHECKED_SUM INTEGER,
    CALL_USER VARCHAR(255),
    COUNT_CHECKS INTEGER,
    COUNT_CHECKS_CURRENT INTEGER,
    COUNT_CHECKS_OLD INTEGER,
    COUNT_CHECKS_P DECIMAL(3,2))
AS
declare variable checked_sum_v decimal(18,2);
declare variable count_checks_v decimal(18,2);
begin
  select count(*) /*всего проверено за период*/
  from contacts cont
  where cont.cc_total_res <> 0 and
        cont.cc_checked_date between :d1 and :d2
  into :checked_sum;

  if (checked_sum is null) then
    checked_sum = 0;

  /* всего проверенных за период в разрезе оператора*/
  for select count(cont.cc_id), cc_call_user
      from contacts cont
      where cont.cc_total_res <> 0 and
            cont.cc_checked_date between :d1 and :d2
      group by cc_call_user
      order by cont.cc_call_user
      into :count_checks, :call_user
  do
  begin
    if (coalesce(checked_sum, 0) > 0) then
    begin
      count_checks_v = count_checks;
      count_checks_p = count_checks_v / checked_sum * 100;
    end
    else
      count_checks_p = 0;

    select count(cont.cc_id)
    from contacts cont
    where cont.cc_total_res <> 0 and
          cont.cc_checked_date between :d1 and :d2 and
          cc_date + 0 < :d1 --  + 0   не убирать, иначе не тот индекс цепляется
          and
          cc_call_user = :call_user
    into :count_checks_old;
    if (count_checks_old is null) then
      count_checks_old = 0;
    count_checks_current = count_checks - count_checks_old;

    suspend;
  end
end
