CREATE PROCEDURE HP_AKB_SET_POCKET_STATUS(
    POCKET_ID D_INTEGER,
    SEND_STATUS D_STR_100,
    POCKET_MSG D_STR_255)
AS
declare variable mp_id d_id;
declare variable k_id d_id;
declare variable kc_num d_integer;
declare variable lp_number d_integer;
declare variable account_no d_str_255;
begin
  if (f_get_db_id() <> 94401) then
    exit;

  if (send_status = 'ok') then
  begin
    update mkr_poket mp
    set mp.mp_is_sended = 1
    where mp.mp_id = :pocket_id;

    for select mp.mp_id, mc.mc_from_k_id
        from mkr_poket mp
        inner join mkr_credit mc on mp.mp_id = mc.mc_from_mp_id
        inner join knclient kc on mc.mc_from_k_id = kc.kc_from_k_id and
              kc.kc_amount_approved > 0
        where mp.mp_id = :pocket_id
        into :mp_id, :k_id
    do
    begin
      update mkr_credit mc
      set mc.mc_is_err = 0
      where mc.mc_from_mp_id = :mp_id and
            mc.mc_from_k_id = :k_id and
            mc.mc_is_err <> 1;
    end
  end

  if (send_status = 'empty') then
  begin
    update mkr_poket mp
    set mp.mp_is_sended = -1
    where mp.mp_id = :pocket_id;
  end

  if (send_status = 'failed') then
  begin
    update mkr_poket mp
    set mp.mp_is_sended = -2,
        mp.mp_send_info = :pocket_msg
    where mp.mp_id = :pocket_id;
  end

  if (send_status = 'processing') then
  begin
    update mkr_poket mp
    set mp.mp_is_sended = 2
    where mp.mp_id = :pocket_id;
  end

  if (send_status = 'processed') then
  begin
    update mkr_poket mp
    set mp.mp_is_sended = 3
    where mp.mp_id = :pocket_id;
  end
end
