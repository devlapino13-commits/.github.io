CREATE PROCEDURE DPD_ANKETA_CHECK(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    KIND D_SMALLINT = 0)
RETURNS (
    RES_CODE D_SMALLINT,
    RES_TEXT D_STR_LONG)
AS
declare variable survey_json d_str_8100;
declare variable jso d_integer;
declare variable gdoc d_smallint;
declare variable dtlastm d_integer;
declare variable dtjob d_str_20;
declare variable employer d_str_255;
declare variable income decimal(18,2);
declare variable taxrate d_smallint;
declare variable nmonth d_smallint;
declare variable incomewt decimal(18,2);
declare variable incomeavg decimal(18,2);
declare variable data_count d_smallint;
declare variable c_id d_id;
begin
    res_code = 1;
    res_text = '';

    if (:obj_kind <> 'miniappclient_income') then
    begin
        res_code = -1;
        res_text = 'Неверный тип объекта';

        suspend;
        exit;
    end

    select c_id,
           survey_json
    from pdn_info_action(:obj_kind, :obj_id)
    into :c_id,
         :survey_json;

    if (trim(:survey_json) = '{}' or :survey_json is null) then
    begin
        res_code = -1;
        res_text = 'Анкета документально подтвержденного дохода не заполнена';

        suspend;
        exit;
    end

    jso = json.parse(:survey_json);

    if (kind in (0, 1)) then
    begin
        gdoc     = json.obj_get_val(jso, 'gdoc', null);
        dtlastm  = json.obj_get_val(jso, 'dtlastm', null);
        employer = json.obj_get_val(jso, 'employer', '');
        dtjob    = json.obj_get_val(jso, 'dtjob', '');

        if (not exists(select *
                       from clientdocs cds
                       where cds.cds_from_c_id = :c_id
                         and cds.cds_is_delete = 0
                         and cds.cds_from_pk_id = 390
                         and f_date_month_begin(cds.cds_doc_date_begin) = f_date_month_begin(current_date))
            and coalesce(:dtjob, '') <> '')
        then
        begin
            res_code = -1;
            res_text = 'Не прикреплен документ о работе у одного работодателя';

            suspend;
            exit;
        end

        if(not exists(select *
                      from clientdocs cds
                      where cds.cds_from_c_id = :c_id
                        and cds.cds_is_delete = 0
                        and cds.cds_from_pk_id = 77
                        and f_date_month_begin(cds.cds_doc_date_begin) = f_date_month_begin(current_date))) then
        begin
            res_code = -1;
            res_text = 'Не прикреплен документ о доходе';

            suspend;
            exit;
        end

        if (gdoc is null) then
        begin
            res_code = -1;
            res_text = 'Нет ответов на п. 1.1 и 1.2. Сохранение Шага 1 невозможно.';

            suspend;
            exit;
        end

        if (dtlastm is null) then
        begin
            res_code = -1;
            res_text = 'Не указан месяц последнего полученного дохода (п. 1.2)';

            suspend;
            exit;
        end

        if (coalesce(employer, '') = '') then
        begin
            res_code = -1;
            res_text = 'Не указано наименование работодателя';

            suspend;
            exit;
        end

        if (coalesce(:dtjob, '') <> '' and dtlastm < 12) then
        begin
            res_code = -1;
            res_text = 'Вопрос 1.3 доступен только при выборе в вопросе 1.2 месяца расчета ПДН или месяца, предшествующего ему.';

            suspend;
            exit;
        end
    end

    if (kind in (0, 2)) then
    begin
        income = json.obj_get_val(jso, 'income', 0);
        data_count = f_json_str_get_count(jso, 'data');

        if (:data_count = 0) then
        begin
            res_code = -1;
            res_text = 'Таблица заполнена не полностью. Сохранение анкеты без ответа на вопрос 2.1 невозможно.';

            suspend;
            exit;
        end

        if (:income <= 0) then
        begin
            res_code = -1;
            res_text = 'Итоговый доход по анкете должен быть больше 0';

            suspend;
            exit;
        end
    end

    if (kind = 0) then
    begin
        gdoc      = json.obj_get_val(jso, 'gdoc', -1);
        taxrate   = json.obj_get_val(jso, 'taxrate', -1);
        nmonth    = json.obj_get_val(jso, 'nmonth', 0);
        incomewt  = json.obj_get_val(jso, 'incomewt', 0);
        incomeavg = json.obj_get_val(jso, 'incomeavg', 0);

        if (:taxrate not in (0, 13)) then
        begin
            res_code = -1;
            res_text = 'Некорректная ставка НДФЛ';

            suspend;
            exit;
        end

        if (:nmonth < 3 or :nmonth > 12) then
        begin
            res_code = -1;
            res_text = 'Количество месяцев для расчёта среднего дохода должно быть от 3 до 12';

            suspend;
            exit;
        end

        if (:incomeavg <= 0) then
        begin
            res_code = -1;
            res_text = 'Не рассчитан среднемесячный подтверждённый доход';

            suspend;
            exit;
        end

        if (:incomewt < 0) then
        begin
            res_code = -1;
            res_text = 'Не рассчитан доход за вычетом НДФЛ';

            suspend;
            exit;
        end
    end

    suspend;
end
