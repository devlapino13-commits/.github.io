CREATE PROCEDURE HP_DO_CHECK_KN_OPERBOOK_ONE(
    CHECK_FULL_CODE D_STR_255,
    REP_DATE DATE,
    REP_KC_ID D_ID)
RETURNS (
    SAB_FULL_CODE D_STR_CODE,
    K_ID ID_TYPE,
    K_KONTRACT_NO D_STR_CODE,
    KC_ID ID_TYPE,
    ACC_ID ID_TYPE,
    CLOSE_STATE ID_TYPE,
    OKS_AMOUNT CURR_TYPE,
    OB_AMOUNT CURR_TYPE)
AS
declare variable sab_id id_type;
declare variable total_oks curr_type;
declare variable total_ob curr_type;
declare variable korzina integer;
declare variable rep_date_begin d_date;
begin
  rep_date_begin = rep_date - extract(day from rep_date) + 1;

  for select sab_id, sab_full_code
      from sys_accbalance
      where sab_full_code = :check_full_code or
            (:check_full_code = 'full' and
            (sab_full_code in ('Микрозаймы.Баланс.ОС', 'Микрозаймы.Баланс.ИзмВБС', 'Микрозаймы.Внебаланс.ОС', 'Микрозаймы.Внебаланс.Процент', 'Микрозаймы.Начислено.Процент', 'Микрозаймы.Погашено.Госпошлина', 'Микрозаймы.Погашено.Переплата') or
            sab_full_code starting 'Микрозаймы.Резерв.Корректировка' or
            sab_full_code starting 'Микрозаймы.Резерв.Госпошлина' or
            sab_full_code starting 'Микрозаймы.Резерв.ОС' or
            sab_full_code starting 'Микрозаймы.Резерв.Процент'))
      into :sab_id, sab_full_code
  do
  begin
    total_oks = 0;
    total_ob = 0;

    k_id = null;
    kc_id = 0;
    k_kontract_no = '';

    if (sab_full_code = 'Микрозаймы.Баланс.ОС') then
    begin
      acc_id = null;
      for select oks_from_k_id, oks_from_kc_id, oks_close_state, oks_balance
          from kn_oksummary
          where oks_date_begin <= :rep_date and
                oks_date_end >= :rep_date and
                oks_from_kc_id = :rep_kc_id
          into k_id, kc_id, close_state, oks_amount
      do
      begin
        ob_amount = f_get_acc_amount(kc_id, sab_id, null, rep_date, 'bal');

        total_oks = total_oks + oks_amount;
        total_ob = total_ob + ob_amount;

        if (oks_amount <> ob_amount) then
        begin
          k_kontract_no = '';

          select k_kontract_no
          from kontract
          where k_id = :k_id
          into k_kontract_no;

          suspend;
        end
      end
    end

    if (sab_full_code = 'Микрозаймы.Начислено.Процент') then
    begin
      acc_id = null;
      for select oks_from_k_id, oks_from_kc_id, oks_close_state, oks_bal_perc
          from kn_oksummary
          where oks_date_begin <= :rep_date and
                oks_date_end >= :rep_date and
                oks_from_kc_id = :rep_kc_id
          into k_id, kc_id, close_state, oks_amount
      do
      begin
        ob_amount = f_get_acc_amount(kc_id, sab_id, null, rep_date, 'bal');

        total_oks = total_oks + oks_amount;
        total_ob = total_ob + ob_amount;

        if (oks_amount <> ob_amount) then
        begin
          k_kontract_no = '';

          select k_kontract_no
          from kontract
          where k_id = :k_id
          into k_kontract_no;

          suspend;
        end
      end
    end

    if (sab_full_code starting 'Микрозаймы.Резерв.Корректировка') then
    begin
      acc_id = null;
      korzina = right(sab_full_code, 1);
      for select oks_from_k_id, oks_from_kc_id, oks_close_state, iif(oks_close_state = 0, oks_diff_ecl_llr, 0)
          from kn_oksummary
          where oks_date_begin <= :rep_date and
                oks_date_end >= :rep_date and
                oks_ecl_acc_kind = :korzina and
                oks_from_kc_id = :rep_kc_id
          into k_id, kc_id, close_state, oks_amount
      do
      begin
        ob_amount = -f_get_acc_amount(kc_id, sab_id, null, rep_date, 'bal');

        total_oks = total_oks + oks_amount;
        total_ob = total_ob + ob_amount;

        if (oks_amount <> ob_amount) then
        begin
          k_kontract_no = '';

          select k_kontract_no
          from kontract
          where k_id = :k_id
          into k_kontract_no;

          suspend;
        end
      end
    end

    if (sab_full_code starting 'Микрозаймы.Резерв.ОС') then
    begin
      acc_id = null;
      korzina = right(sab_full_code, 1);
      for select oks_from_k_id, oks_from_kc_id, oks_close_state, oks_llr_os
          from kn_oksummary
          where oks_date_begin <= :rep_date and
                oks_date_end >= :rep_date and
                oks_ecl_acc_kind = :korzina and
                oks_from_kc_id = :rep_kc_id
          into k_id, kc_id, close_state, oks_amount
      do
      begin
        ob_amount = -f_get_acc_amount(kc_id, sab_id, null, rep_date, 'bal');

        total_oks = total_oks + oks_amount;
        total_ob = total_ob + ob_amount;

        if (oks_amount <> ob_amount) then
        begin
          k_kontract_no = '';

          select k_kontract_no
          from kontract
          where k_id = :k_id
          into k_kontract_no;

          suspend;
        end
      end
    end

    if (sab_full_code starting 'Микрозаймы.Резерв.Процент') then
    begin
      acc_id = null;
      korzina = right(sab_full_code, 1);
      for select oks_from_k_id, oks_from_kc_id, oks_close_state, oks_llr_perc
          from kn_oksummary
          where oks_date_begin <= :rep_date and
                oks_date_end >= :rep_date and
                oks_ecl_acc_kind = :korzina and
                oks_from_kc_id = :rep_kc_id
          into k_id, kc_id, close_state, oks_amount
      do
      begin
        ob_amount = -f_get_acc_amount(kc_id, sab_id, null, rep_date, 'bal');

        total_oks = total_oks + oks_amount;
        total_ob = total_ob + ob_amount;

        if (oks_amount <> ob_amount) then
        begin
          k_kontract_no = '';

          select k_kontract_no
          from kontract
          where k_id = :k_id
          into k_kontract_no;

          suspend;
        end
      end
    end

    if (sab_full_code starting 'Микрозаймы.Резерв.Госпошлина') then
    begin
      acc_id = null;
      korzina = right(sab_full_code, 1);
      for select oks_from_k_id, oks_from_kc_id, oks_close_state, oks_reserv_law
          from kn_oksummary
          where oks_date_begin <= :rep_date and
                oks_date_end >= :rep_date and
                oks_ecl_acc_kind = :korzina and
                oks_from_kc_id = :rep_kc_id
          into k_id, kc_id, close_state, oks_amount
      do
      begin
        ob_amount = -f_get_acc_amount(kc_id, sab_id, null, rep_date, 'bal');

        total_oks = total_oks + oks_amount;
        total_ob = total_ob + ob_amount;

        if (oks_amount <> ob_amount) then
        begin
          k_kontract_no = '';

          select k_kontract_no
          from kontract
          where k_id = :k_id
          into k_kontract_no;

          suspend;
        end
      end
    end

    if (sab_full_code = 'Микрозаймы.Баланс.ИзмВБС') then
    begin
      acc_id = null;
      for select oks_from_k_id, oks_from_kc_id, oks_close_state, iif(oks_close_state = 0, oks_diff_esp_bal, 0)
          from kn_oksummary
          where oks_date_begin <= :rep_date and
                oks_date_end >= :rep_date and
                oks_from_kc_id = :rep_kc_id
          into k_id, kc_id, close_state, oks_amount
      do
      begin
        ob_amount = f_get_acc_amount(kc_id, sab_id, null, rep_date, 'bal');

        total_oks = total_oks + oks_amount;
        total_ob = total_ob + ob_amount;

        if (oks_amount <> ob_amount) then
        begin
          k_kontract_no = '';

          select k_kontract_no
          from kontract
          where k_id = :k_id
          into k_kontract_no;

          suspend;
        end
      end
    end

    if (sab_full_code = 'Микрозаймы.Внебаланс.ОС') then
    begin
      acc_id = null;
      close_state = -1;
      for select oko_from_k_id, kc_id, sum(oko_os + oko_sh_os)
          from operkroff, knclient
          where oko_ol_date <= :rep_date and
                oko_ol_is_active = 1 and
                oko_from_k_id = kc_from_k_id and
                oko_from_c_id = kc_from_c_id and
                kc_amount_approved > 0 and
                kc_id = :rep_kc_id
          group by 1, 2
          into k_id, kc_id, oks_amount
      do
      begin
        ob_amount = f_get_acc_amount(kc_id, sab_id, null, rep_date, 'bal');

        total_oks = total_oks + oks_amount;
        total_ob = total_ob + ob_amount;

        if (oks_amount <> ob_amount) then
        begin
          k_kontract_no = '';

          select k_kontract_no
          from kontract
          where k_id = :k_id
          into k_kontract_no;

          suspend;
        end
      end

      k_id = null;
      kc_id = 0;
      k_kontract_no = '';

      oks_amount = 0;
      for select acc_id, acc_object_id
          from ref_account
          where acc_object_kind = 8 and
                acc_from_sab_id = :sab_id and
                acc_object_id = :rep_kc_id
          into acc_id, kc_id
      do
      begin
        if (exists(select *
                   from operkroff, knclient
                   where oko_ol_date <= :rep_date and
                         oko_ol_is_active = 1 and
                         oko_from_k_id = kc_from_k_id and
                         oko_from_c_id = kc_from_c_id and
                         kc_amount_approved > 0 and
                         kc_id = :rep_kc_id)) then
          continue;

        ob_amount = f_acc_bal(acc_id, rep_date);

        total_ob = total_ob + ob_amount;

        if (ob_amount <> 0) then
        begin
          k_id = null;
          k_kontract_no = '';

          select kc_from_k_id, k_kontract_no
          from knclient, kontract
          where kc_id = :kc_id and
                kc_from_k_id = k_id
          into k_id, k_kontract_no;

          suspend;
        end
      end
    end

    if (sab_full_code = 'Микрозаймы.Внебаланс.Процент') then
    begin
      acc_id = null;
      close_state = -1;
      for select oko_from_k_id, kc_id, sum(oko_perc + oko_sh_perc)
          from operkroff, knclient
          where oko_ol_date <= :rep_date and
                oko_ol_is_active = 1 and
                oko_from_k_id = kc_from_k_id and
                oko_from_c_id = kc_from_c_id and
                kc_amount_approved > 0 and
                kc_id = :rep_kc_id
          group by 1, 2
          into k_id, kc_id, oks_amount
      do
      begin
        ob_amount = f_get_acc_amount(kc_id, sab_id, null, rep_date, 'bal');

        total_oks = total_oks + oks_amount;
        total_ob = total_ob + ob_amount;

        if (oks_amount <> ob_amount) then
        begin
          k_kontract_no = '';

          select k_kontract_no
          from kontract
          where k_id = :k_id
          into k_kontract_no;

          suspend;
        end
      end

      k_id = null;
      kc_id = 0;
      k_kontract_no = '';

      oks_amount = 0;
      for select acc_id, acc_object_id
          from ref_account
          where acc_object_kind = 8 and
                acc_from_sab_id = :sab_id and
                acc_object_id = :rep_kc_id
          into acc_id, kc_id
      do
      begin
        if (exists(select *
                   from operkroff, knclient
                   where oko_ol_date <= :rep_date and
                         oko_ol_is_active = 1 and
                         oko_from_k_id = kc_from_k_id and
                         oko_from_c_id = kc_from_c_id and
                         kc_amount_approved > 0 and
                         kc_id = :rep_kc_id)) then
          continue;

        ob_amount = f_acc_bal(acc_id, rep_date);

        total_ob = total_ob + ob_amount;

        if (ob_amount <> 0) then
        begin
          k_id = null;
          k_kontract_no = '';

          select kc_from_k_id, k_kontract_no
          from knclient, kontract
          where kc_id = :kc_id and
                kc_from_k_id = k_id
          into k_id, k_kontract_no;

          suspend;
        end
      end
    end

    if (sab_full_code = 'Микрозаймы.Погашено.Переплата') then
    begin
      acc_id = null;
      for select kc_from_k_id, ok_from_kc_id, sum(ok_os)
          from kn_operkred, knclient
          where ok_ol_date <= :rep_date and
                ok_ol_is_active = 1 and
                ok_from_kc_id = kc_id and
                ok_type = 9 and
                ok_from_kc_id = :rep_kc_id
          group by 1, 2
          into k_id, kc_id, oks_amount
      do
      begin
        ob_amount = f_get_acc_amount(kc_id, sab_id, null, rep_date, 'bal');

        total_oks = total_oks + oks_amount;
        total_ob = total_ob + ob_amount;

        if (oks_amount <> ob_amount) then
        begin
          k_kontract_no = '';

          select k_kontract_no
          from kontract
          where k_id = :k_id
          into k_kontract_no;

          suspend;
        end
      end

      k_id = null;
      kc_id = 0;
      k_kontract_no = '';

      oks_amount = 0;
      for select acc_id, acc_object_id
          from ref_account
          where acc_object_kind = 8 and
                acc_from_sab_id = :sab_id and
                acc_object_id = :rep_kc_id
          into acc_id, kc_id
      do
      begin
        if (exists(select *
                   from kn_operkred
                   where ok_from_kc_id = :kc_id and
                         ok_ol_date <= :rep_date and
                         ok_ol_is_active = 1 and
                         ok_type = 9)) then
          continue;

        ob_amount = f_acc_bal(acc_id, rep_date);

        total_ob = total_ob + ob_amount;

        if (ob_amount <> 0) then
        begin
          k_id = null;
          k_kontract_no = '';

          select kc_from_k_id, k_kontract_no
          from knclient, kontract
          where kc_id = :kc_id and
                kc_from_k_id = k_id
          into k_id, k_kontract_no;

          suspend;
        end
      end
    end

    if (sab_full_code = 'Микрозаймы.Погашено.Госпошлина') then
    begin
      acc_id = null;
      for select oks_from_k_id, oks_from_kc_id, oks_close_state, oks_bal_law
          from kn_oksummary
          where oks_date_begin <= :rep_date and
                oks_date_end >= :rep_date and
                oks_from_kc_id = :rep_kc_id
          into k_id, kc_id, close_state, oks_amount
      do
      begin
        ob_amount = f_get_acc_amount(kc_id, sab_id, null, rep_date, 'bal');

        total_oks = total_oks + oks_amount;
        total_ob = total_ob + ob_amount;

        if (oks_amount <> ob_amount) then
        begin
          k_kontract_no = '';

          select k_kontract_no
          from kontract
          where k_id = :k_id
          into k_kontract_no;

          suspend;
        end
      end
    end

    k_id = null;
    kc_id = 0;
    k_kontract_no = '';

    k_id = null;
    kc_id = 0;
    k_kontract_no = 'Total amount';
    oks_amount = total_oks;
    ob_amount = total_ob;
    suspend;
  end

  total_oks = 0;
  total_ob = 0;

  k_id = null;
  kc_id = 0;
  k_kontract_no = '';

  if (check_full_code in ('full', 'vbs')) then
  begin
    sab_full_code = 'vbs';
    acc_id = null;
    for select oks_from_k_id, oks_from_kc_id, oks_close_state, oks_bal_esp_end - oks_off_esp_bal
        from kn_oksummary
        where oks_date_begin <= :rep_date and
              oks_date_end >= :rep_date and
              oks_from_kc_id = :rep_kc_id
        into k_id, kc_id, close_state, oks_amount
    do
    begin
      if (close_state = 2) then
        oks_amount = 0;

      ob_amount = 0;
      select sum(kobs_bal_sum)
      from kn_obsummary, ref_account
      where kobs_acc_id = acc_id and
            acc_object_kind = 8 and
            acc_object_id = :kc_id and
            acc_from_sab_id in (12, 13, 15, 16, 20, 21, 54, 55) and
            kobs_date <= :rep_date and
            kobs_date_end >= :rep_date
      into ob_amount;
      if (ob_amount is null) then
        ob_amount = 0;

      total_oks = total_oks + oks_amount;
      total_ob = total_ob + ob_amount;

      if (oks_amount <> ob_amount) then
      begin
        k_kontract_no = '';

        select k_kontract_no
        from kontract
        where k_id = :k_id
        into k_kontract_no;

        suspend;
      end
    end
    k_id = null;
    kc_id = 0;
    k_kontract_no = 'Total amount';
    oks_amount = total_oks;
    ob_amount = total_ob;
    suspend;
  end

  if (check_full_code in ('full', 'reserv')) then
  begin
    sab_full_code = 'reserv all';
    acc_id = null;
    k_id = null;
    kc_id = 0;
    k_kontract_no = '';
    oks_amount = total_oks;
    ob_amount = total_ob;

    for with tab as
        (select oks_from_k_id as k_id, oks_from_kc_id as kc_id, oks_ecl_vbs as oks_amount, 0 as kobs_amount
        from kn_oksummary
        where oks_date_begin <= :rep_date and
              oks_date_end >= :rep_date and
              oks_from_kc_id = :rep_kc_id and
              oks_close_state = 0
        union all
        select kc_from_k_id, kc_id, 0 as oks_amount, -kobs_bal_sum as kobs_amount
        from ref_account, kn_obsummary, knclient
        where acc_object_kind = 8 and
              acc_object_id = :rep_kc_id and
              acc_from_sab_id between 30 and 38 and -- все счета резерва кроме госпошлины
              acc_id = kobs_acc_id and
              kobs_date <= :rep_date and
              kobs_date_end >= :rep_date and
              acc_object_id = kc_id
        )
    select k_id, kc_id, sum(oks_amount), sum(kobs_amount)
    from tab
    group by 1, 2
    having sum(oks_amount) <> sum(kobs_amount)
    into k_id, kc_id, oks_amount, ob_amount
    do
      Suspend;
  end


  if (check_full_code in ('full', 'vbs_out')) then
  begin
    sab_full_code = 'vbs_out';
    acc_id = null;
    k_id = null;
    kc_id = 0;
    k_kontract_no = '';
    oks_amount = total_oks;
    ob_amount = total_ob;

    for with tab as
        (select oks_from_k_id as k_id, oks_from_kc_id as kc_id, oks_off_esp_bal as oks_amount, 0 as kobs_amount
        from kn_oksummary
        where oks_date_begin between :rep_date_begin and :rep_date and
              oks_close_state = 1 and
              oks_from_kc_id = :rep_kc_id
        union all
        select kc_from_k_id, kc_id, 0 as oks_amount, iif(a_dt.acc_from_sab_id in (57, 89), kob_summa, -kob_summa) as kobs_amount
        from ref_account a_dt, ref_account a_kt, kn_operbook, knclient
        where kob_from_kc_id = kc_id and
              ((a_dt.acc_from_sab_id = 13 and a_kt.acc_from_sab_id in (57, 89)) or
               (a_dt.acc_from_sab_id in (57, 89) and a_kt.acc_from_sab_id = 13)) and
              a_dt.acc_id = kob_acc_dt and
              a_kt.acc_id = kob_acc_kt and
              kob_ol_date between :rep_date_begin and :rep_date and
              kob_ol_is_active = 1 and
              kc_id = :rep_kc_id

        )
    select k_id, kc_id, sum(oks_amount), sum(kobs_amount)
    from tab
    group by 1, 2
    having sum(oks_amount) <> sum(kobs_amount)
    into k_id, kc_id, oks_amount, ob_amount
    do
      Suspend;
  end
