CREATE PROCEDURE AAREP_CELLAKTLIST_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    CAL_FROM_CAM_ID DECIMAL(18,0),
    CAL_ID INTEGER,
    CAL_DATE_BEGIN DATE,
    CAL_DATE_END DATE,
    CAL_FROM_C_ID DECIMAL(18,0),
    CAL_CELL_NO VARCHAR(20),
    CAL_IS_CONTACT INTEGER,
    CAL_IS_CORPORATE INTEGER,
    CAL_HAS_LIMIT INTEGER,
    CAL_FROM_COP_ID INTEGER,
    CAL_PS_TEXT VARCHAR(255),
    CAL_FIO VARCHAR(100),
    CAL_REGION_ID INTEGER,
    CAL_BRANCH_ID INTEGER,
    CAL_FROM_JL_ID INTEGER,
    CAL_FROM_JL_NAME VARCHAR(100),
    CAL_REGION_NAME VARCHAR(100),
    CAL_IS_WORKING INTEGER,
    CAL_STATE_STR VARCHAR(20),
    CLK_LIMIT_KIND VARCHAR(100))
AS
declare variable cal_state integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select cal_from_cam_id, cal_id, cal_date_begin, cal_date_end, cal_from_c_id, cal_cell_no, cal_state,
             cal_is_contact, cal_is_corporate, cal_has_limit, cal_from_cop_id, clk_name
      from cellaktlist, cellimitkind
      where cal_date_end >= :report_date_end and
            cal_date_begin <= :report_date_end and
            cal_has_limit = clk_id
      into :cal_from_cam_id, :cal_id, :cal_date_begin, :cal_date_end, :cal_from_c_id, :cal_cell_no, :cal_state,
           :cal_is_contact, :cal_is_corporate, :cal_has_limit, :cal_from_cop_id, :clk_limit_kind
  do
  begin

    cal_ps_text = null;
    cal_region_id = null;
    cal_branch_id = null;
    cal_from_jl_name = null;
    cal_region_name = null;
    cal_from_jl_id = null;
    cal_is_working = null;

    select c_name_fio
    from clients
    where c_id = :cal_from_c_id
    into cal_fio;

    select ps_text, ps_from_reg_id, ps_branch_id, ps_jl_name, ps_reg_name, ps_from_jl_id, ps_is_working
    from hr_personal_last_position(:cal_from_c_id, :report_date_end)
    rows 1
    into :cal_ps_text, :cal_region_id, :cal_branch_id, :cal_from_jl_name, :cal_region_name, :cal_from_jl_id,
         cal_is_working;

    if (cal_state = -1) then
      cal_state_str = 'Аннулирован';
    if (cal_state = 0) then
      cal_state_str = 'На складе';
    if (cal_state = 1) then
      cal_state_str = 'Действующий';
    suspend;
  end
end
