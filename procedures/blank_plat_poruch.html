CREATE PROCEDURE BLANK_PLAT_PORUCH(
    OL DECIMAL(18,0))
RETURNS (
    PP_DATE DATE,
    PP_DOC_NO INTEGER,
    PP_MEMO VARCHAR(1024),
    PP_IS_DELETE SMALLINT,
    PP_CLIENT_NAME VARCHAR(255),
    PP_CLIENT_REGIST_NO VARCHAR(20),
    PP_CLIENT_OKPO VARCHAR(20),
    PP_CLIENT_INN VARCHAR(20),
    PP_CLIENT_BANK_ID VARCHAR(10),
    PP_CLIENT_BANK_NAME VARCHAR(255),
    PP_CLIENT_BANK_ACCOUNT VARCHAR(50),
    PP_OFFICE_NAME VARCHAR(255),
    PP_OFFICE_REGIST_NO VARCHAR(255),
    PP_OFFICE_OKPO VARCHAR(20),
    PP_OFFICE_INN VARCHAR(20),
    PP_OFFICE_BANK_ID VARCHAR(10),
    PP_OFFICE_BANK_NAME VARCHAR(255),
    PP_OFFICE_BANK_ACCOUNT VARCHAR(50),
    PP_SUMMA DECIMAL(18,4),
    PP_PAY_CODE VARCHAR(20),
    PP_CLIENT_BANK_CORR_ACC VARCHAR(255),
    PP_OFFICE_BANK_CORR_ACC VARCHAR(255),
    PP_CLIENT_BANK_VOEN VARCHAR(255),
    PP_OFFICE_BANK_VOEN VARCHAR(255),
    PP_CLIENT_BANK_SWIFT VARCHAR(20),
    PP_OFFICE_BANK_SWIFT VARCHAR(20),
    PP_CLIENT_NAME_FULL VARCHAR(255),
    PP_OFFICE_NAME_FULL VARCHAR(255),
    PP_DATE_STR VARCHAR(255),
    OL_POCKET INTEGER)
AS
declare variable cl decimal(18,0);
declare variable office integer;
declare variable cm_name varchar(255);
declare variable cl_acc_no integer;
declare variable of_acc_no integer;
declare variable br_acc varchar(20);
declare variable reg_id integer;
declare variable ol_oper_type integer;
declare variable ol_oper_subtype integer;
declare variable cm_poruch_transit integer;
declare variable bc_name varchar(255);
declare variable pp_new_mem varchar(255);
declare variable ok_number_check integer;
declare variable ol_doc_client_name varchar(255);
declare variable pp_date_day integer;
declare variable pp_date_month integer;
declare variable pp_date_year integer;
declare variable cm_currency varchar(3);
declare variable st_staff integer;
declare variable st_client integer;
declare variable second_acc varchar(20);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */
  select ob_account_kt, ob_from_pc_id, sum(ob_summa_val)
  from operbook, acctypelist, acc_list
  where ob_from_ol_id = :ol and
        al_id = ob_account_kt and
        al_from_bg_id like atl_acc_mask || '%' and
        atl_from_at_id = 12
  group by ob_account_kt, ob_from_pc_id
  order by ob_account_kt, ob_from_pc_id
  rows 1
  into :br_acc, :pp_pay_code, :pp_summa;
  if (pp_summa is null) then
    pp_summa = 0;
  if (pp_pay_code is null) then
    pp_pay_code = '';

  second_acc = '';
  select ob_account_dt
  from operbook, acctypelist, acc_list
  where ob_from_ol_id = :ol and
        al_id = ob_account_dt and
        al_from_bg_id like atl_acc_mask || '%' and
        atl_from_at_id = 12 rows 1
  into :second_acc;
  if (second_acc is null) then
    second_acc = '';

  select cm_name, cm_poruch_transit, cm_default_currency
  from company
  rows 1
  into :cm_name, :cm_poruch_transit, :cm_currency;

  for select ol_date, ol_doc_no, ol_memo, is_delete, ol_document_client, ol_document_office,
             coalesce(ol_acc_no_client, 1), ol_acc_no_office, ol_from_reg_id, ol_oper_type, ol_oper_subtype,
             ol_doc_client_name, ol_pocket
      from operlist
      where ol_id = :ol
      into :pp_date, :pp_doc_no, :pp_memo, :pp_is_delete, :cl, :office, :cl_acc_no, :of_acc_no, :reg_id, :ol_oper_type,
           :ol_oper_subtype, :ol_doc_client_name, :ol_pocket
  do
  begin
    pp_date_day = extract(day from pp_date);
    pp_date_month = extract(month from pp_date);
    pp_date_year = extract(year from pp_date);
    if (cm_currency = '944') then
    begin
      if (pp_date_month = 1) then
        pp_date_str = pp_date_day || ' ' || ' Yanvar ' || pp_date_year || '-ci il';
      if (pp_date_month = 2) then
        pp_date_str = pp_date_day || ' ' || ' Fevral ' || pp_date_year || '-ci il';
      if (pp_date_month = 3) then
        pp_date_str = pp_date_day || ' ' || ' Mart ' || pp_date_year || '-ci il';
      if (pp_date_month = 4) then
        pp_date_str = pp_date_day || ' ' || ' Aprel ' || pp_date_year || '-ci il';
      if (pp_date_month = 5) then
        pp_date_str = pp_date_day || ' ' || ' May ' || pp_date_year || '-ci il';
      if (pp_date_month = 6) then
        pp_date_str = pp_date_day || ' ' || ' Iyun ' || pp_date_year || '-ci il';
      if (pp_date_month = 7) then
        pp_date_str = pp_date_day || ' ' || ' Iyul ' || pp_date_year || '-ci il';
      if (pp_date_month = 8) then
        pp_date_str = pp_date_day || ' ' || ' Avqust ' || pp_date_year || '-ci il';
      if (pp_date_month = 9) then
        pp_date_str = pp_date_day || ' ' || ' Sentyabr ' || pp_date_year || '-ci il';
      if (pp_date_month = 10) then
        pp_date_str = pp_date_day || ' ' || ' Oktyabr ' || pp_date_year || '-ci il';
      if (pp_date_month = 11) then
        pp_date_str = pp_date_day || ' ' || ' Noyabr ' || pp_date_year || '-ci il';
      if (pp_date_month = 12) then
        pp_date_str = pp_date_day || ' ' || ' Dekabr ' || pp_date_year || '-ci il';
    end
    else
    begin
      if (pp_date_month = 1) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' январь ' || pp_date_year || ' год';
      if (pp_date_month = 2) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' февраль ' || pp_date_year || ' год';
      if (pp_date_month = 3) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' март ' || pp_date_year || ' год';
      if (pp_date_month = 4) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' апрель ' || pp_date_year || ' год';
      if (pp_date_month = 5) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' май ' || pp_date_year || ' год';
      if (pp_date_month = 6) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' июнь ' || pp_date_year || ' год';
      if (pp_date_month = 7) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' июль ' || pp_date_year || ' год';
      if (pp_date_month = 8) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' август ' || pp_date_year || ' год';
      if (pp_date_month = 9) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' сентябрь ' || pp_date_year || ' год';
      if (pp_date_month = 10) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' октябрь ' || pp_date_year || ' год';
      if (pp_date_month = 11) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' ноябрь ' || pp_date_year || ' год';
      if (pp_date_month = 12) then
        pp_date_str = right('00' || pp_date_day, 2) || ' ' || ' декабрь ' || pp_date_year || ' год';
    end
    if (cl is not null) then
    begin
      if ((ol_oper_type <> 2) or (ol_oper_subtype <> 21)) then
      begin
        bc_name = '';
        if (cm_poruch_transit = 1) then
        begin
          select bc_name
          from clients, bankclient
          where c_id = :cl and
                bc_from_c_id = c_id and
                bc_number = :cl_acc_no and
                upper(bc_name) like '%ТРАНЗИТ%' rows 1
          into :bc_name;
          if ((bc_name is null) or (bc_name = '')) then
          begin
            cm_poruch_transit = 0;
          end
        end
        select c_name_fio, c_reg_no, c_okpo, c_inn, b_id, b_name, bc_account_no, bc_name, b_corr_acc, b_mfo_old,
               b_swift
        from clients
        left join(bankclient left join banks on (bc_from_b_id = b_id)) on (bc_from_c_id = c_id and
              bc_number = :cl_acc_no)
        where c_id = :cl rows 1
        into :pp_client_name, :pp_client_regist_no, :pp_client_okpo, :pp_client_inn, :pp_client_bank_id,
             :pp_client_bank_name, :pp_client_bank_account, :bc_name, :pp_client_bank_corr_acc, pp_client_bank_voen,
             pp_client_bank_swift;

        if ((bc_name is not null) and
            (bc_name <> '') and
            (cm_poruch_transit = 1)) then
        begin
          pp_client_name = bc_name;
          pp_new_mem = '';
          select 'ФИО: ' || c_name_fio || ' Паспорт: ' || c_pass_no || ', выдан: ' || c_pass_date || ' ' || c_pass_who
          from clients
          where c_id = :cl
          into :pp_new_mem;
          if ((pp_new_mem is not null) and
              (pp_new_mem <> '')) then
          begin
            pp_memo = pp_memo || ' ' || :pp_new_mem;
          end
        end
        else
        begin
          if ((pp_client_name is null) or (pp_client_name = '')) then
            pp_client_name = ol_doc_client_name;
        end
      end
      else
      begin
        select distinct c_name_fio, c_reg_no, c_okpo, c_inn, b_id, b_name, sack_bank_account, b_corr_acc, b_mfo_old,
                        b_swift
        from opersacard, clients
        left join(sa_cardkind left join banks on (sack_bank_code = b_id)) on (sack_from_c_id = c_id)
        where c_id = :cl and
              osc_from_ol_id = :ol and
              osc_bank_account = sack_bank_account rows 1
        into :pp_client_name, :pp_client_regist_no, :pp_client_okpo, :pp_client_inn, :pp_client_bank_id,
             :pp_client_bank_name, :pp_client_bank_account, :pp_client_bank_corr_acc, pp_client_bank_voen,
             pp_client_bank_swift;
      end
    end
    if (office is not null) then
    begin
      select reg_kont_company, reg_company_name, reg_sfond, reg_okpo, reg_inn, br_from_b_id, b_name, br_account_no,
             b_corr_acc, b_mfo_old, b_swift
      from region
      left join(bankregion left join banks on (br_from_b_id = b_id)) on (br_from_reg_id = reg_unicode and
            br_number = :of_acc_no)
      where reg_unicode = :office
      order by br_is_active desc
      rows 1
      into :pp_client_name_full, :pp_client_name, :pp_client_regist_no, :pp_client_okpo, :pp_client_inn,
           :pp_client_bank_id, :pp_client_bank_name, :pp_client_bank_account, :pp_client_bank_corr_acc,
           pp_client_bank_voen, pp_client_bank_swift;
    end
    if ((office is null) and
        (cl is null) and
        (second_acc <> '')) then
    begin
-- Значит перевод между счетами одного офиса
      select reg_kont_company, reg_company_name, reg_sfond, reg_okpo, reg_inn, br_from_b_id, b_name, br_account_no,
             b_corr_acc, b_mfo_old, b_swift
      from region
      left join(bankregion left join banks on (br_from_b_id = b_id)) on (br_from_reg_id = reg_unicode and
            br_name = :second_acc)
      where reg_unicode = :reg_id
      order by br_is_active desc
      rows 1
      into :pp_client_name_full, :pp_client_name, :pp_client_regist_no, :pp_client_okpo, :pp_client_inn,
           :pp_client_bank_id, :pp_client_bank_name, :pp_client_bank_account, :pp_client_bank_corr_acc,
           pp_client_bank_voen, pp_client_bank_swift;

      if (pp_client_name_full is null) then
        pp_client_name_full = pp_office_name;
    end

    if (pp_client_name is null) then
      pp_client_name = :ol_doc_client_name;
    if (pp_client_name_full is null) then
      pp_client_name_full = :ol_doc_client_name;
    if (pp_client_name_full is null or pp_client_name_full = '') then
      pp_client_name_full = :pp_client_name;

    select reg_kont_company, reg_company_name, reg_sfond, reg_okpo, reg_inn, br_from_b_id, b_name, br_account_no,
           b_corr_acc, b_mfo_old, b_swift
    from region
    left join(bankregion left join banks on (br_from_b_id = b_id)) on (br_from_reg_id = reg_unicode and
          br_name = :br_acc)
    where reg_unicode = :reg_id
    order by br_is_active desc
    rows 1
    into :pp_office_name_full, :pp_office_name, :pp_office_regist_no, :pp_office_okpo, :pp_office_inn,
         :pp_office_bank_id, :pp_office_bank_name, :pp_office_bank_account, :pp_office_bank_corr_acc,
         pp_office_bank_voen, pp_office_bank_swift;

    if (pp_office_name_full is null) then
      pp_office_name_full = pp_office_name;
    suspend;
  end
end
