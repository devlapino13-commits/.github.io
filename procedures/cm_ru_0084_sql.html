CREATE PROCEDURE CM_RU_0084_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kf_disb_date d_date;
declare variable C_ID ID_TYPE;
declare variable PR_CODE str_255;
begin
    select ma.ma_date_oper, pr_code
    from miniapp ma, product
    where ma.ma_is_active = 1
    and ma_id = :ma_id
    and ma_from_pr_id = pr_id
    and ma.ma_date_end = '31.12.9999'
    into KF_DISB_DATE, :pr_code;
    if (pr_code containing '.EMP.') then exit;  -- микрозаем сотруднику, иностранному сотруднику
    --Участники из заявки
    for select mac.mac_from_c_id
    from miniappclient mac
    where mac.mac_from_ma_id = :ma_id
    and mac.mac_amount > 0 --
    into :c_id
    do
    begin         
      select cl.c_name_fio
      from clients cl where cl.c_id = :c_id
      into :res_info_text;
    
    
      --Контакты родственники
      if (not exists (select *
          from contacts cont
          where cont.cc_from_c_id = :c_id
          and cont.cc_is_old_contact = 0
          --and cont.cc_date >= :KF_DISB_DATE-14
          )) then
      begin
        res_error_text = 'Для клиента '||:res_info_text||', не указаны актуальные контактные лица (устаревшие контактные лица не учитываются)';
        res_error_is_critical = 1;
        suspend;
      end
      else
      if (not exists (select *
          from contacts cont
          where cont.cc_from_c_id = :c_id
          and cont.cc_is_old_contact = 0
          and cont.cc_date >= :KF_DISB_DATE-14
          )) then
      begin
        res_error_text = 'Для клиента '||:res_info_text||',  дата выдачи займа превышает срок актуальности контаткных лиц (разница между датой выдачи займа и датой занесения контактного лица в  БД не должна превышать 13 дней) ';
        res_error_is_critical = 1;
        suspend;
      end
    end
end
