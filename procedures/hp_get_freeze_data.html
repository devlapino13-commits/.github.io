CREATE PROCEDURE HP_GET_FREEZE_DATA(
    K_ID D_ID)
RETURNS (
    FREEZE_TIME_H D_INTEGER,
    APPROVAL_DATETIME_MAX D_DATETIME,
    FREEZE_DATETIME_END D_DATETIME)
AS
declare variable k_disb_date d_date;
declare variable loan_sign_datetime d_datetime;
declare variable work_end_time_hours d_integer;
declare variable pr_code d_str_100;
begin
  select freeze_time_h
  from hp_get_freeze_h(:k_id)
  into freeze_time_h;

  select k_disb_date, pr.pr_code
  from kontract k
  inner join product pr on pr.pr_id = k.k_from_pr_id
  where k_id = :k_id
  into k_disb_date, pr_code;

  work_end_time_hours = iif(f_get_db_id() = 81001,
        iif(pr_code containing '.business.', 15, 20),
        23);

  approval_datetime_max = dateadd(hour, work_end_time_hours - freeze_time_h, cast(k_disb_date as timestamp));
  if (f_get_db_id() = 81001 and pr_code containing '.business.') then
    approval_datetime_max = dateadd(minute, 30, approval_datetime_max);

  select min(eds_status_datetime)
  from edoc_sign, edoc_list
  where eds_from_edl_id = edl_id
    and edl_from_edc_id = f_edoc_kind('loan.agreement', 'id')
    and edl_object_kind = 9
    and edl_object_id = :k_id
    and eds_status = 1
    and edl_status = 1
  into loan_sign_datetime;

  freeze_datetime_end = dateadd(hour, freeze_time_h, loan_sign_datetime);

  suspend;
end
