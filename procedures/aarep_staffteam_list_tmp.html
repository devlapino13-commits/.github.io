CREATE PROCEDURE AAREP_STAFFTEAM_LIST_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    RES_LO_ID DECIMAL(18,0),
    RES_REGION_ID INTEGER,
    RES_IS_WORKING INTEGER,
    RES_JOBNAME VARCHAR(100),
    RES_JOBKIND VARCHAR(100),
    RES_JOBMEMO VARCHAR(100),
    RES_LO_FIO VARCHAR(100),
    RES_REGION_NAME VARCHAR(100),
    RES_TEAM_ID INTEGER,
    RES_TEAM_NAME VARCHAR(100),
    RES_TEAM_LEADER INTEGER,
    RES_TEAM_DATE DATE,
    RES_TEAM_REG_NAME VARCHAR(255))
AS
begin
  res_team_id = 0;
  res_team_name = 'Без команды';
  res_team_leader = 0;
  res_team_date = null;
  res_lo_fio = '';
  res_region_name = '';
  res_is_working = 1;
  res_lo_id = 0;
  res_region_id = 0;
  res_jobmemo = '';
  suspend;

  for select rd_id, rd_name
      from refdata
      where rd_main_code = 'StaffTeamList' and
            rd_code <> rd_main_code and
            rd_is_active = 1
      into :res_team_id, :res_team_name
  do
  begin
    suspend;
  end

  res_is_working = 1;
  for select ps_from_c_id as res_lo_id, ps_from_reg_id as res_region_id, jl_name, rd_name
      from hr_pstatus, joblist, jobkind, refdata, region_list(:rep_region_id, 1, -1, 1)
      where ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_c_id > 1000000000004 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner', 'JobSystem.Region_KM') and
            ps_from_reg_id = id_list and
            jk_system_id = rd_id
      into :res_lo_id, res_region_id, res_jobname, res_jobkind
  do
  begin
    res_jobmemo = '';
    if (exists(select *
               from hr_voctravel, hr_orders, sa_worktype
               where hrv_from_ho_id = ho_id and
                     ho_order_type = sawt_id and
                     hrv_from_c_id = :res_lo_id and
                     hrv_date_begin <= :rep_date_end and
                     hrv_date_end_fact >= :rep_date_end and
                     hrv_status = 1 and
                     sawt_base_type = 5 and
                     sawt_include_holiday = 1 and
                     sawt_is_paid = 0 /* дикретный отпуск */
                     rows 1)) then
      res_jobmemo = 'Декретный отпуск';

    res_lo_fio = '';
    res_region_name = '';

    select c_name_fio
    from clients
    where c_id = :res_lo_id
    into :res_lo_fio;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :res_region_id
    into :res_region_name;

    select res_team_id, res_team_name, res_team_leader, res_team_date, res_team_reg_name
    from aarep_get_lo_team(:res_region_id, :res_lo_id, :rep_date_end)
    into :res_team_id, :res_team_name, :res_team_leader, :res_team_date, res_team_reg_name;

    suspend;

    if (f_get_db_id() in (39801, 41701) and
        res_jobkind = 'Кредитный менеджер') then
    begin
      for select chl_reg_id
          from compet_hr_list chl
          where chl.chl_c_id = :res_lo_id and
                :rep_date_end between chl.chl_date_begin and chl.chl_date_end and
                chl_reg_id <> :res_region_id
          into :res_region_id
      do
      begin
        select reg_code || ' ' || reg_name
        from region
        where reg_unicode = :res_region_id
        into :res_region_name;

        suspend;
      end
    end
  end
end
