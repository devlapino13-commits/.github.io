CREATE PROCEDURE AAREP_FUEL_REPORT(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    PET_ID D_INTEGER,
    PET_IS_RENT D_INTEGER,
    PET_NAME D_STR_255,
    PET_LABEL D_STR_CODE,
    PED_ID D_INTEGER,
    PED_PLACE_NAME D_STR_CODE,
    PET_BRANCH_PLACE_NAME D_STR_CODE,
    PED_CLIENT_FIO D_STR_100,
    PED_SALDO_LITER D_CURRENCY,
    PED_SALDO_SUMMA D_CURRENCY,
    PED_FILL_LITER D_CURRENCY,
    PED_FILL_LITER_TALON D_CURRENCY,
    PED_FILL_SUMMA D_CURRENCY,
    PED_WRITEOFF_LITER D_CURRENCY,
    PED_WRITEOFF_SUMMA D_CURRENCY,
    PED_WRITEOFF_KM D_CURRENCY,
    PED_SALDO_END_LITER D_CURRENCY,
    PED_SALDO_END_SUMMA D_CURRENCY,
    PED_EXPENSE_SUMMER D_CURRENCY,
    PED_EXPENSE_WINTER D_CURRENCY,
    PED_EXPENSE_SUMMER_MOUNT D_CURRENCY,
    PED_EXPENSE_WINTER_MOUNT D_CURRENCY)
AS
begin
  for
    select
      pet_id,
      pet_is_rent
    from hp_pettrans_view(1, null)
  into
    :pet_id,
    :pet_is_rent
  do
  begin
    for
      select
        pet_name,
        pet_label,
        ped_client_fio,
        sum(ped_saldo_liter),
        sum(ped_saldo_summa),
        sum(ped_fill_liter),
        sum(ped_fill_liter_talon),
        sum(ped_fill_summa),
        sum(ped_writeoff_liter),
        sum(ped_writeoff_summa),
        sum(ped_writeoff_km),
        max(ped_id),
        max(ped_expense_summer),
        max(ped_expense_winter),
        max(ped_expense_summer_mount),
        max(ped_expense_winter_mount)
      from aarep_transport_petrol_temp(:rep_region_id, :pet_id, :rep_date_begin, :rep_date_end)
      group by
        pet_name,
        pet_label,
        ped_client_fio
    into
      :pet_name,
      :pet_label,
      :ped_client_fio,
      :ped_saldo_liter,
      :ped_saldo_summa,
      :ped_fill_liter,
      :ped_fill_liter_talon,
      :ped_fill_summa,
      :ped_writeoff_liter,
      :ped_writeoff_summa,
      :ped_writeoff_km,
      :ped_id,
      :ped_expense_summer,
      :ped_expense_winter,
      :ped_expense_summer_mount,
      :ped_expense_winter_mount
    do
    begin
      select
        rg.reg_name,
        br.reg_name
      from petdata p
        inner join region rg
          on rg.reg_unicode = ped_region_place_id
        inner join region br
          on br.reg_unicode = rg.parent
      where ped_from_pet_id = :pet_id
        and ped_id = :ped_id
      into
        :ped_place_name,
        :pet_branch_place_name;

      ped_saldo_end_liter = :ped_saldo_liter + :ped_fill_liter + :ped_fill_liter_talon - :ped_writeoff_liter;
      ped_saldo_end_summa = :ped_saldo_summa + :ped_fill_summa - :ped_writeoff_summa;

      suspend;
    end
  end
end
