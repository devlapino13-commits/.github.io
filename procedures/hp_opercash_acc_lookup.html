CREATE PROCEDURE HP_OPERCASH_ACC_LOOKUP(
    OFFICE_ID D_INTEGER,
    OPER_KIND D_STR_CODE,
    OBJ_VALUE D_STR_CODE,
    IS_IPF D_BOOL)
RETURNS (
    AL_ID D_STR_20,
    AL_NAME D_STR_255)
AS
declare variable oper_id d_integer;
declare variable bg_id d_str_20;
declare variable cak_acc_no d_integer;
declare variable def_currency d_str_lang;
begin
  oper_id = f_enum_id('cashoperkind', oper_kind, 0);
  if (coalesce(oper_id, 0) = 0) then
    Exit;

  if (is_ipf is null) then
    is_ipf = 0;

  oper_kind = f_enum_code('cashoperkind', oper_kind, 0);

  select cm_default_currency
  from company
  into def_currency;

  if (oper_kind in ('minus.to_bank', 'plus.from_bank')) then
    office_id = obj_value;

  for select cak_from_bg_id, cak_acc_no
  from cashacckind
  where cak_oper_kind = :oper_id and
        cak_is_ipf = :is_ipf
  into :bg_id, :cak_acc_no
  do
  begin
    if (cak_acc_no is null) then
      cak_acc_no = 0;

    if (char_length(bg_id) = 5) then
    begin
      if (cak_acc_no <> 0) then
      begin
        for select al_id, left(al_account || '    ' || al_name, 255)
        from acc_list
        where al_from_reg_id = :office_id and
              al_from_bg_id = :bg_id and
              al_number = :cak_acc_no and
              al_currency = :def_currency
        into al_id, al_name
        do
          suspend;
      end else
      begin
        for select al_id, left(al_account || '    ' || al_name, 255)
        from acc_list
        where al_from_reg_id = :office_id and
              al_from_bg_id = :bg_id and
              al_currency = :def_currency
        into al_id, al_name
        do
          suspend;
      end
    end else
    begin
      if (cak_acc_no <> 0) then
      begin
        for select al_id, left(al_account || '    ' || al_name, 255)
        from acc_list
        where al_from_reg_id = :office_id and
              al_from_bg_id starting :bg_id and
              al_number = :cak_acc_no and
              al_currency = :def_currency
        into al_id, al_name
        do
          suspend;
      end else
      begin
        for select al_id, left(al_account || '    ' || al_name, 255)
        from acc_list
        where al_from_reg_id = :office_id and
              al_from_bg_id starting :bg_id and
              al_currency = :def_currency
        into al_id, al_name
        do
          suspend;
      end
    end
  end
end
