CREATE PROCEDURE FM_HTML_LO_BONUS_COMMON(
    UN_ID DECIMAL(18,0),
    COLOR_BLACK VARCHAR(100),
    COLOR_RED VARCHAR(100),
    COLOR_GREEN VARCHAR(100),
    REP_DATE_END DATE)
RETURNS (
    LO_ID DECIMAL(18,0),
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable LINE_STR varchar(255);
declare variable S_TEXT varchar(100);
declare variable OVS_VALUE integer;
begin
   select coalesce(res_style, '') from fm_html_styles
   into :res_text;

   res_text = res_text || '<body>';

   select un_from_c_id
   from username
   where un_id = :un_id
   into :lo_id;

   if (lo_id is null) then
    begin
      line_str = 'Пользователь не определен';
      line_str = '<font color="' || color_red || '">' || :line_str || '</font>';
      res_text = BLOB_APPEND(res_text, line_str || '<br>');
      suspend;
      exit;
    end
    line_str = 'Дата: ' || iif(rep_date_end is not null, right('0' || extract(day from rep_date_end), 2) || '.' || right('0' || extract(month from rep_date_end), 2) || '.' || extract(year from rep_date_end), 'Не указан');
    line_str = '<font color="' || color_black || '">' || :line_str || '</font>';
    res_text = BLOB_APPEND(res_text, line_str || '<br>');

    s_text = '';
    select reg_code || ' ' || reg_name
    from region, hr_pstatus
    where reg_unicode = ps_from_reg_id and
          ps_from_c_id = :lo_id and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          ps_is_working = 1 and
          ps_status = 1
    into :s_text;

    if (s_text is null or s_text = '') then
    begin
      line_str = 'У вас нет действующего приказа.';
      line_str = '<font color="' || color_red || '">' || :line_str || '</font>';
      res_text = BLOB_APPEND(res_text, line_str || '<br>');
      suspend;
      exit;
    end

    line_str = 'Офис: ' || s_text;
    line_str = '<font color="' || color_black || '">' || :line_str || '</font>';
    res_text = BLOB_APPEND(res_text, line_str || '<br>');

    select c_name_fio
    from clients
    where c_id = :lo_id
    into :s_text;

    line_str = s_text;
    line_str = '<font color="' || color_black || '">' || :line_str || '</font>';
    res_text = BLOB_APPEND(res_text, line_str || '<br>');

    ovs_value = null;

    select ovs_value
    from hp_objectprops_get('Clients', :lo_id, 'RefData', 'ObjectProps.LO_Partner', :rep_date_end)
    into :ovs_value;

    ovs_value = coalesce(ovs_value, 0);
    s_text = '';
    if (ovs_value = 0) then
      s_text = 'Без статуса';
    if (ovs_value = 1) then
      s_text = 'Ведущий';
    if (ovs_value = 2) then
      s_text = 'Партнер';

    line_str = 'Ваш статус: ' || s_text;
    line_str = '<font color="' || color_green || '">' || :line_str || '</font>';
    res_text = BLOB_APPEND(res_text, line_str || '<br><br>');

    res_text = BLOB_APPEND(res_text, '<table width="100%">
         <col width="70%">
         <col width="30%">
         <tr>' || f_format('<th>%1</th>', 'Показатели') || f_format('<th>%1</th>', 'Значения') || '</tr>');
end
