CREATE PROCEDURE AAREP_CASH_JOURNAL_SUBTMP_1(
    OL_DOC_CLIENT_NAME STR_255,
    OL_DOCUMENT_CLIENT ID_TYPE,
    OL_DOCUMENT_CLIENT2 ID_TYPE,
    CM_DATABASE_ID INTEGER,
    OL_DATE DATE,
    OL_OPER_TYPE INTEGER,
    OL_OPER_SUBTYPE INTEGER,
    OL_ID ID_TYPE,
    OL_KOL_PLUS_INPUT INTEGER,
    HAS_LINE_INPUT INTEGER,
    OL_DOC_NO_STR_INPUT VARCHAR(50),
    REG_CODE VARCHAR(20),
    OL_DOC_NO INTEGER,
    OL_ACCOUNT STR_255,
    OL_SUMMA_PLUS_INPUT CURR_TYPE)
RETURNS (
    OL_CLIENT_NAME STR_255,
    OL_KOL_PLUS INTEGER,
    HAS_LINE INTEGER,
    OL_DOC_NO_STR VARCHAR(50),
    OL_ACCOUNT_2 STR_255,
    OL_SUMMA_PLUS CURR_TYPE,
    C_IS_RESIDENT INTEGER,
    KC_FIO STR_100)
AS
declare variable ol_id_other id_type;
declare variable bl_summa curr_type = 0;
declare variable bl_summa_total curr_type = 0;
declare variable max_line integer;
declare variable kk_id id_type;
declare variable kc_number integer;
declare variable k_amount curr_type = 0;
declare variable kc_amount curr_type = 0;
declare variable k_kommis_perc perc_type;
declare variable kc_kommis_lgota curr_type;
declare variable sub_no integer;
declare variable oek_rd_id integer;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  ol_kol_plus = ol_kol_plus_input;
  has_line = has_line_input;
  ol_doc_no_str = ol_doc_no_str_input;
  ol_summa_plus = ol_summa_plus_input;

  if (ol_doc_client_name is null) then
    ol_doc_client_name = '';
  if (ol_document_client is not null) then
  begin
    select c_name_fio
    from clients cl
    where c_id = :ol_document_client
    into :ol_client_name;
  end
  else
  begin
    if (ol_document_client2 is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :ol_document_client2
      into :ol_client_name;
    end
    else
      ol_client_name = :ol_doc_client_name;
  end
  ol_id_other = 0;

  if ((cm_database_id = 81001) and
      (ol_date >= '10.02.2014') and
      (ol_oper_type = 2) and
      (ol_oper_subtype = 20)) then
  begin
    select ol_id
    from operlist
    where ol_parent_id = :ol_id and
          is_delete = 0 and
          ol_oper_type = 1 and
          ol_oper_subtype = 2
    into :ol_id_other;
    if (ol_id_other is null) then
      ol_id_other = 0;
  end

  if (not(((cm_database_id = 81001 and
      ol_oper_type = 1 and
      ol_oper_subtype <= 2 and
      ol_date >= '01.01.2014')) or (ol_id_other > 0))) then
  begin
    ol_kol_plus = ol_kol_plus + 1;
    has_line = 1;
    ol_doc_no_str = reg_code || '-' || right('000000' || ol_doc_no, 6);

    ol_account_2 = ol_account;

    if ((cm_database_id = 81001) and
        (ol_date >= '01.01.2018')) then
    begin
      ol_account_2 = (case ol_account_2
                        when '50-01' then '20202'
                        when '57-01' then '20209'
                        when '76-02' then '60323'
                        when '76-06' then '47422'
                        when '91-01' then '71801'
                        when '51-01' then '20501'
                        when '70-01' then '60305'
                        when '73-02' then '60308'
                        when '94-01' then '60308'
                        else ol_account_2
                      end);

    end

    suspend;
  end
  else -- Кассовая книга для РФ
  begin
    if (not(ol_oper_type = 1 and
        ol_oper_subtype = 1)) then -- Погашения
    begin
      ol_doc_no_str = '';
      bl_summa = ol_summa_plus;
      bl_summa_total = 0;
      sub_no = 0;
      max_line = 0;
      kk_id = null;

      select distinct ok_from_k_id
      from operkred
      where ((ok_from_ol_id = :ol_id and
            :ol_id_other = 0) or
            (ok_from_ol_id = :ol_id_other)) and
            ok_ol_is_active = 1
      union
      select distinct ox_object_id
      from operextra
      where ox_from_rd_id = :oek_rd_id and
            ((ox_from_ol_id = :ol_id and
            :ol_id_other = 0) or
            (ox_from_ol_id = :ol_id_other)) and
            ox_object_kind = 9 and
            ox_ol_is_active = 1
      union
      select distinct okf.oko_from_k_id
      from operkroff okf
      where okf.oko_from_ol_id = :ol_id and
            okf.oko_ol_is_active = 1 rows 1
      into :kk_id;

      select count(distinct kc_number)
      from knclient
      where kc_from_k_id = :kk_id and
            kc_amount_approved > 0
      into :max_line;
      if (max_line is null) then
        max_line = 0;

      for select distinct kc_number, k_amount_approved, kc_amount_approved, iif(kncl.kc_non_resident = 1, 0, 1),
                          cl.c_name_fio
          from knclient kncl, kontract, clients cl
          where kc_from_k_id = :kk_id and
                kc_amount_approved > 0 and
                k_id = kc_from_k_id and
                cl.c_id = kncl.kc_from_c_id
          order by kc_number
          into :kc_number, :k_amount, :kc_amount, :c_is_resident, :kc_fio
      do
      begin
        sub_no = sub_no + 1;
        ol_summa_plus = cast(bl_summa as double precision) / k_amount;
        ol_summa_plus = round(ol_summa_plus * kc_amount);

        if (max_line = sub_no) then
          ol_summa_plus = bl_summa - bl_summa_total;
        else
        begin
          if ((bl_summa_total + ol_summa_plus) > bl_summa) then
            ol_summa_plus = ol_summa_plus - ((bl_summa_total + ol_summa_plus) - bl_summa);

          bl_summa_total = bl_summa_total + ol_summa_plus;
        end

        ol_doc_no_str = reg_code || '-' || right('000000' || ol_doc_no, 6) || '/' || sub_no;

        ol_kol_plus = ol_kol_plus + 1;
        has_line = 1;

        ol_account_2 = ol_account;

        if ((cm_database_id = 81001) and
            (ol_date >= '01.01.2018')) then
        begin
          ol_account_2 = left(ol_account_2, 5);
          ol_account_2 = (case ol_account_2
                            when '50-01' then '20202'
                            when '57-01' then '20209'
                            when '76-02' then '60323'
                            when '76-06' then '47422'
                            when '91-01' then '71801'
                            when '51-01' then '20501'
                            when '70-01' then '60305'
                            when '73-02' then '60308'
                            when '94-01' then '60308'
                            else ol_account_2
                          end);
          if (c_is_resident = 1) then
            ol_account_2 = (case ol_account_2
                              when '58-03' then '48801'
                              when '76-09' then '48809'
                              when '98-01' then '48809'
                              else ol_account_2
                            end);
          else
            ol_account_2 = (case ol_account_2
                              when '58-03' then '49201'
                              when '76-09' then '49209'
                              when '98-01' then '49209'
                              else ol_account_2
                            end);
        end
        suspend;
      end
    end
    else -- Доп.процент
    begin
      ol_doc_no_str = '';
      bl_summa = ol_summa_plus;
      bl_summa_total = 0;
      sub_no = 0;
      max_line = 0;

      kk_id = null;

      select distinct ok_from_k_id
      from operkred
      where ((ok_from_ol_id = :ol_id and
            :ol_id_other = 0) or
            (ok_from_ol_id = :ol_id_other)) and
            ok_ol_is_active = 1
      union
      select distinct ox_object_id
      from operextra
      where ox_from_rd_id = :oek_rd_id and
            ((ox_from_ol_id = :ol_id and
            :ol_id_other = 0) or
            (ox_from_ol_id = :ol_id_other)) and
            ox_object_kind = 9 and
            ox_ol_is_active = 1
      union
      select distinct okf.oko_from_k_id
      from operkroff okf
      where okf.oko_from_ol_id = :ol_id and
            okf.oko_ol_is_active = 1 rows 1
      into :kk_id;

      select count(distinct kc_number)
      from knclient
      where kc_from_k_id = :kk_id and
            kc_amount_approved > 0
      into :max_line;
      if (max_line is null) then
        max_line = 0;

      for select distinct kc_number, k_amount_approved, kc_amount_approved, k_kommis_perc, kc_kommis_lgota,
                          iif(kncl.kc_non_resident = 1, 0, 1), cl.c_name_fio
          from knclient kncl, kontract, clients cl
          where kc_from_k_id = :kk_id and
                kc_amount_approved > 0 and
                k_id = kc_from_k_id and
                cl.c_id = kncl.kc_from_c_id
          order by kc_number
          into :kc_number, :k_amount, :kc_amount, :k_kommis_perc, :kc_kommis_lgota, :c_is_resident, :kc_fio
      do
      begin
        sub_no = sub_no + 1;
        ol_summa_plus = round(kc_amount * k_kommis_perc / 100 - kc_kommis_lgota);

        if (max_line = sub_no) then
          ol_summa_plus = bl_summa - bl_summa_total;
        else
          bl_summa_total = bl_summa_total + ol_summa_plus;

        ol_doc_no_str = reg_code || '-' || right('000000' || ol_doc_no, 6) || '/' || sub_no;

        ol_kol_plus = ol_kol_plus + 1;
        has_line = 1;

        ol_account_2 = ol_account;

        if ((cm_database_id = 81001) and
            (ol_date >= '01.01.2018')) then
        begin
          ol_account_2 = left(ol_account_2, 5);
          ol_account_2 = (case ol_account_2
                            when '50-01' then '20202'
                            when '57-01' then '20209'
                            when '76-02' then '60323'
                            when '76-06' then '47422'
                            when '91-01' then '71801'
                            when '51-01' then '20501'
                            when '70-01' then '60305'
                            when '73-02' then '60308'
                            when '94-01' then '60308'
                            else ol_account_2
                          end);

          if (c_is_resident = 1) then
          begin
            ol_account_2 = (case ol_account_2
                              when '58-03' then '48801'
                              when '76-09' then '48809'
                              when '98-01' then '48809'
                              else ol_account_2
                            end);
          end
          else
          begin
            ol_account_2 = (case ol_account_2
                              when '58-03' then '49201'
                              when '76-09' then '49209'
                              when '98-01' then '49209'
                              else ol_account_2
                            end);
          end
        end

        suspend;
      end
    end
  end
end
