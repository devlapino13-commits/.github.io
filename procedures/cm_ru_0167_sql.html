CREATE PROCEDURE CM_RU_0167_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable c_id d_id;
declare variable cs_client_circle d_integer;
declare variable kc_id d_id;
declare variable history_not_found d_smallint;
declare variable res_active_count  d_integer;
declare variable res_delinq_count d_integer;
declare variable c_curr_status d_smallint;
declare variable max_amount d_integer;
declare variable res_closed_last_week d_integer;
declare variable memo d_str_255;
declare variable closed_qty d_integer;
declare variable has_doc d_smallint;
declare variable pr_code varchar (100);
begin
--if(k_id = 101010061330300101) then exit;
  if(k_id = 101010038657800701)then exit; 
  for select kncl.kc_from_c_id,
             k.k_disb_date,
             cl.c_name_fio,
             cs.cs_client_circle,
             kncl.kc_amount_approved,
             kncl.kc_id
  from kontract k
  inner join product pr on pr.pr_id = k.k_from_pr_id
  inner join knclient kncl on kncl.kc_from_k_id = k.k_id
  inner join clients cl on cl.c_id = kncl.kc_from_c_id
  left join clientstate cs on cs.cs_from_c_id = cl.c_id
  where k_id = :k_id
  and pr.pr_code like '%.float_percent.%'
  and kncl.kc_amount_approved > 0
  into c_id, kf_disb_date, res_info_text, cs_client_circle, kc_amount, kc_id
  do
  begin
    cs_client_circle = coalesce(cs_client_circle, 0);
   if (exists (select *
            from kontstate ks
            inner join kontract kont on kont.k_id = ks.ks_from_k_id
            inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
            inner join product pr on pr.pr_id = kont.k_from_pr_id
            where kncl.kc_from_c_id = :c_id
            and kncl.kc_amount_approved > 0
            and pr.pr_code like '%.float_percent.%'
            and ks.ks_ol_is_delete = 0
            and ks.ks_state = 1
            and ks.ks_date_begin >= '01.11.2020'
            and ks.ks_from_k_id <> :k_id
    )) then
    begin
        res_error_is_critical=1;
        res_error_text='Клиент в ноябре 2020 года уже получал заем по продукту "Выгодная сторона"';
        suspend;
        exit;
    end
    if (exists (select *
            from knclient kncl
            inner join kontract kon on kon.k_id = kncl.kc_from_k_id
            inner join product pr on pr.pr_id = kon.k_from_pr_id
            inner join kontstate ks on ks.ks_from_k_id = kon.k_id
            where kncl.kc_amount_approved > 0
            and kc_from_c_id = :c_id
            and k_id <> :k_id
            and pr.pr_code like '%.float_percent.%'
            and ks_ol_is_delete = 0
            and ks.ks_state = 1
            and :kf_disb_date between ks_date_begin and ks_date_end
    )) then
    begin
        res_error_is_critical=1;
        res_error_text='У клиента есть активный заем по продукту "выгодная сторона"';
        suspend;
        exit;
    end
    if (exists (
            select *
            from knclient, oksummary oks
            where kc_from_c_id = :c_id and
                kc_amount_approved > 0 and
                kc_from_k_id = oks_from_k_id and
                :kf_disb_date between oks_date_begin and oks_date_end and
                oks.oks_from_k_id <> :k_id and
                oks.oks_close_state = 0 and
                :kf_disb_date - oks.oks_delinq_date > 0)) then
    begin
        res_error_is_critical=1;
        res_error_text='У клиента есть активный просроченный заем в "М Булак"';
        suspend;
        exit;
    end
    max_amount = 25000;
    memo = '';
    if (kc_amount > 25000) then
    begin
        select c_curr_status
        from hp_get_client_status(:c_id, :kf_disb_date)
        into c_curr_status;
        if (c_curr_status <= 3) then
        begin
            if (c_curr_status = 3) then
            begin
                memo = 'У клиента 3 звезды.';
                max_amount = 60000;
            end
            history_not_found = 1;
            res_active_count = 0;
            res_delinq_count = 0;
            res_closed_last_week = 0;
    
            select history_not_found,
                   res_active_count,
                   res_delinq_count,
                   res_closed_last_week,
                   res_loan_count - res_active_count
            from hp_extinfolist_pdn_mfo_info(:kc_id)
            into :history_not_found,
                 :res_active_count,
                 :res_delinq_count,
                 :res_closed_last_week,
                 :closed_qty;
    
            history_not_found = coalesce(history_not_found, 1);
            res_active_count = coalesce(res_active_count, 0);
            res_delinq_count = coalesce(res_delinq_count, 0);
            res_closed_last_week = coalesce(res_closed_last_week, 0);
            if (history_not_found = 1) then
            begin
                res_error_is_critical = 1;
                res_error_text='Кредитная история еще не получена';
                suspend;
                exit;        
            end
            has_doc = 0;
            if (exists(
                    select *
                    from clientdocs
                    where cds_from_c_id = :c_id
                      and cds_from_pk_id = 337
                      and cds_is_delete = 0
                      and (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0))) then
            begin
                has_doc = 1;
            end
            if ((has_doc = 1) and (cs_client_circle =0)) then
            begin
                max_amount = 90000;
                memo = 'Новый клиент с документом "Документы о займе в другом МФО"';
            end
            else  if ((res_active_count - res_delinq_count > 0) or (has_doc = 1)) then   -- если есть активные непросрчоенные займы
            begin
                max_amount = 70000;
                memo = 'У клиента есть активный непросроченный заем в другом МФО.';

                if ((closed_qty >= 5) and (cs_client_circle = 0)) then -- т.к. у клиента есть активный непросроченный заем и это новый клиент и 5 закрытых займов, тогда может брать как 5-и звездочный клиент, т.е. сумма  = 90 000
                begin
                    max_amount = 90000;
                    memo = 'У клиента 5 закрытых займов + это новый клиент + есть актиный заем';
                end
            end
            else if (res_closed_last_week > 0) then
            begin
               max_amount = 70000;
               memo = 'Клиент закрыл заем в другом МФО, в рамках 7 дней.';
            end
        end
        else if (c_curr_status >= 4) then
        begin
            memo = 'У клиента звезд - '||c_curr_status||'.';
            max_amount = 90000;
        end
        if (kc_amount > max_amount) then
        begin
            res_error_is_critical=1;
            res_error_text= memo || 'Максимально допустимая сумма займа = '||max_amount;
            suspend;
            exit;
        end
    end
  end
end
