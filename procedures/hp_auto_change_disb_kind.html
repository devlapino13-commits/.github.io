CREATE PROCEDURE HP_AUTO_CHANGE_DISB_KIND(
    UN_ID D_ID,
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID)
RETURNS (
    DISB_KIND D_SMALLINT)
AS
declare variable res_id d_smallint;
declare variable pr_code d_str_code;
declare variable new_disb_kind d_smallint;
begin
    -- RFCRU-5314
    -- Процедура изменяет тип выдачи автоматом

    if (:obj_kind = 'miniapp') then
        select pr.pr_code, ma.ma_disb_kind
        from miniapp ma
        inner join product pr on ma.ma_from_pr_id = pr.pr_id
        where ma.ma_id = :obj_id
            and ma.ma_is_active = 1
            and ma.ma_date_end = '31.12.9999'
        into :pr_code, :disb_kind;

    if (:obj_kind = 'kontract') then
        select pr.pr_code, k.k_disb_kind
        from kontract k
        inner join product pr on k.k_from_pr_id = pr.pr_id
        where k.k_id = :obj_id
        into :pr_code, :disb_kind;


    -- Если продукт "Бизнес", но способ выдачи не "На расчетный счет", тогда устанавливаем его принудительно
    if (:pr_code containing '.business.' and :disb_kind <> 8) then
      new_disb_kind = 8;

    -- Если продукт не "Бизнес" и способ выдаячи "На расчетный счет, тогда принудительно устанавливаем как "СБП по номеру телефона"
    if (:pr_code not containing '.business.' and :disb_kind = 8) then
      new_disb_kind = 7;


    -- Обновляем способ выдачи
    -- Уточнииь по поводу поля is_online
    if (:obj_kind = 'miniapp' and :new_disb_kind is not null) then
        update miniapp ma
        set ma.ma_disb_kind = :new_disb_kind
        where ma.ma_id = :obj_id;

    if (:obj_kind = 'kontract' and :new_disb_kind is not null) then
        update kontract k
        set k.k_disb_kind = :new_disb_kind
        where k.k_id = :obj_id;


    disb_kind = coalesce(:new_disb_kind, :disb_kind);
    suspend;
end
