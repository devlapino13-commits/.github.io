CREATE PROCEDURE CM_KG_0217_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable CDS_DOC_DATE D_DATE;
begin
  if (ma_id = 3903485) then
    exit;
  for select mac.mac_from_c_id
      from miniapp ma
      inner join miniappclient mac on ma.ma_id = mac.mac_from_ma_id
      where ma.ma_is_active = 1 and
            ma.ma_date_end = '31.12.9999' and
            ma.ma_id = :ma_id
      into :c_id
  do
  begin
    cds_doc_date = null;
    select cds.cds_doc_date_begin
    from clientdocs cds
    inner join passkind pk on cds.cds_from_pk_id = pk.pk_id
    inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
    where cds.cds_from_c_id = :c_id and
          pdt.pdt_code = 'contact_eds' and
          (cds.cds_doc_blob_1 is not null or
          cds.cds_doc_blob_2 is not null)
    order by cds.cds_datetime desc
    rows 1
    into :cds_doc_date;

    if (cds_doc_date <= cast('01.10.2023' as date)) then
    begin
      res_error_text = 'Распечатайте новый Договор ЭЦП';
      res_error_is_critical = 1;
      suspend;
      exit;
    end
  end
end
