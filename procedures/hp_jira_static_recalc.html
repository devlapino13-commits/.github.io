CREATE PROCEDURE HP_JIRA_STATIC_RECALC
AS
declare variable dev_id d_id;
declare variable period_id d_id;
declare variable plan_finish d_date;
declare variable dev_waiting d_integer;
declare variable dev_working d_integer;
declare variable dev_reworking d_integer;
begin
  delete from jira_period
  where 1 = 1;

  insert into jira_period (jrp_dev_id, jrp_period, jrp_level, jrp_tasks_closed, jrp_work_seconds,
                           jrp_estimate_seconds)
  select jri_dev_id, extract(year from jri_finished) * 100 + extract(month from jri_finished), jri_level,
         count(*), sum(jri_worked_secs), sum(jri_estimate_secs)
  from jira_issues, jira_states
  where jri_dev_id <> 0 and
        jri_state_id = jrs_id and
        jrs_result >= 0 and
        jri_finished >= cast('01.01.2024' as timestamp)
  group by 1, 2, 3;

  period_id = ((extract(year from current_timestamp) * 100 +
                extract(month from current_timestamp)) * 100 +
                extract(day from current_timestamp)) * 10000;

  for select jrd_id
  from jira_dev
  where jrd_id = 0 or
        jrd_is_dev = 1
  into dev_id
  do
  begin
    select dev_estimate_busy, dev_waiting, dev_working, dev_reworking
    from aarep_jira_dev_stats(:dev_id)
    into :plan_finish, dev_waiting, dev_working, dev_reworking;

    update or insert into jira_estimate (jre_id, jre_dev_id, jre_date, jre_plan_finish, jre_waiting, jre_working, jre_reworking)
    values (:period_id + :dev_id, :dev_id, current_timestamp, :plan_finish, :dev_waiting, :dev_working, :dev_reworking)
    matching (jre_id);
  end
end
