CREATE PROCEDURE UXP$UXEVENT_KINDS(
    EVENT_TYPE D_STR_10)
RETURNS (
    EVENT_ID D_INTEGER,
    EVENT_CODE D_STR_CODE,
    EVENT_NAME D_STR_100,
    EVENT_MEMO D_STR_255,
    EVENT_KIND D_SMALLINT)
AS
begin
  -- Список доступных событий и обработчиков
  -- При добавлении новой строки, в модуле TUxEventMgr неоходимо реализовать:

  -- Для нового События:
  --   Расширить тип TUxEventKind = (uekOnChange = 1...
  --   Создать новую процедуру типа procedure TUxEventMgr.OnFieldChange(Sender: TField);
  --       ExecHandlers(Sender, uekOnChange);
  --   Дописать процедуру AddEventProc добавив новое название процедуру события
  --       procedure TUxEventMgr.AddEventProc(aComponent: TComponent; aKind: TUxEventKind);


  -- Для нового Обработчика:
  --   Расширить тип TUxHandlerKind = (uhkDoChange = 1, ...
  --   В процедура TUxHandler.Execute добавить строку типа:
  --       if Kind = uhkDoChange then
  --   Создать новую процедуру-обработчик типа:
  --       procedure TUxHandler.DoChange(aExprValue: Variant);


  -- В данной процедуры возвращаемое поле event_kind:
  --   0 - Событие (OnChange и т.д.)
  --   1 - Обработчик, заполнение выражения не образательно
  --   2 - Обработчик с обязательным выражением
  --   3 - Обработчик с обязательным логическим выражением

  if (coalesce(event_type, '') = '') then
    exception e_default 'Empty event_type param';

  if (event_type not in ('On', 'Do')) then
    exception e_default 'Unknown event_type param, needed "On" or "Do"';

  if (event_type = 'On') then
  begin
    event_kind = 0; -- Для всех событий

    event_id = 1;
    event_code = 'OnChange';
    event_name = 'Значение изменилось';
    event_memo = 'Событие срабатывает при изменении значения поля датасета или редактора';
    suspend;

    event_id = 2;
    event_code = 'OnAfterScroll';
    event_name = 'Запись изменилась';
    event_memo = 'Событие срабатывает после перехода на другую запись и после открытия источника';
    suspend;

    event_id = 3;
    event_code = 'OnBeforePost';
    event_name = 'Применение изменений';
    event_memo = 'Событые срабатывает перед применением изменений';
    suspend;

    event_id = 4;
    event_code = 'OnAfterOpen';
    event_name = 'Таблица открылась';
    event_memo = 'Событые срабатывает после открытия таблицы';
    suspend;
  end

  if (event_type = 'Do') then
  begin
    event_id = 1;
    event_code = 'DoChange';
    event_name = 'Установить значение';
    event_memo = 'Устанавливает значение указанному полю или элементу';
    event_kind = 1;
    suspend;

    event_id = 2;
    event_code = 'SetVisible';
    event_name = 'Видимость';
    event_memo = 'Отобразить или скрыть элемент';
    event_kind = 3;
    suspend;

    event_id = 3;
    event_code = 'SetEnabled';
    event_name = 'Доступность';
    event_memo = 'Активировать или деактивировать эелмент';
    event_kind = 3;
    suspend;

    event_id = 4;
    event_code = 'DoRefresh';
    event_name = 'Обновить';
    event_memo = 'Обновить данные, переоткрыть источник';
    event_kind = 1;
    suspend;

    event_id = 5;
    event_code = 'DoCheck';
    event_name = 'Проверка';
    event_memo = 'Проверить выполнение условий';
    event_kind = 3;
    suspend;

    event_id = 6;
    event_code = 'DoShowMessage';
    event_name = 'Показ сообщения';
    event_memo = 'Показать текст в диалоговом окне';
    event_kind = 1;
    suspend;

    event_id = 7;
    event_code = 'SetHilighted';
    event_name = 'Выделение цветом';
    event_memo = 'Выделить ярким цветом';
    event_kind = 3;
    suspend;

    event_id = 8;
    event_code = 'SetColor';
    event_name = 'Установить цвет';
    event_memo = 'Вычисление значения для цвета редактора';
    event_kind = 3;
    suspend;
  end

end
