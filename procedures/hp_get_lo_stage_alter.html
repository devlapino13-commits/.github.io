CREATE PROCEDURE HP_GET_LO_STAGE_ALTER(
    LO_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    LO_STAGE_DATE DATE,
    LO_STAGE_MONTH INTEGER,
    HRV_DATE_END_FACT DATE,
    LO_STAGE_MONTH_H DECIMAL(4,2))
AS
declare variable reg_id integer;
declare variable day_razn integer;
declare variable jk_system_code d_str_100;
declare variable lo_stage_date_tmp d_date;
declare variable ps_is_working d_smallint;
begin
  lo_stage_date = null;
  lo_stage_month = 0;
  lo_stage_month_h = -1; -- если -1 то на СВЗ нету приказа

  if (not exists(select *
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :lo_id and
                       ps_status = 1 and
                       ps_is_working = 1 and
                       :rep_date between ps_date_begin and ps_date_end and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code = 'JobSystem.LO_Standart')) then
  begin
    suspend;
    exit;
  end

  for select ps_date_begin, ps_from_reg_id, jk_system_code, ps_is_working
      from hr_pstatus, joblist, jobkind
      where ps_status = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            ps_from_c_id = :lo_id and
            ps_date_begin <= ps_date_end and
            ps_date_begin <= :rep_date
      order by ps_date_begin
      into lo_stage_date_tmp, reg_id, jk_system_code, ps_is_working
  do
  begin
    if (jk_system_code = 'JobSystem.LO_Standart' and
        ps_is_working = 1) then
    begin
      if (lo_stage_date is null and
          not exists(select *
                     from oksummary
                     where oks_kc_id = :lo_id and
                           :lo_stage_date_tmp between oks_date_begin and oks_date_end and
                           oks_close_state = 0 and
                           oks_region_id = :reg_id and
                           oks_balance > 0)) then
        lo_stage_date = lo_stage_date_tmp;
    end
    else
    begin
      lo_stage_date = null;
    end
  end

  if (lo_stage_date is null) then
  begin
    select ps_date_begin, ps_from_reg_id
    from hr_pstatus, joblist, jobkind
    where ps_status = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code = 'JobSystem.LO_Standart' and
          ps_from_c_id = :lo_id and
          ps_is_working = 1 and
          ps_date_begin <= ps_date_end and
          ps_date_begin <= :rep_date rows 1
    into lo_stage_date, reg_id;
  end

  --  проверка из декретного отпуска когда выходила
  hrv_date_end_fact = null;
  select hrv.hrv_date_end_fact
  from hr_voctravel hrv, hr_orders ho, sa_worktype
  where hrv_from_ho_id = ho_id and
        ho_order_type = sawt_id and
        hrv_from_c_id = :lo_id and
        hrv_status = 1 and
        sawt_id in (16, 17, 18, 20, 21) and
        hrv_date_begin <= :rep_date /*and
        hrv_date_end_fact <= :rep_date */
  order by ho.ho_save_time desc
  rows 1
  into hrv_date_end_fact;

  if (hrv_date_end_fact > rep_date) then
    hrv_date_end_fact = null;

  if (hrv_date_end_fact is not null) then
  begin
    if (not exists(select *
                   from oksummary
                   where oks_kc_id = :lo_id and
                         :hrv_date_end_fact - 1 between oks_date_begin and oks_date_end -- портфель проверять на предыдущий день
                         --              and oks_region_id = :reg_id
                         and
                         oks_close_state = 0 and
                         oks_balance > 0) and
        hrv_date_end_fact > lo_stage_date and
        not exists(select sawt_name
                   from hr_voctravel, hr_orders, sa_worktype
                   where hrv_from_ho_id = ho_id and
                         ho_order_type = sawt_id and
                         hrv_from_c_id = :lo_id and
                         hrv_status = 1 and
                         hrv_date_begin <= :rep_date and
                         hrv_date_end_fact >= :rep_date and
                         sawt_id in (16, 17, 18, 21)
                   order by hrv_date_begin desc
                   rows 1)) then
      lo_stage_date = hrv_date_end_fact;
  end

  -- RFCRU-5099
  if (:lo_id = 1010100018842) then
    lo_stage_date = '15.11.2025';

  if (lo_stage_date is not null) then
  begin
    lo_stage_month = datediff(month, lo_stage_date, rep_date); -- ранее вместо res_date_end было res_date_begin, исправлено по звонку Ироды от 31.01.2018

    if (rep_date <> '31.12.9999' and
        datediff(month, rep_date, rep_date + 1) = 0) then -- если дата отчета не конец месяца то выполняй ниже следующее
      if (extract(day from lo_stage_date) > extract(day from rep_date) and
          lo_stage_month > 0) then
        lo_stage_month = lo_stage_month - 1;

    day_razn = :rep_date - dateadd(month, lo_stage_month, lo_stage_date);
    lo_stage_month_h = lo_stage_month + cast(day_razn as double precision) / 30;
  end

  suspend;
end
