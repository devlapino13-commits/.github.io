CREATE PROCEDURE HP_KKT_SET_STATUS(
    KKM_ID D_ID,
    UUID_STR D_STR_CODE,
    STATUS D_SMALLINT,
    ERROR_TEXT D_STR_LONG)
AS
declare variable err_count d_smallint;
begin
  if (coalesce(status, 999) not in (-2, -1, 1)) then
    exception e_default 'Wrong status: ' || coalesce(status, '<NULL>');

  -- Если статус -1 - "ошибка, передать еще раз", если ошибок > 10, то ставим статус -2
  if (status = -1) then
  begin
    select count(kke_id)
    from kkt_error
    where kke_from_kkm_id = :kkm_id
    into err_count;

    if (err_count > 3) then
      status = -2;
  end
   
  update kkt_main
  set kkm_status = :status,
      kkm_uuid = coalesce(char_to_uuid(:uuid_str), kkm_uuid)
  where kkm_id = :kkm_id;

  if (status < 0) then
    if (coalesce(error_text, '') = '') then
      exception e_default 'Error_text is empty';
    else
      insert into kkt_error (kke_from_kkm_id, kke_error_id)
      values (:kkm_id, f_memo_id(:error_text));
end
