declare procedure get_debt_from_json(jso_in d_integer)
  returns(
    pos d_integer, 
    debt_sign d_bool,
    calc_date d_date,
    debt_date d_date,
    first_sum d_currency,
    debt_op_sum d_currency,
    debt_sum d_currency
  )
  as
    declare variable cnt d_integer;
    declare variable calc_date_str d_str_255;
    declare variable debt_date_str d_str_255;
    declare variable jso d_integer;
  begin
    if (jso_in is null) then
      exit;

    cnt = json.count_get(jso_in);

    pos = 0;
    if (cnt >1) then
      exception e_default json.arr_to_json(jso_in);
    while (pos < cnt) do
    begin
      jso = json.arr_get_obj(jso_in, pos);
      debt_sign = json.obj_get_val(jso, 'sign', 1);
      debt_op_sum  = replace(replace(json.obj_get_val(jso, 'opSum', 0), ',', '.'), '-', 0);
      debt_sum  = replace(replace(json.obj_get_val(jso, 'sum', 0), ',', '.'), '-', 0);
      first_sum  = replace(replace(json.obj_get_val(jso, 'firstSum', 0), ',', '.'), '-', 0);
      calc_date_str = json.obj_get_val(jso, 'calcDate', null);
      debt_date_str = json.obj_get_val(jso, 'date',     null);
      calc_date = iif(calc_date_str = '-', null, calc_date_str);
      debt_date = iif(debt_date_str = '-', null, debt_date_str);

      if (debt_sum = 0) then
        debt_date = null;

      suspend;
      pos = pos + 1;
    end 
  end

declare procedure get_payment_from_json(jso_in d_integer)
  returns(
    lastpayoutdate d_date
  )
  as
    declare variable cnt d_integer;
    declare variable lastpayoutdate_str d_str_255;
    declare variable lastpayoutdate_tmp d_date;
    declare variable pos d_integer;
    declare variable jso d_integer;
  begin
    if (jso_in is null) then  exit;
    cnt = json.count_get(jso_in);
    pos = 0;
    while (pos < cnt) do
    begin

      jso = json.arr_get_obj(jso_in, pos);
      lastpayoutdate_str = json.obj_get_val(jso, 'lastpayoutdate', null);
      lastpayoutdate_tmp = iif(lastpayoutdate_str = '-', null, lastpayoutdate_str);

      if (coalesce(lastpayoutdate_tmp, '01.01.1900') > coalesce(lastpayoutdate, '01.01.1900')) then
        lastpayoutdate = lastpayoutdate_tmp;
      pos = pos + 1;
    end
    suspend;
  end

declare procedure get_contract_amount(jso_in d_integer)
  returns(
    summ d_currency,
    currency d_str_code
  )
  as
    declare variable cnt d_integer;
    declare variable pos d_integer;
    declare variable jso d_integer;
  begin
    if (jso_in is null) then exit;
    cnt = json.count_get(jso_in);
    pos = 0;
    summ = 0;
    while (pos < cnt) do
    begin
      jso = json.arr_get_obj(jso_in, pos);
      pos = pos + 1;
      summ = summ + cast(replace(replace(json.obj_get_val(jso, 'sum', 0), ',', '.'), '-', 0) as d_currency);
      currency = json.obj_get_val(jso, 'currency', null);
    end

    suspend;
  end

returns d_date

declare function my_strtonum (
    num_str d_str_code)
returns d_double
as
begin
  return replace(num_str, ',', '.');
  when any do
    return 0;
end

declare function f_my_obj_get_arr(handle d_integer, path d_str_long)
returns d_integer
as
  declare variable jsa d_integer;
  declare variable val_type d_str_code;
begin
  if (json.contains(handle, path) = 0) then
    return null;

  val_type = json.val_get_type_name(handle, path);

  if (val_type = 'Array') then
    return json.obj_get_arr(handle, path);
  else if (val_type = 'Object') then
  begin
    handle = json.obj_get_obj(handle, path);
    jsa = json.arr_create();
    json.arr_set_obj(jsa, -1, handle);

    return jsa;
  end
  else
    exception e_default path || ' is not Array/Object';
end
