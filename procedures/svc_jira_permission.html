CREATE PROCEDURE SVC_JIRA_PERMISSION(
    LOGIN D_STR_CODE)
RETURNS (
    RESULT D_STR_255)
AS
declare variable SVC_ROLE D_STR_CODE;
declare variable SVC_COUNTRY D_STR_100;
begin
  svc_country = '["KG", "RU", "AZ", "KZ"]';
  svc_role = 'developer';

  if (login in ('igusev', 'hraimjanov')) then svc_role = 'admin';

  if (login = 'rhakimov') then svc_role = 'ceo';

  if (login in ('lbelousova', 'tvorobieva')) then svc_role = 'board_member';

  result = '
    { "role": "'||:svc_role||'",
       "countries": '||:svc_country||'
    }';

  suspend;
end
