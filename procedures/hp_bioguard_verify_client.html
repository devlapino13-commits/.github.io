CREATE PROCEDURE HP_BIOGUARD_VERIFY_CLIENT(
    C_ID D_ID NOT NULL,
    SESSION_ID D_STR_100 NOT NULL,
    DOC_NUMBER D_STR_100 NOT NULL,
    DOC_ISSUE_DATE D_DATE NOT NULL,
    DOC_END_DATE D_DATE NOT NULL,
    DOC_ISSUE_ORG_NAME D_STR_255 NOT NULL,
    CLIENT_PHOTO_ID D_ID NOT NULL,
    PHOTO_1_BINARY D_BLOB_BINARY NOT NULL,
    PHOTO_2_BINARY D_BLOB_BINARY NOT NULL)
AS
declare variable PHOTO_1_BLS_ID D_ID;
declare variable PHOTO_2_BLS_ID D_ID;
declare variable KZ_PASSPORT_CARD_PK_ID D_INTEGER;
begin
    -- Сохраяем полученные фотографии паспорта в BLOBSTORE
    execute procedure hp_insert_into_blobstore('passport_photo_1_'||:session_id||'.jpg', 'clients', :photo_1_binary, 1000000000000, 0, 2)
    returning_values :photo_1_bls_id;

    execute procedure hp_insert_into_blobstore('passport_photo_2_'||:session_id||'.jpg', 'clients', :photo_2_binary, 1000000000000, 0, 2)
    returning_values :photo_2_bls_id;

    select pkk.pk_id
    from passkind pkk
    inner join passdoctype pdt on pkk.pk_from_pdt_id = pdt.pdt_id
    where pdt.pdt_code = 'passport_card'
        and pkk.pk_country = 'kaz'
        and pdt.pdt_is_main_doc = 1
    into :kz_passport_card_pk_id;

    -- Обновляем именно таблицу clients и записываем туда полученные паспортные данные. В триггере clients_au0
    -- автоматически будет обновлена и таблица clientdocs, куда попадут те же самые данные
    -- Сама запись с этим документом в таблице clientdocs создается вместе с созданием клиента
    update clients
    set c_doc_type = :kz_passport_card_pk_id,
        c_pass_no = :doc_number,
        c_pass_date = :doc_issue_date,
        c_pass_date_expire = :doc_end_date,
        c_pass_who = :doc_issue_org_name,
        c_blob_pass1_id = :photo_1_bls_id,
        c_blob_pass2_id = :photo_2_bls_id,
        c_blob_photo_id = :client_photo_id
    where c_id = :c_id;

    -- Верифицируем клиента отдельным запросом
    update clients
    set c_verify_status = 2,
        c_verify_datetime = current_timestamp
    where c_id = :c_id;

    -- Если ни 1 записи не было обновлено, значит, процесс регистрации пошел не по сценарию
    if (row_count = 0) then
        exception e_default f_format('Паспортные данные для клиента c_id=%1 не были сохранены', :c_id);

    -- Удаляем запись, она уже обработана и больше не нужна
    delete from bioguard_temp_data btd
    where btd.btd_session_id = :session_id;
end
