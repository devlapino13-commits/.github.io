CREATE PROCEDURE HP_OFFICERPAR7_RU_UPDATE(
    REP_DATE DATE)
AS
declare variable officer_id decimal(18,0);
declare variable balance decimal(18,4);
declare variable delinq_balance7 decimal(18,4);
declare variable par7 decimal(5,5);
declare variable reg_id integer;
declare variable par7_last decimal(5,5);
begin
  for select oks_kc_id, sum(oks_balance), sum(iif(:rep_date - oks_delinq_date > 30, oks_balance, 0)), -- баланс ПАР30
             oks_region_id
      from oksummary
      where :rep_date between oks_date_begin and oks_date_end and
            oks_close_state = 0 and
            oks_balance > 0 and
            oks_kc_id > 1000000000004
      group by oks_kc_id, oks_region_id
      into officer_id, balance, delinq_balance7, reg_id
  do
  begin
    par7 = cast(delinq_balance7 as double precision) / balance * 100; -- получаю % ПАР 30 по отношению ко всему портфелю
    par7_last = null;

    select op_par8_180
    from officerpar7
    where :rep_date between op_date_begin and op_date_end and
          op_kc_id = :officer_id
    into par7_last;
    par7_last = coalesce(par7_last, 0);

    if (par7_last <> -1.00000 and
        par7_last <= 100.0000 and
        ((par7 >= 9.00000 and
        par7_last < 9.00000) or (par7 < 9.00000 and
        par7_last >= 9.00000))) then
    begin
      insert into officerpar7 (op_kc_id, op_number, op_date_begin, op_date_end, op_balance, op_delinq_bal8_180,
                               op_par8_180, op_reg_id)
      values (:officer_id, null, :rep_date, '31.12.9999', :balance, :delinq_balance7, :par7, :reg_id);
    end
  end
end
