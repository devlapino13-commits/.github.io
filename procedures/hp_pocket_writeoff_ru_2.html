CREATE PROCEDURE HP_POCKET_WRITEOFF_RU_2(
    REP_REGION_ID INTEGER,
    REP_DATE DATE,
    DO_SAVE SMALLINT)
RETURNS (
    K_ID ID_TYPE,
    REG_NAME STR_100,
    K_KONTRACT_NO STR_100,
    KF_NAME STR_100,
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    ZAD_FINE CURR_TYPE,
    TOTAL_ZAD CURR_TYPE,
    OL_ID ID_TYPE,
    OL_REPAY_AMOUNT CURR_TYPE,
    ERROR_TEXT STR_255,
    OE_BALANCE CURR_TYPE,
    DISB_DATE DATE,
    DELIQ_DAYS INTEGER,
    BALANCE_PROBLEM CURR_TYPE)
AS
declare variable balance_date date;
declare variable un_id id_type;
declare variable rd_id integer;
declare variable reg_id integer;
declare variable kf_id id_type;
declare variable oek_rd_id integer;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

--if (rep_date <> current_date) then exit;
  do_save = 0;

  balance_date = current_date;
  un_id = 1010100000278;
  rd_id = f_get_rd_id('Reason_WriteOff.Delinq360');

-- беру контракты, по которым просрочка более 360 дней + контракты должны быть выданы начиная с 28.02.2012
  for select kont.k_id, kont.k_last_reg_id, kont.k_kontract_no, reg.reg_name, kont.k_from_kf_id, kont.k_fact_disb_date,
             cast(:rep_date as date) - oks.oks_delinq_date
      from oksummary oks
      inner join region_list(:rep_region_id, 1, 3, 1) on oks.oks_region_id = id_list
      inner join kontract kont on oks.oks_from_k_id = kont.k_id
      inner join region reg on reg.reg_unicode = kont.k_last_reg_id
      where cast(:rep_date as date) between oks.oks_date_begin and oks.oks_date_end and
            cast(:rep_date as date) - oks.oks_delinq_date > 360 and
            kont.k_fact_disb_date >= '28.02.2012' and
            oks_close_state = 0
      into :k_id, :reg_id, :k_kontract_no, :reg_name, :kf_id, disb_date, deliq_days
  do
  begin
    select kal.kal_new_kf_name
    from knaktlist kal
    where kal.kal_from_kf_id = :kf_id and
          kal.is_delete = 0 and
          :balance_date between kal.kal_date and kal.kal_date_end
    into :kf_name;

    zad_os = 0;
    zad_perc = 0;
    zad_fine = 0;

--  получаю баланс по этим контрактам
    select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine)
    from aarep_make_zad_list(:k_id, :balance_date, 1)
    where sh_date is not null
    into :zad_os, :zad_perc, zad_fine;

    zad_os = coalesce(zad_os, 0);
    zad_perc = coalesce(zad_perc, 0);
    zad_fine = coalesce(zad_fine, 0);
    total_zad = zad_os + zad_perc + zad_fine;

    oe_balance = 0;

    select sum(oe.ox_amount_plus - oe.ox_amount_minus - oe.ox_amount_writeoff)
    from operextra oe
    where oe.ox_from_rd_id = :oek_rd_id and
          ox_object_kind = 9 and
          oe.ox_object_id = :k_id and
          oe.ox_ol_is_active = 1
    into :oe_balance;

    oe_balance = coalesce(oe_balance, 0);

    total_zad = total_zad + oe_balance;

    if (total_zad > 0) then
    begin
      /*
            ol_repay_amount = 0;

            SELECT
                OL_ID,
                OL_REPAY_AMOUNT,
                ERROR_TEXT
            FROM
                HP_DO_REPAYMENT(:K_ID,
                0,
                :balance_date,
                :total_zad,
                1,
                1,
                1,
                1,--1,  -- repay_to_komiss
                :do_save,
                1, --1, -- if_full_repay
                1,
                :reg_id,
                3,
                null,
                0,
                :un_id,
                null,
                'Пакетное списание (просрочка более 360 дней, выданные с 28.02.2012)',
                null,
                :rd_id,
                :total_zad)
                into :ol_id, :ol_repay_amount, error_text;

            balance_problem = total_zad - coalesce(ol_repay_amount, 0);  */
      suspend;
    end
  end
end
