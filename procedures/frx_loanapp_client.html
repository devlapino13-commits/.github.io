CREATE PROCEDURE FRX_LOANAPP_CLIENT(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    CLIENT_NO D_INTEGER,
    CLIENT_FIO D_STR_100,
    CLIENT_BIRTH_DATE D_DATE,
    CLIENT_AMOUNT D_CURRENCY,
    CLIENT_IS_CASHIER D_SMALLINT,
    CLIENT_BUS_SUBTYPE D_STR_CODE,
    CLIENT_LN_TARGET D_STR_255,
    CLIENT_STATUS_STR D_STR_20,
    CLIENT_BUS_TYPE D_STR_CODE,
    C_PASS_NO D_STR_CODE,
    C_PASS_WHO D_STR_255,
    C_PASS_DATE D_DATE,
    C_PASS_WHO_CODE D_STR_20,
    C_FULL_ADDRESS_INDEX D_STR_255,
    QR_CODE D_STR_LONG,
    LANG_DEFAULT D_SMALLINT,
    LANG_OTHER D_SMALLINT,
    ALLOW_TRANSFER_RIGHTS D_SMALLINT,
    C_BIRTH_CITY STR_100,
    CLIENT_INN VARCHAR(30),
    CLIENT_SNILS VARCHAR(30),
    CLIENT_NAME_OLD STR_255,
    CLIENT_NAME_WITH_LAT STR_255,
    CLIENT_PASS_LIST_OLD D_STR_8100,
    CLIENT_SIGN_DATE D_DATE,
    PURPOSE_FUNDING D_STR_255,
    C_INN D_STR_20,
    C_ADDRESS_FACT D_STR_255,
    CELL_MAIN D_STR_100,
    PIN_CODE D_STR_CODE,
    CONSENT_DOC_DATE D_DATE,
    CARDNUMBER D_STR_100,
    CAA_VALUE DECIMAL(18,2),
    REG_NAME_ALTERNAT D_STR_100,
    PR_MAME D_STR_64,
    OGRNIP D_STR_CODE)
AS
declare variable MA_ID integer;
declare variable PR_LOAN_TYPE smallint;
declare variable C_ID ID_TYPE;
declare variable EDL_ID D_ID;
declare variable C_BLANK_LANG D_STR_CODE;
declare variable C_ALTERNAT_FIO STR_100;
declare variable PR_CODE D_STR_100;
declare variable DB_ID D_INTEGER;
declare variable OFFICE_ID D_INTEGER;
declare variable MAC_ID D_ID;
begin
  -- Процедура готовит данные для печати клиентской части кредитной заявки
  db_id = f_get_db_id();

  obj_kind = f_objkind(obj_kind, 'id');

  if (obj_kind = 9) then -- если передан кредит, то вычисляем заявку
  begin
    select k_from_ma_app_id
    from kontract
    where k_id = :obj_id
    into :ma_id;
  end
  else
  begin
    ma_id = :obj_id;
  end

  select pr_loan_type, pr_code, ma_office_id, pr_name
  from miniapp, product
  where ma_from_pr_id = pr_id and
        ma_id = :ma_id and
        ma_date_end >= '31.12.2999'
  into :pr_loan_type, :pr_code, office_id, pr_mame;

  if (pr_code containing '.kardhasan.') then
    select rd_name
    from miniapp_ipf
      inner join refdata
        on rd_id = mai_loan_target
    where mai_from_ma_id = :ma_id
    into :purpose_funding;

  client_no = 0;
  for select c_name_fio, c_birth_date, iif(mac_amount > 0, mac_amount, null),
    mac_is_cashier, rd_name, bs_name, trim(
    case
      when mac_is_poruch = 1 then 'Поручитель'
      when mac_is_cashier = 1 then
        case
          when :db_id = 41701 then 'Клиент'
          else 'Клиент (Казн.)' -- RFCKG-5042
        end
      else 'Клиент'
    end),
    bt_name, cl.c_pass_no, cl.c_pass_who, cl.c_pass_date, cl.c_pass_who_code,
    c_id, c_blank_lang, mac_allow_transfer_rights, c_birth_city, trim(iif(c_citizen = 13, '', c_name_eng)),
    c_inn, mac_id
  from clients cl, miniappclient
    left join refdata on (mac_loan_target = rd_id)
    left join bussubtype on (mac_business_subtype = bs_id)
    left join bustype on (bs_from_bt_id = bt_id)
  where mac_from_c_id = c_id and
    (mac_amount > 0 or
    mac_is_poruch = 1) and
    (:pr_loan_type = 1 or
    mac_is_poruch = 0) and
    mac_from_ma_id = :ma_id
  order by mac_is_poruch asc
  into client_fio, client_birth_date, client_amount, client_is_cashier,
    client_ln_target, client_bus_subtype, client_status_str, client_bus_type,
    c_pass_no, c_pass_who, c_pass_date, c_pass_who_code, c_id, c_blank_lang,
    allow_transfer_rights, c_birth_city, c_alternat_fio, c_inn, mac_id
  do
  begin
    if (c_blank_lang = f_get_const('blank_default_lang')) then
    begin
      lang_default = 1;
      lang_other = 0;
    end else
    begin
      lang_default = 0;
      lang_other = 1;
    end

    client_name_with_lat = client_fio;

    if (c_alternat_fio <> '') then
        client_name_with_lat = client_name_with_lat || ' (' || c_alternat_fio || ') ';

    select left(
             list(distinct(
               coalesce(cds_doc_no,'') || ' ' ||
               lpad(extract(day   from cds_doc_date_begin),2,'0') || '.' ||
               lpad(extract(month from cds_doc_date_begin),2,'0') || '.' ||
               extract(year from cds_doc_date_begin) || ' ' ||
               coalesce(cds_doc_issued_by,'') || ' ' ||
               coalesce(cds_doc_issued_code, f_docrecog_value(drl_id,'authority_code',''))
             ), ' | '),
             8100
           ) as old_passports_line
    from hp_clientdocs_pass_list(:c_id, 1)  -- только старые
    where cds_from_pk_id in (1,6,8,11,12,15,17,34,35,36,42,296,330)
    into client_pass_list_old;

    select cds_doc_no
    from clientdocs
    where cds_from_c_id = :c_id
    and cds_from_pk_id = 61
    and cds_is_delete = 0
    order by cds_doc_date_begin desc 
    rows 1
    into client_snils;
    
    select cds_doc_no
    from clientdocs
    where cds_from_c_id = :c_id
    and cds_from_pk_id = 62
    and cds_is_delete = 0
    order by cds_doc_date_begin desc 
    rows 1
    into client_inn;

    select list(distinct(drv_final_value))
    from docrecogmain
    inner join docrecoglist on drl_from_drm_id = drm_id
    inner join docrecogfield on drf_from_drm_id = drm_id
    inner join docrecogvalue on drv_from_drf_id = drf_id and drv_from_drl_id = drl_id
    where
    drm_code = 'rus.cert.name_change' and
    drf_code = 'client_name_old' and
    drl_from_c_id = :c_id
    into :client_name_old;

    if (client_name_old is null) then
        client_name_old = '';

    c_full_address_index = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :c_id and
          adc_address_type = 1 and
          adc_from_adl_id = adl_id
    into c_full_address_index;

    if (db_id = 39801) then
    begin
      select adl_address_str     -- RFCKZ-7
      from addrclient, addrlist
      where adc_from_c_id = :c_id and
            adc_address_type = 2 and
            adc_from_adl_id = adl_id
      into c_address_fact;

      cell_main = f_cont_info(0, :c_id, 'cell_main', 'ok');

      consent_doc_date = current_date;

      pin_code = '';
      -- пин-код электронной подписи клиента. Текст пустой если не подписан
      if (coalesce(add_params, '') <> '') then
      begin
        edl_id = f_json_str_get_value(add_params, 'edl_id', null);
    
        select eds.eds_pin
        from edoc_sign eds
        where eds.eds_from_edl_id = :edl_id and
              eds.eds_from_c_id = :c_id and
              eds.eds_status = 1
        into pin_code;
      end
        
      select et.et_cardnumber
      from extsys_token et
      where et.et_from_c_id = :c_id and
            et.et_is_actual = 1
      into :cardnumber;

      select coalesce(sum(caa.caa_value), 0)
      from cfp_anketa_answers caa
      inner join cfp_anketa ca on ca.ca_id = caa.caa_from_ca_id
      inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id
      where ca.ca_from_c_id = :c_id and
          ca.ca_object_kind = f_objkind('miniappclient', 'id') and
          ca.ca_object_id = :mac_id and
          cq.cq_code_name in ('salary_statement', 'contract_salary')
      into :caa_value;

      select reg_name_alternat
      from print_company(:office_id, 1, current_date, 0)
      rows 1
      into :reg_name_alternat;
    end
    -- QR-код электронной подписи клиента. Текст пустой если не подписан
    if (coalesce(add_params, '') <> '') then
    begin
      edl_id = f_json_str_get_value(add_params, 'edl_id', null);
      qr_code = f_edoc_qr_by_edl(edl_id, c_id);

      -- Дата подписания. Не используем edoc_sign, чтобы не получилось что дата
      -- подписи больше даты документа
      select edl_blank_date
      from edoc_list
      where edl_id = :edl_id
      into client_sign_date;
    end

    ------ RFCRU-5453 --
    if (db_id = 81001) then
    begin
      select ogrn
      from hp_kontur_ip_get_base_info('clients', :c_id)
      into ogrnip;
    end

    client_no = client_no + 1;
    suspend;
  end
end
