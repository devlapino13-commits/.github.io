CREATE PROCEDURE HP_UPDATE_ADDRDETAILS(
    C_ID D_ID,
    REGISTRATION_DATE D_DATE,
    AUTHORITY_CODE D_STR_20,
    AUTHORITY_NAME D_STR_255)
AS
declare variable add_id d_id;
declare variable adl_id d_id;
begin
  adl_id = (
    select adc_from_adl_id
    from addrclient
    where adc_from_c_id = :c_id
      and adc_address_type = 1);

  if (adl_id is null)
  then
    exception e_message 'Юридический адрес клиента не определен.';

  if (registration_date is null)
  then
    exception e_default 'Введите дату регистрации.';

  update or insert into addrdetails(
    add_adl_id,
    add_registration_date,
    add_authority_code,
    add_authority_name)
  values(
    :adl_id,
    :registration_date,
    trim(coalesce(:authority_code, '')),
    trim(coalesce(:authority_name, '')))
  matching(add_adl_id);
end
