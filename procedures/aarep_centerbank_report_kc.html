CREATE PROCEDURE AAREP_CENTERBANK_REPORT_KC(
    KC_ID D_ID,
    REP_QUERY_DATE D_DATE)
RETURNS (
    REC_NO D_INTEGER,
    C_NAME_FIO D_STR_255,
    C_PASS_SERIA D_STR_CODE,
    C_PASS_NUMBER D_STR_CODE,
    C_TYPE D_STR_CODE,
    C_INN D_STR_CODE,
    K_LOAN_KIND D_STR_CODE,
    K_MSP_SUBJECT D_STR_CODE,
    K_KONTRACT_NO D_STR_CODE,
    K_KONTRACT_DATE D_DATE,
    K_AMOUNT D_CURRENCY,
    K_PERIOD_IN_DAYS D_INTEGER,
    K_KONT_LAST_DATE D_DATE,
    K_KREDLINE_END D_DATE,
    K_PERCENT D_CURRENCY,
    K_DISB_PLACE D_STR_255,
    Z_SECURED D_STR_100,
    Z_NAME D_STR_255,
    Z_COST D_CURRENCY,
    Z_CLIENT_NAME D_STR_255,
    Z_CLIENT_INN D_STR_100,
    Z_INSURANCE_COST D_CURRENCY,
    K_RESTRUCT_DATE D_DATE,
    K_POS_LOAN D_STR_10)
AS
declare variable PASS_NO D_STR_CODE;
declare variable K_ID D_ID;
declare variable Z_REALIZE_COST D_CURRENCY;
declare variable ACC_ID D_ID;
declare variable PR_CODE D_STR_255;
declare variable PR_LOAN_TYPE D_INTEGER;
declare variable K_KOLICH_CLIENT_APPROVED D_INTEGER;
declare variable K_BASE_K_ID D_ID;
begin
  -- Процедура готовит данные с 2018 года по июнь 2020 по одному займу
  -- Запрос от ЦБ от 22.07.2020

  c_pass_seria = '';
  c_pass_number = '';

  if (rep_query_date is null) then
    rep_query_date = '30.06.2020';

  rec_no = 1;
  select k_id, c_name_fio, c_pass_no, iif(c_type = 1, 'ЮЛ', 'ФЛ'), replace(coalesce(c_inn, ''), ' ', ''),
         k_kontract_no, k_kontract_date, kc_amount_approved, datediff(day, k_disb_date, k_kont_last_date),
         k_kont_last_date, k_kredline_end, pr_code, pr_loan_type, round(k_percent_per_day * 365.0000, 2),
         k_restruct_date, k_kolich_client_approved, k_base_k_id
  from knclient, clients, kontract, product
  where kc_id = :kc_id and
        c_id = kc_from_c_id and
        kc_from_k_id = k_id and
        k_from_pr_id = pr_id
  into k_id, c_name_fio, pass_no, c_type, c_inn,
       k_kontract_no, k_kontract_date, k_amount, k_period_in_days,
       k_kont_last_date, k_kredline_end, pr_code, pr_loan_type, k_percent,
       k_restruct_date, k_kolich_client_approved, k_base_k_id;

--  select pass_seria, pass_number
--  from hp_passport_number_split(:pass_no)
--  into :c_pass_seria, :c_pass_number;

  k_pos_loan = 'Нет';
  if (pr_code containing '.patent') then
    k_pos_loan = 'Да';

  k_loan_kind = 'Микрозайм';
  k_msp_subject = 'Нет';

  k_disb_place = '';
  select pl_name
  from kontstate, region, addrlist, places
  where ks_from_k_id = :k_id and
        ks_ol_is_delete = 0 and
        ks_state = 1 and
        ks_region_id = reg_unicode and
        reg_address_fact = adl_id and
        adl_from_pl_id = pl_id
  rows 1
  into k_disb_place;

  z_client_name = '';
  z_client_inn = '';
  z_name = '';
  z_cost = 0;

  if ((pr_code containing '.zalog_car.' or pr_code containing '.zalog.')  and
      pr_loan_type = 1) then
  begin
    select zs_name, c_name_fio, c_inn
    from zalog, clients, zalogsubtype
    where z_from_k_id = :k_id and
          z_type = 1 and
          z_from_c_id = c_id and
          z_subtype = zs_id
    order by coalesce(z_date_begin, :k_kontract_date) desc
    rows 1
    into z_name, z_client_name, z_client_inn;
  end

  z_secured = 'Необеспеченный';
  z_insurance_cost = 0;

  if (k_kolich_client_approved = 1 and k_base_k_id is null) then
  begin
    if (z_client_name <> '') then
    begin
      z_secured = 'Залог';
  
      -- Залоговая стоимоть
      select acc_id
      from ref_account
      where acc_object_kind = 8 and
            acc_object_id = :kc_id and
            acc_from_sab_id = 90 -- sab_full_code = 'Микрозаймы.Внебаланс.Залог'
      into acc_id;
  
      z_realize_cost = 0;
      select -kobs_bal_sum, kobs_dt_sum
      from kn_obsummary
      where kobs_acc_id = :acc_id and
            kobs_date <= :rep_query_date
      order by kobs_acc_id desc, kobs_date desc
      rows 1
      into z_cost, z_realize_cost;
  
      if (z_cost = 0) then
        z_cost = z_realize_cost;
    end
  end else
  begin
    z_secured = 'Поручительство';

    select c_name_fio
    from knclient, clients
    where kc_from_k_id = :k_id and
          kc_id <> :kc_id and
          kc_amount_approved > 0 and
          kc_from_c_id = c_id
    order by kc_id
    rows 1
    into z_client_name;

    if (z_client_name = '' and k_base_k_id is not null) then
    begin
      if (k_base_k_id = k_id) then
      begin
        select k_id
        from kontract
        where k_base_k_id = :k_base_k_id and
              k_id <> k_base_k_id
        rows 1
        into k_id;
      end else
        k_id = k_base_k_id;

      select c_name_fio
      from knclient, clients
      where kc_from_k_id = :k_id and
            kc_id <> :kc_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id
      order by kc_id
      rows 1
      into z_client_name;
    end
  end

  suspend;
end
