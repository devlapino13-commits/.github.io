CREATE PROCEDURE HP_BKI_GET_INFO4REQ_DOC(
    C_ID ID_TYPE)
RETURNS (
    DOC_COUNTRY D_STR_10,
    DOC_TYPE D_STR_10,
    DOC_SERIAL D_STR_10,
    DOC_NUMBER D_STR_20,
    DOC_DATE D_STR_20,
    DOC_ENDDATE D_STR_20,
    DOC_WHO D_STR_255,
    DOC_DEPARTMENT_CODE D_STR_255)
AS
declare variable c_citizen d_integer;
declare variable pk_id d_integer;
declare variable country_citizen d_str_255;
declare variable cds_id d_integer;
declare variable serial_length d_integer;
declare variable c_doc_department_code d_str_255;

declare function del_twice_space(str d_str_255)
  returns d_str_255
  as
  begin
    str = replace(str, ascii_char(9), ' ');
    str = replace(str, '\', '');
    str = f_replace_all(str, ' -', '-');
    str = f_replace_all(str, '- ', '-');
    str = f_replace_all(str, '  ', ' ');

    return trim(str);
  end

declare function f_my_fix_wrong_lat(in_str d_str_255)
  returns d_str_255
  as
  declare variable res_text d_str_255;
  begin
    res_text = in_str;
    if(res_text similar to '%[а-яА-Я]%') then
    begin
      res_text = replace(res_text collate unicode, 'А' collate unicode, 'A');
      res_text = replace(res_text collate unicode, 'В' collate unicode, 'B');
      res_text = replace(res_text collate unicode, 'С' collate unicode, 'C');
      res_text = replace(res_text collate unicode, 'Е' collate unicode, 'E');
      res_text = replace(res_text collate unicode, 'Г' collate unicode, 'F');
      res_text = replace(res_text collate unicode, 'Н' collate unicode, 'H');
      res_text = replace(res_text collate unicode, 'К' collate unicode, 'K');
      res_text = replace(res_text collate unicode, 'М' collate unicode, 'M');
      res_text = replace(res_text collate unicode, 'О' collate unicode, 'O');
      res_text = replace(res_text collate unicode, 'Р' collate unicode, 'P');
      res_text = replace(res_text collate unicode, 'Т' collate unicode, 'T');
      res_text = replace(res_text collate unicode, 'И' collate unicode, 'U');
      res_text = replace(res_text collate unicode, 'Х' collate unicode, 'X');
      res_text = replace(res_text collate unicode, 'У' collate unicode, 'Y');

      res_text = replace(res_text collate unicode, 'а' collate unicode, 'a');
      res_text = replace(res_text collate unicode, 'с' collate unicode, 'c');
      res_text = replace(res_text collate unicode, 'е' collate unicode, 'e');
      res_text = replace(res_text collate unicode, 'г' collate unicode, 'f');
      res_text = replace(res_text collate unicode, 'н' collate unicode, 'h');
      res_text = replace(res_text collate unicode, 'к' collate unicode, 'k');
      res_text = replace(res_text collate unicode, 'м' collate unicode, 'm');
      res_text = replace(res_text collate unicode, 'п' collate unicode, 'n');
      res_text = replace(res_text collate unicode, 'о' collate unicode, 'o');
      res_text = replace(res_text collate unicode, 'р' collate unicode, 'p');
      res_text = replace(res_text collate unicode, 'и' collate unicode, 'u');
      res_text = replace(res_text collate unicode, 'х' collate unicode, 'x');
      res_text = replace(res_text collate unicode, 'у' collate unicode, 'y');

    end
    return res_text;
  end

declare function f_my_fix_wrong_cyr(in_str d_str_255)
  returns d_str_255
  as
  declare variable res_text d_str_255;
  begin
    res_text = in_str;
    if(res_text similar to '%[a-zA-Z]%') then
    begin
      res_text = replace(res_text collate unicode, 'A' collate unicode, 'А');
      res_text = replace(res_text collate unicode, 'B' collate unicode, 'В');
      res_text = replace(res_text collate unicode, 'C' collate unicode, 'С');
      res_text = replace(res_text collate unicode, 'E' collate unicode, 'Е');
      res_text = replace(res_text collate unicode, 'F' collate unicode, 'Г');
      res_text = replace(res_text collate unicode, 'H' collate unicode, 'Н');
      res_text = replace(res_text collate unicode, 'K' collate unicode, 'К');
      res_text = replace(res_text collate unicode, 'M' collate unicode, 'М');
      res_text = replace(res_text collate unicode, 'O' collate unicode, 'О');
      res_text = replace(res_text collate unicode, 'P' collate unicode, 'Р');
      res_text = replace(res_text collate unicode, 'T' collate unicode, 'Т');
      res_text = replace(res_text collate unicode, 'U' collate unicode, 'И');
      res_text = replace(res_text collate unicode, 'X' collate unicode, 'Х');
      res_text = replace(res_text collate unicode, 'Y' collate unicode, 'У');

      res_text = replace(res_text collate unicode, 'a' collate unicode, 'а');
      res_text = replace(res_text collate unicode, 'c' collate unicode, 'с');
      res_text = replace(res_text collate unicode, 'e' collate unicode, 'е');
      res_text = replace(res_text collate unicode, 'f' collate unicode, 'г');
      res_text = replace(res_text collate unicode, 'k' collate unicode, 'к');
      res_text = replace(res_text collate unicode, 'm' collate unicode, 'м');
      res_text = replace(res_text collate unicode, 'n' collate unicode, 'п');
      res_text = replace(res_text collate unicode, 'o' collate unicode, 'о');
      res_text = replace(res_text collate unicode, 'p' collate unicode, 'р');
      res_text = replace(res_text collate unicode, 'u' collate unicode, 'и');
      res_text = replace(res_text collate unicode, 'x' collate unicode, 'х');
      res_text = replace(res_text collate unicode, 'y' collate unicode, 'у');
    end
    return res_text;
  end

begin
  for select c_citizen, c_doc_type, c_pass_no, f_datetostr(c_pass_date), f_datetostr(c_pass_date_expire), coalesce(c_pass_who, ''), coalesce(c_pass_who_code, '')
  from clients cl
  where cl.c_id = :c_id
  into c_citizen, pk_id, doc_number, doc_date, doc_enddate, doc_who, c_doc_department_code
  do
  begin
    for
    with hist_passports as (
          select distinct cds_id, cds_doc_no, f_datetostr(cds_doc_date_begin) cds_doc_date_begin,
              f_datetostr(cds_doc_date_end) cds_doc_date_end, del_twice_space(cds_doc_issued_by) cds_doc_issued_by,
              left(cds_doc_issued_code, 10) doc_department_code
          from hp_clientdocs_pass_list(:c_id, 1)
          order by cds_id desc
          rows 9)
    select null, :c_citizen, :pk_id, :doc_number, :doc_date, :doc_enddate, :doc_who, :c_doc_department_code
    from rdb$database
    union all
    select cds_id, null, null, cds_doc_no, cds_doc_date_begin, cds_doc_date_end, cds_doc_issued_by, doc_department_code
    from hist_passports
    where cds_doc_no is distinct from :doc_number
    into cds_id, c_citizen, pk_id, doc_number, doc_date, doc_enddate, doc_who, doc_department_code
    do
    begin
      doc_country = null;
      doc_type = null;
      doc_serial = null;

      if (coalesce(doc_date, '') = '') then
        continue;

      if (cds_id is not null) then
      begin
        select cds_from_pk_id
        from clientdocs cds
        where cds_id = :cds_id
        into :pk_id;
    
        select pk_country_id
        from passkind
        where pk_id = :pk_id
        into :c_citizen;
      end

      doc_number = f_my_fix_wrong_lat(doc_number);
      country_citizen = f_my_fix_wrong_cyr((select rd_name from refdata where rd_id = :c_citizen));
  
      select rd_memo from refdata
      where rd_main_code = 'CountryOKSM'
        and (rd_name = case :country_citizen
                          when 'Кыргызстан' then 'Киргизия'
                          when 'Республика Молдова' then 'Молдова, Республика'
                          else :country_citizen
                       end)
      into doc_country;
    
      if (doc_country is null) then
        doc_country = '999';

      if (c_citizen = 13) then -- Рф
        doc_type = case pk_id
                     when 12 then '21'    -- Паспорт гражданина РФ
                     when 13 then '22.1'  -- Загран. паспорт РФ
                     else '28' -- Иной документ, удостоверяющий личность гражданина Российской Федерации
                   end;
      else
        doc_type = '31'; -- Паспорт, выданный иностранным государством и признаваемый в Российской Федерации
    
      ---- серия и номер документа -----
      doc_number = replace(doc_number, ' ', '');
  
      serial_length = case pk_id
                        when 12  then 4 -- Паспорт гражданина РФ
                        when 13  then 2 -- Загран. паспорт РФ
                        when 17  then 2 -- Паспорт УЗ (зелёный)
                        when 330 then 2 -- Биометрический паспорт УЗ (бордовый)
                        when 15  then 1 -- Паспорт Тадж.
                        else 0
                      end;
    
      if (char_length(doc_number) > serial_length and serial_length > 0) then
      begin
        doc_serial = left(doc_number, serial_length);
        doc_number = substring(doc_number from serial_length + 1);
      end 
    
      select out_string
      from hp_check_characters_str(:doc_who, '[a-zA-Zа-яА-ЯЁё0-9@- .,'';\/№():_+|"#?!;]*')
      into doc_who;
  
      doc_who = del_twice_space(doc_who);

      if (pk_id = 12) then
      begin
        doc_department_code = replace(doc_department_code, ' ', '');  -- Код подразделения
      
        if (doc_department_code not similar to '[0-9]{3}\-[0-9]{3}' escape '\') then
        begin -- если Код подразделения не подходит по маске, то попытаемся его привести к нужному формату
          select out_string from hp_check_characters_str(:doc_department_code, '[0-9]*')
          into doc_department_code;
      
          if (char_length(doc_department_code) = 6) then  -- если цифр достаточно
            doc_department_code = left(doc_department_code, 3) || '-' || substring(doc_department_code from 4);
          else
            doc_department_code = '999-999'; -- заглушка
        end
      end
      else
        doc_department_code = null;

      if (doc_serial = '') then doc_serial = null;
      if (doc_who = '') then doc_who = null;
      if (doc_enddate = '') then doc_enddate = null;
      if (doc_department_code = '') then doc_department_code = null;

      suspend;
    end
  end
