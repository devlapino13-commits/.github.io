CREATE PROCEDURE HP_COMPET_AZ_OLIMPIC_2022_SPF(
    CPP_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable RES_DATE_BEGIN date;
declare variable RES_DATE_END date;
declare variable STAGE integer;
declare variable RD_ID integer;
declare variable RES_PERIOD_ID integer;
declare variable RES_TEAM_ID integer;
declare variable RES_LO_ID decimal(18,0);
declare variable AD_ACTIVE_BALANCE decimal(18,4) = 0;
declare variable POINTS_1 decimal(18,4) = 0;
declare variable POINTS_2 decimal(18,4) = 0;
declare variable POINTS_ALL decimal(18,4) = 0;
declare variable RES_TEAM_ID_PRIOR integer;
declare variable CPR_ID integer;
declare variable CPR_GROUP_NAME varchar(255);
declare variable CPD_COMMENT blob sub_type 1 segment size 160;
declare variable CPP_ORDERNO integer;
declare variable KIND integer;
declare variable CPR_FINAL_POINTS decimal(18,4) = 0;
declare variable AD_DISB_LOAN decimal(18,4) = 0;
declare variable PLACE_NO integer;
declare variable POINTS_3 decimal(18,4) = 0;
declare variable POINTS_4 decimal(18,4) = 0;
declare variable IS_OTPUSK varchar(100);
declare variable IS_OTPUSK_MONTH varchar(100);
declare variable KOL integer;
declare variable AD_DISB_AMOUNT decimal(18,4) = 0;
declare variable AAREP_FACT_PLANNED decimal(18,4) = 0;
declare variable AAREP_PLAN_ALL decimal(18,4) = 0;
declare variable POINTS_5 decimal(18,4) = 0;
declare variable CLIENT_DIFF decimal(18,4) = 0;
declare variable CLIENT_DIFF_PRIOR decimal(18,4) = 0;
declare variable STAGE_CPP_ID integer;
declare variable AD_DISB_NEW_CLIENT integer;
declare variable DD_TMP date;
declare variable OTPUSK_DAYS smallint;
declare variable N integer;
declare variable AD_DELINQ_BALANCE_30 CURR_TYPE;
declare variable AD_DIFF_BALANCE CURR_TYPE;
declare variable SVZ_PORTFEL_BEGIN CURR_TYPE;
declare variable SVZ_PORTFEL_END CURR_TYPE;
declare variable PRINYAT_PORTFEL CURR_TYPE;
declare variable DATE_BEGIN date;
declare variable DATE_END_PRV date;
declare variable ZARABOTOK_END CURR_TYPE;
declare variable ZARABOTOK_BEGIN CURR_TYPE;
declare variable ZARABOTOK_DIFF CURR_TYPE;
declare variable AD_DELINQ_BALANCE_DIFF_30 CURR_TYPE;
declare variable AVG_DELINQ_BALANCE_30 CURR_TYPE;
declare variable AVG_DELINQ_BALANCE_30_DIFF CURR_TYPE;
declare variable AVG_DISB_AMOUNT CURR_TYPE;
declare variable AVG_ZARABOTOK_DIFF CURR_TYPE;
declare variable AVG_BALANCE_DIFF CURR_TYPE;
declare variable REP_PERIOD integer;
declare variable PLAN_SUM CURR_TYPE;
declare variable DISB_SUM CURR_TYPE;
declare variable DOSROCH_SUM CURR_TYPE;
declare variable PLAN_PERC CURR_TYPE;
declare variable LO_STAGE_DATE date;
declare variable LO_STAGE integer;
declare variable AVG_POINTS CURR_TYPE;
declare variable LO_PANELTY_COUNT CURR_TYPE;
declare variable TOTAL_POINTS CURR_TYPE;
declare variable ADD_POINTS CURR_TYPE;
declare variable KOL_DAY_DECADE smallint;
declare variable KOL_DAY_WORKED smallint;
declare variable KOL_DAY_MONTH integer;
declare variable OTPUSK_KIND smallint;
declare variable CPR_COMMENT STR_255;
declare variable PAR7 PERC_TYPE;
declare variable GROUP_NAME STR_255;
declare variable MAX_POINTS CURR_TYPE;
declare variable AVG_DISB_LOAN CURR_TYPE;
declare variable KOEF_RETURN decimal(15,2);
declare variable TEAM_QTY integer;
declare variable CPD_POINTS_4 decimal(15,2);
declare variable KOLVO integer;
declare variable AVG_KOEF_RETURN_TMP decimal(15,2);
declare variable AVG_KOEF_RETURN decimal(15,2);
declare variable CPD_POINTS_9 decimal(15,2);
declare variable CPD_POINTS_ALL decimal(15,2);
declare variable AVG_KOEF_RETURN_RANK integer;
begin
  -- Расчет итогов олимпиады РФ 2020
  -- Stage:
  -- 0 - период (неделя)
  -- 1 - Промежуточные итоги
  -- 2 - Итоги
  -- 3 - Финальные результаты

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end, cpp_orderno
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end, cpp_orderno;

  kol_day_decade = res_date_end - res_date_begin + 1;

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end
  if (cpp_orderno = 0 and
      res_date_begin = cast('31.01.2022' as date)) then
  begin
    res_date_begin = '01.02.2022';
    res_date_end = '01.02.2022';

    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'standart')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    -- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    for select res_field_1, sum(ad_active_balance)
        from hp_analitica_report('kont_lo_teamname_id', 'tree_group', 8, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 360, 0, 'standart')
        where exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        group by 1
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :res_team_id, :ad_active_balance
    do
    begin
      if (exists(select *
                 from refdata
                 where rd_id = :res_team_id and
                       rd_is_active = 1)) then
      begin
        insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                  cpr_from_un_id, cpr_comment)
        values (:cpp_id, '', -1, :res_team_id, 0, :un_id, '')
        returning cpr_id
        into :cpr_id;
      end
    end
    suspend;
    exit;
  end

  -- Ищем итоги тура
  stage_cpp_id = null;
  select cpp_id
  from competperiod
  where cpp_from_rd_id = :rd_id and
        cpp_stage = 2 and
        cpp_status = 1 and
        cpp_date_end <= :res_date_end
  order by cpp_date_begin desc, cpp_id desc
  rows 1
  into :stage_cpp_id;

  if (stage = 0) then -- Если уровень Декады
  begin
    -- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'standart')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    date_begin = res_date_end - extract(day from res_date_end) + 1;
    date_end_prv = date_begin - 1;
    -- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    for select res_field_1, --
               res_field_2, --
               sum(ad_active_balance), --
               sum(ad_delinq_balance_end3), --
               sum(ad_delinq_balance_diff3), --
               sum(ad_disb_loan), --
               sum(ad_disb_amount)
        from hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group', 8, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 360, 0, 'standart')
        where exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        group by 1, 2
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :res_team_id, --
             :res_lo_id, --
             :ad_active_balance, -- Баланс на конец периода
             :ad_delinq_balance_30, -- Баланс просрочек 7 дней
             :ad_delinq_balance_diff_30, -- Изменение баланс 7 дней
             :ad_disb_loan, -- Выданные займы,
             :ad_disb_amount -- Сумма выдачи СВЗ
    do
    begin
      cpd_comment = '';
      if (ad_disb_loan is null) then
        ad_disb_loan = 0;
      if (ad_disb_amount is null) then
        ad_disb_amount = 0;
      if (aarep_fact_planned is null) then
        aarep_fact_planned = 0;
      if (aarep_plan_all is null) then
        aarep_plan_all = 0;
      if (ad_disb_new_client is null) then
        ad_disb_new_client = 0;

      -- Исключаем если нет действующего
      if (not(exists(select *
                     from hr_pstatus
                     where ps_from_c_id = :res_lo_id and
                           ps_date_end >= :res_date_end and
                           ps_date_begin <= :res_date_end and
                           ps_status = 1 and
                           ps_is_working = 1))) then
        res_team_id = 0;

      if (res_team_id <> 0) then
      begin
        -- Исключаем если не СВЗ/КС
        if (exists(select *
                   from hr_pstatus, joblist, jobkind
                   where ps_from_c_id = :res_lo_id and
                         ps_date_end >= :res_date_end and
                         ps_date_begin <= :res_date_end and
                         ps_status = 1 and
                         ps_from_jl_id = jl_id and
                         jl_kind = jk_id and
                         not(jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner')) and
                         ps_is_working = 1)) then
          res_team_id = 0;
      end

      is_otpusk = '';
      is_otpusk_month = '';
      if (res_team_id <> 0) then
      begin
        -- Исключаем если в отпуске.
        -- Проверяем статус отпуска на каждый день декады:
        -- если из декады проработал менее 5 дней то будем считать что в отпуске
        otpusk_days = 0;
        n = 0;
        while (n <= 9 and dateadd(day, n, res_date_begin) < res_date_end) do
        begin
          dd_tmp = dateadd(day, n, res_date_begin);
          if (exists(select *
                     from hr_voctravel, hr_orders, sa_worktype
                     where hrv_from_ho_id = ho_id and
                           ho_order_type = sawt_id and
                           hrv_from_c_id = :res_lo_id and
                           (sawt_base_type in (4, 5) or sawt_id = 17)  -- отпуск, декретный отпуск
                           and
                           sawt_id <> 20 and
                           sawt_hours_real >= 0 and
                           hrv_date_begin <= :dd_tmp and
                           hrv_date_end_fact >= :dd_tmp and
                           hrv_status = 1)) then
            otpusk_days = otpusk_days + 1;
          n = n + 1;
        end
        kol_day_worked = kol_day_decade - otpusk_days;

        if (kol_day_worked < 5) then
        begin
          if (not(exists(select *
                         from hr_voctravel, hr_orders
                         where hrv_from_ho_id = ho_id and
                               ho_order_type = 20 and
                               hrv_from_c_id = :res_lo_id and
                               hrv_date_begin <= :res_date_end and
                               hrv_date_end_fact >= :res_date_begin and
                               hrv_status = 1))) then
          begin
            is_otpusk = 'Отпуск декады.';
          end
        end
      end

      points_1 = 0.0;
      points_2 = 0.0;
      points_3 = 0.0;
      points_4 = 0.0;
      points_5 = 0.0; --    проверяем, если конец месяца т.е. не 10е и не 20е то включаем дополнительные расчеты
      ad_diff_balance = null;

      cpd_comment = is_otpusk || '<br>';
      if (is_otpusk_month <> '') then
        cpd_comment = cpd_comment || is_otpusk_month || '<br>';
      cpd_comment = cpd_comment || 'Активный баланс: <b>' || round(ad_active_balance, 2) || ' р.</b><br>';
      cpd_comment = cpd_comment || 'Баланс просрочки PAR30: <b>' || round(ad_delinq_balance_30, 2) || ' р.</b><br>';
      cpd_comment = cpd_comment || 'Изменение PAR30: <b>' || round(ad_delinq_balance_diff_30, 2) || ' р.</b><br>';
      cpd_comment = cpd_comment || 'Сумма выдач СВЗ за период: <b>' || round(ad_disb_amount, 0) || '</b><br>';
      cpd_comment = cpd_comment || 'Кол-во выдач СВЗ за период: <b>' || round(ad_disb_loan, 0) || '</b><br>';

      if (res_team_id <> res_team_id_prior) then -- Если команда отличается
      begin
        res_team_id_prior = res_team_id;
        -- Находим нужную строку команды за этот период
        cpr_id = null;
        select cpr_id
        from competresult
        where cpr_from_cpp_id = :cpp_id and
              cpr_object_kind = -1 and
              cpr_object_id = :res_team_id
        into :cpr_id;

        if (cpr_id is null) then -- если такой строки нет, создаем новую
        begin
          cpr_group_name = '';
          stage_cpp_id = null;
          -- Получаем название Группы где последний раз была команда
          if (stage_cpp_id is not null) then
          begin
            -- Если это первая неделя второго этапа, название лиги берем из поля комментария предыдущего расчета
            select coalesce(cpr_comment, '')
            from competresult, competperiod
            where cpr_from_cpp_id = :stage_cpp_id and
                  cpr_object_kind = -1 and
                  cpr_object_id = :res_team_id and
                  cpr_from_cpp_id = cpp_id -- этой привязки небыло
                  rows 1
            into :cpr_group_name;

            if (coalesce(cpr_group_name, '') = '') then
            begin
              select rd_memo
              from refdata
              where rd_id = :res_team_id
              into :cpr_group_name;
            end
          end
          else
          begin -- Если это не первая неделя второго Этапа
            select cpr_group_name
            from competresult, competperiod
            where cpr_from_cpp_id = cpp_id and
                  cpp_from_rd_id = :rd_id and
                  cpr_object_kind = -1 and
                  cpr_object_id = :res_team_id and
                  cpr_group_name <> '' and
                  cpp_orderno < :cpp_orderno
            order by cpp_orderno desc
            rows 1
            into :cpr_group_name;

            if (coalesce(cpr_group_name, '') = '') then
            begin
              select rd_name
              from refdata
              where rd_id = :res_team_id
              into :cpr_group_name;
            end
          end
          if (cpr_group_name is null) then
            cpr_group_name = '';

          insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                    cpr_from_un_id, cpr_comment)
          values (:cpp_id, :cpr_group_name, -1, :res_team_id, 0, :un_id, '')
          returning cpr_id
          into :cpr_id;
        end
      end

      otpusk_kind = 0; -- 0 - сотрудник не в отпуске; 1 - сотрудник в отпуске только по декаде; 2 - сотрудник в отпуске только по месяцу; 3 - сотрудник в отпуске по декаде и по месяцу;
      if (is_otpusk <> '') then
        otpusk_kind = 1;
      if (is_otpusk_month <> '') then
        otpusk_kind = otpusk_kind + 2;

      -- Вставляем данные по сотруднику
      insert into competdetails (cpd_from_cpr_id, cpd_object_kind, cpd_object_id, cpd_points_all, cpd_points_1,
                                 cpd_points_2, cpd_points_3, cpd_points_4, cpd_points_5, cpd_comment, cpd_staff_kolich, -- 1 если не отпуск, иначе 0
                                 cpd_add_data_1, -- Баланс просрчоки более 30 дней
                                 cpd_add_data_2, -- Изменение баланса просрочки более 30 дней
                                 cpd_add_data_3, -- Сумма выдачи СВЗ за декаду в рублях)
                                 cpd_add_data_4, -- Прирост заработка СВЗ
                                 cpd_add_data_5, -- Прирост портфеля СВЗ
                                 cpd_add_data_6, -- Заработок
                                 cpd_add_data_7, -- Количество выдач
                                 cpd_add_data_8, -- Сумма выдачи за месяц
                                 cpd_add_data_9 -- конец месяца или нет 0 - нет; 1 - да
                                 )
      values (:cpr_id, 0, :res_lo_id, :points_all, :points_1, :points_2, :points_3, :points_4, :points_5, :cpd_comment,
              :otpusk_kind, -- iif(:is_otpusk='',1,0),
              :ad_delinq_balance_30, --cpd_add_data_1
              :ad_delinq_balance_diff_30, --cpd_add_data_2
              :ad_disb_amount, --cpd_add_data_3
              :zarabotok_diff, --cpd_add_data_4
              :ad_diff_balance, --cpd_add_data_5
              :plan_perc, --cpd_add_data_6
              :ad_disb_loan, --cpd_add_data_7
              :disb_sum, --cpd_add_data_8
              0); --cpd_add_data_9
    end
    -- Расчет средних данных на одного СВЗ
    client_diff_prior = -777;
    kol = 0;
    place_no = 0;
    for select cpr_id, cpr_group_name, cpr_object_id,
               round(avg(iif(1 = 1, cast(cpd_add_data_1 as double precision), null)), 2),
               round(avg(iif(1 = 1, cast(cpd_add_data_2 as double precision), null)), 2),
               round(avg(iif(cpd_staff_kolich <> 1, cast(cpd_add_data_3 as double precision), null)), 2),
               round(avg(iif(cpd_staff_kolich <> 2, cast(cpd_add_data_4 as double precision), null)), 2),
               round(avg(iif(cpd_staff_kolich <> 2, cast(cpd_add_data_5 as double precision), null)), 2),
               round(avg(iif(cpd_staff_kolich = 0, cast(cpd_add_data_7 as double precision), null)), 2)
        from competresult, competdetails
        where cpr_from_cpp_id = :cpp_id and
              cpr_group_name <> '' and
              cpr_object_kind = -1 and -- Refdata, названия групп
              cpd_from_cpr_id = cpr_id and
              cpd_staff_kolich <> 3 and -- было cpd_staff_kolich<>0
              cpr_object_id <> 0 -- Исключаем всех кто вне команды
        group by 1, 2, 3
        order by 4 desc
        into cpr_id, cpr_group_name, res_team_id, avg_delinq_balance_30, -- баланс просрчоки > 7 в среднем
             avg_delinq_balance_30_diff, -- изменение баланса просрчоки > 7 в среднем
             avg_disb_amount, -- сумма выдачи в среднем
             avg_zarabotok_diff, -- заработок СВЗ в среднем
             avg_balance_diff, -- прирост портфеля в среднем
             avg_disb_loan -- среднее значение выполнения плана по выдачам Команды
    do
    begin
      points_1 = 0;
      points_2 = 0;
      points_3 = 0;
      points_4 = 0;
      points_5 = 0;
      --    1. Изменение баланса просрочек более 7 дней  в среднем на 1 СВЗ в Команде
      if (avg_delinq_balance_30 < 800.0) then
        points_1 = 15;
      else
        points_1 = case
                     when avg_delinq_balance_30_diff <= -300 then 8
                     when avg_delinq_balance_30_diff <= 0 then 2
                     when avg_delinq_balance_30_diff <= 300.00 then -2
                     else -10
                   end;
      --    2. Средняя сумма выдачи Команды за декаду на 1 СВЗ (в рублях)
      points_2 = case
                   when avg_disb_loan > 16.0 then 8
                   when avg_disb_loan > 12.0 then 6
                   when avg_disb_loan > 10.0 then 3
                   when avg_disb_loan > 8.0 then 0
                   when avg_disb_loan > 5.0 then -3
                   else -5
                 end;

      if (client_diff_prior <> client_diff) then
      begin
        place_no = place_no + 1 + kol;
        client_diff_prior = client_diff;
        kol = 0;
      end
      else
        kol = kol + 1;

      update competdetails
      set cpd_points_1 = :points_1,
          cpd_points_2 = :points_2,
          cpd_add_data_7 = :avg_disb_loan
      where cpd_from_cpr_id = :cpr_id and

            cpd_staff_kolich <> 3;

    end
    -- Пересчитываем итоги по командам
    for select cpr_id
        from competresult
        where cpr_from_cpp_id = :cpp_id
        into :cpr_id
    do
    begin
      update competdetails
      set cpd_points_all = cpd_points_1 + cpd_points_2 + cpd_points_3 + cpd_points_4 + cpd_points_5 + cpd_points_6 + cpd_points_7 + cpd_points_8 + cpd_points_9
      where cpd_from_cpr_id = :cpr_id;
      -- данная разработка будет действовать в конце месяца
      avg_points = 0.0;
      lo_panelty_count = 0.0;
      select round(avg(cpd_points_all), 2) -- средний балл
      from competdetails
      where cpd_from_cpr_id = :cpr_id and
            cpd_staff_kolich <> 3

      end
    end
    --  Дополнительное основания для начисления балов, пукнт 4,6
    --  Трем командам с наибольшим выдачами в среднем на СВЗ
    cpd_comment = '';
    for select cpr_object_id, avg(cpd_add_data_7)
        from competperiod, competresult, competdetails
        where cpp_from_rd_id = :rd_id and
              cpp_stage = 0 and
              cpp_date_begin >= :res_date_begin and
              cpp_date_end <= :res_date_end and
              cpp_id = cpr_from_cpp_id and
              cpr_object_id <> 0 and -- Исключаем всех кто вне команды
              cpr_group_name <> '' and
              cpr_id = cpd_from_cpr_id and

              cpd_add_data_7 > 0 -- у кого имеется активный баланс
        group by cpr_object_id

        order by 2 desc
        rows 3
        into res_team_id, avg_disb_amount
    do
    begin
      update competresult
      set cpr_additional_points = cpr_additional_points + 3,
      cpr_comment = iif(cpr_comment = '', '', cpr_comment || '; ') || 'Доп.бал наибольшую выдачу, кол-во выдачи на 1 СВЗ:' || round(:avg_disb_amount, 2)
      where cpr_from_cpp_id = :cpp_id and
           cpr_object_id = :res_team_id;
    end
    cpd_comment = '';
    --  Трем командам с наименьшим ПАР7 в процентном соотношении
    for select cpr_object_id, cast(sum(cpd_add_data_1) as double precision) / sum(cpd_add_data_7) * 100
        from competperiod, competresult, competdetails
        where cpp_from_rd_id = :rd_id and
              cpp_stage = 0 and
              cpp_date_begin >= :res_date_begin and
              cpp_date_end <= :res_date_end and
              cpp_id = cpr_from_cpp_id and
              cpr_object_id <> 0 and -- Исключаем всех кто вне команды
              cpr_group_name <> '' and
              cpr_id = cpd_from_cpr_id and

              cpd_add_data_7 > 0 -- у кого имеется активный баланс
        group by cpr_object_id

        order by 2
        rows 3
        into res_team_id, par7
    do
    begin
      update competresult
      set cpr_additional_points = cpr_additional_points + 3,
      cpr_comment = iif(cpr_comment = '', '', cpr_comment || '; ') || 'Доп.бал за наименьший ПАР30:' || round(:par7, 2)
      where cpr_from_cpp_id = :cpp_id and
           cpr_object_id = :res_team_id;
    end
    cpd_comment = '';
    for select cpr.cpr_object_id, sum(cpr.cpr_final_points)
        from competresult cpr
        inner join competperiod cpp on cpp.cpp_id = cpr.cpr_from_cpp_id and
              cpp.cpp_stage = 0 and
              cpr.cpr_object_kind = -1 and
              cpp.cpp_date_end between :res_date_begin and :res_date_end
              group by 1
        into :res_team_id, cpr_final_points
    do
    begin
      update competresult cpr
      set cpr.cpr_calced_points = cpr.cpr_calced_points + :cpr_final_points,
          cpr.cpr_comment = cpr.cpr_comment || 'Баллы по декадам: ' || :cpr_final_points
      where cpr.cpr_object_id = :res_team_id and
            cpr.cpr_from_cpp_id = :cpp_id;
    end
    -- Пересчитываем итоги по командам
    for select distinct cpr_id, --
                        cpd_points_1 + cpd_points_2 + cpd_points_3 + cpd_points_4 + cpd_points_5 + --
                        cpd_points_6 + cpd_points_7 + cpd_points_8 + cpd_points_9 as sum_points
        from competresult cr
        inner join competdetails cd on cr.cpr_id = cd.cpd_from_cpr_id
        where cpr_from_cpp_id = :cpp_id
        into :cpr_id, :cpd_points_all
    do
    begin
      update competdetails
      set cpd_points_all = :cpd_points_all
      where cpd_from_cpr_id = :cpr_id;

      update competresult
      set cpr_calced_points = cpr_calced_points + :cpd_points_all,
          cpr_final_points = :cpd_points_all + cpr_additional_points + cpr_calced_points
      where (cpr_from_cpp_id = :cpp_id and
            cpr_id = :cpr_id);
    end
    --  Команде, победившей по итогам отчетного месяца
    add_points = 1;
    select cpr_id, max(cpr_final_points)
    from competresult
    where cpr_from_cpp_id = :cpp_id
    group by cpr_id
    order by 2 desc
    rows 1
    into :cpr_id, max_points;

    update competresult
    set cpr_additional_points = cpr_additional_points + :add_points

    where cpr_from_cpp_id = :cpp_id and
          cpr_id = :cpr_id;
    update competdetails cpd
    set cpd.cpd_comment = cpd.cpd_comment || iif(cpd.cpd_comment = '', '', '; ') || 'Балл за 1 место: ' || round(:add_points, 0)
    where cpd.cpd_from_cpr_id = :cpr_id;

    execute procedure hp_competition_recalc_order(:cpp_id);
  end

  if (stage = 2) then -- если уровень этапа
  begin
    -- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    for select cpr_object_kind, cpr_object_id, sum(cpr_final_points)
        from competresult, competperiod
        where cpp_from_rd_id = :rd_id and
              cpp_id = cpr_from_cpp_id and
              cpp_stage = (:stage - 1) and
              cpp_date_end between :res_date_begin and :res_date_end
        group by cpr_object_kind, cpr_object_id
        into :kind, :res_team_id, :cpr_final_points
    do
    begin
      cpr_group_name = '';
      -- Получаем название Группы где последний раз была команда
      select cpr_group_name
      from competresult, competperiod
      where cpr_from_cpp_id = cpp_id and
            cpp_from_rd_id = :rd_id and
            cpr_object_kind = -1 and
            cpr_object_id = :res_team_id and
            cpr_group_name <> '' and
            cpp_orderno < :cpp_orderno
      order by cpp_orderno desc
      rows 1
      into :cpr_group_name;
      if (cpr_group_name is null) then
        cpr_group_name = '';

      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
                                cpr_result_kind, cpr_from_un_id)
      values (:cpp_id, :cpr_group_name, :kind, :res_team_id, :cpr_final_points, 0, :un_id)
      returning cpr_id
      into :cpr_id;
    end

    execute procedure hp_competition_recalc_order(:cpp_id);

    update competresult
    set cpr_additional_points = 10,
        cpr_comment = 'Доп. бал за 1 место: 10'
    where cpr_from_cpp_id = :cpp_id and
          cpr_final_orderno = 1 and
          coalesce(cpr_group_name, 'Не заполнен') not in ('', 'Не заполнен');

    update competresult
    set cpr_additional_points = 6,
        cpr_comment = 'Доп. бал за 2 место: 6'
    where cpr_from_cpp_id = :cpp_id and
          cpr_final_orderno = 2 and
          coalesce(cpr_group_name, 'Не заполнен') not in ('', 'Не заполнен');

    update competresult
    set cpr_additional_points = 3,
        cpr_comment = 'Доп. бал за 3 место: 3'
    where cpr_from_cpp_id = :cpp_id and
          cpr_final_orderno = 3 and
          coalesce(cpr_group_name, 'Не заполнен') not in ('', 'Не заполнен');
  end
  if (stage = 3) then -- если финальные итоги
  begin
    -- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    for select cpr_object_kind, cpr_object_id, avg(cpr_final_points)
        from competresult, competperiod
        where cpp_from_rd_id = :rd_id and
              cpp_id = cpr_from_cpp_id and
              cpp_stage = (:stage - 1) and
              cpp_date_end between :res_date_begin and :res_date_end
        group by cpr_object_kind, cpr_object_id
        into :kind, :res_team_id, :cpr_final_points
    do
    begin
      cpr_group_name = '';
      -- Получаем название Группы где последний раз была команда
      select cpr_group_name
      from competresult, competperiod
      where cpr_from_cpp_id = cpp_id and
            cpp_from_rd_id = :rd_id and
            cpr_object_kind = -1 and
            cpr_object_id = :res_team_id and
            cpr_group_name <> '' and
            cpp_orderno < :cpp_orderno
      order by cpp_orderno desc
      rows 1
      into :cpr_group_name;
      if (cpr_group_name is null) then
        cpr_group_name = '';

      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
                                cpr_result_kind, cpr_from_un_id)
      values (:cpp_id, :cpr_group_name, :kind, :res_team_id, :cpr_final_points, 0, :un_id)
      returning cpr_id
      into :cpr_id;
    end
    execute procedure hp_competition_recalc_order(:cpp_id);
  end
  suspend;
end
