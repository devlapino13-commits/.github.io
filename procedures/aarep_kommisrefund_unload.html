CREATE PROCEDURE AAREP_KOMMISREFUND_UNLOAD
RETURNS (
    REG_NAME D_STR_100,
    KONTRACT_NO D_STR_100,
    DISB_DATE D_DATE,
    C_NAME_FIO D_STR_255,
    AMOUNT D_CURRENCY)
AS
declare variable kf_id id_type;
begin
  for select reg.reg_name, k.k_kontract_no, k.k_disb_date, cl.c_name_fio, sum(kmr.kmr_amount)
      from kommisrefund_cards kmr
      inner join kontract k on kmr.kmr_from_k_id = k.k_id
      inner join kontstate ks on ks.ks_from_k_id = k.k_id
            and ks.ks_ol_is_delete = 0
            and current_date between ks.ks_date_begin and ks.ks_date_end
      inner join region reg on ks.ks_region_id = reg.reg_unicode
      inner join clients cl on kmr.kmr_from_c_id = cl.c_id
      group by 1, 2, 3, 4
      into :reg_name, :kontract_no, :disb_date, :c_name_fio, :amount
  do
  begin
    suspend;
  end
end
