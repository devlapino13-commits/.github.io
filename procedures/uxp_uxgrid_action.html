CREATE PROCEDURE UXP$UXGRID_ACTION(
    GRID_UUID D_UUID,
    NEW_NAME D_STR_100,
    NEW_MEMO D_STR_255,
    ACTION_KIND D_SMALLINT = 0,
    GRID_UUID_CODE D_STR_CODE = null)
AS
declare variable un_id d_id;
declare variable owner_un_id d_id;
begin
  if (coalesce(grid_uuid, '') = '') then
    grid_uuid = uxf$uniq(grid_uuid_code);

  if (action_kind = -1) then -- Удаление
  begin
    un_id = uxf$user();

    if (exists(select * from ux$grshare
    where ugs_ugr_uuid = :grid_uuid and
          ugs_un_id = :un_id and
          coalesce(ugs_owner_un_id, ugs_un_id) = :un_id)) then
    begin
      -- Если свой шаблон, удаляем
      delete from ux$grid
      where ugr_uuid = :grid_uuid;
    end else
    begin
      -- Иначе удаляем предоставленный доступ, отписываемся
      delete from ux$grshare
      where ugs_ugr_uuid = :grid_uuid and
            ugs_un_id = :un_id;
    end
  end else
  if (action_kind = 0) then -- Переименование
  begin
    update ux$grid
    set ugr_name = :new_name, 
        ugr_memo = :new_memo
    where ugr_uuid = :grid_uuid;
  end else
  if (action_kind = 1) then -- Поделится с другим пользователем
  begin
    if (coalesce(new_name, '') = '') then
      exception e_message 'Не указан логин пользователя';

    un_id = uxf$user_id(new_name, false);

    if (un_id is null) then
      exception e_message f_format('Не найден пользователь с логином: ', new_name);

    owner_un_id = uxf$user();

    insert into ux$grshare (ugs_ugr_uuid, ugs_un_id, ugs_owner_un_id)
    values (:grid_uuid, :un_id, :owner_un_id);
  end else
  if (action_kind = 2) then -- Назначить владельца
  begin
    if (coalesce(new_name, '') = '') then
      exception e_message 'Не указан логин пользователя';

    un_id = uxf$user_id(new_name, false);

    if (un_id is null) then
      exception e_message f_format('Не найден пользователь с логином: ', new_name);

    owner_un_id = uxf$user();

    insert into ux$grshare (ugs_ugr_uuid, ugs_un_id, ugs_owner_un_id)
    values (:grid_uuid, :un_id, :owner_un_id);

    update ux$grid
    set ugr_owner_user = :new_name
    where ugr_uuid = :grid_uuid;
  end
end
