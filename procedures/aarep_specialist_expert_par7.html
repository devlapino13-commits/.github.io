CREATE PROCEDURE AAREP_SPECIALIST_EXPERT_PAR7(
    REG_UNICODE ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    C_ID ID_TYPE,
    REG_NAME VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    KE_CODE VARCHAR(20),
    KOL_KL INTEGER,
    KOL_KONT INTEGER,
    PAR7 CURR_TYPE)
AS
declare variable reg_id id_type;
declare variable ma_app_id id_type;
declare variable k_id id_type;
declare variable c_portfel curr_type;
declare variable c_portfel_par7 curr_type;
begin
--¬ыбираю мега-офисы
  for select reg_unicode, reg_name
      from region, region_list(:reg_unicode, 1, -1, 1)
      where (reg_logic_label containing '.mega_office.') and
            reg_status = 3 and
            reg_date_close is null and
            reg_unicode = id_list
      into reg_id, reg_name
  do
  begin
    ke_code = null;
    kol_kl = 0;
    kol_kont = 0;
    par7 = 0;

--Ёксперты, количество выданных кредитов
    for select distinct ma_expert_id, count(distinct k_id) kol_kont
        from miniapp ma, kontract k, kontstate
        where ma_is_active = 1 and
              ma.ma_number = 0 and
              ma.ma_id = k_from_ma_app_id and
              k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_region_id = :reg_id and
              ks_date_begin between :rep_date_begin and :rep_date_end and
              ks_kc_id = 1000000000004 and
              ma_expert_id <> 1000000000004
        group by ma_expert_id
        into c_id, kol_kont
    do
    begin
      select count(distinct kc_from_c_id) kol_kl
      from miniapp ma, kontract k, kontstate ks, knclient kc
      where ma_is_active = 1 and
            ma.ma_id = k_from_ma_app_id and
            k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_region_id = :reg_id and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ma_expert_id = :c_id and
            ma.ma_number = 0 and
            k_id = kc_from_k_id and
            kc_amount_approved > 0
      into kol_kl;

      select sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 7, oks_balance, 0)) / sum(oks_balance) * 100 as par7
      from oksummary oks, kontract k, miniapp ma
      where oks_balance > 0 and
            oks_close_state = 0 and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_region_id = :reg_id and
            oks_from_k_id = k_id and
            k_from_ma_app_id = ma_id and
            ma_is_active = 1 and
            ma.ma_number = 0 and
            ma_expert_id = :c_id
      into :par7;

      select c_name_fio
      from clients
      where c_id = :c_id
      into :c_name_fio;

      select ke_code
      from krexpert
      where :rep_date_end between ke_date and ke_date_end and
            ke_from_p_id = :c_id and
            ke_from_reg_id = :reg_id and
            ke_status = 1
      into ke_code;

      suspend;
    end
  end
end
