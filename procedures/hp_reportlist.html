CREATE PROCEDURE HP_REPORTLIST(
    UN_ID DECIMAL(18,0))
RETURNS (
    RL_ID TYPE OF COLUMN REPORTLIST.RL_ID,
    PARENT TYPE OF COLUMN REPORTLIST.PARENT,
    RL_NAME_RUS TYPE OF COLUMN REPORTLIST.RL_NAME_RUS,
    RL_HINT_RUS TYPE OF COLUMN REPORTLIST.RL_HINT_RUS,
    RL_NAME_ENG TYPE OF COLUMN REPORTLIST.RL_NAME_ENG,
    RL_HINT_ENG TYPE OF COLUMN REPORTLIST.RL_HINT_ENG,
    RL_TYPE TYPE OF COLUMN REPORTLIST.RL_TYPE,
    RL_HOTKEY TYPE OF COLUMN REPORTLIST.RL_HOTKEY,
    RL_DATA_RUS TYPE OF COLUMN REPORTLIST.RL_DATA_RUS,
    RL_DATA_ENG TYPE OF COLUMN REPORTLIST.RL_DATA_ENG,
    RL_IS_READY TYPE OF COLUMN REPORTLIST.RL_IS_READY,
    RL_USER_ID TYPE OF COLUMN REPORTLIST.RL_USER_ID,
    RL_DATE_KIND TYPE OF COLUMN REPORTLIST.RL_DATE_KIND,
    RL_REPORT_KIND TYPE OF COLUMN REPORTLIST.RL_REPORT_KIND,
    RL_SQL TYPE OF COLUMN REPORTLIST.RL_SQL,
    RL_FILTER TYPE OF COLUMN REPORTLIST.RL_FILTER,
    RL_ORDER_NO TYPE OF COLUMN REPORTLIST.RL_ORDER_NO,
    RN_ID INTEGER,
    RL_PARENT_NAME VARCHAR(255))
AS
begin
  /* Выдает список отчетов согласно доступа по ролям */
  rn_id = 0;

  for select rl_id, parent, rl_name_rus, rl_hint_rus, rl_name_eng, rl_hint_eng, rl_type, rl_hotkey, rl_data_rus,
             rl_data_eng, rl_is_ready, rl_user_id, rl_date_kind, rl_report_kind, rl_sql, rl_filter, rl_order_no
      from reportlist
      where rl_id = 0
      into :rl_id, :parent, :rl_name_rus, :rl_hint_rus, :rl_name_eng, :rl_hint_eng, :rl_type, :rl_hotkey, :rl_data_rus,
           :rl_data_eng, :rl_is_ready, :rl_user_id, :rl_date_kind, :rl_report_kind, :rl_sql, :rl_filter, :rl_order_no
  do
  begin
    rl_parent_name = '';
    suspend;
  end

  for select rl_id, parent, rl_name_rus, rl_hint_rus, rl_name_eng, rl_hint_eng, rl_type, rl_hotkey, rl_data_rus,
             rl_data_eng, rl_is_ready, rl_user_id, rl_date_kind, rl_report_kind, rl_sql, rl_filter, rl_order_no
      from reportlist, hp_permission_access_list(:un_id, 'Menulist')
      where rl_id = res_subject_id
      order by parent, rl_order_no, rl_id
      into :rl_id, :parent, :rl_name_rus, :rl_hint_rus, :rl_name_eng, :rl_hint_eng, :rl_type, :rl_hotkey, :rl_data_rus,
           :rl_data_eng, :rl_is_ready, :rl_user_id, :rl_date_kind, :rl_report_kind, :rl_sql, :rl_filter, :rl_order_no
  do
  begin
    rl_parent_name = '';
    suspend;
  end
end
