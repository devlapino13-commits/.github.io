CREATE PROCEDURE AAREP_CLI_CATEGORY_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    KF_OFFICER_NAME VARCHAR(100),
    K_KONTRACT_NO VARCHAR(100),
    K_DISB_DATE DATE,
    KNC_NAME_FIO VARCHAR(100),
    IS_CLIENT INTEGER,
    KNC_REP_ACTIVE_BALANCE CURR_TYPE,
    IS_OWN INTEGER,
    AA_CATEGORY_ID INTEGER,
    KAL_DATE DATE,
    IS_COLLECTOR INTEGER,
    KNC_STR_CITIZEN VARCHAR(100))
AS
declare variable ac_id id_type;
declare variable ac_kontract_id id_type;
declare variable ac_region_id id_type;
declare variable ac_kc_id id_type;
declare variable reg_id_prior integer;
declare variable kc_id_prior id_type;
declare variable k_id_prior id_type;
declare variable st_staff integer;
declare variable st_client integer;
declare variable c_citizen integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  reg_id_prior = 0;
  kc_id_prior = 0;
  k_id_prior = 0;
  for select ac_id, ac_disb_date, ac_kontract_id, ac_is_client, ac_region_id, ac_cli_balance, ac_kc_id, ac_kontract_no
      from aarep_cli_active_list(:rep_date_end), region_list(:rep_region_id, 1, 3, 1)
      where ac_region_id = id_list
      order by ac_region_id, ac_kc_id, ac_kontract_id
      into :ac_id, :k_disb_date, :ac_kontract_id, :is_client, :ac_region_id, :knc_rep_active_balance, :ac_kc_id,
           k_kontract_no
  do
  begin
    if (reg_id_prior <> ac_region_id) then
    begin
      reg_id_prior = ac_region_id;
      reg_name = (select reg_code || ' ' || reg_name
                  from region
                  where reg_unicode = :ac_region_id);
    end
    if (kc_id_prior <> ac_kc_id) then
    begin
      kc_id_prior = ac_kc_id;
      kf_officer_name = (select c_name_fio
                         from clients
                         where c_id = :ac_kc_id);
    end

--    knc_name_fio = (select c_name_fio from clients where c_id = :AC_ID);
    knc_name_fio = null;
    c_citizen = null;
    select c_name_fio, c_citizen
    from clients
    where c_id = :ac_id
    into knc_name_fio, c_citizen;

    knc_str_citizen = null;
    select rd_name
    from refdata
    where rd_id = :c_citizen
    into knc_str_citizen;

    if (k_id_prior <> ac_kontract_id) then
    begin
      k_id_prior = ac_kontract_id;
      if (exists(select *
                 from kontstate
                 where ks_ol_is_delete = 0 and
                       ks_state = 1 and
                       :rep_date_end between ks_date_begin and ks_date_end and
                       ks_from_k_id = :ac_kontract_id and
                       ks_kc_id = :ac_kc_id)) then
      begin
        is_own = 1;
      end
      else
      begin
        is_own = 0;
      end

      aa_category_id = null;
      select aa_category_id
      from aarep_get_category_id(:ac_kontract_id, :rep_date_end, null, null, null, null, null, null)
      into aa_category_id;

      if (is_own = 0) then
      begin
        select kal_date
        from knaktlist, kontract
        where knaktlist.is_delete = 0 and
              :rep_date_end between kal_date and kal_date_end and
              kal_from_kf_id = k_from_kf_id and
              k_id = :ac_kontract_id
        into kal_date;
      end
      else
        kal_date = :rep_date_end - extract(day from :rep_date_end);

      is_collector = null;
      select 1
      from sb_akt, sb_list
      where sba_is_delete = 0 and
            sba_from_sbl_id <> 0 and
            sba_status = 2 and
            :rep_date_end between sba_date_begin and sba_date_end and
            sba_from_k_id = :ac_kontract_id and
            sba_from_sbl_id = sbl_id and
            sbl_type = 1
      into is_collector;
      if (is_collector is null) then
        is_collector = 0;

    end

    if (st_client = 0) then
    begin
      knc_name_fio = 'Depersonalized_' || ac_id;
    end
    suspend;
  end
end
