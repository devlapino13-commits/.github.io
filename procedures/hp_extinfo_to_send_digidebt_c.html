CREATE PROCEDURE HP_EXTINFO_TO_SEND_DIGIDEBT_C(
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID,
    C_ID D_ID)
RETURNS (
    JSON_DATA D_STR_8100)
AS
declare variable target d_str_20;
declare variable slc_id d_id;
declare variable address d_str_long;
declare variable sue_kind d_smallint;
begin
  -- Определение подсудности в DigiDebt
  if (f_objkind(object_kind, 'name') not in ('sud_loanclient', 'sud_maincollect')) then
    exception e_default f_format('Unknown object_kind = %1 for slc_id = %2', object_kind, slc_id);

  slc_id = object_id;
  address = (select slc_address from sud_loanclient where slc_id = :slc_id);

  select slc_address, sue_kind
  from sud_loanclient, sud_extinfo, extinfolist
  where slc_id = :slc_id
    and sue_slc_id = slc_id
    and eil_id = sue_from_eil_id
    and eil_status = 0
  rows 1
  into address, sue_kind;

  target = case sue_kind
             when 1 then '["COURT"]'
             when 2 then '["MAGISTRATE"]'
             when 3 then '["BAILIFF"]'
           end;

  if (target is null) then
    exception e_default 'Unknown target for object_id = ' || coalesce(object_id, '<NULL>');

  json_data = '[{' ||
        f_json_make_pair('TARGET', target , 'array') || ',' ||
        f_json_make_pair('ID_CONTRACT', slc_id, 'num') || ',' ||
        f_json_make_pair('ADDRESS', address, 'str') || '}]';
  suspend;
end
