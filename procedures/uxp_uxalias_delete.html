Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXALIAS_DELETE(
    UXA_UUID D_UUID,
    UXA_UUID_CODE D_STR_CODE = null)
AS
declare variable uxa_name d_str_255;
declare variable mm_uuid d_uuid;
declare variable ua_owner d_uuid;
declare variable ua_uuid d_uuid;
begin
  if (coalesce(uxa_uuid, '') = '') then
    uxa_uuid = uxf$uniq(uxa_uuid_code);

  mm_uuid = uxf$mainmenu();

  for select ua_uuid, ua_owner, Upper(ua_name) || ' из формы ' || Upper(ux_name)
  from ux$action, ux$form, ux$code
  where ua_uxa_uuid = :uxa_uuid and
        ua_owner = ufm_uuid and
        ufm_ux_uuid = ux_uuid
  into ua_uuid, ua_owner, uxa_name
  do
  begin
    if (ua_owner = mm_uuid) then
    begin
      if (exists(select * from uxp$actionmenu(:ua_uuid))) then
        exception e_message 'Запрет на удаление, алиас используется: ' || ascii_char(13) || uxa_name;
    end
    else
      exception e_message 'Запрет на удаление, алиас используется: ' || ascii_char(13) || uxa_name;
  end

  delete from ux$alias
  where uxa_uuid = :uxa_uuid;
end
