CREATE PROCEDURE UXP$UXBUGTEST_CREATE(
    UA_UUID D_UUID,
    UA_VER_KIND D_STR_CODE,
    OPEN_UA_UUID D_UUID,
    OPEN_KIND D_SMALLINT,
    UA_PARAMS D_STR_8100,
    JOB_KIND D_SMALLINT,
    JOB_TEXT D_STR_4096,
    JIRA_ID D_STR_CODE,
    USER_NAME D_STR_CODE)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255,
    RES_UA_UUID D_UUID,
    RES_BTS_UUID D_UUID)
AS
declare variable found_count d_smallint;
begin
  res_text = '';
  res_is_error = 0;

  select count(*)
  from ux$bugtest
  where bts_ua_uuid = :ua_uuid and
        bts_kind = :job_kind and
        bts_open_ua_uuid is not distinct from :open_ua_uuid and
        bts_is_done = 0
  into found_count;

  if (found_count > 5) then
  begin
    res_text = 'В БД имеются более 5 актуальных подобных обращений.';
    res_is_error = 1;
    suspend;
  end

  insert into ux$bugtest (bts_ua_uuid, bts_ua_ver_kind, bts_open_ua_uuid, bts_open_kind, bts_ua_params,
                          bts_kind, bts_text, bts_jira_id, bts_create_user_name)
  values (:ua_uuid, :ua_ver_kind, :open_ua_uuid, :open_kind, :ua_params,
          :job_kind, :job_text, :jira_id, :user_name)
  returning bts_uuid
  into res_bts_uuid;

  res_ua_uuid = ua_uuid;
  res_text = 'Ваш запрос зарегистрован в БД вместе с текущими параметрами';

  suspend;
end
