CREATE PROCEDURE HP_PAYSYS_MAKE_OPERBOOK_KG(
    PSM_ID D_ID)
RETURNS (
    OL_ID D_ID)
AS
declare variable psm_date d_date;
declare variable psm_item_count d_integer;
declare variable psm_memo d_str_long;
declare variable psm_from_un_id d_id;
declare variable psp_name d_str_255;
declare variable psl_sign d_smallint;
declare variable reg_id d_integer;
declare variable psp_sname1 d_str_20;
declare variable psp_sname2 d_str_20;
declare variable psp_sname3 d_str_20;
declare variable cu d_str_lang;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable ol_tmp_id d_id;
declare variable ob_detail d_str_long;
declare variable add_detail d_str_long;
declare variable step_num d_integer;
declare variable is_chbook boolean;
begin
  cu = f_get_const('cm_default_currency');
  for
    select
      psm_date,
      psm_item_count,
      psm_memo,
      psm_from_un_id
    from paysys_main
    where psm_id = :psm_id
      and psm_oper_kind = 0
      and psm_status = 1
      and psm_from_ol_id is null
      and psm_is_delete = 0
  into
    :psm_date,
    :psm_item_count,
    :psm_memo,
    :psm_from_un_id
  do
  begin
    psp_sname1 = 'Чековая книжка';
    psp_sname2 = 'ОВН';
    psp_sname3 = 'РКО';

    if (psm_memo not starting 'Закрепление БСО') then
      add_detail = (
        select
          ', ' || list(name, ' -> ')
        from (
          select
            iif(psl_object_kind = 0, 'Эмитент',
              coalesce(reg_name, c_name_fio)) as name
          from paysys_list
            left join region
              on reg_unicode = psl_object_id
            left join clients
              on c_id = psl_object_id
          where psl_from_psm_id = :psm_id
            and psl_id in (1, 2)
          order by psl_id));

    -- откуда psl_sign = -1
    -- куда psl_sign = 1
    step_num = 0;
    for
      select
        psl_object_id,
        psl_sign,
        psp_name,
        psp_name starting :psp_sname1
      from paysys_list
        inner join paysys_item
          on psi_id = psl_from_psi_id
        inner join paysys_project
          on psp_id = psi_from_psp_id
            and (psp_name starting :psp_sname1
             or  psp_name starting :psp_sname2
             or  psp_name starting :psp_sname3)
      where psl_from_psm_id = :psm_id
        and psl_id in (1, 2)
        and psl_object_kind = 1
    into
      :reg_id,
      :psl_sign,
      :psp_name,
      :is_chbook
    do
    begin
      step_num = step_num + 1;
      acc_dt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :cu
        and al_account = iif(:psl_sign = -1, iif(:is_chbook, '96002/3', '96002/2'), iif(:is_chbook, '96001/3', '96001/2')));
      acc_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :cu
        and al_account = iif(:psl_sign = -1, iif(:is_chbook, '96001/3', '96001/2'), iif(:is_chbook, '96002/3', '96002/2')));
      ob_detail = f_format('БСО: %1', psp_name) || coalesce(add_detail, '') || ';';

      insert into operlist (
        ol_parent_id,
        ol_from_reg_id,
        ol_oper_type, 
        ol_oper_subtype,
        ol_date,
        ol_doc_no,
        ol_doc_kind, 
        ol_memo,
        ol_user_id
       )
      values (
        :ol_id,
        :reg_id,
        2,
        11,
        :psm_date,
        :psm_id,
        9,
        :psm_memo,
        :psm_from_un_id
      )
      returning ol_id
      into :ol_tmp_id;

      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_detail,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_tmp_id,
        :cu,
        1,
        :acc_dt,
        :acc_kt,
        :ob_detail,
        :psm_item_count,
        :psm_item_count
      );

      if (step_num = 1) then
        ol_id = ol_tmp_id;
    end

    if (ol_id > 0) then
    begin
      update paysys_main set
        psm_from_ol_id = :ol_id
      where psm_id = :psm_id;
      suspend;
    end
  end
end
