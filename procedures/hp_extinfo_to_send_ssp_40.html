CREATE PROCEDURE HP_EXTINFO_TO_SEND_SSP_40(
    C_ID ID_TYPE,
    OBJECT_KIND INTEGER,
    OBJECT_ID ID_TYPE)
RETURNS (
    JSON_VALUE D_STR_8100)
AS
declare variable lastname str_100;
declare variable firstname str_100;
declare variable middlename str_100;
declare variable birthday str_100;
declare variable birthplace str_100;
declare variable consentdate str_100;
declare variable admcodeinform str_100;
declare variable json_docs d_str_8100;
declare variable json_fio d_str_8100;
declare variable doc_country d_str_10;
declare variable doc_type d_str_10;
declare variable doc_serial d_str_10;
declare variable doc_number d_str_20;
declare variable doc_date d_date;
declare variable doc_who d_str_255;
declare variable request_reason_code d_str_10;
declare variable consent_reason_code d_str_10;
declare variable app_sum decimal(15,2);
declare variable consenthash d_str_64;
declare variable consentperiod d_str_10;
declare variable app_currency d_str_10;
declare variable contractdate d_str_20;
declare variable c_inn d_str_20;
declare variable inn_checked d_bool;
begin
  json_docs = '';
  json_fio = '';

  for select distinct doc_country, doc_type, doc_serial, doc_number, doc_date, doc_who
  from hp_bki_get_info4req_doc(:c_id)
  into :doc_country, :doc_type, :doc_serial, :doc_number, :doc_date, :doc_who
  do
  begin
      json_docs = json_docs || iif(json_docs <> '', ',', '') ||
          '{' ||
              f_json_make_pair('Гражданство', doc_country, 'str', 1) ||
              f_json_make_pair('КодДУЛ', doc_type, 'str', 1) ||
              f_json_make_pair('Серия', doc_serial, 'str', 1) ||
              f_json_make_pair('Номер', doc_number, 'str', 1) ||
              f_json_make_pair('НаименованиеОргана', doc_who, 'str', 1) ||
              f_json_make_pair('ДатаВыдачи', doc_date, 'date') ||
          '}';
  end

  for select lastname, firstname, middlename
  from hp_bki_get_info4req_fio(:c_id, 1)
  into :lastname, :firstname, :middlename
  do
  begin
    json_fio = json_fio || iif(json_fio <> '', ',', '') ||
          '{' ||
              f_json_make_pair('Фамилия',  lastname,   'str', 1) ||
              f_json_make_pair('Имя',      firstname,  'str', 1) ||
              f_json_make_pair('Отчество', middlename, 'str') ||
          '}';
  end 

  select birthday, birthplace, consentdate, admcodeinform, request_reason_code,
         consent_reason_code, app_sum, consenthash, consentperiod, app_currency, contractdate, c_inn, inn_checked
  from hp_bki_get_info4req(:c_id, :object_kind, :object_id)
  into :birthday, :birthplace, :consentdate, :admcodeinform, :request_reason_code,
       :consent_reason_code, :app_sum, :consenthash, :consentperiod, :app_currency, :contractdate, c_inn, inn_checked;

  json_value = '{"Запрос":{' ||
                    '"Субъект":{' ||
                        '"ФИО":[' || json_fio || '],' ||
                        f_json_make_pair('ДатаРождения', birthday, 'str' ,1) ||
--                        f_json_make_pair('МестоРождения', birthplace, 'str', 1)|| --RFCRU-3680
                        trim(iif(coalesce(c_inn, '') <> '' and (current_date >= '01.03.2025' or not exists(select 1 from hp_db_is_actual)),
                            '"ИНН":{' ||
                              f_json_make_pair('value', c_inn, 'str' ,1) ||
                              f_json_make_pair('ПризнакПроверки', inn_checked, 'str') ||
                            '},',
                        ''))||
                        '"ДокументЛичности":[' || json_docs || ']' ||
                    '},' ||
                    '"Цель":[{' ||
                        f_json_make_pair('КодЦели', request_reason_code, 'str') ||
                    '}],' ||
                    '"Согласие":{' ||
                      '"Цель":[{' ||
                          f_json_make_pair('КодЦели', consent_reason_code, 'str') ||
                      '}],' ||
                      iif(contractdate is null, '',
                        '"Договор":{' ||
                            f_json_make_pair('Дата', contractdate, 'str') ||
                        '},') ||
                      f_json_make_pair('ДатаВыдачи',   consentdate,   'str', 1) ||
                      f_json_make_pair('СрокДействия', consentperiod, 'str', 1) ||
                      f_json_make_pair('ХэшКод',   consenthash,   'str', 1) ||
                      f_json_make_pair('ОбОтветственностиПредупрежден', admcodeInform, 'str') ||
                    '},' ||
                    '"СуммаОбязательства":{' ||
                        f_json_make_pair('Валюта', app_currency, 'str', 1) ||
                        f_json_make_pair('value', app_sum, 'str') ||
                    '}' ||
                  '}' ||
               '}';

  suspend;
end
