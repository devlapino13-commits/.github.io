    for select -sum(okf.oko_os), -sum(okf.oko_perc), 'Погашение внесистемный учет',
               okf.oko_ol_date
        from operkroff okf
        where okf.oko_from_k_id = :k_id and
              okf.oko_ol_is_active = 1 and
              okf.oko_ol_date between :rep_date_begin and :rep_date_end and
              okf.oko_type = 4
        group by okf.oko_ol_date
        into :pay_os, :pay_perc, :pay_type, :ol_date
    do
    begin
      if ((pay_os + pay_perc) <> 0) then
        suspend;
    end

    pay_os = 0;
    pay_perc = 0;
    pay_tax = 0;

    for select sum(oe.ox_amount_minus), 'Погашение по судебным издержкам', oe.ox_ol_date
        from operextra oe
        where oe.ox_object_kind = 9 and
              oe.ox_object_id = :k_id and
              oe.ox_ol_date between :rep_date_begin and :rep_date_end and
              oe.ox_ol_is_active = 1 and
              oe.ox_from_rd_id = :oek_rd_id
        group by oe.ox_ol_date
        into :pay_sud, :pay_type, :ol_date
    do
    begin
      if (pay_sud <> 0) then
        suspend;
    end
  end
