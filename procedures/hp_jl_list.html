CREATE PROCEDURE HP_JL_LIST(
    REG_ID INTEGER,
    DD DATE,
    TEXT_STR VARCHAR(100))
RETURNS (
    LIST_REG_ID INTEGER,
    LIST_DEP_ID INTEGER,
    LIST_JL_ID INTEGER,
    LIST_REG_NAME VARCHAR(100),
    LIST_DEP_NAME VARCHAR(100),
    LIST_JL_NAME VARCHAR(100),
    LIST_JK_NAME VARCHAR(100),
    LIST_JOB_TITLE VARCHAR(100))
AS
begin
  text_str = upper(:text_str);
  /* Procedure Text */
  for select distinct jt_from_reg_id, jt_from_dep_id, jt_from_jl_id, reg_name, dep_name, jl_name, jk_name, jtl_name
      from jobtree, joblist, depart, region, jobkind, jobtitle
      where jt_date_begin <= :dd and
            (jt_date_end is null or
            jt_date_end >= :dd) and
            jt_from_reg_id = reg_unicode and
            (:reg_id = 0 or
            jt_from_reg_id = :reg_id) and
            jt_from_dep_id = dep_id and
            jt_from_jl_id = jl_id and
            jl_kind = jk_id and
            jl_title = jtl_id and
            ((jl_name like '%' || :text_str || '%') or
            (reg_name like '%' || :text_str || '%') or
            (dep_name like '%' || :text_str || '%') or
            (jk_name like '%' || :text_str || '%') or
            (jtl_name like '%' || :text_str || '%') or
            (jtl_memo like '%' || :text_str || '%'))
      into :list_reg_id, :list_dep_id, :list_jl_id, :list_reg_name, :list_dep_name, :list_jl_name, :list_jk_name,
           :list_job_title
  do
    suspend;
end
