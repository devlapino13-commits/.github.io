CREATE PROCEDURE HP_UXFRITEM_LIST(
    UFR_CODE D_STR_CODE,
    REPLACE_READONLY D_BOOL)
RETURNS (
    UFI_CODE D_STR_100,
    UFI_PARENT_CODE D_STR_100,
    UFI_UX_CODE D_STR_CODE,
    UFI_UFR_CODE D_STR_CODE,
    UFI_UFP_CODE D_STR_100,
    UFI_KIND D_INTEGER,
    UFI_CONTROL D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_CAPTION D_STR_CODE,
    UFI_CAPTION_ALIGN D_INTEGER,
    UFI_HINT D_STR_LONG,
    UFI_HIDDEN D_BOOL,
    UFI_READONLY D_BOOL,
    UFI_GROUP_DIRECTION D_INTEGER,
    UFI_H_ALIGN D_INTEGER,
    UFI_V_ALIGN D_INTEGER,
    UFI_WIDTH D_INTEGER,
    UFI_HEIGHT D_INTEGER,
    UFI_ADD_PARAMS D_STR_CODE,
    IMAGE_DETAIL D_INTEGER,
    IMAGE_INDEX D_INTEGER,
    RES_CAPTION D_STR_255,
    RES_COMMENT D_STR_LONG,
    ALLOW_CHILD D_BOOL,
    UFI_IS_PARAM D_SMALLINT,
    UFI_ORIGIN_NAME D_STR_CODE,
    UFD_SOURCE D_STR_CODE,
    UFP_REQUIRED D_BOOL,
    UFP_ACCESS_KIND D_SMALLINT)
AS
declare variable tmp_parent d_str_100;
declare variable main_table d_str_code;
declare variable parent_is_readonly d_bool;
begin
  if (replace_readonly is null) then
    replace_readonly = 0;

  for select ufi_code, ufi_parent_code, ufi_ux_code, ufi_ufr_code, ufi_ufp_code, ufi_kind, ufi_control, ufi_position, ufi_caption,
             ufi_caption_align, ufi_hint, ufi_hidden, ufi_readonly, ufi_group_direction, ufi_h_align, ufi_v_align,
             ufi_width, ufi_height, ufi_add_params, ux_table_name
  from uxfritem, uxcode
  where ufi_ux_code = :ufr_code and
        ufi_parent_code is null and
        ux_code = ufi_ux_code
  order by ufi_position
  into :ufi_code, :ufi_parent_code, :ufi_ux_code, :ufi_ufr_code, :ufi_ufp_code, :ufi_kind, :ufi_control, :ufi_position,
       :ufi_caption, :ufi_caption_align, :ufi_hint, :ufi_hidden, :ufi_readonly, :ufi_group_direction, :ufi_h_align,
       :ufi_v_align, :ufi_width, :ufi_height, :ufi_add_params, :main_table
  do
  begin
    if (ufi_kind = -2) then -- Форма
    begin
      image_detail = 19;
      image_index = 19;
    end else
    begin
      image_detail = 17;
      image_index = 17;
    end

    allow_child = 1;
    ufd_source = '';
    ufi_is_param = 0;
    ufi_origin_name = '';
    ufi_is_param = -1;

    res_comment = ufi_code;
    if (coalesce(ufi_hint, '') <> '')
    then res_comment = res_comment || '  - ' || ufi_hint;

    res_caption = 'Элементы интерфейса';

    suspend;

    parent_is_readonly = ufi_readonly;

    tmp_parent = ufi_code;
    for select ufi_code, ufi_parent_code, ufi_ufr_code, ufi_ux_code, ufi_ufp_code, ufi_kind, ufi_control, ufi_position, ufi_caption,
               ufi_caption_align, ufi_hint, ufi_hidden, iif(:replace_readonly = 1 and :parent_is_readonly = 1, 1, ufi_readonly),
               ufi_group_direction, ufi_h_align, ufi_v_align,
               ufi_width, ufi_height, ufi_add_params, allow_child,
               ufp_is_param, ufp_origin_name, ufd_source, ufp_required, ufp_access_kind
        from hp_uxfritem_tree(:tmp_parent, :replace_readonly)
        into :ufi_code, :ufi_parent_code, :ufi_ufr_code, :ufi_ux_code, :ufi_ufp_code, :ufi_kind, :ufi_control, :ufi_position,
             :ufi_caption, :ufi_caption_align, :ufi_hint, :ufi_hidden, :ufi_readonly,
             :ufi_group_direction, :ufi_h_align,
             :ufi_v_align, :ufi_width, :ufi_height, :ufi_add_params, :allow_child,
             :ufi_is_param, :ufi_origin_name, :ufd_source, :ufp_required, :ufp_access_kind
    do
    begin
      image_detail = 1;
      if (ufi_kind = 0) then
      begin
        select editor_image
        from hp_uxfrparam_editors
        where editor_id = :ufi_control
        into image_detail;
      end else
      begin
        image_detail = case ufi_control
          when 'group' then 0
          when 'label' then 2
          when 'space' then 3
          when 'grid' then 12
          when 'vgrid' then 13
          when 'tree' then 14
          when 'separator' then 15
          when 'splitter' then 16
          when 'frame' then 17
          when 'form' then 19
          else 1
        end;
      end

      image_index = case ufi_control
        when 'group' then 0
        when 'label' then 2
        when 'space' then 3
        when 'grid' then 12
        when 'vgrid' then 13
        when 'tree' then 14
        when 'separator' then 15
        when 'splitter' then 16
        when 'frame' then 17
        when 'form' then 19
        else 1
      end;

      if (ufi_ufr_code is not null) then
      begin
        if (not exists(select * from uxcode
        where ux_code = :ufi_ufr_code and
              ux_table_name = :main_table)) then
        begin
          image_detail = 18;
          image_index = 18;
        end
      end

      if (ufi_kind <> 0) then
        ufi_is_param = -1;
  
      res_comment = ufi_caption;
      if (coalesce(ufi_hint, '') <> '')
      then res_comment = res_comment || '  - ' || ufi_hint;
  
      res_caption = ufi_caption;
      if (coalesce(ufi_origin_name, '') <> '' and ufi_origin_name <> res_caption) then
        res_caption = res_caption || '  (' || ufi_origin_name || ')';

      suspend;
    end
  end

end
