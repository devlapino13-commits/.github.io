CREATE PROCEDURE HP_COM_PACKET_ACTION(
    CURR_CMP_ID D_INTEGER,
    ACTION_KIND D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    CMP_ID D_INTEGER,
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable signing_status d_integer;
declare variable signing_count d_integer;
declare variable cml_id d_id;
declare variable tmp_text d_str_255;
declare variable pdn_sent_status d_integer;
declare variable pdn_signed_status d_integer;
begin
  res_is_error = 1;
  res_text = '';

  if (coalesce(action_kind, '') = '') then
    exception e_default 'Action kind is empty';

  if (action_kind not in ('create', 'delete', 'close')) then
    exception e_default 'Unknown action kind: ' || action_kind;

  cmp_id = curr_cmp_id;

  if (action_kind = 'create') then
  begin
    insert into com_packet (cmp_date)
    values (current_date)
    returning cmp_id
    into :cmp_id;
  
    select un_from_c_id
    from username
    where un_id = :un_id
    into c_id;
  
    if (c_id is not null) then
    begin
      insert into com_staff (cms_from_cmp_id, cms_from_c_id, cms_status)
      values (:cmp_id, :c_id, 2);
    end

    res_is_error = 0;
    res_text = 'Новое заседание успешно создано';
  end
  
  if (action_kind = 'delete') then
  begin
    if (exists(select * from com_action
               where cma_from_cmp_id = :curr_cmp_id)) then
      res_text = 'Имеются ссылки на удаляемое заседание. Удаление отменяется.';

    if (res_text = '') then
    begin
      delete from com_packet
      where cmp_id = :curr_cmp_id;

      res_is_error = 0;
      res_text = 'Заседание успешно удалено';
    end
  end

  if (action_kind = 'close') then
  begin
    signing_status = f_enum('com_action_status', 'signing', 'id');
    pdn_sent_status = f_enum('com_action_status', 'send_edoc_pdn', 'id');
    pdn_signed_status = f_enum('com_action_status', 'signed_edoc_pdn', 'id');

    signing_count = 0;
    select count(*)
    from com_list
    where cml_last_cmp_id = :curr_cmp_id and
          cml_last_status in (:signing_status, :pdn_sent_status, :pdn_signed_status) and
          cml_date >= current_date
    into signing_count;

    if (signing_count > 0) then
    begin
      res_is_error = 1;
      res_text = f_format('Имеются запросы в ожидании подписания: %1', signing_count);
    end else
    begin
      res_is_error = 0;
      signing_count = 0;

      for select cml_id
      from com_list
      where cml_last_cmp_id = :curr_cmp_id and
            cml_last_status in (:signing_status, :pdn_sent_status, :pdn_signed_status) and
            cml_date < current_date
      into cml_id
      do
      begin
        select res_text, res_is_error
        from hp_com_action_add(:cml_id, :curr_cmp_id, 'cancel_edoc', :un_id)
        into :res_text, :res_is_error;

        if (res_is_error = 0) then
        begin
          select res_text, res_is_error
          from hp_com_action_add(:cml_id, :curr_cmp_id, 'cancelled', :un_id)
          into :res_text, :res_is_error;
        end

        if (res_is_error = 0) then
        begin
          signing_count = signing_count + 1;
          select res_text
          from hp_com_memo_add(:cml_id, 'Истек срок подписания', 1, :un_id)
          into :tmp_text;
        end

        if (res_is_error = 1) then
          Break;
      end

      if (res_is_error = 0) then
      begin
        update com_packet
        set cmp_status = 1
        where cmp_id = :curr_cmp_id;

        if (signing_count = 0)
        then res_text = 'Заседание успешно завершено';
        else res_text = f_format('Заседание успешно завершено. Аннулированы из-за истечения срока подписания: %1', signing_count);
      end
    end
  end

  suspend;
end
