CREATE PROCEDURE AAREP_INFO_OFFICE(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RES_REGION_ID INTEGER,
    SVZ_KOL_ACT INTEGER,
    SVZ_DISB INTEGER,
    DISB_KOL_KRED INTEGER,
    DISB_KOL_PARALEL INTEGER,
    DISB_KOL_VSEGO INTEGER,
    SVZ_ONE_DISB CURR_TYPE,
    REP_BALANCE CURR_TYPE,
    DELINQ_BALANCE_1 CURR_TYPE,
    DELINQ_BALANCE_7 CURR_TYPE,
    DELINQ_BALANCE_30 CURR_TYPE,
    PAR1 CURR_TYPE,
    PAR7 CURR_TYPE,
    PAR30 CURR_TYPE,
    KOL_CLOSE_KRED INTEGER,
    KOL_DISB_KRED INTEGER,
    PERC_UDERJ CURR_TYPE,
    REP_BALANCE_LAST CURR_TYPE,
    PRIROST_PORTFEL CURR_TYPE,
    KOL_CLI INTEGER,
    KOL_CLI_LAST INTEGER,
    PRIROST_CLI INTEGER)
AS
begin
  for select res_region_id, sum(svz_kol_act), sum(svz_disb), sum(disb_kol_kred), sum(disb_kol_paralel),
             sum(disb_kol_vsego), sum(svz_one_disb), sum(rep_balance), sum(delinq_balance_1), sum(delinq_balance_7),
             sum(delinq_balance_30), sum(par1), sum(par7), sum(par30), sum(kol_close_kred), sum(kol_disb_kred),
             sum(perc_uderj), sum(rep_balance_last), sum(prirost_portfel), sum(kol_cli), sum(kol_cli_last),
             sum(prirost_cli)
      from aarep_info_office_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
      group by res_region_id
      into :res_region_id, :svz_kol_act, :svz_disb, :disb_kol_kred, :disb_kol_paralel, :disb_kol_vsego, :svz_one_disb,
           :rep_balance, :delinq_balance_1, :delinq_balance_7, :delinq_balance_30, :par1, :par7, :par30,
           :kol_close_kred, :kol_disb_kred, :perc_uderj, :rep_balance_last, :prirost_portfel, :kol_cli, :kol_cli_last,
           :prirost_cli
  do
  begin
    prirost_portfel = rep_balance_last - rep_balance;
    prirost_cli = kol_cli_last - kol_cli;
    suspend;
  end
end
