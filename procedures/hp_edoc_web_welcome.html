CREATE PROCEDURE HP_EDOC_WEB_WELCOME(
    UUID D_STR_CODE)
RETURNS (
    ASK_PIN D_BOOL,
    MSG_TEXT D_STR_255)
AS
declare variable obj_code d_str_code;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable status d_integer;
begin
  -- Возвращает параметры авторизации на странице
  -- Вызывается до авторизации
  -- ask_pin - запрашивать-1, не запрашивать-0 pin-код для авторизации
  -- msg_text возвращается в случае ошибки

  ask_pin = 1;
  obj_code = '';
  msg_text = '';

  select obj_kind, obj_id
  from hp_uuid(:uuid)
  into obj_kind, obj_id;

  if (obj_id is null) then
    msg_text = 'Неверная ссылка';
  else
    obj_code = f_objkind(obj_kind, 'name');

  if (msg_text = '' and obj_code = 'edoc_sign') then
  begin
    select iif(coalesce(edl_date_end, current_date) < current_date, -1, edl_status), edc_need_pin
    from edoc_sign, edoc_list, edoc_cfg
    where eds_id = :obj_id and
          eds_from_edl_id = edl_id and
          edl_from_edc_id = edc_id
    into status, ask_pin;
  
    if (not exists(select * from edoc_msg
                   where edm_from_eds_id = :obj_id)) then msg_text = 'Неверная ссылка';

    if (msg_text = '' and status = -1) then msg_text = 'Данные устарели';
  end

  if (msg_text = '' and obj_code = 'contverify') then
  begin
    select cv_status
    from contverify
    where cv_id = :obj_id
    into status;
  
    if (status >= 1) then
      ask_pin = 0;
  end

  if (msg_text = '' and obj_code = 'contveri') then
  begin
    select cvr_status
    from contveri
    where cvr_id = :obj_id
    into status;
  
    if (status = 1) then
      ask_pin = 0;
  end

  if (msg_text = '') then msg_text = null;

  suspend;
end
