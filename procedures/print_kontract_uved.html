CREATE PROCEDURE PRINT_KONTRACT_UVED(
    K_ID D_ID)
RETURNS (
    K_KONTRACT_NO D_STR_100,
    C_NAME_FIO D_STR_255,
    ZAD_OS D_CURRENCY,
    SH_CALC_PERC D_CURRENCY,
    TOTAL_SUM D_CURRENCY)
AS
declare variable tax_perc d_currency;
declare variable k_disb_date d_date;
begin
  for select k.k_kontract_no, --
             c.c_name_fio, --
             k.k_disb_date
      from kontract k
      inner join knclient kc on kc.kc_from_k_id = k.k_id and
            kc.kc_amount_approved > 0
      inner join clients c on c.c_id = kc.kc_from_c_id
      where k.k_id = :k_id
      into k_kontract_no, --
           c_name_fio, --
           k_disb_date
  do
  begin
    select nsp_percent_for_int
    from hp_get_kontract_nsp_acc(0, :k_disb_date)
    into :tax_perc;

    select sum(zad_os), --
           sum(sh_calc_perc), --
           sum(zad_os + sh_calc_perc )
    from aarep_make_zad_list(:k_id, '31.12.2022', 1)
    into zad_os, --
         sh_calc_perc, --
         total_sum;
    suspend;
  end
end
