             :deliq_unpaidfine, :op_cred_date_nextpayout, :ta_cred_sum_nextpayout, :ta_cred_date_nextpayout,
             :ta_cred_sum_debt, :ta_cred_sum_overdue, :op_cred_sum_paid, :ta_cred_sum_paid, :op_cred_sum_nextpayout,
             :ta_cred_date_overdue, :cred_date_missedpayout, :cder_active_date, :c_id, :cred_sum, :kc_number,
             :kontract_no_orig, :type_, :cred_no, :is_poruch, :cred_account_no, :cred_id, cred_facility, consent,
             cred_sum_limit, cred_insured, guarantee_volume, cred_collateral, cred_way, consentdate, cred_full_cost,
             return_court_data, court_no, court_number, court_source, court_plaintiff, court_resolution, cred_type,
             cred_amount_frequency, k_kolich_client_approved, guarantee_cred_type,
             subject, collateral_id, collateral_no,  collateral_date, collateral_enddate, assessment, assessment_date,
             show_collateral, assessment_currency, principal_amount_olpd, interest_amount_olpd, other_amount_olpd, principal_amount_plpd, interest_amount_plpd, other_amount_plpd,
             kc_id, uid, rep_date_end, is_payed, guarantee_cred_no, guarantee_list_check, op_cred_debt, ta_cred_debt, is_restruct, is_perc_changed, pocket_type,
             guarantee_uid
    do
    begin
      if ((pocket_type containing ('.only_guarantee.')) and (cred_type <> 90)) then continue;
      if ((pocket_type containing ('.only_main.')) and (cred_type = 90)) then continue;

      if (c_id <> c_id_old) then
      begin
        c_id_old = c_id;

        for select cl.c_id,
                   cl.c_doc_type,
                   cl.c_pass_no,

                   cl.c_pass_date,
                   cl.c_pass_date_expire,
                   coalesce(cl.c_pass_who, ' '),

                   cl.c_citizen,
                   cl.c_name_fio,
                   cl.c_sex,

                   cl.c_birth_date,
                   cl.c_inn,
                   trim(coalesce(cl.c_birth_country, '')),

                   trim(coalesce(cl.c_birth_city, '')),
                   trim(coalesce(cl.c_pass_who_code, '')),
                   trim(cl.c_name_eng)
            from clients cl
            where cl.c_id = :c_id
            into :c_id,
                 :c_doc_type,
                 :docno, -- Серия и номер документа

                 :docdate, -- Дата выдачи документа
                 :docenddate,
                 :docplace, -- Место выдачи документа и орган

                 :c_citizen,
                 :c_name_fio,
                 :c_sex,

                 :birthday,
                 :inn,
                 :c_birth_country,

                 :c_birth_city,
                 :docwho,
                 :c_name_eng
        do
        begin
          -- если выгружаем за исторический период, может оказаться, что текущий паспорт был прикреплен уже поле выдачи исторического займа
          -- в таком случае нужно подтянуть старый паспорт, актуальный на момент выдачи займа
          if (docdate > cred_date) then
          begin
            select cds.cds_from_pk_id,
                   cds.cds_doc_no,
                   cds.cds_doc_date_begin,
                   cds.cds_doc_date_end,
                   cds.cds_doc_issued_by
            from clientdocs cds
            where cds.cds_from_c_id = :c_id
            and cds.cds_is_main_doc = 1
            and cds.cds_verify_status = 2
            and cds.cds_doc_date_begin <= :cred_date
            order by cds.cds_id desc
            rows 1
            into :c_doc_type,
                 :docno, -- Серия и номер документа
                 :docdate, -- Дата выдачи документа
                 :docenddate,
                 :docplace; -- Место выдачи документа и орган
          end

          -- По новым клиентам с паспортами гр. Киргизии, Узбекистана и Таджикистана отправку данных о ФИО необходимо формировать на латинице
          -- (согласно ст. 4, Федеральный закон от 30.12.2004 N 218-ФЗ (ред. от 03.08.2018) "О кредитных историях" ).
          if (c_citizen <> 13) then -- если не гражданин РФ
            c_name_fio = c_name_eng;

          select out_string
          from hp_check_characters_str(:c_birth_city, '[ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyzа-я0-9@-@+@(@)@_@?#\/.,№!:;"'' ]+')
          into :c_birth_city;

          select out_string
          from hp_check_characters_str(:docplace, '[ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyzа-я0-9@-@+@(@)@_@?#\/.,№!:;"'' ]+')
          into :docplace;

          --docplace =  replace(docplace, ascii_char(13), '');
          --docplace =  replace(docplace, ascii_char(10), '');

          number = c_id;

          birthplace = c_birth_city;
          if (c_birth_country <> '') then
            birthplace = c_birth_country;

          birthplace = trim(coalesce(birthplace, ''));

          if (birthplace = '') then -- << помни!
          begin
            select ddr.rd_name
            from refdata ddr
            where ddr.rd_id = :c_citizen
            into :birthplace; --  мето роджения беру исходя из гражданства
          end

          if (c_citizen = 13) then
            citizenship = 'RU';
          else
          if (c_citizen = 9) then
            citizenship = 'UZ';
          else
          if (c_citizen = 12) then
            citizenship = 'TJ';
          else
          if (c_citizen = 8) then
            citizenship = 'KG';
          else
            citizenship = '99'; --99 - Неизвестно

          if (c_citizen <> 13) then -- передаю ИНН только для граждан РФ
            inn = '';

          select lower(txt1), lower(txt2), lower(txt3)
          from hp_fio_clear(:c_name_fio)
          into :lastname, :firstname, :middlename;
        
          firstname =coalesce(firstname, ''); -- Имя
          lastname = coalesce(lastname, '');  -- фамилия
          middlename = coalesce(middlename, ''); -- Отчетво
        
          if (c_name_fio like '% уулу %') then lastname = lastname || ' уулу';
          if (c_name_fio like '% кызы %') then lastname = lastname || ' кызы';
          if (c_name_fio like '% uulu %') then lastname = lastname || ' uulu';
          if (c_name_fio like '% kyzy %') then lastname = lastname || ' kyzy';
          if (c_name_fio like '% угли %') then lastname = lastname || ' угли';
          if (c_name_fio like '% кизи %') then lastname = lastname || ' кизи';
          if (c_name_fio like '% qizi %') then lastname = lastname || ' qizi';
          if (c_name_fio like '% oglu %') then lastname = lastname || ' oglu';
          if (c_name_fio like '% ugli %') then lastname = lastname || ' ugli';

          if (firstname <> '') then
              firstname = Upper(substring(firstname from 1 for 1))||substring(firstname from 2);

          if (lastname <> '') then
              lastname = Upper(substring(lastname from 1 for 1))||substring(lastname from 2);

          if (middlename <> '') then
              middlename = Upper(substring(middlename from 1 for 1))||substring(middlename from 2);

          if (c_sex = 0) then
            sex_code = 2;
          else -- женский
          if (c_sex = 1) then
            sex_code = 1;
          else -- мужской
            sex_code = 9; -- неизвестно

          select pask.pk_id
          from passkind pask
          where pask.pk_id = :c_doc_type
          into :pk_id;

          if (c_citizen = 13) then -- Рф
          begin
            if ((pk_id = 12) or (pk_id = 13)) then
              doctype = 1; -- Паспорт гражданина РФ
            if ((pk_id <> 12) and
                (pk_id <> 13)) then
              doctype = 14; -- Неизвестно
          end
          else
            doctype = 6; -- Паспорт, выданный иностранным государством и признаваемый в Российской Федерации

          for select adcl.adc_from_adl_id, adcl.adc_address_type, adcl.adc_is_rent,
                     adl.adl_flat_no, adl.adl_depart_no,
                     adl.adl_section_no, adl.adl_building_no, adl.adl_street_id,
                      adl.adl_from_pl_id, adl.adl_post_index, adl.adl_id, adl.adl_address_str,
                     substring(cast(adl.adl_id as str_255) from 1 for 3), adl.adl_is_edited
              from addrclient adcl
              inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
              where adcl.adc_from_c_id = :c_id and
                    adcl.adc_address_type in (1, 2)
              into :adl_id, :adc_type, :adc_is_rent,
                   :adl_flat_no, :adl_depart_no,
                   :adl_section_no, :adl_building_no, :adl_street_id,
                   :adl_from_pl_id, :adl_post_index, :adl_id, :adl_address_str, :republ_code, :adl_is_edited
          do
          begin
            if (adc_type = 1) then -- место регистрации
            begin
              ar_adl_is_edited = adl_is_edited;

              ar_adl_id = adl_id;
              ar_address_str = adl_address_str;

              if (adc_is_rent = 0) then
                ar_owner = 3; -- аренда
              else
                ar_owner = 9; -- не известно

              if (republ_code = '417') then
                ar_country = 'KG';
              else
              if (republ_code = '810') then
                ar_country = 'RU';
              else
              if (republ_code = '860') then
                ar_country = 'UZ';
              else
              if (republ_code = '972') then
                ar_country = 'TJ';
              else
                ar_country = '99';

              pl_fs_code = null;
              ar_city = null;
              plt_name = null;

              pl_post_index = '';
              select substring(pl_code from 4 for 2), pl.pl_name, pl.pl_post_index
              from places pl
              where pl.pl_id = :adl_from_pl_id
              into :pl_fs_code, :ar_city, :pl_post_index;
              if (coalesce(pl_post_index, '') = '') then
                pl_post_index = '0';

              ar_region = '00';
              if (republ_code = '810') then
              begin
                select rcr.rcr_cib_code
                from rcib_region rcr
                where rcr.rcr_fs_code = coalesce(:pl_fs_code, '00')
                into :ar_region; -- Код региона РФ по месту регистрации
              end

              street_post_index = '';
              street_buf = null;
              plt_name = null;
              select pl.pl_name, pt.plt_name, pl.pl_post_index
              from places pl
              inner join placetype pt on pl.pl_status = pt.plt_id
              where pl.pl_id = :adl_street_id
              into :street_buf, :plt_name, :street_post_index;
              if (coalesce(street_post_index, '') = '') then
                street_post_index = '0';

              -- по умолчанию наибольший приоритет отдаю почтовому индексу, указанному на уровне улицы
              adl_post_index = coalesce(adl_post_index, 0);
              if (street_post_index <> '0') then
                adl_post_index = cast(street_post_index as integer);
              else
              if (pl_post_index <> '0') then
                adl_post_index = cast(pl_post_index as integer);
              ar_index = adl_post_index; -- Почтовый индекс места жительства/нахождения

              if (plt_name in ('улица', 'проспект')) then
                ar_street = street_buf; -- Наименование улицы
              if (plt_name in ('район', 'микрорайон')) then
                ar_district = street_buf; -- Наименование района

              ar_house_no = adl_depart_no; -- Номер дома места регистрации
              ar_build_no = adl_building_no; -- Номер строения места регистрации
              ar_block_no = adl_section_no; -- Номер корпуса места регистрации
              ar_apartment_no = adl_flat_no; -- Номер квартиры/комнаты/офиса места регистрации

            end

            if (adc_type = 2) then -- адрес выдачи
            begin
              af_adl_is_edited = adl_is_edited;

              af_adl_id = adl_id;
              af_address_str = adl_address_str;

              if (adc_is_rent = 0) then
                af_owner = 3; -- аренда
              else
                af_owner = 9; -- не известно

              if (republ_code = '417') then
                af_country = 'KG';
              else
              if (republ_code = '810') then
                af_country = 'RU';
              else
              if (republ_code = '860') then
                af_country = 'UZ';
              else
              if (republ_code = '972') then
                af_country = 'TJ';
              else
                af_country = '99';

              pl_fs_code = null;
              af_city = null;
              plt_name = null;

              pl_post_index = '';
              select substring(pl_code from 4 for 2), pl.pl_name, pl_post_index
              from places pl
              inner join placetype pt on pl.pl_status = pt.plt_id
              where pl.pl_id = :adl_from_pl_id
              into :pl_fs_code, :af_city, :pl_post_index;
              if (coalesce(pl_post_index, '') = '') then
                pl_post_index = '0';

              af_region = '00';
              if (republ_code = '810') then
              begin
                select rcr.rcr_cib_code
                from rcib_region rcr
                where rcr.rcr_fs_code = :pl_fs_code
                into :af_region; -- Код региона РФ по месту регистрации
              end

              street_post_index = '';
              street_buf = null;
              plt_name = null;
              select pl.pl_name, pt.plt_name, pl.pl_post_index
              from places pl
              inner join placetype pt on pl.pl_status = pt.plt_id
              where pl.pl_id = :adl_street_id
              into :street_buf, :plt_name, :street_post_index;
              if (coalesce(street_post_index, '') = '') then
                street_post_index = '0';

              adl_post_index = coalesce(adl_post_index, 0);
              if (street_post_index <> '0') then
                adl_post_index = cast(street_post_index as integer);
              else
              if (pl_post_index <> '0') then
                adl_post_index = cast(pl_post_index as integer);
              af_index = adl_post_index; -- Почтовый индекс места жительства/нахождения

              if (plt_name in ('улица', 'проспект')) then
                af_street = street_buf; -- Наименование улицы
              if (plt_name in ('район', 'микрорайон')) then
                af_district = street_buf; -- Наименование района

              af_house_no = adl_depart_no; -- Номер дома места регистрации
              af_build_no = adl_building_no; -- Номер строения места регистрации
              af_block_no = adl_section_no; -- Номер корпуса места регистрации
              af_apartment_no = adl_flat_no; -- Номер квартиры/комнаты/офиса места регистрации
            end

          end

          middlename = coalesce(middlename, '');

          if (char_length(ar_index) > 6) then
            ar_index_cib = substring(ar_index from 1 for 6);
          else
            ar_index_cib = ar_index;
          if (coalesce(ar_index_cib, '0') = '0') then
            ar_index_cib = '117335'; --'000000';

          if (char_length(af_index) > 6) then
            af_index_cib = substring(af_index from 1 for 6);
          else
            af_index_cib = af_index;
          if (coalesce(af_index_cib, '0') = '0') then
            af_index_cib = '117335'; --'000000';

          ar_house_no_cib = ar_house_no;
          if ((coalesce(ar_house_no_cib, '') = '') or (ar_house_no_cib = '0')) then
            ar_house_no_cib = 'Нет данных';

          af_house_no_cib = af_house_no;
          if ((coalesce(af_house_no_cib, '') = '') or (af_house_no_cib = '0')) then
            af_house_no_cib = 'Нет данных';

          ar_apartment_no_cib = ar_apartment_no;
          if ((coalesce(ar_apartment_no_cib, '') = '') or (ar_apartment_no_cib = '0')) then
            ar_apartment_no_cib = 'Нет данных';

          af_apartment_no_cib = af_apartment_no;
          if ((coalesce(af_apartment_no_cib, '') = '') or (af_apartment_no_cib = '0')) then
            af_apartment_no_cib = 'Нет данных';

          if (coalesce(ar_owner, 0) = 0) then
            ar_owner = 9;
          if (coalesce(af_owner, 0) = 0) then
            af_owner = 9;
          if (coalesce(ar_country, '') = '') then
            ar_country = '99';
          if (coalesce(af_country, '') = '') then
            af_country = '99';
          if (coalesce(ar_region, '') = '') then
            ar_region = '00';
          if (coalesce(af_region, '') = '') then
            af_region = '00';
          if (coalesce(ar_city, '') = '') then
            ar_city = 'Нет данных';
          if (coalesce(af_city, '') = '') then
            af_city = 'Нет данных';
          if (coalesce(ar_street, '') = '') then
            ar_street = '';
          if (coalesce(af_street, '') = '') then
            af_street = '';

          snils = null;

          if (c_citizen = 13) then -- если РФ
          begin
            select cds_doc_no
            from clientdocs, passkind, passdoctype
            where cds_from_c_id = :c_id and
                  cds_from_pk_id = pk_id and
                  pk_from_pdt_id = pdt_id and
                  pdt_code = 'snils' and
                  cds_is_delete = 0
            order by cds_from_c_id desc, cds_doc_date_begin desc
            rows 1
            into :snils;
          end

          select out_string
          from hp_check_characters_str(:snils, '[-]?[0-9]+')
          into :snils;
        end
      end
      suspend;
    end
  end
