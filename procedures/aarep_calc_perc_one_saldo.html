CREATE PROCEDURE AAREP_CALC_PERC_ONE_SALDO(
    K_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RESULT_PAYED D_CURRENCY,
    RESULT_WRITEOFF D_CURRENCY,
    RESULT_PERC_END D_CURRENCY,
    RESULT_OUT_PERC_END D_CURRENCY,
    RESULT_BALANCE_END D_CURRENCY,
    RESULT_CALCED_ALL D_CURRENCY,
    RESULT_PAYED_ALL D_CURRENCY,
    RESULT_WRITEOFF_ALL D_CURRENCY,
    RESULT_DISB_AMOUNT D_CURRENCY,
    RESULT_DISB_DATE D_DATE,
    RESULT_VAC_PERC_SALDO D_CURRENCY)
AS
declare variable pr_percent_kind d_smallint;
declare variable db_id d_integer;
declare variable pr_interest_method d_smallint;
declare variable k_stop_perc d_date;
declare variable stop_days d_integer;
declare variable kont_last_date d_date;
declare variable k_kredline_end d_date;
declare variable dd_0 d_date;
begin
  db_id = f_get_db_id();

  result_perc_end = 0;
  result_out_perc_end = 0;
  result_payed = 0;
  result_writeoff = 0;
  result_balance_end = 0;
  result_payed_all = 0;
  result_writeoff_all = 0;
  result_calced_all = 0;
  result_disb_amount = 0;
  result_disb_date = null;
  result_vac_perc_saldo = 0;

  if (rep_date_begin > rep_date_end) then
  begin
    suspend;
    exit;
  end

  select k_disb_date, k_amount_approved, k_stop_perc, k_kredline_end, pr_interest_method, pr_percent_kind
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into :result_disb_date, :result_disb_amount, k_stop_perc, k_kredline_end, pr_interest_method, pr_percent_kind;

  if (pr_percent_kind = 2) then
  begin
    -- Новый вариант начисления процентов по ИФП продуктам
    -- Вместо использования ставки процента применяется метод амортизации процента
    select result_payed, result_writeoff, result_perc_end, result_balance_end,
           result_calced_all, result_payed_all, result_writeoff_all, result_disb_amount, result_disb_date,
           result_out_perc_end
    from aarep_calc_perc_one_ipf(:k_id, :rep_date_begin, :rep_date_end)
    into :result_payed, :result_writeoff, :result_perc_end, :result_balance_end,
         :result_calced_all, :result_payed_all, :result_writeoff_all, :result_disb_amount, :result_disb_date,
         :result_out_perc_end;
  end
  else
  if (pr_interest_method in (1, 2) and    -- Процент на остаток или аннуитет
      f_new_perc_active(rep_date_end) = 1) then
  begin
    -- Новый механизм начисления процентов
    select result_payed, result_writeoff, result_perc_end, result_balance_end,
           result_calced_all, result_payed_all, result_writeoff_all, result_disb_amount, result_disb_date,
           result_out_perc_end, result_vac_perc_saldo
    from aarep_calc_perc_one_cls(:k_id, :rep_date_begin, :rep_date_end, 0, 1, 1)
    into :result_payed, :result_writeoff, :result_perc_end, :result_balance_end,
         :result_calced_all, :result_payed_all, :result_writeoff_all, :result_disb_amount, :result_disb_date,
         :result_out_perc_end, result_vac_perc_saldo;
  end
  else
  begin
    kont_last_date = k_kredline_end;
  
    select ok_sh_date
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_sh_date > :k_kredline_end and
          ok_ol_date <= :rep_date_end
    order by ok_from_k_id desc, ok_sh_date desc
    rows 1
    into :kont_last_date;

    stop_days = 0;

    if (db_id = 81001 and kont_last_date >= '01.01.2013') then
    begin
      stop_days = 60;
      if (rep_date_end >= '30.06.2014' and kont_last_date > '02.05.2014') then
        stop_days = 61;
    end

    k_stop_perc = minvalue(coalesce(k_stop_perc, kont_last_date + stop_days), kont_last_date + stop_days);

    dd_0 = null;
    select oks_date_begin
    from oksummary
    where oks_from_k_id = :k_id and
          oks_balance <= 0 and
          oks_close_state = 0 and
          oks_date_begin <= :rep_date_end
    order by oks_from_k_id, oks_date_begin
    rows 1
    into :dd_0;

    if (dd_0 is not null) then
    begin
      if (dd_0 < k_stop_perc) then
        k_stop_perc = :dd_0;
    end

    if (k_stop_perc > rep_date_end) then
      k_stop_perc = null;

    select result_payed, result_writeoff, result_perc_end, result_balance_end,
           result_calced_all, result_payed_all, result_writeoff_all, result_disb_amount, result_disb_date,
           result_out_perc_end
    from aarep_calc_perc_one_temp(:k_id, :rep_date_begin, :rep_date_end, 0, :k_stop_perc, :db_id)
    into :result_payed, :result_writeoff, :result_perc_end, :result_balance_end,
         :result_calced_all, :result_payed_all, :result_writeoff_all, :result_disb_amount, :result_disb_date,
         :result_out_perc_end;
  end

  if ((rep_date_end < '01.01.2013' and db_id = 81001) or
      (rep_date_end < '01.06.2013' and db_id in (39801, 41701)) or
      (rep_date_end < '01.06.2013' and db_id = 94401)) then
  begin
    -- проценты обнуляются если дата отчета раньше чем 2013 год
    if (result_perc_end > 0 and
        result_balance_end <= 0) then
      result_perc_end = 0;

    if (result_out_perc_end > 0 and
        result_balance_end <= 0) then
      result_out_perc_end = 0;
  end

  suspend;
end
