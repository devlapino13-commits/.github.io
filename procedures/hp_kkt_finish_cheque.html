CREATE PROCEDURE HP_KKT_FINISH_CHEQUE(
    KKM_ID D_ID NOT NULL,
    KKM_FD D_INTEGER NOT NULL,
    KKM_FPD D_ID NOT NULL,
    KKM_FN D_STR_20 NOT NULL,
    KKM_FISC_DATETIME D_DATETIME NOT NULL,
    UUID_STR D_STR_CODE,
    KKD_CODE D_STR_CODE = null,
    KKM_SHIFT_NUM D_INTEGER = null)
AS
declare variable curr_status d_smallint;
declare variable kkt_kind d_smallint;
declare variable kkf_id d_integer;
declare variable kkd_id d_integer;
declare variable error_text d_str_255;
declare variable msg_subject d_str_255;
declare variable msg_text d_str_long;
declare variable db_id d_integer;
declare variable err_mail d_str_code;
declare variable err_prefix d_str_10;
begin
  select kkm_status, kkm_kkt_kind
  from kkt_main
  where kkm_id = :kkm_id
  into curr_status, kkt_kind;

  if (curr_status is null) then
  begin
    error_text = 'Не найден чек';
    exception e_default 'Not found cheque for kkm_id = ' || kkm_id;
  end 

  if (curr_status = 2) then
  begin
    error_text = 'Чек уже фискализован';
    exception e_default 'Cheque already fiscalized for kkm_id = ' || kkm_id;
  end

  select kkf_id
  from kkt_fn
  where kkf_code = :kkm_fn
  into kkf_id;

  if (kkf_id is null) then
  begin
    db_id = f_get_db_id();

    if (db_id = 81001 and kkt_kind = 1) then
      kkd_code = 'cloud';

    select kkd_id
    from kkt_devices
    where kkd_code = :kkd_code
    into kkd_id;

    if (kkd_id is null and db_id = 39801 and not exists(select 1 from hp_db_is_actual)) then
      kkd_id = 1;

    if (kkd_id is null) then
    begin
      error_text = 'Не найдена ККТ';
      exception e_default 'not found KKT. kkd_code = ' || coalesce(kkd_code, '<NULL>');
    end

    insert into kkt_fn (kkf_from_kkd_id, kkf_code)
    values (:kkd_id, :kkm_fn)
    returning kkf_id
    into :kkf_id;
  end

  update kkt_main
  set kkm_status = 2,
      kkm_fd = :kkm_fd,
      kkm_fpd = :kkm_fpd,
      kkm_from_kkf_id = :kkf_id,
      kkm_fisc_datetime = :kkm_fisc_datetime,
      kkm_uuid = coalesce(char_to_uuid(:uuid_str), kkm_uuid),
      kkm_shift_num = :kkm_shift_num
  where kkm_id = :kkm_id;

  when any do
  begin
    db_id = f_get_db_id();
    err_mail = trim(iif(db_id = 81001, 'ru.finstart@molbulak.com', 'finstart@mbfplanet.com'));
    err_prefix = case db_id
                   when 81001 then 'RU: '
                   when 39801 then 'KZ: '
                   when 41701 then 'KG: '
                   when 94401 then 'AZ: '
                   else ''
                 end;

    msg_subject = err_prefix || 'ККТ: Ошибка в hp_kkt_finish_cheque.';
    msg_text  = 'Ошибка в hp_kkt_finish_cheque: ' || coalesce(error_text, 'gdscode: ' || gdscode) || ascii_char(10) || ascii_char(13) ||
                'Параметры:' || ascii_char(10) || ascii_char(13) ||
                ' kkm_id = ' || coalesce(kkm_id, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' kkm_fd = ' || coalesce(kkm_fd, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' kkm_fpd = ' || coalesce(kkm_fpd, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' kkm_fn = ' || coalesce(kkm_fn, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' kkm_fisc_datetime = ' || coalesce(kkm_fisc_datetime, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' uuid_str = ' || coalesce(uuid_str, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' kkd_code = ' || coalesce(kkd_code, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' kkm_shift_num = ' || coalesce(kkm_shift_num, '<NULL>');

    in autonomous transaction do
      insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id, msg_command_kind)
      values (0, :err_mail, :msg_subject, :msg_text, 14, 1000000000000, '');

    if (curr_status < 2) then
      update kkt_main
      set kkm_status = -2
      where kkm_id = :kkm_id;

    error_text = left(coalesce(error_text, 'gdscode: ' || gdscode), 1024);

    insert into kkt_error (kke_from_kkm_id, kke_error_id)
    values (:kkm_id, f_memo_id(:error_text));
  end 
end
