CREATE PROCEDURE MONETA_FPS_CREDITS_FOR_ADD
RETURNS (
    ESM_ID D_ID,
    PHONE_NO D_STR_CODE,
    BANK_ID D_STR_CODE,
    DISB_AMOUNT D_CURRENCY,
    MNT_DESCRIPTION D_STR_255,
    CLIENT_NAMES D_STR_LONG,
    PAYMENT_ID D_STR_CODE,
    MNT_ID D_STR_10)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable k_kontract_no d_str_100;
declare variable c_name_fio d_str_100;
declare variable ag_id d_integer;
declare variable esm_account d_str_code;
declare variable colon_pos d_integer;
declare variable k_disb_date d_date;
declare variable esm_full_source d_str_long;
declare variable pr_code d_str_100;
declare variable edoc_pmt_project_disb d_str_255;
begin
  -- Процедура предоставляет список платежей, по которым должны быть перечисления (выдачи) через Монету,
  -- по системе быстрых платежей СБП
  edoc_pmt_project_disb = f_get_const('edoc_pmt_project_disb');

  for select esm_id, esm_account, esm_amount, esm_object_id, esm_from_k_id, esm_full_source
      from extsysmain
      where esm_project = f_format('%1_disb', :edoc_pmt_project_disb) and
            esm_action = 'PhoneDisb' and
            esm_status = 0
      into esm_id, esm_account, disb_amount, c_id, k_id, esm_full_source
  do
  begin
    -- Обратная совместимость:
    -- До перехода на ручное подтверждение, payment_id возвращаем пустой, после перехода id из Монеты
    begin
      payment_id = f_json_str_get_value(esm_full_source, 'payment_id', '');
    when any do
      payment_id = '';
    end
    -- esm_account хранит № телефона:ID банка, пример: 79772654344:100000000008
    colon_pos = position(':', esm_account);
    if (colon_pos = 0) then
      exception e_default f_format('Incorrect esm_account format, esm_id:%1', esm_id);

    phone_no = left(esm_account, colon_pos - 1);
    bank_id = right(esm_account, char_length(esm_account) - colon_pos);

    select client_names
    from moneta_client_fps_names(:c_id)
    into client_names;

    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;

    if (k_id is not null) then
    begin
      select k_kontract_no,
             k_disb_date,
             pr.pr_code
      from kontract kont
      inner join product pr on pr.pr_id = kont.k_from_pr_id
      where kont.k_id = :k_id
      into k_kontract_no,
           k_disb_date,
           pr_code;

      mnt_id = iif(:pr_code containing '.oivo_pack.', f_get_const(f_format('%1_oivo_acc', :edoc_pmt_project_disb)), f_get_const(f_format('%1_acc', :edoc_pmt_project_disb)));

      if (k_disb_date is distinct from current_date) then  -- Контроль: если дата отличается не выдаем
        continue;

      mnt_description = 'Выдача займа по договору №' || k_kontract_no || ', ' || c_name_fio || ', без НДС';

      suspend;
    end else
    begin
      select epl_object_id
      from epay_trans, epay_list
      where ept_from_esm_id = :esm_id and
            ept_from_epl_id = epl_id
      into ag_id;

      if (ag_id is not null) then
        mnt_description = 'Выплата по парнерской программе, договор №' || ag_id || ', ' || c_name_fio || ', без НДС';

      suspend;
    end
  end
end
