CREATE PROCEDURE HP_DO_PERC_SET_KN_SALDO(
    K_ID D_ID,
    OL_DATE D_DATE,
    NEW_PERC_SALDO D_CURRENCY)
AS
declare variable client_count d_integer;
declare variable kc_id d_id;
declare variable res_amount d_currency;
declare variable oks_bal_perc d_currency;
begin
  if (new_perc_saldo is null) then
  begin
    select result_perc_end
    from aarep_calc_perc_one_cls(:k_id, :ol_date, :ol_date, 0, 1, 1)
    into :new_perc_saldo;
  end

  if (not exists(select *
                 from kn_oksummary
                 where oks_from_k_id = :k_id and
                       oks_date_begin = :ol_date)) then
    execute procedure hp_kn_oksummary_kont_insert(:k_id, :ol_date, 1);

  select sum(oks_bal_perc)
  from kn_oksummary
  where oks_from_k_id = :k_id and
        oks_date_begin = :ol_date
  into oks_bal_perc;

  if (oks_bal_perc is null) then
    oks_bal_perc = 0;

  if (oks_bal_perc <> new_perc_saldo) then
  begin
    select count(*)
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into client_count;

    if (client_count = 1) then
    begin
      update kn_oksummary
      set oks_need_recalc = 1,
          oks_is_fix_bal_perc = 1,
          oks_bal_perc_fixed = :new_perc_saldo
      where oks_from_k_id = :k_id and
            oks_date_begin = :ol_date;
    end else
    begin
      for select kc_id, res_amount
      from hp_devide_kn_amount(:k_id, 0, :new_perc_saldo, 0)
      into :kc_id, :res_amount
      do
      begin
        update kn_oksummary
        set oks_need_recalc = 1,
            oks_is_fix_bal_perc = 1,
            oks_bal_perc_fixed = :res_amount
        where oks_from_kc_id = :kc_id and
              oks_date_begin = :ol_date;
      end
    end
  end
end
