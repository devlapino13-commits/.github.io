CREATE PROCEDURE HP_COM_LIST_REFDATA_JSON(
    RD_MAIN_CODE D_STR_CODE)
RETURNS (
    COMBO_LIST D_STR_8100)
AS
begin
  -- Возвращает список выпадающих значений для заполнения параметров запроса на КК

  if (coalesce(rd_main_code, '') = '') then
    exception e_default 'Empty refdata main_code';

  combo_list = '';
  select list(f_format('{"id":%1, "name":%2}', id, f_json_escape(name, 'str')))
  from hp_com_list_refdata_combo(:rd_main_code)
  into combo_list;

  combo_list = '{"list":[' || combo_list || ']}';
  suspend;
end
