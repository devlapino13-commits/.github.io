CREATE PROCEDURE AAREP_STATISTICS_LEGITIM(
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    KOLLEG_10_MONTH1 INTEGER,
    KOLLEG_20_MONTH1 INTEGER,
    KOLLEG_30_MONTH1 INTEGER,
    KOLLEG_40_MONTH1 INTEGER,
    KOLLEG_50_MONTH1 INTEGER,
    KOLLEG_60_MONTH1 INTEGER,
    KOLLEG_70_MONTH1 INTEGER,
    KOLLEG_10_MONTH2 INTEGER,
    KOLLEG_20_MONTH2 INTEGER,
    KOLLEG_30_MONTH2 INTEGER,
    KOLLEG_40_MONTH2 INTEGER,
    KOLLEG_50_MONTH2 INTEGER,
    KOLLEG_60_MONTH2 INTEGER,
    KOLLEG_70_MONTH2 INTEGER,
    KOLLEG_10_MONTH3 INTEGER,
    KOLLEG_20_MONTH3 INTEGER,
    KOLLEG_30_MONTH3 INTEGER,
    KOLLEG_40_MONTH3 INTEGER,
    KOLLEG_50_MONTH3 INTEGER,
    KOLLEG_60_MONTH3 INTEGER,
    KOLLEG_70_MONTH3 INTEGER)
AS
declare variable reg_id integer;
begin
  for with tabl1
      as (select kc_from_c_id, max(kc_from_k_id) kc_from_k_id
          from kontstate, knclient
          where ks_ol_is_delete = 0 and
                :rep_date_end between ks_date_begin and ks_date_end and
                ks_state = 1 and
                ks_date_begin >= '01.04.2015' and
                ks_from_k_id = kc_from_k_id and
                kc_amount_approved > 0
          group by kc_from_c_id),
      tabl2
      as (select kc_from_c_id, kc_from_k_id, sum(clp_is_legitimate) as kol_legitimate
          from tabl1, clilegpay
          where kc_from_c_id = clp_from_c_id and
                clp_ol_date <= :rep_date_end
          group by kc_from_c_id, kc_from_k_id)
      select k_last_reg_id, sum(iif(kol_legitimate + 1 = 10, 1, 0)) kolleg_10_month1,
             sum(iif(kol_legitimate + 1 = 20, 1, 0)) kolleg_20_month1,
             sum(iif(kol_legitimate + 1 = 30, 1, 0)) kolleg_30_month1,
             sum(iif(kol_legitimate + 1 = 40, 1, 0)) kolleg_40_month1,
             sum(iif(kol_legitimate + 1 = 50, 1, 0)) kolleg_50_month1,
             sum(iif(kol_legitimate + 1 = 60, 1, 0)) kolleg_60_month1,
             sum(iif(kol_legitimate + 1 = 70, 1, 0)) kolleg_70_month1,

             sum(iif(kol_legitimate + 2 = 10, 1, 0)) kolleg_10_month2,
             sum(iif(kol_legitimate + 2 = 20, 1, 0)) kolleg_20_month2,
             sum(iif(kol_legitimate + 2 = 30, 1, 0)) kolleg_30_month2,
             sum(iif(kol_legitimate + 2 = 40, 1, 0)) kolleg_40_month2,
             sum(iif(kol_legitimate + 2 = 50, 1, 0)) kolleg_50_month2,
             sum(iif(kol_legitimate + 2 = 60, 1, 0)) kolleg_60_month2,
             sum(iif(kol_legitimate + 2 = 70, 1, 0)) kolleg_70_month2,

             sum(iif(kol_legitimate + 3 = 10, 1, 0)) kolleg_10_month3,
             sum(iif(kol_legitimate + 3 = 20, 1, 0)) kolleg_20_month3,
             sum(iif(kol_legitimate + 3 = 30, 1, 0)) kolleg_30_month3,
             sum(iif(kol_legitimate + 3 = 40, 1, 0)) kolleg_40_month3,
             sum(iif(kol_legitimate + 3 = 50, 1, 0)) kolleg_50_month3,
             sum(iif(kol_legitimate + 3 = 60, 1, 0)) kolleg_60_month3, sum(iif(kol_legitimate + 3 = 70, 1, 0)) kolleg_70_month3

      from tabl2, kontract
      where kc_from_k_id = k_id
      group by k_last_reg_id
      into reg_id, kolleg_10_month1, kolleg_20_month1, kolleg_30_month1, kolleg_40_month1, kolleg_50_month1,
           kolleg_60_month1, kolleg_70_month1, kolleg_10_month2, kolleg_20_month2, kolleg_30_month2, kolleg_40_month2,
           kolleg_50_month2, kolleg_60_month2, kolleg_70_month2, kolleg_10_month3, kolleg_20_month3, kolleg_30_month3,
           kolleg_40_month3, kolleg_50_month3, kolleg_60_month3, kolleg_70_month3
  do
  begin

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;

    suspend;
  end

end
