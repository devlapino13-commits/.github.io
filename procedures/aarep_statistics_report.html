CREATE PROCEDURE AAREP_STATISTICS_REPORT(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_REG_NAMEP STR_100,
    RES_REG_NAME STR_100,
    SERTIFIKATOR D_STR_100,
    KOL_PROCESSED INTEGER,
    KOL_REFUSED_NEW INTEGER,
    KOL_REJECTED INTEGER)
AS
declare variable reg_id d_id;
begin
  for select parent_name,
             reg_name,
             officer_name,
             sum(iif(st_name in ('Подтвежден', 'На доработку', 'Отказан'), 1, 0)),
             sum(iif(st_name = 'На доработку' and first_status_reason not in ('Доработка: Доработки вне учета','Доработка: Технические доработки'), 1, 0)),
             sum(iif(st_name = 'Отказан', 1, 0)) -- кол-во отказанных
      from aarep_post_inspection(:rep_region_id, :rep_date_begin, :rep_date_end, 0)
      group by 1, 2, 3
      order by 1
      into :res_reg_namep, :res_reg_name, :sertifikator, :kol_processed, :kol_refused_new, :kol_rejected
  do
  begin
    suspend;
  end
end
