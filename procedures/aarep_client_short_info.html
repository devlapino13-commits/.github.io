CREATE PROCEDURE AAREP_CLIENT_SHORT_INFO(
    CLIENT_ID DECIMAL(18,0))
RETURNS (
    KS_STATE INTEGER,
    KS_STATE_TEXT VARCHAR(500))
AS
declare variable kf_name varchar(100);
declare variable kf_region varchar(100);
declare variable kf_circle integer;
declare variable kf_lo varchar(100);
declare variable k_id decimal(18,0);
declare variable sbl_name varchar(100);
declare variable oko_zad curr_type;
declare variable k_kontract_no d_str_255;
begin
  ks_state = -1;
  ks_state_text = 'Клиент не имеет кредитную историю';
  for select ks_state, kal_new_kf_name, reg_code || ' ' || reg_name, kc_circle, c_name_fio, ks_from_k_id
      from kontract, kontstate, knclient, knaktlist, region, clients
      where ks_from_k_id = kc_from_k_id and
            ks_ol_is_delete = 0 and
            kc_from_c_id = :client_id and
            kc_amount_approved <> 0 and
            k_id = ks_from_k_id and
            k_status > -1 and
            kal_from_kf_id = k_from_kf_id and
            cast(ks_date_begin as date) <= cast('NOW' as date) and
            ks_date_end >= cast('NOW' as date) and
            kal_date <= cast('NOW' as date) and
            kal_date_end >= cast('NOW' as date) and
            knaktlist.is_delete = 0 and
            kal_new_reg_id = reg_unicode and
            kal_new_kf_officer = c_id
      order by ks_state, ks_date_begin desc, kc_circle desc
      rows 1
      into :ks_state, kf_name, kf_region, kf_circle, :kf_lo, :k_id
  do
  begin
    oko_zad = 0;

    ks_state_text = 'Ограниченный просмотр из-за отсутствия действующей заявки';
    if (ks_state = 0) then
      ks_state_text = 'Договор клиента находится в ПРОЦЕССЕ.';
    if (ks_state = 1) then
      ks_state_text = 'Договор клиента является АКТИВНЫМ.';
    if (ks_state = 2) then
    begin
      ks_state_text = 'Договор клиента ЗАКРЫТ.';
      if (f_get_db_id() = 81001) then
      begin
        with tabl1
        as (select k_kontract_no,
                   sum(oko.oko_os + oko.oko_perc + oko.oko_other + oko.oko_sh_os + oko.oko_sh_perc + oko.oko_sh_other) oko_zad
            from operkroff oko1
            inner join operkroff oko on oko1.oko_from_k_id = oko.oko_from_k_id
            inner join kontract on oko.oko_from_k_id = k_id
            inner join operlist ol on ol.ol_id = oko.oko_from_ol_id
            where oko.oko_from_c_id = :client_id and
                  ol.is_delete = 0
            group by 1)
        select left(list(k_kontract_no, ', '), 255), sum(oko_zad) oko_zad
        from tabl1
        where oko_zad > 0
        into k_kontract_no, oko_zad;
        if (oko_zad > 0) then
          ks_state_text = 'Просроченный заем на спец.учете. ' || k_kontract_no;
      end
    end

    ks_state_text = 'Ограниченный просмотр из-за отсутствия актуальной заявки.' || ascii_char(13) || ascii_char(10) || ks_state_text;
    ks_state_text = ks_state_text || ascii_char(13) || ascii_char(10);

    ks_state_text = ks_state_text || ascii_char(13) || ascii_char(10) || 'Кр.папка: ' || kf_name;
    ks_state_text = ks_state_text || ascii_char(13) || ascii_char(10) || 'Офис: ' || kf_region;
    ks_state_text = ks_state_text || ascii_char(13) || ascii_char(10) || 'На учете: ' || kf_lo;
    if (ks_state = 1) then
    begin
      sbl_name = '';
      select sbl_name
      from sb_akt, sb_list
      where sba_is_delete = 0 and
            sba_from_sbl_id <> 0 and
            sba_status = 2 and
            current_date between sba_date_begin and sba_date_end and
            sba_from_k_id = :k_id and
            sba_from_sbl_id = sbl_id rows 1
      into :sbl_name;
      if (sbl_name is null) then
        sbl_name = '';
      if (sbl_name <> '') then
        ks_state_text = ks_state_text || ascii_char(13) || ascii_char(10) || 'Находится у коллекторской службы.';
    end
  end
  suspend;
end
