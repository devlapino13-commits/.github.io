CREATE PROCEDURE HP_MC_TEST_ACCESS_LIST
RETURNS (
    MTA_ID D_INTEGER,
    MTA_PHONE_NUMBER D_STR_CODE,
    MTA_PIN_CODE D_STR_CODE,
    MTA_FROM_C_ID D_ID,
    C_NAME_FIO D_STR_100,
    MTA_DATETIME D_DATETIME,
    MTA_FROM_UN_ID D_ID,
    UN_FIO D_STR_100)
AS
declare variable un_from_c_id d_id;
begin
    for select
        mta_id,
        mta_phone_number,
        mta_pin_code,
        mta_from_c_id,
        c_name_fio,
        mta_datetime,
        mta_from_un_id,
        un_fio,
        un_from_c_id
    from mc_test_access
    inner join clients on c_id = mta_from_c_id
    inner join username on un_id = mta_from_un_id
    into
        :mta_id,
        :mta_phone_number,
        :mta_pin_code,
        :mta_from_c_id,
        :c_name_fio,
        :mta_datetime,
        :mta_from_un_id,
        :un_fio,
        :un_from_c_id
    do
    begin
        -- Если тестовый номер был добавлен сотрудником из отдела кадров, тогда показываем запись
        if (exists
            (
                select 1
                from hr_pstatus ps
                inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
                inner join jobkind jk on jl.jl_kind = jk.jk_id
                    and jk.jk_system_code in ('JobSystem.HR_Service', 'JobSystem.HR_Department')
                where ps.ps_from_c_id = :un_from_c_id
            )
        ) then
            suspend;
    end
end
