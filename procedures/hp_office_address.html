CREATE PROCEDURE HP_OFFICE_ADDRESS(
    REG_UNICODE D_ID)
AS
declare variable reg_id d_id;
declare variable c_id d_id;
declare variable office_name d_str_100;
declare variable adl_address_str d_str_255;
declare variable cell_no d_str_64;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  select reg.reg_unicode, reg.reg_name, adl.adl_address_str
  from region reg
  inner join addrlist adl on adl.adl_id = reg.reg_address_fact
  where reg.reg_unicode = :reg_unicode and
        reg.reg_unicode > 0 and
        current_date between reg.reg_date_open and coalesce(reg.reg_date_close, '31.12.9999')
  into :reg_id, :office_name, :adl_address_str;
  cell_no = '';
  select list(f_cont_info('clients', cl.c_id, 'cell_main', 'ok'), ', ')
  from hr_pstatus ps
  inner join clients cl on cl.c_id = ps.ps_from_c_id
  inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
  inner join jobkind jk on jl.jl_kind = jk.jk_id
  where jk.jk_system_code = 'JobSystem.Region_KM' and
        current_date between ps.ps_date_begin and ps.ps_date_end and
        ps.ps_is_working = 1 and
        ps.ps_status = 1 and
        ps.ps_from_reg_id = :reg_id
  into :cell_no;
  if (cell_no = '') then
  begin
    if (db_id = 41701) then
        cell_no = case
                when :reg_id = 1050200 then '996770700478' --Кочкор-Ата (Эти номера для офисов у которых нет МО. Номера МО которые отвечают за эти офиисы)
                when :reg_id = 1051500 then '996770220093' --Мундуз
                when :reg_id = 1041700 then '996770771702' --Мады
                when :reg_id = 1011500 then '996770002327' --Кара-Коо
                when :reg_id = 1041500 then '996770771703' --Тепе-Курган
              end;
  end

  if (:reg_id is not null) then
  begin
    -- RFCKZ-193
    if (db_id = 41701) then

            'a.imankulova@molbulak.com, z.zhunusbekova@molbulak.com',

            'Новый адрес офиса: ' || :adl_address_str || ', номер телефона МО: ' || :cell_no,
            '', 1, 1000000000000);
        execute procedure hp_send_quick_message('email',
            'm.garifulin@molbulak.com, lawyers@mbulak.kg',
            'Изменился адрес  ОФИСА: ' || :office_name,
            'Новый адрес офиса: ' || :adl_address_str || ', номер телефона МО: ' || :cell_no,
            '', 1, 1000000000000);
        execute procedure hp_send_quick_message('email',
            'aabylkasymova@mbulak.kg, r.hayirdinova@mbulak.kg, ch.mamrasulova@mbulak.kg',
            'Изменился адрес  ОФИСА: ' || :office_name,
            'Новый адрес офиса: ' || :adl_address_str || ', номер телефона МО: ' || :cell_no,
            '', 1, 1000000000000);
        execute procedure hp_send_quick_message('email',
            'admin01@mbulak.kg, admin_osh@mbulak.kg',
            'Изменился адрес  ОФИСА: ' || :office_name,
            'Новый адрес офиса: ' || :adl_address_str || ', номер телефона МО: ' || :cell_no,
            '', 1, 1000000000000);
        execute procedure hp_send_quick_message('email',
            'opermanager@mbulak.kg, g.berdimuratova@molbulak.com',
            'Изменился адрес  ОФИСА: ' || :office_name,
            'Новый адрес офиса: ' || :adl_address_str || ', номер телефона МО: ' || :cell_no,
            '', 1, 1000000000000);
    end
    else if (db_id = 39801) then
    begin
        execute procedure hp_send_quick_message('email',
        'b.djumabekova@mbulak.kz, m.ismailova@mbulak.kz, m.garifulin@molbulak.com',
