CREATE PROCEDURE HP_SURVEY_TIME_EDIT(
    QUES_ID INTEGER,
    ACTION_KIND VARCHAR(100),
    SET_DELTA INTEGER,
    SET_TIME TIMESTAMP)
RETURNS (
    RES_ID INTEGER,
    RES_TEXT VARCHAR(255))
AS
begin
  if (coalesce(ques_id, 0) = 0) then
  begin
    res_id = -1;
    res_text = 'Вопрос не выбран';
    suspend;
    exit;
  end

-- Start_Now     - начать голосование сейчас
-- Finish_Now    - заверишить сейчас
-- Clear_Start   - убрать время начала
-- Clear_Finish  - убрать время завершения
-- Set_Start     - установить точное время запуска
-- Set_Finish    - установить точное время завершения
-- During_minute - Устанавливает дату окончания от Начало+Minute
-- During_hour   - Устанавливает дату окончания от Начало+Hour
-- During_day    - Устанавливает дату окончания от Начало+day

  res_id = -1;
  res_text = 'Команда не распознана.';

  if (action_kind = 'Start_Now') then
  begin
    update survey_ques
    set svq_time_start = current_timestamp,
        svq_time_finish = null
    where svq_id = :ques_id;
    res_id = 1;
    res_text = 'Голосование началось.';
  end

  if (action_kind = 'Finish_Now') then
  begin
    if (exists(select *
               from survey_ques
               where svq_id = :ques_id and
                     (svq_time_start is null or
                     svq_time_start > current_timestamp))) then
    begin
      res_id = -1;
      res_text = 'Голосование еще не началось.';
    end
    else
    begin
      update survey_ques
      set svq_time_finish = current_timestamp
      where svq_id = :ques_id;
      res_id = 1;
      res_text = 'Голосование завершено.';
    end
  end

  if (action_kind = 'Clear_Start') then
  begin
    if (exists(select *
               from survey_ques
               where svq_id = :ques_id and
                     svq_time_start is null)) then
    begin
      res_id = -1;
      res_text = 'Время начала еще не установлено.';
    end
    else
    begin
      update survey_ques
      set svq_time_start = null,
          svq_time_finish = null
      where svq_id = :ques_id;
      res_id = 1;
      res_text = 'Время начала убрано.';
    end
  end

  if (action_kind = 'Clear_Finish') then
  begin
    if (exists(select *
               from survey_ques
               where svq_id = :ques_id and
                     svq_time_finish is null)) then
    begin
      res_id = -1;
      res_text = 'Время завершения еще не установлено.';
    end
    else
    begin
      update survey_ques
      set svq_time_finish = null
      where svq_id = :ques_id;
      res_id = 1;
      res_text = 'Время завершено убрано.';
    end
  end

  if (action_kind = 'Set_Start') then
  begin
    if (set_time is null) then
    begin
      res_id = -1;
      res_text = 'Время запуска не указано.';
    end
    else
    if (set_time < current_timestamp) then
    begin
      res_id = -1;
      res_text = 'Время запуска не может быть прошедшим периодом.';
    end
    else
    if (datediff(day, current_timestamp, set_time) > 60) then
    begin
      res_id = -1;
      res_text = 'Время запуска слишком далеко. Разрешается максимум за 60 дней.';
    end
    else
    begin
      update survey_ques
      set svq_time_start = :set_time,
          svq_time_finish = null
      where svq_id = :ques_id;
      res_id = 1;
      res_text = 'Время начала установлено.';
    end
  end

  if (action_kind = 'Set_Finish') then
  begin
    if (set_time is null) then
    begin
      res_id = -1;
      res_text = 'Время завершения не указано.';
    end
    else
    if (set_time < current_timestamp) then
    begin
      res_id = -1;
      res_text = 'Время завершения не может быть прошедшим периодом.';
    end
    else
    if (exists(select *
               from survey_ques
               where svq_id = :ques_id and
                     svq_time_start is null)) then
    begin
      res_id = -1;
      res_text = 'Время запуска еще не установлено.';
    end
    else
    if (exists(select *
               from survey_ques
               where svq_id = :ques_id and
                     svq_time_start > :set_time)) then
    begin
      res_id = -1;
      res_text = 'Время завершения указано раньше чем время запуска.';
    end
    else
    begin
      update survey_ques
      set svq_time_finish = :set_time
      where svq_id = :ques_id;
      res_id = 1;
      res_text = 'Время завершения установлено.';
    end
  end

  if (action_kind = 'During_minute' or (action_kind = 'During_hour') or (action_kind = 'During_day')) then
  begin
    if (coalesce(set_delta, 0) <= 0) then
    begin
      res_id = -1;
      res_text = 'Задан неправильный период';
    end
    else
    begin
      if (exists(select *
                 from survey_ques
                 where svq_id = :ques_id and
                       svq_time_start is null)) then
      begin
        res_id = -1;
        res_text = 'Время запуска еще не установлено.';
      end
      else
      begin
        if (action_kind = 'During_minute') then
        begin
          update survey_ques
          set svq_time_finish = dateadd(minute, :set_delta, svq_time_start)
          where svq_id = :ques_id;
        end
        if (action_kind = 'During_hour') then
        begin
          update survey_ques
          set svq_time_finish = dateadd(hour, :set_delta, svq_time_start)
          where svq_id = :ques_id;
        end
        if (action_kind = 'During_day') then
        begin
          update survey_ques
          set svq_time_finish = dateadd(day, :set_delta, svq_time_start)
          where svq_id = :ques_id;
        end
        res_id = 1;
        res_text = 'Время завершения установлено.';
      end
    end
  end
  suspend;
end
