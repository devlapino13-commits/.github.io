CREATE PROCEDURE HP_GET_REF_MONEYOPER(
    OL_DATE D_DATE,
    ACC_DT D_ID,
    ACC_KT D_ID,
    CLIENT_DT D_ID,
    CLIENT_KT D_ID,
    KC_ID D_ID)
RETURNS (
    RMM_ID D_ID,
    RMO_ID D_ID)
AS
declare variable ab_dt d_id;
declare variable ab_kt d_id;
declare variable non_resident_dt d_smallint;
declare variable non_resident_kt d_smallint;
declare variable is_noresident d_smallint;
declare variable acc_code_dt d_str_code;
declare variable acc_code_kt d_str_code;
declare variable is_sud d_smallint;
declare variable is_found d_smallint;
declare variable rmm_add_info d_str_code;
declare variable rmm_memo d_str_255;
declare variable rd_id d_integer;
declare variable sab_code_dt d_str_255;
declare variable sab_code_kt d_str_255;
begin
  select acc_code, acc_from_add_ab_id, sab_full_code,
         coalesce(acc_non_resident, 0)
  from ref_account, sys_accbalance
  where acc_id = :acc_dt and
        acc_from_sab_id = sab_id
  into acc_code_dt, ab_dt, non_resident_dt, sab_code_dt;

  select acc_code, acc_from_add_ab_id, sab_full_code,
         coalesce(acc_non_resident, 0)
  from ref_account, sys_accbalance
  where acc_id = :acc_kt and
        acc_from_sab_id = sab_id
  into acc_code_kt, ab_kt, non_resident_kt, sab_code_kt;

  is_noresident = 0;
  if (non_resident_dt = 1 or
      non_resident_kt = 1) then
    is_noresident = 1;

  is_sud = -1;
  is_found = 0;
  for select rmm_id, rmm_add_info, rmm_memo,
             iif(:is_noresident = 1 and coalesce(rmm_from_rmo_noresident, 0) <> 0, rmm_from_rmo_noresident, coalesce(rmm_from_rmo_resident, 0))
      from ref_moneymask
      where rmm_from_ab_dt = :ab_dt and
            rmm_from_ab_kt = :ab_kt and
            :acc_code_dt like rmm_mask_kt and
            :acc_code_kt like rmm_mask_kt
      order by rmm_order_no
      into :rmm_id, :rmm_add_info, :rmm_memo, rmo_id
  do
  begin
    if (rmm_memo containing 'суд' and
        kc_id is not null) then
    begin
      if (is_sud = -1) then
      begin
        if (rd_id is null) then
          rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

        if (exists(select *
                   from kn_operextra
                   where ox_from_kc_id = :kc_id and
                         ox_ol_date <= :ol_date and
                         ox_ol_is_active = 1 and
                         ox_from_rd_id = :rd_id and
                         ox_amount_plus <> 0)) then
          is_sud = 1;
        else
          is_sud = 0;
      end

      if (rmm_memo = 'по решению суда') then
      begin
        if (is_sud = 1) then
        begin
          is_found = 1;
          Break;
        end
      end else
        if (is_sud = 0) then
        begin
          is_found = 1;
          Break;
        end
    end else
    begin
      if (coalesce(rmo_id, 0) <> 0) then
      begin
        is_found = 1;
        Break;
      end
    end
  end

  if (is_found = 0) then
  begin
    rmo_id = 0;
    rmm_id = 0;
  end else
  begin
    if (coalesce(rmo_id, 0) = 0) then
      rmo_id = 0;
  end

  Suspend;
end
