CREATE PROCEDURE HP_BANKRUPT_CHECK(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    KF_CLIENT_FIO D_STR_100,
    KF_CLIENT_PASS_NO D_STR_CODE,
    KF_DISB_DATE D_DATE,
    KC_AMOUNT D_CURRENCY,
    KF_ERROR_TEXT D_STR_255,
    KF_ERROR_IS_CRITICAL D_SMALLINT)
AS
declare variable c_id id_type;
declare variable brt_status d_smallint;
begin
  kc_amount = 0;

  if (obj_kind = 'kontract') then
    select k_disb_date
    from kontract
    where k_id = :obj_id
    into kf_disb_date;
  else if (obj_kind = 'miniapp') then
    select ma_date_oper
    from miniapp
    where ma_id = :obj_id
    and ma_date_end = '31.12.9999'
    into kf_disb_date;

  for select c_id, c_name_fio, c_pass_no
  from knclient, clients
  where :obj_kind = 'kontract' and
        kc_from_k_id = :obj_id and
        kc_amount_approved > 0 and
        kc_from_c_id = c_id
  union
  select c_id, c_name_fio, c_pass_no
  from kontract, product, zalog, clients
  where :obj_kind = 'kontract' and
    k_id = :obj_id and
    z_from_k_id = k_id and
    k_from_pr_id = pr_id and
    z_from_c_id = c_id and
    c_type = 0 and
    pr_code containing '.zalog_car.' and
    z_type = 1 and
    z_from_c_id is not null
  union
  select c_id, c_name_fio, c_pass_no
  from miniappclient, clients
  where :obj_kind = 'miniapp' and
        mac_from_ma_id = :obj_id and
        mac_amount > 0 and
        mac_from_c_id = c_id
  into c_id, kf_client_fio, kf_client_pass_no
  do
  begin
    kf_error_text = '';
    brt_status = f_bankrupt(c_id, kf_disb_date);

    if (brt_status = 1) then
    begin
      kf_error_is_critical = 2;
      kf_error_text = 'По клиенту найдена информация о банкротстве. Выдача невозможна.';
      suspend;
    end
    else if (brt_status = -1) then
    begin
      if (obj_kind = 'miniapp') then
        kf_error_is_critical = 0;
      else
        kf_error_is_critical = 1;
      kf_error_text = 'Не получена информация проверки на банкротство.';
      suspend;
      -- Отправляем на проверку (при необходимости, force_insert = 0)
      execute procedure hp_extinfo_insert_kontur(obj_kind, obj_id, 0, un_id);

    end
  end
end
