CREATE PROCEDURE HP_INN_SAFE_UPDATE(
    C_ID D_ID,
    C_INN D_STR_20,
    CFG_CODE D_STR_CODE = '')
AS
declare variable MSG_BODY D_STR_8100;
declare variable MSG_MAIL_TO D_STR_CODE;
declare variable MSG_SUBJECT D_STR_CODE;
begin
  -- Процедура для обновления ИНН, с логированием на почту ошибок
  -- Назначение: обновить ИНН в клиенте, при получении данных сервисом из ФНС
  uxf$set_user(1000000000000); 

  update clients
  set c_inn = :c_inn
  where c_id = :c_id
    and c_inn is distinct from :c_inn;
  when any do
  begin
    if (f_get_db_id() = 81001) then
      insert into inn_for_update (ifu_from_c_id, ifu_inn) -- RFCRU-3596 Доработка механизма сохранения ИНН
      values (:c_id, :c_inn);

    msg_subject = 'Ошибка при обновлении ИНН клиента';
    msg_mail_to = 'ru.finstart@molbulak.ru';

    with recursive head
    as (select 1 level_, call_s.mon$statement_id, call_s.mon$call_id,
               call_s.mon$object_name, call_s.mon$object_type
        from mon$call_stack call_s
        where call_s.mon$caller_id is null
        union all
        select level_ + 1, call_s.mon$statement_id, call_s.mon$call_id,
               call_s.mon$object_name, call_s.mon$object_type
        from mon$call_stack call_s
        join head on call_s.mon$caller_id = head.mon$call_id),
    t1
    as (select level_, mon$object_name
        from head
        join mon$statements stmt on stmt.mon$statement_id = head.mon$statement_id
        where stmt.mon$attachment_id = current_connection
        order by 1)
    select list(mon$object_name, ascii_char(10))
    from t1
    into msg_body;

    msg_body = f_format('При обновлении клиента, c_id: %1, c_inn: %2 произошла ошибка, gdscode: %3<br> Стек вызовов:<br>', c_id, c_inn, gdscode) || msg_body;

    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body,
                                msg_lifetime_days, msg_from_un_id,
                                msg_command_kind)
    values (0, :msg_mail_to, :msg_subject, :msg_body, 30, 1000000000000,
            'hp_safe_update_inn');

  end
end
