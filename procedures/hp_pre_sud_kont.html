CREATE PROCEDURE HP_PRE_SUD_KONT(
    REP_DATE D_DATE,
    REP_DELINQ_DAYS D_INTEGER)
RETURNS (
    SBL_NAME STR_255,
    K_ID ID_TYPE,
    K_KONTRACT_NO STR_20,
    DISB_DATE D_DATE,
    DELINQ_DAYS D_INTEGER,
    ZAD_ALL CURR_TYPE,
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    ZAD_FINE CURR_TYPE,
    ZAD_EXTRA CURR_TYPE,
    CL_FIO_ALL D_STR_LONG,
    HAS_SUD_REQ D_BOOL)
AS
declare variable oek_rd_id d_integer;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  zad_os = 0;
  zad_perc = 0;
  zad_fine = 0;
  zad_extra = 0;

  for select k_id, k_kontract_no, k_disb_date, :rep_date - oks_delinq_date
      from oksummary
      inner join kontract on k_id = oks_from_k_id
      where :rep_date between oks_date_begin and oks_date_end and
            :rep_date - oks_delinq_date >= :rep_delinq_days and
            oks_close_state = 0
      into k_id, k_kontract_no, disb_date, delinq_days
  do
  begin
    select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine)
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    into zad_os, zad_perc, zad_fine;

    select coalesce(sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff), 0)
    from operextra
    where ox_ol_is_active = 1 and
          ox_from_rd_id = :oek_rd_id and
          ox_object_id = :k_id and
          ox_object_kind = f_objkind('kontract', 'id') and
          ox_ol_date <= :rep_date
    into zad_extra;

    sbl_name = '';

    select sbl_name
    from sb_akt, sb_list
    where sba_from_k_id = :k_id and
          :rep_date between sba_date_begin and sba_date_end and
          sba_is_delete = 0 and
          sba_status = 2 and
          sba_from_sbl_id = sbl_id
    into sbl_name;

    zad_all = zad_os + zad_perc + zad_fine + zad_extra;

    select list(c_name_fio, ', ')
    from knclient
    inner join clients on c_id = kc_from_c_id
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into cl_fio_all;

    has_sud_req = 0;

    if (exists(select *
               from sud_mainreq
               where smr_from_k_id = :k_id)) then
      has_sud_req = 1;
    suspend;
  end
  for select k_id, k_kontract_no, k_disb_date, sum(zad_os), sum(zad_perc), sum(zad_fine), sum(zad_extra)
      from (select k_id, k_kontract_no, k_disb_date, oko_os + oko_sh_os as zad_os, oko_perc + oko_sh_perc as zad_perc,
                   oko_other + oko_sh_other as zad_fine, 0 zad_extra
            from operkroff
            inner join kontract on k_id = oko_from_k_id
            where oko_ol_is_active = 1 and
                  oko_ol_date <= :rep_date
            union all
            select k_id, k_kontract_no, k_disb_date, 0 zad_os, 0 zad_perc, 0 zad_fine,
                   ox_amount_plus - ox_amount_minus - ox_amount_writeoff as zad_extra
            from operextra
            inner join kontract on k_id = ox_object_id and
                  ox_object_kind = 9
            inner join kontstate on ks_from_k_id = k_id
            where ox_ol_is_active = 1 and
                  ox_from_rd_id = :oek_rd_id and
                  ox_object_kind = f_objkind('kontract', 'id') and
                  ox_ol_date <= :rep_date and
                  :rep_date between ks_date_begin and ks_date_end and
                  ks_ol_is_delete = 0 and
                  ks_state = 2)
      group by k_id, k_kontract_no, k_disb_date
      having sum(zad_os) > 0 or sum(zad_perc) > 0 or sum(zad_fine) > 0 or sum(zad_extra) > 0
      into k_id, k_kontract_no, disb_date, zad_os, zad_perc, zad_fine, zad_extra
  do
  begin
    delinq_days = 0;

    select current_date - oks_delinq_date
    from oksummary
    where oks_from_k_id = :k_id and
          oks_delinq_date < '31.12.9999'
    order by oks_date_begin desc rows 1
    into delinq_days;

    sbl_name = '';

    select sbl_name
    from sb_akt, sb_list
    where sba_from_k_id = :k_id and
          :rep_date between sba_date_begin and sba_date_end and
          sba_is_delete = 0 and
          sba_status = 2 and
          sba_from_sbl_id = sbl_id
    into sbl_name;

    zad_all = zad_os + zad_perc + zad_fine + zad_extra;

    select list(c_name_fio, ', ')
    from knclient
    inner join clients on c_id = kc_from_c_id
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into cl_fio_all;

    has_sud_req = 0;

    if (exists(select *
               from sud_mainreq
               where smr_from_k_id = :k_id)) then
      has_sud_req = 1;
    suspend;
  end
end
