CREATE PROCEDURE HP_SUD_GET_STAGE_STATUS(
    OBJECT_KIND D_STR_CODE,
    STATUS_ID D_INTEGER)
RETURNS (
    STAGE_STR D_STR_CODE,
    STATUS_STR D_STR_CODE)
AS
declare variable stage d_str_code;
begin
  stage = f_objkind(object_kind, 'name');

  status_str = (case status_id
      when -2 then 'Отказ'
      when -1 then 'Требуется доработать'
      when 0  then 'Новый'
      when 1  then 'В работе'
      when 2  then 'Завершено'
      end);

  if (stage = 'sud_mainreq') then
  begin
    stage_str = 'Запрос от офиса';
    status_str = (case status_id 
        when 0  then 'Новый'
        when 1  then 'На рассмотрении'
        when 2  then 'Одобрен'
        else status_str
        end);
  end
  else if (stage = 'sud_maindocs') then
  begin
    stage_str = 'Комплектация документов';
    status_str = (case status_id 
        when 0  then 'Сбор документов'
        when 1  then 'Документы собраны'
        else status_str
        end);
  end
  else if (stage = 'sud_mainpay') then
  begin
    stage_str = 'Оплата госпошины';
    status_str = (case status_id 
        when 0  then 'Проверка комплектации'
        when 1  then 'Ожидает оплаты ГП'
        when 2  then 'Госпошлина оплачена'
        else status_str
        end);
  end
  else if (stage = 'sud_mainlist') then
  begin
    stage_str = 'Суд';
    status_str = (case status_id
        when 0  then 'Готов к отправке'
        when 1  then 'Отправлен в Суд'
        when 2  then 'Решение получено'
        else status_str
        end);
  end
  else if (stage = 'sud_maincollect') then
  begin
    stage_str = 'Взыскание';
    status_str = (case status_id 
        when 0  then 'Готов к отправке'
        when 1  then 'Отправлено'
        when 2  then 'ИП завершено'
        else status_str
        end);
  end
  else
    stage_str = '-';

  suspend;
end
