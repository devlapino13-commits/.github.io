CREATE PROCEDURE EXPORT_RU_OPERS_2014(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OB_CURRENCY VARCHAR(3),
    OB_CURR_RATE DECIMAL(18,4),
    OB_ACCOUNT_DT VARCHAR(20),
    OB_SUMMA_VAL DECIMAL(18,4),
    OB_SUMMA DECIMAL(18,4),
    OB_ACCOUNT_KT VARCHAR(20),
    OL_FROM_REG_ID INTEGER,
    OL_DATE DATE,
    OB_ACC_SHORT_DT VARCHAR(20),
    OB_ACC_SHORT_KT VARCHAR(20),
    OL_REGION_NAME VARCHAR(100),
    OB_LOOK_DT_NAME VARCHAR(255),
    OB_LOOK_KT_NAME VARCHAR(255),
    DT_BANK_ACC VARCHAR(20),
    KT_BANK_ACC VARCHAR(20),
    DT_BIK_BANK_ACC VARCHAR(20),
    KT_BIK_BANK_ACC VARCHAR(20))
AS
begin
  for select ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_ol_reg_id, ob_ol_date, ob_short_acc_dt,
             ob_short_acc_kt, reg_name, b_dt.br_account_no, b_dt.br_from_b_id, b_kt.br_account_no, b_kt.br_from_b_id,
             a_dt.al_name, a_kt.al_name, sum(ob_summa_val), sum(ob_summa)
      from region_list(:rep_region_id, 1, 3, 1), region, acc_list a_dt, acc_list a_kt, operbook
      left join bankregion b_dt on (b_dt.br_name = ob_account_dt)
      left join bankregion b_kt on (b_kt.br_name = ob_account_kt)
      where ((ob_ol_reg_id <> 1010100 and
            ob_short_acc_dt <> '71-01/1' and
            ob_short_acc_kt <> '71-01/1'
--      and ((OB_SHORT_ACC_DT='57-01/1' and OB_SHORT_ACC_KT='50-01/1') or (OB_SHORT_ACC_KT='57-01/1' and OB_SHORT_ACC_DT='50-01/1'))
--      and OB_SHORT_ACC_DT<>'57-01/1'
--      and OB_SHORT_ACC_KT<>'57-01/1'
            and
            ob_short_acc_dt <> '90-09/1' and
            ob_short_acc_kt <> '90-09/1') or
            (ob_ol_reg_id = 1010100 and
            ob_short_acc_dt <> '71-01/1' and
            ob_short_acc_kt <> '71-01/1'
--      and ((OB_SHORT_ACC_DT='57-01/1' and OB_SHORT_ACC_KT='50-01/1') or (OB_SHORT_ACC_KT='57-01/1' and OB_SHORT_ACC_DT='50-01/1'))
--      and OB_SHORT_ACC_DT<>'57-01/1'
--      and OB_SHORT_ACC_KT<>'57-01/1'
            and
            (not(ob_short_acc_dt like '51-%') or
            (ob_short_acc_dt = '51-01/12')) and
            (not(ob_short_acc_kt like '51-%') or
            (ob_short_acc_kt = '51-01/12')) and
            ob_short_acc_dt <> '90-09/1' and
            ob_short_acc_kt <> '90-09/1')) and
            ob_ol_date between :rep_date_begin and :rep_date_end and
            ob_ol_reg_id = id_list and
            ob_ol_reg_id = reg_unicode and
            ob_account_dt = a_dt.al_id and
            ob_account_kt = a_kt.al_id and
            ob_ol_is_active = 1
      group by ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_ol_reg_id, ob_ol_date, ob_short_acc_dt, ob_short_acc_kt, reg_name, b_dt.br_account_no, b_dt.br_from_b_id, b_kt.br_account_no, b_kt.br_from_b_id, a_dt.al_name, a_kt.al_name
      into :ob_currency, :ob_curr_rate, :ob_account_dt, :ob_account_kt, :ol_from_reg_id, :ol_date, :ob_acc_short_dt,
           :ob_acc_short_kt, :ol_region_name, :dt_bank_acc, :dt_bik_bank_acc, :kt_bank_acc, :kt_bik_bank_acc,
           :ob_look_dt_name, ob_look_kt_name, :ob_summa_val, :ob_summa
  do
  begin
    if (ob_currency = '810') then
      ob_summa_val = 0;
    if (not(ob_acc_short_dt like '57-04%' and
        ob_acc_short_kt like '57-04%')) then
      suspend;
  end
end
