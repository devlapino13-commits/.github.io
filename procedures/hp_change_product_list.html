CREATE PROCEDURE HP_CHANGE_PRODUCT_LIST(
    ACTION_KIND D_STR_CODE NOT NULL,
    MCP_ID D_ID,
    MCP_FROM_PR_ID D_INTEGER,
    MCP_TITLE D_STR_255,
    MCP_TEXT_1 D_STR_255,
    MCP_TEXT_2 D_STR_255,
    MCP_TEXT_3 D_STR_255,
    MCP_TEXT_4 D_STR_255,
    MCP_ADDITIONAL_INFO D_STR_255,
    MCP_LOAN_PURPOSE D_STR_255,
    MCP_AMOUNT_STEP D_INTEGER,
    MCP_PICTURE_URL D_STR_LONG,
    MCP_SMALL_PICTURE_URL D_STR_LONG,
    MCP_IS_ACTIVE D_SMALLINT,
    MCP_SERIAL_NUMBER D_INTEGER,
    MCP_LANG D_STR_20)
RETURNS (
    RES_CODE D_INTEGER,
    RES_INFO_TEXT D_STR_255)
AS
declare variable mcp_project d_smallint;
declare variable pr_code d_str_100;
declare variable is_extra_product d_boolean; /* Определяет, является ли продукт исключением (RFCKG-5024) */
begin
    res_code = 0;
    res_info_text = '';
    is_extra_product = exists (
        select 1
        from mc_products_extra mpe
        where mpe.mpe_from_pr_id = :mcp_from_pr_id
    );


    if (action_kind not in ('add', 'update', 'deactivate')) then
    begin
        res_code = 1;
        res_info_text = f_format('Неизвестное действие: %1. Модуль настроен некорректно, обратитесь к разработчикам');
        suspend;
        exit;
    end

    -------------------------------------- ВАЛИДАЦИЯ ВХОДНЫХ ДАННЫХ -------------------------------------

    if (action_kind in ('update', 'deactivate')) then
    begin
        if (res_info_text = '' and mcp_id is null) then
            res_info_text = 'Запись не выбрана';
    
        if (res_info_text = '' and
            not exists(select 1
                       from mc_products
                       where mcp_id = :mcp_id)) then
            res_info_text = f_format('Запись с ID=%1 не существует', mcp_id);
    end

    if (action_kind in ('add', 'update')) then
    begin
        -- Валидация продукта
        if (res_info_text = '' and coalesce(mcp_from_pr_id, 0) = 0) then
            res_info_text = 'Продукт не указан';

        if (res_info_text = '' and
            not exists(select 1
                       from prodlogic prl
                       where prl.prl_from_pr_id = :mcp_from_pr_id and
                             (:is_extra_product or current_date between prl.prl_date_begin and prl.prl_date_end)
                       )
            ) then
            res_info_text = f_format('Продукт с ID=%1 не сущесвтует или не настроен', mcp_from_pr_id); 

        pr_code = (select pr_code from product where pr_id = :mcp_from_pr_id);
        mcp_project = case
                        when (pr_code containing '.oivo_pack.') then f_enum_id('mcl_project', 'oivo')
                        else f_enum_id('mcl_project', 'kmp')
                      end;

        -- Валидация текстовок
        if (res_info_text = '' and coalesce(mcp_additional_info, '') = '') then
            res_info_text = 'Укажите дополнительную информацию о продукте';

        if (res_info_text = '' and coalesce(mcp_title, '') = '') then
            res_info_text = 'Укажите заголовок продукта';

        -- Валидация языков
        if (res_info_text = '' and coalesce(mcp_lang, '') = '') then
            res_info_text = 'Укажите язык продукта';
        
        if (res_info_text = '' and
            (mcp_lang not starting with '.' or -- Не начинается с точки
             substring(mcp_lang from char_length(mcp_lang) for 1) <> '.' or -- Не заканчивается точкой
             replace(replace(replace(replace(mcp_lang, '.ru.', '.'), '.kg.', '.'), '.az.', '.'), '.kz.', '.') <> '.')) then
            res_info_text = 'Некорректный формат языков (пример: .ru. или .ru.kg.)'; 

        -- Валидация номера карусели продуктов
        if (res_info_text = '' and mcp_serial_number is null) then
            res_info_text = 'Укажите порядковый номер продукта в карусели';

        if (res_info_text = '' and mcp_serial_number < 0) then
            res_info_text = 'Порядковый номер продукта в карусели не может быть отрицательным';

        -- Валидация значения для ползунка
        if (res_info_text = '' and mcp_amount_step is null) then
            res_info_text = 'Укажите значение для ползунка';

        if (res_info_text = '' and mcp_amount_step <= 0) then
            res_info_text = 'Значение ползунка должно быть больше нуля';

        -- Валидация ссылок
        if (res_info_text = '' and coalesce(mcp_picture_url, '') <> '' and mcp_picture_url not similar to '(http|https)://%|www.%') then
            res_info_text = 'Ссылка на основную картинку имеет некорректный формат';

        if (res_info_text = '' and coalesce(mcp_small_picture_url, '') <> '' and mcp_small_picture_url not similar to '(http|https)://%|www.%') then
            res_info_text = 'Ссылка на основную картинку имеет некорректный формат';
    end

    if (res_info_text <> '') then
    begin
        res_code = 1;
        suspend;
        exit;
    end

    -----------------------------------------------------------------------------------------------------


    if (action_kind = 'add') then
    begin
        insert into mc_products (mcp_from_pr_id, mcp_title, mcp_text_1, mcp_text_2, mcp_text_3,
                                 mcp_text_4, mcp_additional_info, mcp_loan_purpose, mcp_amount_step, mcp_picture_url,
                                 mcp_small_picture_url, mcp_is_active, mcp_serial_number, mcp_lang, mcp_project)
        values (:mcp_from_pr_id, :mcp_title, :mcp_text_1, :mcp_text_2, :mcp_text_3, :mcp_text_4,
                :mcp_additional_info, :mcp_loan_purpose, :mcp_amount_step, :mcp_picture_url, :mcp_small_picture_url,
                :mcp_is_active, :mcp_serial_number, :mcp_lang, :mcp_project);

        res_info_text = 'Новая запись успешно создана';
    end

    if (action_kind = 'update') then
    begin
        update mc_products
        set mcp_from_pr_id = :mcp_from_pr_id,
            mcp_title = :mcp_title,
            mcp_text_1 = :mcp_text_1,
            mcp_text_2 = :mcp_text_2,
            mcp_text_3 = :mcp_text_3,
            mcp_text_4 = :mcp_text_4,
            mcp_additional_info = :mcp_additional_info,
            mcp_loan_purpose = :mcp_loan_purpose,
            mcp_amount_step = :mcp_amount_step,
            mcp_picture_url = :mcp_picture_url,
            mcp_small_picture_url = :mcp_small_picture_url,
            mcp_serial_number = :mcp_serial_number,
            mcp_lang = :mcp_lang,
            mcp_project = :mcp_project
        where (mcp_id = :mcp_id);

        res_info_text = 'Запись успешно обновлена';
    end

    if (action_kind = 'deactivate') then
    begin
        if (not exists(select 1
                       from mc_products
                       where mcp_id = :mcp_id and
                             mcp_is_active = 1)) then
        begin
            res_code = 1;
            res_info_text = 'Ошибка, запись уже деактивирована';
            suspend;
            exit;
        end  

        update mc_products
        set mcp_is_active = 0
        where (mcp_id = :mcp_id);

        res_info_text = 'Запись успешно деактивирована';
    end

    -- Если не было ошибок, обновляем кеш, чтобы пользователи сразу увидели изменения
    if (res_code = 0) then
        update const_var cv
        set cv.cv_value = cast(cv.cv_value as integer) + 1
        where cv.cv_code = 'fsm_client_common_ref_ver';


end
