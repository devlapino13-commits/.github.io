CREATE PROCEDURE AAREP_DISB_POINT(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    K_DISB_DATE D_DATE,
    REG_NAME D_STR_100,
    PR_NAME D_STR_100,
    RES_POINT D_INTEGER,
    RES_POINT_KIND D_STR_CODE,
    RES_CLIENT_STATE D_STR_CODE,
    RES_CLIENT_COUNT D_INTEGER)
AS
declare variable reg_id d_integer;
declare variable pr_id d_integer;
declare variable client_state d_integer;
begin
  for select k_disb_date, reg_id, pr_id, res_point, client_state, count(*)
      from aarep_disb_point_tmp(:rep_date_begin, :rep_date_end)
      group by 1, 2, 3, 4, 5
      into :k_disb_date, :reg_id, :pr_id, :res_point, :client_state, res_client_count
  do
  begin
    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;

    select pr_name
    from product
    where pr_id = :pr_id
    into pr_name;

    res_point_kind = case
                       when res_point <= 2 then 'A. 0-2'
                       when res_point <= 8 then 'B. 3-8'
                       when res_point <= 17 then 'C. 9-17'
                       else 'D. 18+'
                     end;

    res_client_state = case client_state
                       when 0 then '1. Новый клиент'
                       when 1 then '2. Активный клиент'
                       when 2 then '3. Закрыл за 30 дней'
                       else '4. Закрыл больше 30 дней'
                     end;

    suspend;
  end
end
