CREATE PROCEDURE FM_CLIENT_STATUS_OLD(
    UN_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0))
RETURNS (
    STATUS VARCHAR(20),
    REG_ID INTEGER,
    ACCESS_LEV INTEGER,
    IS_ACTIVE_CLIENT INTEGER,
    IS_EMPLOYEE INTEGER)
AS
declare variable user_c_id decimal(18,0);
declare variable c_id_tmp decimal(18,0);
declare variable hq_reg_id integer;
declare variable kc_id decimal(18,0);
begin
-- Возвращает статус контрагента:
-- staff - действующий сотрудник
-- client - клиент, имеющий активный кредит

  access_lev = -1;
  is_active_client = 0;
  is_employee = 0;

  select un_from_c_id
  from username
  where un_id = :un_id
  into :user_c_id;

  status = '';

  reg_id = 0;
  select ps_from_reg_id
  from hr_pstatus
  where ps_from_c_id = :c_id and
        ps_date_end >= '31.12.2999' and
        ps_status = 1 and
        ps_is_working = 1
  into :reg_id;
  if (coalesce(reg_id, 0) <> 0) then
  begin
    status = 'staff';
    is_employee = 1;
  end

  if (status = '') then
  begin
    for select rel_other_c_id
        from hp_client_relation_tmp(:c_id, 1)
        where char_length(rel_code) = 1 or
              rel_level = 0
        into :c_id_tmp
    do
    begin
      if (exists(select *
                 from hr_pstatus
                 where ps_from_c_id = :c_id_tmp and
                       ps_date_end >= '31.12.2999' and
                       ps_status = 1 and
                       ps_is_working = 1)) then
      begin
        status = 'staff';
        is_employee = 1;

        leave;
      end
    end
  end

  reg_id = 0;
  if (status = '') then
  begin
    for select distinct oks_region_id, oks_kc_id
        from knclient, oksummary
        where kc_from_c_id = :c_id and
              kc_amount_approved > 0 and
              kc_from_k_id = oks_from_k_id and
              oks_kc_id <> :user_c_id and
              oks_date_end >= '31.12.2999' and
              oks_balance > 0 and
              oks_close_state = 0
        into :reg_id, :kc_id
    do
    begin
      if (not(exists(select *
                     from hp_staff_changer_list(:un_id, 0)
                     where res_c_id = :kc_id))) then
      begin
        is_active_client = 1;
        if (status = '') then
          status = 'client';
        if (exists(select *
                   from hr_pstatus, joblist, jobkind
                   where ps_from_c_id = :user_c_id and
                         ps_date_begin <= current_date and
                         ps_date_end >= current_date and
                         ps_from_reg_id = :reg_id and
                         ps_is_working = 1 and
                         ps_status = 1 and
                         ps_from_jl_id = jl_id and
                         jl_kind = jk_id and
                         jk_system_code in ('JobSystem.LO_Senior', 'JobSystem.Region_KM'))) then
          status = 'client_office';
      end
    end
  end

  access_lev = 1;

  if (status <> '') then
  begin
    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.SuperUser')
               where has_access = 1)) then
    begin
      suspend;
      exit;
    end
    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Programmer')
               where has_access = 1)) then
    begin
      suspend;
      exit;
    end
  end

  if (status = 'staff') then
  begin
    access_lev = -1;

    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'FSMobile.Staff_Access')
               where has_access = 1)) then
    begin
      access_lev = 0;
    end

    hq_reg_id = f_hq_id();
    if (reg_id = hq_reg_id) then
      access_lev = -1;

    if (access_lev <> 1) then
    begin
      if (exists(select *
                 from miniapp, miniappclient, knaktlist
                 where ma_id = mac_from_ma_id and
                       mac_from_c_id = :c_id and
                       ma_is_active = 1 and
                       mac_amount > 0 and
                       ma_date_app between(current_date - 30) and current_date and
                       ma_date_end >= current_date and
                       knaktlist.is_delete = 0 and
                       kal_date <= current_date and
                       kal_date_end >= current_date and
                       kal_new_kf_officer = :user_c_id and
                       kal_from_kf_id = ma_from_kf_id)) then
      begin
        access_lev = 1;
      end
    end

    if (access_lev <> 1) then
    begin
      if (exists(select *
                 from kontract, knclient, knaktlist
                 where k_id = kc_from_k_id and
                       kc_from_c_id = :c_id and
                       kc_amount_approved > 0 and
                       k_disb_date >= (current_date - 30) and
                       knaktlist.is_delete = 0 and
                       kal_date <= current_date and
                       kal_date_end >= current_date and
                       kal_new_kf_officer = :user_c_id and
                       kal_from_kf_id = k_from_kf_id)) then
      begin
        access_lev = 1;
      end
    end

    suspend;
    exit;
  end

  if (status starting 'client') then
  begin
    access_lev = -1; -- Если клиент другого СВЗ, нет доступа
    if (status = 'client_office') then -- если сотрудник является МО и клиент этого офиса, то есть доступ
    begin
      access_lev = 1;
    end
    else
    begin
-- Иначе доступ в режиме чтения если есть права
      if (exists(select *
                 from hp_permission_access_check(:un_id, 'PermActions', 0, 'FSMobile.Other_Client_Access')
                 where has_access = 1)) then
        access_lev = 0;
    end
    suspend;
    exit;
  end

  suspend;
end
