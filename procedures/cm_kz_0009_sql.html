CREATE PROCEDURE CM_KZ_0009_SQL(
    MA_ID D_INTEGER NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable MA_DATE_SAVE D_DATE;
declare variable C_NAME_FIO D_STR_100;
declare variable K_ID_ACTIVE D_ID;
declare variable MBF_DEPT_SUPRUG D_CURRENCY;
declare variable MBF_DEPT_SUPRUG_TMP D_CURRENCY;
declare variable MA_SUMMA_MINUS D_CURRENCY;
begin
  -- RFCKZ-877
  res_error_is_critical = 2;
  mbf_dept_suprug = 0;

  for select mac.mac_from_c_id,
             c.c_name_fio,
             cast(ma.ma_date_save as d_date),
             ma.ma_summa_minus
      from miniapp ma
      inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
      inner join clients c on c.c_id = mac.mac_from_c_id
      where ma.ma_id = :ma_id and
            ma.ma_is_active = 1 and
            ma.ma_date_end = '31.12.9999' and
            mac.mac_amount > 0
      into c_id,
           c_name_fio,
           ma_date_save,
           ma_summa_minus

  do
  begin

    if (exists(select *
               from miniapp ma
               inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
               where ma.ma_is_active = 1 and
                     ma.ma_id <> :ma_id and
                     ma.ma_date_app = :ma_date_save and
                     ma.ma_date_end = '31.12.9999' and
                     mac.mac_amount > 0 and
                     mac.mac_from_c_id = :c_id)) then
    begin
      res_error_text = 'Возможно взять только 1 заем в день';
      suspend;
    end

    mbf_dept_suprug = ma_summa_minus + mbf_dept_suprug;

    for select ks.ks_from_k_id
        from kontstate ks
        inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
        where ks.ks_state = 1 and
              current_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              kncl.kc_from_c_id = :c_id
        into :k_id_active
    do
    begin
      mbf_dept_suprug_tmp = 0;
      select sum(zad_os + sh_calc_perc)
      from aarep_make_zad_list(:k_id_active, current_date, 1)
      into :mbf_dept_suprug_tmp;
      mbf_dept_suprug_tmp = coalesce(mbf_dept_suprug_tmp, 0);
      mbf_dept_suprug = mbf_dept_suprug + mbf_dept_suprug_tmp;
    end

    if (mbf_dept_suprug > 1500000) then
    begin
      res_error_text = 'Общий совокупный баланс и сумма в новой заявки клиента не должны превышать 1 500 000 тенге';
      suspend;
    end

  end
end
