CREATE PROCEDURE GET_BANKRUPTCY_STATUS(
    C_ID D_ID)
RETURNS (
    IS_BANKRUPTCY D_BOOL)
AS
begin
  -- Процедура возвращает статус — является ли клиент банкротом: "Да" / "Нет"
  is_bankruptcy = 0;

  if (exists(select *
             from bankruptcy_for_exec t
             where t.bfe_from_c_id = :c_id)) then
    is_bankruptcy = 1;

    suspend;
end
