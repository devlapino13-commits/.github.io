CREATE PROCEDURE HP_CHECK_CLIENT_FIO_FT639(
    FIND_C_ID DECIMAL(18,0),
    FIND_FIO VARCHAR(100),
    FIND_PASS_NO VARCHAR(30),
    FIND_INN VARCHAR(20),
    FIND_BIRTH_DATE DATE,
    IS_CRITICAL_FIO SMALLINT)
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(255),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_INN VARCHAR(20),
    C_COMMENT VARCHAR(100),
    C_COMMENT_ID INTEGER)
AS
declare variable find_c_birth_date date;
declare variable find_pass_no_orig varchar(30);
declare variable txt1 varchar(20);
declare variable txt2 varchar(20);
declare variable txt3 varchar(20);
declare variable ss_text varchar(100);
declare variable suspended integer;
declare variable fp_birth_date date;
declare variable found_fp integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable find_fio2 varchar(255);
declare variable find_fio3 varchar(255);
declare variable find_fio4 varchar(255);
declare variable find_inn_ind varchar(20);
declare variable c_type integer;
begin
  if (f_get_db_id() <> 81001) then
    exit;

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  if (find_pass_no is null) then
    find_pass_no = '';
  if (find_fio is null) then
    find_fio = '';

  c_comment_id = 0;
  c_type = 0;
  is_critical_fio = coalesce(is_critical_fio, 0);

  if (find_c_id > 0) then
  begin
    select iif(:find_fio = '', c_name_fio, :find_fio), iif(:find_pass_no = '', c_pass_no, :find_pass_no),
           iif(:find_inn = '', c_inn, :find_inn), iif(:find_c_birth_date is null, c_birth_date, :find_c_birth_date),
           coalesce(c_type, 0)
    from clients
    where c_id = :find_c_id
    into :find_fio, find_pass_no, find_inn, find_c_birth_date, c_type;

  end
  else
    find_c_birth_date = :find_birth_date;

  if (find_pass_no is null) then
    find_pass_no = '';
  if (find_inn is null) then
    find_inn = '';

  find_pass_no_orig = :find_pass_no;
  while (position(' ', find_pass_no) > 0) do
    find_pass_no = replace(:find_pass_no, ' ', '');
  find_fio = upper(:find_fio);
  while (position(' ', find_inn) > 0) do
    find_inn = replace(:find_inn, ' ', '');
  find_inn_ind = find_inn;
  suspended = 0;

-- Сначала проверяем по СРФ
  c_comment = '';
  c_comment_id = 0;
  find_fio2 = find_fio;
  find_fio3 = find_fio;

  find_fio2 = replace(find_fio, '-', ' ');
  find_fio2 = f_replace_all(find_fio2, '  ', ' ');
  find_fio3 = replace(find_fio2, ' уулу', '-уулу'); -- заменил igusev 26.05.2017  было find_fio3=replace(find_fio, ' уулу','-уулу');
  find_fio3 = replace(find_fio3, ' кызы', '-кызы');
  find_fio3 = f_replace_all(find_fio3, '  ', ' ');

  find_fio4 = find_fio3;
  if (find_fio4 starting 'ЧП ') then
    find_fio4 = substring(find_fio4 from 4);
  if (find_fio4 starting 'ИП ') then
    find_fio4 = substring(find_fio4 from 4);

--sd, rg, nd, gr
  for select distinct f.fps_fio, f.docseria, f.docnumber, f.inn_flip, f.birthdate
      from finpolis_639 f
      where (f.fps_fio = :find_fio or
            f.fps_fio = :find_fio2 or
            f.fps_fio = :find_fio3 or
            f.fps_fio = :find_fio4)
      into :c_name_fio, :txt1, :txt2, :txt3, :fp_birth_date
  do
  begin
    if (c_comment_id = 0) then
    begin
      if (is_critical_fio = 1) then
        c_comment_id = c_comment_id + 1;
      c_comment = 'Перечень 639: ФИО';
      if (txt1 is null) then
        txt1 = '';
      if (txt2 is null) then
        txt2 = '';
      if (txt3 is null) then
        txt3 = '';
      ss_text = txt1 || txt2;
      ss_text = replace(ss_text, ' ', '');
      if (find_pass_no <> '') then
      begin
        if (ss_text = find_pass_no) then
        begin
          c_comment_id = c_comment_id + 1;
          c_pass_no = find_pass_no;
          c_comment = c_comment || ' Паспорт;';
        end
      end
      if (find_inn <> '') then
      begin
        ss_text = replace(txt3, ' ', '');
        if (ss_text = find_inn) then
        begin
          c_comment_id = c_comment_id + 1;
          c_inn = find_inn;
          c_comment = c_comment || ' ИНН;';
        end
      end
      if (find_c_birth_date = fp_birth_date) then
      begin
        c_comment_id = c_comment_id + 1;
        c_birth_date = fp_birth_date;
        c_comment = c_comment || ' д.р.;';
      end
      if (c_comment_id > 0) then
      begin
        c_comment_id = 1;
        suspend;
      end
      suspended = 1;
      c_pass_no = :find_pass_no_orig;
      c_inn = :find_inn;
    end
  end
  if (suspended = 0) then
  begin
    for select distinct f.fps_fio, f.docseria, f.docnumber, f.inn_flip, f.birthdate
        from finpolis_639 f
        into c_name_fio, :txt1, :txt2, :txt3, :fp_birth_date
    do
    begin
      if (c_comment_id = 0) then
      begin
        found_fp = 0;
        c_comment = '';
        if (txt1 is null) then
          txt1 = '';
        if (txt2 is null) then
          txt2 = '';
        if (txt3 is null) then
          txt3 = '';
        ss_text = txt1 || txt2;
        ss_text = replace(ss_text, ' ', '');
        if (find_pass_no <> '') then
        begin
          if (ss_text = find_pass_no) then
          begin
            if (c_comment = '') then
              c_comment = 'Перечень 639:';
            found_fp = found_fp + 2;
            c_pass_no = find_pass_no;
            c_comment = c_comment || ' Паспорт;';
          end
        end
        ss_text = replace(txt3, ' ', '');
        if (find_inn <> '') then
        begin
          if (ss_text = find_inn) then
          begin
            if (c_comment = '') then
              c_comment = 'Перечень 639:';
            found_fp = found_fp + 2;
            c_inn = find_inn;
            c_comment = c_comment || ' ИНН;';
          end
        end
        if (found_fp > 0) then
        begin
          if (find_c_birth_date = fp_birth_date) then
          begin
            if (c_comment = '') then
              c_comment = 'Перечень 639:';
            found_fp = found_fp + 1;
            c_birth_date = fp_birth_date;
            c_comment = c_comment || ' д.р.;';
          end
        end
        if (found_fp > 1) then
        begin
          c_comment_id = 1;
          c_pass_no = :find_pass_no_orig;
          c_inn = :find_inn;
          suspend;
        end
      end
    end
  end
end
