CREATE PROCEDURE UXP$UXFRPARAM_VALUE(
    UFP_UUID D_UUID,
    UXA_UUID D_UUID,
    UA_UUID D_UUID,
    CALC_DEF_EXPRESSION D_BOOL = 0)
RETURNS (
    RES_FR_CODE D_STR_100,
    RES_FR_UUID D_UUID,
    RES_FIELD_NAME D_STR_CODE,
    RES_IS_PARAM D_BOOL,
    RES_DEF_VALUE D_STR_100,
    RES_SOURCE_CODE D_STR_100,
    RES_SOURCE_FR_UUID D_UUID,
    RES_SOURCE_FIELD D_STR_CODE,
    RES_SOURCE_IS_PARAM D_BOOL,
    RES_OWNER_UFP_UUID D_UUID,
    RES_OWNER_UFP_CODE D_STR_CODE,
    RES_USE_DEFAULT D_SMALLINT,
    RES_STATE D_SMALLINT,
    RES_LOCKED D_BOOL,
    RES_READONLY D_BOOL,
    RES_LOOKUP_CODE D_STR_100,
    RES_EXEC_PARAM_ID D_SMALLINT)
AS
declare variable ufp_parent d_uuid;
declare variable tmp_ufp_field d_str_code;
declare variable us_is_sql d_bool;
declare variable sql_param_used d_bool;
declare variable uf_uuid d_uuid;
declare variable edi_name d_str_code;
declare variable res_owner_ext_field d_str_code;
declare variable exp_text d_str_100;
declare variable new_date d_datetime;
declare variable ufp_editor d_str_code;
declare variable mm_uuid d_uuid;
declare variable main_ua_uuid d_uuid;
begin
  select ufp_uf_uuid,
         iif(uf_edi_uuid is not null, uxf$code(uf_edi_uuid, 'code'), null),
         ufp_parent, ufp_ufr_uuid, ufr_fcode,
         uf_field_name,
         iif(uf_is_param = 1, 1, iif(uf_pk_field <> 0 and us_is_table = 1, 1, 0)),
         iif(us_kind = -1 and trim(coalesce(us_sql, '')) <> '', 1, 0),
         ufp_editor
  from ux$frparam, ux$frame, ux$source, ux$field
  where ufp_uuid = :ufp_uuid and
        ufr_uuid = ufp_ufr_uuid and
        ufr_us_uuid = us_uuid and
        ufp_uf_uuid = uf_uuid
  into uf_uuid, edi_name, ufp_parent, res_fr_uuid, res_fr_code,
       res_field_name, res_is_param,
       us_is_sql, ufp_editor;

  res_def_value = '';
  res_owner_ufp_code = null;
  res_source_code = '';
  res_source_fr_uuid = null;
  res_source_field = '';
  res_source_is_param = 0;
  res_use_default = -1;
  res_state = 0;
  res_locked = 0;
  res_readonly = 0;
  res_exec_param_id = 0;

  res_lookup_code = '';

  mm_uuid = uxf$mainmenu();

  select ua_uuid
  from ux$action
  where ua_owner = :mm_uuid and
        ua_uxa_uuid = :uxa_uuid and
        ua_usr_id_field = ''
  into main_ua_uuid;

  if (main_ua_uuid is not null) then
  begin
    select uap_exec_param_id
    from ux$actionparam
    where uap_ua_uuid = :main_ua_uuid and
          uap_exec_param_id <> 0 and
          (uap_param_name = :res_field_name or uap_param_name = 'out$' || :res_field_name)
    into res_exec_param_id;
  end

  sql_param_used = 0;
  if (us_is_sql = 1) then
  begin
    select uf_default
    from ux$field
    where uf_uuid = :uf_uuid and
          uf_is_param = 1
    into res_def_value;

    if (coalesce(res_def_value, '') <> '') then
    begin
      res_state = 2;
      res_use_default = 1;
      res_locked = 1;
      sql_param_used = 1;
    end
  end

  if (res_locked = 0) then
  begin
    if (edi_name = 'current$reverse') then
    begin
      res_state = 2;
      res_use_default = 1;
      res_locked = 1;
  
      res_def_value = 0;
    end else
    if (edi_name = 'current$user') then
    begin
      res_state = 2;
      res_use_default = 1;
      res_locked = 1;
  
      res_def_value = uxf$user();
    end else
    if (edi_name = 'current$action_id') then
    begin
      res_state = 2;
      res_use_default = 1;
      res_locked = 1;
  
      res_def_value = '';
      if (ua_uuid is not null) then
        res_def_value = lower(uuid_to_char(ua_uuid));
    end else
    begin
      select upd_def_value, upd_locked
      from ux$paramdef
      where upd_ufp_uuid = :ufp_uuid and
            upd_uxa_uuid is null
      into res_def_value, res_locked;
    
      if (row_count = 1) then
      begin
        res_use_default = 1;
        res_state = 2;
      end
    end
  end

  if (res_locked = 0 and uxa_uuid is not null) then
  begin
    select upd_def_value, upd_locked
    from ux$paramdef
    where upd_ufp_uuid = :ufp_uuid and
          upd_uxa_uuid = :uxa_uuid
    into res_def_value, res_locked;

    if (row_count = 1) then
    begin
      res_use_default = 1;
      res_state = 3;
    end
  end

  if (res_locked = 0 and ua_uuid is not null) then
  begin
    select uap_owner_ufp_uuid, coalesce(uap_owner_ext_field, ''), uap_default,
           uap_use_default, uap_locked
    from ux$actionparam
    where uap_ua_uuid = :ua_uuid and
          (uap_param_name = :res_field_name or uap_param_name = 'out$' || :res_field_name)
    into res_owner_ufp_uuid, res_owner_ext_field, res_def_value,
         res_use_default, res_locked;

    if (row_count = 1) then
      res_state = 4;
    else
    if (res_exec_param_id <> 0) then
      res_state = 3;

    if (res_owner_ufp_uuid is not null and res_use_default = 0) then
    begin
      select ufr_uuid, ufr_fcode, uf_field_name, uf_is_param
      from ux$frparam, ux$field, ux$frame, ux$source
      where ufp_uuid = :res_owner_ufp_uuid and             
            ufp_uf_uuid = uf_uuid and
            ufp_ufr_uuid = ufr_uuid and
            uf_owner = us_uuid
      into res_source_fr_uuid, res_source_code, res_source_field, res_source_is_param;
    end else
    if (res_owner_ext_field <> '' and res_use_default = 0 and ua_uuid is not null) then
    begin
      res_source_fr_uuid = null;
      res_source_code = '';
      res_source_field = res_owner_ext_field;
      res_source_is_param = 0;
    end
  end

  tmp_ufp_field = '';
  if (ufp_parent is not null) then
  begin
    select uf_field_name
    from ux$frparam, ux$field
    where ufp_uuid = :ufp_parent and
          ufp_uf_uuid = uf_uuid
    into tmp_ufp_field;

    res_is_param = 1;
    res_lookup_code = res_fr_code || '$' || tmp_ufp_field;
    res_field_name = tmp_ufp_field || '$' || res_field_name; -- Название параметра Lookup
  end

  if (sql_param_used = 0) then
  begin
    if ((res_state = 2 and uxa_uuid is null and ua_uuid is null) or
        (res_state = 3 and ua_uuid is null) or
         res_state = 4) then
      res_state = 1;
  end

  if (res_locked = 1) then
  begin
    if (res_state <> 1) then
      res_readonly = 1;

    res_state = -res_state;
  end

  if (sql_param_used = 1) then
    res_readonly = 1;
  else
  if (res_state not in (-1, 1)) then
    res_use_default = -1;

  res_owner_ufp_code = null;
  if (res_owner_ufp_uuid is not null) then
    res_owner_ufp_code = uuid_to_char(res_owner_ufp_uuid);

  if (coalesce(calc_def_expression, 0) = 1 and res_def_value starting '=') then
  begin
    exp_text = substring(res_def_value from 2);
    res_def_value = '';
  
    begin
      execute statement 'select ' || exp_text || ' from rdb$database'
      into res_def_value;

      when any do
        res_def_value = '';
    end

    if (res_def_value <> '') then
    begin
      if (ufp_editor = 'date') then
      begin
        new_date = res_def_value;
        res_def_value = f_datetostr(new_date);
        when any do
          res_def_value = '';
      end else
      if (ufp_editor = 'datetime') then
      begin
        new_date = res_def_value;
        res_def_value = f_datetimetostr(new_date, 'dd.mm.yyyy hh:nn:ss');
        when any do
          res_def_value = '';
      end
    end
  end

  suspend;
end
