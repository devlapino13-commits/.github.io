CREATE PROCEDURE AAREP_CL_NOT_NATION(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    KC_AMOUNT_APPROVED DECIMAL(15,2),
    KNC_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_NATION_STR VARCHAR(100),
    C_CITIZEN_STR VARCHAR(100),
    KAL_NEW_KF_NAME VARCHAR(100),
    KF_OFFICER_NAME VARCHAR(100),
    KS_STATE_STR VARCHAR(50),
    K_STATUS_STR VARCHAR(50))
AS
declare variable k_id decimal(18,0);
declare variable c_id decimal(18,0);
declare variable c_nation integer;
declare variable c_citizen integer;
declare variable ks_state integer;
declare variable k_status integer;
declare variable parent_id integer;
declare variable reg_prior integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  reg_prior = 0;
  for select reg_code || ' ' || reg_name, k_id, k_kontract_no, k_disb_date, kc_amount_approved, c1.c_id, c1.c_name_fio,
             c1.c_pass_no, c1.c_nation, dd1.rd_name, c1.c_citizen, dd2.rd_name, kal_new_kf_name, c2.c_name_fio,
             ks_state,
             case ks_state
               when 0 then 'В процессе'
               when 1 then 'Активный'
               when 2 then 'Закрытый'
             end,
             k_status, region.parent,
             case k_status
               when -1 then 'Аннулирован'
               when -2 then 'Заявка'
               when 0 then 'В процессе'
               when 1 then 'Контракт'
             end
      from clients c1, clients c2, knclient, kontract, refdata dd1, refdata dd2, region, region_list(:rep_region_id, 1, 3, 1), knaktlist, kontstate
      where (c1.c_nation = 0 or
            c1.c_citizen = 0) and
            kc_from_c_id = c1.c_id and
            k_id = kc_from_k_id and
            kc_amount_approved <> 0 and
            dd1.rd_id = c1.c_nation and
            dd2.rd_id = c1.c_citizen and
            kal_from_kf_id = k_from_kf_id and
            :rep_date_end between kal_date and kal_date_end and
            (reg_unicode = id_list) and
            kal_new_reg_id = reg_unicode and
            reg_status = 3 and
            knaktlist.is_delete = 0 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_ol_is_delete = 0 and
            ks_from_k_id = k_id and
            kal_new_kf_officer = c2.c_id
      order by parent, reg_code, c2.c_name_fio
      into :reg_name, :k_id, :k_kontract_no, :k_disb_date, :kc_amount_approved, :c_id, :knc_name_fio, :c_pass_no,
           :c_nation, :c_nation_str, :c_citizen, :c_citizen_str, :kal_new_kf_name, :kf_officer_name, :ks_state,
           :ks_state_str, :k_status, :parent_id, :k_status_str
  do
  begin
    if (parent_id <> reg_prior) then
    begin
      select reg_name
      from region
      where reg_unicode = :parent_id
      into :parent_name;
      reg_prior = parent_id;
    end

    suspend;
  end
end
