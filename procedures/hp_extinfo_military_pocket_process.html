CREATE PROCEDURE HP_EXTINFO_MILITARY_POCKET_PROCESS(
    EIL_ID D_ID)
AS
declare variable EIL_STATUS D_SMALLINT;
declare variable EIL_STATUS_DATETIME D_DATETIME;
declare variable JSON_STR D_BLOB_TEXT;
declare variable I D_INTEGER;
declare variable ARR_LENGTH D_INTEGER;
declare variable MCP_ID D_INTEGER;
declare variable ANSWER_DATE D_DATE;
declare variable ANSWER_STATUS D_BOOLEAN;
declare variable INN D_STR_20;
declare variable C_ID D_ID;
declare variable MCP_CREATE_DATE D_DATETIME;
declare variable UN_ID D_ID;
begin
  -- RFCRU-3817 получения информации о статусе прохождения воинской службы в КЗ при покетной обработке
  select cast(blx_blob as d_blob_text),
         eil.eil_status,
         eil.eil_status_datetime
  from extinfolist eil, extinfocfg, hp_get_blobex_blob(eil_from_bls_id)
  where eic_code = 'military_pocket' and
        eil_from_eic_id = eic_id and
        eil_id = :eil_id rows 1
  into json_str,
       eil_status,
       eil_status_datetime;

  if (eil_status = -2) then    -- Если ошибка
  begin
    update military_check_pocket mcp
      set mcp.mcp_status = -2
    where mcp.mcp_from_eil_id = :eil_id;
  end

  if (eil_status = 2) then -- Если мы получили успешный ответ
  begin
    select mcp.mcp_id,
           mcp.mcp_create_date,
           mcp.mcp_from_un_id
    from military_check_pocket mcp
    where mcp.mcp_from_eil_id = :eil_id
    into mcp_id,
         mcp_create_date,
         un_id;

    arr_length = f_json_blob_get_count(json_str, 'data') - 1;
    i = 0;

    while (i <= arr_length) do  -- Проходимся по списку
    begin
      answer_date = cast(f_json_str_get_value(json_str, f_format('data[%1].date', i), null) as d_date);
      answer_status = cast(f_json_str_get_value(json_str, f_format('data[%1].status', i), null) as d_boolean);
      inn = cast(f_json_str_get_value(json_str, f_format('data[%1].iin', i), null) as d_str_20);

      if (coalesce(answer_date, '') <> '' and coalesce(answer_status, '') <> '' and coalesce(inn, '') <> '') then
      begin
        select c.c_id
        from clients c
        where c.c_inn = :inn
        into c_id;
        -- Добавляем запрос по клиенту в таблицу с запросами
        insert into military_check_query (mcq_from_c_id,
                                          mcq_inn,
                                          mcq_status,
                                          mcq_answer_date,
                                          mcq_received,
                                          mcq_from_un_id,
                                          mcq_creation_stage,
                                          mcq_status_datetime,
                                          mcq_create_datetime,
                                          mcq_from_eil_id,
                                          mcq_from_mcp_id)
        values (:c_id,
                :inn,
                iif(:answer_status, 1, 0),
                :answer_date,
                1,
                :un_id,
                f_enum_id('military_request_source', 'pocket_query'),
                current_timestamp,
                current_timestamp,
                :eil_id,
                :mcp_id);
      end

      i = i + 1;
    end

    update military_check_pocket mcp
      set mcp.mcp_status = 2
    where mcp.mcp_id = :mcp_id;
  end

end
