CREATE PROCEDURE RESTORE_CLIENT_FROM_DELETE_BUFFER(
    C_ID D_ID NOT NULL,
    UN_ID D_ID NOT NULL)
AS
begin
    -- RFCRU-5595
    -- Перед тем как обновлять клиента, мы должны отключаем все триггеры таблицы clients
    rdb$set_context('USER_TRANSACTION', 'IGNORE_C_ID', :c_id);

    /*
        Это update запрос, где мы берём запись с таблицы client_delete_buffer и переносим их обратно в таблицу clients
        Данное решение было применено, чтобы избежать и без того здорового select запроса, где мы вытаскиваем все поля
        И перегоняем их в не менее здоровый update запрос
    */

    merge into clients target
    using (
        select
            cdb_from_c_id, c_type, c_name_fio, c_sex, c_birth_date, c_nation, c_citizen, c_pass_no, c_pass_date, c_pass_who,
            c_pass_date_expire, c_from_reg_id, c_email, c_family, c_education, c_kol_family, c_kol_child, c_memo, c_reg_no,
            c_okpo, c_inn, c_ext_inn, c_account, c_org_from_b_id, c_hired_date, c_is_resident, c_army, c_tabel,
            c_plastcard_no, c_others, c_is_personal, c_doc_type, c_form_status_long, c_form_status_short, c_org_type_long,
            c_org_type_short, c_original_id, c_fam_business, c_from_un_id, c_datetime, c_index_c_pass_no,
            c_finpolis_send_date, c_finpolis_proceed, c_pincode, c_from_fl_id, c_fam_adl_id, c_blob_photo_id,
            c_blob_pass1_id, c_blob_pass2_id, c_kib_id, c_birth_year, c_verify_status, c_verify_datetime, c_from_is_id,
            c_pass_last_change, c_pft_risk_koef, c_pft_risk_memo, c_alternat_fio, c_alternat_fl_id, c_birth_country,
            c_birth_city, c_pass_who_code, c_birthday_commited, c_source_system, c_name_full, c_is_public, c_type_contr,
            c_is_founder, c_is_benefic_owner, c_name_eng, c_name_eng_fl_id, c_blank_lang, c_authorized_capital, c_branches,
            c_doc_inform, c_goals, c_current_gni_categ
        from client_delete_buffer
        where cdb_from_c_id = :c_id
    ) source
    on (target.c_id = :c_id)
    when matched then
        update set 
            target.c_type                  = source.c_type,
            target.c_name_fio              = source.c_name_fio,
            target.c_sex                   = source.c_sex,
            target.c_birth_date            = source.c_birth_date,
            target.c_nation                = source.c_nation,
            target.c_citizen               = source.c_citizen,
            target.c_pass_no               = source.c_pass_no,
            target.c_pass_date             = source.c_pass_date,
            target.c_pass_who              = source.c_pass_who,
            target.c_pass_date_expire      = source.c_pass_date_expire,
            target.c_from_reg_id           = source.c_from_reg_id,
            target.c_email                 = source.c_email,
            target.c_family                = source.c_family,
            target.c_education             = source.c_education,
            target.c_kol_family            = source.c_kol_family,
            target.c_kol_child             = source.c_kol_child,
            target.c_memo                  = source.c_memo,
            target.c_reg_no                = source.c_reg_no,
            target.c_okpo                  = source.c_okpo,
            target.c_inn                   = source.c_inn,
            target.c_ext_inn               = source.c_ext_inn,
            target.c_account               = source.c_account,
            target.c_org_from_b_id         = source.c_org_from_b_id,
            target.c_hired_date            = source.c_hired_date,
            target.c_is_resident           = source.c_is_resident,
            target.c_army                  = source.c_army,
            target.c_tabel                 = source.c_tabel,
            target.c_plastcard_no          = source.c_plastcard_no,
            target.c_others                = source.c_others,
            target.c_is_personal           = source.c_is_personal,
            target.c_doc_type              = source.c_doc_type,
            target.c_form_status_long      = source.c_form_status_long,
            target.c_form_status_short     = source.c_form_status_short,
            target.c_org_type_long         = source.c_org_type_long,
            target.c_org_type_short        = source.c_org_type_short,
            target.c_original_id           = source.c_original_id,
            target.c_fam_business          = source.c_fam_business,
            target.c_from_un_id            = source.c_from_un_id,
            target.c_datetime              = source.c_datetime,
            target.c_index_c_pass_no       = source.c_index_c_pass_no,
            target.c_finpolis_send_date    = source.c_finpolis_send_date,
            target.c_finpolis_proceed      = source.c_finpolis_proceed,
            target.c_pincode               = source.c_pincode,
            target.c_from_fl_id            = source.c_from_fl_id,
            target.c_fam_adl_id            = source.c_fam_adl_id,
            target.c_blob_photo_id         = source.c_blob_photo_id,
            target.c_blob_pass1_id         = source.c_blob_pass1_id,
            target.c_blob_pass2_id         = source.c_blob_pass2_id,
            target.c_kib_id                = source.c_kib_id,
            target.c_birth_year            = source.c_birth_year,
            target.c_verify_status         = source.c_verify_status,
            target.c_verify_datetime       = source.c_verify_datetime,
            target.c_from_is_id            = source.c_from_is_id,
            target.c_pass_last_change      = source.c_pass_last_change,
            target.c_pft_risk_koef         = source.c_pft_risk_koef,
            target.c_pft_risk_memo         = source.c_pft_risk_memo,
            target.c_alternat_fio          = source.c_alternat_fio,
            target.c_alternat_fl_id        = source.c_alternat_fl_id,
            target.c_birth_country         = source.c_birth_country,
            target.c_birth_city            = source.c_birth_city,
            target.c_pass_who_code         = source.c_pass_who_code,
            target.c_birthday_commited     = source.c_birthday_commited,
            target.c_source_system         = source.c_source_system,
            target.c_name_full             = source.c_name_full,
            target.c_is_public             = source.c_is_public,
            target.c_type_contr            = source.c_type_contr,
            target.c_is_founder            = source.c_is_founder,
            target.c_is_benefic_owner      = source.c_is_benefic_owner,
            target.c_name_eng              = source.c_name_eng,
            target.c_name_eng_fl_id        = source.c_name_eng_fl_id,
            target.c_blank_lang            = source.c_blank_lang,
            target.c_authorized_capital    = source.c_authorized_capital,
            target.c_branches              = source.c_branches,
            target.c_doc_inform            = source.c_doc_inform,
            target.c_goals                 = source.c_goals,
            target.c_current_gni_categ     = source.c_current_gni_categ;


    -- устанавливаем флаг "восстановлено"
    update client_delete_buffer cdb
    set cdb.cdb_is_restored = 1
    where cdb.cdb_from_c_id = :c_id;

    -- устанавливаем csr_is_restored = 1, чтобы такого клиента больше не вносить в буфер на удаление
    update clientstate_ru csr
    set csr.csr_is_restored = 1,
        csr.csr_is_restored_user = :un_id,
        csr.csr_is_restored_date = current_date
    where csr.csr_from_c_id = :c_id;
end
