CREATE PROCEDURE HP_AFK_CHECK(
    F_MASTER D_STR_64,
    REC_EVENT D_STR_64,
    N1_ID D_STR_64 = null,
    O1_ID D_STR_64 = null,
    N2_ID D_STR_64 = null,
    O2_ID D_STR_64 = null,
    N3_ID D_STR_64 = null,
    O3_ID D_STR_64 = null,
    N4_ID D_STR_64 = null,
    O4_ID D_STR_64 = null,
    N5_ID D_STR_64 = null,
    O5_ID D_STR_64 = null)
AS
declare variable v_text_sql d_str_2048;
begin
  for select text_sql
      from hp_afk_get_sql(:f_master, :rec_event)
      into :v_text_sql
  do
  begin
    if (rec_event = 'deleting') then
      begin
        if (position('o5_id' in v_text_sql) > 0 ) then
            execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id, o3_id := :o3_id, o4_id := :o4_id, o5_id := :o5_id);
        else
        if (position('o4_id' in v_text_sql) > 0 ) then
            execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id, o3_id := :o3_id, o4_id := :o4_id);
        else
        if (position('o3_id' in v_text_sql) > 0 ) then
            execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id, o3_id := :o3_id);
        else
        if (position('o2_id' in v_text_sql) > 0 ) then
            execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id);
        else
            execute statement(:v_text_sql)(o1_id := :o1_id);
      end
    else
    if (rec_event = 'updating') then
      begin
        if (position(':n1_id' in v_text_sql) > 0) then
          begin
            if (position('o5_id' in v_text_sql) > 0 ) then
              execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id, o3_id := :o3_id, o4_id := :o4_id, o5_id := :o5_id,
                                             n1_id := :n1_id, n2_id := :n2_id, n3_id := :n3_id, n4_id := :n4_id, n5_id := :n5_id);
            else
            if (position('o4_id' in v_text_sql) > 0 ) then
              execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id, o3_id := :o3_id, o4_id := :o4_id,
                                             n1_id := :n1_id, n2_id := :n2_id, n3_id := :n3_id, n4_id := :n4_id);
            else
            if (position('o3_id' in v_text_sql) > 0 ) then
              execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id, o3_id := :o3_id,
                                             n1_id := :n1_id, n2_id := :n2_id, n3_id := :n3_id);
            else
            if (position('o2_id' in v_text_sql) > 0 ) then
              execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id,
                                             n1_id := :n1_id, n2_id := :n2_id);
            else
              execute statement(:v_text_sql)(o1_id := :o1_id,
                                             n1_id := :n1_id);

          end
        else
          begin
            if (position('o5_id' in v_text_sql) > 0 ) then
              execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id, o3_id := :o3_id, o4_id := :o4_id, o5_id := :o5_id);
            else
            if (position('o4_id' in v_text_sql) > 0 ) then
              execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id, o3_id := :o3_id, o4_id := :o4_id);
            else
            if (position('o3_id' in v_text_sql) > 0 ) then
              execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id, o3_id := :o3_id);
            else
            if (position('o2_id' in v_text_sql) > 0 ) then
              execute statement(:v_text_sql)(o1_id := :o1_id, o2_id := :o2_id);
            else
              execute statement(:v_text_sql)(o1_id := :o1_id);
          end
      end

  end
end
