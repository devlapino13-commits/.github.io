CREATE PROCEDURE HP_ADDRESS_MAKE_STR(
    PL_ID D_ID,
    STREET_ID D_ID,
    DEPART_NO D_STR_20,
    DEPART_PREF D_STR_20,
    FLAT_NO D_STR_20,
    BUILDING_NO D_STR_20,
    SECTION_NO D_STR_20,
    POST_INDEX D_STR_20,
    IS_WITHOUT_STREET D_SMALLINT,
    NEW_STREET_NAME D_STR_255,
    NEW_STREET_TYPE D_INTEGER,
    LANG_KIND D_SMALLINT,
    UN_ID D_ID,
    VLADEN_NO D_STR_20 = null)
RETURNS (
    ADL_ADDRESS_STR D_STR_255,
    ADL_STREET_ID D_ID,
    HAS_ANY_NO D_INTEGER)
AS
declare variable DB_ID D_INTEGER;
declare variable ADDR_DOM D_STR_20;
declare variable ADDR_KV D_STR_20;
declare variable ADDR_STR D_STR_20;
declare variable ADDR_KORP D_STR_20;
declare variable STR D_STR_255;
declare variable ADDR_VLADEN D_STR_20;
begin
  -- Процедура формирует текст адреса по переданным параметрам

  db_id = f_get_db_id();
  lang_kind = coalesce(lang_kind, 0);

  if (db_id not in (39801, 41701)) then
    lang_kind = 0;

  depart_no = coalesce(depart_no, '');     -- RFCRU-4498  Исправил ошибку при редактировании адреса в окне "Редактирование адреса" (КК Перс данные)
  depart_pref = coalesce(depart_pref, '');
  flat_no = coalesce(flat_no, '');
  building_no = coalesce(building_no, '');
  section_no = coalesce(section_no, '');
  vladen_no = coalesce(vladen_no, '');

  post_index = coalesce(post_index, '');
  if (char_length(post_index) <= 3) then
    post_index = '';

  if (coalesce(street_id, 0) > 0 and is_without_street = 1 and street_id <> pl_id) then
    is_without_street = 0;

  -- Если нет улицы, создаем новую
  if (coalesce(street_id, 0) <= 0 and
      new_street_name <> '' and
      is_without_street = 0) then
  begin
    street_id = 0;
    select pl_id
    from places
    where pl_name = :new_street_name and
          parent = :pl_id
    order by pl_id
    rows 1
    into :street_id;

    if (coalesce(street_id, 0) <= 0) then
    begin
      insert into places (parent, pl_name, pl_status, pl_from_un_id, pl_post_index)
      values (:pl_id, :new_street_name, :new_street_type, :un_id, :post_index)
      returning (pl_id)
      into :street_id;
    end
  end
  adl_street_id = street_id;

  adl_address_str = '';
  -- Сформировали адрес до улицы
  if (coalesce(street_id, 0) > 0) then
  begin
    select place_full
    from aarep_places_lang_link(:street_id, :lang_kind)
    into :adl_address_str;
    if (adl_address_str = 'Не указан') then
      adl_address_str = '';
  end

  if (adl_address_str = '') then
  begin
    select place_full
    from aarep_places_lang_link(:pl_id, :lang_kind)
    into :adl_address_str;
  end

  -- Формируем целиком адрес включая номера дома и т.д.
  if (db_id = 94401) then
  begin
    addr_dom = 'ev.';
    addr_kv = 'm.';
    addr_str = 'tik.';
    addr_korp = 'korp.';
    addr_vladen = '';
  end
  else
  begin
    if (lang_kind = 0) then
    begin
      addr_dom = 'д.'; -- дом
      addr_kv = 'кв.'; -- квартира
      addr_str = 'стр.'; -- строение
      addr_korp = 'корп.'; -- корпус
      addr_vladen = 'вл.';
    end else
    begin
      addr_dom = ''; -- дом
      addr_kv = '-бат'; -- квартира
      addr_str = '-курул.'; -- строение
      addr_korp = 'б.'; -- корпус
      addr_vladen = 'ээлик кылуу ';
    end
  end

  if (post_index <> '') then
    adl_address_str = post_index || ', ' || adl_address_str;
  has_any_no = 0;

  if (depart_no <> '') then
  begin
    has_any_no = 1;
    if (lang_kind = 1) then
    begin
      if (depart_pref <> '')
      then adl_address_str = adl_address_str || ', ' || depart_no || '/' || depart_pref || addr_dom;
      else adl_address_str = adl_address_str || ', ' || depart_no || addr_dom;
    end
    else
    begin
      adl_address_str = adl_address_str || ', ' || addr_dom || depart_no;
      if (depart_pref <> '') then
        adl_address_str = adl_address_str || '/' || depart_pref;
    end
  end
  if (flat_no <> '') then
  begin
    has_any_no = 1;
    if (lang_kind = 1)
    then adl_address_str = adl_address_str || ', ' || flat_no || addr_kv;
    else adl_address_str = adl_address_str || ', ' || addr_kv || flat_no;
  end
  if (building_no <> '') then
  begin
    has_any_no = 1;
    if (lang_kind = 1)
    then adl_address_str = adl_address_str || ', ' || building_no || addr_str;
    else adl_address_str = adl_address_str || ', ' || addr_str || building_no;
  end
  if (section_no <> '') then
  begin
    has_any_no = 1;
    if (lang_kind = 1)
    then adl_address_str = adl_address_str || ', ' || section_no || addr_korp;
    else adl_address_str = adl_address_str || ', ' || addr_korp || section_no;
  end
  if (vladen_no <> '') then
  begin
    has_any_no = 1;
    if (lang_kind = 1)
    then adl_address_str = adl_address_str || ', ' || vladen_no || addr_vladen;
    else adl_address_str = adl_address_str || ', ' || addr_vladen || vladen_no;
  end

  if (has_any_no = 0) then
    adl_address_str = adl_address_str || iif(db_id <> 94401, ', б/н.', ', Nömrəsiz ev');

  while (position('  ', adl_address_str) > 0) do
    adl_address_str = replace(adl_address_str, '  ', ' ');

  for select iif(:lang_kind = 0, plt_short_name, plt_alt_short_name)
      from placetype
      where iif(:lang_kind = 0, plt_short_name, plt_alt_short_name) like '%.' and
            iif(:lang_kind = 0, plt_short_name_side, plt_alt_short_name_side) = 1
      into str
  do
  begin
    if (position(:str || ' ', adl_address_str) > 0) then
      adl_address_str = replace(adl_address_str, :str || ' ', :str || ', ');
  end
  if (adl_address_str like '%. %') then
    adl_address_str = replace(adl_address_str, '. ', '.');

  adl_address_str = replace(adl_address_str, ' ,', ',');

  if (street_id > 0 and pl_id > 0 and street_id <> pl_id) then
  begin
    if (not exists(select * from places
                   where pl_id = :street_id and
                         parent = :pl_id)) then
      adl_address_str = 'Улица из другого населенного пункта';
  end

  suspend;
end
