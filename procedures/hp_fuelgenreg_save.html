CREATE PROCEDURE HP_FUELGENREG_SAVE(
    ID D_INTEGER,
    FG_DTWRITEOFF D_DATE,
    FG_DTBEG D_DATE,
    FG_DTEND D_DATE,
    FG_OFFICE_ID D_INTEGER,
    FG_PET_ID D_INTEGER,
    FG_START D_CURRENCY,
    FG_START_SUM D_CURRENCY,
    FG_FILLUP_SUM D_CURRENCY,
    FG_FILLUP D_CURRENCY,
    FG_WHOURS D_CURRENCY,
    FG_WRATE D_CURRENCY,
    FG_USEDUP D_CURRENCY,
    FG_USEDUP_SUM D_CURRENCY,
    FG_END D_CURRENCY,
    FG_END_SUM D_CURRENCY,
    FG_DRIVER_ID D_ID,
    FG_MANAGER_ID D_ID,
    FG_BOOKER_ID D_ID)
RETURNS (
    FG_ID D_INTEGER,
    ERR_TXT D_STR_255)
AS
begin
  err_txt = (
    select err_txt
    from hp_fuelgenreg_check(
      :id,
      :fg_dtwriteoff,
      :fg_dtbeg,
      :fg_dtend,
      :fg_office_id,
      :fg_pet_id,
      :fg_start,
      :fg_start_sum,
      :fg_fillup,
      :fg_fillup_sum,
      :fg_whours,
      :fg_wrate,
      :fg_usedup,
      :fg_usedup_sum,
      :fg_end,
      :fg_end_sum,
      :fg_driver_id,
      :fg_manager_id,
      :fg_booker_id));

  if (err_txt = '') then
  update or insert into fuelgenreg(
      fg_id,
      fg_dtwriteoff,
      fg_dtbeg,
      fg_dtend,
      fg_office_id,
      fg_pet_id,
      fg_start,
      fg_start_sum,
      fg_fillup,
      fg_fillup_sum,
      fg_whours,
      fg_wrate,
      fg_usedup,
      fg_usedup_sum,
      fg_end,
      fg_end_sum,
      fg_driver_id,
      fg_manager_id,
      fg_booker_id)
  values (
      :id,
      :fg_dtwriteoff,
      :fg_dtbeg,
      :fg_dtend,
      :fg_office_id,
      :fg_pet_id,
      :fg_start,
      :fg_start_sum,
      :fg_fillup,
      :fg_fillup_sum,
      :fg_whours,
      :fg_wrate,
      :fg_usedup,
      :fg_usedup_sum,
      :fg_end,
      :fg_end_sum,
      :fg_driver_id,
      :fg_manager_id,
      :fg_booker_id)
  matching (fg_id)
  returning fg_id
  into
    :fg_id;

  suspend;
end
