CREATE PROCEDURE FULL_WP_LIST(
    KEY_ID DECIMAL(18,0),
    IS_TREE SMALLINT)
RETURNS (
    FWP_ID DECIMAL(18,0),
    FWP_PARENT DECIMAL(18,0),
    FWP_NAME VARCHAR(100),
    FWP_STATUS INTEGER,
    FWP_CAN_INCLUDE INTEGER,
    FWP_REGION_ID INTEGER)
AS
declare variable parent_id decimal(18,0);
declare variable wp_id_temp decimal(18,0);
declare variable reg_id_temp decimal(18,0);
begin
  /* Procedure Text */
  if (key_id > 9999999) then
  begin
    for select id_list, id_parent, wp_name, wp_status, wp_can_include, wp_from_reg_id
        from workplace_list(:key_id, 1, -1, 1), workplace
        where id_list = wp_id
        into :fwp_id, :fwp_parent, :fwp_name, :fwp_status, :fwp_can_include, :fwp_region_id
    do
      suspend;
  end
  else
  begin
    for select id_list, id_parent, reg_name, reg_status, 0, id_list
        from region_orig_list(:key_id, 1, -1, 1), region
        where id_list = reg_unicode
        into :fwp_id, :fwp_parent, :fwp_name, :fwp_status, :fwp_can_include, :fwp_region_id
    do
    begin
      reg_id_temp = :fwp_id;
      if (is_tree = 1) then
        suspend;
      for select distinct wp_id, wp_name, wp_status, wp_from_reg_id
          from workplace
          where wp_from_reg_id = :reg_id_temp and
                (parent is null)
          into :wp_id_temp, :fwp_name, :fwp_status, :fwp_region_id
      do
      begin
        for select id_list, id_parent, wp_name, wp_status, wp_can_include, wp_from_reg_id
            from workplace_list(:wp_id_temp, 1, -1, 1), workplace
            where id_list = wp_id
            into :fwp_id, :fwp_parent, :fwp_name, :fwp_status, :fwp_can_include, :fwp_region_id
        do
        begin
          if (fwp_parent is null) then
            fwp_parent = :reg_id_temp;
          suspend;
        end
      end
    end
  end

end
