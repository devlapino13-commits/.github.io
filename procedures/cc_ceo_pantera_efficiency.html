CREATE PROCEDURE CC_CEO_PANTERA_EFFICIENCY(
    REP_DATE DATE,
    STEP INTEGER,
    REP_PERIOD VARCHAR(10))
RETURNS (
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    CALLED_OPER_STR STR_255,
    ON_PANTERA_QTY INTEGER,
    DELINQ_SUM CURR_TYPE,
    IS_PERIOD_BLOCKED INTEGER,
    PAYED_AFTER_CALL DECIMAL(18,0),
    IS_PAYED_AFTER_CALL INTEGER,
    P_LESS_30 INTEGER,
    P_30_AND_MORE INTEGER,
    IS_GIVED INTEGER)
AS
declare variable k_id id_type;
declare variable pant_date_begin date;
declare variable delinq_days integer;
declare variable i integer;
declare variable rep_date_tmp date;
declare variable max_i integer;
declare variable active_end blob sub_type 1 segment size 80;
declare variable end_str str_100;
declare variable begin_str str_255;
begin
  if (step = 1) then
  begin
    is_gived = 0;
    payed_after_call = 0;
    is_payed_after_call = 0;
    rep_date_end = rep_date;

    if (rep_period = 'day') then
      rep_date_begin = rep_date_end;
    if (rep_period = 'month') then
      rep_date_begin = (select cast('01.' || cast(extract(month from cast(:rep_date as date)) as varchar(2)) || '.' || cast(extract(year from cast(:rep_date as date)) as varchar(4)) as date)
                        from rdb$database);

    active_end = '';
    for select pantk.pant_oper_called, sba.sba_from_k_id, pantk.pant_date_begin,
               pant_oper_called || '-' || cast(sba.sba_from_k_id as varchar(18))
        from pantera_kontracts pantk
        inner join kontstate ks on ks.ks_from_k_id = pantk.pant_from_k_id
        inner join sb_akt sba on sba.sba_from_k_id = pantk.pant_from_k_id
        where pantk.pant_date_end >= pantk.pant_date_begin and
              :rep_date_end between pantk.pant_date_begin and pantk.pant_date_end and
              pantk.pant_date_begin >= sba.sba_date_begin and
              pantk.pant_project_type = 1 and
              sba.sba_date_end >= :rep_date_end and
              sba.sba_date_begin <= sba.sba_date_end and
              sba_is_delete = 0 and
              sba.sba_from_sbl_id = 82 and
              :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1
        into :called_oper_str, :k_id, :pant_date_begin, :end_str
    do
    begin
      delinq_days = null;
      on_pantera_qty = 1;

      active_end = blob_append(active_end, end_str);

      select cast(:rep_date_end as date) - oks_delinq_date as delinq_days
      from oksummary oks
      where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
            cast(:rep_date_end as date) - oks_delinq_date >= 1 -- кредит должне быть в просрочке
            and
            oks_delinq_date < :rep_date and
            oks.oks_close_state = 0 and
            oks.oks_from_k_id = :k_id
      into :delinq_days;

      delinq_days = coalesce(delinq_days, 0);

      delinq_sum = 0;

      if (delinq_days > 0) then
      begin
        delinq_sum = null;

        select sum(zad_os + sh_calc_perc + sh_calc_fine)
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        where sh_date is not null
        into :delinq_sum;

        delinq_sum = coalesce(delinq_sum, 0);
      end

      p_less_30 = 0;
      p_30_and_more = 0;

      if (delinq_days < 30) then
        p_less_30 = 1;
      if (delinq_days >= 30) then
        p_30_and_more = 1;

      is_period_blocked = 0;
      if (pant_date_begin between :rep_date_begin and :rep_date_end) then
        is_period_blocked = 1;

      suspend;
    end

    is_gived = 0;
    on_pantera_qty = 0;
    delinq_sum = 0;
    is_period_blocked = 0;
    payed_after_call = 0;
    is_payed_after_call = 0;
    p_less_30 = 0;
    p_30_and_more = 0;
    is_gived = 0;
    is_payed_after_call = 0;

-- оперделяю суммы погашений
    for select k_id, oper_called, ok_sum
        from aarep_pantera_bonus_rep(:rep_date_begin, :rep_date_end, 1)
        into :k_id, called_oper_str, payed_after_call
    do
    begin
      is_payed_after_call = 1;
      suspend;
    end

-- определяю отданные группы
    begin_str = '';
    payed_after_call = 0;

    for select pantk.pant_oper_called, sba.sba_from_k_id,
               pant_oper_called || '-' || cast(sba.sba_from_k_id as varchar(18))
        from pantera_kontracts pantk
        inner join kontstate ks on ks.ks_from_k_id = pantk.pant_from_k_id
        inner join sb_akt sba on sba.sba_from_k_id = pantk.pant_from_k_id
        where pantk.pant_date_end >= pantk.pant_date_begin and
              :rep_date_begin - 1 between pantk.pant_date_begin and pantk.pant_date_end and
              pantk.pant_date_begin >= sba.sba_date_begin and
              pantk.pant_project_type = 1 and
              sba.sba_date_end >= :rep_date_begin - 1 and
              sba.sba_date_begin <= sba.sba_date_end and
              sba_is_delete = 0 and
              sba.sba_from_sbl_id = 82 and
              :rep_date_begin - 1 between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1
        into :called_oper_str, :k_id, :begin_str
    do
    begin
      is_gived = 0;
      if (active_end not containing (begin_str)) then -- если на начало периода группа на руках у оператора, а на конец периода ее у данного оператра на руках нет
      begin
        is_gived = 1;
        suspend;
      end
    end
  end

  if (step = 2) then -- на этом шаге формирую цикл, т.к. в отчет должны быть видны 7 последних результатов
  begin
    i = 0;
    rep_date_tmp = rep_date;

    max_i = 1;
    if (rep_period = 'day') then
      max_i = 7;

    while (i < max_i) do
    begin
      rep_date_tmp = rep_date - (i);

      for select called_oper_str, rep_date_begin, rep_date_end, sum(on_pantera_qty), sum(delinq_sum),
                 sum(is_period_blocked), sum(payed_after_call), sum(is_payed_after_call), sum(p_less_30),
                 sum(p_30_and_more), sum(is_gived)
          from cc_ceo_pantera_efficiency(:rep_date_tmp, 1, :rep_period)
          group by called_oper_str, rep_date_begin, rep_date_end
          order by called_oper_str
          into :called_oper_str, :rep_date_begin, :rep_date_end, :on_pantera_qty, :delinq_sum, :is_period_blocked,
               :payed_after_call, :is_payed_after_call, :p_less_30, :p_30_and_more, :is_gived
      do
      begin
        suspend;
      end
      i = i + 1;
    end
  end
  if (step = 3) then -- этот шаг, исключительно для сортировки
  begin
    for select called_oper_str, rep_date_begin, rep_date_end, sum(on_pantera_qty), sum(delinq_sum),
               sum(is_period_blocked), sum(payed_after_call), sum(is_payed_after_call), sum(p_less_30),
               sum(p_30_and_more), sum(is_gived)
        from cc_ceo_pantera_efficiency(:rep_date, 2, :rep_period)
        group by called_oper_str, rep_date_begin, rep_date_end
        order by called_oper_str, rep_date_begin desc
        into :called_oper_str, :rep_date_begin, :rep_date_end, :on_pantera_qty, :delinq_sum, :is_period_blocked,
             :payed_after_call, :is_payed_after_call, :p_less_30, :p_30_and_more, :is_gived
    do
    begin
      suspend;
    end
  end
end
