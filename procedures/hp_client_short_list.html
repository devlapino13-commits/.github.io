CREATE PROCEDURE HP_CLIENT_SHORT_LIST(
    K_ID DECIMAL(18,0))
RETURNS (
    KC_NUMBER INTEGER,
    KC_FROM_C_ID DECIMAL(18,0),
    KC_STATUS SMALLINT,
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    CLIENT_NAME VARCHAR(100),
    CLIENT_PASS VARCHAR(100),
    C_TYPE INTEGER,
    C_IS_PERSONAL SMALLINT,
    C_PLASTCARD_NO VARCHAR(80),
    C_EMAIL VARCHAR(100))
AS
declare variable can_suspend integer;
declare variable kc_closed_ol_id decimal(18,0);
begin
  for select kc_number, kc_from_c_id, kc_status, kc_amount_approved, c_name_fio as client_name,
             c_pass_no as client_pass, c_type, c_is_personal,
             (select cont_value
              from hp_contdata_get_cell_main(0, kc_from_c_id)),
             (select cont_value
              from hp_contdata_get_email_main(0, kc_from_c_id)), kc_closed_ol_id
      from knclient, clients
      where kc_from_k_id = :k_id and
            kc_from_c_id = clients.c_id and
            (kc_amount_approved > 0 or
            kc_status = 2)
      order by kc_status desc, kc_number
      into :kc_number, :kc_from_c_id, :kc_status, :kc_amount_approved, :client_name, :client_pass, :c_type,
           :c_is_personal, :c_plastcard_no, :c_email, :kc_closed_ol_id
  do
  begin
    can_suspend = 1;
    if (c_is_personal is null) then
      c_is_personal = 0;
    if (kc_closed_ol_id is not null) then
    begin
      if (exists(select *
                 from operlist
                 where ol_id = :kc_closed_ol_id and
                       is_delete = 0)) then
        can_suspend = 0;
    end
    if (can_suspend = 1) then
      suspend;
  end
end
