CREATE PROCEDURE FM_HTML_CLIENT_ZADOL(
    C_ID DECIMAL(18,0),
    DATE_END DATE,
    WITH_HEADER INTEGER,
    IS_ORIV D_SMALLINT = 0,
    UN_ID D_ID = null)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable K_KONTRACT_NO varchar(100);
declare variable RES_DATE date;
declare variable RES_GR_CURR_AMOUNT decimal(18,4) = 0;
declare variable RES_GR_PLAN_AMOUNT decimal(18,4) = 0;
declare variable RES_GR_TOTAL_AMOUNT decimal(18,4) = 0;
declare variable RES_CL_CURR_AMOUNT decimal(18,4) = 0;
declare variable RES_CL_PLAN_AMOUNT decimal(18,4) = 0;
declare variable RES_CL_TOTAL_AMOUNT decimal(18,4) = 0;
declare variable RES_DELINQ_DATE date;
declare variable DATE_END_STR varchar(20);
declare variable C_NAME_FIO varchar(100);
declare variable DB_ID integer;
declare variable K_ID ID_TYPE;
declare variable RECOM_PAY_SUM CURR_TYPE;
declare variable SH_CALC_FINE decimal(18,4);
declare variable ZAD_OS decimal(18,4);
declare variable ZAD_PERC decimal(18,4);
declare variable SH_CALC_PERC decimal(18,4);
declare variable SH_DATE date;
declare variable PLAN_AMOUNT decimal(18,4);
declare variable URI D_STR_255;
declare variable OK_LAST_PAYED date;
declare variable OK_LAST_PAYED_SUM D_INTEGER;
declare variable OK_LAST_PAYED_STR D_STR_100;
declare variable RD_ID integer;
declare variable OPERKROFF_GR_DEBT decimal(18,2);
declare variable OPERKROFF_CL_DEBT decimal(18,2);
declare variable ROUND_VALUE D_SMALLINT;
declare variable GOS_DUTY D_CURRENCY = 0;
begin
  -- Процедура отображает краткую информацию о клиенте на мобильное приложение
  db_id = f_get_db_id();
  rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');
  round_value = iif(db_id = 94401, 2, 0);

  if (date_end is null) then
    date_end = current_date;
  res_text = '';
  date_end_str = iif(db_id <> 94401, 'сегодня', 'Bu gün');

  if (date_end <> current_date) then
    date_end_str = f_format(date_end);

  if (with_header = 1) then
  begin
    res_text = (select res_style
                from fm_html_styles) || '<body>';
    res_text = BLOB_APPEND(res_text, f_format('<h1 class="green_text">%1 %2</h1>', iif(db_id <> 94401, 'Задолженность на', 'Hesabat tarixinə borc'), date_end_str));
    c_name_fio = '';
    select c_name_fio
    from clients
    where c_id = :c_id
    into :c_name_fio;
    if (coalesce(c_name_fio, '') = '') then
      c_name_fio = iif(db_id <> 94401, 'Данные на уровне группы.', 'Qrup səviyyəsində məlumat.');
    else
      c_name_fio = iif(db_id <> 94401, 'Клиент', 'Müştəri') || ': ' || c_name_fio;
    res_text = BLOB_APPEND(res_text, '<h2>' || c_name_fio || '</h2>');
  end

  for select k_id, k_kontract_no, res_date, res_gr_curr_amount, res_gr_plan_amount, res_gr_total_amount,
             res_cl_curr_amount, res_cl_plan_amount, res_cl_total_amount, res_delinq_date
      from aarep_get_client_zadol(:c_id, :date_end)
      into :k_id, :k_kontract_no, :res_date, :res_gr_curr_amount, :res_gr_plan_amount, :res_gr_total_amount,
           :res_cl_curr_amount, :res_cl_plan_amount, :res_cl_total_amount, :res_delinq_date
  do
  begin
    if (is_oriv = 1) then
    begin
      if (db_id in (39801, 41701)) then
      begin
        if (not exists(select *
                     from sb_akt sa
                     inner join sb_list sl on sa.sba_from_sbl_id = sl.sbl_id
                     inner join username un on sl.sbl_from_c_id = un.un_from_c_id
                     where un.un_id = :un_id and
                           sa.sba_from_k_id = :k_id and
                           current_date between sa.sba_date_begin and sa.sba_date_end and
                           sa.sba_is_delete = 0)) then
        continue;
      end
      else
      begin
        if (not exists(select *
                     from sb_akt sa
                     where sa.sba_from_k_id = :k_id and
                           current_date between sa.sba_date_begin and sa.sba_date_end and
                           sa.sba_is_delete = 0)) then
        continue;
      end 
    end

    uri = f_uri_html(k_kontract_no, 'loan_delinq_list', k_kontract_no, 9, k_id, date_end, date_end);

    res_text = BLOB_APPEND(res_text, f_format('<br><b>%1 %2</b>:<br><br>', iif(db_id <> 94401, 'Инфо по', 'Müqavilə üzrə məlumat'), :uri));

    ok_last_payed = null;
    ok_last_payed_sum = 0;

    select ol_date, sum(payed_sum)
    from (select ok.ok_ol_date as ol_date, -(ok.ok_os + ok.ok_perc + ok.ok_fine) as payed_sum
          from operkred ok
          where ok.ok_from_k_id = :k_id and
                ok.ok_type = 4 and
                ok.ok_ol_is_active = 1 and
                ok.ok_from_c_id = 0

          union all

          select oko_ol_date as ol_date, -(okf.oko_os + okf.oko_perc + okf.oko_other)
          from operkroff okf
          where okf.oko_ol_is_active = 1 and
                okf.oko_from_k_id = :k_id and
                okf.oko_type = 4

          union all

          select oe.ox_ol_date as ol_date, oe.ox_amount_minus
          from operextra oe
          where oe.ox_object_kind = 9 and
                oe.ox_object_id = :k_id and
                oe.ox_ol_is_active = 1 and
                oe.ox_from_rd_id = :rd_id) a
    group by ol_date
    having sum(payed_sum) > 0
    order by ol_date desc
    rows 1
    into :ok_last_payed, :ok_last_payed_sum;

    if (ok_last_payed is not null) then
      ok_last_payed_str = f_datetostr(ok_last_payed);
    else
      ok_last_payed_str = 'Отсутствует';

    ok_last_payed_str = coalesce(ok_last_payed_str, '');
    ok_last_payed_sum = round(coalesce(ok_last_payed_sum, :round_value));

    if (db_id = 81001) then
    begin
      select recom_pay_sum
      from get_recommended_pay_sum(:k_id, :date_end)
      into recom_pay_sum;

      for select sh_calc_fine, zad_os, zad_perc, sh_calc_perc, sh_date
          from aarep_make_zad_list(:k_id, :date_end, 1)
          where sh_id >= 0
          order by sh_date, sh_id asc
          rows 1
          into :sh_calc_fine, zad_os, zad_perc, sh_calc_perc, sh_date
      do
      begin
        if (zad_perc > sh_calc_perc) then
          plan_amount = sh_calc_fine + zad_os + zad_perc;
        else
          plan_amount = sh_calc_fine + zad_os + sh_calc_perc;
      end

      res_text = BLOB_APPEND(res_text, '<table width="100%">
         <col width="50%">
         <col width="50%">
         <tr>
           <th>' || iif(res_delinq_date is not null or res_date is null, iif(db_id <> 94401, 'Просрочен от ', 'Просрочен от') || f_datetostr(res_delinq_date), iif(db_id <> 94401, 'Инфо: Не просрочен', 'Инфо: Не просрочен')) || '</th>' || f_format('<th>%1</th>', iif(db_id <> 94401, 'Сумма', 'Сумма')) || '</tr>' || '<tr>' || '<td align="left">' || iif(db_id <> 94401, 'Рекомендуемый платеж на ', 'Рекомендуемый платеж на ') || :date_end_str || '(РП)*' || '</td>' || '<td align="right">' || round(recom_pay_sum, 0) || '</td>' || '</tr>' || '<tr>' || '<td align="left">' || iif(db_id <> 94401, 'План дата ', 'План дата ') || f_datetostr(sh_date) || '</td>' || '<td align="right">' || round(coalesce(plan_amount, 0), 0) || '</td>' || '</tr>' || '<tr>' || '<td align="left">' || iif(db_id <> 94401, 'Для закрытия ', 'Для закрытия ') || date_end_str || '</td>' || '<td align="right">' || round(res_gr_total_amount, 0) || '</td>' || '</tr>' || '<tr>' || '<td align="left">' || iif(db_id <> 94401, 'Дата последнего погашения ', '') || '</td>' || '<td align="right">' || ok_last_payed_str || '</td>' || '</tr>' || '<tr>' || '<td align="left">' || iif(db_id <> 94401, 'Сумма последнего погашения ', '') || '</td>' || '<td align="right">' || round(ok_last_payed_sum, 0) || '</td>' || '</tr>');

      res_text = BLOB_APPEND(res_text, '</table>');
      res_text = BLOB_APPEND(res_text, '<p><b>*Примечание:</b></p>' || '<ul><li>РП включает (при наличии) просроченную ОС, %%, пени и платеж по графику, выпадающий в ближайшие 10 дней</li></ul>' || '<ul><li>Если сумма РП = 0, то клиент платит по графику либо любую другую сумму. </li></ul>');
      res_text = BLOB_APPEND(res_text, '</body>');
    end
    else
    begin
      if (:db_id in (39801, 41701)) then
      begin
        select sum(ox.ox_amount_plus - ox.ox_amount_minus)
        from operextra ox
        inner join refdata rd on rd.rd_id = ox.ox_from_rd_id
        where ox.ox_object_id = :k_id
        and ox.ox_ol_is_active = 1
        and ox.ox_object_kind = 9
        and rd.rd_code in
            ('OperExtraKind.ExceptionRegister',
             'OperExtraKind.LawClientExpense')
        into gos_duty; --Гос. пошлина RFCKG-3096

        gos_duty = coalesce(gos_duty,0);
        operkroff_gr_debt = null;

        select sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc + oko_other + oko_sh_other)
        from operkroff okf
        where okf.oko_from_k_id = :k_id and
              okf.oko_ol_is_active = 1
        into operkroff_gr_debt; -- Задолженность ГРУППЫ на внесист. учёте

        operkroff_cl_debt = null;

        select sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc + oko_other + oko_sh_other)
        from operkroff okf
        where okf.oko_from_c_id = :c_id and
              okf.oko_from_k_id = :k_id and
              okf.oko_ol_is_active = 1
        into operkroff_cl_debt; -- Задолженность КЛИЕНТА на внесист. учёте

        operkroff_gr_debt = coalesce(operkroff_gr_debt, 0);
        operkroff_cl_debt = coalesce(operkroff_cl_debt, 0);
      end
      res_text = BLOB_APPEND(res_text, '<table width="100%">
         <col width="50%">
         <col width="25%">
         <col width="25%">
         <tr>
           <th>' || iif(res_delinq_date is not null or res_date is null, iif(db_id <> 94401, 'Просрочен от ', 'Gecikmənin yaranma tarixi ') || f_datetostr(res_delinq_date), iif(db_id <> 94401, 'Инфо: Не просрочен', 'Məlumat: Gecikmədə deyil')) || '</th>' || f_format('<th>%1</th>', iif(db_id <> 94401, 'Группа', 'Qrup')) || f_format('<th>%1</th>', iif(db_id <> 94401, 'Клиент', 'Müştəri')) || '</tr>' ||
            '<tr>' || '<td align="left">' || iif(db_id <> 94401, 'Задолж. с проц. на ', 'Faiz üzrə borc ') || :date_end_str || '</td>' || '<td align="right">' || round(res_gr_curr_amount, :round_value) || '</td>' || '<td align="right">' || round(res_cl_curr_amount, :round_value) || '</td>' || '</tr>' || iif(:db_id = 94401, '', iif((operkroff_gr_debt = 0 and operkroff_cl_debt = 0), '',
            '<tr>' || '<td align="left">' || 'Долг на В/У' || '</td>' || '<td align="right">' || round(operkroff_gr_debt, :round_value) || '</td>' || '<td align="right">' || round(operkroff_cl_debt, :round_value) || '</td>' || '</tr>')) ||
            '<tr>' || '<td align="left">' || 'Гос. пошлина ' || '</td>' || '<td align="right">' || round(gos_duty, :round_value) || '</td>' || '<td align="right">' || round(gos_duty, :round_value) || '</td>' || '</tr>' ||
            '<tr>' || '<td align="left">' || iif(db_id <> 94401, 'План дата ', 'Planlaşdırılan ödəniş tarixi ') || f_datetostr(res_date) || '</td>' || '<td align="right">' || round(res_gr_plan_amount, :round_value) || '</td>' || '<td align="right">' || round(res_cl_plan_amount, :round_value) || '</td>' || '</tr>' ||
            '<tr>' || '<td align="left">' || iif(db_id <> 94401, 'Для закрытия ', 'Bağlanma tarixi ') || date_end_str || '</td>' || '<td align="right">' || round(res_gr_total_amount, :round_value) || '</td>' || '<td align="right">' || round(res_cl_total_amount, :round_value) || '</td>' || '</tr>' ||
            '<tr>' || '<td align="left">' || iif(db_id <> 94401, 'Дата последнего погашения ', '') || '</td>' || '<td align="right">' || ok_last_payed_str || '</td>' || '<td align="right">' || ok_last_payed_str || '</td>' || '</tr>' ||
            '<tr>' || '<td align="left">' || iif(db_id <> 94401, 'Сумма последнего погашения ', '') || '</td>' || '<td align="right">' || round(ok_last_payed_sum, :round_value) || '</td>' || '<td align="right">' || round(ok_last_payed_sum, :round_value) || '</tr>');

      res_text = BLOB_APPEND(res_text, '</table>');
      res_text = BLOB_APPEND(res_text, '</body>');
    end
  end

  suspend;
end
