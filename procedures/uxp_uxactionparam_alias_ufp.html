CREATE PROCEDURE UXP$UXACTIONPARAM_ALIAS_UFP(
    UXA_FCODE D_STR_100,
    PARAM_NAME D_STR_CODE)
RETURNS (
    UFP_UUID D_UUID,
    UF_UUID D_UUID,
    UF_DATA_TYPE D_STR_CODE,
    UF_IS_PARAM D_BOOL,
    UF_PK_FIELD D_SMALLINT,
    US_IS_TABLE D_BOOL)
AS
declare variable ufm_uuid d_uuid;
begin
  select ux_draft_uuid
  from ux$alias, ux$code
  where uxa_fcode = :uxa_fcode and
        uxa_owner = ux_uuid
  into ufm_uuid;

  if (ufm_uuid is not null) then
  begin
    select ufp_uuid, uf_uuid, uf_data_type, uf_is_param, uf_pk_field, us_is_table
    from ux$frame, ux$frparam, ux$field, ux$source
    where ufr_owner = :ufm_uuid and
          ufr_uuid = ufp_ufr_uuid and
          ufp_uf_uuid = uf_uuid and
          uf_field_name = :param_name and
          (ufp_public = 1 or uf_is_param = 0) and
          uf_owner = us_uuid
    into ufp_uuid, uf_uuid, uf_data_type, uf_is_param, uf_pk_field, us_is_table;
  end

  if (uf_data_type is null) then
    uf_data_type = '';

  suspend;
end
