CREATE PROCEDURE FIND_US_ACC_LIST(
    FIND_ID D_STR_CODE,
    FIND_TEXT D_STR_255,
    ADD_PARAMS D_STR_8100,
    UN_ID D_ID)
RETURNS (
    RES_ID D_STR_CODE,
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_LONG)
AS
declare variable num_1 d_str_255;
declare variable num_2 d_str_255;
declare variable txt d_str_255;
declare variable max_row d_integer;
declare variable local_currency d_str_code;
declare variable find_reg_id d_integer;
declare variable find_currency d_str_10;
declare variable find_with_local_currency d_smallint;
declare variable find_acc_kind d_str_long;
declare variable find_acc_kind_include d_smallint;
declare variable find_al_account d_str_code;
declare variable find_loro_reg_id d_integer;
declare variable db_id d_integer;
declare variable cash_reg_id d_integer;
declare variable cash_number d_integer;
declare variable cash_code d_str_code;
begin
  if (un_id is null) then
    un_id = 0;

  if (find_text is null) then
    find_text = '';

  find_reg_id = f_json_str_get_value(add_params, 'region_id', '0');
  find_currency = f_json_str_get_value(add_params, 'currency', '');
  find_with_local_currency = f_json_str_get_value(add_params, 'with_local_currency', '0');
  find_acc_kind = f_json_str_get_value(add_params, 'acc_kind', '');
  find_acc_kind_include = f_json_str_get_value(add_params, 'acc_kind_include', '1');
  find_loro_reg_id = f_json_str_get_value(add_params, 'loro_region_id', '0');
  find_al_account = f_json_str_get_value(add_params, 'al_account', '');

  if (find_id is null) then
    find_id = '';

  if (find_id <> '' and
      find_id <> '0' and
      find_text = '') then
  begin
    select al_id, al_account
    from acc_list
    where al_id = :find_id
    into res_id, res_code;

    if (res_id <> '') then
    begin
      res_text = f_objname('acc_list', res_id);
      suspend;
      exit;
    end
  end

  if (find_loro_reg_id <> 0 and
      find_reg_id <> 0) then
  begin
    for select al_id, al_account
        from acc_list, loronostro
        where al_id = ln_account and
              ln_region_loro = :find_reg_id and
              ln_region_nostro = :find_loro_reg_id and
              ln_currency = :find_currency
        into res_id, res_code
    do
    begin
      res_text = f_objname('acc_list', res_id);
      suspend;
    end

    exit;
  end

  max_row = 100;

  select num_1, num_2, txt
  from find_editor_parcer(:find_text)
  into :num_1, :num_2, :txt;

  res_id = '';
  res_code = '';
  res_text = '';

  db_id = f_get_db_id();

  cash_reg_id = 0;
  cash_number = 999999;

  cash_code = '';
  if (db_id in (39801, 41701)) then
  begin
    cash_code = '10001';

    select uc_from_reg_id, uc_num_cash
    from user_cash, username
    where un_from_c_id = uc_from_c_id and
          un_id = :un_id
    into cash_reg_id, :cash_number;
  end

  select cm_default_currency
  from company
  into local_currency;
  if (find_with_local_currency = 0) then
    local_currency = '';

  if (char_length(num_1) = 18) then
  begin
    select al_id, al_account
    from acc_list
    where al_id = :num_1 and
          f_check_acclist_kind(al_from_bg_id, :find_acc_kind) = :find_acc_kind_include
    rows 1
    into res_id, res_code;

    if (res_id <> '') then
    begin
      res_text = f_objname('acc_list', res_id);
      suspend;
      exit;
    end
  end

  if (txt = '') then
  begin
    if (num_1 <> '' and
        num_2 <> '' and
        char_length(num_1) = 5) then
    begin
      for select al_id, al_account
          from acc_list
          where al_from_bg_id = :num_1 and
                al_number = :num_2 and
                (:find_al_account = '' or
                (:find_al_account <> '' and
                al_account = :find_al_account)) and
                (al_date_close is null or
                al_date_close > current_date) and
                (:find_acc_kind = '' or
                (:find_acc_kind <> '' and
                f_check_acclist_kind(al_from_bg_id, :find_acc_kind) = :find_acc_kind_include)) and
                (:find_reg_id = 0 or
                al_from_reg_id = :find_reg_id) and
                (:find_currency = '' or
                al_currency in (:find_currency, :local_currency)) and
                (:cash_number = 0 or
                :cash_reg_id = 0 or
                al_account not starting :cash_code or
                (al_from_reg_id = :cash_reg_id and
                al_account = :cash_code || '/' || :cash_number))
          order by al_from_bg_id, al_id
          rows :max_row
          into res_id, res_code
      do
      begin
        res_text = f_objname('acc_list', res_id);
        suspend;
      end

      if (res_id <> '') then
        exit;
    end

    if (num_1 <> '') then
    begin
      for select al_id, al_account
          from acc_list
          where al_from_bg_id starting :num_1 and
                (:find_al_account = '' or
                (:find_al_account <> '' and
                al_account = :find_al_account)) and
                (:num_2 = '' or
                (:num_2 <> '' and
                al_number = :num_2)) and
                (al_date_close is null or
                al_date_close > current_date) and
                (:find_acc_kind = '' or
                (:find_acc_kind <> '' and
                f_check_acclist_kind(al_from_bg_id, :find_acc_kind) = :find_acc_kind_include)) and
                (:find_reg_id = 0 or
                al_from_reg_id = :find_reg_id) and
                (:find_currency = '' or
                al_currency in (:find_currency, :local_currency)) and
                (:cash_number = 0 or
                :cash_reg_id = 0 or
                al_account not starting :cash_code or
                (al_from_reg_id = :cash_reg_id and
                al_account = :cash_code || '/' || :cash_number))
          order by al_from_bg_id, al_id
          rows :max_row
          into res_id, res_code
      do
      begin
        res_text = f_objname('acc_list', res_id);
        suspend;
      end
    end
    else
    begin
      if (find_reg_id <> 0 and
          find_acc_kind <> '') then
      begin
        for select al_id, al_account
            from acc_list
            where al_from_reg_id = :find_reg_id and
                  (:find_al_account = '' or
                  (:find_al_account <> '' and
                  al_account = :find_al_account)) and
                  (al_date_close is null or
                  al_date_close > current_date) and
                  (:find_acc_kind = '' or
                  (:find_acc_kind <> '' and
                  f_check_acclist_kind(al_from_bg_id, :find_acc_kind) = :find_acc_kind_include)) and
                  (:find_currency = '' or
                  al_currency in (:find_currency, :local_currency)) and
                  (:cash_number = 0 or
                  :cash_reg_id = 0 or
                  al_account not starting :cash_code or
                  (al_from_reg_id = :cash_reg_id and
                  al_account = :cash_code || '/' || :cash_number))
            order by al_from_bg_id, al_id
            rows :max_row
            into res_id, res_code
        do
        begin
          res_text = f_objname('acc_list', res_id);
          suspend;
        end
      end
    end

  end
  else
  begin
    for select al_id, al_account
        from acc_list, find_in_workdoc_words('acc_list', :txt, 1)
        where al_id = wdlink_object_id and
              (:find_al_account = '' or
              (:find_al_account <> '' and
              al_account = :find_al_account)) and
              (al_date_close is null or
              al_date_close > current_date) and
              (:find_acc_kind = '' or
              (:find_acc_kind <> '' and
              f_check_acclist_kind(al_from_bg_id, :find_acc_kind) = :find_acc_kind_include)) and
              (:num_1 = '' or
              (:num_1 <> '' and
              al_from_bg_id starting :num_1)) and
              (:num_2 = '' or
              (:num_2 <> '' and
              al_number = :num_2)) and
              (:find_reg_id = 0 or
              (:find_reg_id <> 0 and
              al_from_reg_id = :find_reg_id)) and
              (:find_currency = '' or
              al_currency in (:find_currency, :local_currency)) and
              (:cash_number = 0 or
              :cash_reg_id = 0 or
              al_account not starting :cash_code or
              (al_from_reg_id = :cash_reg_id and
              al_account = :cash_code || '/' || :cash_number))
        order by al_from_bg_id, al_id
        rows :max_row
        into res_id, res_code
    do
    begin
      res_text = f_objname('acc_list', res_id);
      suspend;
    end
  end
end
