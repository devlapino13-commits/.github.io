CREATE PROCEDURE RUS_STANDART_DO_DISBURSEMENT(
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    DISB_AMOUNT CURR_TYPE)
AS
declare variable psi_id id_type;
declare variable idcard str_100;
begin
  if (exists(select *
             from extsysmain em
             where em.esm_project = 'rus_standart' and
                   em.esm_action = 'CardDisb' and
                   em.esm_from_k_id = :k_id and
                   em.esm_status in (0, 1))) then
    exit;

  for select psi_id
      from hp_paysys_get_cl_info(:c_id, 6) -- 6 - Русский Стандарт
      order by psm_date desc
      rows 1
      into :psi_id
  do
  begin
    select pys.psi_code
    from paysys_item pys
    where pys.psi_id = :psi_id
    into :idcard;
  end

  if (not exists(select *
                 from extsysmain em
                 where em.esm_project = 'rus_standart' and
                       em.esm_action = 'CardDisb' and
                       em.esm_object_kind = 0 and
                       em.esm_object_id = :c_id and
                       em.esm_status = 1 and
                       em.esm_account = :idcard)) then
  begin -- если по данной карте клиент ранее не полукчал кредит
    insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
                            esm_full_source, esm_amount, esm_from_k_id, esm_account)
    values ('rus_standart', 'CardDisb', current_date, 0, :c_id, 0, '', :disb_amount, :k_id, :idcard);
  end
  else -- если текущий заем выдается клиенту по карте, по который выдача ранее уже была в другом займе, тогда не трубется заполнть анкету, достаточно сделать перечисление оп карте
  begin
    insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
                            esm_full_source, esm_amount, esm_from_k_id, esm_account)
    values ('rus_standart', 'CardRepay', current_date, 0, :c_id, 0, '', :disb_amount, :k_id, :idcard);
  end
end
