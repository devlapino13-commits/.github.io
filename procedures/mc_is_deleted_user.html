CREATE PROCEDURE MC_IS_DELETED_USER(
    MCL_ID D_ID NOT NULL)
RETURNS (
    IS_DELETED D_SMALLINT)
AS
declare variable LAST_STATUS D_INTEGER;
declare variable C_ID D_ID;
declare variable RECOVERY_STATUS D_SMALLINT;
declare variable MCL_LOGIN D_STR_20;
declare variable MCL_PROJECT D_ID;
declare variable MCL_PROJECT_TEXT D_STR_20;
declare variable MCL_IS_DELETED D_SMALLINT;
begin
    is_deleted = 0;

    select
        mcl.mcl_from_c_id,
        mcl.mcl_login,
        mcl.mcl_is_delete,
        mcl.mcl_project,
        f_enum('mcl_project', mcl.mcl_project, 'code', 0)
    from mc_login mcl
    where mcl.mcl_id = :mcl_id
    into
        :c_id,
        :mcl_login,
        :mcl_is_deleted,
        :mcl_project,
        :mcl_project_text;
    
    if (coalesce(c_id, 0) = 0
        and mcl_is_deleted = 0
    ) then
        select first 1 mcl.mcl_from_c_id
        from mc_login mcl
        where mcl.mcl_login = :mcl_login
            and mcl.mcl_is_delete = 1
            and mcl.mcl_project = :mcl_project
        order by mcl_id desc
        into :c_id;

    if (coalesce(:c_id, 0) <> 0) then
    begin
        if (mcl_project_text = 'kmp') then
            select cast(ovs_add_value as d_smallint)
            from hp_objectprops_get('clients', :c_id, -1,  'ObjectProps.MC_User_Is_Delete', null)
            into :is_deleted;

        if (mcl_project_text = 'oivo') then
            select cast(ovs_add_value as d_smallint)
            from hp_objectprops_get('clients', :c_id, -1,  'ObjectProps.OIVO_User_Is_Deleted', null)
            into :is_deleted;

        if (coalesce(is_deleted, 0) = 0) then
            is_deleted = 0;
    end

    suspend;
end
