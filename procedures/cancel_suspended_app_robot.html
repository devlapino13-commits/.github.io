CREATE PROCEDURE CANCEL_SUSPENDED_APP_ROBOT
AS
declare variable ma_id d_id;
declare variable days_inactive d_integer;
declare variable pr_id d_integer;
declare variable kf_id d_id;
declare variable pr_code d_str_100;
declare variable kal_new_officer_code varchar(20);
declare variable ma_is_active d_integer;
declare variable k_status d_integer;
declare variable ks_state d_integer;
declare variable fact_disb_date d_date;
declare variable k_disb_date d_date;
declare variable date_save d_datetime;
declare variable k_id d_id;
declare variable ma_number d_integer;
begin
  for select ma.ma_id, ma.ma_from_kf_id, pr.pr_code, datediff(day, ma.ma_date_oper, current_date), ma.ma_number
      from miniapp ma
      inner join product pr on ma.ma_from_pr_id = pr.pr_id
      where ma.ma_is_active = 1 and
            ma.ma_date_end = cast('31.12.9999' as date) and
            ma.ma_date_oper <= current_date - 8
      into :ma_id, :kf_id, :pr_code, :days_inactive, :ma_number
  do
  begin
    -- проверяем наличие записей, говорящих о том, что контракт уже был выдан/закрыт
    if (exists(select *
               from kontract k
               inner join kontstate ks on k.k_id = ks.ks_from_k_id
               where k.k_from_ma_app_id = :ma_id and
                     ks.ks_state in (1, 2) and
                     current_date between ks.ks_date_begin and ks.ks_date_end and
                     ks.ks_ol_is_delete = 0)) then
      continue;

    -- ищем код отдела
    kal_new_officer_code = '';
    select kal.kal_new_officer_code
    from knaktlist kal
    where kal.kal_from_kf_id = :kf_id and
          kal.is_delete = 0 and
          current_date between kal.kal_date and kal.kal_date_end
    into :kal_new_officer_code;

    if (:kal_new_officer_code = '18-3-86') then   --код операционистов, их заявки/договора не трогаем
      continue;

    -- проверяем, если заявка не перешла договор в процессе (не утвержденный/утвержденный), тогда переходим в заявки
    if (not exists(select *
                   from kontract k
                   inner join kontstate ks on k.k_id = ks.ks_from_k_id
                   where k.k_from_ma_app_id = :ma_id and
                         ks.ks_state = 0 and
                         ks.ks_ol_is_delete = 0 and
                         current_date between ks.ks_date_begin and ks.ks_date_end and
                         k.k_status in (0, 1))) then
    begin
      -- заявки по продуктам "Иджара" или "Истисна", которые не редактировались больше 31 дня и не закреплены за оперотделом (код 18-3-86)
      if ((:pr_code containing '.IDJARA.' or :pr_code containing '.ISTISNAA.') and
          :days_inactive > 31) then
      begin
        insert into miniapp (ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay,
                             ma_percent, ma_kolich_client, ma_memo, ma_is_active, ma_is_auto, ma_summa_minus,
                             ma_date_save, ma_from_un_id, ma_office_id, ma_currency_id, ma_no, ma_percent_category,
                             ma_date_end, ma_manager_id, ma_plan_disb_time, ma_expert_id, ma_repay_day,
                             ma_source_system, ma_kommis_perc, ma_refund, ma_add_info)
        select ma_id, -1, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay, ma_percent,
               ma_kolich_client, ma_memo, 0, ma_is_auto, ma_summa_minus, ma_date_save, 1000000000000, ma_office_id,
               ma_currency_id, ma_no, ma_percent_category, '31.12.9999', ma_manager_id, ma_plan_disb_time, ma_expert_id,
               ma_repay_day, ma_source_system, ma_kommis_perc, ma_refund, ma_add_info
        from miniapp
        where ma_id = :ma_id and
              ma_number = :ma_number;
        insert into kontreslist (krl_id, krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                             krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
        values (-1, 91, 1000000000000, null, 0, :ma_id, current_timestamp, current_date, current_date, 0); --krl_from_kr_id = 91 означает аннулирование роботом
      end
      else
      if (:days_inactive > 7 and (not(:pr_code containing '.IDJARA.' or :pr_code containing '.ISTISNAA.'))) then
      begin
        insert into miniapp (ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay,
                             ma_percent, ma_kolich_client, ma_memo, ma_is_active, ma_is_auto, ma_summa_minus,
                             ma_date_save, ma_from_un_id, ma_office_id, ma_currency_id, ma_no, ma_percent_category,
                             ma_date_end, ma_manager_id, ma_plan_disb_time, ma_expert_id, ma_repay_day,
                             ma_source_system, ma_kommis_perc, ma_refund, ma_add_info)
        select ma_id, -1, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay, ma_percent,
               ma_kolich_client, ma_memo, 0, ma_is_auto, ma_summa_minus, ma_date_save, 1000000000000, ma_office_id,
               ma_currency_id, ma_no, ma_percent_category, '31.12.9999', ma_manager_id, ma_plan_disb_time, ma_expert_id,
               ma_repay_day, ma_source_system, ma_kommis_perc, ma_refund, ma_add_info
        from miniapp
        where ma_id = :ma_id and
              ma_number = :ma_number;

        insert into kontreslist (krl_id, krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                             krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
        values (-1, 91, 1000000000000, null, 0, :ma_id, current_timestamp, current_date, current_date, 0); --krl_from_kr_id = 91 означает аннулирование роботом
      end
    end
    else -- заходим в договора в процессе
    begin
      k_status = null;
      ks_state = null;
      k_disb_date = null;
      fact_disb_date = null;
      k_id = null;

      -- получаем только договора в процессе (утвержденные/не утвержденные)
      select k.k_status, ks.ks_state, k.k_disb_date, k.k_fact_disb_date, k.k_id
      from kontract k
      inner join kontstate ks on k.k_id = ks.ks_from_k_id
      where k.k_from_ma_app_id = :ma_id and
            ks.ks_state = 0 and
            k.k_status in (0, 1) and
            ks.ks_ol_is_delete = 0 and
            current_date between ks.ks_date_begin and ks.ks_date_end
      into :k_status, :ks_state, :k_disb_date, :fact_disb_date, :k_id;

      -- договора в процессе, кроме продуктов "Иджара" или "Истисна", которые висят больше 7 дней
      if (current_date - :k_disb_date > 7 and
          :fact_disb_date is null) then
      begin
        if (not(:pr_code containing '.IDJARA.' or :pr_code containing '.ISTISNAA.')) then
        begin
          update kontract k
          set k.k_status = -2
          where k.k_id = :k_id;
        end
      end
    end
  end
end
