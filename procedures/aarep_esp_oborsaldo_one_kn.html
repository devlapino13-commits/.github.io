CREATE PROCEDURE AAREP_ESP_OBORSALDO_ONE_KN(
    REP_KC_ID ID_TYPE,
    REP_K_ID ID_TYPE)
RETURNS (
    KEY_ID ID_TYPE,
    RES_PERIOD D_STR_CODE,
    ACC_NO D_STR_CODE,
    KC_ID ID_TYPE,
    C_ID ID_TYPE,
    K_KONTRACT_NO D_STR_CODE,
    RES_BEGIN_DT CURR_TYPE,
    RES_BEGIN_KT CURR_TYPE,
    RES_OBOR_DT CURR_TYPE,
    RES_OBOR_KT CURR_TYPE,
    RES_END_DT CURR_TYPE,
    RES_END_KT CURR_TYPE,
    OKS_DISB_DATE DATE,
    OKS_LOAN_AMOUNT CURR_TYPE,
    OKS_LAST_SH_DATE DATE,
    OKS_IS_RESIDENT INTEGER,
    OKS_DELINQ_DAY_BEGIN INTEGER,
    OKS_DELINQ_DAY_END INTEGER,
    ACC_MEMO D_STR_255)
AS
declare variable min_date date;
declare variable max_date date;
declare variable rep_date_begin date;
declare variable rep_date_end date;
begin
  if (coalesce(rep_kc_id, 0) = 0) then
    exit;

  if (coalesce(rep_k_id, 0) = 0) then
  begin
    select kc_from_k_id
    from knclient
    where kc_id = :rep_kc_id
    into :rep_k_id;
  end

  key_id = 0;
  select min(oks_date_begin), max(oks_date_begin)
  from kn_oksummary
  where oks_from_kc_id = :rep_kc_id
  into :min_date, max_date;

  min_date = min_date - extract(day from min_date) + 1;
  max_date = dateadd(month, 1, max_date - extract(day from max_date) + 1) - 1;

  rep_date_begin = min_date;
  while (rep_date_begin < max_date) do
  begin
    res_period = extract(year from rep_date_begin) || '-' || right('0' || extract(month from rep_date_begin), 2);

    rep_date_end = dateadd(month, 1, rep_date_begin) - 1;
    for select acc_no, kc_id, c_id, k_kontract_no, res_begin_dt, res_begin_kt, res_obor_dt, res_obor_kt, res_end_dt,
               res_end_kt, oks_disb_date, oks_loan_amount, oks_last_sh_date, oks_is_resident, oks_delinq_day_begin,
               oks_delinq_day_end, acc_memo
        from aarep_esp_oborsaldo(:rep_date_begin, :rep_date_end, :rep_kc_id, :rep_k_id)
        order by acc_no
        into :acc_no, :kc_id, :c_id, :k_kontract_no, :res_begin_dt, :res_begin_kt, :res_obor_dt, :res_obor_kt,
             :res_end_dt, :res_end_kt, :oks_disb_date, :oks_loan_amount, :oks_last_sh_date, :oks_is_resident,
             :oks_delinq_day_begin, :oks_delinq_day_end, :acc_memo
    do
    begin
      key_id = key_id + 1;
      suspend;
    end
    rep_date_begin = dateadd(month, 1, rep_date_begin);
  end
end
