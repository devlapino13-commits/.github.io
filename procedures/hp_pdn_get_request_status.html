CREATE PROCEDURE HP_PDN_GET_REQUEST_STATUS(
    C_ID D_ID,
    REP_DATE_BEGIN D_DATE = null,
    REP_DATE_END D_DATE = null,
    MA_ID D_ID = null)
RETURNS (
    SSP_DATETIME D_DATETIME,
    SSP_RESULT D_STR_CODE,
    SSP_RESULT_ID D_SMALLINT,
    SSP_JSO D_INTEGER,
    KI_DATETIME D_DATETIME,
    KI_RESULT D_STR_CODE,
    KI_RESULT_ID D_SMALLINT,
    KI_JSO D_INTEGER,
    KI_EIL_ID D_ID,
    SSP_EIL_ID D_ID)
AS
declare variable eil_status d_smallint;
declare variable bls_id d_id;
declare variable eic_id d_id;
declare variable json_full_str d_blob_text;
declare variable cli_not_found d_boolean;
declare variable debt_not_present d_boolean;
declare variable all_error d_boolean;
declare variable kbki_jso d_integer;
declare variable actual_days d_integer;
begin
  -- возвращает результаты запросов ССП/КИ
  if (rep_date_end is null) then
    rep_date_end = current_date;

  -- ССП
  -- если не указана дата начала, используем стандартное время "жизни" ответа
  select eic_id, eic_info_actual_days
  from extinfocfg
  where eic_code = 'bki_ssp_40'
  into eic_id, actual_days;

  if (rep_date_begin is null) then
    rep_date_begin = rep_date_end - actual_days;

  if (ma_id is null) then
    select eil_status, eil_status_datetime, eil_from_bls_id, eil_id
    from extinfolist
    where eil_from_eic_id = :eic_id
      and eil_from_c_id = :c_id
      and eil_date between :rep_date_begin and :rep_date_end
    order by eil_status desc, eil_date desc, eil_status_datetime desc
    rows 1
    into eil_status, ssp_datetime, bls_id, ssp_eil_id;
  else
    select eil_status, eil_status_datetime, eil_from_bls_id, eil_id
    from extinfolist
    where eil_from_eic_id = :eic_id
      and eil_from_c_id = :c_id
      and eil_date between :rep_date_begin and :rep_date_end
      and eil_object_kind = f_objkind('Miniapp', 'id')
      and eil_object_id = :ma_id
    order by eil_status desc, eil_date desc, eil_status_datetime desc
    rows 1
    into eil_status, ssp_datetime, bls_id, ssp_eil_id;

  if (eil_status is null) then
  begin
    ssp_result = 'Отчет СП не получен';
    ssp_result_id = 1;
  end
  else if (eil_status < 0) then
  begin
    ssp_result = 'Тех. ошибка';
    ssp_result_id = -1;
  end
  else if (eil_status = 0) then
  begin
    ssp_result = 'В очереди для отправки';
    ssp_result_id = -2;
  end
  else if (eil_status = 1) then
  begin
    ssp_result = 'В ожидании ответа';
    ssp_result_id = -3;
  end
  else if (eil_status = 2) then -- запрос успешно завершен, разбираем ответ
  begin
    if (bls_id is null) then
    begin
      ssp_result = 'СП не получен';
      ssp_result_id = 1;
    end
    else
    begin
      select cast(blx_blob as d_blob_text)
      from hp_get_blobex_blob(:bls_id)
      into :json_full_str;

      if (json_full_str is null) then
      begin
        ssp_result = 'Ошибка: json пустой';
        ssp_result_id = 4;
      end
      else
      begin
        ssp_jso = json.parse(json_full_str);
        all_error = true;
        cli_not_found = true;
        debt_not_present = true;

        for select jso
            from hp_bki_json_get_list(json.obj_get_arr(:ssp_jso, 'КБКИ'), 1)
            into kbki_jso
        do
        begin
          if (json.contains(kbki_jso, 'Ошибка') = 1) then
            continue;
          all_error = false;

          if (json.contains(kbki_jso, 'СубъектНеНайден') = 1) then
            continue;
          cli_not_found = false;

          if (json.contains(kbki_jso, 'ОбязательствНет') = 1) then
            continue;
          debt_not_present = false;
        end

        if (all_error) then
        begin
          ssp_result = 'Все КБКИ вернули ошибку';
          ssp_result_id = 3;
        end
        else if (cli_not_found) then
        begin
          ssp_result = 'Субъект не найден';
          ssp_result_id = 5;
        end
        else if (debt_not_present) then
        begin
          ssp_result = 'Обязательств нет';
          ssp_result_id = 6;
        end
        else
        begin
          ssp_result = 'СП получен';
          ssp_result_id = 9;
        end
      end
    end
  end

  -- КИ
  bls_id = null;
  eil_status = null;

  -- если не указана дата начала, используем стандартное время "жизни" ответа
  select eic_id, eic_info_actual_days
  from extinfocfg
  where eic_code = 'equifax_40'
  into eic_id, actual_days;

  if (rep_date_begin is null) then
    rep_date_begin = rep_date_end - actual_days;

  select eil_status, eil_status_datetime, eil_from_bls_id, eil_id
  from extinfolist
  where eil_from_eic_id = :eic_id
    and eil_from_c_id = :c_id
    and eil_date between :rep_date_begin and :rep_date_end
  order by eil_status desc, eil_date desc, eil_status_datetime desc
  rows 1
  into eil_status, ki_datetime, bls_id, ki_eil_id;

  if (eil_status is null) then
  begin
    ki_result = 'КИ не получена';
    ki_result_id = 1;
  end
  else if (eil_status < 0) then
  begin
    ki_result = 'Тех. ошибка';
    ki_result_id = -1;
  end
  else if (eil_status = 0) then
  begin
    ki_result = 'В очереди для отправки';
    ki_result_id = -2;
  end
  else if (eil_status = 1) then
  begin
    ki_result = 'В ожидании ответа';
    ki_result_id = -3;
  end
  else if (eil_status = 2) then -- запрос успешно завершен, разбираем ответ
  begin
    if (bls_id is null) then
    begin
      ki_result = 'КИ не получена';
      ki_result_id = 1;
    end
    else
    begin
      select cast(blx_blob as d_blob_text)
      from hp_get_blobex_blob(:bls_id)
      into :json_full_str;

      if (json_full_str is null) then
      begin
        ki_result = 'Ошибка: json пустой';
        ki_result_id = 4;
      end
      else
      begin
        ki_jso = json.parse(json_full_str);
        ki_result = iif(json.obj_get_val(ki_jso, 'responsecode') = 3, 'Субъект не найден', 'КИ получена');
        ki_result_id = iif(json.obj_get_val(ki_jso, 'responsecode') = 3, 5, 9);
      end
    end
  end

  suspend;
end
