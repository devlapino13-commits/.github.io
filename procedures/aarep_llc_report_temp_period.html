CREATE PROCEDURE AAREP_LLC_REPORT_TEMP_PERIOD(
    REG_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    LLC_K_ID DECIMAL(18,0),
    LLC_REGION_ID INTEGER,
    LLC_STATUS INTEGER,
    LLC_NAME VARCHAR(50),
    LLC_RESERV_KIND VARCHAR(50),
    LLC_OFFICE_NAME VARCHAR(100),
    LLC_BRANCH_NAME VARCHAR(100),
    LLC_GR_BALANCE DECIMAL(18,4),
    LLC_GR_RPPU DECIMAL(18,4),
    LLC_IND_BALANCE DECIMAL(18,4),
    LLC_IND_RPPU DECIMAL(18,4),
    LLC_GR_PERCENT DECIMAL(18,7),
    LLC_IND_PERCENT DECIMAL(18,7),
    LLC_K_BALANCE DECIMAL(18,4),
    LLC_PERCENT DECIMAL(15,7),
    LLC_RPPU DECIMAL(18,4),
    PR_CODE VARCHAR(100),
    PR_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(20),
    LLC_DATE_STATUS DATE,
    REG_OBLAST VARCHAR(255))
AS
declare variable llc_loan_type smallint;
begin
  for select lk_from_k_id, lk_balance, lk_percent, lk_is_indiv, lk_from_reg_id, lk_status, cs_name, cs_llr_kind_text,
             r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, pr_code, pr_name, k_kontract_no,
             lm_date_status, r1.reg_stat_rep_region
      from llc_main, llc_kont, region_list(:reg_id, 1, -1, 1), classific, kontract, product, region r1, region r2
      where lm_date_status between :rep_date_begin and :rep_date_end and
            (lk_from_reg_id = id_list) and
            lk_from_lm_id = lm_id and
            lm_is_signed = 1 and
            lk_status = cs_id and
            lk_from_reg_id = r1.reg_unicode and
            r1.parent = r2.reg_unicode and
            k_id = lk_from_k_id and
            k_from_pr_id = pr_id
      into :llc_k_id, :llc_k_balance, :llc_percent, :llc_loan_type, :llc_region_id, :llc_status, :llc_name,
           :llc_reserv_kind, :llc_office_name, :llc_branch_name, :pr_code, :pr_name, :k_kontract_no, :llc_date_status,
           :reg_oblast
  do
  begin
    llc_rppu = cast((llc_k_balance / 100) as decimal(18,4));
    llc_rppu = cast((llc_rppu * llc_percent) as decimal(15,2));
    llc_gr_balance = 0;
    llc_gr_rppu = 0;
    llc_ind_balance = 0;
    llc_ind_rppu = 0;
    llc_gr_percent = 0;
    llc_ind_percent = 0;
    if (llc_loan_type = 0) then
    begin
      llc_gr_balance = :llc_k_balance;
      llc_gr_rppu = :llc_rppu;
      llc_gr_percent = :llc_percent;
    end
    else
    begin
      llc_ind_balance = :llc_k_balance;
      llc_ind_rppu = :llc_rppu;
      llc_ind_percent = :llc_percent;
    end
    suspend;
  end
end
