CREATE PROCEDURE HP_OBJECTPROPS_GET_LIST(
    PARAM_NAME VARCHAR(100),
    PARAM_KIND VARCHAR(100),
    PARAM_ID VARCHAR(255),
    RESULT_KIND VARCHAR(100),
    REP_DATE DATE)
RETURNS (
    OVS_FROM_OPR_ID DECIMAL(18,0),
    OVS_ID INTEGER,
    OVS_DATE DATE,
    RESULT_TABLE_ID INTEGER,
    RESULT_UNIQ_ID DECIMAL(18,0),
    RESULT_LOCAL_ID DECIMAL(18,0),
    RESULT_VALUE VARCHAR(255),
    RESULT_ADD_VALUE DECIMAL(18,4))
AS
declare variable uniq_param_id decimal(18,0);
declare variable uniq_param_table_id integer;
begin
-- Процедура возвращает список значений всех свойств по запрощенному объекту ЕСЛИ PARAM_NAME=OBJECT
-- Процедура возвращает список всех объектов у кого свойства совпадает с параметром ЕСЛИ PARAM_NAME=SUBJECT

  select opr_object_kind, opr_object_id, opr_subject_kind, opr_subject_id
  from hp_get_propsparam(:param_kind, :param_id, :result_kind, :param_id)
  into :uniq_param_table_id, :uniq_param_id, :result_table_id, :uniq_param_id;

  ovs_from_opr_id = null;
  ovs_id = null;
  result_value = null;
  result_add_value = 0;

  if (rep_date is null or rep_date >= '31.12.2999') then
  begin
    for select opr_id, opr_subject_id, opr_subject_kind, ovs_id, ovs_value, ovs_add_value, ovs_date_begin
        from objectprops, objectvalues
        where :param_name = 'Object' and
              opr_object_kind = :uniq_param_table_id and
              opr_object_id = :uniq_param_id and
              ovs_id = opr_actual_ovs_id and
              ovs_from_opr_id = opr_id and
              (:result_table_id is null or
              opr_subject_kind = :result_table_id)
        union
        select opr_id, opr_object_id, opr_object_kind, ovs_id, ovs_value, ovs_add_value, ovs_date_begin
        from objectprops, objectvalues
        where :param_name = 'Subject' and
              opr_subject_kind = :uniq_param_table_id and
              opr_subject_id = :uniq_param_id and
              ovs_id = opr_actual_ovs_id and
              ovs_from_opr_id = opr_id and
              (:result_table_id is null or
              opr_object_kind = :result_table_id)
        into :ovs_from_opr_id, :result_uniq_id, :result_table_id, :ovs_id, :result_value, :result_add_value, ovs_date
    do
    begin
      result_local_id = result_uniq_id;
      suspend;
    end
  end
  else
  begin
    for select opr_id, opr_subject_id, opr_subject_kind
        from objectprops
        where :param_name = 'Object' and
              opr_object_kind = :uniq_param_table_id and
              opr_object_id = :uniq_param_id and
              (:result_table_id is null or
              opr_subject_kind = :result_table_id)
        union
        select opr_id, opr_object_id, opr_object_kind
        from objectprops
        where :param_name = 'Subject' and
              opr_subject_kind = :uniq_param_table_id and
              opr_subject_id = :uniq_param_id and
              (:result_table_id is null or
              opr_object_kind = :result_table_id)
        into :ovs_from_opr_id, :result_uniq_id, :result_table_id
    do
    begin
      ovs_id = null;
      result_value = null;
      result_add_value = 0;
      result_local_id = result_uniq_id;
      select ovs_id, ovs_value, ovs_add_value, ovs_date_begin
      from objectvalues
      where ovs_from_opr_id = :ovs_from_opr_id and
            ovs_date_begin <= :rep_date and
            ovs_date_end >= :rep_date and
            ovs_is_active = 1
      into :ovs_id, :result_value, :result_add_value, ovs_date;

      if (ovs_id is not null) then
        suspend;
    end
  end
end
