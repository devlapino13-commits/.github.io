CREATE PROCEDURE HP_DO_KN_OPERBOOK_DIFF_TAX(
    REP_DATE D_DATE)
RETURNS (
    LINE_ID D_INTEGER,
    ACC_ID D_ID,
    ACC_CODE D_STR_CODE,
    AB_CODE D_STR_CODE,
    KOB_SUMMA_BAL D_CURRENCY,
    KOB_SUMMA_TAX D_CURRENCY,
    TAX_KIND D_SMALLINT)
AS
begin
  line_id = 0;
  for select kobs_acc_id, acc_code, ab_code, kobs_bal_sum, kobs_bal_tax, ab_tax_kind
      from kn_obsummary, ref_account, ref_accbalance
      where kobs_date <= :rep_date and
            kobs_date_end >= :rep_date and
            kobs_acc_id = acc_id and
            acc_from_add_ab_id = ab_id and
            ab_tax_kind in (0, 1) and
            ((ab_tax_kind = 0 and kobs_bal_sum <> kobs_bal_tax) or
             (ab_tax_kind = 1 and kobs_bal_tax <> 0))
      into acc_id, acc_code, ab_code, kob_summa_bal, kob_summa_tax, tax_kind
  do
  begin
    line_id = line_id + 1;

--0 - Всегда БУ = НУ
--1 - Всегда НУ = 0
--2 - НУ может отличаться от БУ

    suspend;
  end
end
