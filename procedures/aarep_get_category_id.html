CREATE PROCEDURE AAREP_GET_CATEGORY_ID(
    REP_K_ID ID_TYPE,
    REP_DATE_END DATE,
    DEF_CURR VARCHAR(3),
    AA_CURR_KC_ID ID_TYPE,
    AA_DISB_KC_ID ID_TYPE,
    DD_DISB_DATE DATE,
    KAL_DATE DATE,
    DELINQ_DAYS INTEGER)
RETURNS (
    KONT_CATEGORY VARCHAR(20),
    AA_CATEGORY_ID SMALLINT)
AS
declare variable AA_LOAN_KIND_NEW varchar(20);
declare variable REP_DELINQ_DAYS integer;
declare variable KAL_DATE_CHEK smallint;
declare variable KC_PRIOR_ID decimal(18,0);
declare variable KF_ID decimal(18,0);
declare variable KAL_DATE_PRIOR date;
declare variable SH_DATE date;
declare variable KOL integer;
declare variable REP_DELINQ_BEGIN date;
declare variable K_IS_PROBLEM integer;
declare variable K_FROM_PR_ID integer;
begin
  kont_category = null;
--  DELINQ_DAYS = null;

  if (def_curr is null) then
  begin
    select cm_default_currency
    from company
    into def_curr;
  end

  if (def_curr = '944') then
  begin
    kont_category = 'Категория 1';
    aa_category_id = 1;
    suspend;
    exit;
  end

  if (def_curr = '417') then
  begin
    if (k_is_problem is null or k_from_pr_id is null or kf_id is null or dd_disb_date is null or aa_disb_kc_id is null) then
    begin
      select k_is_problem, k_from_pr_id, k_from_kf_id, k_disb_date, k_fact_disb_lo_id
      from kontract
      where k_id = :rep_k_id
      into k_is_problem, k_from_pr_id, kf_id, :dd_disb_date, aa_disb_kc_id;
    end

    aa_loan_kind_new = 'Категория A';
    aa_category_id = 1;
    if (k_is_problem = 1) then
    begin
      if (exists(select *
                 from sb_akt, sb_list
                 where sba_from_k_id = :rep_k_id and
                       sba_date_begin <= :rep_date_end and
                       sba_date_end >= :rep_date_end and
                       sba_status = 2 and
                       sba_is_delete = 0 and
                       sba_from_sbl_id <> 0 and
                       sba_from_sbl_id = sbl_id and
                       sbl_type = 1)) then
      begin
        aa_loan_kind_new = 'Категория C';
--                   AA_IS_SB=1;
        aa_category_id = 3;
      end
    end
    suspend;
    exit;
  end
  if (def_curr = '810') then
  begin
    if (rep_date_end between '01.01.2016' and '30.07.2016') then
    begin
      if (aa_curr_kc_id is null) then -- Если СВЗ не передан в параметре, то вычисляем
      begin
        select oks_kc_id
        from oksummary
        where oks_from_k_id = :rep_k_id and
              :rep_date_end between oks_date_begin and oks_date_end
              and oks_close_state = 0 
        into :aa_curr_kc_id;
      end
      if (aa_curr_kc_id = 1000000000003) then -- если текущий СВЗ является Вирт.сотр.3, значит 3 категория
      begin
        kont_category = 'Категория 3';
        aa_category_id = 3;
      end
      else
      begin
        kont_category = 'Категория 1';
        aa_category_id = 1;
      end
      suspend;
      exit;
    end

    if (rep_date_end >= '31.07.2016') then
    begin
      if (aa_curr_kc_id is null or delinq_days is null) then -- Если СВЗ не передан в параметре, то вычисляем
      begin
        select oks_kc_id, datediff(day, oks_delinq_date, :rep_date_end)
        from oksummary
        where oks_from_k_id = :rep_k_id and
              :rep_date_end between oks_date_begin and oks_date_end
              and oks_close_state = 0 
        into :aa_curr_kc_id, delinq_days;
        if (delinq_days is null) then
          delinq_days = 0;
      end
      if (aa_curr_kc_id = 1000000000003) then -- если текущий СВЗ является Вирт.сотр.3, значит 3 категория
      begin
        kont_category = 'Категория 3';
        aa_category_id = 3;
      end
      else
      begin
        kont_category = 'Категория 1';
        aa_category_id = 1;
      end
--      Временно отменено так как влият на отчеты акционерам. Sanjari
--      if (delinq_days>360) then
--      begin
--        KONT_CATEGORY = 'Категория 3';
--        AA_CATEGORY_ID=3;
--      end

      suspend;
      exit;
    end

    if (k_is_problem is null or k_from_pr_id is null or kf_id is null or dd_disb_date is null or aa_disb_kc_id is null) then
    begin
      select k_is_problem, k_from_pr_id, k_from_kf_id, k_disb_date, k_fact_disb_lo_id
      from kontract
      where k_id = :rep_k_id
      into k_is_problem, k_from_pr_id, kf_id, :dd_disb_date, aa_disb_kc_id;
    end

    if (rep_date_end < '01.12.2012') then
    begin
      aa_loan_kind_new = 'Категория A';
      aa_category_id = 1;
      if (k_is_problem = 1) then
      begin
        if (exists(select *
                   from sb_akt, sb_list
                   where sba_from_k_id = :rep_k_id and
                         sba_date_begin <= :rep_date_end and
                         sba_date_end >= :rep_date_end and
                         sba_status = 2 and
                         sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1)) then
        begin
          aa_loan_kind_new = 'Категория C';
          aa_category_id = 3;
          suspend;
          exit;
        end
      end
      if ((aa_loan_kind_new = 'Категория A') and
          (k_from_pr_id >= 18) and
          (k_from_pr_id <= 30)) then
      begin
        aa_loan_kind_new = 'Категория B';
        aa_category_id = 2;
      end
      suspend;
      exit;
    end

    if (delinq_days is null) then
    begin
      delinq_days = 0;
      select rep_delinq_days, rep_delinq_begin
      from aarep_kont_delinq_status_new(:rep_k_id, :rep_date_end)
      into :delinq_days, :rep_delinq_begin;
    end
    if (delinq_days is null) then
      delinq_days = 0;
    if ((delinq_days > 180) and
        (rep_date_end > '01.12.2012')) then
    begin
      aa_loan_kind_new = 'Категория 3';
      kont_category = aa_loan_kind_new;
      aa_category_id = 3;
      suspend;
      exit;
    end

    if ((rep_date_end >= '12.04.2013')) then
    begin
      if (exists(select *
                 from operkred
                 where ok_ol_is_active = 1 and
                       ok_from_k_id = :rep_k_id and
                       ok_type in (4, 5, 6) and
                       ok_ol_date between '01.12.2012' and :rep_date_end and
                       (ok_os + ok_perc) < -10 -- 10 руб за погрешность, на мелкие суммы не смотрим
                       and
                       ok_from_c_id = 0 and
                       (ok_ol_date - ok_sh_date) > 180)) then
      begin
        aa_loan_kind_new = 'Категория 3';
        kont_category = aa_loan_kind_new;
        aa_category_id = 3;
        suspend;
        exit;
      end
    end

    if (aa_curr_kc_id is null or kal_date is null) then
    begin
      aa_curr_kc_id = null;
      kal_date = null;
      select kal_new_kf_officer, kal_date
      from knaktlist
      where kal_from_kf_id = :kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end
      into aa_curr_kc_id, kal_date;
    end

    if (rep_date_end >= '14.07.2013') then
    begin
-- 14.07.2013
-- если последняя передача была самому себе (из разных офисов), смотреть на предыдущую дату
-- т.е. находим реальную дату передачи текущему сотруднику от других
      for select kal_new_kf_officer, kal_date
          from knaktlist
          where knaktlist.is_delete = 0 and
                kal_date >= :dd_disb_date and
                kal_date < :kal_date and
                kal_from_kf_id = :kf_id
          order by kal_date desc
          into :kc_prior_id, :kal_date_prior
      do
      begin
        if (kc_prior_id = aa_curr_kc_id) then
          kal_date = kal_date_prior;
        if ((kc_prior_id <> 1000000000000) and
            (kc_prior_id <> aa_curr_kc_id)) then
          break;
      end
    end

    if (rep_k_id is not null) then
    begin
      if (aa_curr_kc_id = aa_disb_kc_id) then
      begin
        if (dd_disb_date >= '01.06.2012') then
        begin
          aa_loan_kind_new = 'Категория 1';
          aa_category_id = 1;
        end
        else
        begin
          rep_delinq_days = 0;
          select rep_delinq_days
          from aarep_kont_delinq_status_new(:rep_k_id, '01.12.2012')
          into :rep_delinq_days;
          if (rep_delinq_days is null) then
            rep_delinq_days = 0;
          if (rep_delinq_days <= 0) then
          begin
            aa_loan_kind_new = 'Категория 1';
            aa_category_id = 1;
          end
          else
          if (rep_delinq_days <= 180) then
          begin
            aa_loan_kind_new = 'Категория 2';
            aa_category_id = 2;
          end
          else
          begin
            aa_loan_kind_new = 'Категория 3';
            aa_category_id = 3;
          end
        end
      end
      else
      begin
        if (kal_date < '01.12.2012') then
          kal_date = '01.12.2012'; -- добавлено от 17.05.2013 после отметы приказа №268-04-13

        rep_delinq_days = 0;
        if (kal_date is not null) then
        begin
          select rep_delinq_days
          from aarep_kont_delinq_status_new(:rep_k_id, :kal_date)
          into :rep_delinq_days;
          if (rep_delinq_days is null) then
            rep_delinq_days = 0;
        end

        if (rep_delinq_days <= 0) then
        begin
          aa_loan_kind_new = 'Категория 1';
          aa_category_id = 1;
        end
        else
        begin
          if (rep_delinq_days <= 180) then
          begin
            aa_loan_kind_new = 'Категория 2';
            aa_category_id = 2;
          end
          else
          begin
            aa_loan_kind_new = 'Категория 3';
            aa_category_id = 3;
          end
        end
      end

      if ((rep_date_end >= '14.07.2013') and
          (aa_category_id = 1) and
          (aa_curr_kc_id <> aa_disb_kc_id)) then
      begin
-- начиная с 14.07.2013 внесены изменения, если папка передана до 14.07.2013 или после 14.07.2013 но числится у виртуального, то
-- если до пережачи и после передачи в ближайшие два платежа были просрочки, считать их Категорией 2.

        sh_date = null; -- SH_Date - это ближайщий второй платеж по графику после передачи папки
        if ((kal_date < '14.07.2013') or (aa_curr_kc_id = 1000000000000)) then
        begin
          kol = 0;
          for select sh_date
              from schedule
              where sh_date >= :kal_date and
                    sh_from_k_id = :rep_k_id
              into :sh_date
          do
          begin
            kol = kol + 1;
            if (kol = 2) then
              break;
          end
          if (sh_date is null) then
            sh_date = kal_date;
        end
        else
          sh_date = kal_date;
        if (sh_date is not null) then
        begin
          if (rep_date_end >= cast('01.01.2016' as date)) then
          begin
            delinq_days = 0;
            select rep_delinq_days, rep_delinq_begin
            from aarep_kont_delinq_status_new(:rep_k_id, :rep_date_end)
            into :delinq_days, :rep_delinq_begin;
          end
          if ((rep_delinq_begin <= :sh_date) and
              (delinq_days > 7)) then
          begin
            aa_loan_kind_new = 'Категория 2';
            aa_category_id = 2;
          end
          else
          begin
            if (sh_date > rep_date_end) then
              sh_date = rep_date_end;
-- были ли просрочки более 7 дней до передачи и ближайшие два платежа после передачи
            if (exists(select *
                       from operkred
                       where ok_from_k_id = :rep_k_id and
                             ok_ol_is_active = 1 and
                             ok_ol_date <= :rep_date_end and
                             ok_sh_date <= :sh_date and
                             (ok_ol_date - ok_sh_date) > 7 and
                             ok_type in (4, 5, 6) and
                             ok_from_c_id = 0 and
                             (ok_os + ok_perc) < -10)) then -- 10 руб за погрешность, на мелкие суммы не смотрим
            begin
              aa_loan_kind_new = 'Категория 2';
              aa_category_id = 2;
            end
          end
        end
      end
      kont_category = aa_loan_kind_new;
      if (kont_category is null) then
        kont_category = 'Не определен';
      suspend;
    end
  end
end
