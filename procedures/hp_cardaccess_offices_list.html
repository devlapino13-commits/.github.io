CREATE PROCEDURE HP_CARDACCESS_OFFICES_LIST
RETURNS (
    UCF_ID D_INTEGER,
    UCF_FROM_REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    UCF_DATE_BEGIN D_DATE,
    UCF_DATE_END D_DATE,
    UCF_ACTIVE D_BOOL,
    UCF_FROM_UN_ID D_ID,
    UN_FIO D_STR_100,
    UCF_CHANGE_TIME D_DATETIME,
    UCF_LIMIT_AMOUNT D_CURRENCY)
AS
declare variable ucf_system d_smallint;
begin
  ucf_system = f_enum_id('access_sytem', 'card_disb_allow');
  for
    select
      ucf_id,
      reg_unicode,
      reg_code || ' ' || reg_name,
      ucf_date_begin,
      ucf_date_end,
      coalesce(ucf_active,0),
      un_id,
      un_fio,
      ucf_change_time,
      ucf_limit_amount
    from region
      left join underwritingcertif
        on ucf_from_reg_id = reg_unicode
          and ucf_system = :ucf_system
      left join username
        on un_id = ucf_from_un_id
    where reg_status = 3
      and current_date between coalesce(reg_date_open, '01.01.2000')
        and coalesce(reg_date_close, '31.12.9999')
    order by reg_name
  into
    :ucf_id,
    :ucf_from_reg_id,
    :reg_name,
    :ucf_date_begin,
    :ucf_date_end,
    :ucf_active,
    :ucf_from_un_id,
    :un_fio,
    :ucf_change_time,
    :ucf_limit_amount
  do
  begin
    suspend;
  end
end
