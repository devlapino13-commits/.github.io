CREATE PROCEDURE AAREP_ANALITICA_PORT_QUICK(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_CLIENT_FIO VARCHAR(100),
    AA_KOLICH_ACTIVE_BEGIN INTEGER,
    AA_KOLICH_ACTIVE_END INTEGER,
    AA_AMOUNT_ACTIVE_BEGIN DECIMAL(18,4),
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_LOAN_AMOUNT DECIMAL(18,4),
    AA_DELIQ_KOLICH INTEGER,
    AA_DELIQ_OS_ALL DECIMAL(18,4),
    AA_DELIQ_PERC_ALL DECIMAL(18,4),
    AA_DELIQ_BALANCE DECIMAL(18,4),
    AA_DELIQ_DAYS_OS INTEGER,
    AA_DELIQ_DAYS_PERC INTEGER,
    AA_DELIQ_REPAY_KOLICH INTEGER,
    AA_OFFICE_POPULATE_NAME VARCHAR(100),
    AA_LOAN_COMMON_NAME VARCHAR(100),
    AA_PRODUCT_NAME VARCHAR(100),
    AA_KOLICH_CL_ACTIVE_BEGIN INTEGER,
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    DD_DISB_DATE DATE,
    AA_DELIQ_CL_KOLICH INTEGER,
    AA_KF_NAME VARCHAR(100),
    AA_KONTRACT_NO VARCHAR(20),
    AA_CURRENCY VARCHAR(3),
    AAREP_PERC_BEGIN DECIMAL(18,4),
    AAREP_PERC_CALCED DECIMAL(18,4),
    AAREP_PERC_PAYED DECIMAL(18,4),
    AAREP_PERC_WRITEOFF DECIMAL(18,4),
    AAREP_PERC_END DECIMAL(18,4),
    AA_CURR_REGION_ID INTEGER,
    AA_LOAN_TYPE INTEGER,
    AA_KREDIT_END DATE)
AS
declare variable level_id integer;
declare variable kolich integer;
declare variable ol_date date;
declare variable ok_sh_date date;
declare variable ok_sh_id integer;
declare variable sh_os decimal(18,4);
declare variable sh_perc decimal(18,4);
declare variable ok_os decimal(18,4);
declare variable ok_perc decimal(18,4);
declare variable ok_type integer;
declare variable balance_begin decimal(18,4);
declare variable balance_end decimal(18,4);
declare variable client_kolich integer;
declare variable curr_kolich integer;
declare variable kont_amount decimal(18,4);
declare variable dd_prior date;
declare variable deliq_os decimal(18,4);
declare variable deliq_perc decimal(18,4);
declare variable deliq_os_all decimal(18,4);
declare variable deliq_perc_all decimal(18,4);
declare variable is_first_line integer;
declare variable first_deliq_dd date;
declare variable aa_amount_disb decimal(18,4);
declare variable aarep_percent_per_day decimal(18,7);
declare variable aarep_calc_method integer;
declare variable k_stop_perc date;
declare variable aa_k_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
declare variable db_id integer;
declare variable k_zero_balance_date date;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  db_id = f_get_db_id();

  for select distinct ks_from_k_id, kal_new_reg_id, kal_new_kf_name, office.reg_code || ' ' || office.reg_name,
                      branch.reg_code || ' ' || branch.reg_name, reg_level.reg_code || ' ' || reg_level.reg_name,
                      reg_level.reg_unicode, c_name_fio, k_kolich_client_approved, k_amount_approved, k_index_curr_code,
                      k_disb_date, office.reg_populate_kind, pr_common_name, pr_name, k_kontract_no,

                      k_percent_per_day, pr_interest_method, k_stop_perc, pr_loan_type, k_kredline_end, k_zero_balance_date

      from kontstate, kontract, knaktlist, region_list(:rep_region_id, 1, -1, 1), clients, product, region office, region branch, region reg_level
      where ks_from_k_id = k_id and
            ((ks_date_end >= :rep_date_begin and
            cast(ks_date_begin as date) <= :rep_date_end and
            ks_state = 1) or
            (ks_date_begin = :rep_date_begin and
            ks_state = 2)) and
            ks_ol_is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            (kal_new_reg_id = id_list) and
            c_id = k_clients_cashier_id and
            office.reg_unicode = kal_new_reg_id and
            branch.reg_unicode = office.parent and
            reg_level.reg_unicode = branch.parent and
            k_from_pr_id = pr_id
      order by kal_new_kf_officer
      into :aa_k_id, :aa_curr_region_id, :aa_kf_name, :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_level_name,
           :level_id, :aa_client_fio, :client_kolich, :kont_amount, :aa_currency, :dd_disb_date,
           :aa_office_populate_name, :aa_loan_common_name, :aa_product_name, :aa_kontract_no, :aarep_percent_per_day,
           :aarep_calc_method, :k_stop_perc, :aa_loan_type, :aa_kredit_end, :k_zero_balance_date
  do
  begin
    aa_amount_disb = :kont_amount;
    if (level_id = 1000000) then
      aa_curr_level_name = :aa_curr_branch_name;
    kolich = 0;

    balance_begin = 0;
    balance_end = 0;

    aa_kolich_active_begin = 0;
    aa_kolich_active_end = 0;

    aa_kolich_cl_active_begin = 0;
    aa_kolich_cl_active_end = 0;

    aa_amount_active_begin = 0;
    aa_amount_active_end = 0;

    aa_deliq_kolich = 0;
    aa_deliq_os_all = 0;
    aa_deliq_perc_all = 0;
    aa_deliq_balance = 0;
    aa_deliq_days_os = 0;
    aa_deliq_days_perc = 0;
    dd_prior = :rep_date_begin;
    deliq_os = 0;
    deliq_perc = 0;
    deliq_os_all = 0;
    deliq_perc_all = 0;
    aa_deliq_repay_kolich = 0;
    is_first_line = 1;
    first_deliq_dd = :rep_date_end;

    aa_deliq_cl_kolich = 0;
    aa_kolich_cl_active_end = 0;
    aa_kolich_cl_active_begin = 0;

    for select ok_ol_date, ok_type, ok_sh_date, ok_sh_id, ok_sh_os, ok_sh_perc, ok_os, ok_perc
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :aa_k_id and
              ok_ol_date <= :rep_date_end and
              ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
              and
              ok_from_c_id = 0
        order by ok_sh_date, ok_sh_id, ok_ol_date, ok_from_ol_id
        into :ol_date, :ok_type, :ok_sh_date, :ok_sh_id, :sh_os, :sh_perc, :ok_os, :ok_perc
    do
    begin
      if (ok_sh_date <= :rep_date_end) then
      begin
        if (ok_sh_id <> 0) then
        begin
          if (ok_type < 8) then
          begin
            if ((is_first_line = 1) or (:ok_sh_date = :dd_prior)) then
            begin
              deliq_os = :deliq_os + :sh_os + :ok_os;
              deliq_perc = :deliq_perc + :sh_perc + :ok_perc;
              if (is_first_line = 1) then
                dd_prior = :ok_sh_date;
            end
            else
            begin
              if ((deliq_os > 0) or (deliq_perc > 0)) then
              begin
                aa_deliq_repay_kolich = :aa_deliq_repay_kolich + 1;
                deliq_os_all = :deliq_os_all + :deliq_os;
                deliq_perc_all = :deliq_perc_all + :deliq_perc;
                if (first_deliq_dd = :rep_date_end) then
                  first_deliq_dd = :dd_prior;
              end
              dd_prior = :ok_sh_date;
              deliq_os = :sh_os + :ok_os;
              deliq_perc = :sh_perc + :ok_perc;
            end
            is_first_line = 0;
          end
        end
      end
      if (ok_type <= 8) then
        balance_end = :balance_end + :ok_os;
      if (:ol_date < :rep_date_begin) then
      begin
        if (ok_type <= 8) then
          balance_begin = :balance_begin + :ok_os;
      end
      kolich = 1;
    end /* Расчет сумм окончен */

    if (balance_begin < 0) then
      balance_begin = 0;
    if (balance_end < 0) then
      balance_end = 0;
    if (balance_begin > 0) then
      aa_kolich_active_begin = 1;
    if (balance_end > 0) then
      aa_kolich_active_end = 1;

    /* CLIENT_KOLICH */
    aa_kolich_cl_active_begin = :aa_kolich_active_begin * :client_kolich;
    aa_kolich_cl_active_end = :aa_kolich_active_end * :client_kolich;

    if (kolich > 0) then
    begin

      if ((:deliq_os > 0) or (:deliq_perc > 0)) then
      begin
        aa_deliq_repay_kolich = :aa_deliq_repay_kolich + 1;
        deliq_os_all = :deliq_os_all + :deliq_os;
        deliq_perc_all = :deliq_perc_all + :deliq_perc;
        if (first_deliq_dd = :rep_date_end) then
          first_deliq_dd = :dd_prior;
      end
      if (:deliq_os_all + :deliq_perc_all > 0) then
      begin
        aa_deliq_kolich = 1;
        aa_deliq_os_all = :deliq_os_all;
        aa_deliq_perc_all = :deliq_perc_all;
        if (:deliq_os_all > 0) then
          aa_deliq_days_os = cast((:rep_date_end - :first_deliq_dd) as integer);
        if (:deliq_perc_all > 0) then
          aa_deliq_days_perc = cast((:rep_date_end - :first_deliq_dd) as integer);
      end
      else
      begin
        aa_deliq_repay_kolich = 0;
        aa_deliq_kolich = 0;
        deliq_os_all = 0;
        deliq_perc_all = 0;
        aa_deliq_os_all = 0;
        aa_deliq_perc_all = 0;
        aa_deliq_days_os = 0;
        aa_deliq_days_perc = 0;
      end
      if (balance_end = 0) then
      begin
        aa_deliq_repay_kolich = 0;
        aa_deliq_kolich = 0;
        deliq_os_all = 0;
        deliq_perc_all = 0;
        aa_deliq_os_all = 0;
        aa_deliq_perc_all = 0;
        aa_deliq_days_os = 0;
        aa_deliq_days_perc = 0;
      end
      curr_kolich = 0;

      curr_kolich = :curr_kolich + 1;
      aa_amount_active_begin = :balance_begin;
      aa_amount_active_end = :balance_end;
      aa_deliq_os_all = :deliq_os_all;
      aa_deliq_perc_all = :deliq_perc_all;

      if (aa_deliq_kolich = 1) then
      begin
        aa_deliq_balance = :aa_amount_active_end;
      end
      aa_deliq_cl_kolich = :aa_deliq_kolich * :client_kolich;
      aa_loan_amount = :aa_amount_disb;

      aarep_perc_begin = 0;
      aarep_perc_calced = 0;
      aarep_perc_payed = 0;
      aarep_perc_writeoff = 0;
      aarep_perc_end = 0;

      if (f_new_perc_active(rep_date_end) = 1) then
      begin
        select result_perc_begin, result_calced, result_payed, result_writeoff, result_perc_end
        from aarep_calc_perc_one_period(:aa_k_id, :rep_date_begin, :rep_date_end)
        into :aarep_perc_begin, :aarep_perc_calced, :aarep_perc_payed, :aarep_perc_writeoff, :aarep_perc_end;
      end else
      begin
        select result_perc_begin, result_calced, result_payed, result_writeoff, result_perc_end
        from aarep_calc_perc_one(:aa_k_id, :rep_date_begin, :rep_date_end, 1, :db_id, :aa_kredit_end, :k_stop_perc, :k_zero_balance_date)
        into :aarep_perc_begin, :aarep_perc_calced, :aarep_perc_payed, :aarep_perc_writeoff, :aarep_perc_end;
      end

      suspend;
    end
  end
end
