CREATE PROCEDURE HP_CLP_CLARIFICATION(
    BS_ID D_INTEGER,
    CQ_ID D_INTEGER)
RETURNS (
    CC_ID D_INTEGER,
    CC_QUESTION_ID D_INTEGER,
    CC_SUBTYPE_ID D_INTEGER,
    CC_NAME D_STR_LONG,
    CC_FROM_UN_ID D_ID,
    UN_FIO D_STR_100,
    CC_DATETIME D_DATETIME,
    CC_IS_DELETE D_BOOL)
AS
begin
  for
    select
      cc_id,
      cc_question_id,
      cc_subtype_id,
      cc_name,
      cc_from_un_id,
      un_fio,
      cc_datetime,
      cc_is_delete
    from clp_clarification
      inner join username
        on un_id = cc_from_un_id
    where cc_subtype_id = :bs_id
      and cc_question_id = :cq_id
  into
    :cc_id,
    :cc_question_id,
    :cc_subtype_id,
    :cc_name,
    :cc_from_un_id,
    :un_fio,
    :cc_datetime,
    :cc_is_delete
  do
  begin
    suspend;
  end
end
