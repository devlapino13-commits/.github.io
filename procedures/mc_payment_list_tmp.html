CREATE PROCEDURE MC_PAYMENT_LIST_TMP(
    MCL_ID D_ID,
    K_ID D_ID)
RETURNS (
    PAYMENT_DATE D_DATE,
    AMOUNT D_STR_100,
    IS_ACTIVE D_SMALLINT,
    IS_DELINQ D_SMALLINT)
AS
declare variable disb_date d_date;
declare variable db_id d_integer;
declare variable c_id id_type;
declare variable d_amount d_currency;
begin

  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  if (coalesce(k_id, 0) <= 0) then
    exception e_default 'parametr "k_id" has invalid value';

  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :c_id;

  if (c_id is null) then
    exit;

  db_id = f_get_db_id();

  select kont.k_fact_disb_date
  from kontract kont
  inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
  where kont.k_id = :k_id and
        kncl.kc_amount_approved > 0 and
        kncl.kc_from_c_id = :c_id
  into :disb_date;

  if (disb_date is null) then
    exit;

  -- актиные платежи ------- >
  is_active = 1;
  for select sh_date, amount
      from (select iif(sh_date is null, cast('31.12.9999' as date), sh_date) as sh_date,
                   sum(zad_os + iif(zad_perc > sh_calc_perc, zad_perc, sh_calc_perc) + sh_calc_fine + zad_kommis + iif(:db_id = 41701, round(zad_perc * 0.02), 0)) as amount
            from aarep_make_zad_list(:k_id, current_date, 1)
            group by sh_date) a
      order by sh_date desc
      into :payment_date, :d_amount
  do
  begin
    is_delinq = 0;

    if ((payment_date = '31.12.9999') or (payment_date < current_date)) then
      is_delinq = 1;

    if (payment_date = '31.12.9999') then
    begin
      select okf.oko_ol_date
      from operkroff okf
      where okf.oko_from_k_id = :k_id and
            okf.oko_ol_is_active = 1 and
            okf.oko_type = 1
      order by okf.oko_ol_date desc
      rows 1
      into :payment_date;
    end

    if (d_amount <> 0) then
    begin
      amount = round(d_amount, 2);

      if (db_id <> 94401) then
        amount = amount || iif(db_id = 41701, ' сом.', iif(db_id = 39801, ' тенге',' руб.'));
      else
        amount = amount || ' manat.';

      amount = replace(amount, '.00', '');
      suspend;
    end
  end
  ----------------------<
  -- закрытые платежи -->
  is_active = 0;
  is_delinq = 0;
  for select ok.ok_sh_date, -sum(ok.ok_os + ok.ok_perc + ok.ok_fine + ok.ok_tax)
      from operkred ok
      where ok.ok_type in (4, 5, 9) and
            ok.ok_ol_is_active = 1 and
            (:db_id = 41701 or ok.ok_sh_date < :payment_date) and
            ok.ok_from_k_id = :k_id and
            ok.ok_ol_date > :disb_date
      group by ok.ok_sh_date
      order by ok.ok_sh_date desc
      into :payment_date, :d_amount
  do
  begin
    if (d_amount <> 0) then
    begin
      amount = round(d_amount, 2);

      if (db_id <> 94401) then
        amount = amount || iif(db_id = 41701, ' сом.', iif(db_id = 39801, ' тенге',' руб.'));
      else
        amount = amount || ' manat.';

      amount = replace(amount, '.00', '');
      suspend;
    end
  end
  ------------------ <
end
