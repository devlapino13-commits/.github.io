CREATE PROCEDURE HP_DO_RESCHEDULE_KZ(
    K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255,
    DELINQ_START_DATE D_DATE,
    SH_DATE D_DATE,
    FINE CURR_TYPE,
    MONTH_ADD D_INTEGER)
AS
declare variable REG_ID D_ID;
declare variable K_KONTRACT_NO D_STR_100;
declare variable K_DISB_DATE D_DATE;
declare variable CONTROL_AMOUNT CURR_TYPE;
declare variable OL_ID D_ID;
declare variable DELINQ_MONTH D_INTEGER;
declare variable MONTHLY_PAYMENT CURR_TYPE;
declare variable CONTROL_AMOUNT_NEW CURR_TYPE;
declare variable C_ID D_ID;
declare variable MONTHLY_PAYMENT_1 CURR_TYPE;
declare variable MONTHLY_PAYMENT_TEMP CURR_TYPE;
declare variable K_PERIOD D_INTEGER;
declare variable FINE_ACCOUNT D_STR_20;
declare variable REPAY_ACCOUNT D_STR_20;
declare variable WRITEOFF_ACCOUNT D_STR_20;
declare variable ADD_PARAMS D_STR_LONG;
declare variable NON_DELINQ_MONTH D_INTEGER;
declare variable FINAL_MONTH_AMOUNT CURR_TYPE;
declare variable K_STOP_PERC D_DATE;
declare variable ADD_MONTH_90 D_INTEGER;
declare variable LAST_MONTH D_DATE;
declare variable OL_MONTH D_INTEGER;
declare variable SH_MONTH D_DATE;
declare variable EFFECT_RATE_COMMON d_percent;
begin
 k_stop_perc = null;
  select k.k_fact_disb_reg_id, k.k_kontract_no, k.k_disb_date, k.k_period, k.k_stop_perc,
         kc.kc_from_c_id
  from kontract k
  inner join knclient kc on kc.kc_from_k_id = k.k_id and
        kc.kc_amount_approved > 0
  where k.k_id = :k_id
  into reg_id, k_kontract_no, k_disb_date, k_period, k_stop_perc, c_id;

  select oks.oks_delinq_date
  from oksummary oks
  where oks.oks_from_k_id = :k_id and
        oks.oks_close_state = 0 and
        :rep_date between oks.oks_date_begin and oks.oks_date_end
  into :delinq_start_date;

  select sum(sh_calc_fine + sh_calc_perc + zad_os), sum(sh_calc_fine)
  from aarep_make_zad_list(:k_id, :rep_date, 1)
  into control_amount, fine; -- общая сумма задолженности и пеня

  with t_zad
  as (select ok_sh_id, ok_sh_date, sum(ok.ok_os + ok_sh_os) sh_os, sum(ok.ok_perc + ok_sh_perc) sh_perc
      from operkred ok
      where ok.ok_from_k_id = :k_id and
            ok.ok_type < 7 and
            ok.ok_ol_is_active = 1 and
            ok_from_c_id = 0
      group by 1, 2)
  select count(ok_sh_date)
  from t_zad
  where :rep_date >= ok_sh_date and
        (sh_os <> 0 or sh_perc <> 0)
  into delinq_month; -- количество просроченных платежей

  non_delinq_month = 0;
  with t_zad
  as (select ok_sh_id, ok_sh_date, sum(ok.ok_os + ok_sh_os) sh_os, sum(ok.ok_perc + ok_sh_perc) sh_perc
      from operkred ok
      where ok.ok_from_k_id = :k_id and
            ok.ok_type < 7 and
            ok.ok_ol_is_active = 1 and
            ok_from_c_id = 0
      group by 1, 2)
  select count(ok_sh_date)
  from t_zad
  where :rep_date < ok_sh_date and
        (sh_os <> 0 or sh_perc <> 0)
  into non_delinq_month; -- количество непросроченных платежей

  month_add = delinq_month + 4; -- по условию пролеваем срок кредитования на кол-во просроченных платежей плюс 2
  monthly_payment = 0;

  with zad_all
  as (select sh_os + sh_perc as payment, sh_id
      from schedule
      where sh_from_k_id = :k_id and
            sh_id > 0)
  select round(avg(payment), 0)
  from zad_all
  where sh_id < (select max(sh_id)
                 from zad_all)
  into monthly_payment;
  if (k_stop_perc is null) then
  begin
    if (coalesce(non_delinq_month, 0) <= 0) then
    begin
      month_add = month_add + 1;
      add_params = f_format('{"protocol_date":"%1","reason":4088,"memo":"реструктуризация графика погашений займа %2 от %3","first_perc_kind":1, "grace_os_list":1, "perc_lgota_value":100, "add_month_kind":2, "add_month_count":%4, "new_annu": %5}', :rep_date, :k_kontract_no, :k_disb_date, :month_add, :monthly_payment);
    end
    else
      add_params = f_format('{"protocol_date":"%1","reason":4088,"memo":"реструктуризация графика погашений займа %2 от %3","first_perc_kind":1, "grace_os_list":"1", "perc_lgota_value":100, "add_month_kind":1, "add_month_count":%4, "new_annu": %5}', :rep_date, :k_kontract_no, :k_disb_date, :month_add, :monthly_payment);

    fine_account = (select al.al_id
                    from acc_list al
                    where al.al_from_reg_id = :reg_id and
                          al.al_is_active = 1 and
                          al.al_account = '1280.09/1');

    select ol_id, error_text
    from hp_do_repayment(:k_id, 0, :rep_date, :fine, 0, 0, 1, 1, 1, 0, 1, :reg_id, 3, :fine_account, 0, 1000000000000, null, 'cписание пении', null, f_get_rd_id('Reason_WriteOff.fin_difficulties'), :control_amount)
    into ol_id, res_text; -- списание пении

    -- делаем реструктуризацию
    /* add_params = f_format('{"protocol_date":"%1","reason":4088,"memo":"реструктуризация графика погашений займа %2 от %3","first_perc_kind":1,"sh_rec_id":1,"perc_lgota_value":100, "count":%4, "amount": %5}', :rep_date, :k_kontract_no, :k_disb_date, :month_add, :monthly_payment);
  select res_is_error, res_text
  from hp_do_reschedule(:k_id, :rep_date, 'load', :add_params, 1000000000000)
  into res_is_error, res_text;

  select res_is_error, res_text
  from hp_do_reschedule(:k_id, :rep_date, 'grace_perc',:add_params, 1000000000000)
  into res_is_error, res_text;

  select res_is_error, res_text
  from hp_do_reschedule(:k_id, :rep_date, 'add_month', :add_params, 1000000000000)
  into res_is_error, res_text;

  select res_is_error, res_text
  from hp_do_reschedule(:k_id, :rep_date, 'new_annu', :add_params, 1000000000000)
  into res_is_error, res_text;
                                */
    select res_is_error, res_text
    from hp_do_reschedule(:k_id, :rep_date, 'auto', :add_params, 1000000000000)
    into res_is_error, res_text;

    select res_is_error, res_text
    from hp_do_reschedule(:k_id, :rep_date, 'save', :add_params, 1000000000000)
    into res_is_error, res_text;

    /*  update kontdata kd
  set kd.kd_ol_date = dateadd(1 day to (dateadd(1 month to :rep_date)))
  where kd.kd_from_k_id = :k_id and
        kd.kd_ol_is_delete = 0 and
        kd.kd_date_end = '31.12.9999'; */

    repay_account = (select al.al_id
                     from acc_list al
                     where al.al_from_reg_id = :reg_id and
                           al.al_is_active = 1 and
                           al.al_account = '1021/1');

    select sum(sh_calc_fine + sh_calc_perc + zad_os)
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    into control_amount_new; -- общая сумма задолженности после реструктуризации

    if (res_is_error = 0) then
      select ol_id, error_text
      from hp_do_repayment(:k_id, 0, :rep_date, 0, 1, 1, 1, 1, 1, 0, 0, :reg_id, 3, :repay_account, 0, 1000000000000, :c_id, '', 0, 0, :control_amount_new)
      into ol_id, res_text; -- делаем нулевое погашение

    monthly_payment_1 = 0;
    monthly_payment_temp = 0;

    -- проверяем первые 2 ежемесячного платежа
    for select first 2 zad_os + zad_perc
        from aarep_make_zad_list(:k_id, :rep_date, 1)
        where sh_date >= :rep_date
        order by sh_date
        into monthly_payment_temp
    do
    begin
      if (monthly_payment_1 = 0) then
        monthly_payment_1 = monthly_payment_temp;
    end
    if ((monthly_payment_1 - monthly_payment_temp) > 0) then -- если первый платеж больше чем второй списываем разницу
    begin
      if (k_period > 12) then
        writeoff_account = (select al.al_id
                            from acc_list al
                            where al.al_from_reg_id = :reg_id and
                                  al.al_is_active = 1 and
                                  al.al_account = '1290.21/1');
      else
        writeoff_account = (select al.al_id
                            from acc_list al
                            where al.al_from_reg_id = :reg_id and
                                  al.al_is_active = 1 and
                                  al.al_account = '2170.25/1');

      select ol_id, error_text
      from hp_do_repayment(:k_id, 0, :rep_date, (:monthly_payment_1 - :monthly_payment_temp), 1, 1, 1, 1, 1, 0, 1, :reg_id, 3, :writeoff_account, 0, 1000000000000, 0, 'списание', 0, f_get_rd_id('Reason_WriteOff.fin_difficulties'), :control_amount_new)
      into ol_id, res_text;
    end
   
  effect_rate_common = 0;
  delete from temp_schedule
  where 1 = 1;

  insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
  select sh_id, sh_date, sh_days, sh_all
  from hp_effect_rate_reschedule_kz(:k_id, :rep_date);

  select calc_eff
  from hp_effect_rate_kg(:k_id)
  into effect_rate_common; -- есп в расчете вбс

  update kontdata kd
  set kd.kd_percent_effect_fifd = :effect_rate_common
  where kd.kd_from_k_id = :k_id and
        kd.kd_ol_is_delete = 0 and
        kd.kd_date_end >= '31.12.9999';
  end
  else
  begin
    add_month_90 = 0;
    last_month = null;
    ol_month = extract(month from :rep_date);
    for select sh_date
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    where zad_os <> 0 or zad_perc <> 0
    order by sh_date desc
    rows 2
    into sh_month
    do
    begin
     if (last_month is null) then
         last_month = sh_month;
       if (extract(month from last_month) = ol_month) then
           add_month_90 = 1;
       else if (last_month > :rep_date) then
           add_month_90 = 1;
       else
       if (last_month <> sh_month) then
       begin
         add_month_90 = datediff(month, last_month, :rep_date);
         if (datediff(month, sh_month, last_month)> 1) then
           add_month_90 = add_month_90 + 1;
       end
    end
    select res_is_error
    from hp_kont_reschedule_move(:k_id, :rep_date, :add_month_90, 1000000000000)
    into res_is_error;
  end
  /* with t_zad
  as (select ok_sh_id, ok_sh_date, sum(ok.ok_os + ok_sh_os) sh_os, sum(ok.ok_perc + ok_sh_perc) sh_perc
      from operkred ok
      where ok.ok_from_k_id = :k_id and
            ok.ok_type < 7 and
            ok.ok_ol_is_active = 1 and
            ok_from_c_id = 0
      group by 1, 2)
  select (sh_os + sh_perc)
  from t_zad
  where ok_sh_date = (select max(ok_sh_date)
                   from t_zad)
  into final_month_amount; -- количество непросроченных платежей

  if (final_month_amount > monthly_payment) then
  begin
  add_params = f_format('{"protocol_date":"%1","reason":4088,"memo":"реструктуризация графика погашений займа %2 от %3","first_perc_kind":1, "grace_os_list":0, "perc_lgota_value":0, "add_month_kind":1, "add_month_count":%4, "new_annu": %5}', :rep_date, :k_kontract_no, :k_disb_date, 1, :monthly_payment);
  select res_is_error, res_text
  from hp_do_reschedule(:k_id, :rep_date, 'auto', :add_params, 1000000000000)
  into res_is_error, res_text;

  select res_is_error, res_text
  from hp_do_reschedule(:k_id, :rep_date, 'save', :add_params, 1000000000000)
  into res_is_error, res_text;

    
  if (res_is_error = 0) then
    select ol_id, error_text
    from hp_do_repayment(:k_id, 0, :rep_date, 0, 1, 1, 1, 1, 1, 0, 0, :reg_id, 3, :repay_account, 0, 1000000000000, :c_id, '', 0, 0, :control_amount_new)
    into ol_id, res_text;
  end */
  suspend;
end
