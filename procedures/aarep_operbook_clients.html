CREATE PROCEDURE AAREP_OPERBOOK_CLIENTS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_ACCOUNT VARCHAR(20),
    REP_IS_DT INTEGER,
    REP_IS_FULL INTEGER)
RETURNS (
    OB_FROM_OL_ID DECIMAL(18,0),
    OB_NUMBER INTEGER,
    OB_CURRENCY VARCHAR(3),
    OB_CURR_RATE DECIMAL(18,4),
    OB_CLIENT_ID DECIMAL(18,0),
    OB_ACCOUNT_DT VARCHAR(20),
    OB_SUMMA_VAL DECIMAL(18,4),
    OB_SUMMA DECIMAL(18,4),
    OB_ACCOUNT_KT VARCHAR(20),
    OB_DETAIL VARCHAR(1024),
    OL_MEMO VARCHAR(1024),
    OL_DATE DATE,
    OL_FROM_REG_ID INTEGER,
    OL_REGION_NAME VARCHAR(100),
    OL_REGION_CODE VARCHAR(20),
    OB_CLIENT_NAME VARCHAR(100))
AS
declare variable ob_client_dt decimal(18,0);
declare variable ob_client_kt decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  if (rep_is_dt = 1) then
  begin
    for select ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa,
               ob_account_kt, ob_client_kt, ob_detail, ol_memo, ol_date, ol_from_reg_id, reg_name, reg_code,
               c_name_fio
        from operlist, region, region_list(:rep_region_id, 1, -1, 1), operbook
        left join clients on (c_id = ob_client_dt)
        where ol_id = ob_from_ol_id and
              ob_ol_is_active = 1 and
              ob_ol_reg_id = id_list and
              reg_unicode = ob_ol_reg_id and
              ob_ol_date between :rep_date_begin and :rep_date_end and
              ob_account_dt like '%' || :rep_account || '%' and
              (((c_id is null) and
              (:rep_is_full = 0)) or
              (:rep_is_full = 1))
        into :ob_from_ol_id, :ob_number, :ob_currency, :ob_curr_rate, :ob_client_dt, :ob_account_dt, :ob_summa_val,
             :ob_summa, :ob_account_kt, :ob_client_kt, :ob_detail, :ol_memo, :ol_date, :ol_from_reg_id, :ol_region_name,
             :ol_region_code, :ob_client_name
    do
    begin
      if (rep_is_dt = 1) then
        ob_client_id = :ob_client_dt;
      else
        ob_client_id = :ob_client_kt;
      suspend;
    end
  end
  else
  begin
    for select ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa,
               ob_account_kt, ob_client_kt, ob_detail, ol_memo, ol_date, ol_from_reg_id, reg_name, reg_code,
               c_name_fio
        from operlist, region, region_list(:rep_region_id, 1, -1, 1), operbook
        left join clients on (c_id = ob_client_kt)
        where ol_id = ob_from_ol_id and
              ob_ol_is_active = 1 and
              ob_ol_reg_id = id_list and
              reg_unicode = ob_ol_reg_id and
              ob_ol_date between :rep_date_begin and :rep_date_end and
              ob_account_kt like '%' || :rep_account || '%' and
              (((c_id is null) and
              (:rep_is_full = 0)) or
              (:rep_is_full = 1))
        into :ob_from_ol_id, :ob_number, :ob_currency, :ob_curr_rate, :ob_client_dt, :ob_account_dt, :ob_summa_val,
             :ob_summa, :ob_account_kt, :ob_client_kt, :ob_detail, :ol_memo, :ol_date, :ol_from_reg_id, :ol_region_name,
             :ol_region_code, :ob_client_name
    do
    begin
      if (rep_is_dt = 1) then
        ob_client_id = :ob_client_dt;
      else
        ob_client_id = :ob_client_kt;
      suspend;
    end
  end
end
