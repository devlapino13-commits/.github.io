CREATE PROCEDURE UXP$UXSOURCE_DIFF_COUNT(
    US_UUID D_UUID,
    US_UUID_CODE D_STR_CODE = null)
RETURNS (
    US_DIFFERENT D_SMALLINT)
AS
declare variable state_count d_integer;
declare variable us_kind d_smallint;
declare variable us_source d_str_code;
begin
  if (coalesce(us_uuid, '') = '') then
    us_uuid = uxf$uniq(us_uuid_code, 0);

  us_different = 0;

  select us_code, us_kind
  from ux$source
  where us_uuid = :us_uuid
  into us_source, us_kind;

  if (us_kind = 0) then
  begin
    select count(*)
    from ux$field
    where uf_owner = :us_uuid and
          uf_state <> 0
    into state_count;
  
    if (state_count <> 0) then
      us_different = 1;
  
    select count(*)
    from ux$index
    where uix_source = :us_source and
          uix_state <> 0
    into state_count;
  
    if (state_count <> 0) then
      us_different = us_different + 2;
  end

  suspend;
end
