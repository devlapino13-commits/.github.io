CREATE PROCEDURE AAREP_KN_REESTR_REPAYMENT(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    KOB_OL_DATE D_DATE,
    KOB_FROM_KC_ID ID_TYPE,
    K_KONTRACT_NO D_STR_20,
    K_FACT_DISB_DATE D_DATE,
    C_NAME_FIO D_STR_100,
    KOB_SUMMA D_CURRENCY,
    TYPE_PAY D_STR_10,
    DT_ACC_ADD_CODE D_STR_CODE,
    KT_ACC_ADD_CODE D_STR_CODE,
    DT_ACC_CODE D_STR_CODE,
    KT_ACC_CODE D_STR_CODE,
    UID D_STR_CODE,
    VID_PAY D_STR_CODE,
    C_BIRTH_DATE D_DATE,
    C_ID_STR D_STR_20)
AS
declare variable kob_id id_type;
declare variable ol_parent_id id_type;
declare variable dt_acc_add_code_bal d_str_code;
declare variable c_id id_type;
declare variable dt_acc_code_bal d_str_code;
begin
  for select distinct kob.kob_id,kob.kob_ol_date, kob.kob_from_kc_id, k.k_kontract_no, k.k_fact_disb_date, c.c_name_fio,
                      c.c_birth_date, c.c_id, kob.kob_summa, dt.acc_add_code dt_acc_add_code,
                      kt.acc_add_code kt_acc_add_code, dt.acc_code dt_acc_code, kt.acc_code kt_acc_code, null
      from kn_operbook kob
      inner join ref_account dt on dt.acc_id = kob.kob_acc_dt
      inner join ref_account kt on kt.acc_id = kob.kob_acc_kt
      inner join knclient kn on kn.kc_id = kob.kob_from_kc_id
      inner join kontract k on k.k_id = kn.kc_from_k_id
      left join clients c on c.c_id = kn.kc_from_c_id
      where kob.kob_ol_is_active = 1 and
          -- kob.kob_from_kc_id=1731013 and
            kob.kob_ol_date between :rep_date_begin and :rep_date_end and
            dt.acc_add_code in ('20501 (00)', '47423 (99)', '47416 (00)', '20202 (00)', '60305 (10)') and
            kt.acc_add_code in ('48801 (00)', '49201 (00)', '61217 (00)', '48809 (10)', '49209 (10)', '49209 (20)', '60322 (10)', '60323 (20)')

      union all

      select distinct kob.kob_id, kob.kob_ol_date, kob.kob_from_kc_id, k.k_kontract_no, k.k_fact_disb_date, c.c_name_fio,
                      c.c_birth_date, c.c_id, kob.kob_summa, dt.acc_add_code dt_acc_add_code,
                      kt.acc_add_code kt_acc_add_code, dt.acc_code dt_acc_code, kt.acc_code kt_acc_code, coalesce(ol.ol_parent_id, ol.ol_id)

      from operkroff ok
      inner join operlist ol on ol.ol_id = ok.oko_from_ol_id
      inner join kn_operbook kob on kob.kob_from_ol_id = coalesce(ol.ol_parent_id, ol.ol_id)
      inner join ref_account dt on dt.acc_id = kob.kob_acc_dt
      inner join ref_account kt on kt.acc_id = kob.kob_acc_kt
      inner join knclient kn on kn.kc_id=kob.kob_from_kc_id
      inner join kontract k on k.k_id = kn.kc_from_k_id
      inner join clients c on c.c_id = k.k_clients_cashier_id
      where kob.kob_ol_date between :rep_date_begin and :rep_date_end and
            kob.kob_ol_is_active = 1 and
           -- kob.kob_from_kc_id=1731013 and
            ok.oko_ol_is_active = 1 and
            dt.acc_add_code = '99999 (00)' and
            kt.acc_add_code in ('91804 (00)', '91702 (00)')

      into :kob_id,:kob_ol_date, :kob_from_kc_id, :k_kontract_no, :k_fact_disb_date, :c_name_fio, :c_birth_date, :c_id,
           :kob_summa, :dt_acc_add_code, :kt_acc_add_code, :dt_acc_code, :kt_acc_code, :ol_parent_id

  do
  begin
    c_id_str = '';
    c_id_str = 'n' || c_id;
    :dt_acc_add_code_bal=null;

    if (dt_acc_add_code = '99999 (00)') then
    begin

      select distinct dt.acc_add_code ,dt.acc_code
      from operlist ol
      inner join kn_operbook kob on kob.kob_from_ol_id = coalesce(ol.ol_parent_id, ol.ol_id)
      inner join ref_account dt on dt.acc_id = kob.kob_acc_dt
      where kob.kob_ol_is_active = 1 and
            ol.ol_id = :ol_parent_id and
            dt.acc_add_code <> '99999 (00)' and
            kob.kob_summa=:kob_summa and
            ol.ol_date between :rep_date_begin and :rep_date_end rows 1
      into :dt_acc_add_code_bal,:dt_acc_code_bal;

      :dt_acc_add_code = :dt_acc_add_code_bal;
      :dt_acc_code= :dt_acc_code_bal;
    end

    type_pay = '';
    if (dt_acc_add_code in ('20501 (00)', '47423 (99)', '47416 (00)') and
        kt_acc_add_code in ('48801 (00)', '49201 (00)', '61217 (00)', '91804 (00)', '48809 (10)', '49209 (10)', '91702 (00)', '60322 (10)', '60323 (20)')) then
      type_pay = 'Безнал.';

    if (dt_acc_add_code in ('20202 (00)') and
        kt_acc_add_code in ('48801 (00)', '49201 (00)', '61217 (00)', '91804 (00)', '48809 (10)', '49209 (10)', '91702 (00)', '49209 (20)', '60322 (10)', '60323 (20)')) then
      type_pay = 'Нал.';

    if (dt_acc_add_code in ('60305 (10)') and
        kt_acc_add_code in ('48801 (00)', '49201 (00)', '61217 (00)', '48809 (10)', '49209 (10)', '91702 (00)', '60322 (10)')) then
      type_pay = 'удерж с ЗП';

    vid_pay = '';
    if (dt_acc_add_code in ('20501 (00)', '47423 (99)', '47416 (00)', '20202 (00)', '60305 (10)') and
        kt_acc_add_code in ('48801 (00)', '49201 (00)', '61217 (00)', '91804 (00)')) then
      vid_pay = 'погашение ОД';

    if (dt_acc_add_code in ('20202 (00)') and
        kt_acc_add_code in ('49209 (20)')) then
      vid_pay = 'погашениие доп %%';

    if (dt_acc_add_code in ('20501 (00)', '47423 (99)', '47416 (00)', '20202 (00)', '60305 (10)') and
        kt_acc_add_code in ('48809 (10)', '49209 (10)', '91702 (00)')) then
      vid_pay = 'погашениие %%';

    if (dt_acc_add_code in ('20501 (00)', '47423 (99)', '47416 (00)', '20202 (00)', '60305 (10)') and
        kt_acc_add_code in ('60322 (10)')) then
      vid_pay = 'погашениие пени';

    if (dt_acc_add_code in ('20501 (00)', '47423 (99)', '20202 (00)') and
        kt_acc_add_code in ('60323 (20)')) then
      vid_pay = 'погашениие госпошлины';

    uid = '';
    uid = f_uuid_cbr(f_uuid('knclient', :kob_from_kc_id));
    suspend;
  end
end
