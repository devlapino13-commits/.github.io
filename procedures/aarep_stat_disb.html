CREATE PROCEDURE AAREP_STAT_DISB(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    ST_PERIOD_KIND INTEGER,
    ST_AMOUNT_KIND INTEGER,
    ST_VALUE DECIMAL(18,4),
    ST_LEVEL INTEGER)
AS
declare variable amount_kind_old integer;
declare variable cl_kolich integer;
declare variable disb decimal(18,4);
declare variable perc decimal(18,4);
declare variable n2 integer;
declare variable cl_t integer;
declare variable disb_t decimal(18,4);
declare variable perc_t decimal(18,4);
declare variable period_t integer;
declare variable st_perc decimal(18,4);
declare variable st_avg_perc_koef decimal(18,7);
declare variable st_disb decimal(18,4);
declare variable st_client_kolich integer;
begin
  amount_kind_old = 0;
  cl_kolich = 0;
  disb = 0;
  perc = 0;
  st_avg_perc_koef = 0;
  st_perc = 0;
  for select st_period_kind, st_amount_kind, sum(st_client_kolich), sum(st_disb), sum(st_perc)
      from aarep_stat_disb_temp(:report_region_id, :report_date_begin, :report_date_end)
      group by st_period_kind, st_amount_kind
      order by st_amount_kind, st_period_kind
      into :st_period_kind, :st_amount_kind, :st_client_kolich, :st_disb, :st_perc
  do
  begin
    if (st_disb <> 0) then
      st_avg_perc_koef = cast(st_perc / st_disb * 100 as decimal(15,1));
    else
      st_avg_perc_koef = 0;
    st_value = st_client_kolich;
    st_level = 1;
    suspend;
    st_value = st_disb;
    st_level = 2;
    suspend;
    st_value = st_avg_perc_koef;
    st_level = 3;
    suspend;
    if ((amount_kind_old = st_amount_kind) or (amount_kind_old = 0)) then
    begin
      cl_kolich = cl_kolich + st_client_kolich;
      disb = disb + st_disb;
      perc = perc + st_perc;
      amount_kind_old = st_amount_kind;
    end
    else
    begin
      n2 = st_amount_kind;
      st_amount_kind = amount_kind_old;
      period_t = st_period_kind;
      st_period_kind = 0;
      cl_t = st_client_kolich;
      st_client_kolich = cl_kolich;
      disb_t = st_disb;
      st_disb = disb;
      perc_t = st_perc;
      st_perc = perc;
      if (st_disb <> 0) then
        st_avg_perc_koef = cast(st_perc / st_disb * 100 as decimal(15,1));
      else
        st_avg_perc_koef = 0;
      st_value = st_client_kolich;
      st_level = 1;
      suspend;
      st_value = st_disb;
      st_level = 2;
      suspend;
      st_value = st_avg_perc_koef;
      st_level = 3;
      suspend;
      st_amount_kind = n2;
      st_period_kind = period_t;
      st_client_kolich = cl_t;
      st_disb = disb_t;
      st_perc = perc_t;

      cl_kolich = st_client_kolich;
      disb = st_disb;
      perc = st_perc;
      amount_kind_old = st_amount_kind;
    end
  end
  st_period_kind = 0;
  st_client_kolich = cl_kolich;
  st_disb = disb;
  st_perc = perc;
  if (st_disb <> 0) then
    st_avg_perc_koef = cast(st_perc / st_disb * 100 as decimal(15,1));
  else
    st_avg_perc_koef = 0;
  st_value = st_client_kolich;
  st_level = 1;
  suspend;
  st_value = st_disb;
  st_level = 2;
  suspend;
  st_value = st_avg_perc_koef;
  st_level = 3;
  suspend;
end
