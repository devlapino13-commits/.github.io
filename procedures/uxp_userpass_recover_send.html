CREATE PROCEDURE UXP$USERPASS_RECOVER_SEND(
    USER_NAME D_STR_CODE NOT NULL,
    SEND_TO D_STR_100,
    SEND_KIND D_STR_CODE NOT NULL)
RETURNS (
    PASS_CODE D_STR_10)
AS
declare variable un_id d_id;
declare variable n d_integer;
declare variable num d_integer;
declare variable msg_subject d_str_255;
declare variable msg_text d_str_long;
begin
  if (send_kind not in ('email', 'sms')) then
    exception e_default f_format('Unknown message_kind: %1', send_kind);

  select un_id
  from username
  where un_name = :user_name
  into un_id;

  if (un_id is null) then
    exception e_message f_format('user not found: %1', user_name);

  n = 0;
  pass_code = '';
  while (n < 4) do
  begin
    n = n + 1;
    num = round(rand() * 10.0000);
    if (num < 0) then
      num = 0;
    else
    if (num >= 10) then
      num = 9;

    pass_code = pass_code || num;
  end

  if (f_get_db_id() <> 94401) then
  begin
    msg_subject = 'Восстановление пароля.';
    msg_text = 'Ваш код для подтверждения: ' || pass_code;
  end else
  begin
    msg_subject = 'Parolun yenilənməsi.';
    msg_text = 'Tasdiq uchun kod: : ' || pass_code;
  end

  execute procedure hp_send_quick_message(:send_kind, :send_to, :msg_subject, :msg_text, 'send_pin', 7, :un_id);
  suspend;
end
