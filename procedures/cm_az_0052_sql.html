CREATE PROCEDURE CM_AZ_0052_SQL(
    MA_ID D_ID,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID ID_TYPE;
declare variable POINT D_INTEGER;
declare variable CLIENTS_CNT D_INTEGER;
declare variable MAC_AMOUNT CURR_TYPE;
declare variable MA_KOLICH_REPAY D_INTEGER;
declare variable MA_DATE_OPER D_DATE;
declare variable REG_NAME D_STR_100;
declare variable REG_ID D_INTEGER;
declare variable KF_ID D_ID;
declare variable PR_CODE D_STR_100;
declare variable KAL_NEW_KF_OFFICER D_ID;
begin
  res_error_is_critical = 2;

  select count(mac_from_c_id)
  from miniappclient
  where mac_from_ma_id = :ma_id and
        mac_amount > 0
  into clients_cnt;

  if (clients_cnt <> 1) then
  begin
    res_error_text = 'В заявке должен быть один клиент';
    suspend;
    exit;
  end

  select ma.ma_kolich_repay,
         ma.ma_date_oper,
         ma.ma_from_kf_id,
         pr.pr_code
  from miniapp ma
  inner join product pr on ma.ma_from_pr_id = pr.pr_id
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end = '31.12.9999'
  into ma_kolich_repay,
       ma_date_oper,
       kf_id,
       pr_code;

  if (ma_kolich_repay is null or ma_date_oper is null or kf_id is null) then
  begin
    res_error_text = 'Заявка заполнена не полностью (срок заявки, дата выдачи, папка заявки)';
    suspend;
    exit;
  end

  select kal.kal_new_reg_id,
         reg.reg_code || ' ' || reg.reg_name,
         kal_new_kf_officer
  from knaktlist kal
  inner join region reg on kal.kal_new_reg_id = reg.reg_unicode
  where kal.kal_from_kf_id = :kf_id and
        kal.is_delete = 0 and
        current_date between kal.kal_date and kal.kal_date_end
  into reg_id,
       reg_name,
       kal_new_kf_officer;

  select macl.mac_from_c_id,
         macl.mac_amount
  from miniappclient macl
  where macl.mac_from_ma_id = :ma_id and
        macl.mac_amount > 0
  into c_id,
       mac_amount;

  point = f_point(:c_id, :ma_date_oper);

  -- условия по периоду
  if (pr_code containing '.test_product.') then
  begin
    if (ma_kolich_repay > 36) then
      res_error_text = 'Максимальный срок займа составляет 36 месяца, клиент не может получить займ на больший срок';

    if (ma_kolich_repay > 24 and point < 24) then
      res_error_text = 'У клиента баллов меньше 24, поэтому cрок займа не может превышать 24 месяцев';

    if (ma_kolich_repay > 18 and point < 6) then
      res_error_text = 'У клиента баллов меньше 6, поэтому cрок займа не может превышать 18 месяцев';

    if (ma_kolich_repay > 18 and point < 3) then
      res_error_text = 'У клиента баллов меньше 3, поэтому cрок займа не может превышать 18 месяцев';
  end
  else
  begin
    if (ma_kolich_repay > 24) then
      res_error_text = 'Максимальный срок займа составляет 24 месяца, клиент не может получить займ на больший срок';

    if (ma_kolich_repay > 18 and point < 6) then
      res_error_text = 'У клиента баллов меньше 6, поэтому cрок займа не может превышать 18 месяцев';

    if (ma_kolich_repay > 18 and point < 3) then
      res_error_text = 'У клиента баллов меньше 3, поэтому cрок займа не может превышать 18 месяцев';
  end

  -- условия по суммам займа
  if (reg_id in (1040600, 1050800)) then -- RFCAZ-432
  begin
    if (mac_amount > 6500) then
      res_error_text = 'Максимальная сумма займа составляет 6500 AZN, клиент не может получить больше этой суммы';

    if (mac_amount > 6000 and point < 48) then
      res_error_text = 'У клиента баллов меньше 48, поэтому он не может получить сумму больше 6000 AZN';

    if (mac_amount > 5500 and point < 42) then
      res_error_text = 'У клиента баллов меньше 42, поэтому он не может получить сумму больше 5500 AZN';

    if (mac_amount > 5000 and point < 36) then
      res_error_text = 'У клиента баллов меньше 36, поэтому он не может получить сумму больше 5000 AZN';

    if (res_error_text is not null) then
      suspend;

    exit;
  end

  if (pr_code containing '.test_product.') then
  begin
    if (mac_amount > 10000) then
      res_error_text = 'Максимальная сумма займа составляет 10000 AZN, клиент не может получить больше этой суммы'; -- RFCAZ-392
  end
  else
  begin
    if (mac_amount > 8500) then -- RFCAZ-611
      res_error_text = 'Максимальная сумма займа составляет 8500 AZN, клиент не может получить больше этой суммы';

    if (mac_amount > 8000 and point < 48) then
      res_error_text = 'У клиента баллов меньше 48, поэтому он не может получить сумму больше 8000 AZN';

    if (mac_amount > 7500 and point < 42) then
      res_error_text = 'У клиента баллов меньше 42, поэтому он не может получить сумму больше 7500 AZN';

    if (mac_amount > 6500 and point < 36) then
      res_error_text = 'У клиента баллов меньше 36, поэтому он не может получить сумму больше 6500 AZN';

    if (mac_amount > 6000 and point < 30) then
      res_error_text = 'У клиента баллов меньше 30, поэтому он не может получить сумму больше 6000 AZN';
  end

  if (mac_amount > 6000 and point < 24) then
    res_error_text = 'У клиента баллов меньше 24, поэтому он не может получить сумму больше 6000 AZN';

  if (mac_amount > 5500 and point < 18) then
    res_error_text = 'У клиента баллов меньше 18, поэтому он не может получить сумму больше 5500 AZN';

  if (mac_amount > 5000 and point < 12) then
    res_error_text = 'У клиента баллов меньше 12, поэтому он не может получить сумму больше 5000 AZN';

  if (mac_amount > 4500 and point < 6) then
    res_error_text = 'У клиента баллов меньше 6, поэтому он не может получить сумму больше 4500 AZN';

  if (mac_amount > 4500 and point < 3) then
    res_error_text = 'У клиента баллов меньше 3, поэтому он не может получить сумму больше 4500 AZN';

--  if (not exists(select *
--                 from permissions
--                 inner join username on un_id = pm_object_id
--                 where current_date between pm_date_begin and pm_date_end and
--                       pm_is_delete = 0 and
--                       pm_access_kind = 1 and
--                       pm_object_kind = 0 and
--                       pm_subject_kind = 3 and
--                       pm_subject_id = f_get_pa_id('FinStart.Permission_Products') and -- СуперКС (RFCAZ-475)
--                       un_from_c_id = :kal_new_kf_officer)) then
--  begin
--    --RFCAZ: 162, 178, 191, 342, 354
--    if (mac_amount > 6500 and reg_id not in (1020400, 1030100, 1030200, 1030300, 1040200, 1040300, 1040500, 1050100, 1050200, 1050300,
--                                             1050400, 1050600, 1060200, 1060300, 1070100, 1080300, 1080400)) then
--      res_error_text = 'Офису ' || reg_name || ' разрешено выдавать займы до 6500 AZN, клиент не может получить больше этой суммы';
--  end

  if (res_error_text is not null) then
    suspend;
end
