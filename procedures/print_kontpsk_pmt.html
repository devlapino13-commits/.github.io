CREATE PROCEDURE PRINT_KONTPSK_PMT(
    KPS_ID D_ID)
RETURNS (
    PMT_NO D_INTEGER,
    OK_OL_DATE D_DATE,
    OK_OS D_CURRENCY,
    OK_PERC D_CURRENCY,
    OK_FINE D_CURRENCY,
    OK_TAX D_CURRENCY)
AS
declare variable k_id d_id;
declare variable ol_id d_id;
declare variable ol_date d_date;
begin
  select kps_from_k_id, ol_id, ol_date
  from kontpsk, operlist
  where kps_id = :kps_id and
        kps_from_ol_id = ol_id
  into k_id, ol_id, ol_date;

  pmt_no = 0;
  for select ok_ol_date, -sum(ok_os), -sum(ok_perc + ok_kommis), -sum(ok_fine), -sum(ok_tax)
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type = 4 and
        ok_ol_date <= :ol_date and
        ok_from_ol_id <= :ol_id
  group by ok_ol_date
  having sum(ok_os) <> 0 or sum(ok_perc + ok_kommis) <> 0 or sum(ok_fine) <> 0
  order by ok_ol_date
  into ok_ol_date, ok_os, ok_perc, ok_fine, ok_tax
  do
  begin
    pmt_no = pmt_no + 1;
    suspend;
  end
end
