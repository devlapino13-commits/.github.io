CREATE PROCEDURE HP_EDOC_INS_LOAN_PDN(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable kc_pdn_koeff d_percent;
declare variable k_status d_integer;
declare variable jk_system_code d_str_100;
declare variable pr_code d_str_100;
begin
  -- Создание документа "Уведомление о расчете ПДН"
  res_text = '';

  select k_status, pr_code
  from kontract, product
  where pr_id = k_from_pr_id
    and k_id = :k_id
  into k_status, pr_code;

  select jk_system_code
  from hr_pstatus h, joblist, jobkind, username
  where ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_status = 1 and
        ps_is_working = 1 and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id and
        ps_from_c_id = un_from_c_id and
        un_id = :un_id
  into :jk_system_code;

  if (jk_system_code is null or jk_system_code not in ('JobSystem.Region_Director', 'JobSystem.Region_KM', 'JobSystem.LO_Standart', 'JobSystem.FinStart_Support')) then
    if (not exists(
      select has_access
      from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.PdnSendAccess')
      where has_access = 1) and un_id <> 1000000000000) then
    res_text = 'Данная опция вам не доступна';

  if (res_text = '' and k_status <> 1) then
    res_text = 'Сначала утвердите договор займа';

  if (res_text = ''
    and pr_code not containing '.selfemployed.'
    and pr_code not containing '.business.') -- RFCRU-2301 -- Для продукта "Самозанятый" ПДН не рассчитывается
  then
  begin
    select kcp.kc_pdn_koef
    from knclient kc, knclient_pdn kcp
    where kc.kc_from_k_id = :k_id and
          kc.kc_amount_approved > 0 and
          kc.kc_id = kcp.kc_id and
          kcp_kind = 0
    into kc_pdn_koeff;

    if (coalesce(kc_pdn_koeff, 0) = 0) then
      res_text = 'ПДН еще не расчитан, обратитесь в техподдержку';
    else
    begin
      if (kc_pdn_koeff <= 50) then
        res_text = 'ПДН не превышает 50%, уведомлять не нужно'; -- RFCRU-1784
      else
        select edl_id, res_text
        from hp_edoc_insert('loan.pdn', f_objkind('kontract', 'id'), :k_id, :un_id)
        into edl_id, res_text;

    end
  end
  suspend;
end
