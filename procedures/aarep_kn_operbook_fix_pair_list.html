CREATE PROCEDURE AAREP_KN_OPERBOOK_FIX_PAIR_LIST(
    REP_DATE D_DATE)
RETURNS (
    ACC_ID D_ID,
    PAIR_ACC_ID D_ID,
    KC_ID D_ID,
    ACC_SALDO D_CURRENCY,
    PAIR_SALDO D_CURRENCY)
AS
begin
  for select acc_id,
             acc_pair_acc_id,
             acc_object_id,
             f_acc_bal(acc_id, :rep_date) acc_saldo,
             f_acc_bal(acc_pair_acc_id, :rep_date) as pair_saldo
  from ref_account
  where acc_from_ab_id in (99, 110) and
        acc_type = 0 and
        acc_object_kind = 8
  into acc_id, pair_acc_id, kc_id, acc_saldo, pair_saldo
  do
  begin
    if ((acc_saldo > 0 and pair_saldo = 0) or
        (acc_saldo = 0 and pair_saldo < 0) or
        (acc_saldo = 0 and pair_saldo = 0)) then
      Continue;

    suspend;
  end
end
