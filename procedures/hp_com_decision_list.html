CREATE PROCEDURE HP_COM_DECISION_LIST(
    CMK_ID D_INTEGER)
RETURNS (
    CMD_ID D_INTEGER,
    CMD_CODE D_STR_CODE,
    CMD_NAME D_STR_100,
    CMD_MEMO D_STR_LONG,
    CMD_INACTIVE D_BOOL,
    CMDK_SELECTED D_BOOL)
AS
begin
  for select cmd_id, cmd_code, cmd_name, cmd_memo, cmd_inactive
  from com_decision
  where cmd_id <> 0
  into :cmd_id, :cmd_code, :cmd_name, :cmd_memo, :cmd_inactive
  do
  begin
    cmdk_selected = 0;
    if (exists(select *
               from com_kind_dec
               where cmkd_from_cmk_id = :cmk_id and
                     cmkd_from_cmd_id = :cmd_id)) then
      cmdk_selected = 1;

    suspend;
  end
end
