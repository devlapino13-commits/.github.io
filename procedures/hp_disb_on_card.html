CREATE PROCEDURE HP_DISB_ON_CARD(
    K_ID ID_TYPE)
AS
declare variable ol_id id_type;
declare variable kc_id id_type;
declare variable c_id id_type;
declare variable amount_approved curr_type;
declare variable et_id id_type;
begin
  select ol.ol_id
  from kontract kont
  inner join operlist ol on ol.ol_id = kont.k_fact_disb_ol_id
  where kont.k_id = :k_id and
        ol.is_delete = 0
  into :ol_id;

  if (ol_id is null) then
    exit;

  update operlist ol
  set ol.is_delete = 1
  where ol.ol_id = :ol_id;

  for select kncl.kc_id, kncl.kc_from_c_id, kncl.kc_amount_approved
      from knclient kncl
      where kncl.kc_from_k_id = :k_id and
            kncl.kc_amount_approved > 0
      into :kc_id, :c_id, :amount_approved
  do
  begin
    select et.et_id
    from extsys_token et
    where et.et_from_c_id = :c_id and
          et.et_is_actual = 1
    into :et_id;

    insert into extsys_card_disb (ecd_from_ol_id, ecd_from_k_id, ecd_from_kc_id, ecd_from_et_id, ecd_amount)
    values (:ol_id, :k_id, :kc_id, :et_id, :amount_approved);
  end
end
