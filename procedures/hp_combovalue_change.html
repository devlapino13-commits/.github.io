CREATE PROCEDURE HP_COMBOVALUE_CHANGE(
    CF_TABLE VARCHAR(100),
    CF_FIELD VARCHAR(100),
    OLD_TEXT VARCHAR(255),
    NEW_TEXT VARCHAR(255))
AS
declare variable sql_text varchar(255);
declare variable cf_kind integer;
begin
  if (position('(', cf_table) = 0) then
  begin
    select cf_kind
    from combofields
    where cf_table_name = :cf_table and
          cf_field_name = :cf_field
    into :cf_kind;

    sql_text = 'update ' || :cf_table || ' set ' || :cf_field || '=''' || :new_text || '''  where ' || :cf_field || '=''' || :old_text || '''';
    execute statement :sql_text;

    if (cf_kind is not null) then
    begin
      delete from combolist
      where cl_kind = :cf_kind and
            cl_text = :old_text;
    end
  end
end
