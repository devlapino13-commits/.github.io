CREATE PROCEDURE HP_SEND_EDOC_PACK_RU(
    K_ID D_ID,
    KIND D_STR_CODE)
AS
declare variable un_id d_id;
declare variable res_text d_str_255;
declare variable db_id d_integer;
declare variable k_status d_integer;
declare variable ks_state d_integer;
declare variable edoc_status d_str_255;
declare variable pr_code d_str_100;
declare variable sh_date d_date;
declare variable k_disb_date d_date;
begin
  --RFCRU-2184
  db_id = f_get_db_id();

  if (db_id <> 81001) then
    exit;

  un_id = 1000000000000;
  res_text = '';

  select k.k_status, k.k_disb_date, pr.pr_code
  from kontract k
  inner join product pr on k.k_from_pr_id = pr.pr_id
  where k_id = :k_id
  into k_status, k_disb_date, pr_code;

  --Получаем дату выдачи из журнала
  select sh.sh_date
  from SCHEDULE sh
  where sh.sh_from_k_id = :k_id
  order by sh.sh_date
  rows 1
  into sh_date;

  if(k_disb_date <> sh_date) then
    res_text = 'Планируемая дата выдачи у договора не совпадает с журналом';

  if (k_status <> 1) then
    res_text = 'Сначала утвердите договор займа';

  if (res_text = '') then
  begin
    select ks_state
    from kontstate
    where ks_from_k_id = :k_id and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999'
    into ks_state;

    if (ks_state = 1) then
      res_text = 'Займ уже действующий';

    if (ks_state = 2) then
      res_text = 'Займ уже закрыт';
  end


    --До подписания ПДН
  if (res_text = '' and kind = 'loan_approve') then
  begin
    -- Проверяем статус ПДН
    edoc_status = f_edoc_check('loan.pdn', :k_id, current_date);
    
    if (edoc_status containing 'документ не сформирован' or edoc_status containing 'документ аннулирован') then
    begin
        --Отправляем ПДН
        select res_text
        from hp_edoc_ins_loan_pdn(:k_id, :un_id)
        into res_text;

         if (res_text = 'ПДН не превышает 50%, уведомлять не нужно' or pr_code similar to '%(.selfemployed.|.business.)%') then  -- RFCRU-2301 -- Для продукта "Самозанятый" и "Бизнес" ПДН не рассчитывается
         begin
             res_text = '';
             --Отправляем Договор займа
             select res_text
             from hp_edoc_ins_loan_agreement(:k_id, :un_id)
             into res_text;
         end
    end
  end

  --После подписания ПДН
  if (res_text = '' and kind = 'loan.agreement_send') then --после подписания ПДН
  begin
    -- Отправляем Договор займа
    res_text = (select res_text
                from hp_edoc_ins_loan_agreement(:k_id, :un_id));
  end
end
