CREATE PROCEDURE HP_MAILING_FP_ANK_UPDATE
AS
declare variable rd_anketa integer;
declare variable k_id id_type;
declare variable kf_id id_type;
declare variable c_id id_type;
declare variable i smallint;
declare variable svz_id id_type;
declare variable rep_date date;
declare variable reg_id integer;
begin
  rep_date = current_date;

  select rd_id
  from refdata
  where rd_code = 'Anketa.AnketaFTUpdate'
  into :rd_anketa;
  if (rd_anketa is null) then
    exit;

-- Беру активные займы, у которых срок 10 месяцев и более, которые были выданы 180 дней назад (~6 месяцев)
  for select kont.k_id, kont.k_from_kf_id
      from kontract kont
      inner join kontstate ks on ks.ks_from_k_id = kont.k_id
      where ks.ks_ol_is_delete = 0 and
            :rep_date - ks.ks_date_begin = 180 and
            ks.ks_state = 1 and
            kont.k_period >= 10 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end
      into :k_id, :kf_id
  do
  begin
    i = 0;

    for select kncl.kc_from_c_id
        from knclient kncl
        inner join clients cl on cl.c_id = kncl.kc_from_c_id
        where kncl.kc_from_k_id = :k_id and
              kncl.kc_amount_approved > 0 and
              cl.c_pft_risk_koef = 2 -- высокий риск
        into :c_id
    do
    begin
      if (i = 0) then
      begin
        i = 1;

        select kal.kal_new_kf_officer, kal.kal_new_reg_id
        from knaktlist kal
        where kal.kal_from_kf_id = :kf_id and
              kal.is_delete = 0 and
              :rep_date between kal.kal_date and kal.kal_date_end
        into :svz_id, :reg_id;
      end

-- Создаем задачу завтрашним днем
      insert into actionlist (acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                              acl_respons_c_id, acl_from_un_id)
      values (:rep_date + 1, :reg_id, 0, :c_id, :rd_anketa, 0, :svz_id, 1000000000001);

      execute procedure fsrobot_insert_event('DoMailing', 'MAILING_FP_ANK_UPDATE',
          'ok;' || k_id || ';' || c_id || ';' || svz_id, 1000000000001);
    end
  end
end
