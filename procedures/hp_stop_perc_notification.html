CREATE PROCEDURE HP_STOP_PERC_NOTIFICATION(
    K_ID D_ID,
    TERMINATION_DATE D_DATE)
AS
declare variable K_KONTRACT_NO D_STR_20;
declare variable K_FACT_DISB_DATE D_DATE;
declare variable K_STOP_PERC D_DATE;
declare variable TOTAL_SUM D_CURRENCY;
declare variable CM_NAME D_STR_255;
declare variable UN_ID D_ID;
declare variable KC_ID D_ID;
declare variable SMS_TEXT D_STR_LONG;
declare variable PHONE_NO D_STR_CODE;
declare variable C_ID D_ID;
declare variable MCL_PROJECT integer;
declare variable MSG_SENDER_CODE d_str_code;
begin
  select cm_name
  from company
  into cm_name;

  select k_kontract_no, k_fact_disb_date, k_stop_perc, ol_user_id
  from kontract, operlist
  where k_id = :k_id
  and ol_id = k_stopped_perc_ol_id
  into k_kontract_no, k_fact_disb_date, k_stop_perc, un_id;

  -- Определяем проект: М Булак или OiVO (RFCRU-4257)
  mcl_project = f_get_mcl_project(f_objkind('kontract','id'), :k_id);

  total_sum = null;
  select sum(zad_os + sh_calc_perc)
  from aarep_make_zad_list(:k_id, :k_stop_perc, 1)
  where sh_date is not null
  into total_sum;
  total_sum = coalesce(total_sum, 0);

  for select kc_from_c_id, kc_id
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into c_id, kc_id
  do
  begin
    select cont_value
    from hp_contdata_get_cell_main(0, :c_id)
    into phone_no;

    phone_no = trim(coalesce(phone_no, ''));
    if (phone_no <> '' and
        un_id = 1000000000001 and
        f_bankrupt(c_id, k_stop_perc) <> 1 and
        f_clientcom_allowed(c_id, 0, 0, 1) = 1) then
     begin
     --  Формируем текст с учетом проекта
      if (mcl_project = 2) then     -- OiVO  RFCRU-4257
      begin
        sms_text =
          'В связи с просрочкой 91 день по договору займа №' || k_kontract_no ||
          ' от ' || f_datetostr(k_fact_disb_date) || 'г. Сумма долга ' ||
          f_numtostr(total_sum, 2) || ' руб. дата возврата до ' ||
          f_datetostr(termination_date) || 'г. Тел. 88005559348. ' || cm_name;
      end
      else -- по умолчанию М Булак
      begin
        sms_text =
          'В связи с просрочкой 91 день по договору займа №' || k_kontract_no ||
          ' от ' || f_datetostr(k_fact_disb_date) || 'г. Сумма долга ' ||
          f_numtostr(total_sum, 2) || ' руб. дата возврата до ' ||
          f_datetostr(termination_date) || 'г. Тел. 88005557171. ' || cm_name;
      end

      msg_sender_code = f_enum_code('sms_sender', iif (:mcl_project = 2, 1, 0));

      f_send_sms(phone_no,  sms_text,  'knclient', kc_id, 'hp_stop_perc_notification', 1095, un_id, 0, msg_sender_code);
      f_send_push('knclient', kc_id, 'Просрочка 91 день', sms_text, 'hp_stop_perc_notification', 1095, un_id);
    end
  end
end
