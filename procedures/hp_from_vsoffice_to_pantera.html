CREATE PROCEDURE HP_FROM_VSOFFICE_TO_PANTERA(
    REP_DATE D_DATE)
AS
declare variable K_ID D_ID;
declare variable SBL_ID D_INTEGER;
declare variable MESS_STR STR_255;
begin
    select rd.rd_str_id
    from refdata rd
    where rd.rd_code = 'OKTELL.Projects.Pantera'
    into :sbl_id;

    for select oks.oks_from_k_id
    from oksummary oks
    where oks.oks_close_state = 0
    and cast(:rep_date as date) - oks.oks_delinq_date >= 7
    and cast(:rep_date as date) between oks.oks_date_begin and oks.oks_date_end
    and oks.oks_kc_id = 1000000000100 -- Виртуальный сотрудник (Офис)
    into :k_id
    do
    begin
        if (exists(
                select *
                from sb_akt sba
                where sba.sba_from_k_id = :k_id
                and :rep_date between sba.sba_date_begin and sba.sba_date_end
                and sba.sba_from_sbl_id = :sbl_id
                and sba.sba_status = 2)) then
            continue;

        insert into sb_akt (sba_from_k_id,  sba_date_begin, sba_from_sbl_id, sba_status, sba_from_un_id)
        values (:k_id,  :rep_date, :sbl_id, 2, 1000000000000);

        /*
        execute procedure okt_sirena_make_call(1, :rep_date, 'BLOCK', 'SirenaTransferReason.ToPanteraFromVSOffice', 'OKTELL.Projects.Pantera', 0, :k_id,
        0, null, null, null, null, null, null, null, null, 0)
        returning_values :mess_str;
        */
    end
end
