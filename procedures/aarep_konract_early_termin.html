CREATE PROCEDURE AAREP_KONRACT_EARLY_TERMIN(
    REP_DATE DATE,
    REP_REGION_ID ID_TYPE)
RETURNS (
    K_KONTRACT_NO VARCHAR(100),
    STATUS VARCHAR(40),
    STOP_PERC_DATE DATE)
AS
declare variable ks_from_k_id decimal(18,0);
declare variable k_stop_perc date;
declare variable ol_id decimal(18,0);
declare variable oks_delinq_date date;
begin
  for select ks_from_k_id, k_kontract_no, k_stop_perc, k_stopped_perc_ol_id, oks_delinq_date
      from kontstate, kontract, knaktlist, region_list(:rep_region_id, 1, -1, 1), oksummary
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date between kal_date and kal_date_end and
            (kal_new_reg_id = id_list) and
            k_id = oks_from_k_id and
            :rep_date between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into ks_from_k_id, k_kontract_no, k_stop_perc, ol_id, oks_delinq_date
  do
  begin
    oks_delinq_date = iif(oks_delinq_date = '31.12.9999', null, oks_delinq_date);
    if (k_stop_perc is not null) then
      status = 'Вручную';
    if ((k_stop_perc is null) and
        (ol_id is null) and
        (cast(:rep_date as date) - oks_delinq_date) > 60 or (oks_delinq_date is null)) then
      status = 'Автомат';
    if ((k_stop_perc is null) and
        (ol_id is null) and
        ((cast(:rep_date as date) - oks_delinq_date) between -100000 and 59)) then
      status = 'Активен';
    begin
      stop_perc_date = iif(status = 'Автомат', iif(oks_delinq_date is null, k_stop_perc, oks_delinq_date + 60), k_stop_perc);
    end
    suspend;
  end
end
