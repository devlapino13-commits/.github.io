CREATE PROCEDURE FM_ANKETA_DETAIL(
    UN_ID DECIMAL(18,0),
    ACL_ID DECIMAL(18,0))
RETURNS (
    QUE_ID INTEGER,
    ANS_ID INTEGER,
    STATE INTEGER,
    ED_TEXT VARCHAR(255),
    ED_KIND INTEGER,
    TITLE VARCHAR(255),
    DETAIL VARCHAR(255),
    MRU VARCHAR(5000),
    IS_NA INTEGER)
AS
begin
  for select res_question_id, -- id вопроса
             res_rd_id, -- id ответа
             res_state, -- Статус (-1, 0, 1)
             res_str_value, -- Текст редактора
             rd_editor_kind, -- Тип редактора
--   Вопросы: 2-флажки, 3-радиокнопки
--   Ответы:  0-обычный, 1-с редактором
             rd_name, -- Текст вопроса/ответа
             rd_description, -- Описание вопроса
             rd_mru_list, -- Список подсказок
             case rd_code
               when 'Results.NoAnswer' then 1
               when 'Results.None' then 2
               else 0
             end -- Системная строка "Нет ответа"
      from hp_actions_short_detail(:un_id, :acl_id)
      into que_id, ans_id, state, ed_text, ed_kind, title, detail, mru, is_na
  do
  begin
-- Тестовое заполнение MRU-списка
--    if (title = 'Таджикистан') then
--    begin
--      mru = 'Москва; Баку; Каракол; Наманган; Бишкек; Санкт Петербург; Ош; Нарын';
--    end
    suspend;
  end
end
