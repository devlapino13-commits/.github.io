CREATE PROCEDURE CM_RU_0183_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable PR_NAME D_STR_100;
declare variable KC_FROM_C_ID D_ID;
declare variable JL_REG_ID D_ID;
declare variable C_CITIZEN D_ID;
declare variable EMP_CITIZEN_TXT D_STR_100;
declare variable JL_NAME D_STR_100;
declare variable C_STAGE_MONTH D_CURRENCY;
declare variable PAR30 D_PERCENT;
declare variable K_PERCENT D_PERCENT;
declare variable K_KOLICH_REPAY D_INTEGER;
declare variable LAST_K_ID D_ID;
declare variable KS_DATE_END D_DATE;
declare variable MAX_DELINQ_DAY D_INTEGER;
begin

  if (ma_id in (1502641)) then exit;

  res_error_is_critical = 1;
  for select ma.ma_date_oper, ma.ma_percent, ma.ma_kolich_repay,
             mac.mac_amount, mac.mac_from_c_id,
             c_name_fio, c.c_citizen, pr.pr_name
      from miniapp ma
      inner join product pr on ma.ma_from_pr_id = pr_id and
            pr_code containing '.EMP.' and
            pr_code containing '.reliable_emp.' and
            pr_code containing '.IND.'
      inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id and
            mac.mac_amount > 0
      inner join clients c on mac.mac_from_c_id = c_id
      where ma.ma_id = :ma_id and
            ma.ma_date_end = '31.12.9999'
      into kf_disb_date, k_percent, k_kolich_repay,
           kc_amount, kc_from_c_id,
           res_info_text, c_citizen, pr_name
  do
  begin
    if (k_kolich_repay not between 3 and 12) then
    begin
      res_error_text = 'Срок займа: от 3 до 12 месяцев.';
      res_error_is_critical = 1;
      suspend;
    end
    jl_name = null;
    select jl.jl_name, hp.ps_from_reg_id
    from hr_pstatus hp, joblist jl
    where :kf_disb_date between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          ps_from_c_id = :kc_from_c_id and
          ps_from_jl_id = jl_id
    into :jl_name, :jl_reg_id;
    jl_name = coalesce(jl_name, '');
    if (jl_name not in ('Администратор', 'Администратор офиса',
                    'Администратор по выдаче займов', 'Администратор-консультант',
                    'Ведущий консультант по выдаче займов', 'Ведущий менеджер',
                    'Ведущий менеджер по выдаче займов', 'Ведущий специалист',
                    'Ведущий специалист по выдаче займов', 'Главный специалист по выдаче займов',
                        'Заместитель менеджера офиса',
                        'И. о. ведущий специалист по выдаче займов',
                        'И.о. заместителя менеджера офиса', 'И.о. менеджера офиса',
                        'И.О. Старшего кассира-оператора', 'И.о. старшего менеджера по выдаче займов',
                        'И.о. Старшего специалиста по выдаче займов', 'Кассир',
                        'Кассир-оператор', 'Консультант',
                        'Консультант по выдаче займов', 'Кредитный менеджер',
                        'Кредитный сотрудник', 'менеджер',
                        'Менеджер офиса', 'Менеджер по выдаче займов',
                        'Менеджер по кредитованию', 'Менеджер-администратор',
                        'Менеджер-консультант', 'Подсобный рабочий',
                        'Специалист', 'Специалист по выдаче займов', 'Старший кассир-оператор',
                        'Старший менеджер по выдаче займов',
                        'Старший специалист по выдаче займов',
                        'Специалист по продажам')) then
    begin
      res_error_text = 'Должность сотрудника не включен в список разрещенных для данного займа';
      res_error_is_critical = 1;
      suspend;
    end
    c_stage_month = 0;
    select emp_stage_month
    from hp_get_employer_stage(:kc_from_c_id, :kf_disb_date)
    into c_stage_month;
    if (coalesce(c_stage_month, '01.01.1900') = '01.01.1900') then
    begin
      res_error_text = 'Стаж не определен, обратитесь к тех.поддержке БД';
      res_error_is_critical = 1;
      suspend;
    end
    else
    begin
      if (c_stage_month < 18.0) then
      begin
        res_error_text = 'Стаж сотрудника менее 18-месяца';
        res_error_is_critical = 1;
        suspend;
      end
      if (c_stage_month < 24) then
      begin
        if (kc_amount > 80000.0) then
        begin
          res_error_text = 'Стаж сотрудника менее 24-месяца, разр.макс.сумма 80 000 р.';
          res_error_is_critical = 1;
          suspend;
        end
        if (k_percent <> 3.5) then
        begin
          res_error_text = 'Стаж сотрудника менее 24-месяца, %%-ставка должно быть 3.5%';
          res_error_is_critical = 0;
          suspend;
        end
      end

      if (c_stage_month >= 24.0 and c_stage_month < 36.0) then
      begin
        if (kc_amount > 100000.0) then
        begin
          res_error_text = 'Стаж сотрудника ' || c_stage_month || '-месяца, разр.макс.сумма 100 000 р.';
          res_error_is_critical = 1;
          suspend;
        end
        if (k_percent <> 3.0) then
        begin
          res_error_text = 'Стаж сотрудника ' || c_stage_month || '-месяца, %%-ставка должно быть 3.0%';
          res_error_is_critical = 0;
          suspend;
        end
      end
      if (c_stage_month >= 36.0) then
      begin
        if (kc_amount > 120000.0) then
        begin
          res_error_text = 'Разр.макс.сумма 120 000 р.';
          res_error_is_critical = 1;
          suspend;
        end
        if (k_percent <> 2.5) then
        begin
          res_error_text = '%%-ставка должно быть 2.5%';
          res_error_is_critical = 0;
          suspend;
        end
      end
      emp_citizen_txt = null;
      select left(rd_name, 100)
      from refdata rd
      where rd.rd_id = :c_citizen
      into emp_citizen_txt;
      if (emp_citizen_txt not in ('Кыргызстан', 'Узбекистан', 'Таджикистан')) then
      begin
        res_error_text = 'По продукту ' || pr_name || ' могут получить только граждане КГ, УЗ и ТЖ. Гражданство сотрудника: ' || emp_citizen_txt;
        res_error_is_critical = 1;
        suspend;
      end
      if (exists(select *
                 from rebuke
                 where rb_from_c_id = :kc_from_c_id and
                       rb_bonus_change = 1 and
                       rb_period_start >= dateadd(month, -12, :kf_disb_date) and
                       rb_is_delete = 0 and
                       rb_type = 1)) then
      begin
        res_error_text = 'У сотрудника есть выговор';
        res_error_is_critical = 1;
        suspend;
      end
      par30 = null;
      if (jl_name not containing 'менед') then
      begin
        select sum(iif(cast(:kf_disb_date as date) - oks_delinq_date > 30, oks_balance, 0)) /
               sum(oks_balance)
        from oksummary,kontract
        where oks_region_id = :jl_reg_id and
        cast(:kf_disb_date as date) between oks_date_begin and oks_date_end and
        oks_balance > 0 and
        cast(:kf_disb_date as date) - oks_delinq_date <= 360 and
        oks_from_k_id = k_id and
        oks_close_state = 0 and
        oks_kc_id = k_fact_disb_lo_id
        and oks_kc_id = :kc_from_c_id
        into par30;
      end
      else
      begin
        select sum(iif(cast(:kf_disb_date as date) - oks_delinq_date > 30, oks_balance, 0)) /
               sum(oks_balance)
        from oksummary,kontract
        where oks_region_id = :jl_reg_id and
        cast(:kf_disb_date as date) between oks_date_begin and oks_date_end and
        oks_balance > 0 and
        cast(:kf_disb_date as date) - oks_delinq_date <= 360 and
        oks_from_k_id = k_id and
        oks_close_state = 0 and
        oks_kc_id = k_fact_disb_lo_id
        into par30;
      end
      par30 = par30 * 100;
      if (par30 >= 7.0) then
      begin
        res_error_text = 'PAR30 сотрудника более 7%';
        res_error_is_critical = 1;
        suspend;
      end
    end
    -- находим предыдущий цикл сотрудника по надежному сотруднику
    last_k_id = null;
    ks_date_end = null;
    select k_id, ks_date_end from knclient, kontract, product, kontstate
    where kc_from_c_id = :kc_from_c_id
    and kc_amount_approved > 0
    and kc_from_k_id = k_id
    and k_from_pr_id = pr_id
    and pr_code containing '.reliable_emp.'
    and k_id = ks_from_k_id
    and ks_ol_is_delete = 0
    and ks_state = 1
    order by k_disb_date desc
    rows 1
    into last_k_id, ks_date_end;

    if (last_k_id is not null) then
    begin
      if (ks_date_end = '31.12.9999') then
      begin
        res_error_text = 'Имеется активный кредит по продукту "Надежный сотрудник"!';
        res_error_is_critical = 1;
        suspend;
      end else
      begin
        ks_date_end = ks_date_end + 1; -- дата закрытие предыдущего цикла

        -- находим максимальную просрочку
        max_delinq_day = null;
        select max(ok_ol_date - ok_sh_date) from operkred
        where ok_ol_is_active = 1
        and ok_from_k_id = :last_k_id
        and ok_type in (4,5)
        and ok_os + ok_perc < 0
        into max_delinq_day;
        if (max_delinq_day is null) then max_delinq_day = 0;

        -- Если у сотрудника была непрерывная просрочка 30 и более дней по предыдущему займу по продукту «Надежный сотрудник»,
        -- то доступ к новому займу по данному продукту возобновляется по истечении одного года с даты полного погашения предыдущего
        -- займа по продукту «Надежный сотрудник».
        if (dateadd(year, 1, ks_date_end) >= kf_disb_date and max_delinq_day >= 30) then
        begin
          res_error_text = 'В предыдущем цикле имеется просрочка более 30 дней!';
          res_error_is_critical = 1;
          suspend;
        end
      end 
    end
  end
end
