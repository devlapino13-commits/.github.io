CREATE PROCEDURE AAREP_PLANFACT_TEMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_AUTO_PERIOD INTEGER,
    REP_AUTO_KOEF DECIMAL(18,7),
    REP_INCLUDE_FIRST_DAY INTEGER)
RETURNS (
    MA_DATE_OPER DATE,
    MA_SUMMA_PLUS_OS DECIMAL(18,4),
    MA_SUMMA_PLUS_PERC DECIMAL(18,4),
    MA_SUMMA_PLUS_ADMIN DECIMAL(18,4),
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_SUMMA_MINUS_REPEAT DECIMAL(18,4),
    MA_SUMMA_MINUS_NEW DECIMAL(18,4),
    MA_SUMMA_MINIAPPNEW DECIMAL(18,4),
    MA_ADMIN_MINUS_PROCESS DECIMAL(18,4),
    MA_REGION_ID INTEGER,
    MA_FACT_OS DECIMAL(18,4),
    MA_FACT_PERC DECIMAL(18,4),
    MA_FACT_ADMIN DECIMAL(18,4),
    MA_FACT_FINE DECIMAL(18,4),
    MA_FACT_DISB_REPEAT DECIMAL(18,4),
    MA_FACT_DISB_NEW DECIMAL(18,4),
    MA_DELIQ_OS DECIMAL(18,4),
    MA_DELIQ_PERC DECIMAL(18,4),
    MA_ADV_OS DECIMAL(18,4),
    MA_ADV_PERC DECIMAL(18,4),
    MA_TOTAL_OS DECIMAL(18,4),
    MA_TOTAL_PERC DECIMAL(18,4))
AS
declare variable ma_circle integer;
declare variable sh_date date;
begin
  /* В основном запросе надо исключить предыдущий день */
  ma_summa_minus_repeat = 0;
  ma_summa_minus_new = 0;
  ma_summa_plus_os = 0;
  ma_summa_plus_perc = 0;
  ma_summa_plus_admin = 0;
  ma_summa_miniappnew = 0;
  ma_admin_minus_process = 0;
  ma_summa_minus = 0;
  ma_fact_os = 0;
  ma_fact_perc = 0;
  ma_fact_admin = 0;
  ma_fact_fine = 0;
  ma_fact_disb_repeat = 0;
  ma_fact_disb_new = 0;
  ma_deliq_os = 0;
  ma_deliq_perc = 0;
  ma_adv_os = 0;
  ma_adv_perc = 0;
  ma_total_os = 0;
  ma_total_perc = 0;
  for select id_list
      from region_orig_list(:rep_region_id, 1, 3, 1)
      order by id_list
      into :ma_region_id
  do
  begin
    ma_date_oper = :rep_date_begin;
    while (ma_date_oper <= :rep_date_end) do
    begin
      suspend;
      ma_date_oper = ma_date_oper + 1;
    end
  end

  /* Прогнозирование по кредитной части */

  for select ma_region_id, ma_date_oper, sum(ma_summa_plus_os), sum(ma_summa_plus_perc), sum(ma_summa_plus_admin),
             sum(ma_summa_minus), sum(ma_summa_minus_repeat), sum(ma_summa_minus_new), sum(ma_summa_miniappnew),
             sum(ma_admin_minus_process)
      from aarep_miniapp_temp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_auto_period, :rep_auto_koef, :rep_include_first_day, 1)
      group by ma_region_id, ma_date_oper
      order by ma_date_oper
      into :ma_region_id, :ma_date_oper, :ma_summa_plus_os, :ma_summa_plus_perc, :ma_summa_plus_admin, :ma_summa_minus,
           :ma_summa_minus_repeat, :ma_summa_minus_new, :ma_summa_miniappnew, :ma_admin_minus_process
  do
  begin
    /*
    if (ma_summa_minus_new>=ma_summa_miniappnew)
    then ma_summa_miniappnew=0;
    else ma_summa_miniappnew=ma_summa_miniappnew-ma_summa_minus_new;
    */
    suspend;
  end

  ma_summa_minus_repeat = 0;
  ma_summa_minus_new = 0;
  ma_summa_plus_os = 0;
  ma_summa_plus_perc = 0;
  ma_summa_plus_admin = 0;
  ma_summa_miniappnew = 0;
  ma_admin_minus_process = 0;
  ma_summa_minus = 0;
  ma_fact_os = 0;
  ma_fact_perc = 0;
  ma_fact_admin = 0;
  ma_fact_fine = 0;
  ma_fact_disb_repeat = 0;
  ma_fact_disb_new = 0;
  ma_deliq_os = 0;
  ma_deliq_perc = 0;
  ma_adv_os = 0;
  ma_adv_perc = 0;
  ma_total_os = 0;
  ma_total_perc = 0;

  /* Результаты фактических операций */

  for select ok_ol_date, ok_ol_reg_id, cast(1 as integer), ok_sh_date, sum(ok_os * cr_rate), cast(0 as decimal(18,4)),
             cast(0 as decimal(18,4)), cast(0 as decimal(18,4)), cast(0 as decimal(18,4))
      from kontract, opercurr, operkred, region_list(:rep_region_id, 1, -1, 1)
      where ok_ol_is_active = 1 and
            ok_type = 8 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_ol_reg_id = id_list and
            ok_ol_date between cr_date and cr_date_end and
            ok_from_k_id = k_id and
            cr_from_cu_id = k_index_curr_code and
            (k_circle + k_circle_internal) = 2 and
            ok_from_c_id = 0
      group by ok_ol_date, ok_ol_reg_id, ok_sh_date
      union
      select ok_ol_date, ok_ol_reg_id, cast(2 as integer), ok_sh_date, sum(ok_os * cr_rate), cast(0 as decimal(18,4)),
             cast(0 as decimal(18,4)), cast(0 as decimal(18,4)), cast(0 as decimal(18,4))
      from kontract, opercurr, operkred, region_list(:rep_region_id, 1, -1, 1)
      where ok_ol_is_active = 1 and
            ok_type = 8 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_ol_reg_id = id_list and
            ok_ol_date between cr_date and cr_date_end and
            ok_from_k_id = k_id and
            cr_from_cu_id = k_index_curr_code and
            (k_circle + k_circle_internal) > 2 and
            ok_from_c_id = 0
      group by ok_ol_date, ok_ol_reg_id, ok_sh_date
      union
      select ok_ol_date, ok_ol_reg_id, cast(0 as integer), ok_sh_date, cast(0 as decimal(18,4)), sum(ok_os * cr_rate),
             sum(ok_perc * cr_rate), sum(ok_fine * cr_rate), sum(ok_kommis * cr_rate)
      from kontract, opercurr, operkred, region_list(:rep_region_id, 1, -1, 1)
      where ok_ol_is_active = 1 and
            ok_type = 4 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_ol_reg_id = id_list and
            ok_ol_date between cr_date and cr_date_end and
            ok_from_k_id = k_id and
            cr_from_cu_id = k_index_curr_code and
            ok_from_c_id = 0 and
            ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
      group by ok_ol_date, ok_ol_reg_id, ok_sh_date
      into :ma_date_oper, :ma_region_id, :ma_circle, :sh_date, :ma_fact_disb_new, :ma_fact_os, :ma_fact_perc,
           :ma_fact_fine, :ma_fact_admin
  do
  begin
    ma_deliq_os = 0;
    ma_deliq_perc = 0;
    ma_adv_os = 0;
    ma_adv_perc = 0;
    ma_fact_disb_repeat = 0;
    ma_total_os = 0;
    ma_total_perc = 0;
    if (ma_circle = 2) then
    begin
      ma_fact_disb_repeat = :ma_fact_disb_new;
      ma_fact_disb_new = 0;
    end
    if (ma_circle = 0) then
    begin
      if (ma_date_oper > :sh_date) then
      begin
        ma_deliq_os = :ma_fact_os;
        ma_deliq_perc = :ma_fact_perc;
        ma_fact_os = 0;
        ma_fact_perc = 0;
      end
      if (ma_date_oper < :sh_date) then
      begin
        ma_adv_os = :ma_fact_os;
        ma_adv_perc = :ma_fact_perc;
        ma_fact_os = 0;
        ma_fact_perc = 0;
      end
      ma_total_os = :ma_fact_os + :ma_adv_os + :ma_deliq_os;
      ma_total_perc = :ma_fact_perc + :ma_adv_perc + :ma_deliq_perc;
    end

    suspend;
  end
end
