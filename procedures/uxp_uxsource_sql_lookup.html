CREATE PROCEDURE UXP$UXSOURCE_SQL_LOOKUP(
    US_UUID D_UUID)
RETURNS (
    LOOKUP_SQL D_STR_LONG,
    ERROR_TEXT D_STR_255)
AS
declare variable us_code d_str_code;
declare variable us_lookup_id d_str_code;
declare variable us_lookup_text d_str_255;
declare variable us_is_table d_bool;
declare variable main_prefix d_str_10;
declare variable pos d_integer;
declare variable sql_text d_str_long;
declare variable usl_fk_index d_str_code;
declare variable param_list d_str_long;
declare variable enter_sym d_str_lang;
declare variable param_name d_str_code;
begin
  enter_sym = ascii_char(13) || ascii_char(10);

  lookup_sql = '';
  usl_fk_index = '';
  error_text = '';

  select us_uuid, us_code, us_is_table, ufl_lookup_id, ufl_lookup_text
  from ux$frlookup, ux$source
  where ufl_ufp_uuid = :us_uuid and
        us_uuid = ufl_us_uuid
  into us_uuid, us_code, us_is_table, us_lookup_id, us_lookup_text;

  if (row_count = 0) then
  begin
    select us_code, us_is_table, us_lookup_id, us_lookup_text
    from ux$source
    where us_uuid = :us_uuid
    into us_code, us_is_table, us_lookup_id, us_lookup_text;
  end

  if (coalesce(us_lookup_text, '') = '' and us_is_table = 1) then
  begin
    error_text = 'Поле или выражение не указано.';
    suspend;
    exit;
  end

  if (coalesce(us_lookup_id, '') = '') then
    us_lookup_id = us_lookup_text;

  sql_text = '';
  main_prefix = '';

  if (us_is_table = 0) then
  begin
    param_list = '';

    for select uf_field_name
    from ux$field
    where uf_owner = :us_uuid and
          uf_is_param = 1
    order by uf_position
    into param_name
    do
    begin
      if (param_list <> '') then
        param_list = param_list || ', ';

      param_list = param_list || ':' || param_name;
    end
  end else
  begin
    pos = position('_', us_lookup_id);
    if (pos > 0) then
      main_prefix = left(us_lookup_id, pos - 1);
    else
      main_prefix = left(us_code, 3);
  end

  if (usl_fk_index = '') then
    main_prefix = '';

  if (us_lookup_id <> us_lookup_text) then
  begin
    if (main_prefix <> '') then
      us_lookup_id = main_prefix || '.' || us_lookup_id;
  
    lookup_sql = 'select ' || us_lookup_id || ' as id, ' || us_lookup_text || ' as name '
                           || enter_sym || ' from ' || us_code
                           || iif(param_list <> '', '(' || param_list || ')', '')
                           || ' ' || main_prefix
                           || iif(sql_text <> '', enter_sym || sql_text, '');
  end else
  begin
    lookup_sql = 'select ' || us_lookup_text || ' as name '
                           || enter_sym || ' from ' || us_code
                           || iif(param_list <> '', '(' || param_list || ')', '')
                           || ' ' || main_prefix
                           || iif(sql_text <> '', enter_sym || sql_text, '');
  end

  lookup_sql = replace(lookup_sql, '  ', ' ');

  suspend;
end
