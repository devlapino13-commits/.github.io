CREATE PROCEDURE HP_CHECK_KONTRACT(
    REP_K_ID DECIMAL(18,0),
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    CHECK_MEMO VARCHAR(255),
    OK_BALANCE CURR_TYPE,
    OB_BALANCE CURR_TYPE,
    OK_OL_DATE DATE,
    OK_FROM_OL_ID ID_TYPE)
AS
declare variable ok_type integer;
declare variable ok_os curr_type;
declare variable ok_perc curr_type;
declare variable ok_fine curr_type;
declare variable ok_kommis curr_type;
declare variable ok_sh_os curr_type;
declare variable acc_balance varchar(20);
declare variable acc_perc_dohod varchar(20);
declare variable acc_percent varchar(20);
declare variable acc_fine varchar(20);
declare variable acc_kommis varchar(20);
declare variable acc_pereplata varchar(20);
declare variable acc_llc_os varchar(20);
declare variable acc_llc_percent varchar(20);
declare variable acc_kommis_dosroch varchar(20);
declare variable acc_balance_old varchar(20);
declare variable acc_perc_dohod_old varchar(20);
declare variable acc_percent_old varchar(20);
declare variable acc_fine_old varchar(20);
declare variable acc_kommis_old varchar(20);
declare variable acc_pereplata_old varchar(20);
declare variable acc_llc_os_old varchar(20);
declare variable acc_llc_percent_old varchar(20);
declare variable acc_llc_percent_old_1 varchar(20);
declare variable acc_kommis_dosroch_old varchar(20);
declare variable ob_summa_val curr_type;
declare variable at_id integer;
declare variable pal_main_acc varchar(20);
declare variable pal_delinq_acc varchar(20);
declare variable at_name varchar(50);
declare variable ok_type_2 decimal(18,4);
declare variable ok_type_3 decimal(18,4);
declare variable sh_kommis curr_type;
declare variable ok_kommis_total curr_type;
declare variable k_disb_date date;
declare variable acc_kommis_1_old varchar(20);
declare variable ok_kommis_adv curr_type;
declare variable k_kolich_client integer;
declare variable kc_kolich_client integer;
declare variable kd_k_new decimal(18,0);
declare variable k_kontract_no_buf varchar(50);
declare variable ok_balance_buf decimal(18,4);
declare variable ks_state_buf integer;
declare variable ok_sum_buf decimal(18,4);
declare variable ok_type_buf integer;
declare variable sh_os decimal(18,4);
declare variable k_amount_approved decimal(18,4);
declare variable cm_database_id integer;
declare variable k_percent_for_days integer;
declare variable pr_percent_for_days integer;
declare variable p_sh_id integer;
declare variable p_sh_date date;
declare variable p_sh_os curr_type;
declare variable p_sh_perc curr_type;
declare variable ok_sh_perc curr_type;
begin
  :cm_database_id = f_get_db_id();

  -- ориентироватся по таблице ACCTYPE, PRODACCTYPE
  -- Счета по новому стандарту
  for select at_id, pal_main_acc || '/' || pal_main_acc_no, at_name, k_kontract_no, k_disb_date,
             k_kolich_client_approved, pal_delinq_acc || '/' || pal_delinq_acc_no
      from kontract, product, prodacclist, acctype
      where k_id = :rep_k_id and
            k_from_pr_id = pr_id and
            pr_from_pat_id = pal_from_pat_id and
            pal_from_at_id = at_id and
            at_kind = 1 and -- выборка только по кредитным счетам
            pal_main_acc is not null
      into at_id, pal_main_acc, at_name, k_kontract_no, k_disb_date, k_kolich_client, pal_delinq_acc
  do
  begin
    if (at_id = 1) then
      acc_balance = :pal_main_acc; -- Портфель
    if (at_id = 2) then
      acc_perc_dohod = :pal_main_acc; -- Начисленные проценты
    if (at_id = 3) then
      acc_percent = :pal_main_acc; -- Полученные проценты
    if (at_id = 5) then
      acc_fine = :pal_main_acc; -- Полученные штрафы
    if (at_id = 7) then
      acc_kommis = :pal_main_acc; -- Полученные комиссионные (ДБП)
    if (at_id = 8) then
      acc_pereplata = :pal_main_acc; -- Счета переплаты по ОС
    if (at_id = 10) then
      acc_llc_os = :pal_main_acc; -- Счета РППУ для списание ОС
    if (at_id = 15) then
      acc_llc_percent = :pal_main_acc; -- Счета РППУ для списание Процентов
    if (at_id = 26) then
      acc_kommis_dosroch = :pal_main_acc; -- Комиссия за досрочное погашениеС
  end

  -- Счета по старому стандарту
  for select sa_account_type, al_account
      from kontract, standacc, acc_list, knaktlist
      where k_id = :rep_k_id and
            sa_from_pr_id = k_from_pr_id and
            sa_account = al_id and
            knaktlist.is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            sa_from_reg_id = kal_new_reg_id and
            '31.12.2013' between kal_date and kal_date_end -- счета на указанную дату, нужно ставить дату операции который корректируешь
      order by kal_date, kal_new_reg_id, sa_account_type
      into at_id, pal_main_acc
  do
  begin
    if (at_id = 1) then
      acc_balance_old = :pal_main_acc; -- Портфель
    if (at_id = 2) then
      acc_perc_dohod_old = :pal_main_acc; -- Начисленные проценты
    if (at_id = 3) then
      acc_percent_old = :pal_main_acc; -- Полученные проценты
    if (at_id = 5) then
      acc_fine_old = :pal_main_acc; -- Полученные штрафы
    if (at_id = 6) then
      acc_kommis_1_old = :pal_main_acc; -- Начисленные комиссионные (Доход)
    if (at_id = 7) then
      acc_kommis_old = :pal_main_acc; -- Полученные комиссионные (ДБП)
    if (at_id = 8) then
      acc_pereplata_old = :pal_main_acc; -- Счета переплаты по ОС
    if (at_id = 10) then
      acc_llc_os_old = :pal_main_acc; -- Счета РППУ для списание ОС
    if (at_id = 15) then
      acc_llc_percent_old = :pal_main_acc; -- Счета РППУ для списание Процентов
    if (at_id = 36) then
      acc_llc_percent_old_1 = :pal_main_acc; -- Счета РППУ для списание Процентов
    if (at_id = 26) then
      acc_kommis_dosroch_old = :pal_main_acc; -- Комиссия за досрочное погашениеС
  end

  ok_type_2 = 0;
  ok_type_3 = 0;
  ok_balance = 0;
  ok_kommis_total = 0;
  for with tabl1
      as (select kd_k_new
          from kontdevides
          where kd_k_orig = :rep_k_id
          union
          select :rep_k_id
          from rdb$database)
      select ok_from_ol_id, ok_type, ok_ol_date, sum(ok_os), sum(ok_perc), sum(ok_fine), sum(ok_kommis), sum(ok_sh_os), sum(ok_sh_perc),
             sum(ok_kommis_adv)
      from operkred ok, tabl1, operlist ol
      where ok_ol_is_active = 1 and
            ok_from_c_id = 0 and
            ok_type in (2, 3, 4, 5, 8, 9) and
            ((:cm_database_id in (39801, 41701) and -- KG
            ok_from_ol_id not in (1090100001421255, 1090100001421275, 1090100002275788, 1090100002278992, 1090100002279002, 1090101327497, 1090100011535270)) -- KG
            or
            (:cm_database_id = 81001 and -- RU
            ok_from_ol_id not in (1010100000201936, 1010100000460877, 1010100001276608, 1010100001295435, 1010100000358388, --
                                  1010100000374960, 1010100001412156, 1010100001474962, 1010100000962761, 1010100000683434, --
                                  1090100004614642, 1010100003217513, 1010100002363099, 1010100003987222, 1010100003986908, --
                                  1010100003986908, 1010100004762165, 1010100004638272, 1010100000228508, 1010100000247785,
                                  1010100000198784, 1010100000186529, 1010100000188264, 1010100000228650, 1010100013223191,
                                  1010100016656891, 1010100016511133, 1010100017714294)) -- RU
            or
            (:cm_database_id = 94401 and --AZ
            ok_from_ol_id not in (1010100000108053, 1020100000059542, 1010100000028406, 1010100000028405, 1010100000415814, 1010100000911364,
                              --
                              1010100000858275, 1010100000507838, 1010100001005248, 1010100000451997,
                                  1010100000681428, 1010100000681436, 1010100001015579, 1010100000974752, 1010100000955187, 1010100001015429))) and
            ok_from_k_id = kd_k_new and
            ol.ol_id = ok.ok_from_ol_id and
            ol.ol_memo <> 'Пакетное списание платежей с просрочкой 3 года и более.' and
            (ol_date <> cast('10.09.15' as date) or
            ol_user_id <> 1000000000001)
      group by ok_from_ol_id, ok_type, ok_ol_date
      order by ok_ol_date
      into ok_from_ol_id, ok_type, ok_ol_date, ok_os, ok_perc, ok_fine, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv
  do
  begin
    ok_fine = ok_fine + ok_kommis_adv;
    if (ok_type = 2) then
      ok_type_2 = ok_type_2 + ok_sh_os;
    if (ok_type = 3) then
      ok_type_3 = ok_type_3 + ok_sh_os;
    if (ok_type in (4, 5, 8)) then
      ok_balance = ok_balance + ok_os;
    ok_kommis_total = ok_kommis_total + ok_kommis;

    if (ok_type = 4 and
        ok_os > 0) then
    begin
      check_memo = 'Сумма погашения по ОС больше 0. operkred.ok_os = ' ||(ok_os);
      suspend;
    end
    if (ok_type = 4 and
        ok_perc > 0) then
    begin
      check_memo = 'Сумма погашения по %% больше 0. operkred.ok_perc = ' ||(ok_perc);
      suspend;
    end
    if (ok_type = 4 and
        ok_sh_os > 0) then
    begin
      check_memo = 'Сумма погашения по ОС больше 0. operkred.ok_sh_os = ' ||(ok_os);
      suspend;
    end
    if (ok_type = 4 and
        ok_sh_perc > 0) then
    begin
      check_memo = 'Сумма погашения по %% больше 0. operkred.ok_sh_perc = ' ||(ok_perc);
      suspend;
    end
    --   Проверка выдачи
    if (ok_type = 8) then
    begin
      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            ob_short_acc_dt in ('180-1/1', '185-1/1', iif(ob_ol_date <= cast('31.12.2013' as date), :acc_balance_old, :acc_balance))
      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;

      ob_balance = ob_summa_val;
      if (ok_os <> ob_summa_val and
          not(cm_database_id = 94401 and --AZ
          rep_k_id = 101010002545700101)) then
      begin
        check_memo = 'Выданная сумма не совпадает на: ' ||(ok_os - ob_summa_val);
        suspend;
      end
    end

    if (ok_type = 4) then
    begin
      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            ((ob_short_acc_kt containing ('450-6') and
            :ok_kommis <> 0) or
            ob_short_acc_kt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_kommis_old, :acc_kommis) or
            ob_short_acc_kt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_kommis_1_old, :acc_kommis))
      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      if (-ok_kommis <> ob_summa_val) then
      begin
        check_memo = 'Разница по полученному доп проценту: ' ||(ok_kommis + ob_summa_val);
        suspend;
      end

      ob_summa_val = null;
      if (cm_database_id = 94401) then
      begin
        select sum(ob_summa_val)
        from operbook ob
        where (ob.ob_short_acc_dt in ('185-1/1', '180-1/1', '21110/1', '21210/1', '21140/1', '21240/1', '21113/1', '21213/1', '21143/1', '21243/1',
                                      '23220/1') or
              ob.ob_short_acc_kt in ('185-1/1', '180-1/1', '21110/1', '21210/1', '21140/1', '21240/1', '21113/1', '21213/1', '21143/1', '21243/1',
                                     '23220/1')) and
              ob_from_ol_id = :ok_from_ol_id
        into ob_summa_val;
      end

      if (cm_database_id <> 94401) then
      begin
        select sum(iif(ob_short_acc_kt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_balance_old, :acc_balance), ob_summa_val, 0) - iif(ob_short_acc_dt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_balance_old, :acc_balance), ob_summa_val, 0))
        from operbook
        where ob_from_ol_id = :ok_from_ol_id and
              (ob_short_acc_dt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_balance_old, :acc_balance) or
              ob_short_acc_kt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_balance_old, :acc_balance))
        into ob_summa_val;
      end

      if (ob_summa_val is null) then
        ob_summa_val = 0;
      ob_balance = ob_balance - ob_summa_val;
      if (-ok_os <> ob_summa_val) then
      begin

        check_memo = 'Разница по полученному ОС: ' ||(ok_os + ob_summa_val);
        suspend;
      end

      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            ((ob_short_acc_kt containing ('450-6') and
            :ok_perc <> 0) or
            (ob_short_acc_kt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_percent_old, :acc_percent)))
      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      if (-ok_perc <> ob_summa_val) then
      begin
        check_memo = 'Разница по выплаченному проценту: ' ||(ok_perc + ob_summa_val);
        suspend;
      end

      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            (ob_short_acc_dt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_fine_old, :acc_fine) or
            ob_short_acc_kt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_fine_old, :acc_fine) or
            ob_short_acc_kt containing '706-1/')
      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      if (-ok_fine <> ob_summa_val and
          ok_ol_date <> '01.04.19') then
      begin
        check_memo = 'Разница по выплаченной пени: ' ||(ok_fine + ob_summa_val);
        suspend;
      end

    end
    --   Переплата кредита
    if (ok_type = 9) then
    begin
      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            (ob_short_acc_dt in ('450-9/1', '76-04/3', iif(ob_ol_date <= cast('31.12.2013' as date), :acc_pereplata_old, :acc_pereplata)) or --
            ob_short_acc_kt in ('450-9/1', '76-04/3', iif(ob_ol_date <= cast('31.12.2013' as date), :acc_pereplata_old, :acc_pereplata)))

      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      if (-ok_os <> ob_summa_val) then
      begin
        check_memo = 'Разница по переплате ОС: ' ||(ok_os + ob_summa_val);
        if (cm_database_id = 81101 and
            ok_from_ol_id not in (1010100002363099)) then
          suspend;
      end
    end
    --   Списание
    if (ok_type = 5) then
    begin
      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            (ob_short_acc_kt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_balance_old, :acc_balance) or --
            ob_short_acc_kt in ('185-1/1', '180-1/1', '21110/1', '21210/1', '21140/1', '21240/1', '21113/1', '21213/1', '21143/1', '21243/1',
                                '23220/1')) and
            ob_short_acc_dt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_llc_os_old, :acc_llc_os)

      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      ob_balance = ob_balance - ob_summa_val;
      if (-ok_os <> ob_summa_val) then
      begin
        check_memo = 'Разница по списанному ОС: ' ||(ok_os + ob_summa_val);
        suspend;
      end
      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            ob_short_acc_kt = iif(ob_ol_date <= cast('31.12.2013' as date), :acc_perc_dohod_old, :acc_perc_dohod) and
            ob_short_acc_dt in (:acc_llc_percent_old, :acc_llc_percent, :acc_llc_percent_old_1, '11002/2') --в всязи с изменением счета 19.02.15
      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      if (-ok_perc <> ob_summa_val) then
      begin
        check_memo = 'Разница по списанному Проценту: ' ||(ok_perc + ob_summa_val);
        suspend;
      end
    end
  end
  ok_ol_date = null;
  if (ok_balance < 0) then
  begin
    check_memo = 'Баланс контракта < 0: ' ||(ok_balance);
    suspend;
  end

  /* Добавил igusev 23.09.2014, вместо кода который идет выше

    Если выявлено, что суммарный баланс включая амнистированные контракты <> 0,
      прохожусь по контрактам связанным с rep_k_id, проверяю их баланс и статус
   */
  if (ok_balance <> 0) then
  begin
    for select kd_k_new, kont.k_kontract_no, ks.ks_state
        from (select kd_k_new
              from kontdevides kd
              where kd_k_orig = :rep_k_id
              union
              select :rep_k_id
              from rdb$database) a
        inner join kontract kont on kont.k_id = a.kd_k_new
        inner join kontstate ks on ks.ks_from_k_id = kont.k_id
        where :rep_date_end between ks.ks_date_begin and ks.ks_date_end
        into :kd_k_new, :k_kontract_no_buf, :ks_state_buf
    do
    begin
      ok_balance_buf = 0;

      for select sum(ok.ok_sh_os), ok_type
          from operkred ok
          where ok.ok_from_k_id = :kd_k_new and
                ok.ok_type in (2, 3) and
                ok.ok_ol_is_active = 1 and
                ok_from_c_id = 0
          group by ok_type
          into :ok_sum_buf, :ok_type_buf
      do
      begin
        if (coalesce(ok_sum_buf, 0) <> 0) then
        begin
          check_memo = k_kontract_no_buf || '; ' || 'У контракта ok_type = ' || ok_type_buf || ' <> 0: ' ||(ok_sum_buf);
          suspend;
        end
      end

      select sum(ok.ok_os)
      from operkred ok
      where ok.ok_from_k_id = :kd_k_new and
            ok.ok_type in (4, 5, 8) and
            ok.ok_ol_is_active = 1 and
            ok_from_c_id = 0
      into :ok_balance_buf;

      if ((ks_state_buf = 2) and
          (ok_balance_buf <> 0)) then
      begin
        check_memo = k_kontract_no_buf || '; ' || 'У закрытого контракта Баланс <> 0: ' ||(ok_balance_buf);
        suspend;
      end

      if (KS_STATE_BUF = 1) then
      begin
        for select OK_SH_ID, OK.OK_SH_DATE, sum(OK.OK_OS + OK_SH_OS) OK_OS, sum(OK_PERC + OK_SH_PERC) SH_PERC
            from OPERKRED OK
            where OK.OK_FROM_K_ID = :KD_K_NEW and
                  OK.OK_TYPE in (1, 2, 3, 4, 5, 6, 9) and
                  OK.OK_OL_IS_ACTIVE = 1 and
                  OK_FROM_C_ID = 0
            group by OK.OK_SH_ID, OK.OK_SH_DATE
            having sum(OK.OK_OS + OK_SH_OS) < 0 or (sum(OK_PERC + OK_SH_PERC) < 0)
            into P_SH_ID, P_SH_DATE, P_SH_OS, P_SH_PERC
        do
        begin
          CHECK_MEMO = K_KONTRACT_NO_BUF || '; ' || 'Планируемое сумма погашении меньше 0(нуля) ' || P_SH_ID || '; ' || P_SH_DATE || '; ' || P_SH_OS || '; ' || P_SH_PERC;
          suspend;
        end
      end
    end
  end

  if (ok_balance <> ob_balance and
      not(cm_database_id = 94401 and --AZ
      rep_k_id = 101010002545700101 and
      -(ob_balance - ok_balance) = 5000)) then
  begin
    check_memo = 'Баланс контракта не идет с обороткой, Разница: ' ||(ob_balance - ok_balance);
    suspend;
  end
  if (ok_type_2 <> 0) then
  begin
    check_memo = 'Сумма реструктуризации по ОС <> 0, ok_type = 2: ' || ok_type_2;
    suspend;
  end
  if (ok_type_3 <> 0) then
  begin
    check_memo = 'Сумма автореструктуризации по ОС <> 0, ok_type = 3: ' || ok_type_3;
    suspend;
  end
  -- Проверка комиссии с графиком
  -- ok_from_c_id=0 добавлен для пары

  sh_kommis = null;
  with tabl1
  as (select kd_k_new
      from kontdevides
      where kd_k_orig = :rep_k_id
      union
      select :rep_k_id
      from rdb$database)
  select sum(sh_kommis)
  from schedule, tabl1
  where sh_from_k_id = kd_k_new
  into sh_kommis;
  if (sh_kommis is null) then
    sh_kommis = 0;
  if (sh_kommis <> -ok_kommis_total) then
  begin
    check_memo = 'Расхождения в сумме комиссии, SCHEDULE <> OPERKRED Разница: ' ||(sh_kommis + ok_kommis_total);
    suspend;
  end

  kc_kolich_client = null;
  select count(*)
  from knclient
  where kc_from_k_id = :rep_k_id and
        kc_amount_approved > 0
  into kc_kolich_client;
  if (kc_kolich_client is null) then
    kc_kolich_client = 0;

  if (k_kolich_client <> kc_kolich_client) then
  begin
    check_memo = 'Разница в кол-ве клиентов, KONTRACT <> KNCLIENT';
    suspend;
  end

  --Проверка на соответствие суммы выдачи в k_amount_approved и sh_os
  select sum(sh_os) sh_os
  from schedule sch
  where sch.sh_from_k_id = :rep_k_id
  into sh_os;

  if (sh_os is null) then
    sh_os = 0;

  select k_amount_approved, coalesce(kont.k_percent_for_days, -777), coalesce(pr.pr_percent_for_days, -777)
  from kontract kont
  inner join product pr on pr.pr_id = kont.k_from_pr_id
  where k_id = :rep_k_id
  into k_amount_approved, k_percent_for_days, pr_percent_for_days;

  if (sh_os <> k_amount_approved) then
  begin
    check_memo = 'Разница в утвержденной сумме, sh_os <> k_amount_approved';
    suspend;
  end

  if (k_percent_for_days <> pr_percent_for_days) then
  begin
    check_memo = 'Неправильное значение ежедневной ставки. k_percent_for_days = ' || k_percent_for_days || ', pr_percent_for_days=' || pr_percent_for_days;
    suspend;
  end
end
