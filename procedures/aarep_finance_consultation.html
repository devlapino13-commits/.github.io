CREATE PROCEDURE AAREP_FINANCE_CONSULTATION(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_100,
    CL_COUNT D_INTEGER,
    FIRST_ENTER_COUNT D_INTEGER,
    ENTER_COUNT D_INTEGER)
AS
declare variable RD_ID D_INTEGER;
declare variable REG_UNICODE D_INTEGER;
begin
  rd_id = f_get_rd_id('CashTransactionIncom.Consultation');

  for select reg.reg_name,
             reg.reg_unicode,
             count(kncl.kc_from_c_id)
      from knclient kncl
      inner join kontract k on k.k_id = kncl.kc_from_k_id
      inner join product pr on pr.pr_id = k.k_from_pr_id and
            pr.pr_code containing '.consultation.'
      inner join kontstate ks on k.k_id = ks.ks_from_k_id and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end
      inner join region reg on reg.reg_unicode = ks.ks_region_id
      where kc_amount_approved > 0
      group by reg.reg_name, reg.reg_unicode
      into :reg_name,
           :reg_unicode,
           :cl_count
  do
  begin
    first_enter_count = 0;

    select count(cvs.cks_first_visit_datetime)
    from clt_kc_visit_svod cvs
    where cvs.cks_first_visit_date between :rep_date_begin and :rep_date_end
    and cvs.cks_office_id = :reg_unicode
    into first_enter_count;

    enter_count = 0;

    select count(cvs.ckv_id)
    from clt_kc_visit cvs
    where cvs.ckv_date between :REP_DATE_BEGIN and :REP_DATE_END
    and cvs.ckv_office_id = :reg_unicode
    into enter_count;

    suspend;
  end
end
