declare variable KC_AMOUNT CURR_TYPE;

  for select kc_from_c_id, kc_amount_approved, pr.pr_code, kont.k_disb_date, k_refund
      from knclient kncl
      inner join kontract kont on kont.k_id = kncl.kc_from_k_id
      inner join product pr on pr.pr_id = kont.k_from_pr_id and
            pr_code containing '.par.'  and
            pr_code containing '.loyal.'
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into c_id, kc_amount, pr_code, kf_disb_date, k_refund
  do
  begin
    if (kf_disb_date < '01.11.2020') then  -- на выдачи с  01.11.2020
      exit;
    select c_name_fio, c_pass_no
    from clients
    where c_id = :c_id
    into res_info_text, kf_client_pass_no;
    select k_id, ks_date_begin
    from  knclient kncl
    inner join kontstate ks on kncl.kc_from_k_id = ks.ks_from_k_id   and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin <= :kf_disb_date and
          :kf_disb_date between ks_date_begin and ks_date_end
    inner join kontract kon on kon.k_id = ks.ks_from_k_id
    inner join product pr on kon.k_from_pr_id = pr.pr_id  and
          (pr_code containing '.shans.' or
          pr_code containing '.express.' or
          pr_code containing '.reliable_client.' or
          pr_code containing '.reliable_client_plus.' or 
          pr_code containing '.advantageous_side.' or
          pr_code containing '.blago.' or
          pr_code containing '.uspeh.' or
          pr_code containing '.zalog.'
          )
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0
    order by ks_date_begin desc
    rows 1
    into activ_k_id, activ_k_disb_date;

    not(k_refund = 1 and kf_disb_date >= '01.03.2022' and pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.)%')) then

      res_error_text = 'Нет активного займа по основному продукту';
      suspend;
    end

    begin
      res_error_text = 'Со дня выдачи основного займа еще не иcтекло 30 дней!';
      res_error_is_critical = 1;
      suspend;
    end
    else
    begin
      if (not exists(select *

                     where ok.ok_type = 4
                     and ok.ok_ol_is_active = 1
                     and ok.ok_from_k_id = :activ_k_id
                     and ((ok.ok_os < 0) or
                          (ok.ok_perc < 0)))) then
      begin
        res_error_text = 'Не было платежей по основному займу';
        res_error_is_critical = 1;
        suspend;
      end
    end*/
