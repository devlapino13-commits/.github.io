CREATE PROCEDURE PRINT_KONT_BUILDER(
    K_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    CURR_DATE D_STR_CODE,
    MAI_OBJECT_ADDRESS D_STR_255,
    C_NAME_FIO D_STR_100,
    C_NAME_FIO_BUIL D_STR_100,
    KIW_MEMO D_STR_255,
    KIW_DAYS D_INTEGER,
    KIW_PRICE D_CURRENCY,
    KIW_PRICE_ALL D_CURRENCY,
    KML_TOTAL_ALL D_CURRENCY,
    MAI_GUARANTEE_PERIOD D_INTEGER,
    QR_CODE_CL D_STR_LONG,
    QR_CODE_BUIL D_STR_LONG,
    DATE_SIGN DATE,
    C_OKPO D_STR_20,
    C_INN D_STR_20,
    BANK_ACCOUNT_NO VARCHAR(50),
    BANK_BIK D_STR_20,
    BANK_NAME D_STR_255,
    AG_ELSOM_ACCOUNT D_STR_20,
    ADL_ADDRESS_STR D_STR_255,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO D_STR_255,
    PANT_DOC_NO VARCHAR(30),
    PANT_DATE_BEGIN DATE,
    REGISTR_DOC_NO VARCHAR(30),
    REGISTR_DATE_BEGIN DATE,
    LAST_YEAR_DATE D_STR_100)
AS
declare variable MA_ID D_ID;
declare variable EDL_ID D_ID;
declare variable C_ID D_ID;
declare variable C_ID_BUIL D_ID;
declare variable KIP_FROM_C_ID D_ID;
declare variable CDS_DOC_NO varchar(30);
declare variable CDS_DOC_DATE_BEGIN date;
declare variable PDT_CODE D_STR_100;
declare variable EDOC_CREATE_DATE date;
begin

  curr_date = f_datetostr(current_date);
  mai_object_address = '';
  kiw_price = 0;
  kml_total_all = 0;
  date_sign = null;

  select coalesce(k.k_from_ma_app_id, 0)
  from kontract k
  where k.k_id = :k_id
  into :ma_id;

  --Указываем точный адрес объекта, период гарантии
  select mif.mai_object_address, mif.mai_guarantee_period
  from miniapp_ipf mif
  where mai_from_ma_id = :ma_id rows 1
  into :mai_object_address, mai_guarantee_period;
  -- ФИО клиента
  c_id = 0;
  select coalesce(c_name_fio, ''), c_id
  from knclient knc
  inner join clients cl on cl.c_id = knc.kc_from_c_id
  where knc.kc_from_k_id = :k_id and
        knc.kc_amount_approved > 0 rows 1
  into :c_name_fio, :c_id;

  --общая стоимость материалов
  select coalesce(sum(kml_total), 0)
  from kontmat
  where kml_object_kind = 9 and
        kml_object_id = :k_id
  into :kml_total_all;

  --общая стоимость за работу
  select coalesce(sum(kiw.kiw_price), 0)
  from kontract_istisnaa_work kiw
  where kiw.kiw_from_k_id = :k_id and
        kiw.kiw_is_delete = 0
  into :kiw_price_all;

  --берем подрядчика
  select kip.kip_from_c_id
  from kontract_istinaa_prepay kip
  where kip.kip_from_k_id = :k_id and
        kip.kip_is_active = 1 rows 1
  into :kip_from_c_id;

  c_inn = '';
  c_okpo = '';
  c_pass_no = '';
  c_pass_date = null;
  c_pass_who = '';

  --инн, окпо
  select cl.c_inn, c_okpo, c_pass_no, c_pass_date, cl.c_pass_who
  from clients cl
  where cl.c_id = :kip_from_c_id
  into :c_inn, c_okpo, c_pass_no, c_pass_date, c_pass_who;

  bank_account_no = '';
  bank_bik = '';
  bank_name = '';
  -- номер счета, бик, банк
  select bc_account_no, bc_from_b_id, b_name
  from bankclient bc, banks
  where bc.bc_from_c_id = :kip_from_c_id and
        bc.bc_is_active = 1 and
        bc.bc_from_b_id = b_id rows 1
  into bank_account_no, bank_bik, bank_name;

  ag_elsom_account = '';
  --Номер электронного кошелька ЭлСом
  select ag_elsom_account
  from agreements a
  where a.ag_object_kind = 0 and
        a.ag_object_id = :kip_from_c_id and
        ag_status = 1
  order by ag_agreement_date desc
  rows 1
  into :ag_elsom_account;

  --юр адрес
  adl_address_str = '';
  select adl_address_str
  from addrclient
  inner join addrlist adl on adl_id = adc_from_adl_id
  where adc_from_c_id = :kip_from_c_id and
        adc_address_type = 1
  into adl_address_str;

  pant_doc_no = '';
  pant_date_begin = null;
  registr_doc_no = '';
  registr_date_begin = null;

  for select cds_doc_no, cds_doc_date_begin, pdt_code
      from clientdocs cd
      inner join passkind pk on cd.cds_from_pk_id = pk_id
      inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cds_from_c_id = :kip_from_c_id and
            current_date between cds_doc_date_begin and cds_doc_date_end and
            cds_is_delete = 0 and
            pdt.pdt_code in ('work_patent', 'certificate_ip')
      into :cds_doc_no, cds_doc_date_begin, pdt_code
  do
  begin
    if (pdt_code = 'work_patent') then
    begin
      pant_doc_no = cds_doc_no;
      pant_date_begin = cds_doc_date_begin;
    end
    else
    begin
      registr_doc_no = cds_doc_no;
      registr_date_begin = cds_doc_date_begin;
    end
  end

  edl_id = 0;
  qr_code_buil = '';
  qr_code_cl = '';
  edoc_create_date = null;
  -- QR-код электронной подписи клиента. Текст пустой если не подписан
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);
    qr_code_cl = f_edoc_qr_by_edl(edl_id, :c_id);

    edl_id = f_json_str_get_value(add_params, 'edl_id', null);

    select cast(eds.eds_status_datetime as date), cast(edl_create_datetime as date)
    from edoc_list edl
    inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id and
          eds.eds_status = 1
    where edl.edl_id = :edl_id rows 1
    into date_sign, edoc_create_date; -- в рамках истисны эл. док уходит только 1 подрядчику

    last_year_date = '';
    last_year_date = '31.12.' || extract(year from :edoc_create_date);
  end

  --инфо по работам
  for select c_name_fio, kiw_memo, kiw_days, kiw_price, c_id
      from kontract_istisnaa_work kiw
      inner join clients cl on cl.c_id = kiw.kiw_from_c_id
      where kiw.kiw_from_k_id = :k_id and
            kiw_is_delete = 0
      into :c_name_fio_buil, kiw_memo, kiw_days, kiw_price, c_id_buil
  do
  begin
    -- QR-код электронной подписи клиента. Текст пустой если не подписан
    if (edl_id > 0) then
      qr_code_buil = f_edoc_qr_by_edl(edl_id, :c_id_buil);
    suspend;
  end
end
