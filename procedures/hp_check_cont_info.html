CREATE PROCEDURE HP_CHECK_CONT_INFO(
    C_ID D_ID,
    CONT_KIND D_STR_CODE,
    ONLY_VERIFIED D_SMALLINT = 0,
    ONLY_INCORRECT D_SMALLINT = 0,
    MAX_ROW D_INTEGER = 0)
RETURNS (
    CDA_ID D_ID,
    CDA_FROM_C_ID D_ID,
    C_NAME_FIO D_STR_100,
    CDA_KIND_KIND D_STR_CODE,
    CDA_CONT_VALUE D_STR_255,
    CDA_IS_MAIN D_SMALLINT,
    CDA_STATUS D_SMALLINT,
    CORRECT_VALUE D_STR_CODE,
    RES_KIND D_STR_CODE)
AS
declare variable rec_no d_integer;
declare variable cont_kind_code d_str_code;
begin
  -- возвращает список неправильных значений.

  -- Входящие параметры:
  -- Если входящий C_ID is null, проверит по всем клиентам
  -- Если MAX_ROW = 0, вернет все, иначе ровно столько сколько указан в этом параметре

  -- CONT_KIND:
  -- Пусто - значит все виды контактных данных
  -- Иначе только совпадающие с Cont_Kind

  -- ONLY_INCORRECT:
  -- 1 - возвращает только ошибочные (res_kind = incorrect)
  -- 0 - возвращает и тех где можно исправить (res_kind = fix)

  -- Результат RES_KIND:
  --   fix       - значит значение можно исправить
  --   incorrect - значение неверное, нельзя автоматом исправить

  if (max_row is null) then
    max_row = 0;

  if (only_incorrect is null) then
    only_incorrect = 0;

  if (only_verified is null) then
    only_verified = 0;

  rec_no = 0;

  cont_kind = trim(coalesce(cont_kind, ''));
  cont_kind_code = null;
  if (cont_kind <> '') then
  begin
    cont_kind = f_enum('cont_kind', cont_kind, 'id');
    cont_kind_code = f_enum('cont_kind', cont_kind, 'code');
  end else
    cont_kind = null;

  for select cda_id, cda_from_c_id, c_name_fio, cda_is_main, cda_status,
             f_cont_code(cda_from_clt_id),
             iif(:cont_kind_code is not null, :cont_kind_code, f_enum('cont_kind', cda_cont_kind, 'code'))
  from contdata, clients
  where cda_from_c_id = c_id and
        cda_from_clt_id <> 0 and
        cda_is_delete = 0 and
        cda_status >= :only_verified and
        (:cont_kind is null or cda_cont_kind = :cont_kind) and
        (:c_id is null or cda_from_c_id = :c_id)
  into cda_id, cda_from_c_id, c_name_fio, cda_is_main, cda_status, cda_cont_value, cda_kind_kind
  do
  begin
    res_kind = '';
    correct_value = '';
    begin
      correct_value = f_cont_trim(cda_cont_value, cda_kind_kind);
      if (correct_value <> cda_cont_value) then
        res_kind = 'fix';

      when any do
      begin
        res_kind = 'incorrect';
      end
    end

    if (res_kind <> '') then
    begin
      if (only_incorrect = 0 or res_kind = 'incorrect') then
      begin
        rec_no = rec_no + 1;
        suspend;
      end
    end

    if (max_row > 0 and rec_no = max_row) then
      Break;
  end
end
