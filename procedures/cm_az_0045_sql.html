CREATE PROCEDURE CM_AZ_0045_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable c_id d_id;
declare variable kol d_smallint;
declare variable k_id_last d_id;
declare variable max_continuing_deling d_integer;
declare variable kc_amount_approved_max d_currency;
declare variable kc_amount_approved_tmp d_currency;
declare variable k_id_tmp d_id;
begin
  res_error_is_critical = 1;
  for select knc.kc_from_c_id, knc.kc_amount_approved, k.k_disb_date, cl.c_name_fio
      from knclient knc
      inner join clients cl on cl.c_id = knc.kc_from_c_id
      inner join kontract k on k.k_id = knc.kc_from_k_id
      where knc.kc_from_k_id = :k_id and
            knc.kc_amount_approved > 0
      into :c_id, :kc_amount, :kf_disb_date, :res_info_text
  do
  begin
       k_id_last = null;
       max_continuing_deling = 0;
       kc_amount_approved_max = 0;
       -- нахожу послдний займ клиента
       for select ks.ks_from_k_id,
              kncl.kc_amount_approved
       from kontstate ks
       inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
       where kncl.kc_from_c_id = :c_id
       and kncl.kc_amount_approved > 0
       and ks.ks_state = 1
       and ks.ks_ol_is_delete = 0
       and ks.ks_date_begin <= :kf_disb_date
       and ks.ks_from_k_id <> :k_id
       order by ks.ks_date_begin desc -- выжно именно последний поймать
       into :k_id_tmp,
            :kc_amount_approved_tmp
       do
       begin
        if (k_id_last is null) then
            k_id_last = k_id_tmp;
        if (kc_amount_approved_max < kc_amount_approved_tmp) then
            kc_amount_approved_max = kc_amount_approved_tmp;
       end
       if (kc_amount_approved_max > 2000) then
        kc_amount_approved_max = 2000;
       if (kc_amount_approved_max < 1000) then
        kc_amount_approved_max = 1000;
       select coalesce(max(ok.ok_ol_date - ok_sh_date), 0)
       from operkred ok
       where ok.ok_ol_is_active = 1
       and ok.ok_ol_date  <= :kf_disb_date
       and ok.ok_ol_date > ok_sh_date
       and ok_type < 7
       and ok_from_k_id = :k_id_last
       and ok_os + ok_perc < 0
       and ok_from_c_id=0
       into :max_continuing_deling;
       if (((k_id_last is null) or (max_continuing_deling > 7)) and (kc_amount > 1000)) then
       begin
            res_error_text = 'Новым клиентам и повторным клиентам, с просрочкой более 7 дней максимальная сумма займа 1000 AZN';
            suspend;
            exit;
       end
       if ((k_id_last is not null) and (max_continuing_deling <= 7)) then
       begin
            if (kc_amount > kc_amount_approved_max) then
            begin
                res_error_text = 'Повторным клиентам у которых не было просрочек и у которых были просрочки до 7 дней включительно на предыдущем цикле, выдаем до максимальной суммы займа на любом из циклов. Но не более 2000 AZN';
                suspend;
                exit;
            end
       end
  end
end
