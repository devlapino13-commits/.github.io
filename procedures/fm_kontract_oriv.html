CREATE PROCEDURE FM_KONTRACT_ORIV(
    UN_ID D_ID,
    DATE_BEGIN DATE,
    DATE_END DATE)
RETURNS (
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    K_KONTRACT_NO D_STR_100,
    REG_NAME D_STR_100,
    C_NAME_FIO D_STR_100,
    DELINQ_SUM_ALL DECIMAL(18,2),
    SROC D_INTEGER)
AS
declare variable lo_id d_id;
declare variable is_curator d_smallint;
declare variable oks_close_state d_smallint;
declare variable kroff decimal(18,2);
declare variable extra decimal(18,2);
begin
  obj_kind = f_objkind('kontract', 'id');

  if (un_id = 0) then
  begin
    exception e_message('Выбранного ОРИВ еще нет в финстарт, обратитесь тех поддержке');
    exit;
  end
  select un_id
  from hp_test_un_id(:un_id)
  into :un_id;

  select u.un_from_c_id
  from username u
  where u.un_id = :un_id
  into :lo_id;
  is_curator = 0;

  if (not exists(select *
                 from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.ViewAccessOriv')
                 where has_access = 1)) then
    exit;

  select c_name_fio
  from clients cl
  where cl.c_id = :lo_id
  into :c_name_fio;

  for select distinct k.k_id, k.k_kontract_no, r.reg_name, current_date - sa.sba_date_begin
      from sb_akt sa
      inner join sb_list sl on sl.sbl_id = sa.sba_from_sbl_id
      inner join kontract k on k.k_id = sa.sba_from_k_id
      inner join region r on r.reg_unicode = k.k_fact_disb_reg_id
      where sl.sbl_from_c_id = :lo_id and
            sl.sbl_id = sa.sba_from_sbl_id and
            iif(sba_date_end = '31.12.9999', current_date, sba_date_end) between :date_begin and :date_end and
            sa.sba_is_delete = 0 and
            sba_status = 2 and
            sbl_type = 1 and
            sa.sba_date_end = '31.12.9999' /*and
            k_id = 101010037776300101*/
      into :obj_id, :k_kontract_no, :reg_name, :sroc
  do
  begin
    oks_close_state = null;
    kroff = 0;
    extra = 0;

    select oks.oks_close_state,
           (oks.oks_out_os + oks.oks_out_perc + oks.oks_out_other), oks.oks_extra
    from oksummary oks
    where current_date between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_from_k_id = :obj_id
    into :oks_close_state, kroff, extra;

    delinq_sum_all = 0;
    if (oks_close_state = 0) then
    begin

      select coalesce(delinq_sum_all, 0)
      from print_kont_delinq_zadol(:obj_id)
      into :delinq_sum_all;
    end

    if (oks_close_state = 1) then
    begin
      delinq_sum_all = kroff + extra;
    end

    if (delinq_sum_all > 0) then
      suspend;
  end
end
