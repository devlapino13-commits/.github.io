CREATE PROCEDURE AAREP_DISB_NOT_SMS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REGION_NAME VARCHAR(60),
    KF_OFFICER_NAME VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    KAL_NEW_KF_NAME VARCHAR(30),
    C_NAME_FIO VARCHAR(100),
    K_DISB_DATE DATE,
    KC_AMOUNT_APPROVED CURR_TYPE,
    C_PLASTCARD_NO VARCHAR(20),
    SV_VERIFIED_STR VARCHAR(100),
    SV_ERRORS VARCHAR(255),
    SV_IN_MINUTES INTEGER,
    SV_COUNT INTEGER,
    SV_UN_FIO VARCHAR(100))
AS
declare variable rep_k_id id_type;
declare variable kc_from_c_id id_type;
declare variable rep_k_id_prior id_type;
declare variable kal_new_kf_officer id_type;
declare variable sv_id integer;
begin
  rep_k_id_prior = 0;
  sv_un_fio = '';

  for select sv_id, rep_k_id, kc_from_c_id, region_name, sv_verified_str, kc_amount_approved, sv_errors, sv_in_minutes,
             sv_count
      from aarep_sms_approve_list_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
      where sv_is_verified in (-1, 0, 1)
      into sv_id, rep_k_id, kc_from_c_id, region_name, sv_verified_str, kc_amount_approved, sv_errors, sv_in_minutes,
           sv_count
  do
  begin
    if (rep_k_id_prior <> rep_k_id) then
    begin
      select k_kontract_no, kal_new_kf_name, kal_new_kf_officer, k_disb_date
      from kontract, knaktlist
      where k_id = :rep_k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            k_disb_date between kal_date and kal_date_end
      into k_kontract_no, kal_new_kf_name, kal_new_kf_officer, k_disb_date;

      kf_officer_name = (select c_name_fio
                         from clients
                         where c_id = :kal_new_kf_officer);
    end
    select c_name_fio,
           (select cont_value
            from hp_contdata_get_cell_valid(0, :kc_from_c_id))
    from clients
    where c_id = :kc_from_c_id
    into c_name_fio, c_plastcard_no;
    suspend;
  end
end
