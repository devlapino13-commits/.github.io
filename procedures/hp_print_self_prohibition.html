CREATE PROCEDURE HP_PRINT_SELF_PROHIBITION(
    C_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_INN D_STR_20,
    C_NAME_FIO D_STR_100,
    C_PASS_NO D_STR_CODE,
    SELF_PROHIBITION_STATUS_STR D_STR_255,
    SELF_PROHIBITION_DATETIME D_DATETIME)
AS
declare variable k_id d_id;
declare variable k_disb_date d_date;
declare variable spm_total_result d_integer;
declare variable cds_create_datetime  d_datetime;
begin
  -- RFCKG-4723

  k_id = f_json_blob_get_value(add_params, 'k_id', 0);

  select k.k_disb_date
  from kontract k
  where k.k_id = :k_id
  into k_disb_date;

  select c.c_inn,
         c.c_name_fio,
         c.c_pass_no
  from clients c
  where c.c_id = :c_id
  into c_inn,
       c_name_fio,
       c_pass_no;

  select
         case
           when spm.spm_total_result = 1 then 'Установлен'
           else 'Не установлен'
         end,
         spm.spm_last_query_datetime,
         spm.spm_total_result
  from self_prohibition_main spm
  where spm.spm_from_c_id = :c_id and
        spm.spm_object_kind = 9 and
        spm.spm_object_id = :k_id
  rows 1
  into :self_prohibition_status_str,
       :self_prohibition_datetime,
       :spm_total_result;

  if (spm_total_result < 0) then
  begin
    select cds.cds_create_datetime
    from clientdocs cds
    inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
    inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
    where pdt.pdt_code = 'self_prohibition' and
          cds.cds_from_c_id = :c_id and
          cds.cds_doc_date_begin <= cast(:self_prohibition_datetime as d_date)
    order by cds.cds_create_datetime desc
    rows 1
    into cds_create_datetime;

    if (cds_create_datetime is not null) then
      self_prohibition_datetime = cds_create_datetime;
  end

  c_inn = coalesce(c_inn, '-');
  c_name_fio = coalesce(c_name_fio, '-');
  c_pass_no = coalesce(c_pass_no, '-');
  self_prohibition_status_str = coalesce(self_prohibition_status_str, '-');
  self_prohibition_datetime = coalesce(self_prohibition_datetime, cast('31.12.9999' as d_datetime));
  suspend;
end
