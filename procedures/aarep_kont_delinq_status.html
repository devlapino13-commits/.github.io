CREATE PROCEDURE AAREP_KONT_DELINQ_STATUS(
    REP_K_ID DECIMAL(18,0),
    REP_DATE_END DATE)
RETURNS (
    REP_IS_DELINQ INTEGER,
    REP_DELINQ_BEGIN DATE,
    REP_DELINQ_DAYS INTEGER)
AS
/*Проверка контракта на просрочку, на указанную дату*/
begin
  rep_is_delinq = 0;
  rep_delinq_days = 0;
  rep_delinq_begin = null;

  if (exists(select *
             from kontstate
             where ks_from_k_id = :rep_k_id and
                   ks_date_begin <= :rep_date_end and
                   ks_date_end >= :rep_date_end and
                   ks_ol_is_delete = 0 and
                   ks_state = 1)) then
  begin
    select ok_sh_date
    from operkred
    where ok_from_k_id = :rep_k_id and
          ok_ol_is_active = 1 and
          ok_type < 7 and
          ok_ol_date <= :rep_date_end and
          ok_sh_date <= :rep_date_end and
          ok_from_c_id = 0
    group by ok_from_k_id, ok_sh_date
    having sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc) > 1
    order by ok_from_k_id, ok_sh_date
    rows 1
    into :rep_delinq_begin;

    if (rep_delinq_begin is not null) then
    begin
      rep_is_delinq = 1;
      rep_delinq_days = cast(rep_date_end - :rep_delinq_begin as integer);
    end
  end
  suspend;
end
