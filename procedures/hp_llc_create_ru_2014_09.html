CREATE PROCEDURE HP_LLC_CREATE_RU_2014_09(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_ID INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    LM_ID DECIMAL(18,0),
    AA_K_ID DECIMAL(18,0),
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_DELIQ_DAYS INTEGER,
    AA_PR_IS_INDIV INTEGER)
AS
declare variable aa_llc_percent decimal(18,7);
declare variable aa_llc_id integer;
declare variable lm_no integer;
declare variable db_id integer;
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate curr_type;
declare variable cr_rate curr_type;
declare variable ks_state integer;
declare variable k_disb_date date;
declare variable pr_id integer;
declare variable pr_product_kind integer;
declare variable is_restruct integer;
declare variable z_name_fio varchar(255);
declare variable section integer;
declare variable rep_period integer;
declare variable kcp_perc_saldo curr_type;
declare variable k_kolich_client_approved integer;
declare variable max_date_graph date;
declare variable max_date_rest date;
declare variable sh_date date;
declare variable pr_code varchar(100);
begin
--Кредиты разделяются на 2 категории, выданные до 28.02.2012 и выданные с 28.02.2012
-- выданные до 28.02.2012 будут классифицироваться по старому
-- выданные с 28.02.2012 будут классифицироваться по новому положенияю РППУ
-- section: 1 - кредит с обеспечением; 2 - кредит без обеспечения;
--  db_id = (select cm_database_id from company);
  db_id = f_get_db_id();
  rep_period = extract(year from cast(:rep_date_end as date)) || right('0' || extract(month from cast(:rep_date_end as date)), 2);

  delete from llc_main
  where lm_date = :rep_date_end;
  for select reg_unicode, reg_code, reg_name
      from region
      where reg_status = 3
      order by reg_code
      into :reg_id, :reg_code, :reg_name
  do
  begin
    select result
    from get_kv_id(10)
    into :lm_id;

    lm_no = right(:lm_id, 4);
    insert into llc_main (lm_id, lm_date, lm_date_status, lm_from_reg_id, lm_no, lm_memo, lm_no_str, lm_is_signed)
    values (:lm_id, :rep_date_end, :rep_date_end, :reg_id, :lm_no, '', :lm_no, 1);

    insert into llc_sign (ls_from_lm_id, ls_number, ls_sign_type, ls_fio)
    values (:lm_id, 1, '1', 'Белоусова Лариса');

--  Вытаскивает закрытые и активные контракты
    for select ks_from_k_id as aa_k_id, 0 as aa_amount_active_end, 0 as aa_deliq_days, 2 as ks_state
        from kontstate -- выборка по закрытым контрактам
        where ks_ol_is_delete = 0 and
              ks_state = 2 and
              ks_region_id = :reg_id and
              cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end
        union
        select oks_from_k_id, oks_balance, cast(:rep_date_end as date) - oks_delinq_date, -- просроченные дни
               1 as ks_state
        from oksummary -- выборка по активным контрактам
        where oks_region_id = :reg_id and
              oks_close_state = 0 and
              :rep_date_end between oks_date_begin and oks_date_end
        into :aa_k_id, :aa_amount_active_end, :aa_deliq_days, ks_state
    do
    begin
      if (aa_deliq_days < 0) then
        aa_deliq_days = 0;

      select pr_loan_type, k_index_curr_code, k_index_curr_rate, k_disb_date, pr_id, pr_product_kind,
             k_kolich_client_approved, pr.pr_code
      from kontract, product pr
      where k_id = :aa_k_id and
            k_from_pr_id = pr_id
      into aa_pr_is_indiv, k_index_curr_code, k_index_curr_rate, k_disb_date, pr_id, pr_product_kind,
           k_kolich_client_approved, :pr_code;

--      Проверка контракта на реструктуризацию
      is_restruct = null;

      select is_restruct
      from aarep_get_restructuring(:aa_k_id, :rep_date_end, 81001)
      into is_restruct;
      is_restruct = coalesce(is_restruct, 0);

      select is_secured
      from aarep_get_secured_loan(:pr_code, :aa_pr_is_indiv, :pr_product_kind, :pr_id, :k_disb_date, :k_kolich_client_approved, :aa_k_id)
      into section;

      if (k_disb_date < '28.02.2012') then
      begin
        select cs_id, cs_percent
        from classific
        where cs_day_min <= :aa_deliq_days and
              cs_day_max >= :aa_deliq_days
        into :aa_llc_id, :aa_llc_percent;

        if (aa_llc_percent is null) then
        begin
          aa_llc_percent = 0;
          aa_llc_id = 1;
        end
      end
      else
      if (k_disb_date >= '28.02.2012') then -- если выдан с 28.02.2014 то будет работать по новому положению РППУ
      begin
        aa_llc_id = 1; -- нужно с Санжаром поговорить, в какую классификацию будет входить, временно присваиваем 1
        /*
        select is_secured
        from aarep_get_secured_loan(:pr_code,:aa_pr_is_indiv,:pr_product_kind,:pr_id,:k_disb_date,:k_kolich_client_approved,:aa_k_id)
        into section;
        */
        if (section = 0) then
          section = 2; -- сделано для того чтобы раньше не обеспеченным присваивали цифру 2

        if (is_restruct = 1) then
          section = section + 2;

        kcp_perc_saldo = null;
        if (:rep_date_end >= '30.09.2014') then -- с этой даты РППУ начисляется и на проценты
        begin
          select kcp_perc_saldo
          from kontcalcperc
          where kcp_from_k_id = :aa_k_id and
                kcp_period = :rep_period
          into kcp_perc_saldo;
          if (kcp_perc_saldo is null) then
            kcp_perc_saldo = 0;
        end

        if (aa_amount_active_end = 0 and
            aa_deliq_days = 0 and
            kcp_perc_saldo > 0 and
            ks_state = 1 and
            :rep_date_end > '30.09.2014') then
        begin
          sh_date = null;
          select sh_date
          from aarep_make_zad_list(:aa_k_id, :rep_date_end, 1)
          where sh_calc_perc > 0 and
                zad_perc = 0 and
                zad_os = 0 and
                sh_repay_type = 1 and
                sh_date is not null
          order by sh_date
          rows 1
          into sh_date;

          if (sh_date is not null) then
          begin
            aa_deliq_days = :rep_date_end - :sh_date;
          end
        end

        aa_llc_percent = case
                           when aa_deliq_days = 0 then case section
                                                         when 1 then 0
                                                         when 2 then 0
                                                         when 3 then 2
                                                         when 4 then 5
                                                       end
                           when aa_deliq_days <= 30 then case section
                                                           when 1 then 1.5
                                                           when 2 then 3
                                                           when 3 then 5
                                                           when 4 then 10
                                                         end
                           when aa_deliq_days <= 60 then case section
                                                           when 1 then 7
                                                           when 2 then 10
                                                           when 3 then 20
                                                           when 4 then 30
                                                         end
                           when aa_deliq_days <= 90 then case section
                                                           when 1 then 15
                                                           when 2 then 20
                                                           when 3 then 40
                                                           when 4 then 50
                                                         end
                           when aa_deliq_days <= 120 then case section
                                                            when 1 then 30
                                                            when 2 then 40
                                                            when 3 then 60
                                                            when 4 then 70
                                                          end
                           when aa_deliq_days <= 180 then case section
                                                            when 1 then 45
                                                            when 2 then 50
                                                            when 3 then 80
                                                            when 4 then 90
                                                          end
                           when aa_deliq_days <= 270 then case section
                                                            when 1 then 60
                                                            when 2 then 65
                                                            when 3 then 95
                                                            when 4 then 100
                                                          end
                           when aa_deliq_days <= 360 then case section
                                                            when 1 then 75
                                                            when 2 then 80
                                                            when 3 then 100
                                                            when 4 then 100
                                                          end
                           else 100
                         end;
      end

      insert into llc_kont (lk_from_lm_id, lk_from_k_id, lk_deliq_days, lk_from_reg_id, lk_balance, lk_status,
                            lk_percent, lk_date, lk_date_end, lk_memo, lk_is_indiv, lk_is_signed, lk_is_changed_mannual,
                            lk_percent_orig, lk_calcperc, lk_is_secured, lk_is_restruct)
      values (:lm_id, :aa_k_id, :aa_deliq_days, :reg_id, :aa_amount_active_end, :aa_llc_id, :aa_llc_percent,
              :rep_date_end, null, '', :aa_pr_is_indiv, 1, 0, :aa_llc_percent, :kcp_perc_saldo,
              iif(:section in (1, 3), 1, 0), :is_restruct);

    end
    suspend;

  end
end
