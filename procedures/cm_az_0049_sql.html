CREATE PROCEDURE CM_AZ_0049_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable c_id id_type;
declare variable mac_from_c_id id_type;
declare variable k_id id_type;
declare variable max_continuing_deling integer;
declare variable ks_date_begin date;
declare variable k_disb_date date;
begin
  if (ma_id in(169998)) then exit;
  res_error_is_critical = 2;
  for select mac_from_c_id, mac_amount
      from miniapp, product
      inner join miniappclient on mac_from_ma_id = :ma_id
      where ma_id = :ma_id and
            ma_is_active = 1 and
            ma_date_end = '31.12.9999' and
            ma_from_pr_id = pr_id and
            mac_amount > 0 and
            pr_code containing '.akcia.'
      into :mac_from_c_id, :kc_amount
  do
  begin
    for select knc.kc_from_c_id, k.k_disb_date, k_id
        from knclient knc
        inner join kontract k on k.k_id = knc.kc_from_k_id
        where knc.kc_from_c_id = :mac_from_c_id and
              knc.kc_amount_approved > 0
        order by k.k_disb_date desc
        rows 1
        into :c_id, :k_disb_date, :k_id
    do
    begin
      max_continuing_deling = null;
      select max(ok_ol_date - ok_sh_date)
      from operkred op
      where op.ok_from_k_id = :k_id and
            op.ok_ol_is_active = 1 and
            op.ok_type in (4, 5) and
            op.ok_ol_date > ok_sh_date
      into :max_continuing_deling;
      max_continuing_deling = coalesce(max_continuing_deling, 0);
      if (max_continuing_deling >= 30) then
      begin
      -- Только повторные клиенты c просрочкой до 30ти дней могут участвовать в Акции
        res_error_text = 'aksiyada yalniz 30 günədək gecikmə ilə təkrar müştərilər iştirak edə bilər';
        suspend;
        exit;
      end
      if (exists(select ks_from_k_id
                 from knclient, kontstate
                 where kc_amount_approved > 0 and
                       kc_from_k_id = :k_id and
                       kc_from_k_id = ks_from_k_id and
                       ks_ol_is_delete = 0 and
                       ks_state = 2 and
                       ks_date_begin > cast('08.07.2021' as date))) then
      begin
      --Дата закрытия кредита должна быть не позднее 20 апреля 2021 года и на момент выдачи у клиента не должно быть активного кредита
        res_error_text = 'kreditin bağlanma tarixi 20 aprel 2021-dən gec olmamali və verildiyi zaman müştərinin aktiv krediti olmamalidir.';
        suspend;
      end
    end
  end
end
