CREATE PROCEDURE HP_KNCLIENT_PDN_VALIDATE(
    K_ID D_ID,
    TAG D_STR_CODE)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable kc_id d_id;
declare variable c_name_fio d_str_100;
declare variable eic_code d_str_code;
declare variable calc_date d_date;
declare variable history_not_found d_bool;
declare variable calc_status d_str_255;
declare variable pr_id d_integer;
declare variable not_check_pdn d_bool;
begin
  -- РФ. Процедура пересчитывает ПДН и контролирует соблюдение условий ПДН/баллы
  err_text = '';

  for select kc_id, c_name_fio, k_disb_date, k_from_pr_id
  from kontract, knclient, clients
  where k_id = :k_id
  and kc_from_k_id = k_id
  and kc_amount_approved > 0
  and kc_from_c_id = c_id
  into kc_id, c_name_fio, calc_date, pr_id
  do
  begin
    calc_date = minvalue(calc_date,  current_date);

    -- Определяем наличие кредитной истории
    eic_code = 'equifax_40';

    select history_not_found, calc_status
    from hp_extinfolist_pdn_get_eil(:kc_id,  :calc_date, 0, :eic_code)
    into history_not_found, calc_status;

    -- Если КИ не найдена, то поищем по НБКИ
    if (history_not_found = 1 or calc_status = 'Полученная история БКИ устарела') then
    begin
      eic_code = 'nbki_40';
      select history_not_found, calc_status
      from hp_extinfolist_pdn_get_eil(:kc_id, :calc_date, 0, :eic_code)
      into history_not_found, calc_status;
    end

    if (history_not_found = 1 or calc_status = 'Полученная история БКИ устарела') then
    begin
      err_text = f_format('%1. Кредитная история не получена. Дождитесь расчета ПДН', c_name_fio);
      suspend;
      exit;
    end
  
    execute procedure hp_knclient_pdn_update(:kc_id, 0, null, null, null, null, null, tag);
  
    if (exists(select * from product where pr_id = :pr_id
    and pr_code containing '.reliable_client.')) then
    begin
      -- RFCRU-2617 Если стоит галка "документально подтвержденный доход", то лимиты по ПДН не считаем до КК
      not_check_pdn = 0;
  
      select 1
      from kontract, miniappclient, miniappclient_income
      where k_id = :k_id
        and mac_from_ma_id = k_from_ma_app_id
        and maci_from_mac_id = mac_id
        and maci_is_documented = 1
      rows 1
      into not_check_pdn;
  
      if (not_check_pdn = 1) then
        exit;
      for select error_text
      from knclient_pdn_limit_check(:kc_id)
      into err_text
      do
        suspend;
        exit;
    end

  end
  suspend;
end
