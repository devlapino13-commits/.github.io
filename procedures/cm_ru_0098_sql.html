CREATE PROCEDURE CM_RU_0098_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare variable c_id id_type;
declare variable k_disb_date date;
declare variable re_kredline_end date;
declare variable par_re_disb_date date;
declare variable kc_circle integer;
declare variable cl_par_max_delinq_day integer;
declare variable cl_par_sum_delinq_day integer;
declare variable max_delinq_day integer;
declare variable sum_delinq_day integer;
declare variable pay_qty integer;
declare variable k_id_active id_type;
begin
  for select kc_from_c_id, kont.k_disb_date
      from knclient kc, kontract kont, product
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      and kc_from_k_id = k_id
      and k_from_pr_id = pr_id
      and pr_code not containing '.PAR.'
      and k_refund = 0
  into :c_id, :k_disb_date
  do
  begin  
   -- if (exists (
    for select kc_from_c_id
    from knclient kc, kontract kont, kontstate ks, product pr
    where ks.ks_from_k_id = kont.k_id
    and kc_from_k_id = ks.ks_from_k_id
    and k_from_pr_id = pr_id
    and kc_amount_approved > 0
    and ks.ks_ol_is_delete = 0
    and :k_disb_date between ks.ks_date_begin and ks.ks_date_end
    and ks.ks_state = 1
    and kc.kc_from_c_id = :c_id
    and pr_code not containing '.PAR.'
    and kont.k_id <> :k_id
    into :k_id_active
    do
               -- )) then
    begin
      pay_qty = 0;
      select count(*)
      from aarep_make_zad_list(:k_id_active, :k_disb_date, 1)
      into :pay_qty;
      pay_qty = coalesce(pay_qty, 0);
      if (pay_qty > 1) then
      begin
          res_info_text = '';
    
          select c_name_fio
          from clients
          where c_id = :c_id
          into res_info_text;
    
          res_error_text = 'У клиента есть активный заем, отправка на кредитный комитет запрещена' ;
          res_error_is_critical = 1;
          suspend; 
       end
    end
  end
end
