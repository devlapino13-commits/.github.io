CREATE PROCEDURE UXP$UXFRCOMBO_UPDATE(
    UFP_UFR_UUID D_UUID,
    UFP_UF_UUID D_UUID,
    FIELD_NAME D_STR_100,
    HIDE_TEXT D_BOOL)
AS
declare variable uf_uuid d_uuid;
declare variable us_code d_str_code;
declare variable uf_field_name d_str_code;
declare variable ufp_uuid d_uuid;
begin
  if (coalesce(field_name, '') = '') then
    uf_uuid = ufp_uf_uuid;
  else
    uf_uuid = uxf$uniq(field_name);

  select us_code, uf_field_name
  from ux$field, ux$source
  where uf_uuid = :uf_uuid and
        uf_owner = us_uuid
  into us_code, uf_field_name;

  select ufp_uuid
  from ux$frparam
  where ufp_ufr_uuid = :ufp_ufr_uuid and
        ufp_uf_uuid = :ufp_uf_uuid
  into ufp_uuid;

  if (ufp_uuid is null) then
  begin
    insert into ux$frparam (ufp_ufr_uuid, ufp_uf_uuid, ufp_editor)
    values (:ufp_ufr_uuid, :ufp_uf_uuid, 'combobox')
    returning ufp_uuid
    into ufp_uuid;
  end

  update or insert into ux$frcombo (ufc_ufp_uuid, ufc_combo_source, ufc_combo_field, ufc_hide_text)
  values (:ufp_uuid, :us_code, :uf_field_name, :hide_text)
  matching (ufc_ufp_uuid);

  update ux$frparam
  set ufp_editor = 'combobox'
  where ufp_uuid = :ufp_uuid and
        ufp_editor <> 'combobox';
end
