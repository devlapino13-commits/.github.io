CREATE PROCEDURE HP_CONTDATA_GET_EMAIL_VALID(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0))
RETURNS (
    CONT_VALUE VARCHAR(255))
AS
begin
  -- упрощенная процедура, возвращает только один актуальный email
  -- Приоритетно берется последний веденный, если такого нет, подтвержденный
