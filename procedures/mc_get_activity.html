CREATE PROCEDURE MC_GET_ACTIVITY(
    MCL_ID D_ID)
RETURNS (
    SCORE D_SMALLINT,
    BECOME_PARTNER_ACTIVE D_BOOL,
    SEND_FRIEND_ACTIVE D_BOOL,
    COMBO_ACTIVE D_BOOL,
    ACCUMULATE_ACTIVE D_BOOL,
    EXCHANGE_ACTIVE D_BOOL)
AS
declare variable date_end d_date;
declare variable c_id d_id;
declare variable show_client_link d_smallint;
declare variable close_date d_date;
declare variable last_k_id d_id;
declare variable rd_id d_integer;
declare variable ag_id d_integer;
declare variable ae_is_exchange d_bool;
begin
  /* Процедура для партнерской программы, возращается активности button'нов */
  date_end = current_date;
  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :c_id;

  -- Условия участия в партнерской программе
  -- Клиент должен быть активным или когда либо бравший кредит
  show_client_link = 1;
  close_date = null;
  select ks_date_end, ks_from_k_id from kontstate, knclient
  where ks_ol_is_delete = 0
  and ks_state = 1
  and ks_from_k_id = kc_from_k_id
  and kc_amount_approved > 0
  and kc_from_c_id = :c_id
  order by ks_date_end desc
  rows 1
  into close_date, last_k_id;
  if (close_date is null) then show_client_link = 0; -- если не имеется кр. история то отменяется
  else
  if (dateadd(month, -6, :date_end) > :close_date) then show_client_link = 0; -- если после закрытия последнего контракта прошло более 6 мес.
  else
  if (exists(select * from moratory
    where m_is_black = 2
    and m_is_active = 1
    and m_from_c_id = :c_id)) then show_client_link = 0; -- если в Стоп-листе
  else
  if (exists(select * from oksummary
      where oks_from_k_id = :last_k_id
      and oks_close_state = 0
      and ((oks_delinq_day > 30 and oks_date_begin <= cast(:date_end as date))
           or (:date_end between oks_date_begin and oks_date_end
               and cast(:date_end as date) - oks_delinq_date > 30))
      )) then show_client_link = 0;
  become_partner_active = show_client_link;


  select rd_id from refdata
  where rd_code = 'AgreementKinds.loyal_agreement'
  into rd_id;

  ag_id = null;
  select ag_id from agreements -- в этом отрезке нужно добавить дополнительное условие на mcl_id
  where ag_object_kind = 0
  and ag_object_id = :c_id
  and ag_status = 1
  and ag_from_rd_id = :rd_id
  and :date_end between ag_date_begin and coalesce(ag_date_end,'31.12.9999')
  order by ag_id desc
  rows 1
  into ag_id;

  send_friend_active = 0; -- Кнопка “Отправить другу”

  score = 0; --балл клиента
  ae_is_exchange = 0; -- обменять баллы или нет
  if (ag_id is not null) then
  begin

    send_friend_active = 1;
    -- если кнопка отправить другу активная то стать партнером не должно быть активным так как уже имеется партнерский договор
    become_partner_active = 0;

    select sum(ogr_amount * ogr_sign * f_loyal(ogr_from_ol_id, :date_end) + ogr_tax_fl * ogr_sign * f_loyal(ogr_from_ol_id, :date_end)) -- Остаток накопленных баллов
    from operagr
    where ogr_ol_date <= :date_end and
          ogr_ol_is_active = 1 and
          ogr_from_ag_id = :ag_id
    into score;
    if (score is null) then score = 0;

    select ae_is_exchange from agreement_exchange
    where ae_from_ag_id = :ag_id
    into ae_is_exchange;
    if (ae_is_exchange is null) then ae_is_exchange = 0;
  end

  if (ae_is_exchange = 0) then
  begin
    accumulate_active = 0;
    exchange_active = 1;
  end else
  begin
    accumulate_active = 1;
    exchange_active = 0;
  end 

  -- Активность комбо-бокса
  -- Если баллы 1200 и более после этого могут обменять (обналичить)
  -- Минимальный порог получения права на выплату заработанных баллов 1200, RFCRU-6095
  if (score >= 1200 and
      extract(day from current_date) between 1 and 9) then
    combo_active = 1;
  else
    combo_active = 0;

  suspend;
end
