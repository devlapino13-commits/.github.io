CREATE PROCEDURE HP_CARDEMBOMAIN_PRINT(
    CEM_ID D_INTEGER)
RETURNS (
    RES_ID D_INTEGER,
    RES_DATE D_DATE,
    RES_OUTNUM D_STR_20,
    RES_KIND D_STR_CODE,
    RES_COMPANY_NAME D_STR_255,
    RES_OWN_NAME D_STR_255,
    RES_APP_COUNT D_INTEGER,
    RES_IS_PINCOVER D_BOOL,
    RES_FILENAME D_STR_100)
AS
declare variable sdt d_str_20;
begin
  res_company_name =
    f_company_name('short', 'rus');
  for
    select
      cem_id,
      cem_date,
      ckd_name,
      iif(ckd_is_own = 1, :res_company_name, 'МПЦ'),
      cem_app_count,
      cem_is_pincover,
      cem_filename
    from cardembomain
      inner join cardkind
        on ckd_id = cem_from_ckd_id
    where cem_id = :cem_id
      and cem_from_ces_id >= 3
  into
    :res_id,
    :res_date,
    :res_kind,
    :res_own_name,
    :res_app_count,
    :res_is_pincover,
    :res_filename
  do
  begin
    sdt = f_datetostr(res_date);
    res_outnum = res_id || '-' || substring(sdt from 4 for 2) || '-' || right(sdt, 2);
    suspend;
  end
end
