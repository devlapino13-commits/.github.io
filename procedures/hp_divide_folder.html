CREATE PROCEDURE HP_DIVIDE_FOLDER(
    REP_K_ID ID_TYPE)
RETURNS (
    NEW_FOLDER_ID ID_TYPE,
    OLD_FOLDER_ID ID_TYPE)
AS
begin
  new_folder_id = (select result
                   from get_kv_id(1));
  insert into kfolder (kf_id, kf_type, kf_memo, is_delete, kf_is_yur, kf_is_new)
  values (:new_folder_id, 0, 'Разделенная папка', 0, 0, 0);

  select k_from_kf_id
  from kontract
  where k_id = :rep_k_id
  into old_folder_id;

  insert into knaktlist (kal_from_ka_id, kal_number, kal_from_kf_id, kal_from_k_id, kal_date, kal_date_end,
                         kal_new_kf_officer, kal_new_kf_name, is_delete, kal_new_reg_id, kal_new_no,
                         kal_new_officer_code)
  select kal_from_ka_id, kal_number, :new_folder_id, kal_from_k_id, kal_date, kal_date_end, kal_new_kf_officer,
         kal_new_kf_name, is_delete, kal_new_reg_id, kal_new_no, kal_new_officer_code
  from knaktlist
  where kal_from_kf_id = :old_folder_id;

  update kontract
  set k_from_kf_id = :new_folder_id
  where k_id = :rep_k_id;

  suspend;
end
