CREATE PROCEDURE HP_UXFIELD_TYPE(
    FIELD_TYPE D_INTEGER,
    FIELD_SIZE D_INTEGER,
    FIELD_SCALE D_INTEGER,
    FIELD_SUB_TYPE D_INTEGER)
RETURNS (
    RES_FIELD_TYPE D_STR_CODE,
    RES_FIELD_SIZE D_INTEGER)
AS
begin
  res_field_size = field_size;

  if (field_type = 7) then
    res_field_type = 'smallint';
  if (field_type = 8) then
    res_field_type = 'integer';
  if (field_type = 10) then
    res_field_type = 'float';
  if (field_type = 12) then
    res_field_type = 'date';
  if (field_type = 16) then
  begin
    if (field_scale = 0) then
      res_field_type = 'bigint';
    else
    if (field_scale <= -6) then
      res_field_type = 'percent';
    else
      res_field_type = 'currency';
  end
  if (field_type = 27) then
    res_field_type = 'double';
  if (field_type = 35) then
    res_field_type = 'datetime';
  if (field_type = 37) then
  begin
    res_field_type = 'string';
    res_field_size = res_field_size / 4;
  end
  if (field_type = 261) then
  begin
    if (field_sub_type = 1) then
      res_field_type = 'memo';
    else
      res_field_type = 'blob';
  end

  suspend;
end
