CREATE PROCEDURE HP_EXTINFOLIST_KK_FULL(
    C_ID D_ID,
    IS_READONLY D_SMALLINT)
RETURNS (
    LINE_ID D_INTEGER,
    LINE_PARENT_ID D_INTEGER,
    EIL_ID D_ID,
    EIL_FROM_C_ID D_ID,
    EIL_FROM_EIC_ID D_ID,
    EIL_DATE D_DATE,
    EIL_OBJECT_KIND D_INTEGER,
    EIL_OBJECT_ID D_ID,
    EIL_EXT_ID D_STR_CODE,
    EIL_IS_JSON D_SMALLINT,
    EIL_FROM_BLS_ID D_ID,
    EIL_STATUS D_SMALLINT,
    EIL_CREATE_DATETIME D_DATETIME,
    EIL_STATUS_DATETIME D_DATETIME,
    EIL_KK_MEMO_ID D_ID,
    EIL_KK_UN_ID D_ID,
    EIL_KK_RESULT D_SMALLINT,
    EIL_KK_DATETIME D_DATETIME,
    EIC_NAME D_STR_255,
    EIC_HAS_API D_SMALLINT,
    EIL_KK_MEMO_STR D_STR_LONG,
    EIL_HAS_BLOB D_SMALLINT,
    EIL_STATUS_STR D_STR_CODE,
    EIC_NEED_KK_RESULT D_SMALLINT,
    EIL_IS_OLD_INFO D_SMALLINT,
    EIL_ERROR_TEXT D_STR_8100,
    EIL_UN_FIO D_STR_100,
    INF_MEMO D_BLOB_TEXT,
    ACM_MEMO D_BLOB_TEXT,
    SFT_MEMO D_BLOB_TEXT,
    RES_REQUEST_BLS_ID D_ID,
    RES_RESPONSE_BLS_ID D_ID)
AS
declare variable kind_id d_integer;
begin
  for select line_id, line_parent_id, eil_id, eil_from_c_id, eil_from_eic_id, eil_date, eil_object_kind, eil_object_id,
             eil_ext_id, eil_is_json, eil_from_bls_id, eil_status, eil_create_datetime, eil_status_datetime,
             eil_kk_memo_id, eil_kk_un_id, eil_kk_result, eil_kk_datetime, eic_name, eic_has_api, eil_kk_memo_str,
             eil_has_blob, eil_status_str, eic_need_kk_result, eil_is_old_info, eil_error_text, eil_un_fio
      from hp_extinfolist_full(:c_id, :is_readonly)
      into :line_id, :line_parent_id, :eil_id, :eil_from_c_id, :eil_from_eic_id, :eil_date, :eil_object_kind,
           :eil_object_id, :eil_ext_id, :eil_is_json, :eil_from_bls_id, :eil_status, :eil_create_datetime,
           :eil_status_datetime, :eil_kk_memo_id, :eil_kk_un_id, :eil_kk_result, :eil_kk_datetime, :eic_name,
           :eic_has_api, :eil_kk_memo_str, :eil_has_blob, :eil_status_str, :eic_need_kk_result, :eil_is_old_info,
           :eil_error_text, :eil_un_fio
  do
  begin
    inf_memo = '';
    acm_memo = '';
    sft_memo = '';
    res_response_bls_id = 0;
    res_request_bls_id = 0;

    select inf_memo, acm_memo, sft_memo
    from hp_get_client_verify_info(:c_id, 0)
    into inf_memo, acm_memo, sft_memo;

    if (coalesce(eil_id, 0) <> 0) then
    begin
      -- XML-отчет
      kind_id = f_enum_id('extinfo_bls_kind', 'equifax_xml_response');
  
      select eib_from_bls_id
      from extinfolistbls
      where eib_from_eil_id = :eil_id
        and eib_kind = :kind_id
      into res_response_bls_id;

      -- XML-запрос
      kind_id = f_enum_id('extinfo_bls_kind', 'equifax_xml_request');
  
      select eib_from_bls_id
      from extinfolistbls
      where eib_from_eil_id = :eil_id
        and eib_kind = :kind_id
      into res_request_bls_id;
    end

    suspend;
  end
end
