CREATE PROCEDURE FIND_DOVLIST_OWN(
    USER_ID DECIMAL(18,0),
    DATE_END DATE)
RETURNS (
    DL_ID INTEGER,
    DL_DATE_BEGIN DATE,
    DL_DATE_END DATE,
    DL_DATE_END_FACT DATE,
    DL_FROM_DC_ID INTEGER,
    DL_FROM_BLN_ID INTEGER,
    DL_DOV_NO VARCHAR(15),
    DL_FROM_C_ID DECIMAL(18,0),
    DL_MANAGER_ID DECIMAL(18,0),
    DL_FROM_REG_ID INTEGER,
    DL_MAIN_TEXT VARCHAR(255),
    DL_CITY VARCHAR(255),
    DL_MEMO_1 VARCHAR(255),
    DL_MEMO_2 VARCHAR(255),
    DL_MEMO_3 VARCHAR(255),
    DL_MEMO_4 VARCHAR(255),
    DL_MEMO_5 VARCHAR(255),
    DL_MEMO_6 VARCHAR(255),
    DL_MEMO_7 VARCHAR(255),
    DL_MEMO_8 VARCHAR(255),
    DL_MEMO_9 VARCHAR(255),
    DL_MEMO_10 VARCHAR(255),
    DL_STATUS INTEGER,
    DL_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    DL_FROM_UN_ID DECIMAL(18,0),
    DL_DATETIME TIMESTAMP,
    DL_BLOB_NAME VARCHAR(255),
    DC_NAME VARCHAR(255),
    C_NAME_LIST VARCHAR(255),
    DL_STATUS_TEXT VARCHAR(50),
    DL_HAS_BLOB INTEGER)
AS
declare variable dl_dov_no_2 integer;
declare variable c_name_fio varchar(100);
declare variable c_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  c_id = null;
  select un_from_c_id
  from username
  where un_id = :user_id
  into :c_id;
  if (c_id is not null) then
  begin
    for select distinct dl_id, dl_date_begin, dl_date_end, dl_date_end_fact, dl_from_dc_id, dl_from_bln_id, dl_dov_no_1,
                        dl_dov_no_2, dl_from_c_id, dl_manager_id, dl_from_reg_id, dl_main_text, dl_city, dl_memo_1,
                        dl_memo_2, dl_memo_3, dl_memo_4, dl_memo_5, dl_memo_6, dl_memo_7, dl_memo_8, dl_memo_9,
                        dl_memo_10, dl_status, dl_blob, dl_from_un_id, dl_datetime, dl_blob_name, dc_name
        from dovlist, dovcateg, dovclients
        where dl_from_dc_id = dc_id and
              dvc_from_dl_id = dl_id and
              dvc_from_c_id = :c_id
        order by dl_date_begin desc
        into :dl_id, :dl_date_begin, :dl_date_end, :dl_date_end_fact, :dl_from_dc_id, :dl_from_bln_id, :dl_dov_no,
             :dl_dov_no_2, :dl_from_c_id, :dl_manager_id, :dl_from_reg_id, :dl_main_text, :dl_city, :dl_memo_1,
             :dl_memo_2, :dl_memo_3, :dl_memo_4, :dl_memo_5, :dl_memo_6, :dl_memo_7, :dl_memo_8, :dl_memo_9,
             :dl_memo_10, :dl_status, :dl_blob, :dl_from_un_id, :dl_datetime, :dl_blob_name, :dc_name
    do
    begin
      /* № договора */
      if (dl_dov_no_2 <> 0) then
        dl_dov_no = dl_dov_no || '/' || dl_dov_no_2;
      /* ФИО получателей */
      c_name_list = '';
      for select c_name_fio
          from clients, dovclients
          where c_id = dvc_from_c_id and
                dvc_from_dl_id = :dl_id
          into :c_name_fio
      do
      begin
        if (char_length(c_name_list) + char_length(c_name_fio) <= 253) then
        begin
          if (c_name_list <> '') then
            c_name_list = c_name_list || ', ' || c_name_fio;
          else
            c_name_list = c_name_fio;
        end
      end
      /* Статус заявки */
      if (dl_status = -1) then
        dl_status_text = 'Ануллированная';
      else
      if (dl_status = 0) then
        dl_status_text = 'В процессе';
      else
      begin
        if ((dl_date_end_fact is null) or (dl_date_end_fact is not null and
            dl_date_end_fact > dl_date_end)) then
        begin
          if (dl_date_end >= :date_end) then
            dl_status_text = 'Действующая';
          else
            dl_status_text = 'Истекщая';
        end
        else
        begin
          if (dl_date_end_fact >= :date_end) then
            dl_status_text = 'Действующая';
          else
            dl_status_text = 'Отозванная';
        end
      end
      if (dl_date_end_fact is not null and
          dl_date_end_fact < dl_date_end) then
        dl_date_end = dl_date_end_fact;
      if (dl_blob is not null) then
        dl_has_blob = 1;
      else
        dl_has_blob = 0;
      suspend;
    end
  end
end
