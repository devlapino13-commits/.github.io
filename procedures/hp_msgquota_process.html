CREATE PROCEDURE HP_MSGQUOTA_PROCESS(
    MQC_CODE D_STR_CODE NOT NULL,
    MQL_CONT_CODE D_STR_CODE)
RETURNS (
    QUOTA_EXCEEDED D_BOOL,
    QUOTA_CODE D_STR_CODE)
AS
declare variable mqc_id d_integer;
declare variable mqc_is_active d_bool;
declare variable start_datetime d_datetime;
declare variable mql_count d_integer;
declare variable mqt_interval d_integer;
declare variable mqt_max_count d_integer;
declare variable mqt_is_common d_bool;
declare variable max_interval d_integer;
declare variable is_whitelist d_bool;
declare variable mqt_add_limit d_smallint;
begin
  -- Процедура обеспечивает обработку квот (лимитов) отправки сообщений

  quota_exceeded = 0;
  quota_code = '';
  mql_cont_code = coalesce(mql_cont_code, '');

  select mqc_id, mqc_is_active
  from msgquotacfg
  where mqc_code = :mqc_code
  into mqc_id, mqc_is_active;

  -- Если кодовое имя квоты передается в процедуру, оно обязано быть и в msgquotacfg
  if (mqc_id is null) then
    exception e_default f_format('mqc_code:%1 not found in table msgquotacfg.', mqc_code);

  -- Допустимо, чтобы конфигурация была отключена, тогда считаем что квота соблюдена
  if (mqc_is_active = 0) then
  begin
    suspend;
    exit;
  end

  -- Проверяем контакт по списку исключений
  is_whitelist = 0;
  if (exists (select *
    from msgquotaexclude
    where mqe_cont_code = :mql_cont_code)) then
    is_whitelist = 1;

  max_interval = 0;
  -- Проходим по всем лимитам текущей конфигурации
  for select mqt_interval, mqt_max_count, mqt_is_common, mqt_add_limit
      from msgquotatimer
      where mqt_from_mqc_id = :mqc_id
        and mqt_is_active = 1
      into mqt_interval, mqt_max_count, mqt_is_common, mqt_add_limit
  do
  begin
    if (max_interval < mqt_interval) then
      max_interval = mqt_interval;

    -- Вычисляем начало интервала для подсчета количества сообщений
    start_datetime = dateadd(second, -mqt_interval * 60, current_timestamp);

    -- Если учет внутри текущих суток, корректируем начало интервала
    if (mqt_add_limit = 1 and start_datetime < current_date) then
      start_datetime = current_date;

    -- Выясняем количество сообщений подпадающих под наш интервал
    if (mqt_is_common = 1) then
      select count(*)
      from msgquotalist
        left join msgquotaexclude on mqe_cont_code = mql_cont_code
      where mql_from_mqc_id = :mqc_id and
        mql_datetime >= :start_datetime and
        (
          (:mqt_add_limit = 0 and mqe_cont_code is null) or -- Считаем всех кроме номеров из списка исключений
          (:mqt_add_limit = 2 and mqe_cont_code is not null) -- Считаем только номера из списка исключений
        )
      into mql_count;
    else
      select count(*)
      from msgquotalist
        left join msgquotaexclude on mqe_cont_code = mql_cont_code
      where mql_from_mqc_id = :mqc_id and
        mql_datetime >= :start_datetime and
        mql_cont_code = :mql_cont_code and
        mqe_cont_code is null
      into mql_count;

    -- При превышении выделенной квоты выходим
    if (mql_count >= mqt_max_count and (is_whitelist = 0 or mqt_add_limit = 2)) then
    begin
      quota_exceeded = 1;
      quota_code = f_format('i%1t%2c%3f%4', mqc_id, mqt_interval, mqt_max_count, mql_count);
      suspend;
      exit;
    end
  end

  -- Проверки пройдены, логируем сообщение
  insert into msgquotalist (mql_from_mqc_id, mql_cont_code)
  values (:mqc_id, :mql_cont_code);

  -- Очистка сообщений, которые вышли за максимальный интервал. Т.е. неактуальны

  -- Выбираем конфигурации, для которых наступило время очистки
  for select mqc_id
      from msgquotacfg
      where dateadd(minute, mqc_cleanup_interval, mqc_cleanup_datetime) < current_timestamp
      into mqc_id
  do
  begin
    -- Удаляем сообщения, которые вышли за границы макс. интервала
    start_datetime = dateadd(second, -max_interval * 60, current_timestamp);
    delete from msgquotalist
    where mql_from_mqc_id = :mqc_id and
      mql_datetime <= :start_datetime;

    -- Обновляем время последней очистки
    update msgquotacfg
    set mqc_cleanup_datetime = current_timestamp
    where mqc_id = :mqc_id;

  when any do
    begin
    -- Тут по плану будут только ошибки блокировки.
    end
  end

  suspend;
end
