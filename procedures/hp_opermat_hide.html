Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_OPERMAT_HIDE(
    OM_FROM_OL_ID D_ID,
    OM_NUMBER D_INTEGER)
RETURNS (
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_BOOL)
AS
begin
  -- добавляет товар в opermat_hide, далее этого товара не будет в счет-фактуре по ИПФ при распечатке (RFCKG-4643)
  -- если надо вернуть товар (снова покзывать в счет-фактуре), то достаточно удалить его из opermat_hide

  res_is_error = 0;
  res_text = '';

  begin
    insert into opermat_hide(omh_ol_id, omh_number)
    values(:om_from_ol_id, :om_number);

    when any do
    begin
      res_is_error = 1;
      res_text = 'При скрытии товара произошла ошибка (hp_opermat_hide)';
      suspend;
      exit;
    end
  end

  suspend;
end
