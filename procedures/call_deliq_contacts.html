CREATE PROCEDURE CALL_DELIQ_CONTACTS(
    DELIQ_START_DATE DATE,
    K_ID DECIMAL(18,0),
    REP_DATE_END DATE,
    CONT_TYPE INTEGER)
RETURNS (
    CCDL_ID DECIMAL(18,0),
    DEBTOR_NAME VARCHAR(255),
    FIO VARCHAR(255),
    IS_DEBTOR SMALLINT,
    CALL_RES SMALLINT)
AS
declare variable contacts_id integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /*
        CREATED BY IGUSEV 17.07.2012
        PURPOSE: Для "аналитического отчета по звонкам должникам".
                 Предоставляет список респондентов (должников + конт. лиц) в разрезе просрочки
    */
  for select cdl.ccdl_id, cl.c_name_fio as debtor_name, coalesce(ccdl_from_contacts_id, 0) contacts_id,
             (
             case
               when ccdl_from_contacts_id is null then cl.c_name_fio
               else cont.cc_name
             end) as fio,
             (
             case
               when ccdl_from_contacts_id is null then 1
               else 0
             end) as is_debtor
      from callcenter_debtors_list cdl
      inner join clients cl on cl.c_id = cdl.ccdl_from_c_id
      left join contacts cont on cont.cc_id = cdl.ccdl_from_contacts_id
      where cdl.ccdl_deliq_start_date = :deliq_start_date and
            cdl.ccdl_from_k_id = :k_id and
            (((ccdl_from_contacts_id is null) and
            (:cont_type = 2)) or
            ((ccdl_from_contacts_id is not null) and
            (:cont_type = 1)) or
            (:cont_type = 0)) and
            cdl.ccdl_create_date <= :rep_date_end
      order by cl.c_name_fio, contacts_id, fio
      into :ccdl_id, :debtor_name, :contacts_id, :fio, :is_debtor
  do
  begin
    call_res = null;

    select cdr.ccdr_result
    from callcenter_debtors_result cdr
    where cdr.ccdr_call_datetime <= :rep_date_end + 1 and
          cdr.ccdr_from_ccdl_id = :ccdl_id
    order by cdr.ccdr_call_datetime desc
    rows 1
    into :call_res;

    suspend;
  end
end
