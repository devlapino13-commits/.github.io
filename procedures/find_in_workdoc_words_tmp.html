CREATE PROCEDURE FIND_IN_WORKDOC_WORDS_TMP(
    FIND_TABLE VARCHAR(100),
    FIND_TXT VARCHAR(255))
RETURNS (
    WDLINK_OBJECT_ID VARCHAR(20),
    WDLINK_FOUND_COUNT INTEGER)
AS
declare variable wdw_id integer;
declare variable k integer;
declare variable txt_tmp varchar(255);
declare variable object_kind integer;
begin
  if (find_table = 'paycode') then
    object_kind = 1;
  if (find_table = 'kassplan') then
    object_kind = 2;
  if (find_table = 'acc_list') then
    object_kind = 3;
  if (find_table = 'balgroup') then
    object_kind = 4;
  if (find_table = 'currlist') then
    object_kind = 5;
  if (find_table = 'banks') then
    object_kind = 6;
  if (find_table = 'places') then
    object_kind = 7;
  wdlink_found_count = 1;

  if (:find_txt <> '') then
  begin
    find_txt = (select res_txt
                from hp_clear_symbols(:find_txt));
    wdlink_found_count = 1;
    while (:find_txt <> '') do
    begin
      k = position(' ', :find_txt) - 1;
      if (k >= 0) then
      begin
        txt_tmp = left(:find_txt, k);
        find_txt = trim(substring(:find_txt from k + 1 for 255));
      end
      else
      begin
        txt_tmp = :find_txt;
        find_txt = '';
      end
      if (char_length(txt_tmp) >= 2) then
      begin
        for select distinct wdlink_object_id
            from workdocword, workdoclink
            where wdw_word starting with :txt_tmp and
                  wdlink_object_kind = :object_kind and
                  wdlink_from_wdw_id = wdw_id
            into :wdlink_object_id
        do
        begin
          suspend;
        end
      end
    end
  end
end
