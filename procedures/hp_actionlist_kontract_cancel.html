CREATE PROCEDURE HP_ACTIONLIST_KONTRACT_CANCEL(
    K_ID D_ID,
    UN_ID D_ID,
    KR_ID D_INTEGER)
AS
declare variable was_error d_bool;
declare variable ma_id d_integer;
begin
  -- Возвращаем договор в заявку
  select res_need_rollback
  from hp_kontract_set_status(:k_id, -2, 1, :un_id)
  into was_error;

  if (was_error = 0) then
  begin
    select k_from_ma_app_id
    from kontract
    where k_id = :k_id
    into ma_id;
    if (ma_id is not null) then
    begin
      execute procedure hp_miniapp_status_change (:ma_id, :un_id, -1);
      insert into kontreslist(
        krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
        krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
      values (:kr_id, :un_id, null, 0, :ma_id, current_timestamp, current_date,
        current_date, 0);
    end
  end
end
