CREATE PROCEDURE AAREP_OBORSALDO_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    BG_ID VARCHAR(20),
    BG_NAME VARCHAR(255),
    BG_TYPE VARCHAR(255),
    BG_SALDO_BEGIN DECIMAL(18,4),
    BG_OBOROT_DT DECIMAL(18,4),
    BG_OBOROT_KT DECIMAL(18,4),
    BG_SALDO_END DECIMAL(18,4))
AS
declare variable tmp_saldo_begin decimal(18,4) = 0;
declare variable tmp_saldo_end decimal(18,4) = 0;
declare variable tmp_oborot_dt decimal(18,4) = 0;
declare variable tmp_oborot_kt decimal(18,4) = 0;
declare variable bg_kind integer;
declare variable al_id varchar(20);
begin
  for select id_list
      from balgroup_list('00', 0, 3, 1), balgroup
      where id_list = bg_id and
            not(bg_id in ('25170', '45170'))
      into bg_id
  do
  begin
    bg_saldo_begin = 0.0;
    bg_oborot_dt = 0.0;
    bg_oborot_kt = 0.0;
    bg_saldo_end = 0.0;
    for select al_id
        from acc_list, region_list(:rep_region_id, 1, -1, 1)
        where al_from_bg_id = :bg_id and
              al_from_reg_id = id_list
        into :al_id
    do
    begin
      tmp_saldo_end = 0;
      tmp_oborot_dt = 0;
      tmp_oborot_kt = 0;

      tmp_saldo_begin = f_al_bal(al_id, rep_date_begin - 1);
      tmp_saldo_end = f_al_bal(al_id, rep_date_end);

      select dt_ekv, kt_ekv
      from hp_al_obor(:al_id, :rep_date_begin, :rep_date_end)
      into :tmp_oborot_dt, :tmp_oborot_kt;

      bg_saldo_begin = bg_saldo_begin + tmp_saldo_begin;
      bg_oborot_dt = bg_oborot_dt + tmp_oborot_dt;
      bg_oborot_kt = bg_oborot_kt + tmp_oborot_kt;
      bg_saldo_end = bg_saldo_end + tmp_saldo_end;
    end

    select bg_name, bg_kind
    from balgroup
    where bg_id = :bg_id
    into :bg_name, bg_kind;

    if (bg_kind = 1) then
      bg_type = '1';
    if (bg_kind = 2) then
      bg_type = '2';
    if (bg_kind = 3) then
      bg_type = '2';
    if (bg_kind = 4) then
      bg_type = '4';
    if (bg_kind = 5) then
      bg_type = '5';
    if (bg_kind = 6) then
      bg_type = '6';

    if (bg_kind in (2, 3, 4)) then
    begin
      bg_saldo_begin = -bg_saldo_begin;
      bg_saldo_end = -bg_saldo_end;
    end
    if (bg_saldo_begin <> 0.0 or bg_saldo_end <> 0.0 or bg_oborot_dt <> 0.0 or bg_oborot_kt <> 0.0) then
      suspend;
  end
end
