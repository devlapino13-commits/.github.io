CREATE PROCEDURE AAREP_CALC_AMORT_KOMMIS_TEST(
    KONT_ID DECIMAL(18,0),
    DATE_BEGIN DATE,
    DATE_END DATE)
RETURNS (
    RES_AMOUNT_BEGIN DECIMAL(18,4),
    RES_AMOUNT_END DECIMAL(18,4),
    RES_AMOUNT_PERIOD DECIMAL(18,4),
    RES_DISB_DATE DATE,
    RES_LAST_DATE DATE,
    RES_KRED_AMOUNT DECIMAL(18,4),
    RES_KOMMIS_AMOUNT DECIMAL(18,4),
    RES_PERC_DAILY DECIMAL(18,9),
    RES_KOMMIS_PAYED DECIMAL(18,4))
AS
declare variable is_closed_end integer;
declare variable day_all integer;
declare variable ttt decimal(18,4);
declare variable day_period integer;
declare variable dd_temp date;
declare variable is_closed_begin integer;
declare variable sh_last_date date;
declare variable sh_kommis_tmp decimal(18,4);
declare variable sh_date_tmp date;
declare variable sh_os_tmp decimal(18,4);
declare variable res_last_date_b date;
declare variable res_kommis_last_b decimal(18,4);
declare variable res_last_date_e date;
declare variable res_kommis_last_e decimal(18,4);
declare variable res_next_date_b date;
declare variable res_next_amount_b decimal(18,4);
declare variable res_next_date_e date;
declare variable res_next_amount_e decimal(18,4);
begin
  is_closed_end = 0;
  is_closed_begin = 0;
  res_amount_begin = 0;
  res_amount_end = 0;
  res_kommis_payed = 0;
  res_amount_period = 0;
  res_perc_daily = 0;
  res_kred_amount = 0;
  res_disb_date = null;
  sh_last_date = null;
  res_kommis_last_b = 0;
  res_last_date_b = null;
  res_next_amount_b = null; -- должен быть именно Null, а не 0
  res_next_date_b = null;
  res_kommis_amount = 0;

  res_kommis_last_e = 0;
  res_last_date_e = null;

  res_next_date_e = null;
  res_next_amount_e = null; -- должен быть именно Null, а не 0

  for select sh_kommis_monthly, sh_date, sh_os
      from schedule
      where sh_from_k_id = :kont_id
      order by sh_date
      into :sh_kommis_tmp, :sh_date_tmp, :sh_os_tmp
  do
  begin
    res_kommis_amount = res_kommis_amount + sh_kommis_tmp; -- сумма выдачи
    res_kred_amount = res_kred_amount + sh_os_tmp; -- сумма выдачи
    if (res_disb_date is null) then
      res_disb_date = :sh_date_tmp; -- Дата выдачи
    sh_last_date = :sh_date_tmp; -- последняя дата по графику
    if (sh_date_tmp < date_begin) then
    begin
      res_kommis_last_b = res_kommis_last_b + sh_kommis_tmp;
      if ((res_last_date_b is null) or (sh_date_tmp > res_last_date_b)) then
        res_last_date_b = :sh_date_tmp;
    end
    if (sh_date_tmp >= date_begin) then
    begin
      if (res_next_date_b is null) then
        res_next_date_b = sh_date_tmp;
      if (res_next_amount_b is null) then
        res_next_amount_b = sh_kommis_tmp;
    end

    if (sh_date_tmp <= date_end) then
    begin
      res_kommis_last_e = res_kommis_last_e + sh_kommis_tmp;
      if ((res_last_date_e is null) or (sh_date_tmp > res_last_date_e)) then
        res_last_date_e = :sh_date_tmp;
    end
    if (sh_date_tmp > date_end) then
    begin
      if (res_next_date_e is null) then
        res_next_date_e = sh_date_tmp;
      if (res_next_amount_e is null) then
        res_next_amount_e = sh_kommis_tmp;
    end

  end

  if (res_kommis_amount <> 0) then
  begin
    if ((res_disb_date >= :date_begin) and
        (res_disb_date <= :date_end)) then
      res_kommis_payed = res_kommis_amount;

    /* на начало */
    if (date_begin < res_disb_date) then
    begin
      res_amount_begin = 0;
      is_closed_begin = 1;
    end
    else
    begin
      if (date_begin > sh_last_date) then
      begin
        res_amount_begin = res_kommis_amount;
        is_closed_begin = 1;
      end
    end

    if (is_closed_begin = 0) then
    begin
      dd_temp = date_begin - 1;
      if (res_next_date_b is null) then
        res_next_date_b = res_last_date_b;
      if (res_next_amount_b is null) then
        res_next_amount_b = 0;
      res_last_date = res_last_date_b;

      if (res_next_date_b > res_last_date_b) then
      begin
        day_all = (res_next_date_b - res_last_date_b);
        day_period = (dd_temp - res_last_date_b);
        res_amount_begin = res_kommis_last_b;
        ttt = 0;
        if (day_all <> 0) then
        begin
          ttt = cast(cast(day_period as decimal(15,7)) / cast(day_all as decimal(15,7)) as decimal(15,7));
          ttt = cast(res_next_amount_b * ttt as decimal(15,2));
        end
        res_amount_begin = res_amount_begin + ttt;
      end
    end

    /* на конец */
    if (not(exists(select *
                   from kontstate
                   where ks_from_k_id = :kont_id and
                         ks_date_begin <= :date_end and
                         ks_date_end >= :date_end and
                         ks_ol_is_delete = 0 and
                         ks_state = 1))) then
      is_closed_end = 1;
    else
      res_amount_end = 0;

    if (is_closed_end = 0) then
    begin
      res_last_date = res_last_date_e;
      if (res_kommis_last_e is null) then
        res_kommis_last_e = 0;
      if (res_next_date_e is null) then
        res_next_date_e = res_last_date_e;
      if (res_next_amount_e is null) then
        res_next_amount_e = 0;

      if (res_next_date_e > res_last_date_e) then
      begin
        day_all = (res_next_date_e - res_last_date_e);
        day_period = (date_end - res_last_date_e);
        res_amount_end = res_kommis_last_e;
        ttt = 0;
        if (day_all <> 0) then
        begin
          ttt = cast(cast(day_period as decimal(15,7)) / cast(day_all as decimal(15,7)) as decimal(15,7));
          ttt = cast(res_next_amount_e * ttt as decimal(15,2));
        end
        res_amount_end = res_amount_end + ttt;
      end
    end
    else
      res_amount_end = res_kommis_amount;

    if (date_begin < res_disb_date) then
    begin
      res_amount_begin = 0;
      is_closed_begin = 1;
    end
    else
    begin
      if (date_begin > sh_last_date) then
      begin
        res_amount_begin = 0;
        is_closed_begin = 1;
      end
      else
        res_amount_begin = res_kommis_amount - res_amount_begin;
    end

    if (res_kommis_payed <> 0) then
      res_amount_begin = 0;

    if (date_end >= sh_last_date) then
      res_amount_end = 0;
    else
    begin
      if (date_end < res_disb_date) then
        res_amount_end = 0;
      else
        res_amount_end = res_kommis_amount - res_amount_end;
    end
    res_amount_period = (:res_amount_end - :res_amount_begin - :res_kommis_payed);
  end
  suspend;
end
