CREATE PROCEDURE AAREP_REPORT_OKUD_SVOD(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    STR_T1 STR_10,
    STR_T2 STR_100,
    STR_T3 D_STR_255,
    STR_T4 D_STR_255,
    STR_T5 D_STR_20,
    STR_T6 D_STR_20,
    STR_T7 D_STR_64,
    STR_T8 D_STR_255,
    STR_T9 DECIMAL(18,2),
    STR_T10 D_INTEGER,
    STR_T11 DECIMAL(18,2),
    STR_T12 DECIMAL(18,2))
AS
begin

  for select str_t1, str_t2, str_t3, str_t4, str_t5, str_t6, str_t7, str_t8, sum(kc_amount), count(k_kontract_no),
             sum(cli_balance), sum(cli_perc_saldo)
      from aarep_report_okud(:rep_date_begin, :rep_date_end)
      group by str_t1, str_t2, str_t3, str_t4, str_t5, str_t6, str_t7, str_t8

           :str_t12
  do
  begin
    suspend;
  end

end
