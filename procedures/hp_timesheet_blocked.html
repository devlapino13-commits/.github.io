CREATE PROCEDURE HP_TIMESHEET_BLOCKED(
    REG_ID D_INTEGER,
    MONTH_INP D_INTEGER,
    YEAR_INP D_INTEGER)
RETURNS (
    CHECKED D_STR_255)
AS
begin

  select jl.jl_name || ' - ' || un.un_fio
  from sa_tabelo_lock stl
  inner join username un on un.un_id = stl.stl_from_un_id
  inner join hr_pstatus ps on ps.ps_from_c_id = un.un_from_c_id
  inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
  where stl.stl_lock = 1 and
        stl.stl_office_id = :reg_id and
        stl.stl_year_month = :year_inp * 100 + :month_inp and
        cast(stl.stl_datetime as date) between ps.ps_date_begin and ps.ps_date_end
  order by stl.stl_datetime desc rows 1
  into checked;

  suspend;

end
