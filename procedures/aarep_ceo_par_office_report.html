CREATE PROCEDURE AAREP_CEO_PAR_OFFICE_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_C_ID DECIMAL(18,0),
    REP_DAYS_IN_PERIOD INTEGER)
RETURNS (
    REG_ID INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    REP_LINE_NO INTEGER,
    REP_DATE DATE,
    REP_BALANCE_ALL DECIMAL(18,4),
    REP_BALANCE_7 DECIMAL(18,4),
    REP_BALANCE_7_010114 DECIMAL(18,4),
    REP_BALANCE_30 DECIMAL(18,4),
    REP_KOLICH_7 INTEGER,
    REP_KOLICH_7_010114 INTEGER,
    REP_KOLICH_30 INTEGER,
    LO_INFO VARCHAR(100),
    REP_DISB_AMOUNT DECIMAL(18,4),
    REP_DISB_CLIENT INTEGER,
    REP_BALANCE_1 DECIMAL(18,4),
    REP_KOLICH_1 INTEGER,
    REP_KOLICH_ALL INTEGER,
    REP_KOLICH_CLI INTEGER,
    REP_DISB_LGOT_AMOUNT CURR_TYPE,
    REP_DISB_IND_AMOUNT CURR_TYPE,
    REP_BALANCE_LGOT CURR_TYPE,
    REP_BALANCE_IND CURR_TYPE,
    REP_BALANCE_180_ CURR_TYPE,
    REP_KOLICH_180_ CURR_TYPE,
    REP_KOLICH_CLI_180_ CURR_TYPE,
    REP_KOLICH_IND INTEGER,
    REP_TRANSFER_CLI INTEGER,
    REP_DISB_LGOT_KOLICH INTEGER,
    REP_DISB_IND_KOLICH INTEGER,
    REP_KOLICH_LGOT INTEGER)
AS
declare variable reg_id_prior integer;
begin
  rep_line_no = 0;
  lo_info = 'Все сотрудники';
  if ((rep_c_id is null) or (rep_c_id = -1)) then
    rep_c_id = 0;

  for select reg_code, reg_name, reg_id, rep_date, rep_line_no, sum(rep_balance_all), sum(rep_kolich_all),
             sum(rep_balance_1), sum(rep_kolich_1), sum(rep_balance_7), sum(rep_kolich_7), sum(rep_balance_30),
             sum(rep_kolich_30), sum(rep_disb_amount), sum(rep_disb_client), sum(rep_kolich_cli),
             sum(rep_disb_lgot_amount), sum(rep_disb_ind_amount), sum(rep_balance_lgot), sum(rep_balance_ind),
             sum(rep_balance_180_), sum(rep_kolich_180_), sum(rep_kolich_cli_180_), sum(rep_kolich_ind),
             sum(rep_transfer_office_cli), sum(rep_disb_lgot_kolich), sum(rep_disb_ind_kolich), sum(rep_kolich_lgot),
             sum(rep_balance_7_010114), sum(rep_kolich_7_010114)
      from aarep_ceo_par_report_tmp(:rep_region_id, :rep_date_end, :rep_c_id, :rep_days_in_period)
      group by reg_code, reg_name, reg_id, rep_date, rep_line_no, parent, parent2
      order by parent2, parent, reg_code, rep_date desc
      into :reg_code, :reg_name, reg_id, :rep_date, rep_line_no, :rep_balance_all, :rep_kolich_all, :rep_balance_1,
           :rep_kolich_1, :rep_balance_7, :rep_kolich_7, :rep_balance_30, :rep_kolich_30, :rep_disb_amount,
           rep_disb_client, :rep_kolich_cli, :rep_disb_lgot_amount, :rep_disb_ind_amount, rep_balance_lgot,
           rep_balance_ind, rep_balance_180_, rep_kolich_180_, rep_kolich_cli_180_, rep_kolich_ind, rep_transfer_cli,
           rep_disb_lgot_kolich, rep_disb_ind_kolich, rep_kolich_lgot, rep_balance_7_010114, rep_kolich_7_010114
  do
  begin
    if (reg_id <> reg_id_prior) then
      lo_info = 'Все сотрудники';
    else
      lo_info = '';

    suspend;
    reg_id_prior = reg_id;
  end
end
