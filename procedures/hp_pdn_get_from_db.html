CREATE PROCEDURE HP_PDN_GET_FROM_DB(
    KC_ID D_ID,
    CALC_DATE D_DATE,
    OUR_OL_ID D_ID = null)
RETURNS (
    CREDENDDATE D_DATE,
    CREDENDDATEFACT D_DATE,
    CREDUPDATE D_DATE,
    CREDSUMDEBT D_CURRENCY,
    CREDSUMOVERDUE D_CURRENCY,
    CREDDAYOVERDUE D_INTEGER,
    CREDFULLCOST D_CURRENCY,
    CREDPARTNERTYPE D_STR_10,
    CREDPARTNERTYPE_ID D_ID,
    CREDTYPE D_STR_10,
    CREDTYPE_ID D_ID,
    CREDCOLLATERAL D_INTEGER,
    CREDCOLLATERALFACTENDDATE D_DATE,
    CREDCOLLATERALITEMTYPE D_STR_10,
    CREDACTIVE_STR D_STR_255,
    CREDIT_LINE_TYPE D_STR_10,
    CRED_OWNER_TYPE D_STR_10,
    DEBTCURR_OPSUM D_CURRENCY,
    DEBTCURR_PERCENTSUM D_CURRENCY,
    DEBTCURR_OTHERSUM D_CURRENCY,
    DEBTOVERDUE_OPSUM D_CURRENCY,
    DEBTOVERDUE_PERCENTSUM D_CURRENCY,
    DEBTOVERDUE_OTHERSUM D_CURRENCY,
    FULLCOST_SUM D_CURRENCY,
    FULLCOST_DATE D_DATE,
    FUTURE_PERCENT_SUM D_CURRENCY)
AS
declare variable k_id d_id;
  declare variable pr_code d_str_100;
  declare variable k_disb_date d_date;
  declare variable debt_overdue_date d_date;
begin
  select k_id, k_disb_date, k_kredline_end, k_percent_effect_fifd, pr_code
  from knclient, kontract, product
  where kc_id = :kc_id
    and kc_from_k_id = k_id
    and k_from_pr_id = pr_id
  into k_id, k_disb_date, credEndDate, credFullCost, pr_code;

  credPartnerType = -1;
  credSumDebt = 0;
  credSumOverdue = 0;
  creddayoverdue = 0;
  credUpdate = calc_date;
  debtcurr_opSum = 0;
  debtcurr_percentSum = 0;
  future_percent_sum = 0;
  debtcurr_otherSum = 0;
  debtOverdue_opSum = 0;
  debtOverdue_percentSum = 0;
  debtOverdue_otherSum = 0;
  debt_overdue_date = null;

  select res_delinq_date, res_delinq_os, res_delinq_perc, res_future_os, res_future_calc_perc, res_future_perc_without_calc
  from hp_pdn_kontract_info(:kc_id, :calc_date, :our_ol_id)
  into debt_overdue_date, debtOverdue_opSum, debtOverdue_percentSum, debtcurr_opSum, debtcurr_percentSum, future_percent_sum;

  -- если выдача, то считаем будущие проценты по повышенной ставке
  if (k_disb_date = calc_date) then
    future_percent_sum = (select sum(sh_perc) from hp_get_base_schedule(:k_id));

  debtcurr_otherSum = future_percent_sum;

  if (debtcurr_opSum < 0) then debtcurr_opSum = 0;
  if (debtcurr_percentSum < 0) then debtcurr_percentSum = 0;
  if (debtcurr_otherSum < 0) then debtcurr_otherSum = 0;
  if (debtOverdue_opSum < 0) then debtOverdue_opSum = 0;
  if (debtOverdue_percentSum < 0) then debtOverdue_percentSum = 0;

  if (debt_overdue_date is not null and debtOverdue_opSum + debtOverdue_percentSum = 0) then
    debt_overdue_date = null;

  if (debt_overdue_date is not null) then
    creddayoverdue = calc_date - debt_overdue_date;

  credSumDebt = debtcurr_opSum;
  credSumOverdue = debtOverdue_opSum + debtOverdue_percentSum;

  -- Получаем актуальную ПСК.
  with psk_tab as(
    select ker_percent_effect_fifd psk, ker_date psk_date
    from konteffrate ker
    where ker.ker_from_k_id = :k_id
      and ker.ker_date <= :calc_date - iif(:calc_date < current_date, 1, 0) -- если расчет задним числом, то изменения в день расчета игнорим
      and ker.ker_state = 0
    union
    select kd_percent_effect_fifd, kd_ol_date
    from kontdata
    where :calc_date between kd_ol_date and kd_date_end
      and kd_ol_is_delete = 0
      and kd_from_k_id = :k_id
      and (:calc_date > kd_ol_date or kd_from_ol_id <= :our_ol_id)
    union
    select kps_psk, ol_date
    from kontpsk, operlist
    where kps_from_k_id = :k_id
      and ol_id = kps_from_ol_id
      and is_delete = 0
      and ol_date <= :calc_date
      and (:calc_date > ol_date or ol_id <= :our_ol_id)
    order by 2 desc
    rows 1)
  select psk, psk_date
  from psk_tab
  into credFullCost, fullcost_date;

  if (k_disb_date = calc_date) then
    fullcost_sum = future_percent_sum;
  else
    select sum(ok_sh_perc + ok_kommis)
    from operkred
    where ok_from_k_id = :k_id
      and ok_ol_is_active = 1
      and ok_type < 4
      and ok_ol_date <= :fullcost_date
      and (:calc_date > ok_ol_date or ok_from_ol_id <= :our_ol_id)
    into fullcost_sum;

  credEnddatefact = null;
  if (credSumDebt + credSumOverdue = 0) then
  begin
    select oks_date_begin
    from kn_oksummary
    where oks_from_kc_id = :kc_id and
          oks_date_begin <= :calc_date and
          oks_close_state = 1
    rows 1
    into credEnddatefact;
  end

  credActive_str = case
    when credSumDebt + credSumOverdue = 0 and coalesce(credEndDateFact, '01.01.1900') <> '01.01.1900' then 'договор закрыт'
    when credDayOverdue > 364 then 'договор активен, просрочка 365+ дней'
    when credDayOverdue > 90 then 'договор активен, просрочка до 90-365 дней'
    when credDayOverdue > 30 then 'договор активен, просрочка до 30-90 дней'
    when credDayOverdue > 0 then 'договор активен, просрочка до 30 дней'
    else 'договор активен'
  end;

  credcollateral = 0;
  credcollateralfactenddate = null;
  credcollateralitemtype = null;
  credit_line_type = 0;
  cred_owner_type = -1;
  credtype = 13;

  if (pr_code containing '.zalog_car.') then
  begin
    credcollateral = 1;
    credcollateralfactenddate = credenddatefact;
    credcollateralitemtype = '2';
    credtype = 17;
  end

  suspend;
end
