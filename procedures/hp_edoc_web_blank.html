CREATE PROCEDURE HP_EDOC_WEB_BLANK(
    UUID D_STR_CODE,
    ID D_ID)
RETURNS (
    FILE_PDF D_BLOB_BINARY)
AS
declare variable edl_id d_id;
declare variable bls_id d_id;
declare variable eds_id d_id;
declare variable edb_id d_id;
declare variable eds_status d_integer;
begin
  -- процедура возвращает blob для просмотра при нажатии на кнопку документа
  -- данная процедура также фиксирует время открытия документа

  select obj_id
  from hp_uuid(:uuid)
  into eds_id;

  select eds_from_edl_id, eds_status
  from edoc_sign
  where eds_id = :eds_id
  into edl_id, eds_status;

  if (edl_id is not null) then
  begin
    select edb_id, coalesce(edbs_from_bls_id, edb_from_bls_id), edb_blob
    from edoc_blank
    left join edoc_blank_scan on (edbs_from_edb_id = edb_id and
          edbs_status = 2)
    where edb_id = :id and
          edb_from_edl_id = :edl_id
    into edb_id, bls_id, file_pdf;

    if (file_pdf is null and
        bls_id is not null) then
    begin
      select blx_blob
      from hp_get_blobex_blob(:bls_id)
      into file_pdf;

      if (eds_status = 0) then
      begin
        update or insert into edoc_open (edo_from_eds_id, edo_from_edb_id, edo_before_sign_datetime)
        values (:eds_id, :edb_id, current_timestamp)
        matching (edo_from_eds_id, edo_from_edb_id);
      end
      else
      begin
        update or insert into edoc_open (edo_from_eds_id, edo_from_edb_id, edo_after_sign_datetime)
        values (:eds_id, :edb_id, current_timestamp)
        matching (edo_from_eds_id, edo_from_edb_id);
      end
    end
  end

  suspend;
end
