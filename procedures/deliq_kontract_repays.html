CREATE PROCEDURE DELIQ_KONTRACT_REPAYS(
    REP_DATE DATE)
RETURNS (
    DK_ID DECIMAL(18,0),
    DK_DATE DATE,
    DK_REPAYS INTEGER,
    DK_OS DECIMAL(18,4),
    DK_PERC DECIMAL(18,4),
    DK_DAYS INTEGER)
AS
declare variable sum_os decimal(18,4);
declare variable sum_perc decimal(18,4);
begin
  /* Procedure Text */
  for select ok_from_k_id, ok_sh_id, ok_sh_date, sum(ok_sh_os + ok_os), sum(ok_sh_perc + ok_perc)
      from operkred
      where ok_from_c_id = 0 and
            ok_sh_date <= :rep_date and
            ok_ol_date <= :rep_date and
            ok_ol_is_active = 1 and
            ok_type < 8 and
            ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
      group by ok_from_k_id, ok_sh_id, ok_sh_date
      having sum(ok_sh_os + ok_os) > 0 or sum(ok_sh_perc + ok_perc) > 0
      into :dk_id, :dk_repays, :dk_date, :dk_os, :dk_perc
  do
  begin
    if (dk_os is null) then
      dk_os = 0;
    if (dk_perc is null) then
      dk_perc = 0;
    dk_days = cast((rep_date - dk_date) as integer);
    if ((dk_os > 0) or (dk_perc > 0)) then
      suspend;
  end
end
