CREATE PROCEDURE HP_FREEZE_CHANGE_STATUS
AS
declare variable K_ID ID_TYPE;
declare variable DISB_DATE D_DATETIME;
declare variable FREEZE_TIME_H D_INTEGER;
declare variable IS_ERROR D_SMALLINT;
declare variable RES_TEXT STR_255;
declare variable K_KONTRACT_NO STR_20;
declare variable FREEZE_TIME_TYPE D_INTEGER;
declare variable C_NAME_FIO STR_100;
declare variable Z_NAME_FIO STR_100;
declare variable EDOC_KIND STR_100;
declare variable UN_ID D_ID;
declare variable KS_KC_ID D_ID;
begin
  if (current_date < '01.09.2025') then
    exit;

  freeze_time_type = null;

  freeze_time_type = case extract(hour from cast(current_time as time))
                       when 16 then 4
                       when 20 then 48
                     end;

  for select ks_from_k_id, k_disb_date, k_kontract_no, ks_kc_id
      from kontstate
      inner join kontract on k_id = ks_from_k_id
      where current_date between ks_date_begin and ks_date_end and
            ks_ol_is_delete = 0 and
            ks_state = 0 and
            k_status = 1
      into k_id, disb_date, k_kontract_no, ks_kc_id
  do
  begin
    freeze_time_h = null;
    is_error = 0;
    res_text = '';

    select u.un_id
    from username u
    where u.un_from_c_id = :ks_kc_id
    into :un_id;

    select freeze_time_h
    from hp_get_freeze_data(:k_id)
    where freeze_datetime_end is null
    into freeze_time_h;

    if (freeze_time_h is not null and
        ((dateadd(hour, freeze_time_h, current_timestamp) > dateadd(hour, 20, disb_date) and
          freeze_time_h = 4 and freeze_time_type = 4) or freeze_time_type = 48)) then
    begin
      select c.c_name_fio
      from clients c
      join knclient kc on kc.kc_from_c_id = c.c_id
      where kc.kc_from_k_id = :k_id
      into :c_name_fio;

      select z_name_fio
      from zalog
      where z_from_k_id = :k_id
        and z_type = 1
        and current_date between coalesce(z_date_begin, current_date) and coalesce(z_date_end, current_date)
      into :z_name_fio;

      for select edc_code
      from edoc_list edl
      join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
      where edl.edl_object_id = :k_id
        and edc.edc_code in ('loan.agreement', 'loan.collateral')
        and edl.edl_object_kind = 9
        and edl.edl_status = 0
      into :edoc_kind
      do
      begin
        if (edoc_kind = 'loan.agreement') then
          res_text = 'Договор займа № ' || k_kontract_no || ' от ' || cast(disb_date as date) || ' ' ||
                     coalesce(c_name_fio,'') || ' - аннулирован';
        else
          res_text = 'Договор залога № ' || k_kontract_no || ' от ' || cast(disb_date as date) || ' ' ||
                     coalesce(z_name_fio,'') || ' - аннулирован';

        if (un_id is not null) then
          f_send_push('username',:un_id,'Аннулирование договора',:res_text,'hp_freeze_change_status_new',7,1000000000000);
      end

      select res_need_rollback, res_text
      from hp_kontract_set_status(:k_id, 0, 0, 1000000000000)
      into is_error, res_text;

      -- если ошибка вернулась, то письмо на финстарт, если нет, то нет
      if (is_error = 1) then
      begin
        insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                    msg_command_kind)
        values (0, 'ru.finstart@molbulak.com',
                'Ошибка при снятии утверждения с договора',
                f_format('Ошибка: "%1", по договору: %2', :res_text, :k_kontract_no), 1, 1000000000001,
                '');
      end
    end
  end
end
