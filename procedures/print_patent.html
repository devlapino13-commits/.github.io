CREATE PROCEDURE PRINT_PATENT(
    PAR1 DECIMAL(18,0))
RETURNS (
    CDS_FROM_C_ID DECIMAL(18,0),
    CDS_ID INTEGER,
    CDS_FROM_PK_ID INTEGER,
    CDS_DOC_NO VARCHAR(30),
    CDS_DOC_DATE_BEGIN D_DATE,
    CDS_DOC_DATE_END D_DATE,
    CDS_DOC_ISSUED_BY VARCHAR(255),
    CDS_DOC_FOR_COMPANY VARCHAR(255),
    CDS_DOC_TERRITORY VARCHAR(255),
    CDS_DOC_BLOB_1 DECIMAL(18,0),
    CDS_DOC_BLOB_2 DECIMAL(18,0),
    CDS_INDEX_DOC_NO VARCHAR(30),
    CDS_FROM_UN_ID DECIMAL(18,0),
    CDS_DATETIME TIMESTAMP,
    CDS_IS_DELETE INTEGER,
    CDS_VERIFY_STATUS INTEGER,
    CDS_VERIFY_DATETIME INTEGER,
    CDS_VERIFY_UN_ID DECIMAL(18,0),
    CDS_MEMO_1 VARCHAR(255),
    CDS_MEMO_2 VARCHAR(255),
    CDS_MEMO_3 VARCHAR(255),
    CDS_MEMO_4 VARCHAR(255),
    CDS_MEMO_5 VARCHAR(255),
    CDS_IS_MAIN_DOC INTEGER,
    CDS_FROM_DRL_ID DECIMAL(18,0),
    CDS_CREATE_DATETIME TIMESTAMP,
    CDS_CREATE_UN_ID BIGINT,
    CDS_DELETE_DATETIME TIMESTAMP,
    CDS_DELETE_UN_ID BIGINT,
    CDS_DOC_NAME CHAR(8))
AS
begin
  select cd.cds_from_c_id, cd.cds_id, cd.cds_from_pk_id, cd.cds_doc_no, cd.cds_doc_date_begin, cd.cds_doc_date_end,
         cd.cds_doc_issued_by, cd.cds_doc_for_company, cd.cds_doc_territory, cd.cds_doc_blob_1, cd.cds_doc_blob_2,
         cd.cds_index_doc_no, cd.cds_from_un_id, cd.cds_datetime, cd.cds_is_delete, cd.cds_verify_status,
         cd.cds_verify_datetime, cd.cds_verify_un_id, cd.cds_memo_1, cd.cds_memo_2, cd.cds_memo_3, cd.cds_memo_4,
         cd.cds_memo_5, cd.cds_is_main_doc, cd.cds_from_drl_id, cd.cds_create_datetime, cd.cds_create_un_id,
         cd.cds_delete_datetime, cd.cds_delete_un_id, 'лиценция' as cds_doc_name

  from knclient
  left join clientdocs cd on cd.cds_from_c_id = kc_from_c_id
  where kc_from_k_id = :par1 and
        cd.cds_is_main_doc = 0 and
        current_date between cds_doc_date_begin and cds_doc_date_end and
        cd.cds_is_delete = 0 and
        cd.cds_from_pk_id = 70

  into :cds_from_c_id, :cds_id, :cds_from_pk_id, :cds_doc_no, :cds_doc_date_begin, :cds_doc_date_end,
       :cds_doc_issued_by, :cds_doc_for_company, :cds_doc_territory, :cds_doc_blob_1, :cds_doc_blob_2,
       :cds_index_doc_no, :cds_from_un_id, :cds_datetime, :cds_is_delete, :cds_verify_status, :cds_verify_datetime,
       :cds_verify_un_id, :cds_memo_1, :cds_memo_2, :cds_memo_3, :cds_memo_4, :cds_memo_5, :cds_is_main_doc,
       :cds_from_drl_id, :cds_create_datetime, :cds_create_un_id, :cds_delete_datetime, :cds_delete_un_id,
       :cds_doc_name;

  suspend;
end
