CREATE PROCEDURE HP_CLIENTCOMREG_AUTO_FILL(
    REP_DATE D_DATE = current_date)
AS
declare variable c_id d_id;
begin
  -- Регламентная процедура, для обработки "Реестра субьектов рассылка которым запрещена"
  -- Таблицы: clientcomreg, clientcomlist
  -- Выполняется раз в сутки, роботом.

  -- По 230-ФЗ запрет автоматически снимается, после закрытия последнего активного займа клиента.
  -- Для этого просматриваем все закрывшиеся займы на дату rep_date
  for with closed as (
    select changed.oks_from_k_id as k_id
    from oksummary changed
    left join oksummary absent on (changed.oks_from_k_id = absent.oks_from_k_id and absent.oks_date_begin = :rep_date)
    where changed.oks_date_end = (:rep_date - 1)
      and absent.oks_from_k_id is null)
  select distinct(kc_from_c_id)
  from closed
    left join knclient on (k_id = kc_from_k_id)
  where kc_amount_approved > 0
  into c_id
  do
  begin
    -- Далее проверяем, у них есть запрет по 230-ФЗ и не осталось никаких действующих займов
    if (f_clientcom_allowed(c_id, 0, 0, 1) = 0 and
      not exists(
        select *
        from knclient, oksummary
        where kc_from_c_id = :c_id and
              kc_amount_approved > 0 and
              oks_from_k_id = kc_from_k_id and
              oks_date_begin >= :rep_date))
    then
      -- Таким снимаем ограничения по ФЗ-230
      execute procedure hp_clientcomreg_set(:c_id, null, 'remove', 0, 0, 1, 'loans_closed');
  end
end
