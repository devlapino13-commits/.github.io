CREATE PROCEDURE OKT_WEBSITE_FILL_ANKETA(
    ESM_ID ID_TYPE)
AS
declare variable un_id id_type;
declare variable res_acl_id id_type;
declare variable field_code str_255;
declare variable value_txt str_255;
declare variable citizenship_code str_255;
declare variable language_code str_255;
declare variable db_id integer;
begin

  :db_id = f_get_db_id();

  if (db_id = 81001) then -- БД РФ
  begin
    select un.un_id
    from username un
    where un.un_name = 'IGUSEV'
    into :un_id;

    execute procedure hp_actionlist_insert(null, null, 'ExtSysMain', :esm_id, 'OKTELL.Projects.Website', 0, :un_id)
        returning_values :res_acl_id;

    for select ef.esf_field_code, ev.esv_value_txt
        from extsysvalues ev
        inner join extsysfields ef on ef.esf_id = ev.esv_from_esf_id
        where ev.esv_from_esm_id = :esm_id
        into :field_code, :value_txt
    do
    begin
      if (field_code = 'cl_name') then
      begin
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Name',
            'AnketaQuestions.OKTELL.WebSite.Cl_Name', 0, '', :un_id);
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Name', 'Results.StrValue',
            1, :value_txt, :un_id);
      end

      if (field_code = 'Phone_No') then
      begin
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Phone_No',
            'AnketaQuestions.OKTELL.WebSite.Phone_No', 0, '', :un_id);
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Phone_No', 'Results.StrValue',
            1, :value_txt, :un_id);
      end

      if (field_code = 'Cl_Citizenship') then
      begin
        citizenship_code = null;

        select rd.rd_code
        from refdata rd
        where rd.rd_main_code = 'Citizen' and
              rd.rd_name = :value_txt
        into :citizenship_code;

        if (citizenship_code is null) then
          citizenship_code = 'Results.Other';

        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Citizenship',
            'AnketaQuestions.OKTELL.WebSite.Cl_Citizenship', 0, '', :un_id);
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Citizenship',
            :citizenship_code, 1, iif(citizenship_code = 'Results.Other', :value_txt, ''), :un_id);
      end

      if (field_code = 'Cl_City') then
      begin
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_City',
            'AnketaQuestions.OKTELL.WebSite.Cl_City', 0, '', :un_id);
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_City', 'Results.StrValue',
            1, :value_txt, :un_id);
      end

      if (field_code = 'Cl_BirthDate') then
      begin
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_BirthDate',
            'AnketaQuestions.OKTELL.WebSite.Cl_BirthDate', 0, '', :un_id);
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_BirthDate',
            'Results.StrValue', 1, :value_txt, :un_id);
      end

      if (field_code = 'Cl_Language') then
      begin
        language_code = null;

        select rd.rd_code
        from refdata rd
        where rd.rd_main_code = 'language' and
              rd.rd_name = :value_txt
        into :language_code;

        if (language_code is null) then
          language_code = 'Results.Other';

        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Language',
            'AnketaQuestions.OKTELL.WebSite.Cl_Language', 0, '', :un_id);
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Language',
            :citizenship_code, 1, iif(language_code = 'Results.Other', :value_txt, ''), :un_id);
      end

      if (field_code = 'Cl_Kolich_Repay') then
      begin
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Kolich_Repay',
            'AnketaQuestions.OKTELL.WebSite.Cl_Kolich_Repay', 0, '', :un_id);
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Kolich_Repay',
            'Results.StrValue', 1, :value_txt, :un_id);
      end

      if (field_code = 'Cl_Loan_Amount') then
      begin
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Loan_Amount',
            'AnketaQuestions.OKTELL.WebSite.Cl_Loan_Amount', 0, '', :un_id);
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Loan_Amount',
            'Results.StrValue', 1, :value_txt, :un_id);
      end

      if (field_code = 'Cl_Next_Call_Value') then
      begin
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Next_Call_Value',
            'AnketaQuestions.OKTELL.WebSite.Cl_Next_Call_Value', 0, '', :un_id);
        execute procedure okt_do_actionsave(:res_acl_id, 'AnketaQuestions.OKTELL.WebSite.Cl_Next_Call_Value',
            'Results.StrValue', 1, :value_txt, :un_id);
      end

      execute procedure okt_do_actionsave(:res_acl_id, '0', '0', 0, '', :un_id);

      update actionlist al
      set al.acl_status = 0
      where al.acl_id = :res_acl_id;
    end
  end
end
