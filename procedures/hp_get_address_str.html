CREATE PROCEDURE HP_GET_ADDRESS_STR(
    ADL_ID D_ID)
RETURNS (
    ADL_ADDRESS_STR D_STR_255,
    NEED_GEO D_BOOL)
AS
declare variable quota_exceeded d_bool;
begin
  -- Процедура вызывается из FsService, возвращает адрес для получения геопозиции
  if (f_get_db_id() <> 81001) then
    exit;

  need_geo = 0;
  -- Проверка на превышение количества запросов в Yandex Geocoder
  -- Передаем adl_id, но это не принципиально, т.к. лимит на уровне всех запросов
  select quota_exceeded
  from hp_msgquota_process('yandex.geocoder', :adl_id)
  into quota_exceeded;
  if (quota_exceeded = 0) then
    need_geo = 1;

  adl_address_str = '';
  select adl_address_str
  from addrlist
  where adl_id = :adl_id and
        (adl_depart_no not in ('0', '') or
        adl_flat_no not in ('0', '') or
        adl_building_no not in ('0', '') or
        adl_section_no not in ('0', ''))
  into adl_address_str;
  if (adl_address_str <> '') then
    suspend;
end
