CREATE PROCEDURE HP_AGREEMENTS_INS(
    C_ID D_ID,
    AG_RD_CODE D_STR_CODE,
    AG_REGION_ID D_INTEGER,
    AG_DATE D_DATE,
    UN_ID D_ID)
RETURNS (
    AG_ID D_ID)
AS
declare variable rd_id d_integer;
declare variable rd_name d_str_255;
begin
  -- 'AgreementKinds.loyal_agreement'
  select rd_id, rd_name
  from refdata
  where rd_code = :ag_rd_code and
        rd_code starting 'AgreementKinds.'
  into rd_id, rd_name;

  if (rd_id is null) then
    exception e_default 'Incorrect rd_code: ' || coalesce(ag_rd_code, '<NULL>');

  if (ag_date is null) then ag_date = current_date;

  if (ag_rd_code = 'AgreementKinds.loyal_agreement')
  then rd_name = 'Программа лояльности клиентов';

  if (ag_region_id is null) then
    ag_region_id = f_hq_id();

  select ag_id
  from agreements
  where ag_object_kind = f_objkind('clients', 'id') and
        ag_object_id = :c_id and
        ag_from_rd_id = :rd_id and
        coalesce(ag_date_end, '31.12.2999') >= current_date and
        coalesce(ag_status, 0) <> -1
  order by ag_date_begin desc
  rows 1
  into ag_id;

  if (ag_id is null) then
  begin
    insert into agreements (ag_from_rd_id, ag_object_kind, ag_object_id, ag_from_reg_id, ag_caption,
                            ag_agreement_date, ag_date_begin, ag_memo, ag_status, ag_from_un_id)
    values (:rd_id, f_objkind('clients', 'id'), :c_id, :ag_region_id, :rd_name,
            :ag_date, :ag_date, '', 0, :un_id)
    returning ag_id
    into :ag_id;
  end

  suspend;
end
