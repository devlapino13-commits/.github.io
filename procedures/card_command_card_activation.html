CREATE PROCEDURE CARD_COMMAND_CARD_ACTIVATION(
    ACTIVATION_CODE D_STR_100,
    OBJECT_KIND D_STR_100,
    OBJECT_ID D_ID,
    PARENT_CS_ID D_INTEGER = null)
AS
declare variable object_kind_str d_str_100;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable cs_request_json d_str_8100;
declare variable cs_id d_integer;
declare variable object_kind_parent d_integer;
declare variable object_id_parent d_id;
declare variable cbo_region_id d_integer;
declare variable un_id d_id;
declare variable card_id_tmp d_integer;
declare variable k_id d_id;
declare variable c_id d_id;
begin
    /*
         Процедура отвечает за формирование операции для МПЦ
         "Активация карты"
    */
    activation_code = coalesce(activation_code, '');

    if (activation_code not in ('card_activation_on_disb', 'card_activation')) then
        exception e_message 'Данный тип активации ['||:activation_code||'] не поддерживается';

    if (parent_cs_id is not null) then
    begin
        select cs.cs_object_kind,
               cs.cs_object_id,
               cs.cs_card_id,
               cs.cs_from_cca_id,
               cs.cs_from_k_id,
               cs.cs_from_c_id
        from cardsync cs
        where cs.cs_id = :parent_cs_id
        into :object_kind_parent,
             :object_id_parent,
             :card_id,
             :cca_id,
             :k_id,
             :c_id;

        if ((object_kind is null) and(object_id is null)) then
        begin
            object_kind = :object_kind_parent;
            object_id = object_id_parent;
        end
    end


    object_kind_str = f_objkind(object_kind, 'name');
    
    if (object_kind_str = 'cardblockopers') then
    begin
        select cbo.cbo_from_cal_id,
               cbo.cbo_region_id,
               cbo.cbo_from_un_id
        from cardblockopers cbo
        where cbo.cbo_id = :object_id
        into :card_id_tmp,
             :cbo_region_id,
             :un_id;

        if (card_id is null) then
            card_id = card_id_tmp;

        if (cca_id is null) then
        begin
            select cal_from_cca_id
            from cardlist cal
            where cal.cal_card_id = :card_id
            into :cca_id;
        end
    end
    else if (object_kind_str = 'cardaktclient') then
    begin
        select ckc.cac_from_reg_id,
               ckc.cac_from_un_id
        from cardaktclient ckc
        where ckc.cac_id = :object_id
        into :cbo_region_id,
             :un_id;
    end

    cs_request_json = f_format('{"card_id":%1}', :card_id);

    insert into cardsync(
      cs_parent_id,
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_request_json,
      cs_office_id,
      cs_from_un_id,
      cs_from_k_id,
      cs_from_c_id
      )
    values(
      :parent_cs_id,
      f_get_cardsynccfg(:activation_code, 'id'),
      :card_id,
      :cca_id,
      :object_kind,
      :object_id,
      :cs_request_json,
      :cbo_region_id,
      :un_id,
      :k_id,
      :c_id
      )
     returning cs_id
     into :cs_id;

     if (object_kind_str = 'cardblockopers') then
     begin
        update cardblockopers cbo
        set cbo.cbo_from_cs_id = :cs_id
        where cbo.cbo_id = :object_id;
     end
end
