CREATE PROCEDURE HP_CHECK_RATING_RESULT(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID)
RETURNS (
    BY_RATING D_SMALLINT,
    K_RATING_SOURCE D_INTEGER)
AS
declare variable amount d_currency;
declare variable c_id d_id;
declare variable ma_date_oper d_date;
declare variable ma_id d_id;
declare variable kib_score d_integer;
declare variable point_kind d_smallint;
declare variable i d_integer;
declare variable min_kib_score d_integer;
declare variable pr_code d_str_100;
declare variable base_k_id d_id;
declare variable base_ma_id d_integer;
declare variable risk_level d_str_10;
declare variable data_source d_integer;
begin
  --JIRA 1519
  if ((select cv.cv_value
       from const_var cv
       where cv.cv_code = 'credit_rating_active') = 0) then
  begin
    by_rating = 0;
    suspend;
    exit;
  end

  by_rating = 1;
  k_rating_source = 0;

  if (f_objkind(:obj_kind, 'name') = 'kontract') then
  begin
    select kt.k_from_ma_app_id
    from kontract kt
    where kt.k_id = :obj_id
    into obj_id;
  end

  select pr.pr_code
  from miniapp ma
  inner join product pr on pr.pr_id = ma.ma_from_pr_id
  where ma.ma_id = :obj_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.2999'
  order by ma.ma_number desc rows 1
  into pr_code;

  if (f_get_db_id() = 41701 and
      pr_code containing '.insur_kontrakt.') then  -- RFCKG-5051 Для Страховки берем данные из КИБ из Основного контракта
  begin
    select lir.lir_kontract_base
    from kontract k
    inner join loan_insurance_relation lir on k.k_id = lir.lir_kontract_insurance
    where k.k_from_ma_app_id = :obj_id
    into base_k_id;

    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :base_k_id
    into base_ma_id;
  end

  i = 0;

  for select ma.ma_summa_minus,
             mac.mac_from_c_id,
             ma.ma_date_oper,
             ma.ma_id,
             iif(pr.pr_code containing '.IPF.', 2, 0) --2 ИПФ, 0 - общий балл
      from miniapp ma
      inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
      inner join product pr on pr.pr_id = ma.ma_from_pr_id
      where ma.ma_id = :obj_id and
            ma.ma_is_active = 1 and
            ma.ma_date_end >= '31.12.2999' and
            mac.mac_amount > 0
      into amount,
           c_id,
           ma_date_oper,
           ma_id,
           point_kind
  do
  begin
    i = i + 1;

    data_source = f_get_rd_id('CfpAnketaCalcSource.Ishenim');
    kib_score = 0;
    select kbm_score_rate
    from hp_kibminiapp_last(:c_id, iif(:pr_code containing '.insur_kontrakt.', :base_ma_id, :ma_id))   -- RFCKG-5051 Для Страховки берем данные из КИБ из Основного контракта
    into kib_score;

    if (coalesce(kib_score, 0) = 0) then    -- RFCKG-5233
    begin
      data_source = f_get_rd_id('CfpAnketaCalcSource.SafeAndSound');
      select risk_level
      from print_report_safe_and_sound(iif(:pr_code containing '.insur_kontrakt.', :base_ma_id, :ma_id), :c_id)
      rows 1
      into risk_level;

      risk_level = coalesce(risk_level, '--');
    end

    kib_score = coalesce(kib_score, 0);

    --Получаем минимальный уровень риска в отчете КБ
    if (i = 1) then
      min_kib_score = kib_score;
    else
      min_kib_score = iif(kib_score < min_kib_score, kib_score, min_kib_score);

    -->--RFCKG-3546(Начало)
    --Если Сумма меньше 5 000 или больше 100 000 или уровень риска в отчете КБ = очень высокий
    if (amount < 5000 or amount > 100000 or
    iif(f_get_rd_code(:data_source) = 'CfpAnketaCalcSource.Ishenim', min_kib_score < 573, risk_level in ('I', 'J'))) then
    begin
      by_rating = 0;
      suspend;
      exit;
    end

    --Нижний порог среднеге риска или когда кредитный рейтинг по КИБ не может быть рассчитан
    if (iif(f_get_rd_code(:data_source) = 'CfpAnketaCalcSource.Ishenim', kib_score < 641, risk_level in ('G', 'H', 'I', 'J')) or
        iif(f_get_rd_code(:data_source) = 'CfpAnketaCalcSource.Ishenim', kib_score = 999, risk_level = '--')) then
    begin
      by_rating = 0;
    end

    --Если Нижни порог среднего риска и сумма более 20 000
    if (iif(f_get_rd_code(:data_source) = 'CfpAnketaCalcSource.Ishenim', min_kib_score < 653, risk_level in ('G', 'H', 'I', 'J'))and amount > 20000) then
    begin
      by_rating = 0;
      suspend;
      exit;
    end
  end
  --<--RFCKG-3546(Конец)
  if (by_rating = 1) then
  begin
    k_rating_source = 1;
    suspend;
    exit;
  end 
  else
  begin
    min_kib_score = coalesce(min_kib_score, 0);

    --Нижний порог высокого риска или когда кредитный рейтинг по КИБ не может быть рассчитан
    if (iif(f_get_rd_code(:data_source) = 'CfpAnketaCalcSource.Ishenim', min_kib_score < 573, risk_level in ('I', 'J')) or
        iif(f_get_rd_code(:data_source) = 'CfpAnketaCalcSource.Ishenim', min_kib_score = 999, risk_level = '--')) then
    begin
      by_rating = 0;
      suspend;
      exit;
    end

    --Если есть активная просрочка, кредит тоже не идёт по кредитному рейтингу k_rating_soucre = 2
    if ((select sum(iif((:ma_date_oper - oks.oks_delinq_date < 0), 0, :ma_date_oper - oks.oks_delinq_date))
         from miniappclient mac
         inner join knclient kc on kc.kc_from_c_id = mac.mac_from_c_id and
               kc.kc_amount_approved > 0
         inner join kontract kt on kt.k_id = kc.kc_from_k_id and
               kt.k_from_ma_app_id <> mac.mac_from_ma_id
         inner join oksummary oks on oks.oks_from_k_id = kt.k_id and
               :ma_date_oper between oks.oks_date_begin and oks.oks_date_end and
               oks.oks_close_state = 0
         where mac.mac_from_ma_id = :obj_id and
               mac.mac_amount > 0) > 0) then
     begin
       by_rating = 0;
       suspend;
       exit;
     end
    
    by_rating = 1;
    k_rating_source = 2;
  end 

  suspend;
end
