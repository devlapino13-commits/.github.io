CREATE PROCEDURE HP_FT_RISK_CATEG(
    C_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    RES_RISK_CATEG D_INTEGER,
    FOUND_SOURCE STR_100)
AS
begin
  -- Процедура возвращает уровень риска клиента для анкеты ФТ

  -- RES_RISK_CATEG:
  -- 0 - стандартный уровен риска
  -- 1 - высокий уровен риска
  -- 2 - критический уровен риска



  if (c_id is null) then
    exception e_default 'Client ID is emptry';

  if (rep_date is null) then
    exception e_default 'Date is emptry';

  res_risk_categ = 0; -- тзначально считаю что уровень риска "Стандартный"

  -- если клиент найден в списке экстремистов, тогд уровень риска - "Критический"
  if (exists(select *
             from ft_info_stoplist fis
             where fis.fisl_from_c_id = :c_id and
                   :rep_date between fis.fisl_date_begin and fis.fisl_date_end)) then
  begin
    found_source = 'blacklist';
    res_risk_categ = 2;
  end
  else
  begin
    if (exists(select *
               from finpolis_639_cli_link fcl -- если найдено совпадение по перечню 639, тогда уровень риска - Высокий
               where fcl.fcl_from_c_id = :c_id
               and fcl.fcl_date <= :rep_date
               and :rep_date < '01.02.2021'
               and fcl.fcl_date < '01.02.2021')) then
    begin
      found_source = '639_list';
      res_risk_categ = 1;
    end
    else
    if (exists(select cl.c_citizen
               from clients cl
               where cl.c_id = :c_id and
                     c_citizen = 12)) then -- Если гражданин Таджикистана, тогда уровень риска - Высокий
    begin
      found_source = 'citizen';
      res_risk_categ = 1;
    end
    else if (exists(select *
             from finpolis_639_cli_link fcl -- если найдено совпадение по перечню 639, тогда уровень риска - Высокий



             where fcl.fcl_from_c_id = :c_id
             and fcl.fcl_date <= :rep_date
             and :rep_date >= '01.02.2021'
             and fcl.fcl_date < '01.02.2021')) then
    begin
      found_source = '639_list_0102_update';
      res_risk_categ = 0;
    end
    /*
    else if (exists(select *
             from finpolis_639_cli_link fcl -- если найдено совпадение по перечню 639, тогда уровень риска - Высокий
             where fcl.fcl_from_c_id = :c_id
             and fcl.fcl_date <= :rep_date
             and :rep_date >= '01.02.2021'
             and fcl.fcl_date >= '01.02.2021')) then
    begin
      found_source = '639_list_0102';
      res_risk_categ = 0;

    end
    */
  end

  suspend;
end
