CREATE PROCEDURE CIB_SET_STATUS(
    CIB_NAME STR_100,
    POCKET_ID INTEGER,
    POCKET_EXTERNAL_STATUS STR_100)
RETURNS (
    POCKET_INTERNAL_STATUS INTEGER)
AS
begin
  if (cib_name = 'ISHENIM') then
  begin
    pocket_internal_status = (case pocket_external_status
                                when 'Success' then 1
                                when 'FinishedWithErrors' then 1
                                when 'IncompleteBatch' then -1
                                when 'NotWellFormed' then -1
                                when 'BatchRejected' then -1
                                else -2
                              end);

    for select cp.cp_id
        from cib_pocket cp
        where cp.cp_id = :pocket_id
        into :pocket_id
    do
    begin
      update cib_pocket cp
      set cp_pocket_status = :pocket_internal_status,
          cp.cp_pocket_kib_status = :pocket_external_status
      where cp.cp_id = :pocket_id;

      update cib_pocket cp
      set cp_pocket_status = 2
      where cp.cp_id <> :pocket_id and
            cp_pocket_status <= 0;
    end
  end
  if (cib_name = 'EQUIFAX') then
  begin
    pocket_internal_status = (case pocket_external_status
                                when 'Success' then 1
                                when 'FinishedWithErrors' then 1
                                when 'BatchRejected' then -1
                                else -2
                              end);
    update cib_pocket cp
    set cp_pocket_status = :pocket_internal_status,
        cp.cp_pocket_kib_status = :pocket_external_status
    where cp.cp_id = :pocket_id;
  end
end
