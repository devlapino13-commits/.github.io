CREATE PROCEDURE HP_GC_TRANSFER_PERFORM(
    IGA_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    NUM_TRANS D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable etj_id d_id;
declare variable ifa_id d_integer;
declare variable igt_id d_integer;
declare variable igt_account_bank_id d_integer;
declare variable igt_account d_str_20;
declare variable igt_amount d_currency;
declare variable obj_kind d_integer;
declare variable request_json_check d_str_2048;
declare variable request_json_pay d_str_2048;
begin
  err_text = '';
  num_trans = 0;

  select
    ifa_id
  from ipc_gc_analiz
    inner join ipc_file_analiz
      on ifa_id = iga_from_ifa_id
        and ifa_file_type = 1
        and ifa_status >= 2
  where iga_id = :iga_id
  into
    :ifa_id;

  if (ifa_id is not null) then
  begin
    request_json_check =
      '{"TerminalCashInValidate":{"MSISDN": "0220018174", "MSISDN2": "%1", "Token": "", "CultureInfo": "ru-RU", "Password": "[pass]", "InstrumentType": "Wallet"}}';
    request_json_pay =
      '{"TerminalCashInReplenish":{"MSISDN": "0220018174", "MSISDN2": "%1", "Token": "", "CultureInfo": "ru-RU", "Password": "[pass]", "Amount": "%2", "ExtRefID": "%3qr", "InstrumentType": "Wallet"}}';
    obj_kind = f_objkind('ipc_gc_transfer', 'id');
    for
      select
        igt_id,
        igt_account_bank_id,
        igt_account,
        igt_amount
      from ipc_gc_transfer
      where igt_from_iga_id = :iga_id
        and igt_status = 0
        and igt_db_error_id is null
        and igt_do_transfer = 1
    into
      :igt_id,
      :igt_account_bank_id,
      :igt_account,
      :igt_amount
    do
    begin
      etj_id = gen_id(gen_elsom_transfer_job_id, 1);
      insert into elsom_transfer_job (
        etj_id,
        etj_object_kind,
        etj_object_id,
        etj_account_bank_id,
        etj_account,
        etj_amount,
        etj_request_json_check,
        etj_request_json_pay,
        etj_from_un_id
      )
      values (
        :etj_id,
        :obj_kind,
        :igt_id,
        :igt_account_bank_id,
        :igt_account,
        :igt_amount,
        f_format(:request_json_check, :igt_account),
        f_format(:request_json_pay, :igt_account, replace(cast(:igt_amount as decimal(18,2)), ',', '.'), :etj_id),
        :un_id
      );
      num_trans = num_trans + 1;
      update ipc_gc_transfer set
        igt_do_transfer = 0,
        igt_status = 1
      where igt_id = :igt_id;
    end
    update ipc_file_analiz set
      ifa_status = iif(ifa_status > 3, ifa_status, 3),
      ifa_from_un_id = :un_id,
      ifa_datetime = current_timestamp
    where ifa_id = :ifa_id;
  end
  if (num_trans = 0) then
    err_text = 'Записи для отправки на перевод через Elsom не определены';
  else
    post_event 'new_elsom_transfer';
end
