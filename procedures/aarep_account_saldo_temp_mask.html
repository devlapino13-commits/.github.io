CREATE PROCEDURE AAREP_ACCOUNT_SALDO_TEMP_MASK(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE,
    REPORT_BG_ID VARCHAR(20),
    REPORT_ACC_NO INTEGER)
RETURNS (
    ACCOUNT VARCHAR(20),
    SUMMA_VAL DECIMAL(18,4),
    SUMMA_EKV DECIMAL(18,4),
    MAX_DATE DATE)
AS
declare variable sql_text varchar(2048);
begin
  if (report_acc_no is null) then
    report_acc_no = -1;
  if (report_bg_id = '') then
    report_bg_id = null;

  sql_text = 'select al_id from acc_list, region_list(:report_region_id, 1, 3, 1) where al_from_reg_id=id_list';
  if (report_acc_no > -1) then
    sql_text = sql_text || ' and al_number=' || report_acc_no;
  if (report_bg_id is not null) then
  begin
    if (char_length(report_bg_id) = 5) then
      sql_text = sql_text || ' and al_from_bg_id=''' || report_bg_id || '''';
    else
      sql_text = sql_text || ' and al_from_bg_id starting ''' || report_bg_id || '''';
  end
  for execute statement(:sql_text)(report_region_id := :report_region_id)
          into :account
  do
  begin
    summa_ekv = f_al_bal(account, report_date_end);
    summa_val = f_al_bal(account, report_date_end, 'val');

    select obs_date
    from obsummary
    where obs_from_al_id = :account and
          obs_date <= :report_date_end
    order by obs_from_al_id desc, obs_date desc
    rows 1
    into :max_date;

    if ((summa_val <> 0) or (summa_ekv <> 0)) then
      suspend;
  end
end
