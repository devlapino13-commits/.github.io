CREATE PROCEDURE HP_ELSOM_OPERATIONS_FILL
AS
declare variable k_id id_type;
declare variable c_id id_type;
declare variable amount curr_type;
declare variable un_id id_type;
declare variable ol_id id_type;
declare variable ol_id_child id_type;
declare variable id_no id_type;
declare variable reg_id integer;
declare variable rep_date date;
declare variable fio str_100;
declare variable kontract_no str_100;
declare variable acc_dt varchar(20);
declare variable acc_kt varchar(20);
declare variable bish_office integer;
declare variable bish_go integer;
declare variable esm_id id_type;
begin
  bish_office = 1060100;
  bish_go = 1090100;

  -- формирую проводки
  for select em.esm_id, em.esm_from_k_id, em.esm_date
      from extsysmain em
      where em.esm_project = 'elsom' and
            em.esm_status = 0 and
            em.esm_action = 'pay'
      into :esm_id, :k_id, :rep_date
  do
  begin
    update extsysmain em
    set em.esm_status = 1,
        em.esm_result_id = 1
    where em.esm_id = :esm_id;

    select kpt.kpr_from_k_id, kpt.kpt_seller_id, kpt.kpt_amount, kpt.kpt_from_un_id
    from kontpaytasks kpt
    where kpt.kpt_from_esm_id = :esm_id
    into :k_id, :c_id, :amount, :un_id;

    select kont.k_last_reg_id, kont.k_kontract_no
    from kontract kont
    where kont.k_id = :k_id
    into reg_id, kontract_no;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into fio;

    /*
      1.  ¬ офисе, где идет финансирование по продукту »ѕ‘ проходит проводка
          ƒебет счет = 21199/99 (счет об¤зательств »ѕ‘)
           редит счет = 11510/4 (счет лоро/ностро Ѕишкекский офис)
      */
    select kv
    from get_kv(:reg_id, 5)
    into :id_no;
    ol_id = :id_no;

    insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete,
                          ol_user_id, ol_document_client)
    values (:ol_id, :reg_id, 2, :rep_date, 9,
            'Оплата поставщику (' || :fio || ') за товар по договору ' || :kontract_no,
            0, 0, :un_id, :c_id);

    select al_id
    from find_acc_list('21199/99')
    where al_from_reg_id = :reg_id
    into :acc_dt;

    select ln_account
    from loronostro
    where ln_region_loro = :reg_id and
          ln_region_nostro = :bish_office
    into :acc_kt;

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                          ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                          ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:ol_id, 1, '417', 1, :c_id, :acc_dt, :amount, :amount, null, null, null, :acc_kt, null,
            'Элсом Оплата за товар поставщику', null, :rep_date, 1, :reg_id);

    select kv
    from get_kv(:bish_office, 5)
    into :id_no;
    ol_id_child = :id_no;

    insert into operlist (ol_id, ol_parent_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status,
                          is_delete, ol_user_id, ol_document_client)
    values (:ol_id_child, :ol_id, :bish_office, 2, :rep_date, 9,
            'Оплата поставщику (' || :fio || ') за товар по договору ' || :kontract_no,
            0, 0, :un_id, :c_id);

    acc_dt = '';
    acc_kt = '';

    select ln_account
    from loronostro
    where ln_region_loro = :bish_office and
          ln_region_nostro = :reg_id
    into :acc_dt;

    select ln_account
    from loronostro
    where ln_region_loro = :bish_office and
          ln_region_nostro = :bish_go
    into :acc_kt;

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                          ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                          ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:ol_id_child, 1, '417', 1, null, :acc_dt, :amount, :amount, null, null, null, :acc_kt, null,
            'Элсом оплата за товар поставщику', null, :rep_date, 1, :bish_office);
    /*
      4. ¬ √ќ проходит проводка:
      ƒебет счет = 11512 (счет лоро/ностро Ѕишкекский офис)
       редит счет = 10101/100 (р/сч ЁЋ—ќћ в KICB)
      */

    select kv
    from get_kv(:bish_go, 5)
    into :id_no;
    ol_id_child = :id_no;

    insert into operlist (ol_id, ol_parent_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status,
                          is_delete, ol_user_id, ol_document_client)
    values (:ol_id_child, :ol_id, :bish_go, 2, :rep_date, 9,
            'Оплата поставщику (' || :fio || ') за товар по договору ' || :kontract_no,
            0, 0, :un_id, :c_id);

    acc_dt = '';
    acc_kt = '';

    select ln_account
    from loronostro
    where ln_region_loro = :bish_go and
          ln_region_nostro = :bish_office
    into :acc_dt;

    select al_id
    from find_acc_list('10101/100')
    where al_from_reg_id = :bish_go
    into :acc_kt;

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                          ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                          ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:ol_id_child, 1, '417', 1, null, :acc_dt, :amount, :amount, null, null, null, :acc_kt, null,
            'Элсом оплата за товар поставщику', null, :rep_date, 1, :bish_go);
  end
end
