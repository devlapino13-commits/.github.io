CREATE PROCEDURE HP_FIND_CLIENTS(
    FIND_C_ID DECIMAL(18,0),
    FIND_FIO VARCHAR(100),
    FIND_BIRTH_DATE DATE,
    FIND_CODE VARCHAR(100),
    FIND_PASS_DATE DATE,
    FIND_PASS_NO VARCHAR(100),
    FIND_CONT_INFO VARCHAR(100),
    FIND_USER_ID DECIMAL(18,0),
    FIND_CITIZEN INTEGER = 0,
    FIND_STATUS D_STR_100 = '',
    IS_PHYS_FACE D_SMALLINT = -1)
RETURNS (
    FC_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    IS_CRITICAL INTEGER,
    RES_SAME_FIO INTEGER,
    RES_SAME_BIRTH_DATE INTEGER,
    RES_SAME_PASS_NO INTEGER,
    RES_SAME_PASS_DATE INTEGER,
    RES_SAME_CONTACT INTEGER,
    RES_SAME_CODE INTEGER,
    RES_COMMENT VARCHAR(255),
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_TYPE INTEGER,
    C_INN VARCHAR(100),
    C_FULL_ADDRESS VARCHAR(255),
    C_STATUS VARCHAR(100),
    C_VERIFY_STATUS INTEGER,
    C_HAS_PHOTO INTEGER,
    C_HAS_SCAN_PASS INTEGER,
    C_PLASTCARD_NO VARCHAR(100),
    C_EMAIL VARCHAR(100),
    C_FROM_FL_ID DECIMAL(18,0),
    C_SEX INTEGER,
    RES_IDENTIFY_KOEF INTEGER,
    C_IS_PUBLIC D_SMALLINT)
AS
declare variable c_blob_photo_id decimal(18,0);
declare variable c_blob_pass1_id decimal(18,0);
declare variable critical_found integer;
declare variable cm_database_id integer;
declare variable pos d_integer;
declare variable sign_qty d_integer;
declare variable sign_qty_first_word d_integer;
begin
  c_full_address = '';
  critical_found = 0;
  is_critical = 0;

  cm_database_id = f_get_db_id();
  if (find_user_id is null) then
    find_user_id = 0;

  if (find_c_id is null) then
    find_c_id = 0;
  if (find_fio is null) then
    find_fio = '';
  if (coalesce(find_birth_date, '01.01.1900') <= '01.01.1900') then
    find_birth_date = null;
  if (coalesce(find_pass_date, '01.01.1900') <= '01.01.1900') then
    find_pass_date = null;
  if (find_code is null) then
    find_code = '';
  if (find_pass_no is null) then
    find_pass_no = '';
  if (find_cont_info is null) then
    find_cont_info = '';
  if (find_user_id is null) then
    find_user_id = 0;
  if (right('000000' || find_code, 6) = '000000') then
    find_code = '';
  if (right('000000' || find_pass_no, 6) = '000000') then
    find_pass_no = '';
  if (right(find_cont_info, 6) = '000000') then
    find_cont_info = '';

  if (is_phys_face = 0 and find_fio <> '' and find_fio not similar to '[-[:DIGIT:]]+') then -- 0 физ лицо, 1 юр лицо, -1 всё вместе
  begin
    pos = position(' ', find_fio); --Находим первый пробел в ФИО
    if (pos = 0) then
      exit;
    sign_qty_first_word = char_length(left(find_fio, pos - 1));
    if (sign_qty_first_word < 4) then     -- по заданию Игоря добавил условие для проверки первого слова
    begin                                  --Если первое слово меньше трёх символов, то проверяем общее количество символов в find_fio
      if (char_length(find_fio) <= 8) then   -- Если количество символов в find_fio меньше 9, то exit;.
      exit;                                  --То есть количество символов в find_fio должно быть больше или равно 8
    end 
    else
    if (pos = 1) then
    begin
      exit;
    end 
    else
    if (coalesce(pos,0) = 0) then
    begin
      exit;
    end 
    else
    if (coalesce(pos,0) <> 0) then
    begin
      sign_qty = char_length(substring(:find_fio from :pos + 1)); -- Считаем количество символов после пробела, если символов нет то exit
      if (sign_qty < 4) then
       exit;
    end
  end 

  if (is_phys_face = 1 and find_fio <> '' and find_fio not similar to '[-[:DIGIT:]]+') then
  begin
    if (character_length(find_fio) < 7) then
      exit;
  end 

  if (find_c_id = 0 and
      find_fio = '' and
      find_birth_date is null and
      find_code = '' and
      find_pass_date is null and
      find_pass_no = '' and
      find_cont_info = '' and
      find_user_id > 0) then
  begin
-- Если критерии поиска отсутствуют, показываем последние для пользователя
    for with tabs
        as (select cul_from_c_id, max(cul_datetime) as cul_datetime
            from clientlast
            where cul_from_un_id = :find_user_id
            group by cul_from_c_id
            order by 2 desc)
        select c_id
        from clients, tabs
        where c_id = cul_from_c_id and
              iif(:is_phys_face = -1, true, c_type = :is_phys_face)
        order by cul_datetime desc
        rows 100
        into :fc_id
    do
    begin
      c_id = fc_id;
      res_comment = '';

      select c_from_fl_id, c_name_fio, c_sex, c_pass_no, c_birth_date, c_type, c_inn,
             (select cont_value
              from hp_contdata_get_cell_valid(0, :c_id)),
             (select cont_value
              from hp_contdata_get_email_valid(0, :c_id)), c_verify_status, c_blob_photo_id, c_blob_pass1_id,
              c_is_public
      from clients
      where c_id = :c_id
      into c_from_fl_id, :c_name_fio, :c_sex, :c_pass_no, :c_birth_date, :c_type, :c_inn, :c_plastcard_no, :c_email,
           c_verify_status, c_blob_photo_id, c_blob_pass1_id, c_is_public;

      if (c_blob_photo_id > 0) then
        c_has_photo = 1;
      else
        c_has_photo = 0;
      if (c_blob_pass1_id > 0) then
        c_has_scan_pass = 1;
      else
        c_has_scan_pass = 0;

      select c_status
      from hp_find_client_status(:fc_id)
      into :c_status;

      if (fc_id <> 1000000000000 and
          fc_id <> 0) then
      begin
        if (find_status = '' or c_status like find_status) then
        suspend;
      end
    end

    exit;
  end

  -- если проверка только по ФИО и ФИО длиной менее 3 символов тогда exit;
  if (find_c_id = 0 and
      (character_length(find_fio) < 3) and
      find_birth_date is null and
      find_code = '' and
      find_pass_date is null and
      find_pass_no = '' and
      find_cont_info = '' and
      find_user_id > 0) then
  begin
    exit;
  end

-- Если в качестве ФИО переданы только числовые значения, ищем по ID
  if (find_c_id = 0 and
      find_fio <> '' and
      find_fio similar to '[-]?[[:DIGIT:]]+' and
      find_birth_date is null and
      find_code = '' and
      find_pass_date is null and
      find_cont_info = '' and
      find_pass_no = '' and
      character_length(find_fio) >= 7) then
  begin
    for select c_id
        from clients
        where c_id = :find_fio
        into :fc_id
    do
    begin
      c_id = fc_id;
      res_comment = '';

      select c_from_fl_id, c_name_fio, c_sex, c_pass_no, c_birth_date, c_type, c_inn,
             (select cont_value
              from hp_contdata_get_cell_valid(0, :c_id)),
             (select cont_value
              from hp_contdata_get_email_valid(0, :c_id)), c_verify_status, c_blob_photo_id, c_blob_pass1_id,
              c_is_public
      from clients
      where c_id = :c_id
      into c_from_fl_id, :c_name_fio, :c_sex, :c_pass_no, :c_birth_date, :c_type, :c_inn, :c_plastcard_no, :c_email,
           c_verify_status, c_blob_photo_id, c_blob_pass1_id, c_is_public;

      if (c_blob_photo_id > 0) then
        c_has_photo = 1;
      else
        c_has_photo = 0;
      if (c_blob_pass1_id > 0) then
        c_has_scan_pass = 1;
      else
        c_has_scan_pass = 0;

      select c_status
      from hp_find_client_status(:fc_id)
      into :c_status;

      if (fc_id <> 1000000000000 and
          fc_id <> 0) then
      begin
        if (find_status = '' or c_status like find_status) then
        suspend;
      end
      exit;
    end
  end

-- Если в качестве ФИО переданы только числовые значения и тире, ищем по номеру телефона, номер группы или договора
  if (find_c_id = 0 and
      find_fio <> '' and
      find_fio similar to '[-[:DIGIT:]]+' and
      find_birth_date is null and
      find_code = '' and
      find_pass_date is null and
      find_cont_info = '' and
      find_pass_no = '' and
      character_length(find_fio) >= 7) then
  begin
    for select fc_id
        from fm_find_client_tmp(:find_fio)
        into :fc_id
    do
    begin
      c_id = fc_id;
      res_comment = '';

      select c_from_fl_id, c_name_fio, c_sex, c_pass_no, c_birth_date, c_type, c_inn,
             (select cont_value
              from hp_contdata_get_cell_valid(0, :c_id)),
             (select cont_value
              from hp_contdata_get_email_valid(0, :c_id)), c_verify_status, c_blob_photo_id, c_blob_pass1_id,
              c_is_public
      from clients
      where c_id = :c_id
      into c_from_fl_id, :c_name_fio, :c_sex, :c_pass_no, :c_birth_date, :c_type, :c_inn, :c_plastcard_no, :c_email,
           c_verify_status, c_blob_photo_id, c_blob_pass1_id, c_is_public;

      if (c_blob_photo_id > 0) then
        c_has_photo = 1;
      else
        c_has_photo = 0;
      if (c_blob_pass1_id > 0) then
        c_has_scan_pass = 1;
      else
        c_has_scan_pass = 0;

      select c_status
      from hp_find_client_status(:fc_id)
      into :c_status;

      if (fc_id <> 1000000000000 and
          fc_id <> 0) then
      begin
        if (find_status = '' or c_status like find_status) then
        suspend;
      end
    end

    exit;
  end

  -- в других случаях ищем через полноценный поиск
  for select c_id, max(is_critical), max(res_same_fio), max(res_same_birth_date), max(res_same_pass_no),
             max(res_same_pass_date), max(res_same_contact), max(res_same_code)
      from hp_find_clients_tmp(:find_c_id, :find_fio, :find_birth_date, :find_code, :find_pass_date, :find_pass_no, :find_cont_info, 0, :find_citizen, :is_phys_face)
      group by c_id
      order by 2 desc, 4 desc
      into :c_id, :is_critical, :res_same_fio, :res_same_birth_date, :res_same_pass_no, :res_same_pass_date,
           :res_same_contact, :res_same_code
  do
  begin
    res_identify_koef = 0;
    res_comment = '';
    c_plastcard_no = '';
    c_email = '';
    fc_id = c_id;
    res_comment = '';
    res_identify_koef = res_same_fio + 1;
    if (res_same_fio = 0) then
      res_comment = res_comment || ', ФИО похоже';
    if (res_same_fio = 1) then
      res_comment = res_comment || ', ФИО очень похож';
    if (res_same_fio = 2) then
      res_comment = res_comment || ', ФИО одинаково';
    if (res_same_code = 1) then
    begin
      if (cm_database_id <> 94401) then
        res_comment = res_comment || ', ИНН';
      else
        res_comment = res_comment || ', Пинкод';
      is_critical = 1; -- Контрольно, т.к. в hp_find_clients_tmp это предусмотрено
    end
    if (res_same_birth_date = 1) then
      res_comment = res_comment || ', день рождения';
    if (res_same_pass_no = 1) then
      res_comment = res_comment || ', паспорт';
    if (res_same_pass_date = 1) then
      res_comment = res_comment || ', выдача паспорта';
    if (res_same_contact = 1) then
      res_comment = res_comment || ', контакт.данные';

    res_identify_koef = res_identify_koef + res_same_code + res_same_birth_date + res_same_pass_no + res_same_pass_date + res_same_contact + is_critical * 2;

    if (is_critical = 1) then
      critical_found = 1;

    if (res_comment <> '') then
    begin
      if (is_critical = 1) then
        res_comment = 'Критично: ' || substring(res_comment from 3);
      else
        res_comment = 'Сходство: ' || substring(res_comment from 3);
    end

    select c_from_fl_id, c_name_fio, c_sex, c_pass_no, c_birth_date, c_type, c_inn,
           (select cont_value
            from hp_contdata_get_cell_valid(0, :c_id)),
           (select cont_value
            from hp_contdata_get_email_valid(0, :c_id)), c_verify_status, c_blob_photo_id, c_blob_pass1_id,
            c_is_public
    from clients
    where c_id = :c_id
    into c_from_fl_id, :c_name_fio, :c_sex, :c_pass_no, :c_birth_date, :c_type, :c_inn, :c_plastcard_no, :c_email,
         c_verify_status, c_blob_photo_id, c_blob_pass1_id, c_is_public;

    if (c_blob_photo_id > 0) then
      c_has_photo = 1;
    else
      c_has_photo = 0;
    if (c_blob_pass1_id > 0) then
      c_has_scan_pass = 1;
    else
      c_has_scan_pass = 0;

    select c_status
    from hp_find_client_status(:fc_id)
    into :c_status;

    if (fc_id <> 1000000000000 and
        fc_id <> 0) then
    begin
      if (critical_found <= is_critical) then
        if (find_status = '' or c_status like find_status) then
        suspend;
    end
 end
