CREATE PROCEDURE INN_QUERY_CREATE_BY_C_ID(
    C_ID D_ID,
    UN_ID D_ID,
    CREATION_STAGE D_STR_CODE)
AS
declare variable c_name_fio d_str_100;
declare variable birth_date d_date;
declare variable pk_id d_integer;
declare variable doc_no d_str_code;
declare variable doc_date d_str_code;
declare variable first_name d_str_code;
declare variable last_name d_str_code;
declare variable middle_name d_str_code;
declare variable name_tail d_str_code;
declare variable eic_id d_id;
declare variable eil_id d_id;
declare variable doc_type d_integer;
declare variable c_alternat_fio d_str_100;
declare variable cfg d_str_20;
declare variable pass_seria d_str_10;
declare variable pass_number d_str_code;
declare variable inn_ru_source d_integer;
begin
    if ((select can_query_inn from inn_query_status(:c_id)) = 0) then
        exit;

    inn_ru_source = f_get_const('inn_ru_source');

    if (creation_stage in ('miniapp_created', 'from_ks')) then
    begin
      select c_name_fio,
             c_birth_date,
             c_alternat_fio
      from clients
      where c_id = :c_id
      into c_name_fio,
           birth_date,
           c_alternat_fio;
    end
    else if (creation_stage = 'verif_query_created') then
    begin
      -- по идее тут должна быть другая логика, т.к. при отправке на верификацию не факт чтот клиентски еданные уже в таблице clients возможно они находятся в буфере.
      -- модель персональные данные
      -- и если операировать данными только из cliens и clientdoс мы не получим актальные данные для запроса.
      select pte.pte_name_fio,
             pte.pte_birth_date,
             pte.pte_alternat_fio
      from perstemp pte
      where pte.pte_from_c_id = :c_id
      into c_name_fio,
           birth_date,
           c_alternat_fio;
    end

    select txt1, txt2, txt3
    from hp_fio_clear(:c_name_fio)
    into last_name, first_name, middle_name;
    name_tail = '';

    if (c_name_fio like '%УУЛУ%') then
      name_tail = 'УУЛУ';
    if (c_name_fio like '%КЫЗЫ%') then
      name_tail = 'КЫЗЫ';
    if (c_name_fio like '%УГЛИ%') then
       name_tail = 'УГЛИ';
    if (c_name_fio like '%КИЗИ%') then
      name_tail = 'КЫЗЫ';

    if (name_tail <> '') then
      if (middle_name <> '') then
        middle_name = middle_name || ' ' || name_tail;
      else
        last_name = last_name || ' ' || name_tail;

    for select distinct
        pk_id,
        cds_doc_no,
        f_datetostr(cds_doc_date_begin)
    from clientdocs cds
    inner join passkind on cds_from_pk_id = pk_id
    inner join passdoctype on pk_from_pdt_id = pdt_id
    where cds_from_c_id = :c_id
    and pdt_is_main_doc = 1
    and cds_verify_status = 2
    and cds.cds_is_delete = 0
    order by cds.cds_doc_date_begin desc
    rows 1
    into pk_id, doc_no, doc_date
    do
    begin
      -- Полный справочник документов:
      -- Паспорт гражданина СССР - "01"
      -- Свидетельство о рождении - "03"
      -- Паспорт иностранного гражданина - "10"
      -- Вид на жительство в России - "12"
      -- Разрешение на временное проживание в России - "15"
      -- Свидетельство о предоставлении временного убежища на территории России - "19"
      -- Паспорт гражданина России - "21"
      -- Свидетельство о рождении, выданное уполномоченным органом иностранного государства - "23"
      -- Вид на жительство иностранного гражданина - "62"
  
      doc_no = replace(doc_no, ' ', '');
      if (coalesce(doc_no, '') = '') then
        continue;

      select pass_seria,
             pass_number
      from split_passport_number(:pk_id, :doc_no) -- RFCKG-3780 Разбивка на серию и номер паспорта
      into :pass_seria,
           :pass_number;

      -- Паспорт/Загранпаспорт РФ
      if (pk_id = 12) then
      begin
        doc_type = 21;
        --Серия и номер паспорта гражданина РФ имеют формат 99 99 9999990, где:
        --9 – десятичная цифра (обязательная к вводу);
        --0 – десятичная цифра (необязательная, может отсутствовать).
        doc_no = left(doc_no, 2) || ' ' || substring(doc_no from 3 for 2) || ' ' || substring(doc_no from 5);
      end
      else
        doc_type = 10;
    end

    if (coalesce(doc_no, '') = '') then
    begin
      if (creation_stage <> 'from_ks') then
        exit;
      else
        exception e_message('Для запроса ИНН требуется верифицированный паспорт клиента');
    end

    if(coalesce(birth_date, '') = '') then
    begin
      if (creation_stage <> 'from_ks') then
        exit;
      else
        exception e_message('Для запроса ИНН требуется дата рождения клиента клиента');
    end

    if(coalesce(c_name_fio, '') = '') then
    begin
      if (creation_stage <> 'from_ks') then
        exit;
      else
        exception e_message('Для запроса ИНН требуется Имя клиента клиента');
    end

    -- если нет родительской записи, создаю ее
    if (not exists (select * from inn_check_main icm where icm.icm_from_c_id = :c_id)) then
    begin
        insert into inn_check_main (icm_from_c_id, icm_is_finished, icm_inn_received)
        values (:c_id, 0, 0);
    end

    for select eic_code, eic_id
    from extinfocfg
    where (:inn_ru_source in (1,3) and eic_code = 'inn_smev')
      or (:inn_ru_source in (2,3) and eic_code = 'inn_gnivc')
    into cfg, eic_id
    do
    begin
      -- создается запрос на получени ИНН
      execute procedure hp_extinfo_insert(:cfg, 'clients', :c_id, 0, :un_id);

      select eil.eil_id
      from extinfolist eil
      where eil.eil_from_c_id = :c_id
      and eil.eil_from_eic_id = :eic_id
      order by eil.eil_id desc
      rows 1
      into :eil_id;

      -- создаю дочернюю запись
      insert into inn_check_queries (icq_from_c_id, icq_fname, icq_sname, icq_patronimic, icq_datebirth, icq_doc_type,
                                     icq_doc_no, icq_doc_date, icq_creation_stage, icq_status, icq_inn_received,
                                     icq_from_eil_id, icq_alternat_fio, icq_pass_seria, icq_pass_number, icq_info_source)
      values (:c_id, :first_name, :last_name, :middle_name, :birth_date, :doc_type, :doc_no,
              :doc_date, f_enum_id('inn_request_source', :creation_stage), 0, 0, :eil_id, :c_alternat_fio, :pass_seria, :pass_number, iif(:cfg = 'inn_gnivc', 0, 1));

      post_event cfg;
    end
end
