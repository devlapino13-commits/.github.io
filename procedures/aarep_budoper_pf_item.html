CREATE PROCEDURE AAREP_BUDOPER_PF_ITEM(
    REP_REGION_ID INTEGER,
    REP_BIT_ID INTEGER,
    REP_WITH_RESERV INTEGER,
    REP_DATE DATE)
RETURNS (
    BOI_MONTH INTEGER,
    BOI_FACT DECIMAL(18,4),
    BOI_PLAN_REAL DECIMAL(18,4),
    BOI_PLAN_OPTIM DECIMAL(18,4),
    BOI_PLAN_PESSIM DECIMAL(18,4),
    BOI_FACT_ALL DECIMAL(18,4),
    BOI_ALL_REAL DECIMAL(18,4),
    BOI_ALL_OPTIM DECIMAL(18,4),
    BOI_ALL_PESSIM DECIMAL(18,4))
AS
declare variable mm integer;
declare variable fact_all decimal(18,4);
begin
  for select boi_month, sum(boi_fact), sum(boi_plan_real), sum(boi_plan_optim), sum(boi_plan_pessim), sum(boi_fact_all),
             sum(boi_all_real), sum(boi_all_optim), sum(boi_all_pessim)
      from aarep_budoper_pf_item_tmp(:rep_region_id, :rep_bit_id, :rep_with_reserv, :rep_date)
      group by boi_month
      order by boi_month
      into :boi_month, :boi_fact, :boi_plan_real, :boi_plan_optim, :boi_plan_pessim, :boi_fact_all, :boi_all_real,
           :boi_all_optim, :boi_all_pessim
  do
  begin
    mm = extract(month from :rep_date);
    if (boi_month = mm) then
      fact_all = boi_fact_all;
    if (boi_month > mm) then
      boi_fact_all = fact_all;
    suspend;
  end
end
