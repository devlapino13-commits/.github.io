CREATE PROCEDURE AAREP_OBJECT_ONE_CONTINFO(
    REP_OBJECT_KIND VARCHAR(100),
    REP_OBJECT_ID DECIMAL(18,0),
    REP_CONTACT_KIND VARCHAR(100),
    REP_IS_SECURED INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    RD_ID INTEGER,
    RD_CODE VARCHAR(255),
    RD_STR_ID VARCHAR(255),
    RD_NAME VARCHAR(255),
    RD_MEMO VARCHAR(255),
    RES_OBJECT_KIND INTEGER,
    RES_OBJECT_ID DECIMAL(18,0),
    RES_OBJECT_NAME VARCHAR(100),
    RES_VALUE_STATUS VARCHAR(100),
    RES_VALUE VARCHAR(100),
    RES_VALUE_ORIG VARCHAR(100),
    RES_VALUE_SHOW VARCHAR(100),
    RES_EDITMASK VARCHAR(255),
    RES_DATETIME TIMESTAMP,
    RES_STATUS_ID INTEGER,
    RES_EXAMPLE VARCHAR(255),
    RES_COMBO_MASK VARCHAR(255),
    RD_TAG INTEGER,
    RES_VERIFY_STATUS INTEGER,
    RES_IS_CRITICAL INTEGER,
    RES_NEED_VERIFY INTEGER,
    RES_JUST_EDITED INTEGER,
    RES_CLIENT_SEX INTEGER,
    RES_IS_EMPTY INTEGER,
    RES_ERROR_TEXT VARCHAR(255))
AS
declare variable ovs_from_un_id decimal(18,0);
declare variable cv_status integer;
declare variable object_kind_name varchar(100);
declare variable cv_error_rd_id integer;
declare variable cv_error_text varchar(255);
declare variable is_main d_smallint;
declare variable cont_kind d_str_code;
declare variable cda_id d_id;
declare variable new_contdata d_integer;
begin
  res_client_sex = -1;

  res_object_kind = f_objkind(rep_object_kind, 'id');
  object_kind_name = f_objkind(rep_object_kind, 'name');

  is_main = -1;
  -- rd_tag здесь цифра означает срок устарения верификации в днях
  if (rep_contact_kind similar to '[-]?[[:DIGIT:]]+') then
  begin
    select rd_id, rd_code, rd_str_id, rd_name, rd_memo, rd_tag
    from refdata
    where rd_id = :rep_contact_kind
    into rd_id, rd_code, rd_str_id, rd_name, rd_memo, rd_tag;
  end
  else
  begin
    select rd_id, rd_code, rd_str_id, rd_name, rd_memo, rd_tag
    from refdata
    where rd_code = :rep_contact_kind
    into rd_id, rd_code, rd_str_id, rd_name, rd_memo, rd_tag;
  end

  if (rd_id is null) then
    exception e_default 'Unknown contact kind';

  cont_kind = replace(rd_code, 'contactinfo.', '');
  if (cont_kind in ('cell_main', 'cell_add')) then
  begin
    if (cont_kind = 'cell_main') then
      is_main = 1;
    if (cont_kind = 'cell_add') then
      is_main = 0;
    cont_kind = 'cell';
  end

  rep_contact_kind = rd_id;

  res_object_id = :rep_object_id;

  if (rep_is_secured is null) then
    rep_is_secured = 0;

  if (object_kind_name = 'Clients') then
  begin
    select c_sex, c_name_fio
    from clients
    where c_id = :res_object_id
    into :res_client_sex, res_object_name;
  end
  else
  begin
    res_object_name = (case object_kind_name
                         when 'Contacts' then (select cc_name || iif(coalesce(cc_status, '') <> '', ' (' || cc_status || ')', '')
                                               from contacts
                                               where cc_id = :res_object_id)
                         when 'Region' then (select reg_code || ' ' || reg_name
                                             from region
                                             where reg_unicode = :res_object_id)
                         else ''
                       end);
  end

  res_just_edited = 0;
  res_status_id = 0;

  res_editmask = '';
  res_value = null;
  res_datetime = null;
  res_value_orig = res_value;
  res_value_status = 'Пусто';
  res_value_show = '-';
  res_example = '';
  res_is_empty = 0;

  new_contdata = f_get_const('new_contdata');
  new_contdata = 2;

  if (new_contdata = 0 or object_kind_name <> 'Clients') then
  begin
    select res_value, res_status_id, res_datetime, res_un_id
    from hp_get_client_continfo(:res_object_id, :rd_id)
    into :res_value, :res_status_id, :res_datetime, ovs_from_un_id;
  end
  else
  begin
    select cda_id, f_cont_code(cda_from_clt_id), cda_status * 2, clg_datetime, clg_from_un_id
    from contdata, contlog
    where cda_from_c_id = :res_object_id and
          cda_cont_kind = f_enum('cont_kind', :cont_kind, 'id') and
          (:is_main = -1 or
          cda_is_main = :is_main) and
          cda_id = clg_from_cda_id and
          cda_is_delete = 0
    order by clg_id desc
    rows 1
    into :cda_id, :res_value, :res_status_id, :res_datetime, ovs_from_un_id;
  end

  if (datediff(minute, res_datetime, current_timestamp) <= 15) then
    res_just_edited = 1;

  if (res_value = '') then
    res_value_status = 'Отсутствует';
  if (res_value is null) then
  begin
    res_value = '';
    res_is_empty = 1;
  end
  res_value_orig = res_value;
  if (res_status_id is null) then
    res_status_id = 0;

  if (res_value <> '') then
  begin
    res_value_status = (case res_status_id
                          when -2 then 'Не подтвердился'
                          when -1 then 'Доработка'
                          when 0 then 'Заполнено'
                          when 1 then 'Проверяется'
                          when 2 then 'Подтвержден'
                          else 'Статус неизвестен'
                        end);
    res_value_status = trim(res_value_status);

    if (rep_is_secured = 1) then
      res_value_show = f_cont_format(res_value, cont_kind, 'secured');
    else
      res_value_show = res_value;
  end

  res_editmask = '';
  res_combo_mask = '';
  if (rd_str_id = 'cell') then
    res_editmask = '';
  if (rd_str_id = 'phone') then
    res_editmask = '';
  if (rd_str_id = 'messenger') then
    res_editmask = '';
  if (rd_str_id = 'email') then
    res_editmask = '[\w\-.]+@[\w\-]+(\.[\w\-]+)+';
  if (rd_str_id = 'other') then
    res_editmask = '[a-zA-Z][a-zA-Z0-9\.,\-_]{5,31}';

  if (rd_str_id = 'cell') then
    res_example = 'Введите в международном формате';
  if (rd_str_id = 'messenger') then
    res_example = 'Введите в международном формате';
  if (rd_str_id = 'phone') then
    res_example = 'Введите в международном формате';
  if (rd_str_id = 'email') then
    res_example = 'Введите email в необходимом формате';
  if (rd_str_id = 'other') then
    res_example = 'Введите значение в необходимом формате';

  if (rd_str_id = 'cell' or rd_str_id = 'phone' or rd_str_id = 'messenger') then
    res_combo_mask = '7;992;996;998;994;';
  if (rep_is_secured = 1 and
      (coalesce(un_id, 0) <> coalesce(ovs_from_un_id, 0))) then
    res_value = res_value_show;
  else
    res_value = res_value_orig;

  -- Данные, которых нужно верифицировать
  res_verify_status = -2;
  res_value_status = res_value_status || ' (Верификация не требуется)';
  res_is_critical = 0;
  res_need_verify = 0;

  if (rd_code = 'contactinfo.cell_main' or rd_code = 'contactinfo.cell_add' or rd_code = 'contactinfo.email') then
  begin
    res_need_verify = 1;
  end

  cv_error_rd_id = 0;
  cv_error_text = '';
  res_error_text = '';

  if (res_need_verify = 1) then
  begin
    res_is_critical = 1;
    res_value_status = '';

    if (res_value_orig <> '') then
    begin
      if (new_contdata = 0) then
      begin
        select cv_status, cv_error_rd_id, cv_error_text
        from contverify
        where cv_object_kind = :res_object_kind and
              cv_object_id = :res_object_id and
              cv_from_rd_id = :rd_id and
              cv_contact_value = :res_value_orig
        order by cv_id desc
        rows 1
        into cv_status, cv_error_rd_id, cv_error_text;
      end
      else
      begin
        if (res_status_id <> 2) then
        begin
          select cvr_status * 2, 0, ''
          from contveri
          where cvr_from_cda_id = :cda_id
          order by cvr_id desc
          rows 1
          into cv_status, cv_error_rd_id, cv_error_text;
        end else
        begin
          cv_status = 2;
          cv_error_rd_id = 0;
          cv_error_text = '';
        end
      end

      if (cv_status is null) then -- Значит ни разу не было верификации
      begin
        res_verify_status = 0;
        res_value_status = 'Требуется верификация (новые данные)';
      end
      else
      begin
        if (cv_status in (1, 2, 3)) then -- верифицирован
        begin
          if (cv_error_rd_id <> 0) then
          begin
            select rd_name
            from refdata
            where rd_id = :cv_error_rd_id
            into :res_error_text;
            if (coalesce(cv_error_text, '') <> '') then
              res_error_text = res_error_text || ': ' || cv_error_text;
          end
          else
          begin
            if (coalesce(cv_error_text, '') <> '') then
              res_error_text = :cv_error_text;
          end
          res_verify_status = 2;
          res_value_status = 'Успешно подтвержден';

          -- Если не верифицирован в ObjectProps, а в cv_status успешная верификация,
          -- Значит мы получили результат старой верификации, и нужно верифицировать заново
          -- Aschegolskiy 01.06.2020, временно, пока с Санжаром не разберемся
          if (res_status_id = 0) then
          begin
            res_verify_status = 0;
            res_value_status = 'Требуется верификация (новые данные)';
          end
        end

        if (cv_status = 0) then -- код отправлен, ожидается ввод кода
        begin
          res_verify_status = 1;
          res_value_status = 'Ожидается подтверждение кода';
        end

        if (cv_status = -2) then -- аннулирован
        begin
          res_verify_status = 0;
          res_value_status = 'Требуется верификация (ранее аннулирован)';
        end
        if (cv_status = -4) then -- была ошибка
        begin
          res_verify_status = 0;
          res_value_status = 'Требуется верификация (ранее была тех.ошибка)';
        end
      end
    end
    else
    begin
      if (res_is_empty = 1) then
      begin
        res_verify_status = -1;
        res_value_status = 'Пусто';
      end
      else
      begin
        res_verify_status = -1;
        res_value_status = 'Отсутствует';
      end
    end
    if (res_value_status <> '') then
    begin
      if (res_verify_status <> 2 and
          rd_code = 'Contactinfo.cell_main') then
        res_is_critical = 1;
      else
        res_is_critical = 0;
    end
  end

  -- res_verify_status -2  Верификация не требуется
  -- res_verify_status -1  Данные не заполнены
  -- res_verify_status  0  Требуется верификация
  -- res_verify_status  1  Ожидается ввод кода
  -- res_verify_status  2  Успешно подтвержден

  suspend;
end
