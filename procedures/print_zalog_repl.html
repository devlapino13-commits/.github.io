    K_KONTRACT_NO D_STR_20,
    K_KONTRACT_DATE D_DATE,
    K_LAST_REG_ID D_INTEGER,
    C_ID ID_TYPE,
    C_NAME_FIO D_STR_100,

    Z_ID1 D_INTEGER,
    Z_NAME1 D_STR_100,
    Z_NAME_FIO1 D_STR_255,
    Z_ADDRESS1 D_STR_255,
    Z_PHONE1 D_STR_100,
    Z_DOCUMENT1 D_STR_255,
    Z_REGISTRY_NO1 VARCHAR(50),
    Z_REGISTRY_DATE1 D_DATE,
    Z_REGISTRY_AGENCY1 VARCHAR(80),

    Z_MEMO2_1 D_STR_2048,
    Z_COST_MARKET1 CURR_TYPE,
    Z_COST_LIKVID1 CURR_TYPE,
    Z_COST_KONTRACT1 CURR_TYPE,
    Z_DATE_BEGIN1 D_DATE,
    Z_DATE_END1 D_DATE,
    C_ID1 ID_TYPE,
    C_NAME_FIO1 D_STR_100,
    C_ALTERNAT_FIO1 D_STR_255,
    C_BIRTH_DATE1 DATE,
    C_PASS_NO1 VARCHAR(30),
    C_PASS_DATE1 D_DATE,
    C_PASS_WHO_CODE1 D_STR_20,
    C_PASS_WHO1 D_STR_255,
    C_PASS_DATE_EXPIRE1 D_DATE,
    C_EMAIL1 D_STR_255,
    Z_ID2 D_INTEGER,
    Z_NAME2 D_STR_100,
    Z_NAME_FIO2 D_STR_255,
    Z_ADDRESS2 D_STR_255,
    Z_PHONE2 D_STR_100,
    Z_DOCUMENT2 D_STR_255,
    Z_REGISTRY_NO2 VARCHAR(50),
    Z_REGISTRY_DATE2 D_DATE,
    Z_REGISTRY_AGENCY2 VARCHAR(80),
    Z_MEMO1_2 VARCHAR(1024),
    Z_MEMO2_2 D_STR_2048,
    Z_COST_MARKET2 CURR_TYPE,
    Z_COST_LIKVID2 CURR_TYPE,
    Z_COST_KONTRACT2 CURR_TYPE,
    Z_DATE_BEGIN2 D_DATE,
    Z_DATE_END2 D_DATE,
    C_ID2 ID_TYPE,
    C_NAME_FIO2 D_STR_100,
    C_ALTERNAT_FIO2 D_STR_255,
    C_BIRTH_DATE2 DATE,
    C_PASS_NO2 VARCHAR(30),
    C_PASS_DATE2 D_DATE,
    C_PASS_WHO_CODE2 D_STR_20,
    C_PASS_WHO2 D_STR_255,
    C_PASS_DATE_EXPIRE2 D_DATE,
    C_EMAIL2 D_STR_255,
    STS_SERIES_NUMBER2 D_STR_255,
    STS_ISSUE_DATE2 D_STR_255,
    STS_AUTHORITY2 D_STR_255,
    STS_PLATE2 D_STR_255,
    PTS_SERIES_NUMBER2 D_STR_255,
    PTS_ISSUE_DATE2 D_STR_255,
    PTS_AUTHORITY2 D_STR_255,
    PTS_VIN2 D_STR_255,
    PTS_MODEL2 D_STR_255,

    PTS_CUSTOMS_ENTRY2 D_STR_255,
    PTS_CUSTOMS_LIMITATIONS2 D_STR_255,
    EDOC_DATE_CREATE D_DATE)
AS
declare variable z_source_z_id d_integer;
declare variable z_date_begin date;
declare variable z_date_end date;
declare variable drl_id id_type;
declare variable edl_id d_id;
begin
  select
    k_id,
    k_kontract_no,
    k_kontract_date,
    k_last_reg_id,
    z_date_begin,
    z_date_end,
    z_source_z_id
  from zalog
    inner join kontract
      on k_id = z_from_k_id
  where z_id = :z_id
    and z_subtype = 4
  into
    :k_id,
    :k_kontract_no,
    :k_kontract_date,
    :k_last_reg_id,
    :z_date_begin,
    :z_date_end,
    :z_source_z_id;

  if (row_count = 1) then
  begin
    if (z_source_z_id > 0) then
    begin
      z_id1 = z_source_z_id;
      z_id2 = z_id;
    end else
    begin
      z_id1 = z_id;
      z_id2 = (
        select first 1 z_id
        from zalog where z_source_z_id = :z_id order by z_id);
    end

    select cast(edl_create_datetime as date)
    from edoc_list
    inner join edoc_cfg on edc_id = edl_from_edc_id
    where edl_object_kind = 9 and
          edl_object_id = :k_id and
          edc_code = 'loan.agreement' and
          edl_status <> -1
    order by edl_create_datetime desc rows 1
    into edoc_date_create;

    if (z_id1 is not null and z_id2 is not null) then
    begin
      c_id = (
        select first 1 kc_from_c_id from knclient
        where kc_from_k_id = :k_id and kc_amount_approved > 0);
      c_name_fio = (
        select c_name_fio from clients
        where c_id = :c_id);

      -- QR-код электронной подписи клиента. Текст пустой если не подписан
      if (coalesce(add_params, '') <> '') then
      begin
        edl_id = f_json_str_get_value(add_params, 'edl_id', null);
        c_qr_code = f_edoc_qr_by_edl(edl_id, c_id);
      end

      -- old
      select
        z_name,
        z_name_fio,
        adl_address_str,
        z_phone,
        z_document,
        z_registry_no,
        z_registry_date,
        z_registry_agency,
        z_memo1,
        z_memo2,
        z_cost_market,
        z_cost_likvid,
        z_cost_kontract,
        z_date_begin,
        z_date_end,
        c_id,
        c_name_fio,
        c_name_eng,
        c_birth_date,
        c_pass_no,
        c_pass_date,
        c_pass_who_code,
        c_pass_who,
        c_pass_date_expire,
        e.cont_value
      from zalog
        inner join clients
          on c_id = z_from_c_id
        left join addrlist
          on adl_id = z_from_adl_id
        left join hp_contdata_get_email_main(0, c_id) as e
          on 1 = 1
      where z_id = :z_id1
      into
        z_name1,
        z_name_fio1,
        z_address1,
        z_phone1,
        z_document1,
        z_registry_no1,
        z_registry_date1,
        z_registry_agency1,
        z_memo1_1,
        z_memo2_1,
        z_cost_market1,
        z_cost_likvid1,
        z_cost_kontract1,
        z_date_begin1,
        z_date_end1,
        c_id1,
        c_name_fio1,
        c_alternat_fio1,
        c_birth_date1,
        c_pass_no1,
        c_pass_date1,
        c_pass_who_code1,
        c_pass_who1,
        c_pass_date_expire1,
        c_email1;
      -- new
      select
        z_name,
        z_name_fio,
        adl_address_str,
        z_phone,
        z_document,
        z_registry_no,
        z_registry_date,
        z_registry_agency,
        z_memo1,
        z_memo2,
        z_cost_market,
        z_cost_likvid,
        z_cost_kontract,
        z_date_begin,
        z_date_end,
        c_id,
        c_name_fio,
        c_name_eng,
        c_birth_date,
        c_pass_no,
        c_pass_date,
        c_pass_who_code,
        c_pass_who,
        c_pass_date_expire,
        e.cont_value
      from zalog
        inner join clients
          on c_id = z_from_c_id
        left join addrlist
          on adl_id = z_from_adl_id
        left join hp_contdata_get_email_main(0, c_id) as e
          on 1 = 1
      where z_id = :z_id2
      into
        z_name2,
        z_name_fio2,
        z_address2,
        z_phone2,
        z_document2,
        z_registry_no2,
        z_registry_date2,
        z_registry_agency2,
        z_memo1_2,
        z_memo2_2,
        z_cost_market2,
        z_cost_likvid2,
        z_cost_kontract2,
        z_date_begin2,
        z_date_end2,
        c_id2,
        c_name_fio2,
        c_alternat_fio2,
        c_birth_date2,
        c_pass_no2,
        c_pass_date2,
        c_pass_who_code2,
        c_pass_who2,
        c_pass_date_expire2,
        c_email2;
      -- доп.инфо для замены авто
      if (add_info = 1) then begin
        -- СТС
        drl_id = (
          select first 1
            cds_from_drl_id
          from clientdocs
          where cds_from_c_id = :c_id
            and cds_is_delete = 0
            and cds_from_drl_id > 0
            and cds_from_pk_id = 310
            and cds_create_datetime >= :z_registry_date2
          order by cds_doc_date_begin desc, cds_datetime desc);
        if (drl_id is not null) then begin
          sts_series_number2 = f_docrecog_value(:drl_id, 'doc_no'); --Серия, № документа
          sts_issue_date2 = f_docrecog_value(:drl_id, 'date_begin'); --Дата выдачи
          sts_authority2 = f_docrecog_value(:drl_id, 'authority'); --Кем выдано
          sts_plate2 = f_docrecog_value(:drl_id, 'plate'); --Регистрационный знак
        end
        -- ПТС
        drl_id = (
          select first 1
            cds_from_drl_id
          from clientdocs
          where cds_from_c_id = :c_id
            and cds_is_delete = 0
            and cds_from_drl_id > 0
            and cds_from_pk_id = 315
            and cds_create_datetime >= :z_registry_date2
          order by cds_doc_date_begin desc, cds_datetime desc);
        if (drl_id is not null) then begin
          pts_series_number2 = f_docrecog_value(:drl_id, 'doc_no'); --Серия, № документа
          pts_issue_date2 = f_docrecog_value(:drl_id, 'date_begin'); --Дата выдачи
          pts_authority2 = f_docrecog_value(:drl_id, 'authority'); --Кем выдан
          pts_vin2 = f_docrecog_value(:drl_id, 'vin'); --Идентификационный номер (VIN)
          pts_model2 = f_docrecog_value(:drl_id, 'model'); --Марка, модель ТС
          pts_type_name2 = f_docrecog_value(:drl_id, 'type_name'); --Наименование (тип ТС)
          pts_category2 = f_docrecog_value(:drl_id, 'category'); --Категория ТС (ABCD, прицеп)
          pts_issue_year2 = f_docrecog_value(:drl_id, 'issue_year'); --Год изготовления ТС
          pts_engine_no2 = f_docrecog_value(:drl_id, 'engine_no'); --Модель, № двигателя
          pts_chassis_no2 = f_docrecog_value(:drl_id, 'chassis_no'); --Шасси (рама) №
          pts_body_no2 = f_docrecog_value(:drl_id, 'body_no'); --Кузов (кабина, прицеп) №
          pts_body_color2 = f_docrecog_value(:drl_id, 'body_color'); --Цвет кузова (кабины, прицепа)
          pts_engine_power2 = f_docrecog_value(:drl_id, 'engine_power'); --Мощность двигателя л.с. (кВт)
          pts_engine_capacity2 = f_docrecog_value(:drl_id, 'engine_capacity'); --Рабочий  объем двигателя, куб. см.
          pts_engine_type2 = f_docrecog_value(:drl_id, 'engine_type'); --Тип двигателя
          pts_environmental_class2 = f_docrecog_value(:drl_id, 'environmental_class'); --Экологический класс
          pts_max_weight2 = f_docrecog_value(:drl_id, 'max_weight'); --Разрешенная максимальная масса, кг
          pts_empty_weight2 = f_docrecog_value(:drl_id, 'empty_weight'); --Масса без нагрузки, кг
          pts_manufacturer2 = f_docrecog_value(:drl_id, 'manufacturer'); --Организация изготовитель ТС (страна)
          pts_approval_info2 = f_docrecog_value(:drl_id, 'approval_info'); --Одобрение типа ТС (№, дата, кем одобрено)
          pts_export_country2 = f_docrecog_value(:drl_id, 'exprort_country'); --Страна вывоза ТС
          pts_customs_entry2 = f_docrecog_value(:drl_id, 'customs_entry'); --Серия, №, ТД, ТПО
          pts_customs_limitations2 = f_docrecog_value(:drl_id, 'customs_limitations'); --Таможенные ограничения
        end
      end
      suspend;
    end
  end
end
