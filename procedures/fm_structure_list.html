CREATE PROCEDURE FM_STRUCTURE_LIST(
    REP_UN_NAME D_STR_CODE,
    REP_C_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REP_ITEM_ID D_INTEGER,
    REP_PARENT_ITEM D_INTEGER,
    REP_ID D_STR_CODE,
    REP_TITLE D_STR_255,
    REP_NEXT_ACTION D_STR_CODE,
    REP_ADD_PARAMS D_STR_LONG,
    REP_HTML D_BLOB_TEXT,
    REP_ITEM_KIND D_INTEGER,
    UN_ID D_ID,
    REP_OBJ_KIND D_STR_CODE,
    REP_OBJ_ID D_STR_CODE,
    REP_GRID_HEADERS D_STR_8100,
    REP_SQL D_STR_8100)
AS
declare variable pos_1 d_integer;
declare variable rep_parent_id d_str_code;
declare variable main_id d_integer;
declare variable tmp_str d_str_long;
declare variable sql_html d_str_8100;
declare variable sql_list d_str_8100;
begin
  pos_1 = 1;

  if (rep_c_id is null) then
    rep_c_id = 1000000000001;

  rep_item_kind = 0;
  rep_grid_headers = '';

  select un_id
  from username
  where un_name = :rep_un_name
  into un_id;

  if (un_id is null) then
    exception e_message f_format('Пользоватен не найден: %1', coalesce(rep_un_name, '<Null>'));

  un_id = (select un_id
           from hp_test_un_id(:un_id));

  rep_id = 'client_info';
  rep_add_params = '';
  rep_title = 'Окно детали клиента';
  rep_next_action = 'html';
  rep_parent_item = null;
  rep_item_id = pos_1;
  rep_parent_id = null;

  rep_obj_kind = 'clients';
  rep_obj_id = rep_c_id;

  sql_html = 'select * from fm_report_html_main(:un_id, :rep_id, :obj_kind, :obj_id, :date_begin, :date_end, :add_params, 0, :un_ip)';
  sql_list = 'select * from fm_report_param_list(:un_id, :date_begin, :date_end, :rep_id, :rep_obj_kind, :rep_obj_id, '''')';

  rep_sql = '';
  if (rep_next_action = 'html') then
    rep_sql = sql_html;

  if (rep_next_action = 'list') then
    rep_sql = sql_list;

  rep_sql = replace(rep_sql, ':un_id', un_id);
  rep_sql = replace(rep_sql, ':rep_id', '''' || rep_id || '''');
  rep_sql = replace(rep_sql, ':obj_kind', '''' || rep_obj_kind || '''');
  rep_sql = replace(rep_sql, ':rep_obj_kind', '''' || rep_obj_kind || '''');
  rep_sql = replace(rep_sql, ':obj_id', rep_obj_id);
  rep_sql = replace(rep_sql, ':rep_obj_id', rep_obj_id);
  rep_sql = replace(rep_sql, ':date_begin', '''' || f_datetostr(rep_date_begin) || '''');
  rep_sql = replace(rep_sql, ':date_end', '''' || f_datetostr(rep_date_end) || '''');
  rep_sql = replace(rep_sql, ':add_params', '''' || rep_add_params || '''');

  rep_html = '';

  suspend;

  rep_next_action = '';
  rep_parent_item = rep_item_id;
  rep_parent_id = rep_id;

  for select 'loans', 'Мои займы'
  from rdb$database
  union
  select 'reports', 'Мои отчеты'
  from rdb$database
  into rep_id, rep_title
  do
  begin
    rep_sql = 'select * from fm_report_list(:un_id, :menu_kind)';
    rep_sql = replace(rep_sql, ':un_id', un_id);
    rep_sql = replace(rep_sql, ':menu_kind', '''' || rep_id || '''');

    rep_item_kind = 1;
    rep_grid_headers = '';
    rep_next_action = 'list';
    rep_parent_item = null;
    rep_parent_id = null;
    pos_1 = pos_1 + 1;
    rep_item_id = pos_1;
    main_id = rep_item_id;
    suspend;
  
    rep_sql = '';
    rep_parent_id = rep_id;
    for select lower(rep_id), rep_title, lower(next_action)
    from fm_report_list(:un_id, :rep_parent_id)
    into :rep_id, :rep_title, :rep_next_action
    do
    begin
      pos_1 = pos_1 + 1;
      rep_add_params = '';
      rep_parent_item = null;
      rep_item_id = pos_1;
      rep_parent_item = main_id;
      rep_grid_headers = '';
      rep_item_kind = 0;

      rep_sql = '';
      if (rep_next_action = 'html') then
        rep_sql = sql_html;
    
      if (rep_next_action = 'list') then
      begin
        rep_sql = sql_list;
        rep_item_kind = 1;
      end
    
      rep_sql = replace(rep_sql, ':un_id', un_id);
      rep_sql = replace(rep_sql, ':rep_id', '''' || rep_id || '''');
      rep_sql = replace(rep_sql, ':obj_kind', '''' || rep_obj_kind || '''');
      rep_sql = replace(rep_sql, ':rep_obj_kind', '''' || rep_obj_kind || '''');
      rep_sql = replace(rep_sql, ':obj_id', rep_obj_id);
      rep_sql = replace(rep_sql, ':rep_obj_id', rep_obj_id);
      rep_sql = replace(rep_sql, ':date_begin', '''' || f_datetostr(rep_date_begin) || '''');
      rep_sql = replace(rep_sql, ':date_end', '''' || f_datetostr(rep_date_end) || '''');
      rep_sql = replace(rep_sql, ':add_params', '''' || rep_add_params || '''');

      if (rep_next_action = 'list') then
      begin
        for select next_action, next_rep_id,
               title_1, title_2, title_3, title_4,
               width_1, width_2, width_3, width_4,
               align_1, align_2, align_3, align_4
        from fm_report_param_view(:rep_id, :un_id)
        as cursor cur
        do
        begin
          tmp_str = f_json_make_pair('next_action', cur.next_action, 'str');
          rep_grid_headers = tmp_str;

          tmp_str = f_json_make_pair('next_rep_id', cur.next_rep_id, 'str');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('title_1', cur.title_1, 'str');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('title_2', cur.title_2, 'str');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('title_3', cur.title_3, 'str');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('title_4', cur.title_4, 'str');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('width_1', cur.width_1, 'num');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('width_2', cur.width_2, 'num');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('width_3', cur.width_3, 'num');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('width_4', cur.width_4, 'num');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('align_1', cur.align_1, 'num');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('align_2', cur.align_2, 'num');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('align_3', cur.align_3, 'num');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          tmp_str = f_json_make_pair('align_4', cur.align_4, 'num');
          rep_grid_headers = rep_grid_headers || ',' || tmp_str;

          rep_grid_headers = '{' || rep_grid_headers || '}';
        end
      end

      suspend;
    end
  end
end
