CREATE PROCEDURE CM_KG_0088_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kc_from_c_id d_id;
declare variable loan_purpose d_str_255;
declare variable bst_name d_str_100;
declare variable pr_code d_str_100;
begin
  for select KC_FROM_C_ID, C_NAME_FIO, RD.RD_NAME, BST.BS_NAME, pr.pr_code
      from KNCLIENT KC
      inner join CLIENTS C on C.C_ID = KC.KC_FROM_C_ID
      inner join BUSSUBTYPE BST on BS_ID = KC.KC_BUSUNESS_SUBTYPE
      inner join REFDATA RD on RD_ID = KC.KC_LOAN_TARGET
      inner join kontract kt on kt.k_id = kc.kc_from_k_id
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      where KC_FROM_K_ID = :K_ID and
            KC_AMOUNT_APPROVED > 0 and
            RD.RD_NAME <> BST.BS_NAME
      into KC_FROM_C_ID, res_info_text, LOAN_PURPOSE, BST_NAME, pr_code
  do
  begin
    --Если цель кредита «Потребительский» и тип бизнеса выбран не «Потребительский (другое)» тогда бдлк
    if (LOAN_PURPOSE = 'Потребительский' and
        BST_NAME <> 'Потребительский') then
    begin
      res_error_text = 'Для цели кредита «Потребительский» допустим только тип бизнеса «Потребительский (другое)»';
      res_error_is_critical = 1;
      suspend;
    end
    --Если цель кредита не «Потребительский» тогда в случае выбора типа бизнеса «Потребительский (другое)» блок
    if (LOAN_PURPOSE <> 'Потребительский' and
        BST_NAME = 'Потребительский') then
    begin
      res_error_text = 'Тип бизнеса «Потребительский (другое)» допустим только для цели бизнеса «Потребительский»';
      res_error_is_critical = 1;
      suspend;
    end
  end
end
