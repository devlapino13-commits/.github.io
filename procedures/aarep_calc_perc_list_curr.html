CREATE PROCEDURE AAREP_CALC_PERC_LIST_CURR(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_K_ID DECIMAL(18,0),
    AAREP_BALANCE_END DECIMAL(18,4),
    AAREP_KONTRACT_NO VARCHAR(20),
    AAREP_CLIENT_NAME VARCHAR(100),
    AAREP_IS_INDIV VARCHAR(10),
    AAREP_PRODUCT_NAME VARCHAR(50),
    AAREP_DISB_DATE DATE,
    AAREP_PERC_END DECIMAL(18,4),
    KF_NAME VARCHAR(100),
    KF_REGION_NAME VARCHAR(50),
    KF_BRANCH_NAME VARCHAR(50),
    AAREP_OUT_PERC_END DECIMAL(18,4))
AS
declare variable curr_prior varchar(3);
declare variable curr_code varchar(3);
declare variable curr_rate decimal(18,4);
declare variable k_curr_rate decimal(18,4);
begin
  curr_prior = '000';
  for select aarep_k_id, aarep_balance_end, aarep_kontract_no, aarep_client_name, aarep_is_indiv, aarep_product_name,
             aarep_disb_date, aarep_perc_end, k_currency, k_index_curr_rate, kal_new_kf_name,
             region.reg_code || ' ' || region.reg_name, parent_name,

             aarep_out_perc_end

      from aarep_calc_perc_list(:rep_region_id, :rep_date_begin, :rep_date_end, '---'), kontract, knaktlist, region, region_list(:rep_region_id, 1, 3, 1)
      where k_id = aarep_k_id and
            ((:report_currency = '---') or
            (:report_currency = '000') or
            (k_currency = :report_currency)) and
            kal_from_kf_id = k_from_kf_id and
            :rep_date_end between kal_date and kal_date_end and
            kal_new_reg_id = reg_unicode and
            (reg_unicode = id_list)
      order by region.parent, reg_unicode
      into :aarep_k_id, :aarep_balance_end, :aarep_kontract_no, :aarep_client_name, :aarep_is_indiv,
           :aarep_product_name, :aarep_disb_date, :aarep_perc_end, :curr_code, :k_curr_rate, :kf_name, :kf_region_name,
           :kf_branch_name, :aarep_out_perc_end
  do
  begin
    if (report_currency = '000') then
    begin
      if (curr_prior <> :curr_code) then
      begin
        select cr_rate
        from opercurr
        where cr_from_cu_id = :curr_code and
              cr_date <= :rep_date_end and
              cr_date_end >= :rep_date_end
        into :curr_rate;
        if (curr_rate is null) then
          curr_rate = 1;
        curr_prior = :curr_code;
      end
      if (curr_rate >= k_curr_rate) then
        aarep_balance_end = cast(:aarep_balance_end * :curr_rate as decimal(15,2));
      else
        aarep_balance_end = cast(:aarep_balance_end * :k_curr_rate as decimal(15,2));
      aarep_perc_end = cast(:aarep_perc_end * :curr_rate as decimal(15,2));
      aarep_out_perc_end = cast(:aarep_out_perc_end * :curr_rate as decimal(15,2));
    end
    suspend;
  end
end
