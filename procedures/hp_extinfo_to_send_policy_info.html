CREATE PROCEDURE HP_EXTINFO_TO_SEND_POLICY_INFO(
    C_ID ID_TYPE,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    JSON_TEXT D_STR_8100)
AS
declare variable CREDIT_NO D_STR_20; /* номер кредита */
declare variable POLICY_NO D_STR_20; /* аналогичен номеру кредита */
declare variable ISSUE_DATE D_DATE; /* дата выдачи кредита */
declare variable EMPLOYEE_ID D_ID; /* фамилия сотрудника выдавшего кредит */
declare variable EMPLOYEE_LAST_NAME D_STR_CODE; /* фамилия сотрудника выдавшего кредит */
declare variable EMPLOYEE_FIRST_NAME D_STR_CODE; /* имя сотрудника выдавшего кредит */
declare variable DATE_START D_DATE; /* дата начала действия полиса, аналогична с датой выдачи кредита */
declare variable DATE_END D_DATE; /* дата конца действия полиса, аналогична с датой конца займа */
declare variable DATE_DIFF D_INTEGER; /* дата конца действия полиса, аналогична с датой конца займа */
declare variable COMMISSION decimal(18,2); /* сумма комиссии (п. №6) */
declare variable INSURANCE_DATA D_STR_LONG;
declare variable ADDRESS D_STR_CODE; /* всегда "Кыргызская Республика" */
declare variable CLIENT D_STR_255; /* фамилия и имя клиента */
declare variable INN D_STR_20; /* ИНН клиента */
declare variable PASSPORT D_STR_20; /* Паспортные данные */
declare variable BIRTHDAY D_STR_20; /* Дата рождения */
declare variable TSP D_CURRENCY; /* Страховая сумма, аналогична сумме займа */
declare variable INSURANCE_SUMM D_CURRENCY; /* Страховая сумма, аналогична сумме займа */
declare variable IK_POLICY_NO_SPARE D_STR_20;
declare variable IK_TSP_SPARE D_CURRENCY;
begin
    json_text = '';
    if (object_kind <> f_objkind('kontract', 'id')
        and f_get_db_id() not in (41701, 39801)
        and f_get_const('kont_insurance_active') <> 1
    ) then
        exit;

    select
        k.k_kontract_no,
        k.k_fact_disb_date,
        k.k_kredline_end,
        iif(k.k_kredline_end - k.k_disb_date < 365, 365, k.k_kredline_end - k.k_disb_date),
        kc.kc_amount_approved,
        kal.kal_new_kf_officer
    from kontract k
    inner join knclient kc on kc.kc_from_k_id = k.k_id
        and kc.kc_from_c_id = :c_id
        and kc.kc_amount_approved > 0
    inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
        and current_date between kal.kal_date and kal.kal_date_end
        and kal.is_delete = 0
    where k.k_id = :object_id
    into
        :credit_no,
        :date_start,
        :date_end,
        :date_diff,
        :insurance_summ,
        :employee_id;

    select
        upper(substring(fl.fl_name1 from 1 for 1)) || lower(substring(fl.fl_name1 from 2)),
        upper(substring(fl.fl_name2 from 1 for 1)) || lower(substring(fl.fl_name2 from 2))
    from clients c
    inner join fiolist fl on fl.fl_id = c.c_from_fl_id
    where c.c_id = :employee_id
    into
        :employee_last_name,
        :employee_first_name;

    select
        c.c_name_fio,
        c.c_inn, 
        c.c_pass_no,
        c.c_birth_date
    from clients c
    where c.c_id = :c_id
    into
        :client, 
        :inn,
        :passport,
        :birthday;

    select coalesce(ag.ag_commis_cards_mb, ag.ag_commis_cards_other, 1)
    from agreements ag
    where current_date between ag.ag_date_begin and ag.ag_date_end
        and ag.ag_from_rd_id = f_get_rd_id('AgreementKinds.insurance')
        and ag.ag_object_kind = f_objkind('clients', 'id')
        and ag.ag_object_id = 1090101889505 /* один статичный c_id страхового агентства*/
        and ag.ag_status = 1
    into tsp;

    select commission_amount
    from get_policy_commission_amount(:object_id)
    into commission;

    select ik_policy_no_spare, ik_tsp_spare
    from insurance_kontract
    where ik_from_k_id = :object_id
    into ik_policy_no_spare, ik_tsp_spare;
    
    policy_no = credit_no;
    if (coalesce(ik_policy_no_spare, '') <> '') then -- RFCKG-4843
    begin
      policy_no = ik_policy_no_spare;
      tsp = ik_tsp_spare;
    end

    insurance_data =
    '{'||
        f_json_make_pair('adres', 'Кыргызская Республика', 'str')||','||
        f_json_make_pair('client', coalesce(:client, '-'), 'str')||','||
        f_json_make_pair('inn', coalesce(:inn, '-'), 'str')||','||
        f_json_make_pair('pasport', coalesce(:passport, '-'), 'str')||','||
        f_json_make_pair('birthday', f_date_format('yyyy-mm-dd', coalesce(:birthday, '01.01.1900')), 'str')||','||
        f_json_make_pair('insurance_summ', coalesce(:insurance_summ, 0), 'num')||','||
        f_json_make_pair('insurance_premium', coalesce(:commission, 0), 'num')||
    '}';

    json_text =
    '{'||
        f_json_make_pair('credit_no', coalesce(:credit_no, '-'), 'str')||','||
        f_json_make_pair('policy_no', coalesce(:policy_no, '-'), 'str')||','||
        f_json_make_pair('issue_date', f_date_format('yyyy-mm-dd', coalesce(:date_start, '01.01.1900')), 'str')||','||
        f_json_make_pair('product_id', 'c5231c2c-b78f-4c28-8c5f-acc467a06995', 'str')||','||
        f_json_make_pair('client', coalesce(:client, '-'), 'str')||','||
        f_json_make_pair('employee_last_name', coalesce(:employee_last_name, '-'), 'str')||','||
        f_json_make_pair('employee_first_name', coalesce(:employee_first_name, '-'), 'str')||','||
        f_json_make_pair('company_id', '24a5d050-3119-4183-b514-6b772be1c780', 'str')||','||
        f_json_make_pair('date_start', f_date_format('yyyy-mm-dd', coalesce(:date_start, '01.01.1900')), 'str')||','||
        f_json_make_pair('date_end', f_date_format('yyyy-mm-dd', coalesce(:date_end, '01.01.1900')), 'str')||','||
        f_json_make_pair('currency_id', '3f994c4c-afbc-404d-9266-7d48f0294467', 'str')||','||
        f_json_make_pair('username', 'admin', 'str')||','||
        f_json_make_pair('diffDays', coalesce(:date_diff, 0), 'num')||','||
        f_json_make_pair('commission', coalesce(:commission * :tsp / 100, 0), 'num')||','||
        f_json_make_pair('insurance_data', coalesce(:insurance_data, '{}'), 'object')||
    '}';

    suspend;
end
