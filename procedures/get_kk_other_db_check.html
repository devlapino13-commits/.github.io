CREATE PROCEDURE GET_KK_OTHER_DB_CHECK(
    REP_FIO STR_100,
    REP_BIRTH_DATE DATE,
    REP_INN STR_20,
    REP_PASS_NO VARCHAR(30),
    EXEC_FROM_SCRIPT D_SMALLINT = 0)
RETURNS (
    RES_TEXT D_STR_8100)
AS
declare variable c_id id_type;
declare variable c_name_fio str_100;
declare variable c_pass_no varchar(30);
declare variable c_pass_date date;
declare variable c_pass_who str_255;
declare variable adl_address_str str_255;
declare variable adc_address_type smallint;
declare variable address_fiz str_255;
declare variable address_yur str_255;
declare variable c_pass_date_text str_20;
declare variable c_birth_date date;
declare variable c_birth_date_str str_20;
declare variable ks_state smallint;
declare variable delinq_day integer;
declare variable k_id id_type;
declare variable max_delinq_day integer;
declare variable kol_kred smallint;
declare variable kol_active smallint;
declare variable kol_delinq_7 smallint;
declare variable sum_ostatok curr_type;
declare variable sum_zad curr_type;
declare variable is_poruch smallint;
declare variable is_zaemshik smallint;
declare variable cc_status str_100;
declare variable cc_name str_100;
declare variable cc_cell_no str_255;
declare variable res_kib d_str_8100;
declare variable db_id d_integer;
declare variable sql_request d_str_255;
declare variable state_str d_str_20;
declare variable ox_amount_plus curr_type;
begin
  db_id = f_get_db_id();
  sql_request = ('select res_text from get_kk_other_db_check(' || '''' || :rep_fio || ''', ' || '''' || cast(:rep_birth_date as date) || ''', ' || '''' || coalesce(:rep_inn, '') || ''', ' || '''' || coalesce(:rep_pass_no, '') || ''', ' || 1 || ')');
  if(exec_from_script = 0) then
  begin
      if (db_id in (81001, 39801, 41701)) then
      begin
        execute statement :sql_request
            with autonomous transaction
            on external (case db_id when 81001 then '192.168.200.250/3050:d:\finance\ib_base\db_1.ib'
                                    when 41701 then '10.1.2.11/3050:e:\ib_base\db_mbfru.ib'
                                    when 39801 then '192.168.200.41/3050:D:\DB\DB_KZ.IB'
                                    else '' end)
            as user 'SYSDBA' password 'F!ora80X'
            into :res_text;
        suspend;
        exit;
      end
  end
  else if (exec_from_script = 1) then
  begin
      res_text = '';
      rep_fio = trim(rep_fio);
      if (rep_fio = '') then
        rep_fio = null;
      rep_inn = replace(rep_inn, ' ', '');
      if (rep_inn = '') then
        rep_inn = null;
      rep_pass_no = replace(rep_pass_no, ' ', '');
      if (rep_pass_no = '') then
        rep_pass_no = null;
      for select c_id, c_name_fio, c_pass_no, -- № паспорта
                 c_pass_date, -- Дата паспорта
                 c_pass_who, -- Кем выдан
                 c_birth_date
          from clients
          where (c_name_fio = :rep_fio and
                c_birth_date = :rep_birth_date) or
                (c_inn = :rep_inn)
          union
          select c_id, c_name_fio, c_pass_no, -- № паспорта
                 c_pass_date, -- Дата паспорта
                 c_pass_who, -- Кем выдан
                 c_birth_date
          from clients, clientdocs
          where c_id = cds_from_c_id and
                cds_index_doc_no = :rep_pass_no
          into c_id, c_name_fio, c_pass_no, c_pass_date, c_pass_who, c_birth_date
      do
      begin
        c_birth_date_str = f_datetostr(c_birth_date);
        c_pass_no = coalesce(c_pass_no, 'Не указан');
        c_pass_date_text = f_datetostr(c_pass_date);
        c_pass_who = coalesce(c_pass_who, 'Не указан');
    
        address_fiz = null;
        address_yur = null;
        for select trim(adl_address_str), adc_address_type
            from addrclient, addrlist
            where adc_from_c_id = :c_id and
                  adc_address_type in (1, 2) and
                  adc_from_adl_id = adl_id and
                  adl_address_str is not null
            into adl_address_str, adc_address_type
        do
        begin
          if (adc_address_type = 1) then
            address_fiz = adl_address_str;
          else
            address_yur = adl_address_str;
        end
        address_fiz = coalesce(address_fiz, 'Не указан');
        address_yur = coalesce(address_yur, 'Не указан');
    
        res_text = res_text || 'ФИО клиента: <b>' || c_name_fio || '</b><br>';
        res_text = res_text || 'Дата рождения: <b>' || c_birth_date_str || '</b><br>';
        res_text = res_text || '№ паспорта: <b>' || c_pass_no || '</b><br>';
        res_text = res_text || 'Дата выдачи паспорта: <b>' || c_pass_date_text || '</b><br>';
        res_text = res_text || 'Кем выдан: <b>' || c_pass_who || '</b><br>';
        res_text = res_text || 'Фактический адрес: <b>' || address_fiz || '</b><br>';
        res_text = res_text || 'Юридический адрес: <b>' || address_yur || '</b><br><br>';
    --  6.  Наличие активных/закрытых займов – в формате кредитной истории (что бы было видно максимальное количество дней просрочки);
        res_text = res_text || iif(db_id = 81001,'<b>Кредитная история клиента в RU:</b><br>', '<b>Кредитная история клиента в KG:</b><br>');
        kol_kred = 0;
        kol_active = 0;
        kol_delinq_7 = 0;
        sum_ostatok = 0;
        for select ks_from_k_id, iif(current_date between ks_date_begin and ks_date_end, 1, 0)
            from knclient, kontstate
            where kc_from_c_id = :c_id and
                  kc_amount_approved > 0 and
                  kc_from_k_id = ks_from_k_id and
                  ks_state = 1
            order by ks_date_begin
            into k_id, ks_state
        do
        begin
          kol_kred = kol_kred + 1;
          kol_active = kol_active + ks_state;
    
          if (ks_state = 1) then
          begin
            delinq_day = null;
            select current_date - oks_delinq_date
            from oksummary
            where current_date between oks_date_begin and oks_date_end and
                  oks_from_k_id = :k_id  and
                  oks_close_state = 0
            into delinq_day;
            delinq_day = coalesce(delinq_day, 0);
    
            if (delinq_day > 7) then
              kol_delinq_7 = kol_delinq_7 + 1;
    
            sum_zad = null;
            select sum(zad_os + sh_calc_perc + sh_calc_fine)
            from aarep_make_zad_list(:k_id, current_date, 1)
            into sum_zad;
            sum_zad = coalesce(sum_zad, 0);
    
            sum_ostatok = sum_ostatok + sum_zad;
          end
    
    --    расчет максимально кол-во просроченных дней
          max_delinq_day = null;
          select max(ok_ol_date - ok_sh_date)
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_k_id = :k_id and
                ok_type in (4, 5) and
                ok_os + ok_perc < 0 and
                ok_ol_date > ok_sh_date
          into max_delinq_day;
          max_delinq_day = coalesce(max_delinq_day, 0);
          if (delinq_day > max_delinq_day) then
            max_delinq_day = delinq_day;
        end
        is_zaemshik = 0;
        if (kol_kred > 0) then
          is_zaemshik = 1;
        is_poruch = 0;
        if (exists(select *
                   from zalog, kontstate
                   where z_type = 0 and
                         z_from_c_id = :c_id and
                         z_from_k_id = ks_from_k_id and
                         ks_ol_is_delete = 0 and
                         ks_state = 1 and
                         current_date between ks_date_begin and ks_date_end)) then
          is_poruch = 1;
    
        if (is_zaemshik + is_poruch = 2) then
        begin
          res_text = res_text || 'Категория: <b>Заёмщик/Поручитель</b><br>';
        end
        else
        if (is_zaemshik = 1) then
        begin
          res_text = res_text || 'Категория: <b>Заёмщик</b><br>';
        end
        else
        if (is_poruch = 1) then
        begin
          res_text = res_text || 'Категория: <b>Поручитель</b><br>';
        end

         ---------Статус займа в просрочке ------ RFCKG-496
        state_str = 'закрытый';
        if (exists(select oks.oks_close_state
                   from oksummary oks
                   where oks.oks_from_k_id = :k_id and
                     current_date between oks.oks_date_begin and oks.oks_date_end
                     and  oks_close_state = 1)) then
           state_str = 'Спец.учет';

        if (exists(select  ks.ks_state
                 from kontstate ks
                 where ks.ks_from_k_id=:k_id
                   and current_date between ks.ks_date_begin and ks.ks_date_end
                   and ks.ks_ol_is_delete=0
                   and ks.ks_state in (1))) then
           state_str= 'Активный';
         -------Остаток суммы для закрытия просрочки------------RFCKG-496

        if (state_str = 'Спец.учет') then
        begin
          select sum (okf.oko_sh_os + okf.oko_sh_perc + okf.oko_sh_other)
          from operkroff okf
          where okf.oko_from_k_id = :k_id
          and okf.oko_ol_is_active = 1
          into sum_ostatok;

          sum_ostatok = coalesce (sum_ostatok,0);

          select sum(ox.ox_amount_plus)
          from operextra ox
          where ox.ox_object_kind = 9
            and ox.ox_object_id = :k_id
            and ox.ox_ol_date <= current_date
         into ox_amount_plus ;

         ox_amount_plus = coalesce (ox_amount_plus, 0);

        end
    
        res_text = res_text || 'Всего получено: <b>' || kol_kred || '</b><br>';
        res_text = res_text || '<font color="green">Из них активных: <b>' || kol_active || '</b></font><br>';
        res_text = res_text || 'Остаток суммы для закрытия по активному займу: <b>' || round(sum_ostatok, 2) || '</b><br>';
        res_text = res_text || '<font color="red">В просрочке более 7 дней: <b>' || kol_delinq_7 || '</b></font><br>';
        res_text = res_text || 'Статус займа в просрочке: <b>' || state_str || '</b><br>';
        res_text = res_text || 'Остаток суммы для закрытия по займу в просрочке : <b>' || round( sum_ostatok, 2)|| '</b><br>';
        res_text = res_text || 'Сумма задолжености судебных издержек : <b>' || round (ox_amount_plus,2) || '</b><br>';
        res_text = res_text || 'Макс. Кол-во просроченных дней: <b>' || max_delinq_day || '</b><br><br>';
        res_text = res_text || '<b>Контактные лица:</b><br>';
    
        for select cc_status, cc_name, cc_cell_no
            from contacts
            where cc_from_c_id = :c_id
            into cc_status, cc_name, cc_cell_no
        do
        begin
          res_text = res_text || :cc_status || ': <b>' || :cc_name || '</b>, телефон № :<b>' || :cc_cell_no || '</b><br>';
        end
    
    
        res_kib = '';
        select client_data from aarep_kib_basic_data(:c_id, 'client')
        into res_kib;
        res_kib = trim(res_kib);
        res_kib = coalesce(res_kib,'');
    
        res_text = res_text || '<br><b>Внешняя КИ:</b><br>';
        if (res_kib <> '') then res_text = res_text || res_kib;
    
      end
      suspend;
   end
end
