CREATE PROCEDURE LAST_DOC_NUMBER_SERVER(
    DD DATE,
    KIND INTEGER,
    REG_ID INTEGER,
    OL_ID DECIMAL(18,0))
RETURNS (
    NUM INTEGER)
AS
declare variable dn_code varchar(255);
declare variable dn_beg_type integer;
declare variable dn_region_kind integer;
declare variable branch_id integer;
declare variable dd_year integer;
declare variable dd_month integer;
declare variable max_n integer;
declare variable dn_id integer;
declare variable max_n2 integer;
declare variable reg_id2 integer;
begin
  /* Procedure Text */
  select dn_admin_opertypes, dn_begin_type, dn_region_kind
  from docname
  where dn_id = :kind
  into :dn_code, :dn_beg_type, dn_region_kind;
  if (dn_region_kind is null) then
    dn_region_kind = 0;
  dd_year = extract(year from :dd);
  dd_month = extract(month from :dd);
  if (dn_code is null) then
    dn_code = '';
  if (dn_code <> '') then
  begin
    num = 0;
    for select dn_id
        from docname
        where dn_admin_opertypes = :dn_code
        into :dn_id
    do
    begin
      max_n = 0;
      if (dn_region_kind = 0) then
      /* на уровне текущего офиса */
      begin
        if (dn_beg_type = 0) then
        /* Нумерация не обнуляется */
        begin
          select max(ol_doc_no)
          from operlist
          where is_delete = 0 and
                ol_doc_kind = :dn_id and
                ol_doc_no is not null and
                ol_from_reg_id = :reg_id and
                ol_id <> :ol_id
          into :max_n;
        end
        if (dn_beg_type = 1) then
        /* Обнуляется раз в год*/
        begin
          select max(ol_doc_no)
          from operlist
          where is_delete = 0 and
                ol_doc_kind = :dn_id and
                ol_doc_no is not null and
                ol_year = :dd_year and
                ol_from_reg_id = :reg_id and
                ol_id <> :ol_id
          into :max_n;
        end
        if (dn_beg_type = 2) then
        /* Обнуляется раз в Месяц*/
        begin
          select max(ol_doc_no)
          from operlist
          where is_delete = 0 and
                ol_doc_kind = :dn_id and
                ol_doc_no is not null and
                ol_year = :dd_year and
                ol_month = :dd_month and
                ol_from_reg_id = :reg_id and
                ol_id <> :ol_id
          into :max_n;
        end
        if (dn_beg_type = 3) then
        /* Обнуляется каждый день*/
        begin
          select max(ol_doc_no)
          from operlist
          where is_delete = 0 and
                ol_doc_kind = :dn_id and
                ol_doc_no is not null and
                ol_date_oper_day = :dd and
                ol_from_reg_id = :reg_id and
                ol_id <> :ol_id
          into :max_n;
        end
      end /* Конец DN_Region_Kind 0 */

      if (dn_region_kind = 1) then
      /* На уровне филиала */
      begin
        select reg_head_office
        from region
        where reg_unicode = :reg_id
        into :branch_id;

        for select reg_unicode
            from region
            where reg_head_office = :branch_id
            into :reg_id2
        do
        begin
          if (dn_beg_type = 0) then
          /* Нумерация не обнуляется */
          begin
            select max(ol_doc_no)
            from operlist
            where operlist.is_delete = 0 and
                  ol_doc_kind = :dn_id and
                  ol_doc_no is not null and
                  ol_from_reg_id = :reg_id2 and
                  ol_id <> :ol_id
            into :max_n2;
          end
          if (dn_beg_type = 1) then
          /* Обнуляется раз в год*/
          begin
            select max(ol_doc_no)
            from operlist
            where operlist.is_delete = 0 and
                  ol_doc_kind = :dn_id and
                  ol_doc_no is not null and
                  ol_year = :dd_year and
                  ol_from_reg_id = :reg_id2 and
                  ol_id <> :ol_id
            into :max_n2;
          end
          if (dn_beg_type = 2) then
          /* Обнуляется раз в Месяц*/
          begin
            select max(ol_doc_no)
            from operlist
            where operlist.is_delete = 0 and
                  ol_doc_kind = :dn_id and
                  ol_doc_no is not null and
                  ol_year = :dd_year and
                  ol_month = :dd_month and
                  ol_from_reg_id = :reg_id2 and
                  ol_id <> :ol_id
            into :max_n2;
          end
          if (dn_beg_type = 3) then
          /* Обнуляется каждый день*/
          begin
            select max(ol_doc_no)
            from operlist
            where operlist.is_delete = 0 and
                  ol_doc_kind = :dn_id and
                  ol_doc_no is not null and
                  ol_date_oper_day = :dd and
                  ol_from_reg_id = :reg_id2 and
                  ol_id <> :ol_id
            into :max_n2;
          end
          if (max_n2 is null) then
            max_n2 = 0;
          if (max_n2 > max_n) then
            max_n = :max_n2;
        end
      end /* Конец DN_Region_Kind 1 */

      if (dn_region_kind = 2) then
      /* на уровне компании */
      begin
        if (dn_beg_type = 0) then
        /* Нумерация не обнуляется */
        begin
          select max(ol_doc_no)
          from operlist
          where is_delete = 0 and
                ol_doc_kind = :dn_id and
                ol_doc_no is not null and
                ol_id <> :ol_id
          into :max_n;
        end
        if (dn_beg_type = 1) then
        /* Обнуляется раз в год*/
        begin
          select max(ol_doc_no)
          from operlist
          where is_delete = 0 and
                ol_doc_kind = :dn_id and
                ol_doc_no is not null and
                ol_id <> :ol_id and
                ol_year = :dd_year
          into :max_n;
        end
        if (dn_beg_type = 2) then
        /* Обнуляется раз в Месяц*/
        begin
          select max(ol_doc_no)
          from operlist
          where is_delete = 0 and
                ol_doc_kind = :dn_id and
                ol_doc_no is not null and
                ol_year = :dd_year and
                ol_month = :dd_month and
                ol_id <> :ol_id
          into :max_n;
        end
        if (dn_beg_type = 3) then
        /* Обнуляется каждый день*/
        begin
          select max(ol_doc_no)
          from operlist
          where is_delete = 0 and
                ol_doc_kind = :dn_id and
                ol_doc_no is not null and
                ol_date_oper_day = :dd and
                ol_id <> :ol_id
          into :max_n;
        end
      end /* Конец DN_Region_Kind 1 */
      if (max_n is null) then
        max_n = 0;
      if (max_n > num) then
        num = max_n;
    end
    if (num is not null) then
      num = num + 1;
    else
      num = 1;
  end
  suspend;
end
