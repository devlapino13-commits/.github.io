CREATE PROCEDURE UXP$DEMO_STAFF_FIND(
    FIND_TEXT D_STR_100)
RETURNS (
    CLI_ID D_INTEGER,
    CLI_TYPE D_SMALLINT,
    CLI_COUNTRY D_STR_CODE,
    CLI_NAME D_STR_100,
    CLI_DATE D_DATE,
    CLI_INN D_STR_CODE,
    CLI_ADDRESS D_STR_255,
    CLI_IS_STAFF D_BOOL,
    STF_TABEL_NO D_INTEGER,
    STF_FROM_REG_ID D_INTEGER,
    STF_JOB_NAME D_STR_100,
    STF_REGION_NAME D_STR_100)
AS
begin
  for select cli_id, cli_type, cli_country, cli_name, cli_date, cli_inn, cli_address, cli_is_staff
  from uxp$demo_client_find(:find_text, 0, 1)
  into :cli_id, :cli_type, :cli_country, :cli_name, :cli_date, :cli_inn, :cli_address, :cli_is_staff
  do
  begin
    stf_tabel_no = '';
    select stf_tabel_no, stf_from_reg_id, stf_job_name, reg_code || ' ' || reg_name
    from ux$demo_staff, region
    where stf_from_cli_id = :cli_id and
          stf_from_reg_id = reg_unicode
    into :stf_tabel_no, :stf_from_reg_id, :stf_job_name, stf_region_name;

    suspend;
  end
end
