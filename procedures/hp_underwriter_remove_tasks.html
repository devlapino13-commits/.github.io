CREATE PROCEDURE HP_UNDERWRITER_REMOVE_TASKS(
    UN_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    PROJECT_CODE VARCHAR(255),
    CURR_STATUS INTEGER)
AS
declare variable rd_id integer;
declare variable acl_id decimal(18,0);
declare variable mark str_255;
begin
  if (project_code = 'UnderWriter_Manager') then
  begin
    project_code = 'UnderWriter';
    mark = 'yes';
  end
  else
    mark = 'no';

  for select rd_id
      from refdata
      where rd_str_id = :project_code and
            rd_main_code = 'ActionGroup'
      into rd_id
  do
  begin
    for select aco_id
        from actionorder
        where aco_from_rd_id = :rd_id and
              aco_respons_c_id = :c_id and
              aco_status = :curr_status
        into :acl_id
    do
    begin
      update actionlist
      set acl_status = 0,
          acl_respons_c_id = null,
          acl_from_un_id = :un_id
      where acl_id = :acl_id;

      if (mark = 'yes') then
        delete from actionmark
        where atm_from_acl_id = :acl_id;
    end
  end
end
