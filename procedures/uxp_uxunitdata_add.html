CREATE PROCEDURE UXP$UXUNITDATA_ADD(
    UNIT_CODE D_STR_CODE,
    DATASOURCE_CODE D_STR_CODE,
    DATASET_CODE D_STR_CODE,
    CLASS_NAME D_STR_CODE,
    SQL_SELECT D_STR_4096,
    SQL_UPDATE D_STR_4096,
    SQL_INSERT D_STR_4096,
    SQL_DELETE D_STR_LONG,
    DS_READONLY D_BOOL)
AS
declare variable uds_uuid d_uuid;
begin
  if (trim(coalesce(dataset_code, '')) <> '') then
  begin
    update or insert into ux$unitdata (uds_unit_code, uds_dataset, uds_cls_name,
                                       uds_sql_select, uds_sql_update, uds_sql_insert, uds_sql_delete, uds_readonly)
    values (:unit_code, :dataset_code, :class_name,
            :sql_select, :sql_update, :sql_insert, :sql_delete, :ds_readonly)
    matching (uds_unit_code, uds_dataset)
    returning (uds_uuid)
    into uds_uuid;
  end

  update or insert into ux$unitsource (usr_unit_code, usr_datasource, usr_uds_uuid)
  values (:unit_code, :datasource_code, :uds_uuid)
  matching (usr_unit_code, usr_datasource);
end
