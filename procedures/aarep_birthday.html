CREATE PROCEDURE AAREP_BIRTHDAY(
    REP_DATE DATE,
    IS_TOP_STAFF SMALLINT = 1)
RETURNS (
    CATEGORY VARCHAR(20),
    EMPLOYEE_FIO VARCHAR(100),
    JL_NAME VARCHAR(100),
    REG_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    C_BIRTH_DATE DATE,
    YEARS SMALLINT,
    STAGE_MONTH DECIMAL(15,2),
    YEAR_STAGE SMALLINT,
    MONTH_STAGE SMALLINT,
    PAR_1 DECIMAL(15,2),
    PAR_30 DECIMAL(15,2),
    OK_CELL_MAIN VARCHAR(100))
AS
declare variable rep_date_begin date;
declare variable rep_date_end date;
declare variable ps_from_jl_id integer;
declare variable db_id integer;
declare variable end_date_last date;
declare variable oks_balance curr_type;
declare variable delinq_balance_30 curr_type;
declare variable delinq_balance_1 curr_type;
declare variable rb_type integer;
declare variable c_id decimal(18,0);
declare variable ps_from_reg_id integer;
declare variable jl_category varchar(50);
declare variable can_suspend smallint;
declare variable jk_system_code varchar(100);
begin

  :db_id = f_get_db_id();

  rep_date_begin = rep_date + 1;
  rep_date_end = rep_date + 14;
  end_date_last = rep_date - extract(day from rep_date); -- дата конец предыдущего месяца

  for select c_id, c_birth_date, ps_from_jl_id, ps_from_reg_id, c_name_fio
      from hr_pstatus, clients
      where :rep_date between ps_date_begin and ps_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            c_id <> 1010100405018 and
            ps_from_c_id = c_id and
            iif(extract(day from c_birth_date) = 29 and extract(month from c_birth_date) = 2,
            cast(extract(day from (c_birth_date+1))||'.'||extract(month from (c_birth_date+1))||'.'||extract(year from :rep_date) as date),
            cast(extract(day from c_birth_date)||'.'||extract(month from c_birth_date)||'.'||extract(year from :rep_date) as date)) between :rep_date_begin and :rep_date_end

      into c_id, c_birth_date, ps_from_jl_id, ps_from_reg_id, employee_fio
  do
  begin
    select jl_category, jl_name, jk_system_code
    from joblist, jobkind
    where jl_id = :ps_from_jl_id and
          jl_kind = jk_id
    into jl_category, jl_name, jk_system_code;
    can_suspend = 0;
    stage_month = 0;
    select lo_stage_month
    from hp_get_lo_stage_full(:c_id, :rep_date, null, 1)
    into stage_month;

    year_stage = trunc(cast(stage_month as double precision) / 12);
    month_stage = stage_month - year_stage * 12;

    par_30 = 0;
    par_1 = 0;

    if (jl_category in ('1 Топ менеджмент', '4 Менеджмент старшего звена', -- КГ
                    '1 Топ.менеджмент' -- РФ
                    )) then
    begin
      category = 'Категория 1';
      can_suspend = 1;
    end
    else
    if (jk_system_code = 'JobSystem.Region_KM') then
    begin
      category = 'Категория 2';
      if (stage_month >= 36) then
      begin
        if (db_id = 81001) then
        begin
          select sum(oks_balance), sum(iif(:end_date_last - oks_delinq_date between 31 and 360, oks_balance, 0))
          from oksummary
          where oks_region_id = :ps_from_reg_id and
                :end_date_last between oks_date_begin and oks_date_end
                and oks_close_state = 0 
          into oks_balance, delinq_balance_30;
          if (oks_balance > 0) then
            par_30 = cast(delinq_balance_30 as double precision) / oks_balance * 100;
          else
            par_30 = 0;

          if (par_30 < 8) then
            can_suspend = 1;
        end
        else
        begin
          select sum(oks_balance), sum(iif(:end_date_last - oks_delinq_date > 30, oks_balance, 0))
          from oksummary
          where oks_region_id = :ps_from_reg_id and
                :end_date_last between oks_date_begin and oks_date_end
                and oks_close_state = 0 
          into oks_balance, delinq_balance_30;
          if (oks_balance > 0) then
            par_30 = cast(delinq_balance_30 as double precision) / oks_balance * 100;
          else
            par_30 = 0;

          if (par_30 < 1) then
            can_suspend = 1;
        end
      end
    end
    else
    if (jk_system_code in ('JobSystem.LO_Senior', 'JobSystem.LO_Standart')) then
    begin
      category = 'Категория 3';
      if (stage_month >= 60) then
      begin
        if (db_id = 81001) then
        begin
          select sum(oks_balance), sum(iif(:end_date_last - oks_delinq_date between 31 and 360, oks_balance, 0))
          from oksummary
          where oks_kc_id = :c_id and
                :end_date_last between oks_date_begin and oks_date_end
                and oks_close_state = 0 
          into oks_balance, delinq_balance_30;
          if (oks_balance > 0) then
            par_30 = cast(delinq_balance_30 as double precision) / oks_balance * 100;
          else
            par_30 = 0;

          if (par_30 < 8) then
            can_suspend = 1;
        end
        else
        begin
          select sum(oks_balance), sum(iif(:end_date_last - oks_delinq_date >= 1, oks_balance, 0))
          from oksummary
          where oks_kc_id = :c_id and
                :end_date_last between oks_date_begin and oks_date_end
                and oks_close_state = 0 
          into oks_balance, delinq_balance_1;
          if (oks_balance > 0) then
            par_1 = cast(delinq_balance_1 as double precision) / oks_balance * 100;
          else
            par_1 = 0;

          if (par_1 < 1) then
            can_suspend = 1;
        end
      end
    end
    else
    begin
      category = 'Категория 4';
      if (stage_month >= 60) then
      begin
        rb_type = null;
        select rb_type
        from rebuke
        where rb_is_delete = 0 and
              rb_from_c_id = :c_id and
              :rep_date between rb_period_start and rb_period_end
        into rb_type;
        if (rb_type is null) then
          rb_type = 0;
        if (rb_type not in (1, 2)) then
          can_suspend = 1;
      end
    end
    if (can_suspend = 0) then
      category = 'Категория 5';
    if (can_suspend = is_top_staff and
        (is_top_staff = 1 or (is_top_staff = 0 and
        stage_month >= 12))) then
    begin
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :ps_from_reg_id and
            r1.parent = r2.reg_unicode
      into reg_name, branch_name;

      years = datediff(year, c_birth_date, rep_date_end);

      ok_cell_main = f_cont_info('clients', c_id, 'cell_main', 'ok');
      suspend;
    end

  end
end
