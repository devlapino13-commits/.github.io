CREATE PROCEDURE HP_UXFRAME_SQL_PARAM(
    UX_CODE D_STR_100,
    FR_CODE D_STR_CODE)
RETURNS (
    SQL_TEXT D_STR_LONG)
AS
declare variable ufd_source d_str_code;
declare variable ufp_name d_str_code;
declare variable field_type d_str_code;
declare variable is_table d_bool;
declare variable ufp_parent_code d_str_100;
declare variable param_name d_str_100;
declare variable ufp_editor d_str_code;
declare variable ufp_is_param d_bool;
begin
  -- Процедура строит искуственный SQL запрос из входящий параметров процедуры
  sql_text = '';
  for select ufd_source, ufp_origin_name, ufd_is_table, ufp_is_param, ufp_editor
  from uxfrparam, uxfrdata
  where ufp_ufr_code = :fr_code and
        ufp_parent_code is null and
        ufp_in_use = 1 and
        (ufp_is_param = 1 or (ufp_is_param = 0 and ufp_pk_field <> 0)) and
        ufp_ufr_code = ufd_ufr_code and
        ufd_kind = 0
  into :ufd_source, ufp_name, is_table, ufp_is_param, ufp_editor
  do
  begin
    if (is_table = 0) then
    begin
      if (exists(select * from uxcodeframe
      where ucf_ux_code = :ux_code and
            ucf_master_ufr_code = :fr_code and
            ucf_detail_field = :ufp_name)) then
        Continue;
    end

    ufd_source = upper(ufd_source);
    ufp_name = upper(ufp_name);

    if (is_table = 0) then
    begin
      for select trim(rdb$field_source)
      from rdb$procedure_parameters
      where rdb$procedure_name = :ufd_source and
            rdb$parameter_name = :ufp_name
      into :field_type
      do
      begin
        if (sql_text <> '') then
          sql_text = sql_text || ', ';
    
        sql_text = sql_text || f_format('cast(null as %1) as %2', field_type, trim(ufp_name));
      end
    end else
    begin
      for select trim(rdb$field_source)
      from rdb$relation_fields
      where rdb$relation_name = :ufd_source and
            rdb$field_name = :ufp_name
      into :field_type
      do
      begin
        if (sql_text <> '') then
          sql_text = sql_text || ', ';
    
        sql_text = sql_text || f_format('cast(null as %1) as %2', field_type, trim(ufp_name));
      end
    end

    if (ufp_is_param = 1 and ufp_editor in ('buttonedit', 'dialogedit')) then
    begin
      if (sql_text <> '') then
        sql_text = sql_text || ', ';

      sql_text = sql_text || f_format('cast(null as %1) as %2', 'D_STR_LONG', trim(ufp_name) || '$text');
    end
  end

  for select ufp_parent_code, ufp_origin_name
  from uxfrparam, uxfrdata
  where ufp_ufr_code = :fr_code and
        ufp_parent_code is not null and
        ufp_is_param = 1 and
        ufp_ufr_code = ufd_ufr_code and
        ufd_kind = 0
  into :ufp_parent_code, ufp_name
  do
  begin
    is_table = null;
    select ufp_origin_name, ulp_source, iif(ulp_source = ulp_table_name, 1, 0)
    from uxfrparam, uxfrlookup, uxlookup
    where ufp_code = :ufp_parent_code and
          ufl_ufp_code = ufp_code and
          ufl_ulp_code = ulp_code
    into param_name, ufd_source, is_table;

    if (is_table is null) then
      Continue;

    param_name = param_name || '$' || ufp_name;

    ufd_source = upper(ufd_source);
    ufp_name = upper(ufp_name);

    if (is_table = 0) then
    begin
      for select trim(rdb$field_source)
      from rdb$procedure_parameters
      where rdb$procedure_name = :ufd_source and
            rdb$parameter_name = :ufp_name
      into :field_type
      do
      begin
        if (sql_text <> '') then
          sql_text = sql_text || ', ';
    
        sql_text = sql_text || f_format('cast(null as %1) as %2', field_type, trim(param_name));
      end
    end else
    begin
      for select trim(rdb$field_source)
      from rdb$relation_fields
      where rdb$relation_name = :ufd_source and
            rdb$field_name = :ufp_name
      into :field_type
      do
      begin
        if (sql_text <> '') then
          sql_text = sql_text || ', ';
    
        sql_text = sql_text || f_format('cast(null as %1) as %2', field_type, trim(param_name));
      end
    end
  end


  if (sql_text <> '') then
  begin
    sql_text = lower(sql_text);
    sql_text = 'select ' || sql_text || ' from rdb$database';
  end
  suspend;
end
