CREATE PROCEDURE HP_COMBOFIELDS_RECREATE
AS
declare variable cf_table varchar(100);
declare variable cf_field varchar(100);
declare variable cf_id integer;
declare variable cf_sp_param varchar(255);
declare variable cf_master_field varchar(100);
declare variable db_id int;
begin
-- запускается роботом в 21:30
-- сначала заполнить справочники родителей
  db_id = f_get_db_id();
  cf_sp_param = '';
  for select cf_kind, cf_table_name, cf_field_name, cf_master_field_name
      from combofields
      where cf_is_ready = 1 and
            not(:db_id in (39801, 41701) and
            cf_kind in (24, 25))
      into :cf_id, :cf_table, :cf_field, cf_master_field
  do
  begin
    execute procedure hp_combofields_fill(:cf_id, :cf_table, :cf_field, :cf_sp_param, :cf_master_field);
  end
end
