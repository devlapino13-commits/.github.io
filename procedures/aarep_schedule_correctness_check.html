CREATE PROCEDURE AAREP_SCHEDULE_CORRECTNESS_CHECK(
    REP_DATE_END D_DATE)
RETURNS (
    KONTRACT_NO D_STR_100,
    KC_ID D_ID,
    C_ID D_STR_255,
    PR_NAME D_STR_255,
    K_AMOUNT_APPROVED CURR_TYPE,
    DEBT_OS CURR_TYPE,
    DEBT_PERC CURR_TYPE,
    OVERDUE_DEBT_OS CURR_TYPE,
    OVERDUE_DEBT_PERC CURR_TYPE,
    ACTIVE_DELINQ_DAYS D_INTEGER,
    K_FACT_DISB_DATE D_DATE,
    INIT_PLAN_CLOSE_DATE D_DATE,
    CURR_PLAN_CLOSE_DATE D_DATE,
    NEXT_PAY_DATE D_DATE,
    NEXT_PAY_AMOUNT CURR_TYPE,
    LAST_FUTURE_PAY_AMOUNT CURR_TYPE,
    PREV_FUTURE_OS_AMOUNT CURR_TYPE,
    FUTURE_NONZERO_CNT D_INTEGER,
    FUTURE_PERC_ONLY_CNT D_INTEGER,
    FUTURE_OS_PERC_CNT D_INTEGER,
    FUTURE_NEG_OS_SUM CURR_TYPE,
    FUTURE_NEG_PERC_SUM CURR_TYPE,
    TERMINATE_DATE D_DATE,
    OKS_DELINQ_DATE D_DATE,
    LAST_OVERDUE_PAY_AMOUNT CURR_TYPE,
    BASE_RATE DECIMAL(18,4),
    REDUCED_RATE DECIMAL(18,4),
    CURRENT_RATE DECIMAL(18,4),
    CURRENT_RATE_BEGIN_DATE D_DATE)
AS
declare variable sh_id d_id;
declare variable sh_date d_date;
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable s_os curr_type;
declare variable s_perc curr_type;
declare variable s_all curr_type;
declare variable pay_amount curr_type;
declare variable k_id d_id;
declare variable last_row_os curr_type;
declare variable v_last_os_pay curr_type;
declare variable v_prev_os_pay curr_type;
declare variable u_cli d_str_255;
begin

  for select oks.oks_from_k_id, k.k_kontract_no, k.k_fact_disb_date, k.k_disb_date, k.k_stop_perc, k.k_from_pr_id,
             oks.oks_delinq_date
      from oksummary oks
      join kontract k on k.k_id = oks.oks_from_k_id
      where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_close_state = 0
      into :k_id, :kontract_no, :k_fact_disb_date, :k_fact_disb_date, :terminate_date, :pr_name, :oks_delinq_date
  do
  begin
    base_rate = null;
    reduced_rate = null;
    current_rate = null;
    current_rate_begin_date = null;

    next_pay_date = null;
    next_pay_amount = null;
    last_future_pay_amount = null;
    prev_future_os_amount = null;

    future_nonzero_cnt = 0;
    future_perc_only_cnt = 0;
    future_os_perc_cnt = 0;
    future_neg_os_sum = 0;
    future_neg_perc_sum = 0;

    last_overdue_pay_amount = null;

    v_last_os_pay = null;
    v_prev_os_pay = null;
    last_row_os = null;

    select pr.pr_name
    from product pr
    join kontract k on k.k_from_pr_id = pr.pr_id
    where k.k_id = :k_id
    into :pr_name;

    select round(k.k_percent_per_day * 365.00, 3)
    from kontract k
    where k.k_id = :k_id
    into :base_rate;

    select reduced_rate_perc_year
    from print_kontract(:k_id)
    into :reduced_rate;

    select first 1 kd_percent_per_day * 365, kd_ol_date
    from kontdata
    where kd_from_k_id = :k_id and
          kd_ol_is_delete = 0 and
          kd_ol_date <= :rep_date_end
    order by kd_ol_date desc
    into :current_rate, :current_rate_begin_date;

    for select z.sh_id, z.sh_date, z.zad_os, z.zad_perc, z.s_os, z.s_perc, z.s_all
        from aarep_make_zad_list(:k_id, :rep_date_end, 1) z
        where z.sh_date is not null
        order by z.sh_date, z.sh_id
        into :sh_id, :sh_date, :zad_os, :zad_perc, :s_os, :s_perc, :s_all
    do
    begin
      if (sh_date < :rep_date_end) then
      begin
        if (coalesce(zad_os, 0) <> 0 or coalesce(zad_perc, 0) <> 0) then
          last_overdue_pay_amount = coalesce(zad_os, 0) + coalesce(zad_perc, 0);
      end

      if (sh_date >= :rep_date_end) then
      begin
        if (coalesce(s_os, 0) <> 0 or coalesce(s_perc, 0) <> 0) then
        begin
          pay_amount = coalesce(s_all, coalesce(s_os, 0) + coalesce(s_perc, 0));

          future_nonzero_cnt = future_nonzero_cnt + 1;

          if (next_pay_date is null) then
          begin
            next_pay_date = sh_date;
            next_pay_amount = pay_amount;
          end

          last_future_pay_amount = pay_amount;
          last_row_os = s_os;

          if (coalesce(s_os, 0) = 0 and coalesce(s_perc, 0) <> 0) then
            future_perc_only_cnt = future_perc_only_cnt + 1;

          if (coalesce(s_os, 0) <> 0 and coalesce(s_perc, 0) <> 0) then
            future_os_perc_cnt = future_os_perc_cnt + 1;

          if (coalesce(s_os, 0) < 0) then
            future_neg_os_sum = future_neg_os_sum + s_os;

          if (coalesce(s_perc, 0) < 0) then
            future_neg_perc_sum = future_neg_perc_sum + s_perc;

          if (coalesce(s_os, 0) <> 0) then
          begin
            v_prev_os_pay = v_last_os_pay;
            v_last_os_pay = pay_amount;
          end
        end
      end
    end

    if (future_nonzero_cnt <= 1) then
      prev_future_os_amount = null;
    else
    begin
      if (coalesce(last_row_os, 0) <> 0) then
        prev_future_os_amount = v_prev_os_pay;
      else
        prev_future_os_amount = v_last_os_pay;
    end

    for select kc.kc_id, 'n' || kc.kc_from_c_id, kc.kc_amount_approved
        from knclient kc
        where kc.kc_from_k_id = :k_id and kc.kc_amount_approved > 0
        into :kc_id, :c_id, :k_amount_approved
    do
    begin
      debt_os = 0;
      debt_perc = 0;
      overdue_debt_os = 0;
      overdue_debt_perc = 0;
      active_delinq_days = 0;

      init_plan_close_date = null;
      curr_plan_close_date = null;

      select oks.oks_balance, oks.oks_bal_perc, oks.oks_kont_last_date, oks.oks_last_sh_date,
             iif(oks.oks_delinq_days > 0, oks.oks_bal_zad_os, 0),
             iif(oks.oks_delinq_days > 0, oks.oks_bal_zad_perc, 0),
             oks.oks_delinq_days
      from kn_oksummary oks
      where oks.oks_from_kc_id = :kc_id and
            :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
            (oks.oks_close_state in (0, 1) or oks.oks_bal_law <> 0 or oks.oks_reserv_law <> 0)
      into :debt_os, :debt_perc, :init_plan_close_date, :curr_plan_close_date,
           :overdue_debt_os, :overdue_debt_perc, :active_delinq_days;

      debt_os = coalesce(debt_os, 0);
      debt_perc = coalesce(debt_perc, 0);
      overdue_debt_os = coalesce(overdue_debt_os, 0);
      overdue_debt_perc = coalesce(overdue_debt_perc, 0);
      active_delinq_days = coalesce(active_delinq_days, 0);

      if (coalesce(debt_os, 0) <> 0 or coalesce(debt_perc, 0) <> 0) then
        suspend;
    end
  end
end
