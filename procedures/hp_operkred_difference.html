CREATE PROCEDURE HP_OPERKRED_DIFFERENCE(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OK_FROM_K_ID DECIMAL(18,0),
    OK_FROM_OL_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    TMP_OK_OS DECIMAL(18,4),
    TMP_OK_PERC DECIMAL(18,4),
    TMP_OK_SH_ID DECIMAL(18,0),
    TMP_OK_SH_DATE DATE,
    TMP_OL_DATE DATE)
AS
declare variable nsum decimal(18,4);
declare variable tmp_ok_from_k_id decimal(18,0);
declare variable tmp_ol_id decimal(18,0);
begin
  for select ok_from_k_id, ol_id, k_kontract_no, sum(ok_os + ok_perc)
      from hp_operkred_difference_tmp(:rep_date_begin, :rep_date_end), kontract
      where ok_from_k_id = k_id
      group by ok_from_k_id, ol_id, k_kontract_no
      having sum(ok_os + ok_perc) = 0
      into :ok_from_k_id, :ok_from_ol_id, :k_kontract_no, :nsum

  do
  begin
    for select ok_from_k_id, ok_sh_id, ok_sh_date,
               /*  max(ol_id),*/
               max(ok_ol_date), sum(ok_os + ok_sh_os) * -1 as os, sum(ok_perc + ok_sh_perc) * -1 as perc
        from operkred, kontstate
        where ok_ol_is_active = 1 and
              ok_ol_date <= :rep_date_end and
              ok_type not in (7, 8) and
              ok_sh_date <= :rep_date_end and
              ks_ol_is_delete = 0 and
              ks_from_k_id = ok_from_k_id and
              ks_state = 1 and
              ok_from_c_id = 0 and
              :rep_date_end between cast(ks_date_begin as date) and ks_date_end and
              ok_from_k_id = :ok_from_k_id
        group by ok_from_k_id, ok_sh_id, ok_sh_date
        having(sum(ok_os + ok_sh_os) <> 0 or (sum(ok_perc + ok_sh_perc) <> 0)) and max(ok_ol_date) >= :rep_date_begin
        into :tmp_ok_from_k_id, :tmp_ok_sh_id, :tmp_ok_sh_date,
             /* :tmp_ol_id,*/
             :tmp_ol_date, :tmp_ok_os, :tmp_ok_perc

    do
    begin
      update operkred
      set ok_os = ok_os + :tmp_ok_os,
          ok_perc = ok_perc + :tmp_ok_perc
      where ok_from_k_id = :tmp_ok_from_k_id and
            ok_from_ol_id = :ok_from_ol_id and
            ok_sh_id = :tmp_ok_sh_id and
            ok_sh_date = :tmp_ok_sh_date and
            ok_type = 4;

      update operbook
      set ob_summa_val = ob_summa_val - :tmp_ok_os
      where ob_from_ol_id = :ok_from_ol_id and
            ob_account_kt like '_______109_1%';

      update operbook
      set ob_summa_val = ob_summa_val - :tmp_ok_perc
      where ob_from_ol_id = :ok_from_ol_id and
            ob_account_kt like '_______21141%';

      suspend;
    end
  end
end
