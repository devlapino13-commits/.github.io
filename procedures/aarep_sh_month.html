CREATE PROCEDURE AAREP_SH_MONTH(
    REP_REGION_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    OKS_REGION_ID ID_TYPE,
    OKS_REGION_ID_OLD ID_TYPE,
    OKS_KC_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REG_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    SH_OS CURR_TYPE,
    SH_PERC CURR_TYPE,
    SH_OS_IPF CURR_TYPE,
    SH_PERC_IPF CURR_TYPE,
    REP_LINE_NO INTEGER)
AS
begin
--Report on credit to employees for the planned amount of return in the past month
--отчет по кредитным служащим по плановой сумме возврата за месяц
  rep_date_begin = :rep_date;
  rep_date_end = cast(dateadd(month, 1, :rep_date_begin) as date) - 1;
  for select distinct oks_region_id, oks_kc_id, sum(ok_os + ok_sh_os) sh_os, sum(ok_perc + ok_sh_perc) sh_perc,
                      sum(iif(pr_code containing '.IPF.', ok_os + ok_sh_os, 0)) sh_os_ipf, sum(iif(pr_code containing '.IPF.', ok_perc + ok_sh_perc, 0)) sh_perc_ipf

      from oksummary oks, operkred ok, kontract k, product p, region_list(:rep_region_id, 1, -1, 1)
      where oks_close_state = 0 and
            :rep_date_begin between oks_date_begin and oks_date_end and
            oks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            oks_from_k_id = ok_from_k_id and
            ok_ol_is_active = 1 and
            ok_from_c_id = 0 and
            ok_ol_date < :rep_date_begin and
            ok_type < 7
--    and oks_region_id = 1040100
--    and oks_kc_id = 1090100006519
--    and oks_from_k_id = 109010000024601801
            and
            ok_sh_date between :rep_date_begin and :rep_date_end and
            oks_region_id = id_list
      group by oks_region_id, oks_kc_id
      having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
      order by oks_region_id, oks_kc_id
      into oks_region_id, oks_kc_id, sh_os, sh_perc, sh_os_ipf, sh_perc_ipf
  do
  begin
    select reg_name
    from region
    where reg_unicode = :oks_region_id
    into reg_name;

    select c_name_fio
    from clients
    where c_id = :oks_kc_id
    into c_name_fio;

    if (oks_region_id_old = oks_region_id) then
      rep_line_no = rep_line_no + 1;
    else
    begin
      rep_line_no = 1;
      oks_region_id_old = oks_region_id;
    end

    suspend;
  end
end
