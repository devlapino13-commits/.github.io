CREATE PROCEDURE HP_GET_TIME_LEFT_FORMATTED(
    MINUTES_CNT D_INTEGER,
    CONT_CODE D_STR_CODE,
    TARGET_MSGQUOTA_CFG_CODE D_STR_CODE)
RETURNS (
    TIME_LEFT D_STR_100)
AS
declare variable first_send_ts timestamp;
declare variable expire_ts timestamp;
declare variable remain_sec bigint;
declare variable days_cnt integer;
declare variable hours_cnt integer;
declare variable mins_cnt integer;
declare variable secs_cnt integer;
begin
    /*
        Процедура возвращает оставшееся время действия
        текущего временного окна msgquota в удобном для пользователя формате.
    
        Логика работы:
        - по target_msgquota_cfg_code и cont_code ищется первая запись в msgquotalist,
          попадающая в текущее окно длиной minutes_cnt минут;
        - определяется момент окончания окна:
              first_send_ts + minutes_cnt минут;
        - рассчитывается разница между current_timestamp и временем окончания;
        - результат возвращается в виде строки:
              N д. HH ч. MM м. SS с.
              HH ч. MM м. SS с.
              M м. SS с.
              S с.
    
        Важно:
        - процедура ищет первую запись только в рамках текущего временного окна,
          а не за всю историю;
        - если записей в текущем окне нет или время уже истекло, возвращается '0 с.';
        - процедура подходит для отображения пользователю времени ожидания до окончания
          ограничения, если логика ограничения считается от первой записи в текущем окне.
    */

    time_left = '0 с.';

    if (coalesce(minutes_cnt, 0) <= 0) then
    begin
        suspend;
        exit;
    end

    first_send_ts = null;

    select first 1 mql.mql_datetime
    from msgquotalist mql
        inner join msgquotacfg mqc on mql.mql_from_mqc_id = mqc.mqc_id
    where mqc.mqc_code = :target_msgquota_cfg_code and
          mql.mql_cont_code = :cont_code and
          mql.mql_datetime >= dateadd(-:minutes_cnt minute to current_timestamp) and
          mql.mql_datetime <= current_timestamp
    order by mql.mql_datetime asc
    into :first_send_ts;

    if (first_send_ts is null) then
    begin
        suspend;
        exit;
    end

    expire_ts = dateadd(:minutes_cnt minute to :first_send_ts);
    remain_sec = datediff(second from current_timestamp to :expire_ts);

    if (coalesce(remain_sec, 0) <= 0) then
    begin
        time_left = '0 с.';
        suspend;
        exit;
    end

    days_cnt = floor(remain_sec / 86400);
    remain_sec = mod(remain_sec, 86400);

    hours_cnt = floor(remain_sec / 3600);
    remain_sec = mod(remain_sec, 3600);

    mins_cnt = floor(remain_sec / 60);
    secs_cnt = mod(remain_sec, 60);

    if (days_cnt > 0) then
    begin
        time_left =
            cast(days_cnt as varchar(10)) || ' д. ' ||
            lpad(cast(hours_cnt as varchar(2)), 2, '0') || ' ч. ' ||
            lpad(cast(mins_cnt as varchar(2)), 2, '0') || ' м. ' ||
            lpad(cast(secs_cnt as varchar(2)), 2, '0') || ' с.';
    end
    else if (hours_cnt > 0) then
    begin
        time_left =
            cast(hours_cnt as varchar(10)) || ' ч. ' ||
            lpad(cast(mins_cnt as varchar(2)), 2, '0') || ' м. ' ||
            lpad(cast(secs_cnt as varchar(2)), 2, '0') || ' с.';
    end
    else if (mins_cnt > 0) then
    begin
        time_left =
            cast(mins_cnt as varchar(10)) || ' м. ' ||
            lpad(cast(secs_cnt as varchar(2)), 2, '0') || ' с.';
    end
    else
    begin
        time_left = cast(secs_cnt as varchar(10)) || ' с.';
    end

    suspend;
end
