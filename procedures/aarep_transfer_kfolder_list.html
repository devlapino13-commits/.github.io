CREATE PROCEDURE AAREP_TRANSFER_KFOLDER_LIST(
    UN_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    RES_OBJECT_KIND VARCHAR(100),
    RES_OBJECT_ID DECIMAL(18,0),
    RES_KF_NAME_CURR VARCHAR(100),
    RES_KF_NAME_PRIOR VARCHAR(100),
    RES_DISB_DATE DATE,
    RES_AKT_DATE DATE,
    RES_DELINQ_DAYS_CURR INTEGER,
    RES_BALANCE_CURR DECIMAL(18,4),
    RES_DELINQ_DAYS_TO_DATE INTEGER,
    RES_BALANCE_TO_DATE DECIMAL(18,4))
AS
declare variable lo_id decimal(18,0);
begin
  select un_from_c_id
  from username
  where un_id = :un_id
  into :lo_id;
  if (lo_id is null) then
    exit;

  if (rep_date is null) then
    rep_date = current_date;

  res_object_kind = 'Kontract';
  for select k_id, kal_new_kf_name, kal_prior_kf_name, k_disb_date, kal_date,
             iif(oks_delinq_date < :rep_date, datediff(day, oks_delinq_date, :rep_date), 0), oks_balance
      from oksummary, kontract, knaktlist
      where oks_date_begin <= :rep_date and
            oks_date_end >= :rep_date and
            oks_from_k_id = k_id and
            oks_balance > 0 and
            oks_close_state = 0 and
            oks_from_k_id = k_id and
            knaktlist.is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            kal_date > k_disb_date and
            kal_date <= :rep_date and
            kal_date_end >= :rep_date and
            kal_prior_lo_id <> oks_kc_id and
            oks_kc_id = :lo_id
      order by kal_date desc, kal_new_kf_name desc
      into :res_object_id, res_kf_name_curr, res_kf_name_prior, res_disb_date, res_akt_date, res_delinq_days_curr,
           res_balance_curr
  do
  begin
    res_delinq_days_to_date = 0;
    res_balance_to_date = 0;
    select iif(oks_delinq_date < :res_akt_date, datediff(day, oks_delinq_date, :res_akt_date), 0), oks_balance
    from oksummary
    where oks_from_k_id = :res_object_id and
          oks_date_begin <= :res_akt_date and
          oks_date_end >= :res_akt_date and
          oks_close_state = 0
    into :res_delinq_days_to_date, :res_balance_to_date;

    suspend;
  end
end
