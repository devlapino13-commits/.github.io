CREATE PROCEDURE AAREP_ANALITICA_PORT_ONE_LO(
    REP_C_ID DECIMAL(18,0),
    REP_DATE_END DATE)
RETURNS (
    LO_OWN_CLIENT_KOLICH INTEGER,
    LO_OWN_PAR7 DECIMAL(18,7),
    LO_BALANCE DECIMAL(18,4),
    LO_DELINQ_BALANCE7 DECIMAL(18,4),
    LO_STAGE DECIMAL(18,4))
AS
declare variable kolich integer;
declare variable ok_sh_date date;
declare variable ok_sh_id integer;
declare variable sh_os decimal(18,4);
declare variable sh_perc decimal(18,4);
declare variable ok_os decimal(18,4);
declare variable ok_perc decimal(18,4);
declare variable ok_type integer;
declare variable balance_end decimal(18,4);
declare variable client_kolich integer;
declare variable currency varchar(3);
declare variable dd_prior date;
declare variable deliq_os decimal(18,4);
declare variable deliq_perc decimal(18,4);
declare variable deliq_os_all decimal(18,4);
declare variable deliq_perc_all decimal(18,4);
declare variable is_first_line integer;
declare variable first_deliq_dd date;
declare variable kont_curr_rate decimal(18,4);
declare variable last_curr_rate decimal(18,4);
declare variable pr_set_rate_in_kont integer;
declare variable ol_id decimal(18,0);
declare variable def_curr varchar(10);
declare variable aa_k_id decimal(18,0);
declare variable aa_kolich_active_end integer;
declare variable aa_amount_active_end decimal(18,4);
declare variable aa_deliq_kolich integer;
declare variable aa_deliq_balance decimal(18,4);
declare variable aa_deliq_days integer;
declare variable aa_deliq_repay_kolich integer;
declare variable aa_kolich_cl_active_end integer;
declare variable lo_stage_date date;
begin
  select cm_default_currency
  from company
  into :def_curr;

  lo_balance = 0;
  lo_delinq_balance7 = 0;
  lo_own_client_kolich = 0;
  lo_stage = 0;
  select min(ks_date_begin)
  from kontstate
  where ks_ol_is_delete = 0 and
        ks_state = 1 and
        ks_kc_id = :rep_c_id
  into :lo_stage_date;
  if (lo_stage_date is null) then
    lo_stage_date = rep_date_end;
  lo_stage = datediff(day from lo_stage_date to :rep_date_end) / 30;

  for select ks_from_k_id, k_kolich_client_approved, k_index_curr_code, k_index_curr_rate, pr_set_rate_in_kont
      from kontstate, kontract, product
      where ks_from_k_id = k_id and
            ks_date_begin between '01.01.1900' and :rep_date_end and
            ks_date_end between :rep_date_end and '31.12.9999' and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_kc_id = :rep_c_id and
            k_from_pr_id = pr_id
      into :aa_k_id, :client_kolich, :currency, :kont_curr_rate, :pr_set_rate_in_kont
  do
  begin
    if (currency <> def_curr) then
    begin
      select cr_rate
      from opercurr
      where cr_from_cu_id = :currency and
            cr_date <= :rep_date_end and
            cr_date_end >= :rep_date_end
      into :last_curr_rate;
      if (last_curr_rate < :kont_curr_rate) then
      begin
        if (pr_set_rate_in_kont = 1) then
        begin
          last_curr_rate = :kont_curr_rate;
        end
      end
    end
    else
      last_curr_rate = 1;

    balance_end = 0;

    aa_kolich_cl_active_end = 0;
    aa_amount_active_end = 0;

    aa_deliq_kolich = 0;
    aa_deliq_balance = 0;
    aa_deliq_days = 0;
    deliq_os = 0;
    deliq_perc = 0;
    deliq_os_all = 0;
    deliq_perc_all = 0;
    aa_deliq_repay_kolich = 0;
    is_first_line = 1;
    first_deliq_dd = :rep_date_end;

    aa_kolich_cl_active_end = 0;

    for select ok_from_ol_id, ok_type, ok_sh_date, ok_sh_id, ok_sh_os, ok_sh_perc, ok_os, ok_perc
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :aa_k_id and
              ok_type <> 7 and
              ok_ol_date <= :rep_date_end and
              ok_from_c_id = 0
        order by ok_sh_date, ok_sh_id, ok_ol_date, ok_from_ol_id
        into :ol_id, :ok_type, :ok_sh_date, :ok_sh_id, :sh_os, :sh_perc, :ok_os, :ok_perc
    do
    begin
      if (ok_sh_date <= :rep_date_end) then
      begin
        if (ok_sh_id <> 0) then
        begin
          if (ok_type < 8) then
          begin
            if ((is_first_line = 1) or (:ok_sh_date = :dd_prior)) then
            begin
              deliq_os = :deliq_os + :sh_os + :ok_os;
              deliq_perc = :deliq_perc + :sh_perc + :ok_perc;
              if (is_first_line = 1) then
                dd_prior = :ok_sh_date;
            end
            else
            begin
              if ((deliq_os > 0) or (deliq_perc > 0)) then
              begin
                aa_deliq_repay_kolich = :aa_deliq_repay_kolich + 1;
                deliq_os_all = :deliq_os_all + :deliq_os;
                deliq_perc_all = :deliq_perc_all + :deliq_perc;
                if (first_deliq_dd = :rep_date_end) then
                  first_deliq_dd = :dd_prior;
              end
              dd_prior = :ok_sh_date;
              deliq_os = :sh_os + :ok_os;
              deliq_perc = :sh_perc + :ok_perc;
            end
            is_first_line = 0;
          end
        end
      end
      if (ok_type <= 8) then
        balance_end = :balance_end + :ok_os;
      kolich = 1;
    end /* Расчет сумм окончен */
    if (balance_end < 0) then
      balance_end = 0;
    if (balance_end > 0) then
      aa_kolich_active_end = 1;

    /* CLIENT_KOLICH */
    aa_kolich_cl_active_end = :aa_kolich_active_end * :client_kolich;

    if (kolich > 0) then
    begin
      if ((:deliq_os > 0) or (:deliq_perc > 0)) then
      begin
        deliq_os_all = :deliq_os_all + :deliq_os;
        deliq_perc_all = :deliq_perc_all + :deliq_perc;
        if (first_deliq_dd = :rep_date_end) then
          first_deliq_dd = :dd_prior;
      end
      if (:deliq_os_all + :deliq_perc_all > 0) then
      begin
        aa_deliq_kolich = 1;
        aa_deliq_days = cast((:rep_date_end - :first_deliq_dd) as integer);
      end
      else
      begin
        aa_deliq_kolich = 0;
        aa_deliq_days = 0;
      end
      if (balance_end = 0) then
      begin
        aa_deliq_repay_kolich = 0;
        aa_deliq_kolich = 0;
        aa_deliq_days = 0;
      end

      aa_amount_active_end = cast(:balance_end * :last_curr_rate as decimal(18,2));

      if (aa_deliq_kolich = 1) then
        aa_deliq_balance = aa_amount_active_end;
      else
        aa_deliq_balance = 0;
      if (not(aa_deliq_days > 7)) then
      begin
        aa_deliq_balance = 0;
      end
      lo_own_client_kolich = lo_own_client_kolich + aa_kolich_cl_active_end;
      lo_balance = lo_balance + aa_amount_active_end;
      lo_delinq_balance7 = lo_delinq_balance7 + aa_deliq_balance;
    end
  end
  lo_own_par7 = 0;
  if (lo_balance <> 0) then
  begin
    lo_own_par7 = lo_delinq_balance7 / lo_balance;
    lo_own_par7 = lo_own_par7 * 100;
  end
  suspend;

end
