CREATE PROCEDURE FM_HTML_MANAGER_DISB_LIMITS(
    UN_ID DECIMAL(18,0),
    REG_ID INTEGER)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable LO_ID decimal(18,0);
declare variable K_ID decimal(18,0);
declare variable C_NAME_FIO varchar(100);
declare variable C_ID decimal(18,0);
declare variable DB_ID integer;
declare variable MCL_APP_VERSION D_SMALLINT;
declare variable MCL_APP_VERSION_TEXT varchar(50);
declare variable DELIQ_DAYS D_STR_20;
declare variable KF_NAME D_STR_20;
declare variable KDP_DATE D_DATE;
declare variable KDP_SUMMA_SALDO D_CURRENCY;
declare variable KDP_SUMMA_APPROVED D_CURRENCY;
declare variable KDP_SUMMA_PROCESS D_CURRENCY;
declare variable KDP_COUNT_APPROVED D_INTEGER;
declare variable KDP_COUNT_PROCESS D_INTEGER;
declare variable UN_FIO D_STR_100;
begin
  db_id = f_get_db_id();

  select un_from_c_id, un_fio
  from username
  where un_id = :un_id
  into lo_id, un_fio; --id sotrudnika
  if (lo_id is null) then
    exit;

  if (coalesce(:reg_id, 0) = 0) then
  begin
    res_text = blob_append(res_text, 'Регион не указан, обратитесь к тех. поддержке');
    suspend;
    exit;
  end

  res_text = (select res_style
              from fm_html_styles) || '<body>';
  res_text = BLOB_APPEND(res_text, f_format('<h1 class="green_text">%1</h1>', iif(db_id <> 94401, 'Лимиты по выдачам', 'Лимиты по выдачам')));
  res_text = BLOB_APPEND(res_text, f_format('<b>%1</b1><br></br>', un_fio));
  res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="22%">
       <col width="26%">
       <col width="26%">
       <col width="26%">
       <tr>' || f_format('<th>%1</th><th>%2</th><th>%3</th><th>%4</th></tr>', iif(db_id <> 94401, 'Дата', 'Дата'),
                iif(db_id <> 94401, 'Остаток лимита', 'Остаток лимита'),
                iif(db_id <> 94401, 'Утвержденные займы', 'Утвержденные займы'),
                iif(db_id <> 94401, 'Договора в процессе', 'Договора в процессе')));

  for select kdp_date, kdp_summa_saldo, kdp_summa_approved, kdp_summa_process, kdp_count_approved, kdp_count_process
      from hp_manager_disb_limits(:un_id, :reg_id)
      into :kdp_date, :kdp_summa_saldo, :kdp_summa_approved, :kdp_summa_process, :kdp_count_approved, :kdp_count_process
  do
  begin
    res_text = BLOB_APPEND(res_text, f_format('<tr><td>%1</td><td align="right">%2</td><td align="right">%3</td><td align="right">%4</td></tr>',
    f_datetostr(kdp_date)||'</a>',
    iif(kdp_summa_saldo > 0, f_numtostr(kdp_summa_saldo,0), '...')||'</a>',
    iif(kdp_summa_approved > 0, f_numtostr(kdp_summa_approved,0), '...')||'</a>',
    iif(kdp_summa_process > 0, f_numtostr(kdp_summa_process,0), '...')
    ));

    res_text = BLOB_APPEND(res_text, f_format('<tr><td>%1</td><td align="right">%2</td><td align="right">%3</td><td align="right">%4</td></tr>',
    ''||'</a>',
    ''|| '</a>',
    iif(kdp_count_approved > 0, f_numtostr(kdp_count_approved,0), '...')||'</a>',
    iif(kdp_count_process > 0, f_numtostr(kdp_count_process,0), '...')
    ));
  end 

  res_text = BLOB_APPEND(res_text, '</table></body>');
  suspend;
end
