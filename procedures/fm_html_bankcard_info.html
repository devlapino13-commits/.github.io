CREATE PROCEDURE FM_HTML_BANKCARD_INFO(
    OBJ_KIND TYPE OF D_STR_CODE,
    OBJ_ID TYPE OF D_ID,
    UN_ID TYPE OF D_ID)
RETURNS (
    RES_TEXT D_STR_8100)
AS
declare variable disb_kind d_smallint;
declare variable lang d_str_lang;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable card_num d_str_code;
declare variable card_info d_str_255;
declare variable card_uri d_str_long;
declare variable disb_date d_date;
declare variable edc_code d_str_code;
declare variable db_id d_integer;
declare variable card_status d_str_code;
begin
  -- Процедура выводит блок информации, для закрепления банковских карт клиентам
  -- Реализовано для заявки и договора

  res_text = '';
  lang = f_get_lang_code(un_id, lang);
  db_id = f_get_db_id();
  obj_kind = f_objkind(obj_kind, 'code');

  edc_code = '';
  disb_kind = -1;
  disb_date = current_date;

  if (db_id = 81001) then
    edc_code = 'bankcard.moneta';
  else if (db_id in (39801, 41701)) then
    edc_code = 'bankcard.elcart';

  if (edc_code = '') then
  begin
    suspend;
    exit;
  end

  if (obj_kind = 'miniapp') then
  begin
    select ma_disb_kind, ma_date_app from miniapp
    where ma_id = :obj_id
      and ma_date_end >= '31.12.2999'
    into disb_kind, disb_date;
  end
  else if (obj_kind = 'kontract') then
  begin
    select k_disb_kind, k_disb_date from kontract
    where k_id = :obj_id
    into disb_kind, disb_date;
  end

  -- Проект должен быть действительным
  if (f_edoc_state2(edc_code, disb_date, un_id)= 'inactive') then
    disb_kind = -1;

  if (disb_kind <> 5) then
  begin
    suspend;
    exit;
  end

  res_text = f_format('<h2>%1</h2>', f_lang(lang, 'Карты клиентов')) ||
    '<table width="100%"><col width="40%"><col width="60%">';

  for select c_id, c_name_fio
  from knclient, clients
  where :obj_kind = 'kontract'
    and kc_from_k_id = :obj_id
    and kc_from_c_id = c_id
    and kc_amount_approved > 0
  union
  select c_id, c_name_fio
  from miniappclient, clients
  where :obj_kind = 'miniapp'
    and mac_from_ma_id = :obj_id
    and mac_from_c_id = c_id
    and mac_amount > 0
  order by 2
  into c_id, c_name_fio
  do
  begin
    card_num = '';
    select et_cardnumber
    from extsys_token
    where et_from_c_id = :c_id
      and et_is_actual = 1
    order by et_id desc
    rows 1
    into card_num;
    card_num = coalesce(card_num, '');
    card_uri = '';

    if (card_num = '') then
    begin
      card_info = f_format('%1: <b>%2</b>', f_lang(lang, 'Карта'), f_lang(lang, 'Не закреплена'));
      card_uri = f_lang(lang, 'Отправить СМС для закрепления');
    end
    else
    begin
      card_info = f_format('%1: <b>%2</b>', f_lang(lang, 'Карта'), card_num);
      card_uri = f_lang(lang, 'Отправить СМС для закрепления другой карты');
    end

    card_status = f_edoc_status(edc_code, c_id, current_date);
    if (card_status = 'unsigned') then
    begin
      card_info = card_info || '<br>' || f_lang(lang, 'СМС для закрепления карты отправлен');
      card_uri = f_lang(lang, 'Повторно отправить СМС');
    end
    card_uri = f_uri_sql(card_uri, 'bankcard_send_sms', 'client', c_id, 'none',
      f_lang(lang, 'Подтвердите отправку клиенту СМС для ввода данных карты'),
      '', null, null, null, null, null, :edc_code);

    res_text = res_text || f_format('<tr><td>%1</td><td>%2<br>%3</td></tr>',
      c_name_fio, card_info, card_uri);
  end

  res_text = res_text || '</table>';
  suspend;
end
