CREATE PROCEDURE FM_EXPRESS_PROCESS(
    EXA_ID D_INTEGER,
    ADD_PARAMS D_STR_20,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_100)
AS
declare variable EDL_ID D_ID;
declare variable EDOC_KIND D_INTEGER;
declare variable OBJ_KIND D_INTEGER;
declare variable OBJ_ID D_ID;
begin
  res_text = '';
  if (add_params = 'next') then
  begin
    update expressapp
    set exa_wait_status = 0,
        exa_from_un_id = :un_id
    where exa_id = :exa_id;
    res_text = 'Успешно выполнена';
    suspend;
  end else
  if (add_params = 'cancel') then
  begin
    select exa_from_edl_id
    from expressapp
    where exa_id = :exa_id
    into edl_id;

    if (edl_id is not null) then
    begin
      select edl_from_edc_id,
             edl_object_kind,
             edl_object_id
      from edoc_list
      where edl_id = :edl_id
      into edoc_kind,
           obj_kind,
           obj_id;
      execute procedure hp_edoc_cancel(:edoc_kind, :obj_kind, :obj_id);
    end

    update expressapp
    set exa_final_status = -1,
        exa_from_un_id = :un_id
    where exa_id = :exa_id;
    res_text = 'Заявка успешно аннулирована.';
    suspend;
  end
end
