CREATE PROCEDURE HP_BKI_XML_LIST_INDICATORS
RETURNS (
    CODE D_STR_CODE,
    NAME D_STR_255,
    DB_FIELD D_STR_255,
    XML_FIELD D_STR_CODE,
    DEFAULT_VALUE D_STR_255,
    XML_ID D_ID,
    USE_IN_VERSIONS D_STR_255)
AS
begin
  for select bki_code, bki_name, bki_db_field, bxf_xml_node, bxf_default_value, bxf_id
  from bki_xml_fields, bki_indicators
  where bki_code = bxf_from_bki_code
  order by bki_code
  into code, name, db_field, xml_field, default_value, xml_id
  do
  begin
    select list(distinct bxv_schema_version)
    from bki_xml_fields_ver, bki_xml_version
    where bxv_id = bfv_from_bxv_id
      and bfv_from_bxf_id = :xml_id
    into use_in_versions;

    suspend;
  end 
end
