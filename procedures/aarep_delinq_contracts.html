CREATE PROCEDURE AAREP_DELINQ_CONTRACTS(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    KF_NAME VARCHAR(20),
    CL_CASHIER_NAME VARCHAR(100),
    DELINQ_DAY INTEGER,
    KC_AMOUNT DECIMAL(18,4),
    KC_BALANCE CURR_TYPE,
    KC_ZAD_OS CURR_TYPE,
    KC_ZAD_PERC CURR_TYPE,
    OFFICER_NAME VARCHAR(100),
    K_PERIOD DECIMAL(15,2),
    PR_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(100),
    KNC_NAME_FIO VARCHAR(100),
    CL_CASHIER_ADDR VARCHAR(255),
    DELINQ_CATEGORY VARCHAR(255),
    DELINQ_REASON VARCHAR(255),
    REPAY_EXPECTATION VARCHAR(10),
    ANK_FILL_FIO VARCHAR(100),
    ANK_FILL_DATE DATE)
AS
declare variable k_id id_type;
declare variable reg_id integer;
declare variable reg_id_prior integer;
declare variable oks_kc_id id_type;
declare variable oks_kc_id_prior id_type;
declare variable delinq_balance curr_type;
declare variable k_clients_cashier_id id_type;
declare variable k_amount_approved curr_type;
declare variable k_from_kf_id id_type;
declare variable kc_from_c_id id_type;
declare variable zad_os curr_type;
declare variable sh_calc_perc curr_type;
declare variable kc_closed_ol_id id_type;
declare variable dolya double precision;
declare variable ank_fill_c_id id_type;
declare variable anketa_id integer;
declare variable is_delinqreport_reason smallint;
declare variable is_delinqreport_repayexpect smallint;
declare variable kc_id d_id;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable obj_name d_str_100;
begin
  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'Anketa.Delinq.1'
  into :anketa_id;
  is_delinqreport_reason = 0;
  if (exists(select *
             from refdata rd
             where rd.rd_code = 'DelinqReport.Reason' and
                   rd.rd_is_active = 1)) then
  begin
    is_delinqreport_reason = 1;
  end
  is_delinqreport_repayexpect = 0;
  if (exists(select *
             from refdata rd
             where rd.rd_code = 'DelinqReport.RepayExpectation' and
                   rd.rd_is_active = 1)) then
  begin
    is_delinqreport_repayexpect = 1;
  end

  reg_id_prior = 0;
  oks_kc_id_prior = 0;
  for select oks_from_k_id, id_list, oks_balance, :rep_date_end - oks_delinq_date, oks_kc_id

      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            :rep_date_end >= oks_delinq_date -- ноль дней просрочки включать или нет?
            and
            oks_region_id = id_list
            and oks_close_state = 0 
      into k_id, reg_id, delinq_balance, delinq_day, oks_kc_id

  do
  begin
--  регион и филиал
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;

      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :reg_id and
            r1.parent = r2.reg_unicode
      into reg_name, branch_name;
    end
--  ФИО КС
    if (oks_kc_id_prior <> oks_kc_id) then
    begin
      oks_kc_id_prior = oks_kc_id;
      select c_name_fio
      from clients
      where c_id = :oks_kc_id
      into officer_name;
    end

    select k_clients_cashier_id, k_amount_approved, k_period, pr_name, k_from_kf_id
    from kontract, product
    where k_id = :k_id and
          k_from_pr_id = pr_id
    into k_clients_cashier_id, k_amount_approved, k_period, pr_name, k_from_kf_id;
--   Кр. папка
    kf_name = '';
    select kal_new_kf_name
    from knaktlist
    where kal_from_kf_id = :k_from_kf_id and
          :rep_date_end between kal_date and kal_date_end and
          knaktlist.is_delete = 0
    into kf_name;
--  ФИО казначея
    cl_cashier_name = '';
    select c_name_fio
    from clients
    where c_id = :k_clients_cashier_id
    into cl_cashier_name;
--  адрес казначея
    cl_cashier_addr = '';
    select adl.adl_address_str
    from addrclient ac, addrlist adl
    where ac.adc_from_c_id = :k_clients_cashier_id and
          ac.adc_address_type = 1 and
          ac.adc_from_adl_id = adl.adl_id
    into cl_cashier_addr;
--  просроченная задолженность
    zad_os = 0;
    sh_calc_perc = 0;
    select sum(zad_os), sum(sh_calc_perc)
    from aarep_make_zad_list(:k_id, :rep_date_end, 1)
    where sh_date <= :rep_date_end
    into zad_os, sh_calc_perc;
    zad_os = coalesce(zad_os, 0);
    sh_calc_perc = coalesce(sh_calc_perc, 0);

    for select kc_from_c_id, kc_amount_approved, kc_closed_ol_id, kc_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        order by kc_closed_ol_id desc -- сортируем наоборот, если были списании то первее было нулы в конце
        into kc_from_c_id, kc_amount, kc_closed_ol_id, :kc_id
    do
    begin
--    если было списание доли то от общей выданной суммы контракта минусуем выданную сумму клиента
      if (kc_closed_ol_id is not null and
          exists(select *
                 from operlist
                 where ol_id = :kc_closed_ol_id and
                       operlist.is_delete = 0 and
                       ol_date <= :rep_date_end)) then
      begin
        k_amount_approved = k_amount_approved - kc_amount;
      end
      else
      begin
        dolya = cast(kc_amount as double precision) / k_amount_approved;
        kc_balance = dolya * delinq_balance;
        kc_zad_os = dolya * zad_os;
        kc_zad_perc = dolya * sh_calc_perc;

--      фио клиента
        select c_name_fio
        from clients
        where c_id = :kc_from_c_id
        into knc_name_fio;

        obj_name = 'KnClient';
        obj_kind = f_objkind(obj_name, 'id');
        obj_id = kc_id;
        if (not exists(
          select *
          from actionlist
          where acl_object_kind = :obj_kind
            and acl_object_id = :obj_id
            and acl_from_rd_id = :anketa_id)) then
        begin
          obj_name = 'Clients';
          obj_kind = f_objkind(obj_name, 'id');
          obj_id = kc_from_c_id;
        end

--       получаю приичну просрочки и категорию просрочки
        delinq_reason = '';
        delinq_category = '';

        if (is_delinqreport_reason = 1) then
        begin
          select res_str_values, -- причина просрочки
                 res_memo -- категория просрочки
          from hp_objectprops_anketa_get(:obj_name, :obj_id, -1, 'DelinqReport.Reason', :rep_date_end, null)
          into :delinq_reason, :delinq_category;
        end

--     Стремление заемщика/ГСО закрыть просрочку
        repay_expectation = '';
        if (is_delinqreport_repayexpect = 1) then
        begin
          select res_str_values
          from hp_objectprops_anketa_get(:obj_name, :obj_id, -1, 'DelinqReport.RepayExpectation', :rep_date_end, null)
          into :repay_expectation;
        end

--      даат заполнения анкеты
        ank_fill_c_id = null;
        ank_fill_date = null;
        select al.acl_respons_c_id, cast(al.acl_last_datetime as date)
        from actionlist al
        where al.acl_object_kind = :obj_kind and
              al.acl_object_id = :obj_id and
              al.acl_status = 2 and
              al.acl_datetime <= :rep_date_end and
              al.acl_from_rd_id = :anketa_id
        order by al.acl_last_datetime desc -- последняя заполненная
        rows 1
        into :ank_fill_c_id, :ank_fill_date;

--      заполнивший анкету
        ank_fill_fio = '';
        select cl.c_name_fio
        from clients cl
        where cl.c_id = :ank_fill_c_id
        into :ank_fill_fio;

        suspend;
      end
    end
  end
end
