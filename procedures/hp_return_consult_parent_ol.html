CREATE PROCEDURE HP_RETURN_CONSULT_PARENT_OL(
    UN_ID D_ID,
    ARC_ID D_ID)
AS
declare variable def_curr d_str_255;
declare variable new_ol_id d_id;
declare variable account_overpayments d_str_20;
declare variable account_doverit d_str_20;
declare variable doc_no d_integer;
declare variable reg_id d_integer;
declare variable memo_date date;
declare variable c_fio d_str_100;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable sum_ovp d_currency;
declare variable kontract_no d_str_20;
declare variable ol_memo d_str_255;
declare variable ob_detail varchar(1024);
begin

    def_curr = f_get_const('cm_default_currency');
    
    new_ol_id = (select max(ob.ob_from_ol_id)
               from operbook ob) + 1;

    select arc.arc_date, c.c_name_fio, arc.arc_from_c_id, arc.arc_from_k_id, arci.arci_amount, k.k_kontract_no, arci.arci_consult_office_id
    from app_return_consult arc
    inner join app_return_consult_info arci on arci.arci_id = arc.arc_id
    inner join clients c on c.c_id = arc.arc_from_c_id
    inner join kontract k on k.k_id = arc.arc_from_k_id
    where arc.arc_id = :arc_id
    into :memo_date, :c_fio, :c_id, :k_id, :sum_ovp, :kontract_no, :reg_id;

    select al.al_id
    from acc_list al
    where al.al_from_reg_id = :reg_id and
            al.al_from_bg_id = '21114' and
            al.al_number = 10
    into :account_overpayments;

    select al.al_id
    from acc_list al
    where al.al_from_reg_id = :reg_id and
            al.al_from_bg_id = '21199' and
            al.al_number = 110
    into :account_doverit;

    select num
    from last_doc_number(current_date, 0, :reg_id)
    into :doc_no;

    ol_memo = right((100 + extract(day from memo_date)), 2) || '.' ||
            right((100 + extract(month from memo_date)), 2) || '.' ||
            extract(year from memo_date) || ' г.';
  
    ol_memo = 'Сумма к возврату по консультационным услугам перед' || :c_fio || 'согл.заявления от ' || :ol_memo;
    ob_detail = '';
    
    execute procedure hp_overpayment_return_save_tmp(:new_ol_id, :reg_id, current_date, :doc_no, :ol_memo, :un_id,
            :c_id, :def_curr, :account_overpayments, :sum_ovp, :account_doverit, :k_id, null, 0, :new_ol_id, :ob_detail);

    update app_return_consult_info
    set arci_from_ol_id = :new_ol_id
    where (arci_id = :arc_id);

end
