CREATE PROCEDURE AAREP_CALC_VOC_DAYS(
    DATE_BEGIN DATE,
    DATE_END DATE,
    REPORT_CLIENT_ID DECIMAL(18,0))
RETURNS (
    KOL_DAYS DECIMAL(18,7))
AS
declare variable MONTH_BEGIN integer;
declare variable MONTH_END integer;
declare variable I integer;
declare variable DAY_1 integer;
declare variable DAY_2 integer;
declare variable MONTH_KOEF decimal(18,7);
declare variable REG_MOUNTAIN_KOEF decimal(18,4);
declare variable DD date;
declare variable MOUNT_KOEF_BEGIN decimal(18,4);
declare variable MOUNT_KOEF_END decimal(18,4);
declare variable LAST_DD date;
declare variable MAX_DAY integer;
declare variable REG_ID D_INTEGER;
begin
  kol_days = 0;
  if (date_begin <= date_end) then
  begin
    if ((extract(year from :date_end)) <> (extract(year from :date_begin))) then
    begin
      date_begin = '01.01.' || cast((extract(year from :date_end)) as varchar(4));
    end
    month_begin = extract(month from :date_begin);
    month_end = extract(month from :date_end);
    
    select ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :report_client_id and
          ps_date_begin <= :date_begin and
          ps_date_end >= :date_begin and
          ps_status = 1
    into :reg_id;
    
    -- RFCKG-4782
    reg_mountain_koef = f_reg_mount_koef(:reg_id, :date_begin);
    
    if (report_client_id = 1010100001212 and date_end <= '31.12.2020') then
      mount_koef_begin = 2.75;
    else
    begin
      if (reg_mountain_koef = 1) then
        mount_koef_begin = 2.33;
      else
        mount_koef_begin = 3.33;
    end
    
    select ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :report_client_id and
          ps_date_begin <= :date_end and
          ps_date_end >= :date_end and
          ps_status = 1
    into :reg_id;
    
    reg_mountain_koef = f_reg_mount_koef(:reg_id, :date_end);
    
    if (report_client_id = 1010100001212 and date_end <= '31.12.2020') then
      mount_koef_end = 2.75;
    else
    begin
      if (reg_mountain_koef = 1) then
        mount_koef_end = 2.33;
      else
        mount_koef_end = 3.33;
    end
    
    i = month_begin + 1;
    while (i < :month_end) do
    begin
      if (i < 9) then
        dd = cast('01.0' || cast(:i + 1 as varchar(1)) || '.' || extract(year from :date_begin) as date) - 1;
      else
        dd = cast('01.' || cast(:i + 1 as varchar(2)) || '.' || extract(year from :date_begin) as date) - 1;
      
      select ps_from_reg_id
      from hr_pstatus
      where ps_from_c_id = :report_client_id and
            ps_date_begin <= :dd and
            ps_date_end >= :dd and
            ps_status = 1
      into :reg_id;
      
      reg_mountain_koef = f_reg_mount_koef(:reg_id, :dd);
      
      if (report_client_id = 1010100001212) then
        month_koef = 2.75;
      else
      begin
        if (reg_mountain_koef = 1) then
          month_koef = 2.33;
        else
          month_koef = 3.33;
      end
      kol_days = kol_days + :month_koef;
      i = i + 1;
    end
    if (month_begin < month_end) then
    begin
      last_dd = dateadd(1 month to :date_begin);
      last_dd = cast('01.' || extract(month from :last_dd) || '.' || extract(year from :last_dd) as date) - 1;
      max_day = extract(day from :last_dd);
      day_1 = extract(day from :date_begin);
      day_1 = max_day - day_1 + 1;
      if (max_day = 0) then
        max_day = 30;
      kol_days = kol_days + (mount_koef_begin / max_day * day_1);
      
      last_dd = dateadd(1 month to :date_end);
      last_dd = cast('01.' || extract(month from :last_dd) || '.' || extract(year from :last_dd) as date) - 1;
      max_day = extract(day from :last_dd);
      day_2 = extract(day from :date_end);
      if (max_day = 0) then
        max_day = 30;
      kol_days = kol_days + (mount_koef_end / max_day * day_2);
    end
    if (month_begin = month_end) then
    begin
      day_1 = extract(day from :date_begin);
      day_2 = extract(day from :date_end);
      last_dd = dateadd(1 month to :date_end);
      last_dd = cast('01.' || extract(month from :last_dd) || '.' || extract(year from :last_dd) as date) - 1;
      max_day = extract(day from :last_dd);
      if (max_day = 0) then
        max_day = 30;
      if (day_2 >= day_1) then
        kol_days = kol_days + (mount_koef_end / max_day * (day_2 - day_1 + 1));
    end
    kol_days = cast(kol_days as decimal(15,2));
  end
  suspend;
end
