CREATE PROCEDURE HR_VOCTRAVEL_CLIENTS(
    HO_ID DECIMAL(18,0))
RETURNS (
    HRV_CLIENT_ID DECIMAL(18,0),
    HRV_DATE_BEGIN DATE,
    HRV_DATE_END DATE,
    HRV_ORDER_TYPE INTEGER,
    HRV_OFFICE_ID INTEGER)
AS
begin
  /* Procedure Text */
  for select hrv_from_c_id, hrv_date_begin, hrv_date_end, hr_orders.ho_order_type, hr_orders.ho_from_reg_id
      from hr_voctravel, hr_orders
      where hrv_from_ho_id = :ho_id and
            hrv_from_ho_id = ho_id and
            hrv_status = 1
      into :hrv_client_id, :hrv_date_begin, :hrv_date_end, :hrv_order_type, :hrv_office_id
  do
  begin
    suspend;
  end
end
