CREATE PROCEDURE HP_SUD_PAY_GET_GP4PAY_LIST
RETURNS (
    KONTRACT_NO D_STR_20,
    GP_SUMM D_CURRENCY,
    COURT_NAME D_STR_255,
    PAEE_NAME D_STR_255,
    INN D_STR_20,
    KPP D_STR_10,
    OKTMO D_STR_20,
    BANKNAME D_STR_255,
    BIK D_STR_10,
    ACC D_STR_20,
    TREASURE_ACC D_STR_20,
    KBK D_STR_20,
    SMR_ID D_ID,
    LAST_SUL_ID D_ID,
    DUTY_PAY_RECEIPT_BLS_ID D_ID)
AS
begin
  -- список заявок на оплату ГП
  for select smp_id, k_kontract_no, smp_sum_sud, sag_name, sag_payee_name,
            sag_inn, sag_kpp, sag_oktmo, sag_bankname, sag_bik,
            sag_acc, sag_treasure_acc, sag_kbk, smr_last_sul_id
  from sud_mainpay, sud_mainreq, kontract, sud_maindocs, sud_agents
  where smp_status = 0
    and smr_id = smp_id
    and k_id = smr_from_k_id
    and smd_id = smp_id
    and sag_id = smd_sud_sag_id
  into smr_id, kontract_no, gp_summ, court_name, paee_name, inn, kpp, oktmo,
       bankname, bik, acc, treasure_acc, kbk, last_sul_id
  do
  begin
    select slb_from_bls_id
    from sud_loanbls, sud_loandocs, sud_cfgdockind
    where sld_id = slb_from_sld_id
      and scd_id = sld_doc_kind
      and sld_from_smd_id = :smr_id
      and scd_code = 'duty_pay_receipt'  -- Квитанция оплаты ГП
      and slb_is_delete = 0
    order by slb_id desc
    rows 1
    into duty_pay_receipt_bls_id;

    suspend;
  end
end
