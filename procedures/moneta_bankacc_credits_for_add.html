CREATE PROCEDURE MONETA_BANKACC_CREDITS_FOR_ADD
RETURNS (
    ESM_ID D_ID,
    MNT_ID D_STR_10,
    DISB_AMOUNT D_CURRENCY,
    MNT_DESCRIPTION D_STR_255,
    CLIENT_NAME D_STR_100,
    CLIENT_ACCOUNT D_STR_CODE,
    CLIENT_INN D_STR_20,
    BANK_BIK D_STR_10,
    BANK_NAME D_STR_255,
    BANK_KS D_STR_255)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable k_kontract_no d_str_100;
declare variable k_disb_date d_date;
begin
  -- Процедура предоставляет список платежей, по которым должны быть
  -- выдачи на расчетный счет клиента через Монету

  for select esm_id, esm_amount, esm_object_id, esm_from_k_id
      from extsysmain
      where esm_project = 'moneta_disb'
        and esm_action = 'AccDisb'
        and esm_status = 0
        and esm_from_k_id is not null
    into esm_id, disb_amount, c_id, k_id
  do
  begin
    select mnt_id, client_account, bank_bik, bank_name, bank_ks, k_kontract_no,
           k_disb_date, client_inn, client_name
    from hp_bankacc_get_info(:k_id, :c_id)
    into mnt_id, client_account, bank_bik, bank_name, bank_ks, k_kontract_no,
         k_disb_date, client_inn, client_name;

    -- Проверяем контрольные значения, если что-то не заполнено, не выдаем
    if (k_disb_date is distinct from current_date or
        coalesce(client_name, '') = '' or
        coalesce(client_inn, '') = '' or
        coalesce(k_kontract_no, '') = '' or
        coalesce(bank_bik, '') = '' or
        coalesce(client_account, '') = '') then
      continue;

    -- client_name начинается с ИП, см. hp_bankacc_get_info
    mnt_description = 'Выдача займа по договору №' || k_kontract_no || ', ' || client_name || ', без НДС';

    suspend;
  end
end
