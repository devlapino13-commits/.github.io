CREATE PROCEDURE AAREP_INFO_OFFICE_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RES_REGION_ID INTEGER,
    SVZ_KOL_ACT INTEGER,
    SVZ_DISB INTEGER,
    DISB_KOL_KRED INTEGER,
    DISB_KOL_PARALEL INTEGER,
    DISB_KOL_VSEGO INTEGER,
    SVZ_ONE_DISB CURR_TYPE,
    REP_BALANCE CURR_TYPE,
    DELINQ_BALANCE_1 CURR_TYPE,
    DELINQ_BALANCE_7 CURR_TYPE,
    DELINQ_BALANCE_30 CURR_TYPE,
    PAR1 CURR_TYPE,
    PAR7 CURR_TYPE,
    PAR30 CURR_TYPE,
    KOL_CLOSE_KRED INTEGER,
    KOL_DISB_KRED INTEGER,
    PERC_UDERJ CURR_TYPE,
    REP_BALANCE_LAST CURR_TYPE,
    PRIROST_PORTFEL CURR_TYPE,
    KOL_CLI INTEGER,
    KOL_CLI_LAST INTEGER,
    PRIROST_CLI INTEGER)
AS
begin
  prirost_portfel = 0;
  prirost_cli = 0;

-- Обнуляем значения для каждого Suspend
  svz_kol_act = 0;
  svz_disb = 0;
  disb_kol_kred = 0;
  disb_kol_paralel = 0;
  disb_kol_vsego = 0;
  svz_one_disb = 0;
  rep_balance = 0;
  delinq_balance_1 = 0;
  delinq_balance_7 = 0;
  delinq_balance_30 = 0;
  par1 = 0;
  par7 = 0;
  par30 = 0;
  kol_close_kred = 0;
  kol_disb_kred = 0;
  perc_uderj = 0;
  rep_balance_last = 0;
  kol_cli = 0;
  kol_cli_last = 0;

-- Кол-во СВЗ
  for select ps_from_reg_id, count(ps_from_c_id)
      from jobkind, joblist, hr_pstatus
      where jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            jk_id = jl_kind and
            jl_id = ps_from_jl_id and
            ps_status = 1 and
            ps_is_working = 1 and
            :rep_date_end between ps_date_begin and ps_date_end
-- Исключаем декретников
            and
            not coalesce((select sawt_id
                          from hr_voctravel, hr_orders, sa_worktype
                          where hrv_from_ho_id = ho_id and
                                ho_order_type = sawt_id and
                                hrv_from_c_id = ps_from_c_id and
                                :rep_date_end between hrv_date_begin and hrv_date_end_fact
                          order by hrv_date_begin desc
                          rows 1), 0) in (16, 17, 18)
      group by ps_from_reg_id
      into res_region_id, svz_kol_act
  do
  begin
    if (svz_kol_act is null) then
      svz_kol_act = 0;
    suspend;
  end

-- Обнуляем значения для каждого Suspend
  svz_kol_act = 0;
  svz_disb = 0;
  disb_kol_kred = 0;
  disb_kol_paralel = 0;
  disb_kol_vsego = 0;
  svz_one_disb = 0;
  rep_balance = 0;
  delinq_balance_1 = 0;
  delinq_balance_7 = 0;
  delinq_balance_30 = 0;
  par1 = 0;
  par7 = 0;
  par30 = 0;
  kol_close_kred = 0;
  kol_disb_kred = 0;
  perc_uderj = 0;
  rep_balance_last = 0;
  kol_cli = 0;
  kol_cli_last = 0;

-- СВЗ выдавших, выдано кредитов, выдано параллельных и всего выданных
  for select ks_region_id, count(distinct ks_kc_id), sum(iif(pr_product_kind <> 2, k_kolich_client_approved, 0)),
             sum(iif(pr_product_kind = 2, k_kolich_client_approved, 0)), sum(k_kolich_client_approved)
      from kontstate, kontract, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id
      group by ks_region_id
      into res_region_id, svz_disb, disb_kol_kred, disb_kol_paralel, disb_kol_vsego
  do
  begin
    if (svz_disb is null) then
      svz_disb = 0;
    if (disb_kol_kred is null) then
      disb_kol_kred = 0;
    if (disb_kol_paralel is null) then
      disb_kol_paralel = 0;
    if (disb_kol_vsego is null) then
      disb_kol_vsego = 0;
    if (disb_kol_vsego > 0) then
      svz_one_disb = cast(svz_disb as double precision) / disb_kol_vsego;
    else
      svz_one_disb = 0;
    suspend;
  end
-- Обнуляем значения для каждого Suspend
  svz_kol_act = 0;
  svz_disb = 0;
  disb_kol_kred = 0;
  disb_kol_paralel = 0;
  disb_kol_vsego = 0;
  svz_one_disb = 0;
  rep_balance = 0;
  delinq_balance_1 = 0;
  delinq_balance_7 = 0;
  delinq_balance_30 = 0;
  par1 = 0;
  par7 = 0;
  par30 = 0;
  kol_close_kred = 0;
  kol_disb_kred = 0;
  perc_uderj = 0;
  rep_balance_last = 0;
  kol_cli = 0;
  kol_cli_last = 0;

-- Вычисление ПАР
  for select oks_region_id, sum(oks_balance), sum(iif(:rep_date_end >= oks_delinq_date, oks_balance, 0)),
             sum(iif(:rep_date_end - 7 > oks_delinq_date, oks_balance, 0)),
             sum(iif(:rep_date_end - 30 > oks_delinq_date, oks_balance, 0))
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_close_state = 0 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
      group by oks_region_id
      into res_region_id, rep_balance, delinq_balance_1, delinq_balance_7, delinq_balance_30
  do
  begin
    if (rep_balance is null) then
      rep_balance = 0;
    if (delinq_balance_1 is null) then
      delinq_balance_1 = 0;
    if (delinq_balance_7 is null) then
      delinq_balance_7 = 0;
    if (delinq_balance_30 is null) then
      delinq_balance_30 = 0;

    if (rep_balance > 0) then
    begin
      par1 = delinq_balance_1 / rep_balance;
      par7 = delinq_balance_7 / rep_balance;
      par30 = delinq_balance_30 / rep_balance;
    end
    else
    begin
      par1 = 0;
      par7 = 0;
      par30 = 0;
    end
    suspend;
  end
-- Обнуляем значения для каждого Suspend
  svz_kol_act = 0;
  svz_disb = 0;
  disb_kol_kred = 0;
  disb_kol_paralel = 0;
  disb_kol_vsego = 0;
  svz_one_disb = 0;
  rep_balance = 0;
  delinq_balance_1 = 0;
  delinq_balance_7 = 0;
  delinq_balance_30 = 0;
  par1 = 0;
  par7 = 0;
  par30 = 0;
  kol_close_kred = 0;
  kol_disb_kred = 0;
  perc_uderj = 0;
  rep_balance_last = 0;
  kol_cli = 0;
  kol_cli_last = 0;

-- Процент удержание закрытых групп, параллельных не включаем
  for select ks1.ks_region_id, count(*), sum((select sum(1)
                                              from kontstate ks2, kontract k2, product pr2
                                              where ks_ol_is_delete = 0 and
                                                    ks_state = 1 and
                                                    ks_date_begin between :rep_date_begin and :rep_date_end -- выдался за текущий период
                                                    and
                                                    ks_region_id = ks1.ks_region_id and
                                                    ks_from_k_id = k_id and
                                                    k_from_pr_id = pr_id and
                                                    pr_product_kind <> 2 and
                                                    k2.k_from_kf_id = k1.k_from_kf_id))
      from kontstate ks1, kontract k1, product pr1
      where ks_ol_is_delete = 0 and
            ks_state = 2 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_product_kind <> 2
      group by ks1.ks_region_id
      into res_region_id, kol_close_kred, kol_disb_kred
  do
  begin
    if (kol_close_kred is null) then
      kol_close_kred = 0;
    if (kol_disb_kred is null) then
      kol_disb_kred = 0;
    if (kol_close_kred > 0) then
      perc_uderj = cast(kol_disb_kred as double precision) / kol_close_kred;
    else
      perc_uderj = 0;
    suspend;
  end
-- Обнуляем значения для каждого Suspend
  svz_kol_act = 0;
  svz_disb = 0;
  disb_kol_kred = 0;
  disb_kol_paralel = 0;
  disb_kol_vsego = 0;
  svz_one_disb = 0;
  rep_balance = 0;
  delinq_balance_1 = 0;
  delinq_balance_7 = 0;
  delinq_balance_30 = 0;
  par1 = 0;
  par7 = 0;
  par30 = 0;
  kol_close_kred = 0;
  kol_disb_kred = 0;
  perc_uderj = 0;
  rep_balance_last = 0;
  kol_cli = 0;
  kol_cli_last = 0;

-- Прирост портфеля
  for select oks_region_id, sum(oks_balance)
      from oksummary
      where :rep_date_begin - 1 between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_close_state = 0 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_begin - 1, null, null, null, null, null, null)) in (1, 2)
      group by oks_region_id
      into res_region_id, rep_balance_last
  do
  begin
    if (rep_balance_last is null) then
      rep_balance_last = 0;
    suspend;
  end
-- Обнуляем значения для каждого Suspend
  svz_kol_act = 0;
  svz_disb = 0;
  disb_kol_kred = 0;
  disb_kol_paralel = 0;
  disb_kol_vsego = 0;
  svz_one_disb = 0;
  rep_balance = 0;
  delinq_balance_1 = 0;
  delinq_balance_7 = 0;
  delinq_balance_30 = 0;
  par1 = 0;
  par7 = 0;
  par30 = 0;
  kol_close_kred = 0;
  kol_disb_kred = 0;
  perc_uderj = 0;
  rep_balance_last = 0;
  kol_cli = 0;
  kol_cli_last = 0;

--   prirost_portfel = rep_balance_last - rep_balance;  -- это будет реализовано в процедуре вызова
-- Кол-во активных клиентов
  for select ac_region_id, sum(ac_is_client)
      from aarep_cli_active_list(:rep_date_end)
      where ac_is_client = 1 and
            (select aa_category_id
             from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
      group by ac_region_id
      into res_region_id, kol_cli
  do
  begin
    if (kol_cli is null) then
      kol_cli = 0;
    suspend;
  end
-- Обнуляем значения для каждого Suspend
  svz_kol_act = 0;
  svz_disb = 0;
  disb_kol_kred = 0;
  disb_kol_paralel = 0;
  disb_kol_vsego = 0;
  svz_one_disb = 0;
  rep_balance = 0;
  delinq_balance_1 = 0;
  delinq_balance_7 = 0;
  delinq_balance_30 = 0;
  par1 = 0;
  par7 = 0;
  par30 = 0;
  kol_close_kred = 0;
  kol_disb_kred = 0;
  perc_uderj = 0;
  rep_balance_last = 0;
  kol_cli = 0;
  kol_cli_last = 0;

-- на предыдущий день
  for select ac_region_id, sum(ac_is_client)
      from aarep_cli_active_list(:rep_date_begin - 1)
      where ac_is_client = 1 and
            (select aa_category_id
             from aarep_get_category_id(ac_kontract_id, :rep_date_begin - 1, null, null, null, null, null, null)) in (1, 2)
      group by ac_region_id
      into res_region_id, kol_cli_last
  do
  begin
    if (kol_cli_last is null) then
      kol_cli_last = 0;
--     prirost_cli = kol_cli_last - kol_cli; это будет реализовано в процедуре вызова
    suspend;
  end
end
