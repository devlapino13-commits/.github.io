Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UI_EREFLIST(
    EREF_UUID_CODE D_STR_CODE)
RETURNS (
    ID D_INTEGER,
    CAPTION D_STR_255,
    NAME D_STR_255,
    CODE D_STR_255,
    MEMO D_STR_255,
    ORDERNO D_INTEGER,
    IS_ACTIVE D_BOOL,
    IS_LOCKED D_SMALLINT,
    STR_ID D_STR_100,
    TAG D_INTEGER)
AS
declare variable erl_code d_str_255;
declare variable erl_kind d_smallint;
declare variable show_inactive d_bool;
declare variable erl_lookup d_smallint;
begin
  select erl_code, erl_kind, erl_lookup
  from ux$ereflist
  where erl_uuid = char_to_uuid(:eref_uuid_code)
  into erl_code, erl_kind, erl_lookup;

  if (row_count = 0) then
    exit;

  id = 0;
  name = '';
  code = '';
  memo = 'Не указан';
  caption = '';
  orderno = 0;
  is_active = 1;
  is_locked = 1;
  str_id = '';
  tag = 0;

  if (erl_kind = 1) then
  begin
    show_inactive = 0;
    if (exists(select * from ux$erefcfg
    where erc_erl_uuid = char_to_uuid(:eref_uuid_code) and
          erc_show_inactive = 1)) then
      show_inactive = 1;

    for select id, caption, name, code, memo, orderno, is_active, is_locked, str_id, tag
    from uxp$refdata(:erl_code, :show_inactive, iif(:erl_lookup in (2, 3), 1, 0))
    into :id, :caption, :name, :code, :memo, :orderno, :is_active, :is_locked, :str_id, :tag
    do
      suspend;
  end else
  if (erl_kind = 0) then
  begin
    if (erl_code = 'office_full_list') then
    begin
      for select reg_unicode, reg_code, reg_code || ' ' || reg_name
      from region
      where reg_status = 3 and
            reg_date_close is null
      order by reg_code
      into :id, :code, :name
      do
        suspend;

      for select reg_unicode, reg_code, reg_code || ' ' || reg_name ||
                 '   (закрыт ' || f_datetostr(reg_date_close) || ')'
      from region
      where reg_status = 3 and
            reg_date_close is not null
      order by reg_code
      into :id, :code, :name
      do
        suspend;

      exit;
    end

    for select el_id, el_code, el_name
    from enum_list
    where el_kind = :erl_code
    order by el_kind, el_id
    into :id, :code, :name
    do
    begin
      if (erl_lookup = 3 or (erl_lookup = 2 and is_locked = 1)) then
        caption = uxf$join_str(name, code);
      else
        caption = name;

      suspend;
    end
  end else
    exception e_default 'Undefined erl_kind: ' || coalesce(erl_kind, '<Null>');
end
