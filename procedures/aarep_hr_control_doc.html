CREATE PROCEDURE AAREP_HR_CONTROL_DOC(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME D_STR_CODE,
    C_NAME_FIO D_STR_100,
    JL_NAME D_STR_100,
    CDS_DOC_DATE_BEGIN D_DATE,
    HIRED_DATE D_DATE,
    ADD_INFO STR_255)
AS
declare variable ER_CANCELLED integer;
declare variable C_ID D_ID;
declare variable CDS_DOC_DATE_END D_DATE;
declare variable IS_IN_ORDER D_SMALLINT;
begin

  for select cl.c_name_fio, ps_from_c_id, r1.reg_code || ' ' || r1.reg_name, j.jl_name, h.ps_from_c_id
      from hr_pstatus h
      inner join region_list(:rep_region_id, 1, 3, 1) on ps_from_reg_id = id_list
      inner join clients cl on cl.c_id = h.ps_from_c_id
      inner join joblist j on ps_from_jl_id = jl_id
      inner join region r1 on id_list = r1.reg_unicode
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_is_working = 1
      into :c_name_fio, c_id, reg_name, jl_name, c_id
  do
  begin
    cds_doc_date_begin = null;

    select cds_doc_date_begin
    from clientdocs cds, passkind, passdoctype
    where cds_from_c_id = :c_id and
          cds_from_pk_id = pk_id and
          cds_is_delete = 0 and
          pk_from_pdt_id = pdt_id and
          :rep_date_end between cds_doc_date_begin and coalesce(cds_doc_date_end, :rep_date_end) and
          pdt_code in ('certificate_of_no_criminal_record')
    order by cds.cds_doc_date_begin desc
    rows 1
    into :cds_doc_date_begin;

    select max(ps_date_begin)
    from hr_pstatus
    inner join hr_orders on ho_id = ps_from_ho_id
    where ps_from_c_id = :c_id and
          ps_status = 1 and
          ho_order_type = 1
    into hired_date;

    add_info = '';
    if (exists(select ps_date_begin
               from hr_pstatus
               inner join hr_orders on ho_id = ps_from_ho_id
               where ps_from_c_id = :c_id and
                     ps_status = 1 and
                     :rep_date_end between ps_date_begin and ps_date_end and
                     ho_order_type = 6 and
                     ho_come_back_date is not null)) then
      add_info = 'Техническое увольнение';

    select list(sawt_name, ', ')
    from hr_voctravel
    inner join hr_orders on ho_id = hrv_from_ho_id
    inner join sa_worktype on sawt_id = ho_order_type
    where hrv_from_c_id = :c_id and
          :rep_date_end between hrv_date_begin and hrv_date_end_fact and
          hrv_status = 1 and
          sawt_base_type = 5 and
          sawt_id <> 20
    into add_info;

    is_in_order = 0;
    if ((jl_name = 'Директор регионального развития') or (jl_name = 'Администратор') or

        (jl_name = 'Юрист департамента розыска и принудительного взыскания') or (jl_name = 'Директор по управлению качеством') or

        (jl_name = 'Специалист отдела контроля качества') or (jl_name = 'Руководитель отдела контроля качества') or

        (jl_name = 'Старший специалист департамента розыска и принудительного взыскания') or (jl_name = 'Специалист департамента розыска и принудительного взыскания') or

        (jl_name = 'Директор регионального развития/Антикризисный менеджер') or (jl_name = 'Специалист') or (jl_name = 'Менеджер офиса') or

        (jl_name = 'Ведущий специалист') or (jl_name = 'Специалист по продажам') or (jl_name = 'Менеджер') or (jl_name = 'Специалист по работе с просроченной задолженностью') or

        (jl_name = 'Специалист службы безопасности') or (jl_name = 'Ведущий специалист отдела по работе с просроченной задолженностью') or (jl_name = 'Кредитный эксперт') or

        -- RFCRU-3082
        (jl_name = 'Руководитель отдела по работе с просроченной задолженностью') or (jl_name = 'И.о. менеджера офиса') or (jl_name = 'Инспектор кредитный') or

        (jl_name = 'Кредитный специалист') or (jl_name = 'Руководитель отдела внутреннего контроля') or (jl_name = 'Специалист отдела внутреннего контроля') or

        (jl_name = 'И.о.руководителя департамента розыска и принудительного взыскания') or (jl_name = 'Руководитель департамента розыска и принудительного взыскания')) then
      is_in_order = 1;

    if (is_in_order = 1) then
      suspend;
  end

end
