CREATE PROCEDURE HP_GET_HTML_IE_STATUS(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_STR_4096)
AS
declare variable obj_kind_str d_str_code;
declare variable req_status d_smallint;
declare variable actuality_date d_date;
declare variable ip_status_str d_str_long;
declare variable ip_status d_smallint;
declare variable color d_str_10;
begin
    -- RFCRU-5314
    -- Процедура отображает небольшой блок по статусу ИП в окне "Статус заявки"

    res_html = '';
    obj_kind_str = f_objkind(:obj_kind, 'code');

    -- Получаем общую информацию о запросе
    -- Был ли вообще отправлен запрос, актуальность запроса и ИП статус, где 0 - не найден и 1 - найден
    select
        req_status,
        actuality_date,
        coalesce(ip_status, 0)
    from hp_kontur_ip_get_base_info(:obj_kind_str, :obj_id)
    into
        req_status,
        actuality_date,
        ip_status;

    -- Если запрос не был отправлен, либо ответ уже не актуален, то специалист видит ссылку.
    -- Через ссылку можно отправить запрос в Контур на проверку статуса ИП
    -- Если с запросом всё ок, тогда просто показываем ответ, был найден либо нет
    -- Сам запрос в Контур, мы формируем через hp_send_query_ie_kontur и дальше через hp_extinfo_to_send_kontur_ip
    color =
        case
            when req_status is null then 'black'
            when req_status = 0 then 'black'
            when req_status = 1 then 'black'
            when req_status = 2 and ip_status = 1 then 'green'
            else 'red'
        end;

    ip_status_str =
        case
            when req_status is null then f_uri_sql('Проверить статус ИП', 'send_query_ie_kontur', :obj_kind, :obj_id, 'none', 'Вы действительно хотите проверить статус ИП', '', 0, 255, '', 'Запрос был отправлен, ожидайте', 'dialog', 'null')
            when current_date >= actuality_date then f_uri_sql('Проверить статус ИП', 'send_query_ie_kontur', :obj_kind, :obj_id, 'none', 'Вы действительно хотите проверить статус ИП', '', 0, 255, '', 'Запрос был отправлен, ожидайте', 'dialog', 'null')
            when req_status < 0 then 'Ошибка'
            when req_status = 0 then 'Ожидайте ответа'
            when req_status = 1 then 'Ожидайте ответа'
            when req_status = 2 and ip_status = 1 then 'Найден'
            when req_status = 2 and ip_status = 0 then 'Не найден'
            else 'Не предвиденный статус, просим обратиться в службу технической поддержки'
        end;


    res_html = blob_append(res_html, '<h2>Проверка статуса ИП</h2><table width="100%"><col width="40%"><col width="60%">');
    res_html = blob_append(res_html, f_html_table_row('Статус ИП:', f_format('<font color="%1">%2</font>', :color, :ip_status_str), '', '', -1, -1, -1, -1, 2, '')||'</table>');
    suspend;
end
