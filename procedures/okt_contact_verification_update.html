CREATE PROCEDURE OKT_CONTACT_VERIFICATION_UPDATE(
    CC_ID ID_TYPE,
    CC_CHECK_RESULT INTEGER,
    CC_CALL_USER STR_255,
    CC_CHECK_MEMO STR_255,
    CC_NAME_OK INTEGER,
    CC_ADDRESS_OK INTEGER,
    CC_CLIENT_OK INTEGER,
    CC_BIRTH_DATE DATE,
    CC_STATUS_OK INTEGER)
AS
begin
  update contacts cont
  set cc_check_result = :cc_check_result,
      cc_in_process = 0,
      cc_try_kol = cc_try_kol + 1,
      cc_call_user = :cc_call_user,
      cc_time_end = current_timestamp,
      cont.cc_check_memo = :cc_check_memo,
      cc_name_ok = :cc_name_ok,
      cc_address_ok = :cc_address_ok,
      cc_client_ok = :cc_client_ok,
      cc_birth_date = iif(:cc_birth_date is null, cc_birth_date, :cc_birth_date),
      cc_status_ok = :cc_status_ok
  where cc_id = :cc_id;
end
