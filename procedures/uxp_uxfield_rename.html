CREATE PROCEDURE UXP$UXFIELD_RENAME(
    UF_UUID D_UUID,
    NEW_CODE D_STR_CODE,
    OLD_SOURCE D_STR_CODE = null,
    OLD_CODE D_STR_CODE = null)
AS
declare variable us_uuid d_uuid;
begin
  -- Переименовывает поле
  if (coalesce(uf_uuid, '') = '') then
    uf_uuid = uxf$uuid(uxf$uniq(old_source), 'uxfield', old_code);

  old_code = '';
  select uf_owner, uf_field_name
  from ux$field
  where uf_uuid = :uf_uuid
  into us_uuid, old_code;

  if (old_code = new_code) then
    exit;

  if (exists(select * from ux$field
  where uf_owner = :us_uuid and
        uf_field_name = :new_code and
        uf_state = 0)) then
    exception e_message f_format('Поле %1 уже существует.', upper(new_code));

  delete from ux$field
  where uf_owner = :us_uuid and
        uf_field_name = :new_code and
        uf_state <> 0;

  update ux$field
  set uf_field_name = :new_code,
      uf_state = 0
  where uf_uuid = :uf_uuid;
end
