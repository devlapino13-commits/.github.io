CREATE PROCEDURE FM_HTML_COMPETITION_DETAILS(
    UN_ID DECIMAL(18,0),
    OBJ_ID DECIMAL(18,0))
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable team_name varchar(255);
declare variable group_name varchar(255);
declare variable final_points decimal(18,4) = 0;
declare variable order_in_group integer;
declare variable period_name varchar(255);
declare variable date_begin date;
declare variable date_end date;
declare variable lo_comment blob sub_type 1 segment size 80;
declare variable lo_id decimal(18,0);
declare variable lo_fio varchar(100);
declare variable color_black varchar(20);
declare variable color_yellow varchar(20);
declare variable color_green varchar(20);
declare variable color_red varchar(20);
declare variable line_str varchar(255);
declare variable max_in_group integer;
declare variable cpr_id integer;
declare variable str varchar(512);
declare variable max_order_no integer;
declare variable curr_color varchar(20);
declare variable top_level integer;
declare variable cm_database_id integer;
declare variable res_team_id integer;
declare variable cpp_stage integer;
declare variable team_id integer;
declare variable rd_code varchar(255);
declare variable is_olimpic integer;
declare variable cpr_comment varchar(255);
declare variable final_orderno integer;
declare variable jk_system_code d_str_100;
declare variable rd_id d_integer;
begin
  /* Procedure Text */
  res_html = (select res_style
              from fm_html_styles) || '<body>';
  select c_id, c_name_fio
  from clients, username
  where un_id = :un_id and
        un_from_c_id = c_id
  into :lo_id, :lo_fio;

  if (lo_id is null) then
    exit;

  select rd_code, rd_id
  from refdata, competperiod
  where cpp_from_rd_id = rd_id and
        cpp_id = :obj_id
  into :rd_code, rd_id;

  color_black = 'black';
  color_yellow = '#F08205';
  color_green = '#02B316';
  color_red = 'red';

  is_olimpic = 0;
  if (rd_code like '%Olimpic%' or rd_code containing 'Champions') then
    is_olimpic = 1;
  if (rd_code containing 'olimpic_mo') then is_olimpic = 0;
  res_team_id = -1;
  if (is_olimpic = 1) then
  begin
  select jk_system_code
  from hr_pstatus, joblist, jobkind
  where ps_from_c_id = :lo_id and
        ps_is_working = 1 and
        ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id
  into jk_system_code;

  if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner', 'JobSystem.Specialist_Expert')) then
  begin
    select res_team_id
    from aarep_get_lo_team(0, :lo_id, null)
    into :res_team_id;
  end
  else
  if (jk_system_code in ('JobSystem.Region_KM')) then
  begin
    select res_team_id
    from aarep_get_mgr_team(:rd_id, :un_id)
    into :res_team_id;
  end

  end

  :cm_database_id = f_get_db_id();
  top_level = 5;
  if (cm_database_id = 94401) then
    top_level = 1;

  if (is_olimpic = 1) then
  begin
    cpr_comment = '';
    select cpr_id, rd_name, cpr_group_name, cpr_final_points, cpr_order_in_group, cpp_period_name, cpp_date_begin,
           cpp_date_end, cpd_comment, cpp_stage, ''
    from competresult, refdata, competdetails, competperiod
    where cpp_id = :obj_id and
          cpr_from_cpp_id = cpp_id and
          cpr_object_kind = -1 and
          cpr_object_id = rd_id and
          cpr_id = cpd_from_cpr_id and
          cpd_object_kind = 0 and
          cpd_object_id = :lo_id and
          cpp_stage = 0
    union
    select cpr_id, rd_name, cpr_group_name, cpr_final_points, cpr_order_in_group, cpp_period_name, cpp_date_begin,
           cpp_date_end, '', cpp_stage, cpr_comment
    from competresult, refdata, competperiod
    where cpp_id = :obj_id and
          cpr_from_cpp_id = cpp_id and
          cpr_object_kind = -1 and
          cpr_object_id = rd_id and
          cpr_object_kind = -1 and
          cpr_object_id = :res_team_id and
          cpp_stage > 0
    into :cpr_id, :team_name, :group_name, final_points, order_in_group, :period_name, :date_begin, :date_end,
         lo_comment, cpp_stage, cpr_comment;

    if (cpr_comment is null) then
      cpr_comment = '';

    max_in_group = 0;
    select count(*), max(cpr_order_in_group)
    from competresult
    where cpr_from_cpp_id = :obj_id and
          cpr_object_kind = -1 and
          cpr_object_id <> 0 and
          cpr_group_name = :group_name
    into :max_in_group, max_order_no;
    if (max_in_group is null) then
      max_in_group = 0;
    if (max_order_no is null) then
      max_order_no = 0;

    if (cpr_comment = '') then
      res_html = BLOB_APPEND(res_html, f_format('<h1 class="green_text">%1<br>%2 %3</h1>', group_name, iif(cm_database_id <> 94401, 'Команда', 'Komanda'), team_name));
    else
      res_html = BLOB_APPEND(res_html, f_format('<h1 class="green_text">%1<br>%2 %3<br>%4</h1>', group_name, iif(cm_database_id <> 94401, 'Команда', 'Komanda'), team_name, cpr_comment));
  end
  else
  begin
    select cpr_id, c_name_fio, cpr_group_name, cpr_final_points, cpr_order_in_group, cpp_period_name, cpp_date_begin,
           cpp_date_end, cpd_comment, cpp_stage, CPR_FINAL_ORDERNO
    from competresult, clients, competdetails, competperiod
    where cpp_id = :obj_id and
          cpr_from_cpp_id = cpp_id and
          cpr_object_kind = 0 and
          cpr_object_id = c_id and
          cpr_id = cpd_from_cpr_id and
          cpd_object_kind = 0 and
          cpd_object_id = :lo_id and
          cpp_stage = 0
    union
    select cpr_id, c_name_fio, cpr_group_name, cpr_final_points, cpr_order_in_group, cpp_period_name, cpp_date_begin,
           cpp_date_end, '', cpp_stage, CPR_FINAL_ORDERNO
    from competresult, clients, competperiod
    where cpp_id = :obj_id and
          cpr_from_cpp_id = cpp_id and
          cpr_object_kind = 0 and
          cpr_object_id = c_id and
          cpr_object_kind = 0 and
          cpr_object_id = :lo_id and
          cpp_stage > 0
    into :cpr_id, :team_name, :group_name, final_points, order_in_group, :period_name, :date_begin, :date_end,
         lo_comment, cpp_stage, FINAL_ORDERNO;

    max_in_group = 0;
    select count(*),
--           max(cpr_order_in_group),
           max(CPR_FINAL_ORDERNO)
    from competresult
    where cpr_from_cpp_id = :obj_id and
          cpr_object_kind = 0 and
          cpr_object_id <> 0
    into :max_in_group, max_order_no;
    if (max_in_group is null) then
      max_in_group = 0;
    if (max_order_no is null) then
      max_order_no = 0;

    select c_name_fio
    from clients
    where c_id = :obj_id
    into :team_name;
    res_html = BLOB_APPEND(res_html, f_format('<h1 class="green_text">%1<br>%2 %3</h1>', group_name, iif(cm_database_id <> 94401, 'Менеджер', 'Menecer'), team_name));
  end

  line_str = f_format('%1 <b>', iif(cm_database_id <> 94401, 'Период с', 'Müddət')) || extract(day from date_begin) || ' ' ||(case
                                                                                                                             when extract(month from date_begin) = 1 then iif(cm_database_id <> 94401, 'янв.', 'yanv')
                                                                                                                             when extract(month from date_begin) = 2 then iif(cm_database_id <> 94401, 'фев.', 'fevr')
                                                                                                                             when extract(month from date_begin) = 3 then iif(cm_database_id <> 94401, 'март', 'mart')
                                                                                                                             when extract(month from date_begin) = 4 then iif(cm_database_id <> 94401, 'апр.', 'aprl')
                                                                                                                             when extract(month from date_begin) = 5 then iif(cm_database_id <> 94401, 'май', 'may')
                                                                                                                             when extract(month from date_begin) = 6 then iif(cm_database_id <> 94401, 'июня.', 'iyun')
                                                                                                                             when extract(month from date_begin) = 7 then iif(cm_database_id <> 94401, 'июля', 'iyul')
                                                                                                                             when extract(month from date_begin) = 8 then iif(cm_database_id <> 94401, 'авг.', 'avqs')
                                                                                                                             when extract(month from date_begin) = 9 then iif(cm_database_id <> 94401, 'сен.', 'sent')
                                                                                                                             when extract(month from date_begin) = 10 then iif(cm_database_id <> 94401, 'окт.', 'oktb')
                                                                                                                             when extract(month from date_begin) = 11 then iif(cm_database_id <> 94401, 'ноя.', 'noyb')
                                                                                                                             when extract(month from date_begin) = 12 then iif(cm_database_id <> 94401, 'дек.', 'dekb')
                                                                                                                           end) || iif(cm_database_id <> 94401, ' по ', ' tarixinədək ') ||

  extract(day from date_end) || ' ' ||(case
                                         when extract(month from date_end) = 1 then iif(cm_database_id <> 94401, 'янв.', 'yanv')
                                         when extract(month from date_end) = 2 then iif(cm_database_id <> 94401, 'фев.', 'fevr')
                                         when extract(month from date_end) = 3 then iif(cm_database_id <> 94401, 'март', 'mart')
                                         when extract(month from date_end) = 4 then iif(cm_database_id <> 94401, 'апр.', 'aprl')
                                         when extract(month from date_end) = 5 then iif(cm_database_id <> 94401, 'май', 'may')
                                         when extract(month from date_end) = 6 then iif(cm_database_id <> 94401, 'июня.', 'iyun')
                                         when extract(month from date_end) = 7 then iif(cm_database_id <> 94401, 'июля', 'iyul')
                                         when extract(month from date_end) = 8 then iif(cm_database_id <> 94401, 'авг.', 'avqs')
                                         when extract(month from date_end) = 9 then iif(cm_database_id <> 94401, 'сен.', 'sent')
                                         when extract(month from date_end) = 10 then iif(cm_database_id <> 94401, 'окт.', 'oktb')
                                         when extract(month from date_end) = 11 then iif(cm_database_id <> 94401, 'ноя.', 'noyb')
                                         when extract(month from date_end) = 12 then iif(cm_database_id <> 94401, 'дек.', 'dekb')
                                       end) || '</b>';
  res_html = BLOB_APPEND(res_html, line_str || '<br>');

  line_str = '<b>' || period_name || '</b>';
  res_html = BLOB_APPEND(res_html, line_str || '<br><br>');

  curr_color = color_black;
  if (order_in_group <= top_level) then
    curr_color = color_green;
  if (max_order_no - order_in_group <= (top_level - 1)) then
    curr_color = color_red;

  if (is_olimpic = 1) then
  begin
    line_str = f_format('%1: <b><font color="%2" style="font-size: 200%">%3</font></b> %4 <b>%5</b>', iif(cm_database_id <> 94401, 'Место в группе', 'Qrupda yer'),
                                                                                                      curr_color,
                                                                                                      order_in_group,
                                                                                                      iif(cm_database_id <> 94401, 'из', 'from'),
                                                                                                      max_in_group);
    res_html = BLOB_APPEND(res_html, line_str || '<br>');

    line_str = f_format('%1: <b>', iif(cm_database_id <> 94401, 'Балл команды', 'Komanda balı')) || round(final_points, 2) || '</b>';
    res_html = BLOB_APPEND(res_html, line_str || '<br><br>');
  end
  else
  begin
    line_str = f_format('%1: <b><font color="', iif(cm_database_id <> 94401, 'Место в рейтинге', 'Reytinq üzrə yer')) || curr_color || '" style="font-size: 200%">' || FINAL_ORDERNO  ||   -- был order_in_group
               f_format('</font></b> %1 <b>%2</b>', iif(cm_database_id <> 94401, 'из', 'from'), max_in_group);
    res_html = BLOB_APPEND(res_html, line_str || '<br>');

    line_str = f_format('%1: <b>%2</b>', iif(cm_database_id <> 94401, 'Балл менеджера', 'Menecerin balı'), round(final_points, 2));
    res_html = BLOB_APPEND(res_html, line_str || '<br><br>');
  end

  res_html = BLOB_APPEND(res_html, '<table width="100%">
       <col width="35%">
       <col width="42%">
       <col width="10%">
       <col width="13%">
       <tr>' || f_format('<th>%1</th><th>%2</th><th>%3</th><th>%4</th>', iif(cm_database_id <> 94401, 'Тройка лидеров', 'Lider Üçlüyü'),
                                                                         iif(cm_database_id <> 94401, 'Офис', 'Filial'),
                                                                         iif(cm_database_id <> 94401, 'Место', 'Yer'),
                                                                         iif(cm_database_id <> 94401, 'Балл', 'Ball')) || '</tr>');
  if (is_olimpic = 1) then
  begin
    for select '<tr>' || '<td align="left">' || rd_name || '<td align="left">' ||(select res_region_name
                                                                                  from aarep_get_team_office(cpr_object_id, cpp_date_end)) || '<td align="center">' || round(cpr_order_in_group, 0) || '</td>' || '<td align="center">' || round(cpr_final_points, 2) || '</td></tr>'
        from competresult, refdata, competperiod
        where cpr_from_cpp_id = :obj_id and
              cpp_id = :obj_id and
              cpr_object_kind = -1 and
              cpr_object_id = rd_id and
              cpr_group_name = :group_name and
              cpr_order_in_group <= 3
        order by cpr_order_in_group
        into str
    do
    begin
      res_html = BLOB_APPEND(res_html, str);
    end
    res_html = BLOB_APPEND(res_html, '</table>');
  end
  else
  begin
    for select '<tr>' || '<td align="left">' || c_name_fio || '<td align="left">' || cpr_group_name || '<td align="center">' || round(CPR_FINAL_ORDERNO/* cpr_order_in_group*/, 0) || '</td>' || '<td align="center">' || round(cpr_final_points, 2) || '</td></tr>'
        from competresult, clients, competperiod
        where cpr_from_cpp_id = :obj_id and
              cpp_id = :obj_id and
              cpr_object_kind = 0 and
              cpr_object_id = c_id
--            and cpr_order_in_group <= 3
--        order by cpr_order_in_group
        order by CPR_FINAL_ORDERNO
        into str
    do
    begin
      res_html = BLOB_APPEND(res_html, str);
    end
    res_html = BLOB_APPEND(res_html, '</table>');
  end

  if (coalesce(lo_comment, '') <> '') then
  begin
    res_html = BLOB_APPEND(res_html, '<br><br>');
    res_html = BLOB_APPEND(res_html, f_format('%1<br> <b>%2</b>:<br><br>', iif(cm_database_id <> 94401, 'Детализация сотрудника', 'Əməkdaş üzrə göstəricilər'), lo_fio));
    res_html = BLOB_APPEND(res_html, lo_comment, '<br>');
  end
  else
  begin
    res_html = BLOB_APPEND(res_html, '<br><br>');
  end

  if (is_olimpic = 1) then
  begin
    if (cpp_stage = 0) then
    begin
      res_html = BLOB_APPEND(res_html, '<table width="100%">
           <col width="75%">
           <col width="25%">
           <tr>' || f_format('<th>%1</th><th>%2</th></tr>', iif(cm_database_id <> 94401, 'Состав команды', 'Состав команды'),
                                                            iif(cm_database_id <> 94401, 'Балл', 'Ball')));

      for select '<tr>' || '<td align="left">' || iif(c_name_fio = :lo_fio, '<b>', '') || c_name_fio || iif(c_name_fio = :lo_fio, '</b>', '') || '</td>' || '<td align="center">' || round(cpd_points_all, 2) || '</td></tr>'
          from competdetails, clients
          where cpd_from_cpr_id = :cpr_id and
                cpd_object_kind = 0 and
                cpd_object_id = c_id
          order by cpd_points_all desc
          into :str
      do
      begin
        res_html = BLOB_APPEND(res_html, str);
      end
      res_html = BLOB_APPEND(res_html, '</table>');
      res_html = BLOB_APPEND(res_html, '</body>');
    end
  end

  line_str = f_format('<p align="center"><font color="%1"<b>%2!</b></font>', color_green, iif(cm_database_id <> 94401, 'Желаем Вам успехов', 'Sizə uğurlar arzulayırıq'));
  res_html = BLOB_APPEND(res_html, line_str || '</p>');

  suspend;
end
