CREATE PROCEDURE HP_SEND_PHOTO_TO_VERIFY(
    FRL_ID D_ID)
AS
declare variable c_id     d_id;
declare variable frl_koef d_currency;
declare variable perc     d_currency;
declare variable reg_id   d_id;
begin
  select frl.frl_from_c_id, frl.frl_koef
  from facerecoglist frl
  where frl.frl_id = :frl_id
  into c_id, frl_koef;

  select c.c_from_reg_id
  from clients c
  where c.c_id = :c_id
  into reg_id;
  if (frl_koef > 92) then
  begin
    update actionlist acl
    set acl.acl_status = 2
    where acl.acl_from_reg_id = :reg_id and
          acl.acl_object_kind = f_objkind('clients', 'id') and
          acl.acl_object_id = :c_id and
          acl.acl_from_rd_id = f_get_rd_id('ActionGroup.Photo_Verify') and
          acl.acl_status in (0,1);

    update clients
    set c_verify_status = 2
    where c_id = :c_id;
    insert into actionlist (acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                          acl_respons_c_id, acl_from_un_id, acl_dont_show_in_order, acl_quota_used)
    values (current_date, :reg_id, 0, :c_id, f_get_rd_id('ActionGroup.Passport_Verify'), 2, null, 1000000000000, 0, 0);
  end
end
