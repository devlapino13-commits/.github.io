CREATE PROCEDURE AAREP_TMP_PERIOD_REPORT(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    PERIOD_KIND VARCHAR(20))
RETURNS (
    DATE_BEGIN DATE,
    DATE_END DATE,
    REG_ID INTEGER,
    PR_ID DECIMAL(18,0),
    DELINQ_KIND VARCHAR(100),
    REP_INDICATOR VARCHAR(100),
    REP_AMOUNT CURR_TYPE)
AS
declare variable tmp_amount decimal(18,4) = 0;
declare variable kol integer;
declare variable tmp_amount2 decimal(18,4) = 0;
declare variable tmp_amount_par0 decimal(18,4) = 0;
declare variable tmp_amount_par7 decimal(18,4) = 0;
declare variable tmp_amount_par30 decimal(18,4) = 0;
declare variable kol2 integer;
declare variable day_end integer;
declare variable tmp_amount_0 decimal(18,4) = 0;
declare variable tmp_amount_30 decimal(18,4) = 0;
declare variable tmp_amount_60 decimal(18,4) = 0;
declare variable tmp_amount_90 decimal(18,4) = 0;
declare variable tmp_amount_120 decimal(18,4) = 0;
declare variable tmp_amount_150 decimal(18,4) = 0;
declare variable tmp_amount_180 decimal(18,4) = 0;
declare variable tmp_amount_210 decimal(18,4) = 0;
declare variable tmp_amount_240 decimal(18,4) = 0;
declare variable tmp_amount_270 decimal(18,4) = 0;
declare variable tmp_amount_300 decimal(18,4) = 0;
declare variable tmp_amount_330 decimal(18,4) = 0;
declare variable tmp_amount_360 decimal(18,4) = 0;
declare variable tmp_amount_over_360 decimal(18,4) = 0;
begin
  if (period_kind = 'month') then
  begin
    day_end = 390;
    date_begin = :rep_date_begin;
    date_begin = date_begin - extract(day from :date_begin) + 1;
    date_end = date_begin + 33;
    date_end = date_end - extract(day from :date_end);
  end
  else
  begin
    day_end = 367;
    date_begin = :rep_date_begin;
    date_end = date_begin + 6;
  end
  while (date_end <= rep_date_end) do
  begin
    rep_amount = 0;

    for select oks_region_id, k_from_pr_id,
               iif(datediff(day, oks_delinq_date, :date_end) <= 360, 'Прос <=360', 'Прос > 360'),
               sum(oks_balance), sum(k_kolich_client_approved),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 361 and :day_end, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 361 and :day_end, k_kolich_client_approved, 0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) >= 0, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) >= 8, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) >= 31, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) < 0, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 0 and 30, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 31 and 60, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 61 and 90, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 91 and 120, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 121 and 150, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 151 and 180, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 181 and 210, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 211 and 240, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 241 and 270, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 271 and 300, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 301 and 330, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) between 331 and 360, oks_balance, 0.0)),
               sum(iif(datediff(day, oks_delinq_date, :date_end) > 360, oks_balance, 0.0))
        from oksummary, kontract
        where oks_date_begin <= :date_end and
              oks_date_end >= :date_end and
              oks_balance > 0 and
              oks_from_k_id = k_id
              and oks_close_state = 0
        group by 1, 2, 3
        into :reg_id, pr_id, delinq_kind, tmp_amount, kol, tmp_amount2, kol2, tmp_amount_par0, tmp_amount_par7,
             tmp_amount_par30, tmp_amount_0, tmp_amount_30, tmp_amount_60, tmp_amount_90, tmp_amount_120,
             tmp_amount_150, tmp_amount_180, tmp_amount_210, tmp_amount_240, tmp_amount_270, tmp_amount_300,
             tmp_amount_330, tmp_amount_360, tmp_amount_over_360
    do
    begin
      rep_indicator = 'КП, сумма';
      rep_amount = tmp_amount;
      suspend;

      rep_indicator = 'КП, кол-во';
      rep_amount = kol;
      suspend;

      rep_indicator = 'Баланс прос >=0';
      rep_amount = tmp_amount_par0;
      suspend;

      rep_indicator = 'Баланс прос >7';
      rep_amount = tmp_amount_par7;
      suspend;

      rep_indicator = 'Баланс прос >30';
      rep_amount = tmp_amount_par30;
      suspend;

      if (tmp_amount2 is null) then
        tmp_amount2 = 0;
      rep_indicator = 'Сумма переход 360+';
      rep_amount = coalesce(tmp_amount2, 0.0);
      suspend;

      if (kol2 is null) then
        kol2 = 0;
      rep_indicator = 'Кол-во переход 360+';
      rep_amount = coalesce(kol2, 0.0);
      suspend;

      rep_indicator = 'A Баланс без прос.';
      rep_amount = coalesce(tmp_amount_0, 0.00);
      suspend;

      rep_indicator = 'B Баланс прос. 0-30';
      rep_amount = coalesce(tmp_amount_30, 0.00);
      suspend;

      rep_indicator = 'C Баланс прос. 31-60';
      rep_amount = coalesce(tmp_amount_60, 0.00);
      suspend;

      rep_indicator = 'D Баланс прос. 61-90';
      rep_amount = coalesce(tmp_amount_90, 0.00);
      suspend;

      rep_indicator = 'E Баланс прос. 91-120';
      rep_amount = coalesce(tmp_amount_120, 0.00);
      suspend;

      rep_indicator = 'F Баланс прос. 121-150';
      rep_amount = coalesce(tmp_amount_150, 0.00);
      suspend;

      rep_indicator = 'G Баланс прос. 151-180';
      rep_amount = coalesce(tmp_amount_180, 0.00);
      suspend;

      rep_indicator = 'H Баланс прос. 181-210';
      rep_amount = coalesce(tmp_amount_210, 0.00);
      suspend;

      rep_indicator = 'I Баланс прос. 211-240';
      rep_amount = coalesce(tmp_amount_240, 0.00);
      suspend;

      rep_indicator = 'J Баланс прос. 241-270';
      rep_amount = coalesce(tmp_amount_270, 0.00);
      suspend;

      rep_indicator = 'K Баланс прос. 271-300';
      rep_amount = coalesce(tmp_amount_300, 0.00);
      suspend;

      rep_indicator = 'L Баланс прос. 301-330';
      rep_amount = coalesce(tmp_amount_330, 0.00);
      suspend;

      rep_indicator = 'M Баланс прос. 331-360';
      rep_amount = coalesce(tmp_amount_360, 0.00);
      suspend;

      rep_indicator = 'N Баланс прос. свыше 360';
      rep_amount = coalesce(tmp_amount_over_360, 0.00);
      suspend;
    end

    tmp_amount = 0;
    rep_indicator = 'Погашения';
    for select oks_region_id, k_from_pr_id,
               iif(datediff(day, oks_delinq_date, :date_end) <= 360, 'Прос <=360', 'Прос > 360'), sum(-ok_os)
        from operkred, oksummary, kontract
        where ok_from_k_id = oks_from_k_id and
              ok_type = 4 and
              ok_ol_is_active = 1 and
              ok_ol_date between :date_begin and :date_end and
              oks_date_begin <= :date_end and
              oks_date_end >= :date_end and
              oks_from_k_id = k_id
              and oks_close_state = 0
        group by 1, 2, 3
        into reg_id, pr_id, delinq_kind, rep_amount
    do
    begin
      suspend;
    end

    delinq_kind = 'Прос <=360';
    for select ks_region_id, k_from_pr_id, sum(k_amount_approved), sum(k_kolich_client_approved)
        from kontstate, kontract
        where ks_date_begin between :date_begin and :date_end and
              ks_state = 1 and
              ks_ol_is_delete = 0 and
              ks_from_k_id = k_id
        group by ks_region_id, k_from_pr_id
        into reg_id, pr_id, tmp_amount, kol
    do
    begin
      rep_indicator = 'Выдачи, сумма';
      rep_amount = tmp_amount;
      suspend;

      rep_indicator = 'Выдачи, кол-во';
      rep_amount = kol;
      suspend;
    end

    tmp_amount = 0;
    rep_indicator = 'Погашения';
    for select ks_region_id, k_from_pr_id, sum(-ok_os)
        from operkred, kontstate, kontract
        where ok_from_k_id = ks_from_k_id and
              ok_type = 4 and
              ok_ol_is_active = 1 and
              ok_ol_date between :date_begin and :date_end and
              ks_date_begin between :date_begin and :date_end and
              ks_ol_is_delete = 0 and
              ks_state = 2 and
              ks_from_k_id = k_id and
              ks_date_end >= '31.12.2999'
        group by 1, 2
        into reg_id, pr_id, rep_amount
    do
    begin
      suspend;
    end

    if (period_kind = 'month') then
    begin
      date_begin = date_end + 1;
      date_end = date_begin + 33;
      date_end = date_end - extract(day from :date_end);
    end
    else
    begin
      date_begin = date_begin + 7;
      date_end = date_end + 7;
    end
  end
end
