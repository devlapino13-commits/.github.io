CREATE PROCEDURE HP_DOCRECOGREQUEST_LIST(
    UN_ID D_ID)
RETURNS (
    DRR_ID D_INTEGER,
    DRL_ID D_ID,
    C_NAME_FIO D_STR_100,
    DP_NAME D_STR_100,
    DRM_NAME D_STR_100,
    DRR_STATUS D_INTEGER,
    DRR_STATUS_STR D_STR_CODE,
    DRR_CREATE_DATETIME D_DATETIME,
    DRR_STATUS_DATETIME D_DATETIME)
AS
begin
  for select drr_id, drr_from_drl_id, c_name_fio, dp_name, drm_name, drr_status, drr_create_datetime, drr_status_datetime
  from docrecogrequest
       inner join clients on (drr_from_c_id = c_id)
       inner join docprop on (drr_from_dp_id = dp_id)
       left join docrecogmain on (drr_from_drm_id = drm_id)
  where drr_from_un_id = :un_id and
        (drr_status in (-1, 0, 1) or
        (drr_status = 2 and drr_status_datetime >= current_date - 1))
  into drr_id, drl_id, c_name_fio, dp_name, drm_name, drr_status, drr_create_datetime, drr_status_datetime
  do
  begin
    if (drm_name is null) then
      drm_name = '';

    drr_status_str = '';

    drr_status_str = case drr_status
      when -2 then 'X'  -- Аннулирован
      when -1 then '!'  -- Переснять
      when 0 then '□'   -- Новый
      when 1 then '•'   -- Прикреплен
      when 2 then '√'   -- Готов
    end;

    suspend;
  end
end
