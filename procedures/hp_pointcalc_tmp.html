CREATE PROCEDURE HP_POINTCALC_TMP(
    C_ID D_ID,
    REP_DATE D_DATE,
    CALC_TYPE D_SMALLINT = 0)
RETURNS (
    K_ID D_ID,
    SH_DATE D_DATE,
    IS_LEGIT D_SMALLINT,
    IS_POINT D_SMALLINT,
    IS_DELINQ D_SMALLINT)
AS
declare variable DB_ID D_INTEGER;
declare variable AMOUNT_PER_CLIENT D_CURRENCY;
declare variable CLIENT_COUNT D_INTEGER;
declare variable ZAD_AMOUNT D_CURRENCY;
declare variable LEGIT_AMOUNT D_CURRENCY;
declare variable POINT_AMOUNT D_CURRENCY;
declare variable MIN_ACTIVE_DATE D_DATE;
declare variable MIN_OL_DATE D_DATE;
declare variable DAYS_1 D_INTEGER;
declare variable DAYS_3 D_INTEGER;
declare variable DAYS_1M1 D_INTEGER;
begin
  if (calc_type is null) then
    calc_type = 0;

  db_id = f_get_db_id();

  if (db_id not in (39801, 41701)) then
    calc_type = 0;

  --Мин. сумма легитимного погашения на клиента
  if (db_id = 81001) then
    amount_per_client = 2500;
  else if (db_id = 41701) then
    amount_per_client = 1500;
  else if (db_id = 94401) then
    amount_per_client = 40;
  else if (db_id = 39801) then
    amount_per_client = 10000;
  else
    exception e_default f_format('unknown db_id: %1', db_id);

  if (db_id = 81001) then
  begin
    days_1 = 8;
    days_3 = 61;
  end else
  begin
    days_1 = 2;
    days_3 = 61;
  end
  days_1m1 = days_1 - 1;

  for select ks_from_k_id, k_kolich_client_approved
  from kontstate, knclient, kontract, product
  where kc_from_c_id = :c_id and
        kc_amount_approved > 0 and
        kc_from_k_id = k_id and
        ks_ol_is_delete = 0 and
        ks_date_begin <= :rep_date and
        ks_date_end >= (:rep_date - 1) and
        ks_state = 1 and
        ks_from_k_id = k_id and
        k_from_pr_id = pr_id and
        (:calc_type = 0 or ((:calc_type = 1 and pr_percent_kind <> 2) or (:calc_type = 2 and pr_percent_kind = 2)))
  into k_id, client_count
  do
  begin
    zad_amount = 0;
    legit_amount = 0;
    point_amount = 0;
    is_legit = 0;
    is_point = 0;
    min_active_date = null;
    is_delinq = 0;

    select min(ok_ol_date)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_date <= :rep_date and
          ok_ol_is_active = 1 and
          ok_type = 4 and
          ((ok_os + ok_perc) < 0) and
          ok_sh_date >= (:rep_date - extract(day from :rep_date) + 1) and
          ok_sh_date - 10 <= ok_ol_date
    into min_ol_date;

    min_ol_date = coalesce(min_ol_date, rep_date);

    for select ok_sh_date
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_date between :min_ol_date and :rep_date and
          ok_ol_is_active = 1 and
          ok_type = 4 and
          (ok_os + ok_perc < 0) and
          ok_sh_date - 10 <= ok_ol_date
    into sh_date
    do
    begin
      zad_amount = 0;
      legit_amount = 0;
      point_amount = 0;
      is_legit = 0;
      is_point = 0;
      select sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc),
             sum(iif(ok_type = 4 and datediff(day, :sh_date, ok_ol_date) >= -10,
                                 -(ok_os + ok_perc), 0)), -- legit_amount
             sum(iif(ok_type = 4 and datediff(day, :sh_date, ok_ol_date) between -10 and :days_1m1,
                                 -(ok_os + ok_perc), 0)) -- point_amount
      from operkred
      where ok_from_k_id = :k_id and
            ok_sh_date = :sh_date and
            ok_ol_is_active = 1 and
            ok_type < 7 and
            ok_ol_date <= :rep_date
      into zad_amount, legit_amount, point_amount;
      if (zad_amount is null) then
        zad_amount = 0;
      if (legit_amount is null) then
        legit_amount = 0;
      if (point_amount is null) then
        point_amount = 0;

       if (zad_amount = 0 and legit_amount >= amount_per_client * client_count) then
       begin
         is_legit = 1;
         if (legit_amount = point_amount) then
           is_point = 1;
       end

       if (is_legit = 1) then
         suspend;
    end

    -- Возращаем даты для штрафов
    is_legit = 0;
    is_point = -1;
    sh_date = null;
    for select ok_sh_date
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_date <= :rep_date and
          ok_ol_is_active = 1 and
          ok_type < 7
    group by ok_sh_date
    having sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc) > 0
    into sh_date
    do
    begin
      if (sh_date between '20.03.2020' and '31.07.2020') then
        sh_date = '31.08.2020';

      is_delinq = 0;
      if (sh_date < rep_date) then
        is_delinq = 1;

      if (min_active_date is null) then
        min_active_date = sh_date;
  
      if (sh_date > dateadd(day, days_3 - days_1, min_active_date)) then
        break;

      suspend;
    end
  end
end
