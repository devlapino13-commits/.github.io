CREATE PROCEDURE AAREP_KN_OPERBOOK_FIX_ZERO_PAIR(
    REP_DATE D_DATE,
    REP_KC_ID D_ID)
AS
declare variable rep_date_begin d_date;
declare variable kobs_acc_id d_id;
declare variable acc_id bigint;
declare variable pair_acc_id bigint;
declare variable kc_id bigint;
declare variable acc_saldo decimal(18,4);
declare variable pair_saldo decimal(18,4);
declare variable ol_date date;
declare variable ol_id bigint;
declare variable orig_ol_id d_id;
declare variable k_id d_id;
declare variable detail_id d_id;
begin
  rep_date_begin = rep_date - extract(day from rep_date) + 1;
  if (coalesce(rep_kc_id, 0) = 0) then
    rep_kc_id = 0;

  for select acc_id,
             acc_pair_acc_id,
             acc_object_id,
             f_acc_bal(acc_id, :rep_date) acc_saldo,
             f_acc_bal(acc_pair_acc_id, :rep_date) as pair_saldo
  from ref_account
  where acc_from_ab_id in (99, 110) and
        acc_type = 0 and
        acc_object_kind = 8 and
        acc_object_id = :rep_kc_id
  into acc_id, pair_acc_id, kc_id, acc_saldo, pair_saldo
  do
  begin
    if ((acc_saldo > 0 and pair_saldo = 0) or
        (acc_saldo = 0 and pair_saldo < 0) or
        (acc_saldo = 0 and pair_saldo = 0)) then
      Continue;

    ol_date = null;
    ol_id = null;

    select kobs_acc_id, kobs_date
    from kn_obsummary
    where kobs_acc_id in (:acc_id, :pair_acc_id) and
          kobs_date between :rep_date_begin and :rep_date
    order by kobs_acc_id desc, kobs_date desc
    rows 1
    into kobs_acc_id, ol_date;

    if (ol_date is not null) then
    begin
      select kob_from_ol_id
      from kn_operbook
      where kob_from_kc_id = :kc_id and
            (kob_acc_dt = :kobs_acc_id or kob_acc_kt = :kobs_acc_id) and
            kob_ol_date = :ol_date and
            kob_ol_is_active = 1
      order by kob_from_ol_id desc
      rows 1
      into ol_id;

      if (ol_id is null) then
      begin
        select kob_from_ol_id
        from kn_operbook
        where kob_from_kc_id = :kc_id and
              kob_ol_date = :rep_date and
              kob_ol_is_active = 1
        order by kob_from_ol_id desc
        rows 1
        into ol_id;
      end
    end

    if (acc_saldo > -pair_saldo) then
      acc_saldo = -pair_saldo;

    if (ol_id is null) then
    begin
      select ol_id
      from hp_do_kn_operbook_ol_memorder(null, :kc_id, '', '', :acc_saldo, 0, :acc_saldo, :acc_saldo,
           'Сворачивание парных счетов', :pair_acc_id, :acc_id, :rep_date, 2, 34, 9, 1010100, 1000000000000)
      into :orig_ol_id;
    end else
    begin
      detail_id = f_kn_detail_id('Сворачивание парных счетов');

      insert into kn_operbook (kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                               kob_summa_tax_dt, kob_summa_tax_kt, kob_from_ol_id, kob_from_kc_id, kob_detail_id)
      values ('810', 1, :pair_acc_id, :acc_id, :acc_saldo, 0, :acc_saldo,
              :acc_saldo, :ol_id, :kc_id, :detail_id);
    end

    select kc_from_k_id
    from knclient
    where kc_id = :kc_id
    into k_id;

--    execute procedure hp_do_kn_operbook_one_online(:k_id, current_date, null, :rep_date);
  end
end
