CREATE PROCEDURE HP_ELDOS_ALLOW_LIST
RETURNS (
    K_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    KF_NAME D_STR_100,
    STATE D_STR_10,
    UN_ID D_ID,
    UN_FIO D_STR_100,
    DT_LOAD D_DATETIME)
AS
declare variable dt_curr d_date;
begin
  dt_curr = current_date;
  for
    select
      k_id,
      iif(k_kontract_no = '', k_id, k_kontract_no),
      k_disb_date,
      kal_new_kf_name,
      case ks_state
        when -1 then 'удален'
        when 0 then 'заявка'
        when 1 then 'активный'
        when 2 then 'закрытый'
      end,
      un_id,
      un_fio,
      eal_datetime
    from eldos_allow_list
      inner join kontract
        on k_id = eal_k_id
      inner join kontstate
        on ks_from_k_id = eal_k_id
          and :dt_curr between ks_date_begin and ks_date_end
          and ks_ol_is_delete = 0
      inner join knaktlist
        on kal_from_kf_id = k_from_kf_id
          and current_date between kal_date and kal_date_end
          and is_delete = 0
      inner join username
        on un_id = eal_un_id
  into
    :k_id,
    :k_kontract_no,
    :k_disb_date,
    :kf_name,
    :state,
    :un_id,
    :un_fio,
    :dt_load
  do
  begin
    suspend;
  end
end
