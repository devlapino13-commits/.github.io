CREATE PROCEDURE HP_EXPRESSAPP_BY_EXTMINIAPP_TMS(
    EMA_ID D_ID)
AS
declare variable C_ID D_ID;
declare variable AMOUNT D_CURRENCY;
declare variable EMA_DATE D_DATE;
declare variable PERIOD D_INTEGER;
declare variable REG_ID D_ID;
declare variable TKT_ID D_ID;
declare variable PR_ID D_INTEGER;
declare variable EXC_ID_BEGIN D_INTEGER;
declare variable PERCENT D_PERCENT;
declare variable REPAY_DAY D_INTEGER;
declare variable EXA_ID D_ID;
declare variable LO_ID D_ID;
declare variable UN_ID D_ID;
declare variable MA_ID D_INTEGER;
declare variable KF_ID D_ID;
declare variable DISB_KIND D_INTEGER;
declare variable LOAN_TARGET D_INTEGER;
declare variable BUSINESS_SUBTYPE D_INTEGER;
declare variable IN_TARIF D_STR_255;
declare variable CLIENT_SHOP_ID D_INTEGER;
declare variable FOREIGN_ECONOM_GOODS_ID D_INTEGER;
declare variable ML_NAME D_STR_100;
declare variable ML_ID D_ID;
declare variable ML_MODEL D_STR_100;
begin
  -- Через МП сотрудника вызывается процедура HP_EXTMINIAPP_WEB_SELL_TMS для формирования предложения для продажи товара в рассрочку
  -- После того как клиент подтверждает через сайт продавца, вызывается процедура HP_EXTMINIAPP_WEB_TMS_UPDATE,
  -- а внутри процедуры HP_EXTMINIAPP_WEB_TMS_UPDATE вызвается текущая процедура.

  -- Данная процедура формировывает заявку на займ

  select ema.ema_from_c_id, ema.ema_amount, ema.ema_date, ema.ema_period, ema.ema_office_id, ema.ema_from_tkt_id,
         ema.ema_from_pr_id, ema.ema_lo_id, ema.ema_from_un_id, ema.ema_address_str
  from extminiapp ema
  where ema.ema_id = :ema_id
  into c_id, amount, ema_date, period, reg_id, tkt_id, pr_id, lo_id, un_id, in_tarif;

  business_subtype = 0;

  -- Уточняем кредитную папку и тип бизнеса клиента по последнему займу
  select k_from_kf_id--, kc_busuness_subtype
  from knclient, kontract
  where kc_from_c_id = :c_id and
        kc_amount_approved > 0 and
        kc_from_k_id = k_id and
        k_status = 1
  order by k_disb_date desc
  rows 1
  into kf_id /*, business_subtype*/;

  -- если нет кредитной папки, создаем новую
  if (kf_id is null) then
  begin
    select kf_id
    from insert_new_kfolder(:reg_id, :lo_id, current_date, 0, '', 0, '')
    into kf_id;
  end

  repay_day = 0; -- день выдачи займа
  percent = 18.25; -- 18% годовых, 1.5% в месяц
  disb_kind = 1; -- На банковски счет
  loan_target = 63; -- уточнить по Товар в рассрочку

  if (coalesce(business_subtype, 0) = 0) then
    business_subtype = 21;

  if (coalesce(period, 0) = 0) then
  begin
    period = case in_tarif
      when 'businessstandart' then 6
      when 'businessstandartkg' then 6
      when 'relationshipstandart' then 6
      when 'migrant' then 9
    end;
  end

  if (in_tarif = 'businessstandart') then
  begin
    ml_name = 'Наставническая Онлайн программа "Энергия Бизнеса"';
    ml_model = 'Тариф "Standart"';
  end else
  if (in_tarif = 'businessstandartkg') then
  begin
    ml_name = 'Наставническая Онлайн программа "Бизнес Энергиясы" (кыргыз тилинде)';
    ml_model = 'Тариф "Standart"';
  end else
  if (in_tarif = 'relationshipstandart') then
  begin
    ml_name = 'Наставническая Онлайн программа "Энергия Отношений"';
    ml_model = 'Тариф "Standart"';
  end else
  if (in_tarif = 'migrant') then
  begin
    ml_name = 'Наставническая Онлайн программа "Наставник Трудового мигранта"';
    ml_model = 'Тариф "PRO"';
  end else
  if (in_tarif = 'energiazhizni') then
  begin
    ml_name = 'Наставническая Онлайн программа "Энергия Жизни"';
    ml_model = 'Тариф "VIP"';
  end

  select ml_id
  from matlist
  where ml_name = :ml_name and
        ml_model = :ml_model
  into ml_id;

  if (ml_id is null) then
  begin
    insert into matlist (ml_name, ml_model, ml_from_yl_id)
    values (:ml_name, :ml_model, 1)
    returning ml_id
    into :ml_id;
  end

  -- создаем заявку на займ Miniapp
  insert into miniapp (ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay,
                       ma_percent, ma_kolich_client, ma_memo, ma_summa_minus,
                       ma_from_un_id, ma_office_id, ma_percent_category, ma_repay_day,
                       ma_disb_kind, ma_from_tkt_id)
  values (current_date, current_date, :kf_id, :pr_id, :period, :percent,
          1, :ml_name, :amount, :un_id,
          :reg_id, :percent, :repay_day, :disb_kind, :tkt_id)
  returning ma_id
  into :ma_id;

  -- добавляем клиента в заявку на займ
  insert into miniappclient (mac_from_reg_id, mac_from_ma_id, mac_from_c_id, mac_amount, mac_is_poruch,
                             mac_loan_target, mac_is_cashier, mac_business_subtype)
  values (:reg_id, :ma_id, :c_id, :amount, 0,
          :loan_target, 1, :business_subtype);

  insert into clp_miniappclient (cmc_id, cmc_from_ma_id, cmc_from_c_id, cmc_questions_id, cmc_clarification_id,
                                 cmc_from_un_id, cmc_datetime)
  values (null, :ma_id, :c_id, 9, 26, :un_id,
          null);

  select exc.exc_id
  from expresscfg exc
  where exc.exc_action_code = 'loan.application' -- 'miniapp'
  into exc_id_begin; -- Проверка заявки

  -- Регистрируем заявку для обработки роботом
  insert into expressapp (exa_from_exp_id, exa_date, exa_group_id, exa_done, exa_from_c_id, exa_base_k_id,
                          exa_from_exc_id, exa_final_status, exa_wait_status, exa_from_ma_id, exa_from_k_id,
                          exa_from_edl_id, exa_disb_date, exa_from_pr_id, exa_amount, exa_period, exa_percent,
                          exa_repay_day, exa_lo_id, exa_from_reg_id, exa_try_count, exa_from_un_id, exa_datetime,
                          exa_reshipping_count)
  values (-2, :ema_date, 31, 0, :c_id, null, :exc_id_begin, 0, 0, :ma_id, null, null, current_date, :pr_id, :amount,
          :period, :percent, :repay_day, :lo_id, :reg_id, 0, :un_id, current_timestamp, 0)
  returning exa_id
  into :exa_id;

  client_shop_id = 6577;           -- Agreement Id
  foreign_econom_goods_id = 1009;  -- Курсы по повышению личностного роста и ораторского искусства
  insert into kontmat (kml_object_kind, kml_object_id, kml_from_ml_id, kml_count, kml_cost, kml_total,
                       kml_guarantee_month, kml_from_ag_id, kml_from_feg_id)
  values (11, :ma_id, :ml_id, 1, :amount, :amount,
          :period, :client_shop_id, :foreign_econom_goods_id);

  /* -- Блок касательно классики, отключен
  client_shop_id = 306;           -- ОсОО Международная Академия Бизнеса и Инвестиций "Мурас"

  -- Прикрепляем товар или усллугу к заявке
  insert into mc_purchase (mp_from_mcl_id, mp_from_cs_id, mp_date, mp_amount,
                           mp_from_feg_id, mp_from_ma_id, mp_from_exa_id)
  values (null, :client_shop_id, current_date, :amount,
         :foreign_econom_goods_id, :ma_id, :exa_id);
  */

  -- Далее запускаем процесс обработки (продвижение) заявки роботом
  execute procedure hp_do_expressapp_process(:exa_id);
end
