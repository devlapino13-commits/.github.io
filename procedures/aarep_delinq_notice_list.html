CREATE PROCEDURE AAREP_DELINQ_NOTICE_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_255,
    K_KONTRACT_NO D_STR_255,
    DELINQ_DAYS D_INTEGER,
    K_DISB_DATE D_DATE,
    SEND_DATE D_DATE,
    PREPARE_DATE D_DATE,
    ROW_NO D_INTEGER,
    IS_RESTRUCT_STR D_STR_10)
AS
declare variable EDL_ID D_ID;
declare variable K_ID D_ID;
declare variable C_ID D_ID;
declare variable DELINQ_DATE D_DATE;
declare variable ROW_COUNTER D_INTEGER;
declare variable IS_RESTRUCT d_str_10 ;
begin

row_counter = 0;
  for select edl.edl_id, 
             k.k_id,
             k.k_kontract_no,
             k.k_disb_date, 
             c.c_name_fio, 
             cast(edl.edl_create_datetime as date)
  from edoc_list edl
  join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id
  join kontract k on k.k_id = edl.edl_object_id
  join knclient kc on kc.kc_from_k_id = k.k_id
  join clients c on c.c_id = kc.kc_from_c_id
  where edl.edl_object_kind = f_objkind('kontract', 'id')
        and edc.edc_code in ('loan.delinq.notif','restruct.loan.delinq.notif')
        and cast(edl.edl_create_datetime as date) between :rep_date_begin and :rep_date_end
        and kc.kc_amount_approved > 0
  order by edl.edl_create_datetime asc
  into :edl_id, 
       :k_id,
       :k_kontract_no,
       :k_disb_date,
       :c_name_fio,
       :prepare_date
  do
  begin
    select cast(edl2.edl_create_datetime as date)
    from edoc_list edl2
    join edoc_blank edb on edb.edb_from_edl_id = edl2.edl_id
    where edl2.edl_id = :edl_id
      and edl2.edl_blank_status = 1
    into :send_date;

    delinq_date = coalesce(send_date, prepare_date);

    select :delinq_date - oks.oks_delinq_date
    from oksummary oks
        where oks.oks_from_k_id = :k_id
          and oks.oks_close_state = 0
          and :delinq_date between oks.oks_date_begin and oks.oks_date_end
    into :delinq_days;

    select is_restruct
    from aarep_get_restructuring(:k_id, :rep_date_end, 39801)
    into :is_restruct;

    if (is_restruct = 1) then
      is_restruct_str = 'Да';
    else
      is_restruct_str = 'Нет';

    row_counter = row_counter + 1;
    row_no = row_counter;
    suspend;
  end
end
