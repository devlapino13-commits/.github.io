CREATE PROCEDURE HP_SUD_COL_DEFINE_COLLECTOR(
    SLC_ID D_ID NOT NULL)
AS
declare variable address d_str_long;
declare variable c_id d_id;
declare variable eil_id d_id;
declare variable kind d_integer;
begin
  exit; -- пока отключаем запросы подсудности

  select slc_address, kc_from_c_id
  from sud_loanclient, knclient
  where slc_id = :slc_id
    and kc_id = slc_from_kc_id
  into address, c_id;

  if (coalesce(address, '') = '') then
    exception e_message 'У клиента не заполнен адрес подсудности';

  kind = 3;

  if (exists(select eil_status
            from hp_sud_get_request_status(:slc_id, :kind)
            where eil_status >= 0))
  then
    exit;

  select eil_id
  from hp_sud_extinfo_insert('DigiDebt_court', 'sud_maincollect', :slc_id, :c_id, 1)
  into eil_id;

  insert into sud_extinfo (sue_slc_id, sue_kind, sue_date, sue_from_eil_id)
  values (:slc_id, :kind, current_date, :eil_id);
end
