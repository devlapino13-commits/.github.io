CREATE PROCEDURE AAREP_ANTIFROD_CONSOLIDATED(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_C_ID D_ID)
RETURNS (
    REP_FIO STR_100,
    REP_BIRTHDATE D_DATE,
    REP_KONTRACT_NO D_STR_20,
    REP_APPLICATE_NO D_INTEGER,
    REP_APPLICATE_DATE D_DATE,
    REP_CHECK_DATETIME D_DATETIME,
    RES_BKI D_INTEGER,
    RES_PHONE D_INTEGER,
    RES_DEVICE D_INTEGER,
    RES_VIDEOCERTIFICATION D_INTEGER,
    RES_TOTAL D_INTEGER,
    RISK_LEVEL D_STR_CODE,
    REP_APP_STATUS D_STR_CODE,
    REG_NAME D_STR_255)
AS
declare variable k_id d_id;
declare variable ma_id d_id;
declare variable k_status d_integer;
declare variable ma_is_active d_integer;

declare function myf_get_res(k_id d_id, afc_code d_str_code)
returns d_percent
as
begin
  return (select all_result
          from antifrod_loanlist, antifrod_categories
          where all_from_afl_id = :k_id
            and afc_id = all_from_afc_id
            and afc_code = :afc_code);
end
 
declare procedure hp_ma_list (reg_id d_integer, rep_date_begin d_date, rep_date_end d_date, rep_c_id d_id)
returns (rep_ma_id d_id)
as
begin
  if (rep_c_id is not null) then
    for select mac_from_ma_id
    from miniappclient
    where mac_from_c_id = :rep_c_id
    into rep_ma_id
    do
      suspend;
  else
    for select ma_id
    from region_list(:reg_id, 1, -1, 1), miniapp
    where ma_office_id = id_list
      and ma_number = 0
      and ma_date_app between :rep_date_begin and :rep_date_end
    into rep_ma_id
    do
      suspend;
end


begin
  for select afl_datetime, afl_risk_level, k_id, k_applicate_no, k_applicate_date, k_kontract_no, k_status, rep_ma_id, reg_code || ' ' || reg_name
  from hp_ma_list(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_c_id), kontract, region, antifrod_loan
  where k_from_ma_app_id = rep_ma_id
    and reg_unicode = k_last_reg_id
    and afl_from_k_id = k_id
    and afl_datetime < current_date -- сегодняшние не показываем
  into rep_check_datetime, risk_level, k_id, rep_applicate_no, rep_applicate_date, rep_kontract_no, k_status, ma_id, reg_name
  do
  begin
    if (f_loan_is_online(k_id) = 0) then
      continue;

    select sum(all_result)
    from antifrod_loanlist
    where all_from_afl_id = :k_id
    into res_total;

    res_total = minvalue(res_total, 100);
    res_total = maxvalue(res_total, 0);

    select c_name_fio, c_birth_date
    from knclient, clients
    where c_id = kc_from_c_id
      and kc_from_k_id = :k_id
    into rep_fio, rep_birthdate;

    res_bki                 = myf_get_res(k_id, '5.8.2.1');
    res_phone               = myf_get_res(k_id, '5.8.2.2');
    res_device              = myf_get_res(k_id, '5.8.2.3');
    res_videocertification  = myf_get_res(k_id, '5.8.2.4');

    risk_level = case risk_level
                   when 0 then 'Низкий'
                   when 1 then 'Значительный'
                   else 'Высокий'
                 end;

    if (k_status > 0) then
    begin
      rep_app_status = 'Утвержден';
    
      select 'Выдан'
      from kontstate
      where ks_from_k_id = :k_id
        and ks_state = 1
        and ks_ol_is_delete = 0
      into rep_app_status;
    end
    else
    begin
      select ma_is_active
      from miniapp
      where ma_id = :ma_id
        and ma_date_end = '31.12.9999'
      into ma_is_active;

      rep_app_status = iif(ma_is_active < 1, 'Отказ', 'В процессе');
    end

    suspend;
  end 
end
