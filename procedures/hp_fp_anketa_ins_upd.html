CREATE PROCEDURE HP_FP_ANKETA_INS_UPD(
    FPA_ID ID_TYPE,
    FPA_C_ID ID_TYPE,
    REP_DATE DATE,
    REP_UN_ID ID_TYPE,
    FPA_TYPE SMALLINT,
    FPA_CITY ID_TYPE,
    FPA_CLIENT_FIO STR_255,
    FPA_CITIZEN INTEGER,
    FPA_DATE_PLACE_BIRTH STR_255,
    FPA_PASS_KIND INTEGER,
    FPA_PASS_PROPS STR_255,
    FPA_MCARD_PROPS STR_255,
    FPA_ACARD_PROPS STR_255,
    FPA_INN STR_255,
    FPA_ADDRESS_LEGAL STR_255,
    FPA_REL_START DATE,
    FPA_ANK_RESP ID_TYPE,
    FPA_DATE_CHECK DATE,
    FPA_CELL_NO STR_20,
    FPA_EMAIL STR_100,
    FPA_SNILS STR_100,
    UPD_ONLY_CHANGE_STR SMALLINT)
AS
declare variable todo smallint;
declare variable old_fpa_client_fio str_255;
declare variable old_fpa_citizen integer;
declare variable old_fpa_date_place_birth str_255;
declare variable old_fpa_pass_kind integer;
declare variable old_fpa_pass_props str_255;
declare variable old_fpa_mcard_props str_255;
declare variable old_fpa_acard_props str_255;
declare variable old_fpa_inn str_255;
declare variable old_fpa_address_legal str_255;
declare variable old_fpa_cell_no str_20;
declare variable old_fpa_email str_100;
declare variable changes_str varchar(1000);
declare variable citizen_str str_100;
declare variable buf_str1 str_100;
declare variable buf_str2 str_100;
declare variable fpa_pass_kind_str str_100;
declare variable fpa_date_str str_100;
declare variable fpa_changes varchar(1000);
declare variable is_insert smallint;
declare variable old_fpa_snils str_100;
declare variable fpa_changes_body varchar(1000);
begin
  todo = 0;
  fpa_changes = '';
  if (fpa_id = 0) then -- insert
  begin
    select max(fpa_id)
    from fp_anketa_cl
    where fpa_c_id = :fpa_c_id
    into fpa_id;

    if (coalesce(fpa_id, 0) = 0) then
      fpa_id = 1;
    else
      fpa_id = fpa_id + 1;

    if (fpa_id = 1) then -- если это первая запись, то обновление не требуется, делаю только insert
    begin
      insert into fp_anketa_cl (fpa_id, fpa_c_id, fpa_from_un_id, fpa_city, fpa_citizen, fpa_client_fio,
                                fpa_date_place_birth, fpa_pass_kind_id, fpa_pass_props, fpa_mcard_props,
                                fpa_acard_props, fpa_inn, fpa_address_legal, fpa_rel_start, fpa_resp_c_id, fpa_type,
                                fpa_date, fpa_date_check, fpa_cell_no, fpa_email, fpa_snils)
      values (:fpa_id, :fpa_c_id, :rep_un_id, :fpa_city, :fpa_citizen, :fpa_client_fio, :fpa_date_place_birth,
              :fpa_pass_kind, :fpa_pass_props, :fpa_mcard_props, :fpa_acard_props, :fpa_inn, :fpa_address_legal,
              :fpa_rel_start, :fpa_ank_resp, :fpa_type, :rep_date, :fpa_date_check, :fpa_cell_no, :fpa_email,
              :fpa_snils);

      todo = 0;
      is_insert = 1;
    end
    else -- если ранне по этому челу уже имется анкета, тогда трубуется обновление
    begin
      todo = 1;
      is_insert = 1;
    end
  end

-- update
  else
  begin
    todo = 1;
    is_insert = 0;
  end

  if (todo = 1) then
  begin
    fpa_changes = '';

    if (fpa_id > 1) then
    begin
      if (upd_only_change_str = 1) then
      begin
        select fpa_client_fio, fpa_citizen, fpa_date_place_birth, fpa_pass_kind_id, fpa_pass_props, fpa_mcard_props,
               fpa_acard_props, fpa_inn, fpa_address_legal, fpa.fpa_cell_no, fpa.fpa_email, fpa.fpa_date, fpa_snils
        from fp_anketa_cl fpa
        where fpa.fpa_c_id = :fpa_c_id and
              fpa.fpa_id = :fpa_id
        into :fpa_client_fio, :fpa_citizen, :fpa_date_place_birth, :fpa_pass_kind, :fpa_pass_props, :fpa_mcard_props,
             :fpa_acard_props, :fpa_inn, :fpa_address_legal, :fpa_cell_no, :fpa_email, :rep_date, :fpa_snils;
      end
-- поднимаю последнюю созданную анкету на дату
      select fpa_client_fio, fpa_citizen, fpa_date_place_birth, fpa_pass_kind_id, fpa_pass_props, fpa_mcard_props,
             fpa_acard_props, fpa_inn, fpa_address_legal, fpa.fpa_cell_no, fpa.fpa_email, fpa_snils
      from fp_anketa_cl fpa
      where fpa.fpa_c_id = :fpa_c_id
      order by fpa_date desc
      rows 1
      into :old_fpa_client_fio, :old_fpa_citizen, :old_fpa_date_place_birth, :old_fpa_pass_kind, :old_fpa_pass_props,
           :old_fpa_mcard_props, :old_fpa_acard_props, :old_fpa_inn, :old_fpa_address_legal, :old_fpa_cell_no,
           :old_fpa_email, :old_fpa_snils;

      changes_str = '';
      if (coalesce(fpa_client_fio, '') <> coalesce(old_fpa_client_fio, '')) then
      begin
        changes_str = changes_str || 'ФИО изменилось: ' || coalesce(fpa_client_fio, '') || '; ';
      end
      if (coalesce(old_fpa_citizen, -777) <> coalesce(fpa_citizen, -777)) then
      begin
        select ddr.rd_name
        from refdata ddr
        where ddr.rd_id = :fpa_citizen
        into :citizen_str;

        changes_str = changes_str || 'гражданство изменилось: ' || coalesce(citizen_str, '') || '; ';

        if (fpa_citizen = 12) then
          changes_str = changes_str || 'степерь (уровень) риска изменилась на "высокий"; ';
        else
          changes_str = changes_str || 'степерь (уровень) риска изменилась на "стандартный"; ';
      end
      if (coalesce(old_fpa_date_place_birth, '') <> coalesce(fpa_date_place_birth, '')) then
      begin
        if ((character_length(old_fpa_date_place_birth) = 10) and
            (character_length(fpa_date_place_birth) >= 10)) then
        begin
          buf_str1 = substring(old_fpa_date_place_birth from 1 for 10);
          buf_str2 = substring(fpa_date_place_birth from 1 for 10);
          if (buf_str1 <> buf_str2) then
            changes_str = changes_str || 'дата, место рождения: ' || coalesce(fpa_date_place_birth, '') || '; ';
        end
        else
          changes_str = changes_str || 'дата, место рождения: ' || coalesce(fpa_date_place_birth, '') || '; ';
      end
      if (coalesce(old_fpa_pass_kind, '') <> coalesce(fpa_pass_kind, '')) then
      begin
        fpa_pass_kind_str = '';
        if (fpa_pass_kind is not null) then
        begin
          select pask.pk_name
          from passkind pask
          where pask.pk_id = :fpa_pass_kind
          into :fpa_pass_kind_str;
        end
        changes_str = changes_str || 'Наименование документа, удостоверяющего личность: ' || coalesce(fpa_pass_kind_str, '') || '; ';
      end
      if (coalesce(old_fpa_pass_props, '') <> coalesce(fpa_pass_props, '')) then
        changes_str = changes_str || 'Реквизиты документа, удостоверяющего личность: ' || coalesce(fpa_pass_props, '') || '; ';

      if (coalesce(old_fpa_mcard_props, '') <> coalesce(fpa_mcard_props, '')) then
        changes_str = changes_str || 'Данные миграционной карты: ' || coalesce(fpa_mcard_props, '') || '; ';

      if (coalesce(old_fpa_acard_props, '') <> coalesce(fpa_acard_props, '')) then
        changes_str = changes_str || 'Легальность в РФ: ' || coalesce(fpa_acard_props, '') || '; ';

      if (coalesce(old_fpa_inn, '') <> coalesce(fpa_inn, '')) then
        changes_str = changes_str || 'ИНН: ' || fpa_inn || '; ';

      if (coalesce(old_fpa_address_legal, '') <> coalesce(fpa_address_legal, '')) then
      begin
        changes_str = changes_str || 'Адрес регистрации: ' || coalesce(fpa_address_legal, '') || '; ';
      end

      if (coalesce(old_fpa_snils, '') <> coalesce(fpa_snils, '')) then
      begin
        changes_str = changes_str || 'СНИЛС: ' || coalesce(fpa_snils, '') || '; ';
      end
--------------------------------
      if ((coalesce(old_fpa_cell_no, '')) <> coalesce(fpa_cell_no, '')) then
      begin
        changes_str = changes_str || 'Номера телефонов: ' || fpa_cell_no;
      end

      if (coalesce(old_fpa_email, '') <> coalesce(fpa_email, '')) then
      begin
        changes_str = changes_str || 'Иная контактная информация: ';
      end
--------------------------------
      changes_str = trim(changes_str);

      fpa_date_str = replace(iif(extract(day from rep_date) < 10, '0', '') || extract(day from rep_date) || '.' || iif(extract(month from rep_date) < 10, '0', '') || extract(month from rep_date) || '.' || extract(year from rep_date), ' ', '');

      if (rep_date < '08.11.2018') then
        fpa_changes_body = 'Клиент отсутствует в Перечне Росфинмониторинга террористов/экстремистов и в Перечне лиц, в отношении которых приняты решения о применении мер по замораживанию/блокированию денежных средств и иного имущества по состоянию на ' || :fpa_date_str || '. Оценка (степень) уровня риска клиента';
      else
        fpa_changes_body = 'Клиент отсутствует в Перечне Росфинмониторинга террористов/экстремистов, в Перечне лиц, в отношении которых приняты решения о применении мер по замораживанию/блокированию денежных средств и иного имущества и Перечне ОНН по состоянию на ' || :fpa_date_str || '. Оценка (степень) уровня риска клиента';

      if (coalesce(changes_str, '') = '') then
        fpa_changes = fpa_changes_body || ' и остальная информация о клиенте не изменилась.';
      else
      begin
        fpa_changes_body = fpa_changes_body || ' не изменилась.';
        fpa_changes = left(fpa_changes_body || ' Информация изменилась: ' || changes_str, 1000);
      end
      fpa_changes = replace(fpa_changes, ';;', ';');
    end
    if ((upd_only_change_str = 0) or (is_insert = 1)) then
    begin
      update or insert into fp_anketa_cl (fpa_id, fpa_c_id, fpa_from_un_id, fpa_city, fpa_citizen, fpa_client_fio,
                                          fpa_date_place_birth, fpa_pass_kind_id, fpa_pass_props, fpa_mcard_props,
                                          fpa_acard_props, fpa_inn, fpa_address_legal, fpa_rel_start, fpa_resp_c_id,
                                          fpa_type, fpa_date, fpa_date_check, fpa_cell_no, fpa_email, fpa_changes,
                                          fpa_snils)
      values (:fpa_id, :fpa_c_id, :rep_un_id, :fpa_city, :fpa_citizen, :fpa_client_fio, :fpa_date_place_birth,
              :fpa_pass_kind, :fpa_pass_props, :fpa_mcard_props, :fpa_acard_props, :fpa_inn, :fpa_address_legal,
              :fpa_rel_start, :fpa_ank_resp, :fpa_type, :rep_date, :fpa_date_check, :fpa_cell_no, :fpa_email,
              :fpa_changes, :fpa_snils)
      matching (fpa_id, fpa_c_id);
    end

    if ((upd_only_change_str = 1) and
        (is_insert = 0)) then
    begin
      update fp_anketa_cl fpa
      set fpa.fpa_changes = :fpa_changes
      where fpa.fpa_id = :fpa_id and
            fpa.fpa_c_id = :fpa_c_id;
    end
  end
end
