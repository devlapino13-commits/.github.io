CREATE PROCEDURE HP_NF_ANALIZ_NEW(
    NF_FNAME D_STR_100,
    NF_DATE D_DATE,
    NF_BLOB D_BLOB_BINARY,
    UN_ID D_ID)
RETURNS (
    IFA_ID D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable bls_id d_id;
begin
  err_text = '';

  execute procedure hp_insert_into_blobstore (
    :nf_fname,
    'NF-файл',
    :nf_blob,
    :un_id,
    2,
    0
  )
  returning_values :bls_id;

  insert into ipc_file_analiz(
    ifa_file_name,
    ifa_file_type,
    ifa_date,
    ifa_from_bls_id,
    ifa_from_un_id
  )
  values (
    :nf_fname,
    3,
    :nf_date,
    :bls_id,
    :un_id
  )
  returning ifa_id
  into :ifa_id;

  suspend;
end
