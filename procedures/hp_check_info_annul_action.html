CREATE PROCEDURE HP_CHECK_INFO_ANNUL_ACTION(
    OBJECT_KIND STR_100,
    OBJECT_ID ID_TYPE)
AS
declare variable ma_id integer;
declare variable c_id id_type;
declare variable acl_id id_type;
declare variable rd_id_check_info integer;
declare variable rd_id_check_info_err integer;
begin
  if (object_kind = 'kontract') then
  begin
    select kont.k_from_ma_app_id
    from kontract kont
    where kont.k_id = :object_id
    into :ma_id;
  end
  else
    ma_id = :object_id;

  if (ma_id is null) then
    exit;

  rd_id_check_info = f_get_rd_id('ActionGroup.Check_Info');
  rd_id_check_info_err = f_get_rd_id('ActionGroup.Check_Info_Error');

  for select macl.mac_from_c_id
      from miniappclient macl
      where macl.mac_from_ma_id = :ma_id and
            macl.mac_amount > 0
      into :c_id
  do
  begin
    acl_id = null;

    for select ao.aco_id
    from actionorder ao
    where ao.aco_status = 0 and
          ao.aco_object_kind = 0 and
          ao.aco_object_id = :c_id and
          ao.aco_from_rd_id in (:rd_id_check_info, :rd_id_check_info_err) and
          ao.aco_respons_c_id is null
    into :acl_id
    do
    begin
      acl_id = coalesce(acl_id, 0);
  
      if (acl_id <> 0) then
      begin
        update actionlist al
        set al.acl_status = -3
        where al.acl_id = :acl_id;
      end
    end
  end
end
