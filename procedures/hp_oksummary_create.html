CREATE PROCEDURE HP_OKSUMMARY_CREATE(
    K_ID D_ID NOT NULL)
AS
declare variable ol_date d_date;
declare variable date_first d_date;
declare variable date_last d_date;
declare variable kf_id d_id;
declare variable ks_date_begin d_date;
declare variable ks_date_end d_date;
begin
  -- Создает полную историю данных в таблице OKSUMMARY.

  delete from oksummary
  where oks_from_k_id = :k_id
  order by oks_from_k_id desc, oks_date_begin desc;

  select ks_date_begin, ks_date_end
  from kontstate
  where ks_from_k_id = :k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0
  into ks_date_begin, ks_date_end;

  if (ks_date_begin is null) then
    exit;

  if (ks_date_end < '31.12.2999') then
    ks_date_end = ks_date_end + 1;

  for select distinct ok_ol_date
  from operkred
  where ok_from_k_id = :k_id and
        not(ok_type = 9 and ok_ol_date > :ks_date_end) and
        ok_ol_is_active = 1
  union
  select ok_sh_date
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1
  union
  select oko_ol_date
  from operkroff
  where oko_from_k_id = :k_id and
        oko_ol_is_active = 1
  union
  select ox_ol_date
  from operextra
  where ox_object_kind = 9 and
        ox_object_id = :k_id and
        ox_ol_is_active = 1
  union
  select or_ol_date
  from operrest
  where or_from_k_id = :k_id and
        or_ol_is_active = 1
  union
  select kd_ol_date
  from kontdata
  where kd_from_k_id = :k_id and
        kd_ol_is_delete = 0
  union
  select ks_date_begin
  from kontstate
  where ks_from_k_id = :k_id and
        ks_ol_is_delete = 0 and
        ks_state = 2
  into ol_date
  do
  begin
    if (ol_date is null) then
      continue;

    if (date_first is null or date_first > ol_date) then
      date_first = ol_date;

    insert into oksummary (oks_from_k_id, oks_date_begin, oks_need_recalc)
    values (:k_id, :ol_date, 1)
    returning iif(oks_close_state = 2, oks_date_begin, oks_date_end)
    into date_last;
  end

  select k_from_kf_id
  from kontract
  where k_id = :k_id
  into kf_id;

  for select kal_date
  from knaktlist a
  where a.kal_from_kf_id = :kf_id and
        a.is_delete = 0 and
        a.kal_date between :date_first and :date_last
  into ol_date
  do
  begin
    if (ol_date is null) then
      continue;

    if (not exists(select * from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin = :ol_date)) then
    begin
      insert into oksummary (oks_from_k_id, oks_date_begin, oks_need_recalc)
      values (:k_id, :ol_date, 1);
    end
  end
end
