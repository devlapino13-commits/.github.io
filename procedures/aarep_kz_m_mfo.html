CREATE PROCEDURE AAREP_KZ_M_MFO(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    OD_SOMNIT_ALL D_CURRENCY,
    PERC_SOMNUT_ALL D_CURRENCY,
    PROVIZ_SOMNIT_ALL D_CURRENCY,
    OD_SOMNIT_1 D_CURRENCY,
    PERC_SOMNUT_1 D_CURRENCY,
    PROVIZ_SOMNIT_1 D_CURRENCY,
    OD_SOMNIT_2 D_CURRENCY,
    PERC_SOMNUT_2 D_CURRENCY,
    PROVIZ_SOMNIT_2 D_CURRENCY,
    OD_SOMNIT_3 D_CURRENCY,
    PERC_SOMNUT_3 D_CURRENCY,
    PROVIZ_SOMNIT_3 D_CURRENCY,
    OD_SOMNIT_4 D_CURRENCY,
    PERC_SOMNUT_4 D_CURRENCY,
    PROVIZ_SOMNIT_4 D_CURRENCY,
    OD_SOMNIT_5 D_CURRENCY,
    PERC_SOMNUT_5 D_CURRENCY,
    PROVIZ_SOMNIT_5 D_CURRENCY,
    OD_SOMNIT_6 D_CURRENCY,
    PERC_SOMNUT_6 D_CURRENCY,
    PROVIZ_SOMNIT_6 D_CURRENCY,
    PERIOD_STR D_STR_100,
    VBS_END_ALL D_CURRENCY,
    VBS_END_1 D_CURRENCY,
    VBS_END_2 D_CURRENCY,
    VBS_END_3 D_CURRENCY,
    VBS_END_4 D_CURRENCY,
    VBS_END_5 D_CURRENCY,
    VBS_END_6 D_CURRENCY)
AS
declare variable OKS_BALANCE D_CURRENCY;
declare variable OKS_BAL_PERC D_CURRENCY;
declare variable OKS_ECL_VBS D_CURRENCY;
declare variable PERC_PROVIZ D_PERCENT;
declare variable OKS_VBS_END D_CURRENCY;
begin
  period_str = 'Отчетный период: c '||f_datetostr(:rep_date_begin)||' по '||f_datetostr(:rep_date_end);
  od_somnit_all = 0;
  perc_somnut_all = 0;
  proviz_somnit_all = 0;

  od_somnit_1 = 0;
  perc_somnut_1 = 0;
  proviz_somnit_1 = 0;

  od_somnit_2 = 0;
  perc_somnut_2 = 0;
  proviz_somnit_2 = 0;

  od_somnit_3 = 0;
  perc_somnut_3 = 0;
  proviz_somnit_3 = 0;

  od_somnit_4 = 0;
  perc_somnut_4 = 0;
  proviz_somnit_4 = 0;

  od_somnit_5 = 0;
  perc_somnut_5 = 0;
  proviz_somnit_5 = 0;

  od_somnit_6 = 0;
  perc_somnut_6 = 0;
  proviz_somnit_6 = 0;

  vbs_end_all = 0;
  vbs_end_1 = 0;
  vbs_end_2 = 0;
  vbs_end_3 = 0;
  vbs_end_4 = 0;
  vbs_end_5 = 0;
  vbs_end_6 = 0;

  for
  select oks_balance, oks_bal_perc, oks_ecl_vbs, perc_proviz, oks_vbs_end
  from aarep_kn_oksummary_rep(:rep_date_begin, :rep_date_end)
  into oks_balance, oks_bal_perc, oks_ecl_vbs, perc_proviz, oks_vbs_end
  do
  begin
    oks_balance = oks_balance / 1000;
    oks_bal_perc = oks_bal_perc / 1000;
    oks_ecl_vbs = oks_ecl_vbs / 1000;

    od_somnit_all = od_somnit_all + oks_balance;
    perc_somnut_all = perc_somnut_all + oks_bal_perc;
    proviz_somnit_all = proviz_somnit_all + oks_ecl_vbs;
    vbs_end_all = vbs_end_all + oks_vbs_end;

    if (perc_proviz between 0.01 and 5) then
    begin
      od_somnit_1 = od_somnit_1 + oks_balance;
      perc_somnut_1 = perc_somnut_1 + oks_bal_perc;
      proviz_somnit_1 = proviz_somnit_1 + oks_ecl_vbs;
      vbs_end_1 = vbs_end_1 + oks_vbs_end;
    end else
    if (perc_proviz between 5.01 and 10) then
    begin
      od_somnit_2 = od_somnit_2 + oks_balance;
      perc_somnut_2 = perc_somnut_2 + oks_bal_perc;
      proviz_somnit_2 = proviz_somnit_2 + oks_ecl_vbs;
      vbs_end_2 = vbs_end_2 + oks_vbs_end;
    end else
    if (perc_proviz between 10.01 and 20) then
    begin
      od_somnit_3 = od_somnit_3 + oks_balance;
      perc_somnut_3 = perc_somnut_3 + oks_bal_perc;
      proviz_somnit_3 = proviz_somnit_3 + oks_ecl_vbs;
      vbs_end_3 = vbs_end_3 + oks_vbs_end;
    end else
    if (perc_proviz between 20.01 and 25) then
    begin
      od_somnit_4 = od_somnit_4 + oks_balance;
      perc_somnut_4 = perc_somnut_4 + oks_bal_perc;
      proviz_somnit_4 = proviz_somnit_4 + oks_ecl_vbs;
      vbs_end_4 = vbs_end_4 + oks_vbs_end;
    end else
    if (perc_proviz between 25.01 and 50) then
    begin
      od_somnit_5 = od_somnit_5 + oks_balance;
      perc_somnut_5 = perc_somnut_5 + oks_bal_perc;
      proviz_somnit_5 = proviz_somnit_5 + oks_ecl_vbs;
      vbs_end_5 = vbs_end_5 + oks_vbs_end;
    end else
    if (perc_proviz between 50.01 and 100) then
    begin
      od_somnit_6 = od_somnit_6 + oks_balance;
      perc_somnut_6 = perc_somnut_6 + oks_bal_perc;
      proviz_somnit_6 = proviz_somnit_6 + oks_ecl_vbs;
      vbs_end_6 = vbs_end_6 + oks_vbs_end;
    end
  end
  suspend;
end
