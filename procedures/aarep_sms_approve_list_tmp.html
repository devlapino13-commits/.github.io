CREATE PROCEDURE AAREP_SMS_APPROVE_LIST_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REP_K_ID DECIMAL(18,0),
    REGION_ID INTEGER,
    REGION_CODE VARCHAR(20),
    REGION_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    OKRUG_NAME VARCHAR(100),
    SV_VERIFIED_STR VARCHAR(100),
    SV_IS_VERIFIED INTEGER,
    KC_FROM_C_ID ID_TYPE,
    KC_AMOUNT_APPROVED CURR_TYPE,
    SV_ERRORS VARCHAR(255),
    SV_IN_MINUTES INTEGER,
    SV_COUNT INTEGER,
    SV_ID INTEGER)
AS
declare variable region_id_prior integer;
declare variable rd_id integer;
begin
  region_id_prior = 0;
  select rd_id
  from refdata
  where rd_code = 'contactinfo.cell_main'
  into :rd_id;

  sv_in_minutes = 0;
  sv_count = 0;
  sv_id = 0;

  for select ks_from_k_id, reg_unicode, reg_code
      from kontstate, region, region_list(:rep_region_id, 1, -1, 1)
      where ks_region_id = reg_unicode and
            reg_need_sms_approve = 1 and
            ks_state = 1 and
            reg_unicode = id_list and
            ks_ol_is_delete = 0 and
            ks_date_begin between :rep_date_begin and :rep_date_end
      into :rep_k_id, :region_id, region_code
  do
  begin
    if (region_id <> region_id_prior) then
    begin
      select aa_curr_office_name, aa_curr_level2_name, aa_curr_level3_name
      from aarep_get_office_categ(:region_id, :rep_date_end)
      into :region_name, branch_name, okrug_name;
      region_id_prior = :region_id;
    end
    for select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :rep_k_id and
              kc_amount_approved > 0
        into :kc_from_c_id, kc_amount_approved
    do
    begin
      sv_verified_str = '';
      sv_is_verified = 0;
      select res_value_status, res_error_text
      from aarep_object_one_continfo('0', :kc_from_c_id, :rd_id, 0, 0)
      into :sv_verified_str, sv_errors;
      suspend;
    end
  end
end
