CREATE PROCEDURE PRINT_PAYMENT_MONITORING_INFO(
    OBJ_ID D_ID)
RETURNS (
    REP_DATE D_DATE,
    KFI_NAME D_STR_100,
    KFI_REGION_NAME D_STR_64,
    KFI_OFFICER_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED DECIMAL(18,2),
    K_BALANCE DECIMAL(18,2),
    K_PAYED_FINE DECIMAL(18,2),
    K_CLOSE_DATE D_DATE,
    K_CIRCLE_TEXT D_STR_10,
    K_KOLICH_CLIENT_APPROVED D_INTEGER,
    K_PERIOD DECIMAL(15,2),
    K_KOLICH_REPAY D_INTEGER,
    K_PERCENT D_PERCENT,
    K_PERCENT_INPUT D_PERCENT,
    PR_NAME D_STR_64,
    PR_MEMO D_STR_255,
    PR_LOAN_TYPE_TEXT D_STR_64,
    PR_IS_SECURED_TEXT D_STR_64,
    K_DELINQ_PAY7 D_INTEGER,
    K_DELINQ_PAY30 D_INTEGER,
    K_DELINQ_PAY60 D_INTEGER,
    K_DELINQ_PAY_MORE D_INTEGER,
    K_DELINQ_CURR_DAYS D_INTEGER,
    SBL_NAME D_STR_100,
    REP_COMPANY_NAME D_STR_255,
    K_STATUS D_INTEGER,
    K_CURR_BONUS_STR D_STR_CODE,
    K_LAST_BONUS_STR D_STR_CODE,
    KS_STATE_TEXT D_STR_CODE)
AS
begin
  rep_date = current_date;

  select t.kfi_name, t.kfi_region_name, t.kfi_officer_name, t.k_kontract_no, t.k_disb_date, t.k_amount_approved, t.k_balance,
         t.k_payed_fine, t.k_close_date, t.k_circle_text, t.k_kolich_client_approved, t.k_period, t.k_kolich_repay, t.k_percent,
         t.k_percent_input, t.pr_name, t.pr_memo, t.pr_loan_type_text, t.pr_is_secured_text,
         t.k_delinq_Pay7, t.k_delinq_Pay30, t.k_delinq_Pay60, t.k_delinq_pay_more, t.k_delinq_curr_days, t.sbl_name,
         t.k_status, t.k_curr_bonus_str, t.k_last_bonus_str, t.ks_state_text
  from kontract_one_full_info(:obj_id, :rep_date) t
  into kfi_name, kfi_region_name, kfi_officer_name, k_kontract_no, k_disb_date, k_amount_approved, k_balance,
       k_payed_fine, k_close_date, k_circle_text, k_kolich_client_approved, k_period, k_kolich_repay, k_percent,
       k_percent_input, pr_name, pr_memo, pr_loan_type_text, pr_is_secured_text,
       k_delinq_Pay7, k_delinq_Pay30, k_delinq_Pay60, k_delinq_pay_more, k_delinq_curr_days, sbl_name, k_status,
       k_curr_bonus_str, k_last_bonus_str, ks_state_text;

  select cm.cm_name
  from company cm
  into rep_company_name;

  suspend;
end
