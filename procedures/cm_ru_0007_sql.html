CREATE PROCEDURE CM_RU_0007_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable c_id id_type;
declare variable kc_circle integer;
declare variable last_close_date date;
declare variable sum_lgota curr_type;
declare variable c_plastcard_no varchar(20);
declare variable c_cell_no varchar(20);
declare variable sds_create_date date;
declare variable kc_kommis_lgota curr_type;
declare variable is_lgota smallint;
begin
  is_lgota = 0;
  for
  select kc_from_c_id,
         kc_circle,
         k_disb_date,
--         c_plastcard_no,
         c_name_fio,
         c_pass_no,
         kc_kommis_lgota
  from knclient, kontract, clients
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  and kc_kommis_lgota > 0
  and kc_from_k_id = k_id
  and kc_from_c_id = c_id
  into c_id,
       kc_circle,
       kf_disb_date,
--       c_plastcard_no,
       res_info_text,
       kf_client_pass_no,
       kc_kommis_lgota
  do
  begin
    c_plastcard_no = null;
    select c_plastcard_no
    from AAREP_GET_CONTINFO('clients', :c_id)
    into c_plastcard_no;
    if (kc_circle > 1) then
    begin
--    вычисляется последний цикл клиента
      last_close_date = null;
      select ks_date_end
      from kontstate, knclient
      where ks_ol_is_delete = 0
      and ks_state = 1
      and ks_from_k_id = kc_from_k_id
      and kc_from_c_id = :c_id
      and kc_amount_approved > 0
      and ks_date_begin < current_date
      order by ks_date_begin desc
      rows 1
      into last_close_date;
      if (last_close_date is null) then last_close_date = '01.01.1900';
      if (last_close_date > dateadd(month,-6,:kf_disb_date)) then
      begin
        res_error_text = 'Клиент повторный, не подлежит к льготе!';
        res_error_is_critical = 1;
        suspend;
      end
    end
    sum_lgota = null;
    select 200 from smsdialogsend
    where (sds_text starting 'Ваш купон активирован'
           or sds_text starting 'Поздравляем! Вы получили скидку в 200 руб. в Мол Булак!')
    and sds_action = 'Kupon_register'
    and sds_cell_no = :c_plastcard_no -- c_cell_no - сотовый номер клиента
    and cast(sds_create_datetime as date) <= :kf_disb_date
    rows 1
    into sum_lgota;
    if (sum_lgota is null) then sum_lgota = 0;
    if (sum_lgota = 0.0000) then
    begin
        for
        select SDS_CELL_NO, sds_create_datetime
        from smsdialogsend
        where (sds_text starting 'Ваш купон активирован'
               or sds_text starting 'Поздравляем! Вы получили скидку в 200 руб. в Мол Булак!')
        and sds_action='Kupon_register'
        and sds_from_c_id = :c_id
        and cast(sds_create_datetime as date) <= :kf_disb_date
        rows 1
        into c_cell_no, sds_create_date
        do
        begin
          -- проверка выданных кредитов по этим номерам
          -- за каждый номер +300
--                Пока нет учета, и если кредит даже закрыт то за данный кредит будет считаться льготой, даже данную льготу получил в денежнем эквиваленте агент
          if (exists(select * from clients, knclient, kontstate
                     where c_plastcard_no = :c_cell_no
                     and c_id = kc_from_c_id
                     and kc_amount_approved > 0
                     and kc_from_k_id = ks_from_k_id
                     and ks_ol_is_delete = 0
                     and ks_state = 1
                     and ks_date_begin >= :sds_create_date -- кредит должен быть выдан после получения купона
                     )) then
          begin
            sum_lgota = sum_lgota + 300;
          end
        end
    end
    if (KC_KOMMIS_LGOTA > 200) then
    begin
      res_error_is_critical=1;
      res_error_text='Скидка по под. проценту не должно превышать 200р.';
      suspend;
    end
    if (is_lgota = 0) then is_lgota = 1; -- это озночает что комиссия проставлено хотябы у одного клиента
    if (sum_lgota > 0) then is_lgota = 2; -- это озночает, что хотябы у одного клиента имеется купон
  end
  if (is_lgota = 1) then
  begin
    res_error_is_critical=1;
    res_error_text='Клиенты не подлежат к льготе!';
    suspend;
  end
end
