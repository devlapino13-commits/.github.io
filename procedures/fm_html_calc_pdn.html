CREATE PROCEDURE FM_HTML_CALC_PDN(
    K_ID DECIMAL(18,0),
    DATE_END D_DATE,
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable STR varchar(1024);
begin
  res_html = (select res_style
              from fm_html_styles) || '<body>';
  res_html = blob_append(res_html, f_format('<h3>%1: %2</h3>', 'ТАБЛИЦА', 'Расчет ПДН'));

  res_html = blob_append(res_html, f_format('<table width="100%">' ||
  '<tr><th>#</th><th>Присутствие займа в КИ / ССП</th><th>UID</th><th>Дата выдачи</th><th>Срок договора</th><th>Статус</th><th>ПСК, %</th><th>Дата получеения истории БКИ</th>'||
  '<th>Дата расчета ССП</th><th>Дата информации КИ</th><th>Дата расчета ПДН</th><th>Среднемесячный платеж ФИНАЛ</th><th>ПДН</th>'||
  '<th>Среднемесячный доход клиента</th><th>Среднемесячный доход согласно Анкете-заявке</th><th>Среднедушевой доход по Росстату</th>' ||
  '<th>Адрес клиента (используется для определения субьекта РФ по стат</th></tr>'));

   for select '<tr><td>' || coalesce(line, '') || '</td>' || 
                    '<td>' || coalesce(tag_str, '') || '</td>' || 
                    '<td>' || coalesce(uid, '') || '</td>' ||  -- <--- добавила UID RFCRU-4136
                    '<td>' || f_datetostr(creddate) || '</td>' || 
                    '<td>' || f_datetostr(credenddate) || '</td>' || 
                    '<td>' || coalesce(credactive_str, '') || '</td>' || 
                    '<td>' || coalesce(credfullcost, '') || '</td>' ||   -- <--- добавила ПСК RFCRU-4136
                    '<td>' || f_datetostr(calc_eil_date) || '</td>' || 
                    '<td>' || f_datetostr(amp_update) || '</td>' || 
                    '<td>' || f_datetostr(credupdate) || '</td>' || 
                    '<td>' || f_datetostr(calc_date) || '</td>' ||
                    '<td>' || coalesce(calc_month_repay_final, '') || '</td>' || 
                    '<td>' || coalesce(final_pdn, '') || '</td>' || 
                    '<td>' || coalesce(kc_final_income, '') || '</td>' ||
                    '<td>' || coalesce(kc_anketa_income, '') || '</td>' || 
                    '<td>' || coalesce(kc_stat_avg_income, '') || '</td>' || 
                    '<td>' || coalesce(adl_address_str, '') || '</td></tr>'

      from aarep_knclient_pdnlist2(null, null, 0, :k_id, 0)
      where calc_month_repay_final > 0
      into :str
  do
  begin
    res_html = blob_append(res_html, str);
  end

  res_html = blob_append(res_html, '</table></body>');

  suspend;
end
