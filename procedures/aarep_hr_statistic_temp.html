CREATE PROCEDURE AAREP_HR_STATISTIC_TEMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    JOB_TITLE VARCHAR(50),
    IS_HIRING INTEGER,
    IS_DISMISSED INTEGER,
    IS_TRANSFER INTEGER,
    IS_REBUKE INTEGER,
    IS_VACATION INTEGER,
    KOL_TRAININGS INTEGER)
AS
declare variable jk_system_code varchar(100);
declare variable reg_id integer;
declare variable c_id decimal(18,0);
declare variable rb_date date;
declare variable id_list integer;
begin
  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into id_list
  do
  begin
    is_hiring = 0;
    is_dismissed = 0;
    is_transfer = 0;
    is_rebuke = 0;
    is_vacation = 0;
    kol_trainings = 0;
-- Найм и расторжение
    for select ps_from_c_id, jk_system_code, ps_from_reg_id, iif(ho_order_type = 1, 1, 0), -- найм
               iif(ho_order_type = 6, 1, 0), -- расторжение
               iif(ho_order_type in (2, 3), 1, 0) --Ротация и изменение условий договора
        from hr_pstatus, joblist, jobkind, hr_orders
        where ps_date_begin between :rep_date_begin and :rep_date_end and
              ps_status = 1
--  and ps_is_working = 1
              and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              ps_from_ho_id = ho_id and
              ps_from_reg_id = :id_list
        into c_id, jk_system_code, reg_id, is_hiring, is_dismissed, is_transfer
    do
    begin
      job_title = case
                    when jk_system_code = 'JobSystem.LO_Standart' then '1. СПФ'
                    when jk_system_code = 'JobSystem.LO_Senior' then '2. CСПФ'
                    when jk_system_code = 'JobSystem.Specialist_KOBD' then '3. КОБД'
                    when jk_system_code = 'JobSystem.Region_KM' then '4. МО'
                    when reg_id <> 1090100 then '5. Прочие должности БРУ'
                    else '6. Должности ГО'
                  end;
      suspend;
    end

    is_hiring = 0;
    is_dismissed = 0;
    is_transfer = 0;
    is_rebuke = 0;
    is_vacation = 0;
    kol_trainings = 0;
--выговора и замечаний
    for select rb_date, rb_from_c_id
        from rebuke
        where rb_is_delete = 0 and
              rb_date between :rep_date_begin and :rep_date_end and
              rb_type <> 3 -- устное замечение не будет включаться
              and
              rb_from_reg_id = :id_list
        into rb_date, c_id
    do
    begin
      is_rebuke = 1;
      for select jk_system_code, ps_from_reg_id
          from hr_pstatus, joblist, jobkind
          where ps_from_c_id = :c_id and
                :rb_date between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id
          into jk_system_code, reg_id
      do
      begin
        job_title = case
                      when jk_system_code = 'JobSystem.LO_Standart' then '1. СПФ'
                      when jk_system_code = 'JobSystem.LO_Senior' then '2. CСПФ'
                      when jk_system_code = 'JobSystem.Specialist_KOBD' then '3. КОБД'
                      when jk_system_code = 'JobSystem.Region_KM' then '4. МО'
                      when reg_id <> 1090100 then '5. Прочие должности БРУ'
                      else '6. Должности ГО'
                    end;
        suspend;
      end
    end

    is_hiring = 0;
    is_dismissed = 0;
    is_transfer = 0;
    is_rebuke = 0;
    is_vacation = 0;
    kol_trainings = 0;
--  отпуска
    for select hrv_date_begin, hrv_from_c_id
        from hr_orders, hr_voctravel
        where ho_order_type between 7 and 20 -- все отпуска и выходы из отпуска, командировка и сверхурочная работа и работа в нерабочее время
              and
              ho_id = hrv_from_ho_id and
              hrv_date_begin between :rep_date_begin and :rep_date_end and
              ho_status = 1 and
              hrv_status = 1 and
              ho_from_reg_id = :id_list
        into rb_date, c_id
    do
    begin
      is_vacation = 1;
      for select jk_system_code, ps_from_reg_id
          from hr_pstatus, joblist, jobkind
          where ps_from_c_id = :c_id and
                :rb_date between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id
          into jk_system_code, reg_id
      do
      begin
        job_title = case
                      when jk_system_code = 'JobSystem.LO_Standart' then '1. СПФ'
                      when jk_system_code = 'JobSystem.LO_Senior' then '2. CСПФ'
                      when jk_system_code = 'JobSystem.Specialist_KOBD' then '3. КОБД'
                      when jk_system_code = 'JobSystem.Region_KM' then '4. МО'
                      when reg_id <> 1090100 then '5. Прочие должности БРУ'
                      else '6. Должности ГО'
                    end;
        suspend;
      end
    end

    is_hiring = 0;
    is_dismissed = 0;
    is_transfer = 0;
    is_rebuke = 0;
    is_vacation = 0;
    kol_trainings = 0;
-- кол-во проведенных тренингов
    for select count(*)
        from tranmain
        where tm_date_begin between :rep_date_begin and :rep_date_end and
              tm_subject is not null and
              tm_from_reg_id = :id_list
        into kol_trainings
    do
    begin
      job_title = '7. Проведенные тренинги';
      suspend;
    end
  end
end
