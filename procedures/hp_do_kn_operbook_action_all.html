CREATE PROCEDURE HP_DO_KN_OPERBOOK_ACTION_ALL(
    RECREATE_ALL D_SMALLINT)
AS
declare variable ol_date d_date;
declare variable max_date d_date;
declare variable curr_minute d_smallint;
begin
  if (recreate_all = 1) then
  begin
    execute statement('delete from kn_operbalbook where 1 = 1')
        with autonomous transaction;
  
    execute statement('delete from kn_balsummary where 1 = 1 order by kbs_date desc')
        with autonomous transaction;
  
    select kob_ol_date
    from kn_operbook
    order by kob_ol_date
    rows 1
    into ol_date;
  
    if (ol_date is null) then exit;
  
    max_date = current_date;
  
    select kob_ol_date
    from kn_operbook
    where kob_ol_date > current_date
    order by kob_ol_date desc
    rows 1
    into max_date;
  
    max_date = minvalue(dateadd(year, 1, current_date), max_date);
  
    while (ol_date < max_date) do
    begin
      execute statement('execute procedure hp_do_kn_operbalbook_fix_date(:ol_date, 0)')(ol_date := ol_date)
          with autonomous transaction;
  
      ol_date = dateadd(day, 1, ol_date);
    end
  end else
  begin
    curr_minute = extract(minute from current_timestamp);

    for select fix_date
    from kn_operbalbook_date
    order by fix_date
    into :ol_date
    do
    begin
      -- текущая дата будет обрабатываться раз в час
      -- другие даты каждые 3 минуты согласно расписания робота

      if (ol_date <> current_date or curr_minute <= 2) then
        in autonomous transaction do
          execute procedure hp_do_kn_operbalbook_fix_date(:ol_date, 0);
    end
  end
end
