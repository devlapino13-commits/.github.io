CREATE PROCEDURE AAREP_REGION_USER_LIST(
    UN_ID DECIMAL(18,0))
RETURNS (
    REG_UNICODE INTEGER,
    PARENT INTEGER,
    REG_STATUS SMALLINT,
    REG_CODE VARCHAR(10),
    REG_NAME VARCHAR(50),
    REG_COMPANY_NAME VARCHAR(255),
    REG_COMPANY_BOSS_TITUL VARCHAR(255),
    REG_COMPANY_BOSS_NAME VARCHAR(255),
    REG_COMPANY_ACCOUNT_TITUL VARCHAR(255),
    REG_COMPANY_ACCOUNT_NAME VARCHAR(255),
    REG_HEAD_OFFICE INTEGER,
    REG_REPAY_ON_WEEKEND INTEGER,
    REG_MOUNTAIN_KOEF DECIMAL(18,7),
    REG_IS_AGENT INTEGER,
    REG_LORONOSTRO_HEAD INTEGER,
    REG_ID INTEGER,
    CODE VARCHAR(10),
    IS_READ INTEGER,
    REG_TAG_STR STR_20)
AS
begin
  for select reg_list, max(is_read)
      from aarep_region_user_tmp(:un_id)
      group by reg_list
      into :reg_unicode, :is_read
  do
  begin
    if (is_read is null) then
      is_read = 0;

    for select reg_unicode, parent, reg_status, reg_code, reg_name, reg_company_name, reg_company_boss_titul,
               reg_company_boss_name, reg_company_account_titul, reg_company_account_name, reg_head_office,
               reg_repay_on_weekend, reg_is_agent, reg_loronostro_head, reg_tag_str
        from region
        where reg_unicode = :reg_unicode
        into :reg_unicode, :parent, :reg_status, :reg_code, :reg_name, :reg_company_name, :reg_company_boss_titul,
             :reg_company_boss_name, :reg_company_account_titul, :reg_company_account_name, :reg_head_office,
             :reg_repay_on_weekend, :reg_is_agent, :reg_loronostro_head, :reg_tag_str
    do
    begin
      -- RFCKG-4782
      reg_mountain_koef = f_reg_mount_koef(:reg_unicode);
      reg_id = reg_unicode;
      code = reg_code;
      suspend;
    end
  end
end
