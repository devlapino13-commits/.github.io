CREATE PROCEDURE CNP_AUTO_DISB
AS
declare variable k_id d_id;
declare variable c_id d_id;
declare variable manager_id d_id;
declare variable un_id d_id;
declare variable res_ol_ids d_id;
declare variable res_text d_str_255;
declare variable reg_id d_integer;
declare variable acc_kt d_str_20;
declare variable sql_text d_str_8100;
begin
  for select cpr.cpr_from_k_id
      from cnp_payment_requests cpr
      inner join kontstate ks on ks.ks_from_k_id = cpr.cpr_from_k_id and
            ks.ks_state = 0 and
            ks.ks_ol_is_delete = 0 and
            current_date between ks.ks_date_begin and ks.ks_date_end
      where cpr.cpr_status = 2 and
            cpr.cpr_action = 'pay'
      into k_id
  do
  begin
    select ks.ks_region_id
    from kontstate ks
    where ks.ks_from_k_id = :k_id and
          ks.ks_state = 0
    into reg_id;

    select kt.k_clients_cashier_id, kt.k_approved_manager
    from kontract kt
    where kt.k_id = :k_id
    into c_id, manager_id;

    --выдача "вешается" на сотрудника который утвердил займ
    select kl.kl_user_id
    from kontlog kl
    where kl.kl_from_k_id = :k_id and
          kl.kl_state = 1
    order by kl.kl_datetime desc rows 1
    into un_id;

    select al_id
    from acc_list
    where al_from_reg_id = 1010100 and
          al_currency = '398' and
          al_account = iif(f_get_disb_source(:k_id) = 'bc', '1021/2', '1021/3')
    into :acc_kt;

    sql_text = 'select res_ol_ids, res_text from hp_do_disbursement('||:un_id||', '||:k_id||', '||:reg_id||', 5, '''||:acc_kt||''', '||:reg_id||', 2, '''||:acc_kt||''', '||:c_id||', '||:manager_id||')';

    execute statement(:sql_text)
    with autonomous transaction
    into res_ol_ids, res_text;

    if (coalesce(res_ol_ids, '') = '') then
    begin
      execute procedure hp_send_quick_message('email', 'h.raimjanov@mbulak.kg',
          'ошибка при выдачи',
          'выдача не осуществилась. ошибка -' || :res_text, '', 7, 1000000000000);
    end
    when any do
      execute procedure hp_send_quick_message('email', 'h.raimjanov@mbulak.kg',
          'ошибка при выдачи',
          'выдача не осуществилась. техническая ошибка. gdscode = ' || gdscode,
          '', 7, 1000000000000);
  end
end
