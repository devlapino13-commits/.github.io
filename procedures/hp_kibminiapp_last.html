CREATE PROCEDURE HP_KIBMINIAPP_LAST(
    C_ID D_ID,
    MA_ID D_ID = null)
RETURNS (
    KBM_ID D_ID,
    KBM_FROM_C_ID D_ID,
    KBM_MA_FROM_REG_ID D_ID,
    KBM_MA_ID D_ID,
    KBM_PR_ID D_STR_255,
    KBM_IS_CHECKED D_BOOL,
    KBM_PR_ID_DATETIME D_DATETIME,
    KBM_CHECKED_DATETIME D_DATETIME,
    KBM_VER_INFO D_SMALLINT,
    KBM_ERR_TEXT D_STR_255,
    KBM_FROM_UN_ID D_ID,
    KBM_DATETIME D_DATETIME,
    KBM_IS_CANDIDATE D_INTEGER,
    KBM_SCORE_RATE D_INTEGER,
    KBM_FROM_RD_ID D_INTEGER)
AS
declare variable ma_date_app d_date;
declare variable kib_change_actualiti_date d_date;
begin
  -- Процедура-обертка для kibminiapp с
  -- возвращает последний запрос в КИБ по клиенту и заявке (RFCKG-2214)
  -- если заявка (ma_id) не передано, то запрос ищется только по клиенту, т.к. могут быть запросы в КИБ
  -- не в рамках заявки, например при онлайн займе клиент нажимает кнопку "Расчитать макс. сумму займа",
  -- при этом система может сделать запрос в КИБ
  -- поле kbm_ma_id не возвращается, т.к. это поле не испоьзуется с
  ma_date_app = null;

  select ma.ma_date_app from miniapp ma
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.9999'
  into ma_date_app;

  kib_change_actualiti_date = f_get_const('kib_change_actualiti_date');

  if (f_get_db_id() = 41701) then
  begin
    if (ma_date_app >= kib_change_actualiti_date) then
    begin
      select kbm_id,
             kbm_from_c_id,
             kbm_ma_from_reg_id,
             kbm_ma_id,
             kbm_pr_id,
             kbm_is_checked,
             kbm_pr_id_datetime,
             kbm_checked_datetime,
             kbm_ver_info,
             kbm_err_text,
             kbm_from_un_id,
             kbm_datetime,
             kbm_is_candidate,
             kbm_score_rate,
             kbm_from_rd_id
      from kibminiapp
      where kbm_from_c_id = :c_id and
      iif(kbm_is_checked = 1,
          cast(kbm_checked_datetime as d_date) between :ma_date_app - 1 and :ma_date_app,
          cast(kbm_datetime as d_date) between :ma_date_app - 1 and :ma_date_app)
      order by kbm_datetime desc
      rows 1
      into kbm_id,
           kbm_from_c_id,
           kbm_ma_from_reg_id,
           kbm_ma_id,
           kbm_pr_id,
           kbm_is_checked,
           kbm_pr_id_datetime,
           kbm_checked_datetime,
           kbm_ver_info,
           kbm_err_text,
           kbm_from_un_id,
           kbm_datetime,
           kbm_is_candidate,
           kbm_score_rate,
           kbm_from_rd_id;
    end
    else
    begin      -- RFCKG-5294
      select kbm_id,
             kbm_from_c_id,
             kbm_ma_from_reg_id,
             kbm_ma_id,
             kbm_pr_id,
             kbm_is_checked,
             kbm_pr_id_datetime,
             kbm_checked_datetime,
             kbm_ver_info,
             kbm_err_text,
             kbm_from_un_id,
             kbm_datetime,
             kbm_is_candidate,
             kbm_score_rate,
             kbm_from_rd_id
      from kibminiapp
      inner join kibminiapp_link on kbl_from_kbm_id = kbm_id
      where kbl_object_kind = f_objkind('miniapp', 'id') and
            kbl_object_id = :ma_id and
            kbm_from_c_id = :c_id and
            kbm_is_checked = 1
      order by kbm_datetime desc
      rows 1
      into kbm_id,
           kbm_from_c_id,
           kbm_ma_from_reg_id,
           kbm_ma_id,
           kbm_pr_id,
           kbm_is_checked,
           kbm_pr_id_datetime,
           kbm_checked_datetime,
           kbm_ver_info,
           kbm_err_text,
           kbm_from_un_id,
           kbm_datetime,
           kbm_is_candidate,
           kbm_score_rate,
           kbm_from_rd_id;
    end
  end
  else
  if (ma_id is null) then
    select kbm_id,
           kbm_from_c_id,
           kbm_ma_from_reg_id,
           kbm_ma_id,
           kbm_pr_id,
           kbm_is_checked,
           kbm_pr_id_datetime,
           kbm_checked_datetime,
           kbm_ver_info,
           kbm_err_text,
           kbm_from_un_id,
           kbm_datetime,
           kbm_is_candidate,
           kbm_score_rate,
           kbm_from_rd_id
    from kibminiapp
    where kbm_from_c_id = :c_id
    order by kbm_datetime desc
    rows 1
    into kbm_id,
         kbm_from_c_id,
         kbm_ma_from_reg_id,
         kbm_ma_id,
         kbm_pr_id,
         kbm_is_checked,
         kbm_pr_id_datetime,
         kbm_checked_datetime,
         kbm_ver_info,
         kbm_err_text,
         kbm_from_un_id,
         kbm_datetime,
         kbm_is_candidate,
         kbm_score_rate,
         kbm_from_rd_id;
  else
    select kbm_id,
           kbm_from_c_id,
           kbm_ma_from_reg_id,
           kbm_ma_id,
           kbm_pr_id,
           kbm_is_checked,
           kbm_pr_id_datetime,
           kbm_checked_datetime,
           kbm_ver_info,
           kbm_err_text,
           kbm_from_un_id,
           kbm_datetime,
           kbm_is_candidate,
           kbm_score_rate,
           kbm_from_rd_id
    from kibminiapp
    inner join kibminiapp_link on kbl_from_kbm_id = kbm_id
    where kbl_object_kind = f_objkind('miniapp', 'id') and
          kbl_object_id = :ma_id and
          kbm_from_c_id = :c_id
    order by kbm_datetime desc
    rows 1
    into kbm_id,
         kbm_from_c_id,
         kbm_ma_from_reg_id,
         kbm_ma_id,
         kbm_pr_id,
         kbm_is_checked,
         kbm_pr_id_datetime,
         kbm_checked_datetime,
         kbm_ver_info,
         kbm_err_text,
         kbm_from_un_id,
         kbm_datetime,
         kbm_is_candidate,
         kbm_score_rate,
         kbm_from_rd_id;

  suspend;
end
