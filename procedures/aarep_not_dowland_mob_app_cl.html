CREATE PROCEDURE AAREP_NOT_DOWLAND_MOB_APP_CL(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    N D_INTEGER,
    C_NAME_FIO D_STR_100,
    C_NAME_FIO_CL D_STR_100,
    KAL_NEW_KF_NAME VARCHAR(50),
    DELIQ_DAYS_STR D_STR_20)
AS
declare variable lo_id decimal(18,0);
declare variable k_id decimal(18,0);
declare variable c_id decimal(18,0);
declare variable deliq_days d_integer;
declare variable reg_id d_id;
begin
  n = 0;
  for select oks.oks_from_k_id, oks.oks_kc_id,
             iif((cast(:report_date_end as date) - oks_delinq_date <= 0), 0, cast(:report_date_end as date) - oks_delinq_date),
             knc.kc_from_c_id
      from oksummary oks, knclient knc, region_list(:report_region_id, 1, -1, 1)
      where knc.kc_from_k_id = oks.oks_from_k_id and
            id_list = oks.oks_region_id and
            :report_date_end between oks.oks_date_begin and oks.oks_date_end and
            knc.kc_from_k_id = oks.oks_from_k_id and
            knc.kc_amount_approved > 0 and
            f_fsm_client_status(kc_from_c_id) = 0
            and oks_close_state = 0
      order by 3, 2
      into :k_id, :lo_id, :deliq_days, :c_id
  do
  begin
    deliq_days_str = '';
    kal_new_kf_name = '';
    c_name_fio_cl = '';
    n = n + 1;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into :c_name_fio_cl;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :lo_id
    into :c_name_fio;

    select kal.kal_new_kf_name
    from knaktlist kal
    inner join kontract k on k.k_from_kf_id = kal.kal_from_kf_id
    where k.k_id = :k_id and
          :report_date_end between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into :kal_new_kf_name;
    deliq_days_str = iif(deliq_days <= 30, 'до 30 дней (вкл.)', 'более 30 дней');
    suspend;
  end

end
