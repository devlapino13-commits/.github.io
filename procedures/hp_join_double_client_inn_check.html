CREATE PROCEDURE HP_JOIN_DOUBLE_CLIENT_INN_CHECK(
    C_STAY D_ID,
    C_REMOVE D_ID)
AS
declare variable ICM_FROM_ICQ_ID D_ID;
declare variable ICM_INN D_STR_20;
begin
  -- RFCRU-3893
  -- Процедура переносит запросы ИНН удаляемого клиента на нового клиента
  update inn_check_queries
  set icq_from_c_id = :c_stay
  where icq_from_c_id = :c_remove;

  if (not (exists (select *
                from inn_check_main icm
                where icm.icm_from_c_id = :c_stay
                and icm.icm_inn_received = 1))) then
  begin
    select icm.icm_from_icq_id, icm.icm_inn
    from inn_check_main icm
    where icm.icm_from_c_id = :c_remove
    and icm.icm_inn_received = 1
    into icm_from_icq_id, icm_inn;

    if (:icm_from_icq_id is not null and :icm_inn is not null) then
    begin
      update inn_check_main icm
      set icm.icm_from_icq_id = :icm_from_icq_id,
          icm.icm_inn = :icm_inn,
          icm.icm_inn_received = 1
      where icm_from_c_id = :c_stay;
    end
  end

  delete from inn_check_main
  where icm_from_c_id = :c_remove;
end
