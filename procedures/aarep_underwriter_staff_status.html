CREATE PROCEDURE AAREP_UNDERWRITER_STAFF_STATUS(
    UN_ID ID_TYPE)
RETURNS (
    RES_UN_ID DECIMAL(18,0),
    RES_UN_NAME VARCHAR(100),
    RES_UN_FIO VARCHAR(100),
    RES_UN_C_ID DECIMAL(18,0),
    RES_ONLINE_STATUS INTEGER,
    RES_ONLINE_TEXT VARCHAR(100),
    RES_DATETIME TIMESTAMP,
    RES_UNITS VARCHAR(255),
    RES_IN_PROCESS INTEGER,
    RES_NEW INTEGER,
    RES_APPROVED INTEGER,
    RES_REJECTED INTEGER,
    RES_REVISION INTEGER)
AS
declare variable db_id integer;
declare variable rd_id integer;
declare variable rd_filter varchar(255);
declare variable rd_str_id varchar(255);
declare variable is_supervisor integer;
declare variable ml_id integer;
declare variable counter integer;
declare variable scores integer;
declare variable rd_id_cld integer; /* ActionGroup.Check_Loan_Data */
begin
  db_id = f_get_db_id();
  scores = 0;

  rd_id = f_get_rd_id('ObjectProps.KK_Staff_Online');
  if (rd_id is null) then
    exit;

  rd_filter = '';
  for select distinct rd_str_id
      from refdata
      where rd_main_code = 'ActionGroup' and
            rd_code <> 'ActionGroup' and
            rd_is_active = 1
      into :rd_str_id
  do
  begin
    if (:db_id in (39801, 41701) and
        :rd_str_id = 'UnderWriter') then
      continue;
    if (exists(select has_access
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'ActionGroup.' || :rd_str_id)
               where has_access = 1)) then
    begin
      if (position(rd_str_id, rd_filter) <= 0) then
        rd_filter = rd_filter || '[' || rd_str_id || ']';
    end
  end

  is_supervisor = 0;
  if (exists(select has_access
             from hp_permission_access_check(:un_id, 'MenuList', 0, 'mSupervisor')
             where has_access = 1)) then
    is_supervisor = 1;
  if (is_supervisor = 0) then
  begin
    if (exists(select has_access
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Superuser')
               where has_access = 1)) then
      is_supervisor = 1;
  end

  select ml_id
  from menulist
  where ml_code = 'mUnderWriter'
  into :ml_id;

  for select un_id, un_name, un_fio, un_from_c_id
      from hp_user_role_list('[menu]' || :ml_id, iif(:is_supervisor = 1,0,:un_id))
      where un_is_active = 1
      union
      select un_id, un_name, un_fio, un_from_c_id
      from username
      where un_id = 1000000000001
      union
      select 0, ' All', ' Общая очередь', 0
      from rdb$database
      where :is_supervisor = 1
      into :res_un_id, res_un_name, res_un_fio, res_un_c_id
  do
  begin
    res_online_status = null;
    res_online_text = '';
    res_datetime = null;

    if (coalesce(res_un_c_id, 0) <> 0) then
    begin
      select ovs_add_value, ovs_datetime
      from hp_objectprops_get('clients', :res_un_c_id, -1, :rd_id, null)
      into :res_online_status, res_datetime;
      if (res_online_status is null) then
        res_online_status = 1;

      if (res_online_status = -1) then
        res_online_text = 'day off';
      if (res_online_status = 0) then
        res_online_text = 'pause';
      if (res_online_status = 1) then
        res_online_text = 'online';
    end

    for select distinct rd_str_id
        from refdata
        where rd_main_code = 'ActionGroup' and
              rd_code <> 'ActionGroup' and
              rd_is_active = 1 and
              :rd_filter like '%[' || rd_str_id || ']%'
        into :rd_str_id
    do
    begin
      res_units = '';
      if (res_un_id <> 0 and
          res_un_id <> 1000000000001) then
      begin
        if (exists(select has_access
                   from hp_permission_access_check(:res_un_id, 'PermActions', 0, 'ActionGroup.' || :rd_str_id)
                   where has_access = 1)) then
          res_units = rd_str_id;
      end
      else
        res_units = rd_str_id;

      if (res_units <> '') then
      begin
        res_in_process = 0;
        res_new = 0;
        res_approved = 0;
        res_rejected = 0;
        res_revision = 0;

        if (res_un_id = 0) then
        begin
          select count(*)
          from actionorder, refdata
          where aco_respons_c_id is null and
                aco_status = 0 and
                aco_from_rd_id = rd_id and
                rd_main_code = 'ActionGroup' and
                rd_str_id = :res_units and
                (:db_id <> 81001 or not exists(select *
                                               from actionmark
                                               where atm_from_acl_id = aco_id))
          into :res_new;
        end
        else
        begin
          select sum(iif(aco_status = 0, 1, 0)), sum(iif(aco_status = 1, 1, 0))
          from actionorder, refdata
          where aco_respons_c_id = :res_un_c_id and
                aco_from_rd_id = rd_id and
                rd_main_code = 'ActionGroup' and
                rd_str_id = :res_units and
                (:db_id <> 81001 or not exists(select *
                                               from actionmark
                                               where atm_from_acl_id = aco_id))
          into :res_new, :res_in_process;
          select sum(iif(acl_status = 2, 1, 0)), sum(iif(acl_status = -2, 1, 0)), sum(iif(acl_status = -1, 1, 0))
          from actionlist, refdata
          where acl_from_rd_id = rd_id and
                rd_main_code = 'ActionGroup' and
                rd_str_id = :res_units and
                acl_respons_c_id = :res_un_c_id and
                acl_dont_show_in_order = 0 and
                acl_last_datetime >= current_date and
                (:db_id <> 81001 or not exists(select *
                                               from actionmark
                                               where atm_from_acl_id = acl_id))
          into :res_approved, :res_rejected, :res_revision;
        end
        if (res_new is null) then
          res_new = 0;
        if (res_in_process is null) then
          res_in_process = 0;
        if (res_approved is null) then
          res_approved = 0;
        if (res_rejected is null) then
          res_rejected = 0;
        if (res_revision is null) then
          res_revision = 0;
        suspend;
      end
    end
  end

  if (:db_id in (39801, 41701)) then
  begin
    rd_str_id = 'UnderWriter';
    rd_filter = '';
    counter = 0;
    scores = 0;

    select rd.rd_orderno
    from refdata rd
    where rd.rd_code = 'PrioritySertif.ScoresLimit'
    into :scores;

    begin
      if (exists(select has_access
                 from hp_permission_access_check(:un_id, 'PermActions', 0, 'ActionGroup.UnderWriter')
                 where has_access = 1)) then
      begin
        if (position(rd_str_id, rd_filter) <= 0) then
          rd_filter = rd_filter || '[' || rd_str_id || ']';
      end
    end

    is_supervisor = 0;
    if (exists(select has_access
               from hp_permission_access_check(:un_id, 'MenuList', 0, 'mSupervisor')
               where has_access = 1)) then
      is_supervisor = 1;
    if (is_supervisor = 0) then
    begin
      if (exists(select has_access
                 from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Superuser')
                 where has_access = 1)) then
        is_supervisor = 1;
    end

    select ml_id
    from menulist
    where ml_code = 'mUnderWriter'
    into :ml_id;

    while (counter < 2) do
    begin
      for select un_id, un_name, un_fio, un_from_c_id
          from hp_user_role_list('[menu]' || :ml_id)
          where un_is_active = 1 and
                (un_id = :un_id or :is_supervisor = 1)
          union
          select un_id, un_name, un_fio, un_from_c_id
          from username
          where un_id = 1000000000001
          union
          select 0, ' All', iif(:counter = 0, ' Общая очередь', ' Приоритетная очередь'),
                 0
          from rdb$database
          where :is_supervisor = 1
          into :res_un_id, res_un_name, res_un_fio, res_un_c_id
      do
      begin
        res_online_status = null;
        res_online_text = '';
        res_datetime = null;

        if (coalesce(res_un_c_id, 0) <> 0) then
        begin
          select ovs_add_value, ovs_datetime
          from hp_objectprops_get('clients', :res_un_c_id, -1, :rd_id, null)
          into :res_online_status, res_datetime;
          if (res_online_status is null) then
            res_online_status = 1;

          if (res_online_status = -1) then
            res_online_text = 'day off';
          if (res_online_status = 0) then
            res_online_text = 'pause';
          if (res_online_status = 1) then
            res_online_text = 'online';
        end

        for select distinct rd_str_id
            from refdata
            where rd_main_code = 'ActionGroup' and
                  rd_code <> 'ActionGroup' and
                  rd_is_active = 1 and
                  :rd_filter like '%[' || rd_str_id || ']%'
            into :rd_str_id
        do
        begin
          res_units = '';
          if (res_un_id <> 0 and
              res_un_id <> 1000000000001) then
          begin
            if (exists(select has_access
                       from hp_permission_access_check(:res_un_id, 'PermActions', 0, 'ActionGroup.' || :rd_str_id)
                       where has_access = 1)) then
              res_units = rd_str_id;
          end
          else
            res_units = rd_str_id;

          if (res_units <> '') then
          begin
            if (counter > 0) then
              res_units = res_units || '.HighPriority';
            res_in_process = 0;
            res_new = 0;
            res_approved = 0;
            res_rejected = 0;
            res_revision = 0;

            if (res_un_id = 0) then
            begin
              if (counter = 0) then
              begin
                select sum(iif(f_point(k_clients_cashier_id, current_date) < :scores, 1, 0))
                from actionorder, refdata, kontract
                where aco_respons_c_id is null and
                      aco_status = 0 and
                      k_id = aco_object_id and
                      aco_from_rd_id = rd_id and
                      rd_main_code = 'ActionGroup' and
                      rd_str_id = :rd_str_id and
                      (:db_id <> 81001 or not exists(select *
                                                     from actionmark
                                                     where atm_from_acl_id = aco_id))
                group by aco_object_kind
                into :res_new;
              end
              if (counter = 1) then
              begin
                select sum(iif(f_point(k_clients_cashier_id, current_date) >= :scores, 1, 0))
                from actionorder, refdata, kontract
                where aco_respons_c_id is null and
                      aco_status = 0 and
                      k_id = aco_object_id and
                      aco_from_rd_id = rd_id and
                      rd_main_code = 'ActionGroup' and
                      rd_str_id = :rd_str_id and
                      (:db_id <> 81001 or not exists(select *
                                                     from actionmark
                                                     where atm_from_acl_id = aco_id))
                group by aco_object_kind
                into :res_new;
              end
            end
            else
            begin
              if (counter = 0) then
              begin
                select sum(iif(aco_status = 0 and f_point(k_clients_cashier_id, current_date) < :scores, 1, 0)),
                       sum(iif(aco_status = 1 and f_point(k_clients_cashier_id, current_date) < :scores, 1, 0))
                from actionorder, refdata, kontract
                where aco_respons_c_id = :res_un_c_id and
                      aco_from_rd_id = rd_id and
                      k_id = aco_object_id and
                      rd_main_code = 'ActionGroup' and
                      rd_str_id = :rd_str_id and
                      (:db_id <> 81001 or not exists(select *
                                                     from actionmark
                                                     where atm_from_acl_id = aco_id))
                into :res_new, :res_in_process;
                select sum(iif(acl_status = 2 and f_point(k_clients_cashier_id, current_date) < :scores, 1, 0)),
                       sum(iif(acl_status = -2 and f_point(k_clients_cashier_id, current_date) < :scores, 1, 0)),
                       sum(iif(acl_status = -1 and f_point(k_clients_cashier_id, current_date) < :scores, 1, 0))
                from actionlist, refdata, kontract
                where acl_from_rd_id = rd_id and
                      rd_main_code = 'ActionGroup' and
                      rd_str_id = :rd_str_id and
                      k_id = acl_object_id and
                      acl_respons_c_id = :res_un_c_id and
                      acl_dont_show_in_order = 0 and
                      acl_last_datetime >= current_date and
                      (:db_id <> 81001 or not exists(select *
                                                     from actionmark
                                                     where atm_from_acl_id = acl_id))
                into :res_approved, :res_rejected, :res_revision;
              end
              if (counter = 1) then
              begin
                select sum(iif(aco_status = 0 and f_point(k_clients_cashier_id, current_date) >= :scores, 1, 0)),
                       sum(iif(aco_status = 1 and f_point(k_clients_cashier_id, current_date) >= :scores, 1, 0))
                from actionorder, refdata, kontract
                where aco_respons_c_id = :res_un_c_id and
                      aco_from_rd_id = rd_id and
                      k_id = aco_object_id and
                      rd_main_code = 'ActionGroup' and
                      rd_str_id = :rd_str_id and
                      (:db_id <> 81001 or not exists(select *
                                                     from actionmark
                                                     where atm_from_acl_id = aco_id))
                into :res_new, :res_in_process;
                select sum(iif(acl_status = 2 and f_point(k_clients_cashier_id, current_date) >= :scores, 1, 0)),
                       sum(iif(acl_status = -2 and f_point(k_clients_cashier_id, current_date) >= :scores, 1, 0)),
                       sum(iif(acl_status = -1 and f_point(k_clients_cashier_id, current_date) >= :scores, 1, 0))
                from actionlist, refdata, kontract
                where acl_from_rd_id = rd_id and
                      rd_main_code = 'ActionGroup' and
                      rd_str_id = :rd_str_id and
                      k_id = acl_object_id and
                      acl_respons_c_id = :res_un_c_id and
                      acl_dont_show_in_order = 0 and
                      acl_last_datetime >= current_date and
                      (:db_id <> 81001 or not exists(select *
                                                     from actionmark
                                                     where atm_from_acl_id = acl_id))
                into :res_approved, :res_rejected, :res_revision;
              end
            end
            if (res_new is null) then
              res_new = 0;
            if (res_in_process is null) then
              res_in_process = 0;
            if (res_approved is null) then
              res_approved = 0;
            if (res_rejected is null) then
              res_rejected = 0;
            if (res_revision is null) then
              res_revision = 0;
            suspend;
          end
        end
      end
      counter = counter + 1;
    end
  end

  --S.K.
  if (db_id = 81001 and
      exists(select *
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Access_ConsiderApp')
             where has_access = 1)) then
  begin
    rd_id_cld = f_get_rd_id('ActionGroup.Check_Loan_Data');
    res_units = 'UnderWriter_Manager';
    for select acl_respons_c_id, sum(iif(acl_status = 0, 1, 0)), sum(iif(acl_status = 1, 1, 0)),
               sum(iif(acl_status = 2, 1, 0)), sum(iif(acl_status = -2, 1, 0)), sum(iif(acl_status = -1, 1, 0))
        from actionlist
        inner join actionmark on atm_from_acl_id = acl_id
        where acl_last_datetime >= current_date and
              acl_respons_c_id > 0 and
              acl_from_rd_id = :rd_id_cld -- Чтобы использовался индекс actionlist_date, иначе забирает actionmark через natural
        group by acl_respons_c_id
        into :res_un_c_id, :res_new, :res_in_process, :res_approved, :res_rejected, :res_revision
    do
    begin
      select un_id, un_name, un_fio
      from username
      where un_from_c_id = :res_un_c_id rows 1
      into res_un_id, res_un_name, res_un_fio;
      res_in_process = coalesce(res_in_process, 0);
      res_new = coalesce(res_new, 0);
      res_approved = coalesce(res_approved, 0);
      res_rejected = coalesce(res_rejected, 0);
      res_revision = coalesce(res_revision, 0);

      res_online_status = null;
      res_online_text = '';
      res_datetime = null;
      select ovs_add_value, ovs_datetime
      from hp_objectprops_get('clients', :res_un_c_id, -1, :rd_id, null)
      into :res_online_status, res_datetime;
      res_online_status = coalesce(res_online_status, 1);
      res_online_text = case res_online_status
                          when -1 then 'day off'
                          when 0 then 'pause'
                          when 1 then 'online'
                        end;
      suspend;
    end
  end
end
