CREATE PROCEDURE HP_ACTLIST_INS_BY_CHANGE_PHONE(
    C_ID D_ID,
    UN_ID D_ID,
    IS_EMPLOYEE D_SMALLINT = 0)
AS
declare variable rd_id d_integer;
declare variable reg_id d_integer;
declare variable edl_id d_integer;
declare variable edc_id d_integer;
declare variable kind d_integer;
declare variable db_id d_id;
declare variable need_to_update d_smallint;
declare variable asp_const d_smallint;
begin
  db_id = f_get_db_id();
  asp_const = f_get_const('asp_const');
  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ActionGroup.SimpleVerify_Phone_Number_Verify'
  into :rd_id;

  select un.un_curr_reg_id
  from username un
  where un.un_id = :un_id
  into :reg_id;

  kind = f_objkind('clients', 'id');

  if (not exists(select *
                 from actionlist acl
                 where acl.acl_object_kind = :kind and
                       acl.acl_object_id = :c_id and
                       acl.acl_from_rd_id = :rd_id and
                       acl.acl_status not in (-2, 2))) then
  begin
    if (is_employee = 0 or not exists(select *
                                      from kontstate ks
                                      inner join kontract k on ks.ks_from_k_id = k.k_id
                                      inner join knclient kc on k.k_id = kc.kc_from_k_id
                                      where kc.kc_from_c_id = :c_id and
                                            kc.kc_amount_approved > 0 and
                                            k.k_circle > 1 and
                                            ks.ks_state = 1 and
                                            current_date between ks.ks_date_begin and ks.ks_date_end and
                                            ks.ks_ol_is_delete = 0)) then
      insert into actionlist (acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                              acl_datetime, acl_from_un_id, acl_result_rd_id, acl_last_datetime, acl_dont_show_in_order,
                              acl_quota_used)
      values (current_date, :reg_id, :kind, :c_id, :rd_id, 0, current_timestamp, :un_id, 0, current_timestamp, 0, 0);

    edl_id = 0;
    edc_id = 1;

    select edc.edc_id
    from edoc_cfg edc
    where edc.edc_code = 'asp.agreement_employee'
    into :edc_id;

    for select edl.edl_id
        from edoc_list edl
        where edl.edl_object_kind = :kind and
              edl.edl_object_id = :c_id and
              edl.edl_from_edc_id in (1, :edc_id) and
              edl.edl_status in (0, 1) and
              coalesce(edl.edl_date_end, '31.12.2999') >= current_date
        into edl_id

    do
    begin
      if (edl_id <> 0 and asp_const = 0) then
      begin
        need_to_update = 1;

        --Если документ был подписан с помощью биометрии, то договор эцп не аннулируется (т.е. срок жизни постоянный)
        if (db_id in (39801, 41701)) then
        begin
          if (exists(select *
                         from edoc_sign eds
                         inner join edoc_sign_recog esr on esr.esr_from_eds_id = eds.eds_id
                         inner join edoc_face edf on edf.edf_id = esr.esr_from_edf_id
                         where eds.eds_from_edl_id = :edl_id and
                               edf.edf_is_success = 1 and
                               eds.eds_status = 1)) then
          begin
            need_to_update = 0;
          end 
        end 

        if (need_to_update = 1) then
        begin
          update edoc_list
          set edl_date_end = (current_date - 1)
          where edl_id = :edl_id;
        end
      end
    end
  end
  else
    exception e_message 'Есть незавершенный процесс проверки основного телефонного номера у текущего клиента.';
end
