CREATE PROCEDURE AAREP_CLOSED_RATE(
    D1 DATE,
    D2 DATE)
RETURNS (
    K_CLOSE_DATE DATE,
    K_CLOSED_CLIENT INTEGER,
    K_REPATED_CLIENT INTEGER,
    K_CLOSED_ADVANCED INTEGER)
AS
declare variable k_id decimal(18,0);
declare variable c_id decimal(18,0);
begin
  for select ks_from_k_id, kc_from_c_id, ks_date_begin
      from kontstate, knclient
      where ks_from_k_id = kc_from_k_id and
            ks_ol_is_delete = 0 and
            ks_date_begin between :d1 and :d2 and
            ks_state = 2 and
            kc_amount_approved is not null and
            kc_amount_approved > 0
      into :k_id, :c_id, :k_close_date
  do
  begin
    k_closed_client = 1;
    k_repated_client = 0;
    k_closed_advanced = 0;

    if (not(exists(select *
                   from knclient, kontstate
                   where ks_from_k_id = kc_from_k_id and
                         ks_ol_is_delete = 0 and
                         ks_date_begin >= :k_close_date and
                         ks_state = 1 and
                         kc_from_c_id = :c_id and
                         kc_amount_approved is not null and
                         kc_amount_approved > 0))) then
    begin
      k_repated_client = 1;
      if (:k_close_date < (select max(sh_date)
                           from schedule
                           where sh_from_k_id = :k_id)) then
        k_closed_advanced = 1;
    end
    suspend;
  end
end
