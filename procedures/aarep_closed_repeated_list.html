CREATE PROCEDURE AAREP_CLOSED_REPEATED_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    REP_KFOLDER_NAME VARCHAR(100),
    REP_LO_FIO VARCHAR(100),
    REP_REG_NAME VARCHAR(100),
    REP_CLOSE_DATE DATE,
    REP_KOLICH_CLIENT INTEGER,
    REP_DISB_DAY0 INTEGER,
    REP_DISB_DAY1 INTEGER,
    REP_DISB_DAY2 INTEGER,
    REP_DISB_DAY3 INTEGER,
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL2_NAME VARCHAR(100),
    AA_CURR_LEVEL3_NAME VARCHAR(100))
AS
declare variable k_id double precision;
declare variable kc_id decimal(18,0);
declare variable reg_id integer;
declare variable kf_id decimal(18,0);
declare variable kol_tmp integer;
declare variable dd date;
declare variable office_prior integer;
begin
  office_prior = 0;
  for select ks_from_k_id, ks_date_begin, k_kontract_no, k_kolich_client_approved, k_from_kf_id, ks_kc_id, ks_region_id,
             ks_kfolder_no, c_name_fio, reg_code || ' ' || reg_name
      from kontstate, clients, region, region_list(:rep_region_id, 1, -1, 1), kontract, product
      where ks_ol_is_delete = 0 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_state = 2 and
            ks_region_id = id_list and
            ks_region_id = reg_unicode and
            ks_from_k_id = k_id and
            ks_kc_id = c_id and
            k_from_pr_id = pr_id and
            pr_loan_type = 0
      into :k_id, :rep_close_date, :k_kontract_no, :rep_kolich_client, :kf_id, :kc_id, :reg_id, :rep_kfolder_name,
           :rep_lo_fio, :rep_reg_name
  do
  begin
    kol_tmp = 0;
    rep_kolich_client = 1;
    rep_disb_day0 = 0;
    rep_disb_day1 = 0;
    rep_disb_day2 = 0;
    rep_disb_day3 = 0;
    dd = null;
    select ks_date_begin, k_kolich_client_approved
    from kontract, kontstate
    where k_from_kf_id = :kf_id and
          k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_close_date and :rep_close_date + 3 rows 1
    into :dd, :kol_tmp;
    if (kol_tmp is null) then
      kol_tmp = 0;
    if (kol_tmp > 0) then
    begin
      if (:dd - :rep_close_date = 0) then
        rep_disb_day0 = 1;
      if (:dd - :rep_close_date = 1) then
        rep_disb_day1 = 1;
      if (:dd - :rep_close_date = 2) then
        rep_disb_day2 = 1;
      if (:dd - :rep_close_date = 3) then
        rep_disb_day3 = 1;
    end

    if (reg_id <> office_prior) then
    begin
      select aa_curr_branch_name, aa_curr_level3_name, aa_curr_level4_name
      from aarep_get_office_categ(:reg_id, :rep_date_end)
      into :aa_curr_branch_name, aa_curr_level2_name, aa_curr_level3_name;
      office_prior = :reg_id;
    end
    suspend;
  end
end
