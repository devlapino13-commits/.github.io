CREATE PROCEDURE HP_GET_CONTRACT_DELINQ_INFO(
    REP_K_ID D_ID)
RETURNS (
    DELINQUENCY_START_DATE D_DATE,
    WRITE_OFF_DATE D_DATE,
    WRITE_OFF_AMOUNT D_CURRENCY,
    MONTHS_ADDED D_INTEGER,
    K_PERIOD D_INTEGER,
    KD_PERCENT_EFFECT_FIFD D_PERCENT)
AS
begin
  select sum(-ok.ok_fine), ok.ok_ol_date
  from operkred ok
  where ok.ok_from_k_id = :rep_k_id and
        ok.ok_type in (5, 6) and
        ok.ok_ol_is_active = 1
  group by ok.ok_ol_date
  order by ok.ok_ol_date desc
  rows 1
  into :write_off_amount, write_off_date;

  select dateadd(1 day to oks.oks_delinq_date) -- дата начала просрочки
  from oksummary oks
  where oks.oks_from_k_id = :rep_k_id and
        oks.oks_close_state = 0 and
        dateadd(-1 day to :write_off_date) between oks.oks_date_begin and oks.oks_date_end -- берем за день до реструктуризации
  into :delinquency_start_date;

  select iif(kn.k_period - kn.k_kolich_repay < 0, 0, kn.k_period - kn.k_kolich_repay), round(kn.k_period, 0) -- тоже, что и в балансе кредита. Если k_period = 22,6 то округляется до 23. В FS также
  from kontract kn
  where kn.k_id = :rep_k_id
  into :months_added, k_period;

  select kd.kd_percent_effect_fifd
  from kontdata kd
  where kd.kd_from_k_id = :rep_k_id and
        kd.kd_ol_is_delete = 0
  order by kd.kd_ol_date desc
  rows 1
  into  kd_percent_effect_fifd;

  suspend;
end
