CREATE PROCEDURE HP_CARD_KONTRACTS_EPS_UPDATE
AS
declare variable k_id d_id;
declare variable effect_rate_common d_percent;
begin
  for select cke.cke_from_k_id
  from card_kontracts_eps cke
  where cke.cke_status = 0
  rows 1
  into :k_id
  do
  begin
    effect_rate_common = 0;

    select effect_rate_common
    from hp_effect_rate_common(:k_id)
    into :effect_rate_common;  

    update card_kontracts_eps cke
    set cke.cke_status = 1,
        cke.cke_new_eps = :effect_rate_common
    where cke.cke_from_k_id = :k_id;
  end
end
