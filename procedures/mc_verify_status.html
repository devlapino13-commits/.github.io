CREATE PROCEDURE MC_VERIFY_STATUS(
    MCL_ID D_ID)
RETURNS (
    STATUS_CODE D_STR_CODE,
    TITLE D_STR_CODE,
    DESCRIPTION D_STR_255)
AS
declare variable VERIFY_STATUS D_INTEGER;
declare variable C_PASS_DATE_EXPIRE D_DATE;
begin
  verify_status = -999;

  select cl.c_verify_status, cl.c_pass_date_expire
  from mc_login mcl
  inner join clients cl on mcl.mcl_from_c_id = cl.c_id
  where mcl.mcl_is_delete = 0 and
        mcl.mcl_id = :mcl_id
  into :verify_status, :c_pass_date_expire;

  if (verify_status = 2) then
  begin
    title = 'Верифицирован';
    status_code = 'verified';
    description = 'Паспорт верифицирован!';

    if ((c_pass_date_expire is not null) and
        (c_pass_date_expire < current_date)) then
    begin
      title = 'Просрочен';
      status_code = 'overdue';
      description = 'Ваш паспорт просрочен. Пожалуйста, добавьте новый документ.';
    end
  end
  else
  if (verify_status = 1) then
  begin
    title = 'Находится в обработке';
    status_code = 'in_process';
    description = 'Паспорт находится в обработке. Пожалуйста подождите.';
  end
  else
  begin
    title = 'Требуется верификация';
    status_code = 'unverified';
    description = 'Требуется верификация';
  end

  suspend;
end
