CREATE PROCEDURE HP_DEVIDE_KONTRACT_CLIENTS(
    K_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0),
    DEV_DATE DATE,
    DEV_MEMO VARCHAR(255),
    DEV_FROM_RD_ID INTEGER)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    K_NEW_ID DECIMAL(18,0),
    K_NEW_NO VARCHAR(20),
    K_CLIENT_KOL INTEGER,
    KC_AMOUNT_APPLICATE DECIMAL(18,4),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    K_NEW_SUBCIRCLE INTEGER)
AS
declare variable kf_id decimal(18,0);
declare variable k_circle integer;
declare variable k_subcircle integer;
declare variable new_k_id_is_found integer;
declare variable k_amount decimal(18,4);
declare variable kkc_from_kp_id integer;
declare variable kkc_status integer;
declare variable lk_from_lm_id decimal(18,0);
declare variable lk_balance decimal(18,4);
declare variable curr_amount decimal(18,4);
declare variable client_amount decimal(18,4);
declare variable koef decimal(15,7);
declare variable okd_from_ol_id decimal(18,0);
declare variable okd_number integer;
declare variable okd_summa_plus decimal(18,4);
declare variable okd_summa_minus decimal(18,4);
declare variable okd_client_plus decimal(18,4);
declare variable okd_client_minus decimal(18,4);
declare variable is_added integer;
declare variable okz_from_z_id integer;
declare variable okz_from_ol_id decimal(18,0);
declare variable okz_number integer;
declare variable okz_summa_plus decimal(18,4);
declare variable okz_summa_minus decimal(18,4);
declare variable okz_is_money integer;
declare variable okz_account varchar(20);
declare variable sh_id integer;
declare variable sh_date date;
declare variable sh_os decimal(18,4);
declare variable sh_perc decimal(18,4);
declare variable sh_kommis decimal(18,4);
declare variable sh_grace integer;
declare variable sh_is_edited integer;
declare variable sh_perc_tax decimal(18,4);
declare variable sh_disb_amount decimal(18,4);
declare variable sh_diff_date integer;
declare variable sh_kommis_monthly decimal(18,4);
declare variable sh_os_client decimal(18,4);
declare variable sh_perc_client decimal(18,4);
declare variable sh_kommis_client decimal(18,4);
declare variable sh_perc_tax_client decimal(18,4);
declare variable sh_disb_amount_client decimal(18,4);
declare variable sh_kommis_monthly_client decimal(18,4);
declare variable total_rows integer;
declare variable inserted_os decimal(18,4);
declare variable row_no integer;
declare variable ok_from_ol_id decimal(18,0);
declare variable ok_sh_id integer;
declare variable ok_sh_date date;
declare variable ok_type integer;
declare variable ok_os decimal(18,4);
declare variable ok_perc decimal(18,4);
declare variable ok_fine decimal(18,4);
declare variable ok_pay_type integer;
declare variable ok_kommis decimal(18,4);
declare variable ok_sh_os decimal(18,4);
declare variable ok_sh_perc decimal(18,4);
declare variable ok_kommis_adv decimal(18,4);
declare variable ok_tax decimal(18,4);
declare variable ok_os_client decimal(18,4);
declare variable ok_perc_client decimal(18,4);
declare variable ok_fine_client decimal(18,4);
declare variable ok_kommis_client decimal(18,4);
declare variable ok_sh_os_client decimal(18,4);
declare variable ok_sh_perc_client decimal(18,4);
declare variable ok_kommis_adv_client decimal(18,4);
declare variable ok_tax_client decimal(18,4);
declare variable k_user decimal(18,0);
declare variable round_sh integer;
declare variable round_ok integer;
declare variable ks_from_ol_id id_type;
declare variable ks_from_k_id id_type;
declare variable ks_date_begin date;
declare variable ks_state integer;
declare variable ks_kc_id id_type;
declare variable ks_region_id integer;
declare variable ks_kfolder_no varchar(100);
declare variable ks_bonus_change integer;
declare variable lk_calcperc curr_type;
declare variable client_percent curr_type;
declare variable curr_percent curr_type;
declare variable new_folder_id curr_type;
declare variable old_folder_id curr_type;
declare variable ox_from_ol_id decimal(18,0);
declare variable ox_from_rd_id integer;
declare variable ox_amount_plus decimal(18,4) = 0;
declare variable ox_amount_minus decimal(18,4) = 0;
declare variable ox_amount_writeoff decimal(18,4) = 0;
declare variable ox_detail varchar(1024);
declare variable ox_client_amount_plus decimal(18,4) = 0;
declare variable ox_client_amount_minus decimal(18,4) = 0;
declare variable ox_client_amount_writeoff decimal(18,4) = 0;
declare variable ox_ol_date date;
declare variable ox_ol_is_active integer;
declare variable k_is_kn_mode integer;
declare variable kc_id id_type;
declare variable ok_number integer;
declare variable ok_ol_date date;
declare variable ok_ol_reg_id integer;
declare variable ok_ol_is_active integer;
declare variable oek_rd_id integer;
declare variable db_id integer;
declare variable cli_fio str_100;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');
  db_id = f_get_db_id();

-- operkred_fix
  if (dev_from_rd_id is null) then
    dev_from_rd_id = 0;
  /* sanjari, finstart, 02.08.2012 */
  /* процедура для разделения одного участника группы на отдельный новый контракт */

  /*
  входящие параметры:
  k_id - основной контракт, из которого планируется вывести участника
  c_id - id клиента, который выходит из группы
  un_id - пользователь, соверщающий разделение группы
  dev_from_rd_id - причина амнистирования
  */

  if (un_id is null) then
    un_id = 1000000000001;

  select k_kontract_no, k_from_kf_id, k_circle, k_amount_approved, k_user_id, k_kolich_client_approved, k_is_kn_mode
  from kontract
  where k_id = :k_id
  into :k_kontract_no, :kf_id, :k_circle, :k_amount, :k_user, :k_client_kol, k_is_kn_mode;

  if (k_amount is null) then
    k_amount = 0;
  if (k_is_kn_mode is null) then
    k_is_kn_mode = 0;

  if (k_client_kol <= 1) then
  begin
    k_new_id = 0;
    k_new_subcircle = 0;
    k_new_no = 'последний в договоре';
    suspend;
    exit;
  end

  if ((kf_id is null) and
      (k_amount = 0)) then
  begin
    k_new_id = 0;
    k_new_subcircle = 0;
    k_new_no = 'нет займа';
    suspend;
    exit;
  end

  select kc_id, kc_amount_applicate, kc_amount_approved
  from knclient
  where kc_from_k_id = :k_id and
        kc_from_c_id = :c_id and
        kc_amount_approved is not null and
        kc_amount_approved > 0 rows 1
  into :kc_id, :kc_amount_applicate, :kc_amount_approved;

  if (kc_amount_applicate is null) then
    kc_amount_applicate = 0;
  if (kc_amount_approved is null) then
    kc_amount_approved = 0;
  if (kc_amount_approved = 0) then -- клиент не получал кредит, завершаем работу
  begin
    k_new_id = 0;
    k_new_subcircle = 0;
    k_new_no = 'нет в договоре';
    suspend;
    exit;
  end

  rdb$set_context('USER_TRANSACTION', 'DEVIDE_IN_PROGRESS', 1);
  round_sh = 0;/* округление графика, кол-во цифр после запятой */
  round_ok = 0;/* округление погашений, кол-во цифр после запятой */
  if (:k_user is null) then
    k_user = :un_id;

  /* определяем последний суб-цикл кредита в данном цикле*/
  select max(k_circle_internal)
  from kontract
  where k_id = :k_id
  into :k_subcircle;

  if (k_subcircle is null) then
    k_subcircle = 0;
  k_new_subcircle = k_subcircle + 1;/* генерируем новый субцикл */
  k_new_no = :k_kontract_no || '/' || :k_new_subcircle;/* генерируем новый номер контракта */
  k_new_id = kf_id * 1000 + k_circle;/* подготовка нового id контракта */
  k_new_id = k_new_id * 100 + k_new_subcircle;/* подготовка нового id контракта */

  if (exists(select *
             from kontract
             where k_id = :k_new_id)) /* проверка отсутствия нового id контракта */
  then
    new_k_id_is_found = 1;
  else
    new_k_id_is_found = 0;

  while (new_k_id_is_found = 1) do
  /* если генерированный id контракта существует, ищем следующую */
  begin
    k_new_subcircle = k_new_subcircle + 1;
    k_new_no = :k_kontract_no || '/' || :k_new_subcircle;
    k_new_id = kf_id * 1000 + k_circle;
    k_new_id = k_new_id * 100 + k_new_subcircle;

    if (exists(select *
               from kontract
               where k_id = :k_new_id)) then
      new_k_id_is_found = 1;
    else
      new_k_id_is_found = 0;
  end
  /* id нового контракта new_k_id */

  /* aschegolskiy, finstart, 09.08.2012 */
  /*если пользователь удален, вставляем пользователя из параметров операции*/
  if (not exists(select un_id
                 from username
                 where un_id = :k_user)) then
    k_user = un_id;

  /*
  создаем новым контракт.
  сумма и кол-во участников в контракте будет уменьшен в основном контракте после успешного завершения разделения */

  insert into kontract (k_id, k_kontract_no, k_circle, k_circle_internal, k_amount_applicate, k_amount_approved,
                        k_clients_leader_id, k_clients_cashier_id, k_from_kf_id, k_currency, k_applicate_date,
                        k_applicate_no, k_kontract_date, k_from_pr_id, k_disb_date, k_stage, k_period, k_kolich_lgota,
                        k_percent, k_percent_lgota, k_kolich_repay, k_from_f_id, k_from_pt_id, k_status, k_close_date,
                        k_percent_per_day, k_memo, k_percent_for_days, k_kolich_client_applicate,
                        k_kolich_client_approved, k_repay_kolich_closed, k_class, k_kommis_perc, k_from_cs_id,
                        k_credit_target, k_from_pac_id, k_is_kredline, k_kredline_end, k_kont_last_date, k_stop_perc,
                        k_stop_fine, k_index_curr_code, k_index_curr_rate, k_from_ma_reg_id, k_from_ma_app_id,
                        k_percent_input, k_percent_category, k_percent_effect, k_percent_effect_fifd, k_user_id,
                        k_percent_flat_annual, k_financed, k_last_reg_id, k_is_problem, k_percent_begin_annual,
                        k_fine_percent, k_stopped_perc_ol_id, k_zero_balance_date, k_expire_date, k_repay_day,
                        k_approved_manager, k_is_devided, k_base_k_id, k_percent_market, k_reserv_kind, k_is_kn_mode)
  select :k_new_id, :k_new_no, :k_circle, :k_new_subcircle, :kc_amount_applicate, :kc_amount_approved, :c_id, :c_id,
         k_from_kf_id, k_currency, k_applicate_date, k_applicate_no, k_kontract_date, k_from_pr_id, k_disb_date,
         k_stage, k_period, k_kolich_lgota, k_percent, k_percent_lgota, k_kolich_repay, k_from_f_id, k_from_pt_id,
         k_status, k_close_date, k_percent_per_day, k_memo, k_percent_for_days, 1, 1, k_repay_kolich_closed, k_class,
         k_kommis_perc, k_from_cs_id, k_credit_target, k_from_pac_id, k_is_kredline, k_kredline_end, k_kont_last_date,
         k_stop_perc, k_stop_fine, k_index_curr_code, k_index_curr_rate, k_from_ma_reg_id, k_from_ma_app_id,
         k_percent_input, k_percent_category, k_percent_effect, k_percent_effect_fifd, :k_user, k_percent_flat_annual,
         k_financed, k_last_reg_id, k_is_problem, k_percent_begin_annual, k_fine_percent, k_stopped_perc_ol_id,
         k_zero_balance_date, k_expire_date, k_repay_day, k_approved_manager, 1, :k_id, k_percent_market, k_reserv_kind, 0 -- k_is_kn_mode в начале устанавливаем на 0, чтобы не было пересчета kn_oksummary
  from kontract
  where k_id = :k_id;

  /* заменяем id контракта на новый в составен участников кредита */
  update knclient
  set kc_from_k_id = :k_new_id --, kc_number = 1
  where kc_id = :kc_id;

  update kontract
  set k_is_kn_mode = 0, -- временно отключаем k_is_kn_mode чтобы повторно не создавались записи в kn_operkred
      k_is_devided = 1,
      k_base_k_id = :k_id,
      k_amount_applicate = k_amount_applicate - :kc_amount_applicate,
      k_amount_approved = k_amount_approved - :kc_amount_approved,
      k_kolich_client_applicate = k_kolich_client_applicate - 1,
      k_kolich_client_approved = k_kolich_client_approved - 1
  where k_id = :k_id;

  update kontract
  set k_is_kn_mode = 0 -- временно отключаем k_is_kn_mode чтобы повторно не создавались записи в kn_operkred
  where k_id = :k_new_id;

  /* сохраняем историю деления: какой контракт на какой и кто выделялся */
  insert into kontdevides (kd_id, kd_k_orig, kd_k_new, kd_c_id, kd_from_un_id, kd_date, kd_memo, kd_from_rd_id)
  values (-1, :k_id, :k_new_id, :c_id, :un_id, :dev_date, :dev_memo, :dev_from_rd_id);

  /* заменяем id контракта на новый, если клиент заполнял анкету в этом контракте*/
  update cloprosmain
  set copm_from_k_id = :k_new_id
  where copm_from_k_id = :k_id and
        copm_from_c_id = :c_id;

  /* заменяем id контракта на новый, если по клиенту заполнялся отчет о мониторинге просрочки в этом контракте */
  update delinqrep
  set dq_from_k_id = :k_new_id
  where dq_from_k_id = :k_id and
        dq_from_c_id = :c_id;

  /* заменяем id контракта на новый в таблице kib_ignore */
  update kib_ignore
  set ki_from_k_id = :k_new_id
  where ki_from_k_id = :k_id and
        ki_from_c_id = :c_id;

  /* заменяем id контракта на новый, если по клиенту заполнен мониторинг в этом контракте */
  update monitoring
  set km_from_k_id = :k_new_id
  where km_from_k_id = :k_id and
        km_from_c_id = :c_id;

  /* заменяем id контракта на новый, если клиент попал в этом цикле в Стоп-лист */
  update moratory
  set m_last_k_id = :k_new_id
  where m_last_k_id = :k_id and
        m_from_c_id = :c_id;

  /* если клиент участвовал при отправке в киб, то включаем нового контракта в каждый пакет отправки */
  for select distinct kkc_from_kp_id, kkc_status
      from kibclients
      where kkc_from_k_id = :k_id and
            kkc_from_c_id = :c_id
      into :kkc_from_kp_id, :kkc_status
  do
  begin
    /* включение нового контракта в пакет отправки */
    insert into kibkredit (kkr_from_kp_id, kkr_from_k_id, kkr_uniq_cred_num, kkr_bank_code, kkr_fil_code,
                           kkr_credtype_code, kkr_credstatex_code, kkr_send_status, kkr_send_result, kkr_transid,
                           kkr_send_time, kkr_repaystatus, kkr_guarantee, kkr_countofdelinquency, kkr_unpaidamount,
                           kkr_unpaidpercentamount, kkr_verify_status, kkr_paymethod)
    select :kkc_from_kp_id, :k_new_id, kkr_uniq_cred_num, kkr_bank_code, kkr_fil_code, kkr_credtype_code,
           kkr_credstatex_code, kkr_send_status, kkr_send_result, kkr_transid, kkr_send_time, kkr_repaystatus,
           kkr_guarantee, kkr_countofdelinquency, kkr_unpaidamount, kkr_unpaidpercentamount, kkr_verify_status,
           kkr_paymethod
    from kibkredit
    where kkr_from_kp_id = :kkc_from_kp_id and
          kkr_from_k_id = :k_id;

    /* перенос клиента из старого контракта на новый в данном пакете отправки */
    update kibclients
    set kkc_from_k_id = :k_new_id
    where kkc_from_k_id = :k_id and
          kkc_from_kp_id = :kkc_from_kp_id and
          kkc_from_c_id = :c_id;
    /* и так проделывается для каждого пакета где был отправлен этот участник в данном цикле (контракте) */
  end

  /* создаем записи для нового контракта */
  insert into kontstate (ks_from_ol_id, ks_from_k_id, ks_date_begin, ks_date_end, ks_ol_is_delete, ks_state)
  select ks_from_ol_id, :k_new_id, ks_date_begin, ks_date_end, ks_ol_is_delete, ks_state
  from kontstate
  where ks_from_k_id = :k_id and
        ks_from_ol_id is not null /* исключаем записи с пустым ol_id, потому что они
                                    создаются автоматически при создании новой записи в kontract  */
  order by ks_number;/* записи должны создаваться по одинаковой очереди */

  /* создать новую запись в реструктрированных, если кредит был реструктурирован */
  if (exists(select *
             from operrest
             where or_from_k_id = :k_id)) then
  begin
    insert into operrest (or_from_ol_id, or_from_k_id, or_kind, or_date_end)
    select or_from_ol_id, :k_new_id, or_kind, or_date_end
    from operrest
    where or_from_k_id = :k_id;
  end

  /* создать новую запись в акте передачи в сб, если кредит был передан в сб */
  if (exists(select *
             from sb_akt
             where sba_from_k_id = :k_id)) then
  begin
    insert into sb_akt (sba_from_k_id, sba_number, sba_date_begin, sba_date_end, sba_from_sbl_id, sba_status,
                        sba_from_un_id, sba_datetime, sba_is_delete)
    select :k_new_id, -1, sba_date_begin, sba_date_end, sba_from_sbl_id, sba_status, sba_from_un_id, sba_datetime,
           sba_is_delete
    from sb_akt
    where sba_from_k_id = :k_id;
  end

  /* разделяем результаты звонков по проекту пантера*/
  execute procedure hp_pantera_devide_kont(:k_id, :c_id, :k_new_id);

  /* создать новую запись в залогах, если на кредит был оформлен залог или поручители */
  if (exists(select *
             from zalog
             where z_from_k_id = :k_id)) then
  begin
    insert into zalog (z_from_k_id, z_id, z_from_c_id, z_type, z_subtype, z_name, z_from_adl_id, z_document,
                       z_cost_market, z_cost_likvid, z_cost_kontract, z_memo1, z_memo2, z_name_fio, z_phone)
    select :k_new_id, -1, z_from_c_id, z_type, z_subtype, z_name, z_from_adl_id, z_document, z_cost_market,
           z_cost_likvid, z_cost_kontract, z_memo1, z_memo2, z_name_fio, z_phone
    from zalog
    where z_from_k_id = :k_id;
  end

  /* если по кредиту был начислен рппу, то создать новую запись с соответствующей суммой баланса */
  for select distinct lk_from_lm_id, lk_balance, lk_calcperc
      from llc_kont
      where lk_from_k_id = :k_id
      into :lk_from_lm_id, :lk_balance, :lk_calcperc
  do
  begin
    koef = :lk_balance / :k_amount;
    client_amount = cast(:koef * :kc_amount_approved as decimal(15,2));
    curr_amount = :lk_balance - :client_amount;
    koef = :kc_amount_approved / :k_amount;
    client_percent = cast(:koef * :lk_calcperc as decimal(15,2));
    curr_percent = :lk_calcperc - :client_percent;

    /* включение нового контракта в рппу с балансом участника */
    insert into llc_kont (lk_from_lm_id, lk_from_k_id, lk_balance, lk_deliq_days, lk_from_reg_id, lk_status, lk_percent,
                          lk_date, lk_date_end, lk_memo, lk_is_indiv, lk_is_signed, lk_is_changed_mannual,
                          lk_percent_orig, lk_calcperc, lk_is_secured, lk_is_restruct)

    for select ok_from_ol_id, ok_sh_id, ok_sh_date, ok_type, sum(ok_os), sum(ok_perc), sum(ok_fine), sum(ok_kommis),
               sum(ok_sh_os), sum(ok_sh_perc), sum(ok_kommis_adv), sum(ok_tax)
        from operkred
        where ok_from_k_id = :k_new_id and
              ok_ol_is_active = 1
        group by ok_from_ol_id, ok_sh_id, ok_sh_date, ok_type
        into :ok_from_ol_id, :ok_sh_id, :ok_sh_date, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_kommis, :ok_sh_os,
             :ok_sh_perc, :ok_kommis_adv, :ok_tax
    do
    begin
      update operkred
      set ok_os = ok_os - :ok_os,
          ok_perc = ok_perc - :ok_perc,
          ok_fine = ok_fine - :ok_fine,
          ok_kommis = ok_kommis - :ok_kommis,
          ok_sh_os = ok_sh_os - :ok_sh_os,
          ok_sh_perc = ok_sh_perc - :ok_sh_perc,
          ok_kommis_adv = ok_kommis_adv - :ok_kommis_adv,
          ok_tax = ok_tax - :ok_tax
      where (ok_from_ol_id = :ok_from_ol_id) and
            (ok_from_k_id = :k_id) and
            (ok_sh_id = :ok_sh_id) and
            (ok_sh_date = :ok_sh_date) and
            (ok_type = :ok_type) order by ok_from_ol_id, ok_number rows 1;
    end
  end
  else
  begin
    update kn_operkred
    set ok_is_actual = 2
    where ok_from_kc_id = :kc_id;

    delete from kn_operkred
    where ok_from_kc_id = :kc_id;

    insert into kn_operkred (ok_from_ol_id, ok_number, ok_from_kc_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                             ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                             ok_ol_reg_id, ok_ol_is_active, ok_is_actual)
    select ok_from_ol_id, ok_number, :kc_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine, ok_pay_type,
           ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active, 2
    from operkred
    where ok_from_k_id = :k_new_id
    order by ok_from_ol_id, ok_number;

    update kn_operkred
    set ok_is_actual = 1
    where ok_from_kc_id = :kc_id and
          ok_is_actual = 2;
  end

  update operkroff
  set oko_from_k_id = :k_new_id
  where oko_from_k_id = :k_id and
        oko_from_c_id = :c_id;

  update kn_oksummary
  set oks_from_k_id = :k_new_id
  where oks_from_kc_id = :kc_id and
        oks_from_k_id <> :k_new_id;

  update kontract
  set k_is_kn_mode = :k_is_kn_mode,
      k_is_devided = 1,
      k_fact_disb_online = -1
  where k_id in (:k_id, :k_new_id) and
        coalesce(k_is_kn_mode, 0) <> :k_is_kn_mode;

  update kontract
  set k_fact_disb_online = -1
  where k_id in (:k_id, :k_new_id);

  if (db_id = 94401 and
      exists(select * from azrepaykont where ark_from_k_id = :k_id)) then
  begin
    select c_name_fio from clients
    where c_id = :c_id
    into cli_fio;

    insert into azrepaykont (ark_id, ark_from_k_id, ark_loan_id, ark_summ, ark_cli_fio, ark_kol_cli, ark_is_finish,
                             ark_error_text, ark_repay_ol_id, ark_ks_state, ark_kontract_no, akr_message_text,
                             ark_fact_repay_sum, ark_from_c_id)
    values (null, :k_new_id, null, 0, :cli_fio, 1, -1, null,
            null, null, :k_new_no, 'Амнистирован', null, null);
  end 

  execute procedure hp_oksummary_create(:k_id);
  execute procedure hp_oksummary_create(:k_new_id);

  rdb$set_context('USER_TRANSACTION', 'DEVIDE_IN_PROGRESS', null);

  suspend;
end
