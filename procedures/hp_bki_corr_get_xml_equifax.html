CREATE PROCEDURE HP_BKI_CORR_GET_XML_EQUIFAX(
    XML_VERSION_ID D_INTEGER,
    BEV_ID D_ID,
    PARENT_NODE D_STR_255,
    REC_NO D_INTEGER,
    PARENT_ID D_ID = null,
    IS_NEW D_BOOL = null)
RETURNS (
    RES_XML D_STR_8100)
AS
declare variable xml_field d_str_255;
declare variable db_field d_str_100;
declare variable bfv_id d_id;
declare variable field_val d_str_8100;
declare variable field_val_tmp d_str_8100;
declare variable attr d_str_8100;
declare variable is_empty boolean;
declare variable bki_code d_str_code;
declare variable jso d_integer;
declare variable bkc_add d_str_long;
declare variable event_date d_date;
declare variable packet_datetime d_datetime;
begin
  res_xml = '';
  attr = '';
  is_empty = true;

  if (parent_id is null) then  -- корень
  begin
    select bev_event_date, bkc_add, bkp_send_datetime
    from bki_events, bki_clients, bki_loans, bki_packet
    where bkc_id = bev_from_bkc_id
      and bkl_id = bkc_from_bkl_id
      and bkp_id = bkl_from_bkp_id
      and bev_id = :bev_id
    into event_date, bkc_add, packet_datetime;

    jso = iif(coalesce(bkc_add, '') = '', json.obj_create(), json.parse(bkc_add));

    -- заполним аттрибуты события
    attr = f_format(' operationCode="C.1" orderNum="%1" eventDateTime="%2" changeReason="%3"', rec_no, f_bki_datetime_to_str(packet_datetime, 3), json.obj_get_val(jso, 'change_reason', '1'));

    -- ищем корневые узлы в коррекции
    select bfv_id, bxf_xml_node
    from bki_xml_fields_ver, bki_xml_fields
    where bxf_id = bfv_from_bxf_id
      and bfv_from_bxv_id = :xml_version_id
      and bxf_from_bki_code = :parent_node
    into parent_id, parent_node;

    for select bfv_id, bki_code, bxf_xml_node
    from bki_xml_fields_ver, bki_xml_fields, bki_indicators
    where bfv_parent_id = :parent_id
      and bxf_id = bfv_from_bxf_id
      and bki_code = bxf_from_bki_code
      and bfv_from_bxv_id = :xml_version_id
      and bki_db_field starting with '<'
    order by bfv_position
    into bfv_id, bki_code, xml_field
    do
    begin
      -- если есть данные для коррекции внутри блока, то заполним его
      if (exists(select bfv_id
                from hp_bki_indicator_list(:bki_code, :xml_version_id), bki_send_data_corr
                where bsd_from_bev_id = :bev_id
                  and bsd_from_bfv_id = bfv_id
                  and bsd_from_bkd_id is distinct from bsd_new_value_id))
      then
      begin
        is_empty = false;
        res_xml = res_xml || '<' || xml_field || '>';
        res_xml = res_xml || (select res_xml from hp_bki_corr_get_xml_equifax(:xml_version_id, :bev_id, :xml_field || '_current', null, :bfv_id));
        res_xml = res_xml || (select res_xml from hp_bki_corr_get_xml_equifax(:xml_version_id, :bev_id, :xml_field || '_new', null, :bfv_id, 1));
        res_xml = res_xml || '</' || xml_field || '>';
      end
    end
  end
  else
  begin
    is_new = coalesce(is_new, 0);

    for select bfv_id, bxf_xml_node, bki_db_field
    from bki_xml_fields_ver, bki_xml_fields, bki_indicators
    where bxf_id = bfv_from_bxf_id
      and bki_code = bxf_from_bki_code
      and bfv_from_bxv_id = :xml_version_id
      and bfv_parent_id = :parent_id
    order by bfv_position
    into bfv_id, xml_field, db_field
    do
    begin
      if (db_field in ('<OBJ>', '<ARR>')) then
        field_val = (select res_xml from hp_bki_corr_get_xml_equifax(:xml_version_id, :bev_id, :xml_field, null, :bfv_id, :is_new));
      else
      begin
        field_val = null;

        select f_bki_data_text(iif(:is_new = 0, bsd_from_bkd_id, bsd_new_value_id))
        from bki_send_data_corr
        where bsd_from_bfv_id = :bfv_id
          and bsd_from_bev_id = :bev_id
        into field_val;
  
        if (coalesce(field_val, '') <> '') then
        begin
          field_val = replace(field_val, '<', 'lt;');
          field_val = replace(field_val, '>', 'gt;');

          if (xml_field containing '<X>') then
            field_val = f_format('<%1></%1>', replace(xml_field, '<X>', field_val));
          else if (db_field starting with '#arr#') then
          begin
            field_val_tmp = field_val;

            select list(f_format('<%1>%2</%1>', :xml_field, res_str), '')
            from hp_split_str(:field_val_tmp, ',')
            into field_val;
          end
          else
            field_val = f_format('<%1>%2</%1>', xml_field, field_val);
        end
      end

      if (coalesce(field_val, '') <> '') then
      begin
        is_empty = false;
        res_xml = res_xml || field_val;
      end
    end
  end

  res_xml = iif(is_empty, null, '<' || parent_node || attr || '>' || res_xml || '</' || parent_node || '>');
  suspend;
end
