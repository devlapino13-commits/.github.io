CREATE PROCEDURE AAREP_NOT_APP_DISB(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    MA_ID INTEGER,
    MA_DATE_APP DATE,
    MA_NO INTEGER,
    K_KONTRACT_NO VARCHAR(20),
    REG_NAME VARCHAR(60),
    KAL_NEW_KF_NAME VARCHAR(20),
    KF_OFFICER VARCHAR(100))
AS
declare variable ma_number integer;
declare variable ma_from_kf_id decimal(18,0);
declare variable k_id decimal(18,0);
declare variable k_state integer;
declare variable kal_new_kf_officer decimal(18,0);
declare variable kal_new_reg_id integer;
begin
--    for
--    with tabl1 as (
--        select ma_from_reg_id miniap_reg_id,
--               ma_id miniap_id,
--               max(ma_number) miniap_number
--        from miniapp
--        group by ma_from_reg_id, ma_id)
--    select ma_from_reg_id, ma_id, ma_number, ma_date_app, ma_from_kf_id, ma_no
--    from tabl1, miniapp
--    where miniap_reg_id = ma_from_reg_id
--    and miniap_id = ma_id
--    and miniap_number = ma_number
--    and ma_is_active = 1 -- активные заявки

  for select ma_id, ma_number, ma_date_app, ma_from_kf_id, ma_no, kal_new_reg_id, kal_new_kf_name, kal_new_kf_officer
      from miniapp, knaktlist, region_list(:rep_region_id, 1, 3, 1)
      where ma_date_end = '31.12.9999' and
            ma_date_app < (:rep_date_end - 10) and
            ma_is_active = 1 -- активные заявки
            and
            kal_new_reg_id = id_list and
            knaktlist.is_delete = 0 and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            kal_from_kf_id = ma_from_kf_id
      into ma_id, ma_number, ma_date_app, ma_from_kf_id, ma_no, kal_new_reg_id, kal_new_kf_name, kal_new_kf_officer
  do
  begin
    k_id = null;
    k_kontract_no = null;
    select k_id, k_kontract_no
    from kontract
    where k_from_kf_id = :ma_from_kf_id and
          k_from_ma_app_id = :ma_id rows 1
    into k_id, k_kontract_no;

    k_state = 0;
    if (k_id is not null) then
    begin
      select 1
      from kontstate
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_from_k_id = :k_id and
            ks_date_begin <= :rep_date_end
      into k_state;
    end

    if (k_state = 0) then
    begin
      kf_officer = (select c_name_fio
                    from clients
                    where c_id = :kal_new_kf_officer);
      reg_name = (select reg_code || ' ' || reg_name
                  from region
                  where reg_unicode = :kal_new_reg_id);
      suspend;
    end
  end
end
