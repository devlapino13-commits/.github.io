CREATE PROCEDURE CRM_MINIAPP_KMP(
    C_ID ID_TYPE)
RETURNS (
    STATUS STR_100,
    REFUSE STR_255,
    SVZ_ID ID_TYPE,
    OFFICE_ID D_INTEGER,
    EMA_DATETIME D_DATETIME,
    STATUS_DATETIME D_DATETIME)
AS
declare variable EMA_STATUS D_INTEGER;
declare variable EMA_REFUSE STR_255;
declare variable EMA_LO_ID ID_TYPE;
declare variable EMA_OFFICE_ID D_ID;
declare variable EMA_STATUS_DATETIME D_DATETIME;
declare variable MA_ID D_INTEGER;
declare variable MA_STATUS D_INTEGER;
declare variable MA_STATUS_DATETIME D_DATETIME;
declare variable MA_REFUSE D_INTEGER;
declare variable MA_LO_ID ID_TYPE;
declare variable MA_OFFICE_ID D_ID;
declare variable K_ID D_ID;
declare variable K_STATUS D_INTEGER;
declare variable K_STATUS_DATETIME D_DATETIME;
declare variable K_REFUSE D_INTEGER;
declare variable K_LO_ID ID_TYPE;
declare variable K_OFFICE_ID D_ID;
declare variable TKT_ID ID_TYPE;
declare variable ema_block_datetime d_datetime;
begin
    for select ema_status, rd_code, ema_lo_id, ema_office_id, ema_datetime, ema_status_datetime, ema_from_tkt_id, ema_block_datetime
    from extminiapp
    inner join refdata on rd_id = ema_result_rd_id
    where ema_from_c_id = :c_id
    and ema_from_mcl_id is not null
    into ema_status, ema_refuse, ema_lo_id, ema_office_id, ema_datetime, ema_status_datetime, tkt_id, ema_block_datetime
    do
    begin 
        ma_id = null;
        k_id = null;
        if ((ema_status = -1 and ema_refuse = 'website_miniapp.Refuse.MiniappCreateCancel' and ema_lo_id = 1000000000000) or (ema_status = 1)) then
        begin
            if (exists(select ma_id
                        from miniapp
                        where ma_from_tkt_id = :tkt_id)) then
            begin
                select ma_id
                from miniapp
                where ma_from_tkt_id = :tkt_id
                rows 1
                into :ma_id;

                select k_id
                from kontract
                where k_from_tkt_id = :tkt_id
                rows 1
                into :k_id;
            end
            else
            begin
                select ma_id
                from miniappclient
                inner join miniapp on ma_id = mac_from_ma_id
                where mac_from_c_id = :c_id
                and ma_date_app >= cast(:ema_datetime as date)
                and ma_date_end = '31.12.9999'
                and mac_amount > 0
                order by ma_date_app
                rows 1
                into :ma_id;

                select k_id
                from kontract
                where k_from_ma_app_id = :ma_id
                rows 1
                into :k_id;

                if (ma_id is not null and k_id is null) then
                begin
                    select k_id
                    from knclient
                    inner join kontract on k_id = kc_from_k_id
                    where kc_from_c_id = :c_id
                    and k_kontract_date >= cast(:ema_datetime as date)
                    order by k_kontract_date
                    rows 1
                    into k_id;
                end 
            end

            select ma_is_active, krl_from_kr_id, un_from_c_id , ma_office_id, ma_date_save
            from miniapp
            inner join username on un_id = ma_from_un_id
            left join kontreslist on krl_ma_id = ma_id
            where ma_id = :ma_id
            and ma_date_end = '31.12.9999'
            rows 1
            into ma_status, ma_refuse, ma_lo_id , ma_office_id,  ma_status_datetime;
            if (k_id is not null) then
            begin
                select k_status, krl_from_kr_id, ks_kc_id, ks_region_id,
                       iif(k_status = 1, ol_datetime, coalesce(krl_datetime, :ma_status_datetime))
                from kontract
                left join kontreslist  on krl_from_k_id = k_id
                left join kontstate on ks_from_k_id = k_id
                left join operkred on ok_from_k_id = k_id and ok_type = 8
                                                          and ok_ol_is_active = 1
                left join operlist on ol_id = ok_from_ol_id
                where k_id = :k_id
                and ks_date_end = '31.12.9999'
                and ks_ol_is_delete = 0
                rows 1
                into k_status, k_refuse, k_lo_id, k_office_id, k_status_datetime;
            end
        end

        if (ema_status = 0 and ema_block_datetime is null) then
            status = 'Не заблокировано ВИПС';
        if (ema_status = 0 and ema_block_datetime is not null) then
            status = 'Заблокировано ВИПС';
        if (ema_status = -1) then
            status = 'Отказ';

        refuse = coalesce(ema_refuse, '');
        svz_id = coalesce(ema_lo_id, 0);
        office_id = coalesce(ema_office_id, 0);
        status_datetime = ema_status_datetime;

        if (ma_id is not null) then
            begin
                status = case ma_status
                         when 1 then 'В процесе'
                         when 0 then 'Отказ'
                         when -1 then 'Отказ'
                         end;
                refuse = coalesce(ma_refuse, '');
                svz_id = coalesce(ma_lo_id, 0);
                office_id = coalesce(ma_office_id, 0);
                status_datetime = ma_status_datetime;
            end
        if (k_id is not null) then
            begin
                status = case k_status
                         when 1 then 'Выдан'
                         when 0 then 'В процессе'
                         when -1 then 'Отказ'
                         when -2 then case ma_status
                                      when 1 then 'В процесе'
                                      when 0 then 'Отказ'
                                      when -1 then 'Отказ'
                                      end
                         end;
                refuse = iif(k_status = -2, coalesce(ma_refuse, ''), coalesce(k_refuse, ''));
                svz_id = coalesce(k_lo_id, 0);
                office_id = coalesce(k_office_id, 0);
                status_datetime = k_status_datetime;
            end
        suspend;
    end
end
