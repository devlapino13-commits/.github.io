CREATE PROCEDURE HP_HR_PSTATUS_UPDATE(
    HO_ID D_ID,
    HO_DATE D_DATE,
    HO_FROM_REG_ID D_INTEGER,
    HO_TEXT D_STR_LONG,
    HO_REASON D_STR_255,
    HO_ORDER_TYPE D_INTEGER,
    HO_ORDER_SUBTYPE D_INTEGER,
    HO_ORDER_NO D_STR_20,
    HO_COME_BACK_DATE D_DATE,
    HO_CAUSE_ID D_INTEGER,
    HO_SIGNER_ID D_INTEGER,
    HO_TRANSFER_OFFICE D_INTEGER,
    PS_ID D_INTEGER,
    PS_DATE_BEGIN D_DATE,
    PS_PERIOD_MONTH D_INTEGER,
    PS_DATE_END_IN_ORDER D_DATE,
    PS_FROM_REG_ID D_INTEGER,
    PS_FROM_DEP_ID D_INTEGER,
    PS_FROM_JL_ID D_INTEGER,
    PS_IS_WORKING D_SMALLINT,
    PS_FROM_C_ID ID_TYPE,
    PS_SALARY CURR_TYPE,
    PS_TRANSFER_OFFICE D_INTEGER,
    PS_TABEL_NO D_STR_10,
    PS_CLASS D_INTEGER,
    PS_IS_MOUNT D_BOOL,
    PS_IS_TIMELESS D_BOOL,
    PS_MOUNT_KOEF D_CURRENCY,
    PS_IS_WORK_BOOK D_BOOL,
    RES_WORK_KIND D_SMALLINT,
    PSA_AGREEMENT_NO D_STR_10,
    HM_MOUNT D_INTEGER,
    UN_ID D_ID,
    PS_NOTIFIC_NO D_STR_CODE,
    PS_NOTIFIC_ADDRESS D_INTEGER)
AS
declare variable PS_IS_ON_STAFF D_SMALLINT;
declare variable PS_IS_PART D_BOOL;
declare variable ORIG_HM_MOUNT D_INTEGER;
declare variable MOUNT_PS_ID D_INTEGER;
declare variable MOUNT_HO_ID D_ID;
begin
  ps_is_on_staff = 1;
  ps_is_part = 0;
  if (res_work_kind = 2) then
    ps_is_part = 1;
  else
    ps_is_on_staff = res_work_kind;

  if (ho_order_type = 6) then
  begin
    ps_date_begin = ps_date_begin + 1;
    ps_date_end_in_order = ps_date_begin;
    ps_is_timeless = 0;
    ps_period_month = 0;
    ps_is_working = 0;
  end else
    ps_is_working = 1;

  if (ho_order_type = 1) then
  begin
    if (exists(select * from hp_relationlink_check(:ps_from_c_id, :un_id, 1)
               where result_error = 1)) then
      exception e_message 'У данного физлица имеются' || ascii_char(13) || 'не одобренные родственники-коллеги.' || ascii_char(13) || 'Сначала получите одобрение.';

    if (exists(select * from hr_pstatus
               where ps_from_c_id = :ps_from_c_id and
                     ps_is_working = 1 and
                     ps_status = 1 and
                     ps_date_end >= '31.12.2099')) then
      exception e_message 'Выбранное физ.лицо уже является действующим сотрудником';
  end

  if (ho_order_type = 2) then
     ps_from_reg_id = ho_transfer_office;

  if (ps_date_end_in_order is null) then
    ps_date_end_in_order = '31.12.2099';

  if (ps_is_timeless is null) then
    ps_is_timeless = 0;

  if (ps_is_timeless = 1) then
    ps_date_end_in_order = '31.12.2099';

  if (ps_is_work_book is null) then
    ps_is_work_book = 0;

  if (coalesce(ho_id, 0) = 0) then
  begin
    insert into hr_orders (ho_date, ho_from_reg_id, ho_text, ho_reason, ho_from_un_id, ho_order_type,
                           ho_order_subtype, ho_order_no, ho_come_back_date, ho_cause_id, ho_signer_id)
    values (:ho_date, :ho_from_reg_id, :ho_text, :ho_reason, :un_id, :ho_order_type,
            :ho_order_subtype, :ho_order_no, :ho_come_back_date, :ho_cause_id, :ho_signer_id)
    returning ho_id
    into :ho_id;
  end else
  begin
    update hr_orders
    set ho_date = :ho_date,
        ho_from_reg_id = :ho_from_reg_id,
        ho_text = :ho_text,
        ho_reason = :ho_reason,
        ho_from_un_id = :un_id,
        ho_order_type = :ho_order_type,
        ho_order_subtype = :ho_order_subtype,
        ho_order_no = :ho_order_no,
        ho_come_back_date = :ho_come_back_date,
        ho_cause_id = :ho_cause_id,
        ho_signer_id = :ho_signer_id
    where ho_id = :ho_id;
  end

  if (ps_is_mount = 0) then
    ps_mount_koef = 0;

  if (coalesce(ps_id, 0) = 0) then
  begin
    insert into hr_pstatus (ps_from_ho_id, ps_date_begin, ps_period_month, ps_date_end_in_order,
                            ps_from_reg_id, ps_from_dep_id, ps_from_jl_id, ps_is_on_staff, ps_is_working, ps_from_c_id,
                            ps_salary, ps_transfer_office, ps_tabel_no, ps_class, ps_is_mount, ps_is_part,
                            ps_is_timeless, ps_mount_koef, ps_is_work_book, ps_notific_no, ps_notific_address)   -- RFCKG-4655
    values (:ho_id, :ps_date_begin, :ps_period_month, :ps_date_end_in_order, :ps_from_reg_id,
            :ps_from_dep_id, :ps_from_jl_id, :ps_is_on_staff, :ps_is_working, :ps_from_c_id, :ps_salary,
            :ps_transfer_office, :ps_tabel_no, :ps_class, :ps_is_mount, :ps_is_part, :ps_is_timeless, :ps_mount_koef, :ps_is_work_book, :ps_notific_no, :ps_notific_address)
    returning (ps_id)
    into ps_id;
  end else
  begin
    update hr_pstatus
    set ps_date_begin = :ps_date_begin,
        ps_period_month = :ps_period_month,
        ps_date_end_in_order = :ps_date_end_in_order,
        ps_from_reg_id = :ps_from_reg_id,
        ps_from_dep_id = :ps_from_dep_id,
        ps_from_jl_id = :ps_from_jl_id,
        ps_is_on_staff = :ps_is_on_staff,
        ps_is_working = :ps_is_working,
        ps_from_c_id = :ps_from_c_id,
        ps_transfer_office = :ps_transfer_office,
        ps_tabel_no = :ps_tabel_no,
        ps_class = :ps_class,
        ps_is_mount = :ps_is_mount,
        ps_is_part = :ps_is_part,
        ps_is_timeless = :ps_is_timeless,
        ps_mount_koef = coalesce(:ps_mount_koef, 0),
        ps_is_work_book = :ps_is_work_book,
        ps_notific_no = :ps_notific_no,
        ps_notific_address = :ps_notific_address
    where ps_from_ho_id = :ho_id and
          ps_id = :ps_id;
  end

  if (coalesce(ps_id, 0) <> 0) then
  begin
    update or insert into hr_pstatus_agreement (psa_from_ho_id, psa_from_ps_id, psa_agreement_no)
    values (:ho_id, :ps_id, :psa_agreement_no)
    matching (psa_from_ho_id, psa_from_ps_id);

    if (ho_order_type = 1) then
    begin
      mount_ps_id = ps_id;
      mount_ho_id = ho_id;
    end else
    begin
      select hm_ps_from_ho_id, hm.hm_from_ps_id
      from hr_pstatus ps
      inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id and
            ho.ho_order_type = 1 and
            ho.ho_status = 1
      left join hr_mount hm on hm.hm_ps_from_ho_id = ps.ps_from_ho_id and
            hm.hm_from_ps_id = ps.ps_id
      where ps.ps_from_c_id = :ps_from_c_id and
            ps.ps_status = 1 and
            ps.ps_date_begin <= :ho_date
      order by ps.ps_date_begin desc
      rows 1
      into :mount_ho_id, mount_ps_id;
    end

    if (mount_ps_id is not null) then
    begin
      orig_hm_mount = 0;

      select hm_mount
      from hr_mount
      where hm_from_ps_id = :mount_ps_id and
            hm_ps_from_ho_id = :mount_ho_id
      into orig_hm_mount;

      if (row_count = 1) then
      begin
        if (orig_hm_mount <> hm_mount) then
        begin
          update hr_mount
          set hm_mount = :hm_mount, 
              hm_datetime = current_timestamp,
              hm_from_un_id = :un_id
          where hm_from_ps_id = :mount_ps_id and
                hm_ps_from_ho_id = :mount_ho_id;
        end
      end else
      begin
        if (hm_mount <> 0 or ho_order_type = 1) then
        begin
          insert into hr_mount (hm_from_ps_id, hm_ps_from_ho_id, hm_mount, hm_datetime, hm_from_un_id)
          values (:mount_ps_id, :mount_ho_id, :hm_mount, current_timestamp, :un_id);
        end
      end
    end
  end
end
