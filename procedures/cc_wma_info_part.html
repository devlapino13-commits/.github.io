CREATE PROCEDURE CC_WMA_INFO_PART(
    WMA_ID INTEGER,
    FIO STR_255,
    BIRTH_DATE DATE,
    PHONE_NO STR_100,
    FIND_BY_WMA_ID SMALLINT = 0)
RETURNS (
    TXT D_BLOB_TEXT)
AS
declare variable c_id id_type;
declare variable res_comment str_255;
declare variable c_name_fio str_100;
declare variable c_birth_date str_10;
declare variable anketa_memo d_str_8100;
declare variable un_name str_100;
declare variable wmat_datetime timestamp;
declare variable fio_tmp str_255;
declare variable birth_date_tmp date;
declare variable phone_no_tmp str_100;
declare variable wma_c_id id_type;
begin
  --  #FF9882  #D1FFE3   #FFDDCC
  txt = '';

  txt = BLOB_APPEND(txt,'ID заявки: <b>'||wma_id||'</b><br>');

  if (find_by_wma_id = 1) then
  begin
      select wmt.wmat_name,
             wmt.wmat_birthday,
             wma.wma_phone_no,
             wma.wma_from_c_id
      from cc_website_miniapp_tmp wmt
      inner join cc_website_miniapp wma on wma.wma_last_wmat_id = wmt.wmat_id
      where wma.wma_id = :wma_id
      into :fio_tmp,
           :birth_date_tmp,
           :phone_no_tmp,
           :wma_c_id;

      fio = fio_tmp;
      birth_date = birth_date_tmp;
      phone_no = phone_no_tmp;
  end

  for select coalesce(wmc.wmc_memo,''),
             iif(un.un_name is null, '[Не указан]', un.un_name),
             wmc.wmc_datetime
  from cc_website_miniapp_close wmc
  left join username un on wmc.wmc_un_id = un.un_id
  where wmc.wmc_from_wma_id = :wma_id
  and wmc.wmc_memo <> ''
  into :anketa_memo, :un_name, :wmat_datetime
  do
  begin
    txt = BLOB_APPEND(txt, '<div style="border: thin solid; padding: 5px;">
                <div style="background-color: #D489E8">
                  ' || :un_name || ' ' || f_datetimetostr(wmat_datetime) || '
                </div>
                <div>
                    ' || :anketa_memo || '
                </div>
                </div>
                <br/>');
  end

  for select wmt.wmat_anketa_memo,
             iif(un.un_name is null, '[Не указан]', un.un_name),
             wmt.wmat_datetime
  from cc_website_miniapp_tmp wmt
  left join username un on wmt.wmat_from_un_id = un.un_id
  where wmt.wmat_from_wma_id = :wma_id
  and wmt.wmat_anketa_memo <> ''
  order by wmt.wmat_no desc
  into :anketa_memo, :un_name, :wmat_datetime
  do
  begin
    txt = BLOB_APPEND(txt, '<div style="border: thin solid; padding: 5px;">
                <div style="background-color: #EAF4FF">
                  ' || :un_name || ' ' || f_datetimetostr(wmat_datetime) || '
                </div>
                <div>
                    ' || :anketa_memo || '
                </div>
                </div>
                <br/>');
  end

  if (wma_c_id is null) then
  begin
      for select c_id, res_comment, c_name_fio, f_datetostr(c_birth_date) c_birth_date
          from hp_find_clients(null, :fio, :birth_date, null, null, null, :phone_no, null, null)
          where is_critical = 1
          into :c_id, :res_comment, :c_name_fio, :c_birth_date
      do
      begin
        txt = BLOB_APPEND(txt,'<div style="border: thin solid; padding: 5px;">');
        txt = BLOB_APPEND(txt,'<div style="background-color: #FFDDCC">');
        txt = BLOB_APPEND(txt,'<b>Совпадение с лицом</b><br><br>');
        txt = BLOB_APPEND(txt,''||res_comment||'<br><br>');
        txt = BLOB_APPEND(txt,'ФИО: <b>'||c_name_fio||'</b><br>');
        txt = BLOB_APPEND(txt,'Дата рождения: <b>'||c_birth_date||'</b><br>');
    
        txt = BLOB_APPEND(txt,(select txt from cc_wma_info_part_tmp(:c_id)));
    
        txt = BLOB_APPEND(txt,'</div>');
        txt = BLOB_APPEND(txt,'</div>');
      end
  end
  else
  begin
    c_id = wma_c_id;
    select cl.c_name_fio, cl.c_birth_date
    from clients cl
    where cl.c_id = :c_id
    into :c_name_fio, :c_birth_date;

    txt = BLOB_APPEND(txt,'<div style="border: thin solid; padding: 5px;">');
    txt = BLOB_APPEND(txt,'<div style="background-color: #D1FFE3">');
    txt = BLOB_APPEND(txt,'<b>Информация по лицу из заявки</b><br><br>');
    txt = BLOB_APPEND(txt,'ФИО: <b>'||c_name_fio||'</b><br>');
    txt = BLOB_APPEND(txt,'Дата рождения: <b>'||c_birth_date||'</b><br>');

    txt = BLOB_APPEND(txt,(select txt from cc_wma_info_part_tmp(:c_id)));

    txt = BLOB_APPEND(txt,'</div>');
    txt = BLOB_APPEND(txt,'</div>');
  end

  suspend;
end
