CREATE PROCEDURE UXP$UXUNITFRAME_SET_ITEM(
    UNI_UUID D_UUID,
    UNF_UUID D_UUID)
AS
declare variable unt_uuid d_uuid;
declare variable tmp_uuid d_uuid;
declare variable is_root d_bool;
declare variable uni_kind d_smallint;
declare variable uni_control d_str_code;
begin
  select uni_unt_uuid, uni_kind, uni_control, iif(uni_parent_uuid is null, 1, 0)
  from ux$unititem
  where uni_uuid = :uni_uuid
  into unt_uuid, uni_kind, uni_control, is_root;

  if (unt_uuid is null) then
    exception e_default 'Ветка структуры не найдена';

  update ux$unitframe
  set unf_uni_uuid = null
  where unf_uni_uuid = :uni_uuid;

  if (unf_uuid is not null) then
  begin
    if (is_root = 1) then
      exception e_message 'Нельзя прикреплять в фрейм на корневую ветку, это форма.';

    if (uni_kind <> -1 or uni_control <> 'group') then
      exception e_message 'Фрейм можно прикреплять только к контейнеру (группе)';

    tmp_uuid = uni_uuid;
    while (tmp_uuid is not null) do
    begin
      select uni_parent_uuid
      from ux$unititem
      where uni_uuid = :tmp_uuid
      into tmp_uuid;

      if (exists(select * from ux$unitframe
      where unf_uni_uuid = :tmp_uuid)) then
        exception e_message 'Фрейм не должен быть внутри другого фрейма, имеется прикрепленный фрейм в родительских ветках';
    end

    for select unf_uni_uuid
    from ux$unitframe
    where unf_owner = :unt_uuid and
          unf_uni_uuid is not null
    into tmp_uuid
    do
    begin
      while (tmp_uuid is not null) do
      begin
        select uni_parent_uuid
        from ux$unititem
        where uni_uuid = :tmp_uuid
        into tmp_uuid;

        if (tmp_uuid = uni_uuid) then
          exception e_message 'Фрейм не должен быть внутри другого фрейма, имеется прикрепленный фрейм в дочерних ветках';
      end
    end

    update ux$unitframe
    set unf_uni_uuid = :uni_uuid
    where unf_uuid = :unf_uuid;
  end
end
