CREATE PROCEDURE ELCARD_TRANSFER_STATUS_RESULT(
    ESM_ID D_ID,
    REQUEST D_STR_8100,
    RESPONSE D_STR_8100)
AS
declare variable PAYMENT_STATUS D_STR_CODE;
declare variable C_ID D_ID;
declare variable COMMIS_AMOUNT D_CURRENCY;
declare variable K_ID D_ID;
declare variable ESM_RESULT_ID D_INTEGER;
declare variable ESM_ACCOUNT D_STR_CODE;
begin
    -- процедура вызывается для присвоения статуса перевода денег при выдачи займа на сторониий Элкарт
    if (coalesce(response, '') = '') then
        exit;

    payment_status = f_json_str_get_value(:response, 'paymentStatus', '');

    if (payment_status = 'success') then
    begin
        update extsysmain em
        set em.esm_full_source = :request,
            em.esm_status = 1
        where em.esm_id = :esm_id
        and em.esm_project = 'elcard_disb' and
            em.esm_action = 'check_transfer_status';
    
        insert into extsysanswer (esa_from_esm_id, esa_message)
        values (:esm_id, :response);

        select em.esm_object_id,
               round((em.esm_amount * 0.3), 2),
               em.esm_from_k_id,
               em.esm_result_id,
               esm_account
        from extsysmain em
        where em.esm_id = :esm_id
        into :c_id,
             :commis_amount,
             :k_id,
             :esm_result_id,
             :esm_account;

        -- формирую задачу на посторение бух проводок
        insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
                                esm_full_source, esm_amount, esm_from_k_id, esm_account, esm_result_id)
        values ('elcard_disb', 'operbookcreate', current_date, 0, :c_id, 0, '', :commis_amount, :k_id, :esm_account, :esm_result_id);
    end
end
