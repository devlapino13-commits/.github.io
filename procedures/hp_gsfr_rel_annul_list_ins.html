CREATE PROCEDURE HP_GSFR_REL_ANNUL_LIST_INS(
    C_ID D_ID,
    OTHER_C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255)
AS
begin
  -- создает новую запись в списке аннулированных родственных связей клиента ГСФР (RFCKG-2614)

  res_info_text = '';
  res_error_text = '';

  if (not exists(select *
                 from username
                 where un_id = :un_id)) then
    res_error_text = 'Ошибка: пользователь не найден';

  if (not exists(select *
                 from clients
                 where c_id = :other_c_id)) then
    res_error_text = 'Ошибка: родственник клиента не найден';

  if (not exists(select *
                 from clients
                 where c_id = :c_id)) then
    res_error_text = 'Ошибка: клиент не найден';

  if (not exists(select *
                 from clientrelation crel
                 where crel_from_c_id = :c_id and
                       crel_other_c_id = :other_c_id) and
      not exists(select *
                 from clientrelation crel
                 where crel.crel_from_c_id = :other_c_id and
                       crel.crel_other_c_id = :c_id)) then
    res_error_text = 'Ошибка: выбранные клиенты/контрагенты не являются родственниками';

  if (res_error_text = '') then
  begin
    insert into gsfr_rel_annul_list (gral_from_c_id,
                                     gral_other_c_id,
                                     gral_from_un_id)
    values (:c_id, :other_c_id, :un_id);

    if (row_count = 1) then
      res_info_text = 'Новая запись добавлена (родственная связь аннулирована)';
    else
      res_error_text = 'Ошибка: добавить новую запись не удалось';
  end

  suspend;
end
