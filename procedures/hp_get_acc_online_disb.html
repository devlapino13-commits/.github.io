CREATE PROCEDURE HP_GET_ACC_ONLINE_DISB(
    DISB_KIND D_INTEGER,
    PR_CODE D_STR_100,
    REG_ID D_INTEGER)
RETURNS (
    ACC_DT D_STR_20,
    ACC_KT D_STR_20,
    RES_TEXT D_STR_100)
AS
declare variable AL_ACCOUNT_KT D_STR_10;
declare variable AL_ACCOUNT_DT D_STR_10;
begin
  --Процедура возвращает дебет и кредит счета для выдач НЕ через кассу
  res_text = '';

  if (pr_code containing '.online_loan.') then
  begin
    al_account_kt = iif(disb_kind = 5, trim('10004/2'), trim('20001/1'));
    al_account_dt = iif(disb_kind = 5, trim('10941/11'), trim('10941/13'));

    acc_kt = (select al_id
              from acc_list
              where al_from_reg_id = iif(:disb_kind = 5, 1090100, :reg_id) and
                    al_currency = '417' and
                    al_account = :al_account_kt);

    acc_dt = (select al_id
              from acc_list
              where al_from_reg_id = :reg_id and
                    al_currency = '417' and
                    al_account = :al_account_dt);
  end
  else
  begin
    al_account_kt = '21199/199';
    al_account_dt = '10941/12';

    if (pr_code containing '.tms.') then
    begin
      al_account_kt = '21199/99';
      al_account_dt = '10941/12';
    end 

    acc_kt = (select al_id
              from acc_list
              where al_from_reg_id = :reg_id and
                    al_currency = 417 and
                    al_account = :al_account_kt);

    acc_dt = (select al_id
              from acc_list
              where al_from_reg_id = :reg_id and
                    al_currency = 417 and
                    al_account = :al_account_dt);
  end

  if (coalesce(acc_kt, '') = '') then
    res_text = 'Счета КТ не определен';
  else
  if (coalesce(acc_dt, '') = '') then
    res_text = 'Счет ДТ не определён';

  suspend;
end
