CREATE PROCEDURE HP_CHECK_USER_PASSWORD(
    UN_LOGIN D_STR_255,
    UN_PASS_HASH D_STR_255,
    UN_IS_LOGIN D_SMALLINT,
    UN_SYSTEM_NAME D_STR_100)
RETURNS (
    UN_ID D_ID,
    UN_MESSAGE D_STR_255,
    UN_CAN_WORK D_INTEGER,
    UN_READONLY D_BOOL)
AS
declare variable last_date date; /* UN_SYSTEM_NAME */
declare variable kol_days integer;
declare variable last_stop_date date;
declare variable day_str varchar(10);
declare variable un_password_new varchar(1024);
declare variable pa_id integer;
declare variable pm_date_end date;
declare variable pass_max_delinq_day integer;
begin
-- Результаты:
-- UN_CAN_WORK = -3   требуется восстановление пароля, без этого не может работать
-- UN_CAN_WORK = -2   требуется смена пароля, без этого не может работать
-- UN_CAN_WORK = -1   не может работать дальше
-- UN_CAN_WORK = 0    пароль неверный, повторная попытка
-- UN_CAN_WORK = 1    все ок, пускаем в систему

-- если UN_Message<>'' то надо будет отобразить сообщение

  pass_max_delinq_day = -30;

  un_id = null;
  un_can_work = -1;
  un_message = '';
  un_readonly = 0;

  select pa_id
  from permactions
  where pa_command_name = :un_system_name and
        pa_parent_id is null
  into :pa_id;

  if (pa_id is null) then
  begin
    un_can_work = -1;
    un_message = 'Система пользователя не определена';
    suspend;
    exit;
  end

  if (exists(select *
             from company
             where cm_is_monopol = 1)) then
  begin
    un_can_work = -1;
    un_message = 'Введутся плановые работы, база недоступна для пользователей. Попробуйте подключиться позже.';
    suspend;
    exit;
  end

  select un_id, un_password_change, un_password_new, un_readonly
  from username
  where un_name = :un_login rows 1
  into :un_id, :last_date, :un_password_new, :un_readonly;
  if (un_id is null) then
    un_id = 0;

  if (un_is_login = 0) then -- Если проверка только соответствия пароля:
  begin
    un_can_work = 1; -- Представим что все верно
    un_message = '';

    if ((un_id = 0) or (coalesce(un_password_new, '') <> coalesce(:un_pass_hash, ''))) then -- пароль неверный
    begin
      un_can_work = -1; -- Проверяться только правильность пароля при смене пароля, поэтому говорим уже нет
      un_message = 'Текущий пароль указан неверно.';
    end
    suspend;
    exit;
  end

  if ((un_id = 0) or (coalesce(un_password_new, '') <> coalesce(:un_pass_hash, ''))) then
  begin
-- неверный пароль с меткой для повторного запроса пароля.
-- Если система три раза получил этот ответ, можно закрывать приложение отказом на вход
    un_can_work = 0;
    un_message = 'Логин или пароль неверный.';
    suspend;
    exit;
  end

  if (un_login = 'fmtest') then
  begin
    un_can_work = 1;
  end
  else
  begin
  if (last_date is null) then
    last_date = dateadd(year, -1, current_date);

  select max(pm_date_end)
  from permissions
  where pm_subject_kind = 1 -- :permactions
        and
        pm_subject_id = :pa_id and
        pm_is_delete = 0 and
        pm_object_kind = 0 -- :username
        and
        pm_object_id = :un_id and
        pm_access_kind = -1 and
        (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
        (coalesce(pm_date_end, '31.12.2999') >= current_date)
  into :pm_date_end;

  if (pm_date_end is not null) then
  begin
    un_can_work = -1; -- закрытие программы
    un_message = 'Ваш доступ заблокирован до ' || pm_date_end;
    suspend;
    exit;
  end

  if (not(exists(select *
                 from hp_permission_access_check(:un_id, 'PermActions', :pa_id, '')
                 where has_access = 1))) then
  begin
    un_can_work = 0; -- закрытие программы
    un_message = 'У вас нет доступа к данной системе.';
    suspend;
    exit;
  end

  if (un_system_name = 'FsClient.Login') then -- если это клиентское приложение, не проверяем соответствие пароля к требованиям
  begin
    suspend;
    exit;
  end

  if (not(exists(select *
                 from permissions
                 where pm_subject_kind = 4 -- :region
                       and
                       pm_object_kind = 0 -- :username
                       and
                       pm_object_id = :un_id and
                       pm_is_delete = 0 and
                       (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                       (coalesce(pm_date_end, '31.12.2999') >= current_date)))) then
  begin
    un_can_work = -1; -- закрытие программы
    un_message = 'У Вас нет доступных офисов для работы.';
    un_message = un_message || ascii_char(13) || ascii_char(10) || 'Обратитесь к тех.поддержке';
    suspend;
    exit;
  end

  last_stop_date = dateadd(month, 1, last_date);
  last_stop_date = last_stop_date + 5;
  kol_days = datediff(day, current_date, last_stop_date);

  if (kol_days <= pass_max_delinq_day) then
  begin
    un_can_work = -3;
    un_message = 'Срок действия пароля истек,' || ascii_char(13) || ascii_char(10) || 'Вам необходимо восстановить пароль.';
    suspend;
    exit;
  end

-- Обновляем дату/время последнего входа, в случае конфликта обновления "съедаем" исключение
  begin
    update or insert into userlogin (ul_from_un_id, ul_system_name, ul_datetime)
    values (:un_id, :un_system_name, current_timestamp)
    matching (ul_from_un_id, ul_system_name);
    when any do
    begin
    end
  end

  if (kol_days > 0) then
  begin
    if (coalesce(:un_pass_hash, '') in ('C4CA4238A0B923820DCC509A6F75849B', -- MD5 из "1"
                                    'D41D8CD98F00B204E9800998ECF8427E')) then -- MD5 из ""
    begin
      un_can_work = -2;
      un_message = 'Пароль не соответствует требованиям.' || ascii_char(13) || ascii_char(10) || 'Вам необходимо сменить пароль.';
      suspend;
      exit;
    end
  end

  if (kol_days > 5) then -- все ок, пропускаем
  begin
    un_can_work = 1;
    suspend;
    exit;
  end

  if (kol_days between 1 and 5) then
  begin
    un_can_work = 1;
    day_str = ' дней.';
    if (kol_days = 1) then
      day_str = ' день.';
    if (kol_days in (1, 2, 3)) then
      day_str = ' дня.';
    if (kol_days = 5) then
      day_str = ' дней.';

    un_message = 'Срок действия пароля истекает через ' || kol_days || day_str || ascii_char(13) || ascii_char(10) || 'Вам необходимо сменить пароль';

    if (un_system_name = 'FinStart.Login') then
      un_message = un_message || ascii_char(13) || ascii_char(10) || 'через меню: Настройки - Смена пароля';
    else
      un_message = un_message || ascii_char(13) || ascii_char(10) || 'через кнопку "Смена пароля"';

    suspend;
    exit;
  end
  if (kol_days between(pass_max_delinq_day + 1) and 0) then
  begin
    un_can_work = -2;
    un_message = 'Срок действия пароля истек.' || ascii_char(13) || ascii_char(10) || 'Вам необходимо сменить пароль.';
    suspend;
    exit;
  end
  end
  suspend;
end
