CREATE PROCEDURE HP_GET_EXPRESSAPP_STAT_ALL(
    REP_DATE D_DATE)
RETURNS (
    EXC_ID D_INTEGER,
    EXC_NAME D_STR_100,
    RES_ACTUAL_1 D_INTEGER,
    RES_CANCEL_1 D_INTEGER,
    RES_CLIENT_CANCEL_1 D_INTEGER,
    RES_ACTUAL_2 D_INTEGER,
    RES_CANCEL_2 D_INTEGER,
    RES_CLIENT_CANCEL_2 D_INTEGER,
    RES_ACTUAL_3 D_INTEGER,
    RES_CANCEL_3 D_INTEGER,
    RES_CLIENT_CANCEL_3 D_INTEGER)
AS
begin
  for select exc_id, exc_name
  from expresscfg
  order by exc_id
  into exc_id, exc_name
  do
  begin
    res_actual_1 = 0;
    res_cancel_1 = 0;
    res_client_cancel_1 = 0;
    res_actual_2 = 0;
    res_cancel_2 = 0;
    res_client_cancel_2 = 0;
    res_actual_3 = 0;
    res_cancel_3 = 0;
    res_client_cancel_3 = 0;

    select
           sum(iif(exa_disb_date = :rep_date and exa_date = :rep_date and exa_final_status >= 0, 1, 0)),
           sum(iif(exa_disb_date = :rep_date and exa_date = :rep_date and exa_final_status = -1, 1, 0)),
           sum(iif(exa_disb_date = :rep_date and exa_date = :rep_date and exa_final_status = -2, 1, 0)),

           sum(iif(exa_disb_date = :rep_date and exa_final_status >= 0, 1, 0)),
           sum(iif(exa_disb_date = :rep_date and exa_final_status = -1, 1, 0)),
           sum(iif(exa_disb_date = :rep_date and exa_final_status = -2, 1, 0)),

           sum(iif(exa_final_status >= 0, 1, 0)),
           sum(iif(exa_final_status = -1, 1, 0)),
           sum(iif(exa_final_status = -2, 1, 0))
    from expressapp
    where exa_from_exc_id = :exc_id and
          exa_disb_date <= :rep_date
    into res_actual_1, res_cancel_1, res_client_cancel_1,
         res_actual_2, res_cancel_2, res_client_cancel_2,
         res_actual_3, res_cancel_3, res_client_cancel_3;

    if (res_actual_1 is null) then
      res_actual_1 = 0;

    if (res_cancel_1 is null) then
      res_cancel_1 = 0;

    if (res_client_cancel_1 is null) then
      res_client_cancel_1 = 0;

    if (res_actual_2 is null) then
      res_actual_2 = 0;

    if (res_cancel_2 is null) then
      res_cancel_2 = 0;

    if (res_client_cancel_2 is null) then
      res_client_cancel_2 = 0;

    if (res_actual_3 is null) then
      res_actual_3 = 0;

    if (res_cancel_3 is null) then
      res_cancel_3 = 0;

    if (res_client_cancel_3 is null) then
      res_client_cancel_3 = 0;

    suspend;
  end
end
