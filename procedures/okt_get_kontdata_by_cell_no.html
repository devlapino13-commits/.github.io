CREATE PROCEDURE OKT_GET_KONTDATA_BY_CELL_NO(
    CELL_NO VARCHAR(20))
RETURNS (
    RES_HTML_TXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable c_id id_type;
declare variable c_txt varchar(100);
declare variable c_name_fio varchar(100);
declare variable k_kontract_no varchar(50);
declare variable ks_state_txt varchar(100);
declare variable clt_id d_integer;
begin
  res_html_txt = '';

  if (f_get_const('new_contdata') = 0) then -- если работаем по старой структуре. После переключения удалить код ниже.
  begin
    for select ci.cci_object_id, ''
        from continfo ci
        where ((ci.ok_cell_main = :cell_no) or
              (ci.last_cell_main = :cell_no)) and
              ci.cci_object_kind = 0
        union
        select distinct ct.cc_from_c_id,
                        'контактное лицо (' || cc_name || ') клиента (' || c_name_fio || ')'
        from contacts ct
        inner join clients cl on cl.c_id = ct.cc_from_c_id
        where ct.cc_cell_no = :cell_no
              -- and CT.CC_CHECK_RESULT = 1
        into c_id, c_name_fio
    do
    begin
      k_kontract_no = '';
      c_txt = '';
      if (c_name_fio = '') then
        select c_name_fio
        from clients
        where c_id = :c_id
        into c_name_fio;

      for select k_kontract_no,
                 case ks_state
                   when -1 then 'аннулирован'
                   when 0 then 'в процессе'
                   when 1 then 'Активный'
                   when 2 then 'Закрытый'
                 end,
                 'Клиент'
          from knclient kc, kontract, kontstate
          where kc_from_c_id = :c_id and
                kc_amount_approved > 0 and
                kc_from_k_id = k_id and
                kc_from_k_id = ks_from_k_id and
                current_date between ks_date_begin and ks_date_end
          union
          select k_kontract_no,
                 case ks_state
                   when -1 then 'аннулирован'
                   when 0 then 'в процессе'
                   when 1 then 'Активный'
                   when 2 then 'Закрытый'
                 end,
                 'Поручитель'
          from knclient kc, kontract, kontstate, zalog z
          where kc_from_c_id = :c_id and
                kc_amount_approved > 0 and
                kc_from_k_id = k_id and
                kc_from_k_id = ks_from_k_id and
                current_date between ks_date_begin and ks_date_end and
                kc_from_k_id = z.z_from_k_id

          into k_kontract_no, ks_state_txt, c_txt
      do
      begin
        if (c_name_fio not starting 'контактное' and
            c_name_fio not starting 'Клиент') then
          c_name_fio = c_txt || ' - (' || c_name_fio || ')';

        res_html_txt = coalesce(res_html_txt, '') || '<br>' || k_kontract_no || ' (' || trim(ks_state_txt) || ') - ' || c_name_fio;
      end
    end
  end
  else
  begin
    clt_id = f_cont_id(cell_no, 'null');

    for select cda_from_c_id, ''
        from contdata ci
        where cda_from_clt_id = :clt_id and
              cda_cont_kind = f_enum('cont_kind', 'cell', 'id') and
              cda_status = 1 and
              cda_is_delete = 0
        union
        select distinct ct.cc_from_c_id,
                        'контактное лицо (' || cc_name || ') клиента (' || c_name_fio || ')'
        from contacts ct
        inner join clients cl on cl.c_id = ct.cc_from_c_id
        where ct.cc_cell_no = :cell_no
        into c_id, c_name_fio
    do
    begin
      k_kontract_no = '';
      c_txt = '';
      if (c_name_fio = '') then
        select c_name_fio
        from clients
        where c_id = :c_id
        into c_name_fio;

      for select k_kontract_no,
                 case ks_state
                   when -1 then 'аннулирован'
                   when 0 then 'в процессе'
                   when 1 then 'Активный'
                   when 2 then 'Закрытый'
                 end,
                 'Клиент'
          from knclient kc, kontract, kontstate
          where kc_from_c_id = :c_id and
                kc_amount_approved > 0 and
                kc_from_k_id = k_id and
                kc_from_k_id = ks_from_k_id and
                current_date between ks_date_begin and ks_date_end
          union
          select k_kontract_no,
                 case ks_state
                   when -1 then 'аннулирован'
                   when 0 then 'в процессе'
                   when 1 then 'Активный'
                   when 2 then 'Закрытый'
                 end,
                 'Поручитель'
          from knclient kc, kontract, kontstate, zalog z
          where kc_from_c_id = :c_id and
                kc_amount_approved > 0 and
                kc_from_k_id = k_id and
                kc_from_k_id = ks_from_k_id and
                current_date between ks_date_begin and ks_date_end and
                kc_from_k_id = z.z_from_k_id

          into k_kontract_no, ks_state_txt, c_txt
      do
      begin
        if (c_name_fio not starting 'контактное' and
            c_name_fio not starting 'Клиент') then
          c_name_fio = c_txt || ' - (' || c_name_fio || ')';

        res_html_txt = coalesce(res_html_txt, '') || '<br>' || k_kontract_no || ' (' || trim(ks_state_txt) || ') - ' || c_name_fio;
      end
    end
  end

  if (res_html_txt = '') then
    res_html_txt = 'Не найден';
  suspend;
end
