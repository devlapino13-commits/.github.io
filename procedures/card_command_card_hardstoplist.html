CREATE PROCEDURE CARD_COMMAND_CARD_HARDSTOPLIST(
    OBJECT_ID D_ID)
AS
declare variable object_kind_str d_str_100;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable cs_request_json d_str_8100;
declare variable cs_id d_integer;
declare variable cbo_region_id d_integer;
declare variable un_id d_id;
declare variable cbo_id d_integer;
declare variable cbr_reason d_str_255;
declare variable ipc_code d_str_10;
begin
  /*
       Процедура отвечает за формирование операции для МПЦ
       "Поместить в жесткий стоп-лист"
  */
  object_kind_str = f_objkind('cardhardstoplist', 'name');

  select chl.chl_card_id,
         cbo.cbo_region_id,
         chl.chl_from_un_id,
         cbo.cbo_id,
         cbr.cbr_reason,
         cbr.cbr_ipc_code
  from cardhardstoplist chl
  inner join cardblockopers cbo on chl.chl_from_cbo_id = cbo.cbo_id
  inner join cardblockreasons cbr on cbr.cbr_id = cbo.cbo_reason
  where chl.chl_id = :object_id
  into :card_id,
       :cbo_region_id,
       :un_id,
       :cbo_id,
       :cbr_reason,
       :ipc_code;

  for select cal_from_cca_id
      from cardlist cal
      where cal.cal_card_id = :card_id
        and cal_status = f_enum('cardlist_status', 'stoplist_in', 'id')
      into :cca_id
  do
  begin
    cs_request_json = f_format('{"card_id":%1, "reason_comment": "%2", "reason_code": "%3", "bank_c": 21, "groupc": "01"}', :card_id, :cbr_reason, :ipc_code);

    insert into cardsync(
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_request_json,
      cs_office_id,
      cs_from_un_id
      )
    values(
      f_get_cardsynccfg('card_hard_stoplist_in', 'id'),
      :card_id,
      :cca_id,
      f_objkind(:object_kind_str, 'id'),
      :object_id,
      :cs_request_json,
      :cbo_region_id,
      :un_id
      )
    returning cs_id
    into :cs_id;
  
    update cardhardstoplist chl
    set chl.chl_from_cs_id = :cs_id,
        chl.chl_is_robot = 1
    where chl.chl_id = :object_id;
  end
end
