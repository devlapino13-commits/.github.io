CREATE PROCEDURE HP_PRINT_SALARY(
    C_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_INN D_STR_20,
    C_NAME_FIO D_STR_100,
    BIN D_STR_CODE,
    PAYMENT_DATE D_STR_CODE,
    PAYMENT_NAME D_STR_255,
    PAYMENT_SUM D_CURRENCY,
    TOTAL_PAYMENT_SUM D_CURRENCY,
    ANSWER_DATE D_DATETIME)
AS
declare variable JSON_DOCUMENT D_BLOB_TEXT;
declare variable I D_INTEGER;
declare variable ARR_LENGTH D_INTEGER;
declare variable BLS_ID D_ID;
declare variable MAC_ID D_ID;
declare variable MA_ID D_ID;
begin
  -- RFCKZ-31
  c_inn = '-';
  c_name_fio = '-';
  bin = '0';
  payment_date = '01.01.1900';
  answer_date = '01.01.1900';
  payment_name = '-';
  payment_sum = 0;
  total_payment_sum = 0;
  i = 0;
  ma_id = add_params;


  select c.c_inn, c.c_name_fio, mac.mac_id
  from miniappclient mac
  inner join clients c on c.c_id = mac.mac_from_c_id
  where mac.mac_from_ma_id = :ma_id and
        mac.mac_from_c_id = :c_id rows 1
  into c_inn, c_name_fio, mac_id;

  select eil.eil_from_bls_id,
         eil.eil_status_datetime
  from income_check_querys icq
  inner join extinfolist eil on eil.eil_id = icq.icq_from_eil_id
  where icq.icq_from_mac_id = :mac_id and
        icq.icq_status = 2 and
        icq.icq_config = 'salary_check'
  order by icq.icq_status_datetime desc rows 1
  into bls_id,
       answer_date;

  bls_id = coalesce(bls_id, 0);

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into json_document;

  if (coalesce(json_document, '') = '') then
  begin
    suspend;
    exit;
  end

  arr_length = f_json_blob_get_count(:json_document, 'baseResponse.deductionDetailed');

  if (arr_length < 0) then  -- Если массив пустой
  begin
  suspend;
  exit;
  end

  while (i < arr_length) do
  begin
    bin = f_json_blob_get_value(:json_document, f_format('baseResponse.deductionDetailed[%1].bin', i), '');
    payment_date = f_datetostr(cast(replace(f_json_blob_get_value(:json_document, f_format('baseResponse.deductionDetailed[%1].date', i), ''), 'T', ' ') as d_datetime));
    payment_name = f_json_blob_get_value(:json_document, f_format('baseResponse.deductionDetailed[%1].name', i), '');
    payment_sum = cast(f_json_blob_get_value(:json_document, f_format('baseResponse.deductionDetailed[%1].paymentSum', i), '') as d_currency);
    total_payment_sum = cast(f_json_blob_get_value(:json_document, f_format('baseResponse.deductionDetailed[%1].TotalpaymentSum', i), '') as d_currency);

    payment_sum = round(payment_sum, 2);
    total_payment_sum = round(total_payment_sum, 2);

    i = i + 1;
    suspend;
  end

end
