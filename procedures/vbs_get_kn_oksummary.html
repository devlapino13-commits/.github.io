    K_KONTRACT_NO VARCHAR(20),
    NEPRERV_PROSROCH D_INTEGER,
    DOPUSHENAYA_PROSROCH D_INTEGER)

declare variable tmp_kc_id d_id;
declare variable tmp_sh_os d_currency;
declare variable tmp_sh_perc d_currency;
declare variable tmp_zad_os d_currency;
declare variable tmp_zad_perc d_currency;
declare variable oks_delinq_date_old d_date;
declare variable oks_date_begin_max d_date;
declare variable i_zap d_integer;
declare variable oks_delinq_date_old_dop d_date;
declare variable k_disb_date d_date;
begin
  if (kc_id is null) then
    kc_id = 0;
  oks_report_date = null;

  if ((select is_kn_mode
       from vbs_is_kn_mode(:k_id)) = 0) then
    exit;
  i_zap = 0;

  neprerv_prosroch = 0;
  select max(oks.oks_date_begin)
  from kn_oksummary oks
  where oks.oks_from_k_id = :k_id and
        (:kc_id = 0 or
        oks.oks_from_kc_id = :kc_id)
  into :oks_date_begin_max;

  for select oks_date_begin, max(oks_date_prv), max(oks_days), max(oks_date_stop_perc), max(oks_percent_norm_prv),
             max(oks_percent_norm), max(oks_percent_esp), sum(0), max(oks_is_sh), max(oks_is_month), max(oks_is_pmt),
             max(oks_is_flat), sum(oks_disb_amount), sum(oks_balance), sum(oks_balance_prv), sum(oks_kommis_amount),
             max(oks_last_sh_date), sum(oks_vbs_prv), sum(oks_vbs_begin), sum(oks_vbs_end), sum(oks_bal_to_perc_prv),
             sum(oks_bal_to_perc), max(oks_delinq_date), max(oks_delinq_days), sum(oks_pmt_os_norm),
             sum(oks_pmt_perc_norm), sum(oks_pmt_kommis), sum(oks_pmt_other_adv), sum(oks_pmt_os_delinq),
             sum(oks_pmt_perc_delinq), sum(oks_pmt_os_adv), sum(oks_pmt_perc_adv), max(oks_pmt_is_adv),
             sum(oks_acr_perc), sum(oks_bal_perc_fixed), sum(oks_bal_perc), sum(oks_bal_perc_prv), sum(oks_acr_kommis),
             sum(oks_bal_kommis_prv), sum(oks_bal_kommis), max(oks_is_reschedule_llr), sum(oks_bal_esp_prv),
             sum(oks_bal_esp_begin), sum(oks_acr_esp_perc), sum(oks_bal_esp_tmp),
             sum(iif(oks_close_state <> 2, oks_bal_esp_end, 0)), sum(oks_pmt_all), sum(oks_acr_kommis_adv),
             sum(oks_diff_esp_perc), sum(oks_diff_esp_bal), sum(oks_bal_zad_os), sum(oks_bal_zad_perc),
             max(oks_delinq_date_prv), sum(oks_bal_esp_norm), sum(oks_bal_esp_delinq), sum(oks_bal_esp_norm_tmp),
             sum(oks_diff_esp_norm), sum(oks_pmt_all_ex_adv), sum(oks_pmt_all_adv), sum(oks_bal_esp_ex_adv),
             max(oks_is_actual), max(iif(oks_is_reschedule_llr = 0, oks_llr_kind, oks_llr_kind + 3)),
             max(oks_llr_percent), max(oks_final_llr_percent), max(oks_ecl_kind), max(oks_ecl_percent),
             max(oks_ecl_period_koef), max(oks_ecl_acc_kind), sum(oks_llr_os), sum(oks_llr_perc), sum(oks_ecl_vbs),
             sum(oks_diff_ecl_llr), sum(oks_off_esp_bal), sum(oks_off_os), sum(oks_off_perc), sum(oks_pmt_os_off),
             sum(oks_pmt_perc_off), sum(oks_acr_law), sum(oks_pmt_law), sum(oks_off_law), sum(oks_bal_law),
             sum(oks_reserv_law), sum(oks_ead_future_vbs), sum(oks_ead_future_perc), min(oks_ead_estimate_delinq_date),
             min(oks_ead_estimate_def_date), sum(oks_ead_amount), sum(oks_rvs_perc)
      from kn_oksummary
      where oks_from_k_id = :k_id and
            (:kc_id = 0 or
            oks_from_kc_id = :kc_id)
      group by oks_date_begin
      order by oks_date_begin
      into :oks_date_begin, :oks_date_prv, :oks_days, :oks_date_stop_perc, :oks_percent_norm_prv, :oks_percent_norm,
           :oks_percent_esp, :oks_kommis_per_day, :oks_is_sh, :oks_is_month, :oks_is_pmt, :oks_is_flat,
           :oks_disb_amount, :oks_balance, :oks_balance_prv, :oks_kommis_amount, :oks_last_sh_date, :oks_vbs_prv,
           :oks_vbs_begin, :oks_vbs_end, :oks_bal_to_perc_prv, :oks_bal_to_perc, :oks_delinq_date, :oks_delinq_days,
           :oks_pmt_os_norm, :oks_pmt_perc_norm, :oks_pmt_kommis, :oks_pmt_other_adv, :oks_pmt_os_delinq,
           :oks_pmt_perc_delinq, :oks_pmt_os_adv, :oks_pmt_perc_adv, :oks_pmt_is_adv, :oks_acr_perc,
           :oks_bal_perc_fixed, :oks_bal_perc, :oks_bal_perc_prv, :oks_acr_kommis, :oks_bal_kommis_prv, :oks_bal_kommis,
           :oks_is_reschedule, :oks_bal_esp_prv, :oks_bal_esp_begin, :oks_acr_esp_perc, :oks_bal_esp_tmp,
           :oks_bal_esp_end, :oks_pmt_all, :oks_acr_kommis_adv, :oks_diff_esp_perc, :oks_diff_esp_bal, :oks_bal_zad_os,
           :oks_bal_zad_perc, :oks_delinq_date_prv, :oks_bal_esp_norm, :oks_bal_esp_delinq, :oks_bal_esp_norm_tmp,
           :oks_diff_esp_norm, :oks_pmt_all_ex_adv, :oks_pmt_all_adv, :oks_bal_esp_ex_adv, :oks_is_actual,
           :oks_llr_kind, :oks_llr_percent, :oks_final_llr_percent, :oks_ecl_kind, :oks_ecl_percent,
           :oks_ecl_period_koef, :oks_ecl_acc_kind, :oks_llr_os, :oks_llr_perc, :oks_ecl_vbs, :oks_diff_ecl_llr,
           :oks_off_esp_bal, :oks_off_os, :oks_off_perc, :oks_pmt_os_off, :oks_pmt_perc_off, :oks_acr_law, :oks_pmt_law,
           :oks_off_law, :oks_bal_law, :oks_reserv_law, :oks_ead_future_vbs, :oks_ead_future_perc,
           :oks_ead_estimate_delinq_date, :oks_ead_estimate_def_date, :oks_ead_amount, :oks_rvs_perc

  do
  begin

    select k_kontract_no, k_disb_date, k_percent_effect_fifd
    from kontract
    where k_id = :k_id
    into k_kontract_no, k_disb_date, oks_effect_rete;
    if (oks_pmt_is_adv > 0) then
      pmt_is_adv = 1;
    else
      pmt_is_adv = 0;

    oks_sh_os = 0;
    oks_sh_perc = 0;
    oks_zad_os = 0;
    oks_zad_perc = 0;

    oks_moved_os = 0;
    if (oks_is_flat = 0) then
      oks_moved_os = oks_balance - oks_bal_to_perc - oks_bal_zad_os;

    for select kc_id
        from knclient
        where ((:kc_id = 0 and
              kc_from_k_id = :k_id) or
              (:kc_id <> 0 and
              kc_id = :kc_id)) and
              kc_amount_approved > 0
        into tmp_kc_id
    do
    begin
      tmp_sh_os = 0;
      tmp_sh_perc = 0;
      tmp_zad_os = 0;
      tmp_zad_perc = 0;

      select sum(ok_sh_os), sum(ok_sh_perc), sum(ok_sh_os + ok_os), sum(ok_sh_perc + ok_perc)
      from kn_operkred
      where ok_from_kc_id = :tmp_kc_id and
            ok_sh_date = :oks_date_begin and
            ok_ol_is_active = 1
      into tmp_sh_os, tmp_sh_perc, tmp_zad_os, tmp_zad_perc;

      oks_sh_os = oks_sh_os + tmp_sh_os;
      oks_sh_perc = oks_sh_perc + tmp_sh_perc;
      oks_zad_os = oks_zad_os + tmp_zad_os;
      oks_zad_perc = oks_zad_perc + tmp_zad_perc;
    end

    oks_calc_plan_all = oks_sh_os + oks_sh_perc;

    if (oks_date_begin > k_disb_date) then
    begin
      select effect_rate
      from hp_effect_rate_to_date(:k_id, :oks_date_begin, 1)
      into :oks_effect_rete;
      when any do
        oks_effect_rete = null;
    end

    perc_calced = oks_acr_perc;
    perc_saldo = oks_bal_perc;
    perc_base = oks_bal_to_perc_prv;
    perc_diff = 0;

    if (kc_id = 0 and
        oks_date_begin <> oks_date_prv) then
    begin
      select result_calced, result_perc_end, result_perc_base
      from aarep_calc_perc_one_cls(:k_id, :oks_date_prv + 1, :oks_date_begin, 0, 1, 1)
      into :perc_calced, :perc_saldo, :perc_base;

      perc_calced = perc_calced - oks_rvs_perc;

      perc_diff = perc_saldo - oks_bal_perc;

    end
    dopushenaya_prosroch = 0;
    neprerv_prosroch = 0;

    if (i_zap > 0) then
    begin
      neprerv_prosroch = datediff(day, oks_delinq_date_old, oks_date_begin);
      if (neprerv_prosroch is null) then
        neprerv_prosroch = 0;
    end

    oks_delinq_date_old = oks_delinq_date;

    if ((i_zap > 0) and
        ((oks_delinq_date is null) or (oks_date_begin_max = oks_date_begin))) then
    begin
      dopushenaya_prosroch = datediff(day, oks_delinq_date_old_dop, oks_date_begin);
    end
    if ((i_zap > 0) and
        (oks_delinq_date is not null) and
        (oks_delinq_date_old_dop is null)) then
      oks_delinq_date_old_dop = oks_delinq_date;
    if ((i_zap > 0) and
        (oks_delinq_date is null)) then
      oks_delinq_date_old_dop = null;

    i_zap = i_zap + 1;

    suspend;
  end
end
