CREATE PROCEDURE UXP$DATATYPE(
    DATA_TYPE D_INTEGER NOT NULL,
    DATA_SIZE D_INTEGER = 0,
    IS_PARAM D_BOOL = 0)
RETURNS (
    RES_FB_DOMAIN D_STR_CODE,
    RES_FB_TYPE D_STR_CODE)
AS
begin
  if (data_size is null) then
    data_size = 0;

  if (is_param = 1) then
  begin
    res_fb_domain = case data_type
      when 0 then ''   -- ftUnknown
      when 1 then 'string'   -- ftString
      when 2 then 'd_smallint'   -- ftSmallint
      when 3 then 'd_integer'   -- ftInteger
      when 4 then 'd_integer'   -- ftWord
      when 5 then 'd_bool'   -- ftBoolean
      when 6 then 'd_float'   -- ftFloat
      when 7 then 'd_currency'   -- ftCurrency
      when 8 then 'd_id'   -- ftBCD
      when 9 then 'd_date'   -- ftDate
      when 10 then 'd_time'   -- ftTime
      when 11 then 'd_datetime'   -- frDateTime
      when 12 then 'd_uuid'   -- frBytes
      when 13 then ''   -- frVarBytes
      when 14 then ''   -- ftAutoInc
      when 15 then 'd_blob_binary'   -- ftBlob
      when 16 then 'string'   -- ftMemo
      when 17 then ''   -- ftGraphic
      when 18 then ''   -- ftFmtMemo
      when 19 then ''   -- ftParadoxOle
      when 20 then ''   -- ftDBaseOle
      when 21 then ''   -- ftTypedBinary
      when 22 then ''   -- ftCursor
      when 23 then ''   -- ftFixedChar
      when 24 then 'string'   -- ftWideString
      when 25 then 'd_id'   -- ftLargeint
      when 26 then ''   -- frADT
      when 27 then ''   -- ftArray
      when 28 then ''   -- ftReference
      when 29 then ''   -- ftDataSet
      when 30 then ''   -- ftOraBlob
      when 31 then ''   -- ftOraClob
      when 32 then ''   -- ftVariant
      when 33 then ''   -- ftInterface
      when 34 then ''   -- ftIDispatch
      when 35 then 'd_uuid'   -- ftGuid
      when 36 then 'd_datetime'   -- ftTimeStamp
      when 37 then ''   -- ftFMTBcd
      when 38 then ''   -- ftFixedWideChar
      when 39 then 'd_blob_text'   -- ftWideMemo
      when 40 then ''   -- ftOraTimeStamp
      when 41 then ''   -- ftOraInterval
      when 42 then 'd_id'   -- ftLongWord
      when 43 then 'd_smallint'   -- ftShortint
      when 44 then 'd_smallint'   -- ftByte
      when 45 then 'd_id'   -- ftExtended
      when 46 then ''   -- ftConnection
      when 47 then ''   -- ftParams
      when 48 then ''   -- ftStream
      when 49 then ''   -- ftTimeStampOffset
      when 50 then ''   -- ftObject
      when 51 then ''   -- ftSingle
      else ''
    end;
  end else
  begin
    res_fb_domain = case data_type
      when 0 then ''
      when 1 then 'd_bool'
      when 2 then 'd_smallint'
      when 3 then 'd_integer'
      when 4 then 'd_integer'
      when 5 then 'd_id'
      when 6 then 'd_smallint'
      when 7 then 'd_integer'
      when 8 then 'd_integer'
      when 9 then 'd_id'
      when 10 then 'd_float'
      when 11 then 'd_double'
      when 12 then 'd_double'
      when 13 then 'd_currency'
      when 14 then 'd_currency'
      when 15 then 'd_currency'
      when 16 then 'd_datetime'
      when 17 then 'd_time'
      when 18 then 'd_date'
      when 19 then 'd_datetime'
      when 20 then ''
      when 21 then ''
      when 22 then ''
      when 23 then 'string'
      when 24 then 'string'
      when 25 then 'd_uuid'
      when 26 then 'd_blob_binary'
      when 27 then 'd_blob_text'
      when 28 then 'd_blob_text'
      when 29 then 'd_blob_text'
      when 30 then ''
      when 31 then ''
      when 32 then ''
      when 33 then ''
      when 34 then ''
      when 35 then ''
      when 36 then ''
      when 37 then ''
      when 38 then ''
      when 39 then 'd_uuid'
      when 40 then ''
      else ''
    end;
  end

  if (res_fb_domain <> '') then
  begin
    res_fb_domain = lower(res_fb_domain);

    res_fb_type = case res_fb_domain
      when 'd_blob_binary' then 'blob'
      when 'd_blob_text' then 'memo'
      when 'd_bool' then 'smallint'
      when 'd_currency' then 'currency'
      when 'd_date' then 'date'
      when 'd_datetime' then 'datetime'
      when 'd_float' then 'float'
      when 'd_id' then 'bigint'
      when 'd_integer' then 'integer'
      when 'd_smallint' then 'smallint'
      when 'd_time' then 'time'
      when 'd_uuid' then 'char'
      when 'string' then 'string'
      else ''
    end;

    if (res_fb_domain = 'string') then
    begin
      if (data_size > 4096) then res_fb_domain = 'd_str_8100';
      else if (data_size > 2048) then res_fb_domain = 'd_str_4096';
      else if (data_size > 1024) then res_fb_domain = 'd_str_2048';
      else if (data_size > 255)  then res_fb_domain = 'd_str_long';
      else if (data_size > 100)  then res_fb_domain = 'd_str_255';
      else if (data_size > 64)   then res_fb_domain = 'd_str_100';
      else if (data_size > 50)   then res_fb_domain = 'd_str_64';
      else if (data_size > 20)   then res_fb_domain = 'd_str_code';
      else if (data_size > 10)   then res_fb_domain = 'd_str_20';
      else if (data_size > 3)    then res_fb_domain = 'd_str_10';
      else if (data_size > 1) then res_fb_domain = 'd_str_lang';
      else if (data_size = 1) then res_fb_domain = 'd_str_sym';
      else res_fb_domain = '';
    end
  end

  if (res_fb_domain = '') then
    exception e_default f_format('Undefined domain for type: %1', data_type);
  else
  if (res_fb_type = '') then
    exception e_default f_format('Undefined data type for fomain: %1', res_fb_domain);

  suspend;
end
