Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UI_UXFRAMES(
    UA_UUID D_UUID,
    VER_KIND D_STR_CODE = null,
    UA_UUID_CODE D_STR_CODE = null)
RETURNS (
    FR_CODE D_STR_CODE,
    FR_CAPTION D_STR_100,
    FR_UUID D_UUID,
    UFR_CAN_UPDATE D_BOOL,
    UFR_CAN_INSERT D_BOOL,
    UFR_CAN_DELETE D_BOOL,
    UFR_KEY_FIELDS D_STR_100,
    UFR_PARAM_SQL D_STR_8100,
    UFR_SELECT_SQL D_STR_8100,
    UFR_UPDATE_SQL D_STR_8100,
    UFR_INSERT_SQL D_STR_8100,
    UFR_DELETE_SQL D_STR_8100,
    UFR_MASTER_FR_UUID D_UUID,
    UFR_MASTER_FR_CODE D_STR_100,
    UFR_MASTER_FIELD D_STR_100,
    UFR_DETAIL_FIELD D_STR_100,
    UFR_IS_MAIN D_BOOL,
    UFR_HAS_EXEC_PROC D_BOOL,
    UFR_IS_GRID D_BOOL,
    UFR_SELECT_IS_EXEC D_BOOL,
    UFR_AUTOINCFIELD D_STR_CODE,
    UFR_ACCESS_UPDATE D_BOOL,
    UFR_ACCESS_INSERT D_BOOL,
    UFR_ACCESS_DELETE D_BOOL,
    UFR_DEFAULT_UA_UUID D_UUID,
    UFR_INPLACE_UUID D_UUID,
    UFR_APPEND_FIELD D_STR_CODE,
    UFR_CLONE_UUID D_UUID,
    UFR_CLONE_KIND D_STR_CODE,
    UFR_TREE_PARENT_FIELD D_STR_CODE,
    UFR_TREE_IMAGE_FIELD D_STR_CODE,
    UFR_REFRESH_PARAMS D_STR_LONG,
    UFR_TREE_MOVE_SQL D_STR_8100,
    UFR_TREE_POSITION_FIELD D_STR_CODE,
    UFR_TREE_CAN_DRAG_FIELD D_STR_CODE,
    UFR_TREE_CAN_DROP_FIELD D_STR_CODE)
AS
declare variable ux_uuid d_uuid;
declare variable main_fr_uuid d_uuid;
declare variable ua_usr_id_field d_str_code;
declare variable uxa_uuid d_uuid;
declare variable ux_kind d_integer;
declare variable ux_kind_code d_str_code;
declare variable ua_action_kind d_str_code;
declare variable uxa_kind d_integer;
begin
  if (coalesce(ua_uuid, '') = '') then
    ua_uuid = uxf$uniq(ua_uuid_code);

  ufr_inplace_uuid = null;

  select uxa_uuid, uxf$ver(ua_uxa_owner, :ver_kind), ua_usr_id_field, ua_uxa_owner_kind, uxa_kind
  from uxp$uxalias_info(:ua_uuid)
  into uxa_uuid, ux_uuid, ua_usr_id_field, ux_kind, uxa_kind;

  ux_kind_code = uxf$kind(ux_kind, 'code');

  if (ux_kind_code = 'uxcommand') then
    ua_action_kind = 'command';
  else
  if (ux_kind_code = 'uxreport') then
    ua_action_kind = 'report';
  else
  begin
    ua_action_kind = trim(case
      when ua_usr_id_field <> '' then 'search'
      when uxa_kind = 2 then 'insert'
      else 'open'
    end);
  end

  select ufr_uuid
  from ux$frame
  where ufr_owner = :ux_uuid and
        ufr_is_mainframe = 1
  into main_fr_uuid;

  for select fr_uuid, fr_code, res_caption, master_fr_uuid, ufr_tree_parent_field, ufr_tree_image_field,
             fr_master_field, fr_detail_field, fr_is_grid, ufr_refresh_params,
             ufr_tree_position_field, ufr_tree_can_drag_field, ufr_tree_can_drop_field
  from uxp$uxframe_tree(:ux_uuid)
  into :fr_uuid, fr_code, fr_caption, ufr_master_fr_uuid, ufr_tree_parent_field, ufr_tree_image_field,
       :ufr_master_field, :ufr_detail_field, :ufr_is_grid, ufr_refresh_params,
       ufr_tree_position_field, ufr_tree_can_drag_field, ufr_tree_can_drop_field
  do
  begin
    ufr_is_main = 0;

    ufr_access_update = 1;
    ufr_access_insert = 1;
    ufr_access_delete = 1;

    ufr_clone_uuid = null;
    ufr_clone_kind = ''; -- clone or link

    fr_caption = uxf$lang(fr_caption);

    if (fr_uuid = main_fr_uuid) then
      ufr_is_main = 1;

    ufr_default_ua_uuid = null;

    if (ua_action_kind in ('open', 'search', 'report')) then
    begin
      select uaa_ua_uuid
      from ux$aliasaction, ux$action
      where uaa_uxa_uuid = :uxa_uuid and
            uaa_is_default = 1 and
            uaa_ua_uuid = ua_uuid and
            ua_owner = :fr_uuid
      rows 1
      into ufr_default_ua_uuid;

      if (ufr_is_main = 1 and ufr_default_ua_uuid is null) then
      begin
        select uaa_ua_uuid
        from ux$aliasaction, ux$action
        where uaa_uxa_uuid = :uxa_uuid and
              uaa_is_default = 1 and
              uaa_ua_uuid = ua_uuid and
              ua_owner = :ux_uuid
        rows 1
        into ufr_default_ua_uuid;
      end
    end

    ufr_append_field = '';
    if (ua_action_kind = 'open') then
    begin
      select uf_field_name || '$text'
      from ux$frparam, ux$field
      where ufp_ufr_uuid = :fr_uuid and
            ufp_uf_uuid = uf_uuid and
            ufp_editor = 'dialogedit' and
            uf_is_param = 0
      order by uf_position
      rows 1
      into ufr_append_field;
    end

    ufr_master_fr_code = '';
    if (ufr_master_fr_uuid is not null) then
      ufr_master_fr_code = uxf$code(ufr_master_fr_uuid, 'fcode');

    ufr_has_exec_proc = 0;
    for select ufr_can_update, ufr_can_insert, ufr_can_delete,
               ufr_key_fields, ufr_param_sql, ufr_select_sql,
               ufr_update_sql, ufr_insert_sql, ufr_delete_sql,
               ufr_has_exec_proc, ufr_select_is_exec, ufr_autoincfield,
               ufr_tree_move_sql
    from uxp$uxframe_sql(:fr_uuid)
    into :ufr_can_update, :ufr_can_insert, :ufr_can_delete,
         :ufr_key_fields, :ufr_param_sql, :ufr_select_sql,
         :ufr_update_sql, :ufr_insert_sql, :ufr_delete_sql,
         :ufr_has_exec_proc, :ufr_select_is_exec, :ufr_autoincfield,
         :ufr_tree_move_sql
    do
      suspend;
  end
end
