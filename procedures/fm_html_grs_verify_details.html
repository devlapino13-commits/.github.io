CREATE PROCEDURE FM_HTML_GRS_VERIFY_DETAILS(
    C_ID DECIMAL(18,0),
    UN_ID D_ID)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable c_verify_status           integer;
declare variable color_black               varchar(100);
declare variable color_yellow              varchar(100);
declare variable color_green               varchar(100);
declare variable color_red                 varchar(100);
declare variable res_color                 varchar(100);
declare variable str_tmp                   varchar(512);
declare variable db_id                     integer;
declare variable acl_id                    decimal(18,0);
declare variable rd_id                     integer;
declare variable rd1_id                    d_integer;
declare variable acm_memo                  varchar(4000);
declare variable str_memo                  varchar(512);
declare variable json_key                  varchar(255);
declare variable json_value                varchar(255);
declare variable need_passport_second_page integer;
declare variable c_doc_type                integer;
declare variable c_citizen                 integer;
declare variable str_comment               varchar(255);
declare variable str_tmp_1                 d_str_100;
declare variable str_tmp_2                 d_str_100;
declare variable acl_status                d_smallint;
declare variable rd_id_ver                 d_integer;
declare variable gc_status                 smallint;
declare variable gc_status_datetime        timestamp;
declare variable uri_text                  d_str_255;
declare variable uri_description           d_str_255;
declare variable gc_from_eil_id            d_id;
declare variable res_text_tmp              d_str_255;
declare variable state                     d_smallint;
declare variable answer                    d_str_8100;
declare variable lang                      d_str_lang;
declare variable kib_name                  d_str_10;
declare variable confirmations_status      d_smallint;
declare variable gdfl_descr                d_str_255;
begin
  db_id = f_get_db_id();
  kib_name = iif(db_id <> 39801, 'ГРС', 'ГБД ФЛ'); -- RFCKG-4140
  lang = f_get_lang_code(un_id, null);
  rd_id = f_get_rd_id('ActionGroup.Passport_Verify');
  rd1_id = f_get_rd_id('ActionGroup.Passport_Verify_OiVO');
  uri_text = '';
  res_text = '';

  confirmations_status = f_confirmations_status(:c_id,  current_date);

  if(db_id = 39801) then
  begin
    select description
    from hp_gdfl_pass_check(:c_id)
    rows 1
    into gdfl_descr;
  end

  select gc.gc_status, gc.gc_status_datetime, gc.gc_from_eil_id
  from grs_clients gc
  where gc.gc_from_c_id = :c_id and
        gc.gc_is_last = 1
        rows 1
  into gc_status, gc_status_datetime, gc_from_eil_id;

  select coalesce(eir.eir_answer, '')
  from extinfoerr eir
  where eir.eir_from_eil_id = :gc_from_eil_id rows 1
  into answer;

  if (gc_status is null) then
    gc_status = 1;

  color_black = 'black';
  color_yellow = '#F08205';
  color_green = '#02B316';
  color_red = 'red';

  res_color = color_black;
  res_color = (case gc_status
                 when -4 then color_red
                 when -3 then color_red
                 when -2 then color_red
                 when -1 then color_red
                 when 0 then color_black
                 when 1 then color_yellow
                 when 2 then color_green
                 when 3 then color_green
               end);

  select res_text, state
  from hp_check_pasport_actual_grs(:c_id)
  into res_text_tmp, state;

  res_text = BLOB_APPEND(res_text, f_format('<br>%1:<br>', 'Проверка клиента в '|| :kib_name));

  str_tmp = iif(answer starting 404, f_lang(lang, 'В ходе верификации возникла ошибка, прошу проверить ИНН и номер паспорта и отправить ещё раз запрос в ' || :kib_name), f_lang(lang,  f_enum_name('grs_status', gc_status)));
  str_tmp = '<font color="' || res_color || '">' || trim(:str_tmp) || '</font><br>';

  if(db_id <> 39801) then
    res_text = BLOB_APPEND(res_text, str_tmp);
  else if (db_id = 39801 and state <> 0) then    -- RFCKG-4140
  begin
    res_text = BLOB_APPEND(res_text, str_tmp);
    if (gdfl_descr <> '') then
      res_text = BLOB_APPEND(res_text, f_format('<div style="color: red"><b>Статус уд. личности: </b>%1</div>', gdfl_descr));
  end

  if (gc_status in (-2, -3)) then
  begin
    uri_description = f_lang(lang, 'Подтверждаете отправку данные клиента повторно в ' || :kib_name || '?');
    uri_text = f_lang(lang, 'Отправить повторно');
  end
  if (gc_status = 1) then
  begin
    uri_description = f_lang(lang, 'Подтверждаете отправку данные клиента в ' || :kib_name || '?');
    uri_text = f_lang(lang, 'Отправить данные клиента для проверки в '|| :kib_name || '?');
  end

  if (state = 0) then
  begin
    if (res_text_tmp <> 'Информация с ' || :kib_name || ' просроченная') then
        res_text = BLOB_APPEND(res_text, :res_text_tmp || '<br>');      
    else
    if (res_text_tmp <> 'Запрос, в процессе обработки') then
    begin
      res_color = iif(db_id = 39801, 'red', :res_color);
      res_text = BLOB_APPEND(res_text,'<font color="' || :res_color || '">' || :res_text_tmp || '</font><br>' || '<br>');
      uri_description = f_lang(lang, 'Подтверждаете отправку данные клиента в ' || :kib_name || '?');
      uri_text = f_lang(lang, 'Отправить данные клиента для проверки в ' || :kib_name);

      if (db_id = 39801) then
      begin
        if(confirmations_status = 1) then
          res_text = BLOB_APPEND(res_text, f_uri_sql(:uri_text, 'send_passport_grs', 'client', :c_id, 'none', uri_description, '', 0, 0, '', '', 'dialog', '') || '<br>');
        else if (confirmations_status = 0) then
          res_text = BLOB_APPEND(:res_text, 'Ожидается ознакомление от клиента<br>');
        else
        begin
          res_text = BLOB_APPEND(:res_text, 'Перед повторной проверкой необходимо ознакомление с согласиями от клиента<br>');
          res_text = BLOB_APPEND(res_text, f_uri_sql('Отправить ознакомление с согласиями', 'send_confirmations', 'client', :c_id, 'none', 'Отправить ознакомление с согласиями?', '', 0, 0, '', '', 'dialog', ''));
        end
      end
      else
        res_text = BLOB_APPEND(res_text, f_uri_sql(:uri_text, 'send_passport_grs', 'client', :c_id, 'none', uri_description, '', 0, 0, '', '', 'dialog', '') || '<br>');
      suspend;
      exit;
    end
  end

  if (uri_text <> '') then
  begin
    if (gc_status = -2 and
        answer not starting 404) then
    begin
      if (not exists(select *
                     from actionorder aco
                     where aco.aco_from_rd_id in (:rd_id, :rd1_id) and
                           aco.aco_object_kind = f_objkind('clients', 'id') and
                           aco.aco_object_id = :c_id)) then
      begin
        res_text = BLOB_APPEND(res_text, f_uri_sql(f_lang(lang, 'Отправить запрос на верификацию'), 'query_verify', 'Clients', :c_id, 'none', f_lang(lang, 'Подтверждаете?'), '', 0, 0, '', '', 'dialog', '') || '<br>');
      end
      else
      if (exists(select *
                 from actionorder aco
                 where aco.aco_from_rd_id in (:rd_id, :rd1_id) and
                       aco.aco_object_kind = f_objkind('clients', 'id') and
                       aco.aco_object_id = :c_id)) then
      begin
        update grs_clients gc
        set gc.gc_is_last = 0
        where gc.gc_from_c_id = :c_id and
              gc.gc_is_last = 1;

        insert into grs_clients (gc_from_c_id, gc_from_eil_id, gc_status, gc_status_datetime, gc_is_last)
        values (:c_id, :gc_from_eil_id, -3, current_timestamp, 1);
      end
    end
    else
    begin
      if (db_id = 39801) then
      begin
        if(confirmations_status = 1) then
          res_text = BLOB_APPEND(res_text, f_uri_sql(:uri_text, 'send_passport_grs', 'client', :c_id, 'none', uri_description, '', 0, 0, '', '', 'dialog', '') || '<br>');
        else if (confirmations_status = 0) then
          res_text = BLOB_APPEND(:res_text, 'Ожидается ознакомление от клиента<br>');
        else
        begin
          res_text = BLOB_APPEND(:res_text, 'Перед повторной проверкой необходимо ознакомление с согласиями от клиента<br>');
          res_text = BLOB_APPEND(res_text, f_uri_sql('Отправить ознакомление с согласиями', 'send_confirmations', 'client', :c_id, 'none', 'Отправить ознакомление с согласиями?', '', 0, 0, '', '', 'dialog', ''));
        end
      end
      else
        res_text = BLOB_APPEND(res_text, f_uri_sql(:uri_text, 'send_passport_grs', 'client', :c_id, 'none', uri_description, '', 0, 0, '', '', 'dialog', '') || '<br>');
    end

  end
  res_text = BLOB_APPEND(res_text, '<br>');

  suspend;
end
