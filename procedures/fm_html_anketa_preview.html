CREATE PROCEDURE FM_HTML_ANKETA_PREVIEW(
    OBJ_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable color_red varchar(20);
declare variable str varchar(512);
declare variable lang d_str_lang;
declare variable db_id d_id;
declare variable ma_id d_integer;
declare variable pr_code str_100;
declare variable k_id id_type;
begin
  db_id = f_get_db_id();
  res_html = (select res_style from fm_html_styles) || '<body>';
  color_red = 'red';
  lang = f_get_lang_code(un_id, null);

  res_html = BLOB_APPEND(res_html, f_format('<h1 class="green_text">%1</h1>',
    iif(db_id in (39801, 41701),'Анкета и Анализ клиента',f_lang(lang, 'Анкетные данные'))) ||
    f_uri_open(f_lang(lang, 'Открыть анкету'), 'anketa', 'clients', obj_id,
    f_lang(lang, 'Личный лист'), 'rd_id=Anketa.Main&kind=0&can_cancel=0', un_id) ||
    '<br><br>');

  res_html = BLOB_APPEND(res_html,
    f_format('<table width="100%"><col width="50%"><col width="50%">'||
      '<tr><th>%1</th><th>%2</th></tr>',
      f_lang(lang, 'Вопрос'), f_lang(lang, 'Ответ')));

   if (db_id <> 81001) then
   begin
      for select '<tr><td align="left">' ||
        iif(res_answer = '-', '<font color="' || :color_red || '">' || res_question || '</font>', res_question) ||
        '</td><td align="left"><b>' || res_answer || '</b></td></tr>'
      from hp_actions_print_detail(:obj_id)
      into str
      do
      begin
        res_html = BLOB_APPEND(res_html, str);
      end
  end
  if (db_id = 81001) then
  begin
    ma_id = null;
    pr_code = '';

    select ma_id, pr_code
    from miniapp
    inner join miniappclient on mac_from_ma_id = ma_id
    inner join product on pr_id = ma_from_pr_id
    where mac_from_c_id = :obj_id and
          ma_is_active = 1 and
          ma_date_end = '31.12.9999'
    order by ma_id desc rows 1
    into ma_id, pr_code;

    k_id = null;

    select k_id
    from kontract
    inner join kontstate on ks_from_k_id = k_id
    where k_from_ma_app_id = :ma_id and
          ks_ol_is_delete = 0 and
          current_date between ks_date_begin and ks_date_end and
          ks_state = 0
    into k_id;

    if (coalesce(k_id, ma_id) is not null and
        (pr_code containing '.universal.' or pr_code containing '.oivo_pack.')) then
    begin
      for select '<tr><td align="left">' || iif(iif(res_date <= current_date - 90 and ques_code in ('AnketaQuestions.LoanPurpose', 'AnketaQuestions.InfoSource', 'work_place_name', 'work_address_orientir', 'work_address_metro', 'work_staj_last_place', 'work_status', 'work_monthly_income', 'work_income_period', 'work_income_days',
                                                                                                    'work_business_kind', 'work_job_name', 'work_boss_phone_no', 'work_extra_income', 'work_extra_amount', 'AnketaQuestions.MonetaryObligations', 'AnketaQuestions.DefendantInCourt', 'AnketaQuestions.previous_fio'), '-', res_text) = '-', '<font color="' || :color_red || '">' || ques_text || '</font>', ques_text) || '</td><td align="left"><b>' || res_text || '</b></td></tr>'
          from hp_anketa_print(0, :obj_id, 'anketa.main', null)
          into str
      do
      begin
        res_html = blob_append(res_html, str);
      end
    end

    else
    begin
      for select '<tr><td align="left">' || iif(iif(res_date <= current_date - 90 and ques_code not in ('AnketaQuestions.5905', 'AnketaQuestions.Use_BankCard', 'AnketaQuestions.Education', 'AnketaQuestions.InfoSource', 'AnketaQuestions.Ownership', 'AnketaQuestions.5943', 'AnketaQuestions.Nation', 'AnketaQuestions.CanBeSurety', 'AnketaQuestions.previous_fio', 'AnketaQuestions.5915'), '-', res_text) = '-', '<font color="' || :color_red || '">' || ques_text || '</font>', ques_text) || '</td><td align="left"><b>' || res_text || '</b></td></tr>'
          from hp_anketa_print(0, :obj_id, 'anketa.main', null)
          into str
      do
      begin
            res_html = BLOB_APPEND(res_html, str);
      end
    end
  end

  res_html = BLOB_APPEND(res_html, '</table></body>');
  suspend;
end
