CREATE PROCEDURE AAREP_CLOSEDAY_MEMO(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    CD_DATE DATE,
    CD_DATETIME TIMESTAMP,
    CD_OPEN_MEMO VARCHAR(255),
    CD_REGION VARCHAR(50),
    CD_PARENT_NAME VARCHAR(50),
    CD_UN_FIO VARCHAR(100))
AS
begin
  for select cd_date, cd_datetime, cd_open_memo, reg_code || ' ' || reg_name, parent_name, un_fio
      from closeday, region_orig_list(:report_region_id, 1, 3, 1), region, username
      where cd_date >= :report_date_begin and
            cd_date <= :report_date_end and
            (cd_is_closed = 0 or
            cd_is_cash_closed = 0) and
            cd_from_reg_id = reg_unicode and
            reg_unicode = id_list and
            un_id = cd_user_id
      order by id_parent, id_list, cd_datetime
      into :cd_date, :cd_datetime, :cd_open_memo, :cd_region, :cd_parent_name, :cd_un_fio
  do
  begin
    suspend;
  end
end
