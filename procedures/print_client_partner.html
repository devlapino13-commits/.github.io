CREATE PROCEDURE PRINT_CLIENT_PARTNER(
    C_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    FIO_SUPRUG D_STR_100,
    NATION_SUPRUG D_STR_100,
    C_SEX_SUPRUG D_INTEGER,
    INN_SUPRUG D_STR_100,
    C_BIRTH_DATE D_DATE,
    CL_HAVE_SUPRUG D_INTEGER,
    CL_HAVE_CHILDREN D_INTEGER)
AS
declare variable CREL_OTHER_C_ID D_ID;
declare variable HUSBEND_C_ID D_ID;
declare variable REL_CODE D_INTEGER;
begin
     husbend_c_id = null;
     cl_have_suprug = 0;
     cl_have_children = 0;

     for select crt.rel_other_c_id,
                crt.rel_code
     from hp_client_relation_tmp(:c_id, 2) crt
     where crt.rel_code in (1,2)
     into crel_other_c_id, rel_code
     do
     begin
        if (rel_code in (1,2)) then
        begin
            cl_have_suprug = 1;
            husbend_c_id = crel_other_c_id;
        end

        if (rel_code in (5,6)) then
            cl_have_children = 1;
     end

     fio_suprug = null;
     nation_suprug = null;
     c_birth_date  = null;
     inn_suprug  = null;
     c_sex_suprug = null;

     if (husbend_c_id is not null) then
     begin
         select cl.c_name_fio,
                rd.rd_name as nation,
                cl.c_inn  ,
                cl.c_sex,
                cl.c_birth_date
         from clients cl
         left join refdata rd on rd.rd_id = cl.c_nation
         where cl.c_id = :husbend_c_id
         into fio_suprug,
              nation_suprug,
              inn_suprug,
              c_sex_suprug,
              c_birth_date;

     end

     suspend;
end
