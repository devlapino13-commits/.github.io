CREATE PROCEDURE HP_CONTDATA_UPDATE(
    NEW_FROM_C_ID D_ID NOT NULL,
    NEW_CONT_KIND D_STR_CODE NOT NULL,
    NEW_CONT_VALUE D_STR_100 NOT NULL,
    NEW_IS_MAIN D_SMALLINT NOT NULL,
    UN_ID D_ID NOT NULL,
    NEW_DATETIME D_DATETIME = null)
RETURNS (
    CDA_ID D_ID)
AS
declare variable cda_from_clt_id d_integer;
declare variable cda_is_delete d_smallint;
declare variable cda_cont_kind d_integer;
declare variable new_date d_date;
declare variable is_changed d_smallint;
declare variable tmp_id d_id;
declare variable cda_date d_date;
declare variable new_status d_integer;
begin
  cda_cont_kind = f_enum('cont_kind', new_cont_kind, 'id');
  new_cont_kind = f_enum('cont_kind', new_cont_kind, 'code');

  if (trim(coalesce(new_from_c_id, 0)) = 0) then
    exception e_message 'Физлицо не указано';

  new_cont_value = f_cont_trim(new_cont_value, new_cont_kind);

  cda_from_clt_id = f_cont_id(new_cont_value, 'add');

  if (new_is_main = 1 and cda_from_clt_id = 0 and new_cont_kind = 'cell' and f_get_db_id() in (39801, 41701)) then
    exception e_message 'Необходимо указать основной номер телефона.';

  cda_is_delete = 0;
  new_status = 0;

  if (new_datetime is null) then new_datetime = current_timestamp;
  new_date = cast(new_datetime as date);

  is_changed = 0;
  select cda_id, cda_is_delete, cda_date
  from contdata
  where cda_from_c_id = :new_from_c_id and
        cda_cont_kind = :cda_cont_kind and
        cda_from_clt_id = :cda_from_clt_id and
        cda_is_main = :new_is_main
  order by cda_id desc
  rows 1
  into :cda_id, :cda_is_delete, cda_date;

  -- контакты кроме cell и email сразу вставляются как подтвержденные
  if (cda_cont_kind not in (f_enum('cont_kind', 'cell', 'id'), f_enum('cont_kind', 'email', 'id'))) then
    new_status = 1;

  if (cda_id is null) then
  begin
    insert into contdata (cda_from_c_id, cda_cont_kind, cda_from_clt_id, cda_date, cda_status, cda_is_main, cda_is_delete, cda_from_un_id)
    values (:new_from_c_id, :cda_cont_kind, :cda_from_clt_id, :new_date, :new_status, :new_is_main, 0, :un_id)
    returning cda_id, cda_is_delete
    into :cda_id, cda_is_delete;

    is_changed = 1;
  end
  else
  begin
    if (cda_is_delete = 1) then
    begin
      update contdata
      set cda_from_un_id = :un_id
      where cda_id = :cda_id;

      delete from contdelete
      where cdl_from_cda_id = :cda_id;

      is_changed = 1;
    end

    if (new_date <> cda_date) then
    begin
      update contdata
      set cda_date = :new_date,
          cda_from_un_id = :un_id
      where cda_id = :cda_id;
    end
  end

  for select cda_id
  from contdata
  where cda_from_c_id = :new_from_c_id and
        cda_cont_kind = :cda_cont_kind and
        cda_status = :new_status and
        cda_is_main = :new_is_main and
        cda_is_delete = 0 and
        cda_date < :new_datetime and
        cda_id <> :cda_id
  into tmp_id
  do
  begin
    execute procedure hp_contdata_delete(:tmp_id, 'changed', :cda_id, :un_id);
    is_changed = 1;
  end

  suspend;
end
