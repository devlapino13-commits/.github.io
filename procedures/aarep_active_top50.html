CREATE PROCEDURE AAREP_ACTIVE_TOP50(
    REP_DATE DATE)
RETURNS (
    AC_K_ID DECIMAL(18,0),
    AC_DISB DECIMAL(18,4),
    AC_NO INTEGER,
    AC_C_ID DECIMAL(18,0),
    AC_C_BALANCE DECIMAL(18,4),
    C_NAME_FIO VARCHAR(100),
    CL_DISB_DATE DATE,
    SH_DELINQ_DATE DATE,
    DAYS_IN_ARREARS INTEGER,
    CL_CURRENCY VARCHAR(3),
    PR_COMMON_NAME VARCHAR(100),
    BUS_TYPE VARCHAR(100),
    BUS_SUBTYPE VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    CL_DISB_AMOUNT DECIMAL(18,4))
AS
declare variable curr_prior varchar(3);
declare variable curr_code varchar(3);
declare variable curr_rate decimal(18,4);
declare variable k_curr_rate decimal(18,4);
declare variable cm_def_curr varchar(3);
declare variable pr_set_rate_in_kont integer;
declare variable disb_dd date;
declare variable ac_k_balance decimal(18,4);
declare variable sh_os decimal(18,4);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  ac_no = 0;
  for select ac_k_id, ac_disb, ac_c_id, ac_c_balance, cl_disb_amount, disb_dd
      from aarep_active_top50_tmp(:rep_date)
      order by ac_c_balance desc
      rows 50
      into :ac_k_id, :ac_disb, :ac_c_id, :ac_c_balance, :cl_disb_amount, :cl_disb_date
  do
  begin
    ac_no = ac_no + 1;
    select c_name_fio
    from clients
    where c_id = :ac_c_id
    into :c_name_fio;

    days_in_arrears = 0;
    sh_delinq_date = null;
    select ok_sh_date, sum(ok_sh_os + ok_os)
    from operkred
    where ok_from_k_id = :ac_k_id and
          ok_ol_date <= :rep_date and
          ok_sh_date <= :rep_date and
          ok_type <= 6 and
          ok_ol_is_active = 1 and
          ok_from_c_id = 0
    group by ok_sh_date
    having sum(ok_sh_os + ok_os) > 0
    order by ok_sh_date
    rows 1
    into :sh_delinq_date, :sh_os;
    if (sh_delinq_date is not null) then
    begin
      days_in_arrears = cast(:rep_date - :sh_delinq_date as integer);
    end

    select k_kontract_no, k_index_curr_code, pr_common_name, bt_name, bs_name
    from kontract, knclient, product, bustype, bussubtype
    where k_from_pr_id = pr_id and
          kc_from_k_id = k_id and
          kc_busuness_subtype = bs_id and
          bs_from_bt_id = bt_id and
          k_id = :ac_k_id and
          kc_from_c_id = :ac_c_id and
          kc_amount_approved is not null and
          kc_amount_approved > 0
    into :k_kontract_no, :cl_currency, :pr_common_name, :bus_type, :bus_subtype;

    suspend;
  end
end
