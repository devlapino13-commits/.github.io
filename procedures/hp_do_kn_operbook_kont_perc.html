CREATE PROCEDURE HP_DO_KN_OPERBOOK_KONT_PERC(
    KC_ID D_ID,
    OL_DATE D_DATE,
    OL_REG_ID D_INTEGER,
    OL_IS_ACTIVE D_INTEGER,
    UN_ID D_ID)
AS
declare variable ob_detail d_str_255;
declare variable amount curr_type;
declare variable dt_acc_id id_type;
declare variable kt_acc_id id_type;
declare variable loan_acc_group d_str_code;
declare variable kontract_no d_str_255;
declare variable kontract_date date;
declare variable ol_doc_no integer;
declare variable ol_doc_no_str d_str_code;
declare variable oks_acr_perc curr_type;
declare variable oks_bal_kommis curr_type;
declare variable dt_sys_acc d_str_code;
declare variable kt_sys_acc d_str_code;
declare variable non_resident smallint;
declare variable client_type smallint;
declare variable ol_id d_id;
declare variable oks_rvs_perc d_currency;
begin
-- Процедура создает новые доп.проводки по начислению процентов

  -- Не формировать проводки ЕПС кроме РФ даже если учет идет в KN_OKSUMMARY (KZ)
  if (f_get_db_id() <> 81001) then
    exit;

  loan_acc_group = 'Микрозаймы'; -- в будущем нужно прикрепить название раздела системного счета в таблицу продуктов.

  if (ol_date is null) then
    exception e_default 'Date is empty';

  if (ol_date >= cast('01.03.2019' as date)) then
  begin
    select ol_id
    from hp_get_kn_operbook_ol(:ol_date, :ol_reg_id, 'calc_perc')
    into :ol_id;
  end
  else
  begin
    select k_kontract_no, k_kontract_date
    from kontract, knclient
    where kc_id = :kc_id and
          kc_from_k_id = k_id
    into :kontract_no, kontract_date;
    kontract_no = 'договору займа ' || kontract_no || ' от ' ||f_datetostr(:kontract_date) || ' г.';

    ob_detail = 'Начисления по ' || kontract_no;
    insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_doc_no,
                          ol_doc_no_str, ol_memo, ol_parent_id, is_delete, ol_user_id)
    values (:ol_reg_id, 2, 30, :ol_date, 9, :ol_doc_no, :ol_doc_no_str, :ob_detail, null, 0, :un_id)
    returning ol_id
    into :ol_id;
  end

  select non_resident, client_type
  from hp_get_kn_client_resident(:kc_id, :ol_date)
  into non_resident, client_type;

-- начисления процентов по займу линейный метод
  for select oks_acr_perc, oks_rvs_perc, oks_bal_kommis
      from kn_oksummary
      where oks_from_kc_id = :kc_id and
            oks_date_begin = :ol_date and
            oks_date_prv <> oks_date_begin
      into oks_acr_perc, oks_rvs_perc, oks_bal_kommis
  do
  begin
    if (oks_acr_perc is null) then
      oks_acr_perc = 0;
    if (oks_bal_kommis is null) then
      oks_bal_kommis = 0;
    if (oks_rvs_perc is null) then
      oks_rvs_perc = 0;

-- Бух.проводки по начислению процентов
    oks_acr_perc = oks_acr_perc - oks_rvs_perc;
    amount = oks_acr_perc;
    dt_sys_acc = 'Процент.Начислено';
    kt_sys_acc = 'Процент.Доход';
    ob_detail = 'Начисление процента';

    select acc_id
    from get_ref_account_knclient(:loan_acc_group, :dt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
    into dt_acc_id;

    select acc_id
    from get_ref_account_office(:loan_acc_group, :kt_sys_acc, :ol_reg_id, '', -1, :non_resident, :client_type, :ol_date)
    into kt_acc_id;

    execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
        :ol_date, :ol_reg_id, :ol_is_active, :ob_detail, 2);

    if (oks_rvs_perc <> 0) then
    begin
      oks_acr_perc = oks_acr_perc + oks_rvs_perc;
      amount = oks_acr_perc;

      execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
          :ol_date, :ol_reg_id, :ol_is_active, :ob_detail, 2);
    end

    amount = -oks_bal_kommis;
    dt_sys_acc = 'ДопПроцент.Погашено';
    kt_sys_acc = 'ДопПроцент.Начислено';
    ob_detail = 'Списание доначисленного доп процента';

    select acc_id
    from get_ref_account_knclient(:loan_acc_group, :dt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
    into dt_acc_id;

    select acc_id
    from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
    into kt_acc_id;

    execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, amount, 0, amount, :ol_date,
        :ol_reg_id, :ol_is_active, :ob_detail, 1, 0);

    amount = oks_bal_kommis;
    dt_sys_acc = 'ДопПроцент.Начислено';
    kt_sys_acc = 'ДопПроцент.Доход';
    ob_detail = 'Начисление доп.процента';

    select acc_id
    from get_ref_account_knclient(:loan_acc_group, :dt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
    into dt_acc_id;

    select acc_id
    from get_ref_account_office(:loan_acc_group, :kt_sys_acc, :ol_reg_id, '', -1, :non_resident, :client_type, :ol_date)
    into kt_acc_id;

    execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, 0, 0, 0, :ol_date,
        :ol_reg_id, :ol_is_active, :ob_detail, 1);
  end
end
