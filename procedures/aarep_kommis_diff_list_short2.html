CREATE PROCEDURE AAREP_KOMMIS_DIFF_LIST_SHORT2(
    NEW_VALUE D_PERCENT,
    SHOW_ROW_COUNT D_INTEGER,
    FIX_3 D_INTEGER,
    CUSTOM_KOEF D_PERCENT)
RETURNS (
    GROUP_KIND D_STR_CODE,
    REC_ID D_INTEGER,
    K_ID D_ID,
    K_DISB_DATE D_DATE,
    CLIENT_COUNT D_INTEGER,
    ORIG_RATE_YEAR D_PERCENT,
    ORIG_AMOUNT D_CURRENCY,
    ORIG_PERC D_CURRENCY,
    ORIG_KOMMIS D_CURRENCY,
    NEW_RATE_YEAR D_PERCENT,
    NEW_AMOUNT D_CURRENCY,
    NEW_PERC D_CURRENCY,
    DIFF_CLIENT_COUNT D_INTEGER,
    DIFF_ALL_PERC D_CURRENCY,
    DIFF_PLUS_PERC D_CURRENCY,
    LAST_DATE D_DATE,
    PERCENT_WITH_KOEF D_PERCENT)
AS
declare variable k_percent_per_day d_percent;
declare variable sh_date d_date;
declare variable sh_date_prv d_date;
declare variable tmp_os d_currency;
declare variable tmp_perc d_currency;
declare variable tmp_kommis d_currency;
declare variable days_list d_str_255;
declare variable annu d_currency;
declare variable balance d_currency;
declare variable sh_is_lgota d_smallint;
declare variable days d_integer;
declare variable iter d_integer;
declare variable k_percent_per_day_orig d_percent;
declare variable min_iter d_integer;
declare variable max_iter d_integer;
declare variable pr_percent_for_days d_integer;
begin
  -- Если new_perc <= 2, то это коэффициент
  -- Иначе это абсолютное значение к проценту

  /*
    last_date - дата погашения по графику
    percent_with_koef - Измененная % ставка
  */

  if (show_row_count is null) then
    show_row_count = 0;

  if (fix_3 is null) then
    fix_3 = 0;

  if (fix_3 = 0) then
  begin
    min_iter = -10;
    max_iter = -1;
  end
  if (fix_3 = 1) then
  begin
    min_iter = -1;
    max_iter = 1;
  end
  if (fix_3 = -1) then
  begin
    min_iter = 0;
    max_iter = 0;
  end


  rec_id = 0;
  for select ks_from_k_id, ks_date_begin, k_percent_per_day, k_kolich_client_approved, pr.pr_percent_for_days
  from kontstate ks, kontract, product pr
  where ks_state = 1 and
        ks_date_begin >= '08.07.2020' and
        ks_ol_is_delete = 0 and
        ks_from_k_id = k_id and
        k_from_pr_id = pr_id and
        pr_percent_kind in (0, 1) and
        k_kommis_perc <> 0
  into k_id, k_disb_date, k_percent_per_day, client_count, :pr_percent_for_days
  do
  begin
    rec_id = rec_id + 1;
    k_percent_per_day_orig =  k_percent_per_day;

    orig_amount = 0;
    orig_perc = 0;
    orig_kommis = 0;
    orig_rate_year = round(k_percent_per_day_orig * 365.0, 1);

    sh_date_prv = null;
    days_list = '';
    sh_date = null;

    for select sh_date, sh_os, sh_perc, sh_kommis
    from schedule
    where sh_from_k_id = :k_id
    order by sh_from_k_id, sh_id
    into sh_date, tmp_os, tmp_perc, tmp_kommis
    do
    begin
      orig_amount = orig_amount + tmp_os;
      orig_perc = orig_perc + tmp_perc;
      orig_kommis = orig_kommis + tmp_kommis;

      if (sh_date_prv is not null) then
      begin
        if (days_list <> '') then
          days_list = days_list || ',';
        days_list = days_list || datediff(day, sh_date_prv, sh_date);
      end

      sh_date_prv = sh_date;
    end
    last_date = sh_date;

    if (orig_kommis = 0) then
      exception e_default f_format('Zero kommis, k_id = %1', k_id);

    new_amount = orig_amount - orig_kommis;

    iter = min_iter;

    while (iter <= max_iter) do
    begin
      if (fix_3 = 0) then
      begin
        if (iter = 0) then
        begin
          new_value = 9.34;
          group_kind = 'Увеличение на ' || round(new_value, 2);
        end else
        if (iter < 0) then
        begin
          new_value = (940.00 + iter) / 100.00;
          group_kind = 'Увеличение на ' || round(new_value, 2);
        end
        else
        begin
          new_value = 1 + cast(iter as double precision) / 100.00;
          group_kind = 'Коэф. повышения ' || round(new_value, 2);
        end
      end
      if (fix_3 = 1) then
      begin
        if (iter = 0) then
        begin
          new_value = 9.34;
          group_kind = 'Увеличение на ' || round(new_value, 2);
        end
        if (iter = -1) then
        begin
          new_value = 1.25374;
          group_kind = 'Коэф. повышения ' || round(new_value, 5);
        end
        if (iter = 1) then
        begin
          new_value = custom_koef;
          group_kind = 'Коэф. повышения ' || round(new_value, 5);
        end
      end

      if (fix_3 = -1) then
      begin
        new_value = custom_koef;
        group_kind = 'Коэф. повышения ' || round(new_value, 5);
      end

      k_percent_per_day = k_percent_per_day_orig;

      if (new_value <= 2)
      then k_percent_per_day = round(k_percent_per_day * new_value, 7);
      else
      begin
        k_percent_per_day = round(k_percent_per_day * 365.0000000, 1);
        k_percent_per_day = round((k_percent_per_day + new_value) / 365.0000000, 7);
      end
  
      if (k_percent_per_day = 0) then
        exception e_default f_format('Zero new interest rate, k_id = %1', k_id);
  
      new_rate_year = round(k_percent_per_day * 365.0, 2);
  
      annu = f_annu(new_amount, 0, 1, k_percent_per_day, days_list, 0, 0, 0, 0, 0);
  
      new_perc = 0;
  
      sh_date_prv = k_disb_date;
      balance = new_amount;
    
      for select sh_date, iif(sh_os = 0, 1, 0)
      from schedule
      where sh_from_k_id = :k_id and
            sh_id > 0
      order by sh_from_k_id, sh_id
      into sh_date, sh_is_lgota
      do
      begin
        days = datediff(day, sh_date_prv, sh_date);
        tmp_perc = cast(balance / 100.0000 as double precision);
        tmp_perc = cast(tmp_perc * k_percent_per_day as double precision);
        tmp_perc = cast(tmp_perc * days as double precision);
        tmp_perc = round(tmp_perc, 0);
        new_perc = new_perc + tmp_perc;
  
        tmp_os = 0;
    
        if (sh_is_lgota = 0) then
          tmp_os = annu - tmp_perc;
    
        if (sh_date = last_date) then
          tmp_os = balance;
    
        sh_date_prv = sh_date;
        balance = balance - tmp_os;
      end
  
      diff_client_count = 0;
      diff_all_perc = 0;
      diff_plus_perc = 0;
  
      diff_all_perc = orig_perc + orig_kommis - new_perc;
  
      if (diff_all_perc <> 0) then
      begin
        diff_client_count = client_count;
        diff_plus_perc = diff_all_perc;
      end

      percent_with_koef = 0;

      select round(cast(kd.kd_percent_per_day as double precision) * :pr_percent_for_days * :custom_koef, 2)
      from kontdata kd
      where kd.kd_from_k_id = :k_id
      and kd.kd_ol_is_delete =0
      and current_date between kd.kd_ol_date and kd.kd_date_end
      into :percent_with_koef;
  
      suspend;
      iter = iter + 1;
    end

    if (show_row_count > 0 and rec_id >= show_row_count) then
      exit;
  end
end
