CREATE PROCEDURE HP_CLIENTDOCS_LIST(
    C_ID D_ID)
RETURNS (
    CDS_ID D_INTEGER,
    CDS_VERIFY_STATUS D_INTEGER,
    CDS_VERIFY_STATUS_NAME D_STR_255,
    PK_NAME D_STR_100,
    CDS_DOC_NO VARCHAR(30),
    CDS_CREATE_DATETIME D_DATETIME,
    CDS_DOC_DATE_BEGIN D_DATE,
    CDS_DOC_DATE_END D_DATE,
    CDS_DOC_ISSUED_BY D_STR_255,
    CDS_IS_DELETE D_INTEGER)
AS
begin
--RFCKG-4715
--Форма "Изменить даты документа"
    for select cds_id,
               case
                 when cds_verify_status = -2 then 'Данные не верны'
                 when cds_verify_status = -1 then 'Скан не требуется'
                 when cds_verify_status = 0 then 'В процессе редактирования'
                 when cds_verify_status = 1 then 'Отправлен на проверку'
                 when cds_verify_status = 2 then 'Подтвержден'
                 when cds_verify_status = 3 then 'Отправлен на проверку*'
               else 'Неизвестный статус'
               end as cds_verify_status_name,
               pk_name,
               cds_doc_no,
               cds_create_datetime,
               cds_doc_date_begin,
               cds_doc_date_end,
               cds_doc_issued_by,
               cds_is_delete
        from aarep_clientdocs_list(:c_id)
        order by cds_is_old_doc, cds_id
    into
        :cds_id,
        :cds_verify_status_name,
        :pk_name,
        :cds_doc_no,
        :cds_create_datetime,
        :cds_doc_date_begin,
        :cds_doc_date_end,
        :cds_doc_issued_by,
        :cds_is_delete
    do
        suspend;
end
