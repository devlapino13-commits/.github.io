CREATE PROCEDURE HP_IPF_KOMMIS_FOR_DISB_CREATE(
    REP_REGION_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    DO_SAVE SMALLINT = 0)
RETURNS (
    REGION_NAME STR_100,
    OFFICE_NAME STR_100,
    ACC_KOMMENT STR_255,
    SUM_KOMMIS CURR_TYPE)
AS
declare variable reg_id id_type;
declare variable new_ol_id decimal(18,0);
declare variable dt_acc_id varchar(20);
declare variable kt_acc_id varchar(20);
declare variable kt_al_name varchar(200);
declare variable dt_al_name varchar(200);
begin
  --Скрипт который вытащит все взносы за выдачу ИПФ за период:
  for select parent_name, reg_name, id_list, rd.rd_name, sum(ok.ok_amount) ok_amount
      from operkontract ok
      inner join refdata rd on rd.rd_id = ok.ok_operation_type and
            rd.rd_code = 'CashTransactionIncom.IPFCommission'
      inner join operlist ol on ol.ol_id = ok.ok_from_ol_id and
            ol.is_delete = 0 and
            ol.ol_date between :rep_date_begin and :rep_date_end
      inner join region_list(:rep_region_id, 1, -1, 1) on ol.ol_from_reg_id = id_list
      inner join region on reg_unicode = ol_from_reg_id
      inner join kontstate ks on ks.ks_from_k_id = ok.ok_from_k_id and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 -- условие добавил igusev т.к. клиент в последний момнет может отказаться и тогда будет возврат комиссии, который не учтен.
      group by parent, parent_name, reg_name, id_list, rd_name
      having sum(ok.ok_amount) <> 0
      into region_name, office_name, reg_id, acc_komment, sum_kommis
  do
  begin
    if (do_save = 1) then
    begin
      --Получаем айди для новой операции
      new_ol_id = null;
      select kv
      from get_kv(:reg_id, 5)
      into :new_ol_id;

      --Дт. 21141/1000 (Единовременный сбор по ИПФ)
      --Кт. 60591/8 Доп.доход от единовременных сборов по ИПФ
      dt_acc_id = null;
      select al.al_id, al.al_name
      from acc_list al
      where al.al_from_reg_id = :reg_id and
            al.al_from_bg_id = '21141' and
            al.al_number = 1000
      into dt_acc_id, dt_al_name;

      kt_acc_id = null;
      select al.al_id, al.al_name
      from acc_list al
      where al.al_from_reg_id = :reg_id and
            al.al_from_bg_id = '60591' and
            al.al_number = 8
      into kt_acc_id, kt_al_name;

      --Создаем операцию
      insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete,
                            ol_user_id, ol_doc_client_name)
      values (:new_ol_id, :reg_id, 2, :rep_date_end, 8,
              'Признание доходов по един. сборам за финансирования', 0, 0,
              1090100000385, ''); --1090100000385 = igusev

      --Создаем проводки в бух.таблице operbook
      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                            ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                            ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date,
                            ob_ol_is_active, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt)
      values (:new_ol_id, -1, '417', 1, null, :dt_acc_id, :sum_kommis, :sum_kommis, null, null, null, :kt_acc_id, null,
              :kt_al_name, null, :rep_date_end, 1, :reg_id, '21141/1000', '60591/8');
    end
    else
      suspend;
  end
end
