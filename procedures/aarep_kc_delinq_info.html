CREATE PROCEDURE AAREP_KC_DELINQ_INFO(
    REP_REGION_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    OFFICE_NAME VARCHAR(100),
    K_KONTRACT_NO VARCHAR(50),
    K_KF_NO VARCHAR(100),
    K_DISB_DATE DATE,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    PR_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    K_AMOUNT_APPROVED CURR_TYPE,
    K_DELINQ_DAYS INTEGER,
    C_PLASTCARD_NO VARCHAR(100),
    C_BIRTH_DATE DATE,
    ADL_ADDRESS_STR VARCHAR(255),
    C_CITIZEN_TXT VARCHAR(255),
    C_BIRTH_COUNTRY VARCHAR(100),
    K_STOP_PERC DATE,
    K_CITY VARCHAR(100),
    KC_AMOUNT_APPROVED CURR_TYPE,
    K_KOLICH_REPAY INTEGER,
    K_PERCENT PERC_TYPE,
    K_DELINQ_DATE DATE,
    K_PAYED_SUM CURR_TYPE,
    K_ALL_BALANCE CURR_TYPE,
    K_BALANCE CURR_TYPE,
    K_BALANCE_PERC CURR_TYPE,
    C_PAYED_SUM CURR_TYPE,
    C_ALL_BALANCE CURR_TYPE,
    C_BALANCE CURR_TYPE,
    C_BALANCE_PERC CURR_TYPE,
    PORUCH VARCHAR(50),
    K_KREDLINE_END DATE,
    C_AMNESTY VARCHAR(10))
AS
declare variable k_id id_type;
declare variable k_id_temp id_type;
declare variable reg_id id_type;
declare variable reg_id_tmp id_type;
declare variable c_id id_type;
declare variable kf_id id_type;
declare variable pr_id id_type;
declare variable pr_id_tmp id_type;
declare variable reg_address_fact id_type;
declare variable c_dolya perc_type;
declare variable pr_code varchar(100);
begin
  for select oks_from_k_id, oks_region_id, oks_balance, cast(:rep_date as date) - oks_delinq_date as k_delinq_days,
             oks_delinq_date, --
             k_kontract_no, k_amount_approved, k_fact_disb_date, k_from_kf_id, k.k_kolich_client_approved, k_from_pr_id,
             k_stop_perc, k_kolich_repay, k_percent, k.k_kredline_end
      from oksummary oks, region_list(:rep_region_id, 1, -1, 1), kontract k
      where oks_close_state = 0 and
            :rep_date between oks_date_begin and oks_date_end and
            :rep_date > oks_delinq_date and
            oks_region_id = id_list and
            oks_from_k_id = k_id
      order by oks_region_id, k_from_pr_id
      into k_id, reg_id, :k_balance, k_delinq_days, k_delinq_date, --
           :k_kontract_no, :k_amount_approved, :k_disb_date, kf_id, :k_kolich_client_approved, pr_id, k_stop_perc,
           k_kolich_repay, k_percent, k_kredline_end
  do
  begin

    if (reg_id <> coalesce(reg_id_tmp, 0)) then
    begin
      reg_id_tmp = reg_id;
      select reg_name, reg.reg_address_fact
      from region reg
      where reg_unicode = :reg_id
      into office_name, reg_address_fact;

      k_city = null;
      select pl.pl_name
      from addrlist adl, places pl
      where adl.adl_id = :reg_address_fact and
            adl.adl_from_pl_id = pl.pl_id and
            pl.is_delete = 0
      into :k_city;
    end

    if (pr_id <> coalesce(pr_id_tmp, 0)) then
    begin
      pr_id_tmp = pr_id;
      select pr_name, pr_code
      from product pr
      where pr.pr_id = :pr_id
      into :pr_name, pr_code;
    end

    k_kf_no = null;
    select kal.kal_new_kf_name
    from knaktlist kal
    where kal.kal_from_kf_id = :kf_id and
          kal.is_delete = 0 and
          :rep_date between kal_date and kal_date_end
    into k_kf_no;

    poruch = 'Не определен';
    if (pr_code containing '.GR.' and
        pr_code containing '.FAMILY') then
    begin
      if (k_disb_date between '05.10.15' and '21.10.15') then
      begin
        poruch = 'Групповой без поручительства';
      end
      if (k_disb_date >= '22.10.15') then
      begin
        poruch = 'Групповой с поручительством';
      end
    end

    if (pr_code containing '.GR.' and
        pr_code not containing '.FAMILY') then
    begin
      if (:k_disb_date < '15.12.14') then
      begin
        poruch = 'Групповой с поручительством';
      end
      if (:k_disb_date between '15.12.14' and '14.01.15') then
      begin
        poruch = 'Групповой без поручительства';
      end
      if (:k_disb_date between '15.01.15' and '30.06.15') then
      begin
        poruch = 'Групповой с поручительством';
      end
      if (:k_disb_date between '01.07.15' and '01.01.2016') then
      begin
        poruch = 'Групповой без поручительства';
      end

      if (pr_code containing '.PATENT.') then
      begin
        poruch = 'Групповой с поручительством';
      end
    end

    if (pr_code containing '.IND.') then
    begin
      if (pr_code containing '.ONE.') then
      begin
        if (:k_disb_date between '15.12.14' and '14.01.15') then
        begin
          poruch = 'Индивидуальный без поручительства';
        end
        if (:k_disb_date between '15.01.15' and '30.06.15') then
        begin
          poruch = 'Индивидуальный с поручительством';
        end
        if (:k_disb_date between '01.07.15' and '01.01.2016') then
        begin
          poruch = 'Индивидуальный без поручительства';
        end
      end

      if (pr_name containing 'Будущий сотрудник' or (pr_code similar to '%(.EMP.|.PP.)%')) then
      begin
        poruch = 'Индивидуальный с поручительством';
      end
      if (pr_code similar to '%(.PATENT.|.BARAKAT|.LOYAL.|.WEEK.)%') then
      begin
        poruch = 'Индивидуальный без поручительства';
      end
    end

    k_payed_sum = 0;
    k_all_balance = 0;
    k_balance_perc = 0;
    select -sum(iif(ok_type in (4, 9), ok_os + ok_sh_os + ok_perc + ok_sh_perc, 0)) payed_sum,
           sum(iif(ok_type < 7, ok_os + ok_sh_os + ok_perc + ok_sh_perc, 0)) balance_all,
           sum(iif(ok_type < 7, ok_perc + ok_sh_perc, 0)) k_balance_perc
    from operkred ok
    where ok_ol_is_active = 1 and
          ok_from_c_id = 0 and
          ok_from_k_id = :k_id
    into k_payed_sum, k_all_balance, k_balance_perc;

    for select kc_from_c_id, kc_amount_approved
        from knclient kc
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id, kc_amount_approved
    do
    begin
      select c_name_fio, c_birth_date, rd.rd_name, cl.c_birth_country
      from clients cl, refdata rd
      where c_id = :c_id and
            c_citizen = rd_id and
            rd.rd_main_code = 'Citizen'
      into c_name_fio, c_birth_date, c_citizen_txt, c_birth_country;

      c_plastcard_no = null;
      select cont_value
      from hp_contdata_get_cell_main(0, :c_id)
      into c_plastcard_no;

      adl_address_str = null;
      select adl.adl_address_str
      from addrclient adc, addrlist adl
      where adc.adc_from_c_id = :c_id and
            adc.adc_address_type = 2 and
            adc.adc_from_adl_id = adl.adl_id
      into adl_address_str;

      c_dolya = kc_amount_approved / k_amount_approved;

      c_payed_sum = k_payed_sum * c_dolya;
      c_all_balance = k_all_balance * c_dolya;
      c_balance = k_balance * c_dolya;
      c_balance_perc = k_balance_perc * c_dolya;

--амнистирован или нет 09-февраль 17-г
      if (exists(select *
                 from kontdevides kd
                 where kd.kd_date <= :rep_date and
                       kd.kd_c_id = :c_id and
                       kd.kd_k_new = :k_id)) then
        c_amnesty = 'Да';
      else
        c_amnesty = 'Нет';

      suspend;
    end
  end
end
