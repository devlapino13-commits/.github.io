CREATE PROCEDURE UXP$UXEVENT_CONTROL_LINKS(
    CONTROL_ID D_STR_CODE,
    EVENT_TYPE D_STR_10)
RETURNS (
    EVENT_ID D_INTEGER,
    EVENT_CAPTION D_STR_255,
    EVENT_EXP_KIND D_STR_100)
AS
declare variable event_kind d_integer;
begin
  if (event_type not in ('On', 'Do')) then
    exception e_default f_format('Undefined event_kind: %1', coalesce(event_type, '<Null>'));

  --   0 - Событие (OnChange и т.д.)
  --   1 - Обработчик, заполнение выражения не образательно
  --   2 - Обработчик с обязательным выражением
  --   3 - Обработчик с обязательным логическим выражением


  for select event_id, event_code || '  - ' || event_name, event_kind
  from uxp$uxevent_links, uxp$uxevent_kinds(:event_type)
  where evl_control = :control_id and
        evl_event_code = event_code
  into :event_id, :event_caption, :event_kind
  do
  begin
    event_exp_kind = case event_kind
      when 0 then '-'
      when 1 then 'Выражение не образательно'
      when 2 then 'Арифметическое выражение'
      when 3 then 'Логическое выражение'
    end;

    event_exp_kind = trim(event_exp_kind);

    suspend;
  end
end
