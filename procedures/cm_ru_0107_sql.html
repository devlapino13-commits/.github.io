CREATE PROCEDURE CM_RU_0107_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable PR_CODE varchar(100);
declare variable KC_FROM_C_ID ID_TYPE;
declare variable C_CITIZEN integer;
declare variable RD_CODE varchar(255);
declare variable RD_NAME varchar(255);
declare variable KC_CIRCLE integer;
declare variable K_KREDLINE_END date;
declare variable PDT_CODE varchar(100);
declare variable CDS_DOC_DATE_BEGIN date;
declare variable CDS_DOC_DATE_END date;
declare variable PDT_NAME varchar(100);
declare variable C_LAST_STATUS integer;
begin
  for select KC_FROM_C_ID, KC_AMOUNT_APPROVED, KC_CIRCLE, PR_CODE, K_DISB_DATE
      from KNCLIENT, KONTRACT, PRODUCT
      where KC_FROM_K_ID = :K_ID and
            KC_AMOUNT_APPROVED > 0 and
            KC_FROM_K_ID = K_ID and
            K_FROM_PR_ID = PR_ID and
            PR_CODE similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.|.BLAGO.|.USPEH.|.RELIABLE.|.RELIABLE_CLIENT.|.RELIABLE_CLIENT_PLUS.|.PAR.LOYAL.)%'
      into KC_FROM_C_ID, KC_AMOUNT, KC_CIRCLE, PR_CODE, KF_DISB_DATE
  do
  begin
    select C_NAME_FIO, C_CITIZEN
    from CLIENTS
    where C_ID = :KC_FROM_C_ID
    into res_info_text, C_CITIZEN;
    select RD_CODE, RD_NAME
    from REFDATA
    where RD_ID = :C_CITIZEN
    into RD_CODE, RD_NAME;
    if ((PR_CODE containing '.BARAKAT.' or pr_code containing '.BLAGO.' or pr_code containing '.RELIABLE_CLIENT.' or pr_code containing '.PAR.LOYAL.') and
        RD_CODE not in ('Citizen.KG', 'Citizen.UZ', 'Citizen.TJ', 'Citizen.RU', 'Citizen.KZ')) then
    begin
      res_error_text = 'По выбранному продукту граждане ' || RD_NAME || ' не могут брать.';
      res_error_is_critical = 1;
      suspend;
    end       
    if ((PR_CODE containing '.BARAKAT PLUS.' or pr_code containing '.USPEH.' or pr_code containing '.RELIABLE_CLIENT_PLUS.')and
        RD_CODE not in ('Citizen.KG', 'Citizen.UZ', 'Citizen.TJ', 'Citizen.RU')) then
    begin
      res_error_text = 'По выбранному продукту граждане ' || RD_NAME || ' не могут брать.';
      res_error_is_critical = 1;
      suspend;
    end
    if ((PR_CODE containing 'BARAKAT'
         or pr_code containing '.BLAGO.'
         or pr_code containing '.USPEH.'
         or pr_code containing '.RELIABLE.'
         or pr_code containing '.RELIABLE_CLIENT.'
         or pr_code containing '.RELIABLE_CLIENT_PLUS.'
         or pr_code containing '.PAR.LOYAL.')
        and :RD_CODE = 'Citizen.KG'
        and not exists(select *
                   from CLIENTDOCS CD, PASSKIND, PASSDOCTYPE
                   where CDS_FROM_C_ID = :KC_FROM_C_ID and
                         CDS_FROM_PK_ID = PK_ID and
                         PK_FROM_PDT_ID = PDT_ID and
                         CDS_IS_DELETE = 0 and
                         coalesce(cds_doc_blob_1, 0) > 0 and
                         coalesce(CDS_DOC_DATE_END, cast('31.12.99' as date)) >= :KF_DISB_DATE and
                         (PDT_CODE in ('magration_card', 'registration_rvp', -- Разрешение на временное проживание (РВП)
                                   'registration_vnj', 'job_agreement') -- Вид на жительство (ВНЖ) и труд.договор
                         or PK_ID = 87))) then
    begin
      res_error_text = 'Для гражд. КГ не имеется документ миграционного учета.';
      res_error_is_critical = 1;
      suspend;
    end
    /*
    if ((PR_CODE containing '.BARAKAT.'
         or pr_code containing '.blago.')
        and
        RD_CODE <> 'Citizen.RU' and
        RD_CODE <> 'Citizen.KG' and
        not exists(select *
                   from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                   where CDS_FROM_C_ID = :KC_FROM_C_ID and
                         CDS_FROM_PK_ID = PK_ID and
                         PK_FROM_PDT_ID = PDT_ID and
                         CDS_IS_DELETE = 0 and
                         coalesce(cds_doc_blob_1, 0) > 0 and
                         :KF_DISB_DATE between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END  and
                         PDT_CODE in ('work_patent', 'work_permit'))) then
    begin
      res_error_text = 'У клиента должно быть патент или разрешения на работу!';
      res_error_is_critical = 1;
      suspend;
    end
    */
    if ((PR_CODE containing '.BARAKAT PLUS.'
         or pr_code containing '.USPEH.'
         or pr_code containing '.RELIABLE.'
         or pr_code containing '.RELIABLE_CLIENT_PLUS.')
        and rd_code not in ('Citizen.KG', 'Citizen.RU')
        and 1<>1 -- данная проверка убрана так как проверки усвлоии документов настравивается в другом модуле
        ) then
    begin
      if (not exists(select *
                     from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                     where CDS_FROM_C_ID = :KC_FROM_C_ID and
                           CDS_FROM_PK_ID = PK_ID and
                           PK_FROM_PDT_ID = PDT_ID and
                           CDS_IS_DELETE = 0 and
                           coalesce(cds_doc_blob_1, 0) > 0 and
                           :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END and
                           (PDT_CODE in ('registration_rvp', -- Разрешение на временное проживание (РВП)
                                     'registration_vnj', -- Вид на жительство (ВНЖ)
                                     'work_patent', -- патент
                                     'work_permit',  -- разрешение наработу
                                     'student_card'
                                     ) or (PDT_CODE in ('Docs_VUZ_SUZ') and pk_id = 326)  or PK_ID = 65 -- 'Свидетельство участника программы добровольного переселения'
                           ))) then
      begin
        res_error_text = 'У клиента должен быть "РВП" или "ВНЖ" или "Свидетельство участника программы добровольного переселения"';
        res_error_is_critical = 1;
        suspend;
      end
    end
    --    Обязательно наличие у клиента Патента или Разрешения на работу
    --    Разработан по пункту 5
    for select PDT_CODE, min(CDS_DOC_DATE_BEGIN), max(CDS_DOC_DATE_END), PDT_NAME
        from CLIENTDOCS, PASSKIND, PASSDOCTYPE
        where CDS_FROM_C_ID = :KC_FROM_C_ID and
              CDS_FROM_PK_ID = PK_ID and
              PK_FROM_PDT_ID = PDT_ID and
              CDS_IS_DELETE = 0 and
              coalesce(cds_doc_blob_1, 0) > 0 and
              :KF_DISB_DATE between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
        group by PDT_CODE, PDT_NAME
        into PDT_CODE, CDS_DOC_DATE_BEGIN, CDS_DOC_DATE_END, PDT_NAME
    do
    begin
      if (PDT_CODE similar to 'registration_yur|registration_fiz' and
          CDS_DOC_DATE_END < K_KREDLINE_END and
          :RD_CODE = 'Citizen.KG' and
          exists(select *
                 from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                 where CDS_FROM_C_ID = :KC_FROM_C_ID and
                       CDS_FROM_PK_ID = PK_ID and
                       PK_FROM_PDT_ID = PDT_ID and
                       CDS_IS_DELETE = 0 and
                       coalesce(cds_doc_blob_1, 0) > 0 and
                       :KF_DISB_DATE between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END and
                       PDT_CODE similar to 'registration_yur|registration_fiz' and
                       CDS_DOC_DATE_END >= :K_KREDLINE_END)) then
        CDS_DOC_DATE_END = K_KREDLINE_END + 10;
      if (PDT_CODE containing 'registration_rvp' and
          :RD_CODE = 'Citizen.KG' and
          exists(select *
                 from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                 where CDS_FROM_C_ID = :KC_FROM_C_ID and
                       CDS_FROM_PK_ID = PK_ID and
                       PK_FROM_PDT_ID = PDT_ID and
                       CDS_IS_DELETE = 0 and
                       coalesce(cds_doc_blob_1, 0) > 0 and
                       :KF_DISB_DATE between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END and
                       PDT_CODE containing 'registration_vnj')) then
        CDS_DOC_DATE_END = :K_KREDLINE_END + 10;
      if (:RD_CODE in ('Citizen.TJ', 'Citizen.UZ')) then
      begin
        if (PR_CODE containing '.BARAKAT.'
            or pr_code containing '.BLAGO.'
            or pr_code containing '.RELIABLE.'
            or pr_code containing '.RELIABLE_CLIENT.'
            or pr_code containing '.PAR.LOYAL.') then
        begin
          if (PDT_CODE in ('work_permit', -- Разрешение на работу
                       'work_patent') -- Патент) then
              and
              not exists(select *
                         from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                         where CDS_FROM_C_ID = :KC_FROM_C_ID and
                               CDS_FROM_PK_ID = PK_ID and
                               PK_FROM_PDT_ID = PDT_ID and
                               CDS_IS_DELETE = 0 and
                               coalesce(cds_doc_blob_1, 0) > 0 and
                               PDT_CODE similar to 'certificate_approving_doc|application_citizenship_rassian_lang_speaker')) then
          begin
            if (
                --            dateadd(month, 2, :KF_DISB_DATE) > CDS_DOC_DATE_END and  --Новое условие от 01.04.18 Заменен на условие ниже
                KF_DISB_DATE > coalesce(CDS_DOC_DATE_END, '01.01.1990') and
                PDT_CODE in ('work_permit')) then -- Разрешение на работу
            begin
              res_error_text = PDT_NAME || ' должен быть действительны на момент получения микрозайма';
              res_error_is_critical = 1;
              suspend;
            end
            select C_LAST_STATUS
            from HP_GET_CLIENT_STATUS(:KC_FROM_C_ID, :KF_DISB_DATE)
            into C_LAST_STATUS;
            if (CDS_DOC_DATE_END < KF_DISB_DATE and
                PDT_CODE in ('work_patent') and
                C_LAST_STATUS = 0) then
            begin
              res_error_text = PDT_NAME || ' должен быть действительны на момент получения микрозайма';
              res_error_is_critical = 1;
              suspend;
            end
          end
        end
        else
        if (PR_CODE containing '.BARAKAT PLUS.'
            or pr_code containing '.USPEH.'
            or pr_code containing '.RELIABLE.'
            or pr_code containing '.RELIABLE_CLIENT_PLUS.') then
        begin
          if (PDT_CODE in ('registration_rvp') -- Разрешение на временное проживание (РВП)
              and
              not exists(select *
                         from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                         where CDS_FROM_C_ID = :KC_FROM_C_ID and
                               CDS_FROM_PK_ID = PK_ID and
                               PK_FROM_PDT_ID = PDT_ID and
                               coalesce(cds_doc_blob_1, 0) > 0 and
                               CDS_IS_DELETE = 0 and
                               CDS_DOC_DATE_END > :CDS_DOC_DATE_END and
                               PDT_CODE containing 'registration_vnj')) then
          begin
            if (not exists(select *
                           from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                           where CDS_FROM_C_ID = :KC_FROM_C_ID and
                                 CDS_FROM_PK_ID = PK_ID and
                                 PK_FROM_PDT_ID = PDT_ID and
                                 CDS_IS_DELETE = 0 and
                                 coalesce(cds_doc_blob_1, 0) > 0 and
                                 PDT_CODE containing 'certificate_approving_doc') and

                --and dateadd(month, 3, :kf_disb_date) > CDS_DOC_DATE_END) then действительными на момент получения займа документами
                KF_DISB_DATE > CDS_DOC_DATE_END) then
            begin
              res_error_text = 'Для гражд. ТЖ и УЗ ' || PDT_NAME || ' должны быть действительны на момент получения микрозайма';
              res_error_is_critical = 1;
              suspend;
            end
          end
          if (PDT_CODE in ('registration_vnj') -- Вид на жительство (ВНЖ)
              ) then
          begin
            if (not exists(select *
                           from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                           where CDS_FROM_C_ID = :KC_FROM_C_ID and
                                 CDS_FROM_PK_ID = PK_ID and
                                 PK_FROM_PDT_ID = PDT_ID and
                                 CDS_IS_DELETE = 0 and
                                 coalesce(cds_doc_blob_1, 0) > 0 and
                                 PDT_CODE containing 'certificate_approving_doc') and
                --                and dateadd(month, 3, :kf_disb_date) > CDS_DOC_DATE_END) then действительными на момент получения займа документами
                KF_DISB_DATE > CDS_DOC_DATE_END) then
            begin
              res_error_text = 'Для гражд. ТЖ и УЗ ' || PDT_NAME || ' должны быть действительны на момент получения микрозайма';
              res_error_is_critical = 1;
              suspend;
            end
          end
        end
      end
    end
  end
end
