CREATE PROCEDURE HP_APPROVE_LOAN(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID,
    IS_FINSTART D_SMALLINT)
RETURNS (
    RES_TEXT D_STR_LONG,
    RES_NEED_ROLLBACK D_SMALLINT)
AS
declare variable DB_ID integer;
declare variable K_ID decimal(18,0);
declare variable KSR_ID integer;
declare variable REG_ID integer;
declare variable CRITICAL_LEVEL integer;
declare variable K_DISB_DATE date;
declare variable UN_NAME varchar(50);
declare variable C_ID D_ID;
declare variable KC_ID D_ID;
declare variable EIL_STATUS D_SMALLINT;
declare variable MA_ID D_INTEGER;
declare variable ACL_STATUS D_INTEGER;
declare variable EDL_ID D_ID;
declare variable PR_CODE D_STR_100;
declare variable KC_STATUS D_INTEGER;
declare variable IM_ID D_ID;
declare variable IM_TEXT D_STR_255;
declare variable AMOUNT_APPROVED D_CURRENCY;
declare variable KID_RES_TEXT D_STR_255; -- RFCKZ-805
begin
  -- Процедура для утверждения кредита.
  -- Если вызывается из мобильного приложения (через FM_APPROVE_LOAN),
  -- то is_finstart = 0 и не будет запускаться проверка Check loan condition.

  res_need_rollback = 1;

  db_id = f_get_db_id();
  if (is_finstart is null) then is_finstart = 0;

  res_text = '';

  obj_kind = f_objkind(obj_kind, 'name');

  if (obj_kind = 'Miniapp') then
    res_text = iif(db_id <> 94401, 'Отмена. Утвердить можно только договора в процессе', 'Ləğv olunur. Ancaq prossesdə olan müqavilələri təsdiqləmək mümkündür');

  if (res_text = '') then
  begin
    if (obj_kind = 'Clients') then
    begin
      select k_id, k_last_reg_id, k_disb_date
      from kontract, knclient
      where kc_from_k_id = k_id and
            k_status = 0 and
            kc_amount_approved > 0 and
            kc_from_c_id = :obj_id
      order by k_disb_date desc
      rows 1
      into :k_id, reg_id, k_disb_date;
    end

    if (obj_kind = 'Kontract') then
    begin
      select k_id, k_last_reg_id, k_disb_date, pr.pr_code
      from kontract kt
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      where kt.k_id = :obj_id and
            kt.k_status = 0
      into :k_id, reg_id, k_disb_date, pr_code;

      select kc.kc_from_c_id
      from knclient kc
      where kc.kc_from_k_id = :k_id and
            kc.kc_amount_approved > 0
      rows 1
      into c_id;
    end

    if (k_id is null)
    then res_text = iif(db_id <> 94401, 'Отмена. Не найден договор в процессе.', 'Ləğv olunur. Prossesdə olan müqavilə mövcud deyil');
    else
    begin
      obj_kind = 'Kontract';
      obj_id = :k_id;
    end
  end

  -- S.K.
  if (res_text = '' and db_id <> 94401) then
  begin
    acl_status = (select acl_status from hp_actionlist_kontract_get(:k_id));
    if ((acl_status <> 2) and (acl_status <> -4))
      then res_text = 'Нет одобрения от кредитного комитета.';
  end

  if (db_id in (39801, 41701) and
      pr_code containing '.express_similar.' and
      pr_code containing '.auto_kk.') then
  begin
    res_text = '';
    acl_status = -4; --Рассмотрение КК не требуется
  end

  if (res_text = '' and k_disb_date < current_date and
    not exists(select * from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.ApproveLoan_BackDating') where has_access = 1))
  then res_text = iif(db_id <> 94401, 'Отмена. Дата выдачи меньше сегодняшней даты.', 'Ləğv olunur.  Buraxılış tarixi bu günün tarixindən fərqlənir.');

  if (res_text = '' and db_id = 81001 and datediff(day, current_date, k_disb_date) > 15) then
    res_text = 'Дата выдачи не может быть больше чем 15 дней после утверждения займа.';

  -- Проверка прав пользователя
  if (res_text = '') then
  begin
    ksr_id = 0;
    select ksr_id
    from aarep_kontsign_access(:k_id, :un_id, :reg_id, '')
    into :ksr_id;
    if (ksr_id = 0) then
      res_text = iif(db_id <> 94401, 'Отмена. У вас нет прав на утверждение в этом регионе.', 'Ləğv olunur. Sizin bu reqion(filial) üzrə müqavilə təsdiqləmək hüququnuz yoxdur.');
  end

  if (res_text = '') then
  begin
    -- Проверка условий
    critical_level = 0;

    -- Имеет ли право утверждать нестандартные кредиты
    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'Access_To_Approve_NonStandart_Loans')
               where has_access = 1)) then
      critical_level = 1;

    select un_name
    from username
    where un_id = :un_id
    into un_name;

    if (is_finstart = 0) then
    begin
      rdb$set_context('USER_SESSION', 'FINSTART_USER_LOGIN', un_name);

      if (exists(select *   -- добавлено по задаче RFCRU-1500 чтобы утверждающая стороно могла видеть блок из-за лимита
                 from hp_check_loan_konditions(:k_id, 0, 0, 'Loan_Approve', :un_id)
                 where kf_error_is_critical > :critical_level
                 and kf_error_text containing 'Исчерпан лимит на выдачу на'
                 )) then res_text = 'Исчерпан лимит на выдачу на: ' || f_datetostr(k_disb_date);
      else
      if (exists(select *
                 from hp_check_loan_konditions(:k_id, 0, 0, 'Loan_Approve', :un_id)
                 where kf_error_is_critical > :critical_level)) then
        res_text = iif(db_id <> 94401, 'Отмена. Имеются критические несоответствия. Устраните их изучив детали.', 'Ləğv olunur. Kritik uyğunsuzluqlar mövcuddur. Məlumatları dəqiqləşdirib səhvləri aradan qaldırın.');

      rdb$set_context('USER_SESSION', 'FINSTART_USER_LOGIN', null);

    when any do
      begin
        rdb$set_context('USER_SESSION', 'FINSTART_USER_LOGIN', null);
        exception;
      end
    end
  end

  -- RFCRU-5572
  if (res_text = '' and db_id = 81001) then
  begin
    if (c_id is null) then
    begin
      select kc.kc_from_c_id
      from knclient kc
      where kc.kc_from_k_id = :k_id and
            kc.kc_amount_approved > 0
      rows 1
      into :c_id;
    end

    if (c_id is not null) then
    begin
      if (not exists(
        select *
        from clientdocs
          inner join addresstype_registration
            on atr_from_cds_id = cds_id
        where cds_from_c_id = :c_id
          and cds_is_delete = 0))
      then
        res_text = 'Не закреплен код регистрации - обратитесь к КК (на ру мониторинг)';
    end
  end

  if (db_id = 81001) then
  begin
    if (res_text = '') then
    begin
      -- Проверка на банкротство при необходимости
      execute procedure hp_extinfo_insert_kontur('kontract', :k_id, 0, un_id);

      -- Сохранение баллов клиента для договора RFCRU-5380
      execute procedure hp_point_save(9, :k_id);
    end
  end

  if (res_text = '' and db_id in (39801, 41701)) then
  begin
    -- Переподготовка бланков заявки (Анкета ПОФТЭ) в КГ
    edl_id = null;
    select edl_id
    from kontract, edoc_list, edoc_cfg
    where k_id = :k_id and
          edl_from_edc_id = edc_id and
          edc_code = 'loan.application' and
          edl_object_kind = f_objkind('miniapp', 'id') and
          edl_object_id = k_from_ma_app_id and
          edl_status = 1
    into edl_id;
    if (edl_id is not null) then
      execute procedure hp_edoc_blank_approve(edl_id);
  end

  if (res_text = '' and db_id = 81001) then
  begin
    kc_status = null;

    for select kc_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into kc_id
    do
    begin
      select kc_status
      from knclient_pdn
      where kc_id = :kc_id
        and kcp_kind = 0
      into kc_status;

      if (coalesce(kc_status, -1) = -1) then
        res_text = 'По клиенту отсутствует расчет ПДН';
    end
  end

  if (res_text = '') then
  begin
    -- Утверждение кредита
    update kontract
    set k_status = 1,
        k_user_id = :un_id
    where k_id = :k_id;

    -- RFCKZ-805
    -- res_need_rollback здесь еще равен 1, поэтому выход откатывает утверждение
    kid_res_text = '';
    if (db_id = 39801) then
      select res_text
      from hp_edoc_ins_loan_kid(:k_id, :un_id)
      into kid_res_text;

    if (coalesce(kid_res_text, '') <> '') then
    begin
      res_text = 'Отмена. Не сформирован КИД и проект договора: ' || kid_res_text;
      suspend;
      exit;
    end

    if (db_id = 81001 and pr_code containing '.business.'       -- RFCRU-5576 перенес создания "ВС об операции новым ИП" на этап утверждения займа
        and exists(
                   select * from hp_kontur_ip_get_base_info('clients', :c_id)
                   where datediff(month, :k_disb_date, reg_date) < 12)) then
    begin
      im_id = (select im_id        -- Создаем "ВС об операции новым ИП"
               from hp_imessage_create(null, current_date, 0, 1, :un_id, :k_id));

      select res_txt       -- Подписываем "ВС об операции новым ИП"
      from hp_imessage_sign(:un_id, :im_id || ';', 1)
      into im_text;
    end

    res_text = iif(db_id <> 94401, 'Договор успешно утвержден.', 'Müqavilə uğurla təsdiqləndi.');
    res_need_rollback = 0;

    if (db_id = 39801) then
    begin
      select kn.k_from_ma_app_id, kc.kc_from_c_id, kn.k_amount_approved
      from kontract kn
      inner join knclient kc on kc.kc_from_k_id = kn.k_id
      where kn.k_id = :k_id and
            kc.kc_amount_approved > 0
      into :ma_id, :c_id, :amount_approved;

      insert into pkb_miniapp_info(pmi_from_ma_id, pmi_from_c_id, pmi_create_datetime, pmi_status, pmi_is_sent, pmi_sum, pmi_rep_type, pmi_project)
      values (:ma_id, :c_id, current_timestamp, 3, 0, :amount_approved, 1, f_get_rd_id('cib.pkb'));

      insert into pkb_miniapp_info(pmi_from_ma_id, pmi_from_c_id, pmi_create_datetime, pmi_status, pmi_is_sent, pmi_sum, pmi_rep_type, pmi_project)
      values (:ma_id, :c_id, current_timestamp, 3, 0, :amount_approved, 1, f_get_rd_id('cib.gkb'));
    end 
  end

  suspend;
end
