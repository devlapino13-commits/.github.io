CREATE PROCEDURE AAREP_STAJE_IN_MOLBULAK(
    REP_REGION_ID ID_TYPE,
    REP_DATE D_DATE = current_date)
RETURNS (
    EM_REGION STR_100,
    EM_OFFICE STR_100,
    EM_NAME STR_100,
    EM_BIRTH_DATE D_DATE,
    EM_JOB_NAME STR_100,
    EM_STAJ_YEARS D_INTEGER,
    EM_STAJ_MONTHS D_INTEGER,
    EM_WORKED_DAYS D_INTEGER,
    EM_STAJ STR_100,
    PS_CURR_STATUS STR_100,
    EM_STAJ_M_M D_STR_100,
    EM_STAJ_M_D D_INTEGER)
AS
declare variable c_id id_type;
declare variable date_begin d_date;
declare variable date_end d_date;
declare variable prev_date_end d_date;
declare variable date_end_fact d_date;
declare variable worked_days_total d_integer;
declare variable m_y d_integer;
declare variable m_m d_integer;
declare variable m_d d_integer;
declare variable m_d_total d_integer;
declare variable y d_integer;
declare variable m d_integer;
declare variable d d_integer;
declare variable d_total d_integer;
begin

  for select apl.ps_from_c_id,
             apl.branch_name,
             apl.reg_code_name,
             apl.c_name_fio,
             apl.c_birth_date,
             apl.jl_name,
             apl.ps_curr_status
      from aarep_personal_list(:rep_region_id, :rep_date, 1) apl
      order by branch_name,
          reg_code_name,
          c_name_fio,
          c_birth_date
      into c_id,
           em_region,
           em_office,
           em_name,
           em_birth_date,
           em_job_name,
           ps_curr_status
  do
  begin
    -- общий стаж работы
    em_staj_years = 0;
    em_staj_months = 0;
    em_worked_days = 0;
    worked_days_total = 0;

    select c_worked_years,
           c_worked_months,
           c_worked_days,
           c_worked_days_total
    from aarep_clients_info_for_cio(:c_id, :rep_date)
    into em_staj_years,
         em_staj_months,
         em_worked_days,
         worked_days_total;

    -- считаем стаж работы без декретных отпусков
    m_y = 0;
    m_m = 0;
    m_d = 0;
    m_d_total = 0;

    -- перебираем декретные отпуска
    for select hrv.hrv_date_begin,
               hrv.hrv_date_end_fact
        from hr_voctravel hrv
          inner join hr_orders ho on ho.ho_id = hrv.hrv_from_ho_id
        where ho.ho_status = 1 and -- приказ утвержден
              ho.ho_order_type in (17, 18) and -- типы приказов согласно RFCKG-1402
              hrv.hrv_status = 1 and -- отпуск утвержден
              hrv.hrv_from_c_id = :c_id and -- фильтр по сотруднику
              hrv.hrv_date_begin <= :rep_date -- начало отпуска до отчетной даты
        into date_begin, -- первый день отпуска
             date_end -- последний день отпуска
    do
    begin
      y = 0;
      m = 0;
      d = 0;
      d_total = 0;

      -- проверка выхода из отпуска досрочно
      date_end_fact = null;
      select hrv_date_end_fact
      from hr_voctravel hrv
        inner join hr_orders ho on ho.ho_id = hrv.hrv_from_ho_id
      where ho.ho_status = 1 and -- приказ утвержден
            ho.ho_order_type = 20 and -- тип приказа - выход из декретного отпуска
            hrv.hrv_status = 1 and -- отпуск утвержден
            hrv.hrv_from_c_id = :c_id and -- фильтр по сотруднику
            hrv.hrv_date_end_fact between :date_begin and :date_end -- выход из отпуска внутри отпуска
      into date_end_fact; -- первый рабочий день после досрочного выхода из отпуска

      if (date_end_fact < date_end) then
        date_end = date_end_fact - 1; -- был досрочный выход из декретного отпуска

      if (date_end > rep_date) then
        date_end = rep_date; -- если конец отпуска больше отчетной даты, то считаем до отчетной даты

      -- получаем кол-во полных лет/мес/дней в текущем отпуске
      select cnt_years,
             cnt_months,
             cnt_days,
             days_total
      from aarep_staj(:date_begin, :date_end)
      into y,
           m,
           d,
           d_total;

      -- добавляем текущий отпуск к предыдущим
      m_y = m_y + y;
      m_m = m_m + m;
      m_d = m_d + d;
      m_d_total = m_d_total + d_total;

      if (date_begin = prev_date_end + 1) then
        m_d = m_d - 1; -- текущий период идет сразу за предыдущим

      prev_date_end = date_end;
    end

    m_m = m_m + m_d / 30;
    m_d = mod(m_d, 30);
    m_y = m_y + m_m / 12;
    m_m = mod(m_m, 12);

    if (m_d_total = 0) then
    begin
      -- отпускных дней нет, общий стаж и стаж без декретов равны
      y = em_staj_years;
      m = em_staj_months;
      d = em_worked_days;
      d_total = worked_days_total;
    end
    else
    begin
      -- вычитаем из общего стажа сумарный декретный период
      y = em_staj_years - m_y;
      m = em_staj_months - m_m;
      d = em_worked_days - m_d;
      d_total = worked_days_total - m_d_total;

      while (d < 0) do
      begin
        m = m - 1;
        d = d + 30;
      end

      while (m < 0) do
      begin
        y = y - 1;
        m = m + 12;
      end
    end

    m = y * 12 + m;
    em_staj_m_m = f_period_name(0, m, d, 0); -- стаж без декретных (мес+дни)
    em_staj_m_d = d_total; -- стаж без декретных (дни)

    -- Округляем общий стаж: если дней 16 и больше, то обнуляем с увеличением месяца, иначе просто обнуляем
    if (em_worked_days > 15) then
    begin
      em_staj_months = em_staj_months + em_worked_days / 16;
      em_staj_years = em_staj_years + em_staj_months / 12;
      em_staj_months = mod(em_staj_months, 12);
    end
    em_worked_days = 0;

    if (em_staj_years = 0 and em_staj_months = 0) then
      em_staj = '0 мес';
    else
      em_staj = f_period_name(em_staj_years, em_staj_months, em_worked_days, 0);

    suspend;
  end
end
