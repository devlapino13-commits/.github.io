CREATE PROCEDURE HP_ANKETA_ANSWER_GET(
    ACL_ID D_ID,
    QUESTION_ID D_INTEGER,
    ANS_ID D_INTEGER)
RETURNS (
    RES_STATE D_INTEGER,
    RES_STR_VALUE D_STR_2048)
AS
declare variable acg_id d_id;
declare variable acl_object_kind d_integer;
declare variable acl_object_id d_id;
declare variable ovs_id d_id;
begin

  select acl_object_kind, acl_object_id
  from actionlist
  left join actiongroup on acl_from_rd_id = acg_group_rd_id
  where acg_is_delete = 0
  and acl_id = :acl_id
  and acg_from_rd_id = :question_id
  into acl_object_kind, acl_object_id;

  select ovs_id
  from hp_objectprops_get(:acl_object_kind, :acl_object_id, -1, :question_id, current_date)
  into ovs_id;

  select ovr_state, ovr_value
  from objectprops
  inner join objectvalres on ovr_from_opr_id = opr_id and
        ovr_from_ovs_id = :ovs_id and
        ovr_result_rd_id = :ans_id
  where opr_object_kind = :acl_object_kind and
        opr_object_id = :acl_object_id and
        opr_subject_kind = -1 and
        opr_subject_id = :question_id
  into res_state, res_str_value;
  res_state = coalesce(res_state, 0);
  res_str_value = coalesce(res_str_value, '');
  suspend;
end
