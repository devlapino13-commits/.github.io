CREATE PROCEDURE UXP$UXMENULIST_TREE_TMP(
    PARENT_ID D_INTEGER = null)
RETURNS (
    ML_ID D_INTEGER,
    ML_PARENT_ID D_INTEGER,
    ML_CODE D_STR_CODE,
    ML_CAPTION D_STR_255,
    ML_INDEX D_INTEGER,
    ML_IS_FOLDER D_BOOL,
    ML_IS_UX D_BOOL,
    ML_UA_CODE D_STR_100,
    ML_CAN_MOVE D_BOOL,
    ML_UA_UUID_CODE D_STR_CODE,
    ML_FS_DISABLED D_BOOL,
    ML_WEB_DISABLED D_BOOL,
    ML_UA_UUID D_UUID,
    ML_UXA_CAPTION D_STR_255,
    ML_POSITION D_INTEGER)
AS
declare variable tmp_index d_integer;
begin
  ml_ua_code = null;
  ml_can_move = 0;
  tmp_index = -1;
  for select ml_id, ml_parent_id, ml_code, ml_caption, ml_index, iif(ml_kind = 0, 0, 1),
             uml_ua_uuid, uml_fs_disabled, uml_web_disabled, iif(ml_kind = 0, ml_has_child, iif(ml_kind = 1, 1, 0)),
             ml_position
  from menulist left join ux$menulist on (ml_id = uml_ml_id)
  where ml_parent_id is not distinct from :parent_id
  order by ml_position, ml_index
  into :ml_id, :ml_parent_id, :ml_code, :ml_caption, :ml_index, :ml_is_ux,
       :ml_ua_uuid, :ml_fs_disabled, :ml_web_disabled, :ml_is_folder, :ml_position
  do
  begin
    tmp_index = tmp_index + 1;

    ml_can_move = ml_is_ux;
    if (ml_id > 0) then
      ml_can_move = 1;

    if (coalesce(ml_index, -1) = -1) then
    begin
      if (ml_id > 0) then
      begin
        select rl_order_no
        from reportlist
        where rl_id = :ml_id
        into ml_index;
      end
    end

    if (coalesce(ml_index, -1) = -1) then
      ml_index = tmp_index;
    else
      tmp_index = ml_index;

    ml_ua_code = null;
    ml_ua_uuid_code = '';
    ml_uxa_caption = '';

    if (ml_ua_uuid is not null) then
    begin
      select ua_name, ua_uxa_fcode
      from ux$action
      where ua_uuid = :ml_ua_uuid
      into ml_uxa_caption, ml_ua_code;
  
      ml_ua_uuid_code = lower(uuid_to_char(ml_ua_uuid));
      ml_is_ux = 1;
      ml_code = 'uxmenulist_' || replace(uuid_to_char(ml_ua_uuid), '-', '_');
    end

    suspend;
  end
end
