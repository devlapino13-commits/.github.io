CREATE PROCEDURE AAREP_ORIV_CONTRACTS(
    REP_REGION_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    K_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    KAL_NEW_KF_NAME D_STR_100,
    SB_ID D_ID,
    SB_FIO D_STR_100,
    STATUS D_STR_10)
AS
begin
    for select distinct
        sbl.sbl_id,
        sbl.sbl_name,
        k.k_kontract_no,
        k.k_id,
        knl.kal_new_kf_name,
        iif(oks.oks_close_state = 0, 'Активный', 'Спец. учёт')
    from sb_list sbl
    inner join sb_akt sba on sba.sba_from_sbl_id = sbl.sbl_id
        and sba.sba_is_delete = 0
        and sba.sba_status = 2
        and :rep_date between sba.sba_date_begin and sba.sba_date_end
    inner join kontract k on sba.sba_from_k_id = k.k_id
    inner join knaktlist knl on k.k_from_kf_id = knl.kal_from_kf_id
            and knl.is_delete = 0
            and :rep_date between knl.kal_date and knl.kal_date_end
    inner join oksummary oks on sba.sba_from_k_id = oks.oks_from_k_id
        and :rep_date between oks.oks_date_begin and oks.oks_date_end 
        and oks.oks_close_state in (0,1)
    inner join region_list(:rep_region_id, 1, 3, 1) on id_list = k.k_fact_disb_reg_id
    where sbl.sbl_type = 1
        and :rep_date between sbl.sbl_date_begin and coalesce(sbl.sbl_date_end, current_date)
    order by
        sbl.sbl_id
    into
        :sb_id,
        :sb_fio, 
        :k_kontract_no,
        :k_id,
        :kal_new_kf_name,
        :status
    do
        suspend;

end
