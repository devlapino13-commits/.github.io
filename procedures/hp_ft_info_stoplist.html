CREATE PROCEDURE HP_FT_INFO_STOPLIST
RETURNS (
    FISL_ID D_ID,
    FISL_FROM_C_ID D_ID,
    C_NAME_FIO D_STR_100,
    FISL_DATE_BEGIN D_DATE,
    FISL_DATE_END D_DATE,
    FISL_FOR_CHECK_DAYS_TXT D_STR_LONG,
    FISL_FOR_UPDATE_TXT D_STR_LONG,
    FISL_FROM_UN_ID D_ID,
    UN_FIO D_STR_100,
    FISL_CREATE_DATETIME D_DATETIME,
    FISL_CATEGORY_ID D_INTEGER,
    RD_NAME D_STR_255,
    FISL_FREEZE_DATE_BEGIN D_DATE,
    FISL_FREEZE_DATE_END D_DATE,
    FISL_FREEZE_TXT D_STR_LONG)
AS
begin
  for
    select
      fisl_id,
      fisl_from_c_id,
      c_name_fio,
      fisl_date_begin,
      fisl_date_end,
      fisl_for_check_days_txt,
      fisl_for_update_txt,
      fisl_from_un_id,
      un_fio,
      fisl_create_datetime,
      fisl_category_id,
      rd_name,
      fisl_freeze_date_begin,
      fisl_freeze_date_end,
      fisl_freeze_txt
    from Ft_Info_Stoplist
      inner join Clients
        on c_id = fisl_from_c_id
      inner join Username
        on un_id = fisl_from_un_id
      inner join Refdata
        on rd_id = fisl_category_id
  into
      :fisl_id,
      :fisl_from_c_id,
      :c_name_fio,
      :fisl_date_begin,
      :fisl_date_end,
      :fisl_for_check_days_txt,
      :fisl_for_update_txt,
      :fisl_from_un_id,
      :un_fio,
      :fisl_create_datetime,
      :fisl_category_id,
      :rd_name,
      :fisl_freeze_date_begin,
      :fisl_freeze_date_end,
      :fisl_freeze_txt
  do suspend;
end
