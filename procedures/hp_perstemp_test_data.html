CREATE PROCEDURE HP_PERSTEMP_TEST_DATA(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    IS_VERIFIED D_SMALLINT,
    CLIENT_TYPE D_SMALLINT,
    RES_NAME D_STR_255,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    C_RESULT D_STR_CODE)
AS
begin
  if (rep_date_begin is null or rep_date_end is null) then
  begin
    rep_date_begin = '01.01.1990';
    rep_date_end = '31.12.1990';
  end

  is_verified = 1;
  while (is_verified in (1, 2)) do
  begin
    c_result = '-';
    client_type = 1; -- Без истории
    for select c_id, c_name_fio, c_birth_date, c_id || ', ' || c_name_fio || ', ' || f_datetostr(c_birth_date)
    from clients, clientstate
    where cs_from_c_id = c_id and
          c_birth_date between :rep_date_begin and :rep_date_end and
          ((:is_verified = 1 and c_verify_status <> 2) or
           (:is_verified = 2 and c_verify_status = 2)) and
          cs_client_state = -1 and
          cs_staff_state = -1
    order by cs_date desc
    into c_id, c_name_fio, c_birth_date, res_name
    do
    begin
      if (exists(select res_check from hp_perstemp_test_check(:c_id)
                 where res_check = 0)) then
        Continue;
  
      c_result = 'Найден';
      suspend;
      break;
    end

    if (c_result = '-') then
      Suspend;
  
    c_result = '-';
    client_type = 2; -- Активные
    for select c_id, c_name_fio, c_birth_date, c_id || ', ' || c_name_fio || ', ' || f_datetostr(c_birth_date)
    from clients, clientstate
    where cs_from_c_id = c_id and
          c_birth_date between :rep_date_begin and :rep_date_end and
          ((:is_verified = 1 and c_verify_status <> 2) or
           (:is_verified = 2 and c_verify_status = 2)) and
          cs_client_state = 1
    order by cs_date desc
    into c_id, c_name_fio, c_birth_date, res_name
    do
    begin
      if (exists(select res_check from hp_perstemp_test_check(:c_id)
                 where res_check = 0)) then
        Continue;

      c_result = 'Найден';
      suspend;
      break;
    end

    if (c_result = '-') then
      Suspend;
  
    c_result = '-';
    client_type = 3; -- Закрытые
    for select c_id, c_name_fio, c_birth_date, c_id || ', ' || c_name_fio || ', ' || f_datetostr(c_birth_date)
    from clients, clientstate
    where cs_from_c_id = c_id and
          c_birth_date between :rep_date_begin and :rep_date_end and
          ((:is_verified = 1 and c_verify_status <> 2) or
           (:is_verified = 2 and c_verify_status = 2)) and
          cs_client_state = 2
    order by cs_date desc
    into c_id, c_name_fio, c_birth_date, res_name
    do
    begin
      if (exists(select res_check from hp_perstemp_test_check(:c_id)
                 where res_check = 0)) then
        Continue;

      c_result = 'Найден';
      suspend;
      break;
    end

    if (c_result = '-') then
      Suspend;
  
    c_result = '-';
    client_type = 4; -- Сотрудник без кредитной истории
    for select c_id, c_name_fio, c_birth_date, c_id || ', ' || c_name_fio || ', ' || f_datetostr(c_birth_date)
    from clients, clientstate
    where cs_from_c_id = c_id and
          c_birth_date between :rep_date_begin and :rep_date_end and
          ((:is_verified = 1 and c_verify_status <> 2) or
           (:is_verified = 2 and c_verify_status = 2)) and
          cs_client_state = -1 and
          cs_staff_state = 1
    order by cs_date desc
    into c_id, c_name_fio, c_birth_date, res_name
    do
    begin
      if (exists(select res_check from hp_perstemp_test_check(:c_id)
                 where res_check = 0)) then
        Continue;

      c_result = 'Найден';
      suspend;
      break;
    end

    if (c_result = '-') then
      Suspend;


    c_result = '-';
    client_type = 5; -- ТиЭ

    for select c.c_id, c.c_name_fio, c.c_birth_date, c.c_id || ', ' || c.c_name_fio || ', ' || f_datetostr(c.c_birth_date)
    from clients c, aarep_check_finpolis_tmp(null, 0) t
    where c.c_id = t.c_id and
          ((:is_verified = 1 and c_verify_status <> 2) or
           (:is_verified = 2 and c_verify_status = 2))
    into c_id, c_name_fio, c_birth_date, res_name
    do
    begin
      if (exists(select res_check from hp_perstemp_test_check(:c_id)
                 where res_check = 0)) then
        Continue;

      c_result = 'Найден';
      suspend;
      break;
    end

    if (c_result = '-') then
      Suspend;

    is_verified = is_verified + 1;
  end
end
