CREATE PROCEDURE HP_CUSTOMDATACFG_ONE(
    OBJECT_KIND INTEGER,
    OBJECT_SUBKIND INTEGER,
    OBJECT_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    CDC_ID INTEGER,
    CDC_CODE VARCHAR(255),
    CDC_CAPTION VARCHAR(255),
    CDC_OBJECT_KIND INTEGER,
    CDC_DATA_TYPE VARCHAR(50),
    CDC_CATEGORY VARCHAR(255),
    CDC_ORDER_NO INTEGER,
    CDC_COMBO_TEXTS VARCHAR(1024),
    CDC_IS_DELETE INTEGER,
    CDC_MEMO VARCHAR(255),
    CDV_VALUE_NUMBER DECIMAL(15,2),
    CDV_VALUE_DATE DATE,
    CDV_VALUE_STRING VARCHAR(255))
AS
begin
  if (rep_date is null) then
    rep_date = cast('31.12.9999' as date) - 1;
  for select cdc_id, cdc_code, cdc_caption, cdc_object_kind, cdc_data_type, cdc_category, cdc_order_no, cdc_combo_texts,
             cdc_is_delete, cdc_memo
      from customdatacfg
      where cdc_object_kind = :object_kind and
            cdc_object_subkind = :object_subkind and
            cdc_is_delete = 0
      order by cdc_category, cdc_order_no, cdc_id
      into :cdc_id, :cdc_code, :cdc_caption, :cdc_object_kind, :cdc_data_type, :cdc_category, :cdc_order_no,
           :cdc_combo_texts, :cdc_is_delete, :cdc_memo
  do
  begin
    cdv_value_number = null;
    cdv_value_date = null;
    cdv_value_string = null;

    if (cdc_data_type = 'number') then
    begin
      select cdv_value_number
      from customdatavalue
      where cdv_object_kind = :object_kind and
            cdv_object_id = :object_id and
            cdv_from_cdc_id = :cdc_id and
            cdv_date_begin <= :rep_date and
            cdv_date_end >= :rep_date rows 1
      into :cdv_value_number;
    end
    if (cdc_data_type = 'date') then
    begin
      select cdv_value_date
      from customdatavalue
      where cdv_object_kind = :object_kind and
            cdv_object_id = :object_id and
            cdv_from_cdc_id = :cdc_id and
            cdv_date_begin <= :rep_date and
            cdv_date_end >= :rep_date rows 1
      into :cdv_value_date;
    end
    if (cdc_data_type = 'string') then
    begin
      select cdv_value_string
      from customdatavalue
      where cdv_object_kind = :object_kind and
            cdv_object_id = :object_id and
            cdv_from_cdc_id = :cdc_id and
            cdv_date_begin <= :rep_date and
            cdv_date_end >= :rep_date rows 1
      into :cdv_value_string;
    end
    suspend;
  end
end
