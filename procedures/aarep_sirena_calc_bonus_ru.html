CREATE PROCEDURE AAREP_SIRENA_CALC_BONUS_RU(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    CC_PROJECT STR_100,
    OPER_CALLED STR_100,
    PAYED_SUM_ALL_VIPS CURR_TYPE,
    PAYED_SUM_ALL_SIRENA CURR_TYPE,
    PAYED_SUM_ALL CURR_TYPE,
    PAYED_QTY INTEGER,
    CONTRACTS_BEGIN INTEGER,
    CONTRACTS_RECEIPTS INTEGER,
    CONTRACTS_CUR_UNSUCCESSFUL INTEGER,
    CONTRACTS_RECALLED INTEGER,
    EFFICIENCY PERC_TYPE,
    BONUS_FOLDER_SVZ CURR_TYPE,
    BONUS_FOLDER_SIRENA CURR_TYPE,
    BONUS_FOR_PERFORMANCE CURR_TYPE,
    BONUS_TO_ACCRUAL CURR_TYPE)
AS
declare variable rep_project str_100;
declare variable cc_project_salary curr_type;
declare variable perc_recovery_svz perc_type;
declare variable perc_recovery_sirena perc_type;
declare variable for_efficiency curr_type;
declare variable recovery_plan curr_type;
declare variable not_payed_1 integer;
declare variable not_payed_2 integer;
begin
  /* Процедура создана для расчета бонусов по КЦ Сирена!
  Дата начало разработки: 11-июня 2018-года
  Разработчик: Рахманов Музаффар */

  cc_project_salary = 9000;
  perc_recovery_svz = 0; --Взыскание папок СВЗ
  perc_recovery_sirena = 0; --Взыскание папок Сирена 1
  recovery_plan = 0;
  /*
  REP_TXT = 'Расчет бонусов за период с ' || --Дата начало(далее внизу)
  (select SVAL
   from DATETOSTR(:REP_DATE_BEGIN)) --
  || ' по ' || --Дата конца(далее внизу)
  (select SVAL
   from DATETOSTR(:REP_DATE_END));
                                      */
  rep_project = 'OKTELL.PROJECTS.SIRENA';
  for --
      select cc_project, oper_called, --
             sum(payed_sum) payed_sum_all, --
             sum(iif(project = 'Сирена', payed_sum, 0)) payed_sum_all_sirena, --
             sum(iif(project = 'Сирена', 0, payed_sum)) payed_sum_all_vips, --
             count(distinct k_kontract_no) payed_qty
      from aarep_sirena_payed_rep(:rep_project, :rep_date_begin, :rep_date_end, 0)
      where coalesce(oper_called, 'IGUSEV') <> 'IGUSEV'
      group by cc_project, oper_called
      into cc_project, oper_called, payed_sum_all, payed_sum_all_sirena, payed_sum_all_vips, payed_qty
  do
  begin
    bonus_folder_svz = 0; -- Бонус за папки СВЗ
    bonus_folder_sirena = 0; --Бонус за папки Сирены
    bonus_for_performance = 0; --Бонус за эффективность
    bonus_to_accrual = 0; --Бонус к начислению

    contracts_begin = 0; --Кол-во договоров на начало
    contracts_receipts = 0; --Поступления договоров
    contracts_cur_unsuccessful = 0; --Текущие неуспешные
    contracts_recalled = 0; --Отозванные в СБ (без текущих)
    not_payed_1 = 0;
    not_payed_2 = 0;

    select count(*)
    from pantera_kontracts pk, refdata rd, oksummary oks
    where :rep_date_begin - 1 between pk.pant_date_begin and pk.pant_date_end and
          pk.pant_oper_called = :oper_called and
          pk.pant_rd_project_type = rd.rd_id and
          rd.rd_code containing :rep_project and
          pk.pant_from_k_id = oks_from_k_id and
          :rep_date_begin - 1 between oks_date_begin and oks_date_end and
          :rep_date_begin - 1 > oks_delinq_date
          and oks_close_state = 0
    into contracts_begin;

    with t1
    as (select pant_from_k_id, pant_date_begin
        from pantera_kontracts pk, refdata rd
        where :rep_date_end between pk.pant_date_begin and pk.pant_date_end and
              pk.pant_oper_called = :oper_called and
              pk.pant_rd_project_type = rd.rd_id and
              rd.rd_code containing :rep_project and
              :rep_date_end < pant_date_begin + 8)
    select count(*) contracts_cur_unsuccessful
    from t1
    where not exists(select *
                     from operkred ok
                     where pant_from_k_id = ok_from_k_id and
                           ok_type = 4 and
                           ok_ol_is_active = 1 and
                           ok_ol_date between pant_date_begin and :rep_date_end)

    into contracts_cur_unsuccessful;

    select count(distinct pk.pant_from_k_id), --
           sum(iif(:rep_date_end > pant_date_begin + 8, 1, 0))
    from pantera_kontracts pk, refdata rd
    where pk.pant_date_begin between :rep_date_begin and :rep_date_end and
          pant_date_begin <= pant_date_end and
          pk.pant_oper_called = :oper_called and
          pk.pant_rd_project_type = rd.rd_id and
          rd.rd_code containing :rep_project
    into contracts_receipts, --разница на 1 контракт за май, нужно найти и прописать доп.условие
         not_payed_2; --не используется

    contracts_cur_unsuccessful = coalesce(contracts_cur_unsuccessful, 0);
    efficiency = 0; --Эффективность
    /*
        SYSDBA

        |*Эффективность = Кол-во договоров с успешными погашениями / Общее кол-во договоров  в работе за период
        кроме текущих** неуспешных -Неправильный
        **Текущая папка = папка в работе менее 8 дней

        Новая формула от Аюб:
        Эффективность сотрудника =
Количество договоров на начало+Поступления договоров-Текущие неуспешные%Количество погашенных договоров

    */
--Эффек.=(колличеств.взыска.договоров/ (договора на начало+ поступление договоров)- текущее не успешные)*100
    efficiency = (payed_qty / (contracts_begin + contracts_receipts) - contracts_cur_unsuccessful);

    if (cc_project = 'Сирена 1') then
    begin
      /*
            SYSDBA
            |Для операторов Сирена 1:
Оклад: 
9 000 сомов. В него входит 500 000 руб. взыскание по папкам действующих СВЗ.
Бонусы: 
- за взыскание свыше 500 000 руб. 1% за папки уволенных СВЗ, и 0,5% за папки действующих СВЗ.
- 250 руб. за эффективность свыше 70% (за каждый 1%)
Штраф:
- минус 250 руб. за эффективность ниже 70% (за каждый 1%)

        */
      recovery_plan = coalesce(payed_sum_all_vips, 0) - 500000;
      perc_recovery_svz = 0.5; --0,5% за папки действующих СВЗ
      perc_recovery_sirena = 1; --1% за папки уволенных СВЗ (папки Сирена)
      for_efficiency = 70; --Бонус за эффективность если больше 70%
    end
    else
    if (cc_project = 'Сирена 2') then
    begin
      recovery_plan = coalesce(payed_sum_all_vips, 0) - 100000;
      perc_recovery_svz = 1.3;
      perc_recovery_sirena = 2.6;
      for_efficiency = 32.5; --Бонус за эффективность если больше 32.5%
    end

    perc_recovery_svz = perc_recovery_svz / 100;
    perc_recovery_sirena = perc_recovery_sirena / 100;

    bonus_folder_svz = round(recovery_plan * perc_recovery_svz); -- Бонус за папки СВЗ
    bonus_folder_sirena = round(payed_sum_all_sirena * perc_recovery_sirena); --Бонус за папки Сирены
    bonus_for_performance = (efficiency - for_efficiency) * 250; --Бонус за эффективность
    bonus_to_accrual = bonus_folder_svz + bonus_folder_sirena + bonus_for_performance; --Бонус к начислению

    if (bonus_to_accrual < 0) then
      bonus_to_accrual = 0;

    suspend;
  end
end
