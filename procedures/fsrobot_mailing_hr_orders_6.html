CREATE PROCEDURE FSROBOT_MAILING_HR_ORDERS_6(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable c_name_fio varchar(100);
declare variable reg_name varchar(100);
declare variable jl_name varchar(100);
declare variable ps_date date;
declare variable ho_come_back_date date;
declare variable c_login varchar(100);
declare variable res_tmp_bode blob sub_type 1 segment size 80;
declare variable c_email varchar(100);
declare variable c_cell_no varchar(100);
declare variable c_id decimal(18,0);
declare variable kol integer;
declare variable hr_order_no varchar(20);
declare variable date_str varchar(255);
declare variable date_format_str varchar(20);
declare variable jl_id d_id;
declare variable db_id integer;
declare variable ps_from_reg_id integer;
declare variable quick_mail_to varchar(100);
begin
  -- Процедура вызывается из FSROBOT_ON_MAILING
  -- Процедура готовит список уведомлений для отправки роботом по почте или sms
  -- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
  -- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
  -- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

  -- RefData.RD_Memo    - Список получателей
  -- RefData.RD_Tag     - Срок жизни сообщения
  -- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

  -- Rep_Params - входящий параметры в зависимости от команды
  -- Rep_UN_ID - пользователь
  :db_id = f_get_db_id();
  rep_params = (trim(rep_params));
  rep_email = (trim(rep_email));
  res_maillist = :rep_email;
  res_c_id = 0;

  -- Адрес службы поддержки для отдельного короткого письма
  quick_mail_to = case :db_id
                    when 39801 then 'help@mbulak.kz'
                    when 41701 then 'help@mbulak.kg'
                    else null
                  end;

  -- Если этот адрес уже входит в список рассылки робота (RefData.RD_Memo),
  -- короткое письмо не шлём: длинное письмо содержит те же данные и больше,
  -- иначе получатель получает два письма на один приказ
  if (:quick_mail_to is not null and
      position(lower(:quick_mail_to) in lower(coalesce(:res_maillist, ''))) > 0) then
    quick_mail_to = null;

  -- HR_Orders_6 - Увольнение
  res_body = '<b>Внимание !</b><br>';
  res_body = BLOB_APPEND(res_body, '<br>');

  res_tmp_bode = '';
  kol = 0;
  for select c_id, c_name_fio, reg_code || ' ' || reg_name, jl_name, jl_id, ps_date_begin,
             (select cont_value
              from hp_contdata_get_email_main(0, c_id)),
             (select cont_value
              from hp_contdata_get_cell_main(0, c_id)), ho_order_no, ho.ho_come_back_date, ps_from_reg_id
      from hr_pstatus, clients, region, joblist, hr_orders ho
      where ps_from_ho_id = :rep_params and
            ps_status = 1 and
            ps_from_c_id = c_id and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id and
            ps_from_ho_id = ho_id
      into c_id, c_name_fio, reg_name, jl_name, jl_id, ps_date, c_email, c_cell_no, hr_order_no, ho_come_back_date, ps_from_reg_id
  do
  begin
    -- КЗ: ГО - все должности, регионы - только МО (jl_id = 9)
    if (:db_id = 39801) then
    begin
      if (:ps_from_reg_id <> 1010100 and :jl_id <> 9) then
        continue;
    end

    if (c_email = '') then
      c_email = 'Отсутствует';
    if (c_cell_no = '') then
      c_cell_no = 'Отсутствует';

    ps_date = ps_date - 1;
    date_format_str = right('0' || extract(day from ps_date), 2) || '.' || right('0' || extract(month from ps_date), 2) || '.' || extract(year from ps_date);

    kol = kol + 1;
    c_login = '';
    select iif(un_is_active = 1, un_name || ', доступ актуальный', coalesce(un_name || ' - ', '') || 'Доступ закрыт')
    from hp_client_un_is_active(:c_id)
    into :c_login;
    if (c_login = '') then
      c_login = 'Отсутствует';

    date_str = 'Последний рабочий день: <B>' || date_format_str || '</B>';

    if (ho_come_back_date is not null) then
      res_tmp_bode = res_tmp_bode || '<B>Техническое увольнение сотрудника</B><br>';
    res_tmp_bode = res_tmp_bode || 'ФИО: <B>' || c_name_fio || '</B><br>';
    res_tmp_bode = res_tmp_bode || 'ОП: <B>' || reg_name || '</B><br>';
    res_tmp_bode = res_tmp_bode || 'Должность: <B>' || jl_name || '</B><br>';
    res_tmp_bode = res_tmp_bode || 'Сотовый номер в БД: <B>' || c_cell_no || '</B><br>';
    res_tmp_bode = res_tmp_bode || 'Логин в FinStart: <B>' || c_login || '</B><br>';
    res_tmp_bode = res_tmp_bode || 'Email в БД: <B>' || c_email || '</B><br>';
    res_tmp_bode = res_tmp_bode || date_str || '<br>';
    if (:jl_id in (225, 226, 227, 240)) then
      res_tmp_bode = res_tmp_bode || '<B>' || '<font color="red"> Закрыть доступ к порталу Тундук </font></B><br>';

    if (:quick_mail_to is not null and :jl_id not in (226, 227)) then
      execute procedure hp_send_quick_message('email', :quick_mail_to,
            'Увольнение сотрудника',
            'ФИО: ' || :c_name_fio || '<br>' || 'Последний рабочий день: ' || :date_format_str || '<br>' || 'ОП: ' || :reg_name || '<br>' || 'Должность: ' || :jl_name || '<br>' || 'Сотовый номер в БД: ' || :c_cell_no || '<br>' || 'Email в БД: ' || :c_email,
            '', 1, 1000000000000);

    if (:db_id in (39801, 41701) and
        exists(select *
               from tranmembers tb, tranmain tm
               where tb_is_trainer = 0 and
                     tm_type <> 0 and
                     tm_id = tb_from_tm_id and
                     tm_date_end >= :ps_date - 365 and
                     tb_from_c_id = :c_id)) then
      res_tmp_bode = res_tmp_bode || '<B>' || '<font color="red">Участвовал в тренингах в течении года</font></B><br>';

    res_tmp_bode = res_tmp_bode || '<br>';
  end

  if (kol = 0) then
    exit;

  res_body = '<b>Внимание !</b><br>';
  res_body = BLOB_APPEND(res_body, '<br>');

  if (kol = 1) then
  begin
    res_subject = 'Уволен сотрудник, приказ № ' || hr_order_no;
    res_body = BLOB_APPEND(res_body, '<b>Уволен сотрудник:</b><br>');
  end
  else
  begin
    res_subject = 'Уволены сотрудники, приказ № ' || hr_order_no;
    res_body = BLOB_APPEND(res_body, '<b>Уволены сотрудники:</b><br>');
  end
  res_body = BLOB_APPEND(res_body, '<br>');
  res_body = BLOB_APPEND(res_body, res_tmp_bode);
  res_body = BLOB_APPEND(res_body, '<br>');
  if (user_fio <> '') then
  begin
    res_body = BLOB_APPEND(res_body, 'Автор изменений: ' || user_fio);
    res_body = BLOB_APPEND(res_body, '<br>');
    res_body = BLOB_APPEND(res_body, '<br>');
  end
  res_body = BLOB_APPEND(res_body, 'Всем службам произвести необходимые действия!');
  res_body = BLOB_APPEND(res_body, '<br>');

  suspend;
end
