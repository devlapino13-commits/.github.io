CREATE PROCEDURE EQUIFAX_GET_KNCLIENT_INFO(
    KC_ID ID_TYPE)
RETURNS (
    LASTNAME D_STR_100,
    FIRSTNAME D_STR_100,
    MIDDLENAME D_STR_100,
    GENDER D_SMALLINT,
    BIRTHDAY D_DATE,
    BIRTHPLACE D_STR_255,
    DOCTYPE D_STR_20,
    DOCNO D_STR_20,
    DOCDATE D_DATE,
    DOCENDDATE D_DATE,
    DOCPLACE D_STR_255,
    INN D_STR_20,
    PFNO D_STR_100,
    REASON D_SMALLINT,
    CONSENT D_SMALLINT,
    CONSENTDATE D_DATE,
    CONSENTENDDATE D_DATE,
    ADMCODE_INFORM D_SMALLINT,
    INCOME D_CURRENCY,
    APPLICATION_NUM D_STR_20,
    APPLICATION_DATE D_DATE,
    CRED_TYPE D_STR_10,
    CRED_CURRENCY D_STR_10,
    CRED_SUM D_CURRENCY,
    CRED_FREQUENCY_PAYMENT D_SMALLINT,
    CRED_DURATION D_INTEGER,
    CITIZENSHIP D_STR_100,
    MARRIAGE D_SMALLINT,
    DEPENDANTS_BEL18 D_SMALLINT,
    DEPENDANTS_UND18 D_SMALLINT,
    EDUCATION D_SMALLINT,
    PHONE_MOBILE D_STR_100,
    PHONE_HOME D_STR_100,
    PHONE_WORK D_STR_100,
    EMAIL D_STR_100,
    PROFESSION D_STR_100,
    COMPANY_NAME D_STR_100,
    REG_OWNER D_SMALLINT,
    REG_INDEX D_STR_100,
    ADDR_REG_TOTAL D_STR_255,
    REG_COUNTRY D_STR_10,
    REG_REGION D_STR_10,
    REG_DISTRICT D_STR_100,
    REG_STREET D_STR_100,
    REG_HOUSE D_STR_100,
    REG_FLAT D_STR_100,
    FACT_OWNER D_SMALLINT,
    FACT_INDEX D_STR_100,
    ADDR_FACT_TOTAL D_STR_255,
    FACT_COUNTRY D_STR_10,
    FACT_REGION D_STR_10,
    FACT_DISTRICT D_STR_100,
    FACT_STREET D_STR_100,
    FACT_HOUSE D_STR_100,
    FACT_FLAT D_STR_100)
AS
declare variable c_id id_type;
declare variable k_id id_type;
declare variable pk_id id_type;
declare variable adl_from_pl_id id_type;
declare variable adl_street_id id_type;
declare variable adl_post_index d_str_100;
declare variable pl_post_index d_str_100;
declare variable street_post_index d_str_100;
declare variable adc_type d_smallint;
declare variable iindex d_str_100;
declare variable address d_str_255;
declare variable owner d_smallint;
declare variable country d_str_10;
declare variable pl_fs_code d_str_10;
declare variable rregion d_str_10;
declare variable street_buf d_str_100;
declare variable plt_name d_str_100;
declare variable district d_str_100;
declare variable street d_str_100;
declare variable house d_str_100;
declare variable flat d_str_100;
begin

  reason = 0; -- 0    Заключение и исполнение договора
  consent = 1; -- 1    Согласие клиента
  consentenddate = null;
  admcode_inform = 1; -- 1    Пользователь КИ проинформирован об административной ответственности
  cred_frequency_payment = 1; --      Периодичность платежей    Количество месяцев между платежами.
  cred_type = '05'; --      Потребительский кредит

  select kncl.kc_from_c_id, kncl.kc_from_k_id, fl.fl_name1, fl.fl_name2, fl.fl_name3, cl.c_sex, cl.c_birth_date,
         trim(replace(coalesce(cl.c_birth_country, cl.c_birth_city, ''), '`', '')), cl.c_doc_type, cl.c_pass_no,
         cl.c_pass_date, cl.c_pass_date_expire, replace(replace(cl.c_pass_who, '`', ''), '*', ''),
         coalesce(cl.c_inn, ''), cl.c_citizen, cl.c_family, cl.c_education, cl.c_kol_child, cl.c_kol_family
  from knclient kncl
  inner join clients cl on cl.c_id = kncl.kc_from_c_id
  inner join fiolist fl on fl.fl_id = cl.c_from_fl_id
  where kncl.kc_id = :kc_id
  into c_id, k_id, lastname, firstname, middlename, gender, birthday, birthplace, doctype, docno, docdate, docenddate,
       docplace, inn, citizenship, marriage, education, dependants_bel18, -- Количество иждивенцев до 18 лет включительно
       dependants_und18; -- Количество иждивенцев старше 18 лет

----------------------------------------------------------------------------

  phone_mobile = f_cont_info('clients', c_id, 'cell_main', 'ok');
  phone_home = f_cont_info('clients', c_id, 'phone_home', 'ok');
  phone_work = f_cont_info('clients', c_id, 'phone_work', 'ok');
  email = f_cont_info('clients', c_id, 'email', 'ok');

  select pask.pk_id
  from passkind pask
  where pask.pk_id = :doctype
  into :pk_id;

  if (citizenship = 13) then -- рф
  begin
    if ((pk_id = 12) or (pk_id = 13)) then
      doctype = 1; -- паспорт гражданина рф
    else
      doctype = 14; -- неизвестно
  end
  else
    doctype = 6; -- паспорт, выданный иностранным государством и признаваемый в российской федерации

  if (birthplace = '') then -- << помни!
  begin
    select ddr.rd_name
    from refdata ddr
    where ddr.rd_id = :citizenship
    into :birthplace; --  мето роджения беру исходя из гражданства
  end

  citizenship = case citizenship
                  when 13 then 'RU'
                  when 9 then 'UZ'
                  when 12 then 'TJ'
                  when 8 then 'KG'
                  else 99
                end;

  gender = case gender
             when 0 then 2 --    Женский
             when 1 then 1 --    Мужской
             else 9 --    неизвестно
           end;

  pfno = null;
  select cds_doc_no
  from clientdocs, passkind, passdoctype
  where cds_from_c_id = :c_id and
        cds_from_pk_id = pk_id and
        pk_from_pdt_id = pdt_id and
        pdt_code = 'snils' and
        cds_is_delete = 0
  order by cds_from_c_id desc, cds_doc_date_begin desc
  rows 1
  into :pfno;
  pfno = coalesce(pfno, '');
  pfno = replace(pfno, '-', '');
  pfno = replace(pfno, ' ', '');

  select k.k_applicate_date, k.k_kontract_no, k.k_kontract_date, k.k_currency, k.k_amount_approved, -------  или на уровне клиента?
         k.k_period

  from kontract k
  where k.k_id = :k_id
  into :consentdate, :application_num, :application_date, :cred_currency, :cred_sum, :cred_duration;

  select cdv_value_number
  from hp_customdatacfg_one(0, 1, :c_id, current_date)
  where cdc_code = 'work_yield_sum'
  into :income;

  marriage = case marriage
               when 14 then 0 -- Холост / Не замужем
               when 15 then 1 -- Женат / Замужем
               when 16 then 2 -- Разведен / Разведена
               when 17 then 3 -- Вдовец / Вдова
               when 50 then 4 -- Гражданский брак / Совместное проживание
               else 9 -- Неизвестно
             end;

  education = case education
                when 18 then 0 -- Начальная школа
                when 19 then 1 -- Средняя школа
                when 20 then 2 -- Специализированная средняя школа
                when 21 then 3 -- Незаконченное высшее образование
                when 22 then 4 -- Высшее образование
--when then 5    -- Два и более высших образования
                when 23 then 6 -- Ученая степень
--when then 8    -- Другое
                else 9 -- Неизвестно
              end;

  select ovs_value
  from hp_objectprops_get(0, :c_id, -1, 'work_place_name', current_date)
  into :company_name;
  select ovs_value
  from hp_objectprops_get(0, :c_id, -1, 'work_job_name', current_date)
  into :profession;

----------------------------------------------------------------------------  address

  for select coalesce(adl.adl_post_index, 0), adl.adl_from_pl_id, adl.adl_street_id, adl.adl_address_str,
             adcl.adc_address_type, adc_is_rent, substring(cast(adl.adl_id as str_255) from 1 for 3),
             iif(coalesce(adl_depart_no, 0) <> 0, cast(adl_depart_no as varchar(20)) || '|', '') || iif(coalesce(adl_section_no, 0) <> 0, cast(adl_section_no as varchar(20)) || '|', '') || iif(coalesce(adl_building_no, 0) <> 0, cast(adl_building_no as varchar(20)), ''), -- дом корпус строение
             iif(coalesce(adl_flat_no, 0) <> 0, cast(adl_flat_no as varchar(20)), '')

      from addrclient adcl
      inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
      where adcl.adc_from_c_id = :c_id and
            adcl.adc_address_type in (1, 2)
      into :adl_post_index, :adl_from_pl_id, :adl_street_id, :address, :adc_type, :owner, :country, :house, -- дом корпус строение
           :flat
  do
  begin

    select coalesce(pl.pl_post_index, 0), substring(pl_code from 4 for 2)
    from places pl
    where pl.pl_id = :adl_from_pl_id
    into :pl_post_index, :pl_fs_code;

    if (country = '810') then
    begin
      select rcr.rcr_cib_code
      from rcib_region rcr
      where rcr.rcr_fs_code = :pl_fs_code
      into :rregion; -- Код региона РФ по месту регистрации
    end

    select coalesce(pl.pl_name, ''), coalesce(pt.plt_name, ''), coalesce(pl.pl_post_index, 0)
    from places pl
    inner join placetype pt on pl.pl_status = pt.plt_id
    where pl.pl_id = :adl_street_id
    into :street_buf, :plt_name, :street_post_index;

    street = '';
    district = '';
    if (:plt_name = 'улица') then
      street = street_buf; -- Наименование улицы
    if (:plt_name in ('район', 'микрорайон')) then
      district = street_buf; -- Наименование района

    /* по умолчанию наибольший приоритет отдаю почтовому индексу, указанному на уровне улицы*/
    if (street_post_index <> '0') then
      iindex = cast(street_post_index as integer);
    else
    if (pl_post_index <> '0') then
      iindex = cast(pl_post_index as integer); -- Почтовый индекс места жительства/нахождения

    if (owner = 0) then
      owner = 3; -- аренда
    else
      owner = 9; -- не известно

    country = case country
                when '417' then 'KG'
                when '810' then 'RU'
                when '860' then 'UZ'
                when '972' then 'TJ'
                else '99'
              end;

    if (adc_type = 1) then
    begin
      reg_owner = owner;
      reg_country = country;
      reg_region = rregion;
      reg_district = district;
      reg_street = street;
      reg_house = house;
      reg_flat = flat;
      reg_index = iindex;
      addr_reg_total = address;

    end
    else
    if (adc_type = 2) then
    begin
      fact_owner = owner;
      fact_country = country;
      fact_region = rregion;
      fact_district = district;
      fact_street = street;
      fact_house = house;
      fact_flat = flat;
      fact_index = iindex;
      addr_fact_total = address;
    end

  end

  suspend;

end
