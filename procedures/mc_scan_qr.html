CREATE PROCEDURE MC_SCAN_QR(
    MCL_ID D_ID,
    UID D_STR_CODE)
RETURNS (
    ERROR_TXT D_STR_255,
    JSON_TEXT D_STR_255,
    SUCCESS_SCAN D_BOOL)
AS
declare variable object_kind d_str_10;
declare variable object_id d_id;
declare variable has_access_to_scan d_integer;
begin
  success_scan = 0;
  begin
    select udl_object_kind, udl_object_id
    from uuidlist
    where udl_uuid = char_to_uuid(:uid)
    into object_kind, object_id;
    when any do
    begin
      error_txt = 'Не товар: ' || uid;
      suspend;
      exit;
    end
  end

  has_access_to_scan = (select has_access_to_scan
                        from mc_credit_limit(:mcl_id));
  if (has_access_to_scan = 0) then
  begin
    error_txt = 'У вас нет доступа к скану QR. Кредитный лимит не расчитан или равен нулю.';
    suspend;
    exit;
  end

  if (coalesce(:object_id, -777) = -777) then
  begin
    error_txt = 'Ошибка';
    suspend;
    exit;
  end

  if (f_objkind(:object_kind, 'name') = 'mc_purchase') then
  begin
    if (exists(select *
               from mc_purchase mp
               where mp.mp_id = :object_id and
                     (mp.mp_is_finished = 1 or
                     mp.mp_from_mcl_id is not null))) then
    begin
      json_text = '';
      error_txt = 'Операция недоступна';
      suspend;
      exit;
    end

    begin
      update mc_purchase mp
      set mp.mp_from_mcl_id = :mcl_id
      where mp.mp_id = :object_id and
            mp.mp_status = 1 and
            mp.mp_from_ma_id is null; --QR может отсканиться только если он еще не выдан
      when any do
      begin
        error_txt = 'Операция не найдена';
        suspend;
        exit;
      end
    end

    success_scan = 1;
    json_text = '{ "mp_id":"' || :object_id || '"}';
    error_txt = '';
    suspend;
    exit;
  end
  else
  begin
    json_text = '';
    error_txt = 'Операция не найдена';
    suspend;
    exit;
  end
end
