CREATE PROCEDURE CB24_RFC1322_LOAN_REPAYMENTS(
    DISB_PERIOD_BEGIN D_DATE,
    DISB_PERIOD_END D_DATE,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    LOAN_ID D_ID,
    LOAN_UID D_STR_CODE,
    PAYMENT_FACT_SUM D_CURRENCY,
    PAYMENT_FACT_SUM_PRINCIPAL D_CURRENCY,
    PAYMENT_FACT_SUM_INTEREST D_CURRENCY,
    PAYMENT_FACT_SUM_PENALTIES D_CURRENCY,
    PAYMENT_FACT_SUM_STATEDUTIES D_CURRENCY,
    PAYMENT_FACT_DATE D_DATE,
    DUE_ARREAR D_CURRENCY,
    DUE_ARREAR_PRINCIPAL D_CURRENCY,
    DUE_ARREAR_INTEREST D_CURRENCY,
    PAST_DUE_ARREAR D_CURRENCY,
    PAST_DUE_ARREAR_PRINCIPAL D_CURRENCY,
    PAST_DUE_ARREAR_INTEREST D_CURRENCY,
    SUBJECT_TYPE D_STR_20,
    SUBJECT_ID D_ID,
    SUBJECT_PRIVATE_FULLNAME D_STR_100,
    SUBJECT_PRIVATE_BIRTH_DATE D_DATE,
    SUBJECT_PRIVATE_COUNTRY_CODE D_STR_20,
    SUBJECT_PRIVATE_DOC_SERIAL D_STR_CODE,
    SUBJECT_PRIVATE_DOC_NUMBER D_STR_CODE,
    SUBJECT_LEGAL_FULLNAME D_STR_255,
    SUBJECT_LEGAL_INN D_STR_20,
    SUBJECT_LEGAL_OGRN D_STR_20,
    RES_KONTRACT_NO D_STR_20,
    RES_DISB_DATE D_DATE,
    RES_DOC_ID D_ID,
    RES_PK_NAME D_STR_100)
AS
declare variable kc_id_prv d_id;
declare variable kc_disb_passport_id d_integer;
declare variable pk_id d_integer;
declare variable cds_doc_no d_str_code;
begin
  subject_type = 'ФЛ';
  subject_legal_fullname = '';
  subject_legal_inn = '';
  subject_legal_ogrn = '';
  kc_id_prv = 0;

  for select oks_from_kc_id, k_kontract_no, k_disb_date, oks_date_begin, kc_disb_passport_id,
             kc_from_c_id, c_name_fio, c_birth_date,
             -(oks_pmt_os_norm + oks_pmt_os_adv + oks_pmt_os_delinq),  -- payment_fact_sum_principal
             -(oks_pmt_perc_norm + oks_pmt_perc_adv + oks_pmt_perc_delinq),  -- payment_fact_sum_interest
             oks_pmt_law, -- payment_fact_sum_stateduties
             oks_balance, -- due_arrear_principal
             oks_bal_perc, -- due_arrear_interest
--             iif(oks_delinq_date is not null, oks_bal_zad_os, 0.00), -- past_due_arrear_principal
--             iif(oks_delinq_date is not null, oks_bal_zad_perc, 0.00) -- past_due_arrear_interest
             iif(oks_delinq_days > 0, oks_bal_zad_os, 0.00), -- past_due_arrear_principal
             iif(oks_delinq_days > 0, oks_bal_zad_perc, 0.00) -- past_due_arrear_interest
  from kn_oksummary, knclient, kontract, clients
  where oks_date_begin between :rep_date_begin and :rep_date_end and
        oks_from_k_id = k_id and
        oks_from_kc_id = kc_id and
        kc_from_c_id = c_id and
        oks_is_pmt = 1 and
        k_disb_date + 0 between :disb_period_begin and :disb_period_end
  order by oks_from_kc_id desc, oks_date_begin desc
  into loan_id, res_kontract_no, res_disb_date, payment_fact_date, kc_disb_passport_id,
       subject_id, subject_private_fullname, subject_private_birth_date,
       payment_fact_sum_principal,
       payment_fact_sum_interest,
       payment_fact_sum_stateduties,
       due_arrear_principal,
       due_arrear_interest,
       past_due_arrear_principal,
       past_due_arrear_interest
  do
  begin
    payment_fact_sum_penalties = 0;
    select -sum(ok_fine)
    from kn_operkred
    where ok_from_kc_id = :loan_id and
          ok_ol_date = :payment_fact_date and
          ok_type = 4 and
          ok_ol_is_active = 1
    into payment_fact_sum_penalties;

    if (payment_fact_sum_penalties is null) then
      payment_fact_sum_penalties = 0;

    if (payment_fact_sum_principal = 0 and
        payment_fact_sum_interest = 0 and
        payment_fact_sum_penalties = 0 and
        payment_fact_sum_stateduties = 0) then
      Continue;

    if (kc_id_prv <> loan_id) then
    begin
      kc_id_prv = loan_id;
      loan_uid = f_uuid_cbr(f_uuid('knclient', loan_id));

      res_doc_id = null;
      subject_private_doc_serial = '';
      subject_private_doc_number = '';
      subject_private_country_code = '';
      res_pk_name = '';

      if (kc_disb_passport_id is not null) then
      begin
        res_doc_id = kc_disb_passport_id;

        select cds_doc_no, pk_id, pk_name, left(pk_country, 2)
        from clientdocs, passkind
        where cds_id = :res_doc_id and
              cds_from_pk_id = pk_id
        into cds_doc_no, pk_id, res_pk_name, subject_private_country_code;

        select res_doc_seria, res_doc_number
        from hp_passport_serial_num(:cds_doc_no, :pk_id)
        into subject_private_doc_serial, subject_private_doc_number;
      end
    end

    payment_fact_sum = payment_fact_sum_principal + payment_fact_sum_interest + payment_fact_sum_penalties + payment_fact_sum_stateduties;

    if (due_arrear_interest < 0) then
      due_arrear_interest = 0;

    if (past_due_arrear_interest > due_arrear_interest) then
      past_due_arrear_interest = due_arrear_interest;

    due_arrear_principal = due_arrear_principal - past_due_arrear_principal;
    due_arrear_interest = due_arrear_interest - past_due_arrear_interest;
    due_arrear = due_arrear_principal + due_arrear_interest;

    past_due_arrear = past_due_arrear_principal + past_due_arrear_interest;

    suspend;
  end
end
