CREATE PROCEDURE HP_KONTDISBPLAN_LIST(
    REP_DATE_END D_DATE)
RETURNS (
    KDP_ID D_INTEGER,
    KDP_OFFICE_ID D_INTEGER,
    KDP_MGR_ID D_INTEGER,
    KDP_DATE_BEGIN D_DATE,
    KDP_SUMMA_PLAN D_CURRENCY,
    KDP_SUMMA_PROCESS D_CURRENCY,
    KDP_SUMMA_APPROVED D_CURRENCY,
    KDP_SUMMA_SALDO D_CURRENCY,
    KDP_COUNT_PROCESS D_INTEGER,
    KDP_COUNT_APPROVED D_INTEGER,
    LML_NAME D_STR_100,
    RES_SUMMA_DISB D_CURRENCY,
    RES_COUNT_DISB D_INTEGER,
    RES_SHORTFALL D_CURRENCY)
AS
declare variable k_amount_approved d_currency;
declare variable lo_id d_id;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  kdp_date_begin = rep_date_end;

  if (db_id = 81001) then
  begin
    for select lml_id,
               lml_from_reg_id,
               lml_name
        from loanmgrlist, region
        where lml_from_reg_id = reg_unicode and
              lml_name <> reg_code || ' ' || reg_code and
              lml_date_begin <= :rep_date_end and
              lml_date_end >= :rep_date_end
        order by lml_name,
            lml_id
        into kdp_mgr_id,
             kdp_office_id,
             lml_name
    do
    begin
      kdp_id = null;
      kdp_summa_plan = 0;
      kdp_summa_process = 0;
      kdp_summa_approved = 0;
      kdp_summa_saldo = 0;
      kdp_count_process = 0;
      kdp_count_approved = 0;

      res_count_disb = 0;
      res_summa_disb = 0;

      select kdp_id,
             kdp_summa_plan,
             kdp_summa_process,
             kdp_summa_approved,
             kdp_summa_saldo,
             kdp_count_process,
             kdp_count_approved
      from kontdisbplan
      where kdp_mgr_id = :kdp_mgr_id and
            kdp_date_begin = :kdp_date_begin
      into :kdp_id,
           :kdp_summa_plan,
           :kdp_summa_process,
           :kdp_summa_approved,
           :kdp_summa_saldo,
           :kdp_count_process,
           :kdp_count_approved;

      for select k_amount_approved,
                 ks_kc_id
          from kontstate, kontract
          where ks_state = 1 and
                ks_ol_is_delete = 0 and
                ks_date_begin = :rep_date_end and
                ks_region_id = :kdp_office_id and
                ks_from_k_id = k_id
          into k_amount_approved,
               lo_id
      do
      begin
        if (f_lo_mgr_id(null, rep_date_end, lo_id, kdp_office_id) <> kdp_mgr_id) then
          continue;

        res_summa_disb = res_summa_disb + k_amount_approved;
        res_count_disb = res_count_disb + 1;
      end

      suspend;
    end
  end
  else
  begin
    for select lml_id,
               lml_from_reg_id,
               lml_name
        from loanmgrlist, region
        where lml_from_reg_id = reg_unicode and
              lml_name <> reg_code || ' ' || reg_code and
              lml_date_begin <= :rep_date_end and
              lml_date_end >= :rep_date_end and
              :rep_date_end between reg_date_open and coalesce(reg_date_close, '31.12.9999') -- только действующие офисы (RFCAZ-207)
        order by lml_name,
            lml_id
        into kdp_mgr_id,
             kdp_office_id,
             lml_name
    do
    begin
      kdp_id = null;
      kdp_summa_plan = 0;
      kdp_summa_process = 0;
      kdp_summa_approved = 0;
      kdp_summa_saldo = 0;
      kdp_count_process = 0;
      kdp_count_approved = 0;

      res_shortfall = 0;
      res_count_disb = 0;
      res_summa_disb = 0;

      select kdp_id,
             kdp_summa_plan,
             kdp_summa_process,
             kdp_summa_approved,
             kdp_summa_saldo,
             kdp_count_process,
             kdp_count_approved
      from kontdisbplan
      where kdp_mgr_id = :kdp_mgr_id and
            kdp_date_begin = :kdp_date_begin
      into :kdp_id,
           :kdp_summa_plan,
           :kdp_summa_process,
           :kdp_summa_approved,
           :kdp_summa_saldo,
           :kdp_count_process,
           :kdp_count_approved;

      res_shortfall = kdp_summa_saldo - kdp_summa_process;

      for select k_amount_approved,
                 ks_kc_id
          from kontstate, kontract
          where ks_state = 1 and
                ks_ol_is_delete = 0 and
                ks_date_begin = :rep_date_end and
                ks_region_id = :kdp_office_id and
                ks_from_k_id = k_id
          into k_amount_approved,
               lo_id
      do
      begin
        if (f_lo_mgr_id(null, rep_date_end, lo_id, kdp_office_id) <> kdp_mgr_id) then
          continue;

        res_summa_disb = res_summa_disb + k_amount_approved;
        res_count_disb = res_count_disb + 1;
      end

      suspend;
    end
  end
end
