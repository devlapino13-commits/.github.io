CREATE PROCEDURE UXP$UNITITEM_TREE(
    UNIT_UUID D_UUID,
    PARENT_UUID D_UUID,
    VISIBLE_PARENT D_UUID,
    SHOW_GRID D_BOOL)
RETURNS (
    UNI_UUID_CODE D_STR_CODE,
    UNI_PARENT_UUID_CODE D_STR_CODE,
    UNI_UUID D_UUID,
    UNI_UNT_UUID D_UUID,
    UNI_PARENT_UUID D_UUID,
    UNI_CODE D_STR_CODE,
    UNI_NAME D_STR_100,
    UNI_JSON_PATH D_STR_LONG,
    UNI_POSITION D_INTEGER,
    UNI_INACTIVE D_BOOL,
    UNI_KIND D_SMALLINT,
    UNI_CAPTION D_STR_100,
    UNI_CONTROL D_STR_CODE,
    UNI_HINT D_STR_LONG,
    UNI_ADD_INFO D_STR_8100,
    UNI_CAPTION_ALIGN D_SMALLINT,
    UNI_HIDDEN D_SMALLINT,
    UNI_READONLY D_BOOL,
    UNI_GROUP_DIRECTION D_SMALLINT,
    UNI_H_ALIGN D_SMALLINT,
    UNI_V_ALIGN D_SMALLINT,
    UNI_LEFT D_INTEGER,
    UNI_TOP D_INTEGER,
    UNI_WIDTH D_INTEGER,
    UNI_HEIGHT D_INTEGER,
    UNI_SIZABLE D_BOOL,
    UNI_ORDER_INDEX D_SMALLINT,
    UNI_GROUP_INDEX D_SMALLINT,
    UNI_FIXED_KIND D_SMALLINT,
    UNI_SUMMARY_KIND D_SMALLINT,
    UNI_DATASOURCE D_STR_CODE,
    UNI_FIELDNAME D_STR_CODE)
AS
declare variable cls_visual d_smallint;
declare variable tmp_parent d_uuid;
declare variable uni_linked_label d_str_code;
declare variable tmp_visible d_uuid;
begin
  if (show_grid is null) then
    show_grid = 0;

  if (visible_parent is null) then
    visible_parent = parent_uuid;

  if (show_grid = 1) then
  begin
    for select uni_uuid,
               uni_unt_uuid,
               uni_parent_uuid,
               uni_code,
               uni_name,
               uni_json_path,
               uni_position,
               uni_inactive,
               uni_kind, 
               uni_caption, 
               uni_control, 
               uni_hint, 
               uni_add_info, 
               uni_caption_align, 
               uni_hidden, 
               uni_readonly, 
               uni_group_direction, 
               uni_h_align, 
               uni_v_align, 
               uni_left,
               uni_top,
               uni_width, 
               uni_height, 
               uni_sizable, 
               uni_order_index, 
               uni_group_index, 
               uni_fixed_kind, 
               uni_summary_kind,
               uni_datasource,
               uni_fieldname,
               cls_visual
        from ux$unititem, ux$class
        where uni_name = cls_name and
              uni_uuid = :parent_uuid
        order by uni_position
        into :uni_uuid,
             :uni_unt_uuid,
             :uni_parent_uuid,
             :uni_code,
             :uni_name,
             :uni_json_path,
             :uni_position,
             :uni_inactive,
             :uni_kind, 
             :uni_caption, 
             :uni_control, 
             :uni_hint, 
             :uni_add_info, 
             :uni_caption_align, 
             :uni_hidden, 
             :uni_readonly, 
             :uni_group_direction, 
             :uni_h_align, 
             :uni_v_align, 
             :uni_left,
             :uni_top,
             :uni_width, 
             :uni_height, 
             :uni_sizable, 
             :uni_order_index, 
             :uni_group_index, 
             :uni_fixed_kind, 
             :uni_summary_kind,
             :uni_datasource,
             :uni_fieldname,
             :cls_visual
    do
    begin
      unit_uuid = uni_unt_uuid;
      uni_uuid_code = uuid_to_char(uni_uuid);
      uni_parent_uuid_code = uuid_to_char(uni_parent_uuid);
  
      if (show_grid = 0 and uni_control in ('grid', 'tree')) then
        suspend;
    end
  end

  for select uni_uuid,
             uni_unt_uuid,
             uni_code,
             uni_name,
             uni_json_path,
             uni_position,
             uni_inactive,
             uni_kind, 
             uni_caption, 
             uni_control, 
             uni_hint, 
             uni_add_info, 
             uni_caption_align, 
             uni_hidden, 
             uni_readonly, 
             uni_group_direction, 
             uni_h_align, 
             uni_v_align, 
             uni_left,
             uni_top,
             uni_width, 
             uni_height, 
             uni_sizable, 
             uni_order_index, 
             uni_group_index, 
             uni_fixed_kind, 
             uni_summary_kind,
             uni_datasource,
             uni_fieldname,
             cls_visual,
             uni_linked_label
      from ux$unititem, ux$class
      where uni_unt_uuid = :unit_uuid and
            uni_name = cls_name and
            uni_parent_uuid is not distinct from :parent_uuid
      order by uni_position
      into :uni_uuid,
           :uni_unt_uuid,
           :uni_code,
           :uni_name,
           :uni_json_path,
           :uni_position,
           :uni_inactive,
           :uni_kind, 
           :uni_caption, 
           :uni_control, 
           :uni_hint, 
           :uni_add_info, 
           :uni_caption_align, 
           :uni_hidden, 
           :uni_readonly, 
           :uni_group_direction, 
           :uni_h_align, 
           :uni_v_align, 
           :uni_left,
           :uni_top,
           :uni_width, 
           :uni_height, 
           :uni_sizable, 
           :uni_order_index, 
           :uni_group_index, 
           :uni_fixed_kind, 
           :uni_summary_kind,
           :uni_datasource,
           :uni_fieldname,
           :cls_visual,
           :uni_linked_label
  do
  begin
    uni_parent_uuid = visible_parent;
    uni_uuid_code = uuid_to_char(uni_uuid);
    uni_parent_uuid_code = uuid_to_char(uni_parent_uuid);

    if (coalesce(uni_linked_label, '') <> '') then
    begin
      select uni_caption
      from ux$unititem
      where uni_parent_uuid = :uni_parent_uuid and
            uni_code = :uni_linked_label
      into uni_caption;
    end

    if (uni_control = 'label') then
    begin
      if (exists(select * from ux$unititem
      where uni_parent_uuid = :uni_parent_uuid and
            uni_linked_label = :uni_code)) then
        Continue;
    end

    tmp_visible = visible_parent;
    if (cls_visual = 1 and uni_inactive = 0) then
    begin
      suspend;

      tmp_visible = uni_uuid;
    end

    if (show_grid = 0 and uni_control in ('grid', 'tree')) then
      Continue;

    if (uni_inactive = 1) then
      Continue;

    tmp_parent = uni_uuid;
    for select uni_uuid_code, uni_uuid, uni_unt_uuid, uni_parent_uuid, uni_parent_uuid_code, uni_code, uni_name,
               uni_json_path, uni_position, uni_inactive, uni_kind, uni_caption, uni_control, uni_hint, uni_add_info,
               uni_caption_align, uni_hidden, uni_readonly, uni_group_direction, uni_h_align, uni_v_align, uni_left,
               uni_top, uni_width, uni_height, uni_sizable, uni_order_index, uni_group_index, uni_fixed_kind,
               uni_summary_kind, uni_datasource, uni_fieldname
        from uxp$unititem_tree(:unit_uuid, :tmp_parent, :tmp_visible, :show_grid)
        into :uni_uuid_code, :uni_uuid, :uni_unt_uuid, :uni_parent_uuid, :uni_parent_uuid_code, :uni_code, :uni_name,
             :uni_json_path, :uni_position, :uni_inactive, :uni_kind, :uni_caption, :uni_control, :uni_hint, :uni_add_info,
             :uni_caption_align, :uni_hidden, :uni_readonly, :uni_group_direction, :uni_h_align, :uni_v_align, :uni_left,
             :uni_top, :uni_width, :uni_height, :uni_sizable, :uni_order_index, :uni_group_index, :uni_fixed_kind,
             :uni_summary_kind, :uni_datasource, :uni_fieldname
    do
      suspend;
  end
end
