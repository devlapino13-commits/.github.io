CREATE PROCEDURE GET_METADATA_FK
RETURNS (
    FK_TABLE VARCHAR(50),
    FK_FIELD VARCHAR(255),
    FK_CONSTAINT VARCHAR(255),
    MAIN_TABLE VARCHAR(50),
    MAIN_FIELD VARCHAR(255))
AS
begin
  for with tab
      as (select trim(a.rdb$relation_name) as main_table, trim(e.rdb$field_name) as main_field,
                 trim(a.rdb$constraint_name) as fk_constaint, trim(c.rdb$relation_name) as fk_table,
                 trim(d.rdb$field_name) as fk_field
          from rdb$ref_constraints b, rdb$relation_constraints a, rdb$relation_constraints c, rdb$index_segments d, rdb$index_segments e, rdb$indices i
          where (a.rdb$constraint_type = 'FOREIGN KEY') and
                (a.rdb$constraint_name = b.rdb$constraint_name) and
                (b.rdb$const_name_uq = c.rdb$constraint_name) and
                (c.rdb$index_name = d.rdb$index_name) and
                (a.rdb$index_name = e.rdb$index_name) and
                (a.rdb$index_name = i.rdb$index_name) and
                (d.rdb$field_position = e.rdb$field_position) and
                (c.rdb$constraint_type = 'PRIMARY KEY')
          order by c.rdb$relation_name, a.rdb$constraint_name, d.rdb$field_position)
      select lower(fk_table), list(lower(fk_field)), lower(fk_constaint), lower(main_table), list(lower(main_field))
      from tab
      group by fk_table, fk_constaint, main_table
      into :fk_table, :fk_field, :fk_constaint, :main_table, :main_field
  do
  begin
    suspend;
  end
end
