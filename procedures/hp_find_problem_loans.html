CREATE PROCEDURE HP_FIND_PROBLEM_LOANS(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID D_ID,
    OL_ID D_ID,
    OL_DATE DATE,
    KS_STATE D_SMALLINT)
AS
declare variable k_id_old d_id;
declare variable problem smallint;
declare variable ok_sh_date_old date;
declare variable ok_sh_id integer;
declare variable ok_sh_date date;
begin
  k_id_old = null;
  problem = 0;

  for select ok.ok_from_k_id,
             ok.ok_from_ol_id,
             ok.ok_ol_date
  from operkred ok
  where ok.ok_ol_is_active = 1
  and ok.ok_ol_date between :rep_date_begin and :rep_date_end
  and ok.ok_type in (4,9)
  --and ok.ok_from_k_id in (109010087926100201)
  group by ok.ok_from_k_id,
           ok.ok_from_ol_id,
           ok.ok_ol_date
  order by ok.ok_from_k_id,
           ok.ok_from_ol_id,
           ok.ok_ol_date
  into :k_id,
       :ol_id,
       :ol_date
  do
  begin
    if (k_id_old is null) then
        k_id_old = k_id;

    ok_sh_date_old = null;

    if (k_id_old <> k_id) then
        problem = 0;

    if (problem = 0) then
    begin
        for select ok.ok_sh_id,
                  ok.ok_sh_date
        from operkred ok
        where ok.ok_from_ol_id = :ol_id
        and ok.ok_from_k_id = :k_id
        and ok.ok_type in (4,9)
        order by  ok.ok_sh_id,
                  ok.ok_sh_date
        into :ok_sh_id, :ok_sh_date
        do
        begin
            if (ok_sh_date_old is null) then
                ok_sh_date_old = ok_sh_date;
    
            if (ok_sh_date < ok_sh_date_old) then
                problem = 1;
        end
    end

    if (problem = 1) then
    begin
        select ks.ks_state
        from kontstate ks
        where ks.ks_ol_is_delete = 0
        and current_date between ks.ks_date_begin and ks.ks_date_end
        and ks.ks_from_k_id = :k_id
        into :ks_state;

        suspend;
    end

    k_id_old = k_id;
  end
end
