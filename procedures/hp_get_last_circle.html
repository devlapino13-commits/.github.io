CREATE PROCEDURE HP_GET_LAST_CIRCLE(
    REP_K_ID ID_TYPE,
    REP_MA_ID INTEGER,
    REP_C_ID_LIST VARCHAR(255),
    REP_DISB_DATE DATE)
RETURNS (
    RE_K_ID ID_TYPE,
    RE_KOL_CLI INTEGER,
    RE_PERCENT_PER_DAY PERC_TYPE,
    RE_KOMMIS_PERC PERC_TYPE)
AS
declare variable k_from_kf_id id_type;
declare variable k_disb_date date;
declare variable kol_cli integer;
declare variable pr_product_kind integer;
declare variable k integer;
declare variable sql_text varchar(4000);
declare variable re_kf_id id_type;
begin
-- определить предыдущий цикл контракта
  if (rep_k_id is null) then
    rep_k_id = 0;
  if (rep_ma_id is null) then
    rep_ma_id = 0;
  if (rep_c_id_list is null) then
    rep_c_id_list = '';
  if (rep_disb_date is null) then
    rep_disb_date = current_date;

  if (rep_k_id <> 0) then
  begin
    k_from_kf_id = null;
    k_disb_date = null;
    kol_cli = null;

    select k_from_kf_id, k_disb_date, k_kolich_client_approved, pr_product_kind
    from kontract, product
    where k_id = :rep_k_id and
          k_from_pr_id = pr_id
    into k_from_kf_id, k_disb_date, kol_cli, pr_product_kind;

    re_k_id = null;
    re_kol_cli = null;
    re_kf_id = null;

    if (kol_cli = 1 or pr_product_kind = 2) then
    begin
      select k_id, k_kolich_client_approved, k_percent_per_day, k_kommis_perc
      from kontstate, kontract, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin < :k_disb_date and
            ks_from_k_id = k_id and
            k_from_kf_id = :k_from_kf_id and
            k_from_pr_id = pr_id and
            pr_product_kind <> 2
      order by ks_date_begin desc
      rows 1
      into re_k_id, re_kol_cli, re_percent_per_day, re_kommis_perc;
    end
    else
    if (kol_cli >= 2) then
    begin
      with tabl1
      as (select kc_from_c_id,
                 (select k_from_kf_id as re_kf_id
                  from knclient kc2, kontract, product, kontstate
                  where kc2.kc_from_c_id = kc1.kc_from_c_id and
                        kc2.kc_amount_approved > 0 and
                        kc2.kc_from_k_id = ks_from_k_id and
                        ks_ol_is_delete = 0 and
                        ks_date_begin < :k_disb_date and
                        ks_state = 1 and
                        ks_from_k_id = k_id and
                        k_from_pr_id = pr_id and
                        pr_product_kind <> 2
                  order by ks_date_begin desc
                  rows 1)
          from knclient kc1
          where kc1.kc_from_k_id = :rep_k_id and
                kc1.kc_amount_approved > 0)
      select re_kf_id,
             (select k_id
              from kontract, product, kontstate
              where k_from_kf_id = re_kf_id and
                    k_id = ks_from_k_id and
                    ks_ol_is_delete = 0 and
                    ks_state = 1 and
                    k_from_pr_id = pr_id and
                    pr_product_kind <> 2 and
                    ks_date_begin < :k_disb_date
              order by ks_date_begin desc
              rows 1) as re_k_id,
             (select k_percent_per_day
              from kontract, product, kontstate
              where k_from_kf_id = re_kf_id and
                    k_id = ks_from_k_id and
                    ks_ol_is_delete = 0 and
                    ks_state = 1 and
                    k_from_pr_id = pr_id and
                    pr_product_kind <> 2 and
                    ks_date_begin < :k_disb_date
              order by ks_date_begin desc
              rows 1) as k_percent_per_day,
             (select k_kommis_perc
              from kontract, product, kontstate
              where k_from_kf_id = re_kf_id and
                    k_id = ks_from_k_id and
                    ks_ol_is_delete = 0 and
                    ks_state = 1 and
                    k_from_pr_id = pr_id and
                    pr_product_kind <> 2 and
                    ks_date_begin < :k_disb_date
              order by ks_date_begin desc
              rows 1) as k_kommis_perc, count(*) as re_kol_cli
      from tabl1
      where re_kf_id is not null
      group by re_kf_id, re_k_id, k_percent_per_day, k_kommis_perc
      order by re_kol_cli desc, k_percent_per_day
      rows 1
      into re_kf_id, re_k_id, re_percent_per_day, re_kommis_perc, re_kol_cli;

      if (re_k_id is not null and
          (cast(kol_cli as double precision) / 2 > cast(re_kol_cli as double precision) or re_kol_cli = 1)) then
      begin
        re_k_id = null;
      end
    end
  end
  else
  if (rep_ma_id <> 0) then
  begin

    k_from_kf_id = null;
    k_disb_date = null;
    kol_cli = null;
    re_kf_id = null;

    select ma_from_kf_id, ma_date_oper, ma_kolich_client, pr_product_kind
    from miniapp, product
    where ma_id = :rep_ma_id and
          ma_date_end = cast('31.12.9999' as date) and
          ma_from_pr_id = pr_id rows 1
    into k_from_kf_id, k_disb_date, kol_cli, pr_product_kind;

    re_k_id = null;
    re_kol_cli = null;
    if (kol_cli = 1 or pr_product_kind = 2) then
    begin
      select k_id, k_kolich_client_approved, k_percent_per_day, k_kommis_perc
      from kontstate, kontract, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin < :k_disb_date and
            ks_from_k_id = k_id and
            k_from_kf_id = :k_from_kf_id and
            k_from_pr_id = pr_id and
            pr_product_kind <> 2
      order by ks_date_begin desc
      rows 1
      into re_k_id, re_kol_cli, re_percent_per_day, re_kommis_perc;
    end
    else
    if (kol_cli >= 2) then
    begin
      with tabl1
      as (select mac_from_c_id,
                 (select k_from_kf_id as re_kf_id
                  from knclient kc2, kontract, product, kontstate
                  where kc2.kc_from_c_id = mac_from_c_id and
                        kc2.kc_amount_approved > 0 and
                        kc2.kc_from_k_id = ks_from_k_id and
                        ks_ol_is_delete = 0 and
                        ks_date_begin < :k_disb_date and
                        ks_state = 1 and
                        ks_from_k_id = k_id and
                        k_from_pr_id = pr_id and
                        pr_product_kind <> 2
                  order by ks_date_begin desc
                  rows 1)
          from miniappclient
          where mac_from_ma_id = :rep_ma_id and
                mac_amount > 0)
      select re_kf_id,
             (select k_id
              from kontract, product, kontstate
              where k_from_kf_id = re_kf_id and
                    k_id = ks_from_k_id and
                    ks_ol_is_delete = 0 and
                    ks_state = 1 and
                    k_from_pr_id = pr_id and
                    pr_product_kind <> 2 and
                    ks_date_begin < :k_disb_date
              order by ks_date_begin desc
              rows 1) as re_k_id,
             (select k_percent_per_day
              from kontract, product, kontstate
              where k_from_kf_id = re_kf_id and
                    k_id = ks_from_k_id and
                    ks_ol_is_delete = 0 and
                    ks_state = 1 and
                    k_from_pr_id = pr_id and
                    pr_product_kind <> 2 and
                    ks_date_begin < :k_disb_date
              order by ks_date_begin desc
              rows 1) as k_percent_per_day,
             (select k_kommis_perc
              from kontract, product, kontstate
              where k_from_kf_id = re_kf_id and
                    k_id = ks_from_k_id and
                    ks_ol_is_delete = 0 and
                    ks_state = 1 and
                    k_from_pr_id = pr_id and
                    pr_product_kind <> 2 and
                    ks_date_begin < :k_disb_date
              order by ks_date_begin desc
              rows 1) as k_kommis_perc, count(*) as re_kol_cli
      from tabl1
      where re_kf_id is not null
      group by re_kf_id, re_k_id, k_percent_per_day, k_kommis_perc
      order by re_kol_cli desc, k_percent_per_day
      rows 1
      into re_kf_id, re_k_id, re_percent_per_day, re_kommis_perc, re_kol_cli;

      if (re_k_id is not null and
          (cast(kol_cli as double precision) / 2 > cast(re_kol_cli as double precision) or re_kol_cli = 1)) then
      begin
        re_k_id = null;
      end
    end

  end
  else
  if (rep_c_id_list <> '') then
  begin
    re_kf_id = null;
    re_k_id = null;
    re_kol_cli = null;

    rep_c_id_list = replace(rep_c_id_list, ';', ',');

--  определить кол-во клиентов по запятым
    kol_cli = char_length(:rep_c_id_list) - char_length(replace(rep_c_id_list, ',', '')) + 1;

    if (kol_cli = 1) then
    begin
      select k_id, k_kolich_client_approved, k_percent_per_day, k_kommis_perc, k_from_kf_id
      from kontstate, kontract, product, knclient
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin < :rep_disb_date and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_product_kind <> 2 and
            pr_loan_type = 1 and
            k_id = kc_from_k_id and
            kc_from_c_id = cast(:rep_c_id_list as id_type) and
            kc_amount_approved > 0
      order by ks_date_begin desc
      rows 1
      into re_k_id, re_kol_cli, re_percent_per_day, re_kommis_perc, re_kf_id;
    end
    else
    if (kol_cli >= 2) then
    begin
      sql_text = 'with tabl1 as (
          select c_id,
                 (select k_from_kf_id as re_kf_id
                  from knclient kc2, kontract, product, kontstate
                  where kc2.kc_from_c_id = c_id
                  and kc2.kc_amount_approved > 0
                  and kc2.kc_from_k_id = ks_from_k_id
                  and ks_ol_is_delete = 0
                  and ks_date_begin < :k_disb_date
                  and ks_state = 1
                  and ks_from_k_id = k_id
                  and k_from_pr_id = pr_id
                  and pr_product_kind <> 2
                  and pr_loan_type = 0
                  order by ks_date_begin desc
                  rows 1)
          from clients
          where c_id in (' || :rep_c_id_list || '))
        select  re_kf_id,
               (select k_id
               from kontract, product, kontstate
               where k_from_kf_id = re_kf_id
               and k_id = ks_from_k_id
               and ks_ol_is_delete = 0
               and ks_state = 1
               and k_from_pr_id = pr_id
               and pr_product_kind <> 2
               and ks_date_begin < :k_disb_date
               order by ks_date_begin desc
               rows 1) as re_k_id,
               (select k_percent_per_day
               from kontract, product, kontstate
               where k_from_kf_id = re_kf_id
               and k_id = ks_from_k_id
               and ks_ol_is_delete = 0
               and ks_state = 1
               and k_from_pr_id = pr_id
               and pr_product_kind <> 2
               and ks_date_begin < :k_disb_date
               order by ks_date_begin desc
               rows 1) as k_percent_per_day,
               (select k_kommis_perc
               from kontract, product, kontstate
               where k_from_kf_id = re_kf_id
               and k_id = ks_from_k_id
               and ks_ol_is_delete = 0
               and ks_state = 1
               and k_from_pr_id = pr_id
               and pr_product_kind <> 2
               and ks_date_begin < :k_disb_date
               order by ks_date_begin desc
               rows 1) as k_kommis_perc,
               count(*) as re_kol_cli
        from tabl1
        where re_kf_id is not null
        group by re_kf_id, re_k_id, k_percent_per_day, k_kommis_perc
        order by re_kol_cli desc, k_percent_per_day
        rows 1';

      execute statement(sql_text)(k_disb_date := rep_disb_date)
          into re_kf_id, re_k_id, re_percent_per_day, re_kommis_perc, re_kol_cli;

      if (re_k_id is not null and
          (cast(kol_cli as double precision) / 2 > cast(re_kol_cli as double precision) or re_kol_cli = 1)) then
      begin
        re_k_id = null;
      end
    end
  end
  if (re_k_id is not null) then
    suspend;
end
