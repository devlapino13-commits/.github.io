CREATE PROCEDURE HP_GET_LINKED_KNCLIENT_LIST(
    KC_ID ID_TYPE,
    SELF_INCLUDE INTEGER = 1)
RETURNS (
    RES_KC_ID ID_TYPE)
AS
declare variable k_id id_type;
declare variable tmp_k_id id_type;
begin
  select kc_from_k_id
  from knclient
  where kc_id = :kc_id
  into :k_id;

  for select res_k_id
      from hp_get_linked_kontract_list(:k_id, 1)
      into tmp_k_id
  do
  begin
    for select kc_id
        from knclient
        where kc_from_k_id = :tmp_k_id and
              kc_amount_approved > 0
        into :res_kc_id
    do
    begin
      if (self_include = 1 or res_kc_id <> kc_id) then
        suspend;
    end
  end
end
