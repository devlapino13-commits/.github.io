CREATE PROCEDURE CB_RU_SB_2015_03(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    PORT_A CURR_TYPE,
    PORT_B CURR_TYPE,
    PORT_C CURR_TYPE,
    KOL_CLI INTEGER,
    PAR7 CURR_TYPE,
    BONUS CURR_TYPE,
    OFFICE_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    KOL_SB INTEGER,
    FIO_SB VARCHAR(2048),
    REG_ID INTEGER)
AS
declare variable rep_date_begin date;
declare variable k_from_kf_id id_type;
declare variable k_disb_date date;
declare variable reg_id_begin integer;
declare variable kal_date date;
declare variable rep_date_end_2 date;
declare variable oks_balance curr_type;
declare variable delinq_balance decimal(18,4);
declare variable category_id smallint;
declare variable k_id decimal(18,0);
declare variable just_cl integer;
declare variable oks_delinq_date date;
declare variable oks_kc_id id_type;
begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;
  rep_date_end_2 = rep_date_begin - 1;
  port_a = 0;
  port_b = 0;
  port_c = 0;

  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into reg_id
  do
  begin
    port_a = 0;
    port_b = 0;
    port_c = 0;

    delinq_balance = 0;
    oks_balance = 0;
    kol_cli = 0;
    for select oks_from_k_id, oks_balance, oks_delinq_date, oks_kc_id
        from oksummary
        where oks_region_id = :reg_id and
              :rep_date_end between oks_date_begin and oks_date_end and
              oks_balance > 0 and
              oks_close_state = 0
--      and :rep_date_end - oks_delinq_date between 8 and 180
        into k_id, delinq_balance, oks_delinq_date, oks_kc_id
    do
    begin
--      Изменено по техническому дополнению к бонусной политике от 01.08.2016 Улан
      if (:rep_date_end < '31.07.2016') then
      begin
        category_id = null;
        select aa_category_id
        from aarep_get_category_id(:k_id, :rep_date_end, null, :oks_kc_id, null, null, null, null)
        into category_id;
      end
      else
      begin
        if (:rep_date_end - :oks_delinq_date <= 360) then
        begin
          category_id = null;
          select aa_category_id
          from aarep_get_category_id(:k_id, :rep_date_end, null, :oks_kc_id, null, null, null, null)
          into category_id;
        end
        else
        begin
          category_id = 3;
        end
      end

      if (category_id < 3 and
          category_id is not null) then
      begin
--    Если контракт выдан до начало отчетного периода, то офис проверяет на конец прошлого месяца
        select k_from_kf_id, k_disb_date
        from kontract
        where k_id = :k_id
        into k_from_kf_id, k_disb_date;

--          if (:rep_date_end - :oks_delinq_date between 8 and 180) then
        if (:rep_date_end - :oks_delinq_date >= 8) then
        begin
          port_a = port_a + delinq_balance;

          if (k_disb_date < rep_date_begin) then
          begin
            select kal_new_reg_id
            from knaktlist
            where knaktlist.is_delete = 0 and
                  :rep_date_begin - 1 between kal_date and kal_date_end and
                  kal_from_kf_id = :k_from_kf_id
            into reg_id_begin;
            if (reg_id_begin <> reg_id) then
            begin
              port_b = port_b + delinq_balance;
            end
          end
          else
          begin
--       Если выдан после начало отчетного периода, то проверяет следующее
            select kal_date
            from knaktlist
            where knaktlist.is_delete = 0 and
                  :rep_date_end between kal_date and kal_date_end and
                  kal_from_kf_id = :k_from_kf_id
            into kal_date;
--       если после выдачи была передача, а выдача было произведено во время отчетного периода
--       то проверяет до даты передачи офис контракта
            if (kal_date > k_disb_date) then
            begin
              select kal_new_reg_id
              from knaktlist
              where knaktlist.is_delete = 0 and
                    :kal_date - 1 between kal_date and kal_date_end and
                    kal_from_kf_id = :k_from_kf_id
              into reg_id_begin;
              if (reg_id_begin <> reg_id) then
              begin
                port_b = port_b + delinq_balance;
              end
            end
          end
        end

        just_cl = null;
        select res_just_client_count
        from hp_get_just_client_in_loan(:rep_date_end, :k_id, :k_disb_date)
        into just_cl;
        if (just_cl is null) then
          just_cl = 0;
        kol_cli = kol_cli + just_cl;
      end
    end
--    Теперь по этому региону портфель на начало нужно вычислить т.е. портфель C
    for select oks_from_k_id, oks_balance, oks_kc_id, oks_delinq_date
        from oksummary
        where oks_region_id = :reg_id and
              :rep_date_end_2 between oks_date_begin and oks_date_end and
              oks_balance > 0 and
              oks_close_state = 0
        into k_id, oks_balance, oks_kc_id, oks_delinq_date
    do
    begin
--      Изменено по техническому дополнению к бонусной политике от 01.08.2016 Улан
      if (:rep_date_end_2 < '31.07.2016') then
      begin
        category_id = null;
        select aa_category_id
        from aarep_get_category_id(:k_id, :rep_date_end_2, null, :oks_kc_id, null, null, null, null)
        into category_id;
      end
      else
      begin
        if (:rep_date_end_2 - :oks_delinq_date <= 360) then
        begin
          category_id = null;
          select aa_category_id
          from aarep_get_category_id(:k_id, :rep_date_end_2, null, :oks_kc_id, null, null, null, null)
          into category_id;
        end
        else
        begin
          category_id = 3;
        end
      end

      if (category_id < 3) then
      begin
        port_c = port_c + oks_balance;
      end
    end

    if (port_a + port_b + port_c <> 0) then
    begin
      if (port_c <> 0) then
      begin
        par7 = (port_a - port_b) / port_c * 100;
      end
      else
      begin
        if (port_a - port_b > 0) then
        begin
          par7 = 100;
        end
        else
        begin
          par7 = 0;
        end
      end
--  пункт 2.2 если кол-во клиентов менее 500 то небудет начисляться
      bonus = 0;
      if (kol_cli >= 1000) then
      begin
        bonus = case
                  when par7 < 3.0000 then 35000
                  when par7 < 5.0000 then 25000
                  when par7 <= 7.0000 then 15000
                  else 0
                end;
      end
      else
      if (kol_cli >= 500) then
      begin
        bonus = case
                  when par7 < 3.0000 then 17500
                  when par7 < 5.0000 then 12500
                  when par7 <= 7.0000 then 7500
                  else 0
                end;
      end

      select r1.reg_code || ' ' || r1.reg_name office_name, r2.reg_code || ' ' || r2.reg_name branch_name
      from region r1, region r2
      where r1.reg_unicode = :reg_id and
            r1.parent = r2.reg_unicode
      into office_name, branch_name;

--    Вычисление кол-во СБ
      kol_sb = 0;
      fio_sb = '';
      select count(*), list(c_name_fio, '; ')
      from hr_pstatus, joblist, jobkind, clients
      where ps_status = 1 and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.Specialist_SB' and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = :reg_id and
            ps_from_c_id = c_id
      into kol_sb, fio_sb;
      if (kol_sb is null) then
        kol_sb = 0;

      suspend;
    end
  end
end
