CREATE PROCEDURE AAREP_MINIAPP_CLIENTS(
    MA_FROM_REG_ID INTEGER,
    MA_ID INTEGER)
RETURNS (
    KF_CLIENT_ID DECIMAL(18,0),
    KF_CLIENT_NAME VARCHAR(100),
    KF_CLIENT_PASS_NO VARCHAR(30),
    KF_CLIENT_BIRTH_DATE DATE,
    KF_AMOUNT DECIMAL(18,4),
    MA_KIB_SEND INTEGER,
    MA_KIB_RECIEVE INTEGER,
    MA_KIB_REC_TIME TIMESTAMP,
    KBM_ID INTEGER,
    KH_ALL_BALANCE DECIMAL(18,4),
    KH_ALL_KREDIT INTEGER,
    KBM_ERR_TEXT VARCHAR(255),
    KBM_IS_ERROR INTEGER,
    KF_IS_PORUCH INTEGER,
    KF_CLIENT_HAS_PHOTO INTEGER,
    KF_CLIENT_HAS_PASS INTEGER,
    C_CIRCLE INTEGER,
    KF_TYPE STR_100,
    SCORE D_INTEGER,
    MA_KIB_SAFE_AND_SOUND_SEND D_INTEGER,
    MA_KIB_SAFE_AND_SOUND_RECIEVE D_INTEGER,
    MA_KIB_SAFE_AND_SOUND_REC_TIME D_DATETIME)
AS
declare variable kh_aprox_balance decimal(18,4);
declare variable kh_dateaward date;
declare variable kh_amount decimal(18,4);
declare variable kh_period_in_month decimal(15,2);
declare variable kh_clients_kol integer;
declare variable month_left decimal(15,2);
declare variable ma_date_oper date;
declare variable st_staff integer;
declare variable st_client integer;
declare variable c_blob_photo_id decimal(18,0);
declare variable c_blob_pass1_id decimal(18,0);
declare variable cu_symbol_old varchar(3);
declare variable kbm_checked_date date;
declare variable cu_symbol varchar(3);
declare variable kh_unpaidamount curr_type;
declare variable cu_id varchar(3);
declare variable cr_rate curr_type;
declare variable k_id id_type;
declare variable k_disb_date d_date;
declare variable k_status d_integer;
declare variable is_ipf boolean;
declare variable pkb_eil_status d_integer;
declare variable pkb_eil_answer_datetime d_datetime;
declare variable safe_and_sound_eil_status d_smallint;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select ma_date_oper
  from miniapp
  where ma_id = :ma_id and
        ma_date_end = '31.12.9999'
  into :ma_date_oper;

  k_id = null;
  select
    k_id,
    k_disb_date,
    k_status,
    pr_code containing '.ipf.'
  from kontract
    inner join product
      on pr_id = k_from_pr_id
  where k_from_ma_app_id = :ma_id and
        k_status >= 0
  order by k_disb_date
  rows 1
  into
    :k_id,
    :k_disb_date,
    :k_status,
    :is_ipf;

  for
    select
      mac_from_c_id,
      c_name_fio,
      c_pass_no,
      c_birth_date,
      mac_amount,
      mac_is_poruch,
      c_blob_photo_id,
      c_blob_pass1_id,
      case mac_is_poruch
        when -2 then 'Потребитель услуги'
        when -1 then 'Лицо, дающее согласие'
        when  1 then 'Поручитель'
      else iif(coalesce(mac_is_cashier, 0) = 1, 'Казначей', 'Клиент') end
    from miniappclient, clients
    where mac_from_ma_id = :ma_id
      and c_id = mac_from_c_id
    into
      :kf_client_id,
      :kf_client_name,
      :kf_client_pass_no,
      :kf_client_birth_date,
      :kf_amount,
      :kf_is_poruch,
      :c_blob_photo_id,
      :c_blob_pass1_id,
      :kf_type
  do
  begin
--  вычисляем цикл клиента
--  если по заявке сформирован договор то из договора будем брать цикл клиента
    c_circle = null;
    if (k_id is not null) then
    begin
      select kc_circle /* + 1 */
      from knclient -- закомментировал + 1 igusev т.е. логика с +1 видится не корректной
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = :kf_client_id
      into c_circle;
    end

    if (c_circle is null) then
    begin
      select count(*)
      from kontstate, knclient
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin < :ma_date_oper and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = :kf_client_id
      into c_circle;
      c_circle = coalesce(c_circle, 0) + 1;
    end

    if (c_blob_photo_id > 0) then
      kf_client_has_photo = 1;
    else
      kf_client_has_photo = 0;

    if (c_blob_pass1_id > 0) then
      kf_client_has_pass = 1;
    else
      kf_client_has_pass = 0;

    ma_kib_send = 0;
    ma_kib_recieve = 0;
    kbm_err_text = null;
    kbm_id = null;

    select kbm_id, kbm_err_text
    from hp_kibminiapp_last(:kf_client_id, :ma_id)
    into :kbm_id, :kbm_err_text;
    if (kbm_id is not null) then
      ma_kib_send = 1;

    kbm_id = null;
    ma_kib_rec_time = null;
    select kbm_id, kh_create_date
    from hp_kibminiapp_last(:kf_client_id, :ma_id)
    inner join kibhistory on kh_from_kbm_id = kbm_id
    order by kh_create_date desc
    rows 1
    into :kbm_id, :ma_kib_rec_time;

    kh_all_balance = 0;
    kh_all_kredit = 0;
    if ((kbm_err_text <> '') and
        (kbm_err_text is not null)) then
      kbm_is_error = 1;
    else
      kbm_is_error = 0;
    if (ma_kib_rec_time is not null) then
    begin
      ma_kib_recieve = 1;
      cu_symbol_old = '';

      for select kh_dateaward, kh_amount, kh_period_in_month, kh_clients_kol, left(
                 case kh_currency
                   when 'Сом (Кыргызский)' then 'KGS'
                   when 'Доллар США' then 'USD'
                   when 'Евро' then 'EUR'
                   when 'Российский рубл' then 'RUB'
                   else 'KGS'
                 end, 3), cast(kbm_checked_datetime as date), kh_unpaidamount
          from hp_kibminiapp_last(:kf_client_id, :ma_id)
          inner join kibhistory on kh_from_kbm_id = kbm_id
          where kh_creditstatus in ('Выдан', 'Просрочен', 'Просрочен  - <не выбрано>', 'Просрочен  -> Просрочки', 'Просрочен -> Просрочки', 'Реструктуризирован', 'Своевременный платеж', 'Начисление процентов', 'Granted_And_Activated', 'Restructured', 'Open_Under_Legal_Proceedings') and
                kh_is_our = 0
          into :kh_dateaward, :kh_amount, :kh_period_in_month, :kh_clients_kol, :cu_symbol, :kbm_checked_date,
               :kh_unpaidamount
      do
      begin
        if (kh_unpaidamount is null) then
          kh_unpaidamount = 0;

        if (cu_symbol <> cu_symbol_old) then
        begin
          select cl.cu_id
          from currlist cl
          where cl.cu_symbol = :cu_symbol
          into :cu_id;

          select opc.cr_rate
          from opercurr opc
          where opc.cr_from_cu_id = :cu_id and
                :kbm_checked_date between opc.cr_date and opc.cr_date_end
          into :cr_rate;

          cr_rate = coalesce(cr_rate, 1);
          cu_symbol_old = cu_symbol;
        end

        kh_aprox_balance = 0;
        kh_all_kredit = kh_all_kredit + 1;
        month_left = datediff(month, :kh_dateaward, :ma_date_oper);
        if (month_left < cast(month_left as integer)) then
          month_left = cast(month_left as integer) - 1;
        else
          month_left = cast(month_left as integer);
        if (month_left < kh_period_in_month) then
        begin
          if ((kh_clients_kol <> 0) and
              (kh_period_in_month <> 0)) then
          begin
            kh_aprox_balance = kh_amount - (((kh_amount / kh_clients_kol) / kh_period_in_month) * month_left);
            kh_aprox_balance = kh_aprox_balance * cr_rate;
          end
        end
        kh_all_balance = kh_all_balance + kh_unpaidamount /*KH_APROX_BALANCE*/ * cr_rate;
      end
    end
    /*
    else
    begin
      select KBM_ID, KH_CREATE_DATE from KIBMINIAPP, KIBHISTORY
      where KBM_FROM_C_ID=:KF_CLIENT_ID
      and KBM_ID=KH_FROM_KBM_ID
      order by KH_CREATE_DATE DESC
      rows 1
      into :KBM_ID, :MA_KIB_REC_TIME;
      if (MA_KIB_REC_TIME is not null) then
      begin
        ma_kib_recieve=2;
      end
    end  */
    if (ma_kib_recieve = 0) then
    begin
      select kbm_checked_datetime
      from kibminiapp
      inner join kibminiapp_link on kbl_from_kbm_id = kbm_id
      where kbl_object_kind = f_objkind('miniapp', 'id') and
          kbl_object_id = :ma_id and
          kbm_from_c_id = :kf_client_id and
          --KBM_VER_INFO in (3,4,5) and 
          kbm_checked_datetime is not null rows 1
      into :ma_kib_rec_time;
      if (ma_kib_rec_time is not null) then
        ma_kib_recieve = 1;
    end

    score = iif(kf_is_poruch <> 0 , null,
      (select
         iif(:is_ipf, c_curr_status_ipf, c_curr_status)
       from hp_get_client_status_str(:kf_client_id, current_date)));

    -- RFCKZ-198
    if (db_id = 39801) then
    begin
        pkb_eil_status = null;
        pkb_eil_answer_datetime = null;

        select first 1 eil.eil_status, eil.eil_status_datetime
        from extinfolist eil
        inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
        where eic.eic_code = 'pkb_cib_report' and
              eil.eil_object_kind = f_objkind('miniapp', 'id') and
              eil.eil_object_id = :ma_id and
              eil.eil_from_c_id = :kf_client_id
        order by eil.eil_status_datetime desc
        into :pkb_eil_status, :pkb_eil_answer_datetime;

        if (pkb_eil_status is not null) then
            ma_kib_send = 1;

        if (pkb_eil_status = 2) then
        begin
            ma_kib_recieve = 1;
            ma_kib_rec_time = :pkb_eil_answer_datetime;
        end
    end
    -- RFCKG-4861
    ma_kib_safe_and_sound_send = 0;
    ma_kib_safe_and_sound_recieve = 0;
    ma_kib_safe_and_sound_rec_time = null;
    safe_and_sound_eil_status = null;

    if (db_id = 41701) then
    begin

      select first 1 eil.eil_status,
                     eil.eil_status_datetime
      from extinfolist eil
      inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
      where eic.eic_code = 'ss_cib_report' and
            eil.eil_object_kind = f_objkind('miniapp', 'id') and
            ((eil.eil_date between :ma_date_oper - 1 and :ma_date_oper)
            or
            (eil.eil_object_id = :ma_id)) and -- RFCKG-5396
            eil.eil_from_c_id = :kf_client_id
      order by eil.eil_status_datetime desc
      into safe_and_sound_eil_status,
           ma_kib_safe_and_sound_rec_time;

      if (coalesce(safe_and_sound_eil_status, -777) <> -777) then
        ma_kib_safe_and_sound_send = 1;

      if (safe_and_sound_eil_status = 2) then
        ma_kib_safe_and_sound_recieve = 1;

    end

    suspend;
  end
end
