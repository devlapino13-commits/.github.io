CREATE PROCEDURE MC_EDOC_COUNT(
    MCL_ID D_ID)
RETURNS (
    EDL_COUNT D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable OBJ_KIND D_ID;
declare variable OBJ_ID D_ID;
declare variable PR_CODE D_STR_100;
declare variable MCL_PROJECT D_STR_CODE;
begin
  edl_count = 0;

  select
    mcl_from_c_id,
    f_enum('mcl_project', mcl_project, 'code', 0)
  from mc_login
  where mcl_id = :mcl_id
  into
    c_id,
    mcl_project;

  if (c_id is not null) then
  begin
    for select distinct edl_object_kind, edl_object_id
    from edoc_list, edoc_sign, edoc_frx
    where eds_from_c_id = :c_id and
          eds_from_edl_id = edl_id and
          edf_from_edc_id = edl_from_edc_id and
          eds_status = 0 and
          edl_status = 0
    into obj_kind, obj_id
    do
    begin
      pr_code = '';
      if (f_objkind(obj_kind, 'name') = 'kontract') then
      begin
        select pr_code
        from kontract
        inner join product on pr_id = k_from_pr_id
        where k_id = :obj_id
        into pr_code;
      end
      else
      if (f_objkind(obj_kind, 'name') = 'miniapp') then
      begin
        select pr_code
        from miniapp
        inner join product on pr_id = ma_from_pr_id
        where ma_id = :obj_id
        order by ma_number desc
        rows 1
        into pr_code;
      end

      -- RFCRU-3766
      if (mcl_project = 'kmp' and pr_code <> '' and pr_code containing '.oivo_pack.') then
        continue;

      if (mcl_project = 'oivo' and pr_code <> '' and pr_code not containing '.oivo_pack.') then
        continue;

      edl_count = edl_count + 1;
    end
  end

  suspend;
end
