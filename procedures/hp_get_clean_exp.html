CREATE PROCEDURE HP_GET_CLEAN_EXP(
    REP_DATE_END D_DATE,
    REP_C_ID D_ID = null)
RETURNS (
    REG_NAME D_STR_100,
    C_INN D_STR_20,
    C_NAME_FIO D_STR_100,
    PS_DATE_BEGIN D_DATE,
    FIRST_DISB_DATE D_DATE,
    LAST_EXP D_INTEGER,
    LAST_DECR_PERIOD D_INTEGER,
    PURE_EXP_LAST D_INTEGER,
    TOTAL_EXP D_INTEGER,
    TOTAL_DECR_PERIOD D_INTEGER,
    AAREP_DATE D_DATE,
    OUTPUT_C_ID D_ID)
AS
declare variable avg_day_month decimal(18,4);
declare variable jl_id_list d_str_255;
declare variable reg_id_old d_integer;
declare variable c_id d_id;
declare variable reg_id d_integer;
declare variable ho_id d_id;
declare variable jl_id_standart d_str_100;
declare variable jl_id_senior d_str_100;
declare variable date_begin d_date;
declare variable date_end d_date;
declare variable jl_id d_integer;
declare variable previous_jl_id d_id;
declare variable next_ps_date_begin d_date;
declare variable tmp_exp d_integer;
declare variable ho_text d_str_255;
declare variable order_type d_integer;
declare variable hrv_date_tmp d_date;
declare variable hrv_date_begin d_date;
declare variable hrv_date_end_fact d_date;
declare variable min_ps_date_begin d_date;
declare variable next_jl_id d_id;
begin
  avg_day_month = 30.4167;

  aarep_date = :rep_date_end;

  select '.' || list(jl.jl_id, '.') || '.'
  from joblist jl
  inner join jobkind jk on jk.jk_id = jl.jl_kind
  where jk.jk_system_code in ('jobsystem.lo_standart')
  into jl_id_standart;

  select '.' || list(jl.jl_id, '.') || '.'
  from joblist jl
  inner join jobkind jk on jk.jk_id = jl.jl_kind
  where jk.jk_system_code in ('jobsystem.lo_senior')
  into jl_id_senior;

  jl_id_list = trim(jl_id_standart || right(jl_id_senior, character_length(jl_id_senior) - 1));

  reg_id_old = 0;
  for select
        ps_from_c_id, ps_from_reg_id
      from hr_pstatus ps
      where :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
            ps.ps_from_c_id = coalesce(:rep_c_id, ps.ps_from_c_id) and
            ps.ps_status = 1 and
            ps.ps_is_working = 1 and
            cast(:jl_id_list as d_str_255) containing '.' || ps.ps_from_jl_id || '.'   --cast нужен для того чтобы дебаггер не ломался
      order by ps.ps_date_begin desc, ps_from_reg_id asc
      into c_id, reg_id
  do
  begin
    output_c_id = :c_id;
    reg_name = null;
    if (reg_id <> reg_id_old) then
    begin
      select rg.reg_code || ' ' || rg.reg_name
      from region rg
      where rg.reg_unicode = :reg_id
      into reg_name;
    end

    total_exp = 0;

    ps_date_begin = null;

    for select ps_date_begin, ps_date_end, ps_from_ho_id, ps_from_jl_id,
               lead(ps_date_begin) over (partition by ps_from_c_id order by ps_date_begin),
               lag(ps_from_jl_id) over (partition by ps_from_c_id order by ps_date_begin),
               lead(ps_from_jl_id) over (partition by ps_from_c_id order by ps_date_begin),
               min(ps_date_begin) over (partition by ps_from_c_id order by ps_date_begin)
    from hr_pstatus ps
    where ps.ps_from_c_id = :c_id and
          ps.ps_is_working = 1 and
          ps.ps_status = 1 and
          cast(:jl_id_list as d_str_255) containing '.' || ps.ps_from_jl_id || '.'
    order by ps_date_begin
    into date_begin, date_end, ho_id, jl_id, next_ps_date_begin, previous_jl_id, next_jl_id, min_ps_date_begin
    do
    begin
--       if (:jl_id_list not containing '.' || jl_id || '.') then
--         continue;

       if (date_end >= '31.12.9999') then
         date_end = :rep_date_end;

       if (extract(day from date_begin) <> extract(day from date_end + 1)) then
         tmp_exp = datediff(month, date_begin, date_end + 1) - 1;
       else
         tmp_exp = datediff(month, date_begin, date_end + 1);

       total_exp = total_exp + tmp_exp;

       if (exists(select *
                  from hr_pstatus ps
                  where ps.ps_from_c_id = :c_id and
                        ps.ps_status = 1 and
                        ps.ps_date_begin > :date_begin and
                        (:jl_id_list not containing '.' || ps.ps_from_jl_id || '.' or
                        ps.ps_is_working = 0) and
                        ps.ps_date_begin < :rep_date_end + 1)) then
         continue;

       if (ps_date_begin is null) then
         ps_date_begin = date_begin;

       ho_text = null;
       order_type = null;
       select ho.ho_text, ho.ho_order_type
       from hr_orders ho
       where ho.ho_id = :ho_id
       into ho_text, order_type;

       if (ho_text containing 'переимен' or
           ho_text containing 'перевести' or
           ho_text containing 'продлить') then
         continue;

       if (:jl_id_standart containing '.' || previous_jl_id || '.' and
           :jl_id_senior containing '.' || jl_id || '.') then
         continue;

       if (:jl_id_senior containing '.' || previous_jl_id || '.' and
           :jl_id_standart containing '.' || jl_id || '.') then
         continue;

       if (datediff(day, date_end, next_ps_date_begin) = 1 and order_type = 2) then
         continue;

       if (order_type in (2, 3)) then
         continue;

       if (date_begin > :rep_date_end) then
         continue;

       ps_date_begin = date_begin;
    end 

    if (extract(day from ps_date_begin) > extract(day from :rep_date_end + 1)) then
      last_exp = datediff(month, :ps_date_begin, :rep_date_end + 1) - 1;
    else
      last_exp = datediff(month, :ps_date_begin, :rep_date_end + 1) ;

    if (last_exp > total_exp) then
      total_exp = last_exp;

    last_decr_period = 0;
    total_decr_period = 0;
    for with
    cte_decr_period as (
        select distinct
            hrv.hrv_date_begin as hrv_date_begin,
            lead(hrv_date_begin) over (partition by hrv_from_c_id order by hrv_date_begin) as next_hrv_date_begin,
            lag(hrv_date_end_fact) over (partition by hrv_from_c_id order by hrv_date_begin) as prev_hrv_date_end_fact,
            1 + iif(hrv.hrv_date_end_fact >= :rep_date_end, :rep_date_end, hrv.hrv_date_end_fact) as hrv_date_end_fact
        from hr_voctravel hrv
        inner join hr_orders ho on ho.ho_id = hrv.hrv_from_ho_id and
              ho.ho_status = 1 and
              ho.ho_order_type in (16, 17, 18, 24) --всё что связано с декретом
        inner join hr_pstatus ps on ps.ps_from_c_id = hrv.hrv_from_c_id and
              ps.ps_status = 1 and
              ps.ps_is_working = 1 and
              cast(:jl_id_list as d_str_255) containing '.' || ps.ps_from_jl_id || '.'
        where hrv.hrv_from_c_id = :c_id and
              ho.ho_date between ps.ps_date_begin and ps.ps_date_end and
              ho.ho_date <= :rep_date_end
        order by hrv_date_begin
    )
    select
       iif(hrv_date_begin < prev_hrv_date_end_fact + 1, prev_hrv_date_end_fact + 1, hrv_date_begin) as hrv_date_begin,
       hrv_date_end_fact as hrv_date_end_fact
    from cte_decr_period
    into hrv_date_begin, hrv_date_end_fact
    do
    begin
      hrv_date_tmp = null;
      select hrv.hrv_date_begin
      from hr_voctravel hrv
      inner join hr_orders ho on ho.ho_id = hrv.hrv_from_ho_id and
            ho.ho_status = 1 and
            ho.ho_order_type = 20 --выход издекретного отпуска
      where hrv.hrv_from_c_id = :c_id and
            hrv.hrv_date_begin between :hrv_date_begin and :hrv_date_end_fact and
            hrv.hrv_status = 1
      into hrv_date_tmp;

      if (hrv_date_tmp is not null) then
        hrv_date_end_fact = hrv_date_tmp;

      total_decr_period = total_decr_period +  datediff(day, hrv_date_begin, hrv_date_end_fact);

      if (hrv_date_begin >= :ps_date_begin) then
        last_decr_period = last_decr_period + datediff(day, hrv_date_begin, hrv_date_end_fact);
    end 

    total_decr_period = round(total_decr_period / avg_day_month, 0);
    last_decr_period = round(last_decr_period / avg_day_month, 0);

    if (last_decr_period is not null) then
      pure_exp_last = last_exp - last_decr_period;
    else
      pure_exp_last = last_exp;

    if (total_exp < 0) then
      total_exp = 0;
    if (last_exp < 0) then
      last_exp = 0;
    if (pure_exp_last < 0) then
      pure_exp_last = 0;

    c_inn = null;
    c_name_fio = null;
    select cl.c_inn, cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into c_inn, c_name_fio;

    first_disb_date = null;
    select ks.ks_date_begin
    from kontstate ks
    where ks.ks_date_begin >= :ps_date_begin and
          ks.ks_kc_id = :c_id and
          ks.ks_state = 1 and
          ks.ks_ol_is_delete = 0
    order by ks.ks_date_begin asc
    rows 1
    into first_disb_date;

    suspend;
  end
end
