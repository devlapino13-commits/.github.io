CREATE PROCEDURE AAREP_AA_BOOKVIEW_EXPORT(
    REGION_ID INTEGER,
    DATE_BEGIN DATE,
    DATE_END DATE)
RETURNS (
    OB_CURRENCY VARCHAR(3),
    OB_CURR_RATE DECIMAL(18,4),
    OB_CLIENT_DT VARCHAR(20),
    OB_ACCOUNT_DT VARCHAR(20),
    OB_SUMMA_VAL DECIMAL(18,4),
    OB_SUMMA DECIMAL(18,4),
    OB_FROM_PC_ID VARCHAR(20),
    OB_FROM_KP_ID VARCHAR(10),
    OB_ACCOUNT_KT VARCHAR(20),
    OB_CLIENT_KT VARCHAR(20),
    OB_DETAIL VARCHAR(1024),
    OB_SUBKONTO_DT INTEGER,
    OB_SUBKONTO_KT INTEGER,
    OL_ID VARCHAR(20),
    OL_FROM_REG_ID INTEGER,
    OL_DATE DATE,
    OL_DOC_NO INTEGER,
    OL_DOC_NO_STR VARCHAR(20),
    OL_MEMO VARCHAR(1024),
    IS_DELETE SMALLINT,
    OL_DATETIME TIMESTAMP,
    OL_DOC_CLIENT_NAME VARCHAR(255),
    OL_YEAR INTEGER,
    OL_MONTH INTEGER,
    UN_NAME VARCHAR(30),
    DN_NAME VARCHAR(255),
    MO_NAME VARCHAR(100),
    CLIENT_DT VARCHAR(255),
    CLIENT_KT VARCHAR(255),
    IS_DELETE_NAME VARCHAR(20),
    OL_CALC_MEMO VARCHAR(1024),
    OB_ACC_SHORT_DT VARCHAR(20),
    OB_ACC_SHORT_KT VARCHAR(20),
    OL_REGION_NAME VARCHAR(100),
    OL_BRANCH_NAME VARCHAR(100),
    OB_LOOK_DT_NAME VARCHAR(200),
    OB_LOOK_KT_NAME VARCHAR(200),
    OL_OPER_TYPE INTEGER,
    OL_DOC_KIND INTEGER,
    OL_IS_CLIENT INTEGER,
    OL_POCKET INTEGER,
    OB_SUB_DT_NAME VARCHAR(255),
    OB_SUB_KT_NAME VARCHAR(255),
    OL_IS_SALARY_LIST INTEGER,
    OL_OPER_SUBTYPE INTEGER,
    DT_BANK_ACC VARCHAR(20),
    KT_BANK_ACC VARCHAR(20),
    DT_BIK_BANK_ACC VARCHAR(20),
    KT_BIK_BANK_ACC VARCHAR(20),
    OB_NUMBER INTEGER)
AS
declare variable ol_id_prior decimal(18,0);
declare variable ob_number_prior integer;
declare variable c_id_dt decimal(18,0);
declare variable c_id_kt decimal(18,0);
declare variable ol_id_num decimal(18,0);
declare variable ccc decimal(18,0);
begin
  ol_id_prior = -1;
  ob_number_prior = -1;
  for select distinct ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_from_pc_id,
                      ob_from_kp_id, ob_account_kt, ob_client_kt, ob_detail, ob_subkonto_dt, ob_subkonto_kt, ol_id,
                      ob_number, ol_from_reg_id, ol_date, ol_doc_no, ol_doc_no_str, ol_memo, is_delete, ol_datetime,
                      ol_doc_client_name, ol_year, ol_month, un_name, dn_name, mo_name, client_dt, client_kt,
                      is_delete_name, ol_calc_memo, ob_acc_short_dt, ob_acc_short_kt, ol_region_name, ol_branch_name,
                      ob_look_dt_name, ob_look_kt_name, ol_oper_type, ol_doc_kind, ol_is_client, ol_pocket,
                      ob_sub_dt_name, ob_sub_kt_name, ol_is_salary_list, ol_oper_subtype, b_dt.br_account_no,
                      b_dt.br_from_b_id, b_kt.br_account_no, b_kt.br_from_b_id
      from aarep_aa_bookview(:region_id, :date_begin, :date_end, 1, '', '', '', '', -1)
      left join bankregion b_dt on (b_dt.br_name = ob_account_dt)
      left join bankregion b_kt on (b_kt.br_name = ob_account_kt)
      where ((ol_from_reg_id <> 1010100 and
            ob_acc_short_dt <> '71-01/1' and
            ob_acc_short_kt <> '71-01/1' and
            ((ob_acc_short_dt = '57-01/1' and
            ob_acc_short_kt = '50-01/1') or
            (ob_acc_short_kt = '57-01/1' and
            ob_acc_short_dt = '50-01/1'))
--      and OB_ACC_SHORT_DT<>'57-01/1'
--      and OB_ACC_SHORT_KT<>'57-01/1'
            and
            ob_acc_short_dt <> '90-09/1' and
            ob_acc_short_kt <> '90-09/1') or
            (ol_from_reg_id = 1010100 and
            ob_acc_short_dt <> '71-01/1' and
            ob_acc_short_kt <> '71-01/1' and
            ((ob_acc_short_dt = '51-01/12') or
            (ob_acc_short_dt = '57-01/1' and
            ob_acc_short_kt = '50-01/1') or
            (ob_acc_short_kt = '57-01/1' and
            ob_acc_short_dt = '50-01/1'))
--      and OB_ACC_SHORT_DT<>'57-01/1'
--      and OB_ACC_SHORT_KT<>'57-01/1'
            and
            not(ob_acc_short_dt like '51-%') and
            not(ob_acc_short_kt like '51-%') and
            ob_acc_short_dt <> '90-09/1' and
            ob_acc_short_kt <> '90-09/1'))

      order by ol_id, ob_number, b_dt.br_number desc, b_kt.br_number desc
      into :ob_currency, :ob_curr_rate, :c_id_dt, :ob_account_dt, :ob_summa_val, :ob_summa, :ob_from_pc_id,
           :ob_from_kp_id, :ob_account_kt, :c_id_kt, :ob_detail, :ob_subkonto_dt, :ob_subkonto_kt, :ol_id_num,
           :ob_number, :ol_from_reg_id, :ol_date, :ol_doc_no, :ol_doc_no_str, :ol_memo, :is_delete, :ol_datetime,
           :ol_doc_client_name, :ol_year, :ol_month, :un_name, :dn_name, :mo_name, :client_dt, :client_kt,
           :is_delete_name, :ol_calc_memo, :ob_acc_short_dt, :ob_acc_short_kt, :ol_region_name, :ol_branch_name,
           :ob_look_dt_name, :ob_look_kt_name, :ol_oper_type, :ol_doc_kind, :ol_is_client, :ol_pocket, :ob_sub_dt_name,
           :ob_sub_kt_name, :ol_is_salary_list, :ol_oper_subtype, :dt_bank_acc, :dt_bik_bank_acc, :kt_bank_acc,
           :kt_bik_bank_acc
  do
  begin
    if (not(ob_acc_short_dt like '57-04%' and
        ob_acc_short_kt like '57-04%')) then
    begin
      ccc = null;
      if (ob_acc_short_dt like '58-%') then
      begin
        select kc_from_c_id
        from operkred, knclient
        where ok_from_ol_id = :ol_id_num and
              ok_from_k_id = kc_from_k_id and
              kc_amount_approved is not null and
              kc_amount_approved > 0 and
              kc_status = 2 and
              ok_from_c_id = 0 rows 1
        into :ccc;
        if (ccc is not null) then
          c_id_dt = :ccc;
      end
      if (ob_acc_short_kt like '58-%') then
      begin
        select kc_from_c_id
        from operkred, knclient
        where ok_from_ol_id = :ol_id_num and
              ok_from_k_id = kc_from_k_id and
              kc_amount_approved is not null and
              kc_amount_approved > 0 and
              kc_status = 2 and
              ok_from_c_id = 0 rows 1
        into :ccc;
        if (ccc is not null) then
          c_id_kt = :ccc;
      end
      ob_client_dt = 'N' || cast(c_id_dt as varchar(15));
      ob_client_kt = 'N' || cast(c_id_kt as varchar(15));
      ol_id = cast(ol_id_num as varchar(20));
      if ((ol_id_prior <> :ol_id_num) or (ob_number_prior <> :ob_number)) then
        suspend;
    end
    ol_id_prior = :ol_id_num;
    ob_number_prior = :ob_number;
  end
end
