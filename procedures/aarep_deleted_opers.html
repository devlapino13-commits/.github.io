CREATE PROCEDURE AAREP_DELETED_OPERS(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    OL_DOC_NO INTEGER,
    OL_REG_NAME VARCHAR(100),
    OL_MEMO VARCHAR(1024),
    OL_DATETIME TIMESTAMP,
    OL_OPER_SUMMA DECIMAL(18,4))
AS
declare variable ol_id decimal(18,0);
declare variable ol_parent_id decimal(18,0);
declare variable kol integer;
declare variable dt varchar(20);
declare variable kt varchar(20);
declare variable ob_summa decimal(18,4) = 0;
begin
  for select ol_id, ol_doc_no, reg_code || ' ' || reg_name, ol_memo, ol_datetime, ol_parent_id
      from operlist, region, region_list(:rep_region_id, 1, -1, 1)
      where operlist.is_delete = 1 and
            ol_date_oper_day = :rep_date and
            reg_unicode = ol_from_reg_id and
            reg_unicode = id_list
      order by reg_code, ol_doc_no
      into :ol_id, :ol_doc_no, :ol_reg_name, :ol_memo, :ol_datetime, :ol_parent_id
  do
  begin
    if (ol_parent_id is null) then
    begin
      ol_oper_summa = 0;
      kol = 0;
      for select ob_account_dt, ob_account_kt, ob_summa
          from operbook
          where ob_from_ol_id = :ol_id
          into :dt, :kt, :ob_summa
      do
      begin
        if (exists(select *
                   from acc_list, acctypelist
                   where ((al_id = :dt) or
                         (al_id = :kt)) and
                         al_from_bg_id like atl_acc_mask || '%' and
                         atl_from_at_id in (11, 12) /* Касса или банк*/ )) then
        begin
          ol_oper_summa = ol_oper_summa + ob_summa;
          kol = kol + 1;
        end
      end
      if (kol is null) then
        kol = 0;
      if (ol_oper_summa is null) then
        ol_oper_summa = 0;
      if (kol > 0) then
        suspend;
    end
  end
end
