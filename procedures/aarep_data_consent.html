CREATE PROCEDURE AAREP_DATA_CONSENT(
    REP_REGION_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_CODE,
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    CONT_INFO D_STR_100,
    EDL_STATUS D_STR_CODE,
    K_DISB_DATE D_DATE,
    KC_NAME_FIO D_STR_100)
AS
declare variable c_id d_id;
declare variable edc_id d_id;
begin
    edc_id = f_edoc_kind('persdata.consent', 'id');

    for select reg.reg_name,
               c.c_name_fio,
               k.k_kontract_no,
               f_cont_info('clients', c.c_id, 'cell_main', 'ok'),
               k.k_disb_date,
               ks.c_name_fio,
               c.c_id
        from oksummary oks
        inner join region_list(:rep_region_id, 1, 3, 1) on id_list = oks.oks_region_id
        inner join region reg on reg.reg_unicode = id_list
        inner join knclient kc on kc.kc_from_k_id = oks.oks_from_k_id
        inner join clients c on c.c_id = kc.kc_from_c_id
        inner join kontract k on k.k_id = kc.kc_from_k_id
        inner join clients ks on ks.c_id = oks.oks_kc_id
        where oks_close_state = 0 and
              :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
              kc.kc_amount_approved > 0
        into :reg_name,
             :c_name_fio,
             :k_kontract_no,
             :cont_info,
             :k_disb_date,
             :kc_name_fio,
             :c_id
    do
    begin
        edl_status = 'Не подписан';

        select iif(edl.edl_status = 1, 'Подписан', 'Не подписан')
        from edoc_list edl
        where edl.edl_object_id = :c_id and
              edl.edl_from_edc_id = :edc_id and
              edl.edl_object_kind = 0 and
              :rep_date_end between edl_blank_date and coalesce(edl_date_end, :rep_date_end)
        order by edl.edl_status desc, edl.edl_id desc
        rows 1
        into :edl_status;

        suspend;
    end
end
