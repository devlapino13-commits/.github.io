CREATE PROCEDURE HP_GET_RESCHEDULE(
    KC_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    KOL_RESH_NOT_KK D_SMALLINT,
    FIRST_RESH_DATE_NOT_KK D_DATE,
    FIRST_RESH_DATE_KK D_DATE,
    KOL_RESH_KK D_SMALLINT,
    RESH_KIND_STR D_STR_10,
    LAST_RESH_DATE_NOT_KK D_DATE)
AS
declare variable K_ID D_ID;
declare variable RD_CODE3 D_STR_255;
begin

  select kc_from_k_id from knclient
  where kc_id = :kc_id
  into k_id;

  kol_resh_not_kk = null;

  with tabl1 as (
    select distinct oks_last_resh_date, 1 schet, null rd_code from kn_oksummary
    where oks_from_kc_id = :kc_id
    and oks_last_resh_date is not null
    and oks_date_begin <= :rep_date
    union
    select or_ol_date, -1, rd_code
    from operrest, refdata
    where or_from_k_id = :k_id
    and or_ol_is_active = 1
    and or_ol_date <= :rep_date
    and or_from_rd_id = rd_id
    and rd_code in ('Reason_Reschedule.8790', 'Reason_Reschedule.10086')
    ),
  tabl2 as (
    select oks_last_resh_date,
           sum(schet) schet2,
           max(rd_code) rd_code2
    from tabl1
    group by 1
    ),
  tabl3 as (
    select tabl2.*,
           -- сортирует по дате где rd_code не нул и берет первый запись
           iif( (dense_rank() over (order by iif(rd_code2 is not null, oks_last_resh_date, '31.12.9999'))) = 1, rd_code2, null ) rd_code3
    from tabl2
  )
  select sum(iif(schet2 = 1, 1, 0)), -- 19.1 "Кол-во реструктуризаций (кроме КК)"
         min(iif(schet2 = 1, oks_last_resh_date, null)), -- 19.2 "Дата первой реструктуризации (кроме КК)"
         min(iif(schet2 <= 0, oks_last_resh_date, null)), -- 19.3 "Дата первой реструктуризации КК"
         sum(iif(schet2 <= 0, 1, 0)), -- 19.4 "Кол-во реструктуризаций КК"
         max(rd_code3), -- 19.5 "Вид последней рестртуктуризации КК"
         max(iif(schet2 = 1, oks_last_resh_date, null)) --19. "Дата последней реструктуризации (кроме КК)"

  from tabl3
  into kol_resh_not_kk,
       first_resh_date_not_kk,
       first_resh_date_kk,
       kol_resh_kk,
       rd_code3,
       last_resh_date_not_kk
       ;

  resh_kind_str = null;
  if (rd_code3 = 'Reason_Reschedule.8790') then resh_kind_str = 'КК СВО';
  else if (rd_code3 = 'Reason_Reschedule.10086') then resh_kind_str = 'КК ТЖС';

  suspend;
end
