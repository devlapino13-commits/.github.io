CREATE PROCEDURE HP_SURVEY_FILL(
    KC_ID D_ID,
    RD_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    ACL_ID D_ID)
AS
declare variable obj_kind d_integer;
declare variable obj_kind1 d_integer;
declare variable obj_kind2 d_integer;
declare variable obj_id d_id;
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable respons_c_id d_id;
declare variable new_opr_id d_id;
declare variable opr_id d_id;
declare variable ovs_id d_integer;
declare variable acl_date d_date;
begin
  acl_id = 0;
  obj_kind1 = f_objkind('Clients', 'id');
  obj_kind2 = f_objkind('KnClient', 'id');

  if (exists(
    select * from actionlist
    where acl_object_kind = :obj_kind2
      and acl_object_id = :kc_id
      and acl_from_rd_id = :rd_id))
  then exit;

  select
    kc_from_c_id,
    k_last_reg_id
  from knclient
    inner join kontract
      on k_id = kc_from_k_id
  where kc_id = :kc_id
  into
    :c_id,
    :reg_id;

  for
    with tmp as (
      select
        acl_id,
        acl_date,
        acl_respons_c_id,
        acl_object_kind,
        acl_object_id
      from knclient
        inner join actionlist
          on acl_object_kind = :obj_kind2
            and acl_object_id = kc_id
            and acl_from_rd_id = :rd_id
      where kc_from_c_id = :c_id
        and kc_amount_approved > 0
        and kc_id < :kc_id
      union
      select
        acl_id,
        acl_date,
        acl_respons_c_id,
        acl_object_kind,
        acl_object_id
      from actionlist
      where acl_object_kind = :obj_kind1
        and acl_object_id = :c_id
        and acl_from_rd_id = :rd_id
    )
    select first 1
      acl_date,
      acl_respons_c_id,
      acl_object_kind,
      acl_object_id
    from tmp
    order by acl_id desc
  into
    :acl_date,
    :respons_c_id,
    :obj_kind,
    :obj_id
  do
  begin
    acl_date = coalesce(acl_date, current_date);
    if (respons_c_id is null) then
      respons_c_id = (
        select un_from_c_id
        from username where un_id = :un_id);
    for
      select
        opr_id,
        opr_actual_ovs_id
      from actiongroup
        inner join objectprops
          on opr_object_kind = :obj_kind
            and opr_object_id = :obj_id
            and opr_subject_kind = -1
            and opr_subject_id = acg_from_rd_id
      where acg_group_rd_id = :rd_id
        and acg_parent_id is null
        and acg_is_delete = 0
    into
      :opr_id,
      :ovs_id
    do
    begin
      new_opr_id = gen_id(gen_objectprops_id, 1);
      insert into objectprops (
        opr_id,
        opr_object_id,
        opr_subject_id,
        opr_object_kind,
        opr_subject_kind,
        opr_actual_ovs_id,
        opr_fact_ovs_id
      )
      select
        :new_opr_id,
        :kc_id,
        opr_subject_id,
        :obj_kind2,
        opr_subject_kind,
        1,
        1
      from objectprops
      where opr_id = :opr_id;

      insert into objectvalues (
        ovs_from_opr_id,
        ovs_id,
        ovs_date_begin,
        ovs_value,
        ovs_from_un_id,
        ovs_add_value
      )
      select
        :new_opr_id,
        1,
        current_date,
        ovs_value,
        :un_id,
        ovs_add_value
      from objectvalues
      where ovs_from_opr_id = :opr_id
        and ovs_id = :ovs_id;

      insert into objectvalres (
        ovr_from_opr_id,
        ovr_from_ovs_id,
        ovr_result_rd_id,
        ovr_state,
        ovr_value
      )
      select
        :new_opr_id,
        1,
        ovr_result_rd_id,
        ovr_state,
        ovr_value
      from objectvalres
      where ovr_from_opr_id = :opr_id
        and ovr_from_ovs_id = :ovs_id;
    end

    if (new_opr_id > 0) then
      insert into actionlist (
        acl_date,
        acl_from_reg_id,
        acl_object_kind,
        acl_object_id,
        acl_from_rd_id,
        acl_respons_c_id,
        acl_from_un_id)
      values (
        :acl_date,
        :reg_id,
        :obj_kind2,
        :kc_id,
        :rd_id,
        :respons_c_id,
        :un_id)
      returning acl_id
      into :acl_id;

    suspend;
  end
end
