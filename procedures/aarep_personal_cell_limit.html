CREATE PROCEDURE AAREP_PERSONAL_CELL_LIMIT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    LIM_DATE_BEGIN DATE,
    LIM_HO_DATE DATE,
    LIM_HO_TEXT VARCHAR(1024),
    LIM_HO_ORDER_NO VARCHAR(20),
    LIM_JL_NAME VARCHAR(100),
    LIM_JL_ORDER_NO INTEGER,
    LIM_REG_CODE_NAME VARCHAR(50),
    LIM_IS_WORKING_TEXT VARCHAR(50),
    LIM_JL_CATEGORY VARCHAR(50),
    LIM_FIO VARCHAR(100),
    LIM_FROM_C_ID DECIMAL(18,0),
    LIM_JL_ID INTEGER,
    LIM_JL_LIMIT_AMOUNT DECIMAL(18,4),
    LIM_JL_LIMIT_DATE DATE,
    LIM_PERS_LIMIT_AMOUNT DECIMAL(18,4),
    LIM_PERS_LIMIT_DATE DATE,
    LIM_PERS_LIMIT_NEW DECIMAL(18,4))
AS
declare variable jl_id_prior integer;
declare variable jl_kind integer;
declare variable cm_kc_cell_limit decimal(18,4);
declare variable rep_client_kolich integer;
declare variable cm_kc_cell_limit_min decimal(18,4);
declare variable cm_kc_cell_limit_max decimal(18,4);
declare variable rep_client_kolich_ind integer;
declare variable kol_kc integer;
declare variable rep_client_kolich_all integer;
begin
  select cm_kc_cell_limit, cm_kc_cell_limit_min, cm_kc_cell_limit_max
  from company
  into cm_kc_cell_limit, :cm_kc_cell_limit_min, :cm_kc_cell_limit_max;
  if (cm_kc_cell_limit is null) then
    cm_kc_cell_limit = -1;
  if (cm_kc_cell_limit_min is null) then
    cm_kc_cell_limit_min = 0;
  if (cm_kc_cell_limit_max is null) then
    cm_kc_cell_limit_max = 0;
  jl_id_prior = 0;
  lim_jl_limit_amount = 0;
  lim_jl_limit_date = null;
  for select ps_date_begin, ho_date, ho_text, ho_order_no, jl_id, jl_name, jl_order_no, reg_code_name,
             ps_is_working_text, jl_category, c_name_fio, ps_from_c_id, jl_kind
      from aarep_personal_cell_list(:report_region_id, :report_date_end)
      order by jl_order_no, jl_id
      into :lim_date_begin, :lim_ho_date, :lim_ho_text, :lim_ho_order_no, :lim_jl_id, :lim_jl_name, :lim_jl_order_no,
           :lim_reg_code_name, :lim_is_working_text, :lim_jl_category, :lim_fio, :lim_from_c_id, :jl_kind
  do
  begin
    if (jl_id_prior <> lim_jl_id) then
    begin
      if ((jl_kind not in (5, 6)) or (cm_kc_cell_limit = -1)) then -- Изменен от 04.06.2013, было if ((jl_kind<>5) or (cm_kc_cell_limit=-1)) then
      begin
        lim_jl_limit_amount = 0;
        select sum(cjl_amount), max(cjl_date)
        from celljoblimit
        where cjl_from_jl_id = :lim_jl_id and
              cjl_date <= :report_date_end
        into :lim_jl_limit_amount, :lim_jl_limit_date;
        if (lim_jl_limit_amount is null) then
          lim_jl_limit_amount = 0;
      end
      jl_id_prior = lim_jl_id;
    end
-- Если НЕ КС
    if ((jl_kind not in (5, 6)) or (cm_kc_cell_limit = -1)) then -- Изменен от 04.06.2013, было if ((jl_kind<>5) or (cm_kc_cell_limit=-1)) then
    begin
      select max(cpl_date), sum(cpl_amount)
      from cellperlimit
      where cpl_from_c_id = :lim_from_c_id
      into :lim_pers_limit_date, :lim_pers_limit_amount;
      if (lim_pers_limit_amount is null) then
        lim_pers_limit_amount = 0;
      if (lim_is_working_text <> 'Уволен') then
        lim_pers_limit_new = lim_jl_limit_amount;
      else
        lim_pers_limit_new = 0;
    end
    else
    begin /* Лимит КС согласно кол-во клиентов */
      if (jl_kind = 5) then
      begin
        select rep_client_kolich, rep_client_kolich_ind
        from aarep_get_kc_client(:lim_from_c_id, :report_date_end)
        into :rep_client_kolich, :rep_client_kolich_ind;
        if (report_date_end < '01.05.2013') then
        begin
          lim_pers_limit_amount = :rep_client_kolich * cm_kc_cell_limit;
          lim_pers_limit_amount = lim_pers_limit_amount + :rep_client_kolich_ind * cm_kc_cell_limit * 5;
          if (lim_pers_limit_amount < :cm_kc_cell_limit_min) then
            lim_pers_limit_amount = cm_kc_cell_limit_min;
          if (lim_pers_limit_amount > :cm_kc_cell_limit_max) then
            lim_pers_limit_amount = :cm_kc_cell_limit_max;
          lim_jl_limit_amount = :lim_pers_limit_amount;
          lim_pers_limit_new = lim_jl_limit_amount;
        end
        else
        begin
          lim_pers_limit_amount = (:rep_client_kolich + :rep_client_kolich_ind) * cm_kc_cell_limit; -- добавлено по приказу 103-06-13 от 03.06.2013
          lim_jl_limit_amount = :lim_pers_limit_amount;
          lim_pers_limit_new = lim_jl_limit_amount;
        end
      end
      else
      begin
--     Лимит по КМ, находит закрепленных КСников
        kol_kc = 0;
--     Расчет кол-во кредитов
        lim_pers_limit_amount = 0;
        lim_jl_limit_amount = :lim_pers_limit_amount;
        lim_pers_limit_new = lim_jl_limit_amount;
      end

      if (lim_is_working_text = 'Уволен') then
      begin
        lim_pers_limit_new = 0;
        lim_pers_limit_amount = 0;
      end
    end

    suspend;
  end
end
