CREATE PROCEDURE HP_COM_LIST_AUTO_APPROVE(
    CML_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable cmp_id d_id;
begin
  -- автоподтверждение
  res_is_error = 0;
  res_text = '';

  if (not exists(select *
      from com_cfg, com_list
      where cml_id = :cml_id and
            cmc_id = cml_from_cmc_id and
            cmc_auto_approve = 1 and
            cmc_from_edc_id is not null)) then
  begin
    suspend;
    exit;
  end

  select cmp_id
  from com_packet
  where cmp_date = current_date
  into cmp_id;

  if (cmp_id is null) then
  begin
    select cmp_id
    from hp_com_packet_action(null , 'create', 1010100000050) -- Пользователь Зара Гитинова
    into :cmp_id;
  end

  select res_text, res_is_error
  from hp_com_action_add(:cml_id, :cmp_id, 'process', 1000000000000)
  into :res_text, :res_is_error;

  if (res_is_error = 0) then
  begin
    select res_text, res_is_error
    from hp_com_action_add(:cml_id, :cmp_id, 'approved', 1000000000000)
    into :res_text, :res_is_error;
  end

  if (res_is_error = 0) then
  begin
    select res_text, res_is_error
    from hp_com_action_add(:cml_id, :cmp_id, 'signing', 1000000000000)
    into :res_text, :res_is_error;
  end

  if (res_is_error = 1) then
  begin
    select res_is_error
    from hp_com_action_add(:cml_id, :cmp_id, 'rework', 1000000000000)
    into :res_is_error;

    select res_is_error
    from hp_com_memo_add(:cml_id, :res_text, 1, 1000000000000)
    into :res_is_error;

    res_is_error = 1;
  end

  suspend;
end
