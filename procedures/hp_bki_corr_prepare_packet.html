CREATE PROCEDURE HP_BKI_CORR_PREPARE_PACKET(
    USE_OLD_DATA_IN_TITLE D_BOOL = 0,
    CHANGE_REASON D_INTEGER = null)
AS
declare variable packet_id d_id;
declare variable bkc_id d_id;
declare variable bev_id d_id;
declare variable bfv_id d_id;
declare variable bkd_id d_id;
declare variable bkk_id d_id;
declare variable c_id d_id;
declare variable xml_version_id d_integer;
declare variable jso_client d_integer;
declare variable bkc_add d_str_long;
begin
  select bxv_id
  from bki_xml_version
  where bxv_date_begin <= current_date
  order by bxv_date_begin desc
  rows 1
  into xml_version_id;

  if (xml_version_id is null) then
    exception e_default 'Not found xml_version_id';

  for select bkp_id, bkk_id
  from bki_packet, bki_kind, bki_packet_type
  where bkp_status = -5
    and bkk_id = bkp_from_bkk_id
    and bkt_id = bkp_from_bkt_id
  into packet_id, bkk_id
  do
  begin
    while (true)
    do
    begin
      select bev_id, bkc_id, bkc_from_c_id
      from bki_loans, bki_clients, bki_events
      where bkc_from_bkl_id = bkl_id
        and bev_from_bkc_id = bkc_id
        and bkl_from_bkp_id = :packet_id
        and bev_status = 0
      order by bkl_object_id
      rows 1
      into bev_id, bkc_id, c_id;
   
      bkc_add = f_format('{"change_reason":"%1"}', coalesce(change_reason, 1));

      update bki_clients
      set bkc_add = :bkc_add
      where bkc_id = :bkc_id
        and bkc_add is distinct from :bkc_add;


      if (row_count = 0) then
         leave;              -- выход из цикла
     if (use_old_data_in_title = 1) then
     begin
        -- если титульная часть (ТЧ) содержится в изменяемых данных, то скопируем старые значения в ТЧ записи, что бы правильно идентифицировать СКИ
        -- пока считаем что ТЧ обязательно есть полностью
        for with recursive tab as(
          select bfv_id, bxf_from_bki_code as bki_code, bfv_parent_id, bxf_from_bki_code
          from bki_xml_fields_ver, bki_xml_fields
          where bxf_id = bfv_from_bxf_id
            and bfv_from_bxv_id = :xml_version_id
          order by bfv_position
        ), t1 (id, parent_id, bki_code) as(
          select bfv_id, bfv_parent_id, bxf_from_bki_code
          from tab
          where bki_code = 'Title'
          union all
          select bfv_id, bfv_parent_id, bxf_from_bki_code
          from t1, tab
          where bfv_parent_id = id
        )
        select id, bsd_from_bkd_id
        from t1, bki_send_data_corr, bki_xml_fields_ver, bki_xml_fields
        where bfv_id = bsd_from_bfv_id
          and bxf_id = bfv_from_bxf_id
          and bsd_from_bev_id = :bev_id
          and bxf_from_bki_code = bki_code
          and bsd_from_bkd_id is not null
        into bfv_id, bkd_id
        do
          insert into bki_send_data_oper (bsd_from_bfv_id, bsd_obj_kind, bsd_obj_id, bsd_from_bkd_id)
          values (:bfv_id, 1, :bkc_id, :bkd_id);
      end
      else
      begin
        -- Заполним ТЧ из БД
        jso_client = json.obj_create();
        json.obj_set_s(jso_client, 'bkk_id', bkk_id);
        json.obj_set_s(jso_client, 'deal_sign_credit', 1);
        execute procedure hp_bki_get_client_info5(:c_id, current_date, :jso_client);
        execute procedure hp_bki_calc_save('Title', xml_version_id, 1, bkc_id, jso_client);
      end

      update bki_events
      set bev_status = 1,
          bev_event_time = localtime,
          bev_event_date = current_date
      where bev_id = :bev_id;
    end

    -- пакет подготовлен
    update bki_packet
    set bkp_status = 1,
        bkp_result_status = 'Prepared'
    where bkp_id = :packet_id;
  end
end
