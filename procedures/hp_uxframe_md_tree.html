CREATE PROCEDURE HP_UXFRAME_MD_TREE(
    UX_CODE D_STR_CODE,
    MASTER_FR_CODE D_STR_CODE)
RETURNS (
    FR_CODE D_STR_100,
    MASTER_FRAME D_STR_CODE,
    MASTER_FIELD D_STR_CODE,
    DETAIL_FIELD D_STR_CODE,
    SHOW_KIND D_SMALLINT,
    RES_CAPTION D_STR_100,
    RES_IMAGE D_INTEGER,
    RES_TABLE_NAME D_STR_CODE)
AS
declare variable tmp_fr_code d_str_code;
declare variable ucf_is_mainframe d_bool;
declare variable form_table d_str_code;
begin
  if (trim(coalesce(master_fr_code, '')) = '') then
    master_fr_code = null;

  select ux_table_name
  from uxcode
  where ux_code = :ux_code
  into form_table;

  for select ucf_ufr_code, ux_name, ucf_master_ufr_code, ucf_master_field, ucf_detail_field, ufr_show_kind, ucf_is_mainframe, ux_table_name
  from uxcodeframe, uxframe, uxcode
  where ucf_ux_code = :ux_code and
        ucf_ufr_code = ufr_code and
        ufr_code = ux_code and
        ucf_master_ufr_code is not distinct from :master_fr_code
  into fr_code, res_caption, master_frame, master_field, detail_field, show_kind, ucf_is_mainframe, res_table_name
  do
  begin
    res_image = 17;

    if (master_field is null) then
      master_field = '';

    if (detail_field is null) then
      detail_field = '';

    if (ucf_is_mainframe = 1) then
    begin
      res_caption = res_caption || '  (основной)';

      res_image = 47;
    end else
      if (res_table_name <> form_table) then
        res_image = 18;

    suspend;

    tmp_fr_code = fr_code;
    for select fr_code, res_caption, master_frame, master_field, detail_field, show_kind,
               res_table_name, res_image
    from hp_uxframe_md_tree(:ux_code, :tmp_fr_code)
    into :fr_code, :res_caption, :master_frame, :master_field, :detail_field, :show_kind,
         :res_table_name, :res_image
    do
      suspend;
  end
end
