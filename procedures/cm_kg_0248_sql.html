CREATE PROCEDURE CM_KG_0248_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable COMMISSION D_CURRENCY;
declare variable OK_AMOUNT D_CURRENCY;
declare variable k_amount_approved D_CURRENCY;
declare variable C_ID D_ID;
declare variable POINT D_INTEGER;
declare variable INSURANCE_REQUIRED D_SMALLINT;
declare variable k_disb_date d_date;
begin
  if (f_get_const('new_loan_conditions') <> 1) then
    exit;

  res_error_is_critical = 1;
  begin
    for select cl.c_name_fio,
           cl.c_id,
           ik.ik_commission,
           k.k_disb_date,
           k.k_amount_approved
    from kontract k
    inner join knclient kc on k.k_id = kc.kc_from_k_id
    inner join product p on k.k_from_pr_id = p.pr_id
    inner join clients cl on cl.c_id = kc.kc_from_c_id
    inner join insurance_kontract ik on ik.ik_from_k_id = k.k_id
    where k.k_id = :k_id
        and kc.kc_amount_approved > 0
        and p.pr_code not containing '.ipf.'
        and p.pr_code containing '.ind.'
        and p.pr_code containing '.standart.'
        and p.pr_code containing '.consultation.'
        and p.pr_code containing '.INSUR.'
    into :res_info_text, :c_id , :commission, k_disb_date, k_amount_approved
    do
    begin
      point = coalesce(f_point(:c_id, :k_disb_date), 0);

      if (point < 20) then
      -- Баллы 0–19 страховка ОБЯЗАТЕЛЬНО
        insurance_required = 1;
      else
        -- Баллы 20 и выше → смотрим сумму кредита
      insurance_required = iif(k_amount_approved >= 250000, 1, 0);

      if (insurance_required = 0) then
        exit;

      ok_amount = 0;
      select sum(ok.ok_amount) from operkontract ok
      inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
      where ok.ok_from_k_id = :k_id
      and ok.ok_operation_type = f_get_rd_id('CashTransactionIncom.Insurance')
      and ol.is_delete = 0
      and ol_document_client = :c_id
      into :ok_amount;
    
      ok_amount = coalesce(ok_amount, 0);
    
      res_error_text='';
    
      if (ok_amount = 0) then
      begin
        res_error_text = 'Оплата за страхование не произведена';
        suspend;
      end
      else if (ok_amount < commission) then
      begin
        res_error_text = 'Оплата за страхование произведена не в полном объеме. Цена за страхование = '||round(commission)||' сом . Оплачено: '||round(ok_amount)||' сом';
        suspend;
      end
      else if (ok_amount > commission) then
      begin
        res_error_text = 'Оплата за страхование произведена в большем объеме. Цена за страхование = '||round(commission)||' сом . Оплачено: '||round(ok_amount)||' сом';
        suspend;
      end
    end
  end
end
