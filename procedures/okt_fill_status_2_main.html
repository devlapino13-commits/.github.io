CREATE PROCEDURE OKT_FILL_STATUS_2_MAIN(
    PROJECT STR_100,
    OBJ_ID ID_TYPE,
    OBJECT_KIND STR_100,
    OBJECT_ID ID_TYPE,
    CALL_RESULT_CODE STR_100,
    SUBJECT_KIND STR_100 = 'ContactInfo.Cell_Main')
RETURNS (
    MESS_STR STR_255)
AS
declare variable CALL_RESULT D_INTEGER;
declare variable UN_ID D_ID;
begin
  select rd.rd_code
  from pantera_kontracts pak
  inner join refdata rd on rd.rd_id = pak.pant_rd_project_type
  where pak.pant_id = :obj_id
  into :project;

  un_id = 1000000000000;

  select rd.rd_id
  from refdata rd
  where rd.rd_code = :call_result_code
  into :call_result;

  select mess_str
  from okt_sirena_make_call(0, current_date, 'MAKE_CALL', null, :project, :obj_id, 0, 0, :object_kind, :object_id, :subject_kind, :un_id, null, :call_result, '', null, 0)
  into :mess_str;

  suspend;
end
