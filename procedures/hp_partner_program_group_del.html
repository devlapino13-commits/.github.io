CREATE PROCEDURE HP_PARTNER_PROGRAM_GROUP_DEL(
    PPG_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
begin
  res_is_error = 1;
  res_text = '';

  if (not exists(select *
                 from partner_program_group
                 where ppg_id = ppg_id)) then
    res_text = 'Ошибка клиент не найден';

  if (res_text = '') then
  begin
    delete from partner_program_group
    where ppg_id = ppg_id;

    res_is_error = 0;
  end

  suspend;
end
