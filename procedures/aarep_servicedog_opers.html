CREATE PROCEDURE AAREP_SERVICEDOG_OPERS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    SD_ID INTEGER,
    SD_DOGOVOR_NO VARCHAR(20),
    SD_FROM_REG_ID INTEGER,
    SD_FROM_C_ID DECIMAL(18,0),
    SD_CATEGORY VARCHAR(255),
    SD_MEMO VARCHAR(1024),
    SD_DATE_BEGIN DATE,
    SD_DATE_END DATE,
    SD_DATE_CLOSED DATE,
    SD_ACC_MAIN VARCHAR(20),
    SD_ACC_ADVANCE VARCHAR(20),
    SD_ACC_EXPENSE VARCHAR(20),
    SD_USER_ID DECIMAL(18,0),
    SD_DATETIME TIMESTAMP,
    SD_REGION_NAME VARCHAR(100),
    SD_IS_INDIV_STR VARCHAR(50),
    SD_ONCE_ONLY_PAY_STR VARCHAR(50),
    SD_NEED_PATENT_STR VARCHAR(50),
    SD_NEED_SPOLIS_STR VARCHAR(50),
    SD_ONLY_WORKDAYS_STR VARCHAR(50),
    SD_CLIENT_NAME VARCHAR(100),
    SD_CLIENT_PASS_NO VARCHAR(30),
    SD_STATUS_STR VARCHAR(50),
    SD_STATUS INTEGER,
    SD_KIND_STR VARCHAR(100),
    SD_BRANCH_ID INTEGER,
    SD_BRANCH_NAME VARCHAR(100),
    SD_ONLY_WORKDAYS INTEGER,
    SK_SFOND_KOEF DECIMAL(18,7),
    SK_PDNAL_KOEF DECIMAL(18,7),
    SK_COMP_SFOND_KOEF DECIMAL(18,7),
    SK_SFOND_MIN_ZP DECIMAL(18,4),
    SSC_AMOUNT_DOGOVOR DECIMAL(18,4),
    SSC_OTHER_PLUS DECIMAL(18,4),
    SSC_TAX_CLIENT_SFOND DECIMAL(18,4),
    SSC_TAX_CLIENT_PDNAL DECIMAL(18,4),
    SSC_TAX_CLIENT_OTHER DECIMAL(18,4),
    SSC_TAX_COMPANY_SFOND DECIMAL(18,4),
    SSC_CLIENT_SALDO DECIMAL(18,4),
    SSC_TOTAL_DAYS DECIMAL(18,4),
    SSC_DISB_AMOUNT DECIMAL(18,4),
    SSC_SALDO_BEGIN DECIMAL(18,4),
    SSC_SALDO_END DECIMAL(18,4),
    SD_ACC_BALANCE VARCHAR(20),
    SD_CLIENT_INN VARCHAR(20),
    SD_BIRTH_DATE DATE,
    SD_SEX_STR VARCHAR(20),
    SD_COMMENT_SF VARCHAR(100),
    SD_SF_LGOTA DECIMAL(18,7),
    HR_REGION D_STR_100,
    GNI_CODE D_STR_100,
    RES_CARD_PROJECT D_STR_100)
AS
declare variable sd_is_indiv integer;
declare variable sd_once_only_pay integer;
declare variable sd_need_patent integer;
declare variable sd_need_spolis integer;
declare variable c_sex integer;
declare variable dd date;
begin
  for select sd_id, sd_dogovor_no, sd_from_reg_id, sd_from_c_id, sd_category, sd_memo, sd_is_indiv, sd_date_begin,
             sd_date_end, sd_date_closed, sd_once_only_pay, sd_acc_main, sd_acc_advance, sd_acc_expense, sd_need_patent,
             sd_need_spolis, sd_only_workdays, sd_user_id, sd_datetime, r1.reg_code || ' ' || r1.reg_name, c_name_fio,
             c_pass_no, sd_status, sk_name, r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name, sk_sfond_koef,
             sk_pdnal_koef, sk_comp_sfond_koef, r1.reg_sfond_min_zp, c_inn, c_birth_date, c_sex
      from servicedog, region r1, region_list(:rep_region_id, 1, -1, 1), clients, servicekind, region r2
      where r1.reg_unicode = sd_from_reg_id and
            r2.reg_unicode = r1.reg_head_office and
            (sd_from_reg_id = id_list) and
            c_id = sd_from_c_id and
            sd_kind = sk_id
      into :sd_id, :sd_dogovor_no, :sd_from_reg_id, :sd_from_c_id, :sd_category, :sd_memo, :sd_is_indiv, :sd_date_begin,
           :sd_date_end, :sd_date_closed, :sd_once_only_pay, :sd_acc_main, :sd_acc_advance, :sd_acc_expense,
           :sd_need_patent, :sd_need_spolis, :sd_only_workdays, :sd_user_id, :sd_datetime, :sd_region_name,
           :sd_client_name, :sd_client_pass_no, :sd_status, :sd_kind_str, :sd_branch_id, :sd_branch_name,
           :sk_sfond_koef, :sk_pdnal_koef, :sk_comp_sfond_koef, :sk_sfond_min_zp, :sd_client_inn, :sd_birth_date,
           :c_sex
  do
  begin
    hr_region = '';

    select rd.rd_name
    from region reg
    inner join refdata rd on rd.rd_id = reg.reg_hr_region
    where reg.reg_unicode = :sd_from_reg_id
    into :hr_region;

    gni_code = '';

    select first 1 rd.rd_str_id
    from client_gni_category cgi
    inner join refdata rd on rd.rd_id = cgi.cgc_from_category
    where :rep_date_end between cgi.cgc_date_begin and cgi.cgc_date_end
    and cgi.cgc_is_delete = 0
    and cgi.cgc_from_c_id = :sd_from_c_id
    order by cgc_id
    into :gni_code;

    if (sd_is_indiv = 0) then
      sd_is_indiv_str = 'Юр.лицо';
    else
      sd_is_indiv_str = 'Физ.лицо';
    if (sd_once_only_pay = 1) then
      sd_once_only_pay_str = 'Разовый';
    else
      sd_once_only_pay_str = 'Ежемесячный';
    if (sd_need_patent = 1) then
      sd_need_patent_str = 'Требуется';
    else
      sd_need_patent_str = 'Не требуется';
    if (sd_need_spolis = 1) then
      sd_need_spolis_str = 'Требуется';
    else
      sd_need_spolis_str = 'Не требуется';
    if (sd_only_workdays = 1) then
      sd_only_workdays_str = 'Почасовая';
    else
      sd_only_workdays_str = 'За месяц';
    if (sd_status = -1) then
      sd_status_str = 'Аннулирован';
    if (sd_status = 0) then
      sd_status_str = 'В процессе';
    if (sd_status = 1) then
      sd_status_str = 'Утвержден';
    if (sd_status = 2) then
      sd_status_str = 'Закрыт';
    if (sk_sfond_min_zp is null) then
      sk_sfond_min_zp = 0;
    if (sd_acc_main is not null) then
      sd_acc_balance = '';
    begin
      select al_from_bg_id
      from acc_list
      where al_id = :sd_acc_main
      into sd_acc_balance;
    end

    ssc_amount_dogovor = 0;
    ssc_other_plus = 0;
    ssc_tax_client_sfond = 0;
    ssc_tax_client_pdnal = 0;
    ssc_tax_client_other = 0;
    ssc_tax_company_sfond = 0;
    ssc_client_saldo = 0;
    ssc_total_days = 0;
    ssc_disb_amount = 0;
    ssc_saldo_begin = 0;
    ssc_saldo_end = 0;

    select sum(ssc_client_saldo)
    from servicecalc, operlist
    where ssc_from_sd_id = :sd_id and
          ol_id = ssc_from_ol_id and
          operlist.is_delete = 0 and
          ol_date < :rep_date_begin
    into :ssc_saldo_begin;
    if (ssc_saldo_begin is null) then
      ssc_saldo_begin = 0;

    select sum(ssc_amount_dogovor), sum(ssc_other_plus), sum(ssc_tax_client_sfond), sum(ssc_tax_client_pdnal),
           sum(ssc_tax_client_other), sum(ssc_tax_company_sfond), sum(ssc_client_saldo), sum(ssc_total_days),
           sum(ssc_disb_amount)
    from servicecalc, operlist
    where ssc_from_sd_id = :sd_id and
          ol_id = ssc_from_ol_id and
          operlist.is_delete = 0 and
          ol_date_oper_day between :rep_date_begin and :rep_date_end
    into :ssc_amount_dogovor, :ssc_other_plus, :ssc_tax_client_sfond, :ssc_tax_client_pdnal, :ssc_tax_client_other,
         :ssc_tax_company_sfond, :ssc_client_saldo, :ssc_total_days, :ssc_disb_amount;
    if (ssc_amount_dogovor is null) then
      ssc_amount_dogovor = 0;
    if (ssc_other_plus is null) then
      ssc_other_plus = 0;
    if (ssc_tax_client_sfond is null) then
      ssc_tax_client_sfond = 0;
    if (ssc_tax_client_pdnal is null) then
      ssc_tax_client_pdnal = 0;
    if (ssc_tax_client_other is null) then
      ssc_tax_client_other = 0;
    if (ssc_tax_company_sfond is null) then
      ssc_tax_company_sfond = 0;
    if (ssc_client_saldo is null) then
      ssc_client_saldo = 0;
    if (ssc_total_days is null) then
      ssc_total_days = 0;
    if (ssc_disb_amount is null) then
      ssc_disb_amount = 0;

    ssc_saldo_end = ssc_saldo_begin + ssc_client_saldo;

    select sum(sao_sf_lgota)
    from sa_oklad
    where sao_from_c_id = :sd_from_c_id
    into :sd_sf_lgota;
    if (sd_sf_lgota is null) then
      sd_sf_lgota = 0;
    if (c_sex = 0) then
      sd_sex_str = 'Женщина';
    else
      sd_sex_str = 'Мужчина';

    sd_comment_sf = '001 Стандарт';
    if (c_sex = 0) then
      dd = cast('01.01.1969' as date);
    else
      dd = cast('01.01.1964' as date);
    if (sd_birth_date < dd) then
      sd_comment_sf = '007 Муж.ранее 01.01.1964 и жен.ранее 01.01.1969';
    if (sd_sf_lgota > 0) then
      sd_comment_sf = '003 Работники пенсионеры';

    res_card_project = '';
    select sack_agreement_name
    from sa_cardno, sa_cardkind
    where sacn_from_c_id = :sd_from_c_id and
          sacn_date_begin <= :rep_date_end and
          sacn_from_sack_id = sack_id
    order by sacn_from_c_id desc, sacn_date_begin desc
    rows 1
    into res_card_project;

    suspend;
  end
end
