CREATE PROCEDURE HP_EDOC_WEB_TOKEN_CHECK(
    TRANS_ID D_STR_20)
RETURNS (
    RES_CODE D_SMALLINT,
    RES_TEXT D_STR_100)
AS
declare variable et_id d_id;
declare variable et_token d_str_100;
declare variable client_id bigint;
begin
  -- Проверка состояния токена, вызывается с сайта (часто)
  -- res_code:
  --  0 - Токен получен, необходимо прекратить повторные запросы
  --  1 - Токен еще не получен, запросы продолжаются
  --  2 - Неправильные параметры, необходимо прекратить повторные запросы

  res_code = 0;
  res_text = 'Данные карты успешно сохранены';

  if (trans_id is null) then
  begin
    res_code = 2;
    res_text = 'ID транзакции не передан';
  end

  select esm_object_id from extsysmain
  where esm_project = 'moneta_disb' and
  esm_trans_id = :trans_id
  rows 1
  into client_id;

  if (client_id is null) then
  begin
    res_code = 2;
    res_text = 'Транзакция не найдена';
  end

  if (res_code = 0) then
  begin
    et_token = null;
    et_id = null;

    select et_id, et_token
    from extsys_token
    where et_from_c_id = :client_id
    order by et_id desc
    rows 1
    into et_id, et_token;

    if (coalesce(et_id, 0) = 0) then
    begin
      res_code = 2;
      res_text = 'Ошибка при обработке запроса'; -- Токен не создавался
    end
    else
    if (coalesce(et_token, '') = '') then
    begin
      res_code = 1;
      res_text = 'Данные карты в процессе обработки'; -- Ждем ответа
    end
  end

  suspend;
end
