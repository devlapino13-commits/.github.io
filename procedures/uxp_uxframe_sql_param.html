CREATE PROCEDURE UXP$UXFRAME_SQL_PARAM(
    UFR_UUID D_UUID,
    NEED_PUBLIC_PARAM D_BOOL,
    UFR_UUID_CODE D_STR_CODE = null)
RETURNS (
    SQL_TEXT D_STR_LONG,
    SQL_PARAMS D_STR_LONG)
AS
declare variable uf_field_name d_str_code;
declare variable is_table d_bool;
declare variable ufp_parent d_uuid;
declare variable ufp_editor d_str_code;
declare variable is_param d_bool;
declare variable db_domain d_str_code;
declare variable field_list d_str_long;
declare variable enter_sym d_str_10;
declare variable us_code d_str_code;
declare variable ux_uuid d_uuid;
declare variable fr_uuid d_uuid;
declare variable ufp_public d_bool;
declare variable ufr_master_ufr_uuid d_uuid;
declare variable ufr_master_field d_str_code;
declare variable ufr_detail_field d_str_code;
declare variable tmp_domain d_str_code;
declare variable uf_select_name d_str_long;
declare variable uf_pk_field d_smallint;
begin
  -- Процедура строит искуственный SQL запрос из входящий параметров процедуры

  if (coalesce(ufr_uuid, '') = '') then
    ufr_uuid = uxf$uniq(ufr_uuid_code);

  if (need_public_param is null) then
    need_public_param = 0;

  select ufr_owner
  from ux$frame
  where ufr_uuid = :ufr_uuid
  into ux_uuid;

  enter_sym = ascii_char(13) || ascii_char(10);

  sql_text = '';
  sql_params = '';

  for select ufr_uuid
  from ux$frame
  where ufr_owner = :ux_uuid
  into fr_uuid
  do
  begin
    if (need_public_param = 0 and fr_uuid <> ufr_uuid) then
      Continue;

    for select uf_field_name, uf_select_name, us_is_table, uf_is_param, uf_domain, ufp_editor,
               ufp_parent, us_code, ufp_public,
               ufr_master_ufr_uuid, ufr_master_field, ufr_detail_field, uf_pk_field
    from ux$frparam, ux$frame, ux$field, ux$source
    where ufr_uuid = :fr_uuid and
          ufr_us_uuid = us_uuid and
          ufp_ufr_uuid = ufr_uuid and
          ufp_uf_uuid = uf_uuid and
          (uf_is_param = 1 or (ufp_parent is null and uf_pk_field <> 0 and us_is_table = 1))
    into uf_field_name, uf_select_name, is_table, is_param, db_domain, ufp_editor,
         ufp_parent, us_code, ufp_public,
         ufr_master_ufr_uuid, ufr_master_field, ufr_detail_field, uf_pk_field
    do
    begin
      if (need_public_param = 1 and ufp_public <> 1) then
        Continue;

      if (is_table = 0) then
      begin
        if (ufr_detail_field = uf_field_name) then
          Continue;

        /*
        if (exists(select * from ux$frame
        where ufr_owner = :ux_uuid and
              ufr_master_ufr_uuid = :fr_uuid and
              ufr_detail_field = :uf_field_name)) then
          Continue;
        */
      end
  
      if (db_domain starting 'rdb$') then
        exception e_default 'Field domain not defined: ' || us_code || '.' || uf_field_name || ' -> ' || db_domain;

      if (is_table = 0 and
          ufr_master_ufr_uuid is not null and
          uf_field_name = ufr_detail_field and
          is_param = 1) then
        uf_field_name = ufr_master_field;

      if (sql_text <> '') then
        sql_text = sql_text || ', ';
  
      if (ufp_parent is not null) then
      begin
        tmp_domain = '';
        select uf_field_name
        from ux$frparam, ux$field
        where ufp_uuid = :ufp_parent and
              ufp_uf_uuid = uf_uuid
        into tmp_domain;

        uf_field_name = uf_field_name || '$' || tmp_domain;
        uf_select_name = uf_select_name || '$' || tmp_domain;
      end
  
      if (sql_params <> '') then
        sql_params = sql_params || ', ';
      sql_params = sql_params || uf_field_name;







      field_list = '';
      if (ufp_editor in ('blob_edoc', 'blob_bls', 'blob')) then
      begin
        field_list = field_list || f_format('cast(null as %1) as %2', db_domain, trim(uf_field_name));
  
        field_list = field_list || ', ' || enter_sym;
        field_list = field_list || 'cast(null as d_str_8100) as ' || uf_field_name || '$json';
        field_list = field_list || ', ' || enter_sym;
        field_list = field_list || 'cast(''(пусто)'' as d_str_100) as ' || uf_field_name || '$text';
  
        if (ufp_editor = 'blob') then
        begin
          field_list = field_list || ', ' || enter_sym;
          field_list = field_list || 'cast(null as d_blob_binary) as ' || uf_field_name || '$blob';
          field_list = field_list || ', ' || enter_sym;
          field_list = field_list || 'cast(null as d_str_255) as ' || uf_field_name || '$file';
        end
  
        ufp_editor = 'blob_json';
      end else
      if (ufp_editor = 'blob_json') then
      begin
        field_list = field_list || 'cast(null as d_str_8100) as ' || uf_field_name || '$json';
        field_list = field_list || ', ' || enter_sym;
        field_list = field_list || 'cast(''(пусто)'' as d_str_100) as ' || uf_field_name || '$text';
      end
      else
      begin
        if (is_param = 0 and uf_pk_field <> 0) then
        begin
          if (db_domain in ('d_id', 'id_type') or uf_field_name containing '_ol_id') then
            db_domain = 'd_str_20';

          field_list = f_format('cast(null as %1) as %2', db_domain, trim(uf_field_name));
        end
        else
          field_list = uf_select_name;
      end
  
      sql_text = sql_text || field_list;
  
      if (is_param = 1 and ufp_editor in ('buttonedit', 'dialogedit')) then
      begin
        if (sql_text <> '') then
          sql_text = sql_text || ', ';
  
        sql_text = sql_text || f_format('cast(null as %1) as %2', 'D_STR_LONG', trim(uf_field_name) || '$text');
      end

      if (is_param = 1 and ufp_editor in ('curredit', 'currfixed')) then
      begin
        if (not exists(select * from ux$frparam, ux$field
                       where ufp_ufr_uuid = :fr_uuid and
                             ufp_uf_uuid = uf_uuid and
                             uf_is_param = 1 and
                             uf_field_name = (:uf_field_name || '$bond'))) then
        begin
          if (ufp_editor = 'currfixed') then
            exception e_default f_format('%1 - поле %2$bond отсутствует, field: %2', ufp_editor, uf_field_name);

          if (sql_text <> '') then
            sql_text = sql_text || ', ';
  
          sql_text = sql_text || 'cast(null as d_str_long) as ' || trim(uf_field_name) || '$bond';
        end
      end
    end
  end

  if (sql_text <> '') then
  begin
    sql_text = lower(sql_text);
    sql_text = 'select ' || sql_text || ' from rdb$database';
  end
  suspend;
end
