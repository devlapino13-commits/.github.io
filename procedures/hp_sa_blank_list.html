CREATE PROCEDURE HP_SA_BLANK_LIST(
    SAWT_ID D_INTEGER)
RETURNS (
    SAB_FROM_SAWT_ID D_INTEGER,
    SAB_FROM_BLN_ID D_INTEGER,
    SAB_WITH_MOUNT D_BOOL,
    SAB_WITHOUT_MOUNT D_BOOL,
    SAB_FOR_STAFF D_BOOL,
    SAB_FOR_FREELANCER D_BOOL,
    SAB_FOR_PARTTIME D_BOOL,
    RES_BLANK_NAME D_STR_255,
    RES_ACTIVE D_BOOL,
    RES_DISABLE_DATE D_DATE,
    RES_BLANK_LANG D_STR_10)
AS
begin
  for select sab_from_sawt_id, sab_from_bln_id,
             sab_with_mount, sab_without_mount, sab_for_staff, sab_for_freelancer, sab_for_parttime,
             bln_name, bln_lang
  from sa_blank, blankname
  where sab_from_sawt_id = :sawt_id and
        sab_from_bln_id = bln_id
  order by sab_from_sawt_id, sab_from_bln_id
  into :sab_from_sawt_id, :sab_from_bln_id,
       :sab_with_mount, :sab_without_mount, :sab_for_staff, :sab_for_freelancer, :sab_for_parttime,
       :res_blank_name, :res_blank_lang
  do
  begin
    res_active = 0;

    select max(blb_date_end)
    from blankblob
    where blb_from_bln_id = :sab_from_bln_id and
          (blb_blob is not null or blb_frx_blob is not null)
    into res_disable_date;

    if (res_disable_date >= current_date) then
    begin
      res_active = 1;
      res_disable_date = null;
    end else
    if (datediff(day, res_disable_date, current_date) > 30) then
      Continue;

    suspend;
  end
end
