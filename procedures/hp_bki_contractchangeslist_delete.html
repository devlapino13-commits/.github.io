CREATE PROCEDURE HP_BKI_CONTRACTCHANGESLIST_DELETE(
    BOD_ID D_ID)
AS
declare variable bod_add d_str_long;
declare variable jso d_integer;
declare variable status d_integer;
declare variable cml_id d_id;
declare variable k_id d_id;
declare variable cnt d_integer;
begin
  select bod_add, bod_object_id
  from bki_once_data
  where bod_id = :bod_id
  into bod_add, k_id;

  jso = json.parse(bod_add);
  status = json.obj_get_val(jso, 'status');
  if (status is null or status not in (0, 1)) then
    exception e_message 'Запись может быть удалена только со статусом Новый или Готово';

  cml_id = json.obj_get_val(jso, 'cmlId');
  select count(*)
  from bki_once_data
  where bod_kind = 3
    and bod_object_id = :k_id
    and json.obj_get_val(json.parse(bod_add), 'cmlId') = :cml_id
  into :cnt;

  if (cnt <= 1) then
    exception e_message 'Нельзя удалить — это единственная запись по данному запросу';

  delete from bki_once_data
  where bod_id = :bod_id;
end
