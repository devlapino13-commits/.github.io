CREATE PROCEDURE AAREP_CURRLIST_USED
RETURNS (
    CU_ID VARCHAR(3),
    CU_SYMBOL VARCHAR(10),
    CU_NAME VARCHAR(50))
AS
begin
  cu_id = '---';
  cu_name = '--- По всем валютам';
  suspend;
  cu_id = '000';
  cu_name = '000 Сводный (эквив. валют)';
  suspend;

  for select cu_id, cu_symbol, cu_id || ' ' || cu_symbol || ' ' || cu_name
      from currlist, accposit
      where cu_id = ap_from_cl_id
      into :cu_id, :cu_symbol, :cu_name
  do
  begin
    suspend;
  end
end
