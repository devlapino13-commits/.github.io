CREATE PROCEDURE HP_GET_UI_UXACTION(
    UA_CODE D_STR_100,
    UN_ID D_STR_CODE)
RETURNS (
    UA_ACTION_KIND D_STR_CODE,
    UA_DISABLE_ON_EDIT D_BOOL,
    UA_READONLY D_BOOL,
    UA_SAFE_MODE D_BOOL,
    UXA_CODE D_STR_100,
    UX_CAPTION D_STR_100,
    UX_CODE D_STR_CODE,
    UX_KIND D_INTEGER,
    UX_WIDTH D_INTEGER,
    UX_HEIGHT D_INTEGER,
    UX_ROOT_CODE D_STR_100,
    UX_DIALOGEDIT_UA_CODE D_STR_100,
    UX_DIALOGEDIT_FIELD D_STR_CODE,
    UX_DIALOGEDIT_IS_PARAM D_BOOL,
    UX_DIALOGEDIT_FIND_FIELD D_STR_CODE,
    UX_SOURCE_ROWS D_SMALLINT,
    UX_SOURCE_REFRESH D_BOOL,
    UX_SOURCE_LOCATE_FIELDS D_STR_255,
    CMD_COMMIT_EXPR D_STR_255,
    CMD_MESSAGE_FIELD D_STR_CODE,
    UX_FORM_INVISIBLE D_BOOL,
    UXA_DEFAULT_UA_CODE D_STR_100)
AS
declare variable mainframe_code d_str_code;
declare variable dialog_uxa_code d_str_100;
declare variable ux_open_kind d_smallint; /* Нужно ли открывать окно перед выполнением команды */
declare variable ufp_code d_str_100;
declare variable ufp_origin_name d_str_code;
declare variable unlinked_fields d_str_255;
declare variable ufr_show_kind d_smallint;
declare variable ufr_can_update d_bool;
declare variable ufr_can_insert d_bool;
declare variable ufr_can_delete d_bool;
declare variable ufr_code d_str_code;
declare variable pk_count d_integer;
declare variable ufr_name d_str_100;
declare variable edit_frames d_bool;
declare variable is_linked d_bool;
declare variable uxa_kind d_smallint;
declare variable ua_usr_id_field d_str_code;
begin
  if (trim(coalesce(ua_code, '')) = '') then
    exception e_default 'Emptry code';

  ux_form_invisible = 0;
  ux_source_rows = 0; -- если не команда, а Update или Delete SQL тоже поправить
  ux_source_refresh = 0; -- если не команда, а Update или Delete SQL тоже поправить
  cmd_commit_expr = '';
  cmd_message_field = '';
  ux_source_locate_fields = '';

  select ux_code, ux_kind, uxa_code, ua_name,
         ua_disable_on_edit, ua_readonly, ua_safe_mode, uxa_kind, ua_usr_id_field
  from hp_get_uxalias(:ua_code)
  into :ux_code, :ux_kind, :uxa_code, :ux_caption,
       :ua_disable_on_edit, :ua_readonly, :ua_safe_mode, uxa_kind, ua_usr_id_field;

  if (row_count = 0) then
    exception e_message f_format('UxAction not found: %1', ua_code);

  ua_action_kind = trim(case
    when ua_usr_id_field <> '' then 'search'
    when uxa_kind = 2 then 'insert'
    when uxa_kind = 4 then 'command'
    else 'open'
  end);

  if (ua_action_kind is null) then
    exception e_default f_format('Unknown action_kind, ua_code: %1', ua_code);

  uxa_default_ua_code = '';

  if (ua_action_kind in ('open', 'search')) then
  begin
    select uaa_ua_code
    from uxaliasaction
    where uaa_uxa_code = :uxa_code and
          uaa_is_default = 1
    rows 1
    into uxa_default_ua_code;
  end

  select ucf_ufr_code, ufr_can_insert
  from uxcodeframe, uxframe
  where ucf_ux_code = :ux_code and
        ucf_is_mainframe = 1 and
        ucf_ufr_code = ufr_code
  into mainframe_code, ufr_can_insert;

  if (uxa_kind = 4) then
  begin
    select cmd_open_kind, cmd_source_rows, cmd_source_refresh, cmd_commit_expr, cmd_message_field
    from uxcommand
    where cmd_code = :ux_code
    into :ux_open_kind, :ux_source_rows, :ux_source_refresh, :cmd_commit_expr, :cmd_message_field;

    if (ux_open_kind in (0, 2)) then
    begin
      ux_form_invisible = 1;
      unlinked_fields = '';
      for select ufp_code, ufp_origin_name
      from uxfrparam
      where ufp_ufr_code = :mainframe_code and
            ufp_is_param = 1 and
            ufp_is_delete = 0 and
            ufp_in_use = 1 and
            ufp_editor <> 'dialogedit'
      into ufp_code, ufp_origin_name
      do
      begin
        is_linked = 0;
        select iif(res_state <> 0 and (res_source_field <> '' or res_def_value <> ''), 1, 0)
        from hp_uxparam_final_value(:ufp_code, :uxa_code, :ua_code)
        into :is_linked;

        if (is_linked = 0) then
        begin
          if (not exists(select * from uxactionparam
          where uap_ua_code = :ua_code and
                uap_kind = 0 and
                uap_link_ufp_code = :ufp_code)) then
          begin
            if (char_length(unlinked_fields || ', ' || ufp_origin_name) <= 255) then
            begin
              if (unlinked_fields <> '') then
                unlinked_fields = unlinked_fields || ', ';
              unlinked_fields = unlinked_fields || ufp_origin_name;
            end

            ux_form_invisible = 0;
          end
        end
      end

      if (ux_open_kind = 2 and ux_form_invisible = 0) then
        exception e_message f_format('Окно команды %1 невидимое.', upper(ux_code)) || ascii_char(13) ||
                                     'Имеются параметры без связи:' || ascii_char(13) ||
                                     upper(unlinked_fields);
    end
  end else
  begin
    ux_source_rows = 0;
    ux_source_refresh = 0;

    if (ua_action_kind = 'insert') then
    begin
      ux_source_refresh = 1;

      if (ufr_can_insert = 0) then
        exception e_message f_format('Фрейм не поддерживает создание записи, fr_code: %1', mainframe_code);
    end

    if (exists(select * from uxactionparam
    where uap_ua_code = :ua_code and
          uap_from_ufp_code is not null and
          uap_kind = 0)) then -- Если есть входящие параметры
      ux_source_rows = 1;
  end

  edit_frames = 0;
  for select ucf_ufr_code, ux_name, ufr_show_kind, ufr_can_update, ufr_can_insert, ufr_can_delete
  from uxcodeframe, uxframe, uxfrdata, uxcode
  where ucf_ux_code = :ux_code and
        ucf_ufr_code = ufr_code and
        ufr_code = ufd_ufr_code and
        ufd_kind = 0 and
        ufr_code = ux_code
  into ufr_code, ufr_name, ufr_show_kind, ufr_can_update, ufr_can_insert, ufr_can_delete
  do
  begin
    if (exists(select * from uxfritem
    where ufi_ux_code = :ufr_code and
          ufi_control = 'html' and
          ufi_parent_code = :ufr_code || '#g')) then
      exception e_message 'HTML-просмотр не может быть внутри таблицы.';

    unlinked_fields = '';

    if (ufr_can_update = 1 or ufr_can_insert = 1 or ufr_can_delete = 1) then
    begin
      edit_frames = 1;
      pk_count = 0;
      for select ufp_code, ufp_origin_name
      from uxfrparam
      where ufp_ufr_code = :ufr_code and
            ufp_pk_field = 1 and
            ufp_is_delete = 0 and
            ufp_in_use = 1
      into ufp_code, ufp_origin_name
      do
      begin
        pk_count = pk_count + 1;

        if (ufr_can_update = 1) then
        begin
          if (not exists(select * from uxframesql
          where ufs_ufr_code = :ufr_code and
                ufs_sql_kind = 1 and
                ufs_ufp_code = :ufp_code)) then
          exception e_message f_format('Фрейм %1 (%2), ключевое поле %3 не прикреплен к Update SQL полям.',  upper(ufr_code), ufr_name, upper(ufp_origin_name));
        end

        if (ufr_can_insert = 1) then
        begin
          if (not exists(select * from uxframesql
          where ufs_ufr_code = :ufr_code and
                ufs_sql_kind = 2 and
                ufs_ufp_code = :ufp_code)) then
          exception e_message f_format('Фрейм %1 (%2), ключевое поле %3 не прикреплен к Insert SQL полям.',  upper(ufr_code), ufr_name, upper(ufp_origin_name));
        end

        if (ufr_can_delete = 1) then
        begin
          if (not exists(select * from uxframesql
          where ufs_ufr_code = :ufr_code and
                ufs_sql_kind = 3 and
                ufs_ufp_code = :ufp_code)) then
          exception e_message f_format('Фрейм %1 (%2), ключевое поле %3 не прикреплен к Delete SQL полям.',  upper(ufr_code), ufr_name, upper(ufp_origin_name));
        end
      end
      if (pk_count = 0) then
        exception e_message f_format('Фрейм %1 (%2) обозначен', upper(ufr_code), ufr_name) || ascii_char(13) ||
                                     'как модифицируемый, но ключевые поля не указаны';
    end

    -- если фрейм однострочный, и не связан как дочернее,
    -- то проверить наличие привязки параметров по ключевым полям
    if (ufr_show_kind = 1 and ua_action_kind <> 'insert') then
    begin
      for select ufp_code, ufp_origin_name
      from uxfrparam
      where ufp_ufr_code = :ufr_code and
            ufp_is_param = 0 and
            ufp_pk_field <> 0 and
            ufp_is_delete = 0
      into ufp_code, ufp_origin_name
      do
      begin
        if (not exists(select * from uxactionparam
        where uap_ua_code = :ua_code and
              uap_kind = 0 and
              uap_link_ufp_code = :ufp_code and
              uap_from_ufp_code is not null)) then
        begin
          if (char_length(unlinked_fields || ', ' || ufp_origin_name) <= 255) then
          begin
            if (unlinked_fields <> '') then
              unlinked_fields = unlinked_fields || ', ';
            unlinked_fields = unlinked_fields || ufp_origin_name;
          end
        end
      end
    end

    if (unlinked_fields <> '') then
      exception e_message f_format('Фрейм %1 (%2)', upper(ufr_code), ufr_name) || ascii_char(13) ||
                                   'является однострочным. ключевые поля без связи:' || ascii_char(13) ||
                                   upper(unlinked_fields);
  end

  if (edit_frames = 0 and ua_action_kind <> 'command') then
    ua_readonly = 1;

  -- Вызывается как поисковик, превращаем в безопасный режим
  if (ua_readonly = 0 and ua_action_kind = 'search' and ua_safe_mode = 0) then
    ua_safe_mode = 1;

  select list(fp.ufp_origin_name || '=' || lp.ufp_origin_name)
  from uxactionparam, uxfrparam as lp, uxfrparam as fp
  where uap_ua_code = :ua_code and
        uap_from_ufp_code = fp.ufp_code and
        uap_link_ufp_code = lp.ufp_code and
        uap_kind = 2
  into ux_source_locate_fields;

  if (ux_source_locate_fields is null) then
    ux_source_locate_fields = '';

  ux_width = 0;
  ux_height = 0;

  select ufi_code, ufi_width, ufi_height
  from uxfritem
  where ufi_ux_code = :ux_code and
        ufi_parent_code is null
  into ux_root_code, :ux_width, :ux_height;

  ux_dialogedit_is_param = 0;
  ux_dialogedit_ua_code = '';
  ux_dialogedit_field = '';

  -- Диалоговое окно НЕ открывается для полей если это открыть или действие.
  -- Открыааем если Insert и поле, либо это параметр
  -- Если их несколько, берем только первую

  select ua_uxa_code, usr_id_field, ua_usr_id_field, ufp_is_param
  from uxaction, uxfrparam, uxsearch
  where ua_ux_code = :mainframe_code and
        ufp_ufr_code = ua_ux_code and
        ufp_origin_name = ua_usr_id_field and
        ufp_editor = 'dialogedit' and
        ua_uxa_code = usr_code
  order by ufp_is_param desc, ufp_index
  rows 1
  into dialog_uxa_code, ux_dialogedit_find_field, ux_dialogedit_field, ux_dialogedit_is_param;

  if (dialog_uxa_code is not null) then
  begin
    if (ua_action_kind = 'insert' or ux_dialogedit_is_param = 1) then
    begin
      select ua_code
      from uxaction
      where ua_ux_code = :mainframe_code and
            ua_usr_id_field = :ux_dialogedit_field and
            ua_uxa_code = :dialog_uxa_code
      into ux_dialogedit_ua_code;
    end else
    begin
      ux_dialogedit_is_param = 0;
      ux_dialogedit_ua_code = '';
      ux_dialogedit_field = '';
    end
  end

  suspend;
end
