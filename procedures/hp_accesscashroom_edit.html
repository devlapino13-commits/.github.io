CREATE PROCEDURE HP_ACCESSCASHROOM_EDIT(
    ID D_INTEGER)
RETURNS (
    ACR_ID D_INTEGER,
    ACR_DATE D_DATE,
    ACR_OFFICE_ID D_INTEGER,
    ACR_REG_ID D_INTEGER,
    ACR_START_DATE D_DATE,
    ACR_END_DATE D_DATE,
    ACR_BASIS_ID D_INTEGER,
    ACR_EDL_ID D_ID,
    ACR_STATUS D_SMALLINT,
    STATUS D_STR_CODE,
    ACR_ISSUER_ID D_ID,
    ISSUER_NAME D_STR_100,
    ACR_CASHIER_ID D_ID,
    CASHIER_NAME D_STR_100,
    ACR_PERSON_ID D_ID,
    PERSON_NAME D_STR_100,
    ACR_DIRECTOR_ID D_ID,
    DIRECTOR_NAME D_STR_100,
    ACR_ISSUER_DATESIGN D_DATETIME,
    ACR_CASHIER_DATESIGN D_DATETIME,
    ACR_PERSON_DATESIGN D_DATETIME,
    ACR_DIRECTOR_DATESIGN D_DATETIME)
AS
begin
  for
    select
      acr_id,
      acr_date,
      acr_office_id,
      acr_reg_id,
      acr_start_date,
      acr_end_date,
      acr_basis_id,
      acr_edl_id,
      acr_status,
      case acr_status
        when -1 then 'аннулирован'
        when 0 then 'создан'
        when 1 then 'завизирован'
        when 2 then 'частично подписан'
        when 3 then 'подписан'
      end,
      acr_issuer_id,
      cl1.c_name_fio,
      acr_cashier_id,
      cl2.c_name_fio,
      acr_person_id,
      cl3.c_name_fio,
      acr_director_id,
      cl4.c_name_fio,
      acr_issuer_datesign,
      acr_cashier_datesign,
      acr_person_datesign,
      acr_director_datesign
    from accesscashroom
      inner join clients cl1
        on cl1.c_id = acr_issuer_id
      inner join clients cl2
        on cl2.c_id = acr_cashier_id
      inner join clients cl3
        on cl3.c_id = acr_person_id
      inner join clients cl4
        on cl4.c_id = acr_director_id
    where acr_id = :id
  into
    :acr_id,
    :acr_date,
    :acr_office_id,
    :acr_reg_id,
    :acr_start_date,
    :acr_end_date,
    :acr_basis_id,
    :acr_edl_id,
    :acr_status,
    :status,
    :acr_issuer_id,
    :issuer_name,
    :acr_cashier_id,
    :cashier_name,
    :acr_person_id,
    :person_name,
    :acr_director_id,
    :director_name,
    :acr_issuer_datesign,
    :acr_cashier_datesign,
    :acr_person_datesign,
    :acr_director_datesign
  do
  begin
    if (acr_edl_id > 0) then
    begin
      if (acr_cashier_datesign is null) then
        acr_cashier_datesign = (
          select eds_status_datetime
          from edoc_sign
          where eds_from_edl_id = :acr_edl_id
            and eds_from_c_id = :acr_cashier_id
            and eds_status = 1);
      if (acr_person_datesign is null) then
        acr_person_datesign = (
          select eds_status_datetime
          from edoc_sign
          where eds_from_edl_id = :acr_edl_id
            and eds_from_c_id = :acr_person_id
            and eds_status = 1);
      if (acr_director_datesign is null) then
        acr_director_datesign = (
          select eds_status_datetime
          from edoc_sign
          where eds_from_edl_id = :acr_edl_id
            and eds_from_c_id = :acr_director_id
            and eds_status = 1);
    end
    suspend;
  end
end
