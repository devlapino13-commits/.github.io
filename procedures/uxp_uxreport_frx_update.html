CREATE PROCEDURE UXP$UXREPORT_FRX_UPDATE(
    URT_UUID D_UUID,
    FRX_BLOB D_BLOB_BINARY,
    FRX_JSON_DATA D_BLOB_TEXT)
AS
declare variable datasets d_str_long;
declare variable ds_count d_integer;
declare variable idx d_integer;
declare variable ds_name d_str_code;
declare variable ds_sql d_str_long;
declare variable ds_fields d_str_8100;
declare variable ufr_uuid d_uuid;
declare variable var_count d_integer;
declare variable var_code d_str_code;
declare variable is_mainframe d_bool;
declare variable mainframe_uuid d_uuid;
declare variable us_uuid d_uuid;
begin
  if (frx_blob is not null) then
  begin
    update ux$report
    set urt_fastrep = :frx_blob
    where urt_uuid = :urt_uuid;
  end

  if (coalesce(frx_json_data, '') = '') then
    Exit;

  ds_count = f_json_blob_get_count(frx_json_data, 'frx');

  if (ds_count = -1) then
    exit;

  datasets = '';
  idx = -1;
  while (idx < ds_count - 1) do
  begin
    idx = idx + 1;
    ds_name = f_json_blob_get_value(frx_json_data, 'frx[' || idx || '].name', '');
    datasets = datasets || '[' || ds_name || ']';
  end

  delete from ux$frame
  where ufr_owner = :urt_uuid and
        (ufr_code = '' or :datasets not containing '[' || ufr_code || ']');

  idx = -1;
  while (idx < ds_count - 1) do
  begin
    idx = idx + 1;
    ds_name = f_json_blob_get_value(frx_json_data, 'frx[' || idx || '].name', '');
    ds_sql = f_json_blob_get_value(frx_json_data, 'frx[' || idx || '].data.sql', '');
    ds_fields = f_json_blob_get_value(frx_json_data, 'frx[' || idx || '].data.fields', '');

    if (ds_fields <> '') then
      ds_fields = '{"fields":' || ds_fields || '}';

    is_mainframe = 0;
    ufr_uuid = null;
    select ufr_uuid, ufr_is_mainframe
    from ux$frame
    where ufr_owner = :urt_uuid and
          ufr_code = :ds_name
    into ufr_uuid, is_mainframe;

    if (ufr_uuid is null) then
    begin
      insert into ux$frame (ufr_owner, ufr_code, ufr_name, ufr_is_grid)
      values (:urt_uuid, :ds_name, :ds_name, 1)
      returning ufr_uuid, ufr_is_mainframe
      into :ufr_uuid, :is_mainframe;
    end

    if (is_mainframe = 1) then
      mainframe_uuid = ufr_uuid;

    execute procedure uxp$uxframe_update_sql(:ufr_uuid, :ds_sql, :ds_fields, 1);
  end

  if (mainframe_uuid is null) then
    exception e_default 'Mainframe not found';

  select ufr_us_uuid
  from ux$frame
  where ufr_uuid = :mainframe_uuid
  into us_uuid;

  update ux$field
  set uf_state = -2
  where uf_owner = :us_uuid and
        uf_type > 0;

  var_count = f_json_blob_get_count(frx_json_data, 'vars');

  idx = -1;
  while (idx < var_count - 1) do
  begin
    idx = idx + 1;
    var_code = f_json_blob_get_value(frx_json_data, 'vars[' || idx || ']', '');
    if (var_code <> '') then
    begin
      update ux$field
      set uf_state = 0
      where uf_field_name = :var_code and
            uf_owner = :us_uuid;

      if (row_count = 0) then
      begin
        insert into ux$field (uf_owner, uf_field_name, uf_is_param, uf_type, uf_domain, uf_data_type)
        values (:us_uuid, :var_code, 1, 1, 'd_str_255', 'string');
      end
    end
  end

  delete from ux$field
  where uf_owner = :us_uuid and
        uf_state = -2;
end
