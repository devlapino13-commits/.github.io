CREATE PROCEDURE AAREP_GET_LO_TEAM(
    REG_ID INTEGER,
    C_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    RES_TEAM_ID INTEGER,
    RES_TEAM_NAME VARCHAR(255),
    RES_TEAM_LEADER INTEGER,
    RES_TEAM_DATE DATE,
    RES_TEAM_REG_NAME VARCHAR(255))
AS
declare variable reg_name varchar(100);
begin
  if (reg_id is null) then
    reg_id = 0;

  reg_name = 'xxxx';
  if (reg_id <> 0) then
  begin
    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into :reg_name;
  end

  res_team_id = 0;
  res_team_date = null;
  res_team_name = 'Без команды';

-- Получаем рабочую команду сотрудника
  select res_rd_id, ovs_value, ovs_add_value, ovs_date
  from hp_objectprops_get(0, :c_id, -1, 'objectprops.staffteam', :rep_date)
  into :res_team_id, res_team_name, res_team_leader, res_team_date;

  if (not(exists(select *
                 from refdata
                 where rd_id = :res_team_id and
                       rd_is_active = 1))) then
    res_team_id = 0;

  if (res_team_id is null) then
    res_team_id = 0;

  if (res_team_id = 0) then
  begin
    res_team_name = 'Без команды';
    res_team_leader = 0;
  end
  res_team_reg_name = res_team_name;
  suspend;

end
