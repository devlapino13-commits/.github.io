CREATE PROCEDURE HP_ANALITICA_CREATE_PERIOD(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_NEED_CLIENT_DETAIL INTEGER,
    REP_INCLUDE_CLOSED DATE,
    REP_EXCLUDE_ACTIVE DATE,
    REP_DELINQ_DAYS INTEGER,
    REP_LO_ID DECIMAL(18,0),
    REP_KIND VARCHAR(100))
RETURNS (
    RES_PERIOD_ID INTEGER)
AS
declare variable reg_id integer;
declare variable reg_branch_name varchar(100);
declare variable reg_okrug_name varchar(100);
declare variable reg_oblast_name varchar(100);
declare variable rd_id integer;
declare variable lo_id decimal(18,0);
declare variable res_team_name varchar(255);
declare variable res_team_id integer;
declare variable c_stage integer;
declare variable ps_date_begin date;
begin
-- Основная процедура, заполняюшая данные за период для приготовления отчета о портфеле по новой схеме
-- Схема работы:
-- на каждый период запускается цикл 1-3. Далее после завершения запускается п.4
-- 1. hp_analitica_create_period
-- 2.    hp_analitica_fill_temptable
-- 3.       hp_analitica_get_list
-- 4. hp_analitica_report

  if (coalesce(rep_kind, '') = '') then
    rep_kind = 'standart';

  if (rep_need_client_detail is null) then
    rep_need_client_detail = 0;
  if (rep_delinq_days is null) then
    rep_delinq_days = 30;
  if (rep_lo_id is null) then
    rep_lo_id = 0; -- Если код СВЗ передан, отчет будет готовиться только по этому сотруднику
  if (rep_region_id = 0 and
      rep_lo_id = 0) then
    exit;

  if (rep_lo_id <> 0) then -- Если передан ID СВЗ, то отчет будет готовиться только по нему
  begin
    rep_region_id = 0;
    select ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :rep_lo_id and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          ps_status = 1 and
          ps_is_working = 1
    into :rep_region_id;
    if (coalesce(rep_region_id, 0) = 0) then
      exit;
  end

-- Готовим данные если только параметры были изменены
  delete from temp_analitica_period;
  delete from temp_analitica_regions;

  insert into temp_analitica_regions (reg_id, result_reg_id)
  with tab
  as (select id_list, 0 as result_reg_id
      from region_list(1000000, 1, 3, 1)
      union
      select id_list, id_list as result_reg_id
      from region_list(:rep_region_id, 1, 3, 1))
  select id_list, max(result_reg_id)
  from tab
  group by id_list;

  for select reg_unicode
      from region
      where reg_status = 3
      into :reg_id
  do
  begin
    reg_branch_name = '';
    reg_oblast_name = '';
    reg_okrug_name = '';
    select aa_curr_branch_name, aa_office_stat_name, aa_curr_level3_name
    from aarep_get_office_categ(:reg_id, :rep_date_end)
    into :reg_branch_name, reg_oblast_name, reg_okrug_name;

    update temp_analitica_regions
    set result_branch_name = :reg_branch_name,
        result_okrug_name = :reg_okrug_name,
        result_oblast_name = :reg_oblast_name
    where (reg_id = :reg_id);
  end

  insert into temp_analitica_period (ap_date_begin, ap_date_end, ap_region_id, ap_orderno, ap_rep_kind)
  values (:rep_date_begin, :rep_date_end, :rep_region_id, 1, :rep_kind)
  returning (ap_id)
  into :res_period_id;

  execute procedure hp_analitica_fill_temptable(:rep_region_id, :rep_date_begin, :rep_date_end, :res_period_id,
      :rep_need_client_detail, :rep_include_closed, :rep_exclude_active, :rep_delinq_days, :rep_lo_id, :rep_kind);

  select rd_id
  from refdata
  where rd_code = 'ObjectProps.StaffTeam'
  into :rd_id;

  for select tal_lo_id
      from temp_analitica_lo
      where tal_team_name = '' and
            tal_from_ap_id = :res_period_id
      into :lo_id
  do
  begin
    if (rd_id is not null) then
    begin
      res_team_name = '';
      select res_team_id, res_team_name
      from aarep_get_lo_team(0, :lo_id, :rep_date_end)
      into res_team_id, :res_team_name;

      update temp_analitica_lo
      set tal_team_id = :res_team_id,
          tal_team_name = :res_team_name
      where tal_from_ap_id = :res_period_id and
            tal_lo_id = :lo_id;
    end

    c_stage = 0;
    ps_date_begin = null;
    select min(ps_date_begin)
    from hr_pstatus, joblist, jobkind
    where ps_from_c_id = :lo_id and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')
    into :ps_date_begin;

    if (ps_date_begin is not null) then
    begin
      c_stage = datediff(month, ps_date_begin, (rep_date_end + 1));
      if (extract(day from ps_date_begin) > extract(day from (rep_date_end + 1)) and
          c_stage > 0) then
        c_stage = c_stage - 1;
    end
    else
      c_stage = 0;

    update temp_analitica_lo
    set tal_lo_staj = right('000' || :c_stage, 3)
    where tal_from_ap_id = :res_period_id and
          tal_lo_id = :lo_id;
  end

  suspend;
end
