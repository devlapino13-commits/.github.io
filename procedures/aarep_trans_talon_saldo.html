CREATE PROCEDURE AAREP_TRANS_TALON_SALDO(
    REP_TRANS_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    PEO_BILL_ID DECIMAL(18,0),
    PEO_BILL_LINE_NO INTEGER,
    PEO_FROM_C_ID DECIMAL(18,0),
    PEO_COST DECIMAL(18,4),
    PEO_FROM_PEN_ID INTEGER,
    PEO_LITER DECIMAL(18,4),
    PEO_SUMMA DECIMAL(18,4),
    PEO_OPER_DATE DATE,
    PEO_FROM_REG_ID INTEGER,
    PEO_MATOTVET_FIO VARCHAR(100),
    PEO_PETROL_NAME VARCHAR(100),
    PEO_ACC_BALANCE VARCHAR(20),
    PEO_ACC_NUMBER INTEGER,
    PEO_LITER_IN_ONE DECIMAL(18,4),
    PEO_TRANS_ID INTEGER,
    PEO_BRANCH_ID INTEGER,
    PEO_BRANCH_NAME VARCHAR(100))
AS
declare variable st_staff integer;
declare variable st_client integer;
declare variable db_id integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  :db_id = f_get_db_id();

  for select peo_bill_id, peo_bill_line_no, peo_from_c_id, peo_cost, peo_from_pen_id, sum(peo_trans_liter),
             sum(peo_trans_summa), peo_oper_date, peo_from_reg_id, c_name_fio, pen_name, pen_acc_balance,
             pen_acc_number, pen_liter_in_one, r1.reg_head_office, r2.reg_code || ' ' || r2.reg_name

      from petopers, operlist, clients, petname, region r1, region r2
      where peo_from_pet_id = :rep_trans_id and
            peo_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            ((ol_date <= :rep_date) or
            ((peo_oper_type in (3, 7, 8, 9)) and
            ((peo_liter < 0) or
            (peo_trans_liter < 0)))) and
            peo_from_c_id = c_id and
            peo_from_pen_id = pen_id and
            ol_from_reg_id = r1.reg_unicode and
            r1.reg_head_office = r2.reg_unicode
      group by peo_bill_id, peo_bill_line_no, peo_from_c_id, peo_cost, peo_from_pen_id, peo_oper_date, peo_from_reg_id, c_name_fio, pen_name, pen_acc_balance, pen_acc_number, pen_liter_in_one, r1.reg_head_office, r2.reg_code, r2.reg_name
      having sum(peo_trans_liter) <> 0 or sum(peo_trans_summa) <> 0
      order by peo_oper_date, peo_bill_id, peo_bill_line_no
      into :peo_bill_id, :peo_bill_line_no, :peo_from_c_id, :peo_cost, :peo_from_pen_id, :peo_liter, :peo_summa,
           :peo_oper_date, :peo_from_reg_id, :peo_matotvet_fio, :peo_petrol_name, :peo_acc_balance, :peo_acc_number,
           :peo_liter_in_one, :peo_branch_id, :peo_branch_name
  do
  begin
    if (db_id = 94401) then
      peo_branch_id = peo_from_reg_id;
    peo_trans_id = :rep_trans_id;
    suspend;
  end
end
