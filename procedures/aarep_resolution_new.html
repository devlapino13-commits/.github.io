CREATE PROCEDURE AAREP_RESOLUTION_NEW(
    LINE_B D_INTEGER,
    LINE_END D_INTEGER)
RETURNS (
    RESOLUTION_OLD D_STR_8100,
    RESOLUTION D_STR_8100,
    STAGE_CODE VARCHAR(50),
    SUM_OLD D_CURRENCY,
    SUM_NEW D_CURRENCY,
    SLC_ID BIGINT)
AS
declare variable c_id d_id;
declare variable c_name_fio d_str_255;
declare variable c_birth_date_str d_date;
declare variable c_pass_no d_str_100;
declare variable address_legal d_str_long;
declare variable debt_summ decimal(18,2);
declare variable k_kontract_no d_str_20;
declare variable k_disb_date_str d_date;
declare variable gp_summ decimal(18,2);
declare variable smr_id d_id;
declare variable k_id d_id;
declare variable cl_count d_integer;
declare variable sul_object_kind d_integer;
declare variable is_ok d_integer;
declare variable stage_id d_integer;
declare variable status d_integer;
declare variable sul_from_smr_id d_id;
declare variable lawsuit_no d_str_100;
declare variable decision_bls_id d_id;
declare variable exdoc_bls_id d_id;
begin

  for select slc_id, kc_from_c_id, slc_sum_os + slc_sum_perc, slc_from_smr_id, slc_resolution ,slc_sum_os + slc_sum_perc+ slc_sum_fine,slc_sum_os + slc_sum_perc
      from sud_loanclient, knclient
      where --slc_from_smr_id = :smr_id
            kc_id = slc_from_kc_id --and
           -- slc_resolution is not null-- and
           -- slc_id >= :line_b and
           -- slc_id <= :line_end
            --and kc_id=2206726
            order by SLC_SUM_DATE desc
      into slc_id, c_id, debt_summ, smr_id, resolution_old ,sum_old,sum_new
  do
  begin
    stage_id=null;
    status=null;
    sul_from_smr_id=null;

    ------------------------------------------------------------------------
    select sul_object_kind, sul_status, sul_from_smr_id
    from sud_mainlog
    where sul_from_smr_id = :smr_id
    order by sul_datetime desc
    rows 1
    into :stage_id, :status, :sul_from_smr_id;

    decision_bls_id = null;
    lawsuit_no = null;
    select sml_decision_bls_id, sml_lawsuit_no,sml_exdoc_bls_id
    from sud_mainlist
    where sml_id = :sul_from_smr_id
    into :decision_bls_id, lawsuit_no,:exdoc_bls_id;

    stage_code=null;
    is_ok = 0;
    stage_code = case stage_id
                   when f_objkind('sud_mainreq', 'id') then 'R'
                   when f_objkind('sud_maindocs', 'id') then 'D'
                   when f_objkind('sud_mainpay', 'id') then 'P'
                   when f_objkind('sud_mainlist', 'id') then 'J'
                   when f_objkind('sud_maincollect', 'id') then 'JC'
                   else 'error'
                 end;

    if (stage_code = 'error') then
      exception e_default 'Unknown stage_id = ' || coalesce(stage_id, '<NULL>');

    if (stage_code in ('R', 'D', 'P')) then
      is_ok = 1;

    if (stage_code = 'J') then
    begin
      stage_code = case
                     when status = 0 then '|J1|'
                     when status = 2 then '|J5|'
                     when status = -2 then '|J6|'
                     when decision_bls_id is null then '|J2|'
                     when decision_bls_id is not null and lawsuit_no is null then '|J3|'
                     when decision_bls_id is not null and lawsuit_no is not null and exdoc_bls_id is null then '|J4|'
                   end;

      if (stage_code = '') then
        stage_code = '|J0|';

      if (stage_code in ('|J1|', '|J2|', '|J0|')) then
        is_ok = 1; else

      is_ok = 0;
    end
    ----------------------------------------------------------------------------------

    if (is_ok = 1) then
    begin
      k_id = (select smr_from_k_id
              from sud_mainreq
              where smr_id = :smr_id);

      select k_kontract_no, f_datetostr(k_disb_date)
      from kontract
      where k_id = :k_id
      into :k_kontract_no, :k_disb_date_str;

      cl_count = (select count(slc_id)
                  from sud_loanclient
                  where slc_from_smr_id = :smr_id);
      gp_summ = (select smp_sum_sud
                 from sud_mainpay
                 where smp_id = :smr_id) / cl_count;

      select cl.c_name_fio, f_datetostr(cl.c_birth_date), cl.c_pass_no
      from clients cl
      where cl.c_id = :c_id
      into :c_name_fio, :c_birth_date_str, :c_pass_no;

      select adl_address_str
      from addrlist adl
      inner join addrclient adcl on adcl.adc_from_adl_id = adl.adl_id
      where adcl.adc_from_c_id = :c_id and
            adcl.adc_address_type = 1 -- Юридический (Адрес регистрации)
      into :address_legal;

      -- резолюцию

      resolution = f_format('Взыскать с %1 %2 г.р. пасп. %3 адр. %4, задолж. по ОС и %% в сумме %5р. ', c_name_fio, c_birth_date_str, c_pass_no, coalesce(:address_legal, 'XXX'), debt_summ) || f_format('по дог. №%1 от %2г. и расходы по оплате ГП - %3р.', k_kontract_no, k_disb_date_str, gp_summ);

      update sud_loanclient
    set slc_resolution = :resolution
    where slc_id = :slc_id;
      suspend;
    end

      end
    end
