CREATE PROCEDURE PRINT_CLIENT(
    AAREP_K_ID DECIMAL(18,0),
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_BIRTH_DATE_SLASH STR_20,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_TYPE INTEGER,
    C_CITIZEN_STR VARCHAR(255),
    C_VERIFY_STATUS INTEGER,
    KC_STATUS INTEGER,
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KC_AMOUNT_APPROVED_EKV DECIMAL(18,4),
    KC_NUMBER INTEGER,
    KC_PATENT_NO VARCHAR(40),
    KC_PATENT_DATE DATE,
    CA_ADDRESS VARCHAR(255),
    C_FULL_ADDRESS VARCHAR(255),
    C_FULL_ADDRESS_INDEX VARCHAR(255),
    KC_CIRCLE INTEGER,
    C_BANK_ACCOUNT_NO VARCHAR(50),
    C_BANK_CODE VARCHAR(10),
    C_BANK_NAME VARCHAR(255),
    C_BANK_CORR_ACC VARCHAR(255),
    K_KOMMIS_AMOUNT DECIMAL(18,4),
    KC_KOMMIS INTEGER,
    KC_BONUS_TEXT VARCHAR(30),
    C_ADDRESS_FACT VARCHAR(255),
    C_ADDRESS_RODINA_YUR VARCHAR(255),
    C_ADDRESS_RODINA_FACT VARCHAR(255),
    KC_NSP_AMOUNT DECIMAL(18,4),
    KC_KOMMIS_AND_NSP DECIMAL(18,4),
    C_ALTERNAT_FIO VARCHAR(100),
    C_PINCODE VARCHAR(100),
    KC_LOAN_TARGET VARCHAR(255),
    C_CELL_NO VARCHAR(20),
    C_EMAIL VARCHAR(100),
    CONTACT_DATA VARCHAR(5000),
    CONTACT_DATA_RF VARCHAR(5000),
    C_PASS_WHO_CODE VARCHAR(20),
    C_TEL VARCHAR(100),
    KC_LOAN_TARGET_LANG VARCHAR(255),
    C_UNIQ_ACC VARCHAR(12),
    FACT_LOAN_TARGET VARCHAR(100),
    FOR_PURPOSE_USED VARCHAR(100),
    ANALIZ_STATUS VARCHAR(100),
    ANALIZ_DATE DATE,
    ANALIZ_LO_FIO VARCHAR(100),
    KC_INN VARCHAR(20),
    KC_BANK_SWIFT VARCHAR(20),
    KC_BANK_MFO_OLD VARCHAR(10),
    LAST_NAME VARCHAR(100),
    FIRST_NAME VARCHAR(100),
    PATRONYMIC VARCHAR(100),
    C_SEX SMALLINT,
    C_BIRTH_COUNTRY VARCHAR(100),
    C_BIRTH_CITY VARCHAR(100),
    SNILS_NO VARCHAR(30),
    PK_NAME VARCHAR(100),
    PASSPORT_SERIES VARCHAR(30),
    PASSPORT_ID VARCHAR(30),
    C_INN VARCHAR(20),
    C_INN_MEMO VARCHAR(50),
    C_LSCHET_KG VARCHAR(20),
    PHONE_HOME STR_100,
    FAMILY_STATE STR_100,
    FAMILY_COUNT D_INTEGER,
    SPOUSE_NAME STR_100,
    SPOUSE_BIRTH D_DATE,
    SPOUSE_CELL STR_20,
    QR_CODE D_STR_LONG,
    EDL_ID D_ID,
    ALLOW_TRANSFER_RIGHTS D_SMALLINT,
    KC_PDN_KOEF DECIMAL(18,2),
    EDS_STATUS_DATE STR_20,
    EDS_STATUS_STR D_STR_CODE,
    DISB_KIND_STR D_STR_255,
    UUID D_STR_CODE,
    OA_PAN_MASKED D_STR_20,
    KC_LOAN_TARGET1 D_STR_255,
    OGRNIP D_STR_CODE)
AS
declare variable kc_from_c_id decimal(18,0); /* kc_loan_target */
declare variable k_kommis_perc decimal(18,7);
declare variable k_amount_approved decimal(18,4);
declare variable k_disb_date date;
declare variable c_citizen integer;
declare variable k_nsp_percent decimal(18,7);
declare variable k_kolich_client_approved integer;
declare variable k_nsp_amount decimal(18,4);
declare variable kc_nsp_amount_all decimal(18,4);
declare variable kol integer;
declare variable kc_kommis_all decimal(18,4);
declare variable cont_data_temp varchar(700);
declare variable kk integer;
declare variable cc_type integer;
declare variable acl_id id_type;
declare variable cl_analiz_date date;
declare variable acl_status integer;
declare variable c_from_fl_id id_type;
declare variable n smallint;
declare variable c_doc_type integer;
declare variable c_family integer;
declare variable pk_country varchar(3);
declare variable tmp_id d_id;
declare variable ma_id integer;
declare variable kc_id d_id;
declare variable k_disb_kind d_integer;
declare variable db_id d_integer;
declare variable c_id d_id;
declare variable obj_kind d_integer;
begin
  db_id = f_get_db_id();

  -- если передан НЕ K_ID, возможно это ID из рефинансирования. K_ID берем оттуда
  if (not exists(select *
                 from kontract
                 where k_id = :aarep_k_id)) then
  begin
    select krd_from_k_id
    from kontrefund_disb
    where krd_id = :aarep_k_id
    into :tmp_id;

    if (tmp_id is not null) then
      aarep_k_id = tmp_id;
  end

  k_nsp_percent = 0;
  select k_kommis_perc, k_amount_approved, k_disb_date, k_kolich_client_approved,
    k_from_ma_app_id, k_disb_kind
  from kontract
  where k_id = :aarep_k_id
  into k_kommis_perc, k_amount_approved, k_disb_date, k_kolich_client_approved,
    ma_id, k_disb_kind;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -2 -- нсп от доп.процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end

  oa_pan_masked = '';
  if (db_id in (39801, 41701)) then
    oa_pan_masked = coalesce((select oa_pan_masked from online_app where oa_from_k_id = :aarep_k_id), '');

  k_nsp_amount = 0;
  kc_kommis_all = 0;
  kc_nsp_amount_all = 0;
  kol = 0;
  for select c_id, c_name_fio, c_birth_date, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire, c_type, c_citizen,
             kc_status, kc_amount_approved, kc_amount_approved * k_index_curr_rate, kc_number, kc_patent_no,
             kc_patent_date, adl_address_str, kc_circle, kc_from_c_id, c_name_eng, c_pincode, rd_name, rd_memo,
             (select cont_value
              from hp_contdata_get_cell_main(0, c_id)),
             (select cont_value
              from hp_contdata_get_email_main(0, c_id)), cl.c_pass_who_code, coalesce(cl.c_inn, '-'), c_from_fl_id,
             c_sex, c_birth_country, c_birth_city, c_doc_type, c_inn, cl.c_family, cl.c_kol_family, cl.c_verify_status,
             kc_id, f_uuid_cbr(f_uuid('knclient', kc_id))
      from clients cl, knclient, addrclient, addrlist, kontract, refdata
      where kc_from_k_id = k_id and
            kc_from_c_id = c_id and
            k_id = :aarep_k_id and
            kc_amount_approved > 0 and
            adc_from_c_id = c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id and
            kc_loan_target = rd_id
      order by kc_number
      into c_id, c_name_fio, c_birth_date, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire, c_type, c_citizen,
           kc_status, kc_amount_approved, kc_amount_approved_ekv, kc_number, kc_patent_no, kc_patent_date,
           c_full_address, kc_circle, kc_from_c_id, c_alternat_fio, c_pincode, kc_loan_target, kc_loan_target_lang,
           c_cell_no, c_email, c_pass_who_code, kc_inn, c_from_fl_id, c_sex, c_birth_country, c_birth_city,
           c_doc_type, c_inn, c_family, family_count, c_verify_status, kc_id, uuid
  do
  begin
    c_birth_date_slash = extract(year from c_birth_date) || '/' || right(0 || extract(month from c_birth_date), 2) || '/' || right(0 || extract(day from c_birth_date), 2);

    c_inn = trim(c_inn);
    c_inn_memo = '';
    if (coalesce(c_inn, '') = '') then
    begin
      c_inn = '3';
      c_inn_memo = 'отсутствует';
    end

    if (not((c_alternat_fio containing ('uulu')) or (c_alternat_fio containing ('kyzy')))) then
    begin
      kk = position(' ', :c_alternat_fio);
      if (kk > 0) then
      begin
        kk = position(' ', c_alternat_fio, kk + 1);
        if (kk > 0) then
          c_alternat_fio = substring(:c_alternat_fio from 1 for kk - 1);
      end
    end
    kc_loan_target1 = kc_loan_target;
    if (kc_loan_target = 'Не указан') then
      kc_loan_target = '';

    select rd_name
    from refdata
    where rd_id = :c_citizen
    into :c_citizen_str;

    kol = kol + 1;

    if (db_id = 81001) then
    begin
      if (k_disb_date <= '15.10.2017') then
      begin
        select c_last_bonus_status_str
        from hp_kontstate_client_status(:kc_from_c_id, :k_disb_date)
        into :kc_bonus_text;
      end
      else
      begin
        select c_last_status_str
        from hp_get_client_status_str(:kc_from_c_id, :k_disb_date)
        into kc_bonus_text;
      end
    end
    ca_address = c_full_address;

    c_address_fact = '';
    c_address_rodina_yur = '';
    c_address_rodina_fact = '';

    select adl_address_str,
           left(iif(adc_phone is null or (adc_phone = ''), 'Номер не указан', adc_phone), 100)
    from addrclient, addrlist
    where adc_from_c_id = :kc_from_c_id and
          adc_address_type = 2 and
          adc_from_adl_id = adl_id
    into :c_address_fact, c_tel;
    if (c_address_fact is null) then
      c_address_fact = '';
    if (c_address_fact = 'Не указан') then
      c_address_fact = '';

    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :kc_from_c_id and
          adc_address_type = 4 and
          adc_from_adl_id = adl_id
    into :c_address_rodina_yur;
    if (c_address_rodina_yur is null) then
      c_address_rodina_yur = '';
    if (c_address_rodina_yur = 'Не указан') then
      c_address_rodina_yur = '';

    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :kc_from_c_id and
          adc_address_type = 5 and
          adc_from_adl_id = adl_id
    into :c_address_rodina_fact;
    if (c_address_rodina_fact is null) then
      c_address_rodina_fact = '';
    if (c_address_rodina_fact = 'Не указан') then
      c_address_rodina_fact = '';

    select res
    from fhp_round_bank(:k_amount_approved * :k_kommis_perc / 100.00, 0)
    into k_kommis_amount;
    select res
    from fhp_round_bank(:k_kommis_amount * :k_nsp_percent / 100, 0)
    into k_nsp_amount;

    select res
    from fhp_round_bank(:kc_amount_approved * :k_kommis_perc / 100, 2)
    into kc_kommis;
    if (kol < k_kolich_client_approved) then
      kc_kommis_all = kc_kommis_all + kc_kommis;
    else
      kc_kommis = k_kommis_amount - kc_kommis_all;

    select res
    from fhp_round_bank(:kc_kommis * :k_nsp_percent / 100, 0)
    into kc_nsp_amount;
    if (kol < k_kolich_client_approved) then
      kc_nsp_amount_all = kc_nsp_amount_all + kc_nsp_amount;
    else
      kc_nsp_amount = k_nsp_amount - kc_nsp_amount_all;

    kc_kommis_and_nsp = kc_kommis + kc_nsp_amount;

    kc_bank_swift = null;
    kc_bank_mfo_old = null;
    c_bank_corr_acc = null;
    c_bank_account_no = '';
    c_bank_code = '';
    c_bank_name = '';
    select bc_account_no, bc_from_b_id, b_name, b_corr_acc, b_swift, b_mfo_old
    from bankclient, banks
    where bc_from_c_id = :kc_from_c_id and
          bc_from_b_id = b_id and
          bc_is_main = 1 and
          bc_is_active = 1 rows 1
    into :c_bank_account_no, :c_bank_code, :c_bank_name, :c_bank_corr_acc, kc_bank_swift, kc_bank_mfo_old;

    contact_data_rf = '';
    contact_data = '';
    for select cc_type, contact_data
        from aarep_kontract_contacts(:kc_from_c_id)
        into cc_type, :cont_data_temp
    do
    begin
      if (cc_type = 0) then
        contact_data_rf = left(contact_data_rf || ascii_char(13) || cont_data_temp, 5000);
      if (cc_type = 1) then
        contact_data = left(contact_data || ascii_char(13) || cont_data_temp, 5000);
    end

    c_uniq_acc = 'XXX';
    select uniq_acc_str
    from hp_show_clientuniq_acc_no(:kc_from_c_id)
    into c_uniq_acc; --Лиц.счет добавлен 01.07.16

    if (c_uniq_acc = 'XXX') then
      c_uniq_acc = 'Не закреплен';
    else
      c_lschet_kg = extract(year from c_birth_date) || right(0 || extract(month from c_birth_date), 2) || right(0 || extract(day from c_birth_date), 2) || '-' || c_uniq_acc;

    if (c_full_address containing 'г.Санкт-Петербург') then
    begin
      if (c_full_address starting 'г') then
        c_full_address_index = '190000, ' || c_full_address;
      else
        c_full_address_index = '190000' || substring(c_full_address from 7);
    end
    else
      c_full_address_index = c_full_address;

    acl_id = null;

    fact_loan_target = '-'; -- фактическая цель займа
    for_purpose_used = '-'; -- целевое использование
    analiz_status = '-'; -- Статус проведения повторного анализа платежеспособности клиента
    analiz_lo_fio = '-';

    select al.acl_id, cast(al.acl_last_datetime as date), cl.c_name_fio, al.acl_object_kind
    from actionlist al
    inner join refdata rd on rd.rd_id = al.acl_from_rd_id
    inner join clients cl on cl.c_id = al.acl_respons_c_id
    where al.acl_date >= :k_disb_date --+ 59
          and
          al.acl_status = 2 and
          ((al.acl_object_kind = 0 and al.acl_object_id = :kc_from_c_id) or
           (al.acl_object_kind = 8 and al.acl_object_id = :kc_id)) and
          rd.rd_code = 'Anketa.TargetUseLoan'
    order by al.acl_date
    rows 1
    into :acl_id, :analiz_date, :analiz_lo_fio, :obj_kind;

    analiz_date = coalesce(analiz_date, current_date);
    cl_analiz_date = null;

    if (acl_id is not null) then -- если есть заполненная анкета "Мониторинг целевого использования"
    begin
      select res_str_values
      from hp_objectprops_anketa_get(
        iif(:obj_kind = 0, 'Clients', 'KnClient'),
        iif(:obj_kind = 0, :kc_from_c_id, :kc_id),
        -1,
        'AnketaQuestions.LoanUseFact',
        :analiz_date,
        null)
      into :fact_loan_target;

      select cast(al.acl_last_datetime as date), acl_status
      from actionlist al
      inner join refdata rd on rd.rd_id = al.acl_from_rd_id
      inner join clients cl on cl.c_id = al.acl_respons_c_id
      where al.acl_parent_id = :acl_id and
            al.acl_status >= 0 -- важно чтобы анкета балы, хотябы в процессе
            and
            al.acl_object_kind = 0 and
            al.acl_object_id = :kc_from_c_id and
            rd.rd_code = 'Anketa.ClientAnaliz' and
            rd.rd_is_active = 1 rows 1
      into :cl_analiz_date, acl_status;

      if (cl_analiz_date is null) then -- если анкета присутствует, значит уже выявлено, что использование кредита - "не целевое"
        for_purpose_used = 'Да';
      else
      begin
        for_purpose_used = 'Нет';

        if ((exists(select *
                    from refdata rd
                    where rd.rd_code = 'AnketaQuestions.Reanalysis' and
                          rd.rd_is_active = 1)) and
            (acl_status = 2)) then
        begin
          select res_str_values
          from hp_objectprops_anketa_get('Clients', :kc_from_c_id, -1, 'AnketaQuestions.Reanalysis', :cl_analiz_date, null)
          into :analiz_status;
        end
        analiz_status = 'Да'; -- Временно, пока идет аудит 12.12.2016, чтобы в анкете было видно, что анализ проведен
      end

      if (k_disb_date < '23.10.2016') then
        analiz_date = k_disb_date + 60;
    end

    last_name = null;
    first_name = null;
    patronymic = null;
    select fl_name1, fl_name2, fl_name3
    from fiolist
    where fl_id = :c_from_fl_id
    into last_name, first_name, patronymic;
    last_name = coalesce(last_name, '');
    first_name = coalesce(first_name, '');
    patronymic = coalesce(patronymic, '');

    if (last_name <> '') then
    begin
      n = char_length(last_name);
      last_name = left(last_name, 1) || right(lower(last_name), n - 1);
    end
    if (first_name <> '') then
    begin
      n = char_length(first_name);
      first_name = left(first_name, 1) || right(lower(first_name), n - 1);
    end
    if (patronymic <> '') then
    begin
      n = char_length(patronymic);
      patronymic = left(patronymic, 1) || right(lower(patronymic), n - 1);
    end

    snils_no = '';
    select max(iif(pdt_code = 'snils', cds_doc_no, ''))
    from clientdocs, passkind, passdoctype
    where cds_from_c_id = :kc_from_c_id and
          cds_is_delete = 0 and
          coalesce(cds_doc_blob_1, 0) > 0 and
          cds_from_pk_id = pk_id and
          pk_from_pdt_id = pdt_id and
          current_date between cds_doc_date_begin and coalesce(cds_doc_date_end, '31.12.2999')

    pk_country = '';

    if (c_pass_no <> '' and

    begin
      n = char_length(c_pass_no);

      begin
        passport_series = left(c_pass_no, 2);
        passport_id = right(c_pass_no, n - 2);
      end
      else
      if (pk_country in ('rus')) then
      begin
        passport_series = left(c_pass_no, 4);
        passport_id = right(c_pass_no, n - 4);
      end
    end

    phone_home = f_cont_info('clients', kc_from_c_id, 'Phone_Home', 'last');

    family_state = null;
    spouse_name = null;
    spouse_birth = null;
    spouse_cell = null;
    select rd.rd_name
    from refdata rd
    where rd.rd_id = :c_family
    into family_state;

    select cl1.c_name_fio, cl1.c_birth_date, f_cont_info('clients', cl1.c_id, 'cell_main', 'last')
    from clientrelation cr
    left join relationtype rt on (rt.rel_id = cr.crel_from_rel_id and
          rt.rel_can_select = 1)
    left join clients cl1 on cl1.c_id = cr.crel_other_c_id
    where cr.crel_from_c_id = :kc_from_c_id and
          cr.crel_from_rel_id in (1, 2) and
          cr.crel_is_active = 1 rows 1
    into spouse_name, spouse_birth, spouse_cell;

    -- QR-код электронной подписи клиента. Текст пустой если не подписан
    if (coalesce(add_params, '') <> '') then

    -- Уступка прав требования
    allow_transfer_rights = 1;
    if (ma_id is not null) then

            mac_from_c_id = :kc_from_c_id
      order by mac_amount desc

    end

    if (k_disb_date >= '01.09.2020' and db_id = 81001) then
    begin
      kc_pdn_koef = 0;
      select coalesce(round(kp.kc_pdn_koef, 2), 0)
      from knclient_pdn kp
      where kp.kc_id = :kc_id
        and kcp_kind = 0
      into :kc_pdn_koef;

      --< Добавлено по задаче на jira 2812

      eds_status_str = '';

      select f_datetostr(eds_status_datetime), eds_status_str

      rows 1
      into :eds_status_date, :eds_status_str; -->
    end

    if (k_disb_kind = 1) then
      disb_kind_str = 'Наличными денежными средствами';
    else
      disb_kind_str = 'Перечисление на банковскую карту';

    if (db_id = 81001 and k_disb_kind = 7) then
    begin
      select f_format('Перечисление денежных средств на счет в кредитную организацию с использованием системы быстрых платежей согласно реквизитам: %1, %2',
        cell_no, coalesce(fpb_name_full, fpb_name))

    if (db_id = 81001 and k_disb_kind = 8) then
      disb_kind_str = f_format('Перечисление денежных средств на расчетный счет, согласно реквизитам: БИК: %1, Номер счета:%2', :c_bank_code, :c_bank_account_no);

      into ogrnip;
