  res_show_kind = 'dialog';
  db_id = f_get_db_id();
  start_date = '08.09.2021';
  tmp_number = 0;
  date_20250101 = cast('01.01.2025' as date); -- нужно поменять на 01.01.2025
  date_20250710 = cast('10.07.2025' as date);  
  date_20260210 = cast('10.02.2026' as date);
  select max(ks_state)
  from kontstate
  where ks_from_k_id = :rep_k_id and
        ks_ol_is_delete = 0
  into :tmp_number;
  if (tmp_number is null) then
    tmp_number = 0;
  if (tmp_number = 1) then
  begin
    res_show_kind = 'dialog';
    res_text = 'Договор уже активный, займ ранее выдан.';

  if (tmp_number = 2) then
  begin
    res_show_kind = 'dialog';
    res_text = 'Договор уже закрыт, нельзя повторно выдавать.';
    suspend;
    exit;
  end
  if (db_id = 81001) then
  begin
    if (exists(select *
               from kontract
               inner join product on pr_id = k_from_pr_id
               where k_id = :rep_k_id
               and (pr_code containing '.zalog.' or pr_code containing '.zalog_car.'))) then
    begin
      if (f_edoc_status('loan.collateral', :rep_k_id, current_date) <> 'signed') then
      begin
        res_show_kind = 'dialog';
        res_text = 'Не подписан Договор залога';
        suspend;
        exit;
      end
    end
  end 
  if (rep_disb_kind = 1 and
      db_id <> 94401) then
  begin
    if (exists(select *
               from extsysmain
               where esm_from_k_id = :rep_k_id and
                     esm_project in ('moneta_disb', 'elcard_disb') and
                     esm_action in ('CardDisb', 'PhoneDisb') and

    begin

      suspend;
      exit;

      rep_disb_reg_id <> 1010100) then -- Монета
  begin

    exit;
  end
  if (rep_disb_kind = 1 and
      rep_repay_kind <> 1) then
  begin

    res_text = 'При наличной выдаче доп.процент тоже должен приниматься наличными';

    exit;
  end
  if (db_id = 41701 and
      rep_disb_kind = 5 and
      rep_disb_reg_id <> 1090100) then -- Бишкек ГО
  begin
    res_show_kind = 'dialog';

    exit;

    select lir.lir_kontract_insurance
    from loan_insurance_relation lir

    into kontract_insurance;

    if (coalesce(kontract_insurance, 0) = 0 or
        f_edoc_status('loan.agreement', :kontract_insurance, current_date) <> 'signed') then
    begin
      res_show_kind = 'dialog';
      res_text = 'Cначала нужно оформить кредит на страхование';
      suspend;
      exit;

  end

  if (db_id = 41701 and f_get_const('self_prohibition') = 1) then
  begin
    select res
    from hp_self_prohibition_get_status(:rep_cashier_id, 'kontract', :rep_k_id)
    into :prohibition_status;

    if (prohibition_status in (0, -1, -2, 1, -3)) then
    begin
      res_show_kind = 'dialog';

      res_text = case prohibition_status
                   when 0  then 'Выдача невозможна, запрос в сервис Самозапрет в обработке'
                   when -1 then 'Выдача невозможна, ошибка в запросе в сервис Самозапрет'
                   when -2 then 'Выдача невозможна, требуется прикрепить скриншот с Тундука по отсутствию самозапрета'
                   when 1  then 'Выдача невозможна, у клиента действует Самозапрет'
                   when -3  then 'Выдача невозможна, запрос в сервис Самозапрет не осуществлялся'

 /*   if (db_id = 39801 and
      rep_disb_kind = 5 and

  end
  select list(kf_client_fio || ': ' || kf_error_text)

  if (res_text is null) then

         k_kommis_perc, k_index_curr_code, k_kontract_no, pr_round_admin, pr_code, k_prepayment_summa,
         k_kolich_client_approved, k_add_month, k_percent, k_percent_per_day, k_disb_kind, pr_percent_kind,

  from kontract, product
  where k_id = :rep_k_id and
        k_from_pr_id = pr_id
  into tmp_number, kf_id, k_disb_date, pr_id, k_kontract_date, k_disb_amount, k_kommis_perc, k_currency, k_kontract_no,

  if (k_kommis_perc is null) then

  -- для КР начиная от 02.09.2020 Доп.процент можно платить отдельно отложенно

  --  need_repay_add_perc = 0;

  into :loan_office, :folder_no;

    exit;

  begin

    suspend;
    exit;
  end

      rep_disb_kind = 4) or (db_id = 94401 and

    if (db_id = 81001 and
        rep_disb_kind = 4) then
      paysys_proj_id = 6;

        paysys_proj_id = 1;

        paysys_proj_id = 2;

    suspend;
    exit;
  end
  if (tmp_number <> 1) then
  begin
    res_show_kind = 'dialog';
    res_text = 'Договор не утвержден. Выдача отменяется.';
    suspend;
    exit;

  -- корректируем тип клиента и резидентство перед выдачей
  for select kc_id, kc_from_c_id
      from knclient
      where kc_from_k_id = :rep_k_id
      into :kc_id, :kc_from_c_id
  do
  begin
    select res_client_type, res_non_resident
    from hp_get_knclient_type(:kc_id, null, null)  -- изменен от 02.04.2026г Улан, Было hp_get_client_type(:kc_from_c_id, null, null)
    into kc_client_type, kc_non_resident;
    update knclient
    set kc_client_type = :kc_client_type,
        kc_non_resident = :kc_non_resident
    where kc_id = :kc_id and
          (coalesce(kc_client_type, -1) <> :kc_client_type or
          coalesce(kc_non_resident, -1) <> :kc_non_resident);
  end
  execute procedure hp_do_kn_resident_update(:rep_k_id, null, null, null, :un_id);
  -- Сумма комиссии
  k_kommis_amount = round(k_disb_amount * k_kommis_perc / 100.0, -pr_round_admin);
  select sum(kc_kommis_lgota)
  from knclient
  where kc_from_k_id = :rep_k_id and
        kc_amount_approved > 0
  into :tmp_amount;
  if (tmp_amount is null) then
    tmp_amount = 0;
  k_kommis_amount = k_kommis_amount - tmp_amount;
  if (k_kommis_amount < 0) then
    k_kommis_amount = 0;

  curr_region_id = f_hq_id(db_id);

  select res_account
  from hp_make_kredit_acclist(:pr_id, :loan_office, 1) -- Основная сумма
  into :acc_os;
  if (coalesce(acc_os, '') = '') then
  begin
    res_show_kind = 'dialog';
    res_text = 'Счет учета кредита не найден';
    suspend;
    exit;
  end
  k_nsp_amount = 0;
  delay_acc_kommis = rep_repay_account;
  delay_acc_kommis_nsp = rep_repay_account;
  if (k_kommis_amount <> 0.0) then
  begin
    select nsp_percent_for_extra_int, nsp_acc_extra_int_repay
    from hp_get_kontract_nsp_acc(:loan_office, :k_disb_date)
    into :nsp_perc, acc_nsp;
    if ((need_repay_add_perc = 0) and
        (not(k_disb_kind = 6 and
        db_id in (39801, 41701)))) then
    begin
      select res_account
      from hp_make_kredit_acclist(:pr_id, :loan_office, 43) -- Отложенный доп.процент
      into :delay_acc_kommis;
      if (coalesce(delay_acc_kommis, '') = '') then
      begin
        res_show_kind = 'dialog';
        res_text = 'Счет учета отложенного доп.процента не найден';
        suspend;
        exit;
      end
      if (nsp_perc <> 0.0) then

  from hp_do_kontmat_sell(:rep_k_id, :ol_id_disb, :rep_disb_account)
  into :res_id, :res_text;
  if (res_id = -1) then
  begin
    res_ol_ids = '';
    res_show_kind = 'dialog';
    suspend;
    exit;
  end
  if (k_disb_kind = 6 and
      db_id in (39801, 41701)) then
    k_kommis_amount = 0;
  if (k_kommis_amount <> 0) then -- если нет комиссии, завершаем работу
  begin
    ol_memo = (case
                 when (db_id in (39801, 41701)) then 'Комиссия за кассовое обслуживание по договору ' || k_kontract_no || ' от ' || k_kontract_date_str || ' г. Кр.папка № ' || folder_no
                 when (db_id = 81001) then 'Приход доп.процент за использование займа согл. договора займа ' || k_kontract_no || ' от ' || k_kontract_date_str || ' г. Кр.папка № ' || folder_no
                 when (db_id = 94401) then k_kontract_date_str || ' tarixli ' || k_kontract_no || ' nomreli muqavileye esasen istifade olunmush kredit uchun elave faizin odenishi. Kredit papkasi № ' || folder_no
               end);
    -- Заполняем таблицы для комиссии
    ol_id_kommis = 0;
    ol_id_doc = null;
    -- Проводки с учетом лоро-ностро если выдается из другого офиса
    for select res_reg_id, res_acc_dt, res_acc_kt, total_line, res_line_no
        from aarep_loronostro_link(iif(:need_repay_add_perc = 1, :rep_repay_reg_id, :loan_office), :delay_acc_kommis, :acc_kommis)
        into :res_reg_id, :res_acc_dt, :res_acc_kt, :total_line, :res_line_no
    do
    begin
      --RFCKG-2443
      --Редактируем только ДТ счет, КТ счет автоматически берётся из счетов кредитования
      is_online_loan_kg = 0;
      if (db_id in (39801, 41701) and pr_code containing '.online_loan.' and k_disb_kind = 5) then
      begin
        is_online_loan_kg = 1;
        
        acc_dt_kommis_online_loan = (select al_id
                                     from acc_list al
                                     where al.al_from_reg_id = :rep_repay_reg_id and
                                           al.al_currency = '417' and
                                           al.al_account = '11531/101');

        acc_dt_nsp_online_loan = (select al_id
                                  from acc_list al
                                  where al.al_from_reg_id = :rep_repay_reg_id and
                                        al.al_currency = '417' and
                                        al.al_account = '11531/108');

        ol_memo = 'Комиссия по договору №' || :k_kontract_no || ' от ' || :k_kontract_date_str || ' Кр. папка №' || :folder_no;
      end 

      select kv
      from get_kv(:curr_region_id, 5)
      into :ol_id;
      if (ol_id_doc is null) then
        ol_id_doc = ol_id;
      ol_id_kommis = ol_id;
      if (res_line_no = 1) then
      begin
        ol_doc_kind = (case
                         when rep_repay_kind = 1 then 1 -- приходный ордер
                         when rep_repay_kind = 2 then 7 -- поступление в банк
                         when rep_repay_kind in (3, 4, 5, 7, 8) then 9 -- мемориальный ордер
                       end);
        -- если доп.процент отложен, печатаем мем.ордер
        if (need_repay_add_perc = 0) then
          ol_doc_kind = 9;
        -- очередной номер документа
        select num
        from last_doc_number(:k_disb_date, :ol_doc_kind, :rep_repay_reg_id)
        into :ol_doc_no;
        ol_doc_no_str = :ol_doc_no;
      end
      else
        ol_doc_kind = 9;
      -- тип операции
      if (res_line_no = total_line) then
      begin
        -- выдача кредита, приход доп.процента
        ol_oper_type = 1;
        ol_oper_subtype = 1;
      end
      else
      begin
        -- перевод между офисами
        ol_oper_type = 2;
        ol_oper_subtype = 20;
      end
      -- Вставка новой операции в Operlist
      insert into operlist (ol_id, ol_from_reg_id, ol_date, ol_doc_no, ol_doc_no_str, ol_oper_type, ol_oper_subtype,
                            ol_doc_kind, ol_memo, ol_user_id, ol_document_client, ol_doc_client_name, ol_is_client,
                            ol_parent_id, ol_manager_c_id, ol_from_rd_id, ol_acc_no_client)
      values (:ol_id, :res_reg_id, :k_disb_date, :ol_doc_no, :ol_doc_no_str, :ol_oper_type, :ol_oper_subtype,
              :ol_doc_kind, :ol_memo, :un_id, :rep_cashier_id, '', iif(:rep_cashier_id is not null, 1, 0), :ol_id_main,
              :rep_manager_id, 0, :bc_number);
      if (res_line_no = total_line) then
      begin
        update operloan
        set opl_from_al_id = :rep_repay_account,
            opl_acc_kind = iif(:rep_repay_kind in (1, 2), :rep_repay_kind, 3)
        where opl_from_ol_id = :ol_id_doc;
      end
      -- Вставка новой проводки в Operbook
      if (res_line_no < total_line) then -- через лоро-ностро, общая сумма
      begin
        res_memo = ol_memo;
        insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val,
                              ob_summa, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
        values (:ol_id, :k_currency, 1, :res_acc_dt, :res_acc_kt, :k_kommis_amount + :k_nsp_amount,
                :k_kommis_amount + :k_nsp_amount, :res_memo, :k_disb_date, 1, :res_reg_id);
      end
      else
      begin
        res_memo = ol_memo;
        insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val,
                              ob_summa, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
        values (:ol_id, :k_currency, 1,
                iif(:is_online_loan_kg = 1, trim(:acc_dt_kommis_online_loan), trim(:res_acc_dt)), --RFCKG-2443
                :res_acc_kt, :k_kommis_amount, :k_kommis_amount, :res_memo, :k_disb_date, 1, :res_reg_id);
        if (k_nsp_amount <> 0) then
        begin
          res_memo = iif(db_id in (39801, 41701), 'Выплачен НсП на комиссию за кассовое обслуживание кредита', 'Выплачен НсП на доп.процент');
          if (need_repay_add_perc = 0) then
          begin
            res_acc_dt = delay_acc_kommis_nsp;
            res_memo = iif(db_id in (39801, 41701), 'Признание НсП на комиссию за кассовое обслуживание кредита', 'Признание НсП на доп.процент');
          end

          if (is_online_loan_kg = 1) then
            res_memo = 'НсП Комиссии по договору №' || :k_kontract_no || ' от ' || :k_kontract_date_str || ' Кр. папка №' || :folder_no;

          insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val,
                                ob_summa, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
          values (:ol_id, :k_currency, 1,
                  iif(:is_online_loan_kg = 1, trim(:acc_dt_nsp_online_loan) , trim(:res_acc_dt)), --RFCKG-2443
                  :acc_nsp, :k_nsp_amount, :k_nsp_amount, :res_memo, :k_disb_date, 1, :res_reg_id);
        end
      end
      -- S.K.
      if (need_repay_add_perc = 1) then
        res_ol_ids = res_ol_ids || ';' || ol_id_doc || '-' || ol_doc_kind;
    end
    -- Создаем записи в Operkred
    if (need_repay_add_perc = 1) then
    begin
      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                            ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
                            -- Строка комиссии
      select :ol_id_kommis, -1, :rep_k_id, sh_id, sh_date, 4, 0, 0, 0, :rep_repay_kind, -:k_kommis_amount, 0, 0, 0,
             -:k_nsp_amount, :k_disb_date, :loan_office, 1, 0
      from schedule
      where sh_id = 0 and
            sh_from_k_id = :rep_k_id;
    end

    if (db_id = 94401 and rep_repay_kind <> 1) then
    begin
        acc_dt_account_no = '45170/1';
        acc_kt_account_no = '25031/1';
      select al_id
      from find_acc_list(:acc_dt_account_no)
      where al_from_reg_id = :res_reg_id and
            al_account = :acc_dt_account_no
      into :acc_dt;
  
      select al_id
      from find_acc_list(:acc_kt_account_no)
      where al_from_reg_id = :res_reg_id and
            al_account = :acc_kt_account_no
      into :acc_kt;
  
      insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val,
                            ob_summa, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
      values (:ol_id, :k_currency, 1, :acc_dt, :acc_kt, :k_kommis_amount, :k_kommis_amount,
              :res_memo,
              :k_disb_date, 1, :res_reg_id);
    end

  end

  -- RFCKG-4552
  if (db_id = 39801) then
  begin
    k_kommis_amount = round(k_disb_amount * 0.0116, 2);
    if (k_kommis_amount > 0) then
    begin
      acc_dt = (
        select al_id from acc_list
        where al_from_reg_id = :res_reg_id
          and al_currency = :k_currency
          and al_account = '7470.21/1');
      res_memo = f_format('Комиссия по договору %1 от %2г. Кр.папка № %3', k_kontract_no, k_kontract_date_str, folder_no);
      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_summa_val,
        ob_summa,
        ob_detail,
        ob_ol_date,
        ob_ol_is_active,
        ob_ol_reg_id)
      select
        ob_from_ol_id,
        ob_currency,
        1,
        iif(ol_id = :ol_id_main, :acc_dt, ob_account_dt),
        ob_account_kt,
        :k_kommis_amount,
        :k_kommis_amount,
        :res_memo,
        ob_ol_date,
        1,
        ob_ol_reg_id
      from operbook
        inner join operlist
          on ol_parent_id = :ol_id_main
            and ob_from_ol_id = ol_id;
    end
  end

  if (db_id = 81001 and
      k_disb_date >= '01.01.2018') then
    execute procedure hp_do_kn_operbook_one_online(:rep_k_id, :k_disb_date, null, null);
  if (res_ol_ids <> '') then
  begin
    if ((db_id = 94401) and
        (rep_disb_kind in (4, 8)) and
        (k_kolich_client_approved = 1)) then -- Millicard (4 - Standart, 8 - Business)
    begin
      execute procedure millicard_do_disbursement(:rep_k_id, :kc_from_c_id, :k_disb_amount_no_kommis);
    end
    if (db_id = 81001 and
        rep_disb_kind in (5, 7) and
        k_kolich_client_approved = 1) then
    begin
      if (:k_disb_date >= '08.04.2019') then
        execute procedure moneta_do_disbursement(:rep_k_id, :kc_from_c_id, :k_disb_amount); -- #moneta_disb
    end
    -- RFCKZ-471
--    if (db_id = 41701 and
--        rep_disb_kind = 5) then
--    begin
--      execute procedure elcard_do_transfer(:rep_k_id); -- инициирует задачу на создание платежного ордера при выдаче на стороннюю Элкарт карту
--    end
    if ((db_id = 81001) and
        (rep_disb_kind = 4) and
        (k_kolich_client_approved = 1)) then
    begin
      execute procedure rus_standart_do_disbursement(:rep_k_id, :kc_from_c_id, /*:k_disb_amount*/ 40); -- временно ставлю 40 рублей  #rus_standart
    end
    if (db_id = 41701 and
        f_tmp_const('use_checkpost') = '1') then
      select acl_id
      from create_postcheck_action(:rep_k_id, :rep_disb_reg_id)
      into acl_id;

  end

  if (db_id = 41701 and pr_code containing '.insur_base.') then
  begin
    select lir.lir_kontract_insurance
    from loan_insurance_relation lir
    where lir.lir_kontract_base = :rep_k_id
    into kontract_insurance;

    execute procedure hp_insurance_auto_disb(:kontract_insurance, :un_id);
  end

  if (db_id = 41701) then
  begin
    if (f_get_const('new_loan_conditions') = 1) then
      insurance_min_amount = cast(f_get_const('kont_new_insurance_min_amount') as d_integer);
    else
      insurance_min_amount = cast(f_get_const('kont_insurance_min_amount') as d_integer);
    
    if (pr_code containing '.ipf.') then
      insurance_min_amount = cast(f_get_const('kont_insurance_ipf_min_amount') as d_integer);

    if (f_get_const('kont_insurance_active') = 1
        and pr_code containing '.IND.'
        and (pr_code containing '.STANDART.' or
             pr_code containing '.IPF.') -- RFCKG-5348
        and pr_code not containing '.ignore_insurance.'
        and k_disb_amount >= :insurance_min_amount
        and exists(select 1 from insurance_kontract ik where ik.ik_from_k_id = :rep_k_id)
        and (   
             (pr_code not containing '.consultation.')
             or (f_get_const('new_loan_conditions') = 1 and pr_code containing '.consultation.' and pr_code containing '.INSUR.') -- RFCKG-4724
            )
    ) then
    begin
      update insurance_kontract ik
      set ik.ik_enum_status = f_enum_id('insurance_kontract', 'loan_is_granted')
      where ik.ik_from_k_id = :rep_k_id;
    end

    pair_k_id = null;
    if (pr_code containing '.insur_base.' ) then
    begin
      select lir.lir_kontract_insurance
      from loan_insurance_relation lir
      where lir.lir_kontract_base = :rep_k_id
      into pair_k_id;
    end

    if(pr_code containing '.insur_kontrakt.')   then
    begin
      select lir.lir_kontract_base
      from loan_insurance_relation lir
      where lir.lir_kontract_insurance = :rep_k_id
      into pair_k_id;
    end

    select
      sum(iif(pr.pr_code not containing '.ipf.', oks.oks_balance, 0)),
      sum(iif(pr.pr_code containing '.ipf.', oks.oks_balance, 0))
    from knclient kc
    inner join kontract k on kc.kc_from_k_id = k.k_id
    inner join product pr on k.k_from_pr_id = pr.pr_id
    inner join oksummary oks on kc.kc_from_k_id = oks.oks_from_k_id
      and :k_disb_date between oks.oks_date_begin and oks.oks_date_end
      and oks.oks_close_state in (0,1)
      and oks.oks_from_k_id <> :rep_k_id
      and (:pair_k_id is null or oks.oks_from_k_id <> :pair_k_id)
    where kc.kc_from_c_id = :kc_from_c_id
      and kc.kc_amount_approved > 0
    into
      :standard_balance,
      :ipf_balance;

    update or insert into kontract_issue_state(kis_from_k_id, kis_standard_points, kis_standard_balance, kis_ipf_points, kis_ipf_balance)
    values (:rep_k_id, f_point(:kc_from_c_id, :k_disb_date, 1), coalesce(:standard_balance, 0), f_point(:kc_from_c_id, :k_disb_date, 2), coalesce(:ipf_balance, 0))
    matching (kis_from_k_id);

  end

  suspend;
