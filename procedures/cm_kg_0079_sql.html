CREATE PROCEDURE CM_KG_0079_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable KAL_NEW_KF_OFFICER ID_TYPE;
declare variable UN_FROM_C_ID ID_TYPE;
declare variable PR_CODE varchar(100);
declare variable JK_SYSTEM_CODE varchar(100);
begin
  if (K_ID = 0) then
    exit;
  if (k_id = 109010278901200101) then
    exit; --RFCKG-490
  if (K_ID = 109010239773700101) then
    exit; ---RFCKG-613
  if (k_id = 109010242287300101) then
    exit; -- RFCKG-689
  if (k_id = 109010244875700101) then
    exit; --RFCKG-749
  if (k_id = 109010245540400101) then
    exit; --RFCKG-768
  if (k_id = 109010248810200101) then
    exit; --RFCKG-847
  if (k_id = 109010249331800101) then
    exit; --RFCKG-878
  if (k_id = 109010294303800101) then
    exit; --RFCKG-2905

  select KAL_NEW_KF_OFFICER, K.K_AMOUNT_APPROVED, K_DISB_DATE, PR_CODE
  from KONTRACT K
  inner join PRODUCT PR on K_FROM_PR_ID = PR_ID
  inner join KNAKTLIST KAL on (K_FROM_KF_ID = KAL_FROM_KF_ID and
        KAL.IS_DELETE = 0 and
        K_DISB_DATE between KAL_DATE and KAL_DATE_END)
  where K_ID = :K_ID
  into KAL_NEW_KF_OFFICER, KC_AMOUNT, KF_DISB_DATE, PR_CODE;
  if (coalesce(KAL_NEW_KF_OFFICER, 0) <> 0) then
  begin
    select UN.UN_FROM_C_ID
    from USERNAME UN
    where UN_ID = :UN_ID
    into UN_FROM_C_ID;

    select JK_SYSTEM_CODE
    from HR_PSTATUS, HR_ORDERS, JOBLIST JL, JOBKIND JK
    where PS_STATUS = 1 and
          PS_FROM_HO_ID = HO_ID and
          PS_FROM_JL_ID = JL_ID and
          PS_FROM_C_ID = :UN_FROM_C_ID and
          :KF_DISB_DATE between PS_DATE_BEGIN and PS_DATE_END and
          JK.JK_ID = JL.JL_KIND
    order by PS_DATE_BEGIN desc rows 1
    into JK_SYSTEM_CODE;
    if (PR_CODE containing '.farmer.') then
    begin
      /*
        SYSDBA
        |условия утверждения по суммам, CCПФ, МО до 100 000, РМ 100 001 до 200 000 для продукта Фермер:
        в коде продукта слово "farmer"
    */
      if (KC_AMOUNT > 100000 and
          JK_SYSTEM_CODE in ('JobSystem.LO_Senior', 'JobSystem.Region_KM')) then
      begin
        res_error_text = 'МО/ССПФ может утвердить максимум 100000 с.';
        res_error_is_critical = 1;
        suspend;
      end
      if (KC_AMOUNT > 250000 and
          JK_SYSTEM_CODE in ('JobSystem.Region_Director')) then
      begin
        res_error_text = 'РМ может утвердить максимум 250000 с.';
        res_error_is_critical = 1;
        suspend;
      end
    end
    if (KAL_NEW_KF_OFFICER = coalesce(UN_FROM_C_ID, 0)) then
    begin
      res_error_text = 'Вы не можете утвердить свой контракт!';
      res_error_is_critical = 1;
      suspend;
    end
  end
end
