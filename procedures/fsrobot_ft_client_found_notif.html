CREATE PROCEDURE FSROBOT_FT_CLIENT_FOUND_NOTIF(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0),
    CL_NAME STR_100)
AS
declare variable c_id decimal(18,0);
declare variable c_alternat_fio str_100;
declare variable field_code varchar(1000);
declare variable field_value varchar(1000);
declare variable c_comment str_255;
declare variable ft_name_fio str_255;
declare variable reg_id integer;
declare variable manager_id id_type;
declare variable c_name_fio str_100;
declare variable c_pass_no str_100;
declare variable db_id integer;
declare variable manager_fio str_100;
declare variable manager_cell_no str_100;
declare variable reg_name str_100;
declare variable finpolis_type str_100;
declare variable finpolis_type_name str_100;
begin
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID  - пользователь

-- СМС передаче клиента в Департамент розыска и принудительного взыскания для судебного разбирательства
--C_ID       = substring(rep_params from 1 for position(';' in rep_params) - 1);

--Res_MailList=:rep_email;

  for select t.field_code, t.field_value
      from hp_extsys_parcer(:rep_params, null) t
      into :field_code, :field_value
  do
  begin
    if (field_code = 'C_COMMENT') then
      c_comment = field_value;
    if (field_code = 'FT_NAME_FIO') then
      ft_name_fio = field_value;
    if (field_code = 'C_ID') then
      c_id = cast(iif(field_value = '', '0', field_value) as id_type);
    if (field_code = 'REG_ID') then
      reg_id = cast(iif(field_value = '', '0', field_value) as integer);
    if (field_code = 'LO_ID') then
      manager_id = cast(iif(field_value = '', '0', field_value) as id_type);
    if (field_code = 'DB_ID') then
      db_id = field_value;
    if (field_code = 'FINPOLIS_TYPE') then
      finpolis_type = field_value;
  end

  finpolis_type = coalesce(finpolis_type, '');

  select cl.c_name_fio, cl.c_pass_no
  from clients cl
  where cl.c_id = :c_id
  into :c_name_fio, :c_pass_no;

  select cl.c_name_fio,
         (select cont_value
          from hp_contdata_get_cell_valid(0, :manager_id))
  from clients cl
  where cl.c_id = :manager_id
  into :manager_fio, :manager_cell_no;

  select reg.reg_name
  from region reg
  where reg.reg_unicode = :reg_id
  into :reg_name;

  if (finpolis_type = '639') then
    finpolis_type_name = 'Перечня 639';
  else
    finpolis_type_name = 'Перечня Росфинмониторинга/ Решения о замораживании (блокировании)';

  res_subject = 'Совпадение контактной информации о клиенте с лицом из ' || finpolis_type_name;

  res_body = '<b>Совпадение контактной информации о клиенте с лицом из ' || finpolis_type_name || ' ' || coalesce(c_comment, '') || ' ' || coalesce(ft_name_fio, '') || '</b><br>' || 'Клиент: ' || coalesce(:c_name_fio, '') || ' <br>' || 'Офис: ' || coalesce(:reg_name, '') || ' <br>' || 'СВЗ: ' || coalesce(:manager_fio, '') || ' <br>' || 'Тел СВЗ: ' || coalesce(:manager_cell_no, '') || ' <br>';

  res_maillist = 'oft@molbulak.com';
  res_c_id = :c_id;
  suspend;
end
