CREATE PROCEDURE HP_PARSE_JSON_GBDFL(
    BLS_ID D_ID)
RETURNS (
    IIN D_STR_20,
    GENDER D_STR_10,
    STREET D_STR_255,
    BUILDING D_STR_255,
    FLAT D_STR_255,
    COUNTRY D_STR_100,
    COUNTRY_TXT D_STR_255,
    CITY D_BLOB_TEXT,
    ID_NUMBER D_STR_20,
    ID_BEGIN_DATE D_DATE,
    ID_END_DATE D_DATE,
    ID_SURNAME D_STR_100,
    ID_NAME D_STR_100,
    ID_PATRONYMIC D_STR_100,
    ENG_SURNAME D_STR_100,
    ENG_FIRSTNAME D_STR_100,
    ID_BIRTH_DATE D_DATE,
    ISSUE_ORG_CODE D_STR_10,
    ISSUE_ORG_NAME_RU D_STR_255,
    PARENT_ID D_ID,
    DISTRICT_CODE D_STR_20,
    DISTRICT_TXT D_STR_255,
    PL_DISTRICT_ID D_ID,
    REGION_CODE D_STR_100,
    REGION_TXT D_STR_255,
    JSON_ADDRESS D_BLOB_TEXT,
    ADDR_COUNTRY_CODE D_STR_20)
AS
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable JSON_DATA D_BLOB_TEXT;
declare variable JSON_PERSON D_BLOB_TEXT;
declare variable JSON_GENDER D_BLOB_TEXT;
declare variable JSON_NATIONALITY D_BLOB_TEXT;
declare variable JSON_CITIZENSHIP D_BLOB_TEXT;
declare variable JSON_BIRTH_PLACE D_BLOB_TEXT;
declare variable JSON_BIRTH_COUNTRY D_BLOB_TEXT;
declare variable JSON_BIRTH_REGION D_BLOB_TEXT;
declare variable JSON_BIRTH_DISTRICT D_BLOB_TEXT;
declare variable JSON_ADDR_COUNTRY D_BLOB_TEXT;
declare variable JSON_ADDR_DISTRICT D_BLOB_TEXT;
declare variable JSON_ADDR_REGION D_BLOB_TEXT;
declare variable JSON_DOCUMENTS D_BLOB_TEXT;
declare variable DOC_ITEM D_BLOB_TEXT;
declare variable I integer;
declare variable TEMP_TYPE_CODE D_STR_10;
declare variable TEMP_STATUS_CODE D_STR_10;
declare variable TEMP_END_DATE D_DATE;
declare variable PL_TYPE D_INTEGER;
declare variable PL_REGION_ID D_ID;
declare variable UN_VIRT D_ID;
declare variable DOC_COUNT integer;
declare variable DOC_PATH varchar(100);
begin
    select tg.json_text
    from temp_grs tg
    where tg.bls_id = :bls_id
    into json_text;
    if (coalesce(json_text, '') = '') then
    begin
      select blx_blob
      from hp_get_blobex_blob(:bls_id)
      into json_text;
    end

    un_virt = 1000000000000;
    
    iin = f_json_str_get_value(:json_text, 'iin', '');  -- пин
    gender = f_json_str_get_value(:json_text, 'gender.code', '');  --пол (1-М/2-Ж)
    eng_surname = f_json_str_get_value(:json_text, 'engSurname', '');  -- Фамилия на латинице 
    eng_firstname = f_json_str_get_value(:json_text, 'engFirstName', '');  -- Имя на латинице 
                                    -- ADRESS --
    city = f_json_str_get_value(:json_text, 'address.city', '');  -- город
    street = f_json_str_get_value(:json_text, 'address.street', ''); -- улица
    building = f_json_str_get_value(:json_text, 'address.building', ''); -- дом
    flat = f_json_str_get_value(:json_text, 'address.flat', '');  -- квартира
    country = f_json_str_get_value(:json_text, 'address.country.code', '');  -- страна (код)
    country_txt = f_json_str_get_value(:json_text, 'address.country.name_ru', '') ;  -- страна (текст)
    district_code = f_json_str_get_value(:json_text, 'address.district.code', ''); -- область (код)
    district_txt  = f_json_str_get_value(:json_text, 'address.district.name_ru', ''); -- область (текст)
    region_code = f_json_str_get_value(:json_text, 'address.region.code', '');   -- город / регион (код)
    region_txt  = f_json_str_get_value(:json_text, 'address.region.name_ru', ''); -- город / регион (текст)

    json_address = cast(f_json_blob_get_value(:json_text, 'address', '') as d_blob_text);
                                    -- DOCUMENTS --
    json_documents = cast(f_json_blob_get_value(:json_text, 'documents', '')as d_blob_text);
    doc_count = coalesce(f_json_blob_get_count(:json_documents, 'document'),0);
    
    i = 0;
    while (i < doc_count) do
    begin
    doc_path = 'document[' || :i || ']';
    
    doc_item = cast(f_json_blob_get_value(:json_documents, :doc_path, '')as d_blob_text);
    
    if (doc_item is null or doc_item = '') then
    begin
      i = i + 1;
      continue;
    end
    
    temp_type_code   = f_json_str_get_value(:doc_item, 'type.code', '');
    temp_status_code = f_json_str_get_value(:doc_item, 'status.code', '');
    
    if (temp_type_code = '002' and temp_status_code = '00') then
    begin
      temp_end_date = cast(left(f_json_str_get_value(:doc_item, 'end_date', ''), 10)as date);
    
      if (temp_end_date is not null and temp_end_date >= current_date) then
      begin
        id_number = f_json_str_get_value(:doc_item, 'number', '');
    
        id_begin_date = cast(left(f_json_str_get_value(:doc_item, 'begin_date', ''), 10)as date);
        id_end_date = temp_end_date;
        id_birth_date = cast(left(f_json_str_get_value(:doc_item, 'birth_date', ''), 10)as date);
    
        issue_org_code = f_json_str_get_value(:doc_item, 'issue_organization.code', '');    -- Орган выдавший документ (КОД)
        issue_org_name_ru = f_json_str_get_value(:doc_item, 'issue_organization.name_ru', '');  -- Орган выдавший документ

        id_surname = f_json_str_get_value(:doc_item, 'surname', '');  -- фамилия
        id_name = f_json_str_get_value(:doc_item, 'name', '');  -- имя
        id_patronymic = f_json_str_get_value(:doc_item, 'patronymic', '');  -- отчество
    
        -- Выходим из цикла — дальнейшие документы нам не нужны
        leave;
      end
      -- если temp_end_date < current_date, документ просрочен — продолжаем цикл
    end
    -- Переход к следующему индексу
    i = i + 1;
    end


   -- проверяем наличие дочернего элемента и ес его нет добавляем
    if (addr_country_code <> '3980') then
      exception e_default 'страна не Казахстан';

    parent_id = 398000000001; -- казахстан

    if (region_code <> '0') then
    begin
      pl_type = 0;
      pl_region_id = (select pl.pl_id
                      from places pl
                      where pl.pl_code = :region_code rows 1);

      if (left(:region_txt, 2) = 'г.') then
        pl_type = 398005;
      else
        pl_type = 398002;
      if (pl_region_id is null) then
        pl_region_id = (select pl_id
                        from hp_insert_place_node(:parent_id, :region_txt, :pl_type, :un_virt, :region_code, 1));

      parent_id = :pl_region_id;
    end

    if (district_code <> '0') then
    begin
      pl_district_id = (select pl.pl_id
                        from places pl
                        where pl.pl_code = :district_code rows 1);
      pl_type = 0;
      if (left(:district_txt, 2) = 'г.') then
        pl_type = 398005;
      else
      if (right(:district_txt, 3) = 'р-н') then
        pl_type = 398003;

      if (pl_district_id is null) then
        pl_district_id = (select pl_id
                          from hp_insert_place_node(:parent_id, :district_txt, :pl_type, :un_virt, :district_code, 1));

      parent_id = :pl_district_id;
    end

  suspend;
end
