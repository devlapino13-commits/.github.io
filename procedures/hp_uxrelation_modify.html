CREATE PROCEDURE HP_UXRELATION_MODIFY(
    UFP_CODE D_STR_100 NOT NULL,
    ACTION_KIND D_STR_CODE NOT NULL,
    TABLE_NAME D_STR_CODE NOT NULL)
AS
declare variable source_name d_str_code;
declare variable field_name d_str_code;
begin
  if (action_kind not in ('delete', 'insert')) then
    exception e_default 'Undefined action_kind: ' || action_kind;

  select ufd_source, ufp_origin_name
  from uxfrparam, uxfrdata
  where ufp_code = :ufp_code and
        ufp_ufr_code = ufd_ufr_code and
        ufd_kind = 0
  into source_name, field_name;

  if (action_kind = 'insert') then
  begin
    if (not exists(select *
    from rdb$ref_constraints b,
         rdb$relation_constraints a,
         rdb$relation_constraints c,
         rdb$index_segments d,
         rdb$index_segments e,
         rdb$indices i
    where (a.rdb$constraint_type = 'FOREIGN KEY') and
          (a.rdb$constraint_name = b.rdb$constraint_name) and
          (b.rdb$const_name_uq = c.rdb$constraint_name) and
          (c.rdb$index_name = d.rdb$index_name) and
          (a.rdb$index_name = e.rdb$index_name) and
          (a.rdb$index_name = i.rdb$index_name) and
          (a.rdb$relation_name = upper(:source_name)) and
          (e.rdb$field_name = upper(:field_name)) and
          c.rdb$relation_name = upper(:table_name))) then
    begin
      update or insert into uxrelation (url_source, url_field_name, url_fk_table)
      values (:source_name, :field_name, :table_name)
      matching (url_source, url_field_name, url_fk_table);
    end
  end else
  begin
    delete from uxrelation
    where url_source = :source_name and
          url_field_name = :field_name and
          url_fk_table = :table_name;
  end
end
