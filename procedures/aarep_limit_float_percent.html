CREATE PROCEDURE AAREP_LIMIT_FLOAT_PERCENT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_FACT_DISB_DATE DATE,
    KAL_NEW_KF_NAME D_STR_100,
    CLIENT_STATUS D_STR_100,
    OVER_BALANC D_CURRENCY,
    OVERLIMIT D_CURRENCY,
    CLOSE_OVERLIMIT D_CURRENCY,
    KF_OFFICER_NAME VARCHAR(100),
    APPROVED_MANAGER_FIO STR_100)
AS
declare variable k_id d_id;
declare variable max_amount d_currency;
declare variable c_id d_id;
declare variable kol smallint; /* declare variable k_id d_id; */
declare variable kc_id d_id;
declare variable kc_amount_approved d_currency; /* declare variable oks_balance_all d_currency; */
declare variable oks_balance d_currency;
declare variable star d_currency;
declare variable c_curr_status d_smallint;
declare variable history_not_found d_smallint;
declare variable res_active_count d_integer;
declare variable res_delinq_count d_integer;
declare variable res_closed_last_week d_integer;
declare variable k_from_kf_id d_id;
declare variable k_fact_disb_lo_id d_id;
declare variable k_approved_manager d_id;
declare variable iter_k_id d_id;
declare variable sh_calc_perc d_currency;
begin
  for select r1.reg_name, k.k_kontract_no, k.k_amount_approved, k.k_id, k_fact_disb_date, k.k_from_kf_id,
             k.k_fact_disb_lo_id, k.k_approved_manager
      from kontract k
      inner join region r1 on r1.reg_unicode = k.k_last_reg_id
      inner join region_list(:rep_region_id, 1, 3, 1) on k.k_fact_disb_reg_id = id_list
      inner join product p on p.pr_id = k.k_from_pr_id
      where k.k_fact_disb_date between :rep_date_begin and :rep_date_end and
            p.pr_code containing '.FLOAT_PERCENT.' --and k.k_id =101010060891100101
      into reg_name,  k_kontract_no, k_amount_approved, k_id, k_fact_disb_date,
           k_from_kf_id, k_fact_disb_lo_id, k_approved_manager --нахожу нужного контракта
  do
  begin
    kal_new_kf_name = null;
    approved_manager_fio = '';
    kf_officer_name = '';
    select cl.c_name_fio
    from clients cl
    where cl.c_id = :k_fact_disb_lo_id
    into kf_officer_name;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :k_approved_manager
    into approved_manager_fio;

    select kal.kal_new_kf_name
    from knaktlist kal
    where :k_fact_disb_date between kal.kal_date and kal.kal_date_end and
          kal.kal_from_kf_id = :k_from_kf_id -- id кредитной папки
          and
          kal.is_delete = 0
    into kal_new_kf_name;

    c_name_fio = '';
    kc_id = null;
    c_id = null;
    kc_amount_approved = 0;

    select cl.c_name_fio, kc_id, kc_from_c_id, kc_amount_approved
    from knclient knc
    inner join clients cl on cl.c_id = knc.kc_from_c_id
    where knc.kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into c_name_fio, :kc_id, :c_id, :kc_amount_approved; --нахожу нужного клиента

    over_balanc = 0;

    max_amount = 0;
    for select coalesce(oks.oks_balance, 0)
        from knclient knc
        inner join oksummary oks on oks.oks_from_k_id = knc.kc_from_k_id
        where knc.kc_from_c_id = :c_id and
              knc.kc_amount_approved > 0 and
              :rep_date_end between oks.oks_date_begin and oks.oks_date_end
        into :oks_balance
    do
    begin
      over_balanc = over_balanc + oks_balance; --вычисляю его общий баланс
    end
    c_curr_status = 0;
    client_status = null;

    select c_curr_status
    from hp_get_client_status(:c_id, :rep_date_end)
    into :c_curr_status;

    client_status = case c_curr_status
                                      when 5 then 'Статус: *****'
                                      when 4 then 'Статус: ****'
                                      when 3 then 'Статус: ***'
                                      when 2 then 'Статус: **'
                                      when 1 then 'Статус: *'
                                      else 'Без статуса'
                                    end;
    max_amount = 0;
    max_amount = case c_curr_status
                   when 5 then 90000.00
                   when 4 then 90000.00
                   when 3 then 80000.00
                   when 2 then 60000.00
                   when 1 then 45000.00
                   else 30000
                 end;

    if (c_curr_status <= 2) then
    begin
      history_not_found = 1;
      res_active_count = 0;
      res_delinq_count = 0;
      res_closed_last_week = 0;

      select history_not_found, res_active_count, res_delinq_count, res_closed_last_week
      from hp_extinfolist_pdn_mfo_info(:kc_id)
      into :history_not_found, :res_active_count, :res_delinq_count, :res_closed_last_week;

      history_not_found = coalesce(history_not_found, 1);
      res_active_count = coalesce(res_active_count, 0);
      res_delinq_count = coalesce(res_delinq_count, 0);
      res_closed_last_week = coalesce(res_closed_last_week, 0);

      if ((res_active_count - res_delinq_count > 0) or (res_closed_last_week > 0) or
          -- S.K.
          exists(select *
                 from clientdocs
                 where cds_from_c_id = :c_id and
                       cds_from_pk_id = 337 and
                       cds_is_delete = 0 and
                       (cds_doc_blob_1 > 0 or
                       cds_doc_blob_2 > 0))) then -- если есть активные непросрчоенные займы
        max_amount = 70000;

    end

    overlimit = 0;
    close_overlimit = 0;
    if (over_balanc > max_amount) then
    begin
      overlimit = over_balanc - max_amount;

      for select oks.oks_from_k_id
          from knclient knc
          inner join oksummary oks on oks.oks_from_k_id = knc.kc_from_k_id
          where knc.kc_from_c_id = :c_id and
                knc.kc_amount_approved > 0 and
                :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
                knc.kc_from_k_id <> :k_id
          into :iter_k_id
      do
      begin
        select coalesce(sum(sh_calc_perc + sh_add_perc), 0)
        from aarep_make_zad_list(:iter_k_id, :rep_date_end, 1, 0)
        into :sh_calc_perc;

        close_overlimit = close_overlimit + sh_calc_perc;
      end
      close_overlimit = close_overlimit + overlimit;
      suspend;
    end

  end
end
