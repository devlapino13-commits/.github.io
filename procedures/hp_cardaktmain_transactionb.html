CREATE PROCEDURE HP_CARDAKTMAIN_TRANSACTIONB(
    CAM_ID D_INTEGER)
RETURNS (
    OL_ID D_ID,
    ERR_CODE D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable curr_code d_str_lang;
declare variable reg_id d_integer;
declare variable reg_name d_str_code;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable card_count d_integer;
declare variable ckd_is_own d_bool;
declare variable ckd_name d_str_code;
declare variable un_id d_id;
declare variable acc_dt1 d_str_20;
declare variable acc_kt1 d_str_20;
declare variable acc_dt2 d_str_20;
declare variable acc_kt2 d_str_20;
declare variable ol_memo d_str_255;
begin
  err_code = 0;
  err_text = '';

  curr_code = '417';
  select
    cam_card_count,
    ckd_is_own,
    ckd_name,
    un_id,
    cam_from_ol_id
  from hp_cardaktmain_view(:cam_id)
  where cam_status = 0
  into
    :card_count,
    :ckd_is_own,
    :ckd_name,
    :un_id,
    :ol_id;

  if (ol_id is null or
    exists(
      select *
      from operlist
      where ol_id = :ol_id
        and is_delete = 1)) then
  begin
    select
      reg_unicode,
      reg_name,
      c_id,
      c_name_fio
    from cardplace
      inner join clients
        on c_id = cpl_from_c_id
      inner join region
        on reg_unicode = cpl_from_reg_id
    where cpl_id = 0
    into
      :reg_id,
      :reg_name,
      :c_id,
      :c_name_fio;

    acc_dt1 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/12');
    acc_kt1 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/10');
    acc_dt2 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/10');
    acc_kt2 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/12');
    if (acc_dt1 is null or acc_kt1 is null or acc_dt2 is null or acc_kt2 is null) then
    begin
      err_code = 2;
      err_text = 'В ' || reg_name || ' не открыты счета:' ||
        iif(acc_dt1 is null, ' 96101/12', '') || iif(acc_kt1 is null, ' 96101/10', '') ||
        iif(acc_dt2 is null, ' 96102/10', '') || iif(acc_kt2 is null, ' 96102/12', '');
      suspend;
      exception e_message err_text;
    end
    ol_memo = 'Передача заготовок карт в МПЦ по АПП № ' || cam_id || ', в количестве ' || card_count || 'шт.';

    insert into operlist (
      ol_from_reg_id,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_memo,
      ol_user_id,
      ol_document_client
     )
    values (
      :reg_id,
      2,
      2,
      current_date,
      :cam_id,
      0,
      :ol_memo,
      :un_id,
      :c_id
    )
    returning ol_id
    into :ol_id;

    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_summa,
      ob_summa_val
    )
    values (
      :ol_id,
      :curr_code,
      1,
      :acc_dt1,
      :acc_kt1,
      :card_count,
      :card_count
    );

    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_summa,
      ob_summa_val
    )
    values (
      :ol_id,
      :curr_code,
      1,
      :acc_dt2,
      :acc_kt2,
      :card_count,
      :card_count
    );
    suspend;
  end else
  begin
    err_code = 1;
    err_text = 'AПП заготовок карт № ' || cam_id || ' не может быть проведен.';
    suspend;
    exception e_message err_text;
  end
end
