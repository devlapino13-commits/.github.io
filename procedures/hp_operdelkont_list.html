CREATE PROCEDURE HP_OPERDELKONT_LIST(
    IN_ODM_ID D_INTEGER)
RETURNS (
    ODK_ID D_INTEGER,
    ODK_FROM_ODM_ID D_INTEGER,
    ODK_LOAN_OL_ID D_ID,
    ODK_FROM_K_ID D_ID,
    ODK_AMOUNT D_CURRENCY,
    RES_KONTRACT_NO D_STR_CODE,
    RES_INFO D_STR_255,
    RES_OL_DATE D_DATE,
    RES_OL_MEMO D_STR_LONG,
    ODM_STATE D_SMALLINT,
    RES_CHECK_TEXT D_STR_255,
    RES_CHECK_IS_ERROR D_SMALLINT,
    RES_NEW_AMOUNT D_CURRENCY)
AS
declare variable odm_main_ol_id d_id;
begin
  select odm_main_ol_id, odm_state
  from operdelmain
  where odm_id = :in_odm_id
  into odm_main_ol_id, odm_state;

  res_check_is_error = 0;
  res_check_text = '';

  select res_text
  from hp_operdelmain_check(:in_odm_id)
  into :res_check_text;

  if (res_check_text <> '') then
    res_check_is_error = 1;

  for select odk_id, odk_from_odm_id, odk_loan_ol_id, odk_from_k_id, odk_amount,
             k_kontract_no, ol_date, ol_memo
  from operdelkont
       inner join kontract on (odk_from_k_id = k_id)
       left join operlist on (odk_loan_ol_id = ol_id)
  where odk_from_odm_id = :in_odm_id
  into :odk_id, :odk_from_odm_id, :odk_loan_ol_id, :odk_from_k_id,
       :odk_amount,
       :res_kontract_no, res_ol_date, res_ol_memo
  do
  begin
    res_info = 'Основной';

    if (odk_loan_ol_id is null) then
      res_info = 'Новый';

    res_new_amount = 0;
    select odl_new_amount
    from operdellist
    where odl_from_odm_id = :in_odm_id and
          odl_from_k_id = :odk_from_k_id and
          odl_main_ol_id = :odm_main_ol_id
    into res_new_amount;

    if (res_new_amount is null) then
      res_new_amount = 0;

    suspend;
  end
end
