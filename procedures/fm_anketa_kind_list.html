CREATE PROCEDURE FM_ANKETA_KIND_LIST(
    UN_ID D_ID)
RETURNS (
    RD_ID D_INTEGER,
    RD_NAME D_STR_255,
    RD_KIND D_INTEGER,
    RD_COUNT D_INTEGER)
AS
declare variable c_id d_id;
declare variable manager_c_id d_id;
declare variable db_id d_id;
declare variable reg_id d_id;
begin
  db_id = f_get_db_id();

    -- Возвращает список типов анкет пользователя с их количеством
  if (db_id = 41701) then    -- RFCKG-3667 --

  begin
    select un_from_c_id
    from username
    where un_id = :un_id
    into :c_id;

    select ps.ps_from_c_id
    from hr_pstatus ps
    inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
    inner join jobkind jk on jl.jl_kind = jk.jk_id
        and jk.jk_system_code = 'JobSystem.Region_KM'
    where current_date between ps.ps_date_begin and ps.ps_date_end
      and ps.ps_from_c_id = :c_id
      and ps.ps_is_on_staff = 1
      and ps.ps_is_working = 1
      and ps.ps_status = 1
    into :manager_c_id;

    -- RFCKG-4460
    if (manager_c_id is not null) then
    begin
      select ps.ps_from_reg_id
      from hr_pstatus ps
      where ps.ps_from_c_id = :manager_c_id
        and ps.ps_is_on_staff = 1
        and ps.ps_is_working = 1
        and ps.ps_status = 1
        and current_date between ps.ps_date_begin and ps.ps_date_end
      rows 1
      into :reg_id;
    end
  end

  for
    with t as (
      select
        rd_id,
        rd_name,
        iif(rd_str_id = 'Anketa', 0, 1) as rd_kind,
        count(1) as rd_count
      from actionorder
      inner join refdata
        on rd_id = aco_from_rd_id
        and rd_main_code = 'ActionGroup'
        and rd_str_id in ('Anketa', 'CheckAnketa', 'UnderWriter_CheckLoanFix')
        and rd_is_active = 1
      inner join hp_staff_changer_list(:un_id, 1)
        on res_c_id = aco_respons_c_id
      where aco_object_kind in (
        f_objkind('Clients','id'),
        f_objkind('KnClient','id'),
        f_objkind('MiniAppClient','id'),
        f_objkind('Kontract', 'id'))
      group by rd_id, rd_name, rd_str_id
    ),

    -- RFCKG-4460
    manager_tasks as (
      select
        rd.rd_id,
        rd.rd_name,
        1 as rd_kind,
        count(*) as rd_count
      from actionorder ao
      inner join refdata rd
        on rd.rd_id = ao.aco_from_rd_id
        and rd.rd_main_code = 'ActionGroup'
        and rd.rd_id in (8941, 1622)
        and rd.rd_is_active = 1
      inner join username un
        on un.un_from_c_id = ao.aco_respons_c_id
      inner join hr_pstatus ps
        on ps.ps_from_c_id = un.un_from_c_id
        and ps.ps_is_on_staff = 1
        and ps.ps_is_working = 1
        and ps.ps_status = 1
        and current_date between ps.ps_date_begin and ps.ps_date_end
      inner join joblist jl
        on jl.jl_id = ps.ps_from_jl_id
      inner join jobkind jk
        on jl.jl_kind = jk.jk_id
        and jk.jk_system_code = 'JobSystem.LO_Standart'
      where :manager_c_id is not null
        and ps.ps_from_reg_id = :reg_id
        and ao.aco_object_kind in (
          f_objkind('Clients','id'),
          f_objkind('KnClient','id'),
          f_objkind('MiniAppClient','id'),
          f_objkind('Kontract','id'))
      group by rd.rd_id, rd.rd_name
    ),

    -- RFCKG-4460
    all_data as (
      select
        0 as rd_id,
        'Все задачи' as rd_name,
        -1 as rd_kind,
        (select sum(rd_count) from t) +
        (select coalesce(sum(rd_count), 0) from manager_tasks) as rd_count
      from rdb$database
      union all
      select
        rd_id,
        rd_name,
        rd_kind,
        rd_count
      from t
      union all
      select
        rd_id,
        rd_name,
        rd_kind,
        rd_count
      from manager_tasks
    )

    select
      rd_id,
      rd_name,
      rd_kind,
      rd_count
    from all_data
    into :rd_id, :rd_name, :rd_kind, :rd_count
  do
  begin
    suspend;
  end
end
