CREATE PROCEDURE HP_CLIENT_UN_IS_ACTIVE(
    C_ID DECIMAL(18,0))
RETURNS (
    UN_ID DECIMAL(18,0),
    UN_IS_ACTIVE INTEGER,
    UN_NAME VARCHAR(30))
AS
begin
  un_is_active = 0;
  select un_id
  from username
  where un_from_c_id = :c_id rows 1
  into :un_id;

  select un_is_active, un_name
  from hp_un_is_active(:un_id)
  into :un_is_active, :un_name;

  suspend;
end
