CREATE PROCEDURE HP_SEPARATE_FIO_CHECKVALID(
    C_ID D_ID,
    FNAME D_STR_100 = null,
    MNAME D_STR_100 = null,
    SNAME D_STR_100 = null,
    FNAME_ENG D_STR_100 = null,
    MNAME_ENG D_STR_100 = null,
    SNAME_ENG D_STR_100 = null,
    FULL_FIO_LAT D_STR_100 = null)
RETURNS (
    IS_CORRECT BOOLEAN,
    ERR_TXT D_STR_8100)
AS
declare variable is_check boolean;
declare variable c_type d_integer;
declare variable div_ch d_str_lang;
begin
  is_correct = true;
  err_txt = '';
  div_ch = ascii_char(13);

  --suspend;
  --exit;

  if (c_id < 0) then
     select
       c_type,
       fl1.fl_name2,
       fl1.fl_name3,
       fl1.fl_name1,
       fl2.fl_name2,
       fl2.fl_name3,
       fl2.fl_name1,
       c_alternat_fio
     from clients
       left join fiolist as fl1
         on fl1.fl_id = c_from_fl_id
       left join fiolist as fl2
         on fl2.fl_id = c_name_eng_fl_id
     where c_id = -:c_id
     into
       :c_type,
       :fname,
       :mname,
       :sname,
       :fname_eng,
       :mname_eng,
       :sname_eng,
       :full_fio_lat;
  else
    c_type = (select c_type from clients where c_id = :c_id);

  fname = trim(coalesce(fname, ''));
  mname = trim(coalesce(mname, ''));
  sname = trim(coalesce(sname, ''));
  fname_eng = trim(coalesce(fname_eng, ''));
  mname_eng = trim(coalesce(mname_eng, ''));
  sname_eng = trim(coalesce(sname_eng, ''));
  full_fio_lat = trim(coalesce(full_fio_lat, ''));

  if (fname <> '') then
  begin
    is_check = (select result from check_regular_exp(:fname, '[А-ЯЁ№()".[:SPACE:]' || trim(iif(:c_type = 0, '', '[:DIGIT:]')) || '-]+'));
    if (not is_check) then
    begin
      is_correct = false;
      err_txt = err_txt || iif(err_txt = '', '', div_ch) || f_format('  - Имени: "%1";', fname);
    end
  end

  if (mname <> '') then
  begin
    is_check = (select result from check_regular_exp(:mname, '[А-ЯЁ№()".[:SPACE:]' || trim(iif(:c_type = 0, '', '[:DIGIT:]')) || '-]+'));
    if (not is_check) then
    begin
      is_correct = false;
      err_txt = err_txt || iif(err_txt = '', '', div_ch) || f_format('  - Отчестве: "%1";', mname);
    end
  end

  if (sname <> '') then
  begin
    is_check = (select result from check_regular_exp(:sname, '[А-ЯЁ№()".[:SPACE:]' || trim(iif(:c_type = 0, '', '[:DIGIT:]')) || '-]+'));
    if (not is_check) then
    begin
      is_correct = false;
      err_txt = err_txt || iif(err_txt = '', '', div_ch) || f_format('  - Фамилии: "%1";', sname);
    end
  end

  if (fname_eng <> '') then
  begin
    is_check = (select result from check_regular_exp(:fname_eng, '[A-Z№()".[:SPACE:]' || trim(iif(:c_type = 0, '', '[:DIGIT:]')) || '-]+'));
    if (not is_check) then
    begin
      is_correct = false;
      err_txt = err_txt || iif(err_txt = '', '', div_ch) || f_format('  - Имени на английском: "%1";', fname_eng);
    end
  end

  if (mname_eng <> '') then
  begin
    is_check = (select result from check_regular_exp(:mname_eng, '[A-Z№()".[:SPACE:]' || trim(iif(:c_type = 0, '', '[:DIGIT:]')) || '-]+'));
    if (not is_check) then
    begin
      is_correct = false;
      err_txt = err_txt || iif(err_txt = '', '', div_ch) || f_format('  - Отчестве на английском: "%1";', mname_eng);
    end
  end

  if (sname_eng <> '') then
  begin
    is_check = (select result from check_regular_exp(:sname_eng, '[A-Z№()".[:SPACE:]' || trim(iif(:c_type = 0, '', '[:DIGIT:]')) || '-]+'));
    if (not is_check) then
    begin
      is_correct = false;
      err_txt = err_txt || iif(err_txt = '', '', div_ch) || f_format('  - Фамилии на английском: "%1";', sname_eng);
    end
  end

  if (full_fio_lat <> '') then
  begin
    is_check = (select result from check_regular_exp(:full_fio_lat, '[A-Z№()".[:SPACE:]' || trim(iif(:c_type = 0, '', '[:DIGIT:]')) || '-]+'));
    if (not is_check) then
    begin
      is_correct = false;
      err_txt = err_txt || iif(err_txt = '', '', div_ch) || f_format('  - ФИО на латинице: "%1";', full_fio_lat);
    end
  end

  if (not is_correct) then
    err_txt = 'Найдены недопустимые символы в' || div_ch || err_txt;

  suspend;
end
