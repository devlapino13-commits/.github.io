CREATE PROCEDURE HP_JSON_ACTION_SAVE_OPERLIST(
    JA_NAME D_STR_CODE,
    ID D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    RES_ID D_STR_CODE,
    RES_ERROR_TEXT D_STR_255,
    OL_DATE D_DATE,
    OL_FROM_REG_ID D_INTEGER,
    OL_IS_ACTIVE D_SMALLINT)
AS
declare variable ja_json d_str_8100;
declare variable ol_doc_no_str d_str_code;
declare variable ol_doc_no d_integer;
begin
  if (coalesce(id, '') = '') then
    id = '0';
  res_id = id;
  res_error_text = '';


  ol_is_active = 1;

  for select ja_json
      from temp_json_action
      where ja_name = :ja_name
      into :ja_json
  do
  begin
    for select f_json_str_get_value(:ja_json, 'ol_from_reg_id',null) as ol_from_reg_id,
               f_json_str_get_value(:ja_json, 'ol_date',null) as ol_date,
               f_json_str_get_value(:ja_json, 'ol_doc_no_str','') as ol_doc_no_str,
               f_json_str_get_value(:ja_json, 'ol_doc_kind',null) as ol_doc_kind,
               f_json_str_get_value(:ja_json, 'ol_oper_type',null) as ol_oper_type,
               f_json_str_get_value(:ja_json, 'ol_oper_subtype',null) as ol_oper_subtype,
               f_json_str_get_value(:ja_json, 'ol_document_client',null) as ol_document_client,
               f_json_str_get_value(:ja_json, 'ol_doc_client_name','') as ol_doc_client_name,
               f_json_str_get_value(:ja_json, 'ol_memo',null) as ol_memo
        from rdb$database as cursor ol
    do
    begin
      ol_date = ol.ol_date;
      ol_from_reg_id = ol.ol_from_reg_id;
      ol_doc_no_str = ol.ol_doc_no_str;
      ol_doc_no = null;
      if (coalesce(ol_doc_no_str, '') <> '') then
      begin
        ol_doc_no = ol_doc_no_str;
        when any do
          ol_doc_no = null;
      end

      if (ol_date is null) then
        res_error_text = 'Дата проводки не указана';

      if (coalesce(ol.ol_oper_type, 0) = 0) then
        res_error_text = 'Неверный тип операции';

      if (coalesce(ol.ol_oper_subtype, 0) = 0) then
        res_error_text = 'Неверный субтип операции';

      if (coalesce(ol_from_reg_id, 0) = 0) then
        res_error_text = 'Офис проведения не указан';

      if (res_error_text <> '') then
      begin
        suspend;
        exit;
      end

      if (id <> '0') then
      begin
        update operlist
        set ol_from_reg_id = ol.ol_from_reg_id,
            ol_date = ol.ol_date,
            ol_doc_no = :ol_doc_no,
            ol_doc_no_str = iif(coalesce(ol_doc_no_str, '') = coalesce(ol_doc_no, ''), ol.ol_doc_no_str, ol_doc_no_str),
            ol_oper_type = ol.ol_oper_type,
            ol_oper_subtype = ol.ol_oper_subtype,
            ol_memo = ol.ol_memo,
            ol_doc_kind = ol.ol_doc_kind,
            ol_document_client = ol.ol_document_client,
            ol_doc_client_name = ol.ol_doc_client_name
        where ol_id = :id
        returning 1 - is_delete
        into ol_is_active;
      end
      else
      begin
        if (ol_doc_no is null and
            coalesce(ol_doc_no_str, '') = '') then
        begin
          select num
          from last_doc_number(:ol_date, ol.ol_doc_kind, :ol_from_reg_id)
          into ol_doc_no;
          ol_doc_no_str = ol_doc_no;
        end

        insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo, ol_user_id,
                              ol_oper_subtype, ol_document_client, ol_doc_client_name)
        values (ol.ol_from_reg_id, ol.ol_oper_type, ol.ol_date, :ol_doc_no, :ol_doc_no_str, ol.ol_doc_kind, ol.ol_memo, :un_id,
                ol.ol_oper_subtype, ol.ol_document_client, ol.ol_doc_client_name)
        returning ol_id, 1 - is_delete
        into :id, :ol_is_active;
      end
    end
  end

  res_id = id;
  suspend;
end
