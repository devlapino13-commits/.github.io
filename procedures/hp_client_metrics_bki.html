declare function f_my_fix_wrong_lat(in_str d_str_255)

      res_text = replace(res_text collate unicode, 'Е' collate unicode, 'E');
      res_text = replace(res_text collate unicode, 'Г' collate unicode, 'F');
      res_text = replace(res_text collate unicode, 'Н' collate unicode, 'H');
      res_text = replace(res_text collate unicode, 'К' collate unicode, 'K');

declare procedure hp_my_split_fio(c_name_fio d_str_255)
  returns(
    name_last d_str_255,
    name_first d_str_255,
    name_middle d_str_255
  )
  as
  begin
    select lower(txt1), lower(txt2), lower(txt3)
    from hp_fio_clear(:c_name_fio)
    into :name_last, :name_first, :name_middle;
  
    name_first =coalesce(name_first, ''); -- Имя
    name_last = coalesce(name_last, '');  -- фамилия
    name_middle = coalesce(name_middle, ''); -- Отчетво
  
    if (c_name_fio like '% уулу %') then name_last = name_last || ' уулу';
    if (c_name_fio like '% кызы %') then name_last = name_last || ' кызы';
    if (c_name_fio like '% uulu %') then name_last = name_last || ' uulu';
    if (c_name_fio like '% kyzy %') then name_last = name_last || ' kyzy';
    if (c_name_fio like '% угли %') then name_last = name_last || ' угли';
    if (c_name_fio like '% кизи %') then name_last = name_last || ' кизи';
    if (c_name_fio like '% qizi %') then name_last = name_last || ' qizi';
    if (c_name_fio like '% oglu %') then name_last = name_last || ' oglu';
    if (c_name_fio like '% ugli %') then name_last = name_last || ' ugli';
  
    if (name_first <> '') then
        name_first = upper(substring(name_first from 1 for 1))||substring(name_first from 2);
  
    if (name_last <> '') then
        name_last = upper(substring(name_last from 1 for 1))||substring(name_last from 2);
  
    if (name_middle <> '') then
        name_middle = upper(substring(name_middle from 1 for 1))||substring(name_middle from 2);

    name_first = replace(name_first, ascii_char(10) ,'');
    name_first = replace(name_first, ascii_char(13) ,'');
  
    suspend;
  end

begin
  if (coalesce(delimeter, '') = '') then
    delimeter = ';';

  loan_count_bki = 0;


  select kc_from_c_id, k_from_ma_app_id
  from knclient, kontract
  where kc_id = :kc_id
    and k_id = kc_from_k_id
  into c_id, ma_id;

  jso = rdb$get_context('USER_TRANSACTION', 'bki_ki_jso_for_c_id_' || c_id);
  eil_id = rdb$get_context('USER_TRANSACTION', 'bki_ki_eil_for_c_id_' || c_id);

  if (jso is null) then
  begin
    select ki_result_id, ki_result, ki_jso, ssp_jso, ki_eil_id, ssp_eil_id
    from hp_pdn_get_request_status(:c_id, null, :calc_date, :ma_id)
    into ki_result_id, res_text, jso, ssp_jso, eil_id, ssp_eil_id;
  
    criteria = 'БКИ.';
    bki_text = null;
    db_text = null;
    critical_level = 2;
  
    if (ki_result_id not in (5,9)) then
    begin
      suspend;
      exit;
    end 
  
    rdb$set_context('USER_TRANSACTION', 'bki_ssp_jso_for_c_id_' || c_id, ssp_jso);
    rdb$set_context('USER_TRANSACTION', 'bki_ki_jso_for_c_id_' || c_id, jso);
    rdb$set_context('USER_TRANSACTION', 'bki_ki_eil_for_c_id_' || c_id, eil_id);
    rdb$set_context('USER_TRANSACTION', 'bki_ssp_eil_for_c_id_' || c_id, ssp_eil_id);
  end
  else
    ki_result_id = iif(json.obj_get_val(jso, 'responsecode') = 3, 5, 9);

  birth_bki = '';
  fio_bki_index = '#';
  fio_bki = '';
  main_doc_bki = '';
  doc_bki_index = '#';
  hist_doc_bki = '';
  addr_bki = '';
  addr_bki_index = '#';
  phone_bki = '';
  phone_bki_index = '#';
  loan_count_bki = 0;

  if (ki_result_id <> 5) then -- если не "клиент не найден"
  begin
    title_jso = coalesce(json.obj_get_obj(jso, 'response.titlePart._private'), json.obj_create());
  
    tmp_jso = json.obj_get_arr(title_jso, 'birth');
  
    if (tmp_jso is not null and json.count_get(tmp_jso) > 0) then
      birth_bki = json.obj_get_val(json.arr_get_obj(tmp_jso, 0), 'date', null);
  
    line = json.obj_get_val(title_jso, 'name.last', '') || ' ' || json.obj_get_val(title_jso, 'name.first', '');
    fio_bki_index = '#' || replace(line, ' ', '') || '#';
    -- сверка без учета отчества
    fio_bki = line || ' ' || json.obj_get_val(title_jso, 'name.middle', '');
  
    for select jso
    from hp_bki_json_get_list(json.obj_get_arr(:title_jso, 'history.name'), 1)
    into tmp_jso
    do
    begin
      line = json.obj_get_val(tmp_jso, 'last', '') || ' ' || json.obj_get_val(tmp_jso, 'first', '');
      fio_bki_index = fio_bki_index || replace(line, ' ', '') || '#';
      -- сверка без учета отчества
      fio_bki = fio_bki || delimeter || line || ' ' || json.obj_get_val(tmp_jso, 'name.middle', '');
    end
  
    line = json.obj_get_val(title_jso, 'doc.serial', '') || ' ' || json.obj_get_val(title_jso, 'doc.number', '');
    main_doc_bki = line;
    line_index = replace(line, ' ', '');
    if (line_index <> '') then
      doc_bki_index = '#' || line_index || '#';
  
    for select jso
    from hp_bki_json_get_list(json.obj_get_arr(:title_jso, 'history.doc'), 1)
    into tmp_jso
    do
    begin
      line = json.obj_get_val(tmp_jso, 'serial', '') || ' ' || json.obj_get_val(tmp_jso, 'number', '');
  
      line_index = replace(line, ' ', '');
      if (line_index = '') then
        continue;
  
      hist_doc_bki = hist_doc_bki || delimeter || line;
      doc_bki_index = doc_bki_index || line_index || '#';
    end
  
    for select jso
    from hp_bki_json_get_list(json.obj_get_arr(:jso, 'response.basePart.addrReg'), 1)
    union
    select jso
    from hp_bki_json_get_list(json.obj_get_arr(:jso, 'response.basePart.addrFact'), 1)
    into tmp_jso
    do
    begin
      okato = json.obj_get_val(tmp_jso, 'okato', '');
  
      line = '';
      if (okato <> '') then
        select pl_name || ', '
        from places
        where pl_add_code = :okato
        order by pl_datetime desc
        rows 1
        into line;
  
      line = line ||
             json.obj_get_val(tmp_jso, 'otherStatement', '')  || ', ' ||
             json.obj_get_val(tmp_jso, 'street', '')  || ', ' ||
             json.obj_get_val(tmp_jso, 'house', '')  || ', ' ||
             json.obj_get_val(tmp_jso, 'domain', '')  || ', ' ||
             json.obj_get_val(tmp_jso, 'block', '')  || ', ' ||
             json.obj_get_val(tmp_jso, 'build', '')  || ', ' ||
             json.obj_get_val(tmp_jso, 'apartment', '');
  
      line_index = replace(replace(line, ' ', ''), ',', '');
      if (line_index = '') then
        continue;
  
      if (addr_bki_index not containing ('#' || line_index || '#')) then
      begin
        if (char_length(addr_bki) + char_length(line) > 8080) then
           leave;
  
        addr_bki = iif(addr_bki <> '', addr_bki || delimeter , '') || line;
        addr_bki_index = addr_bki_index || line_index || '#';
      end
    end
  
    for select jso
    from hp_bki_json_get_list(json.obj_get_arr(:jso, 'response.basePart.contacts.phone'), 1)
    into tmp_jso
    do
    begin
      line = json.obj_get_val(tmp_jso, 'number', '');
  
      if (line = '') then
        continue;
      line = replace(line, ' ', '');
      phone_bki = phone_bki || delimeter || line;
      phone_bki_index = phone_bki_index || line || '#';
      phone_bki_index = phone_bki_index || '7' || line || '#'; -- кто-то отправляет без кода страны
    end
  
    for select jso
    from hp_bki_json_get_list(json.obj_get_arr(:jso, 'response.basePart.contract'), 1)
    into tmp_jso
    do
    begin
      uid = json.obj_get_val(tmp_jso, 'uid.id', '');
      creddate = json.obj_get_val(tmp_jso, 'deal.date', '01.01.1900');
      credenddatefact = replace(json.obj_get_val(tmp_jso, 'contractEnd.date', '01.01.1900'), '-', '01.01.1900');
  
      if ((char_length(uid) <> 36 or not exists(select obj_id from hp_find_cbr_uuid(:uid))) -- не имеет uid или не наш uid
        and (creddate + 7) > calc_date -- за последние 7 дней
        and credenddatefact = '01.01.1900') -- не закрыт
      then
        loan_count_bki = loan_count_bki + 1;
    end
  end

  ------------------------------
  criteria = 'БКИ. ФИО.';
  criteria_code = 'bki_fio';
  critical_level = 1;
  bki_text = fio_bki;
  db_text = '';

  for select f_my_fix_wrong_cyr(c_name_fio)
    from clients
    where c_id = :c_id
  union
  select f_my_fix_wrong_lat(c_name_eng)
    from clients
    where c_id = :c_id
  union
  select f_my_fix_wrong_cyr(f_docrecog_value(cds.cds_from_drl_id, 'client_name_old'))
    from clientdocs cds
    where cds_from_pk_id = 66 -- Свидетельство о перемене Фамилии, Имени, Отчества
      and cds_from_c_id = :c_id
      and cds_is_delete = 0
  union
  select f_my_fix_wrong_lat(f_docrecog_value(cds.cds_from_drl_id, 'client_name_old_eng'))
    from clientdocs cds
    where cds_from_pk_id = 66
      and cds_from_c_id = :c_id
      and cds_is_delete = 0
  into :line
  do
  begin
    select name_last, name_first, name_middle
    from hp_fio_clear_ru(:line)
    into :name_last, :name_first, :name_middle;

    db_text = iif(db_text <> '', db_text || delimeter, '') || name_last || ' ' || name_first || ' ' || name_middle;
    -- сверка без учета отчества
    if (fio_bki_index containing ('#' || replace(name_last || name_first, ' ', '') || '#')) then
      critical_level = 0;
  end

  res_text = iif(critical_level > 0, 'нет совпадений', 'есть совпадение');
  antifrod_res = iif(critical_level > 0, 1, 0);
  antifrod_res_tmp = antifrod_res;
  suspend;
  --------------------------------
  criteria = 'БКИ. Дата рождения.';
  criteria_code = 'bki_birthdate';
  bki_text = birth_bki;
  db_text = '';

  select f_datetostr(c_birth_date)
  from clients
  where c_id = :c_id
  into db_text;

  critical_level = iif(db_text is distinct from bki_text, 2, 0);
  res_text = iif(critical_level > 0, 'нет совпадений', 'есть совпадение');
  antifrod_res = iif(critical_level > 0, 1, 0);
  antifrod_res_tmp = minvalue(antifrod_res_tmp, antifrod_res);
  suspend;
  ------------------------------
  criteria = 'БКИ. Итог: ФИО + Дата рождения.';  -- нужно для отчета Антифрод
  criteria_code = 'bki_fio_birthdate';
  critical_level = 0;
  bki_text = null;
  db_text = null;
  antifrod_res = antifrod_res_tmp;
  res_text = iif(antifrod_res > 0, 'нет совпадений', 'есть совпадение');
  suspend;
  ------------------------------
  criteria = 'БКИ. ДУЛ.';
  criteria_code = 'bki_doc';
  critical_level = 2; -- по просьбе Ильи убираю критичность (было 2)
  bki_text = hist_doc_bki;
  db_text = '';
  main_doc_db = '';
  antifrod_res = 0;

  for select distinct f_my_fix_wrong_lat(cds_index_doc_no)
  from clientdocs cds
  where cds.cds_from_c_id = :c_id
    and cds_is_main_doc = 1
    and cds_index_doc_no <> ''
  order by cds_is_delete, cds_create_datetime desc
  into line
  do
  begin
    if (replace(line, ' ', '') = '') then
      continue;

    if (main_doc_db = '') then
    begin
      main_doc_db = line;
    end
    else
      db_text = db_text || delimeter || line;

    if (doc_bki_index containing ('#' || replace(line, ' ', '') || '#')) then
      critical_level = 0;
  end

  if (delimeter starting with '<') then
  begin
    format = f_format('<b><font size = +1 color = "%1">%2</font></b>%3',
                      iif(replace(main_doc_bki, ' ', '') = replace(main_doc_db, ' ', ''), '#006600'/*зеленый*/, '#990000'/*красный*/),
                      '%1');

    db_text = f_format(format, main_doc_db) || db_text;
    bki_text = f_format(format, main_doc_bki) || bki_text;
  end

  res_text = iif(critical_level > 0, 'нет совпадений', 'есть совпадение');
  antifrod_res = iif(critical_level > 0, 1, 0);
  critical_level = 0; -- по просьбе Ильи убираю критичность для КК
  suspend;
  ------------------------------
  criteria = 'БКИ. Адреса.';
  criteria_code = 'bki_address';
  critical_level = 1;
  bki_text = addr_bki;
  db_text = '';

  a_other_statement = null; --  Иной населенный пункт
  a_street = null;     -- Улица
  a_house = null;      -- Дом
  a_block = null;      -- Корпус
  a_build = null;      -- Строение
  a_apartment = null;  -- Квартира
  line_tmp = '';

  for select adl_flat_no, adl_depart_no, adl_section_no,
    adl_building_no, adl_street_id, adl_from_pl_id,
    substring(cast(adl_id as str_255) from 1 for 3)
  from addrclient, addrlist
  where adl_id = adc_from_adl_id
    and adc_from_c_id = :c_id
    and adc_address_type in (1, 2)
  union
  select adl_flat_no, adl_depart_no, adl_section_no,
    adl_building_no, adl_street_id, adl_from_pl_id,
    substring(cast(adl_id as str_255) from 1 for 3)
  from addrclog, addrloglist
  where adl_id = adcl_from_adl_id
    and adcl_from_c_id = :c_id
    and adcl_address_type in (1, 2)
  into a_apartment, a_house, a_block, a_build, adl_street_id,
       adl_from_pl_id, republ_code
  do
  begin
    if (a_apartment = '0') then a_apartment = null;
    if (a_house = '0') then a_house = null;
    if (a_block = '0') then a_block = null;
    if (a_build = '0') then a_build = null;

    a_country = (select rd2.rd_memo
                from refdata rd1, refdata rd2
                where rd1.rd_str_id = :republ_code
                  and rd1.rd_main_code = 'Citizen'
                  and rd2.rd_main_code = 'CountryOKSM'
                  and rd2.rd_name = case rd1.rd_name
                                        when 'Кыргызстан' then 'Киргизия'
                                        when 'Республика Молдова' then 'Молдова, Республика'
                                        else rd1.rd_name
                                     end);

    a_country_text = '';
    if (coalesce(a_country, '') = '') then
    begin
      a_country = '999';
      a_country_text = (select rd_name
                        from refdata rd
                        where rd_str_id = :republ_code
                          and rd_main_code = 'Citizen');
  
      if (coalesce(a_country_text, '') = '') then
        a_country_text = '-';
     end 

    select pl_name
    from places
    where pl_id = :adl_from_pl_id
    into a_other_statement;
  
    select pl_name
    from places
    inner join placetype on pl_status = plt_id
    where pl_id = :adl_street_id
    into a_street;
  
    select out_string
    from hp_check_characters_str(:a_street, '[а-яА-ЯЁё0-9- ]*')
    into a_street;

    line = coalesce(a_other_statement || ', ', '') ||
           coalesce(a_street || ', ', '') ||
           coalesce(a_house || ', ', '') ||
           coalesce(a_block || ', ', '') ||
           coalesce(a_build || ', ', '') ||
           coalesce(a_apartment, '');

    line_index = replace(replace(line, ' ', ''), ',', '');

    if (line_index = '' or line_index is not distinct from line_tmp) then
      continue;

    line_tmp = line_index;
    db_text = iif(db_text <> '', db_text || delimeter, '') || line;
    if (addr_bki_index containing ('#' || line_index || '#')) then
      critical_level = 0;
  end

  res_text = iif(critical_level > 0, 'нет совпадений', 'есть совпадение');
  antifrod_res = iif(critical_level > 0, 1, 0);
  suspend;

  ------------------------------
  criteria = 'БКИ. Номер телефона:';
  criteria_code = 'bki_phone';
  critical_level = 1;
  bki_text = phone_bki;
  db_text = '';

  for select distinct coalesce(f_cont_code(cda_from_clt_id), '')
  from contdata cda, contlog clg
  where clg_from_cda_id = cda_id
    and cda_from_c_id = :c_id
    and cda_cont_kind = f_enum('cont_kind', 'cell', 'id')
    and cda_is_main = 1
    and cda_status = 1
    and clg_datetime < :calc_date + 1
  into line
  do
    line = replace(line, ' ', '');

    if (line <> '') then
    begin
      db_text = db_text || delimeter || line;

      if (phone_bki_index containing line) then
        critical_level = 0;
    end

  res_text = iif(critical_level = 0, 'Есть совпадение', 'Нет совпадения');
  antifrod_res = iif(critical_level = 0, 0, 1);

  suspend;
  ------------------------------
  criteria = 'БКИ. Количество активных займов не в Компании за 7 дней:';
  criteria_code = 'bki_activ_other_7';
  res_text = loan_count_bki;
  critical_level = iif(loan_count_bki > 1, 2, 0);
  antifrod_res = iif(loan_count_bki > 1, 1, 0);
  bki_text = null;
  db_text = null;

  suspend;
  ------------------------------
end
