CREATE PROCEDURE INN_KK_INFO(
    C_ID D_ID)
RETURNS (
    HTML_TXT D_BLOB_TEXT)
AS
declare variable C_NAME_FIO D_STR_100;
declare variable C_BIRTH_DATE D_DATE;
declare variable EIL_CREATE_DATETIME D_DATETIME;
declare variable ICQ_CREATION_STAGE D_STR_CODE;
declare variable ICQ_FNAME D_STR_CODE;
declare variable ICQ_SNAME D_STR_CODE;
declare variable ICQ_PATRONIMIC D_STR_CODE;
declare variable ICQ_DOC_TYPE D_SMALLINT;
declare variable ICQ_DOC_NO D_STR_20;
declare variable TOTAL_STATUS D_STR_CODE;
declare variable TOTAL_STATUS_COLOR D_STR_20;
declare variable ICQ_ID D_ID;
declare variable COUNTER D_INTEGER;
declare variable TOTAL_STATUS_ID D_INTEGER;
declare variable EIR_ANSWER D_STR_255;
declare variable EIL_STATUS D_INTEGER;
declare variable ICQ_DOC_DATE D_DATE;
declare variable request_html_txt D_BLOB_TEXT;
declare variable icm_inn_received d_bool;
declare variable icq_info_source d_smallint;
declare variable eil_status_datetime d_datetime;
begin
  html_txt = '<body>';
  counter = 0;

  select
      c_name_fio
  from clients
  where c_id = :c_id
  into :c_name_fio;

  select icm.icm_inn_received
  from inn_check_main icm
  where icm.icm_from_c_id = :c_id
  into icm_inn_received;

  for select
         icq.icq_id,
         eil.eil_create_datetime,
         f_enum('inn_request_source', icq.icq_creation_stage, 'name'),
         icq.icq_fname,
         icq.icq_sname,
         icq.icq_patronimic,
         icq.icq_datebirth,
         icq.icq_doc_type,
         icq.icq_doc_no,
         icq.icq_doc_date,
         icq.icq_info_source,
         eil.eil_status_datetime
  from inn_check_queries icq
  inner join extinfolist eil on icq.icq_from_eil_id = eil.eil_id
  where icq.icq_from_c_id = :c_id
  order by icq.icq_id desc
  into icq_id,
       eil_create_datetime,
       icq_creation_stage,
       icq_fname,
       icq_sname,
       icq_patronimic,
       c_birth_date,
       icq_doc_type,
       icq_doc_no,
       icq_doc_date,
       icq_info_source,
       eil_status_datetime
  do
  begin
    counter = counter + 1;
    icq_info_source = coalesce(icq_info_source, 0);

    select
        case total_status
            when 0 then 'в процессе'
            when 1 then 'обработан, ИНН не найден'
            when 2 then 'обработан, ИНН найден'
            else 'ошибка определения '
        end,
        case total_status
            when 0 then 'blue'
            when 2 then 'green'
            else 'red'
        end,
        total_status
    from inn_kk_query_status(:icq_id)
    into total_status,
         total_status_color,
         total_status_id;

    eil_status = 0;
    eir_answer = '';

    if (total_status_id = 1) then
    begin
        select eil.eil_status,
               left(err.eir_answer, 255)
        from extinfolist eil
        inner join inn_check_queries icq on eil.eil_id = icq.icq_from_eil_id
        left join extinfoerr err on err.eir_from_eil_id = eil.eil_id
        where icq.icq_id = :icq_id
        and eil.eil_status = -2
        into :eil_status,
             :eir_answer;

        if (eil_status = -2) then
        begin
            total_status = 'Проверка не удалась. Сервис ФНС вернул ошибку';
            total_status_color = 'red';
        end
    end

    request_html_txt = blob_append(request_html_txt, f_format('%1: %2<br>', 'Источник данных', iif(icq_info_source = 0, 'ФНС', 'СМЭВ')));
    request_html_txt = blob_append(request_html_txt, f_format('%1: %2<br>', 'Дата/время запроса', f_datetimetostr(:eil_create_datetime)));
    request_html_txt = blob_append(request_html_txt, f_format('%1: %2<br>', 'Дата/время получения ответа', f_datetimetostr(:eil_status_datetime)));
    request_html_txt = blob_append(request_html_txt, f_format('%1: <span style="color: %3">%2</span><br>', 'Результат', total_status, total_status_color));
    if (eir_answer <> '') then
        request_html_txt = blob_append(request_html_txt, f_format('%1: <span style="color: %3">%2</span><br>', 'Ошибка', eir_answer, total_status_color));

    request_html_txt = blob_append(request_html_txt, f_format('%1: %2<br>', 'Источник', icq_creation_stage));
    request_html_txt = blob_append(request_html_txt, f_format('%1: %2<br>', 'Имя', icq_fname));
    request_html_txt = blob_append(request_html_txt, f_format('%1: %2<br>', 'Фамилия', icq_sname));
    request_html_txt = blob_append(request_html_txt, f_format('%1: %2<br>', 'Отчество', icq_patronimic));
    request_html_txt = blob_append(request_html_txt, f_format('%1: %2<br>', 'Дата рождения', f_datetostr(:c_birth_date)));
    request_html_txt = blob_append(request_html_txt, f_format('%1: %2<br>', 'Вид документа', iif(icq_doc_type = 10, 'Паспорт иностранного гражданина', 'Паспорт гражданина России')));
    request_html_txt = blob_append(request_html_txt, f_format('%1: %2<br>', 'Номер документа', icq_doc_no));
    request_html_txt = blob_append(request_html_txt, f_format('%1: %2<br>', 'Дата выдачи документа', f_datetostr(:icq_doc_date)));
    request_html_txt = blob_append(request_html_txt, '<hr />');
  end

  html_txt = blob_append(html_txt, f_format('<h4>%1</h4>', :c_name_fio));

  if (counter <> 0) then   -- RFCRU-3523
    html_txt = blob_append(html_txt, f_format('<span style="color: %2">%1</span>', iif(icm_inn_received = 0, 'ИНН не найден', 'ИНН найден'), iif(icm_inn_received = 0, 'red', 'green')));

  html_txt = blob_append(html_txt, f_format('<p>%1</p>', 'Список запросов:'));

  html_txt = blob_append(html_txt, request_html_txt);

  if (counter = 0) then
    html_txt = blob_append(html_txt, '<p>Запросов не было</p><hr />');

  html_txt = blob_append(html_txt, '</body>');
  suspend;
end
