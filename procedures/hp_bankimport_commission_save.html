CREATE PROCEDURE HP_BANKIMPORT_COMMISSION_SAVE(
    OL_ID ID_TYPE,
    SUMM CURR_TYPE)
AS
declare variable reg_id integer;
declare variable parent_id id_type;
declare variable account_dt varchar(20);
declare variable account_kt varchar(120);
declare variable def_curr varchar(3);
declare variable c_id id_type;
declare variable ol_id_child id_type;
declare variable k_kontract_no str_100;
declare variable oper_date date;
begin
  select ol.ol_from_reg_id, ol.ol_parent_id, ol_document_client, ol.ol_date
  from operlist ol
  where ol.ol_id = :ol_id
  into :reg_id, :parent_id, :c_id, :oper_date;

  if (parent_id is not null) then
  begin
    ol_id_child = :ol_id;

    select ol.ol_from_reg_id, ol_document_client
    from operlist ol
    where ol.ol_id = :parent_id
    into :reg_id, :c_id;

    ol_id = parent_id;
  end
  else
  begin
    select ol.ol_id
    from operlist ol
    where ol.ol_parent_id = :ol_id rows 1
    into :ol_id_child;
  end

  select kont.k_kontract_no
  from operkred ok
  inner join kontract kont on kont.k_id = ok.ok_from_k_id
  where ok.ok_from_ol_id = :ol_id_child rows 1
  into :k_kontract_no;

  select al_id
  from acc_list
  where al_from_bg_id = '91-02' and
        al_from_reg_id = :reg_id and
        acc_list.is_delete = 0 and
        al_number = 3
  into account_dt;

  select ob.ob_account_dt
  from operbook ob
  where ob.ob_from_ol_id = :ol_id
  rows 1
  into :account_kt;

  select cm.cm_default_currency
  from company cm
  into def_curr;

  insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                        ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                        ob_from_sac_id, ob_ol_date) --, ob_ol_is_active, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt) -- они через триггер обновляются
  values (:ol_id, -1, :def_curr, 1, :c_id, :account_dt, :summ, :summ, null, null, null, :account_kt, null,
          'ком. за кред. займ ' || :k_kontract_no, null, :oper_date);
end
