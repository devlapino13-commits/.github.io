CREATE PROCEDURE AAREP_ACREPLINE_OBOROT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    AJ_BALANCE_DT VARCHAR(20),
    AJ_ACC_NUMBER_DT INTEGER,
    AJ_BALANCE_KT VARCHAR(20),
    AJ_ACC_NUMBER_KT INTEGER,
    AJ_SUMMA DECIMAL(18,4))
AS
begin

  for select a1.al_from_bg_id, a1.al_number, a2.al_from_bg_id, a2.al_number, sum(ob_summa)
      from acc_list a1, acc_list a2, operbook, region_list(:report_region_id, 1, -1, 1)
      where a1.al_id = ob_account_dt and
            a2.al_id = ob_account_kt and
            ob_ol_date between :report_date_begin and :report_date_end and
            ob_ol_is_active = 1 and
            ob_ol_reg_id = id_list
      group by a1.al_from_bg_id, a1.al_number, a2.al_from_bg_id, a2.al_number
      into :aj_balance_dt, :aj_acc_number_dt, :aj_balance_kt, :aj_acc_number_kt, :aj_summa
  do
  begin
    suspend;
  end
