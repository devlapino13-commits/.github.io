CREATE PROCEDURE HP_INVENTORY_CHECK_APPROVE(
    IM_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    RES_STATUS D_SMALLINT)
AS
declare variable has_access d_smallint;
begin
  has_access = null;
  select has_access
  from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.invmain_check_approve')
  into has_access;

  if (coalesce(has_access, 0) = 0) then
  begin
    res_text = 'У вас нет доступа для перевода акта в статус "Проверено"';
    res_status = 0;
    suspend;
    exit;
  end 

  if (not exists(select *
                 from invmain im
                 where im.im_id = :im_id)) then
  begin
    res_text = 'Акт не сохранён. Действие отменяется.';
    res_status = 0;
    suspend;
    exit;
  end 

  select im.im_status
  from invmain im
  where im.im_id = :im_id
  into res_status;

  if (res_status = 0) then
  begin
    update invmain im
    set im.im_status = 2
    where im.im_id = :im_id;

    res_text = 'Статус акта успешно изменён на "Проверено"';
    res_status = 2;
  end
  else
  if (res_status = 2) then
    res_text = 'Акт уже в статусе "Проверено"';
  else
  if (res_status = -1) then
    res_text = 'Акт аннулирован. Невозможно перевести акт в статус "Проверено"';

  suspend;
end
