Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_UUID(
    UUID D_STR_CODE)
RETURNS (
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID)
AS
begin
  -- По UUID возвращает тип и ID объекта
  select udl_object_kind, udl_object_id
  from uuidlist
  where udl_uuid = char_to_uuid(:uuid)
  into obj_kind, obj_id;

  suspend;
end
