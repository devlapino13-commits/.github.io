CREATE PROCEDURE HP_OPERCASH_AKT_FACT_SAVE(
    CFL_FROM_CIL_ID D_INTEGER,
    CFL_FROM_CSP_ID D_INTEGER,
    CFL_DATE D_DATE,
    CFL_CASHIER_ID D_ID,
    CFL_INSPECTOR_ID D_ID,
    CFL_FACT_AMOUNT D_CURRENCY,
    CFL_FACT_AMOUNT$BOND D_STR_LONG,
    PERSON_ID D_ID,
    DIFF_ACCOUNT D_STR_CODE,
    CFL_FROM_UN_ID D_ID)
AS
declare variable cil_inv_amount d_currency;
declare variable oper_kind d_integer;
declare variable akt_oper_kind_code d_str_code;
declare variable opc_corr_reg_id d_integer;
declare variable opc_is_ipf d_integer;
declare variable add_oper_sign d_smallint;
declare variable add_memo d_str_long;
declare variable opc_corr_al_id d_str_code;
declare variable opc_id d_integer;
declare variable cfl_id d_integer;
declare variable csp_office_id d_integer;
begin
  select cil_inv_amount, cil_oper_kind
  from cashinvlist
  where cil_id = :cfl_from_cil_id
  into cil_inv_amount, oper_kind;

  if (row_count = 0) then                     --
    exception e_message 'Акт не найден, id: ' || coalesce(cfl_from_cil_id, '< Null>');

  select cil_id
  from cashinvlist
  where cil_parent_id = :cfl_from_cil_id and
        cil_is_delete = 0
  into cfl_id;

  if (cfl_id is not null) then
      exception e_message 'Акт ранее обработан';

  akt_oper_kind_code = f_enum_code('cashoperkind', oper_kind);

  select csp_office_id
  from cashplace
  where csp_id = :cfl_from_csp_id
  into csp_office_id;

  if (exists(select *
             from hp_is_day_closed(:csp_office_id, :cfl_date, 1)
             where day_closed = 1)) then
    exception e_message 'Опер.день кассы закрыт: ' || f_datetostr(cfl_date);

  insert into cashinvlist (cil_parent_id, cil_from_csp_id, cil_date, cil_oper_kind, cil_cashier_id,
                           cil_inspector_id, cil_inv_amount, cil_from_un_id)
  values (:cfl_from_cil_id, :cfl_from_csp_id, :cfl_date, :oper_kind, :cfl_cashier_id,
          :cfl_inspector_id, :cfl_fact_amount, :cfl_from_un_id)
  returning cil_id
  into :cfl_id;

  execute procedure hp_cashbond_update(:cfl_id, :cfl_fact_amount$bond);

  if (akt_oper_kind_code = 'plus.inkassa') then
  begin
    execute procedure hp_cashterminal_book_save(:cfl_from_cil_id, :cfl_from_un_id);
    exit;
  end

  opc_corr_reg_id = null;
  opc_is_ipf = 0;

  if (akt_oper_kind_code in ('minus.office', 'minus.to_kassa')) then
  begin
    oper_kind = f_enum_id('cashoperkind', 'plus.transfer');

    select opc_office_id, opc_is_ipf, opc_corr_al_id
    from opercash
    where opc_from_cil_id = :cfl_from_cil_id and
          opc_parent_id is null and
          opc_is_delete = 0
    order by opc_id desc
    rows 1
    into opc_corr_reg_id, opc_is_ipf, opc_corr_al_id;
  end

  opc_id = null;

  if (akt_oper_kind_code <> 'minus.to_bank') then
  begin
    insert into opercash (opc_date, opc_from_csp_id, opc_oper_kind, opc_sign,
                          opc_amount, opc_from_c_id, opc_is_ipf, opc_corr_al_id,
                          opc_corr_reg_id, opc_manager_id, opc_from_cil_id, opc_memo, opc_from_un_id)
    values (:cfl_date, :cfl_from_csp_id, :oper_kind, 1,
            :cil_inv_amount, :person_id, :opc_is_ipf, :opc_corr_al_id,
            :opc_corr_reg_id, :cfl_inspector_id, :cfl_id, 'Приход инкассы', :cfl_from_un_id)
    returning opc_id
    into opc_id;
  end

  if (cil_inv_amount <> cfl_fact_amount) then
  begin
    if (cil_inv_amount > cfl_fact_amount) then
    begin
      oper_kind = f_enum_id('cashoperkind', 'minus.saldo'); -- если по чему больше по факту
      add_oper_sign = -1;
      add_memo = 'Недостача при инкассе';
    end
    else
    begin
      oper_kind = f_enum_id('cashoperkind', 'plus.saldo');
      add_oper_sign = 1;
      add_memo = 'Излишка при инкассе';
    end

    insert into opercash (opc_parent_id, opc_date, opc_from_csp_id, opc_oper_kind, opc_sign,
                          opc_amount, opc_from_c_id, opc_is_ipf, opc_corr_al_id,
                          opc_corr_reg_id, opc_manager_id, opc_from_cil_id, opc_memo, opc_from_un_id)
    values (:opc_id, :cfl_date, :cfl_from_csp_id, :oper_kind, :add_oper_sign,
            abs(:cil_inv_amount - :cfl_fact_amount), :person_id, :opc_is_ipf, :diff_account,
            :opc_corr_reg_id, :cfl_inspector_id, :cfl_id, :add_memo, :cfl_from_un_id);
  end
end
