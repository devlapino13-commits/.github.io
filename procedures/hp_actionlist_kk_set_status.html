CREATE PROCEDURE HP_ACTIONLIST_KK_SET_STATUS(
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID,
    RD_ID_MAIN D_INTEGER,
    RD_ID_DETAIL D_INTEGER,
    NEW_STATUS D_ID,
    NEW_MEMO D_STR_255,
    NEW_COMMENT_CODE D_STR_CODE,
    NEW_COMMENT_TEXT D_STR_255,
    UN_ID D_ID,
    RD_ID_REASON D_INTEGER = null)
AS
declare variable oper_kind d_str_code;
declare variable res_memo d_str_long;
begin
  -- устанавливаем значение (состояние) ActionList
  rd_id_reason = coalesce(rd_id_reason, 0);
  new_comment_code = coalesce(new_comment_code, '');
  new_comment_text = coalesce(new_comment_text, '');

  if (new_memo is null) then
    new_memo = ''; -- Комментарии при смене статуса

  if (object_kind = 9 and new_comment_text = '') then
  begin
    if (new_status = -1) then
      exception e_message 'Выберите стандартную причину доработки.';
    else
    if (new_status = -2) then
      exception e_message 'Выберите стандартную причину отказа.';
  end

  if (new_status < 0 and new_memo = '') then
    exception e_message 'В случае отказа или доработки, необходимо заполнить комментарий.';

  oper_kind = '';
  oper_kind = trim(case new_status
    when  0 then 'Для сведения:'
    when  1 then 'Для сведения:'
    when -1 then 'Доработка:'
    when -2 then 'Отказ:'
    when  2 then 'Подтверждён:'
  end);

  res_memo = '';
  if (new_status < 0 and object_kind = 9) then
    res_memo =  '[' || oper_kind || ' ' || new_comment_text || ']' || ascii_char(13) || ascii_char(10);
  else
  begin
    if (new_memo = '') then
      res_memo = new_memo;
    else
      res_memo = oper_kind || ' ' || new_memo;
  end

  if (new_comment_code <> '') then
    res_memo = Left(res_memo, 254 - char_Length(new_comment_code)) || '#' || new_comment_code;
  else
    res_memo = Left(res_memo, 255);

  execute procedure hp_actionlist_set_status(:object_kind, :object_id, :rd_id_main, :rd_id_detail, :new_status, :res_memo, :un_id, :rd_id_reason);
end
