CREATE PROCEDURE AAREP_EFFECTIVENESS_WEEK_OLD(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(50),
    WEEK_1_DISB_SVZ_ONE CURR_TYPE,
    WEEK_2_DISB_SVZ_ONE CURR_TYPE,
    WEEK_3_DISB_SVZ_ONE CURR_TYPE,
    WEEK_4_DISB_SVZ_ONE CURR_TYPE,
    WEEK_5_DISB_SVZ_ONE CURR_TYPE,
    WEEK_6_DISB_SVZ_ONE CURR_TYPE,
    WEEK_1_SVZ_CLI_AVG CURR_TYPE,
    WEEK_2_SVZ_CLI_AVG CURR_TYPE,
    WEEK_3_SVZ_CLI_AVG CURR_TYPE,
    WEEK_4_SVZ_CLI_AVG CURR_TYPE,
    WEEK_5_SVZ_CLI_AVG CURR_TYPE,
    WEEK_6_SVZ_CLI_AVG CURR_TYPE)
AS
declare variable reg_unicode integer;
declare variable i smallint;
declare variable rep_date date;
declare variable svz_kol_act integer;
declare variable disb_kred_week integer;
declare variable rep_date_begin date;
declare variable kol_cli integer;
declare variable svz_disb_kol curr_type;
declare variable svz_cli_kol curr_type;
begin
  rep_date = :rep_date_end;
  for select reg_unicode, reg_name
      from region, region_list(:rep_region_id, 1, 3, 1)
      where reg_unicode = id_list
--  and reg_unicode = 1240100
      order by reg_name
      into reg_unicode, office_name
  do
  begin
    rep_date = :rep_date_end;
    i = 0;
    while (i < 6) do
    begin
      rep_date = rep_date - (6 - i) * 7;
      rep_date_begin = rep_date - 6; -- начало недели
      i = i + 1;
-- Кол-во СВЗ
      svz_kol_act = null;
      select count(ps_from_c_id)
      from jobkind, joblist, hr_pstatus
      where jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            jk_id = jl_kind and
            jl_id = ps_from_jl_id and
            ps_status = 1 and
            ps_is_working = 1 and
            :rep_date between ps_date_begin and ps_date_end and
            ps_from_reg_id = :reg_unicode
-- Исключаем декретников
            and
            not coalesce((select sawt_id
                          from hr_voctravel, hr_orders, sa_worktype
                          where hrv_from_ho_id = ho_id and
                                ho_order_type = sawt_id and
                                hrv_from_c_id = ps_from_c_id and
                                :rep_date between hrv_date_begin and hrv_date_end_fact
                          order by hrv_date_begin desc
                          rows 1), 0) in (16, 17, 18)
      into svz_kol_act;
      if (svz_kol_act is null) then
        svz_kol_act = 0;
-- Кол-во выдачи
      disb_kred_week = 0;
      select sum(k_kolich_client_approved)
      from kontstate, kontract, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            cast(ks_date_begin as date) between :rep_date_begin and :rep_date and
            ks_region_id = :reg_unicode and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_product_kind <> 2
      into disb_kred_week;
      if (disb_kred_week is null) then
        disb_kred_week = 0;
-- Кол-во активных клиентов
      select sum(ac_is_client)
      from aarep_cli_active_list(:rep_date)
      where ac_region_id = :reg_unicode and
            ac_is_client = 1 and
            (select aa_category_id
             from aarep_get_category_id(ac_kontract_id, :rep_date, null, null, null, null, null, null)) in (1, 2)
      into kol_cli;
      if (kol_cli is null) then
        kol_cli = 0;
--    ВЫДАЧИ НА 1 АКТИВНОГО СВЗ
--    СРЕДНЕЕ КОЛ-ВО КЛИЕНТОВ НА 1 СВЗ
      if (svz_kol_act > 0) then
      begin
        svz_disb_kol = cast(disb_kred_week as double precision) / svz_kol_act;
        svz_cli_kol = cast(kol_cli as double precision) / svz_kol_act;
      end
      else
      begin
        svz_disb_kol = 0;
        svz_cli_kol = 0;
      end

      if (i = 1) then
      begin
        week_1_disb_svz_one = svz_disb_kol;
        week_1_svz_cli_avg = svz_cli_kol;
      end
      if (i = 2) then
      begin
        week_2_disb_svz_one = svz_disb_kol;
        week_2_svz_cli_avg = svz_cli_kol;
      end
      if (i = 3) then
      begin
        week_3_disb_svz_one = svz_disb_kol;
        week_3_svz_cli_avg = svz_cli_kol;
      end
      if (i = 4) then
      begin
        week_4_disb_svz_one = svz_disb_kol;
        week_4_svz_cli_avg = svz_cli_kol;
      end
      if (i = 5) then
      begin
        week_5_disb_svz_one = svz_disb_kol;
        week_5_svz_cli_avg = svz_cli_kol;
      end
      if (i = 6) then
      begin
        week_6_disb_svz_one = svz_disb_kol;
        week_6_svz_cli_avg = svz_cli_kol;
      end

    end
    suspend;
  end
end
