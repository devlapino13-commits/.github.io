      from equifax_fip_list_temp(:pocket_id, 4)
      into :rep_date_begin, :status_date, :cred_missedpay_120, :order_no, :status_code, :status, :cred_full_pay,

           :snils, :ma_id, :k_id, uid
  do
  begin
    application_date = '';
    if (application_date_tmp is not null) then
    begin
      y = extract(year from application_date_tmp);
      m = extract(month from application_date_tmp);
      d = extract(day from application_date_tmp);

      application_date = iif(d < 10, '0' || d, d) || '.' || iif(m < 10, '0' || m, m) || '.' || y || substring(application_date_tmp from 11 for 9);
    end

-- replace(left(cast(APPLICATION_DATE_TMP as str_100), 19),'-', '.'),

    counter = counter + 1;

    xml_middlename = 1;
    xml_docenddate = 1;
    xml_docwho = 1;
    xml_inn = 1;
    xml_snils = 1;

    xml_ar_district = 1;
    xml_ar_street = 1;
    xml_ar_build_no = 1;
    xml_ar_block_no = 1;
    xml_ar_apartment_no = 1;

    xml_af_district = 1;
    xml_af_street = 1;
    xml_af_build_no = 1;
    xml_af_block_no = 1;
    xml_af_apartment_no = 1;

    if (coalesce(middlename, '') = '') then
      xml_middlename = 0;

    if (docenddate is null) then
      xml_docenddate = 0;

    if (coalesce(docwho, '') = '') then
      xml_docwho = 0;

    if (coalesce(inn, '') = '') then

    if (coalesce(snils, '') = '') then

      xml_ar_district = 0;

    if (coalesce(ar_street, '') = '') then
      xml_ar_street = 0;

    if (coalesce(ar_build_no, '0') = '0') then
      xml_ar_build_no = 0;

    if (coalesce(ar_block_no, '0') = '0') then
      xml_ar_block_no = 0;

    if (coalesce(ar_apartment_no, '0') = '0') then
      xml_ar_apartment_no = 0;

--
    if (coalesce(af_district, '') = '') then
      xml_af_district = 0;

    if (coalesce(af_street, '') = '') then
      xml_af_street = 0;

    if (coalesce(af_build_no, '0') = '0') then
      xml_af_build_no = 0;

    if (coalesce(af_block_no, '0') = '0') then
      xml_af_block_no = 0;

    if (coalesce(af_apartment_no, '0') = '0') then
      xml_af_apartment_no = 0;

    xml_application = 0;
    xml_approval_date = 0;
    xml_failure = 0;

    if (type_ = 0) then
      xml_application = 1;

    if (status_code = 2) then
      xml_approval_date = 1;

    if (status_code = 3) then
      xml_failure = 1;

    xml_credit = 0;

    if (status in ('cont_120', 'closed', 'disb')) then
    begin
      xml_credit = 1;
    end

    do_suspend = 1;

    if (status = 'cont_120') then
    begin
      if (exists(select *
                 from equifax_missedpay_120 tt
                 where tt.emp_from_k_id = :k_id and
                       tt.emp_from_cp_id = :pocket_id)) then
      begin
        do_suspend = 0;
      end
      else
      begin
        insert into equifax_missedpay_120 (emp_from_k_id, emp_from_cp_id)
        values (:k_id, :pocket_id);
      end
    end

    if (do_suspend = 1) then
      suspend;
  end
