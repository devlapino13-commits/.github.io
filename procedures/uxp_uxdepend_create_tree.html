CREATE PROCEDURE UXP$UXDEPEND_CREATE_TREE(
    UX_UUID D_UUID,
    OWNER_UUID D_UUID,
    PARENT_CODE D_STR_100,
    CURR_LEVEL D_INTEGER,
    MAX_LEVEL D_INTEGER,
    SKIP_CHILDS D_BOOL = 0)
AS
declare variable ux_kind d_str_code;
declare variable res_image d_integer;
declare variable lock_delete d_smallint;
declare variable lock_child d_bool;
declare variable tmp_uuid d_uuid;
declare variable uxa_kind d_integer;
declare variable img_add_code d_str_code;
declare variable tmp_count d_integer;
declare variable skip_insert d_bool;
declare variable mm_uuid d_uuid;
declare variable is_mainmenu d_bool;
begin
  ux_kind = uxf$code(ux_uuid, 'kind');
  skip_insert = 0;

  mm_uuid = uxf$mainmenu();

  if (skip_childs is null) then
    skip_childs = 0;

  if (ux_kind = 'uxframe') then
  begin
    if (exists(select * from ux$depend
               where udp_owner_uuid = :owner_uuid and
                     udp_uuid = :ux_uuid)) then
      exit;

    if (exists(select * from ux$frame
               where ufr_uuid = :ux_uuid and
                     uxf$code(ufr_owner, 'kind') = 'uxreport')) then
    skip_insert = 1;
  end else
  if (ux_kind = 'uxalias') then
  begin
    select count(*)
    from ux$alias a1, ux$alias a2
    where a1.uxa_uuid = :ux_uuid and
          a2.uxa_owner = a1.uxa_owner
    into tmp_count;

    if (tmp_count = 1) then
      skip_insert = 1;
  end

  if (coalesce(parent_code, '') = '') then
    parent_code = null;

  lock_delete = 0;
  lock_child = 0;

  img_add_code = '';

  if (skip_insert = 0 or skip_childs = 1) then
  begin
    if (curr_level is null) then
      curr_level = 0;
  
    if (max_level is null) then
      max_level = 1;
  
    curr_level = curr_level + 1;
  
    if (curr_level > max_level) then
      exit;

    if (ux_kind = 'uxaction') then
    begin
      select ua_uuid, uxa_kind, iif(ua_owner = :mm_uuid, 1, 0)
      from ux$action, ux$alias
      where ua_uuid = :ux_uuid and
            ua_uxa_uuid = uxa_uuid
      into tmp_uuid, uxa_kind, is_mainmenu;
  
      if (is_mainmenu = 1) then
      begin
        img_add_code = 'mainmenu';

        if (exists(select * from uxp$actionmenu(:ux_uuid))) then
          lock_delete = 1;
      end
      else
        img_add_code = uxa_kind;

    end else
    if (ux_kind = 'uxalias') then
    begin
      if (exists(select * from ux$search
                 where usr_uuid = :ux_uuid)) then
        img_add_code = 'find';
      else
      if (exists(select * from ux$alias
                 where uxa_uuid = :ux_uuid and
                       uxa_kind = 2)) then
        img_add_code = 'new';
    end else
    if (ux_kind = 'uxfrparam') then
    begin
      select ufp_editor
      from ux$frparam
      where ufp_uuid = :ux_uuid
      into img_add_code;
    end else
    if (ux_kind in ('uxform', 'uxreport', 'uxcommand')) then
    begin
      img_add_code = '';
      select iif(ufm_uuid = ux_release_uuid, 'release',
                 iif(ufm_uuid = ux_draft_uuid, '', 'version'))
      from ux$form, ux$code
      where ufm_uuid = :ux_uuid and
            ufm_ux_uuid = ux_uuid
      into img_add_code;

      if (img_add_code = 'release') then
      begin
        lock_delete = 1;
        lock_child = 1;
      end
    end
  
    select res_image_id
    from uxp$uxkind_image_id(:ux_kind, :img_add_code)
    into :res_image;
  
    insert into ux$depend (udp_uuid, udp_parent_code, udp_owner_uuid, udp_ux_kind, udp_image_id,
                           udp_lock_delete, udp_child_locked)
    values (:ux_uuid, :parent_code, :owner_uuid, :ux_kind, :res_image,
            :lock_delete, :lock_child)
    returning udp_code
    into parent_code;
  end

  if (skip_childs = 1) then
    Exit;

  if (ux_kind = 'uxframe') then
  begin
    select ufr_owner
    from ux$frame
    where ufr_uuid = :ux_uuid
    into tmp_uuid;

    execute procedure uxp$uxdepend_create_tree(:tmp_uuid, :owner_uuid, :parent_code, :curr_level, :max_level);
  end else
  if (ux_kind in ('uxform', 'uxreport', 'uxcommand')) then
  begin
    for select uxa_uuid
    from ux$form, ux$alias
    where ufm_uuid = :ux_uuid and
          ufm_ux_uuid = uxa_owner
    into tmp_uuid
    do
      execute procedure uxp$uxdepend_create_tree(:tmp_uuid, :owner_uuid, :parent_code, :curr_level, :max_level);
  end else
  if (ux_kind = 'uxalias') then
  begin
    select count(*)
    from ux$alias, ux$form
    where uxa_uuid = :ux_uuid and
          uxa_owner = ufm_ux_uuid
    into tmp_count;

    for select ua_uuid, iif(ua_owner = :mm_uuid, 1, 0)
    from ux$action
    where ua_uxa_uuid = :ux_uuid
    into tmp_uuid, is_mainmenu
    do
    begin
      execute procedure uxp$uxdepend_create_tree(:tmp_uuid, :owner_uuid, :parent_code, :curr_level, :max_level);

      lock_delete = 0;

      if (tmp_count = 1 and is_mainmenu = 1) then
      begin
        if (exists(select * from uxp$actionmenu(:tmp_uuid))) then
          lock_delete = 1;
      end

      if (tmp_count = 1 and lock_delete = 1) then
      begin
        update ux$depend
        set udp_lock_delete = 1,
            udp_child_locked = 1
        where udp_code = :parent_code and
              (udp_lock_delete <> 1 or udp_child_locked <> 1);
      end
    end
  end else
  if (ux_kind = 'uxaction') then
  begin
    if (img_add_code <> 'mainmenu') then
    begin
      select ua_owner
      from ux$action
      where ua_uuid = :ux_uuid
      into tmp_uuid;
  
      execute procedure uxp$uxdepend_create_tree(:tmp_uuid, :owner_uuid, :parent_code, :curr_level, :max_level);
    end
  end else
  if (ux_kind = 'uxsource') then
  begin
    for with tab as
    (select ufr_uuid as ufr_uuid, 0 as kind
    from ux$frame
    where ufr_us_uuid = :ux_uuid
    union
    select ufd_ufr_uuid as ufr_uuid, 1 as kind
    from ux$frdata
    where ufd_us_uuid = :ux_uuid)
    select ufr_uuid
    from tab
    group by ufr_uuid
    order by min(kind)
    into tmp_uuid
    do
      execute procedure uxp$uxdepend_create_tree(:tmp_uuid, :owner_uuid, :parent_code, :curr_level, :max_level);

    for select ufm_uuid
    from ux$code, ux$form
    where ux_us_uuid = :ux_uuid and
          ux_uuid = ufm_ux_uuid
    into tmp_uuid
    do
    begin
      if (not exists(select * from ux$depend
                     where udp_owner_uuid = :owner_uuid and
                           udp_uuid = :tmp_uuid)) then
        execute procedure uxp$uxdepend_create_tree(:tmp_uuid, :owner_uuid, :parent_code, :curr_level, :max_level);
    end

    for select ufl_ufp_uuid
    from ux$frlookup
    where ufl_us_uuid = :ux_uuid
    into tmp_uuid
    do
    begin
      if (not exists(select * from ux$depend
                     where udp_owner_uuid = :owner_uuid and
                           udp_uuid = :tmp_uuid)) then
        execute procedure uxp$uxdepend_create_tree(:tmp_uuid, :owner_uuid, :parent_code, :curr_level, :max_level);
    end
  end else
  if (ux_kind = 'uxfrparam') then
  begin
    for select ufp_ufr_uuid
    from ux$frparam
    where ufp_uuid = :ux_uuid
    into tmp_uuid
    do
      execute procedure uxp$uxdepend_create_tree(:tmp_uuid, :owner_uuid, :parent_code, :curr_level, :max_level);
  end else
  if (ux_kind = 'uxfield') then
  begin
    for select uf_owner
    from ux$field
    where uf_uuid = :ux_uuid
    into tmp_uuid
    do
      execute procedure uxp$uxdepend_create_tree(:tmp_uuid, :owner_uuid, :parent_code, :curr_level, :max_level, 1);

    for select ufp_uuid
    from ux$frparam
    where ufp_uf_uuid = :ux_uuid
    into tmp_uuid
    do
      execute procedure uxp$uxdepend_create_tree(:tmp_uuid, :owner_uuid, :parent_code, :curr_level, :max_level);
  end
end
