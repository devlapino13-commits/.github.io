CREATE PROCEDURE MC_SELFIE_SAVE(
    MCL_ID D_ID,
    PHOTO D_BLOB_BINARY)
AS
declare variable file_name d_str_100;
declare variable caption d_str_100;
declare variable bls_id d_id;
declare variable un_id d_id;
declare variable c_id d_id;
begin
  caption = 'kmp_user_photo';
  file_name = 'user_photo_' || :mcl_id || '.jpeg';
  un_id = 1000000000000; -- ID робота

  -- Сохранение фото в blobstore
  execute procedure hp_insert_into_blobstore(:file_name, :caption, :photo, :un_id, 0, 2)
      returning_values :bls_id;

  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into c_id;

  update clients c
  set c.c_blob_photo_id = :bls_id
  where c.c_id = :c_id;

end
