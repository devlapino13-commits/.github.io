Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_DO_KN_OPERBALBOOK_FIX_DATE(
    REP_DATE D_DATE,
    NEED_DELETE D_SMALLINT = 0)
AS
begin
  if (coalesce(need_delete, 0) = 0) then
  begin
    update kn_operbalbook
    set obb_need_fix = 1
    where obb_ol_date = :rep_date;
  end else
  begin
    delete from kn_operbalbook
    where obb_ol_date = :rep_date;
  
    delete from kn_balsummary
    where kbs_date = :rep_date;
  end

  for select dt.acc_from_ab_id as Dt, kt.acc_from_ab_id as Kt,
         sum(kob_summa) as summa,
         sum(kob_summa_val) as summa_val,
         sum(kob_summa_tax_dt) as summa_tax_dt,
         sum(kob_summa_tax_kt) as summa_tax_kt
  from kn_operbook, ref_account dt, ref_account kt
  where kob_acc_dt = dt.acc_id and
        kob_acc_kt = kt.acc_id and
        kob_ol_is_active = 1 and
        kob_ol_date = :rep_date
  group by 1, 2
  as cursor book
  do
  begin
    update or insert into kn_operbalbook (obb_ol_date, obb_ab_dt, obb_ab_kt, obb_summa, obb_summa_val,
                                          obb_summa_tax_dt, obb_summa_tax_kt, obb_is_extended, obb_need_fix)
    values (:rep_date, book.dt, book.kt, book.summa, book.summa_val, book.summa_tax_dt, book.summa_tax_kt,
            0, 0)
    matching (obb_ol_date,obb_ab_dt,obb_ab_kt);
  end

  for select dt.acc_from_add_ab_id as Dt, kt.acc_from_add_ab_id as Kt,
         sum(kob_summa) as summa,
         sum(kob_summa_val) as summa_val,
         sum(kob_summa_tax_dt) as summa_tax_dt,
         sum(kob_summa_tax_kt) as summa_tax_kt
  from kn_operbook, ref_account dt, ref_account kt
  where kob_acc_dt = dt.acc_id and
        kob_acc_kt = kt.acc_id and
        kob_ol_is_active = 1 and
        kob_ol_date = :rep_date
  group by 1, 2
  as cursor book
  do
  begin
    update or insert into kn_operbalbook (obb_ol_date, obb_ab_dt, obb_ab_kt, obb_summa, obb_summa_val,
                                          obb_summa_tax_dt, obb_summa_tax_kt, obb_is_extended, obb_need_fix)
    values (:rep_date, book.dt, book.kt, book.summa, book.summa_val, book.summa_tax_dt, book.summa_tax_kt,
            1, 0)
    matching (obb_ol_date,obb_ab_dt,obb_ab_kt);
  end

  delete from kn_operbalbook
  where obb_ol_date = :rep_date and
        obb_need_fix = 1;

  delete from kn_operbalbook_date
  where fix_date = :rep_date;
end
