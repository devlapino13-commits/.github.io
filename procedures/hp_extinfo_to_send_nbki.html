CREATE PROCEDURE HP_EXTINFO_TO_SEND_NBKI(
    C_ID ID_TYPE,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    JSON_VALUE D_BLOB_TEXT)
AS
declare variable pass_no str_100;
declare variable counter integer;
declare variable str_leng integer;
declare variable ch str_100;
declare variable fl_id id_type;
declare variable c_birth_city str_100;
declare variable c_birth_country str_100;
declare variable street str_100;
declare variable housenumber d_str_20;
declare variable adr_post_index str_100;
declare variable street_post_index str_100;
declare variable city_post_index str_100;
declare variable city_name str_100;
declare variable is_finish smallint;
declare variable birthdt str_100;
declare variable gender smallint;
declare variable issuedate str_100;
declare variable issueauthority str_255;
declare variable issuecountry str_100;
declare variable placeofbirth str_100;
declare variable first_n str_100;
declare variable name1 str_100;
declare variable paternal str_100;
declare variable seria str_100;
declare variable number str_100;
declare variable addresstype smallint;
declare variable ar_housenumber d_str_20;
declare variable ar_street str_100;
declare variable ar_city str_100;
declare variable ar_postal str_100;
declare variable af_housenumber d_str_20;
declare variable af_street str_100;
declare variable af_city str_100;
declare variable af_postal str_100;
declare variable ar_addresstype smallint;
declare variable af_addresstype smallint;
declare variable create_date date;
declare variable close_date date;
declare variable consent_date d_str_code;
declare variable consent_expire_date d_str_code;
declare variable ma_id d_id;
declare variable pass_type d_id;
begin

  if (coalesce(object_kind, 0) not in (9,11)) then
  begin
    exception e_default 'object_kind must be 11 or 9';
  end 

  if (object_kind = 9) then
  begin
    select k_kredline_end, k_from_ma_app_id
    from kontract
    where k_id = :object_id
    into close_date, ma_id;

    if (close_date is null) then
    begin
        select max(sh_date)
        from schedule sh
        where sh.sh_from_k_id = :object_id
        into :close_date;
    end
  end else

  if (object_kind = 11) then
  begin
    ma_id = object_id;
    select max(sh_date) from schedule
    where sh_from_k_id = -:object_id
    into close_date;

    if (close_date is null) then
    begin
        select max(sh_date)
        from schedule sh
        inner join kontract kont on kont.k_id = sh.sh_from_k_id
        where kont.k_from_ma_app_id = :ma_id
        into close_date;
    end
  end 

  select cast(ma_date_save as date)
  from miniapp
  where ma_id = :ma_id
  order by ma_number
  rows 1
  into create_date;

  consent_date = f_datetostr(create_date);
  consent_expire_date = f_datetostr(close_date);

  select f_datetostr(cl.c_birth_date),
             coalesce(cl.c_pass_no, ''),
              (
               case c_sex
                 when 1 then 1
                 when 0 then 2
                 else 9
              end),
             iif(c_citizen = 13, 21, 10),
             f_datetostr(cl.c_pass_date),
             coalesce(cl.c_pass_who, ''),
             rd.rd_name,
             cl.c_from_fl_id,
             cl.c_birth_city,
             c_birth_country
  from clients cl
  left join refdata rd on rd.rd_id = cl.c_citizen
  where cl.c_id = :c_id
  into birthDt,
       pass_no,
       gender,
       pass_type,
       issueDate,
       issueAuthority,
       issueCountry,
       fl_id,
       c_birth_city,
       c_birth_country;

  placeOfBirth = c_birth_city;
  if (c_birth_country <> '') then
    placeOfBirth = c_birth_country;
  placeOfBirth= trim(coalesce(placeOfBirth, ''));

  select fl.fl_name1, fl.fl_name2, fl.fl_name3
  from fiolist fl
  where fl.fl_id = :fl_id
  into :first_n, :name1, :paternal;

  pass_no = replace(pass_no, ' ', '');
  seria = '';
  number = pass_no;

  if (pass_type = 21) then
  begin
    if(pass_no similar to '[-]?[[:DIGIT:]]+') then
    begin
        seria = substring(pass_no from 1 for 4);
        number = substring(pass_no from 5);
    end
    else
    begin
        counter = 1;
        is_finish = 0;
        str_leng = character_length(pass_no);
        while ((counter < str_leng-1) and (is_finish = 0)) do
        begin
            ch = substring(pass_no from counter for 1);
            if(ch not similar to '[-]?[[:DIGIT:]]+') then
                seria = seria||ch;
            else
            begin
                number = substring(pass_no from counter);
                is_finish = 1;
            end
            counter = counter + 1;
        end
    end
  end

  for select adcl.adc_address_type,
             pl_street.pl_name,
             adl.adl_depart_no,
             iif(coalesce(adl_post_index,   '') = '', '000000', adl_post_index),
             iif(coalesce(pl_street.pl_post_index, '') = '', '000000', pl_street.pl_post_index),
             iif(coalesce(pl_city.pl_post_index, '') = '', '000000', pl_city.pl_post_index),
             pl_city.pl_name
      from addrclient adcl
      inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
      left join places pl_street on pl_street.pl_id = adl.adl_street_id
      left join places pl_city on pl_city.pl_id = adl.adl_from_pl_id
      where adcl.adc_from_c_id = :c_id and
            adcl.adc_address_type in (1, 2)
      into :addressType,
           :street,
           :houseNumber,
           :adr_post_index,
           :street_post_index,
           :city_post_index,
           :city_name
  do
  begin
      if (adr_post_index = '000000') then
      begin
        if (street_post_index <> '') then
          adr_post_index = :street_post_index;
        else
          adr_post_index = :city_post_index;
      end

      if (addressType = 1) then
      begin
          ar_addressType = addressType;
          ar_houseNumber = houseNumber;
          ar_street = street;
          ar_city = city_name;
          ar_postal = adr_post_index;
      end
      else if (addressType = 2) then
      begin
          af_addressType = addressType;
          af_houseNumber = houseNumber;
          af_street = street;
          af_city = city_name;
          af_postal = adr_post_index;
      end
  end

  json_value =
    '{
      "prequest": {
          "req": {
            "InquiryReq": {
                "ConsentReq": {' ||
                  f_json_make_pair('consentFlag', 'Y', 'str') || ', ' ||
                  f_json_make_pair('liability', 'Y', 'str') || ', ' ||
                  f_json_make_pair('consentDate', consent_date, 'str') || ', ' ||
                  f_json_make_pair('consentExpireDate', consent_expire_date, 'str') || ', ' ||
                  f_json_make_pair('consentPurpose', 1, 'num') || ' ' ||'
                  }
              },
              "IdReq": [{'||
                  f_json_make_pair('idNum', number, 'str') || ', ' ||
                  f_json_make_pair('seriesNumber', seria, 'str') || ', ' ||
                  f_json_make_pair('idType', pass_type, 'str') || ', ' ||
                  f_json_make_pair('issueDate', issuedate, 'str') || ', ' ||
                  f_json_make_pair('issueAuthority', issueAuthority, 'str') || ', ' ||
                  f_json_make_pair('issueCountry', issueCountry, 'str') || ' ' ||'
                      }],
              "PersonReq": {'||
                  f_json_make_pair('first', first_n, 'str') || ', ' ||
                  f_json_make_pair('name1', name1, 'str') || ', ' ||
                  f_json_make_pair('paternal', paternal, 'str') || ', ' ||
                  f_json_make_pair('birthDt', birthDt, 'str') || ', ' ||
                  f_json_make_pair('gender', gender, 'str') || ', ' ||
                  f_json_make_pair('placeOfBirth', placeOfBirth, 'str') || ' ' ||'
               },
              "AddressReq": [{'||
                  f_json_make_pair('houseNumber', ar_houseNumber, 'str') || ', ' ||
                  f_json_make_pair('street', ar_street, 'str') || ', ' ||
                  f_json_make_pair('city', ar_city, 'str') || ', ' ||
                  f_json_make_pair('postal', ar_postal, 'str') || ', ' ||
                  f_json_make_pair('addressType', ar_addresstype, 'str') || ' ' ||'
               },{'||
                  f_json_make_pair('houseNumber', af_houseNumber, 'str') || ', ' ||
                  f_json_make_pair('street', af_street, 'str') || ', ' ||
                  f_json_make_pair('city', af_city, 'str') || ', ' ||
                  f_json_make_pair('postal', af_postal, 'str') || ', ' ||
                  f_json_make_pair('addressType', af_addresstype, 'str') || ' ' ||'
                 }]
              }
          }
      }';
  suspend;
end
