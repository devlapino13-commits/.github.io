CREATE PROCEDURE CONTROLLER_FOR_BLOCK_LIST_NEW(
    REP_DATE DATE,
    DAY_COUNT_PRIOR INTEGER,
    REP_REGION_ID ID_TYPE)
RETURNS (
    REG_NAME VARCHAR(50),
    KF_NAME STR_100,
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    CL_CITIZEN_STR STR_100,
    REP_C_ADDRESS_JOB VARCHAR(255),
    K_DISB_DATE DATE,
    COUNTER INTEGER,
    STATE VARCHAR(10),
    CONTROL_FIO STR_100,
    KONT_IS_CHECKED SMALLINT,
    CL_IS_CHECKED SMALLINT,
    IS_CONTROL_BLOCK SMALLINT,
    CHECK_OK SMALLINT,
    CHECK_NOT_OK SMALLINT,
    KF_COUNTER INTEGER)
AS
declare variable kf_id id_type;
declare variable reg_id_prior integer;
declare variable reg_id integer;
declare variable cl_citizen integer;
declare variable dw smallint;
declare variable cl_checked_id id_type;
declare variable kk integer;
declare variable lname str_100;
declare variable fname str_100;
declare variable patronymic str_100;
declare variable contr_check_result smallint;
declare variable k_fact_disb_date date;
declare variable k_fact_closed_date date;
declare variable is_ok smallint;
begin
  reg_id_prior = 0;
  counter = 0;
  kf_counter = 0;
  dw = extract(weekday from rep_date);

  for select ks_from_k_id, kont.k_from_kf_id, k_last_reg_id, k_disb_date,
             (
             case ks_state
               when 0 then 'В процессе'
               else 'Выдан'
             end),
             kont.k_fact_disb_date, kont.k_fact_closed_date
      from kontstate ks
      inner join kontract kont on kont.k_id = ks.ks_from_k_id
      inner join region_list(:rep_region_id, 1, -1, 1) on cast(kont.k_last_reg_id as id_type) = id_list
      where ks_ol_is_delete = 0 and
            ((ks_state = 0) and
            (:dw not in (0, 6)) or
            ((ks_state in (0, 1)) and
            (:dw in (0, 6)))) and
            :rep_date between ks_date_begin and ks_date_end and
            ks.ks_date_begin >= :rep_date - :day_count_prior and
            kont.k_status > -1
            /*and kont.k_id not in
  (
    select cont.contr_from_k_id from controller cont where cont.contr_is_delete = 0
  )  */
      order by kont.k_disb_date, :reg_id, :kf_id
      into :k_id, :kf_id, :reg_id, :k_disb_date, :state, :k_fact_disb_date, :k_fact_closed_date
  do
  begin
    is_ok = 0;
    if ((k_fact_disb_date is null) and
        (:dw not in (0, 6))) then
      is_ok = 1; -- в будние только договора в процессе

    if (:dw in (0, 6)) then
    begin
      is_ok = 1;

      if (k_fact_disb_date is not null) then
      begin
        k_disb_date = k_fact_disb_date;
      end
    end

    /*
      if ((k_fact_disb_date is not null) and (:dw in (0,6))) then
      begin
        k_disb_date =  k_fact_disb_date;
        is_ok = 1;  -- в выходные и договора в процессе и выданные
      end
      */

    if (k_fact_closed_date is not null) then
      is_ok = 0;

    if (is_ok = 1) then
    begin

      if (reg_id <> reg_id_prior) then
      begin
        select reg.reg_name
        from region reg
        where reg.reg_unicode = :reg_id
        into :reg_name;

        reg_id_prior = reg_id;
      end

      cl_checked_id = null;
      control_fio = '';
      kont_is_checked = 0;

      check_ok = 0;
      check_not_ok = 0;
      contr_check_result = 0;

      is_control_block = 0;

      select cl_cont.c_name_fio,
             (
             case
               when cont.contr_checked_date is not null then 1
               else 0
             end),
             cont.contr_from_c_id, cont.contr_check_result
      from controller cont
      inner join username un on un.un_id = cont.contr_from_un_id
      inner join clients cl_cont on cl_cont.c_id = un.un_from_c_id
      where cont.contr_from_k_id = :k_id and
            cont.contr_is_delete = 0
      into :control_fio, :kont_is_checked, :cl_checked_id, :contr_check_result;

      if (control_fio <> '') then
        is_control_block = 9;

      if (contr_check_result = 1) then
      begin
        check_ok = 1;
        is_control_block = 1;
      end

      if (contr_check_result = -1) then
      begin
        check_not_ok = 1;
        is_control_block = -1;
      end

      kk = 0;
      fname = '';
      lname = '';
      patronymic = '';

      if (control_fio <> '') then
      begin
        kk = position(' ', :control_fio) - 1;
        if (kk > 0) then
        begin
          lname = substring(:control_fio from 1 for kk);
          control_fio = substring(:control_fio from kk + 2 for 255);
          kk = position(' ', control_fio) - 1;
          if (kk > 0) then
          begin
            fname = substring(:control_fio from 1 for kk);
            control_fio = substring(:control_fio from kk + 2 for 255);
            patronymic = :control_fio;
          end
          else
            fname = :control_fio;
          control_fio = lname || ' ' || left(fname, 1) || '. ' || iif(patronymic <> '', left(patronymic, 1) || '.', '');
        end
      end

      select kal.kal_new_kf_name
      from knaktlist kal
      where kal_from_kf_id = :kf_id and
            :rep_date between kal_date and kal_date_end and
            kal.is_delete = 0
      into :kf_name;

      kf_counter = kf_counter + 1;

      for select kc.kc_from_c_id,
                 (
                 case c_citizen
                   when 8 then 'КГ'
                   when 9 then 'УЗ'
                   when 10 then 'КЗ'
                   when 12 then 'ТЖ'
                   when 13 then 'РФ'
                   else '--'
                 end) as c_citizen_str,
                 c_citizen
          from knclient kc, clients cl
          where kc_from_k_id = :k_id and
                kc_amount_approved > 0 and
                kc.kc_from_c_id = c_id
          into :c_id, cl_citizen_str, cl_citizen
      do
      begin
        cl_is_checked = 0;
        if (cl_checked_id = c_id) then
          cl_is_checked = 1;

        rep_c_address_job = null;
        select adl_address_str
        from addrlist, addrclient
        where adc_from_c_id = :c_id and
              adc_from_adl_id = adl_id and
              adc_address_type = 3
        into rep_c_address_job;
        rep_c_address_job = iif(rep_c_address_job is not null, rep_c_address_job, 'Адрес не указан');

        if ((cl_citizen_str = '--') and
            (cl_citizen <> 0)) then
        begin
          select ddr.rd_name
          from refdata ddr
          where ddr.rd_id = :cl_citizen
          into :cl_citizen_str;
        end
        counter = counter + 1;
        suspend;
      end
    end
  end
end
