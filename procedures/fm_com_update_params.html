CREATE PROCEDURE FM_COM_UPDATE_PARAMS(
    CML_ID D_ID,
    PARAM_NAME D_STR_CODE,
    PARAM_VALUE D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    NEED_REFRESH D_SMALLINT)
AS
declare variable lang d_str_lang;
begin
  -- Редактирование параметров автоматической - реструктуризации в КК
  res_text = '';
  need_refresh = 0;
  lang = f_get_lang_code(un_id, null);

  if (res_text = '' and param_name = 'grace_os_list') then
  begin
    -- В списке льготных погашений должны быть только цифры и запятые
    -- Пробелы убираем, на остальное ругаемся
    param_value = replace(coalesce(param_value, ''), ' ', '');
    if (param_value not similar to '[0123456789,]*') then
      res_text = f_lang(lang, 'Ошибка при вводе. Допускаются только цифры и запятые');
  end

  if (res_text = '' and param_name in ('new_annu', 'monthly_income')) then
  begin
    -- Сумма погашения и ежемесячный доход должны быть числом
    param_value = replace(coalesce(param_value, ''), ' ', '');
    if (param_value = '') then param_value = '0';
    if (param_value not similar to '[0123456789]+') then
      res_text = f_lang(lang, 'Ошибка при вводе. Допускаются только цифры');
    else
      param_value = cast(param_value as bigint);
  end

  if (param_value is null) then
    res_text = f_lang(lang, 'Параметр не указан');

  if (res_text = '') then
  begin
    update com_list
    set cml_add_params = f_json_str_set_value(cml_add_params, :param_name, :param_value)
    where cml_id = :cml_id;

    delete from com_list_resh
    where cmr_from_cml_id = :cml_id;

    need_refresh = 1;
  end
  suspend;
end
