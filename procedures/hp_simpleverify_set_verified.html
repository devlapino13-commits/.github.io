CREATE PROCEDURE HP_SIMPLEVERIFY_SET_VERIFIED(
    QRL_ID D_ID,
    RESPONS_C_ID D_ID)
AS
declare variable qrl_gps_lat d_double;
declare variable qrl_gps_long d_double;
declare variable tmp_un_id d_id;
begin
  if (qrl_id is null) then
    exception e_default 'qrl_id is null';

  select qrl_gps_lat, qrl_gps_long, qrf_from_un_id
  from qr_list, qr_photo
  where qrl_id = :qrl_id and
        qrf_from_qrl_id = qrl_id
  into :qrl_gps_lat, :qrl_gps_long, tmp_un_id;

  if (row_count = 0) then
    exception e_default f_format('Record not found, qrl_id: %1', qrl_id);

  if (respons_c_id is null) then
  begin
    select un_from_c_id
    from username
    where un_id = :tmp_un_id
    into respons_c_id;
  end

  insert into qr_verify (qrv_from_qrl_id, qrv_from_c_id, qrv_gps_lat, qrv_gps_long)
  values (:qrl_id, :respons_c_id, :qrl_gps_lat, :qrl_gps_long);

  update qr_list
  set qrl_state = 2 -- Подтверждение
  where qrl_id = :qrl_id;
end
