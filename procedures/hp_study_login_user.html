CREATE PROCEDURE HP_STUDY_LOGIN_USER(
    UN_ID D_ID)
RETURNS (
    ERROR_TEXT D_STR_255,
    USER_NAME D_STR_100,
    REGION_NAME VARCHAR(50),
    OFFICE_NAME VARCHAR(50),
    ACCESS D_SMALLINT,
    JOBPOSITION D_STR_100,
    TEST_ACCESS D_SMALLINT)
AS
declare variable const_err d_str_255;
declare variable c_id d_id;
declare variable ps_is_working d_smallint;
declare variable list_jl_id d_str_255;
declare variable jl_id d_id;
declare variable reg_id d_integer;
declare variable visit_qty d_integer;
declare variable qty_days d_integer;
declare variable date_off d_date;
declare variable ho_id d_id;
declare variable i_date_begin d_date;
declare variable jk_system_code d_str_100;
declare variable test_access_hour d_integer;
declare variable test_access_day d_integer;
begin
  test_access = 0;

  --Для теста
  if (un_id = 9990100005551) then
  begin
    error_text = '';
    user_name = 'Иванов Иван Иванович';
    region_name = 'Иссык-Кульский р.';
    office_name = 'Балыкчи';
    access = 1;
    jobposition = 'Стажер СпФ';
    test_access = 1;
    suspend;
    exit;
  end 

  --Для теста
  if (un_id = 9990100005552) then
  begin
    error_text = '';
    user_name = 'Петров Петр Петрович';
    region_name = 'Ошский р.';
    office_name = 'Ош';
    access = 1;
    jobposition = 'Стажер Кассир';
    suspend;
    exit;
  end 

  if (un_id = 9990100005553) then
  begin
    error_text = '';
    user_name = 'Олегов Олег Олегович';
    region_name = 'Головной офис';
    office_name = 'Бишкек г. (ГО)';
    access = 1;
    jobposition = 'Операционный менеджер';
    suspend;
    exit;
  end 

  if (un_id = 9990100005554) then
  begin
    error_text = '';
    user_name = 'Тестов Тест Тестович';
    region_name = 'Иссык-Кульский р.';
    office_name = 'Балыкчи';
    access = 1;
    jobposition = 'Заместитель менеджера офиса';
    suspend;
    exit;
  end 

  if (un_id = 9990100005555) then
  begin
    error_text = '';
    user_name = 'Тестовый ВИПС';
    region_name = 'Иссык-Кульский р.';
    office_name = 'Балыкчи';
    access = 1;
    jobposition = 'Специалист по финансированию';
    test_access = 1;
    suspend;
    exit;
  end 

  const_err = 'У пользователя нет доступа на данный ресурс';
  error_text = '';
  begin
    select un.un_id, un.un_from_c_id
    from username un
    where un.un_id = :un_id
    into un_id, c_id;
    when any do
    begin
      error_text = 'Идентификатор пользователя передан не корректно';
      suspend;
      exit;
    end
  end

  select cl.c_name_fio
  from clients cl
  where cl.c_id = :c_id
  into :user_name;

  --По запросу Jira KG 742
  if (un_id in (1090100001522,  --AKHAMZEEVA
                1090100003285,  --NMUSURMANKULOV
                1090100001547,  --CHEDILBEKOVA
                1090100000023,  --RHAYIRDINOVA
                1090100001337,  --SHTASHHODJAEV
                1090100001797,  --KABDYMALIKOVA
                1090100005519,  --ASANALIEVAM
                1090100005727,  --NTYNYMBEK
                1090100005871,  --AOMURBEKOV
                1090100000524,
                1090100001499,  --AZHORAEV
                1090100002738,   --CHBOZHOKOVA
                1090100003947
                )) then
  begin
    error_text = '';
    region_name = 'Головной офис';
    office_name = 'Бишкек г. (ГО)';
    access = 1;
    jobposition = iif(un_id in (1090100000023, 1090100001337, 1090100001797, 1090100000524), trim('Старший Региональный менеджер'), trim('Региональный менеджер'));

    if (un_id = 1090100001499) then
      jobposition = 'Специалист по финансированию';
    if (un_id = 1090100003947) then
    begin
      jobposition = 'Специалист по финансированию';
      user_name = 'Демо 1';
    end

    reg_id = 1090100;

    if (un_id in (1090100005519, 1090100005727, 1090100005871, 1090100002738)) then
    begin
      jobposition = 'Менеджер офиса';

      select upvs.upvs_first_visit_date
      from un_portal_visit_svod upvs
      where upvs.upvs_from_un_id = :un_id
      into date_off;

      if (current_date > dateadd(month, 3, date_off)) then
      begin
        access = -1; --Доступа нет
        error_text = const_err;
        suspend;
        exit;
      end 
    end

    insert into un_portal_visit (upv_from_un_id, upv_office_id, upv_datetime, upv_date)
    values (:un_id, :reg_id, current_timestamp, current_date);

    select count(*)
    from un_portal_visit upv
    where upv.upv_from_un_id = :un_id
    into visit_qty;

    visit_qty = coalesce(visit_qty, 0);

    if (not(exists(select *
                   from un_portal_visit_svod upvs
                   where upvs.upvs_from_un_id = :un_id))) then
    begin
      insert into un_portal_visit_svod (upvs_from_un_id, upvs_visit_qty, upvs_first_visit_datetime, upvs_first_visit_date, upvs_office_id)
      values (:un_id, :visit_qty + 1, current_timestamp, current_date, :reg_id);
    end
    else
    begin
      update un_portal_visit_svod upvs
      set upvs.upvs_visit_qty = :visit_qty + 1
      where upvs.upvs_from_un_id = :un_id;
    end

    suspend;
    exit;
  end 

  select '.' || list(jl.jl_id, '.') || '.'
  from joblist jl
  inner join jobkind jk on jk.jk_id = jl.jl_kind
  where (jk.jk_system_code in ('jobsystem.lo_standart',
                               'jobsystem.lo_senior',
                               'jobsystem.region_km',
                               'jobsystem.region_director',
                               'JobSystem.Specialist_KOBD',
                               'JobSystem.Operational_manager') or
         jk.jk_name containing 'Заместитель менеджера офиса')
  into list_jl_id;

  select rg.reg_name, rg2.reg_name, ps.ps_is_working, ps.ps_from_jl_id, rg.reg_unicode, jls.jl_name, ps.ps_from_ho_id, jk.jk_system_code
  from hr_pstatus ps
  inner join joblist jls on ps.ps_from_jl_id = jls.jl_id
  inner join jobkind jk on jk.jk_id = jls.jl_kind
  inner join region rg on rg.reg_unicode = ps.ps_from_reg_id
  inner join region rg2 on rg2.reg_unicode = rg.parent
  where ps.ps_from_c_id = :c_id and
        ps.ps_status = 1 and
        current_date between ps.ps_date_begin and ps.ps_date_end
  into office_name, region_name, ps_is_working, jl_id, reg_id, jobposition, ho_id, jk_system_code;

  if (coalesce(ho_id, -1) = -1 or  coalesce(ps_is_working, 0) = 0) then
  begin
    select rg.reg_name, rg2.reg_name, 1, i.i_from_jl_id, rg.reg_unicode, jls.jl_name, i.i_date_begin
    from interns i
    inner join joblist jls on jls.jl_id = i.i_from_jl_id
    inner join region rg on rg.reg_unicode = i.i_from_reg_id
    inner join region rg2 on rg2.reg_unicode = rg.parent
    where i.i_from_c_id = :c_id and
          i.i_is_delete = 0
    order by i.i_date_begin desc
    rows 1
    into office_name, region_name, ps_is_working, jl_id, reg_id, jobposition, i_date_begin;

    ps_is_working = coalesce(ps_is_working, 0);

    if (exists(select *
               from un_portal_visit_svod upvs
               where upvs.upvs_from_un_id = :un_id and
                     cast(upvs.upvs_first_visit_datetime as date) < :i_date_begin)) then
    begin
      select f_date_work_nday(upv.upv_date, 10)
      from un_portal_visit upv
      where upv.upv_from_un_id = :un_id and
            upv.upv_date >= :i_date_begin
      order by upv.upv_date
      rows 1
      into date_off;
    end
    else
    begin
      select f_date_work_nday(upvs.upvs_first_visit_date, 10)
      from un_portal_visit_svod upvs
      where upvs.upvs_from_un_id = :un_id
      into date_off;
    end 

    date_off = coalesce(date_off, current_date);

    if (current_date > date_off) then
    begin
      access = -1; --Доступа нет
      error_text = const_err;
      suspend;
      exit;
    end
  end 

  if ((coalesce(:ps_is_working,0) = 0) or (:list_jl_id not containing ('.' || :jl_id || '.'))) then
  begin
    access = -1; --Доступа нет
    error_text = const_err;
  end
  else
    access = 1; -- Доступ есть

  if (un_id in (1090100002567,         --EKYDYKBEK
                1090100000343,         --ADERKEMBAEVA
                1090100004389,         --RBOGDANOVA
                1090100003399)) then   --JSAPARBAEV
  begin
    jobposition = 'Менеджер офиса';
  end 


  if (access = 1) then
  begin
    insert into un_portal_visit (upv_from_un_id, upv_office_id, upv_datetime, upv_date)
    values (:un_id, :reg_id, current_timestamp, current_date);

    select count(*)
    from un_portal_visit upv
    where upv.upv_from_un_id = :un_id
    into visit_qty;

    visit_qty = coalesce(visit_qty, 0);

    if (not(exists(select *
                   from un_portal_visit_svod upvs
                   where upvs.upvs_from_un_id = :un_id))) then
    begin
      insert into un_portal_visit_svod (upvs_from_un_id, upvs_visit_qty, upvs_first_visit_datetime, upvs_first_visit_date, upvs_office_id)
      values (:un_id, :visit_qty + 1, current_timestamp, current_date, :reg_id);
    end
    else
    begin
      update un_portal_visit_svod upvs
      set upvs.upvs_visit_qty = :visit_qty + 1
      where upvs.upvs_from_un_id = :un_id;
    end 
    
    test_access_hour = extract(hour from current_timestamp) * 100 + extract(minute from current_timestamp);
    test_access_day = extract(day from current_date);

    if (jk_system_code in ('jobsystem.lo_standart', 'jobsystem.lo_senior') and
        test_access_day between 10 and 15 and
        test_access_hour between 1900 and 2359) then
      test_access = 1;
  end 

  suspend;
end
