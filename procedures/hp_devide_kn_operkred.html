CREATE PROCEDURE HP_DEVIDE_KN_OPERKRED(
    OK_FROM_K_ID ID_TYPE,
    OK_SH_ID INTEGER,
    OK_SH_DATE DATE,
    OK_TYPE INTEGER,
    OK_OS CURR_TYPE,
    OK_PERC CURR_TYPE,
    OK_FINE CURR_TYPE,
    OK_KOMMIS CURR_TYPE,
    OK_SH_OS CURR_TYPE,
    OK_SH_PERC CURR_TYPE,
    OK_KOMMIS_ADV CURR_TYPE,
    OK_TAX CURR_TYPE,
    OK_OL_IS_ACTIVE INTEGER,
    OK_OL_DATE DATE)
RETURNS (
    KC_ID ID_TYPE,
    RES_OK_OS CURR_TYPE,
    RES_OK_PERC CURR_TYPE,
    RES_OK_FINE CURR_TYPE,
    RES_OK_KOMMIS CURR_TYPE,
    RES_OK_SH_OS CURR_TYPE,
    RES_OK_SH_PERC CURR_TYPE,
    RES_OK_KOMMIS_ADV CURR_TYPE,
    RES_OK_TAX CURR_TYPE)
AS
declare variable k_amount curr_type;
declare variable kc_amount curr_type;
declare variable kc_count d_integer;
declare variable curr_line d_integer;
declare variable saldo_ok_os curr_type;
declare variable saldo_ok_perc curr_type;
declare variable saldo_ok_fine curr_type;
declare variable saldo_ok_kommis curr_type;
declare variable saldo_ok_sh_os curr_type;
declare variable saldo_ok_sh_perc curr_type;
declare variable saldo_ok_kommis_adv curr_type;
declare variable saldo_ok_tax curr_type;
declare variable koef double precision;
declare variable round_scale integer;
declare variable tmp_amount curr_type;
declare variable max_diff curr_type;
declare variable diff_ok_fine curr_type;
declare variable diff_ok_kommis curr_type;
declare variable diff_ok_kommis_adv curr_type;
declare variable diff_ok_tax curr_type;
declare variable res_ok_annu curr_type;
declare variable diff_ok_annu curr_type;
declare variable last_kc_id id_type;
declare variable gr_zad_os curr_type;
declare variable gr_zad_perc curr_type;
declare variable arr_zad_os varchar(400);
declare variable arr_zad_perc varchar(400);
declare variable zad_os decimal(18,4);
declare variable zad_perc decimal(18,4);
declare variable ol_id decimal(18,0);
declare variable ok_number integer;
declare variable k_count integer;
declare variable try_count integer;
declare variable round_scale_perc integer;
declare variable need_round_os_2 integer;
declare variable need_round_perc_2 integer;
declare variable min_diff_os curr_type;
declare variable min_diff_perc curr_type;
declare variable last_kc_line integer;
declare variable new_saldo_os curr_type;
declare variable new_saldo_perc curr_type;
declare variable round_scale_os integer = 0;
declare variable zad_balance curr_type;
declare variable arr_zad_balance varchar(400);
declare variable gr_zad_balance curr_type;
declare variable disb_date date;
declare variable kn_disb_amount curr_type;
declare variable new_saldo_balance curr_type;
declare variable min_diff_balance curr_type;
begin
  round_scale = 2;

  select k_amount_approved, k_kolich_client_approved, k_disb_date
  from kontract
  where k_id = :ok_from_k_id
  into k_amount, k_count, disb_date;

-- если участников группы только 1, возвращаем как есть
  if (k_count = 1) then
  begin
    select kc_id
    from knclient
    where kc_from_k_id = :ok_from_k_id and
          kc_amount_approved > 0
    into :kc_id;

    res_ok_os = ok_os;
    res_ok_perc = ok_perc;
    res_ok_fine = ok_fine;
    res_ok_kommis = ok_kommis;
    res_ok_sh_os = ok_sh_os;
    res_ok_sh_perc = ok_sh_perc;
    res_ok_kommis_adv = ok_kommis_adv;
    res_ok_tax = ok_tax;

    suspend;
    exit;

    when any do
      exception e_default 'Client count is differ: kontract: 1, knclient > 1';
  end

  kc_count = 0;

  gr_zad_os = 0;
  gr_zad_perc = 0;
  gr_zad_balance = 0;

  min_diff_os = 0;
  min_diff_perc = 0;
  min_diff_balance = 0;

  arr_zad_os = '';
  arr_zad_perc = '';
  arr_zad_balance = '';

  if (ok_ol_date > disb_date and
      ok_type = 1 and
      ok_sh_os <> 0) then
  begin
-- если идет строка графика, проверяем общую сумму.
-- если она уже выровнена, не распределяем
    select sum(ok_sh_os)
    from kn_operkred, knclient
    where kc_from_k_id = :ok_from_k_id and
          kc_amount_approved > 0 and
          ok_from_kc_id = kc_id and
          ok_ol_is_active = 1 and
          ok_type = 1
    into :kn_disb_amount;
    if (kn_disb_amount is null) then
      kn_disb_amount = 0;
    if (kn_disb_amount = k_amount) then
      ok_sh_os = 0;
  end

  need_round_os_2 = 0;
  need_round_perc_2 = 0;
  round_scale_os = 0;
  round_scale_perc = 0;
  try_count = 1;

  if (ok_os + ok_sh_os <> round(ok_os + ok_sh_os, 0) or
     (ok_os <> 0 and abs(ok_os) < (k_count * 2))) then
  begin
    need_round_os_2 = 1;
    round_scale_os = 2;
  end

  if (ok_perc + ok_sh_perc <> round(ok_perc + ok_sh_perc, 0) or
     (ok_perc <> 0 and abs(ok_perc) < (k_count * 2))) then
  begin
    need_round_perc_2 = 1;
    round_scale_perc = 2;
  end

  if (ok_type not in (1, 8) and
      (ok_os <> 0 or ok_sh_os <> 0 or ok_perc <> 0 or ok_sh_perc <> 0)) then
  begin
    for select kc_id
        from knclient
        where kc_from_k_id = :ok_from_k_id and
              kc_amount_approved > 0
        into kc_id
    do
    begin
      zad_os = 0;
      zad_perc = 0;
      zad_balance = 0;

      if (ok_type = 9 and
          ok_os > 0) then
      begin
        zad_os = 0;
        select sum(ok_os)
        from kn_operkred
        where ok_from_kc_id = :kc_id and
              ok_ol_is_active = 1 and
              ok_ol_date <= :ok_ol_date and
              ok_type = 9
        into zad_os;
        if (zad_os is null) then
          zad_os = 0;
        zad_balance = zad_os;

        if (ok_os <> 0 and
            zad_os <> round(zad_os)) then
        begin
          need_round_os_2 = 1;
          round_scale_os = 2;
        end
      end
      else
      begin
        select sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from kn_operkred
        where ok_from_kc_id = :kc_id and
              ok_sh_date = :ok_sh_date and
              ok_ol_is_active = 1 and
              ok_type < 7
        into zad_os, zad_perc;

        if (zad_os is null) then
          zad_os = 0;
        if (zad_perc is null) then
          zad_perc = 0;

        select oks_balance
        from kn_oksummary
        where oks_from_kc_id = :kc_id and
              oks_date_begin <= :ok_ol_date
        order by oks_from_kc_id desc, oks_date_begin desc
        rows 1
        into zad_balance;

        if (zad_balance is null) then
          zad_balance = 0;

        if ((ok_sh_os <> 0 and
            zad_os <> round(zad_os)) or (ok_os <> 0 and
            zad_balance <> round(zad_balance))) then
        begin
          need_round_os_2 = 1;
          round_scale_os = 2;
        end
        if ((ok_perc <> 0 or ok_sh_perc <> 0) and
            zad_perc <> round(zad_perc)) then
        begin
          need_round_perc_2 = 1;
          round_scale_perc = 2;
        end
      end

      gr_zad_os = gr_zad_os + zad_os;
      gr_zad_perc = gr_zad_perc + zad_perc;
      gr_zad_balance = gr_zad_balance + zad_balance;

      arr_zad_os = arr_zad_os || right('                    ' || zad_os, 20);
      arr_zad_perc = arr_zad_perc || right('                    ' || zad_perc, 20);
      arr_zad_balance = arr_zad_balance || right('                    ' || zad_balance, 20);
    end
    if (gr_zad_balance = gr_zad_os) then
      arr_zad_os = arr_zad_balance;
    if (gr_zad_balance > gr_zad_os) then
    begin
      arr_zad_balance = arr_zad_os;
      gr_zad_balance = gr_zad_os;
    end

    new_saldo_os = ok_os + ok_sh_os + gr_zad_os;
    new_saldo_perc = ok_perc + ok_sh_perc + gr_zad_perc;
    new_saldo_balance = ok_os + ok_sh_os + gr_zad_balance;

    if (ok_sh_os <> 0 and
        abs(new_saldo_os) >= k_count) then
      min_diff_os = sign(new_saldo_os);

    if (ok_os <> 0 and
        abs(new_saldo_balance) >= k_count) then
      min_diff_balance = sign(new_saldo_balance);

    if ((ok_sh_perc <> 0 or ok_perc <> 0) and
        abs(new_saldo_perc) >= k_count) then
      min_diff_perc = sign(new_saldo_perc);
  end

-- максимально допустимые отклонения, меньше которой будем выравнивать
  max_diff = (case
                when k_count = 2 then 2
                when k_count = 3 then 3
                when k_count <= 6 then 4
                else 5
              end);

  max_diff = 5;
  try_count = 0;

  while (try_count < 2) do
  begin
-- при try_count = 1 проверяем ожидаемые остатки по ОС и %% для определения разряда округления
-- при try_count = 2 округляем по нужнему разряду и выводим результаты
    try_count = try_count + 1;

    if (try_count = 2) then
    begin
      if (need_round_os_2 = 1) then
      begin
        round_scale_os = 2;
        min_diff_os = min_diff_os / 100.0;
        min_diff_balance = min_diff_balance / 100.0;
      end
      if (need_round_perc_2 = 1) then
      begin
        round_scale_perc = 2;
        min_diff_perc = min_diff_perc / 100.0;
      end
    end

    saldo_ok_os = ok_os;
    saldo_ok_perc = ok_perc;
    saldo_ok_fine = ok_fine;
    saldo_ok_kommis = ok_kommis;
    saldo_ok_sh_os = ok_sh_os;
    saldo_ok_sh_perc = ok_sh_perc;
    saldo_ok_kommis_adv = ok_kommis_adv;
    saldo_ok_tax = ok_tax;

    diff_ok_annu = 0;
    diff_ok_fine = 0;
    diff_ok_kommis = 0;
    diff_ok_kommis_adv = 0;
    diff_ok_tax = 0;

    curr_line = 0;
    kc_count = 0;

    for select kc_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :ok_from_k_id and
              kc_amount_approved > 0
        into kc_id, kc_amount
    do
    begin
      kc_count = kc_count + 1;

      curr_line = curr_line + 1;
      zad_os = 0;
      zad_perc = 0;
      zad_balance = 0;
      if (arr_zad_os <> '') then
      begin
        zad_os = substring(arr_zad_os from ((curr_line - 1) * 20 + 1) for 20);
        zad_perc = substring(arr_zad_perc from ((curr_line - 1) * 20 + 1) for 20);
        zad_balance = substring(arr_zad_balance from ((curr_line - 1) * 20 + 1) for 20);
      end

      koef = cast(kc_amount as double precision) / k_amount;

      res_ok_os = 0;
      res_ok_sh_os = 0;
      res_ok_perc = 0;
      res_ok_sh_perc = 0;

      if (ok_type in (1, 8)) then -- выдача и график погашений
      begin
        res_ok_os = round(koef * ok_os, round_scale_os);
        res_ok_sh_os = round(koef * ok_sh_os, round_scale_os);

        res_ok_perc = round(koef * ok_perc, round_scale_perc);
        res_ok_sh_perc = round(koef * ok_sh_perc, round_scale_perc);
      end
      else
      begin
-- погашения
        if (ok_os <> 0) then
          res_ok_os = round(koef * new_saldo_balance, round_scale_os) - zad_balance;
        else
        if (ok_sh_os <> 0) then
          res_ok_sh_os = round(koef * new_saldo_os, round_scale_os) - zad_os;

        if (ok_perc <> 0) then
          res_ok_perc = round(koef * new_saldo_perc, round_scale_perc) - zad_perc;
        else
        if (ok_sh_perc <> 0) then
          res_ok_sh_perc = round(koef * new_saldo_perc, round_scale_perc) - zad_perc;
      end

      if (curr_line <> k_count) then
      begin
        res_ok_annu = round(koef * (ok_os + ok_perc) + diff_ok_annu, round_scale);
        res_ok_fine = round(koef * ok_fine + diff_ok_fine, round_scale);
        res_ok_kommis = round(koef * ok_kommis + diff_ok_kommis, round_scale);
        res_ok_kommis_adv = round(koef * ok_kommis_adv + diff_ok_kommis_adv, round_scale);
        res_ok_tax = round(koef * ok_tax + diff_ok_tax, round_scale);

-- если это выдача, берем сумму займа участника
        if (ok_type = 8 and
            ok_ol_is_active = 1) then
        begin
          if (ok_os = k_amount) then
            res_ok_os = kc_amount;
          else
          begin
            res_ok_os = round(koef * ok_os, 0);
            if (abs(res_ok_os - kc_amount) <= max_diff * 2) then
              res_ok_os = kc_amount;
          end
        end

        if (min_diff_os <> 0 and
            new_saldo_os <> 0) then
        begin
          if (ok_sh_os <> 0 and
              zad_os + res_ok_sh_os = 0) then
            res_ok_sh_os = res_ok_sh_os + min_diff_os;
        end

        if (min_diff_balance <> 0 and
            new_saldo_balance <> 0) then
        begin
          if (ok_os <> 0 and
              zad_balance + res_ok_os = 0) then
            res_ok_os = res_ok_os + min_diff_balance;
        end

        if (min_diff_perc <> 0 and
            new_saldo_perc <> 0) then
        begin
          if (ok_perc <> 0 and
              zad_perc + res_ok_perc = 0) then
            res_ok_perc = res_ok_perc + min_diff_perc;

          if (ok_sh_perc <> 0 and
              zad_perc + res_ok_sh_perc = 0) then
            res_ok_sh_perc = res_ok_sh_perc + min_diff_perc;
        end

-- если общая сумма ОС и %% на уровне группы равен нулю,
-- результат тоже выдаем такой же
        if (ok_os <> 0 and
            ok_perc <> 0 and
            ok_os + ok_perc = 0) then
          res_ok_perc = -res_ok_os;

        if (ok_sh_os <> 0 and
            ok_sh_perc <> 0 and
            ok_sh_os + ok_sh_perc = 0) then
          res_ok_sh_perc = -res_ok_sh_os;
-- конец проверки равенства

        if (k_count > 2 and
            mod(ok_sh_id + k_count - 1, k_count) + 1 <> curr_line) then
        begin
          diff_ok_annu = round(koef * (ok_os + ok_perc) + diff_ok_annu, 4) - res_ok_annu;
          diff_ok_fine = round(koef * ok_fine + diff_ok_fine, 4) - res_ok_fine;
          diff_ok_kommis = round(koef * ok_kommis + diff_ok_kommis, 4) - res_ok_kommis;
          diff_ok_kommis_adv = round(koef * ok_kommis_adv + diff_ok_kommis_adv, 4) - res_ok_kommis_adv;
          diff_ok_tax = round(koef * ok_tax + diff_ok_tax, 4) - res_ok_tax;
        end

        saldo_ok_os = saldo_ok_os - res_ok_os;
        saldo_ok_perc = saldo_ok_perc - res_ok_perc;
        saldo_ok_fine = saldo_ok_fine - res_ok_fine;
        saldo_ok_kommis = saldo_ok_kommis - res_ok_kommis;
        saldo_ok_sh_os = saldo_ok_sh_os - res_ok_sh_os;
        saldo_ok_sh_perc = saldo_ok_sh_perc - res_ok_sh_perc;
        saldo_ok_kommis_adv = saldo_ok_kommis_adv - res_ok_kommis_adv;
        saldo_ok_tax = saldo_ok_tax - res_ok_tax;

        if (try_count = 1) then
        begin
          if (ok_type between 2 and 7 and
              ok_sh_os <> 0) then
          begin
            if (new_saldo_os = 0 and
                (zad_os + res_ok_os + res_ok_sh_os) <> 0) then
              need_round_os_2 = 1;
            if (new_saldo_os <> 0 and
                (zad_os + res_ok_os + res_ok_sh_os) = 0) then
              need_round_os_2 = 1;
          end

          if ((ok_type between 2 and 7 or ok_type = 9) and
              ok_os <> 0) then
          begin
            if (new_saldo_balance = 0 and
                (zad_balance + res_ok_os) <> 0) then
              need_round_os_2 = 1;
            if (new_saldo_balance <> 0 and
                (zad_balance + res_ok_os) = 0) then
              need_round_os_2 = 1;
          end

          if (ok_type between 2 and 7 and
              (ok_perc <> 0 or ok_sh_perc <> 0)) then
          begin
            if (new_saldo_perc = 0 and
                (zad_perc + res_ok_perc + res_ok_sh_perc) <> 0) then
              need_round_perc_2 = 1;
            if (new_saldo_perc <> 0 and
                (zad_perc + res_ok_perc + res_ok_sh_perc) = 0) then
              need_round_perc_2 = 1;
          end
        end
        else
          suspend;

      end
      else
      begin
        last_kc_id = kc_id;
        last_kc_line = curr_line;
      end

      if (ok_type = 8 and
          ok_ol_is_active = 1) then
      begin
        tmp_amount = 0;
        select sum(ok_sh_os), max(ok_from_ol_id), max(ok_number)
        from kn_operkred
        where ok_from_kc_id = :kc_id and
              ok_ol_is_active = 1 and
              ok_type = 1 and
              ok_sh_os <> 0
        into :tmp_amount, ol_id, ok_number;
        if (tmp_amount is null) then
          tmp_amount = 0;
        if (ok_number is null) then
          ok_number = 0;

        if (tmp_amount <> kc_amount and
            ok_number > 0) then
        begin
          update kn_operkred
          set ok_sh_os = ok_sh_os - (:tmp_amount - :kc_amount)
          where ok_from_ol_id = :ol_id and
                ok_number = :ok_number and
                ok_from_kc_id = :kc_id;
        end
      end
    end -- KnClient

    if (kc_count <> k_count) then
      exception e_default 'Client count is differ: kontract: ' || k_count || ', knclient: ' || kc_count;

    if (last_kc_id is null) then
      exception e_default 'Last client not found: ' || ok_from_k_id;

    kc_id = last_kc_id;
    zad_os = 0;
    zad_perc = 0;
    zad_balance = 0;
    if (arr_zad_os <> '') then
    begin
      zad_os = substring(arr_zad_os from ((last_kc_line - 1) * 20 + 1) for 20);
      zad_perc = substring(arr_zad_perc from ((last_kc_line - 1) * 20 + 1) for 20);
      zad_balance = substring(arr_zad_balance from ((last_kc_line - 1) * 20 + 1) for 20);
    end

    res_ok_os = saldo_ok_os;
    res_ok_perc = saldo_ok_perc;
    res_ok_sh_os = saldo_ok_sh_os;
    res_ok_sh_perc = saldo_ok_sh_perc;

    res_ok_fine = saldo_ok_fine;
    res_ok_kommis = saldo_ok_kommis;
    res_ok_kommis_adv = saldo_ok_kommis_adv;
    res_ok_tax = saldo_ok_tax;

    if (try_count = 1) then
    begin
      if (ok_type between 2 and 7 and
          ok_sh_os <> 0) then
      begin
        if (new_saldo_os = 0 and
            (zad_os + res_ok_os + res_ok_sh_os) <> 0) then
          need_round_os_2 = 1;
        if (new_saldo_os <> 0 and
            (zad_os + res_ok_os + res_ok_sh_os) = 0) then
          need_round_os_2 = 1;
      end

      if ((ok_type between 2 and 7 or ok_type = 9) and
          ok_os <> 0) then
      begin
        if (new_saldo_balance = 0 and
            (zad_balance + res_ok_os) <> 0) then
          need_round_os_2 = 1;
        if (new_saldo_balance <> 0 and
            (zad_balance + res_ok_os) = 0) then
          need_round_os_2 = 1;
      end

      if (ok_type between 2 and 7 and
          (ok_perc <> 0 or ok_sh_perc <> 0)) then
      begin
        if (new_saldo_perc = 0 and
            (zad_perc + res_ok_perc + res_ok_sh_perc) <> 0) then
          need_round_perc_2 = 1;
        if (new_saldo_perc <> 0 and
