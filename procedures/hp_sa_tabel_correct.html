CREATE PROCEDURE HP_SA_TABEL_CORRECT
RETURNS (
    SAT_FROM_C_ID DECIMAL(18,0),
    SAT_YEAR_MONTH INTEGER,
    SAT_FROM_SAWT_ID INTEGER,
    DD DATE,
    OFFICE_MIN INTEGER,
    OFFICE_MAX INTEGER,
    PS_FROM_REG_ID INTEGER,
    REG_FOUND INTEGER,
    SAT_TOTAL_HOURS INTEGER)
AS
declare variable sat_year integer;
declare variable sat_month integer;
begin
  for select sat_from_c_id, sat_year_month, sat_from_sawt_id, max(sat_total_hours), min(sat_office_id),
             max(sat_office_id)
      from sa_tabel
      where sat_is_ready = 1 and
            (sat_total_hours <> 0 or
            sat_total_calc <> 0)
      group by sat_from_c_id, sat_year_month, sat_from_sawt_id
      having count(*) > 1
      into sat_from_c_id, sat_year_month, sat_from_sawt_id, sat_total_hours, office_min, office_max
  do
  begin
    sat_month = mod(sat_year_month, 100);
    sat_year = (sat_year_month - sat_month) / 100;

    if (sat_month < 12) then
      dd = cast('01.' || right('00' ||(sat_month + 1), 2) || '.' || sat_year as date) - 1;
    else
      dd = cast('01.01.' || sat_year as date);

    ps_from_reg_id = null;
    select ps_from_reg_id
    from hr_pstatus
    where ps_status = 1 and
          ps_from_c_id = :sat_from_c_id and
          ps_date_begin <= :dd and
          ps_date_end >= :dd
    order by ps_is_working desc
    rows 1
    into :ps_from_reg_id;

    if ((ps_from_reg_id = office_min) or (ps_from_reg_id = office_max)) then
      reg_found = 1;
    else
      reg_found = 0;

    if (reg_found = 1) then
    begin
      delete from sa_tabel
      where sat_is_ready = 1 and
            (sat_total_hours <> 0 or
            sat_total_calc <> 0) and
            sat_from_c_id = :sat_from_c_id and
            sat_year_month = :sat_year_month and
            sat_from_sawt_id = :sat_from_sawt_id and
            sat_office_id <> :ps_from_reg_id;
    end
    suspend;
  end
end
