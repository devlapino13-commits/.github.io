CREATE PROCEDURE HP_UNDO_RESCHEDULE_FLOAT(
    K_ID D_ID,
    REP_DATE D_DATE,
    UN_ID D_ID)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable kd_ol_date d_date;
declare variable reg_id d_integer;
declare variable kf_id d_id;
declare variable k_kontract_no d_str_code;
declare variable k_kontract_date d_date;
declare variable ol_memo d_str_long;
declare variable tmp_amount d_currency;
declare variable kd_percent_effect d_percent;
declare variable res_ol_id d_id;
declare variable new_kd_id d_integer;
declare variable db_id d_integer;
declare variable close_date d_date;
declare variable kd_id d_integer;
declare variable old_ol_id d_id;
declare variable kd_is_primary d_bool;
begin
  -- Отмена последнего повышения плавающей процентной ставки
  -- Используется где по вине клиента погашение не попал куда надо во время и ставка была повышена.

  db_id = f_get_db_id();
  err_text = '';

  select k_from_kf_id, k_kontract_no, k_kontract_date
  from kontract, product pr
  where k_id = :k_id and
        k_from_pr_id = pr_id and
        pr_percent_kind = 1 -- плавающая процентная ставка
  into kf_id, k_kontract_no, k_kontract_date;

  if (kf_id is null) then
    err_text = 'Контракт с плавающей процентной ставкой не найден';

  if (err_text = '') then
  begin
    select kd_ol_date, kd_from_ol_id, kd_is_primary
    from kontdata
    where kd_from_k_id = :k_id and
          kd_ol_is_delete = 0 and
          kd_ol_date <= kd_date_end and
          kd_ol_date <= :rep_date
    order by kd_ol_date desc
    rows 1
    into kd_ol_date, old_ol_id, kd_is_primary;
  
    if (kd_is_primary = 1) then
      err_text = 'По контракту не было повыщений процентной ставки';
  end

  if (err_text = '' and db_id = 81001) then
  begin
    close_date = f_get_const('kn_operbook_close_date');
    if (rep_date <= close_date) then
      err_text = 'Месяц закрыт';
  end

  if (exists(select * from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        (ok_ol_date > :rep_date or
        (ok_ol_date = :rep_date and ok_from_ol_id > :old_ol_id)))) then
    err_text = 'Существуют операции после даты восстановления';

  if (err_text = '' and kd_ol_date = rep_date) then
  begin
    -- Если дата восстановления равна дате повышения, просто аннулируем операцию повышения ставки
    update operlist
    set is_delete = 1
    where ol_id = :old_ol_id;

    exit;
  end

  if (err_text = '') then
  begin
    select kal_new_reg_id
    from knaktlist k
    where k.kal_from_kf_id = :kf_id and
          k.is_delete = 0 and
          k.kal_date <= :rep_date and
          k.kal_date_end >= :rep_date
    into reg_id;
  
    if (reg_id is null) then
      err_text = 'Офис учета кредита не найден';
  end

  if (err_text = '') then
  begin
    select kd_id
    from kontdata
    where kd_from_k_id = :k_id and
          kd_ol_is_delete = 0 and
          kd_ol_date <= kd_date_end and
          kd_ol_date < :kd_ol_date
    order by kd_ol_date desc
    rows 1
    into kd_id;
  
    if (kd_id is null) then
      err_text = 'Не найдена предыдущая ставка';
  end

  if (err_text <> '') then
  begin
    suspend;
    exit;
  end

  ol_memo = f_format('Восстановление плавающей процентной ставки по договору %1 от %2', k_kontract_no, f_datetostr(k_kontract_date));

  insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_memo, is_delete,
                        ol_user_id)
  values (:reg_id, 1, 12, :rep_date, 9, :ol_memo, 0, :un_id)
  returning ol_id
  into :res_ol_id;

  insert into kontdata (kd_from_k_id, kd_is_primary, kd_from_ol_id, kd_ol_date, kd_ol_is_delete, kd_percent,
                        kd_percent_per_day, kd_percent_effect, kd_percent_effect_fifd, kd_memo,
                        kd_fine_percent, kd_float_perc_level)
  select kd_from_k_id, kd_is_primary, :res_ol_id, :rep_date, 0, kd_percent, kd_percent_per_day,
         kd_percent_effect, kd_percent_effect_fifd, 'Восстановление предыдущей плавающей процентной ставки', kd_fine_percent, kd_float_perc_level
  from kontdata
  where kd_id = :kd_id
  returning kd_id
  into new_kd_id;

  -- Построение задолженности по графику на дату
  select res_amount
  from hp_repay_fill(:k_id, 0, :rep_date, 0, 1, 1, 1, 1, 1)
  into :tmp_amount;

  -- Автореструктуризация графика
  tmp_amount = 0;
  select has_advanced_pay
  from hp_repay_reschedule(:rep_date, 3, 1, :k_id)
  into :tmp_amount; -- Имеются ли досрочные погашения
  tmp_amount = 0;

  -- Сохранение на Operkred сумм автореструктуризации
  execute procedure hp_repay_save_operkred(:k_id, 0, :res_ol_id, :rep_date, 0, 3, 0);

  select kps_psk
  from hp_edoc_ins_kontpsk(:k_id, :res_ol_id, :un_id)
  into :kd_percent_effect;

  if (kd_percent_effect is not null) then
  begin
    update kontdata
    set kd_percent_effect_fifd = :kd_percent_effect
    where kd_id = :new_kd_id;
  end

  if (db_id = 81001) then
    execute procedure hp_do_kn_operbook_one_online(:k_id, current_date, null, :rep_date - 1);

  suspend;
end
