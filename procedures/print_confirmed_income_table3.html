CREATE PROCEDURE PRINT_CONFIRMED_INCOME_TABLE3(
    K_ID D_ID)
RETURNS (
    NUMBER D_INTEGER,
    RES_MONTH D_STR_20,
    MONTHLY_INCOME D_CURRENCY,
    AMOUNT1 D_CURRENCY,
    AMOUNT2 D_CURRENCY,
    AMOUNT3 D_CURRENCY,
    AMOUNT4 D_CURRENCY)
AS
declare variable ki_survey d_str_8100;
declare variable ncount d_integer;
declare variable numb d_integer;
begin
  -- RFCRU-4782
  -- ПДН; Профсуждение;
  for
    select first 1
      ki_survey
    from kk_income
    where ki_from_k_id = :k_id
      and ki_is_manual = 0
    order by ki_id desc
  into
    :ki_survey
  do
  begin
    ncount = f_json_str_get_count(ki_survey, 'data');
    numb = 0;
    while (numb < ncount) do
    begin
      res_month = f_json_str_get_value(ki_survey, 'data[' || numb || '].nmonth', '');
      monthly_income = f_json_str_get_value(ki_survey, 'data[' || numb || '].sumt', null);
      amount1 = f_json_str_get_value(ki_survey, 'data[' || numb || '].sum1', null);
      amount2 = f_json_str_get_value(ki_survey, 'data[' || numb || '].sum2', null);
      amount3 = f_json_str_get_value(ki_survey, 'data[' || numb || '].sum3', null);
      amount4 = f_json_str_get_value(ki_survey, 'data[' || numb || '].sum4', null);
      number = f_json_str_get_value(ki_survey, 'data[' || numb || '].numb', null);
      numb = numb + 1;
      suspend;
    end
  end
end
