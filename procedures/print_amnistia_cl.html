CREATE PROCEDURE PRINT_AMNISTIA_CL(
    AAREP_K_ID ID_TYPE)
RETURNS (
    K_KONTRACT_NO VARCHAR(50),
    C_AMNISTIA_FIO VARCHAR(100),
    KC_AMOUNT_APPROVED CURR_TYPE,
    ADDRESS_YUR VARCHAR(255),
    ZAD_OS CURR_TYPE,
    ZAD_CALC_PERC CURR_TYPE,
    ZAD_OS_PERC CURR_TYPE,
    C_CITIZEN INTEGER,
    C_BIRTH_DATE DATE,
    CITIZEN VARCHAR(255),
    CZ_CL_FINE CURR_TYPE,
    C_PASS_NO VARCHAR(30),
    C_PASS_WHO STR_255,
    C_PASS_DATE DATE,
    C_INN STR_20,
    C_SNILS STR_100)
AS
declare variable kd_k_new d_id;
declare variable kd_c_id d_id;
declare variable k_id_min d_id;
declare variable k_id_max d_id;
begin
  k_id_min = aarep_k_id / 10 * 10; -- 101010017976100301 -> 101010017976100300
  k_id_max = aarep_k_id / 10 * 10 + 9; -- 101010017976100301 -> 101010017976100309
  for select kc_from_k_id, kc_from_c_id, kc_amount_approved
      from kontstate ks, knclient kc
      where ks_from_k_id between :k_id_min and :k_id_max and
            ks_ol_is_delete = 0 and
            ks_state in (1, 2) and
            current_date between ks_date_begin and ks_date_end and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0
      into kd_k_new, kd_c_id, kc_amount_approved
  do
  begin
    select c_name_fio, cl.c_citizen, c_birth_date, cl.c_pass_no,cl.c_pass_who, cl.c_pass_date, c_inn
    from clients cl
    where c_id = :kd_c_id
    into c_amnistia_fio, c_citizen, c_birth_date,c_pass_no,c_pass_who,c_pass_date,c_inn;

    select cds_doc_no
    from clientdocs
    inner join passkind on pk_id = cds_from_pk_id
    inner join passdoctype on pdt_id = pk_from_pdt_id
    where cds_from_c_id = :kd_c_id and
          pdt_code = 'snils' and
          cds_is_delete = 0
          order by cds_datetime desc rows 1
    into c_snils;

    select k_kontract_no
    from kontract
    where k_id = :kd_k_new
    into k_kontract_no;

    address_yur = null;
    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :kd_c_id and
          adc_address_type = 1 and
          adc_from_adl_id = adl_id
    into :address_yur;
    if (address_yur is null or address_yur = 'не указан') then
      address_yur = 'адрес не указан';

    select sum(cur_cl_os), sum(cur_cl_perc), sum(cur_cl_os + cur_cl_perc), sum(cz_cl_fine)
    from print_client_zadol(:kd_k_new)
    where cz_name_fio = :c_amnistia_fio
    into zad_os, zad_calc_perc, zad_os_perc, cz_cl_fine;

    select rd.rd_name
    from refdata rd
    where rd.rd_main_code = 'citizen' and
          rd_id = :c_citizen
    into citizen;

    suspend;
  end
end
