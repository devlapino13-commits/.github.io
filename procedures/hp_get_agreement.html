CREATE PROCEDURE HP_GET_AGREEMENT(
    K_ID DECIMAL(18,0))
RETURNS (
    AG_ID INTEGER,
    AG_CAPTION VARCHAR(255),
    AG_MEMO VARCHAR(255),
    AG_OBJECT_ID D_ID)
AS
declare variable k_disb_date date;
declare variable reg_id integer;
declare variable rd_id id_type;
declare variable pr_id d_integer;
begin
  for select rd_id
      from refdata
      where rd_code in ('AgreementKinds.IPF_Sellers', 'AgreementKinds.tsp_and_ipf', 'AgreementKinds.IPF_Idjara', 'AgreementKinds.IPF_WithCommission')
      into :rd_id

  do
  begin
    k_disb_date = null;
    reg_id = null;
    pr_id = null;
    select k_disb_date, kal_new_reg_id, pr_id
    from kontract, knaktlist, product
    where kal_from_kf_id = k_from_kf_id and
          k_disb_date between kal_date and kal_date_end and
          k_id = :k_id and
          knaktlist.is_delete = 0 and
          k_from_pr_id = pr_id
    into k_disb_date, reg_id, pr_id;

    for select ag_id, ag_caption, ag_memo, ag_object_id
        from agreements ag, region_list(:reg_id, 1, -1, 0), prodagreement
        where ag_status = 1 and
              ag_from_reg_id = id_list and
              ag.ag_from_rd_id = :rd_id and
              :k_disb_date between coalesce(ag_date_begin, '01.01.1900') and coalesce(ag_date_end, '31.12.2999') and
              pag_from_ag_id = ag_id and
              pag_from_pr_id = :pr_id
        into :ag_id, :ag_caption, :ag_memo, :ag_object_id
    do
      suspend;
  end
end
