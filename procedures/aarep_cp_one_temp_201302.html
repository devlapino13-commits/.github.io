CREATE PROCEDURE AAREP_CP_ONE_TEMP_201302(
    KONT_ID DECIMAL(18,0),
    DATE_BEGIN DATE,
    DATE_END DATE,
    PERC_PER_DAY DECIMAL(18,7),
    CALC_METHOD INTEGER,
    DISB_AMOUNT DECIMAL(18,4),
    INCLUDE_LAST_DAY_PERC INTEGER,
    K_STOP_PERC DATE)
RETURNS (
    RESULT_BALANCE_BEGIN DECIMAL(18,4),
    RESULT_PERC_BEGIN DECIMAL(18,4),
    RESULT_CALCED DECIMAL(18,4),
    RESULT_PAYED DECIMAL(18,4),
    RESULT_WRITEOFF DECIMAL(18,4),
    RESULT_PERC_END DECIMAL(18,4),
    RESULT_BALANCE_END DECIMAL(18,4),
    RESULT_CALCED_ALL DECIMAL(18,4),
    RESULT_PAYED_ALL DECIMAL(18,4),
    RESULT_WRITEOFF_ALL DECIMAL(18,4),
    RESULT_DISB_AMOUNT DECIMAL(18,4),
    RESULT_DISB_DATE DATE,
    KOL_DAYS INTEGER,
    DD_PRIOR DATE,
    DD DATE,
    ROUND_PERC INTEGER,
    RESULT_PAYED_TAX DECIMAL(18,4),
    RESULT_WRITEOFF_TAX DECIMAL(18,4),
    RESULT_PAYED_ALL_TAX DECIMAL(18,4),
    RESULT_WRITEOFF_ALL_TAX DECIMAL(18,4),
    RESULT_CALCED_TAX DECIMAL(18,4),
    RESULT_CALCED_ALL_TAX DECIMAL(18,4),
    RESULT_PERC_END_TAX DECIMAL(18,4))
AS
declare variable total_balance_begin decimal(18,4);
declare variable date_end_orig date;
declare variable payed_perc decimal(18,4);
declare variable balance_new decimal(18,4);
declare variable ok_t integer;
declare variable sh_dd date;
declare variable dd2 date;
declare variable cm_calc_perc_next_morning integer;
declare variable tax decimal(18,4);
declare variable pr_calc_perc_tax integer;
declare variable pr_is_360 integer;
declare variable total_balance decimal(18,4);
declare variable perc_a decimal(18,4);
declare variable round_fine integer;
declare variable dd_orig date;
declare variable calc_perc_for_deliq integer;
declare variable payed_tax decimal(18,4);
declare variable perc_t decimal(18,4);
declare variable tax_orig decimal(18,7);
begin
  /* Procedure Text */
  round_perc = 0;
  round_fine = 0;
  tax = 0;
  dd_prior = null;
  kol_days = 0;
  result_perc_end = 0;
  total_balance = 0;
  total_balance_begin = 0;
  result_perc_begin = 0;
  result_payed = 0;
  result_writeoff = 0;
  result_balance_begin = 0;
  result_balance_end = 0;
  result_payed_all = 0;
  result_writeoff_all = 0;
  result_payed = 0;
  result_calced_all = 0;
  date_end_orig = date_end;
  result_disb_amount = 0;
  result_disb_date = null;

  result_perc_end_tax = 0;
  result_payed_tax = 0;
  result_writeoff_tax = 0;
  result_payed_all_tax = 0;
  result_writeoff_all_tax = 0;
  result_payed_tax = 0;
  result_calced_all_tax = 0;
  result_payed_tax = 0;
  result_writeoff_tax = 0;
  tax_orig = 0;

  if (date_begin <= date_end) then
  begin
    select min(ok_ol_date), sum(ok_os), cm_calc_perc_next_morning, pr_calc_perc_tax, pr_is_360, pr_round_perc,
           pr_round_fine, pr_calc_perc_for_deliq
    from operkred, company, kontract, product
    where ok_ol_date <= :date_end and
          ok_type = 8 and
          ok_from_k_id = :kont_id and
          ok_ol_is_active = 1 and
          ok_from_k_id = k_id and
          k_from_pr_id = pr_id and
          ok_from_c_id = 0
    group by cm_calc_perc_next_morning, pr_calc_perc_tax, pr_is_360, pr_round_perc, pr_round_fine, pr_calc_perc_for_deliq
    into :result_disb_date, :result_disb_amount, :cm_calc_perc_next_morning, :pr_calc_perc_tax, :pr_is_360, :round_perc,
         :round_fine, :calc_perc_for_deliq;
    disb_amount = result_disb_amount;

    tax = 0;
    tax_orig = :tax;

    if (include_last_day_perc = 1) then
    begin
      date_end = :date_end + :include_last_day_perc;
    end

    date_end = :date_end - :cm_calc_perc_next_morning;
    for select pdd, pok_t, psh_dd, sum(pbalance), sum(ppayed_perc), sum(ppayed_tax), pdd_orig
        from aarep_cp_one_new_temp2_201302(:kont_id, :date_end, :calc_perc_for_deliq)
        group by pdd, pok_t, psh_dd, pdd_orig
        order by pdd, psh_dd, pok_t
        into :dd, :ok_t, :sh_dd, :balance_new, :payed_perc, :payed_tax, :dd_orig
    do
    begin

      if (dd >= '01.01.2009') then
        tax = 0;
      else
        tax = :tax_orig;
      if (balance_new is null) then
        balance_new = 0;
      if (payed_perc is null) then
        payed_perc = 0;
      if (payed_tax is null) then
        payed_tax = 0;
      dd2 = dd_orig;
      if ((dd_orig >= date_begin) and
          (dd_orig <= date_end_orig)) then
      begin
        if (ok_t = 4) then
        begin
          result_payed = result_payed - payed_perc;
          result_payed_tax = result_payed_tax - payed_tax;
        end
        if ((ok_t = 5) or (ok_t = 6)) then
        begin
          result_writeoff = result_writeoff - payed_perc;
          result_writeoff_tax = result_writeoff_tax - payed_tax;
        end
      end
      if (dd_orig <= date_end_orig) then
      begin
        if (ok_t = 4) then
        begin
          result_payed_all = result_payed_all - payed_perc;
          result_payed_all_tax = result_payed_all_tax - payed_tax;
        end
        if ((ok_t = 5) or (ok_t = 6)) then
        begin
          result_writeoff_all = result_writeoff_all - payed_perc;
          result_writeoff_all_tax = result_writeoff_all_tax - payed_tax;
        end
      end
      if ((dd_orig <= date_end_orig) and
          (ok_t <> 0)) then
        result_balance_end = result_balance_end + balance_new;

      if (dd_prior is null) then
      begin
        kol_days = 0;
        total_balance = total_balance + balance_new;
        dd_prior = dd;
      end
      else
      begin
        if (dd_orig <= date_end_orig) then
        begin
          result_perc_end = result_perc_end + payed_perc;
          result_perc_end_tax = result_perc_end_tax + payed_tax;
        end
        if (total_balance <> 0) then
        begin
          if ((k_stop_perc is null) or (k_stop_perc >= dd)) then
            execute procedure fhp_agein360(:dd_prior, :dd, :pr_is_360)
                returning_values :kol_days;
          else
            execute procedure fhp_agein360(:dd_prior, :k_stop_perc, :pr_is_360)
                returning_values :kol_days;

          if (kol_days < 0) then
            kol_days = 0;
          if (kol_days > 0) then
          begin
            if (calc_method > 0) then
              perc_a = total_balance * perc_per_day / 100 * kol_days;
            else
              perc_a = disb_amount * perc_per_day / 100 * kol_days;
            perc_t = 0;
            if (tax <> 0) then
            begin
              perc_t = perc_a * tax / 100;
              perc_a = perc_a - perc_t;
            end
            result_perc_end = result_perc_end + perc_a;
            result_perc_end_tax = result_perc_end_tax + perc_t;
          end
        end
        if (dd2 <= date_end_orig) then
        begin
          total_balance = total_balance + balance_new;
          if (total_balance < 0) then
            total_balance = 0;
        end
        dd_prior = dd;
      end
    end
  end
  /*  if (:CM_CALC_PERC_NEXT_MORNING=0) then */
  begin
    if (result_perc_end_tax < 0) then
      result_perc_end_tax = 0;
    if (result_perc_end < 0) then
      result_perc_end = 0;
  end
--  if ((result_perc_end>0) and (result_balance_end<=0)) then result_perc_end=0;
--  if ((result_perc_end_tax>0) and (result_balance_end<=0)) then result_perc_end_tax=0;
  if (round_perc = 0) then
  begin
    result_perc_end = cast((result_perc_end + result_perc_end_tax) as decimal(18,0));
    result_perc_end_tax = cast(result_perc_end_tax as decimal(18,0));
    result_perc_end = result_perc_end - result_perc_end_tax;
  end
  else
  begin
    result_perc_end = cast((result_perc_end + result_perc_end_tax) as decimal(18,2));
    result_perc_end_tax = cast(result_perc_end_tax as decimal(18,2));
    result_perc_end = result_perc_end - result_perc_end_tax;
  end
  result_calced = result_perc_end + (result_payed + result_writeoff);
  result_calced_tax = result_perc_end_tax + (result_payed_tax + result_writeoff_tax);
  result_calced_all = result_perc_end + result_payed_all + result_writeoff_all;
  result_calced_all_tax = result_perc_end_tax + result_payed_all_tax + result_writeoff_all_tax;
  if (result_calced is null) then
    result_calced = 0;
  if (result_calced_tax is null) then
    result_calced_tax = 0;
  suspend;
end
