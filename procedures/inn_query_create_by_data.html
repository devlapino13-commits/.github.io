CREATE PROCEDURE INN_QUERY_CREATE_BY_DATA(
    C_ID D_ID,
    UN_ID D_ID,
    FIRST_NAME D_STR_CODE,
    LAST_NAME D_STR_CODE,
    MIDDLE_NAME D_STR_CODE,
    BIRTH_DATE D_DATE,
    DOC_NO D_STR_CODE,
    DOC_DATE D_DATE,
    DOC_TYPE D_INTEGER,
    ACL_ID D_ID = null,
    PASS_SERIA D_STR_10 = '',
    PASS_NUMBER D_STR_20 = '')
RETURNS (
    ICQ_ID D_ID,
    ICQ_ID1 D_ID)
AS
declare variable eil_id d_id;
declare variable cfg d_str_20;
declare variable inn_received d_bool;
declare variable is_finished d_bool;
declare variable en_id d_integer;
declare variable inn_ru_source d_integer;
begin
  en_id = f_enum_id('inn_request_source', 'from_kk');
  inn_ru_source = f_get_const('inn_ru_source');
  icq_id = 0;
  icq_id1 = 0;

  select
    inn_received,
    is_finished
  from inn_query_status(:c_id)
  into
    :inn_received,
    :is_finished;

  if (((inn_received = 1) /*or (is_finished = 0)*/)
    and (exists(select * from inn_check_main icm where icm.icm_from_c_id = :c_id))) then
  begin
    suspend;
    exit;
  end

  -- если нет родительской записи, создаю ее
  if (not exists (select * from inn_check_main icm where icm.icm_from_c_id = :c_id)) then
    insert into inn_check_main (icm_from_c_id, icm_is_finished, icm_inn_received)
    values (:c_id, 0, 0);

  cfg = 'inn_gnivc';
  execute procedure hp_extinfo_insert(:cfg, 'clients', :c_id, 0, :un_id);
  eil_id = (
    select first 1
      eil_id
    from extinfolist
      inner join extinfocfg
        on eic_code = :cfg
          and eic_id = eil_from_eic_id
    where eil_from_c_id = :c_id
    order by eil_id desc);
  -- создаю дочернюю запись ИНН ФНС
  insert into inn_check_queries (
    icq_from_c_id, icq_fname, icq_sname, icq_patronimic, icq_datebirth, icq_doc_type,
    icq_doc_no, icq_doc_date, icq_creation_stage, icq_status, icq_inn_received,
    icq_from_eil_id, icq_acl_id, icq_pass_seria, icq_pass_number, icq_info_source)
  values (
    :c_id, :first_name, :last_name, :middle_name, :birth_date, :doc_type,
    :doc_no, :doc_date, :en_id, 0, 0,
    :eil_id, :acl_id, :pass_seria, :pass_number, 0)
  returning
    icq_id
  into
    :icq_id;

  inn_ru_source = f_get_const('inn_ru_source');

  if (inn_ru_source in (1,3)) then
  begin
    cfg = 'inn_smev';
    execute procedure hp_extinfo_insert(:cfg, 'clients', :c_id, 0, :un_id);
    eil_id = (
      select first 1
        eil_id
      from extinfolist
        inner join extinfocfg
          on eic_code = :cfg
            and eic_id = eil_from_eic_id
      where eil_from_c_id = :c_id
      order by eil_id desc);
    -- создаю дочернюю запись ИНН СМЭВ
    insert into inn_check_queries (
      icq_link_id, icq_from_c_id, icq_fname, icq_sname, icq_patronimic, icq_datebirth, icq_doc_type,
      icq_doc_no, icq_doc_date, icq_creation_stage, icq_status, icq_inn_received,
      icq_from_eil_id, icq_acl_id, icq_pass_seria, icq_pass_number, icq_info_source)
    values (
      :icq_id, :c_id, :first_name, :last_name, :middle_name, :birth_date, :doc_type,
      :doc_no, :doc_date, :en_id, 0, 0,
      :eil_id, :acl_id, :pass_seria, :pass_number, 1)
    returning
      icq_id
    into
      :icq_id1;
  end

  suspend;


  post_event 'inn_gnivc';
  if (inn_ru_source in (1,3)) then
    post_event 'inn_smev';
end
