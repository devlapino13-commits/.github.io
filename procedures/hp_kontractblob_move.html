CREATE PROCEDURE HP_KONTRACTBLOB_MOVE
RETURNS (
    KB_FROM_K_ID TYPE OF COLUMN KONTRACTBLOB.KB_FROM_K_ID,
    KB_ID TYPE OF COLUMN KONTRACTBLOB.KB_ID,
    KB_NAME TYPE OF COLUMN KONTRACTBLOB.KB_NAME,
    KB_FILENAME TYPE OF COLUMN KONTRACTBLOB.KB_FILENAME,
    KB_FROM_UN_ID TYPE OF COLUMN KONTRACTBLOB.KB_FROM_UN_ID,
    KB_DATETIME TYPE OF COLUMN KONTRACTBLOB.KB_DATETIME,
    KB_FROM_BLS_ID TYPE OF COLUMN KONTRACTBLOB.KB_FROM_BLS_ID,
    BLS_ID DECIMAL(18,0))
AS
declare variable kb_blob blob sub_type 0 segment size 8;
begin
  for select kb_from_k_id, kb_id, kb_name, kb_blob, kb_filename, kb_from_un_id, kb_datetime, kb_from_bls_id
      from kontractblob
      where kb_blob is not null
      into :kb_from_k_id, :kb_id, :kb_name, :kb_blob, :kb_filename, :kb_from_un_id, :kb_datetime, :kb_from_bls_id
  do
  begin
    execute procedure hp_insert_into_blobstore (:kb_filename, :kb_name, :kb_blob, :kb_from_un_id,
            2, 0)
    returning_values :bls_id;

    update kontractblob
    set kb_blob = null,
        kb_from_bls_id = :bls_id
    where kb_from_k_id = :kb_from_k_id and
          kb_id = :kb_id;
  end
  suspend;
end
