    KAL_NEW_KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    KS_DATE_BEGIN D_DATE,
    REG_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(50),
    KAL_NEW_KF_OFFICER D_STR_100,
    TYPE_FINANCING D_STR_100,
    PR_LOAN_TYPE_STR D_STR_100,
    IS_RESTRUCTED VARCHAR(50),

declare variable kf_id d_id;
declare variable k_id d_id;
declare variable c_id id_type;

begin
  k_id_old = 0;
 kol_pros = 0;
  for select kt.k_kontract_no,
             ks.ks_date_begin,
             kt.k_from_kf_id,

             kt.k_id,
             pr.pr_name,
             (case pr.pr_loan_type when 0 then 'группа'
                                   when 1 then 'индив.'
                                   else 'другое' end),
             iif(pr.pr_code containing ('.ipf.'), 'ипф', 'классика'),
             kncl.kc_amount_approved,
             k_percent,

--             kt.k_period,

             cl.c_pass_no,

             cl.c_pass_date_expire,
             kncl.kc_circle,
             kncl.kc_id,

             kncl.kc_business_type,
             kncl.kc_loan_target,
             iif(kc_circle <= 1, 'новый клиент', 'повторный клиент'),

             kt.k_from_ma_app_id,
             cl.c_family,
             pr.pr_code,

             kt.k_kont_last_date,
             kt.k_disb_date

      from kontract kt
      inner join kontstate ks on ks.ks_from_k_id = kt.k_id
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      inner join knclient kncl on kncl.kc_from_k_id = kt.k_id
      inner join clients cl on cl.c_id = kncl.kc_from_c_id
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(ks.ks_region_id as d_integer)
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            kncl.kc_amount_approved > 0
      order by ks.ks_date_begin, k_id
      into :k_kontract_no,
           :ks_date_begin,
           :kf_id,

           :k_id,
           :pr_name,
           :pr_loan_type_str,

           :type_financing,

--           :k_period,        некорректно выводит срок кредита после реструктуризации

           :c_sex_str,
           :c_pass_no,

           :c_pass_date_expire,
           :kc_circle,
           :kc_id,

           :kc_business_type,
           :kc_loan_target,
           :new_or_rep_cl_str,

           :c_family,
           :pr_code,

           :k_disb_date

  for select kml_from_ml_id
      from kontmat km, agreements ag
      where km.kml_object_kind = 9 and
            km.kml_object_id = :k_id and
            km.kml_from_ml_id is not null and
            km.kml_from_ag_id = ag_id and
            ag.ag_from_rd_id in (:rd_id, :rd_id1)
      into :kml_from_ml_id
    do
    begin
        select ml.ml_name
        from matlist ml
        where ml.ml_id = :kml_from_ml_id
        into kml_tmp;
        kml_name = kml_name || ' ' || kml_tmp;
    end


    if (k_id_old <> k_id) then
    begin
        kal_new_kf_name = null;
        kal_new_kf_officer = null;
        reg_name = null;
        reg_unicode= null;
    
        select kal.kal_new_kf_name,
               cl1.c_name_fio,
               rg.reg_name,
               rg.reg_unicode
        from knaktlist kal
        inner join clients cl1 on cl1.c_id = kal.kal_new_kf_officer
        inner join region rg on rg.reg_unicode = kal.kal_new_reg_id
        where kal.kal_from_kf_id = :kf_id and
              kal.is_delete = 0 and
              :rep_date_end between kal.kal_date and kal.kal_date_end
        into kal_new_kf_name,
             kal_new_kf_officer,
             reg_name,
             reg_unicode;

        select coalesce(max(ok.ok_ol_date - ok_sh_date), 0)
        from operkred ok
        where ok.ok_ol_date > ok_sh_date and
              ok_type < 7 and
              ok_from_k_id = :k_id and
              ok_os + ok_perc < 0 and
              ok_from_c_id = 0
        into :kol_pros;

        is_restructed = null;

        select iif(is_restruct = 1, 'реструктуризирован' , 'не реструкт.')
        from aarep_get_restructuring(:k_id,  cast('31.12.9999' as date) , f_get_db_id()) agr

        ma_percent = null;
        ma_kolich_repay = null;
        ma_kommis_perc = null;

        select cast(ma.ma_date_save as date),
               ma.ma_percent,
               ma.ma_kolich_repay,
               ma.ma_kommis_perc
        from miniapp ma
        where ma.ma_date_end = '31.12.9999' and
              ma.ma_is_active = 1 and
              ma.ma_id = :ma_id
        into :ma_date_app,
             :ma_percent,
             :ma_kolich_repay,
             :ma_kommis_perc;

    end

    k_perc = null;
    k_perc_ipf = null;
    k_perc = iif ( :pr_code not containing '.IPF.' , round(:k_kommis_perc , 2) , null);
    k_perc_ipf = iif ( :pr_code containing '.IPF.' , 150 , null);

    ma_perc = null;
    ma_perc_ipf = null;
    ma_perc = iif( :pr_code not containing '.IPF.' , :ma_kommis_perc , null);
    ma_perc_ipf = iif( :pr_code containing '.IPF.' , 150 , null);

    mac_amount = null;
    mac_business_subtype = null;
    mac_loan_target = null;

    select mac.mac_amount,
           mac.mac_business_subtype,
           ref.rd_name
    from miniappclient mac
    left join refdata ref on ref.rd_id = mac.mac_loan_target
    where mac.mac_from_c_id = :c_id and
          mac.mac_from_ma_id = :ma_id and
          mac.mac_amount > 0
    into :mac_amount,
         :mac_business_subtype,
         :mac_loan_target;

    mac_business = (select bss.bt_name
                    from bustype bss
                    inner join bussubtype bst on bst.bs_from_bt_id = bss.bt_id
                    where bst.bs_id = :mac_business_subtype);

    bt_name = (select bt.bt_name from bustype bt where bt.bt_id = :kc_business_type);
    type_business = (select rd.rd_name from refdata rd where rd.rd_id = :kc_loan_target);

    month_repay = null;

    select (psf.sh_os + psf.sh_perc + psf.sh_perc_tax)
    from print_schedule_full (:k_id, 'RU') psf
    rows 1
    into :month_repay;

    period = datediff(month, :k_disb_date, :K_Kont_last_date);
    if (extract(day from :k_disb_date) < extract(day from :K_Kont_last_date)) then
        period = period + round((extract(day from :K_Kont_last_date) - extract(day from :k_disb_date)) / 30.00, 2);
    else
        period = period - round((extract(day from :k_disb_date) - extract(day from :K_Kont_last_date)) / 30.00, 2);
    period = round(period, 1);


    indexed_adr = null;

    select adl.adl_address_str
    from addrclient adc
    inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
    where adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 1
    into indexed_adr;

    fact_adr = null;

    select adl.adl_address_str
    from addrclient adc
    inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
    where adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 2
    into fact_adr;

    family_status = (select rd_name from refdata rd where rd.rd_id = :c_family);

    crel_other_c_id = null;

    select crt.rel_other_c_id
    from HP_CLIENT_RELATION_TMP(:c_id, 2) crt
    where iif(crt.rel_code = '', null, crt.rel_code) in (1,2)
    rows 1
    into crel_other_c_id;

    fio_suprug = (select cl2.c_name_fio from clients cl2 where cl2.c_id = :crel_other_c_id);

    pass_no_suprug = null;
    date_expire_suprug = null;
    fam_status_suprug = null;

    select c.c_pass_no,
           c.c_pass_date_expire,
           red.rd_name
    from clients c
    inner join refdata red on red.rd_id = c.c_family
    where c.c_id = :crel_other_c_id
    into :pass_no_suprug,
         :date_expire_suprug,
         :fam_status_suprug;

    indexed_adr_suprug = null;

    select adl2.adl_address_str
    from addrclient adc2
    inner join addrlist adl2 on adl2.adl_id = adc2.adc_from_adl_id
    where adc2.adc_from_c_id = :crel_other_c_id and
          adc2.adc_address_type = 1
    into indexed_adr_suprug;

    fact_adr_suprug = null;

    select adl3.adl_address_str
    from addrclient adc3
    inner join addrlist adl3 on adl3.adl_id = adc3.adc_from_adl_id
    where adc3.adc_from_c_id = :crel_other_c_id and
          adc3.adc_address_type = 2
    into fact_adr_suprug;

--    fam_status_suprug = null;
--    select rd.rd_name
--    from clients clint
--    inner join refdata rd on rd.rd_id = clint.c_family
--    where clint.c_id = :crel_other_c_id
--    into :fam_status_suprug;

    report_kib = 'не имеется';
    if (exists (select *
                from hp_kibminiapp_last(:c_id, :ma_id)
                inner join kibhistory on kh_from_kbm_id = kbm_id)) then
      report_kib = 'имеется';
    
    date_of_mon = null;
    fact_use = null;
    memo = null;
    loan_target = null;
    am_descr_of_loan = null;
    svz_memo = null;

    select  ptm.action_date,
            ptm.fact_loan_target,
            ptm.monitoring_memo,
            ptm.loan_target,
            ptm.descr_of_loan,
            ptm.svz_memo
    from print_target_monitoring (:kc_id) ptm
    into date_of_mon,
         fact_use,
         memo,
         loan_target,
         am_descr_of_loan,
         svz_memo;
         
          
    target_app = null;
    comments = null;
    work_address_orientir = null;
    descr_income = null;
    monthly_income = null;
    extra_amount = null;
    monthly_expenses = null;
    fam_expenses = null;
    legal_docs_expenses = null;
    other_business_expenses = null;
    repayment_obligations = null;
    net_profit = null;
    amount_monthly_payments = null;
    kpd = null;
    aak_descr_of_loan = null;
    
if (pr_code not containing '.express_online.') then
 begin 
    select pca.loan_target,
           pca.svz_memo,
           pca.work_address_orientir,
           pca.work_place_name,
           pca.monthly_income,
           pca.extra_amount,
           pca.communal_expenses,
           pca.living_expenses,
           legal_docs_expenses ,
           other_business_expenses,
           pca.repayment_obligations,
           net_profit ,
           amount_monthly_payments,
           pca.kpd,
           pca.descr_of_loan

    from print_client_analiz(:kc_id) pca
    into  target_app,
          comments,
          work_address_orientir,
          descr_income,
          monthly_income,
          extra_amount,
          monthly_expenses,
          fam_expenses,
          legal_docs_expenses,
          other_business_expenses,
          repayment_obligations,
          net_profit ,
          amount_monthly_payments,
          kpd,
          aak_descr_of_loan;
    end
     date_notification = null;
     delinq_reason = null;
     delinq_memo = null;
     repay_expectation = null;
     
     select pdm.analiz_date,
            pdm.delinq_reason,
            pdm.delinq_memo,
            pdm.REPAY_WHEN
     from print_delinq_monitoring(:kc_id) pdm
     into :date_notification,
          :delinq_reason,
          :delinq_memo,
          :repay_expectation;
          
    branch_name = null;
    
    select r2.reg_name
    from region reg
    inner join region r2 on reg.parent = r2.reg_unicode
    where reg.reg_unicode = :reg_unicode
    into branch_name;

    /*select llc.branch_name
    from aarep_llc_clients_reports(:reg_unicode, :rep_date_end) llc rows 1
    into :branch_name;   */

   /* res_data = null;
    select res_data
    from hp_json_kontract_docs(:k_id, 0)
    into res_data;

    asp_signed = null;
    if(res_data containing '"Согласие супруга"') then
        asp_signed = 'Да';

    if (res_data not containing '"Согласие супруга"') then
        asp_signed = 'Нет';  --индив

    if((asp_signed = 'Да') and (c_family in (0,14,16,17) or (fio_suprug is null))) then
        asp_signed = 'Пусто\Нет';  --группа   */
        Asp_signed = 'Нет';

        If(exists( select * from clientdocs cds
        Inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
        Where cds.cds_from_c_id = :crel_other_c_id
        And pk.pk_id = 310)) then
        Asp_signed = 'Да';

    suspend;
    k_id_old = k_id;

  end
end
