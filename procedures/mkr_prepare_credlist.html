CREATE PROCEDURE MKR_PREPARE_CREDLIST(
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_CATEG SMALLINT)
RETURNS (
    CR_TYPE SMALLINT,
    KNC_ID DECIMAL(18,0),
    ID VARCHAR(255),
    NAME VARCHAR(255),
    PLACEOFBIRTH VARCHAR(255),
    PINCODE VARCHAR(255),
    COUNTRYCODE VARCHAR(3),
    BANKRUPTCYSTATUS SMALLINT,
    DATEOFBIRTH DATE,
    K_ID DECIMAL(18,0),
    ACCOUNTNO VARCHAR(255),
    CREDITTYPE VARCHAR(3),
    CREDITLINEAMOUNT DECIMAL(18,4),
    PURPOSEOFCREDIT VARCHAR(3),
    CURRENCYOFCREDIT VARCHAR(3),
    INITIALAMOUNTOFCREDIT DECIMAL(18,4),
    DISOUTAMOUNTOFCREDIT DECIMAL(18,4),
    ANNUALINTERESTRATE DECIMAL(18,7),
    DATEOFGRANT DATE,
    DUETIMEFIRSTCONTRACT DATE,
    DUETIMELASTCONTRACT DATE,
    LASTPAYMENTDATE DATE,
    MONTHLYPAYMENTAMOUNT DECIMAL(18,4),
    DAYSMAINSUMISOVERDUE INTEGER,
    DAYSINTERESTISOVERDUE INTEGER,
    OIAFORREPPERIOD DECIMAL(18,4),
    OIAFORLASTTHREEMONTHS DECIMAL(18,4),
    NUMBEROFPROLONGS INTEGER,
    CREDITCLASSCODE VARCHAR(3),
    CREDITSTATUSCODE VARCHAR(3),
    PROVISION DECIMAL(18,4),
    CREDITPERIODINMONTHS INTEGER,
    KC_NUMBER INTEGER)
AS
declare variable K_DELIQ_DAYS integer;
declare variable KOL_PROLONG integer;
declare variable KNC_BUSINESS_SUBTYPE_ID integer;
declare variable KNC_DELIQ_PERC decimal(18,4);
declare variable KNC_DELIQ_OS decimal(18,4);
declare variable K_AMOUNT_APPROVED decimal(18,4);
declare variable K_FROM_C_ID decimal(18,0);
declare variable OIAFORLASTTHREEMONTHS_BUF decimal(18,4);
declare variable BUFDATE date;
declare variable REP_DATE_BEGIN_TEMP date;
declare variable ACTIVE_ID smallint;
declare variable CLOSE_ID smallint;
declare variable IS_FINISH integer;
declare variable SH_DATE_TMP date;
declare variable SUMM_TMP CURR_TYPE;
declare variable PR_CODE STR_100;
declare variable LASTPAYMENTDATE_OPERKROFF date;
declare variable PR_ID ID_TYPE;
declare variable OKF_DEBT_OS CURR_TYPE;
declare variable OKF_DEBT_PERC CURR_TYPE;
declare variable K_KOLICH_CLIENT_APPROVED D_INTEGER;
declare variable OKS_DELINQ_DATE D_DATE;
declare variable MIN_SH_DATE D_DATE;
declare variable LOAN_CLOSED_DATE date;
declare variable BASE_K_ID D_ID;
declare variable OLD_DISOUTAMOUNTOFCREDIT CURR_TYPE;
declare variable DAYSMAINSUMISOVERDUE_FIX D_INTEGER;
begin
  /*
    CREATED BY IGUSEV 29.07.2012

    PURPOSE: Для МКР. формирование данных для кредитного пакета

    */
  /* ПОЛЯ КОТОРЫЕ НЕ ПЕРЕДАЮ  (ПОЛЯ НЕ ОБЯЗАТЕЛЬНЫЕ ДЛЯ ЗАПОЛНЕНИЯ)

       PlaceOfBirth   -  Место рождения кредитополучателя
       PinCode        -  Пин код кредитополучателя
    */

  /* <PurposeOfCredit>
      001    Потребительские кредиты физическим лицам   <== Выставил этот сектор для всех кредитов
      002    Нефтяной сектор
      003    Ипотечные кредиты физическим лицам
      004    Сектор торговли и обслуживания
      005    Сектор Энергетики, химии и природных ресурсов
      006    Сельскохозяйственный сектор сектор обработки
      007    Строительный и имущественный сектор
      008    Промышленный и производственный сектор
      009    Сектор транспорта и связи
      010    Другие сектора

      <CreditClassCode>
      001    Обеспеченные наличными средствами
      002    Удовлетворительные кредиты
      003    Подконтрольные кредиты
      004    Неудовлетворительные кредиты
      005    Опасные кредиты
      006    Безнадежные кредиты

      <CreditStatusCode>    было до 18.09.2015
      Если реструкторизирован то 004, в простивном случае 006
      001    Полностью оплаченные
      002    С неучтенными процентами
      003    Списанные с баланса
      004    Реструктуризованные
      005    Реструктуризованные и  с неучтенными процентами
      006    Другие статусы

      с  18.09.2015
      007 - iwlek - рабочие кредиты - актиные кредиты с просрочкой до 89 дней
      008 - qeyri iwlek - нерабочие - активные кредиты с просрочкой 90 или более дней
      003 - silinmis - списанные - списанные с баланса компании
      001 - bagl? - закрытые кредиты

      <CreditType>
      001    Кредит
      002    Кредитная линия
      003    Кредитная карта
      004    гарантия

      duetimefirstcontract -  Дата погашения кредита с первоначальным договором
      duetimelastcontract - Дата погашения кредита с окончательным соглашением

    */
  if (report_categ is null) then
    report_categ = 0;

  delete from mkr_temp_dcredits;
  delete from mkr_temp_pcredits;

  if (report_date_begin is null) then
    report_date_begin = '01.01.1899';

  /*Заполняю список просроченный кредтов */

  for select k_id, knc_id, knc_rep_deliq_perc, knc_rep_deliq_os, rep_deliq_days_kolich
      from aarep_kont_clients_list(1000000, null, :report_date_end, 0, '000', 5)
      into :k_id, :k_from_c_id, :knc_deliq_perc, :knc_deliq_os, :k_deliq_days
  do
  begin
    insert into mkr_temp_dcredits (k_id, k_from_c_id, knc_deliq_perc, knc_deliq_os, k_deliq_days)
    values (:k_id, :k_from_c_id, :knc_deliq_perc, :knc_deliq_os, :k_deliq_days);
  end

  /*Заполняю список пролонгированных кредитов */
  for select k_id, kol_prolog
      from aarep_restrukt_dt(1000000, :report_date_end)
      into :k_id, :kol_prolong
  do
  begin
    insert into mkr_temp_pcredits (k_id, kol_prolog)
    values (:k_id, :kol_prolong);
  end

  /*  Значения которые будут выставлены для всех кредитов---------------------------------------->*/
  placeofbirth = 'Azerbaycan';
  credittype = '001';/* тип - кредит*/
  creditlineamount = 0;/*Сумма кредитной линии*/
  provision = 0;/* Сумма созданного резерва */
  /*<---------------------------------------------------------------------------------------------*/

  if (report_categ = 0) then -- общий
  begin
    rep_date_begin_temp = null;
    active_id = 3;
    close_id = 4;
  end
  if (report_categ = 1) then -- выданные
  begin
    rep_date_begin_temp = :report_date_begin;
    active_id = 2;
    close_id = -10;
  end
  if (report_categ = 2) then -- закрытые
  begin
    rep_date_begin_temp = null;
    active_id = -10;
    close_id = 4;
  end

  for select cr_type,
             common_list.knc_id,
             c.c_pass_no,

             iif(upper(pk_country) = 'AZN', 'AZE', upper(pk_country)) as countrycode,
             0 as bankruptcystatus, --Статус для банкротства 1, статус не обанкрочен 0
             knc_birth_date,

             common_list.k_id,
             cu_symbol as currencyofcredit,
             coalesce(k_amount_approved, 0),

             coalesce(knc_amount_approved, 0) as initialamountofcredit,
             coalesce(k_percent, 0), --Сумма кредитного процента
             k_disb_date, --Дата выдачи кредта

             k_kredline_end, --Предположительная дата закрытия кредита
             coalesce(k_deliq_days, 0),
             coalesce(knc_deliq_perc, 0),

             coalesce(kol_prolog, 0), --число пролонгаций по кредиту
             trim(coalesce(c_pincode, '')) as pincode,
             knc_business_subtype_id,

             pr_code,
             pr_id,
             c.c_name_fio
      from (--********************** клиенты активных кредитов ********************************************
          select 1 as cr_type,
                 akcl.knc_id,
                 knc_birth_date,
                 akcl.k_id,
                 k_amount_approved,
                 knc_amount_approved,
                 k_percent,   -- Сумма кредитного процента
                 k_disb_date, -- Дата выдачи кредта
                 k_kredline_end, --Предположительная дата закрытия кредита
                 k_currency, knc_doc_type,
                 tc.knc_deliq_perc,
                 tc.k_deliq_days,
                 knc_business_subtype_id,
                 pr.pr_code,
                 pr_id
          from aarep_kont_clients_list(1000000, :rep_date_begin_temp, :report_date_end, 0, '000', :active_id) akcl
          inner join product pr on pr.pr_id = k_from_pr_id
          left join mkr_temp_dcredits tc on tc.k_id = akcl.k_id and
                tc.k_from_c_id = akcl.knc_id /* привязываю просроченные кредиты*/
          where knc_type_str = 'Физ.лицо'
          union all
          --************************* клиенты закрытых кредитов ***********************************
          select 2 as cr_type,
                 ckcl.knc_id,
                 knc_birth_date,
                 ckcl.k_id,
                 k_amount_approved,
                 knc_amount_approved,
                 k_percent, /*Сумма кредитного процента*/
                 k_disb_date, /* Дата выдачи кредта */
                 k_kredline_end, /*Предположительная дата закрытия кредита*/
                 k_currency,
                 knc_doc_type,
                 0 as knc_deliq_perc,
                 0 as k_deliq_days,
                 knc_business_subtype_id,
                 pr.pr_code,
                 pr_id
          from aarep_kont_clients_list(1000000, :report_date_begin , :report_date_end, 0, '000', :close_id) ckcl
          inner join product pr on pr.pr_id = k_from_pr_id
          where knc_type_str = 'Физ.лицо'
      ) as common_list
      inner join clients c on c_id = knc_id
      inner join currlist cl on cl.cu_id = k_currency
      left join passkind p on p.pk_id = knc_doc_type
      left join mkr_temp_pcredits tp on tp.k_id = common_list.k_id /* привязываю пролонгированные кредиты   */
      into :cr_type,
           :knc_id,
           :id,

           :countrycode,
           :bankruptcystatus,
           :dateofbirth,

           :k_id,
           :currencyofcredit,
           :k_amount_approved,

           :initialamountofcredit,
           :annualinterestrate,
           :dateofgrant,

           :duetimelastcontract,
           :daysmainsumisoverdue,
           :oiaforrepperiod, --сумма процента не полученная до даты отчета

           :numberofprolongs,
           :pincode,
           :knc_business_subtype_id,

           :pr_code,
           :pr_id,
           :name
  do
  begin
      -- Дата погашения кредита с первоначальным договором
      select max(sh.sh_date)
      from schedule sh
      where sh.sh_from_k_id = :k_id
      into :duetimefirstcontract;

      daysinterestisoverdue = daysmainsumisoverdue;  -- т.к. к нас в отчетности ни где нет разделения между числом дней просрочки по ОС и % делаю эти даты одинаковыми

      id = coalesce(id, '');
      id = replace(id, '№', '');

      select cast(iif(kont.k_base_k_id is null,kont.k_id, kont.k_base_k_id) as str_100) || '-' || cast(kncl.kc_number as str_100),
             kc_number,
             kont.k_kolich_client_approved,
             kont.k_base_k_id
      from knclient kncl
      inner join kontract kont on kncl.kc_from_k_id = kont.k_id
      where kncl.kc_from_k_id = :k_id and
            kncl.kc_from_c_id = :knc_id and
            kncl.kc_amount_approved > 0
      rows 1
      into :accountno, :kc_number, :k_kolich_client_approved, :base_k_id;

      lastpaymentdate = null;

      select max(ok_ol_date)
      from operkred ok
      where ok_type = 4 and
            ok_ol_is_active = 1 and
            ok_from_k_id = :k_id and
            ok_ol_date <= :report_date_end and
            ok_kommis = 0 and
            ok_from_c_id = 0 and
            (ok.ok_os + ok.ok_perc) <> 0
      into :lastpaymentdate; -- Последняя дата оплаты

      is_finish = 0;
      monthlypaymentamount = 0;

      for select ok.ok_sh_date,
                 sum(ok.ok_os + ok.ok_sh_os + ok.ok_perc + ok.ok_sh_perc) as summ
          from operkred ok
          where ok.ok_from_k_id = :k_id and
                ok.ok_type in (1, 2, 3) and
                ok.ok_ol_is_active = 1 and
                ok.ok_ol_date <= :report_date_end
          group by 1
          having sum(ok.ok_os + ok.ok_sh_os + ok.ok_perc + ok.ok_sh_perc) <> 0
          into :sh_date_tmp, :summ_tmp
      do
      begin
        if (is_finish = 0) then
        begin
          if (sh_date_tmp >= :report_date_end) then
          begin
            monthlypaymentamount = summ_tmp;
            is_finish = 1;
          end
          else
          begin
            monthlypaymentamount = summ_tmp;
            is_finish = 0;
          end
        end
      end

      if (k_kolich_client_approved > 1) then
        monthlypaymentamount = round(monthlypaymentamount / k_amount_approved * initialamountofcredit, 2);

      oiaforlastthreemonths = 0;
      oiaforlastthreemonths_buf = 0;

      if (daysinterestisoverdue > 0) then
      begin
        bufdate = report_date_end - 90;
        if (daysinterestisoverdue > 90) then
        begin
          for select coalesce(zad_perc, 0)
              from aarep_make_zad_list(:k_id, :report_date_end, 0) amzl
              where sh_date >= :bufdate
              into :oiaforlastthreemonths_buf
          do
          begin
            oiaforlastthreemonths = oiaforlastthreemonths + oiaforlastthreemonths_buf; -- вытаскиваю общую задолжность по %
          end

          if (oiaforlastthreemonths > oiaforrepperiod) then
            oiaforlastthreemonths = oiaforrepperiod;
        end
        else
          oiaforlastthreemonths = oiaforrepperiod;
      end

      if ((oiaforrepperiod > 0) and
          (oiaforlastthreemonths = 0)) then
        oiaforlastthreemonths = oiaforrepperiod;
      if (oiaforrepperiod = 0) then
        oiaforlastthreemonths = 0;

      if (oiaforlastthreemonths < 0) then
        oiaforlastthreemonths = 0;
      if (oiaforrepperiod < 0) then
        oiaforrepperiod = 0;

      /*  <CreditClassCode>
      001    Обеспеченные наличными средствами
      002    Удовлетворительные кредиты
      003    Подконтрольные кредиты
      004    Неудовлетворительные кредиты
      005    Опасные кредиты
      006    Безнадежные кредиты
      */

      if (daysmainsumisoverdue is null) then
        creditclasscode = '002';
      else
      begin
        if (daysmainsumisoverdue <= 30) then
          creditclasscode = '002';
        if ((daysmainsumisoverdue > 30) and
            (daysmainsumisoverdue <= 90)) then
          creditclasscode = '003';
        if ((daysmainsumisoverdue > 90) and
            (daysmainsumisoverdue <= 180)) then
          creditclasscode = '004';
        if ((daysmainsumisoverdue > 180) and
            (daysmainsumisoverdue <= 360)) then
          creditclasscode = '005';
        if (daysmainsumisoverdue > 360) then
          creditclasscode = '006';
      end

      disoutamountofcredit = 0;

      if (cr_type = 1) then -- активные кредиты
      begin
        select oks.oks_balance,
               oks.oks_delinq_date
        from oksummary oks
        where :report_date_end between oks.oks_date_begin and oks.oks_date_end and
              oks.oks_close_state = 0 and
              oks.oks_from_k_id = :k_id
        into :disoutamountofcredit,   -- Остаточная сумма кредита для все группы только ОС
             :oks_delinq_date;  -- дата начала просрочки

        -- Если это разделенный кредит, необходимо провериь какая сумма долга по ОС отправлялась по клиенту до разделения,
        -- т.к. если отпаравить в МКР сумму большую чем отправлялась ранее, МКР запись не примет.
        -- Поэтому если разница между суммой которая отправлялась до разделения и текущей суммой долга по ОС в пределах 0.1 AZN тогда отправляем старую сумму.
        if (base_k_id is not null) then
        begin
            old_disoutamountofcredit = 0;
        
            select round(mcc.mc_disoutamountofcredit,2)
            from mkr_credit mcc
            where mcc.mc_from_k_id = :base_k_id
            and mcc.mc_from_c_id = :knc_id
            order by mcc.mc_from_mp_id desc
            rows 1
            into :old_disoutamountofcredit;
        
            if (abs(disoutamountofcredit - old_disoutamountofcredit) <= 0.1) then
                disoutamountofcredit = old_disoutamountofcredit;
        end

        if (k_kolich_client_approved > 1) then
            disoutamountofcredit = round(disoutamountofcredit / k_amount_approved * initialamountofcredit, 2); -- расчиываю остаточную сумму для человека

        -- Сумма ежемесячной оплаты которую должен заплатить кредитополучатель не может превышать суммы остатка и процентного долга. Это требование МКР
        if (coalesce(:disoutamountofcredit, 0) < coalesce(:monthlypaymentamount, 0)) then
          monthlypaymentamount = disoutamountofcredit;

        if (daysmainsumisoverdue < 90) then
          creditstatuscode = '007'; -- iwlek - рабочие кредиты - актиные кредиты с просрочкой до 89 дней
        else
        if (daysmainsumisoverdue >= 90) then
          creditstatuscode = '008'; -- qeyri iwlek - нерабочие - активные кредиты с просрочкой 90 или более дней

        /*
            закомментировал igusev, в соответствии с запросом в письме от Азера. Тема: Pandemiya ilə bağlı tövsiyənin qüvvədən düşməsi
            
        if ((daysmainsumisoverdue > 0) and
           (oks_delinq_date between cast('01.03.2020' as date) and cast('30.09.2020' as date))) then -- если просрочка за период с 1.03.2020 по 30.09.2020тогда код = 009 "Обязательство, просроченное в связи с пандемией"
            creditstatuscode = '009';
        */
      end
      else -- если закрытый заем
      begin
        okf_debt_os = 0;
        okf_debt_perc = 0;

        -- Igusev 14.11.2019 утвердили между собой алгоритм(Я, Лариса, Эрнест, Алтын, Светлана Морозова) алгоритм:
        -- Если заем закрыт не важно списанием или нет и не важно какая причина списания, но у участника группы есть долг на внесистмке, в рамках этого займа, тогда статус  кредита «Внесистемный учет»
        select sum(okf.oko_os + okf.oko_sh_os),
               sum(okf.oko_perc + okf.oko_sh_perc)
        from operkroff okf
        where okf.oko_from_k_id = :k_id and
              okf.oko_from_c_id = :knc_id and
              okf.oko_ol_is_active = 1 and
              okf.oko_ol_date <= :report_date_end
        into :okf_debt_os,   -- задолжность по ОС
             :okf_debt_perc; -- задолжность по %

        okf_debt_os = coalesce(okf_debt_os, 0);
        okf_debt_perc = coalesce(okf_debt_perc, 0);

        if ((okf_debt_os > 0) or (okf_debt_perc > 0))  then
        begin
          creditclasscode = '006';
          creditstatuscode = '003';           -- silinmis - списанные - списанные с баланса компании
          disoutamountofcredit = okf_debt_os; -- задолжность по ОС
          oiaforrepperiod = okf_debt_perc;    -- задолжность по %

          if ((disoutamountofcredit <> 0) or (oiaforrepperiod <> 0)) then
          begin
            loan_closed_date = null;

            select ks.ks_date_begin
            from kontstate ks
            where ks.ks_state = 2
            and ks.ks_from_k_id = :k_id
            and ks.ks_ol_is_delete = 0
            and :report_date_end  between ks.ks_date_begin and ks.ks_date_end
            into :loan_closed_date;

            /*
            select :report_date_end - min(ok_sh_date)
            from kontract k, operkred ok
            where k_id = :k_id and
                  ok.ok_from_k_id = k.k_id and
                  k.k_fact_closed_ol_id = ok_from_ol_id and
                  ok_type = 5 and
                  ok_sh_date < ok_ol_date and
                  ok_ol_is_active = 1
            into daysmainsumisoverdue;
            */

             -- igusev 01.12.2020 число дней активной просрочки расчитываю так: получаю дату начала просчроки на состояние - дата закрытия займа минус 1 день и вычисляю сколько дней просрочки накопилось с того дня, до отчетной даты.

            if (:loan_closed_date is not null) then
            begin
                daysmainsumisoverdue_fix = null;

                select :report_date_end - (mdd.mdd_last_renewal - mdd.mdd_delinq_days)
                from mkr_delinq_days mdd
                where mdd.mdd_account_no = :accountno
                into :daysmainsumisoverdue_fix;

                if (daysmainsumisoverdue_fix is not null) then
                    daysmainsumisoverdue = daysmainsumisoverdue_fix;
                else
                begin
                    select :report_date_end - oks.oks_delinq_date
                    from oksummary oks
                    where oks.oks_from_k_id = :k_id
                    and oks.oks_close_state = 0
                    and :loan_closed_date - 1 between oks.oks_date_begin and oks.oks_date_end
                    into daysmainsumisoverdue;
                end

                daysinterestisoverdue = daysmainsumisoverdue;
            end
          end

          lastpaymentdate_operkroff = null;

          select okf.oko_ol_date
          from operkroff okf
          where okf.oko_from_k_id = :k_id and
                okf.oko_from_c_id = :knc_id and
                okf.oko_ol_is_active = 1 and
                okf.oko_ol_date <= :report_date_end and
                okf.oko_type = 4 and
                (okf.oko_os + okf.oko_perc) <> 0
          order by okf.oko_ol_date desc
          rows 1
          into :lastpaymentdate_operkroff; -- Последняя дата оплаты

          lastpaymentdate_operkroff = coalesce(lastpaymentdate_operkroff, cast('01.01.1900' as date));

          if (lastpaymentdate_operkroff > lastpaymentdate) then
            lastpaymentdate = lastpaymentdate_operkroff;
        end
        else
          creditstatuscode = '001'; -- bagl? - закрытые кредиты

          -- если кредиты закрылся и на дату закрытия был закрыт платеж, просрочка по которому за период с 01.03.2020 по 30.09.2020 тогда статус  010 - Кредит, просроченный в связи с пандемией и полностью оплаченный
          /*
            закомментировал igusev, в соответствии с запросом в письме от Азера. Тема: Pandemiya ilə bağlı tövsiyənin qüvvədən düşməsi

          min_sh_date = null;

          select min(ok.ok_sh_date)
          from operkred ok
          where ok.ok_from_k_id = :k_id
          and ok.ok_type = 4
          and ok.ok_ol_date = :lastpaymentdate
          and ok.ok_sh_date < :lastpaymentdate
          and ok.ok_ol_is_active = 1
          and ((ok.ok_os < 0) or (ok.ok_perc < 0) or (ok.ok_fine < 0))
          into :min_sh_date;

          if ((min_sh_date is not null) and (min_sh_date between cast('01.03.2020' as date) and cast('30.09.2020' as date))) then
          begin
            creditstatuscode = '010';
            daysmainsumisoverdue = lastpaymentdate - min_sh_date;
            daysinterestisoverdue = daysmainsumisoverdue;
          end
          */
      end

      if ((dateofgrant is not null) and
          (duetimelastcontract is not null)) then
      begin
        creditperiodinmonths = (extract(year from duetimelastcontract) - extract(year from dateofgrant)) * 12 + extract(month from duetimelastcontract) - extract(month from dateofgrant) + 1;
      end
      if (creditperiodinmonths = 0) then
        creditperiodinmonths = 1;

      purposeofcredit = null;

      select purposeofcredit
      from mkr_purposeofcredit_get(:pr_id, :knc_business_subtype_id, :dateofgrant)
      into :purposeofcredit;


      disoutamountofcredit = coalesce(disoutamountofcredit, 0);
      oiaforrepperiod = coalesce(oiaforrepperiod, 0);
      daysinterestisoverdue = coalesce(daysinterestisoverdue, 0);

      if (disoutamountofcredit > initialamountofcredit) then
        disoutamountofcredit = initialamountofcredit;

      if (disoutamountofcredit < 0) then
        disoutamountofcredit = 0;

      suspend;
  end
