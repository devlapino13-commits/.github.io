CREATE PROCEDURE KMP_PASSPORT_ADD_CHECK(
    MCL_ID D_ID,
    INN D_STR_20)
RETURNS (
    HAS_INN D_BOOL,
    INN_SAME D_BOOL,
    PHONE_NUMBER_SAME D_BOOL,
    IS_KG_DB D_BOOL)
AS
declare variable USER_INN D_STR_20;
declare variable CITIZEN_CODE D_STR_255;
declare variable MCL_LOGIN D_STR_CODE;
declare variable C_ID_BY_INN D_ID;
declare variable PHONE_NUMBER D_STR_20;
begin
  is_kg_db = 0;
  if (f_get_db_id() in (39801, 41701)) then
    is_kg_db = 1;

  select cl.c_inn,
         rd.rd_code,
         mcl.mcl_login
  from mc_login mcl
  left join clients cl on cl.c_id = mcl.mcl_from_c_id
  left join refdata rd on rd.rd_id = cl.c_citizen
  where mcl.mcl_id = :mcl_id
  into :user_inn,
       :citizen_code,
       :mcl_login;

  user_inn = coalesce(user_inn, '');
  citizen_code = coalesce(citizen_code, '');

  has_inn = 1;
  if ((user_inn = '') or (citizen_code <> 'Citizen.KG'))  then
    has_inn = 0;

  inn_same = 0;
  if ((inn = user_inn) and (has_inn = 1)) then
    inn_same = 1;

  phone_number_same = 0;
  if (has_inn = 0) then
  begin
    c_id_by_inn = (select cl.c_id from clients cl where cl.c_inn = :inn rows 1);

    if (c_id_by_inn is not null) then
        phone_number = f_cont_info('clients', :c_id_by_inn, 'cell_main', 'ok');

    phone_number = coalesce(phone_number, '');
    if (phone_number = mcl_login) then
        phone_number_same = 1;
  end

  suspend;
end
