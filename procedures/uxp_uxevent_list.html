CREATE PROCEDURE UXP$UXEVENT_LIST(
    UFI_UUID D_UUID,
    UFI_EVENT_KIND D_INTEGER,
    IS_TARGET D_BOOL)
RETURNS (
    EVT_UUID D_UUID,
    EVT_SENDER_UUID D_UUID,
    EVT_EVENT_KIND D_INTEGER,
    EVT_TARGET_UUID D_UUID,
    EVT_HANDLER_KIND D_INTEGER,
    EVT_POSITION D_INTEGER,
    EVT_UFR_UUID D_UUID,
    EVT_EXPRESSION D_STR_8100,
    RES_CAPTION D_STR_100,
    RES_SENDER_UUID_CODE D_STR_CODE,
    RES_TARGET_UUID_CODE D_STR_CODE,
    RES_EVENT_KIND_STR D_STR_CODE,
    RES_HANDLER_KIND_STR D_STR_CODE,
    RES_MEMO D_STR_LONG,
    RES_UUID_CODE D_STR_CODE,
    RES_UFR_UUID_CODE D_STR_CODE,
    RES_LINK_KIND D_SMALLINT)
AS
declare variable sender_caption d_str_100;
declare variable target_caption d_str_100;
declare variable handler_name d_str_code;
declare variable event_code d_str_code;
declare variable event_name d_str_code;
declare variable tmp_kind d_integer;
begin
  for select evt_uuid, evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position,
             evt_expression, evt_ufr_uuid, evt_link_kind
  from uxp$uxevent_list_st(:ufi_uuid, :ufi_event_kind, :is_target)
  into :evt_uuid, :evt_sender_uuid, :evt_event_kind, :evt_target_uuid, :evt_handler_kind, :evt_position,
       :evt_expression, evt_ufr_uuid, res_link_kind
  do
  begin
    res_sender_uuid_code = uuid_to_char(evt_sender_uuid);
    res_target_uuid_code = uuid_to_char(evt_target_uuid);

    res_memo = '%1: %2   ->   %3: %4';
    sender_caption = uxf$code(evt_sender_uuid, 'caption');

    if (uxf$code(evt_target_uuid, 'kind') <> 'uxevent') then
      target_caption = uxf$code(evt_target_uuid, 'caption');
    else
    begin
      select uxf$code(evt_target_uuid, 'caption')
      from ux$event
      where evt_uuid = :evt_target_uuid
      into target_caption;
    end

    if (is_target = 0) then
    begin
      res_caption = target_caption;
      res_uuid_code = uuid_to_char(evt_target_uuid);
    end
    else
    begin
      res_caption = sender_caption;
      res_uuid_code = uuid_to_char(evt_sender_uuid);
    end

    res_ufr_uuid_code = '';
    if (evt_ufr_uuid is not null) then
      res_ufr_uuid_code = uuid_to_char(evt_ufr_uuid);

    res_memo = '';
    res_event_kind_str = '';
    res_handler_kind_str = '';

    select event_code, event_name
    from uxp$uxevent_kinds('On')
    where event_id = :evt_event_kind
    into :event_code, :event_name;

    if (row_count = 0) then
      exception e_message f_format('"On" событие не найдено: %1', evt_event_kind);

    res_event_kind_str = event_code || '       - ' || event_name;

    event_code = '';
    handler_name = '';

    if (res_link_kind = 0) then
    begin
      select event_code, event_name
      from uxp$uxevent_kinds('Do')
      where event_id = :evt_handler_kind
      into :event_code, :handler_name;
  
      if (row_count = 0) then
        exception e_message f_format('"Do" обработчик не найден: %1', evt_handler_kind);
    end
    else
    if (res_link_kind = 1) then
    begin
      select evt_handler_kind
      from ux$event
      where evt_uuid = :evt_target_uuid
      into tmp_kind;

      select event_code, event_name
      from uxp$uxevent_kinds('Do')
      where event_id = :tmp_kind
      into :event_code, :handler_name;
  
      if (row_count = 0) then
        exception e_message f_format('"Do" обработчик не найден: %1', tmp_kind);
  
      target_caption = target_caption || ' (копия)';
    end else
    if (res_link_kind = 2) then
    begin
      select event_code, event_name
      from uxp$uxevent_kinds('On')
      where event_id = :evt_handler_kind
      into :event_code, :handler_name;
  
      if (row_count = 0) then
        exception e_message f_format('"On" обработчик не найден: %1', evt_handler_kind);

      target_caption = target_caption || ' (копия)';
    end

    res_handler_kind_str = event_code || '      - ' || handler_name;

    res_memo = f_format('%1: %2  -> %3: %4', sender_caption, event_name, handler_name, target_caption);

    suspend;
  end
end
