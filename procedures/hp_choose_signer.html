CREATE PROCEDURE HP_CHOOSE_SIGNER(
    UN_ID D_ID)
RETURNS (
    M_ID D_ID,
    M_NAME D_STR_100)
AS
declare variable reg_id d_integer;
begin
  select ps_from_reg_id
  from hr_pstatus, username
  where un_id = :un_id and
        ps_from_c_id = un_from_c_id and
        ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_status = 1 and
        ps_is_working = 1
  into :reg_id;

  if (reg_id is null) then
  begin
    reg_id = f_hq_id();
  end

  for select distinct c_id, c_name_fio
      from region r, kontsignreg, clients, kontsignmen
      where reg_unicode = :reg_id and
            ksr_from_reg_id in (reg_unicode, r.parent) and
            ksm_id = ksr_from_ksm_id and
            ksm_from_c_id = c_id and
            ksm_is_active = 1 and
            ksr_is_active = 1 and
            coalesce(ksr_date_begin, '01.01.1990') <= current_date and
            coalesce(ksr_date_end, '31.12.9999') >= current_date
      union
      -- Добавил Мамрасулову Чынар и Хайирдинова Роза Сайфудиновна (привязаны к КГ) в качестве подписанта
      select c_id, c_name_fio
      from kontsignmen ksm
      inner join clients c on c.c_id = ksm.ksm_from_c_id
      where ksm.ksm_from_c_id in (1090100181509, 1090100001937) and
            ksm.ksm_is_active = 1

      into :m_id, :m_name
  do
    suspend;

end
