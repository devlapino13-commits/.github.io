CREATE PROCEDURE HP_ACTIONLIST_IN_PROCESS1(
    SERVICE_KIND VARCHAR(255),
    UN_ID DECIMAL(18,0),
    WITH_DETAIL INTEGER)
RETURNS (
    REC_ID INTEGER,
    REC_PARENT_ID INTEGER,
    RD_ID INTEGER,
    RD_CODE VARCHAR(255),
    ACL_OBJECT_KIND INTEGER,
    RES_BOX_NAME VARCHAR(100),
    RES_COUNT INTEGER,
    RES_STATUS INTEGER,
    RES_IS_OWN INTEGER,
    ACL_ID DECIMAL(18,0),
    RES_PICTURE_ID INTEGER,
    RES_NODE_MEMO VARCHAR(255),
    RES_IS_BOLD INTEGER,
    RES_CAN_SELECT INTEGER,
    RD_STR_ID VARCHAR(100))
AS
declare variable c_id decimal(18,0);
declare variable id_new_in_waiting integer;
declare variable id_own_in_waiting integer;
declare variable id_new_in_process integer;
declare variable id_own_in_process integer;
declare variable res_new_in_waiting integer;
declare variable res_own_in_waiting integer;
declare variable res_new_in_process integer;
declare variable res_own_in_process integer;
declare variable all_new_in_waiting integer;
declare variable all_own_in_waiting integer;
declare variable all_new_in_process integer;
declare variable all_own_in_process integer;
declare variable rd_name varchar(255);
declare variable acl_object_id decimal(18,0);
declare variable id_rd_code_line integer;
declare variable rec_id_last integer;
declare variable all_high_priority integer;
declare variable id_high_priority integer;
declare variable res_high_priority integer;
declare variable scores d_integer;
declare variable db_id d_integer;
declare variable rd_post_check_id d_integer;
declare variable rd_post_check_code varchar(255);
declare variable all_post_check d_integer;
declare variable id_post_check d_integer;
declare variable res_post_check d_integer;
declare variable is_post_check d_bool;
declare variable str_id_post_check varchar(100);
declare variable rd_id_old integer;
declare variable rd_code_old varchar(255);
declare variable rd_str_id_old varchar(100);
begin
  -- Проверяем наличие новых задач в дереве по пунктам меню
  rd_post_check_code = 'ActionGroup.Check_Loan_Data_PostCheck';
  rec_id = 0;
  scores = 0;
  db_id = f_get_db_id();

  if (db_id in (39801, 41701)) then
  begin
    select rd.rd_orderno
    from refdata rd
    where rd.rd_code = 'PrioritySertif.ScoresLimit'
    into :scores;

    select rd.rd_id, rd.rd_str_id
    from refdata rd
    where rd.rd_code = :rd_post_check_code
    into :rd_post_check_id, :str_id_post_check;
  end

  rd_post_check_id = coalesce(rd_post_check_id, 0);

  for select rec_id, rec_parent_id, rd_id, rd_code, acl_object_kind, res_box_name, res_count, res_status, res_is_own,
             acl_id, res_picture_id, res_node_memo, res_is_bold, res_can_select, rd_str_id
      from hp_reminder_in_tree(:un_id)
      into :rec_id, :rec_parent_id, :rd_id, :rd_code, :acl_object_kind, :res_box_name, :res_count, :res_status,
           :res_is_own, :acl_id, :res_picture_id, :res_node_memo, :res_is_bold, :res_can_select, :rd_str_id
  do
    suspend;

  -- Service_kind - имя службы для которой выдается список задач в очереди
  -- Service_kind будет закрепляться в свойстве RD_Str_ID элемента ActionGroup
  -- Текущие варианты значений Service_kind:
  -- UnderWriter, Digitising
  -- в будущем могут быть Digitising, CallCenter_Calling, LO_Calling, Visit_Monitoring, Visit_Inspection и т.д.
  res_node_memo = '';
  res_is_bold = 0;
  res_picture_id = 6;
  if (coalesce(service_kind, '') = '') then
    exit;

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  res_can_select = 0;
  rec_parent_id = null;
  all_new_in_waiting = 0;
  all_own_in_waiting = 0;
  all_new_in_process = 0;
  all_own_in_process = 0;
  all_high_priority = 0;
  acl_id = null;
  all_post_check = 0;

  for select rd_id, rd_code, rd_name, rd_str_id
      from hp_actionlist_un_access(:service_kind, :un_id)
      into :rd_id, :rd_code, :rd_name, :rd_str_id
  do
  begin
    if (:rd_code = :rd_post_check_code and :db_id in (39801, 41701)) then
      continue;

    rec_id = rec_id + 1;
    id_rd_code_line = rec_id;
    rec_id = rec_id + 1;
    id_new_in_waiting = rec_id;
    rec_id = rec_id + 1;
    id_own_in_waiting = rec_id;
    rec_id = rec_id + 1;
    id_new_in_process = rec_id;
    rec_id = rec_id + 1;
    id_own_in_process = rec_id;
    rec_id = rec_id + 1;
    id_high_priority = rec_id;
    rec_id = rec_id + 1;
    id_post_check = rec_id;

    all_new_in_waiting = 0;
    all_own_in_waiting = 0;
    all_new_in_process = 0;
    all_own_in_process = 0;
    all_high_priority = 0;
    all_post_check = 0;

    res_new_in_waiting = 0;
    res_own_in_waiting = 0;
    res_new_in_process = 0;
    res_own_in_process = 0;
    res_high_priority = 0;
    res_post_check = 0;

    if (rd_str_id = 'UnderWriter') then
    begin
      for select aco_object_kind,
          sum(res_new_in_waiting),
          sum(res_high_priority),
          sum(res_own_in_waiting),
          sum(res_new_in_process),
          sum(res_own_in_process),
          sum(res_post_check)
          from
          (
            select aco_object_kind,
                   sum(iif(aco_status = 0 and aco_respons_c_id is null and f_point(k_clients_cashier_id, current_date) < :scores, 1, 0)) as res_new_in_waiting,
                   sum(iif(aco_status = 0 and aco_respons_c_id is null and f_point(k_clients_cashier_id, current_date) >= :scores, 1, 0)) as res_high_priority,
                   sum(iif(aco_status = 0 and aco_respons_c_id is not null, 1, 0)) as res_own_in_waiting,
                   sum(iif(aco_status = 1 and aco_respons_c_id is null, 1, 0)) as res_new_in_process,
                   sum(iif(aco_status = 1 and aco_respons_c_id is not null, 1, 0)) as res_own_in_process,
                   0 as res_post_check
            from actionorder, user_reg_list(:un_id), actionlist, kontract
            where (aco_respons_c_id = :c_id or aco_respons_c_id is null) and
                  aco_from_rd_id = :rd_id and
                  aco_from_reg_id = r_list and
                  acl_id = aco_id and
                  acl_dont_show_in_order = 0 and
                  k_id = aco_object_id and
                  aco_from_rd_id <> :rd_post_check_id
            group by aco_object_kind
            union all
            select aco_object_kind,
                   0 as res_new_in_waiting,
                   0 as res_high_priority,
                   0 as res_own_in_waiting,
                   0 as res_new_in_process,
                   sum(iif(aco_status = 1 and aco_respons_c_id is not null, 1, 0)) as res_own_in_process,
                   sum(iif(aco_status = 0 and aco_respons_c_id is null,1,0)) as res_post_check
            from actionorder, user_reg_list(:un_id), actionlist, kontract
            where (aco_respons_c_id = :c_id or aco_respons_c_id is null) and
                  aco_from_reg_id = r_list and
                  acl_id = aco_id and
                  acl_dont_show_in_order = 0 and
                  k_id = aco_object_id and
                  aco_from_rd_id = :rd_post_check_id and
                  :db_id in (39801, 41701) and
                  :rd_code = 'ActionGroup.Check_Loan_Data'
            group by aco_object_kind
          )
          group by aco_object_kind
          order by 3 desc
          into acl_object_kind, :res_new_in_waiting, -- Новые в ожидании
               :res_high_priority, --
               :res_own_in_waiting, -- Мои в очереди
               :res_new_in_process, -- В обработке у других
               :res_own_in_process, -- Мои в обработке
               :res_post_check --Псст-проверка
      do
      begin

        acl_id = null;

        if (res_new_in_waiting is null) then
          res_new_in_waiting = 0;
        if (res_high_priority is null) then
          res_high_priority = 0;
        if (res_own_in_waiting is null) then
          res_own_in_waiting = 0;
        if (res_new_in_process is null) then
          res_new_in_process = 0;
        if (res_own_in_process is null) then
          res_own_in_process = 0;
        if (res_post_check is null) then
          res_post_check = 0;

        if (db_id not in (39801, 41701)) then
          res_new_in_waiting = res_new_in_waiting + res_high_priority;

        if (res_own_in_process <> 0) then
        begin
          -- Мои в обработке
          rec_parent_id = id_own_in_process;
          res_status = 1;
          res_is_own = 1;
          res_picture_id = 6;
          res_is_bold = 0;
          res_count = res_own_in_process;
          all_own_in_process = all_own_in_process + res_own_in_process;
          res_can_select = 1;

          if (with_detail = 1) then
          begin
            res_node_memo = 'Нажмите два раза чтоб открыть детали';

            for select aco_id, aco_object_id, is_post_check
                from
                (
                  select aco_id, aco_object_id, 0 as is_post_check
                  from actionorder
                  where aco_respons_c_id = :c_id and
                        aco_status = :res_status and
                        aco_object_kind = :acl_object_kind and
                        aco_from_rd_id = :rd_id
                  union all
                  select aco_id, aco_object_id, 1
                  from actionorder
                  where aco_respons_c_id = :c_id and
                        aco_status = :res_status and
                        aco_object_kind = :acl_object_kind and
                        aco_from_rd_id = :rd_post_check_id and
                        :db_id in (39801, 41701) and
                        :rd_code = 'ActionGroup.Check_Loan_Data'
                )
                into acl_id, acl_object_id, is_post_check
            do
            begin
              rec_id = rec_id + 1;
              res_box_name = '';
              select res_object_name
              from hp_actionlist_object_name(:acl_object_kind, :acl_object_id)
              into :res_box_name;
              rd_id_old = :rd_id;
              rd_code_old = :rd_code;
              rd_str_id_old = :rd_str_id;

              if (is_post_check = 1) then
              begin
                rd_id = :rd_post_check_id;
                rd_code = :rd_post_check_code;
                rd_str_id = :str_id_post_check;
              end
              suspend;
              rd_id = :rd_id_old;
              rd_code = :rd_code_old;
              rd_str_id = :rd_str_id_old;
            end
            acl_id = null;
          end
        end
        if (res_own_in_waiting <> 0) then
        begin
          -- Мои в очереди
          rec_parent_id = id_own_in_waiting;
          res_status = 0;
          res_is_own = 1;
          res_picture_id = 6;
          res_is_bold = 0;
          res_count = res_own_in_waiting;
          all_own_in_waiting = all_own_in_waiting + res_own_in_waiting;
          res_node_memo = 'Раскройте ветку и выберите объект для начала обработки';
          res_can_select = 1;

          if (with_detail = 1) then
          begin
            res_node_memo = 'Нажмите два раза чтоб открыть детали';
            for select aco_id, aco_object_id
                from actionorder
                where aco_respons_c_id = :c_id and
                      aco_status = :res_status and
                      aco_object_kind = :acl_object_kind and
                      aco_from_rd_id = :rd_id
                into acl_id, acl_object_id
            do
            begin
              rec_id = rec_id + 1;
              res_box_name = '';
              select res_object_name
              from hp_actionlist_object_name(:acl_object_kind, :acl_object_id)
              into :res_box_name;
              suspend;
            end
            acl_id = null;
          end

        end
        if (res_new_in_process <> 0) then
        begin
          -- В обработке у других
          res_count = res_new_in_process;
          all_new_in_process = all_new_in_process + res_new_in_process;
        end

        if (res_new_in_waiting <> 0) then
        begin
          -- Новые в ожидании
          res_count = res_new_in_waiting;
          all_new_in_waiting = all_new_in_waiting + res_new_in_waiting;
        end

        if (res_high_priority <> 0) then
        begin
          -- Приоритетные в ожидании
          res_count = res_high_priority;
          all_high_priority = all_high_priority + res_high_priority;
        end

        if (res_post_check <> 0) then
        begin
          -- Пост проверка
          res_count = res_post_check;
          all_post_check = all_post_check + res_post_check;
        end
      end
    end

    else
    begin
      for select aco_object_kind, sum(iif(aco_status = 0 and aco_respons_c_id is null, 1, 0)),
                 sum(iif(aco_status = 0 and aco_respons_c_id is not null, 1, 0)),
                 sum(iif(aco_status = 1 and aco_respons_c_id is null, 1, 0)),
                 sum(iif(aco_status = 1 and aco_respons_c_id is not null, 1, 0))
          from actionorder, user_reg_list(:un_id), actionlist
          where (aco_respons_c_id = :c_id or aco_respons_c_id is null) and
                aco_from_rd_id = :rd_id and
                aco_from_reg_id = r_list and
                acl_id = aco_id and
                acl_dont_show_in_order = 0
          group by aco_object_kind
          into acl_object_kind, :res_new_in_waiting, -- Новые в ожидании
               :res_own_in_waiting, -- Мои в очереди
               :res_new_in_process, -- В обработке у других
               :res_own_in_process -- Мои в обработке
      do
      begin

        acl_id = null;

        if (res_new_in_waiting is null) then
          res_new_in_waiting = 0;
        if (res_own_in_waiting is null) then
          res_own_in_waiting = 0;
        if (res_new_in_process is null) then
          res_new_in_process = 0;
        if (res_own_in_process is null) then
          res_own_in_process = 0;

        if (res_own_in_process <> 0) then
        begin
          -- Мои в обработке
          rec_parent_id = id_own_in_process;
          res_status = 1;
          res_is_own = 1;
          res_picture_id = 6;
          res_is_bold = 0;
          res_count = res_own_in_process;
          all_own_in_process = all_own_in_process + res_own_in_process;
          res_can_select = 1;

          if (with_detail = 1) then
          begin
            res_node_memo = 'Нажмите два раза чтоб открыть детали';

            for select aco_id, aco_object_id
                from actionorder
                where aco_respons_c_id = :c_id and
                      aco_status = :res_status and
                      aco_object_kind = :acl_object_kind and
                      aco_from_rd_id = :rd_id
                into acl_id, acl_object_id
            do
            begin
              rec_id = rec_id + 1;
              res_box_name = '';
              select res_object_name
              from hp_actionlist_object_name(:acl_object_kind, :acl_object_id)
              into :res_box_name;
              suspend;
            end
            acl_id = null;
          end
        end
        if (res_own_in_waiting <> 0) then
        begin
          -- Мои в очереди
          rec_parent_id = id_own_in_waiting;
          res_status = 0;
          res_is_own = 1;
          res_picture_id = 6;
          res_is_bold = 0;
          res_count = res_own_in_waiting;
          all_own_in_waiting = all_own_in_waiting + res_own_in_waiting;
          res_node_memo = 'Раскройте ветку и выберите объект для начала обработки';
          res_can_select = 1;

          if (with_detail = 1) then
          begin
            res_node_memo = 'Нажмите два раза чтоб открыть детали';
            for select aco_id, aco_object_id
                from actionorder
                where aco_respons_c_id = :c_id and
                      aco_status = :res_status and
                      aco_object_kind = :acl_object_kind and
                      aco_from_rd_id = :rd_id
                into acl_id, acl_object_id
            do
            begin
              rec_id = rec_id + 1;
              res_box_name = '';
              select res_object_name
              from hp_actionlist_object_name(:acl_object_kind, :acl_object_id)
              into :res_box_name;
              suspend;
            end
            acl_id = null;
          end

        end
        if (res_new_in_process <> 0) then
        begin
          -- В обработке у других
          res_count = res_new_in_process;
          all_new_in_process = all_new_in_process + res_new_in_process;
        end

        if (res_new_in_waiting <> 0) then
        begin
          -- Новые в ожидании
          res_count = res_new_in_waiting;
          all_new_in_waiting = all_new_in_waiting + res_new_in_waiting;
        end
      end
    end

    acl_id = null;

    res_can_select = 0;
    res_is_bold = 0;
    rec_id_last = rec_id;

    rec_id = id_rd_code_line;
    rec_parent_id = null;
    res_picture_id = 0;
    res_status = -1;
    res_is_own = -1;
    res_node_memo = '';
    res_count = all_new_in_waiting + all_own_in_waiting + all_own_in_process + all_high_priority + all_post_check; /* +all_new_in_process */
    res_box_name = rd_name;
    if (res_count <> 0) then
    begin
      res_box_name = res_box_name || ' (' || res_count || ')';
      res_is_bold = 1;
    end
    res_node_memo = 'Здесь будут ваши объекты в обработке';
    suspend;

    rec_parent_id = id_rd_code_line;

    rec_id = id_own_in_process;
    res_picture_id = 1;
    res_status = 1;
    res_is_own = 1;
    res_box_name = 'Мои в обработке';
    res_count = all_own_in_process;
    res_is_bold = 0;
    res_node_memo = 'Здесь будут ваши объекты в обработке';
    res_can_select = 0;
    if (res_count <> 0) then
    begin
      res_can_select = 1;
      res_node_memo = 'Объекты в режиме ожидания вашего решения';
      res_is_bold = 1;
      res_box_name = res_box_name || ' (' || res_count || ')';
    end
    suspend;

    rec_id = id_own_in_waiting;
    res_picture_id = 2;
    res_status = 0;
    res_is_own = 1;
    res_box_name = 'Мои в очереди';
    res_count = all_own_in_waiting;
    res_is_bold = 0;
    res_node_memo = 'Здесь будут объекты в очереди, закрепленные за вами';
    res_can_select = 0;
    if (res_count <> 0) then
    begin
      res_can_select = 1;
      res_node_memo = 'Объекты следующие в очереди';
      res_is_bold = 1;
      res_box_name = res_box_name || ' (' || res_count || ')';
    end
    suspend;

    rec_id = id_new_in_process;
    res_picture_id = 3;
    res_status = 1;
    res_is_own = 0;
    res_box_name = 'В обработке у других';
    res_count = all_new_in_process;
    res_is_bold = 0;
    res_node_memo = 'Здесь находятся объекты в очереди, закрепленные за другими сотрудниками';
    res_can_select = 0;
    if (res_count <> 0) then
    begin
      res_node_memo = 'Объекты не доступны для вас';
      res_is_bold = 1;
      res_box_name = res_box_name || ' (' || res_count || ')';
    end
    suspend;

    rec_id = id_new_in_waiting;
    res_picture_id = 4;
    res_status = 0;
    res_is_own = 0;
    res_box_name = 'Новые в ожидании';
    res_count = all_new_in_waiting;
    res_is_bold = 0;
    res_node_memo = 'Здесь будут находиться объекты в общей очереди';
    res_can_select = 0;
    if (res_count <> 0) then
    begin
      res_can_select = 1;
      res_node_memo = 'Нажмите два раза, чтоб забрать очередной объект на обработку';
      res_is_bold = 1;
      res_box_name = res_box_name || ' (' || res_count || ')';
    end
    suspend;

    rec_id = id_high_priority;
    res_picture_id = 4;
    res_status = 0;
    res_is_own = 0;
    res_box_name = 'Приоритетные в ожидании';
    res_count = all_high_priority;
    res_is_bold = 0;
    res_node_memo = 'Здесь будут находиться объекты с высоким приоритетом в общей очереди';
    res_can_select = 0;
    if (res_count <> 0) then
    begin
      res_can_select = 1;
      res_node_memo = 'Нажмите два раза, чтоб забрать очередной объект на обработку';
      res_is_bold = 1;
      res_box_name = res_box_name || ' (' || res_count || ')';
    end
    if (rd_id = 1287 and
        db_id in (39801, 41701)) then
