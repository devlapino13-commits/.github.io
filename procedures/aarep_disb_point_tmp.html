CREATE PROCEDURE AAREP_DISB_POINT_TMP(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_ID D_INTEGER,
    PR_ID D_INTEGER,
    K_DISB_DATE D_DATE,
    RES_POINT D_INTEGER,
    CLIENT_STATE D_INTEGER)
AS
declare variable c_id d_id;
declare variable last_close_date d_date;
begin
  for select kc_from_c_id, ks_date_begin, k_from_pr_id, ks_region_id
  from kontstate, kontract, knclient
  where ks_date_begin between :rep_date_begin and :rep_date_end and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_from_k_id = k_id and
        k_id = kc_from_k_id and
        kc_amount_approved > 0
  into c_id, k_disb_date, pr_id, reg_id
  do
  begin
    res_point = f_point(c_id, k_disb_date);

    client_state = 0; -- Клиент новый
    last_close_date = null;
    select ks_date_end
    from kontstate, knclient
    where ks_state = 1 and
          ks_ol_is_delete = 0 and
          ks_date_begin < :k_disb_date and
          ks_from_k_id = kc_from_k_id and
          kc_from_c_id = :c_id and
          kc_amount_approved > 0
    order by ks_date_end desc
    rows 1
    into last_close_date;
    if (last_close_date is not null) then
    begin
      if (last_close_date < '31.12.9999') then
        last_close_date = last_close_date + 1;

      if (last_close_date > k_disb_date)
      then client_state = 1; -- Клиент активный на момент выдачи
      else
        if (last_close_date >= dateadd(day, -30, k_disb_date))
        then client_state = 2; -- Клиент закрыл за последние 30 дней
        else client_state = 3; -- Клиент закрыт ранее 30 дней
    end

    suspend;
  end
end
