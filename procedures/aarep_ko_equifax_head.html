CREATE PROCEDURE AAREP_KO_EQUIFAX_HEAD(
    BLS_ID D_ID)
RETURNS (
    FIO D_STR_255,
    BIRTH_DATA D_STR_255,
    PASS_DATA D_STR_LONG,
    ADDRESS_REG D_STR_LONG,
    ADDRESS_FACT D_STR_LONG,
    KO_REP_DATE_STR D_STR_CODE,
    KO_CREATE_DATETIME_STR D_STR_CODE,
    KO_RESPONSE_STR D_STR_255)
AS
declare variable jso d_integer;
declare variable jso_tmp d_integer;
declare variable okato d_str_code;

declare function getcatval(kind d_str_code, code d_str_code)
  returns d_str_255
  as
  begin
    return coalesce((select bct_name from bki_catalog where bct_kind = :kind and bct_code = :code), 'Неопределен');
  end

declare function my_get_last_el(jso_in d_integer)
returns d_integer
as
  declare variable jso d_integer;
  declare variable res d_integer;
  declare variable res_tmp d_integer;
  declare variable max_date d_date;
  declare variable cur_date d_date;
begin
  res = null;
  if (jso_in is not null) then
  begin
    max_date = '01.01.1900';
  
    for select jso
    from hp_bki_json_get_list(:jso_in, 1)
    into jso
    do
    begin
      res_tmp = jso;
  
      cur_date = json.obj_get_val(jso, 'extraData.7', current_date);
  
      if (max_date < cur_date) then
      begin
        max_date = cur_date;
        res = res_tmp;
      end
    end
  end

  if (res is null) then
    res = json.obj_create();

  return res;
end

begin
  jso = f_bki_bls2json(bls_id);

  -- ФИО текущие
  fio = '';
  jso_tmp = json.obj_get_obj(jso, 'response.titlePart._private.name');
  if (jso_tmp is not null) then
    fio = json.obj_get_val(jso_tmp, 'last', '') || ' ' ||
          json.obj_get_val(jso_tmp, 'first', '') || ' ' ||
          json.obj_get_val(jso_tmp, 'middle', '');

  -- Данные о рождении
  birth_data = '';
  jso_tmp = json.obj_get_arr(jso, 'response.titlePart._private.birth');

  select json.obj_get_val(jso, 'date', '') || ' ' || json.obj_get_val(jso, 'place', '')
  from hp_bki_json_get_list(:jso_tmp, 1)
  into birth_data;

  -- ДУЛ текущий
  pass_data = '';
  jso_tmp = json.obj_get_obj(jso, 'response.titlePart._private.doc');
  if (jso_tmp is not null) then
    pass_data = getcatval('bki_doc_type', json.obj_get_val(jso_tmp, 'typeValue')) || ' <b>' ||
                json.obj_get_val(jso_tmp, 'serial', '') || ' ' ||
                json.obj_get_val(jso_tmp, 'number', '') || '</b> выдан <b>' ||
                json.obj_get_val(jso_tmp, 'date', '') || '</b> ' ||
                json.obj_get_val(jso_tmp, 'who', '');

  -- Адрес регистрации
  okato = null;
  address_reg = '';
  jso_tmp = my_get_last_el(json.obj_get_arr(jso, 'response.basePart.addrReg'));

  if (jso_tmp is not null) then
  begin
    okato = json.obj_get_val(jso_tmp, 'okato');

    address_reg = coalesce(json.obj_get_val(jso_tmp, 'otherStatement')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'street')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'house')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'domain')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'block')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'build')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'apartment'), '');
  end

  if (okato is not null) then
  begin
    with recursive t1 (lvl, parent_id, name, code, okato) as(
      select 0, parent, pl_name || ' ' || plt_short_name, pl_code, pl_add_code
      from places, placetype
      where plt_id = pl_status
        and pl_add_code = :okato
    union all
      select lvl + 1, parent, pl_name || ' ' || plt_short_name || ', ' || name, pl_code, pl_add_code
      from t1, places, placetype
      where pl_id = parent_id
        and parent is not null
        and plt_id = pl_status
    )
    select name || ', ' || :address_reg
    from t1
    order by lvl desc
    rows 1
    into address_reg;

    address_reg = 'OKATO: ' || okato || ', ' || :address_reg;
  end

  -- Адрес проживания
  okato = null;
  address_fact = '';
  jso_tmp = my_get_last_el(json.obj_get_arr(jso, 'response.basePart.addrFact'));

  if (jso_tmp is not null) then
  begin
    okato = json.obj_get_val(jso_tmp, 'okato');

    address_fact = coalesce(json.obj_get_val(jso_tmp, 'otherStatement')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'street')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'house')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'domain')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'block')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'build')||', ', '')||
      coalesce(json.obj_get_val(jso_tmp, 'apartment'), '');
  end

  if (okato is not null) then
  begin
    with recursive t1 (lvl, parent_id, name, code, okato) as(
      select 0, parent, pl_name || ' ' || plt_short_name, pl_code, pl_add_code
      from places, placetype
      where plt_id = pl_status
        and pl_add_code = :okato
    union all
      select lvl + 1, parent, pl_name || ' ' || plt_short_name || ', ' || name, pl_code, pl_add_code
      from t1, places, placetype
      where pl_id = parent_id
        and parent is not null
        and plt_id = pl_status
    )
    select name || ', ' || :address_fact
    from t1
    order by lvl desc
    rows 1
    into address_fact;

    address_fact = 'OKATO: ' || okato || ', ' || :address_fact;
  end

  ko_rep_date_str = json.obj_get_val(jso, 'response.dateofreport', '');
  ko_create_datetime_str = json.obj_get_val(jso, 'datetime', '');
  ko_response_str = json.obj_get_val(jso, 'responsestring', '');

  suspend;
end
