CREATE PROCEDURE HP_APPROVED_EXA_PROCESS(
    EXA_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable EXA_WAIT_STATUS D_INTEGER;
begin
  exa_wait_status = null;
  select exa_wait_status from expressapp, expresscfg
  where exa_id = :exa_id
  and exa_from_exc_id = exc_id
  and exc_action_code = 'loan.next'
  into exa_wait_status;

  if (exa_wait_status is not null) then
  begin
    if (exa_wait_status = 1) then
    begin
      update expressapp set exa_wait_status = 0, exa_from_un_id = :un_id
      where exa_id = :exa_id;
      res_text = 'Успешно подтвержден!';
    end else
    if (exa_wait_status = 0) then
    begin
      res_text = 'Уже был подствержден, подождите 5 мин';
    end
  end else
  begin
    res_text = 'Этап пройден!';
  end
  suspend;
end
