CREATE PROCEDURE MC_LOAN_LIST(
    MCL_ID D_ID)
RETURNS (
    K_ID D_ID,
    AMOUNT_APPROVED D_STR_100,
    TOTAL_DEBT D_STR_100,
    DEBT_OS D_STR_100,
    DEBT_PERC D_STR_100,
    DEBT_FINE D_STR_100,
    DEBT_SH_PERC D_STR_100,
    DEBT_OTHER D_STR_100,
    RECOMENDED_AMOUNT D_STR_100,
    PAYMENT_TXT D_STR_255,
    DAYS_LEFT D_STR_100,
    DELINQ_STATUS D_SMALLINT,
    K_KONTRACT_NO D_STR_100,
    KF_NAME D_STR_100,
    K_KOLICH_CLIENT_APPROVED D_INTEGER,
    DAYS_TO_PAYMENT D_INTEGER,
    DAYS_BETWEEN_PAYMENTS D_INTEGER,
    DEBT_KOMMIS_AMOUNT D_STR_100,
    K_KONTRACT_NO_STR D_STR_CODE,
    RECOMENDED_AMOUNT_NUM DECIMAL(18,2),
    RECOMENDED_AMOUNT_STR_FULL D_STR_100,
    FIO_SHORT D_STR_100,
    PR_NAME D_STR_100,
    LOAN_MESSAGE D_STR_2048,
    CURRENT_AMOUNT D_STR_100,
    CURRENT_OS D_STR_100,
    CURRENT_PERC D_STR_100,
    CURRENT_FINE D_STR_100,
    CURRENT_OTHER D_STR_100,
    SHOW_RESTRUCT_BUTTON D_BOOL)
AS
declare variable C_ID D_ID;
declare variable TEXT D_STR_20;
declare variable RECOM_PAY_SUM D_CURRENCY;
declare variable K_TOTAL_DEBT D_CURRENCY;
declare variable K_DEBT_OS D_CURRENCY;
declare variable K_DEBT_PERC D_CURRENCY;
declare variable K_DEBT_FINE D_CURRENCY;
declare variable K_DEBT_OTHER D_CURRENCY;
declare variable K_AMOUNT D_CURRENCY;
declare variable REP_DATE D_DATE;
declare variable SH_DATE D_DATE;
declare variable DAYS D_SMALLINT;
declare variable OKS_DELINQ_DATE D_DATE;
declare variable DB_ID D_INTEGER;
declare variable LANG_CODE D_STR_LANG;
declare variable ZAD_PERC D_CURRENCY;
declare variable SH_PERC_TAX D_CURRENCY;
declare variable RES_KOMMIS_AMOUNT D_CURRENCY;
declare variable RES_KOMMIS_AMOUNT_TAX D_CURRENCY;
declare variable SH_CALC_PERC_TAX D_CURRENCY;
declare variable PR_CODE D_STR_100;
declare variable K_DISB_DATE D_DATE;
declare variable STR_MONEY D_STR_20;
declare variable MCL_PROJECT D_STR_20;
declare variable K_TEMP_OS D_CURRENCY;
declare variable K_TEMP_PERC D_CURRENCY;
declare variable K_CURR_OS D_CURRENCY;
declare variable K_CURR_PERC D_CURRENCY;
declare variable K_CURR_FINE D_CURRENCY;
declare variable K_CURR_OTHER D_CURRENCY;
declare variable KFR_ID D_INTEGER;
declare variable KRD_IS_DONE D_SMALLINT;
declare variable KRD_FROM_EDL_ID D_ID;
declare variable KRD_ACTION_DATE D_DATE;
declare variable need_to_calc_debt d_bool;
begin
  db_id = f_get_db_id();
  lang_code = f_get_lang_code(null, :mcl_id);
  debt_sh_perc = '';
  debt_kommis_amount = '';

  -- Определяет, нужно ли вычислять оставшийся долг по кредиту. Может быть ситуация, где всё нужно вывести по нулям
  -- тогда делать тяжелые вычисления, а затем всё обнулять нет никакого смысла, проще изначально не считать
  need_to_calc_debt = 1;

  /*
    delinq_status:
    0 - погашение в срок
    1 - скоро просрочка
    2 - заем в просрочке
  */

  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  select
    mcl.mcl_from_c_id,
    f_enum('mcl_project', mcl.mcl_project, 'code', 0)
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into
    :c_id,
    :mcl_project;

  if (c_id is null) then
    exit;

  select short_fio
  from clients, hp_fio_clear(c_name_fio)
  where c_id = :c_id
  into fio_short;

  days_between_payments = 30; -- считаем данную величину константой
  rep_date = current_date;

  str_money = case db_id
    when 41701 then ' сом.'
    when 39801 then ' тенге'
    when 81001 then ' руб.'
    when 94401 then ' manat.'
    else ''
  end;

  for select ks_from_k_id
      from knclient, kontstate
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            kc_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date between ks_date_begin and ks_date_end
      union
      select oko_from_k_id
      from operkroff, knclient
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            kc_from_k_id = oko_from_k_id and
            oko_ol_is_active = 1
      group by oko_from_k_id
      having sum(oko_os + oko_sh_os) > 0 or sum(oko_perc + oko_sh_perc) > 0
      into k_id
  do
  begin
    select pr.pr_code
    from kontract kont
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    where kont.k_id = :k_id
    into :pr_code;

      -- RFCRU-3766
    if (mcl_project = 'kmp' and pr_code <> '' and pr_code containing '.oivo_pack.') then
      continue;

    if (mcl_project = 'oivo' and pr_code <> '' and pr_code not containing '.oivo_pack.') then
      continue;

    show_restruct_button = 0;
    days_to_payment = 0;

    select k_amount_approved
    from kontract
    where k_id = :k_id
    into k_amount;

    k_total_debt = 0;
    k_debt_os = 0;
    k_debt_perc = 0;
    k_debt_fine = 0;
    k_debt_other = 0;
    sh_date = null;
    zad_perc = 0;
    sh_perc_tax = 0;
    sh_calc_perc_tax = 0;

    select sum(zad_os + sh_calc_perc + sh_calc_fine + zad_kommis), sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine),
           sum(zad_kommis), min(coalesce(sh_date, cast('01.01.1900' as date))), sum(zad_perc),
           sum(round(zad_perc / 100 * 2, 0)), sum(round(sh_calc_perc / 100 * 2, 0))
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    into k_total_debt, k_debt_os, k_debt_perc, k_debt_fine, k_debt_other, sh_date, zad_perc, sh_perc_tax,
         sh_calc_perc_tax;

    if (db_id in (39801, 41701)) then
      select res_kommis_amount, -- комиссия при досрочном погашении в КГ
             round(res_kommis_amount / 100 * 2, 0)
      from hp_repay_calc_kommis_kg(:k_id, :c_id, :rep_date)
      into :res_kommis_amount, :res_kommis_amount_tax;

    recom_pay_sum = 0;

    select recom_pay_sum
    from get_recommended_pay_sum(:k_id, :rep_date)
    into recom_pay_sum;

    -- вынес расчет госпошлины выше, чтобы использовать для КЗ тоже
    if (db_id in (81001, 39801)) then
    begin
      select coalesce(sum(res_amount_saldo), 0)
      from hp_repay_extra_saldo(:k_id, 0, :rep_date)
      into :k_curr_other;

      k_curr_other = round(k_curr_other, 2);

      -- для КЗ добавляем это в рекомендуемую сумму
      if (db_id = 39801 and k_curr_other > 0) then
        recom_pay_sum = recom_pay_sum + k_curr_other;
    end

    oks_delinq_date = null;
    select oks_delinq_date
    from oksummary
    where oks_from_k_id = :k_id and
          :rep_date between oks_date_begin and oks_date_end and
          oks_close_state = 0
    into oks_delinq_date;
    days = 0;
    payment_txt = null;
    days_left = null;

    if (oks_delinq_date >= '31.12.2999') then
      days = 0;
    else
      days = abs(rep_date - oks_delinq_date);

    text = case days
             when 1 then days || ' ' || f_lang(lang_code, 'день')
             when 2 then days || ' ' || f_lang(lang_code, 'дня')
             when 3 then days || ' ' || f_lang(lang_code, 'дня')
             when 4 then days || ' ' || f_lang(lang_code, 'дня')
             else days || ' ' || f_lang(lang_code, 'дней')
           end;

    if (sh_date = '01.01.1900') then
    begin
      payment_txt = f_lang(lang_code, 'Просрочка более 360 дней');
      delinq_status = 2;
    end
    else
    if (oks_delinq_date < rep_date) then
    begin
      payment_txt = f_lang(lang_code, 'Просрочка с') || ' ' || f_datetostr(oks_delinq_date);
      days_left = text || ' ' || f_lang(lang_code, 'просрочки');
      delinq_status = 2;
    end
    else
    if (oks_delinq_date >= rep_date) then
    begin
      payment_txt = f_lang(lang_code, 'Оплатить до') || ' ' || f_datetostr(oks_delinq_date);
      days_left = f_lang(lang_code, 'Осталось') || ' ' || text;
      if (days < 7) then
        delinq_status = 1;
      else
        delinq_status = 0;
      days_to_payment = days;
    end

    total_debt = round(k_total_debt, 2);
    debt_perc = round(k_debt_perc, 2);

    if (db_id = 39801) then
    begin
      -- RFCKZ-587
      select first 1 kfr.kfr_id, krd.krd_is_done, krd.krd_from_edl_id, krd.krd_action_date
      from kont_for_restruct_kz kfr
      left join kont_for_restruct_kz_dtl krd on krd.krd_from_kfr_id = kfr.kfr_id
      where kfr.kfr_from_k_id = :k_id
        and kfr.kfr_is_delete = 0
      order by kfr.kfr_id desc, krd.krd_is_done desc
      into :kfr_id, :krd_is_done, :krd_from_edl_id, :krd_action_date;

      if (show_restruct_button = 0
        and days > 0
        and cast(current_time as time) < time '22:45:00'
        and coalesce(kfr_id, 0) > 0
        and coalesce(krd_is_done, 0) = 0
        and current_date <= cast(f_get_const('km_finis_restruct_date') as date)
      ) then
      begin
        if (show_restruct_button = 0
          and f_edoc_status('restruct_application', :k_id, current_date) in ('empty', 'cancelled')
        ) then
          show_restruct_button = 1;

        if (show_restruct_button = 0
          and :krd_action_date + 15 >= current_date
          and f_edoc_status('restruct_application', :k_id, current_date) = 'signed'
          and f_edoc_status('add_loan_consent', :k_id, current_date) in ('empty', 'unsigned', 'cancelled')
        ) then
          show_restruct_button = 1;

      end

      -- RFCKZ-705: Проверка дополнительных условий на реструктуризацию
      if (show_restruct_button = 1) then
      begin
        select is_restructuring_allowed
        from hp_check_restruct_conditions(:k_id)
        into :show_restruct_button;
      end
    end

    if (db_id in (39801, 41701)) then
    begin
      if (pr_code containing '.IPF.') then
      begin
        total_debt = round((k_debt_os + zad_perc + sh_perc_tax), 0); -- Закрыть заем сегодня" если продукт ИПФ
        debt_perc = round(zad_perc, 2); -- сумма наценки
        debt_sh_perc = round(sh_perc_tax, 2); -- налог с продаж
      end
      else
      begin
        res_kommis_amount_tax = coalesce(res_kommis_amount_tax, 0);
        --k_debt_other = k_debt_other + res_kommis_amount + sh_calc_perc_tax + res_kommis_amount_tax;
        total_debt = round((k_total_debt + sh_calc_perc_tax + res_kommis_amount + res_kommis_amount_tax), 0); -- значение поля "Закрыть заем сегодня" по Классике
        debt_sh_perc = round(res_kommis_amount_tax + sh_calc_perc_tax, 2); --налог с продаж
      end
      debt_kommis_amount = round(res_kommis_amount, 2); --комиссия за досрочное
      debt_kommis_amount = replace(debt_kommis_amount, '.00', '') || str_money;

      debt_sh_perc = replace(debt_sh_perc, '.00', '') || str_money;
    end

    amount_approved = round(k_amount, 2);
    debt_os = round(k_debt_os, 2);
    debt_fine = round(k_debt_fine, 2);
    debt_other = round(k_debt_other, 2);
    recomended_amount = round(recom_pay_sum, 2);
    recomended_amount_num = round(recom_pay_sum, 2);

    amount_approved = replace(amount_approved, '.00', '') || str_money;
    total_debt = replace(total_debt, '.00', '') || str_money;
    debt_perc = replace(debt_perc, '.00', '') || str_money;
    debt_os = replace(debt_os, '.00', '') || str_money;
    debt_fine = replace(debt_fine, '.00', '') || str_money;
    debt_other = replace(debt_other, '.00', '') || str_money;
    recomended_amount = replace(recomended_amount, '.00', '') || str_money;


    recomended_amount_str_full = 'Рекомендуемая сумма ' || recomended_amount;

    -- добавил igusev для удобства тестирвоания ------------- >
    k_kontract_no = '';
    kf_name = '';
    k_kolich_client_approved = 0;

    select kont.k_kontract_no, kal.kal_new_kf_name, kont.k_kolich_client_approved, k_disb_date, pr.pr_name
    from kontract kont
    inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    where kont.k_id = :k_id and
          kal.is_delete = 0 and
          :rep_date between kal.kal_date and kal.kal_date_end
    into :k_kontract_no, :kf_name, :k_kolich_client_approved, :k_disb_date, :pr_name;
    -- <
    k_kontract_no_str = 'договор №' || k_kontract_no || ' от ' || f_datetostr(:k_disb_date);

    if (db_id = 81001) then
    begin
        select loan_message
        from hp_get_kontract_sud_info(:k_id, :c_id)
        into :loan_message;

        -- RFCRU-5274: Если клиент банкрот в статусе "Реализация имущества завершена", то долг выводим по нулям
        if (exists(select *
            from bankrupt brt, hp_get_kontur_info(brt.brt_from_eil_id) ki
            where brt.brt_from_c_id = :c_id and
                  current_date between brt.brt_date_begin and brt.brt_date_end and
                  ki.status = 'Реализация имущества завершена')
        ) then
        begin
            need_to_calc_debt = 0;

            total_debt = '0' || str_money;
            debt_perc = '0' || str_money;
            debt_os = '0' || str_money;
            debt_fine = '0' || str_money;
            debt_other = '0' || str_money;

            -- В рекомендуемую суммму платежа выводим 0 со звездочкой
            recomended_amount = '0*' || str_money;
        end

        if (need_to_calc_debt = 1) then
        begin
            -- Общая сумма просроченной задолженности
            select
                sum(zad_os), -- Основная сумма
                sum(iif(coalesce(sh_date, :rep_date) <= :rep_date, sh_calc_perc, zad_perc)), -- Проценты
                sum(sh_calc_fine) -- Пени
            from aarep_make_zad_list(:k_id, :rep_date, 1)
            where sh_date < :rep_date
            into :k_temp_os, :k_temp_perc, :k_curr_fine;
    
            k_curr_os = coalesce(k_temp_os, 0);
            k_curr_perc = coalesce(k_temp_perc, 0);
    
            k_temp_os = null;
            k_temp_perc = null;
    
            -- Текущая задолженность
            select first 1
                iif(sh_date = :rep_date, zad_os, 0), -- Основная сумма
                sh_calc_perc -- Проценты
            from aarep_make_zad_list(:k_id, :rep_date, 1)
            where sh_date >= :rep_date
            order by sh_date
            into :k_temp_os, :k_temp_perc;
    
            k_curr_os = k_curr_os + coalesce(k_temp_os, 0);
            k_curr_perc = k_curr_perc + coalesce(k_temp_perc, 0);
    
            -- Округляем числа до 2 знаков после запятой
            k_curr_os = round(k_curr_os , 2);
            k_curr_perc = round(k_curr_perc, 2);
            k_curr_fine = round(k_curr_fine, 2);
        end

        current_os = nullif(cast(trunc(k_curr_os) as d_str_100), '0') || str_money;
        current_perc = nullif(cast(trunc(k_curr_perc) as d_str_100), '0') || str_money;
        current_fine = nullif(cast(trunc(k_curr_fine) as d_str_100), '0') || str_money;
        current_other = nullif(cast(trunc(k_curr_other) as d_str_100), '0') || str_money;

        current_amount = nullif(cast(trunc(coalesce(k_curr_os, 0) + coalesce(k_curr_perc, 0) + coalesce(k_curr_fine, 0) + coalesce(k_curr_other, 0)) as d_str_100), '0') || str_money;
    end

    suspend;
  end
end
