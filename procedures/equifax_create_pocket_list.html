CREATE PROCEDURE EQUIFAX_CREATE_POCKET_LIST(
    REP_DATE D_DATE,
    CREATE_STANDARD_DAILY_POCKET D_SMALLINT = 1)
RETURNS (
    POCKET_ID D_INTEGER,
    POCKET_NAME STR_255,
    POCKET_DATE D_DATE,
    POCKET_TYPE VARCHAR(50))
AS
declare variable OL_DATE D_DATE;
begin
  if (create_standard_daily_pocket = 1) then -- 1 = создать стандартный дневной пакет
  begin
      pocket_type = '.d.kont.standard_daily.';
      for select pocket_id,
             pocket_name,
             pocket_date
      from equifax_create_pocket(:rep_date, :pocket_type)
      into :pocket_id,
           :pocket_name,
           :pocket_date
      do
      begin
        suspend;
      end

      /* Создаем пакеты по исправленным ошибкам. Ошибки испраляются чз FS. Просмотр - > Журнал ошибок КИБ
         т.к. робот отправляет пакет в 5 утра поэтому на дату rep_date еще испралений не было, т.к. слишком рано, поэтому
         в пакет созданный сегодня должны попасть кредиты котоыре иправлялись вчера.
      ------------------------------------------------------------------------ > */
      for select distinct cp.cp_date_end
      from cib_kontract_errors cke
      inner join cib_pocket cp on cp.cp_id = cke.cke_from_cp_id
      ------- т.к. в сегодняшний пакет REP_DATE передаесядата = вчерашнему дню (т.к. задержка в один день), поэтому испраления которые были сделаны вчера и должны попасть в пакет->
      where cke.cke_checked_datetime >= :rep_date
      and cke.cke_checked_datetime <  :rep_date + 1
      -------------------------------------------------------------------------<
      and cp.cp_cib_id = f_get_rd_id('CIB.EQUIFAX')
      into :pocket_date
      do
      begin
          pocket_type = '.d.kont.errors_corrected.';
          for select pocket_id,
                     pocket_name,
                     pocket_date
          from equifax_create_pocket(:pocket_date, :pocket_type)
          into :pocket_id,
               :pocket_name,
               :pocket_date
          do
          begin
            suspend;
          end  
      end
      ------------------------------------------------------------------------ <

      rep_date = rep_date - 1; -- намеренно уменьшаю дату пакета на 1 день, т.к. ниже пойдет логика, относящаяся к пакетам, где фактическая дата операции отличается от учетной.
  end

  for select ol_date,
             replace('.d.kont.' ||
             trim(iif(ok_repay_by_prev_date > 0, '.ok_pay_prev_d.', '')) ||
             trim(iif(okf_repay_by_prev_date > 0, '.okf_pay_prev_d.', '')) ||
             trim(iif(perc_changed_by_prev_date > 0, '.perc_ch.', '')) ||
             trim(iif(restruct_by_prev_date > 0, '.rest_prev_d.', '')), '..', '.')
  from equifax_create_pocket_list_tmp(:rep_date, 1)
  order by ol_date desc
  into :ol_date, :pocket_type
  do
  begin
      for select pocket_id,
         pocket_name,
         pocket_date
      from equifax_create_pocket(:ol_date, :pocket_type)
      into :pocket_id,
           :pocket_name,
           :pocket_date
      do
      begin
        suspend;
      end
  end

  if (extract(month from :rep_date) <> extract(month from :rep_date + 1)) then
  begin
    for select pocket_id,
         pocket_name,
         pocket_date
    from equifax_create_pocket(:rep_date, '.d.kont.standard_daily.monthly.')
    into :pocket_id,
       :pocket_name,
       :pocket_date
    do
    begin
        suspend;
    end  
  end
end
