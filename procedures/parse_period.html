CREATE PROCEDURE PARSE_PERIOD(
    P_TEXT VARCHAR(50))
RETURNS (
    YEARS INTEGER,
    MONTHS INTEGER)
AS
declare variable v_text varchar(50);
declare variable v_num_str varchar(50);
declare variable v_num numeric(10,2);
begin
  years = 0;
  months = 0;

  v_text = lower(trim(:p_text));

  -- "0"
  if (v_text = '0' or v_text = '') then
  begin
    suspend;
    exit;
  end

  -- "11 и более" / "5 и более лет"
  if (v_text containing 'и более') then
  begin
    years = cast(substring(v_text from 1 for position(' ' in v_text) - 1) as integer);
    months = 0;
  end

  if (v_text containing '-лет') then
  begin
    -- извлекаем число до суффикса "-лет"
    years = cast(trim(substring(v_text from 1 for position('-лет' in v_text) - 1)) as integer);
    months = 0;  -- поскольку это только "лет", месяцев нет
  end

  -- диапазон "2-2.5 года" / "2-2,5 года"
  else if (v_text containing '-') then
  begin
    years = cast(trim(substring(v_text from 1 for position('-' in v_text) - 1)) as integer);

    v_num_str = trim(substring(v_text from position('-' in v_text) + 1));

    if (position(' ' in v_num_str) > 0) then
      v_num_str = substring(v_num_str from 1 for position(' ' in v_num_str) - 1);

    v_num_str = replace(v_num_str, ',', '.');
    v_num = cast(v_num_str as numeric(10,2));

    months = cast((v_num - floor(v_num)) * 12 as integer);
  end

  -- "0,5 года" / "11 лет" / "1,5 года"
  else
  begin
    -- если есть пробел > берем число до него
    if (position(' ' in v_text) > 0) then
      v_num_str = substring(v_text from 1 for position(' ' in v_text) - 1);
    else
      v_num_str = v_text;

    v_num_str = replace(v_num_str, ',', '.');
    v_num = cast(v_num_str as numeric(10,2));

    years = floor(v_num);
    months = cast((v_num - years) * 12 as integer);
  end

  if (v_text containing 'мес') then
  begin
    years = 0;
    if (position(' ' in v_text) > 0) then
      months = cast(substring(v_text from 1 for position(' ' in v_text) - 1) as integer);
    else
      months = cast(v_text as integer);
  end

  if (v_text containing 'дней') then
  begin
    years = 0;
    months = 0;
  end

  -- ?? ограничение: максимум 10 лет
  if (years >= 10) then
  begin
    years = 10;
    months = 0;
  end

  suspend;
end
