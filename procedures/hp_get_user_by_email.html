CREATE PROCEDURE HP_GET_USER_BY_EMAIL(
    USER_EMAIL VARCHAR(100),
    USER_SUBJECT VARCHAR(255))
RETURNS (
    USER_ID DECIMAL(18,0),
    USER_NAME VARCHAR(30),
    USER_C_ID DECIMAL(18,0),
    USER_FIO VARCHAR(100),
    USER_RESULT_TXT VARCHAR(255),
    USER_COMMAND VARCHAR(255))
AS
declare variable un_active integer;
declare variable c_email_verified integer;
begin
  user_result_txt = '';
  user_command = '';
  user_name = '';
  user_c_id = null;

-- получаем C_ID и ФИО через email
  select un_id, un_name, res_object_id, c_name_fio, res_is_verified
  from hp_contdata_find_object(:user_email, 'email'), clients, username
  where res_object_kind = 0 and
        res_object_id = c_id and
        c_id = un_from_c_id rows 1
  into :user_id, :user_name, :user_c_id, user_fio, c_email_verified;

  if (user_c_id is not null) then
  begin
    select un_is_active
    from hp_un_is_active(:user_id)
    into :un_active;

    if (un_active = 0) then -- Если пользователь найден, но он не активный, то ругаемся
    begin
      user_id = null;
      user_result_txt = 'Пользователь заблокирован';
    end
    else
    begin
-- с пользователем все ок, ищем команду для выполнения через Subject письма
      if (c_email_verified <> 1) then
      begin
        user_id = null;
        user_result_txt = 'Ваш email не подтвержден. Обратитесь в службу тех.поддержки за консультацией.';
      end
      else
      begin
        select distinct fsr_sys_command
        from fsrobot
        where :user_subject starting fsr_text rows 1
        into :user_command;
      end
    end
  end
  else
  begin
    user_result_txt = 'Пользователь с такими email не найден';
  end
  suspend;
end
