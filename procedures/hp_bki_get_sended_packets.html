CREATE PROCEDURE HP_BKI_GET_SENDED_PACKETS(
    BKK_ID D_ID,
    BKT_FORMAT D_SMALLINT,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    BKP_ID D_ID,
    BKP_DATE D_DATE,
    BKP_CODE D_STR_CODE,
    ACTION_CODE D_STR_CODE,
    BKP_STATUS D_INTEGER,
    BKP_RESULT_STATUS D_STR_100,
    BKP_SEND_DATETIME D_DATETIME,
    BKP_DATETIME D_DATETIME,
    TOTAL_LOANS D_INTEGER,
    EMPTY_LOANS D_INTEGER,
    SENDED_LOANS D_INTEGER,
    TOTAL_EVENTS D_INTEGER,
    SENDED_EVENTS D_INTEGER,
    REPEATED D_INTEGER,
    ERR_REPEATED D_INTEGER,
    ERRORS D_INTEGER,
    NOT_SEND D_INTEGER,
    LINE_STATUS D_SMALLINT)
AS
declare variable bkl_id d_id;
declare variable bev_id d_id;
declare variable status d_integer;
begin
  if (rep_date_begin is null) then rep_date_begin = current_date;

  for select bkp.bkp_id, bkp_date, bkp_code, bkp_status, bkp.bkp_result_status, bkp_send_datetime, bkp.bkp_datetime,
            f_enum('bki_packet_action', bkt_action, 'code') action_code
  from bki_packet bkp, bki_packet_type
  where bkt_id = bkp_from_bkt_id
    and bkp_date between :rep_date_begin and coalesce(:rep_date_end, '31.12.9999')
    and (:bkk_id is null or bkp_from_bkk_id = :bkk_id)
    and (:bkt_format is null or bkt_format = :bkt_format)
  order by 1
  into bkp_id, bkp_date, bkp_code, bkp_status, bkp_result_status, bkp_send_datetime, bkp_datetime, action_code
  do
  begin
    total_loans   = 0;
    empty_loans   = 0;
    sended_loans  = 0;
    total_events  = 0;
    sended_events = 0;
    repeated      = 0;
    err_repeated  = 0;
    errors        = 0;
    not_send      = 0;

    for select bkl_id
    from bki_loans
    where bkl_from_bkp_id = :bkp_id
    into bkl_id
    do
    begin
      total_loans = total_loans + 1;

      for select bev_id, bev_status
      from bki_clients, bki_events
      where bkc_from_bkl_id = :bkl_id
        and bev_from_bkc_id = bkc_id
      into bev_id, status
      do
      begin
        total_events = total_events + 1;
        if (bkp_status > 1 and status in (0,1)) then
          not_send = not_send + 1;

        if (status > 0) then
          sended_events = sended_events + 1;
        else if (status = - 3) then
          empty_loans = empty_loans + 1;
        else if (status = - 1) then
          errors = errors + 1;
        else if (status = - 2) then
          if (exists(select 1
                     from bki_errors
                     where bke_from_bev_id = :bev_id
                       and bke_error_code in ('2011', '2010')))
          then
            err_repeated = err_repeated + 1;
          else
            repeated = repeated + 1;

      end 
    end 

    line_status = case     -- 0 - норма, 1 - обратить внимание, 2 - аларм
                    when bkp_status < 0 then 2
                    when not_send > 0 then 2
                    when bkp_status < 3 then 1
                    else 0
                  end;

    suspend;
  end 
end
