CREATE PROCEDURE PRINT_SCHEDULE_FULL(
    KONT_ID DECIMAL(18,0),
    LANG VARCHAR(2) = 'RU')
RETURNS (
    SH_ID VARCHAR(10),
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_TOTAL DECIMAL(18,4),
    SH_KOMMIS DECIMAL(18,4),
    SH_DISB_AMOUNT DECIMAL(18,4),
    SH_NAME_FIO VARCHAR(100),
    SH_PERC_TAX DECIMAL(18,4),
    SH_IS_HEADER SMALLINT,
    SH_SUM_DISB CURR_TYPE,
    SH_TOTAL_EX_TAX DECIMAL(18,4),
    KC_BONUS_TEXT VARCHAR(20),
    SH_CL_FULL_TOTAL DECIMAL(18,4),
    SH_CL_FULL_TOTAL_COST DECIMAL(18,4))
AS
declare variable sh_os1 decimal(18,4);
declare variable sh_perc1 decimal(18,4);
declare variable sh_kommis1 decimal(18,4);
declare variable disb_all decimal(18,4);
declare variable sh_perc_tax1 decimal(18,4);
declare variable k_kolich_repay smallint;
declare variable last_sum_os decimal(18,4);
declare variable sh_os_summ curr_type;
declare variable sh_perc_summ curr_type;
declare variable sh_total_summ curr_type;
declare variable sh_kommis_summ curr_type;
declare variable sum_perc_kommis curr_type;
declare variable k_disb_date date;
declare variable k_nsp_percent decimal(18,7);
declare variable sh_total_ex_tax_summ curr_type;
declare variable sh_perc_tax_summ curr_type;
declare variable c_id decimal(10,0);
declare variable kc_kommis_lgota decimal(18,4);
declare variable sum_perc decimal(18,4);
declare variable k_kommis_perc decimal(18,7);
declare variable sh_os2 decimal(18,4);
declare variable sh_perc2 decimal(18,4);
declare variable sh_kommis2 decimal(18,4);
declare variable sh_perc_tax2 decimal(18,4);
declare variable client_count integer;
declare variable k_prepayment_summa decimal(18,4);
declare variable db_id integer;
begin
  db_id = f_get_db_id();

  k_nsp_percent = 0;
  select k_disb_date, kont.k_kommis_perc, coalesce(kont.k_prepayment_summa, 0)
  from kontract kont
  where k_id = :kont_id
  into k_disb_date, k_kommis_perc, k_prepayment_summa;

  select count(*)
  from knclient
  where kc_from_k_id = :kont_id and
        kc_amount_approved > 0
  into :client_count;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end

  select sum(sh_os), sum(sh_perc), sum(sh_kommis)
  from hp_get_base_schedule(:kont_id)
  where ((:db_id in (39801, 41701) and sh_id > 0) or
        :db_id not in (39801, 41701))
  into :sh_os1, :sh_perc1, :sh_kommis1;

  if (sh_os1 is null) then
    sh_os1 = 0;
  if (sh_perc1 is null) then
    sh_perc1 = 0;
  if (sh_kommis1 is null) then
    sh_kommis1 = 0;

  for select k_amount_approved,
             cast(iif(:lang = 'KG', 'Келишим боюнча жалпы т�©л�©�© графиги ', 'Общий график погашений по договору') as varchar(100)),
             k_amount_approved, k_kolich_repay, cast(0 as decimal(18,0)), cast(0 as decimal(18,4))
      from kontract
      where k_id = :kont_id
      union
      select k_amount_approved, c_name_fio, kc_amount_approved, k_kolich_repay, c_id,
             coalesce(kc.kc_kommis_lgota, 0.0)
      from knclient kc, clients, kontract
      where :client_count > 1 and
            kc_from_c_id = c_id and
            kc_from_k_id = :kont_id and
            kc_amount_approved <> 0 and
            k_id = :kont_id
      order by 1
      into disb_all, sh_name_fio, sh_disb_amount, k_kolich_repay, c_id, kc_kommis_lgota
  do
  begin
    if (c_id = 0) then
    begin
      select sum(coalesce(kc_kommis_lgota, 0.0))
      from knclient
      where kc_from_k_id = :kont_id and
            kc_amount_approved > 0
      into :kc_kommis_lgota;
    end
    select c_last_bonus_status_str
    from hp_kontstate_client_status(:c_id, :k_disb_date)
    into  kc_bonus_text;

    sh_os_summ = 0;
    sh_perc_summ = 0;
    sh_perc_tax_summ = 0;
    sh_total_summ = 0;
    sh_total_ex_tax_summ = 0;
    sh_kommis_summ = 0;
    last_sum_os = sh_disb_amount;

    sum_perc = 0;
    sh_cl_full_total = 0;

    sum_perc = sh_perc1;
    sum_perc = cast(sum_perc * (sh_disb_amount / disb_all) as decimal(18,2));

    for select sh_id, sh_date, sh_os, sh_perc, sh_kommis
    from hp_get_base_schedule(:kont_id)
    where ((:db_id in (39801, 41701) and sh_id > 0) or
          :db_id not in (39801, 41701))
    order by sh_date
    into :sh_id, :sh_date, :sh_os1, :sh_perc1, :sh_kommis1
    do
    begin
      if (db_id in (39801, 41701) and
          k_disb_date <= '28.06.2018') then
        sh_perc_tax1 = (select res
                        from fhp_round_bank_old(((:sh_perc1 + :sh_kommis1) * :k_nsp_percent / 100.0), 0));
      else
        sh_perc_tax1 = (select res
                        from fhp_round_bank(((:sh_perc1 + :sh_kommis1) * :k_nsp_percent / 100.0), 0));

      sum_perc_kommis = 0;

      sh_os = cast(sh_os1 * (sh_disb_amount / disb_all) as decimal(18,2));
      sh_perc = cast(sh_perc1 * (sh_disb_amount / disb_all) as decimal(18,2));

      if (sh_kommis1 > 0) then
        sh_kommis = round(sh_disb_amount * k_kommis_perc / 100 - kc_kommis_lgota, 0);
      else
        sh_kommis = 0;

      sum_perc_kommis = sum_perc + sh_kommis;

      sh_perc_tax = cast(sh_perc_tax1 * (sh_disb_amount / disb_all) as decimal(18,2));
      sh_total = sh_os + sh_perc + sh_perc_tax + sh_kommis;
      sh_total_ex_tax = sh_os + sh_perc + sh_kommis;
      sh_sum_disb = null;
      if (sh_id = 0) then
      begin
        if (k_disb_date <> sh_date) then
          exception e_message 'Дата выдачи в договоре отличается от даты выдачи в графике';

        sh_is_header = 1;
        sh_sum_disb = sum_perc_kommis;

        sh_cl_full_total = 0;

        sh_os2 = cast(sh_os1 * (sh_disb_amount / disb_all) as decimal(18,2));
        sh_perc2 = cast(sh_perc1 * (sh_disb_amount / disb_all) as decimal(18,2));

        if (sh_kommis1 > 0) then
          sh_kommis2 = sh_disb_amount * k_kommis_perc / 100 - kc_kommis_lgota;
        else
          sh_kommis2 = 0;

        if (db_id in (39801, 41701) and
            k_disb_date <= '28.06.2018') then
          sh_perc_tax2 = (select res
                          from fhp_round_bank_old(((:sh_perc2 + :sh_kommis2) * :k_nsp_percent / 100.0), 0));
        else
          sh_perc_tax2 = (select res
                          from fhp_round_bank(((:sh_perc2 + :sh_kommis2) * :k_nsp_percent / 100.0), 0));

        sh_perc_tax2 = cast(sh_perc_tax2 * (sh_disb_amount / disb_all) as decimal(18,2));
        sh_cl_full_total = sh_cl_full_total + sh_os2 + sh_perc2 + sh_perc_tax2 + sh_kommis2;
      end
      else
      begin
        sh_is_header = 0;
        sh_sum_disb = null;
      end
      if (sh_id = k_kolich_repay) then
      begin
        sh_os = last_sum_os;
        sh_total = sh_os + sh_perc + sh_perc_tax;
        sh_total_ex_tax = sh_os + sh_perc;
      end

      if (sh_id = 0) then
      begin
        if (k_prepayment_summa > 0) then
        begin
          sh_os = sh_os + k_prepayment_summa;
          sh_total_ex_tax = sh_total_ex_tax + k_prepayment_summa;
          sh_total = sh_total + k_prepayment_summa;
        end
      end

      if (db_id <> 94401) then --Кроме АЗ можно округлять
      begin
        if (sh_kommis <> 0) then
          select res
          from fhp_round_bank(:sh_kommis, 0)
          into sh_kommis;

        if (sh_total <> 0) then
          select res
          from fhp_round_bank(:sh_total, 0)
          into sh_total;
      end

      sh_cl_full_total_cost = round(sh_cl_full_total - sh_disb_amount, 0);

      suspend;

      if (sh_id = 0) then
      begin
        if (k_prepayment_summa > 0) then
        begin
          sh_os = sh_os - k_prepayment_summa;
          sh_total_ex_tax = sh_total_ex_tax - k_prepayment_summa;
          sh_total = sh_total - k_prepayment_summa;
        end
      end

      sh_os_summ = sh_os_summ + sh_os;
      sh_perc_summ = sh_perc_summ + sh_perc;
      sh_perc_tax_summ = sh_perc_tax_summ + sh_perc_tax;
      sh_total_summ = sh_total_summ + sh_total;
      sh_total_ex_tax_summ = sh_total_ex_tax_summ + sh_total_ex_tax;
      if (sh_id = k_kolich_repay) then
      begin
        if (db_id <> 94401) then --Кроме АЗ можно округлять
        begin
          if (sh_kommis <> 0) then
            select res
            from fhp_round_bank(:sh_kommis_summ, 0)
            into sh_kommis;

          if (sh_total <> 0) then
            select res
            from fhp_round_bank(:sh_total_summ + :k_prepayment_summa, 0)
            into sh_total;
        end

        sh_os = sh_os_summ + k_prepayment_summa;
        sh_perc = sh_perc_summ;
        sh_perc_tax = sh_perc_tax_summ;
        sh_total_ex_tax = sh_total_ex_tax_summ;
        sh_id = null;
        sh_date = null;
        sh_sum_disb = :sh_disb_amount;
        sh_id = 'Всего:';
        if (lang = 'KG') then
          sh_id = 'Жалпы:';
        suspend;
      end

      last_sum_os = last_sum_os - coalesce(sh_os, 0);
    end
  end
end
