CREATE PROCEDURE MC_ONLINE_LOAN_EDOC_LIST(
    MCL_ID D_ID,
    ONLINE_APP_ID D_INTEGER)
RETURNS (
    EDS_ID D_ID,
    EDC_NAME D_STR_255)
AS
declare variable c_id d_id;
declare variable exa_id d_id;
declare variable ma_id d_id;
declare variable k_id d_id;
declare variable miniapp_obj_kind d_integer;
declare variable kontract_obj_kind d_integer;
declare variable miniapp_edoc_kind d_integer;
declare variable kontract_edoc_kind d_integer;
begin
  /*
  Необходимо реализовать процедуру на базе MC_EDOC_LIST только чтобы она возращала электроныне документы по онлайн займу
  Ожидается что будут видны документы: заявка, кредитный договор
  */
  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into c_id;

  if (c_id is null) then
    exit;

  select oa.oa_from_exa_id
  from online_app oa
  where oa.oa_id = :online_app_id
  into exa_id;

  select exa.exa_from_ma_id, exa.exa_from_k_id
  from expressapp exa
  where exa.exa_id = :exa_id
  into ma_id, k_id;

  miniapp_obj_kind = f_objkind('miniapp', 'id');
  kontract_obj_kind = f_objkind('kontract', 'id');
  miniapp_edoc_kind = f_edoc_kind('loan.application', 'id');
  kontract_edoc_kind = f_edoc_kind('loan.agreement', 'id');

  for with table1
      as (
      --Заявки
      select edl.edl_id as edl_id, edl.edl_from_edc_id as edc_id
      from edoc_list edl
      where edl.edl_object_kind = :miniapp_obj_kind and
            edl.edl_object_id = :ma_id and
            edl.edl_status >= 0 and
            edl.edl_from_edc_id = :miniapp_edoc_kind
      union
      --Кредитные договора
      select edl.edl_id as edl_id, edl.edl_from_edc_id as edc_id
      from edoc_list edl
      where edl.edl_object_kind = :kontract_obj_kind and
            edl.edl_object_id = :k_id and
            edl.edl_status >= 0 and
            edl.edl_from_edc_id = :kontract_edoc_kind)
      select eds.eds_id, edc.edc_name
      from table1 t1
      inner join edoc_sign eds on eds.eds_from_edl_id = t1.edl_id and
            eds.eds_from_c_id = :c_id --Показываем только документы клиента
      inner join edoc_cfg edc on edc.edc_id = t1.edc_id
      into eds_id, edc_name
  do
    suspend;
end
