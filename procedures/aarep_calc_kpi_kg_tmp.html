CREATE PROCEDURE AAREP_CALC_KPI_KG_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    K_KOLICH INTEGER,
    K_KC_ID DECIMAL(18,0),
    K_REGION_ID INTEGER,
    K_BALANCE DECIMAL(18,4))
AS
declare variable curr_curr varchar(3);
declare variable k_curr varchar(3);
declare variable k_curr_rate decimal(18,4);
declare variable kb decimal(18,4);
declare variable ok_type integer;
declare variable k_id decimal(18,0);
declare variable new_rate decimal(18,4);
declare variable pr_set_rate_in_kont integer;
begin
  select cm_default_currency
  from company
  into :curr_curr;

  for select k_id, k_index_curr_code, k_index_curr_rate, k_kolich_client_approved, kal_new_kf_officer, kal_new_reg_id,
             pr_set_rate_in_kont
      from kontract, knaktlist, kontstate, region_list(:rep_region_id, 1, -1, 1), product
      where k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            cast(ks_date_begin as date) <= :rep_date_end and
            ks_date_end >= :rep_date_end and
            ks_state = 1 and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            (kal_new_reg_id = id_list) and
            k_from_pr_id = pr_id
      into :k_id, :k_curr, :k_curr_rate, :k_kolich, :k_kc_id, :k_region_id, :pr_set_rate_in_kont
  do
  begin
    k_balance = 0;
    select oks_balance
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_close_state = 0
    into :k_balance;
    if ((k_balance is null) or (k_balance <= 0)) then
      k_balance = 0;

    new_rate = 1;
    if (curr_curr <> k_curr) then
    begin
      select cr_rate
      from opercurr
      where cr_from_cu_id = :k_curr and
            cr_date <= :rep_date_end and
            cr_date_end >= :rep_date_end
      into :new_rate;
      if (new_rate is null) then
        new_rate = 1;
      if (new_rate < k_curr_rate) then
      begin
        if (pr_set_rate_in_kont = 1) then
          new_rate = :k_curr_rate;
      end
      k_balance = k_balance * :new_rate;
    end
    if (k_balance > 0) then
      suspend;
  end
end
