CREATE PROCEDURE AAREP_OPER_CLASSIFIC_NERES_BANK(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RES_BANK_ID VARCHAR(10),
    RES_GROUP_REP_1_2 VARCHAR(20),
    RES_CLIENT_COUNTRY VARCHAR(100),
    RES_OPERCODE_STR VARCHAR(20),
    RES_CURRENCY VARCHAR(10),
    REP_AMOUNT_PLUS CURR_TYPE,
    REP_AMOUNT_MINUS CURR_TYPE,
    RES_BANK_NAME VARCHAR(255),
    ERR_TEXT VARCHAR(1024))
AS
declare variable res_bank_id_prior varchar(10);
begin
  err_text = '';
  if (not(exists(select *
                 from temp_moneyclassific
                 rows 1))) then
  begin
-- Отчет запускается первый раз, заполняем временную таблицу.
    select err_text
    from aarep_oper_classific_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
    into :err_text;
    if (err_text <> '') then
    begin
      suspend;
      exit;
    end
  end
-- Ниже данные будут браться из временной таблице

  res_bank_id_prior = '';
  for select res_bank_id, res_group_rep_1_2, res_client_country, res_opercode_str, res_currency,
             sum(iif(res_oper_sign = 1, res_amount, 0.0)), -- Зачисление
             sum(iif(res_oper_sign = -1, res_amount, 0.0)) -- Списание
      from temp_moneyclassific
      where res_is_bank <> 0 and
            res_client_is_resident = 0 and
            res_bank_id <> ''
      group by res_bank_id, res_group_rep_1_2, res_client_country, res_opercode_str, res_currency
      into :res_bank_id, :res_group_rep_1_2, res_client_country, res_opercode_str, res_currency, :rep_amount_plus,
           :rep_amount_minus
  do
  begin
    if (res_bank_id_prior <> res_bank_id) then
    begin
      res_bank_name = '';
      select b_name
      from bankregion, banks
      where b_id = :res_bank_id rows 1
      into :res_bank_name;
    end
    suspend;
  end
end
