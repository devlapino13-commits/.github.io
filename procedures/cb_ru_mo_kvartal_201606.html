CREATE PROCEDURE CB_RU_MO_KVARTAL_201606(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    OKRUG_NAME VARCHAR(100),
    BALANCE CURR_TYPE,
    DELINQ_BALANCE_30 CURR_TYPE,
    PAR30 DECIMAL(15,2),
    BONUS_FACTOR INTEGER,
    KOL_SVZ_ON_BONUS INTEGER,
    BONUS_MO INTEGER,
    KOL_SSVZ INTEGER,
    FIO_SSVZ VARCHAR(1024))
AS
declare variable month1_begin date;
declare variable month1_end date;
declare variable month2_begin date;
declare variable month2_end date;
declare variable month3_begin date;
declare variable month3_end date;
declare variable reg_id integer;
begin
  month3_end = :rep_date_end;
  month3_begin = month3_end - extract(day from :month3_end) + 1;
  month2_end = month3_begin - 1;
  month2_begin = month2_end - extract(day from :month2_end) + 1;
  month1_end = month2_begin - 1;
  month1_begin = month1_end - extract(day from :month1_end) + 1;

  for select oks_region_id, sum(oks_balance), sum(iif(:rep_date_end - oks_delinq_date > 30, oks_balance, 0))
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_region_id = id_list and
            :rep_date_end - oks_delinq_date <= 360 and
            oks_kc_id <> 1000000000003 and
            oks_close_state = 0
      group by oks_region_id
      into reg_id, balance, delinq_balance_30
  do
  begin
    par30 = cast(delinq_balance_30 as double precision) / balance * 100;
    bonus_factor = 0;
    kol_svz_on_bonus = 0;
    bonus_mo = 0;
    if (par30 < 9) then
    begin
      bonus_factor = case
                       when par30 < 3 then 5000
                       when par30 < 5 then 4000
                       when par30 < 7 then 2000
                       when par30 < 9 then 1000
                       else 0
                     end;

      kol_svz_on_bonus = 0;
      with tabl1
      as (select c_id
          from cb_ru_lo_one_lo_from_mo(:reg_id, 0, :month1_begin, :month1_end)
          where aarep_bonus_total_to_pay > 0
          union all
          select c_id
          from cb_ru_lo_one_lo_from_mo(:reg_id, 0, :month2_begin, :month2_end)
          where aarep_bonus_total_to_pay > 0
          union all
          select c_id
          from cb_ru_lo_one_lo_from_mo(:reg_id, 0, :month3_begin, :month3_end)
          where aarep_bonus_total_to_pay > 0),
      tabl2
      as (select c_id, count(*) kol
          from tabl1
          where exists(select *
                       from hr_pstatus -- не включать уволенных СВЗ
                       where :rep_date_end between ps_date_begin and ps_date_end and
                             ps_status = 1 and
                             ps_is_working = 1 and
                             ps_from_c_id = c_id)
          group by c_id
          having count(*) >= 2)
      select count(*)
      from tabl2
      into kol_svz_on_bonus;
      bonus_mo = bonus_factor * kol_svz_on_bonus;
    end

--    Вычисление кол-во ССВЗ
    kol_ssvz = 0;
    fio_ssvz = '';
    select count(*), list(c_name_fio, '; ')
    from hr_pstatus, joblist, jobkind, clients
    where ps_status = 1 and
          ps_is_working = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code = 'JobSystem.Region_KM' and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_reg_id = :reg_id and
          ps_from_c_id = c_id
    into kol_ssvz, fio_ssvz;
    if (kol_ssvz is null) then
      kol_ssvz = 0;
    fio_ssvz = coalesce(fio_ssvz, '');
    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
    from region r1, region r2, region r3
    where r1.reg_unicode = :reg_id and
          r1.parent = r2.reg_unicode and
          r2.parent = r3.reg_unicode
    into reg_name, branch_name, okrug_name;

    suspend;
  end
end
