CREATE PROCEDURE HP_KN_OKSUMMARY_BANKRUPT_UPDATE(
    REP_DATE D_DATE)
AS
declare variable brt_date_begin d_date;
declare variable k_stop_perc d_date;
declare variable month_begin d_date;
declare variable k_kredline_end d_date;
declare variable k_from_kf_id d_id;
declare variable kal_new_reg_id d_integer;
declare variable kal_new_kf_officer d_id;
declare variable kal_new_no d_integer;
declare variable kal_new_kf_name d_str_100;
declare variable ka_memo d_str_long;
declare variable email_sent d_smallint;
declare variable sirena_name d_str_100;
declare variable officer_name d_str_100;
declare variable client_count d_smallint;
declare variable un_id_fsrobot d_id;
declare variable rd_id_finished d_integer;
declare variable k_new_id d_id;
declare variable err_text d_str_code;
declare variable c_name_fio d_str_100;
declare variable k_kontract_no d_str_20;
declare variable sirena_id d_id;
declare variable reg_id_hq d_integer;
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable ka_id d_id;
declare variable brt_date_fact d_date;
begin
  -- Процедура обновляет записи по банкротству и перестраивает проводки ЕПС

  un_id_fsrobot = 1000000000000;
  rd_id_finished = f_get_rd_id('Reason_Amnesty.bankruptcy_finished');
  sirena_id = 1000000000004;
  reg_id_hq = 1010100; -- Москва (ГО)

--  if (rep_date <> f_date_month_end(rep_date)) then
--    exception e_default 'Дата должна быть концом месяца';

  month_begin = dateadd(day, -extract(day from rep_date) + 1, rep_date);

  select c_name_fio
  from clients
  where c_id = :sirena_id
  into sirena_name;
  sirena_name = coalesce(sirena_name, 'КЦ Сирена');

  --Цикл по всем банкротам
  for select brt_from_c_id, min(brt_date_begin), min(brt_date_fact)
  from bankrupt
  where brt_from_c_id not in (1010101293164,1010101293494) -- RFCRU-5571
  group by 1
  into c_id, brt_date_begin, brt_date_fact
  do
  begin
    email_sent = 0;

    for select kc_id, kc_from_k_id, k_stop_perc, k_kredline_end, k_from_kf_id
    from knclient, kontract
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = k_id
    into kc_id, k_id, k_stop_perc, k_kredline_end, k_from_kf_id
    do
    begin
      if (k_stop_perc is null and brt_date_fact >= month_begin and brt_date_begin < k_kredline_end) then
      begin
        -- Если на дату банкротства договор активный и не расторгнут
        if (exists(select * from kontstate
        where ks_from_k_id = :k_id and
              ks_date_begin <= :brt_date_begin and  --на дату проверки
              ks_date_end >= :brt_date_begin and
              ks_ol_is_delete = 0 and
              ks_state = 1) and
        --------------RFCRU-3222 и нет записи в operkred, расторгаем датой судебного решения
        not exists(select * from operkred ok
        where ok.ok_from_k_id = :k_id and
              ok.ok_ol_date between :brt_date_fact and :rep_date and --между датой судебного решения и датой проверки
              ok.ok_type not in (3, 5) and
              ok.ok_ol_is_active = 1)) then
        begin
          execute procedure hp_kontract_stop_perc(k_id, brt_date_fact, brt_date_fact);
        end
      end

      if (brt_date_begin >= month_begin) then
      begin
        kal_new_kf_name = '';
        kal_new_reg_id = null;

        select kal_new_kf_name, kal_new_reg_id, kal_new_no, kal_new_kf_officer
        from kontstate
        inner join kontract on k_id = ks_from_k_id
        inner join knaktlist on kal_from_kf_id = k_from_kf_id
        where ks_from_k_id = :k_id and
              ks_date_begin <= :brt_date_begin and
              ks_date_end >= :brt_date_begin and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              is_delete = 0 and
              kal_date_end = '31.12.9999'
        into kal_new_kf_name, kal_new_reg_id, kal_new_no, kal_new_kf_officer;

        if (kal_new_kf_name <> '' and kal_new_reg_id is not null) then
        begin
          -- Отправка уведомления по почте
          if (email_sent = 0) then
          begin
            execute procedure hp_transfer_kfolder(:kal_new_reg_id, null, null, null, null, :kal_new_kf_name, null, null, :c_id, 1);
            email_sent = 1;
          end

          if (kal_new_kf_officer > sirena_id) then
          begin
            select c_name_fio
            from clients
            where c_id = :kal_new_kf_officer
            into officer_name;
            officer_name = coalesce(officer_name, '<Неизвестно>');

            ka_memo = 'Прием-передача кредитных папок (Выборочно) от ' || officer_name || ' к КС ' || sirena_name;
            execute procedure hp_transfer_kfolder(kal_new_reg_id, rep_date, ka_memo, k_from_kf_id, sirena_id,
                kal_new_kf_name, kal_new_no, '.ВС', c_id, 0);
          end
        end
      end

      if (exists(select * from kn_oksummary
      where oks_from_kc_id = :kc_id and
            oks_date_begin = :rep_date)) then
      begin
        execute procedure hp_kn_oksummary_recalc(:kc_id, :rep_date);
        execute procedure hp_do_kn_operbook_one_online(:k_id, current_date, null, :rep_date - 1);
      end
    end
  end

  -- Для клиентов с ЗАВЕРШЕННЫМ банкротством отправляем папки в КЦ Сирена, ГО

  -- Собираем завершенных банкротов
  for select c_id, c_name_fio, min(brt_date_fact)
  from bankrupt, clients
  where brt_from_c_id = c_id
    and brt_is_finished = 1
    and brt_from_c_id not in (1010101293164,1010101293494) -- RFCRU-5571
  group by c_id, c_name_fio
  into c_id, c_name_fio, brt_date_fact
  do
  begin
    -- Проходим по всем незакрытым кредитам клиента
    for select k_id, kc_id, k_kontract_no
    from knclient, kontract, oksummary
    where kc_from_c_id = :c_id and
          kc_from_k_id = k_id and
          kc_amount_approved > 0 and
          oks_from_k_id = k_id and
          :rep_date between oks_date_begin and oks_date_end and
          oks_close_state in (0, 1)
    into k_id, kc_id, k_kontract_no
    do
    begin
      -- Считаем количество клиентов
      select count(*)
      from knclient
      where kc_from_k_id = :k_id
        and kc_amount_approved > 0
      into client_count;

      -- Для групповых кредитов, выделяем клиента из договора (амнистируем)
      client_count = coalesce(client_count, 0);
      k_new_id = null;
      if (client_count > 1) then
      begin
        select k_new_id, k_new_no
        from hp_devide_kontract_clients(:k_id, :c_id, :un_id_fsrobot, :rep_date, 'Банкротство завершено', :rd_id_finished)
        into k_new_id, err_text;

        if (coalesce(k_new_id, 0) = 0) then
          exception e_default f_format('Клиент: %1 (%2), договор: :%3 (%4), ошибка:%5.<br>', c_name_fio, c_id, k_kontract_no, k_id, err_text);
        else
          k_id = k_new_id;
      end

      select kal_new_kf_name, kal_new_reg_id, kal_new_no, kal_new_kf_officer, k_from_kf_id
      from kontract
      left join knaktlist on kal_from_kf_id = k_from_kf_id
      where k_id = :k_id and
            is_delete = 0 and
            kal_date_end = '31.12.9999'
      into kal_new_kf_name, kal_new_reg_id, kal_new_no, kal_new_kf_officer, k_from_kf_id;

      -- Передаем в ГО, если еще не передано или СВЗ не КЦ Сирена
      if (kal_new_reg_id <> reg_id_hq or kal_new_kf_officer <> sirena_id) then
      begin
        ka_memo =  f_format('Прием-передача кредитных папок клиента банкрота в %2 ГО', sirena_name);
        execute procedure hp_transfer_kfolder(reg_id_hq, rep_date, ka_memo, k_from_kf_id, sirena_id,
            kal_new_kf_name, kal_new_no, '.ВС', c_id, 0);

        -- Получаем созданный ka_id и делаем проводки
        select kal_from_ka_id
        from knaktlist
        where kal_from_kf_id = :k_from_kf_id and
              is_delete = 0 and
              kal_date_end = '31.12.9999'
        into ka_id;
        execute procedure hp_save_knaktlist_operbook(ka_id);
      end
      -- Вставляем в БКИ
      if (not exists (
          select * from bki_bankrupt
          where bbt_from_k_id = :k_id)) then
        insert into bki_bankrupt (bbt_from_k_id, bbt_date)
        values (:k_id, :brt_date_fact);

      -- Пересчитываем ЕПС
      if (exists(select * from kn_oksummary
      where oks_from_kc_id = :kc_id and
            oks_date_begin = :rep_date)) then
      begin
        execute procedure hp_kn_oksummary_recalc(:kc_id, :rep_date);
        execute procedure hp_do_kn_operbook_one_online(:k_id, current_date, null, :rep_date - 1);
      end
    end
  end
end
