CREATE PROCEDURE SAVE_PASS_CHECK_RESULT(
    C_ID D_ID,
    RESULT_INFO D_BLOB_TEXT)
AS
declare variable eil_id d_id;
declare variable eic_id d_integer;
declare variable c_name_fio d_str_100;
declare variable bls_caption d_str_100;
declare variable bls_id d_id;
begin
  select eic.eic_id
  from extinfocfg eic
  where eic.eic_code = 'passport_status'
  into eic_id;

  select c.c_name_fio
  from clients c
  where c.c_id = :c_id
  into c_name_fio;

  bls_caption = 'Инфоком, статус паспорта, ' || :c_name_fio;

  insert into extinfolist (eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_is_json, eil_from_eip_id,
                           eil_from_un_id, eil_from_c_id)
  values (:eic_id, current_date, f_objkind('clients', 'id'), :c_id, 1, null, 1000000000000, :c_id)
  returning eil_id
  into :eil_id;

  select bls_id
  from hp_insert_into_blobstore('passport_status_json.txt', :bls_caption, :result_info, 1000000000000, 2, 3)
  into :bls_id;

  update extinfolist eil
  set eil.eil_status = 2,
      eil.eil_from_bls_id = :bls_id
  where eil.eil_id = :eil_id and
        eil.eil_from_c_id = :c_id;
end
