CREATE PROCEDURE AAREP_EXPRESSAPP(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    REGION_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    RES_TEXT D_STR_100,
    EXC_NAME D_STR_100,
    KS_DATE_BEGIN D_DATE,
    LO_NAME D_STR_100,
    EXA_GROUP_ID D_INTEGER,
    EDL_BLANK_DATE D_DATE,
    EXA_RESHIPPING_COUNT D_INTEGER,
    EXA_DATE D_DATE,
    MA_EDL_DATE D_DATE,
    K_EDL_DATE D_DATE,
    CONSULT_EDL_DATE D_DATE)
AS
declare variable c_id id_type;
declare variable exc_id d_integer;
declare variable exa_final_status d_smallint;
declare variable reg_id d_integer;
declare variable k_id id_type;
declare variable lo_id id_type;
declare variable exa_id d_integer;
declare variable exa_wait_status d_smallint;
declare variable ma_id d_id;
declare variable obj_kind_miniapp d_integer;
declare variable obj_kind_kontract d_integer;
declare variable obj_kind_expressapp d_integer;
declare variable ma_edl_status d_smallint;
declare variable k_edl_status d_smallint;
declare variable exa_edl_status d_smallint;
declare variable edoc_kind d_integer;
declare variable exa_disb_date d_date;
declare variable ma_is_active d_smallint;
begin
  obj_kind_miniapp = f_objkind('miniapp', 'id');
  obj_kind_kontract = f_objkind('kontract', 'id');
  obj_kind_expressapp = f_objkind('expressapp', 'id');
  edoc_kind = f_edoc_kind('consulting.agreement', 'id');

  for select exa.exa_from_c_id, exa.exa_from_exc_id, exa.exa_final_status, exa.exa_from_reg_id, exa.exa_from_k_id,
             exa.exa_lo_id, exa.exa_group_id, exa.exa_id, exa.exa_reshipping_count, exa.exa_wait_status, exa.exa_date,
             exa.exa_from_ma_id, exa.exa_disb_date
      from expressapp exa
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = exa.exa_from_reg_id
      where exa.exa_id > 0 and
            exa.exa_date between :rep_date_begin and :rep_date_end
      into c_id, exc_id, exa_final_status, reg_id, k_id, lo_id, exa_group_id, exa_id, exa_reshipping_count,
           exa_wait_status, exa_date, ma_id, exa_disb_date
  do
  begin
    c_name_fio = null;
    c_inn = null;
    select cl.c_name_fio, cl.c_inn
    from clients cl
    where cl.c_id = :c_id
    into c_name_fio, c_inn;

    lo_name = null;
    select cl.c_name_fio
    from clients cl
    where cl.c_id = :lo_id
    into lo_name;

    office_name = null;
    region_name = null;
    select rg.reg_code || ' ' || rg.reg_name, rg2.reg_name
    from region rg
    inner join region rg2 on rg2.reg_unicode = rg.parent
    where rg.reg_unicode = :reg_id and
          rg2.reg_unicode > 0
    into office_name, region_name;

    select ma.ma_is_active
    from miniapp ma
    where ma.ma_id = :ma_id and
          ma.ma_date_end >= '31.12.2999'
    into :ma_is_active;

    ks_date_begin = null;
    select ks.ks_date_begin
    from kontstate ks
    where ks.ks_from_k_id = :k_id and
          ks.ks_ol_is_delete = 0 and
          ks.ks_state = 1 and
          :rep_date_end between ks.ks_date_begin and ks.ks_date_end
    into ks_date_begin;

    edl_blank_date = null;
    exa_edl_status = null;
    select edl.edl_blank_date, edl.edl_status
    from edoc_list edl
    where edl.edl_object_kind = :obj_kind_expressapp and
          edl.edl_object_id = :exa_id and
          edl.edl_blank_date >= :exa_date
    order by edl.edl_id desc
    rows 1
    into edl_blank_date, exa_edl_status;

    exc_name = null;
    select exc.exc_name
    from expresscfg exc
    where exc.exc_id = :exc_id
    into exc_name;

    ma_edl_status = null;
    ma_edl_date = null;
    select edl.edl_status, edl.edl_blank_date
    from edoc_list edl
    where edl.edl_object_kind = :obj_kind_miniapp and
          edl.edl_object_id = :ma_id
    order by edl.edl_id desc
    rows 1
    into ma_edl_status, ma_edl_date;

    consult_edl_date = null;
    k_edl_status = null;
    k_edl_date = null;

    if (ma_edl_status = 1) then
    begin
        select edl.edl_blank_date
        from edoc_list edl
        where edl.edl_object_kind = 0 and
              edl.edl_object_id = :c_id and
              edl.edl_from_edc_id = :edoc_kind and
              edl.edl_blank_date between :exa_date and :exa_disb_date and
              edl.edl_date_end is null
        order by edl.edl_id desc
        rows 1
        into consult_edl_date;
    

        select edl.edl_status, edl.edl_blank_date
        from edoc_list edl
        where edl.edl_object_kind = :obj_kind_kontract and
              edl.edl_object_id = :k_id
        order by edl.edl_id desc
        rows 1
        into k_edl_status, k_edl_date;
    end
    res_text = '';
    --------------------------------------------------------------------------------
    if(ma_is_active = -1 or
       exa_final_status = -1 and
       exc_id <> 30) then
      res_text='Отказ компанией';
    --------------------------------------------------------------------------------
    else
    if (exc_id <= 10 and
        exa_final_status = 0) then
      res_text = 'В ожидании отправки предложения клиенту';
    --------------------------------------------------------------------------------
    else
    if ((exc_id < 30) and
        (exa_final_status < 0)) then
      res_text = 'Отказ компанией';

    --------------------------------------------------------------------------------
    else
    if (exc_id = 30 and
        exa_wait_status = 2 and
        exa_edl_status >= 0) then
      res_text = 'В ожидании ответа клиента на предложение';

    --------------------------------------------------------------------------------
    else
    if (exc_id = 30 and
        exa_final_status = -2) then
      res_text = 'Клиент отказался';

    --------------------------------------------------------------------------------
    else
    if (exc_id = 30 and
        exa_final_status = -1) then
      res_text = 'Срок действия предложения истек';

    --------------------------------------------------------------------------------
    else
    if (exc_id = 30 and
        exa_wait_status = 2) then
      res_text = 'Ожидание ответа от клиента';
    --------------------------------------------------------------------------------
    else
    if (exc_id = 45 and
        exa_wait_status = 1) then
      res_text = 'Робот ждёт подтверждения актуальности данных от ВИПСа';
    --------------------------------------------------------------------------------
    else
    if (exc_id = 55 and
        exa_wait_status = 1) then
      res_text = 'Робот ждёт отправки эл. заявки от ВИПСа';
    --------------------------------------------------------------------------------
    else
    if ((exc_id = 60) and
        (exa_edl_status = 1) and
        (ma_edl_status <= 0)) then
      res_text = 'Предложение принял, но клиент не подписал Заявку';
    --------------------------------------------------------------------------------
    else
    if (exc_id = 60 and
        exa_wait_status = 2) then
      res_text = 'Ожидание ответа от клиента';
    --------------------------------------------------------------------------------
    else
    if (exc_id > 60 and
        exc_id <= 150) then
      res_text = 'Возникла ошибка';
    --------------------------------------------------------------------------------
    else
    if (exc_id = 150 and
        exa_final_status = -1) then
      res_text = 'Отказ компанией';
    --------------------------------------------------------------------------------
    else
    if (exc_id = 151 and
        exa_wait_status = 1) then
      res_text = 'Робот ждёт отправки договора на оказание конс. услуг от ВИПСа';
    --------------------------------------------------------------------------------
    else
    if (exc_id = 152 and
        exa_wait_status = 2) then
      res_text = 'Ожидание ответа от клиента';
    --------------------------------------------------------------------------------
    else
    if (exc_id = 155 and
        exa_wait_status = 1) then
      res_text = 'Робот ждёт отправки договора займа от ВИПСа';
    --------------------------------------------------------------------------------
    else
    if ((exc_id >= 160) and
        (k_edl_status <= 0) and
        (exa_wait_status <> -1)) then
      res_text = 'Предложение принял,подписал Заявку,но клиент не подписал Договор';
    --------------------------------------------------------------------------------
    else
    if (exc_id >= 160 and
        exa_wait_status = -1) then
      res_text = 'Предложение принял, подписал Заявку, Договор отправлен, но появилась ошибка у робота';
    --------------------------------------------------------------------------------
    else
    if ((ks_date_begin is null) and
        (exa_edl_status = 1) and
        (ma_edl_status = 1) and
        (k_edl_status = 1)) then
      res_text = 'Предложение принял, подписал Заявку и Договор, но не забрал ДС';
    --------------------------------------------------------------------------------
    else
    if (ks_date_begin is not null) then
      res_text = 'Кредит выдан';
    --------------------------------------------------------------------------------
    else
    if (exa_wait_status = -1) then
    res_text = 'Возникла ошибка у робота';
    --------------------------------------------------------------------------------
    else
    if (exc_id > 160 and ks_date_begin is null) then
    res_text = 'Было аннулирование со стороны ВИПС';
    --------------------------------------------------------------------------------

    suspend;
  end
end
