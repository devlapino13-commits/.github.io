CREATE PROCEDURE HP_PERSTEMP_SEND_TO_VERIFY(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_BOOL)
AS
declare variable only_photo_changed d_smallint;
declare variable c_type             d_integer;
declare variable c_blob_pass1_id    d_id;
declare variable c_blob_pass2_id    d_id;
declare variable c_blob_photo_id    d_id;
declare variable c_verify_status    d_smallint;
declare variable cancel_verify      d_bool;
begin
  only_photo_changed = 0;
  res_is_error = 1;
  res_text = '';

  if (not exists(select * from refdata where rd_code starting 'ActionGroup.Passport_Verify' and rd_is_active = 1)) then
    res_text = 'Модуль верификации паспорта отключен';

  if (exists(select * from actionorder
              where aco_object_kind = 0 and
                    aco_object_id = :c_id and
                    aco_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO')))) then
    res_text = 'Данные уже находятся в очереди на верификацию';

  if (res_text = '') then
  begin
    select c_type, c_blob_pass1_id, c_blob_pass2_id, c_blob_photo_id, c_verify_status
    from clients
    where c_id = :c_id
    into c_type, c_blob_pass1_id, c_blob_pass2_id, c_blob_photo_id, c_verify_status;

    select pte_blob_pass1_id, pte_blob_pass2_id, pte_blob_photo_id
    from perstemp
    where pte_from_c_id = :c_id
    into c_blob_pass1_id, c_blob_pass2_id, c_blob_photo_id;

    if (c_type <> 0) then
      res_text = 'Юр.лица не подлежат верификации';
    else
    if (coalesce(c_blob_pass1_id, 0) = 0 or coalesce(c_blob_pass2_id, 0) = 0) then
      res_text = 'Скан паспорта не прикреплен';
    else
    if (coalesce(c_blob_photo_id, 0) = 0) then
      res_text = 'Фото не прикреплено';
  end

  if (res_text = '') then
  begin
    if (exists(select * from aarep_check_finpolis_one(:c_id))) then
    begin
      res_text = 'ВНИМАНИЕ,' ||
                 'Необходимо связаться с комплайнс менеджером ГО.' ||
                 'Дальнейшие операции с этим контрагентом запрещены.' ||
                 'Детали в меню "Проверка по спискам ТиЭ"';
    end
  end

  if (res_text = '') then
  begin
    cancel_verify = 1;
    if (f_tmp_const('use_perstemp') = '1') then
    begin
      if (not exists(select * from perstemp
                     where pte_from_c_id = :c_id)) then
      begin
        insert into perstemp (pte_from_c_id)
        values (:c_id);
      end

      if (c_verify_status = 2) then
        cancel_verify = 0;
    end

    execute procedure hp_send_client_to_verify(:c_id, :un_id, :only_photo_changed);

    if (cancel_verify = 1) then
    begin
      update clients
      set c_verify_status = 0
      where c_id = :c_id and
            c_verify_status <> 0;
    end

    res_is_error = 0;
    res_text = 'Данные успешно отправлены на верификацию';
  end

  suspend;
end
