CREATE PROCEDURE CHANGE_CLIENT_NAMES
AS
declare variable c_id decimal(18,0);
declare variable c_type integer;
declare variable c_sex integer;
declare variable c_name varchar(100);
declare variable fl integer;
declare variable yl integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  fl = 0;
  yl = 0;
  for select c_id, c_type, c_sex
      from clients
      order by c_id
      into :c_id, :c_type, :c_sex
  do
  begin
    c_name = '';
    if (c_type = 0) then
      fl = fl + 1;
    else
      yl = yl + 1;
    if (c_sex = 0) then
    begin
      if (c_type = 0) then
        c_name = 'Физ.лицо Ж ' || fl;
      else
        c_name = 'Юр.лицо ' || yl;
    end
    else
    begin
      if (c_type = 0) then
        c_name = 'Физ.лицо М ' || fl;
      else
        c_name = 'Юр.лицо ' || yl;
    end
    update clients
    set c_name_fio = :c_name
    where c_id = :c_id;
  end
end
