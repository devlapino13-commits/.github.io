CREATE PROCEDURE AAREP_PAYMENTIPF_WITHCOMMIS(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_CODE,
    KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    C_NAME_FIO D_STR_100,
    K_DISB_DATE D_DATE,
    K_KREDLINE_END D_DATE,
    SELLER_NAME D_STR_100,
    ML_NAME D_STR_255,
    KML_TOTAL D_CURRENCY,
    KML_COMMIS D_CURRENCY,
    KML_PAYMENT D_CURRENCY)
AS
declare variable kf_id d_id;
declare variable rd_id d_integer;
declare variable reg_lst d_str_8100;
begin
  reg_lst = (select list(id_list, ';') from region_list(:rep_region_id, 1, -1, 1));
  rd_id = f_get_rd_id('AgreementKinds.IPF_WithCommission');
  for
    select
      k_kontract_no,
      k_from_kf_id,
      c1.c_name_fio,
      k_disb_date,
      k_kredline_end,
      c2.c_name_fio,
      ml_name,
      kml_total,
      kml_commis,
      kml_total-kml_commis
    from agreements
      inner join clients as c2
        on c2.c_id = ag_object_id
      inner join kontmat
        on kml_from_ag_id = ag_id
          and kml_commis > 0
      inner join kontract
        on k_id = kml_object_id
          and k_disb_date between :rep_date_begin and :rep_date_end
      inner join knclient
        on kc_from_k_id = k_id
          and kc_amount_approved > 0
          and kc_status = 2
      inner join clients as c1
        on c1.c_id = kc_from_c_id
      inner join matlist
        on ml_id = kml_from_ml_id
    where ag_from_rd_id = :rd_id
  into
    :k_kontract_no,
    :kf_id,
    :c_name_fio,
    :k_disb_date,
    :k_kredline_end,
    :seller_name,
    :ml_name,
    :kml_total,
    :kml_commis,
    :kml_payment
   do
   begin
     for
       select
         kal_new_kf_name,
         reg_name
       from knaktlist
         inner join region
           on reg_unicode = kal_new_reg_id
       where kal_from_kf_id = :kf_id
         and :rep_date_end between kal_date and kal_date_end
         and :reg_lst containing kal_new_reg_id
         and knaktlist.is_delete = 0
     into
       :kf_name,
       :reg_name
     do
     begin
       suspend;
     end
   end
end
