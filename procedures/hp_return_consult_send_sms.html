CREATE PROCEDURE HP_RETURN_CONSULT_SEND_SMS(
    ARC_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable status_datetime timestamp;
declare variable status d_smallint;
begin

  select eds.eds_status_datetime, eds.eds_status
  from app_return_consult_info arci
  inner join edoc_sign eds on eds.eds_from_edl_id = arci.arci_from_edl_id
  where arci.arci_id = :arc_id
  into :status_datetime, :status;

  if (:status_datetime is not null and datediff(second, :status_datetime, current_timestamp) < 300) then
    begin
        if (:status = 0) then
            :res_text = 'С момента прошлой отправки СМС прошло менее 5-ти минут.';
        else
            :res_text = 'Заявление было подписано клиентом.';
        suspend;
        exit;
    end

  select edl_id, res_text
  from hp_edoc_insert('app.return_consultation', f_objkind('app_return_consult', 'id'), :arc_id, :un_id)
  into :edl_id, :res_text;

  if (:res_text = '' and
      :edl_id > 0) then

  begin
    update app_return_consult_info
    set arci_from_edl_id = :edl_id
    where (arci_id = :arc_id);
  end

  suspend;
end
