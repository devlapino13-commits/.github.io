CREATE PROCEDURE HP_KNCLIENT_ANKETA_INCOME(
    KC_ID D_INTEGER,
    CALC_DATE D_DATE)
RETURNS (
    MONTHLY_INCOME D_CURRENCY,
    WORK_INCOME D_CURRENCY,
    WORK_INCOME_PERIOD D_STR_100,
    WORK_EXTRA_INCOME D_CURRENCY)
AS
declare variable c_id d_id;
declare variable ovs_value str_255;
begin
  monthly_income = 0;

  select kc_from_c_id
  from knclient
  where kc_id = :kc_id
  into c_id;

  if (c_id is not null) then
  begin
    work_income = 0;
    ovs_value = '';
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'Work_Monthly_Income', :calc_date)
    into ovs_value;
  
    ovs_value = coalesce(ovs_value, 0);
    ovs_value = trim(replace(ovs_value, 'Значение: ', ''));
    if (ovs_value similar to '[-]?[[:DIGIT:]]+') then
    begin
      work_income = abs(cast(ovs_value as id_type));

      work_income_period = null;
      select res_rd_code
      from hp_objectprops_get(0, :c_id, -1, 'work_income_period', :calc_date)
      into work_income_period;
      work_income_period = coalesce(work_income_period, '');

      monthly_income = (case work_income_period
        when 'work_income_period.daily' then  work_income * 30
        when 'work_income_period.three_times_per_week' then work_income / 7 * 30 * 3
        when 'work_income_period.twice_per_week' then work_income / 7 * 30 * 2
        when 'work_income_period.weekly' then work_income / 7 * 30
        when 'work_income_period.three_times_per_month' then work_income * 3
        when 'work_income_period.twice_per_month' then work_income * 2
        when 'work_income_period.monthly' then work_income
        else work_income
      end);

      -- По запросу Елены Грабовской, если превышаем 200 тыс, берем оригинальную цифру
      -- 17.08.2020 по запросу Елены, начиная со второго квартала не обращаем внимание на work_income_period
      -- if (monthly_income > 200000 or calc_date >= '01.04.2020') then
        monthly_income = work_income;
    end

    work_extra_income = 0;
    ovs_value = '';
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'Work_Extra_Amount', :calc_date)
    into :ovs_value;

    ovs_value = coalesce(ovs_value, 0);
    ovs_value = trim(replace(ovs_value, 'Значение: ', ''));
    if (ovs_value similar to '[-]?[[:DIGIT:]]+') then
      work_extra_income = abs(cast(ovs_value as id_type));

    monthly_income = monthly_income + work_extra_income;
  end

  -- По запросу Елены Грабовской, если менее 1000, игнорируем сумму дохода
  if (monthly_income < 1000) then
    monthly_income = 0;

  -- Переводим на русский язык
  work_income_period = trim(case work_income_period
    when 'work_income_period.daily' then  'Ежедневно'
    when 'work_income_period.three_times_per_week' then 'Трижды в неделю'
    when 'work_income_period.twice_per_week' then 'Дважды в неделю'
    when 'work_income_period.weekly' then 'Раз в неделю'
    when 'work_income_period.three_times_per_month' then 'Трижды в месяц'
    when 'work_income_period.twice_per_month' then 'Дважды в месяц'
    when 'work_income_period.monthly' then 'Раз в месяц'
    else 'Другое'
  end);

  suspend;
end
