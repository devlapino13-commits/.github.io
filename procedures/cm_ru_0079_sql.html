CREATE PROCEDURE CM_RU_0079_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID ID_TYPE;
declare variable Z_FROM_C_ID ID_TYPE;
declare variable C_NAME_FIO STR_100;
declare variable C_COMMENT STR_255;
declare variable K_KONTRACT_NO D_STR_20;
begin
  -- проверка под фт на уровне контракта
  for select kncl.kc_from_c_id, k.k_kontract_no
      from kontract k
      inner join knclient kncl on k.k_id = kncl.kc_from_k_id
      where k.k_id = :k_id and
            kncl.kc_amount_approved > 0
      into :c_id, k_kontract_no
  do
  begin
    for select c_name_fio, c_comment
        from hp_check_client_fio_ft(:c_id, 1)
        into :c_name_fio, :c_comment
    do
    begin
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into :res_info_text;
      --            if (c_comment='рфм (сфр): фио') then
      res_error_is_critical = 2;
      --            else
      --                res_error_is_critical = 2;
      res_error_text = 'Совпадение с лицом по ' || c_comment || ' ' || c_name_fio;

      if (check_kind = '.loan_disb.') then
      begin
        --письмо комплайенс  RFCRU-4399
        execute procedure hp_send_quick_message('email', 'ru.oft@molbulak.com', 'Совпадение с лицом по ' || c_comment,
          'клиент ' || :c_name_fio ||
          ', совпадение с лицом по: ' ||  :c_comment ||
          ', номер договора: ' || :k_kontract_no,
          '', null, :un_id);
      end
      suspend;
    end
  end
  for select z.z_from_c_id, cl.c_name_fio
      from zalog z
      inner join clients cl on  z.z_from_c_id = cl.c_id
      where z.z_from_k_id = :k_id
      into :z_from_c_id, :res_info_text
  do
  begin
    for select c_name_fio, c_comment
        from hp_check_client_fio_ft(:z_from_c_id, 1)
        into :c_name_fio, :c_comment
    do
    begin
      res_error_is_critical = 2;
      res_error_text = 'Залогодатель в списке ФТ, выдача кредита запрещена';
      suspend;
    end
  end
end
