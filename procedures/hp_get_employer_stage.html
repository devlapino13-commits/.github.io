CREATE PROCEDURE HP_GET_EMPLOYER_STAGE(
    EMP_C_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    EMP_STAGE_YEAR D_CURRENCY,
    EMP_STAGE_MONTH D_CURRENCY,
    EMP_STAGE_DAY D_CURRENCY)
AS
begin
  /* Трудовой стаж сотрудника */

  emp_stage_year = 0;
  emp_stage_month = 0;
  emp_stage_day = 0;
  select sum(datediff(day, ps_date_begin, iif(ps_date_end > :rep_date, :rep_date, ps_date_end)) + 1)
  from hr_pstatus
  where ps_from_c_id = :emp_c_id and
        ps_is_working = 1 and
        ps_status = 1 and
        ps_date_begin < ps_date_end and
        ps_date_begin <= :rep_date
  into emp_stage_day;

  emp_stage_month = emp_stage_day / 30.0;
  emp_stage_year = emp_stage_month / 12.0;

  suspend;
end
