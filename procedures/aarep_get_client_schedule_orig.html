CREATE PROCEDURE AAREP_GET_CLIENT_SCHEDULE_ORIG(
    C_ID DECIMAL(18,0),
    DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(255),
    K_KONTRACT_DATE DATE,
    RES_ID INTEGER,
    RES_DATE DATE,
    RES_GR_PLAN_AMOUNT DECIMAL(18,4),
    RES_CL_PLAN_AMOUNT DECIMAL(18,4),
    RES_GR_OS DECIMAL(18,4),
    RES_CL_OS DECIMAL(18,4),
    RES_GR_PERC DECIMAL(18,4),
    RES_CL_PERC DECIMAL(18,4),
    RES_GR_FINE DECIMAL(18,4),
    RES_CL_FINE DECIMAL(18,4),
    K_DISB_DATE DATE)
AS
declare variable K_ID decimal(18,0);
declare variable GR_AMOUNT decimal(18,4) = 0;
declare variable CL_AMOUNT decimal(18,4) = 0;
declare variable KOEF double precision;
declare variable KC_CLOSED_OL_ID decimal(18,0);
declare variable KC_AMOUNT_APPROVED decimal(18,4) = 0;
declare variable KC_FROM_C_ID decimal(18,0);
declare variable TAX_PERC decimal(18,7);
declare variable DB_ID D_ID;
declare variable ROUND_VALUE D_SMALLINT;
declare variable CLIENT_COUNT D_INTEGER;
begin
  db_id = f_get_db_id();
  round_value = iif(db_id = 94401, 2, 0);

-- Если параметр Client
  for select kc_from_k_id,
             '<b>' || k_kontract_no || '</b> от <b>' || right('0' || extract(day from k_disb_date), 2) || '.' || right('0' || extract(month from k_disb_date), 2) || '.' || extract(year from k_disb_date) || '</b><br>Кр.продукт: <b>' || pr_name || '</b>',
             ks_date_begin,
             k_disb_date -- чтобы вытащить первоначальный график то проверям на дату выдачи
      from knclient, kontstate, kontract, product
      where ks_from_k_id = kc_from_k_id and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            ks_state = 1 and
            ks_date_end >= '31.12.2999' and
            ks_ol_is_delete = 0 and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id
      order by k_disb_date
      into :k_id, k_kontract_no, k_kontract_date, k_disb_date
  do
  begin
    tax_perc = 0;
    select nsp_percent_for_int
    from hp_get_kontract_nsp_acc(0, :k_kontract_date)
    into :tax_perc;

-- по индексированным кредитам не должно начисляться НСП, добавил Улан от 30.03.2016
    if (k_id in (109010020831300101, 109010020831500101, 109010020831700101, 109010020831900101)) then
    begin
      tax_perc = 0;
    end

    select count(kc_from_c_id)
    from knclient
    where kc_from_k_id = :k_id
      and kc_amount_approved > 0
    into :client_count;

    gr_amount = 0;
    cl_amount = 0;
    res_gr_os = 0;
    res_cl_os = 0;
    res_gr_perc = 0;
    res_cl_perc = 0;
    res_gr_fine = 0;
    res_cl_fine = 0;

    for select kc_from_c_id, kc_amount_approved, kc_closed_ol_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into :kc_from_c_id, :kc_amount_approved, :kc_closed_ol_id
    do
    begin
      if (kc_from_c_id = c_id) then
        cl_amount = kc_amount_approved;
      if (kc_closed_ol_id is null) then
        gr_amount = gr_amount + kc_amount_approved;
      else
      if (not(exists(select *
                     from operlist
                     where ol_id = :kc_closed_ol_id and
                           is_delete = 0))) then
        gr_amount = gr_amount + kc_amount_approved;
      else
      if (kc_from_c_id = c_id) then
        cl_amount = 0;
    end
    if (gr_amount <> 0) then
      koef = cl_amount / gr_amount;
    else
      koef = 1.0;

    for select sh_id, sh_date, zad_os, zad_perc, sh_calc_fine + iif(sh_id < 0, zad_kommis, 0.0),
               s_all + sh_calc_fine + round((zad_perc * :tax_perc) / 100.00, :round_value)
        from aarep_make_zad_list(:k_id, :k_disb_date, 1)  -- date_end заменили на k_disb_date
        where sh_date is not null
        into :res_id, :res_date, res_gr_os, res_gr_perc, res_gr_fine, res_gr_plan_amount
    do
    begin
      res_cl_plan_amount = round(res_gr_plan_amount * koef, :round_value);
      res_cl_os = round(res_gr_os * koef, :round_value);
      res_cl_perc = round(res_gr_perc * koef, :round_value);
      res_cl_fine = round(res_gr_fine * koef, :round_value);

      if (:client_count = 1 and
          :db_id = 39801) then
      begin
        res_gr_plan_amount = 0;
        res_gr_os = 0;
        res_gr_perc = 0;
        res_gr_fine = 0;
      end
      suspend;
    end
  end

-- Если параметр Kontract
  for select ks_from_k_id,
             '<b>' || k_kontract_no || '</b> от <b>' || right('0' || extract(day from k_disb_date), 2) || '.' || right('0' || extract(month from k_disb_date), 2) || '.' || extract(year from k_disb_date) || '</b><br>Кр.продукт: <b>' || pr_name || '</b>',
             ks_date_begin, k_amount_approved
      from kontstate, kontract, product
      where ks_from_k_id = :c_id and
            ks_state = 1 and
            ks_date_end >= '31.12.2999' and
            ks_ol_is_delete = 0 and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id
      order by k_disb_date
      into :k_id, k_kontract_no, k_kontract_date, gr_amount
  do
  begin
    tax_perc = 0;
    select nsp_percent_for_int
    from hp_get_kontract_nsp_acc(0, :k_kontract_date)
    into :tax_perc;

-- по индексированным кредитам не должно начисляться НСП, добавил Улан от 30.03.2016
    if (k_id in (109010020831300101, 109010020831500101, 109010020831700101, 109010020831900101)) then
    begin
      tax_perc = 0;
    end

    cl_amount = gr_amount;
    koef = 1.0;

    for select sh_id, sh_date, zad_os, zad_perc, sh_calc_fine,
               s_all + sh_calc_fine + round((zad_perc * :tax_perc) / 100.00, :round_value)
        from aarep_make_zad_list(:k_id, :k_disb_date, 1)  -- date_end заменили на k_disb_date
        where sh_date is not null
        into :res_id, :res_date, res_gr_os, res_gr_perc, res_gr_fine, res_gr_plan_amount
    do
    begin
      res_cl_plan_amount = res_gr_plan_amount;
      res_cl_os = res_gr_os;
      res_cl_perc = res_gr_perc;
      res_cl_fine = res_gr_fine;

      if (:client_count = 1 and
          :db_id = 39801) then
      begin
        res_gr_plan_amount = 0;
        res_gr_os = 0;
        res_gr_perc = 0;
        res_gr_fine = 0;
      end
      suspend;
    end
  end

end
