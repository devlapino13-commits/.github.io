CREATE PROCEDURE HP_BLOB_MOVE(
    CB_FROM_C_ID DECIMAL(18,0),
    CB_ID INTEGER,
    CB_KIND INTEGER,
    UN_ID DECIMAL(18,0))
AS
declare variable cb_blob blob sub_type 0 segment size 80;
declare variable cb_from_un_id decimal(18,0);
declare variable cb_datetime timestamp;
declare variable cb_name varchar(255);
declare variable cb_filename varchar(255);
declare variable bls_id integer;
begin
  for select cb_name, cb_blob, cb_filename, cb_from_un_id
      from clientblob
      where cb_from_c_id = :cb_from_c_id and
            cb_id = :cb_id and
            cb_blob is not null and
            (cb_from_bls_id is null or
            cb_from_bls_id = 0)
      into :cb_name, :cb_blob, :cb_filename, :cb_from_un_id
  do
  begin

    if (cb_blob is not null) then
    begin
      execute procedure hp_insert_into_blobstore (:cb_filename, :cb_name, :cb_blob, :un_id, 2, 0)
      returning_values :bls_id;

      if (cb_kind = 0) then -- Прочие документы
      begin
        update clientblob
        set cb_from_bls_id = :bls_id,
            cb_blob = null,
            cb_from_un_id = :un_id
        where cb_from_c_id = :cb_from_c_id and
              cb_id = :cb_id;
      end
      if (cb_kind = 1) then -- Фото клиента
      begin
        update clients
        set c_blob_photo_id = :bls_id
        where c_id = :cb_from_c_id;
      end
      if (cb_kind = 2) then -- Лицевая часть паспорта
      begin
        update clients
        set c_blob_pass1_id = :bls_id
        where c_id = :cb_from_c_id;
      end
      if (cb_kind = 3) then -- Обратная часть паспорта, прописка
      begin
        update clients
        set c_blob_pass2_id = :bls_id
        where c_id = :cb_from_c_id;
      end
      if (cb_kind <> 0) then -- Прочие документы
      begin
        delete from clientblob
        where cb_from_c_id = :cb_from_c_id and
              cb_id = :cb_id;
      end
    end
  end

end
