CREATE PROCEDURE CB_PERSONAL_DATA(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_C_ID DECIMAL(18,0))
RETURNS (
    CB_PERSONAL_ID DECIMAL(18,0),
    CB_REGION_ID INTEGER,
    CB_PERSONAL_FIO VARCHAR(100),
    CB_POSITION_NAME VARCHAR(100),
    CB_JOBKIND_NAME VARCHAR(100),
    CB_JOBTITLE_NAME VARCHAR(100),
    CB_IS_WORKING_STR VARCHAR(20),
    CB_OFFICE_NAME VARCHAR(100),
    CB_BRANCH_NAME VARCHAR(100),
    CB_REGION_NAME VARCHAR(100),
    CB_OKRUG_NAME VARCHAR(100),
    CB_IS_WORKING INTEGER,
    CB_PERSONAL_STAGE DATE)
AS
declare variable ps_is_working integer;
begin
  select c_name_fio
  from clients
  where c_id = :rep_c_id
  into :cb_personal_fio;
  cb_personal_id = :rep_c_id;
  cb_region_id = :rep_region_id;

  cb_personal_stage = null;
  select min(ks_date_begin)
  from kontstate
  where ks_kc_id = :rep_c_id
--  and ks_region_id=:REP_REGION_ID
        and
        ks_state = 1 and
        ks_ol_is_delete = 0
  into :cb_personal_stage;
  if (cb_personal_stage is null) then
  begin
    select min(ps_date_begin)
    from hr_pstatus
    where ps_from_c_id = :rep_c_id
--    and ps_from_reg_id=:REP_REGION_ID
    into :cb_personal_stage;
  end

  cb_position_name = null;
  for select distinct jl_name, jk_name, jtl_name, ps_is_working
      from hr_pstatus, joblist, jobkind, jobtitle
      where ps_from_c_id = :rep_c_id
--  and ps_from_reg_id=:REP_REGION_ID
            and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jl_title = jtl_id
      into :cb_position_name, :cb_jobkind_name, :cb_jobtitle_name, :cb_is_working
  do
  begin
    if (cb_is_working = 1) then
      cb_is_working_str = 'Работает';
    else
      cb_is_working_str = 'Уволен';
    if (cb_position_name is null) then
    begin
      cb_position_name = 'Нет приказа';
      cb_jobkind_name = 'Нет приказа';
      cb_jobtitle_name = 'Нет приказа';
      cb_is_working_str = 'Нет приказа';
    end

    select aa_curr_office_name, aa_curr_branch_name, aa_curr_level3_name, aa_curr_level4_name
    from aarep_get_office_categ(:rep_region_id, :rep_date_end)
    into :cb_office_name, :cb_branch_name, :cb_region_name, cb_okrug_name;

    suspend;
  end
end
