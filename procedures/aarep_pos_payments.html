CREATE PROCEDURE AAREP_POS_PAYMENTS(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_TYPE D_SMALLINT)
RETURNS (
    PARENT_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    TERM_NAME D_STR_255,
    SUM_AMOUNT CURR_TYPE,
    KOLVO_TRANS D_INTEGER,
    TERM_TYPE D_STR_10,
    COMMIS_PERC D_PERCENT)
AS
declare variable trm_id d_id;
declare variable rd_id_tsp d_integer;
declare variable rd_id_tsp_ipf d_integer;
declare variable ag_id d_integer;
declare variable cs_obj_kind d_integer;
declare variable cl_obj_kind d_integer;
begin
  rep_type = coalesce(rep_type, 0);

  if (rep_type = 1) then  -- Для всех ПВН кроме МБулак
  begin
    rd_id_tsp = f_get_rd_id('AgreementKinds.tsp');
    rd_id_tsp_ipf = f_get_rd_id('AgreementKinds.tsp_and_ipf');
    cs_obj_kind = f_objkind('clientshops', 'id');
    cl_obj_kind = f_objkind('clients', 'id');

    for select ita.ita_term_name, rg.reg_name, rg2.reg_name, its.trm_id, coalesce(ag.ag_commis_cards_other, 0)
        from ipc_terminals its
        inner join ipc_terminal_app ita on ita.ita_from_trm_id = its.trm_id
        inner join ipc_terminal_ownership ito on ito.ito_trm_id = its.trm_id
        inner join region rg on rg.reg_unicode = ita.ita_from_reg_id
        inner join region rg2 on rg2.reg_unicode = rg.parent
        inner join clientshops cs on cs.cs_id = ito.ito_object_id and
            cs.cs_from_rd_id = :rd_id_tsp
        inner join agreements ag on ag.ag_object_id = cs.cs_from_c_id
        where :rep_date_end between ito.ito_date_begin and ito.ito_date_end and
              ito.ito_status > -1 and
              rg.reg_unicode > 0 and
              rg2.reg_unicode > 0 and
              ita.ita_term_name <> 'М Булак' and
              ita.ita_doc_date between ito.ito_date_begin and ito.ito_date_end and
              ito.ito_object_kind = :cs_obj_kind and
              ag.ag_object_kind = :cl_obj_kind and
              ag.ag_status = 1 and
              ag.ag_from_rd_id in (:rd_id_tsp, :rd_id_tsp_ipf) and
              :rep_date_end between ag.ag_date_begin and coalesce(ag.ag_date_end, '31.12.9999')
        into term_name, office_name, parent_name, trm_id, commis_perc
    do
    begin
      kolvo_trans = null;
      sum_amount = null;

      select count(t.igdp_terminal_id), sum(t.igdp_amount)
      from ipc_gc_analiz_detail_pos t
      inner join ipc_gc_analiz iga on iga.iga_id = t.igdp_from_iga_id
      inner join ipc_file_analiz ifa on ifa.ifa_id = iga.iga_from_ifa_id
      where ifa.ifa_date between :rep_date_begin and :rep_date_end and
            t.igdp_terminal_id = :trm_id
      group by igdp_from_ag_id
      into kolvo_trans, sum_amount;

      sum_amount = coalesce(sum_amount, 0);
      kolvo_trans = coalesce(kolvo_trans, 0);
      term_type = 'POS';

      suspend;
    end
  end
  else  -- Для ПВН МБулак
  if (rep_type = 0) then
  begin
    for select ita.ita_term_name, rg.reg_name, rg2.reg_name, its.trm_id
        from ipc_terminals its
        inner join ipc_terminal_app ita on ita.ita_from_trm_id = its.trm_id
        inner join ipc_terminal_ownership ito on ito.ito_trm_id = its.trm_id
        inner join region rg on rg.reg_unicode = ita.ita_from_reg_id
        inner join region rg2 on rg2.reg_unicode = rg.parent
        where :rep_date_end between ito.ito_date_begin and ito.ito_date_end and
              ito.ito_status > -1 and
              rg.reg_unicode > 0 and
              rg2.reg_unicode > 0 and
              ita.ita_term_name = 'М Булак' and
              ita.ita_doc_date between ito.ito_date_begin and ito.ito_date_end
        into term_name, office_name, parent_name, trm_id
    do
    begin
      kolvo_trans = null;
      sum_amount = null;

      select count(t.igdp_terminal_id), sum(t.igdp_amount)
      from ipc_gc_analiz_detail_pos t
      inner join ipc_gc_analiz iga on iga.iga_id = t.igdp_from_iga_id
      inner join ipc_file_analiz ifa on ifa.ifa_id = iga.iga_from_ifa_id
      where ifa.ifa_date between :rep_date_begin and :rep_date_end and
            t.igdp_terminal_id = :trm_id
      group by igdp_from_ag_id
      into kolvo_trans, sum_amount;

      commis_perc = 0;
      sum_amount = coalesce(sum_amount, 0);
      kolvo_trans = coalesce(kolvo_trans, 0);
      term_type = 'POS';

      suspend;
    end
  end 
end
