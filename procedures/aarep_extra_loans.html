CREATE PROCEDURE AAREP_EXTRA_LOANS(
    REP_REGION D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    NUM D_INTEGER,
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    C_BIRTH_DATE D_DATE,
    K_KONTRACT_NO D_STR_20,
    KAL_NEW_KF_NAME D_STR_100,
    KC_AMOUNT_APPROVED D_CURRENCY,
    K_FACT_DISB_DATE D_DATE,
    K_PERCENT D_PERCENT,
    K_FACT_CLOSED_DATE D_DATE,
    OKO_OS D_CURRENCY,
    OKO_PERC D_CURRENCY,
    OKO_TOTAL D_CURRENCY,
    OFFICE_NAME D_STR_CODE,
    ADL_ADDRESS_REG D_STR_255,
    ADL_ADDRESS_FACT D_STR_255,
    REGION_REG D_STR_255,
    KS_FIO D_STR_255,
    SBL_NAME D_STR_255)
AS
declare variable reg_id d_integer;
declare variable k_id id_type;
declare variable c_id id_type;
begin
  --RFCKG-1143, отчет по внесистемному учету на дату

  num = 0;

  for select id_list
      from region_list(:rep_region, 1, 3, 1)
      into reg_id -- все офисы по переданному региону/стране
  do
  begin

    for select ks.ks_from_k_id
        from kontstate ks
        inner join operkroff okf on okf.oko_from_k_id = ks.ks_from_k_id
        where ks.ks_region_id = :reg_id and -- фильтр по региону
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 2 and -- контракты закрытые
              :rep_date between ks.ks_date_begin and ks.ks_date_end and -- закрытые на дату
              okf.oko_ol_is_active = 1 and
              okf.oko_ol_date <= :rep_date -- даты операций до отчетной даты (включительно)
        group by ks.ks_from_k_id
        having sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc) > 0 -- есть внесистемный долг
        into k_id -- id контракта
    do
    begin
      for select c.c_id,
                 c.c_name_fio,
                 c.c_inn,
                 c.c_birth_date,
                 round(kc.kc_amount_approved, 2)
          from knclient kc
          inner join clients c on c.c_id = kc.kc_from_c_id
          where kc.kc_from_k_id = :k_id and
                kc.kc_amount_approved > 0
          into c_id, -- id клиента
               c_name_fio, -- ФИО клиента
               c_inn, -- ПИН (ИНН) клиента
               c_birth_date, -- дата рождения клиента
               kc_amount_approved -- cобственная сумма займа клиента
      do
      begin
        num = num + 1; -- порядковый номер строки

        select k.k_kontract_no,
               k.k_fact_disb_date,
               round(k.k_percent, 2),
               k.k_fact_closed_date,
               kal.kal_new_kf_name,
               reg.reg_name,
               c.c_name_fio
        from kontract k
        inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
        inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
        inner join clients c on c.c_id = kal.kal_new_kf_officer
        where k.k_id = :k_id and
              kal.is_delete = 0 and
              :rep_date between kal.kal_date and kal.kal_date_end
        into k_kontract_no, -- номер контракта
             k_fact_disb_date, -- дата выдачи
             k_percent, -- процентная ставка
             k_fact_closed_date, -- дата закрытия контракта
             kal_new_kf_name, -- номер группы (папки)
             office_name, -- текущий офис
             ks_fio; -- текущий КС

        oko_os = 0;
        oko_perc = 0;
        oko_total = 0;
        select round(sum(okf.oko_sh_os - okf.oko_os), 2),
               round(sum(okf.oko_sh_perc - okf.oko_perc), 2)
        from operkroff okf
        where okf.oko_from_k_id = :k_id and
              okf.oko_from_c_id = :c_id and
              okf.oko_ol_is_active = 1
        group by okf.oko_from_c_id
        into oko_os,
             oko_perc;

        oko_total = oko_os + oko_perc;

        adl_address_reg = '';
        select adl.adl_address_str,
               f_get_oblast_kg(adl.adl_address_str)
        from addrclient adc
        inner join addrlist adl on adc.adc_from_adl_id = adl.adl_id
        inner join places pl on pl.pl_id = adl.adl_from_pl_id
        where adc.adc_from_c_id = :c_id and
              adc.adc_address_type = 1
        into adl_address_reg, -- адрес регистрации (прописки)
             region_reg; -- регион регистрации (область прописки)

        adl_address_fact = '';
        select adl.adl_address_str
        from addrclient adc
        inner join addrlist adl on adc.adc_from_adl_id = adl.adl_id
        where adc.adc_from_c_id = :c_id and
              adc.adc_address_type = 2
        into adl_address_fact; -- фактический адрес проживания

        select coalesce(sbl.sbl_name, '')
        from sb_akt sba
        inner join sb_list sbl on sbl_id = sba_from_sbl_id
        where sba.sba_from_k_id = :k_id and
              sba.sba_status = 2 and
              sba.sba_is_delete = 0 and
              :rep_date between sba.sba_date_begin and sba.sba_date_end and
              sbl.sbl_id > 0 rows 1
        into sbl_name; -- Подразделение (КЦ, ОВР, Юр. отдел)

        suspend;
      end
    end
  end
end
