CREATE PROCEDURE HP_UX_MAIN_LIST
RETURNS (
    RES_ID D_STR_255,
    RES_PARENT_ID D_STR_255,
    RES_KIND D_STR_CODE,
    RES_IMAGE D_SMALLINT,
    RES_NAME D_STR_100,
    RES_CODE D_STR_100,
    RES_MEMO D_STR_LONG,
    RES_KIND_CAPTION D_STR_CODE,
    RES_DATETIME D_DATETIME,
    RES_TABLE_NAME D_STR_CODE,
    RES_WARNING_STATE D_SMALLINT)
AS
declare variable ux_root d_str_code;
declare variable tmp_str d_str_code;
declare variable ux_root_ui d_str_code;
declare variable ux_root_main d_str_code;
declare variable tmp_table d_str_code;
begin
  res_warning_state = 0;

  res_image = 70; -- папка
  res_table_name = '';

  res_kind_caption = '';
  res_datetime = null;

  res_image = 70; -- папка
  res_kind = 'table.ui';
  res_parent_id = null;
  res_id = 'table.ui';
  res_code = '';
  res_memo = 'Таблицы, участвующие в интерфейсах или формах поисках';
  res_name = 'Таблицы с интерфейсами';
  ux_root_ui = res_id;
  suspend;


  res_kind = 'object.main';
  res_parent_id = null;
  res_id = 'object.main';
  res_code = '';
  res_memo = 'Служебные объекты';
  res_name = 'Объекты';
  suspend;

  res_parent_id = res_id;

  res_image = 72; -- Картинка с Fx
  res_kind = 'keyname';
  res_id = 'keyname';
  res_code = '';
  res_memo = 'SQL код получения названия объекта по ключевому полю (F_KeyName)';
  res_name = 'Названия по значению';
  suspend;

  res_image = 43; -- Картинка портфеля
  res_kind = 'mainmenu';
  res_id = 'mainmenu';
  res_code = 'fm_Mainmenu';
  res_memo = 'Главное меню приложения';
  res_name = 'Главное меню';
  suspend;

  res_image = 70; -- папка

  res_kind = 'table.main';
  res_parent_id = null;
  res_id = 'table.main';
  res_code = '';
  res_memo = 'Таблицы без привязки к интерфейсам';
  res_name = 'Другие таблицы';
  ux_root_main = res_id;
  suspend;

  res_parent_id = ux_root_ui;

  execute procedure hp_temp_uxtable_fill;

  res_kind = 'table';
  res_kind_caption = 'Таблица';
  res_datetime = null;

  for select lower(ut_code), coalesce(ut_name, ''), ut_memo
  from hp_uxtable_list
  into :res_code, :res_name, :res_memo
  do
  begin
    res_image = 21; -- Таблица
    ux_root = ux_root_main;

    for select res_table_name
    from hp_uxtable_link(:res_code)
    into :tmp_table
    do
    begin
      if (exists(select * from temp_uxtable
      where table_name = :tmp_table)) then
      begin
        ux_root = ux_root_ui;
        break;
      end
    end

    res_id = ux_root || '.' || res_code;
    res_parent_id = ux_root;

    tmp_str = '';
    select res_fk_table
    from hp_uxtable_link(:res_code)
    where res_level = 0
    into tmp_str;

    if (tmp_str <> '') then
      res_parent_id = ux_root || '.' || tmp_str;

    if (exists(select * from uxcode
               where ux_table_name = :res_code and
                     ux_kind = uxf$kind('uxform', 'id'))) then
      res_image = 38; -- Форма

    if (res_name <> '' and res_name <> res_code)
    then res_name = f_format('%1 (%2)', res_code, res_name);
    else res_name = res_code;

    res_id = lower(res_id);
    res_parent_id = lower(res_parent_id);

    res_table_name = res_code;
    suspend;
  end

  res_kind_caption = '';
  res_table_name = 'enum_kind';

  res_image = 70;
  res_kind = 'enum.main';
  res_parent_id = null;
  res_id = 'enum';
  res_code = '';
  res_memo = 'Список перечислений';
  res_name = 'Перечисления';
  ux_root = res_id;
  suspend;

  res_parent_id = ux_root;

  res_table_name = 'enum_list';
  res_image = 71; -- Enum
  res_kind = 'enum';
  res_kind_caption = 'Перечисления';

  for select 'enum.' || ek_code, ek_code, ek_code, ek_name
  from enum_kind
  into res_id, res_code, res_name, res_memo
  do
    suspend;

end
