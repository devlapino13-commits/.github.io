CREATE PROCEDURE FSROBOT_MAILING_SMS_REPAY_ADV(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable C_NAME_FIO varchar(100);
declare variable OL_DATE date;
declare variable C_CELL_NO varchar(100);
declare variable C_ID decimal(18,0);
declare variable KOL integer;
declare variable DATE_FORMAT_STR varchar(20);
declare variable K_ID decimal(18,0);
declare variable AMOUNT decimal(18,4) = 0;
declare variable UNIQ_ACC_STR varchar(10);
declare variable DOC_NO varchar(20);
declare variable AMOUNT_STR varchar(20);
declare variable REG_ID integer;
declare variable CM_DATABASE_ID integer;
declare variable AMOUNT_SYM varchar(10);
declare variable KS_STATE integer;
declare variable OKS_DELINQ_DATE_STR varchar(20);
declare variable KONT_LAST_DATE date;
declare variable OKS_DELINQ_DATE date;
declare variable RES_BODY_1 blob sub_type 1 segment size 80;
declare variable RES_BODY_2 blob sub_type 1 segment size 80;
declare variable REPAY_DATE_STR varchar(10);
declare variable CELL_ALL varchar(255);
declare variable LOGIC_TAG smallint;
declare variable OKS_DELINQ_DATE_TMP date;
declare variable K_KONTRACT_NO varchar(20);
declare variable K_DISB_DATE date;
declare variable ZAD_PERC decimal(15,2);
declare variable K_NSP_PERCENT decimal(15,2);
declare variable ZAD_TAX decimal(15,2);
declare variable K_REFUND smallint;
begin
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID - пользователь

-- СМС после погашения кредита
  if (user_fio is null) then
    user_fio = '';
  res_maillist = :rep_email;

  :cm_database_id = f_get_db_id();
  amount_sym = (case
                  when (cm_database_id = 81001) then 'p.'
                  when (cm_database_id in (39801, 41701)) then 'c.'
                  when (cm_database_id = 94401) then 'AZN'
                end);

  res_body = '';
  kol = 0;
  for select k_id, k_kontract_no, k_kontract_date
      from kontract kont
      where k_id = :rep_params
      into :k_id, k_kontract_no, k_disb_date
  do
  begin
    date_format_str = right('0' || extract(day from k_disb_date), 2) || '.' || right('0' || extract(month from k_disb_date), 2) || '.' || extract(year from k_disb_date);

-- СМС о досрочно-частичном погашении
    res_body = '';
    res_subject = 'Досрочно-частичное погашение';

-- Второй СМС, о состоянии
    ks_state = null;
    select ks_state
    from kontstate
    where ks_from_k_id = :k_id and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999'
    into :ks_state;

    if (ks_state = 1) then
    begin
-- Кредит все еще активный, показываем состояние
      select oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :ol_date and
            oks_date_end >= :ol_date and
            oks_close_state = 0
      into :oks_delinq_date;

      logic_tag = 0;

      if (cm_database_id = 81001) then
      begin
        if (reg_id in (1090100, --0901    Нижний Новгород
                       1260100, --2601    Калининград
                       1240100, --2401    Калуга
                       1220100, --2201    Мурманск
                       1420100, --4201    Ярославль
                       1430100, --4301    Липецк
                       1440100  --4401    Тверь
                       )) then
        begin
          logic_tag = 1;
        end
        else
        if (reg_id in (1031300, --0313    Красный текстильщик (Санкт-Петербург)
                       1031500  --0315    Спасский (Санкт-Петербург))
                       )) then
        begin
          logic_tag = 2;
        end
      end

      if (logic_tag = 0) then
      begin
        oks_delinq_date_str = right('0' || extract(day from oks_delinq_date), 2) || '.' || right('0' || extract(month from oks_delinq_date), 2) || '.' || extract(year from oks_delinq_date);
      end
      else
      begin
        oks_delinq_date_tmp = dateadd(-5 day to :oks_delinq_date);

        if (logic_tag = 1) then
        begin
-- если рекомендованный платеж выпадает на выходной день, тогда указываю дату ближайшего рабочего дня
          if ((select sc.scd_work_hours
               from sa_calendays sc
               where sc.scd_date = :oks_delinq_date_tmp) = 0) then
          begin
            select sc.scd_date
            from sa_calendays sc
            where sc.scd_date > :oks_delinq_date_tmp and
                  sc.scd_work_hours > 0
            order by sc.scd_date
            rows 1
            into :oks_delinq_date_tmp;
          end
        end
-- дата рекомендованного платежа не может быть позже даты, следующего платежа по графику
        if (oks_delinq_date_tmp > oks_delinq_date) then
          oks_delinq_date_tmp = oks_delinq_date;

        oks_delinq_date_str = right('0' || extract(day from oks_delinq_date_tmp), 2) || '.' || right('0' || extract(month from oks_delinq_date_tmp), 2) || '.' || extract(year from oks_delinq_date_tmp);
      end

      if (oks_delinq_date > ol_date) then
      begin
        if (cm_database_id in (39801, 41701)) then
        begin
          if (logic_tag = 0) then
            res_body_2 = 'Spasibo! Dolga net. Sleduyushiy platej: ' || oks_delinq_date_str;
          else
            res_body_2 = 'Spasibo! Rekomend. data sled. plateja: ' || oks_delinq_date_str;
        end
        else
        if (cm_database_id = 81001) then
        begin
          if (logic_tag = 0) then
            res_body_2 = 'Spasibo! Dolga net. Sleduyshee pogashenie: ' || oks_delinq_date_str;
          else
            res_body_2 = 'Spasibo! Rekomend. data sled. pogashenija: ' || oks_delinq_date_str;
        end
        else
        if (cm_database_id = 94401) then
        begin
          res_body_2 = 'Sagolun! Novbeti odenish tarixiniz: ' || oks_delinq_date_str;
        end
      end
      else
      begin

        amount = 0;
        if (kont_last_date > ol_date) then
        begin
          select sum(zad_os + zad_perc + sh_calc_fine), sum(zad_perc)
          from aarep_make_zad_list_temp(:k_id, :ol_date, 0)
          where sh_date <= :ol_date
          into :amount, zad_perc;
        end
        else
        begin
          select sum(zad_os + sh_calc_perc + sh_calc_fine), sum(sh_calc_perc)
          from aarep_make_zad_list_temp(:k_id, :ol_date, 1)
          where sh_date <= :ol_date
          into :amount, zad_perc;
        end
        if (amount is null) then
          amount = 0;

        if (cm_database_id in (39801, 41701) and
            zad_perc > 0 and
            amount > 0) then
        begin
          select nsd_percent
          from nspconfig_data
          where nsd_from_nsm_id = -1 -- нсп от процента
                and
                :ol_date between nsd_date_begin and nsd_date_end
          into :k_nsp_percent;

          select res
          from fhp_round_bank(:zad_perc * :k_nsp_percent / 100, 0)
          into zad_tax;

          amount = amount + coalesce(zad_tax, 0);
        end

        if (amount > 0) then
        begin
          amount_str = round(amount, 0);
          if (char_length(amount_str) > 6) then
            amount_str = left(amount_str, char_length(amount_str) - 6) || ',' || left(right(amount_str, 6), 3) || ',' || right(amount_str, 3);
          else
          if (char_length(amount_str) > 3) then
            amount_str = left(amount_str, char_length(amount_str) - 3) || ',' || right(amount_str, 3);
          amount_str = amount_str || amount_sym;

          if (cm_database_id <> 94401) then
          begin
            res_body_2 = 'Oplatite tekushiy dolg. Na ' || repay_date_str || ' summa: ' || amount_str;
          end
          else
          begin
            res_body_2 = 'Gecikmede olan qaliq borc: ' || amount_str;
          end
        end
        else
        begin
          if (cm_database_id <> 94401) then
          begin
            res_body_2 = 'Spasibo! Dolga net.';
          end
          else
          begin
            res_body_2 = 'Sagolun!';
          end
        end
      end
    end

    if (ks_state = 2) then
    begin
-- Кредит закрыт
      if (cm_database_id in (39801, 41701)) then
      begin
        res_body_2 = 'Dolg pogashen polnostyu. Spasibo za sotrudnichestvo! ' || repay_date_str;
      end
      else
      if (cm_database_id = 81001) then
      begin
        res_body_2 = 'Zaym pogashen polnostyu. Spasibo za sotrudnichestvo! ' || repay_date_str;
      end
      else
      begin
        res_body_2 = 'MBulaka olan borcunuz tam odenilmishdir.' || repay_date_str;
      end
    end

    cell_all = '';
    for select c_id,
               (select cont_value
                from hp_contdata_get_cell_main(0, c_id)), c_name_fio
        from knclient, clients
        where kc_amount_approved > 0 and
              kc_from_k_id = :k_id and
              kc_from_c_id = c_id
        into :c_id, :c_cell_no, c_name_fio
    do
    begin
      k_refund = 0;

      if (cm_database_id in (39801, 41701)) then
      begin
          -- если на момент операции у клиента есть заявка по рефинансирвоанию k_refund = 1
          if (exists (select *
                      from miniapp maa
                      inner join miniappclient macl on macl.mac_from_ma_id = maa.ma_id
                      where macl.mac_from_c_id = :c_id
                      and macl.mac_amount > 0
                      and maa.ma_is_active = 1
                      and maa.ma_date_end = '31.12.9999'
                      and maa.ma_refund = 1
                      and maa.ma_date_oper >= :ol_date
          )) then
            k_refund = 1;
      end

      res_c_id = c_id;
      if ((right(coalesce(c_cell_no, '000000'), 6) <> '000000') and
          (position(c_cell_no, cell_all) <= 0) and
                    (not ((cm_database_id in (39801, 41701)) and (k_refund = 1) and (ks_state = 2)))) then -- igusev 26.04.2020 добавил условие, на основании письма от Бахтияра "Subject: Сервисный смс" Sent: Saturday, April 25, 2020 12:03 PM
      begin
        cell_all = cell_all || c_cell_no || ';';
        res_maillist = :c_cell_no;

        uniq_acc_str = (select uniq_acc_str
                        from hp_show_clientuniq_acc_no(:c_id));

        res_subject = 'Погашение кредита, СМС';
        res_body = f_format('%1:%2; %3; %4', iif(cm_database_id <> 94401, 'L/chet', 'Hesab N'), uniq_acc_str, res_body_1, res_body_2);
        suspend;
      end
    end
  end
end
