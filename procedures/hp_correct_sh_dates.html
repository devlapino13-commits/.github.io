CREATE PROCEDURE HP_CORRECT_SH_DATES(
    K_ID DECIMAL(18,0),
    WITH_CORRECT INTEGER)
RETURNS (
    SH_DATE DATE,
    PR_IS_360 INTEGER,
    PR_PERCENT_FOR_DAYS INTEGER,
    K_PERCENT_PER_DAY DECIMAL(18,7),
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    RES_KOL_DAYS INTEGER,
    RES_CALC_PERC DECIMAL(18,4),
    PERC_TEMP DECIMAL(18,4),
    RES_DIFF_PERC DECIMAL(18,4),
    KK_ID DECIMAL(18,0),
    SH_DIFF_DATE INTEGER,
    SH_ID INTEGER,
    SH_PERC_NEXT DECIMAL(18,4),
    BALANCE_NEW DECIMAL(18,4),
    DIFF_DAY DECIMAL(18,4))
AS
declare variable dd_prior date;
declare variable dd date;
declare variable kk_close_date date;
declare variable n decimal(18,4);
begin
  for select distinct k_id, max(sh_date)
      from kontract, kontstate, schedule
      where ks_from_k_id = k_id and
            ks_state = 1 and
            sh_from_k_id = k_id and
            ((k_id = :k_id) or
            (:k_id = 0))
      group by k_id
      order by k_id
      into :kk_id, :kk_close_date
  do
  begin
    sh_perc_next = 0;
    sh_diff_date = 0;
    res_calc_perc = 0;
    res_kol_days = 0;
    balance_new = 0;
    dd_prior = null;
    perc_temp = 0;
    for select sh_date, pr_is_360, pr_percent_for_days, k_percent_per_day, sh_diff_date, sh_id, sum(sh_os),
               sum(sh_perc)
        from hp_correct_sh_dates_temp(:kk_id)
        group by sh_date, pr_is_360, pr_percent_for_days, k_percent_per_day, sh_id, sh_diff_date
        order by sh_date
        into :sh_date, :pr_is_360, :pr_percent_for_days, :k_percent_per_day, :sh_diff_date, :sh_id, :sh_os, :sh_perc
    do
    begin
      if (dd_prior is not null) then
      begin
        if (sh_diff_date = 0) then
        begin
          execute procedure fhp_agein360(:dd_prior, :sh_date, :pr_is_360)
              returning_values :res_kol_days;
          perc_temp = :perc_temp + balance_new * res_kol_days * k_percent_per_day / 100;
          perc_temp = :perc_temp + :sh_perc_next;
          sh_perc_next = 0;
        end
        else
        begin
          dd = :sh_date - :sh_diff_date;
          if (not((sh_diff_date > 0) and
              (dd = dd_prior))) then
          begin
            perc_temp = :perc_temp + :sh_perc_next;
            sh_perc_next = 0;
          end
          if (sh_diff_date > 0) then
          begin
            execute procedure fhp_agein360(:dd, :sh_date, :pr_is_360)
                returning_values :res_kol_days;
            sh_perc_next = :sh_perc_next + cast(:balance_new * :res_kol_days * :k_percent_per_day / 100 as decimal(15,0));

            execute procedure fhp_agein360(:dd_prior, :dd, :pr_is_360)
                returning_values :res_kol_days;
            if (res_kol_days > 0) then
              perc_temp = :perc_temp + balance_new * res_kol_days * k_percent_per_day / 100;
          end
          else
          begin
            execute procedure fhp_agein360(:dd_prior, :sh_date, :pr_is_360)
                returning_values :res_kol_days;
            if (res_kol_days > 0) then
              perc_temp = :perc_temp + balance_new * res_kol_days * k_percent_per_day / 100;

            execute procedure fhp_agein360(:sh_date, :dd, :pr_is_360)
                returning_values :res_kol_days;
            if (res_kol_days > 0) then
              perc_temp = :perc_temp + (:balance_new + :sh_os) * res_kol_days * k_percent_per_day / 100;
          end
        end
      end
      if (sh_perc <> 0) then
      begin
        if (sh_date = kk_close_date) then
        begin
          perc_temp = :perc_temp + :sh_perc_next;
        end
        res_calc_perc = cast(:perc_temp as decimal(15,0));
        perc_temp = 0;
        res_diff_perc = :res_calc_perc - :sh_perc;
        if ((res_diff_perc >= -1) and
            (res_diff_perc <= 1)) then
        begin
          res_diff_perc = 0;
          res_calc_perc = :sh_perc;
        end
      end
      else
      begin
        res_calc_perc = 0;
        res_diff_perc = 0;
      end
      diff_day = 0;
      if ((res_diff_perc < 0) and
          (with_correct = 1)) then
      begin
        if ((balance_new <> 0) and
            (k_percent_per_day <> 0)) then
        begin
          n = cast(((:balance_new + :sh_os) * :k_percent_per_day / 100) as decimal(15,0));
          if (n <> 0) then
            diff_day = cast((res_diff_perc / n) as decimal(15,0));
        end
        if (diff_day <> 0) then
        begin
          if (not((extract(day from :sh_date) >= 27) and
              (extract(month from :sh_date) = 2))) then
          begin
            update schedule
            set sh_diff_date = cast(:diff_day as integer)
            where sh_from_k_id = :kk_id and
                  sh_id = :sh_id and
                  sh_diff_date = 0;
          end
          else
          begin
            if ((extract(day from :sh_date) = 27) and
                (extract(month from :sh_date) = 2)) then
            begin
              update schedule
              set sh_diff_date = -1
              where sh_from_k_id = :kk_id and
                    sh_id = :sh_id and
                    sh_diff_date = 0;
            end
          end
        end
      end
      else
        diff_day = 0;
      suspend;
      if (sh_diff_date >= 0) then
        dd_prior = :sh_date;
      else
        dd_prior = :sh_date - :sh_diff_date;
      balance_new = balance_new + sh_os;
    end
  end
end
