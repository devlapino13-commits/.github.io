CREATE PROCEDURE HP_REGISTRATION_CLIENT_TO_MPC(
    CAC_ID D_INTEGER NOT NULL,
    SHOW_ADD_INFO SMALLINT = 0)
AS
declare variable c_id d_id;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable csc_id d_integer;
declare variable new_account d_str_2048;
declare variable new_agreement d_str_2048;
declare variable new_customer d_str_2048;
declare variable cs_request_json d_str_8100;
declare variable cac_from_reg_id d_integer;
declare variable un_id d_id;
declare variable k_id d_id;
begin
  for
    select
      cac_from_c_id,
      cac_card_id,
      cac_from_cca_id,
      cac_from_reg_id,
      ck.cac_from_un_id
    from cardaktclient  ck
    where cac_id = :cac_id
    into
      :c_id,
      :card_id,
      :cca_id,
      :cac_from_reg_id,
      :un_id
  do
  begin
    -- получаю утвержденный займ клиента по продукту связанному с картами "М Булак"
    select kont.k_id
    from kontract kont
    inner join knclient kncl on kont.k_id = kncl.kc_from_k_id
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    where kncl.kc_from_c_id = :c_id
          and kncl.kc_amount_approved > 0
          and pr.pr_mbulak_cards_kind > 0 -- продукт по картам "М Булак"
          and kont.k_status = 1
          and kont.k_fact_disb_ol_id is null
    rows 1
    into :k_id;

    select csc.csc_id
    from cardsynccfg csc
    where csc.csc_code = 'registr_client'
    into :csc_id;

    select new_account
    from cr_json_new_account(:c_id, :show_add_info)
    into :new_account;

    select new_agreement
    from cr_json_new_agreement(:c_id, :show_add_info)
    into :new_agreement;

    select new_customer
    from cr_json_new_customer(:c_id, :show_add_info)
    into :new_customer;

    cs_request_json = '{' ||
        '"account": ' || :new_account || ', ' ||
        '"agreement": ' || :new_agreement || ', ' ||
        '"customer": ' || :new_customer ||
        '}';

    insert into cardsync (cs_from_csc_id, cs_card_id, cs_from_cca_id, cs_object_kind, cs_object_id, cs_request_json, cs_office_id, cs_from_un_id, cs_from_c_id, cs_from_k_id)
    values (:csc_id, :card_id, :cca_id, f_objkind('cardaktclient', 'id'), :cac_id, :cs_request_json, :cac_from_reg_id, :un_id, :c_id, :k_id);
  end
end
