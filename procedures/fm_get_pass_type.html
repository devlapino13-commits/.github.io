CREATE PROCEDURE FM_GET_PASS_TYPE(
    COUNTRY_ID INTEGER)
RETURNS (
    PK_ID INTEGER,
    PK_NAME VARCHAR(100),
    PK_IS_LATIN INTEGER)
AS
declare variable type_count integer;
begin
  select count(*)
  from passkind, passdoctype
  where pk_country_id = :country_id and
        pk_id > 0 and
        pk_from_pdt_id = pdt_id and
        pk_is_delete = 0 and
        pdt_is_main_doc = 1
  into type_count;
  if (type_count > 1) then
  begin
    pk_id = -1;
    pk_name = 'Выберите тип...';
    pk_is_latin = 0;
    suspend;
  end

  for select pk_id, pk_name, pk_is_latin
      from passkind, passdoctype
      where pk_country_id = :country_id and
            pk_id > 0 and
            pk_from_pdt_id = pdt_id and
            pk_is_delete = 0 and
            pdt_is_main_doc = 1
      order by pk_order, pk_id
      into pk_id, pk_name, pk_is_latin
  do
  begin
    suspend;
  end
end
