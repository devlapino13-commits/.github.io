CREATE PROCEDURE AAREP_DELINQ_DISB_MONTHLY(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_DELINQ_DAY INTEGER)
RETURNS (
    DISB_PERIOD INTEGER,
    PRODUCT_KIND VARCHAR(100),
    MONTH_ID INTEGER,
    DEL_BALANCE DECIMAL(18,4),
    DEL_COUNT INTEGER)
AS
begin
  for select disb_period, product_kind, month_id, sum(del_balance), sum(del_count)
      from aarep_delinq_disb_monthly_tmp(:rep_date_begin, :rep_date_end, :rep_delinq_day)
      group by disb_period, product_kind, month_id
      into :disb_period, :product_kind, :month_id, :del_balance, :del_count
  do
  begin
    suspend;
  end
end
