CREATE PROCEDURE HP_ROBOT_INS_FACERECOG
AS
declare variable c_id d_id;
declare variable photo_id d_id;
declare variable c_photo_id d_id;
declare variable photo_exists d_smallint;
begin
  for select ps.c_id,
             ps.photo_id
  from photo_save_temp ps
  into :c_id,
       :photo_id
  do
  begin
    photo_exists = 0;

    -- проверка фото из временной таблицы
    if (:photo_id > 0) then
    begin
      select 1
      from hp_get_blobex_blob(:photo_id)
      rows 1
      into :photo_exists;
    end

    -- если не найдено — проверяем фото клиента
    if (:photo_exists = 0) then
    begin
      select c.c_blob_photo_id
      from clients c
      where c.c_id = :c_id
      into :c_photo_id;

      if (:c_photo_id > 0) then
      begin
        select 1
        from hp_get_blobex_blob(:c_photo_id)
        rows 1
        into :photo_exists;
      end
    end

    -- единая точка обработки
    if (:photo_exists = 1) then
    begin
      execute procedure hp_ins_facerecoglist(:c_id);

      delete from photo_save_temp
      where c_id = :c_id;
    end
  end
end
