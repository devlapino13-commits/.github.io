CREATE PROCEDURE AAREP_PAR_WEEKS_6(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(50),
    WEEK_1_PAR7 CURR_TYPE,
    WEEK_2_PAR7 CURR_TYPE,
    WEEK_3_PAR7 CURR_TYPE,
    WEEK_4_PAR7 CURR_TYPE,
    WEEK_5_PAR7 CURR_TYPE,
    WEEK_6_PAR7 CURR_TYPE,
    WEEK_1_PAR30 CURR_TYPE,
    WEEK_2_PAR30 CURR_TYPE,
    WEEK_3_PAR30 CURR_TYPE,
    WEEK_4_PAR30 CURR_TYPE,
    WEEK_5_PAR30 CURR_TYPE,
    WEEK_6_PAR30 CURR_TYPE,
    WEEK_1_BALANCE DECIMAL(18,4),
    WEEK_2_BALANCE DECIMAL(18,4),
    WEEK_3_BALANCE DECIMAL(18,4),
    WEEK_4_BALANCE DECIMAL(18,4),
    WEEK_5_BALANCE DECIMAL(18,4),
    WEEK_6_BALANCE DECIMAL(18,4))
AS
declare variable reg_unicode integer;
declare variable i smallint;
declare variable rep_date date;
declare variable rep_balance curr_type;
declare variable delinq_balance_7 curr_type;
declare variable delinq_balance_30 curr_type;
declare variable par7 curr_type;
declare variable par30 curr_type;
declare variable can_suspend integer;
begin
  for select reg_unicode, reg_name
      from region, region_list(:rep_region_id, 1, 3, 1)
      where reg_unicode = id_list
      order by reg_name
      into reg_unicode, office_name
  do
  begin
    can_suspend = 0;
    i = 0;
    while (i < 6) do
    begin
      i = i + 1;
      rep_date = rep_date_end - (6 - i) * 7;
      rep_balance = null;
      delinq_balance_7 = null;
      delinq_balance_30 = null;
      select sum(oks_balance), sum(iif(:rep_date - 7 > oks_delinq_date, oks_balance, 0)),
             sum(iif(:rep_date - 30 > oks_delinq_date, oks_balance, 0))
      from oksummary
      where oks_region_id = :reg_unicode and
            oks_close_state = 0 and
            :rep_date between oks_date_begin and oks_date_end and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date, null, null, null, null, null, null)) in (1, 2)
      into rep_balance, delinq_balance_7, delinq_balance_30;
      if (rep_balance is null) then
        rep_balance = 0;
      if (delinq_balance_7 is null) then
        delinq_balance_7 = 0;
      if (delinq_balance_30 is null) then
        delinq_balance_30 = 0;
      if (rep_balance > 0) then
        can_suspend = 1;
      if (rep_balance > 0) then
      begin
        par7 = delinq_balance_7 / rep_balance;
        par30 = delinq_balance_30 / rep_balance;
      end
      else
      begin
        par7 = 0;
        par30 = 0;
      end
      if (i = 1) then
      begin
        week_1_par7 = par7;
        week_1_par30 = par30;
        week_1_balance = rep_balance;
      end
      if (i = 2) then
      begin
        week_2_par7 = par7;
        week_2_par30 = par30;
        week_2_balance = rep_balance;
      end
      if (i = 3) then
      begin
        week_3_par7 = par7;
        week_3_par30 = par30;
        week_3_balance = rep_balance;
      end
      if (i = 4) then
      begin
        week_4_par7 = par7;
        week_4_par30 = par30;
        week_4_balance = rep_balance;
      end
      if (i = 5) then
      begin
        week_5_par7 = par7;
        week_5_par30 = par30;
        week_5_balance = rep_balance;
      end
      if (i = 6) then
      begin
        week_6_par7 = par7;
        week_6_par30 = par30;
        week_6_balance = rep_balance;
      end

    end
    if (can_suspend = 1) then
      suspend;
  end
end
