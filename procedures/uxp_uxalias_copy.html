CREATE PROCEDURE UXP$UXALIAS_COPY(
    UXA_OWNER D_UUID,
    UXA_KIND D_SMALLINT,
    UXA_CODE D_STR_CODE,
    UXA_NAME D_STR_CODE,
    INCLUDE_ACTIONS D_BOOL,
    ORIG_UXA_UUID D_UUID)
RETURNS (
    UXA_UUID D_UUID)
AS
declare variable uu_is_base d_bool;
declare variable ux_uuid d_uuid;
begin
  select uu_is_base
  from ux$uniq
  where uu_id = :uxa_owner
  into uu_is_base;

  if (uu_is_base = 1) then
    ux_uuid = uxa_owner;
  else
  begin
    select ufm_ux_uuid
    from ux$form
    where ufm_uuid = :uxa_owner
    into ux_uuid;

    if (ux_uuid is null) then
      exception e_default f_format('Unknown UXA_OWNER uuid: %1', uuid_to_char(uxa_owner));
  end

  select ux_uuid
  from uxp$uxalias_insert(:ux_uuid, :uxa_kind, :uxa_code, :uxa_name)
  into :uxa_uuid;

  if (include_actions = 1) then
  begin
    insert into ux$aliasaction (uaa_uxa_uuid, uaa_ua_uuid, uaa_is_default)
    select :uxa_uuid, uaa_ua_uuid, uaa_is_default
    from ux$aliasaction
    where uaa_uxa_uuid = :orig_uxa_uuid;
  end

  suspend;
end
