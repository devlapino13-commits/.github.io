CREATE PROCEDURE HP_EXTINFO_GET_BLANK_INFO(
    EIC_ID D_ID,
    BLANK_DATE D_DATE)
RETURNS (
    JSON_MODEL D_BLOB_TEXT,
    RES_TEXT D_STR_255,
    BLN_ID INTEGER)
AS
begin
  res_text = '';

  select eib_from_bln_id
  from extinfoblank
  where eib_from_eic_id = :eic_id
  rows 1
  into bln_id;

  if (bln_id is null) then
    res_text = 'Не настроен бланк для просмотра';

  if (res_text = '') then
  begin
    select blb_json_blob
    from blankblob
    where blb_from_bln_id = :bln_id and
          blb_date_begin <= :blank_date and
          blb_date_end >= :blank_date
    into json_model;

    if (json_model is null) then
      res_text = 'Не настроен шаблон бланка № '|| bln_id|| ' на дату: ' || f_datetostr(blank_date);
  end
  bln_id = coalesce(bln_id, 0);

  suspend;
end
