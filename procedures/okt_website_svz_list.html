CREATE PROCEDURE OKT_WEBSITE_SVZ_LIST(
    REP_REGION_ID INTEGER)
RETURNS (
    LO_REG_NAME STR_100,
    LO_C_ID ID_TYPE,
    LO_FIO_PROBLEM STR_100,
    LO_TEL_NO VARCHAR(20),
    LO_PROBLEM VARCHAR(20),
    LO_LANG VARCHAR(50),
    LO_LANG_RU SMALLINT,
    LO_LANG_UZ SMALLINT,
    LO_LANG_TJ SMALLINT,
    LO_LANG_KG SMALLINT,
    SEND_QTY INTEGER)
AS
declare variable lo_reg_id id_type;
declare variable reg_id_tmp id_type;
declare variable lo_lang_tmp varchar(255);
declare variable lo_mark_speak smallint;
declare variable lo_is_native smallint;
begin
  send_qty = 0;

  for select ps_from_reg_id, ps_from_c_id
      from hr_pstatus hps
      inner join joblist jl on jl.jl_id = ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      inner join region_list(:rep_region_id, 1, -1, 1) on id_list = ps_from_reg_id
      where current_date between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            jk.jk_system_code = 'JobSystem.LO_Standart'
--and ps_from_c_id = 1010100220900
      order by 1
      into lo_reg_id, lo_c_id
  do
  begin
    if (lo_c_id not in (1010100001577, 1010100313141, 1010100059335, 1010100342995, 1010100083879, 1010100102705, 1010100080070, 1010100134297, 1010100153454, 1010100219563,

                        1010100471481, 1010100511138, 1010100298147,

                        1010100149843, 1010100150776, 1010100470964, 1010100455226, 1010100542476, 1010100341359, 1010100312741,

                        )) then
    begin
      if (not exists(select *
                     from svz_par_problem spp
                     where spp.spp_is_del = 0 and
                           spp.svz_id = :lo_c_id)) then
      begin
        if (lo_reg_id <> coalesce(reg_id_tmp, 0)) then
        begin
          select reg_name
          from region
          where reg_unicode = :lo_reg_id
          into lo_reg_name;

          reg_id_tmp = lo_reg_id;
        end

        lo_lang = 'Не указан';
        lo_mark_speak = 0;
        lo_is_native = null;
        lo_lang_ru = lo_mark_speak;
        lo_lang_uz = lo_mark_speak;
        lo_lang_tj = lo_mark_speak;
        lo_lang_kg = lo_mark_speak;

        for select rd.rd_name, cl.cl_mark_speak, cl.cl_is_native
            from clanguage cl, refdata rd
            where cl.cl_from_c_id = :lo_c_id and
                  cl.cl_lang_no = rd_id and
                  rd_main_code = 'Language'
            into lo_lang_tmp, lo_mark_speak, lo_is_native
        do
        begin
          if (lo_is_native = 1) then
            lo_lang = lo_lang_tmp;

          if (lo_lang_tmp = 'Русский') then
            lo_lang_ru = lo_mark_speak;
          if (lo_lang_tmp = 'Узбекский') then
            lo_lang_uz = lo_mark_speak;
          if (lo_lang_tmp = 'Таджикский') then
            lo_lang_tj = lo_mark_speak;
          if (lo_lang_tmp = 'Кыргызский') then
            lo_lang_kg = lo_mark_speak;

        end

        lo_tel_no = '';
        select cont_value
        from hp_contdata_get_cell_main(0, :lo_c_id)
        where cont_value is not null
        into lo_tel_no;

        if (lo_tel_no = '') then
          lo_tel_no = 'Не указан';

        lo_problem = '';
        lo_fio_problem = '';
        if (lo_tel_no = 'Не указан') then
          lo_problem = 'Телефон не указан';

        if (lo_problem <> '') then
          lo_fio_problem = (select c_name_fio
                            from clients cl
                            where cl.c_id = :lo_c_id);

        select wls.wls_counter
        from cc_website_lo_sms wls
        where wls.wls_lo_id = :lo_c_id
        order by wls_lo_id desc, wls.wls_counter desc
        rows 1
        into :send_qty;

        send_qty = coalesce(send_qty, 0);

        suspend;
      end
    end
  end
