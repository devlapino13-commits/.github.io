CREATE PROCEDURE HP_CORRECT_K_LAST_REG_ID
AS
declare variable kf_id decimal(18,0);
declare variable reg_id integer;
declare variable d1 date;
declare variable d2 date;
declare variable k_id decimal(18,0);
begin
  for select distinct min(kal_date), max(kal_date_end), kal_from_kf_id, kal_new_reg_id
      from knaktlist
      where is_delete = 0
      group by kal_from_kf_id, kal_new_reg_id
      into :d1, :d2, :kf_id, :reg_id
  do
  begin
    for select k_id
        from kontract, kontstate
        where ks_state = 2 and
              ks_date_begin between :d1 and :d2 and
              ks_ol_is_delete = 0 and
              ks_from_k_id = k_id and
              k_from_kf_id = :kf_id
        into :k_id
    do
    begin
      update kontract
      set k_last_reg_id = :reg_id
      where k_id = :k_id and
            (k_last_reg_id <> :reg_id or
            k_last_reg_id is null);
    end
    if (d2 = '31.12.9999') then
    begin
      for select k_id
          from kontract, kontstate
          where ks_state <= 1 and
                ks_date_end >= '01.01.2999' and
                ks_date_begin < ks_date_end and
                ks_ol_is_delete = 0 and
                ks_from_k_id = k_id and
                k_from_kf_id = :kf_id
          into :k_id
      do
      begin
        update kontract
        set k_last_reg_id = :reg_id
        where k_id = :k_id and
              (k_last_reg_id <> :reg_id or
              k_last_reg_id is null);
      end
    end
  end
end
