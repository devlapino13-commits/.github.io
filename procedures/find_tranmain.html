CREATE PROCEDURE FIND_TRANMAIN(
    REGION_ID INTEGER,
    DATE_BEGIN DATE,
    DATE_END DATE,
    WITH_DATE INTEGER,
    FIND_TEXT VARCHAR(255),
    FIND_FIO_TEXT VARCHAR(255))
RETURNS (
    TM_ID INTEGER,
    TM_DATE_BEGIN DATE,
    TM_DATE_END DATE,
    TM_SUBJECT VARCHAR(255),
    TM_CATEGORY VARCHAR(255),
    TM_PLACE VARCHAR(255),
    TM_TARGET VARCHAR(255),
    TM_MEMO D_STR_2048,
    TM_COMPANY VARCHAR(255),
    TM_REGION_NAME VARCHAR(100))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */
  if (find_text <> '') then
    find_text = upper(:find_text);
  if (find_fio_text <> '') then
    find_fio_text = upper(:find_fio_text);
  if (find_fio_text = '') then
  begin
    for select tm_id, tm_date_begin, tm_date_end, tm_subject, tm_category, tm_place, tm_target, tm_memo, tm_company,
               reg_name
        from tranmain, region, region_list(:region_id, 1, -1, 1)
        where tm_from_reg_id = reg_unicode and
              (tm_from_reg_id = id_list) and
              (upper(tm_subject) like '%' || :find_text || '%' or
              upper(tm_category) like '%' || :find_text || '%' or
              upper(tm_place) like '%' || :find_text || '%' or
              upper(tm_target) like '%' || :find_text || '%' or
              upper(tm_memo) like '%' || :find_text || '%' or
              upper(tm_company) like '%' || :find_text || '%') and
              ((:with_date = 0) or
              ((:with_date = 1) and
              ((tm_date_begin <= :date_end) and
              (tm_date_end >= :date_begin))))
        into :tm_id, :tm_date_begin, :tm_date_end, :tm_subject, :tm_category, :tm_place, :tm_target, :tm_memo,
             :tm_company, :tm_region_name
    do
    begin
      suspend;
    end
  end
  else
  begin /* Поиск с ФИО участников или тренеров */
    for select distinct tm_id, tm_date_begin, tm_date_end, tm_subject, tm_category, tm_place, tm_target, tm_memo,
                        tm_company, reg_name
        from tranmain, region, region_list(:region_id, 1, -1, 1), clients, tranmembers
        where tm_from_reg_id = reg_unicode and
              (tm_from_reg_id = id_list) and
              tb_from_tm_id = tm_id and
              tb_from_c_id = c_id and
              upper(c_name_fio) like '%' || :find_fio_text || '%' and
              (upper(tm_subject) like '%' || :find_text || '%' or
              upper(tm_category) like '%' || :find_text || '%' or
              upper(tm_place) like '%' || :find_text || '%' or
              upper(tm_target) like '%' || :find_text || '%' or
              upper(tm_memo) like '%' || :find_text || '%' or
              upper(tm_company) like '%' || :find_text || '%') and
              ((:with_date = 0) or
              ((:with_date = 1) and
              ((tm_date_begin <= :date_end) and
              (tm_date_end >= :date_begin))))
        into :tm_id, :tm_date_begin, :tm_date_end, :tm_subject, :tm_category, :tm_place, :tm_target, :tm_memo,
             :tm_company, :tm_region_name
    do
    begin
      suspend;
    end
  end
end
