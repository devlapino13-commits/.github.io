AS
declare variable pr_is_secured integer;
declare variable pr_int_calc_type integer;
declare variable k_is_kredline integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable pr_id_prior id_type;
declare variable kf_region_id_prior integer;
declare variable kf_officer_id_prior id_type;
declare variable trans_refdata_id integer;
declare variable trans_refdata_id_prior integer;
declare variable woff_memo1 str_255;
declare variable db_id integer;
begin
  :db_id = f_get_db_id();

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  pr_id_prior = 0;
  kf_region_id_prior = 0;
  kf_officer_id_prior = 0;
  trans_refdata_id_prior = -9999;
  for select trans_kontract_id, trans_date, trans_doc_no, trans_region_id, trans_type, trans_type_text, trans_disb_os,
             trans_os, trans_perc, trans_fine, trans_kommis, trans_kommis_adv, trans_pay_type, trans_pay_type_text,
             trans_operlist_id, trans_pay_all, trans_other_client, trans_datetime, trans_other_day, trans_refdata_id,
             trans_plan_date
      from aarep_transkred_id(:report_region_id, :report_date_begin, :report_date_end, :report_currency)
      where trans_type = 5
      into :k_id, :trans_date, :trans_doc_no, :trans_region_id, :trans_type, :trans_type_text, :trans_disb_os,
           :trans_os, :trans_perc, :trans_fine, :trans_kommis, :trans_kommis_adv, :trans_pay_type, :trans_pay_type_text,
           :trans_operlist_id, :trans_pay_all, :trans_other_client, :trans_datetime, :trans_other_day,
           :trans_refdata_id, :trans_plan_date
  do
  begin
    if (:trans_type_text not starting 'Списание') then
      continue; --   Данные из таблицы KONTRACT
    select k_from_kf_id, k_currency, k_applicate_date, k_applicate_no, k_kontract_no, k_kontract_date, k_from_pr_id,
           k_disb_date, k_amount_applicate, k_stage, k_period, k_kolich_lgota, k_percent, k_percent_lgota,
           k_kolich_repay, k_from_f_id, k_from_pt_id, k_circle, k_circle_internal, k_percent_per_day, k_memo,
           k_amount_approved, k_percent_for_days, k_kolich_client_applicate, k_kolich_client_approved,
           k_repay_kolich_closed, k_class, k_kommis_perc, k_from_cs_id, k_credit_target, k_is_kredline, k_kredline_end,
           k_clients_leader_id, k_clients_cashier_id
    from kontract
    where k_id = :k_id
    into :k_from_kf_id, :k_currency, :k_applicate_date, :k_applicate_no, :k_kontract_no, :k_kontract_date,
         :k_from_pr_id, :k_disb_date, :k_amount_applicate, :k_stage, :k_period, :k_kolich_lgota, :k_percent,
         :k_percent_lgota, :k_kolich_repay, :k_from_f_id, :k_from_pt_id, :k_circle, :k_circle_internal,
         :k_percent_per_day, :k_memo, :k_amount_approved, :k_percent_for_days, :k_kolich_client_applicate,
         :k_kolich_client_approved, :k_repay_kolich_closed, :k_risk_category, :k_kommis_perc, :k_from_cs_id,
         :k_credit_target, :k_is_kredline, :k_kredline_end, :cl_leader_id, :cl_cashier_id;
    --    Данные из продукта, если продукт не повторется то будет вытаскивать данные
    if (:k_from_pr_id <> pr_id_prior) then
    begin
      pr_id_prior = :k_from_pr_id;
      select pr_code, pr_name, pr_loan_type, pr_is_secured, pr_interest_method, pr_percent_for_days
      from product
      where pr_id = :k_from_pr_id
      into :pr_code, :pr_name, :pr_loan_type, :pr_is_secured, :pr_int_calc_type, :pr_int_for_days;
    end
    --  Данные лидера
    cl_leader_name = null;
    cl_leader_pass_no = null;
    cl_leader_pass_date = null;
    cl_leader_pass_who = null;
    select c_name_fio, c_pass_no, c_pass_date, c_pass_who
    from clients
    where c_id = :cl_leader_id
    into cl_leader_name, cl_leader_pass_no, cl_leader_pass_date, cl_leader_pass_who;
    --  Адрес лидера
    cl_leader_full_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_address_type = 1 and
          adc_from_c_id = :cl_leader_id and
          adc_from_adl_id = adl_id
    into :cl_leader_full_address;
    --  Данные казначея
    cl_cashier_name = null;
    cl_cashier_pass_no = null;
    cl_cashier_pass_date = null;
    cl_cashier_pass_who = null;
    select c_name_fio, c_pass_no, c_pass_date, c_pass_who
    from clients
    where c_id = :cl_cashier_id
    into cl_cashier_name, cl_cashier_pass_no, cl_cashier_pass_date, cl_cashier_pass_who;
    --  Адрес казначея
    cl_cashier_full_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_address_type = 1 and
          adc_from_c_id = :cl_cashier_id and
          adc_from_adl_id = adl_id
    into :cl_cashier_full_address;
    --  Текущие данные из таблицы knaktlist
    kf_region_id = null;
    kf_officer_id = null;
    kf_name = null;
    kf_number = null;
    kf_officer_code = null;
    kf_date_in_to_krs = null;
    kf_date_out_from_krs = null;
    select kal_new_reg_id, kal_new_kf_officer, kal_new_kf_name, kal_new_no, kal_new_officer_code, kal_date,
           kal_date_end
    from knaktlist
    where kal_from_kf_id = :k_from_kf_id and
          knaktlist.is_delete = 0 and
          :trans_date between kal_date and kal_date_end
    into kf_region_id, kf_officer_id, :kf_name, :kf_number, :kf_officer_code, :kf_date_in_to_krs, :kf_date_out_from_krs;
    if (kf_region_id is null) then
      kf_region_id = -1;
    if (kf_officer_id is null) then
      kf_officer_id = -1;
    --  Параметры региона
    if (kf_region_id <> kf_region_id_prior) then
    begin
      kf_region_id_prior = kf_region_id;
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
      from region r1, region r2, region r3
      where r1.reg_unicode = :kf_region_id and
            r1.parent = r2.reg_unicode and
            r2.parent = r3.reg_unicode
      into reg_name, branch_name, okrug_name;
    end
    trans_region_name = (select reg_code || ' ' || reg_name
                         from region
                         where reg_unicode = :trans_region_id);
    --  Параметры Кредитного сотрудника
    if (kf_officer_id <> kf_officer_id_prior) then
    begin
      kf_officer_name = null;
      kf_officer_id_prior = kf_officer_id;
      select c_name_fio
      from clients
      where c_id = :kf_officer_id
      into kf_officer_name;
      if (kf_officer_name is null) then
        kf_officer_name = '';
    end
    --  Дата контракта
    kf_date_create = (select min(kal_date)
                      from knaktlist
                      where kal_from_kf_id = :k_from_kf_id and
                            knaktlist.is_delete = 0 and
                            kal_from_ka_id = 0);

    k_disb_year = extract(year from k_disb_date);
    k_disb_month = extract(month from k_disb_date);
    trans_year = extract(year from trans_date);
    trans_month = extract(month from trans_date);

    if (k_is_kredline = 0) then
      k_is_kredline_str = 'Нет';
    if (k_is_kredline = 1) then
      k_is_kredline_str = 'Да';

    pr_loan_type_str = '';
    if (pr_loan_type = 0) then
      pr_loan_type_str = 'Группа';
    if (pr_loan_type = 1) then
      pr_loan_type_str = 'Индив.';

    pr_is_secured_str = '';
    if (pr_is_secured = 0) then
      pr_is_secured_str = 'Без залога';
    if (pr_is_secured = 1) then
      pr_is_secured_str = 'Залог';
    if (pr_is_secured = 2) then
      pr_is_secured_str = 'Поручители';
    if (pr_is_secured = 3) then
      pr_is_secured_str = 'Залог,поручит.';

    pr_int_calc_type_str = '';
    if (pr_int_calc_type = 0) then
      pr_int_calc_type_str = 'На сумму кредита';
    if (pr_int_calc_type = 1) then
      pr_int_calc_type_str = 'На остаток';
    if (pr_int_calc_type = 2) then
      pr_int_calc_type_str = 'Аннуитет';

    k_circle_real = :k_circle + :k_circle_internal - 1;
    k_percent_real = :k_percent - :k_percent_lgota;

    if ((extract(weekday from :trans_date) = 0) or (extract(weekday from :trans_date) = 6)) then
      trans_weekend = 'Выходной день';
    else
      trans_weekend = 'Рабочий день';

    if (trans_type = 5) then
    begin
      if (trans_refdata_id <> 0 and
          trans_refdata_id_prior <> trans_refdata_id) then
      begin
        trans_refdata_id_prior = trans_refdata_id;
        select rd_name
        from refdata
        where rd_id = :trans_refdata_id
        into woff_memo1;
      end
      else
      if (trans_refdata_id = 0) then
        woff_memo1 = 'Не заполнен';

      woff_memo = woff_memo1;
    end
    else
    begin
      woff_memo = '';
    end

    if (db_id = 94401) then
    begin
      if (exists(select *
                 from azrepaykont ar
                 where ar.ark_from_k_id = :k_id)) then
        is_azrepaykont = 1;
      else
        is_azrepaykont = 0;
    end

    suspend;
  end
end
