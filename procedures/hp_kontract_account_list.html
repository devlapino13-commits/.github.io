CREATE PROCEDURE HP_KONTRACT_ACCOUNT_LIST(
    K_ID DECIMAL(18,0),
    K_DATE DATE)
RETURNS (
    ACC_BALANCE VARCHAR(20),
    ACC_PERCENT VARCHAR(20))
AS
declare variable reg_id integer;
declare variable pr_id integer;
begin
  /* Procedure Text */
  acc_balance = null;
  acc_percent = null;

  select kal_new_reg_id, k_from_pr_id
  from kontract, knaktlist
  where k_id = :k_id and
        k_from_kf_id = kal_from_kf_id and
        kal_date <= :k_date and
        kal_date_end >= :k_date and
        knaktlist.is_delete = 0
  into :reg_id, :pr_id;

  select res_account
  from hp_make_kredit_acclist(:pr_id, :reg_id, 1)
  into :acc_balance;

  select res_account
  from hp_make_kredit_acclist(:pr_id, :reg_id, 3)
  into :acc_percent;

  suspend;
end
