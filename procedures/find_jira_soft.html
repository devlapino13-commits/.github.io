CREATE PROCEDURE FIND_JIRA_SOFT
RETURNS (
    JSW_ID D_INTEGER,
    JSW_TEXT D_STR_255)
AS
begin
  for select jsw_id, jsw_parent_id, jsw_code, jsw_full_code, jsw_name, jsw_memo, jsw_inactive
  from jira_soft
  order by jsw_full_code
  as cursor cur
  do
  begin
    jsw_id = cur.jsw_id;
    jsw_text = cur.jsw_full_code;
    if (jsw_id <> 0) then
    begin
      if (position(cur.jsw_name, cur.jsw_full_code) = 0) then
        jsw_text = jsw_text || '   - ' || cur.jsw_name;
    end else
      jsw_text = 'Не указана';

    suspend;
  end
end
