CREATE PROCEDURE HP_INTERN_TO_STAFF(
    I_ID D_INTEGER)
RETURNS (
    IS_ERR D_SMALLINT)
AS
declare variable ho_text d_str_long;
declare variable ho_reason d_str_255;
declare variable jl_name d_str_100;
declare variable date_end d_date;
declare variable reg_id d_integer;
declare variable ho_id d_id;
declare variable jl_id d_integer;
declare variable c_id d_id;
begin
  is_err = 0;

  select i.i_date_end_fact, i.i_from_reg_id, i.i_from_jl_id, i.i_from_c_id
  from interns i
  where i.i_id = :i_id
  into :date_end, :reg_id, :jl_id, :c_id;

  if (:jl_id = 270) then
    jl_id = 227; --СПФ
  if (:jl_id = 271) then
    jl_id = 7; --Кассир

  select jl.jl_name
  from joblist jl
  where jl.jl_id = :jl_id
  into :jl_name;

  ho_text = 'Принять на должность ' || :jl_name || ' с ' || f_datetostr(:date_end) || ' г.';
  ho_reason = 'Заявление сотрудника и трудовой договор от ' || f_datetostr(:date_end);
  ho_id = (select kv
           from get_kv(1090100, 11));

  insert into hr_orders (ho_id, ho_date, ho_from_reg_id, ho_text, ho_reason, ho_from_un_id, ho_save_time, ho_order_type,
                         ho_order_no, ho_status, ho_status_user_id, ho_status_datetime)
  values (:ho_id, :date_end, :reg_id, :ho_text, :ho_reason, 1000000000000, current_timestamp, 1, '', 1, 1000000000000,
          current_timestamp);

  execute procedure hp_hr_pstatus_insert(1, :ho_id, :date_end, '31.12.2999',
      datediff(month, :date_end, cast('31.12.2099' as date)), '31.12.2099', :reg_id, 3, :jl_id, 1, 1, :c_id, 0, null,
      null, 0, 1, null);

  suspend;
end
