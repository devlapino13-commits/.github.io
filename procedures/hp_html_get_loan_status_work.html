CREATE PROCEDURE HP_HTML_GET_LOAN_STATUS_WORK(
    K_ID D_ID)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable c_name_fio d_str_100;
declare variable c_id d_id;
declare variable kiw_from_c_id d_str_100;
declare variable kiw_memo d_str_255;
declare variable kiw_days d_integer;
declare variable kiw_price d_currency;
declare variable kip_prepay_sum d_currency;
declare variable kip_balance_sum d_currency;
begin
  -- Процедура отображения работ по контракту в окне "Статус заявки"
  res_text = '';

  for select kiw_from_c_id, c_name_fio, kiw_memo, kiw_days, kiw_price
      from hp_find_detail_work_istisna(:k_id)
      into c_id, c_name_fio, kiw_memo, kiw_days, kiw_price
  do
  begin
    -- Получаем сумму предоплаты и сумму остатка к оплате
    select  kip.kip_prepay_sum, kip.kip_balance_sum
    from kontract_istinaa_prepay kip
    where kip.kip_from_k_id = :k_id and
          kip.kip_from_c_id = :c_id and
          kip.kip_is_active = 1
    into kip_prepay_sum, kip_balance_sum;

    res_text = f_format('<h3>%1</h3><table width="100%"><col width="40%"><col width="60%">', '');
    res_text = blob_append(res_text, f_html_table_row('<b>Наименование работы</b>', :kiw_memo, '', '', -1, -1, -1, -1, 2, ''));
    res_text = blob_append(res_text, f_html_table_row('<b>Стоимость работ</b>', :kiw_price, '', '', -1, -1, -1, -1, 2, ''));
    res_text = blob_append(res_text, f_html_table_row('<b>Срок выполнения</b>', :kiw_days, '', '', -1, -1, -1, -1, 2, ''));
    res_text = blob_append(res_text, f_html_table_row('<b>Подрядчик</b>', :c_name_fio, '', '', -1, -1, -1, -1, 2, ''));
    res_text = blob_append(res_text, f_html_table_row('<b>Сумма предоплаты</b>', :kip_prepay_sum, '', '', -1, -1, -1, -1, 2, ''));
    res_text = blob_append(res_text, f_html_table_row('<b>Осталось оплатить</b>', :kip_balance_sum, '', '', -1, -1, -1, -1, 2, ''));
    res_text = blob_append(res_text, '</table>');
    suspend;
  end

  if (coalesce(:kiw_memo, '') = '') then
  begin
    res_text = '';
    suspend;
  end
end
