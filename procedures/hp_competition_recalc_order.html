CREATE PROCEDURE HP_COMPETITION_RECALC_ORDER(
    CPP_ID INTEGER)
AS
declare variable cpr_id integer;
declare variable total_points decimal(18,4);
declare variable cpr_orderno integer;
declare variable prior_points decimal(18,4);
declare variable group_name_prior varchar(255);
declare variable group_name varchar(255);
declare variable cl_count integer;
declare variable need_order_by_group integer;
declare variable rd_code varchar(255);
declare variable ad_diff_client curr_type;
declare variable stage integer;
declare variable res_date_begin date;
declare variable res_date_end date;
declare variable rd_id integer;
declare variable ad_disb_loan integer;
declare variable cpd_add_data_6 integer;
declare variable cpd_add_data_2 integer;
declare variable db_id d_integer;
declare variable cpr_group_name d_str_255;
declare variable cpr_object_id d_id;
declare variable cpr_calced_points decimal(15,2);
declare variable cpr_final_points decimal(15,2);
declare variable cpd_points_all decimal(15,2);
declare variable cpr_rank decimal(15,2);
declare variable avg_disb decimal(15,2);
begin
  -- Расчет занимаемых

  select rd_code, cpp_stage, cpp_date_begin, cpp_date_end, rd_id
  from refdata, competperiod
  where rd_id = cpp_from_rd_id and
        cpp_id = :cpp_id
  into :rd_code, :stage, -- 0 - уровень недели; 1 - уровень тура;
       :res_date_begin, :res_date_end, :rd_id;

  db_id = f_get_db_id();
  if (db_id in (39801, 41701) and
      :stage = 1 and
      :rd_code = 'Competitions.Olimpic_2021_mo') then
  begin
    cl_count = 0;
    for select cpr_id, cpr_group_name, cpr_final_points, avg(cpd_add_data_3 + cpd_add_data_4) cpd_add_data_6,
               avg(cpd_add_data_2) cpd_add_data_2, substring(cpr_comment from 12 for 2) avg_disb
        from competresult, competdetails
        where cpr_from_cpp_id = 526 and
              cpd_from_cpr_id = cpr_id
        group by cpr_id, cpr_group_name, cpr_final_points, avg_disb

        order by cpr_group_name, cpr_final_points desc, avg_disb desc, cpd_add_data_6 desc, -- сортировка по сумме выдачи
            cpd_add_data_2 desc -- сортировка по приросту портфеля
        into :cpr_id, :group_name, :total_points, cpd_add_data_6, cpd_add_data_2, avg_disb
    do
    begin

      cpr_orderno = 0;
      prior_points = -999999;

      cl_count = cl_count + 1;
      cpr_orderno = cl_count;

      update competresult
      set cpr_order_in_group = :cpr_orderno,
          cpr_final_orderno = :cpr_orderno
      where cpr_id = :cpr_id;
    end
  end
  else
  if (db_id in (39801, 41701)) then
  begin
    -- Пересчитываем итоги по командам
    for with t_cpd_all_points
        as (select distinct cpr_id, --
                            cpr_calced_points + cpr_additional_points as sum_points
            from competresult cr
            where cpr_from_cpp_id = :cpp_id)
        select cpr_id, rank() over(order by sum_points desc)
        from t_cpd_all_points
        into :cpr_id, :cpr_rank
    do
    begin
      update competresult
      set cpr_final_points = cpr_calced_points + cpr_additional_points,
          cpr_order_in_group = :cpr_rank,
          cpr_final_orderno = :cpr_rank,
          cpr_comment = coalesce(cpr_comment, '')
      where (cpr_from_cpp_id = :cpp_id and
            cpr_id = :cpr_id);
    end

    exit;
  end

  if ((rd_code starting 'Competitions.Champions_League_201' or rd_code starting 'Competitions.Olimpic_201') and
      db_id <> 81001) then
  begin
    if (stage = 0) then
    begin
      cpr_orderno = 0;
      prior_points = -999999;
      cl_count = 0;

      group_name_prior = '[{x}]';
      for select cpr_id, cpr_group_name, cpr_final_points, avg(cpd_add_data_3 + cpd_add_data_4) cpd_add_data_6,
                 avg(cpd_add_data_2) cpd_add_data_2
          from competresult, competdetails
          where cpr_from_cpp_id = :cpp_id and
                cpr_group_name <> '' and
                cpr_object_kind = -1 -- Refdata, названия групп
                and
                cpd_from_cpr_id = cpr_id
          group by cpr_id, cpr_group_name, cpr_final_points

          order by cpr_group_name, cpr_final_points desc, cpd_add_data_6 desc, -- сортировка по сумме выдачи
              cpd_add_data_2 desc -- сортировка по приросту портфеля
          into :cpr_id, :group_name, :total_points, cpd_add_data_6, cpd_add_data_2
      do
      begin
        if (group_name <> group_name_prior) then
        begin
          cpr_orderno = 0;
          prior_points = -999999;
          group_name_prior = group_name;
          cl_count = 0;
        end
        cl_count = cl_count + 1;
        cpr_orderno = cl_count;

        update competresult
        set cpr_order_in_group = :cpr_orderno,
            cpr_final_orderno = :cpr_orderno
        where cpr_id = :cpr_id;
      end
    end
    else
    begin
      cpr_orderno = 0;
      prior_points = -999999;
      cl_count = 0;

      group_name_prior = '[{x}]';
      for select cpr_id, cpr_group_name, cpr_final_points,
                 --             если в итоговом расчете одинаковые балы то места определяется по кол-ву выдач
                 avg((select sum(cpd_add_data_3 + cpd_add_data_4)
                      from competresult, competperiod, competdetails
                      where cpr_from_cpp_id = cpp_id and
                            cpp_from_rd_id = :rd_id and
                            cpp_date_begin between :res_date_begin and :res_date_end and
                            cpp_date_end between :res_date_begin and :res_date_end and
                            cpp_stage = 0 and
                            cpr_id = cpd_from_cpr_id and
                            cpr_object_id = cpr.cpr_object_id)) sum_vydach,
                 --             если в итоговом расчете одинаковые балы то места определяется по увеличению клиентов за итоговый период
                 avg((select sum(cpd_add_data_5)
                      from competresult, competperiod, competdetails
                      where cpr_from_cpp_id = cpp_id and
                            cpp_from_rd_id = :rd_id and
                            cpp_date_begin between :res_date_begin and :res_date_end and
                            cpp_date_end between :res_date_begin and :res_date_end and
                            cpp_stage = 0 and
                            cpr_id = cpd_from_cpr_id and
                            cpr_object_id = cpr.cpr_object_id)) prirost_portfel
          from competresult cpr
          where cpr_from_cpp_id = :cpp_id
          group by cpr_id, cpr_group_name, cpr_final_points
          order by cpr_group_name, cpr_final_points desc, sum_vydach desc, prirost_portfel desc
          into :cpr_id, :group_name, :total_points, :ad_disb_loan, :ad_diff_client
      do
      begin
        if (group_name <> group_name_prior) then
        begin
          cpr_orderno = 0;
          prior_points = -999999;
          group_name_prior = group_name;
          cl_count = 0;
        end
        cl_count = cl_count + 1;
        cpr_orderno = cl_count;

        update competresult
        set cpr_order_in_group = :cpr_orderno,
            cpr_final_orderno = :cpr_orderno
        where cpr_id = :cpr_id;
      end
    end

    exit;
  end

  need_order_by_group = 0; -- Присваивать ли места на уровне Группы
  if (rd_code starting 'Competitions.Olimpic') then
    need_order_by_group = 1;

  -- Перерасчет занимаемых общих мест
  cpr_orderno = 0;
  prior_points = -999999;
  cl_count = 0;
  for select cpr_id, cpr_final_points
      from competresult
      where cpr_from_cpp_id = :cpp_id
      order by cpr_final_points desc
      into :cpr_id, :total_points
  do
  begin
    cl_count = cl_count + 1;
    if (total_points <> prior_points) then
    begin
      cpr_orderno = cl_count;
      prior_points = total_points;
    end
    if (need_order_by_group = 1) then
    begin
      update competresult
      set cpr_final_orderno = :cpr_orderno
      where cpr_id = :cpr_id;
    end
    else
    begin
      update competresult
      set cpr_final_orderno = :cpr_orderno,
          cpr_order_in_group = :cpr_orderno
      where cpr_id = :cpr_id;
    end
  end

  if (rd_code not starting 'Competitions.Rating_MO') then
  begin
    if (need_order_by_group = 1) then
    begin
      -- Перерасчет занимаемых мест внутри группы
      cpr_orderno = 0;
      prior_points = -999999;
      cl_count = 0;

      group_name_prior = '[{x}]';
      for select cpr_id, cpr_group_name, cpr_final_points
          from competresult
          where cpr_from_cpp_id = :cpp_id
          order by cpr_group_name, cpr_final_points desc
          into :cpr_id, :group_name, :total_points
      do
      begin
        if (group_name <> group_name_prior) then
        begin
          cpr_orderno = 0;
          prior_points = -999999;
          group_name_prior = group_name;
          cl_count = 0;
        end
        cl_count = cl_count + 1;
        if (total_points <> prior_points) then
        begin
          cpr_orderno = cl_count;
          prior_points = total_points;
        end
        update competresult
        set cpr_order_in_group = :cpr_orderno
        where cpr_id = :cpr_id;
      end
    end
  end
  else
  begin
    -- Перерасчет занимаемых мест внутри группы
    cpr_orderno = 0;
    prior_points = -999999;
    cl_count = 0;

    group_name_prior = '[{x}]';
    for select cpr_id, left(cpr_group_name, 1), cpr_final_points
        from competresult
        where cpr_from_cpp_id = :cpp_id
        order by 2, 3 desc
        into :cpr_id, :group_name, :total_points
    do
    begin
      if (group_name <> group_name_prior) then
      begin
        cpr_orderno = 0;
        prior_points = -999999;
        group_name_prior = group_name;
        cl_count = 0;
      end
      cl_count = cl_count + 1;
      if (total_points <> prior_points) then
      begin
        cpr_orderno = cl_count;
        prior_points = total_points;
      end
      update competresult
      set cpr_order_in_group = :cpr_orderno
      where cpr_id = :cpr_id;
    end

  end

end
