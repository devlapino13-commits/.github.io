CREATE PROCEDURE HP_FPS_KONTRACT_CHECK(
    K_ID D_ID,
    CHECK_KIND D_STR_CODE)
RETURNS (
    C_NAME_FIO D_STR_255,
    ERROR_TEXT D_STR_LONG)
AS
declare variable kc_id d_id;
declare variable fcb_id d_integer;
declare variable fcb_status d_integer;
declare variable k_disb_kind d_integer;
begin
  if (f_get_db_id() <> 81001) then
    exit;

  select k_disb_kind
  from kontract
  where k_id = :k_id
  into k_disb_kind;

  if (k_disb_kind <> 7) then
    exit;

  for select kc_id, c_name_fio
  from knclient, clients
  where kc_from_k_id = :k_id and
       kc_amount_approved > 0 and
       kc_from_c_id = c_id
  into kc_id, c_name_fio
  do
  begin
    error_text = '';
    
    select fbl_from_fcb_id
    from fps_loan_banks
    where flb_from_kc_id = :kc_id
    into fcb_id;
    
    if (fcb_id is null) then
    begin
      error_text = 'СБП банк не указан';
    end
    else if (f_get_const('fps_manual_disb') <> 1) then
    begin
      select fcb_status
      from fps_client_banks
      where fcb_id = :fcb_id
      into fcb_status;

      if (fcb_status = -1) then
        error_text = 'СБП: Клиент не найден в этом банке';
      else if (fcb_status = 0) then
          error_text = 'СБП: Дождитесь окончания проверки клиента';
      else if (fcb_status = 1 and check_kind <> '.send_kk.') then
          error_text = 'СБП: Требуется подтверждение клиента от КК';
      -- Тут контроль на всякий случай, по-идее не должен доходить
      else if (fcb_status <> 2 and check_kind in ('.loan_approve.','.loan_disb.')) then
          error_text = f_format('СБП: Неверный статус:%1, обратитесь в техподдержку', fcb_status);
    end
    if (error_text <> '') then
      suspend;
  end
end
