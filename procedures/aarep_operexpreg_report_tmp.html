CREATE PROCEDURE AAREP_OPEREXPREG_REPORT_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OB_BALANCE VARCHAR(20),
    OB_ACC_SHORT VARCHAR(20),
    REG_ID INTEGER,
    OB_AMOUNT DECIMAL(18,4),
    BG_KIND_STR VARCHAR(20),
    CC_NAME VARCHAR(255),
    CC_ID INTEGER)
AS
declare variable bg_kind_dt integer;
declare variable bg_kind_kt integer;
declare variable ob_balance_dt varchar(20);
declare variable ob_balance_kt varchar(20);
declare variable ob_acc_short_dt varchar(20);
declare variable ob_acc_short_kt varchar(20);
declare variable ob_from_ol_id decimal(18,0);
declare variable ob_number integer;
declare variable ob_summa decimal(18,4);
declare variable n integer;
declare variable ob_summa_tmp decimal(18,4);
declare variable reg_tmp integer;
declare variable reg_tmp_0 integer;
declare variable reg_head_office integer;
declare variable cc_name_default varchar(255);
declare variable cu decimal(18,4);
begin
  select cc_name
  from costcenter
  where cc_id = 1
  into :cc_name_default;

  for select ob_from_ol_id, ob_number, ob_summa, bg_dt.bg_id, bg_kt.bg_id, ob_ol_reg_id, al_dt.al_account,
             al_kt.al_account, bg_dt.bg_kind, bg_kt.bg_kind
      from operbook, region_list(:rep_region_id, 1, -1, 1), region, acc_list al_dt, acc_list al_kt, balgroup bg_dt, balgroup bg_kt
      where ob_ol_is_active = 1 and
            ob_ol_date between :rep_date_begin and :rep_date_end and
            ob_ol_reg_id = id_list and
            ob_account_dt = al_dt.al_id and
            ob_account_kt = al_kt.al_id and
            al_dt.al_from_bg_id = bg_dt.bg_id and
            al_kt.al_from_bg_id = bg_kt.bg_id and
            ((bg_dt.bg_kind >= 4 and
            bg_kt.bg_kind <> 3) or
            (bg_kt.bg_kind >= 4 and
            bg_dt.bg_kind <> 3)) and
            ob_summa is not null and
            ob_summa <> 0 and
            ob_ol_reg_id = reg_unicode
      into ob_from_ol_id, ob_number, :ob_summa, :ob_balance_dt, :ob_balance_kt, :reg_id, ob_acc_short_dt,
           ob_acc_short_kt, bg_kind_dt, :bg_kind_kt
  do
  begin
    ob_amount = 0;
    n = 0;
    for select cc_id, cc_name, occ_amount
        from opercc, costcenter
        where occ_from_ol_id = :ob_from_ol_id and
              occ_ob_number = :ob_number and
              occ_from_cc_id = cc_id and
              occ_amount <> 0
        into :cc_id, :cc_name, :cu
    do
    begin
      n = n + 1;
      if (bg_kind_dt = 4) then
      begin
        bg_kind_str = 'Доходы';
        ob_balance = :ob_balance_dt;
        ob_acc_short = :ob_acc_short_dt;
        ob_amount = -cu;
        suspend;
      end
      if (bg_kind_kt = 4) then
      begin
        bg_kind_str = 'Доходы';
        ob_balance = :ob_balance_kt;
        ob_acc_short = :ob_acc_short_kt;
        ob_amount = cu;
        suspend;
      end
      if (bg_kind_dt = 5) then
      begin
        bg_kind_str = 'Расходы';
        ob_balance = :ob_balance_dt;
        ob_acc_short = :ob_acc_short_dt;
        ob_amount = -cu;
        suspend;
      end
      if (bg_kind_kt = 5) then
      begin
        bg_kind_str = 'Расходы';
        ob_balance = :ob_balance_kt;
        ob_acc_short = :ob_acc_short_kt;
        ob_amount = cu;
        suspend;
      end

    end
    if (n = 0) then
    begin
      cc_name = :cc_name_default;
      cc_id = 1;
      if (bg_kind_dt = 4) then
      begin
        bg_kind_str = 'Доходы';
        ob_balance = :ob_balance_dt;
        ob_acc_short = :ob_acc_short_dt;
        ob_amount = -ob_summa;
        suspend;
      end
      if (bg_kind_kt = 4) then
      begin
        bg_kind_str = 'Доходы';
        ob_balance = :ob_balance_kt;
        ob_acc_short = :ob_acc_short_kt;
        ob_amount = ob_summa;
        suspend;
      end
      if (bg_kind_dt = 5) then
      begin
        bg_kind_str = 'Расходы';
        ob_balance = :ob_balance_dt;
        ob_acc_short = :ob_acc_short_dt;
        ob_amount = -ob_summa;
        suspend;
      end
      if (bg_kind_kt = 5) then
      begin
        bg_kind_str = 'Расходы';
        ob_balance = :ob_balance_kt;
        ob_acc_short = :ob_acc_short_kt;
        ob_amount = ob_summa;
        suspend;
      end
    end
  end
end
