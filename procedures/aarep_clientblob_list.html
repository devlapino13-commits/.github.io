CREATE PROCEDURE AAREP_CLIENTBLOB_LIST(
    C_ID DECIMAL(18,0))
RETURNS (
    CB_FROM_C_ID TYPE OF COLUMN CLIENTBLOB.CB_FROM_C_ID,
    CB_ID TYPE OF COLUMN CLIENTBLOB.CB_ID,
    CB_NAME TYPE OF COLUMN CLIENTBLOB.CB_NAME,
    CB_BLOB TYPE OF COLUMN CLIENTBLOB.CB_BLOB,
    CB_FILENAME TYPE OF COLUMN CLIENTBLOB.CB_FILENAME,
    CB_TYPE TYPE OF COLUMN CLIENTBLOB.CB_TYPE,
    CB_FROM_UN_ID TYPE OF COLUMN CLIENTBLOB.CB_FROM_UN_ID,
    CB_DATETIME TYPE OF COLUMN CLIENTBLOB.CB_DATETIME,
    CB_FROM_BLS_ID TYPE OF COLUMN CLIENTBLOB.CB_FROM_BLS_ID,
    CB_PHOTO_KIND VARCHAR(50))
AS
begin
  cb_photo_kind = 'Прочие';
  for select cb_from_c_id, cb_id, cb_name, cb_blob, cb_filename, cb_type, cb_from_un_id, cb_datetime, cb_from_bls_id
      from clientblob
      where cb_from_c_id = :c_id
      into :cb_from_c_id, :cb_id, :cb_name, :cb_blob, :cb_filename, :cb_type, :cb_from_un_id, :cb_datetime,
           :cb_from_bls_id
  do
  begin
    if (cb_from_bls_id <> 0) then
    begin
      cb_type = -1;
      cb_blob = null;
      select bls_filename, bls_caption, bls_from_un_id, bls_datetime
      from blobstore
      where bls_id = :cb_from_bls_id
      into :cb_filename, :cb_name, :cb_from_un_id, :cb_datetime;
      suspend;
    end
    else
    begin
      if (cb_blob is not null) then
        suspend;
    end
  end

  cb_blob = null;
  cb_from_bls_id = 0;
  cb_id = -1;
  select c_blob_photo_id
  from clients
  where c_id = :c_id
  into :cb_from_bls_id;
  if (cb_from_bls_id is null) then
    cb_from_bls_id = 0;
  if (cb_from_bls_id <> 0) then
  begin
    cb_type = -1;
    cb_photo_kind = 'Фото контрагента';
    select bls_filename, bls_caption, bls_from_un_id, bls_datetime
    from blobstore
    where bls_id = :cb_from_bls_id
    into :cb_filename, :cb_name, :cb_from_un_id, :cb_datetime;
    suspend;
  end

  cb_from_bls_id = 0;
  cb_id = -2;
  select c_blob_pass1_id
  from clients
  where c_id = :c_id
  into :cb_from_bls_id;
  if (cb_from_bls_id is null) then
    cb_from_bls_id = 0;
  if (cb_from_bls_id <> 0) then
  begin
    cb_type = -1;
    cb_photo_kind = 'Паспорт, лицевая';
    select bls_filename, bls_caption, bls_from_un_id, bls_datetime
    from blobstore
    where bls_id = :cb_from_bls_id
    into :cb_filename, :cb_name, :cb_from_un_id, :cb_datetime;
    suspend;
  end

  cb_from_bls_id = 0;
  cb_id = -3;
  select c_blob_pass2_id
  from clients
  where c_id = :c_id
  into :cb_from_bls_id;
  if (cb_from_bls_id is null) then
    cb_from_bls_id = 0;
  if (cb_from_bls_id <> 0) then
  begin
    cb_type = -1;
    cb_photo_kind = 'Паспорт, обратная';
    select bls_filename, bls_caption, bls_from_un_id, bls_datetime
    from blobstore
    where bls_id = :cb_from_bls_id
    into :cb_filename, :cb_name, :cb_from_un_id, :cb_datetime;
    suspend;
  end

end
