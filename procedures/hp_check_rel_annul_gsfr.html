CREATE PROCEDURE HP_CHECK_REL_ANNUL_GSFR(
    CREL_FROM_C_ID D_ID,
    CREL_OTHER_C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable c_name_fio d_str_100;
declare variable c_pass_no d_str_20;
declare variable c_inn d_str_20;
declare variable c_birth_date d_date;
declare variable phone_number d_str_20;
declare variable rel_other_c_id d_id;
begin
  res_text = '';

  if (exists(select *
             from gsfr_rel_annul_list gral
             where gral.gral_from_c_id = :crel_from_c_id and
                   gral.gral_other_c_id = :crel_other_c_id and
                   gral.gral_is_delete = 0) or
      exists(select *
      from gsfr_rel_annul_list gral
      where gral.gral_from_c_id = :crel_other_c_id and
            gral.gral_other_c_id = :crel_from_c_id and
            gral.gral_is_delete = 0)) then
    exit;

  c_name_fio = '';
  c_pass_no = '';
  c_inn = '';
  c_birth_date = null;
  select cl.c_name_fio, cl.c_pass_no, cl.c_inn, cl.c_birth_date
  from clients cl
  where cl.c_id = :rel_other_c_id
  into c_name_fio, c_pass_no, c_inn, c_birth_date;

  phone_number = f_cont_info('clients', :rel_other_c_id, 'cell_main', 'last');

  if (exists(select *
             from hp_check_client_fio_ft_fuzzy(:rel_other_c_id, :c_name_fio, :c_pass_no, :c_inn, :c_birth_date, :phone_number, 1, 1))) then
  begin
    if (not exists(select *
                   from hp_permission_access_check(:un_id, 'permactions', 0, 'FinStart.GSFR_relation_annul')
                   where has_access = 1)) then
      res_text = 'Совпданеие с лциом по списку ГСФР. Обратитесь к комплаенс-офицеру. ФИО: ' || :c_name_fio;
  end

  if (res_text = '') then
  begin
    c_name_fio = '';
    c_pass_no = '';
    c_inn = '';
    c_birth_date = null;
    select cl.c_name_fio, cl.c_pass_no, cl.c_inn, cl.c_birth_date
    from clients cl
    where cl.c_id = :rel_other_c_id
    into c_name_fio, c_pass_no, c_inn, c_birth_date;

    phone_number = f_cont_info('clients', :rel_other_c_id, 'cell_main', 'last');

    if (exists(select *
               from hp_check_client_fio_ft_fuzzy(:rel_other_c_id, :c_name_fio, :c_pass_no, :c_inn, :c_birth_date, :phone_number, 1, 1))) then
    begin
      if (not exists(select *
                     from hp_permission_access_check(:un_id, 'permactions', 0, 'FinStart.GSFR_relation_annul')
                     where has_access = 1)) then
        res_text = 'Совпданеие с лциом по списку ГСФР. Обратитесь к комплаенс-офицеру. ФИО: ' || :c_name_fio;
    end
  end

  suspend;
end
