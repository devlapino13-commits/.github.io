CREATE PROCEDURE HP_CASHCHECK_SIGN(
    CCM_ID D_INTEGER,
    C_ID D_ID = null)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable dt_curr d_datetime;
declare variable ccm_ol_id d_id;
declare variable ol_id d_id;
declare variable signed boolean;
declare variable signed_all boolean;
begin
  err_text = '';
  dt_curr = current_timestamp;

  ol_id = (
    select
      ccm_ol_id
    from cashcheckmain
    where ccm_id = :ccm_id);

  update cashcheckmain set
    ccm_auditor_datesign = :dt_curr
  where ccm_id = :ccm_id
    and ccm_auditor = coalesce(:c_id, ccm_auditor)
    and ccm_auditor_datesign is null;
  signed = row_count <> 0;

  update cashcheckmain set
    ccm_cashier_curr_datesign = :dt_curr
  where ccm_id = :ccm_id
    and ccm_cashier_curr = coalesce(:c_id, ccm_cashier_curr)
    and ccm_cashier_curr_datesign is null;
  signed = signed or (row_count <> 0);

  update cashcheckmain set
    ccm_cashier_new_datesign = :dt_curr
  where ccm_id = :ccm_id
    and ccm_cashier_new = coalesce(:c_id, ccm_cashier_new)
    and ccm_cashier_new_datesign is null;
  signed = signed or (row_count <> 0);

  update cashcheckpart set
    ccp_datesign = :dt_curr
  where ccp_from_ccm_id = :ccm_id
    and ccp_from_c_id = coalesce(:c_id, ccp_from_c_id)
    and ccp_datesign is null;
  signed = signed or (row_count <> 0);

  if (signed) then
  begin
    signed_all = (select signed_all from hp_cashcheck_signedall(:ccm_id));

    if (signed_all and ol_id is null) then
      ol_id = (select ol_id from hp_cashcheckmain_acc_entry(:ccm_id));

    update cashcheckmain set
      ccm_status = iif(:signed_all, 2, 1),
      ccm_ol_id = :ol_id
    where ccm_id = :ccm_id
      and ccm_status in (0, 1);
  end else
    err_text = f_format('Ошибка подписания "акта инвентаризации кассы" №%1', :ccm_id);

  suspend;
end
