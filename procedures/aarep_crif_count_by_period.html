CREATE PROCEDURE AAREP_CRIF_COUNT_BY_PERIOD(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    STATUS_DESCRIPTION D_STR_64,
    REQUEST_COUNT D_INTEGER)
AS
begin
    for select 
            case eil.eil_status
                when 0 then 'В процессе'
                when 1 then 'В процессе'
                when 2 then 'Ответ получен'
                else 'Ответ не получен'
            end as status_description,
            count(*)
        from extinfolist eil
        join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
        where eic.eic_code = 'stop_cr_crif'
            and eil.eil_date between :rep_date_begin and :rep_date_end
        group by status_description
        into status_description,
             request_count
    do
    begin
        suspend;
    end
end
