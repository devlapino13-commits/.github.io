CREATE PROCEDURE HP_JSON_FROM_SQL(
    JSON_KIND VARCHAR(100),
    JSON_NAME VARCHAR(100),
    JSON_SQL BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
RETURNS (
    RES_JSON BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable long_str varchar(3000);
declare variable n_start integer;
declare variable n_finish integer;
declare variable sym varchar(1);
declare variable n integer;
declare variable as_name varchar(100);
declare variable f_name varchar(100);
declare variable rec_count integer;
declare variable tab_sql blob sub_type 1 segment size 80;
declare variable ss varchar(10);
declare variable len integer;
declare variable i integer;
declare variable k1 integer;
declare variable k2 integer;
declare variable v_delimiter varchar(3);
begin
  if (json_name is null) then
    json_name = '';
  json_name = lower(json_name);

  if (not(json_kind in ('array', 'object'))) then
    exception e_default 'Incorrect JSON_Kind param';
  if (trim(coalesce(json_sql, '')) = '') then
    exception e_default 'Empty json_sql param';

  json_sql = replace(json_sql, ascii_char(10), '');
  json_sql = replace(json_sql, ascii_char(13), '');
  json_sql = trim(json_sql);
  while (position('  ', json_sql) > 0) do
  begin
    json_sql = replace(json_sql, '  ', ' ');
  end
  json_sql = ' ' || json_sql;
  json_sql = replace(json_sql, ' from ', ', from ');

  res_json = '';
  n_start = position(' select ', json_sql);
  if (n_start = 0) then
    exception e_default 'No "select" keyword found';
  n_start = n_start + 7;
  n_finish = position(' from ', json_sql, n_start);
  if (n_finish = 0) then
    exception e_default 'No "from" keyword found';

  i = n_start;
  f_name = '';
  tab_sql = '';
  v_delimiter = ' ''';
  while (i < n_finish) do
  begin
    i = i + 1;
    sym = substring(json_sql from i for 1);
    if (sym = ',') then
    begin
      n = position(' as ', f_name);
      if (position(f_name, '(') > 0 and
          n = 0) then
      begin
        exception e_default 'Aggragated field without alias name';
      end
      else
      begin
        if (n > 0) then
          as_name = substring(f_name from n + 4);
        else
          as_name = f_name;
      end
      as_name = trim(as_name);
      as_name = lower(as_name);
      tab_sql = BLOB_APPEND(tab_sql, v_delimiter, '•' || as_name || '•:•''||coalesce(' || as_name || ','''')||''•');
      v_delimiter = ', ';
      f_name = '';
    end
    else
      f_name = f_name || sym;
  end
  tab_sql = 'select ' || tab_sql || ''' from Tab';
  json_sql = replace(json_sql, ', from ', ' from ');

  tab_sql = BLOB_APPEND('with tab as (', json_sql, ') ', tab_sql);

  res_json = '';
  v_delimiter = '';
  long_str = '';

  rec_count = 0;
  for execute statement(tab_sql)
          into :long_str
  do
  begin
    i = 0;
    len = char_length(long_str);
    while (i < len) do
    begin
      i = i + 1;
      k1 = position('•', long_str, i);
      if (k1 > 0) then
      begin
        k2 = position('•', long_str, k1 + 1);
        if (k2 - k1 = 11) then
        begin
          ss = substring(long_str from k1 + 1 for 10);
          if (ss similar to '[[:DIGIT:]]{4}\-[[:DIGIT:]]{2}\-[[:DIGIT:]]{2}' escape '\') then
          begin
            ss = right(ss, 2) || '.' || substring(ss from 6 for 2) || '.' || left(ss, 4);
            long_str = left(long_str, k1) || ss || substring(long_str from k2);
          end
        end
        i = k1;
      end
      else
        i = len;
    end

    long_str = replace(long_str, '\', '\\');
    long_str = replace(long_str, '"', '\"');
    long_str = replace(long_str, '•', '"');
    res_json = BLOB_APPEND(res_json, v_delimiter, '{' || long_str || '}');
    v_delimiter = ',';
    rec_count = rec_count + 1;
  end
  if (json_kind = 'array') then
    res_json = BLOB_APPEND(iif(json_name <> '', '"' || json_name || '":', ''), '[', res_json, ']');
  else
  begin
    if (rec_count = 0) then
      exception e_default 'No record found in JSON object';
    if (rec_count > 1) then
      exception e_default 'Too many records in JSON object';
    if (rec_count = 1) then
      res_json = BLOB_APPEND(iif(json_name <> '', '"' || json_name || '":', ''), res_json);
  end
  suspend;
end
