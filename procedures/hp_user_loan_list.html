CREATE PROCEDURE HP_USER_LOAN_LIST(
    LIST_KIND D_STR_CODE NOT NULL,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE NOT NULL,
    ONLY_OWN_LOANS D_BOOL NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_K_ID D_ID,
    RES_DATE_BEGIN D_DATE,
    RES_IS_OWN D_BOOL)
AS
begin
  if (list_kind not in ('active', 'disb')) then
    exception e_default f_format('Incorrect list_kind: %1', list_kind);

  if (list_kind = 'active') then
  begin
    for select res_k_id, min(res_date_begin), min(res_is_own)
    from hp_tmp_loan_list_active(:rep_date_end, :only_own_loans, :un_id)
    group by 1
    into :res_k_id, :res_date_begin, :res_is_own
    do
      suspend;
  end

  if (list_kind = 'disb') then
  begin
    for select res_k_id, min(res_date_begin), min(res_is_own)
    from hp_tmp_loan_list_disb(:rep_date_begin, :rep_date_end, :only_own_loans, :un_id)
    group by 1
    into :res_k_id, :res_date_begin, :res_is_own
    do
      suspend;
  end

end
