CREATE PROCEDURE UXP$UXALIASACTION_UPDATE(
    UXA_UUID D_UUID,
    ITEM_CODE D_STR_CODE,
    UA_UUID D_UUID,
    NEW_CHECK D_BOOL)
AS
declare variable ux_uuid d_uuid;
begin
  select uxa_owner
  from ux$alias
  where uxa_uuid = :uxa_uuid
  into ux_uuid;

  if (ua_uuid is not null) then
  begin
    if (new_check = 1) then
    begin
      update or insert into ux$aliasaction (uaa_uxa_uuid, uaa_ua_uuid)
      values (:uxa_uuid, :ua_uuid)
      matching (uaa_uxa_uuid, uaa_ua_uuid);
    end else
    begin
      delete from ux$aliasaction
      where uaa_uxa_uuid = :uxa_uuid and
            uaa_ua_uuid = :ua_uuid;
    end
  end else
  begin
    if (right(item_code, 2) = '00') then
    begin
      item_code = left(item_code, char_length(item_code) - 2);

      if (new_check = 1) then
      begin
        for select ua_uuid
        from ux$action
        where ua_owner = :ux_uuid and
              ua_menu_index starting :item_code
        into ua_uuid
        do
        begin
          update or insert into ux$aliasaction (uaa_uxa_uuid, uaa_ua_uuid)
          values (:uxa_uuid, :ua_uuid)
          matching (uaa_uxa_uuid, uaa_ua_uuid);
        end
      end else
      begin
        for select ua_uuid
        from ux$action
        where ua_owner = :ux_uuid and
              ua_menu_index starting :item_code
        into ua_uuid
        do
        begin
          delete from ux$aliasaction
          where uaa_uxa_uuid = :uxa_uuid and
                uaa_ua_uuid = :ua_uuid;
        end
      end
    end
  end
end
