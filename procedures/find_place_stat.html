CREATE PROCEDURE FIND_PLACE_STAT(
    FIND_ID D_STR_CODE,
    FIND_TEXT D_STR_100,
    ADD_PARAMS D_STR_8100,
    UN_ID D_ID)
RETURNS (
    PL_ID D_ID,
    PL_PARENT_ID D_ID,
    PL_NAME D_STR_100,
    PL_PARENT_NAME D_STR_100)
AS
declare variable db_id d_integer;
declare variable main_pl_id d_id;
begin
  db_id = f_get_db_id();
  if (find_text is null) then
    find_text = '';

  if (un_id is null) then un_id = 1000000000000;
  if (add_params is null) then add_params = '';

  pl_parent_id = null;
  pl_parent_name = '';

  if (find_text = '') then
  begin
    if (find_id <> '') then
    begin
      main_pl_id = pl_id;
      for select pl_id, parent,
                 iif(plt_short_name_side = -1, plt_name || ' ', '') ||
                 pl_name ||
                 lower(iif(plt_short_name_side = 1, ' '|| plt_name, ''))
      from places, placetype
      where pl_id = :find_id and
            pl_status = plt_id
      into pl_id, pl_parent_id, pl_name
      do
        Suspend;

      Exit;
    end

    select pl_id, pl_name
    from places
    where pl_id starting left(:db_id, 3) and
          parent is null
    into pl_id, pl_name;

    suspend;

    main_pl_id = pl_id;
    for select pl_id, parent,
               iif(plt_short_name_side = -1, plt_name || ' ', '') ||
               pl_name ||
               lower(iif(plt_short_name_side = 1, ' '|| plt_name, ''))
    from places, placetype
    where parent = :main_pl_id and
          pl_status = plt_id
    order by pl_name, pl_id
    into pl_id, pl_parent_id, pl_name
    do
      Suspend;
  end else
  begin
    for select pl_id, parent,
               iif(plt_short_name_side = -1, plt_name || ' ', '') ||
               pl_name ||
               lower(iif(plt_short_name_side = 1, ' '|| plt_name, ''))
    from places, placetype
    where pl_status = plt_id and
          pl_name starting :find_text and
          plt_can_select = 0 and
          pl_id starting left(:db_id, 3)
    order by pl_name, pl_id
    into pl_id, pl_parent_id, pl_name
    do
    begin
      select iif(plt_short_name_side = -1, plt_name || ' ', '') ||
             pl_name ||
             lower(iif(plt_short_name_side = 1, ' '|| plt_name, ''))
      from places, placetype
      where pl_id = :pl_parent_id and      
            pl_status = plt_id
      into pl_parent_name;

      Suspend;
    end
  end

end
