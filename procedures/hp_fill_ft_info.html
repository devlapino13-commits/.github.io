CREATE PROCEDURE HP_FILL_FT_INFO(
    C_ID D_ID,
    REP_DATE D_DATE,
    REG_ID D_INTEGER,
    NEED_CREATE D_SMALLINT,
    UN_ID D_ID)
AS
declare variable FIF_ID D_ID;
declare variable FIF_FIELDNAME D_STR_CODE;
declare variable FIL_ID D_ID;
declare variable NEW_VALUE D_STR_LONG;
declare variable FIB_ID D_ID;
declare variable IS_EDITED D_SMALLINT;
begin
  if (c_id is null) then
    exception e_default 'Empty C_ID';

  if (rep_date is null) then
    exception e_default 'Empty Rep_date';

  -- на дату должен быть только один действующий бланк
  select fib_id
  from ft_infoblank
  where fib_date <= :rep_date and
        coalesce(fib_date_close, cast('31.12.9999' as date)) >= :rep_date and
        fib_state = 1 -- актуальный бланк
  into fib_id;

  if (fib_id is null) then
    exception e_default 'Actual FIB_ID not found';

  -- если анкета закрыта в день заполнения данных, то обратно открываем ее
  if (need_create = 1) then
  begin
    update ft_infolist
    set fil_date_close = null
    where fil_from_fib_id = :fib_id and
          fil_from_c_id = :c_id and
          fil_date_open <= :rep_date and
          fil_date_close = :rep_date;
  end

  -- по всем действующим анкетам заполняем поля
  select fil_id
  from ft_infolist
  where fil_from_fib_id = :fib_id and
        fil_from_c_id = :c_id and
        fil_date_open <= :rep_date and
        coalesce(fil_date_close, cast('31.12.9999' as date)) >= :rep_date
  into :fil_id;

  -- если нет действующей анкеты, создаем новую
  if (need_create = 1 and fil_id is null) then
  begin
    insert into ft_infolist (fil_from_fib_id, fil_from_c_id, fil_date_open, fil_last_un_id, fil_from_reg_id)
    values (:fib_id, :c_id, :rep_date, :un_id, :reg_id)
    returning (fil_id)
    into fil_id;
  end

  for select fif_id, fif_fieldname
      from ft_infofield
      where fif_from_fib_id = :fib_id and
            fif_date <= :rep_date and
            fif_is_delete = 0
      into :fif_id, fif_fieldname
  do
  begin
    new_value = '';
    select res_text, is_edited
    from hp_ft_fieldvalue_calc(:fil_id, :c_id, :fif_fieldname, :rep_date, 0)
    into new_value, is_edited;

    execute procedure hp_ft_fieldvalue_update(:fil_id, :c_id, :fif_id, :rep_date, :new_value, :un_id, :is_edited, '');
  end
end
