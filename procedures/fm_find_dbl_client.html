CREATE PROCEDURE FM_FIND_DBL_CLIENT(
    FIND_FIO VARCHAR(255),
    FIND_BIRTH_DATE DATE,
    FIND_CODE VARCHAR(20),
    FIND_PASS_DATE DATE,
    FIND_PASS_NO VARCHAR(100),
    UN_ID DECIMAL(18,0),
    FIND_DRL_ID ID_TYPE = 0,
    CITIZEN_ID INTEGER = 0)
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    IS_CRITICAL INTEGER,
    ACCESS_LEVEL INTEGER,
    ADD_DOC_TEXT D_STR_255)
AS
begin
  citizen_id = coalesce(citizen_id, 0);
  if (coalesce(find_drl_id, 0) <> 0) then
  begin
    find_fio = f_docrecog_value(find_drl_id, 'calc_name_full', '');
    find_birth_date = f_docrecog_value(find_drl_id, 'birth_date', null);
    find_code = f_docrecog_value(find_drl_id, 'inn', '');
    find_pass_no = f_docrecog_value(find_drl_id, 'doc_no', '');
    find_pass_date = f_docrecog_value(find_drl_id, 'date_begin', null);
    add_doc_text = ''; -- временно пустой. В будущем установить текст вопрос для добавления как документ к найденному дубликату.

    select pk_country_id
    from docrecoglist, passkind
    where drl_id = :find_drl_id and
          pk_id = drl_from_pk_id
    into citizen_id;
  end

  for select c_id, is_critical, c_name_fio, c_birth_date, c_pass_no
      from hp_find_clients(null, :find_fio, :find_birth_date, :find_code, :find_pass_date, :find_pass_no, '', 0, :citizen_id)
      where c_type = 0
      into c_id, is_critical, c_name_fio, c_birth_date, c_pass_no
  do
  begin
    select access_lev
    from fm_client_status(:un_id, :c_id)
    into access_level;

    if (char_length(c_pass_no) >= 5) then
      c_pass_no = left(c_pass_no, char_length(c_pass_no) - 5) || '***' || right(c_pass_no, 2);

    suspend;
  end
end
