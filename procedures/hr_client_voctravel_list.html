CREATE PROCEDURE HR_CLIENT_VOCTRAVEL_LIST(
    HO_ID DECIMAL(18,0),
    DD1 DATE,
    DD2 DATE,
    CL_ID DECIMAL(18,0),
    REG_ID INTEGER)
RETURNS (
    HRV_CLIENT_ID DECIMAL(18,0),
    HRV_DATE_BEGIN DATE,
    HRV_DATE_END DATE,
    HRV_ORDER_TYPE INTEGER,
    HRV_OFFICE_ID INTEGER)
AS
begin
  for select distinct hrv_from_c_id, hrv_date_begin, hrv_date_end_fact, ho_order_type, ho_from_reg_id
      from hr_voctravel, hr_orders, sa_worktype
      where hrv_from_ho_id <> :ho_id and
            hrv_from_ho_id = ho_id and
            hrv_date_begin <= :dd2 and
            hrv_date_end_fact >= :dd1 and
            hrv_from_c_id = :cl_id and
            hrv_status = 1 and
            ho_order_type = sawt_id and
            coalesce(sawt_hide_in_tabel, 0) = 0
      order by hrv_date_begin, ho_order_type
      into :hrv_client_id, :hrv_date_begin, :hrv_date_end, :hrv_order_type, :hrv_office_id
  do
  begin
    if (hrv_client_id is null) then
      hrv_client_id = 0;
    if (hrv_order_type is null) then
      hrv_order_type = 0;
    suspend;
  end
end
