CREATE PROCEDURE HP_CARD_REISSUE_CASH(
    CARD_ID D_INTEGER)
RETURNS (
    CAC_ID D_INTEGER,
    CCA_ID D_INTEGER,
    REASON D_STR_255,
    REG_NAME D_STR_CODE,
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    AMOUNT D_CURRENCY,
    STATUS D_STR_100)
AS
declare variable c_id d_id;
declare variable cac_status1 d_integer;
declare variable pay_amount d_currency;
begin
  cac_status1 = f_enum_id('cardaktclient', 'issued_delayed');
  pay_amount = 350.00;
  status = 'карта не определена';

  select first 1
    cac_id,
    cac_from_cca_id,
    cac_from_c_id,
    cac_amount,
    rd_name,
    reg_name,
    c_name_fio,
    k_kontract_no,
    'акт выдачи №' || cac_id
  from cardaktclient
    inner join refdata
      on rd_id = cac_from_rd_id
    inner join region
      on reg_unicode = cac_from_reg_id
    inner join clients
      on c_id = cac_from_c_id
    left join kontract
      on k_id = cac_from_k_id
  where cac_card_id = :card_id
    and cac_status = cac_status
    and cac_amount < :pay_amount
    and cac_from_edl_id is null
  order by cac_id desc
  into
    :cac_id,
    :cca_id,
    :c_id,
    :amount,
    :reason,
    :reg_name,
    :c_name_fio,
    :k_kontract_no,
    :status;

  if (cac_id > 0 and cca_id is null) then
    cca_id = (
      select
        max(cca_id)
      from cardclientid
        inner join cardclientacc
          on cca_from_cci_id = cci_id
            and cca_status = 0
      where cci_from_c_id = :c_id
        and cci_is_stopped = 0
        and cci_status = 1);

  suspend;
end
