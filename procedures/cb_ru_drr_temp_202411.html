CREATE PROCEDURE CB_RU_DRR_TEMP_202411(
    STAGE_MONTH D_SMALLINT,
    LO_ID D_ID,
    REG_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    LO_BALANCE_ALL D_CURRENCY,
    LO_BALANCE_OWN D_CURRENCY,
    OWN_BALANCE_PERC D_CURRENCY,
    LO_BONUS D_CURRENCY,
    RES D_SMALLINT,
    LO_BONUS_TOTAL_FACT D_CURRENCY,
    LO_BONUS_TOTAL D_CURRENCY)
AS
begin
  res = 0;
  if (stage_month not between 7 and 11) then
  begin
    suspend;
    exit;
  end
  --проверка доли портфеля
  lo_balance_all = null;
  lo_balance_own = null;
  select sum(oks_balance), sum(iif(ks_kc_id = oks_kc_id, oks_balance, 0)) from oksummary, kontstate
  where :rep_date_end between oks_date_begin and oks_date_end
  and oks_kc_id = :lo_id
  and oks_region_id = :reg_id
  and oks_close_state = 0
  and oks_from_k_id = ks_from_k_id
  and ks_ol_is_delete = 0
  and ks_state = 1
  and :rep_date_end between ks_date_begin and ks_date_end
  into lo_balance_all, lo_balance_own;
  lo_balance_all = coalesce(lo_balance_all, 0);
  lo_balance_own = coalesce(lo_balance_own, 0);

  own_balance_perc = iif(lo_balance_all > 0, lo_balance_own / lo_balance_all, 0) * 100;

  lo_bonus = 0;
  select  coalesce(aarep_bonus, 0), --maxvalue(coalesce(aarep_bonus_total_to_pay, 0), coalesce(aarep_bonus_alternative, 0))
          coalesce(aarep_bonus_total_fact, 0),
          coalesce(aarep_bonus_total, 0)
  from CB_RU_LO_20241201_ONE_LO_DISB(:reg_id, :lo_id, :rep_date_begin,:rep_date_end,0)
  into lo_bonus, lo_bonus_total_fact, lo_bonus_total;

--  if (own_balance_perc >= 80.0 and lo_bonus > 0.0) then
  if (lo_bonus_total >= 25000) then
  begin
    res = 1;
  end
  suspend;
end
