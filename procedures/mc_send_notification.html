CREATE PROCEDURE MC_SEND_NOTIFICATION(
    PHONE D_STR_CODE,
    PROJECT D_STR_CODE,
    PUSH_TITLE D_STR_CODE,
    PUSH_BODY D_STR_CODE)
RETURNS (
    RES_CODE D_STR_CODE)
AS
declare variable mcl_id d_id;
begin

    res_code = '';

    if (not exists
        (
            select 1
            from mc_test_access mta
            where mta.mta_phone_number = :phone    
        )
    ) then
    begin
        res_code = f_format('not a test phone: %1', :phone);
        suspend;
        exit;
    end

    select mcl.mcl_id
    from mc_login mcl
    where mcl.mcl_login = :phone
        and mcl.mcl_is_delete = 0
        and mcl.mcl_project = f_enum('mcl_project', :project, 'id')
    into :mcl_id;

    if (coalesce(:mcl_id, 0) = 0) then
    begin
        res_code = 'mcl_id was not found';
        suspend;
        exit;
    end

    res_code = 'send';
    execute procedure mc_send_push('mc_login', :mcl_id, :push_title, :push_body);

    suspend;
end
