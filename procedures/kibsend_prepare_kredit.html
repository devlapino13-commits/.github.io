CREATE PROCEDURE KIBSEND_PREPARE_KREDIT(
    REP_DATE DATE,
    REP_DELIQ_DAYS INTEGER)
RETURNS (
    BANK_CODE INTEGER,
    FIL_CODE INTEGER,
    UNIQ_CRED_NUM DECIMAL(18,0),
    CREDTYPE_CODE VARCHAR(20),
    CREDSTATEX_CODE VARCHAR(20),
    UNIQ_K_ID DECIMAL(18,0),
    POCKET_ID INTEGER,
    REPAYSTATUS INTEGER,
    GUARANTEE VARCHAR(100),
    COUNTOFDELINQUENCY INTEGER,
    UNPAIDAMOUNT DECIMAL(18,4),
    UNPAIDPERCENTAMOUNT DECIMAL(18,4),
    UNPAIDFINE DECIMAL(18,4),
    STAT INTEGER,
    FACT_AMOUNT DECIMAL(18,4),
    PLAN_AMOUNT DECIMAL(18,4),
    DELIQ_DAYS INTEGER,
    DELIQ_OS DECIMAL(18,4),
    DELIQ_PERC DECIMAL(18,4),
    PAY_FREQUENCY SMALLINT,
    PAYMETHOD SMALLINT,
    FINE DECIMAL(18,4),
    PAYOUTDAY SMALLINT,
    FACT_REPAYMENT_DATE DATE,
    OVERDUEPAYOUTNUMBER INTEGER,
    WRITE_OFF_OS DECIMAL(18,4),
    WRITE_OFF_PERC DECIMAL(18,4))
AS
declare variable pr_kind integer;
declare variable dd_start date;
declare variable max_day integer;
declare variable last_date date;
declare variable date_close date;
declare variable kol integer;
declare variable dd date;
declare variable pok_id integer;
declare variable deliq_amount decimal(18,4);
declare variable reg_id integer;
declare variable zad_os decimal(18,4);
declare variable zad_perc decimal(18,4);
declare variable sh_calc_fine decimal(18,4);
declare variable deliq_days_kolich integer;
declare variable disb_date date;
declare variable fact_amount_temp decimal(18,4);
declare variable ol_date date;
declare variable pr_wm_kolich integer;
declare variable pr_week_or_month smallint;
declare variable payoutday_buf smallint;
declare variable cu_id varchar(3);
declare variable ol_id_temp decimal(18,0);
declare variable max_countofdelinquency integer;
declare variable max_repaystatus integer;
declare variable sh_date date;
begin
  /*
    PURPOSE: Процедура заполняет кредитный пакет

    CHANGED BY IGUSEV 22.06.2012
    changes:  КИБ произвел изменения кодов статуса погашения.

                1    Платежи по графику
                10   до 30 дней просрочки
                13   Погашен за счет права уступки требования
                15   Погашен гарантом
                16   Погашен за счет обеспечения во внесудебном порядке
                17   Погашен через судебные органы (кредиты без залога)
                2    30 дней до 59 дней просрочки
                3    60 дней до 89 дней просрочки
                4    90 дней  до 119 дней просрочки
                5    120 дней просрочки и более
                8    Погашен за счет обеспечения в судебном порядке

             В соответст с этим, неактуальные стутусы в теле hp были убраны (закоментированны)
             и вместо них выставлены актуальные статусы.

    CHANGED BY IGUSEV 16.07.2012
    changes:  КИБ произвел изменения статусов кредита.
              На данный момент активные статусы:
              '1-0'  Рассмотрение заявки
              '2-0'  Выдан
              '3-0'  Отказано
              '4-0'  Погашен
              '5-0'  Просрочен
              '8-0'  Не погашен/списан
              '14-0' Реструктуризирован
              '15-0' Судебное разбирательство
              '16-0' Своевременный платеж

     CHANGED BY IGUSEV 23.07.2012
     changes: Добавлены следющие исходящие параметры:
              UNPAIDFINE decimal(18,4),    - остаточная сумма по пени
              FACT_AMOUNT decimal(18,4)    - фактическая сумма платежа
              PLAN_AMOUNT decimal(18,4)    - плановая сумма платежа
              COUNTOFDELINQUENCY -  кол-во просрочек
              OverduePayoutNumber - теперь обозначает НЕ порядковыйномер просроченной выплаты, а просто порядковый номер выплаты
              PAYMETHOD  - метод платежа (ануитет или нет)
              FINE - пеня
              PAYOUTDAY -  оговоренный день выплаты 
              PAY_FREQUENCY - частота пдлатежей (ежемесячно или нет)

     Передаю только следующие статусы : Выдан, Закрыт, Просрочки, Своевременные платежи.

     Предварительные заявки не передаю!!!

     Когда я прдаю кредиты наших сотрудников я не оиентируюсь на pr_send_to_kib=1 и pr_is_employer=1
     т.к. было выявлено что для большинства кредитов наших сотрудников, в продукте, эти параметры выставлены не верно.

  */

  delete from temp_kib_dk_list;

  if ((rep_deliq_days is null) or (rep_deliq_days = 0)) then
    rep_deliq_days = 30;
  countofdelinquency = 0;
  guarantee = ' *П* ; ';
  unpaidpercentamount = 0;
  overduepayoutnumber = 0;
  unpaidamount = 0;

  /* ==================== ВЫДАННЫЕ КРЕДИТЫ ============================================*/
  write_off_os = 0;
  write_off_perc = 0;

  stat = 1;/* Выданные кредиты тип/ - 1 */
  credstatex_code = '2-0';/* Выданные кредиты */
  select distinct kp_date
  from kibpocket
  where kp_status = :stat
  order by kp_date desc
  rows 1
  into :dd_start;
  if (dd_start is null) then
    dd_start = :rep_date - 1;

  pocket_id = gen_id(gen_kibpocket_id, 1);
  insert into kibpocket (kp_id, kp_status, kp_date, kp_memo, kp_create_time)
  values (:pocket_id, :stat, :rep_date, 'Выданные кредиты', cast('NOW' as timestamp));

  for select cm_kib_bank_code, reg_kib_code, k_id, pr_loan_type, pr_interest_method, k_disb_date, pr_wm_kolich,
             pr_week_or_month
      from (select cm_kib_bank_code, reg_kib_code, k_id, pr_loan_type, pr_interest_method, k_disb_date, pr_wm_kolich,
                   pr_week_or_month
            from (
            /**********  Выданные кредиты НЕ наших сотрудников ************/
            select distinct cm_kib_bank_code, reg_kib_code, k_id, pr_loan_type, pr_interest_method, k_disb_date,
                            prod.pr_wm_kolich, prod.pr_week_or_month
            from kontract, knclient kncl, clients cl, knaktlist, kontstate, region, company, product prod
            where kncl.kc_from_k_id = k_id and
                  cl.c_id = kncl.kc_from_c_id and
                  lower(cl.c_name_fio) not like '%client%' and
                  k_from_kf_id = kal_from_kf_id and
                  kal_date <= k_disb_date and
                  kal_date_end >= k_disb_date and
                  kal_new_reg_id = reg_unicode and
                  ks_from_k_id = k_id and
                  ks_date_begin between(:dd_start + 1) and :rep_date and
                  ks_date_begin <= ks_date_end and
                  ks_state = :stat and
                  ks_ol_is_delete = 0 and
                  k_from_pr_id = pr_id and
                  pr_send_to_kib = 1 and
                  pr_is_employer <> 1) m) a
      union all
      select cm_kib_bank_code, reg_kib_code, k_id, pr_loan_type, pr_interest_method, k_disb_date, pr_wm_kolich,
             pr_week_or_month
      from (/**********  Выданные кредиты наших сотрудников ************/
      select distinct cm_kib_bank_code, reg_kib_code, k_id, pr_loan_type, pr_interest_method, kont.k_disb_date,
                      pr_wm_kolich, pr_week_or_month
      from kontract kont, knclient kncl, clients cl, knaktlist, kontstate, region, company, product prod
      where kncl.kc_from_k_id = k_id and
            cl.c_id = kncl.kc_from_c_id and
            lower(cl.c_name_fio) like '%client%' and
            k_from_kf_id = kal_from_kf_id and
            kal_date <= k_disb_date and
            kal_date_end >= k_disb_date and
            kal_new_reg_id = reg_unicode and
            ks_from_k_id = k_id and
            ks_date_begin between(:dd_start + 1) and :rep_date and
            ks_date_begin <= ks_date_end and
            ks_state = :stat and
            ks_ol_is_delete = 0 and
            k_from_pr_id = pr_id
            /*and pr_send_to_kib=1
      and pr_is_employer=1 */
      ) b
      into :bank_code, :fil_code, :uniq_cred_num, :pr_kind, :paymethod, :disb_date, :pr_wm_kolich, :pr_week_or_month
  do
  begin
    uniq_k_id = :uniq_cred_num;
    if (pr_kind = 1) then
      credtype_code = '7';
    if (pr_kind = 0) then
      credtype_code = '3';

    unpaidamount = 0;
    unpaidpercentamount = 0;
    unpaidfine = 0;
    /*---------- рассчитываю остати (сколько клиенту осталось заплатить) по ОС, %, пеня ---------------------------*/
    for select zad_os, zad_perc, sh_calc_fine
        from aarep_make_zad_list(:uniq_cred_num, :disb_date, 1)
        where sh_date is not null
        into :zad_os, :zad_perc, :sh_calc_fine
    do
    begin
      unpaidamount = unpaidamount + zad_os;
      unpaidpercentamount = unpaidpercentamount + zad_perc;
      unpaidfine = unpaidfine + sh_calc_fine;
    end
    /*--------------------------------------------------------------------------------------------------------------*/
    fact_amount = 0;
    /* -------- частота выплат ------------------------------------------------------------------------*/
    if (pr_wm_kolich = 1 and
        pr_week_or_month = 1) then
      pay_frequency = 2;
    else
    /* еженедельно */
    if (pr_wm_kolich = 2 and
        pr_week_or_month = 1) then
      pay_frequency = 3;
    else
    /* раз в две недели */
    if (pr_wm_kolich = 1 and
        pr_week_or_month = 2) then
      pay_frequency = 4;
    else
    /* ежемесячно */
    if (pr_wm_kolich = 2 and
        pr_week_or_month = 2) then
      pay_frequency = 5;
    else
    /* раз в два месяца */
    if (pr_wm_kolich = 3 and
        pr_week_or_month = 2) then
      pay_frequency = 6;
    else
    /* поквартально */
    if (pr_wm_kolich = 4 and
        pr_week_or_month = 2) then
      pay_frequency = 7;
    else
    /* три раза год */
    if (pr_wm_kolich = 6 and
        pr_week_or_month = 2) then
      pay_frequency = 7;
    else
    /* раз в полгода */
    if (pr_wm_kolich = 12 and
        pr_week_or_month = 2) then
      pay_frequency = 9;
    else
      /* ежегодно */
      pay_frequency = 10;/*другое*/
    /*-------------------------------------------------------------------------------------------------*/
    if (paymethod = 2) then
      /* ес в нашей БД метод Ануитет */
      paymethod = 1;/* тогда и для КИБ-а я выставляю метод Ануитет*/
    else
      /* в противном случае выставляю метод "Другое"*/
      paymethod = 4;
    /*------------------------------------------------------------------------*/
    plan_amount = 0;/* оговоренная сумма платежа. не передаю т.к. для выданных кредитов это поле не обязат.*/
    /*------------------------------------------------------------------------*/
    /*---------- оговоренный день выплаты ------------------------------------*/
    payoutday = 0;/* не передаю т.к для выданных кредитов это поле не обязательное*/
    /*------------------------------------------------------------------------*/
    deliq_days = 0;
    deliq_os = 0;
    deliq_perc = 0;
    fine = 0;
    countofdelinquency = 0;
    repaystatus = 1;
    fact_repayment_date = null;
    overduepayoutnumber = 0;
    suspend;
  end
  /****************************************************************************/

  stat = 2;/* Закрытые кредиты тип - 2 */

  select distinct kp_date
  from kibpocket
  where kp_status = :stat
  order by kp_date desc
  rows 1
  into :dd_start;
  if (dd_start is null) then
    dd_start = :rep_date - 1;

  pocket_id = gen_id(gen_kibpocket_id, 1);
  insert into kibpocket (kp_id, kp_status, kp_date, kp_memo, kp_create_time)
  values (:pocket_id, :stat, :rep_date, 'Закрытые кредиты', cast('NOW' as timestamp));

  for select cm_kib_bank_code, reg_kib_code, k_id, pr_loan_type, k_kredline_end, pr_interest_method, k_disb_date,
             pr_wm_kolich, pr_week_or_month, ks_date_begin
      from (select cm_kib_bank_code, reg_kib_code, k_id, pr_loan_type, k_kredline_end, pr_interest_method, k_disb_date,
                   pr_wm_kolich, pr_week_or_month, ks_date_begin
            from (
            /**********  закрытые кредиты НЕ наших сотрудников ************/
            select distinct cm_kib_bank_code, reg_kib_code, k_id, pr_loan_type, k_kredline_end, pr_interest_method,
                            kont.k_disb_date /*дата выдачи*/ , pr_wm_kolich, pr_week_or_month, ks_date_begin
            from kontract kont, knclient kncl, clients cl, knaktlist, kontstate ks, region, company, product
            where kncl.kc_from_k_id = k_id and
                  cl.c_id = kncl.kc_from_c_id and
                  lower(cl.c_name_fio) not like '%client%' and
                  k_from_kf_id = kal_from_kf_id and
                  kal_date <= k_disb_date and
                  kal_date_end >= k_disb_date and
                  kal_new_reg_id = reg_unicode and
                  ks_from_k_id = k_id and
                  ks_date_begin between(:dd_start + 1) and :rep_date and
                  ks_date_begin <= ks_date_end and
                  ks_state = :stat and
                  ks_ol_is_delete = 0 and
                  k_from_pr_id = pr_id and
                  pr_send_to_kib = 1 and
                  pr_is_employer <> 1) m) a
      union all
      select cm_kib_bank_code, reg_kib_code, k_id, pr_loan_type, k_kredline_end, pr_interest_method, k_disb_date,
             pr_wm_kolich, pr_week_or_month, ks_date_begin
      from (
      /**********  закрытые кредиты наших сотрудников ************/
      select distinct cm_kib_bank_code, reg_kib_code, k_id, pr_loan_type, k_kredline_end, pr_interest_method,
                      kont.k_disb_date /*дата выдачи*/ , pr_wm_kolich, pr_week_or_month, ks_date_begin
      from kontract kont, knclient kncl, clients cl, knaktlist, kontstate, region, company, product
      where kncl.kc_from_k_id = k_id and
            cl.c_id = kncl.kc_from_c_id and
            lower(cl.c_name_fio) like '%client%' and
            k_from_kf_id = kal_from_kf_id and
            kal_date <= k_disb_date and
            kal_date_end >= k_disb_date and
            kal_new_reg_id = reg_unicode and
            ks_from_k_id = k_id and
            ks_date_begin between(:dd_start + 1) and :rep_date and
            ks_date_begin <= ks_date_end and
            ks_state = :stat and
            ks_ol_is_delete = 0 and
            k_from_pr_id = pr_id
            /* and pr_send_to_kib=1
         and pr_is_employer=1 */
      ) b
      into :bank_code, :fil_code, :uniq_cred_num, :pr_kind, :date_close, :paymethod, :disb_date, :pr_wm_kolich,
           pr_week_or_month, fact_repayment_date /* фактическая дата закрытия кредита*/
  do
  begin
    write_off_os = 0;
    write_off_perc = 0;
    /* Проверяю, был ли контракт списан последней опирацией */
    for select ks.ks_from_ol_id
        from kontstate ks
        where --:REP_DATE  between ks.ks_date_begin and ks.ks_date_end  and
              ks.ks_from_k_id = :uniq_cred_num and
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 2
        into :ol_id_temp
    do
    begin
      select -sum(ok.ok_os), -sum(ok.ok_perc)
      from operkred ok
      where ok.ok_from_ol_id = :ol_id_temp and
            ok.ok_from_k_id = :uniq_cred_num and
            ok.ok_type = 5 and
            ok_from_c_id = 0
      into :write_off_os, :write_off_perc;
    end
    /* Под списанные, здесь попадают контракты,
       по которым было списание по ОС последней операцией.
       Если списали только % и пеню, контракт не считается списанным
    */
    write_off_os = coalesce(write_off_os, 0);
    write_off_perc = coalesce(write_off_perc, 0);

    credstatex_code = '4-0';
    if (write_off_os > 0) then
      credstatex_code = '8-0';/*было списание по ОС*/
    else
      write_off_perc = 0;/*если нет списания по ОС, проценты не передаю */

    repaystatus = 1;
    max_repaystatus = 1;

    uniq_k_id = :uniq_cred_num;
    if (pr_kind = 1) then
      credtype_code = '7';
    if (pr_kind = 0) then
      credtype_code = '3';
--CredStatEx_Code='4-0';

    max_day = 0;
    /*максимально просроченные дни и просроченная сумма*/
    select max(ok_ol_date - ok_sh_date), sum(ok_os + ok_perc)
    from operkred
    where ok_ol_is_active = 1 and
          ok_from_k_id = :uniq_cred_num and
          ok_type = 4 and
          ok_ol_date > ok_sh_date and
          ok_ol_date <= :rep_date and
          ok_from_c_id = 0 and
          ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
          and
          ok_os + ok_perc <> 0 /*  !!! Иначе ты можешь поймать погашения просроченной пени (отложенные пени, или по новому руководству, пеня берется после погашения ОС и процентов). */
    into :max_day, :deliq_amount;
    if (max_day is null) then
      max_day = 0;
    if (deliq_amount is null) then
      deliq_amount = 0;

    countofdelinquency = 0;/*число просрочек*/

    /* ВАЖНО! КИБ ПРОИЗВЕЛ ИЗМЕНЕНИЯ ТЕПЕРЬ overduepayoutnumber - ЭТО НЕ ПОРЯДКОВЫЙ НОМЕР ПРОСРОЧЕННОЙ ВЫПЛАТЫ, А ПРОСТО ПОРЯДКОВЫЙ НОМЕР ВЫПЛАТЫ*/
    overduepayoutnumber = 0;

    for select distinct ok.ok_ol_date
        from operkred ok
        where ok.ok_type = 4 and
              ok.ok_ol_is_active = 1 and
              ok.ok_from_k_id = :uniq_cred_num and
              ok.ok_ol_date <= :rep_date and
              ok.ok_kommis = 0 /*не комиссия*/
              and
              ok_from_c_id = 0
        group by ok.ok_ol_date
        into :dd
    do
    begin
      overduepayoutnumber = overduepayoutnumber + 1;/*порядковый номер выплаты, рассчитываю как общее число выплат до этого момента*/
    end

    deliq_days = 0;
    if (deliq_amount < -10) then
    begin
      for select distinct ok_sh_date
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_k_id = :uniq_cred_num and
                ok_type = 4 and
                ok_ol_date > ok_sh_date and
                ok_ol_date <= :rep_date and
                ok_from_c_id = 0
          into :dd
      do
      begin
        countofdelinquency = countofdelinquency + 1;/*число просрочек*/
      end

      max_countofdelinquency = null;
      max_repaystatus = null;

      select max(kibkr.kkr_countofdelinquency), max(kibkr.kkr_repaystatus)
      from kibkredit kibkr
      where kibkr.kkr_from_k_id = :uniq_cred_num and
            kibkr.kkr_transid > 1 -- важно!
      into :max_countofdelinquency, :max_repaystatus;

      max_countofdelinquency = coalesce(max_countofdelinquency, 0);
      max_repaystatus = coalesce(max_repaystatus, 0);

      if (countofdelinquency < max_countofdelinquency) then
        countofdelinquency = :max_countofdelinquency;

      last_date = null;
      select ok_ol_date
      from operkred
      where ok_ol_is_active = 1 and
            ok_from_k_id = :uniq_cred_num and
            ok_type = 4 and
            ok_ol_date > ok_sh_date and
            ok_from_c_id = 0
      order by ok_from_k_id, ok_ol_date desc
      rows 1
      into :last_date;

      if (last_date > :date_close) then
      begin
        kol = (last_date - date_close);
        deliq_days = kol;

      order by ok_from_k_id desc, ok_sh_date desc
      rows 1
      into :ol_date;

      /* ------------оговоренный день выплаты --------------------------*/
      payoutday = null;
      select extract(day from sh_date)
      from aarep_make_zad_list(:uniq_cred_num, :ol_date - 1, 1)
      where sh_date is not null rows 1
      into :payoutday;

      /* если клиент произвел первый платеж по кредиту в тот-же день когда и взял кредито тогда
           на дату :ol_date - 1   ничего не вернется, нужно исходить из даты ol_date
        */

      if (payoutday is null) then
        select extract(day from sh_date)
        from aarep_make_zad_list(:uniq_cred_num, :ol_date, 1)
        where sh_date is not null rows 1
        into :payoutday;
      /*----------------------------------------------------------------*/

      select -sum(ok.ok_os + ok.ok_perc + ok.ok_fine + ok.ok_tax)
      from operkred ok
      where ok.ok_type in (4, 9) and
            ok.ok_ol_is_active = 1 and
            ok.ok_from_k_id = :uniq_cred_num and
            ok.ok_ol_date <= :rep_date and
            ok.ok_kommis = 0 /*не комиссия*/
            and
            ok_from_c_id = 0
      group by ok.ok_ol_date
      order by ok.ok_ol_date desc
      rows 1
      into :fact_amount;
      fact_amount = coalesce(fact_amount, 0);
      /*----------------------------------------------------------------------*/
      /* частота выплат ---------------------------------------------------------------------------------*/
      if (pr_wm_kolich = 1 and
          pr_week_or_month = 1) then
        pay_frequency = 2;
      else
      /* единовременно */
      if (pr_wm_kolich = 2 and
          pr_week_or_month = 1) then
        pay_frequency = 3;
      else
      /* раз в две недели */
      if (pr_wm_kolich = 1 and
          pr_week_or_month = 2) then
        pay_frequency = 4;
      else
      /* ежемесячно */
      if (pr_wm_kolich = 2 and
          pr_week_or_month = 2) then
        pay_frequency = 5;
      else
      /* раз в два месяца */
      if (pr_wm_kolich = 3 and
          pr_week_or_month = 2) then
        pay_frequency = 6;
      else
      /* поквартально */
      if (pr_wm_kolich = 4 and
          pr_week_or_month = 2) then
        pay_frequency = 7;
      else
      /* три раза год */
      if (pr_wm_kolich = 6 and
          pr_week_or_month = 2) then
        pay_frequency = 7;
      else
      /* раз в полгода */
      if (pr_wm_kolich = 12 and
          pr_week_or_month = 2) then
        pay_frequency = 9;
      else
        /* ежегодно */
        pay_frequency = 10;
      /*-------------------------------------------------------------------------------------------------*/

      if (paymethod = 2) then
        /* ес в нашей БД метод Ануитет */
        paymethod = 1;/* тогда и для КИБ-а я выставляю метод Ануитет*/
      else
        /* в противном случае выставляю метод "Другое"*/
        paymethod = 4;

      deliq_os = 0;
      deliq_perc = 0;
      fine = 0;
      repaystatus = 1;

      if (repaystatus < max_repaystatus) then
        repaystatus = max_repaystatus;

      fact_repayment_date = null;/*фактическая дата закрытия кредита*/
      suspend;
    end
  end

  countofdelinquency = 0;
  guarantee = ' *П* ; ';
  unpaidpercentamount = 0;
  unpaidamount = 0;
  deliq_os = 0;
  deliq_perc = 0;
  overduepayoutnumber = 0;
