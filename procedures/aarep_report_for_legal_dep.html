CREATE PROCEDURE AAREP_REPORT_FOR_LEGAL_DEP(
    REP_DATE D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_LEGAL_ADDRESS D_STR_255,
    K_KONTRACT_NO D_STR_20,
    K_PERCENT PERC_TYPE,
    KAL_NEW_KF_NAME D_STR_100,
    K_DISB_DATE D_DATE,
    PR_NAME D_STR_64,
    SBA_DATE_BEGIN D_DATE,
    ZAD_OS D_CURRENCY,
    ZAD_PERC D_CURRENCY,
    ZAD_SUM D_CURRENCY)
AS
declare variable SBL_ID D_INTEGER;
declare variable K_ID ID_TYPE;
declare variable C_ID ID_TYPE;
begin
  select sbl.sbl_id
  from sb_list sbl
  where sbl.sbl_name = 'Юридичекий Отдел Без ПП'
  into :sbl_id;

  for select sba.sba_from_k_id,
             sba.sba_date_begin
      from sb_akt sba
      where sba.sba_from_sbl_id = :sbl_id and
            :rep_date between sba.sba_date_begin and sba.sba_date_end and
            sba.sba_status = 2
  into :k_id,
       :sba_date_begin
  do
  begin
    select kn.k_kontract_no,
           kn.k_percent,
           kn.k_disb_date,
           kal.kal_new_kf_name,
           pr.pr_name
    from kontract kn
    inner join knaktlist kal on kal.kal_from_kf_id = kn.k_from_kf_id
    inner join product pr on pr.pr_id = kn.k_from_pr_id
    where kn.k_id = :k_id and
          kal.is_delete = 0 and
          :rep_date between kal.kal_date and kal.kal_date_end
    into :k_kontract_no,
         :k_percent,
         :k_disb_date,
         :kal_new_kf_name,
         :pr_name;

    select sum(zad_os),
           sum(zad_perc),
           sum(zad_os + zad_perc)
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    into :zad_os,
         :zad_perc,
         :zad_sum;

    for select kc.kc_from_c_id
        from knclient kc
        where kc.kc_from_k_id = :k_id and
              kc.kc_amount_approved > 0
    into :c_id
    do
    begin
      select cl.c_name_fio,
             adl.adl_address_str
      from addrlist adl
      inner join addrclient adc on  adc.adc_from_adl_id = adl.adl_id
      inner join clients cl on cl.c_id = adc.adc_from_c_id
      where adc.adc_from_c_id = :c_id and
            adc.adc_address_type = 1
      into :c_name_fio,
           :c_legal_address;

      suspend;
    end
  end
end
