CREATE PROCEDURE HP_FERTILIZERPRICE_VIEW(
    AG_ID D_INTEGER,
    FEG_ID D_INTEGER)
RETURNS (
    FPR_ID D_INTEGER,
    FPR_FROM_AG_ID D_INTEGER,
    FPR_FROM_FEG D_INTEGER,
    FPR_PRICE_NORTH D_CURRENCY,
    FPR_PRICE_SOUTH D_CURRENCY,
    FPR_DT_BEGIN D_DATE,
    FPR_DT_END D_DATE,
    FPR_DATETIME D_DATETIME,
    FPR_FROM_UN_ID D_ID,
    UN_FIO D_STR_100)
AS
begin
  for
    select
      fpr_id,
      fpr_from_ag_id,
      fpr_from_feg,
      fpr_price_north,
      fpr_price_south,
      fpr_dt_begin,
      fpr_dt_end,
      fpr_datetime,
      fpr_from_un_id,
      un_fio
    from fertilizerprice
      inner join username
        on un_id = fpr_from_un_id
    where fpr_from_ag_id = :ag_id
      and fpr_from_feg = :feg_id
  into
    :fpr_id,
    :fpr_from_ag_id,
    :fpr_from_feg,
    :fpr_price_north,
    :fpr_price_south,
    :fpr_dt_begin,
    :fpr_dt_end,
    :fpr_datetime,
    :fpr_from_un_id,
    :un_fio
  do
  begin
    if (fpr_dt_end = '31.12.9999') then
      fpr_dt_end = null;
    suspend;
  end
end
