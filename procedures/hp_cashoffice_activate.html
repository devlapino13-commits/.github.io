CREATE PROCEDURE HP_CASHOFFICE_ACTIVATE(
    OFFICE_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable oper_kind d_integer;
declare variable ol_id d_id;
declare variable cash_id d_integer;
declare variable c_id d_id;
declare variable csp_cashier_id d_id;
begin
  res_is_error = 1;
  res_text = '';

  oper_kind = f_enum_id('cashoperkind', 'check.close');

  /*
  if (exists(select * from opercash
             where opc_date = :rep_date and
                   opc_office_id = :office_id and
                   opc_oper_kind <> :oper_kind)) then
    res_text = 'В офисе на указанную дату уже существуют кассовые операции';

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end
  */

  res_is_error = 0;

  c_id = uxf$usr_cid();

  insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo, ol_oper_subtype)
  values (:office_id, 2, :rep_date - 1, 0, '0', 1, 'Остатки перед активацией РМК', 1)
  returning ol_id
  into :ol_id;

  for select csp_id, csp_cashier_id
  from cashplace
  where csp_office_id = :office_id
  into :cash_id, csp_cashier_id
  do
  begin
    if (csp_cashier_id is null) then
      csp_cashier_id = c_id;

    insert into opercash (opc_date, opc_office_id, opc_from_csp_id, opc_oper_kind, opc_sign, opc_status, opc_is_signed,
                          opc_from_ol_id, opc_cashier_id, opc_memo)
    values (:rep_date - 1, :office_id, :cash_id, :oper_kind, 0, 1, 1,
            :ol_id, :csp_cashier_id, 'Остатки перед активацией РМК');

    insert into cashtotal (ctt_from_csp_id, ctt_date, ctt_cashier_id, ctt_need_recalc)
    values (:cash_id, :rep_date - 1, :csp_cashier_id, 1);
  end

  update cashoffice
  set cso_is_active = 1,
      cso_activate_date = :rep_date
  where cso_office_id = :office_id and
        (cso_is_active = 0 or cso_activate_date is distinct from :rep_date);

  suspend;
end
