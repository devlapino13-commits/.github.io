CREATE PROCEDURE HP_CANCEL_MINIAPP_BKI(
    REP_DATE DATE)
AS
declare variable MA_ID integer;
declare variable K_ID bigint;
declare variable IS_DISB smallint;
declare variable EDS_STATUS_DATE date;
declare variable KOL_DAY_APP smallint;
declare variable MSG_DATE date;
declare variable MSG_DATE_END date;
begin
  /*
  задача RFCRU-2466
  2. Если с момента подписания клиентом Заявления на займ прошло 30 дней и клиенту не была направлена ссылка с договором на подпись,
  то Заявление/заявка аннулируется без возможности дальнейшей работы с ним вне зависимости от других параметров, например, утверждение КК
  или менеджером и т.п. Ориентируемся только на факт отправки договора на подпись клиенту.

  3. По истечении 5 (пяти) рабочих дней после отправки первой ссылки на подписание договора по заявке и если займ не выдан,
  то Заявление/Заявка аннулируется без возможности дальнейшей работы с ним.
  */
  -- находим все активные заявки по которым еще выдачи не произведены
  for
  select ma_id from miniapp
  where ma_date_app <= :rep_date
  and ma_date_app >= '21.10.2024'  -- до этой даты не имеются заявки которые попадают под данное условие
  and ma_date_end = '31.12.9999'
  and ma_is_active = 1
  into ma_id
  do
  begin
    k_id = null;
    is_disb = 0;
    for
    select k_id from kontract
    where k_from_ma_app_id = :ma_id
    into k_id
    do
    begin
      if (exists(select * from kontstate
                   where ks_from_k_id = :k_id
                   and ks_ol_is_delete = 0
                   and ks_state in (1,2)
                   and :rep_date between ks_date_begin and ks_date_end))
      then is_disb = 1;
    end

    if (is_disb = 0) then
    begin

      eds_status_date = null;  -- дата подписьм заявлений
      select eds_status_datetime
      from edoc_sign, edoc_list, edoc_msg, fsrobot_msgbox
      where eds_from_edl_id = edl_id
      and edm_from_eds_id = eds_id
      and msg_id = edm_from_msg_id
      and edl_from_edc_id = f_edoc_kind('loan.application','id')
      and edl_object_kind = 11
      and edl_object_id = :ma_id
      and eds_status = 1
      and edm_status = 1
      and edl_status = 1
      into eds_status_date;

      kol_day_app = rep_date - eds_status_date;

      msg_date = null;   -- дата первой отправки договора
      select min(cast(msg_datetime as date))
      from edoc_sign, edoc_list, edoc_msg, fsrobot_msgbox
      where eds_from_edl_id = edl_id
      and edm_from_eds_id = eds_id
      and msg_id = edm_from_msg_id
      and edl_from_edc_id = f_edoc_kind('loan.agreement','id')
      and edl_object_kind = 9
      and edl_object_id = :k_id
      into msg_date;

      msg_date_end = f_date_work_nday(:msg_date, 5);


      if (kol_day_app > 30 and msg_date is null) then
      begin
        if (k_id is not null) then
        begin
          update kontract set k_status = -1
          where k_id = :k_id;
        end else
        begin
          execute procedure hp_miniapp_status_change(:ma_id, 1000000000000, 0);
        end
      end else
      if (msg_date is not null) then
      begin
        if (rep_date > msg_date_end) then -- если прошло 5 рабочих дней
        begin
          -- исключение на период охлаждения
          if (not exists(select 1 from hp_get_freeze_data(:k_id)
            where freeze_datetime_end > current_timestamp))
          then
            update kontract set k_status = -1
            where k_id = :k_id;
        end 
      end
    end
  end
end
