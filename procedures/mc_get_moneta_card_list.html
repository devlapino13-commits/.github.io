CREATE PROCEDURE MC_GET_MONETA_CARD_LIST(
    MCL_ID D_ID)
RETURNS (
    OPER_ID D_STR_CODE,
    CARD_NUMBER D_STR_CODE)
AS
declare variable c_id d_id;
declare variable card_prv d_str_code;
declare variable esm_id d_id;
begin
  oper_id = '';

  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into :c_id;

  if (c_id is not null) then
  begin
    card_prv = '';
    for with tab
        as (select esm_id,
                   (select field_value
                    from hp_extsys_parcer(esm_full_source, 'cardnumber')) as card_number
            from extsysmain
            where esm_project = 'moneta' and
                  esm_action = 'pay' and
                  esm_status in (0, 1) and
                  esm_object_kind = 0 and
                  esm_object_id = :c_id and
                  esm_date >= dateadd(year, -2, current_date))
        select card_number, max(esm_id)
        from tab
        group by 1
        order by 2 desc
        rows 5
        into card_number, esm_id
    do
    begin
      oper_id = '';
      select (select field_value
              from hp_extsys_parcer(esm_full_source, 'mnt_operation_id'))
      from extsysmain
      where esm_id = :esm_id
      into oper_id;
      if (oper_id is null) then
        oper_id = '';

      if (oper_id <> '' and
          card_number <> '') then
        suspend;
    end
  end

  oper_id = '';
  card_number = 'Использовать новую карту';
  suspend;
end
