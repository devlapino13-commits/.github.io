CREATE PROCEDURE FM_UNIREF_LIST(
    UN_ID D_ID,
    M_OBJ_KIND D_STR_255,
    M_OBJ_ID D_ID,
    M_REF_KIND D_STR_255,
    M_REF_ADD_KIND D_STR_255,
    M_FIND_TXT D_STR_255,
    M_ADD_PARAMS D_BLOB_TEXT = '')
RETURNS (
    TEXT1 D_STR_255,
    TEXT2 D_STR_255,
    MEMO1 D_STR_255,
    MEMO2 D_STR_255,
    OBJ_KIND D_STR_255,
    OBJ_ID D_ID,
    ADD_PARAMS D_BLOB_TEXT)
AS
declare variable T D_INTEGER;
declare variable TMP_STR D_STR_255;
declare variable JSON_TMP D_BLOB_TEXT;
declare variable FIND_TXT_LAT D_STR_255;
declare variable FIND_TXT_CYR D_STR_255;
declare variable DB_ID D_ID;
declare variable ARRAY_SIZE D_INTEGER;
declare variable ARRAY_IDX D_INTEGER;
begin
  db_id = f_get_db_id();
  m_find_txt = coalesce(m_find_txt, '');
  if (m_add_params is null) then
    m_add_params = '';
  add_params = '';

  if (m_ref_kind = 'ClientDocuments') then
  begin
    -- Возвращает орган выдавший паспорт
    obj_kind = 'RefData';
    if (m_obj_kind = 'Clients') then
    begin
      if (m_find_txt <> '') then
      begin
        for select rd_id, '<b>' || rd_str_id || '</b>', rd_name
            from find_refdata(:m_find_txt, '', '', :m_ref_add_kind, 1, 0)
            into obj_id, memo1, text1
        do
          suspend;
      end
      else
      begin
        -- Просматриваем предыдущие органы выдавшие документ
        for select distinct cds_doc_issued_by
            from passkind, clientdocs
            where cds_from_pk_id = pk_id and
                  cds_from_c_id = :m_obj_id and
                  pk_passport_issuers = :m_ref_add_kind and
                  coalesce(cds_doc_issued_by, '') <> ''
            into tmp_str
        do
        begin
          for select rd_id, '<b>' || rd_str_id || '</b>', rd_name
              from refdata
              where rd_main_code = :m_ref_add_kind and
                    rd_name = :tmp_str
              into obj_id, memo1, text1
          do
            suspend;
        end
      end
    end
  end -- m_kind = 'PassportIssuers'

  if (m_ref_kind = 'StreetList') then
  begin
    obj_kind = 'Places';
    t = 0;
    for select street_name, pl_name, full_text, street_id, street_name_orig,
      '{' ||
      f_json_make_pair('street_kind', stree_kind, 'str') || ',' ||
      f_json_make_pair('street_name', street_name_orig, 'str') || ',' ||
      f_json_make_pair('post_index', coalesce(iif(pl_post_index = '0', null, pl_post_index), ''), 'str') || ',' ||
      f_json_make_pair('pl_id', pl_id, 'str') ||
      '}'
    from hp_find_places(:m_obj_id, '', :m_find_txt)
    rows 50
    into text1, text2, memo1, obj_id, tmp_str, add_params
    do
    begin
      suspend;
      if (tmp_str = m_find_txt) then
        t = 1;
    end

    -- Если нет точных совпадений, даем возможность добавить
    if (m_find_txt <> '' and
        t = 0) then
    begin
      select '<b>Добавить улицу: </b>' || :m_find_txt,
             case plt_short_name_side
               when -1 then plt_short_name || ' ' || pl_name
               when 0 then pl_name
               when 1 then pl_name || ' ' || plt_short_name
             end,
             '', -1,
             '{' ||
             f_json_make_pair('street_name', :m_find_txt, 'str') || ',' ||
             f_json_make_pair('post_index', coalesce(iif(pl_post_index = '0', null, pl_post_index), ''), 'str') || ',' ||
             f_json_make_pair('pl_id', pl_id, 'str') ||
             '}'

      from places, placetype
      where pl_id = :m_obj_id and
            pl_status = plt_id
      into text1, text2, memo1, obj_id, add_params;
      text2 = replace(text2, '. ', '.'); -- Также, как в hp_find_places
      suspend;
    end
  end

  if (m_ref_kind = 'AppClientList') then
  begin
    obj_kind = 0;
    for select c_id, c_name_fio, f_datetostr(c_birth_date), c_pass_no
    from fm_find_client(:m_find_txt, :un_id)
    into obj_id, text1, text2, memo2
    do
      suspend;
  end

  if (m_ref_kind = 'AppProductList') then
  begin
    obj_kind = 19;
    text2 = '';
    memo2 = '';
    select ma_result_txt
    from fm_miniapp_action(:un_id, 'getproductlist', :m_add_params)
    into json_tmp;

    array_size = f_json_blob_get_count(json_tmp, 'product');
    array_idx = 0;
    while (array_idx < array_size) do
    begin
      add_params = f_json_blob_get_value(json_tmp, f_format('product[%1]', array_idx), null);
      obj_id = f_json_str_get_value(add_params, 'id', null);
      text1 = f_json_str_get_value(add_params, 'name', null);
      memo1 = f_json_str_get_value(add_params, 'memo', null);
      add_params = '{"product":' || add_params || '}';
      if (db_id in (39801, 41701)) then
      begin
        text1 = memo1;
        memo1 = '';
      end
      if (text1 containing m_find_txt or memo1 containing m_find_txt ) then
        suspend;
      array_idx = array_idx + 1;
    end

    if (array_size <= 0) then
    begin
      json_tmp = '';
      text2 = '';
      memo1 = '';
      memo2 = '';
      add_params = '';
      obj_id = -1;
      text1 = 'Нет доступных продуктов';
      text2 = '';
      memo1 = 'Проверьте условия в заявке';
      memo2 = '';
      suspend;
    end
    exit;
  end

  if (m_ref_kind = 'BusinessSubtypeList') then
  begin
    obj_kind = 20;
    json_tmp = '';
    text1 = '';
    text2 = '';
    memo1 = '';
    memo2 = '';

    for select bs_id, f_format('<b>%1</b>', bs_name), bt_name
    from bustype, bussubtype
    where bs_from_bt_id = bt_id
      and coalesce(bs_is_delete, 0) = 0
      and bs_id <> 0
      and (:m_find_txt = '' or bs_name containing :m_find_txt or bt_name containing :m_find_txt)
    order by bt_name, bs_name
    into obj_id, text1, memo1
    do
    begin
      if (db_id in (39801, 41701) and m_obj_kind = -1) then
      begin
        -- Реализация только для КГ.
        -- Поле Тип бизнеса переименовано в Отрасль кредитования
        -- Если цель кредита: Потребительский, то и Отрасль кредитования: Потребительский
        -- Либо оба поля должны быть не Потребительские
        if ((m_obj_id = 63 and obj_id = 21) or (m_obj_id <> 63 and obj_id <> 21)) then
          suspend;
      end
      else if (db_id = 81001 and m_obj_kind = -1) then
      begin
        -- Реализация только для РФ.  RFCRU-3456
        -- Если цель кредита: Потребительский (refdata = 73), то и Отрасль кредитования: Потребительский bussubtype id = 21
        -- Если цель кредита: Развитие бизнеса (refdata = 72) то в списке будут новые отрасли c bussubtype id >= 41
        if ((m_obj_id = 72 and obj_id <= 41) or (m_obj_id = 73 and obj_id <> 21))  then
          continue;
        else
          suspend;
      end
      else
        suspend;
    end
    exit;
  end

  if (m_ref_kind = 'BirthCityList') then
  begin
    obj_kind = 0;
    obj_id = 0;
    json_tmp = '';
    text1 = '';
    text2 = '';
    memo1 = '';
    memo2 = '';
    t = 0;

    if (m_find_txt <> '') then
    begin
      m_find_txt = f_replace_all(trim(m_find_txt), '  ', ' ');

      select result_txt
      from hp_convert_format(:m_find_txt, 'lat')
      into :find_txt_lat;

      select result_txt
      from hp_convert_format(:m_find_txt, 'rus')
      into :find_txt_cyr;

      m_find_txt = '%' || replace(m_find_txt, ' ', '%') || '%';
      find_txt_lat = '%' || replace(find_txt_lat, ' ', '%') || '%';
      find_txt_cyr = '%' || replace(find_txt_cyr, ' ', '%') || '%';

      for select text_list
          from aarep_get_combolist('clients', 'c_birth_city', '', null, :m_ref_add_kind)
          where text_list like :m_find_txt or
                text_list like :find_txt_lat or
                text_list like :find_txt_cyr rows 100
          into text1
      do
        suspend;
    end
    else
    begin
      for select text_list
          from aarep_get_combolist('clients', 'c_birth_city', '', null, :m_ref_add_kind)
          rows 100
          into text1
      do
        suspend;
    end
    exit;
  end

  if (m_ref_kind = 'ContactRelationList') then
  begin
    obj_kind = 0;
    obj_id = 0;
    text1 = '';
    text2 = '';
    memo1 = '';
    memo2 = '';
    add_params = '';
    for select rd_name
    from refdata
    where rd_main_code = 'relation'
      and rd_is_active = 1
      and rd_code <> rd_main_code
      and coalesce(rd_str_id, '') <> ''
    order by rd_orderno
    into :text1
    do
      suspend;

    text1 = 'Коллега';
    suspend;
    text1 = 'Друг';
    suspend;
    text1 = 'Знакомый';
    suspend;
    text1 = 'Дальний родственник';
    suspend;
    text1 = 'Работодатель';
    suspend;

    exit;
  end

  if (m_ref_kind = 'MatList') then
  begin
    obj_kind = null;
    obj_id = 0; -- Тут будет лежать kontmat.kml_from_feg_id
    text1 = '';
    text2 = '';
    memo1 = '';
    memo2 = '';
    add_params = '';

    for select ref.feg_id,
      f_format('<b>%1</b>', left(ref.feg_name, 248)),
      f_format('<small>%1</small>', left(coalesce(ref_p.feg_name, ''), 240)),
      f_format('{"show_imei":%1}', iif(ref.feg_code = '8517120000',1,0))
    from hp_foreign_econom_goods_ref(:m_obj_id) ref
      left join foreign_econom_goods_ref ref_p on ref.feg_parent_id = ref_p.feg_id
    where ref.feg_code is not null
      and (:m_find_txt = '' or ref.feg_name containing :m_find_txt or ref_p.feg_name containing :m_find_txt)

    rows 50
    into obj_id, text1, memo1, add_params
    do
      suspend;
    exit;
  end

  if (m_ref_kind = 'AgreementList') then
  begin
    obj_kind = null; -- Agreemtnt отсутствует в f_objkind
    obj_id = 0;
    text1 = '';
    text2 = '';
    memo1 = '';
    memo2 = '';
    add_params = '';

    -- Находим регион и дату из договора
    -- Получаем список договоров для региона и даты
    for select ag_id, ag_caption
        from hp_get_agreement(:m_obj_id)
        into :obj_id, :text1
    do
      suspend;
    exit;
  end

  if (m_ref_kind = 'IstisnaaSellers') then
  begin
    obj_kind = 0;
    obj_id = 0;
    text1 = '';
    text2 = '';
    memo1 = '';
    memo2 = '';
    add_params = '';
    
    for select ag.ag_object_id, c.c_name_fio
        from agreements ag
        inner join clients c on c.c_id = ag.ag_object_id
        where ag.ag_from_rd_id = f_get_rd_id('AgreementKinds.IPF_SellersI') and --Продавцы Истиснаа
              current_date between ag.ag_date_begin and ag.ag_date_end and
              ag.ag_object_kind = f_objkind('Clients', 'ID') and
              ag.ag_status = 1
        into :obj_id, :text1
    do
      suspend;
    exit;
  end 
end
