CREATE PROCEDURE HP_CHECK_FORMAT(
    NEW_PASS VARCHAR(255))
RETURNS (
    RESULT_TXT VARCHAR(255))
AS
declare variable chars_count integer;
declare variable ch varchar(1) collate unicode;
declare variable i integer;
declare variable str_uppers varchar(255) collate unicode;
declare variable str_lowers varchar(255) collate unicode;
declare variable has_lower integer;
declare variable has_upper integer;
declare variable has_symbol integer;
declare variable has_number integer;
begin
-- Проверяет формат пароля
-- Если исходящий Result_TXT<>'' значит что-то не так.

  result_txt = '';
  new_pass = coalesce(new_pass, '');
  if (new_pass = '') then
  begin
    result_txt = 'Новый пароль не указан';
    suspend;
    exit;
  end
  if (char_length(new_pass) < 7) then
  begin
    result_txt = 'Длина пароля должна быть больше 6 символов.';
    suspend;
    exit;
  end
  chars_count = char_length(new_pass);
  i = 0;
  has_number = 0;
  has_symbol = 0;
  has_upper = 0;
  has_lower = 0;
  str_uppers = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
  str_lowers = 'abcdefghijklmnopqrstuvwxyz';
  while (i < chars_count) do
  begin
    i = i + 1;
    ch = substring(new_pass from i for 1);
    if (position(ch, '0123456789') > 0) then
      has_number = 1;
    if (position(ch, '.,~!@#$%^&*()_+-') > 0) then
      has_symbol = 1;
    if (position(ch, str_uppers) > 0) then
      has_upper = 1;
    if (position(ch, str_lowers) > 0) then
      has_lower = 1;
  end
  if ((has_number + has_symbol + has_upper + has_lower) < 3) then
  begin
    result_txt = 'Пароль должен содержать минимум три категории из:' || ascii_char(13) || ascii_char(10);
    result_txt = result_txt || 'Латинские заглавные (А-Z);' || ascii_char(13) || ascii_char(10);
    result_txt = result_txt || 'Латинские строчные (a-z);' || ascii_char(13) || ascii_char(10);
    result_txt = result_txt || 'Цифры (0-9)' || ascii_char(13) || ascii_char(10);
    result_txt = result_txt || 'Отличные от букв и цифр знаки (например: !, $, #, % и др.).';
    suspend;
    exit;
  end
  if ((new_pass = 'Qwerty123') or (new_pass = 'Qwerty!@#') or (new_pass = '@Werty123')) then
  begin
    result_txt = 'Запрещенная комбинация символов';
    suspend;
    exit;
  end
  suspend;
end
