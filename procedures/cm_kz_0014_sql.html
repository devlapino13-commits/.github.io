CREATE PROCEDURE CM_KZ_0014_SQL(
    MA_ID D_ID,
    CHECK_KIND D_STR_255,
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable IS_EMPLOYED D_STR_255;
declare variable IS_ASP_RECEIVER D_STR_255;
declare variable IS_PENSIONER D_STR_255;
begin
    -- RFCRU-3940
    res_error_text = '';
    res_error_is_critical = 0;

    for select mac.mac_from_c_id
    from miniappclient mac
    where mac.mac_from_ma_id = :ma_id and
          mac.mac_amount > 0
    into :c_id

    do
    begin
        select max(case when opr_subject_id = 9262 then rd_code end) as is_employed,
               max(case when opr_subject_id = 9265 then rd_code end) as is_asp_receiver,
               max(case when opr_subject_id = 9264 then rd_code end) as is_pensioner
        from objectvalues, objectprops, objectvalres, refdata
        where opr_object_kind = 0 and
              opr_object_id = :c_id and
              opr_subject_kind = '-1' and
              opr_subject_id in (9262, 9265, 9264) and -- 9265 - Является получателем АСП (да)
              ovs_from_opr_id = opr_id and -- 9262 - Является ли клиент трудоустроенным в настоящее время (Да/Нет) см. refdata
              ovr_result_rd_id = rd_id and   -- 9264 - Является ли клиент пенсионером
              ovs_id = opr_actual_ovs_id and
              ovr_from_opr_id = ovs_from_opr_id and
              ovr_from_ovs_id = ovs_id
        into :is_employed, is_asp_receiver, is_pensioner;

        if (is_employed containing 'No'
            and is_pensioner containing 'No'
            and is_asp_receiver containing 'Yes') then
        begin
            res_error_text = 'Клиент не является трудоустроенным, займ не может быть выдан';
            res_error_is_critical = 2; 
            suspend;
        end
    end
end
