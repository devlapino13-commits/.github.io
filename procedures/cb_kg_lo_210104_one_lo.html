    AAREP_PAR_BALANCE7 DECIMAL(18,4),
    AAREP_PORT_BALANCE_OWN DECIMAL(18,4),
    AAREP_PORT_KOLICH_OWN INTEGER,

AS
declare variable aarep_port_kolich_all_tmp integer;
declare variable aarep_par_balance1_tmp decimal(18,4);
declare variable aarep_par_balance7_tmp decimal(18,4);
declare variable aarep_port_balance_all_tmp decimal(18,4);
declare variable aarep_month_plan_os decimal(18,4) = 0;
declare variable aarep_month_plan_perc decimal(18,4) = 0;
declare variable aarep_plan_all decimal(18,4);
declare variable aarep_fact_all decimal(18,4);
declare variable aarep_fact_planned decimal(18,4);
declare variable aarep_fact_other decimal(18,4);
declare variable koef_bonus decimal(18,4) = 0;
declare variable koef_penalty decimal(18,4) = 0;
declare variable aarep1_plan_all decimal(18,4) = 0;
declare variable aarep1_fact_all decimal(18,4) = 0;
declare variable aarep1_fact_planned decimal(18,4) = 0;
declare variable aarep_fact_kommis_ipf decimal(18,4);
declare variable aarep1_fact_other decimal(18,4) = 0;
declare variable c_id decimal(18,0);
declare variable c_email varchar(100);
declare variable reg_id integer;
declare variable k_id_tmp id_type;
declare variable k_disb_date_tmp date;
declare variable clients_qty_tmp integer;
declare variable aarep1_delinq_os decimal(18,4) = 0;
declare variable ps_date_begin date;
declare variable date_dismissal date;
declare variable ps_date_begin_first date;
declare variable bonus_koef curr_type;
declare variable bonus_sum curr_type;
declare variable bonus_memo varchar(255);
declare variable koef_consult decimal(18,4) = 0;
declare variable sawt_name str_100;
declare variable aarep1_consult_summa decimal(18,4);
declare variable rep_date_begin d_date;
begin

  koef_bonus = 90.0;
  koef_penalty = 400.0;
  aarep_port_kolich_ex_tranfer = 0;
  koef_consult = 90.0;
  rep_date_begin = '01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end);
  for select ps_from_c_id, c_email, ps_from_reg_id, ps_date_begin, jl_name
      from hr_pstatus, clients, joblist, jobkind, region_list(:rep_region_id, 1, -1, 1)
      where :rep_c_id = 0 and
            ps_from_c_id = c_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_c_id > 1000000000003 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            ps_from_reg_id = id_list
      union
      select c_id, c_email, :rep_region_id, current_date, ''
      from clients
      where :rep_c_id <> 0 and
            c_id = :rep_c_id
      order by 3
      into :c_id, :c_email, :reg_id, :ps_date_begin, jl_name
  do
  begin

    select c_name_fio
    from clients
    where c_id = :c_id
    into :c_name_fio;

    aarep_lo_status = '';

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into :reg_name;

    aarep_memo = '';
    if (c_email is null) then
      c_email = '';
    if (c_email like '%@molbulak.com') then
      aarep_memo = 'Ok';
    else
      aarep_memo = 'Не будет отправлено, нет EMAIL';

    aarep_port_balance_all = 0;
    aarep_port_kolich_all = 0;

    aarep_par_balance1 = 0;
    aarep_par_balance7 = 0;

    aarep_port_balance_all_tmp = 0;
    aarep_port_kolich_all_tmp = 0;
    aarep_par_balance1_tmp = 0;
    aarep_par_balance7_tmp = 0;

    aarep_clients_qty = 0;

    for select kont.k_id, kont.k_disb_date, sum(oks_balance), sum(k_kolich_client_approved),
               sum(iif(:rep_date_end - oks_delinq_date > 1, oks_balance, 0.0)),
               sum(iif(:rep_date_end - oks_delinq_date > 7, oks_balance, 0.0))
        from oksummary, kontract kont
        where oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end and
              oks_kc_id = :c_id and
              oks_balance <> 0 and
              oks_from_k_id = k_id and
              (:rep_date_end - oks_delinq_date <= 180) and
              oks_close_state = 0
        group by k_id, k_disb_date
        into :k_id_tmp, :k_disb_date_tmp, :aarep_port_balance_all_tmp, :aarep_port_kolich_all_tmp,
             :aarep_par_balance1_tmp, :aarep_par_balance7_tmp
    do
    begin
      if (:k_id_tmp in (109010107649000201, 109010114207900201, 109010114207900301, 109010111341700101, 109010111337500201, 109010072951100201, 109010072951100301, 109010072951100401, 109010093060500301, 109010115136100101,
                        109010116846500101, 109010108614000101)) then
        continue;
      aarep_port_balance_all = aarep_port_balance_all + aarep_port_balance_all_tmp;
      aarep_port_kolich_all = aarep_port_kolich_all + aarep_port_kolich_all_tmp;
      aarep_par_balance1 = aarep_par_balance1 + aarep_par_balance1_tmp;
      aarep_par_balance7 = aarep_par_balance7 + aarep_par_balance7_tmp;

      select res_just_client_count
      from hp_get_just_client_in_loan(:rep_date_end, :k_id_tmp, :k_disb_date_tmp)
      into clients_qty_tmp;

      if (clients_qty_tmp is null) then
        clients_qty_tmp = 0;

      aarep_clients_qty = aarep_clients_qty + clients_qty_tmp;
    end

    if (aarep_port_balance_all is null) then
      aarep_port_balance_all = 0;
    if (aarep_port_kolich_all is null) then
      aarep_port_kolich_all = 0;
    if (aarep_par_balance1 is null) then
      aarep_par_balance1 = 0;
    if (aarep_par_balance7 is null) then
      aarep_par_balance7 = 0;

    ----------------------------------------------------------------------- <<
    if (aarep_port_balance_all = 0) then
      aarep_memo = 'Не будет отправлено, нет КП.';

    aarep_date = :rep_date_end;
    aarep_plan_all = 0;
    aarep_fact_all = 0;
    aarep_fact_planned = 0;
    aarep_fact_other = 0;
    aarep_delinq_os = 0;
    aarep_delinq_perc = 0;
    aarep_fact_os = 0;
    aarep_fact_perc = 0;
    aarep_fact_fine = 0;
    aarep_fact_delinq_os = 0;
    aarep_fact_delinq_perc = 0;
    aarep_fact_adv_os = 0;
    aarep_fact_adv_perc = 0;
    aarep_port_balance_own = 0;
    aarep_port_kolich_own = 0;
    aarep_oklad = 0;
    select sum(aarep_plan_all), sum(aarep_fact_all), sum(aarep_fact_planned), sum(aarep_fact_other),
           sum(aarep_month_plan_os), sum(aarep_month_plan_perc), sum(aarep_delinq_os), sum(aarep_delinq_perc),
           sum(aarep_fact_os), sum(aarep_fact_perc), sum(aarep_fact_fine), sum(aarep_fact_delinq_os),
           sum(aarep_fact_delinq_perc), sum(aarep_fact_adv_os), sum(aarep_fact_adv_perc),
           sum(aarep_port_balance_own), sum(aarep_port_kolich_own), sum(aarep_fact_kommis), sum(aarep_fact_kommis_ipf),
           sum(transfer_sb), sum(aarep_consult_summa)
    from cb_kg_lo_210104_one_tmp(:reg_id, :c_id, :rep_date_end)
    into :aarep_plan_all, aarep_fact_all, aarep_fact_planned, aarep_fact_other, aarep_month_plan_os,
         aarep_month_plan_perc, aarep_delinq_os, aarep_delinq_perc, aarep_fact_os, aarep_fact_perc, aarep_fact_fine,
         aarep_fact_delinq_os, aarep_fact_delinq_perc, aarep_fact_adv_os, aarep_fact_adv_perc,
         aarep_port_balance_own, aarep_port_kolich_own, aarep_fact_kommis, aarep_fact_kommis_ipf, kol_transfer_sb,
         aarep_consult_summa;

    if (aarep_plan_all is null) then
      aarep_plan_all = 0; -- Сумма новых просрочек + выплаченные по графику за период
    if (aarep_fact_all is null) then
      aarep_fact_all = 0; -- Сумма новых просрочек + выплаченные по графику за период
    if (aarep_fact_planned is null) then
      aarep_fact_planned = 0; -- выплаченные по графику за период
    if (aarep_fact_other is null) then
      aarep_fact_other = 0; -- выплаченные по графику за период
    if (aarep_month_plan_os is null) then
      aarep_month_plan_os = 0;
    if (aarep_month_plan_perc is null) then
      aarep_month_plan_perc = 0;
    if (aarep_delinq_os is null) then
      aarep_delinq_os = 0;
    if (aarep_delinq_perc is null) then
      aarep_delinq_perc = 0;
    if (aarep_fact_os is null) then
      aarep_fact_os = 0;
    if (aarep_fact_perc is null) then
      aarep_fact_perc = 0;
    if (aarep_fact_fine is null) then
      aarep_fact_fine = 0;
    if (aarep_fact_kommis is null) then
      aarep_fact_kommis = 0;
    if (aarep_fact_delinq_os is null) then
      aarep_fact_delinq_os = 0;
    if (aarep_fact_delinq_perc is null) then
      aarep_fact_delinq_perc = 0;
    if (aarep_fact_adv_os is null) then
      aarep_fact_adv_os = 0;
    if (aarep_fact_adv_perc is null) then
      aarep_fact_adv_perc = 0;
    if (aarep_consult_summa is null) then
      aarep_consult_summa = 0;
    select sum(osd_value)
    from opersalmain osm
    inner join opersaldetail osd on osd.osd_from_osm_id = osm.osm_id
    inner join sa_config sac on sac.sac_id = osd.osd_from_sac_id and
          sac.sac_id = 19
    where osm_ol_is_active = 1 and
          osm_ol_date between :rep_date_begin and :rep_date_end and
          osm_from_c_id = :c_id and
          osd_value <> 0
          group by sac.sac_id
    into aarep_oklad;
        if (aarep_oklad is null) then
      aarep_oklad = 0;

    aarep_returnment_koef = 1.0;
    if (aarep_plan_all <> 0) then
      aarep_returnment_koef = round(aarep_fact_planned / aarep_plan_all, 4);
    else
      aarep_returnment_koef = 0.0;
    if (aarep_returnment_koef > 1.0) then
      aarep_returnment_koef = 1.0;

    aarep_bonus_plus = round((aarep_fact_all / 1000.0) * koef_bonus, 0);
    aarep_bonus_minus = round((aarep_delinq_os / 1000.0) * koef_penalty, 0);
    aarep_bonus_consult = round((aarep_consult_summa / 1000.0) * koef_consult, 0);
    aarep_bonus_minus_sb = kol_transfer_sb * 50;

    aarep_bonus_total_fact = 0;
    aarep_bonus_total_to_pay = 0;
    aarep_bonus_total = aarep_bonus_plus + aarep_bonus_consult - aarep_bonus_minus - aarep_bonus_minus_sb;
    aarep_bonus_total_fact = aarep_bonus_total - aarep_oklad;
    aarep_bonus_total_to_pay = aarep_bonus_total_fact; -- бонус к начислению
    if (aarep_bonus_total_to_pay < 0) then
      aarep_bonus_total_to_pay = 0;
    aarep_bonus_ideal = round(((aarep_fact_delinq_perc + aarep_fact_kommis + aarep_month_plan_perc + aarep_fact_adv_perc + aarep_fact_fine + aarep_fact_kommis_ipf) / 1000.0) * koef_bonus, 0);

    aarep_day_bonus_plus = aarep_bonus_plus;
    aarep_day_bonus_consult = aarep_bonus_consult;
    aarep_day_bonus_minus = aarep_bonus_minus;
    aarep_day_bonus_total = aarep_bonus_total;
    aarep_day_returnment_koef = aarep_returnment_koef;

    aarep1_plan_all = 0;
    aarep1_fact_all = 0;
    aarep1_fact_planned = 0;
    aarep1_fact_other = 0;

    if ((extract(day from :rep_date_end) > 1) and
        (need_one_day = 1)) then
    begin
      select sum(aarep_plan_all), sum(aarep_fact_all), sum(aarep_fact_planned), sum(aarep_fact_other),
             sum(aarep_delinq_os), sum(aarep_fact_kommis), sum(aarep_consult_summa)
      from cb_kg_lo_210104_one_tmp(:reg_id, :c_id, :rep_date_end - 1)
      into :aarep1_plan_all, :aarep1_fact_all, :aarep1_fact_planned, :aarep1_fact_other, aarep1_delinq_os,
           aarep_fact_kommis, aarep1_consult_summa;

      if (aarep1_plan_all is null) then
        aarep1_plan_all = 0;
      if (aarep1_fact_all is null) then
        aarep1_fact_all = 0;
      if (aarep1_fact_planned is null) then
        aarep1_fact_planned = 0;
      if (aarep1_fact_other is null) then
        aarep1_fact_other = 0;
      if (aarep1_delinq_os is null) then
        aarep1_delinq_os = 0;

      aarep_day_returnment_koef = 1.0;
      if (aarep_plan_all - aarep1_plan_all <> 0) then
        aarep_day_returnment_koef = round((aarep_fact_planned - aarep1_fact_planned) / (aarep_plan_all - aarep1_plan_all), 4);
      else
        aarep_day_returnment_koef = 0.0;
      if (aarep_day_returnment_koef > 1.0) then
        aarep_day_returnment_koef = 1.0;
      if (aarep_day_returnment_koef < 0) then
        aarep_day_returnment_koef = 1.0;

      aarep_day_bonus_plus = round((aarep1_fact_all / 1000.0) * koef_bonus, 0);
      aarep_day_bonus_consult = round((aarep1_consult_summa / 1000.0) * koef_consult, 0);
      aarep_day_bonus_minus = round((aarep1_delinq_os / 1000.0) * koef_penalty, 0);
      aarep_day_bonus_total = aarep_day_bonus_plus + aarep_day_bonus_consult - aarep_day_bonus_minus;

      aarep_day_bonus_consult = aarep_bonus_consult - aarep_day_bonus_consult;
      aarep_day_bonus_plus = aarep_bonus_plus - aarep_day_bonus_plus;
      aarep_day_bonus_minus = aarep_bonus_minus - aarep_day_bonus_minus;
      aarep_day_bonus_total = aarep_bonus_total - aarep_day_bonus_total;
    end
    aarep_bonus_memo = '';
    if (aarep_bonus_total_to_pay > 0 and
        ps_date_begin > dateadd(month, -3, :rep_date_end) -- менее трех месяцев
        ) then
    begin
      --      если бонус к начислению имеется то только тогда проверяем стаж
      --      если сотрудник уволился и заново устроился, то стаж будет считаться с той даты когда повторный раз устроился
      --      вычисляем последнюю дату увольнения
      date_dismissal = null;
      select max(ps_date_begin)
      from hr_pstatus
      where ps_is_working = 0 and
            ps_status = 1 and
            ps_from_c_id = :c_id
      into date_dismissal;
      date_dismissal = coalesce(date_dismissal, '01.01.1990');
      --      вычисляем дату когда последний раз устроился
      select min(ps_date_begin)
      from hr_pstatus
      where ps_is_working = 1 and
            ps_status = 1 and
            ps_from_c_id = :c_id and
            ps_date_begin > :date_dismissal
      into ps_date_begin_first;

      if (ps_date_begin_first > dateadd(month, -3, :rep_date_end)) then
      begin
        if (not(rep_date_end between '01.09.20' and '31.10.20' and
            c_id = 1090100633822)) then
        begin
          --        если дата когда устроился последний раз менее 3х месяцев то обнуляем бонус к начислению
          aarep_bonus_total_to_pay = 0;
          aarep_bonus_memo = 'Стаж менее 3х месяцев';
        end
      end
    end

    --  Вытаскивает текущий статус СПФ и ССПФ
    sawt_name = null;
    select sawt_name
    from hr_voctravel, hr_orders, sa_worktype
    where hrv_from_ho_id = ho_id and
          ho_order_type = sawt_id and
          hrv_from_c_id = :c_id and
          :rep_date_end between hrv_date_begin and hrv_date_end_fact
    order by hrv_date_begin desc
    rows 1
    into sawt_name;
    -- Если Трудовой отпуск декретницы отпуск то обнуляет бонус
    if (sawt_name containing 'Трудовой отпуск декретницы') then
    begin
      aarep_bonus_total_to_pay = 0;
    end
    --    вытаскивает данные по выговорам и бонус коэффициент
    bonus_memo = '';
    bonus_koef = 0;
    bonus_sum = 0;
    if (aarep_bonus_total_to_pay > 0) then
    begin
      select bonus_koef, bonus_sum, bonus_memo
      from hp_get_vigovor_koef(:c_id, :rep_date_end)
      where bonus_memo <> 'Нет замечаний и выговоров'
      into bonus_koef, bonus_sum, bonus_memo;

      if (bonus_memo <> '') then
      begin
        aarep_bonus_total_to_pay = aarep_bonus_total_to_pay * bonus_koef / 100 - bonus_sum;
        if (aarep_bonus_total_to_pay < 0) then
          aarep_bonus_total_to_pay = 0;
        aarep_bonus_memo = bonus_memo;
      end
    end

    if (rep_c_id <> 0) then
      suspend;
    else
    begin
      if ((aarep_port_balance_all <> 0) or (aarep_bonus_total <> 0)) then
        suspend;
    end
  end
end
