CREATE PROCEDURE AAREP_CALC_PERC_ONE_TEMP2(
    KONT_ID DECIMAL(18,0),
    DATE_END DATE)
RETURNS (
    PDD DATE,
    POK_T INTEGER,
    PSH_DD DATE,
    PBALANCE DECIMAL(18,4),
    PPAYED_PERC DECIMAL(18,4),
    PDD_ORIG DATE,
    PPAYED_TAX DECIMAL(18,4))
AS
declare variable psh_dd2 date;
declare variable psh_dd_prior date;
declare variable pbalance_all decimal(18,4);
declare variable pbalance_orig decimal(18,4);
declare variable pdd_orig2 date;
declare variable psh_dd_orig date;
declare variable last_sh_dd date;
declare variable moved_sh_date d_date;
begin
  psh_dd_prior = null;
  pbalance_all = 0;

  psh_dd2 = null;
  moved_sh_date = null;
  for select ok_ol_date,
             iif(ok_sh_date >= '01.03.2020' and ok_type = 2 and ol_oper_type = 1 and ol_oper_subtype = 9 and ok_sh_os > 0, -2, ok_type), -- если это перенос даты, отмечаем -2
             ok_sh_date,
             sum(ok_os + ok_sh_os), sum(ok_perc), sum(ok_tax)
      from operkred, operlist
      where ok_ol_is_active = 1 and
            ok_from_k_id = :kont_id and
            ok_type <= 7 and
            ok_from_c_id = 0 and
            ok_from_ol_id = ol_id
      group by 1, 2, 3
      having sum(ok_os + ok_sh_os) <> 0 or (sum(ok_perc) <> 0) or (sum(ok_tax) <> 0)
      order by 3, 1
      into :pdd, :pok_t, :psh_dd, :pbalance, :ppayed_perc, :ppayed_tax
  do
  begin
    if (pok_t = -2) then
    begin
      pok_t = 2;
      moved_sh_date = psh_dd;
    end

    if (date_end >= '01.04.2020') then
    begin
      if (moved_sh_date = psh_dd) then -- если платеж перенесен, считаем что он исполнен и не начисляем %% на эту часть
        pbalance = 0;
    end

    pbalance_orig = :pbalance;
    pdd_orig2 = :pdd;
    psh_dd_orig = :psh_dd;
    if ((pok_t <> 3) and
        (pdd <= :date_end)) then
    begin
      pdd_orig = :pdd;
      if (pdd > psh_dd) then
        pdd = :psh_dd;
      if (pbalance is null) then
        pbalance = 0;
      if (ppayed_perc is null) then
        ppayed_perc = 0;
      if (ppayed_tax is null) then
        ppayed_tax = 0;
      suspend;
    end

    if ((pok_t <> 3) and
        (psh_dd >= :date_end) and
        (psh_dd2 is null)) then
    begin
      psh_dd2 = :psh_dd;
      pdd = :date_end;
      pdd_orig = :date_end;
      pbalance = 0;
      ppayed_perc = 0;
      ppayed_tax = 0;
      pok_t = 4;
      suspend;
    end
    pdd = :pdd_orig2;

    if ((pdd <= :date_end) and
        (psh_dd <= :date_end)) then
    begin
      last_sh_dd = psh_dd_orig;
      if ((psh_dd = :psh_dd_prior) or
          (psh_dd_prior is null)) then
      begin
        pbalance = :pbalance_orig;
        pbalance_all = pbalance_all - pbalance;
        pdd = :psh_dd;
        pdd_orig = :psh_dd;
        pbalance = 0;
        ppayed_perc = 0;
        ppayed_tax = 0;
        pok_t = 0;
      end
      else
      begin
        if (pbalance_all < 0) then
        begin
          pbalance = :pbalance_all;
          psh_dd = :psh_dd_prior;
          pdd = :psh_dd_prior;
          pdd_orig = :psh_dd_prior;
          ppayed_perc = 0;
          ppayed_tax = 0;
          pok_t = 0;
          suspend;
        end
        pbalance_all = 0;
        pbalance = :pbalance_orig;
        pbalance_all = pbalance_all - pbalance;
        psh_dd = :psh_dd_orig;
        pdd = :psh_dd;
        pdd_orig = :psh_dd;
        pbalance = 0;
        ppayed_perc = 0;
        ppayed_tax = 0;
        pok_t = 0;
      end
    end

    psh_dd_prior = :psh_dd;
  end

  if (pbalance_all < 0) then
  begin
    pbalance = :pbalance_all;
    psh_dd = :last_sh_dd;
    pdd = :psh_dd;
    pdd_orig = :psh_dd;
    ppayed_perc = 0;
    ppayed_tax = 0;
    pok_t = 0;
    suspend;
  end
end
