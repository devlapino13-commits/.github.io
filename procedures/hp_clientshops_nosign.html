CREATE PROCEDURE HP_CLIENTSHOPS_NOSIGN(
    NOSIGN_LST D_STR_8100,
    KIND_LST D_STR_CODE)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    ID D_ID,
    NAME D_STR_255)
AS
declare variable dt_curr d_date;
begin
  dt_curr = current_date;
  for
    select
      res_str
    from hp_split_str(:nosign_lst, ';')
  into
    :id
  do
  begin
    if (kind_lst = 'clientshops') then
    for
      select
        c_id,
        c_name_fio,
        cs_name
      from clientshops
        inner join clients
          on c_id = cs_from_c_id
      where cs_id = :id
    into
      :c_id,
      :c_name_fio,
      :name
    do suspend;

    if (kind_lst = 'terminals') then
    for
      select
        c_id,
        c_name_fio,
        trm_invent_str
      from ipc_terminals
        inner join ipc_terminal_ownership
          on ito_trm_id = trm_id
            and :dt_curr between ito_date_begin and ito_date_end
        inner join clientshops
          on cs_id = ito_object_id
        inner join clients
          on c_id = cs_from_c_id
      where trm_id = :id
        and trm_type = 1
        and trm_actual = 1
    into
      :c_id,
      :c_name_fio,
      :name
    do suspend;
  end
end
