CREATE PROCEDURE CES_EDOC_SIGN(
    C_ID D_ID NOT NULL,
    EDS_ID D_STR_20 NOT NULL,
    REQUEST_JSON D_STR_8100 NOT NULL,
    ANSWER_JSON D_STR_8100 NOT NULL)
RETURNS (
    RES_CODE D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable EDL_ID D_ID;
declare variable ESM_ID D_ID;
declare variable SIGN_TEXT D_BLOB_TEXT;
begin
  res_text = 'Документы успешно подписаны';
  res_code = 0;

  select edl_id
  from edoc_sign, edoc_list
  where eds_id = :eds_id and
        eds_from_edl_id = edl_id
  into edl_id;

  if (edl_id is null) then
  begin
    res_text = 'Произошла ошибка при подписании эл. документа';
    res_code = 1;
    suspend;
    exit;
  end 

  sign_text = f_json_blob_get_value(:answer_json, 'sign', '');

  if (sign_text = '') then
  begin
    res_text = 'Произошла ошибка. Отсутствует подпись.';
    res_code = 1;
    suspend;
    exit;
  end
  begin
    update edoc_sign
    set eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'ces', 'id'),
        eds_status = 1
    where eds_id = :eds_id;
    
    insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
                          esm_full_source, esm_trans_id, esm_account)
    values ('ces', 'sign', current_date, 0, :c_id, 1, :request_json, :eds_id, :edl_id)
    returning esm_id
    into :esm_id;
    
    insert into extsysanswer (esa_from_esm_id, esa_message)
    values (:esm_id, :answer_json);
    
    insert into edoc_ces_sign (ecs_from_eds_id, eds_from_esm_id, eds_sign_text)
    values (:eds_id, :esm_id, :sign_text);

    when any do
    begin
        res_code = 1;
        res_text = 'Что-то пошло не так, пожалуйста обратитесь к своему кредитному служащему';
    end
  end

  suspend;
end
