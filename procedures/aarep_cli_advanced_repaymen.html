CREATE PROCEDURE AAREP_CLI_ADVANCED_REPAYMEN(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_AND DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_KREDLINE_END DATE,
    KS_DATE_BEGIN DATE,
    REG_NAME VARCHAR(61),
    PARENT_NAME VARCHAR(50),
    KAL_NEW_KF_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KF_OFFICER VARCHAR(100),
    PR_CODE VARCHAR(100),
    PR_NAME VARCHAR(50),
    LOAN_TYPE_STR VARCHAR(20),
    OS_BEFORE_CLOSING CURR_TYPE,
    OS_BEFORE_CLOS_CLI CURR_TYPE)
AS
declare variable st_staff integer;
declare variable st_client integer;
declare variable k_id id_type;
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate curr_type;
declare variable def_curr varchar(3);
declare variable ks_from_ol_id id_type;
declare variable curr_rate curr_type;
declare variable dolya perc_type;
begin
-- operkred_fix

-- Процедура которая вытаскивает список клиентов которые досрочно закрыли кредит
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select cm_default_currency
  from company
  into :def_curr;

  for select k_id, k_kontract_no, k_disb_date, k_amount_approved, k_kredline_end, ks_date_begin,
             reg_code || ' ' || reg_name as reg_name, parent_name, kal_new_kf_name,
--           cli.c_name_fio,                           -----------------------
--           kc_amount_approved,                       -----------------------
             off.c_name_fio as kf_officer, pr_code, pr_name,
             case pr_loan_type
               when 0 then 'Групповой'
               when 1 then 'Индив.'
               else 'неопред.'
             end,
             k_index_curr_code, k_index_curr_rate, ks_from_ol_id

      from kontstate, kontract, -- knclient, clients cli,
      clients off, knaktlist, region, region_list(:rep_region_id, 1, 3, 1), product
      where ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            ks_state = 2 and
            cast(ks_date_begin as date) between :rep_date_begin and :rep_date_and and
            k_kredline_end > cast(ks_date_begin as date) and
            cast(ks_date_begin as date) between kal_date and kal_date_end and
            kal_from_kf_id = k_from_kf_id and
            id_list = kal_new_reg_id and
            knaktlist.is_delete = 0 and
            reg_unicode = id_list
--    and kc_from_k_id = k_id
--    and kc_from_c_id = cli.c_id
--    and kc_amount_approved <> 0
            and
            kal_new_kf_officer = off.c_id and
            k_from_pr_id = pr_id
      into k_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :k_kredline_end, :ks_date_begin, :reg_name,
           :parent_name, :kal_new_kf_name,
--         :C_NAME_FIO,
--         :KC_AMOUNT_APPROVED,
           :kf_officer, :pr_code, :pr_name, :loan_type_str, k_index_curr_code, k_index_curr_rate, ks_from_ol_id
  do
  begin
    select coalesce(-sum(ok_os), 0)
    from operkred
    where ok_from_ol_id = :ks_from_ol_id -- выборка только по операции который закрыл контракт
          and
          ok_from_k_id = :k_id -- ограничение по контракту
          and
          ok_type = 4 and
          ok_sh_date > :ks_date_begin -- дата по графику должна > c даты операции
          and
          ok_from_c_id = 0
    into os_before_closing;

    if (k_index_curr_code <> def_curr) then
    begin
      select cr_rate
      from opercurr
      where cr_from_cu_id = :k_index_curr_code and
            :ks_date_begin between cr_date and cr_date_end rows 1
      into :curr_rate;
      if ((curr_rate is null) or (curr_rate = 0)) then
        curr_rate = 1;

      if (k_index_curr_rate > curr_rate) then
        os_before_closing = os_before_closing * k_index_curr_rate;
      else
        os_before_closing = os_before_closing * curr_rate;
      k_amount_approved = k_amount_approved * k_index_curr_rate;
      kc_amount_approved = kc_amount_approved * k_index_curr_rate;

    end

    for select c_name_fio, kc_amount_approved
        from knclient, clients
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = c_id
        into c_name_fio, kc_amount_approved
    do
    begin
      dolya = kc_amount_approved / k_amount_approved;
      os_before_clos_cli = os_before_closing * dolya;

      suspend;
    end
  end
end
