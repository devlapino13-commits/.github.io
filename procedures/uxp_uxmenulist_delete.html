CREATE PROCEDURE UXP$UXMENULIST_DELETE(
    ML_ID D_INTEGER,
    NEED_DISABLE D_BOOL)
AS
declare variable ml_kind d_integer;
declare variable ml_ua_uuid d_uuid;
begin
  -- Удаляет или отключает пункт меню или открепляет действие от меню
  if (need_disable is null) then
    need_disable = 0;

  select ml_kind
  from menulist
  where ml_id = :ml_id
  into ml_kind;

  if (row_count = 0) then
    exception e_message('Элемент меню не найден');

  ml_ua_uuid = null;
  select uml_ua_uuid
  from ux$menulist
  where uml_ml_id = :ml_id
  into ml_ua_uuid;

  if (need_disable = 0) then
  begin
    if (ml_kind = 0 and ml_ua_uuid is null) then
      exception e_message('Удалять можно только динамическое меню с действиями');
  
    if (exists(select * from menulist
    where ml_parent_id = :ml_id)) then
      exception e_message('Имеются подменю, сначала удалите их');
  
    if (ml_ua_uuid is not null) then
    begin
      delete from ux$menulist
      where uml_ml_id = :ml_id and
            uml_ua_uuid = :ml_ua_uuid;
    end else
    begin
      delete from menulist
      where ml_id = :ml_id and
            ml_kind <> 0;
    end
  end else
  begin
    if (ml_kind = 0 and ml_ua_uuid is null) then
      exception e_message('Отключать/включать можно только динамическое меню с действиями');
  
    update ux$menulist
    set uml_fs_disabled = 1 - uml_fs_disabled
    where uml_ml_id = :ml_id;
  end
end
