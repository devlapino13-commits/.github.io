CREATE PROCEDURE HP_PKB_MINIAPP_INFO_SAVE_RES(
    PMI_ID D_ID,
    JSON_INFO D_BLOB_TEXT,
    RES_ID D_SMALLINT,
    ERR_TEXT D_STR_LONG)
AS
declare variable BLS_ID D_ID;
begin
  bls_id = null;
  if (json_info is null or json_info = '' or json_info = '{}') then
    json_info = '';

  if (json_info <> '') then
  begin
    execute procedure hp_insert_into_blobstore('Ответ от сервиса(ПКБ)', 'Ответ от сервиса(ПКБ)', :json_info, 1000000000000, 2, 3)
    returning_values :bls_id;
  end
  update pkb_miniapp_info
      set
      pmi_is_sent = iif(:res_id = -1, -1, 1),
      pmi_bls_id = :bls_id,
      pmi_err_text = :err_text,
      pmi_last_datetime = current_timestamp
  where (pmi_id = :pmi_id);
end
