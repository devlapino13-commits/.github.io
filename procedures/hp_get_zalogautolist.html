CREATE PROCEDURE HP_GET_ZALOGAUTOLIST(
    ZAP_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    ZAL_ID D_INTEGER,
    ZAL_FROM_ZAP_ID D_INTEGER,
    ZAL_COST D_CURRENCY,
    ZAL_WEBPAGE D_STR_LONG,
    ZAL_STATUS D_SMALLINT,
    ZAL_IS_DELETE D_SMALLINT,
    ZAL_FROM_UN_ID D_ID,
    ZAL_DATETIME D_DATETIME,
    ZAL_STATUS_STR D_STR_CODE,
    ZAL_IMAGE_COUNT D_INTEGER,
    ZAL_IMAGE_LIST D_STR_8100,
    ZAL_IS_MAIN D_SMALLINT,
    TOTAL_ZAL_COST D_CURRENCY,
    TOTAL_ZAL_COUNT D_INTEGER,
    TOTAL_AVG_COST D_CURRENCY)
AS
declare variable blob_info d_str_8100;
declare variable bls_id d_id;
declare variable main_zal_id d_integer;
begin
  select za_from_zal_id
  from zalogautoprof, zalogauto
  where zap_id = :zap_id and
        zap_from_za_id = za_id
  into main_zal_id;

  select sum(zal_cost), count(*)
  from zalogautolist
  where zal_from_zap_id = :zap_id and
        zal_is_delete = 0 and
        zal_id <> :main_zal_id
  into total_zal_cost, total_zal_count;

  if (total_zal_cost is null) then
    total_zal_cost = 0;

  if (total_zal_count is null) then
    total_zal_count = 0;

  total_avg_cost = 0;
  if (total_zal_count <> 0) then
    total_avg_cost = round(total_zal_cost / total_zal_count, 0);

  for select zal_id, zal_from_zap_id, zal_cost, zal_webpage, zal_status, zal_is_delete, zal_from_un_id, zal_datetime
      from zalogautolist
      where zal_from_zap_id = :zap_id
      order by zal_id
      into zal_id, zal_from_zap_id, zal_cost, zal_webpage, zal_status, zal_is_delete, zal_from_un_id, zal_datetime
  do
  begin
    zal_status_str = '';
    if (zal_is_delete = 1) then
      zal_status_str = 'Удалена';
    else
    begin
      if (coalesce(zal_webpage, '') = '') then
        zal_status_str = 'Нет ссылки';
      else
        zal_status_str = case zal_status
                           when -2 then 'Ошибка'
                           when -1 then 'Доработка'
                           when 0 then 'Новая'
                           when 1 then 'В обработке'
                           when 2 then 'Обработан'
                         end;
    end

    zal_image_count = 0;
    zal_image_list = '';
    zal_is_main = iif(main_zal_id = zal_id, 1, 0);

    for select zab_from_bls_id
        from zalogautoblob
        where zab_from_zal_id = :zal_id and
              zab_from_bls_id is not null
        order by zab_id
        into bls_id
    do
    begin
      zal_image_count = zal_image_count + 1;
      blob_info = f_image_item('blobstore', bls_id, 'рис.' || zal_image_count, un_id);
      if (zal_image_list <> '') then
        zal_image_list = zal_image_list || ',';

      zal_image_list = zal_image_list || blob_info;
    end

    zal_image_list = '[{"caption": "Авто", "blobs":[' || zal_image_list || ']}]';
    suspend;
  end
end
