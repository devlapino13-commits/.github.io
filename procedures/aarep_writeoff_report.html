CREATE PROCEDURE AAREP_WRITEOFF_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    KONT_ID D_STR_20,
    REG_ID INTEGER,
    REP_C_NAME VARCHAR(100),
    REP_OFF_BEGIN_OS DECIMAL(18,4),
    REP_OFF_BEGIN_PERC DECIMAL(18,4),
    REP_OFF_BEGIN_OTHER DECIMAL(18,4),
    REP_OFF_SH_OS DECIMAL(18,4),
    REP_OFF_SH_PERC DECIMAL(18,4),
    REP_OFF_SH_OTHER DECIMAL(18,4),
    REP_OFF_PAY_OS DECIMAL(18,4),
    REP_OFF_PAY_PERC DECIMAL(18,4),
    REP_OFF_PAY_OTHER DECIMAL(18,4),
    REP_OFF_DEAD_OS DECIMAL(18,4),
    REP_OFF_DEAD_PERC DECIMAL(18,4),
    REP_OFF_DEAD_OTHER DECIMAL(18,4),
    REP_OFF_END_OS DECIMAL(18,4),
    REP_OFF_END_PERC DECIMAL(18,4),
    REP_OFF_END_OTHER DECIMAL(18,4),
    REP_WRITEOFF_DATE DATE,
    REP_FACT_WRITEOFF_DATE DATE,
    REP_LASTPAY_DATE DATE,
    K_DISB_DATE DATE,
    K_KONTRACT_NO VARCHAR(20),
    KF_WRITEOFF_NAME VARCHAR(100),
    KF_LAST_NAME VARCHAR(100),
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL2_NAME VARCHAR(100),
    AA_OFFICE_STAT_NAME VARCHAR(100),
    AA_PRODUCT_NAME VARCHAR(100),
    AA_PRODUCT_KIND VARCHAR(100),
    CURR_OFFICER VARCHAR(100),
    CURR_OFFICE VARCHAR(60),
    CURR_STAUS VARCHAR(20),
    C_BIRTH_DATE DATE,
    KS_STATE INTEGER,
    K_STATUS VARCHAR(10),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_PERIOD INTEGER,
    OL_DOC_CLIENT_NAME VARCHAR(255),
    RD_NAME VARCHAR(255),
    DELINQ_DAYS_WR_DATE INTEGER,
    K_CLOSE_DATE DATE,
    K_KREDLINE_END DATE,
    DELINQ_DAYS INTEGER,
    SBA_DATE_BEGIN DATE,
    SBL_NAME VARCHAR(255),
    SBL_TEAM_NAME VARCHAR(255),
    ZAD_OS CURR_TYPE,
    SH_CALC_PERC CURR_TYPE,
    SH_CALC_FINE CURR_TYPE,
    CAT_CR_AC VARCHAR(10),
    PER_OF_LIMIT DATE,
    C_ID ID_TYPE,
    KC_ID ID_TYPE,
    KC_NUMBER D_INTEGER,
    C_PASS_NO STR_100,
    C_ID_STR STR_100,
    KC_AMOUNT_APPROVED CURR_TYPE,
    K_PERCENT PERC_TYPE,
    IS_AZREPAYKONT SMALLINT,
    K_BASE_K_ID D_STR_20,
    AAREP_DISB_LO_FIO STR_100,
    DISB_REG_NAME VARCHAR(50),
    SH_TOTAL CURR_TYPE,
    REP_WRITEOFF_DATE_OS_PERC DATE,
    C_INN D_STR_20,
    FACT_ADR D_STR_255,
    INDEXED_ADR D_STR_255,
    LAL_NAME D_STR_255,
    NOTIFICATION D_STR_10)
AS
declare variable K_FROM_KF_ID decimal(18,0);
declare variable PR_IS_INDIV integer;
declare variable EMP integer;
declare variable PARAL integer;
declare variable PR_ID integer;
declare variable PR_ID_PRIOR integer;
declare variable CLIENT_KOLICH integer;
declare variable KAL_NEW_KF_OFFICER ID_TYPE;
declare variable KAL_NEW_KF_OFFICER_PRIOR ID_TYPE;
declare variable RD_ID ID_TYPE;
declare variable DB_ID integer;
declare variable K_FACT_DISB_LO_ID ID_TYPE;
declare variable K_FACT_DISB_REG_ID integer;
declare variable K_KONT_LAST_DATE2 D_DATE;
declare variable K_KOLICH_LGOTA D_INTEGER;
declare variable PR_CODE D_STR_100;
declare variable REP_PERIOD D_INTEGER;
declare variable REP_OFF_DATE D_DATE;
begin
  db_id = f_get_db_id();
  pr_id_prior = 0;
  kal_new_kf_officer_prior = 0;
  for select distinct oko_from_k_id, oko_from_c_id, k_last_reg_id, pr_name, pr_loan_type, pr_is_employer, pr_category,
                      pr_id, pr_code, k_kolich_client_approved, k_from_kf_id, kont.k_percent,
                      cast(iif(k_base_k_id is null, k_id, k_base_k_id) as d_str_20) k_base_k_id, kont.k_fact_disb_lo_id,
                      kont.k_fact_disb_reg_id, r.reg_code || ' ' || r.reg_name
      from operkroff, kontract kont, product, region r
      where k_id = oko_from_k_id and
            k_from_pr_id = pr_id and
            kont.k_fact_disb_reg_id = r.reg_unicode
      into :k_id, :c_id, :reg_id, :aa_product_name, :pr_is_indiv, :emp, :paral, :pr_id, :pr_code, :client_kolich, k_from_kf_id,
           :k_percent, k_base_k_id, k_fact_disb_lo_id, k_fact_disb_reg_id, disb_reg_name
  do
  begin
    kont_id = cast(k_id as d_str_20);

    select kc.kc_id, kc.kc_amount_approved, kc_number
    from knclient kc
    where kc_from_c_id = :c_id and
          kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into :kc_id, :kc_amount_approved, kc_number;

    select kal_new_kf_officer, kal_new_kf_name
    from knaktlist
    where knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal_from_kf_id = :k_from_kf_id
    into kal_new_kf_officer, kf_last_name;

    if (kal_new_kf_officer_prior <> kal_new_kf_officer) then
    begin
      curr_officer = (select c_name_fio
                      from clients
                      where c_id = :kal_new_kf_officer);
      curr_office = null;
      curr_staus = null;
      select reg_code || ' ' || reg_name, iif(ps_is_working = 1, 'Работает', 'Уволен')
      from hr_pstatus, region
      where ps_status = 1 and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_c_id = :kal_new_kf_officer and
            ps_from_reg_id = reg_unicode
      into curr_office, curr_staus;
      kal_new_kf_officer_prior = kal_new_kf_officer;
    end

    aarep_disb_lo_fio = null;

    select c_name_fio
    from clients
    where c_id = :k_fact_disb_lo_id
    into :aarep_disb_lo_fio;
    if (aarep_disb_lo_fio is null) then
      aarep_disb_lo_fio = 'не найден';

    c_birth_date = null;
    rep_c_name = '';
    c_pass_no = '';
    c_id_str = '';

    if (c_id is not null) then
    begin
      select c_name_fio, c_birth_date, cl.c_pass_no, cl.c_inn
      from clients cl
      where c_id = :c_id
      into :rep_c_name, :c_birth_date, :c_pass_no, :c_inn;

      c_id_str = cast(c_id as str_100);
    end

    if (pr_id <> pr_id_prior) then
    begin
      if ((pr_is_indiv = 0) and
          (emp = 0) and
          (paral = 0) and
          (client_kolich <> 2)) then
      begin
        aa_product_kind = 'A Групповой';
      end
      if ((pr_is_indiv = 1) and
          (emp = 0) and
          (paral = 0) and
          (client_kolich = 2)) then -- было if ((PR_IS_INDIV=0) and (emp=0) and (paral=0) and (CLIENT_KOLICH=2)) then
      begin
        aa_product_kind = 'E Парный';
      end
      if (paral = 1) then
      begin
        aa_product_kind = 'B Параллельный';
      end
      if ((pr_is_indiv = 1) and
          (emp = 0) and
          (paral = 0) and
          (client_kolich < 2)) then -- было if ((PR_IS_INDIV=1) and (emp=0) and (paral=0))
      begin
        aa_product_kind = 'C Индивидуальный';
      end
      if ((pr_is_indiv = 1) and
          (emp = 1) and
          (paral = 0)) then
      begin
        aa_product_kind = 'D Сотрудники';
      end
      pr_id_prior = pr_id;
    end
    rep_off_begin_os = 0;
    rep_off_begin_perc = 0;
    rep_off_begin_other = 0;

    rep_off_end_os = 0;
    rep_off_end_perc = 0;
    rep_off_end_other = 0;

    rep_off_sh_os = 0;
    rep_off_sh_perc = 0;
    rep_off_sh_other = 0;

    rep_off_pay_os = 0;
    rep_off_pay_perc = 0;
    rep_off_pay_other = 0;

    rep_off_dead_os = 0;
    rep_off_dead_perc = 0;
    rep_off_dead_other = 0;

    if (exists(select *
               from region_list(:reg_id, 1, -1, 0)
               where id_list = :rep_region_id)) then
    begin
      select sum(oko_sh_os + oko_os), sum(oko_sh_perc + oko_perc), sum(oko_sh_other + oko_other)
      from operkroff, operlist
      where oko_from_ol_id = ol_id and
            oko_from_k_id = :k_id and
            oko_from_c_id = :c_id and
            ol_date < :rep_date_begin and
            operlist.is_delete = 0
      into :rep_off_begin_os, :rep_off_begin_perc, :rep_off_begin_other;
      if (rep_off_begin_os is null) then
        rep_off_begin_os = 0;
      if (rep_off_begin_perc is null) then
        rep_off_begin_perc = 0;
      if (rep_off_begin_other is null) then
        rep_off_begin_other = 0;

      select sum(oko_sh_os), sum(oko_sh_perc), sum(oko_sh_other), max(oko_ol_date)
      from operkroff, operlist
      where oko_from_ol_id = ol_id and
            oko_from_k_id = :k_id and
            oko_from_c_id = :c_id and
            ol_date_oper_day between :rep_date_begin and :rep_date_end and
            oko_type = 1 and
            operlist.is_delete = 0
      into :rep_off_sh_os, :rep_off_sh_perc, :rep_off_sh_other, :rep_off_date;
      if (rep_off_sh_os is null) then
        rep_off_sh_os = 0;
      if (rep_off_sh_perc is null) then
        rep_off_sh_perc = 0;
      if (rep_off_sh_other is null) then
        rep_off_sh_other = 0;

      select sum(oko_os), sum(oko_perc), sum(oko_other)
      from operkroff, operlist
      where oko_from_ol_id = ol_id and
            oko_from_k_id = :k_id and
            oko_from_c_id = :c_id and
            ol_date_oper_day between :rep_date_begin and :rep_date_end and
            oko_type in (4, 7, 9) and
            operlist.is_delete = 0
      into :rep_off_pay_os, :rep_off_pay_perc, :rep_off_pay_other;
      if (rep_off_pay_os is null) then
        rep_off_pay_os = 0;
      if (rep_off_pay_perc is null) then
        rep_off_pay_perc = 0;
      if (rep_off_pay_other is null) then
        rep_off_pay_other = 0;

      select sum(oko_os), sum(oko_perc), sum(oko_other)
      from operkroff, operlist
      where oko_from_ol_id = ol_id and
            oko_from_k_id = :k_id and
            oko_from_c_id = :c_id and
            ol_date_oper_day between :rep_date_begin and :rep_date_end and
            oko_type = 5 and
            operlist.is_delete = 0
      into :rep_off_dead_os, :rep_off_dead_perc, :rep_off_dead_other;
      if (rep_off_dead_os is null) then
        rep_off_dead_os = 0;
      if (rep_off_dead_perc is null) then
        rep_off_dead_perc = 0;
      if (rep_off_dead_other is null) then
        rep_off_dead_other = 0;

      rep_off_end_os = rep_off_begin_os + rep_off_sh_os + rep_off_pay_os + rep_off_dead_os;
      rep_off_end_perc = rep_off_begin_perc + rep_off_sh_perc + rep_off_pay_perc + rep_off_dead_perc;
      rep_off_end_other = rep_off_begin_other + rep_off_sh_other + rep_off_pay_other + rep_off_dead_other;

      -- Дата первого списания по контракту только по ОД и %%
      if ((rep_off_begin_os <> 0) or (rep_off_begin_perc <> 0) or (rep_off_end_os <> 0) or (rep_off_end_perc <> 0) or (rep_off_sh_os <> 0) or (rep_off_sh_perc <> 0) or (rep_off_pay_os <> 0) or (rep_off_pay_perc <> 0) or (rep_off_dead_os <> 0) or (rep_off_dead_perc <> 0)) then


        select min(ol_date_oper_day)
        from operkroff, operlist
        where oko_from_k_id = :k_id and
              oko_from_ol_id = ol_id and
              oko_from_c_id = :c_id and
              operlist.is_delete = 0 and
              (oko_sh_os <> 0 or
              oko_sh_perc <> 0 or
              oko_os <> 0 or
              oko_perc <> 0)
        into :rep_writeoff_date_os_perc;

      if ((rep_off_begin_os <> 0) or (rep_off_begin_perc <> 0) or (rep_off_begin_other <> 0) or (rep_off_end_os <> 0) or (rep_off_end_perc <> 0) or (rep_off_end_other <> 0) or (rep_off_sh_os <> 0) or (rep_off_sh_perc <> 0) or (rep_off_sh_other <> 0) or (rep_off_pay_os <> 0) or (rep_off_pay_perc <> 0) or (rep_off_pay_other <> 0) or (rep_off_dead_os <> 0) or (rep_off_dead_perc <> 0) or (rep_off_dead_other <> 0)) then
      begin
        select min(ol_date_oper_day), max(ol_date_oper_day)
        from operkroff, operlist
        where oko_from_k_id = :k_id and
              oko_from_ol_id = ol_id and
              oko_from_c_id = :c_id and
              operlist.is_delete = 0
        into :rep_writeoff_date, :rep_lastpay_date;

        --Добавил новое поле. Дата перехода на внесистемный учет 09.04.15 Музаффар >>>>>
        ol_doc_client_name = null;
        rd_id = null;
        rep_fact_writeoff_date = null;
        select distinct ok_ol_date, ol.ol_doc_client_name, --Dobavlen 090316
                        ol.ol_from_rd_id --Dobavlen 090316
        from operkred ok, operkroff, operlist ol
        where ok_ol_is_active = 1 and
              ok_from_c_id = 0 and
              ok_type = 5 and
              ok_from_k_id = :k_id and
              ok_from_ol_id = oko_from_ol_id and
              oko_from_c_id = :c_id and
              (ok.ok_os + ok.ok_perc) < 0 and
              ok_from_ol_id = ol_id and
              ol.is_delete = 0
        order by ol_date desc
        rows 1
        into :rep_fact_writeoff_date, ol_doc_client_name, rd_id; --Конец <<<<<
        if (ol_doc_client_name is null) then
          ol_doc_client_name = '';

        rd_name = null;
        select rd_name
        from refdata rd
        where rd.rd_is_active = 1 and
              rd.rd_id = :rd_id
        into rd_name;
        if (rd_name is null) then
          rd_name = 'Не заполнен';

        select k_disb_date, k_kontract_no, k_from_kf_id, k_amount_approved, round(k_period), k_close_date,
               k_kredline_end, k_kolich_lgota --Dobavlen 090316
        from kontract
        where k_id = :k_id
        into :k_disb_date, k_kontract_no, k_from_kf_id, k_amount_approved, k_period, k_close_date, k_kredline_end,
             :k_kolich_lgota;

        select kal_new_kf_name
        from knaktlist
        where kal_from_kf_id = :k_from_kf_id and
              kal_date <= :rep_writeoff_date and
              kal_date_end >= :rep_writeoff_date and
              knaktlist.is_delete = 0
        into :kf_writeoff_name;

        select aa_curr_office_name, aa_curr_branch_name, aa_curr_level3_name, aa_office_stat_name
        from aarep_get_office_categ(:reg_id, :rep_date_end)
        into :aa_curr_office_name, :aa_curr_branch_name, aa_curr_level2_name, aa_office_stat_name;

        select ks_state
        from kontstate ks
        where ks_ol_is_delete = 0 and
              :rep_date_end between ks_date_begin and ks_date_end and
              ks_from_k_id = :k_id
        into ks_state;

        k_status = '';
        if (ks_state = 1) then
          k_status = 'Активный';
        if (ks_state = 2) then
          k_status = 'Закрытый';

        delinq_days_wr_date = 0;
        select iif(:rep_fact_writeoff_date - oks_delinq_date > 0, :rep_fact_writeoff_date - oks_delinq_date, 0)
        from oksummary
        where oks_close_state = 0 and
              oks_from_k_id = :k_id and
              :rep_fact_writeoff_date - 1 between oks_date_begin and oks_date_end
        into delinq_days_wr_date;

        delinq_days = 0;
        select iif(:rep_date_end - oks_delinq_date > 0, :rep_date_end - oks_delinq_date, 0)
        from oksummary
        where oks_close_state = 0 and
              oks_from_k_id = :k_id and
              :rep_date_end between oks_date_begin and oks_date_end
        into delinq_days;
        if (delinq_days is null) then
          delinq_days = 0;

        sba_date_begin = null;
        sbl_name = null;
        sbl_team_name = '';
        select sba.sba_date_begin, sbl.sbl_name, sbl.sbl_team_name
        from sb_akt sba, sb_list sbl
        where sba.sba_is_delete = 0 and
              sba.sba_from_k_id = :k_id and
              sba_from_sbl_id <> 0 and
              sba.sba_from_sbl_id = sbl.sbl_id and
              :rep_date_end between sba_date_begin and sba_date_end
        into sba_date_begin, sbl_name, sbl_team_name;
        if (sbl_name is null) then
        begin
          sbl_name = '';
          cat_cr_ac = 'A';
        end
        else
          cat_cr_ac = 'C';
        if (sbl_team_name is null) then
          sbl_team_name = '';

        zad_os = 0;
        sh_calc_perc = 0;
        sh_calc_fine = 0;
        select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine)
        from aarep_make_zad_list(:k_id, :sba_date_begin, 1)
        where sh_date is not null
        into zad_os, sh_calc_perc, sh_calc_fine;

        --срок исковой давности
        per_of_limit = dateadd(year, 3, k_kredline_end);

        if (db_id = 94401) then
        begin
          if (exists(select *
                     from azrepaykont ar
                     where ar.ark_from_k_id = :k_id)) then
            is_azrepaykont = 1;
          else
            is_azrepaykont = 0;

          lal_name = '';
          select coalesce(lal.lal_name, '')
          from lawmain lam
          inner join lawkredit lak on lak.lak_from_lam_id = lam.lam_id
          inner join lawcourts lac on lac.lac_from_lam_id = lam.lam_id
          inner join lawlist lal on lal.lal_id = lac.lac_from_lal_id
          where lak.lak_from_k_id = :k_id and
                lam.lam_is_active = 1 and
                :rep_date_end between lac.lac_date_begin and lac.lac_date_end
          order by lac.lac_date_begin desc
          rows 1
          into :lal_name;

          notification = '';
          if (exists(select *
                     from printcert prc
                     where prc.prc_is_active = 1 and
                           prc.prc_from_k_id = :k_id and
                           prc.prc_from_bln_id = 1225)) then   -- İcra qeydi
            notification = 'Отправлено';
        end

        select max(sh_date), max(iif(:k_kolich_lgota + 1 = sh_id, sh_os + sh_perc, 0))
        from schedule
        where sh_from_k_id = :k_id
        into k_kont_last_date2, sh_total;

        indexed_adr = null;

        select adl.adl_address_str
        from addrclient adc
        inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
        where adc.adc_from_c_id = :c_id and
              adc.adc_address_type = 1
        into indexed_adr;

        fact_adr = null;

        select adl.adl_address_str
        from addrclient adc
        inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
        where adc.adc_from_c_id = :c_id and
              adc.adc_address_type = 2
        into fact_adr;

        suspend;
      end
