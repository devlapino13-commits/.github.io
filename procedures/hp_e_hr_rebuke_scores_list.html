CREATE PROCEDURE HP_E_HR_REBUKE_SCORES_LIST(
    RS_FROM_RB_ID D_INTEGER)
RETURNS (
    RS_ID D_INTEGER,
    RS_FROM_RBK_ID D_INTEGER,
    RS_REBUKE_TYPE D_INTEGER,
    RS_SCORES D_SMALLINT,
    RBK_CODE D_STR_CODE,
    RBK_NAME D_STR_LONG,
    RES_REBUKE_TYPE_STR D_STR_100,
    RES_SCORE_LIMITS D_STR_20)
AS
begin
  for select rs_id, rs_from_rbk_id, rs_rebuke_type, rs_scores,
             rbk_code, rbk_name
  from rebuke_scores, rebuke_kind
  where rs_from_rb_id = :rs_from_rb_id and
        rs_is_delete = 0 and
        rs_from_rbk_id = rbk_id
  order by rs_id desc
  into :rs_id, :rs_from_rbk_id, :rs_rebuke_type, :rs_scores,
       :rbk_code, :rbk_name
  do
  begin
    res_rebuke_type_str = f_enum_name('rebuke_type', rs_rebuke_type);

    res_score_limits = '0';
    select res_score_limits
    from hp_e_hr_rebuke_get_scores(:rs_from_rbk_id, :rs_rebuke_type)
    into :res_score_limits;

    suspend;
  end
end
