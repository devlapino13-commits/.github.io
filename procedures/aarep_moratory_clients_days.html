declare variable pr_loan_type integer;
declare variable cc_id decimal(18,0);
declare variable cc_date date;
declare variable cc_state integer;
declare variable cc_k_id decimal(18,0);
declare variable cc_prior_id decimal(18,0);
declare variable cc_m_id decimal(18,0);
declare variable c_sex smallint;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  cc_prior_id = 0;
  for select distinct kc_from_c_id, ks_date_begin, ks_state, k_id, kal_new_reg_id, kal_new_kf_officer, kal_new_kf_name,
                      c_name_fio, reg_name
      from kontstate, kontract, knclient, knaktlist, clients, region, region_list(:report_region_id, 1, -1, 1)
      where ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            kc_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            ks_date_begin between(:report_date_end - 180) and :report_date_end and
            ks_date_end >= :report_date_end and
            kal_date <= :report_date_end and
            kal_date_end >= :report_date_end and
            kal_new_kf_officer = c_id and
            reg_unicode = kal_new_reg_id and
            kal_new_reg_id = id_list and
            kc_amount_approved <> 0 and
            kc_amount_approved is not null
      order by kc_from_c_id, ks_date_begin desc, ks_state
      into :cc_id, :cc_date, :cc_state, :cc_k_id, :m_from_reg_id, :kf_officer_id, :kf_name, :kf_officer_name,
           :kf_reg_name
  do
  begin
    if (cc_id <> :cc_prior_id) then
    begin
      if (cc_state = 2) then
      begin
        if ((cc_date + 7) <= :report_date_end) then
        begin
          cc_m_id = -1;
          select distinct m_id
          from moratory
          where m_from_c_id = :cc_id and
                m_is_active = 1 and
                m_is_black <= 1 -- Мораторий и отказанные. Стоп-лист не показываем.
                and
                m_date_begin >= :cc_date rows 1
          into :cc_m_id;
          if ((cc_m_id = -1) or (cc_m_id is null)) then
          begin
            for select k_id, k_from_kf_id, k_currency, k_applicate_date, k_applicate_no, k_kontract_no, k_kontract_date,
                       k_from_pr_id, k_disb_date, k_amount_applicate, k_stage, k_period, k_kolich_lgota, k_percent,
                       k_percent_lgota, k_kolich_repay, k_from_f_id, k_from_pt_id, k_status, k_close_date, k_circle,
                       k_circle_internal, k_percent_per_day, k_memo, k_amount_approved, k_percent_for_days,
                       k_kolich_client_applicate, k_kolich_client_approved, k_repay_kolich_closed, k_class,
                       k_kommis_perc, k_from_cs_id, k_clients_leader_id, k_clients_cashier_id, k_credit_target,
                       k_from_pac_id, k_is_kredline, k_kredline_end, k_stop_perc, k_stop_fine, k_index_curr_code,
                       k_index_curr_rate, k_from_ma_reg_id, k_from_ma_app_id, k_percent_input, k_percent_category, c_id,
                       c_name_fio, c_sex, c_birth_date, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire,
                       c_from_reg_id,
                       (select cont_value
                        from hp_contdata_get_email_main(0, :cc_id)), c_kol_family, c_kol_child, c_memo, c_reg_no,
                       c_okpo, c_inn, c_account, c_hired_date, c_tabel,
                       (select cont_value
                        from hp_contdata_get_cell_main(0, :cc_id)), c_others, adl_address_str, kc_from_k_id, kc_number,
                       kc_from_c_id, kc_amount_applicate, kc_from_pt_id, kc_business_type, kc_is_new_business,
                       kc_is_registered, kc_business_name, kc_monthly_oborot, kc_kapital, kc_month_profit,
                       kc_amount_approved, kc_work_kolich, kc_work_kolich_family, kc_is_business_full_time,
                       kc_busuness_subtype, kc_is_new_added, kc_status, kc_circle, kc_patent_no, kc_patent_date,
                       bt_name, bs_name, pr_name, pr_loan_type

                from product, bustype, bussubtype, knclient, clients, kontract, addrlist, addrclient
                where k_id = :cc_k_id and
                      kc_from_k_id = :cc_k_id and
                      kc_from_c_id = c_id and
                      c_id = :cc_id and
                      kc_busuness_subtype = bs_id and
                      kc_business_type = bt_id and
                      k_from_pr_id = pr_id and
                      adc_address_type = 1 and
                      adc_from_adl_id = adl_id and
                      adc_from_c_id = c_id
                into :k_id, :k_from_kf_id, :k_currency, :k_applicate_date, :k_applicate_no, :k_kontract_no,
                     :k_kontract_date, :k_from_pr_id, :k_disb_date, :k_amount_applicate, :k_stage, :k_period,
                     :k_kolich_lgota, :k_percent, :k_percent_lgota, :k_kolich_repay, :k_from_f_id, :k_from_pt_id,
                     :k_status, :k_close_date, :k_circle, :k_circle_internal, :k_percent_per_day, :k_memo,
                     :k_amount_approved, :k_percent_for_days, :k_kolich_client_applicate, :k_kolich_client_approved,
                     :k_repay_kolich_closed, :k_class, :k_kommis_perc, :k_from_cs_id, :k_clients_leader_id,
                     :k_clients_cashier_id, :k_credit_target, :k_from_pac_id, :k_is_kredline, :k_kredline_end,
                     :k_stop_perc, :k_stop_fine, :k_index_curr_code, :k_index_curr_rate, :k_from_ma_reg_id,
                     :k_from_ma_app_id, :k_percent_input, :k_percent_category, :c_id, :c_name_fio, :c_sex,
                     :c_birth_date, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_from_reg_id,
                     :c_email, :c_kol_family, :c_kol_child, :c_memo, :c_reg_no, :c_okpo, :c_inn, :c_account,
                     :c_hired_date, :c_tabel, :c_plastcard_no, :c_others, :c_full_address, :kc_from_k_id, :kc_number,
                     :kc_from_c_id, :kc_amount_applicate, :kc_from_pt_id, :kc_business_type, :kc_is_new_business,
                     :kc_is_registered, :kc_business_name, :kc_monthly_oborot, :kc_kapital, :kc_month_profit,
                     :kc_amount_approved, :kc_work_kolich, :kc_work_kolich_family, :kc_is_business_full_time,
                     :kc_busuness_subtype, :kc_is_new_added, :kc_status, :kc_circle, :kc_patent_no, :kc_patent_date,
                     :bt_name, :bs_name, :pr_name, :pr_loan_type
            do
            begin

              m_from_c_id = :cc_id;
              m_last_k_id = :cc_k_id;
              m_is_black = 0;
              m_date_begin = :cc_date;
              m_month = 0;
              m_is_active = 1;
              m_comment_group = '';
              m_comment_officer = '';
              m_comment_manager = '';
              m_client_fio = :c_name_fio;
              m_client_pass_no = :c_pass_no;
              m_client_address = :c_full_address;
              m_client_birth_date = :c_birth_date;
              m_our_clients = 1;
              m_motive = 'Не заполнена форма моратория';
              m_is_force_text = 'По собств. желанию';
              if (pr_loan_type = 0) then
                pr_loan_type_text = 'Групповой кредит';
              if (pr_loan_type = 1) then
                pr_loan_type_text = 'Индив. кредит';
              m_morat_year = extract(year from m_date_begin);
              m_morat_month = extract(month from m_date_begin);
              m_days = cast((:report_date_end - cc_date) as integer);
              if (m_days <= 15) then
                m_days_category = 'A 7-15 дней';
              else
              if (m_days <= 30) then
                m_days_category = 'B 15-30 дней';
              else
              if (m_days <= 60) then
                m_days_category = 'C 2 месяца';
              else
              if (m_days <= 90) then
                m_days_category = 'D 3 месяца';
              else
              if (m_days <= 120) then
                m_days_category = 'E 4 месяца';
              else
                m_days_category = 'F Более 4 месяца';
              suspend;
            end
          end
        end
      end
    end
    cc_prior_id = :cc_id;
  end
end
