Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_PUSH_LOG(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    PL_ID D_INTEGER,
    PL_CREATE_DATETIME D_DATETIME,
    PL_TITLE D_STR_255,
    PL_CONTENT D_STR_LONG,
    PL_DATE_BEGIN D_DATE,
    PL_DATE_END D_DATE,
    PL_PROJECT D_STR_CODE,
    PL_FROM_UN_ID D_ID,
    PL_FROM_C_ID D_ID,
    UN_FIO D_STR_255,
    C_NAME_FIO D_STR_255)
AS
begin
    for select pl_id,
               pl_create_datetime,
               pl_title,
               pl_content,
               pl_date_begin,
               pl_date_end,
               iif(pl_project = 1, 'МБулак', iif(pl_project = 2, 'Ойво', 'Не установлен')),
               pl_from_un_id,
               pl_from_c_id,
               un.un_fio,
               c.c_name_fio
    from push_log pl
    join clients c on c.c_id = pl.pl_from_c_id
    join username un on un.un_id = pl.pl_from_un_id
    where pl.pl_create_datetime between :rep_date_begin and :rep_date_end
    into :pl_id,
         :pl_create_datetime,
         :pl_title,
         :pl_content,
         :pl_date_begin,
         :pl_date_end,
         :pl_project,
         :pl_from_un_id,
         :pl_from_c_id,
         :un_fio,
         :c_name_fio
    do
        suspend;
end
