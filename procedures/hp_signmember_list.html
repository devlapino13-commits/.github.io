CREATE PROCEDURE HP_SIGNMEMBER_LIST(
    SGN_ID D_INTEGER,
    DT_CURR D_DATE = null)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    IS_SIGN D_INTEGER,
    SIGN_DATE D_DATETIME)
AS
declare variable sgn_edl_id d_id;
declare variable sm_kind d_smallint;
declare variable sm_obj_id d_id;
declare variable numb d_integer;
begin
  numb = 0;
  delete from temp_cli where 1 = 1;

  if (dt_curr is null) then
    dt_curr = (select sgn_datetime from signdoc where sgn_id = :sgn_id);

  for
    select
      sgn_edl_id,
      sm_kind,
      sm_obj_id
    from signdoc
      inner join signmember
        on sm_sgn_id = sgn_id
    where sgn_id = :sgn_id
  into
    :sgn_edl_id,
    :sm_kind,
    :sm_obj_id
  do
  begin
    -- сотрудник
    if (sm_kind = 0) then
    for
      select
        c_id
      from clients
        inner join hr_pstatus
          on ps_from_c_id = c_id
            and :dt_curr between ps_date_begin and ps_date_end
            and ps_status = 1
            and ps_is_working = 1
      where c_id = :sm_obj_id
    into
      :c_id
    do
    begin
      numb = numb + 1;
      insert into temp_cli (tc_number, tc_from_c_id)
      values (:numb, :c_id);
    end

    -- должность
    if (sm_kind = 1) then
    for
      select
        c_id
      from joblist
        inner join hr_pstatus
          on ps_from_jl_id = jl_id
            and :dt_curr between ps_date_begin and ps_date_end
            and ps_status = 1
            and ps_is_working = 1
        inner join clients
          on c_id = ps_from_c_id
      where jl_id = :sm_obj_id
    into
      :c_id
    do
    begin
      numb = numb + 1;
      insert into temp_cli (tc_number, tc_from_c_id)
      values (:numb, :c_id);
    end

    -- отдел
    if (sm_kind = 2) then
    for
      select
        c_id
      from depart
        inner join hr_pstatus
          on ps_from_dep_id = dep_id
            and :dt_curr between ps_date_begin and ps_date_end
            and ps_status = 1
            and ps_is_working = 1
        inner join clients
          on c_id = ps_from_c_id
      where dep_id = :sm_obj_id
    into
      :c_id
    do
    begin
      numb = numb + 1;
      insert into temp_cli (tc_number, tc_from_c_id)
      values (:numb, :c_id);
    end

    -- регион
    if (sm_kind = 3) then
    for
      select
        c_id
      from region_list(:sm_obj_id, 1, 3, 1)
        inner join hr_pstatus
          on ps_from_reg_id = id_list
            and :dt_curr between ps_date_begin and ps_date_end
            and ps_status = 1
            and ps_is_working = 1
        inner join clients
          on c_id = ps_from_c_id
    into
      :c_id
    do
    begin
      numb = numb + 1;
      insert into temp_cli (tc_number, tc_from_c_id)
      values (:numb, :c_id);
    end
  end

  for
    select
      tc_from_c_id
    from temp_cli
    group by tc_from_c_id
  into
    :c_id
  do
  begin
    is_sign = 0;
    sign_date = null;
    c_name_fio = (select c_name_fio
      from clients where c_id = :c_id);

    if (sgn_edl_id > 0) then
    for
      select
        eds_status,
        eds_status_datetime
      from edoc_sign
      where eds_from_edl_id = :sgn_edl_id
        and eds_from_c_id = :c_id
        and eds_status = 1
    into
      :is_sign,
      :sign_date
    do
    begin
    end

    suspend;
  end
end
