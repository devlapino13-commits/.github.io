CREATE PROCEDURE HP_GET_CLIENT_STATUS_STR(
    REP_C_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    C_LAST_STATUS INTEGER,
    C_CURR_STATUS INTEGER,
    C_LAST_STATUS_CLASSIC INTEGER,
    C_LAST_STATUS_IPF INTEGER,
    C_CURR_STATUS_CLASSIC INTEGER,
    C_CURR_STATUS_IPF INTEGER,
    POLEZ_PLATEJ_TOTAL INTEGER,
    C_LAST_STATUS_STR VARCHAR(100),
    C_CURR_STATUS_STR VARCHAR(100),
    C_LAST_STATUS_CLASSIC_STR VARCHAR(100),
    C_LAST_STATUS_IPF_STR VARCHAR(100),
    C_CURR_STATUS_CLASSIC_STR VARCHAR(100),
    C_CURR_STATUS_IPF_STR VARCHAR(100))
AS
begin
  select c_last_status,
         c_curr_status,
         c_last_status_classic,
         c_last_status_ipf,
         c_curr_status_classic,
         c_curr_status_ipf,
         polez_platej_total
  from hp_get_client_status(:rep_c_id, :rep_date_end)
  into :c_last_status,
       :c_curr_status,
       :c_last_status_classic,
       :c_last_status_ipf,
       :c_curr_status_classic,
       :c_curr_status_ipf,
       :polez_platej_total;

  c_last_status_str = 'Балл: '||c_last_status;
  c_curr_status_str = 'Балл: '||c_curr_status;

  c_last_status_classic_str = 'Балл: '||c_last_status_classic;
  c_last_status_ipf_str = 'Балл: '||c_last_status_ipf;

  c_curr_status_classic_str = 'Балл: '||c_curr_status_classic;
  c_curr_status_ipf_str = 'Балл: '||c_curr_status_ipf;
  suspend;
end
