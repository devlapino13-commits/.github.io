CREATE PROCEDURE HP_DO_EXPRESSAPP_ACT_SEND_KIB(
    EXA_ID D_INTEGER)
RETURNS (
    RES_STATUS D_SMALLINT)
AS
declare variable ma_id d_id;
declare variable un_id d_id;
declare variable tmp_int d_integer;
declare variable obj_kind d_str_10;
declare variable res_text d_blob_text;
begin
  obj_kind = f_objkind('miniapp', 'code');
  res_status = 0;
  ma_id = null;
  un_id = null;
  select exa.exa_from_ma_id, exa.exa_from_un_id
  from expressapp exa
  where exa.exa_id = :exa_id
  into ma_id, un_id;

  select res_text
  from fm_send_to_kib(:obj_kind, :ma_id, :un_id)
  into res_text;

  if (res_text containing 'Данные успешно отправлены на проверку в КИБ') then
  res_status = 0;
  else
  if (exists(select *
             from kibminiapp kbm
             where kbm.kbm_ma_id = :ma_id and
                   kbm.kbm_is_checked = 1 )) then
  res_status = 0;
  else
  res_status = -1;

  suspend;
end
