CREATE PROCEDURE HP_CHECK_CAN_SIGN_STAFF(
    UN_ID DECIMAL(18,0),
    SIGN_DATE DATE)
RETURNS (
    CAN_SIGN INTEGER,
    CAN_SIGN_TEXT VARCHAR(255))
AS
declare variable c_id decimal(18,0);
declare variable cm_database_id integer;
begin
  can_sign = 0;
  can_sign_text = '';
  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  :cm_database_id = f_get_db_id();
  if (cm_database_id = 94401 and
      un_id = 1010100000001) then
  begin
    can_sign = 1;
    suspend;
    exit;
  end

  if (exists(select has_access
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Superuser')
             where has_access = 0)) then
  begin
    if (c_id is not null) then
    begin
      if (exists(select *
                 from hr_pstatus
                 where ps_from_c_id = :c_id and
                       ps_date_begin <= :sign_date and
                       ps_date_end >= :sign_date and
                       ps_is_working = 1)) then
      begin
        if (exists(select *
                   from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Alien_Personnel_Orders')
                   where has_access = 1) or exists(select *
                                                   from dovlist, dovclients, dovcateg
                                                   where dvc_from_dl_id = dl_id and
                                                         dvc_from_c_id = :c_id and
                                                         dl_date_begin <= :sign_date and
                                                         dl_date_end >= :sign_date and
                                                         (dl_date_end_fact is null or
                                                         dl_date_end_fact >= :sign_date) and
                                                         dl_from_dc_id = dc_id and
                                                         dc_sign_staff = 1)) then
        begin
          can_sign = 1;
        end
        else
          can_sign_text = 'У Вас нет действующей доверенности позволяющая утверждать кадровые приказы';
      end
      else
      begin
        can_sign_text = 'У Вас нет действующего трудового договора';
      end
    end
    else
      can_sign_text = 'Ваш логин не привязан к контрагентам';
  end
  else
    can_sign = 1;
  suspend;
end
