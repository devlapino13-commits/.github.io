CREATE PROCEDURE HP_SIRENA_SB_AKT_UNBLOCK(
    K_ID ID_TYPE,
    UNBLOCK_DATE D_DATE = null)
AS
declare variable SBA_DATE_BEGIN D_DATE;
declare variable SBL_ID D_INTEGER;
declare variable PANT_ID ID_TYPE;
declare variable TRANSFER_REASON_CODE STR_100;
declare variable MESS_STR STR_255;
declare variable RD_PROJECT_TYPE STR_100;
begin
  for select sba.sba_date_begin, sba.sba_from_sbl_id
      from sb_akt sba
      where sba.sba_from_k_id = :k_id and
            sba.sba_status > 0 and
            sba.sba_is_delete = 0 and
            sba.sba_date_begin <= sba.sba_date_end and
            sba_date_end = '31.12.9999'
      into :sba_date_begin, :sbl_id -- на какой департамен передано
  do
  begin
    pant_id = null;

    for select pantk.pant_id, rd.rd_code
        from pantera_kontracts pantk
        inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
        where pantk.pant_from_k_id = :k_id and
              pantk.pant_date_end = '30.12.9999'
        into :pant_id, :rd_project_type -- на комом департаменте сейчас висит группа, по КЦ
    do
    begin
      transfer_reason_code = (select rd.rd_code
                              from refdata rd
                              where rd.rd_main_code = 'SirenaTransferReason'
                              and rd.rd_str_id = cast(:sbl_id as varchar(10)));

      transfer_reason_code = coalesce(transfer_reason_code, 'SirenaTransferReason.ToSB');

      execute procedure okt_sirena_make_call(1, iif(:unblock_date is null, :sba_date_begin - 1, unblock_date), 'UNBLOCK', :transfer_reason_code, :rd_project_type,
          :pant_id, null, 0, null, null, null, null, null, null, null, null, 0)
      returning_values :mess_str;
    end
  end
end
