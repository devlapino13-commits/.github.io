CREATE PROCEDURE AAREP_CELLBILL_LIST(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    CBL_DATE DATE,
    CBL_BILL_NO VARCHAR(20),
    CBL_ID INTEGER,
    CBL_CELL_NO VARCHAR(20),
    CBL_AMOUNT DECIMAL(18,4),
    CBL_USER_NAME VARCHAR(30),
    CBL_TIME TIMESTAMP,
    CBL_OPERATOR VARCHAR(50))
AS
begin
  for select cb_date, cbl_bill_no, cbl_id, cbl_cell_no, cbl_amount, un_name, cbl_time, cop_name
      from cellbilloper, cellbill, celloperators, username
      where cbl_bill_no = cb_bill_no and
            cb_from_cop_id = cop_id and
            cbl_user_id = un_id and
            cbl_amount <> 0 and
            cb_date >= :rep_date_begin and
            cb_date <= :rep_date_end
      into :cbl_date, :cbl_bill_no, :cbl_id, :cbl_cell_no, :cbl_amount, :cbl_user_name, :cbl_time, :cbl_operator
  do
  begin
    suspend;
  end
end
