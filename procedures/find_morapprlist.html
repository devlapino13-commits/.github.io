CREATE PROCEDURE FIND_MORAPPRLIST(
    FM_REGION_ID INTEGER,
    FM_CLIENT_FIO VARCHAR(255),
    FM_APPR_STATUS SMALLINT)
RETURNS (
    M_ID ID_TYPE,
    M_NUMBER INTEGER,
    MAL_ID INTEGER,
    MAL_APPR_STATUS SMALLINT,
    MAL_REQ_STATE SMALLINT,
    M_IS_ACTIVE SMALLINT,
    M_CLIENT_FIO VARCHAR(100),
    M_CLIENT_PASS_NO VARCHAR(30),
    M_CLIENT_ADDRESS VARCHAR(255),
    M_CLIENT_BIRTH_DATE DATE,
    M_FROM_REG_ID INTEGER,
    M_NO INTEGER,
    M_COMMENT_BL VARCHAR(2048),
    MAL_REQ_COMMENT VARCHAR(512),
    MAL_APPR_COMMENT VARCHAR(256),
    REG_NAME VARCHAR(50),
    MAL_REQ_DATETIME TIMESTAMP,
    MAL_APPR_DATETIME TIMESTAMP,
    C_ID DECIMAL(18,0),
    UN_FIO D_STR_100)
AS
declare variable mal_appr_user d_id;
declare variable db_id d_id;
begin
  db_id = f_get_db_id();
  if (coalesce(fm_client_fio, '') = '') then
  begin
    for select m_id, m_number, mal_id, mal_appr_status, mal_req_state, m_is_active, m_client_fio, m_client_pass_no,
               m_client_address, m_client_birth_date, mal_from_reg_id, m_no,
               left(coalesce('Клиент: ' || m_comment_group || '; ', '') || coalesce('Сотрудник: ' || m_comment_officer || '; ', '') || coalesce('Менеджер: ' || m_comment_manager, ''), 255) m_comment_bl,
               mal_req_comment, mal_appr_comment, reg_name, mal_req_datetime, mal_appr_datetime, m_from_c_id,
               mal_appr_user
        from morapprlist m, moratory, region, region_list(:fm_region_id, 1, -1, 1)
        where mal_from_m_id = m_id and
              mal_from_m_number = m_number and
              mal_from_reg_id = reg_unicode and
              mal_from_reg_id = id_list and
              (:fm_appr_status = mal_appr_status or
              :fm_appr_status = 3)
              rows iif (:db_id in (39801, 41701), 1000, 50)
        into m_id, m_number, mal_id, mal_appr_status, mal_req_state, m_is_active, m_client_fio, m_client_pass_no,
             m_client_address, m_client_birth_date, m_from_reg_id, m_no, m_comment_bl, mal_req_comment,
             mal_appr_comment, reg_name, mal_req_datetime, mal_appr_datetime, c_id, mal_appr_user
    do
    begin
      un_fio = '';

      if (coalesce(mal_appr_user, 0) > 0) then
      begin
        select c_name_fio
        from username
        inner join clients on c_id = un_from_c_id
        where un_id = :mal_appr_user
        into :un_fio;
      end

      suspend;
    end
  end
  else
  begin
    for select distinct m_id, m_number, mal_id, mal_appr_status, mal_req_state, m_is_active, m_client_fio,
                        m_client_pass_no, m_client_address, m_client_birth_date, mal_from_reg_id, m_no,
                        left(coalesce('Клиент: ' || m_comment_group || '; ', '') || coalesce('Сотрудник: ' || m_comment_officer || '; ', '') || coalesce('Менеджер: ' || m_comment_manager, ''), 255) m_comment_bl,
                        mal_req_comment, mal_appr_comment, reg_name, mal_req_datetime, mal_appr_datetime, m_from_c_id, mal_appr_user
        from morapprlist, moratory, hp_find_from_namelist(0, 14, :fm_client_fio), region, region_list(:fm_region_id, 1, -1, 1)
        where mal_from_m_id = m_id and
              mal_from_m_number = m_number and
              m_id = object_id and
              mal_from_reg_id = reg_unicode and
              mal_from_reg_id = id_list and
              (:fm_appr_status = mal_appr_status or
              :fm_appr_status = 3) rows 50
        into m_id, m_number, mal_id, mal_appr_status, mal_req_state, m_is_active, m_client_fio, m_client_pass_no,
             m_client_address, m_client_birth_date, m_from_reg_id, m_no, m_comment_bl, mal_req_comment,
             mal_appr_comment, reg_name, mal_req_datetime, mal_appr_datetime, c_id, mal_appr_user
    do
    begin
      un_fio = '';

      if (coalesce(mal_appr_user, 0) > 0) then
      begin
        select c_name_fio
        from username
        inner join clients on c_id = un_from_c_id
        where un_id = :mal_appr_user
        into :un_fio;
      end

      suspend;
    end
  end
end
