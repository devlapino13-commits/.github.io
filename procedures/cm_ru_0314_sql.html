CREATE PROCEDURE CM_RU_0314_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;  
declare variable ma_id d_id;
declare variable kc_id d_id;
declare variable disb_date d_date;       
declare variable ma_date d_date;
declare variable cli_balance curr_type;
declare variable amount curr_type;
declare variable delinq_balance d_integer;
declare variable office_id d_integer; 
declare variable external_balance d_integer;
begin
  res_error_is_critical = 2;

  for select kc_id, kc_from_c_id, k_disb_date, k_fact_disb_reg_id, kc_amount_approved, k_from_ma_app_id
      from kontract
      inner join knclient on kc_from_k_id = k_id
      inner join product on pr_id = k_from_pr_id
      where k_id = :k_id and
            pr_code containing '.b_refinance.' and
            kc_amount_approved > 0
      into kc_id, c_id, disb_date, office_id, amount, ma_id
  do
  begin
    if (office_id in (1490600, 1050600)) then
    begin
      res_error_text = 'В текущем офисе выдача запрещена';
      suspend;
      exit;
    end

    --заем может быть выдан только 1 раз за акцию
    if (exists
       (select 1
        from kontstate, knclient, kontract, product
        where ks_ol_is_delete = 0 and
              ks_state >= 1 and
              ks_from_k_id = k_id and
              k_from_pr_id = pr_id and
              ks_from_k_id = kc_from_k_id and
              :disb_date between ks_date_begin and ks_date_end and
              kc_amount_approved > 0 and
              kc_from_c_id = :c_id and
              pr_code containing '.b_refinance.')) then
    begin
      res_error_text = f_format('Заем может быть выдан не более 1 раза');
      suspend;
    end

    if (exists(select *
               from schedule sh
               where sh.sh_from_k_id = :k_id and
                     sh.sh_id > 0 and
                     sh.sh_os = 0 and
                     sh.sh_grace = 1 and
                     extract(month from sh_date) not in (1, 2))) then
    begin
      res_error_text = 'По продукту льготный период можно получить только на январь и/или февраль ';
      suspend;
    end 

    --    Заем может быть выдан при условии, что после его выдачи общая задолженность клиента по основному долгу не превысит 500 000 руб.
    cli_balance = 0;
    delinq_balance = 0;
    select sum(oks_balance), sum(iif(:disb_date > oks_delinq_date, oks_delinq_os + oks_delinq_perc, 0))
    from knclient, oksummary
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = oks_from_k_id and
          oks_balance > 0 and
          oks_close_state = 0 and
          :disb_date between oks_date_begin and oks_date_end
    into cli_balance, delinq_balance;
    cli_balance = coalesce(cli_balance, 0);
    delinq_balance = coalesce(delinq_balance, 0);

    if (cli_balance + amount > 500000) then
    begin
      res_error_text = 'Общий баланс всех займов превысит 500 000 р.';
      suspend;
    end

    if (delinq_balance > 0) then
    begin
      res_error_text = 'У клиента имеется активная просрочка';
      suspend;
    end

    select ma_date_app
    from miniapp
    where ma_id = :ma_id and
          ma_is_active = 1 and
          ma_date_end = '31.12.9999'
    into :ma_date;

    if (check_kind = '.send_kk.' and
      not exists (select 1
                  from clientdocs cds
                  where cds_from_c_id = :c_id
                    and cds_from_pk_id = 337
                    and cds_is_delete = 0
                    and (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0)
                    and cds.cds_doc_date_begin >= :ma_date)) then
    begin
      res_error_text = 'Не приложен документ о наличии займа в другом МФО';
      suspend;
    end

    external_balance = 0;

    select sum(coalesce(kpl.kpl_debtcurr_opsum, 0) +
               coalesce(kpl.kpl_debtcurr_percentsum, 0) +
               coalesce(kpl.kpl_debtoverdue_opsum, 0) +
               coalesce(kpl.kpl_debtoverdue_percentsum, 0) +
               coalesce(kpl.kpl_debtoverdue_othersum, 0))
    from knclient_pdn kcp
    join knclient_pdnlist kpl on kpl.kpl_from_kcp_id = kcp.kcp_id
    where kcp.kc_id = :kc_id
      and kpl.kpl_history_kc_id is null
      and (kpl.kpl_credsumdebt > 0
        or kpl.kpl_credenddatefact is null 
        or kpl.kpl_credenddatefact >= :ma_date
        or kpl.kpl_credsumoverdue > 0)
    into :external_balance;

    external_balance = coalesce(:external_balance, 0);

    if (amount > external_balance + 50000) then
    begin
      res_error_text = 'Размер займа не должен превышать cовокупный объем внешнего долга клиента: '|| external_balance ||' и дополнительных 50 000 рублей';
      suspend;
    end
  end
end
