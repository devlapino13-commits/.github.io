CREATE PROCEDURE HP_REPAYMENT_DETAIL_INFO_KZ(
    K_ID D_ID,
    REP_DATE D_DATE,
    KONT_NO D_STR_CODE = null)
RETURNS (
    REP_DATE_OUT D_DATE,
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    K_KONTRACT_NO D_STR_CODE,
    K_KONTRACT_DATE D_DATE,
    K_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    CU_SYMBOL D_STR_20,
    K_PERIOD INTEGER,
    K_PERCENT_YEAR DECIMAL(18,2),
    K_PERCENT_EFFECT DECIMAL(18,2),
    KOMISS DECIMAL(18,2),
    PENALTY DECIMAL(18,2),
    DELINQ_DAYS D_STR_20,
    OS_LOAN DECIMAL(18,2),
    PERC_LOAN DECIMAL(18,2),
    FINE_LOAN DECIMAL(18,2),
    GOSP_LOAN DECIMAL(18,2),
    BALANCE_BEGIN DECIMAL(18,2),
    MAIL DECIMAL(18,2),
    TOTAL DECIMAL(18,2))
AS
declare variable k_percent_per_day d_percent;
declare variable rd_id_exec id_type;
begin
  if (k_id is null and kont_no <> '') then
  begin
    select k_id
    from kontract
    where k_kontract_no = :kont_no
    into k_id;
  end

  select c_name_fio, c_inn, k_kontract_no, k_kontract_date, k_disb_date, k_amount_approved,
         cu_symbol, k_period, k_percent_effect, k_percent_per_day
  from kontract
  inner join knclient on kc_from_k_id = k_id
  inner join clients on c_id = kc_from_c_id
  inner join currlist on k_currency = cu_id
  where k_id = :k_id
  and kc_amount_approved > 0
  into c_name_fio, c_inn, k_kontract_no, k_kontract_date, k_disb_date, k_amount_approved,
       cu_symbol, k_period, k_percent_effect, k_percent_per_day;

  k_percent_year = round(k_percent_per_day * 365.0, 2);
  komiss  = 0;
  penalty = 0.5;
  mail    = 2300;

  delinq_days = 0;
  select iif((cast(:rep_date as date) - oks_delinq_date < 0), 0,
              cast(:rep_date as date) - oks_delinq_date)
  from oksummary
  where oks_from_k_id = :k_id and
        oks_date_begin <= :rep_date and
        oks_date_end   >= :rep_date and
        oks_close_state = 0
  into delinq_days;

  os_loan   = 0;
  perc_loan = 0;
  fine_loan = 0;

  select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine)
  from aarep_make_zad_list(:k_id, :rep_date, 1)
  where sh_date is not null
  into :os_loan, :perc_loan, :fine_loan;

  os_loan   = coalesce(os_loan,   0);
  perc_loan = coalesce(perc_loan, 0);
  fine_loan = coalesce(fine_loan, 0);

  rd_id_exec    = f_get_rd_id('OperExtraKind.ExecutiveInscription');
  balance_begin = 0;

  select sum(o.ox_amount_plus)
  from operextra o
  inner join operlist ol on ol.ol_id = o.ox_from_ol_id
  where o.ox_object_kind = 9 and
        o.ox_object_id = :k_id and
        cast(ol.ol_datetime as date) <= :rep_date and
        o.ox_ol_is_active = 1 and
        o.ox_from_rd_id = :rd_id_exec
  into :balance_begin;

  balance_begin = coalesce(balance_begin, 0);

  gosp_loan = ceiling((os_loan + perc_loan + fine_loan) * 0.03);

  total = os_loan + perc_loan + fine_loan + balance_begin + gosp_loan + mail;
  rep_date_out = rep_date;
  suspend;
end
