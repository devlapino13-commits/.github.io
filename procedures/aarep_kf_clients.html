CREATE PROCEDURE AAREP_KF_CLIENTS(
    KF_ID DECIMAL(18,0))
RETURNS (
    KF_CLIENT_ID DECIMAL(18,0),
    KF_CLIENT_NAME VARCHAR(100),
    KF_CLIENT_PASS_NO VARCHAR(30),
    KF_AMOUNT DECIMAL(18,4),
    KF_DISB_DATE DATE,
    KF_STATE_TEXT VARCHAR(20))
AS
declare variable kf_state integer;
declare variable c_id_prior decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */
  c_id_prior = 0;
  for select distinct c_id, c_name_fio, c_pass_no, kc_amount_approved, k_disb_date, ks_state
      from clients, knclient, kontract, kontstate
      where c_id = kc_from_c_id and
            kc_from_k_id = k_id and
            k_id = ks_from_k_id and
            ks_date_end = '31.12.9999' and
            k_from_kf_id = :kf_id and
            kc_amount_approved <> 0
      order by c_id, k_disb_date desc
      into :kf_client_id, :kf_client_name, kf_client_pass_no, :kf_amount, :kf_disb_date, :kf_state
  do
  begin
    if (kf_state = 0) then
      kf_state_text = 'В процессе';
    if (kf_state = 1) then
      kf_state_text = 'Активный';
    if (kf_state = 2) then
      kf_state_text = 'Закрыт';
    if (c_id_prior <> kf_client_id) then
      suspend;
    c_id_prior = kf_client_id;
  end
end
