CREATE PROCEDURE KFOLDER_INFO_KONTRACT(
    ID DECIMAL(18,0))
RETURNS (
    KFI_ID DECIMAL(18,0),
    KFI_OFFICER_ID DECIMAL(18,0),
    KFI_OFFICER_NAME VARCHAR(100),
    KFI_NAME VARCHAR(100),
    KFI_REG_ID INTEGER,
    KFI_REG_NAME VARCHAR(100),
    KFI_CODE VARCHAR(20),
    KFI_NO INTEGER,
    KFI_DATE DATE,
    KFI_REG_CODE VARCHAR(10),
    KFI_KONT_COMPANY VARCHAR(255),
    KFI_REG_NAME_CURR VARCHAR(100))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */
  for select kal1.kal_from_kf_id, kal1.kal_new_kf_officer, c_name_fio, kal1.kal_new_kf_name, kal1.kal_new_reg_id,
             reg1.reg_name, kal1.kal_new_officer_code, kal1.kal_new_no, kal1.kal_date, reg1.reg_code,
             reg1.reg_kont_company, reg2.reg_name
      from knaktlist kal1, clients, region reg1, kontract, knaktlist kal2, region reg2
      where kal1.kal_new_kf_officer = c_id and
            kal1.kal_new_reg_id = reg1.reg_unicode and
            kal1.kal_from_kf_id = k_from_kf_id and
            k_id = :id and
            kal1.is_delete = 0
            /* and kal1.KAL_Date<=k_disb_date */
            and
            kal1.kal_date_end >= k_disb_date and
            kal2.kal_from_kf_id = k_from_kf_id and
            current_date between kal2.kal_date and kal2.kal_date_end and
            kal2.kal_new_reg_id = reg2.reg_unicode
      into :kfi_id, :kfi_officer_id, :kfi_officer_name, :kfi_name, :kfi_reg_id, :kfi_reg_name, :kfi_code, :kfi_no,
           :kfi_date, :kfi_reg_code, :kfi_kont_company, :kfi_reg_name_curr
  do
  begin
    suspend;
  end
end
