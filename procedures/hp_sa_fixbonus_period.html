CREATE PROCEDURE HP_SA_FIXBONUS_PERIOD(
    REP_DATE_END D_DATE)
RETURNS (
    SFB_ID D_INTEGER,
    SFB_PERIOD D_INTEGER,
    SFB_FIO D_STR_100,
    SFB_FROM_C_ID D_ID,
    SFB_TOTAL_WORK D_INTEGER,
    SFB_FACT_WORK D_INTEGER,
    SFB_FIX_BONUS D_CURRENCY,
    RES_FIO D_STR_100,
    RES_OFFICE D_STR_100,
    RES_JOB D_STR_100,
    RES_DATE_BEGIN D_DATE,
    RES_DATE_END D_DATE,
    RES_STATUS D_STR_CODE,
    RES_HIRE_DATE D_DATE,
    RES_FOUND D_BOOL)
AS
declare variable rep_period d_integer;
begin
  rep_period = extract(year from rep_date_end) * 100 + extract(month from rep_date_end);

  for select sfb_id,
             sfb_period,
             sfb_fio,
             sfb_from_c_id,
             sfb_total_work,
             sfb_fact_work,
             sfb_fix_bonus,
             c_name_fio
      from sa_fixbonus
           left join clients on (sfb_from_c_id = c_id)
      where sfb_period = :rep_period
      into :sfb_id,
           :sfb_period,
           :sfb_fio,
           :sfb_from_c_id,
           :sfb_total_work,
           :sfb_fact_work,
           :sfb_fix_bonus,
           :res_fio
  do
  begin
    res_hire_date = null;
    res_office = '';
    res_job = '';
    res_date_begin = null;
    res_date_end = null;
    res_status = '';
    res_found = 0;

    if (coalesce(res_fio, '') <> '') then
    begin
      res_found = 1;
      sfb_fio = res_fio;

      select reg_code || ' ' || reg_name, jl_name,
             ps_date_begin, ps_date_end,
             trim(iif(ps_is_working = 0, 'Уволен ' || f_datetostr(ps_date_begin), ''))
      from hr_pstatus, region, joblist
      where ps_from_c_id = :sfb_from_c_id and
            ps_status = 1 and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id
      into res_office, res_job,
           res_date_begin, res_date_end,
           res_status;

      if (row_count = 1) then
      begin
        select ps_date_begin
        from hr_pstatus, hr_orders
        where ps_from_c_id = :sfb_from_c_id and
              ps_date_begin <= :rep_date_end and
              ps_status = 1 and
              ps_from_ho_id = ho_id and
              ho_order_type = 1
        order by ps_date_begin desc
        rows 1
        into res_hire_date;
      end
    end

    suspend;
  end
end
