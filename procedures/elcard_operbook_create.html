CREATE PROCEDURE ELCARD_OPERBOOK_CREATE(
    ESM_ID ID_TYPE)
AS
declare variable k_kontract_no str_100;
declare variable id_no id_type;
declare variable ol_id id_type;
declare variable amount curr_type;
declare variable ob_memo varchar(1024);
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable esa_message d_blob_text;
declare variable doc_no d_integer;
declare variable acc_dt_id d_id;
declare variable acc_kt_id d_id;
declare variable moneta_acc d_str_code;
declare variable k_id d_id;
declare variable reg_id d_integer;
declare variable parent_ol_id d_id;
declare variable pr_code d_str_100;
declare variable acc d_str_20;
declare variable esm_date d_date;
declare variable ol_memo d_str_255;
declare variable al_number d_integer;
declare variable reg_code d_str_100;
begin
  esa_message = '';

  select em.esm_amount, kont.k_fact_disb_reg_id, pr.pr_code, kont.k_kontract_no, em.esm_date, reg.reg_code
  from extsysmain em
  inner join kontract kont on kont.k_id = em.esm_from_k_id
  inner join product pr on kont.k_from_pr_id = pr.pr_id
  inner join region reg on reg.reg_unicode = kont.k_fact_disb_reg_id
  where em.esm_id = :esm_id
  and em.esm_project = 'elcard_disb'
  and em.esm_action = 'OperbookCreate'
  and em.esm_status = 0
  into :amount, :reg_id, :pr_code, :k_kontract_no, :esm_date, :reg_code;

  ob_memo = 'Комиссия банка 0,25% по Зачислению на пл.карту Элкарт суммы займа по договору ' || k_kontract_no;
  ol_memo = 'Комиссия за выдачу на карту Элькарт';

  al_number = 1;
  if (pr_code like '.KARDHASAN.') then
    al_number = 101;

  -- 0 . ======== создаем родительский Operlist ============================== >
  select kv
  from get_kv(1090100, 5) -- Бишкек ГО
  into :id_no;

  select num
  from last_doc_number(:esm_date, 9, 1090100)
  into :doc_no;

  parent_ol_id = :id_no;

  insert into operlist (ol_id, ol_doc_no, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete, ol_user_id)
  values (:parent_ol_id, :doc_no, 1090100, 2, :esm_date, 9, :ol_memo, 0, 0, 1000000000000);
  -- ========================================================================= <
  -- 1 . ============= создаем операцию и проводку в офисе выдачи займа ====== >
  -- создаем операцию (дочернюю) в офисе выдачи займа ------------------------->
  select kv
  from get_kv(:reg_id, 5)
  into :id_no;

  select num
  from last_doc_number(:esm_date, 9, :reg_id)
  into :doc_no;

  ol_id = :id_no;

  insert into operlist (ol_id, ol_parent_id,  ol_doc_no, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete, ol_user_id)
  values (:ol_id, :parent_ol_id, :doc_no, :reg_id, 2, :esm_date, 9, :ol_memo, 0, 0, 1000000000000);
  -----------------------------------------------------------------------------<
  -- создаем проводку --- офис: офис выдачи займа, Дт 80509/..., Кт 11510/2 ----->
  acc_dt = '';
  acc_kt ='';


  if (pr_code like '.KARDHASAN.') then
    acc = '80509/101';
  else
    acc = '80509/1';

  select al_id
  from find_acc_list(:acc)
  where al_from_reg_id = :reg_id
  and al_number = :al_number
  into :acc_dt;

  select al_id
  from find_acc_list('11510/2')
  where al_from_reg_id = :reg_id

  into :acc_kt;

  insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                        ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                        ob_from_sac_id)
  values (:ol_id, 1, '417', 1, null, :acc_dt, :amount, :amount, null, null, null, :acc_kt, null, :ob_memo, null);
  -----------------------------------------------------------------------------<
  -- ========================================================================= <
  -- 2 . ============= создаем операцию и проводку в Бишкекском офисе ======== >
  -- создаем операцию (дочернюю) в Бишкекском офисе --------------------------->
  select kv
  from get_kv(1060100, 5) -- Бишкекский офис
  into :id_no;

  select num
  from last_doc_number(:esm_date, 9, 1060100)
  into :doc_no;

  ol_id = :id_no;

  insert into operlist (ol_id, ol_parent_id, ol_doc_no, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete, ol_user_id)
  values (:ol_id, :parent_ol_id, :doc_no, 1060100, 2, :esm_date, 9, :ol_memo, 0, 0, 1000000000000);
  -----------------------------------------------------------------------------<
  -- создаем проводку --- офис: Бишкекский офис, Дт 11512/152, Кт 11510/1 ----->
  acc_dt = '';
  acc_kt ='';

  select al_id
  from find_acc_list('11512/')
  where al_from_reg_id = 1060100
  and al_name like '%'||:reg_code||'%'
  into :acc_dt;

  select al_id
  from find_acc_list('11510/1')
  where al_from_reg_id = 1060100
  into :acc_kt;

  insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                        ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                        ob_from_sac_id)
  values (:ol_id, 1, '417', 1, null, :acc_dt, :amount, :amount, null, null, null, :acc_kt, null, :ob_memo, null);
  -----------------------------------------------------------------------------<
  -- ========================================================================= <
  -- ========================================================================= <
  -- 3 . ============= создаем операцию и проводку в Бишкек ГО =============== >
  -- создаем операцию (дочернюю) в Бишкек ГО ---------------------------------->
  select kv
  from get_kv(1090100, 5) -- Бишкек ГО
  into :id_no;

  select num
  from last_doc_number(:esm_date, 9, 1090100)
  into :doc_no;

  ol_id = :id_no;

  insert into operlist (ol_id, ol_parent_id, ol_doc_no, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete, ol_user_id)
  values (:ol_id, :parent_ol_id, :doc_no, 1090100, 2, :esm_date, 9, :ol_memo, 0, 0, 1000000000000);
  -----------------------------------------------------------------------------<
  -- создаем проводку --- офис: Бишкек г. (ГО), Дт 11512/152, Кт 10004/2 ----->
  acc_dt = '';
  acc_kt ='';

  select al_id
  from find_acc_list('11512/5')
  where al_from_reg_id = 1090100
  into :acc_dt;

  select al_id
  from find_acc_list('10004/2')
  where al_from_reg_id = 1090100
  into :acc_kt;

  insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                        ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                        ob_from_sac_id)
  values (:ol_id, 1, '417', 1, null, :acc_dt, :amount, :amount, null, null, null, :acc_kt, null, :ob_memo, null);
  -----------------------------------------------------------------------------<
  -- ========================================================================= <

  insert into extsysanswer (esa_from_esm_id, esa_message)
  values (:esm_id, 'ol_id = ' || :parent_ol_id);
end
