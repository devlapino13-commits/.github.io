CREATE PROCEDURE PRINT_ZAL_AUTO_CL(
    REP_C_ID ID_TYPE)
RETURNS (
    STS_SERIES_NUMBER STR_255,
    STS_ISSUE_DATE STR_255,
    STS_AUTHORITY STR_255,
    STS_PLATE STR_255,
    PTS_SERIES_NUMBER STR_255,
    PTS_ISSUE_DATE STR_255,
    PTS_AUTHORITY STR_255,
    PTS_VIN STR_255,
    PTS_MODEL STR_255,
    PTS_TYPE_NAME STR_255,
    PTS_CATEGORY STR_255,
    PTS_ISSUE_YEAR STR_255,
    PTS_ENGINE_NO STR_255,
    PTS_CHASSIS_NO STR_255,
    PTS_BODY_NO STR_255,
    PTS_BODY_COLOR STR_255,
    PTS_ENGINE_POWER STR_255,
    PTS_ENGINE_CAPACITY STR_255,
    PTS_ENGINE_TYPE STR_255,
    PTS_ENVIRONMENTAL_CLASS STR_255,
    PTS_MAX_WEIGHT STR_255,
    PTS_EMPTY_WEIGHT STR_255,
    PTS_MANUFACTURER STR_255,
    PTS_APPROVAL_INFO STR_255,
    PTS_EXPORT_COUNTRY STR_255,
    PTS_CUSTOMS_ENTRY STR_255,
    PTS_CUSTOMS_LIMITATIONS STR_255)
AS
declare variable find_drl_id decimal(18,0);
declare variable c_id decimal(18,0);
begin
  c_id = :rep_c_id;
  find_drl_id = null;
  select cd.cds_from_drl_id
  from clientdocs cd
  where cd.cds_from_c_id = :c_id and
        cd.cds_is_delete = 0 and
        cds_from_drl_id > 0 and
        cd.cds_from_pk_id = 310 --Свидетельство о рег. трансп. средства (СТС)
  order by cds_doc_date_begin desc, cd.cds_datetime desc
  rows 1
  into find_drl_id;

  sts_series_number = f_docrecog_value(:find_drl_id, 'doc_no'); --Серия, № документа
  sts_issue_date = f_docrecog_value(:find_drl_id, 'date_begin'); --Дата выдачи
  sts_authority = f_docrecog_value(:find_drl_id, 'authority'); --Кем выдано
  sts_plate = f_docrecog_value(:find_drl_id, 'plate'); --Регистрационный знак

  find_drl_id = null;
  select cd.cds_from_drl_id
  from clientdocs cd
  where cd.cds_from_c_id = :c_id and
        cd.cds_is_delete = 0 and
        cds_from_drl_id > 0 and
        cd.cds_from_pk_id = 315 --Паспорт трансп. средства (ПТС)
  order by cds_doc_date_begin desc, cd.cds_datetime desc
  rows 1
  into find_drl_id;

  pts_series_number = f_docrecog_value(:find_drl_id, 'doc_no'); --Серия, № документа
  pts_issue_date = f_docrecog_value(:find_drl_id, 'date_begin'); --Дата выдачи
  pts_authority = f_docrecog_value(:find_drl_id, 'authority'); --Кем выдан
  pts_vin = f_docrecog_value(:find_drl_id, 'vin'); --Идентификационный номер (VIN)
  pts_model = f_docrecog_value(:find_drl_id, 'model'); --Марка, модель ТС
  pts_type_name = f_docrecog_value(:find_drl_id, 'type_name'); --Наименование (тип ТС)
  pts_category = f_docrecog_value(:find_drl_id, 'category'); --Категория ТС (ABCD, прицеп)
  pts_issue_year = f_docrecog_value(:find_drl_id, 'issue_year'); --Год изготовления ТС
  pts_engine_no = f_docrecog_value(:find_drl_id, 'engine_no'); --Модель, № двигателя
  pts_chassis_no = f_docrecog_value(:find_drl_id, 'chassis_no'); --Шасси (рама) №
  pts_body_no = f_docrecog_value(:find_drl_id, 'body_no'); --Кузов (кабина, прицеп) №
  pts_body_color = f_docrecog_value(:find_drl_id, 'body_color'); --Цвет кузова (кабины, прицепа)
  pts_engine_power = f_docrecog_value(:find_drl_id, 'engine_power'); --Мощность двигателя л.с. (кВт)
  pts_engine_capacity = f_docrecog_value(:find_drl_id, 'engine_capacity'); --Рабочий  объем двигателя, куб. см.
  pts_engine_type = f_docrecog_value(:find_drl_id, 'engine_type'); --Тип двигателя
  pts_environmental_class = f_docrecog_value(:find_drl_id, 'environmental_class'); --Экологический класс
  pts_max_weight = f_docrecog_value(:find_drl_id, 'max_weight'); --Разрешенная максимальная масса, кг
  pts_empty_weight = f_docrecog_value(:find_drl_id, 'empty_weight'); --Масса без нагрузки, кг
  pts_manufacturer = f_docrecog_value(:find_drl_id, 'manufacturer'); --Организация изготовитель ТС (страна)
  pts_approval_info = f_docrecog_value(:find_drl_id, 'approval_info'); --Одобрение типа ТС (№, дата, кем одобрено)
  pts_export_country = f_docrecog_value(:find_drl_id, 'exprort_country'); --Страна вывоза ТС
  pts_customs_entry = f_docrecog_value(:find_drl_id, 'customs_entry'); --Серия, №, ТД, ТПО
  pts_customs_limitations = f_docrecog_value(:find_drl_id, 'customs_limitations'); --Таможенные ограничения

  suspend;
end
