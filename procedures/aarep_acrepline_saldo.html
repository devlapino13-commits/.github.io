CREATE PROCEDURE AAREP_ACREPLINE_SALDO(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    ACC_BALANCE VARCHAR(255),
    ACC_NUMBER INTEGER,
    END_DT DECIMAL(18,4),
    END_KT DECIMAL(18,4))
AS
begin
  for select acc_balance, acc_number, sum(acc_summa)
      from aarep_acrepline_saldo_temp(:report_region_id, :report_date_end)
      group by acc_balance, acc_number
      having sum(acc_summa) <> 0
      into :acc_balance, :acc_number, :end_dt
  do
  begin
    if (end_dt >= 0) then
    begin
      end_kt = 0;
    end
    else
    begin
      end_kt = -1 * :end_dt;
      end_dt = 0;
    end
    suspend;
  end
end
