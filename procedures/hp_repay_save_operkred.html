CREATE PROCEDURE HP_REPAY_SAVE_OPERKRED(
    K_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    OL_ID DECIMAL(18,0),
    REP_DATE DATE,
    IS_WRITEOFF INTEGER,
    REPAY_DT_ACCOUNT_KIND INTEGER,
    NEED_WRITEOFF_OPEREXTRA INTEGER = 0)
AS
declare variable saldo_os decimal(18,4) = 0;
declare variable saldo_perc decimal(18,4) = 0;
declare variable saldo_fine decimal(18,4) = 0;
declare variable saldo_kommis decimal(18,4) = 0;
declare variable cm_database_id integer;
declare variable s_rec_no integer;
declare variable zad_os decimal(18,4);
declare variable zad_perc decimal(18,4);
declare variable zad_fine decimal(18,4);
declare variable zad_kommis decimal(18,4);
declare variable pay_os decimal(18,4);
declare variable pay_perc decimal(18,4);
declare variable pay_fine decimal(18,4);
declare variable pay_kommis decimal(18,4);
declare variable pay_tax decimal(18,4);
declare variable pay_pereplat decimal(18,4);
declare variable graph_diff_os decimal(18,4);
declare variable graph_diff_perc decimal(18,4);
declare variable pr_kommis_is_percent decimal(18,4) = 0;
declare variable s_id integer;
declare variable s_date date;
declare variable need_close_loan integer; /* необходимо закрыть кредит */
declare variable ok_type integer;
declare variable ok_os decimal(18,4);
declare variable ok_perc decimal(18,4);
declare variable ok_fine decimal(18,4);
declare variable ok_pay_type integer;
declare variable ok_kommis decimal(18,4);
declare variable ok_sh_os decimal(18,4);
declare variable ok_sh_perc decimal(18,4);
declare variable ok_kommis_adv decimal(18,4);
declare variable ok_tax decimal(18,4);
declare variable writeoff_os decimal(18,4);
declare variable writeoff_perc d_currency;
declare variable writeoff_other d_currency;
declare variable client_koef double precision;
declare variable fact_loan_amount curr_type;
declare variable kc_amount_approved curr_type;
declare variable kc_closed_ol_id id_type;
declare variable fact_cl_qty integer;
declare variable counter integer;
declare variable writeoff_os_cl curr_type;
declare variable writeoff_perc_cl curr_type;
declare variable writeoff_other_cl d_currency;
declare variable writeoff_os_cl_sum curr_type;
declare variable writeoff_perc_cl_sum curr_type;
declare variable kroff_rd_tag integer;
declare variable payed_extra decimal(18,4) = 0;
declare variable woff_extra decimal(18,4) = 0;
declare variable pay_extra decimal(18,4) = 0;
declare variable has_sh_changes integer;
declare variable fix_os integer;
declare variable loan_is_active integer;
declare variable c_id_tmp id_type;
declare variable pay_add_perc d_currency;
declare variable pr_id d_id;
declare variable pr_code d_str_100;
declare variable ipf_commission d_currency;
declare variable pr_ipf_commission d_currency;
declare variable is_ipf d_smallint;
declare variable is_mbh boolean;
declare variable rd_id d_integer;
declare variable tom_perc d_currency;
declare variable tom_other d_currency;
declare variable cl_count d_integer;
declare variable cl_c_id d_id;
declare variable writeoff_kommis d_currency;
declare variable writeoff_kommis_cl d_currency;
declare variable is_vac_perc d_smallint;
declare variable vac_perc_pay d_currency;
declare variable vac_perc_resh d_currency;
declare variable vac_delta_sh_id d_integer;
declare variable payed_all d_currency;
declare variable vac_max_part_limit d_currency;
declare variable vac_date_begin d_date;
declare variable vac_date_end d_date;
declare variable vac_days d_integer;
declare variable easy_start d_boolean;
declare variable fix_perc d_currency;
begin
  cm_database_id = f_get_db_id();
  has_sh_changes = 0;

  easy_start = false;
  if (is_writeoff = -1) then
  begin
    is_writeoff = 0;
    easy_start = true;
  end

  select
    pr_id,
    pr_code
  from kontract
    inner join product
      on pr_id = k_from_pr_id
  where k_id = :k_id
  into
    :pr_id,
    :pr_code;

  select count(*), max(kc_from_c_id)
  from knclient
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into cl_count, cl_c_id;

  select pr_kommis_is_percent, rd_exclude_from_operkroff, vac_date_begin, vac_date_end,
         vac_max_part_limit, coalesce(vac_updated_sh_id, 0) - coalesce(vac_last_sh_id, 0)
  from temp_params
  into :pr_kommis_is_percent, :kroff_rd_tag, vac_date_begin, vac_date_end, vac_max_part_limit, vac_delta_sh_id;

  select sum(zad_os - pay_os), sum(calcing_perc - iif(s_id = 0 and is_extra = 1, 0, pay_perc)),
         sum(zad_fine - pay_fine), sum(zad_kommis - pay_kommis), sum(pay_os),
         sum(iif(s_id = 0 and is_extra = 1, 0, pay_perc)),
         sum(iif(s_date >= :vac_date_begin, pay_os + pay_perc, 0))
  from temp_vrepay
  where s_date is not null -- #operkroff_pay 11.04.2019
  into :saldo_os, :saldo_perc, :saldo_fine, :saldo_kommis, writeoff_os, writeoff_perc, payed_all;

  if (saldo_os is null) then
    saldo_os = 0;
  if (saldo_perc is null) then
    saldo_perc = 0;
  if (saldo_fine is null) then
    saldo_fine = 0;
  if (saldo_kommis is null) then
    saldo_kommis = 0;
  if (writeoff_os is null) then
    writeoff_os = 0;
  if (writeoff_perc is null) then
    writeoff_perc = 0;
  if (payed_all is null) then
    payed_all = 0;

  need_close_loan = 0;
  fix_os = 0;
  loan_is_active = 0;

  if (exists(select *
             from kontstate
             where ks_from_k_id = :k_id and
                   ks_state = 1 and
                   ks_date_end >= '31.12.2999' and
                   ks_ol_is_delete = 0)) then
    loan_is_active = 1;

  if (loan_is_active = 1) then
  begin
    if (f_ipf_newrepmode(rep_date, k_id, is_writeoff) = 1) then
      if (saldo_os <= 0) then
        need_close_loan = 1;

    if (saldo_os <= 0 and
        saldo_perc <= 0) then -- если долг по ос и %% закрыт
    begin
      if (saldo_fine <= 0) then
        need_close_loan = 1; -- если долга по пеням нет, закрываем кредит
      else
      -- рф и кр: закрываем если даже имеется долг по пеням. пеню списываем
      if (cm_database_id in (81001, 39801, 41701)) then
        need_close_loan = 1;
    end
    if (saldo_os <= 0) then
      fix_os = 1;
  end

  if (easy_start) then
    need_close_loan = 1;

  ok_kommis_adv = 0;
  pay_add_perc = 0;

  vac_perc_pay = 0;
  vac_perc_resh = 0;

  if (easy_start) then
    fix_perc = saldo_perc;
  else
    fix_perc = 0;

  for select s_rec_no, s_id, s_date, is_vac_perc, zad_os, zad_perc, zad_fine, zad_kommis, pay_os,
             iif(s_id = 0 and is_extra = 1, 0, pay_perc), pay_fine, pay_kommis, pay_tax, pay_pereplat, graph_diff_os,
             graph_diff_perc, pay_extra, iif(s_id = 0 and is_extra = 1, pay_perc, 0)
      from temp_vrepay
      where s_date is not null
      into s_rec_no, s_id, s_date, is_vac_perc, zad_os, zad_perc, zad_fine, zad_kommis, pay_os, pay_perc, pay_fine, pay_kommis,
           pay_tax, pay_pereplat, graph_diff_os, graph_diff_perc, pay_extra, pay_add_perc
  do
  begin
    if (is_vac_perc >= 1) then
    begin
      vac_perc_pay = vac_perc_pay + pay_perc;
      vac_perc_resh = vac_perc_resh + graph_diff_perc;
    end

    -- обычное погашение или списание
    if (pay_os <> 0 or pay_perc <> 0 or pay_fine <> 0 or pay_tax <> 0 or pay_kommis <> 0 or pay_add_perc <> 0) then
    begin
      ok_os = 0;
      ok_perc = 0;
      ok_fine = 0;
      ok_kommis = -pay_add_perc;
      ok_sh_os = 0;
      ok_sh_perc = 0;
      ok_tax = 0;
      ok_kommis_adv = 0;
      ok_type = 4 + is_writeoff;
      ok_os = -pay_os;
      ok_perc = -pay_perc;
      ok_fine = -pay_fine;
      ok_tax = -pay_tax;
      ok_kommis_adv = -pay_kommis;
      ok_pay_type = repay_dt_account_kind;
      ok_sh_os = 0;
      ok_sh_perc = 0;

      -- если комиссия за досрочное погашение считается как процент,
      -- то прибавляем к графе процент
      if (pr_kommis_is_percent = 1) then
      begin
        ok_perc = ok_perc + ok_kommis_adv;
        ok_kommis_adv = 0;
      end

      if (cm_database_id in (39801, 41701) and
          exists(select *
                 from product pr
                 where pr.pr_id = :pr_id and
                       pr.pr_code containing '.ipf.' and
                       (coalesce(pr.pr_ipf_commission, 0) <> 0))) then
        is_ipf = 1;

      if (is_ipf = 1) then
      begin
        ok_kommis = -ok_kommis;
        rd_id = f_get_rd_id('CashTransactionIncom.IPFCommission');

        ipf_commission = 0;

        select sum(ok.ok_amount)
        from operkontract ok
        inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
        where ok.ok_from_k_id = :k_id
        and ok.ok_operation_type = :rd_id
        and ol.is_delete = 0
        into ipf_commission;

        ipf_commission = coalesce(ipf_commission, 0);

        select pr.pr_ipf_commission
        from product pr
        where pr.pr_id = :pr_id
        into pr_ipf_commission;

        if (pr_ipf_commission <> ipf_commission) then
        begin
          if (ok_kommis > pr_ipf_commission) then
            ok_kommis = ok_kommis - pr_ipf_commission;
          else
          begin
            pr_ipf_commission = ok_kommis;
            ok_kommis = 0;
          end

          insert into operkontract (ok_id, ok_from_k_id, ok_from_ol_id, ok_amount, ok_operation_type)
          values (null, :k_id, :ol_id, :pr_ipf_commission, 2208);
        end
      end

      if (is_ipf = 1 and
          ok_kommis = 0 and
          ok_perc = 0 and
          ok_os = 0 and
          ok_tax = 0 and
          ok_kommis_adv = 0 and
          ok_fine = 0) then
            continue;
      else
      begin
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:ol_id, -1, :k_id, :s_id, :s_date, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_pay_type, :ok_kommis,
                :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax);

        if (easy_start and fix_perc <> 0) then
        begin
          insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                                ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
          values (:ol_id, -1, :k_id, :s_id, :s_date, 7, 0, -:fix_perc, 0, :ok_pay_type, 0,
                  0, 0, 0, :ok_tax);

          fix_perc = 0;
        end
     end

      -- если комиссия за досрочное погашение считается как процент,
      -- то дополнительно начисляем процент через ok_type=7
      if (pr_kommis_is_percent = 1 and
          pay_kommis <> 0) then
      begin
        ok_os = 0;
        ok_perc = 0;
        ok_fine = 0;
        ok_kommis = 0;
        ok_sh_os = 0;
        ok_sh_perc = 0;
        ok_tax = 0;
        ok_kommis_adv = 0;
        ok_type = 7;
        ok_perc = pay_kommis;
        ok_sh_perc = 0;
        ok_pay_type = 3;

        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:ol_id, -1, :k_id, :s_id, :s_date, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_pay_type, :ok_kommis,
                :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax);

        ok_os = 0;
        ok_perc = 0;
        ok_fine = 0;
        ok_kommis = 0;
        ok_sh_os = 0;
        ok_sh_perc = 0;
        ok_tax = 0;
        ok_kommis_adv = 0;
        ok_type = 3;
        ok_perc = 0;
        ok_sh_perc = pay_kommis;
        ok_pay_type = 3;

        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:ol_id, -1, :k_id, :s_id, :s_date, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_pay_type, :ok_kommis,
                :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax);

        if (ok_type in (2, 3) and
            ok_sh_os <> 0) then
          has_sh_changes = 1;
      end
    end

    -- переплата по кредиту
    if (pay_pereplat <> 0) then
    begin
      ok_os = 0;
      ok_perc = 0;
      ok_fine = 0;
      ok_kommis = 0;
      ok_sh_os = 0;
      ok_sh_perc = 0;
      ok_tax = 0;
      ok_kommis_adv = 0;
      ok_type = 9;
      ok_os = -pay_pereplat;
      if (:ok_pay_type is null) then
        :ok_pay_type = 1;
      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
      values (:ol_id, -1, :k_id, :s_id, :s_date, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_pay_type, :ok_kommis,
              :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax);
    end

    -- оплата по доп.обязательствам
    if (pay_extra <> 0) then
    begin
      if ((need_writeoff_operextra = 0) and
          (is_writeoff = 1)) then
      begin
        --  в случае если это списание и судебные издержки списывать не нужно, тогда ничего не делаю
      end
      else
      begin
        payed_extra = 0;
        woff_extra = 0;

        if (is_writeoff = 0) then
          payed_extra = pay_extra;
        else
          woff_extra = pay_extra;

        if (cm_database_id in (39801, 41701) and
            exists(select *
                   from product pr
                   where pr.pr_id = :pr_id and
                         (coalesce(pr.pr_ipf_commission, 0) <> 0))) then
        begin
          if (ok_kommis_adv <> 0) then
            ok_kommis = ok_kommis + ok_kommis_adv;

          rd_id = f_get_rd_id('CashTransactionIncom.IPFCommission');

          ipf_commission = 0;

          select ok.ok_amount
          from operkontract ok
          inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
          where ok.ok_from_k_id = :k_id
          and ok.ok_operation_type = :rd_id
          and ol.is_delete = 0
          into ipf_commission;

          ipf_commission = coalesce(ipf_commission, 0);

          select pr.pr_ipf_commission
          from product pr
          where pr.pr_id = :pr_id
          into pr_ipf_commission;

          if (pr_ipf_commission <> ipf_commission) then
          begin
            if (payed_extra > pr_ipf_commission) then
              payed_extra = payed_extra - pr_ipf_commission;
            else
            begin
              pr_ipf_commission = :payed_extra;
              payed_extra = 0;
            end
            insert into operkontract (ok_id, ok_from_k_id, ok_from_ol_id, ok_amount, ok_operation_type)
            values (null, :k_id, :ol_id, :pr_ipf_commission, 2208);
          end
        end
        insert into operextra (ox_from_ol_id, ox_from_reg_id, ox_ol_date, ox_ol_is_active, ox_object_kind, ox_object_id,
                               ox_from_rd_id, ox_amount_plus, ox_amount_minus, ox_amount_writeoff, ox_detail)
        values (:ol_id, null, :rep_date, 1, 9, :k_id, abs(:s_id), 0.0, :payed_extra, :woff_extra,
                iif(:is_writeoff = 0, 'Оплата', 'Списание'));
      end
    end

    -- автореструктуризация
    if (graph_diff_os <> 0 or graph_diff_perc <> 0 or (need_close_loan = 1)) then
    begin
      ok_os = 0;
      ok_perc = 0;
      ok_fine = 0;
      ok_kommis = 0;
      ok_sh_os = 0;
      ok_sh_perc = 0;
      ok_tax = 0;
      ok_kommis_adv = 0;
      ok_type = 3;
      ok_pay_type = 3;
      ok_sh_os = -graph_diff_os;
      ok_sh_perc = -graph_diff_perc;

      if (ok_sh_os <> 0 or ok_sh_perc <> 0) then
      begin
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:ol_id, -1, :k_id, :s_id, :s_date, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_pay_type, :ok_kommis,
                :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax);

        if (ok_type in (2, 3) and
            ok_sh_os <> 0) then
          has_sh_changes = 1;
      end
    end

    if (need_close_loan = 1) then -- если кредит закрывается
    begin
      -- списание невыплаченной пени и комиссии
      ok_os = 0;
      ok_perc = 0;
      ok_fine = 0;
      ok_kommis = 0;
      ok_sh_os = 0;
      ok_sh_perc = 0;
      ok_tax = 0;
      ok_kommis_adv = 0;
      ok_type = 5;
      ok_pay_type = 3;
      ok_fine = -(zad_fine - pay_fine);
      ok_kommis_adv = zad_kommis - pay_kommis;
      if (ok_fine <> 0 or ok_kommis_adv <> 0) then
      begin
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:ol_id, -1, :k_id, :s_id, :s_date, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_pay_type, :ok_kommis,
                :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax);
      end
    end
  end

  if (vac_delta_sh_id is null) then
    vac_delta_sh_id = 0;

  vac_days = 0;
  if (payed_all >= vac_max_part_limit) then
    vac_days = datediff(day, vac_date_end, rep_date);

  if (vac_perc_pay <> 0 or (payed_all > 0 and rep_date between :vac_date_begin and :vac_date_end)) then
  begin
    vac_days = 0;
    if (payed_all >= vac_max_part_limit) then
      vac_days = datediff(day, vac_date_end, rep_date);

    -- Платежи или списания по кредитным каникулам
    insert into opervac (ov_from_ol_id, ov_from_k_id, ov_type, ov_os, ov_perc, ov_last_os_sh_id, ov_max_part_limit, ov_vac_days)
    values (:ol_id, :k_id, iif(:is_writeoff = 1, 5, 4), 0, -:vac_perc_pay, :vac_delta_sh_id, -:payed_all, :vac_days);

    vac_delta_sh_id = 0;
  end

  if (vac_perc_resh <> 0) then
  begin
    -- Автокоррективрока %% по кредитным каникулам
    insert into opervac (ov_from_ol_id, ov_from_k_id, ov_type, ov_os, ov_perc, ov_last_os_sh_id)
    values (:ol_id, :k_id, 3, 0, -:vac_perc_resh, :vac_delta_sh_id);
  end

  -- Погашение ранее списанных платежей
  for select pay_os, pay_perc, pay_fine, pay_pereplat
      from temp_vrepay
      where s_date is null
      into writeoff_os, writeoff_perc, writeoff_kommis, pay_pereplat
  do
  begin
    if (writeoff_kommis is null) then
      writeoff_kommis = 0;

    -- если переплата --------------->>  -- данный блок был добавлен igusev 26.09.2018   #operkroff_pay
    if (pay_pereplat > 0) then
    begin
      ok_os = 0;
      ok_perc = 0;
      ok_fine = 0;
      ok_kommis = 0;
      ok_sh_os = 0;
      ok_sh_perc = 0;
      ok_tax = 0;
      ok_kommis_adv = 0;
      ok_type = 9;
      ok_os = -pay_pereplat;

      if (s_date is null) then
        s_date = (select max(ok.ok_sh_date)
                  from operkred ok
