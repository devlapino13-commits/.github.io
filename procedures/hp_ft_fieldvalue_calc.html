CREATE PROCEDURE HP_FT_FIELDVALUE_CALC(
    FIL_ID D_ID,
    C_ID D_ID,
    FIELD_NAME D_STR_CODE,
    REP_DATE D_DATE,
    FROM_HISTORY D_SMALLINT)
RETURNS (
    RES_TEXT D_STR_LONG,
    IS_EDITED D_SMALLINT)
AS
declare variable edit_value d_str_long;
declare variable fif_id d_id;
declare variable fie_id d_id;
begin
  is_edited = 0;

  select fif_id
  from ft_infofield, ft_infolist
  where fil_id = :fil_id and
        fil_from_fib_id = fif_from_fib_id and
        fif_fieldname = :field_name
  into fif_id;

  edit_value = '';

  select fie_id, fie_edit_value
  from ft_infoeditvalue
  where fie_from_c_id = :c_id and
        fie_from_fif_id = :fif_id and
        fie_date = :rep_date
  order by fie_datetime desc, fie_id desc
  rows 1
  into fie_id, edit_value;

  if (fie_id is not null) then
  begin
    is_edited = 1;
    if (edit_value is null) then edit_value = '';
    res_text = edit_value;
  end else
  begin
    select res_text
    from hp_ft_fieldvalue_calc_tmp(:fil_id, :c_id, :field_name, :rep_date, :from_history)
    into :res_text;
    if (res_text is null) then res_text = '';
  end

  Suspend;
end
