CREATE PROCEDURE FM_GET_OWN_GOOD(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    ML_NAME D_STR_255,
    ML_MODEL D_STR_255,
    BILL_ID D_ID,
    REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    WP_ID D_ID,
    WP_NAME D_STR_100,
    MOTV_ID D_ID,
    MOTV_NAME D_STR_100,
    ML_ID D_ID,
    YL_CODE D_STR_20,
    QUANTITY D_CURRENCY,
    COST D_CURRENCY,
    BILL_LINE_NO D_INTEGER)
AS
begin
    for select bill_id,
               reg_id,
               reg_name,
               wp_id,
               wp_name,
               motv_id,
               motv_name,
               ml_id,
               ml_name,
               ml_model,
               yl_code,
               quantity,
               cost,
               bill_line_no
        from hp_get_own_good(:c_id)
        into bill_id,
             reg_id,
             reg_name,
             wp_id,
             wp_name,
             motv_id,
             motv_name,
             ml_id,
             ml_name,
             ml_model,
             yl_code,
             quantity,
             cost,
             bill_line_no
    
    do
    begin
      suspend;
    end
end
