CREATE PROCEDURE HP_CARDCLIENT_UPDATE_CUSTOMER(
    C_ID D_ID,
    C_NEW_ID D_ID = null,
    SHOW_ADD_INFO SMALLINT = 0,
    UN_ID D_ID = null)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable is_union d_bool;
declare variable request_json d_str_8100;
declare variable csc_reg_id d_integer;
declare variable csc_upd_id d_integer;
declare variable pos_add d_integer;
declare variable str_add d_str_100;
declare variable new_inn d_str_20;
declare variable old_inn d_str_20;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable k_id d_id;
declare variable reg_id d_integer;
begin
  err_text = '';
  --exit;

  un_id = coalesce(un_id, 1000000000000);
  csc_reg_id = f_get_cardsynccfg_id('registr_client');
  csc_upd_id = f_get_cardsynccfg_id('update_client_info');

  if (not exists(
    select * from cardsync
    where cs_from_c_id = :c_id
      and cs_from_csc_id = :csc_reg_id
      and cs_status = 2))
  then err_text = 'Не определена операция регистрации нового клиента в МПЦ.';

  if (err_text = '' and c_new_id > 0 and c_new_id <> c_id and
    not exists(
      select * from clients
      where c_id = :c_new_id
        and coalesce(c_inn, '') <> ''))
  then err_text = 'Клиент (' || c_new_id || ') не определен.';

  if (err_text = '') then
  begin
    is_union = iif(c_new_id is null or (c_new_id = c_id) , 0, 1);

    request_json = (
      select '{"customer": ' || trim(new_customer) || '}'
      from cr_json_new_customer(
        iif(:is_union = 0, :c_id, :c_new_id),
        :show_add_info));

    new_inn = f_json_str_get_value(request_json, 'customer.person_code', '');

    select first 1
      cs_card_id,
      cs_from_cca_id,
      cs_from_k_id,
      cs_office_id,
      f_json_str_get_value(cs_request_json, 'customer.person_code' || trim(iif(cs_from_csc_id = :csc_upd_id, '_change_to', '')) , '')
    from cardsync
    where cs_from_c_id = :c_id
      and cs_from_csc_id in (:csc_reg_id, :csc_upd_id)
      and cs_status = 2
    order by cs_upd_datetime desc
    into
      :card_id,
      :cca_id,
      :k_id,
      :reg_id,
      :old_inn;

    str_add = '';
    if (new_inn <> old_inn) then
    begin
      request_json = replace(request_json, '"person_code":', '"person_code_change_to":');
      str_add = str_add || ',"person_code":"' || old_inn || '"';
    end
    if (is_union = 1) then
    begin
      request_json = replace(request_json, '"cid":', '"cid_change_to":');
      str_add = str_add || ',"cid":' || c_id;
    end

    if (str_add <> '') then
    begin
      pos_add = char_length(request_json) - 1;
      request_json = overlay(request_json placing str_add from pos_add for 0);
    end

    if (card_id is not null) then
      insert into cardsync (
        cs_from_csc_id,
        cs_card_id,
        cs_from_cca_id,
        cs_object_kind,
        cs_object_id,
        cs_request_json,
        cs_office_id,
        cs_from_un_id,
        cs_from_k_id,
        cs_from_c_id,
        cs_date)
      values(
        :csc_upd_id,
        :card_id,
        :cca_id,
        f_objkind('clients', 'id'),
        coalesce(:c_new_id, :c_id),
        :request_json,
        :reg_id,
        :un_id,
        :k_id,
        coalesce(:c_new_id, :c_id),
        current_date);
  end

  suspend;
end
