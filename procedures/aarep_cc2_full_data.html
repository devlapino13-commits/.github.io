CREATE PROCEDURE AAREP_CC2_FULL_DATA(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    CALL_TYPE STR_100,
    CPC_QESTION_DATE DATE,
    CPC_QESTION_TEXT VARCHAR(1000),
    CPC_QESTION_OPEN SMALLINT,
    CPC_NOT_ANSWERED SMALLINT,
    CPC_ANSWER_DATE DATE,
    CPC_ANSWER_TEXT VARCHAR(1000),
    CPC_CALLED_OPER_DATE DATE,
    CPC_MEMO STR_255,
    CP_FIO STR_100,
    CP_CELL_NO STR_100,
    CP_CITY STR_100,
    CP_ADDIT_CELL_NO STR_100,
    CP_MEMO STR_255,
    IS_CLIENT SMALLINT,
    CP_EXT_INFO STR_255,
    QESTION_OPER_FIO STR_100,
    ANSWER_OPER_FIO STR_100,
    CALLED_OPER_FIO STR_100,
    LAWYER_FIO STR_100,
    INFO_SOURCE STR_100,
    CITIZEN STR_100,
    CP_CREATE_DATE DATE,
    CPC_CATEG_STR STR_100)
AS
declare variable cpc_qestion_oper_id id_type;
declare variable cpc_answer_oper_id id_type;
declare variable cpc_called_oper_id id_type;
declare variable cpc_lawyer_id id_type;
declare variable cp_citizen integer;
declare variable cp_from_is_id integer;
begin
  for select (case cpc.cpc_call_type
                when 1 then 'Исходящий'
                when 2 then 'Входящий'
              end) as call_type, cpc.cpc_qestion_date, cpc.cpc_qestion_text,

                   cpc.cpc_qestion_oper_id, cpc.cpc_qestion_open, cpc.cpc_not_answered,

                   cpc.cpc_answer_date, cpc.cpc_answer_text, cpc.cpc_answer_oper_id,

                   cpc.cpc_called_oper_id, cpc.cpc_called_oper_date, cpc.cpc_lawyer_id,

                   cpc.cpc_memo,
                   (
                   case
                     when cp.cp_from_c_id is not null then 1
                     else 0
                   end),

                   cp.cp_fio, cp.cp_cell_no, cp.cp_citizen,

                   cp.cp_city, cp.cp_addit_cell_no, cp.cp_create_date,

                   cp.cp_from_is_id, cp.cp_memo, cp.cp_ext_info,

                   cpc.cpc_categ_str
      from cc_person_calls cpc
      inner join cc_persons cp on cp.cp_id = cpc.cpc_from_cp_id
      where cpc.cpc_qestion_date between :rep_date_begin and :rep_date_end

      into :call_type, :cpc_qestion_date, :cpc_qestion_text, :cpc_qestion_oper_id, :cpc_qestion_open, :cpc_not_answered,
           :cpc_answer_date, :cpc_answer_text, :cpc_answer_oper_id, :cpc_called_oper_id, :cpc_called_oper_date,
           :cpc_lawyer_id, :cpc_memo, :is_client, :cp_fio, :cp_cell_no, :cp_citizen, :cp_city, :cp_addit_cell_no,
           :cp_create_date, :cp_from_is_id, :cp_memo, :cp_ext_info, :cpc_categ_str
  do
  begin
    qestion_oper_fio = '';
    select cl.c_name_fio
    from username un
    inner join clients cl on cl.c_id = un.un_from_c_id
    where un.un_id = :cpc_qestion_oper_id
    into :qestion_oper_fio;

    if (qestion_oper_fio is null) then
      qestion_oper_fio = '';
------
    answer_oper_fio = '';
    select cl.c_name_fio
    from username un
    inner join clients cl on cl.c_id = un.un_from_c_id
    where un.un_id = :cpc_answer_oper_id
    into :answer_oper_fio;

    if (answer_oper_fio is null) then
      answer_oper_fio = '';
------
    called_oper_fio = '';
    select cl.c_name_fio
    from username un
    inner join clients cl on cl.c_id = un.un_from_c_id
    where un.un_id = :cpc_called_oper_id
    into :called_oper_fio;

    if (called_oper_fio is null) then
      called_oper_fio = '';
------
    lawyer_fio = '';
    select cl.c_name_fio
    from username un
    inner join clients cl on cl.c_id = un.un_from_c_id
    where un.un_id = :cpc_lawyer_id
    into :lawyer_fio;

    if (lawyer_fio is null) then
      lawyer_fio = '';
------
    info_source = '';

    select rd.rd_name
    from refdata rd
    where rd.rd_id = :cp_from_is_id
    into :info_source;

    if (info_source is null) then
      info_source = '';
-------
    citizen = '';

    select rd.rd_name
    from refdata rd
    where rd.rd_id = :cp_citizen
    into :citizen;

    if (citizen is null) then
      citizen = '';

    suspend;
  end
end
