CREATE PROCEDURE CM_KG_0097_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable c_id id_type;
declare variable pr_is_employer integer;
declare variable pr_code d_str_100;
begin
    select pr.pr_is_employer, pr.pr_code
    from kontract ktr
    inner join product pr on ktr.k_from_pr_id=pr.pr_id
    where ktr.k_id=:k_id and
          ktr.k_amount_approved > 0
    into:pr_is_employer, pr_code;

    if (:pr_is_employer=1) then exit;                  -- Доступные кредиты для сотрудников
    if (:pr_is_employer=0) then                        -- Ловим недоступные кредиты для сотрудников
        for select knc.kc_from_c_id                    -- Проверка клиента среди сотрудников
        from knclient knc
        where knc.kc_from_k_id=:k_id
        and knc.kc_amount_approved>0
        into:c_id
        do
        begin
            if (c_id = 1090100062382) then --Для теста c_id Игоря
              exit;

            if (exists(select *
                        from hr_pstatus hps
                        where current_date between PS_DATE_BEGIN and PS_DATE_END and
                        PS_STATUS = 1 and
                        PS_IS_WORKING = 1 and
                        PS_FROM_C_ID = :c_id)) then
            begin
            res_error_text = 'По данному продукту выдача сотрудникам запрещена';
            res_error_is_critical = 1;
            suspend;
            end
        end
end
