CREATE PROCEDURE AAREP_OFFICE_LIST(
    REP_REGION_ID INTEGER,
    REP_YEAR INTEGER,
    REP_MONTH INTEGER)
RETURNS (
    REG_ID INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    REG_LOAN_COUNT INTEGER)
AS
declare variable rep_date_begin date;
declare variable rep_date_end date;
begin
  if (rep_year > 0) then
  begin
    if (rep_month < 10) then
      rep_date_begin = '01.0' || :rep_month || '.' || :rep_year;
    else
      rep_date_begin = '01.' || :rep_month || '.' || :rep_year;
    rep_date_end = dateadd(month, 1, :rep_date_begin) - 1;
  end

  for select id_list, reg_code, reg_name
      from region_list(:rep_region_id, 1, 3, 1), region
      where id_list = reg_unicode
      into :reg_id, :reg_code, :reg_name
  do
  begin
    reg_loan_count = 0;
    if (rep_year > 0) then
    begin
      select distinct count(*)
      from kontract, knaktlist, kontstate
      where ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin <= :rep_date_end and
            ks_date_end >= (:rep_date_begin - 1) and
            knaktlist.is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            kal_new_reg_id = :reg_id
      into :reg_loan_count;
      if (reg_loan_count is null) then
        reg_loan_count = 0;
    end
    suspend;
  end
end
