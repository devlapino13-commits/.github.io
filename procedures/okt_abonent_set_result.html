CREATE PROCEDURE OKT_ABONENT_SET_RESULT(
    JSON_STR D_STR_LONG)
AS
declare variable RES_ID D_INTEGER;
declare variable DELETE_FROM_ABONENT_LIST D_BOOL;
declare variable ADD_IGNORE_LIST D_BOOL;
declare variable PI_ID D_INTEGER;
declare variable PANT_ID D_ID;
declare variable OBJECT_KIND D_INTEGER;
declare variable OBJECT_ID D_ID;
declare variable PHONE D_STR_CODE;
declare variable RESULT_CODE D_STR_100;
declare variable TIMEZONE D_INTEGER;
declare variable ATTEMPTS_REMAINING_QTY D_INTEGER;
declare variable IS_IVR D_BOOL;
declare variable DURATION D_INTEGER;
declare variable JSON_KEY D_STR_255;
declare variable JSON_VALUE D_STR_255;
declare variable OPERATOR_PHONE D_STR_CODE;
declare variable PHONE_ID D_INTEGER;
declare variable OPERATOR_PHONE_ID D_INTEGER;
begin
  for select json_key, json_value
      from fm_json_parse(:json_str, 0)
      into :json_key, :json_value
  do
  begin
    if (json_key = 'pant_id') then
      pant_id = json_value;

    if (json_key = 'object_kind') then
      object_kind = json_value;

    if (json_key = 'object_id') then
      object_id = json_value;

    if (json_key = 'phone') then
      phone = json_value;

    if (json_key = 'result_code') then
      result_code = json_value;

    if (json_key = 'timezone') then
      timezone = json_value;

    if (json_key = 'attempts_remaining_qty') then
      attempts_remaining_qty = json_value;

    if (json_key = 'is_ivr') then
      is_ivr = json_value;

    if (json_key = 'duration') then
      duration = json_value;

    if (json_key = 'operator_phone') then
        operator_phone = json_value;
  end

  -- RFCRU-5338
  phone = (case
                when char_length(:phone) = 11
                     and :phone starting with '89'
                then '7' || substring(:phone from 2)
                else phone
          end);

  result_code = 'IVRCallStatus.'||result_code;
  res_id = f_get_rd_id(result_code);

  delete_from_abonent_list = 0;
  add_ignore_list = 0;

  if (result_code = 'IVRCallStatus.Successful') then
  begin
    delete_from_abonent_list = 1;
    add_ignore_list = 1;
  end
  else
  begin
    if (attempts_remaining_qty = 0) then
    begin
        delete_from_abonent_list = 1;
        add_ignore_list = 1;
    end
  end

  if ((add_ignore_list = 1) and (f_get_const('okt_ivr_test') = 0)) then
    pi_id = (select pi_id from hp_pantera_ignore_insert(null, :object_kind, :object_id, f_get_rd_id('contactinfo.cell_main'), :result_code, :pant_id, null));

  if (delete_from_abonent_list = 1) then
    delete from okt_active_abonent_list t where t.oal_phone = :phone;

  if (coalesce(phone, '') <> '') then
    phone_id = f_cont_id(phone, 'add');

  if (coalesce(operator_phone, '') <> '') then
    operator_phone_id = f_cont_id(operator_phone, 'add');

  insert into okt_abonent_calls (oas_object_kind, oas_object_id, oas_from_pant_id, oas_result, oas_timezone, oas_duration, oas_attempts_remaining_qty, oas_is_ivr, oas_abonent_phone, oas_operator_phone, oas_ignore_id, oas_json)
  values (:object_kind, :object_id, :pant_id, :res_id, :timezone,  :duration, :attempts_remaining_qty, :is_ivr, :phone_id, :operator_phone_id, :pi_id, :json_str);
end
