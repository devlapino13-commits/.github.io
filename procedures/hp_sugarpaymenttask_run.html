CREATE PROCEDURE HP_SUGARPAYMENTTASK_RUN(
    SPT_ID D_INTEGER,
    UN_ID D_ID,
    OPER D_BOOL)
AS
declare variable k_id d_id;
declare variable acl_id d_id;
begin
  -- закрепление ЭЦП
  if (oper = 0) then
    update sugarpaymenttask set
      spt_eds_fixed = 1,
      spt_eds_un_id = :un_id
    where spt_id = :spt_id;
  else
  -- формирование задач на перевод
  if (oper = 1) then
  begin
    k_id = (select spt_k_id from sugarpaymenttask where spt_id = :spt_id);
    if (k_id > 0) then
    begin
      execute procedure hp_epaytask_create(:k_id);

      acl_id = (
        select acl_id
        from actionlist
        where acl_object_kind = 9
          and acl_object_id = :k_id
          and acl_from_rd_id = f_get_rd_id('EPayProjects')
          and acl_status = 0);

      update sugarpaymenttask set
        spt_created = iif(:acl_id is null, 0, 1),
        spt_created_un_id = :un_id,
        spt_acl_id = :acl_id
      where spt_id = :spt_id;
    end
  end
end
