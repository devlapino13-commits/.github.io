CREATE PROCEDURE FM_PRINT_PURCHASE_SELLER(
    OL_ID D_ID)
RETURNS (
    K_ID D_ID,
    SELLER_NAME D_STR_100,
    SELLER_INN D_STR_20,
    SELLER_PASS_NO VARCHAR(30),
    SELLER_ADDRESS D_STR_255,
    SELLER_REGDOC_NAME D_STR_100,
    SELLER_REGDOC_NO VARCHAR(30),
    SELLER_REGDOC_DATE D_DATE,
    SELLER_TAX_CODE D_STR_255,
    SELLER_TAX_NAME D_STR_255,
    SELLER_ELSOM D_STR_20,
    SELLER_DATE D_DATE,
    SELLER_DOCNO D_INTEGER,
    CUSTOMER_NAME D_STR_255,
    CUSTOMER_INN D_STR_20,
    CUSTOMER_ADR1 D_STR_2048,
    CUSTOMER_ADR2 D_STR_255,
    CUSTOMER_FIO D_STR_100,
    CUSTOMER_JOBNAME D_STR_100,
    PROD_CODE D_STR_10,
    PROD_NAME D_STR_255,
    PROD_MODEL D_STR_255,
    PROD_COUNT CURR_TYPE,
    PROD_COST CURR_TYPE,
    PROD_TOTAL CURR_TYPE,
    PROD_UNIT D_STR_20,
    KONT_NO D_STR_20,
    BANK_BIK D_STR_10,
    BANK_NAME D_STR_255,
    BANK_NO VARCHAR(50))
AS
begin
  select k_id
  from aarep_purchase_kontract(:ol_id)
  into :k_id;

  select seller_name, seller_inn, seller_pass_no, seller_address, seller_regdoc_name, seller_regdoc_no,
         seller_regdoc_date, seller_tax_code, seller_tax_name, seller_elsom, seller_date, seller_docno
  from aarep_purchase_seller(:ol_id)
  into :seller_name, :seller_inn, :seller_pass_no, :seller_address, :seller_regdoc_name, :seller_regdoc_no,
       :seller_regdoc_date, :seller_tax_code, :seller_tax_name, :seller_elsom, :seller_date, :seller_docno;

  select customer_name, customer_inn, customer_adr1, customer_adr2, customer_fio, customer_jobname
  from aarep_purchase_customer1(:ol_id)
  into :customer_name, :customer_inn, :customer_adr1, :customer_adr2, :customer_fio, :customer_jobname;

  select bank_bik, bank_name, bank_no
  from aarep_purchase_bank(:ol_id)
  into :bank_bik, :bank_name, :bank_no;

  for
  select prod_code, prod_name, prod_model, prod_count, prod_cost, prod_total, prod_unit, kont_no
  from aarep_purchase_product(:ol_id)
  into :prod_code, :prod_name, :prod_model, :prod_count, :prod_cost, :prod_total, :prod_unit, :kont_no

  do
  begin
    suspend;
  end
end
