CREATE PROCEDURE AAREP_FOR_TSB_SECTION_1_5(
    REP_DATE_END DATE)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_INN VARCHAR(20),
    KC_AMOUNT CURR_TYPE,
    K_KONTRACT_NO VARCHAR(20),
    RES_CLIENT_IS_RESIDENT INTEGER,
    K_DISB_DATE DATE,
    DELINQ_DAY INTEGER,
    KC_BALANCE DECIMAL(18,4))
AS
declare variable C_ID ID_TYPE;
declare variable KOL_KRED integer;
declare variable KC_AMOUNT2 CURR_TYPE;
declare variable K_KONTRACT_NO2 varchar(20);
declare variable K_DISB_DATE2 date;
declare variable CITIZEN_RU_ID integer;
declare variable DOC_VNJ_ID integer;
begin
  citizen_ru_id = null;
  select rd_id
  from refdata
  where rd_code = 'Citizen.RU'
  into :citizen_ru_id;

  doc_vnj_id = null;
  select pk_id
  from passkind, passdoctype
  where pdt_id = pk_from_pdt_id and
        pk_is_delete = 0 and
        pdt_code = 'registration_vnj' rows 1
  into :doc_vnj_id;

  for with tabl1
      as (select kc_from_c_id,
                 cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance as kc_balance,
                 cast(:rep_date_end as date) - oks_delinq_date as delinq_day, kc_amount_approved, k_kontract_no,
                 k_disb_date
          from oksummary, kontract, knclient
          where :rep_date_end between oks_date_begin and oks_date_end and
                oks_close_state = 0 and
                oks_balance > 0 and
                oks_from_k_id = k_id and
                oks_from_k_id = kc_from_k_id and
                kc_amount_approved > 0)
      select kc_from_c_id, sum(kc_balance) kc_balance, max(delinq_day) delinq_day, count(*) as kol_kred,
             max(kc_amount_approved), max(k_kontract_no), max(k_disb_date)
      from tabl1
      group by kc_from_c_id
      order by kc_balance desc, delinq_day
      rows 20
      into c_id, kc_balance, delinq_day, kol_kred, kc_amount2, k_kontract_no2, k_disb_date2
  do
  begin
    if (delinq_day < 0) then
      delinq_day = 0;

    select '№:' || c_id, c_inn
    from clients
    where c_id = :c_id
    into c_name_fio, c_inn;

    select 1 - res_non_resident
    from hp_get_client_type(:c_id, :citizen_ru_id, :doc_vnj_id, :rep_date_end)
    into :res_client_is_resident;

    if (kol_kred > 1) then
    begin
      for select kc_amount_approved, k_kontract_no, k_disb_date,
                 cast(:rep_date_end as date) - oks_delinq_date as delinq_day,
                 cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance
          from knclient, kontract, oksummary
          where kc_from_c_id = :c_id and
                kc_amount_approved > 0 and
                kc_from_k_id = k_id and
                k_id = oks_from_k_id and
                :rep_date_end between oks_date_begin and oks_date_end
                and oks_close_state = 0 
          into kc_amount, k_kontract_no, k_disb_date, delinq_day, kc_balance
      do
      begin

        suspend;
      end

    end
    else
    begin
      kc_amount = kc_amount2;
      k_kontract_no = k_kontract_no2;
      k_disb_date = k_disb_date2;

      suspend;
    end
  end
end
