CREATE PROCEDURE AAREP_MOVEMENT_CLIENTSCARD_NEW(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    CARD_ID D_INTEGER,
    CCA_ID D_INTEGER,
    DATE_ISSUE D_DATE,
    CREDIT_LIMIT D_CURRENCY,
    NEW_LIMIT D_CURRENCY,
    BALANCE_BEG D_CURRENCY,
    SUMMA_DT D_CURRENCY,
    SUMMA_KT D_CURRENCY,
    BALANCE_END D_CURRENCY)
AS
declare variable reg_lst d_str_8100;
declare variable acc d_str_10;
begin
  select
    coalesce(list(id_list, ';'), '')
  from region_list(:rep_region_id, 1, 3, 1)
  into
    :reg_lst;

  for
    select
      reg_unicode,
      reg_name,
      c_id,
      c_name_fio,
      cal_card_id,
      cal_from_cca_id,
      cal_date_issue
    from cardlistlnk
      inner join cardlist
        on cal_card_id = cll_card_id
      inner join clients
        on c_id = cll_from_c_id
      inner join cardplace
        on cpl_id = cal_last_cpl_id
      inner join region
        on reg_unicode = cpl_from_reg_id
  into
    :reg_id,
    :reg_name,
    :c_id,
    :c_name_fio,
    :card_id,
    :cca_id,
    :date_issue
  do
  begin
    if (reg_lst not containing reg_id)
      then continue;

    --credit_limit = (select amount from hp_balance_cardloans('cardlist', :card_id, :rep_date_begin));
    new_limit = (select amount from hp_balance_cardloans('cardlist', :card_id, :rep_date_end));

    acc = '20001/1';
    with cards as
    (
      select distinct clo2.clo_card_id as card_id
      from cardlistloan as clo1
        inner join cardlistloan as clo2
          on clo2.clo_from_k_id = clo1.clo_from_k_id
      where clo1.clo_card_id = :card_id
    ), kont as
    (
      select distinct clo_from_k_id as ko_id
      from cardlistloan
        inner join cards
          on card_id = clo_card_id
    ), blim as
    (
      select
        sum(k_amount_approved) as amount
      from kontract
        inner join kont
          on ko_id = k_id
      where k_disb_date <= :rep_date_end
    )
    select
      sum(iif(ol_date < :rep_date_begin, iif(ob_short_acc_kt = :acc, ob_summa, 0) - iif(ob_short_acc_dt = :acc, ob_summa, 0), 0)),
      sum(iif(ol_date between :rep_date_begin and :rep_date_end and ob_short_acc_dt = :acc, ob_summa, 0)),
      sum(iif(ol_date between :rep_date_begin and :rep_date_end and ob_short_acc_kt = :acc, ob_summa, 0)),
      sum(iif(ol_date <= :rep_date_end, iif(ob_short_acc_kt = :acc, ob_summa, 0) - iif(ob_short_acc_dt = :acc, ob_summa, 0), 0)),
      coalesce((select sum(amount) from blim), 0)
    from operbook
      inner join (
        select distinct ok_from_ol_id as olo_id
          from operkred
              inner join kont
                on ko_id = ok_from_k_id
        union
        select ol_id as olo_id
          from pvnopers
              inner join cards
                on card_id = po_card_id
              inner join operlist
                on ol_parent_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = po_from_ol_id )
                  or ol_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = po_from_ol_id )
            where --po_card_k_id = :k_id and
                  po_is_mb_card = 1
              and po_from_ol_id > 0
              and po_is_delete = 0
        union
        select ol_id as olo_id
          from ipc_exd_detail
            inner join cards
              on card_id = ied_from_card_id
            inner join operlist
              on ied_from_ol_id in (ol_id, ol_parent_id)
          where --ied_from_k_id = :k_id and
             ied_from_ol_id > 0
        ) as a
      on a.olo_id = ob_from_ol_id
      inner join operlist
        on ol_id = ob_from_ol_id
          and is_delete = 0
    where ob_short_acc_dt = :acc or ob_short_acc_kt = :acc
    into
      :balance_beg,
      :summa_dt,
      :summa_kt,
      :balance_end,
      :credit_limit;

    if (balance_beg <> 0 or summa_kt <> 0 or summa_dt <> 0)
      then suspend;
  end
end
