CREATE PROCEDURE GET_CONTRACTS_SS(
    CATEGORY D_STR_20,
    BASE_PATH D_STR_CODE,
    BLS_ID D_ID,
    C_NAME_FIO D_STR_255)
RETURNS (
    PROVIDER D_STR_255,
    OWNERSHIP_TYPE D_STR_20,
    START_DATE D_DATE,
    END_DATE D_DATE,
    BORROWERS D_STR_2048,
    GUARANTORS D_STR_2048,
    LOAN_AMOUNT D_CURRENCY,
    CURRENCY D_STR_20,
    ROLE_STR D_STR_20,
    PAYMENTS D_CURRENCY,
    DURATION D_INTEGER,
    OUTSTANDING D_CURRENCY,
    INTEREST_OUTSTANDING D_CURRENCY,
    LOAN_STATUS D_STR_100,
    PAYMENT_STATUS D_STR_100,
    MAX_PAST_DUE D_STR_100,
    IS_OUR D_SMALLINT,
    INTEREST_RATE D_STR_10)
AS
declare variable PATH D_STR_255;
declare variable JSON_BLB D_BLOB_TEXT;
declare variable ARR_LENGTH D_INTEGER;
declare variable I D_INTEGER;
declare variable LINKED_ARR_LENGTH D_INTEGER;
declare variable X D_INTEGER;
declare variable OWNERSHIP D_INTEGER;
declare variable LOAN_STATUS_CODE D_STR_10;
declare variable PAYMENT_STATUS_CODE D_STR_10;
declare variable MAX_PAST_DUE_CODE D_STR_10;
declare variable ROLE_CODE D_STR_10;
declare variable TMPSTR D_STR_100;
begin
  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into :json_blb;

  path = base_path || '.credit_report.contracts_history.' || category || '.granted_contract';
  arr_length = f_json_blob_get_count(json_blb, path);
  i = 0;
  while (i < arr_length) do
  begin
    is_our = 0;
    provider = f_json_blob_get_value(json_blb, path || f_format('[%1].granted_contract.provider_code_desc', i), '');
    if (provider = 'MKK M Bulak') then
      is_our = 1;
    if (category <> 'services') then
    begin
      ownership = cast(f_json_blob_get_value(json_blb, path || f_format('[%1].granted_contract.ownership_type', i), 0) as integer);
      if (ownership = 1) then
      ownership_type = 'Индивидуальный займ';
      if (ownership = 2) then
      ownership_type = 'Групповой займ';
    end

    tmpstr = f_json_blob_get_value(json_blb, path || f_format('[%1].granted_contract.contract_start_date', i), '');
    start_date = cast(tmpstr as date);

    tmpstr = f_json_blob_get_value(json_blb, path || f_format('[%1].granted_contract.contract_end_date', i), '');
    end_date = case when tmpstr <> '' then cast(tmpstr as d_date) else null end;

    interest_rate = '';
    if(current_date >= cast(f_get_const('send_interest_rate_to_kib') as date)) then
      interest_rate = f_json_blob_get_value(json_blb, path || f_format('[%1].granted_contract.interest_rate', i), '');

    linked_arr_length = f_json_blob_get_count(json_blb, path || f_format('[%1].linked_subject', i));  --Заемщики
    borrowers = '';
    if (linked_arr_length = 0) then
    borrowers = c_name_fio;
    else
    begin
      x = 0;
      while (x < linked_arr_length) do
      begin
        borrowers = borrowers || f_json_blob_get_value(json_blb, path || f_format('[%1].linked_subject[%2].name', i, x), '');
        x = x + 1;
      end
    end
    
    linked_arr_length = f_json_blob_get_count(json_blb, path || f_format('[%1].guarantee', i));  --Поручители
    guarantors = '';
    if (linked_arr_length > 0) then
    begin
      x = 0;
      while (x < linked_arr_length) do
      begin
        guarantors = guarantors || f_json_blob_get_value(json_blb, path || f_format('[%1].guarantee[%2].guarantor_name', i, x), '');
        x = x + 1;
      end
    end
    
    loan_amount = cast(f_json_blob_get_value(json_blb, path || f_format('[%1].granted_installment.financed_amount', i), 0) as d_currency);
    duration = cast(f_json_blob_get_value(json_blb, path || f_format('[%1].granted_installment.installments_number', i), 0) as d_integer);
    currency = f_json_blob_get_value(json_blb, path || f_format('[%1].granted_contract.original_currency', i), '');
    role_code = f_json_blob_get_value(json_blb, path || f_format('[%1].granted_contract.role', i), '');
    role_str = case
                when role_code = 'B' then 'Заемщик'
                when role_code = 'C' then 'Созаемщик'
                when role_code = 'G' then 'Поручитель'
                else '--'
               end;

    if (category = 'installments' or category = 'credit_cards') then
    begin
      payments = cast(f_json_blob_get_value(json_blb, path || f_format('[%1].granted_installment.monthly_payment_amount', i), 0) as d_currency);
      outstanding = cast(f_json_blob_get_value(json_blb, path || f_format('[%1].granted_installment.principal_outstanding_amount', i), 0) as d_currency);
      interest_outstanding = cast(f_json_blob_get_value(json_blb, path || f_format('[%1].granted_installment.interest_outstanding_amount', i), 0) as d_currency);
    end
    
    loan_status_code = f_json_blob_get_value(json_blb, path || f_format('[%1].granted_contract.contract_phase', i), '');
    loan_status = case
                    when loan_status_code = 'RQ' then 'Запрошеный'
                    when loan_status_code = 'RN' then 'Отозваный (по желанию клиента)'
                    when loan_status_code = 'RF' then 'Отклоненый (учреждением)'
                    when loan_status_code = 'AC' then 'Открытый'
                    when loan_status_code = 'CL' then 'Закрытый'
                    when loan_status_code = 'CA' then 'Закрытый досрочно'
                    else '--'
                  end;
    
    if (category <> 'non_installments') then
    begin
    payment_status_code = cast(f_json_blob_get_value(json_blb, path || f_format('[%1].granted_installment.days_past_due', i), 0) as d_integer);
    payment_status = case
                      when payment_status_code = 0 then 'Слишком новый'
                      when payment_status_code = 1 then 'Плата по согласованию'
                      when payment_status_code = 2 then 'Задержка 1-29 дней'
                      when payment_status_code = 3 then 'Задержка 30-59 дней'
                      when payment_status_code = 4 then 'Задержка 60-89 дней'
                      when payment_status_code = 5 then 'Задержка 90-119 дней'
                      when payment_status_code = 6 then 'Задержка 120-149 дней'
                      when payment_status_code = 7 then 'Задержка 150-179 дней'
                      when payment_status_code = 8 then 'Более 180 дней задержки'
                      else '--'
                     end;
    
    max_past_due_code = cast(f_json_blob_get_value(json_blb, path || f_format('[%1].maximum.max_days_past_due', i), 0) as d_integer);
    max_past_due = case
                    when max_past_due_code = 0 then 'Слишком новый'
                    when max_past_due_code = 1 then 'Плата по согласованию'
                    when max_past_due_code = 2 then 'Задержка 1-29 дней'
                    when max_past_due_code = 3 then 'Задержка 30-59 дней'
                    when max_past_due_code = 4 then 'Задержка 60-89 дней'
                    when max_past_due_code = 5 then 'Задержка 90-119 дней'
                    when max_past_due_code = 6 then 'Задержка 120-149 дней'
                    when max_past_due_code = 7 then 'Задержка 150-179 дней'
                    when max_past_due_code = 8 then 'Более 180 дней задержки'
                    else '--'
                    end;
    end
    
    i = i + 1;
    suspend;
  end
end
