CREATE PROCEDURE FSROBOT_PUSHING_PHONE_CHANGED(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable c_id id_type;
declare variable eds_status_date d_date;
declare variable old_cell_no d_str_100;
declare variable rd_id d_integer;
declare variable new_cell_no d_str_100;
declare variable client_fio d_str_100;
declare variable reg_name d_str_100;
declare variable reg_region_director d_id;
declare variable kc_from_k_id id_type;
declare variable ovs_datetime d_datetime;
declare variable res_date_save d_datetime;
declare variable ma_id d_id;
declare variable edl_from_reg_id d_integer;
begin
  res_body = '';

  /*Задача, если ранее уже был заключен договор эцп и клиент сменил телефон номер должен утверждать РМ */

  /* Анатолий �©егольский
        Кыял добрый день
        Когда в МП сотрудник меняет номер телефона вызывается FM_CONTINFO_UPDATE.
        Но саму проверку и отправку пуш-уведомления лучше сделать в отдельной процедуре для КГ
        и вызывать в HP_CONTINFO_UPDATE, т.к. она общая для всех систем, не только для МП.
    */

  -- Отправка PUSH региональному директору, если ранее уже был заключен договор эцп и клиент сменил телефон.

  -- Процедура вызывается из FSROBOT_ON_MAILING
  -- Процедура готовит список уведомлений для отправки роботом по почте или sms
  -- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
  -- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
  -- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

  -- RefData.RD_Memo    - Список получателей
  -- RefData.RD_Tag     - Срок жизни сообщения
  -- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

  -- Rep_Params - входящий параметры в зависимости от команды
  -- Rep_UN_ID - пользователь

  c_id = cast(rep_params as d_id); -- id клиента

  if (user_fio is null) then
    user_fio = '';
  res_maillist = :rep_email;

  select cl.c_id, cl.c_name_fio
  from clients cl
  where cl.c_id = :c_id
  into :res_c_id, :client_fio;

  select knc.kc_from_k_id
  from knclient knc
  where knc.kc_from_c_id = :c_id and
        knc.kc_amount_approved > 0
  order by kc_from_k_id desc
  rows 1
  into :kc_from_k_id;

  select k_from_ma_app_id
  from kontract k
  where k.k_id = :kc_from_k_id
  into :ma_id;

  -- номер на дату подписания послднего договора АСП (до текущей смены номера)
  rd_id = f_get_rd_id('ContactInfo.Cell_Main');

  select first 1 ma_date_save
  from miniapp
  where ma_id = :ma_id and
        ma_is_active = 1
  order by ma_number
  into :res_date_save;

  -- номер на дату заявки
  old_cell_no = null;
  select first 1 ovs_value, ovs_datetime
  from objectprops
  inner join objectvalues on ovs_from_opr_id = opr_id and
        ovs_datetime <= :res_date_save and
        ovs_is_active = 1 and
        ovs_value <> '' and
        ovs_add_value = 2
  where opr_object_kind = '0' and
        opr_object_id = :res_c_id and
        opr_subject_kind = '-1' and
        opr_subject_id = :rd_id
  order by ovs_datetime desc
  into :old_cell_no, :ovs_datetime; -- возвращает подтвер тел номер (старый)

  -- если у клиента ранее был подписан договор АСП
  eds_status_date = null;
  select cast(eds_status_datetime as date), el.edl_from_reg_id
  from edoc_list el
  inner join edoc_sign es on es.eds_from_edl_id = el.edl_id
  where el.edl_object_kind = 0 and
        el.edl_object_id = :c_id and
        el.edl_blank_status = 1 and --бланки приготовлены
        el.edl_from_edc_id = f_edoc_kind('asp.agreement', 'id') and
        el.edl_status = 1 and --статус подписан
        cast(eds_status_datetime as date) < :ovs_datetime rows 1
  into :eds_status_date, :edl_from_reg_id;

  if (eds_status_date is null) then
    exit;

  new_cell_no = f_cont_info('clients', :c_id, 'cell_main', 'last'); -- новый номер

  reg_name = '';
  select office.reg_name, reg.reg_region_director
  from region office
  inner join region reg on reg.reg_unicode = office.parent
  where office.reg_unicode = :edl_from_reg_id
  into :reg_name, :reg_region_director;

  if (reg_region_director = 1090100319785) then ----Jira.BOARD-4918 по запросу Алмаз байке
    reg_region_director = 1090100507011;

  --------  направляю пуш-уведомление PM -------------->

  select un_google_id
  from username
  where un_from_c_id = :reg_region_director and
        un_google_id is not null rows 1
  into :res_maillist;

  if (coalesce(res_maillist, '') <> '' and
      coalesce(new_cell_no, '') <> coalesce(old_cell_no, '')) then
  begin

    -- зная id клиента нужно выйти на id РД
    res_subject = 'У повторного Клиента изменен основной номер телефона.';

    res_body = BLOB_APPEND(res_body, 'Смена номера клиента ' || coalesce(:client_fio, '') || ascii_char(10) || ascii_char(13));
    res_body = BLOB_APPEND(res_body, 'Офис:  ' || coalesce(reg_name, '') || ascii_char(10) || ascii_char(13));
    res_body = BLOB_APPEND(res_body, 'старый: ' || coalesce(old_cell_no, '') || ascii_char(10) || ascii_char(13));
    res_body = BLOB_APPEND(res_body, 'новый: ' || coalesce(new_cell_no, '') || ascii_char(10) || ascii_char(13));
    res_body = BLOB_APPEND(res_body, 'Сравните подпись клиента на новом Договоре ЭЦП и в паспорте.' || ascii_char(10) || ascii_char(13));
    res_body = BLOB_APPEND(res_body, 'Позвоните на старый и новый номер для проверки смены номера.' || ascii_char(10) || ascii_char(13));

  end

  if (coalesce(res_body, '') <> '') then
    suspend;
end
