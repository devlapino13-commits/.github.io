CREATE PROCEDURE PRINT_FT_ANKETA_FIELDS(
    OBJ_ID D_ID,
    ADD_PARAMS D_STR_LONG)
RETURNS (
    FAF_CAPTION D_STR_LONG,
    FAF_TRUE_CAPTION D_STR_255,
    FAF_FALSE_CAPTION D_STR_255,
    FAV_TRUE_CHECKED D_BOOL,
    FAV_FALSE_CHECKED D_BOOL,
    FAV_TRUE_COMMENT D_STR_255,
    FAV_FALSE_COMMENT D_STR_255)
AS
declare variable fav_comment d_str_255;
declare variable fav_from_fal_id d_id;
declare variable faf_default_value d_bool;
declare variable company_name d_str_255;
begin
  -- Печать вопросов/ответов анкеты ПОД/ФТ
  company_name = f_company_name('short', 'local');

  for select faf_caption, faf_default_value, faf_true_caption, faf_false_caption,
    fav_from_fal_id, fav_comment
  from ft_anketa_field faf
  inner join ft_anketa_list fal on fal.fal_id = :obj_id and fal.fal_from_fak_id = faf.faf_from_fak_id
  left join ft_anketa_value on fav_from_fal_id = fal.fal_id and fav_from_faf_id = faf_id
  where faf_inactive = 0
  order by faf_position
  into faf_caption, faf_default_value, faf_true_caption, faf_false_caption,
    fav_from_fal_id, fav_comment
  do
  begin
    faf_caption = replace(faf_caption, '[company_name]', company_name);
    fav_true_checked = 0;
    fav_false_checked = 0;
    fav_true_comment = '';
    fav_false_comment = '';
    fav_comment = coalesce(fav_comment, '');

    if (fav_from_fal_id is null) then
    begin
      -- Ответа нет, значит используем ответ по-умолчанию
      if (faf_default_value = 0) then
        fav_false_checked = 1;
      else
        fav_true_checked = 1;
    end
    else
    begin
      -- Ответ есть, значит используем ответ НЕ по-умолчанию
      if (faf_default_value = 1) then
        fav_false_checked = 1;
      else
        fav_true_checked = 1;
    end

    -- Какой ответ выбран, для того и комментарий
    if (fav_true_checked = 1) then
      fav_true_comment = fav_comment;
    else
      fav_false_comment = fav_comment;
    suspend;
  end
end
