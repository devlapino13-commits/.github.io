    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)

    K_KONTRACT_NO VARCHAR(20),
    CLIENT_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_NAME_FIO VARCHAR(100),

    KC_FINAL_INCOME_OLD CURR_TYPE,
    KC_STAT_AVG_INCOME_OLD CURR_TYPE)

declare variable C_ID ID_TYPE;
declare variable PR_CODE STR_100;
declare variable PR_ID_OLD integer;
declare variable SAB_ID D_ID;
declare variable SAB_ID_ZALOG_COST D_ID;
declare variable K_ID decimal(18,0);
declare variable PR_ID D_INTEGER;
declare variable K_ID_PRIOR ID_TYPE;
declare variable PR_ID_PRIOR ID_TYPE;
declare variable OKS_IS_REFUND_LLR smallint;
declare variable OKS_IS_RESCHEDULE_LLR smallint;
declare variable KC_DATE D_DATE;
declare variable CLIENT_TYPE D_STR_20;
declare variable IS_REF D_INTEGER;
begin
  pr_id_old = 0;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Поручительство'
  into sab_id;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Залог'
  into :sab_id_zalog_cost;

  for
  --------списанные
      with tabl1
      as (select oko_from_k_id
          from operkroff
          where oko_ol_is_active = 1 and
                oko_ol_date between :rep_date_begin and :rep_date_end and
                oko_type = 1 and
                (oko_sh_os > 0 or
                oko_sh_perc > 0)
          group by 1)
      select 'списан' status_zap, k_id, k_kontract_no, k_disb_date, k_stop_perc,
             cast(k_percent_effect_fifd as decimal(15,3)) as cred_full_cost, k_from_pr_id

      from tabl1, kontract
      where oko_from_k_id = k_id and
            k_disb_date > '30.09.2019'
      union all
      ----на балансе.реестр--------------

             cast(kont.k_percent_effect_fifd as decimal(15,3)) as cred_full_cost, k_from_pr_id
      from oksummary oks, kontract kont
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks.oks_from_k_id = kont.k_id and
            oks_close_state = 0 and
            k_disb_date > '30.09.2019'
      into :status_zap, k_id, k_kontract_no, k_disb_date, k_stop_perc, cred_full_cost, pr_id
  do
  begin
    :status_zap = trim(:status_zap);
    k_percent_flat_annual = 0;
    select round(k_percent_per_day * 365.00, 3)
    from kontract kon
    where kon.k_id = :k_id
    into k_percent_flat_annual;

    percent_year_r = null;
    date_off = null;
    if (:status_zap = 'списан') then
    begin
      date_off = null;
      select ok_ol_date
      from operkred
      where ok_ol_is_active = 1 and
            ok_from_k_id = :k_id and
            ok_type = 5 and
            ok_os + ok_perc < 0
      order by ok_ol_date
      rows 1
      into date_off;

      select kd_percent_per_day * 365
      from kontdata
      where :date_off between kd_ol_date and kd_date_end and
            kd_ol_is_delete = 0 and
            kd_from_k_id = :k_id
      into percent_year_r;
    end
    else
    if (:status_zap = 'На балансе\Рестр\Реф') then
    begin
      select round(k_percent_per_day * 365.00, 3)
      from kontract kon
      where kon.k_id = :k_id
      into percent_year_r;
    end
    if (pr_id <> pr_id_old) then
    begin
      select pr_code, pr_name
      from product
      where pr_id = :pr_id
      into pr_code, pr_name;

      pr_id_old = pr_id;
    end

    client_type = 'ФЛ';
    if (pr_code containing '.selfemployed.') then client_type = 'СЗ';
      else
      if (pr_code containing '.business.') then client_type = 'ИП';

    if (client_type <> 'ФЛ') then
      continue;
    for select kc_id, kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              kc_client_type = 3

    do

      client_name_fio = null;
      c_birth_date = null;

      into client_name_fio, c_birth_date;

      kc_stat_avg_income_old = null;

      from knclient_pdn
      where kc_id = :kc_id and
            kcp_kind = 0

      kc_pdn_koef_end = null;

      from knclient_pdn
      left join com_list cml on cml_from_kcp_id = kcp_id
      left join com_action cma on cma_from_cml_id = cml_id
      left join com_cfg cmlf on cml_from_cmc_id = cmc_id

      rows 1
      into kc_pdn_koef_end, kc_anketa_income, kc_stat_avg_income, kc_final_income, kc_date_max;

        kc_pdn_koef_end = final_pdn;

        kc_final_income = :kc_final_income_old;

               when :status_zap = 'списан' and :date_off - 1 between oks_date_begin and oks_date_end then 1
