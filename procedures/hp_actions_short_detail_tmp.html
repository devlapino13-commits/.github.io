CREATE PROCEDURE HP_ACTIONS_SHORT_DETAIL_TMP(
    ACL_OBJECT_KIND D_INTEGER,
    ACL_OBJECT_ID D_ID,
    ANKETA_NAME D_STR_100,
    ACL_DATE D_DATE = null)
RETURNS (
    RES_ACL_ID D_ID,
    RES_QUESTION_ID D_INTEGER,
    RES_RD_ID D_INTEGER,
    RES_STATE D_INTEGER,
    RES_STR_VALUE D_STR_2048,
    RD_EDITOR_KIND D_INTEGER,
    RD_ORDERNO FLOAT,
    RD_CODE D_STR_100,
    RD_NAME D_STR_255,
    RD_DESCRIPTION D_BLOB_TEXT,
    RD_MRU_LIST D_STR_8100,
    TRACK_RESULT D_INTEGER,
    RES_FINAL_STR_VALUE D_STR_2048,
    DB_ID D_INTEGER,
    ORIG_STATE D_INTEGER)
AS
declare variable acg_caption d_str_255; /* Альтернативный заголовок */
declare variable anketa_id d_integer; /* ID шаблона анкеты */
declare variable acg_track_result d_integer;
declare variable rd_str_id d_str_100;
declare variable isvisitcheck d_integer;
declare variable need_show_uncheckall d_integer;
declare variable res_id_noanswer d_integer;
declare variable answer_count d_integer;
declare variable answer_no d_integer;
declare variable rd_ques_orderno d_integer;
declare variable line_id d_integer;
declare variable line_parent_id d_integer;
declare variable need_metro d_integer;
declare variable ovs_id d_integer;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable ma_id d_id;
declare variable rep_date d_date;
declare variable loan_target d_str_255;
declare variable pr_code str_100;
declare variable loan_description d_str_long;
begin
-- Процедура возвращает форму анкеты по ID задачи
-- если ID задачи не указана, возвращает первую из очереди
-- Если очередь пуст, ничего не возвращается
  acl_date = coalesce(acl_date, current_date);

  rd_mru_list = '';
  res_final_str_value = '';

  select
    rd_id,
    rd_str_id
  from refdata
  where rd_code = :anketa_name
  into
    :anketa_id,
    :rd_str_id;

  if (anketa_id is null) then
    exit;

  db_id = f_get_db_id();
  res_id_noanswer = (select rd_id from refdata where rd_code = 'Results.NoAnswer');

  if (db_id = 81001 and
      anketa_name = 'anketa.main') then
  begin
    ma_id = null;
    pr_code = '';

    select ma_id, pr_code
    from miniapp
    inner join miniappclient on mac_from_ma_id = ma_id
    inner join product on pr_id = ma_from_pr_id
    where mac_from_c_id = :acl_object_id and
          ma_is_active = 1 and
          ma_date_end = '31.12.9999'
    order by ma_id desc rows 1
    into ma_id, pr_code;

    k_id = null;

    select k_id
    from kontract
    inner join kontstate on ks_from_k_id = k_id
    where k_from_ma_app_id = :ma_id and
          ks_ol_is_delete = 0 and
          current_date between ks_date_begin and ks_date_end and
          ks_state = 0
    into k_id;
  end

  -- RFCKG-28
  if (:db_id in (39801, 41701) and
      :acl_object_kind = f_objkind('knclient', 'id') and
      :anketa_name = 'Anketa.TargetUseLoan') then
  begin
    select kc.kc_from_c_id, kc.kc_from_k_id, k.k_from_ma_app_id
    from knclient kc
    inner join kontract k on k.k_id = kc.kc_from_k_id
    where kc.kc_id = :acl_object_id
    into c_id, k_id, ma_id;

    -- Цель кредитования из заявки
    select rd.rd_code
    from miniappclient macl
    left join refdata rd on rd.rd_id = mac_loan_target
    where macl.mac_from_ma_id = :ma_id and
          macl.mac_from_c_id = :c_id
    into :loan_target;
  end
  -- RFCKG-28

  select
    acl_id
  from actionlist
  where acl_object_kind = :acl_object_kind
    and acl_object_id = :acl_object_id
    and acl_from_rd_id = :anketa_id
  order by acl_id desc
  rows 1
  into
    :res_acl_id;

  if (res_acl_id is null) then
    res_acl_id = 0;

  isvisitcheck = 0;
  if (rd_str_id <> 'Anketa') then
    isvisitcheck = 1;

  need_metro = 0;
-- получаем строки (вопросы-ответы) анкеты
  for
    select
      acg_id,
      acg_parent_id,
      acg_from_rd_id,
      acg_caption,
      acg_editor_kind,
      acg_track_result,
      acg_orderno,
      iif(coalesce(acg_caption, '') <> '', acg_caption, rd_name),
      rd_code,
      acg_str_tag
    from actiongroup
      inner join refdata
        on rd_id = acg_from_rd_id
    where acg_group_rd_id = :anketa_id
      and acg_is_delete = 0
    order by acg_orderno, acg_id
  into
    :line_id,
    :line_parent_id,
    :res_rd_id,
    :acg_caption,
    :rd_editor_kind,
    :acg_track_result,
    :rd_orderno,
    :rd_name,
    :rd_code,
    :rd_mru_list
  do
  begin
    res_state = 0;
    res_str_value = '';
    if (rd_mru_list is null) then
      rd_mru_list = '';
    res_final_str_value = '';
    orig_state = 0;

    if (line_parent_id is null) then -- если это главная строка (вопрос), уточняем вид отслеживания
    begin
      if (rd_code = 'work_address_metro') then
        need_metro = 1;

      rd_ques_orderno = rd_orderno;
      answer_no = 0;
      answer_count = 0;

-- Вычисляем количество ответов
      select
        count(*)
      from actiongroup
      where acg_parent_id = :line_id
        and acg_is_delete = 0
      into
        :answer_count;

-- если track_result=1 значит значение храниться с привязкой к вопросу
      track_result = acg_track_result;
      res_question_id = res_rd_id;

      need_show_uncheckall = 1;
-- если есть хоть одна запись с ответом, отменяем галочку "Ответ не выбран"
      if (exists(
        select *
        from objectprops
          inner join objectvalres
            on ovr_from_opr_id = opr_id
              and ovr_from_ovs_id = opr_actual_ovs_id
        where opr_object_kind = :acl_object_kind
          and opr_object_id = :acl_object_id
          and opr_subject_kind = -1
          and opr_subject_id = :res_rd_id))
      then need_show_uncheckall = 0;

      -- проверяем актуальность ответов (за исключением некоторых), если они не актуальны, то ставим отметку, что вопрос не имеет ответа
      if (db_id = 81001 and
          anketa_name = 'Anketa.Main') then
      begin
        if (coalesce(k_id, ma_id) is not null and
            (pr_code containing '.universal.' or pr_code containing '.oivo_pack.')) then
        begin
          if (exists(select *
                     from objectprops
                     inner join objectvalres on ovr_from_opr_id = opr_id and
                           ovr_from_ovs_id = opr_actual_ovs_id
                     inner join refdata on opr_subject_id = rd_id
                     inner join objectvalues on ovs_id = opr_actual_ovs_id and
                           ovs_from_opr_id = opr_id
                     where opr_object_kind = :acl_object_kind and
                           opr_object_id = :acl_object_id and
                           opr_subject_kind = -1 and
                           opr_subject_id = :res_rd_id and
                           (ovs_date_begin <= current_date - 90 and
                           rd_code in ('AnketaQuestions.LoanPurpose', 'AnketaQuestions.InfoSource', 'work_place_name', 'work_address_orientir', 'work_address_metro', 'work_staj_last_place', 'work_status', 'work_monthly_income', 'work_income_period', 'work_income_days',
                                       'work_business_kind', 'work_job_name', 'work_boss_phone_no', 'work_extra_income', 'work_extra_amount', 'AnketaQuestions.MonetaryObligations', 'AnketaQuestions.DefendantInCourt', 'AnketaQuestions.previous_fio')))) then
          begin
            need_show_uncheckall = 1;
          end
        end
        else
        begin
          if (exists(select *
                     from objectprops
                     inner join objectvalres on ovr_from_opr_id = opr_id and
                           ovr_from_ovs_id = opr_actual_ovs_id
                     inner join refdata on opr_subject_id = rd_id
                     inner join objectvalues on ovs_id = opr_actual_ovs_id and
                           ovs_from_opr_id = opr_id
                     where opr_object_kind = :acl_object_kind and
                           opr_object_id = :acl_object_id and
                           opr_subject_kind = -1 and
                           opr_subject_id = :res_rd_id and
                           (ovs_date_begin <= current_date - 90 and
                           rd_code not in ('AnketaQuestions.5905', 'AnketaQuestions.Use_BankCard', 'AnketaQuestions.Education', 'AnketaQuestions.InfoSource', 'AnketaQuestions.Ownership', 'AnketaQuestions.5943', 'AnketaQuestions.Nation', 'AnketaQuestions.CanBeSurety', 'AnketaQuestions.previous_fio', 'AnketaQuestions.5915')))) then
          begin
            need_show_uncheckall = 1;
          end
        end
      end

-- Уточняем RD_ID выбранного ответа в переменной checked_radio_id,
-- чтоб в дальнейшем этот ответ отметить едининой (как checked)
      res_state = 0;
      orig_state = 0;
      select
        ovs_id,
        ovs_add_value,
        ovs_add_value
      from hp_objectprops_get(
        :acl_object_kind,
        :acl_object_id,
        -1,
        :res_rd_id,
        :acl_date)
      into
        :ovs_id,
        :res_state,
        :orig_state;
      if (res_state is null) then
        res_state = 0;
      if (orig_state is null) then
        orig_state = 0;
    end
    else
    begin
      if (need_metro = 1 and
          rd_code = 'Results.StrValue') then
      begin
        need_metro = 0;
        select
          list(rd_name, ';')
        from refdata
        where rd_main_code = 'Metropoliten'
          and rd_main_code <> rd_code
        into
          :rd_mru_list;
      end
      answer_no = answer_no + 1;
    end

    if (line_parent_id is not null) then -- если это ответы
    begin
      res_state = 0;
      orig_state = 0;
      res_str_value = '';
      res_final_str_value = '';
      select
        ovr_state,
        ovr_state,
        ovr_value
      from objectprops
        inner join objectvalres
          on ovr_from_opr_id = opr_id
            and ovr_from_ovs_id = :ovs_id
            and ovr_result_rd_id = :res_rd_id
      where opr_object_kind = :acl_object_kind
        and opr_object_id = :acl_object_id
        and opr_subject_kind = -1
        and opr_subject_id = :res_question_id
      into
        :res_state,
        :orig_state,
        :res_str_value;

      -- RFCKG-28 Автозаполнение анкеты ЦМ из Анкеты анализа клиента
      if (:db_id in (39801, 41701) and
          :anketa_name = 'Anketa.TargetUseLoan' and
          :acl_object_kind = f_objkind('knclient', 'id')) then
      begin
        -- Выбор типа кредитования
        if (:res_question_id = f_get_rd_id('AnketaQuestions.LoanUseFact') and
            :loan_target = :rd_code) then
        begin
            :res_state = 1;
            :need_show_uncheckall = 0;
        end
        -- Заполнение примечания
        else if (:res_question_id = f_get_rd_id('Monitoring.Memo') and
                 :rd_code = 'Results.StrValue' and
                 coalesce(:res_str_value, '') = '') then
        begin
            select coalesce(cast(max(kl.kl_datetime) as date), current_date)
            from kontlog kl
            where kl.kl_from_k_id = :k_id and
                  kl.kl_state = 1
            into :rep_date;
    
            -- RFCKG-4309: Доработка текста, который автоматически выводится в поле "Примечание"
            select loan_description
            from hp_get_loan_description(:c_id, :ma_id)
            into :loan_description;

            select pr.pr_code
            from miniapp ma
            inner join product pr on ma.ma_from_pr_id = pr.pr_id
            where
                ma.ma_id = :ma_id and
                ma.ma_is_active = 1 and
                ma.ma_date_end = '31.12.9999'
            order by ma.ma_date_app desc
            rows 1
            into :pr_code;

            -- RFCKG-4494: сокращаем длину текста до 255 символов
            if (pr_code containing '.IPF.') then
                :res_str_value = left('Да, клиент использовал финансирование на ' || loan_description, 255);
            else
                :res_str_value = left('Да, клиент использовал кредит на ' || loan_description, 255);

            :res_state = 1;
            :need_show_uncheckall = 0;
        end 
      end 
      -- RFCKG-28 END

      -- если ответ не актуальный, то скрываем его
      if (db_id = 81001 and
          anketa_name = 'anketa.main') then
      begin
        if (coalesce(k_id, ma_id) is not null and
            (pr_code containing '.universal.' or pr_code containing '.oivo_pack.')) then
        begin
          select iif(ovs_date_begin <= current_date - 90 and rd_code in ('AnketaQuestions.LoanPurpose', 'AnketaQuestions.InfoSource', 'work_place_name', 'work_address_orientir', 'work_address_metro', 'work_staj_last_place', 'work_status', 'work_monthly_income', 'work_income_period', 'work_income_days',
                                                                         'work_business_kind', 'work_job_name', 'work_boss_phone_no', 'work_extra_income', 'work_extra_amount', 'AnketaQuestions.MonetaryObligations', 'AnketaQuestions.DefendantInCourt', 'AnketaQuestions.previous_fio'), 0, ovr_state),
                 ovr_state, ovr_value
          from objectprops
          inner join objectvalres on ovr_from_opr_id = opr_id and
                ovr_from_ovs_id = :ovs_id and
                ovr_result_rd_id = :res_rd_id
          inner join refdata on opr_subject_id = rd_id
          inner join objectvalues on ovs_id = opr_actual_ovs_id and
                ovs_from_opr_id = opr_id
          where opr_object_kind = :acl_object_kind and
                opr_object_id = :acl_object_id and
                opr_subject_kind = -1 and
                opr_subject_id = :res_question_id
          into :res_state, :orig_state, :res_str_value;

          if (res_question_id = f_get_rd_id('AnketaQuestions.CanBeSurety')) then
          begin
            res_state = 0;
            if (rd_code = 'AnketaAnswers.6719') then
            begin
              res_state = 1;
              need_show_uncheckall = 0;
            end
          end

          if (res_question_id = f_get_rd_id('AnketaQuestions.Nation')) then
          begin
            res_state = 0;
            if (rd_code = 'Nation.RU') then
            begin
              res_state = 1;
              need_show_uncheckall = 0;
            end
          end
        end
        else
          select iif(ovs_date_begin <= current_date - 90 and rd_code not in ('AnketaQuestions.5905', 'AnketaQuestions.Use_BankCard', 'AnketaQuestions.Education', 'AnketaQuestions.InfoSource', 'AnketaQuestions.Ownership', 'AnketaQuestions.5943', 'AnketaQuestions.Nation', 'AnketaQuestions.CanBeSurety', 'AnketaQuestions.previous_fio', 'AnketaQuestions.5915'), 0, ovr_state),
                 ovr_state, ovr_value
          from objectprops
          inner join objectvalres on ovr_from_opr_id = opr_id and
                ovr_from_ovs_id = :ovs_id and
                ovr_result_rd_id = :res_rd_id
          inner join refdata on opr_subject_id = rd_id
          inner join objectvalues on ovs_id = opr_actual_ovs_id and
                ovs_from_opr_id = opr_id
          where opr_object_kind = :acl_object_kind and
                opr_object_id = :acl_object_id and
                opr_subject_kind = -1 and
                opr_subject_id = :res_question_id
          into :res_state, :orig_state, :res_str_value;
      end

      if (res_state is null) then
        res_state = 0;
      if (orig_state is null) then
        orig_state = 0;
      if (res_str_value is null) then
        res_str_value = '';
      if (res_str_value <> '') then
        res_final_str_value = res_str_value;
      if (res_state <> 1 and
          res_str_value <> '') then
        res_str_value = '';
      if (res_str_value <> '') then
        res_final_str_value = res_str_value;
    end

    select
      rd_description
    from hp_actions_make_description(
      :anketa_id,
      :res_rd_id,
      :rd_code,
      :acl_object_kind,
      :acl_object_id)
    into
      :rd_description;

    rd_description = acg_caption;
    suspend;

    if (answer_no = answer_count and
        isvisitcheck = 0) then -- если это не визит осмотр, а анкета, добавляем спец.ответы
    begin
      rd_editor_kind = 0;
      rd_description = '';
      rd_mru_list = '';
      res_rd_id = res_id_noanswer;
      acg_caption = '';
      rd_orderno = rd_ques_orderno + 0.5;
      rd_name = iif(db_id <> 94401, 'Вопрос еще не задан', 'cavab seçilməyib');
      rd_code = 'Results.NoAnswer';
      res_state = need_show_uncheckall;
      orig_state = 0;
      res_str_value = '';
      res_final_str_value = '';
      suspend;
    end
  end
end
