CREATE PROCEDURE HP_COMPET_AZ_2024_SUPERP(
    C_ID D_ID,
    REG_ID D_ID,
    REP_PERIOD D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    IN_SORTING_ORDER D_SMALLINT = 0)
RETURNS (
    SUPER_POKAZATEL D_PERCENT,
    SUPER_POKAZATEL_TXT D_STR_255,
    ROWS_DESC D_SMALLINT)
AS
declare variable sp_type d_smallint;
declare variable bonus_total d_currency;
declare variable bonus_total_prev d_currency;
declare variable k_count d_percent;
declare variable kc_count d_percent;
begin
  if (in_sorting_order = 1) then
  begin
    if (rep_period in (202401, 202402)) then
      rows_desc = 1; --Если нужно сортировать показатели с крупных до малого (10,9,8 и т.д)
    else
      rows_desc = 0; --Если нужно сортировать с наименьшего до наибольшего -1,0,1 и т.д.

    suspend;
    exit; --Для расчета показателей идет отдельный запрос!!!
  end
  if (c_id is not null) then
    sp_type = 0;
  else
    sp_type = 1;

  if (rep_period = 202401) then
  begin
    super_pokazatel = 100;
    super_pokazatel_txt = 'Супер-показатель за январь: Тестовый показатель 100';
  end

  if (rep_period in (202402, 202405)) then
  begin
    select sum(aarep_bonus_total)
    from cb_az_lo_201602_one_lo_2_new(:reg_id, 0, :rep_date_end, 0)
    where jk_name in ('Baş kredit mütəxəssisi', 'Кредитный сотрудник')
    into bonus_total;

    select sum(aarep_bonus_total)
    from cb_az_lo_201602_one_lo_2_new(:reg_id, 0, '30.04.2024', 0)
    where jk_name in ('Baş kredit mütəxəssisi', 'Кредитный сотрудник')
    into bonus_total_prev;

    super_pokazatel = bonus_total -  bonus_total_prev;
    super_pokazatel_txt = 'Супер-показатель за февраль: Прирост заработка КС';
  end

  if (rep_period = 202406) then
  begin
    super_pokazatel = 0;
    super_pokazatel_txt = 'Супер-показатель за июнь: Среднее кол-во выдач на одного КС в команде';

    if (reg_id not in (1070400, 1040500)) then
    begin
      select count(*)
      from kontstate ks
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_region_id =:reg_id
      into :k_count;

      select count(*)
      from hr_pstatus ps
      inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where ps.ps_from_reg_id = :reg_id and
            :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
            jk.jk_system_code containing '.lo_standart' and
            ps.ps_status = 1 and
            ps.ps_is_working = 1
      into :kc_count;

      super_pokazatel = k_count;
    end

    if (reg_id = 1070400) then
    begin
      select count(*)
      from kontstate ks
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_region_id in (1070400, 1040500)
      into :k_count;

      select count(*)
      from hr_pstatus ps
      inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where ps.ps_from_reg_id in (1070400, 1040500) and
            :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
            jk.jk_system_code containing '.lo_standart' and
            ps.ps_status = 1 and
            ps.ps_is_working = 1
      into :kc_count;

       super_pokazatel = k_count;
    end
  end

  suspend;
end
