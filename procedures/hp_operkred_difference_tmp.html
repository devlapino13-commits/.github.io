CREATE PROCEDURE HP_OPERKRED_DIFFERENCE_TMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OK_FROM_K_ID DECIMAL(18,0),
    OK_SH_ID INTEGER,
    OK_SH_DATE DATE,
    OL_ID DECIMAL(18,0),
    OL_DATE DATE,
    OK_OS DECIMAL(18,4),
    OK_PERC DECIMAL(18,4))
AS
begin
  for select ok_from_k_id, ok_sh_id, ok_sh_date,
             /*  max(ol_id) as ol_id,*/
             max(ok_ol_date) as ol_date, sum(ok_os + ok_sh_os) as ok_os, sum(ok_perc + ok_sh_perc) as ok_perc
      from operkred, kontstate
      where ok_ol_is_active = 1 and
            ok_ol_date <= :rep_date_end and
            ok_type not in (7, 8) and
            ok_sh_date <= :rep_date_end and
            ks_ol_is_delete = 0 and
            ks_from_k_id = ok_from_k_id and
            ks_state = 1 and
            ok_from_c_id = 0 and
            :rep_date_end between cast(ks_date_begin as date) and ks_date_end
            /* and ol_from_reg_id=1070400        /* временная строка, ограничение по офисам*/
            /* and ok_from_k_id=109010004063600101*/
      group by ok_from_k_id, ok_sh_id, ok_sh_date
      having(sum(ok_os + ok_sh_os) <> 0 or (sum(ok_perc + ok_sh_perc) <> 0)) and (max(ok_ol_date) >= :rep_date_begin)
      into :ok_from_k_id, :ok_sh_id, :ok_sh_date,
           /*  :OL_ID,*/
           :ol_date, :ok_os, :ok_perc
  do
  begin
    for select ok_from_ol_id
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :ok_from_k_id and
              ok_ol_date = :ol_date and
              ok_sh_id = :ok_sh_id and
              ok_type = 4 and
              ok_from_c_id = 0
        into :ol_id
    do
    begin
      suspend;
    end
  end
end
