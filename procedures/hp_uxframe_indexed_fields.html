CREATE PROCEDURE HP_UXFRAME_INDEXED_FIELDS(
    FR_CODE D_STR_CODE)
RETURNS (
    ID D_STR_255,
    ID_VISIBLE D_STR_255,
    TEXT D_STR_255)
AS
declare variable ufd_source d_str_code;
declare variable is_table d_bool;
declare variable idx_name d_str_code;
declare variable field_name d_str_code;
declare variable field_caption d_str_code;
begin
  select ufd_source, ufd_is_table
  from uxfrdata
  where ufd_ufr_code = :fr_code and
        ufd_kind = 0
  into ufd_source, is_table;

  if (row_count = 0) then
    exit;

  ufd_source = upper(ufd_source);

  if (is_table = 1) then
  begin
    for select i.rdb$index_name
    from rdb$indices i
    where  i.rdb$relation_name = :ufd_source
    into idx_name
    do
    begin
      id = '';
      text = '';
      for select trim(lower(s.rdb$field_name))
      from rdb$index_segments s
      where s.rdb$index_name = :idx_name
      order by rdb$field_position
      into field_name
      do
      begin
        field_caption = '';
        select ufp_name
        from uxfrparam
        where ufp_ufr_code = :fr_code and
              ufp_origin_name = :field_name and
              ufp_parent_code is null
        into field_caption;

        if (row_count = 0) then
          break;

        if (id <> '') then
        begin
          id = id || ';';
          text = text || ',  ';
        end
        id = id || field_name;
        text = text || field_caption;

        id_visible = id;
        suspend;
      end
    end
  end else
  begin
    for select trim(lower(p.rdb$parameter_name))
    from rdb$procedure_parameters p
    where p.rdb$procedure_name = :ufd_source and
          p.rdb$parameter_type = 0
    into id
    do
    begin
      text = '';
      select ufp_name
      from uxfrparam
      where ufp_ufr_code = :fr_code and
            ufp_origin_name = :id and
            ufp_parent_code is null
      into text;

      if (row_count = 0) then
        text = id;

      id_visible = id;
      suspend;
    end
  end
end
