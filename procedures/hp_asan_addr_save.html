CREATE PROCEDURE HP_ASAN_ADDR_SAVE(
    ADRRLIST D_STR_255)
RETURNS (
    ADL_ID D_ID)
AS
declare variable address d_str_255;
declare variable parent_id d_id;
declare variable pl_id_temp d_id;
declare variable un_virt d_id;
declare variable address_part d_str_255;
declare variable plt_part d_str_255;
declare variable plt_id d_id;
declare variable pl_id d_id;
begin
  parent_id = 944000000001;
  un_virt = 1000000000000;
  for select lower(value_out)
      from STR_WITH_SEPARATOR_PARSER_ASAN(trim(:adrrlist), ', ')
      into :address
  do
  begin
    --    ADDRESS_PART = null;
    --    select first 1 VALUE_OUT
    --    from STR_WITH_SEPARATOR_PARSER(:ADDRESS, ' ')
    --    into ADDRESS_PART;

    plt_part = null;

    select first 1 lower(reverse(value_out))
    from STR_WITH_SEPARATOR_PARSER_ASAN(reverse(:address), ' ')
    into plt_part;

    address_part = trim(lower(replace(:address, :plt_part, '')));

    plt_id = null;

    if (plt_part containing '.') then
    begin
      plt_part = replace(plt_part, 'ray.', 'rayonu');
      plt_part = replace(plt_part, 'şəh.', 'şəhəri');
      plt_part = replace(plt_part, 's.', 'şəhəri');
      plt_part = replace(plt_part, 'küç.', 'küçəsi');
      plt_part = replace(plt_part, 'k.', 'kəndisi'); --
      plt_part = replace(plt_part, '', '');
      plt_part = replace(plt_part, '', '');
    end

    select plt.plt_id
    from placetype plt
    where :plt_part containing lower(plt.plt_name)
    rows 1
    into plt_id;

    plt_id = coalesce(:plt_id, 944000);

    if (plt_id = 944004) then
    begin
      address_part = replace(:address_part, 'ş', 's');
      address_part = replace(:address_part, 'ə', 'a');
      address_part = replace(:address_part, 'ğ', 'g');
      address_part = replace(address_part, 'ü', 'u');
      address_part = replace(:address_part, 'ç', 'c');
    end

    pl_id = null;
    pl_id_temp = null;

    select pl.pl_id
    from places pl
    where pl.pl_status = :plt_id and
          pl.is_delete = 0 and
          lower(pl.pl_name) = :address_part rows 1
    into :pl_id_temp;

    if (pl_id_temp is null) then
    begin
      pl_id_temp = (select pl_id
                    from hp_insert_place_node(:parent_id, :address_part, :plt_id, :un_virt, null, 1));

      pl_id = pl_id_temp;

      parent_id = pl_id_temp;
    end
    else
    begin
      pl_id = pl_id_temp;

      parent_id = pl_id_temp;
    end

    if (:plt_id > 944007) then
    begin
      insert into addrlist (adl_from_pl_id, adl_depart_no, adl_depart_pref, adl_flat_no, adl_building_no,
                            adl_section_no, adl_cross_street_id, adl_address_str, adl_comment, adl_post_index,
                            adl_is_edited, adl_from_un_id, adl_street_id, adl_is_without_street, adl_is_filled,
                            adl_long, adl_lat, adl_geo_count, adl_place_pls_id, adl_street_pls_id, adl_lang_kind)
      values (:parent_id, 0, '', null, 0, 0, null, :adrrlist, null, null, 0, :un_virt, :pl_id,
              iif(:pl_id is null, 1, 0), 1, null, null, -1, null, null, 0)
      returning adl_id
      into :adl_id;
      suspend;
      exit;
    end

  end
    insert into addrlist (adl_from_pl_id, adl_depart_no, adl_depart_pref, adl_flat_no, adl_building_no,
                            adl_section_no, adl_cross_street_id, adl_address_str, adl_comment, adl_post_index,
                            adl_is_edited, adl_from_un_id, adl_street_id, adl_is_without_street, adl_is_filled,
                            adl_long, adl_lat, adl_geo_count, adl_place_pls_id, adl_street_pls_id, adl_lang_kind)
      values (:parent_id, 0, '', null, 0, 0, null, :adrrlist, null, null, 0, :un_virt, null,
              0, 1, null, null, -1, null, null, 0)
      returning adl_id
      into :adl_id;
  suspend;
end
