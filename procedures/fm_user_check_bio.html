CREATE PROCEDURE FM_USER_CHECK_BIO(
    UN_NAME D_STR_CODE,
    C_ID D_ID,
    JSON_VALIDATION_DETAILS D_STR_8100)
RETURNS (
    ERR_TEXT D_STR_255,
    SAME_PERSON SMALLINT,
    P_INFO VARCHAR(255))
AS
declare variable un_fio d_str_255;
declare variable job_name d_str_255;
declare variable quota_exceeded d_smallint;
declare variable msg_subject d_str_code;
declare variable msg_body d_str_255;
declare variable un_id d_id;
declare variable liveness_score d_percent;
begin
  /* RFCKG-3940
    Первым делом в ответе от /user/check_bio/ проверяю поле err_text. Если оно
    не пустое, то я его показываю и выхожу из окна Фотоидентификации в окно Логина.
    (причиной непустого err_text могут быть закончившиеся попытки или критические ошибки)

    Если в ответ на /user/check_bio/ поле same_person = 1, т.е. фотоидентификация
    прошла успешно, я показываю сообщение "Подтверждаете отправку пароля на номер XXXXXXX?"
    и выхожу из окна Фотоидентификации в окно Логина.

    Если в ответ на /user/check_bio/ поле same_person = 0, т.е. фотоидентификация
    прошла неуспешно (но попытки есть), я показываю сообщение из поля p_info и
    предлагаю пройти фотоидентификацию повторно.
  */
  same_person = 0;
  err_text = '';
  quota_exceeded = 0;

  select un_id, un_fio
  from username
  where un_name = :un_name and
        un_from_c_id = :c_id rows 1
  into un_id, un_fio;

  select quota_exceeded
  from hp_msgquota_process('mobile.staff.bio_reset_pwd', :un_name)
  into quota_exceeded;

  if (quota_exceeded = 1) then
  begin
      err_text = 'Количество попыток превышено, повторите попытку через 15 минут';

      select jl.jl_name
      from hr_pstatus ps
      inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
      where ps.ps_from_c_id = :c_id and
            ps.ps_is_working = 1 and
            current_date between ps.ps_date_begin and ps.ps_date_end
      into job_name;

      msg_subject = 'Неудачная попытка смены пароля';
      msg_body = :un_fio || ', ' || :job_name || ', ' || cast(current_timestamp as timestamp) || ', не смог(ла) завершить процесс смены пароля телефона в мобильном приложении сотрудника.';

      insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                  msg_object_kind, msg_object_id)
      values (0, 'control_kg.mirm@mbulak.kg', :msg_subject, :msg_body, 7, 1000000000000, 0, 0); -- на время проверки
      --        values (0, 'h.raimjanov@mbulak.kg', :msg_subject, :msg_body, 1, 1000000000000, 0, 0); -- на время проверки
      --        values (0, 'control_kg.mirm@mbulak.kg', :msg_subject, :msg_body, 7, 1000000000000, 0, 0);

      suspend;
      exit;
  end

  same_person = f_json_blob_get_value(:json_validation_details, 'same_person', 0);
  liveness_score = f_json_blob_get_value(:json_validation_details, 'liveness_score', 0);
  p_info = f_json_blob_get_value(:json_validation_details, 'p_info', '');

  insert into bio_results (br_from_un_id, br_result, br_match_percentage, br_attempts_number, br_validation_details)
  values (:un_id, :same_person, :liveness_score, 1, :json_validation_details);
  suspend;
end
