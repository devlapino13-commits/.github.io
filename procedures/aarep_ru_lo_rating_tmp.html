CREATE PROCEDURE AAREP_RU_LO_RATING_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    RES_WITH_LO INTEGER)
RETURNS (
    RES_REGION_ID INTEGER,
    RES_LO_ID DECIMAL(18,0),
    RES_JUST_CL_END INTEGER,
    RES_BALANCE_END DECIMAL(18,4),
    RES_DELINQ_BALANCE7_END DECIMAL(18,4),
    RES_JUST_CL_BEGIN INTEGER,
    RES_BALANCE_BEGIN DECIMAL(18,4),
    RES_DELINQ_BALANCE7_BEGIN DECIMAL(18,4),
    RES_DELINQ_BALANCE30_BEGIN DECIMAL(18,4),
    RES_DELINQ_BALANCE30_END DECIMAL(18,4),
    RES_DELINQ_BALANCE1_BEGIN DECIMAL(18,4),
    RES_DELINQ_BALANCE1_END DECIMAL(18,4),
    RES_DISB_NEW INTEGER,
    RES_DISB_REPEAT INTEGER,
    RES_PLAN_OS DECIMAL(18,4),
    RES_FACT_OS DECIMAL(18,4),
    IS_TRANSFERED INTEGER,
    OKS_DELINQ_DATE DATE,
    K_ID DECIMAL(18,0),
    NEED_SUSPEND INTEGER)
AS
declare variable res_balance_tmp decimal(18,4) = 0;
declare variable kf_id decimal(18,0);
declare variable rep_dd date;
declare variable rep_dd_begin date;
declare variable os_tmp decimal(18,4) = 0;
declare variable dd_disb_date date;
declare variable dd date;
declare variable res_lo_id_end decimal(18,0);
declare variable reg_new integer;
begin
-- ПОКАЗАТЕЛИ НА НАЧАЛО ПЕРИОДА
  rep_dd = :rep_date_begin - 1;
  rep_dd_begin = (:rep_date_begin - 10);
  if (res_with_lo is null) then
    res_with_lo = 1;

  res_just_cl_begin = 0;
  res_balance_begin = 0;
  res_delinq_balance7_begin = 0;
  res_delinq_balance30_begin = 0;

  res_just_cl_end = 0;
  res_balance_end = 0;
  res_delinq_balance7_end = 0;
  res_delinq_balance30_end = 0;

  res_delinq_balance1_begin = 0;
  res_delinq_balance1_end = 0;
  res_disb_new = 0;
  res_disb_repeat = 0;

  res_plan_os = 0;
  res_fact_os = 0;
  is_transfered = 0;

-- Баланс и кол-во клиентов на конец по 1 и 2 категории
  for select oks_from_k_id, oks_region_id, oks_kc_id, oks_balance, oks_delinq_date, k_disb_date, k_from_kf_id
      from oksummary, kontract, region_list(:rep_region_id, 1, 3, 1)
      where oks_date_begin <= :rep_dd and
            oks_date_end >= :rep_dd and
            oks_balance > 0
--  and (:REP_DD-oks_delinq_date)<=180
            and
            oks_from_k_id = k_id and
            oks_close_state = 0 and
            oks_region_id = id_list
--  and oks_from_k_id=101010001185300701
            and
            ((oks_kc_id > 1000000000004 and
            :res_with_lo = 1) or
            (:res_with_lo = 0 and
            oks_kc_id <> 1000000000003))
      into :k_id, :res_region_id, :res_lo_id, :res_balance_tmp, :oks_delinq_date, :dd_disb_date, :kf_id
  do
  begin
    is_transfered = 0;
    select kal_new_kf_officer, kal_new_reg_id
    from knaktlist
    where kal_from_kf_id = :kf_id and
          kal_date <= :rep_date_end and
          kal_date_end >= :rep_date_end and
          knaktlist.is_delete = 0
    into :res_lo_id_end, :reg_new;
    if (res_lo_id_end <> res_lo_id) then
      is_transfered = 1;

    if ((res_with_lo = 0) and
        (is_transfered = 1) and
        (res_lo_id_end <> 1000000000003)
--    and (res_lo_id_end<>1000000000004)
        ) then
      is_transfered = 0;

    if ((res_with_lo = 0) and
        (is_transfered = 0) and
        (reg_new <> res_region_id) and
        (res_lo_id_end <> res_lo_id) and
        (res_lo_id_end <> 1000000000004)) then
      is_transfered = 1;
    if (res_with_lo = 0) then
      res_lo_id = 0;

    need_suspend = 0;
    res_balance_begin = res_balance_tmp;
    res_delinq_balance7_begin = 0;
    res_delinq_balance30_begin = 0;
    res_delinq_balance1_begin = 0;

    if (is_transfered = 0) then
    begin
      if (:rep_dd - oks_delinq_date > 7) then
        res_delinq_balance7_begin = res_balance_tmp;
      if (:rep_dd - oks_delinq_date > 30) then
        res_delinq_balance30_begin = res_balance_tmp;
      if (:rep_dd - oks_delinq_date >= 1) then
        res_delinq_balance1_begin = res_balance_tmp;
    end

    need_suspend = 1;
    if (:rep_date_end - oks_delinq_date > 180 and
        rep_date_end < '24.01.2016') then
    begin
-- находящиеся у внешних коллекторов не будут считатся
      if (exists(select *
                 from sb_akt, sb_list
                 where sba_is_delete = 0 and
                       sba_from_sbl_id <> 0 and
                       sba_status = 2 and
                       :rep_dd between sba_date_begin and sba_date_end and
                       sba_from_k_id = :k_id and
                       sba_from_sbl_id = sbl_id and
                       sbl_type = 1)) then
        need_suspend = 0;
      else
      begin
-- находящиеся у внешних коллекторов не будут считатся
        if (exists(select *
                   from sb_akt, sb_list
                   where sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_status = 2 and
                         :rep_date_end between sba_date_begin and sba_date_end and
                         sba_from_k_id = :k_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1)) then
        begin
          res_delinq_balance7_begin = 0;
          res_delinq_balance30_begin = 0;
          res_delinq_balance1_begin = 0;
        end
      end
      if (need_suspend = 1) then
      begin
        res_delinq_balance7_begin = 0;
        res_delinq_balance30_begin = 0;
        res_delinq_balance1_begin = 0;
      end
    end
    res_just_cl_begin = 0;
    if ((is_transfered = 0) and
        (reg_new <> res_region_id)) then
      res_region_id = reg_new;
    if ((need_suspend = 1) and
        (is_transfered = 0)) then
    begin
      select res_just_client_count
      from hp_get_just_client_in_loan(:rep_dd, :k_id, :dd_disb_date)
      into :res_just_cl_begin;

      if (res_with_lo = 1) then
        res_region_id = 0;
      suspend;
    end
  end
-- ПОКАЗАТЕЛИ НА КОНЕЦ ПЕРИОДА
  rep_dd = :rep_date_end;
  rep_dd_begin = :rep_date_begin;

  res_just_cl_begin = 0;
  res_balance_begin = 0;
  res_delinq_balance7_begin = 0;
  res_delinq_balance30_begin = 0;

  res_just_cl_end = 0;
  res_balance_end = 0;
  res_delinq_balance7_end = 0;
  res_delinq_balance30_end = 0;

  res_delinq_balance1_begin = 0;
  res_delinq_balance1_end = 0;
  res_disb_new = 0;
  res_disb_repeat = 0;

  res_plan_os = 0;
  res_fact_os = 0;
  is_transfered = 0;

-- Баланс и кол-во клиентов на конец по 1 и 2 категории
  for select oks_from_k_id, oks_region_id, oks_kc_id, oks_balance, oks_delinq_date, k_disb_date, k_from_kf_id
      from oksummary, kontract, region_list(:rep_region_id, 1, 3, 1)
      where oks_date_begin <= :rep_dd and
            oks_date_end >= :rep_dd and
            oks_balance > 0
--  and (:REP_DD-oks_delinq_date)<=180
            and
            oks_from_k_id = k_id and
            oks_close_state = 0 and
            oks_region_id = id_list
--  and oks_from_k_id=101010001185300701
            and
            ((oks_kc_id > 1000000000004 and
            :res_with_lo = 1) or
            (:res_with_lo = 0 and
            oks_kc_id <> 1000000000003))
      into :k_id, :res_region_id, :res_lo_id, :res_balance_tmp, :oks_delinq_date, :dd_disb_date, :kf_id
  do
  begin
    need_suspend = 0;
    res_balance_end = res_balance_tmp;
    res_delinq_balance7_end = 0;
    res_delinq_balance30_end = 0;
    res_delinq_balance1_end = 0;
    if (:rep_dd - oks_delinq_date > 7) then
      res_delinq_balance7_end = res_balance_tmp;
    if (:rep_dd - oks_delinq_date > 30) then
      res_delinq_balance30_end = res_balance_tmp;
    if (:rep_dd - oks_delinq_date >= 1) then
      res_delinq_balance1_end = res_balance_tmp;
-- считать только не переданные за текущий период
    if (res_with_lo = 1) then
    begin
      need_suspend = 1;
--        if (exists(
--          select * from knaktlist
--          where knaktlist.is_delete = 0
--          and :REP_DD between kal_date and kal_date_end
--          and kal_from_kf_id = :kf_id
--          and kal_new_kf_officer = :res_lo_id
--          and kal_new_reg_id = :res_region_id
--          and (kal_date < :rep_dd_begin or kal_date <= :dd_disb_date)
--          )) then Need_Suspend=1;

      if (exists(select *
                 from knaktlist
                 where knaktlist.is_delete = 0 and
                       :rep_dd between kal_date and kal_date_end and
                       kal_from_kf_id = :kf_id and
                       kal_new_kf_officer = :res_lo_id and
                       kal_prior_lo_id <> :res_lo_id
--          and kal_new_reg_id = :res_region_id
                       and
                       kal_date >= :rep_dd_begin and
                       kal_date > :dd_disb_date)) then
        need_suspend = 0;
    end
    else
    begin
      if (res_lo_id > 1000000000004) then
      begin
        if (exists(select *
                   from knaktlist
                   where knaktlist.is_delete = 0 and
                         maxvalue((:rep_date_begin - 1), :dd_disb_date) between kal_date and kal_date_end and
                         kal_from_kf_id = :kf_id and
                         kal_new_reg_id = :res_region_id
--          and kal_new_kf_officer=:res_lo_id
            )) then
          need_suspend = 1;
      end
      else
      begin
        if ((res_lo_id <> 1000000000003)
--        and (res_lo_id<>1000000000004)
            ) then
          need_suspend = 1;
      end
    end

    if (res_with_lo = 0) then
      res_lo_id = 0;

    if (need_suspend = 1) then
    begin
-- находящиеся у внешних коллекторов не будут считатся
      if (:rep_date_end - oks_delinq_date > 180 and
          rep_date_end < '24.01.2016') then
      begin

        if (exists(select *
                   from sb_akt, sb_list
                   where sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_status = 2 and
                         :rep_dd between sba_date_begin and sba_date_end and
                         sba_from_k_id = :k_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1)) then
          need_suspend = 0;
      end
    end
    if (need_suspend = 1) then
    begin
      res_just_cl_end = 0;
      select res_just_client_count
      from hp_get_just_client_in_loan(:rep_dd, :k_id, :dd_disb_date)
      into :res_just_cl_end;

      suspend;
    end
  end

  res_just_cl_begin = 0;
  res_balance_begin = 0;
  res_delinq_balance7_begin = 0;
  res_delinq_balance30_begin = 0;

  res_just_cl_end = 0;
  res_balance_end = 0;
  res_delinq_balance7_end = 0;
  res_delinq_balance30_end = 0;

  res_delinq_balance1_begin = 0;
  res_delinq_balance1_end = 0;
  res_disb_new = 0;
  res_disb_repeat = 0;
  res_plan_os = 0;
  res_fact_os = 0;
  is_transfered = 0;

  for select ks_region_id, ks_kc_id, sum(iif(kc_kommis_lgota > 0.0, 1, 0)), count(*)
      from kontstate, knclient, kontract, product, region_list(:rep_region_id, 1, 3, 1)
      where ks_from_k_id = kc_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            kc_amount_approved > 0 and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_category = 0 and
            ks_region_id = id_list
--  and ks_from_k_id=101010001185300701
      group by ks_region_id, ks_kc_id
      into :res_region_id, :res_lo_id, res_disb_new, -- выданные через агентов
           res_disb_repeat -- выданные ВСЕ
  do
  begin
    if (res_with_lo = 0) then
      res_lo_id = 0;
    suspend;
  end

  res_just_cl_begin = 0;
  res_balance_begin = 0;
  res_delinq_balance7_begin = 0;
  res_delinq_balance30_begin = 0;

  res_just_cl_end = 0;
  res_balance_end = 0;
  res_delinq_balance7_end = 0;
  res_delinq_balance30_end = 0;

  res_delinq_balance1_begin = 0;
  res_delinq_balance1_end = 0;
  res_disb_new = 0;
  res_disb_repeat = 0;
  res_plan_os = 0;
  res_fact_os = 0;
  is_transfered = 0;

  for select distinct ks_from_k_id, kal_new_reg_id, kal_new_kf_officer, ks_date_begin
      from kontstate, kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1)
      where ks_ol_is_delete = 0 and
            ks_date_end >= (:rep_date_begin - 1) and
            ks_date_begin <= :rep_date_end and
            ks_state = 1 and
            kal_from_kf_id = k_from_kf_id and
            ks_from_k_id = k_id and
            kal_date <= maxvalue(:rep_date_begin, ks_date_begin) and
            kal_date_end >= maxvalue(:rep_date_begin, ks_date_begin) and
            knaktlist.is_delete = 0 and
            ks_region_id = id_list
--  and ks_from_k_id=101010001185300701
      into :k_id, :res_region_id, :res_lo_id, :dd_disb_date
  do
  begin
    if (res_with_lo = 0) then
      res_lo_id = 0;
    dd = maxvalue(:rep_date_begin, (dd_disb_date + 1));

    res_plan_os = 0;
    res_fact_os = 0;
    os_tmp = 0;

    for select sum(ok_os + ok_sh_os)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_date < :dd and
              ok_sh_date between :rep_date_begin and :rep_date_end and
              ok_ol_is_active = 1 and
              ok_type < 7 and
              ok_from_c_id = 0
        group by ok_sh_date
        having(sum(ok_os + ok_sh_os) <> 0)
        into :os_tmp
    do
    begin
      if (os_tmp is null) then
        os_tmp = 0;
      res_plan_os = res_plan_os + os_tmp;
    end

    for select -sum(ok_os)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_date between :rep_date_begin and :rep_date_end and
              ok_sh_date between :rep_date_begin and :rep_date_end and
              ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_from_c_id = 0 having(sum(ok_os) <> 0)
        into :os_tmp
    do
    begin
      if (os_tmp is null) then
        os_tmp = 0;
      res_fact_os = res_fact_os + os_tmp;
    end
    if ((res_plan_os <> 0) or (res_fact_os <> 0)) then
      suspend;
  end
end
