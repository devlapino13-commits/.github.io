CREATE PROCEDURE CM_KG_0057_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable ma_expert_fio varchar(100);
declare variable ma_expert_id id_type;
declare variable count_cl integer;
begin
  for
  select ma.ma_expert_id
  from miniapp ma
  where ma_id = :ma_id
  and current_date between ma.ma_date_app and ma_date_end
  and ma.ma_is_active = 1
  into ma_expert_id
  do
  begin
    count_cl = null;
    select count(distinct kc_from_c_id)
    from actionorder ao, refdata rd, knclient, kontstate
    where ao.aco_date <= current_date
    and ao.aco_respons_c_id = :ma_expert_id
    --and ao.aco_respons_c_id = 1090100437141
    and ao.aco_from_rd_id = rd_id  -- Мониторинг целевого использования
    and rd.rd_code = 'Anketa.TargetUseLoan'
    and ((ao.aco_object_kind = 0 and ao.aco_object_id = kc_from_c_id) or
         (ao.aco_object_kind = 8 and ao.aco_object_id = kc_id))
    and kc_amount_approved > 0
    and kc_from_k_id = ks_from_k_id
    and ks_state = 1
    and current_date between ks_date_begin and ks_date_end
    and ks_date_begin <= current_date-60
    into count_cl;
    if (count_cl is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :ma_expert_id
      into ma_expert_fio;
      res_error_text = 'У СПФ '||ma_expert_fio||' есть активные задачи по мониторингу, от даты выдачи прошло 60 дней. Количество: '||count_cl;
      res_error_is_critical = 1;
      suspend;
    end
  end
end
