CREATE PROCEDURE AAREP_NBKR_PREDPIS(
    REP_REGION_ID D_INTEGER)
RETURNS (
    REG_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    KONTRACT_NO D_STR_100,
    KS_FIO D_STR_100,
    C_NAME_FIO D_STR_100,
    HAS_ASP D_SMALLINT,
    HAS_VERIF_CELL_NO D_SMALLINT,
    PREDPISANIYE_STATUS D_STR_100,
    KONTRACT_STATUS D_STR_100,
    CLIENT_STATUS D_STR_100,
    CL_ACT_REG_NAME D_STR_100,
    CL_ACT_KONTRACT_NO D_STR_100,
    CL_ACT_KS_FIO D_STR_100,
    K_DISB_DATE D_DATE)
AS
declare variable c_id d_id;
declare variable cl_act_kf_id d_id;
declare variable k_id d_id;
declare variable kc_id d_id;
declare variable kf_id d_id;
declare variable edoc_status d_smallint;
begin
  for select reg2.reg_name, reg.reg_name, kont.k_kontract_no, cl.c_id, cl.c_name_fio, kont.k_id, kc_id,
             kont.k_from_kf_id, kont.k_disb_date
      from nbkr_predpisaniye t
      inner join knclient kncl on kncl.kc_id = t.np_from_kc_id
      inner join clients cl on kncl.kc_from_c_id = cl.c_id
      inner join kontract kont on kont.k_id = kncl.kc_from_k_id
      inner join region reg on reg.reg_unicode = kont.k_last_reg_id
      inner join region reg2 on reg2.reg_unicode = reg.parent
      inner join region_list(:rep_region_id, 1, 3, 1) on id_list = reg.reg_unicode
      into :reg_name, :office_name, :kontract_no, :c_id, :c_name_fio, :k_id, :kc_id, :kf_id,
           :k_disb_date
  do
  begin
    has_asp = 0;

    if (exists(select *
               from hp_edoc_status('asp.agreement', 0, :c_id)
               where client_sign_status = 1)) then
      has_asp = 1;

    has_verif_cell_no = 0;

    if (f_cont_info('clients', :c_id, 'cell_main', 'ok') <> '') then
      has_verif_cell_no = 1;

    ks_fio = '';

    predpisaniye_status = '';

    select edoc_status_str, edoc_status
    from hp_edoc_status('nbkr_predpisaniye.dop_perc', 8, :kc_id)
    into :predpisaniye_status, :edoc_status;
    if (:edoc_status <> 1) then
    begin
      if (exists(select *
                 from clientdocs cds
                 where cds.cds_from_c_id = :c_id and
                       cds.cds_from_pk_id = 337)) then
      begin
        :predpisaniye_status = 'Подписан вручную';
      end
    end

    select cl.c_name_fio
    from knaktlist kal
    inner join clients cl on kal.kal_new_kf_officer = cl.c_id
    where kal.kal_from_kf_id = :kf_id and
          current_date between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into :ks_fio;

    kontract_status = 'Закрытый';
    if (exists(select ks.ks_state
               from kontstate ks
               where ks.ks_ol_is_delete = 0 and
                     current_date between ks.ks_date_begin and ks.ks_date_end and
                     ks.ks_state = 1 and
                     ks.ks_from_k_id = :k_id)) then
    begin
      kontract_status = 'Активный';
      client_status = 'Активный';
    end

    cl_act_reg_name = null;
    cl_act_kf_id = null;
    cl_act_kontract_no = null;
    client_status = null;
    cl_act_ks_fio = null;

    if (kontract_status = 'Закрытый') then
    begin
      client_status = 'Закрытый';

      for select reg.reg_name, kont.k_from_kf_id, kont.k_kontract_no
          from kontstate ks
          inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
          inner join kontract kont on kont.k_id = ks.ks_from_k_id
          inner join region reg on reg.reg_unicode = kont.k_last_reg_id
          where ks.ks_ol_is_delete = 0 and
                current_date between ks.ks_date_begin and ks.ks_date_end and
                ks.ks_state = 1 and
                kncl.kc_from_c_id = :c_id
          order by ks.ks_date_begin desc
          rows 1
          into :cl_act_reg_name, :cl_act_kf_id, :cl_act_kontract_no
      do
      begin
        client_status = 'Активный';
        cl_act_ks_fio = '';

        select cl.c_name_fio
        from knaktlist kal
        inner join clients cl on kal.kal_new_kf_officer = cl.c_id
        where kal.kal_from_kf_id = :cl_act_kf_id and
              current_date between kal.kal_date and kal.kal_date_end and
              kal.is_delete = 0
        into :cl_act_ks_fio;
      end
    end

    suspend;
  end
end
