CREATE PROCEDURE UXP$UXFRITEM_TEMP(
    UX_UUID D_UUID,
    FULL_UI_ITEMS D_SMALLINT,
    UFM_UUID D_UUID = null,
    RES_KIND D_STR_CODE = null,
    UX_UUID_CODE D_STR_100 = null,
    UFM_UUID_CODE D_STR_CODE = null)
RETURNS (
    UFI_UUID D_UUID,
    UFI_PARENT_UUID D_UUID,
    UFI_UX_CODE D_STR_100,
    UFI_UFR_UUID D_UUID,
    UFI_UFR_CODE D_STR_100,
    UFI_UF_UUID D_UUID,
    UFI_KIND D_INTEGER,
    UFI_CONTROL D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_NAME D_STR_100,
    UFI_CAPTION_ALIGN D_INTEGER,
    UFI_HINT D_STR_LONG,
    UFI_HIDDEN D_SMALLINT,
    UFI_READONLY D_BOOL,
    UFI_GROUP_DIRECTION D_INTEGER,
    UFI_H_ALIGN D_INTEGER,
    UFI_V_ALIGN D_INTEGER,
    UFI_WIDTH D_INTEGER,
    UFI_HEIGHT D_INTEGER,
    UFI_SIZABLE D_BOOL,
    UFI_ORDER_INDEX D_SMALLINT,
    UFI_GROUP_INDEX D_SMALLINT,
    UFI_FIXED_KIND D_SMALLINT,
    UFI_SUMMARY_KIND D_SMALLINT,
    IMAGE_INDEX D_INTEGER,
    RES_CAPTION D_STR_255,
    RES_COMMENT D_STR_LONG,
    ALLOW_CHILD D_BOOL,
    UF_IS_PARAM D_SMALLINT,
    UF_FIELD_NAME D_STR_CODE,
    UF_SOURCE D_STR_CODE,
    UF_NOT_NULL D_BOOL,
    UF_SECURED D_BOOL,
    UFI_REFRESH D_BOOL,
    UFI_UUID_STR D_STR_CODE,
    UFI_PARENT_UUID_STR D_STR_CODE,
    UFP_PUBLIC D_BOOL,
    UFI_EVENT_CONTROL D_STR_CODE,
    UFI_BAND_INDEX D_INTEGER,
    UFI_HEADER_ALIGN D_SMALLINT,
    UFI_VALUE_ALIGN D_SMALLINT,
    UFI_FONT_SIZE D_SMALLINT)
AS
declare variable res_ufi_uuid d_uuid;
declare variable res_ufi_readonly d_bool;
declare variable ux_kind d_str_code;
declare variable uni_code d_str_code;
declare variable uf_type d_smallint;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code, 0);

  if (coalesce(ufm_uuid, '') = '') then
    ufm_uuid = uxf$uniq(ufm_uuid_code, 0);

  if (coalesce(ux_uuid, '') = '') then
    exit;

  if (full_ui_items is null) then
    full_ui_items = 0;

  ufi_band_index = -1;

  ufi_ux_code = uxf$code(ux_uuid, 'fcode');
  ux_kind = uxf$code(ufm_uuid, 'kind');

  res_kind = trim(coalesce(res_kind, ''));
  ufi_header_align = 0;
  ufi_value_align = 0;
  ufi_font_size = 0;

  if (res_kind = '' or res_kind = 'event') then
  begin
    if (res_kind = 'event') then
      full_ui_items = -1;

    for select res_ufi_uuid, res_parent_ufi, res_ufi_readonly, res_event_control,
               ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_h_align, ufi_v_align
    from uxp$uxfritem_tree(:ux_uuid, null, :full_ui_items, 0)
    into :res_ufi_uuid, :ufi_parent_uuid, :res_ufi_readonly, ufi_event_control,
         ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_h_align, ufi_v_align
    do
    begin
      ufi_readonly = res_ufi_readonly;
      for select ufi_uuid, ufi_ufr_uuid, ufi_uf_uuid, ufi_kind, ufi_control, ufr_fcode,
                 ufi_hidden, ufi_group_direction, ufi_width, ufi_height,
                 uf_is_param, uf_field_name, us_code, uf_not_null, uf_secured,
                 ufi_sizable, ufi_order_index, ufi_group_index, ufi_fixed_kind, ufi_summary_kind,
                 ufi_font_size, ufi_value_align
          from ux$fritem
               left join ux$field on (ufi_uf_uuid = uf_uuid)
               left join ux$source on (uf_owner = us_uuid)
               left join ux$frame on (ufr_uuid = ufi_ufr_uuid)
          where ufi_uuid = :res_ufi_uuid
          into :ufi_uuid, :ufi_ufr_uuid, :ufi_uf_uuid, :ufi_kind, :ufi_control, ufi_ufr_code,
               :ufi_hidden, :ufi_group_direction, :ufi_width, :ufi_height,
               :uf_is_param, :uf_field_name, :uf_source, :uf_not_null, :uf_secured,
               :ufi_sizable, :ufi_order_index, :ufi_group_index, :ufi_fixed_kind, :ufi_summary_kind,
               :ufi_font_size, :ufi_value_align
      do
      begin
        if (ufi_kind = 1) then
        begin
          select ufi_group_direction,
                 iif(:ufi_width = 0, ufi_width, :ufi_width),
                 iif(:ufi_height = 0, ufi_height, :ufi_height)
          from ux$fritem
          where ufi_uuid = :ufi_ufr_uuid
          into ufi_group_direction, ufi_width, ufi_height;
        end else
        if (ufi_uuid = ufi_ufr_uuid and full_ui_items = 0) then
        begin
          ufi_kind = 1;
          ufi_control = 'frame';
        end

        ufi_uuid_str = uuid_to_char(ufi_uuid);
        ufi_parent_uuid_str = uuid_to_char(ufi_parent_uuid);

        res_caption = ufi_name;
  
        if (res_caption = '') then
        begin
          res_caption = '<Пусто> ';
  
          if (ufi_kind = 1 and ufi_ufr_uuid is not null) then
            res_caption = res_caption || uxf$code(ufi_ufr_uuid, 'name');
          else
          if (ufi_parent_uuid is null) then
            res_caption = res_caption || uxf$code(ux_uuid, 'name');
          else
          if (coalesce(uf_field_name, '') <> '') then
            res_caption = res_caption || uf_field_name;
          else
            res_caption = res_caption || ufi_control || ufi_position;
        end else
        if (coalesce(uf_field_name, '') <> '' and uf_field_name <> res_caption) then
          res_caption = res_caption || '  (' || uf_field_name || ')';

        suspend;
      end

      ufi_value_align = 0;
    end

    if (ux_kind = 'uxreport' and res_kind = '') then
    begin
      ufi_parent_uuid = ux_uuid;
      ufi_parent_uuid_str = uuid_to_char(ufi_parent_uuid);
  
      uf_not_null = 0;
      uf_secured = 0;
      ufi_position = 0;
  
      uf_is_param = 0;
      uf_field_name = '';
      uf_source = '';
      uf_not_null = 0;
      uf_secured = 0;
      ufi_uf_uuid = null;
      ufi_kind = -1;
      ufi_control = 'grid';
      ufi_caption_align = -1; -- Не видно
  
      ufi_group_direction = -1; -- Не группа
      ufi_h_align = 3; -- На всю ширину
      ufi_v_align = 3; -- На всю высоту
      uf_secured = 0;
      ufi_readonly = 1;
      image_index = -1;

      for select ugr_uuid, ufr_uuid, ugr_name, ugr_memo, ugr_inactive
      from ux$frame, ux$grid
      where ufr_owner = :ufm_uuid and
            ufr_uuid = ugr_ufr_uuid and
            ugr_inactive = 0 and
            ugr_default = 1
      rows 1
      into :ufi_uuid, :ufi_ufr_uuid, :ufi_name, ufi_hint, ufi_hidden
      do
      begin
        ufi_uuid_str = uuid_to_char(ufi_uuid);
        ufi_ufr_code = uuid_to_char(ufi_ufr_uuid);
    
        suspend;
      end
    end
  end else
  if (res_kind = 'split') then
  begin
    ufi_position = 0;
    ufi_parent_uuid = null;

    ufi_ufr_uuid = null;
    ufi_uf_uuid = null;
    ufi_ufr_code = '';
    uf_field_name = '';
    ufi_name = 'Параметры';
    uf_not_null = 0;
    uf_source = '';
    ufi_control = 'group';
    ufi_refresh = 0;
    ufp_public = 0;
    ufi_width = 0;
    ufi_height = 0;

    ufi_sizable = 1;
    ufi_order_index = 0;
    ufi_group_index = 0;
    ufi_fixed_kind = 0;
    ufi_summary_kind = 0;

    ufi_uuid_str = ufi_ux_code;
    ufi_uuid = f_md5_uuid(ufi_ux_code);

    ufi_kind = 0;
    ufi_hint = '';
    ufi_hidden = 0;
    ufi_group_direction = -1; -- Не группа
    ufi_h_align = 3; -- На всю ширину
    ufi_v_align = 4; -- По умолчанию
    ufi_caption_align = 0; -- Слева
    ufi_value_align = 0;
    uf_secured = 0;
    ufi_readonly = 0;
    image_index = -1;

    allow_child = 0;
    uf_is_param = 1;
    ufi_parent_uuid_str = ufi_uuid_str;

    for select ufp_ufr_uuid, ufp_uf_uuid, ufr_fcode,
               uf_is_param, uf_field_name, uf_name, ufp_not_null, uf_type,
               us_code, ufp_editor, ufp_refresh, ufp_public
    from ux$frame, ux$frparam, ux$field, ux$source
    where ufr_owner = :ufm_uuid and
          ufr_uuid = ufp_ufr_uuid and
          ufp_public = 1 and
          ufp_uf_uuid = uf_uuid and
          uf_is_param = 1 and
          uf_owner = us_uuid and
          ufp_parent is null
    order by ufr_is_mainframe desc, uf_position
    into ufi_ufr_uuid, ufi_uf_uuid, ufi_ufr_code,
         uf_is_param, uf_field_name, ufi_name, uf_not_null, uf_type,
         uf_source, ufi_control, ufi_refresh, ufp_public
    do
    begin
      if (uf_type = 1 and uf_field_name containing 'company_') then
        Continue;

      ufi_uuid = ufi_uf_uuid;
      ufi_position = ufi_position + 1;
      res_comment = ufi_name;
      res_caption = ufi_name;

      ufi_uuid_str = uuid_to_char(ufi_uuid);

      suspend;
    end
  end else
  if (res_kind = 'uxgrid') then
  begin
    for select ufi_uuid, ufi_parent_uuid, ufi_ux_code, ufi_ufr_uuid, ufi_ufr_code, ufi_uf_uuid, ufi_kind, ufi_control,
               ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_hidden, ufi_readonly, ufi_group_direction,
               ufi_h_align, ufi_v_align, ufi_width, ufi_height, ufi_sizable, ufi_order_index, ufi_group_index,
               ufi_fixed_kind, ufi_summary_kind, image_index, uf_is_param, uf_field_name, uf_source, uf_not_null,
               uf_secured, ufi_parent_uuid_str, ufi_band_index, ufi_header_align, ufi_value_align
        from uxp$uxgrid_items(:ux_uuid, :ufm_uuid, 0)
        into :ufi_uuid, :ufi_parent_uuid, :ufi_ux_code, :ufi_ufr_uuid, :ufi_ufr_code, :ufi_uf_uuid, :ufi_kind, :ufi_control,
             :ufi_position, :ufi_name, :ufi_caption_align, :ufi_hint, :ufi_hidden, :ufi_readonly, :ufi_group_direction,
             :ufi_h_align, :ufi_v_align, :ufi_width, :ufi_height, :ufi_sizable, :ufi_order_index, :ufi_group_index,
             :ufi_fixed_kind, :ufi_summary_kind, :image_index, :uf_is_param, :uf_field_name, :uf_source, :uf_not_null,
             :uf_secured, :ufi_parent_uuid_str, ufi_band_index, ufi_header_align, ufi_value_align
    do
    begin
      ufi_uuid_str = uuid_to_char(ufi_uuid);
      ufi_parent_uuid_str = uuid_to_char(ufi_parent_uuid);

      suspend;
    end
  end else
  if (res_kind = 'unit') then
  begin
    ufi_ux_code = uxf$code(ux_uuid, 'code');
    ufi_ufr_uuid = ux_uuid;
    ufi_ufr_code = uuid_to_char(ufi_ufr_uuid);
    image_index = -1;
    uf_is_param = 0;
    uf_not_null = 0;
    uf_secured = 0;
    ufi_value_align = 0;

    for select uni_uuid, uni_parent_uuid, uni_kind, uni_control,
               uni_position, uni_caption, uni_caption_align, uni_hint, uni_hidden, uni_readonly, uni_group_direction,
               uni_h_align, uni_v_align, uni_width, uni_height, uni_sizable, uni_order_index, uni_group_index,
               uni_fixed_kind, uni_summary_kind, uni_fieldname, uni_datasource,
               uni_uuid_code, uni_parent_uuid_code, uni_code
        from uxp$unititem_tree(:ux_uuid, null, null, 0)
        into :ufi_uuid, :ufi_parent_uuid, :ufi_kind, :ufi_control,
             :ufi_position, :ufi_name, :ufi_caption_align, :ufi_hint, :ufi_hidden, :ufi_readonly, :ufi_group_direction,
             :ufi_h_align, :ufi_v_align, :ufi_width, :ufi_height, :ufi_sizable, :ufi_order_index, :ufi_group_index,
             :ufi_fixed_kind, :ufi_summary_kind, :uf_field_name, :uf_source,
             :ufi_uuid_str, :ufi_parent_uuid_str, uni_code
    do
    begin
      ufi_uf_uuid = null;
      ufi_hint = uni_code;

      if (coalesce(uf_field_name, '') <> '' and coalesce(uf_source, '') <> '') then
        ufi_uf_uuid = f_md5_uuid(uf_source || '.' || uf_field_name);

      suspend;
    end
  end else
    exception e_default f_format('Undefined res_kind: %1', res_kind);
end
