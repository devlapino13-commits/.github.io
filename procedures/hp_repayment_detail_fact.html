CREATE PROCEDURE HP_REPAYMENT_DETAIL_FACT(
    K_ID D_ID,
    REP_DATE D_DATE,
    K_KONTRACT_NO D_STR_CODE = null)
RETURNS (
    OKS_DATE D_DATE,
    OKS_PMT_OS D_CURRENCY,
    OKS_PMT_PERC D_CURRENCY,
    OKS_PMT_ALL D_CURRENCY,
    OKS_DISB_AMOUNT D_CURRENCY,
    OKS_BAL_TO_PERC D_CURRENCY,
    OKS_DAYS D_INTEGER,
    OKS_CALC_PERC D_CURRENCY,
    OKS_BALANCE D_CURRENCY,
    OKS_BAL_PERC D_CURRENCY,
    OKS_PERCENT_NORM D_PERCENT)
AS
declare variable PRV_DATE D_DATE;
declare variable K_STOP_PERC D_DATE;
declare variable OL_DATE D_DATE;
declare variable DB_ID D_INTEGER;
begin
  db_id = f_get_db_id();

  if (k_id is null and
      k_kontract_no <> '') then
  begin
    select k_id
    from kontract
    where k_kontract_no = :k_kontract_no
    into k_id;
  end

  select k_stop_perc
  from kontract
  where k_id = :k_id
  into k_stop_perc;

  ol_date = null;

  if (db_id = 81001) then
  begin
    select ol.ol_date
    from operkroff oko
    inner join operlist ol on ol.ol_id = oko.oko_from_ol_id
    where oko.oko_from_k_id = :k_id and
          ol.is_delete = 0 and
          oko.oko_type = 1
    order by oko.oko_ol_date
    rows 1
    into :ol_date;
  end

  prv_date = null;
  for select oks_date_begin, -sum(oks_disb_amount), sum(oks_balance), sum(oks_bal_to_perc_prv),

             -sum(oks_pmt_os_norm + oks_pmt_os_delinq + oks_pmt_os_adv),
             -sum(oks_pmt_perc_norm + oks_pmt_perc_delinq + oks_pmt_perc_adv), sum(oks_bal_perc)
      from kn_oksummary knoks

      where oks_from_k_id = :k_id and
            (oks_is_sh = 1 or
            (oks_is_pmt = 1 and
            oks_date_begin <= :rep_date)) and
            iif(:ol_date is null, 1 = 1, oks_date_begin < :ol_date)
      group by 1
      order by 1
      into oks_date, oks_disb_amount, oks_balance, oks_bal_to_perc, oks_pmt_os, oks_pmt_perc, oks_bal_perc
  do
  begin
    oks_percent_norm = 0;
    oks_pmt_all = oks_pmt_os + oks_pmt_perc;

    if (prv_date is null) then
      prv_date = oks_date;

    if (k_stop_perc is null or oks_date <= k_stop_perc) then
    begin
      select oks_percent_norm
      from kn_oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin = :prv_date rows 1
      into oks_percent_norm;
    end

    oks_calc_perc = 0;

    select sum(oks_acr_perc)
    from kn_oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin between :prv_date + 1 and :oks_date
    into oks_calc_perc;

    if (oks_calc_perc is null) then
      oks_calc_perc = 0;

    oks_days = oks_date - prv_date;

    suspend;
    prv_date = oks_date;
  end
  if (db_id = 81001) then
  begin
    for select oko.oko_ol_date, -sum(oko.oko_os), -sum(oko.oko_perc)
        from operkroff oko
        inner join operlist ol on ol.ol_id = oko.oko_from_ol_id
        where oko.oko_from_k_id = :k_id and
              oko.oko_type = 4 and
              oko.oko_ol_is_active = 1
        group by 1
        into :oks_date, oks_pmt_os, oks_pmt_perc
    do
    begin

      oks_percent_norm = 0;
      oks_pmt_all = oks_pmt_os + oks_pmt_perc;

      if (k_stop_perc is null or oks_date <= k_stop_perc) then
      begin
        select oks_percent_norm
        from kn_oksummary
        where oks_from_k_id = :k_id and
              oks_date_begin = :prv_date rows 1
        into oks_percent_norm;
      end

      oks_calc_perc = 0;

      select sum(oks_acr_perc)
      from kn_oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin between :prv_date + 1 and :oks_date
      into oks_calc_perc;

      if (oks_calc_perc is null) then
        oks_calc_perc = 0;

      oks_days = oks_date - prv_date;

      oks_balance = oks_balance - oks_pmt_os;
      oks_bal_perc = oks_bal_perc - oks_pmt_perc;

      suspend;
      prv_date = oks_date;
    end
  end
end
