CREATE PROCEDURE UXP$UX_LIST_DETAIL(
    UX_UUID D_UUID,
    UX_UUID_CODE D_STR_100 = null)
RETURNS (
    RES_ID D_STR_255,
    RES_PARENT_ID D_STR_255,
    RES_KIND D_STR_CODE,
    RES_IMAGE D_SMALLINT,
    RES_CAPTION D_STR_100,
    RES_FCODE D_STR_100,
    RES_MEMO D_STR_LONG,
    RES_KIND_CAPTION D_STR_CODE,
    RES_SOURCE_UUID D_UUID,
    RES_DATETIME D_DATETIME,
    RES_WARNING_STATE D_SMALLINT,
    RES_UUID D_UUID,
    RES_UUID_STR D_STR_CODE,
    RES_NAME D_STR_100,
    RES_SCODE D_STR_CODE,
    RES_READONLY D_BOOL,
    RES_SOURCE_CODE D_STR_CODE,
    RES_WEB_HELP D_STR_LONG)
AS
declare variable ux_root d_str_255;
declare variable ux_root_forms d_str_255;
declare variable tmp_root d_str_255;
declare variable tmp_code d_str_100;
declare variable ux_kind d_str_code;
declare variable table_name d_str_code;
declare variable us_is_table d_bool;
declare variable tmp_uuid d_uuid;
declare variable master_fr_uuid d_uuid;
declare variable fr_us_uuid d_uuid;
declare variable ux_us_uuid d_uuid;
declare variable ux_release_uuid d_uuid;
declare variable ux_draft_uuid d_uuid;
declare variable tmp_ux_uuid d_uuid;
declare variable ux_draft_user d_str_code;
declare variable ufm_datetime d_datetime;
declare variable ux_root_commands d_str_255;
declare variable ux_root_reports d_str_255;
declare variable form_count d_integer;
declare variable form_version_root d_str_255;
declare variable form_image d_integer;
declare variable orig_scode d_str_code;
declare variable orig_fcode d_str_100;
declare variable orig_caption d_str_100;
declare variable ufm_updated d_datetime;
declare variable ver_kind d_smallint;
declare variable ufm_user d_str_code;
declare variable is_doc d_bool;
begin
  res_image = 0; -- папка
  res_warning_state = 0;
  res_readonly = 0;
  res_web_help = '';

  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  -- Images
  -- 4 - Свой поисковик
  -- 27 - Чужой поисковик

  -- 17 - Свой фрейм
  -- 18 - Чужой фрейм
  -- 19 - Своя форма
  -- 53 - Чужая форма
  -- 21 - Таблица
  -- 22 - Набор фреймов
  -- 23 - Набор форм
  -- 24 - Неиспользуемые

  -- 25 - Команды
  -- 26 - Бланки


  res_datetime = null;

  if (ux_uuid is null) then
    exit;

  select us_code,
         us_is_table,
         iif(position(us_name, us_code) = 0,
             iif(position(us_code, us_name) = 0, us_code || '  - ' || us_name, us_name),
             us_code)
  from ux$source
  where us_uuid = :ux_uuid
  into table_name, us_is_table, res_caption;

  if (row_count = 0) then
    Exit;

  res_scode = table_name;
  res_name = table_name;
  res_uuid = ux_uuid;
  res_uuid_str = uuid_to_char(res_uuid);

  if (us_is_table = 1) then
  begin
    res_kind_caption = 'Таблица';
    res_kind = 'table';
    res_memo = 'Формы и интерфейсы таблицы';
    res_image = 21;
  end
  else
  begin
    res_kind_caption = 'Процедура';
    res_kind = 'procedure';
    res_memo = 'Процедуры используемые в интерфейсах';
    res_image = 75;
  end

  res_source_uuid = ux_uuid;
  res_fcode = table_name;
  res_id = table_name;
  ux_root = res_id;
  res_parent_id = null;

  if (res_caption is null) then
    res_caption = table_name;

  res_source_code = uxf$code(res_source_uuid, 'code');
  suspend;

  res_uuid = null;
  res_uuid_str = '';

  res_kind_caption = '';
  res_memo = '';
  res_source_uuid = null;
  res_source_code = '';
  res_datetime = null;

  tmp_code = res_fcode;

  res_id = tmp_code || '.forms';
  res_fcode = '';
  res_scode = '';
  res_name = '';
  ux_root_forms = res_id;
  res_parent_id = ux_root;
  res_kind = 'table.forms';
  res_caption = 'Формы';
  res_memo = 'Формы принадлежащие к таблице и используемые в формах';
  res_image = 23;
  suspend;

  res_id = tmp_code || '.commands';
  res_fcode = '';
  res_scode = '';
  res_name = '';
  ux_root_commands = res_id;
  res_parent_id = ux_root;
  res_kind = 'table.commands';
  res_caption = 'Команды';
  res_memo = 'Команды принадлежащие к таблице';
  res_image = 100;
  suspend;

  res_id = tmp_code || '.reports';
  res_fcode = '';
  res_scode = '';
  res_name = '';
  ux_root_reports = res_id;
  res_parent_id = ux_root;
  res_kind = 'table.reports';
  res_caption = 'Отчеты и бланки';
  res_memo = 'Отчет и бланки системы';
  res_image = 37;
  suspend;

  for select distinct ux_uuid, ux_us_uuid, ux_scode, ux_fcode, ux_name, uu_kind, uu_datetime,
             ux_release_uuid, ux_draft_uuid, ux_draft_user
  from ux$code, ux$uniq
  where ux_us_uuid = :ux_uuid and
        ux_uuid = uu_id
  union
  select ux_uuid, ux_us_uuid, ux_scode, ux_fcode, ux_name, uu_kind, uu_datetime,
             ux_release_uuid, ux_draft_uuid, ux_draft_user
  from ux$frame, ux$form, ux$code, ux$uniq
  where (ufr_us_uuid = :ux_uuid or ufr_add_table_fields = :table_name) and
        ufr_owner = ufm_uuid and
        ufm_ux_uuid = ux_uuid and
        (ux_us_uuid <> ufr_us_uuid or ufr_add_table_fields <> '') and
        ux_uuid = uu_id
  order by 4, 2
  into tmp_ux_uuid, ux_us_uuid, res_scode, res_fcode, res_caption, ux_kind, res_datetime,
       ux_release_uuid, ux_draft_uuid, ux_draft_user
  do
  begin
    res_memo = '';

    select count(*)
    from ux$form
    where ufm_ux_uuid = :tmp_ux_uuid
    into form_count;

    res_uuid = null;
    ux_kind = uxf$kind(ux_kind, 'code');

    res_name = res_caption;
    orig_scode = res_scode;
    orig_fcode = res_fcode;
    orig_caption = res_caption;

    form_image = case ux_kind
      when 'uxcommand' then 35
      when 'uxreport' then 26
      else 19
    end;

    if (exists(select *
                from ux$alias, ux$search
                where uxa_owner = :tmp_ux_uuid and
                      uxa_uuid = usr_uuid)) then
      form_image = 4;

    form_version_root = null;
    if (form_count > 1) then
    begin
      res_id = uuid_to_char(tmp_ux_uuid) || '.uxcode';
      form_version_root = res_id;
      res_parent_id = ux_root_forms;
      res_kind = 'uxcode';
      res_kind_caption = 'Версии';
      res_memo = 'Версии формы и отчетов';
      res_image = form_image;
      res_fcode = orig_fcode || '   от ' || f_datetimetostr(res_datetime);

      if (ux_kind = 'uxcommand') then
        res_parent_id = ux_root_commands;
      else
      if (ux_kind = 'uxreport') then
        res_parent_id = ux_root_reports;

      res_uuid = null;
      res_uuid_str = '';

      suspend;
    end

    for select ufm_uuid, uu_datetime, ufm_updated, ufm_user,
               iif(ufm_uuid = :ux_draft_uuid and ufm_uuid is distinct from :ux_release_uuid, 0,
                   iif(ufm_uuid = :ux_release_uuid, 1, 2)),
               ufm_web_help
    from ux$form, ux$uniq
    where ufm_ux_uuid = :tmp_ux_uuid and
          ufm_uuid = uu_id
    order by 4, uu_datetime desc
    into res_uuid, ufm_datetime, ufm_updated, ufm_user, ver_kind, res_web_help
    do
    begin
      res_warning_state = 0;
      res_readonly = 0;
      res_kind = ux_kind;

      is_doc = uxf$is_doc(res_uuid);

      res_caption = orig_caption;
      res_scode = orig_scode;
      res_fcode = orig_fcode;

      res_parent_id = ux_root_forms;
      res_image = form_image;

      if (is_doc = 1) then
        res_image = 148; -- Зеленая книга

      if (ux_kind = 'uxcommand') then
        res_parent_id = ux_root_commands;
      else
      if (ux_kind = 'uxreport') then
        res_parent_id = ux_root_reports;

      if (ver_kind = 2) then
      begin
        res_parent_id = coalesce(form_version_root, res_parent_id);
        res_id = res_fcode || ', ver.' || f_datetimetostr(ufm_datetime);
        res_memo = 'Версия объекта, изменен: ' || f_datetimetostr(ufm_updated);

        res_readonly = 1;
        res_warning_state = 4; -- Версия
        res_image = 91;

        if (form_count > 1) then
          res_caption = 'Версия от ' || f_datetimetostr(ufm_datetime);
      end else
      begin
        if (ver_kind = 0) then
        begin
          res_parent_id = coalesce(form_version_root, res_parent_id);
          res_id = res_fcode || ', ' || f_datetimetostr(ufm_datetime);

          res_readonly = 0;
          res_warning_state = 1; -- Свой драфт
          res_memo = 'Драфт версия, изменен: ' || f_datetimetostr(ufm_updated);

          if (ufm_user <> current_user and ux_draft_user <> current_user) then
          begin
            res_memo = res_memo || ', пользователь: ' || ufm_user;
--            res_readonly = 1;
            res_warning_state = 2; -- Чужой драфт
          end

          if (form_count > 1) then
            res_caption = 'Драфт,  ' || f_datetimetostr(ufm_updated);
        end else
        begin
          res_parent_id = coalesce(form_version_root, res_parent_id);
          res_id = res_fcode || ' rel.' || f_datetimetostr(ufm_datetime);

          res_memo = 'Релиз версия, изменен: ' || f_datetimetostr(ufm_updated);
          res_readonly = 1;
          res_warning_state = 3; -- Релиз
          res_image = 92;

          if (is_doc = 1) then
            res_image = 149; -- Зеленая книга со знаком $

          if (form_count > 1) then
            res_caption = 'Релиз,   ' || f_datetimetostr(ufm_updated);
        end
      end

      if (form_count = 1) then
      begin
        res_image = form_image;

        if (is_doc = 1) then
        begin
          if (ver_kind <> 0) then
            res_image = 149; -- Зеленая книга со знаком $
          else
            res_image = 148; -- Зеленая книга
        end
      end

      res_uuid_str = uuid_to_char(res_uuid);
      res_kind_caption = uxf$kind(ux_kind, 'caption');

      if (is_doc = 1) then
        res_kind_caption = 'Документ(ы)';

      -- Фреймы и формы таблицы
      res_source_uuid = ux_uuid;
      res_source_code = uxf$code(res_source_uuid, 'code');

      if (form_count > 1) then
        res_fcode = orig_fcode || '   от ' || f_datetimetostr(ufm_datetime);

      if (coalesce(res_web_help, '') <> '') then
      begin
        if (trim(coalesce(res_memo, '')) = '') then
          res_memo = res_web_help;
        else
          res_memo = res_memo || ascii_char(13) || res_web_help;
      end

      suspend;

      res_web_help = '';

      if (ux_kind <> 'uxreport') then
      begin
        res_uuid_str = '';
        res_parent_id = res_id;

        tmp_root = res_id;
        tmp_code = res_fcode;
        tmp_uuid = res_uuid;
    
        res_memo = 'Фрейм используемый в форме';
    
        res_kind = 'uxframe';
        res_kind_caption = uxf$kind(res_kind, 'caption');
        res_datetime = null;
  
        for select ufr_uuid,
                   ufr_us_uuid,
                   ufr_code,
                   ufr_fcode,
                   ufr_name,
                   ufr_master_ufr_uuid,
                   uu_datetime
        from ux$frame, ux$uniq
        where ufr_owner = :tmp_uuid and
              ufr_uuid = uu_id
        into res_uuid, fr_us_uuid, res_scode,  res_fcode, res_caption, master_fr_uuid, res_datetime
        do
        begin
          res_name = res_caption;
          res_uuid_str = uuid_to_char(res_uuid);
          res_id = tmp_code || '#' || res_fcode;
    
          if (fr_us_uuid = ux_us_uuid) then
            res_image = 17;
          else
            res_image = 18;

          if (master_fr_uuid is null) then
            res_parent_id = tmp_root;
          else
            res_parent_id = tmp_code || '#' || uxf$code(master_fr_uuid, 'fcode');
    
          res_source_uuid = fr_us_uuid;
          res_source_code = uxf$code(res_source_uuid, 'code');
          suspend;
        end
      end
  
      res_uuid = null;
      res_source_uuid = ux_uuid;
      res_source_code = uxf$code(res_source_uuid, 'code');
      res_kind_caption = '';
      res_memo = '';
      res_datetime = null;
      res_scode = '';
      res_warning_state = 0;
      res_readonly = 0;
    end
  end
end
