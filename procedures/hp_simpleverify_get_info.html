CREATE PROCEDURE HP_SIMPLEVERIFY_GET_INFO(
    ACL_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT,
    RES_IMAGE_LIST D_STR_8100,
    RES_ACTIONS D_STR_LONG,
    RES_MEMO_TEMPLATES D_STR_8100,
    IS_READONLY D_SMALLINT)
AS
declare variable tmp_html d_str_8100;
declare variable tmp_str d_str_long;
declare variable obj_kind d_integer;
declare variable obj_code d_str_code;
declare variable obj_id d_id;
declare variable tmp_image_list d_str_8100;
declare variable blob_info d_str_long;
declare variable bls_id d_id;
declare variable c_id d_id;
declare variable tmp_un_id d_id;
declare variable tmp_date d_date;
declare variable lo_un_id d_id;
declare variable lo_c_id d_id;
declare variable tmp_datetime d_datetime;
declare variable rd_id d_integer;
declare variable rd_code d_str_255;
begin
  -- Процедура возвращает параметры для показа окна SimpleVerifyForm
  -- SimpleVerifyForm - стандартное окно верификации информации

  res_html = '';           -- HTML для показа информации
  res_image_list = '';     -- Список изображений для показа
  res_memo_templates = ''; -- Массив шаблонов для коментариев
  is_readonly = 0;         -- Окно в режиме просмотра или нет
  res_actions = '';    -- Доступные действия в окне
  -- Доступные действия:
  --   show_memo - Показать вкладку коментарии или скрыть
  --   approve   - Подтверждение
  --   back      - Вернуть в общую очередь
  --   reject    - Отказ
  --   revision  - Доработка

  -- Значения по умолчанию.
  res_actions = '{"show_memo":1,"approve":1,"back":1,"reject":1,"revision":0}';

  select acl_object_kind, acl_object_id, iif(acl_status in (0, 1), 0, 1), acl_from_rd_id
  from actionlist
  where acl_id = :acl_id
  into obj_kind, obj_id, is_readonly, rd_id;

  select rd.rd_code
  from refdata rd
  where rd.rd_id = :rd_id
  into :rd_code;

  if (obj_id is null) then
    exception e_default 'Acl_ID not found';

  obj_code = f_objkind(obj_kind, 'name');

  if (obj_code = 'qr_photo') then
  begin
    res_memo_templates = '{"memo":["Плохое качество фото", "Клиент отсутствует в фото"]}';

    select qrl_from_c_id, qrf_from_un_id, qrf_from_bls_id, c_name_fio,
      c_birth_date, c_from_un_id, qrl_datetime
    from qr_photo, qr_list, clients
    where qrf_from_qrl_id = :obj_id and
          qrf_from_qrl_id = qrl_id and
          qrl_from_c_id = c_id
    into c_id, tmp_un_id, bls_id, tmp_str, tmp_date, lo_un_id, tmp_datetime;

    tmp_html =
      f_format('Клиент <b>%1</b>', tmp_str) || '<br>' ||
      f_format('Дата рождения <b>%1</b>', f_datetostr(tmp_date)) || '<br><br>';

    select c_id, c_name_fio
    from username, clients
    where un_id = :lo_un_id and
          un_from_c_id = c_id
    into lo_c_id, tmp_str;

    tmp_html = tmp_html ||
      f_format('Пользователь, создавший клиента: <b>%1</b>', tmp_str) || '<br>';

    tmp_str = 'Не определен';
    select reg_name
    from hr_pstatus, region
    where current_date between ps_date_begin and ps_date_end and
          ps_status = 1 and
          ps_is_working = 1 and
          ps_from_c_id = :lo_c_id
          and ps_from_reg_id = reg_unicode
    into tmp_str;
    tmp_html = tmp_html ||
      f_format('Офис пользователя: <b>%1</b>', tmp_str) || '<br><br>';

    select c_name_fio
    from username, clients
    where un_id = :tmp_un_id and
          un_from_c_id = c_id
    into tmp_str;
    tmp_html = tmp_html ||
      f_format('Пользователь, сделавший фото: <b>%1</b>', tmp_str) || '<br>' ||
      f_format('Дата и время загрузки фото: <b>%1</b>', f_datetimetostr(tmp_datetime)) || '<br>';

    res_html = tmp_html;

    if (bls_id is not null) then
    begin
      blob_info = f_image_item('blobstore', bls_id, 'Фото', un_id, 1);
      res_image_list = f_format('{"caption": "Клиент и специалист", "blobs":[%1]}', blob_info);
    end

    select res_image_list
    from hp_get_client_doc_images(:c_id, 1, 1, '', :un_id)
    into :tmp_image_list;
  
    if (tmp_image_list <> '') then
    begin
      if (res_image_list = '')
      then res_image_list = tmp_image_list;
      else
      begin
        res_image_list = res_image_list || ',' || tmp_image_list;
      end
    end
  end

  if (rd_code = 'ActionGroup.SimpleVerify_Phone_Number_Verify' and obj_kind = 0) then
    select tmp_html, res_image_list
    from hp_html_verify_phone(:obj_id, :un_id)
    into :res_html, :res_image_list;


  res_image_list = '[' || res_image_list || ']';
  res_html = blob_append((select res_style from fm_html_styles), '<body>', res_html, '</body>');

  suspend;
end
