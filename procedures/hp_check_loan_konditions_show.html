CREATE PROCEDURE HP_CHECK_LOAN_KONDITIONS_SHOW(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    CHECK_KIND D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    CCR_ID D_INTEGER,
    CCR_FROM_CDM_ID D_INTEGER,
    CDM_MEMO D_STR_255,
    CCR_INFO D_STR_255,
    CCR_MESSAGE D_STR_LONG,
    CCR_IS_CRITICAL D_SMALLINT,
    CCR_DATETIME D_STR_CODE,
    RES_CHECK_STATE D_SMALLINT)
AS
declare variable ccs_id d_integer;
declare variable check_date d_date;
begin
  if (trim(coalesce(obj_kind, '')) = '') then
    exception e_default 'Empty obj_kind';

  check_kind = f_enum('condition_check_kind', check_kind, 'id');

  obj_kind = f_objkind(obj_kind, 'id');

  select ccs_id
  from condcheckstate
  where ccs_object_kind = :obj_kind and
        ccs_object_id = :obj_id and
        ccs_check_kind = :check_kind
  into ccs_id;

  if (ccs_id is null) then
    exit;

  select blank_date
  from hp_edoc_object_info(:obj_kind, :obj_id, :un_id)
  into :check_date;
  if (check_date is null) then
    check_date = current_date;

  for select ccr_id, ccr_from_cdm_id, cdm_memo, ccr_info, ccr_message, ccr_is_critical, f_datetimetostr(ccr_datetime)
  from condcheckresult, conditionmain
  where ccr_from_ccs_id = :ccs_id and
        ccr_from_cdm_id = cdm_id
  into :ccr_id, :ccr_from_cdm_id, :cdm_memo, :ccr_info, :ccr_message, :ccr_is_critical, :ccr_datetime
  do
  begin
    res_check_state = 0;

    if (ccr_is_critical > 0) then
    begin
      if (exists(select * from condcheckignore
      where cci_object_kind = :obj_kind and
            cci_object_id = :obj_id and
            cci_from_cdm_id = :ccr_from_cdm_id and
            cci_date_end >= :check_date and
            cci_state = 1)) then
      begin
        ccr_is_critical = 0;
        res_check_state = -1;
      end
      else
        res_check_state = -2;
    end
    else
      res_check_state = -1;

    suspend;
  end
end
