CREATE PROCEDURE CM_KZ_0030_SQL(
    MA_ID D_INTEGER NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
begin
  res_error_is_critical = 0;
  res_info_text = 'Проверка ПКБ';
  res_error_text = '';

  -- RFCKZ-634
  -- 1. Проверка pkb_available_reports
  if (not exists(
        select 1
        from extinfolist eil
        join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
        where eic.eic_code = 'pkb_available_reports'
          and eil.eil_object_kind = f_objkind('miniapp', 'id')
          and eil.eil_object_id = :ma_id
          and eil.eil_status = 2
          and eil.eil_from_bls_id is not null
      )) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Не получены доступные отчеты ПКБ';
    suspend;
  end

  -- 2. Проверка pkb_cib_report, если есть приоритетный отчет
  if (coalesce((select priority_id from hp_get_priority_rep_id(f_objkind('miniapp', 'id'), :ma_id)), -1) <> -1
      and not exists(
        select 1
        from extinfolist eil
        join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
        where eic.eic_code = 'pkb_cib_report'
          and eil.eil_object_kind = f_objkind('miniapp', 'id')
          and eil.eil_object_id = :ma_id
          and eil.eil_status = 2
          and eil.eil_from_bls_id is not null
      )) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Не получен кредитный отчет ПКБ';
    suspend;
  end
end
