CREATE PROCEDURE HP_FUELGENREG_VIEW(
    REG_ID D_INTEGER,
    DATE_BEG D_DATE,
    DATE_END D_DATE,
    ID D_INTEGER = null)
RETURNS (
    FG_ID D_INTEGER,
    FG_DTWRITEOFF D_DATE,
    FG_DTBEG D_DATE,
    FG_DTEND D_DATE,
    FG_OFFICE_ID D_INTEGER,
    FG_PET_ID D_INTEGER,
    FG_START D_CURRENCY,
    FG_START_SUM D_CURRENCY,
    FG_FILLUP D_CURRENCY,
    FG_FILLUP_SUM D_CURRENCY,
    FG_WHOURS D_CURRENCY,
    FG_WRATE D_CURRENCY,
    FG_USEDUP D_CURRENCY,
    FG_USEDUP_SUM D_CURRENCY,
    FG_END D_CURRENCY,
    FG_END_SUM D_CURRENCY,
    FG_DRIVER_ID D_ID,
    FG_MANAGER_ID D_ID,
    FG_BOOKER_ID D_ID,
    FG_DRIVER_DTSIGN D_DATETIME,
    FG_MANAGER_DTSIGN D_DATETIME,
    FG_BOOKER_DTSIGN D_DATETIME,
    FG_DRIVER_EDL_ID D_ID,
    FG_MANAGER_EDL_ID D_ID,
    FG_BLS_ID D_ID,
    FG_STATUS D_INTEGER,
    PET_NAME D_STR_255,
    PET_LABEL D_STR_CODE,
    PET_IS_RENT D_INTEGER,
    PET_NORMA_KM D_INTEGER,
    REG_NAME D_STR_CODE,
    DRIVER_FIO D_STR_100,
    MANAGER_FIO D_STR_100,
    BOOKER_FIO D_STR_100,
    IS_CAR D_SMALLINT,
    STATE D_STR_20)
AS
begin
  if (id > 0) then
  begin
    reg_id = 1000000;
    date_beg = '01.01.1900';
    date_end = '31.12.9999';
  end

  for
    select
      fg_id,
      fg_dtwriteoff,
      fg_dtbeg,
      fg_dtend,
      fg_office_id,
      fg_pet_id,
      fg_start,
      fg_start_sum,
      fg_fillup,
      fg_fillup_sum,
      fg_whours,
      fg_wrate,
      fg_usedup,
      fg_usedup_sum,
      fg_end,
      fg_end_sum,
      fg_driver_id,
      fg_manager_id,
      fg_booker_id,
      fg_driver_dtsign,
      fg_manager_dtsign,
      fg_booker_dtsign,
      fg_driver_edl_id,
      fg_manager_edl_id,
      fg_bls_id,
      fg_status,
      pet_name,
      pet_label,
      pet_is_rent,
      pet_norma_km,
      reg_name,
      cs.c_name_fio, 
      mn.c_name_fio,
      bk.c_name_fio,
      iif(pet_norma_yediz = 1, 0, 1),
      case fg_status
        when -1 then 'аннулирован'
        when  0 then 'создан'
        when  1 then 'подписан'
      end
    from fuelgenreg
      inner join region_list(:reg_id, 1, 3, 1)
        on id_list = fg_office_id
      inner join region
        on reg_unicode = fg_office_id
      inner join pettrans
        on pet_id = fg_pet_id
      inner join clients as cs
        on cs.c_id = fg_driver_id
      inner join clients as mn
        on mn.c_id = fg_manager_id
      left join clients as bk
        on bk.c_id = fg_booker_id
    where (:id > 0 and fg_id = :id) or
      (:id is null and fg_dtwriteoff between :date_beg and :date_end)
  into
    :fg_id,
    :fg_dtwriteoff,
    :fg_dtbeg,
    :fg_dtend,
    :fg_office_id,
    :fg_pet_id,
    :fg_start,
    :fg_start_sum,
    :fg_fillup,
    :fg_fillup_sum,
    :fg_whours,
    :fg_wrate,
    :fg_usedup,
    :fg_usedup_sum,
    :fg_end,
    :fg_end_sum,
    :fg_driver_id,
    :fg_manager_id,
    :fg_booker_id,
    :fg_driver_dtsign,
    :fg_manager_dtsign,
    :fg_booker_dtsign,
    :fg_driver_edl_id,
    :fg_manager_edl_id,
    :fg_bls_id,
    :fg_status,
    :pet_name,
    :pet_label,
    :pet_is_rent,
    :pet_norma_km,
    :reg_name,
    :driver_fio,
    :manager_fio,
    :booker_fio,
    :is_car,
    :state
  do
  begin
    suspend;
  end
end
