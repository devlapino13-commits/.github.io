CREATE PROCEDURE AAREP_CASH_DISB_CHECK(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    END_DISB_SALDO DECIMAL(18,4),
    END_OTHER_SALDO DECIMAL(18,4),
    END_TOTAL_SALDO DECIMAL(18,4),
    BEGIN_DISB_SALDO DECIMAL(18,4),
    BEGIN_OTHER_SALDO DECIMAL(18,4),
    BEGIN_TOTAL_SALDO DECIMAL(18,4),
    IN_FROM_BANK DECIMAL(18,4),
    IN_FROM_REPAY DECIMAL(18,4),
    OUT_TO_BANK DECIMAL(18,4),
    OUT_TO_BANK_FROM_DISB DECIMAL(18,4),
    OUT_TO_DISB DECIMAL(18,4),
    START_DATE DATE,
    START_SALDO DECIMAL(18,4),
    TOTAL_TO_BANK DECIMAL(18,4),
    TOTAL_REPAY DECIMAL(18,4),
    TOTAL_FROM_BANK DECIMAL(18,4),
    TOTAL_DISB DECIMAL(18,4),
    REP_REGION_NAME VARCHAR(100))
AS
declare variable cash_str varchar(20);
declare variable ol_date date;
declare variable needed_minus_from decimal(18,4) = 0;
declare variable total_needed_minus decimal(18,4) = 0;
declare variable database_id integer;
declare variable total_saldo2 decimal(18,4) = 0;
declare variable total_out_to_bank_from_disb decimal(18,4) = 0;
declare variable tmp_amount decimal(18,4);
declare variable al_id varchar(20);
declare variable total_disb2 decimal(18,4);
declare variable total_from_bank2 decimal(18,4);
declare variable total_repay2 decimal(18,4);
declare variable total_to_bank2 decimal(18,4);
begin
--  start_date='01.06.2014';
  start_date = '17.09.2014';
  start_saldo = 0;

  select reg_code || ' ' || reg_name
  from region
  where reg_unicode = :rep_region_id
  into :rep_region_name;

  begin_disb_saldo = 0;
  begin_other_saldo = 0;
  begin_total_saldo = 0;

  end_disb_saldo = 0;
  end_other_saldo = 0;
  end_total_saldo = 0;

  in_from_bank = 0;
  in_from_repay = 0;

  out_to_bank = 0;
  out_to_disb = 0;

  :database_id = f_get_db_id();

  if (database_id in (39801, 41701)) then
  begin
    cash_str = '10001'; -- кр
  end

  if (database_id = 81001) then
  begin
    cash_str = '50-01'; -- рф
  end

  for select al_id
      from acc_list
      where al_from_bg_id = :cash_str and
            al_from_reg_id = :rep_region_id
      into :al_id
  do
  begin
    tmp_amount = f_al_bal(al_id, start_date - 1);
    start_saldo = start_saldo + tmp_amount; -- сальдо на начало запуска нового учета
    tmp_amount = f_al_bal(al_id, rep_date_end - 1);
    begin_total_saldo = begin_total_saldo + tmp_amount; -- сальдо на начало периода
    tmp_amount = f_al_bal(al_id, rep_date_end);
    end_total_saldo = end_total_saldo + tmp_amount; -- сальдо на конец периода
  end

  total_to_bank = 0;
  total_repay = 0;
  total_from_bank = 0;
  total_disb = 0;

  end_other_saldo = start_saldo;
  begin_other_saldo = start_saldo;

  for select ol_date, sum(in_from_bank), sum(in_from_repay - in_from_bank), sum(out_to_bank),
             sum(out_to_disb - out_to_bank)
      from aarep_cash_disb_check_tmp(:rep_region_id, :rep_date_end)
      group by ol_date
      into :ol_date, :in_from_bank, :in_from_repay, :out_to_bank, :out_to_disb
  do
  begin
    out_to_bank_from_disb = 0;
    if (ol_date < rep_date_end) then
    begin
      begin_disb_saldo = begin_disb_saldo + in_from_bank - out_to_disb;
      begin_other_saldo = begin_other_saldo + in_from_repay - out_to_bank;
    end
    end_disb_saldo = end_disb_saldo + in_from_bank - out_to_disb;
    end_other_saldo = end_other_saldo + in_from_repay - out_to_bank;

    if (end_other_saldo < 0) then
    begin
      end_disb_saldo = end_disb_saldo + end_other_saldo;
      out_to_bank_from_disb = -end_other_saldo;
      end_other_saldo = 0;
    end
    if (begin_other_saldo < 0) then
    begin
      begin_disb_saldo = begin_disb_saldo + begin_other_saldo;
      begin_other_saldo = 0;
    end
  end
  if (rep_region_id = 1010100) then
    end_disb_saldo = 1000000;
  suspend;
end
