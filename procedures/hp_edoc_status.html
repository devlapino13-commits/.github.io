CREATE PROCEDURE HP_EDOC_STATUS(
    EDOC_KIND D_STR_CODE,
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    ONE_C_ID D_ID = null)
RETURNS (
    EDL_ID D_ID,
    EDOC_STATUS D_SMALLINT,
    EDOC_STATUS_STR D_STR_100,
    EDOC_NAME D_STR_100,
    EDOC_BLANK_DATE D_DATE,
    EDOC_CREATE_DATETIME D_DATETIME,
    EDS_ID D_ID,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    CONT_VALUE D_STR_100,
    CONT_IS_VERIFIED D_SMALLINT,
    CLIENT_SIGN_STATUS D_SMALLINT,
    CLIENT_SIGN_STATUS_STR D_STR_CODE,
    CLIENT_SIGN_DATETIME D_DATETIME,
    CLIENT_STATUS_DATETIME D_DATETIME,
    CLIENT_STATUS_UN_ID D_ID,
    MESSAGE_SEND_DATETIME D_DATETIME,
    MESSAGE_SEND_UN_ID D_ID,
    MESSAGE_SEND_COUNT D_INTEGER,
    CLIENT_SIGN_KIND D_INTEGER)
AS
declare variable obj_kind_str d_str_code;
declare variable edc_id d_integer;
declare variable edc_obj_kind d_str_code;
declare variable edl_blank_status d_smallint;
declare variable edm_status d_smallint;
declare variable msg_send_status d_smallint;
declare variable message_status d_str_100; /* Статус отправки сообщения */
begin
  obj_kind_str = f_objkind(obj_kind, 'name');
  obj_kind = f_objkind(obj_kind, 'id');

  select edc_id, f_objkind(edc_object_kind, 'name'), edc_name
  from edoc_cfg
  where edc_code = :edoc_kind
  into :edc_id, :edc_obj_kind, edoc_name;

  if (edc_id is null) then
    exit;

  if (obj_kind_str <> edc_obj_kind) then
    exception e_default 'Incorrect obj_kind. Expected: ' || edc_obj_kind || ', actual: ' || obj_kind_str;

  if (coalesce(one_c_id, 0) = 0) then
    one_c_id = 0;

  edl_id = null;
  edoc_status = -2; --1; igusev 16.05.2021 внес правку, т.к. если эл. документ не был создан, процедура возвращала статус = 1, который соотвествует статусу = подписан.
  cont_is_verified = 0;
  edl_blank_status = 1;

  message_send_count = 1;
  select edl_id, edl_status, edl_blank_status, edl_blank_date, edl_create_datetime, edl_create_un_id
  from edoc_list
  where edl_object_kind = :obj_kind and
        edl_object_id = :obj_id and
        edl_from_edc_id = :edc_id
  order by edl_id desc
  rows 1
  into :edl_id, :edoc_status, edl_blank_status, edoc_blank_date, edoc_create_datetime, message_send_un_id;
  client_status_un_id = message_send_un_id;

  edoc_status_str = '';
  edoc_status_str = case edoc_status
    when -2 then 'Не сформирован'
    when -1 then 'Аннулирован'
    when  0 then 'Не подписан'
    when  1 then 'Подписан'
  end;
  if (edoc_status = 0) then
  begin
    if (edl_blank_status = -1) then
      edoc_status_str = 'Ошибка приготовления бланка';

    if (edl_blank_status = 0) then
      edoc_status_str = 'Бланк не готов';
  end
  edoc_status_str = trim(edoc_status_str);

  for select c_id
  from hp_edoc_sign_clients(:edoc_kind, :obj_kind, :obj_id, '')
  where (:one_c_id = 0 or c_id = :one_c_id)
  into :c_id
  do
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;

    eds_id = null;
    client_sign_status = -2;
    client_status_datetime = null;
    client_sign_datetime = null;
    cont_value = '';
    client_sign_status_str = 'Код не выслан';
    message_send_datetime = null;
    message_status = '---';
    message_send_un_id = null;
    client_sign_kind = null;

    if (edl_id is not null) then
    begin
      select eds_id, eds_status, eds_status_datetime, eds_enum_edoc_sign_kind
      from edoc_sign
      where eds_from_edl_id = :edl_id and
            eds_from_c_id = :c_id
      order by eds_id desc
      rows 1
      into eds_id, client_sign_status, client_status_datetime, client_sign_kind;

      if (client_sign_status = 1) then
        client_sign_datetime = client_status_datetime;

      client_sign_status_str = case client_sign_status
        when  0 then 'Не подписан'
        when  1 then 'Подписан'
      end;

      edm_status = -2;
      select f_cont_format(f_cont_code(edm_from_clt_id), 'cell', 'secured'),
             edm_status, msg_send_status, msg_datetime, cda_status
      from edoc_msg, fsrobot_msgbox, contdata
      where edm_from_eds_id = :eds_id and
            edm_from_msg_id = msg_id and
            edm_from_cda_id = cda_id
      order by edm_from_eds_id desc, edm_status desc, edm_from_msg_id desc
      rows 1
      into cont_value, edm_status, msg_send_status, message_send_datetime, cont_is_verified;

      select sds_name
      from sms_delivery_status
      where sds_id = :msg_send_status
      into message_status;

      if (client_sign_status = 0) then
      begin
        if (edm_status = -2) then
          client_sign_status_str = 'Код не выслан';
        else
          if (edm_status = 0 and msg_send_status not in (-1, 0, 1)) then
            client_sign_status_str = 'Сообщение' || ' ' || message_status;
      end

      client_sign_status_str = trim(client_sign_status_str);
    end

    suspend;
  end
end
