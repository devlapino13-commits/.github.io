CREATE PROCEDURE CB_RU_DRR_202411(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    LO_NAME_FIO D_STR_100,
    LO_STAGE_DATE D_DATE,
    STAGE_MONTH D_INTEGER,
    LO_BALANCE_ALL D_CURRENCY,
    LO_BALANCE_OWN D_CURRENCY,
    OWN_BALANCE_PERC D_CURRENCY,
    RES D_SMALLINT,
    LO_BONUS D_CURRENCY,
    BONUS_DRR D_INTEGER,
    RES_MEMO D_STR_255,
    LO_BONUS_TOTAL_FACT D_CURRENCY,
    LO_BONUS_TOTAL D_CURRENCY)
AS
declare variable rep_date_begin d_date;
declare variable lo_id id_type;
declare variable reg_id d_integer;
declare variable sfb_fix_bonus d_currency;
declare variable sfb_total_work d_integer;
declare variable sfb_fact_work decimal(18,2);
declare variable sfb_fact_work_eff d_currency;
declare variable rep_period_end d_integer;
declare variable total_days d_smallint;
declare variable dd d_date;
declare variable lp_amount d_currency;
declare variable aarep_bonus_minus_disb d_integer;
declare variable kol_disb_kont curr_type;
declare variable lo_stage_month_full curr_type;
declare variable jk_system_code d_str_code;
declare variable disb_norma d_currency;
declare variable k d_smallint;
declare variable i d_smallint;
declare variable re_date_begin d_date;
declare variable re_date_end d_date;
declare variable re_res d_smallint;
begin
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;

  for
  select ps_from_c_id,
         ps_from_reg_id,
         jk_system_code,
         c_name_fio
  from hr_pstatus, clients, joblist, jobkind, region_list(:rep_region_id, 1, -1, 1)
  where ps_from_c_id = c_id and
        ps_date_begin <= :rep_date_end and
        ps_date_end >= :rep_date_end and
        ps_status = 1 and
        ps_is_working = 1 and
        ps_from_c_id > 1000000000003 and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id and
        jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
        ps_from_reg_id = id_list --and c_id = 1010101139724
  into lo_id, reg_id, jk_system_code, lo_name_fio
  do
  begin
    lo_stage_date = null;
    select lo_stage_date from hp_get_lo_stage_alter(:lo_id, :rep_date_end)
    into lo_stage_date;

    k = 0;
    if (extract(day from :lo_stage_date) = 1) then
    begin
      lo_stage_date = lo_stage_date - 1; -- если первое число то на конец периода должен быть один месяц
      k = 1;
    end 

    stage_month = datediff(month, cast(:lo_stage_date as date), cast(:rep_date_end as date));
    if (k = 1) then lo_stage_date = lo_stage_date + 1;

    if (lo_stage_date >= '01.11.2021' and
        stage_month between 7 and 11) then
    begin
      select lo_balance_all, lo_balance_own, own_balance_perc, lo_bonus, res, lo_bonus_total_fact, lo_bonus_total
      from cb_ru_drr_temp_202411(:stage_month, :lo_id, :reg_id, :rep_date_begin, :rep_date_end)
      into :lo_balance_all, :lo_balance_own, :own_balance_perc, :lo_bonus, :res, lo_bonus_total_fact, lo_bonus_total;

      res_memo = '';
      if (res = 1 and stage_month = 11 and dateadd(month, -11, rep_date_end) >= lo_stage_date) then
      begin
        res = 0;
        res_memo = 'стаж более 11 мес.';
      end 

      re_res = 0;
      bonus_drr = 0;

      if (res = 1) then
      begin
        i = stage_month;
        re_date_begin = rep_date_begin;
        while (i > 7) do -- нужно проверить получал ли за предыдущие периоды
        begin
          i = i - 1;
          re_date_end = re_date_begin - 1;
          re_date_begin = re_date_end - extract(day from re_date_end) + 1;

          select res
          from cb_ru_drr_temp(:stage_month, :lo_id, :reg_id, :re_date_begin, :re_date_end)
          into :re_res;

          if (re_res = 1) then  -- предыдущий раз получал бонусы по текущему сотруднику
          begin
            break;
          end
        end

        if (re_res = 0) then -- положено к бонусу
        begin
          bonus_drr = case stage_month
                        when 7 then 5000
                        when 8 then 3500
                        when 9 then 2000
                        else 500
                      end;
        end else
        begin
          res_memo = 'ранее было начислено когда стаж:'||i||' мес.';
        end
      end
      suspend;
    end
  end
end
