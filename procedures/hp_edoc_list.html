CREATE PROCEDURE HP_EDOC_LIST(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDS_ID D_ID,
    EDS_FROM_EDL_ID D_ID,
    EDS_STATUS D_INTEGER,
    EDS_ENUM_EDOC_SIGN_KIND D_INTEGER,
    EDS_STATUS_DATETIME D_DATETIME,
    EDL_OBJECT_KIND D_INTEGER,
    EDL_OBJECT_ID D_ID,
    EDL_ENUM_EDOC_KIND D_INTEGER,
    EDL_STATUS D_INTEGER,
    EDL_CREATE_UN_ID D_ID,
    EDL_CREATE_DATETIME D_DATETIME,
    EDL_MANAGER_C_ID D_ID,
    EDS_STATUS_STR D_STR_CODE,
    EDL_STATUS_STR D_STR_CODE,
    EDOC_SIGN_KIND D_STR_CODE,
    EDOC_KIND D_STR_100,
    OBJ_INFO D_STR_255,
    EDL_FROM_EDC_ID D_INTEGER,
    EDL_BLANK_STATUS D_SMALLINT,
    EDL_BLANK_STATUS_STR D_STR_CODE,
    EDL_DATE_END D_DATE,
    EDL_FORM_CAPTION D_STR_255,
    EDL_BLANK_DATE D_DATE,
    VER_CES D_BOOL)
AS
declare variable eds_from_c_id d_id;
declare variable tupical_freelancer_kontract d_smallint;
declare variable edc_code d_str_code;
declare variable lst_roles d_str_255;
declare variable db_id d_integer;
declare variable sk_ces d_integer;
declare variable sk_pkb d_integer;
begin
  db_id = f_get_db_id();
  obj_kind = f_objkind(obj_kind, 'name');
  edl_form_caption = f_objname(obj_kind, obj_id);
  tupical_freelancer_kontract = 0;
  sk_ces = f_enum_id('edoc_sign_kind', 'ces');
  sk_pkb = f_enum_id('edoc_sign_kind', 'pkb');
  ver_ces = 0;

  if (db_id in (39801, 41701)) then
  begin
    select lst_roles
    from hp_get_user_roles(:un_id)
    into :lst_roles;

    --Если бухгалтер.
    if (:lst_roles containing 'accountant' or exists(select *
                                                     from permissions pm
                                                     where pm.pm_object_kind = 0 and
                                                           pm.pm_object_id = :un_id and
                                                           pm.pm_subject_kind = 3 and
                                                           pm.pm_subject_id = (select pa.pa_id
                                                                               from permactions pa
                                                                               where pa.pa_command_name = 'FinStart.CanReadFreelancerKontract') and
                                                           current_date between coalesce(pm.pm_date_begin, current_date) and coalesce(pm.pm_date_end, current_date) and
                                                           pm.pm_is_delete = 0 and
                                                           pm.pm_access_kind = 1)) then
    begin
      tupical_freelancer_kontract = 1;
    end
  end

  -- документы и соглашения подписанные клиентов
  for select eds_id
      from hp_edoc_list_tmp(:obj_kind, :obj_id, :un_id)
      into :eds_id
  do
  begin
    for select eds_from_edl_id, eds_status, eds_enum_edoc_sign_kind, eds_status_datetime, edl_object_kind,
               edl_object_id, edl_from_edc_id, edl_status, edl_create_un_id, edl_create_datetime, edl_manager_c_id,
               edc_name, edl_blank_status, edl_date_end, eds_from_c_id, edc_code, edl_blank_date
        from edoc_sign, edoc_list, edoc_cfg
        where eds_id = :eds_id and
              eds_from_edl_id = edl_id and
              edl_from_edc_id = edc_id and
              eds_from_c_id is not null
        into :eds_from_edl_id, :eds_status, :eds_enum_edoc_sign_kind, :eds_status_datetime, :edl_object_kind,
             :edl_object_id, :edl_from_edc_id, :edl_status, :edl_create_un_id, :edl_create_datetime, :edl_manager_c_id,
             edoc_kind, edl_blank_status, edl_date_end, :eds_from_c_id, :edc_code, edl_blank_date
    do
    begin
      if (db_id in (39801, 41701) and
          tupical_freelancer_kontract = 0 and
          edc_code in ('freelancer_kontract', 'freelancer_akt', 'probationer_kontract', 'kochegar_kontract')) then
        continue;

      edl_enum_edoc_kind = edl_from_edc_id;

      eds_status_str = trim(case eds_status
                              when 0 then 'не подписан'
                              when 1 then 'подписан'
                            end);

      edl_status_str = trim(case edl_status
                              when 0 then 'не подписан'
                              when 1 then 'подписан'
                              when -1 then 'аннулирован'
                            end);
      if (coalesce(edl_date_end, current_date) < current_date) then
      begin
        edl_status_str = edl_status_str || ', срок истёк';

        if (db_id = 81001 and edc_code = 'asp.agreement') then
          edl_status_str = edl_status_str || ' ' || f_datetostr(edl_date_end + 1);
      end

      edl_blank_status_str = trim(case edl_blank_status
                                    when 0 then 'В очереди'
                                    when 1 then 'Готов'
                                    when -1 then 'Ошибка'
                                  end);

      if (eds_status = 1) then
        edoc_sign_kind = f_enum('edoc_sign_kind', eds_enum_edoc_sign_kind, 'name');
      else
        edoc_sign_kind = '';

      if (obj_kind = 'clients') then
        obj_info = f_objname(edl_object_kind, edl_object_id);
      else
        obj_info = f_objname('clients', eds_from_c_id);

      if (obj_info <> '') then
        obj_info = edoc_kind || ', ' || obj_info;
      else
        obj_info = edoc_kind;

      if ((eds_status = 1) and (
        (db_id = 41701 and eds_enum_edoc_sign_kind = sk_ces) or
        (db_id = 39801 and eds_enum_edoc_sign_kind = sk_pkb)))
      then
        ver_ces = 1;
      else
        ver_ces = 0;

      suspend;
    end
  end
end
