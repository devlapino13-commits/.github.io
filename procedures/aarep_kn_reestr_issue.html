CREATE PROCEDURE AAREP_KN_REESTR_ISSUE(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    KOB_OL_DATE D_DATE,
    KOB_FROM_KC_ID ID_TYPE,
    K_KONTRACT_NO D_STR_20,
    K_FACT_DISB_DATE D_DATE,
    C_NAME_FIO D_STR_100,
    KOB_SUMMA D_CURRENCY,
    TYPE_PAY D_STR_10,
    DT_ACC_ADD_CODE D_STR_CODE,
    KT_ACC_ADD_CODE D_STR_CODE,
    DT_ACC_CODE D_STR_CODE,
    KT_ACC_CODE D_STR_CODE,
    UID D_STR_CODE,
    C_BIRTH_DATE D_DATE,
    C_ID_STR D_STR_20)
AS
declare variable c_id id_type;
begin
  for select kob.kob_ol_date, kob.kob_from_kc_id, k.k_kontract_no, k.k_fact_disb_date, c.c_name_fio, c.c_birth_date,
             c_id, kob.kob_summa,
             case
               when dt.acc_add_code in ('48801 (00)', '49201 (00)') and kt.acc_add_code = '20202 (00)' then 'Нал.'
               when dt.acc_add_code in ('48801 (00)', '49201 (00)') and kt.acc_add_code = '20501 (00)' then 'Безнал.'
             end type_pay,
             dt.acc_add_code dt_acc_add_code, kt.acc_add_code kt_acc_add_code, dt.acc_code dt_acc_code,
             kt.acc_code kt_acc_code
      from kn_operbook kob
      inner join ref_account dt on dt.acc_id = kob.kob_acc_dt
      inner join ref_account kt on kt.acc_id = kob.kob_acc_kt
      inner join knclient kn on kn.kc_id = kob.kob_from_kc_id
      inner join kontract k on k.k_id = kn.kc_from_k_id
      left join clients c on c.c_id = kn.kc_from_c_id
      where kob.kob_ol_is_active = 1 and
            kob.kob_ol_date between :rep_date_begin and :rep_date_end and
            dt.acc_add_code in ('48801 (00)', '49201 (00)') and
            kt.acc_add_code in ('20202 (00)', '20501 (00)')
      into :kob_ol_date, :kob_from_kc_id, :k_kontract_no, :k_fact_disb_date, :c_name_fio, :c_birth_date, :c_id,
           :kob_summa, :type_pay, :dt_acc_add_code, :kt_acc_add_code, :dt_acc_code, :kt_acc_code
  do
  begin
    c_id_str = '';
    c_id_str = 'n' ||c_id;
    uid = '';
    uid = f_uuid_cbr(f_uuid('knclient', :kob_from_kc_id));
    suspend;
  end
end
