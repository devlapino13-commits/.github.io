CREATE PROCEDURE CM_RU_0281_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable PR_CODE STR_100;
declare variable MA_REFUND D_INTEGER;
declare variable IS_REFINANCE d_smallint;
begin
  res_error_is_critical = 2;

  select pr_code, ma_refund
  from miniapp
  inner join product on pr_id = ma_from_pr_id
  where ma_id = :ma_id and
        ma_date_end = '31.12.9999'
  into pr_code, ma_refund;

  -- флаг
  is_refinance = iif(
    pr_code containing '.refinance.' or
    pr_code containing '.refinance_oivo.' or
    pr_code containing '.b_refinance.',
    1, 0
  );

  if (ma_refund = 1) then
  begin
    -- внешние займы
    if (exists(select 1
               from kontrefinance
               where kr_from_ma_id = :ma_id
                 and kr_from_k_id is null)
        and is_refinance = 0) then
    begin
      res_error_text = 'Рефинансирование внешних займов приостановлено';
      suspend;
    end

    -- внутренние займы
    if (exists(select 1
               from kontrefinance
               where kr_from_ma_id = :ma_id
                 and kr_from_k_id is not null)
        and is_refinance = 1) then
    begin
      res_error_text = 'По продукту "Рефинанс" и "Рефинанс OiVO" и "Б-Рефинанс" нельзя выбирать внутренние займы для рефинансирования';
      suspend;
    end
  end

  -- запрет без флага
  if (is_refinance = 1 and ma_refund <> 1) then
  begin
    res_error_text = 'Продукт "Рефинанс", "Рефинанс OiVO" и "Б-Рефинанс" выдается только для рефинансирования';
    suspend;
  end
end
