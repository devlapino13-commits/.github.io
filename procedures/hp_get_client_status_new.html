CREATE PROCEDURE HP_GET_CLIENT_STATUS_NEW(
    REP_C_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    C_LAST_STATUS INTEGER,
    C_CURR_STATUS INTEGER,
    POLEZ_PLATEJ_TOTAL INTEGER)
AS
declare variable sh_date date;
declare variable ol_date date;
declare variable repay_sum curr_type;
declare variable delinq_day integer;
declare variable polez_platej smallint;
declare variable dolya double precision;
declare variable is_parallel smallint;
declare variable tpp_id integer;
declare variable tpp_polez_platej smallint;
declare variable number_platej integer;
declare variable prior integer;
declare variable k_id id_type;
declare variable pr_product_kind integer;
declare variable ks_date_end date;
declare variable cs_star_total integer;
declare variable cs_date_begin date;
declare variable i integer;
declare variable k integer;
declare variable polez_platej_19 integer;
declare variable max_delinq_day_19 integer;
declare variable kol_delinq_do_7_19 integer;
declare variable kol_delinq_bolshe_8_19 integer;
declare variable polez_platej_18 integer;
declare variable max_delinq_day_18 integer;
declare variable kol_delinq_do_7_18 integer;
declare variable kol_delinq_bolshe_8_18 integer;
declare variable polez_platej_12 integer;
declare variable max_delinq_day_12 integer;
declare variable kol_delinq_do_7_12 integer;
declare variable kol_delinq_bolshe_8_12 integer;
declare variable polez_platej_14 integer;
declare variable max_delinq_day_14 integer;
declare variable kol_delinq_do_7_14 integer;
declare variable kol_delinq_bolshe_8_14 integer;
declare variable polez_platej_9 integer;
declare variable max_delinq_day_9 integer;
declare variable kol_delinq_do_7_9 integer;
declare variable kol_delinq_bolshe_8_9 integer;
declare variable polez_platej_6 integer;
declare variable max_delinq_day_6 integer;
declare variable kol_delinq_do_7_6 integer;
declare variable kol_delinq_bolshe_8_6 integer;
declare variable woff_sum curr_type;
declare variable pr_code varchar(100);
begin
--  rep_c_id = 1010100061060;
-- перед вычислением вытаскиваем данный из таблицы CLIENTSTATUS
  i = 0;
  cs_star_total = null;
  cs_date_begin = null;
  select cs_star_total, cs_date_begin, cs_polez_platej
  from clientstatus
  where cs_from_c_id = :rep_c_id and
        cs_ol_is_delete = 0 and
        :rep_date_end between cs_date_begin and cs_date_end
  into cs_star_total, cs_date_begin, polez_platej_total;
  cs_star_total = coalesce(cs_star_total, 0);
  cs_date_begin = coalesce(cs_date_begin, '31.12.2999');

  c_last_status = cs_star_total; -- статус на начало
  if (rep_date_end = cs_date_begin) then
  begin
    c_curr_status = cs_star_total;
    suspend;
    exit;
  end

  if (cs_date_begin < '31.12.2999') then -- было if (cs_date_begin is not null)
  begin
--  проверяем после фиксирования статусов не было ли погашении по текущему клиенту
--  если не было то статус не пересчитывается
    repay_sum = null;
    select -sum(ok_os + ok_perc)
    from knclient, operkred
    where kc_from_c_id = :rep_c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = ok_from_k_id and
          ok_ol_is_active = 1 and
          ok_ol_date > :cs_date_begin and
          ok_ol_date <= :rep_date_end and
          ok_type in (4, 5) and
          ok_os + ok_perc < 0
    into repay_sum;
    repay_sum = coalesce(repay_sum, 0);
    if (repay_sum = 0) then
    begin
      c_curr_status = cs_star_total;
      suspend;
      exit;
    end
  end

  delete from temp_payable_payment
  where tpp_id > 0;
  k = 0;
  for select k_id, pr_product_kind, ks_date_end,
             cast(kc_amount_approved as double precision) / k_amount_approved, pr_code
      from knclient, kontstate, kontract, product
      where kc_from_c_id = :rep_c_id and
            kc_amount_approved > 0 and
            kc_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            ks_date_begin <= :rep_date_end
      order by ks_date_begin desc
      into k_id, pr_product_kind, ks_date_end, dolya, pr_code
  do
  begin
    delinq_day = 0;

    if (ks_date_end > rep_date_end - 1) then
    begin
--       если кредит активный то проверяем на текущую просрочку
      select :rep_date_end - oks_delinq_date
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_close_state = 0 and
            oks_from_k_id = :k_id
      into delinq_day;
      if (delinq_day < 0) then
        delinq_day = 0;
    end

    if (pr_product_kind = 2 or pr_code containing '.express.') then
      is_parallel = 1;
    else
      is_parallel = 0;

    if (delinq_day > 0) then
    begin
      i = i + 1;
      insert into temp_payable_payment (tpp_id, tpp_sh_date, tpp_ol_date, tpp_repay_sum, tpp_delinq_day,
                                        tpp_polez_platej, tpp_is_parallel)
      values (:i, :rep_date_end, :rep_date_end, 0, :delinq_day, 0, :is_parallel);
    end

    for with tabl1
        as (select ok_sh_date,
--                   max(ok_ol_date) as ol_date,
                   max(iif(ok_sh_date between '20.03.2020' and '31.08.2020' and
                           ok_ol_date > ok_sh_date, ok_sh_date - 11, ok_ol_date)) as ol_date, -- Sanjari, 28.07.2020
                   sum(ok_os + ok_sh_os) as os, sum(ok_perc + ok_sh_perc) as perc
--               ,-sum(iif(ok_type = 4, ok_os, 0)) as repay_os,
--               -sum(iif(ok_type = 4, ok_perc, 0)) as repay_perc
            from operkred
            where ok_ol_is_active = 1 and
                  ok_type < 7 and
                  ok_ol_date <= :rep_date_end and
                  ok_from_k_id = :k_id and
                  ok_sh_id > 0 -- дату выдачу не включать
                  and
                  (ok_os <> 0 or
                  ok_perc <> 0 or
                  ok_sh_os <> 0 or
                  ok_sh_perc <> 0) -- погашении только по пеням не включать
            group by ok_sh_date, ok_from_k_id, ok_from_c_id
            having sum(ok_os + ok_sh_os) <= 0 and sum(ok_perc + ok_sh_perc) <= 0)
        select ok_sh_date, ol_date,
--          repay_os + repay_perc as repay_sum,
--          iif(ol_date > ok_sh_date, ol_date - ok_sh_date, 0) as delinq_day,
               iif(ok_sh_date - ol_date > 10, 0, 1) as polez_platej
        from tabl1
        order by ok_sh_date desc
        into sh_date, ol_date,
--          repay_sum,
--          delinq_day,
             polez_platej
    do
    begin
--     погашенная сумма который закрывает платеж не должно быть менее 2500 руб.
      /*
       select -sum(ok_os + ok_perc) from operkred
       where ok_ol_is_active = 1
       and ok_from_k_id = :k_id
       and ok_sh_date = :sh_date
       and ok_ol_date = :ol_date
       and ok_type = 4
       into repay_sum;
       */
--     добавлено по письму от 18.10.2017 Зара, При учете легитимности погашения учитывать суммарно все НЕДОСРОЧНЫЕ погашения по ОДНОМУ платежу.
      repay_sum = null;
      delinq_day = null;
      woff_sum = null;
      select -sum(iif(ok_type = 4, ok_os + ok_perc, 0)),
--              max(ok_ol_date - ok_sh_date),
              max(iif(ok_sh_date between '20.03.2020' and '31.08.2020' and
                      ok_ol_date > ok_sh_date, ok_sh_date - 11, ok_ol_date) - ok_sh_date),
             -sum(iif(ok_type = 5, ok_os + ok_perc, 0))
      from operkred, operlist
      where ok_ol_is_active = 1 and
            ok_sh_date = :sh_date and
            ok_from_k_id = :k_id and
            ok_type in (4, 5) and
            ok_os + ok_perc < 0 and
            ok_sh_date - ok_ol_date <= 10 and
            ok_from_ol_id = ol_id and
            ol_memo not starting 'Корректировка согласно протокола кредитного комитета №29 от 01.04.2019г. Контракт №:'
      into repay_sum, delinq_day, woff_sum;
      repay_sum = coalesce(repay_sum, 0);
      delinq_day = coalesce(delinq_day, 0);
      woff_sum = coalesce(woff_sum, 0);

      if (delinq_day < 0) then
        delinq_day = 0;
      repay_sum = dolya * repay_sum;

      if (is_parallel = 1 or repay_sum < 2500.00 or woff_sum > 0) then
        polez_platej = 0;

      if ((polez_platej = 1 or delinq_day > 0)
--            and k < 19
          and
          repay_sum > 0) then
      begin
        i = i + 1;
        if (polez_platej = 1) then
          k = k + 1;
        insert into temp_payable_payment (tpp_id, tpp_sh_date, tpp_ol_date, tpp_repay_sum, tpp_delinq_day,
                                          tpp_polez_platej, tpp_is_parallel)
        values (:i, :sh_date, :ol_date, :repay_sum, :delinq_day, :polez_platej, :is_parallel);
      end
    end
  end
--   присваиваем номер платежа
  number_platej = 0;
  prior = -1;
  for select tpp_id, tpp_polez_platej
      from temp_payable_payment
      order by tpp_sh_date desc
      into tpp_id, tpp_polez_platej
  do
  begin
    if (number_platej = 0 and
        tpp_polez_platej = 1) then
      number_platej = 1;
    if (tpp_polez_platej = 1 and
        prior = 1) then
      number_platej = number_platej + 1;
    if (tpp_polez_platej = 0 and
        prior = 1) then
      number_platej = number_platej + 1;

    prior = tpp_polez_platej;

    update temp_payable_payment
    set tpp_number_platej = :number_platej
    where tpp_id = :tpp_id;
  end
  c_curr_status = 0;

  polez_platej_19 = 0;
  max_delinq_day_19 = 0;
  kol_delinq_do_7_19 = 0;
  kol_delinq_bolshe_8_19 = 0;
  polez_platej_18 = 0;
  max_delinq_day_18 = 0;
  kol_delinq_do_7_18 = 0;
  kol_delinq_bolshe_8_18 = 0;
  polez_platej_12 = 0;
  max_delinq_day_12 = 0;
  kol_delinq_do_7_12 = 0;
  kol_delinq_bolshe_8_12 = 0;
  polez_platej_14 = 0;
  max_delinq_day_14 = 0;
  kol_delinq_do_7_14 = 0;
  kol_delinq_bolshe_8_14 = 0;
  polez_platej_9 = 0;
  max_delinq_day_9 = 0;
  kol_delinq_do_7_9 = 0;
  kol_delinq_bolshe_8_9 = 0;
  polez_platej_6 = 0;
  max_delinq_day_6 = 0;
  kol_delinq_do_7_6 = 0;
  kol_delinq_bolshe_8_6 = 0;
  polez_platej_total = 0;

  select sum(iif(tpp_number_platej <= 19, tpp_polez_platej, 0)) polez_platej_19,
         max(iif(tpp_number_platej <= 19, tpp_delinq_day, 0)) max_delinq_day_19,
         sum(iif(tpp_number_platej <= 19 and tpp_delinq_day between 1 and 7, 1, 0)) kol_delinq_do_7_19,
         sum(iif(tpp_number_platej <= 19 and tpp_delinq_day >= 8, 1, 0)) kol_delinq_bolshe_8_19,

         sum(iif(tpp_number_platej <= 18, tpp_polez_platej, 0)) polez_platej_18,
         max(iif(tpp_number_platej <= 18, tpp_delinq_day, 0)) max_delinq_day_18,
         sum(iif(tpp_number_platej <= 18 and tpp_delinq_day between 1 and 7, 1, 0)) kol_delinq_do_7_18,
         sum(iif(tpp_number_platej <= 18 and tpp_delinq_day >= 8, 1, 0)) kol_delinq_bolshe_8_18,

         sum(iif(tpp_number_platej <= 12, tpp_polez_platej, 0)) polez_platej_12,
         max(iif(tpp_number_platej <= 12, tpp_delinq_day, 0)) max_delinq_day_12,
         sum(iif(tpp_number_platej <= 12 and tpp_delinq_day between 1 and 7, 1, 0)) kol_delinq_do_7_12,
         sum(iif(tpp_number_platej <= 12 and tpp_delinq_day >= 8, 1, 0)) kol_delinq_bolshe_8_12,

         sum(iif(tpp_number_platej <= 14, tpp_polez_platej, 0)) polez_platej_14,
         max(iif(tpp_number_platej <= 14, tpp_delinq_day, 0)) max_delinq_day_14,
         sum(iif(tpp_number_platej <= 14 and tpp_delinq_day between 1 and 7, 1, 0)) kol_delinq_do_7_14,
         sum(iif(tpp_number_platej <= 14 and tpp_delinq_day >= 8, 1, 0)) kol_delinq_bolshe_8_14,

         sum(iif(tpp_number_platej <= 9, tpp_polez_platej, 0)) polez_platej_9,
         max(iif(tpp_number_platej <= 9, tpp_delinq_day, 0)) max_delinq_day_9,
         sum(iif(tpp_number_platej <= 9 and tpp_delinq_day between 1 and 7, 1, 0)) kol_delinq_do_7_9,
         sum(iif(tpp_number_platej <= 9 and tpp_delinq_day >= 8, 1, 0)) kol_delinq_bolshe_8_9,

         sum(iif(tpp_number_platej <= 6, tpp_polez_platej, 0)) polez_platej_6,
         max(iif(tpp_number_platej <= 6, tpp_delinq_day, 0)) max_delinq_day_6,
         sum(iif(tpp_number_platej <= 6 and tpp_delinq_day between 1 and 7, 1, 0)) kol_delinq_do_7_6,
         sum(iif(tpp_number_platej <= 6 and tpp_delinq_day >= 8, 1, 0)) kol_delinq_bolshe_8_6,

         sum(tpp_polez_platej) polez_platej_total
  from temp_payable_payment
--    where tpp_number_platej <= 19
  into polez_platej_19, max_delinq_day_19, kol_delinq_do_7_19, kol_delinq_bolshe_8_19, polez_platej_18,
       max_delinq_day_18, kol_delinq_do_7_18, kol_delinq_bolshe_8_18, polez_platej_12, max_delinq_day_12,
       kol_delinq_do_7_12, kol_delinq_bolshe_8_12, polez_platej_14, max_delinq_day_14, kol_delinq_do_7_14,
       kol_delinq_bolshe_8_14, polez_platej_9, max_delinq_day_9, kol_delinq_do_7_9, kol_delinq_bolshe_8_9,
       polez_platej_6, max_delinq_day_6, kol_delinq_do_7_6, kol_delinq_bolshe_8_6, polez_platej_total;

  polez_platej_19 = coalesce(polez_platej_19, 0);
  max_delinq_day_19 = coalesce(max_delinq_day_19, 0);
  kol_delinq_do_7_19 = coalesce(kol_delinq_do_7_19, 0);
  kol_delinq_bolshe_8_19 = coalesce(kol_delinq_bolshe_8_19, 0);
  polez_platej_18 = coalesce(polez_platej_18, 0);
  max_delinq_day_18 = coalesce(max_delinq_day_18, 0);
  kol_delinq_do_7_18 = coalesce(kol_delinq_do_7_18, 0);
  kol_delinq_bolshe_8_18 = coalesce(kol_delinq_bolshe_8_18, 0);
  polez_platej_12 = coalesce(polez_platej_12, 0);
  max_delinq_day_12 = coalesce(max_delinq_day_12, 0);
  kol_delinq_do_7_12 = coalesce(kol_delinq_do_7_12, 0);
  kol_delinq_bolshe_8_12 = coalesce(kol_delinq_bolshe_8_12, 0);
  polez_platej_14 = coalesce(polez_platej_14, 0);
  max_delinq_day_14 = coalesce(max_delinq_day_14, 0);
  kol_delinq_do_7_14 = coalesce(kol_delinq_do_7_14, 0);
  kol_delinq_bolshe_8_14 = coalesce(kol_delinq_bolshe_8_14, 0);
  polez_platej_9 = coalesce(polez_platej_9, 0);
  max_delinq_day_9 = coalesce(max_delinq_day_9, 0);
  kol_delinq_do_7_9 = coalesce(kol_delinq_do_7_9, 0);
  kol_delinq_bolshe_8_9 = coalesce(kol_delinq_bolshe_8_9, 0);
  polez_platej_6 = coalesce(polez_platej_6, 0);
  max_delinq_day_6 = coalesce(max_delinq_day_6, 0);
  kol_delinq_do_7_6 = coalesce(kol_delinq_do_7_6, 0);
  kol_delinq_bolshe_8_6 = coalesce(kol_delinq_bolshe_8_6, 0);
  polez_platej_total = coalesce(polez_platej_total, 0);
--   ***** - 18 последних полезных платежей произведены  с просрочкой не более 0 дней
  if (polez_platej_18 = 18 and
      max_delinq_day_18 = 0 and
      kol_delinq_do_7_18 = 0 and
      kol_delinq_bolshe_8_18 = 0) then
  begin
    c_curr_status = 5;
    suspend;
    exit;
  end
--   ***** - Клиент, взявший заем в статусе пяти звезд имеет право на одну просрочку до 7 дней
  if (c_last_status = 5) then
  begin
    if (polez_platej_19 = 19 and
        max_delinq_day_19 <= 7 and
        kol_delinq_do_7_19 <= 1 and
        kol_delinq_bolshe_8_19 = 0) then
    begin
      c_curr_status = 5;
      suspend;
      exit;
    end
  end
--   **** -12 последних полезных платежей произведены  с просрочкой не более 0 дней
  if (polez_platej_12 = 12 and
      max_delinq_day_12 = 0 and
      kol_delinq_do_7_12 = 0 and
      kol_delinq_bolshe_8_12 = 0) then
  begin
    c_curr_status = 4;
    suspend;
    exit;
  end
--   **** - Клиент, взявший заем в статусе  как минимум четырех звезд имеет право на две просрочки до 7 дней
  if (c_last_status >= 4) then
  begin
    if (polez_platej_14 = 14 and
        max_delinq_day_14 <= 7 and
        kol_delinq_do_7_14 <= 2 and
        kol_delinq_bolshe_8_14 = 0) then
    begin
      c_curr_status = 4;
      suspend;
      exit;
    end
  end
--   *** - 9 последних полезных платежей произведены  с просрочкой не более 0 дней
  if (polez_platej_9 = 9 and
      max_delinq_day_9 = 0 and
      kol_delinq_do_7_9 = 0 and
      kol_delinq_bolshe_8_9 = 0) then
  begin
    c_curr_status = 3;
    suspend;
    exit;
  end
--   *** - Клиент, взявший заем в статусе как минимум трех звезд имеет право на три просрочки до 7 дней
  if (c_last_status >= 3) then
  begin
    if (polez_platej_12 = 12 and
        max_delinq_day_12 <= 7 and
        kol_delinq_do_7_12 <= 3 and
        kol_delinq_bolshe_8_12 = 0) then
    begin
      c_curr_status = 3;
      suspend;
      exit;
    end
  end
--   ** - 9 последних полезных платежей произведены с просрочкой не более 7 дней
  if (polez_platej_9 = 9 and
      max_delinq_day_9 <= 7 and
      kol_delinq_do_7_9 <= 9 and
      kol_delinq_bolshe_8_9 = 0) then
  begin
    c_curr_status = 2;
    suspend;
    exit;
  end
--   *** - Клиент, взявший заем в статусе двух звезд и более и допустивший одну просрочку
--   в 8 дней и выше спускается до статуса 1 звезды
  if (c_last_status >= 2) then
  begin
    if (polez_platej_9 = 9 and
        max_delinq_day_9 < 21 and
        kol_delinq_do_7_9 <= 9 and
        kol_delinq_bolshe_8_9 <= 1) then
    begin
      c_curr_status = 1;
      suspend;
      exit;
    end
  end

--   *  (одна звезда) – 6 последних полезных платежей произведены  с просрочкой не более 7 дней
  if (polez_platej_6 = 6 and
      max_delinq_day_6 <= 7 and
      kol_delinq_do_7_6 <= 6 and
      kol_delinq_bolshe_8_6 = 0) then
  begin
    c_curr_status = 1;
    suspend;
  end
  else
  if (max_delinq_day_6 >= 21 or (kol_delinq_bolshe_8_6 >= 1) or (polez_platej_6 < 6)) then
  begin
    c_curr_status = 0;
    suspend;
  end
  else
    suspend;

end
