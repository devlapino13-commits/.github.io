CREATE PROCEDURE HP_MA_TIMER_KZ_START(
    MA_ID D_ID NOT NULL)
AS
declare variable SIGN_DT D_DATETIME;
declare variable REG_ID D_INTEGER;
declare variable C_NAME D_STR_100;
declare variable MO_C_ID D_ID;
declare variable SENT_TXT D_STR_255;
begin
  -- RFCKZ-659: Старт таймера 55 минут и первый пуш КМ
  -- Вызывается из HP_EDOC_AFTER_SIGN_ACTION при подписании loan.application

  -- Момент подписания заявки клиентом
  select eds.eds_status_datetime
  from edoc_list edl
  inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
  inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
  where edl.edl_object_kind = f_objkind('miniapp', 'id')
    and edl.edl_object_id = :ma_id
    and edc.edc_code = 'loan.application'
    and eds.eds_status = 1
  order by edl.edl_id desc
  rows 1
  into sign_dt;

  if (sign_dt is null) then exit;

  -- RFCKZ-956
  select ma.ma_office_id
  from miniapp ma
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.9999'
  rows 1
  into reg_id;

  select c.c_name_fio
  from miniappclient mac
  inner join clients c on c.c_id = mac.mac_from_c_id
  where mac.mac_from_ma_id = :ma_id
    and mac.mac_amount > 0
  rows 1
  into c_name;

  -- Записываем таймер
  if (not exists(select *
                 from ma_timer_kz
                 where mtk_from_ma_id = :ma_id
                   and mtk_is_closed = 0)) then
  begin
    insert into ma_timer_kz (mtk_from_ma_id, mtk_sign_datetime)
    values (:ma_id, :sign_dt);
  end

  -- Пуш
  for select ps.ps_from_c_id
      from hr_pstatus ps
      inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where current_date between ps.ps_date_begin and ps.ps_date_end
        and ps.ps_is_working = 1
        and ps.ps_status = 1
        and ps.ps_from_reg_id = :reg_id
        and jk.jk_system_code = 'JobSystem.Region_KM'
      into mo_c_id
  do
  begin
    sent_txt = mo_c_id || '; У вас 30 минут для принятия решения по заявке ' || c_name;
    execute procedure fsrobot_insert_event('DoMailing', 'push_sent_app', left(:sent_txt, 255), 1000000000000);
  end                                     
end
