CREATE PROCEDURE AAREP_UNDERWRITING_ANALYS
RETURNS (
    ACL_ID INTEGER,
    ACL_DATE DATE,
    ACTION_KIND VARCHAR(100),
    REG_NAME VARCHAR(100),
    STAFF_FIO VARCHAR(100),
    ACL_START_TIME TIMESTAMP,
    ACL_FINISH_TIME TIMESTAMP,
    ACL_DELTA_TIME VARCHAR(20),
    ACL_DELTA_SECOND INTEGER,
    OPR_ID DECIMAL(18,0))
AS
declare variable acl_from_reg_id integer;
declare variable acl_respons_c_id decimal(18,0);
declare variable acl_datetime timestamp;
declare variable line_no integer;
declare variable acl_status_prior integer;
declare variable acl_status integer;
declare variable sec integer;
declare variable delta_day integer;
declare variable delta_hour integer;
declare variable delta_min integer;
declare variable delta_sec integer;
declare variable st varchar(20);
begin
  for select acl_id, acl_date, acl_from_reg_id, acl_respons_c_id, acl_datetime, opr_id
      from actionlist, objectprops
      where opr_object_kind = acl_object_kind and
            opr_object_id = acl_object_id and
            opr_subject_kind = -1 and
            opr_subject_id = acl_from_rd_id and
            acl_dont_show_in_order = 0 and
            acl_status <> 0
      into :acl_id, acl_date, acl_from_reg_id, acl_respons_c_id, acl_datetime, opr_id
  do
  begin
    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :acl_from_reg_id
    into :reg_name;

    select c_name_fio
    from clients
    where c_id = :acl_respons_c_id
    into :staff_fio;

    acl_status_prior = 0;
    acl_start_time = null;
    line_no = 0;
    for select ovs_datetime, ovs_add_value
        from objectvalues
        where ovs_from_opr_id = :opr_id and
              ovs_is_active = 1
        order by ovs_datetime
        into acl_finish_time, acl_status
    do
    begin
      if (acl_start_time is null) then
      begin
        acl_start_time = acl_finish_time;
        acl_status_prior = acl_status;
      end
      st = 'XX';
      if (acl_status_prior = -2) then
        st = 'Отказан';
      if (acl_status_prior = -1) then
        st = 'На доработку';
      if (acl_status_prior = 0) then
        st = 'В очереди';
      if (acl_status_prior = 1) then
        st = 'Принят на обработку';
      if (acl_status_prior = 2) then
        st = 'Подтвежден';
      action_kind = st;
      action_kind = action_kind || ' -> ';
      st = 'XX';
      if (acl_status = -2) then
        st = 'Отказан';
      if (acl_status = -1) then
        st = 'На доработку';
      if (acl_status = 0) then
        st = 'В очереди';
      if (acl_status = 1) then
        st = 'Принят на обработку';
      if (acl_status = 2) then
        st = 'Подтвежден';
      action_kind = action_kind || st;

      if (line_no = 0) then
        action_kind = 'Новая заявка';

      sec = datediff(second, acl_start_time, acl_finish_time);
      acl_delta_second = sec;

      delta_day = round(sec / 86400, 0);
      if (delta_day > sec / 86400) then
        delta_day = delta_day - 1;
      sec = sec - delta_day * 86400;

      delta_hour = round(sec / 3600, 0);
      if (delta_hour > sec / 3600) then
        delta_hour = delta_hour - 1;
      sec = sec - delta_hour * 3600;

      delta_min = round(sec / 60, 0);
      if (delta_min > sec / 60) then
        delta_min = delta_min - 1;
      sec = sec - delta_min * 60;

      delta_sec = sec;

      acl_delta_time = right('000' || delta_day, 3) || ':' || right('00' || delta_hour, 2) || ':' || right('00' || delta_min, 2) || ':' || right('00' || delta_sec, 2);

      suspend;
      acl_start_time = acl_finish_time;
      acl_status_prior = acl_status;
      line_no = line_no + 1;
    end
  end
end
