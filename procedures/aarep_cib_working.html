CREATE PROCEDURE AAREP_CIB_WORKING
RETURNS (
    CW_ID D_INTEGER,
    CW_DATETIME D_DATETIME,
    UN_FIO D_STR_100,
    ACTION_TYPE D_STR_20)
AS
begin
  for select cw_id,
             cw_datetime,
             un.un_fio,
             trim(iif(cw_is_working = 1, 'Включил', 'Выключил'))
  from cib_working cw
  inner join username un on un.un_id = cw.cw_from_un_id
  order by cw_id desc
  into :cw_id,
       :cw_datetime,
       :un_fio,
       :action_type
  do
    suspend;
end
