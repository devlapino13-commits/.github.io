CREATE PROCEDURE HP_SPLITB_STR(
    STR_VALUE D_BLOB_TEXT,
    DELIMITER D_STR_LANG)
RETURNS (
    RES_STR D_STR_8100)
AS
declare variable txt d_blob_text;
declare variable delim_pos d_integer;
begin
  if (coalesce(str_value, '') = '') then
    exit;

  delim_pos = 1;
  txt = str_value;
  while (txt <> '') do
  begin
    res_str = '';
    delim_pos = position(delimiter, txt);
    if (delim_pos > 0) then
    begin
      res_str = substring(txt from 1 for delim_pos - 1);
      txt = substring(txt from delim_pos + char_length(delimiter));
    end
    else
    begin
      res_str = txt;
      txt = '';
    end
    if (trim(res_str) <> '') then
    suspend;
  end
end
