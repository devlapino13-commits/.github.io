CREATE PROCEDURE HP_INTERNS_VIEW(
    UN_ID_INP D_ID,
    USER_ROLE D_STR_100,
    REG_ID_INP D_INTEGER,
    DATE_BEGIN_INP D_DATE,
    DATE_END_INP D_DATE,
    FULL_INP D_SMALLINT,
    I_ID_INP D_INTEGER = 0,
    MP D_SMALLINT = 0)
RETURNS (
    I_ID D_INTEGER,
    C_ID D_ID,
    REG_ID D_INTEGER,
    UN_ID D_ID,
    JL_ID D_INTEGER,
    NAME_FIO D_STR_100,
    REG_NAME D_STR_100,
    JL_NAME D_STR_100,
    PC_RESULT D_SMALLINT,
    PC_USERNAME D_STR_100,
    PC_DATE D_DATETIME,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    ASP_STATUS D_STR_20,
    SD_DOG_NO D_STR_20,
    SD_EDL_STATUS D_STR_20,
    EC_RESULT D_SMALLINT,
    EC_DATE D_DATETIME,
    EC_USERNAME D_STR_100,
    IS_WORKING D_SMALLINT,
    DATE_END_FACT D_DATE,
    ASP_STATUS_CODE D_STR_CODE,
    SD_EDL_STATUS_CODE D_STR_CODE,
    REGION D_STR_CODE)
AS
declare variable sd_id d_integer;
declare variable sshe_id d_integer;
declare variable reg_id_tmp d_integer;
declare variable i_id_tmp d_integer;
begin
  reg_id_tmp = 0;

  for select i.i_id, i.i_from_c_id, i.i_from_reg_id, i.i_registred_by, i.i_from_jl_id, c.c_name_fio, jl.jl_name,
             i.i_date_begin, i.i_date_end, f_edoc_status('asp.agreement', i.i_from_c_id, current_date),
             i.i_date_end_fact
      from interns i
      inner join hp_user_region(:user_role, :un_id_inp, :reg_id_inp) ur on i.i_from_reg_id = ur.reg_id
      inner join clients c on i.i_from_c_id = c.c_id
      inner join joblist jl on jl.jl_id = i.i_from_jl_id
      where i.i_is_delete = 0 and
            iif(:mp = 1, i.i_from_jl_id = 270, true) and
            iif(:i_id_inp = 0, i.i_id <> 0, i.i_id = :i_id_inp) and
            iif(:full_inp = 1, (i.i_date_begin is null) or
            (i.i_date_begin between :date_begin_inp and :date_end_inp), i.i_date_begin between :date_begin_inp and :date_end_inp)
      into :i_id, :c_id, :reg_id, :un_id, :jl_id, :name_fio, :jl_name, :date_begin, :date_end, :asp_status_code,
           :date_end_fact
  do
  begin
    pc_result = null;
    pc_username = null;
    pc_date = null;
    sd_dog_no = null;
    sd_id = null;
    ec_result = null;
    ec_date = null;
    ec_username = null;

    if (exists(select *
               from intern_committee_result icr
               where icr.icr_from_i_id = :i_id)) then
    begin
      i_id_tmp = null;

      select icr.icr_from_i_id, icr.icr_personnel_com_res, un.un_fio, icr.icr_personnel_com_timestamp, sd.sd_dogovor_no,
             sd.sd_id, icr.icr_educational_com_res, icr.icr_educational_com_timestamp, un1.un_fio
      from intern_committee_result icr
      inner join username un on icr.icr_personnel_com = un.un_id
      left join servicedog sd on icr.icr_from_sd_id = sd.sd_id
      left join username un1 on icr.icr_educational_com = un1.un_id
      where icr.icr_from_i_id = :i_id and
            ((icr.icr_personnel_com_res = 0 and
            datediff(day, icr.icr_personnel_com_timestamp, current_timestamp) <= 1) or
            (icr.icr_personnel_com_res = 1 and
            icr.icr_educational_com_res = 0 and
            datediff(day, icr.icr_educational_com_timestamp, current_timestamp) <= 1) or
            (icr.icr_personnel_com_res = 1 and
            icr.icr_educational_com_res = 1 and
            datediff(day, icr.icr_educational_com_timestamp, current_timestamp) <= 1) or
            (icr.icr_personnel_com_res = 1 and
            icr.icr_educational_com_res is null))
      into :i_id_tmp, :pc_result, :pc_username, :pc_date, :sd_dog_no, :sd_id, :ec_result, :ec_date, :ec_username;

      if (:i_id_tmp is null) then
        continue;

      if (:mp = 1 and
          :ec_result is not null) then
        continue;

      if (:mp = 1 and
          coalesce(:pc_result, 0) <> 1) then
        continue;
    end

    if (:reg_id_tmp <> :reg_id) then
    begin
      reg_id_tmp = :reg_id;
      select reg.reg_code || ' ' || reg.reg_name, reg2.reg_name
      from region reg
      inner join region reg2 on reg.parent = reg2.reg_unicode
      where reg.reg_unicode = :reg_id
      into :reg_name, :region;
    end

    if (:asp_status_code = 'signed') then
      asp_status = 'Подписан';
    if (:asp_status_code = 'unsigned') then
      asp_status = 'Не подписан';
    if (:asp_status_code = 'cancelled') then
      asp_status = 'Аннулирован';
    if (:asp_status_code = 'empty') then
      asp_status = 'Не отправлялся';

    -- смотрим по прикреплению в доках
    if (exists(
      select *
      from clientdocs
        inner join passkind
          on pk_id = cds_from_pk_id
        inner join passdoctype
          on pdt_id = pk_from_pdt_id
            and pdt_code = 'contact_eds'
      where cds_from_c_id = :c_id
        and current_date between cds_doc_date_begin
        and coalesce(cds_doc_date_end, '31.12.9999')
        and cds_is_delete = 0
        and cds_doc_blob_1 > 0)) then
    begin
      asp_status_code = 'signed';
      asp_status = 'Прикреплен';
    end

    sd_edl_status = null;
    sd_edl_status_code = null;

    if (sd_id is not null) then
    begin
      sd_edl_status_code = f_edoc_status('probationer_kontract', :sd_id, current_date);

      if (sd_edl_status_code = 'empty') then
        sd_edl_status = 'Не отправлялся';
      if (sd_edl_status_code = 'signed') then
        sd_edl_status = 'Подписан';
      if (sd_edl_status_code = 'unsigned') then
        sd_edl_status = 'Не подписан';
      if (sd_edl_status_code = 'cancelled') then
        sd_edl_status = 'Аннулирован';
    end

    if (:mp = 1 and
        coalesce(sd_edl_status_code, 'empty') <> 'signed') then
      continue;

    is_working = null;
    if (:ec_result is not null and
        :ec_result = 1 and
        exists(select *
               from hr_pstatus ps
               where ps.ps_from_c_id = :c_id and
                     current_date between ps.ps_date_begin and ps.ps_date_end and
                     ps.ps_is_working = 1)) then
      is_working = 1;

    suspend;
  end
end
