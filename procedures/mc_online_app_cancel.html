CREATE PROCEDURE MC_ONLINE_APP_CANCEL(
    MCL_ID D_ID,
    ONLINE_APP_ID D_ID)
RETURNS (
    ERROR_TEXT D_STR_LONG)
AS
declare variable MA_ID D_ID;
declare variable K_ID D_ID;
declare variable UN_ID D_ID;
declare variable MCL_LANG_CODE D_STR_LANG;
begin
  error_text = '';

  select exa.exa_from_ma_id, exa.exa_from_k_id, exa.exa_from_un_id
  from online_app oa
  inner join expressapp exa on exa.exa_id = oa.oa_from_exa_id
  where oa.oa_id = :online_app_id
  into ma_id, k_id, un_id;

  select mcl.mcl_lang_code
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :mcl_lang_code;

  if (exists(select *
             from kontstate ks
             where ks.ks_from_k_id = :k_id and
                   ks.ks_state = 1 and
                   ks.ks_ol_is_delete = 0)) then
  begin
    error_text = f_lang(:mcl_lang_code, 'Аннулировать заявку не удалось.');
    suspend;
    exit;
  end 

  if (coalesce(ma_id, 0) <> 0) then
  begin
    -- Инсертим запись в kontreslist со значением "Перерегистрация клиента"
    insert into kontreslist (krl_id, krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                             krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
    values (-1, 89, :un_id, null, 0, :ma_id, current_timestamp, current_date, current_date, 0);
    execute procedure hp_miniapp_status_change(:ma_id, :un_id, 0);
  end

  if (coalesce(k_id, 0) <> 0) then
  begin
    --Таблица kontlog автоматически обновиться со статусом -1
    update kontract kt
    set kt.k_status = -1
    where kt.k_id = :k_id;
  end 

  update online_app oa
  set oa.oa_status = -1
  where oa.oa_id = :online_app_id and
        oa.oa_from_mcl_id = :mcl_id;

  suspend;
end
