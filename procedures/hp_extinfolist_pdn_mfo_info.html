CREATE PROCEDURE HP_EXTINFOLIST_PDN_MFO_INFO(
    NEW_KC_ID D_ID)
RETURNS (
    RES_LOAN_COUNT D_INTEGER,
    RES_ACTIVE_COUNT D_INTEGER,
    RES_DELINQ_COUNT D_INTEGER,
    RES_TOTAL_BALANCE D_CURRENCY,
    RES_DELINQ_BALANCE D_CURRENCY,
    RES_CLOSED_LAST_WEEK D_INTEGER,
    HISTORY_NOT_FOUND D_SMALLINT)
AS
declare variable json_loan d_str_8100;
declare variable loan_count d_integer;
declare variable json_full d_blob_text;
declare variable bls_id d_id;
declare variable k_id d_id;
declare variable c_id d_id;
declare variable k_disb_date d_date;
declare variable fact_rec_no d_integer;
declare variable json_full_str d_str_8100;
declare variable has_blob d_smallint;
declare variable creddate d_date;
declare variable credenddatefact d_date;
declare variable credupdate d_date;
declare variable credsumdebt d_currency;
declare variable credsumoverdue d_currency;
declare variable credpartnertype d_integer;
declare variable kc_id d_id;
declare variable credsum d_currency;
begin
  exception e_default 'Еще не переведено на новый формат КО БКИ';

  select kc_from_k_id, kc_from_c_id, k_disb_date
  from knclient, kontract
  where kc_id = :new_kc_id and
        kc_from_k_id = k_id
  into :k_id, :c_id, creddate;
  k_disb_date = creddate;

  if (k_id is null or c_id is null) then
    exception e_default 'Нет займа KC_ID: ' || coalesce(new_kc_id, '<NULL>');

  select bls_id, has_blob, history_not_found
  from hp_extinfolist_pdn_get_eil(:new_kc_id, current_date, 1)
  into :bls_id, :has_blob, history_not_found;

  json_full = '';

  if (has_blob = 1) then
  begin
    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into :json_full;

    when any do
    begin
      json_full = '';
      history_not_found = 1;
    end
  end

  res_loan_count = 0;
  res_active_count = 0;
  res_total_balance = 0;
  res_delinq_balance = 0;
  res_closed_last_week = 0;
  res_delinq_count = 0;

  if (json_full = '') then
  begin
    suspend;
    exit;
  end

  json_full_str = '';
  if (char_length(json_full) <= 8100) then
    json_full_str = json_full;

  loan_count = -1;
  if (json_full_str <> '' and json_full_str <> '{}') then
  begin
    loan_count = f_json_str_get_count(json_full_str, 'response.basePart.credit');
  end else
  begin
    if (json_full <> '' and json_full <> '{}') then
      loan_count = f_json_blob_get_count(json_full, 'response.basePart.credit');
  end

  fact_rec_no = 0;
  if (loan_count = -1) then loan_count = 0;
  while (fact_rec_no < loan_count) do
  begin
    fact_rec_no = fact_rec_no + 1;

    if (json_full_str <> '')
    then json_loan = f_json_str_get_value(json_full_str, 'response.basePart.credit[' || (fact_rec_no - 1) || ']', '');
    else json_loan = f_json_blob_get_value(json_full, 'response.basePart.credit[' || (fact_rec_no - 1) || ']', '');

    credDate = f_json_str_get_value(json_loan, 'credDate', null);

    if (credDate > k_disb_date) then
      Continue;

    credPartnerType = f_strtonum(f_json_str_get_value(json_loan, 'credPartnerType', 0));
    if (credPartnerType <> 3) then -- если не МФО, не обрабатываем дальше
      Continue;

    credEnddatefact = f_json_str_get_value(json_loan, 'credEnddatefact', null);
    credSumDebt = f_strtonum(f_json_str_get_value(json_loan, 'credSumDebt', 0));
    credSumOverdue = f_strtonum(f_json_str_get_value(json_loan, 'credSumOverdue', 0));
    credSum = f_strtonum(f_json_str_get_value(json_loan, 'credSum', 0));

    if (credUpdate > k_disb_date) then credUpdate = k_disb_date;
    if (credEnddatefact > k_disb_date) then credEnddatefact = k_disb_date;

    if (credUpdate < dateadd(month, -3, k_disb_date) and credSumOverdue = 0) then
      Continue;

    -- учитывать в расчете ПДН
    kc_id = null;
    select kc_id
    from knclient, kontstate, kontract
    where kc_from_c_id = :c_id and
          kc_amount_approved = :credsum and
          kc_from_k_id = ks_from_k_id and
          kc_from_k_id = k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin = :credDate
    rows 1
    into kc_id;

    if (kc_id is null) then
    begin
      res_loan_count = res_loan_count + 1;

      if (credEnddatefact >= k_disb_date) then
      begin
        res_closed_last_week = res_closed_last_week + 1;
        credSumDebt = 0;
        credSumOverdue = 0;
      end

      if (credSumDebt + credSumOverdue > 0) then
        res_active_count = res_active_count + 1;

      if (credsumoverdue > 0) then
        res_delinq_count = res_delinq_count + 1;

      res_total_balance = res_total_balance + credSumDebt + credSumOverdue;
      res_delinq_balance = res_delinq_balance + credsumoverdue;

    end

    kc_id = null;
  end


  Suspend;
end
