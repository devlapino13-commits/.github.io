CREATE PROCEDURE AAREP_EXTSYSMAIN_LIST_PROJECT(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_ONLY_PAY D_INTEGER,
    REP_PROJECT STR_255)
RETURNS (
    ESM_ID ID_TYPE,
    ESM_TRANS_ID D_STR_20,
    ESM_PROJECT D_STR_20,
    ESM_ACCOUNT_PR D_STR_20,
    ESM_ACTION D_STR_20,
    ESM_DATE D_DATE,
    ESM_ACCOUNT D_STR_20,
    ESM_OBJECT_KIND D_INTEGER,
    ESM_OBJECT_ID ID_TYPE,
    ESM_DATETIME D_DATETIME,
    ESM_RESULT_ID D_INTEGER,
    ESM_STATUS D_INTEGER,
    ESM_FULL_SOURCE D_BLOB_TEXT,
    ESM_EXT_DATETIME D_DATETIME,
    ESM_AMOUNT D_CURRENCY,
    ESM_STATUS_STR D_STR_100,
    ESM_OBJECT_NAME D_STR_255,
    OL_ID ID_TYPE,
    REG_NAME D_STR_100,
    REG_ID D_INTEGER,
    ESM_FROM_K_ID ID_TYPE,
    K_KONTRACT_NO D_STR_20,
    PRODUCT_TYPE D_STR_10,
    TERM_NUM D_STR_100,
    TERM_OFFICE D_STR_CODE,
    ISSUING_METHOD D_STR_20)
AS
declare variable db_id d_integer;
declare variable pos d_integer;
begin
  db_id = f_get_db_id();

  for select distinct esm_id, esm_trans_id, esm_project, esm_action, esm_date, esm_account, esm_object_kind,
                      esm_object_id, esm_datetime, esm_result_id, esm_status, esm_full_source, esm_ext_datetime,
                      esm_amount, esm_from_k_id
      from extsysmain
      where esm_date between :rep_date_begin and :rep_date_end and
            esm_project = coalesce(:rep_project, esm_project) and
            esm_project <> 'elsom' and
            esm_project <> 'elcard_disb' and
            esm_project <> 'transfer_on_phone' and
            esm_id <> 987330
      order by esm_id desc
      into :esm_id, :esm_trans_id, :esm_project, :esm_action, :esm_date, :esm_account, :esm_object_kind, :esm_object_id,
           :esm_datetime, :esm_result_id, :esm_status, :esm_full_source, :esm_ext_datetime, :esm_amount,
           :esm_from_k_id
  do
  begin
    esm_account_pr = null;
    term_office = null;
    if (db_id in (39801, 41701) and
        esm_project not in ('kicb', 'moneta', 'paysoft')) then
    begin
      if (esm_project = 'qr') then --JIRA 1561 (в оплатах по qr кодам esm_account = 'оплата ч/з QR-код')
        esm_account = null;
      esm_account_pr = replace(esm_account, '-', '');
      esm_account_pr = substring(esm_account from position(':', esm_account, 5) for 5) || '-' -->>
      || substring(/*'` ' ||*/ esm_account_pr from 1 for position(':', esm_account_pr, 5)); -- a.kim 29.11.2018 закомментрировал задача в jira board-261

      esm_account_pr = replace(esm_account_pr, ':', '');
    end
    else
      esm_account_pr = esm_account;

    term_num = null;
    select field_value
    from hp_extsys_parcer(:esm_full_source, 'terminal_number')
    into :term_num;

    if (esm_project in ('doscredo', 'umai', 'paysoft') and
        term_num <> '') then
    begin
      term_office = (select reg_name
                     from terminals
                     inner join region on reg_unicode = tm_office
                     where tm_terminal_id = :term_num and
                           tm_project = 2 and
                           :esm_date between coalesce(tm_date_begin, '01.01.1900') and coalesce(tm_date_end, '31.12.9999'));

      -- Кыял Абдымаликова попросила выводить номер терминала после знака "-"
      -- pos = position('-', term_num, 1) + 1;
      -- term_num = substring(term_num from pos);
    end


    esm_status_str = (case
                        when esm_status = -3 then 'Ошибка пр обработке'
                        when esm_status = -2 then 'Тестовый платеж'
                        when esm_status = -1 then 'Системная ошибка'
                        when esm_status = 0 then 'В процессе'
                        when esm_status = 1 then 'Обработан'
                      end);

    esm_object_name = '';
    ol_id = null;
    reg_name = '';
    reg_id = 0;
    if ((rep_only_pay = 1 and
        esm_action = 'pay') or (rep_only_pay <> 1)) then
    begin
      if (esm_object_kind = 0) then
      begin
        select c_name_fio
        from clients
        where c_id = :esm_object_id
        into :esm_object_name;
      end

      if (esm_action = 'pay') then
      begin
        select eso_from_ol_id
        from extsysoper
        where eso_from_esm_id = :esm_id and
              exists(select *
                     from operlist ol
                     where ol.ol_id = eso_from_ol_id and
                           ol.is_delete = 0) rows 1
        into :ol_id;
      end

      k_kontract_no = '';
      product_type = '';

      if (esm_from_k_id is not null) then
      begin
        select k_kontract_no, reg_code || ' ' || reg_name, iif(pr_code containing '.ipf.', 'ИПФ', 'Класика'),
               iif(k.k_disb_kind containing '6', 'Карта Мбулак', 'Наличными')
        from kontract k, knaktlist kal, region r, product pr
        where k_id = :esm_from_k_id and
              k_from_kf_id = kal_from_kf_id and
              kal.is_delete = 0 and
              :esm_date between kal_date and kal_date_end and
              kal.kal_new_reg_id = reg_unicode and
              k.k_from_pr_id = pr.pr_id
        into :k_kontract_no, reg_name, :product_type, :issuing_method;
      end

      suspend;
    end
  end
end
