CREATE PROCEDURE HP_SELKONT_CASHARRIVAL(
    K_ID D_ID)
RETURNS (
    RES_PR_CODE D_STR_100,
    RES_IS_IPF D_BOOL,
    RES_AMOUNT_APPROVED D_CURRENCY,
    RES_PREPAYMENT_SUMMA D_CURRENCY,
    RES_IPF_COMMISSION D_CURRENCY,
    RES_CLI_LIST D_STR_2048,
    RES_ERR_TEXT D_STR_255)
AS
begin
  res_err_text = '';
  res_cli_list = '';
  for
    select
      pr_code,
      k_amount_approved,
      coalesce(k_prepayment_summa,0),
      coalesce(pr_ipf_commission,0)
    from kontract
      inner join product
        on pr_id = k_from_pr_id
    where k_id = :k_id
  into
    :res_pr_code,
    :res_amount_approved,
    :res_prepayment_summa,
    :res_ipf_commission
  do
  begin
    res_is_ipf = iif(res_pr_code containing '.ipf.', 1, 0);

    if (not (res_is_ipf = 1 or res_pr_code containing '.kons.')) then
      res_err_text = 'Договор должен быть ИПФ или классика с консультационными услугами';

    if (res_err_text = '') then
      res_cli_list = (
        select
          list(c_id || ',' || c_name_fio, ';')
        from knclient
          inner join clients
            on c_id = kc_from_c_id
        where kc_from_k_id = :k_id
          and kc_amount_approved > 0);

    suspend;
  end
end
