CREATE PROCEDURE AAREP_PLANNING_REPORT_TEMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_AUTO_PERIOD INTEGER,
    REP_AUTO_KOEF DECIMAL(18,7),
    REP_INCLUDE_FIRST_DAY INTEGER)
RETURNS (
    MA_DATE_OPER DATE,
    MA_SUMMA_PLUS_OS DECIMAL(18,4),
    MA_SUMMA_PLUS_PERC DECIMAL(18,4),
    MA_SUMMA_PLUS_ADMIN DECIMAL(18,4),
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_ADMIN_PLUS DECIMAL(18,4),
    MA_ADMIN_MINUS DECIMAL(18,4),
    MA_TRANSFER_PLUS DECIMAL(18,4),
    MA_TRANSFER_MINUS DECIMAL(18,4),
    MA_SUMMA_MINUS_REPEAT DECIMAL(18,4),
    MA_SUMMA_MINUS_NEW DECIMAL(18,4),
    MA_SUMMA_MINIAPPNEW DECIMAL(18,4),
    MA_ADMIN_MINUS_PROCESS DECIMAL(18,4),
    MA_REGION_ID INTEGER,
    MA_DISB_KOLICH INTEGER)
AS
begin
  ma_date_oper = :rep_date_begin - 1 + rep_include_first_day;
  ma_summa_plus_os = 0;
  ma_summa_plus_perc = 0;
  ma_summa_plus_admin = 0;
  ma_summa_minus = 0;
  ma_admin_plus = 0;
  ma_admin_minus = 0;
  ma_transfer_plus = 0;
  ma_transfer_minus = 0;
  ma_summa_minus_repeat = 0;
  ma_summa_minus_new = 0;
  ma_summa_miniappnew = 0;
  ma_admin_minus_process = 0;
  ma_disb_kolich = 0;

  /* Прогнозирование по кредитной части */

  for select ma_region_id, ma_date_oper, sum(ma_summa_plus_os), sum(ma_summa_plus_perc), sum(ma_summa_plus_admin),
             sum(ma_summa_minus), sum(ma_admin_plus), sum(ma_admin_minus), sum(ma_transfer_plus),
             sum(ma_transfer_minus), sum(ma_summa_minus_repeat), sum(ma_summa_minus_new), sum(ma_summa_miniappnew),
             sum(ma_admin_minus_process), sum(ma_disb_kolich)
      from aarep_miniapp_temp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_auto_period, :rep_auto_koef, :rep_include_first_day, 1)
      group by ma_region_id, ma_date_oper
      order by ma_date_oper
      into :ma_region_id, :ma_date_oper, :ma_summa_plus_os, :ma_summa_plus_perc, :ma_summa_plus_admin, :ma_summa_minus,
           :ma_admin_plus, :ma_admin_minus, :ma_transfer_plus, :ma_transfer_minus, :ma_summa_minus_repeat,
           :ma_summa_minus_new, :ma_summa_miniappnew, :ma_admin_minus_process, :ma_disb_kolich
  do
  begin
    /*
    if (ma_summa_minus_new>=ma_summa_miniappnew)
    then ma_summa_miniappnew=0;
    else ma_summa_miniappnew=ma_summa_miniappnew-ma_summa_minus_new;
    */
    suspend;
  end
  ma_disb_kolich = 0;

  /* Планирование админ.части */

  for select ma_date_oper, sum(ma_summa_plus_os), sum(ma_summa_plus_perc), sum(ma_summa_plus_admin),
             sum(ma_summa_minus), sum(ma_admin_plus), sum(ma_admin_minus), sum(ma_transfer_plus),
             sum(ma_transfer_minus), sum(ma_summa_minus_repeat), sum(ma_summa_minus_new), sum(ma_summa_miniappnew),
             sum(ma_admin_minus_process), ma_region_id
      from aarep_miniapp_temp_admin(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_include_first_day)
      group by ma_date_oper, ma_region_id
      into :ma_date_oper, :ma_summa_plus_os, :ma_summa_plus_perc, :ma_summa_plus_admin, :ma_summa_minus, :ma_admin_plus,
           :ma_admin_minus, :ma_transfer_plus, :ma_transfer_minus, :ma_summa_minus_repeat, :ma_summa_minus_new,
           :ma_summa_miniappnew, :ma_admin_minus_process, :ma_region_id
  do
  begin
    suspend;
  end

end
