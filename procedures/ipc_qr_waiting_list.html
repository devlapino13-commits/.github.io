CREATE PROCEDURE IPC_QR_WAITING_LIST
RETURNS (
    CS_ID D_ID,
    IQS_SHOP_NAME VARCHAR(275))
AS
declare variable rd_id_tsp d_integer;
begin
  rd_id_tsp = f_get_rd_id('AgreementKinds.tsp');
  for select cs_id, cs_name || '||' || cs_id
      from clientshops
      where cs_qr_data is null and
            cs_is_delete = 0 and
            cs_from_rd_id = :rd_id_tsp and
            exists(select *
                   from agreements
                   where ag_object_kind = 0 and
                         ag_object_id = cs_from_c_id and
                         ag_status = 1 and
                         ag_from_rd_id = :rd_id_tsp and
                         f_edoc_status('TSP.Agreement', ag_id, current_date) = 'signed') and
            exists(select *
                   from ipc_qr_app
                   inner join ipc_qr_app_shops on iqs_from_iqa_id = iqa_id and
                         iqs_from_cs_id = cs_id and
                         iqs_has_qr = 0
                   where iqa_status = 1)
      into :cs_id, :iqs_shop_name
  do
  begin
    suspend;
  end
end
