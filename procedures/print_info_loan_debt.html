CREATE PROCEDURE PRINT_INFO_LOAN_DEBT(
    ILC_ID D_ID)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    K_SUM_OS D_CURRENCY,
    K_SUM_PERC D_CURRENCY,
    K_SUM_NSP D_CURRENCY,
    K_SUM_TOTAL D_CURRENCY)
AS
declare variable k_id id_type;
declare variable rep_date d_date;
begin
  k_sum_os = 0;
  k_sum_perc = 0;
  k_sum_nsp = 0;
  k_sum_total = 0;

  select ilc_date
  from info_loan_client
  where ilc_id = :ilc_id
  into rep_date;

  for select ild_from_k_id
      from info_loan_debt
      where ild_from_ilc_id = :ilc_id
      into k_id
  do
  begin
    select k_kontract_no
    from kontract
    where k_id = :k_id
    into k_kontract_no;

    select sum(zad_os),
           sum(sh_calc_perc)
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    into k_sum_os,
         k_sum_perc;

    select res
    from fhp_round_bank(:k_sum_perc * 2.00 / 100, 0)
    into k_sum_nsp;

    k_sum_total = k_sum_os + k_sum_perc + k_sum_nsp;

    suspend;
  end
end
