CREATE PROCEDURE HP_CLIENTS_PERMISSIONS_DIFF(
    C_ID_FROM DECIMAL(18,0),
    C_ID_TO DECIMAL(18,0),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PM_OBJECT_ID DECIMAL(18,0),
    PM_SUBJECT_KIND TYPE OF COLUMN PERMISSIONS.PM_SUBJECT_KIND,
    PM_SUBJECT_ID TYPE OF COLUMN PERMISSIONS.PM_SUBJECT_ID,
    REP_SUBJECT_NAME VARCHAR(255),
    REP_SUBJECT_MEMO VARCHAR(255),
    PM_DATE_BEGIN DATE,
    PM_DATE_END DATE,
    PM_ACCESS_KIND INTEGER,
    PM_UN_KIND INTEGER,
    RN_ONLY_SUPERUSER INTEGER)
AS
declare variable un_id_2 decimal(18,0);
declare variable un_id_1 decimal(18,0);
declare variable uniqtab_username integer;
declare variable uniqtab_rolename integer;
declare variable uniqtab_menulist integer;
declare variable uniqtab_permactions integer;
declare variable uniqtab_region integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;
  select ukt_id
  from hp_get_uniqtable_id('Menulist')
  into :uniqtab_menulist;
  select ukt_id
  from hp_get_uniqtable_id('PermActions')
  into :uniqtab_permactions;
  select ukt_id
  from hp_get_uniqtable_id('Region')
  into :uniqtab_region;

  rn_only_superuser = 0;
  if (c_id_from is null) then
    exit;
  if (c_id_to is null) then
    exit;

  select un_id
  from username
  where un_from_c_id = :c_id_from rows 1
  into :un_id_1;

  select un_id
  from username
  where un_from_c_id = :c_id_to rows 1
  into :un_id_2;

  pm_un_kind = -1;
  if ((c_id_to = 0 or (c_id_to > 0 and
      un_id_2 is not null)) and
      (un_id_1 is not null)) then
  begin
    pm_object_id = un_id_1;
    pm_subject_kind = uniqtab_permactions;
    for select pa_id, pa_name, pa_memo
        from permactions
        where pa_parent_id is null
        into :pm_subject_id, :rep_subject_name, :rep_subject_memo
    do
    begin
      if (exists(select *
                 from hp_permission_access_check(:un_id_1, 'PermActions', :pm_subject_id, '')
                 where has_access = 1)) then
      begin
        pm_access_kind = -1;
        pm_date_begin = rep_date_begin;
        pm_date_end = rep_date_end;
        suspend;
        pm_access_kind = 1;
        pm_date_begin = rep_date_end + 1;
        pm_date_end = null;
        suspend;
      end
    end
    if (c_id_to = 0) then
      exit;
  end
  else
  begin
    if ((un_id_1 is null) and
        (c_id_to > 0 and
        un_id_2 is null)) then
    begin
      rep_subject_name = 'Передающий и принимающий не являются пользователеми.';
      suspend;
      exit;
    end

    if (un_id_1 is null) then
    begin
      rep_subject_name = 'Передающий не является пользователем.';
      suspend;
      exit;
    end

    if (c_id_to > 0 and
        un_id_2 is null) then
    begin
      rep_subject_name = 'Принимающий не является пользователем.';
      suspend;
      exit;
    end
  end

  pm_un_kind = 1;
  pm_access_kind = 1;
  pm_object_id = un_id_2;

  for with tab
      as -- Права передающего
      (select pm_subject_kind, pm_subject_id, 1 as diff,
              maxvalue(:rep_date_begin, coalesce(pm_date_begin, '01.01.2000')) as access_begin,
              minvalue(:rep_date_end, coalesce(pm_date_end, '31.12.2999')) as access_end
       from permissions
       where pm_object_kind = :uniqtab_username and
             pm_object_id = :un_id_1 and
             pm_is_delete = 0 and
             pm_access_kind > 0 and
             (coalesce(pm_date_end, '31.12.2999') >= current_date)
       union
-- Права принимающего
       select pm_subject_kind, pm_subject_id, -1 as diff, null as access_begin, null as access_end
       from permissions
       where pm_object_kind = :uniqtab_username and
             pm_object_id = :un_id_2 and
             pm_is_delete = 0 and
             pm_access_kind > 0 and
             (coalesce(pm_date_begin, '01.01.2000') <= :rep_date_begin) and
             (coalesce(pm_date_end, '31.12.2999') >= :rep_date_end))
      select pm_subject_kind, pm_subject_id, max(access_begin) as access_begin, max(access_end) as access_end
      from tab
      group by pm_subject_kind, pm_subject_id
      having sum(diff) > 0
      into :pm_subject_kind, :pm_subject_id, :pm_date_begin, :pm_date_end
  do
  begin
    rn_only_superuser = 0;
    if (pm_subject_kind = :uniqtab_rolename) then
    begin
      if (exists(select *
                 from permissions, menulist
                 where pm_object_kind = :uniqtab_rolename and
                       pm_object_id = :pm_subject_id and
                       pm_subject_kind = :uniqtab_menulist and
                       pm_subject_id = ml_id and
                       pm_is_delete = 0 and
                       ml_secure_kind = -1)) then
        rn_only_superuser = 1;

      if (rn_only_superuser = 0) then
      begin
        if (exists(select *
                   from permissions, permactions
                   where pm_object_kind = :uniqtab_rolename and
                         pm_object_id = :pm_subject_id and
                         pm_subject_kind = :uniqtab_permactions and
                         pm_subject_id = pa_id and
                         pm_is_delete = 0 and
                         pa_secure_kind = -1)) then
          rn_only_superuser = 1;
      end
    end
    if ((pm_subject_kind = :uniqtab_menulist) and
        (rn_only_superuser = 0)) then
    begin
      if (exists(select *
                 from menulist
                 where ml_secure_kind = -1 and
                       ml_id = :pm_subject_id)) then
        rn_only_superuser = 1;
    end
    if ((pm_subject_kind = :uniqtab_permactions) and
        (rn_only_superuser = 0)) then
    begin
      if (exists(select *
                 from permactions
                 where pa_secure_kind = -1 and
                       pa_id = :pm_subject_id)) then
        rn_only_superuser = 1;
    end

    if (pm_subject_kind = :uniqtab_rolename) then
    begin
      select rn_name, rn_memo
      from rolename
      where rn_id = :pm_subject_id
      into :rep_subject_name, :rep_subject_memo;
    end
    if (pm_subject_kind = :uniqtab_menulist) then
    begin
      select ml_caption, ml_description
      from menulist
      where ml_id = :pm_subject_id
      into :rep_subject_name, :rep_subject_memo;
    end
    if (pm_subject_kind = :uniqtab_permactions) then
    begin
      select pa_name, pa_memo
      from permactions
      where pa_id = :pm_subject_id
      into :rep_subject_name, :rep_subject_memo;
    end
    if (pm_subject_kind = :uniqtab_region) then
    begin
      select reg_code || ' ' || reg_name, reg_address
      from region
      where reg_unicode = :pm_subject_id
      into :rep_subject_name, :rep_subject_memo;
    end
    suspend;
  end
end
