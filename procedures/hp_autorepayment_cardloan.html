CREATE PROCEDURE HP_AUTOREPAYMENT_CARDLOAN(
    CS_ID D_INTEGER = null)
AS
declare variable csc_id d_integer;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable k_id d_id;
declare variable c_id d_id;
declare variable un_id d_id;
declare variable reg_id d_integer;
declare variable amount d_currency;
declare variable cs_amount d_currency;
declare variable dt_curr d_date;
declare variable acc_dt d_str_20;
declare variable ol_id d_id;
begin
  dt_curr = current_date;
  un_id = 1000000000000;
  csc_id = f_get_cardsynccfg_id('check_balance_on_autorepayment');

  if (cs_id > 0) then
  begin
    amount = coalesce((select arc_amount from autorepayment_cardloan where arc_from_cs_id = :cs_id), 0);
    select
      cs_card_id,
      cs_from_cca_id,
      cs_from_k_id,
      cs_from_c_id,
      cs_office_id,
      cast(f_json_str_get_value(cs_answer_json, 'data.availableAmount', 0) as d_currency)
    from cardsync
    where cs_id = :cs_id
      and cs_from_csc_id = :csc_id
      and cs_status = 2
    into
      :card_id,
      :cca_id,
      :k_id,
      :c_id,
      :reg_id,
      :cs_amount;
    if (row_count = 1) then
      in autonomous transaction do
      begin
        update cardsync set
          cs_amount = :cs_amount
        where cs_id = :cs_id;

        update autorepayment_cardloan set
          arc_cs_amount = :cs_amount,
          arc_is_completed = 1
        where arc_from_cs_id = :cs_id;

        if (cs_amount > 0 and amount > 0) then
        begin
          if (amount > cs_amount) then
            amount = cs_amount;

          acc_dt = (
            select al_id from acc_list
            where al_from_reg_id = :reg_id
              and al_currency = '417'
              and al_account = '20001/1');
          ol_id = (
            select
              ol_id
            from hp_do_repayment (
              :k_id,
              0,              -- c_id
              :dt_curr,       -- rep_date
              :amount,        -- repay_amount
              1,              -- repay_to_os
              1,              -- repay_to_perc
              1,              -- repay_to_fine
              1,              -- repay_to_kommis
              1,              -- need_save
              0,              -- is_full_repay
              0,              -- is_writeoff
              :reg_id,        -- repay_fact_office_id
              3,              -- repay_dt_account_kind
              :acc_dt,        -- repay_dt_account_no
              1,              -- repay_curr_rate
              :un_id,         -- repay_user_id
              :c_id,          -- repay_document_client
              null,           -- repay_doc_client_name_reason
              0,              -- repay_manager_c_id
              0));             -- repay_from_rd_id
          if (ol_id > 0) then
          begin
            update operlist set
              ol_memo = ol_memo || coalesce(f_format(' номер ID %1 (сч.%2)', :card_id, :cca_id), '')
            where :ol_id in (ol_id, ol_parent_id);

            update autorepayment_cardloan set
              arc_from_ol_id = :ol_id,
              arc_ol_date = :dt_curr
            where arc_from_cs_id = :cs_id;
          end
        end
      end
  end
  else
  begin
    for
      select
        clo_card_id,
        clo_from_c_id,
        k_id,
        k_last_reg_id
      from cardlistloan
        inner join kontract
          on k_id = clo_from_k_id
            and (k_fact_closed_date is null or k_fact_closed_ol_id is null)
            and exists (
                  select *
                  from operkred
                  where ok_from_k_id = k_id
                    and ok_sh_date = :dt_curr
                    and ok_ol_is_active = 1
                    and ok_sh_id > 0
                    and ok_type = 1)
            and not exists (
                  select *
                  from autorepayment_cardloan
                  where arc_from_k_id = k_id
                    and arc_is_completed = 1
                    and arc_amount >= arc_cs_amount)
    into
      :card_id,
      :c_id,
      :k_id,
      :reg_id
    do
    begin
      amount = coalesce((
        select
          round(sum(zad_os) + round(1.02 * sum(zad_perc), 1) + sum(sh_calc_fine), 0)
        from aarep_make_zad_list(:k_id, :dt_curr, 1)
        where sh_date <= :dt_curr), 0);

      if (amount > 0) then
        in autonomous transaction do
        begin
          insert into cardsync(
            cs_from_csc_id,
            cs_card_id,
            cs_from_cca_id,
            cs_object_kind,
            cs_object_id,
            cs_request_json,
            cs_office_id,
            cs_from_k_id,
            cs_from_c_id,
            cs_from_un_id)
          select first 1
            :csc_id,
            :card_id,
            cs_from_cca_id,
            9,
            :k_id,
            f_format('{"card_id": %1}', cs_card_id),
            :reg_id,
            :k_id,
            :c_id,
            :un_id
          from cardsync
          where cs_card_id = :card_id
            and cs_from_k_id = :k_id
          order by cs_id desc
          returning cs_id into :cs_id;

          insert into autorepayment_cardloan (
            arc_from_k_id,
            arc_from_reg_id,
            arc_from_cs_id,
            arc_amount )
          values (
            :k_id,
            :reg_id,
            :cs_id,
            :amount );
        end
    end
  end
end
