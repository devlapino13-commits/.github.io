CREATE PROCEDURE HP_CLOSEMONTHMAIN_INIT(
    REP_PERIOD D_INTEGER)
AS
declare variable rep_date_begin d_date;
declare variable rep_date_end d_date;
begin
  rep_date_begin = cast('01.' || right(rep_period, 2) || '.' || left(rep_period, 4) as date);
  rep_date_end = f_date_month_end(rep_date_begin);

  delete from closemonthlist
  where 1 = 1;

  update closemonthmain
  set cmm_status = 0,
      cmm_date_begin = :rep_date_begin,
      cmm_date_end = :rep_date_end,
      cmm_res_data = null
  where 1 = 1;
end
