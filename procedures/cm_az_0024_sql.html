CREATE PROCEDURE CM_AZ_0024_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable PR_ID ID_TYPE;
declare variable PR_CODE STR_100;
declare variable C_ID ID_TYPE;
declare variable REL_C_ID ID_TYPE;
declare variable KC_CIRCLE ID_TYPE;
declare variable PORUCH_K_ID ID_TYPE;
begin
  for select KC_FROM_C_ID, KC_AMOUNT_APPROVED, KC_CIRCLE, K_DISB_DATE, K_FROM_PR_ID, PR_CODE
      from KONTRACT K, PRODUCT PR, KNCLIENT KC
      where K_ID = :K_ID and
            K_FROM_PR_ID = PR_ID and
            PR_ID not in (67, 66, 65, 64, 57, 56, 55, 54) and
            K_ID = KC_FROM_K_ID and
            KC_AMOUNT_APPROVED > 0
      into C_ID, KC_AMOUNT, KC_CIRCLE, KF_DISB_DATE, PR_ID, PR_CODE
  do
  begin
    --Клиент не может получить займ по другим продуктам если является родственником другого активного клиента
    for select CR.CREL_FROM_C_ID
        from CLIENTRELATION CR
        where CR.CREL_OTHER_C_ID = :C_ID and
              CR.CREL_IS_ACTIVE = 1
        into REL_C_ID
    do
    begin
      if (exists(select *
                 from KNCLIENT KC, KONTSTATE KS, KONTRACT K
                 where KC_FROM_C_ID = :REL_C_ID and
                       KC_AMOUNT_APPROVED > 0 and
                       KC_FROM_K_ID = KS_FROM_K_ID and
                       KS_OL_IS_DELETE = 0 and
                       KS_STATE = 1 and
                       :KF_DISB_DATE between KS_DATE_BEGIN and KS_DATE_END and
                       KS_FROM_K_ID = K_ID and
                       K_FROM_PR_ID in (
                                    --68, 69, 70, 71, 72, 73 обновил от 21.11.18
                                    68, 69, 70, 71, 72, 73, 74, 75, 76, 77,
                                        78, 79))) then
      begin
        --Комментарий: “Родственники клиентов активных кредитов по продуктам “AQRO” могут брать кредиты
        --только по продуктам “Ail? Istehlak”. Для исключений по клиенту просьба обратиться к Кредитному департаменту.”
        res_error_text = '”AQRO” məhsulu üzrə aktiv müştərilərin ailə üzvləri yalnız “Ailə İstehlak” məhsulu üzrə müraciyəti mümkündür. İstisna hallar üçün Kredit Departamentinə müraciyət edin.';
        res_error_is_critical = 1;
        suspend;
      end
      --Клиент не может получить займ по другим продуктам если является родственником поручителя другого активного клиента
      for select distinct Z.Z_FROM_K_ID
          from ZALOG Z
          where Z_FROM_C_ID = :REL_C_ID and
                Z_TYPE = 0
          into :PORUCH_K_ID
      do
      begin
        if (exists(select *
                   from KONTSTATE KS, KONTRACT K
                   where KS_FROM_K_ID = :PORUCH_K_ID and
                         KS_OL_IS_DELETE = 0 and
                         KS_STATE = 1 and
                         :KF_DISB_DATE between KS_DATE_BEGIN and KS_DATE_END and
                         KS_FROM_K_ID = K_ID and
                         K_FROM_PR_ID in (
                                      --68, 69, 70, 71, 72, 73
                                      68, 69, 70, 71, 72, 73, 74, 75, 76, 77,
                                          78, 79))) then
        begin
          --Комментарий: “Родственники поручителей активных клиентов по продуктам “AQRO” могут брать кредиты
          --только по продуктам “Ailə İstehlak”.”
          res_error_text = '”AQRO” məhsulu üzrə zamin qismində çıxış edən şəxslərin ailə üzvləri “Ailə İstehlak” məhsulunun qaydalarına uyğun kredit üçün müraciət edə bilər.';
          res_error_is_critical = 1;
          suspend;
        end
      end
      if (PR_ID in (
                --68, 69, 70, 71, 72, 73
                68, 69, 70, 71, 72, 73, 74, 75, 76, 77,
                    78, 79) and
          exists(select *
                 from HR_PSTATUS PS
                 where PS.PS_FROM_C_ID = :REL_C_ID and
                       PS.PS_IS_WORKING = 1 and
                       :KF_DISB_DATE between PS.PS_DATE_BEGIN and PS.PS_DATE_END)) then
      begin
        --Комментарий: “Родственники Сотрудников компании не имеют право брать кредит по продукту “AQRO”
        res_error_text = 'Təşkilatın əməkdaşlarının ailə üzvləri “AQRO” məhsulu üzrə kredit ala bilməzlər.';
        res_error_is_critical = 1;
        suspend;
      end
    end
  end
end
