CREATE PROCEDURE HP_PLACES_GET_TIMEZONE_C(
    C_ID D_ID,
    PHONE_NO D_STR_100)
RETURNS (
    REG_TIMEZONE D_SMALLINT,
    PH_TIMEZONE D_SMALLINT,
    MSK_CALL_TIME_START D_TIME,
    MSK_CALL_TIME_END D_TIME)
AS
declare variable pl_id d_id;
declare variable msk_time_shift d_smallint;
begin
  select adl_from_pl_id
  from addrclient, addrlist
  where adl_id = adc_from_adl_id
    and adc_from_c_id = :c_id
    and adc_address_type = 1
  into pl_id;

  select timezone
  from hp_places_get_timezone_pl(:pl_id)
  into reg_timezone;

  if (reg_timezone is null) then
    exit;

  ph_timezone = case
                  when phone_no starting with '996' then 6
                  when phone_no starting with '998' then 5
                  when phone_no starting with '992' then 5
                  when phone_no starting with '994' then 4
                  when phone_no starting with '77' then 5
                  else reg_timezone
                end;

  msk_time_shift = 3 - minvalue(reg_timezone, ph_timezone);
  msk_call_time_start = dateadd(hour, iif(msk_time_shift < -9, -9, msk_time_shift), cast('09:00' as time));

  msk_time_shift = 3 - maxvalue(reg_timezone, ph_timezone);
  msk_call_time_end = dateadd(hour, iif(msk_time_shift > 3, 3, msk_time_shift), cast('20:00' as time));

  suspend;
end
