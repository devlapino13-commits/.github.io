CREATE PROCEDURE HP_SUD_LOANBLS_ADD(
    SMD_ID D_ID NOT NULL,
    SLC_ID D_ID,
    DOC_KIND D_INTEGER NOT NULL,
    OBJ_KIND D_INTEGER NOT NULL,
    OBJ_ID D_ID NOT NULL,
    BLS_ID D_ID NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_SLD_ID D_ID,
    RES_SLB_ID D_ID)
AS
begin
  if (slc_id = 0) then
    slc_id = null;

  select res_sld_id
  from hp_sud_loandocs_add(:smd_id, :slc_id, :doc_kind, :un_id)
  into :res_sld_id;

  select slb_id
  from sud_loanbls
  where slb_from_sld_id = :res_sld_id and
        slb_from_bls_id = :bls_id and
        slb_object_kind = :obj_kind and
        slb_object_id = :obj_id and
        slb_is_delete = 0
  into res_slb_id;

  if (res_slb_id is null) then
  begin
    insert into sud_loanbls (slb_from_sld_id, slb_from_bls_id, slb_object_kind, slb_object_id, slb_from_un_id)
    values (:res_sld_id, :bls_id, :obj_kind, :obj_id, :un_id)
    returning slb_id
    into res_slb_id;
  end

  suspend;
end
