CREATE PROCEDURE GET_AMORT_INFO(
    BILL_ID DECIMAL(18,0),
    BILL_LINE_NO INTEGER,
    DD DATE)
RETURNS (
    GAI_OS DECIMAL(18,4),
    GAI_AMORT DECIMAL(18,4),
    GAI_CALC_TYPE INTEGER,
    GAI_IS_STARTED SMALLINT,
    GAI_START_YEAR INTEGER,
    GAI_START_MONTH INTEGER,
    GAI_LIFETIME INTEGER,
    GAI_DONTCALC INTEGER)
AS
declare variable kolich integer;
begin
  /* Procedure Text */
  gai_start_year = 0;
  gai_start_month = 0;
  select distinct oam_calc_type, oam_start_year, oam_start_month
  from operamormain
  where oam_bill_id = :bill_id and
        oam_bill_line_no = :bill_line_no and
        oam_is_delete = 0 rows 1
  into :kolich, :gai_start_year, :gai_start_month;

  gai_os = 0;
  gai_amort = 0;
  gai_lifetime = 0;
  gai_dontcalc = 0;
  if (kolich is not null) then
  begin
    gai_is_started = 1;
    gai_calc_type = kolich;
    select sum(oal_summa_os), sum(oal_summa_amort), sum(oal_lifetime_month), sum(oal_dont_calc)
    from operamorlist
    where oal_is_delete = 0 and
          oal_bill_id = :bill_id and
          oal_bill_line_no = :bill_line_no
    into :gai_os, :gai_amort, :gai_lifetime, :gai_dontcalc;
    if (gai_os is null) then
      gai_os = 0;
    if (gai_amort is null) then
      gai_amort = 0;
    if (gai_lifetime is null) then
      gai_lifetime = 0;
    if (gai_dontcalc is null) then
      gai_dontcalc = 0;
  end
  else
  begin
    gai_is_started = 0;
    select distinct mc_is_calc_amort, mc_lifetime
    from matcateg, operbill
    where mc_id = ob_from_mc_id and
          ob_bill_id = :bill_id and
          ob_bill_line_no = :bill_line_no and
          ob_date_begin <= :dd and
          ob_date_end >= :dd and
          ob_is_delete = 0
    into :gai_calc_type, :gai_lifetime;
    if (gai_calc_type is null) then
      gai_calc_type = 0;
    if (gai_lifetime is null) then
      gai_lifetime = 0;
  end
  suspend;

end
