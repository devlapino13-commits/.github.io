CREATE PROCEDURE AAREP_CLIENT_CARD_NUM(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    C_PASS_NO VARCHAR(30),
    C_PINCODE D_STR_20,
    C_PASS_DATE D_DATE,
    C_PASS_DATE_EXPIRE D_DATE,
    ADL_ADDRESS_STR D_STR_255,
    CARD_NUMBER D_STR_100,
    K_DISB_DATE D_DATE)
AS
declare variable C_ID D_ID;
declare variable K_ID D_ID;
begin
  k_disb_date = null;
  c_id = 0;
  k_id = 0;

  for select kc.kc_from_c_id,
             ks.ks_date_begin,
             ks.ks_from_k_id
      from kontstate ks
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            kc.kc_amount_approved > 0
      into :c_id,
           :k_disb_date,
           :k_id
  do
  begin
    card_number = null;

    select em.esm_account
    from extsysmain em
    where em.esm_action = 'addCredit' and
          em.esm_project in ('millicard_biz_disb','millicard_disb')and
          em.esm_from_k_id = :k_id and
          em.esm_status = 1
    into :card_number;

    c_name_fio = null;
    c_birth_date = null;
    c_pass_no = null;
    c_pincode = null;
    c_pass_date = null;
    c_pass_date_expire = null;

    select c.c_name_fio,
           c.c_birth_date,
           c.c_pass_no,
           c.c_pincode,
           c.c_pass_date,
           c.c_pass_date_expire
    from clients c
    where c.c_id = :c_id
    into :c_name_fio,
         :c_birth_date,
         :c_pass_no,
         :c_pincode,
         :c_pass_date,
         :c_pass_date_expire;

    adl_address_str = null;

    select adl.adl_address_str
    from addrclient adc
    inner join addrlist adl on adc.adc_from_adl_id = adl.adl_id
    where adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 1
    into :adl_address_str;

    suspend;
  end
end
