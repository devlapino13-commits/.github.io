CREATE PROCEDURE HP_PACKET_1C_IMP_BANK_INSERT(
    PCB_DATE DATE,
    PCB_DOC_NO VARCHAR(50),
    PCB_DETAIL VARCHAR(255),
    PCB_SIGN SMALLINT,
    PCB_ACC_DT_CODE VARCHAR(50),
    PCB_ACC_KT_CODE VARCHAR(50),
    PCB_AMOUNT_SUM DECIMAL(18,4),
    PCB_AMOUNT_VAL DECIMAL(18,4),
    PCB_AMOUNT_TAX_DT DECIMAL(18,4),
    PCB_AMOUNT_TAX_KT DECIMAL(18,4),
    PCB_FROM_CU_ID VARCHAR(3),
    PCB_MEMO VARCHAR(1024))
AS
begin
  insert into packet_1c_imp_bank (pcb_date, pcb_doc_no, pcb_detail, pcb_sign, pcb_acc_dt_code, pcb_acc_kt_code,
                                  pcb_amount_sum, pcb_amount_val, pcb_amount_tax_dt, pcb_amount_tax_kt, pcb_from_cu_id,
                                  pcb_memo)
  values (:pcb_date, :pcb_doc_no, :pcb_detail, :pcb_sign, :pcb_acc_dt_code, :pcb_acc_kt_code, :pcb_amount_sum,
          :pcb_amount_val, :pcb_amount_tax_dt, :pcb_amount_tax_kt, :pcb_from_cu_id, :pcb_memo);
end
