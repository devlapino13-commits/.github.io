CREATE PROCEDURE PRINT_KONT_DOCS(
    PRINT_K_ID DECIMAL(18,0))
RETURNS (
    OL_DOC_NO INTEGER,
    OL_DATE DATE,
    DOC_TYPE VARCHAR(20),
    OK_OS DECIMAL(18,4),
    OK_PERC DECIMAL(18,4),
    OK_FINE DECIMAL(18,4),
    OK_KOMMIS DECIMAL(18,4))
AS
declare variable ol_parent_id decimal(18,0);
declare variable ol_id decimal(18,0);
begin
  for select ol.ol_id, ol.ol_parent_id, ol.ol_doc_no, ol.ol_date,
             (
             case ok_type
               when 8 then 'Расходный'
               when 4 then 'Приходный'
             end) as doc_type,
             sum(ok_os), sum(ok_perc), sum(ok_fine), sum(ok_kommis)
      from operkred, operlist ol
      where ok_ol_is_active = 1 and
            ok_from_ol_id = ol.ol_id and
            ok_type in (8, 4) and
            ok_from_k_id = :print_k_id and
            ok_from_c_id = 0
      group by ol.ol_id, ol.ol_parent_id, ol_doc_no, ol.ol_date, ok_type
      into :ol_id, :ol_parent_id, :ol_doc_no, :ol_date, :doc_type, :ok_os, :ok_perc, :ok_fine, :ok_kommis
  do
  begin
    if (ol_parent_id is not null) then
    begin
      select ol_doc_no
      from operlist ol
      where ol.ol_id = :ol_parent_id
      into :ol_doc_no;
    end

    suspend;
  end
end
