CREATE PROCEDURE FSROBOT_SQL_EXECUTOR(
    COMMAND_KIND VARCHAR(255))
AS
declare variable rep_params varchar(255);
declare variable fel_id integer;
declare variable is_success integer;
begin
-- Процедура вызывается из через сервис по графику
-- Выполняет SQL команды по заданным фильтрам (Event_Name)

-- Пример создания записи
--  insert into FSROBOT_EVENTLIST (FEL_EVENT_NAME, FEL_COMMAND_KIND, FEL_PARAMS, FEL_FROM_UN_ID)
--  values ('Exec_SQL', :COMMAND_KIND, 'update clients set ....', :UN_ID)

  for select fel_id, fel_params
      from fsrobot_eventlist
      where fel_event_name = 'Exec_SQL' and
            fel_command_kind = :command_kind
      into :fel_id, :rep_params
  do
  begin
    is_success = 1;
    if (rep_params <> '') then
    begin
      execute statement :rep_params;
      when any do
      begin
        is_success = 0;
      end
    end
-- Удаляем задачу из списка, она уже исполнена
    if (is_success = 1) then
    begin
      delete from fsrobot_eventlist
      where fel_id = :fel_id;
    end
  end
end
