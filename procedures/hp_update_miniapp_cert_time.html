CREATE PROCEDURE HP_UPDATE_MINIAPP_CERT_TIME(
    MA_ID INTEGER,
    MANAGER_ID DECIMAL(18,0),
    PLAN_DISB_TIME TIMESTAMP,
    UN_ID DECIMAL(18,0),
    DISB_DATE DATE)
AS
declare variable orig_manager_id decimal(18,0);
declare variable orig_plan_disb_time timestamp;
declare variable ma_number integer;
declare variable last_date_app date;
declare variable last_un_id decimal(18,0);
declare variable ma_date_oper date;
begin
  if (manager_id is null) then
    manager_id = 0;

  select ma_number, ma_manager_id, ma_plan_disb_time, ma_date_app, ma_from_un_id, ma_date_oper
  from miniapp
  where ma_id = :ma_id and
        ma_date_end >= '31.12.2999'
  order by ma_id desc, ma_number desc
  into :ma_number, :orig_manager_id, :orig_plan_disb_time, last_date_app, last_un_id, ma_date_oper;

  if (orig_manager_id is null) then
    orig_manager_id = 0;
  if (manager_id = 0) then
    manager_id = orig_manager_id;
  if (disb_date is null) then
    disb_date = ma_date_oper;
  else
  if (disb_date <> ma_date_oper) then
    plan_disb_time = null;

  if ((manager_id <> orig_manager_id) or (coalesce(orig_plan_disb_time, '01.01.1900') <> coalesce(plan_disb_time, '01.01.1900'))) then
  begin
    if (coalesce(un_id, 0) = 0) then
    begin
      un_id = null;
      select un_id
      from username
      where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN')
      into :un_id;
      if (un_id is null) then
        un_id = 1000000000001;
    end
    if (last_date_app <> current_date or last_un_id <> un_id) then
    begin
      insert into miniapp (ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay,
                           ma_percent, ma_kolich_client, ma_memo, ma_is_active, ma_is_auto, ma_summa_minus,
                           ma_date_save, ma_from_un_id, ma_office_id, ma_currency_id, ma_no, ma_percent_category,
                           ma_date_end, ma_manager_id, ma_plan_disb_time, ma_expert_id, ma_repay_day, ma_source_system,
                           ma_kommis_perc, ma_refund, ma_add_info)
      select ma_id, -1, current_date, :disb_date, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay, ma_percent,
             ma_kolich_client, ma_memo, ma_is_active, ma_is_auto, ma_summa_minus, ma_date_save, :un_id, ma_office_id,
             ma_currency_id, ma_no, ma_percent_category, '31.12.9999', :manager_id, :plan_disb_time, ma_expert_id,
             ma_repay_day, ma_source_system, ma_kommis_perc, ma_refund, ma_add_info
      from miniapp
      where ma_id = :ma_id and
            ma_number = :ma_number;
    end
    else
    begin
      update miniapp
      set ma_manager_id = :manager_id,
          ma_plan_disb_time = :plan_disb_time,
          ma_date_oper = :disb_date
      where ma_id = :ma_id and
            ma_number = :ma_number;
    end
  end
end
