CREATE PROCEDURE HP_EPAY_DOC_DISB_OPERBOOK(
    OL_ID D_ID)
AS
declare variable ag_id d_integer;
declare variable epay_amount d_currency;
declare variable epay_tax_fl d_currency;
declare variable non_resident d_integer;
declare variable loyal_acc_id d_id;
declare variable curr_saldo d_currency;
declare variable reg_id d_integer;
declare variable ol_is_active d_integer;
declare variable ob_detail d_str_long;
declare variable ol_date d_date;
declare variable acc_id d_id;
declare variable ob_amount d_currency;
declare variable acc_code d_str_code;
declare variable al_id d_str_code;
begin
  select ol_date, ol_from_reg_id, 1 - is_delete
  from operlist
  where ol_id = :ol_id
  into ol_date, reg_id, ol_is_active;

  if (ol_date is null) then
    exception e_default 'ol_id is not found';

  for select ogr_from_ag_id, sum(ogr_amount), sum(ogr_tax_fl)
  from operagr
  where ogr_from_ol_id = :ol_id
  group by ogr_from_ag_id
  into ag_id, epay_amount, epay_tax_fl
  do
  begin
    loyal_acc_id = null;
    non_resident = null;

    select epd_non_resident
    from epay_list, epay_doc
    where epl_from_ol_id = :ol_id and
          epd_from_epl_id = epl_id
    into non_resident;
    if (non_resident is null) then
      exception e_default 'non_resident value not defined';

    select acc_id
    from get_ref_account_agreements('Расчеты.ЛоялПартнеры', '', :ag_id, :non_resident, :ol_date)
    into :loyal_acc_id;

    if (loyal_acc_id is null) then
      exception e_default 'Account not found';

    -- проверяем, есть ли другой счет отличающийся по резидентству
    acc_id = null;
    for select acc_id
    from ref_account
    where acc_object_kind = f_objkind('agreements', 'id') and
          acc_object_id = :ag_id and
          acc_non_resident = 1 - :non_resident
    into acc_id
    do
    begin
      -- если счет есть, проверяем остаток
      curr_saldo = 0;
      if (acc_id is not null) then
        curr_saldo = f_acc_bal(acc_id, ol_date);
  
      -- если есть остаток на другом счете, переносим на новый счет
      if (curr_saldo <> 0) then
      begin
        ob_detail = 'Перенос сальдо из-за смены резиденства участника';
        execute procedure hp_do_kn_operbook_ins(:ol_id, null, :acc_id, :loyal_acc_id, null, null, 0, 0, 0, :ol_date,
            :reg_id, :ol_is_active, :ob_detail, 1, 0);
      end
    end

    if (epay_amount + epay_tax_fl <> 0) then
    begin
      ob_amount = epay_amount + epay_tax_fl;

      acc_code = '71802810005540710101'; --  Расходы на рекламу
      acc_id = f_get_acc_id(acc_code);
      if (acc_id is null) then
        exception e_default 'Account ' || acc_code || ' not found';

      ob_detail = 'Сумма начисленных баллов в рублевом эквиваленте к выплате';
      execute procedure hp_do_kn_operbook_ins(:ol_id, null, :acc_id, :loyal_acc_id, null, null, :ob_amount, 0, :ob_amount, :ol_date,
          :reg_id, :ol_is_active, :ob_detail, 0, 1);
    end

    if (epay_tax_fl <> 0) then
    begin
      ob_amount = epay_tax_fl;

      acc_code = '60301810000000000001'; --  Расчеты по НДФЛ при исполнении обязанностей налогового агента
      acc_id = f_get_acc_id(acc_code);
      if (acc_id is null) then
        exception e_default 'Account ' || acc_code || ' not found';

      ob_detail = 'Сумма НДФЛ на сумму выплат свыше 4000р. за календарный год';
      execute procedure hp_do_kn_operbook_ins(:ol_id, null, :loyal_acc_id, :acc_id, null, null, :ob_amount, 0, ob_amount, :ol_date,
          :reg_id, :ol_is_active, :ob_detail, 0, 1);
    end

    if (epay_amount <> 0) then
    begin
      ob_amount = epay_amount;

      acc_code = '20501810000000000047'; --  Расчетный счет ООО Монета
      acc_id = f_get_acc_id(acc_code);
      if (acc_id is null) then
        exception e_default 'Account ' || acc_code || ' not found';

      ob_detail = 'Выплата начисленных баллов в рублевом эквиваленте';
      execute procedure hp_do_kn_operbook_ins(:ol_id, null, :loyal_acc_id, :acc_id, null, null, :ob_amount, 0, :ob_amount, :ol_date,
          :reg_id, :ol_is_active, :ob_detail, 0, 1);

      select al_id
      from acc_list
      where al_from_acc_id = :acc_id
      into al_id;
      if (al_id is null) then
        exception e_default 'Счет по Минфин не найден для ' || acc_code;

      insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_summa_val, ob_summa, ob_account_kt, ob_detail)
      values (:ol_id, '810', 1, '010181076-05000005', :ob_amount, :ob_amount, :al_id, :ob_detail);
    end
  end
end
