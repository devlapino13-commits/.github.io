CREATE PROCEDURE AAREP_CASH_DISB_CHECK_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    IN_FROM_BANK DECIMAL(18,4),
    IN_FROM_REPAY DECIMAL(18,4),
    OUT_TO_BANK DECIMAL(18,4),
    OUT_TO_DISB DECIMAL(18,4),
    OL_DATE DATE)
AS
declare variable cash_str varchar(20);
declare variable bank_str varchar(20);
declare variable database_id integer;
declare variable bank_str2 varchar(20);
declare variable al_id2 varchar(20);
declare variable start_date date;
declare variable al_id varchar(20);
declare variable bank_str3 varchar(20);
begin
  --  Start_Date='01.06.2014';
  start_date = '17.09.2014';

  :database_id = f_get_db_id();

  if (database_id in (39801, 41701)) then
  begin
    cash_str = '10001'; -- КР
    bank_str = '10150'; -- КР
    bank_str2 = '';
    bank_str3 = '';
  end

  if (database_id = 81001) then
  begin
    cash_str = '50-01'; -- РФ
    bank_str = '51-01'; -- РФ
    bank_str2 = '57-01';
    bank_str3 = '94-01';
  end

  in_from_bank = 0;
  in_from_repay = 0;

  out_to_bank = 0;
  out_to_disb = 0;

  for select al_id
      from acc_list
      where al_from_bg_id = :cash_str and
            al_from_reg_id = :rep_region_id
      into :al_id
  do
  begin
    if (database_id = 81001) then
      for select ob.ob_ol_date, sum(ob.ob_summa), -- Сдано в банк
                 0
          from operbook ob
          inner join operlist ol on ob.ob_from_ol_id = ol.ol_parent_id
          inner join operbook ob2 on (ol.ol_id = ob2.ob_from_ol_id and
                ob2.ob_short_acc_kt starting '57-04/' and
                ob2.ob_short_acc_dt starting '51-01/')
          where ob.ob_ol_is_active = 1 and
                ob.ob_ol_date between :start_date and :rep_date_end and
                ob.ob_short_acc_dt = '57-04/1' and
                ob.ob_account_kt = :al_id and
                ob.ob_ol_reg_id = :rep_region_id
          group by ob_ol_date
          into :ol_date, :out_to_bank, :in_from_bank
      do
      begin
        suspend;
        out_to_bank = 0;
        ol_date = null;
      end

    for select ob_ol_date, sum(iif(ob_account_dt = :al_id, ob_summa, 0)), -- все погашения включая из банка
               sum(iif(ob_account_kt = :al_id, ob_summa, 0)) -- все расходы, всключая сдачи в банк
        from operbook
        where ob_ol_is_active = 1 and
              ob_ol_date between :start_date and :rep_date_end and
              (ob_account_dt = :al_id or
              ob_account_kt = :al_id) and
              ob_ol_reg_id = :rep_region_id
        group by ob_ol_date
        into :ol_date, :in_from_repay, :out_to_disb
    do
    begin
      suspend;
    end
    in_from_repay = 0;
    out_to_disb = 0;

    for select al_id
        from acc_list
        where (al_from_bg_id = :bank_str or
              al_from_bg_id = :bank_str2 or
              al_from_bg_id = :bank_str3) and
              al_from_reg_id = :rep_region_id
        into :al_id2
    do
    begin
      for select ob_ol_date, sum(iif(ob_account_dt = :al_id2 and ob_account_kt = :al_id, ob_summa, 0)), -- Сдано в банк
                 sum(iif(ob_account_dt = :al_id and ob_account_kt = :al_id2, ob_summa, 0)) -- Прием из банка
          from operbook
          where ob_ol_is_active = 1 and
                ob_ol_date between :start_date and :rep_date_end and
                (ob_account_dt in (:al_id2, :al_id) or
                ob_account_kt in (:al_id2, :al_id)) and
                ob_ol_reg_id = :rep_region_id
          group by ob_ol_date
          into :ol_date, :out_to_bank, :in_from_bank
      do
      begin
        suspend;
      end
    end
  end
end
