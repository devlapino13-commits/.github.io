CREATE PROCEDURE HP_FLOATING_CHARTS_INFO(
    K_ID D_ID = 0)
RETURNS (
    RES_ERROR_TEXT D_STR_255,
    RES_INFO_TEXT D_STR_LONG)
AS
declare variable k_status d_integer;
declare variable ok_from_ol_id d_id;
declare variable sh_date d_date;
declare variable first_update_percent d_date;
declare variable second_update_percent d_date;
begin
  -- RFCKG-3422 обертка для модуля "Тестирование плавающих графиков"
  res_error_text = '';
  res_info_text = '<body>';

  if (exists(select * from hp_db_is_actual)) then
    res_error_text = 'БД дожна быть тестовой';


  select k.k_status
  from kontract k
  where k.k_id = :k_id
  into k_status;

  if (k_status <> 1) then
    res_error_text = 'Контракт не активный';

  select ok.ok_from_ol_id
  from operkred ok
  where ok.ok_from_k_id = :k_id
    and ok.ok_type = 8 --Выдача
    and ok_ol_is_active = 1
  into ok_from_ol_id;

  if(ok_from_ol_id is null) then
    res_error_text = 'По контракту не была произведена выдача';


  select sh.sh_date
  from schedule sh
  where sh.sh_from_k_id = :k_id    --Дата ближайшего погашения
    and sh.sh_date >= current_date
  order by sh.sh_date
  rows 1
  into sh_date;

  if (k_id = 0 or k_id is null ) then
    res_error_text = '';
    
  if(res_error_text = '' and (k_id is not null or k_id = 0)) then
  begin
    first_update_percent = f_date_work_nday(sh_date, 3);
    second_update_percent = f_date_work_nday(sh_date, 8);

    res_info_text = res_info_text || f_format('<p>Дата ближайшего погашения: <b>%1</b></p><p>Первое повышение плавающей процентной ставки: <b>%2</b></p><p>Второе повышение  плавающей процентной ставки: <b>%3</b></p>',
    f_datetostr(:sh_date), f_datetostr(first_update_percent), f_datetostr(second_update_percent));
  end
  else
    res_info_text = res_info_text || f_format('<p>%1</P>', res_error_text);

  res_info_text = res_info_text || '</body>';
  suspend;
end
