CREATE PROCEDURE AAREP_VAC_REQ_STAFF_REPORT(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    C_ID TYPE OF COLUMN CLIENTS.C_ID,
    C_NAME_FIO TYPE OF COLUMN CLIENTS.C_NAME_FIO,
    JL_NAME TYPE OF COLUMN JOBLIST.JL_NAME,
    REG_NAME TYPE OF COLUMN REGION.REG_NAME,
    LAST_HIRE D_DATE,
    DAYS_USED INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    REG_NAME_ORDR TYPE OF COLUMN REGION.REG_NAME,
    C_NAME_FIO_ORDR TYPE OF COLUMN CLIENTS.C_NAME_FIO)
AS
declare variable ps_from_reg_id d_integer;
declare variable c_id_tmp id_type;
declare variable ps_from_jl_id d_integer;
begin
  c_id_tmp = -1;
  for select ps.ps_from_c_id, ps.ps_from_reg_id, ps.ps_from_jl_id
      from hr_pstatus ps
      inner join region_list(:rep_region_id, 1, -1, 1) rl on ps.ps_from_reg_id = rl.id_list
      where ps.ps_date_begin <= :rep_date_end and
            ps.ps_date_end >= :rep_date_begin and
            ps.ps_status = 1 and
            ps.ps_is_working = 1 and
            exists(select 1
                   from vacationrequest vr
                   where vr.vr_from_c_id = ps.ps_from_c_id and
                         vr_date_begin >= :rep_date_begin and
                         vr_date_end <= :rep_date_end)
      order by ps.ps_from_c_id, ps.ps_date_begin desc
      into :c_id, :ps_from_reg_id, :ps_from_jl_id
  do
  begin
    if (c_id <> c_id_tmp) then
    begin
      c_id_tmp = :c_id;
      select c_name_fio
      from clients
      where c_id = :c_id
      into :c_name_fio;

      select jl_name
      from joblist
      where jl_id = :ps_from_jl_id
      into :jl_name;

      select ps.ps_date_begin
      from hr_pstatus ps
      inner join hr_orders ho on ps.ps_from_ho_id = ho.ho_id and
            ho.ho_order_type = 1 and
            ho.ho_status = 1
      where ps.ps_from_c_id = :c_id and
            ps.ps_date_begin <= :rep_date_begin
      order by 1 desc
      rows 1
      into last_hire;

      select reg_name
      from region
      where reg_unicode = :ps_from_reg_id
      into :reg_name;
      reg_name_ordr = :reg_name;
      c_name_fio_ordr = :c_name_fio;
      for select vr.vr_date_begin, vr.vr_date_end, vr.vr_days
          from vacationrequest vr
          where vr_from_c_id = :c_id and
                vr_date_begin >= :rep_date_begin and
                vr_date_end <= :rep_date_end
          order by vr.vr_date_begin
          into :date_begin, :date_end, :days_used
      do
      begin
        suspend;
        reg_name = '';
        c_name_fio = '';
        jl_name = '';
        last_hire = null;
      end
    end
  end
end
