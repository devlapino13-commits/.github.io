CREATE PROCEDURE AAREP_STAJ(
    DATE_BEGIN D_DATE,
    DATE_END D_DATE)
RETURNS (
    CNT_YEARS D_INTEGER,
    CNT_MONTHS D_INTEGER,
    CNT_DAYS D_INTEGER,
    DAYS_TOTAL D_INTEGER)
AS
declare variable year_a d_smallint;
    declare variable month_a d_smallint;
    declare variable day_a d_smallint;
    declare variable year_b d_smallint;
    declare variable month_b d_smallint;
    declare variable day_b d_smallint;
    declare variable first_date_a d_date;
    declare variable last_day_a d_smallint;
    declare variable last_day_b d_smallint;
    declare variable str_a d_str_10;
    declare variable str_b d_str_10;
begin
  cnt_years = 0;
  cnt_months = 0;
  cnt_days = 0;
  days_total = datediff(day, date_begin, date_end) + 1;

  year_a = extract(year from date_begin);
  month_a = extract(month from date_begin);
  day_a = extract(day from date_begin);

  year_b = extract(year from date_end);
  month_b = extract(month from date_end);
  day_b = extract(day from date_end);

  first_date_a = '01.' || month_a || '.' || year_a;
  last_day_a = extract(day from f_date_month_end(date_begin));
  last_day_b = extract(day from f_date_month_end(date_end));
  str_a = right(cast(date_begin as varchar(10)), 5);
  str_b = right(cast(date_end + 1 as varchar(10)), 5);

  if (year_a = year_b and month_a = month_b) then
  begin
    -- начало и конец периода совпадают по месяцу и году, считаем только дни
    cnt_days = day_b - day_a + 1;
    cnt_months = 0;
  end
  else if (str_a = str_b) then
  begin
    -- начало и конец периода совпадают по дню и месяцу, считаем только месяцы
    cnt_days = 0;
    cnt_months = datediff(month, date_begin, date_end);
  end
  else if (date_begin = first_date_a and day_b = last_day_b) then
  begin
    -- начало периода = первый день месяца и конец периода = последний день месяца
    cnt_days = 0;
    cnt_months = datediff(month, date_begin, date_end) + 1;
  end
  else if (date_begin = first_date_a) then
  begin
    -- начало периода = первый день месяца
    cnt_days = day_b;
    cnt_months = datediff(month, date_begin, date_end);
  end
  else if (day_b = last_day_b) then
  begin
    -- конец периода = последний день месяца
    cnt_days = last_day_a - day_a + 1;
    cnt_months = datediff(month, date_begin, date_end);
  end
  else
  begin
    -- начало периода <> первый день месяца, конец периода <> последний день месяца
    cnt_days = last_day_a - day_a + day_b + 1;
    cnt_months = datediff(month, date_begin, date_end) - 1;
  end

  cnt_months = cnt_months + cnt_days / 30;
  cnt_days = mod(cnt_days, 30);
  cnt_years = cnt_months / 12;
  cnt_months = mod(cnt_months, 12);

  suspend;
end
