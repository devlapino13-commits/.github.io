CREATE PROCEDURE HP_CHANGE_TRILLIONER_PASS
AS
declare variable c_id d_id;
declare variable un_id d_id;
declare variable hq_id d_integer;
declare variable reg_id d_integer;
declare variable ps_is_working d_smallint;
declare variable jl_name d_str_100;
declare variable new_pass d_str_code;
declare variable mail_to d_str_code;
declare variable un_name d_str_code;
begin
  -- Процедура смены пароля для Мираполис (сайт Триллионер уже не работает)
  hq_id = f_hq_id();
  for select un_id, un_from_c_id, un_name
      from username, clients
      where un_from_c_id = c_id
      into un_id, c_id, un_name
  do
  begin
    -- Либо пользователь работает либо уволен за последний месяц
    for select ps_is_working, jl_name, ps_from_reg_id
    from hr_pstatus, joblist
    where ps_date_begin <= current_date and
          ps_date_end >= current_date and
          ps_from_c_id = :c_id and
          ps_from_jl_id = jl_id and
          (ps_is_working = 1 or
          (ps_is_working = 0 and ps_date_begin >= dateadd(month, -1, current_date)))
    into ps_is_working, jl_name, reg_id
    do
    begin
      new_pass = f_gen_password(8, 1, 1, 1, 2);
      update username
      set un_trillioner_pass = :new_pass
      where un_id = :un_id;

      if (ps_is_working = 1 and (reg_id = hq_id or jl_name in ('Администратор', 'Кассир-оператор'))) then
      begin
        -- Отправляем письмо про смену пароля
        mail_to = coalesce(f_cont_info(0, c_id, 'email', 'ok'), '');
        if (mail_to <> '') then
          execute procedure hp_send_message('email', c_id, mail_to, 'Доступ в Академию М Булак',
          'Ваш логин и пароль для входа в Академию М Булак:<br>' ||
          'Логин: ' || un_name|| '<br>' ||
          'Пароль: ' || new_pass || '<br>' ||
          'Адрес Академии М Булак: https://academy.molbulak.com/ <br>',
          'hp_change_trillioner_pass', 7, un_id);
      end
    end
  end
end
