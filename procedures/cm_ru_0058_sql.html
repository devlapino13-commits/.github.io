CREATE PROCEDURE CM_RU_0058_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable k_amount_approved curr_type;
declare variable kc_circle integer;
declare variable kc_from_c_id id_type;
declare variable k_kredline_end date;
declare variable dolya double precision;
declare variable k_kolich_repay integer;
declare variable sh_total_sum curr_type;
declare variable k_add_info varchar(255);
declare variable c_curr_bonus_status integer;
begin
  k_add_info = null;
  if (k_id is not null and
    not exists(
    select * from schedule
    where sh_from_k_id = :k_id
    )) then
  begin
    res_error_text = 'В контракте не построен график погашения, постройте и повторите выдать.';
    res_error_is_critical = 2;
    suspend;
  end else
  for select kc_from_c_id, kc_amount_approved, kc_circle, k_amount_approved, k_kredline_end,
             k_kolich_repay, k_add_info, k_disb_date
    from knclient, kontract, product
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0  
    and kc_from_k_id = k_id
    and k_from_pr_id = pr_id
    and pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.)%'
    and pr_code not containing '.pilot.'
  into kc_from_c_id, kc_amount, kc_circle, k_amount_approved, k_kredline_end,
       k_kolich_repay, k_add_info, kf_disb_date
  do
  begin
    dolya = cast(kc_amount as double precision) /:k_amount_approved;
    k_add_info = coalesce(k_add_info, '');
    /*
  --Проверка суммы ежем. погашения. Добавлено 31.05.16 Музаффар
    if (exists (select * from schedule sch
              where sch.sh_from_k_id = :k_id
              and sh_date <> :k_kredline_end
              and sch.sh_id > 0
              and sh_os <> 0
              and :kc_amount/:k_amount_approved * (sch.sh_os + sh_perc) > 8500
              )) then
    begin
      res_error_text = 'Сумма ежемесячного платежа не должна быть более 8 500 руб.!';
      res_error_is_critical = 1;
      res_info_text = (select c_name_fio from clients where c_id = :kc_from_c_id);
      suspend;
    end
    */
    if (kf_disb_date <= '15.10.2017') then
    begin
        c_curr_bonus_status = null;
        select c_curr_bonus_status
        from hp_kontstate_loan_status(:k_id, 0, 0, :kf_disb_date, '')
        into c_curr_bonus_status;
        c_curr_bonus_status = coalesce(c_curr_bonus_status, 0);
    end else
    begin
      select k_curr_status from hp_get_kontract_status(:k_id, :kf_disb_date)
      into c_curr_bonus_status;
    end 
    for
    select :dolya * (sh_os + sh_perc)
    from schedule
    where sh_from_k_id = :k_id
    and sh_id between 1 and :k_kolich_repay - 1
    and sh_grace = 0
    into sh_total_sum
    do
    begin
      if (sh_total_sum < 3000.0 and c_curr_bonus_status <= 0) then
      begin
        select c_name_fio from clients
        where c_id = :kc_from_c_id
        into res_info_text;
        res_error_text = 'Сумма ежемесячного платежа не должна быть менее 3000,00 руб.';
        res_error_is_critical = 1;
        suspend;
        exit;
      end else
      if (sh_total_sum > 9000.0) then
      begin
        select c_name_fio from clients
        where c_id = :kc_from_c_id
        into res_info_text;
        if (kf_disb_date <= '15.10.2017') then
        begin
            if (k_add_info = 'new_product' and k_kolich_repay <> 6) then
            begin
              res_error_text = 'Сумма ежемесячного платежа не должна быть более 9000,00 руб.';
              res_error_is_critical = 1;
              suspend;
            end
            if (k_add_info <> 'new_product' and k_kolich_repay not between 6 and 12) then
            begin
              res_error_text = 'Сумма ежемесячного платежа не должна быть более 9000,00 руб.';
              res_error_is_critical = 1;
              suspend;
            end
        end else
        begin
          if (k_kolich_repay <> 6 and
              not(kc_amount between 50000 and 90000
                  and k_kolich_repay >= 6)) then
          begin
            res_error_text = 'Сумма ежемесячного платежа не должна быть более 9000,00 руб.';
            res_error_is_critical = 2;
            suspend;
          end
        end
      end
    end
  end
end
