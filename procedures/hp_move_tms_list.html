CREATE PROCEDURE HP_MOVE_TMS_LIST(
    REP_DATE DATE,
    REG_FROM INTEGER,
    WP_FROM DECIMAL(18,0),
    MO_FROM DECIMAL(18,0))
RETURNS (
    OM_ACCOUNT VARCHAR(20),
    OM_BILL_ID DECIMAL(18,0),
    OM_BILL_LINE_NO INTEGER,
    OM_WORKPLACE_ID DECIMAL(18,0),
    OM_MATOTVET_ID DECIMAL(18,0),
    OM_KOLICH_TMS DECIMAL(18,4),
    OM_AMOUNT_TMS DECIMAL(18,4),
    OB_INVENT_STR VARCHAR(20),
    OB_WP_NAME VARCHAR(100),
    OB_MO_NAME VARCHAR(100),
    ML_NAME VARCHAR(255))
AS
declare variable wp_id_prior decimal(18,0);
declare variable mo_id_prior decimal(18,0);
begin
  wp_id_prior = 0;
  mo_id_prior = 0;
  for select om_account, om_bill_id, om_bill_line_no, om_workplace_id, om_matotvet_id,
             sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus),
             sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus)
      from opermat, operlist
      where om_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            ol_date <= :rep_date and
            (:wp_from = 0 or
            om_workplace_id = :wp_from) and
            (:mo_from = 0 or
            om_matotvet_id = :mo_from) and
            ol_from_reg_id = :reg_from
      group by om_account, om_bill_id, om_bill_line_no, om_workplace_id, om_matotvet_id
      having(sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) <> 0) or (sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) <> 0)
      into :om_account, :om_bill_id, :om_bill_line_no, :om_workplace_id, :om_matotvet_id, :om_kolich_tms,
           :om_amount_tms
  do
  begin
    ob_invent_str = null;
    ml_name = null;
    select ob_invent_str, ml_name
    from operbill, matlist
    where ob_bill_id = :om_bill_id and
          ob_bill_line_no = :om_bill_line_no and
          ob_date_begin <= :rep_date and
          ob_date_end >= :rep_date and
          ob_is_delete = 0 and
          ml_id = ob_from_ml_id
    into :ob_invent_str, ml_name;

    if (wp_id_prior <> om_workplace_id) then
    begin
      select wp_name
      from workplace
      where wp_id = :om_workplace_id
      into :ob_wp_name;
      wp_id_prior = om_workplace_id;
    end
    if (mo_id_prior <> om_matotvet_id) then
    begin
      select c_name_fio
      from clients
      where c_id = :om_matotvet_id
      into :ob_mo_name;
      mo_id_prior = om_matotvet_id;
    end
    if (ob_invent_str is not null) then
      suspend;
  end
end
