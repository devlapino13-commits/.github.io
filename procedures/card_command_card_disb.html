CREATE PROCEDURE CARD_COMMAND_CARD_DISB(
    OBJECT_KIND D_STR_100,
    OBJECT_ID D_ID,
    PARENT_CS_ID D_INTEGER)
AS
declare variable k_amount_approved d_currency;
declare variable object_kind_str d_str_100;
declare variable k_id d_id;
declare variable c_id d_id;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable cca_code d_str_code;
declare variable cs_request_json d_str_8100;
declare variable disb_ol_id d_id;
declare variable cs_id d_integer;
declare variable k_last_reg_id d_integer;
declare variable un_id d_id;
declare variable new_limit d_currency;
begin
  /*
    Процедура отвечает за формирование операции для МПЦ
    "установка кредитного лимита по карт-счету",
    по факту выдачи займа
  */
  object_kind_str = f_objkind(object_kind, 'name');

  if (object_kind_str = 'kontract') then
    k_id = object_id;
  else
    exception e_message 'object_kind = '||object_kind_str||' не реализован';

  select
    k_amount_approved,
    k_fact_disb_ol_id,
    k_last_reg_id
  from kontract
  where k_id = :k_id
  into
    :k_amount_approved,
    :disb_ol_id,
    :k_last_reg_id;

  select
    ol_user_id
  from operlist
  where ol_id = :disb_ol_id
  into
    :un_id;

  select
    kc_from_c_id
  from knclient
  where kc_from_k_id = :k_id
    and kc_amount_approved > 0
  into
    :c_id;

  select
    card_id,
    cca_id,
    cca_code
  from hp_client_bank_card_mb(:c_id, current_date)
  into
    :card_id,
    :cca_id,
     :cca_code;

  new_limit = (select amount from hp_balance_cardloans('cardacc', :cca_id));
  cs_request_json = f_format('{"bank_c":"21", "groupc":"01", "limit":%1, "card_acct":"%2"}',
    cast(new_limit as d_id), cca_code);

  insert into cardsync(
    cs_parent_id,
    cs_from_csc_id,
    cs_card_id,
    cs_from_cca_id,
    cs_object_kind,
    cs_object_id,
    cs_amount,
    cs_request_json,
    cs_office_id,
    cs_from_un_id,
    cs_from_k_id,
    cs_from_c_id
  )
  values(
    :parent_cs_id,
    f_get_cardsynccfg('card_disb', 'id'),
    :card_id,
    :cca_id,
    f_objkind(:object_kind, 'id'),
    :object_id,
    :new_limit,
    :cs_request_json,
    :k_last_reg_id,
    :un_id,
    :k_id,
    :c_id
  )
  returning cs_id
  into :cs_id;

  update cardclientacc set
    cca_credit_limit = :new_limit
  where cca_id = :cca_id;

  -- создаю запись в cardsyncopers для связки cardsync с operlist
  insert into cardsyncopers (cso_from_cs_id, cso_from_ol_id, cso_after_exec)
  values (:cs_id, :disb_ol_id, 0);

  -- устанавливаю лимит в разрезе карт.счет - займ = сумме выдачи
  insert into cardclientacc_limit (
    ccl_from_cca_id, ccl_card_id, ccl_from_c_id, ccl_from_cs_id, ccl_from_ol_id,
    ccl_amount_limit_up, ccl_from_k_id, ccl_new_limit)
  values (
    :cca_id, :card_id, :c_id, :cs_id, :disb_ol_id,
    :k_amount_approved, :k_id, :k_amount_approved);
end
