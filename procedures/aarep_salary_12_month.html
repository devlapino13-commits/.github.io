CREATE PROCEDURE AAREP_SALARY_12_MONTH(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_255,
    C_INN D_STR_CODE,
    C_PINCODE D_STR_CODE,
    RES_1 D_CURRENCY,
    RES_2 D_CURRENCY,
    RES_3 D_CURRENCY,
    RES_4 D_CURRENCY,
    RES_5 D_CURRENCY,
    RES_6 D_CURRENCY,
    RES_7 D_CURRENCY,
    RES_8 D_CURRENCY,
    RES_9 D_CURRENCY,
    RES_10 D_CURRENCY,
    RES_11 D_CURRENCY,
    RES_12 D_CURRENCY)
AS
declare variable c_id d_id;
declare variable tmp_amount d_currency;
declare variable total_amount d_currency;
declare variable dd_begin d_date;
declare variable dd_end d_date;
declare variable sac_id d_integer;
declare variable month_no d_integer;
begin
  rep_date_begin = rep_date_begin - extract(day from rep_date_begin) + 1;
  rep_date_end = dateadd(month, 1, rep_date_end - extract(day from rep_date_end) + 1) - 1;

  for select distinct osm_from_c_id, c_name_fio, c_inn, c_pincode
      from opersalmain, clients
      where osm_from_c_id = c_id and
            osm_ol_is_active = 1 and
            osm_ol_date between :rep_date_begin and :rep_date_end
      order by c_id
      into c_id, c_name_fio, c_inn, c_pincode
  do
  begin
    res_1 = 0;
    res_2 = 0;
    res_3 = 0;
    res_4 = 0;
    res_5 = 0;
    res_6 = 0;
    res_7 = 0;
    res_8 = 0;
    res_9 = 0;
    res_10 = 0;
    res_11 = 0;
    res_12 = 0;

    dd_begin = rep_date_begin - extract(day from rep_date_begin) + 1;
    while (dd_begin <= rep_date_end) do
    begin
      dd_end = dateadd(month, 1, dd_begin) - 1;

      total_amount = 0;
      for select osd_from_sac_id, sum(osd_value)
          from opersalmain, opersaldetail
          where osm_id = osd_from_osm_id and
                osm_ol_is_active = 1 and
                osm_ol_date between :dd_begin and :dd_end and
                osm_from_c_id = :c_id and
                osd_from_sac_id in (22, 13, 14, 58)
          group by osd_from_sac_id
          into sac_id, tmp_amount
      do
      begin
        if (tmp_amount is null) then
          tmp_amount = 0;
        if (sac_id = 22) then
          total_amount = total_amount + tmp_amount;
        else
          total_amount = total_amount - tmp_amount;
      end
      month_no = extract(year from dd_begin) * 12 + extract(month from dd_begin) - extract(year from rep_date_begin) * 12 - extract(month from rep_date_begin) + 1;

      if (month_no = 1) then
        res_1 = total_amount;
      else
      if (month_no = 2) then
        res_2 = total_amount;
      else
      if (month_no = 3) then
        res_3 = total_amount;
      else
      if (month_no = 4) then
        res_4 = total_amount;
      else
      if (month_no = 5) then
        res_5 = total_amount;
      else
      if (month_no = 6) then
        res_6 = total_amount;
      else
      if (month_no = 7) then
        res_7 = total_amount;
      else
      if (month_no = 8) then
        res_8 = total_amount;
      else
      if (month_no = 9) then
        res_9 = total_amount;
      else
      if (month_no = 10) then
        res_10 = total_amount;
      else
      if (month_no = 11) then
        res_11 = total_amount;
      else
      if (month_no = 12) then
        res_12 = total_amount;

      dd_begin = dateadd(month, 1, dd_begin);
    end
    suspend;
  end
end
