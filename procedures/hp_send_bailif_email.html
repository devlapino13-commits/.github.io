CREATE PROCEDURE HP_SEND_BAILIF_EMAIL(
    EMAIL_RD_ID_LIST D_STR_255,
    MEMO D_STR_2048,
    C_ID D_ID,
    UN_ID D_ID,
    ACL_ID D_ID,
    EMAIL_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    MAIL_SUBJECT D_STR_255 = null)
AS
declare variable mail_to d_str_255;
declare variable rd_id d_integer;
declare variable rd_code d_str_100;
declare variable msg_id d_integer;
declare variable email d_str_100;
begin
  rd_code = 'Bailiff.Email';

  if (mail_subject = 'null' or mail_subject is null) then
    mail_subject = 'Оповещение пристава.';
  mail_to = '';
  email = '';

  if (:memo <> '') then
    email_body = :email_body || '<br><b>Примечание: </b><br>' || :memo;

  for select value_out
      from str_with_separator_parser(:email_rd_id_list)
      into :rd_id
  do
  begin
    select rd.rd_name
    from refdata rd
    where rd.rd_id = :rd_id
    into :email;

    mail_to = :mail_to || '; ' || :email;
  end

  mail_to = overlay(:mail_to placing '' from 1 for 2);

  if (mail_to = '') then
    exception e_message 'Произошла ошибка при отправке. Список адресатов пуст.';

  insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                              msg_command_kind)
  values (0, :mail_to, :mail_subject, :email_body, -1, :un_id, :rd_code)
  returning msg_id
  into :msg_id;

  insert into bailiff_msg (bm_from_msg_id, bm_from_c_id, bm_from_acl_id, bm_memo)
  values (:msg_id, :c_id, :acl_id, :memo);
end
