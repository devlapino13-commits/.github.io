CREATE PROCEDURE HP_CASHBOOKPAGES_GETPAGE(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    RES_PAGE_NUM INTEGER,
    RES_SALDO_BEGIN DECIMAL(18,4),
    RES_SALDO_END DECIMAL(18,4),
    RES_ERR_TEXT VARCHAR(255))
AS
declare variable CURR_YEAR D_INTEGER;
declare variable PREV_DATE D_DATE;
declare variable EPS_CASH_DATE_BEGIN D_DATE;
begin
  /*
  Возвращает начальный номер по дате и офису, или возвращает ошибку, если нужно
  сначала напечатать предыдущие дни
  */
  prev_date = null;
  res_page_num = 0;
  res_saldo_begin = null;
  res_saldo_end = null;
  res_err_text = '';
  eps_cash_date_begin = f_tmp_const('eps_cash_date_begin');

  -- Проверяем были ли движения на дату, если они были, заодно получаем сальдо
  if (eps_cash_date_begin > rep_date) then
    select ol_saldo_begin_dt - ol_saldo_begin_kt,
           ol_saldo_end_dt - ol_saldo_end_kt
    from aarep_cash_journal(:rep_region_id, :rep_date)
    rows 1
    into res_saldo_begin, res_saldo_end;
  else
    select ol_saldo_begin_dt - ol_saldo_begin_kt,
           ol_saldo_end_dt - ol_saldo_end_kt
    from aarep_cash_journal_eps(:rep_region_id, :rep_date)
    rows 1
    into res_saldo_begin, res_saldo_end;

  if (res_saldo_begin is null) then
  begin
    -- Если нет движений по кассе, то нельзя давать печатать,
    -- иначе появится дополнительная страница
    res_err_text = 'На указанную дату нет операций по кассе.';
    suspend;
    exit;
  end

  -- Получаем № страницы
  curr_year = extract(year from rep_date);
  select cbp_date + 1, cbp_last_page_num
  from cashbookpages
  where cbp_date < :rep_date and
        extract(year from cbp_date) = :curr_year and
        cbp_from_reg_id = :rep_region_id and
        cbp_is_actual = 1
  order by cbp_date desc, cbp_datetime desc
  rows 1
  into prev_date, res_page_num;

  if (prev_date is null) then
  begin
    -- Первая книга в этом году
    prev_date = cast('01.01.' || curr_year as date);
  end

  /* Проверяем, что за этот период не было ни одной даты,
  для которой отчет AAREP_CASH_JOURNAL вернет записи  */
  while (prev_date < rep_date) do
  begin
    /* Хватит только начальной даты и по текущий день */
    if ((eps_cash_date_begin > prev_date and exists(select * from aarep_cash_journal(:rep_region_id, :prev_date)))
      or (eps_cash_date_begin <= prev_date and exists(select * from aarep_cash_journal_eps(:rep_region_id, :prev_date)))) then
    begin
      res_err_text = 'Для соблюдения нумерации необходимо сначала ' || 'напечатать кассовую книгу начиная с: ' || f_datetostr(prev_date);
      break;
    end
    prev_date = prev_date + 1;
  end
  suspend;
end
