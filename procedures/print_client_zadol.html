CREATE PROCEDURE PRINT_CLIENT_ZADOL(
    PRINT_K_ID DECIMAL(18,0))
RETURNS (
    CZ_OS DECIMAL(18,4),
    CZ_PERC DECIMAL(18,4),
    CZ_PERC_SH DECIMAL(18,4),
    CZ_FINE DECIMAL(18,4),
    CZ_ALL DECIMAL(18,4),
    CZ_DATE DATE,
    CZ_OS_PERC DECIMAL(18,4),
    CL_ALL_SUB CURR_TYPE,
    CZ_DEL_DAYS INTEGER,
    CZ_FROM_C_ID DECIMAL(18,0),
    CZ_NAME_FIO VARCHAR(100),
    CZ_AMOUNT DECIMAL(18,4),
    CZ_CL_OS DECIMAL(18,4),
    CZ_CL_PERC DECIMAL(18,4),
    CZ_CL_FINE DECIMAL(18,4),
    CZ_CL_ALL DECIMAL(18,4),
    CZ_CL_OS_PERC DECIMAL(18,4),
    CZ_CL_TAX DECIMAL(18,4),
    CZ_CL_ALL_TAX DECIMAL(18,4),
    CZ_OS_SP DECIMAL(18,4),
    CZ_ALL_SP DECIMAL(18,4),
    CZ_CL_OS_SP DECIMAL(18,4),
    CZ_CL_ALL_SP DECIMAL(18,4),
    CUR_CL_OS DECIMAL(18,4),
    CUR_CL_PERC DECIMAL(18,4),
    CUR_CL_FINE DECIMAL(18,4),
    CUR_CL_ALL DECIMAL(18,4),
    CZ_TAX_NSP CURR_TYPE,
    CZ_CL_ALL_TAX_NSP CURR_TYPE,
    C_ADDRESS_YUR VARCHAR(255),
    C_ADDRESS_YUR_DISB VARCHAR(255),
    C_ADDRESS_RODINA_YUR VARCHAR(255),
    CZ_USE_DAYS INTEGER,
    C_BIRTH_DATE DATE,
    FIRST_ALL_PERC DECIMAL(18,4),
    CL_FIRST_ALL_PERC DECIMAL(18,4),
    PAY_OS DECIMAL(18,4),
    PAY_PERC DECIMAL(18,4),
    PAY_FINE DECIMAL(18,4),
    CL_PAY_OS DECIMAL(18,4),
    CL_PAY_PERC DECIMAL(18,4),
    CL_PAY_FINE DECIMAL(18,4),
    DELINQ_DATE DATE,
    CL_CITIZENSHIP VARCHAR(255),
    SH_MIN_DATE DATE,
    SH_MIN_FINE CURR_TYPE,
    SH_MIN_PERC CURR_TYPE,
    SH_MIN_OS CURR_TYPE,
    SH_MIN_ALL CURR_TYPE,
    CL_SH_MIN_FINE CURR_TYPE,
    CL_SH_MIN_PERC CURR_TYPE,
    CL_SH_MIN_OS CURR_TYPE,
    CL_SH_MIN_ALL CURR_TYPE,
    CZ_DEL_NSP CURR_TYPE,
    CZ_DEL_NSP_SH CURR_TYPE,
    CZ_CL_DEL_NSP CURR_TYPE,
    CZ_CL_DEL_ALL CURR_TYPE,
    CL_OS_SUB CURR_TYPE,
    CL_PERC_SUB CURR_TYPE)
AS
declare variable sh_date date; /* CL_PAY_FINE */
declare variable sh_calc_fine decimal(18,4);
declare variable sh_calc_perc decimal(18,4);
declare variable zad_os decimal(18,4);
declare variable zad_perc decimal(18,4);
declare variable rep_date date;
declare variable kont_amount decimal(18,4);
declare variable st_staff integer;
declare variable st_client integer;
declare variable sum_os decimal(18,4) = 0;
declare variable sum_perc decimal(18,4) = 0;
declare variable sum_fine decimal(18,4) = 0;
declare variable cur_sum_os decimal(18,4) = 0;
declare variable cur_sum_perc decimal(18,4) = 0;
declare variable cur_sum_fine decimal(18,4) = 0;
declare variable sum_loan decimal(18,4) = 0;
declare variable k_disb_date date;
declare variable k_nsp_percent decimal(18,7);
declare variable sh_perc_cur decimal(18,4);
declare variable sh_kommis_cur decimal(18,4);
declare variable sh_perc_temp decimal(18,4);
declare variable sh_kommis_temp decimal(18,4);
declare variable initial_tax_cur decimal(18,4);
declare variable payed_tax decimal(18,4);
declare variable kz_tax_cur decimal(18,4);
declare variable sum_tax decimal(18,4);
declare variable k_stop_perc date;
declare variable sum_os_sp decimal(18,4);
declare variable cur_os decimal(18,4);
declare variable cur_perc decimal(18,4);
declare variable cur_fine decimal(18,4);
declare variable cur_all decimal(18,4);
declare variable c_citizen integer;
declare variable cl_dolya perc_type;
declare variable oko_ol_date date;
declare variable oko_os curr_type;
declare variable oko_perc curr_type;
declare variable oko_other curr_type;
declare variable db_id integer;
begin

  :db_id = f_get_db_id();
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  cz_os = 0;
  cz_os_sp = 0;
  cz_perc = 0;
  cz_perc_sh = 0;
  cz_fine = 0;
  cz_all = 0;
  cz_all_sp = 0;

  cur_os = 0;
  cur_perc = 0;
  cur_fine = 0;
  cur_all = 0;
  cl_all_sub = 0;
  cl_os_sub = 0;
  cl_perc_sub = 0;

  sh_min_date = null;
  rep_date = cast('NOW' as date);

  select k_amount_approved, k_disb_date, kont.k_stop_perc,
         datediff(day, kont.k_disb_date, kont.k_stop_perc) as cz_use_days
  from kontract kont
  where k_id = :print_k_id
  into :kont_amount, :k_disb_date, :k_stop_perc, cz_use_days;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
  end
  if (k_nsp_percent is null) then
    k_nsp_percent = 0;

  for select sh_date, sh_calc_fine, sh_calc_perc, zad_os, zad_perc
      from aarep_make_zad_list(:print_k_id, :rep_date, 1)
      --  where SH_DATE is not null
      into :sh_date, :sh_calc_fine, :sh_calc_perc, :zad_os, :zad_perc
  do
  begin
    if ((sh_min_date is null) and
        (coalesce(sh_date, '01.01.1990') < :rep_date)) then
    begin
      if ((zad_os <> 0) or (sh_calc_perc <> 0) /*or (SH_CALC_FINE<>0)*/ ) then
      begin
        if (sh_date is not null) then
          sh_min_date = :sh_date;
        /*Добавлено по просьбе Светланы от 18 марта 16-года*/
        sh_min_fine = :sh_calc_fine;
        sh_min_perc = :sh_calc_perc;
        sh_min_os = :zad_os;
        sh_min_all = zad_os + sh_calc_perc + coalesce(sh_calc_fine, 0);
        /*Добавлено по просьбе Светланы от 18 марта 16-года*/
      end
    end
    if (coalesce(sh_date, '01.01.1990') <= :rep_date) then
    begin
      cz_os = cz_os + zad_os;
      cz_fine = cz_fine + sh_calc_fine;
      cz_perc = cz_perc + sh_calc_perc;
      cz_perc_sh = cz_perc_sh + zad_perc;
    end
    select res
    from fhp_round_bank(:cz_perc * :k_nsp_percent / 100, 0)
    into cz_del_nsp;

    select res
    from fhp_round_bank(:cz_perc_sh * :k_nsp_percent / 100, 0)
    into cz_del_nsp_sh;

    if (sh_date is not null) then
    begin
      cur_os = round(cur_os + zad_os);
      cur_perc = round(cur_perc + sh_calc_perc);
      cur_fine = round(cur_fine + sh_calc_fine);
    end
    if (k_stop_perc is not null) then
      cz_os_sp = round(cz_os_sp + zad_os);
    else
      cz_os_sp = cz_os;
  end
  cur_all = round(cur_os + cur_perc + cur_fine);

  cz_all = round(cz_os + cz_perc + cz_fine);
  cz_os_perc = round(cz_os + cz_perc);

  if (k_stop_perc is not null) then
    cz_all_sp = round(cz_os_sp + cz_perc + cz_fine);
  else
    cz_all_sp = cz_all;

  cz_date = :rep_date;
  if (db_id = 81001 and
      sh_min_date is null) then
  begin
    oko_ol_date = null;
    oko_os = 0;
    oko_perc = 0;
    oko_other = 0;
    select min(oko_ol_date), sum(oko_os + oko_sh_os), sum(oko_perc + oko_sh_perc), sum(oko_other + oko_sh_other)
    from operkroff
    where oko_ol_is_active = 1 and
          oko_from_k_id = :print_k_id and
          oko_ol_date <= current_date
    into oko_ol_date, oko_os, oko_perc, oko_other;
    if (oko_os is null) then
      oko_os = 0;
    if (oko_perc is null) then
      oko_perc = 0;
    if (oko_other is null) then
      oko_other = 0;

    if (oko_ol_date is not null) then
    begin
      select oks_delinq_date
      from oksummary
      where :oko_ol_date - 1 between oks_date_begin and oks_date_end and
            oks_from_k_id = :print_k_id and
            oks_close_state = 0
      into sh_min_date;
    end

    cur_os = round(cur_os + oko_os);
    cur_perc = round(cur_perc + oko_perc);
    cur_fine = round(cur_fine + oko_other);
  end

  if (sh_min_date is not null) then
    cz_del_days = (rep_date - sh_min_date);
  else
    cz_del_days = 0;

  sh_perc_cur = 0;
  sh_kommis_cur = 0;

  for select coalesce(sh_perc, 0), coalesce(sh_kommis, 0)
      from schedule
      where sh_date < :rep_date and
            sh_from_k_id = :print_k_id
      into :sh_perc_temp, :sh_kommis_temp
  do
  begin
    sh_perc_cur = sh_perc_cur + sh_perc_temp;
    sh_kommis_cur = sh_kommis_cur + sh_kommis_temp;
  end
  initial_tax_cur = cast(((sh_kommis_cur + sh_perc_cur) * k_nsp_percent / 100) as decimal(18,2));
  select sum(ok.ok_tax)
  from operkred ok
  where ok.ok_from_k_id = :print_k_id and
        ok.ok_ol_is_active = 1 and
        ok_from_c_id = 0
  into :payed_tax;

  payed_tax = coalesce(payed_tax, 0);
  kz_tax_cur = round(initial_tax_cur + payed_tax);

  sum_os = 0;
  sum_perc = 0;
  sum_fine = 0;
  sum_loan = 0;
  sum_tax = 0;
  sum_os_sp = 0;

  cur_sum_os = 0;
  cur_sum_perc = 0;
  cur_sum_fine = 0;

  for select c_name_fio, kc_amount_approved, kc_from_c_id, cl.c_birth_date, c_citizen
      from clients cl, knclient
      where kc_from_k_id = :print_k_id and
            kc_from_c_id = c_id and
            kc_amount_approved is not null and
            kc_amount_approved <> 0
      into :cz_name_fio, :cz_amount, cz_from_c_id, c_birth_date, c_citizen
  do
  begin
    sum_loan = sum_loan + cz_amount;
    if (sum_loan < kont_amount) then
    begin
      cl_dolya = cz_amount / kont_amount;
      cz_cl_os = round(cz_os * cl_dolya);
      sum_os = round(sum_os + cz_cl_os);

      cur_cl_os = round(cur_os * cl_dolya);
      cur_sum_os = round(cur_sum_os + cur_cl_os);

      if (k_stop_perc is not null) then
      begin
        cz_cl_os_sp = round(cz_os_sp * cl_dolya);
        sum_os_sp = round(sum_os_sp + cz_cl_os_sp);
      end
      else
      begin
        cz_cl_os_sp = cz_cl_os;
        sum_os_sp = sum_os;
      end

      cz_cl_perc = round(cz_perc * cl_dolya);
      sum_perc = round(sum_perc + cz_cl_perc);

      cur_cl_perc = round(cur_perc * cl_dolya);
      cur_sum_perc = round(cur_sum_perc + cur_cl_perc);

      cz_cl_fine = round(cz_fine * cl_dolya);
      sum_fine = round(sum_fine + cz_cl_fine);

      cur_cl_fine = round(cur_fine * cl_dolya);
      cur_sum_fine = round(cur_sum_fine + cur_cl_fine);

      cz_cl_tax = round(kz_tax_cur * cl_dolya);
      sum_tax = round(sum_tax + cz_cl_tax);
    end
    else
    begin
      cz_cl_os = round(cz_os - sum_os);
      cur_cl_os = round(cur_os - cur_sum_os);

      if (k_stop_perc is not null) then
        cz_cl_os_sp = round(cz_os_sp - sum_os_sp);
      else
        cz_cl_os_sp = cz_cl_os;

      cz_cl_perc = round(cz_perc - sum_perc);
      cz_cl_fine = round(cz_fine - sum_fine);

      cur_cl_perc = round(cur_perc - cur_sum_perc);
      cur_cl_fine = round(cur_fine - cur_sum_fine);

      cz_cl_tax = round(kz_tax_cur - sum_tax);
    end

    cz_cl_del_nsp = round(cz_cl_perc * k_nsp_percent / 100);
    cz_cl_tax = round(cz_cl_tax);
    cz_tax_nsp = round(cur_cl_perc * k_nsp_percent / 100);

    cz_cl_all = cz_cl_os + cz_cl_perc + cz_cl_fine;
    cur_cl_all = round(cur_cl_os + cur_cl_perc + cur_cl_fine);

    if (k_stop_perc is not null) then
      cz_cl_all_sp = round(cz_cl_os_sp + cz_cl_perc + cz_cl_fine);
    else
      cz_cl_all_sp = cz_cl_all;

    cz_cl_os_perc = round(cz_cl_os + cz_cl_perc);
    cz_cl_del_all = round(cz_cl_os + cz_cl_perc + cz_cl_fine + cz_cl_del_nsp);
    cz_cl_all_tax = round(cz_cl_os + cz_cl_perc + cz_cl_fine + cz_cl_tax);
    cz_cl_all_tax_nsp = round(cz_cl_os + cz_cl_perc + cz_cl_fine + cz_tax_nsp); -- добавлено от 14.09.2015 Улан

    -->>>>>>>> ДОбавил Музаффар по просьбе Светланы от 6 октября 15 года
    c_address_yur = '';
    c_address_rodina_yur = '';

    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :cz_from_c_id and
          adc_address_type = 1 and
          adc_from_adl_id = adl_id
    into :c_address_yur;
    if (c_address_yur is null) then
      c_address_yur = '';
    if (c_address_yur = 'Не указан') then
      c_address_yur = '';

    c_address_yur_disb = c_address_yur;

    select pva_text_value
    from hp_persinfo_loglist(:cz_from_c_id, 'addrclient', 'address_legal')
    where pva_datetime <= :k_disb_date
      and pva_text_value <> 'Не указан'
    order by pva_datetime desc
    rows 1
    into c_address_yur_disb;

    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :cz_from_c_id and
          adc_address_type = 4 and
          adc_from_adl_id = adl_id
    into :c_address_rodina_yur;
    if (c_address_rodina_yur is null) then
      c_address_rodina_yur = '';
    if (c_address_rodina_yur = 'Не указан') then
      c_address_rodina_yur = '';

    select sum(sh_perc) sh_perc
    from schedule
    where sh_from_k_id = :print_k_id
    into :first_all_perc;

    select -sum(ok_os) ok_os, -sum(ok_perc) ok_perc, -sum(ok_fine) ok_fine
    from operkred ok
    where ok_ol_is_active = 1 and
          ok_from_c_id = 0 and
          ok_type in (4, 5, 9) and
          ok_from_k_id = :print_k_id
    into pay_os, pay_perc, pay_fine;

    cl_first_all_perc = round(first_all_perc * cl_dolya);
    cl_pay_os = round(pay_os * cl_dolya);
    cl_pay_perc = round(pay_perc * cl_dolya);
    cl_pay_fine = round(pay_fine * cl_dolya);
    --<<<<< Конец от 06 октября 15 года

    delinq_date = sh_min_date;

    select ref.rd_name
    from refdata ref
    where ref.rd_id = :c_citizen and
          ref.rd_main_code = 'Citizen'
    into cl_citizenship;

    /*Добавлено по просьбе Светланы от 18 марта 16-года*/
    cl_sh_min_fine = round(cl_sh_min_fine * cl_dolya);
    cl_sh_min_perc = round(cl_sh_min_perc * cl_dolya);
    cl_sh_min_os = round(cl_sh_min_os * cl_dolya);

    cl_sh_min_all = round(cl_sh_min_os + cl_sh_min_perc + cl_sh_min_fine);

    /*Добавлено по просьбе Светланы от 18 марта 16-года Конец*/

    cl_all_sub = cur_cl_all-cz_all;
    cl_os_sub=cur_cl_os-cz_cl_os;
    cl_perc_sub=cur_cl_perc-cz_cl_perc;

     suspend;
  end

end
