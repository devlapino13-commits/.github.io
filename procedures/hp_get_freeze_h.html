CREATE PROCEDURE HP_GET_FREEZE_H(
    K_ID D_ID)
RETURNS (
    FREEZE_TIME_H D_INTEGER)
AS
declare variable pr_code d_str_100;
declare variable disb_summ d_currency;
begin
  freeze_time_h = 0;

  select k_amount_approved, pr_code
  from kontract, product
  where k_id = :k_id
    and pr_id = k_from_pr_id
  into disb_summ, pr_code;
    
  -------------- РФ -----------------
  if (f_get_db_id() = 81001) then
  begin
    if (pr_code containing '.selfemployed.' or pr_code containing '.business.') then  -- Для  продуктов Самозанятый и Бизнес, период охлаждения не работает
    begin
      suspend;
      exit;
    end 

    freeze_time_h = case
                      when disb_summ > 200000 then 48
                      when disb_summ >= 50000 then 4
                      else 0
                    end;
  end
  -------------- KZ -----------------
  else if (f_get_db_id() = 39801) then
  begin
    freeze_time_h = iif(disb_summ > cast(f_get_const('kz_freeze_start_amount') as integer), 24, 0);
  end 

  suspend;
end
