CREATE PROCEDURE HP_EXTINFO_SET_RESULT(
    ID D_ID,
    JSON_INFO D_BLOB_TEXT)
AS
declare variable FILE_NAME D_STR_255;
declare variable CAPTION D_STR_255;
declare variable C_ID D_ID;
declare variable C_NAME_FIO D_STR_255;
declare variable BLS_ID D_ID;
declare variable DB_ID D_ID;
begin
  -- процедура сохраняет результат полученный от внешних источников

  bls_id = null;
  if (json_info is null or json_info = '' or json_info = '{}') then json_info = '';

  if (json_info <> '') then
  begin
    db_id = f_get_db_id();

    file_name = '';
    select eic_code, eic_name, eil_from_c_id
    from extinfocfg, extinfolist
    where eil_id = :id and
          eil_from_eic_id = eic_id
    into file_name, caption, c_id;

    c_name_fio = '';
    if (c_id is not null) then
    begin
      select c_name_fio from clients
      where c_id = :c_id
      into c_name_fio;
    end
    if (c_name_fio <> '') then
      caption = caption ||', '||c_name_fio;

    if (file_name = '') then
      exception e_default 'ID not found: '||coalesce(id, '<NULL>');

    file_name = file_name || '_json.txt';

    execute procedure hp_insert_into_blobstore (:file_name, :caption, :json_info, 1000000000000, 2, 3)
    returning_values :bls_id;

    if (db_id = 41701) then
      insert into temp_grs(bls_id, json_text)
      values (:bls_id, :json_info);
  end


  update extinfolist
  set eil_status = 2,
      eil_from_bls_id = :bls_id
  where eil_id = :id;
end
