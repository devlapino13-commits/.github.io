CREATE PROCEDURE HP_GET_BASE_SCHEDULE(
    K_ID D_ID)
RETURNS (
    SH_ID D_INTEGER,
    SH_DATE D_DATE,
    SH_OS D_CURRENCY,
    SH_PERC D_CURRENCY,
    SH_KOMMIS D_CURRENCY,
    SH_GRACE D_INTEGER,
    SH_IS_EDITED D_INTEGER,
    SH_PERC_TAX D_CURRENCY,
    SH_DISB_AMOUNT D_CURRENCY,
    SH_DIFF_DATE D_INTEGER,
    SH_KOMMIS_MONTHLY D_CURRENCY)
AS
declare variable sh_date_prv d_date;
declare variable last_sh_id d_integer;
declare variable new_annu d_currency;
declare variable first_date d_date;
declare variable days_list d_str_long;
declare variable sh_balance d_currency;
declare variable new_perc_per_day d_percent;
declare variable k_disb_date d_date;
declare variable orig_sh_os d_currency;
declare variable start_date d_date;
declare variable loan_amount d_currency;
declare variable pr_percent_kind d_integer;
declare variable is_other_base_schedule d_integer;
begin
  if (coalesce(k_id, 0) = 0) then
    exception e_default 'K_ID is emptry';

  start_date = cast('08.09.2021' as date);

  select k_percent_per_day, k_disb_date, pr_percent_kind
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into new_perc_per_day, k_disb_date, pr_percent_kind;

  is_other_base_schedule = 0;

  if (f_get_db_id() = 81001 and pr_percent_kind = 1 and k_disb_date >= start_date) then
    is_other_base_schedule = 1; -- Необходимо показать другой график как базовую

  if (is_other_base_schedule = 0) then
  begin
    for select sh_id, sh_date, sh_os, sh_perc, sh_kommis,
               sh_grace, sh_is_edited, sh_perc_tax, sh_disb_amount,
               sh_diff_date, sh_kommis_monthly
        from schedule
        where sh_from_k_id = :k_id
        order by sh_id
        into :sh_id, :sh_date, :sh_os, :sh_perc, :sh_kommis,
             sh_grace, sh_is_edited, sh_perc_tax, sh_disb_amount,
             sh_diff_date, sh_kommis_monthly
    do
      suspend;
  end else
  begin
    sh_kommis = 0;
  
    days_list = '';
    loan_amount = 0;
    sh_date_prv = null;
  
    for select sh_id, sh_date, sh_os, sh_perc
    from schedule
    where sh_from_k_id = :k_id and
          sh_id >= 1
    order by sh_id
    into sh_id, sh_date, sh_os, sh_perc
    do
    begin
      if (sh_id > 1) then
      begin
        loan_amount = loan_amount + sh_os;
  
        if (sh_os <> 0) then
        begin
          if (days_list <> '') then
            days_list = days_list || ',';
    
          days_list = days_list || datediff(day, sh_date_prv, sh_date);
        end
      end
  
      if (sh_id = 1) then
        first_date = sh_date;
  
      sh_date_prv = sh_date;
      last_sh_id = sh_id;
    end
  
    new_annu = f_annu(loan_amount, 0, 0, new_perc_per_day, days_list, 0, 0, 0, 0, 0);
  
    sh_balance = loan_amount;
    for select sh_id, sh_date, sh_os, sh_perc, sh_kommis,
               sh_grace, sh_is_edited, sh_perc_tax, sh_disb_amount,
               sh_diff_date, sh_kommis_monthly
    from schedule
    where sh_from_k_id = :k_id
    order by sh_id
    into sh_id, sh_date, sh_os, sh_perc, sh_kommis,
         sh_grace, sh_is_edited, sh_perc_tax, sh_disb_amount,
         sh_diff_date, sh_kommis_monthly
    do
    begin
      orig_sh_os = sh_os;
      if (sh_id >= 2) then
      begin
        if (sh_id = 2) then
          sh_date_prv = first_date;
  
        sh_perc = sh_balance / cast(100.0000 as float) * new_perc_per_day * datediff(day, sh_date_prv, sh_date);
  
        sh_perc = round(sh_perc, 0);
  
        if (orig_sh_os <> 0) then
        begin
          if (sh_id < last_sh_id) then
          begin
            sh_os = new_annu - sh_perc;
            if (sh_os < 0) then
              sh_os = 0;
          end
          else sh_os = sh_balance;
  
          sh_balance = sh_balance - sh_os;
        end else sh_os = 0;
  
        if (sh_os < 0) then
          exception e_default 'Отрицательная основная сумма! Дождитесь решения от разработчиков.';
      end
  
      suspend;
  
      sh_date_prv = sh_date;
    end
  end
end
