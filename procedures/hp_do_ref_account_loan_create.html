CREATE PROCEDURE HP_DO_REF_ACCOUNT_LOAN_CREATE(
    K_ID ID_TYPE,
    WITH_SALDO INTEGER = 0)
AS
declare variable kc_id id_type;
declare variable sab_id id_type;
declare variable saldo_date date;
declare variable kc_non_resident integer;
declare variable kc_client_type integer;
declare variable disb_reg_id integer;
declare variable acc_id id_type;
declare variable disb_date date;
begin
-- создает лицевые счета по займу
  saldo_date = '31.12.2017';

  for select kc_id, kc_non_resident, kc_client_type, ks_date_begin, ks_region_id
      from knclient, kontstate
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            ks_from_k_id = :k_id and
            ks_state = 1 and
            ks_ol_is_delete = 0
      into :kc_id, :kc_non_resident, :kc_client_type, disb_date, disb_reg_id
  do
  begin
    if (disb_date < saldo_date) then
      disb_date = saldo_date;

-- создаем счета по займу
    for select sab_id
        from sys_accbalance
        where sab_object_kind = 8
        order by sab_full_code
        into :sab_id
    do
    begin
      select acc_id
      from get_ref_account_knclient(:sab_id, '', :kc_id, :kc_non_resident, :kc_client_type, :disb_date, :disb_reg_id)
      into :acc_id;
    end
  end

  if (with_saldo = 1) then
    execute procedure hp_do_ref_account_loan_ob_saldo(:k_id);
end
