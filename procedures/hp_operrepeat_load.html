CREATE PROCEDURE HP_OPERREPEAT_LOAD(
    K_ID D_ID,
    ORP_ID D_INTEGER)
RETURNS (
    RES_TEXT D_STR_255,
    READ_ONLY D_SMALLINT,
    RES_ORP_ID D_ID,
    CAPTION D_STR_255,
    ORP_FROM_K_ID D_ID,
    ORP_BEGIN_DATE D_DATE,
    ORP_OL_DATE D_DATE,
    ORP_LOAN_REVERSE D_SMALLINT,
    ORP_BOOK_REVERSE D_SMALLINT,
    ORP_STATE D_SMALLINT,
    ORP_MEMO D_STR_LONG,
    ORP_SQL D_STR_8100,
    ORP_CHANGE_LOG D_BLOB_TEXT)
AS
declare variable k_kontract_no d_str_code;
begin
  res_text = ''; -- Если текст пустой, значит без ошибки
  read_only = 0;
  caption = 'Перепроведение операций по займу';

  if (orp_id is null) then
    orp_id = 0;

  if (k_id is null) then
    k_id = 0;

  if (orp_id = 0 and k_id <> 0) then
  begin
    select orp_id, orp_state
    from operrepeat
    where orp_from_k_id = :k_id and
          f_enum('operrepeat_state', orp_state, 'code') not in ('done', 'cancelled')
    into orp_id, orp_state;

    if (orp_id = 0) then
      orp_state = 0;
  end

  if (orp_id <> 0) then
  begin
    select orp_from_k_id, orp_ol_date, orp_begin_date, orp_book_reverse, orp_loan_reverse,
           orp_state, orp_memo, orp_sql, orp_change_log
    from operrepeat
    where orp_id = :orp_id
    into :orp_from_k_id, :orp_ol_date, :orp_begin_date, :orp_book_reverse, :orp_loan_reverse,
         :orp_state, :orp_memo, :orp_sql, :orp_change_log;
  
    if (row_count = 0)
    then res_text = 'Нет запроса с указанным ID';
    else
    begin
      if (f_enum('operrepeat_state', orp_state, 'code') in ('cancelled', 'done'))
      then read_only = 1;
      else orp_change_log = '';
    end
  end else
  begin
    orp_from_k_id = k_id;
    orp_begin_date = current_date - (extract(day from current_date) - 1);
    orp_ol_date = orp_begin_date;
    orp_loan_reverse = 0;
    orp_book_reverse = 0;
    orp_state = 0;
    orp_change_log = '';
  end

  if (orp_state is null) then
    orp_state = 0;

  if (orp_memo is null) then
    orp_memo = '';

  if (orp_sql is null) then
    orp_sql = '';

  select k_kontract_no
  from kontract
  where k_id = :orp_from_k_id
  into k_kontract_no;

  caption = f_format('Перепроведение операций по займу %1, запрос: %2 %3',
                     k_kontract_no,
                     f_enum('operrepeat_state', orp_state, 'name'));

  res_orp_id = orp_id;
  suspend;
end
