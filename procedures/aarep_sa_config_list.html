CREATE PROCEDURE AAREP_SA_CONFIG_LIST(
    C_ID DECIMAL(18,0))
RETURNS (
    SAC_ID INTEGER,
    SAN_DETAIL VARCHAR(255))
AS
begin
  if ((c_id <> 0) and
      (c_id is not null)) then
  begin
    for select sd_id, sd_category || ', договор от ' || sd_date_begin
        from servicedog
        where sd_from_c_id = :c_id
          and sd_status in (0, 1)
        order by sd_date_begin desc
        into :sac_id, :san_detail
    do
    begin
      sac_id = 0 - sac_id;
      suspend;
    end
  end
  for select sac_id, sac_name || ' (' || sac_code || ')'
      from sa_config
      where sac_is_credit = 11
      into :sac_id, san_detail
  do
  begin
    suspend;
  end
end
