CREATE PROCEDURE HP_CONTVERI_SEND(
    C_ID D_ID,
    CONT_KIND D_STR_CODE,
    CONT_VALUE D_STR_100,
    CONT_IS_MAIN D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable RD_ID integer;
declare variable LAST_DATETIME timestamp;
declare variable CV_ID decimal(18,0);
declare variable CV_PASSWORD_ENCODED varchar(255);
declare variable MSG_BODY D_STR_8100;
declare variable C_NAME_FIO D_STR_100;
declare variable DB_ID D_INTEGER;
declare variable C_SEX D_SMALLINT;
declare variable MSG_SUBJECT D_STR_255;
declare variable LINK_PARAMS D_STR_255;
declare variable MSG_ID D_INTEGER;
declare variable EDS_UUID D_STR_CODE;
declare variable CONT_KIND_CODE D_STR_CODE;
declare variable PIN_CODE D_STR_CODE;
declare variable CDA_ID D_ID;
declare variable CVR_ID D_INTEGER;
declare variable CLT_ID D_INTEGER;
declare variable IS_NEW_CONTDATA D_SMALLINT;
declare variable CDA_STATUS D_SMALLINT;
declare variable OTHER_CLIENT_COUNT D_INTEGER;
begin
  -- Процедура отправляет контактную информация на верификацию

  res_is_error = 1;
  res_text = '';

  if (coalesce(cont_value, '') = '') then
  begin
    res_text = 'Контактная информация не заполнена';
    suspend;
    exit;
  end

  is_new_contdata = f_get_const('new_contdata');

  is_new_contdata = 2;

  -- если тип объекта передан через название, определяем числовой код
  db_id = f_get_db_id();

  -- cont_kind могут передаваться как старый, так и новый вариант кодовых названий
  -- поэтому конверетируем их между собой
  select rd_id, cont_kind_code, cont_kind_id, cont_is_main
  from hp_contkind_info(:cont_kind)
  into :rd_id, :cont_kind_code, :cont_kind, cont_is_main;

  -- если ранее успешно выслан код, уточняем время отправки.
  -- Ниже в коде понадобиться
  if (is_new_contdata in (0, 2)) then -- если работаем по старому
  begin
    select cv_datetime
    from contverify
    where cv_object_kind = f_objkind('clients', 'id') and
          cv_object_id = :c_id and
          cv_from_rd_id = :rd_id and
          cv_contact_value = :cont_value and
          cv_status = 0
    order by cv_id desc
    rows 1
    into :last_datetime;
  end

  if (is_new_contdata in (1, 2)) then -- если работаем по новому
  begin
    clt_id = f_cont_id(cont_value, 'null');

    if (clt_id is not null) then
    begin
      select cda_id, cda_status
      from contdata
      where cda_from_c_id = :c_id and
            cda_from_clt_id = :clt_id and
            cda_cont_kind = :cont_kind and
            cda_is_main = :cont_is_main
      into cda_id, cda_status;
    end

    if (f_get_const('new_contdata') = 1) then -- если работаем по новому
    begin
      if (cda_id is null)
      then res_text = 'Указанная контактная информация не прикреплена';
      else
      begin
        -- если это основной номер и он уже подтвержден, не разрешаем отправлять СМС
        if (cda_status = 1 and f_enum('cont_kind', cont_kind, 'code') = 'cell') then
          res_text = 'Данный номер ранее уже подтвержлен';
      end
    end
    if (is_new_contdata = 1 and res_text = '') then
    begin
      select cvr_datetime
      from contveri
      where cvr_from_cda_id = :cda_id and
            cvr_status = 0
      order by cvr_id desc
      rows 1
      into :last_datetime;
    end
  end

  if (res_text = '' and last_datetime is not null and
      datediff(minute, last_datetime, current_timestamp) < 3) then
    res_text = 'После последней отправки кода еще не прошло 3 минуты';

  -- Если была ошибка, завершаем работу
  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  -- Добавляем проверку для email: если уже верифицирован у другого клиента
  other_client_count = 0;

  if (cont_kind_code = 'email' and db_id in (81001, 39801)) then
  begin
    if (is_new_contdata in (0, 2)) then -- старый способ
    begin
      select count(*)
      from contverify
      where cv_object_kind = f_objkind('clients', 'id') and
            cv_from_rd_id = :rd_id and
            cv_contact_value = :cont_value and
            cv_status > 0 and
            cv_object_id <> :c_id
      into :other_client_count;
    end
    if (is_new_contdata in (1, 2)) then -- новый способ
    begin
      clt_id = f_cont_id(cont_value, 'null');
      if (clt_id is not null) then
      begin
        select count(*)
        from contdata
        where cda_from_clt_id = :clt_id and
              cda_cont_kind = :cont_kind and
              cda_status = 1 and
              cda_is_delete = 0 and
              cda_from_c_id <> :c_id
        into :other_client_count;
      end
    end
    if (other_client_count > 0) then
    begin
      res_text = 'Данный e-mail верифицирован у другого клиента. Измените e-mail';
      suspend;
      exit;
    end
  end

  -- Регистрация нового пароля для будущего подтверждения
  pin_code = right('0000' || cast(trunc(rand() * 100000) as integer), 4);
  cv_password_encoded = f_coder_pass(pin_code);

  if (is_new_contdata in (0, 2)) then -- если работаем по старому
  begin
    insert into contverify (cv_object_kind, cv_object_id, cv_from_rd_id, cv_contact_value, cv_password, cv_status,
                            cv_from_un_id)
    values (f_objkind('clients', 'id'), :c_id, :rd_id, :cont_value, :cv_password_encoded, 0, :un_id)
    returning (cv_id)
    into cv_id;

    eds_uuid = f_uuid('contverify', cv_id);
  end

  if (is_new_contdata in (1, 2)) then -- если работаем по новому
  begin
    insert into contveri (cvr_from_cda_id, cvr_hash_pin, cvr_status, cvr_sign_kind, cvr_from_un_id)
    values (:cda_id, :cv_password_encoded, 0, 0, :un_id)
    returning (cvr_id)
    into cvr_id;

    if (is_new_contdata = 1) then
      eds_uuid = f_uuid('contveri', cvr_id);
  end

  msg_id = next value for gen_fsrobot_msgbox_id;

  -- отправляем сообщение сразу
  msg_subject = '';
  msg_body = '';
  c_name_fio = '';
  c_sex = 2;

  link_params = case db_id
    when 81001 then 'https://edoc.molbulak.ru/'
    when 41701 then 'https://edoc.molbulak.kg/'
    when 94401 then 'https://edoc.molbulak.com/'
    when 39801 then 'https://edoc.mbulak.kz/'
  end;
  link_params = trim(link_params);

  link_params = link_params || eds_uuid || '/' || iif(db_id = 39801, 'check-pin-code', msg_id || '/');
  if (cont_kind_code = 'email') then
    link_params = '<a href="' || link_params || '">' || link_params || '</a>';

  if (cont_kind_code = 'cell') then
  begin
    if (db_id <> 94401)
    then msg_body = 'Podtverdite nomer telefona otpraviv kod ' || pin_code || ' na nomer ' || f_get_const('company_cell') || ' ili najmite ' || link_params;
    else msg_body = 'Nomreni tesdiqlemek uchun ' || pin_code || ' kodunu ' || f_get_const('company_cell') || ' nomresine gonderin ve ya linki achin: ' || link_params;
  end

  if (cont_kind_code = 'email') then
  begin
    select c_name_fio, c_sex
    from clients
    where c_id = :c_id
    into c_name_fio, c_sex;

    if (db_id <> 94401) then
    begin
      msg_subject = 'Подтвердить адрес электронной почты';

      if (c_name_fio <> '') then
      begin
        if (c_sex = 0)
        then msg_body = 'Уважаемая ' || c_name_fio || ',<br><br>';
        else msg_body = 'Уважаемый ' || c_name_fio || ',<br><br>';
      end

      msg_body = msg_body || 'Компания M Bulak просит подтвердить ваш электронный адрес. <br>';
      msg_body = msg_body || 'Ваш код для подтверждения email: <strong>' || pin_code || '</strong><br>';
      msg_body = msg_body || 'Пройдите по ссылке и подтвердите email: ' || link_params || '<br><br>';
      msg_body = msg_body || 'Если Вы подключены к сети компании, также можете подтвердить ваш электронный адрес перейдя по данной ';
      msg_body = msg_body || '<br>Если Вы это сообщение получили ошибочно, просим проигнорировать или связаться с представителем компании.';
    end else
    begin
      msg_subject = 'Email təsdiqlənməsi';

      if (c_name_fio <> '') then
        msg_body = 'Hörmətli ' || c_name_fio || ',<br><br>';

      msg_body = msg_body || 'M Bulak BOKT elektron poçt ünvanını təsdiq etməyinizi xahiş edir. <br>';
      msg_body = msg_body || 'Poçt ünvanın təsdiqi üçün şifrəniz: <strong>' || pin_code || '</strong><br>';
      msg_body = msg_body || 'Linke keçin və email təsdiqləyin: ' || link_params || '<br><br>';
      msg_body = msg_body || 'Əgər təşkilatın internet şəbəkəsinə qoşulusunuz, elektron poçt ünvanınızı qeyd olunan ';
      msg_body = msg_body || '<br>Əgər bu mesaj səhvən göndərilib, nəzərə almamağınız xahiş olunur və ya təşkilatın əməkdaşı ilə əlaqə saxlıyın.<br><br>';
    end
  end

  if (msg_body <> '') then
  begin
    insert into fsrobot_msgbox (
      msg_id, msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days,
      msg_from_un_id, msg_command_kind, msg_object_kind, msg_object_id,
      msg_sender)
    values (:msg_id, iif(:cont_kind_code = 'email', 4, 3), :cont_value,
      :msg_subject, :msg_body, 7, :un_id, 'hp_contveri_send',
      f_objkind('clients', 'id'), :c_id, 2);
  end

  res_is_error = 0; -- если дошли до этой точки, значит нет ошибок
  suspend;
end
