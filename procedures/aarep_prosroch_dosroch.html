CREATE PROCEDURE AAREP_PROSROCH_DOSROCH(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_DISB_DATE DATE,
    K_KREDLINE_END DATE,
    KS_DATE_BEGIN DATE,
    KS_DATE_END DATE,
    KAL_NEW_KF_NAME VARCHAR(100),
    KAL_NEW_KF_OFFICER DECIMAL(18,0),
    KF_OFFICER_NAME VARCHAR(100),
    KF_REGION_ID INTEGER,
    KF_REGION_NAME VARCHAR(50),
    KF_BRANCH_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(20),
    K_AMOUNT_APPROVED DECIMAL(18,0),
    OK_SUM D_CURRENCY,
    OK_OL_DATE D_DATE)
AS
declare variable reg_prior integer;
declare variable parent_id integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable ok_os decimal(15,2);
declare variable ok_perc decimal(15,2);
declare variable ok_fine decimal(15,2);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  reg_prior = 0;
  for select distinct k_id, k_disb_date, k_kredline_end, k_amount_approved, ks_date_begin, ks_date_end, kal_new_kf_name,
                      kal_new_kf_officer, c_name_fio, kal_new_reg_id, reg_name, k_kontract_no, parent
      from kontstate, kontract, operkred, knaktlist, region, region_list(:report_region_id, 1, -1, 1), clients
      where ks_from_k_id = k_id and
            ks_state = 2 and
            ks_date_begin < k_kredline_end and
            ok_from_k_id = k_id and
            ok_fine <> 0 and
            kal_from_kf_id = k_from_kf_id and
            :report_date_begin >= kal_date and
            :report_date_end <= kal_date_end and
            (id_list = reg_unicode) and
            kal_new_reg_id = reg_unicode and
            ks_date_begin between :report_date_begin and :report_date_end and
            kal_new_kf_officer = c_id and
            knaktlist.is_delete = 0 and
            ok_from_c_id = 0
      into :k_id, :k_disb_date, :k_kredline_end, :k_amount_approved, :ks_date_begin, :ks_date_end, :kal_new_kf_name,
           :kal_new_kf_officer, :kf_officer_name, :kf_region_id, :kf_region_name, :k_kontract_no, :parent_id
  do
  begin
    :ok_sum = 0;

    select -sum(coalesce(ok.ok_os, 0)), -sum(coalesce(ok.ok_perc, 0)), -sum(coalesce(ok.ok_fine, 0))
    from operkred ok
    where ok.ok_from_k_id = :k_id and
          ok.ok_type = 4 and
          ok.ok_ol_date between :report_date_begin and :report_date_end
    into ok_os, ok_perc, ok_fine;

    :ok_sum = ok_os + ok_perc + ok_fine;

    select ok.ok_ol_date
    from operkred ok
    where ok.ok_from_k_id = :k_id and
          ok.ok_type = 4 and
          ok.ok_ol_date between :report_date_begin and :report_date_end
    order by ok.ok_ol_date desc
    rows 1
    into ok_ol_date;

    if (parent_id <> reg_prior) then
    begin
      select reg_name
      from region
      where reg_unicode = :parent_id
      into :kf_branch_name;
      reg_prior = parent_id;
    end

    suspend;
  end
end
