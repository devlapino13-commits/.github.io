CREATE PROCEDURE HP_FSROBOT_KK_DO_PROCESS
AS
declare variable acl_id d_id;
declare variable c_id d_id;
declare variable is_online d_integer;
declare variable rd_online d_integer;
declare variable task_count d_integer;
declare variable role_underwriter d_integer;
begin
  -- Распределение свободных задач на работающих андеррайтеров

  select rd_id
  from refdata
  where rd_code = 'ObjectProps.KK_Staff_Online'
  into rd_online;

  select rn_id
  from rolename
  where rn_name = 'ru_monitoring'
  into role_underwriter;

  if (rd_online is null) then
    exit;

  -- Внешний цикл очереди получаем андеррайтеров
  for with
  ru_monitoring as (
    select un_id
    from permissions, username
    where
      pm_object_kind = 0 and
      pm_object_id = un_id and
      pm_subject_kind = 1 and
      pm_subject_id = :role_underwriter and
      pm_is_delete = 0 and
      (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
      (coalesce(pm_date_end, '31.12.2999') >= current_date) and
      pm_access_kind = 1),
  fs_users as (
    select un_id as fs_un_id from ru_monitoring, hp_permission_access_check(un_id, 'Permactions', 0, 'FinStart.Login')
    where has_access = 1)
  select un_from_c_id from username, fs_users
  where un_id = fs_un_id
  into c_id do
  begin
    -- Проверяем что работает
    select ovs_add_value
    from hp_objectprops_get('clients', :c_id, -1, :rd_online, null)
    into is_online;
    is_online = coalesce(is_online, 1);
    if (is_online <> 1) then
      continue;

    -- Проверяем количество задач
    select sum(iif(aco_status = 0, 1, 0))
    from refdata
      inner join actionorder on aco_from_rd_id = rd_id
      left join actionmark on atm_from_acl_id = aco_id
      left join actionlist on acl_id = aco_id
    where aco_respons_c_id = :c_id and
          rd_main_code = 'ActionGroup' and
          rd_str_id = 'UnderWriter' and
          atm_from_acl_id is null
    into task_count;
    if (task_count > 1) then
      continue;

    -- Нашли бездельника, готовим ему две задачи
    -- На будущее лучше бы раздавать по одной, иначе первые в списке будут запахиваться
    for select aco_id
    from refdata
      inner join actionorder on aco_from_rd_id = rd_id
      left join actionmark on atm_from_acl_id = aco_id
      left join actionlist on acl_id = aco_id
    where
      rd_main_code = 'ActionGroup' and
      rd_str_id = 'UnderWriter' and
      aco_respons_c_id is null and
      aco_status = 0 and
      atm_from_acl_id is null
    order by acl_last_datetime asc
    rows 1
    into acl_id do
    begin
      execute procedure hp_underwriter_set_to_staff(1000000000000, acl_id, c_id);
    end
  end
end
