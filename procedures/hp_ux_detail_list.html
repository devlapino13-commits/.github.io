CREATE PROCEDURE HP_UX_DETAIL_LIST(
    UX_CODE D_STR_CODE NOT NULL,
    SHOW_LINK_KIND D_SMALLINT)
RETURNS (
    RES_ID D_STR_255,
    RES_PARENT_ID D_STR_255,
    RES_KIND D_STR_CODE,
    RES_IMAGE D_SMALLINT,
    RES_NAME D_STR_100,
    RES_CODE D_STR_100,
    RES_MEMO D_STR_LONG,
    RES_IS_OWN D_BOOL,
    RES_KIND_CAPTION D_STR_CODE,
    RES_TABLE_NAME D_STR_CODE,
    RES_DATETIME D_DATETIME,
    RES_WARNING_STATE D_SMALLINT)
AS
declare variable ux_root d_str_255;
declare variable ux_root_frames d_str_255;
declare variable ux_root_forms d_str_255;
declare variable ux_root_new d_str_255;
declare variable ux_root_link d_str_255;
declare variable ufi_master_ufi_code d_str_255;
declare variable tmp_root d_str_255;
declare variable tmp_code d_str_code;
declare variable ux_kind d_str_code;
declare variable table_name d_str_code;
declare variable tmp_table d_str_code;
declare variable tmp_code2 d_str_code;
declare variable us_found d_bool;
declare variable orig_res_id d_str_255;
declare variable any_found d_bool;
declare variable link_found d_bool;
begin
  res_image = 0; -- папка
  res_is_own = 1;
  link_found = 0;
  res_warning_state = 0;

  if (show_link_kind is null) then
    show_link_kind = 0;

  -- show_link_kind:
  -- 0 - Не показывать
  -- 1 - Показывать внутри ветки
  -- 2 - Показывать в отдельной ветка

  -- Images
  -- 4 - Свой поисковик
  -- 27 - Чужой поисковик

  -- 17 - Свой фрейм
  -- 18 - Чужой фрейм
  -- 19 - Своя форма
  -- 53 - Чужая форма
  -- 21 - Таблица
  -- 22 - Набор фреймов
  -- 23 - Набор форм
  -- 24 - Неиспользуемые

  -- 25 - Команды
  -- 26 - Бланки


  if (trim(coalesce(ux_code, '')) = '') then
    exit;

  select ux_table_name,
         iif(position(ut_caption, ut_table_name) = 0,
             iif(position(ut_table_name, ut_caption) = 0, ut_table_name || '  - ' || ut_caption, ut_caption),
             ut_table_name)
  from uxcode, uxtable
  where ux_code = :ux_code and
        ux_table_name = ut_table_name
  into table_name, res_name;

  if (table_name is null) then
  begin
    select ut_table_name,
           iif(position(ut_caption, ut_table_name) = 0,
               iif(position(ut_table_name, ut_caption) = 0, ut_table_name || '  - ' || ut_caption, ut_caption),
               ut_table_name)
    from uxtable
    where ut_table_name = :ux_code
    into table_name, res_name;

    if (table_name is null) then
      exit;
  end

  res_kind_caption = 'Таблица';
  res_table_name = table_name;
  res_datetime = null;

  res_code = table_name;
  res_kind = 'table';
  res_memo = 'Формы и интерфейсы таблицы';
  res_id = table_name;
  ux_root = res_id;
  res_parent_id = null;
  res_image = 21;

  if (res_name is null) then
    res_name = table_name;

  suspend;

  res_kind_caption = '';
  res_memo = '';
  res_table_name = '';
  res_datetime = null;

  tmp_code = res_code;

  res_id = tmp_code || '.forms';
  res_code = '';
  ux_root_forms = res_id;
  res_parent_id = ux_root;
  res_kind = 'table.forms';
  res_name = 'Формы';
  res_memo = 'Формы принадлежащие к таблице и используемые в формах';
  res_image = 23;
  suspend;

  res_id = tmp_code || '.frames';
  res_code = '';
  ux_root_frames = res_id;
  res_parent_id = ux_root;
  res_kind = 'table.frames';
  res_name = 'Фреймы';
  res_memo = 'Фреймы принадлежащие к таблице и используемые в формах';
  res_image = 22;
  suspend;

  res_id = tmp_code || '.new';
  res_code = '';
  ux_root_new = res_id;
  res_parent_id = ux_root;
  res_kind = 'table.new';
  res_name = 'Неиспользуемые';
  res_memo = 'Неиспользуемые формы и фреймы';
  res_image = 24;
  suspend;

  ux_root_link = null;

  if (show_link_kind = 2) then -- Отдельная ветка для связей
  begin
    res_id = tmp_code || '.link';
    res_code = '';
    ux_root_link = res_id;
    res_parent_id = ux_root;
    res_kind = 'table.new';
    res_name = 'Ссылки на объекты';
    res_memo = 'Использование форм и фреймов в других таблицах';
    res_image = 31;
  end

  for select ux_code,
             ux_memo,
             ux_name,
             uxf$kind(ux_kind, 'code'),
             uxf$kind(ux_kind, 'caption'),
             ux_create_datetime
  from uxcode
  where ux_table_name = :table_name
  into res_code, res_memo, res_name, ux_kind, res_kind_caption, res_datetime
  do
  begin
    -- Фреймы и формы таблицы
    res_table_name = table_name;

    res_id = res_code;
    res_kind = ux_kind;
    res_parent_id = ux_root_new;
    res_is_own = 1;

    if (ux_kind = 'uxframe') then
    begin
      if ((exists(select *
                  from uxcodeframe
                  where ucf_ufr_code = :res_code)) or
          (exists(select *
                  from uxcodeframe
                  where ucf_master_ufr_code = :res_code))) then
        res_parent_id = ux_root_frames;

      res_image = 17;
    end else
    begin
      if (exists(select *
                  from uxfritem
                  where ufi_ux_code = :res_code)) then
        res_parent_id = ux_root_forms;

      res_image = 19;
      res_image = case ux_kind
        when 'uxform' then 19
        when 'uxcommand' then 25
        when 'uxreport' then 26
      end;

      if (exists(select *
                from uxalias, uxsearch
                where uxa_ux_code = :res_code and
                      uxa_code = usr_code)) then
        res_image = 4;
    end

    suspend;

    res_parent_id = res_id;
    res_is_own = 0;

    tmp_root = res_id;
    tmp_code = res_code;

    if (ux_kind <> 'uxframe') then
    begin -- ux_kind <> 'uxframe'
      res_memo = 'Фрейм используемый в форме';

      res_kind = 'uxframe';

      for select :tmp_code || '#' || fr_code, fr_code, :tmp_code || '#' || master_frame
      from hp_uxframe_md_tree(:tmp_code, null)
      into :res_id, :res_code, :ufi_master_ufi_code
      do
      begin
        res_table_name = '';
        res_kind_caption = '';
        res_memo = '';
        res_datetime = null;

        res_image = 17;
        res_is_own = 1;
        res_name = 'Объект не найден';
        tmp_table = '';

        res_memo = 'Фрейм испольщуемый в форме';

        if (ufi_master_ufi_code is null) then
          res_parent_id = tmp_root;

        select ux_table_name,
               ux_memo,
               ux_name,
               uxf$kind(ux_kind, 'caption'),
               ux_create_datetime
        from uxcode
        where ux_code = :res_code
        into tmp_table, res_memo, res_name, res_kind_caption, res_datetime;

        res_table_name = tmp_table;

        if (tmp_table <> table_name) then
        begin
          res_is_own = 0;
          res_image = 18;
        end

        suspend;
      end
    end

    res_table_name = '';
    res_kind_caption = '';
    res_memo = '';
    res_datetime = null;

    if (show_link_kind = 0) then
      continue;

    any_found = 0;
    if (show_link_kind = 2) then
    begin
      res_id = 'link.' || res_id;
      res_code = tmp_code;
      res_parent_id = ux_root_link;
      res_is_own = 1;

      orig_res_id = res_id;
--      suspend;
  
      tmp_root = res_id;
      res_parent_id = tmp_root;
    end

    if (ux_kind = 'uxframe' and show_link_kind <> 0) then
    begin
      res_memo = 'Форма из другой таблицы использующая фрейм выше';
      res_image = 53;

      for select distinct ux_code, uxf$kind(ux_kind, 'kind')
      from uxcodeframe, uxcode
      where (ucf_ufr_code = :tmp_code or ucf_master_ufr_code = :tmp_code) and
             ucf_ux_code = ux_code and
             ux_table_name <> :table_name
      into tmp_code2, res_kind
      do
      begin
        res_id = null;

        res_image = case res_kind
          when 'uxform' then 53
          when 'uxcommand' then 28 -- также как и свой
          when 'uxreport' then 29   -- также как и свой
        end;

        if (exists(select *
                  from uxaction, uxalias
                  where uxa_ux_code = :tmp_code2 and
                        uxa_code = ua_uxa_code)) then
          res_image = 27;

        select ufi_code, ux_code, ux_memo, ux_name,
               ux_table_name,
               uxf$kind(ux_kind, 'caption'),
               ux_create_datetime
        from uxfritem, uxcode
        where ufi_ux_code = :tmp_code2 and
              ufi_parent_code is null and
              ufi_ux_code = ux_code
        into res_id, res_code, res_memo, res_name, res_table_name, res_kind_caption, res_datetime;

        if (res_id is not null) then
        begin
          any_found = 1;
          suspend;
        end
      end
    end

    res_table_name = '';
    res_kind_caption = '';
    res_memo = '';
    res_datetime = null;

    if (ux_kind = 'uxform' and show_link_kind <> 0) then
    begin
      if (show_link_kind = 1) then
      begin
        res_parent_id = tmp_code || '.find';
        res_memo = 'Фрейм из другой таблицы использующий эту форму';
      end

      res_kind = 'uxframe';
      res_image = 18;
      res_is_own = 0;

      us_found = 0;


      for select ufp_code, ufp_ufr_code
      from uxalias, uxaction, uxfrparam
      where uxa_ux_code = :tmp_code and
            ua_uxa_code = uxa_code and
            ua_ux_code = ufp_ufr_code and
            ua_usr_id_field = ufp_origin_name
      into :res_id, :res_code
      do
      begin
        us_found = 1;

        select ux_name,
               ux_table_name,
               ux_memo,
               uxf$kind(ux_kind, 'caption'),
               ux_create_datetime
        from uxcode
        where ux_code = :res_code
        into res_name, res_table_name, res_memo, res_kind_caption, res_datetime;

        any_found = 1;
        suspend;
      end

      res_table_name = '';
      res_kind_caption = '';
      res_memo = '';
      res_datetime = null;

      if (us_found = 1 and show_link_kind = 1) then
      begin
        res_is_own = 0;
        res_id = tmp_code || '.find';
        res_code = '';
        res_parent_id = tmp_root;
        res_kind = 'uxsearch.find';
        res_name = 'Участие в редакторах поиска';
        res_memo = 'Использование формы в качестве поисковика';
        res_image = 30;
        suspend;
      end
    end

    if (any_found = 1 and show_link_kind = 2) then
    begin
      link_found = 1;
      res_id = orig_res_id;
      res_code = tmp_code;
      res_parent_id = ux_root_link;
      res_is_own = 1;
      res_memo = '';
      suspend;
    end
  end

  res_table_name = '';
  res_kind_caption = '';
  res_memo = '';
  res_datetime = null;

  if (link_found = 1 and show_link_kind = 2) then
  begin
    res_id = tmp_code || '.link';
    res_code = '';
    ux_root_link = res_id;
    res_parent_id = ux_root;
    res_kind = 'table.new';
    res_name = 'Ссылки на объекты';
    res_memo = 'Использование форм и фреймов в других таблицах';
    res_image = 31;
    suspend;
  end

end
