CREATE PROCEDURE MC_COMMON_REF_GET(
    MCL_ID ID_TYPE,
    LANG D_STR_100 = 'ru')
RETURNS (
    COMMON_REF_JSON D_BLOB_TEXT)
AS
declare variable db_id d_integer;
declare variable user_code d_integer;
declare variable mcl_project d_str_20;
declare variable new_products_active d_bool;
begin
    db_id = f_get_db_id();
    new_products_active = cast(f_get_const('kmp_new_product_list_active') as integer);

    lang = coalesce(lang, '');
    if (db_id = 94401) then lang = 'az';
    if (lang = '') then lang = 'ru';

    -- json для окна "Гость"
    if (mcl_id = -1) then
    begin
        common_ref_json = '{';
        common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('picture_list', :lang)||',');
        common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('company_phone_list', :lang)||',');

        if (db_id = 94401) then
            common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('company_mail_list', :lang)||',');

        common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('Information_for_clients', :lang)||',');
        common_ref_json = BLOB_APPEND(common_ref_json, f_get_json_from_sql('office_list', :db_id)||'');
        common_ref_json = BLOB_APPEND(common_ref_json,'}');

        suspend;
        exit;
    end

    common_ref_json = '{';
    common_ref_json = BLOB_APPEND(common_ref_json, f_get_json_from_sql('city_list', :db_id)||',');

    if (db_id <> 81001) then
    begin
        common_ref_json = BLOB_APPEND(common_ref_json, f_get_json_from_sql('office_list', :db_id)||',');
        common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('company_phone_list', :lang)||',');
        common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('reliable_client', :lang)||',');
        common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('Information_for_clients', :lang)||',');
        
        if (new_products_active = 1) then
            common_ref_json = BLOB_APPEND(common_ref_json, (select product_list_json from hp_get_product_list_json(:lang))||',');
        else
            common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('product_list', :lang)||',');

        common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('currency_str', :lang)||',');
        common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('paysis_projects', :lang)||',');   -- выбор оплаты
    end 

    if (db_id in (39801, 41701)) then
    begin
        select ml.mcl_user_code
        from mc_login ml
        where ml.mcl_id = :mcl_id
        into user_code;
    
        if (user_code = 1) then
            common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('installment_sale_agent', :lang)||','); -- товары в рассрочку для агента
        else
            common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('installment_sale_client', :lang)||','); --товары в рассрочку для клиекнта
    
        common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('my_point', :lang)||',');   -- раздел мои баллы только для КГ
        common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('information_for_partners', :lang)||',');   -- Информация для партнеров для КГ
        common_ref_json = BLOB_APPEND(common_ref_json, f_get_json_from_sql('magazine_list', :db_id)||',');
        common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('loan_limit', :lang))||','); -- кредитный лимит
        common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('online_loan', :lang))||',');
        common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('doc_links', :lang))||',');
        common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('ces_description', :lang))||',');
        common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('credit_limit_description', :lang))||',');
        common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('calc_settings', :lang))||',');
    end
    else if (db_id = 81001) then
    begin
        -- RFCRU-3766
        select f_enum('mcl_project', mcl.mcl_project, 'code', 0)
        from mc_login mcl
        where mcl.mcl_id = :mcl_id
        into mcl_project;

        common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('payment_and_transfer_list', :lang))||',');
        common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('payment_description', :lang))||','); -- информация по оплате

        if (mcl_project = 'kmp') then
        begin
            -- RFCRU-4242
            common_ref_json = BLOB_APPEND(common_ref_json, f_get_json_from_sql('office_list', :db_id, :mcl_project)||',');
            common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('company_phone_list', :lang))||',');

            if (new_products_active = 1) then
                common_ref_json = BLOB_APPEND(common_ref_json, (select product_list_json from hp_get_product_list_json(:lang))||',');
            else
                common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('product_list', :lang))||',');

            common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('partner_program', :lang))||',');
            common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('reliable_client', :lang))||',');
            common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('Information_for_clients', :lang))||',');
            common_ref_json = BLOB_APPEND(common_ref_json, (f_get_common_json('credit_limit_description', :lang))||',');
        end

        if (mcl_project = 'oivo') then
        begin
            -- RFCRU-4242
            common_ref_json = BLOB_APPEND(common_ref_json, f_get_json_from_sql('office_list', :db_id)||',');
            common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('oivo_company_phone_list', :lang)||',');

            if (new_products_active = 1) then
                common_ref_json = BLOB_APPEND(common_ref_json, (select product_list_json from hp_get_product_list_json(:lang, 'oivo'))||',');
            else
                common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('oivo_product_list', :lang)||',');

            common_ref_json = BLOB_APPEND(common_ref_json, f_get_common_json('oivo_information_for_clients', :lang)||',');
        end
    end

    common_ref_json = BLOB_APPEND(common_ref_json, f_get_json_from_sql('passkind', :db_id)||'');
    common_ref_json = BLOB_APPEND(common_ref_json,'}');

    suspend;
end
