CREATE PROCEDURE FM_HTML_CFP_INFO(
    MAC_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable C_ID D_ID;
declare variable MA_ID D_ID;
declare variable C_NAME_FIO D_STR_255;
declare variable LANG D_STR_CODE;
declare variable NAME D_STR_255;
declare variable TYPE_NAME D_STR_255;
declare variable CP_PARENT_ID D_ID;
declare variable CP_PARENT_TMP D_ID;
declare variable NET_PROFIT D_INTEGER;
declare variable EXPENSE D_INTEGER;
declare variable EX_PROFIT D_INTEGER;
declare variable CQ_CODE_NAME D_STR_255;
declare variable CQ_NAME D_STR_255;
declare variable CQ_BASIC_INCOM D_SMALLINT;
declare variable ANSW_VALUE D_INTEGER;
declare variable K_TEMP D_ID;
declare variable AMOUNT_MONTHLY_PAYMENTS decimal(18,0);
declare variable REPAYMENT_OBLIGATIONS decimal(18,0);
declare variable KBM_ID D_ID;
declare variable SUM_DEBT decimal(18,0);
declare variable K_ID D_ID;
declare variable CA_ID D_ID;
declare variable CP_ID D_ID;
declare variable KPD decimal(15,2);
declare variable CW_IS_WORKING D_BOOL;
declare variable CW_ID D_ID;
declare variable MONTHLY_INCOME D_CURRENCY;
declare variable DB_ID D_INTEGER;
declare variable KDD D_CURRENCY;
declare variable acl_id d_id;
begin
  db_id = f_get_db_id();

  if(coalesce(mac_id, 0) < 0) then    -- RFCKG-5565
  begin
    acl_id = mac_id * -1;

    select acl.acl_object_id
    from actionlist acl
    where acl.acl_id = :acl_id
    into mac_id;
  end

  for select mac.mac_from_c_id, mac.mac_from_ma_id
      from miniappclient mac
      where mac.mac_id = :mac_id and
            mac.mac_amount > 0
      into c_id, ma_id
  do
  begin
    select ca.ca_id, ca.ca_from_cp_id, ca.ca_from_cw_id
    from cfp_anketa ca
    where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
          ca.ca_object_id = :mac_id and
          ca.ca_from_c_id = :c_id and
          (
            coalesce(:acl_id, 0) = 0                    -- RFCKG-5566
            or ca.ca_from_acl_id = :acl_id
          ) and
          (
            coalesce(:acl_id, 0) <> 0
            or ca.ca_from_acl_id is null
          )
    into ca_id, cp_id, cw_id;

    select cw.cw_is_working
    from cib_working cw
    where cw.cw_id = :cw_id
    into cw_is_working;

    select c.c_name_fio, c.c_blank_lang
    from clients c
    where c.c_id = :c_id
    into c_name_fio, lang;

    res_html = (select res_style
                from fm_html_styles) || f_format('<body><br><h2>%1</h2><br>', 'Анализ доходов/расходов') || f_format('<table width="100%"><col width="40%"><col width="60%">');
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'ФИО клиента') || '</b>', :c_name_fio, '', '', -1, -1, -1, -1, 2, ''));

    name = '';
    type_name = '';

    select iif(:lang = 'kg', cp.cp_name_kg, cp.cp_name_ru), cp.cp_parenet_id,
           f_lang(:lang, f_enum_name('cfp_positions', cp.cp_type))
    from cfp_positions cp
    where cp.cp_id = :cp_id
    into name, cp_parent_id, type_name;

    res_html = blob_append(res_html, f_html_table_row('<b>' || :type_name || '</b>', :name, '', '', -1, -1, -1, -1, 2, ''));
    while (coalesce(cp_parent_id, 0) > 0) do
    begin
      select iif(:lang = 'kg', cp.cp_name_kg, cp.cp_name_ru), cp.cp_parenet_id,
             f_lang(:lang, f_enum_name('cfp_positions', cp.cp_type))
      from cfp_positions cp
      where cp.cp_id = :cp_parent_id
      into name, cp_parent_tmp, type_name;

      res_html = blob_append(res_html, f_html_table_row('<b>' || :type_name || '</b>', :name, '', '', -1, -1, -1, -1, 2, ''));

      cp_parent_id = cp_parent_tmp;
    end


    -- RFCKG-4242: Добавление поля "Ежемесячный основной доход" в анкету анализ доходов/расходов
    if (f_get_db_id() = 41701) then
    begin
        select ca.ca_avr_income
        from cfp_anketa ca
        where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
              ca.ca_object_id = :mac_id and
              ca.ca_from_c_id = :c_id and
              (
                coalesce(:acl_id, 0) = 0                    -- RFCKG-5566
                or ca.ca_from_acl_id = :acl_id
               ) and
              (
                coalesce(:acl_id, 0) <> 0
                or ca.ca_from_acl_id is null
              )
        into monthly_income;

        if (monthly_income is not null) then
        begin
            res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'Ежемесячный основной доход') || '</b>', round(:monthly_income, 2), '', '', -1, -1, -1, -1, 2, ''));
        end
    end


    net_profit = 0;
    expense = 0;
    ex_profit = 0;
    for select cq.cq_code_name, iif(:lang = 'kg', cq.cq_name_kg, cq.cq_name_ru), cq.cq_basic_incom, caa.caa_value
        from cfp_anketa_answers caa
        inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id
        where caa.caa_from_ca_id = :ca_id
        order by cq.cq_basic_incom desc, cq.cq_id
        into cq_code_name, cq_name, cq_basic_incom, answ_value
    do
    begin
      res_html = blob_append(res_html, f_html_table_row('<b>' || :cq_name || '</b>', :answ_value, '', '', -1, -1, -1, -1, 2, ''));

      if (cq_code_name in ('monthly_housing_expenses', 'monthly_family_expenses', 'monthly_tax_expenses', 'monthly_other_expenses')) then
      begin
        if (expense = 0) then
          expense = :answ_value;
        else
          expense = expense + :answ_value;
      end
      if (cq_basic_incom = 1) then
      begin
        if (net_profit = 0) then
          net_profit = answ_value;
        else
        begin
          if (:db_id <> 39801) then
            net_profit = net_profit * :answ_value;
          else
            net_profit = net_profit + :answ_value;
        end
      end
      if (cq_code_name in ('cl_add_income_spouse_salary', 'cl_add_income_household', 'cl_add_income_service')) then
        ex_profit = ex_profit + :answ_value;
    end
    select kpd, repayment_obligations, net_profit, amount_monthly_payments
    from get_cfp_kpd(:mac_id)
    into :kpd, :repayment_obligations, :net_profit, :amount_monthly_payments;

    if (:db_id = 39801) then
    begin
      select cakdn.cakdn_kdn,
             cakdn.cakdn_kdd,
             cakdn.cakdn_net_profit,
             cakdn.cakdn_other_dept,
             cakdn.cakdn_monthly_payment
      from cfp_anketa_kdn cakdn
      where cakdn.cakdn_from_mac_id = :mac_id
      rows 1
      into kpd,
           kdd,
           net_profit,
           repayment_obligations,
           amount_monthly_payments;

      kpd = coalesce(kpd, 0);
      kdd = coalesce(kdd, 0);
      net_profit = coalesce(net_profit, 0);
      repayment_obligations = coalesce(repayment_obligations, 0);
      amount_monthly_payments = coalesce(amount_monthly_payments, 0);
    end

    if (cw_is_working = 1) then
      res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'Погашение других долговых обязательств') || ': </b>', :repayment_obligations, '', '', -1, -1, -1, -1, 2, ''));
    if (:db_id <> 39801) then
      res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'Чистая прибыль в месяц') || ': </b>', :net_profit, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'Сумма ежемесячной выплаты') || ': </b>', :amount_monthly_payments, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, iif(db_id <> 39801, 'КПД', 'КДН')) || ': </b>', iif(db_id = 39801, round(:kpd, 2), :kpd), '', '', -1, -1, -1, -1, 2, ''));
    if (:db_id = 39801) then
      res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'КДД') || ': </b>', round(:kdd, 2), '', '', -1, -1, -1, -1, 2, '')); -- RFCKZ-56
    res_html = blob_append(res_html, '</table><br>');
    suspend;
  end
end
