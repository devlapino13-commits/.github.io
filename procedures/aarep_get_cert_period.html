CREATE PROCEDURE AAREP_GET_CERT_PERIOD(
    REP_OFFICE_ID INTEGER,
    REP_DATE DATE,
    REP_DELTA INTEGER,
    REP_MA_ID INTEGER)
RETURNS (
    REP_TIME TIMESTAMP,
    REP_GROUP_COUNT INTEGER,
    REP_MEMO VARCHAR(255),
    REP_SELECTED INTEGER)
AS
declare variable rep_time_value decimal(18,4);
declare variable tmp_min integer;
declare variable tmp_hour integer;
begin
-- Возвращает время сертификаций в офисе на определенную дату

  rep_selected = 0;
  for select rep_time_value, sum(rep_group_count), sum(rep_selected), left(list(rep_memo, '; '), 255)
      from aarep_get_cert_period_tmp(:rep_office_id, :rep_date, :rep_delta, :rep_ma_id)
      group by rep_time_value
      into :rep_time_value, :rep_group_count, :rep_selected, :rep_memo
  do
  begin
    tmp_hour = ceiling(rep_time_value);
    tmp_min = mod(rep_time_value * 100, 100);
    if (tmp_min <> 0) then
      tmp_hour = tmp_hour - 1;
    rep_time = rep_date;
    rep_time = dateadd(hour, tmp_hour, rep_time);
    rep_time = dateadd(minute, tmp_min, rep_time);

    if (coalesce(rep_memo, '') <> '') then
      rep_memo = substring(rep_memo from 3 for 255);

    if ((rep_time_value <> 0) or (rep_group_count <> 0)) then
      suspend;
  end
end
