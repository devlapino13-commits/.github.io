CREATE PROCEDURE HP_NF_ANALIZ_ACCOUNTING_ENTRY(
    IFA_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable ifa_file_name d_str_100;
declare variable ifa_date d_date;
declare variable ifa_is_detail_problem d_bool;
declare variable id d_integer;
declare variable rec_no d_integer;
declare variable main_bank_id d_integer;
declare variable sub_bank_id d_integer;
declare variable subaccount d_integer;
declare variable recieved_tran d_integer;
declare variable must_pay_tran d_currency;
declare variable netto_count_trans d_integer;
declare variable netto_amount d_currency;
declare variable reg_id d_integer;
declare variable acc_dt d_str_20;
declare variable acc_issuer d_str_20;
declare variable acc_dt1 d_str_20;
declare variable acc_kt d_str_20;
declare variable ol_id d_id;
declare variable ob_detail d_str_long;
declare variable curr_code d_str_lang;
begin
  err_text = '';
  curr_code = '417';

  select
    ifa_file_name,
    ifa_date,
    ifa_is_detail_problem
  from ipc_file_analiz
  where ifa_id = :ifa_id
    and ifa_file_type = 3
    and ifa_status = 0
  into
    :ifa_file_name,
    :ifa_date,
    :ifa_is_detail_problem;
  if (row_count = 0) then
    err_text = 'Данные для построения бухгалтерских проводок не определены.';

  if (err_text = '' and ifa_is_detail_problem = 1) then
    err_text = 'Выявлены ошибки! Бухгалтерские проводки могут быть сформированы только после их исправления.';

  if (err_text = '') then
  begin
    reg_id = 1090100;
    acc_dt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '10150/101');
    acc_dt1 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '80509/10');
    acc_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '21114/13');

    insert into operlist (
      ol_from_reg_id,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_memo,
      ol_user_id
    )
    values (
      :reg_id,
      2,
      11,
      :ifa_date,
      :ifa_id,
      9,
      f_format('NF-файл "%1" от %2 г.', :ifa_file_name, f_datetostr(:ifa_date)),
      :un_id
    )
    returning ol_id
    into :ol_id;

    for
      select
        ind_id,
        ind_rec_no,
        ind_main_bank_id,
        ind_sub_bank_id,
        ib_subaccount,
        ind_recieved_tran,
        ind_must_pay_tran,
        ind_netto_count_trans,
        ind_netto_amount
      from ipc_nonfin_detail
        inner join ipc_banks
          on ib_id = ind_sub_bank_id
      where ind_from_ifa_id = :ifa_id
        and ind_error_id is null
    into
      :id,
      :rec_no,
      :main_bank_id,
      :sub_bank_id,
      :subaccount,
      :recieved_tran,
      :must_pay_tran,
      :netto_count_trans,
      :netto_amount
    do
    begin
      acc_issuer = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '10004/' || :subaccount);

      if (recieved_tran > 0) then
      begin
        ob_detail = f_format('Комиссия по нефинансовым транзакциям по пл.картам за %1 (%2)', f_datetostr(ifa_date), rec_no);

        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_detail,
          ob_summa,
          ob_summa_val
        )
        values (
          :ol_id,
          :curr_code,
          1,
          :acc_dt,
          :acc_issuer,
          :ob_detail,
          :recieved_tran,
          :recieved_tran
        );
        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_detail,
          ob_summa,
          ob_summa_val
        )
        values (
          :ol_id,
          :curr_code,
          1,
          :acc_issuer,
          :acc_kt,
          :ob_detail,
          :recieved_tran,
          :recieved_tran
        );
      end

      if (must_pay_tran > 0) then
      begin
        ob_detail = f_format('Расходы по нефинансовым транзакциям по пл.картам за %1 (%2)', f_datetostr(ifa_date), rec_no);

        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_detail,
          ob_summa,
          ob_summa_val
        )
        values (
          :ol_id,
          :curr_code,
          1,
          :acc_dt1,
          :acc_issuer,
          :ob_detail,
          :must_pay_tran,
          :must_pay_tran
        );
        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_detail,
          ob_summa,
          ob_summa_val
        )
        values (
          :ol_id,
          :curr_code,
          1,
          :acc_issuer,
          :acc_dt,
          :ob_detail,
          :must_pay_tran,
          :must_pay_tran
        );
      end
    end

    update ipc_file_analiz set
      ifa_status = 1,
      ifa_from_un_id = :un_id,
      ifa_datetime = current_timestamp,
      ifa_from_ol_id = :ol_id
    where ifa_id = :ifa_id;
  end

  suspend;
end
