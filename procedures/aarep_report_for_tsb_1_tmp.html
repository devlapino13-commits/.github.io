CREATE PROCEDURE AAREP_REPORT_FOR_TSB_1_TMP(
    REP_DATE_END DATE)
RETURNS (
    RES_CLIENT_IS_RESIDENT INTEGER,
    CLI_BALANCE CURR_TYPE,
    CLI_PERC_SALDO CURR_TYPE,
    CLI_KOMMIS_SALDO CURR_TYPE,
    CLI_RPPU_BAL_DO_28022012 CURR_TYPE,
    CLI_RPPU_BAL_POSLE_28022012 CURR_TYPE,
    CLI_RPPU_PERC_POSLE_28022012 CURR_TYPE,
    PERCENT_YEAR CURR_TYPE,
    PERCENT_EFFECT CURR_TYPE,
    K_KOLICH_REPAY INTEGER,
    K_ID DECIMAL(18,0),
    K_DISB_DATE DATE,
    LOAN_KIND SMALLINT,
    K_PERIOD_DAY INTEGER)
AS
declare variable oks_balance curr_type;
declare variable k_amount_approved curr_type;
declare variable c_id id_type;
declare variable kc_amount_approved curr_type;
declare variable dolya double precision;
declare variable rep_period integer;
declare variable kcp_perc_saldo curr_type;
declare variable kcp_kommis_saldo curr_type;
declare variable lk_balance curr_type;
declare variable lk_calcperc curr_type;
declare variable lk_percent perc_type;
declare variable rppu_bal_do_28022012 curr_type;
declare variable rppu_bal_posle_28022012 curr_type;
declare variable rppu_perc_posle_28022012 curr_type;
declare variable citizen_ru_id integer;
declare variable doc_vnj_id integer;
declare variable pr_id integer;
declare variable pr_product_kind integer;
declare variable max_sh_date date;
declare variable k_kredline_end date;
begin
  rep_period = extract(year from cast(:rep_date_end as date)) || right('0' || extract(month from cast(:rep_date_end as date)), 2);
  citizen_ru_id = null;
  select rd_id
  from refdata
  where rd_code = 'Citizen.RU'
  into :citizen_ru_id;

  doc_vnj_id = null;
  select pk_id
  from passkind, passdoctype
  where pdt_id = pk_from_pdt_id and
        pk_is_delete = 0 and
        pdt_code = 'registration_vnj' rows 1
  into :doc_vnj_id;

  for select oks_from_k_id, oks_balance
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end
            and oks_close_state = 0
      into k_id, oks_balance
  do
  begin
    select k_disb_date, k_amount_approved, k_kolich_repay, k_percent_effect_fifd, k_from_pr_id, k_kredline_end
    from kontract
    where k_id = :k_id
    into k_disb_date, k_amount_approved, k_kolich_repay, percent_effect, pr_id, k_kredline_end;

    max_sh_date = null;
    select max(sh_date)
    from schedule
    where sh_from_k_id = :k_id
    into max_sh_date;

    if (max_sh_date is null) then
      max_sh_date = k_kredline_end;

    k_period_day = max_sh_date - k_disb_date;

    select pr_product_kind
    from product
    where pr_id = :pr_id
    into pr_product_kind;

    kcp_perc_saldo = null;
    kcp_kommis_saldo = null;
    select kcp_perc_saldo, kcp_kommis_saldo
    from kontcalcperc
    where kcp_from_k_id = :k_id and
          kcp_period = :rep_period
    into kcp_perc_saldo, kcp_kommis_saldo;
    kcp_perc_saldo = coalesce(kcp_perc_saldo, 0);
    kcp_kommis_saldo = coalesce(kcp_kommis_saldo, 0);

    lk_balance = null;
    lk_calcperc = null;
    lk_percent = null;
    select lk_balance, lk_calcperc, lk_percent
    from llc_main, llc_kont
    where lm_is_signed = 1 and
          lm_id = lk_from_lm_id and
          :rep_date_end between lk_date and lk_date_end and
          lk_from_k_id = :k_id
    into lk_balance, lk_calcperc, lk_percent;
    lk_balance = coalesce(lk_balance, 0);
    lk_calcperc = coalesce(lk_calcperc, 0);
    lk_percent = coalesce(lk_percent, 0);

    rppu_bal_do_28022012 = 0;
    rppu_bal_posle_28022012 = 0;
    rppu_perc_posle_28022012 = 0;
    if (k_disb_date < '28.02.2012') then
    begin
      rppu_bal_do_28022012 = cast(lk_balance as double precision) * lk_percent / 100;
    end
    else
    begin
      rppu_bal_posle_28022012 = cast(lk_balance as double precision) * lk_percent / 100;
      rppu_perc_posle_28022012 = cast(lk_calcperc as double precision) * lk_percent / 100;
    end

    percent_year = null;
    select kd_percent_per_day * 365
    from kontdata
    where :k_disb_date between kd_ol_date and kd_date_end and
          kd_ol_is_delete = 0 and
          kd_from_k_id = :k_id
    into percent_year;

    for select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id, kc_amount_approved
    do
    begin
      dolya = cast(kc_amount_approved as double precision) / k_amount_approved;

      select 1 - res_non_resident
      from hp_get_client_type(:c_id, :citizen_ru_id, :doc_vnj_id, :rep_date_end)
      into :res_client_is_resident;

      cli_balance = dolya * oks_balance;
      cli_perc_saldo = dolya * kcp_perc_saldo;
      cli_kommis_saldo = dolya * kcp_kommis_saldo;

      cli_rppu_bal_do_28022012 = dolya * rppu_bal_do_28022012;
      cli_rppu_bal_posle_28022012 = dolya * rppu_bal_posle_28022012;
      cli_rppu_perc_posle_28022012 = dolya * rppu_perc_posle_28022012;

      if (pr_id = 44) then
        loan_kind = 2;
      else
      if (pr_product_kind = 4) then
        loan_kind = 1;
      else
        loan_kind = 3;

      suspend;
    end
  end
end
