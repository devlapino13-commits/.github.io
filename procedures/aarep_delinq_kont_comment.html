CREATE PROCEDURE AAREP_DELINQ_KONT_COMMENT(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    K_ID ID_TYPE,
    K_KONTRACT_NO STR_20,
    KF_NAME STR_100,
    CL_CASHIER_NAME STR_100,
    DISB_DATE D_DATE,
    KS_ID ID_TYPE,
    KS_NAME STR_100,
    DAYS_DELAY D_INTEGER,
    KONTRACT_COMMENT STR_255,
    KONTRACT_COMMENT_DATETIME D_DATETIME,
    KONTRACT_COMMENT_CC_CL D_STR_LONG,
    KONTRACT_COMMENT_CC_CONT D_STR_LONG)
AS
declare variable comm_datetime d_datetime;
declare variable comm_pers d_str_100;
declare variable comm_fio d_str_100;
declare variable comm_text d_str_long;
declare variable razd str_20;
declare variable comm_kind d_integer;
declare variable razd_cl str_20;
declare variable razd_cont str_20;
begin
    for select k_id, k_kontract_no, k_disb_date, kf_officer_id, kf_officer_name, rep_deliq_days_kolich,
               kf_name, cl_cashier_name
    from aarep_kontract_list(:rep_region_id, :rep_date_begin, :rep_date_end, null, 810, 5)
    into k_id, k_kontract_no, disb_date, ks_id, ks_name, days_delay,
         kf_name, cl_cashier_name
    do
    begin
        kontract_comment = null;
        kontract_comment_datetime = null;
        kontract_comment_cc_cl = '';
        kontract_comment_cc_cont = '';
        select left(obm_memo, 255), obm_datetime
        from hp_objectprops_memo_get('kontract', :k_id, 'refdata', 'objectprops.kontract_memo', 1)
        into kontract_comment, kontract_comment_datetime;

        comm_datetime = null;
        comm_pers = null;
        comm_fio = null;
        comm_text = null;
        razd = '';

        for select pcl_called_datetime, c_name_fio, pcl_memo
        from pantera_kontracts
        inner join pantera_clients on pcl_from_pan_id = pant_id   -- clients comments
        inner join clients on c_id = pcl_from_c_id
        where pant_from_k_id = :k_id
        and pcl_has_memo = 1
        order by pcl_called_datetime desc
        into comm_datetime, comm_fio, comm_text
        do
        begin
            kontract_comment_cc_cl = left(kontract_comment_cc_cl || razd || f_datetimetostr(comm_datetime) || ' - клиент - ' || comm_fio || ' - ' || trim(coalesce(comm_text, '')), 1024);
            razd = ascii_char(10);
        end

        comm_datetime = null;
        comm_pers = null;
        comm_fio = null;
        comm_text = null;
        razd = '';

        for select pcon_called_datetime, cc_status, cc_name, pcon_memo
        from pantera_kontracts
        inner join pantera_contacts on pcon_from_pan_id = pant_id   -- contaacts comments
        inner join contacts on cc_id = pcon_from_cc_id
        where pant_from_k_id = :k_id
        and pcon_has_memo = 1
        order by pcon_called_datetime desc
        into comm_datetime, comm_pers, comm_fio, comm_text
        do
        begin
            kontract_comment_cc_cont = left(kontract_comment_cc_cont || razd || f_datetimetostr(comm_datetime) || ' - ' || trim(coalesce(comm_pers, '')) || ' - ' || comm_fio || ' - ' || trim(coalesce(comm_text, '')), 1024);
            razd = ascii_char(10);
        end

        razd_cl = iif(kontract_comment_cc_cl <> '', ascii_char(10), '');
        razd_cont = iif(kontract_comment_cc_cont <> '', ascii_char(10), '');

        comm_datetime = null;
        comm_pers = null;
        comm_fio = null;
        comm_text = null;

        for select omu_datetime, omu_object_kind, iif(omu_object_kind = 0, c_name_fio, cc_name), cc_status, omu_memo
        from pantera_kontracts
        inner join okt_manager_update on omu_from_pant_id = pant_id   -- manager comments
        left join clients on c_id = omu_object_id and omu_object_kind = 0
        left join contacts on cc_id = omu_object_id and omu_object_kind = 13
        where pant_from_k_id = :k_id
        and omu_object_kind in (0, 13)
        and coalesce(omu_memo, '') <> ''
        order by omu_datetime desc
        into comm_datetime, comm_kind, comm_fio, comm_pers, comm_text
        do
        begin
            if (comm_kind = 0) then
            begin
                kontract_comment_cc_cl = left(kontract_comment_cc_cl || razd_cl || f_datetimetostr(comm_datetime) || ' - клиент - ' || comm_fio || ' - ' || trim(coalesce(comm_text, '')), 1024);
                razd_cl = ascii_char(10);
            end
            else
            begin
                kontract_comment_cc_cont = left(kontract_comment_cc_cont || razd_cont || f_datetimetostr(comm_datetime) || ' - ' || trim(coalesce(comm_pers, '')) || ' - ' || comm_fio || ' - ' || trim(coalesce(comm_text, '')), 1024);
                razd_cont = ascii_char(10);
            end
        end

        suspend;
    end
end
