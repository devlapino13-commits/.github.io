CREATE PROCEDURE PRINT_AKB_GUARANTEE_INFO(
    KBM_ID D_ID)
RETURNS (
    GUAACCOUNTNO D_STR_20,
    GUABANKNAME D_STR_255,
    GUACREDITPURPOSEDESCRIPTION D_STR_255,
    GUACURRENCY D_STR_10,
    GUAMONTHLYPAYMENTAMOUNT D_CURRENCY,
    GUAINITIALAMOUNT D_CURRENCY,
    GUAGRANTEDON D_DATE,
    GUACONTRACTDUEON D_DATE,
    GUAINTERESTAMOUNT D_CURRENCY,
    GUAOUTSTANDINGDEBT D_CURRENCY)
AS
declare variable CHECKED_DATE D_DATETIME;
declare variable JSON_TEXT D_BLOB_BINARY;
declare variable GUARANTEECOUNT D_INTEGER;
declare variable GUARANTEEITEM D_BLOB_TEXT;
declare variable G D_INTEGER;
declare variable GUACREDITPURPOSE varchar(10);
declare variable CREDITPURPOSE varchar(10);
declare variable GUAGRANTEDON_STR varchar(30);
declare variable GUACONTRACTDUEON_STR varchar(30);
declare variable DB_ID D_ID;
begin
  db_id = f_get_db_id();
  json_text = '';
  select bls.bls_blob
  from extinfolist eil
  inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
  inner join kibminiapp kbm on eil.eil_object_id = kbm.kbm_id
  inner join blobstore bls on eil.eil_from_bls_id = bls.bls_id
  where eil.eil_status = 2 and
        kbm.kbm_ver_info = 3 and
        kbm.kbm_id = :kbm_id and
        eic.eic_code = 'akb'
  order by kbm.kbm_checked_datetime desc
  rows 1
  into :json_text;

  if (json_text <> '' or json_text <> '{}' or json_text is not null) then
  begin
    guaranteecount = f_json_blob_get_count(:json_text, 'guarantee[0].guarantee') - 1;

    if (guaranteecount >= 0) then
    begin
      g = 0;
      while (g <= guaranteecount) do
      begin 
        guaranteeitem = f_json_blob_get_value(:json_text, 'guarantee[0].guarantee[' || :g ||']', '');

        guaaccountno = f_json_blob_get_value(:guaranteeitem, 'guaAccountNo', '');
        guabankname = f_json_blob_get_value(:guaranteeitem, 'guaBankName', '');
        guacreditpurpose = f_json_blob_get_value(:guaranteeitem, 'guaCreditPurpose', '');
        guacreditpurposedescription = case
                                          when guacreditpurpose = '001' then iif(db_id <> 94401, 'Потребительские кредиты физическим лицам', 'Fiziki şəxslərə istehlak kreditləri')
                                          when guacreditpurpose = '002' then iif(db_id <> 94401, 'Нефтяной сектор', 'Neft sektoru')
                                          when guacreditpurpose = '003' then iif(db_id <> 94401, 'Ипотечные кредиты физическим лицам', 'Fiziki şəxslərə ipoteka kreditləri')
                                          when guacreditpurpose = '004' then iif(db_id <> 94401, 'Сектор торговли и обслуживания', 'Ticarət və xidmət sektoru')
                                          when guacreditpurpose = '005' then iif(db_id <> 94401, 'Сектор энергетики, химии и природных ресурсов', 'Energetika, kimya və təbii ehtiyatlar sektoru')
                                          when guacreditpurpose = '006' then iif(db_id <> 94401, 'Сектор сельского хозяйства и переработки', 'Kənd təsərrüfatı və emal sektoru')
                                          when guacreditpurpose = '007' then iif(db_id <> 94401, 'Сектор строительства и недвижимости', 'İnşaat və əmlak sektoru')
                                          when guacreditpurpose = '008' then iif(db_id <> 94401, 'Промышленность и производственный сектор', 'Sənaye və istehsal sektoru')
                                          when guacreditpurpose = '009' then iif(db_id <> 94401, 'Транспортно-коммуникационный сектор', 'Nəqliyyat və rabitə sektoru')
                                          when guacreditpurpose = '010' then iif(db_id <> 94401, 'Другие секторы', 'Digər sektorlar')
                                          when guacreditpurpose = '011' then iif(db_id <> 94401, 'Аренда жилого помещения с правом выкупа', 'Yaşayış sahəsinin satmaq öhdəliyi ilə kirayəyə alınması')
                                        end;
        guacurrency = f_json_blob_get_value(:guaranteeitem, 'guaCurrency', '');
        guamonthlypaymentamount = f_json_blob_get_value(:guaranteeitem, 'guaMonthlyPaymentAmount', '');
        guamonthlypaymentamount = cast(guamonthlypaymentamount as d_currency);
        guainitialamount = f_json_blob_get_value(:guaranteeitem, 'guaInitialAmount', '');
        guainitialamount = cast(guainitialamount as d_currency);
        guagrantedon_str = f_json_blob_get_value(:guaranteeitem, 'guaGrantedOn', '');
        guagrantedon_str = substring(guagrantedon_str from 1 for 10);
        guagrantedon = cast(guagrantedon_str as d_date);
        guacontractdueon_str = f_json_blob_get_value(:guaranteeitem, 'guaContractDueOn', '');
        guacontractdueon_str = substring(guacontractdueon_str from 1 for 10);
        guacontractdueon = cast(guacontractdueon_str as d_date);
        guainterestamount = f_json_blob_get_value(:guaranteeitem, 'guaInterestAmount', '');
        guainterestamount = cast(guainterestamount as d_currency);
        guaoutstandingdebt = f_json_blob_get_value(:guaranteeitem, 'guaOutstandingDebt', '');
        guaoutstandingdebt = cast(guaoutstandingdebt as d_currency);
        g = g + 1;
        suspend;
      end
    end
  end
end
