CREATE PROCEDURE CM_RU_0060_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable KF_CLIENT_PASS_NO varchar(30);
declare variable C_ID ID_TYPE;
declare variable IS_PARALEL integer;
declare variable LAST_IS_PARALEL integer;
declare variable K_KONTRACT_NO varchar(20);
declare variable PR_CODE varchar(100);
declare variable RE_PR_CODE varchar(100);
declare variable K_REFUND integer;
declare variable K_ACTIVE_DISB_DATE D_DATE;
declare variable MA_ID D_INTEGER;
declare variable REFUND_K_ID D_ID;
begin
  for select kc_from_c_id,
         k_disb_date,
         iif(pr_code not containing '.PAR.', 0, 1) is_paralel,
         kc_amount_approved, pr_code, k_refund, k_from_ma_app_id
  from kontract, product pr, knclient
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and k_id = kc_from_k_id
  and kc_amount_approved > 0
  and pr.pr_code not containing '.food.'
  and pr.pr_code not containing '.float_percent.'
  and pr.pr_code not containing '.express_online.'
  and pr_code not containing '.support.'
  and pr_code not containing '.summer_plus.'
  and pr_code not containing '.comfortable.'
  and pr_code not containing '.zalog.'
  and pr_code not containing '.selfemployed.'
  and pr_code not containing '.business.'
  and pr_code not containing '.universal.'
  and pr_code not containing '.privilege.'
  and pr_code not containing '.privilege_plus.'
  and pr_code not containing '.oivo.'
  and pr_code not containing '.plus_60.'
  and pr_code not containing '.plus_100.'
  and pr_code not containing '.plus_100_b.'
  into c_id, kf_disb_date, is_paralel, kc_amount, pr_code, k_refund, ma_id
  do
  begin
    for
    select k_kontract_no,
           iif(pr_code not containing '.PAR.', 0, 1) last_is_paralel, pr_code,
           k_disb_date,
           k_id
    from kontract, product, knclient, kontstate
    where k_from_pr_id = pr_id
    and k_id = kc_from_k_id
    and k_id <> :k_id
    and kc_amount_approved > 0
    and kc_from_c_id = :c_id
    and k_id = ks_from_k_id
    and ks_ol_is_delete = 0
    and ks_state = 1
    and :kf_disb_date between ks_date_begin and ks_date_end
    and pr_code not containing '.express_online.'
    and pr_code not containing '.summer_plus.'
    and pr_code not containing '.comfortable.'
    and pr_code not containing '.zalog.'
    and pr_code not containing '.selfemployed.'
    and pr_code not containing '.business.'
    and pr_code not containing '.privilege.'
    and pr_code not containing '.privilege_plus.'
    and pr_code not containing '.universal.'
    and pr_code not containing '.oivo.'
    and pr_code not containing '.plus_60.'
    and pr_code not containing '.plus_100.'
    and pr_code not containing '.plus_100_b.'
    into k_kontract_no, last_is_paralel, re_pr_code, k_active_disb_date, refund_k_id
    do
    begin
      if (k_refund = 0) then
      begin
        if (re_pr_code containing '.float_percent.') then
            continue;

        if (is_paralel + last_is_paralel = 2 ) then
        begin
          if (k_active_disb_date + 3  > kf_disb_date) then
          begin
            select c_name_fio, c_pass_no from clients
            where c_id = :c_id
            into  res_info_text, kf_client_pass_no;
  
  --          res_error_text = 'У клиента с параллельным продуктом уже имеется активный параллельный займ в след. контракте:'||k_kontract_no;
            res_error_text = 'Новый заем "Параллельный" может быть выдан не ранее '||f_datetostr((k_active_disb_date + 3));
            res_error_is_critical = 2;
            suspend;
          end
        end else
        if (is_paralel + last_is_paralel = 0 and
            pr_code = re_pr_code) then  ----RFCRU-4953
        begin
          select c_name_fio, c_pass_no from clients
          where c_id = :c_id
          into  res_info_text, kf_client_pass_no;

          if (pr_code containing '.reliable_client.' and
              re_pr_code containing '.reliable_client.') then    --RFCRU-4953
          begin
            if (k_active_disb_date + 3  > kf_disb_date) then
            begin
              res_error_text = 'Новый заем "Надежный клиент" может быть выдан не ранее '||f_datetostr((k_active_disb_date + 3));
              res_error_is_critical = 2;
              suspend;
            end
          end else
          if (pr_code not containing '.shans.') then
          begin
            res_error_text = 'У клиента имеется активный займ в след. контракте:'||k_kontract_no;
            res_error_is_critical = 1;
            if(pr_code containing '.barakat.' and kc_amount = 15000) then
                    res_error_is_critical = 0;
            --По просьбе Светланы 11.05.20
            if (re_pr_code containing '.food.' and
                  not exists(select * from oksummary oks
                             where :kf_disb_date between oks_date_begin and oks_date_end and
                             oks_delinq_date < :kf_disb_date  and 
                             oks_from_k_id = :k_id and oks_close_state = 0)) then
            res_error_is_critical = 0;
            suspend;
          end
        end
      end else
      begin
        -- k_refund = 1
        if (exists (select *
                    from kontrefinance kr                 -- RFCRU-3486
                    where kr.kr_from_ma_id = :ma_id
                    and kr.kr_from_k_id = :refund_k_id)) then
        begin
          select c_name_fio, c_pass_no from clients
          where c_id = :c_id
          into  res_info_text, kf_client_pass_no;
          res_error_text = 'Внимание, при процедуре Рефинансирования требуется закрыть активный микрозайм по контракту:'||k_kontract_no;
          res_error_is_critical = 0;
          suspend;
        end
      end
    end
  end
end
