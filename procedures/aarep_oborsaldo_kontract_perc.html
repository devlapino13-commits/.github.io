CREATE PROCEDURE AAREP_OBORSALDO_KONTRACT_PERC(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ACC_BALANCE_NO VARCHAR(20),
    ACCOUNT VARCHAR(20),
    REG_NAME VARCHAR(100),
    K_ID ID_TYPE,
    K_KONTRACT_NO VARCHAR(20),
    K_AMOUNT CURR_TYPE,
    BEG_DT CURR_TYPE,
    BEG_KT CURR_TYPE,
    OB_DT CURR_TYPE,
    OB_KT CURR_TYPE,
    END_DT CURR_TYPE,
    END_KT CURR_TYPE,
    KOL_CLI INTEGER)
AS
declare variable id_list integer;
declare variable reg_code varchar(10);
declare variable k_from_pr_id integer;
declare variable acc_no integer;
declare variable k_index_curr_code varchar(3);
declare variable pr_id_prior integer;
declare variable rep_period_begin integer;
declare variable rep_period_end integer;
declare variable ks_date_begin date;
declare variable rep_begin smallint;
declare variable rep_end smallint;
begin
  pr_id_prior = 0;
  beg_dt = 0;
  beg_kt = 0;
  end_dt = 0;
  end_kt = 0;
  rep_begin = 0;

  if (extract(day from :rep_date_begin) = 1) then
  begin
    rep_date_begin = rep_date_begin - 1;

    rep_period_begin = extract(year from :rep_date_begin) * 100 + extract(month from :rep_date_begin);

    if (exists(select *
               from kontcalcperc
               where kcp_period = :rep_period_begin)) then
      rep_begin = 1;

    rep_date_begin = rep_date_begin + 1;
  end
  else
    rep_period_begin = extract(year from :rep_date_begin) * 100 + extract(month from :rep_date_begin);

  rep_period_end = extract(year from :rep_date_end) * 100 + extract(month from :rep_date_end);

  if (exists(select *
             from kontcalcperc
             where kcp_period = :rep_period_end) and
      extract(day from :rep_date_end + 1) = 1) then
    rep_end = 1;
  else
    rep_end = 0;

  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into id_list
  do
  begin
    select reg_code, reg_name
    from region
    where reg_unicode = :id_list
    into reg_code, reg_name;
    reg_name = reg_code || ' ' || reg_name;

    begin
      for select k_id, ks_date_begin, k_kontract_no, k_amount_approved, k_from_pr_id, k_index_curr_code,
                 k_kolich_client_approved
          from kontstate, kontract
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_begin <= :rep_date_end and
                ks_date_end >= :rep_date_begin - 1 and
                ks_from_k_id = k_id
--    and ks_from_k_id = 101010014557100101

                and
                k_last_reg_id = :id_list
          into k_id, ks_date_begin, k_kontract_no, k_amount, k_from_pr_id, k_index_curr_code, kol_cli
      do
      begin
        beg_dt = 0;
        ob_kt = 0;
        ob_dt = 0;
        end_dt = 0;
        if (rep_end = 1) then
        begin
          select kcp.kcp_perc_begin_saldo, kcp.kcp_perc_payed + kcp.kcp_perc_writeoff, kcp.kcp_perc_saldo
          from kontcalcperc kcp
          where kcp_period = :rep_period_end and
                kcp.kcp_from_k_id = :k_id
          into beg_dt, ob_kt, end_dt;
        end
        else
        if (rep_begin = 1) then
        begin
          select kcp.kcp_perc_saldo
          from kontcalcperc kcp
          where kcp_period = :rep_period_begin and
                kcp.kcp_from_k_id = :k_id
          into beg_dt;

          ob_kt = 0;
          select -sum(ok_perc)
          from operkred
          where ok_from_k_id = :k_id and
                ok_ol_is_active = 1 and
                ok_ol_date between :rep_date_begin and :rep_date_end and
                ok_type in (4, 5) -- погашение или списание
          into ob_kt;
          if (ob_kt is null) then
            ob_kt = 0;

          end_dt = 0;
          select sum(sh_calc_perc)
          from aarep_make_zad_list(:k_id, :rep_date_end, 1)
          where sh_date is not null
          into end_dt;
        end
        else
        if (rep_begin = 0 and
            rep_end = 0) then
        begin
--   вычисляем cсальдо начисленных процентов на начало
          beg_dt = 0;
          if (ks_date_begin < rep_date_begin) then
          begin
            select sum(sh_calc_perc)
            from aarep_make_zad_list(:k_id, :rep_date_begin - 1, 1)
            where sh_date is not null
            into beg_dt;

          end

          ob_kt = 0;
          select -sum(ok_perc)
          from operkred
          where ok_from_k_id = :k_id and
                ok_ol_is_active = 1 and
                ok_ol_date between :rep_date_begin and :rep_date_end and
                ok_type in (4, 5) -- погашение или списание
          into ob_kt;
          if (ob_kt is null) then
            ob_kt = 0;

          end_dt = 0;
          select sum(sh_calc_perc)
          from aarep_make_zad_list(:k_id, :rep_date_end, 1)
          where sh_date is not null
          into end_dt;
        end

        end_dt = coalesce(end_dt, 0);
        beg_dt = coalesce(beg_dt, 0);
        ob_dt = ob_kt - (beg_dt - end_dt);

        if (pr_id_prior <> k_from_pr_id) then
        begin
          pr_id_prior = k_from_pr_id;

          select pal_main_acc, pal_main_acc_no
          from product, prodacclist, acctype
          where pr_id = :k_from_pr_id and
                pr_from_pat_id = pal_from_pat_id and
                pal_from_at_id = at_id and
                at_kind = 1 -- выборка только по кредитным счетам
                and
                at_id = 2 and
                pal_main_acc = '76-09'
          into acc_balance_no, acc_no;
        end

        suspend;
      end
    end
  end
end
