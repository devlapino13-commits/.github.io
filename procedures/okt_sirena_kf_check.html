CREATE PROCEDURE OKT_SIRENA_KF_CHECK(
    KF_NAME STR_100,
    KONTRACT_NO STR_100,
    PROJEC_TYPE_CODE STR_100)
RETURNS (
    MESSAGE_TEXT D_STR_8100)
AS
declare variable REP_DATE D_DATE;
declare variable QTY D_INTEGER;
declare variable K_ID ID_TYPE;
declare variable KF_ID ID_TYPE;
declare variable IS_OK D_SMALLINT;
declare variable K_KONTRACT_NO STR_20;
declare variable DELINQ_DAYS D_INTEGER;
declare variable PANT_OPER_CALLED STR_100;
declare variable OP STR_20;
begin
  rep_date = current_date;
  message_text = '';
  qty = 0;

  kf_name = trim(coalesce(kf_name, ''));
  kontract_no = trim(coalesce(kontract_no, ''));
  projec_type_code = trim(coalesce(projec_type_code, ''));

  select rd.rd_name
  from refdata rd
  where rd.rd_code = :projec_type_code
  into :op;

  if (kontract_no <> '') then
  begin
    select kont.k_id
    from kontract kont
    where kont.k_kontract_no = :kontract_no
    into :k_id;
  end
  else
  if (kf_name <> '') then
  begin
    select kal.kal_from_kf_id
    from knaktlist kal
    where kal.kal_new_kf_name = :kf_name and
          :rep_date between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into :kf_id;
  end

  is_ok = 0;
  if (k_id is null) then
    k_id = 0;
  if (kf_id is null) then
    kf_id = 0;
  if (((k_id > 0) or (kf_id > 0)) and
      (projec_type_code <> '')) then
    is_ok = 1;

  if (is_ok = 1) then
  begin
    for select ks.ks_from_k_id, kont.k_kontract_no
        from kontstate ks
        inner join kontract kont on kont.k_id = ks.ks_from_k_id
        where ks.ks_state = 1 and
              :rep_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              ks.ks_from_k_id = :k_id
        union
        select ks.ks_from_k_id, kont.k_kontract_no
        from kontstate ks
        inner join kontract kont on kont.k_id = ks.ks_from_k_id
        where ks.ks_state = 1 and
              :rep_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              kont.k_from_kf_id = :kf_id
        into :k_id, :kontract_no
    do
    begin
      qty = qty + 1;
      k_kontract_no = null;

      select k_kontract_no, kont.k_from_kf_id as kf_id, cast(:rep_date as date) - oks_delinq_date as delinq_days
      from oksummary oks
      inner join kontract kont on kont.k_id = oks.oks_from_k_id
      where :rep_date between oks_date_begin and oks_date_end and
            oks_delinq_date < :rep_date and
            oks_close_state = 0 and
            kont.k_id = :k_id
      into :k_kontract_no, :kf_id, delinq_days;

      if (k_kontract_no is null) then
        k_kontract_no = '';
      if (k_kontract_no <> '') then
      begin
        message_text = message_text || ' Контракт ' || k_kontract_no;

        if (exists(select *
                   from knaktlist kal
                   where kal.kal_from_kf_id = :kf_id and
                         kal.is_delete = 0 and
                         kal.kal_new_kf_officer = 1000000000004 --  КЦ Сирена
                         and
                         :rep_date between kal.kal_date and kal.kal_date_end)) then
        begin
          message_text = message_text || ', числится за "КЦ Сирена", ';
        end

        message_text = message_text || ', число дней просрочки = ' || cast(delinq_days as varchar(10));

        pant_oper_called = null;

        select pantk.pant_oper_called
        from pantera_kontracts pantk
        inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
        where pantk.pant_from_k_id = :k_id and
              :rep_date between pantk.pant_date_begin and pantk.pant_date_end and
              rd.rd_code = :projec_type_code
        into :pant_oper_called;

        if (pant_oper_called is not null) then
        begin
          message_text = message_text || ', числится за ' || pant_oper_called;
        end
        else
        begin
          message_text = message_text || ', за операторами ' || op || ' не числится';
        end
      end
      else
      begin
        message_text = message_text || 'Контракт ' || :kontract_no || ' не имеет текущей просрочки';
      end
    end
    if (message_text <> '') then
    begin
      suspend;
      exit;
    end

    if (qty = 0) then
    begin
      message_text = 'Активный контракт не найден.';
      suspend;
      exit;
    end
  end
  else
  begin
    message_text = 'Данные для посика указаны не корректно.';
    suspend;
    exit;
  end
  suspend;
end
