CREATE PROCEDURE HP_SET_UNDERWRITER_STATUS(
    UN_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    NEW_STATUS INTEGER)
AS
declare variable last_status integer;
begin
  if (coalesce(c_id, 0) = 0) then
  begin
    select un_from_c_id
    from username
    where un_id = :un_id
    into :c_id;
  end
  if (coalesce(c_id, 0) = 0) then
    exit;

  select ovs_add_value
  from hp_objectprops_get('clients', :c_id, -1, 'ObjectProps.KK_Staff_Online', null)
  into :last_status;

  if (last_status is null) then
    last_status = -777;

  if (last_status <> new_status) then
  begin
    execute procedure hp_objectprops_set('Clients', :c_id, -1, 'ObjectProps.KK_Staff_Online', current_date, :new_status,
        :new_status, :un_id);
  end
end
