CREATE PROCEDURE HP_MINIAPP_CHECK_GOODS_ON_STOCK(
    MA_ID D_INTEGER)
RETURNS (
    RES_VALUE D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable k_id d_id;
declare variable kont_amount d_currency;
declare variable kf_disb_date d_date;
declare variable ol_date d_date;
declare variable kml_total_tmp d_currency;
declare variable kml_total d_currency;
declare variable ma_date_app d_date;
declare variable pr_code d_str_100;
declare variable ag_id d_integer;
begin
  select res_value, res_text
  from hp_miniapp_check_goods(:ma_id)
  into :res_value, :res_text;

  if (res_value in (0, -1)) then
  begin
    suspend;
    exit;
  end

  select kont.k_id, k_amount_approved + coalesce(kont.k_prepayment_summa, 0), k_disb_date, pr.pr_code
  from kontract kont
  inner join product pr on pr.pr_id = kont.k_from_pr_id
  where kont.k_from_ma_app_id = :ma_id
  into :k_id, :kont_amount, :kf_disb_date, :pr_code;

  if (pr_code not containing '.murabaha.') then
  begin
    res_value = 1;
    suspend;
    exit;
  end

  select ma.ma_date_app
  from miniapp ma
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        :kf_disb_date between ma.ma_date_app and ma_date_end rows 1
  into ma_date_app;

  kml_total = 0;
  for select ol.ol_date, km.kml_from_ag_id, sum(km.kml_total)
      from kontmat km, matlist ml, opermat om, operlist ol
      where kml_object_kind = 9 and
            kml_object_id = :k_id and
            kml_from_ml_id = ml_id and
            kml_bill_id = om_bill_id and
            kml_bill_line_no = om_bill_line_no and
            om_oper_type = 1 and
            om_from_ol_id = ol_id and
            ol.is_delete = 0
      group by ol.ol_date, km.kml_from_ag_id
      into ol_date, ag_id, kml_total_tmp
  do
  begin
    if (kml_total_tmp is null) then
      kml_total_tmp = 0;
    kml_total = kml_total + kml_total_tmp;

    if (not exists(
      select *
      from agreement_esign_access
      where aea_from_ag_id = :ag_id
        and aea_sign))
    then continue;

    if (ma_date_app > :ol_date) then
    begin
      res_text = 'Дата покупки меньше даты заявки';
      res_value = -1;
      suspend;
      exit;
    end

    if (ol_date > kf_disb_date) then
    begin
      res_text = 'Дата покупки больше даты выдачи!';
      res_value = -1;
      suspend;
      exit;
    end
  end

  if (kml_total = 0) then
  begin
    res_text = 'Товары на складе не найдены';
    res_value = -1;
    suspend;
    exit;
  end
  else
  if (kont_amount <> kml_total) then
  begin
    res_text = 'Cумма товар(а,ов) отличается от суммы финансирования.';
    res_value = -1;
    suspend;
    exit;
  end

  res_value = 1;
  suspend;
end
