CREATE PROCEDURE HP_UPDATE_STOPPERC_137_07_16(
    REP_K_ID ID_TYPE,
    REP_DATE DATE)
AS
declare variable ks_region_id integer;
declare variable k_kontract_no varchar(20);
declare variable ol_memo varchar(255);
declare variable k_stop_perc date;
declare variable sh_id integer;
declare variable sh_date date;
declare variable s_perc decimal(18,4);
declare variable reg_id integer;
declare variable new_ol_id decimal(18,0);
declare variable un_id decimal(18,0);
declare variable bal decimal(18,4) = 0;
declare variable sh_os decimal(18,4) = 0;
declare variable sh_perc decimal(18,4) = 0;
declare variable s_perc_all decimal(18,4) = 0;
declare variable cm_database_id integer;
declare variable termination_date date;
declare variable termination_date_str str_100;
declare variable k_kredline_end date;
declare variable do_calc smallint;
declare variable kd_percent_effect d_percent;
begin
  -- Автоматизация по приказу 137-07-16 о/д
  -- send_sms

  cm_database_id = f_get_db_id();
  if (cm_database_id <> 81001) then
    exit;

  new_ol_id = null;

  bal = 0;
  s_perc_all = 0;

  for select k_stop_perc, ks_region_id, k_kontract_no, kont.k_kredline_end
      from kontract kont, kontstate
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            current_date between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_stop_perc is not null and
            k_id = :rep_k_id
      into k_stop_perc, ks_region_id, k_kontract_no, k_kredline_end
  do
  begin
    -- Согласно приказу, все будущие платежи сворачиваются в один платеж, и этот платеж выставляется на дату расторжения + 30 дней.
    -- Но, если платеж выпадает на выходной день, его нужное перенести на ближайший рабочий день после выходного
    ---------->>
    if (rep_date < cast('01.10.2020' as date)) then
    begin
      termination_date = dateadd(30 day to :rep_date);

      if ((select sc.scd_work_hours
           from sa_calendays sc
           where sc.scd_date = :termination_date) = 0) then
      begin
        select sc.scd_date
        from sa_calendays sc
        where sc.scd_date > :termination_date and
              sc.scd_work_hours > 0
        order by sc.scd_date
        rows 1
        into :termination_date;
      end
    end
    else
      termination_date = rep_date; -- начиная с 01.10.2020 график сворачивается в дату расторжения

    do_calc = 1;

    if (k_kredline_end < termination_date) then
    begin
      do_calc = 0;
      termination_date = k_kredline_end;
    end

    termination_date_str = f_datetostr(termination_date);
    -----------<<

    un_id = null;
    select un_id
    from username
    where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN')
    into :un_id;
    if (un_id is null) then
      un_id = 1000000000001;

    if (do_calc = 1) then
    begin
      for select sh_id, sh_date, zad_os, zad_perc, coalesce(sh_calc_perc, 0)
          from aarep_make_zad_list(:rep_k_id, :rep_date, 1)
          where (zad_perc <> 0 or
                zad_os <> 0 or
                coalesce(sh_calc_perc, 0) <> 0) and
                sh_date > :rep_date
          into sh_id, sh_date, sh_os, sh_perc, s_perc
      do
      begin
        if (:new_ol_id is null) then
        begin
          reg_id = f_hq_id(cm_database_id);

          select kv
          from get_kv(:reg_id, 5)
          into :new_ol_id;

          ol_memo = 'Приостановление процентов договора №' || k_kontract_no || '. Дата приостановления ' || :k_stop_perc;
          insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status,
                                is_delete, ol_user_id)
          values (:new_ol_id, :ks_region_id, 1, :rep_date, 8, :ol_memo, 0, 0, :un_id);
        end
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:new_ol_id, -1, :rep_k_id, :sh_id, :sh_date, 3, 0, 0, 0, 3, 0, -:sh_os, -:sh_perc, 0, 0);
        bal = bal + sh_os;
        s_perc_all = s_perc_all + s_perc;
      end
    end
  end
  if (new_ol_id is not null) then
  begin
    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                          ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
    values (:new_ol_id, -1, :rep_k_id, :sh_id, :termination_date, 3, 0, 0, 0, 3, 0, :bal, :s_perc_all, 0, 0);
  end

  if (new_ol_id is not null) then
  begin
    update kontract
    set k_stopped_perc_ol_id = :new_ol_id
    where k_id = :rep_k_id;
  end

  execute procedure hp_stop_perc_notification(rep_k_id, termination_date);

  -- Формируем Информационный расчет ПСК, для уведомления клиента
  select kps_psk
  from hp_edoc_ins_kontpsk(:rep_k_id, :new_ol_id, :un_id)
  into :kd_percent_effect;
end
