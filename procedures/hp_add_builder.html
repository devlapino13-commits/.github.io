CREATE PROCEDURE HP_ADD_BUILDER(
    K_ID D_ID)
AS
declare variable c_id d_id;
declare variable count_builder d_smallint;
declare variable all_sum_builder d_currency;
declare variable kml_total d_currency;
declare variable kiw_price d_currency;
declare variable k_prepayment_summa d_currency;
begin

  update kontract_istinaa_prepay
  set kip_is_active = 0
  where kip_from_k_id = :k_id and
  kip_is_active = 1;

  count_builder = 0;

  for with tab1
      as (select ag_object_id
          from kontmat
          inner join agreements on (kml_from_ag_id = ag_id)
          where kml_object_kind = 9 and
                kml_object_id = :k_id
          union all
          select kiw_from_c_id
          from kontract_istisnaa_work kiw
          where kiw.kiw_from_k_id = :k_id and
                kiw_is_delete = 0)
      select distinct ag_object_id, tab2.count_builder
      from tab1
      inner join(select count(distinct ag_object_id) as count_builder
                 from tab1) as tab2 on 1 = 1
      into :c_id, count_builder
  do
  begin
    all_sum_builder = 0;
    kml_total = 0;
    kiw_price = 0;

    select coalesce(sum(kml_total), 0)
    from kontmat
    inner join agreements on (kml_from_ag_id = ag_id)
    where kml_object_kind = 9 and
          kml_object_id = :k_id and
          ag_object_id = :c_id
    into :kml_total;

    select coalesce(sum(kiw.kiw_price), 0)
    from kontract_istisnaa_work kiw
    where kiw.kiw_from_k_id = :k_id and
          kiw.kiw_from_c_id = :c_id and
          kiw.kiw_is_delete = 0
    into :kiw_price;

    all_sum_builder = kiw_price + kml_total;
    if (count_builder = 1) then
    begin
      k_prepayment_summa = 0;

      select k_prepayment_summa
      from kontract k
      where k.k_id = :k_id
      into :k_prepayment_summa;
    end

    insert into kontract_istinaa_prepay (kip_from_k_id, kip_from_c_id, kip_pay_sum, kip_prepay_sum)
    values (:k_id, :c_id, :all_sum_builder, :k_prepayment_summa);

  end
end
