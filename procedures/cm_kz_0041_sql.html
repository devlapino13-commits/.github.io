CREATE PROCEDURE CM_KZ_0041_SQL(
    K_ID D_ID,
    CHECK_KIND D_STR_255,
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable MA_ID D_ID;
declare variable RES_STATUS D_SMALLINT;
begin
  -- RFCKZ-996
  res_info_text = 'Проверка статуса СУСН';
  res_error_text = '';
  res_error_is_critical = 0;

  select kn.k_from_ma_app_id
  from kontract kn
  where kn.k_id = :k_id
  into :ma_id;

  for select kc.kc_from_c_id
      from knclient kc
      where kc.kc_from_k_id = :k_id and
            kc.kc_amount_approved > 0
      into :c_id
  do
  begin
    res_status = null;

    select res_status
    from hp_susn_get_status(:c_id, :ma_id)
    into :res_status;

    if (coalesce(res_status, 0) <> 2) then
    begin
      res_error_text = 'Статус СУСН еще не получен. Для утверждения займа без данного статуса необходимо выполнить утверждение через ФС.';
      res_error_is_critical = 1;
      suspend;
      exit;
    end
  end
end
