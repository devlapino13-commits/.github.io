CREATE PROCEDURE HP_OPEREXTRA_EXECUTION_SAVE(
    REP_RD_ID INTEGER,
    REP_OBJECT_KIND INTEGER,
    REP_OBJECT_ID DECIMAL(18,0),
    REP_DATE DATE,
    REP_AMOUNT DECIMAL(18,4),
    REP_DETAIL VARCHAR(255),
    REP_OPER_MEMO VARCHAR(1024),
    UN_ID DECIMAL(18,0))
AS
declare variable DB_ID d_integer;
declare variable LAST_KN_DATE D_DATE;
declare variable rd_id_cancel d_integer;
declare variable rd_id_restoration d_integer;
declare variable rep_region_id d_integer;
declare variable ol_id d_id;
declare variable ox_id d_integer;
declare variable cancel_ol_id d_id;
declare variable cancel_ox_id d_integer;
declare variable restoration_ol_id d_id;
declare variable restoration_ox_id d_integer;
declare variable ol_parent_id d_id;
declare variable ox_from_rd_id d_integer;
declare variable oen_id d_id;
begin
  -- RFCKZ-639 сохраненеие для модуля Отмена/Восстановление исполнительной надписи OperExtraCancelAndRestorationForm
  rd_id_cancel = f_get_rd_id('OperExtraKind.LawClientExpenseCancel');
  rd_id_restoration = f_get_rd_id('OperExtraKind.LawClientExpenseRestoration');
  ox_from_rd_id = f_get_rd_id('OperExtraKind.ExecutiveInscription');
  db_id = f_get_db_id();

  if (f_objkind(:rep_object_kind, 'code') = 'kontract') then
  begin
    select k_last_reg_id
    from kontract
    where k_id = :rep_object_id
    into :rep_region_id;
  end

  select kv
  from get_kv(:rep_region_id, 5)
  into :ol_id;

  select ox.ox_from_ol_id
  from operextra ox
  inner join refdata rd on rd.rd_id = ox.ox_from_rd_id
  where ox_object_kind = :rep_object_kind
    and ox_object_id = :rep_object_id
    and ox_ol_is_active = 1
    and rd.rd_code in ('OperExtraKind.ExecutiveInscription')
    and ox_amount_plus = :rep_amount
  order by ox.ox_from_ol_id desc
  rows 1
  into ol_parent_id;

  insert into operlist (ol_id, ol_parent_id, ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind,
                        ol_memo, ol_status, is_delete, ol_user_id)
  values (:ol_id, :ol_parent_id, :rep_region_id, 2, 29, :rep_date, 9, :rep_oper_memo, 0, 0, :un_id);

  if (rep_rd_id = rd_id_cancel) then
  begin

    insert into operextra (ox_from_ol_id, ox_from_reg_id, ox_ol_date, ox_ol_is_active, ox_object_kind, ox_object_id,
                           ox_from_rd_id, ox_amount_plus, ox_amount_minus, ox_amount_writeoff, ox_detail)
    values (:ol_id, :rep_region_id, :rep_date, 1, :rep_object_kind, :rep_object_id, :ox_from_rd_id, 0.0, :rep_amount, 0.0,
            :rep_detail)
    returning ox_from_ol_id, ox_id
    into cancel_ol_id, cancel_ox_id;

    select ox.ox_from_ol_id, ox.ox_id
    from operextra ox
    where ox.ox_object_kind = :rep_object_kind and
          ox.ox_object_id = :rep_object_id and
          ox.ox_from_rd_id = :rd_id_restoration and
          ox_ol_is_active = 1
    order by ox.ox_from_ol_id desc
    rows 1
    into restoration_ol_id, restoration_ox_id;

    insert into operextra_execution_note (oen_from_ol_id_cancel, oen_from_ox_id_cancel, oen_from_rd_id, oen_object_kind, oen_object_id, oen_from_ol_id_restoration, oen_from_ox_id_restoration)
    values (:cancel_ol_id, :cancel_ox_id, :rep_rd_id, :rep_object_kind, :rep_object_id, :restoration_ol_id, :restoration_ox_id)
    returning oen_id
    into :oen_id;

    if (not exists(select *
                   from restruct_block_cancel rbc
                   where rbc.rbc_from_k_id = :rep_object_id and
                         rbc.rbc_date_begin = current_date and
                         rbc.rbc_date_end = dateadd(day, cast(f_get_const('kz_restruct_block_cancel_days') as integer), current_date))
    ) then
    begin
        -- RFCKZ-705. При отмене испол. надписи блок на реструктуризацию снимается на период 30 календарных дней
        insert into restruct_block_cancel (rbc_from_k_id, rbc_date_begin, rbc_date_end, rbc_from_oen_id)
        values (:rep_object_id, current_date, dateadd(day, cast(f_get_const('kz_restruct_block_cancel_days') as integer), current_date), :oen_id);
    end
  end
  else
  begin
    insert into operextra (ox_from_ol_id, ox_from_reg_id, ox_ol_date, ox_ol_is_active, ox_object_kind, ox_object_id,
                           ox_from_rd_id, ox_amount_plus, ox_amount_minus, ox_amount_writeoff, ox_detail)
    values (:ol_id, :rep_region_id, :rep_date, 1, :rep_object_kind, :rep_object_id, :ox_from_rd_id, :rep_amount, 0.0, 0.0,
            :rep_detail)
    returning ox_from_ol_id, ox_id
    into restoration_ol_id, restoration_ox_id;

    select ox.ox_from_ol_id, ox.ox_id
    from operextra ox
    where ox.ox_object_kind = :rep_object_kind and
          ox.ox_object_id = :rep_object_id and
          ox.ox_from_rd_id = :rd_id_cancel and
          ox_ol_is_active = 1
    order by ox.ox_from_ol_id desc
    rows 1
    into cancel_ol_id, cancel_ox_id;

    insert into operextra_execution_note (oen_from_ol_id_cancel, oen_from_ox_id_cancel, oen_from_rd_id, oen_object_kind, oen_object_id, oen_from_ol_id_restoration, oen_from_ox_id_restoration)
    values (:cancel_ol_id, :cancel_ox_id, :rep_rd_id, :rep_object_kind, :rep_object_id, :restoration_ol_id, :restoration_ox_id);
  end
end
