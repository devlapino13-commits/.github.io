CREATE PROCEDURE HP_CORRECT_GRAPH(
    REP_K_ID D_ID,
    REP_DATE D_DATE,
    REP_UN_ID D_ID)
RETURNS (
    BALANCE_CORRECT D_CURRENCY,
    BALANCE_LAST D_CURRENCY,
    BALANCE_RAZN D_CURRENCY,
    REG_ID D_INTEGER,
    NEW_OL_ID D_ID,
    ACC_OS D_STR_20,
    ACC_PERC D_STR_20,
    RAZ_PERC D_CURRENCY,
    ACCOUNT_STORNO VARCHAR(20))
AS
declare variable AL_ACCOUNT D_STR_20;
declare variable AT_ID D_SMALLINT;
declare variable SH_ID D_SMALLINT;
declare variable SH_DATE D_DATE;
declare variable SH_OS_RAZ D_CURRENCY;
declare variable SH_PERC_RAZ D_CURRENCY;
declare variable SH_FINE_RAZ D_CURRENCY;
declare variable MIN_SH_DATE D_DATE;
declare variable KD_ID D_INTEGER;
declare variable KD_PERCENT D_PERCENT;
declare variable KD_PERCENT_LAST D_PERCENT;
declare variable KC_ID D_ID;
declare variable ORDER_NO D_SMALLINT;
declare variable BAL_STORNO D_CURRENCY;
declare variable ACCOUNT D_STR_20;
declare variable K_KONTRACT_NO D_STR_20;
declare variable LN_ACCOUNT D_STR_20;
declare variable NEW_OL_ID_PARENT D_ID;
declare variable TEXT_MEMO D_STR_255;
declare variable PERC1 D_PERCENT;
declare variable PERC2 D_PERCENT;
declare variable PERC_DAY D_PERCENT;
declare variable PERC_MONTH D_PERCENT;
declare variable OK_TYPE D_SMALLINT;
begin

  select k_kontract_no from kontract
  where k_id = :rep_k_id
  into k_kontract_no;

  ok_type = 3; -- 2 - реструктуризация; 3 - просто корректировка

  if (ok_type = 3) then
    text_memo = 'Корректировка графика погашений займа '||k_kontract_no||' от '|| f_datetostr(:rep_date);
  else
    text_memo = 'Реструктуриация графика погашений займа '||k_kontract_no||' от '|| f_datetostr(:rep_date);

  reg_id = null;
  select oks_region_id from oksummary
  where oks_from_k_id = :rep_k_id
  and :rep_date between oks_date_begin and oks_date_end
  into reg_id;
  if (reg_id is null) then exit;

  if (rep_un_id is null) then rep_un_id = 1010100000493;
  new_ol_id = null;

  -- текущий процент
  kd_percent_last = 0;
  select kd_percent from kontdata
  where kd_ol_is_delete = 0
  and current_date between kd_ol_date and kd_date_end
  and kd_from_k_id = :rep_k_id
  into kd_percent_last;

  -- процент до повышение
  kd_id = null;
  kd_percent = 0;
  select kd_id, kd_percent from kontdata
  where kd_ol_is_delete = 0
  and kd_from_k_id = :rep_k_id
  order by kd_percent, kd_ol_date
  rows 1
  into kd_id, kd_percent;

  if (kd_percent_last > kd_percent and 1<>1) then
  begin
    if (new_ol_id is null) then
    begin

      select kv
      from get_kv(:reg_id, 5)
      into :new_ol_id;
    
      insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, --
                            ol_status, is_delete, ol_user_id, ol_doc_client_name)
      values (:new_ol_id, :reg_id, 1, :rep_date, 5,
              :text_memo, 0, 0, :rep_un_id, :text_memo);
    end

    insert into kontdata (kd_from_k_id, kd_is_primary, kd_from_ol_id, kd_ol_date, kd_ol_is_delete, kd_percent,
                          kd_percent_per_day, kd_percent_effect, kd_percent_effect_fifd, kd_date_end, kd_memo,
                          kd_fine_percent, kd_float_perc_level)
    select kd_from_k_id, 0, :new_ol_id, :rep_date, 0, kd_percent, kd_percent_per_day,
           kd_percent_effect, kd_percent_effect_fifd, '31.12.9999', :text_memo, kd_fine_percent, kd_float_perc_level
    from kontdata
    where kd_id = :kd_id;
  end

  if (rep_k_id = 101010046290000601) then
  begin
    perc1 = 41.37;
    perc2 = 82.73;
  end else
  if (rep_k_id = 101010046290000701) then
  begin
    perc1 = 39.54;
    perc2 = 79.08;
  end

  if (rep_k_id in (101010046290000601, 101010046290000701)) then
  begin
    if (new_ol_id is null) then
    begin

      select kv
      from get_kv(:reg_id, 5)
      into :new_ol_id;
    
      insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, --
                            ol_status, is_delete, ol_user_id, ol_doc_client_name)
      values (:new_ol_id, :reg_id, 1, :rep_date, 5,
              :text_memo, 0, 0, :rep_un_id, :text_memo);
    end

    perc_day = perc1 / 365;
    perc_month = perc_day * 30;
    insert into kontdata (kd_from_k_id, kd_is_primary, kd_from_ol_id, kd_ol_date, kd_ol_is_delete, kd_percent,
                          kd_percent_per_day, kd_date_end, kd_memo,
                          kd_fine_percent, kd_float_perc_level)
    select kd_from_k_id, 0, :new_ol_id, '27.10.2022', 0, :perc_month, :perc_day,
           '31.12.9999', :text_memo, kd_fine_percent, kd_float_perc_level
    from kontdata
    where kd_id = :kd_id;

    perc_day = perc2 / 365;
    perc_month = perc_day * 30;
    insert into kontdata (kd_from_k_id, kd_is_primary, kd_from_ol_id, kd_ol_date, kd_ol_is_delete, kd_percent,
                          kd_percent_per_day, kd_date_end, kd_memo,
                          kd_fine_percent, kd_float_perc_level)
    select kd_from_k_id, 0, :new_ol_id, '11.10.2023', 0, :perc_month, :perc_day,
           '31.12.9999', :text_memo, kd_fine_percent, kd_float_perc_level
    from kontdata
    where kd_id = :kd_id;

    new_ol_id = null;
  end

  if (rep_k_id in (101010110288900101, 101010114353400101, 101010114353400201, 101010097922100301, 101010097922100401,
  101010066656000801, 101010066656000701, 101010040828900901)) then
  begin
    if (new_ol_id is null) then
    begin
      select kv
      from get_kv(:reg_id, 5)
      into :new_ol_id;
    
      insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, --
                            ol_status, is_delete, ol_user_id, ol_doc_client_name)
      values (:new_ol_id, :reg_id, 1, :rep_date, 5,
              :text_memo, 0, 0, :rep_un_id, :text_memo);
    end

    insert into kontdata (kd_from_k_id, kd_is_primary, kd_from_ol_id, kd_ol_date, kd_ol_is_delete, kd_percent,
                          kd_percent_per_day, kd_percent_effect, kd_percent_effect_fifd, kd_date_end, kd_memo,
                          kd_fine_percent, kd_float_perc_level)
    select kd_from_k_id, 0, :new_ol_id, :rep_date, 0, 0, 0,
           kd_percent_effect, kd_percent_effect_fifd, '31.12.9999', :text_memo, kd_fine_percent, kd_float_perc_level
    from kontdata
    where kd_id = :kd_id;
  end




  -- вытаскиваем корректный баланс контракта
  balance_correct = null;
  select sum(tsc_sh_os), min(tsc_sh_id), min(tsc_sh_date) from temp_schedule_correct
  where tsc_k_id = :rep_k_id
  into balance_correct, sh_id, sh_date;
  balance_correct = coalesce(balance_correct,0);
  min_sh_date = sh_date;

  if (balance_correct > 0) then
  begin
    -- некорректный баланс
    balance_last = null;
    select oks_balance, oks_region_id from oksummary
    where oks_from_k_id = :rep_k_id
    and :rep_date between oks_date_begin and oks_date_end
    and oks_close_state = 0
    into balance_last, reg_id;
    balance_last = coalesce(balance_last, 0);

    balance_razn = 0;
    if (balance_correct <> balance_last) then
    begin
      balance_razn = balance_correct - balance_last;

      --Создаем операцию
      if (new_ol_id is null) then
      begin
        select kv
        from get_kv(:reg_id, 5)
        into :new_ol_id;
      
        insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, --
                              ol_status, is_delete, ol_user_id, ol_doc_client_name)
        values (:new_ol_id, :reg_id, 1, :rep_date, 5,
                :text_memo, 0, 0, :rep_un_id, :text_memo);
      end
      -- создание аналитики по корректировке баланса
      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                            ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
      values (:new_ol_id, -1, :rep_k_id, :sh_id, :sh_date, 4, :balance_razn, -:balance_razn, 0, 1, 0, 0, 0, 0, 0, :rep_date, null, 1, 0);

      -- создание проводки
      acc_os = null;
      acc_perc = null;
      for
      select at_id, pal_main_acc || '/' || pal_main_acc_no, k_kontract_no
      from kontract, product, prodacclist, acctype
      where k_id = :rep_k_id and
            k_from_pr_id = pr_id and
            pr_from_pat_id = pal_from_pat_id and
            pal_from_at_id = at_id and
            at_kind = 1 and -- выборка только по кредитным счетам
            pal_main_acc is not null and
            at_id in (1,3) -- портфель, полученные проценты
      into at_id, al_account, k_kontract_no
      do
      begin
        select al_id from acc_list
        where al_from_reg_id = :reg_id
        and al_account = :al_account
        into al_account;
        if (at_id = 1) then acc_os = al_account;
        if (at_id = 3) then acc_perc = al_account;
      end

      account_storno = '';

      for
      with tabl1 as (
        select distinct ok_from_ol_id from operkred
        where ok_ol_is_active = 1
        and ok_from_k_id = :rep_k_id)
      select case
               when ob_account_dt containing '57-04' then 1
               when ob_account_dt containing '50-01' then 2
               else 3
             end order_no,
             ob_account_dt, sum(ob_summa) from tabl1, operbook
      where ok_from_ol_id = ob_from_ol_id
      and ob_account_dt not containing '58-03'
      and ob_account_dt not containing '76-09'
      group by 1,2
      into order_no, account, bal_storno
      do
      begin
        if (bal_storno > -balance_razn) then
        begin
          account_storno = account;
          break;
        end
      end

      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_from_pc_id,
                   ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date)
      values (:new_ol_id, -1, 810, 1, null, :account_storno, :balance_razn, :balance_razn, null, null, null, :acc_perc, null, 'Корректировка:'||:k_kontract_no, null, :rep_date);

      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_from_pc_id,
                   ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date)
      values (:new_ol_id, -1, 810, 1, null, :account_storno, -:balance_razn, -:balance_razn, null, null, null, :acc_os, null, 'Корректировка:'||:k_kontract_no, null, :rep_date);

      if (exists(select * from loronostro
        where ln_account = :account_storno)) then
      begin
        select ln_account from LORONOSTRO
        where ln_region_loro = 1010100
        and ln_region_nostro = :reg_id
        into ln_account;

        select kv
        from get_kv(1010100, 5)
        into :new_ol_id_parent;
      
        insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, --
                              ol_status, is_delete, ol_user_id, ol_doc_client_name)
        values (:new_ol_id_parent, 1010100, 1, :rep_date, 5,
                :text_memo, 0, 0, :rep_un_id, :text_memo);

        update operlist set ol_parent_id = :new_ol_id_parent
        where ol_id = :new_ol_id;

        insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_from_pc_id,
                   ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date)
        values (:new_ol_id_parent, -1, 810, 1, null, '010181051-01000018', :balance_razn, :balance_razn, null, null, null, :ln_account, null, 'Корректировка:'||:k_kontract_no, null, :rep_date);

      end

    end
      -- Исправление графика
      for
      with tabl1 as (
          select sh_id, sh_date, -zad_os os, -zad_perc perc, -sh_calc_fine fine
          from aarep_make_zad_list(:rep_k_id, :rep_date, 0)
          union all
          select tsc_sh_id, tsc_sh_date, tsc_sh_os, tsc_sh_perc, tsh_sh_fine
          from temp_schedule_correct
          where tsc_k_id = :rep_k_id)
      select sh_id, sh_date, sum(os), sum(perc), sum(fine) from tabl1
      group by 1,2
      having sum(os) <> 0 or sum(perc) <> 0 or sum(fine) <> 0
      into sh_id, sh_date, sh_os_raz, sh_perc_raz, sh_fine_raz
      do
      begin
        if (new_ol_id is null) then
        begin
          --Создаем операцию
          select kv
          from get_kv(:reg_id, 5)
          into :new_ol_id;
        
          insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, --
                                ol_status, is_delete, ol_user_id, ol_doc_client_name)
          values (:new_ol_id, :reg_id, 1, :rep_date, 5,
              :text_memo, 0, 0, :rep_un_id, :text_memo);
        end 

        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                            ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
        values (:new_ol_id, -1, :rep_k_id, :sh_id, :sh_date, :ok_type, 0, 0, :sh_fine_raz, 3, 0, :sh_os_raz, :sh_perc_raz, 0, 0, :rep_date, null, 1, 0);
      end

      select min(sh_date) from aarep_make_zad_list(:rep_k_id,:rep_date,1)
      where sh_date >= :rep_date
      into min_sh_date;

      raz_perc = null;
      select sh_id, sh_date, zad_perc - sh_calc_perc from aarep_make_zad_list(:rep_k_id,:min_sh_date,1)
      where sh_date = :min_sh_date
      order by sh_date 
      rows 1
      into sh_id, sh_date, raz_perc;

      raz_perc = coalesce(raz_perc,0);

      -- поправка накопленных процентов
      if (raz_perc <> 0 and 1<>1) then
      begin
        if (new_ol_id is null) then
        begin
          --Создаем операцию
          select kv
          from get_kv(:reg_id, 5)
          into :new_ol_id;
        
          insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, --
                                ol_status, is_delete, ol_user_id, ol_doc_client_name)
          values (:new_ol_id, :reg_id, 1, :rep_date, 5,
              :text_memo||', накопленные проценты', 0, 0, :rep_un_id, :text_memo);
        end 

        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                            ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
        values (:new_ol_id, -1, :rep_k_id, :sh_id, :sh_date, 7, 0, :raz_perc, 0, 3, 0, 0, 0, 0, 0, :rep_date, null, 1, 0);
      end

      if (ok_type = 2) then -- если реструктуризация до добавляем в таблицу operrest
      begin
        insert into OPERREST (OR_FROM_OL_ID, OR_FROM_K_ID, OR_KIND, OR_DATE_END, OR_OL_DATE, OR_OL_IS_ACTIVE, OR_FROM_RD_ID,
                              OR_FROM_REG_ID, OR_DATE_PROT, OR_LLR_DEPENDS, OR_LLR_MANUAL)
        values (:new_ol_id, :rep_k_id, 0, '31.12.9999', :rep_date, 1, 8235,
                :reg_id, :rep_date, 1, 0);
      end 

    kc_id = null;
    select kc_id from knclient
    where kc_from_k_id = :rep_k_id
    into kc_id;

    execute procedure hp_kn_oksummary_recalc(:kc_id,:rep_date);
    execute procedure hp_do_kn_operbook_one_online(:rep_k_id, current_date, null, :rep_date - 1);

    suspend;
  end
end
