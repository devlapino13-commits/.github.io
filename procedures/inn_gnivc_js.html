CREATE PROCEDURE INN_GNIVC_JS(
    EIL_ID D_ID)
RETURNS (
    JSON_DATA D_STR_8100)
AS
begin
  json_data = (
    select
      f_format('{"fio":"%1","num_doc":"%2","dt_doc":"%3","dt_query":"%4","dt_answer":"%5",',
        f_json_str_get_value(json_value, 'fam', '') || ' ' ||
          f_json_str_get_value(json_value, 'nam', '') || ' ' ||
            f_json_str_get_value(json_value, 'otch', ''),
        f_json_str_get_value(json_value, 'docs[0].docno', ''),
        f_json_str_get_value(json_value, 'docs[0].docdt', ''),
        f_datetimetostr(eil_create_datetime),
        f_datetimetostr(eil_status_datetime)) ||
      f_format('"inn":"%1"}',
        f_json_str_get_value(blx_blob, 'inn', ''))
    from extinfolist
      inner join hp_extinfo_to_send_inn_gnivc(eil_from_c_id, eil_id)
        on 1 = 1
      inner join hp_get_blobex_blob(eil_from_bls_id)
        on 1 = 1
    where eil_id = :eil_id);
  suspend;
end
