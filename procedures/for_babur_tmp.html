CREATE PROCEDURE FOR_BABUR_TMP
RETURNS (
    C_ID ID_TYPE,
    AGE_TXT STR_100,
    C_LANGUAGE STR_100,
    C_SEX STR_100)
AS
declare variable c_birth_date date;
declare variable c_age integer;
declare variable rep_date date;
begin
  age_txt = '';

  rep_date = '18.09.2017';

  for select distinct kncl.kc_from_c_id
      from knclient kncl
      inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id
      where ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            kncl.kc_amount_approved > 0
      into :c_id
  do
  begin
    select cl.c_birth_date, rd.rd_name, iif(cl.c_sex = 0, 'Женский', 'Мужской')
    from clients cl
    inner join refdata rd on rd.rd_id = cl.c_nation
    where cl.c_id = :c_id
    into :c_birth_date, :c_language, :c_sex;

    if (c_language not in ('Таджикский', 'Русский', 'Кыргызский', 'Узбекский')) then
      c_language = 'Другое';

    c_age = 0; -- Возаст СВЗ
    if (c_birth_date is not null) then
    begin
      c_age = trunc(datediff(month, :c_birth_date, :rep_date) / 12);
    end

    if (c_age < 21) then
      age_txt = ' до 21 года';

    if (c_age between 21 and 25) then
      age_txt = '21 - 25 лет';
    if (c_age between 26 and 30) then
      age_txt = '26 - 30 лет';

    if (c_age between 31 and 35) then
      age_txt = '31 - 35 лет';
    if (c_age between 36 and 40) then
      age_txt = '36 - 40 лет';

    if (c_age between 41 and 45) then
      age_txt = '41 - 45 лет';
    if (c_age between 46 and 50) then
      age_txt = '46 - 50 лет';

    if (c_age between 51 and 55) then
      age_txt = '51 - 55 лет';
    if (c_age between 56 and 60) then
      age_txt = '56 - 60 лет';

    if (c_age between 61 and 65) then
      age_txt = '61 - 65 лет';
    /*
    if (c_age between 66 and 70) then  age_txt =  '66 - 70 лет';

    if (c_age between 71 and 75) then  age_txt =  '71 - 75 лет';
    if (c_age between 76 and 80) then  age_txt =  '76 - 80 лет';

    if (c_age > 80) then  age_txt =  '81 год и более';
    */
    if (c_age > 65) then
      age_txt = '66 лет и старше';

    suspend;
  end
end
