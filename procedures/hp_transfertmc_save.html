CREATE PROCEDURE HP_TRANSFERTMC_SAVE(
    JSDATA D_BLOB_TEXT)
RETURNS (
    TRT_ID D_INTEGER)
AS
declare variable dt_act d_date;
declare variable nm_act d_integer;
declare variable offices_id d_integer;
declare variable officer_id d_integer;
declare variable mols_id d_id;
declare variable molr_id d_id;
declare variable wpr_id d_id;
declare variable numb d_integer;
begin
  numb = f_json_blob_get_count(jsdata, 'tmc_lst');
  if (numb > 0) then
  begin
    dt_act     = f_json_blob_get_value(jsdata, 'head.dt_act', null);
    nm_act     = f_json_blob_get_value(jsdata, 'head.nm_act', null);
    offices_id = f_json_blob_get_value(jsdata, 'head.offices_id', null);
    officer_id = f_json_blob_get_value(jsdata, 'head.officer_id', null);
    mols_id    = f_json_blob_get_value(jsdata, 'head.mols_id', null);
    molr_id    = f_json_blob_get_value(jsdata, 'head.molr_id', null);
    wpr_id     = f_json_blob_get_value(jsdata, 'head.wpr_id', null);

    insert into transfertmc (
      trt_number,
      trt_date,
      trt_json,
      trt_offices_id,
      trt_officer_id,
      trt_mols_id,
      trt_molr_id,
      trt_wpr_id
    )
    values (
      :nm_act,
      :dt_act,
      :jsdata,
      :offices_id,
      :officer_id,
      :mols_id,
      :molr_id,
      :wpr_id)
    returning
      trt_id
    into
      :trt_id;

    suspend;
  end
end
