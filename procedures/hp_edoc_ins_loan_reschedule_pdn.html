CREATE PROCEDURE HP_EDOC_INS_LOAN_RESCHEDULE_PDN(
    CML_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable kc_pdn_koeff d_percent;
declare variable kc_id d_id;
declare variable eic_code d_str_code;
declare variable history_not_found d_smallint;
declare variable calc_status d_str_100;
declare variable cml_date d_date;
declare variable c_id d_id;
begin
  -- Создание документа "Уведомление о расчете ПДН"

  res_text = '';

  select kc_id, cml_date, kc_from_c_id
  from com_list, knclient
  where cml_id = :cml_id
    and kc_from_k_id = cml_object_id
    and kc_amount_approved > 0
  into kc_id, cml_date, c_id;

  -- Проверяем допустимость рассылки смс/push
  if (f_clientcom_allowed(c_id, 0, 1, 0) = 0) then
    res_text = 'Отправка сообщений данному клиенту запрещена по причине отсутствия согласий';
  else
  begin
    -- Определяем наличие кредитной истории
    eic_code = 'equifax_40';
  
    select history_not_found, calc_status
    from hp_extinfolist_pdn_get_eil(:kc_id,  :cml_date, 0, :eic_code, 1)
    into history_not_found, calc_status;
  
    -- Если КИ не найдена, то поищем по НБКИ
    if (history_not_found = 1 or calc_status = 'Полученная история БКИ устарела') then
    begin
      eic_code = 'nbki_40';
      select history_not_found, calc_status
      from hp_extinfolist_pdn_get_eil(:kc_id,  :cml_date, 0, :eic_code, 1)
      into history_not_found, calc_status;
    end
  
    if (history_not_found = 1 or calc_status = 'Полученная история БКИ устарела') then
      res_text = 'ПДН не расчитан, убедитесь что запрос в БКИ отправлен. ' || coalesce(calc_status, '');
  end

  if (res_text = '') then
  begin
    -- Рассчитываем ПДН
    execute procedure hp_knclient_pdn_update(kc_id, 0, cml_id, null, null, null, null, 'edoc_pdn_create');
  
    select kc_pdn_koef
    from com_list, knclient_pdn
    where cml_id = :cml_id
      and cml_from_kcp_id = kcp_id
    into kc_pdn_koeff;
  
    if (coalesce(kc_pdn_koeff, 0) = 0) then
      res_text = 'ПДН не расчитан, отбратитесь в техподдержку.';
    else
    begin
      select edl_id, res_text
      from hp_edoc_insert('loan.reschedule.pdn', f_objkind('com_list', 'id'), :cml_id, :un_id)
      into edl_id, res_text;
    end
  end

  suspend;
end
