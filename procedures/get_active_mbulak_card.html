CREATE PROCEDURE GET_ACTIVE_MBULAK_CARD(
    C_ID D_ID)
RETURNS (
    CAL_CARD_ID D_INTEGER,
    CARD_PAN_HIDEN_LONG D_STR_20,
    CARD_PAN_HIDEN_SHORT D_STR_10)
AS
declare variable CARD_TRANSFER_TO_CLIENT_STATUS D_INTEGER;
begin
  -- процедура возвращает активню карту клиента

  card_transfer_to_client_status = f_enum_id('cardlist_status', 'transfer_to_client');

  select cal.cal_card_id,
         ('**** **** **** ' || right(cal_pan_short, 4)),
         ('** ' || right(cal_pan_short, 4))
  from cardlist cal
  inner join cardclientacc on cca_id = cal_from_cca_id
  inner join cardclientid on cci_id = cca_from_cci_id and
        cci_from_c_id = :c_id
  where cal_status = :card_transfer_to_client_status
  into
    :cal_card_id,
    :card_pan_hiden_long,
    :card_pan_hiden_short;

  card_pan_hiden_long = coalesce(card_pan_hiden_long, '');
  card_pan_hiden_short = coalesce(card_pan_hiden_short, '');

  suspend;
end
