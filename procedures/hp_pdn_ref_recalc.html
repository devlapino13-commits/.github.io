CREATE PROCEDURE HP_PDN_REF_RECALC(
    REP_DATE_ONE D_DATE,
    REP_DATE_TWO D_DATE,
    ACTUAL_TIME D_DATETIME)
RETURNS (
    PRR_K_ID D_ID,
    PRR_K_KONTRACT_NO D_STR_20,
    PRR_K_FACT_DISB_DATE D_DATE,
    PRR_DISB_DAYS D_INTEGER,
    PRR_K_FACT_CLOSED_DATE D_DATE,
    PRR_REF_CNT_LIST D_STR_20,
    PRR_REF_OUR D_STR_LONG,
    PRR_REF_OUR_ACTIVE D_STR_LONG,
    PRR_REF_OUR_CNT D_SMALLINT,
    PRR_REF_OUR_CNT_ACTIVE D_SMALLINT,
    PRR_REF_EXT D_STR_LONG,
    PRR_REF_EXT_ACTIVE D_STR_LONG,
    PRR_REF_EXT_CNT D_SMALLINT,
    PRR_REF_EXT_CNT_ACTIVE D_SMALLINT,
    PRR_PDN_STATUS D_STR_CODE,
    PRR_PDN_STATUS_ID D_SMALLINT,
    PRR_PDN_DATE D_DATE,
    PRR_PDN_DAYS D_INTEGER,
    PRR_SSP_DATE D_DATE,
    PRR_SSP_NOTE D_STR_255,
    PRR_SSP_NOTE_ID D_SMALLINT,
    PRR_KI_DATE D_DATE,
    PRR_KI_NOTE D_STR_255,
    PRR_KI_NOTE_ID D_SMALLINT,
    PRR_IS_SELECTED D_BOOL,
    PRR_DATETIME D_DATETIME)
AS
begin
  -- ПДН - расчет в связи с неисполнением цели по рефинансированию (RFCRU-4309)

  for select prr_k_id,
             prr_k_kontract_no,
             prr_k_fact_disb_date,
             prr_disb_days,
             prr_k_fact_closed_date,
             prr_ref_cnt_list,
             prr_ref_our,
             prr_ref_our_active,
             prr_ref_our_cnt,
             prr_ref_our_cnt_active,
             prr_ref_ext,
             prr_ref_ext_active,
             prr_ref_ext_cnt,
             prr_ref_ext_cnt_active,
             prr_pdn_status,
             prr_pdn_status_id,
             prr_pdn_date,
             prr_pdn_days,
             prr_ssp_date,
             prr_ssp_note,
             prr_ssp_note_id,
             prr_ki_date,
             prr_ki_note,
             prr_ki_note_id,
             prr_is_selected,
             prr_datetime
      from pdn_ref_recalc
      into :prr_k_id,
           :prr_k_kontract_no,
           :prr_k_fact_disb_date,
           :prr_disb_days,
           :prr_k_fact_closed_date,
           :prr_ref_cnt_list,
           :prr_ref_our,
           :prr_ref_our_active,
           :prr_ref_our_cnt,
           :prr_ref_our_cnt_active,
           :prr_ref_ext,
           :prr_ref_ext_active,
           :prr_ref_ext_cnt,
           :prr_ref_ext_cnt_active,
           :prr_pdn_status,
           :prr_pdn_status_id,
           :prr_pdn_date,
           :prr_pdn_days,
           :prr_ssp_date,
           :prr_ssp_note,
           :prr_ssp_note_id,
           :prr_ki_date,
           :prr_ki_note,
           :prr_ki_note_id,
           :prr_is_selected,
           :prr_datetime
  do
    suspend;
end
