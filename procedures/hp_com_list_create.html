CREATE PROCEDURE HP_COM_LIST_CREATE(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    CMC_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    CML_ID D_ID,
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_INTEGER)
AS
begin
  select cml_id, res_text, res_is_error
  from fm_com_list_create(:obj_kind, :obj_id, :cmc_id, :un_id)
  into :cml_id, :res_text, :res_is_error;
  suspend;
end
