CREATE PROCEDURE HP_GET_LOAN_PRODUCTS(
    LOAN_PR_ID INTEGER,
    LOANDISBDATE DATE,
    CLIENT_COUNT INTEGER,
    ALL_CL_AMOUNT CURR_TYPE,
    MIN_CL_AMOUNT CURR_TYPE,
    MAX_CL_AMOUNT CURR_TYPE,
    C_ID_LIST VARCHAR(255),
    ADD_MONTH SMALLINT = 0,
    REFUND SMALLINT = 0)
RETURNS (
    PR_ID INTEGER,
    PR_NAME VARCHAR(255),
    PR_MEMO VARCHAR(255),
    PERC_MIN PERC_TYPE,
    PERC_MAX PERC_TYPE,
    PERC_DEFAULT PERC_TYPE,
    KOMMIS_PERC_MIN PERC_TYPE,
    KOMMIS_PERC_MAX PERC_TYPE,
    KOMMIS_PERC_DEFAULT PERC_TYPE,
    PERIOD_MIN INTEGER,
    PERIOD_MAX INTEGER,
    PERIOD_DEFAULT INTEGER,
    ADD_INFO VARCHAR(255))
AS
declare variable is_correct integer;
declare variable db_id d_integer;
begin
  if (loan_pr_id is null) then
    loan_pr_id = 0;

  db_id = f_get_db_id();

  if (loan_pr_id = 0) then
  begin
    for select pr_id, pr_name, pr_memo
        from product
        where pr_date_begin <= coalesce(:loandisbdate, current_date) and
              coalesce(pr_date_end, '31.12.2999') >= coalesce(:loandisbdate, current_date) and
              pr_active = 1 and
              iif (:db_id in (39801, 41701), (pr_code not containing '.express_online.' and
                                    pr_code not containing '.installment_goods.' and
                                    pr_code not containing '.sugarmill.' and
                                    pr_code not containing '.online_loan.'), true)
        into :pr_id, :pr_name, :pr_memo
    do
    begin
      for select perc_min, perc_max, perc_default, kommis_perc_min, kommis_perc_max, kommis_perc_default, period_min,
                 period_max, period_default, add_info
          from hp_get_percent(:pr_id, :loandisbdate, :client_count, :all_cl_amount, -1, :min_cl_amount, :max_cl_amount, :c_id_list, :add_month, :refund)
          where is_correct = 1
          into :perc_min, :perc_max, :perc_default, :kommis_perc_min, :kommis_perc_max, :kommis_perc_default,
               :period_min, :period_max, :period_default, add_info
      do
      begin
        suspend;
      end
    end
  end
  else
  begin
    for select pr_id, pr_name, pr_memo
        from product
        where pr_id = :loan_pr_id
        into :pr_id, :pr_name, :pr_memo
    do
    begin
      for select perc_min, perc_max, perc_default, kommis_perc_min, kommis_perc_max, kommis_perc_default, period_min,
                 period_max, period_default, is_correct, add_info
          from hp_get_percent(:pr_id, :loandisbdate, :client_count, :all_cl_amount, -1, :min_cl_amount, :max_cl_amount, :c_id_list, :add_month, :refund)
          into :perc_min, :perc_max, :perc_default, :kommis_perc_min, :kommis_perc_max, :kommis_perc_default,
               :period_min, :period_max, :period_default, is_correct, add_info
      do
      begin
        if (is_correct = 0) then
        begin
          period_default = -1;
        end
        suspend;
      end
    end
  end
end
