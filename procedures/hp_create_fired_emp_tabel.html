CREATE PROCEDURE HP_CREATE_FIRED_EMP_TABEL(
    SAT_FROM_C_ID D_ID,
    SAT_YEAR_MONTH D_INTEGER,
    SAT_OFFICE_ID D_INTEGER,
    SAT_FROM_SAWT_ID D_INTEGER)
AS
declare variable date_begin d_date;
declare variable date_end d_date;
declare variable date_gone_number d_smallint;
declare variable sql_request_select d_str_long;
declare variable sql_request_update d_str_long;
declare variable sql_request_update_tmp d_str_100;
declare variable tmp_var d_smallint;
declare variable tmp_hours d_smallint;
declare variable tmp_qty d_integer;
declare variable i d_smallint;
begin
  if (exists(select *
             from region rg
             inner join refdata rd on rd.rd_id = rg.reg_office_hours_id
             where rg.reg_unicode = :sat_office_id and
                   rd.rd_code in ('office_hours_5', 'office_hours_5mob'))) then
    exit;

  date_begin = '01.' || right(:sat_year_month, 2) || '.' || left(:sat_year_month, 4);
  date_end = dateadd(month, 1, date_begin) - 1;

  date_gone_number = null;
  select extract(day from ps.ps_date_begin)
  from hr_pstatus ps
  inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id and
        ho.ho_order_type = 6 --увольнение сотрудника
  where ps.ps_from_c_id = :sat_from_c_id and
        ps.ps_from_reg_id = :sat_office_id and
        ps.ps_date_begin between :date_begin and :date_end and
        ps.ps_status = 1 and
        ps.ps_is_working = 0
  into date_gone_number;

  --Заполняю дни согласно составленному табелю, но только до дня увольнения
  sql_request_update_tmp = 'update sa_tabel sat set sat_day_';

  i = 1;
  tmp_var = 0;
  tmp_hours = 0;
  tmp_qty = 0;
  while (i < date_gone_number)
  do
  begin
    sql_request_select = 'select sat_day_' || :i || ' ' ||
                         'from sa_tabelo sat
                          where sat.sat_year_month = ' || :sat_year_month || ' and ' ||
                             'sat.sat_from_c_id = ' || :sat_from_c_id || ' and ' ||
                             'sat.sat_from_sawt_id = ' || :sat_from_sawt_id || ' and ' ||
                             'sat.sat_office_id = ' || :sat_office_id;
    execute statement sql_request_select
    into tmp_var;

    tmp_hours = tmp_hours + tmp_var;   --Количество часов
    tmp_qty = tmp_qty + iif(tmp_var > 0, 1, 0); --Количество дней (оператор iif добавлен на случай когда есть 7часовые рабочие дни в календаре)

    sql_request_update = sql_request_update_tmp || :i || ' = ' || :tmp_var || ' where sat.sat_year_month = ' || :sat_year_month || ' and ' ||
                                                                           'sat.sat_from_c_id = ' || :sat_from_c_id || ' and ' ||
                                                                           'sat.sat_from_sawt_id = ' || :sat_from_sawt_id || ' and ' ||
                                                                           'sat.sat_office_id = ' || :sat_office_id;
    execute statement sql_request_update;

    i = i + 1;
  end 

  --Дозаполняю оставшиеся дни
  i = date_gone_number;
  while (i <= 31)
  do
  begin
    sql_request_update = sql_request_update_tmp || :i || ' = 0' || ' where sat.sat_year_month = ' || :sat_year_month || ' and ' ||
                                                                 'sat.sat_from_c_id = ' || :sat_from_c_id || ' and ' ||
                                                                 'sat.sat_from_sawt_id = ' || :sat_from_sawt_id || ' and ' ||
                                                                 'sat.sat_office_id = ' || :sat_office_id;
    execute statement sql_request_update;
    i = i + 1;
  end

  update sa_tabel sat
  set sat_total_hours = :tmp_hours,
      sat_total_calc = :tmp_hours,
      sat_days_total = :tmp_qty,
      sat_days_calc = :tmp_qty
  where sat_year_month = :sat_year_month and
        sat_from_c_id = :sat_from_c_id and
        sat_office_id = :sat_office_id and
        sat_from_sawt_id = :sat_from_sawt_id;
end
