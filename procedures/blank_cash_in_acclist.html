CREATE PROCEDURE BLANK_CASH_IN_ACCLIST(
    OL DECIMAL(18,0))
RETURNS (
    BL_SUMMA DECIMAL(18,4),
    BL_CORR_ACCOUNT VARCHAR(20),
    BL_BALANCE_ID VARCHAR(20),
    BL_REGION_CODE VARCHAR(10),
    BL_ACCOUNT_NO INTEGER)
AS
declare variable n integer;
declare variable cm_database_id integer;
declare variable ol_date date;
begin
  /* procedure text */
  n = 0;

  :cm_database_id = f_get_db_id();

  for select ob_account_kt, sum(ob_summa_val), a_kt.al_from_bg_id, reg_code, a_kt.al_number, ol_date
      from operbook, acc_list a_dt, acc_list a_kt, acctypelist, region, operlist
      where ob_from_ol_id = :ol and
            ol_id = :ol and
            a_dt.al_id = ob_account_dt and
            a_kt.al_id = ob_account_kt and
            a_dt.al_from_bg_id like atl_acc_mask || '%' and
            atl_from_at_id in (11, 14, 16) and
            ol_from_reg_id = reg_unicode
      group by ob_account_kt, a_kt.al_from_bg_id, reg_code, a_kt.al_number, ol_date
      into :bl_corr_account, :bl_summa, :bl_balance_id, :bl_region_code, :bl_account_no, :ol_date
  do
  begin
    if (bl_summa is null) then
      bl_summa = 0;
    n = n + 1;

    if ((cm_database_id = 81001) and
        (ol_date >= '01.01.2018')) then
    begin
      if (bl_balance_id = '76-05' and
          right(bl_corr_account, 1) = '4') then
        bl_balance_id = '60323';

      bl_balance_id = (case bl_balance_id
                         when '57-01' then '20209'
                         when '91-01' then '71801'
                         when '73-02' then '60308'
                         when '76-02' then '60323'
                         when '51-01' then '20501'
                         else bl_balance_id
                       end);

      if (bl_balance_id starting '70-') then
        bl_balance_id = '60306';
    end

    suspend;
  end
  bl_balance_id = '';
  bl_region_code = '';
  bl_account_no = null;
  bl_corr_account = null;
  bl_summa = null;
  if (n = 0) then
  begin
    suspend;
    suspend;
    suspend;
  end
  if (n = 1) then
  begin
    suspend;
    suspend;
  end
  if (n = 2) then
  begin
    suspend;
  end
end
