CREATE PROCEDURE HP_ZALOGAUTO_WEB_UPDATE(
    ZAL_ID D_INTEGER,
    ZAL_STATUS D_SMALLINT,
    ZAL_ADD_PARAMS D_STR_8100,
    ZAL_BLOB D_BLOB_BINARY)
AS
declare variable zal_cost d_currency;
declare variable bls_id d_id;
begin
  -- Ключи, приходящие в zal_add_params
  -- Марка ТС: brand
  -- Модель ТС: model
  -- Год выпуска: issue_year
  -- Новая машина: is_new
  -- Комплектация: equipment
  -- Цена: cost

  if (zal_status = 2) then
  begin
    zal_cost = f_json_str_get_value(zal_add_params, 'cost', 0);
    update zalogautolist
    set zal_cost = :zal_cost,
        zal_status = :zal_status,
        zal_add_params = :zal_add_params
    where zal_id = :zal_id;

    execute procedure hp_insert_into_blobstore ('car_screenshot.jpg', 'Скриншот авто', :zal_blob,
      1000000000000, 2, 3)
    returning_values :bls_id;

    insert into zalogautoblob (zab_from_zal_id, zab_from_bls_id)
    values (:zal_id, :bls_id);
  end
  else if (zal_status = -2) then
  begin
    update zalogautolist
    set
      zal_status = :zal_status,
      zal_add_params = :zal_add_params
    where zal_id = :zal_id;
  end
  else
    exception e_default f_format('Invalid status %1. Allowed statuses: -2 or 2.', zal_status);
end
