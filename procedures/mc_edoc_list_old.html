CREATE PROCEDURE MC_EDOC_LIST_OLD(
    MCL_ID D_ID)
RETURNS (
    EDS_ID D_ID,
    EDC_NAME D_STR_255,
    EDL_BLANK_DATE D_DATE,
    DOC_NUMBER D_STR_CODE,
    EDS_STATUS D_INTEGER,
    EDS_STATUS_STR D_STR_CODE)
AS
declare variable C_ID D_ID;
declare variable LANG D_STR_LANG;
declare variable OBJ_KIND D_STR_CODE;
declare variable OBJ_ID D_ID;
declare variable EDC_ID D_INTEGER;
declare variable DB_ID D_INTEGER;
declare variable EDC_CODE D_STR_100;
declare variable MCL_LOGIN D_STR_CODE;
declare variable EDL_ID bigint;
begin
  db_id = f_get_db_id();
  lang = f_get_lang_code(null, mcl_id);
  select mcl_from_c_id, mcl_login from mc_login
  where mcl_id = :mcl_id
  into c_id, mcl_login;

  if (c_id is null) then
    exit;

  -- RFCRU-196
  -- Если логин КМП отличается от подтвержденного № телефона клиента,
  -- тогда эл. документы не показываем. А то проходят с новым номером оцифровку
  -- и могут просматривать и подписывать, что не есть хорошо
  if (db_id = 81001 and mcl_login <> f_cont_info('clients', c_id, 'cell_main', 'ok')) then
    exit;

  for select eds_id, edl_blank_date, eds_status, edc_name,
    edl_object_kind, edl_object_id, edc_id, edc_code, null edl_id
  from edoc_list, edoc_sign, edoc_cfg
  where eds_from_c_id = :c_id and
        eds_from_edl_id = edl_id and
        edl_from_edc_id = edc_id and
        edl_status >= 0 and edl_blank_status = 1
  union
  select null as eds_id, edl_blank_date, 1 as eds_status, edc_name,
    edl_object_kind, edl_object_id, edc_id, edc_code, edl_id
  from edoc_list, edoc_cfg, knclient
  where --eds_from_c_id = :c_id and
        edl_from_edc_id = edc_id and
        edl_status >= 0 and
        edl_blank_status = 1 and
        edc_code = 'loan.debt' and
        edl_object_kind = 9 and
        edl_object_id = kc_from_k_id and
        kc_amount_approved > 0 and
        kc_from_c_id = :c_id
  order by 2 desc
  into eds_id, edl_blank_date, eds_status, edc_name, obj_kind, obj_id, edc_id,
       :edc_code, edl_id
  do
  begin
    if ((db_id in (39801, 41701)) and  -- igusev для КГ скрываетс эл. документ заявка на заем, если он подписан. 02.06.2021 было согласовано на собрании с Бахтияром, Азаматом М., Бактыгуль
        (edc_code = 'loan.application') and
        (eds_status = 1)) then
        continue;

    if (edc_code = 'loan.debt') then eds_id = -edl_id;

    if (exists(select * from edoc_frx
    where edf_from_edc_id = :edc_id)) then -- Только документы с бланками
    begin
      eds_status_str =
        case eds_status
          when 0 then f_lang(lang, 'Ожидает подписания')
          else f_lang(lang, 'Подписан')
        end;
      eds_status_str = trim (eds_status_str);
  
      obj_kind = f_objkind(obj_kind, 'name');
      doc_number = '';
      if (obj_kind = 'kontract') then
      begin
        select k_kontract_no
        from kontract
        where k_id = :obj_id
        into doc_number;
      end else
      if (obj_kind = 'miniapp') then
      begin
        select ma_no
        from miniapp
        where ma_id = :obj_id
        order by ma_number desc
        rows 1
        into doc_number;
      end
      suspend;
    end
  end
end
