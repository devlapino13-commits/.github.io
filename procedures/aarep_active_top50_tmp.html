CREATE PROCEDURE AAREP_ACTIVE_TOP50_TMP(
    REP_DATE DATE)
RETURNS (
    AC_K_ID DECIMAL(18,0),
    AC_DISB DECIMAL(18,4),
    AC_C_ID DECIMAL(18,0),
    AC_C_BALANCE DECIMAL(18,4),
    CL_DISB_AMOUNT DECIMAL(18,4),
    DISB_DD DATE)
AS
declare variable curr_code varchar(3);
declare variable curr_rate decimal(18,4);
declare variable k_curr_rate decimal(18,4);
declare variable cm_def_curr varchar(3);
declare variable pr_set_rate_in_kont integer;
declare variable avg_amount decimal(18,4);
declare variable ac_k_balance d_currency;
begin
  select cm_default_currency
  from company
  into cm_def_curr;

  avg_amount = 100000;
  if (cm_def_curr = '944') then
  avg_amount = 1000;
  -- Один раз вычисляем курс доллара на дату
  for select k_currency, k_amount_approved, ks_from_k_id, k_index_curr_rate, pr_set_rate_in_kont, ks_date_begin
      from kontstate, kontract, product
      where ks_date_begin <= :rep_date and
            ks_date_end >= :rep_date and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            k_id = ks_from_k_id and
            pr_id = k_from_pr_id
            and
            cast((k_amount_approved * k_index_curr_rate) as double precision) / k_kolich_client_approved >= :avg_amount
      into :curr_code, :ac_disb, :ac_k_id, :k_curr_rate, :pr_set_rate_in_kont, :disb_dd
  do
  begin
    ac_k_balance = 0;
    select sum(ok_os)
    from operkred
    where ok_from_k_id = :ac_k_id and
          ok_ol_is_active = 1 and
          ok_ol_date between :disb_dd and :rep_date and
          ok_type <= 8 and
          ok_type >= 2 and
          ok_from_c_id = 0
    into :ac_k_balance;

    if (ac_k_balance > 0) then
    begin
      if (cm_def_curr <> curr_code) then
      begin
        select cr_rate
        from opercurr
        where cr_from_cu_id = :curr_code and
              cr_date <= :rep_date and
              cr_date_end >= :rep_date
        into :curr_rate;
        if (k_curr_rate > curr_rate) then
        begin
          if (pr_set_rate_in_kont = 1) then
            curr_rate = :k_curr_rate;
        end
        if (curr_rate is null) then
          curr_rate = 1;
        ac_k_balance = cast(:ac_k_balance * :curr_rate as decimal(15,2));
        ac_disb = cast(:ac_disb * :curr_rate as decimal(15,2));
      end
      else
        curr_rate = 1;
    end

    ac_c_balance = 0;
    for select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :ac_k_id and
              kc_amount_approved <> 0 and
              kc_amount_approved is not null
        into :ac_c_id, :cl_disb_amount
    do
    begin
      cl_disb_amount = cast(cl_disb_amount * curr_rate as decimal(18,4));
      ac_c_balance = round(cast (ac_k_balance as double precision) * cast( (cl_disb_amount / ac_disb) as double precision), 4);
      suspend;
    end
  end
end
