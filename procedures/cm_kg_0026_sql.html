CREATE PROCEDURE CM_KG_0026_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kf_disb_date d_date;
declare variable kf_id id_type;
declare variable reg_id integer;
declare variable k_kolich_repay integer;
declare variable k_kolich_client_approved integer;
declare variable k_percent_per_day perc_type;
declare variable k_percent_per_day_old perc_type;
declare variable pr_product_kind integer;
declare variable k_circle integer;
declare variable k_kommis_perc perc_type;
declare variable k_kommis_perc_old perc_type;
declare variable kommis_snijen perc_type;
declare variable k_amount_approved curr_type;
declare variable pr_name varchar(50);
declare variable pr_code varchar(100);
declare variable pr_code_old varchar(100);
declare variable kolich_poruch integer;
declare variable c_id id_type;
declare variable k_id_temp id_type;
declare variable k_kontract_no_temp varchar(50);
declare variable oks_delinq_day integer;
declare variable k_period integer;
declare variable amount_approved curr_type;
begin
  for
  select k_disb_date, k_from_kf_id, k_last_reg_id, k_kolich_repay, k_kolich_client_approved, K_Percent_Per_Day,
           pr_product_kind, k_circle, k_kommis_perc, k_amount_approved, pr_name, pr_code, round(kont.k_period)
  from kontract kont, product
  where k_id=:k_id
  and k_from_pr_id = pr_id
  into :kf_disb_date, :kf_id, :reg_id, :k_kolich_repay, :k_kolich_client_approved, K_Percent_Per_Day,
                 :pr_product_kind, k_circle, k_kommis_perc, k_amount_approved, pr_name, pr_code, :k_period
  do
  begin
        if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
          exit;
        select round(K_Percent_Per_Day * 365,4), k_kommis_perc, k_id, pr_code
        from kontract, kontstate, product
        where k_from_kf_id = :kf_id
        and k_id = ks_from_k_id
        and ks_state = 1
        and ks_date_begin < :kf_disb_date  
        and k_from_pr_id = pr_id     
        and pr_product_kind = 1
--        and pr_code containing '.GR.'
        and pr_code not containing '.PAR.'
--        and pr_code not containing '.VYGOD.'
        order by ks_date_begin desc
        rows 1
        into K_Percent_Per_Day_old, k_kommis_perc_old, k_id_temp, pr_code_old;
--          По групповым кредитам процент не должен превышать 42.5
    if (pr_product_kind = 1 and pr_code not containing 'VYGOD') then
    begin
      if (round(K_Percent_Per_Day * 365,4) > 42.5) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'У данного контракта по методу аннуитета, годовая процентная ставка превысила 42.5%.';
        suspend;
      end
      if (k_circle <= 1 and round(K_Percent_Per_Day * 365,4) < 35) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Годовая процентная ставка не должно быть не ниже 35%.';
        suspend;
      end
      
      if (k_circle > 1 and round(K_Percent_Per_Day * 365,4) < 33) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Годовая процентная ставка не должно быть не ниже 33%.';
        suspend;
      end
        select max(ok_ol_date - ok_sh_date)
        from operkred
        where ok_ol_is_active = 1
        and ok_from_k_id = :k_id_temp
        and ok_type in (4,5,9)
        and ok_os + ok_perc < 0
        into oks_delinq_day;
        
        kommis_snijen = case
                            when oks_delinq_day <= 0 then -0.3
                            when oks_delinq_day between 1 and 7 then 0.3
                            when oks_delinq_day between 8 and 15 then 0.6
                            when oks_delinq_day between 16 and 30 then 0.9
                            else 100
                          end;
        if (kommis_snijen between 3 and 3.9) then  k_kommis_perc_old = k_kommis_perc_old + kommis_snijen; 
        if (kommis_snijen > 3.9) then  k_kommis_perc_old = 3.9;
        if (kommis_snijen < 3) then  k_kommis_perc_old = 3;
        if (k_kommis_perc < k_kommis_perc_old) then
          begin
            res_error_is_critical = 1;
            res_error_text='Доп. процент должен быть: '||k_kommis_perc_old;
            suspend;
          end else
      if (pr_code not containing 'IND' and k_circle = 1 and :k_kommis_perc <> 3.9) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Дополнительная процентная ставка должен быть 3.9%!.';
          suspend;
        end else
      if (pr_code not containing 'IND' and k_kommis_perc < 3) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Дополнительная процентная ставка не должен быть меньше 3%.';
          suspend;
        end
    end
    if (pr_product_kind = 2) then
    begin
      if (pr_code containing '.GR.' and pr_code containing '.par.') then
        begin
          if (round(K_Percent_Per_Day * 365,4) > 42.5) then
            begin
              res_error_is_critical = 1;
              res_error_text = 'Контракт параллельный, годовая процентная ставка не должно быть выше 42.5%.';
              suspend;
            end
            if (round(K_Percent_Per_Day * 365,4) < 33) then
            begin
              res_error_is_critical = 1;
              res_error_text = 'Контракт параллельный, годовая процентная ставка не должно быть ниже 33%.';
              suspend;
            end                                                      
    
            if (pr_code_old not containing '.VYGOD.' and K_Percent_Per_Day_old - 2 <> round(K_Percent_Per_Day * 365,4) and K_Percent_Per_Day_old >= 35) then
            begin                  
              res_error_is_critical = 1;
              res_error_text = 'Процентная ставка указана не верно!';
              suspend;
            end
            if (pr_code_old containing '.VYGOD.') then
            begin                  
                select round(K_Percent_Per_Day * 365,4)
                from kontract, kontstate, product
                where k_from_kf_id = :kf_id
                and k_id = ks_from_k_id
                and ks_state = 1
                and ks_date_begin < :kf_disb_date  
                and k_from_pr_id = pr_id     
                and pr_product_kind = 1
                and pr_code containing '.GR.'
                and pr_code not containing '.PAR.'
                and pr_code not containing '.VYGOD.'
                order by ks_date_begin desc
                rows 1
                into K_Percent_Per_Day_old;
                if (K_Percent_Per_Day_old is not null and
                    K_Percent_Per_Day_old - 2 <> round(K_Percent_Per_Day * 365,4) and
                    K_Percent_Per_Day_old >= 35) then
                  begin
                    res_error_is_critical = 1;
                    res_error_text = 'Процентная ставка указана не верно!';
                    suspend;
                  end
                if (K_Percent_Per_Day_old is not null and round(K_Percent_Per_Day * 365,4) <> 40.5) then
                  begin
                    res_error_is_critical = 1;
                    res_error_text = 'Процентная ставка указана не верно!';
                    suspend;
                  end
            end
            if (K_Percent_Per_Day_old is null) then K_Percent_Per_Day_old = 42.5;
    
            if (k_kommis_perc <> 3) then
            begin                  
              res_error_is_critical = 1;
              res_error_text = 'Дополнительная процентная ставка должен быть 3%!';
              suspend;
            end
    
            if (k_kolich_repay not between 3 and 18) then
            begin                  
              res_error_is_critical = 1;
              res_error_text = 'Срок кредита указано не верно!';
              suspend;
            end
        end
      if (pr_code containing '.IND.' and pr_code containing '.PAR.') then
      begin
        if (k_amount_approved not between 15000 and 30000) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Сумма кредита должен быть от 15 000 до 30 000.';
          suspend;
        end
        if (round(K_Percent_Per_Day * 365,4) > 36.9 and round(K_Percent_Per_Day * 365,4) - K_Percent_Per_Day_old <> 2) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Процентная ставка должна быть меньше: '||(K_Percent_Per_Day_old - 2);
          suspend;
        end else
        if (round(K_Percent_Per_Day * 365,4) < 35) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Процентная ставка не должна быть меньше: 35%';
          suspend;
        end
        if (:k_kommis_perc <> 3.5) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Доп. процентная ставка должна быть 3,5%';
          suspend;
        end
      end
    end 
    if (pr_product_kind = 3 and pr_code containing '.BIRGE.') then -- проверка продукта "Бирге"
    begin              
        select round(K_Percent_Per_Day * 365,4), k_kommis_perc, k_id, pr_code
        from kontract, kontstate, product
        where k_from_kf_id = :kf_id
        and k_id = ks_from_k_id
        and ks_state = 1
        and ks_date_begin < :kf_disb_date  
        and k_from_pr_id = pr_id
        and pr_code containing '.IND.'
        and pr_code not containing '.PAR.'
        order by ks_date_begin desc
        rows 1
        into K_Percent_Per_Day_old, k_kommis_perc_old, k_id_temp, pr_code_old;
        select max(ok_ol_date - ok_sh_date)
        from operkred
        where ok_ol_is_active = 1
        and ok_from_k_id = :k_id_temp
        and ok_type in (4,5,9)
        and ok_os + ok_perc < 0
        into oks_delinq_day;
        if (pr_code_old not containing '.EC.') then
        begin
        kommis_snijen = 0;
        kommis_snijen = case
                          when oks_delinq_day <= 0 then -0.1
                          when oks_delinq_day between 1 and 7 then 0.1
                          when oks_delinq_day between 8 and 15 then 0.2
                          when oks_delinq_day between 16 and 30 then 0.3
                          else 100
                        end;
        if (kommis_snijen between 3.5 and 3.9) then  k_kommis_perc_old = k_kommis_perc_old + kommis_snijen;
        if (kommis_snijen > 3.9) then  k_kommis_perc_old = 3.9;
        if (kommis_snijen < 3.5) then  k_kommis_perc_old = 3.5;
        if (k_kommis_perc < k_kommis_perc_old and pr_code not containing '.PAR.') then
          begin
            res_error_is_critical = 1;
            res_error_text='Доп. процент должен быть: '||k_kommis_perc_old;
            suspend;
          end
        end else
        if (pr_code_old containing '.EC.') then
        begin
        kommis_snijen = 0;
        kommis_snijen = case
                          when oks_delinq_day <= 0 then -0.1
                          when oks_delinq_day between 1 and 7 then 0.3
                          when oks_delinq_day between 8 and 15 then 0.5
--                          when oks_delinq_day between 16 and 30 then 0.3
                          else 100
                        end;
        if (kommis_snijen between 4.5 and 6.9) then  k_kommis_perc_old = k_kommis_perc_old + kommis_snijen;
        if (kommis_snijen > 6.9) then  k_kommis_perc_old = 6.9;
        if (kommis_snijen < 4.5) then  k_kommis_perc_old = 4.5;
        if (k_kommis_perc < k_kommis_perc_old and pr_code not containing '.PAR.') then
          begin
            res_error_is_critical = 1;
            res_error_text='Доп. процент должен быть: '||k_kommis_perc_old;
            suspend;
          end    
        if (k_circle = 1 and :k_kommis_perc < 6.9) then
          begin
            res_error_is_critical = 1;
            res_error_text = 'Дополнительная процентная ставка должен быть 6.9%!.';
            suspend;
          end
        end
    --Сумма кредита
      if (k_amount_approved > 55000) then
      begin
         res_error_is_critical = 1;
         res_error_text = 'Сумма индив. кредита не должно превышает 55тыс.';
         suspend;
      end 
      if (k_amount_approved < 15000) then
      begin
         res_error_is_critical = 1;
         res_error_text = 'Сумма индив. кредита не должно быть меньше 15 000.';
         suspend;
      end     
        
      if (:k_circle = 1 and pr_code_old containing '.EC.' and :k_kommis_perc  < 6.9) then
        begin
          res_error_is_critical = 1;
          res_error_text='Доп. процент должен быть: 6.9%';
          suspend;
        end
--            Проверка процентной ставки
      if (not(round(K_Percent_Per_Day * 365,4) between 37 and 43.5)) then
      begin
         if (pr_code not containing '.EC.') then   -- < 21.09.2015 igusev  добавил строку
         begin
             res_error_is_critical = 1;
             res_error_text = 'Контракт Индив., годовая процентная ставка не правильно указана.';
             suspend;
         end
      end
    end
  end
end
