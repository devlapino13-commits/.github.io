CREATE PROCEDURE HP_PVNOPERS_SAVE(
    PO_ID D_INTEGER,
    IS_MB_CARD D_BOOL,
    OPER_TYPE D_SMALLINT,
    C_ID D_ID,
    CARD_ID D_INTEGER,
    AMOUNT D_CURRENCY,
    OFFICE_ID D_INTEGER,
    ACC_KT D_STR_20,
    K_ID D_ID,
    TICKET_NO D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    OL_ID D_ID)
AS
declare variable acc_dt d_str_20;
declare variable is_cashing d_bool;
declare variable status_id d_smallint;
declare variable old_oper_type d_smallint;
declare variable old_card_id d_integer;
declare variable old_c_id d_id;
declare variable old_k_id d_id;
declare variable old_amount d_currency;
declare variable old_ticket_no d_str_code;
declare variable old_un_id d_id;
declare variable old_office_id d_integer;
declare variable old_is_mb_card d_smallint;
declare variable old_acc_kt d_str_20;
declare variable card_disb_office_id d_id;
begin
  if (po_id is null and is_mb_card = 1) then
    exception e_message 'Не передан po_id';

  if (oper_type = 0 and is_mb_card = 0) then
    exception e_message 'операция обналичивания доступна только для карт МБулак';
  else if (oper_type = 0 and is_mb_card = 1) then
    status_id = f_enum('pvnopers_status', 'writeoff_check', 'id');
  else if (oper_type = 1 and is_mb_card = 0) then
    status_id = f_enum('pvnopers_status', 'operation_ok', 'id');
  else if (oper_type = 1 and is_mb_card = 1) then
    status_id = f_enum('pvnopers_status', 'writeoff_check', 'id');
    
  if (oper_type = 0) then
  begin
      if (coalesce(acc_kt, '') = '') then
        exception e_message 'Счет Кт не определен';
    
      acc_dt = (select al_id from acc_list where al_from_reg_id = :office_id and al_currency = 417 and al_account = '10024/2'); -- Операции по ПВН
    
      if (coalesce(acc_dt, '') = '') then
        exception e_message 'Счет Дт не определен';
    
      select c_id,
             is_cashing
      from find_client_by_mb_card(:card_id)
      into :c_id,
           :is_cashing;
    
      if (c_id is null) then
        exception e_message 'Клиент не определен';
    
      if (is_cashing <> 1) then
        exception e_message 'С данной карты запрещено обналичивание';
  end
  else if (oper_type = 1) then
  begin
      if (k_id is null) then
        exception e_message 'id кредита не предано';
  end

  if ((is_mb_card = 0) and (ticket_no is null)) then
    exception e_message 'Не указан № чека';

  if (po_id is not null) then
  begin
      select po_oper_type,
             po_card_id,
             po_from_c_id,
             po_from_k_id,
             po_amount,
             po_ticket_no,
             po_un_id,
             po_office_id,
             po_is_mb_card,
             po_acc_kt
      from pvnopers po
      where po.po_id = :po_id
      into :old_oper_type,
           :old_card_id,
           :old_c_id,
           :old_k_id,
           :old_amount,
           :old_ticket_no,
           :old_un_id,
           :old_office_id,
           :old_is_mb_card,
           :old_acc_kt;

      if ((old_oper_type <> oper_type) or
          (old_card_id <> card_id) or
          (old_c_id <> c_id) or
          --(coalesce(old_k_id, -777) <> coalesce(k_id, -777)) or
          (old_amount <> amount) or
          (coalesce(old_ticket_no, '') <> coalesce(ticket_no, '')) or
          (old_un_id <> un_id) or
          (old_office_id <> office_id) or
          (old_is_mb_card <> is_mb_card) or
          (coalesce(old_acc_kt, '') <> coalesce(acc_kt, ''))) then
      exception e_message 'Данные были изменены, хотя не должны были меняться, с момента проверки баланса';
  end
  else
  begin
      if (card_id is not null) then
      begin
          select cpl.cpl_from_reg_id
          from cardlist cal
          inner join cardplace cpl on cal.cal_last_cpl_id = cpl.cpl_id
          where cal.cal_card_id = :card_id
          into :card_disb_office_id;
      end

      insert into pvnopers (po_date, po_oper_type, po_card_id, po_from_c_id, po_from_k_id, po_amount,
                            po_ticket_no, po_un_id, po_office_id, po_status, po_is_mb_card, po_acc_kt,
                            po_card_office_id)
      values (current_date, :oper_type, :card_id, :c_id, :k_id, :amount,
              :ticket_no, :un_id, :office_id, :status_id, :is_mb_card, :acc_kt,
              :card_disb_office_id)
      returning po_id
      into :po_id;
  end
  --------------
  if (oper_type = 1) then
  begin
    if ((select kont.k_disb_date
        from kontract kont
        where kont.k_id = :k_id) = current_date) then
            exception e_message 'Погашение кредита в день выдачи запрещено';
  end

  if (oper_type = 0) then -- если обналичивание
     execute procedure hp_pvnopers_writeoff(:po_id); -- создается задача на списание с карты
  else if (oper_type = 1) then
  begin
    if (is_mb_card = 0) then
        execute procedure hp_pvnopers_repayment(:po_id); -- формируется операция погашения по займу
    else
        execute procedure hp_pvnopers_writeoff(:po_id); -- создается задача на списание с карты
  end

  select po_from_ol_id
  from pvnopers
  where po_id = :po_id
  into :ol_id;

  if (ol_id is not null and (oper_type = 0 or (oper_type = 1 and is_mb_card <> 0))) then -- если обналичивание
    execute procedure hp_opercash_pvnoper_save(:po_id);

  suspend;
end
