CREATE PROCEDURE HP_FILL_KN_KONTCALCPERC(
    K_ID ID_TYPE,
    LAST_PERIOD INTEGER)
AS
declare variable k_is_devided integer;
declare variable k_base_k_id id_type;
declare variable loan_kc_list d_str_255;
declare variable loan_amount_list d_str_255;
declare variable tmp_kc_list d_str_255;
declare variable tmp_amount_list d_str_255;
declare variable loan_amount curr_type;
declare variable tmp_loan_amount curr_type;
declare variable kcp_period integer;
declare variable kc_id id_type;
declare variable kc_amount curr_type;
declare variable tmp_k_id id_type;
declare variable close_date date;
declare variable kcp_perc_begin_saldo curr_type;
declare variable kcp_kommis_begin_saldo curr_type;
declare variable kcp_out_perc_begin_saldo curr_type;
declare variable kcp_period_prv integer;
declare variable kcp_perc_calced curr_type = 0;
declare variable kcp_kommis_calced curr_type = 0;
declare variable kcp_out_perc_calced curr_type;
declare variable diff_kcp_perc_begin_saldo curr_type;
declare variable diff_kcp_kommis_begin_saldo curr_type;
declare variable diff_kcp_out_perc_begin_saldo curr_type;
declare variable diff_kcp_perc_calced curr_type = 0;
declare variable diff_kcp_kommis_calced curr_type = 0;
declare variable diff_kcp_out_perc_calced curr_type;
declare variable curr_year integer;
declare variable curr_month integer;
declare variable kc_kommis curr_type;
declare variable loan_kommis curr_type;
declare variable loan_kommis_list d_str_255;
declare variable tmp_kommis_list d_str_255;
declare variable tmp_loan_kommis curr_type;
begin
-- процедура делит проценты на уровне участников и заполняет таблицу kn_kontcalcperc
  select k_is_devided, k_base_k_id
  from kontract
  where k_id = :k_id
  into k_is_devided, k_base_k_id;

  if (coalesce(last_period, 0) = 0) then
    exception e_default 'Last period is empty';

-- амнистированные (дочерние) займы не делим, они заполняются параллельно при делении главного займа
  if (k_is_devided = 1 and
      k_base_k_id <> :k_id) then
    exit;

  loan_kc_list = '';
  loan_amount = 0;
  kcp_period_prv = 0;
  select list(kc_id), list(kc_amount_approved),
         list(iif(kc_amount_approved > 0, coalesce(kc_amount_approved * k_kommis_perc, 0) - coalesce(kc_kommis_lgota, 0), 0)),
         sum(kc_amount_approved),
         sum(iif(kc_amount_approved > 0, coalesce(kc_amount_approved * k_kommis_perc, 0) - coalesce(kc_kommis_lgota, 0), 0))
  from knclient, kontract
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0 and
        k_id = kc_from_k_id
  into loan_kc_list, loan_amount_list, loan_kommis_list, loan_amount, loan_kommis;

  if (loan_kc_list is null) then
    loan_kc_list = '';
  if (loan_amount_list is null) then
    loan_amount_list = '';
  if (loan_kommis_list is null) then
    loan_kommis_list = '';
  if (loan_amount is null) then
    loan_amount = 0;
  if (loan_kommis is null) then
    loan_kommis = 0;

  if (loan_amount = 0) then
    exit;

  if (k_is_devided = 0) then
  begin
    delete from kn_kontcalcperc
    where kcp_from_k_id = :k_id;
  end
  else
  begin
    delete from kn_kontcalcperc
    where kcp_from_k_id = :k_id;

    for select kd_k_new
        from kontdevides
        where kd_k_orig = :k_id
        into tmp_k_id
    do
    begin
      delete from kn_kontcalcperc
      where kcp_from_k_id = :tmp_k_id;
    end
  end

  for select kcp_period
      from kontcalcperc
      where kcp_from_k_id = :k_id
      order by kcp_period
      into :kcp_period
  do
  begin
    curr_year = kcp_period / 100;
    curr_month = mod(kcp_period, 100);

    if (curr_month > 1) then
      curr_month = curr_month - 1;
    else
    begin
      curr_month = 12;
      curr_year = curr_year - 1;
    end
    kcp_period_prv = curr_year * 100 + curr_month;

    tmp_kc_list = loan_kc_list;
    tmp_amount_list = loan_amount_list;
    tmp_loan_amount = loan_amount;
    tmp_kommis_list = loan_kommis_list;
    tmp_loan_kommis = loan_kommis;

    kcp_perc_begin_saldo = 0;
    kcp_kommis_begin_saldo = 0;
    kcp_out_perc_begin_saldo = 0;
    kcp_perc_calced = 0;
    kcp_kommis_calced = 0;
    kcp_out_perc_calced = 0;

    diff_kcp_perc_begin_saldo = 0;
    diff_kcp_kommis_begin_saldo = 0;
    diff_kcp_out_perc_begin_saldo = 0;
    diff_kcp_perc_calced = 0;
    diff_kcp_kommis_calced = 0;
    diff_kcp_out_perc_calced = 0;

    if (k_is_devided = 1) then
    begin
-- к текущему составу займов добавляем ранее амнистированных, если по ним нет начислений в ведомости
      for select kd_k_new
          from kontdevides
          where kd_k_orig = :k_id and
                kd_k_new <> :k_id
          into :tmp_k_id
      do
      begin
        if (not(exists(select *
                       from kontcalcperc
                       where kcp_from_k_id = :tmp_k_id and
                             kcp_period = :kcp_period))) then
        begin
-- начисление в ведомости нет. Проверяем, не закрыт ли займ к этому периоду
          close_date = null;

          select ks_date_begin
          from kontstate
          where ks_from_k_id = :tmp_k_id and
                ks_ol_is_delete = 0 and
                ks_state = 2
          into :close_date;

-- добавляем в текущий состав участников
          if (close_date is null or (kcp_period <= extract(year from close_date) * 100 + extract(month from close_date))) then
          begin
            for select kc_id, kc_amount_approved,
                       iif(kc_amount_approved > 0, coalesce(kc_amount_approved * k_kommis_perc, 0) - coalesce(kc_kommis_lgota, 0), 0)
                from knclient, kontract
                where kc_from_k_id = :tmp_k_id and
                      kc_from_k_id = k_id and
                      kc_amount_approved > 0
                into :kc_id, kc_amount, kc_kommis
            do
            begin
              tmp_kc_list = kc_id || ',' || tmp_kc_list;
              tmp_amount_list = kc_amount || ',' || tmp_amount_list;
              tmp_loan_amount = tmp_loan_amount + kc_amount;

              tmp_kommis_list = kc_kommis || ',' || tmp_kommis_list;
              tmp_loan_kommis = tmp_loan_kommis + kc_kommis;

            end
          end
        end
        else
        begin
          kc_id = null;
          select kc_id, kc_amount_approved,
                 iif(kc_amount_approved > 0, coalesce(kc_amount_approved * k_kommis_perc, 0) - coalesce(kc_kommis_lgota, 0), 0)
          from knclient, kontract
          where kc_from_k_id = :tmp_k_id and
                kc_from_k_id = k_id and
                kc_amount_approved > 0
          into :kc_id, kc_amount, kc_kommis;

          if (not(exists(select *
                         from kontcalcperc
                         where kcp_from_k_id = :tmp_k_id and
                               kcp_period = :kcp_period_prv))) then
          begin
-- если амнистированный займ уже имеется в ведомости и ввстречается первый раз,
-- переносим без изменений, с актуальным сальдо
            kcp_perc_begin_saldo = 0;
            kcp_kommis_begin_saldo = 0;
            kcp_out_perc_begin_saldo = 0;
            kcp_perc_calced = 0;
            kcp_kommis_calced = 0;
            kcp_out_perc_calced = 0;

            insert into kn_kontcalcperc (kcp_period, kcp_from_kc_id, kcp_from_k_id, kcp_from_reg_id, kcp_perc_saldo,
                                         kcp_transfer_reg_id, kcp_perc_calced, kcp_perc_payed, kcp_perc_writeoff,
                                         kcp_perc_begin_saldo, kcp_perc_transfer_payed, kcp_perc_transfer_writeoff,
                                         kcp_kommis_calced, kcp_kommis_payed, kcp_kommis_writeoff, kcp_kommis_saldo,
                                         kcp_kommis_begin_saldo, kcp_kommis_transfer_payed,
                                         kcp_kommis_transfer_writeoff, kcp_perc_account, kcp_kommis_account,
                                         kcp_loan_is_closed, kcp_out_perc_begin_saldo, kcp_out_perc_calced,
                                         kcp_out_perc_saldo)
            select :kcp_period, :kc_id, :tmp_k_id, cl_kcp_from_reg_id, cl_kcp_perc_saldo, cl_kcp_transfer_reg_id,
                   cl_kcp_perc_calced, cl_kcp_perc_payed, cl_kcp_perc_writeoff, cl_kcp_perc_begin_saldo,
                   cl_kcp_perc_transfer_payed, cl_kcp_perc_transfer_writeoff, cl_kcp_kommis_calced, cl_kcp_kommis_payed,
                   cl_kcp_kommis_writeoff, cl_kcp_kommis_saldo, cl_kcp_kommis_begin_saldo, cl_kcp_kommis_transfer_payed,
                   cl_kcp_kommis_transfer_writeoff, cl_kcp_perc_account, cl_kcp_kommis_account, cl_kcp_loan_is_closed,
                   cl_kcp_out_perc_begin_saldo, cl_kcp_out_perc_calced, cl_kcp_out_perc_saldo
            from hp_devide_kn_kontcalcperc(:tmp_k_id, :kcp_period, :kc_id, :kc_amount, :kc_kommis)
            returning kcp_perc_begin_saldo, kcp_kommis_begin_saldo, kcp_out_perc_begin_saldo, kcp_perc_calced, kcp_kommis_calced, kcp_out_perc_calced
            into kcp_perc_begin_saldo, kcp_kommis_begin_saldo, kcp_out_perc_begin_saldo, kcp_perc_calced,
                 kcp_kommis_calced, kcp_out_perc_calced;

            select :kcp_perc_begin_saldo - kcp_perc_begin_saldo, :kcp_kommis_begin_saldo - kcp_kommis_begin_saldo,
                   :kcp_out_perc_begin_saldo - kcp_out_perc_begin_saldo, :kcp_perc_calced - kcp_perc_calced,
                   :kcp_kommis_calced - kcp_kommis_calced, :kcp_out_perc_calced - kcp_out_perc_calced
            from kontcalcperc
            where kcp_from_k_id = :tmp_k_id and
                  kcp_period = :kcp_period
            into kcp_perc_begin_saldo, kcp_kommis_begin_saldo, kcp_out_perc_begin_saldo, kcp_perc_calced,
                 kcp_kommis_calced, kcp_out_perc_calced;

            diff_kcp_perc_begin_saldo = diff_kcp_perc_begin_saldo + kcp_perc_begin_saldo;
            diff_kcp_kommis_begin_saldo = diff_kcp_kommis_begin_saldo + kcp_kommis_begin_saldo;
            diff_kcp_out_perc_begin_saldo = diff_kcp_out_perc_begin_saldo + kcp_out_perc_begin_saldo;
            diff_kcp_perc_calced = diff_kcp_perc_calced + kcp_perc_calced;
            diff_kcp_kommis_calced = diff_kcp_kommis_calced + kcp_kommis_calced;
            diff_kcp_out_perc_calced = diff_kcp_out_perc_calced + kcp_out_perc_calced;
          end
          else
          begin
-- в ведомости данный займ встречается не первый месяц,
-- поэтому переносим без изменений
            insert into kn_kontcalcperc (kcp_period, kcp_from_kc_id, kcp_from_k_id, kcp_from_reg_id, kcp_perc_saldo,
                                         kcp_transfer_reg_id, kcp_perc_calced, kcp_perc_payed, kcp_perc_writeoff,
                                         kcp_perc_begin_saldo, kcp_perc_transfer_payed, kcp_perc_transfer_writeoff,
                                         kcp_kommis_calced, kcp_kommis_payed, kcp_kommis_writeoff, kcp_kommis_saldo,
                                         kcp_kommis_begin_saldo, kcp_kommis_transfer_payed,
                                         kcp_kommis_transfer_writeoff, kcp_perc_account, kcp_kommis_account,
                                         kcp_loan_is_closed, kcp_out_perc_begin_saldo, kcp_out_perc_calced,
                                         kcp_out_perc_saldo)
            select kcp_period, :kc_id, :tmp_k_id, kcp_from_reg_id, kcp_perc_saldo, kcp_transfer_reg_id, kcp_perc_calced,
                   kcp_perc_payed, kcp_perc_writeoff, kcp_perc_begin_saldo, kcp_perc_transfer_payed,
                   kcp_perc_transfer_writeoff, kcp_kommis_calced, kcp_kommis_payed, kcp_kommis_writeoff,
                   kcp_kommis_saldo, kcp_kommis_begin_saldo, kcp_kommis_transfer_payed, kcp_kommis_transfer_writeoff,
                   kcp_perc_account, kcp_kommis_account, kcp_loan_is_closed, kcp_out_perc_begin_saldo,
                   kcp_out_perc_calced, kcp_out_perc_saldo
            from kontcalcperc
            where kcp_period = :kcp_period and
                  kcp_from_k_id = :tmp_k_id;
          end
        end -- конец если займ встречается
      end -- конец списка амнистированных
    end -- конец формирование составка

    insert into kn_kontcalcperc (kcp_period, kcp_from_kc_id, kcp_from_k_id, kcp_from_reg_id, kcp_perc_saldo,
                                 kcp_transfer_reg_id, kcp_perc_calced, kcp_perc_payed, kcp_perc_writeoff,
                                 kcp_perc_begin_saldo, kcp_perc_transfer_payed, kcp_perc_transfer_writeoff,
                                 kcp_kommis_calced, kcp_kommis_payed, kcp_kommis_writeoff, kcp_kommis_saldo,
                                 kcp_kommis_begin_saldo, kcp_kommis_transfer_payed, kcp_kommis_transfer_writeoff,
                                 kcp_perc_account, kcp_kommis_account, kcp_loan_is_closed, kcp_out_perc_begin_saldo,
                                 kcp_out_perc_calced, kcp_out_perc_saldo)
    select :kcp_period, kc_id, null, cl_kcp_from_reg_id, cl_kcp_perc_saldo, cl_kcp_transfer_reg_id, cl_kcp_perc_calced,
           cl_kcp_perc_payed, cl_kcp_perc_writeoff, cl_kcp_perc_begin_saldo, cl_kcp_perc_transfer_payed,
           cl_kcp_perc_transfer_writeoff, cl_kcp_kommis_calced, cl_kcp_kommis_payed, cl_kcp_kommis_writeoff,
           cl_kcp_kommis_saldo, cl_kcp_kommis_begin_saldo, cl_kcp_kommis_transfer_payed,
           cl_kcp_kommis_transfer_writeoff, cl_kcp_perc_account, cl_kcp_kommis_account, cl_kcp_loan_is_closed,
           cl_kcp_out_perc_begin_saldo, cl_kcp_out_perc_calced, cl_kcp_out_perc_saldo
    from hp_devide_kn_kontcalcperc(:k_id, :kcp_period, :tmp_kc_list, :tmp_amount_list, :tmp_kommis_list, :diff_kcp_perc_begin_saldo, :diff_kcp_kommis_begin_saldo, :diff_kcp_out_perc_begin_saldo, :diff_kcp_perc_calced, :diff_kcp_kommis_calced, :diff_kcp_out_perc_calced);

  end -- конец периода

  if (kcp_period is null) then
    exit;

-- основной займ закрылся раньше, переводим остальных самостоятельно
  while (kcp_period < last_period) do
  begin
    kcp_period_prv = kcp_period;

    curr_year = kcp_period / 100;
    curr_month = mod(kcp_period, 100);

    if (curr_month < 12) then
      curr_month = curr_month + 1;
    else
    begin
      curr_month = 1;
      curr_year = curr_year + 1;
    end
    kcp_period = curr_year * 100 + curr_month;

    for select kd_k_new
        from kontdevides
        where kd_k_orig = :k_id and
              kd_k_new <> :k_id
        into :tmp_k_id
    do
    begin
      if (exists(select *
                 from kontcalcperc
                 where kcp_from_k_id = :tmp_k_id and
                       kcp_period = :kcp_period)) then
      begin
        kc_id = null;
        select kc_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :tmp_k_id and
              kc_amount_approved > 0
        into :kc_id, kc_amount;

-- в ведомости данный займ встречается не первый месяц,
-- поэтому переносим без изменений
        insert into kn_kontcalcperc (kcp_period, kcp_from_kc_id, kcp_from_k_id, kcp_from_reg_id, kcp_perc_saldo,
                                     kcp_transfer_reg_id, kcp_perc_calced, kcp_perc_payed, kcp_perc_writeoff,
                                     kcp_perc_begin_saldo, kcp_perc_transfer_payed, kcp_perc_transfer_writeoff,
                                     kcp_kommis_calced, kcp_kommis_payed, kcp_kommis_writeoff, kcp_kommis_saldo,
                                     kcp_kommis_begin_saldo, kcp_kommis_transfer_payed, kcp_kommis_transfer_writeoff,
                                     kcp_perc_account, kcp_kommis_account, kcp_loan_is_closed, kcp_out_perc_begin_saldo,
                                     kcp_out_perc_calced, kcp_out_perc_saldo)
        select kcp_period, :kc_id, :tmp_k_id, kcp_from_reg_id, kcp_perc_saldo, kcp_transfer_reg_id, kcp_perc_calced,
               kcp_perc_payed, kcp_perc_writeoff, kcp_perc_begin_saldo, kcp_perc_transfer_payed,
               kcp_perc_transfer_writeoff, kcp_kommis_calced, kcp_kommis_payed, kcp_kommis_writeoff, kcp_kommis_saldo,
               kcp_kommis_begin_saldo, kcp_kommis_transfer_payed, kcp_kommis_transfer_writeoff, kcp_perc_account,
               kcp_kommis_account, kcp_loan_is_closed, kcp_out_perc_begin_saldo, kcp_out_perc_calced,
               kcp_out_perc_saldo
        from kontcalcperc
        where kcp_period = :kcp_period and
              kcp_from_k_id = :tmp_k_id;
      end -- конец если займ встречается
    end -- конец списка амнистированных

  end
end
