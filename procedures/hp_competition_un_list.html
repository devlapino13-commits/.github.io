CREATE PROCEDURE HP_COMPETITION_UN_LIST(
    RD_CODE VARCHAR(255),
    UN_ID DECIMAL(18,0))
RETURNS (
    CPP_ID INTEGER,
    CPP_PERIOD_NAME VARCHAR(255),
    CPR_FINAL_POINTS DECIMAL(18,4),
    CPP_ORDER_TXT VARCHAR(100),
    CPP_ORDERNO INTEGER)
AS
declare variable lo_id decimal(18,0);
declare variable rd_id integer;
declare variable order_diff integer;
declare variable cpr_order_last integer;
declare variable cpr_order_in_group integer;
declare variable cpp_date_end date;
declare variable color_green varchar(20);
declare variable color_red varchar(20);
declare variable res_team_id integer;
declare variable db_id integer;
declare variable jk_system_code d_str_100;
begin

  :db_id = f_get_db_id();
  select un_from_c_id
  from username
  where un_id = :un_id
  into :lo_id;
  if (lo_id is null) then
    exit;

  select rd_id
  from refdata
  where rd_code = :rd_code
  into :rd_id;
  if (rd_id is null) then
    exit;

  color_green = '#02B316';
  color_red = 'red';

  select jk_system_code
  from hr_pstatus, joblist, jobkind
  where ps_from_c_id = :lo_id and
        ps_is_working = 1 and
        ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id
  into jk_system_code;

  if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner', 'JobSystem.Specialist_Expert')) then
  begin
    select res_team_id
    from aarep_get_lo_team(0, :lo_id, null)
    into :res_team_id;
  end
  else
  if (jk_system_code in ('JobSystem.Region_KM') and :db_id = 81001) then
  begin
    select res_team_id
    from aarep_get_mgr_team(:rd_id, :un_id)
    into :res_team_id;
  end

  cpr_order_last = null;
  if (:db_id in (39801, 41701) and
      :rd_code containing 'Champions_League') then
  begin
    for select cpp_id, cpp_period_name, cpr_order_in_group, cpr_final_points, cpp_orderno, cpp_date_end
        from competdetails, competresult, competperiod
        where cpd_object_kind = 0 and
              cpd_object_id = :lo_id and
              cpd_from_cpr_id = cpr_id and
              cpr_from_cpp_id = cpp_id and
              cpp_status = 1 -- подтвежденные
              and
              cpp_from_rd_id = :rd_id

        order by 5
        into :cpp_id, cpp_period_name, cpr_order_in_group, cpr_final_points, cpp_orderno, cpp_date_end
    do
    begin
      cpp_order_txt = '';
      cpp_period_name = cpp_period_name || ' ' || iif(db_id <> 94401, 'на', 'на') || ' <b>' || extract(day from cpp_date_end) || ' ' ||(case
                                                                                                                                              when extract(month from cpp_date_end) = 1 then iif(db_id <> 94401, 'янв.', 'yanv')
                                                                                                                                              when extract(month from cpp_date_end) = 2 then iif(db_id <> 94401, 'фев.', 'fevr')
                                                                                                                                              when extract(month from cpp_date_end) = 3 then iif(db_id <> 94401, 'март', 'mart')
                                                                                                                                              when extract(month from cpp_date_end) = 4 then iif(db_id <> 94401, 'апр.', 'aprl')
                                                                                                                                              when extract(month from cpp_date_end) = 5 then iif(db_id <> 94401, 'май', 'may')
                                                                                                                                              when extract(month from cpp_date_end) = 6 then iif(db_id <> 94401, 'июня.', 'iyun')
                                                                                                                                              when extract(month from cpp_date_end) = 7 then iif(db_id <> 94401, 'июля', 'iyul')
                                                                                                                                              when extract(month from cpp_date_end) = 8 then iif(db_id <> 94401, 'авг.', 'avqs')
                                                                                                                                              when extract(month from cpp_date_end) = 9 then iif(db_id <> 94401, 'сен.', 'sent')
                                                                                                                                              when extract(month from cpp_date_end) = 10 then iif(db_id <> 94401, 'окт.', 'oktb')
                                                                                                                                              when extract(month from cpp_date_end) = 11 then iif(db_id <> 94401, 'ноя.', 'noyb')
                                                                                                                                              when extract(month from cpp_date_end) = 12 then iif(db_id <> 94401, 'дек.', 'dekb')
                                                                                                                                            end);

      cpp_order_txt = '';
      if (cpr_order_last is not null) then
      begin
        order_diff = cpr_order_last - cpr_order_in_group;
        if (order_diff = 0) then
          cpp_order_txt = cpr_order_in_group || ' (0)';
        if (order_diff > 0) then
          cpp_order_txt = cpr_order_in_group || ' (<font color="' || color_green || '">+' || order_diff || '</font>)';
        if (order_diff < 0) then
          cpp_order_txt = cpr_order_in_group || ' (<font color="' || color_red || '">' || order_diff || '</font>)';
      end
      else
        cpp_order_txt = cpr_order_in_group;

      suspend;
      cpr_order_last = cpr_order_in_group;
    end
  end
  else
  begin
    for select cpp_id, cpp_period_name, cpr_order_in_group, cpr_final_points, cpp_orderno, cpp_date_end
        from competdetails, competresult, competperiod
        where cpd_object_kind = 0 and
              cpd_object_id = :lo_id and
              cpd_from_cpr_id = cpr_id and
              cpr_from_cpp_id = cpp_id and
              cpp_status = 1 -- подтвежденные
              and
              cpp_from_rd_id = :rd_id and
              cpp_stage = 0
        union
        select cpp_id, cpp_period_name, cpr_order_in_group, cpr_final_points, cpp_orderno, cpp_date_end
        from competresult, competperiod
        where cpr_object_id = :res_team_id and
              cpr_object_kind = -1 and
              cpr_from_cpp_id = cpp_id and
              cpp_status = 1 -- подтвежденные
              and
              cpp_from_rd_id = :rd_id and
              cpp_stage > 0
        order by 5
        into :cpp_id, cpp_period_name, cpr_order_in_group, cpr_final_points, cpp_orderno, cpp_date_end
    do
    begin
      cpp_order_txt = '';
      cpp_period_name = cpp_period_name || ' ' || iif(db_id <> 94401, 'на', 'на') || ' <b>' || extract(day from cpp_date_end) || ' ' ||(case
                                                                                                                                              when extract(month from cpp_date_end) = 1 then iif(db_id <> 94401, 'янв.', 'yanv')
                                                                                                                                              when extract(month from cpp_date_end) = 2 then iif(db_id <> 94401, 'фев.', 'fevr')
                                                                                                                                              when extract(month from cpp_date_end) = 3 then iif(db_id <> 94401, 'март', 'mart')
                                                                                                                                              when extract(month from cpp_date_end) = 4 then iif(db_id <> 94401, 'апр.', 'aprl')
                                                                                                                                              when extract(month from cpp_date_end) = 5 then iif(db_id <> 94401, 'май', 'may')
                                                                                                                                              when extract(month from cpp_date_end) = 6 then iif(db_id <> 94401, 'июня.', 'iyun')
                                                                                                                                              when extract(month from cpp_date_end) = 7 then iif(db_id <> 94401, 'июля', 'iyul')
                                                                                                                                              when extract(month from cpp_date_end) = 8 then iif(db_id <> 94401, 'авг.', 'avqs')
                                                                                                                                              when extract(month from cpp_date_end) = 9 then iif(db_id <> 94401, 'сен.', 'sent')
                                                                                                                                              when extract(month from cpp_date_end) = 10 then iif(db_id <> 94401, 'окт.', 'oktb')
                                                                                                                                              when extract(month from cpp_date_end) = 11 then iif(db_id <> 94401, 'ноя.', 'noyb')
                                                                                                                                              when extract(month from cpp_date_end) = 12 then iif(db_id <> 94401, 'дек.', 'dekb')
                                                                                                                                            end);

      cpp_order_txt = '';
      if (cpr_order_last is not null) then
      begin
        order_diff = cpr_order_last - cpr_order_in_group;
        if (order_diff = 0) then
          cpp_order_txt = cpr_order_in_group || ' (0)';
        if (order_diff > 0) then
          cpp_order_txt = cpr_order_in_group || ' (<font color="' || color_green || '">+' || order_diff || '</font>)';
        if (order_diff < 0) then
          cpp_order_txt = cpr_order_in_group || ' (<font color="' || color_red || '">' || order_diff || '</font>)';
      end
      else
        cpp_order_txt = cpr_order_in_group;

      suspend;
      cpr_order_last = cpr_order_in_group;
    end
  end
  if (:rd_code containing '2022_mo' and
      :db_id in (39801, 41701)) then
  begin
    for select cpp_id, cpp_period_name, cpr_order_in_group, cpr_final_points, cpp_orderno, cpp_date_end
        from competresult cpr
        inner join competdetails cpd on cpr.cpr_id = cpd.cpd_from_cpr_id

        inner join competperiod cpp on cpp.cpp_id = cpr.cpr_from_cpp_id and
              cpp.cpp_from_rd_id = :rd_id and
              cpp.cpp_status = 1
        where cpr.cpr_object_kind = f_objkind('clients', 'id') and
              cpr.cpr_object_id = :lo_id
        order by 5
        into :cpp_id, cpp_period_name, cpr_order_in_group, cpr_final_points, cpp_orderno, cpp_date_end
    do
    begin
      cpp_order_txt = '';
      cpp_period_name = cpp_period_name || ' ' || iif(db_id <> 94401, 'на', 'на') || ' <b>' || extract(day from cpp_date_end) || ' ' ||(case
                                                                                                                                              when extract(month from cpp_date_end) = 1 then iif(db_id <> 94401, 'янв.', 'yanv')
                                                                                                                                              when extract(month from cpp_date_end) = 2 then iif(db_id <> 94401, 'фев.', 'fevr')
                                                                                                                                              when extract(month from cpp_date_end) = 3 then iif(db_id <> 94401, 'март', 'mart')
                                                                                                                                              when extract(month from cpp_date_end) = 4 then iif(db_id <> 94401, 'апр.', 'aprl')
                                                                                                                                              when extract(month from cpp_date_end) = 5 then iif(db_id <> 94401, 'май', 'may')
                                                                                                                                              when extract(month from cpp_date_end) = 6 then iif(db_id <> 94401, 'июня.', 'iyun')
                                                                                                                                              when extract(month from cpp_date_end) = 7 then iif(db_id <> 94401, 'июля', 'iyul')
                                                                                                                                              when extract(month from cpp_date_end) = 8 then iif(db_id <> 94401, 'авг.', 'avqs')
                                                                                                                                              when extract(month from cpp_date_end) = 9 then iif(db_id <> 94401, 'сен.', 'sent')
                                                                                                                                              when extract(month from cpp_date_end) = 10 then iif(db_id <> 94401, 'окт.', 'oktb')
                                                                                                                                              when extract(month from cpp_date_end) = 11 then iif(db_id <> 94401, 'ноя.', 'noyb')
                                                                                                                                              when extract(month from cpp_date_end) = 12 then iif(db_id <> 94401, 'дек.', 'dekb')
                                                                                                                                            end);

      cpp_order_txt = '';
      if (cpr_order_last is not null) then
      begin
        order_diff = cpr_order_last - cpr_order_in_group;
        if (order_diff = 0) then
          cpp_order_txt = cpr_order_in_group || ' (0)';
        if (order_diff > 0) then
          cpp_order_txt = cpr_order_in_group || ' (<font color="' || color_green || '">+' || order_diff || '</font>)';
        if (order_diff < 0) then
          cpp_order_txt = cpr_order_in_group || ' (<font color="' || color_red || '">' || order_diff || '</font>)';
      end
      else
        cpp_order_txt = cpr_order_in_group;

      suspend;
      cpr_order_last = cpr_order_in_group;
    end
  end
end
