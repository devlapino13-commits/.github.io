CREATE PROCEDURE REFDATA_LIST(
    CODE_FILTER VARCHAR(255))
RETURNS (
    RD_ID INTEGER,
    RD_PARENT_ID INTEGER,
    RD_CODE VARCHAR(255),
    RD_MAIN_CODE VARCHAR(255),
    RD_NAME VARCHAR(255),
    RD_MEMO VARCHAR(255),
    RD_STR_ID VARCHAR(100),
    RD_ORDERNO INTEGER,
    RD_IS_ACTIVE INTEGER,
    RD_IS_LOCKED INTEGER,
    RD_TAG INTEGER,
    RD_CHILD_COUNT INTEGER)
AS
begin
  if (code_filter is null) then
    code_filter = '';
  if (code_filter = '') then
  begin
    for select rd_id, rd_parent_id, rd_code, rd_main_code, rd_name, rd_memo, rd_str_id, rd_orderno, rd_is_active,
               rd_is_locked, rd_tag
        from refdata
        order by rd_main_code, rd_parent_id, rd_id
        into :rd_id, :rd_parent_id, :rd_code, :rd_main_code, :rd_name, :rd_memo, :rd_str_id, :rd_orderno, :rd_is_active,
             :rd_is_locked, :rd_tag
    do
    begin
      rd_child_count = 0;
      select count(*)
      from refdata
      where rd_parent_id = :rd_id
      into :rd_child_count;
      if (rd_child_count is null) then
        rd_child_count = 0;
      suspend;
    end
  end
  else
  begin
    for select rd_id, rd_parent_id, rd_code, rd_main_code, rd_name, rd_memo, rd_str_id, rd_orderno, rd_is_active,
               rd_is_locked, rd_tag
        from refdata
        where (rd_main_code = :code_filter or
              rd_id = 0 or
              rd_code = :code_filter)
        order by rd_main_code, rd_parent_id, rd_id
        into :rd_id, :rd_parent_id, :rd_code, :rd_main_code, :rd_name, :rd_memo, :rd_str_id, :rd_orderno, :rd_is_active,
             :rd_is_locked, rd_tag
    do
    begin
      rd_child_count = 0;
      select count(*)
      from refdata
      where rd_parent_id = :rd_id
      into :rd_child_count;
      if (rd_child_count is null) then
        rd_child_count = 0;
      suspend;
    end
  end
end
