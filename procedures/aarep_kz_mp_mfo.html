    REP_REGION_ID D_INTEGER,

    REP_DATE_END D_DATE)
RETURNS (
    OD_11 D_CURRENCY,
    PERC_CALCED_11 D_CURRENCY,
    LLC_11 D_CURRENCY,
    KOL_KRED_11 D_INTEGER,
    KOL_CLI_11 D_INTEGER,
    OD_112 D_CURRENCY,
    PERC_CALCED_112 D_CURRENCY,
    LLC_112 D_CURRENCY,
    KOL_KRED_112 D_INTEGER,
    KOL_CLI_112 D_INTEGER,
    OD_112222 D_CURRENCY,
    PERC_CALCED_112222 D_CURRENCY,
    LLC_112222 D_CURRENCY,
    KOL_KRED_112222 D_INTEGER,
    KOL_CLI_112222 D_INTEGER,
    OD_12 D_CURRENCY,
    PERC_CALCED_12 D_CURRENCY,
    LLC_12 D_CURRENCY,
    KOL_KRED_12 D_INTEGER,
    KOL_CLI_12 D_INTEGER,
    OD_122 D_CURRENCY,
    PERC_CALCED_122 D_CURRENCY,
    LLC_122 D_CURRENCY,
    KOL_KRED_122 D_INTEGER,

    OD_1222 D_CURRENCY,
    PERC_CALCED_1222 D_CURRENCY,
    LLC_1222 D_CURRENCY,
    KOL_KRED_1222 D_INTEGER,
    KOL_CLI_1222 D_INTEGER,
    OD_122222 D_CURRENCY,
    PERC_CALCED_122222 D_CURRENCY,
    LLC_122222 D_CURRENCY,
    KOL_KRED_122222 D_INTEGER,
    KOL_CLI_122222 D_INTEGER,
    OD_4 D_CURRENCY,
    PERC_CALCED_4 D_CURRENCY,
    LLC_4 D_CURRENCY,
    KOL_KRED_4 D_INTEGER,
    KOL_CLI_4 D_INTEGER,

    OD_1222221 D_CURRENCY,
    OD_1222222 D_CURRENCY,
    OD_1222223 D_CURRENCY,
    OD_1222224 D_CURRENCY,
    PERC_CALCED_1222221 D_CURRENCY,
    PERC_CALCED_1222222 D_CURRENCY,
    PERC_CALCED_1222223 D_CURRENCY,
    PERC_CALCED_1222224 D_CURRENCY,
    LLC_1222221 D_CURRENCY,
    LLC_1222222 D_CURRENCY,
    LLC_1222223 D_CURRENCY,
    LLC_1222224 D_CURRENCY,
    KOL_KRED_1222221 D_INTEGER,
    KOL_KRED_1222222 D_INTEGER,
    KOL_KRED_1222223 D_INTEGER,
    KOL_KRED_1222224 D_INTEGER,
    KOL_CLI_1222221 D_INTEGER,
    KOL_CLI_1222222 D_INTEGER,
    KOL_CLI_1222223 D_INTEGER,
    KOL_CLI_1222224 D_INTEGER)
AS
declare variable OKS_DELINQ_DATE D_DATE;
declare variable OKS_BALANCE D_CURRENCY;
declare variable PRIOD D_INTEGER;
declare variable KCP_PERC_CALCED D_CURRENCY;
declare variable K_DISB_DATE D_DATE;
declare variable K_KOLICH_CLIENT_APPROVED D_SMALLINT;
declare variable RES_JUST_CLIENT_COUNT D_SMALLINT;
declare variable RD_ID D_INTEGER;
declare variable LK_PERCENT D_PERCENT;
declare variable K_ID D_ID;
declare variable K_AMOUNT_APPROVED D_CURRENCY;
declare variable CLI_DOLYA D_DOUBLE;
declare variable C_ID_TMP D_ID;
declare variable CHET_KOL_END D_SMALLINT;
declare variable DD_CLOSE_TMP D_DATE;
declare variable K_PERIOD_DAY D_INTEGER;
declare variable OKS_BAL_PERC D_CURRENCY;
declare variable OKS_ECL_VBS D_CURRENCY;
declare variable OKS_DELINQ_DAYS D_INTEGER;
declare variable OKS_BAL_ZAD_OS D_CURRENCY;
declare variable OKS_BAL_ZAD_PERC D_CURRENCY;
declare variable DEL_OS D_CURRENCY;
declare variable DEL_PERC D_CURRENCY;
declare variable DELINQ_DAYS D_INTEGER;
declare variable C_ID D_ID;
declare variable KOL_CLI D_INTEGER;
begin
  select rd_id from refdata
  where rd_code = 'Loan_Target.Consumer'
  into rd_id;

  od_11 = 0;
  priod = extract(year from rep_date_end) * 100 + extract(month from rep_date_end);
  perc_calced_11 = 0;
  kol_kred_11 = 0;
  kol_cli_11 = 0;
  llc_11 = 0;
  od_112 = 0;
  perc_calced_112 = 0;
  kol_kred_112 = 0;
  kol_cli_112 = 0;
  llc_112 = 0;
  od_12 = 0;
  perc_calced_12 = 0;
  llc_12 = 0;
  kol_kred_12 = 0;
  kol_cli_12 = 0;

  od_1222221 = 0;
  od_1222222 = 0;
  od_1222223 = 0;
  od_1222224 = 0;
  perc_calced_1222221 = 0;
  perc_calced_1222222 = 0;
  perc_calced_1222223 = 0;
  perc_calced_1222224 = 0;
  llc_1222221 = 0;
  llc_1222222 = 0;
  llc_1222223 = 0;
  llc_1222224 = 0;
  kol_kred_1222221 = 0;
  kol_kred_1222222 = 0;
  kol_kred_1222223 = 0;
  kol_kred_1222224 = 0;
  kol_cli_1222221 = 0;
  kol_cli_1222222 = 0;
  kol_cli_1222223 = 0;
  kol_cli_1222224 = 0;
  del_os = 0;
  del_perc = 0;
  kol_cli = 0;
  /*
  for
  select oks_from_k_id,
         oks_delinq_date,
         oks_balance
  from oksummary, region_list(:rep_region_id,1,3,1)
  where :rep_date_end between oks_date_begin and oks_date_end
  and oks_close_state = 0
  and oks_region_id = id_list
  into k_id,
       oks_delinq_date,
       oks_balance
  */
  for
  select oks_from_k_id,
         oks_balance,
         oks_bal_perc,
         oks_ecl_vbs,
         oks_delinq_days,
         oks_bal_zad_os,
         oks_bal_zad_perc
  from aarep_kn_oksummary_rep(:rep_date_begin, :rep_date_end)
  where oks_balance > 0
  into k_id,
       oks_balance,
       oks_bal_perc,
       oks_ecl_vbs,
       oks_delinq_days,
       oks_bal_zad_os,
       oks_bal_zad_perc

  do
  begin
    oks_balance = oks_balance / 1000;
    oks_bal_perc = oks_bal_perc / 1000;
    oks_ecl_vbs = oks_ecl_vbs / 1000;
    oks_bal_zad_os = oks_bal_zad_os / 1000;
    oks_bal_zad_perc = oks_bal_zad_perc / 1000;

    select k_kolich_client_approved, k_disb_date, k_amount_approved, k_kredline_end - k_disb_date
    from kontract
    where k_id = :k_id
    into k_kolich_client_approved, k_disb_date, k_amount_approved, k_period_day;

    kcp_perc_calced = null;
    select kcp_perc_calced from kontcalcperc
    where kcp_period = :priod
    and kcp_from_k_id = :k_id
    into kcp_perc_calced;
    kcp_perc_calced = coalesce(kcp_perc_calced,0);

    lk_percent = null;
    select lk_percent from llc_kont
    where lk_from_k_id = :k_id
    and :rep_date_end between lk_date and lk_date_end
    into lk_percent;
    lk_percent = coalesce(lk_percent,0);

    select res_just_client_count
    from hp_get_just_client_in_loan(:rep_date_end, :k_id, :k_disb_date)
    into res_just_client_count;

    kol_cli = kol_cli + res_just_client_count;

    if (oks_delinq_days < 1) then -- не просроченные займы
    begin
      od_11 = od_11 + oks_balance;

      perc_calced_11 = perc_calced_11 + oks_bal_perc;

      kol_kred_11 = kol_kred_11 + k_kolich_client_approved;

--      kol_cli_11 = kol_cli_11 + res_just_client_count;

      llc_11 = llc_11 + oks_ecl_vbs;

    end else  -- просроченные займы
    begin
      od_12 = od_12 + oks_balance;

      perc_calced_12 = perc_calced_12 + oks_bal_perc;

      kol_kred_12 = kol_kred_12 + k_kolich_client_approved;

--      kol_cli_12 = kol_cli_12 + res_just_client_count;

      llc_12 = llc_12 + oks_ecl_vbs;

      del_os = del_os + oks_bal_zad_os;
      del_perc = del_perc + oks_bal_zad_perc;

    end 

--    od_112 = 0;
--    perc_calced_112 = 0;
--    llc_112 = 0;
    kol_kred_112 = 0;
    kol_cli_112 = 0;

    for
    select iif(:k_kolich_client_approved > 1, cast(kc_amount_approved as double precision) / :k_amount_approved, 1),
           kc_from_c_id
    from knclient
    where kc_loan_target = :rd_id
    and kc_amount_approved > 0
    and kc_from_k_id = :k_id
    into cli_dolya, c_id_tmp
    do
    begin
      od_112 = od_112 + cli_dolya * oks_balance;

      perc_calced_112 = perc_calced_112 + cli_dolya * oks_bal_perc;

      llc_112 = llc_112 + cli_dolya * oks_ecl_vbs;

      kol_kred_112 = kol_kred_112 + 1;
      /*
      chet_kol_end = 1;
      for select kc_date_close_os
          from knclient
          where ((kc_date_disb between '01.01.1900' and (:k_disb_date - 1)) or
                (kc_date_disb = :k_disb_date and
                kc_from_k_id < :k_id)) and
                kc_date_close_os between :k_disb_date and '31.12.9999' and
                kc_from_c_id = :c_id_tmp
          into :dd_close_tmp
      do
      begin
        if (dd_close_tmp >= :rep_date_end) then
        begin
          chet_kol_end = 0;
        end
      end
      kol_cli_112 = kol_cli_112 + chet_kol_end; */

    end

    if (oks_delinq_days >= 1) then -- просроченные займы
    begin
      if (oks_delinq_days <= 30) then od_1222221 = od_1222221 + oks_balance;
        else if (oks_delinq_days <= 60) then od_1222222 = od_1222222 + oks_balance;
          else if (oks_delinq_days <= 90) then od_1222223 = od_1222223 + oks_balance;
            else od_1222224 = od_1222224 + oks_balance;
    
      if (oks_delinq_days <= 30) then perc_calced_1222221 = perc_calced_1222221 + oks_bal_perc;
        else if (oks_delinq_days <= 60) then perc_calced_1222222 = perc_calced_1222222 + oks_bal_perc;
          else if (oks_delinq_days <= 90) then perc_calced_1222223 = perc_calced_1222223 + oks_bal_perc;
            else perc_calced_1222224 = perc_calced_1222224 + oks_bal_perc;
    
    
      if (oks_delinq_days <= 30) then llc_1222221 = llc_1222221 + oks_ecl_vbs;
        else if (oks_delinq_days <= 60) then llc_1222222 = llc_1222222 + oks_ecl_vbs;
          else if (oks_delinq_days <= 90) then llc_1222223 = llc_1222223 + oks_ecl_vbs;
            else llc_1222224 = llc_1222224 + oks_ecl_vbs;
    
      if (oks_delinq_days <= 30) then kol_kred_1222221 = kol_kred_1222221 + kol_kred_112;
        else if (oks_delinq_days <= 60) then kol_kred_1222222 = kol_kred_1222222 + kol_kred_112;
          else if (oks_delinq_days <= 90) then kol_kred_1222223 = kol_kred_1222223 + kol_kred_112;
            else kol_kred_1222224 = kol_kred_1222224 + kol_kred_112;
    
--      if (oks_delinq_days <= 30) then kol_cli_1222221 = kol_cli_1222221 + kol_cli_112;
--        else if (oks_delinq_days <= 60) then kol_cli_1222222 = kol_cli_1222222 + kol_cli_112;
--          else if (oks_delinq_days <= 90) then kol_cli_1222223 = kol_cli_1222223 + kol_cli_112;
--            else kol_cli_1222224 = kol_cli_1222224 + kol_cli_112;
    end

  end

  for
  select kc_from_c_id, max(:rep_date_end - oks_delinq_date)
  from oksummary, kontract, knclient
  where :rep_date_end between oks_date_begin and oks_date_end
  and :rep_date_end - oks_delinq_date > 0
  and oks_from_k_id = k_id
  and k_id = kc_from_k_id
  and kc_amount_approved > 0
  group by 1
  into c_id, delinq_days
  do
  begin
    kol_cli_112 = 1;
    if (delinq_days <= 30) then kol_cli_1222221 = kol_cli_1222221 + kol_cli_112;
        else if (delinq_days <= 60) then kol_cli_1222222 = kol_cli_1222222 + kol_cli_112;
          else if (delinq_days <= 90) then kol_cli_1222223 = kol_cli_1222223 + kol_cli_112;
            else kol_cli_1222224 = kol_cli_1222224 + kol_cli_112;
    kol_cli_12 = kol_cli_12 + kol_cli_112;
  end 
  kol_cli_11 = kol_cli - kol_cli_12;


--  od_112222 = od_112 - del_os;
  od_112222 = od_112 - od_12;

--  perc_calced_112222 = perc_calced_112 - del_perc;
  perc_calced_112222 = perc_calced_112 - perc_calced_12;

  llc_112222 = llc_112 - llc_12;
--  kol_kred_112222 = kol_kred_112;
--  kol_cli_112222 = kol_cli_112;

--  od_112222 = od_112;
--  perc_calced_112222 = perc_calced_11;
--  llc_112222 = llc_11;
  kol_kred_112222 = kol_kred_11;
  kol_cli_112222 = kol_cli_11;

  od_122 = od_112;
  perc_calced_122 = perc_calced_112;
  llc_122 = llc_112;
  kol_kred_122 = kol_kred_112;
  kol_cli_122 = kol_cli_112;

  od_1222 = od_112;
  perc_calced_1222 = perc_calced_112;
  llc_1222 = llc_112;
  kol_kred_1222 = kol_kred_112;
  kol_cli_1222 = kol_cli_112;

  od_122222 = od_112;
  perc_calced_122222 = perc_calced_112;
  llc_122222 = llc_112;
  kol_kred_122222 = kol_kred_112;
  kol_cli_122222 = kol_cli_112;

  od_4 = od_11 + od_12;
  perc_calced_4 = perc_calced_11 + perc_calced_12;
  llc_4 = llc_11 + llc_12;
  kol_kred_4 = kol_kred_11 + kol_kred_12;
  kol_cli_4 = kol_cli_11 +kol_cli_12;

  period_str = 'Отчетный период: по состоянию на '||f_datetostr(:rep_date_end)||' года';
  suspend;
end
