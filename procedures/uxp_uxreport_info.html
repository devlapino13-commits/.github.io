CREATE PROCEDURE UXP$UXREPORT_INFO(
    UFM_UUID D_UUID)
RETURNS (
    URT_UUID TYPE OF D_UUID,
    URT_KIND TYPE OF D_SMALLINT,
    URT_FASTREP TYPE OF D_BLOB_BINARY,
    URT_EXCEL TYPE OF D_BLOB_BINARY,
    URT_IMPORT_KIND D_INTEGER,
    URT_IMPORT_ID D_INTEGER,
    URT_KIND_INFO D_STR_255)
AS
begin
  urt_excel = null;

  for select urt_uuid,
             urt_kind,
             urt_fastrep,
             urt_import_kind,
             urt_import_id
      from ux$report
      where urt_uuid = :ufm_uuid
      into :urt_uuid,
           :urt_kind,
           :urt_fastrep,
           :urt_import_kind,
           :urt_import_id
  do
  begin
    urt_kind_info = '';

    if (urt_import_kind = 0) then
    begin
      if (urt_kind = 0) then
        urt_kind_info = 'Настраиваемые шаблоны';
      else
      if (urt_kind = 1) then
        urt_kind_info = 'Таблица с Excel шаблонами';
      else
      if (urt_kind = 2) then
      begin
        if (urt_fastrep is null) then
          urt_kind_info = 'Необходим FastReport шаблон';
        else
          urt_kind_info = 'FastReport шаблон прикреплен';
      end else
      if (urt_kind = 3) then
      begin
        if (urt_fastrep is null) then
          urt_kind_info = 'Необходим FastReport шаблон';
        else
          urt_kind_info = 'FastReport шаблон прикреплен';
      end
    end else
    if (urt_import_kind > 0) then
    begin
      if (coalesce(urt_import_id, 0) = 0) then
        urt_kind_info = 'Не импортирован';
      else
        urt_kind_info = 'Импортирован';
    end

    suspend;
  end
end
