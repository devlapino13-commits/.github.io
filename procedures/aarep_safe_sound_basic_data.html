CREATE PROCEDURE AAREP_SAFE_SOUND_BASIC_DATA(
    OBJECT_ID D_ID,
    OBJ_KIND D_STR_10)
RETURNS (
    C_ID D_ID,
    CLIENT_DATA BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable c_name_fio d_str_100;
declare variable ma_id d_integer;
declare variable bls_id d_id;
declare variable eil_date d_date;
declare variable json d_blob_text;
declare variable base_path d_str_100;
declare variable ma_date_app d_date;
declare variable outstanding_other d_currency; /* активный баланс в других фку */
declare variable amount_other d_integer; /* кол-во активных займов в других фку */
declare variable max_past_due d_integer;
declare variable max_past_due_str d_str_100;
declare variable max_past_due_days d_integer;
declare variable outstanding_not_our d_currency;
declare variable amount_not_our d_currency;
declare variable guarantor d_integer;
declare variable guarantor_other d_integer;
declare variable pr_code d_str_100;
declare variable base_k_id d_id;
declare variable base_ma_id d_integer;
declare variable insur_ma_id d_integer;
begin
  obj_kind = trim(coalesce(obj_kind, ''));

  if ((obj_kind is null) or (object_id is null)) then
    exit;
  client_data = '';

  ma_id = 0;
  if (obj_kind in ('kontract', '9')) then
    select k_from_ma_app_id
    from kontract
    where k_id = :object_id
    into ma_id;
  else
  if (obj_kind in ('miniapp', '11')) then
    ma_id = object_id;

  select pr.pr_code
  from miniapp ma
  inner join product pr on pr.pr_id = ma.ma_from_pr_id
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.2999'
  into pr_code;

  if (pr_code containing '.insur_kontrakt.') then  -- RFCKG-5051
  begin
    select lir.lir_kontract_base
    from kontract k
    inner join loan_insurance_relation lir on k.k_id = lir.lir_kontract_insurance
    where k.k_from_ma_app_id = :ma_id
    into base_k_id;

    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :base_k_id
    into base_ma_id;

    if (coalesce(base_ma_id, 0) <> 0) then
    begin
      insur_ma_id = ma_id;
      ma_id = base_ma_id;
    end
  end

  for select c.c_id, c.c_name_fio
  from clients c
  inner join miniappclient mac on c.c_id = mac.mac_from_c_id
  inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
  where ma.ma_id = :ma_id
  and ma.ma_date_end = '31.12.9999'
  into :c_id, :c_name_fio
  do
  begin
    client_data = blob_append(client_data, '<b>' || c_name_fio || '</b><br>');

    select ma.ma_date_app
    from miniapp ma
    where ma.ma_id = :ma_id
    order by ma.ma_date_app
    rows 1
    into ma_date_app;

    select eil.eil_from_bls_id, cast(eil.eil_status_datetime as date)
    from extinfolist eil
    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id and
          eic.eic_code = 'ss_cib_report'
    where eil.eil_from_c_id = :c_id and
          eil.eil_is_json = 1 and
          eil.eil_status = 2 and
          ((eil.eil_date between :ma_date_app - 1 and :ma_date_app)
          or
          (eil.eil_object_kind = f_objkind('miniapp', 'id') and eil.eil_object_id = :ma_id))  -- RFCKG-5396
    order by eil.eil_id desc
    rows 1
    into bls_id, eil_date;

    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into :json;
  
    if (row_count is null or coalesce(:json, '') = '') then
    begin
        client_data = BLOB_APPEND(client_data, '&nbsp&nbspКлиент еще не проверен<br><br>');
        suspend;
        exit;
    end
  
    if (f_json_blob_get_value(:json, 'result', '') = 'NO SUBJECT FOUND') then
    begin
        client_data = BLOB_APPEND(client_data, '&nbsp&nbspКлиент не найден<br><br>');
        suspend;
        exit;
    end

    client_data = blob_append(client_data, '&nbsp&nbspДата проверки: <b>' || :eil_date || '</b><br>');
    base_path = 'other_element.monitoring_enquiry.credit_report.contracts_history';

    max_past_due_days = 0;
    outstanding_other = 0;
    amount_other = 0;
    guarantor_other = 0;
    for select max_past_due, total_outstanding_not_our, total_amount_not_our, guarantor
    from get_fm_data_safe_sound('installments', :bls_id, :base_path)
    into :max_past_due, outstanding_not_our, amount_not_our, guarantor
    do
    begin
      if (max_past_due > max_past_due_days) then
          max_past_due_days = max_past_due;
      outstanding_other = outstanding_not_our;
      amount_other = amount_not_our;
      guarantor_other = guarantor_other + guarantor;
    end

    max_past_due_str = case
                        when max_past_due_days = 0 then 'Слишком новый'
                        when max_past_due_days = 1 then 'Плата по согласованию'
                        when max_past_due_days = 2 then 'Задержка 1-29 дней'
                        when max_past_due_days = 3 then 'Задержка 30-59 дней'
                        when max_past_due_days = 4 then 'Задержка 60-89 дней'
                        when max_past_due_days = 5 then 'Задержка 90-119 дней'
                        when max_past_due_days = 6 then 'Задержка 120-149 дней'
                        when max_past_due_days = 7 then 'Задержка 150-179 дней'
                        when max_past_due_days = 8 then 'Более 180 дней задержки'
                        else '--'
                       end;
    if (max_past_due_days = 0 or max_past_due_days = 1) then
      client_data = blob_append(client_data, '&nbsp&nbsp<b>В других ФКУ нарушений не обнаружено</b><br>');

    client_data = BLOB_APPEND(client_data, '&nbsp&nbspМакс. дни просрочки в других ФКУ<br>' || '&nbsp&nbsp&nbsp&nbsp<b>');
    if (max_past_due_days = 0 or max_past_due_days = 1) then
      client_data = BLOB_APPEND(client_data, '0 дней</b><br>');
    else
    begin
      client_data = BLOB_APPEND(client_data, max_past_due_str || '</b><br>');
    end
    
    client_data = BLOB_APPEND(client_data, '&nbsp&nbspАктивный баланс в других ФКУ<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || cast(round(outstanding_other) as d_str_10) || ' сом</b><br>');
    client_data = BLOB_APPEND(client_data, '&nbsp&nbspАктивные займы в других ФКУ<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || cast(amount_other as d_str_10) || ' кредит(ов)</b><br>');
    client_data = BLOB_APPEND(client_data, '&nbsp&nbspАктивные в других ФКУ, где он(а) поручитель<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || cast(guarantor_other as d_str_10) || ' кредит(ов)</b><br>');
      
    if (ma_id <> 0) then
        client_data = BLOB_APPEND(client_data, f_uri_html('Просмотр отчета', 'ssound_report',
          'Отчет КИБ', 'miniapp', iif(coalesce(:insur_ma_id, 0) <> 0, :insur_ma_id, :ma_id), current_date, current_date,
          '{"c_id":' || :c_id || '}') || '<br>');
  end
  suspend;
end
