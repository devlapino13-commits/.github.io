Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE MC_FIXED_MSG(
    MCL_ID D_ID)
RETURNS (
    SUBJECT D_STR_255,
    BODY D_STR_2048,
    BG_COLOR D_STR_CODE,
    TEXT_COLOR D_STR_CODE)
AS
declare variable mcl_login d_str_100;
begin
  select mcl.mcl_login
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :mcl_login;

  select fm.fm_subject, fm.fm_body, f_enum_code('fixed_msg_colores',fm.fm_bg_colore,null), f_enum_code('fixed_msg_colores',fm.fm_text_colore,null)
  from fixed_msg fm
  where current_date between fm.fm_date_begin and fm.fm_date_end
  into :subject, :body, :bg_color, :text_color;

  if (subject is null or body is null or bg_color is null or text_color is null) then
    exit;

  suspend;
end
