CREATE PROCEDURE CARD_COMMAND_SET_CARD_NAME(
    CS_ID D_INTEGER)
AS
declare variable cca_id d_integer;
declare variable cs_object_id d_id;
declare variable cs_object_kind d_integer;
declare variable card_id d_integer;
declare variable cs_request_json d_str_8100;
declare variable office_id d_integer;
declare variable un_id d_id;
declare variable c_id d_id;
declare variable c_name_eng d_str_100;
declare variable k_id d_id;
begin
    select cs_from_cca_id,
         cs_object_id,
         cs_object_kind,
         cs_card_id,
         cs_from_k_id,
         cs_from_c_id
    from cardsync cs
    where cs.cs_id = :cs_id
    into :cca_id,
         :cs_object_id,
         :cs_object_kind,
         :card_id,
         :k_id,
         :c_id;

    -- если картсчет подвязан
    select cac.cac_from_un_id,
           cac.cac_from_reg_id
    from cardaktclient cac
    where cac_id = :cs_object_id
    into :un_id,
         :office_id;

    if (c_id is null) then
    begin
        select cac_from_c_id
        from cardaktclient
        where cac_id = :cs_object_id
        into :c_id;
    end

    -- если карта активирована
    -- новая задача по редактированию имени на карте
    select res_text
    from clients
    inner join ipc_fio_for_card(c_name_eng) on 1 = 1
    where c_id = :c_id
    into :c_name_eng;

    cs_request_json = f_format('{"card_id":%1, "c_name_eng":"%2"}', :card_id, :c_name_eng);
    insert into cardsync(
      cs_parent_id,
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_request_json,
      cs_office_id,
      cs_from_un_id,
      cs_from_k_id,
      cs_from_c_id
      )
    values(
      :cs_id,
      f_get_cardsynccfg('set_card_name', 'id'),
      :card_id,
      :cca_id,
      :cs_object_kind,
      :cs_object_id,
      :cs_request_json,
      :office_id,
      :un_id,
      :k_id,
      :c_id
      );
end
