CREATE PROCEDURE STRING_CHAR_TO_DECIMAL(
    STRING_TXT_IN D_STR_100)
RETURNS (
    STRING_TXT_ORIG D_STR_100,
    STRING_TXT_VARCHAR D_STR_100,
    STRING_TXT_OUT D_PERCENT,
    KOL_TOCH D_INTEGER,
    CHAR_QTY D_INTEGER)
AS
declare variable string_txt_tmp varchar(1);
declare variable i integer;
begin
  /* Процедура используется если нужно преобразовать текстовую строку в числовые данные с фиксированной точностью */
  if (string_txt_in is null) then
    exit;
  else
  begin
    char_qty = char_length(string_txt_in);

    string_txt_orig = left(string_txt_in, char_qty);

    while (coalesce(i, 0) <= char_qty) do
    begin
      i = coalesce(i, 0) + 1;
      string_txt_tmp = substring(string_txt_orig from i for 1);
      if (string_txt_tmp in (',', '.')) then
      begin
        if (coalesce(kol_toch, 0) = 0) then
          string_txt_tmp = '.';
        else
          string_txt_tmp = '';

        kol_toch = 1;
      end

      if (string_txt_tmp similar to '[-]?[[:DIGIT:]]+' or --
          (string_txt_tmp = '.' and kol_toch = 1)) then
        string_txt_varchar = coalesce(string_txt_varchar, '') || string_txt_tmp;
    end

    string_txt_varchar = left(iif(string_txt_varchar = '.', 0, string_txt_varchar), 11);
    string_txt_out = cast(string_txt_varchar as d_percent);
    suspend;
  end
end
