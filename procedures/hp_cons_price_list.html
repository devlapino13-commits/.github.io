CREATE PROCEDURE HP_CONS_PRICE_LIST
RETURNS (
    CP_ID D_ID,
    CP_FROM_PR_ID D_ID,
    CP_FROM_PRL_ID D_ID,
    PR_NAME D_STR_255,
    CP_MIN_POINT D_SMALLINT,
    CP_MAX_POINT D_SMALLINT,
    CP_MIN_AMOUNT D_CURRENCY,
    CP_MAX_AMOUNT D_CURRENCY,
    CP_PRICE D_CURRENCY,
    CP_LEVEL D_INTEGER,
    CP_LEVEL_STR D_STR_64,
    CP_PERIOD D_SMALLINT,
    CP_IS_ACTIVE D_STR_10,
    CP_CREATE_DATETIME D_DATETIME,
    CP_DATETIME D_DATETIME,
    CP_FROM_UN_ID D_ID)
AS
begin
    for select cp.cp_id, cp.cp_from_pr_id, cp.cp_from_prl_id, cp.cp_min_point, cp.cp_max_point,cp.cp_min_amount, cp.cp_max_amount, cp.cp_price,
               cp.cp_level, iif(cp.cp_level in (0, 1, 2, 3, 4, 5), f_enum('consul_lvl', cp.cp_level, 'name'), 'Не указан либо некорректно указан уровень консультации'), cp.cp_period, cp.cp_is_active, cp.cp_create_datetime, cp.cp_datetime, cp.cp_from_un_id, pr.pr_name
    from consul_price cp
    join product pr on pr.pr_id = cp.cp_from_pr_id
    into :cp_id, :cp_from_pr_id, :cp_from_prl_id, :cp_min_point, :cp_max_point, :cp_min_amount, :cp_max_amount, :cp_price,
         :cp_level, :cp_level_str, :cp_period, :cp_is_active, :cp_create_datetime,  :cp_datetime, :cp_from_un_id, :pr_name
    do
        suspend;
end
