CREATE PROCEDURE FM_DEMO_SCHED_ACTION(
    UN_ID D_ID,
    ACTION_KIND D_STR_CODE,
    ACTION_DATA D_STR_8100)
RETURNS (
    RESULT_ID D_INTEGER,
    RESULT_TXT D_BLOB_TEXT)
AS
declare variable LONG_TXT D_STR_8100;
declare variable LOAN_AMOUNT D_CURRENCY = 0;
declare variable LOAN_PERIOD D_INTEGER;
declare variable LOAN_PRODUCT D_INTEGER;
declare variable LOAN_PRODUCT_ADD D_INTEGER;
declare variable LOAN_PERCENT D_PERCENT;
declare variable LOAN_DISB_DATE D_DATE;
declare variable LOAN_REPAY_DAY D_INTEGER;
declare variable LOAN_STATUS D_INTEGER;
declare variable LOAN_GRACE_PERIODS D_STR_255;
declare variable DB_ID D_INTEGER;
declare variable JSON_KEY D_STR_100;
declare variable JSON_VALUE D_STR_255;
declare variable LOAN_PERIOD_MIN D_INTEGER;
declare variable LOAN_PERIOD_MAX D_INTEGER;
declare variable LOAN_AMOUNT_MIN D_INTEGER;
declare variable LOAN_AMOUNT_MAX D_INTEGER;
declare variable LOAN_ADD_MONTH D_INTEGER;
declare variable I D_SMALLINT;
declare variable LANG D_STR_LANG;
declare variable TMP_BLOB D_BLOB_TEXT;
declare variable LOAN_TOTAL_PERC_AMOUNT D_INTEGER;
declare variable LOAN_CLI_ID D_ID;
begin
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);

  result_id = 0;
  result_txt = f_lang(lang, 'Невыполнимая команда');

  if (action_kind = 'do_load') then
  begin
    result_id = 1;
    loan_amount = iif(db_id <> 94401, 20000, 100);
    loan_period = 6;
    loan_period_min = 1;
    loan_period_max = 12;
    loan_percent = 0;
    loan_repay_day = 0;

    select min(prl_repay_min), max(prl_repay_max), min(prl_amount_cl_min), max(prl_amount_cl_max)
    from product, prodlogic
    where pr_date_begin <= current_date and
          coalesce(pr_date_end, '31.12.2999') >= current_date and
          pr_active = 1 and
          prl_from_pr_id = pr_id
    into loan_period_min, loan_period_max, loan_amount_min, loan_amount_max;

    if (loan_amount < loan_amount_min) then
      loan_amount = loan_amount_min;
    if (loan_amount > loan_amount_max) then
      loan_amount = loan_amount_max;

    result_txt =
      '"amount":' || loan_amount ||
      ',"min_amount": ' || loan_amount_min ||
      ',"max_amount": ' || loan_amount_max ||
      ',"period":' || loan_period ||
      ',"min_period": ' || loan_period_min ||
      ',"max_period": ' || loan_period_max ||
      ',"disb_date":"' || f_datetostr(current_date) || '"' ||
      ',"repay_day": ' || loan_repay_day || ',' ||
      iif(db_id in (39801, 41701, 94401), '"percent": ' || loan_percent ||
      ', "min_percent": 0.0, "max_percent": 100.0,', '');

    if (db_id = 81001) then
    begin
      -- Массив статусов, только для РФ
      long_txt = '';
      i = 0;
      while (i <= 81) do
      begin
        long_txt = long_txt || iif(long_txt = '', '', ',') || f_format('{"id":"%1","name":"%2"}', i, trim(iif(i = 81, i || ' и более', i)));
        i = i + 1;
      end
      result_txt = BLOB_APPEND(result_txt, '"status":0, "statuses":[' || long_txt || '],');
    end

    -- Массив продуктов без параллельных
    select res_json
    from hp_json_from_sql('array', 'products', '
    select pr_id, pr_memo as pr_name
    from product
    where pr_date_begin <= current_date
    and coalesce(pr_date_end, ''31.12.2999'') >= current_date
    and pr_active = 1
    and pr_code not containing ''.loyal.''
    and pr_code not containing ''.emp.''
     ')
    into long_txt;

    result_txt = BLOB_APPEND(result_txt, long_txt);

    -- Массив только параллельных
    if (db_id = 81001) then
    begin
      long_txt = '{"pr_id": -1,"pr_name":"Нет"}';
      for select pr_id, pr_memo as pr_name
          from product
          where pr_date_begin <= current_date and
                coalesce(pr_date_end, '31.12.2999') >= current_date and
                pr_active = 1 and
                pr_code containing '.loyal.' and
                pr_code not containing '.emp.'
      as cursor cur_prod
      do
      begin
        long_txt = long_txt || ',{"pr_id":' || cur_prod.pr_id || ', "pr_name":"' || cur_prod.pr_name || '"}';
      end
      long_txt = ', "products_add":[' || long_txt || ']';
      result_txt = BLOB_APPEND(result_txt, long_txt);
    end

    -- Отсрочки COVID
    /* Отключены 11.11.2021, т.к. больше года не предоставляются
       Если включить, то в МП появится этот контрол для выбора
    long_txt = '[{"id":"0","name":"' || f_lang(lang, 'Стандартно') || '"}]';
    if ((db_id in (39801, 41701) and current_date >= '26.03.2020') or
        (db_id = 81001 and current_date >= '28.03.2020')) then
      long_txt =
        '[{"id":"0","name":"' || f_lang(lang, 'Стандартно') || '"},'||
        '{"id":"1","name":"' || f_lang(lang, 'Добавить 1 месяц') || '"},'||
        '{"id":"2","name":"' || f_lang(lang, 'Добавить 2 месяца') || '"}]';
    result_txt = BLOB_APPEND(result_txt, ', "add_month_list":' || long_txt );
    */

    result_txt = '{' || result_txt || '}';
  end
  else
  if (action_kind = 'do_schedule') then
  begin
    result_id = 1;
    result_txt = '';
    loan_add_month = 0;

    for select json_key, json_value
        from fm_json_parse(:action_data)
        into :json_key, :json_value
    do
    begin
      if (json_key = 'amount') then
        loan_amount = json_value;
      if (json_key = 'period') then
        loan_period = json_value;
      if (json_key = 'disb_date') then
        loan_disb_date = json_value;
      if (json_key = 'repay_day') then
        loan_repay_day = json_value;
      if (db_id in (39801, 41701, 94401) and
          json_key = 'percent') then
      begin
        loan_percent = json_value;
      when any do
        begin
          result_id = 0;
          result_txt = f_lang(lang, 'Процентная ставка не может быть нулевой');
          suspend;
          exit;
        end
      end
      if (json_key = 'status') then
        loan_status = json_value;
      if (json_key = 'product') then
        loan_product = json_value;
      if (json_key = 'product_add') then
        loan_product_add = json_value;
      if (json_key = 'grace_periods') then
        loan_grace_periods = json_value;
      if (json_key = 'add_month') then
        loan_add_month = json_value;
      if (json_key = 'client_id') then
        loan_cli_id = json_value;
    end

    if (db_id = 81001) then
    begin
      select perc_default
      from fm_get_percent(:loan_product, :loan_status, :loan_disb_date, :loan_amount, :loan_period, :loan_product_add)
      into loan_percent;
      if (loan_percent is null) then
      begin
        result_id = 0;
        result_txt = f_lang(lang, 'Введенные параметры не соответствуют условиям выбранного продукта');
        suspend;
        exit;
      end
    end
    else
    begin
      if (coalesce(loan_percent, 0) = 0) then
      begin
        result_id = 0;
        result_txt = f_lang(lang, 'Процентная ставка не может быть нулевой');
        suspend;
        exit;
      end
    end

    -- Базовый график
    execute procedure hp_make_schedule_date(loan_amount, loan_period, loan_repay_day, loan_disb_date, loan_product,
        0, 0, loan_grace_periods, 1, loan_add_month);

    execute procedure hp_make_schedule_perc(loan_amount, loan_period, loan_product, loan_percent, 0, 0, loan_add_month, -1, 1);

    result_id = 1;
    result_txt = '';
    select res_json || '}'
    from hp_json_from_sql('array', 'schedule', '
    select sh_id as id,
           sh_date,
           sh_os as os,
           sh_perc as perc,
           sh_grace as grace,
           0 as tax,
           sh_os + sh_perc as sh_all,
           sh_os + sh_perc as sh_all_tax
    from temp_schedule
    where sh_id <> 0')
    into tmp_blob;

    select sum(sh_perc)
    from temp_schedule
    where sh_id <> 0
    into loan_total_perc_amount;
    loan_total_perc_amount = coalesce(loan_total_perc_amount, 0);

    result_txt = 
      '{"title":"' || f_lang(lang, 'Базовый') || '",' ||
      '"detail": "' || f_lang(lang, 'График платежей с <b>базовой</b> процентной ставкой') || '",' ||
      '"summary": "' || f_format(f_lang(lang, 'Итого сумма %% по займу: <b>%1</b>'), loan_total_perc_amount) || '",' ||
      tmp_blob;

    -- Если продукт с плавающей ставкой, добавляем график с пониженной ставкой
    if (db_id = 81001 and exists(
      select * from product
      where pr_id = :loan_product
      and pr_percent_kind = 1)) then
    begin
      -- RFCRU-5436
      -- Т.к. сотрудников нет, строим сниженный
      execute procedure hp_make_schedule_perc(loan_amount, loan_period, loan_product, loan_percent, 0, 0, loan_add_month, :loan_cli_id, 0);
      select res_json || '}'
      from hp_json_from_sql('array', 'schedule', '
      select sh_id as id,
             sh_date,
             sh_os as os,
             sh_perc as perc,
             sh_grace as grace,
             0 as tax,
             sh_os + sh_perc as sh_all,
             sh_os + sh_perc as sh_all_tax
      from temp_schedule
      where sh_id <> 0')
      into tmp_blob;
  
      select sum(sh_perc)
      from temp_schedule
      where sh_id <> 0
      into loan_total_perc_amount;
      loan_total_perc_amount = coalesce(loan_total_perc_amount, 0);

      result_txt = BLOB_APPEND(result_txt,
        ',{"title":"' || f_lang(lang, 'Пониженный') || '",' ||
        '"detail": "' || f_lang(lang, 'График платежей с <b>пониженной</b> процентной ставкой') || '",' ||
        '"summary": "' || f_format(f_lang(lang, 'Итого сумма %% по займу: <b>%1</b>'), loan_total_perc_amount) || '",' ||
        tmp_blob);
   end
   -- Упаковка всех графиков в массив
   result_txt = BLOB_APPEND('{"tab_list": [', result_txt, ']}');
  end

  suspend;

  delete from temp_schedule
  where 1 = 1;
end
