CREATE PROCEDURE HP_GET_FPS_APPLICATION_STATUS(
    MAC_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    RES_LINK D_STR_LONG)
AS
declare variable c_id d_id;
declare variable cell_no d_str_100;
declare variable clt_id d_integer;
declare variable fpb_id d_id;
declare variable fpb_name d_str_255;
declare variable fcb_status d_integer;
declare variable fcb_status_str d_str_255;
declare variable lang d_str_lang;
begin
    res_text = '';
    res_link = '';
    fcb_status_str = '';

    if (f_tmp_const('moneta_fps_active') <> 1) then
    begin
        suspend;
        exit;
    end

    select mac.mac_from_c_id
    from miniappclient mac
    inner join miniapp ma on mac.mac_from_ma_id = ma.ma_id
        and ma.ma_disb_kind = 7
        and ma.ma_is_active = 1
        and ma.ma_date_end >= '31.12.2999'
    where mac.mac_id = :mac_id
    into :c_id;


    if (c_id is null) then
    begin
        suspend;
        exit;
    end

    lang = f_get_lang_code(un_id, null);
    cell_no = f_cont(c_id, 'cell', 'ok', 'main');
    if (cell_no = '') then
    begin
        res_text = 'СБП не доступен: Основной номер отсутствует.';
        suspend;
        exit;
    end

    clt_id = f_cont_id(cell_no, 'add');
    cell_no = f_cont_format(cell_no, 'cell', 'secured');

    select fb.fpb_id, fb.fpb_name, fcb.fcb_status
    from fps_application_banks fab
    inner join fps_client_banks fcb on fab.fab_from_fcb_id = fcb.fcb_id
        and fcb.fcb_from_c_id = :c_id
        and fcb.fcb_from_clt_id = :clt_id
    inner join fps_banks fb on fcb.fcb_from_fpb_id = fb.fpb_id
    where fab.fab_from_mac_id = :mac_id
    into fpb_id, fpb_name, fcb_status;


    if (fpb_id is not null) then
    begin
        if (f_get_const('fps_manual_disb') <> 1) then
        begin
            fcb_status_str =
                case fcb_status
                    when -2 then 'Отказано, другой клиент'
                    when -1 then 'Клиент не найден в этом банке'
                    when 0 then 'На проверке'
                    when 1 then 'Требуется доп. проверка от КК'
                    when 2 then 'Подтвержден'
                    else 'Неизвестный статус'
                end;

            res_text = f_format('СБП: <b>%1</b>, %2<br><b>%3</b>', cell_no, fcb_status_str, fpb_name);
        end
        else
            res_text = f_format('СБП: <b>%1</b>, <b>%2</b>', cell_no, fpb_name);

        res_link = f_uri_sql( f_lang(lang, 'Изменить банк'), 'set_fps_bank', 'miniappclient', :mac_id, 'combo', f_lang(lang, 'Выберите банк для выдачи займа'), '', 0, 0, fpb_id, '', 'dialog', '');

    end
    else
    begin
        res_text = f_format('СБП: %1<br>', cell_no);
        res_link = f_uri_sql(f_lang(lang, 'Выбрать банк'), 'set_fps_bank', 'miniappclient', :mac_id, 'combo', f_lang(lang, 'Выберите банк для выдачи займа'), '', 0, 0, '', '', 'dialog', '');

    end

    suspend;
end
