CREATE PROCEDURE HP_GET_PASSKIND(
    COUNTRY_ID INTEGER)
RETURNS (
    PK_ID INTEGER,
    PK_NAME VARCHAR(100),
    PK_MIN INTEGER,
    PK_MAX INTEGER,
    PK_CAN_SPACE INTEGER,
    PK_IS_LATIN INTEGER,
    PDT_ID INTEGER,
    PDT_CODE VARCHAR(100),
    PDT_NAME VARCHAR(100),
    PDT_MEMO VARCHAR(255),
    PDT_NEED_DOC_NO INTEGER,
    PDT_NEED_DATE_BEGIN INTEGER,
    PDT_NEED_DATE_END INTEGER,
    PDT_NEED_DISB_WHO INTEGER,
    PDT_NEED_FOR_COMPANY INTEGER,
    PDT_NEED_TERRITORY INTEGER,
    PK_ORDER INTEGER,
    PK_CAPTION_FOR_MEMO_1 VARCHAR(100),
    PK_CAPTION_FOR_MEMO_2 VARCHAR(100),
    PK_CAPTION_FOR_MEMO_3 VARCHAR(100),
    PK_CAPTION_FOR_MEMO_4 VARCHAR(100),
    PK_CAPTION_FOR_MEMO_5 VARCHAR(100),
    PK_PASSPORT_ISSUERS VARCHAR(255),
    PDT_IS_MAIN_DOC INTEGER)
AS
begin
  for select pk_id, pk_name, pk_min, pk_max, pk_can_space, pk_is_latin, pk_order, pdt_id, pdt_code, pdt_name, pdt_memo,
             pdt_need_doc_no, pdt_need_date_begin, pdt_need_date_end, pdt_need_disb_who, pdt_need_for_company,
             pdt_need_territory, pk_caption_for_memo_1, pk_caption_for_memo_2, pk_caption_for_memo_3,
             pk_caption_for_memo_4, pk_caption_for_memo_5, pk_passport_issuers, pdt_is_main_doc
      from passkind, passdoctype
      where pdt_id = pk_from_pdt_id and
            pk_id > 0 and
            (((pk_country_id = :country_id or
            :country_id = 0) and
            pdt_is_main_doc = 1) or
            (pdt_is_main_doc = 0)) and
            pk_is_delete = 0
      order by pk_order, pk_name
      into :pk_id, :pk_name, :pk_min, :pk_max, :pk_can_space, :pk_is_latin, :pk_order, :pdt_id, :pdt_code, :pdt_name,
           :pdt_memo, :pdt_need_doc_no, :pdt_need_date_begin, :pdt_need_date_end, :pdt_need_disb_who,
           :pdt_need_for_company, :pdt_need_territory, :pk_caption_for_memo_1, :pk_caption_for_memo_2,
           :pk_caption_for_memo_3, :pk_caption_for_memo_4, :pk_caption_for_memo_5, :pk_passport_issuers, :pdt_is_main_doc

  do
  begin
    suspend;
  end
end
