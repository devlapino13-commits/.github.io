CREATE PROCEDURE HP_PRE_SUD_PAYED(
    REP_K_ID ID_TYPE)
RETURNS (
    PAY_DATE D_DATE,
    PAY_SUM CURR_TYPE,
    PAY_TYPE STR_255,
    PAY_CLI D_STR_LONG)
AS
declare variable ol_id id_type;
begin
  for select ol_id, ol_date, pay_cli, pay_type, sum(pay_sum)
      from (select ol_id, ol_date,
                   c_name_fio || iif(ol_doc_client_name is not null, ' через ' || ol_doc_client_name, '') as pay_cli,
                   case ok_pay_type
                     when 1 then 'Налич.'
                     when 2 then 'Безнал'
                     when 3 then 'Внутр.'
                   end as pay_type,
                   (ok_os + ok_perc + ok_fine + ok_kommis) * -1 as pay_sum
            from operkred
            inner join operlist on ol_id = ok_from_ol_id
            left join clients on c_id = ol_document_client
            where ok_from_k_id = :rep_k_id and
                  ok_ol_is_active = 1 and
                  ok_type = 4
            union all
            select ol_id, ol_date,
                   c_name_fio || iif(ol_doc_client_name is not null, ' через ' || ol_doc_client_name, '') as pay_cli,
                   null as pay_type, ox_amount_minus as pay_sum
            from operextra
            inner join operlist on ol_id = ox_from_ol_id
            left join clients on c_id = ol_document_client
            where ox_object_kind = 9 and
                  ox_object_id = :rep_k_id and
                  ox_ol_is_active = 1 and
                  ox_amount_minus <> 0
            union all
            select ol_id, ol_date,
                   c_name_fio || iif(ol_doc_client_name is not null, ' через ' || ol_doc_client_name, '') as pay_cli,
                   null as pay_type,
                   (oko_os + oko_perc + oko_other) * -1 as pay_sum
            from operkroff
            inner join operlist on ol_id = oko_from_ol_id
            left join clients on c_id = oko_from_c_id
            where oko_from_k_id = :rep_k_id and
                  oko_type = 4 and
                  oko_ol_is_active = 1)
      group by 1, 2, 3, 4
      order by ol_date
      into ol_id, pay_date, pay_cli, pay_type, pay_sum
  do
  begin
    suspend;
  end
end
