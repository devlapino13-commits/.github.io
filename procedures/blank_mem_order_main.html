CREATE PROCEDURE BLANK_MEM_ORDER_MAIN(
    OL DECIMAL(18,0),
    POCKET_ID INTEGER,
    REG_ID INTEGER,
    MEM_DATE DATE)
RETURNS (
    BL_ID DECIMAL(18,0),
    BL_DATE DATE,
    BL_DOC_NO INTEGER,
    BL_MEMO VARCHAR(1024),
    BL_OFFICE_FULL_NAME VARCHAR(255),
    BL_POCKET_NAME VARCHAR(100),
    UN_NAME VARCHAR(30),
    UN_FIO VARCHAR(100))
AS
declare variable OFFICE varchar(255);
declare variable COMPANY_NAME varchar(255);
declare variable DT varchar(20);
declare variable KT varchar(20);
begin
  /* Procedure Text */
  if (ol <> 0) then
  begin
    for select ol_id, ol_date, ol_doc_no, ol_memo, cm_name, reg_name, mo_name, un_name, un_fio
        from company, region, operlist, mopocket, username un
        where ol_id = :ol and
              reg_unicode = ol_from_reg_id and
              ol_pocket = mo_id and
              un_id  = ol_user_id
        into :bl_id, :bl_date, :bl_doc_no, :bl_memo, :company_name, :office, :bl_pocket_name, :un_name, :un_fio
    do
    begin
      bl_office_full_name = company_name || ' ' || office;
      suspend;
    end
  end
  else
  begin
    for select distinct ol_id, ol_date, ol_doc_no, ol_memo, cm_name, reg_name, mo_name, ob_account_dt, ob_account_kt, un_name, un_fio
        from company, region, operlist, mopocket, operbook, username
        where ol_pocket = :pocket_id and
              operlist.is_delete = 0 and
              ob_from_ol_id = ol_id and
              ol_from_reg_id = :reg_id and
              mo_id = :pocket_id and
              reg_unicode = ol_from_reg_id and
              ol_date_oper_day = :mem_date and
              un_id  = ol_user_id
        into :bl_id, :bl_date, :bl_doc_no, :bl_memo, :company_name, :office, :bl_pocket_name, :dt, :kt, :un_name, :un_fio
    do
    begin
      bl_office_full_name = company_name || ' ' || office;

      if (not(exists(select *
                     from acc_list, acctypelist
                     where ((al_id = :dt) or
                           (al_id = :kt)) and
                           al_from_bg_id like atl_acc_mask || '%' and
                           atl_from_at_id = 11 /* Касса */ ))) then
      begin


        suspend;
      end
    end
  end
end
