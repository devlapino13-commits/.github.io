CREATE PROCEDURE AAREP_KIB_MINIAPP_CLI_REPORT(
    MA_FROM_REG_ID INTEGER,
    MA_ID INTEGER,
    C_ID ID_TYPE,
    FULL_HISTORY SMALLINT = 0)
RETURNS (
    BASIC_AMOUNT D_CURRENCY,
    MONTHLY_REPAY D_CURRENCY,
    ACTIVE_CNT D_SMALLINT,
    ALL_CNT D_SMALLINT,
    SHOWN_CNT D_SMALLINT,
    KF_CLIENT_ID DECIMAL(18,0),
    KF_CLIENT_NAME VARCHAR(100),
    KF_CLIENT_PASS_NO VARCHAR(30),
    KF_AMOUNT DECIMAL(18,4),
    MA_KIB_SEND INTEGER,
    MA_KIB_RECIEVE INTEGER,
    KH_ALL_BALANCE DECIMAL(18,4),
    KH_ALL_KREDIT DECIMAL(18,4),
    KH_APROX_BALANCE DECIMAL(18,4),
    KH_DATEAWARD DATE,
    KH_AMOUNT DECIMAL(18,4),
    KH_PERIOD_IN_MONTH DECIMAL(18,2),
    KH_CLIENTS_KOL INTEGER,
    KH_LIMIT DECIMAL(18,2),
    KH_DATESTOP DATE,
    KH_DATEPROLONGATION DATE,
    KH_AMONTHOFPROLONGATION DECIMAL(18,2),
    KH_UNPAIDAMOUNT DECIMAL(18,2),
    KH_UNPAIDPERCENAMOUNT DECIMAL(18,2),
    KH_COUNTOFDELINQUENCY INTEGER,
    KH_CREDITSTATUS VARCHAR(255),
    KH_CREDITTYPE VARCHAR(255),
    KH_CURRENCY VARCHAR(50),
    KH_REPAYSTATUS VARCHAR(50),
    KH_IS_OUR INTEGER,
    KH_CREATE_DATE TIMESTAMP,
    KH_CLIENTS_FIO VARCHAR(1024),
    KH_PORUCH_FIO VARCHAR(1024),
    KH_IS_ACTIVE INTEGER,
    KH_MONTHLY_REPAY DECIMAL(18,4),
    KF_IS_PORUCH INTEGER,
    UN_FIO VARCHAR(100),
    KH_ALL_ACTIVE_KOL INTEGER,
    KH_NO INTEGER,
    KF_DATE_BIRTH DATE,
    KF_INN VARCHAR(20),
    KF_IS_PORUCH_STR VARCHAR(20),
    KH_DELIQDAYSCOUNT INTEGER,
    KH_OWNAMOUNT DECIMAL(18,4),
    KH_HIST_QTY SMALLINT,
    KH_HIST_SHOWN SMALLINT,
    IS_REAL_HIST SMALLINT,
    MA_KOLICH_REPAY INTEGER,
    KBM_CHECKED_DATE DATE,
    KH_DELREPAYMENT VARCHAR(255),
    SCORE_RATE D_STR_100,
    RISK_RATE VARCHAR(2),
    RISK_RATE_DESCRIPTION D_STR_100,
    IPF_POINTS D_INTEGER,
    COMMON_POINTS D_INTEGER,
    KH_COMPANY_NAME D_STR_255,
    LOAN_PERCENT DECIMAL(18,2))
AS
declare variable CLIENT_FIO varchar(250);
declare variable KBM_SCORE_RATE D_INTEGER;
declare variable CHECK_DATE D_DATE;
declare variable PR_CODE D_STR_100;
declare variable BASE_K_ID D_ID;
declare variable BASE_MA_ID D_INTEGER;
declare variable INSUR_MA_ID D_INTEGER;
begin
  select pr.pr_code
  from miniapp ma
  inner join product pr on pr.pr_id = ma.ma_from_pr_id
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.2999'
  into pr_code;

  if (pr_code containing '.insur_kontrakt.') then  -- RFCKG-5051
  begin
    select lir.lir_kontract_base
    from kontract k
    inner join loan_insurance_relation lir on k.k_id = lir.lir_kontract_insurance
    where k.k_from_ma_app_id = :ma_id
    into base_k_id;

    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :base_k_id
    into base_ma_id;

    if (coalesce(base_ma_id, 0) <> 0) then
    begin
      insur_ma_id = ma_id;
      ma_id = base_ma_id;
    end
  end

  score_rate = null;
  select cl.c_name_fio
  from clients cl
  where cl.c_id = :c_id rows 1
  into :client_fio;

  kbm_score_rate = 0;
  score_rate = null;
  risk_rate_description = null;
  check_date = null;

  select kbm_score_rate, cast(kbm_checked_datetime as date)
  from hp_kibminiapp_last(:c_id, :ma_id)
  into kbm_score_rate, check_date;

  ipf_points = f_point(:c_id, :check_date, 2);
  common_points = f_point(:c_id, :check_date, 0);

  :score_rate = null;

  score_rate = (case
                  when (kbm_score_rate is not null and kbm_score_rate <> 999) then cast(kbm_score_rate as str_100)
                  when kbm_score_rate = 999 then 'Не может быть рассчитан'
                  when kbm_score_rate is null then 'Отсутствует'
                end);

  if (check_date < '30.01.2024') then
    risk_rate = (case
                   when kbm_score_rate between 250 and 250 then 'E3'
                   when kbm_score_rate between 251 and 379 then 'E2'
                   when kbm_score_rate between 380 and 529 then 'E1'
                   when kbm_score_rate between 530 and 559 then 'D3'
                   when kbm_score_rate between 560 and 574 then 'D2'
                   when kbm_score_rate between 575 and 598 then 'D1'
                   when kbm_score_rate between 599 and 619 then 'C3'
                   when kbm_score_rate between 620 and 641 then 'C2'
                   when kbm_score_rate between 642 and 664 then 'C1'
                   when kbm_score_rate between 665 and 679 then 'B3'
                   when kbm_score_rate between 680 and 694 then 'B2'
                   when kbm_score_rate between 695 and 709 then 'B1'
                   when kbm_score_rate between 710 and 724 then 'A3'
                   when kbm_score_rate between 725 and 739 then 'A2'
                   when kbm_score_rate between 740 and 900 then 'A1'
                   else '--'
                 end);
  else
    risk_rate = (case
                   when kbm_score_rate between 250 and 250 then 'E3'
                   when kbm_score_rate between 251 and 529 then 'E2'
                   when kbm_score_rate between 530 and 573 then 'E1'
                   when kbm_score_rate between 574 and 603 then 'D3'
                   when kbm_score_rate between 604 and 619 then 'D2'
                   when kbm_score_rate between 620 and 640 then 'D1'
                   when kbm_score_rate between 641 and 652 then 'C3'
                   when kbm_score_rate between 653 and 664 then 'C2'
                   when kbm_score_rate between 665 and 676 then 'C1'
                   when kbm_score_rate between 677 and 688 then 'B3'
                   when kbm_score_rate between 689 and 700 then 'B2'
                   when kbm_score_rate between 701 and 712 then 'B1'
                   when kbm_score_rate between 713 and 730 then 'A3'
                   when kbm_score_rate between 731 and 742 then 'A2'
                   when kbm_score_rate between 743 and 900 then 'A1'
                   else '--'
                 end);

  risk_rate_description = (case
                             when risk_rate in ('E3', 'E2', 'E1') then 'Очень высокий риск'
                             when risk_rate in ('D3', 'D2', 'D1') then 'Высокий риск'
                             when risk_rate in ('C3', 'C2', 'C1') then 'Средний риск'
                             when risk_rate in ('B3', 'B2', 'B1') then 'Низкий риск'
                             when risk_rate in ('A3', 'A2', 'A1') then 'Очень низкий риск'
                             else '--'
                           end);

  select sum(t.kh_aprox_balance), sum(t.kh_monthly_repay), sum(t.kh_is_active), max(t.kh_hist_qty),
         count(t.kh_is_active)
  from aarep_kib_miniapp_report(:ma_from_reg_id, :ma_id, :client_fio, :full_history) t
  into basic_amount, monthly_repay, active_cnt, all_cnt, shown_cnt;

  for select kf_client_id, kf_client_name, kf_client_pass_no, kf_amount, ma_kib_send, ma_kib_recieve, kh_all_balance,
             kh_all_kredit, kh_aprox_balance, kh_dateaward, kh_amount, kh_period_in_month, kh_clients_kol, kh_limit,
             kh_datestop, kh_dateprolongation, kh_amonthofprolongation, kh_unpaidamount, kh_unpaidpercenamount,
             kh_countofdelinquency, kh_creditstatus, kh_credittype, kh_currency, kh_repaystatus, kh_is_our,
             kh_create_date, kh_clients_fio, kh_poruch_fio, kh_is_active, kh_monthly_repay, kf_is_poruch, un_fio,
             kh_all_active_kol, kh_no, kf_date_birth, kf_inn, kf_is_poruch_str, kh_deliqdayscount, kh_ownamount,
             kh_hist_qty, kh_hist_shown, is_real_hist, ma_kolich_repay, kbm_checked_date, kh_delrepayment,
             iif(kh_company_name='',null,kh_company_name), loan_percent
      from aarep_kib_miniapp_report(:ma_from_reg_id, :ma_id, :client_fio, :full_history)
      into :kf_client_id, :kf_client_name, :kf_client_pass_no, :kf_amount, :ma_kib_send, :ma_kib_recieve,
           :kh_all_balance, :kh_all_kredit, :kh_aprox_balance, :kh_dateaward, :kh_amount, :kh_period_in_month,
           :kh_clients_kol, :kh_limit, :kh_datestop, :kh_dateprolongation, :kh_amonthofprolongation, :kh_unpaidamount,
           :kh_unpaidpercenamount, :kh_countofdelinquency, :kh_creditstatus, :kh_credittype, :kh_currency,
           :kh_repaystatus, :kh_is_our, :kh_create_date, :kh_clients_fio, :kh_poruch_fio, :kh_is_active,
           :kh_monthly_repay, :kf_is_poruch, :un_fio, :kh_all_active_kol, :kh_no, :kf_date_birth, :kf_inn,
           :kf_is_poruch_str, :kh_deliqdayscount, :kh_ownamount, :kh_hist_qty, :kh_hist_shown, :is_real_hist,
           :ma_kolich_repay, :kbm_checked_date, :kh_delrepayment, :kh_company_name, :loan_percent
  do
  begin
    if (coalesce(insur_ma_id, 0) <> 0) then  -- RFCKG-5051
    begin
      select ma.ma_kolich_repay
      from miniapp ma
      where ma_id = :insur_ma_id and
            ma_date_end = '31.12.9999' and
            ma.ma_is_active = 1
      into :ma_kolich_repay;

      select mac.mac_amount
      from miniappclient mac
      where mac.mac_from_ma_id = :insur_ma_id and
            mac.mac_amount > 0
      into kf_amount;
    end

    suspend;
  end
end
