CREATE PROCEDURE FM_ADD_CLIENT_REL(
    C_NAME_FIO D_STR_100,
    C_SEX D_SMALLINT,
    C_BIRTH_DATE D_DATE,
    C_NATION D_INTEGER,
    C_CITIZEN D_INTEGER,
    C_PHONE_NO D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    C_ID D_ID,
    ERR_TEXT D_STR_255)
AS
begin
  -- Добавление родственника клиента. Сокращенный вариант полей + № телефона

  err_text = '';
  begin
    -- Генерируем c_id
    select result
    from get_kv_id(2)
    into :c_id;

    if (coalesce(trim(c_phone_no), '') = '') then
    begin
      err_text = 'Номер телефона не введен.';
      suspend;
      exit;
    end

    insert into clients (c_id, c_type, c_name_fio, c_sex, c_birth_date, c_nation, c_citizen, c_is_personal,
                         c_from_reg_id, c_from_un_id, c_source_system)
    values (:c_id, 0, :c_name_fio, :c_sex, :c_birth_date, :c_nation, :c_citizen, 1, 1000000, :un_id, 1)
    returning c_id
    into c_id;

    -- Сохранение номера телефона
    execute procedure hp_continfo_update(:c_id, 'contactinfo.cell_main', :c_phone_no, 0, :un_id);

    when any do
      err_text = 'Ошибка при добавлении родственника клиента. gds_code=' || gdscode;
  end
  suspend;
end
