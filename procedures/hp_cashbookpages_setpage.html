CREATE PROCEDURE HP_CASHBOOKPAGES_SETPAGE(
    CBP_FROM_REG_ID INTEGER,
    CBP_DATE DATE,
    CBP_FIRST_PAGE_NUM INTEGER,
    CBP_LAST_PAGE_NUM INTEGER,
    CBP_SALDO_BEGIN DECIMAL(18,4),
    CBP_SALDO_END DECIMAL(18,4),
    CBP_FROM_UN_ID DECIMAL(18,0))
AS
declare variable prev_last_page_num integer;
declare variable cbp_year integer;
declare variable tmp_cbp_date date;
declare variable prev_saldo_end decimal(18,4);
begin
  /* Сохранение нумерации после очередной печати кассовой книги
     Подразумевается что если указанную дату отчет был приготовлен,
     то в этой дате всегда есть запись с cbp_number=0
  */

-- Запоминаем предыдущий номер последней страницы отчета
  prev_last_page_num = null;
  prev_saldo_end = null;
  select cbp_last_page_num, cbp_saldo_end
  from cashbookpages
  where cbp_date = :cbp_date and
        cbp_from_reg_id = :cbp_from_reg_id and
        cbp_number = 0
  into prev_last_page_num, prev_saldo_end;

  if (prev_last_page_num is not null) then
  begin
-- Запись с cbp_number = 0 есть, следовательно, отчет до этого печатали
-- Записи в текущем дне сдвигаем вниз
    update cashbookpages
    set cbp_number = cbp_number + 1
    where cbp_date = :cbp_date and
          cbp_from_reg_id = :cbp_from_reg_id order by cbp_number desc;

-- Затем проверяем изменился ли номер последней страницы отчета
-- или изменилось сальдо
    if (cbp_last_page_num <> prev_last_page_num or cbp_saldo_end <> prev_saldo_end) then
    begin
-- Нумерация в следующих датах "сломалась". Каждой следующей
-- напечатанной книге до конца года убираем признак актуальности
      cbp_year = extract(year from cbp_date);

-- Аннулируем нумерацию, со следующей даты и до конца года
-- где есть cbp_number = 0 и нумерация актуальна
      for select cbp_date
          from cashbookpages
          where extract(year from cbp_date) = :cbp_year and
                cbp_date > :cbp_date and
                cbp_from_reg_id = :cbp_from_reg_id and
                cbp_number = 0 and
                cbp_is_actual = 1
          into tmp_cbp_date
      do
      begin
-- Сдвигаем записи в очередном дне
        update cashbookpages
        set cbp_number = cbp_number + 1
        where cbp_date = :tmp_cbp_date and
              cbp_from_reg_id = :cbp_from_reg_id order by cbp_number desc;
-- Добавляем новую запись cbp_actual = 0 (делаем неактуальной)
        insert into cashbookpages (cbp_date, cbp_from_reg_id, cbp_number, cbp_first_page_num, cbp_last_page_num,
                                   cbp_saldo_begin, cbp_saldo_end, cbp_from_un_id, cbp_is_actual, cbp_changed_date)
        values (:tmp_cbp_date, :cbp_from_reg_id, 0, 0, 0, 0, 0, :cbp_from_un_id, 0, :cbp_date);
      end
    end
  end

-- Вставляем новую актуальную запись, в текущем дне
  insert into cashbookpages (cbp_date, cbp_from_reg_id, cbp_number, cbp_first_page_num, cbp_last_page_num,
                             cbp_saldo_begin, cbp_saldo_end, cbp_from_un_id, cbp_is_actual)
  values (:cbp_date, :cbp_from_reg_id, 0, :cbp_first_page_num, :cbp_last_page_num, :cbp_saldo_begin, :cbp_saldo_end,
          :cbp_from_un_id, 1);
end
