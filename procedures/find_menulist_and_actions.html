CREATE PROCEDURE FIND_MENULIST_AND_ACTIONS(
    FIND_TXT VARCHAR(100),
    FIND_KIND VARCHAR(100))
RETURNS (
    RES_ID INTEGER,
    RES_TEXT VARCHAR(255),
    RES_MEMO VARCHAR(255),
    RES_PATH VARCHAR(255),
    RES_KIND VARCHAR(100))
AS
declare variable res_parent integer;
declare variable tmp_txt varchar(255);
begin
--  find_kind:
--            all - искать везде
--            menu - искать среди меню
--            permission - искать среди действий

  if (find_txt = '') then
    exit;
  if ((find_kind = 'all') or (find_kind = 'menu')) then
  begin
    res_kind = 'Меню';
    for select ml_id, ml_parent_id, ml_caption, ml_description
        from menulist
        where (ml_caption like '%' || :find_txt || '%' or
              ml_description like '%' || :find_txt || '%')
        into :res_id, :res_parent, :res_text, :res_memo
    do
    begin
      res_path = '';
      while (res_parent is not null) do
      begin
        select ml_parent_id, ml_caption
        from menulist
        where ml_id = :res_parent
        into :res_parent, :tmp_txt;
        if (res_path <> '') then
          res_path = tmp_txt || ' -> ' || res_path;
        else
          res_path = tmp_txt;
      end
      suspend;
    end
  end
  if ((find_kind = 'all') or (find_kind = 'permission')) then
  begin
    res_kind = 'Полномочия';
    for select pa_id, pa_parent_id, pa_name, pa_memo
        from permactions
        where (pa_name like '%' || :find_txt || '%' or
              pa_memo like '%' || :find_txt || '%')
        into :res_id, :res_parent, :res_text, :res_memo
    do
    begin
      res_path = '';
      while (res_parent is not null) do
      begin
        select pa_parent_id, pa_name
        from permactions
        where pa_id = :res_parent
        into :res_parent, :tmp_txt;
        if (res_path <> '') then
          res_path = tmp_txt || ' -> ' || res_path;
        else
          res_path = tmp_txt;
      end
      suspend;
    end
  end
end
