CREATE PROCEDURE HP_FIX_UNIQ_ACC(
    C_ID ID_TYPE)
RETURNS (
    UNIQ_ACC_STR VARCHAR(100))
AS
declare variable c_birth_date date;
declare variable un_id id_type;
begin
  for select cl.c_birth_date, cl.c_from_un_id
      from clients cl
      where cl.c_id = :c_id and
            cl.c_birthday_commited = 1
      into :c_birth_date, :un_id
  do
  begin
    select uniq_acc_str
    from hp_show_clientuniq_acc_no(:c_id)
    into :uniq_acc_str;

    if (uniq_acc_str = 'XXX') then
    begin
      execute procedure hp_generate_c_uniq_id(:c_id, :c_birth_date, 1, :un_id);

      select uniq_acc_str
      from hp_show_clientuniq_acc_no(:c_id)
      into :uniq_acc_str;
    end
  end
  if (uniq_acc_str is null) then
    uniq_acc_str = 'Клиент не найден';
  suspend;
end
