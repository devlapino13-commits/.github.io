CREATE PROCEDURE HP_CAN_CALL_CHECK(
    K_ID ID_TYPE,
    OBJ_KIND STR_100,
    OBJ_ID ID_TYPE,
    REP_DATETIME D_DATETIME = null,
    IGNORE_NEXT_CALL_CHECK D_BOOL = 0)
RETURNS (
    CAN_CALL D_SMALLINT,
    REASON STR_255,
    NEXT_CALL_DATE D_DATE,
    TIMEZONE D_INTEGER,
    MSK_CALL_TIME_START D_TIME,
    MSK_CALL_TIME_END D_TIME)
AS
declare variable C_ID ID_TYPE;
declare variable CC_ID ID_TYPE;
declare variable CELL_NO STR_100;
begin
  -- доработка согласно задаче RFCRU-3495
  rep_datetime = coalesce(rep_datetime,  current_timestamp);
  -- процедура отвечает за проверку, наступила ли дата + время звонка абоненту

  can_call = 1;
  reason = '';

  c_id = null;
  cc_id = null;

  if (obj_kind = 'Clients') then
    c_id = :obj_id;
  else if (obj_kind = 'Contacts') then
    cc_id = :obj_id;
  else
    exception e_message 'obj_kind is not defined';

  if (c_id is not null) then
  begin
    select cast(pds.pds_next_call_datetime as date)
    from pantera_data_svod pds
    inner join pantera_kontracts pak on pak.pant_id = pds.pds_from_pan_id
    where pds.pds_from_c_id = :c_id
    and pak.pant_from_k_id = :k_id
    order by pds.pds_id desc
    rows 1
    into :next_call_date;

    if ((next_call_date > current_date) and (ignore_next_call_check = 0)) then
        reason = 'Дата для следующего звонка еще не наступила';
    else
    begin
        cell_no = f_cont_info('clients', c_id, 'cell_main', 'last');
    
        select msk_call_time_start,
               msk_call_time_end,
               reg_timezone
        from hp_places_get_timezone_c(:c_id, :cell_no)
        into msk_call_time_start,
             msk_call_time_end,
             timezone;
    
        if (cast(current_time as d_time) not between msk_call_time_start and msk_call_time_end) then
          reason = 'Текущее время вне допустимого диапазона для звонков';
     end
  end

  if (cc_id is not null) then
  begin
    select cast(pds.pds_next_call_datetime as date)
    from pantera_data_svod pds
    inner join pantera_kontracts pak on pak.pant_id = pds.pds_from_pan_id
    where pds.pds_from_cc_id = :cc_id
    and pak.pant_from_k_id = :k_id
    order by pds.pds_id desc
    rows 1
    into :next_call_date;

    if ((next_call_date > current_date) and (ignore_next_call_check = 0)) then
        reason = 'Дата для следующего звонка еще не наступила';
    else
        begin
        select timezone,
               msk_call_time_start,
               msk_call_time_end
        from hp_places_get_timezone_cc(:cc_id)
        into :timezone,
             :msk_call_time_start,
             :msk_call_time_end;
    
        if (cast(current_time as d_time) not between msk_call_time_start and msk_call_time_end) then
          reason = 'Текущее время вне допустимого диапазона для звонков клиенту';
     end
  end

  if (reason <> '') then
    can_call = 0;

  suspend;
end
