CREATE PROCEDURE HP_E_HR_REBUKE_LIST(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    IN_C_ID D_ID,
    BONUS_CHANGE D_SMALLINT,
    DOC_TYPE D_INTEGER,
    DOC_NUMBER D_STR_255)
RETURNS (
    RB_ID INTEGER,
    CURRENT_REG_NAME VARCHAR(255),
    DOC_REG_NAME VARCHAR(255),
    RB_NUMBER VARCHAR(255),
    RB_DATE DATE,
    C_NAME_FIO VARCHAR(255),
    RB_TYPE VARCHAR(255),
    RB_BONUS_CHANGE SMALLINT,
    RB_BONUS_KOEF DECIMAL(18,2),
    RB_PERIOD_START DATE,
    RB_PERIOD_END DATE,
    INIT_FIO VARCHAR(255),
    IS_WORKING SMALLINT,
    RB_REASON VARCHAR(255),
    JL_NAME VARCHAR(255),
    RS_SCORES D_SMALLINT,
    RES_SCORE_LIST D_STR_100,
    RES_BLANK_CODE D_STR_100,
    RB_IS_DELETE D_BOOL)
AS
declare variable rb_from_c_id id_type;
declare variable rb_type_id d_integer;
declare variable rb_type_prv d_integer;
begin
  /*
    PURPOSE : Модуль - "Регистрация выговоров и замечаний"
              Журнал документов. Предоставляет список документов "Выговоров" в соответствии с настройками фильтра

    Выборка осуществляется по последнему офису сотрудника

  */
  if (coalesce(bonus_change, 2) = 2) then
    bonus_change = null;

  if (doc_number is null) then
    doc_number = '';

  if (coalesce(doc_type, 0) = 0) then
    doc_type = null;

  res_blank_code = 'hr_rebuke';
  if (in_c_id is not null) then
  begin
    for select rb_id, reg.reg_name, reg_doc.reg_name, rb_number, rb_date, cl.c_name_fio, rb_type,
               rb_bonus_change, rb.rb_bonus_koef, rb.rb_period_start, rb.rb_period_end, cli.c_name_fio as init_fio,
               hrps.ps_is_working, rb.rb_reason, rb_from_c_id, rb.rb_is_delete
        from rebuke rb
        inner join clients cl on cl.c_id = rb.rb_from_c_id
        left join clients cli on cli.c_id = rb.rb_initiator_id
        inner join hr_pstatus hrps on hrps.ps_from_c_id = cl.c_id /* для выявления конечного офиса работы сотрудника*/
        inner join region_list(:rep_region_id, 1, -1, 1) rl on hrps.ps_from_reg_id = rl.id_list
        inner join region reg on reg.reg_unicode = rl.id_list /*текущий офис сотрудника*/
        inner join region reg_doc on reg_doc.reg_unicode = rb.rb_from_reg_id /*офис документа*/
        where rb.rb_is_delete = 0 and
              current_date between hrps.ps_date_begin and hrps.ps_date_end and /*всегда только опследний офис работы сорудника на текущую дату*/
              rb_from_c_id = :in_c_id and
              rb.rb_type = coalesce(:doc_type, rb.rb_type) and
              rb.rb_number containing :doc_number and
              rb.rb_bonus_change = coalesce(:bonus_change, rb.rb_bonus_change)
        order by rb.rb_date desc, rb.rb_number_n desc, rb.rb_sub_number desc
        into :rb_id, :current_reg_name, :doc_reg_name, :rb_number, :rb_date, :c_name_fio, :rb_type_id,
             :rb_bonus_change, :rb_bonus_koef, :rb_period_start, :rb_period_end, :init_fio,
             :is_working, :rb_reason, :rb_from_c_id, :rb_is_delete
    do
    begin
      if (rb_type_prv is distinct from rb_type_id) then
      begin
        rb_type = f_enum_name('rebuke_type', rb_type_id);
        rb_type_prv = rb_type_id;
      end

      rs_scores = 0;
      select sum(rs_scores)
      from rebuke_scores
      where rs_from_rb_id = :rb_id and
            rs_is_delete = 0
      into rs_scores;

      if (rs_scores is null) then
        rs_scores = 0;
  
      res_score_list = '';
      select list(distinct rbk_code)
      from rebuke_scores, rebuke_kind
      where rs_from_rb_id = :rb_id and
            rs_is_delete = 0 and
            rs_from_rbk_id = rbk_id
      into res_score_list;

      jl_name = '';
      select jl.jl_name
      from hr_pstatus hrps
      inner join joblist jl on jl.jl_id = hrps.ps_from_jl_id
      where hrps.ps_from_c_id = :rb_from_c_id and
            :rb_date between hrps.ps_date_begin and hrps.ps_date_end
      into :jl_name;
  
      suspend;
    end
  end else
  begin
    for select rb_id, reg.reg_name, reg_doc.reg_name, rb_number, rb_date, cl.c_name_fio, rb_type,
               rb_bonus_change, rb.rb_bonus_koef, rb.rb_period_start, rb.rb_period_end, cli.c_name_fio as init_fio,
               hrps.ps_is_working, rb.rb_reason, rb_from_c_id, rb.rb_is_delete
        from rebuke rb
        inner join clients cl on cl.c_id = rb.rb_from_c_id
        left join clients cli on cli.c_id = rb.rb_initiator_id
        inner join hr_pstatus hrps on hrps.ps_from_c_id = cl.c_id /* для выявления конечного офиса работы сотрудника*/
        inner join region_list(:rep_region_id, 1, -1, 1) rl on hrps.ps_from_reg_id = rl.id_list
        inner join region reg on reg.reg_unicode = rl.id_list /*текущий офис сотрудника*/
        inner join region reg_doc on reg_doc.reg_unicode = rb.rb_from_reg_id /*офис документа*/
        where rb.rb_is_delete = 0 and
              current_date between hrps.ps_date_begin and hrps.ps_date_end and /*всегда только опследний офис работы сорудника на текущую дату*/
              rb.rb_date between :rep_date_begin and :rep_date_end and
              rb.rb_type = coalesce(:doc_type, rb.rb_type) and
              rb.rb_number containing :doc_number and
              rb.rb_bonus_change = coalesce(:bonus_change, rb.rb_bonus_change)
        order by rb.rb_date desc, rb.rb_number_n desc, rb.rb_sub_number desc
        into :rb_id, :current_reg_name, :doc_reg_name, :rb_number, :rb_date, :c_name_fio, :rb_type_id,
             :rb_bonus_change, :rb_bonus_koef, :rb_period_start, :rb_period_end, :init_fio,
             :is_working, :rb_reason, :rb_from_c_id, :rb_is_delete
    do
    begin
      if (rb_type_prv is distinct from rb_type_id) then
      begin
        rb_type = f_enum_name('rebuke_type', rb_type_id);
        rb_type_prv = rb_type_id;
      end
  
      rs_scores = 0;
      select sum(rs_scores)
      from rebuke_scores
      where rs_from_rb_id = :rb_id and
            rs_is_delete = 0
      into rs_scores;

      if (rs_scores is null) then
        rs_scores = 0;
  
      res_score_list = '';
      select list(distinct rbk_code)
      from rebuke_scores, rebuke_kind
      where rs_from_rb_id = :rb_id and
            rs_is_delete = 0 and
            rs_from_rbk_id = rbk_id
      into res_score_list;

      jl_name = '';
      select jl.jl_name
      from hr_pstatus hrps
      inner join joblist jl on jl.jl_id = hrps.ps_from_jl_id
      where hrps.ps_from_c_id = :rb_from_c_id and
            :rb_date between hrps.ps_date_begin and hrps.ps_date_end
      into :jl_name;
  
      suspend;
    end
  end
end
