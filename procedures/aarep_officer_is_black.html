CREATE PROCEDURE AAREP_OFFICER_IS_BLACK(
    REP_DATE_END DATE)
RETURNS (
    PS_IS_ON_STAFF_STR VARCHAR(20),
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_BIRTH_DATE DATE,
    JL_NAME VARCHAR(100),
    REG_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(50),
    REG_NAME_REGION VARCHAR(50),
    REG_NAME_OKRUG VARCHAR(50),
    M_CLIENT_FIO VARCHAR(100),
    M_CLIENT_PASS_NO VARCHAR(50),
    M_FROM_REG_ID INTEGER,
    M_COMMENT_GROUP VARCHAR(2048),
    M_COMMENT_OFFICER VARCHAR(2048),
    M_COMMENT_MANAGER VARCHAR(2048),
    M_CLIENT_ADDRESS VARCHAR(255),
    M_IS_BLACK_STR VARCHAR(50),
    M_CLIENT_BIRTH_DATE DATE)
AS
declare variable m_is_black smallint;
declare variable ps_is_on_staff integer;
declare variable ps_from_c_id decimal(18,0);
declare variable ps_from_reg_id integer;
declare variable ps_from_jl_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
declare variable m_id decimal(18,0);
declare variable m_number integer;
declare variable c_inn varchar(20);
declare variable c_plastcard_no varchar(100);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ps_from_c_id, ps_from_reg_id, ps_is_on_staff, ps_date_begin, ps_date_end, ps_from_jl_id, c_name_fio,
             c_pass_no, c_birth_date, c_inn,
             (select cont_value
              from hp_contdata_get_cell_main(0, c_id))
      from hr_pstatus, clients
      where ps_from_c_id = c_id and
            ps_is_working = 1 and
            :rep_date_end between ps_date_begin and ps_date_end
      into :ps_from_c_id, :ps_from_reg_id, :ps_is_on_staff, :ps_date_begin, :ps_date_end, :ps_from_jl_id, :c_name_fio,
           :c_pass_no, :c_birth_date, :c_inn, :c_plastcard_no
  do
  begin
    for select m_id, m_number
        from hp_blacklist_search(:ps_from_c_id, :c_name_fio, :c_birth_date, :c_pass_no, :c_inn, :c_plastcard_no, 1, 1, 0)
        into :m_id, :m_number
    do
    begin
      for select m_client_fio, m_client_pass_no, m_from_reg_id, m_comment_group, m_comment_officer, m_comment_manager,
                 m_client_address, m_is_black, m_client_birth_date
          from moratory
          where m_id = :m_id and
                m_number = :m_number
          into :m_client_fio, :m_client_pass_no, :m_from_reg_id, :m_comment_group, :m_comment_officer,
               :m_comment_manager, :m_client_address, :m_is_black, :m_client_birth_date

      do
      begin
        select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name,
               r4.reg_code || ' ' || r4.reg_name
        from region r1
        left join region r2 on (r1.parent = r2.reg_unicode)
        left join region r3 on (r2.parent = r3.reg_unicode)
        left join region r4 on (r3.parent = r4.reg_unicode)
        where :ps_from_reg_id = r1.reg_unicode rows 1
        into :reg_name, :branch_name, :reg_name_region, :reg_name_okrug;

        select jl_name
        from joblist
        where :ps_from_jl_id = jl_id rows 1
        into :jl_name;

        m_is_black_str = case :m_is_black
                           when 1 then 'коричневый список (отказано в заявке)'
                           when 2 then  'стоп лист'
                         end;

        if (ps_is_on_staff is not null) then
          if (ps_is_on_staff = 1) then
            ps_is_on_staff_str = 'Штатный';
          else
            ps_is_on_staff_str = 'Внештатный';
        else
          ps_is_on_staff_str = 'Приказ не оформлен';

        suspend;
      end
    end
  end
end
