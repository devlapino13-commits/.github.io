CREATE PROCEDURE MILLICARD_DO_DISBURSEMENT(
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    DISB_AMOUNT CURR_TYPE)
AS
declare variable front_id str_100;
declare variable psi_id id_type;
declare variable pan str_100;
declare variable pr_code str_100;
declare variable project varchar(20);
declare variable front_id_source str_100;
declare variable project_id integer;
begin
  select pr.pr_code
  from kontract kont
  inner join product pr on kont.k_from_pr_id = pr.pr_id
  where kont.k_id = :k_id
  into :pr_code;

  project = 'millicard_disb';
  front_id_source = 'ObjectProps.Millicard_FrontId';
  project_id = 1;

  if (pr_code like '%.BizCardDisb.%') then
  begin
    project =  'millicard_biz_disb';
    front_id_source = 'ObjectProps.Millicard_Biz_FrontId';
    project_id = 2;
  end

  if (exists(select *
             from extsysmain em
             where em.esm_project = :project and
                   em.esm_action = 'addCredit' and
                   em.esm_from_k_id = :k_id and
                   em.esm_status in (0, 1))) then
    exit;

  -- Если id клиента, ранее уже был зарегистрирован в millicard тогда нет смысла создавать задачу на регистрацию нового клиента в millicard -->
  front_id = null;

  select ovs_value
  from hp_objectprops_get(0, :c_id, -1, :front_id_source, current_date)
  into :front_id;

  front_id = coalesce(front_id, '');
  ---------- <
  for select psi_id
      from hp_paysys_get_cl_info(:c_id, :project_id)
      order by psm_date desc
      rows 1
      into :psi_id
  do
  begin
    select pys.psi_code
    from paysys_item pys
    where pys.psi_id = :psi_id
    into :pan;
  end

  if (front_id = '') then -- клиент не был зарегистрирован в millicard, создаю задачу на регитсрацию клиента
  begin
    insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
                            esm_full_source, esm_amount, esm_from_k_id, esm_account)
    values (:project, 'addClient', current_date, 0, :c_id, 0, '', :disb_amount, :k_id, :pan);
  end
  else --  клиентранее уже был зарегистрирован в Millicard, сразу создаю задачу на выдачу кредита
  begin
    insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
                            esm_full_source, esm_amount, esm_from_k_id, esm_account)
    values (:project, 'addCredit', current_date, 0, :c_id, 0, '', :disb_amount, :k_id, :pan);
  end
end
