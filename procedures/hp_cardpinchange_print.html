CREATE PROCEDURE HP_CARDPINCHANGE_PRINT(
    CPH_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    CPH_DATE D_DATE,
    CARD_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    SIGNER_C_NAME_FIO D_STR_100,
    SIGNER_C_INN D_STR_20,
    SIGNER_C_PASS_NO D_STR_CODE,
    SIGNER_C_PASS_DATE D_DATE,
    SIGNER_C_PASS_WHO D_STR_255,
    SIGNER_C_ADDR D_STR_255,
    SIGNER_JL_NAME D_STR_100,
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    C_PASS_NO D_STR_CODE,
    C_PASS_DATE D_DATE,
    C_PASS_WHO D_STR_255,
    C_ADDR D_STR_255,
    COMPANY_NAME D_STR_255,
    COMPANYF_NAME D_STR_255,
    QR_CODE D_STR_LONG,
    CELL_NO D_STR_100,
    EDOC_CARD_ACT_NUMBER D_ID,
    EDOC_CARD_ACT_DATE D_DATE,
    CAL_PAN_SHORT D_STR_20)
AS
declare variable C_ID D_ID;
declare variable EDL_ID D_ID;
declare variable UN_ID D_ID;
declare variable SIGNER_ID D_ID;
begin
  cell_no = '';
  company_name = f_company_name('short', 'rus');
  companyf_name = f_company_name('full', 'rus');

  for
    select
      cph_date,
      cph_card_id,
      reg_name,
      c_id,
      cph_from_un_id,
      cph_res_edl_id
    from cardpinchange
      inner join region
        on reg_unicode = cph_region_id
      inner join clients
        on c_id = cph_from_c_id
    where cph_id = :cph_id
  into
    :cph_date,
    :card_id,
    :reg_name,
    :c_id,
    :un_id,
    :edl_id
  do
  begin
    cal_pan_short = (select cal_pan_short from cardlist where cal_card_id = :card_id);

    -- от кого
    un_id = 1090100000637;
    select
      c_id,
      c_name_fio,
      c_inn,
      c_pass_no,
      c_pass_date,
      c_pass_who
    from username
      inner join clients
        on c_id = un_from_c_id
    where un_id = :un_id
    into
      :signer_id,
      :signer_c_name_fio,
      :signer_c_inn,
      :signer_c_pass_no,
      :signer_c_pass_date,
      :signer_c_pass_who;
    signer_c_addr = (
      select
        adl_address_str
      from addrclient
        inner join addrlist
          on adl_id = adc_from_adl_id
      where adc_from_c_id = :signer_id
        and adc_address_type = 1);
    signer_jl_name = (
      select
        jl_name
      from hr_pstatus
        inner join joblist
          on jl_id = ps_from_jl_id
      where ps_from_c_id = :signer_id
        and :cph_date between ps_date_begin and ps_date_end
        and ps_is_working = 1
        and ps_status = 1);

    -- кому
    select
      c_name_fio,
      c_inn,
      c_pass_no,
      c_pass_date,
      c_pass_who
    from clients
    where c_id = :c_id
    into
      :c_name_fio,
      :c_inn,
      :c_pass_no,
      :c_pass_date,
      :c_pass_who;
    c_addr = (
      select
        adl_address_str
      from addrclient
        inner join addrlist
          on adl_id = adc_from_adl_id
      where adc_from_c_id = :c_id
        and adc_address_type = 1);
    if (coalesce(add_params, '') <> '') then
    begin
      edl_id = f_json_str_get_value(add_params, 'edl_id', null);
      qr_code = f_edoc_qr_by_edl(edl_id, :c_id);

      -- информация должна тянуться из договора на выпуск и обсуживание карт Элькарт,
      -- который активен на дату создания данного эл. документа.
      select first 1
        edl_id,
        cast(eds_status_datetime as date)
      from edoc_list
        inner join edoc_sign
          on eds_from_edl_id = edl_id
      where edl_object_kind = 0
        and edl_object_id = :c_id
        and edl_from_edc_id = f_edoc_kind('mbulak_card.agreement', 'id')
        and edl_status = 1
        and edl_id < :edl_id
      order by edl_id desc
      into
        :edoc_card_act_number,
        :edoc_card_act_date;
    end
    cell_no = f_cont_info('clients', :c_id, 'cell_main', 'ok');

    suspend;
  end
end
