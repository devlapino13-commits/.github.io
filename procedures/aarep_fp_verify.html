CREATE PROCEDURE AAREP_FP_VERIFY
RETURNS (
    FPV_ID INTEGER,
    FPV_MA_ID INTEGER,
    FPV_C_ID ID_TYPE,
    FPV_DATE_CHECK TIMESTAMP,
    FPV_FP_ID D_ID,
    FPV_FIO STR_255,
    FPV_BIRTHDATE DATE,
    FPV_PASS_NO STR_20,
    FPV_TERR_CATEG_ID INTEGER,
    FPV_IS_CONFIRM INTEGER,
    FPV_UN_ID ID_TYPE,
    FPV_CONFIRM_DATE TIMESTAMP,
    REG_NAME VARCHAR(50),
    C_NAME_FIO STR_100,
    C_BIRTH_DATE DATE,
    TERR_CATEG STR_255,
    UN_FIO STR_100,
    FPV_DESCRIPT STR_255,
    FPV_REASON STR_255,
    PERSON_DESCRIPTION D_STR_100)
AS
declare variable reg_id    integer;
declare variable is_client d_smallint;
declare variable is_intern boolean;
begin
  for
    select
      fpv_id,
      fpv_ma_id,
      fpv_c_id,
      fpv_date_check,
      fpv_fp_id,
      fpv_fio,
      fpv_birthdate,
      fpv_pass_no,
      fpv_terr_categ_id,
      fpv_is_confirm,
      fpv_un_id,
      fpv_confirm_date,
      descript,
      fpv_reason
    from fp_verify
      left join finpolis
        on fp_id = fpv_fp_id
    where fpv_is_confirm = 0
  into
    :fpv_id,
    :fpv_ma_id,
    :fpv_c_id,
    :fpv_date_check,
    :fpv_fp_id,
    :fpv_fio,
    :fpv_birthdate,
    :fpv_pass_no,
    :fpv_terr_categ_id,
    :fpv_is_confirm,
    :fpv_un_id,
    :fpv_confirm_date,
    :fpv_descript,
    :fpv_reason
  do
  begin
    is_intern = fpv_ma_id < 0;

    if (is_intern) then
    begin
      reg_id = (
        select i_from_reg_id
        from interns
        where i_id = abs(:fpv_ma_id));
      person_description = 'Стажёр';
    end else begin
      reg_id = (
        select first 1 ma_office_id from miniapp
        where ma_id = :fpv_ma_id
        order by ma_number desc);
      is_client = (
        select mac_is_poruch
        from miniappclient mac
        where mac.mac_from_ma_id = :fpv_ma_id
          and mac.mac_from_c_id = :fpv_c_id);
      person_description =
        case
          when is_client = 0 then 'Клиент'
          when is_client <> 0 or is_client is null then 'Родственник'
        else '' end;
      -- RFCKG-3647
      if (exists(
        select *
        from agreements
        where ag_object_id = :fpv_c_id
          and ag_object_kind = 0))
      then
        person_description = 'Контрагент';
    end

    reg_name = (
      select reg_name from region
      where reg_unicode = :reg_id);
    terr_categ = (
      select rd_name from refdata
      where rd_id = :fpv_terr_categ_id);
    un_fio = (
      select un_fio from username
      where un_id = :fpv_un_id);
    select
      c_name_fio,
      c_birth_date
    from clients
    where c_id = :fpv_c_id
    into
      :c_name_fio,
      :c_birth_date;

    suspend;
  end
end
