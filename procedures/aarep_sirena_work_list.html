CREATE PROCEDURE AAREP_SIRENA_WORK_LIST(
    REP_PROJECT STR_100,
    REP_DATE DATE)
RETURNS (
    OPER_CALLED STR_100,
    PROJECT STR_100,
    COMMAND_NAME STR_100,
    REGION_NAME STR_100,
    OFFICE_NAME STR_100,
    K_KONTRACT_NO STR_100,
    KAL_NEW_KF_NAME D_STR_100,
    OKS_DELINQ_DAY INTEGER,
    DELINQ_ZAD_OS CURR_TYPE,
    DELINQ_ZAD_PERC CURR_TYPE,
    DELINQ_ZAD_FINE CURR_TYPE,
    DELINQ_ZAD_ALL CURR_TYPE,
    BLOCK_DATE DATE,
    BLOCK_DELINQ_DAYS INTEGER,
    BLOCK_DELINQ_OS CURR_TYPE,
    BLOCK_DELINQ_PERC CURR_TYPE,
    BLOCK_DELINQ_FINE CURR_TYPE,
    BLOCK_DELINQ_ALL CURR_TYPE,
    BLOCK_REASON STR_100,
    CC_PROJECT STR_100,
    TRANSFER_CATEGORY STR_100)
AS
declare variable K_ID ID_TYPE;
declare variable KF_ID ID_TYPE;
declare variable OFFICE_ID_OLD integer;
declare variable REG_ID_OLD integer;
declare variable OFFICE_ID integer;
declare variable REG_ID integer;
declare variable PT_ID ID_TYPE;
declare variable PANT_DATE_BEGIN date;
declare variable PANT_ID ID_TYPE;
declare variable PROJECT_RD_ID integer;
declare variable OKF_DEBT D_CURRENCY;
declare variable OKS_CLOSE_STATE D_SMALLINT;
declare variable DELINQ_DATE D_DATE;
declare variable BLOCK_CLOSE_STATE D_SMALLINT;
begin
-- отражает группы в работе у операторов, в рамках проекта Сирена
  office_id_old = 0;
  reg_id_old = 0;

  for select pantk.pant_from_k_id, kont.k_kontract_no, kont.k_from_kf_id, pantk.pant_oper_called, pantk.pant_id,
             kont.k_last_reg_id, pantk.pant_date_begin, rd.rd_name, rd.rd_id, pantk.pant_id
      from pantera_kontracts pantk
      inner join kontract kont on kont.k_id = pantk.pant_from_k_id
      inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
      where :rep_date between pantk.pant_date_begin and pantk.pant_date_end and
            rd.rd_code containing (:rep_project)and
            pantk.pant_date_end >= pantk.pant_date_begin and
            ((pantk.pant_oper_called <> 'IGUSEV' and :rep_project <> 'OKTELL.PROJECTS.AUTO_INFORMER') or :rep_project = 'OKTELL.PROJECTS.AUTO_INFORMER') -- исключаю свой логин, т.к. выступаю виртуальным оператором (Исключение для проекта Автоинформатор BOARD-5273)
      order by pant_oper_called
      into :k_id, :k_kontract_no, :kf_id, :oper_called, :pant_id, :office_id, :pant_date_begin, :cc_project,
           :project_rd_id, :pant_id
  do
  begin
    kal_new_kf_name = '';
    select kal.kal_new_kf_name
    from knaktlist kal
    where kal.kal_from_kf_id = :kf_id and
          kal.is_delete = 0 and
          :rep_date between kal.kal_date and kal.kal_date_end
    into :kal_new_kf_name;

    command_name = '';
    if (:rep_project in ('OKTELL.PROJECTS.PANTERA','OKTELL.PROJECTS.SIRENA','OKTELL.PROJECTS.KS_FINANCE')) then
    begin
        select pcm.pcm_name
        from pantera_kontracts_com pkc
        inner join pantera_commands pcm on pkc.pkc_from_pcm_id = pcm.pcm_id
        where pkc.pkc_from_pant_id = :pant_id
        into command_name;
    end
     
    oks_delinq_day = null;
    delinq_zad_os = null;
    delinq_zad_perc = null;
    delinq_zad_fine = null;
    delinq_zad_all = null;
    oks_close_state = null;

    select (cast(:rep_date as date) - oks_delinq_date) oks_delinq_day,
           oks_close_state,
           oks.oks_out_os,
           oks.oks_out_perc,
           oks.oks_out_other,
           oks.oks_out_os + oks.oks_out_perc + oks.oks_out_other
    from oksummary oks
    where oks.oks_from_k_id = :k_id and
          :rep_date between oks_date_begin and oks_date_end
    into oks_delinq_day, -- определяю число дней просрочки
         oks_close_state,
         delinq_zad_os,
         delinq_zad_perc,
         delinq_zad_fine,
         okf_debt;

    if (oks_close_state is null) then
        continue;

    oks_delinq_day = coalesce(oks_delinq_day, 0);

    if (oks_delinq_day < 0) then
        oks_delinq_day = 0;

    if (oks_close_state = 1) then
    begin
       delinq_zad_all = okf_debt;
       delinq_date = null;

       select oks_delinq_date
       from oksummary
       where oks_from_k_id = :k_id
       and oks_delinq_date < '31.12.9999'
       order by oks_date_begin desc
       rows 1
       into delinq_date;
        
       if (delinq_date is not null) then
            oks_delinq_day = :rep_date - delinq_date;
    end

    if (oks_delinq_day <> 0) then
    begin
      if (office_id <> office_id_old) then
      begin
        office_id_old = office_id;

        select reg.reg_name, reg.parent
        from region reg
        where reg.reg_unicode = :office_id
        into :office_name, -- офис
             :reg_id;
      end

      if (reg_id_old <> reg_id) then
      begin
        reg_id_old = reg_id;

        select reg.reg_name
        from region reg
        where reg.reg_unicode = :reg_id
        into :region_name; -- регион
      end

      select iif(kal.kal_new_kf_officer = 1000000000004, 'ВС Сирена', 'СВЗ')
      from knaktlist kal
      where kal.is_delete = 0 and
            kal.kal_from_kf_id = :kf_id and
            :pant_date_begin between kal.kal_date and kal.kal_date_end
      into :project; -- за кем числится группа

      if (oks_close_state = 0) then
      begin
          -- просроченная задолжность на дату отчета
          select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine), sum(zad_os + sh_calc_perc + sh_calc_fine)
          from aarep_make_zad_list(:k_id, :rep_date, 1)
          where sh_date < :rep_date
          into :delinq_zad_os, :delinq_zad_perc, :delinq_zad_fine, :delinq_zad_all;
    
          delinq_zad_os = coalesce(delinq_zad_os, 0);
          delinq_zad_perc = coalesce(delinq_zad_perc, 0);
          delinq_zad_fine = coalesce(delinq_zad_fine, 0);
          delinq_zad_all = coalesce(delinq_zad_all, 0);
      end

      block_date = null;
      block_delinq_days = null;
      block_delinq_os = null;
      block_delinq_perc = null;
      block_delinq_fine = null;
      block_delinq_all = null;
      block_reason = null;

      pt_id = null;

      -- данные на момент блокировки
      select pantt.pt_id, pantt.pt_date, pantt.pt_delinq_days, pantt.pt_delinq_os, pantt.pt_delinq_perc,
             pantt.pt_delinq_fine, pantt.pt_delinq_os + pantt.pt_delinq_perc + pantt.pt_delinq_fine, rd.rd_name
      from pantera_transfer pantt
      inner join refdata rd on rd.rd_id = pantt.pt_reason_rd_id
      where pantt.pt_input_pant_id = :pant_id
      into :pt_id, :block_date, :block_delinq_days, :block_delinq_os, :block_delinq_perc, :block_delinq_fine,
           :block_delinq_all, :block_reason;

      if (pt_id is null) then -- если нет записи в pantera_transfer, тогда рассчитваю данные на дату блокировки, а не на момент блокировки
        block_date = pant_date_begin;

      block_close_state = null;

      select (cast(:block_date as date) - oks_delinq_date) oks_delinq_day,
                oks.oks_close_state
      from oksummary oks
      where oks.oks_from_k_id = :k_id and
            :block_date between oks_date_begin and oks_date_end
      into block_delinq_days,
           block_close_state;
        
      if (block_close_state = 1) then
      begin
           delinq_date = null;
        
           select oks_delinq_date
           from oksummary
           where oks_from_k_id = :k_id
           and oks_delinq_date < '31.12.9999'
           order by oks_date_begin desc
           rows 1
           into delinq_date;
            
           if (delinq_date is not null) then
                 block_delinq_days = :block_date - delinq_date;
      end
        
      select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine), sum(zad_os + sh_calc_perc + sh_calc_fine)
      from aarep_make_zad_list(:k_id, :block_date, 1)
      where coalesce(sh_date, cast('01.01.1900' as date)) < :block_date
      into :block_delinq_os, :block_delinq_perc, :block_delinq_fine, :block_delinq_all;

      block_delinq_os = coalesce(block_delinq_os, 0);
      block_delinq_perc = coalesce(block_delinq_perc, 0);
      block_delinq_fine = coalesce(block_delinq_fine, 0);
      block_delinq_days = coalesce(block_delinq_days, 0);
      block_delinq_all = coalesce(block_delinq_all, 0);
      block_reason = coalesce(block_reason, '');

      if (rep_project = 'OKTELL.PROJECTS.PANTERA') then
      begin
        transfer_category = 'Первичная';
        if (exists(select *
                   from pantera_kontracts pantk
                   where pantk.pant_id <> :pant_id and
                         pantk.pant_from_k_id = :k_id and
                         pantk.pant_date_begin <= :block_date and
                         coalesce(pantk.pant_is_dismiss, 0) = 0 -- важно, чтобы не поймать увольнения
                         and
                         pantk.pant_rd_project_type = :project_rd_id and
                         pantk.pant_oper_called <> :oper_called)) then
        begin
          transfer_category = 'Вторичная';
        end
      end

      suspend;
    end
  end
end
