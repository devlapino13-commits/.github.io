CREATE PROCEDURE HP_UXIMPORT_LIST
RETURNS (
    RES_ID D_STR_255,
    RES_PARENT_ID D_STR_255,
    RES_KIND D_STR_CODE,
    RES_IMAGE D_SMALLINT,
    RES_NAME D_STR_100,
    RES_CODE D_STR_100,
    RES_MEMO D_STR_LONG,
    RES_IS_OWN D_BOOL,
    RES_KIND_CAPTION D_STR_CODE,
    RES_TABLE_NAME D_STR_CODE,
    RES_DATETIME D_DATETIME,
    RES_WARNING_STATE D_SMALLINT,
    RES_CAN_SELECT D_BOOL,
    RES_SELECT_KIND D_SMALLINT)
AS
declare variable uim_new_item d_bool;
declare variable uim_new_child d_bool;
declare variable own_table d_str_code;
begin
  res_image = 0; -- папка
  res_is_own = 1;
  res_warning_state = 0;

  own_table = '';
  select uim_table
  from uximport
  where uim_id = 1
  into own_table;

  for select uim_id, uim_parent_id, uim_code, uim_name, uim_kind, uim_table, uim_new_item, uim_new_child, uim_can_select,
             uim_select_kind
  from uximport
  into :res_id, :res_parent_id, :res_code, :res_name, :res_kind, :res_table_name, :uim_new_item, :uim_new_child,
       :res_can_select, :res_select_kind
  do
  begin
    res_is_own = 1;
    if (uim_new_item = 1 or uim_new_child = 1) then
      res_is_own = 0;

    res_warning_state = 0;
    if (uim_new_item = 1) then
      res_warning_state = 1;
    else
    if (uim_new_child = 1) then
      res_warning_state = 2;

    if (res_can_select = 0) then
      res_select_kind = null;

    res_datetime = null;
    res_memo = '';

    if (uim_new_item = 0) then
    begin
      if (res_kind <> 'alias') then
      begin
        select ux_create_datetime, ux_memo
        from uxcode
        where ux_code = :res_code
        into res_datetime, res_memo;

        res_kind_caption = uxf$kind(res_kind, 'caption');
      end else
        res_kind_caption = 'Алиас';
    end else
      res_memo = 'Новый объект';

    if (res_table_name = own_table) then
    begin
      res_image = case
        when res_kind = 'uxform' then 19
        when res_kind = 'uxcommand' then 25
        when res_kind = 'uxreport' then 26
        when res_kind = 'alias' then 20
        else 19
      end;
    end else
    begin
      res_image = case
        when res_kind = 'uxform' then 53
        when res_kind = 'uxcommand' then 25
        when res_kind = 'uxreport' then 26
        when res_kind = 'alias' then 20
        else 19
      end;
    end

    suspend;
  end
end
