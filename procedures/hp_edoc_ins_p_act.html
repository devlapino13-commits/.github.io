CREATE PROCEDURE HP_EDOC_INS_P_ACT(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable pr_code d_str_100;
declare variable ma_id d_id;
declare variable c_id d_id;
declare variable k_status d_smallint;
declare variable ks_state d_smallint;
declare variable app_date d_date;
declare variable k_disb_date date;
declare variable kma_id d_id;
declare variable kml_purchase_datetime d_datetime;
declare variable db_id d_id;
declare variable prepay_sum curr_type;
begin
  --p_act = purchase act  акт прием передачи
  -- Процедура создает электронный документ по займу.
  -- Готовит бланки и отправляет СМС.
  -- Создан на основе HP_EDOC_INS_LOAN_AGREEMENT
  res_text = '';
  db_id = f_get_db_id();

  select k_from_ma_app_id, pr_code, k_status, k_disb_date
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into ma_id, pr_code, k_status, k_disb_date;

  if (k_status <> 1) then
    res_text = 'Сначала утвердите договор займа';

  if (res_text = '') then
  begin
    select ks_state
    from kontstate
    where ks_from_k_id = :k_id and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999'
    into ks_state;

    if (ks_state = 1) then
      res_text = 'Займ уже действующий';

    if (ks_state = 2) then
      res_text = 'Займ уже закрыт';
  end

  if (res_text = '') then
  begin
    for select ag_object_id, kma.kma_id
        from kontmatag kma
        inner join agreements ag on kma.kma_from_ag_id = ag.ag_id
        where kma.kma_from_k_id = :k_id
        into :c_id, :kma_id
    do
    begin
      if (coalesce(:c_id, 0) <> 0) then
      begin
        select kml_purchase_datetime
        from print_kontmat(:k_id, :c_id)
        order by kml_purchase_datetime desc
        rows 1
        into kml_purchase_datetime;

        if (coalesce(kml_purchase_datetime, '31.12.9999') = '31.12.9999') then
          res_text = 'Товар ещё не куплен';
      end
    end
  end
  
  --Перед отправкой акта приёма-передачи по мурабаха сначала проверяем наличие частичной оплаты
  --Если частичная оплата > 0, то проверяем оприходована ли эта сумма в базе
  if (db_id in (39801, 41701) and res_text = '' and pr_code containing '.murabaha.') then
  begin
    select ma.ma_prepayment_summa
    from miniapp ma
    where ma.ma_id = :ma_id and
          ma.ma_is_active = 1 and
          ma.ma_date_end >= '31.12.2999'
    into prepay_sum;

    if (coalesce(prepay_sum, 0) > 0) then
    begin
      if (not exists(select *
                     from operkontract ok
                     where ok.ok_from_k_id = :k_id and
                           ok.ok_operation_type = f_get_rd_id('CashTransactionIncom.PartialPrice') and
                           ok.ok_amount = :prepay_sum)) then
        res_text = 'Частичная сумма не оприходована в кассу. Оприходуйте сумму.';
    end
  end 


  --Цикл for нужен потому что товары могут быть куплены у нескольких контрагентов
  if (res_text = '') then
  begin
    for select kma.kma_id
        from kontmatag kma
        inner join agreements ag on kma.kma_from_ag_id = ag.ag_id
        where kma.kma_from_k_id = :k_id
        into kma_id
    do
    begin
      if (f_edoc_status('purchase_act.agreement', :kma_id, current_date) <> 'signed') then
      begin
        select edl_id, res_text
        from hp_edoc_insert('purchase_act.agreement', f_objkind('kontmatag', 'id'), :kma_id, :un_id)
        into edl_id, res_text;
      end
    end
  end 

  suspend;
end
