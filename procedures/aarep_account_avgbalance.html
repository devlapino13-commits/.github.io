CREATE PROCEDURE AAREP_ACCOUNT_AVGBALANCE(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_BALANCE_ID VARCHAR(20),
    REPORT_ACCOUNT_NO INTEGER,
    REPORT_CURRENCY VARCHAR(3),
    REPORT_IS_BALANCE_ACC SMALLINT)
RETURNS (
    ACC_CODE VARCHAR(20),
    ACC_BALANCE_END DECIMAL(18,4),
    ACC_BALANCE_AVG DECIMAL(18,4),
    ACC_NAME VARCHAR(200),
    ACC_BG_NAME VARCHAR(255))
AS
declare variable acc_dd date;
declare variable acc_summa decimal(18,4);
declare variable last_dd date;
declare variable has_oper integer;
declare variable total_summa decimal(18,4);
declare variable temp_summa decimal(18,4);
declare variable kol_day integer;
declare variable dd_delta integer;
declare variable al_id varchar(20);
declare variable tmp_summa decimal(18,4);
begin
  dd_delta = :report_date_end - :report_date_begin + 1;
  if (dd_delta > 0) then
  begin
    if (report_is_balance_acc = 0) then
    begin /* По лицевым счетам */
      for select al_id, al_name, bg_id || ' ' || bg_name
          from acc_list, balgroup, region_list(:report_region_id, 1, -1, 1)
          where al_from_bg_id = bg_id and
                al_from_reg_id = id_list and
                (bg_id = :report_balance_id or
                :report_balance_id = '') and
                (al_currency = :report_currency or
                :report_currency = '---') and
                (al_number = :report_account_no or
                :report_account_no = -1)

          into :acc_code, :acc_name, :acc_bg_name
      do
      begin
        acc_balance_end = f_al_bal(acc_code, report_date_end);

        temp_summa = acc_balance_end;
        total_summa = 0;
        last_dd = report_date_end;
        has_oper = 0;
        kol_day = 0;
        for select obs_date
            from obsummary
            where obs_from_al_id = :acc_code
            order by obs_date desc
            into :acc_dd
        do
        begin
          select dt_ekv - kt_ekv
          from hp_al_obor(:acc_code, :acc_dd, :acc_dd)
          into acc_summa;

          if (has_oper = 0) then
          begin
            kol_day = (last_dd - acc_dd) + 1;
            total_summa = total_summa + kol_day * temp_summa;
            last_dd = acc_dd;
            temp_summa = temp_summa - acc_summa;
          end
          else
          begin
            kol_day = (last_dd - acc_dd);
            total_summa = total_summa + kol_day * temp_summa;
            last_dd = acc_dd;
            temp_summa = temp_summa - acc_summa;
          end
          has_oper = 1;
        end
        if (has_oper = 1) then
        begin
          if (report_date_begin < last_dd) then
          begin
            kol_day = (last_dd - report_date_begin);
            total_summa = total_summa + kol_day * temp_summa;
          end
          acc_balance_avg = cast(total_summa / dd_delta as decimal(18,4));
        end
        else
          acc_balance_avg = acc_balance_end;
        suspend;
      end
    end
    else
    begin /* По балансовым счетам */
      acc_balance_end = 0;
      for select bg_id, bg_name, bg_id || ' ' || bg_name
          from balgroup
          where bg_id = :report_balance_id or
                :report_balance_id = ''
          into :acc_code, :acc_name, :acc_bg_name
      do
      begin
        acc_balance_end = 0;
        for select al_id
            from acc_list, region_list(:report_region_id, 1, -1, 1)
            where al_from_bg_id = :acc_code and
                  al_from_reg_id = id_list and
                  (al_currency = :report_currency or
                  :report_currency = '---') and
                  (al_number = :report_account_no or
                  :report_account_no = -1)
            into :al_id
        do
        begin
          tmp_summa = f_al_bal(al_id, report_date_end);
          acc_balance_end = acc_balance_end + tmp_summa;
        end

        temp_summa = acc_balance_end;
        total_summa = 0;
        last_dd = report_date_end;
        has_oper = 0;
        kol_day = 0;
        for select obs_date, obs_from_al_id
            from obsummary, acc_list, region_list(:report_region_id, 1, -1, 1)
            where obs_from_al_id = al_id and
                  al_from_bg_id = :acc_code and
                  al_from_reg_id = id_list and
                  obs_date between :report_date_begin and :report_date_end
            order by obs_date desc
            into :acc_dd, al_id
        do
        begin
          select dt_ekv - kt_ekv
          from hp_al_obor(:al_id, :acc_dd, :acc_dd)
          into acc_summa;

          if (has_oper = 0) then
          begin
            kol_day = (last_dd - acc_dd) + 1;
            total_summa = total_summa + kol_day * temp_summa;
            last_dd = acc_dd;
            temp_summa = temp_summa - acc_summa;
          end
          else
          begin
            kol_day = (last_dd - acc_dd);
            total_summa = total_summa + kol_day * temp_summa;
            last_dd = acc_dd;
            temp_summa = temp_summa - acc_summa;
          end
          has_oper = 1;
        end
        if (has_oper = 1) then
        begin
          if (report_date_begin < last_dd) then
          begin
            kol_day = (last_dd - report_date_begin);
            total_summa = total_summa + kol_day * temp_summa;
          end
          acc_balance_avg = cast(total_summa / dd_delta as decimal(18,4));
        end
        else
          acc_balance_avg = acc_balance_end;
        suspend;
      end
    end
  end
end
