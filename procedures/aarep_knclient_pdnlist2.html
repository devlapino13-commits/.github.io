      if (repeated = 1) then
        disb_datetime = null;

      if (iif(repeated in (0,2,3), k_disb_date, calc_date) > credenddate and calc_period_month = 1) then
        calc_period_month = 0;

      kpl_is_zeroperc_str = case kpl_is_zeroperc
                                 when 0 then 'Нет'
                                 when 1 then 'Да'
                                 when 2 then 'Руч'
                                 else null
                            end;

      credit_line_sign = null;

      if (credit_line_type is not null) then
        credit_line_sign = iif(left(credit_line_type, 1) = '0', 0, 1);

      if (repeated = 2) then     -- RFCRU-3093
      begin
        ref_check_date = null;  -- Дата проверки закрытия рефинансируемых займов
        ref_check_days = null;  -- Кол-во дней между датой проверки и датой выдачи
        ref_close_date = null;  -- Дата закрытия рефинансируемого займа
        ref_calc_date = null;   -- Дата повторного расчета ПДН в связи с невыполн цели реф-ия
        ref_calc_days = null;   -- Кол-во дней между датой повторного расчета ПДН и датой выдачи
        ref_k_id = null;
        ref_bls_id = null;

        select kr_from_k_id
        from kontract, kontrefinance
        where k_id = :k_id
          and kr_from_ma_id = k_from_ma_app_id
          and kr_loan_uuid = :credid
        into ref_k_id;

        if (row_count > 0) then
        begin
          ref_check_date = minvalue(ref_date_end, current_date);

          if (ref_k_id is not null) then
          begin -- для наших рефинансов
            select max(oks_date_end)
            from oksummary
            where oks_from_k_id = :ref_k_id
              and (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc) > 0
            into ref_close_date;

            ref_close_date = iif(ref_close_date > current_date + 1, null, ref_close_date + 1);
          end
          else
          begin -- для чужих рефинансов
            if (ref_bls_id is null) then
              select eil_from_bls_id
              from extinfolist, extinfocfg
              where eil_from_c_id = :c_id
                and eil_from_eic_id = eic_id
                and eic_code = 'equifax_40'
                and eil_status = 2
                and eil_date > :k_disb_date
              order by eil_date desc, eil_id
              rows 1
              into ref_bls_id;

            if (ref_bls_id is not null) then
            begin
              in autonomous transaction do
              begin
                json_full = '';
              
                if (ref_bls_id is not null) then
                begin
                  select cast(blx_blob as d_blob_text)
                  from hp_get_blobex_blob(:ref_bls_id)
                  into :json_full;
                end
              
                if (coalesce(json_full, '') not starting with '{') then
                  exception e_default 'wrong json in bls_id: ' || ref_bls_id;
              
                jso_full = json.parse(json_full);
              
                select credenddatefact
                from hp_pdn_get_from_json(:jso_full, :ref_date_end, 'equifax_40')
                where creduid = :credid
                rows 1
                into ref_close_date;
              end
            end
          end
  
          select kc_date
          from knclient_pdn
          where kc_id = :new_kc_id
            and kcp_kind = 2
          into ref_calc_date;

          ref_check_days = ref_check_date - k_disb_date;
          ref_calc_days = ref_calc_date - k_disb_date;
        end
      end

      if (repeated = 3 and rec_no = -1) then     -- RFCRU-5519
      begin
        loansign_datetime = null;
        final_pdn_last = null;
        calc_date_last = null;
        amp_sum_new_last = null;
        itogi_amp_our_sum_last = null;
        itogi_amp_ext_sum_last = null;
        calc_month_repay_final_last = null;
        kc_anketa_income_last = null;
        kc_stat_avg_income_last = null;
        kc_final_income_last = null;
        cmc_name = null;

        -- выбираем последний повторный, в выбранном периоде
        select final_pdn, kc_date, amp_sum_new, itogi_amp_our_sum, itogi_amp_ext_sum,
            calc_month_repay_final, kc_anketa_income, kc_stat_avg_income, kc_final_income,
            cmc_name, adl_address_str, str_id
        from aarep_knclient_one_pdnlist2(:new_kc_id, '1,2')
        where rec_no = -1
          and kc_date between :rep_date_begin and coalesce(:repeated_date_end, :rep_date_end)
        order by kc_date desc
        rows 1
        into final_pdn_last, calc_date_last, amp_sum_new_last, itogi_amp_our_sum_last, itogi_amp_ext_sum_last,
            calc_month_repay_final_last, kc_anketa_income_last, kc_stat_avg_income_last, kc_final_income_last,
            cmc_name, adl_address_str_last, str_id_last;

        select eds_status_datetime
        from knclient, edoc_list, edoc_cfg, edoc_sign
        where kc_id = :new_kc_id
          and edl_object_kind = 9
          and edl_object_id = kc_from_k_id
          and edl_from_edc_id = edc_id
          and eds_from_edl_id = edl_id
          and edc_code = 'loan.agreement'
          and eds_status = 1
          and edl_status = 1
          into loansign_datetime;
      end

      line = line + 1;
      suspend;
    end
  end
