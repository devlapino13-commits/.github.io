CREATE PROCEDURE HP_GET_REPANALITICACONFIG(
    RAT_ID INTEGER)
RETURNS (
    RAC_SECTION VARCHAR(255),
    RAC_OBJECT_NAME VARCHAR(255),
    RAC_OBJECT_VALUE VARCHAR(255))
AS
declare variable sec varchar(255);
begin
--MainParams              Параметры в главном окне
--gFieldsFlat_Selected    Порядок сортировки
--gFieldsTree_Selected    Поля с группировкой храняться отрицательным знаком
--gRepData_Bands_Order    если значение минусовое, значит не видимый
--gRepData_Fields_Band    Индекс Band, принадлежность поля к Band
--gRepData_Fields_Order    Порядок сортировки внутри Band, если значение минусовое, значит не видимый
--gRepData_Fields_Visible  Порядок сортировки внутри Band, если значение минусовое, значит не видимый
--gRepData_CalcFields      Вычисляемые поля
--gRepData_Summary         Формула расчета значения групп и итогов

  sec = 'MainParams';
  for select rac_section, rac_object_name, rac_object_value
      from repanaliticaconfig
      where rac_from_rat_id = :rat_id and
            rac_section = :sec
      into :rac_section, :rac_object_name, :rac_object_value
  do
    suspend;

  sec = 'gRepData_CalcFields';
  for select rac_section, rac_object_name, rac_object_value
      from repanaliticaconfig
      where rac_from_rat_id = :rat_id and
            rac_section = :sec
      into :rac_section, :rac_object_name, :rac_object_value
  do
    suspend;

  sec = 'gRepData_CalcFields_Caption';
  for select rac_section, rac_object_name, rac_object_value
      from repanaliticaconfig
      where rac_from_rat_id = :rat_id and
            rac_section = :sec
      into :rac_section, :rac_object_name, :rac_object_value
  do
    suspend;

  sec = 'gFieldsFlat_Selected';
  for select rac_section, rac_object_name, rac_object_value
      from repanaliticaconfig
      where rac_from_rat_id = :rat_id and
            rac_section = :sec
      into :rac_section, :rac_object_name, :rac_object_value
  do
    suspend;

  sec = 'gFieldsTree_Selected';
  for select rac_section, rac_object_name, rac_object_value
      from repanaliticaconfig
      where rac_from_rat_id = :rat_id and
            rac_section = :sec
      into :rac_section, :rac_object_name, :rac_object_value
  do
    suspend;

  sec = 'gRepData_Bands_Order';
  for select rac_section, rac_object_name, rac_object_value
      from repanaliticaconfig
      where rac_from_rat_id = :rat_id and
            rac_section = :sec
      into :rac_section, :rac_object_name, :rac_object_value
  do
    suspend;

  sec = 'gRepData_Fields_Band';
  for select rac_section, rac_object_name, rac_object_value
      from repanaliticaconfig
      where rac_from_rat_id = :rat_id and
            rac_section = :sec
      into :rac_section, :rac_object_name, :rac_object_value
  do
    suspend;

  sec = 'gRepData_Fields_Order';
  for select rac_section, rac_object_name, rac_object_value
      from repanaliticaconfig
      where rac_from_rat_id = :rat_id and
            rac_section = :sec
      order by cast(rac_object_value as integer)
      into :rac_section, :rac_object_name, :rac_object_value
  do
    suspend;

  sec = 'gRepData_Fields_Visible';
  for select rac_section, rac_object_name, rac_object_value
      from repanaliticaconfig
      where rac_from_rat_id = :rat_id and
            rac_section = :sec
      into :rac_section, :rac_object_name, :rac_object_value
  do
    suspend;

  sec = 'gRepData_Summary';
  for select rac_section, rac_object_name, rac_object_value
      from repanaliticaconfig
      where rac_from_rat_id = :rat_id and
            rac_section = :sec
      into :rac_section, :rac_object_name, :rac_object_value
  do
    suspend;
end
