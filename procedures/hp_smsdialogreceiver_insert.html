CREATE PROCEDURE HP_SMSDIALOGRECEIVER_INSERT(
    CELL_FROM VARCHAR(20),
    CELL_TO VARCHAR(20),
    SMS_TEXT VARCHAR(255))
AS
declare variable sdr_system_id integer;
declare variable db_id d_integer;
begin
  cell_from = replace(cell_from, '+', '');
  if (left(sms_text, 9) = 'Molbulak:') then
    sms_text = trim(substring(sms_text from 10 for 255));

  if (coalesce(sms_text, '') <> '') then
  begin
    db_id = f_get_db_id();
    if (db_id = 81001) then
      if (cell_from not starting '79') then Exit;

    if (db_id in (39801, 41701)) then
      if (cell_from not starting '996') then Exit;

    if (db_id = 94401) then
      if (cell_from not starting '994') then Exit;

    if (sms_text like '%#%') then
    begin
      execute procedure hp_do_sanjar_survey(:cell_from, :sms_text);
    end
    else
    begin
-- Ищем системную команду
      select sdsw_id
      from hp_smsdialog_sysword_finder(:sms_text)
      into sdr_system_id;

      if (sdr_system_id <> 0) then
      begin
        insert into smsdialogreciever (sdr_cell_no, sdr_text, sdr_system_id)
        values (:cell_from, :sms_text, :sdr_system_id);
      end
      else
      begin
        if (cell_to is null) then
          cell_to = '';
        insert into survey_result (svr_source_kind, svr_source_id, svr_text, svr_server_sim)
        values (1, :cell_from, :sms_text, :cell_to);
      end
    end
  end
end
