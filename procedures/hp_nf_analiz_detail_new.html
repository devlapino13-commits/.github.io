CREATE PROCEDURE HP_NF_ANALIZ_DETAIL_NEW(
    IFA_ID D_INTEGER,
    REC_NO D_INTEGER,
    MAIN_BNK D_STR_CODE,
    RECIEVED_TRAN D_INTEGER,
    MUST_PAY_TRAN D_CURRENCY,
    SUB_BNK D_STR_CODE,
    NETTO_COUNT_TRANS D_INTEGER,
    NETTO_AMOUNT D_CURRENCY)
RETURNS (
    IND_ID D_INTEGER)
AS
declare variable main_bnk_id d_integer;
declare variable sub_bnk_id d_integer;
begin
  main_bnk_id = (select ib_id from ipc_banks where ib_bank_code = :main_bnk);
  sub_bnk_id = (select ib_id from ipc_banks where ib_bank_code = :sub_bnk);

  insert into ipc_nonfin_detail (
    ind_from_ifa_id,
    ind_rec_no,
    ind_main_bank_id,
    ind_sub_bank_id,
    ind_recieved_tran,
    ind_must_pay_tran,
    ind_netto_count_trans,
    ind_netto_amount
  )
  values (
    :ifa_id,
    :rec_no,
    :main_bnk_id,
    :sub_bnk_id,
    :recieved_tran,
    :must_pay_tran,
    :netto_count_trans,
    :netto_amount
  )
  returning ind_id
  into :ind_id;

  suspend;
end
