CREATE PROCEDURE FM_HTML_SPF_EXAM_ACCESS(
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable reg_name d_str_100;
declare variable reg_id d_id;
declare variable link d_str_long;
declare variable obj_kind d_integer;
declare variable has_access d_smallint;
begin
  res_text = '';
  obj_kind = f_objkind('region', 'id');
  res_text = (select res_style
              from fm_html_styles) || '<body>';

  res_text = BLOB_APPEND(res_text, '<h2>Список офисов</h2><br>');

  res_text = BLOB_APPEND(res_text, '<table width="100%"><col width="75%"><col width="25%">');
  res_text = BLOB_APPEND(res_text, f_html_table_row('<b>Офис</b>', '<b>Доступ</b>', '', '', -1, -1, -1, -1, 2, ''));
  for select rg.reg_code || ' '  || rg.reg_name, rg.reg_unicode
      from region_list(1000000, 1, 3, 1) t
      inner join region rg on rg.reg_unicode = t.id_list
      into reg_name, reg_id
  do
  begin
    link = f_uri_sql('Разрешить аттестацию', 'give_exam_access', 'region', :reg_id, 'none', 'Подтвердите выдачу доступа к аттестации', '', null, null, null, null, 'dialog', null);
    has_access = 0;
    if (exists(select *
               from exam_access ea
               where ea.ea_object_kind = :obj_kind and
                     ea.ea_object_id = :reg_id and
                     current_date between ea.ea_date_begin and ea.ea_date_end)) then
      has_access = 1;

    res_text = BLOB_APPEND(res_text, f_html_table_row(:reg_name, iif(:has_access = 1, trim('Доступ есть'), trim(:link)), '', '', -1, -1, -1, -1, 2, ''));
  end
  res_text = BLOB_APPEND(res_text, '</table></body>');

  suspend;
end
