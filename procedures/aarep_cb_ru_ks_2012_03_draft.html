CREATE PROCEDURE AAREP_CB_RU_KS_2012_03_DRAFT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(50),
    AA_CURR_BRANCH_NAME VARCHAR(50),
    AA_CURR_KC_FIO VARCHAR(100),
    PS_IS_ON_STAFF_STR VARCHAR(20),
    AA_AMOUNT_DISB INTEGER,
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_DELIQ_BALANCE7 DECIMAL(18,4),
    AA_PAR DECIMAL(15,2),
    AA_KOLICH_CL_DISB INTEGER,
    AA_BONUS_FACTOR INTEGER,
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    JL_NAME VARCHAR(100),
    AA_KOLICH_CL_CLOSE INTEGER)
AS
declare variable ps_is_on_staff smallint;
declare variable aa_curr_kc_id decimal(18,0);
declare variable ps_is_working integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  aa_amount_disb = 0;
  aa_kolich_cl_disb = 0;
  aa_kolich_cl_close = 0;

  for select aa_curr_kc_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_kc_fio, sum(aa_amount_active_end),
             sum(aa_deliq_balance7), sum(aa_kolich_cl_active_end)
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 0, 0, 0, 0, 0, 0)
      group by aa_curr_kc_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_kc_fio
      into :aa_curr_kc_id, :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_kc_fio,

           :aa_amount_active_end, :aa_deliq_balance7, :aa_kolich_cl_active_end
  do
  begin
    aa_par = 0;
    if (:aa_amount_active_end <> 0) then
    begin
      aa_par = :aa_deliq_balance7 * 100;
      aa_par = cast(aa_par / :aa_amount_active_end as decimal(15,2));
    end
    aa_bonus_factor = 0;

    aa_bonus_factor = case
                        when aa_par <= 3.0 then 200
                        when aa_par between 3.01 and 7.0 then 150
                        when aa_par between 7.01 and 10.0 then 100
                        when aa_par between 10.01 and 15.0 then 50
                        when aa_par >= 15.01 then 0
                      end;

    ps_is_on_staff = null;
    select jl_name, ps_is_on_staff, ps_is_working
    from hr_pstatus, joblist
    where ps_from_c_id = :aa_curr_kc_id and
          ps_date_begin <= ps_date_end and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_jl_id = jl_id
    order by ps_is_working desc
    rows 1
    into :jl_name, :ps_is_on_staff, :ps_is_working;

    if (ps_is_on_staff is not null) then
    begin
      if (ps_is_on_staff = 1) then
        ps_is_on_staff_str = 'Штатный';
      else
        ps_is_on_staff_str = 'Внештатный';
      if (ps_is_working = 0) then
        ps_is_on_staff_str = ps_is_on_staff_str || ', ' || 'уволен';
    end
    else
      ps_is_on_staff_str = 'Приказ не оформлен';

    suspend;
  end
  aa_curr_office_name = '';
  aa_curr_branch_name = '';
  jl_name = '';
  ps_is_on_staff_str = '';

  aa_par = 0;
  aa_amount_active_end = 0;
  aa_deliq_balance7 = 0;
  aa_kolich_cl_active_end = 0;
  aa_bonus_factor = 0;

  for select kal_new_kf_officer, c_name_fio, sum(k_kolich_client_approved), sum(k_amount_approved)
      from kontract, kontstate, knaktlist, region r1, region r2, clients, region_list(:rep_region_id, 1, -1, 1)
      where k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_state = 1 and
            knaktlist.is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= ks_date_begin and
            kal_date_end >= ks_date_begin and
            knaktlist.kal_new_reg_id = r1.reg_unicode and
            r2.reg_unicode = r1.parent and
            c_id = kal_new_kf_officer and
            kal_new_reg_id = id_list
      group by kal_new_kf_officer, c_name_fio
      into :aa_curr_kc_id, :aa_curr_kc_fio,

           :aa_kolich_cl_disb, :aa_amount_disb
  do
  begin
    suspend;
  end
  aa_kolich_cl_disb = 0;
  aa_amount_disb = 0;

  for select kal_new_kf_officer, c_name_fio, sum(k_kolich_client_approved)
      from kontract, kontstate, knaktlist, region r1, region r2, clients, region_list(:rep_region_id, 1, -1, 1)
      where k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_state = 2 and
            knaktlist.is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            knaktlist.kal_new_reg_id = r1.reg_unicode and
            r2.reg_unicode = r1.parent and
            c_id = kal_new_kf_officer and
            kal_new_reg_id = id_list and
            k_disb_date >= '01.03.2012' and
            k_disb_date <= '31.03.2012'
      group by kal_new_kf_officer, c_name_fio
      into :aa_curr_kc_id, :aa_curr_kc_fio, :aa_kolich_cl_close
  do
  begin
    suspend;
  end
end
