CREATE PROCEDURE HP_DO_EXPRESSAPP_ACTION(
    EXA_ID D_INTEGER NOT NULL,
    ACTION_CODE D_STR_CODE NOT NULL)
RETURNS (
    RES_STATUS D_SMALLINT)
AS
declare variable DB_ID D_ID;
declare variable UN_ID D_ID;
begin
  -- Процедура выполняет действие переданное как параметр

  -- res_status может быть:
  --  -1 ошибка
  --   0 действия завершено, идем дальше
  --   1 действия не совершено, ждем
  db_id = f_get_db_id();
  if (action_code not in ('new',
                          'start',            -- готово
                          'check',            -- готово
                          'miniapp.create',   -- готово
                          'bankcard',         -- готово
                          'loan.create',      -- готово
                          'kk.send',          -- готово
                          'extinfo.ready',
                          'kk.wait',
                          'loan.approve',
                          'loan.disb',
                          'loan.active',
                          'send_to_kib',
                          'data_check',
                          'application_send',
                          'loan_send',
                          'consult_send',
                          'do_disbursement',
                          'finish',
                          'loan.next')) then
    exception e_default f_format('Unknown action_code: %1', action_code);

  -- Здесь должен быть код для каждой вышеуказанной команды.
  -- Перед выполнением команды, сначала нужно проверять, может этот этап уже пройден.
  -- Например, перед созданием Miniapp, нужно проверить, может Miniapp уже создан
  if (action_code = 'start') then
  begin
    res_status = 0;
    suspend;
    exit;
  end 

  if (action_code = 'finish') then
  begin
    res_status = 0;
    suspend;
    exit;
  end 

  -- первоначальная проверка, клиент попадает по текущую акцию
  if (action_code = 'check') then
  begin
    select res_status from hp_do_expressapp_action_check(:exa_id)
    into res_status;

    suspend;
    exit;
  end
  -- Создание новой заявки
  if (action_code = 'miniapp.create') then
  begin
    select res_status from hp_do_expressapp_action_macreat(:exa_id)
    into res_status;
    suspend;
    exit;
  end
  if (db_id in (39801, 41701) and action_code = 'data_check') then
  begin
    if ((select exa.exa_date
        from expressapp exa
        where exa.exa_id = :exa_id) < '10.04.2022') then
    begin
      res_status = 0;
      suspend;
      exit;
    end 
    else
    begin
      res_status = 1;
      suspend;
      exit;
    end 
  end 
  --Отправка эл. заявки ВИПСом
  if (db_id in (39801, 41701) and action_code = 'application_send') then
  begin
    res_status = 1;
    suspend;
    exit;
  end
  -- Отправка заявки в киб
  if (db_id in (39801, 41701) and action_code = 'send_to_kib') then
  begin
    select res_status from hp_do_expressapp_act_send_kib(:exa_id)
    into res_status;
    suspend;
    exit;
  end 
  -- проверка банковской карты или создание эл. документа для закрепление банковской карты
  if (action_code = 'bankcard') then
  begin
    select res_status from hp_do_expressapp_action_bankcar(:exa_id)
    into res_status;
    suspend;
    exit;
  end
  -- создание кредитного договора
  if (action_code = 'loan.create') then
  begin
    select res_status from hp_do_expressapp_action_loancr(:exa_id)
    into res_status;
    suspend;
    exit;
  end
  -- отправка на КК
  if (action_code = 'kk.send') then
  begin

    if (db_id in (39801, 41701)) then
    begin
      res_status = 0;
      suspend;
      exit;
    end
     
    select res_status from hp_do_expressapp_action_kksend(:exa_id)
    into res_status;
    suspend;
    exit;
  end
  -- Ожидание ответа от внешних источников
  if (action_code = 'extinfo.ready') then
  begin
    select wait_status from hp_do_expressapp_action_extinfo(:exa_id)
    into res_status;
    suspend;
    exit;
  end
  -- Ожидание одобрения КК
  if (action_code = 'kk.wait') then
  begin
    if (db_id in (39801, 41701)) then
    begin
      res_status = 0;
      suspend;
      exit;
    end

    select wait_status from hp_do_expressapp_action_kkwait(:exa_id)
    into res_status;
    suspend;
    exit;
  end
  -- Займ утвержден
  if (action_code = 'loan.approve') then
  begin
    select wait_status from hp_do_expressapp_action_approve(:exa_id)
    into res_status;
    suspend;
    exit;
  end
  -- Отправка эл. договора ВИПСом
  if (db_id in (39801, 41701) and action_code = 'loan_send') then
  begin
    res_status = 1;
    suspend;
    exit;
  end
  -- Отправка договора на оказание услуг ВИПСом
  if (db_id in (39801, 41701) and action_code = 'consult_send') then
  begin
    res_status = 1;
    suspend;
    exit;
  end 
  -- Отправлен на выдачу
  if (action_code = 'loan.disb') then
  begin
    select wait_status from hp_do_expressapp_action_disb(:exa_id)
    into res_status;
    suspend;
    exit;
  end
  -- Проверка Займ выдан
  if (action_code = 'loan.active') then
  begin
    select wait_status from hp_do_expressapp_action_active(:exa_id)
    into res_status;
    suspend;
    exit;
  end
  if (db_id in (39801, 41701) and action_code = 'do_disbursement') then
  begin
    select wait_status
    from hp_expressapp_do_disbursement(:exa_id)
    into res_status;
    suspend;
    exit;
  end 

  if (action_code = 'loan.next') then
  begin
    select wait_status
    from hp_expressapp_loan_next(:exa_id)
    into res_status;
    suspend;
    exit;
  end 

  res_status = -1;
  suspend;
end
