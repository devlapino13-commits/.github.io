CREATE PROCEDURE HP_ONLINE_PASS_SCORE
RETURNS (
    OPS_ID D_ID,
    OPS_DATE_BEGIN D_DATE,
    OPS_MIN_POINT D_INTEGER)
AS
begin
  -- процедура возвращает на текущий день минимальный балл клиента, необходимый для онлайн займа (RFCKG-2253)
  -- если на текущий день минимальный балл клиента не задан, то возвращается 100, т.е. по умолчанию
  -- доступ к онлайн займу открыт только клиентам, у которых 100 баллов и больше

  select ops_id,
         ops_date_begin,
         ops_min_point
  from online_passing_score
  where current_date between ops_date_begin and ops_date_end
  into ops_id,
       ops_date_begin, -- дата начала действия
       ops_min_point; -- текущий минимальный проходной балл клиента

  if (ops_min_point is null) then
    ops_min_point = 100;

  suspend;
end
