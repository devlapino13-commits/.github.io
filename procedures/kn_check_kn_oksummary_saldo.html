CREATE PROCEDURE KN_CHECK_KN_OKSUMMARY_SALDO
RETURNS (
    SQL_STR D_STR_LONG)
AS
declare variable bal_perc curr_type;
declare variable bal_kommis curr_type;
declare variable balance curr_type;
declare variable prv_bal_perc curr_type;
declare variable prv_bal_kommis curr_type;
declare variable prv_balance curr_type;
declare variable orig_bal_perc curr_type;
declare variable orig_bal_kommis curr_type;
declare variable orig_balance curr_type;
declare variable prv_kc_id id_type;
declare variable k_id decimal(18,0);
declare variable k_id_prior id_type;
begin
  k_id_prior = 0;
  for select oks_b.oks_from_k_id
      from kn_oksummary oks_b, kn_oksummary oks_e
      where oks_b.oks_from_kc_id = oks_e.oks_from_kc_id and
            oks_b.oks_date_begin <> oks_b.oks_date_prv and
            oks_b.oks_date_prv = oks_e.oks_date_begin and
            (oks_b.oks_bal_perc_prv <> oks_e.oks_bal_perc or
            oks_b.oks_bal_kommis_prv <> oks_e.oks_bal_kommis or
            oks_b.oks_balance_prv <> oks_e.oks_balance)
      into k_id
  do
  begin
    if (k_id <> k_id_prior) then
    begin
      sql_str = 'execute procedure hp_fill_kn_oksummary(' || k_id || ', 1);';
      suspend;
    end

    k_id_prior = k_id;
  end
end
