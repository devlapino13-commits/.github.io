CREATE PROCEDURE HP_SET_CARDLIST_STATUSREG(
    CARD_LIST D_BLOB_TEXT)
RETURNS (
    RES_CARD_ID D_INTEGER,
    ERR_CODE D_INTEGER)
AS
declare variable card_id d_integer;
declare variable pan_short d_str_20;
declare variable answer_json d_str_8100;
declare variable expire_period d_integer;
declare variable i d_integer;
declare variable n d_integer;
begin
  i = 0;
  n = f_json_blob_get_count(:card_list, 'cards');

  while (i < n) do
  begin
    card_id = f_json_str_get_value(:card_list, 'cards[' || :i || '].id', null);
    answer_json = f_json_str_get_value(:card_list, 'cards[' || :i || ']', null);

    select
      res_card_id,
      err_code
    from hp_set_card_statusreg(:card_id, :answer_json)
    into
      :res_card_id,
      :err_code;
    suspend;

    i = i + 1;
  end
end
