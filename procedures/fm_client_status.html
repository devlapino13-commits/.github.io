CREATE PROCEDURE FM_CLIENT_STATUS(
    UN_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0))
RETURNS (
    ACCESS_LEV INTEGER,
    IS_ACTIVE_CLIENT INTEGER,
    IS_EMPLOYEE INTEGER)
AS
declare variable USER_C_ID decimal(18,0);
declare variable C_ID_TMP decimal(18,0);
declare variable HQ_REG_ID integer;
declare variable REG_ID integer;
declare variable LAST_KC_ID decimal(18,0);
declare variable MAX_DAYS integer;
declare variable CS_DATE date;
declare variable CS_CLIENT_STATE integer;
declare variable CS_STAFF_STATE integer;
declare variable STATUS varchar(20);
declare variable HAS_REGION_ACCESS integer;
declare variable STAFF_REG_ID integer;
declare variable DB_ID integer;
declare variable DB_IS_ACTUAL d_integer;
declare variable UNIQTAB_USERNAME integer;
declare variable UNIQTAB_ROLENAME integer;
declare variable RN_NAME varchar(30);
declare variable SBL_FROM_C_ID D_ID;
declare variable IS_CURATOR D_SMALLINT;
declare variable SBL_ID D_INTEGER;
begin
  -- Возвращает статус контрагента:
  -- staff - действующий сотрудник
  -- client - чужой клиент, имеющий активный кредит
  -- Client_Own - свой клиент
  -- Client_Old - старый чужой клиент
  -- Если пустое значение, значит либо свой клиент либо свободное физ.лицо
  un_id = (select un_id
           from hp_test_un_id(:un_id));

  select un_from_c_id
  from username
  where un_id = :un_id
  into :user_c_id;

  db_id = f_get_db_id();
  db_is_actual = 0;
  if (exists
    (   
        select *
        from hp_db_is_actual
    )
  ) then
    db_is_actual = 1;

  status = '';
  reg_id = 0;
  is_active_client = 0;
  is_employee = 0;
  max_days = 7;
  if (db_id = 81001) then
    max_days = 60;

  has_region_access = 0;


  select coalesce(cs_client_state, -1), coalesce(cs_staff_state, -1), cs_date, cs_region_id, cs_lo_id
  from clientstate
  where cs_from_c_id = :c_id
  into :cs_client_state, :cs_staff_state, :cs_date, :reg_id, :last_kc_id;

  if (cs_staff_state = 1) then -- Проверка на доступ к сотрудникам
  begin
    status = 'staff';
    select ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :c_id and
          ps_date_end >= '31.12.2999' and
          ps_status = 1
    into :staff_reg_id;
  end

  -- Проверка на родственников если не действующий клиент
  if (cs_client_state = -1 or (cs_client_state in (0, 2) and
      cs_date < (current_date - max_days))) then
  begin
    for select rel_other_c_id
        from hp_client_relation_tmp(:c_id, 1)
        where char_length(rel_code) = 1 or
              rel_level = 0
        into :c_id_tmp
    do
    begin
      if (exists(select *
                 from clientstate
                 where cs_from_c_id = :c_id_tmp and
                       cs_staff_state = 1)) then
      begin
        status = 'staff_rel';
        leave;
      end
    end
  end

  if (cs_client_state <> -1) then
  begin
    -- если клиент активный или есть заявка (или закрыт) за последние MAX_DAYS дней
    if (status not in ('staff', 'staff_rel')) then
    begin
      if (cs_client_state = 1 or cs_date >= (current_date - max_days)) then
        status = 'client_not_own';
      else
      if (status <> 'staff') then
        status = 'client_old'; -- неактивный клиент (нет движения за последние MAX_DAYS дней и не сотрудник
    end

    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Superuser')
               where has_access = 1)) then
      status = 'client_own';
    else
    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Programmer')
               where has_access = 1)) then
      status = 'client_own';

    if (status <> 'client_own') then
    begin
      -- Проверка: Свой клиент, либо клиент кого заменяет текущий пользователь
      if (exists(select *
                 from hp_staff_changer_list(:un_id, 1)
                 where res_c_id = :last_kc_id)) then
      begin
        if (cs_client_state = 0 or status not in ('staff', 'staff_rel')) then
          status = 'client_own';
      end
      else
      begin
        if (exists(select *
                   from aarep_region_user_tmp(:un_id)
                   where reg_list = :reg_id)) then
        begin
          has_region_access = 1;

          -- Проверка для менеджерского доступа: клиент находится в подчиняемом офиса
          if (exists(select *
                     from hr_pstatus, joblist, jobkind
                     where ps_from_c_id = :user_c_id and
                           ps_date_begin <= current_date and
                           ps_date_end >= current_date and
                           ps_is_working = 1 and
                           ps_status = 1 and
                           ps_from_jl_id = jl_id and
                           jl_kind = jk_id and
                           (jk_system_code in (
                             'JobSystem.Region_KM', -- это поменять на переменную
                             'JobSystem.FinStart_Support',
                             'JobSystem.Region_Director',
                             'JobSystem.Director_Executive',
                             'JobSystem.Director_Islamic_Window') or    -- RFCKG-3288
                           (jk_system_code = 'JobSystem.LO_Senior' and :db_id not in (39801, 41701))))) then   -- RFCKG-379 Убран доступ к не своим клиентам для ССПФ в КГ
          begin
            if (cs_client_state = 0 or status not in ('staff', 'staff_rel')) then
              status = 'client_own';
          end

          -- RFCKG-379 Доступ на просмотр клиентов других СПФ для ССПФ завязано на роль в КГ
          if (:db_id in (39801, 41701)) then
          begin
            select ukt_id
            from hp_get_uniqtable_id('Username')
            into :uniqtab_username;

            select ukt_id
            from hp_get_uniqtable_id('Rolename')
            into :uniqtab_rolename;

            if (exists(select *
                       from permissions, rolename
                       where pm_object_kind = :uniqtab_username and
                             pm_object_id = :un_id and
                             pm_subject_kind = :uniqtab_rolename and
                             pm_is_delete = 0 and
                             (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                             (coalesce(pm_date_end, '31.12.2999') >= current_date) and
                             pm_subject_id = rn_id and
                             rn_name = 'SKS')) then
              begin
                if (cs_client_state = 0 or status not in ('staff', 'staff_rel')) then
                    status = 'client_own';
              end
           end
           --RFCKG-379 END
        end
      end
    end
  end
  if (status = 'staff_rel') then
    status = 'staff';

  access_lev = -1; -- По умолчанию доступ закрыт

  if (status = 'staff') then -- Проверка на доступ к сотрудникам
  begin
    if (cs_staff_state = 1) then
      is_employee = 1;
    if (staff_reg_id is not null) then
    begin
      if (exists(select *
                 from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Superuser')
                 where has_access = 1)) then
        access_lev = 1;
      else
      begin
        if (exists(select *
                   from aarep_region_user_tmp(:un_id)
                   where reg_list = :staff_reg_id)) then
        begin
          if (exists(select *
                     from hp_permission_access_check(:un_id, 'PermActions', 0, 'FSMobile.Staff_Access')
                     where has_access = 1)) then
          begin
            access_lev = 0;
            -- Если сотрудник работает в ГО, через мобильное приложение к нему нет доступа
            hq_reg_id = f_hq_id(db_id);
            if (staff_reg_id = hq_reg_id) then
              access_lev = -1;
          end
        end
      end
    end
  end
  else
  begin
    if (cs_client_state = 1) then
      is_active_client = 1;

    if (status = '' -- не клиент
        or status = 'client_own') -- свой клиент
    then
      access_lev = 1;

    if (status = 'client_old') --  Это чужой неактивный клиент, нужно обсудить
    then
    begin
      -- RFCRU-6006: Открываем полный доступ к неактивным клиентам всем випсам данного офиса
      if (db_id = 81001 and has_region_access = 1) then
        access_lev = 1;
      else
        access_lev = 2;
   end

    if (status = 'client_not_own') then --  Это чужой клиент, нужно обсудить
    begin
      access_lev = -1;

      if (has_region_access = 1) then
      begin
        if (exists(select *
                   from hp_permission_access_check(:un_id, 'PermActions', 0, 'FSMobile.Other_Client_Access')
                   where has_access = 1)) then
          access_lev = 0;
      end
    end
  end

  if (exists(select *
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.ViewAccessOriv')
             where has_access = 1)) then
  begin
    access_lev = -1;

    is_curator = 0;

    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.ViewAccessToAll')
               where has_access = 1)) then
      is_curator = 1;

    for select distinct sbl_from_c_id, sl.sbl_id
        from knclient knc, sb_akt sa, sb_list sl
        where kc_from_c_id = :c_id and
              sba_from_k_id = knc.kc_from_k_id and
              kc_amount_approved > 0 and
              sba_is_delete = 0 and
              sba_from_sbl_id <> 0 and
              sba_status = 2 and
               current_date between sba_date_begin and sba_date_end and
              sba_from_sbl_id = sbl_id and
              sbl_type = 1
        into :sbl_from_c_id, :sbl_id
    do
    begin
      if (sbl_id = 55 and db_id in (39801, 41701)) then -- исключаю из списка "КЦ Пантера"
          continue;

      if (sbl_from_c_id = user_c_id or is_curator = 1) then
        access_lev = 1;
    end

  end

  -- Проверить влияние access_lev при регистрации заявки: окно открыто и выбирается чужой клиент
  -- Обсудить access_lev=2 - окно открывается в сокращенном ввиде и возможность регистрации заявки

  if (db_id = 81001 and
      access_lev in (-1, 2)) then
  begin
    select ukt_id
    from hp_get_uniqtable_id('Username')
    into :uniqtab_username;
    select ukt_id
    from hp_get_uniqtable_id('Rolename')
    into :uniqtab_rolename;

    rn_name = null;
    select rn_name
    from permissions, rolename
    where pm_object_kind = :uniqtab_username and
          pm_object_id = :un_id and
          pm_subject_kind = :uniqtab_rolename and
          pm_is_delete = 0 and
          (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
          (coalesce(pm_date_end, '31.12.2999') >= current_date) and
          pm_subject_id = rn_id and
          rn_name in ('SB', 'SB_KG')
    into rn_name;
    rn_name = coalesce(rn_name, '');

    -- проверка, находится ли клиент в СБ: ДЕПАРТАМЕНТ РВ
    if (rn_name = 'SB' and
        exists(select *
               from knclient, sb_akt
               where kc_from_c_id = :c_id and
                     kc_amount_approved > 0 and
                     kc_from_k_id = sba_from_k_id and
                     current_date between sba_date_begin and sba_date_end and
                     sba_from_sbl_id = 85 -- ДЕПАРТАМЕНТ РВ
                     and
                     sba_status = 2 and
                     sba_is_delete = 0)) then
    begin
      access_lev = 0;
    end
    if (rn_name = 'SB_KG' and
        exists(select *
               from knclient, sb_akt sb_list
               where kc_from_c_id = :c_id and
                     kc_amount_approved > 0 and
                     kc_from_k_id = sba_from_k_id and
                     current_date between sba_date_begin and sba_date_end and
                     sba_from_sbl_id = 156 -- KG Назанов Марс
                     and
                     sba_status = 2 and
                     sba_is_delete = 0)) then
    begin
      access_lev = 0;
    end
  end

  -- RFCAZ-311
  if (db_id = 94401
      and db_is_actual = 0
      and un_id = 1010100000305) then -- Азер
    access_lev = 1;

  suspend;
end
