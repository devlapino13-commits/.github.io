CREATE PROCEDURE AAREP_DELINQ_KONT_STAJE_KC(
    REP_REGION_ID INTEGER,
    REP_DATE DATE,
    REP_C_ID ID_TYPE = 0)
RETURNS (
    REG_ID ID_TYPE,
    KS_KC_ID ID_TYPE,
    K_APPROVED_MANAGER ID_TYPE,
    K_ID ID_TYPE,
    SH_ID INTEGER,
    DELINQ_DATE DATE,
    KS_LO_EXPERIENCE_FULL DECIMAL(4,2),
    KF_NAME STR_100,
    AAREP_DISB_LO STR_100,
    UN_FIO_APPROVED STR_100,
    K_KONTRACT_NO STR_100,
    K_DISB_DATE DATE)
AS
declare variable TMP smallint;
begin
  if (rep_c_id is null or rep_c_id = -1) then rep_c_id = 0;
  if (not exists(select * from loanmgrlist
  where lml_from_c_id = :rep_c_id
  and :rep_date between lml_date_begin and lml_date_end)) then rep_c_id = 0;

  /* Procedure Text */
  /*
  for select oks_region_id, oks_delinq_date, oks_from_k_id
      from oksummary oks
      inner join region_list(:rep_region_id, 1, -1, 1) on (oks_region_id = id_list)
      where oks_close_state = 0 and
            :rep_date between oks_date_begin and oks_date_end and
            :rep_date - oks_delinq_date = 1
  */
  for
  select oks_region_id,
         oks_delinq_date,
         oks_from_k_id
  from oksummary, region_list(:rep_region_id, 1, -1, 1)
  where :rep_date between oks_date_begin and oks_date_end
  and :rep_date - oks_delinq_date = 1
  and oks_close_state = 0
  and oks_region_id = id_list
  and (:rep_c_id = 0 or :rep_c_id = f_lo_mgr_c_id(f_lo_mgr_id(oks_from_k_id, :rep_date, oks_kc_id,oks_region_id)))
  into :reg_id, :delinq_date, :k_id
  do
  begin
    ks_kc_id = null;
    k_kontract_no = null;
    k_disb_date = null;
    k_approved_manager = null;
    kf_name = null;
    ks_lo_experience_full = 0;
    select ks.ks_kc_id, k_kontract_no, k_disb_date, k.k_approved_manager, k.k_fact_disb_kf_no
    from kontstate ks
    inner join kontract k on (ks_from_k_id = k_id)
    where ks_from_k_id = :k_id and
          ks_ol_is_delete = 0 and
          :rep_date between ks_date_begin and ks_date_end
    into ks_kc_id, k_kontract_no, k_disb_date, k_approved_manager, kf_name;

    select lo_stage_month_h
    from hp_get_lo_stage(:ks_kc_id, :k_disb_date)
    into ks_lo_experience_full;

    if (coalesce(ks_lo_experience_full, 0) <= 12) then
    begin
      aarep_disb_lo = null;
      select c_name_fio
      from clients c
      where c.c_id = :ks_kc_id
      into aarep_disb_lo;

      un_fio_approved = null;
      select c_name_fio
      from clients c
      where c.c_id = :k_approved_manager
      into un_fio_approved;

      sh_id = null;
      select sh_id
      from schedule sh
      where sh_from_k_id = :k_id and
            sh_date = :delinq_date
      into sh_id;

      if (sh_id <= 3) then
      begin
        tmp = 0;
        suspend;
      end
    end
  end
  if (tmp is null) then
  begin
    kf_name = 'Нет данных';
    k_approved_manager = null;
    k_id = null;
    sh_id = null;
    delinq_date = null;
    ks_lo_experience_full = null;
    aarep_disb_lo = null;
    un_fio_approved = null;
    k_kontract_no = null;
    k_disb_date = null;
    suspend;
  end
end
