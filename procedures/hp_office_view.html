CREATE PROCEDURE HP_OFFICE_VIEW
RETURNS (
    REG_ID D_INTEGER,
    REG_CODE D_STR_10,
    REG_NAME D_STR_CODE,
    IS_LINK D_BOOL)
AS
declare variable dt_curr d_date;
declare variable object_kind d_integer;
begin
  dt_curr = current_date;
  object_kind = f_objkind('region', 'id');
  for
    select
      r.reg_unicode,
      r.reg_code,
      r.reg_name
    from region as r
    where r.reg_status = 3
      and r.is_delete = 0
      and current_date between coalesce(r.reg_date_open, '01.01.1900')
        and coalesce(r.reg_date_close, '31.12.9999')
    order by r.reg_code
  into
    :reg_id,
    :reg_code,
    :reg_name
  do
  begin
    if (exists(
      select * from ipc_terminal_ownership
      where ito_object_id = :reg_id
        and ito_object_kind = :object_kind
        and :dt_curr between ito_date_begin and ito_date_end
        and ito_status > -1))
    then is_link = 1;
      else is_link = 0;
    suspend;
  end
end
