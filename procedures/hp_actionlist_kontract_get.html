CREATE PROCEDURE HP_ACTIONLIST_KONTRACT_GET(
    K_ID DECIMAL(18,0))
RETURNS (
    ACL_ID DECIMAL(18,0),
    ACL_STATUS INTEGER,
    ACL_DATETIME TIMESTAMP)
AS
declare variable RD_CODE D_STR_100;
declare variable DB_ID D_INTEGER;
begin
  db_id = f_get_db_id();

  if (db_id = 39801) then  -- RFCKZ-154
  begin
    acl_status = -4;
    exit;
  end

  select rd_code
  from hp_get_check_loan_data(:k_id)
  into :rd_code;

  -- Получает значение проверки кредита
  select acl_id, acl_status, acl_datetime
  from hp_actionlist_get_status('Kontract', :k_id, :rd_code)
  into :acl_id, :acl_status, acl_datetime;
  if (acl_datetime is null) then
    acl_status = -3;
  suspend;
end
