CREATE PROCEDURE CM_KG_0122_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable c_id d_id;
declare variable k_id_last d_id;
declare variable kc_max_amount_approved_last d_currency;
declare variable pr_code d_str_100;
declare variable kc_circle d_integer;
declare variable prl_id d_integer;
declare variable prl_amount_cl_max d_currency;
begin
  res_error_is_critical = 2;
  select k.k_disb_date, pr_code, max(pl.prl_amount_cl_max)
  from kontract k
  inner join product pr on pr.pr_id = k.k_from_pr_id
  inner join prodlogic pl on pr_id = pl.prl_from_pr_id
  where k_id = :k_id and
        k_add_month = 0 and
        k_refund = 0
  group by 1, 2
  into :kf_disb_date, pr_code, prl_amount_cl_max;

  if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
   exit;

  if (pr_code containing '.IPF.') then --и не распространяется на ИПФ
  begin
    exit;
  end
  else
  begin
    for select knc.kc_from_c_id, knc.kc_amount_approved, cl.c_name_fio, kc_circle
        from knclient knc
        inner join clients cl on cl.c_id = knc.kc_from_c_id
        where knc.kc_from_k_id = :k_id and
              knc.kc_amount_approved > 0 and
              knc.kc_circle > 1
        into :c_id, :kc_amount, :res_info_text, kc_circle
    do
    begin
      if (kc_circle < 4) then
      begin
        kc_max_amount_approved_last = 0;
        select max(knc.kc_amount_approved) as kc_max_amount_approved_last
        from kontstate ks
        inner join knclient knc on ks.ks_from_k_id = knc.kc_from_k_id and
              knc.kc_from_c_id = :c_id and
              knc.kc_amount_approved > 0
        where ks.ks_state = 1 and
              ks.ks_ol_is_delete = 0 and
              ks.ks_date_begin <= :kf_disb_date and
              ks.ks_from_k_id <> :k_id
        into :kc_max_amount_approved_last;
        kc_max_amount_approved_last = coalesce(kc_max_amount_approved_last, 0);
        if (kc_max_amount_approved_last <= 50000) then
          kc_max_amount_approved_last = 50000;
        if (kc_amount > kc_max_amount_approved_last) then --Если у клиента в истории были кредиты не более 50 000 сом, то он может взять кредит не более 50 000 сом.
        begin
          res_error_text = 'Неактивным клиентам МБ сумма выдачи не может быть больше чем макс. сумма выдачи в закрытых кредитах.';
          suspend;
        end
      end
      else
      begin
        if (kc_amount > prl_amount_cl_max) then --Данные условия не распространяются на клиентов/ГСО с 4-го цикла и выше и не распространяется на ИПФ
        begin
          res_error_text = 'Сумма клиента не может быть больше: ' || round(prl_amount_cl_max);
          suspend;
        end
      end
    end
  end
end
