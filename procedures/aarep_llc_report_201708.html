CREATE PROCEDURE AAREP_LLC_REPORT_201708(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_IS_PRIOR INTEGER)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    C_NAME_FIO VARCHAR(100),
    K_DISB_DATE DATE,
    LK_IS_SECURED_STR VARCHAR(5),
    LK_IS_RESTRUCT_STR VARCHAR(5),
    LK_DELIQ_DAYS INTEGER,
    PERCENT_RVPZ PERC_TYPE,
    RPPU_BALANCE CURR_TYPE,
    RPPU_CALCPERC CURR_TYPE,
    REG_ID INTEGER,
    KC_AMOUNT DECIMAL(18,4),
    PERIOD_DAY INTEGER,
    LK_BALANCE_CLI CURR_TYPE,
    LK_CALCPERC_CLI CURR_TYPE,
    GROUP_NUMBER INTEGER,
    IS_RESIDENT VARCHAR(10))
AS
declare variable k_id decimal(18,0);
declare variable lk_is_secured integer;
declare variable lk_is_restruct integer;
declare variable k_amount_approved curr_type;
declare variable dolya double precision;
declare variable c_id id_type;
declare variable sh_close_date date;
declare variable lk_balance decimal(18,4);
declare variable lk_calcperc decimal(18,4);
declare variable kol_cli integer;
declare variable k integer;
declare variable balance_remain curr_type;
declare variable calcperc_remain curr_type;
declare variable citizen_ru_id integer;
declare variable doc_vnj_id integer;
declare variable res_client_is_resident smallint;
declare variable report_old smallint;
declare variable pr_id integer;
declare variable pr_code varchar(100);
declare variable pr_loan_type integer;
declare variable pr_product_kind integer;
begin
  citizen_ru_id = null;
  select rd_id
  from refdata
  where rd_code = 'Citizen.RU'
  into :citizen_ru_id;

  doc_vnj_id = null;
  select pk_id
  from passkind, passdoctype
  where pdt_id = pk_from_pdt_id and
        pk_is_delete = 0 and
        pdt_code = 'registration_vnj' rows 1
  into :doc_vnj_id;

  if (rep_date_end < '31.08.2017') then
    report_old = 1;
  else
    report_old = 0;

  for select ks_from_k_id, ks_date_begin, lk_is_secured, lk_is_restruct, lk_balance, lk_calcperc, lk_deliq_days,
             id_list
      from llc_main, llc_kont, kontstate, region_list(:rep_region_id, 1, 3, 1)
      where lm_is_signed = 1 and
            lm_id = lk_from_lm_id and
            :rep_date_end between lk_date and lk_date_end and
            lk_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            (:rep_is_prior = 1 and
            ks_date_begin < '28.02.2012' or
            :rep_is_prior = 0 and
            ks_date_begin >= '28.02.2012') and
            lk_from_reg_id = id_list
      into k_id, k_disb_date, lk_is_secured, lk_is_restruct, lk_balance, lk_calcperc, lk_deliq_days, reg_id
  do
  begin
    select k_kontract_no, k_amount_approved, k_kolich_client_approved, k_from_pr_id
    from kontract
    where k_id = :k_id
    into k_kontract_no, k_amount_approved, kol_cli, pr_id;

    select max(sh_date)
    from schedule
    where sh_from_k_id = :k_id
    into sh_close_date;

    period_day = sh_close_date - k_disb_date;

    if (:report_old = 1 and
        :rep_is_prior = 1) then
    begin
      select pr_code, pr_loan_type, pr_product_kind
      from product
      where pr_id = :pr_id
      into pr_code, pr_loan_type, pr_product_kind;

      select is_secured
      from aarep_get_secured_loan(:pr_code, :pr_loan_type, :pr_product_kind, :pr_id, :k_disb_date, :kol_cli, :k_id)
      into lk_is_secured;
    end

    if (lk_is_secured is null) then
      lk_is_secured = 0;
    if (lk_is_restruct is null) then
      lk_is_restruct = 0;

    if (lk_is_secured = 1) then
      lk_is_secured_str = 'Да';
    else
      lk_is_secured_str = 'Нет';

    if (lk_is_restruct = 1) then
      lk_is_restruct_str = 'Да';
    else
      lk_is_restruct_str = 'Нет';

    k = 0;
    balance_remain = lk_balance;
    calcperc_remain = lk_calcperc;
    for select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id, kc_amount
    do
    begin
      dolya = cast(kc_amount as double precision) / k_amount_approved;

      select c_name_fio
      from clients
      where c_id = :c_id
      into c_name_fio;
      c_name_fio = '№' || c_id;

      select 1 - res_non_resident
      from hp_get_client_type(:c_id, :citizen_ru_id, :doc_vnj_id, :rep_date_end)
      into :res_client_is_resident;

      if (res_client_is_resident = 1) then
        is_resident = 'Да';
      else
        is_resident = 'Нет';

      select group_number, percent_rvpz
      from hp_get_rvpz(:kc_amount, :period_day, :lk_is_secured, :lk_is_restruct, :lk_deliq_days)
      into group_number, percent_rvpz;

      k = k + 1;

      if (k < kol_cli) then
      begin
        lk_balance_cli = dolya * lk_balance;
        lk_calcperc_cli = dolya * lk_calcperc;

        balance_remain = balance_remain - lk_balance_cli;
        calcperc_remain = calcperc_remain - lk_calcperc_cli;
      end
      else
      begin
        lk_balance_cli = balance_remain;
        lk_calcperc_cli = calcperc_remain;
      end

      rppu_balance = cast(lk_balance_cli as double precision) * percent_rvpz / 100;
      rppu_calcperc = cast(lk_calcperc_cli as double precision) * percent_rvpz / 100;

      suspend;
    end
  end
end
