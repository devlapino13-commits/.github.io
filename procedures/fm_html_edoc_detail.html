CREATE PROCEDURE FM_HTML_EDOC_DETAIL(
    EDL_ID BIGINT,
    UN_ID BIGINT,
    IS_SUBREPORT SMALLINT)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable LANG D_STR_LANG;
declare variable EDS_ID D_ID;
declare variable C_NAME_FIO D_STR_100;
declare variable EDS_STATUS D_SMALLINT;
declare variable LINK_SMS D_STR_LONG;
declare variable LINK_PRINT D_STR_LONG;
declare variable LINK_SCAN D_STR_255;
declare variable LINK_SEND_CHECK D_STR_255;
declare variable TMP_STR D_STR_100;
declare variable CELL_NO D_STR_CODE;
declare variable EDB_STATUS_STR D_STR_CODE;
declare variable EDBS_STATUS_STR D_STR_255;
declare variable EDB_ID D_ID;
declare variable EDB_STATUS D_SMALLINT;
declare variable EDBS_STATUS D_SMALLINT;
declare variable EDF_PRINT_NAME D_STR_255;
declare variable CAN_PRINT D_SMALLINT;
declare variable EDBS_FROM_RD_ID D_INTEGER;
declare variable EDBS_FROM_RD_NAME D_STR_255;
declare variable EDC_CODE D_STR_CODE;
declare variable DB_ID D_INTEGER;
declare variable C_ID D_ID;
declare variable EDF_FROM_EDC_ID D_ID;
declare variable EDBS_FROM_BLS_ID D_ID;
begin
  if (is_subreport = 0) then
  begin
    -- Вызываем fm_html_edoc_list для отображения заголовка
    -- эта процедура будет вызвана с параметром is_subreport =1
    -- Для этого передаем edl_id текущего документа
    select res_text
    from fm_html_edoc_list('edoc_list', :edl_id, :un_id)
    into res_text;
    suspend;
    exit;
  end

  -- Клиенты
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);
  res_text = f_format(
    '<br><table width="100%"><col width="60%"><col width="40%">'||
    '<tr><th>%1</th><th>%2</th></tr>',
    f_lang(lang, 'Клиенты'),
    f_lang(lang, 'Действия'));

  for select eds_id, c_name_fio, eds_status, edc_code, c_id
  from edoc_sign
    inner join clients on eds_from_c_id = c_id
    inner join edoc_list on eds_from_edl_id = edl_id
    inner join edoc_cfg on edl_from_edc_id = edc_id
  where eds_from_edl_id = :edl_id
  into eds_id, c_name_fio, eds_status, edc_code, c_id
  do
  begin
    -- Если не подписан, создаем ссылку для повторной отправки СМС
    link_sms = '';
    if (eds_status = 0 and coalesce(edc_code, '') <> 'pretrial_claim') then
    begin
      link_sms = f_uri_sql(f_lang(lang, 'Повторный СМС'), 'edoc_msg_ins',
        'edoc_list', edl_id, 'none', c_name_fio,
        f_lang(lang, 'Подтвердите повторную отправку СМС'),
        null, null, null, null, 'dialog', :eds_id);
    end

    -- Формируем статус подписания
    if (eds_status = 1) then
      tmp_str = f_lang(lang, 'Подписан');
    else
      tmp_str = f_lang(lang, 'СМС отправлен');
  
    -- Получаем номер клиента
    cell_no = '';
    select msg_mail_to
    from edoc_msg, fsrobot_msgbox
    where edm_from_eds_id = :eds_id and
          edm_from_msg_id = msg_id
          and msg_type in (1, 3)
    order by msg_datetime desc
    rows 1
    into cell_no;

    res_text = BLOB_APPEND(res_text,
      f_format('<tr><td class="filled" colspan="2"><b>%1</b></td></tr>', c_name_fio) ||
      f_format('<tr><td>%1: <b>%2</b><br>%3: <b>%4</b></td>',
        f_lang(lang, 'Статус'), tmp_str,
        f_lang(lang, 'Номер'), f_uri_phone(cell_no, 0, c_id)) ||
      f_format('<td align="center">%1</td></tr>', link_sms));
  end

  -- Бланки
  res_text = BLOB_APPEND(res_text, f_format(
    '</table><br><table width="100%"><col width="60%"><col width="40%">' ||
    '<tr><th>%1</th><th>%2</th>',
    f_lang(lang, 'Бланки документа'),
    f_lang(lang, 'Действия')));

  for select edb.edb_id,
             edf.edf_print_name,
             c.c_name_fio,
             edb.edb_status,
             edbs.edbs_status,
             iif(edb.edb_from_bls_id is not null or edb.edb_blob is not null , 1, 0),
             edbs.edbs_from_rd_id,
             edf.edf_from_edc_id,
             edbs.edbs_from_bls_id
  from edoc_blank edb
  join edoc_frx edf on edb.edb_from_edf_id = edf.edf_id
  left join edoc_blank_scan edbs on edbs.edbs_from_edb_id = edb.edb_id
  left join clients c on edb.edb_from_c_id = c.c_id
  where edb.edb_from_edl_id = :edl_id
  into :edb_id,
       :edf_print_name,
       :c_name_fio,
       :edb_status,
       :edbs_status,
       :can_print,
       :edbs_from_rd_id,
       :edf_from_edc_id,
       :edbs_from_bls_id
  do
  begin
    c_name_fio = coalesce(c_name_fio, '');
    link_scan = '';
    link_print = '';
    link_send_check = '';
    edbs_status_str = '';

    edb_status_str =
      case edb_status
        when 0 then f_lang(lang, 'Готовится')
        when 1 then f_lang(lang, 'Готов')
        when -1 then f_lang(lang, 'Ошибка')
        else ''
      end;

    if (edbs_status is not null) then
    begin
      if (edbs_from_bls_id > 0 and edbs_status = 0) then
        edbs_status_str = f_lang(lang, 'Прикреплен');
      else
        edbs_status_str = case edbs_status
                            when 0 then f_lang(lang, 'Ожидает закрепления')
                            when 1 then f_lang(lang, 'На проверке')
                            when 2 then f_lang(lang, 'Документ одобрен')
                            when -1 then f_lang(lang, 'Доработка')
                            else ''
                          end;

      edbs_from_rd_name = '';
      if (edbs_status = -1 and edbs_from_rd_id is not null) then
      begin
        select rd_name
        from refdata
        where rd_id = :edbs_from_rd_id
        into edbs_from_rd_name;
        edbs_status_str = edbs_status_str || ': ' || edbs_from_rd_name;
      end

      if (edbs_status in (-1, 0)) then
      begin
        link_scan = f_uri_open(f_lang(lang, 'Закрепить скан с подписью'),
          'blobpdfdoc', f_objkind('edoc_blank', 'id'), edb_id, edf_print_name,
          'can_load=1&save_kind=edoc_blank_scan', un_id);
      end

      if (db_id = 81001 and f_edoc_kind(:edf_from_edc_id, 'code') = 'cell.application') then
      begin
        if (f_get_jobkind_code(:un_id) = 'JobSystem.Region_KM' and edbs_status not in (1, 2)) then
        link_send_check = f_uri_sql('Отправить на проверку', 'cell.application_send_check', '', :edb_id, 'none', 'Отправить на проверку?',
                    '', 0, 0, '', '', 'dialog', '', 0);
        else link_scan = '';
      end
    end

    if (db_id in (39801, 41701)) then
      can_print = iif (exists(
        select has_access
        from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_print_edoc')
        where has_access = 1), 1, 0);

    if (can_print = 1) then
    begin
      link_print = f_uri_blank(f_lang(lang, 'Печать'), 'print_pdf',
        f_objkind('edoc_blank', 'id'), edb_id, 0, f_lang(lang, 'Печать'),
        0, 0, '');
    end

    res_text = BLOB_APPEND(res_text,
      f_format('<tr><td class="filled" colspan="2"><b>%1</b></td></tr><tr><td>', edf_print_name)||
      iif(c_name_fio <> '', f_format('%1: %2<br>', f_lang(lang, 'Клиент'), c_name_fio), '') ||
      iif(edb_status_str <> '', f_format('%1: <b>%2</b><br>', f_lang(lang, 'Статус бланка'), edb_status_str), '') ||
      iif(edbs_status_str <> '', f_format('%1: <b>%2</b><br>', f_lang(lang, 'Статус скана'), edbs_status_str), '') ||
      '</td><td>');

    if (link_print <> '' or link_scan <> '') then
    begin
      res_text = BLOB_APPEND(res_text,
      iif(link_scan <> '', link_scan, '') ||
      iif(link_print <> '' and link_scan <> '', '<br><br>', '') ||
      iif(link_print <> '', link_print, '') ||
      iif(link_send_check <> '' and link_scan <> '', '<br><br>', '') ||
      iif(link_send_check <> '' and link_print <> '', '<br><br>', '') ||
      iif(link_send_check <> '', link_send_check, ''));
    end
    res_text = BLOB_APPEND(res_text, '</td></tr>');
  end
  res_text = BLOB_APPEND(res_text, '</table>');

  suspend;
end
