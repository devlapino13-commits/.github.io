CREATE PROCEDURE AAREP_TOP_OFFISE(
    REP_DATE D_DATE)
RETURNS (
    C_NAME_FIO_ALL VARCHAR(512),
    CELL_NO D_STR_100,
    REG_NAME D_STR_100,
    PLACE D_STR_100,
    K_KOL DECIMAL(18,2),
    DATE_BEGIN DATE,
    DATE_END DATE)
AS
declare variable kc_id d_id;
declare variable reg_id d_integer;
declare variable kol d_integer;
declare variable svz_kol d_currency;
declare variable otpusk_days d_smallint;
declare variable n d_smallint;
declare variable dd_tmp date;
declare variable lo_work_days d_smallint;
begin
  --отчет будет формироваться в субботу, за отчетный период прошлая суббота - текущая пятница

  --ФИО менеджера, город, ср кол-во выдач на 1 випс.
  date_begin = dateadd(-7 day to :rep_date); --прошлая суббота
  date_end = dateadd(-1 day to :rep_date); --текущая пятница

  for select reg.reg_unicode, reg.reg_name, pl.pl_name
      from region reg
      inner join addrlist al on adl_id = reg_address_fact
      inner join places pl on al.adl_from_pl_id = pl.pl_id
      where reg.reg_status = 3 and
            reg.is_delete = 0 and
            current_date <= coalesce(reg.reg_date_close, '31.12.9999') and
            reg.reg_unicode not in (1012100, 1012300, 1010100, 1012700)
      into :reg_id, reg_name, place
  do
  begin
    k_kol = 0;
    svz_kol = 0;
    c_name_fio_all = '';
    for select hps.ps_from_c_id
        from hr_pstatus hps
        inner join joblist jl on jl.jl_id = ps_from_jl_id
        inner join jobkind jk on jk.jk_id = jl.jl_kind and
              jk.jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior')
        where :rep_date between ps_date_begin and ps_date_end and
              ps_is_working = 1 and
              hps.ps_status = 1 and
              hps.ps_from_reg_id = :reg_id
        into :kc_id
    do
    begin
      -- Общее количество СВЗ, за исключением сотрудников в отпуске
      -- Проверяем статус отпуска на каждый день недели:
      -- если все 7 дней был в отпуске, исключаем из списка
      otpusk_days = 0;
      n = 0;
      while (n <= 6) do
      begin
        dd_tmp = dateadd(day, n, date_begin);
        if (exists(select *
                   from hr_voctravel, hr_orders, sa_worktype
                   where hrv_from_ho_id = ho_id and
                         ho_order_type = sawt_id and
                         hrv_from_c_id = :kc_id and
                         sawt_base_type in (4, 5) and -- отпуск, декретный отпуск
                         sawt_id <> 20 and
                         sawt_hours_real >= 0 and
                         hrv_date_begin <= :dd_tmp and
                         hrv_date_end_fact >= :dd_tmp and
                         hrv_status = 1)) then
          otpusk_days = otpusk_days + 1;
        n = n + 1;
      end
      lo_work_days = 7 - otpusk_days;

      if (lo_work_days > 0) then
        svz_kol = svz_kol + 1;

    end

    if (svz_kol > 0) then
    begin
      select list(cl.c_name_fio), list(f_cont_info('Clients', c_id, 'cell_main', 'ok'),',')
      from hr_pstatus hps
      inner join clients cl on cl.c_id = hps.ps_from_c_id
      inner join joblist jl on jl.jl_id = ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where :rep_date between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            hps.ps_status = 1 and
            jk.jk_system_code containing 'JobSystem.Region_KM' and
            hps.ps_from_reg_id = :reg_id
      into :c_name_fio_all, cell_no; --находим МО офиса

      select coalesce(count(distinct k.k_id), 0)
      from kontstate, kontract k
      where ks_from_k_id = k_id and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_date_begin between :date_begin and :date_end and
            k.k_last_reg_id = :reg_id
      into :k_kol; -- количество выдач
      k_kol = k_kol / svz_kol;
      suspend;
    end
  end
end
