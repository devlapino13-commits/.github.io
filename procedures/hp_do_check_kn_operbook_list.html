CREATE PROCEDURE HP_DO_CHECK_KN_OPERBOOK_LIST(
    REP_SAB_FULL_CODE D_STR_CODE NOT NULL,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE NOT NULL)
RETURNS (
    KC_ID D_ID,
    K_ID D_ID,
    C_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    SAB_ID D_INTEGER,
    SAB_FULL_CODE D_STR_CODE,
    OKS_CLOSE_STATE D_INTEGER,
    OKS_AMOUNT D_CURRENCY,
    EPS_AMOUNT D_CURRENCY,
    ACC_ID_LIST D_STR_255)
AS
begin
  for select kc_id
  from hp_do_check_kn_operbook_tmp(:rep_date_begin, :rep_date_end)
  into :kc_id
  do
  begin
    for select k_id, c_id, sab_id, k_kontract_no, sab_full_code, oks_close_state, oks_amount, eps_amount, acc_id_list
    from hp_do_check_kn_operbook_kc(:rep_sab_full_code, :kc_id, :rep_date_begin, :rep_date_end)
    into :k_id, :c_id, :sab_id, :k_kontract_no, :sab_full_code, :oks_close_state, :oks_amount, :eps_amount, :acc_id_list
    do
      suspend;
  end
end
