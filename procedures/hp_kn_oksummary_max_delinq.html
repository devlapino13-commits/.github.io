CREATE PROCEDURE HP_KN_OKSUMMARY_MAX_DELINQ(
    KC_ID D_ID,
    C_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    LOAN_COUNT D_INTEGER,
    MAX_DELINQ_DAY D_INTEGER)
AS
declare variable k_id d_id;
declare variable tmp_kc_id d_id;
declare variable kc_delinq_days d_integer;
declare variable kc_refunded_delinq_day d_integer;
declare variable oks_last_resh_delinq_before d_integer;
declare variable oks_delinq_date d_date;
declare variable oks_delinq_days d_integer;
begin
  loan_count = 0;
  max_delinq_day = 0;

  for select kc_id, kc_from_k_id, coalesce(kc_refunded_delinq_day, 0)
  from knclient, kontstate
  where ks_state = 1 and
        ks_date_begin <= :rep_date and
        ks_date_end >= :rep_date and
        ks_ol_is_delete = 0 and
        ks_from_k_id = kc_from_k_id and
        kc_from_c_id = :c_id and
        kc_id <> :kc_id and
        kc_amount_approved > 0 and
        kc_date_close >= :rep_date
  into tmp_kc_id, k_id, kc_refunded_delinq_day
  do
  begin
    if (rep_date < '01.11.2021' and kc_refunded_delinq_day > 0) then
      kc_refunded_delinq_day = kc_refunded_delinq_day + 1;

    oks_last_resh_delinq_before = 0;
    oks_delinq_days = 0;
    oks_delinq_date = null;
    kc_delinq_days = 0;

    select oks_delinq_date
    from kn_oksummary
    where oks_from_kc_id = :tmp_kc_id and
          oks_date_begin <= :rep_date
    order by oks_from_kc_id desc, oks_date_begin desc
    rows 1
    into oks_delinq_date;
    if (oks_delinq_date is not null) then
    begin
      oks_delinq_days = datediff(day, oks_delinq_date, rep_date);
      if (rep_date < '01.11.2021') then
        oks_delinq_days = oks_delinq_days + 1;

      select or_delinq_day_before
      from operrest
      where or_from_k_id = :k_id and
            or_llr_depends = 1 and
            or_ol_is_active = 1 and
            or_ol_date <= :rep_date
      order by or_ol_date desc
      rows 1
      into oks_last_resh_delinq_before;

      kc_delinq_days = oks_delinq_days + kc_refunded_delinq_day + oks_last_resh_delinq_before;

      if (max_delinq_day < kc_delinq_days) then
        max_delinq_day = kc_delinq_days;
    end

    loan_count = loan_count + 1;
  end
  suspend;
end
