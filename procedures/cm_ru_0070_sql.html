CREATE PROCEDURE CM_RU_0070_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable c_id id_type;
declare variable err integer;
declare variable rd_id integer;
declare variable adl_id id_type;
declare variable cc_name str_100;
declare variable cc_status str_100;
declare variable pr_code d_str_100;
declare variable rd_code d_str_100;
begin
  res_error_is_critical = 1;
  for select kncl.kc_from_c_id, coalesce(cont.cc_name, 'ФИО не указано'), cont.cc_from_adl_id,
             coalesce(cont.cc_status, ''), p.pr_code
      from knclient kncl
      inner join contacts cont on cont.cc_from_c_id = kncl.kc_from_c_id
      inner join kontract k on k.k_id = kncl.kc_from_k_id
      inner join product p on p.pr_id = k.k_from_pr_id and
            kncl.kc_amount_approved > 0 and
            kncl.kc_from_k_id = :k_id and
            cont.cc_is_old_contact = 0 and
            cont.cc_type = 1 -- родина
      into :c_id, :cc_name, :adl_id, :cc_status, pr_code
  do
  begin
    err = 0;
    if (adl_id is null) then
      err = 1;
    else
    if (right(adl_id, 9) = '000000000') then
      err = 1;
    if (err = 1 and
        pr_code not containing  '.universal.' and pr_code not containing '.oivo.') then
    begin
      rd_code = '';
      select rd.rd_code
      from refdata rd
      where rd.rd_name = :cc_status and
            rd.rd_main_code = 'Relation' and
            rd.rd_main_code is not null
      into :rd_code;

      if (rd_code in ('Relation.91', 'Relation.93', 'Relation.94', 'Relation.95', 'Relation.100', 'Relation.115', 'Relation.119', 'Relation.99', 'Relation.105', 'Relation.108',
                      'Relation.103', 'Relation.104', 'Relation.125', 'Relation.126', 'Relation.128', 'Relation.129')) then
      begin
        select cl.c_name_fio
        from clients cl
        where cl.c_id = :c_id
        into :res_info_text;
        res_error_text = 'Для контактного лица ' || cc_name || ', не указан адрес';
        suspend;
      end
    end
  end
end
