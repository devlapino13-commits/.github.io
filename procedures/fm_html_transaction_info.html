CREATE PROCEDURE FM_HTML_TRANSACTION_INFO(
    OL_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable str varchar(1024);
declare variable ol_date date;
declare variable ol_doc_no integer;
declare variable ol_memo varchar(1024);
declare variable ol_document_client decimal(18,0);
declare variable ol_doc_client_name varchar(255);
declare variable ol_parent_id decimal(18,0);
begin
  res_text = '';
  for select ol_date, ol_doc_no, ol_memo, ol_document_client, ol_doc_client_name, ol_parent_id
      from operlist
      where ol_id = :ol_id
      into :ol_date, ol_doc_no, ol_memo, ol_document_client, ol_doc_client_name, ol_parent_id
  do
  begin
    with tab
    as (select iif(ok_type in (4, 9), 'Погаш:', iif(ok_type = 5, 'Списан:', 'Выдача:')) as ol_kind,
               round(sum(-ok_os - ok_perc - ok_fine - ok_kommis - ok_kommis_adv - ok_tax), 2) as ol_amount
        from operkred
        where ok_from_ol_id = :ol_id and
              ok_type in (4, 5, 8, 9)
        group by 1
        --having sum(-ok_os - ok_perc - ok_fine - ok_kommis - ok_kommis_adv - ok_tax) <> 0  -- закомментировал igusev 01.04.2021 т.к. операции типа: "Корректировка доп.%% Корректировка доп.%% ...." стали давать кучу ошибок ol_id = 1090100031042943
        )
    select list(ol_kind || ' <b>' || ol_amount, ',</b><br>')
    from tab
    into :res_text;
    res_text = BLOB_APPEND('<b>Детали операции</b><br><br>', res_text, '</b><br>');

    res_text = BLOB_APPEND(res_text, 'Дата: <b>' || right('0' || extract(day from ol_date), 2) || '.' || right('0' || extract(month from ol_date), 2) || '.' || right('0' || extract(year from ol_date), 4) || '</b><br>');

    res_text = BLOB_APPEND(res_text, 'Док.№: <b>' || coalesce(ol_doc_no, '-') || '</b><br>');

    if (ol_parent_id is not null) then
    begin
      str = '';
      select reg_code || ' ' || reg_name
      from region, operlist
      where ol_id = :ol_parent_id and
            ol_from_reg_id = reg_unicode
      into str;
      res_text = BLOB_APPEND(res_text, 'Место платежа: <b>' || str || '</b><br>');
    end

    str = '';
    if (ol_document_client is not null) then
    begin
      select f_uri_open(c_name_fio, 'client', 'Clients', c_id, c_name_fio, '', :un_id)
      from clients
      where c_id = :ol_document_client
      into str;
    end

    if (coalesce(ol_doc_client_name, '') <> '') then
      str = str || ' через ' || ol_doc_client_name;

    res_text = BLOB_APPEND(res_text, 'Клиент: <b>' || str || '</b><br>');

    res_text = BLOB_APPEND(res_text, '<br>Детали: <b>' || coalesce(ol_memo, '-') || '</b><br>');
    suspend;
  end
end
