CREATE PROCEDURE AAREP_KIB_MINIAPP_REPORT(
    MA_FROM_REG_ID INTEGER,
    MA_ID INTEGER,
    CLIENT_FIO VARCHAR(100),
    FULL_HISTORY SMALLINT = 0)
RETURNS (
    KF_CLIENT_ID DECIMAL(18,0),
    KF_CLIENT_NAME VARCHAR(100),
    KF_CLIENT_PASS_NO VARCHAR(30),
    KF_AMOUNT DECIMAL(18,4),
    MA_KIB_SEND INTEGER,
    MA_KIB_RECIEVE INTEGER,
    KH_ALL_BALANCE DECIMAL(18,4),
    KH_ALL_KREDIT DECIMAL(18,4),
    KH_APROX_BALANCE DECIMAL(18,4),
    KH_DATEAWARD DATE,
    KH_AMOUNT DECIMAL(18,4),
    KH_PERIOD_IN_MONTH DECIMAL(18,2),
    KH_CLIENTS_KOL INTEGER,
    KH_LIMIT DECIMAL(18,2),
    KH_DATESTOP DATE,
    KH_DATEPROLONGATION DATE,
    KH_AMONTHOFPROLONGATION DECIMAL(18,2),
    KH_UNPAIDAMOUNT DECIMAL(18,2),
    KH_UNPAIDPERCENAMOUNT DECIMAL(18,2),
    KH_COUNTOFDELINQUENCY INTEGER,
    KH_CREDITSTATUS VARCHAR(255),
    KH_CREDITTYPE VARCHAR(255),
    KH_CURRENCY VARCHAR(50),
    KH_REPAYSTATUS VARCHAR(50),
    KH_IS_OUR INTEGER,
    KH_CREATE_DATE TIMESTAMP,
    KH_CLIENTS_FIO VARCHAR(1024),
    KH_PORUCH_FIO VARCHAR(1024),
    KH_IS_ACTIVE INTEGER,
    KH_MONTHLY_REPAY DECIMAL(18,4),
    KF_IS_PORUCH INTEGER,
    UN_FIO VARCHAR(100),
    KH_ALL_ACTIVE_KOL INTEGER,
    KH_NO INTEGER,
    KF_DATE_BIRTH DATE,
    KF_INN VARCHAR(20),
    KF_IS_PORUCH_STR VARCHAR(20),
    KH_DELIQDAYSCOUNT INTEGER,
    KH_OWNAMOUNT DECIMAL(18,4),
    KH_HIST_QTY SMALLINT,
    KH_HIST_SHOWN SMALLINT,
    IS_REAL_HIST SMALLINT,
    MA_KOLICH_REPAY INTEGER,
    KBM_CHECKED_DATE DATE,
    KH_DELREPAYMENT VARCHAR(255),
    KBM_ID DECIMAL(18,0),
    KH_COMPANY_NAME VARCHAR(255),
    LOAN_PERCENT DECIMAL(18,2))
AS
declare variable MA_DATE_OPER date;
declare variable UN_ID decimal(18,0);
declare variable MONTH_LEFT decimal(15,2);
declare variable ST_STAFF integer;
declare variable ST_CLIENT integer;
declare variable CLOSED_H_QTY integer;
declare variable CURRENT_CLOSED integer;
declare variable H_QTY integer;
declare variable KH_ACTIVE_DELINQ_DAYS_COUNT integer;
declare variable PR_CODE D_STR_100;
declare variable BASE_K_ID D_ID;
declare variable BASE_MA_ID D_INTEGER;
declare variable INSUR_MA_ID D_INTEGER;
begin

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  ma_from_reg_id = 0;
  /*
    CHANGED BY IGUSEV 27.06.2012
    changes:  добавлена выборка по полю KH_DELIQDAYSCOUNT - число дней просрочки

    CHANGED BY IGUSEV 26.09.2012
    if ((KH_CREDITSTATUS='Выдан') || (KH_CREDITSTATUS='Своевременный платеж')) then

    CHANGED BY IGUSEV 31.01.2013
    добавлен исходящий парамет KH_OWNAMOUNT + добавлена логика рассчета для KH_APROX_BALANCE, KH_Monthly_Repay с учетом если KH_OWNAMOUNT <> 0

    CHANGED BY IGUSEV 02.07.2013
    Добавлены 2 исходящих параметра 
    KH_HIST_QTY   - кол-во кред. историй у клиента
    KH_HIST_SHOWN - кол-во отображенных кред. историй

    также теперь отчет попадают только 9 кред. историй 
    order by  KH_DATEAWARD desc
    rows 9

    c 18.11.13 в отчет попадают только 8 кред. историй
    order by  KH_DATEAWARD desc
    rows 8

  */
  kh_hist_qty = 0;
  kh_hist_shown = 0;

  if (client_fio is null) then
    client_fio = '';
  select ma_date_oper, maa.ma_kolich_repay
  from miniapp maa
  where ma_id = :ma_id and
        ma_date_end = '31.12.9999'
  into :ma_date_oper, :ma_kolich_repay;

  select first 1 pr.pr_code, ma_date_oper, ma.ma_kolich_repay
  from miniapp ma
  inner join product pr on pr.pr_id = ma.ma_from_pr_id
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.2999'
  order by ma.ma_number desc
  into pr_code, ma_date_oper, ma_kolich_repay;

  if (pr_code containing '.insur_kontrakt.') then  -- RFCKG-5051
  begin
    select lir.lir_kontract_base
    from kontract k
    inner join loan_insurance_relation lir on k.k_id = lir.lir_kontract_insurance
    where k.k_from_ma_app_id = :ma_id
    into base_k_id;

    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :base_k_id
    into base_ma_id;

    if (coalesce(base_ma_id, 0) <> 0) then
    begin
      insur_ma_id = ma_id;
      ma_id = base_ma_id;
    end
  end

  for select mac_from_c_id, c_name_fio, c_pass_no, mac_amount, mac_is_poruch, c_birth_date, c_inn
      from miniappclient macl, clients
      where mac_from_ma_id = :ma_id and
            c_id = mac_from_c_id and
            ((:client_fio = '') or
            (:client_fio <> '' and
            c_name_fio = :client_fio))
      into :kf_client_id, :kf_client_name, :kf_client_pass_no, :kf_amount, :kf_is_poruch, :kf_date_birth, :kf_inn
  do
  begin
    if (pr_code containing '.insur_kontrakt.') then   -- RFCKG-5051
    begin
      select mac_amount
      from miniappclient mac
      where mac.mac_from_ma_id = :insur_ma_id and
            mac.mac_from_c_id = :kf_client_id
      into kf_amount;

      kf_amount = coalesce(kf_amount, 0);
    end

    kh_no = 0;
    ma_kib_send = 0;
    ma_kib_recieve = 0;

    kbm_id = 0;
    select kbm_from_un_id, kbm_id, kbm_checked_datetime
    from hp_kibminiapp_last(:kf_client_id, :ma_id)
    into un_id, kbm_id, kh_create_date;

    if (kbm_id > 0) then
      ma_kib_send = 1;

    kh_all_balance = 0;
    kh_all_kredit = 0;
    kh_all_active_kol = 0;

    if ((not exists(select *
                    from kibhistory kh
                    where kh.kh_from_kbm_id = :kbm_id and
                          kh_creditstatus not containing ('Отказано') -- если отказано, это значит клиент кредит не получил
                          and
                          kh.kh_creditstatus not containing ('Рассмотрение заявки') -- кредитный договор еще не оформлен
        )) and
        (kbm_id > 0)) then
    begin
      un_fio = 'Не определен';

      select c_name_fio
      from clients, username
      where un_id = :un_id and
            un_from_c_id = c_id
      into :un_fio;
      if (un_fio is null) then
        un_fio = 'Не определен';

      ma_kib_recieve = 1;

      kh_all_balance = 0;
      kh_all_kredit = 0;
      kh_aprox_balance = 0;
      kh_amount = 0;
      kh_period_in_month = 0;
      kh_clients_kol = 0;
      kh_limit = 0;
      kh_amonthofprolongation = 0;
      kh_unpaidamount = 0;
      kh_unpaidpercenamount = 0;
      kh_countofdelinquency = 0;
      kh_is_our = 0;
      kh_is_active = 0;
      kh_monthly_repay = 0;
      kf_is_poruch = 0;
      kh_all_active_kol = 0;
      kh_no = 0;
      kh_deliqdayscount = 0;
      kh_ownamount = 0;
      kh_hist_qty = 0;
      kh_hist_shown = 0;
      kh_repaystatus = null;
      kh_creditstatus = null;
      kh_currency = null;

      is_real_hist = 0;

      suspend;
    end

    kh_hist_qty = 0;
    closed_h_qty = 0;
    h_qty = 0;

    kh_delrepayment = '';
    for select kh_amount, kh_limit, kh_dateaward, kh_datestop, kh_dateprolongation, kh_amonthofprolongation,
               kh_unpaidamount, kh_unpaidpercenamount,
               iif(coalesce(kh_max_delinq_days_count, 0) > 0, kh_max_delinq_days_count, kh_countofdelinquency), -- igusev 24.01.2019 до 24.01.2019 тянулось просто поле kh_countofdelinquency,
               kh_creditstatus, kh_credittype, kh_currency, kh_repaystatus, kh_create_date,
               kh_period_in_month, kh_clients_kol, kh_clients_fio, kh_poruch_fio, kh_is_our, kbm_from_un_id,
               kh_deliqdayscount, kh_ownamount, iif(kh_is_debtor = 0, 1, 0), cast(kbm_checked_datetime as date),
               kh_delrepayment, coalesce(kh_active_delinq_days_count, 0), coalesce(kh_company_name, ''), kh_loan_percent
        from hp_kibminiapp_last(:kf_client_id, :ma_id)
        inner join kibhistory on kh_from_kbm_id = kbm_id
        where --KH_IS_DEBTOR = 1 and
              kh_creditstatus not containing ('Отказано') and -- если отказано, это значит клиент кредит не получил
              kh_creditstatus not containing ('Рассмотрение заявки') -- кредитный договор еще не оформлен
        order by kh_dateaward desc
            --rows 8
        into :kh_amount, :kh_limit, :kh_dateaward, :kh_datestop, :kh_dateprolongation, :kh_amonthofprolongation,
             :kh_unpaidamount, :kh_unpaidpercenamount, :kh_countofdelinquency, :kh_creditstatus, :kh_credittype,
             :kh_currency, :kh_repaystatus, :kh_create_date, :kh_period_in_month, :kh_clients_kol,
             :kh_clients_fio, :kh_poruch_fio, :kh_is_our, :un_id, :kh_deliqdayscount, :kh_ownamount, :kf_is_poruch,
             :kbm_checked_date, :kh_delrepayment, :kh_active_delinq_days_count, :kh_company_name, :loan_percent
    do
    begin
      if(current_date < cast(f_get_const('send_interest_rate_to_kib') as date)) then
        loan_percent = 0;

      if (kh_company_name <> '') then
      begin
        kh_is_our = 0;
        if ((kh_company_name containing 'М Булак') or (kh_company_name containing 'МБулак')) then
            kh_is_our = 1;
      end

      if (kh_credittype = 'Individual') then
        kh_credittype = 'Индив. кредит';
      else
      if (kh_credittype = 'Group_Loan') then
        kh_credittype = 'Групповой. кредит';

      h_qty = h_qty + 1;
      current_closed = 0;

      if (kh_ownamount is null) then
        kh_ownamount = 0;
      if (kf_is_poruch is null) then
        kf_is_poruch = 0;
      if (kf_is_poruch = 0) then
        kf_is_poruch_str = 'Заемщик';
      if (kf_is_poruch = 1) then
        kf_is_poruch_str = 'Поручитель';

      is_real_hist = 1;
      kh_no = kh_no + 1;
      un_fio = 'Не определен';

      select c_name_fio
      from clients, username
      where un_id = :un_id and
            un_from_c_id = c_id
      into :un_fio;
      if (un_fio is null) then
        un_fio = 'Не определен';

      if (kh_hist_qty = 0) then
      begin
        select count(*)
        from kibminiapp
        inner join kibminiapp_link on kbl_from_kbm_id = kbm_id
        inner join kibhistory on kh_from_kbm_id = kbm_id
        where kbl_object_kind = f_objkind('miniapp', 'id') and
              kbl_object_id = :ma_id and
              kbm_from_c_id = :kf_client_id
        into  kh_hist_qty;

        kh_hist_shown = kh_hist_qty;
        if (kh_hist_shown > 8) then
          kh_hist_shown = 8;
      end

      ma_kib_recieve = 1;
      kh_aprox_balance = 0;
      kh_is_active = 0;
      kh_monthly_repay = 0;

      if (kh_creditstatus in ('Выдан', 'Реструктуризирован', 'Своевременный платеж', 'Начисление процентов',

                          'Granted_And_Activated', 'Restructured', 'Open_Under_Legal_Proceedings') or (kh_creditstatus containing ('Просрочен'))) then
      begin
        kh_is_active = 1; -- Активный заем

        month_left = datediff(month, :kh_dateaward, :ma_date_oper);
        if (month_left < cast(month_left as integer)) then
          month_left = cast(month_left as integer) - 1;
        else
          month_left = cast(month_left as integer);
        if (month_left < kh_period_in_month) then
        begin
          if ((kh_clients_kol <> 0) and
              (kh_period_in_month <> 0)) then
          begin
            if (kh_ownamount = 0) then
            begin
              kh_aprox_balance = kh_amount - (kh_amount / kh_period_in_month) * month_left;
              kh_aprox_balance = kh_aprox_balance / kh_clients_kol;
            end
            else
              kh_aprox_balance = kh_ownamount - (kh_ownamount / kh_period_in_month) * month_left;
          end
        end
        /*            if (KH_APROX_BALANCE<(KH_UNPAIDAMOUNT/KH_CLIENTS_KOL))  then KH_APROX_BALANCE=(KH_UNPAIDAMOUNT/KH_CLIENTS_KOL); */

        if ((kh_clients_kol <> 0) and
            (kh_period_in_month <> 0)) then
        begin
          if (kh_ownamount = 0) then
            kh_monthly_repay = (kh_amount / kh_clients_kol) / kh_period_in_month;
          else
            kh_monthly_repay = kh_ownamount / kh_period_in_month;
        end

      end
      else
      begin
        closed_h_qty = closed_h_qty + 1;
        current_closed = 1;
      end

      kh_all_balance = kh_all_balance + kh_aprox_balance;
      kh_all_kredit = kh_all_kredit + kh_monthly_repay;
      kh_all_active_kol = kh_all_active_kol + kh_is_active;

      /*
        В отчет КИБ по клиенту подтягиваю все активные займы (не важно сколько их, в нашей 
        компании/не в нашей, подтягиваю все) + 3 последних закрытых займа.

        */

      if (coalesce(kh_repaystatus, '') = '') then
      begin
        select rd.rd_name
        from refdata rd
        where rd.rd_main_code = 'Ishenim_CreditStatus' and
              rd.rd_parent_id is not null and
              rd.rd_code = :kh_creditstatus
        into :kh_creditstatus;

        kh_repaystatus = 'Акт. пр-ка: ' || cast(kh_active_delinq_days_count as str_100) || ' дн';
      end

      if (current_closed = 0) then
        suspend;
      else
      begin
        if (full_history = 1) then
        begin
          suspend;
        end
        else
        begin
          if ((h_qty <= 7) and
              (closed_h_qty <= 5)) then
            suspend;
        end
      end
    end
  end
end
