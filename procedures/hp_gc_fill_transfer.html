CREATE PROCEDURE HP_GC_FILL_TRANSFER(
    IGA_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable ifa_id d_integer;
declare variable c_id d_id;
declare variable ag_id d_integer;
declare variable amount d_currency;
declare variable num d_integer;
begin
  err_text = '';
  ifa_id = (select iga_from_ifa_id from ipc_gc_analiz where iga_id = :iga_id);

  if (exists(
        select *
        from ipc_file_analiz
        where ifa_id = :ifa_id
          and ifa_file_type = 1
          and ifa_status in (-1,1,2))) then
  begin
    delete from ipc_gc_transfer
      where igt_from_iga_id = :iga_id;

    num = 0;
    for
      with temp as (
        select
          igdp_from_c_id as c_id,
          igdp_from_ag_id as ag_id,
          sum(igdp_amount - coalesce(igdp_tsp_commis_amount,0)) as amount
        from ipc_gc_analiz_detail_pos
        where igdp_from_iga_id = :iga_id
          and igdp_is_error = 0
          and igdp_type = 1
          and igdp_from_c_id > 0
          and igdp_from_ag_id > 0
        group by igdp_from_c_id, igdp_from_ag_id
        union all
        select
          igqp_from_c_id as c_id,
          igqp_from_ag_id as ag_id,
          sum(igqp_amount - coalesce(igqp_tsp_commis_amount,0)) as amount
        from ipc_gc_analiz_qr_payments
        where igqp_from_iga_id = :iga_id
          and igqp_is_error = 0
          and igqp_from_c_id > 0
          and igqp_from_ag_id > 0
        group by igqp_from_c_id, igqp_from_ag_id
      )
      select
        c_id,
        ag_id,
        sum(amount)
      from temp
      group by c_id, ag_id
    into
      :c_id,
      :ag_id,
      :amount
    do
    begin
      insert into ipc_gc_transfer (
        igt_from_iga_id,
        igt_from_c_id,
        igt_from_ag_id,
        igt_account,
        igt_account_bank_id,
        igt_amount,
        igt_status,
        igt_do_transfer
      )
      select
        :iga_id,
        :c_id,
        :ag_id,
        ag_elsom_account,
        ag_elsom_bank_id,
        round(:amount,2),
        0,
        1
      from agreements
      where ag_id = :ag_id;
      num = num + 1;
    end

    if (num > 0) then
      update ipc_file_analiz set
        ifa_status = 2,
        ifa_from_un_id = :un_id,
        ifa_datetime = current_timestamp
      where ifa_id = :ifa_id;
  end else err_text = 'Нет данных для формирования списка переводов. Проверьте актуальность операции';
end
