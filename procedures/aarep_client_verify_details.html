CREATE PROCEDURE AAREP_CLIENT_VERIFY_DETAILS(
    OBJ_KIND INTEGER,
    OBJ_ID DECIMAL(18,0))
RETURNS (
    ACL_ID DECIMAL(18,0),
    RESULT_C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_VERIFY_STATUS INTEGER,
    C_VERIFY_STATUS_STR VARCHAR(100),
    ACL_DATETIME TIMESTAMP,
    RES_FIO INTEGER,
    RES_SEX INTEGER,
    RES_BIRTH_DATE INTEGER,
    RES_PASSNO INTEGER,
    RES_PASSDATE INTEGER,
    RES_PASSDATE_EXPIRE INTEGER,
    RES_INN INTEGER,
    RES_PHOTO INTEGER,
    RES_PASSPORT INTEGER,
    RES_PASSPORT_WHO INTEGER,
    RES_BIRTH_PLACE INTEGER,
    RES_PASSKIND INTEGER,
    ACL_FINAL_DATETIME TIMESTAMP,
    ACL_MEMO VARCHAR(255),
    RES_LEG_ADDR INTEGER)
AS
declare variable c_id             decimal(18,0);
declare variable k_id             decimal(18,0);
declare variable ma_id            integer;
declare variable acm_memo         varchar(4000);
declare variable acl_respons_c_id decimal(18,0);
declare variable acl_status       integer;
declare variable json_key         varchar(255);
declare variable json_value       varchar(255);
declare variable check_kind       varchar(20);
declare variable kol              integer;
declare variable aco_id           decimal(18,0);
declare variable cds_create_date  date;
begin
  c_id = 0;
  k_id = 0;
  ma_id = 0;
  if (obj_kind = 0) then
    c_id = obj_id;
  if (obj_kind = 9) then
    k_id = obj_id;
  if (obj_kind = 11) then
    ma_id = obj_id;

  for select kc_from_c_id
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      union
      select c_id
      from clients
      where c_id > 0 and
            c_id = :c_id
      union
      select mac_from_c_id
      from miniappclient
      where mac_from_ma_id = :ma_id and
            (mac_amount > 0 or
            mac_is_poruch = 1)
      into :result_c_id
  do
  begin
    acl_memo = '';
    res_fio = 0;
    res_sex = 0;
    res_birth_date = 0;
    res_passno = 0;
    res_passdate = 0;
    res_passdate_expire = 0;
    res_inn = 0;
    res_photo = 0;
    res_passport = 0;
    res_passport_who = 0;
    res_birth_place = 0;
    res_leg_addr = 0;

    c_verify_status_str = '';
    select c_name_fio, c_verify_status
    from clients
    where c_id = :result_c_id
    into :c_name_fio, :c_verify_status;

    acl_id = null;
    acl_datetime = null;
    acl_final_datetime = null;
    acl_respons_c_id = null;
    acl_status = null;

    select acl_id, acl_datetime, acl_last_datetime, acl_respons_c_id, acl_status
    from actionlist
    where acl_object_kind = 0 and
          acl_object_id = :result_c_id and
          acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO'))
    order by acl_id desc
    rows 1
    into :acl_id, acl_datetime, acl_final_datetime, acl_respons_c_id, acl_status;

    acm_memo = '';
    select acm_memo
    from actionmemo
    where acm_from_acl_id = :acl_id
    into :acm_memo;
    if (coalesce(acm_memo, '') <> '') then
    begin
      for select json_key, json_value
          from fm_json_parse(:acm_memo, 0, 'verify')
          into :json_key, :json_value
      do
      begin
        if (json_key = 'comment') then
          acl_memo = json_value;
      end
    end

    if (c_verify_status = 2) then
    begin
      c_verify_status_str = 'Подтверждена: ОК';
      select cast(cds.cds_create_datetime as date)
      from clientdocs cds
      where cds.cds_from_c_id = :result_c_id and
            cds.cds_is_main_doc = 1 and
            cds.cds_is_delete = 0
      into cds_create_date;
      if (exists(select *
                 from grs_clients gc
                 where gc.gc_from_c_id = :result_c_id and
                       gc.gc_is_last = 1 and
                       gc.gc_status in (2, 3) and
                       cast(gc.gc_status_datetime as date) = :cds_create_date)) then
        c_verify_status_str = c_verify_status_str || ' (Информация получена с ГРС)';
      res_fio = 1;
      res_sex = 1;
      res_birth_date = 1;
      res_passno = 1;
      res_passdate = 1;
      res_passdate_expire = 1;
      res_inn = 1;
      res_photo = 1;
      res_passport = 1;
      res_passport_who = 1;
      res_birth_place = 1;
      res_leg_addr = 1;
    end

    if (c_verify_status = -2) then
    begin
      c_verify_status_str = 'Результат: Отказ';
      if (acl_id is not null) then
      begin
        res_fio = 1;
        res_sex = 1;
        res_birth_date = 1;
        res_passno = 1;
        res_passdate = 1;
        res_passkind = 1;
        res_passdate_expire = 1;
        res_inn = 1;
        res_photo = 1;
        res_passport = 1;
        res_passport_who = 1;
        res_birth_place = 1;
        res_leg_addr = 1;

        if (coalesce(acm_memo, '') <> '') then
        begin
          for select json_key, json_value
              from fm_json_parse(:acm_memo, 0, 'verify')
              into :json_key, :json_value
          do
          begin
            check_kind = 'fio';
            if (json_key = check_kind) then
              res_fio = -1;
            check_kind = 'sex';
            if (json_key = check_kind) then
              res_sex = -1;
            check_kind = 'birth_date';
            if (json_key = check_kind) then
              res_birth_date = -1;
            check_kind = 'passno';
            if (json_key = check_kind) then
              res_passno = -1;
            check_kind = 'passdate_disb';
            if (json_key = check_kind) then
              res_passdate = -1;
            check_kind = 'passdate_expire';
            if (json_key = check_kind) then
              res_passdate_expire = -1;
            check_kind = 'inn';
            if (json_key = check_kind) then
              res_inn = -1;
            check_kind = 'photo';
            if (json_key = check_kind) then
              res_photo = -1;
            check_kind = 'passport';
            if (json_key = check_kind) then
              res_passport = -1;
            check_kind = 'passport_who';
            if (json_key = check_kind) then
              res_passport_who = -1;
            check_kind = 'birth_place';
            if (json_key = check_kind) then
              res_birth_place = -1;
            check_kind = 'passkind';
            if (json_key = check_kind) then
              res_passkind = -1;
          end -- Конец персинга
        end
        else
        begin
          res_birth_date = 0;
          res_passno = 0;
          res_passdate = 0;
          res_passdate_expire = 0;
          res_passkind = 0;
          res_inn = 0;
          res_photo = 0;
          res_passport = 0;
          res_passport_who = 0;
          res_birth_place = 0;
          res_leg_addr = 0;
        end
      end
    end

    if (c_verify_status = -1) then
    begin
      c_verify_status_str = 'Результат: Верификация не требуется';
    end

    if (c_verify_status = 0) then
    begin
      c_verify_status_str = 'Результат: Данные не полные';
      res_fio = 1;
      res_sex = 1;
      res_birth_date = 1;
      res_passno = 1;
      res_passdate = 1;
      res_passkind = 1;
      res_passdate_expire = 1;
      res_inn = 1;
      res_photo = 1;
      res_passport = 1;
      res_passport_who = 1;
      res_birth_place = 1;
      res_leg_addr = 1;

      select iif(coalesce(c_birth_date, '') = '', -1, 1), iif(coalesce(c_pass_no, '') = '', -1, 1),
             iif(coalesce(c_pass_date, '') = '', -1, 1), iif(coalesce(c_pass_date_expire, '') = '', -1, 1),
             iif(coalesce(c_blob_photo_id, 0) <= 0, -1, 1), iif(coalesce(c_blob_pass1_id, 0) <= 0, -1, 1),
             iif(coalesce(c_pass_who, '') = '', -1, 1), iif(coalesce(c_birth_country, '') = '', -1, 1),
             iif(coalesce(c_doc_type, 0) <= 0, -1, 1), iif(coalesce(adl.adl_is_filled, 0) <= 0, -1, 1)
      from clients
      inner join addrclient adr on adr.adc_from_c_id = c_id
      inner join addrlist adl on adr.adc_from_adl_id = adl.adl_id and
            adr.adc_address_type = 1
      where c_id = :result_c_id
      into res_birth_date, res_passno, res_passdate, res_passdate_expire, res_photo, res_passport, res_passport_who,
           res_birth_place, res_passkind, res_leg_addr;
    end

    if (c_verify_status = 1) then
    begin
      kol = 0;
      if (acl_respons_c_id is not null) then
        c_verify_status_str = 'Результат: Рассматривается';
      else
      begin
        for select aco_id
            from actionorder
            where aco_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO'))
            order by aco_orderno
            into :aco_id
        do
        begin
          if (aco_id <= acl_id) then
            kol = kol + 1;
        end
        c_verify_status_str = 'Результат: В ожидании № ' || kol;
      end
    end
    suspend;
  end
end
