CREATE PROCEDURE AAREP_GET_COMBOLIST(
    TABLE_NAME VARCHAR(100),
    FIELD_NAME VARCHAR(100),
    SP_PARAM VARCHAR(255),
    UPDATED_TIME TIMESTAMP,
    CATEGORY_VALUE VARCHAR(255) = '')
RETURNS (
    TEXT_LIST VARCHAR(255),
    SERVER_TIME TIMESTAMP)
AS
declare variable CF_KIND integer;
declare variable SQL_TEXT varchar(1024);
declare variable CF_STORED_PROC_PARAM varchar(255);
declare variable CF_IS_READY integer;
declare variable TXT1 varchar(100);
declare variable TXT2 varchar(100);
declare variable N integer;
declare variable CF_MASTER_FIELD_NAME varchar(100);
begin
  server_time = current_timestamp;
  if (sp_param is null) then
    sp_param = '';
  if (category_value is null) then
    category_value = '';
  if (updated_time is null) then
    updated_time = '01.01.1900';

  if ((table_name = 'clients') and
      (field_name = 'c_birth_country')) then
  begin
    for select rd_name
        from refdata
        where rd_main_code = 'citizen' and
              rd_parent_id is not null
        order by rd_orderno, rd_name
        into :text_list
    do
    begin
      suspend;
    end
    exit;
  end

  if ((table_name = 'clients') and
      (field_name = 'fio_list_1')) then
  begin
    txt1 = '';
    txt2 = '';
    n = position(' ', :sp_param, 1);
    if (n > 0) then
    begin
      txt1 = trim(left(sp_param, n - 1));
    end
    else
      txt1 = trim(sp_param);

    if (txt1 <> '') then
    begin
      for select distinct fl_name2
          from fiolist
          where fl_name1 = :txt1 and
                fl_name2 <> '' and
                fl_name2 is not null
          into :text_list
      do
      begin
        text_list = left(text_list, 1) || lower(right(text_list, char_length(:text_list) - 1));
        suspend;
      end
    end
    exit;
  end

  if ((table_name = 'clients') and
      (field_name = 'fio_list_2')) then
  begin
    txt1 = '';
    txt2 = '';
    n = position(' ', :sp_param, 1);
    if (n > 0) then
    begin
      txt1 = trim(left(sp_param, n - 1));
      txt2 = trim(substring(sp_param from n + 1 for 255));
    end
    else
      txt1 = trim(sp_param);

    if ((txt1 <> '') and
        (txt2 <> '')) then
    begin
      text_list = '-';
      suspend;
      for select distinct fl_name3
          from fiolist
          where fl_name1 = :txt1 and
                fl_name2 = :txt2 and
                fl_name3 <> '' and
                fl_name3 <> '-' and
                fl_name3 is not null
          into :text_list
      do
      begin
        text_list = left(text_list, 1) || lower(right(text_list, char_length(:text_list) - 1));
        suspend;
      end
    end
    exit;
  end

  if ((table_name = 'clients') and
      (field_name = 'verify_comment')) then
  begin
    text_list = 'Размытое изображение';
    suspend;
    text_list = 'Плохое качество изображения';
    suspend;
    text_list = 'Из-за освещения невозможно прочитать текст';
    suspend;
    text_list = 'Требуется фото с оригинала';
    suspend;
    text_list = 'На 1странице паспорта номер паспорта не читабелен, необходимо приложить паспорт в полном объеме.';
    suspend;
    text_list = 'Паспорт гр.УЗБ старого образца не принимается на обслуживание необходимо приложить биометрический/заграничный паспорт';
    suspend;
    text_list = 'Данные в приложенном паспорте и данные указанные в БД не совпадают';
    suspend;
    text_list = 'Основное фото клиента не соответствует требованиям - приложите повторное фото клиента.';
    suspend;
    text_list = 'В паспорте отсутствует подпись клиента.';
    suspend;
    text_list = 'В паспорте  отсутствуют/не отображаются водяные знаки, необходимо проверить паспорт через UV и приложить повторно в  хорошем качестве.';
    suspend;
    text_list = 'Отсутствует  оборотная/ 2 сторона паспорта.';
    suspend;
    text_list = 'Лица без гражданства не принимаются на обслуживание.';
    suspend;
    text_list = 'Паспорт значится среди не действительных по базе ГРС.';
    suspend;
    text_list = 'Паспорт испорчен и является не действительным ( разрисован чернилами/порван/деформирован и пр.)';
    suspend;
   -- RFCRU-3851
    text_list = 'На фото в паспорте имеются недопустимые элементы.';
    suspend;
    text_list = 'Приложено основное фото другого клиента.';
    suspend;
    text_list = 'Отсутствует основное фото клиента.';
    suspend;
    text_list = 'На фото в паспорте лицо клиента засвечено.';
    suspend;
    text_list = 'Отсутствует паспорт клиента.';
    suspend;
    text_list = 'Срок действия паспорта истек.';
    suspend;
    text_list = 'Фото паспорта с экрана телефона, приложите оригинал.';
    suspend;
    text_list = 'Паспорт не в полном объеме.';
    suspend;
    text_list = 'Паспорт приложен в ПДФ формате не отображается в БД.';
    suspend;
    -- RFCRU-5033
    text_list = 'Требуется верификация клиента по QR-коду';
    suspend;
    exit;
  end

  select cf_kind, cf_is_ready, coalesce(cf_stored_proc_param, ''), coalesce(cf_master_field_name, '')
  from combofields
  where cf_table_name = :table_name and
        cf_field_name = :field_name and
        coalesce(cf_stored_proc_param, '') = :sp_param rows 1
  into :cf_kind, :cf_is_ready, :cf_stored_proc_param, cf_master_field_name;
  if (cf_stored_proc_param is null) then
    cf_stored_proc_param = '';

  if (cf_kind is not null) then
  begin
    if (cf_is_ready = 1) then
    begin
      if (cf_master_field_name <> '' and
          category_value <> '') then
      begin
        for select cl_text
            from combolist
            where cl_kind = :cf_kind and
                  cl_category = :category_value and
                  cl_datetime >= :updated_time
            into :text_list
        do
        begin
          suspend;
        end
      end
      else
      begin
        for select cl_text
            from combolist
            where cl_kind = :cf_kind and
                  cl_datetime >= :updated_time
            into :text_list
        do
        begin
          suspend;
        end
      end
    end
    else
    begin
      text_list = 'Нет данных';
      suspend;
    end
  end
  else
  begin
    if (coalesce(cf_stored_proc_param, '') = '') then
    begin
      sql_text = 'select distinct ' || :field_name || ' from ' || :table_name || ' where ' || :field_name || '<>'''' and ' || :field_name || ' is not null';
      for execute statement :sql_text
              into :text_list
      do
      begin
        suspend;
      end
    end
  end
end
