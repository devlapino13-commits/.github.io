CREATE PROCEDURE HP_ACTIONLIST_OBJECT_NAME(
    REP_OBJECT_KIND INTEGER,
    REP_OBJECT_ID DECIMAL(18,0))
RETURNS (
    RES_OBJECT_NAME VARCHAR(255))
AS
declare variable k_id d_id;
declare variable ma_id d_integer;
begin
-- Процедура возвращает название объекта

  res_object_name = '';
  if (rep_object_kind = 0) then -- Контрагент
  begin
    select c_name_fio
    from clients
    where c_id = :rep_object_id
    into :res_object_name;
  end
  if (rep_object_kind = 8) then -- Контрагент в кредите
  begin
    select k_kontract_no || ', ' || c_name_fio
    from knclient, clients, kontract
    where kc_id = :rep_object_id and
          kc_from_k_id = k_id and
          kc_from_c_id = c_id
    into :res_object_name;
  end
  if (rep_object_kind = 9) then -- Кредит
  begin
    select k_kontract_no
    from kontract
    where k_id = :rep_object_id
    into :res_object_name;
  end
  if (rep_object_kind = 11) then -- Заявка
  begin
    select ma_no
    from miniapp
    where ma_id = :rep_object_id
    order by ma_id, ma_number
    rows 1
    into :res_object_name;
  end
  if (rep_object_kind = 10) then -- Документ клиента
  begin
    select pk_name
    from clientdocs, passkind
    where cds_id = :rep_object_id and
          cds_from_pk_id = pk_id
    into :res_object_name;
  end
  if (rep_object_kind = 31) then -- Распознавание документов
  begin
    select iif(drl_from_c_id is null, 'Клиент:', 'Документ:') || pk_name
    from docrecoglist, passkind
    where drl_id = :rep_object_id and
          drl_from_pk_id = pk_id
    into :res_object_name;
  end
  if (rep_object_kind = 54) then -- Профсуждение залога
  begin
    select za_from_k_id, za_from_ma_id
    from zalogautoprof, zalogauto
    where zap_id = :rep_object_id and
          zap_from_za_id = za_id
    into k_id, ma_id;

    if (k_id is not null) then
      select res_object_name from hp_actionlist_object_name(9, :k_id)
      into res_object_name;
    else
      select res_object_name from hp_actionlist_object_name(11, :ma_id)
      into res_object_name;
  end
  if (rep_object_kind = 58) then -- Фото встречи, клиент
  begin
    select c_name_fio
    from qr_list, clients
    where qrl_id = :rep_object_id and
          qrl_from_c_id = c_id
    into :res_object_name;
  end
  suspend;
end
