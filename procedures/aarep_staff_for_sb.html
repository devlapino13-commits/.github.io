CREATE PROCEDURE AAREP_STAFF_FOR_SB(
    UN_ID DECIMAL(18,0))
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(255),
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(255),
    JL_NAME VARCHAR(255),
    JL_SYSTEM_NAME VARCHAR(255),
    C_CELL_NO VARCHAR(255),
    C_EMAIL VARCHAR(255),
    HAS_CELL INTEGER,
    HAS_EMAIL INTEGER)
AS
declare variable reg_id integer;
begin
  reg_id = 0;
  reg_id = f_hq_id();

  if ((exists(select has_access
              from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Superuser')
              where has_access = 1)) or (exists(select has_access
                                                from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Administrator')
                                                where has_access = 1))) then
    reg_id = 0;

  for select c_id, c_name_fio, reg_code, reg_name, jl_name, rd_name
      from hr_pstatus, clients, joblist, region, jobkind, refdata
      where ps_from_c_id = c_id and
            ps_from_jl_id = jl_id and
            ps_from_reg_id = reg_unicode and
            ps_from_reg_id <> :reg_id and
            ps_is_working = 1 and
            ps_status = 1 and
            ps_date_end >= '31.12.2999' and
            jl_kind = jk_id and
            jk_system_id = rd_id
      into :c_id, c_name_fio, reg_code, reg_name, jl_name, jl_system_name
  do
  begin
    has_cell = 0;
    has_email = 0;

    select cont_value
    from hp_contdata_get_cell_valid(0, :c_id)
    into :c_cell_no;

    select cont_value
    from hp_contdata_get_email_valid(0, :c_id)
    into :c_email;

    if (c_cell_no <> '') then
      has_cell = 1;
    if (c_email <> '') then
      has_email = 1;

    suspend;
  end
end
