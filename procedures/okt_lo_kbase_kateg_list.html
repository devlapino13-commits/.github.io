CREATE PROCEDURE OKT_LO_KBASE_KATEG_LIST
RETURNS (
    CKC_ID INTEGER,
    CKC_NAME STR_255,
    CKC_MEMO STR_255,
    CKC_TYPE SMALLINT,
    CKC_IS_DELETE SMALLINT,
    CKC_OPER_ID ID_TYPE)
AS
begin
  for select ckc_id, ckc_name, ckc_memo, ckc_type, ckc_is_delete, ckc_oper_id
      from cc_kbase_categ ckc
      where ckc.ckc_is_delete = 0
      into :ckc_id, :ckc_name, :ckc_memo, :ckc_type, :ckc_is_delete, :ckc_oper_id
  do
  begin
    suspend;
  end
end
