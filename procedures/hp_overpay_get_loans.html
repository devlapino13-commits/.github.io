CREATE PROCEDURE HP_OVERPAY_GET_LOANS(
    OPM_ID D_ID,
    OPL_TYPE D_SMALLINT)
RETURNS (
    REC_NO D_INTEGER,
    K_KONTRACT_NO D_STR_CODE,
    K_DISB_DATE D_DATE,
    OPL_SUM D_CURRENCY,
    OPL_RES_SUM D_CURRENCY,
    OPL_DATE D_DATE,
    OPL_OVERDUE_DAYS D_INTEGER)
AS
begin
  for select opl_no, k_kontract_no, opl_sum, opl_res_sum, opl_date, k_disb_date, opl_overdue_days
  from overpay_loans, kontract
  where k_id = opl_from_k_id
    and opl_from_opm_id = :opm_id
    and opl_type = :opl_type
  order by opl_no nulls last, opl_overdue_days desc, k_disb_date
  into rec_no, k_kontract_no, opl_sum, opl_res_sum, opl_date, k_disb_date, opl_overdue_days
  do
    suspend;
end
