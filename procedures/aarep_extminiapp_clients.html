CREATE PROCEDURE AAREP_EXTMINIAPP_CLIENTS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    C_NAME_FIO_PARTNYER D_STR_100,
    DISB_REG_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    C_NAME_FIO D_STR_100,
    K_FACT_DISB_DATE D_DATE,
    KC_AMOUNT_APPROVED D_CURRENCY,
    DISB_LO_FIO D_STR_100,
    CLK_ADD_DATETIME D_DATE,
    BONUS_GROUP D_INTEGER,
    RES_TEXT D_STR_100)
AS
declare variable db_id d_integer;
declare variable ogr_object_id d_id;
declare variable ogr_object_kind d_integer;
declare variable clk_from_un_id d_id;
declare variable clk_from_mcl_id d_id;
declare variable clk_link_cell_no d_str_20;
declare variable k_id d_id;
declare variable ol_id d_id;
declare variable c_id d_id;
begin
  if (f_get_db_id() <> 81001) then
    exit;

  for select aarep_k_id, ol_id
      from aarep_kontract_id(:rep_region_id, :rep_date_begin, :rep_date_end, 0, '000', 2)
      into :k_id, ol_id
  do
  begin

    ogr_object_kind = 0;
    ogr_object_id = 0;

    select ogr.ogr_object_kind, ogr.ogr_object_id
    from operagr ogr
    where ogr.ogr_from_ol_id = :ol_id and
          ogr.ogr_from_rd_id = f_get_rd_id('AgreementKinds.loyal_agreement') and
          ogr.ogr_object_id is not null and
          ogr.ogr_ol_is_active = 1
    into :ogr_object_kind, ogr_object_id;

    if (ogr_object_kind = 34) then
    begin
      clk_link_cell_no = '';
      clk_from_un_id = 0;
      clk_from_mcl_id = 0;
      res_text = '';

      select clk.clk_link_cell_no, coalesce(clk.clk_from_un_id, 0), coalesce(clk.clk_from_mcl_id, 0), cl.c_id, cl.c_name_fio,
             clk.clk_add_datetime
      from clientlink clk
      inner join clients cl on cl.c_id = clk.clk_from_c_id
      where clk.clk_id = :ogr_object_id
      into :clk_link_cell_no, clk_from_un_id, clk_from_mcl_id, c_id, c_name_fio_partnyer, clk_add_datetime;

      bonus_group = null;
      select ppg_bonus
      from partner_program_group
      where ppg_from_c_id = :c_id
      into bonus_group;

      k_kontract_no = '';
      k_fact_disb_date = null;
      kc_amount_approved = null;
      c_name_fio = '';
      disb_reg_name = '';
      disb_lo_fio = '';
      select k.k_kontract_no, k.k_fact_disb_date, knc.kc_amount_approved, cl.c_name_fio, r.reg_name,
             cl_disb.c_name_fio
      from kontract k
      inner join knclient knc on knc.kc_from_k_id = k.k_id
      inner join clients cl on cl.c_id = knc.kc_from_c_id
      inner join region r on r.reg_unicode = k.k_fact_disb_reg_id
      inner join clients cl_disb on cl_disb.c_id = k.k_fact_disb_lo_id
      where knc.kc_amount_approved > 0 and
            k.k_id = :k_id
      into :k_kontract_no, k_fact_disb_date, kc_amount_approved, c_name_fio, disb_reg_name, disb_lo_fio;

      if (exists(select *
                 from extminiweb ex
                 where ex.emw_phone = :clk_link_cell_no)) then
        res_text = 'Друг Партнера заполнил анкету на WS';

      else
      begin
        if (clk_from_un_id > 0) then
          res_text = 'Контакт друга Партнера в МП сохранил ВиПС';
        else
        if (clk_from_mcl_id > 0) then
          res_text = 'Его контакт Партнер сохранил в КМП ';

      end
      suspend;
    end

  end

end
