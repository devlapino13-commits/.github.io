declare function my_f_div_sum (
    summ decimal(15,2),
    om_perc decimal(18,7))
returns decimal(15,2)
as
declare variable result decimal(15,2);
begin
  -- для деления сумм между участниками (групповой кредит)
  if (coalesce(summ, 0) = 0) then
    result = 0;
  else
  begin
    result = summ * om_perc / 100;
    if (result = 0 and
        summ > 0) then -- бывает при делении остается 0, т.е. количество копеек меньше количества участников
      result = 0.01;
  end
  return result;
end

  if (pr_percent_kind = 1) then
  begin

    and kd_ol_is_delete = 0

  -- если договор заключен раньше, чем запустили передачу в БКИ, то не отправляем
  project_date = (select date_begin from bki_kind bkk where bkk_code = 'ru.equifax' rows 1);

  if (approve_date is null and status in ('disb', 'close')) then
  begin

  end

  into close_oksummary_date;

  -- если есть долг в будущем, то договор нельзя закрывать даже с нулевым долгом - при передаче за исторический период могут быть сторнирования поздней датой
  is_future_debt = close_oksummary_date >= :rep_date;
  close_oksummary_date = iif(is_future_debt, const_null_date, close_oksummary_date + 1);

  if (coalesce(deal_date, '31.12.2999') < project_date and show_forced = 0) then
    exit;

  if (k_stop_perc > :rep_date) then
     k_stop_perc = null;

  -- если договор расторгнут, то выставим дату окончания договора датой расторжения
  if (k_stop_perc is not null and k_stop_perc < deal_end_date) then

  if (deal_date is null) then
  ----------------- если контракт не найден ----------------------------------
  begin
    ------  Информационная часть кредитной истории  -----------------------
    -- если заявка не подписана, то не отправляем

    begin

        select f_uuid_cbr(f_uuid('knclient', kc_id)), kc_id
        from knclient kc, kontract k
          where k_id = kc_from_k_id
            and k_from_ma_app_id = :ma_id
            and kc_from_c_id = :c_id
        into uid, kc_id;

      json.arr_set_obj(jsa, -1, jso);

      my_obj_set_s(jso, 'c_id', c_id);
      my_obj_set_s(jso, 'approve_date', approve_date);
      my_obj_set_s(jso, 'approve_datetime', approve_datetime);

      my_obj_set_s(jso, 'info_app_num', info_app_num);
      my_obj_set_s(jso, 'info_app_source_type', info_app_source_type);
      my_obj_set_s(jso, 'info_app_way', info_app_way);
      my_obj_set_s(jso, 'info_app_approval_date', info_app_approval_date);
      my_obj_set_s(jso, 'info_app_review_deadline', info_app_review_deadline);
      my_obj_set_s(jso, 'info_app_purpose_code', info_app_purpose_code);
      my_obj_set_s(jso, 'info_failure_date', info_failure_date);
      my_obj_set_s(jso, 'info_failure_datetime', info_failure_datetime);
      my_obj_set_s(jso, 'info_failure_reason', info_failure_reason);
      my_obj_set_s(jso, 'object_kind', object_kind);
      my_obj_set_s(jso, 'object_id', object_id);
      my_obj_set_s(jso, 'status', status);
      my_obj_set_s(jso, 'deal_sign_credit', deal_sign_credit);

      if (info_app_loan_sum is not null) then
      begin
        my_obj_set_s(jso, 'info_app_loan_sum', info_app_loan_sum);
        my_obj_set_s(jso, 'info_app_loan_currency', 'RUB');

      if (bkt_format = 2) then
      begin
        my_obj_set_s(jso, 'uid', uid);

      -- Блок 29_1
      pdn = 0;
      avg_load_range = '-';

      avg_income_info_source = '-';

                             when pdn <= 50 then 3
                             when pdn <= 60 then 4

                           end;
  
          select maci_is_documented

      exit;

    select count(distinct kc_from_c_id)

      and kc_from_k_id in (k_id, k_base_k_id)

    end
    else
      select sum(ok_sh_perc + ok_kommis)

      where ok_from_k_id = :k_id
        and ok_ol_is_active = 1
        and ok_type < 4
        and ok_ol_date <= :full_cost_date
      into full_cost_sum;

       full_cost_percent = 0;
    ------------------------------------------------

    if (status = 'close') then
      select min(oko_ol_date), min(ol_datetime)

      where oko_from_k_id = :k_id
        and oko_from_ol_id = ol_id
        and oko.oko_ol_is_active = 1
        and oko.oko_type = 1
        and oko_ol_date <= :rep_date
        and oko_sh_os + oko_sh_perc + oko_other > 0
      into writeoff_date, writeoff_datetime;

    if (cast(writeoff_datetime as timestamp) <> writeoff_date) then
      writeoff_datetime = cast(writeoff_date as timestamp);

    accounting_sign = iif(writeoff_date is null, 1, 0);

    -- Платежи -------------------------------------------
    delete from bki_payments_tmp where 1=1;

    paid_op_sum = 0;
    last_payout_op_sum = 0;
    paid_percent_sum = 0;
    last_payout_percent_sum = 0;
    last_payout_other_sum = 0;
    paid_other_sum = 0;
    last_payout_date = :const_null_date;
    last_payout_sum_30 = 0;     -- Сумма последнего внесенного платежа (части платежа) с просрочкой более 30 календарных дней
    paid_sum_24 = 0;            -- Сумма внесенных платежей за 24 календарных месяца, за исключением платежей с просрочкой более 30 календарных дней
  
    month_begin = dateadd(day, -extract(day from :rep_date) + 1, :rep_date);

    for with t1(ol_id, os, perc, other, payout_sum_30) as (
          select ok_from_ol_id, -ok_os, -ok_perc, -(ok_fine + ok_tax + ok_kommis), iif(ok_ol_date - ok_sh_date > 30, -(ok_os + ok_perc), 0) -- части платежей с просрочкой более 30 дней
          from operkred ok
          where ok.ok_from_k_id = :k_id
            and ok.ok_type = 4
            and ok.ok_ol_date <= :rep_date
            and ok_from_c_id = 0
          union all
          select oko_from_ol_id, -oko_os, -oko_perc, -oko_other, -(oko_os + oko_perc)
          from operkroff oko
          where oko_ol_date <= :rep_date
            and oko_from_k_id = :k_id
            and oko_type = 4
    )
    select ol_date,
          iif(ol_date <= :deal_end_date     -- до наступления ответственности поручителей, все платежи считаем общими
                or (coalesce(opl_acc_kind, 0) = 1
                    and coalesce(ol_doc_client_name, '') <> ''),
            -1, ol_document_client),
          max(ol.ol_datetime),
          sum(os), sum(perc), sum(other), sum(payout_sum_30)
    from t1
    inner join operlist ol on ol.ol_id = t1.ol_id
                          and ol.is_delete = 0
                          and ol.ol_document_client is not null -- что бы не нарваться на сторнирование (незарегистрированное)
                          and not exists(select rl_id from reversallist rl where rl.rl_old_ol_id = ol.ol_id and rl.rl_ol_is_delete = 0) -- эта операция не сторнировалась
                          and not exists(select rl_id from reversallist rl where rl.rl_new_ol_id = ol.ol_id and rl.rl_ol_is_delete = 0) -- это не стороно
    left join operloan on opl_from_ol_id = coalesce(ol.ol_parent_id, ol.ol_id)
    group by 1, 2
    order by ol_date
    into ol_date, ol_document_client, ol_datetime, os, perc, other, last_payout_sum_30
    do
    begin
      if (ol_date <> cast(ol_datetime as d_date)) then
      begin
        new_ol_datetime = null;

        select max(esm_datetime)
        from extsysmain
        where esm_from_k_id = :k_id
        and esm_action = 'pay'
        and esm_date = :ol_date
        into new_ol_datetime;

        if (new_ol_datetime is null) then
          new_ol_datetime = cast(ol_date || ' 23:59:00' as d_datetime);

        ol_datetime = new_ol_datetime;
      end 

      paid_op_sum = paid_op_sum + os;     --   Сумма внесенных платежей по основному долгу
      paid_percent_sum = paid_percent_sum + perc; -- Сумма внесенных платежей по процентам
      paid_other_sum = paid_other_sum + other;    -- Сумма внесенных платежей по иным требованиям

      if (os < 0) then os = 0;
      if (perc < 0) then perc = 0;
      if (other < 0) then other = 0;
      if (last_payout_sum_30 < 0) then last_payout_sum_30 = 0;

      if (os + perc + other > 0) then
      begin
        insert into bki_payments_tmp (bpt_c_id, bpt_date, bpt_op_sum, bpt_percent_sum, bpt_other_sum, bpt_sum_30, bpt_ol_datetime)
        values (:ol_document_client, :ol_date, :os, :perc, :other, :last_payout_sum_30, :ol_datetime);

        last_payout_date = ol_date;         --   Дата последнего внесенного платежа
        last_payout_datetime = ol_datetime;
        last_payout_op_sum = os;            --   Сумма последнего внесенного платежа по основному долгу
        last_payout_percent_sum = perc;     --   Сумма последнего внесенного платежа по процентам
        last_payout_other_sum = other;      --   Сумма последнего внесенного платежа по иным требованиям
      end
    end

    if (last_payout_op_sum is null) then last_payout_op_sum = 0;
    if (last_payout_percent_sum is null) then last_payout_percent_sum = 0;
    if (last_payout_other_sum is null) then last_payout_other_sum = 0;

    last_payout_sum = last_payout_op_sum + last_payout_percent_sum + last_payout_other_sum;   -- Сумма последнего внесенного платежа
    paid_sum = paid_op_sum + paid_percent_sum + paid_other_sum;                               -- Сумма всех внесенных платежей по обязательству
    ---------------------------------------------------------------------------

    -- Задолженность. Срочная и просроченная. ----------------------
    op_next_payout_sum = 0;
    op_next_payout_date = const_null_date;
    percent_next_payout_sum = 0;
    percent_next_payout_date = const_null_date;
    debt_current_op_sum = 0;
    debt_current_percent_sum = 0;
    future_percent_sum = 0;
    debt_current_other_sum = 0;
    debt_overdue_op_sum = 0;
    debt_overdue_percent_sum = 0;
    debt_overdue_other_sum = 0;
    debt_overdue_sum = 0;
    debt_overdue_date = const_null_date;
    debt_current_date = deal_date;
    debt_sum = 0;
    debt_current_sum = 0;
    overdue_op_miss_payout_date = const_null_date;
    overdue_perc_miss_payout_date = const_null_date;


    --- для закрытия договора(ов) для клиента-банкрота
    if (rep_date = crutch_date) then
    begin
      is_future_debt = false;
      contract_end_date = crutch_date;
      contract_end_datetime = cast(crutch_date as d_datetime);
      contract_end_reason = '16';
      status = 'close';
      status_date = crutch_date;
      debt_calc_date = crutch_date;
      debt_calc_datetime = contract_end_datetime;

      if (k_id in (101010081114601001, 101010131168700101, 101010088872901101, 101010088872901301, 101010096750200401, 101010024215900901)) then
        contract_end_reason = '17';

      if (k_id in (101010111390900201, 101010111390900301, 101010110486900101)) then
        contract_end_reason = '11';
    end
    else
    begin
      -- дата расчета: расчет когда оплата или наступила дата оплаты или прошло 30 дней после последнего расчета
      debt_sign_calc_last_payout = 1;
      last_sh_payout_date = const_null_date;    -- дата последнего платежа по графику
  
      select ok_sh_date
      from operkred
      where ok_from_k_id = :k_id
        and ok_ol_is_active = 1
        and ok_ol_date <= :rep_date
        and ok_sh_date <= :rep_date
        and ok_type in (1,2,3,4,6)
      group by 1
      having sum(ok_os + ok_sh_os) <> 0
        or sum(ok_perc + ok_sh_perc) <> 0
      order by 1 desc
      rows 1
      into last_sh_payout_date;
  
      -- надо выловить последнее движение денег, что бы принять во внимание все операции (сторно и т.д.), такие операции будем ловить начиная с 01.11.2022 - ввод в бой формата 4.0
      last_money_move_date = const_null_date;
      last_money_move_datetime = const_null_datetime;
  
      select ok_ol_date, max(ol_datetime)
      from operkred ok, operlist
      where ok.ok_ol_is_active = 1
        and ok.ok_from_k_id = :k_id
        and ok_from_ol_id = ol_id
        and ok_ol_date between '01.11.2022' and minvalue(:rep_date, iif(:close_oksummary_date = :const_null_date, :rep_date, :close_oksummary_date))  -- что бы не попали корректировки которые Улан делал после закрытия
        and ok_type in (1,2,3,4,6)
      group by 1
      having sum(ok_os + ok_sh_os) <> 0
        or sum(ok_perc + ok_sh_perc) <> 0
      union
      select oko_ol_date, ol_datetime
      from operkroff oko, operlist
      where oko_ol_is_active = 1
        and oko_from_k_id = :k_id
        and oko_from_ol_id = ol_id
        and oko_ol_date between '01.11.2022' and :rep_date
        and oko_type in (2,3,4,6)
      order by 1 desc
      rows 1
      into last_money_move_date, last_money_move_datetime;
  
      if (cast(last_money_move_datetime as date) <> last_money_move_date) then
        last_money_move_datetime = cast(last_money_move_date as timestamp);

      -- Текущая просрочка
      curr_overdue_date = null;

      select oks_delinq_date + 1
      from oksummary
      where oks_from_k_id = :k_id
        and oks_delinq_date < :rep_date
        and :rep_date between oks_date_begin and oks_date_end
        and oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc > 0
      into curr_overdue_date;

      -- дата расчета. если еще не было ни оплаты ни пропущенного платежа, то считаем на дату выдачи, учитываем дату расторжения
      debt_calc_date = maxvalue(last_money_move_date,   -- последний денежный движ
                                last_payout_date,       -- дата оплаты
                                last_sh_payout_date,    -- плановая дата платежа
                                iif(curr_overdue_date is null or last_sh_payout_date >= rep_date, const_null_date, last_sh_payout_date + 1), -- + к просрочке, если платеж не погашен, на следующий день
                                deal_date,              -- дата выдачи
                                iif(deal_end_date > :rep_date, const_null_date, deal_end_date), -- дата планового окончания
                                close_oksummary_date,   -- дата факт закрытия
                                last_debt_calc_date,   -- последняя переданная дата расчета
                                coalesce(writeoff_date, const_null_date));     --дата списания RFCRU-4944

      debt_calc_datetime = maxvalue(last_money_move_datetime,   -- последний денежный движ
                                    last_payout_datetime,       -- дата оплаты
                                    iif (last_payout_date < last_sh_payout_date, cast(last_sh_payout_date || ' 23:59:00' as d_datetime), const_null_datetime),    -- плановая дата платежа
                                    iif(curr_overdue_date is null or last_sh_payout_date >= rep_date, const_null_datetime, cast(last_sh_payout_date + 1 as d_datetime)), -- + к просрочке, если платеж не погашен, на следующий день
                                    deal_datetime,              -- дата выдачи
                                    iif(deal_end_date > :rep_date, const_null_datetime, cast(deal_end_date as d_datetime)), -- дата планового окончания
                                    cast(close_oksummary_date as d_datetime),   -- дата факт закрытия
                                    cast(last_debt_calc_date as d_datetime),   -- последняя переданная дата расчета
                                    coalesce(writeoff_datetime, const_null_datetime));     --дата списания RFCRU-4944

      if (rep_date - debt_calc_date >= 31) then
      begin
        -- рассчитывать надо каждые 30 дней
        debt_sign_calc_last_payout = 0;
        debt_calc_date = rep_date - mod(rep_date - debt_calc_date, 31);
        debt_calc_datetime = null;
      end
      else
      if (rep_date <> last_payout_date) then
      begin
        -- не день следующего платежа RFCRU-5081
        debt_sign_calc_last_payout = null;
      end
    
      for select sum(zad_os), sum(zad_perc), sum(sh_calc_perc), coalesce(sh_date, cast(:const_null_date as date)) -- в списанных бывает sh_date = null
      from aarep_make_zad_list(:k_id, :debt_calc_date, 1)
      group by sh_date
      order by sh_date
      into :zad_os, :zad_perc, :sh_calc_perc, :sh_date
      do
      begin
        if (sh_date < debt_calc_date) then
        begin
          if (zad_os > 0) then
            overdue_op_miss_payout_date = sh_date;                        -- Дата последнего пропущенного платежа по основному долгу
          if (zad_perc > 0) then
            overdue_perc_miss_payout_date = sh_date;                      -- Дата последнего пропущенного платежа по процентам
  
          debt_overdue_op_sum = debt_overdue_op_sum + zad_os;                 -- Сумма ТЕКУ�©ЕЙ просроченной задолженности по основному долгу
          debt_overdue_percent_sum = debt_overdue_percent_sum + zad_perc; -- Сумма просроченной задолженности по процентам
        end
        else
        begin
          debt_current_op_sum = debt_current_op_sum + zad_os;             -- Сумма срочной задолженности по основному долгу
          future_percent_sum = future_percent_sum + zad_perc;            -- Сумма процентов по графику (для ВСП (ССП))
          debt_current_percent_sum = debt_current_percent_sum + sh_calc_perc;  -- Срочные %%
    
          if ((op_next_payout_date = const_null_date) and
              (zad_os > 0)) then
          begin
            op_next_payout_date = sh_date; -- Дата след платежа ОС
            op_next_payout_sum = zad_os;
          end
          if ((percent_next_payout_date = const_null_date) and
              (zad_perc > 0)) then
          begin
            percent_next_payout_date = sh_date; -- Дата след платежа %%
            percent_next_payout_sum = zad_perc;
          end
        end
      end
  
      -- если выдача, то считаем будущие проценты по повышенной ставке
      if (cred_start_debt_date = rep_date) then
        future_percent_sum = base_percent_sum;
  
      if (debt_overdue_op_sum < 0) then debt_overdue_op_sum = 0;
      if (debt_overdue_percent_sum < 0) then debt_overdue_percent_sum = 0;
      if (debt_overdue_other_sum < 0) then debt_overdue_other_sum = 0;
      if (debt_current_op_sum < 0) then debt_current_op_sum = 0;
      if (future_percent_sum < 0) then future_percent_sum = 0;
      if (debt_current_percent_sum < 0) then debt_current_percent_sum = 0;
  
      regularity = iif(op_next_payout_sum > 0 or percent_next_payout_sum > 0, '3', null);   -- регулярность платежа - ежемесячно, если есть следующий платеж
      debt_current_sum = debt_current_op_sum + debt_current_percent_sum + debt_current_other_sum; -- Сумма срочной задолженности
      debt_overdue_sum = debt_overdue_op_sum + debt_overdue_percent_sum + debt_overdue_other_sum;   -- Сумма просроченной задолженности

      if (debt_calc_date = last_sh_payout_date  -- если расчет на дату погашения по графику
        and debt_overdue_sum + debt_current_sum = 0) -- а долга нет, т.е. текущий по графику уже оплачен
      then  -- то датой расчета будет последний платеж (т.к. в этом случае он явно был)
      begin
        debt_calc_date = last_payout_date;
        debt_calc_datetime = last_payout_datetime;
        debt_calc_date = rep_date - mod(rep_date - debt_calc_date, 31);
      end
      else if (rep_date = op_next_payout_date) then  -- наступила дата платежа
        debt_calc_date = rep_date;

      -- Дата возникновения ПОСЛЕДНЕЙ просроченной задолженности
      -- Просрочка начинается на следующий день после пропущенного платежа (RFCRU-1059)
      select coalesce(max(oks_delinq_date) + 1, :const_null_date)
      from oksummary
      where oks_from_k_id = :k_id
        and oks_delinq_date < :debt_calc_date
        and oks_date_begin <= :debt_calc_date
        and (oks_date_end <> oks_date_begin or oks_delinq_date + 1 <> oks_date_begin)   -- игнорим долг если он был на след. день после платежа и схлопнулся в тот же день
        and oks_delinq_os + oks_delinq_perc > 0
      into debt_overdue_date;

      -- Сумма внесенных платежей за 24 календарных месяца, за исключением платежей с просрочкой более 30 календарных дней
      month_begin = dateadd(day, -extract(day from :debt_calc_date) + 1, :debt_calc_date);

      select coalesce(sum(bpt_op_sum + bpt_percent_sum - bpt_sum_30), 0)
      from bki_payments_tmp
      where bpt_date between dateadd(month, -24, :month_begin) and :month_begin - 1
      into paid_sum_24;

      -- Дата последнего пропущенного платежа по основному долгу, если дата рассчитана неверно
      if (debt_overdue_op_sum > 0 and overdue_op_miss_payout_date = const_null_date) then
        if (accounting_sign = 0) then  -- списан
          select max(ok.ok_sh_date)    -- максимальная дата попавшая в списание
          from operkred ok
          where ok.ok_type = 5
          and ok.ok_ol_is_active = 1
          and ok.ok_from_k_id = :k_id
          and ok.ok_os <> 0           -- по ОС
          into overdue_op_miss_payout_date;
        else
          overdue_op_miss_payout_date = last_sh_payout_date;

      if (debt_overdue_percent_sum > 0 and overdue_perc_miss_payout_date = const_null_date) then
        overdue_perc_miss_payout_date = overdue_op_miss_payout_date;

      -- Дата последнего пропущенного платежа по процентам, если дата рассчитана неверно
      if (debt_overdue_percent_sum > 0 and overdue_perc_miss_payout_date = const_null_date) then
        if (accounting_sign = 0) then  -- списан
          select max(ok.ok_sh_date)    -- максимальная дата попавшая в списание
          from operkred ok
          where ok.ok_type = 5
          and ok.ok_ol_is_active = 1
          and ok.ok_from_k_id = :k_id
          and ok.ok_perc <> 0          -- по %%
          into overdue_perc_miss_payout_date;
        else
          overdue_perc_miss_payout_date = last_sh_payout_date;
      -----------------------------------------------

      -- если есть долг но договор закрыт то обнуляем долг (есть старые косячные договоры которые закрыты, но в них остался долг)
      if (debt_current_sum + debt_overdue_sum > 0 and not is_future_debt) then
      begin
        debt_current_sum = 0;
        debt_current_op_sum = 0;
        debt_current_percent_sum = 0;
        debt_current_other_sum = 0;

        debt_overdue_sum = 0;
        debt_overdue_op_sum = 0;
        debt_overdue_percent_sum = 0;
        debt_overdue_other_sum = 0;
      end

      if (op_next_payout_sum + percent_next_payout_sum > 0 and not is_future_debt) then
      begin
        op_next_payout_sum = 0;
        percent_next_payout_sum = 0;
        op_next_payout_date = const_null_date;
        percent_next_payout_date = const_null_date;
      end
    end
  
    debt_op_sum = debt_current_op_sum + debt_overdue_op_sum;               -- Сумма задолженности по основному долгу
    debt_percent_sum = debt_current_percent_sum + debt_overdue_percent_sum;-- Сумма задолженности по процентам
    debt_other_sum = debt_current_other_sum + debt_overdue_other_sum;      -- Сумма задолженности по иным требованиям
    debt_sum = debt_op_sum + debt_percent_sum + debt_other_sum;            -- Сумма задолженности
  
    overdue_day = 0;
    last_overdue_day = 0;
    debt_overdue_date_paid = const_null_date;

    if (debt_overdue_date > const_null_date) then
    begin
      if (debt_overdue_sum = 0) then -- если просрочки нет, то значит последняя просрочка погашена
        debt_overdue_date_paid = debt_overdue_date;
      else
      begin
        overdue_day = debt_calc_date - debt_overdue_date + 1;                -- Даты начала просрочки и расчета обе включаются (RFCRU-1059)

        -- найдем дату предыдущей просрочки
        select coalesce(max(oks_delinq_date) + 1, :const_null_date)
        from oksummary
        where oks_from_k_id = :k_id
          and oks_delinq_date < :debt_overdue_date - 1
          and oks_date_begin <= :debt_overdue_date - 1
          and oks_delinq_os + oks_delinq_perc > 0
        into debt_overdue_date_paid;
      end
    end

    if (debt_overdue_date_paid > const_null_date) then  -- найдем продолжительность последней погашенной просрочки
      select coalesce(min(oks_date_begin), :rep_date) - :debt_overdue_date_paid + 1  -- если запись не найдена, то значит просрочка до последнего
      from oksummary
      where oks_from_k_id = :k_id
        and oks_date_begin between :debt_overdue_date_paid and :rep_date
        and oks_delinq_os + oks_delinq_perc = 0
      into last_overdue_day;
    --------------------------------------------------------------------
  
    -- если долга нет, то договор закрыт датой последнего платежа
    if (debt_sum = 0 and not is_future_debt and rep_date is distinct from crutch_date) then
    begin
      contract_end_date = maxvalue(last_payout_date, close_oksummary_date);
      contract_end_datetime = maxvalue(last_payout_datetime, cast(close_oksummary_date as d_datetime));
      contract_end_reason = '1'; -- Надлежащее исполнение обязательства

      if (k_id = 101010070923801801 and rep_date = '01.12.2025') then
      begin
        contract_end_date = null;
        contract_end_datetime = null;
        contract_end_reason = null;
      end
    end

    size_payments_type = case                                                -- Код соблюдения размера платежей
                            when last_payout_date = const_null_date then 3   -- Платеж не внесен (ни разу)
                            when debt_overdue_sum = 0 then 1                 -- Платеж внесен в полном размере
                            else 2                                           -- Платеж внесен не в полном размере
                         end;
  
    payments_deadline_type = case                   -- Код соблюдения срока внесения платежей
                                 when (last_payout_date = const_null_date and debt_overdue_sum = 0) then 1      -- Срок внесения платежа не наступил (новый договор)
                                 when debt_overdue_sum = 0 then 2      -- Платежи вносятся своевременно
                                 else 3             -- Платежи вносятся несвоевременно
                             end;
  
  
    -- Величина среднемесячного платежа -------------
    average_payment_sum = 0;
    sum_total_remain = 0;

    --- найдем дату на которую рассчитываем показатель
    average_payment_date = debt_calc_date;

    if (debt_sum > 0) then
    begin
      sum_total_remain = debt_current_op_sum + future_percent_sum;
      months = (extract(year from deal_end_date) - extract(year from average_payment_date)) * 12 +
                        extract(month from deal_end_date) - extract(month from average_payment_date) +
                        iif(extract(day from deal_end_date) > extract(day from average_payment_date), 1, 0);
      average_payment_sum = sum_total_remain / cast((iif(months <= 0, 1, months)) as d_currency) + debt_overdue_sum;
    end
    --------------------------------------

    if (debt_overdue_sum > 0 or is_future_debt) then
       status = trim(iif(accounting_sign = 0, 'writeoff', 'disb'));
  
    is_obesp = false;
  
    -- проверять обеспеченный заем или нет имеет смысл проверять только если это групповой заем или займ как-то связан амнистией
    if ((joint_debtors_count > 1) or (k_orig_id is not null))  then
    begin
        select is_secured = 1   -- 1 - обеспеченный, 0 - не обеспеченный
        from aarep_get_secured_loan(null, null, null, null, null, null, :k_id)
        into is_obesp;
    end

    if (is_obesp) then  -- од + %% по первичному графику
      select sum(sh_os + sh_perc)
      from schedule
      where sh_from_k_id = :k_id
      into contract_full_first_sum;

    info_credit_date = deal_date;

    if (joint_debtors_count > 1) then
    begin
      ------------------------------------------------------
      -- Далее суммы будут делиться между участниками
      op_next_payout_sum_tmp       = op_next_payout_sum;
      percent_next_payout_sum_tmp  = percent_next_payout_sum;
      full_cost_sum_tmp            = full_cost_sum;
      debt_current_op_sum_tmp      = debt_current_op_sum;
      debt_current_percent_sum_tmp = debt_current_percent_sum;
      debt_current_other_sum_tmp   = debt_current_other_sum;
      debt_overdue_op_sum_tmp      = debt_overdue_op_sum;
      debt_overdue_percent_sum_tmp = debt_overdue_percent_sum;
      debt_overdue_other_sum_tmp   = debt_overdue_other_sum;
      average_payment_sum_tmp      = average_payment_sum;
      contract_full_first_sum_tmp  = contract_full_first_sum;
      -------------------------------------------------------------------
    end
