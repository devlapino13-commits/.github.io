CREATE PROCEDURE HP_BKI_SCORING_CHECK(
    JSO D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    SCORE_LEVEL D_SMALLINT,
    RES_TXT D_STR_255,
    CARD_ID D_INTEGER,
    SCORE D_INTEGER)
AS
declare variable bsl_level1 d_integer;
  declare variable bsl_level2 d_integer;
begin
  score_level = -1;
  res_txt = 'Без оценки';

  if (json.contains(:jso, 'response.basePart.scoring') = 0 or json.val_get_type_name(:jso, 'response.basePart.scoring') <> 'Array') then
  begin
    suspend;
    exit;
  end

  select json.obj_get_val(jso, 'scorCardId'), json.obj_get_val(jso, 'score')
  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'response.basePart.scoring'), 1)
  where json.obj_get_val(jso, 'scorDate') <= :rep_date
    and json.obj_get_val(jso, 'scorCardId') in (75, 76)
  order by json.obj_get_val(jso, 'scorDate') desc, json.obj_get_val(jso, 'scorCardId')
  rows 1
  into card_id, score;

  if ((card_id + score) is null) then
  begin
    suspend;
    exit;
  end

  if (card_id = 76) then  -- бюро не может провести оценку
  begin
    suspend;
    exit;
  end

  select bsl_level1, bsl_level2
  from bki_scoring_levels
  where bsl_date_begin <= :rep_date
  order by bsl_date_begin desc
  rows 1
  into bsl_level1, bsl_level2;

  if (bsl_level1 is null) then -- нет критериев для оценки
  begin
    suspend;
    exit;
  end

  if (score >= bsl_level2) then
  begin
    score_level = 2;
    res_txt = 'Положительный';

    suspend;
    exit;
  end

  if (score < bsl_level1) then
  begin
    score_level = 0;
    res_txt = 'Отрицательный';

    suspend;
    exit;
  end

  score_level = 1;
  res_txt = 'Средний';

  suspend;
end
