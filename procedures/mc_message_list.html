CREATE PROCEDURE MC_MESSAGE_LIST(
    MCL_ID D_ID,
    LAST_READ_UPDATE D_SMALLINT = 1)
RETURNS (
    MSG_ID D_ID,
    MSG_DATETIME STR_100,
    MSG_BODY D_BLOB_TEXT)
AS
declare variable c_id d_id;
declare variable eds_id d_id;
declare variable edl_id d_id;
declare variable is_show d_bool;
declare variable date_begin d_date;
declare variable mcl_project d_smallint;
declare variable db_id d_integer;
declare variable mcl_login d_str_code;
declare variable penult_mcl_id d_id;
begin
  db_id = f_get_db_id();

  select
    ml.mcl_from_c_id,
    ml.mcl_project,
    ml.mcl_login
  from mc_login ml
  where ml.mcl_id = :mcl_id
  into
    c_id,
    mcl_project,
    mcl_login;

  select first 1 mcl.mcl_id
  from mc_login mcl
  where mcl.mcl_from_c_id = :c_id and
        mcl.mcl_login starting :mcl_login and
        mcl.mcl_is_delete = 1 and
        mcl.mcl_project = :mcl_project
  order by mcl.mcl_id desc
  into :penult_mcl_id;

  date_begin = current_date - 30;

  for with
    user_topics as (
      select '/topics/' || topic_name as topic_name
      from hp_fcm_user_topics(null, :mcl_id)
    ),
    msg as (
      -- RFCRU-3766
      -- Уведомления пользователю через mcl_id
      select msg_id, msg_body, msg_datetime
      from fsrobot_msgbox
      where msg_object_kind = f_objkind('mc_login', 'id') and
            msg_object_id in (:mcl_id, :penult_mcl_id) and
            msg_type = 2 and
            msg_status >= 0 and
            msg_datetime >= :date_begin
      union
      -- Уведомления пользователю через c_id
      select msg_id, msg_body, msg_datetime
      from fsrobot_msgbox
      where msg_object_kind = f_objkind('clients', 'id') and
            msg_object_id = :c_id and
            msg_type = 2 and
            msg_status >= 0 and
            msg_datetime >= :date_begin
      union
      -- Уведомления пользователю через kc_id
      select msg_id, msg_body, msg_datetime
      from knclient, kontract, product, fsrobot_msgbox
      where kc_from_c_id = :c_id and
            msg_object_kind = f_objkind('knclient', 'id') and
            msg_object_id = kc_id and
            msg_type = 2 and
            msg_status >= 0 and
            msg_datetime >= :date_begin and
            kc_from_k_id = k_id and
            k_from_pr_id = pr_id and
            iif(:mcl_project = 1,pr_code not containing '.oivo_pack.', pr_code containing '.oivo_pack.')
      union
      -- Уведомления через топики (группы рассылок), пользователь должен быть подписан
      select msg_id, msg_body, msg_datetime
      from fsrobot_msgbox, user_topics
      where msg_object_kind = f_objkind('clients', 'id') and
            msg_object_id = 1000000000000 and
            msg_mail_to = topic_name and
            msg_type = 2 and
            msg_status >= 0 and
            msg_datetime >= :date_begin
      union
      -- RFCKZ-894
      select msg_id, msg_body, msg_datetime
      from (
        select
            msg_id,
            msg_body,
            msg_datetime,
            row_number() over (order by msg_datetime desc) as rn
        from fsrobot_msgbox, mc_login
        where :db_id = 39801
          and mcl_from_c_id = :c_id
          and mcl_project = :mcl_project
          and msg_object_kind = f_objkind('mc_login', 'id')
          and msg_object_id = mcl_id
          and msg_type = 2
          and msg_status >= 0
          and msg_datetime >= :date_begin
          and msg_subject containing 'Ознакомление с согласиями'
      ) t
      where :db_id = 39801
        and rn = 1
    )
    select  msg_id,
            '{' || msg_body || '}',
            f_datetimetostr(msg_datetime)
    from msg
    order by msg_datetime desc
    into msg_id, msg_body, msg_datetime
  do
  begin
    is_show = 1; -- по умолчанию уведомления показывать
    eds_id = cast(f_json_str_get_value(msg_body, 'data.eds_id', '0') as bigint);

    if (eds_id > 0) then
    begin
      select eds_from_edl_id
      from edoc_sign
      where eds_id = :eds_id
      into edl_id;

      select is_show
      from mc_edoc_sign_status(:edl_id, :c_id)
      into is_show;
    end

    if (is_show = 1) then
      suspend;
  end

  if (last_read_update = 1) then
    execute procedure mc_last_read_update(:mcl_id);
end
