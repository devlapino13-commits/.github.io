CREATE PROCEDURE AAREP_CEO_MONTHLY(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(50),
    SVZ_KOL_ACT INTEGER,
    SVZ_DISB INTEGER,
    DISB_KOL_KRED INTEGER,
    DISB_KOL_PARALEL INTEGER,
    DISB_KOL_VSEGO INTEGER,
    SVZ_ONE_DISB CURR_TYPE,
    PAR1 CURR_TYPE,
    PAR7 CURR_TYPE,
    PAR30 CURR_TYPE,
    PERC_UDERJ CURR_TYPE,
    REP_BALANCE CURR_TYPE,
    PRIROST_PORTFEL CURR_TYPE,
    KOL_CLI INTEGER,
    PRIROST_CLI INTEGER)
AS
declare variable reg_unicode integer;
declare variable rep_date_begin date;
declare variable week_day smallint;
begin
  rep_date_begin = :rep_date_end - extract(day from :rep_date_end) + 1;

  for select res_region_id, svz_kol_act, svz_disb, disb_kol_kred, disb_kol_paralel, disb_kol_vsego, svz_one_disb, par1,
             par7, par30, perc_uderj, rep_balance, prirost_portfel, kol_cli, prirost_cli
      from aarep_info_office(:reg_unicode, :rep_date_begin, :rep_date_end)
      into reg_unicode, svz_kol_act, svz_disb, disb_kol_kred, disb_kol_paralel, disb_kol_vsego, svz_one_disb, par1,
           par7, par30, perc_uderj, rep_balance, prirost_portfel, kol_cli, prirost_cli
  do
  begin
    select reg_unicode, reg_name
    from region
    where reg_unicode = :reg_unicode
    into reg_unicode, office_name;
    suspend;
  end
