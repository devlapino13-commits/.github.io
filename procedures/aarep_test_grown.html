CREATE PROCEDURE AAREP_TEST_GROWN(
    DD DATE)
RETURNS (
    C_NAME_FIO INTEGER,
    REP_MONTH INTEGER,
    KOL INTEGER)
AS
declare variable c_id decimal(18,0);
begin
  rep_month = extract(month from :dd);
  for select ks_kc_id, sum(k_kolich_client_approved)
      from oksummary, kontract, kontstate, product
      where oks_from_k_id = k_id and
            oks_date_begin <= :dd and
            oks_date_end >= :dd and
            oks_balance > 0 and
            oks_kc_id = ks_kc_id and
            ks_state = 1 and
            ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            ks_date_begin <= :dd and
            ks_date_end >= :dd and
            k_from_pr_id = pr_id and
            pr_product_kind <> 2 and
            oks_close_state = 0
      group by ks_kc_id
      into :c_id, kol
  do
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into :c_name_fio;
    suspend;
  end
end
