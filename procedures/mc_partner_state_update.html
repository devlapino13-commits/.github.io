CREATE PROCEDURE MC_PARTNER_STATE_UPDATE(
    MCL_ID D_ID,
    NEW_STATE D_STR_CODE)
RETURNS (
    RES_TEXT D_STR_255,
    RES_CODE D_BOOL)
AS
declare variable ag_id d_integer;
declare variable date_end d_date;
declare variable c_id d_id;
declare variable rd_id d_integer;
declare variable ae_is_exchange d_bool;
declare variable new_is_exchange d_integer;
begin
  /* Процедура обновляет обменять баллы или копить баллы*/
  date_end = current_date;
  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  if (coalesce(new_state,'') not in ('save_up','exchange')) then
    exception e_default 'parametr "new_state" has invalid value';

  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :c_id;

  select rd_id from refdata
  where rd_code = 'AgreementKinds.loyal_agreement'
  into rd_id;

  ag_id = null;
  select ag_id from agreements -- в этом отрезке нужно добавить дополнительное условие на mcl_id
  where ag_object_kind = 0
  and ag_object_id = :c_id
  and ag_status = 1
  and ag_from_rd_id = :rd_id
  and :date_end between ag_date_begin and coalesce(ag_date_end, '31.12.9999')
  order by ag_id desc
  rows 1
  into ag_id;

  ae_is_exchange = 0;
  select ae_is_exchange from agreement_exchange
  where ae_from_ag_id = :ag_id
  into ae_is_exchange;
  ae_is_exchange = coalesce(ae_is_exchange,0);

  new_is_exchange = 0;
  if (new_state = 'exchange') then new_is_exchange = 1;

  res_text = 'Статус не изменен!';
  res_code = 0;
  if (new_is_exchange <> ae_is_exchange) then
  begin
    update or insert into agreement_exchange (ae_from_ag_id, ae_is_exchange)
    values (:ag_id, :new_is_exchange)
    matching (ae_from_ag_id);
    res_text = 'Статус успешно изменен!';

    if (new_state = 'save_up') then
      res_text = 'Благодарим за выбор. Не забывайте соблюдать условия для сохранения статуса Партнера.';
    else if (new_state = 'exchange') then
      res_text = 'Благодарим за выбор. Деньги в обмен на бонусы вам будут переведены 17-19 числа. Пожалуйста, до 15 числа актуализируйте ваши документы через КМП или через вашего Специалиста.';
  end
  suspend;
end
