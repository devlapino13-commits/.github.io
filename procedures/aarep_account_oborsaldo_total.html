CREATE PROCEDURE AAREP_ACCOUNT_OBORSALDO_TOTAL(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    ACC_CLASS VARCHAR(255),
    BEG_DT DECIMAL(18,4),
    BEG_KT DECIMAL(18,4),
    OB_DT DECIMAL(18,4),
    OB_KT DECIMAL(18,4),
    END_DT DECIMAL(18,4),
    END_KT DECIMAL(18,4))
AS
begin
  beg_dt = 0;
  beg_kt = 0;
  ob_dt = 0;
  ob_kt = 0;
  end_dt = 0;
  end_kt = 0;
  for select acc_class, sum(beg_dt), sum(beg_kt), sum(ob_dt), sum(ob_kt), sum(end_dt), sum(end_kt)
      from aarep_account_oborsaldo(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency)
      where acc_kind = 3
      group by acc_class
      order by acc_class
      into :acc_class, :beg_dt, :beg_kt, :ob_dt, :ob_kt, :end_dt, :end_kt
  do
  begin
    suspend;
  end
  for select sum(beg_dt), sum(beg_kt), sum(ob_dt), sum(ob_kt), sum(end_dt), sum(end_kt)
      from aarep_account_oborsaldo(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency)
      where (acc_kind = 2 or
            acc_kind = 3)
      into :beg_dt, :beg_kt, :ob_dt, :ob_kt, :end_dt, :end_kt
  do
  begin
    acc_class = 'ИТОГО:';
    suspend;
  end
end
