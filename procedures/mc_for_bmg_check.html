CREATE PROCEDURE MC_FOR_BMG_CHECK(
    MCL_ID D_ID)
RETURNS (
    INN D_STR_20,
    PHONE_NUMBER D_STR_20)
AS
begin
  select cl.c_inn,
         f_cont_info('clients', cl.c_id, 'cell_main', 'ok')
  from mc_login ml
  inner join clients cl on cl.c_id = ml.mcl_from_c_id
  where ml.mcl_id = :mcl_id
  into :inn,
       :phone_number;

  inn = coalesce(inn, '');
  phone_number = coalesce(phone_number, '');

  suspend;
end
