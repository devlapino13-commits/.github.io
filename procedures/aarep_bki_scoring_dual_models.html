CREATE PROCEDURE AAREP_BKI_SCORING_DUAL_MODELS(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    FIO D_STR_255,
    BIRTH_DATE D_DATE,
    C_ID D_ID,
    APP_NO D_STR_CODE,
    APP_DATE D_DATE,
    PR_NAME D_STR_CODE,
    KO_DATE D_DATE,
    KONTRACT_NO D_STR_CODE,
    SC_71 D_STR_CODE,
    SC_72 D_STR_CODE,
    SC_75 D_STR_CODE,
    SC_76 D_STR_CODE)
AS
declare variable ma_id d_id;
  declare variable bls_id d_id;
  declare variable json_full d_blob_text;
  declare variable jsa d_integer;
  declare variable card_id d_integer;
  declare variable score d_integer;
  declare variable json_score d_str_8100;
begin
--- RFCRU-3736     Отчет по скорингам
-- Поля: ФИО, дата рождения, id клиента, номер заявки, дата заявки (дата ответа от БКИ?), номер договора? (при наличии),
-- результат из БКИ по обоим моделям (sc 71,72,75,76).
  for select c_name_fio, c_birth_date, c_id, ma_no, ma_date_app, ma_id, pr_name
  from miniapp, miniappclient, clients, product
  where mac_from_ma_id = ma_id
    and c_id = mac_from_c_id
    and pr_id = ma_from_pr_id
    and ma_date_app between :rep_date_begin and :rep_date_end
    and ma_number = 0
  into fio, birth_date, c_id, app_no, app_date, ma_id, pr_name
  do
  begin
    kontract_no = null;
    ko_date = null;
    sc_71 = null;
    sc_72 = null;
    sc_75 = null;
    sc_76 = null;
    bls_id = null;
    json_full = null;

    select k_kontract_no, pr_name
    from kontract, product
    where pr_id = k_from_pr_id
      and k_from_ma_app_id = :ma_id
    into kontract_no, pr_name;

    select eil_date, eil_from_bls_id
    from extinfolist
    where eil_from_eic_id = 14
      and eil_date >= :app_date
      and eil_from_c_id = :c_id
    order by eil_date, eil_id
    rows 1
    into ko_date, bls_id;

    if (bls_id is not null) then
    begin
      select blx_blob
      from hp_get_blobex_blob(:bls_id)
      into json_full;

      if (coalesce(json_full, '') = '') then
        continue;

      in autonomous transaction do
      begin
        json_score = '{"arr":' || f_json_blob_get_value(json_full, 'response.basePart.scoring', null) || '}';

        if (coalesce(json_score, '') = '') then
          continue;

        jsa = json.obj_get_arr(json.parse(json_score), 'arr');
  
        for select json.obj_get_val(jso, 'scorCardId'), json.obj_get_val(jso, 'score')
        from hp_bki_json_get_list(:jsa, 1)
        into card_id, score
        do
        begin
          if (card_id = 71) then
            sc_71 = score;
          else if (card_id = 72) then
            sc_72 = score;
          else if (card_id = 75) then
            sc_75 = score;
          else if (card_id = 76) then
            sc_76 = score;
        end
      end
      suspend;
    end
  end
end
