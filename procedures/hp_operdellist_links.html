CREATE PROCEDURE HP_OPERDELLIST_LINKS(
    IN_ODM_ID D_INTEGER,
    IN_K_ID D_ID)
RETURNS (
    ODL_ID D_INTEGER,
    ODL_FROM_ODM_ID D_INTEGER,
    ODL_FROM_K_ID D_ID,
    ODL_OLD_KKM_ID D_ID,
    ODL_MAIN_OL_ID D_ID,
    ODL_LOAN_OL_ID D_ID,
    ODL_NEW_OL_ID D_ID,
    ODL_NEW_AMOUNT D_CURRENCY,
    ODL_NEW_ACCOUNT D_STR_20,
    ODL_NEW_REG_ID D_INTEGER,
    ODL_NEW_ACTION D_SMALLINT,
    ODL_IS_CHECKED D_BOOL,
    ODL_STATE D_SMALLINT,
    RES_OL_DATE D_DATE,
    RES_OL_MEMO D_STR_LONG,
    RES_KONTRACT_NO D_STR_CODE,
    RES_IS_MAIN D_BOOL,
    RES_PAYMENT_KIND_STR D_STR_CODE,
    RES_OPER_CODE D_STR_CODE,
    RES_OPER_NAME D_STR_100)
AS
declare variable odl_new_payment_kind d_smallint;
declare variable ol_oper_type d_smallint;
declare variable ol_oper_subtype d_smallint;
begin
  for select odl_id, odl_from_odm_id, odl_from_k_id, odl_old_kkm_id, odl_main_ol_id, odl_loan_ol_id, odl_new_ol_id, odl_new_amount,
             odl_new_account, odl_new_reg_id, odl_new_action, odl_is_checked, odl_state, odl_new_payment_kind,
             ol_date, ol_memo, k_kontract_no, iif(odl_main_ol_id = odm_main_ol_id, 1, 0),
             ol_oper_type, ol_oper_subtype
  from operdellist
       inner join operdelmain on (odl_from_odm_id = odm_id)
       inner join operlist on (odl_loan_ol_id = ol_id)
       inner join kontract on (odl_from_k_id = k_id)
  where odl_from_odm_id = :in_odm_id and
        odl_from_k_id = :in_k_id
  order by ol_date, odl_loan_ol_id
  into :odl_id, :odl_from_odm_id, :odl_from_k_id, :odl_old_kkm_id, :odl_main_ol_id, :odl_loan_ol_id, :odl_new_ol_id, :odl_new_amount,
       :odl_new_account, :odl_new_reg_id, :odl_new_action, :odl_is_checked, :odl_state, odl_new_payment_kind,
       :res_ol_date, :res_ol_memo, :res_kontract_no, res_is_main,
       :ol_oper_type, :ol_oper_subtype
  do
  begin
    res_payment_kind_str = trim(case odl_new_payment_kind
      when 1 then 'Касса'
      when 2 then 'Банк'
      when 3 then 'Онлайн'
      else '-'
    end);

    res_oper_code = '';
    res_oper_name = '';

    select res_code, res_name
    from hp_operdel_operkind(:odl_loan_ol_id, :ol_oper_type, :ol_oper_subtype)
    into :res_oper_code, :res_oper_name;

    suspend;
  end
end
