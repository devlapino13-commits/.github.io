CREATE PROCEDURE HP_LLC_UPDATE_CALCPERC_AZ(
    REP_DATE_END DATE)
RETURNS (
    LK_FROM_LM_ID ID_TYPE,
    LK_FROM_K_ID ID_TYPE,
    LK_PERCENT CURR_TYPE,
    SH_CALC_PERC CURR_TYPE,
    REP_PERIOD INTEGER)
AS
begin
  rep_period = extract(year from cast(:rep_date_end as date)) || right('0' || extract(month from cast(:rep_date_end as date)), 2);
  for select lk_from_lm_id, lk_from_k_id, lk_percent
      from llc_main, llc_kont, kontstate
      where lm_is_signed = 1 and
            lm_id = lk_from_lm_id and
            :rep_date_end between lk_date and lk_date_end and
            lk_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end
      into lk_from_lm_id, lk_from_k_id, lk_percent
  do
  begin
    sh_calc_perc = null;
    /*   select sum(sh_calc_perc)
    from aarep_make_zad_list(:lk_from_k_id, :rep_date_end, 1)
--    where sh_date <= :rep_date_end
    into sh_calc_perc; */

    if (:rep_date_end >= '01.02.2015') then -- с этой даты РППУ начисляется и на проценты
    begin
      select kcp_perc_saldo
      from kontcalcperc
      where kcp_from_k_id = :lk_from_k_id and
            kcp_period = :rep_period
      into sh_calc_perc;
      if (sh_calc_perc is null) then
        sh_calc_perc = 0;
    end
    if (sh_calc_perc is null) then
      sh_calc_perc = 0;

    if (sh_calc_perc > 0) then
    begin
      update llc_kont
      set lk_calcperc = :sh_calc_perc
      where lk_from_lm_id = :lk_from_lm_id and
            lk_from_k_id = :lk_from_k_id;
    end

    suspend;
  end
end
