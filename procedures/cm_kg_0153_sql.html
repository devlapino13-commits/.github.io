CREATE PROCEDURE CM_KG_0153_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable acl_status integer;
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable pr_code d_str_100;
declare variable rd_id d_integer;
begin

  select kal_new_reg_id, kal_new_kf_officer, pr.pr_code
  from kontract kt
  inner join knaktlist kal on kal.kal_from_kf_id = kt.k_from_kf_id
  inner join product pr on pr.pr_id = kt.k_from_pr_id and
        pr.pr_id > 0
  where kt.k_id = :k_id and
        current_date between kal.kal_date and kal.kal_date_end and
        kal.is_delete = 0
  into :reg_id, :c_id, :pr_code;

  -- Для продукта с данными pr_code проверка КК не требуется
  if (pr_code containing '.express_similar.' or pr_code containing '.express_online.') then
    exit;

  if (exists(select *
             from underwritingcertif
             inner join underwritingcertifspf on ucfs_from_reg_id = ucf_from_reg_id and
                   ucfs_from_c_id = :c_id and
                   current_date between ucfs_date_begin and ucfs_date_end and
                   ucfs_active = 1
             where ucf_from_reg_id = :reg_id and
                   current_date between ucf_date_begin and ucf_date_end and
                   ucf_system = f_enum_id('access_sytem', 'undervriter_certif') and
                   ucf_active = 1)) then
  begin


    select rd_id
    from hp_get_check_loan_data(:k_id)
    into:rd_id;

    --acl_status = (select acl_status from hp_actionlist_kontract_get(:k_id));
    acl_status = (select first 1 iif(acl_datetime is null, -3, acl_status)
                  from actionlist
                  where acl_object_kind = 9 and
                        acl_object_id = :k_id and
                        acl_from_rd_id = :rd_id
                  order by acl_id desc);

    res_error_text = case acl_status
                       when -3 then 'Данные еще не отправлены на сертификацию'
                       when -2 then 'Отказано сертификатором'
                       when -1 then 'Сертификатором отправлено на доработку'
                       when 0 then 'В ожидании сертификации'
                       when 1 then 'На рассмотрении у сертификатора'
                       else ''
                     end;
    if (res_error_text <> '') then
    begin
      res_error_is_critical = 2;
      suspend;
    end
  end
end
