declare variable c_id1 id_type;
declare variable ext_code integer;

declare variable pl_id id_type;

declare variable pl_name str_100;
declare variable plt_name str_100;
declare variable num_node d_integer;

begin
  res_full_xml = '';
  ext_code = 10; ext_passport = 'Паспорт иностранного гражданина';
  int_code = 21; int_passport = 'Паспорт гражданина Российской Федерации';

  res_temp_xml =
  '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
  <PledgeNotificationToNotary xmlns="http://fciit.ru/eisn/ruzdi/types/2.3" version="2.3" SendType="2" >

                    <URN>1107746627779</URN>
                    <UINN>7728744752</UINN>

                </Organization>

                    <BirthDate>1991-10-19</BirthDate>
                    <BaseAuthority>
                        <Name>Доверенность в машиночитаемом виде</Name>

                        <IssuerInn>7728744752</IssuerInn>
                        <IssueDate>2025-02-26</IssueDate>
                        <FileDesc>
                            <File>
                                <FileLocation>ON_EMCHD_20250226_997fb76d-ac28-45fe-91f5-4766b867e57d.xml</FileLocation>
                                <Md5sum>F290F8F8F1FD6682E3A111FA1548B189</Md5sum>
                            </File>
                            <Signature>
                                <FileLocation>ON_EMCHD_20250226_997fb76d-ac28-45fe-91f5-4766b867e57d.xml.sig</FileLocation>
                                <Md5sum>2551F02DD8E8F1CBA09233467D50CFEE</Md5sum>
                            </Signature>
                            <AttachmentId>5d638520-83a3-4253-92bb-f4b4d8565fee</AttachmentId>
                        </FileDesc>
                    </BaseAuthority>
                    <PersonDocument>
                        <Code>21</Code>
                        <Name>Паспорт гражданина Российской Федерации</Name>
                        <SeriesNumber>36 11 538894</SeriesNumber>
                    </PersonDocument>
                    <PersonAddress>
                        <AddressRF registration="true">
                            <RegionCode>63</RegionCode>
                            <Region>Самарская область</Region>
                            <City>г. Самара</City>
                            <Street>ул. Ново-Садовая</Street>
                            <House>дом 38</House>

                    </PersonAddress>
                </Attorney>

        </FormUZ1>
    </NotificationData>
  </PledgeNotificationToNotary>';

  for
    select first 1
      z_id,
      k_id,
      k_kontract_date, -- RFCRU-4549
      k_kredline_end,
      k_last_reg_id,
      kc_from_c_id,
      c_id,
      c_name_fio,
      left(ru.fl_name1,1) || lower(substring(ru.fl_name1 from 2)),
      left(ru.fl_name2,1) || lower(substring(ru.fl_name2 from 2)),
      left(ru.fl_name3,1) || lower(substring(ru.fl_name3 from 2)),
      c_alternat_fio,
      left(en.fl_name1,1) || lower(substring(en.fl_name1 from 2)),
      left(en.fl_name2,1) || lower(substring(en.fl_name2 from 2)),
      left(en.fl_name3,1) || lower(substring(en.fl_name3 from 2)),
      c_birth_date,
      replace(c_pass_no,' ','')
    from kontract
      inner join product
        on pr_id = k_from_pr_id
          and (pr_code containing '.ZALOG_CAR.' or pr_code containing '.zalog.')
      inner join knclient
        on kc_id = (select first 1 kc_id from knclient
          where kc_from_k_id = k_id and kc_amount_approved > 0)
      inner join zalog
        on z_id = (select first 1 z_id from zalog
          where z_from_k_id = k_id and z_is_ready = 1 order by z_id desc)
      inner join clients
        on c_id = z_from_c_id
          --and c_type = 0
      left join fiolist as ru
        on ru.fl_id = c_from_fl_id
      left join fiolist as en

    where k_kontract_no = :k_kontract_no
  into
    :z_id,
    :k_id,
    :k_disb_date,
    :k_kredline_end,
    :reg_id,
    :cl_id,
    :c_id,
    :c_name_fio,
    :c_lastname,
    :c_firstname,
    :c_middlename,
    :c_alternat_fio,
    :c_flastname,
    :c_ffirstname,
    :c_fmiddlename,
    :c_birth_date,
    :c_pass_no
  do
  begin
    if (c_ffirstname in ('uulu', 'kyzy')) then begin
      c_lastname = c_lastname || ' ' || iif(c_ffirstname='uulu', 'Уулу', 'Кызы');
      c_flastname = c_flastname || ' ' || c_ffirstname;
      c_ffirstname = c_fmiddlename;
      c_fmiddlename = '';
    end

    -- паспорт
    if ((
      select first 1 trim(pk_country)
      from clientdocs
        inner JOIN passkind
          on pk_id = cds_from_pk_id
            and pk_from_pdt_id in (1,2,3)
      where cds_from_c_id = :c_id
        and cds_is_delete = 0
        and :k_disb_date between cds_doc_date_begin and cds_doc_date_end
      order by cds_id desc) = 'rus') then
    begin
      c_pass_code = int_code;
      c_pass_name = int_passport;
    end else begin
      c_pass_code = ext_code;
      c_pass_name = ext_passport;
    end

    -- адрес
    select
      adl_id,
      adl_is_edited,
      adl_address_str,
      adl_from_pl_id,
      adl_street_id,
      adl_is_without_street,
      iif(adl_depart_no='0','',adl_depart_no||iif(adl_depart_pref='','','/')||adl_depart_pref),
      iif(adl_flat_no='0','',adl_flat_no),
      iif(adl_building_no='0', '', adl_building_no)
      adl_address_str
    from addrclient
      inner join addrlist
        on adl_id = adc_from_adl_id
    where adc_from_c_id = :c_id
      and adc_address_type = 1
    into
      :adl_id,
      :res_address_is_edited,
      :res_address_str,
      :adl_from_pl_id,
      :adl_street_id,
      :adl_is_without_street,
      :adl_depart_no,
      :adl_flat_no,
      :adl_building_no;

    -- залог
    c_id1 = (select first 1 kc_from_c_id from knclient where kc_from_k_id = :k_id);
    vin = '';
    pin = '';
    chassisnumber = '';
    bodynumber = '';
    select first 1
      f_docrecog_value(cds_from_drl_id, 'vin'),
      f_docrecog_value(cds_from_drl_id, 'pin'),
      f_docrecog_value(cds_from_drl_id, 'body_no'),
      f_docrecog_value(cds_from_drl_id, 'chassis_no')
    from clientdocs
    where cds_from_c_id = :c_id1
      and cds_from_pk_id = 315
      and cds_is_delete = 0
      and cds_from_drl_id > 0
    order by cds_create_datetime desc
    into
      vin,
      pin,
      bodynumber,
      chassisnumber;

    pos = position(' ', bodynumber);
    if (pos > 0) then bodynumber = left(bodynumber, pos-1);

    if (vin containing 'от' or vin containing 'не')
      then vin = '';
    if (pin containing 'от' or pin containing 'не')
      then pin = '';
    if (chassisnumber containing 'от' or chassisnumber containing 'не')
      then chassisnumber = '';
    if (bodynumber containing 'от' or bodynumber containing 'не')
      then bodynumber = '';

    res_temp_xml = replace(res_temp_xml, '&Id', '00000000-0000-0000-0000-' || lpad(z_id, 12, 0));
    res_full_xml = res_temp_xml;

    -- контракт
    res_full_xml = replace(res_full_xml, '&KontNumber', k_kontract_no);
    res_full_xml = replace(res_full_xml, '&KontDtBeg', k_disb_date);
    res_full_xml = replace(res_full_xml, '&KontDtEnd', k_kredline_end);

    -- залогодатель
    res_full_xml = replace(res_full_xml, '&LastName', c_lastname);
    res_full_xml = replace(res_full_xml, '&FirstName', c_firstname);
    res_full_xml = replace(res_full_xml, '&MiddleName', c_middlename);
    res_full_xml = replace(res_full_xml, '&FLastName', c_flastname);
    res_full_xml = replace(res_full_xml, '&FFirstName', c_ffirstname);
    res_full_xml = replace(res_full_xml, '&FMiddleName', c_fmiddlename);
    res_full_xml = replace(res_full_xml, '&Birthdate', c_birth_date);

    -- документ залогодателя
    res_full_xml = replace(res_full_xml, '&CodeDoc', c_pass_code);
    res_full_xml = replace(res_full_xml, '&NameDoc', c_pass_name);
    res_full_xml = replace(res_full_xml, '&SeriesNumber', c_pass_no);

    -- залог
    descr = 'Автомобиль';
    res_full_xml = replace(res_full_xml, '&Vin', iif(vin = '', '', '<VIN>' || vin || '</VIN>'));
    res_full_xml = replace(res_full_xml, '&Pin', iif(pin = '', '', '<PIN>' || pin || '</PIN>'));
    res_full_xml = replace(res_full_xml, '&ChassisNumber', iif(chassisnumber = '', '', '<ChassisNumber>' || chassisnumber || '</ChassisNumber>'));
    res_full_xml = replace(res_full_xml, '&BodyNumber', iif(bodynumber = '', '', '<BodyNumber>' || bodynumber || '</BodyNumber>'));
    res_full_xml = replace(res_full_xml, '&Descr', '<Description>' || descr || '</Description>');

    -- адрес залогодателя
    if (res_address_is_edited = 1) then begin
      adl_region_code = '';
      adl_region = '';
      adl_district = '';
      adl_city = '';
      adl_locality = '';
      adl_street = '';
      adl_depart_no = '';
      adl_flat_no = '';
      adl_building_no = '';
    end else begin
      -- улица
      if (adl_street_id > 0 and adl_is_without_street = 0) then
        select
          case plt_alt_short_name_side
            when -1 then plt_alt_short_name || ' ' || pl_name
            when  0 then pl_name
            when  1 then pl_name || ' ' || plt_alt_short_name
          end
        from places
          inner join placetype
            on plt_id = pl_status
        where pl_id = :adl_street_id
        into adl_street;
      else adl_street = '';

      -- субъект, район, город, населенный пункт
      num_node = 0;
      pl_id = adl_from_pl_id;
      adl_region_code = '';
      adl_region = '';
      adl_district = '';
      adl_city = '';
      adl_locality = '';
      while (pl_id <> 810000000001) DO begin
        num_node = num_node + 1;
        select
          parent,
          case plt_alt_short_name_side
            when -1 then plt_alt_short_name || ' ' || pl_name
            when  0 then pl_name
            when  1 then pl_name || ' ' || plt_alt_short_name
          end,
          plt_name
        from places
          inner join placetype
            on plt_id = pl_status
        where pl_id = :pl_id
        into
          pl_parent,
          pl_name,
          plt_name;
        if (plt_name in ('область', 'край', 'республика', 'автономный округ') or
          pl_name in ('г. Москва', 'г. Санкт-Петербург')) then
        begin
          select
            rd_name,
            rd_memo
          from refdata
          where rd_main_code = 'PlaceRegion'
            and rd_parent_id > 0
            and rd_code = :pl_id
          into
            adl_region_code,
            adl_region;
        end
        else
        if (plt_name = 'район') then adl_district = pl_name;
        else
        if (plt_name = 'город') then adl_city = pl_name;
        else adl_locality = pl_name;
        pl_id = pl_parent;
      end
    end
    res_full_xml = replace(res_full_xml, '&CodeRegion', iif(adl_region_code = '', '', '<RegionCode>' || adl_region_code || '</RegionCode>'));
    res_full_xml = replace(res_full_xml, '&Region', iif(adl_region = '', '', '<Region>' || adl_region || '</Region>'));
    res_full_xml = replace(res_full_xml, '&City', iif(adl_city = '', '', '<City>' || adl_city || '</City>'));
    res_full_xml = replace(res_full_xml, '&District', iif(adl_district = '', '', '<District>' || adl_district || '</District>'));
    res_full_xml = replace(res_full_xml, '&Locality', iif(adl_locality = '', '', '<Locality>' || adl_locality || '</Locality>'));
    res_full_xml = replace(res_full_xml, '&Street', iif(adl_street = '', '', '<Street>' || adl_street || '</Street>'));
    res_full_xml = replace(res_full_xml, '&Building', iif(adl_building_no = '', '', '<Building>' || adl_building_no || '</Building>'));
    res_full_xml = replace(res_full_xml, '&House', iif(adl_depart_no = '', '', '<House>' || adl_depart_no || '</House>'));
    res_full_xml = replace(res_full_xml, '&Apartment', iif(adl_flat_no = '', '', '<Apartment>' || adl_flat_no || '</Apartment>'));

    jsdoc = '';
    for
      select
        max(cds_id)
      from clientdocs
        inner join passkind
          on pk_id = cds_from_pk_id
      where cds_from_c_id = :c_id
        and :k_disb_date between cds_doc_date_begin and coalesce(cds_doc_date_end, :k_disb_date)
        and cds_is_delete = 0
        and cds_verify_status = 2
        and (pk_id in (315, 310, 43, 304) or cds_is_main_doc = 1)
      group by pk_id
    into
      :cds_id
    do
    begin
      jsdoc = jsdoc || iif(jsdoc <> '', ',', '') ||
        f_docs_item(:cds_id, null, 0, 0, 0);
    end
    if (jsdoc <> '') then
      jsdoc = '[{' || f_json_make_pair('caption', :c_name_fio, 'str') || ',' ||
        '"auto_load":0, "items":[' || jsdoc || ']}]' ;

    jsloan = coalesce((
      select
        f_format('[{"caption" : "Договор залога", "blobs":[%1]}]',
          f_image_item('blobstore', edb_from_bls_id, 'Оригинал', null, 0, 0, 0))
      from edoc_list
        inner join edoc_blank
          on edb_from_edl_id = edl_id
      where edl_object_kind = 9
        and edl_object_id = :k_id
        and edl_from_edc_id = f_edoc_kind('loan.collateral', 'id')
        and edl_status = 1), '');


    suspend;
  end
end
