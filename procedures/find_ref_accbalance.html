CREATE PROCEDURE FIND_REF_ACCBALANCE(
    FIND_STR VARCHAR(255),
    FIND_KIND INTEGER)
RETURNS (
    AB_ID TYPE OF D_ID,
    AB_PARENT_ID TYPE OF D_ID,
    AB_CODE TYPE OF D_STR_CODE,
    AB_NAME TYPE OF D_STR_LONG)
AS
declare variable bg_full d_str_code;
begin
--  find_kind 0 все, 1 - классы, 2 - группы и балансы, 3 - балансовые счета
  if (find_str is null) then
    find_str = '';
  find_str = trim(find_str);
  if (find_str <> '') then
  begin
    if (char_length(find_str) = 5 and
        position('-', find_str) > 0) then
    begin
-- значит это старый балансовый счет, нужно вытащить через mapping
      for select ab.ab_id, ab.ab_parent_id, ab.ab_code, ab.ab_code || ' ' || ab.ab_name || '   / ' || parent.ab_name
          from balgroupmap m, ref_accbalance ab, ref_accbalance parent
          where ab.ab_level = 3 and
                ab.ab_parent_id = parent.ab_id and
                ab.ab_id = m.ab_code and
                m.bg_code = :find_str
          order by ab_code
          into :ab_id, :ab_parent_id, ab_code, ab_name
      do
        suspend;

    end
    else
    begin
      for select left(ab_id || '00000', 5), ab_id, ab_parent_id, ab_code, ab_name
          from ref_accbalance, find_in_workdoc_words('ref_accbalance', :find_str, 1)
          where ab_id = wdlink_object_id and
                ab_level >= :find_kind
          order by 1, 3
          into :bg_full, :ab_id, :ab_parent_id, ab_code, ab_name
      do
      begin
        suspend;
      end
    end
  end
  else
  begin
    for select ab.ab_id, ab.ab_parent_id, ab.ab_code, ab.ab_code || ' ' || ab.ab_name || '   / ' || parent.ab_name
        from ref_accbalance ab, ref_accbalance parent
        where ab.ab_level = 3 and
              ab.ab_parent_id = parent.ab_id
        order by ab_code
        into :ab_id, :ab_parent_id, ab_code, ab_name
    do
      suspend;
  end
end
