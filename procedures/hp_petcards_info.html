CREATE PROCEDURE HP_PETCARDS_INFO(
    REP_CARD_NO VARCHAR(20),
    REP_DATE DATE)
RETURNS (
    PCS_FROM_REG_ID INTEGER,
    PCS_REGION_ID INTEGER,
    PCS_ERROR INTEGER,
    PCS_C_ID DECIMAL(18,0),
    PCS_IS_TRANS INTEGER,
    PCS_REGION_NAME VARCHAR(100),
    PCS_MEMO VARCHAR(255),
    PCS_C_FIO VARCHAR(100),
    PCS_FROM_PET_ID INTEGER,
    PCS_BRANCH_ID INTEGER,
    PCS_BRANCH_NAME VARCHAR(100))
AS
declare variable txt varchar(255);
declare variable pcs_from_c_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
declare variable db_id integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  pcs_error = 0;

  db_id = f_get_db_id();

  for select pcs_from_reg_id, pcs_from_pet_id, pcs_from_c_id, reg_name
      from petcards, region
      where pcs_card_no = :rep_card_no and
            pcs_date <= :rep_date and
            reg_unicode = pcs_from_reg_id
      order by pcs_date desc
      rows 1
      into :pcs_from_reg_id, :pcs_from_pet_id, :pcs_from_c_id, :pcs_region_name
  do
  begin
    if (db_id <> 94401) then
    begin
      select r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :pcs_from_reg_id and
            r2.reg_unicode = r1.reg_head_office
      into :pcs_branch_id, :pcs_branch_name;
    end
    else
    begin
      select reg_unicode, reg_code || ' ' || reg_name
      from region
      where reg_unicode = :pcs_from_reg_id
      into :pcs_branch_id, :pcs_branch_name;
    end

    pcs_is_trans = 0;
    if (pcs_from_pet_id is not null) then
    begin
      pcs_region_id = null;
      pcs_c_id = null;
      txt = '';
      select ped_region_place_id, ped_from_c_id, pet_name || ' ' || pet_label || ', ' || reg_name
      from petdata, pettrans, region
      where ped_from_pet_id = :pcs_from_pet_id and
            pet_id = :pcs_from_pet_id and
            ped_date <= :rep_date and
            ped_region_place_id = reg_unicode
      order by ped_date desc
      rows 1
      into :pcs_region_id, :pcs_c_id, :txt;
      pcs_is_trans = 1;

      select c_name_fio
      from clients
      where c_id = :pcs_c_id
      into :pcs_c_fio;

      if (pcs_region_id <> pcs_from_reg_id) then
      begin
        pcs_error = 1;
      end
    end
    else
    begin
      if (pcs_from_c_id is not null) then
      begin
        pcs_c_id = pcs_from_c_id;
        select c_name_fio
        from clients
        where c_id = :pcs_from_c_id
        into :pcs_c_fio;

        select jl_name || ', ' || reg_name, ps_from_reg_id
        from hr_pstatus, joblist, region
        where ps_from_c_id = :pcs_c_id and
              ps_date_begin <= :rep_date and
              ps_date_end >= :rep_date and
              ps_from_jl_id = jl_id and
              ps_from_reg_id = reg_unicode
        order by ps_date_begin desc
        rows 1
        into :txt, :pcs_region_id;
        pcs_is_trans = 0;
        if (pcs_region_id <> pcs_from_reg_id) then
        begin
          pcs_error = 1;
        end
        if (txt is null) then
          txt = 'No office';
      end
    end
    pcs_memo = :txt;
    if (pcs_memo is null) then
      pcs_memo = '';
    suspend;
  end
end
