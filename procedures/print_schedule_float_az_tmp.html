CREATE PROCEDURE PRINT_SCHEDULE_FLOAT_AZ_TMP(
    K_ID D_ID,
    LEVEL_NO D_INTEGER)
RETURNS (
    SH_ID D_INTEGER,
    SH_DATE D_DATE,
    SH_OS D_CURRENCY,
    SH_PERC D_CURRENCY,
    SH_KOMMIS D_CURRENCY,
    SH_ALL D_CURRENCY,
    SH_TOTAL D_CURRENCY,
    SH_DISB_AMOUNT D_CURRENCY,
    CL_QTY D_SMALLINT,
    SH_PERC_TAX D_CURRENCY,
    SH_BALANCE D_CURRENCY)
AS
declare variable sh_date_prv d_date;
declare variable perc_new1 d_percent;
declare variable last_sh_id d_integer;
declare variable new_annu d_currency;
declare variable k_percent_per_day d_percent;
declare variable calced_perc d_currency;
declare variable level_day1 d_integer;
declare variable new_perc_per_day1 d_percent;
declare variable first_date d_date;
declare variable days_list d_str_long;
declare variable loan_amount d_currency;
declare variable orig_annu d_currency;
declare variable level_day2 d_integer;
declare variable perc_new2 d_percent;
declare variable new_perc_per_day2 d_percent;
declare variable level_day d_integer;
declare variable new_perc_per_day d_percent;
declare variable k_disb_date d_date;
declare variable orig_perc d_percent;
declare variable orig_sh_os d_currency;
declare variable sh_date_1 d_date;
declare variable sh_date_2 d_date;
declare variable tmp_dd d_date;
declare variable start_date d_date;
declare variable k_nsp_percent d_percent;
declare variable pr_percent_kind d_integer;
declare variable pr_code d_str_100;
declare variable c_id d_id;
declare variable point d_integer;
begin
  if (coalesce(k_id, 0) = 0) then
    exception e_default 'K_ID is emptry';

  if (level_no is null) then
    exception e_default 'level_no is emptry';

  if (not(level_no in (1, 2))) then
    exception e_default 'Incorrect level_no: ' || level_no;

  start_date = cast('29.07.2021' as date);

  select k_percent_per_day, k_amount_approved, k_disb_date, k.k_kolich_client_approved, pr.pr_percent_kind, pr.pr_code,
         kc.kc_from_c_id, round(cast(k_percent_per_day as double precision) * 360.0, 2)
  from kontract k
  inner join product pr on k.k_from_pr_id = pr_id
  inner join knclient kc on kc.kc_from_k_id = k.k_id
  where k_id = :k_id
  into k_percent_per_day, sh_disb_amount, k_disb_date, cl_qty, pr_percent_kind, pr_code, c_id, :orig_perc;

  if (pr_percent_kind <> 1) then
    exit;

  select nsd_percent
  from nspconfig_data
  where nsd_from_nsm_id = -1 and -- нсп от процента
        :k_disb_date between nsd_date_begin and nsd_date_end
  into :k_nsp_percent;
  if (k_nsp_percent is null) then
    k_nsp_percent = 0;

  --orig_perc = round(cast(k_percent_per_day as double precision) * 365.0, 2);

  --point = f_point(:c_id, :k_disb_date);

  sh_kommis = 0;

  days_list = '';
  loan_amount = 0;
  sh_date_prv = null;

  select sh_date
  from schedule
  where sh_from_k_id = :k_id and
        sh_id = 1
  into sh_date_1;

  select sh_date
  from schedule
  where sh_from_k_id = :k_id and
        sh_id = 2
  into sh_date_2;

  /*
  orig_perc =(case  when point between 0 and 2 then 42.0
                    when point between 3 and 5 then 38.0
                    when point between 6 and 8 then 34.0
                    when point between 9 and 11 then 30.0
                    when point between 12 and 14 then 28.0
                    when point between 15 and 17 then 26.0
                    when point >= 18 then 24.0
                    end);  */

  --Новый продукт в АЗ 01.08.2021
  level_day1 = 2;
  perc_new1 = 100.0;
  if (pr_code containing '.score_rate.') then
  begin
    if (k_disb_date < '01.04.2022') then
    begin
      perc_new1 = orig_perc + 5.0;
    end
    else
    begin
      perc_new1 = (case orig_perc --действуют с 01.04.2022 jira 5145
                     when 42 then 47
                     when 38 then 43
                     when 34 then 39
                     when 32 then 37
                     when 30 then 35
                     when 28 then 33
                     when 26 then 31
                     when 24 then 30
                     when 23 then 30
                     else 0
                   end);
    end
  end
  new_perc_per_day1 = perc_new1 / 360.000000;

  level_day2 = 7;
  perc_new2 = 100.0;
  if (pr_code containing '.score_rate.') then
  begin
    if (k_disb_date < '01.04.2022') then
    begin
      perc_new2 = case
                    when orig_perc = 42.0 then orig_perc + 7.0
                    when orig_perc = 38.0 then orig_perc + 11.0
                    when orig_perc = 34.0 then orig_perc + 14.0
                    when orig_perc = 30.0 then orig_perc + 14.0
                    when orig_perc = 28.0 then orig_perc + 14.0
                    when orig_perc = 26.0 then orig_perc + 14.0
                    when orig_perc = 24.0 then orig_perc + 14.0
                  end;
    end
    else
    begin
      perc_new2 = (case orig_perc --действуют с 01.04.2022 jira 5145
                     when 42 then 49
                     when 38 then 49
                     when 34 then 48
                     when 32 then 44
                     when 30 then 42
                     when 28 then 40
                     when 26 then 40
                     when 24 then 40
                     when 23 then 40
                     else 0
                   end);
    end
  end
  new_perc_per_day2 = perc_new2 / 360.000000;

  if (sh_date_1 is not null and
      sh_date_2 is not null) then
  begin
    -- Вычисляем рабочие дня соответствующие на
    tmp_dd = dateadd(day, level_day1, sh_date_1);
    if (tmp_dd >= sh_date_2) then
      tmp_dd = sh_date_2 - 1;
    level_day1 = datediff(day, sh_date_1, tmp_dd);

    tmp_dd = dateadd(day, level_day2, sh_date_1);
    if (tmp_dd >= sh_date_2) then
      tmp_dd = sh_date_2 - 1;
    level_day2 = datediff(day, sh_date_1, tmp_dd);
  end

  if (level_no = 1) then
  begin
    level_day = level_day1;
    new_perc_per_day = new_perc_per_day1;
  end

  if (level_no = 2) then
  begin
    level_day = level_day2;
    new_perc_per_day = new_perc_per_day2;
  end

  for select sh_id, sh_date, sh_os, sh_perc
      from schedule
      where sh_from_k_id = :k_id and
            sh_id >= 1
      order by sh_id
      into sh_id, sh_date, sh_os, sh_perc
  do
  begin
    if (sh_id > 1) then
      loan_amount = loan_amount + sh_os;

    if (orig_annu is null and
        sh_os <> 0) then
      orig_annu = sh_os + sh_perc;

    if (sh_date_prv is not null and
        sh_os <> 0) then
    begin
      if (days_list <> '') then
        days_list = days_list || ',';

      days_list = days_list || datediff(day, sh_date_prv, sh_date);
    end

    if (sh_id = 1) then
    begin
      sh_date_prv = dateadd(day, level_day, sh_date);
      first_date = sh_date_prv;
    end
    else
      sh_date_prv = sh_date;

    last_sh_id = sh_id;
  end

  sh_id = 1;
  sh_date = null;

  calced_perc = round((loan_amount / 100.0000 * k_percent_per_day) * level_day1, 0);

  if (level_no >= 2) then
    calced_perc = calced_perc + round((loan_amount / 100.0000 * new_perc_per_day1) * (level_day2 - level_day1), 0);

  new_annu = f_annu(loan_amount, calced_perc, 1, new_perc_per_day, days_list, 0, 0, 0, 0, 0);

  sh_balance = loan_amount;
  for select sh_id, sh_date, sh_os, sh_perc, sh_kommis
      from schedule
      where sh_from_k_id = :k_id and
            sh_id > 0
      order by sh_id
      into sh_id, sh_date, sh_os, sh_perc, sh_kommis
  do
  begin
    orig_sh_os = sh_os;
    if (sh_id >= 2) then
    begin
      if (sh_id = 2) then
        sh_date_prv = first_date;

      sh_perc = sh_balance / cast(100.0000 as float) * new_perc_per_day * datediff(day, sh_date_prv, sh_date);

      if (sh_id = 2) then
        sh_perc = sh_perc + calced_perc;

      sh_perc = round(sh_perc, 0);

      if (orig_sh_os <> 0) then
      begin
        if (sh_id < last_sh_id) then
        begin
          sh_os = new_annu - sh_perc;
          if (sh_os < 0) then
            sh_os = 0;
        end
        else
          sh_os = sh_balance;

        sh_balance = sh_balance - sh_os;
      end
      else
        sh_os = 0;

      if (sh_os < 0) then
        exception e_default 'Отрицательная основная сумма! Дождитесь решения от разработчиков.';
    end

    sh_perc_tax = (select res
                   from fhp_round_bank(((:sh_perc + :sh_kommis) * :k_nsp_percent / 100.0), 0));

    sh_all = sh_os + sh_perc;
    sh_total = sh_os + sh_perc + sh_kommis + sh_perc_tax;
    suspend;

    sh_date_prv = sh_date;
  end
end
