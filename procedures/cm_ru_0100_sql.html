CREATE PROCEDURE CM_RU_0100_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable ma_refund d_smallint;
declare variable pr_code d_str_100;
begin
  for
  select ma_date_oper, ma_refund, pr_code
  from miniapp, product
  where ma_id = :ma_id
  and ma_date_end = '31.12.9999'
  and ma_is_active = 1
  and ma_from_pr_id = pr_id
  and pr_product_kind = 2
  and pr_code not containing '.express_online.'
  into kf_disb_date, ma_refund, pr_code
  do
  begin
    ma_refund = coalesce(ma_refund,0);
    if (not exists(
      select *
      from miniappclient
        inner join knclient
          on kc_from_c_id = mac_from_c_id
            and kc_amount_approved > 0
        inner join kontstate
          on ks_from_k_id = kc_from_k_id + 0
            and :kf_disb_date between ks_date_begin and ks_date_end
            and ks_state = 1
            and ks_ol_is_delete = 0
        inner join kontract
          on k_id = ks_from_k_id
        inner join product
          on pr_id = k_from_pr_id
            and pr_product_kind <> 2
      where mac_from_ma_id = :ma_id
        and mac_amount > 0
    /*
    select * from miniappclient, knclient, kontstate, kontract, product
    where mac_from_ma_id = :ma_id 
    and mac_amount > 0
    and mac_from_c_id = kc_from_c_id
    and kc_amount_approved > 0
    and kc_from_k_id = ks_from_k_id
    and ks_ol_is_delete = 0
    and ks_state = 1
    and :kf_disb_date between ks_date_begin and ks_date_end
    and ks_from_k_id = k_id
    and k_from_pr_id = pr_id
    and pr_product_kind <> 2
    */
    )) then
    begin
      if (not (ma_refund = 1 and kf_disb_date >= '01.03.2022' and pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.)%')) then
      begin
        res_error_text = 'Не имеется активный основной кредит.';
        res_error_is_critical = 2;
        suspend;
      end
    end
  end
end
