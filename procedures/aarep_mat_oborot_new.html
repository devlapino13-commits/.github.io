CREATE PROCEDURE AAREP_MAT_OBOROT_NEW(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_KIND SMALLINT)
RETURNS (
    REP_ML_NAME VARCHAR(255),
    REP_INVENT_STR VARCHAR(50),
    REP_OPER_DATE DATE,
    REP_REGION_NAME VARCHAR(100),
    REP_ACCOUNT VARCHAR(20),
    REP_KOLICH_PLUS DECIMAL(18,4),
    REP_KOLICH_MINUS DECIMAL(18,4),
    REP_SUMMA_PLUS DECIMAL(18,4),
    REP_SUMMA_MINUS DECIMAL(18,4),
    REP_MATOTVET VARCHAR(100),
    REP_WORKPLACE VARCHAR(100),
    REP_CATEGORY VARCHAR(100),
    REP_CATEGORY_PARENT VARCHAR(100),
    REP_CATEGORY_TYPE VARCHAR(100),
    REP_ACCOUNT_NAME VARCHAR(100),
    REP_ACC_BALANCE VARCHAR(20),
    REP_BALANCE_NAME VARCHAR(100),
    REP_OPER_KIND VARCHAR(50))
AS
declare variable rep_cat_type integer;
declare variable ot_type integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ml_name, ob_invent_str, ol_date_oper_day, reg_name, om_account, a1.al_from_bg_id, a1.al_name, b1.bg_name,
             om_kolich_plus, om_kolich_minus, om_summa_plus, om_summa_minus, c1.c_name_fio, w1.wp_name, m1.mc_name,
             m2.mc_name, m1.mc_status, om_oper_type
      from opermat, operlist, operbill, matlist, region, clients c1, workplace w1, matcateg m1, matcateg m2, acc_list a1, region_list(:report_region_id, 1, -1, 1), balgroup b1
      where om_from_ol_id = ol_id and
            ((:report_kind = 0) or
            (om_oper_type = :report_kind)) and
            ob_bill_id = om_bill_id and
            ob_bill_line_no = om_bill_line_no and
            ob_from_ml_id = ml_id and
            ob_date_begin <= ol_date and
            ob_date_end >= ol_date and
            ol_date_oper_day between :report_date_begin and :report_date_end and
            reg_unicode = ol_from_reg_id and
            (id_list = ol_from_reg_id) and
            om_matotvet_id = c_id and
            om_workplace_id = wp_id and
            operlist.is_delete = 0 and
            ob_from_mc_id = m1.mc_id and
            m1.parent = m2.mc_id and
            a1.al_id = om_account and
            a1.al_from_bg_id = bg_id
      order by ol_date_oper_day
      into :rep_ml_name, :rep_invent_str, :rep_oper_date, :rep_region_name, :rep_account, :rep_acc_balance,
           :rep_account_name, :rep_balance_name, :rep_kolich_plus, :rep_kolich_minus, :rep_summa_plus, :rep_summa_minus,
           :rep_matotvet, :rep_workplace, :rep_category, :rep_category_parent, :rep_cat_type, :ot_type
  do
  begin
    if (rep_cat_type = 0) then
      rep_category_type = 'Основные средства';
    if (rep_cat_type = 1) then
      rep_category_type = 'МБП';
    if (rep_cat_type = 2) then
      rep_category_type = 'Материалы, расходники';
    if (rep_cat_type = 3) then
      rep_category_type = 'Нематеральные активы';

    if (ot_type = 1) then
      rep_oper_kind = 'Покупка, приобретение';
    if (ot_type = 2) then
      rep_oper_kind = 'Приход от производства';
    if (ot_type = 3) then
      rep_oper_kind = 'Перемещение внутри офиса';
    if (ot_type = 4) then
      rep_oper_kind = 'Внутренние списания из баланса';
    if (ot_type = 5) then
      rep_oper_kind = 'Продажа, Передача филиалам (Выбытие)';
    if (ot_type = 6) then
      rep_oper_kind = 'Модернизация, реконструкция';
    if (ot_type = 7) then
      rep_oper_kind = 'Разделение цельного / Объединение';
    if (ot_type = 8) then
      rep_oper_kind = 'Комплектация / Декомплектация';
    if (ot_type = 9) then
      rep_oper_kind = 'Переоценка (доход, убыток)';
    if (ot_type = 10) then
      rep_oper_kind = 'Переоценка за счет амортизации';
    suspend;
  end
end
