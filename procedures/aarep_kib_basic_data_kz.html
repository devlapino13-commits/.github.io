CREATE PROCEDURE AAREP_KIB_BASIC_DATA_KZ(
    OBJECT_ID D_ID,
    OBJECT_TYPE D_STR_10)
RETURNS (
    C_ID D_ID,
    CLIENT_DATA D_BLOB_TEXT)
AS
declare variable UNPAID_OWN_AMOUNT CURR_TYPE;
declare variable ACTIVE_KONT smallint;
declare variable MAX_DELINQ_DAYS integer;
declare variable PORUCH_KONT_QTY integer;
declare variable KBM_CHECKED_DATE D_DATE;
declare variable C_NAME_FIO varchar(100);
declare variable BLS_ID D_ID;
declare variable BLS_ID_PMI D_ID;
declare variable TOTAL_SUBMITTED_MINIAPP D_STR_10;
declare variable TOTAL_APPROVED_MINIAPP D_STR_10;
declare variable TOTAL_REJECTED_MINIAPP D_STR_10;
declare variable TOTAL_CREATED_MINIAPP D_STR_10;
declare variable TOTAL_REVOKED_MINIAPP D_STR_10;
declare variable PERIOD D_STR_100;
declare variable IS_OVERDUE_IN_60 D_SMALLINT;
declare variable OVERDUE_7_30 D_INTEGER;
declare variable OVERDUE_31_60 D_INTEGER;
declare variable OVERDUE_61_90 D_INTEGER;
declare variable OVERDUE_91_360 D_INTEGER;
declare variable OVERDUE_360 D_INTEGER;
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable I D_INTEGER;
declare variable CNT_TERMINATED D_INTEGER;
declare variable DAYS_STR D_STR_20;
declare variable DAYS D_INTEGER;
declare variable HAS_PARSED_DATA D_SMALLINT;
begin
  overdue_7_30 = 0;
  overdue_31_60 = 0;
  overdue_61_90 = 0;
  overdue_91_360 = 0;
  overdue_360 = 0;

  if (object_type = 'kontract') then
  begin
    select kn.k_from_ma_app_id
    from kontract kn
    inner join knclient kc on kc.kc_from_k_id = kn.k_id
        and kc.kc_amount_approved > 0
    where kn.k_id = :object_id
    into :object_id;

    object_type = 'miniapp';
  end 

  select eil.eil_from_bls_id, cast(eil.eil_status_datetime as date), eil.eil_from_c_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eil.eil_object_kind = f_objkind(:object_type, 'id') and
        eil.eil_object_id = :object_id and
        eic.eic_code = 'pkb_cib_report' and
        eil.eil_status = 2 and
        eil.eil_from_bls_id is not null
  order by eil.eil_status_datetime desc rows 1
  into :bls_id, :kbm_checked_date, :c_id;

  if (:bls_id is null) then
    exit;

  has_parsed_data = 0;

  select prp.prp_max_overdue_days, prp.prp_out_amount_total, prp.prp_active_contract_count,
         prp.prp_poruch_kont_qty, prp.prp_overdue_7_30_count, prp.prp_overdue_31_60_count,
         prp.prp_overdue_61_90_count, prp.prp_overdue_91_360_count, prp.prp_overdue_over_360_count
  from pkb_report_parsed prp
  where prp.prp_from_ma_id = :object_id
  order by prp.prp_from_eil_id desc rows 1
  into :max_delinq_days, :unpaid_own_amount, :active_kont,
       :poruch_kont_qty, :overdue_7_30, :overdue_31_60,
       :overdue_61_90, :overdue_91_360, :overdue_360;

  if (row_count = 0) then
  begin
    select all_k_overdue_max, outstanding_amount_sum, active_kont, poruch_kont_qty
    from hp_parse_pkb_report(:bls_id)
    rows 1
    into :max_delinq_days, unpaid_own_amount, active_kont, poruch_kont_qty;
  end
  else
    has_parsed_data = 1;

  select nkh_is_overdue, cl.c_name_fio
  from negativ_kib_history nkh
  inner join clients cl on cl.c_id = nkh.nkh_from_c_id
  where nkh.nkh_from_c_id = :c_id and
        nkh.nkh_last = 1
  into :is_overdue_in_60, c_name_fio;

  client_data = BLOB_APPEND(client_data, '<b>' || c_name_fio || '</b><br>');
  client_data = BLOB_APPEND(client_data, '&nbsp&nbspДата проверки: <b>' || kbm_checked_date || '</b><br>');
  client_data = BLOB_APPEND(client_data, '&nbsp&nbsp<b>В других ФКУ нарушений ' || iif(is_overdue_in_60 = 1, 'обнаружено', 'не обнаружено') || '</b><br>');
  client_data = BLOB_APPEND(client_data, '&nbsp&nbspМакс. дни просрочки в других ФКУ<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || cast(max_delinq_days as varchar(10)) || ' дней</b><br>');
  client_data = BLOB_APPEND(client_data, '&nbsp&nbspАктивный баланс в других ФКУ<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || cast(round(unpaid_own_amount) as varchar(10)) || '</b><br>');
  client_data = BLOB_APPEND(client_data, '&nbsp&nbspАктивные займы в других ФКУ<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || cast(active_kont as varchar(10)) || ' кредит(ов)</b><br>');
  client_data = BLOB_APPEND(client_data, '&nbsp&nbspАктивные займы в других ФКУ, где он(а) поручитель<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || cast(poruch_kont_qty as varchar(10)) || ' кредит(ов)</b><br>');

    client_data = BLOB_APPEND(client_data, f_uri_html('Просмотр отчета', 'kib_report_kz',
    'Отчет КИБ', object_type, :object_id, current_date, current_date,
    '{"c_id":' || c_id || '}') || '<br>');
--------------------------------------------------------------------------------
  select pmi.pmi_bls_id
  from pkb_miniapp_info pmi
  where pmi.pmi_from_ma_id = :object_id
        and pmi.pmi_is_sent = 1
  order by pmi.pmi_last_datetime desc rows 1
  into :bls_id_pmi;

  if (:bls_id_pmi is not null) then
  begin
    select period, total_submitted_miniapp, total_approved_miniapp, total_rejected_miniapp, total_created_miniapp, total_revoked_miniapp
    from hp_parse_pkb_miniapp_info(:bls_id_pmi)
    into :period, :total_submitted_miniapp, :total_approved_miniapp, :total_rejected_miniapp, :total_created_miniapp, :total_revoked_miniapp;

    client_data = BLOB_APPEND(client_data, 'Информация по поданным заявкам за период с <br>' || '&nbsp&nbsp' || :period || '</b><br>');
    client_data = BLOB_APPEND(client_data, 'Общее количество подданных заявок <br>' || '&nbsp&nbsp' || :total_submitted_miniapp || ' кредит(ов)</b><br>');
    client_data = BLOB_APPEND(client_data, 'Количество заявок со статусом "Одобрено" <br>' || '&nbsp&nbsp' || :total_approved_miniapp || ' кредит(ов)</b><br>');
    client_data = BLOB_APPEND(client_data, 'Количество заявок со статусом "Отказано кредитором" <br>' || '&nbsp&nbsp' || :total_rejected_miniapp || ' кредит(ов)</b><br>');
    client_data = BLOB_APPEND(client_data, 'Количество заявок со статусом "Создано" <br>' || '&nbsp&nbsp' || :total_created_miniapp || ' кредит(ов)</b><br>');
    client_data = BLOB_APPEND(client_data, 'Количество заявок со статусом "Отозвано заемщиком" <br>' || '&nbsp&nbsp' || :total_revoked_miniapp || ' кредит(ов)</b><br>');
  end
  --  RFCKZ-557
    select cast(blx_blob as d_blob_text)
    from hp_get_blobex_blob(:bls_id)
    into :json_text;

    if (json_text is not null and has_parsed_data = 0) then
    begin
      cnt_terminated = f_json_blob_get_count(json_text, 'Report.Root.TerminatedContracts.Contract');

      if (cnt_terminated > 0) then
      begin
        i = 0;
        while (i < cnt_terminated) do
        begin
          days_str = f_json_blob_get_value(json_text,'Report.Root.TerminatedContracts.Contract[' || i || '].NumberOfOverdueInstalmentsMax.value','');
                days = iif(days_str <> '', cast(days_str as integer), 0);
    
                if (days between 7 and 30) then overdue_7_30 = overdue_7_30 + 1;
                else if (days between 31 and 60) then overdue_31_60 = overdue_31_60 + 1;
                else if (days between 61 and 90) then overdue_61_90 = overdue_61_90 + 1;
                else if (days between 91 and 360) then overdue_91_360 = overdue_91_360 + 1;
                else if (days > 360) then overdue_360 = overdue_360 + 1;
    
                i = i + 1;
            end
        end
    end
        client_data = BLOB_APPEND(client_data, '<br><b>Информация о просроченных задолженностях из завершенных контрактов:</b><br>');
        client_data = BLOB_APPEND(client_data, 'Просрочка 7-30 дней: <b>'  || iif(overdue_7_30 > 0, 'Да (' || overdue_7_30 || ')', 'Нет') || '</b><br>');
        client_data = BLOB_APPEND(client_data, 'Просрочка 31-60 дней: <b>' || iif(overdue_31_60 > 0, 'Да (' || overdue_31_60 || ')', 'Нет') || '</b><br>');
        client_data = BLOB_APPEND(client_data, 'Просрочка 61-90 дней: <b>' || iif(overdue_61_90 > 0, 'Да (' || overdue_61_90 || ')', 'Нет') || '</b><br>');
        client_data = BLOB_APPEND(client_data, 'Просрочка 91-360 дней: <b>' || iif(overdue_91_360 > 0, 'Да (' || overdue_91_360 || ')', 'Нет') || '</b><br>');
        client_data = BLOB_APPEND(client_data, 'Просрочка свыше 360 дней: <b>' || iif(overdue_360 > 0, 'Да (' || overdue_360 || ')', 'Нет') || '</b><br>');
        suspend;
end
