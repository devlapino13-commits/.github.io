CREATE PROCEDURE HP_EDOC_INS_BLOCK_CARD(
    EDOC_KIND D_STR_CODE,
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable cbo_id d_id;
begin
  for select cbo.cbo_id
      from cardblockopers cbo
      where cbo.cbo_from_c_id = :c_id
      into cbo_id
  do
  begin

    res_text = '';
    select edl_id, res_text
    from hp_edoc_insert(:edoc_kind, f_objkind('cardblockopers', 'id'), :cbo_id, :un_id)
    into edl_id, res_text;

    suspend;
  end
end
