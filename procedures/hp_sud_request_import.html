CREATE PROCEDURE HP_SUD_REQUEST_IMPORT(
    K_KONTRACT_NO D_STR_20)
RETURNS (
    K_ID D_ID,
    K_FACT_DISB_DATE D_DATE,
    OFFICE_ID D_INTEGER,
    OFFICE_NAME D_STR_100,
    KF_NAME D_STR_100,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    ZAD_TOTAL D_CURRENCY,
    UN_ID D_ID,
    UN_NAME D_STR_CODE,
    NOTE D_STR_255,
    NOTE_ID D_SMALLINT)
AS
begin
  -- создание заявок судебного модуля путем импорта из excel (RFCRU-3149)

  begin
    un_id = uxf$user();
    when any do
      note = 'Ошибка: пользователь не найден';
  end

  select k_id,
         k_fact_disb_date,
         kal_new_reg_id,
         kal_new_kf_name
  from kontract
  inner join knaktlist on k_from_kf_id = kal_from_kf_id
  where k_kontract_no = :k_kontract_no and
        is_delete = 0 and
        current_date between kal_date and kal_date_end
  into k_id,
       k_fact_disb_date,
       office_id,
       kf_name;

  if (not exists(select *
                 from aarep_region_user_tmp(:un_id)
                 where reg_list = :office_id)) then
    note = 'Ошибка: нет доступа к офису';

  if (exists(select smr_id
             from sud_mainreq jmr, sud_mainlog jml
             where jml.sul_id = jmr.smr_last_sul_id and
                   jmr.smr_from_k_id = :k_id and
                   jml.sul_status in (-1, 0, 1))) then
    note = 'Ошибка: договор уже добавлен';

  select lower(un_name)
  from username
  where un_id = :un_id
  into un_name;

  select reg_code || ' ' || reg_name
  from region
  where reg_unicode = :office_id
  into office_name;

  select cz_from_c_id,
         cz_name_fio,
         cur_cl_all
  from print_client_zadol(:k_id)
  rows 1
  into c_id,
       c_name_fio,
       zad_total;

  if (coalesce(note, '') = '') then
  begin
    note = 'Готов к импорту';
    note_id = 1;
  end
  else
    note_id = -1;

  suspend;
end
