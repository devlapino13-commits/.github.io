CREATE PROCEDURE HP_GET_KNCLIENT_TYPE(
    KC_ID D_ID,
    CURR_CITIZEN_ID D_INTEGER,
    DOC_VNJ_ID D_INTEGER,
    REP_DATE D_DATE = null,
    K_ID D_ID = null)
RETURNS (
    RES_CLIENT_TYPE D_INTEGER,
    RES_NON_RESIDENT D_INTEGER,
    RES_FROM_CDS_ID D_INTEGER,
    RES_COUNTRY_CODE D_STR_20,
    RES_COUNTRY_ID D_INTEGER,
    PDT_MEMO D_STR_255)
AS
declare variable C_ID D_ID;
declare variable CLIENT_TYPE D_INTEGER;
begin
  if (k_id is null) then
  begin
    select kc_from_c_id, kc_from_k_id from knclient
    where kc_id = :kc_id
    into c_id, k_id;
  end else
  begin
    c_id = :kc_id;
  end 

  client_type = 0;
  if (exists (select * from kontract, product where k_id = :k_id
      and k_from_pr_id = pr_id
      and pr_code containing '.business.')) then client_type = 2;

  for
  select res_client_type, res_non_resident, res_from_cds_id, res_country_code, res_country_id, pdt_memo
  from hp_get_client_type(:c_id, :curr_citizen_id, :doc_vnj_id, :rep_date)
  into :res_client_type, :res_non_resident, :res_from_cds_id, :res_country_code, :res_country_id, :pdt_memo
  do
  begin
    if (client_type <> 0) then res_client_type = client_type;

    suspend;
  end
end
