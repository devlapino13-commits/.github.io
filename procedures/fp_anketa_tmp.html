CREATE PROCEDURE FP_ANKETA_TMP(
    REP_DATE DATE,
    C_ID ID_TYPE)
RETURNS (
    FPA_CITY ID_TYPE,
    FPA_CLIENT_FIO STR_255,
    FPA_CITIZEN INTEGER,
    FPA_DATE_PLACE_BIRTH STR_255,
    FPA_PASS_KIND INTEGER,
    FPA_PASS_PROPS STR_255,
    FPA_MCARD_PROPS STR_255,
    FPA_ACARD_PROPS STR_255,
    FPA_INN STR_255,
    FPA_ADDRESS_LEGAL STR_255,
    FPA_REL_START DATE,
    FPA_ANK_RESP ID_TYPE,
    FPA_DATE_CHECK DATE,
    FPA_CELL_NO STR_20,
    FPA_EMAIL STR_100,
    FPA_SNILS STR_100)
AS
declare variable c_birth_date date;
declare variable c_pass_no str_255;
declare variable c_pass_date date;
declare variable c_pass_who str_255;
declare variable ca_m_card_seria str_255;
declare variable ca_m_card_num str_255;
declare variable ca_m_card_dstart date;
declare variable ca_al_doc_name str_255;
declare variable ca_al_doc_seria str_255;
declare variable ca_al_doc_num str_255;
declare variable ca_al_doc_dstart date;
declare variable ca_al_doc_dend date;
declare variable birth_place_str str_255;
declare variable buf_place_str str_255;
declare variable buf_country str_255;
declare variable adl_from_pl_id id_type;
declare variable adc_from_adl_id id_type;
declare variable ca_m_card_dstart_str str_255;
declare variable ca_m_card_dend_str str_255;
declare variable ca_al_doc_dstart_str str_255;
declare variable ca_al_doc_dend_str str_255;
declare variable c_birth_date_str str_255;
declare variable ca_m_card_dend date;
declare variable c_pass_date_str str_255;
declare variable c_birth_country str_100;
declare variable c_birth_city str_100;
declare variable ma_id id_type;
declare variable fpa_city_1 id_type;
declare variable fpa_city_2 id_type;
declare variable fpa_ank_resp_1 id_type;
declare variable fpa_ank_resp_2 id_type;
declare variable fpa_rel_start_1 date;
declare variable fpa_rel_start_2 date;
declare variable ma_id_1 id_type;
declare variable ma_id_2 id_type;
begin
  if (rep_date is null) then
    rep_date = current_date;

  -- город в шапке и ответственное лицо беру из адреса офиса первой выдачи займа клиента где лицо является клиентом или залогодателем.
  -- проверяю среди клиентов
  select reg.reg_from_pl_id, cl.c_id, ks.ks_date_begin, kont.k_from_ma_app_id
  from knclient kncl
  inner join kontract kont on kont.k_id = kncl.kc_from_k_id
  inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id
  inner join region reg on reg.reg_unicode = ks.ks_region_id
  inner join operlist ol on ol.ol_id = ks.ks_from_ol_id
  inner join username un on un.un_id = ol.ol_user_id
  inner join clients cl on cl.c_id = un.un_from_c_id
  where ks.ks_ol_is_delete = 0 and
        ks.ks_state = 1 and
        kncl.kc_amount_approved > 0 and
        kont.k_status <> -1 and
        kncl.kc_from_c_id = :c_id
  order by ks.ks_date_begin
  rows 1
  into :fpa_city_1, :fpa_ank_resp_1, :fpa_rel_start_1, :ma_id_1;

  -- проверяю среди залогодателей
  select reg.reg_from_pl_id, cl.c_id, ks.ks_date_begin, kont.k_from_ma_app_id
  from zalog zal
  inner join kontract kont on kont.k_id = zal.z_from_k_id
  inner join kontstate ks on ks.ks_from_k_id = k_id
  inner join region reg on reg.reg_unicode = ks.ks_region_id
  inner join operlist ol on ol.ol_id = ks.ks_from_ol_id
  inner join username un on un.un_id = ol.ol_user_id
  inner join clients cl on cl.c_id = un.un_from_c_id
  where ks.ks_ol_is_delete = 0 and
        ks.ks_state = 1 and
        zal.z_from_c_id is not null  and
        kont.k_status <> -1 and
        zal.z_from_c_id = :c_id and
        :rep_date between coalesce(zal.z_date_begin, '01.01.1900') and coalesce(zal.z_date_end, '31.12.9999') and
        zal.z_is_ready = 1
  order by ks.ks_date_begin
  rows 1
  into :fpa_city_2, :fpa_ank_resp_2, :fpa_rel_start_2, :ma_id_2;

  -- определяю кем лицо стал реньше клиенто или залогодателем, исходя из этого заполняю шапку
  if (coalesce(fpa_rel_start_1, '01.01.1900') > coalesce(fpa_rel_start_2, '01.01.1900')) then
  begin
    fpa_city = fpa_city_1 ;
    fpa_ank_resp = fpa_ank_resp_1;
    fpa_rel_start = fpa_rel_start_1;
    ma_id = ma_id_1;
  end
  else
  begin
    fpa_city = fpa_city_2;
    fpa_ank_resp = fpa_ank_resp_2;
    fpa_rel_start = fpa_rel_start_2;
    ma_id = ma_id_2;
  end

  select maa.ma_date_app
  from miniapp maa
  where maa.ma_id = :ma_id and
        maa.ma_date_end = '31.12.9999'
  into :fpa_date_check;

  select trim(coalesce(cl.c_name_fio, '')),
         coalesce(cl.c_citizen, 0),
         c_birth_date,

         coalesce(c_pass_no, ''),
         c_pass_date,
         coalesce(c_pass_who, ''),

         coalesce(c_inn, ''),
         coalesce(cl.c_doc_type, 0), -- тип документа
         coalesce(cl.c_birth_country, ''),

         coalesce(cl.c_birth_city, ''),
         (select cont_value
          from hp_contdata_get_cell_main(0, :c_id)),
         (select cont_value
          from hp_contdata_get_email_main(0, :c_id))
  from clients cl
  where cl.c_id = :c_id
  into :fpa_client_fio, :fpa_citizen,   :c_birth_date,
       :c_pass_no,      :c_pass_date,   :c_pass_who,
       :fpa_inn,        :fpa_pass_kind, :c_birth_country,
       :c_birth_city,   :fpa_cell_no,   :fpa_email;

  ca_m_card_seria = '';
  ca_m_card_num = '';
  ca_m_card_dstart = null;
  ca_m_card_dend = null;

  ca_al_doc_name = '';
  ca_al_doc_seria = '';
  ca_al_doc_num = '';
  ca_al_doc_dstart = null;
  ca_al_doc_dend = null;
  fpa_snils = null;

  select cds_doc_no, cds_doc_date_begin, cds_doc_date_end
  from clientdocs, passkind, passdoctype
  where cds_from_c_id = :c_id and
        cds_from_pk_id = pk_id and
        pk_from_pdt_id = pdt_id and
        pdt_code = 'magration_card' and
        cds_is_delete = 0
  order by cds_from_c_id desc, cds_doc_date_begin desc
  rows 1
  into :ca_m_card_num, :ca_m_card_dstart, :ca_m_card_dend;

  select cds_doc_no
  from clientdocs, passkind, passdoctype
  where cds_from_c_id = :c_id and
        cds_from_pk_id = pk_id and
        pk_from_pdt_id = pdt_id and
        pdt_code = 'snils' and
        cds_is_delete = 0
  order by cds_from_c_id desc, cds_doc_date_begin desc
  rows 1
  into :fpa_snils;

  fpa_snils = coalesce(fpa_snils, '');

  select pdt_code, cds_doc_no, cds_doc_date_begin, cds_doc_date_end
  from clientdocs, passkind, passdoctype
  where cds_from_c_id = :c_id and
        cds_from_pk_id = pk_id and
        pk_from_pdt_id = pdt_id and
        pdt_code in ('registration_fiz', 'registration_yur', 'registration_mbf', 'registration_rvp', 'registration_vnj') and
        cds_is_delete = 0
  order by cds_from_c_id desc, cds_doc_date_begin desc
  rows 1
  into :ca_al_doc_name, :ca_al_doc_num, :ca_al_doc_dstart, :ca_al_doc_dend;

  if (ca_al_doc_name in ('registration_fiz', 'registration_yur', 'registration_mbf')) then
    ca_al_doc_name = 'временная регистрация';
  else
  if (ca_al_doc_name = 'registration_rvp') then
    ca_al_doc_name = 'РВП';
  else
  if (ca_al_doc_name = 'registration_vnj') then
    ca_al_doc_name = 'вид на жит-во';

  /*Реквизиты документа, удостоверяющего личность*/
  if (trim(c_birth_city) = '-') then
    c_birth_city = '';
  birth_place_str = c_birth_country || ' ' || c_birth_city;

  c_pass_date_str = '';
  if (c_pass_date is not null) then
  begin
    c_pass_date_str = replace(iif(extract(day from c_pass_date) < 10, '0', '') || extract(day from c_pass_date) || '.' || iif(extract(month from c_pass_date) < 10, '0', '') || extract(month from c_pass_date) || '.' || extract(year from c_pass_date), ' ', '');
  end

  fpa_pass_props = substring('Серия, Номер:' || c_pass_no || '; Дата Выдачи:' || c_pass_date_str || '; Орган, выдавший документ:' || c_pass_who from 1 for 255);

  /*Данные миграционной карты --------------------------------------------->*/
  fpa_mcard_props = '';

  if (coalesce(ca_m_card_seria, '') <> '') then
    fpa_mcard_props = 'Серия: ' || ca_m_card_seria || ';';
  if (coalesce(ca_m_card_num, '') <> '') then
    fpa_mcard_props = fpa_mcard_props || ' Номер: ' || ca_m_card_num || ';';

  ca_m_card_dstart_str = '';
  if (ca_m_card_dstart is not null) then
  begin
    ca_m_card_dstart_str = replace(iif(extract(day from ca_m_card_dstart) < 10, '0', '') || extract(day from ca_m_card_dstart) || '.' || iif(extract(month from ca_m_card_dstart) < 10, '0', '') || extract(month from ca_m_card_dstart) || '.' || extract(year from ca_m_card_dstart), ' ', '');
  end

  if (ca_m_card_dstart_str <> '') then
    fpa_mcard_props = fpa_mcard_props || ' Дата выдачи: ' || ca_m_card_dstart_str || ';';

  ca_m_card_dend_str = '';
  if (ca_m_card_dend is not null) then
  begin
    ca_m_card_dend_str = replace(iif(extract(day from ca_m_card_dend) < 10, '0', '') || extract(day from ca_m_card_dend) || '.' || iif(extract(month from ca_m_card_dend) < 10, '0', '') || extract(month from ca_m_card_dend) || '.' || extract(year from ca_m_card_dend), ' ', '');
  end

  if (ca_m_card_dend_str <> '') then
    fpa_mcard_props = fpa_mcard_props || ' Дата окончания: ' || ca_m_card_dend_str || ';';

  fpa_mcard_props = trim(fpa_mcard_props);
  /*<-----------------------------------------------------------------------*/

  /* Данные документа, подтверждающего право иностранного гражданина или лица без гражданства на пребывание (проживание) в РФ */

  fpa_acard_props = '';
  ca_al_doc_name = trim(ca_al_doc_name);
  if (coalesce(ca_al_doc_name, '') <> '') then
    fpa_acard_props = 'Тип документа: ' || ca_al_doc_name || ';';
  if (coalesce(ca_al_doc_seria, '') <> '') then
    fpa_acard_props = fpa_acard_props || ' Серия: ' || ca_al_doc_seria || ';';
  if (coalesce(ca_al_doc_num, '') <> '') then
    fpa_acard_props = fpa_acard_props || ' Номер: ' || ca_al_doc_num || ';';

  ca_al_doc_dstart_str = '';
  if (ca_al_doc_dstart is not null) then
  begin
    ca_al_doc_dstart_str = replace(iif(extract(day from ca_al_doc_dstart) < 10, '0', '') || extract(day from ca_al_doc_dstart) || '.' || iif(extract(month from ca_al_doc_dstart) < 10, '0', '') || extract(month from ca_al_doc_dstart) || '.' || extract(year from ca_al_doc_dstart), ' ', '');
  end

  if (ca_al_doc_dstart_str <> '') then
    fpa_acard_props = fpa_acard_props || ' Дата прибытия: ' || ca_al_doc_dstart_str || ';';

  ca_al_doc_dend_str = '';
  if (ca_al_doc_dend is not null) then
  begin
    ca_al_doc_dend_str = replace(iif(extract(day from ca_al_doc_dend) < 10, '0', '') || extract(day from ca_al_doc_dend) || '.' || iif(extract(month from ca_al_doc_dend) < 10, '0', '') || extract(month from ca_al_doc_dend) || '.' || extract(year from ca_al_doc_dend), ' ', '');
  end

  if (ca_al_doc_dend_str <> '') then
    fpa_acard_props = fpa_acard_props || ' Дата окончания: ' || ca_al_doc_dend_str || ';';

  fpa_acard_props = trim(fpa_acard_props);

  fpa_date_place_birth = '';

  c_birth_date_str = '';
  if (c_birth_date is not null) then
  begin
    c_birth_date_str = replace(iif(extract(day from c_birth_date) < 10, '0', '') || extract(day from c_birth_date) || '.' || iif(extract(month from c_birth_date) < 10, '0', '') || extract(month from c_birth_date) || '.' || extract(year from c_birth_date), ' ', '');
  end

  fpa_date_place_birth = c_birth_date_str || ' ' || birth_place_str;
  fpa_date_place_birth = trim(fpa_date_place_birth);
  /*<< ---------------------------------------------------------------------*/
  fpa_address_legal = '';

  for select adc_from_adl_id
      from hp_addrclient(:c_id)
      where adc_address_type = 1 -- Юридически = Адрес регистрации
      into :adc_from_adl_id
  do
  begin
    buf_country = '';
    buf_place_str = '';

    select adl.adl_from_pl_id, adl_address_str
    from addrlist adl
    where adl.adl_id = :adc_from_adl_id
    into :adl_from_pl_id, :buf_place_str;

    if (coalesce(buf_place_str, '') = 'Не указан') then
      buf_place_str = '';
    if ((adl_from_pl_id is not null) and
        (buf_place_str <> '')) then
    begin
      select pl.pl_name
      from places pl
      where pl.pl_id = cast(substring(cast(:adl_from_pl_id as varchar(18)) from 1 for 3) || '000000001' as decimal(18,0))
      into :buf_country;
    end

    fpa_address_legal = coalesce(buf_country, '') || ' ' || coalesce(buf_place_str, '');
  end
  suspend;
end
