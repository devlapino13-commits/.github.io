CREATE PROCEDURE HP_OPERCASH_DO_INSURANCE(
    OPC_ID D_ID,
    OL_ID D_ID)
AS
declare variable office_id d_integer;
declare variable office_name d_str_100;
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable pr_code d_str_100;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable opc_date d_date;
declare variable opc_amount d_currency;
declare variable opc_is_ipf d_bool;
declare variable al_account d_str_code;
declare variable al_id_go d_str_20;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable ob_detail d_str_255;
declare variable ik_mbulak_share d_currency;
begin
  -- Построение доп. проводок по проекту страхования кредита

  select opc_office_id,
         opc_from_k_id,
         opc_from_c_id,
         opc_date,
         opc_amount,
         opc_is_ipf
  from opercash
  where opc_id = :opc_id and
        opc_is_delete = 0 and
        opc_oper_kind = f_enum_id('cashoperkind', 'plus.loan.insurance')
  into office_id,
       k_id,
       c_id,
       opc_date,
       opc_amount,
       opc_is_ipf;

  if (office_id is null or
      k_id is null or
      c_id is null or
      opc_date is null or
      opc_amount is null) then
    exit;

  if (opc_is_ipf = 1) then
  begin
    select pr_code
    from kontract
    inner join product on pr_id = k_from_pr_id
    where k_id = :k_id
    into pr_code;

    if (pr_code containing '.ipf.' and
        pr_code not containing '.ignore_insurance.') then
      exit; -- при оплате страхования ИПФ доп. проводки не строятся (RFCKG-5390)
  end

  select reg_code || ' ' || reg_name
  from region
  where reg_unicode = :office_id
  into office_name;

  select k_kontract_no,
         k_disb_date,
         ik_mbulak_share,
         ag_from_al_id
  from kontract
  inner join insurance_kontract on ik_from_k_id = k_id
  inner join agreements on ag_id = ik_from_ag_id
  where k_id = :k_id
  into k_kontract_no,
       k_disb_date,
       ik_mbulak_share, -- доля МБулак (RFCKG-4937)
       al_id_go; -- счет страхования в ГО

  select c_name_fio
  from clients
  where c_id = :c_id
  into c_name_fio;

  -- Проводка "Комиссия МБулак"
  ob_detail = 'Комиссия МБулак по дог. о сотрудничестве по страхованию';

  select al_account
  from acc_list
  where al_id = :al_id_go
  into al_account; -- короткий код счета (группа счетов/субсчет контрагента, например 21122/301)

  select al_id
  from acc_list
  where al_currency = '417' and
        al_from_reg_id = :office_id and
        al_account = :al_account
  into acc_dt; -- счет страхования в офисе, где принимается платеж (RFCKG-4907)

  if (acc_dt = '') then
    exception e_message 'Счет Дт для комиссии МБулак не определен, офис ' || office_name;

  if (opc_is_ipf = 1) then
    al_account = '21114/201';
  else
    al_account = '21114/20';

  select al_id
  from acc_list
  where al_currency = '417' and
        al_from_reg_id = :office_id and
        al_account = :al_account
  into acc_kt;

  if (acc_kt = '') then
    exception e_message 'Счет Kт для комиссии МБулак не определен, офис ' || office_name;

  insert into operbook (ob_from_ol_id, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_detail)
  values (:ol_id, :acc_dt, :acc_kt, :ik_mbulak_share, :ik_mbulak_share, :opc_date, :office_id, :ob_detail);
end
