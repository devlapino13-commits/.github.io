CREATE PROCEDURE FIND_SB_AKT(
    REP_REGION_ID INTEGER,
    REP_DATE DATE,
    REP_KIND INTEGER)
RETURNS (
    SBA_FROM_K_ID DECIMAL(18,0),
    SBA_NUMBER INTEGER,
    SBA_DATE_BEGIN DATE,
    SBA_DATE_END DATE,
    SBA_FROM_SBL_ID INTEGER,
    SBA_STATUS INTEGER,
    SBA_FROM_UN_ID DECIMAL(18,0),
    SBA_DATETIME TIMESTAMP,
    SBA_IS_DELETE INTEGER,
    SBA_REGION_ID INTEGER,
    SBA_LO_ID DECIMAL(18,0),
    SBA_KF_NAME VARCHAR(100),
    SBA_REGION_NAME VARCHAR(100),
    SBA_LO_FIO VARCHAR(100),
    SBA_KONT_STATE VARCHAR(20),
    SBA_KONT_STATE_DATE DATE,
    SBL_NAME VARCHAR(255),
    SBL_FROM_C_ID DECIMAL(18,0),
    SBL_OKRUG_ID INTEGER,
    SBL_TERRITORY VARCHAR(255),
    SBL_TEAM_NAME VARCHAR(255),
    SBL_CATEGORY VARCHAR(255),
    SBL_COORDINATOR DECIMAL(18,0),
    SBL_MEMO VARCHAR(255),
    SBL_DATE_BEGIN DATE,
    SBL_DATE_END DATE,
    SBL_TYPE_STR VARCHAR(20),
    SBL_PHONE VARCHAR(255),
    SBL_EMAIL VARCHAR(255),
    SBL_C_NAME_FIO VARCHAR(100),
    SBL_OKRUG_NAME VARCHAR(100),
    SBL_COUNTRY_NAME VARCHAR(100),
    SBL_COORDINATOR_FIO VARCHAR(100),
    SBA_KONTRACT_NO VARCHAR(20),
    AAREP_CLIENTS_KOL INTEGER,
    AAREP_CASHIER_ID DECIMAL(18,0),
    AAREP_LEADER_ID DECIMAL(18,0),
    AAREP_CASHIER_FIO VARCHAR(100),
    AAREP_LEADER_FIO VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_OFFICE_POPULATE_NAME VARCHAR(100),
    AA_OFFICE_STAT_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_CURR_LEVEL2_NAME VARCHAR(100),
    AA_CURR_LEVEL3_NAME VARCHAR(100),
    SBA_DISB_DATE DATE,
    SBA_BALANCE DECIMAL(18,4),
    REP_BALANCE DECIMAL(18,4))
AS
declare variable reg_id_prior integer;
declare variable lo_id_prior decimal(18,0);
declare variable ks_state integer;
declare variable sbl_id_prior integer;
declare variable sbl_type integer;
declare variable lev_id integer;
declare variable s_level varchar(100);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  reg_id_prior = 0;
  lo_id_prior = 0;
  sbl_id_prior = 0;

  for select sba_from_k_id, sba_number, sba_date_begin, sba_date_end, sba_from_sbl_id, sba_status, sba_from_un_id,
             sba_datetime, sba_is_delete,

             kal_new_kf_officer, kal_new_reg_id, kal_new_kf_name, ks_state, ks_date_begin,

             k_kontract_no, k_kolich_client_approved, k_clients_cashier_id, k_clients_leader_id, k_disb_date
      from sb_akt, kontract, knaktlist, kontstate, region_list(:rep_region_id, 1, -1, 1)
      where sba_from_k_id = k_id and
            sba_is_delete = 0 and
            sba_from_sbl_id <> 0 and
            ks_from_k_id = k_id and
            ks_date_begin <= :rep_date and
            ks_date_end >= :rep_date and
            ks_ol_is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_date <= :rep_date and
            kal_date_end >= :rep_date and
            sba_from_sbl_id <> 0 and
            ((:rep_kind >= 1 and
            sba_status = :rep_kind and
            sba_date_end >= :rep_date) or
            (:rep_kind in (-1, 0) and
            sba_status = :rep_kind) or
            (:rep_kind = -2)) and
            kal_new_reg_id = id_list and
            sba_date_end > sba_date_begin
      into :sba_from_k_id, :sba_number, :sba_date_begin, :sba_date_end, :sba_from_sbl_id, :sba_status, :sba_from_un_id,
           :sba_datetime,

           :sba_is_delete, :sba_lo_id, :sba_region_id, :sba_kf_name, :ks_state, :sba_kont_state_date,

           :sba_kontract_no, :aarep_clients_kol, :aarep_cashier_id, :aarep_leader_id, :sba_disb_date
  do
  begin
    if (ks_state = 0) then
      sba_kont_state = 'заявка';
    else
    if (ks_state = 1) then
      sba_kont_state = 'активный';
    else
    if (ks_state = 2) then
      sba_kont_state = 'закрытый';

    if (sba_region_id <> reg_id_prior) then
    begin
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :sba_region_id
      into :sba_region_name;
      reg_id_prior = sba_region_id;

      lev_id = 0;
      select branch.reg_code || ' ' || branch.reg_name, office.reg_populate_kind, office.reg_stat_rep_region,
             r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name
      from region office, region branch, region r2
      where office.reg_unicode = :sba_region_id and
            branch.reg_unicode = office.parent and
            r2.reg_unicode = branch.parent
      into :aa_curr_branch_name, :aa_office_populate_name, :aa_office_stat_name, :lev_id, :s_level;

      if ((lev_id is not null) and
          (lev_id <> 0) and
          (lev_id <> 1000000)) then
      begin
        aa_curr_level_name = s_level;
      end
      else
      begin
        aa_curr_level_name = aa_curr_branch_name;
        lev_id = 0;
      end

      select r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :lev_id and
            r2.reg_unicode = r1.parent
      into :lev_id, :s_level;
      if ((lev_id is not null) and
          (lev_id <> 0) and
          (lev_id <> 1000000)) then
      begin
        aa_curr_level2_name = s_level;
      end
      else
      begin
        aa_curr_level2_name = aa_curr_level_name;
        lev_id = 0;
      end

      select r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :lev_id and
            r2.reg_unicode = r1.parent
      into :lev_id, :s_level;
      if ((lev_id is not null) and
          (lev_id <> 0) and
          (lev_id <> 1000000)) then
      begin
        aa_curr_level3_name = s_level;
      end
      else
      begin
        aa_curr_level3_name = aa_curr_level2_name;
        lev_id = 0;
      end

    end
    if (sba_lo_id <> lo_id_prior) then
    begin
      select c_name_fio
      from clients
      where c_id = :sba_lo_id
      into :sba_lo_fio;
      lo_id_prior = sba_lo_id;
    end

    aarep_cashier_fio = null;
    select c_name_fio
    from clients
    where c_id = :aarep_cashier_id
    into :aarep_cashier_fio;

    aarep_leader_fio = null;
    select c_name_fio
    from clients
    where c_id = :aarep_leader_id
    into :aarep_leader_fio;

    if (sbl_id_prior <> sba_from_sbl_id) then
    begin
      select sbl_name, sbl_from_c_id, sbl_okrug_id, sbl_territory, sbl_team_name, sbl_category, sbl_coordinator,
             sbl_memo, sbl_date_begin, sbl_date_end, sbl_type, sbl_phone, sbl_email, c1.c_name_fio, c2.c_name_fio,
             okl_name, rd_name
      from sb_list, clients c1, clients c2, okruglist, refdata
      where sbl_id = :sba_from_sbl_id and
            sbl_from_c_id = c1.c_id and
            sbl_coordinator = c2.c_id and
            sbl_okrug_id = okl_id and
            okl_country = rd_id
      into :sbl_name, :sbl_from_c_id, :sbl_okrug_id, :sbl_territory, :sbl_team_name, :sbl_category, :sbl_coordinator,
           :sbl_memo, :sbl_date_begin, :sbl_date_end, :sbl_type, :sbl_phone, :sbl_email, :sbl_c_name_fio,
           :sbl_coordinator_fio, :sbl_okrug_name, :sbl_country_name;
      sbl_type_str = 'Не определен';
      if (sbl_type = 0) then
        sbl_type_str = 'СБ';
      else
      if (sbl_type = 1) then
        sbl_type_str = 'Коллекторы';
      sbl_id_prior = sba_from_sbl_id;
    end
    select sum(ok_os)
    from operkred
    where ok_from_k_id = :sba_from_k_id and
          ok_ol_date between :sba_disb_date and :sba_date_begin and
          ok_ol_is_active = 1 and
          ok_from_c_id = 0 and
          ok_type <= 8
    into :sba_balance;
    if (sba_balance is null) then
      sba_balance = 0;

    if (exists(select *
               from operkroff okf
               where okf.oko_from_k_id = :sba_from_k_id and
                     okf.oko_ol_date <= :rep_date)) then
    begin
      select sum(coalesce(op.oko_os, 0) + coalesce(op.oko_perc, 0) + coalesce(op.oko_other, 0) + coalesce(op.oko_sh_other, 0) + coalesce(op.oko_sh_os, 0) + coalesce(op.oko_sh_perc, 0))
      from operkroff op
      where oko_from_k_id = :sba_from_k_id and
            oko_ol_date <= :rep_date and
            op.oko_ol_is_active = 1
      into :rep_balance; -- текущий баланс
      rep_balance = coalesce(:rep_balance, 0);
    end
    else
    begin
      select sum(zad_os)
      from aarep_make_zad_list(:sba_from_k_id, :rep_date, 1)
      into rep_balance; -- текущий баланс
      rep_balance = coalesce(:rep_balance, 0);
    end
    if (rep_balance < 0) then
      :rep_balance = 0;
    suspend;
  end
end
