CREATE PROCEDURE AAREP_LAW_TRANS_ORIV(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RES_ID INTEGER,
    RES_DATE_BEGIN DATE,
    RES_DELO_NO STR_100,
    RES_KONTRACT_NO STR_20,
    RES_K_FACT_DISB_DATE DATE,
    RES_STATE_STR STR_20,
    RES_COST_ALL CURR_TYPE,
    RES_SUM_JACT CURR_TYPE,
    RES_SUM_COLL CURR_TYPE,
    LAW_FIO STR_100,
    REG_NAME_OFFICE STR_100,
    REG_NAME STR_100,
    PAY_OS CURR_TYPE,
    PAY_PERC CURR_TYPE,
    PAY_TAX CURR_TYPE,
    PAY_SUD CURR_TYPE,
    PAY_OKF_OS CURR_TYPE,
    PAY_OKF_PERC CURR_TYPE,
    DEBT_SUD CURR_TYPE,
    RES_DELINQ_DAYS INTEGER,
    LAST_DATE DATE,
    DAYS_ACT_1 INTEGER,
    K_AMOUNT_APPROVED D_CURRENCY,
    TOTAL_DEBT D_CURRENCY,
    RES_DATE_END DATE)
AS
declare variable res_k_id id_type;
declare variable res_state integer;
declare variable rd_id integer;
declare variable ol_date_kred date;
declare variable ol_date_krof date;
declare variable ol_date_extra date;
declare variable sba_date_begin date;
declare variable ok_ol_date date;
declare variable max_3 date;
declare variable days_act integer;
declare variable ok_ol_date_1 date;
declare variable standart_debt d_currency;
declare variable okf_debt d_currency;
declare variable oe_debt d_currency;
begin
  RD_ID = F_GET_RD_ID('OperExtraKind.LawClientExpense');

  for select LAM_ID, LAM_DATE_BEGIN, LAM_DELO_NO, LAM_COST_ALL, LAM_SUM_JACT, LAM_SUM_COLL, CL.C_NAME_FIO, LAM_DATE_RESULT
      from LAWMAIN LAM
      inner join CLIENTS CL on CL.C_ID = LAM.LAM_LAWER_ID
      where LAM_DATE_BEGIN between :REP_DATE_BEGIN and :REP_DATE_END and
            LAM_IS_ACTIVE = 1
      order by LAM_DATE_BEGIN, LAM_ID
      into :RES_ID, :RES_DATE_BEGIN, :RES_DELO_NO, :RES_COST_ALL, :RES_SUM_JACT, :RES_SUM_COLL, :LAW_FIO, :RES_DATE_END
  do
  begin
    for select first 1 K_ID, K_KONTRACT_NO, KS_STATE, REG.REG_NAME, RG.REG_NAME, KT.K_FACT_DISB_DATE, KT.K_AMOUNT_APPROVED
        from LAWKREDIT
        inner join KONTRACT KT on K_ID = LAK_FROM_K_ID
        inner join REGION REG on KT.K_LAST_REG_ID = REG.REG_UNICODE
        inner join REGION RG on RG.REG_UNICODE = REG.PARENT
        inner join KONTSTATE on KS_FROM_K_ID = K_ID and
              KS_OL_IS_DELETE = 0 and
              :REP_DATE_END between KS_DATE_BEGIN and KS_DATE_END
        where LAK_FROM_LAM_ID = :RES_ID
        INTO :RES_K_ID, :RES_KONTRACT_NO, :RES_STATE, :REG_NAME_OFFICE, :REG_NAME, :RES_K_FACT_DISB_DATE, :K_AMOUNT_APPROVED
    do
    --========================
    begin
      standart_debt = 0;
      okf_debt = 0;
      oe_debt = 0;

      select sum(zad_os + sh_calc_perc + sh_calc_fine)
      from aarep_make_zad_list(:RES_K_ID, :REP_DATE_END, 1)
      into :standart_debt;

      standart_debt = coalesce(standart_debt, 0);

      select sum(okf.oko_os + okf.oko_perc + okf.oko_other + okf.oko_sh_os + okf.oko_sh_perc + okf.oko_sh_other)
      from operkroff okf
      where okf.oko_ol_is_active = 1
      and okf.oko_ol_date <= :rep_date_end
      and okf.oko_from_k_id = :res_k_id
      into :okf_debt;

      okf_debt = coalesce(okf_debt, 0);

      select sum(oe.ox_amount_plus - oe.ox_amount_minus - oe.ox_amount_writeoff)
      from operextra oe
      where oe.ox_ol_is_active = 1
      and oe.ox_object_kind = 9
      and oe.ox_object_id = :res_k_id
      and oe.ox_from_rd_id = :rd_id
      into :oe_debt;

      oe_debt = coalesce(oe_debt, 0);

      total_debt = standart_debt + okf_debt + oe_debt;

      :OL_DATE_KRED = null;
      :OL_DATE_KROF = null;
      :OL_DATE_EXTRA = null; --last_DATE для данного поля находим последнюю дату операции по данному кредиту из OPERKRED,OPERKROFF,OPEREXTRA

      select OK.OK_OL_DATE
      from OPERKRED OK
      where OK.OK_FROM_K_ID = :RES_K_ID and
            OK.OK_TYPE in (4, 9) and
            OK.OK_OL_IS_ACTIVE = 1 and
            OK.OK_OL_DATE <= :REP_DATE_END and
            ((OK.OK_OS <> 0) or (OK.OK_PERC <> 0) or (OK.OK_FINE <> 0))
      order by OK.OK_OL_DATE desc rows 1
      into :OL_DATE_KRED; --(стандартный учет)

      select OPO.OKO_OL_DATE
      from OPERKROFF OPO
      where OPO.OKO_FROM_K_ID = :RES_K_ID and
            OPO.OKO_ID = 4 and
            OPO.OKO_OL_IS_ACTIVE = 1 and
            OPO.OKO_OL_DATE <= :REP_DATE_END and
            ((OPO.OKO_OS <> 0) or (OPO.OKO_PERC <> 0))
      order by OPO.OKO_OL_DATE desc rows 1
      into :OL_DATE_KROF; -- (внесистемка)

      select OP.OX_OL_DATE
      from OPEREXTRA OP
      where OP.OX_OBJECT_ID = :RES_K_ID and
            OP.OX_AMOUNT_MINUS > 0 and
            OP.OX_OL_IS_ACTIVE = 1 and
            OP.OX_OL_DATE <= :REP_DATE_END
      order by OP.OX_OL_DATE desc rows 1
      into :OL_DATE_EXTRA; -- (госпошлина)

      OL_DATE_KRED = coalesce(OL_DATE_KRED, '01.01.1001');
      OL_DATE_KROF = coalesce(OL_DATE_KROF, OL_DATE_KRED);
      OL_DATE_EXTRA = coalesce(OL_DATE_EXTRA, OL_DATE_KRED);

      if (:OL_DATE_KRED >= :OL_DATE_KROF) then
        :MAX_3 = :OL_DATE_KRED;
      else
        :MAX_3 = :OL_DATE_KROF;

      if (:OL_DATE_EXTRA >= :MAX_3) then
        :MAX_3 = :OL_DATE_EXTRA;

      if (:MAX_3 = '01.01.1001') then
        :LAST_DATE = null;
      else
        :LAST_DATE = :MAX_3;
      ------------------------------------------------------------------------------
      SBA_DATE_BEGIN = null; --Алгоритм для расчета поля days_act_1
      OK_OL_DATE = null; --Если  КД передан в ОРиВ   после списания то берем количество просроченных дней на дату списания,
      DAYS_ACT = null;
      DAYS_ACT_1 = null;

      select S.SBA_DATE_BEGIN
      from SB_AKT S
      where S.SBA_FROM_K_ID = :RES_K_ID and
            S.SBA_IS_DELETE = 0 and
            S.SBA_FROM_SBL_ID = 30 and
            S.SBA_DATE_BEGIN <= :REP_DATE_END
      order by S.SBA_DATE_BEGIN desc rows 1
      into :SBA_DATE_BEGIN; -- дата передачи в ОРиВ

      select cast(:SBA_DATE_BEGIN as date) - OKS.OKS_DELINQ_DATE -- число дней активной просрочки
      from OKSUMMARY OKS
      where :SBA_DATE_BEGIN between OKS.OKS_DATE_BEGIN and OKS.OKS_DATE_END and
            OKS.OKS_FROM_K_ID = :RES_K_ID
            and oks_close_state = 0 
      into :DAYS_ACT; -- активная просрочка на дату передачи в ОРиВ

      if (:DAYS_ACT < 0) then
        :DAYS_ACT_1 = 0;
      else
      if (:DAYS_ACT is null) then -- если кредит не активный на день передачи в ОРИВ
      begin
        select OL.OL_DATE
        from KONTRACT KONT
        inner join OPERLIST OL on OL.OL_ID = KONT.K_FACT_CLOSED_OL_ID
        where KONT.K_ID = :RES_K_ID and
              OL.IS_DELETE = 0
        into :OK_OL_DATE;

        :OK_OL_DATE_1 = OK_OL_DATE - 1;

        select cast(:OK_OL_DATE_1 as date) - OKS.OKS_DELINQ_DATE -- число дней активной просрочки
        from OKSUMMARY OKS
        where :OK_OL_DATE_1 between OKS.OKS_DATE_BEGIN and OKS.OKS_DATE_END and
              OKS.OKS_FROM_K_ID = :RES_K_ID
              and oks_close_state = 0 
        into :DAYS_ACT;
        if (:DAYS_ACT < 0) then
          :DAYS_ACT_1 = 0;
        else
        if (:DAYS_ACT is null) then
          :DAYS_ACT_1 = null;
        else
          :DAYS_ACT_1 = DAYS_ACT + 1;
      end
      else
        :DAYS_ACT_1 = :DAYS_ACT;

      ------------------------------------------------------------------------------
      RES_DELINQ_DAYS = 0;
      select :RES_DATE_BEGIN - OKS_DELINQ_DATE
      from OKSUMMARY OKS
      where OKS_FROM_K_ID = :RES_K_ID and
            :RES_DATE_BEGIN between OKS_DATE_BEGIN and OKS_DATE_END
            and oks_close_state = 0 
      into RES_DELINQ_DAYS;
      if (coalesce(RES_DELINQ_DAYS, -1) < 0) then
        RES_DELINQ_DAYS = 0;

      PAY_OS = 0;
      PAY_PERC = 0;
      PAY_TAX = 0;

      -- погашения по стандартному учету
      select -sum(OK.OK_OS), -sum(OK.OK_PERC), -sum(OK.OK_FINE)
      from OPERKRED OK
      where OK.OK_FROM_K_ID = :RES_K_ID and
            OK.OK_TYPE in (4, 9) and
            OK.OK_OL_IS_ACTIVE = 1 and
            OK.OK_OL_DATE between :REP_DATE_BEGIN and :REP_DATE_END
      into :PAY_OS, :PAY_PERC, :PAY_TAX;

      PAY_OS = coalesce(PAY_OS, 0);
      PAY_PERC = coalesce(PAY_PERC, 0);
      PAY_TAX = coalesce(PAY_TAX, 0);

      ----- госпошлина (судебные издержки)
      PAY_SUD = 0;

      select sum(OE.OX_AMOUNT_MINUS)
      from OPEREXTRA OE
      where OE.OX_OBJECT_KIND = 9 and
            OE.OX_OBJECT_ID = :RES_K_ID and
            OE.OX_OL_DATE between :REP_DATE_BEGIN and :REP_DATE_END and
            OE.OX_OL_IS_ACTIVE = 1 and
            OE.OX_FROM_RD_ID = :RD_ID
      into :PAY_SUD;

      PAY_SUD = coalesce(PAY_SUD, 0);

      DEBT_SUD = 0;

      select sum(coalesce(OE.OX_AMOUNT_PLUS, 0) - coalesce(OE.OX_AMOUNT_MINUS, 0) - coalesce(OE.OX_AMOUNT_WRITEOFF, 0))
      from OPEREXTRA OE
      where OE.OX_OBJECT_KIND = 9 and
            OE.OX_OBJECT_ID = :RES_K_ID and
            OE.OX_OL_DATE between :REP_DATE_BEGIN and :REP_DATE_END and
            OE.OX_OL_IS_ACTIVE = 1 and
            OE.OX_FROM_RD_ID = :RD_ID
      into :DEBT_SUD;

      DEBT_SUD = coalesce(DEBT_SUD, 0);

      --===============================================

      PAY_OKF_OS = 0;
      PAY_OKF_PERC = 0;

      select -sum(OKF.OKO_OS), -sum(OKF.OKO_PERC)
      from OPERKROFF OKF
      where OKF.OKO_FROM_K_ID = :RES_K_ID and
            OKF.OKO_OL_IS_ACTIVE = 1 and
            OKF.OKO_OL_DATE between :REP_DATE_BEGIN and :REP_DATE_END and
            OKF.OKO_TYPE = 4
      into :PAY_OKF_OS, :PAY_OKF_PERC;

      PAY_OKF_OS = coalesce(PAY_OKF_OS, 0);
      PAY_OKF_PERC = coalesce(PAY_OKF_PERC, 0);

      begin
        if (RES_STATE = 2) then
        begin
          RES_STATE_STR = iif(exists(select OKO_FROM_K_ID
                                     from OPERKROFF
                                     where OKO_FROM_K_ID = :RES_K_ID and
                                           OKO_OL_IS_ACTIVE = 1 and
                                           OKO_OL_DATE <= :REP_DATE_END
                                     group by OKO_FROM_K_ID
                                     having sum(OKO_OS + OKO_PERC + OKO_SH_OS + OKO_SH_PERC) > 0), 'Внесистемный учет', 'Закрытый');
        end
        else
        if (RES_STATE = 1) then
          RES_STATE_STR = 'Активный';
        else
        if (RES_STATE = 0) then
          RES_STATE_STR = 'В процессе';
        suspend;
      end

      --===============================================
    end
  end
end
