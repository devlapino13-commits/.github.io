CREATE PROCEDURE HP_OPERDELMAIN_CHECK(
    IN_ODM_ID D_INTEGER)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable oper_type d_smallint;
declare variable oper_subtype d_smallint;
declare variable res_code d_str_code;
declare variable res_name d_str_100;
declare variable odm_oper_date d_date;
declare variable closed_date d_date;
declare variable min_ol_date d_date;
declare variable ch_count d_integer;
begin
  res_text = '';

  select odm_oper_date
  from operdelmain d
  where odm_id = :in_odm_id
  into odm_oper_date;

  closed_date = cast(f_get_const('kn_operbook_close_date') as date);

  if (odm_oper_date <= closed_date) then
    res_text = 'Опер.день по ЕПС закрыт: ' || f_datetostr(odm_oper_date);

  if (res_text = '') then
  begin
    select min(ol.ol_date)
    from operdellist, operlist ol
    where odl_from_odm_id = :in_odm_id and
          odl_loan_ol_id = ol.ol_id and
          ol.is_delete = 0
    into min_ol_date;
  
    if (min_ol_date <= closed_date) then
      res_text = 'Операция ранней датой: ' || f_datetostr(min_ol_date) || ', ' || ascii_char(13) ||
                 'Сторно не поддерживается';
  end

  if (res_text = '') then
  begin
    for select distinct coalesce(ol_oper_type, 0), coalesce(ol_oper_subtype, 0)
    from operdellist, operlist ol
    where odl_from_odm_id = :in_odm_id and
          odl_loan_ol_id = ol.ol_id and
          ol.is_delete = 0
    order by ol_date desc, odl_id desc
    into oper_type, oper_subtype
    do
    begin
      res_code = '';
      res_name = '';
  
      select res_code, res_name
      from hp_operdel_operkind(null, :oper_type, :oper_subtype)
      into :res_code, :res_name;
  
      /*
      when 1 then 'disb'  X
      when 2 then 'repay'
      when 3 then 'writeoff'
      when 4 then 'reschedule'  X
      when 9 then 'transfer'
      when 10 then 'overpayback'
      when 11 then 'overpayoff'
      when 12 then 'floatperc'
      when 13 then 'reverse'  X
      */
  
      if (res_code in ('disb', 'reschedule', 'reverse', 'overpayback', 'overpayoff')) then
      begin
        res_text = 'Обработка невозможна из-за операции:' || ascii_char(res_name);
        break;
      end
    end
  end

  if (res_text = '') then
  begin
    select count(*)
    from operdellist
    where odl_from_odm_id = :in_odm_id and
         odl_is_checked = 0
    into ch_count;

    if (ch_count <> 0) then
      res_text = 'Имеются не проверенные строки, количество: ' || ch_count;
  end

  suspend;
end
