CREATE PROCEDURE ELCARD_OPERBOOK_CREATE_BY_LIST
AS
declare variable esm_id d_id;
declare variable base_esm_id d_id;
begin
   -- процедура предназначется для запуска роботом fsrobot_sh_main

   for select em.esm_id
   from extsysmain em
   where em.esm_project = 'elcard_disb'
   and em.esm_action = 'OperbookCreate'
   and em.esm_status = 0
   into :esm_id
   do
   begin
        -- строю проводку по комисии за выдачу чз Элькарт
        execute procedure elcard_operbook_create(:esm_id);

        update extsysmain em
        set em.esm_status = 1
        where em.esm_id = :esm_id;
   end
end
