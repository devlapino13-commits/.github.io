CREATE PROCEDURE UXP$UXREPORT_IMPORT_FRX(
    IMPORT_KIND D_SMALLINT,
    IMPORT_ID D_INTEGER,
    JSON_DATA D_BLOB_TEXT = null)
RETURNS (
    REP_UX_UUID D_UUID,
    RES_UUID_STR D_STR_CODE,
    RES_DATETIME D_DATETIME)
AS
declare variable rl_name d_str_100;
declare variable ufm_uuid d_uuid;
declare variable ux_owner d_uuid;
declare variable uxa_uuid d_uuid;
declare variable preblank_uuid d_uuid;
declare variable ua_uuid d_uuid;
begin
  if (import_kind = 1) then
  begin
    select rl_name_rus, rl_datetime
    from reportlist
    where rl_id = :import_id
    into rl_name, res_datetime;
  end else
  begin
    select bln_name, blb_datetime
    from blankblob, blankname
    where bln_id = :import_id and
          blb_from_bln_id = bln_id
    order by blb_date_begin desc
    rows 1
    into rl_name, res_datetime;
  end

  select ufm_ux_uuid
  from ux$report, ux$form
  where urt_import_kind = :import_kind and
        urt_import_id = :import_id and
        urt_uuid = ufm_uuid
  into rep_ux_uuid;

  if (rep_ux_uuid is null) then
  begin
    if (import_kind = 1) then
    begin
      execute procedure uxp$uxsource_create('reportlist', 'Отчеты', 1, 0, '', 1);
  
      ux_owner = uxf$uniq('reportlist');
  
      select ux_uuid, ufm_uuid
      from uxp$uxcode_insert(:ux_owner, 'uxreport', :import_id, :rl_name)
      into :rep_ux_uuid, :ufm_uuid;
  
      insert into ux$report (urt_uuid, urt_kind)
      values (:ufm_uuid, 2);
    end else
    begin
      execute procedure uxp$uxsource_create('blankname', 'Бланки', 1, 0, '', 1);
  
      ux_owner = uxf$uniq('blankname');
  
      select ux_uuid, ufm_uuid
      from uxp$uxcode_insert(:ux_owner, 'uxreport', :import_id, :rl_name)
      into :rep_ux_uuid, :ufm_uuid;
  
      insert into ux$report (urt_uuid, urt_kind)
      values (:ufm_uuid, 3);

      select uxa_uuid
      from ux$alias, ux$form
      where ufm_uuid = :ufm_uuid and
            ufm_ux_uuid = uxa_owner
      into uxa_uuid;
    end
  end else
  begin
    select ufm_uuid
    from ux$form
    where ufm_ux_uuid = :rep_ux_uuid
    into ufm_uuid;

    update ux$report
    set urt_kind = iif(:import_kind = 1, 2, 3)
    where urt_uuid = :ufm_uuid and
          urt_kind <> iif(:import_kind = 1, 2, 3);
  end

  execute procedure uxp$uxreport_frx_update(:ufm_uuid, null, :json_data);

  if (uxa_uuid is not null) then
  begin
    -- Импортирован бланк, настраиваем параметры

    select ux_draft_uuid
    from ux$code
    where ux_fcode = 'fm_ux$preblanks'
    into preblank_uuid;

    select ua_uuid
    from ux$action
    where ua_owner = :preblank_uuid and
          ua_uxa_uuid = :uxa_uuid and
          ua_usr_id_field = ''
    into ua_uuid;

    if (ua_uuid is not null) then
      execute procedure uxp$uxactionparam_auto_fill(:ua_uuid);
  end

  res_uuid_str = uuid_to_char(ufm_uuid);

  suspend;
end
