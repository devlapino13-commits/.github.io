CREATE PROCEDURE AAREP_AGENT_AKT_MAIN(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    AG_AGENT_ID DECIMAL(18,0),
    AG_AGENT_FIO VARCHAR(100),
    AG_PASS_NO VARCHAR(30),
    AG_PASS_DATE DATE,
    AG_PASS_WHO VARCHAR(255),
    AG_DISB_KOLICH INTEGER,
    AG_DISB_ZP DECIMAL(18,4),
    AG_DISB_CALC DECIMAL(18,4),
    AG_AGENT_ADDRESS VARCHAR(255),
    AG_DISB_PDNAL DECIMAL(18,4),
    AG_DISB_SF DECIMAL(18,4),
    S_KOEF DOUBLE PRECISION,
    AG_DISB_ALL_CUMULATIVE INTEGER)
AS
declare variable ag_disb_kolich_temp integer;
declare variable ag_disb_all integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  ag_disb_all = 0;
  select scv_value
  from sa_consvalue
  where scv_date_begin <= :report_date_end and
        scv_from_scn_id = 3
  order by scv_date_begin desc
  rows 1
  into :s_koef;
  if (s_koef is null) then
    s_koef = 8;
  for select kc_agent_id, count(*), c_name_fio, c_pass_no, c_pass_date, c_pass_who, adl_address_str
      from kontstate, knclient, clients, operlist, region_list(:report_region_id, 1, -1, 1), addrlist, addrclient
      where kontstate.ks_ol_is_delete = 0 and
            ks_date_begin between :report_date_begin and :report_date_end and
            ks_state = 1 and
            ks_from_k_id = kc_from_k_id and
            kc_agent_id is not null and
            kc_agent_id = c_id and
            kc_amount_approved is not null and
            kc_amount_approved <> 0 and
            ks_from_ol_id = ol_id and
            (ol_from_reg_id = id_list) and
            adc_from_adl_id = adl_id and
            adc_address_type = 1 and
            adc_from_c_id = c_id
      group by kc_agent_id, c_name_fio, c_pass_no, c_pass_date, c_pass_who, adl_address_str
      into :ag_agent_id, :ag_disb_kolich, :ag_agent_fio, :ag_pass_no, :ag_pass_date, :ag_pass_who, :ag_agent_address
  do
  begin

    select count(*)
    from kontstate, knclient, operlist, region_list(:report_region_id, 1, -1, 1)
    where kontstate.ks_ol_is_delete = 0 and
          cast(ks_date_begin as date) <= :report_date_end and
          ks_state = 1 and
          ks_from_k_id = kc_from_k_id and
          kc_agent_id = :ag_agent_id and
          kc_amount_approved is not null and
          kc_amount_approved <> 0 and
          ks_from_ol_id = ol_id and
          (ol_from_reg_id = id_list)
    into :ag_disb_all;
    if (ag_disb_all is null) then
      ag_disb_all = 0;

    ag_disb_all_cumulative = ag_disb_all;

    ag_disb_zp = 0;
    ag_disb_calc = 0;
    ag_disb_kolich_temp = :ag_disb_kolich;
    if ((ag_disb_all > 200) and
        (ag_disb_kolich_temp > 0)) then
    begin
      if ((ag_disb_all - 200) > ag_disb_kolich_temp) then
      begin
        ag_disb_zp = ag_disb_zp + ag_disb_kolich_temp * 60;
        ag_disb_all = ag_disb_all - ag_disb_kolich_temp;
        ag_disb_kolich_temp = 0;
      end
      else
      begin
        ag_disb_zp = ag_disb_zp + (ag_disb_all - 200) * 60;
        ag_disb_kolich_temp = ag_disb_kolich_temp - (ag_disb_all - 200);
        ag_disb_all = 200;
      end
    end

    if ((ag_disb_all > 100) and
        (ag_disb_kolich_temp > 0)) then
    begin
      if ((ag_disb_all - 100) > ag_disb_kolich_temp) then
      begin
        ag_disb_zp = ag_disb_zp + ag_disb_kolich_temp * 40;
        ag_disb_all = ag_disb_all - ag_disb_kolich_temp;
        ag_disb_kolich_temp = 0;
      end
      else
      begin
        ag_disb_zp = ag_disb_zp + (ag_disb_all - 100) * 40;
        ag_disb_kolich_temp = ag_disb_kolich_temp - (ag_disb_all - 100);
        ag_disb_all = 100;
      end
    end

    if ((ag_disb_all > 0) and
        (ag_disb_kolich_temp > 0)) then
    begin
      if ((ag_disb_all) > ag_disb_kolich_temp) then
      begin
        ag_disb_zp = ag_disb_zp + ag_disb_kolich_temp * 30;
        ag_disb_all = ag_disb_all - ag_disb_kolich_temp;
        ag_disb_kolich_temp = 0;
      end
      else
      begin
        ag_disb_zp = ag_disb_zp + ag_disb_all * 30;
        ag_disb_kolich_temp = ag_disb_kolich_temp - ag_disb_all;
        ag_disb_all = 0;
      end
    end

    /*
    if ((AG_DISB_KOLICH_temp<=100) and (AG_DISB_KOLICH_temp>0)) then
    begin
      ag_disb_zp=ag_disb_zp+AG_DISB_KOLICH_temp*30;
    end else
    begin
      ag_disb_zp=ag_disb_zp+30*100;
      AG_DISB_KOLICH_temp=:AG_DISB_KOLICH_temp-100;
      if ((AG_DISB_KOLICH_temp<=100) and (AG_DISB_KOLICH_temp>0)) then
      begin
        ag_disb_zp=ag_disb_zp+AG_DISB_KOLICH_temp*40;
      end else
      begin
        ag_disb_zp=ag_disb_zp+40*100;
        AG_DISB_KOLICH_temp=:AG_DISB_KOLICH_temp-100;
        if (AG_DISB_KOLICH_temp>0) then
        begin
          ag_disb_zp=ag_disb_zp+AG_DISB_KOLICH_temp*60;
        end
      end
    end
    */
    ag_disb_calc = cast(:ag_disb_zp / (100 - 10 - :s_koef) * 100 as decimal(18,2));
    ag_disb_sf = cast(:ag_disb_calc * (s_koef / 100) as decimal(18,2));
    ag_disb_pdnal = :ag_disb_calc - :ag_disb_sf - :ag_disb_zp;
    suspend;
  end
end
