CREATE PROCEDURE HP_UXFRPARAM_RENAME(
    UFP_CODE D_STR_100,
    NEW_ORIGIN_NAME D_STR_CODE)
AS
declare variable fr_code d_str_code;
declare variable old_origin_name d_str_code;
begin
  -- Переименовывает поле фрейма
  select ufp_ufr_code, ufp_origin_name
  from uxfrparam
  where ufp_code = :ufp_code
  into fr_code, old_origin_name;

  if (old_origin_name = new_origin_name) then
    exit;

  if (exists(select * from uxfrparam
  where ufp_ufr_code = :fr_code and
        ufp_origin_name = :new_origin_name and
        ufp_state = 0)) then
    exception e_message f_format('Поле % уже существует.', upper(new_origin_name));

  delete from uxfrparam
  where ufp_ufr_code = :fr_code and
        ufp_origin_name = :new_origin_name and
        ufp_state <> 0;

  update uxfrparam
  set ufp_origin_name = :new_origin_name,
      ufp_state = 0
  where ufp_code = :ufp_code;
end
