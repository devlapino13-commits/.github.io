CREATE PROCEDURE FSROBOT_REQUEST_FOR_TRANSFER(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
begin
  -- Процедура вызывается из FSROBOT_ON_MAILING
  -- Процедура готовит список уведомлений для отправки роботом по почте или sms
  -- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
  -- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
  -- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

  -- RefData.RD_Memo    - Список получателей
  -- RefData.RD_Tag     - Срок жизни сообщения
  -- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

  -- Rep_Params - входящий параметры в зависимости от команды
  -- Rep_UN_ID - пользователь

  -- Request for Money Transfer - Запрос на перевод Денежных Средств
  select coalesce(r1.reg_email_manager || '; ', '') || coalesce(r1.reg_email_operator || '; ', '') || --
         coalesce(r2.reg_email_manager || '; ', '') || coalesce(r2.reg_email_operator || '; ', '') || --
         iif(1090100 in (rl_corr_office, rm_from_reg_id), 'a.turdukulova@molbulak.kg; ', '')

  from reqmain rm
  inner join region r1 on r1.reg_unicode = rm.rm_from_reg_id
  inner join reqlist rl on rm.rm_id = rl.rl_from_rm_id
  inner join region r2 on r2.reg_unicode = rl.rl_corr_office
  where rm.rm_id = :rep_params and
        rm.rm_from_rk_id = 0
  into res_maillist;

  res_maillist = coalesce(res_maillist, '') || coalesce(:rep_email, '');

  res_c_id = 0;
  user_fio = 'Finstart';

  res_subject = 'Внимание ! По БД проставлен запрос на перевод Денежных Средств !';
  -- RFCKG-4430 Было изменено ФИО в рассылке -->
  -- RFCKG-2456 Было изменено ФИО в рассылке -->
  res_body = '<b>Добрый день, уважаемые коллеги!</b><br><br>
  По БД проставлен запрос на <b>перевод Денежных Средств</b><br>
  Пожалуйста, согласуйте его.<br>
  ДС нужно перевести на счета которые указаны в запросе, если банк не такой как у вас перевод делаем <font color="blue">«ГРОССОМ»</font><br>
  Если будут вопросы, можете обратиться к <b>Жуманазарова Ширин<br><br>
  Контактные данные:</b><br>
  E-mail: sh.zhumanazarova@mbulak.kg <br>тел: +996 (772) 73-12-70<br><br>
  <font color="blue"><b>Примечание:</b><br>
  1.    (ИПФ средства переводите только с ИПФ счёта на ИПФ)<br>
  2.    (Классические средства только с Классики счёта на Классику)<br>
  3.    (Запросы ДС принимаются до 17-00 на завтра на почту)</font><br><br>
  <font color="red">P.S. Ожидать письменного ответа подтверждения не нужно, так как запросы все обрабатываются и вносятся в список на распределение.</font>';

  suspend;
end
