CREATE PROCEDURE AAREP_EQUIFAX_CRED_GUARANTEE(
    JSA D_INTEGER)
RETURNS (
    UID D_STR_255,
    GUARANTEE_SUM D_CURRENCY,
    GUARANTEE_DATE D_DATE,
    END_DATE D_DATE,
    FACT_END_DATE D_DATE,
    END_REASON_STR D_STR_255)
AS
declare variable i d_integer;
  declare variable cnt d_integer;
  declare variable jso d_integer;

  declare function getcatval(kind d_str_code, code d_str_code)
  returns d_str_255
  as
  begin
    return coalesce((select bct_name from bki_catalog where bct_kind = :kind and bct_code = :code), 'Неопределен');
  end

  declare function my_strtonum (
    num_str d_str_code)
  returns d_currency
  as
  begin
    return replace(num_str, ',', '.');
    when any do
      return 0;
  end

  declare function my_strtodate (
    date_str d_str_code)
  returns d_date
  as
  begin
    return date_str;
    when any do
      return null;
  end

begin
  if (coalesce(jsa, 0) = 0) then
    exit;

  i = 0;
  cnt = json.count_get(jsa);

  while (i < cnt) do
  begin
    jso = coalesce(json.arr_get_obj(jsa, i), json.obj_create());
    i = i + 1;

    uid = json.obj_get_val(jso, 'uid');
    guarantee_sum = my_strtonum(json.obj_get_val(jso, 'sum'));
    guarantee_date = my_strtodate(json.obj_get_val(jso, 'date'));
    end_date = my_strtodate(json.obj_get_val(jso, 'endDate'));
    fact_end_date = my_strtodate(json.obj_get_val(jso, 'factEndDate'));
    end_reason_str = getcatval('guarantee_endReason', json.obj_get_val(jso, 'endReason'));

    suspend;
  end
end
