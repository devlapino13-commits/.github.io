Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXUNIT_ADD(
    UNT_CODE D_STR_CODE,
    UNT_NAME D_STR_100,
    UNT_DFM_NAME D_STR_CODE,
    UNT_DFM_JSON D_BLOB_TEXT,
    UNT_DFM_HASH D_STR_CODE,
    UNT_PAS_NAME D_STR_CODE,
    UNT_PAS_CODE D_BLOB_TEXT,
    UNT_PAS_HASH D_STR_CODE)
AS
declare variable unt_uuid d_uuid;
begin
  update or insert into ux$unit (unt_code, unt_name,
                                 unt_dfm_name, unt_dfm_json, unt_dfm_hash,
                                 unt_pas_name, unt_pas_code, unt_pas_hash)
  values (:unt_code, :unt_name,
          :unt_dfm_name, :unt_dfm_json, :unt_dfm_hash,
          :unt_pas_name, :unt_pas_code, :unt_pas_hash)
  matching (unt_code)
  returning unt_uuid
  into unt_uuid;

  delete from ux$unititem
  where uni_uuid = :unt_uuid;

  delete from ux$unitsource
  where usr_unit_code = :unt_code;

  delete from ux$unitdata
  where uds_unit_code = :unt_code;
end
