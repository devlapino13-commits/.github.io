CREATE PROCEDURE AAREP_AUTO_OWNERNAME(
    K_ID ID_TYPE)
RETURNS (
    OWNER_NAME STR_255)
AS
declare variable k_disb_date date;
declare variable c_id id_type;
begin
  k_disb_date = (select k_disb_date from kontract where k_id = :k_id);
  for
    select z_from_c_id
    from zalog where z_from_k_id = :k_id
    union
    select kc_from_c_id
    from knclient where kc_from_k_id = :k_id
  into
    :c_id
  do
  begin
    select
      f_docrecog_value(cds_from_drl_id, 'calc_name_full')
    from clientdocs
    where cds_from_c_id = :c_id
      and cds_from_pk_id = 310
      and cds_is_delete = 0
      and cds_from_drl_id > 0
      and datediff(day, :k_disb_date, cds_create_datetime) <= 0
    order by cds_create_datetime desc rows 1
    into :owner_name;
    if (owner_name<>'') then suspend;
  end
end
