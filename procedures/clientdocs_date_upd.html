CREATE PROCEDURE CLIENTDOCS_DATE_UPD(
    CDS_ID D_ID,
    DT_BEG D_DATE,
    DT_END D_DATE,
    UN_ID D_ID)
AS
declare variable drl_id d_id;
begin
  drl_id = (
    select cds_from_drl_id
    from clientdocs where cds_id = :cds_id);
  update clientdocs set
    cds_doc_date_begin = :dt_beg,
    cds_doc_date_end = :dt_end,
    cds_from_un_id = :un_id
  where cds_id = :cds_id;

  if (drl_id > 0) then
  begin
    update docrecogvalue set
      drv_final_value = f_datetostr(:dt_beg)
    where drv_id = (
      select first 1 drv_id
      from docrecogvalue, docrecogfield
      where drv_from_drl_id = :drl_id and
        drv_from_drf_id = drf_id and
        drf_inactive = 0 and
        drf_code = 'date_begin');
    update docrecogvalue set
      drv_final_value = f_datetostr(:dt_end)
    where drv_id = (
      select first 1 drv_id
      from docrecogvalue, docrecogfield
      where drv_from_drl_id = :drl_id and
        drv_from_drf_id = drf_id and
        drf_inactive = 0 and
        drf_code = 'date_end');
  end
end
