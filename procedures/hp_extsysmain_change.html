CREATE PROCEDURE HP_EXTSYSMAIN_CHANGE(
    ESM_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    K_ID DECIMAL(18,0),
    NEW_ESM_DATE D_DATE = null)
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable ol_id decimal(18,0);
declare variable esm_date date;
declare variable is_once boolean;
declare variable is_restruct boolean;
declare variable restruct_ol_id d_id;
begin
  res_text = '';
  select eso_from_ol_id, ol_date
  from operlist, extsysoper
  where eso_from_ol_id = ol_id and
        eso_from_esm_id = :esm_id and
        operlist.is_delete = 0
  into :ol_id, :esm_date;

  if (ol_id is null) then
  begin
    res_text = 'Данная транзакция еще не обработана либо аннулирована.';
    suspend;
    exit;
  end

  if (coalesce(k_id, 0) = 0) then
    k_id = null;

  with t1 as (
      select distinct ok_from_ol_id, ok_type
      from operkred
      where ok_from_k_id = :k_id
        and ok_ol_is_active = 1
        and ok_ol_date >= :esm_date)
  select count(ok_from_ol_id) = 1,                             -- операция единственная
      count(ok_from_ol_id) = 1 and list(ok_type) = 3,          -- это реструктуризация
      iif(count(ok_from_ol_id) = 1, list(ok_from_ol_id), null) -- id операции, если операция единственная
  from t1
  into is_once, is_restruct, restruct_ol_id;

  -- если есть только одна более поздняя операция и это реструктуризация, то удалим ее.
  if (is_once
    and is_restruct
    and (select count(distinct ok_from_k_id) = 1 from operkred where ok_from_ol_id = :restruct_ol_id) -- индивидуальная операция
    and (exists(select kd_id
                from kontdata
                where kd_from_ol_id = :restruct_ol_id
                  and kd_float_perc_level > 0)))                                                      -- операция повышения %% ставки
  then
  begin
    update operlist ol
    set ol.is_delete = 1
    where ol_id = :restruct_ol_id;

    when any do
    begin
      res_text = 'Не удалось удалить операцию повышения %% ставки по новому договору. Обратитесь к тех.поддержке.';
      suspend;
      exit;
    end 
  end

  if (exists(select *
             from operkred
             where ok_ol_date >= :esm_date and
                   ok_from_ol_id > :ol_id and
                   ok_from_k_id = :k_id and
                   ok_ol_is_active = 1)) then
  begin
    res_text = 'По новому договору существуют более поздние операции. Обратитесь к тех.поддержке.';
    suspend;
    exit;
  end

  if (exists(select *
             from operkred ok
             where ok_ol_date = :esm_date and
                   ok_from_k_id = :k_id and
                   ok.ok_type = 3 and
                   ok_ol_is_active = 1)) then
  begin
    res_text = 'По новому договору существует операция реструктуризации этой датой. Обратитесь к тех.поддержке.';
    suspend;
    exit;
  end

  update operlist
  set is_delete = 1
  where ol_id = :ol_id;

  -- если указана new_esm_date, то будет меняться только дата
  if (new_esm_date is not null) then
  begin
      k_id = (select esm_from_k_id from extsysmain esm where esm_id = :esm_id);

      update extsysmain
      set esm_date = : new_esm_date,
          esm_from_k_id = :k_id
      where esm_id = :esm_id;
  end 
  else
    if (coalesce(c_id, 0) <> 0) then
    begin
      update extsysmain
      set esm_object_id = :c_id,
          esm_from_k_id = :k_id
      where esm_id = :esm_id;
    end

  update extsysmain
  set esm_status = 0
  where esm_id = :esm_id;

  select err_text
  from hp_extsys_process_one(:esm_id)
  into res_text;

  suspend;
end
