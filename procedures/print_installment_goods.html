CREATE PROCEDURE PRINT_INSTALLMENT_GOODS(
    OBJ_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    CHECK_NO D_STR_20,
    OPERATION_ID D_STR_20,
    KS_DATE_BEGIN D_DATE,
    KAGENT_NAME D_STR_100,
    ADDRESS_STR D_STR_255,
    QR_KAGENT D_STR_LONG,
    CLIENT_NAME D_STR_100,
    PASS_NO VARCHAR(30),
    PASS_WHO D_STR_255,
    PASS_DATE D_DATE,
    QR_CLIENT D_STR_LONG,
    FEG_NAME D_STR_255,
    MP_AMOUNT D_CURRENCY,
    C_TYPE D_SMALLINT,
    CB_BEN_NAME D_STR_100,
    RD_NAME D_STR_255,
    SELLER_DOC_NAME VARCHAR(30),
    SELLER_DOC_DATE_BEGIN D_DATE,
    SELLER_DOC_NO D_STR_100)
AS
declare variable cs_id d_id;
declare variable mcl_id d_id;
declare variable feg_id d_id;
declare variable kagent_c_id d_id;
declare variable client_c_id d_id;
declare variable edl_id d_id;
begin
  select mp.mp_receipt_number, cast(ks.ks_from_ol_id as d_str_20), ks.ks_date_begin, mp.mp_from_cs_id, mp.mp_from_mcl_id,
         mp.mp_from_feg_id, mp.mp_amount, rg.reg_name
  from kontract kt
  inner join mc_purchase mp on mp.mp_from_ma_id = kt.k_from_ma_app_id
  inner join kontstate ks on ks.ks_from_k_id = kt.k_id
  inner join region rg on rg.reg_unicode = ks.ks_region_id
  where mp.mp_id = :obj_id and
        ks.ks_state = 1 and
        ks.ks_ol_is_delete = 0
  into check_no, operation_id, ks_date_begin, cs_id, mcl_id, feg_id, mp_amount, address_str;

  select cl.c_name_fio, cl.c_type, cl.c_id
  from clientshops cs
  inner join clients cl on cl.c_id = cs.cs_from_c_id
  where cs.cs_id = :cs_id and
        cs.cs_is_delete = 0
  into kagent_name, c_type, kagent_c_id;

  select cds.cds_doc_no, f_datetostr(cds.cds_doc_date_begin), p.pk_name
  from clientdocs cds
  inner join passkind p on p.pk_id = cds.cds_from_pk_id
  inner join passdoctype pdt on pdt.pdt_id = p.pk_from_pdt_id
  where cds.cds_from_c_id = :kagent_c_id and
        cds.cds_is_main_doc = 0 and
        :ks_date_begin between cds_doc_date_begin and cds_doc_date_end and
        cds.cds_is_delete = 0 and
        iif(:c_type = 0, pdt_code = 'certificate_ip', pdt_code = 'certificate_yur')
  rows 1
  into seller_doc_no, seller_doc_date_begin, seller_doc_name;

  if (c_type = 1) then
  begin
    select t.cb_ben_name, t.rd_name
    from hp_client_benefic(:kagent_c_id) t
    into cb_ben_name, rd_name;
  end 

  select cl.c_name_fio, cl.c_pass_no, cl.c_pass_who, cl.c_pass_date, cl.c_id
  from mc_login mcl
  inner join clients cl on cl.c_id = mcl.mcl_from_c_id
  where mcl.mcl_id = :mcl_id and
        mcl.mcl_is_delete = 0
  into client_name, pass_no, pass_who, pass_date, client_c_id;

  select feg.feg_name
  from foreign_econom_goods_ref feg
  where feg.feg_id = :feg_id
  into feg_name;

  cb_ben_name = coalesce(cb_ben_name, '');
  rd_name = coalesce(rd_name, '');
  seller_doc_name = coalesce(seller_doc_name, '');
  seller_doc_date_begin = coalesce(seller_doc_date_begin, current_date);
  seller_doc_no = coalesce(seller_doc_no, '');

  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(:add_params, 'edl_id', null);
    qr_client = f_edoc_qr_by_edl(:edl_id, :client_c_id);
    qr_kagent = f_edoc_qr_by_edl(:edl_id, :kagent_c_id);
  end 

  suspend;
end
