CREATE PROCEDURE AAREP_REPORT_SUPER_LO(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    ASL_OFFICE_ID D_INTEGER,
    ASL_LO_ID D_ID,
    ASL_DATE_BEGIN D_DATE,
    ASL_DATE_END D_DATE,
    ASL_OFFICE_MANAGER_ID D_ID,
    ASL_REGION_NAME D_STR_100,
    ASL_OFFICE_NAME D_STR_100,
    ASL_OFFICE_PAR30_360 D_PERCENT,
    ASL_OFFICE_MANAGER_NAME D_STR_100,
    ASL_OFFICE_MANAGER_EXPERIENCE D_SMALLINT,
    ASL_LO_NAME D_STR_100,
    ASL_SUPER_CATEGORY D_SMALLINT,
    ASL_PAR30_360 D_PERCENT,
    ASL_EXPERIENCE D_SMALLINT,
    ASL_EARNINGS D_INTEGER,
    ASL_BALANCE D_CURRENCY,
    ASL_DELIQ30 D_CURRENCY,
    ASL_DELIQ360 D_CURRENCY,
    ASL_BALANCE_WITHOUT360 D_CURRENCY,
    ASL_DELIQ30_60 D_CURRENCY)
AS
declare variable TMP_OFFICE_ID D_INTEGER;
declare variable C_NAME_FIO D_STR_100;
declare variable C_EXPERIENCE D_SMALLINT;
declare variable AAREP_BONUS_TOTAL D_CURRENCY;
declare variable OFFICE_DELIQ_BALANCE D_CURRENCY;
declare variable OFFICE_ACTIVE_BALANCE D_CURRENCY;
declare variable REP_DATE_BEGIN D_DATE;
begin
  -- возвращает список Супер КС (RFCAZ-427)

  if (f_get_db_id() <> 94401) then
    exit;

  if (rep_region_id is null) then
    exception e_message 'Ошибка: не указан регион';

  if (rep_date_end is null) then
    exception e_message 'Ошибка: не указана дата отчета';

  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1; -- начало отчетного месяца

  delete from analitica_super_lo;

  insert into analitica_super_lo (asl_office_id, asl_lo_id, asl_date_begin, asl_date_end, asl_region_name, asl_office_name, asl_lo_name, asl_balance, asl_deliq30, asl_deliq360)
  select aa_curr_office_id, aa_curr_kc_id, :rep_date_begin, :rep_date_end, min(aarep_name_1), min(aarep_name_2), min(aarep_name_3), sum(aarep_active_balance_e), sum(aarep_deliq_balance30), sum(aarep_deliq_balance360)
  from aarep_analitica_portfolio(:rep_region_id, :rep_date_begin, :rep_date_end, -1, 0, null)
  group by aa_curr_office_id, aa_curr_kc_id
  having sum(aarep_active_balance_e) > 0;

  update analitica_super_lo
  set asl_balance_without360 = asl_balance - asl_deliq360, -- КП чистый от просрочек 360+
      asl_deliq30_60 = asl_deliq30 - asl_deliq360; -- Баланс кредитов с просрочкой 30-360

  update analitica_super_lo
  set asl_par30_360 = 0
  where asl_deliq30_60 = 0 and
        asl_balance_without360 = 0;

  update analitica_super_lo
  set asl_par30_360 = asl_deliq30_60 / asl_balance_without360 * 100 -- ПАР 30-360 КС в процентах
  where asl_balance_without360 > 0;

  for select distinct asl_office_id
      from analitica_super_lo
      into tmp_office_id
  do
  begin
    office_deliq_balance = null;
    office_active_balance = null;
    select sum(asl_deliq30_60), sum(asl_balance_without360)
    from analitica_super_lo
    where asl_office_id = :tmp_office_id
    into office_deliq_balance, office_active_balance;

    if (coalesce(office_active_balance, 0) = 0) then
      asl_office_par30_360 = 0;
    else
      asl_office_par30_360 = office_deliq_balance / office_active_balance * 100; -- ПАР 30-360 офиса в процентах

    update analitica_super_lo
    set asl_office_par30_360 = :asl_office_par30_360
    where asl_office_id = :tmp_office_id;

    asl_office_manager_id = null;
    select ps.ps_from_c_id
    from hr_pstatus ps
    inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
    inner join jobkind jk on jk.jk_id = jl.jl_kind
    where ps.ps_from_reg_id = :tmp_office_id and
          ps.ps_status = 1 and
          ps.ps_is_working = 1 and
          :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
          jk.jk_system_code = 'JobSystem.Region_KM' and
          jl.jl_name containing 'filialının meneceri'
    into asl_office_manager_id;

    c_name_fio = null;
    select c_name_fio
    from clients
    where c_id = :asl_office_manager_id
    into c_name_fio;

    c_experience = null;
    select c_stage
    from aarep_work_experience(:rep_date_end, :asl_office_manager_id)
    into c_experience;

    update analitica_super_lo
    set asl_office_manager_id = :asl_office_manager_id,
        asl_office_manager_name = :c_name_fio,
        asl_office_manager_experience = :c_experience
    where asl_office_id = :tmp_office_id;
  end

  for select distinct asl_lo_id
      from analitica_super_lo
      where asl_lo_id <> coalesce(asl_office_manager_id, 0) -- менеджера офисов не входят в список Супер КС
      into asl_lo_id
  do
  begin
    if (not exists(select *
                   from hr_pstatus
                   where ps_from_c_id = :asl_lo_id and
                         ps_status = 1 and
                         ps_is_working = 1 and
                         :rep_date_end between ps_date_begin and ps_date_end)) then
      continue; -- КС не работает (уволен) на момент обновления списка Супер КС

    if (exists(select *
               from hr_pstatus ps
               inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
               inner join jobkind jk on jk.jk_id = jl.jl_kind
               where ps.ps_from_c_id = :asl_lo_id and
                     ps.ps_status = 1 and
                     ps.ps_is_working = 1 and
                     :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
                     jk.jk_system_code = 'JobSystem.Region_KM')) then -- региональные менеджера не входят в список Супер КС
      continue;

    c_experience = null;
    select c_stage
    from aarep_work_experience(:rep_date_end, :asl_lo_id)
    into c_experience;

    update analitica_super_lo
    set asl_experience = :c_experience
    where asl_lo_id = :asl_lo_id;
  end

  -- формирование списка 2-й категории Супер КС
  for select asl_office_id, asl_lo_id
      from analitica_super_lo
      where asl_office_id not in (1050900) and -- исключенные офисы (RFCAZ-563)
            asl_office_par30_360 < 1 and -- ПАР офиса менее 1
            asl_experience >= 6 and -- стаж КС 6+ месяцев
            asl_par30_360 < 1 -- ПАР КС менее 1
      into asl_office_id, asl_lo_id
  do
  begin
    update analitica_super_lo
    set asl_super_category = 2
    where asl_office_id = :asl_office_id and
          asl_lo_id = :asl_lo_id;
  end

  for select asl_office_id, asl_lo_id
      from analitica_super_lo
      where asl_super_category is null
      into asl_office_id, asl_lo_id
  do
  begin
    aarep_bonus_total = null;
    select aarep_bonus_total
    from cb_az_lo_201602_one_lo_2_new(:asl_office_id, :asl_lo_id, :rep_date_end, 0)
    into aarep_bonus_total;

    update analitica_super_lo
    set asl_earnings = :aarep_bonus_total
    where asl_office_id = :asl_office_id and
          asl_lo_id = :asl_lo_id;
  end

  -- формирование списка 1-й категории Супер КС
  for select asl_office_id, asl_lo_id
      from analitica_super_lo
      where asl_super_category is null and -- из тех, кто не вошел во 2-ую категорию
            asl_experience >= 6 and -- стаж КС 6+ месяцев
            asl_par30_360 < 0.7 and -- ПАР КС менее 0,7
            asl_earnings > 1000 -- зароботок более 1000 RFCAZ-618
      into asl_office_id, asl_lo_id
  do
  begin
    update analitica_super_lo
    set asl_super_category = 1
    where asl_office_id = :asl_office_id and
          asl_lo_id = :asl_lo_id;
  end

  -- показ списка Супер КС
  for select asl_office_id, asl_lo_id, asl_date_begin, asl_date_end, asl_region_name, asl_office_name, asl_office_par30_360, asl_office_manager_name, asl_office_manager_id, asl_office_manager_experience, asl_lo_name, asl_super_category, asl_balance, asl_deliq30, asl_deliq360, asl_balance_without360, asl_deliq30_60, asl_par30_360, asl_experience, asl_earnings
      from analitica_super_lo
      where asl_super_category is not null
      order by asl_region_name, asl_office_name, asl_lo_name
      into asl_office_id, asl_lo_id, asl_date_begin, asl_date_end, asl_region_name, asl_office_name, asl_office_par30_360, asl_office_manager_name, asl_office_manager_id, asl_office_manager_experience, asl_lo_name, asl_super_category, asl_balance, asl_deliq30, asl_deliq360, asl_balance_without360, asl_deliq30_60, asl_par30_360, asl_experience, asl_earnings
  do
    suspend;
end
