CREATE PROCEDURE HP_CASHINVENTORY_LOAD(
    IN_CIL_ID D_INTEGER,
    IN_CASH_ID D_INTEGER,
    IN_DATE D_DATE,
    IN_OPER_KIND D_STR_100,
    IN_MANAGER_ID D_ID,
    UN_ID D_ID)
RETURNS (
    CIL_ID D_INTEGER,
    CIL_FROM_CSP_ID D_INTEGER,
    CIL_DATE D_DATE,
    CIL_OPER_KIND D_INTEGER,
    CIL_INSPECTOR_ID D_ID,
    CIL_INV_AMOUNT D_CURRENCY,
    CFL_FACT_AMOUNT D_CURRENCY,
    CFL_FACT_AMOUNT$BOND D_STR_LONG,
    DIFF_ACCOUNT D_STR_CODE,
    RES_OFFICE_ID D_INTEGER,
    AKT_PERSON_ID D_ID,
    AKT_AUDIT_ID D_ID,
    AKT_MEMBER_ID D_ID)
AS
declare variable cfl_id d_integer;
begin
  if (in_cil_id is null) then
    in_cil_id = 0;

  diff_account = '';
  akt_person_id = null;  --  CIM_ROLE_KIND = -1 как мат.ответственный
  akt_audit_id = null;   --  CIM_ROLE_KIND = 2 как аудит
  akt_member_id = null;  --  CIM_ROLE_KIND = 3 как участник

  if (in_cil_id <> 0) then
  begin
    cil_id = in_cil_id;

    select cil_from_csp_id, cil_date, cil_oper_kind, cil_inspector_id, cil_inv_amount, cil_office_id
    from cashinvlist
    where cil_id = :in_cil_id
    into :cil_from_csp_id, :cil_date, :cil_oper_kind, :cil_inspector_id, :cil_inv_amount, res_office_id;

    if (row_count = 0) then
      exception e_message 'Акт инкассации не найден, Id: ' || in_cil_id;

    select cil_id, cil_inv_amount
    from cashinvlist
    where cil_parent_id = :in_cil_id and
          cil_is_delete = 0
    into :cfl_id, :cfl_fact_amount;

    if (row_count = 0) then
    begin
      cfl_fact_amount = cil_inv_amount;
      cfl_fact_amount$bond = '';
    end else
    begin
      select res_bond_list
      from hp_cashbond_load(:cfl_id)
      into :cfl_fact_amount$bond;
    end

    select cim_from_c_id
    from cashinvmembers
    where cim_from_cil_id = :in_cil_id and
          cim_role_kind = -1
    into akt_person_id;

    select cim_from_c_id
    from cashinvmembers
    where cim_from_cil_id = :in_cil_id and
          cim_role_kind = 2
    into akt_audit_id;

    select cim_from_c_id
    from cashinvmembers
    where cim_from_cil_id = :in_cil_id and
          cim_role_kind = 3
    into akt_member_id;
  end else
  begin
    cil_id = 0;

    if (in_cash_id is null) then
      exception e_message 'Касса не указана';

    if (in_manager_id is null) then
      exception e_message 'Контролер не указан';
  
    if (in_date is null) then
      in_date = current_date;

    if (exists(select * from opercash
    where opc_from_csp_id = :in_cash_id and
          opc_date = :in_date and
          opc_is_delete = 0 and
          opc_amount <> 0 and
          opc_from_ol_id is null)) then
      exception e_message 'Имеются не проведенные кассовые операции';

    cil_oper_kind = f_enum_id('cashoperkind', in_oper_kind, 0);

    select csp_office_id, csp_cashier_id
    from cashplace
    where csp_id = :in_cash_id
    into res_office_id, akt_person_id;

    cil_from_csp_id = in_cash_id;
    cil_date = in_date;
    cil_inspector_id = in_manager_id;
    cil_inv_amount = round(f_cash_saldo(in_cash_id, in_date, 0));
    cfl_fact_amount = 0;
  end

  suspend;
end
