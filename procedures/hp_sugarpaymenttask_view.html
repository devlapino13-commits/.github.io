CREATE PROCEDURE HP_SUGARPAYMENTTASK_VIEW(
    REG_ID D_INTEGER)
RETURNS (
    SPT_ID D_INTEGER,
    SPT_K_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    SPT_C_ID D_ID,
    C_NAME_FIO D_STR_100,
    SPT_SELLER_ID D_ID,
    SELLER_NAME D_STR_100,
    SPT_REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    SPT_AMOUNT D_CURRENCY,
    SPT_EDS_FIXED D_BOOL,
    SPT_CREATED D_SMALLINT,
    SPT_ACL_ID D_ID,
    SPT_DATETIME D_DATETIME)
AS
declare variable office_id d_integer;
declare variable seller_id d_integer;
begin
  for
    select
      reg_unicode
    from region
      inner join region_list(:reg_id, 1, -1, 1)
        on id_list = reg_unicode
  into
    :office_id
  do
  begin
    for
      select
        spt_id,
        spt_k_id,
        k_kontract_no,
        k_disb_date,
        spt_c_id,
        c1.c_name_fio,
        spt_seller_id,
        c2.c_name_fio,
        spt_reg_id,
        reg_name,
        spt_amount,
        spt_eds_fixed,
        spt_created,
        spt_acl_id,
        spt_datetime
      from sugarpaymenttask
        inner join kontract
          on k_id = spt_k_id
        inner join clients as c1
          on c1.c_id = spt_c_id
        inner join clients as c2
          on c2.c_id = spt_seller_id
        inner join region
          on reg_unicode = spt_reg_id
      where spt_reg_id = :office_id
        and spt_created = -1
    into
      :spt_id,
      :spt_k_id,
      :k_kontract_no,
      :k_disb_date,
      :spt_c_id,
      :c_name_fio,
      :spt_seller_id,
      :seller_name,
      :spt_reg_id,
      :reg_name,
      :spt_amount,
      :spt_eds_fixed,
      :spt_created,
      :spt_acl_id,
      :spt_datetime
    do
    begin
      suspend;
    end
  end
end
