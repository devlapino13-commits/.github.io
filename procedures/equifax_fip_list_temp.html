              ar_house_no = adl_depart_no; -- номер дома места регистрации
              ar_build_no = adl_building_no; -- номер строения места регистрации
              ar_block_no = adl_section_no; -- номер корпуса места регистрации
              ar_apartment_no = adl_flat_no; -- номер квартиры/комнаты/офиса места регистрации

            end

            if (adc_type = 2) then -- адрес выдачи
            begin
              af_adl_is_edited = adl_is_edited;

              af_adl_id = adl_id;
              af_address_str = adl_address_str;

              if (adc_is_rent = 0) then
                af_owner = 3; -- аренда
              else
                af_owner = 9; -- не известно

              if (republ_code = '417') then
                af_country = 'kg';
              else
              if (republ_code = '810') then
                af_country = 'ru';
              else
              if (republ_code = '860') then
                af_country = 'uz';
              else
              if (republ_code = '972') then
                af_country = 'tj';
              else
                af_country = '99';

              pl_fs_code = null;
              af_city = null;
              plt_name = null;

              pl_post_index = '';
              select substring(pl_code from 4 for 2), pl.pl_name, pl_post_index
              from places pl
              inner join placetype pt on pl.pl_status = pt.plt_id
              where pl.pl_id = :adl_from_pl_id
              into :pl_fs_code, :af_city, :pl_post_index;
              if (coalesce(pl_post_index, '') = '') then
                pl_post_index = '0';

              af_region = '00';
              if (republ_code = '810') then
              begin
                select rcr.rcr_cib_code
                from rcib_region rcr
                where rcr.rcr_fs_code = :pl_fs_code
                into :af_region; -- код региона рф по месту регистрации
              end

              street_post_index = '';
              street_buf = null;
              plt_name = null;
              select pl.pl_name, pt.plt_name, pl.pl_post_index
              from places pl
              inner join placetype pt on pl.pl_status = pt.plt_id
              where pl.pl_id = :adl_street_id
              into :street_buf, :plt_name, :street_post_index;
              if (coalesce(street_post_index, '') = '') then
                street_post_index = '0';

              adl_post_index = coalesce(adl_post_index, 0);
              if (street_post_index <> '0') then
                adl_post_index = cast(street_post_index as integer);
              else
              if (pl_post_index <> '0') then
                adl_post_index = cast(pl_post_index as integer);
              af_index = adl_post_index; -- почтовый индекс места жительства/нахождения

              if (plt_name = 'улица') then
                af_street = street_buf; -- наименование улицы
              if (plt_name in ('район', 'микрорайон')) then
                af_district = street_buf; -- наименование района

              af_house_no = adl_depart_no; -- номер дома места регистрации
              af_build_no = adl_building_no; -- номер строения места регистрации
              af_block_no = adl_section_no; -- номер корпуса места регистрации
              af_apartment_no = adl_flat_no; -- номер квартиры/комнаты/офиса места регистрации
            end

          end

          middlename = coalesce(middlename, '');

          --117335
          if (char_length(ar_index) > 6) then
            ar_index_cib = substring(ar_index from 1 for 6);
          else
            ar_index_cib = ar_index;
          if (coalesce(ar_index_cib, '0') = '0') then
            ar_index_cib = '117335'; --'000000';

          if (char_length(af_index) > 6) then
            af_index_cib = substring(af_index from 1 for 6);
          else
            af_index_cib = af_index;
          if (coalesce(af_index_cib, '0') = '0') then
            af_index_cib = '117335'; --'000000';

          ar_house_no_cib = ar_house_no;
          if ((coalesce(ar_house_no_cib, '') = '') or (ar_house_no_cib = '0')) then
            ar_house_no_cib = 'нет данных';

          af_house_no_cib = af_house_no;
          if ((coalesce(af_house_no_cib, '') = '') or (af_house_no_cib = '0')) then
            af_house_no_cib = 'нет данных';

          ar_apartment_no_cib = ar_apartment_no;
          if ((coalesce(ar_apartment_no_cib, '') = '') or (ar_apartment_no_cib = '0')) then
            ar_apartment_no_cib = 'нет данных';

          af_apartment_no_cib = af_apartment_no;
          if ((coalesce(af_apartment_no_cib, '') = '') or (af_apartment_no_cib = '0')) then
            af_apartment_no_cib = 'нет данных';

          if (coalesce(ar_owner, 0) = 0) then
            ar_owner = 9;
          if (coalesce(af_owner, 0) = 0) then
            af_owner = 9;
          if (coalesce(ar_country, '') = '') then
            ar_country = '99';
          if (coalesce(af_country, '') = '') then
            af_country = '99';
          if (coalesce(ar_region, '') = '') then
            ar_region = '00';
          if (coalesce(af_region, '') = '') then
            af_region = '00';
          if (coalesce(ar_city, '') = '') then
            ar_city = 'нет данных';
          if (coalesce(af_city, '') = '') then
            af_city = 'нет данных';
          if (coalesce(ar_street, '') = '') then
            ar_street = '';
          if (coalesce(af_street, '') = '') then
            af_street = '';

          snils = null;

          select cds_doc_no
          from clientdocs, passkind, passdoctype
          where cds_from_c_id = :c_id and
                cds_from_pk_id = pk_id and
                pk_from_pdt_id = pdt_id and
                pdt_code = 'snils' and
                cds_is_delete = 0
          order by cds_from_c_id desc, cds_doc_date_begin desc
          rows 1
          into :snils;

          snils = coalesce(snils, '');
          snils = replace(snils, '-', '');
          snils = replace(snils, ' ', '');
          if (snils not similar to '[[:DIGIT:]]{11}') then  --050421.Кравцов. СНИЛС должен быть 11 цифр. Если это не так, то не передаем его вообще. (необязателен)
            snils = '';
        end
      end

      suspend;
    end
  end
