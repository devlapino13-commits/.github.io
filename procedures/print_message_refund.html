CREATE PROCEDURE PRINT_MESSAGE_REFUND(
    FR_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO VARCHAR(50),
    K_DISB_DATE D_DATE,
    FR_AMOUNT D_CURRENCY,
    FR_CELL_NO D_STR_20)
AS
declare variable DB_ID D_ID;
begin
  :db_id = f_get_db_id();
  if (db_id in (39801, 41701)) then
  begin
    for select cl.c_name_fio, kt.k_kontract_no, kt.k_disb_date, fr.fr_amount, fr.fr_cell_no
        from finerefund fr
        inner join kontract kt on kt.k_id = fr.fr_from_k_id
        inner join clients cl on cl.c_id = fr.fr_from_c_id
        where fr.fr_id = :fr_id and
              fr.fr_status = 1
        into :c_name_fio, :k_kontract_no, :k_disb_date, :fr_amount, :fr_cell_no
    do
    begin
      suspend;
    end
  end
end
