Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_OPERREPEAT_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    ORP_ID D_INTEGER,
    ORP_FROM_K_ID D_ID,
    ORP_MEMO D_STR_LONG,
    ORP_OL_DATE D_DATE,
    ORP_BEGIN_DATE D_DATE,
    ORP_BOOK_REVERSE D_SMALLINT,
    ORP_LOAN_REVERSE D_SMALLINT,
    ORP_STATE D_SMALLINT,
    ORP_FROM_OL_ID D_STR_CODE,
    K_KONTRACT_NO D_STR_CODE,
    ORP_BOOK_REVERSE_STR D_STR_CODE,
    ORP_LOAN_REVERSE_STR D_STR_CODE,
    ORP_STATE_STR D_STR_CODE)
AS
begin
  for select orp_id, orp_from_k_id, orp_memo, orp_ol_date, orp_begin_date, orp_book_reverse, orp_loan_reverse,
             orp_state, orp_from_ol_id, k_kontract_no
      from operrepeat, kontract
      where orp_ol_date between :rep_date_begin and :rep_date_end and
            orp_from_k_id = k_id
      into :orp_id, :orp_from_k_id, :orp_memo, :orp_ol_date, :orp_begin_date, :orp_book_reverse, :orp_loan_reverse,
           :orp_state, :orp_from_ol_id, :k_kontract_no
  do
  begin
    orp_book_reverse_str = case orp_book_reverse
                             when 0 then 'Не указан'
                             when 1 then 'Перестроить'
                             when 2 then 'Сторнировать'
                           end;

    orp_loan_reverse_str = case orp_loan_reverse
                             when 0 then 'Не указан'
                             when 1 then 'Перестроить'
                             when 2 then 'Сторнировать'
                           end;

    orp_state_str = case orp_state
                             when 0 then 'Новый запрос'
                             when 1 then 'В обработке'
                             when 2 then 'Подготовлен'
                             when 3 then 'Проверен'
                             when 4 then 'Обработан'
                             when -1 then 'Отменен'
                           end;

    suspend;
  end
end
