CREATE PROCEDURE HP_IMPORT_PLAN(
    IMP_TXT1 D_BLOB_TEXT,
    IMP_TXT2 D_BLOB_TEXT,
    DT_PLAN DATE,
    UN_ID ID_TYPE)
AS
declare variable imp_txt d_blob_text;
declare variable db_id integer;
declare variable c_id id_type;
declare variable reg_id integer;
declare variable regp_id integer;
declare variable c_name str_100;
declare variable sum_classic_all integer;
declare variable sum_classic integer;
declare variable sum_ipf_all integer;
declare variable sum_ipf integer;
declare variable rd_classic_id integer;
declare variable rd_ipf_id integer;
declare variable pp_classic_id integer;
declare variable pp_ipf_id integer;
declare variable pm_id integer;
declare variable pr_plan integer;
declare variable rec_txt str_255;
declare variable rec_len integer;
declare variable rec_posp integer;
declare variable rec_posc integer;
declare variable fld_pos1 integer;
declare variable fld_pos2 integer;
declare variable iter integer;
begin
  imp_txt = imp_txt1 || imp_txt2;

  db_id = f_get_db_id();
  rd_classic_id = f_get_rd_id('PlanIndicators.Disb_Amount');
  rd_ipf_id = f_get_rd_id('PlanIndicators.Disb_Amount_IPF');

  dt_plan = dateadd(1 - extract(day from dt_plan) day to dt_plan);
  pr_plan = 100 * extract(year from dt_plan) + extract(month from dt_plan);

-- Чистка данных
  delete from planoper
  where po_date = :dt_plan and
        po_from_pm_id in (select pm_id
                          from planmain
                          where pm_from_rd_id in (:rd_classic_id, :rd_ipf_id));
  rec_posp = 0;
  sum_classic_all = 0;
  sum_ipf_all = 0;
  while (1 = 1) do
  begin
    rec_posc = position('|', imp_txt, rec_posp + 1);
    if (rec_posc > rec_posp) then
    begin
      rec_txt = substring(imp_txt from rec_posp + 1 for rec_posc - rec_posp - 1);
      rec_len = char_length(rec_txt);
      fld_pos1 = position(';', rec_txt);
      if (fld_pos1 > 0) then
      begin
        fld_pos2 = position(';', rec_txt, fld_pos1 + 1);
        c_name = left(rec_txt, fld_pos1 - 1);

        regp_id = (select reg_unicode
                   from region
                   where reg_name = :c_name and
                         reg_date_close is null);
        if (regp_id is not null) then
        begin
          reg_id = regp_id;
          c_id = 0;
        end
        else
        begin
          if ((db_id = 81001) and
              (c_name = 'КЦ Сирена')) then
            c_id = (select first 1 c_id
                    from clients
                    where c_name_fio = :c_name);
          else
            c_id = (select first 1 c_id
                    from clients
                    where c_name_fio = :c_name and
                          exists(select *
                                 from hr_pstatus
                                 inner join hr_orders on ho_id = ps_from_ho_id and
                                       ho_status = 1
                                 where ps_from_c_id = c_id and
                                       current_date between ps_date_begin and ps_date_end and
                                       ps_is_working = 1));
        end
        if (c_id is not null) then
        begin
-- Классика
          sum_classic = iif(fld_pos2 = 0, right(rec_txt, rec_len - fld_pos1), substring(rec_txt from fld_pos1 + 1 for fld_pos2 - fld_pos1 - 1));
          if (sum_classic > 0) then
          begin
            if (c_id > 0) then
            begin
              if (pp_classic_id is null) then
              begin
                insert into planpocket (pp_date, pp_memo, pp_from_un_id)
                values (:dt_plan,
                        'План выдач на ' || f_monthname(:dt_plan) || '(классика/импорт)',
                        :un_id)
                returning pp_id
                into :pp_classic_id;
              end

              pm_id = (select pm_id
                       from planmain
                       where pm_from_rd_id = :rd_classic_id and
                             pm_from_reg_id = :reg_id and
                             pm_from_c_id = :c_id and
                             pm_is_mgr = 0);
              if (pm_id is null) then
                insert into planmain (pm_from_rd_id, pm_from_reg_id, pm_from_c_id)
                values (:rd_classic_id, :reg_id, :c_id)
                returning pm_id
                into :pm_id;
              insert into planoper (po_from_pp_id, po_from_pm_id, po_date, po_period, po_calc, po_minus)
              values (:pp_classic_id, :pm_id, :dt_plan, :pr_plan, :sum_classic, 0);
            end
            else
              sum_classic_all = sum_classic_all + sum_classic;
          end

-- ИПФ
          sum_ipf = iif(fld_pos2 = 0, 0, right(rec_txt, rec_len - fld_pos2));
          if (sum_ipf > 0 and
              db_id in (39801, 41701)) then
          begin
            if (c_id > 0) then
            begin
              if (pp_ipf_id is null) then
              begin
                insert into planpocket (pp_date, pp_memo, pp_from_un_id)
                values (:dt_plan, 'План выдач на ' || f_monthname(:dt_plan) || '(ИПФ/импорт)',
                        :un_id)
                returning pp_id
                into :pp_ipf_id;
              end

              pm_id = (select pm_id
                       from planmain
                       where pm_from_rd_id = :rd_ipf_id and
                             pm_from_reg_id = :reg_id and
                             pm_from_c_id = :c_id and
                             pm_is_mgr = 0);
              if (pm_id is null) then
                insert into planmain (pm_from_rd_id, pm_from_reg_id, pm_from_c_id)
                values (:rd_ipf_id, :reg_id, :c_id)
                returning pm_id
                into :pm_id;
              insert into planoper (po_from_pp_id, po_from_pm_id, po_date, po_period, po_calc, po_minus)
              values (:pp_ipf_id, :pm_id, :dt_plan, :pr_plan, :sum_ipf, 0);
            end
            else
              sum_ipf_all = sum_ipf_all + sum_ipf;
          end
        end
      end
    end
    else
      leave;
    rec_posp = rec_posc;
  end

-- Группировка (офисы)
  insert into planoper (po_from_pp_id, po_from_pm_id, po_date, po_period, po_calc, po_minus)
  select a.pp_id, pm_id, :dt_plan, :pr_plan, a.po_calc, 0
  from (select pm_from_reg_id as reg_id, po_from_pp_id as pp_id, sum(po_calc) as po_calc
        from planoper
        inner join planmain on pm_id = po_from_pm_id and
              pm_from_c_id > 0
        where po_from_pp_id in (:pp_classic_id, :pp_ipf_id)
        group by pm_from_reg_id, po_from_pp_id) as a
  inner join planmain on pm_from_reg_id = a.reg_id and
        pm_from_c_id = 0 and
        pm_is_mgr = 0 and
        pm_from_rd_id = iif(a.pp_id = :pp_classic_id, :rd_classic_id, :rd_ipf_id);

-- Группировка (регионы)
  reg_id = 1000000;
  iter = 0;
  while (iter < 3) do
  begin
    iter = iter + 1;
    insert into planoper (po_from_pp_id, po_from_pm_id, po_date, po_period, po_calc, po_minus)
    select a.pp_id, pm_id, :dt_plan, :pr_plan,
           iif(:iter = 3, iif(pm_from_rd_id = :rd_classic_id, :sum_classic_all, :sum_ipf_all), a.po_calc), 0
    from (select parent as reg_id, po_from_pp_id as pp_id, sum(po_calc) as po_calc
          from planoper
          inner join planmain on pm_id = po_from_pm_id and
                pm_from_c_id = 0
          inner join region on reg_unicode = pm_from_reg_id and
                reg_status = iif(:iter = 1, 3, 2) and
                iif(:iter = 3, parent = :reg_id, parent <> :reg_id)
          where po_from_pp_id in (:pp_classic_id, :pp_ipf_id)
          group by parent, po_from_pp_id) as a
    inner join planmain on pm_from_reg_id = a.reg_id and
          pm_from_c_id = 0 and
          pm_is_mgr = 0 and
          pm_from_rd_id = iif(a.pp_id = :pp_classic_id, :rd_classic_id, :rd_ipf_id);
  end
end
