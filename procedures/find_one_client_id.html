CREATE PROCEDURE FIND_ONE_CLIENT_ID(
    C_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_BIRTH_DATE DATE)
RETURNS (
    C_ID DECIMAL(18,0),
    CC_NAME_FIO VARCHAR(100),
    CC_PASS_NO VARCHAR(30),
    CC_BIRTH_DATE DATE,
    V_FOUND_FIO INTEGER,
    V_FOUND_PASS_NO INTEGER,
    V_FOUND_BIRTH_DATE INTEGER)
AS
declare variable found_kol integer;
declare variable is_critical integer;
declare variable found_fields integer;
begin
  if (c_pass_no is null) then
    c_pass_no = '';
  if (c_fio is null) then
    c_fio = '';
  c_pass_no = replace(c_pass_no, ' ', '');

-- Поиск клиента по трем параметрам во время импорта списка КИБ

-- Новый вариант поиска
  found_kol = 0;
  c_id = null;
  v_found_fio = 0;
  v_found_pass_no = 0;
  v_found_birth_date = 0;

  for select c_id, max(is_critical), max(res_same_fio) + max(res_same_birth_date) + max(res_same_pass_no),
             max(res_same_fio), max(res_same_birth_date), max(res_same_pass_no)
      from hp_find_clients_tmp(0, :c_fio, :c_birth_date, '', null, :c_pass_no, '', 0)
      group by c_id
      order by 2 desc, 3 desc
      into c_id, is_critical, found_fields, v_found_fio, v_found_birth_date, v_found_pass_no
  do
  begin
    found_kol = 1;
    if (v_found_fio <> -1) then
      v_found_fio = 1;
    if (exists(select *
               from knclient
               where (kc_from_c_id = :c_id or
                     kc_from_c_id = (select c_id
                                     from clients
                                     where c_original_id = :c_id)) and
                     kc_amount_approved > 0)) then
    begin
      select c_name_fio, c_pass_no, c_birth_date
      from clients
      where c_id = :c_id
      into cc_name_fio, :cc_pass_no, :cc_birth_date;
      suspend;
      exit;
    end
  end
  if (found_kol > 0) then
  begin
    suspend;
    exit;
  end

end
