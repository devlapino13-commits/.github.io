CREATE PROCEDURE HP_INSURANCE_TARIF_ADD(
    IT_FROM_PR_ID D_ID,
    UN_ID D_ID,
    IT_MIN_POINT D_INTEGER,
    IT_MAX_POINT D_INTEGER,
    IT_TARIF D_PERCENT,
    IT_TARIF_TYPE D_SMALLINT,
    AG_ID D_ID,
    IAC_COMMIS D_CURRENCY)
AS
declare variable it_from_prl_id d_id;
declare variable it_id d_id;
begin
    select first 1 prl.prl_id
    from prodlogic prl
    where prl.prl_from_pr_id = :it_from_pr_id
      and current_date between prl.prl_date_begin and prl.prl_date_end
    order by prl.prl_date_begin desc
    into :it_from_prl_id;

    insert into insurance_tarif (
        it_from_pr_id,
        it_from_prl_id,
        it_create_datetime,
        it_last_datetime,
        it_from_un_id,
        it_min_point,
        it_max_point,
        it_tarif,
        it_tarif_type)
    values (
        :it_from_pr_id,
        :it_from_prl_id,
        current_timestamp,
        current_timestamp,
        :un_id,
        :it_min_point,
        :it_max_point,
        :it_tarif,
        :it_tarif_type)
    returning it_id
    into :it_id;


    insert into insur_ag_commis (
        iac_from_ag_id,
        iac_from_it_id,
        iac_commis,
        iac_datetime,
        iac_from_un_id)
    values (
        :ag_id,
        :it_id,
        :iac_commis,
        current_timestamp,
        :un_id);
end
