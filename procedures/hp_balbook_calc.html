CREATE PROCEDURE HP_BALBOOK_CALC(
    CALC_TEXT D_STR_255,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_AMOUNT D_CURRENCY,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable command_str d_str_code;
declare variable is_tax d_smallint;
declare variable par1 d_str_code;
declare variable par2 d_str_code;
declare variable comma d_integer;
declare variable param_pos d_integer;
begin
  res_error_text = '';
  res_amount = 0;

  calc_text = replace(calc_text, ' ', '');

  if (calc_text is null) then
    res_error_text = 'Неправильный формат формулы';

  if (res_error_text = '' and left(calc_text, 3) <> 'fs_') then
    res_error_text = 'Неправильный формат формулы';

  param_pos =  position('[', calc_text);
  if (res_error_text = '' and param_pos = 0) then
    res_error_text = 'Неправильный формат формулы';

  command_str = left(calc_text, param_pos - 1);
  if (res_error_text = '' and command_str not in ('fs_o', 'fs_s', 'fs_ot', 'fs_st')) then
    res_error_text = 'Неправильная функция для FinStart: '||command_str;

  if (res_error_text <> '') then
  begin
    Suspend;
    exit;
  end

  if (right(calc_text, 1) = ']') then
    calc_text = replace(calc_text, ']', '');

-- fs_s(4xx01)
-- fs_o(112312,131231)

  is_tax = 0;
  if (right(command_str, 1) = 't') then
  begin
    is_tax = 1;
    command_str = left(command_str, 4);
  end

  par1 = '';
  par2 = '';
  if (command_str = 'fs_s') then
  begin
    par1 = substring(calc_text from param_pos + 1);
    res_amount = f_balbook_saldo(par1, rep_date_end, is_tax);
    when any do
      res_error_text = 'Неверные параметры';
  end else
  begin
    comma = position(',', calc_text, param_pos);
    if (comma = 0) then
      res_error_text = 'Неправильный формат формулы';

    par1 = substring(calc_text from param_pos + 1 for comma - param_pos - 1);
    par2 = substring(calc_text from comma + 1);
    res_amount = f_balbook_obor(par1, par2, rep_date_begin, rep_date_end, is_tax);
    when any do
      res_error_text = 'Неверные параметры';
  end

  Suspend;
end
