CREATE PROCEDURE HP_PDN_RECALCLIST_CALC(
    PDR_ID D_ID)
AS
declare variable pdr_status d_smallint;
  declare variable pdr_kind d_smallint;
  declare variable kc_id d_id;
  declare variable kcp_id d_id;
  declare variable k_id d_id;
  declare variable close_date d_date;
begin
  select pdr_status, kc_id, pdr_kind, kc_from_k_id
  from pdn_manual_recalclist, knclient
  where pdr_id = :pdr_id
    and kc_from_k_id = pdr_from_k_id
  into pdr_status, kc_id, pdr_kind, k_id;

  if (pdr_status <> 1) then
    exception e_message 'Расчет недоступен для данного статуса';

  select iif(oks_date_end < current_date, oks_date_end + 1, null)
  from oksummary oks
  where oks_from_k_id = :k_id
  order by oks_from_k_id, oks_date_end desc
  rows 1
  into close_date;

  if (close_date is null) then
  begin
    update pdn_manual_recalclist
    set pdr_status = -3
    where pdr_id = :pdr_id;
  end
  else
  begin
    execute procedure hp_knclient_pdn_update(kc_id, 0, null, null, null, 2, null, 'manual_pdnmodule');

    select kcp_id 
    from knclient_pdn
    where kc_id = :kc_id
      and kcp_kind = :pdr_kind
    into kcp_id;
  
    update pdn_manual_recalclist
    set pdr_from_kcp_id = :kcp_id
    where pdr_id = :pdr_id;
  end
end
