CREATE PROCEDURE MC_GOODS_CATEGORY_LIST(
    MCL_ID D_ID)
RETURNS (
    ID D_INTEGER,
    NAME D_STR_255)
AS
declare variable AG_ID D_INTEGER;
begin
    --  через пользователя КМП получаю, к какой торговой точке пользователь относится и далее получаю id действующего договора с контрагентом по товарам в рассрочку
    select ag.ag_id
    from mc_login ml
    inner join clientshops ch on ml.mcl_login = ch.cs_phone
    inner join agreements ag on ag.ag_object_id = ch.cs_from_c_id
    where current_date between ag.ag_date_begin and ag.ag_date_end
    and ag.ag_status = 1
    and ag.ag_from_rd_id = f_get_rd_id('AgreementKinds.installment_goods')
    and ml.mcl_id = :mcl_id
    rows 1
    into :ag_id;

    if (ag_id is not null) then
    begin
        -- получаю список товаров допустимых к продаже чз договор с контрагентом
        for select feg.feg_id,
                   feg.feg_name
        from foreign_econom_goods_ref feg
        inner join feacagreement_get (:ag_id)
        on rd_id = feg_from_rd_id
        and feg_code is null
        order by feg_name
        into :id,
             :name
        do
        begin
            suspend;
        end
    end
end
