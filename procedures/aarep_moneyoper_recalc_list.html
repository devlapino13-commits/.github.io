CREATE PROCEDURE AAREP_MONEYOPER_RECALC_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_DATE D_DATE,
    RES_TOTAL D_INTEGER,
    RES_RECALC D_INTEGER,
    RES_DATE_RECALC D_INTEGER)
AS
declare variable cash_acc_id d_id;
begin
  res_date = rep_date_begin;
  while (res_date <= rep_date_end) do
  begin
    res_total = 0;
    res_recalc = 0;
    res_date_recalc = 0;

    select count(*), sum(iif(coalesce(kob_from_rml_id, 0) = 0, 1, 0))
    from kn_operbook
    where kob_ol_date = :res_date and
          kob_ol_is_active = 1 and
          coalesce(kob_from_rml_id, 0) <> -1
    into res_total, res_recalc;

    if (res_total is null) then
      res_total = 0;

    if (res_recalc is null) then
      res_recalc = 0;

    for select acc_id
    from ref_account, sys_accbalance
    where sab_full_code = 'ПлатСистемы.Погашено' and
          acc_from_sab_id = sab_id
    into cash_acc_id
    do
    begin
      for select kob_id, kob_from_ol_id, kob_ol_date, kob_cashflow_date
      from kn_operbook
      where kob_ol_date = :res_date and
            kob_ol_is_active = 1 and
            kob_acc_dt + 0 = :cash_acc_id
      as cursor cur
      do
      begin
        if (not exists(select kob_ol_date
        from operperiod, kn_operbook, ref_account, sys_accbalance
        where cur.kob_ol_date between olp_date_begin and olp_date_end and
              kob_ol_date = cur.kob_cashflow_date and
              kob_ol_is_active = 1 and
              olp_from_ol_id = kob_from_ol_id and
              kob_acc_kt + 0 = :cash_acc_id and
              kob_acc_dt = acc_id and
              acc_from_sab_id = sab_id and
              sab_full_code starting 'Деньги.Безнал.Банк.')) then
          res_date_recalc = res_date_recalc + 1;
      end
    end

    suspend;

    res_date = res_date + 1;
  end
end
