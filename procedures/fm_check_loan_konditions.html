CREATE PROCEDURE FM_CHECK_LOAN_KONDITIONS(
    CHECK_KIND VARCHAR(100),
    K_ID DECIMAL(18,0),
    MA_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    RESULT_IS_OK INTEGER,
    RESULT_TXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable db_id integer;
declare variable un_name varchar(30);
declare variable client_fio varchar(100);
declare variable client_fio_prior varchar(100);
declare variable error_is_critical integer;
declare variable long_txt varchar(2000);
begin
  result_is_ok = 1;
  client_fio_prior = '';

  :db_id = f_get_db_id();

  select un_name
  from username
  where un_id = :un_id
  into un_name;

  result_txt = '';
  begin
    rdb$set_context('USER_SESSION', 'FINSTART_USER_LOGIN', :un_name);

    for select kf_client_fio, kf_error_is_critical,
               iif(kf_error_is_critical = 1, '<font color="red">' || iif(:db_id <> 94401, 'Критично', 'Kritik')|| ':' || '</font>', iif(kf_error_is_critical = 2, '<font color="red">' || iif(:db_id <> 94401, 'Очень критично', 'Kritik hədd')|| ':' || '</font>', iif(:db_id <> 94401, 'Сигнал', 'Xəbərdarlıq')|| ':')) || ' ' || coalesce(kf_error_text, '')
        from hp_check_loan_konditions(:k_id, 0, :ma_id, :check_kind, :un_id)
        order by kf_client_fio
        into client_fio, :error_is_critical, :long_txt
    do
    begin
      if (client_fio <> client_fio_prior) then
      begin
        result_txt = BLOB_APPEND(result_txt, iif(client_fio_prior <> '', '<br><br>', '') || '<b>' || client_fio || '</b>');
        client_fio_prior = client_fio;
      end
      result_txt = BLOB_APPEND(result_txt, '<br>' || long_txt);
      if (error_is_critical > 0) then
        result_is_ok = 0;
    end
    result_txt = result_txt || ''; --закроем блоб, читая его
    when any do
    begin
      rdb$set_context('USER_SESSION', 'FINSTART_USER_LOGIN', null);
      exception;
    end
  end
  suspend;
end
