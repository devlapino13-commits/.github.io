CREATE PROCEDURE HP_NEW_ACC_LIST_CREATE
RETURNS (
    REG_CODE VARCHAR(20),
    AL_ID VARCHAR(20),
    AL_NAME VARCHAR(255),
    AL_MEMO VARCHAR(255),
    AL_FROM_BG_ID VARCHAR(20),
    AL_FROM_REG_ID INTEGER,
    AL_CURRENCY VARCHAR(3),
    AL_TYPE INTEGER,
    NEW_AL_ID VARCHAR(20),
    NEW_BG_ID VARCHAR(20))
AS
declare variable no_str varchar(10);
declare variable next_no integer;
declare variable bg_id varchar(20);
declare variable bg_is_budget integer;
declare variable bg_is_recost integer;
declare variable bg_report integer;
begin
-- Создает необходимые лицевые счета согласно трансформационной таблицы Balgroup_Link
  for select bg_new_bg_id, b_old.bg_is_budget, b_old.bg_is_recost, b_old.bg_report
      from balgroup_list('0', 1, -1, 1), balgroup_link, balgroup b_old
      where id_list = bg_old_bg_id and
            bg_old_bg_id = b_old.bg_id
      into bg_id, bg_is_budget, bg_is_recost, bg_report
  do
  begin
    update balgroup
    set bg_is_budget = :bg_is_budget,
        bg_is_recost = :bg_is_recost,
        bg_report = :bg_report
    where bg_id = :bg_id;
  end

  update acc_list
  set al_date_close = '01.01.2013'
  where position('-', al_id) > 0 or
        al_from_bg_id starting '9X';

  for select reg_code, al_id, al_from_reg_id, al_from_bg_id, al_currency, al_type, al_name, al_memo
      from acc_list, region
      where al_from_reg_id = reg_unicode
      order by al_id
      into :reg_code, :al_id, :al_from_reg_id, :al_from_bg_id, :al_currency, :al_type, :al_name, :al_memo
  do
  begin
-- ищем есть ли соответствующий балансовый счет по новому формату
    new_bg_id = null;
    new_al_id = null;
    select bg_new_bg_id
    from balgroup_link
    where bg_old_bg_id = :al_from_bg_id rows 1
    into :new_bg_id;
    if (new_bg_id is not null) then
    begin
-- если балансовые счет есть, проверяем, открыт ли лицевой счет
      new_al_id = null;
      select al_new_al_id
      from acc_list_link
      where al_old_al_id = :al_id
      into :new_al_id;
      if (new_al_id is null) then
      begin
-- если лицевого счета нета, открываем заново
        next_no = 0;
        select max(al_number)
        from acc_list
        where al_from_bg_id = :new_bg_id and
              al_from_reg_id = :al_from_reg_id and
              al_currency = :al_currency
        into :next_no;
        if (next_no is null) then
          next_no = 0;
        next_no = next_no + 1;
        no_str = right('000000' || next_no, 6);
        new_al_id = reg_code || al_currency || new_bg_id || no_str;

        insert into acc_list (al_id, al_from_reg_id, al_date, al_from_bg_id, al_currency, al_number, al_type, al_name,
                              al_memo, al_date_close, al_is_active, is_delete)
        values (:new_al_id, :al_from_reg_id, '01.01.2013', :new_bg_id, :al_currency, :next_no, :al_type, :al_name,
                :al_memo, null, 1, 0);

        insert into acc_list_link (al_old_al_id, al_new_al_id)
        values (:al_id, :new_al_id);
      end
    end
-- если new_al_id пустое, нужно остановить замену счетов
    suspend;
  end
end
