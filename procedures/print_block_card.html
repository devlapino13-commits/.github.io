CREATE PROCEDURE PRINT_BLOCK_CARD(
    CBO_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME_FIO D_STR_255,
    C_PASS_NO D_STR_100,
    C_PASS_DATE D_DATE,
    C_PASS_WHO D_STR_255,
    C_PASS_DATE_EXPIRE D_DATE,
    C_INN D_STR_20,
    CELL_MAIN D_STR_20,
    C_ADDRESS_YUR D_STR_255,
    C_ADDRESS_FACT D_STR_255,
    BLANK_DATE D_DATE,
    QR_CODE D_STR_LONG,
    CBR_REASON D_STR_255,
    CAL_PAN_SHORT D_STR_20,
    FIO_SVZ D_STR_255)
AS
declare variable EDL_ID D_ID;
declare variable CBO_REASON D_ID;
declare variable C_ID D_ID;
declare variable ADDRESS_TYPE smallint;
declare variable C_ADDRESS D_STR_255;
declare variable CAL_ID D_ID;
declare variable UN_ID D_ID;
begin
  qr_code = '';
  fio_svz = '';
  for select cbo.cbo_from_c_id, cbo.cbo_reason, cbo.cbo_from_cal_id
      from cardblockopers cbo
      where cbo.cbo_id = :cbo_id
      into c_id, cbo_reason, cal_id
  do
  begin
    select cbr.cbr_reason
    from cardblockreasons cbr
    where cbr.cbr_id = :cbo_reason
    into cbr_reason;

    select cal.cal_pan_short
    from cardlist cal
    where cal.cal_card_id = :cal_id
    into cal_pan_short;
    cell_main = 0;
    select c.c_name_fio, c.c_pass_no, c.c_pass_date, c.c_pass_who, c.c_pass_date_expire, c.c_inn
    from clients c
    where c.c_id = :c_id
    into c_name_fio, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire, c_inn;

    cell_main = f_cont_info(0, :c_id, 'cell_main', 'ok');

    c_address_yur = null;
    c_address_fact = null;
    for select adc_address_type, adc_address_str
        from hp_addrclient(:c_id)
        into :address_type, :c_address
    do
    begin
      if (address_type = 1) then
        c_address_yur = :c_address;
      if (address_type = 2) then
        c_address_fact = :c_address;
    end

    if (coalesce(add_params, '') <> '') then
    begin
      edl_id = f_json_str_get_value(add_params, 'edl_id', null);
      qr_code = f_edoc_qr_by_edl(edl_id, c_id);

      select edl.edl_blank_date, edl.edl_create_un_id
      from edoc_list edl
      where edl.edl_id = :edl_id
      into blank_date, un_id;

      select un.un_fio
      from username un
      where un.un_id = :un_id
      into fio_svz;

    end
    suspend;
  end
end
