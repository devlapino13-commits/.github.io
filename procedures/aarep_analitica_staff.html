CREATE PROCEDURE AAREP_ANALITICA_STAFF(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REP_STAFF_ALL INTEGER,
    REP_STAFF_HQ INTEGER,
    REP_STAFF_BRANCH INTEGER,
    REP_STAFF_LO INTEGER,
    REP_LO_TO_STAFF FLOAT,
    REP_BRANCH_NO INTEGER,
    REP_OFFICE_NO INTEGER,
    REP_ALL_CLIENTS INTEGER,
    REP_ALL_DISB DECIMAL(18,4),
    REP_MOBILE_NO INTEGER)
AS
declare variable hq_id integer;
declare variable reg_id integer;
declare variable jl_code varchar(100);
declare variable rep_minioffice integer;
declare variable rep_start_date date;
declare variable c_name_fio varchar(100);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  rep_staff_all = 0;
  rep_staff_hq = 0;
  rep_staff_branch = 0;
  rep_staff_lo = 0;
  rep_branch_no = 0;
  rep_office_no = 0;
  rep_minioffice = 0;
  rep_mobile_no = 0;

  rep_all_clients = 0;
  rep_all_disb = 0;

  hq_id = f_hq_id();

  for select ps_from_reg_id, jk_system_code
      from hr_pstatus, joblist, jobkind
      where ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_from_jl_id = jl_id and
            ps_is_working = 1 and
            jl_kind = jk_id
      into :reg_id, :jl_code
  do
  begin
    rep_staff_all = rep_staff_all + 1;
    if (reg_id = hq_id) then
      rep_staff_hq = rep_staff_hq + 1;
    if (jl_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
      rep_staff_lo = rep_staff_lo + 1;
  end
  rep_staff_branch = rep_staff_all - rep_staff_hq;
  if (rep_staff_all <> 0) then
  begin
    rep_lo_to_staff = rep_staff_lo * 100 / rep_staff_all;
  end

  for select distinct reg_unicode, reg_head_office, reg_detail_kolich
      from region
      where reg_status = 3 and
            reg_is_agent = 0
      into :reg_id, :hq_id, :rep_minioffice
  do
  begin
    select min(ol_date_oper_day)
    from operlist
    where ol_from_reg_id = :reg_id
    into :rep_start_date;

    if (rep_start_date <= :rep_date_end) then
    begin
      rep_office_no = rep_office_no + 1;
      rep_office_no = rep_office_no + rep_minioffice;
      rep_mobile_no = rep_mobile_no + rep_minioffice;
    end
  end
  for select reg_unicode
      from region
      where reg_status = 3 and
            reg_head_office = reg_unicode
      into :reg_id
  do
  begin
    rep_branch_no = rep_branch_no + 1;
  end
  if (rep_branch_no > 1) then
    rep_branch_no = rep_branch_no - 1;/* Убираем ГО */

  for select distinct c_name_fio
      from kontstate, knclient, clients
      where ks_from_k_id = kc_from_k_id and
            kc_from_c_id = c_id and
            ks_ol_is_delete = 0 and
            cast(ks_date_begin as date) <= :rep_date_end and
            ks_date_end > cast(ks_date_begin as date) and
            kc_amount_approved is not null and
            kc_amount_approved > 0 and
            ks_state = 1
      into :c_name_fio
  do
  begin
    rep_all_clients = rep_all_clients + 1;
  end

  for select sum(ok_os * k_index_curr_rate)
      from operkred, kontract
      where ok_ol_is_active = 1 and
            ok_type = 8 and
            k_id = ok_from_k_id and
            ok_ol_date <= :rep_date_end and
            ok_from_c_id = 0
      into :rep_all_disb
  do
  begin
    if (rep_all_disb is null) then
      rep_all_disb = 0;
  end
  rep_all_disb = cast(rep_all_disb as decimal(15,2));
  suspend;
end
