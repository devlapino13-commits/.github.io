CREATE PROCEDURE HP_PRE_SUD_ACTION_LOG_INS(
    ACTION_TYPE D_INTEGER,
    ACTION_ID D_ID,
    ACTION_K_ID D_ID,
    ACTION_DATETIME D_DATETIME,
    ACTION_UN_FIO D_STR_100)
AS
begin
  -- создает запись в журнале учета действий модуля досудебных взысканий (RFCRU-4792)

  -- pre_sud_action_log - таблица логов
  -- поле psal_action_type - тип действия: select * from enum_list where el_kind = 'pre_sud_actions'
  -- поле psal_action_id - ID действия, значение зависит от psal_action_type:
  -- register_in  -> psal_action_id = 0, контракт с активной просрочкой (hp_pre_sud_kontract_in_out)
  -- register_out -> psal_action_id = 0, контракт без активной просрочки (hp_pre_sud_kontract_in_out)
  -- mail_in      -> psal_action_id это omk_id из outmailkont при "outmailing.oml_is_out = 0" (outmailkont_ai)
  -- mail_out     -> psal_action_id это omk_id из outmailkont при "outmailing.oml_is_out = 1" (outmailkont_ai)
  -- message_sms  -> psal_action_id это msg_id из fsrobot_msgbox при "msg_type = 1" (fsrobot_msgbox_ai20)
  -- message_push -> psal_action_id это msg_id из fsrobot_msgbox при "msg_type = 2" (fsrobot_msgbox_ai20)
  -- phone_calls  -> psal_action_id это pnc_id из phonecalls (phonecalls_ai)
  -- auto_inform  -> psal_action_id это oic_id из okt_ivr_calls (okt_ivr_calls_ai)
  -- operator_comment -> psal_action_id это pcl_id из pantera_clients при "pcl_has_memo = 1" (pantera_clients_ai20)
  -- pretrial_claim -> psal_action_id это edl_id из edoc_list (edoc_list_au0)

  if (coalesce(action_un_fio, '') = '') then
    exception e_message 'Сотрудник не указан';

  if (action_k_id is null) then
    exception e_message 'Контракт не указан';

  if (action_type is null or
      action_id is null or
      action_datetime is null) then
    exception e_message 'Действие не указано';

  if (action_type = f_enum_id('pre_sud_actions', 'register_in', 1) and
      not exists(select *
                 from pre_sud_kontract_list
                 where pskl_k_id = :action_k_id)) then
    insert into pre_sud_kontract_list (pskl_k_id)
    values (:action_k_id);

  insert into pre_sud_action_log (psal_action_datetime, psal_action_type, psal_action_id, psal_action_k_id, psal_action_un_fio)
  values (:action_datetime, :action_type, :action_id, :action_k_id, :action_un_fio);

  if (action_type = f_enum_id('pre_sud_actions', 'register_out', 1)) then
    delete from pre_sud_kontract_list
    where pskl_k_id = :action_k_id;
end
