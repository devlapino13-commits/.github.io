CREATE PROCEDURE AAREP_PROSROCH_PERIOD(
    REP_REGION INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OK_FROM_K_ID DECIMAL(18,0),
    OK_SH_ID DECIMAL(18,0),
    OK_SH_DATE DATE,
    K_KONTRACT_NO VARCHAR(20),
    KAL_NEW_KF_NAME VARCHAR(50),
    KAL_NEW_REG_ID INTEGER,
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_KOLICH_REPAY INTEGER,
    K_CIRCLE INTEGER,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_PERCENT_CATEGORY DECIMAL(18,7),
    K_PERCENT DECIMAL(18,7),
    PR_CODE VARCHAR(100),
    PR_NAME VARCHAR(255),
    PR_LOAN_TYPE INTEGER,
    OL_DATE_1 DATE,
    OK_OS DECIMAL(18,4),
    OK_PERC DECIMAL(18,4),
    OK_FINE DECIMAL(18,4),
    PR_LOAN_TYPE_STR VARCHAR(20))
AS
declare variable reg_prior integer;
declare variable parent_id integer;
begin
  reg_prior = 0;
  for select ok_from_k_id, ok_sh_id, ok_sh_date, k_kontract_no, kal_new_kf_name, kal_new_reg_id,
             reg_code || ' ' || reg_name, k_disb_date, k_amount_approved, k_kolich_repay, k_circle,
             k_kolich_client_approved, k_percent_category, k_percent, pr_code, pr_name, pr_loan_type, max(ok_ol_date),
             sum(ok_os), sum(ok_perc), sum(ok_fine), parent
      from operkred, kontract, knaktlist, region, region_list(:rep_region, 1, 3, 1), product
      where ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_type in (1, 4, 5, 6, 7, 9) and
            k_id = ok_from_k_id and
            kal_from_kf_id = k_from_kf_id and
            :rep_date_end between kal_date and kal_date_end and
            kal_new_reg_id = reg_unicode and
            (id_list = reg_unicode) and
            k_from_pr_id = pr_id and
            ok_from_c_id = 0
      group by ok_from_k_id, ok_sh_id, ok_sh_date, k_kontract_no, kal_new_kf_name, kal_new_reg_id, reg_code, reg_name, k_disb_date, k_amount_approved, k_kolich_repay, k_circle, k_kolich_client_approved, k_percent_category, k_percent, pr_code, pr_name, pr_loan_type, parent
      having sum(ok_os) <> 0 and max(ok_ol_date) > ok_sh_date
      into :ok_from_k_id, :ok_sh_id, :ok_sh_date, :k_kontract_no, :kal_new_kf_name, :kal_new_reg_id, :reg_name,
           :k_disb_date, :k_amount_approved, :k_kolich_repay, :k_circle, :k_kolich_client_approved, :k_percent_category,
           :k_percent, :pr_code, :pr_name, :pr_loan_type, :ol_date_1, :ok_os, :ok_perc, :ok_fine, :parent_id

  do
  begin
    if (parent_id <> reg_prior) then
    begin
      select reg_name
      from region
      where reg_unicode = :parent_id
      into :parent_name;
      reg_prior = parent_id;
    end

    if (pr_loan_type = 1) then
      pr_loan_type_str = 'Индив.';
    else
      pr_loan_type_str = 'Группа';

    suspend;
  end
end
