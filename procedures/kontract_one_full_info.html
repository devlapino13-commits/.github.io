CREATE PROCEDURE KONTRACT_ONE_FULL_INFO(
    KONT_ID D_ID,
    REPORT_DATE_END D_DATE)
RETURNS (
    K_ID D_ID,
    K_FROM_KF_ID D_ID,
    K_CURRENCY D_STR_LANG,
    K_APPLICATE_DATE D_DATE,
    K_APPLICATE_NO D_INTEGER,
    K_KONTRACT_NO D_STR_20,
    K_KONTRACT_DATE D_DATE,
    K_FROM_PR_ID D_INTEGER,
    K_DISB_DATE D_DATE,
    K_AMOUNT_APPLICATE D_CURRENCY,
    K_STAGE D_INTEGER,
    K_PERIOD DECIMAL(15,2),
    K_KOLICH_LGOTA D_INTEGER,
    K_PERCENT D_PERCENT,
    K_PERCENT_LGOTA D_PERCENT,
    K_KOLICH_REPAY D_INTEGER,
    K_FROM_F_ID D_STR_20,
    K_FROM_PT_ID D_STR_20,
    K_STATUS D_INTEGER,
    K_STATUS_NAME VARCHAR(30),
    K_CLOSE_DATE D_DATE,
    K_CIRCLE D_INTEGER,
    K_CIRCLE_INTERNAL D_INTEGER,
    K_PERCENT_PER_DAY D_PERCENT,
    K_MEMO D_STR_LONG,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_PERCENT_FOR_DAYS D_INTEGER,
    K_KOLICH_CLIENT_APPLICATE D_INTEGER,
    K_KOLICH_CLIENT_APPROVED D_INTEGER,
    K_REPAY_KOLICH_CLOSED D_INTEGER,
    K_CLASS D_STR_20,
    K_KOMMIS_PERC D_PERCENT,
    K_CREDIT_TARGET D_STR_255,
    K_IS_KREDLINE_STR D_STR_10,
    K_KREDLINE_END D_DATE,
    KFI_NAME D_STR_100,
    KFI_OFFICER_CODE D_STR_20,
    KFI_OFFICER_NAME D_STR_100,
    KFI_REGION_NAME D_STR_CODE,
    KFI_REGION_ID D_INTEGER,
    KFI_DATE D_DATE,
    KFI_DATE_END D_DATE,
    PR_CODE D_STR_100,
    PR_NAME D_STR_CODE,
    PR_IS_SECURED_TEXT D_STR_CODE,
    PR_LOAN_TYPE_TEXT D_STR_CODE,
    K_CIRCLE_TEXT D_STR_10,
    K_BALANCE D_CURRENCY,
    K_PAYED_FINE D_CURRENCY,
    PR_LOAN_TYPE D_INTEGER,
    KS_STATE D_INTEGER,
    KS_STATE_TEXT D_STR_CODE,
    K_INDEX_CURR_CODE D_STR_LANG,
    K_INDEX_CURR_RATE D_CURRENCY,
    K_LAST_BONUS_STR VARCHAR(30),
    K_IS_MAIN_LOAN VARCHAR(30),
    K_CURR_BONUS_STR D_STR_CODE,
    K_DELINQ_PAY7 D_INTEGER,
    K_DELINQ_PAY30 D_INTEGER,
    K_DELINQ_PAY60 D_INTEGER,
    K_DELINQ_PAY_MORE D_INTEGER,
    K_DELINQ_CURR_DAYS D_INTEGER,
    K_PERCENT_INPUT D_PERCENT,
    K_STOP_PERC D_DATE,
    K_STOP_FINE D_DATE,
    SBL_NAME D_STR_255,
    PR_MEMO D_STR_255,
    IS_MBH D_BOOL)
AS
declare variable pr_is_secured d_integer;
declare variable k_is_kredline d_integer;
declare variable st_staff d_integer;
declare variable st_client d_integer;
declare variable c_last_bonus_status d_integer;
declare variable c_curr_bonus_status d_integer;
declare variable pr_category d_integer;
declare variable sh_date d_date;
declare variable ol_delinq_pay_dd d_date;
declare variable balance d_currency;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select k_id, k_from_kf_id, k_currency, k_applicate_date, k_applicate_no, k_kontract_no, k_kontract_date, k_from_pr_id,
         k_disb_date, k_amount_applicate, k_stage, k_period, k_kolich_lgota, kd_percent, k_percent_lgota,
         k_kolich_repay, k_from_f_id, k_from_pt_id, kontract.k_status, k_close_date, k_circle, k_circle_internal,
         kd_percent_per_day, k_memo, k_amount_approved, k_percent_for_days, k_kolich_client_applicate,
         k_kolich_client_approved, k_repay_kolich_closed, k_class, k_kommis_perc, k_credit_target, k_is_kredline,
         k_kredline_end, k_index_curr_code, k_index_curr_rate, kal_new_kf_name, kal_new_officer_code, c_name_fio,
         reg_name, reg_unicode, kal_date, kal_date_end, pr_code, pr_name, pr_is_secured, pr_loan_type, ks_state,
         pr_category, k_percent_input, k_stop_perc, k_stop_fine, pr_memo
  from kontract, knaktlist, clients, region, product, kontstate, kontdata
  where k_id = :kont_id and
        kal_from_kf_id = k_from_kf_id and
        kal_new_kf_officer = c_id and
        k_from_pr_id = pr_id and
        reg_unicode = kal_new_reg_id and
        kal_date <= :report_date_end and
        kal_date_end >= :report_date_end and
        kd_from_k_id = k_id and
        kd_ol_date <= :report_date_end and
        kd_date_end >= :report_date_end and
        kd_ol_is_delete = 0 and
        ks_from_k_id = k_id and
        cast(ks_date_begin as date) <= :report_date_end and
        ks_date_end >= :report_date_end and
        ks_ol_is_delete = 0 and
        knaktlist.is_delete = 0
  into :k_id, :k_from_kf_id, :k_currency, :k_applicate_date, :k_applicate_no, :k_kontract_no, :k_kontract_date,
       :k_from_pr_id, :k_disb_date, :k_amount_applicate, :k_stage, :k_period, :k_kolich_lgota, :k_percent,
       :k_percent_lgota, :k_kolich_repay, :k_from_f_id, :k_from_pt_id, :k_status, :k_close_date, :k_circle,
       :k_circle_internal, :k_percent_per_day, :k_memo, :k_amount_approved, :k_percent_for_days,
       :k_kolich_client_applicate, :k_kolich_client_approved, :k_repay_kolich_closed, :k_class, :k_kommis_perc,
       :k_credit_target, :k_is_kredline, :k_kredline_end, :k_index_curr_code, :k_index_curr_rate, :kfi_name,
       :kfi_officer_code, :kfi_officer_name, :kfi_region_name, :kfi_region_id, :kfi_date, :kfi_date_end, :pr_code,
       :pr_name, :pr_is_secured, :pr_loan_type, :ks_state, :pr_category, :k_percent_input, :k_stop_perc, :k_stop_fine, pr_memo;
  is_mbh = iif(f_new_markup_mbh(:kont_id), 1, 0);

  if ((k_disb_date >= '26.10.2013') and
      (db_id in (39801, 41701))) then
    k_percent_input = null;
  k_close_date = null;
  select sum(ok_os), -1 * sum(ok_fine), max(ok_ol_date)
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type <= 8 and
        ok_ol_date <= :report_date_end and
        ok_from_c_id = 0
  into :k_balance, :k_payed_fine, :k_close_date;
  if (k_balance is null) then
    k_balance = 0;
  if (k_payed_fine is null) then
    k_payed_fine = 0;

  if (k_balance is null) then
    k_balance = 0;
  if (k_payed_fine is null) then
    k_payed_fine = 0;
  if (k_close_date is null) then
    k_close_date = cast('31.12.9999' as date);

  k_status_name = '';
  if (k_status = -1) then
    k_status_name = 'Аннулирован';
  if (k_status = -2) then
    k_status_name = 'Заявка';
  if (k_status = 0) then
    k_status_name = 'В процессе';
  if (k_status = 1) then
    k_status_name = 'Контракт';

  if (k_is_kredline = 0) then
    k_is_kredline_str = 'Нет';
  if (k_is_kredline = 1) then
    k_is_kredline_str = 'Да';

  ks_state_text = '';
--    if ((ks_state=0) or (ks_state=1)) then ks_state_text='В процессе';
--    if (ks_state=2) then ks_state_text='Активный';
--    if (ks_state=3) then ks_state_text='Закрытый';
--  Исправлено на следующее
  if (ks_state = 0) then
    ks_state_text = 'В процессе';
  else
  if (ks_state = 1) then
    ks_state_text = 'Активный';
  else
    ks_state_text = 'Закрытый';

  if (pr_is_secured = 1) then
    pr_is_secured_text = 'С залогом';
  else
    pr_is_secured_text = 'Без залога';

  if (pr_loan_type = 1) then
    pr_loan_type_text = iif(is_mbh = 0, 'Индивидуальный кредит', 'Индивидуальное финансирование');
  else
    pr_loan_type_text = iif(is_mbh = 0, 'Групповой кредит', 'Групповое финансирование');

  k_percent = k_percent - k_percent_lgota;

  if (k_circle_internal > 1) then
    k_circle_text = (cast(k_circle as varchar(3)) || ' / ' || cast(k_circle_internal as varchar(3)));
  else
    k_circle_text = cast(k_circle as varchar(5));

  if (db_id = 81001) then
  begin
    if (report_date_end <= '15.10.2017') then
    begin
      select c_last_bonus_status, c_curr_bonus_status, c_last_bonus_status_str, c_curr_bonus_status_str
      from hp_kontstate_loan_status(:kont_id, 0, 0, :report_date_end, '')
      into :c_last_bonus_status, :c_curr_bonus_status, :k_last_bonus_str, k_curr_bonus_str;
    end
    else
    begin
      select k_curr_status, k_curr_status_str, k_last_status, k_last_status_str
      from hp_get_kontract_status(:kont_id, :report_date_end)
      into :c_curr_bonus_status, :k_curr_bonus_str, :c_last_bonus_status, :k_last_bonus_str;
    end
  end
  if (pr_category = 0) then
    k_is_main_loan = 'Основной кредит';
  else
    k_is_main_loan = 'Параллельный кредит';

  k_delinq_pay7 = 0;
  k_delinq_pay30 = 0;
  k_delinq_pay60 = 0;
  k_delinq_pay_more = 0;
  for select distinct ok_sh_date
      from operkred
      where ok_ol_is_active = 1 and
            ok_from_k_id = :kont_id and
            ok_ol_date <= :report_date_end and
            ok_from_c_id = 0
      into :sh_date
  do
  begin
    ol_delinq_pay_dd = null;
    select max(ok_ol_date)
    from operkred
    where ok_ol_is_active = 1 and
          ok_ol_date <= :report_date_end and
          ok_sh_date < ok_ol_date and
          ok_sh_date = :sh_date and
          ok_from_k_id = :kont_id and
          ok_type = 4 and
          (ok_os + ok_perc) <= -1 and
          ok_from_c_id = 0
    into :ol_delinq_pay_dd;
    if (ol_delinq_pay_dd is not null) then
    begin
      if ((ol_delinq_pay_dd - sh_date) <= 7) then
        k_delinq_pay7 = k_delinq_pay7 + 1;
      else
      if ((ol_delinq_pay_dd - sh_date) <= 30) then
        k_delinq_pay30 = k_delinq_pay30 + 1;
      else
      if ((ol_delinq_pay_dd - sh_date) <= 60) then
        k_delinq_pay60 = k_delinq_pay60 + 1;
      else
        k_delinq_pay_more = k_delinq_pay_more + 1;
    end
  end
  k_delinq_curr_days = 0;
  if (ks_state = 1) then
  begin
    for select ok_sh_date, sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc)
        from operkred
        where ok_from_k_id = :kont_id and
              ok_ol_is_active = 1 and
              ok_sh_date <= :report_date_end and
              ok_ol_date <= :report_date_end and
              ok_type <= 6 and
              ok_sh_id > 0 and
              ok_from_c_id = 0
        group by ok_sh_date
        having sum(ok_os + ok_sh_os) >= 1 or sum(ok_perc + ok_sh_perc) >= 1
        into :sh_date, :balance
    do
    begin
      if (k_delinq_curr_days < (:report_date_end - sh_date)) then
        k_delinq_curr_days = :report_date_end - sh_date;
    end
  end
  sbl_name = null;
  select sbl_name
  from sb_akt, sb_list
  where sba_from_k_id = :kont_id and
        sba_status = 2 and
        sba_is_delete = 0 and
        current_date between sba_date_begin and sba_date_end and
        sba_from_sbl_id = sbl_id and
        sbl_id > 0 rows 1
  into sbl_name;
  if (sbl_name is null) then
    sbl_name = '';
  if (sbl_name <> '' and db_id in (39801, 41701)) then
  kfi_officer_name = 'ОВР';

  suspend;
end
