CREATE PROCEDURE PRINT_KONTPSK_INFO(
    KPS_ID D_ID)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    K_KONTRACT_DATE D_DATE,
    K_PERCENT_PER_YEAR D_PERCENT,
    K_ZAD_OS D_CURRENCY,
    K_ZAD_PERC D_CURRENCY,
    K_ZAD_FINE D_CURRENCY,
    K_ZALOG_STR D_STR_CODE,
    KC_AMOUNT_APPROVED D_CURRENCY,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    OL_DATE D_DATE,
    LAST_SH_DATE D_DATE,
    PERIOD_MONTH D_CURRENCY,
    KPS_PSK D_PERCENT,
    KPS_PSK_AMOUNT D_CURRENCY,
    C_INN STR_100,
    PAID_OS D_CURRENCY,
    PAID_PERC D_CURRENCY,
    PAID_FINE D_CURRENCY,
    PAID_TOTAL D_CURRENCY,
    OVERDUE_OS D_CURRENCY,
    OVERDUE_PERC D_CURRENCY,
    OVERDUE_FINE D_CURRENCY,
    OVERDUE_TOTAL D_CURRENCY,
    PERCENT_YEAR_R DECIMAL(15,3))
AS
declare variable K_ID D_ID;
declare variable OL_ID D_ID;
declare variable PR_ID D_INTEGER;
declare variable K_DISB_DATE D_DATE;
declare variable DB_ID D_ID;
begin
  db_id = f_get_db_id();

  paid_os = 0;
  paid_perc = 0;
  paid_fine = 0;
  paid_total = 0;
  overdue_os = 0;
  overdue_perc = 0;
  overdue_fine = 0;
  overdue_total = 0;
  c_inn = '';

  select kps_from_k_id, kps_psk, ol_id, ol_date
  from kontpsk, operlist
  where kps_id = :kps_id and
        kps_from_ol_id = ol_id
  into k_id, kps_psk, ol_id, ol_date;

  select k_kontract_no, k_kontract_date, k_disb_date, k_percent_per_day * 365,
    k_from_pr_id
  from kontract
  where k_id = :k_id
  into k_kontract_no, k_kontract_date, k_disb_date, k_percent_per_year,
    pr_id;

  k_zalog_str = 'Нет';
  if (exists(select * from product where pr_id = :pr_id and pr_code containing '.ZALOG_CAR.')) then
    k_zalog_str = 'Да';

  select ok_sh_date
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type < 7 and
        ok_ol_date <= :ol_date and
        ok_from_ol_id <= :ol_id
  group by ok_sh_id, ok_sh_date
  having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
  order by ok_sh_date desc
  rows 1
  into last_sh_date;

  --Расчет выплаченных сумм для кз
  if (db_id = 39801) then
  begin
    select -sum(ok_os + ok_sh_os),-sum(ok_perc + ok_sh_perc), -sum(ok_fine)
    from operkred
    where ok_from_k_id = :k_id and
    ok_ol_is_active = 1 and
    ok_type = 4  and
    ok_ol_date <= :ol_date and
    ok_from_ol_id <= :ol_id
    into :paid_os, :paid_perc, :paid_fine;
    paid_total = coalesce(paid_os, 0) + coalesce(paid_perc, 0) + coalesce(paid_fine, 0);
  end

  period_month = round(datediff(day, k_disb_date, last_sh_date) / (365.0 / 12.0), 1);

  select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine)
  from aarep_make_zad_list(:k_id, :ol_date, 1)
  where sh_date is not null
  into k_zad_os, k_zad_perc, k_zad_fine;

  --Расчет сумм в просрочке для кз
  if (db_id = 39801) then
  begin
    select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine)
    from aarep_make_zad_list(:k_id, :ol_date, 1)
    where sh_date is not null and
    sh_date < cast(:ol_date as date)
    into :overdue_os, :overdue_perc, :overdue_fine;
    overdue_total = coalesce(overdue_os, 0) + coalesce(overdue_perc, 0) + coalesce(overdue_fine, 0);
  end

  select sum(ok_sh_perc + ok_kommis)
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type < 4 and
        ok_ol_date <= :ol_date and
        ok_from_ol_id <= :ol_id
  into kps_psk_amount;

  select c_id, c_name_fio, kc_amount_approved, c_birth_date, c_inn
  from clients, knclient
  where kc_from_k_id = :k_id and
        kc_from_c_id = c_id and
        kc_amount_approved > 0
  rows 1
  into c_id, c_name_fio, kc_amount_approved, c_birth_date, c_inn;

  if (kps_id = 2047259) then
  begin
    ol_date = '23.09.2025';
    k_percent_per_year = 59.617;
    k_zad_perc = 11240;
  end

  select kd_percent_per_day * 365
  from kontdata
  where :ol_date between kd_ol_date and kd_date_end and
        kd_ol_is_delete = 0 and
        kd_from_k_id = :k_id
  rows 1
  into percent_year_r; -- годовая процентная ставка на дату погашения (ставка после погашения), RFCRU-6086
  
  suspend;

end
