CREATE PROCEDURE MKR_PURPOSEOFCREDIT_GET(
    PR_ID D_ID,
    KNC_BUSINESS_SUBTYPE_ID D_INTEGER,
    K_DISB_DATE D_DATE)
RETURNS (
    PURPOSEOFCREDIT VARCHAR(3))
AS
declare variable pr_code d_str_100;
declare variable bsl_name d_str_255;
begin
  /*
          Коды кредитных назначений <PurposeOfCredit>
    
          Код    Описание
          001    Потребительские кредиты физическим лицам
          002    Нефтяной сектор
          003    Ипотечные кредиты физическим лицам
          004    Сектор торговли и обслуживания
          005    Сектор Энергетики, химии и природных ресурсов
          006    Сельскохозяйственный сектор сектор обработки
          007    Строительный и имущественный сектор
          008    Промышленный и производственный сектор
          009    Сектор транспорта и связи
          010    Другие сектора

  */

  select pr.pr_code
  from product pr
  where pr.pr_id = :pr_id
  into :pr_code;

  if (pr_code like '%.MKR_EXCEPTION1.%') then
  begin
    purposeofcredit = '004';
    if (knc_business_subtype_id in (8, 9, 10, 11, 17)) then
      purposeofcredit = '006';
  end
  else
  begin
    if ((pr_code like '%.IND.%') and
        (pr_code like '%.EXPRESS.%')) then
      purposeofcredit = '006';
    else
    if (pr_code like '%.ABIZ.%') then
    begin
      select substring(bsl.bsl_name from 1 for 3)
      from busstandartlink bslink
      inner join busstandartlist bsl on bsl.bsl_id = bslink.bslk_from_bsl_id and
            bslink.bslk_from_bsm_id = bsl.bsl_from_bsm_id
      where bslink.bslk_from_bs_id = :knc_business_subtype_id and
            bslink.bslk_from_bsm_id = 2
      into :purposeofcredit;
    end
    else
    begin
      if (k_disb_date >= '14.01.2016') then
        purposeofcredit = '001';
      else
      begin
        select bsl.bsl_name
        from busstandartlink bslink
        inner join busstandartlist bsl on bsl.bsl_id = bslink.bslk_from_bsl_id and
              bslink.bslk_from_bsm_id = bsl.bsl_from_bsm_id
        where bslink.bslk_from_bs_id = :knc_business_subtype_id and
              bslink.bslk_from_bsm_id = 2 rows 1
        into :bsl_name;

        purposeofcredit = '010';
        if (bsl_name = '') then
          purposeofcredit = '010';
        else
          purposeofcredit = left(bsl_name, 3);
      end
    end
  end

  if (pr_id in (68, 69, 70, 71, 72, 73, 74, 75, 76, 77,
                78, 79, 80, 81, 82, 83, 84, 85)) then
    purposeofcredit = '010';
  if (pr_id in (86, 87)) then
    purposeofcredit = '001';
  if (pr_id in (88, 89, 90, 91)) then
    purposeofcredit = '010';
  if (pr_code like '%.Aile.%' or pr_code like '%.Aqro.%') then
    purposeofcredit = '001';
  if (pr_code like '%.Micro_Biznes.%') then
    purposeofcredit = '004';

  suspend;
end
