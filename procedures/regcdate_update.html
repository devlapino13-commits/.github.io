CREATE PROCEDURE REGCDATE_UPDATE(
    REG_ID INTEGER,
    RCV_ID INTEGER)
AS
declare variable rc_id integer;
declare variable dd_prior date;
declare variable rcd_id integer;
declare variable rcd_date_begin date;
begin
  /* Procedure Text */
  select rcv_from_rc_id
  from regcvalue
  where rcv_id = :rcv_id
  into :rc_id;
  dd_prior = cast('31.12.9999' as date);
  for select rcd_id, rcd_date_begin
      from regcdate, regcvalue
      where rcd_from_rcv_id = rcv_id and
            rcv_from_rc_id = :rc_id and
            rcd_from_reg_id = :reg_id and
            rcd_is_delete = 0
      order by rcd_date_begin desc, rcd_id desc
      into :rcd_id, :rcd_date_begin
  do
  begin
    update regcdate
    set rcd_date_end = :dd_prior
    where rcd_id = :rcd_id and
          (rcd_date_end <> :dd_prior or
          rcd_date_end is null);
    dd_prior = :rcd_date_begin - 1;
  end

end
