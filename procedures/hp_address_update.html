CREATE PROCEDURE HP_ADDRESS_UPDATE(
    UPD_ADL_ID D_ID,
    PL_ID D_ID,
    STREET_ID D_ID,
    DEPART_NO D_STR_20,
    DEPART_PREF D_STR_20,
    FLAT_NO D_STR_20,
    BUILDING_NO D_STR_20,
    SECTION_NO D_STR_20,
    COMMENT_STR D_STR_255,
    POST_INDEX D_STR_20,
    IS_WITHOUT_STREET D_SMALLINT,
    IS_EDITED D_SMALLINT,
    FACT_ADDRESS D_STR_255,
    LANG_KIND D_SMALLINT,
    COUNTRY_PL_ID D_ID,
    OBLAST_PL_ID D_ID,
    UN_ID D_ID,
    VLADEN_NO D_STR_20 = null)
RETURNS (
    ADL_ID D_ID,
    ADL_ADDRESS_STR D_STR_255)
AS
declare variable db_id integer;
declare variable has_any_no integer;
declare variable res_text d_str_255;
begin
  res_text = '';
  lang_kind = coalesce(lang_kind, 0);

  db_id = f_get_db_id();

  if (res_text = '' and coalesce(pl_id, 0) = 0) then
    res_text = iif(db_id <> 94401, 'Населенный пункт не указан', 'Yaşayış məntəqəsi qeyd olunmayıb');

  if (coalesce(street_id, 0) > 0 and is_without_street = 1 and street_id <> pl_id) then
    is_without_street = 0;

  if (res_text = '' and
      coalesce(street_id, 0) <= 0 and
      is_without_street = 0) then
    res_text = iif(db_id <> 94401, 'Улица не указана', 'Küçə qeyd olunmayıb');

  if (res_text = '' and street_id > 0 and pl_id > 0 and street_id <> pl_id) then
  begin
    if (not exists(select * from places
                   where pl_id = :street_id and
                         parent = :pl_id)) then
      res_text = trim(iif(db_id <> 94401, 'Улица из другого населенного пункта', 'Küçə başqa rayondan'));
  end

  if (coalesce(trim(post_index), '') = '0') then
    post_index = '';

  if (res_text = '' and char_length(coalesce(trim(post_index), '')) not in (0, 6)) then
    res_text = iif(db_id <> 94401, 'Почтовый индекс указан неверно', 'Poçt kodu səhvdir');

  if (res_text <> '') then
    exception e_message res_text;

  depart_no = coalesce(depart_no, '0');
  depart_pref = coalesce(depart_pref, '');
  flat_no = coalesce(flat_no, '0');
  building_no = coalesce(building_no, '0');
  section_no = coalesce(section_no, '0');
  post_index = coalesce(post_index, '');
  vladen_no = coalesce(vladen_no, '');

  if (char_length(post_index) <= 3) then
    post_index = '';
  comment_str = coalesce(comment_str, '');

  if (is_edited is null) then
    is_edited = 0;

  has_any_no = 0;

  adl_address_str = '';
  select adl_address_str
  from hp_make_address_str(:pl_id, :street_id, :depart_no, :depart_pref, :flat_no, :building_no, :section_no,
                           :post_index, :is_without_street,
                           '', 0, :lang_kind, :un_id, :vladen_no)
  into :adl_address_str;

  if (adl_address_str = fact_address) then
    is_edited = 0;

  if (is_edited = 1 and coalesce(fact_address, '') <> '') then
  begin
    if (coalesce(fact_address, '') <> '') then
      adl_address_str = fact_address;
  end else
    is_edited = 0;

  -- Конец формирования
  if (post_index = '') then
    post_index = '0';
  if (is_without_street = 1) then
    street_id = pl_id; -- Т.к. улицу не добавляли
  if (is_without_street = 0 and
      has_any_no = 0) then
    is_without_street = 1;

  insert into addrlist (adl_from_pl_id, adl_depart_no, adl_depart_pref, adl_flat_no, adl_building_no,
                        adl_section_no, adl_address_str, adl_comment, adl_post_index, adl_is_edited, adl_from_un_id,
                        adl_street_id, adl_is_without_street, adl_lang_kind, adl_vladen_no)
  values (:pl_id, :depart_no, :depart_pref, :flat_no, :building_no, :section_no, :adl_address_str,
          :comment_str, :post_index, :is_edited, :un_id, :street_id, :is_without_street, :lang_kind, :vladen_no)
  returning adl_id
  into adl_id;

  if (post_index <> '' and
      street_id > 0 and
      is_without_street = 0) then
  begin
    update places
    set pl_post_index = :post_index
    where pl_id = :street_id and
          coalesce(pl_post_index, '') = '';
  end

  suspend;
end
