CREATE PROCEDURE MILLICARD_CLIENTS_FOR_ADD(
    PROJECT STR_100 = 'millicard_disb')
RETURNS (
    ESM_ID ID_TYPE,
    FRONTID ID_TYPE,
    SERIALNUM STR_100,
    FIRSTNAME STR_100,
    LASTNAME STR_100,
    MIDDLENAME STR_100,
    PASPNUM STR_100,
    PASPPIN STR_100,
    PASPNUM_ORIGINAL STR_100,
    GENDER SMALLINT)
AS
begin
  project = coalesce(project, '');
  if (project = '') then
    project = 'millicard_disb';

  for select em.esm_id, esm_object_id
      from extsysmain em
      where em.esm_project = :project and
            em.esm_action = 'addClient' and
            em.esm_status = 0
      into :esm_id, :frontid
  do
  begin
    for select fl.fl_name1, -- Фамилия
               fl.fl_name2, -- Имя
               fl.fl_name3, -- Отчество
               cl.c_pass_no, cl.c_pincode, iif(cl.c_sex = 1, 1, 2)
        from clients cl
        inner join fiolist fl on cl.c_from_fl_id = fl.fl_id
        where cl.c_id = :frontid
        into :lastname, :firstname, :middlename, :paspnum, :pasppin, :gender
    do
    begin
      paspnum_original = paspnum;

      if (paspnum starting 'AZE') then
      begin
        serialnum = left(paspnum, 3);
        paspnum = substring(paspnum from 4);
      end
      else
      begin
        serialnum = left(paspnum, 2);
        paspnum = substring(paspnum from 3);
      end

      /*
        frontid * String This parameter is defined by NBCO. Each new customer should be given a unique id. The length of Frontid should not be less than 4 characters. Example: 0000001.
        firstName * String Customer Name.
        lastName * String Customer's Last Name
        middleName* String Customer's  Father Name birth String Customer's
        birth date. Format: "dd /MM/yyyy". For example: 21/02/1989
        gender String 1 – male ;     2 - Female
        maritalStatus String 1 – single;    2 – married;
        mobilePhone String It is important to fill when the customer enables SMS banking service.
        email String Customer's email address
        serialNum * String Identity card serial number. For example: "AZE"
        paspNum * String Identity card number. For example: "11111111"
        paspDate String Date of issue of identity card. Format: " dd /MM/yyyy ". For example: "22/02/2012"
        paspExpiry String Expiry date of ID card. Format: " dd /MM/yyyy ". For example: "22/02/2032"
        paspOrg String Identity card issuer.
        country String Customer's country.
        city String Customer City
        address String Customer's registered address.
        paspPin * String Customer ID's PIN.     */

      suspend;
    end
  end
end
