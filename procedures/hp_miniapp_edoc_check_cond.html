CREATE PROCEDURE HP_MINIAPP_EDOC_CHECK_COND(
    MA_ID D_INTEGER)
RETURNS (
    RES_VALUE D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable K_ID D_ID;
declare variable PR_ID D_ID;
declare variable PR_CODE D_STR_100;
declare variable C_ID D_ID;
declare variable IS_PORUCH D_SMALLINT;
declare variable OTHER_C_ID D_ID;
declare variable REL_NAME D_STR_100;
declare variable DB_ID D_ID;
declare variable IS_OK smallint;
declare variable MA_DATE_OPER date;
begin
  res_text = '';
  res_value = 0;
  is_ok = 0;

  db_id = f_get_db_id();
  if (db_id not in (39801, 41701)) then
    exit;

  select pr.pr_code
  from product pr
  inner join miniapp ma on pr.pr_id = ma.ma_from_pr_id and
        ma.ma_id = :ma_id and
        current_date between ma.ma_date_app and ma.ma_date_end
  into pr_code;

  if ((:pr_code containing '.kontmat.') or (:pr_code containing '.idjara.') or (:pr_code containing '.fertilizer.') or (:pr_code containing '.selitra.')) then
  begin
    select k.k_id
    from kontract k
    where k.k_from_ma_app_id = :ma_id
    into k_id;

    if (k_id is null and
      not exists(select * from kontmat
      where kml_object_kind = f_objkind('miniapp', 'id')
      and kml_object_id = :ma_id))
    then
      res_text = 'Товар не найден';
    else
    if (not exists(select * from kontmat km
         where km.kml_object_kind = f_objkind('kontract', 'id') and
         km.kml_object_id = :k_id)  and
        not exists(select * from kontmat
        where kml_object_kind = f_objkind('miniapp', 'id')
        and kml_object_id = :ma_id)) then
    begin
      res_value = -1;
      res_text = 'Товар не найден';
      suspend;
      exit;
    end
    else
    begin
      select res_text
      from hp_verify_amount_goods(:k_id)
      into :res_text;
      if (:res_text <> '') then
      begin
        res_value = -1;
        suspend;
        exit;
      end
    end
  end
  if (not exists(select *
                 from miniapp ma
                 where ma.ma_is_active = 1 and
                       ma.ma_id = :ma_id and
                       ma.ma_date_end = '31.12.9999')) then
  begin
    res_value = -1;
    res_text = 'Заявка не действительная';
    suspend;
    exit;
  end
  if (exists(select *
             from miniappclient mac
             where mac.mac_from_ma_id = :ma_id and
                   mac.mac_is_poruch = -1)) then -- is_poruch = -1 - лицо дающее согласие
  begin
    for select mac.mac_from_c_id
        from miniappclient mac
        where mac.mac_from_ma_id = :ma_id and
              mac.mac_amount > 0
        into c_id
    do
    begin
      other_c_id = 0;
      rel_name = '';

      select crel_other_c_id, rel_name
      from hp_client_relation_list(:c_id, 0)
      where rel_name in ('супруг', 'супруга') --
            rows 1
      into other_c_id, rel_name;

      if (exists(select *
                 from miniappclient mac
                 where mac.mac_from_ma_id = :ma_id and
                       mac.mac_from_c_id = :other_c_id) and
          exists(select *
                 from hp_client_relation_list(:other_c_id, 0)
                 where crel_from_rel_id in ('1', '2'))) then
        is_ok = 1;
    end
    if (is_ok < 1) then
      res_text = 'Лицом дающее согласие должен быть супруг или супруга!';
  end

  for select mac.mac_from_c_id, ma.ma_date_oper
      from miniappclient mac
      inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id and
            ma.ma_date_end = '31.12.9999' and
            ma.ma_is_active = 1
      where mac.mac_from_ma_id = :ma_id and
            mac.mac_is_poruch <> -2
      into c_id, ma_date_oper
  do
  begin
    if (not exists(select *
                   from clientdocs cds
                   inner join passkind pk on cds.cds_from_pk_id = pk.pk_id
                   inner join passdoctype pdt on pk.pk_from_pdt_id = pdt.pdt_id and
                         pdt.pdt_code in ('contact_eds', 'emploee_eds') --  договор ЭЦП, договор ЭПЦ сотрудника
                   where cds.cds_from_c_id = :c_id and
                         coalesce(cds_doc_blob_1, 0) > 0 and
                         iif(pdt.pdt_code = 'contact_eds', coalesce(cds_doc_blob_2, 0) > 0, true) and --Если договор ЭЦП сотрудника, то пропустить проверку на cds_doc_blob_2
                         cds.cds_is_delete = 0 and
                         cds.cds_doc_date_begin <= :ma_date_oper and
                         cds.cds_doc_date_begin >= (select cast(ovs.ovs_datetime as date)
                                                    from objectprops opr
                                                    inner join objectvalues ovs on ovs.ovs_from_opr_id = opr.opr_id and
                                                          cast(ovs_datetime as d_date) <= :ma_date_oper and
                                                          ovs.ovs_is_active = 1 and
                                                          ovs.ovs_value <> '' and
                                                          ovs.ovs_add_value = 2
                                                    where opr.opr_object_kind = 0 and
                                                          opr.opr_object_id = :c_id and
                                                          opr.opr_subject_kind = -1 and
                                                          opr.opr_subject_id = f_get_rd_id('contactinfo.cell_main')
                                                    order by ovs_datetime desc
                                                    rows 1))) then
    begin
      if (f_get_const('asp_const') = 1 and f_asp_signed(:c_id, :ma_date_oper) = 1) then
        continue;
      else
        res_text = res_text || ' У клиента, не закреплен фото договора ЭЦП';
    end

  end
  suspend;
end
