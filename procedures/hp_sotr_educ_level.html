CREATE PROCEDURE HP_SOTR_EDUC_LEVEL(
    C_ID D_ID)
RETURNS (
    IS_MIDDLE D_INTEGER,
    ED_LEVEL D_STR_100,
    INST_NAME D_STR_100,
    SPEC_NAME D_STR_100,
    CS_DATE_BEGIN D_STR_100,
    CS_DATE_END D_STR_100)
AS
begin
  select iif(cs.cs_education_level in (19, 18), 1, 0), rd.rd_name
  from cstudy cs
  inner join refdata rd on rd.rd_id = cs.cs_education_level
  where cs.cs_from_c_id = :c_id and
        cs.cs_number = (select max(cs.cs_number)
                        from cstudy cs
                        where cs.cs_from_c_id = :c_id)
  into :is_middle, :ed_level;

  if (is_middle = 1) then
  begin
    suspend;
  end
  else
  begin
    for select cs.cs_inst_name, cs.cs_spec_name, cs.cs_year_begin, cs.cs_year_end
        from cstudy cs
        where cs.cs_from_c_id = :c_id
        into :inst_name, :spec_name, :cs_date_begin, :cs_date_end
    do
    begin
      suspend;
    end
  end
end
