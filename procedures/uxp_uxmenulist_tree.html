CREATE PROCEDURE UXP$UXMENULIST_TREE(
    PARENT_ID D_INTEGER = null)
RETURNS (
    ML_ID D_INTEGER,
    ML_PARENT_ID D_INTEGER,
    ML_CODE D_STR_CODE,
    ML_CAPTION D_STR_255,
    ML_INDEX D_INTEGER,
    ML_IS_FOLDER D_BOOL,
    ML_IS_UX D_BOOL,
    ML_UA_CODE D_STR_100,
    IMAGE_INDEX D_INTEGER,
    CAN_MOVE D_BOOL,
    ML_UA_UUID D_UUID,
    ML_UA_UUID_CODE D_STR_CODE,
    ML_FS_DISABLED D_BOOL,
    ML_WEB_DISABLED D_BOOL,
    ML_UXA_CAPTION D_STR_255,
    ML_POSITION D_INTEGER)
AS
declare variable tmp_parent d_integer;
declare variable tmp_index d_integer;
begin
  tmp_index = -1;
  for select ml_id, ml_code, ml_parent_id, ml_caption, ml_is_folder, ml_is_ux, ml_ua_code, ml_can_move,
             ml_ua_uuid, ml_ua_uuid_code, ml_fs_disabled, ml_web_disabled, ml_uxa_caption, ml_position
  from uxp$uxmenulist_tree_tmp(:parent_id)
  into ml_id, ml_code, ml_parent_id, ml_caption, ml_is_folder, ml_is_ux, ml_ua_code, can_move,
       ml_ua_uuid, ml_ua_uuid_code, ml_fs_disabled, ml_web_disabled, ml_uxa_caption, ml_position
  do
  begin
    tmp_index = tmp_index + 1;
    ml_index = tmp_index;

--  Иконки FinStart:
--    Folder     37
--    Item       20
--
--  Иконки Ux
--    Folder     36
--    Item       38 для окон

    if (ml_uxa_caption <> '') then
    begin
      if (ml_caption not containing ml_uxa_caption) then
        ml_uxa_caption = ml_uxa_caption || '  (' || ml_ua_code || ')';
      else
        ml_uxa_caption = ml_ua_code;
    end

    if (ml_is_ux = 0) then
    begin
      if (ml_is_folder = 1) then
        image_index = 37;
      else
        image_index = 101;
    end else
    begin
      if (ml_is_folder = 1) then
        image_index = 36;
      else
        image_index = 38; -- Форма
    end
    suspend;

    if (ml_is_folder = 1) then
    begin
      tmp_parent = ml_id;

      for select ml_id, ml_code, ml_parent_id, ml_caption, ml_index, ml_is_folder, ml_is_ux,
                 ml_ua_code, image_index, can_move, ml_ua_uuid, ml_ua_uuid_code, ml_fs_disabled, ml_web_disabled,
                 ml_uxa_caption, ml_position
      from uxp$uxmenulist_tree(:tmp_parent)
      into ml_id, ml_code, ml_parent_id, ml_caption, ml_index, ml_is_folder, ml_is_ux,
           ml_ua_code, image_index, can_move, ml_ua_uuid, ml_ua_uuid_code, ml_fs_disabled, ml_web_disabled,
           ml_uxa_caption, ml_position
      do
        suspend;
    end
  end
end
