CREATE PROCEDURE HP_CFP_ANKETA_SHOW(
    UN_ID D_ID,
    MA_ID D_INTEGER,
    LANG D_STR_LANG,
    K_DISB_DATE D_DATE)
RETURNS (
    RES_HTML D_STR_4096)
AS
declare variable MAC_ID D_INTEGER;
declare variable C_ID D_ID;
declare variable C_NAME_FIO D_STR_100;
declare variable DB_ID D_INTEGER;
declare variable CIB_IS_WORKING D_BOOL;
declare variable PR_CODE D_STR_100;
declare variable BASE_K_ID D_ID;
declare variable BASE_MA_ID D_INTEGER;
declare variable DISB_DATE D_DATE;
declare variable IS_SUSN D_SMALLINT; /* RFCKZ-996 */
declare variable SUSN_STATUS D_SMALLINT; /* RFCKZ-996 */
declare variable KDN_LIMIT D_CURRENCY; /* RFCKZ-996 */
declare variable CAN_EDIT D_SMALLINT;
declare variable ERR_TEXT D_STR_255;
begin
  db_id = f_get_db_id();
  res_html = '';
  can_edit = 1;

  select pr.pr_code,
         ma.ma_date_oper
  from miniapp ma
  inner join product pr on ma.ma_from_pr_id = pr.pr_id
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.2999'
  order by ma.ma_number desc
  rows 1
  into pr_code,
       disb_date;

  -- RFCKG-3574
  select cw.is_working
  from cib_working_status cw
  into cib_is_working;

  if (db_id = 41701 and
      pr_code containing '.insur_kontrakt.') then  -- RFCKG-5051 Для данного продукта не запрашивается История КИБ
  begin
    select lir.lir_kontract_base
    from kontract k
    inner join loan_insurance_relation lir on k.k_id = lir.lir_kontract_insurance
    where k.k_from_ma_app_id = :ma_id
    into base_k_id;

    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :base_k_id
    into base_ma_id;
  end

  for select mac.mac_id,
             mac.mac_from_c_id
      from miniappclient mac
      where mac.mac_from_ma_id = :ma_id and
            mac.mac_amount > 0
      into mac_id,
           c_id
  do
  begin
    select c.c_name_fio
    from clients c
    where c.c_id = :c_id
    into c_name_fio;

    -- RFCKZ-996: статус СУСН и предельное значение КДН для него
    is_susn = 0;
    susn_status = 0;
    kdn_limit = 0.5;

    if (db_id = 39801) then
    begin
      select res_status, is_susn
      from hp_susn_get_status(:c_id, :ma_id)
      into susn_status, is_susn;

      if (susn_status <> 2) then
        res_html = blob_append(res_html,
          '<span style="color: red;">Статус СУСН не получен</span><br>');

      if (is_susn = 1) then
        kdn_limit = 0.25;
    end

    select err.eir_answer
    from extinfolist eil
    inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
    inner join extinfoerr err on err.eir_from_eil_id = eil.eil_id
    where eic.eic_code = 'kdn_score' and               -- RFCKZ-871
          eil.eil_object_kind = f_objkind('miniappclient', 'id') and
          eil.eil_object_id = :mac_id and
          eil.eil_status = -2
    order by eil.eil_id desc
    rows 1
    into err_text;

    if (db_id = 39801 and exists (select *                             -- RFCKZ-871
                                  from extinfolist eil
                                  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                                  where eic.eic_code = 'kdn_score' and
                                        eil.eil_object_kind = f_objkind('miniappclient', 'id') and
                                        eil.eil_status in (-1, 0) and
                                        eil.eil_object_id = :mac_id)) then
      can_edit = 0;

    if (db_id = 39801 and not exists(select *
                                     from extinfolist eil
                                     where eil.eil_from_eic_id in (9, 10, 12) and -- RFCKZ-237
                                           eil.eil_object_kind = f_objkind('clients', 'id') and
                                           eil.eil_from_c_id = :c_id and
                                           eil.eil_object_id = :c_id and
                                           eil.eil_status = 2 and
                                           eil.eil_date = current_date) and not exists(select *
                                                                                       from clientdocs cds
                                                                                       inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
                                                                                       inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id and
                                                                                             pdt.pdt_code = 'card_statement'
                                                                                       where cds.cds_from_c_id = :c_id and
                                                                                             cds.cds_is_delete = 0 and
                                                                                             cds.cds_doc_date_begin = current_date)) then --RFCKZ-90
      res_html = blob_append(res_html, 'Для заполнения Анкеты необходимо прикрепить актуальную выписку по карте либо получить справку о (ЗП, пенсии, АСП) <br>');

    else
    if (db_id = 39801 and
       (not exists(select 1
                   from extinfolist eil
                   inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                   where eic.eic_code = 'pkb_available_reports' and  -- RFCKZ-1026
                         eil.eil_object_kind = f_objkind('miniapp', 'id') and
                         eil.eil_object_id = :ma_id and
                         eil.eil_status = 2)
        or
       ((select priority_id
         from hp_get_priority_rep_id(f_objkind('miniapp', 'id'), :ma_id)) <> -1
        and not exists(select 1
                       from extinfolist eil
                       inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                       where eic.eic_code = 'pkb_cib_report' and
                             eil.eil_object_kind = f_objkind('miniapp', 'id') and
                             eil.eil_object_id = :ma_id and
                             eil.eil_status = 2 and
                             eil.eil_from_bls_id is not null)))) then
      res_html = blob_append(res_html, 'Для заполнения Анкеты необходимо получить кредитный отчет (ответ от ПКБ) <br>');

    else
    if (db_id = 39801 and coalesce((select cak.cakdn_kdn
                                    from cfp_anketa_kdn cak
                                    where cak.cakdn_from_c_id = :c_id and
                                          cak.cakdn_from_mac_id = :mac_id), 1) <= :kdn_limit) then
    begin
      res_html = blob_append(res_html, '<span style="color: green;">Анкета доходов/расходов заполнена успешно!</span><br>');
      res_html = blob_append(res_html, f_uri_html(f_lang(lang, 'Просмотр'), 'cfp_anketa_info', f_lang(lang, 'Анализ доходов/расходов'), 'miniappclient', :mac_id, current_date, current_date, ''));
    end
    else
    begin
      res_html = blob_append(res_html, '<table width="100%">
       <col width="40%">
       <col width="30%">
       <col width="30%">
       <tr>');

      if (exists(select * /* 'Не заполнен ' ||*/
                 from cfp_anketa ca
                 where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
                       ca.ca_object_id = :mac_id and
                       ca.ca_from_c_id = :c_id)) then
        -- RFCKG-3249
        res_html = blob_append(res_html, f_html_table_row(:c_name_fio, 'Заполнен<br>' || f_uri_html(f_lang(lang, 'Просмотр'), 'cfp_anketa_info', f_lang(lang, 'Анализ доходов/расходов'), 'miniappclient', :mac_id, current_date, current_date, ''), iif(k_disb_date < current_date, '', iif(can_edit = 0, '',  f_uri_open(f_lang(lang, 'Изменить'), 'cli_financial_profile', f_objkind('miniappclient', 'id'), :mac_id, null, null, un_id))), '', -1, -1, -1, -1, 3, ''));
      else
      begin

        if (:db_id <> 41701 and cib_is_working = 0) then
          res_html = blob_append(res_html, f_html_table_row(:c_name_fio, 'Не заполнен', f_uri_open(f_lang(lang, 'Заполнить'), 'cli_financial_profile', f_objkind('miniappclient', 'id'), :mac_id, null, null, un_id), '', -1, -1, -1, -1, 3, ''));

        else
        if (exists(select 1
                   from kibminiapp kbm
                   where kbm.kbm_from_c_id = :c_id and
                         cast(kbm.kbm_checked_datetime as d_date) between :disb_date - 1 and :disb_date and
                   kbm.kbm_is_checked = 1)) then
        begin
          res_html = blob_append(res_html, f_html_table_row(:c_name_fio, 'Не заполнен', f_uri_open(f_lang(lang, 'Заполнить'), 'cli_financial_profile', f_objkind('miniappclient', 'id'), :mac_id, null, null, un_id), '', -1, -1, -1, -1, 3, ''));
        end
        else if (:db_id = 41701 and exists (select 1
                                            from extinfolist eil
                                            inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
                                            where ((eil.eil_date between :disb_date - 1 and :disb_date) or (eil.eil_object_kind = f_objkind('miniapp', 'id') and  eil.eil_object_id = :ma_id)) and  -- RFCKG-5396
                                                   eil.eil_from_c_id = :c_id and
                                                   eic.eic_code = 'ss_cib_report' and
                                                   eil.eil_status = 2)) then
        begin
          res_html = blob_append(res_html, f_html_table_row(:c_name_fio, 'Не заполнен', f_uri_open(f_lang(lang, 'Заполнить'), 'cli_financial_profile', f_objkind('miniappclient', 'id'), :mac_id, null, null, un_id), '', -1, -1, -1, -1, 3, ''));
        end
      end
    end
    res_html = blob_append(res_html, '</table>');

    if (coalesce(:err_text, '') <> '' and :err_text <> 'OK') then
      res_html = blob_append(res_html, f_format('<p style="color: red; margin: 5px 0;">%1</p>', :err_text));
  end
  suspend;
end
