CREATE PROCEDURE HP_REPAYMENT_DETAIL_PAYS_KZ(
    K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    OK_SH_DATE D_DATE,
    OK_OS D_CURRENCY,
    OK_PERC D_CURRENCY,
    OK_FINE D_CURRENCY,
    SS D_CURRENCY)
AS
begin
  for select ok_ol_date,
             abs(sum(ok_os + ok_sh_os)),
             abs(sum(ok_perc + ok_sh_perc)),
             abs(sum(ok_fine)),
             abs(sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc + ok_fine))
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_ol_date <= :rep_date and
            ok_type = 4
      group by ok_ol_date, ok_sh_date
      order by ok_ol_date, ok_sh_date
      into :ok_sh_date, :ok_os, :ok_perc, :ok_fine, :ss
  do
    suspend;
end
