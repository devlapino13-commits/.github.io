CREATE PROCEDURE IPC_QR_PAYMENTS_OL_CREATE_ONE(
    IQP_ID D_INTEGER)
AS
declare variable acc_dt d_str_20;
declare variable id_no d_id;
declare variable ol_id d_id;
declare variable ol_doc_no d_integer;
declare variable amount d_currency;
declare variable iqp_check_number d_str_100;
declare variable iqp_ipc_datetime d_datetime;
declare variable acc_kt d_str_20;
declare variable office_id d_integer;
declare variable oper_date d_date;
declare variable c_id d_id;
declare variable doc_kind d_smallint;
begin
    for select iqp_amount,
               iqp.iqp_check_number,
               iqp_ipc_datetime,
               ag.ag_from_al_id,
               ag.ag_object_id
    from ipc_qr_payments iqp
    inner join agreements ag on ag.ag_id = iqp.iqp_from_ag_id
    where iqp.iqp_id = :iqp_id
    and iqp.iqp_state = 0
    and iqp.iqp_from_cs_id > 0
    into :amount,
         :iqp_check_number,
         :iqp_ipc_datetime,
         :acc_kt,
         :c_id
    do
    begin
        oper_date = cast(iqp_ipc_datetime as date);
        office_id = 1090100;

        acc_dt = (select al_id from acc_list where al_from_reg_id = :office_id and al_currency = 417 and al_account = '10024/1'); -- единый котловой счет в Бишкек ГО

        select kv
        from get_kv(:office_id, 5)
        into :id_no;
        
        ol_id = :id_no;
        doc_kind = 9; -- 9 - т.к. нас интерсует мемориальный ордер
        
        -- очередной номер документа
        select num
        from last_doc_number(:oper_date, :doc_kind, :office_id)
        into :ol_doc_no;
        
        insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete,
                              ol_user_id, ol_document_client, ol_doc_no, ol_doc_no_str)
        values (:ol_id, :office_id, 2, :oper_date, :doc_kind,
                'Покупка по QR', 0, 0, 1000000000000, :c_id, :ol_doc_no, :ol_doc_no);
        
        insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                              ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                              ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
        values (:ol_id, 1, '417', 1, null, :acc_dt, :amount, :amount, null, null, null, :acc_kt, :c_id,
                f_format('Номер чека:%1; от: %2; ', :iqp_check_number,  f_datetimetostr(:iqp_ipc_datetime)), null, :oper_date, 1, :office_id);

        update ipc_qr_payments iqp
        set iqp.iqp_state = 1,
            iqp.iqp_from_ol_id = :ol_id
        where iqp.iqp_id = :iqp_id;
    end
end
