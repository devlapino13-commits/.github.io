Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXFRAME_SQL(
    FR_UUID D_UUID,
    FR_UUID_CODE D_STR_CODE = null)
RETURNS (
    UFR_IS_GRID D_BOOL,
    UFR_CAN_UPDATE D_BOOL,
    UFR_CAN_INSERT D_BOOL,
    UFR_CAN_DELETE D_BOOL,
    UFR_KEY_FIELDS D_STR_100,
    UFR_PARAM_SQL D_STR_8100,
    UFR_SELECT_SQL D_STR_8100,
    UFR_UPDATE_SQL D_STR_8100,
    UFR_INSERT_SQL D_STR_8100,
    UFR_DELETE_SQL D_STR_8100,
    UFR_HAS_EXEC_PROC D_BOOL,
    UFR_SELECT_IS_EXEC D_BOOL,
    UFR_AUTOINCFIELD D_STR_CODE,
    UFR_TREE_MOVE_SQL D_STR_8100)
AS
declare variable exec_proc d_bool;
begin
  if (coalesce(fr_uuid, '') = '') then
    fr_uuid = uxf$uniq(fr_uuid_code);

  ufr_select_sql = '';
  ufr_update_sql = '';
  ufr_insert_sql = '';
  ufr_delete_sql = '';
  ufr_tree_move_sql = '';



  ufr_has_exec_proc = 0;
  ufr_select_is_exec = 0;

  for select ufr_is_grid, ufr_can_update, ufr_can_insert, ufr_can_delete, us_keyfields, us_autoincfield
  from ux$frame, ux$source
  where ufr_uuid = :fr_uuid and
        ufr_us_uuid = us_uuid
  into :ufr_is_grid, :ufr_can_update, :ufr_can_insert, :ufr_can_delete, :ufr_key_fields, ufr_autoincfield
  do
  begin
    if (ufr_key_fields is null) then
      ufr_key_fields = '';
    ufr_param_sql = '';

    select sql_text
    from uxp$uxframe_sql_param(:fr_uuid, 0)
    into ufr_param_sql;

    select sql_text, sql_is_exec
    from uxp$uxframe_sql_select(:fr_uuid)
    into ufr_select_sql, ufr_select_is_exec;

    if (ufr_can_update = 1) then
    begin
      select sql_text, exec_proc
      from uxp$uxframe_sql_modify(:fr_uuid, 1)
      into :ufr_update_sql, exec_proc;

      if (ufr_update_sql <> '' and exec_proc = 1) then
        ufr_has_exec_proc = 1;
    end

    if (ufr_can_insert = 1) then
    begin
      select sql_text, exec_proc
      from uxp$uxframe_sql_modify(:fr_uuid, 2)
      into :ufr_insert_sql, exec_proc;

      if (ufr_insert_sql <> '' and exec_proc = 1) then
        ufr_has_exec_proc = 1;
    end

    if (ufr_can_delete = 1) then
    begin
      select sql_text, exec_proc
      from uxp$uxframe_sql_modify(:fr_uuid, 3)
      into :ufr_delete_sql, exec_proc;

      if (ufr_delete_sql <> '' and exec_proc = 1) then
        ufr_has_exec_proc = 1;
    end

    if (ufr_can_update = 1 or ufr_can_insert = 1 or ufr_can_delete = 1) then
    begin
      select res_tree_move_sql
      from uxp$uxframe_sql_tree_move(:fr_uuid, false)
      into :ufr_tree_move_sql;
    end

    suspend;
  end
end
