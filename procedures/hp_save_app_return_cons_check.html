CREATE PROCEDURE HP_SAVE_APP_RETURN_CONS_CHECK(
    C_ID D_ID,
    K_ID D_ID,
    ARC_DATE DATE,
    OL_DATE DATE,
    ARC_AMOUNT D_CURRENCY,
    COUNT_ INTEGER,
    OL_DATE_RETURN DATE)
RETURNS (
    RES_TXT D_STR_255)
AS
declare variable k_no d_str_20;
declare variable days integer;
declare variable c_fio d_str_100;
begin
  res_txt = '';

  if (:count_ > 1) then
  begin
    if (:arc_amount = 0) then
    begin
      res_txt = 'По выбранному договору возврат средств был произведен ' || :ol_date_return;
    end
    else
    begin
      res_txt = 'По выбранному договору разница составляет ' || :arc_amount;
    end
    suspend;
    exit;
  end

  if (k_id is not null and
      (exists(select *
              from app_return_consult arc
              where arc.arc_from_c_id = :c_id and
                    arc.arc_from_k_id = :k_id and
                    arc.arc_status <> -2))) then
  begin
    select k.k_kontract_no
    from kontract k
    where k.k_id = :k_id
    into :k_no;
    res_txt = 'Заявление по договору ' || :k_no || ' уже было зарегистрировано.';
    suspend;
    exit;
  end

  if (k_id is null and
      (exists(select *
              from app_return_consult arc
              where arc.arc_from_c_id = :c_id))) then
  begin
    select c.c_name_fio
    from clients c
    where c.c_id = :c_id
    into :c_fio;
    res_txt = 'Заявление на имя ' || :c_fio || ' уже было зарегистрировано.';
    suspend;
    exit;
  end

  days = :arc_date - :ol_date;
  if (days > 14) then
  begin
    res_txt = 'Возврат возможен, если с момента оплаты прошло менее 14-ти дней. По выбранному договору оплата производилась ' || days || ' дня(ей) назад.';
    suspend;
    exit;
  end

  if (coalesce(:arc_amount, 0) <= 0) then
  begin
    res_txt = 'Cумма возврата не может быть меньше или равной 0.';
    suspend;
    exit;
  end

  suspend;

end
