CREATE PROCEDURE CM_KG_0012_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable rd_code str_255;
declare variable business_type integer;
declare variable busuness_subtype integer;
declare variable pr_code d_str_100;
begin
  for select rd.rd_code,
             kncl.kc_business_type,
             kncl.kc_busuness_subtype,
             cl.c_name_fio,
             pr.pr_code
  from knclient kncl
  inner join clients cl on cl.c_id = kncl.kc_from_c_id
  inner join refdata rd on rd.rd_id = kncl.kc_loan_target
  inner join kontract kt on kt.k_id = kncl.kc_from_k_id
  inner join product pr on pr.pr_id = kt.k_from_pr_id
  where kncl.kc_from_k_id = :k_id
  and kncl.kc_amount_approved >0
  into :rd_code,
       :business_type,
       :busuness_subtype,
       :res_info_text,
       :pr_code
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;

        if (current_date <= cast('31.07.2015' as date)) then
        begin
            if (rd_code = 'Loan_Target.Consumer') then -- Потребительский
            begin
                res_error_text='Клиент: '||:res_info_text||', до конца месяца (31.07.2015), "Потребительские" кредиты не выдаются';
                res_error_is_critical=1;
                suspend;
            end
        end
    end
end
