CREATE PROCEDURE HP_PROC_DEPENDED_TABLES(
    PROC_NAME D_STR_CODE)
RETURNS (
    RES_TABLE_NAME D_STR_CODE)
AS
declare variable depend_type d_integer;
declare variable tmp_source d_str_code;
begin
  proc_name = trim(upper(proc_name));

  for select distinct rdb$depended_on_name, rdb$depended_on_type
  from rdb$dependencies
  where rdb$dependent_name = :proc_name and
        rdb$package_name is null and
        rdb$depended_on_type in (0, 5)
  into tmp_source, depend_type
  do
  begin
    tmp_source = trim(tmp_source);

    if (depend_type = 0) then
    begin
      res_table_name = tmp_source;
      suspend;
    end
    else
    begin
      for select res_table_name
      from hp_proc_depended_tables(:tmp_source)
      into :res_table_name
      do
        suspend;
    end
  end
end
