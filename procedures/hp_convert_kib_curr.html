CREATE PROCEDURE HP_CONVERT_KIB_CURR(
    KH_CURRENCY D_STR_CODE)
RETURNS (
    CU_ID D_INTEGER,
    CURR_RATE D_CURRENCY)
AS
begin
    -- RFCKG-3810
    cu_id =
    case
        when kh_currency in ('Сом (Кыргызский)', 'KGS') then 417
        when kh_currency in ('Доллар США', 'USD') then 840
        when kh_currency in ('Евро', 'EUR') then 978
        when kh_currency in ('Российский рубл', 'RUB') then 810
        else 0
    end;

    select curr_rate
    from hp_get_curr_rate(:cu_id, current_date)
    into :curr_rate;

    suspend;
end
