CREATE PROCEDURE AAREP_CLIENT_CHANGED_PNUMBER(
    REG_ID D_INTEGER,
    DATE_BEG D_DATE,
    DATE_END D_DATE)
RETURNS (
    RES_REG_NAME D_STR_CODE,
    RES_C_NAME_VIPS D_STR_100,
    RES_KF_NAME D_STR_100,
    RES_K_ID D_ID,
    RES_KONTRACT_NO D_STR_20,
    RES_C_ID D_ID,
    RES_C_NAME_FIO D_STR_100,
    RES_DATE_SAVE D_DATETIME,
    RES_DISB_DATE D_DATE,
    RES_CHANGE_DATE D_DATETIME,
    RES_CELL_NO D_STR_255,
    RES_CELL_NO_NEW D_STR_255,
    RES_PREV_OWNER D_STR_255,
    PR_NAME VARCHAR(50),
    PORUCH_FIO D_STR_100,
    EDS_STATUS_DATETIME_PRIMARY D_DATE,
    EDS_STATUS_DATETIME_ACTUAL D_DATE)
AS
declare variable ma_id d_id;
declare variable rd_id d_integer;
declare variable ovs_datetime d_datetime;
declare variable c_prev_id d_id;
declare variable poruch_c_id d_id;
begin
  rd_id = f_get_rd_id('ContactInfo.Cell_Main');
  for
    select
      k_id,
      k_from_ma_app_id,
      reg_name,
      c1.c_name_fio, 
      kal_new_kf_name,
      k_kontract_no,
      c2.c_id, 
      c2.c_name_fio,
      k_disb_date,
      pr_name,
      mac_from_c_id
    from kontstate
      inner join kontract
        on k_id = ks_from_k_id
      inner join knaktlist
        on kal_from_kf_id = k_from_kf_id
          and :date_end between kal_date and kal_date_end
      inner join clients c1
        on c1.c_id = kal_new_kf_officer
      inner join knclient
        on kc_from_k_id = k_id
          and kc_amount_approved > 0
      inner join clients c2
        on c2.c_id = kc_from_c_id
      inner join miniappclient
        on mac_from_ma_id = k_from_ma_app_id
      inner join region_list(:reg_id, 1, 3, 1) rl
        on rl.id_list = ks_region_id
      inner join region
        on reg_unicode = ks_region_id
      inner join product
        on pr_id = k_from_pr_id
    where ks_date_begin between :date_beg and :date_end
      and ks_state = 1
      and ks_ol_is_delete = 0
      and mac_is_poruch = -1
  into
    :res_k_id,
    :ma_id,
    :res_reg_name,
    :res_c_name_vips,
    :res_kf_name,
    :res_kontract_no,
    :res_c_id,
    :res_c_name_fio,
    :res_disb_date,
    :pr_name,
    :poruch_c_id
  do
  begin
    select first 1
      ma_date_save
    from miniapp
    where ma_id = :ma_id
      and ma_is_active = 1
    order by ma_number
    into
      :res_date_save;

    poruch_fio = '';
    select c_name_fio
    from clients
    where c_id = :poruch_c_id
    into :poruch_fio;

    eds_status_datetime_primary = null;
    select first 1 cast(eds.eds_status_datetime as date)
    from edoc_list edl
    inner join edoc_sign eds on edl.edl_id = eds.eds_from_edl_id
    where eds.eds_from_c_id = :res_c_id
          and eds.eds_status = 1
    into :eds_status_datetime_primary;

    eds_status_datetime_actual = null;
    select eds.eds_status_datetime
    from edoc_list edl
    inner join edoc_sign eds on edl.edl_id = eds.eds_from_edl_id
    where eds.eds_from_c_id = :res_c_id
          and eds.eds_status = 1
    order by eds.eds_status_datetime desc
    rows 1
    into :eds_status_datetime_actual;

    -- номер на дату заявки
    res_cell_no = null;
    select first 1
      ovs_value,
      ovs_datetime
    from objectprops
      inner join objectvalues
        on ovs_from_opr_id = opr_id
          and ovs_datetime <= :res_date_save
          and ovs_is_active = 1
          and ovs_value <> ''
          and ovs_add_value = 2
    where opr_object_kind = '0'
      and opr_object_id = :res_c_id
      and opr_subject_kind = '-1'
      and opr_subject_id = :rd_id
    order by ovs_datetime desc
    into
      :res_cell_no,
      :ovs_datetime;

    -- прежний владелец номера
    c_prev_id = (
      select first 1
        cda_from_c_id
      from contdata
        inner join contlist
          on cda_from_clt_id = clt_id
            and clt_code = :res_cell_no
      where cda_date < :ovs_datetime
        and cda_from_c_id <> :res_c_id
      order by cda_date desc);
    res_prev_owner = (
      select
        c_name_fio || ', ' || rel_name
      from hp_client_relation_list(:res_c_id , 0)
      where crel_other_c_id = :c_prev_id);

    -- измененный номер
    res_cell_no_new = null;
    res_change_date = null;
    select first 1
      ovs_value,
      ovs_datetime
    from objectprops
      inner join objectvalues
        on ovs_from_opr_id = opr_id
          and ovs_datetime > :ovs_datetime and ovs_datetime <= (:res_disb_date + 1)
          and ovs_is_active = 1
          and ovs_value not in ('', :res_cell_no)
          and ovs_add_value = 2
    where opr_object_kind = '0'
      and opr_object_id = :res_c_id
      and opr_subject_kind = '-1'
      and opr_subject_id = :rd_id
    order by ovs_datetime desc
    into
      :res_cell_no_new,
      :res_change_date;
    if (res_change_date is not null)
      then suspend;
  end
end
