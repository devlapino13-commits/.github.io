CREATE PROCEDURE AAREP_CALC_PERC_FILL_LIST(
    REPORT_REGION_ID D_INTEGER,
    REPORT_YEAR D_INTEGER,
    REPORT_MONTH D_INTEGER)
AS
declare variable db_id d_integer;
declare variable k_kredline_end d_date;
declare variable closed_date d_date;
declare variable k_zero_balance_date d_date;
declare variable rep_date_begin d_date;
declare variable rep_date_end d_date;
declare variable reg_id d_integer;
declare variable rep_period d_integer;
declare variable rep_period_prior d_integer;
declare variable aarep_kommis_begin d_currency = 0;
declare variable aarep_kommis_end d_currency = 0;
declare variable aarep_kommis_period d_currency = 0;
declare variable aarep_kommis_payed d_currency = 0;
declare variable reg_id_prior d_integer;
declare variable perc_repay d_currency = 0;
declare variable perc_writeoff d_currency = 0;
declare variable kommis_repay d_currency = 0;
declare variable tmp_period d_integer;
declare variable sa_account d_str_20;
declare variable delta d_integer;
declare variable pr_id d_integer;
declare variable pr_id_prior d_integer;
declare variable aarep_calc_method d_integer;
declare variable aarep_k_id d_id;
declare variable aarep_perc_begin d_currency;
declare variable aarep_perc_calced d_currency;
declare variable aarep_perc_payed d_currency;
declare variable aarep_perc_writeoff d_currency;
declare variable aarep_perc_end d_currency;
declare variable aarep_percent_per_day d_percent;
declare variable k_stop_perc d_date;
declare variable k_id d_id;
declare variable aarep_out_perc_end d_currency = 0;
declare variable aarep_out_perc_begin d_currency = 0;
declare variable aarep_out_perc_calced d_currency = 0;
declare variable loan_kc_list d_str_255;
declare variable loan_amount_list d_str_255;
declare variable loan_kommis_list d_str_255;
declare variable k_disb_date d_date;
declare variable dtbeg d_date;
declare variable dtend d_date;
declare variable dated d_date;
declare variable k_amount_approved d_currency;
declare variable nday d_integer;
declare variable lday boolean;
declare variable period_beg d_integer;
declare variable period_end d_integer;
declare variable delayed_perc d_currency;
declare variable is_mbh boolean;
begin
  db_id = f_get_db_id();
  pr_id_prior = 0;

  -- очистка временной таблицы
  delete from temp_kontcalcperc
  where 1 = 1;

  rep_period = (:report_year * 100 + :report_month);
  if (report_month > 1) then
    rep_period_prior = :report_year * 100 + :report_month - 1;
  else
    rep_period_prior = (:report_year - 1) * 100 + 12;

  if (report_month < 10) then
    rep_date_begin = '01.0' || :report_month || '.' || :report_year;
  else
    rep_date_begin = '01.' || :report_month || '.' || :report_year;
  rep_date_end = dateadd(month, 1, :rep_date_begin) - 1;
  dtbeg = rep_date_begin;
  dtend = rep_date_end;

  if (rep_date_end > current_date) then
    rep_date_end = current_date;

  -- очистка данных текущего месяца, ниже будет создаваться заново
  for
    select
      id_list
    from region_list(:report_region_id, 1, 3, 1)
    into
      :reg_id
  do
  begin
    delete from kontcalcperc
    where kcp_period = :rep_period
      and kcp_from_reg_id = :reg_id;

    delete from kn_kontcalcperc
    where kcp_period = :rep_period
      and kcp_from_reg_id = :reg_id;

    -- заполнение остатка на начало периода
    insert into temp_kontcalcperc (
      kcp_from_k_id, kcp_from_reg_id, kcp_perc_saldo_begin, kcp_perc_saldo_end,
      kcp_out_perc_saldo_begin, kcp_out_perc_saldo_end, kcp_kommis_saldo_begin,
      kcp_kommis_saldo_end, kcp_last_reg_id, kcp_delayed_perc)
    select
      kcp_from_k_id, kal_new_reg_id, kcp_perc_saldo, kcp_perc_saldo,
      kcp_out_perc_saldo, kcp_out_perc_saldo, kcp_kommis_saldo,
      kcp_kommis_saldo, kcp_from_reg_id, kcp_delayed_perc_end
    from knaktlist
      inner join kontract
        on k_from_kf_id = kal_from_kf_id
      inner join kontcalcperc
        on kcp_from_k_id = k_id
          and kcp_period = :rep_period_prior
          and kcp_from_c_id = 0
    where :rep_date_end between kal_date and kal_date_end
      and kal_new_reg_id = :reg_id
      and is_delete = 0;

    for
      select
        k_id
      from knaktlist
        inner join kontract
          on k_from_kf_id = kal_from_kf_id
        inner join kontstate
          on ks_from_k_id = k_id
            and ks_date_begin between :rep_date_begin and :rep_date_end
            and ks_ol_is_delete = 0
            and ks_state = 2
      where :rep_date_end between kal_date and kal_date_end
        and kal_new_reg_id = :reg_id
        and is_delete = 0
    into
      :k_id
    do
    begin
      update temp_kontcalcperc set
        kcp_perc_saldo_end = 0,
        kcp_out_perc_saldo_end = 0,
        kcp_kommis_saldo_end = 0
      where kcp_from_k_id = :k_id
        and (kcp_perc_saldo_end <> 0
          or kcp_out_perc_saldo_end <> 0
          or kcp_kommis_saldo_end <> 0);
    end

    -- список кредитов по которым нужно начислить проценты
    for
      select
        k_id,
        k_disb_date,
        k_amount_approved,
        kal_new_reg_id,
        k_percent_per_day,
        k_stop_perc,
        k_kredline_end,
        minvalue(k_kredline_end, coalesce(k_fact_closed_date, '31.12.9999')),
        k_zero_balance_date,
        k_from_pr_id,
        :db_id in (39801, 41701) and :rep_period >= 202308 and pr_code containing '.murabaha.'
      from knaktlist
        inner join kontract
          on k_from_kf_id = kal_from_kf_id
        inner join product
          on pr_id = k_from_pr_id
        inner join kontstate
          on ks_from_k_id = k_id
            and ks_date_begin <= :rep_date_end and ks_date_end >= (:rep_date_begin - 1)
            and ks_ol_is_delete = 0
            and ks_state = 1
      where :rep_date_end between kal_date and kal_date_end
        and kal_new_reg_id = :reg_id
        and knaktlist.is_delete = 0
    into
      :aarep_k_id,
      :k_disb_date,
      :k_amount_approved,
      :reg_id,
      :aarep_percent_per_day,
      :k_stop_perc,
      :k_kredline_end,
      :closed_date,
      :k_zero_balance_date,
      :pr_id,
      :is_mbh
    do
    begin
      aarep_perc_calced = null;

      -- амортизация доп.процента
      select res_amount_begin, res_amount_end, res_amount_period, res_kommis_payed
      from aarep_calc_amort_kommis_temp(:aarep_k_id, :rep_date_begin, :rep_date_end, :db_id)
      into :aarep_kommis_begin, :aarep_kommis_end, :aarep_kommis_period, :aarep_kommis_payed;

      if (pr_id <> pr_id_prior) then
      begin
        select
          pr_interest_method
        from product
        where pr_id = :pr_id
        into
          :aarep_calc_method;
      end
      -- если предыдущий месяц заполнен, дополняем данные на конец периода
      if ((report_year * 100 + report_month) <= 201303) then
      begin
        select result_payed, result_writeoff, result_perc_end, 0
        from aarep_calc_perc_one_201302(:aarep_k_id, :rep_date_begin, :rep_date_end, :aarep_percent_per_day, :aarep_calc_method, 0, 2, :k_stop_perc)
        into :aarep_perc_payed, :aarep_perc_writeoff, :aarep_perc_end, :aarep_out_perc_end;
      end else
      begin
        if (f_new_perc_active(:rep_date_end) = 1) then
        begin
          select result_payed, result_writeoff, result_perc_end, result_out_perc_end
          from aarep_calc_perc_one_saldo(:aarep_k_id, :rep_date_begin, :rep_date_end)
          into :aarep_perc_payed, :aarep_perc_writeoff, :aarep_perc_end, :aarep_out_perc_end;
        end else
        begin
          select result_payed, result_writeoff, result_perc_end, result_out_perc_end
          from aarep_calc_perc_one(:aarep_k_id, :rep_date_begin, :rep_date_end, 2, :db_id, :k_kredline_end, :k_stop_perc, :k_zero_balance_date)
          into :aarep_perc_payed, :aarep_perc_writeoff, :aarep_perc_end, :aarep_out_perc_end;
        end

        if (is_mbh) then
        begin
          period_beg = extract(year from k_disb_date)*100 + extract(month from k_disb_date);
          period_end = extract(year from closed_date)*100 + extract(month from closed_date);
          delayed_perc = coalesce((
            select
              kcp_delayed_perc_end
            from kontcalcperc
            where kcp_period = :rep_period_prior
              and kcp_from_k_id = :aarep_k_id), 0);
          dated = (
            select rep_delinq_begin + 90
            from aarep_kont_delinq_status(:aarep_k_id, :dtend)
            where rep_delinq_days >= 90);

          if (dated <= dtbeg) -- просрочка более 90дн.
            then aarep_perc_calced = 0;
          else
          if (period_end <= rep_period) -- период закрытия займа
            then aarep_perc_calced = delayed_perc;
          else
          begin
            if (period_beg = rep_period) then
            begin
              nday = extract(day from k_disb_date);
              -- последний день месяца
              lday = extract(day from dateadd(month, 1, k_disb_date - nday + 1) - 1) = nday;
              nday = iif(lday or nday >= 30, 0, 30 - nday);
            end else
            begin
              if (dated is null or dated >= dtend)
                then nday = 30;
              else
                nday = extract(day from dated) - 1;
            end
            aarep_perc_calced = round(k_amount_approved * nday * aarep_percent_per_day / 100, 0);
            if (rep_period > period_beg and aarep_perc_calced > delayed_perc)
              then aarep_perc_calced = delayed_perc;
          end
        end else
          aarep_perc_calced = null;
      end

      update or insert into temp_kontcalcperc (
        kcp_from_k_id, kcp_from_reg_id, kcp_perc_calc, kcp_perc_payed, kcp_perc_writeoff,
        kcp_perc_saldo_end, kcp_out_perc_saldo_end, kcp_kommis_payed, kcp_kommis_saldo_end)
      values (
        :aarep_k_id, :reg_id, :aarep_perc_calced, :aarep_perc_payed, :aarep_perc_writeoff,
        :aarep_perc_end, :aarep_out_perc_end, :aarep_kommis_payed, :aarep_kommis_end)
      matching (kcp_from_k_id);
    end

    -- временная таблица заполнена, переносим в боевую таблицу.
    insert into kontcalcperc (
      kcp_period, kcp_from_k_id, kcp_from_c_id, kcp_from_reg_id, kcp_perc_calced,
      kcp_perc_saldo, kcp_perc_payed, kcp_perc_writeoff, kcp_perc_begin_saldo,
      kcp_transfer_reg_id, kcp_kommis_calced, kcp_kommis_saldo, kcp_kommis_payed,
      kcp_kommis_begin_saldo, kcp_out_perc_calced, kcp_out_perc_saldo, kcp_out_perc_begin_saldo,
      kcp_delayed_perc_begin)
    select
      :rep_period, kcp_from_k_id, 0, kcp_from_reg_id, kcp_perc_calc,
      kcp_perc_saldo_end, kcp_perc_payed, kcp_perc_writeoff, kcp_perc_saldo_begin,
      kcp_last_reg_id, kcp_kommis_calc, kcp_kommis_saldo_end, kcp_kommis_payed,
      kcp_kommis_saldo_begin, kcp_out_perc_calc, kcp_out_perc_saldo_end, kcp_out_perc_saldo_begin,
      kcp_delayed_perc
    from temp_kontcalcperc
    where kcp_from_reg_id = :reg_id;

    delete from temp_kontcalcperc
    where 1 = 1;

    for
      select
        kcp_period,
        kcp_from_k_id,
        kcp_from_reg_id,
        kcp_transfer_reg_id
      from kontcalcperc
      where kcp_period = :rep_period
        and kcp_from_reg_id = :reg_id
        and kcp_from_reg_id <> kcp_transfer_reg_id
    into
      :tmp_period,
      :aarep_k_id,
      :reg_id,
      :reg_id_prior
    do
    begin
      perc_repay = 0;
      perc_writeoff = 0;
      kommis_repay = 0;
      select
        sum(iif(ok_type = 4, -ok_perc, 0)),
        sum(iif(ok_type in (5, 6), -ok_perc, 0)),
        sum(iif(ok_type = 4, -ok_kommis, 0))
      from operkred
      where ok_from_k_id = :aarep_k_id
        and ok_ol_is_active = 1
        and ok_ol_date between :rep_date_begin and :rep_date_end
        and ok_ol_reg_id = :reg_id_prior
        and ok_from_c_id = 0
      into
        :perc_repay,
        :perc_writeoff,
        :kommis_repay;
      perc_repay = coalesce(perc_repay, 0);
      perc_writeoff = coalesce(perc_writeoff, 0);
      kommis_repay = coalesce(kommis_repay, 0);

      -- проценты и комиссия, выплаченная или списанная в предыдущем офисе определены,
      -- теперь нужно закинуть в kontcalc_perc
      if (perc_repay <> 0 or perc_writeoff <> 0 or kommis_repay <> 0) then
      begin
        update kontcalcperc set
          kcp_perc_transfer_payed = :perc_repay,
          kcp_perc_transfer_writeoff = :perc_writeoff,
          kcp_kommis_transfer_payed = :kommis_repay
        where kcp_from_k_id = :aarep_k_id
          and kcp_period = :tmp_period
          and kcp_from_c_id = 0;
      end
    end

    -- закрепление бух.счета по кредитам
    for
      select
        k_id,
        sa_account
      from kontcalcperc
        inner join kontract
          on k_id = kcp_from_k_id
        inner join standacc
          on sa_from_reg_id = kcp_from_reg_id
            and sa_from_pr_id = k_from_pr_id
            and sa_from_f_id = k_from_f_id
            and sa_account_type = 2
      where kcp_period = :rep_period
        and kcp_from_reg_id = :reg_id
    into
      :aarep_k_id,
      :sa_account
    do
    begin
      update kontcalcperc set
        kcp_perc_account = :sa_account
      where kcp_period = :rep_period
        and kcp_from_k_id = :aarep_k_id
        and kcp_from_c_id = 0;
    end

    for
      select
        k_id,
        sa_account
      from kontcalcperc
        inner join kontract
          on k_id = kcp_from_k_id
        inner join standacc
          on sa_from_reg_id = kcp_from_reg_id
            and sa_from_pr_id = k_from_pr_id
            and sa_from_f_id = k_from_f_id
            and sa_account_type = 6
      where kcp_period = :rep_period
        and kcp_from_reg_id = :reg_id
    into
      :aarep_k_id,
      :sa_account
    do
    begin
      update kontcalcperc set
        kcp_kommis_account = :sa_account
      where kcp_period = :rep_period
        and kcp_from_k_id = :aarep_k_id
        and kcp_from_c_id = 0;
    end

    for
      select
        kcp_from_k_id
      from kontcalcperc
      where kcp_period = :rep_period
        and kcp_from_reg_id = :reg_id
        and kcp_perc_begin_saldo = 0
        and kcp_perc_calced = 0
        and kcp_perc_payed = 0
        and kcp_perc_writeoff = 0
        and kcp_perc_saldo = 0
        and kcp_out_perc_begin_saldo = 0
        and kcp_out_perc_calced = 0
        and kcp_out_perc_saldo = 0
        and kcp_perc_transfer_payed = 0
        and kcp_perc_transfer_writeoff = 0
        and kcp_kommis_begin_saldo = 0
        and kcp_kommis_calced = 0
        and kcp_kommis_payed = 0
        and kcp_kommis_writeoff = 0
        and kcp_kommis_saldo = 0
        and kcp_kommis_transfer_payed = 0
        and kcp_kommis_transfer_writeoff = 0
    into
      :k_id
    do
    begin
      if (exists(
        select *
        from kontstate
        where ks_from_k_id = :k_id
          and ks_ol_is_delete = 0
          and ks_state = 2
          and ks_date_begin < :rep_date_begin)) then
      delete from kontcalcperc
      where kcp_period = :rep_period
        and kcp_from_k_id = :k_id
        and kcp_from_k_id = 0;
    end

    for
      select
        kcp_period,
        kcp_from_k_id,
        kcp_period - (extract(year from ks_date_begin) * 100 + extract(month from ks_date_begin))
      from kontcalcperc
        inner join kontstate
          on ks_from_k_id = kcp_from_k_id
            and ks_state = 2
            and ks_ol_is_delete = 0
            and (extract(year from ks_date_begin) * 100 + extract(month from ks_date_begin)) < kcp_period
      where kcp_period = :rep_period
        and kcp_from_reg_id = :reg_id
        and kcp_loan_is_closed = 0
    into
      :tmp_period,
      :aarep_k_id,
      :delta
    do
    begin
      update kontcalcperc set
        kcp_loan_is_closed = :delta
      where kcp_period = :tmp_period
        and kcp_from_k_id = :aarep_k_id
        and kcp_from_c_id = 0;
    end

    for
      select
        kcp_from_k_id
      from kontcalcperc
      where kcp_period = :rep_period
        and kcp_from_reg_id = :reg_id
    into
      :aarep_k_id
    do
    begin
      -- Деление на участников
      loan_kc_list = '';
      loan_amount_list = '';
      loan_kommis_list = '';

      select
        list(kc_id),
        list(kc_amount_approved),
        list(iif(kc_amount_approved > 0, coalesce(kc_amount_approved * k_kommis_perc, 0) - coalesce(kc_kommis_lgota, 0), 0))
      from knclient
        inner join kontract
          on k_id = kc_from_k_id
      where kc_from_k_id = :aarep_k_id
        and kc_amount_approved > 0
      into
        :loan_kc_list,
        :loan_amount_list,
        :loan_kommis_list;

      insert into kn_kontcalcperc (
        kcp_period, kcp_from_kc_id, kcp_from_k_id, kcp_from_reg_id, kcp_perc_saldo,
        kcp_transfer_reg_id, kcp_perc_calced, kcp_perc_payed, kcp_perc_writeoff,
        kcp_perc_begin_saldo, kcp_perc_transfer_payed, kcp_perc_transfer_writeoff,
        kcp_kommis_calced, kcp_kommis_payed, kcp_kommis_writeoff, kcp_kommis_saldo,
        kcp_kommis_begin_saldo, kcp_kommis_transfer_payed, kcp_kommis_transfer_writeoff,
        kcp_perc_account, kcp_kommis_account, kcp_loan_is_closed, kcp_out_perc_begin_saldo,
        kcp_out_perc_calced, kcp_out_perc_saldo)
      select
        :rep_period, kc_id, :aarep_k_id, cl_kcp_from_reg_id, cl_kcp_perc_saldo,
        cl_kcp_transfer_reg_id, cl_kcp_perc_calced, cl_kcp_perc_payed, cl_kcp_perc_writeoff,
        cl_kcp_perc_begin_saldo, cl_kcp_perc_transfer_payed, cl_kcp_perc_transfer_writeoff,
        cl_kcp_kommis_calced, cl_kcp_kommis_payed, cl_kcp_kommis_writeoff, cl_kcp_kommis_saldo,
        cl_kcp_kommis_begin_saldo, cl_kcp_kommis_transfer_payed, cl_kcp_kommis_transfer_writeoff,
        cl_kcp_perc_account, cl_kcp_kommis_account, cl_kcp_loan_is_closed, cl_kcp_out_perc_begin_saldo,
        cl_kcp_out_perc_calced, cl_kcp_out_perc_saldo
      from hp_devide_kn_kontcalcperc(:aarep_k_id, :rep_period, :loan_kc_list, :loan_amount_list, :loan_kommis_list);
    end
  end

  -- в таблицу KN_Oksummary добавляем строки для судебных госпошлин, в случае если займ закрыт давно
  execute procedure hp_do_kn_oksummary_add_law(:rep_date_end);
  execute procedure hp_do_kn_oksummary_add_law(dateadd(month, 1, :rep_date_end + 1) - 1);
end
