CREATE PROCEDURE AAREP_REPAY_EARLY_IPF(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    PR_NAME STR_100,
    REPAY_OS CURR_TYPE,
    REPAY_PERC CURR_TYPE)
AS
declare variable REG_ID integer;
begin
  for
  select id_list from region_list(:rep_region_id,1,3,1)
  into reg_id
  do
  begin
    select reg_code||' '||reg_name from region
    where reg_unicode = :reg_id
    into reg_name;

    for
    select pr_name, -sum(ok_os), -sum(ok_perc)
    from operkred, kontract, product
    where ok_ol_is_active = 1
    and ok_type = 4
    and ok_ol_reg_id = :reg_id
    and ok_ol_date between :rep_date_begin and :rep_date_end
    and ok_sh_date > :rep_date_end
    and ok_from_k_id = k_id
    and k_from_pr_id = pr_id
    and pr_code containing '.ipf.'
    group by 1
    into pr_name, repay_os, repay_perc
    do
    begin

      suspend;
    end
  end
end
