CREATE PROCEDURE HP_DO_MONEYOPER_IMPORT_PROCESS(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    NEED_CLEAR_OLD_LIST INTEGER = 0,
    NEED_FILL_MONEY_OPER INTEGER = 0,
    NEED_CASH INTEGER = 0,
    NEED_BANK INTEGER = 0,
    NEED_1C INTEGER = 0)
RETURNS (
    RES_KIND D_STR_CODE,
    RES_TOTAL D_INTEGER,
    RES_IMPORTED D_INTEGER,
    RES_CREATED D_INTEGER,
    RES_LINKED D_INTEGER)
AS
begin
-- перед запуском импорта из 1с, необходимо заполнить таблицы счетов и проводок через отдельную программу
-- при необходимости, очистить таблицу KN_OPERMONEY и PACKET_1C_IMP_BANK за старые периоды

  res_total = 0;
  res_imported = 0;
  res_created = 0;
  res_linked = 0;

  res_kind = 'Импорт новых счетов 1с';
  select new_acc_count, imported_count
  from hp_import_packet_1c_acc
  into :res_total, :res_imported;
  suspend;

  update packet_1c_imp_bank
  set pcb_acc_dt_code = pcb_acc_dt_code
  where pcb_acc_dt_id is null;

  update packet_1c_imp_bank
  set pcb_acc_kt_code = pcb_acc_kt_code
  where pcb_acc_kt_id is null;

  res_total = 0;
  res_imported = 0;
  res_created = 0;
  res_linked = 0;

  if (need_clear_old_list = 1) then
  begin
    res_kind = 'Удаление старого списка денеж.опер.';
    delete from kn_opermoney
    where kom_ol_date < :rep_date_begin and
          kom_is_done = 1;

    res_total = row_count;
    suspend;

    res_kind = 'Удаление старого списка выписки 1C';
    delete from packet_1c_imp_bank
    where pcb_date < :rep_date_begin and
          pcb_from_ol_id is not null;

    res_total = row_count;
    suspend;
  end

  if (need_fill_money_oper = 1) then
  begin
    res_total = 0;
    res_imported = 0;
    res_created = 0;
    res_linked = 0;
    res_kind = 'Составление списка кассовых опер.';
    select res_total, res_imported, res_created, res_linked
    from hp_do_ref_account_cash_obor(:rep_date_begin, :rep_date_end, -1)
    into :res_total, :res_imported, :res_created, :res_linked;
    suspend;

    res_total = 0;
    res_imported = 0;
    res_created = 0;
    res_linked = 0;
    res_kind = 'Составление списка банковских опер.';
    select res_total, res_imported, res_created, res_linked
    from hp_do_ref_account_bank_obor(:rep_date_begin, :rep_date_end, -1, 0)
    into :res_total, :res_imported, :res_created, :res_linked;
    suspend;
  end

  if (need_cash = 1) then
  begin
    res_total = 0;
    res_imported = 0;
    res_created = 0;
    res_linked = 0;
    res_kind = 'Обработка кассовых операций';
    select res_total, res_imported, res_created, res_linked
    from hp_do_ref_account_cash_obor(:rep_date_begin, :rep_date_end, 0)
    into :res_total, :res_imported, :res_created, :res_linked;
    suspend;
  end

  if (need_bank = 1) then
  begin
    res_total = 0;
    res_imported = 0;
    res_created = 0;
    res_linked = 0;
    res_kind = 'Обработка банковских операций';
    select res_total, res_imported, res_created, res_linked
    from hp_do_ref_account_bank_obor(:rep_date_begin, :rep_date_end, 0, 0)
    into :res_total, :res_imported, :res_created, :res_linked;
    suspend;
  end

  if (need_1c = 1) then
  begin
    res_total = 0;
    res_imported = 0;
    res_created = 0;
    res_linked = 0;
    res_kind = 'Обработка операций из 1С';
    select res_total, res_imported, res_created, res_linked
    from hp_do_packet_1c_imp_bank(1000000000001, :rep_date_begin, :rep_date_end, 0)
    into :res_total, :res_imported, :res_created, :res_linked;
    suspend;
  end
end
