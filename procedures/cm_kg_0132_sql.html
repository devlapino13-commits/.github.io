CREATE PROCEDURE CM_KG_0132_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kf_disb_date d_date;
declare variable c_id id_type;
declare variable un_c_id id_type;
declare variable jk_system_code str_100;
declare variable res_cell_no d_str_255;
declare variable res_cell_no_new d_str_255;
declare variable rd_id d_integer;
declare variable ovs_datetime d_datetime;
declare variable res_date_save d_datetime;
declare variable ma_id d_id;
declare variable eds_status_date date;
declare variable un_name varchar(30);
begin
  if (k_id = 109010189842200101) then
    exit;

  rd_id = f_get_rd_id('ContactInfo.Cell_Main');
  res_error_is_critical = 2;
  select un.un_from_c_id, un.un_name
  from username un
  where un_id = :un_id
  into :un_c_id, un_name;

  if (un_name = 'SHTASHHODJAEV') then
    exit;

  jk_system_code = '';

  if (un_c_id is not null) then
  begin
    select jk.jk_system_code
    from hr_pstatus hps
    inner join joblist jl on jl.jl_id = ps_from_jl_id
    inner join jobkind jk on jk.jk_id = jl.jl_kind
    where current_date between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          hps.ps_from_c_id = :un_c_id
    into :jk_system_code;
  end
  jk_system_code = coalesce(jk_system_code, '');
  for select cl.c_id, cl.c_name_fio, k.k_disb_date, k_from_ma_app_id
      from knclient knc
      inner join kontract k on knc.kc_from_k_id = k.k_id
      inner join clients cl on cl.c_id = knc.kc_from_c_id
      where knc.kc_from_k_id = :k_id and
            knc.kc_amount_approved > 0
      into :c_id, :res_info_text, :kf_disb_date, :ma_id
  do
  begin
    select first 1 ma_date_save
    from miniapp
    where ma_id = :ma_id and
          ma_is_active = 1
    order by ma_number
    into :res_date_save;
    -- номер на дату заявки
    res_cell_no = null;
    select first 1 ovs_value, ovs_datetime
    from objectprops
    inner join objectvalues on ovs_from_opr_id = opr_id and
          ovs_datetime <= :res_date_save and
          ovs_is_active = 1 and
          ovs_value <> '' and
          ovs_add_value = 2
    where opr_object_kind = '0' and
          opr_object_id = :c_id and
          opr_subject_kind = '-1' and
          opr_subject_id = :rd_id
    order by ovs_datetime desc
    into :res_cell_no, :ovs_datetime; -- возвращает подтвер тел номер (старый)

    eds_status_date = null;
    select cast(eds_status_datetime as date)
    from edoc_list el
    inner join edoc_sign es on es.eds_from_edl_id = el.edl_id
    where el.edl_object_kind = 0 and
          el.edl_object_id = :c_id and
          el.edl_blank_status = 1 and --бланки приготовлены
          el.edl_from_edc_id = f_edoc_kind('asp.agreement', 'id') and
          el.edl_status = 1 and --статус подписан
          cast(eds_status_datetime as date) < :ovs_datetime rows 1
    into :eds_status_date;
    -- измененный номер
    res_cell_no_new = null;

    select first 1 ovs_value
    from objectprops
    inner join objectvalues on ovs_from_opr_id = opr_id and
          ovs_datetime > :ovs_datetime and
          ovs_datetime <= (:kf_disb_date + 1) and
          ovs_is_active = 1 and
          ovs_value not in ('', :res_cell_no) and
          ovs_add_value = 2
    where opr_object_kind = '0' and
          opr_object_id = :c_id and
          opr_subject_kind = '-1' and
          opr_subject_id = :rd_id
    order by ovs_datetime desc
    into :res_cell_no_new; --возращает новый тел номер
    --Если ранее клиент подписывал договор ЭЦП, и при получение нового кредита сменил тел.номер то такие кредиты утверждает РМ.
    if (res_cell_no <> res_cell_no_new) then
    begin
     if (eds_status_date is not null and
          jk_system_code <> 'JobSystem.Region_KM' and jk_system_code <> 'JobSystem.region_director' ) then -- РМ --МО
      begin
        res_error_text = 'Данную выдачу может утвердить MO т.к. был изменен тел.номер.';
        suspend;
      end
      -- Если ранее договора ЭЦП не было то смена телефона идет через МО с заключением договора ЭЦП.
      if (eds_status_date is null and
          jk_system_code = 'JobSystem.LO_Senior') then --Старший СПФ
      begin
        res_error_text = 'Данную выдачу может утвердить МО т.к. был изменен тел.номер.';
        suspend;
      end
    end
  end
end
