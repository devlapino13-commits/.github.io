CREATE PROCEDURE HP_EDOC_CONF_STAFF_INS(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    BLANK_LIST D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable edoc_kind d_str_code;
declare variable obj_kind_code d_str_code;
declare variable doc_date d_date;
declare variable reg_id d_integer;
declare variable need_confirm d_bool;
begin
  obj_kind = f_objkind(obj_kind, 'id');
  obj_kind_code = f_objkind(obj_kind, 'code');
  res_text = '';
  edl_id = 0;

  if (obj_kind_code not in ('hr_orders', 'servicedog')) then
    exception e_default 'Incorrect object_kind: ' || coalesce(obj_kind, '<Null>');

  if (obj_kind_code = 'hr_orders') then
  begin
    select ho_date, ho_from_reg_id
    from hr_orders
    where ho_id = :obj_id
    into doc_date, reg_id;

    if (row_count = 0) then
      res_text = 'Приказ не найден, id: ' || coalesce(obj_id, '<Null>');

    edoc_kind = f_edoc_kind('staff.orders', 'id');
  end else
  begin
    select sd_date_begin, sd_from_reg_id
    from servicedog
    where sd_id = :obj_id and
          sd_kind = 0
    into doc_date, reg_id;

    if (row_count = 0) then
      res_text = 'Договор на оказание услуг не найден, id: ' || coalesce(obj_id, '<Null>');

    edoc_kind = f_edoc_kind('staff.servicedog', 'id');
  end

  if (res_text = '') then
  begin
    insert into edoc_conf (edn_object_kind, edn_object_id, edn_date, edn_from_edc_id,
                           edn_create_un_id, edn_from_reg_id, edn_blank_list)
    values (:obj_kind, :obj_id, :doc_date, :edoc_kind,
            :un_id, :reg_id, :blank_list)
    returning edn_edl_id, edn_need_confirm
    into :edl_id, need_confirm;
  
    if (need_confirm = 0) then
      res_text = 'Эл.документ успешно сформирован';
    else
      res_text = 'Эл.документ отправлен на согласование в бухгалтерию';
  end

  suspend;
end
