CREATE PROCEDURE HP_SUD_LIGRES_SEND(
    SLC_ID D_ID,
    UN_ID D_ID)
RETURNS (
    ERROR_TEXT D_STR_255)
AS
declare variable obj_kind d_integer;
declare variable td d_date;
declare variable c_id d_id;
declare variable k_id d_id;
begin
  obj_kind = f_objkind('sud_loanclient', 'id');
  error_text = '';

  select 'Клиент в списке "нет информации".'
  from sud_object_list
  where sol_kind = 1
     and sol_object_kind = :obj_kind
     and sol_object_id = :slc_id
  rows 1
  into error_text;

  if (error_text = '') then
  begin
    td = (select max(smc_begin_date) from sud_maincollect where smc_status < 2 and smc_from_slc_id = :slc_id);

    if (td > dateadd(month, -1, current_date)) then
      error_text = 'Не прошло месяца с даты подачи.';
  end

  if (error_text = '') then
  begin
    select kc_from_c_id, kc_from_k_id
    from sud_loanclient, knclient
    where slc_id = :slc_id
      and kc_id = slc_from_kc_id
    into c_id, k_id;

    select max(eil_date)
    from extinfolist, extinfocfg
    where eil_from_c_id = :c_id
      and eic_id = eil_from_eic_id
      and eic_code = 'ligres_fssp_cli_check'
      and eil_status > 0
    into td;

    if (td > dateadd(month, -1, current_date)) then
      error_text = 'Не прошло месяца с даты последнего запроса.';

    if (error_text = '') then
    begin
      if (not exists(
                    select 1
                    from oksummary oks
                    where oks_from_k_id = :k_id
                      and current_date between oks.oks_date_begin and oks.oks_date_end
                      and (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc + oks_out_other + oks_extra) > 0))
      then
        error_text = 'Договор закрыт.';
    end
  end

  if (error_text = '') then
    execute procedure hp_extinfo_insert_bki_cfg('ligres_fssp_cli_check', :c_id, :obj_kind, :slc_id, 0, :un_id);

  suspend;
end
