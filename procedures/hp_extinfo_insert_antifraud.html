CREATE PROCEDURE HP_EXTINFO_INSERT_ANTIFRAUD(
    INN D_STR_20 NOT NULL,
    UN_ID D_ID = 1000000000000)
RETURNS (
    AC_ID D_ID)
AS
declare variable EIC_ID D_ID;
declare variable EIL_ID D_ID;
begin
    -- RFCKG-302
    -- Если сегодня уже был запрос, новый мы не отправляем
    select ac.ac_id
    from antifraud_check ac
    where ac.ac_inn = :inn
        and ac.ac_date = current_date
        and f_enum('antifraud_status', ac.ac_status, 'code') in ('in_process', 'not_found', 'was_found')
    into :ac_id;

    if(coalesce(ac_id, 0) > 0) then
    begin
        suspend;
        exit;
    end

    eic_id = (select eic.eic_id from extinfocfg eic where eic.eic_code = 'antifraud_client_check');
    insert into antifraud_check (ac_inn)
    values (:inn)
    returning ac_id into :ac_id;

    insert into extinfolist (eil_from_eic_id, eil_object_kind, eil_object_id, eil_is_json, eil_from_un_id)
    values (:eic_id, f_objkind('antifraud_check', 'id'), :ac_id, 1, :un_id)
    returning eil_id into :eil_id;

    update antifraud_check ac
    set ac.ac_from_eil_id = :eil_id
    where ac.ac_id = :ac_id;

    post_event 'antifraud_client_check';
    suspend;
end
