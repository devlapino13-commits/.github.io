CREATE PROCEDURE HP_INSURANCE_TARIF_LIST
RETURNS (
    IT_ID D_ID,
    IT_FROM_PR_ID D_ID,
    IT_FROM_PRL_ID D_ID,
    IT_CREATE_DATETIME D_DATETIME,
    IT_IS_ACTIVE D_SMALLINT,
    IT_LAST_DATETIME D_DATETIME,
    IT_FROM_UN_ID D_ID,
    IT_MIN_POINT D_INTEGER,
    IT_MAX_POINT D_INTEGER,
    IT_TARIF D_PERCENT,
    IT_TARIF_TYPE D_INTEGER,
    IT_TARIF_TYPE_STR D_STR_CODE,
    PR_NAME D_STR_255,
    AG_CAPTION D_STR_255,
    IAC_COMMIS D_CURRENCY)
AS
begin
    for select it.it_id,
               it.it_from_pr_id,
               it.it_from_prl_id,
               it.it_create_datetime,
               it.it_is_active,
               it.it_last_datetime,
               it.it_from_un_id,
               it.it_min_point,
               it.it_max_point,
               it.it_tarif,
               it.it_tarif_type,
               iif(it.it_tarif_type = 0, 'Дневной', 'Годовой'),
               pr.pr_name,
               ag.ag_caption,
               iac.iac_commis
    from insurance_tarif it
    join insur_ag_commis iac on iac.iac_from_it_id = it.it_id
    join agreements ag on ag.ag_id = iac.iac_from_ag_id
    join product pr on pr.pr_id = it.it_from_pr_id
    into :it_id,
         :it_from_pr_id,
         :it_from_prl_id,
         :it_create_datetime,
         :it_is_active,
         :it_last_datetime,
         :it_from_un_id,
         :it_min_point,
         :it_max_point,
         :it_tarif,
         :it_tarif_type,
         :it_tarif_type_str,
         :pr_name,
         :ag_caption,
         :iac_commis
    do
    suspend;
end
