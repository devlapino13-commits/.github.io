CREATE PROCEDURE HP_USER_CASH_INS(
    OFFICE_ID D_INTEGER,
    ACC_TYPE D_INTEGER,
    C_ID D_ID,
    NUM_CASH D_INTEGER,
    DATE_BEGIN D_DATETIME,
    DATE_END D_DATETIME)
RETURNS (
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_BOOL)
AS
declare variable bg_id d_str_20;
declare variable db_id d_integer;
begin
  -- создает запись в user_cash (RFCKZ-396)
  -- используется в форме "Документы -> Соответствие кассы пользователю", поле "Счет кассы"

  db_id = f_get_db_id();

  if (db_id <> 39801) then
    exit;

  res_is_error = 1;
  res_text = 'Возникла ошибка при добавлении записи';

  if (acc_type not in (0, 1)) then
  begin
    res_text = 'Ошибка: счет ' || acc_type || ' не настроен';
    suspend;
    exit;
  end

  if (date_begin is null) then
    date_begin = current_timestamp;

  insert into user_cash (uc_from_c_id, uc_from_reg_id, uc_num_cash, uc_acctype, uc_date_beg)
  values (:c_id, :office_id, :num_cash, :acc_type, :date_begin);

  res_is_error = 0;
  res_text = 'Запись успешно добавлена';
  suspend;
end
