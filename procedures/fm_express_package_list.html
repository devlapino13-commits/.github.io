CREATE PROCEDURE FM_EXPRESS_PACKAGE_LIST(
    UN_ID D_ID,
    EXP_ID D_INTEGER)
RETURNS (
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID,
    TEXT_1 D_STR_255,
    TEXT_2 D_STR_255,
    TEXT_3 D_STR_255,
    TEXT_4 D_STR_255)
AS
declare variable exa_final_status d_integer;
declare variable lo_id d_id;
declare variable exa_from_exc_id d_integer;
declare variable exc_name d_str_100;
declare variable wait_text d_str_100;
declare variable ma_id d_integer;
declare variable k_id d_id;
begin
  select un_id
  from hp_test_un_id(:un_id)
  into un_id;

  select un_from_c_id
  from username
  where un_id = :un_id
  into lo_id;

  obj_kind = f_objkind('expressapp', 'id');
  text_3 = '';
  for select exa_id,
             c_name_fio,
             iif(exa_from_exc_id > 0,
             f_numtostr(exa_amount, 0) || '<br>' || f_datetostr(exa_disb_date),
             f_numtostr(exa_amount, 0)),
             exa_final_status,
             exa_from_exc_id,
             exc_name,
             exa_from_ma_id,
             exa_from_k_id
  from expressapp, clients, expresscfg
  where exa_from_c_id = c_id and
        exa_from_exc_id = exc_id and
        exa_lo_id = :lo_id  and
        exa_from_exp_id = :exp_id -- ID пакета, это временное решение чтобы быстрее запустить акцию
  order by iif(exa_wait_status = 1, 1, 2)
  into obj_id, text_1, text_2, exa_final_status, exa_from_exc_id, exc_name, ma_id, k_id
  do
  begin
    obj_kind = f_objkind('expressapp', 'id');
    text_3 = ''; -- Статус
    text_2 = trim(text_2);

    if (exa_from_exc_id = 0) then
      exc_name = '';

    if (exa_from_exc_id <> 0) then
    begin
      if (k_id is not null) then
      begin
        obj_kind = f_objkind('kontract', 'id');
        obj_id = k_id;

        text_4 = '';
        select kt.k_kontract_no
        from kontract kt
        where kt.k_id = :k_id
        into text_4;

      end else
      if (ma_id is not null) then
      begin
        obj_kind = f_objkind('miniapp', 'id');
        obj_id = ma_id;

        text_4 = '';
        select kal.kal_new_kf_name
        from miniapp ma
        inner join knaktlist kal on kal.kal_from_kf_id = ma.ma_from_kf_id
        where ma.ma_id = :ma_id and
              ma.ma_is_active = 1 and
              ma.ma_date_end >= '31.12.2999' and
              current_date between kal.kal_date and kal.kal_date_end
        into text_4;

      end

      if (exa_final_status in (0, 1)) then
      begin
        select res_text
        from hp_expressapp_status_info(:obj_kind, :obj_id, :un_id)
        into wait_text;
        wait_text = replace(wait_text, 'Робот ', '');
        text_3 = wait_text || ' ' || exc_name;
      end else
      begin
        text_3 = case exa_final_status
          when -2 then '&#10006; Клиент отказался'
          when -1 then '&#10060; Отказано'
          when  2 then '&#9989; Займ выдан'
        end;
      end
    end else
    begin
      text_3 = case exa_final_status
        when -2 then '&#10006; Клиент отказался'
        when -1 then '&#10060; Отказано'
        when  2 then '&#9989; Займ выдан'
      end;
    end

    suspend;
  end
end
