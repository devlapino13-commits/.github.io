CREATE PROCEDURE AAREP_SA_TABEL_PRINT_MASTER(
    REPORT_REGION INTEGER,
    REPORT_YEAR INTEGER,
    REPORT_MONTH INTEGER)
RETURNS (
    SAT_FROM_C_ID DECIMAL(18,0),
    SAT_YEAR INTEGER,
    SAT_MONTH INTEGER,
    SAT_OFFICE_ID INTEGER,
    SAT_OFFICE_NAME VARCHAR(100),
    SAT_TOTAL_NORMA DECIMAL(18,4),
    SAT_TOTAL_HOURS DECIMAL(18,4),
    SAT_TOTAL_CALC DECIMAL(18,4),
    SAT_PERSONAL_FIO VARCHAR(100),
    SAT_NUMBER INTEGER,
    SAT_VOCATION_PAYED DECIMAL(18,4),
    SAT_VOCATION_BEGIN DECIMAL(18,4),
    SAT_VOCATION_END DECIMAL(18,4),
    SAT_VOCATION_CURR DECIMAL(18,4))
AS
declare variable sat_payed decimal(18,4);
declare variable tmp_day decimal(18,4);
declare variable tmp_day2 decimal(18,4);
declare variable total_days integer;
declare variable dd date;
declare variable start_work decimal(15,2);
declare variable stage_date date;
begin
  sat_number = 0;
  total_days = 30;
  for select sat_from_c_id, sat_year, sat_month, sat_office_id, sat_office_name, sum(sat_days_norma),
             sum(sat_days_total), sum(sat_days_vocation), sum(sat_days_calc), sat_personal_fio
      from aarep_sa_tabel_print_mtemp(:report_region, :report_year, :report_month)
      group by sat_from_c_id, sat_year, sat_month, sat_office_id, sat_office_name, sat_personal_fio
      order by sat_office_id, sat_personal_fio
      into :sat_from_c_id, :sat_year, :sat_month, :sat_office_id, :sat_office_name, :sat_total_norma, :sat_total_hours,
           :sat_vocation_payed, :sat_total_calc, :sat_personal_fio

  do
  begin
    sat_number = sat_number + 1;
    sat_vocation_payed = 0;
    sat_vocation_begin = 0;
    sat_vocation_end = 0;
    sat_vocation_curr = 0;
    select sat_vocation_payed, sat_vocation_begin, sat_vocation_end, sat_vocation_curr, sat_total_norma,
           sat_total_hours, sat_total_calc
    from aarep_sa_tabel_one_voc(:sat_office_id, :sat_from_c_id, :report_year, :report_month)
    rows 1
    into :sat_vocation_payed, :sat_vocation_begin, :sat_vocation_end, :sat_vocation_curr, :sat_total_norma,
         :sat_total_hours, :sat_total_calc;

    suspend;
  end
end
