CREATE PROCEDURE AAREP_SA_TABEL_ONE_VOC(
    REPORT_REGION_ID INTEGER,
    REPORT_CLIENT_ID DECIMAL(18,0),
    REPORT_YEAR INTEGER,
    REPORT_MONTH INTEGER)

    SAT_VOCATION_PAYED DECIMAL(18,4),
    SAT_VOCATION_BEGIN DECIMAL(18,4),
    SAT_VOCATION_END DECIMAL(18,4),
    SAT_VOCATION_CURR DECIMAL(18,4),
    SAT_TOTAL_NORMA DECIMAL(18,4),
    SAT_TOTAL_HOURS DECIMAL(18,4),
    SAT_TOTAL_CALC DECIMAL(18,4))
AS

declare variable TOTAL_DAYS integer;

declare variable STAGE_DATE date;

declare variable SAT_PAYED decimal(18,4);
declare variable DD1 date;
declare variable DD2 date;
declare variable UNPAID_HOLIDAY_BEGIN decimal(18,4);
declare variable UNPAID_HOLIDAY_END decimal(18,4);
declare variable DD2_ date;
declare variable REAL_DATE_GONE date;
declare variable DB_ID integer;
declare variable REG_ID integer;
begin
  day_1 = 0;
  day_2 = 0;
  day_3 = 0;
  last_day = cast('01.' || report_month || '.' || report_year as date);
  last_day = dateadd(1 month to :last_day) - 1;
  begin_of_year = cast('01.01.' || report_year as date);
  end_of_prior_month = cast('01.' || report_month || '.' || report_year as date) - 1;
  db_id = f_get_db_id();
  -- RFCKG-4782
  reg_mountain_koef = f_reg_mount_koef(:report_region_id, end_of_prior_month); -- RFCKG-5508
  if (reg_mountain_koef = 1) then
    sat_cons = 2.33;
  else
    sat_cons = 3.33;

  sat_number = 0;
  total_days = 30;
  sat_vocation_curr = 0;
  sat_payed = 0;
  sat_vocation_begin = 0;
  sat_vocation_end = 0;
  sat_vocation_payed = 0;
  sat_total_norma = 0;
  sat_total_hours = 0;
  sat_total_calc = 0;

  select sum(sat_days_norma), sum(sat_days_total), sum(sat_days_vocation), sum(sat_days_calc)
  from aarep_sa_tabel_one_tmp(:report_region_id, :report_client_id, :report_year, :report_month)
  into :sat_total_norma, :sat_total_hours, :sat_vocation_payed, :sat_total_calc;

  if (sat_vocation_payed is null) then
    sat_vocation_payed = 0;
  dd = null;

  if (db_id = 94401) then
  begin
-- S.K. 05.02.2019
    sat_cons = (select swn_month_norma
                from sa_weekends_norma
                where swn_from_c_id = :report_client_id and
                      swn_year = :report_year);
  end
  else
  begin
    if (report_client_id = 1010100001212 and (report_year * 100 + report_month)<= 202012) then
      sat_cons = 2.75;
    else
    begin
      if (reg_mountain_koef = 1) then
        sat_cons = 2.33;
      else
        sat_cons = 3.33;
    end
  end
  m_koef = sat_cons;

  /*------------ igusev 27.08.2012*/
  select hrps.ps_date_begin
  from hr_pstatus hrps
  inner join hr_orders ho on ho.ho_id = hrps.ps_from_ho_id
  where hrps.ps_from_c_id = :report_client_id and
        ho.ho_order_type = 1 and
        ps_status in (0, 1) -- включает только аннулированные и утвержденные
  order by hrps.ps_date_begin desc
  rows 1
  into :employing_date;/* получаю дату, когда сотрудник последний раз был принят на работу*/
  if (employing_date < cast('01.01.' || :report_year as date)) then
  begin
    employing_date = cast('01.01.' || :report_year as date);
  end

  select c_hired_date
  from clients
  where c_id = :report_client_id

  begin
    stage_date = cast('01.01.' || :report_year as date);
  end

    employing_date = stage_date;
  if (employing_date is null) then
    employing_date = cast('01.01.' || :report_year as date);

  stage_month = extract(month from employing_date);
  /*------------*/
  /* ---- igusev 14.11.2012 ------ процедура вытаскивает из табеля кол-во дней неоплачиваемого отпуска взятых с начала года до REPORT_MONTH (включительно)*/
  unpaid_holiday_end = 0;

  select unpaid_holiday
  from calc_unpaid_holidays(:report_client_id, :report_year, :report_month, :sat_cons, :stage_month)
  into :unpaid_holiday_end;

  unpaid_holiday_begin = 0;
  if (:report_month > 1) then
  begin
    select unpaid_holiday
    from calc_unpaid_holidays(:report_client_id, :report_year, :report_month - 1, :sat_cons, :stage_month)
    into :unpaid_holiday_begin;
  end
  /*-----------------------------*/

  select sum(sat_days_total * sawt_hours_real)
  from sa_tabel, sa_worktype sa
  where sat_year_month < (:report_year * 100 + :report_month) and
        sat_year_month >= (:report_year * 100 + :stage_month) /* igusev 24.08.2012 добавил, чтобы учитывать все оплачиваемые отпуска, которые сотрудник брал с даты принятия на работу (ес был принят не в этом году, то считаю с начала года) */
        and
        sat_year_month >= (:report_year * 100 + 1) and
        sat_is_ready = 1 and
        sawt_id = sat_from_sawt_id and
        sawt_base_type = 4 and
        sat_from_c_id = :report_client_id and
        sa.sawt_is_paid = 1 /*igusev добавил 06.09.2012*/
  into :sat_payed;
  if (sat_payed is null) then
    sat_payed = 0;

  /* minus_num = 0; */

  select ps_date_begin - 1
  from hr_pstatus
  where ps_from_c_id = :report_client_id and
        ps_is_working = 0 and
        ps_status in (0, 1) -- включает только аннулированные и утвержденные
  order by ps_date_begin desc
  rows 1
  into :date_gone;

  real_date_gone = date_gone;

  /*
    if (:DATE_GONE between cast('01.'||report_month||'.'||report_year as date) and :LAST_DAY) then

        minus_num = 1;
    end
    */

  if ((date_gone is null) or (date_gone > :last_day)) then
  begin
    date_gone = :last_day;
  end
  else
  begin
    is_work = 1;

      dd = dateadd(1 month to :dd);

            sat_cons = 2.33;
          else
            sat_cons = 3.33;
        end
      end

      day_1 = total_days_begin - extract(day from :stage_date) + 1;
      day_1 = cast(day_1 / total_days_begin * sat_cons as decimal(15,2));

      i1 = extract(month from :stage_date) + 1;
      i2 = extract(month from :date_gone) - 1;
      i = i1;
      day_2 = 0;
      while (i <= i2) do
      begin
        if (i < 9) then

        else
          dd = cast('01.' || cast(:i + 1 as varchar(2)) || '.' || extract(year from :dd) as date) - 1;
        select ps_from_reg_id
        from hr_pstatus
        where ps_from_c_id = :report_client_id and

        into :reg_id;
        reg_mountain_koef = f_reg_mount_koef(:reg_id, :dd);

        if (db_id <> 94401) then
        begin
          if (report_client_id = 1010100001212 and dd <= cast('31.12.2020' as date)) then
            sat_cons = 2.75;
          else
          begin
            if (reg_mountain_koef = 1) then
              sat_cons = 2.33;
            else
              sat_cons = 3.33;
          end
        end

        day_2 = day_2 + sat_cons;
        i = i + 1;
      end

      dd = dateadd(1 month to :date_gone);
      dd = cast('01.' || extract(month from :dd) || '.' || extract(year from :dd) as date) - 1;
      select ps_from_reg_id
      from hr_pstatus
      where ps_from_c_id = :report_client_id and
            ps_date_begin <= :dd and
            ps_date_end >= :dd and
            ps_status in (0, 1) -- включает только аннулированные и утвержденные
      into :reg_id;
      reg_mountain_koef = f_reg_mount_koef(:reg_id, :dd);

      if (db_id <> 94401) then
      begin
        if (report_client_id = 1010100001212 and dd <= cast('31.12.2020' as date)) then
          sat_cons = 2.75;
        else
        begin
          if (reg_mountain_koef = 1) then
            sat_cons = 2.33;
          else
            sat_cons = 3.33;
        end
      end

      day_3 = extract(day from :date_gone);/*- minus_num*/
      day_3 = cast(day_3 * sat_cons as decimal(15,2));
      day_3 = cast(day_3 / total_days_end as decimal(15,2));
      if ((sat_total_hours = 0) and
          (real_date_gone between(cast('01.' || report_month || '.' || report_year as date)) and last_day)) then
      begin
        day_3 = 0;
      end
    end
    else
    begin
      sat_cons = :m_koef;
      day_3 = extract(day from :date_gone) - extract(day from :stage_date) + 1;
      if (day_3 < 0) then
        day_3 = 0;
      day_3 = cast(day_3 * sat_cons as decimal(15,2));
      day_3 = cast(day_3 / total_days_end as decimal(15,2));
    end
    if (extract(month from :stage_date) < :report_month) then
    begin
      sat_vocation_begin = sat_vocation_begin + day_1 + day_2;
    end
    else
    begin
      sat_vocation_curr = sat_vocation_curr + day_1;
    end
    if (extract(month from :date_gone) = :report_month) then
    begin
      sat_vocation_curr = sat_vocation_curr + day_3;
    end
    else
    begin
      sat_vocation_begin = sat_vocation_begin + day_3;
    end
  end
  else
  begin
    sat_vocation_begin = 0;
    sat_vocation_end = 0;
    sat_vocation_curr = 0;
    sat_vocation_payed = 0;
  end

  /* не начисление отпускных дней во время дикретных отпусков */
  for select hrv_date_begin, hrv_date_end_fact

            hrv_date_begin <= hrv_date_end_fact and

            ho_order_type = sawt_id and
            hrv_from_ho_id = ho_id and
            sawt_base_type = 5 /* Отпуска */

            sawt_is_paid = 0 /* неоплачиваемые, дикретные */
            and
            hrv_from_c_id = :report_client_id and
            hrv_date_end_fact >= :stage_date
      into :dd1, :dd2
  do
  begin
    dd2_ = null;
    kol_days = 0;
    if (dd1 < :begin_of_year) then
      dd1 = :begin_of_year;
    if (dd2 <= end_of_prior_month) then
    begin
      select hrv_date_begin - 1
      from hr_voctravel, sa_worktype swt, hr_orders
      where hrv_date_begin <= :dd2 --:last_day
            and
            hrv_date_end_fact >= :begin_of_year and
            hrv_date_begin <= hrv_date_end_fact and
            hrv_status = 1 and
            ho_order_type = sawt_id and
            hrv_from_ho_id = ho_id and
            swt.sawt_id = 15 --19 /*Отзыв с отпуска (дни сохран-ся)*/
            and
            hrv_from_c_id = :report_client_id and
            hrv_date_end_fact >= :stage_date and
            hrv_date_begin > :dd1
      order by hrv_date_begin
      rows 1
      into :dd2_;

      if (dd2_ is not null) then
        dd2 = dd2_;

      /* все декретные отпуска были до этого месяца */

      if (date_gone < dd2) then
        dd2 = date_gone;

      select kol_days
      from aarep_calc_voc_days(:dd1, :dd2, :report_client_id)
      into :kol_days;
      sat_vocation_begin = sat_vocation_begin - :kol_days;
    end
    if (dd2 > end_of_prior_month) then
    begin
      if (dd2 > last_day) then
        dd2 = :last_day;

      select hrv_date_begin - 1
      from hr_voctravel, sa_worktype swt, hr_orders
      where hrv_date_begin <= :last_day and
            hrv_date_end_fact >= :begin_of_year and
            hrv_date_begin <= hrv_date_end_fact and
            hrv_status = 1 and
            ho_order_type = sawt_id and
            hrv_from_ho_id = ho_id and
            swt.sawt_id = 15 /*Отзыв с отпуска (дни сохран-ся)*/
            /*and sawt_is_paid=0 */
            and
            hrv_from_c_id = :report_client_id and
            hrv_date_end_fact >= :stage_date and
            hrv_date_begin > :dd1
      order by hrv_date_begin
      rows 1
      into :dd2_;
      if (dd2_ is not null) then
        dd2 = dd2_;

      if (dd1 > end_of_prior_month) then
      begin
        /* Декретные отпуск начался в этом месяце */
        select kol_days
        from aarep_calc_voc_days(:dd1, :dd2, :report_client_id)
        into :kol_days;
        sat_vocation_curr = sat_vocation_curr - :kol_days;
      end
      else
      begin
        /* декретный отпуск начался раньше и действует в текущем месяце */
        if (date_gone < end_of_prior_month) then
          end_of_prior_month = date_gone;

        select kol_days
        from aarep_calc_voc_days(:dd1, :end_of_prior_month, :report_client_id)
        into :kol_days;
        sat_vocation_begin = sat_vocation_begin - :kol_days;

        if (date_gone < dd2) then
          dd2 = date_gone;

        select kol_days
        from aarep_calc_voc_days(:end_of_prior_month + 1, :dd2, :report_client_id)
        into :kol_days;

        if (sat_vocation_curr <> 0) then
          sat_vocation_curr = sat_vocation_curr - :kol_days;
      end
    end
  end

  /*

  Для сотрудниц, взявших отпуска по беременности и родам (ID = 17) с 01.01.2015 года,
  должны начисляться в полном объеме дни оплачиваемого отпуска вплоть до конца отпуска.
  согласовано с Чинарой М. 24.02.2015
  */
  for select hrv_date_begin, hrv_date_end_fact
      from hr_voctravel, sa_worktype saw, hr_orders
      where hrv_date_begin <= :last_day and
            hrv_date_end_fact >= :begin_of_year and
            hrv_date_begin <= hrv_date_end_fact and
            hrv_status = 1 and
            ho_order_type = sawt_id and
            hrv_from_ho_id = ho_id and
            saw.sawt_id = 17 /* Отпуск по беременности и родам*/
            and
            hrv_from_c_id = :report_client_id and
            hrv_date_end_fact >= :stage_date
      into :dd1, :dd2
  do
  begin
    if (dd2 >= last_day) then
      dd2 = last_day;

--Декретные отпуска с начала года(за прошлые аннулируются) добавлено по просьбе Чинары Мамрасуловой 25 явн 17г
    if (dd1 < :begin_of_year) then
      dd1 = :begin_of_year;

    if (dd1 > end_of_prior_month) then
    begin
      /* Декретные отпуск начался в этом месяце */
      select kol_days
      from aarep_calc_voc_days(:dd1, :dd2, :report_client_id)
      into :kol_days;
      sat_vocation_curr = sat_vocation_curr + :kol_days;
    end
    else
    begin
      if (dd2 <= end_of_prior_month) then
      begin
        /* декретный отпуск начался раньше и действует в текущем месяце */
        select kol_days
        from aarep_calc_voc_days(:dd1, :dd2, :report_client_id)
        into :kol_days;
        sat_vocation_begin = sat_vocation_begin + :kol_days;
      end
      else
      begin
        /* декретный отпуск начался раньше и действует в текущем месяце */
        select kol_days
        from aarep_calc_voc_days(:dd1, :end_of_prior_month, :report_client_id)
        into :kol_days;
        sat_vocation_begin = sat_vocation_begin + :kol_days;

        select kol_days
        from aarep_calc_voc_days(:end_of_prior_month + 1, :dd2, :report_client_id)
        into :kol_days;
        sat_vocation_curr = sat_vocation_curr + :kol_days;
      end
    end
  end

  /* Уменьшение, если в текущем году увольнялся и нанимался обратно */
  /*for select ps_date_begin, ps_date_end from hr_pstatus
  where ps_from_c_id=:REPORT_CLIENT_ID
  and ps_is_working=0
  and ps_date_end>=:begin_of_year
  and ps_date_end<=:last_day
  order by ps_date_begin
  rows 1
  into :dd1, :dd2
  do
  begin
    if (exists(select * from hr_pstatus
    where ps_from_c_id=:REPORT_CLIENT_ID
    and ps_is_working=1
    and ps_date_begin>:dd2)) then
    begin
        kol_days=0;
        if (dd1<:BEGIN_OF_YEAR) then dd1=:BEGIN_OF_YEAR;
        if (dd2<=END_OF_PRIOR_MONTH) then
        begin
          */
  /* все декретные отпуска были до этого месяца */
  /*
          select KOL_DAYS
          from AAREP_CALC_VOC_DAYS(:dd1, :dd2, :REPORT_CLIENT_ID)
          into :kol_days;
          SAT_VOCATION_BEGIN=SAT_VOCATION_BEGIN-:kol_days;
        end
        if (dd2>END_OF_PRIOR_MONTH) then
        begin
          if (dd2>last_day) then dd2=:last_day;
          if (dd1>END_OF_PRIOR_MONTH) then
          begin
            */
  /* Декретные отпуск начался в этом месяце */
  /*
