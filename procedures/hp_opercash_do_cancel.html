CREATE PROCEDURE HP_OPERCASH_DO_CANCEL(
    IN_OPC_ID D_ID,
    RD_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable reg_id d_integer;
declare variable ol_id d_id;
declare variable opc_from_k_id d_id;
declare variable opc_date d_date;
declare variable tmp_oper d_str_255;
declare variable opc_oper_kind d_smallint;
declare variable kommis_ol_id d_id;
declare variable kommis_opc_id d_id;
declare variable cil_id d_integer;
begin
  res_is_error = 0;

  select coalesce(opc_parent_id, opc_id), opc_office_id, opc_from_ol_id, opc_from_k_id, opc_date, opc_oper_kind
  from opercash
  where opc_id = :in_opc_id
  into in_opc_id, reg_id, ol_id, opc_from_k_id, opc_date, opc_oper_kind;

  if (row_count = 0) then
    exception e_message 'Операция не найдена';

  if (exists(select *
             from hp_is_day_closed(:reg_id, :opc_date, 1)
             where day_closed = 1)) then
    exception e_message 'Опер.день кассы закрыт: ' || f_datetostr(opc_date);

  if (ol_id is not null) then
  begin
    if (exists(select *
               from pvnopers
               where po_from_ol_id = :ol_id and
                     coalesce(po_ol_is_active, 1) = 1)) then
      exception e_message 'Удаление карточных операций запрещена. Обратитесь к курирующему бухгалтеру';

    if (opc_from_k_id is not null) then
    begin
      tmp_oper = '';
      select 'ol_id: ' || ok_from_ol_id || ' от ' || f_datetostr(ol_date)
      from operkred, operlist
      where ok_from_k_id = :opc_from_k_id and
            ok_ol_is_active = 1 and
            ok_from_ol_id = ol_id and
            (ol_id is distinct from :ol_id and ol_parent_id is distinct from :ol_id) and
            (ok_ol_date > :opc_date or (ok_ol_date = :opc_date and ok_from_ol_id > :ol_id))
      rows 1
      into tmp_oper;

      if (tmp_oper <> '') then
        exception e_message 'Имеется более поздняя операция по займу, ' || tmp_oper;
    end

    if (opc_oper_kind = f_enum_id('cashoperkind', 'plus.loan.kommis')) then
    begin
      res_is_error = 1;
      res_text = 'Удаление комиссии за кассовое обслуживание запрещено'; -- RFCKG-4146
    end

    if (exists(select *
               from operlistappr op
               where op.olr_from_ol_id = :ol_id and
                     op.olr_reg_state = 1 and
                     op.olr_appr_status = 0)) then
    begin
      res_is_error = 1;
      res_text = 'Подождите, данный запрос рассматривается.';
    end

    if (res_is_error = 0) then
    begin
      if (rd_id is null) then
        rd_id = 0;

      insert into operlistappr (olr_from_ol_id, olr_reg_state, olr_reg_user, olr_reg_rd_id, olr_reg_datetime,
                                olr_appr_status, olr_appr_user, olr_appr_datetime, olr_appr_rd_id, olr_from_reg_id,
                                olr_is_cashplace)
      values (:ol_id, 1, :un_id, :rd_id, current_timestamp, 0, null, null, :rd_id, :reg_id, 1);

      if (opc_oper_kind = f_enum_id('cashoperkind', 'minus.loan.disb')) then
      begin
        -- при удалении операции выдачи удаляется и комиccия за кассовое обслуживание (RFCKG-4146)
        kommis_ol_id = 0;
        select opc_from_ol_id
        from opercash
        where opc_is_delete = 0 and
              opc_parent_id = :in_opc_id and
              opc_oper_kind = f_enum_id('cashoperkind', 'plus.loan.kommis')
        into kommis_ol_id;

        if (coalesce(kommis_ol_id, 0) <> 0) then
         insert into operlistappr (olr_from_ol_id, olr_reg_state, olr_reg_user, olr_reg_rd_id, olr_reg_datetime,
                                   olr_appr_status, olr_appr_user, olr_appr_datetime, olr_appr_rd_id, olr_from_reg_id,
                                   olr_is_cashplace)
         values (:kommis_ol_id, 1, :un_id, :rd_id, current_timestamp, 0, null, null, :rd_id, :reg_id, 1);
      end

      res_text = 'Создан запрос на аннулирование, ждите подтверждения';
    end
  end else
  begin
    update opercash
    set opc_is_delete = 1
    where opc_id = :in_opc_id and
          opc_is_delete = 0;

    if (opc_oper_kind = f_enum_id('cashoperkind', 'minus.loan.disb')) then
    begin
      -- при удалении операции выдачи удаляется и комиccия за кассовое обслуживание (RFCKG-4146)
      kommis_opc_id = 0;
      select opc_id
      from opercash
      where opc_is_delete = 0 and
            opc_parent_id = :in_opc_id and
            opc_oper_kind = f_enum_id('cashoperkind', 'plus.loan.kommis')
      into kommis_opc_id;

      if (coalesce(kommis_opc_id, 0) <> 0) then
        update opercash
        set opc_is_delete = 1
        where opc_from_ol_id is null and
              opc_id = :kommis_opc_id;
    end

    res_text = 'Операция успешно аннулирована';
  end

  -- RFCKG-5475
  if (opc_oper_kind = f_enum_id('cashoperkind', 'plus.inkassa')) then
  begin
    cil_id = (select opc_from_cil_id from opercash where opc_id = :in_opc_id);
    delete from cashterminal
    where ctm_from_cil_id = :cil_id;
    update cashinvlist set
      cil_is_delete = 1
    where cil_id = :cil_id;
  end

  suspend;
end
