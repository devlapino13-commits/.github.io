CREATE PROCEDURE HP_EDOC_TRANSFORM_OBJ_KIND(
    EDL_OBJECT_KIND D_INTEGER,
    EDL_OBJECT_ID D_ID)
RETURNS (
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID)
AS
declare variable spm_from_ma_id d_integer;
declare variable spm_from_sbc_id d_id;
begin
    /*
        RFCRU-4254
        Процедура преобразует edl_object_kind, который отличается от clients/miniapp/kontract
        к одному из этих типов, чтобы затем по ним определить mcl_project
    */

    if (f_get_db_id() <> 81001) then
        exit;
    
    if (edl_object_kind = 51) then -- com_list
    begin
        -- Из этой таблицы мы можем достать контракт
        select
            cml.cml_object_kind,
            cml.cml_object_id
        from com_list cml
        where cml.cml_id = :edl_object_id
        into 
            :obj_kind,
            :obj_id;
        
        -- Если тип объекта не совпадает с ожидаемым, явно вызываем ошибку
        if (obj_kind <> f_objkind('kontract', 'id')) then
            exception e_default f_format('Unexpected obj_kind. Expected: kontract, actual: %1', f_objkind(:obj_kind, 'name'));

    end
    else if (edl_object_kind = 56) then -- kontpsk
    begin
        obj_kind = f_objkind('kontract', 'id');

        select kps.kps_from_k_id
        from kontpsk kps
        where kps.kps_id = :edl_object_id
        into :obj_id;
    end
    else if (edl_object_kind = 59) then -- ft_anketa_list
    begin
        obj_kind = f_objkind('clients', 'id');

        select fal.fal_from_c_id
        from ft_anketa_list fal
        where fal.fal_id = :edl_object_id
        into :obj_id;
    end
    else if (edl_object_kind = 74) then -- sud_mainreq
    begin
        obj_kind = f_objkind('kontract', 'id');

        select smr.smr_from_k_id
        from sud_mainreq smr
        where smr.smr_id = :edl_object_id
        into :obj_id;
    end
    else if (edl_object_kind = 117) then -- overpay_main
    begin
        obj_kind = f_objkind('clients', 'id');

        select opm.opm_from_c_id
        from overpay_main opm
        where opm.opm_id = :edl_object_id
        into :obj_id;
    end
    else if (edl_object_kind = 118) then -- smevpass_msg
    begin
        select spm.spm_from_sbc_id, spm.spm_from_ma_id
        from smevpass_msg spm
        where spm.spm_id = :edl_object_id
        into :spm_from_sbc_id, :spm_from_ma_id;

        if (spm_from_ma_id is not null) then
        begin
            obj_kind = f_objkind('miniapp', 'id');
            obj_id = spm_from_ma_id;
        end
        else
        begin
            obj_kind = f_objkind('clients', 'id');

            select sbc.sbc_from_c_id
            from smevpass_blockclients sbc
            where sbc.sbc_id = :spm_from_sbc_id
            into :obj_id;
        end
    end

    suspend;
end
