CREATE PROCEDURE HP_BKI_CONTRACTCHANGESLIST_UX
RETURNS (
    BOD_ID D_ID,
    K_ID D_ID,
    K_KONTRACT_NO D_STR_CODE,
    STATUS D_STR_CODE,
    SEND_STATUS D_STR_CODE,
    SEND_DATE D_DATE,
    CHANGE_DATE D_DATE,
    CHANGE_TYPE D_STR_CODE,
    CHANGE_TYPE_TXT D_STR_LONG,
    SPECIAL_TYPE D_STR_CODE,
    SPECIAL_TYPE_TXT D_STR_LONG,
    TYPE_TEXT D_STR_LONG,
    APPLY_DATE D_DATE,
    END_DATE D_DATE,
    STATUS_INT D_INTEGER)
AS
declare variable bod_add d_str_long;
declare variable bkc_id d_id;
declare variable bkp_status d_integer;
declare variable bev_status d_integer;
declare variable jso d_integer;
declare variable bkb_id d_id;
begin
  for select bod_id, bod_object_id, bod_add, bod_from_bkc_id, k_kontract_no
  from bki_once_data
  left join kontract on k_id = bod_object_id
  where bod_kind = 3
  into bod_id, k_id, bod_add, bkc_id, k_kontract_no
  do
  begin
    send_date = null;
    change_date = null;
    change_type = null;
    change_type_txt = null;
    special_type = null;
    special_type_txt = null;
    type_text = null;
    apply_date = null;
    end_date = null;
    bkb_id = null;
    status_int = null;
    status = null;

    jso = json.parse(bod_add);
    change_date = json.obj_get_val(jso, 'date');
    change_type = json.obj_get_val(jso, 'type');
    special_type = json.obj_get_val(jso, 'specialType');
    type_text = json.obj_get_val(jso, 'typeText');
    apply_date = json.obj_get_val(jso, 'applyDate');
    end_date = json.obj_get_val(jso, 'endDate');
    status_int = json.obj_get_val(jso, 'status');

    select bct_code || ' - ' || bct_name
    from bki_catalog
    where bct_kind = 'contractChanges_type'
      and bct_code = :change_type
    into change_type_txt;

    select bct_code || ' - ' || bct_name
    from bki_catalog
    where bct_kind = 'contractChanges_subtype' || trim(iif(:change_type = '1', '', :change_type))
      and bct_code = :special_type
    into special_type_txt;

    select bkp_date, bkp_status, bev_status
    from bki_packet, bki_loans, bki_clients, bki_events
    where bkl_from_bkp_id = bkp_id
      and bkc_from_bkl_id = bkl_id
      and bev_from_bkc_id = bkc_id
      and coalesce(bev_delete_bev_id, 0) = 0
      and bkc_id = :bkc_id
      and bev_event_id = 8
      and bev_event_date = :change_date
    order by bkp_status desc, bev_status desc
    rows 1
    into send_date, bkp_status, bev_status;

    if (send_date is null) then
      select bkb_id
      from bki_buffer
      where bkb_object_kind = 9
        and bkb_object_id = :k_id
        and bkb_date = :change_date
        and bkb_type = 0
      rows 1
      into bkb_id;

    send_status = case
                when send_date is null and bkb_id is null then 'Не отправлено'
                when send_date is null then 'В очереди'
                when bkp_status < 0 then 'Ошибка пакета'
                when bev_status < 0 then 'Ошибка БКИ'
                when bev_status < 3 then 'Отправлено'
                when bev_status = 3 then 'Обработано'
                else 'Неопределено'
              end;

    if (status_int is not null) then
    begin
      status = case
                when status_int = 0 then 'Новый'
                when status_int = 1 then 'Готово'
                when status_int = 2 then 'В очереди'
                when status_int = -1 then 'Отмена отправки'
                else '-'
              end;
    end

    suspend;
  end
end
