CREATE PROCEDURE FHP_CURRENCYSTR_KG(
    VAL NUMERIC(15,2))
RETURNS (
    CURR_STR VARCHAR(1000))
AS
declare variable sign_of_val d_str_10;
declare variable cents d_str_10;
declare variable grade d_smallint;
declare variable val_str d_str_20;
declare variable num d_str_20;
declare variable i d_integer;
declare variable res d_str_255;
declare variable buf d_str_255;
begin
  curr_str = '';
  sign_of_val = ''; -- знак числа
  if (val < 0) then
  begin
    sign_of_val = 'минус ';
    val = -val;
  end

  -- выбираем и запоминаем тыйыны, убираем их из числа
  val_str = cast(val as varchar(20));
  i = position('.' in val_str);
  cents = lpad(substring(val_str from i + 1 for 2), 2, '0');
  val_str = lpad(substring(val_str from 1 for i - 1), ((i + 1) / 3 * 3), '0');

  -- разбираем число
  grade = 1;
  while (val_str != '') do
  begin
    res = '';
    num = right(val_str, 3); -- берём первый класс числа (первые три цифры справа)

    if (num != '000') then
    begin
      -- сотни
      i = cast(substring(num from 1 for 1) as int);
      res = trim(case
                   when i = 1 then 'жүз'
                   when i = 2 then 'эки жүз'
                   when i = 3 then 'үч жүз'
                   when i = 4 then 'т�©рт жүз'
                   when i = 5 then 'беш жүз'
                   when i = 6 then 'алты жүз'
                   when i = 7 then 'жети жүз'
                   when i = 8 then 'сегиз жүз'
                   when i = 9 then 'тогуз жүз'
                   else ''
                 end);

      -- десятки
      i = cast(substring(num from 2 for 1) as int);
      buf = case
              when i = 1 then 'он'
              when i = 2 then 'жыйырма'
              when i = 3 then 'отуз'
              when i = 4 then 'кырк'
              when i = 5 then 'элүү'
              when i = 6 then 'алтымыш'
              when i = 7 then 'жетимиш'
              when i = 8 then 'сексен'
              when i = 9 then 'токсон'
              else ''
            end;
      res = trim(res || ' ' || buf);

      -- единицы
      i = cast(substring(num from 3 for 1) as int);
      buf = case
              when i = 1 then 'бир'
              when i = 2 then 'эки'
              when i = 3 then 'үч'
              when i = 4 then 'т�©рт'
              when i = 5 then 'беш'
              when i = 6 then 'алты'
              when i = 7 then 'жети'
              when i = 8 then 'сегиз'
              when i = 9 then 'тогуз'
              else ''
            end;
      res = trim(res || ' ' || buf);

      -- класс числа
      buf = case
              when grade = 2 then 'миң'
              when grade = 3 then 'миллион'
              when grade = 4 then 'миллиард'
              when grade = 5 then 'триллион'
              when grade = 6 then 'квадриллион'
              when grade = 7 then 'квинтиллион'
              else ''
            end;
      res = trim(res || ' ' || buf);
    end

    if (curr_str != '' and res != '') then
      curr_str = ' ' || curr_str;
    curr_str = res || curr_str; -- словесное представление класса ставим в начало результирующей строки

    val_str = left(val_str, char_length(val_str) - 3); -- следующий класс числа
    grade = grade + 1; -- увеличиваем счётчик класса
  end

  curr_str = sign_of_val || curr_str; -- знак числа
  curr_str = upper(substring(curr_str from 1 for 1)) || substring(curr_str from 2); -- первая буква прописная

  curr_str = curr_str || ' сом';
  if (cents > 0) then
    curr_str = curr_str || ' ' || cents || ' тыйын';

  suspend;
end
