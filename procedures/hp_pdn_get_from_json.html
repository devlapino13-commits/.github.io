CREATE PROCEDURE HP_PDN_GET_FROM_JSON(
    JSO_FULL D_INTEGER,
    REP_DATE D_DATE,
    EIC_CODE D_STR_CODE)
RETURNS (
    CREDUID D_STR_CODE,
    CREDCURRENCY D_STR_CODE,
    CREDDATE D_DATE,
    CREDENDDATE D_DATE,
    CREDENDDATEFACT D_DATE,
    CREDUPDATE D_DATE,
    CREDSUMDEBT D_CURRENCY,
    CREDSUMOVERDUE D_CURRENCY,
    CREDDAYOVERDUE D_INTEGER,
    CREDFULLCOST D_CURRENCY,
    CREDSUM D_CURRENCY,
    CREDPARTNERTYPE D_STR_10,
    CREDTYPE D_STR_10,
    CREDCOLLATERAL D_INTEGER,
    CREDCOLLATERALFACTENDDATE D_DATE,
    CREDGUARANTEE D_INTEGER,
    CREDGUARANTEEFACTENDDATE D_DATE,
    CREDCATEGORY D_STR_10,
    CREDCONSUMER D_SMALLINT,
    CREDRATIO D_STR_10,
    CREDPURPOSE D_STR_10,
    CREDCOLLATERALITEMTYPE D_STR_10,
    CREDACTIVE_STR D_STR_255,
    SIGNCREDITCARD D_BOOL,
    FULLCURRENTDEBT D_CURRENCY,
    MINSUMPAYCC D_CURRENCY,
    PERCENTENDDATE D_DATE,
    CREDIT_LINE_TYPE D_STR_10,
    CRED_OWNER_TYPE D_STR_10,
    AVERAGE_PAYMENT_DATE D_DATE,
    AVERAGE_PAYMENT_SUM D_CURRENCY,
    GRACEDATE D_DATE,
    GRACEENDDATE D_DATE,
    DEALSIGNCREDITCARD D_SMALLINT,
    FULLCOST_SUM D_CURRENCY,
    FULLCOST_DATE D_DATE,
    DEBTCURR_OPSUM D_CURRENCY,
    DEBTCURR_OTHERSUM D_CURRENCY,
    DEBTCURR_PERCENTSUM D_CURRENCY,
    DEBTOVERDUE_OPSUM D_CURRENCY,
    DEBTOVERDUE_PERCENTSUM D_CURRENCY,
    DEBTOVERDUE_OTHERSUM D_CURRENCY,
    DEBTOVERDUE_DATE D_DATE)
AS
begin
  if (eic_code = 'equifax_40') then
    for select creduid, credcurrency, creddate, credenddate, credenddatefact, credupdate, credsumdebt, credsumoverdue,
               creddayoverdue, credfullcost, credsum, credpartnertype, credtype, credcollateral, credcollateralfactenddate,
               credguarantee, credguaranteefactenddate, credcategory, credconsumer, credratio, credpurpose,
               credcollateralitemtype, credactive_str, signcreditcard, fullcurrentdebt, minsumpaycc, percentenddate,
               credit_line_type, cred_owner_type, average_payment_date, average_payment_sum,
               gracedate, graceenddate, dealsigncreditcard, fullcost_sum, fullcost_date, debtcurr_opsum,
               debtcurr_othersum, debtcurr_percentsum, debtoverdue_opsum, debtoverdue_percentsum, debtoverdue_othersum, debtoverdue_date
    from hp_pdn_get_from_equifax_json(:jso_full, :rep_date)
    into :creduid, :credcurrency, :creddate, :credenddate, :credenddatefact, :credupdate, :credsumdebt, :credsumoverdue,
         :creddayoverdue, :credfullcost, :credsum, :credpartnertype, :credtype, :credcollateral,
         :credcollateralfactenddate, :credguarantee, :credguaranteefactenddate, :credcategory, :credconsumer,
         :credratio, :credpurpose, :credcollateralitemtype, :credactive_str, :signcreditcard, :fullcurrentdebt,
         :minsumpaycc, :percentenddate, :credit_line_type, :cred_owner_type, :average_payment_date, :average_payment_sum,
         :gracedate, :graceenddate, :dealsigncreditcard, :fullcost_sum, :fullcost_date, :debtcurr_opsum,
         :debtcurr_othersum, :debtcurr_percentsum, :debtoverdue_opsum, :debtoverdue_percentsum, :debtoverdue_othersum, debtoverdue_date
    do
      suspend;

  else if (eic_code = 'nbki_40') then
    for select creduid, credcurrency, creddate, credenddate, credenddatefact, credupdate, credsumdebt, credsumoverdue,
               creddayoverdue, credfullcost, credsum, credpartnertype, credtype, credcollateral, credcollateralfactenddate,
               credguarantee, credguaranteefactenddate, credcategory, credconsumer, credratio, credpurpose,
               credcollateralitemtype, credactive_str, signcreditcard, fullcurrentdebt, minsumpaycc, percentenddate,
               average_payment_date, average_payment_sum
    from hp_pdn_get_from_nbki_json(:jso_full, :rep_date)
    into :creduid, :credcurrency, :creddate, :credenddate, :credenddatefact, :credupdate, :credsumdebt, :credsumoverdue,
         :creddayoverdue, :credfullcost, :credsum, :credpartnertype, :credtype, :credcollateral,
         :credcollateralfactenddate, :credguarantee, :credguaranteefactenddate, :credcategory, :credconsumer,
         :credratio, :credpurpose, :credcollateralitemtype, :credactive_str, :signcreditcard, :fullcurrentdebt,
         :minsumpaycc, :percentenddate, :average_payment_date, :average_payment_sum
    do
      suspend;

  else
    exception e_default 'unknown eic_code: ' || coalesce(eic_code, '<NULL>');
end
