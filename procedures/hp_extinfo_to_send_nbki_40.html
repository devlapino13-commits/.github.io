CREATE PROCEDURE HP_EXTINFO_TO_SEND_NBKI_40(
    C_ID ID_TYPE,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    JSON_VALUE D_BLOB_TEXT)
AS
declare variable lastname d_str_100;
declare variable firstname d_str_100;
declare variable middlename d_str_100;
declare variable birthday d_str_20;
declare variable consentdate d_str_20;
declare variable admcodeinform d_str_10;
declare variable json_docs d_str_8100;
declare variable json_fio d_str_8100;
declare variable doc_country d_str_10;
declare variable doc_type d_str_10;
declare variable doc_serial d_str_10;
declare variable doc_number d_str_20;
declare variable doc_date d_str_20;
declare variable doc_who d_str_255;
declare variable request_reason_code d_str_10;
declare variable app_sum decimal(15,2);
declare variable consenthash d_str_64;
declare variable consentperiod d_str_10;
declare variable contractdate d_str_20;
declare variable doc_department_code d_str_255;
declare variable c_inn d_str_20;
declare variable inn_checked d_bool;
declare variable first_call d_bool;
begin
  json_docs = '';
  json_fio = '';

  for select doc_country, doc_type, doc_serial, doc_number, doc_date, doc_who, doc_department_code
  from hp_bki_get_info4req_doc(:c_id)
  into :doc_country, :doc_type, :doc_serial, :doc_number, :doc_date, :doc_who, :doc_department_code
  do
  begin
      json_docs = json_docs || iif(json_docs <> '', ',', '') ||
          '{' ||
              f_json_make_pair('id_type', doc_type, 'str', 1) ||
              f_json_make_pair('oksm', doc_country, 'str', 1) ||
              f_json_make_pair('series_number', doc_serial, 'str', 1) ||
              f_json_make_pair('id_num', doc_number, 'str', 1) ||
              f_json_make_pair('issue_authority', doc_who, 'str', 1) ||
              f_json_make_pair('issue_date', doc_date, 'date', 1) ||
              f_json_make_pair('div_code', doc_department_code, 'str') ||
          '}';
  end

  select birthday, consentdate, admcodeinform, request_reason_code,
         app_sum, consenthash, consentperiod, contractdate, c_inn, inn_checked
  from hp_bki_get_info4req(:c_id, :object_kind, :object_id)
  into :birthday, :consentdate, :admcodeinform, :request_reason_code,
       :app_sum, :consenthash, :consentperiod, :contractdate, c_inn, inn_checked;

  first_call = 1;

  for select lastname, firstname, middlename
  from hp_bki_get_info4req_fio(:c_id)
  into :lastname, :firstname, :middlename
  do
  begin
    json_fio = json_fio || iif(json_fio <> '', ',', '') ||
          '{' ||
              f_json_make_pair('name1',  lastname,  'str', 1) ||
              f_json_make_pair('first', firstname, 'str', 1) ||
              f_json_make_pair('paternal', middlename, 'str', first_call) ||
              iif(first_call = 1, f_json_make_pair('birth_dt', birthday, 'str'), '')||
          '}';
    first_call = 0;
  end 

  json_value ='{'||
             '"person_req":['|| json_fio || '],'||
             '"id_req":[' || json_docs || '],'||
                        trim(iif(coalesce(c_inn, '') <> '' and (current_date >= '01.03.2025' or not exists(select 1 from hp_db_is_actual)),
                            '"inn":{' ||
                              f_json_make_pair('code', 1, 'str' ,1) ||
                              f_json_make_pair('no', c_inn, 'str' ,1) ||
                              f_json_make_pair('checked', inn_checked, 'str') ||
                            '},',
                        ''))||
             '"inquiry_req":{'||
                f_json_make_pair('inq_purpose', request_reason_code, 'str', 1) ||
                f_json_make_pair('inq_amount', replace(cast(app_sum as d_str_20), '.', ','), 'str', 1) ||

                '"consent_req": {' ||
                  f_json_make_pair('liability', admcodeInform, 'str', 1) ||
                  f_json_make_pair('consent_date', consentdate, 'str', 1) ||
                  f_json_make_pair('consent_period', consentperiod, 'str', 1) ||
                  f_json_make_pair('agreement_date', contractdate, 'str', 1) ||
                  f_json_make_pair('consent_hash', consenthash, 'str', 1) ||
                  trim(iif(coalesce(c_inn, '') <> '' and (current_date >= '01.03.2025' or not exists(select 1 from hp_db_is_actual)),
                    f_json_make_pair('report_user_tax_id', c_inn, 'str', 1),
                  ''))||
                  f_json_make_pair('consent_purpose', request_reason_code, 'str') ||
                '}'||
             '}'||
           '}';
  suspend;
end
