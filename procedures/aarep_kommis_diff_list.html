CREATE PROCEDURE AAREP_KOMMIS_DIFF_LIST(
    NEW_VALUE D_PERCENT,
    SHOW_ROW_COUNT D_INTEGER)
RETURNS (
    REC_ID D_INTEGER,
    K_ID D_ID,
    K_DISB_DATE D_DATE,
    CLIENT_COUNT D_INTEGER,
    ORIG_RATE_YEAR D_PERCENT,
    ORIG_AMOUNT D_CURRENCY,
    ORIG_PERC D_CURRENCY,
    ORIG_KOMMIS D_CURRENCY,
    NEW_RATE_YEAR D_PERCENT,
    NEW_AMOUNT D_CURRENCY,
    NEW_PERC D_CURRENCY,
    DIFF_CLIENT_COUNT D_INTEGER,
    DIFF_ALL_PERC D_CURRENCY,
    DIFF_PLUS_PERC D_CURRENCY)
AS
declare variable k_percent_per_day d_currency;
begin
  -- Если new_perc <= 2, то это коэффициент
  -- Иначе это абсолютное значение к проценту

  if (show_row_count is null) then
    show_row_count = 0;

  rec_id = 0;
  for select ks_from_k_id, ks_date_begin, k_percent_per_day, k_kolich_client_approved
  from kontstate, kontract, product
  where ks_state = 1 and
        ks_date_begin >= '08.07.2020' and
        ks_ol_is_delete = 0 and
        ks_from_k_id = k_id and
        k_from_pr_id = pr_id and
        pr_percent_kind in (0, 1) and
        k_kommis_perc <> 0
  into k_id, k_disb_date, k_percent_per_day, client_count
  do
  begin
    rec_id = rec_id + 1;

    orig_amount = 0;
    orig_perc = 0;
    orig_kommis = 0;
    orig_rate_year = round(k_percent_per_day * 365.0, 1);

    select sum(sh_os), sum(sh_perc), sum(sh_kommis)
    from schedule
    where sh_from_k_id = :k_id
    into orig_amount, orig_perc, orig_kommis;
    if (orig_amount is null) then
      orig_amount = 0;
    if (orig_perc is null) then
      orig_perc = 0;
    if (orig_kommis is null) then
      orig_kommis = 0;

    if (orig_kommis = 0) then
      exception e_default f_format('Zero kommis, k_id = %1', k_id);

    new_amount = orig_amount - orig_kommis;

    if (new_value <= 2)
    then k_percent_per_day = round(k_percent_per_day * new_value, 7);
    else k_percent_per_day = k_percent_per_day + round(new_value / 365.0000000, 7);

    if (k_percent_per_day = 0) then
      exception e_default f_format('Zero new interest rate, k_id = %1', k_id);

    new_rate_year = round(k_percent_per_day * 365.0, 2);

    new_perc = 0;
    select sum(sh_perc)
    from aarep_kommis_diff_sh(:k_id, :new_amount, :k_percent_per_day)
    into :new_perc;
    if (new_perc is null) then
      new_perc = 0;

    diff_client_count = 0;
    diff_all_perc = 0;
    diff_plus_perc = 0;

    diff_all_perc = orig_perc - new_perc;

    if (diff_all_perc > 0) then
    begin
      diff_client_count = client_count;
      diff_plus_perc = diff_all_perc;
    end

    suspend;

    if (show_row_count > 0 and rec_id > show_row_count) then
      exit;
  end
end
