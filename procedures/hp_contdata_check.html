CREATE PROCEDURE HP_CONTDATA_CHECK(
    NEED_FIX D_SMALLINT = 0,
    EDIT_CHECK_STATUS D_SMALLINT = 1,
    CHECK_ALL_CLIENTS D_SMALLINT = 0)
RETURNS (
    C_ID D_ID,
    RD_CODE D_STR_100,
    STATUS D_STR_100,
    VALUE_OLD D_STR_100,
    VALUE_NEW D_STR_100)
AS
begin
  if (need_fix is null) then
    need_fix = 0;

  if (edit_check_status is null) then
    edit_check_status = 1;

  if (check_all_clients is null) then
    check_all_clients = 0;

  for select cci_object_id
  from continfo
  where coalesce(cci_check_status, 0) <= :check_all_clients
  into c_id
  do
  begin
    if (need_fix = 1) then
    begin
      rd_code = '';
      for select rd_code
      from hp_continfo_fix(:c_id)
      into :rd_code
      do
        rd_code = '';
    end

    for select rd_code, status, value_old, value_new
        from hp_contdata_one_check(:c_id, :edit_check_status)
        into :rd_code, :status, :value_old, :value_new
    do
      Suspend;
  end
end
