CREATE PROCEDURE UXP$UXGRID_FIELD_MODIFY(
    RES_CHECKED D_SMALLINT,
    GRID_UUID D_UUID,
    GRID_UUID_CODE D_STR_CODE,
    UI_FIELD_NAME D_STR_CODE,
    UI_CAPTION D_STR_100,
    UI_POSITION D_INTEGER,
    UI_WIDTH D_INTEGER,
    UI_SIZABLE D_BOOL,
    UI_ORDER_INDEX D_INTEGER,
    UI_GROUP_INDEX D_INTEGER,
    UI_FIXED_KIND D_INTEGER,
    UI_SUMMARY_KIND D_INTEGER,
    UI_BAND_ID D_INTEGER,
    UI_HINT D_STR_LONG,
    UI_HIDDEN D_SMALLINT,
    UI_READONLY D_BOOL,
    UI_HEADER_ALIGN D_SMALLINT = null,
    UI_VALUE_ALIGN D_SMALLINT = null)
AS
declare variable orig_ui_position d_integer;
declare variable pos d_integer;
declare variable uf_uuid d_uuid;
declare variable ui_uuid d_uuid;
declare variable ui_ufr_uuid d_uuid;
begin
  if (coalesce(grid_uuid, '') = '') then
  begin
    if (grid_uuid_code <> '') then
    begin
      pos = position('__', grid_uuid_code);
      if (pos > 0) then
      begin
        grid_uuid_code = substring(grid_uuid_code from pos + 2);
        grid_uuid_code = replace(grid_uuid_code, '_', '-');
      end
    end

    grid_uuid = uxf$uniq(grid_uuid_code, 0);
  end

  if (ui_header_align is null) then
    ui_header_align = 0;

  if (ui_value_align is null) then
    ui_value_align = 0;

  select ugr_ufr_uuid
  from ux$grid
  where ugr_uuid = :grid_uuid
  into ui_ufr_uuid;

  if (ui_ufr_uuid is null) then
    exception e_default 'Grid frame not found';

  if (ui_field_name similar to '[-]?[[:DIGIT:]]+') then
  begin
    if (res_checked = 1) then
    begin
      if (ui_band_id = -1) then
        ui_band_id = null;

      update or insert into ux$band (bnd_grid_uuid, bnd_id, bnd_parent_id, bnd_position, bnd_caption,
                                     bnd_hidden, bnd_width, bnd_fixed_kind)
      values (:grid_uuid, :ui_field_name, :ui_band_id, :ui_position, :ui_caption, :ui_hidden,
              :ui_width, :ui_fixed_kind)
      matching (bnd_grid_uuid, bnd_id);
    end
  end else
  begin
    select uf_uuid
    from ux$frparam, ux$field
    where ufp_ufr_uuid = :ui_ufr_uuid and
          ufp_uf_uuid = uf_uuid and
          uf_field_name = :ui_field_name
    rows 1
    into uf_uuid;
  
    if (uf_uuid is null) then
      exception e_default 'Field not found: ' || coalesce(ui_field_name, '<Null>');
  
    select ugi_uuid, ugi_position
    from ux$gritem
    where ugi_ugr_uuid = :grid_uuid and
          ugi_uf_uuid = :uf_uuid
    into ui_uuid, orig_ui_position;
  
    if (ui_order_index is null) then
      ui_order_index = 0;
  
    if (ui_group_index is null) then
      ui_group_index = 0;
  
    if (ui_position < 0) then
      ui_position = 0;
  
    if (res_checked = 0) then
    begin
      delete from ux$gritem
      where ugi_uuid = :ui_uuid;
    end else
    begin
      if (ui_position is null) then
        ui_position = orig_ui_position;

      select coalesce(:ui_header_align, ugi_header_align), coalesce(:ui_value_align, ugi_value_align)
      from ux$gritem
      where ugi_uuid = :ui_uuid
      into :ui_header_align, :ui_value_align;

      update or insert into ux$gritem
             (ugi_ugr_uuid, ugi_caption, ugi_uf_uuid, ugi_position, ugi_width, ugi_bnd_id,
              ugi_sizable, ugi_order_index, ugi_group_index, ugi_fixed_kind, ugi_summary_kind,
              ugi_hint, ugi_hidden, ugi_readonly, ugi_header_align, ugi_value_align)
      values (:grid_uuid, :ui_caption, :uf_uuid, :ui_position, :ui_width, :ui_band_id,
              :ui_sizable, :ui_order_index, :ui_group_index, :ui_fixed_kind, :ui_summary_kind,
              :ui_hint, :ui_hidden, :ui_readonly, :ui_header_align, :ui_value_align)
      matching (ugi_ugr_uuid, ugi_uf_uuid);
    end
  end
end
