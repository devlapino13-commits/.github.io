CREATE PROCEDURE MC_REL_CLIENT_CALCULATOR_PARAMS(
    MCL_ID D_ID,
    BY_POINT D_SMALLINT,
    POINT D_INTEGER)
RETURNS (
    AMOUNT_MIN D_INTEGER,
    AMOUNT_MAX D_INTEGER,
    PERIOD_MIN D_INTEGER,
    PERIOD_MAX D_INTEGER)
AS
declare variable c_id d_id;
declare variable pr_id d_integer;
declare variable point_cl d_integer;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  mcl_id = coalesce(mcl_id, -777);
  if (mcl_id <= 0) then
    exception e_default '"mcl_id" is not valid';

  by_point = coalesce(by_point, -777);
  if (by_point not in (0, 1)) then
    exception e_default '"by_point" is not valid';

  point = coalesce(point, -777);
  if (point < 0) then
    exception e_default '"point" is not valid';

  period_min = 3;
  period_max = 12;

  select mc.mcl_from_c_id
  from mc_login mc
  where mc.mcl_id = :mcl_id
  into :c_id;

  if (db_id in (39801, 41701)) then
    select pr.pr_id
    from product pr
    where pr.pr_code like '%.consultation.%' rows 1
    into :pr_id;

  if (db_id = 81001) then
    select pr.pr_id
    from product pr
    where pr.pr_code like '%.reliable_client.%' and
          current_date between pr.pr_date_begin and pr.pr_date_end and
          pr.pr_active = 1
    into :pr_id;

  if (db_id = 94401) then
    select pr.pr_id
    from product pr
    where pr.pr_code like '%.score_rate.%' and
          pr.pr_active = 1
    order by pr.pr_id desc
    rows 1
    into :pr_id;

  point_cl = point;
  if (by_point = 0) then -- расчет исходя из баллов клиента
    point_cl = f_point(c_id, current_date);

  select amount_min, amount_max
  from perc_amount_interval_by_point(:pr_id, :point_cl)
  into :amount_min, :amount_max;

  if (amount_min is null) then
    exception e_default 'amount_min is null';
  if (amount_max is null) then
    exception e_default 'amount_max is null';
  if (period_min is null) then
    exception e_default 'period_min is null';
  if (period_max is null) then
    exception e_default 'period_max is null';

  suspend;
end
