CREATE PROCEDURE AAREP_KANT_FACTORY_REPORT(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_DISB_DATE D_DATE,
    REG_NAME D_STR_255,
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    SUM_TOTAL_PRICE DECIMAL(18,2),
    GUARANTEE_NO D_STR_100,
    OB_SUMMA_VAL D_CURRENCY)
AS
declare variable params d_str_long;
declare variable trans_id d_str_20;
declare variable k_id d_id;
begin
  for select kt.k_id, kt.k_disb_date, cl.c_name_fio, kt.k_kontract_no, kt.k_amount_approved, esw.esw_trans_id,
             rg.reg_name
      from kontract kt
      inner join region rg on rg.reg_unicode = kt.k_fact_disb_reg_id
      inner join extminiapp ema on ema.ema_from_tkt_id = kt.k_from_tkt_id
      inner join clients cl on cl.c_id = ema.ema_from_c_id
      inner join extsysweb esw on esw.esw_id = ema.ema_from_esw_id
      inner join kontstate ks on ks.ks_from_k_id = kt.k_id
      where ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0 and
            esw.esw_project = 'kant_factory' and
            ema.ema_status = 1 and
            esw.esw_command = 'app_save'
      into k_id, k_disb_date, c_name_fio, k_kontract_no, sum_total_price, trans_id, reg_name
  do
  begin

    select esw.esw_full_source
    from extsysweb esw
    where esw.esw_trans_id = :trans_id and
          esw.esw_command = 'loan_disb' and
          esw.esw_project = 'kant_factory'
    order by esw.esw_datetime desc
    rows 1
    into params;

    guarantee_no = (select field_value
                    from hp_extsys_parcer(:params, 'act_number'));

    select sum(ob.ob_summa_val)
    from operbook ob
    inner join operlist ol on ol.ol_id = ob.ob_from_ol_id
    inner join kontstate ks on ks.ks_from_ol_id = ol.ol_parent_id
    where ks.ks_from_k_id = :k_id and
          ob.ob_short_acc_dt = '11531/99'
    into ob_summa_val;

    suspend;
  end
end
