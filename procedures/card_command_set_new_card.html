CREATE PROCEDURE CARD_COMMAND_SET_NEW_CARD(
    CS_ID D_INTEGER)
AS
declare variable cci_id d_integer;
declare variable cca_id d_integer;
declare variable cs_object_id d_id;
declare variable cs_object_kind d_integer;
declare variable card_id d_integer;
declare variable cs_request_json d_str_8100;
declare variable office_id d_integer;
declare variable un_id d_id;
declare variable k_id d_id;
declare variable c_id d_id;
begin
  select cs_from_cca_id,
         cs_object_id,
         cs_object_kind,
         cs_card_id,
         cs.cs_from_k_id,
         cs.cs_from_c_id
  from cardsync cs
  where cs.cs_id = :cs_id
  into :cca_id,
       :cs_object_id,
       :cs_object_kind,
       :card_id,
       :k_id,
       :c_id;

    -- если клиент зарегистрирован
    cci_id = (select cca_from_cci_id
              from cardclientacc
              where cca_id = :cca_id);

    select ca.cac_from_reg_id,
           ca.cac_from_un_id
    from cardaktclient ca
    where cac_id = :cs_object_id
    into :office_id,
         :un_id;

    update cardclientid set
      cci_status = 1
    where cci_id = :cci_id;

    update cardclientacc set
      cca_status = 0
    where cca_id = :cca_id;

    -- новая задача по подвязке карты к картсчету
    cs_request_json = f_format('{"card_id":%1, "card_acct":%2, "product":"%3"}',
      :card_id,
      :cca_id,
      trim(iif(exists(select * from hp_db_is_actual), '21', '021')));
    insert into cardsync(
      cs_parent_id,
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_request_json,
      cs_office_id,
      cs_from_un_id,
      cs_from_k_id,
      cs_from_c_id
      )
    values(
      :cs_id,
      f_get_cardsynccfg('set_new_card', 'id'),
      :card_id,
      :cca_id,
      :cs_object_kind,
      :cs_object_id,
      :cs_request_json,
      :office_id,
      :un_id,
      :k_id,
      :c_id
      );
end
