CREATE PROCEDURE HP_INTERN_SEND_EMAIL(
    I_ID D_INTEGER,
    PROCESS D_STR_100)
AS
declare variable mail_to d_str_255;
declare variable mail_subject d_str_255;
declare variable mail_body blob sub_type 1 segment size 80;
declare variable un_fio d_str_100;
declare variable reg_name d_str_100;
declare variable jl_name d_str_100;
declare variable intern_fio d_str_100;
begin
  select un.un_fio, reg.reg_code || ' ' || reg.reg_name, jl.jl_name, c.c_name_fio
  from interns i
  inner join username un on i.i_registred_by = un.un_id
  inner join region reg on i.i_from_reg_id = reg.reg_unicode
  inner join joblist jl on i.i_from_jl_id = jl.jl_id
  inner join clients c on i.i_from_c_id = c.c_id
  where i.i_id = :i_id
  into :un_fio, :reg_name, :jl_name, :intern_fio;

  if (process = 'registration') then
  begin
    mail_to = 'hr_kg.mirm@mbulak.kg';
    mail_subject = 'Уведомление о регистрации стажера. ' || :reg_name;
    mail_body = 'От <b>' || :un_fio || '</b><br><br>' || 'В офис <b>' || :reg_name || '</b> на позицию <b>' || :jl_name || '</b> был(а) зарегистрирован(а) ' || :intern_fio;
  end
  if (process = 'personnel_committee') then
  begin
    mail_to = 'hr_branch@molbulak.kg';
    mail_subject = 'Уведомление об утверждении рекомендации на стажировку. ' || :reg_name;
    mail_body = 'В офис <b>' || :reg_name || '</b> на позицию <b>' || :jl_name || '</b> был(а) утвржден(а) на стажировку <b>' || :intern_fio || '</b>';
  end
  if (process = 'hr_first_step') then
  begin
    mail_to = 'acc.branch@mbulak.kg, acc.zp@molbulak.kg';
    mail_subject = 'Уведомление о стажировке. ' || :reg_name;
    mail_body = 'Был составлен типовой договор: Стажер ' || :intern_fio || '('|| :jl_name ||') офис: ' || :reg_name || '.';
  end
  if (process = 'educational_committee') then
  begin
    mail_to = 'acc.branch@mbulak.kg, acc.zp@molbulak.kg';
    mail_subject = 'Уведомление о завершении стажировки. ' || :reg_name;
    mail_body = 'Стажер ' || :intern_fio || '('|| :jl_name ||') офис: ' || :reg_name || ' успешно завершил стажировку.';

    execute procedure hp_send_quick_message('email', :mail_to, :mail_subject, :mail_body, '', 1, 1000000000000);

    mail_to = 'hr_branch@molbulak.kg, acc.zp@molbulak.kg';
    mail_subject = 'Уведомление об окончании стажировки. ' || :reg_name;
    mail_body = 'Принять в штат: Стажер ' || :intern_fio || '('|| :jl_name ||') офис: ' || :reg_name || '.';
  end

  execute procedure hp_send_quick_message('email', :mail_to, :mail_subject, :mail_body, '', 1, 1000000000000);
end
