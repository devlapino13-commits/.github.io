CREATE PROCEDURE HP_FIND_KONTRACT_UX(
    FIND_TEXT D_STR_100,
    REP_REGION_ID D_INTEGER,
    KONT_STATE D_SMALLINT,
    DISB_KIND D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    RES_K_ID D_ID,
    RES_CAPTION D_STR_100,
    RES_KONTRACT_NO D_STR_20,
    RES_DISB_DATE D_DATE,
    RES_DISB_AMOUNT D_CURRENCY,
    RES_CLIENT_COUNT D_INTEGER,
    RES_OFFICE_CODE D_STR_100,
    RES_FOLDER_CODE D_STR_CODE,
    RES_LO_FIO D_STR_100,
    RES_PRODUCT_NAME D_STR_100,
    RES_CLIENT_LIST D_STR_4096,
    RES_STATUS_CODE D_STR_CODE,
    RES_OFFICE_ID D_INTEGER,
    RES_LO_ID D_ID,
    RES_FROM_PR_ID D_INTEGER)
AS
declare variable kf_id d_id;
declare variable oks_close_state d_smallint;
begin
  for select distinct res_k_id
  from hp_find_kontract_ux_tmp(:find_text, :rep_region_id, :kont_state, :disb_kind, :un_id)
  into :res_k_id
  do
  begin
    for select k_kontract_no, k_disb_date, k_amount_approved, k_kolich_client_approved,
               pr_id, pr_name, k_from_kf_id, k_kontract_no || ' от ' || f_datetostr(k_disb_date)
    from kontract, product
    where k_id = :res_k_id and
          k_from_pr_id = pr_id
    into res_kontract_no, res_disb_date, res_disb_amount, res_client_count,
         res_from_pr_id, res_product_name, kf_id, res_caption
    do
    begin
      res_office_code = '';
      res_lo_fio = '';
      res_folder_code = '';
      res_client_list = '';
      res_status_code = '';

      if (kont_state = 0) then
        res_status_code = 'В процессе';
      else
      begin
        oks_close_state = null;

        select oks_close_state
        from oksummary
        where oks_from_k_id = :res_k_id and
              oks_date_begin <= current_date and
              oks_date_end >= current_date
        into oks_close_state;

        res_status_code = trim(case coalesce(oks_close_state, 2)
          when 0 then 'Активный'
          when 1 then 'Спец.учет'
          when 2 then 'Закрыт'
        end);
      end

      select reg_code || ' ' || reg_name, c_name_fio, a.kal_new_kf_name,
             reg_unicode, c_id
      from knaktlist a, region, clients
      where a.kal_from_kf_id = :kf_id and
            a.kal_date_end >= '31.12.2999' and
            a.is_delete = 0 and
            a.kal_new_reg_id = reg_unicode and
            a.kal_new_kf_officer = c_id
      into res_office_code, res_lo_fio, res_folder_code,
           res_office_id, res_lo_id;

      select left(list(c_name_fio), 4000)
      from knclient, clients
      where kc_from_k_id = :res_k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id
      into res_client_list;

      if (char_length(res_client_list) = 4000) then
        res_client_list = res_client_list || '...';

      suspend;
    end
  end
end
