CREATE PROCEDURE AAREP_REPORT_CENTER_BANK(
    REP_DATE_END DATE)
RETURNS (
    PLAN_OS_91_ CURR_TYPE,
    PLAN_OS_31_90 CURR_TYPE,
    PLAN_OS_1_30 CURR_TYPE,
    DELINQ_OS_31_ CURR_TYPE,
    DELINQ_OS_6_30 CURR_TYPE,
    DELINQ_OS_0_5 CURR_TYPE,
    CALC_PERC_91_ CURR_TYPE,
    CALC_PERC_31_90 CURR_TYPE,
    CALC_PERC_1_30 CURR_TYPE,
    DELINQ_CALC_PERC_31_ CURR_TYPE,
    DELINQ_CALC_PERC_6_30 CURR_TYPE,
    DELINQ_CALC_PERC_0_5 CURR_TYPE)
AS
declare variable oks_from_k_id decimal(18,0);
declare variable oks_balance decimal(18,4);
declare variable oks_delinq_date date;
declare variable sh_date date;
declare variable zad_os decimal(18,4);
declare variable sh_calc_perc decimal(18,4);
declare variable delinq_day integer;
declare variable t_plan_os_91_ curr_type;
declare variable t_plan_os_31_90 curr_type;
declare variable t_plan_os_1_30 curr_type;
declare variable t_delinq_os_31_ curr_type;
declare variable t_delinq_os_6_30 curr_type;
declare variable t_delinq_os_0_5 curr_type;
declare variable t_calc_perc_91_ curr_type;
declare variable t_calc_perc_31_90 curr_type;
declare variable t_calc_perc_1_30 curr_type;
declare variable t_delinq_calc_perc_31_ curr_type;
declare variable t_delinq_calc_perc_6_30 curr_type;
declare variable t_delinq_calc_perc_0_5 curr_type;
begin
  plan_os_91_ = 0;
  plan_os_31_90 = 0;
  plan_os_1_30 = 0;
  delinq_os_31_ = 0;
  delinq_os_6_30 = 0;
  delinq_os_0_5 = 0;

  calc_perc_91_ = 0;
  calc_perc_31_90 = 0;
  calc_perc_1_30 = 0;
  delinq_calc_perc_31_ = 0;
  delinq_calc_perc_6_30 = 0;
  delinq_calc_perc_0_5 = 0;

  for select oks_from_k_id, oks_balance, oks_delinq_date
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end
--  and oks_delinq_date >= :rep_date_end
--  and oks_from_k_id = 101010015302800101
            and oks_close_state = 0
      into oks_from_k_id, oks_balance, oks_delinq_date
  do
  begin
    t_plan_os_91_ = 0;
    t_plan_os_31_90 = 0;
    t_plan_os_1_30 = 0;
    t_delinq_os_31_ = 0;
    t_delinq_os_6_30 = 0;
    t_delinq_os_0_5 = 0;

    t_calc_perc_91_ = 0;
    t_calc_perc_31_90 = 0;
    t_calc_perc_1_30 = 0;
    t_delinq_calc_perc_31_ = 0;
    t_delinq_calc_perc_6_30 = 0;
    t_delinq_calc_perc_0_5 = 0;

    for select sh_date, zad_os, sh_calc_perc
        from aarep_make_zad_list(:oks_from_k_id, :rep_date_end, 1)
        where sh_date is not null
        into sh_date, zad_os, sh_calc_perc
    do
    begin
--      if (zad_os < 0) then zad_os = 0;
--      if (sh_calc_perc < 0) then sh_calc_perc = 0;

      delinq_day = rep_date_end - sh_date;

      if (delinq_day <= -91) then
      begin
--      план платежи 91 и более дней
        t_plan_os_91_ = t_plan_os_91_ + zad_os;
        t_calc_perc_91_ = t_calc_perc_91_ + sh_calc_perc;
      end
      else
      if (delinq_day <= -31) then
      begin
--      план плотежи от 31 до 90 дней
        t_plan_os_31_90 = t_plan_os_31_90 + zad_os;
        t_calc_perc_31_90 = t_calc_perc_31_90 + sh_calc_perc;
      end
      else
      if (delinq_day <= -1) then
      begin
--      план платежи от 1 до 30 дней
        t_plan_os_1_30 = t_plan_os_1_30 + zad_os;
        t_calc_perc_1_30 = t_calc_perc_1_30 + sh_calc_perc;
      end
      else
      if (delinq_day >= 31) then
      begin
--      просроченные более 30 дней
        t_delinq_os_31_ = t_delinq_os_31_ + zad_os;
        t_delinq_calc_perc_31_ = t_delinq_calc_perc_31_ + sh_calc_perc;
      end
      else
      if (delinq_day >= 6) then
      begin
--      просроченные 6 - 30 дней
        t_delinq_os_6_30 = t_delinq_os_6_30 + zad_os;
        t_delinq_calc_perc_6_30 = t_delinq_calc_perc_6_30 + sh_calc_perc;
      end
      else
      if (delinq_day >= 0) then
      begin
--      просроченные 0 - 5 дней
        t_delinq_os_0_5 = t_delinq_os_0_5 + zad_os;
        t_delinq_calc_perc_0_5 = t_delinq_calc_perc_0_5 + sh_calc_perc;
      end
--      suspend;
    end
    if (t_plan_os_91_ < 0) then
      t_plan_os_91_ = 0;
    if (t_plan_os_31_90 < 0) then
      t_plan_os_31_90 = 0;
    if (t_plan_os_1_30 < 0) then
      t_plan_os_1_30 = 0;
    if (t_delinq_os_31_ < 0) then
      t_delinq_os_31_ = 0;
    if (t_delinq_os_6_30 < 0) then
      t_delinq_os_6_30 = 0;
    if (t_delinq_os_0_5 < 0) then
      t_delinq_os_0_5 = 0;

    if (t_calc_perc_91_ < 0) then
      t_calc_perc_91_ = 0;
    if (t_calc_perc_31_90 < 0) then
      t_calc_perc_31_90 = 0;
    if (t_calc_perc_1_30 < 0) then
      t_calc_perc_1_30 = 0;
    if (t_delinq_calc_perc_31_ < 0) then
      t_delinq_calc_perc_31_ = 0;
    if (t_delinq_calc_perc_6_30 < 0) then
      t_delinq_calc_perc_6_30 = 0;
    if (t_delinq_calc_perc_0_5 < 0) then
      t_delinq_calc_perc_0_5 = 0;

    plan_os_91_ = plan_os_91_ + t_plan_os_91_;
    plan_os_31_90 = plan_os_31_90 + t_plan_os_31_90;
    plan_os_1_30 = plan_os_1_30 + t_plan_os_1_30;
    delinq_os_31_ = delinq_os_31_ + t_delinq_os_31_;
    delinq_os_6_30 = delinq_os_6_30 + t_delinq_os_6_30;
    delinq_os_0_5 = delinq_os_0_5 + t_delinq_os_0_5;

    calc_perc_91_ = calc_perc_91_ + t_calc_perc_91_;
    calc_perc_31_90 = calc_perc_31_90 + t_calc_perc_31_90;
    calc_perc_1_30 = calc_perc_1_30 + t_calc_perc_1_30;
    delinq_calc_perc_31_ = delinq_calc_perc_31_ + t_delinq_calc_perc_31_;
    delinq_calc_perc_6_30 = delinq_calc_perc_6_30 + t_delinq_calc_perc_6_30;
    delinq_calc_perc_0_5 = delinq_calc_perc_0_5 + t_delinq_calc_perc_0_5;

  end
  suspend;
end
