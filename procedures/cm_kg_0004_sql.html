CREATE PROCEDURE CM_KG_0004_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_client_pass_no varchar(30);
declare variable c_id id_type;
declare variable c_birth_date d_date;
declare variable c_cell_no_date d_date;
declare variable c_doc_type d_integer;
declare variable c_verify_status d_integer;
declare variable k_kontract_no d_str_20;
declare variable crel_other_c_id id_type;
declare variable ma_refund d_smallint;
declare variable pr_code d_str_100;
declare variable oks_delinq_date d_date;
declare variable relation_type d_str_20;
declare variable c_name_fio d_str_100;
begin
  if (ma_id = 3573740) then -- RFCKG-608 запрос от Чолпон
    exit;
  for select mac_from_c_id, c_name_fio, c_pass_no, c_birth_date, c_verify_status, c_doc_type, ma_refund, pr_code
      from miniappclient macl, clients, miniapp maa, product
      where mac_from_ma_id = :ma_id and
            (mac_amount > 0 or
            mac_is_poruch = 1) and
            c_type = 0 and
            mac_from_c_id = c_id and
            maa.ma_id = macl.mac_from_ma_id and
            maa.ma_is_active = 1 and
            maa.ma_date_end = '31.12.9999' and
            maa.ma_from_pr_id = pr_id
      into :c_id, :res_info_text, :kf_client_pass_no, :c_birth_date, c_verify_status, c_doc_type, :ma_refund, pr_code
  do
  begin
    if (c_doc_type = 1) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'С 1-апреля 2017-года, паспорта образца 1994-года являются недействительными';
      suspend;
    end
    c_cell_no_date = null;
    select c_cell_no_date
    from aarep_get_continfo('clients', :c_id)
    into c_cell_no_date;
    res_error_text = '';
    res_error_is_critical = 0;
    if (:c_birth_date is null) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Дата рождения не заполнена';
      suspend;
    end
    if (coalesce(:kf_client_pass_no, '') = '') then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Номер паспорта не заполнен';
      suspend;
    end
    --      Пункт 7, проверка клиента на просрочку
    k_kontract_no = null;
    for select k_kontract_no, oks_delinq_date
        from oksummary, knclient, kontract
        where kc_from_c_id = :c_id and
              kc_amount_approved > 0 and
              kc_from_k_id = oks_from_k_id and
              current_date between oks_date_begin and oks_date_end and
              current_date > oks_delinq_date and
              oks_from_k_id = k_id and
              oks_close_state = 0
        into k_kontract_no, oks_delinq_date
    do
    begin
      if (k_kontract_no is not null and
          not(oks_delinq_date between '01.12.20' and '31.01.21' and
          pr_code containing '.par.' and
          (pr_code containing '.ind.' or pr_code containing '.gr.' or pr_code containing '.birge.') and
          exists(select *
                 from knclient_compens kcc
                 inner join knclient kncl on kncl.kc_id = kcc.kcc_from_kc_id and
                       kc_from_c_id = :c_id
                 where kcc.kcc_date_end > kcc.kcc_date_begin))) then


        if (exists(select *
                   from miniapp maa
                   inner join product pr on pr.pr_id = ma_from_pr_id
                   where maa.ma_id = :ma_id and
                         maa.ma_date_end = '31.12.9999' and
                         pr.pr_code containing '.Actiya_Ishenim_Refin.' and
                         maa.ma_is_active = 1)) then
          exit;

      begin
        res_error_is_critical = iif(ma_refund = 1, 0, 1);
        res_error_text = 'У клиента имеется активный просроченный кредит в контракте №' || k_kontract_no;
        suspend;
      end
    end

    -- Находит родственников клиента и проверяет на просрочку (RFCKG-2429, RFCKG-4542)
    for select crel_other_c_id, c_name_fio, lower(rel_name)
        from hp_client_relation_list(:c_id, 0)
        where crel_from_rel_id in ('1', '2', '3', '4', '5', '6')
        into crel_other_c_id, c_name_fio, relation_type
    do
    begin
        if (crel_other_c_id is null) then
          crel_other_c_id = 0;
        k_kontract_no = null;
        select k_kontract_no
        from oksummary, knclient, kontract
        where kc_from_c_id = :crel_other_c_id and
              kc_amount_approved > 0 and
              kc_from_k_id = oks_from_k_id and
              current_date between oks_date_begin and oks_date_end and
              current_date > oks_delinq_date and
              oks_close_state = 0 and
              oks_from_k_id = k_id rows 1
        into k_kontract_no;
        if (k_kontract_no is not null) then
        begin
          res_error_is_critical = iif(ma_refund = 1, 0, 1);
          res_error_text = c_name_fio || ' (' || relation_type || ' клиента) имеет активный просроченный займ № ' || k_kontract_no;
          suspend;
        end
    end
  end
end
