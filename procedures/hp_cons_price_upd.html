CREATE PROCEDURE HP_CONS_PRICE_UPD(
    CP_ID D_ID,
    CP_FROM_PR_ID D_ID,
    CP_MIN_POINT D_SMALLINT,
    CP_MAX_POINT D_SMALLINT,
    CP_MIN_AMOUNT D_CURRENCY,
    CP_MAX_AMOUNT D_CURRENCY,
    CP_PRICE D_CURRENCY,
    CP_LEVEL D_SMALLINT,
    CP_PERIOD D_SMALLINT,
    CP_IS_ACTIVE D_SMALLINT,
    CP_FROM_UN_ID D_ID)
AS
begin
    update consul_price
    set cp_min_point = :cp_min_point,
        cp_max_point = :cp_max_point,
        cp_min_amount = :cp_min_amount,
        cp_max_amount = :cp_max_amount,
        cp_price = :cp_price,
        cp_level = :cp_level,
        cp_period = :cp_period,
        cp_is_active = :cp_is_active,
        cp_datetime = current_timestamp,
        cp_from_un_id = :cp_from_un_id
    where cp_id = :cp_id;
end
