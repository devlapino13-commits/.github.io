CREATE PROCEDURE HP_UNDERWRITINGCERTIFSPF_REGUPD(
    C_ID D_ID,
    REG_ID D_INTEGER,
    DATE_BEGIN D_DATE,
    UN_ID D_ID)
AS
declare variable old_reg d_integer;
declare variable ucfs_id d_id;
declare variable old_date_end d_date;
begin
  select ucfs.ucfs_id, ucfs.ucfs_from_reg_id, ucfs.ucfs_date_end
  from underwritingcertifspf ucfs
  where ucfs.ucfs_from_c_id = :c_id
  into :ucfs_id, :old_reg, :old_date_end;

  if (:ucfs_id is null) then
  begin
    insert into underwritingcertifspf (ucfs_from_c_id, ucfs_from_reg_id, ucfs_date_begin, ucfs_active, ucfs_from_un_id)
    values (:c_id, :reg_id, :date_begin, 1, :un_id);
  end
  else
  begin
    if (:old_reg is null or :old_reg = :reg_id) then
      exit;

    else
      if (old_date_end < :date_begin) then
        date_begin = :old_date_end;

      update underwritingcertifspf
      set ucfs_from_reg_id = :reg_id,
          ucfs_date_begin = :date_begin,
          ucfs_from_un_id = :un_id
      where (ucfs_id = :ucfs_id);
  end
end
