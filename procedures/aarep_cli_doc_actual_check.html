CREATE PROCEDURE AAREP_CLI_DOC_ACTUAL_CHECK(
    REGION_ID D_INTEGER = 1000000,
    REP_DATE D_DATE = current_date)
RETURNS (
    NUMB D_INTEGER,
    C_ID D_ID,
    C_NAME_FIO STR_100,
    C_BIRTH_DATE D_DATE,
    C_IS_RESIDENT D_INTEGER,
    RESIDENT D_STR_10,
    KS_STATE D_INTEGER,
    STATE D_STR_10,
    DOC_TYPE D_STR_20,
    DOC_DATE D_DATE,
    DOC_DATE_EXPIRE D_DATE,
    K_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    K_KFOLDER_NAME D_STR_100,
    OFFICE_NAME D_STR_CODE,
    CPDS_TIMESTAMP D_DATETIME,
    CITIZEN D_STR_100)
AS
declare variable REG_ID D_INTEGER;
declare variable KF_ID D_ID;
declare variable DT_FINISH D_DATE;
declare variable RCOUNT D_INTEGER;
begin
  numb = 0;
  delete from temp_rep;
  for
    select
      k_last_reg_id,
      k_from_kf_id,
      kal_new_kf_name,
      k_id,
      k_kontract_no,
      ks_state,
      k_disb_date,
      kc_from_c_id
  from kontstate
    inner join kontract
      on k_id = ks_from_k_id
    inner join knclient
      on kc_from_k_id = k_id
        and kc_amount_approved > 0
    inner join knaktlist
      on kal_from_kf_id = k_from_kf_id
        and :rep_date between kal_date and kal_date_end
        and knaktlist.is_delete = 0
    inner join region_list(:region_id, 1, -1, 1)
      on id_list = kal_new_reg_id
  where :rep_date between ks_date_begin and ks_date_end
    and ks_state in (1, 2)
    and ks_ol_is_delete = 0
  order by k_disb_date
  into
    :reg_id,
    :kf_id,
    :k_kfolder_name,
    :k_id,
    :k_kontract_no,
    :ks_state,
    :k_disb_date,
    :c_id
  do
  begin
    -- повтор
    if (exists(
      select * from temp_rep
      where id = :c_id))
    then continue;

    -- прекращение отношений
    dt_finish = coalesce((
      select last_date
      from ft_date_close_calc(:k_id, :c_id, :rep_date)), '31.12.9999');
    if (dt_finish <= rep_date) then
      continue;

    -- свидетельство о смерти
    if (exists(
      select * from clientdocs
      where cds_from_c_id = :c_id
        and cds_from_pk_id = 91
        and cds_doc_date_begin <= :rep_date))
    then continue;

    -- закрытый займ
    if (ks_state = 2 and not exists(
      select oko_from_k_id
      from operkroff
      where oko_from_k_id = :k_id
        and oko_ol_is_active = 1
      group by oko_from_k_id
      having sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc) > 0))
    then continue;

    for
      select
        c_id,
        c_name_fio,
        c_birth_date,
        c_is_resident,
        c_pass_date,
        c_pass_date_expire,
        rd.rd_name
      from clients
      inner join refdata rd on rd.rd_id = c_citizen
      where c_id = :c_id
    into
      :c_id,
      :c_name_fio,
      :c_birth_date,
      :c_is_resident,
      :doc_date,
      :doc_date_expire,
      :citizen
    do
    begin
      resident = iif(c_is_resident = 1, '+', '');
      state = iif(ks_state = 2, '+', '');
      office_name = (
        select
          reg_name
        from region
        where reg_unicode = :reg_id);
      cpds_timestamp = (
        select first 1
          cpds_timestamp
        from clipassdelaysms
        where cpds_c_id = :c_id
        order by cpds_timestamp desc);

      if (not (rep_date between doc_date and coalesce(doc_date_expire, '31.12.9999')))
      then begin
        doc_type = 'Паспорт';

        numb = numb + 1;
        insert into temp_rep(id) values(:c_id);
      if (coalesce(doc_date, '31.12.9999') <> '31.12.9999' and
          coalesce(doc_date_expire, '31.12.9999') <> '31.12.9999') then
           suspend;
      end

      rcount = 0;
      if (c_is_resident = 0) then
      begin
        for
          select first 1
            iif(cds_from_pk_id = 39, 'ВНЖ', 'РВП'),
            cds_doc_date_begin,
            coalesce(cds_doc_date_end, '31.12.9999')
          from clientdocs
          where cds_from_c_id = :c_id
            and cds_from_pk_id in (39, 46)
            and cds_is_delete = 0
            and cds_verify_status = 2
          order by cds_datetime desc
        into
          :doc_type,
          :doc_date,
          :doc_date_expire
        do
        begin
          -- дата добавления миграционной карты > даты окончания действия РВП | ВНЖ
          if (not (rep_date between doc_date and doc_date_expire)
            and exists(
              select *
              from clientdocs
              where cds_from_c_id = :c_id
                and cds_from_pk_id = 47
                and cds_create_datetime > :doc_date_expire
                and cds_is_delete = 0
                and cds_verify_status = 2))
          then
            rcount = 0;
          else
            rcount = 1;
        end
      end
      if (rcount = 0 or (rep_date between doc_date and coalesce(doc_date_expire, '31.12.9999')))
        then continue;

      numb = numb + 1;
      insert into temp_rep(id) values(:c_id);
      if (coalesce(doc_date, '31.12.9999') <> '31.12.9999' and
          coalesce(doc_date_expire, '31.12.9999') <> '31.12.9999') then
           suspend;
    end
  end
end
