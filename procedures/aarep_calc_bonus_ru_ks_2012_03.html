CREATE PROCEDURE AAREP_CALC_BONUS_RU_KS_2012_03(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(50),
    AA_CURR_BRANCH_NAME VARCHAR(50),
    AA_CURR_KC_FIO VARCHAR(100),
    PS_IS_ON_STAFF_STR VARCHAR(20),
    AA_AMOUNT_DISB INTEGER,
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_DELIQ_BALANCE7 DECIMAL(18,4),
    AA_PAR DECIMAL(15,2),
    AA_KOLICH_CL_DISB INTEGER,
    AA_BONUS DECIMAL(18,4),
    AA_BONUS_FACTOR INTEGER,
    AA_BONUS_END DECIMAL(18,4),
    JL_NAME VARCHAR(100),
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    AA_KOLICH_CL_CLOSE INTEGER,
    AA_BONUS_CLOSE DECIMAL(18,4))
AS
begin
  for select max(aa_curr_office_name), max(aa_curr_branch_name), aa_curr_kc_fio, max(jl_name), max(ps_is_on_staff_str),
             sum(aa_amount_disb), sum(aa_amount_active_end), sum(aa_deliq_balance7), sum(aa_par),
             sum(aa_kolich_cl_disb), sum(aa_bonus_factor), sum(aa_kolich_cl_active_end), sum(aa_kolich_cl_close)
      from aarep_cb_ru_ks_2012_03_draft(:rep_region_id, :rep_date_begin, :rep_date_end)
      group by aa_curr_kc_fio
      into :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_kc_fio, :jl_name, :ps_is_on_staff_str, :aa_amount_disb,
           :aa_amount_active_end, :aa_deliq_balance7, :aa_par, :aa_kolich_cl_disb, :aa_bonus_factor,
           :aa_kolich_cl_active_end, :aa_kolich_cl_close

  do
  begin
    aa_bonus_close = 0;
    aa_bonus = :aa_kolich_cl_disb * 500;
    aa_bonus_end = :aa_bonus / 100 * aa_bonus_factor;
    if (aa_bonus_factor > 0) then
      aa_bonus_close = aa_kolich_cl_close * 1000;
    aa_bonus_end = :aa_bonus / 100 * aa_bonus_factor + aa_bonus_close;
    suspend;
  end

end
