CREATE PROCEDURE AAREP_MINIAPP_TEMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_AUTO_PERIOD INTEGER,
    REP_AUTO_KOEF DECIMAL(18,7),
    REP_INCLUDE_FIRST_DAY INTEGER,
    REP_WITH_REPAY INTEGER)
RETURNS (
    MA_DATE_OPER DATE,
    MA_SUMMA_PLUS_OS DECIMAL(18,4),
    MA_SUMMA_PLUS_PERC DECIMAL(18,4),
    MA_SUMMA_PLUS_ADMIN DECIMAL(18,4),
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_ADMIN_PLUS DECIMAL(18,4),
    MA_ADMIN_MINUS DECIMAL(18,4),
    MA_TRANSFER_PLUS DECIMAL(18,4),
    MA_TRANSFER_MINUS DECIMAL(18,4),
    MA_SUMMA_MINUS_REPEAT DECIMAL(18,4),
    MA_SUMMA_MINUS_NEW DECIMAL(18,4),
    MA_SUMMA_MINIAPPNEW DECIMAL(18,4),
    MA_ADMIN_MINUS_PROCESS DECIMAL(18,4),
    MA_REGION_ID INTEGER,
    MA_KF_ID DECIMAL(18,0),
    MA_DISB_KOLICH INTEGER,
    MA_UNIQ_ID INTEGER,
    LAST_DD DATE)
AS
declare variable ma_id_prior integer;
declare variable ma_id integer;
declare variable ma_number integer;
declare variable ma_is_active integer;
declare variable k_disb_date date;
declare variable k_amount decimal(18,4);
declare variable ma_office_id integer;
declare variable ma_admin_perc decimal(18,7);
declare variable ma_admin_perc1 decimal(18,7);
declare variable dd_new date;
declare variable sh_dd date;
declare variable k_id_prior decimal(18,0);
declare variable kf_id decimal(18,0);
declare variable kf_id1 decimal(18,0);
declare variable sh_amount decimal(18,4);
declare variable k_id decimal(18,0);
declare variable ma_is_new integer;
declare variable prod_id integer;
declare variable k_status integer;
declare variable ma_pr_id integer;
declare variable ma_pr_id_prior integer;
declare variable new_app_amount decimal(18,4);
declare variable is_disb integer;
declare variable cm_miniapp_clear_day integer;
begin
  select cm_miniapp_clear_day
  from company
  into :cm_miniapp_clear_day;
  if (cm_miniapp_clear_day is null) then
    cm_miniapp_clear_day = 3;

  ma_uniq_id = 0;
  if (rep_with_repay is null) then
    rep_with_repay = 1;
  ma_disb_kolich = 0;
  ma_pr_id_prior = -1;
  dd_new = :rep_date_begin + :rep_auto_period;
  ma_id_prior = -1;
  ma_summa_minus_repeat = 0;
  ma_summa_minus_new = 0;
  ma_summa_plus_os = 0;
  ma_summa_plus_perc = 0;
  ma_summa_plus_admin = 0;
  ma_transfer_plus = 0;
  ma_transfer_minus = 0;
  ma_admin_plus = 0;
  ma_admin_minus = 0;
  ma_is_new = 0;
  ma_summa_miniappnew = 0;
  ma_admin_minus_process = 0;
  ma_summa_minus = 0;

  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      order by id_list
      into :ma_region_id
  do
  begin
    ma_date_oper = :rep_date_begin;
    while (ma_date_oper <= :rep_date_end) do
    begin
      suspend;
      ma_date_oper = ma_date_oper + 1;
    end
  end
  /* Выдачи по предварительным заявкам */
  for select ma_id, ma_number, ma_date_oper, ma_is_active, ma_summa_minus, ma_office_id, ma_from_kf_id, ma_from_pr_id
      from miniapp, region_list(:rep_region_id, 1, -1, 1)
      where ma_date_app < (:rep_date_begin + :rep_include_first_day) and
            ma_date_end >= (:rep_date_begin + :rep_include_first_day) and
            ma_date_oper between :rep_date_begin - 60 and :rep_date_end and
            ma_office_id = id_list
      order by ma_id desc, ma_number desc
      into :ma_id, :ma_number, :ma_date_oper, :ma_is_active, :ma_summa_minus, :ma_office_id, :ma_kf_id, :ma_pr_id
  do
  begin
    if ((ma_pr_id <> ma_pr_id_prior) or (ma_pr_id_prior is null)) then
    begin
      ma_admin_perc = 0;

      select prl_admin_perc_min
      from prodlogic
      where prl_from_pr_id = :ma_pr_id and
            prl_date_begin <= :ma_date_oper and
            prl_date_end >= :ma_date_oper rows 1
      into :ma_admin_perc;
      if (ma_admin_perc is null) then
        ma_admin_perc = 0;
    end
    ma_region_id = :ma_office_id;
    ma_summa_plus_admin = 0;
    if (ma_id_prior <> ma_id) then
    begin
      if (ma_is_active <= 0) then
        ma_summa_minus = 0;
      k_amount = 0;
      k_status = 0;
      if (ma_is_active <= 0) then
      begin
        ma_summa_minus = 0;
        ma_admin_perc = 0;
      end
      ma_is_new = 0;

      /* Проверка цикла */
      if (ma_summa_minus > 0) then
      begin
        if (ma_date_oper = :rep_date_begin and
            rep_include_first_day = 1 and
            rep_with_repay = 1) then
        begin
          if (exists(select distinct ks_state
                     from kontstate, kontract
                     where k_id = ks_from_k_id and
                           ks_date_begin = :rep_date_begin and
                           ks_state = 1 and
                           k_from_kf_id = :ma_kf_id and
                           ks_ol_is_delete = 0)) then
            is_disb = 1;
          else
            is_disb = 0;
          if (is_disb = 1) then
          begin
            ma_summa_minus = 0;
          end
        end
        if (ma_summa_minus > 0) then
        begin
          if (ma_date_oper >= :rep_date_begin and
              ma_date_oper <= :rep_date_end and
              ma_summa_minus <> 0) then
          begin
            if (exists(select distinct ks_state
                       from kontstate, kontract
                       where k_id = ks_from_k_id and
                             ks_date_begin < :rep_date_begin and
                             ks_state = 1 and
                             k_from_kf_id = :ma_kf_id and
                             ks_ol_is_delete = 0)) then
              ma_is_new = 0;
            else
              ma_is_new = 1;

            if (ma_admin_perc is null) then
              ma_admin_perc = 0;
            ma_summa_plus_admin = :ma_summa_minus * :ma_admin_perc / 100;
            if (ma_is_new = 1) then
            begin
              ma_summa_minus_new = ma_summa_minus;
              ma_summa_minus = 0;
            end
            else
            begin
              ma_summa_minus_new = 0;
            end
            if ((ma_date_oper >= :rep_date_begin) and
                (ma_date_oper <= :rep_date_end)) then
            begin
              ma_disb_kolich = 1;
              suspend;
              ma_disb_kolich = 0;
            end
          end
        end
      end
    end
    ma_id_prior = :ma_id;
  end
  ma_is_new = 0;
  ma_summa_plus_admin = 0;
  ma_summa_minus_new = 0;
  ma_summa_minus = 0;

  /* Планируемые погашения */
  ma_kf_id = null;
  if (rep_with_repay = 1) then
  begin
    for select aarep_region_id, aarep_date_action, sum(aarep_plan_os), sum(aarep_plan_perc)
        from aarep_planning_repays(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_include_first_day)
        group by aarep_region_id, aarep_date_action
        into :ma_region_id, :ma_date_oper, :ma_summa_plus_os, :ma_summa_plus_perc
    do
    begin
      suspend;
    end
  end

  ma_summa_plus_os = 0;
  ma_summa_plus_perc = 0;
  ma_summa_plus_admin = 0;
  ma_summa_minus = 0;
  ma_transfer_plus = 0;
  ma_transfer_minus = 0;
  ma_admin_plus = 0;
  ma_admin_minus = 0;
  k_id_prior = 0;
  prod_id = 0;
  if (dd_new <= :rep_date_end) then
  begin
-- Список последних погашений
    ma_pr_id = -1;
    for select kk_region_id, kk_from_pr_id, kk_from_kf_id, kk_id, kk_amount_approved * kk_disb_curr_rate
        from aarep_active_kont_id_in_period(:rep_region_id, null, (:rep_date_begin + :rep_include_first_day - 1), 0, 0, 1, 0, 0)
        where kk_kredline_end <= :rep_date_end
        into :ma_region_id, :prod_id, :kf_id, :k_id, :sh_amount
    do
    begin
      sh_dd = null;
      for select ok_sh_date
          from operkred
          where ok_from_k_id = :k_id and
                ok_ol_date < (:rep_date_begin + :rep_include_first_day) and
                ok_sh_date >= :dd_new and
                ok_ol_is_active = 1 and
                ok_from_c_id = 0
          group by ok_sh_date
          having sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc) > 0
          order by ok_sh_date desc
          rows 1
          into :sh_dd
      do
      begin
        if (sh_dd <= :rep_date_end) then
        begin
          if (not(exists(select *
                         from miniapp, product
                         where ma_date_oper >= :rep_date_begin and
                               ma_date_app < (:rep_date_begin + :rep_include_first_day) and
                               ma_date_end >= (:rep_date_begin + :rep_include_first_day) and
                               ma_from_kf_id = :kf_id and
                               ma_from_pr_id = pr_id and
                               pr_category = 0))) then
          begin
            ma_kf_id = kf_id;
            kf_id1 = 0;
            ma_date_oper = :sh_dd;
            ma_summa_minus_repeat = :sh_amount + :sh_amount * rep_auto_koef / 100;

            if (prod_id <> ma_pr_id) then
            begin
              ma_admin_perc = 0;
              select distinct prl_admin_perc_min
              from prodlogic
              where prl_from_pr_id = :prod_id and
                    prl_date_begin <= :ma_date_oper and
                    prl_date_end >= :ma_date_oper rows 1
              into :ma_admin_perc;
              if (ma_admin_perc is null) then
                ma_admin_perc = 0;
            end
            ma_pr_id = prod_id;
            ma_summa_plus_admin = :ma_summa_minus_repeat * :ma_admin_perc / 100;
            ma_disb_kolich = 1;
            suspend;
            ma_disb_kolich = 0;
          end
        end
      end
    end
  end
  kf_id = null;
  ma_kf_id = null;
  ma_summa_plus_os = 0;
  ma_summa_plus_perc = 0;
  ma_summa_plus_admin = 0;
  ma_summa_minus = 0;
  ma_transfer_plus = 0;
  ma_transfer_minus = 0;
  ma_admin_plus = 0;
  ma_admin_minus = 0;
  ma_summa_miniappnew = 0;
  ma_summa_minus_repeat = 0;

  /* Прогнозируемые выдачи новым группам */
  for select man_office, man_disb_date, sum(man_amount)
      from miniappnew, region_list(:rep_region_id, 1, -1, 1)
      where (man_office = id_list) and
            man_date < (:rep_date_begin + :rep_include_first_day) and
            man_disb_date between :rep_date_begin and :rep_date_end
      group by man_office, man_disb_date
      having sum(man_amount) <> 0
      into :ma_region_id, :ma_date_oper, :ma_summa_miniappnew
  do
  begin
    new_app_amount = 0;
    if (ma_date_oper < (:rep_date_begin + :cm_miniapp_clear_day)) then
      ma_summa_miniappnew = 0;
    if (ma_summa_miniappnew > 0) then
    begin
      select sum(ma_summa_minus_new)
      from aarep_miniapp_temp2(:ma_region_id, :ma_date_oper, :ma_date_oper, :rep_auto_period, :rep_include_first_day)
--      where MA_DATE_OPER=:ma_date_oper
--      and ma_region_id=:MA_REGION_ID
      into :new_app_amount;
      if (new_app_amount is null) then
        new_app_amount = 0;
      if (ma_summa_miniappnew < new_app_amount) then
        ma_summa_miniappnew = 0;
      else
      begin
        ma_summa_miniappnew = :ma_summa_miniappnew - :new_app_amount;
      end
      if (ma_summa_miniappnew > 0) then
      begin
        ma_uniq_id = :ma_uniq_id + 1;
        suspend;
      end
    end
  end
end
