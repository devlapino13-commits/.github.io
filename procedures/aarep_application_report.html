CREATE PROCEDURE AAREP_APPLICATION_REPORT(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    MA_ID D_INTEGER,
    K_KONTRACT_NO D_STR_20,
    MA_DATE_APP D_DATE,
    MA_KONT_STAT_STR D_STR_20,
    KR_TEXT D_STR_100,
    OFFICE_NAME D_STR_100,
    REGION_NAME D_STR_100,
    KC_AMOUNT_APPLICATE DECIMAL(18,2),
    PR_NAME D_STR_100,
    K_CIRCLE D_INTEGER,
    K_PERIOD D_INTEGER,
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    C_EDUCATION_TEXT D_STR_64,
    C_FAMILY_TEXT D_STR_20,
    C_CITIZEN_TEXT D_STR_20,
    C_NATION_TEXT D_STR_20,
    C_KOL_CHILD D_INTEGER,
    C_KOL_FAMILY D_INTEGER,
    AGE D_INTEGER,
    C_SEX D_STR_10,
    KF_CLIENT_NAME D_STR_100,
    MONTHLY_REPAY DECIMAL(18,2),
    KIB_MBF_NON_DELINQ D_INTEGER,
    KIB_MBF_DELINQ D_INTEGER,
    SUM_MBF_KH_OWNAMOUNT DECIMAL(18,2),
    SUM_MBF_KH_AMOUNT DECIMAL(18,2),
    MAX_MBF_KH_DELIQDAYSCOUNT D_INTEGER,
    ACT_MBF_NON_DELINQ D_INTEGER,
    ACT_MBF_DELINQ D_INTEGER,
    SUM_MBF_KH_NON_DELINQ_BALANCE DECIMAL(18,2),
    SUM_MBF_KH_DELINQ_BALANCE DECIMAL(18,2),
    KIB_MBF_ACT_MAX_DELINQ D_INTEGER,
    KIB_NOMBF_NON_DELINQ D_INTEGER,
    KIB_NOMBF_DELINQ D_INTEGER,
    SUM_NOMBF_KH_OWNAMOUNT DECIMAL(18,2),
    SUM_NOMBF_KH_AMOUNT DECIMAL(18,2),
    MAX_NOMBF_KH_DELIQDAYSCOUNT D_INTEGER,
    ACT_NOMBF_NON_DELINQ D_INTEGER,
    ACT_NOMBF_DELINQ D_INTEGER,
    SUM_NOMBF_KH_NON_DELINQ_BALANCE DECIMAL(18,2),
    SUM_NOMBF_KH_DELINQ_BALANCE DECIMAL(18,2),
    KIB_NOMBF_ACT_MAX_DELINQ D_INTEGER,
    MIN_KH_DATEAWARD D_DATE,
    SCORE_RATE D_STR_100,
    RISK_RATE D_STR_100,
    IS_NEW_CLIENT D_STR_20,
    C_SCORE D_INTEGER)
AS
declare variable REGION_ID D_INTEGER;
declare variable K_ID D_ID;
declare variable K_STATUS D_INTEGER;
declare variable MA_KONT_STATE D_INTEGER;
declare variable C_ID D_ID;
declare variable KBM_ID D_ID;
declare variable KS_CLOSED_DATE D_DATE;
declare variable KH_CREATE_DATE D_DATE;
declare variable K_FACT_DISB_KF_NO D_STR_100;
declare variable KBM_CHECKED_DATE D_DATE;
declare variable C_FAMILY D_INTEGER;
declare variable C_EDUCATION D_INTEGER;
declare variable C_CITIZEN D_INTEGER;
declare variable C_NATION D_INTEGER;
declare variable C_BIRTH_DATE D_DATE;
declare variable PR_ID D_ID;
declare variable MA_IS_ACTIVE D_INTEGER;
begin

  ma_id = 0;
  ma_is_active = 0;
  region_id = 0;
  k_id = 0;
  k_kontract_no = 0;
  k_period = 0;
  k_circle = 0;
  pr_name = '';
  kc_amount_applicate = 0;
  c_id = 0;
  ma_is_active = 0;

  for select ma.ma_id, --№зявки
             ma.ma_is_active, --статус заявки
             ma.ma_date_app, --дата заявки
             ma.ma_office_id,
             k.k_id,
             k.k_status,
             k.k_kontract_no, --№контракта
             k.k_period, --срок
             k.k_circle, --цикл
             pr.pr_name, -- продукт
             kc.kc_amount_applicate, --сумма кл
             kc.kc_from_c_id,
             iif(kc.kc_circle <= 1, 'Новый', 'Повторный')
      from miniapp ma
      inner join kontract k on k.k_from_ma_app_id = ma.ma_id
      inner join knclient kc on kc.kc_from_k_id = k.k_id
      inner join product pr on k.k_from_pr_id = pr.pr_id
      where ma.ma_date_app between :rep_date_begin and :rep_date_end and
            ma.ma_date_end = '31.12.9999' and
            pr.pr_code not containing '.ipf.' and
            kc.kc_amount_approved > 0
      into :ma_id, --№зявки
           :ma_is_active, --статус заявки
           :ma_date_app, --дата заявки
           :region_id,
           :k_id,
           :k_status,
           :k_kontract_no, --№контракта
           :k_period, --срок
           :k_circle, --цикл
           :pr_name, -- продукт
           :kc_amount_applicate, --сумма кл
           :c_id,
           :is_new_client
  do
  begin
 /*По просбе Мирзаитовой.М статус кредита на момент формир-ия отчета*/
    ma_kont_state = null;
    select ks.ks_state
    from kontstate ks
    where ks_from_k_id = :k_id and
          ks_ol_is_delete = 0 and
          current_date between ks.ks_date_begin and ks.ks_date_end
    order by ks.ks_date_begin desc
    rows 1
    into :ma_kont_state;

    if (ma_kont_state is null) then
      ma_kont_state = -1;
    if ((k_status = -1) and
        (ma_kont_state = 0)) then
      ma_kont_state = -2;
    if (ma_kont_state = 0 and
        k_status = -2) then
      ma_kont_state = -1;
    if (ma_kont_state = -2) then
      ma_kont_stat_str = 'Аннулирован';
    if (ma_kont_state = -1) then
      ma_kont_stat_str = 'Отказ';
    if (ma_kont_state = 0) then
      ma_kont_stat_str = 'В процессе';
    if (ma_kont_state = 1) then
      ma_kont_stat_str = 'Активный';
    if (ma_kont_state = 2) then
      ma_kont_stat_str = 'Закрытый';

    office_name = '';
    region_name = '';
    select reg.reg_code || ' ' || reg.reg_name, reg2.reg_name
    from region reg
    inner join region reg2 on reg2.reg_unicode = reg.parent
    where reg.reg_unicode = :region_id and
          reg2.reg_unicode > 0
    into :office_name, --Регион
         :region_name; --Код офиса

    c_name_fio = '';
    c_family = 0;
    c_kol_family = 0;
    c_inn = 0;
    c_kol_child = 0;
    c_sex = 0;
    for select c.c_name_fio, --фио клиета
               c.c_family, --семейное положения
               c.c_kol_family, --кол-во членов семьи
               c.c_education, --оброзования
               c.c_citizen,
               c.c_nation,
               c.c_inn,
               c.c_birth_date,
               c.c_kol_child,
               iif(c.c_sex = 1, 'муж', 'жен') --пол клииента
        from clients c
        where c.c_id = :c_id
        into :c_name_fio,
             :c_family,
             :c_kol_family,
             :c_education,
             :c_citizen,
             :c_nation,
             :c_inn,
             :c_birth_date,
             :c_kol_child,
             :c_sex
    do
    begin

      c_education_text = '';
      select rd.rd_name
      from refdata rd
      where rd.rd_id = :c_education
      into :c_education_text; --Образование кл.

      c_citizen_text = '';
      select rd.rd_name
      from refdata rd
      where rd.rd_id = :c_citizen
      into :c_citizen_text; --Гражд-во кл.

      c_nation_text = '';
      select rd.rd_name
      from refdata rd
      where rd.rd_id = :c_nation
      into :c_nation_text; --Нац-ность кл.

      c_family_text = '';
      select rf.rd_name
      from refdata rf
      where rf.rd_id = :c_family
      into :c_family_text; --семейное положения

      age = 0;
      age = f_age(:c_birth_date, current_date);

      kr_text = '';
      select kr.kr_text
      from kontreslist krl
      inner join kontres kr on krl.krl_from_kr_id = kr.kr_id
      where krl.krl_ma_id = :ma_id
      rows 1
      into :kr_text;--Причина отказа

      kf_client_name = null;
      kbm_checked_date = null;
      monthly_repay = null;
      kib_mbf_non_delinq = null;
      kib_mbf_delinq = null;
      sum_mbf_kh_ownamount = null;
      sum_mbf_kh_amount = null;
      max_mbf_kh_deliqdayscount = null;
      act_mbf_non_delinq = null;
      act_mbf_delinq = null;
      sum_mbf_kh_non_delinq_balance = null;
      sum_mbf_kh_delinq_balance = null;
      kib_mbf_act_max_delinq = null;
      kib_nombf_non_delinq = null;
      kib_nombf_delinq = null;
      sum_nombf_kh_ownamount = null;
      sum_nombf_kh_amount = null;
      max_nombf_kh_deliqdayscount = null;
      act_nombf_non_delinq = null;
      act_nombf_delinq = null;
      sum_nombf_kh_non_delinq_balance = null;
      sum_nombf_kh_delinq_balance = null;
      kib_nombf_act_max_delinq = null;
      min_kh_dateaward = null;

      select list(distinct kf_client_name, ', '),
             --
             min(kbm_checked_date) kbm_checked_date, min(kh_create_date) kh_create_date, --
             max(monthly_repay) monthly_repay, --
             -->> MBulak
             sum(iif(kh_is_our = 1 and kh_deliqdayscount = 0, 1, 0)) kib_mbf_non_delinq, --
             sum(iif(kh_is_our = 1 and kh_deliqdayscount > 0, 1, 0)) kib_mbf_delinq, --
             sum(iif(kh_is_our = 1, kh_ownamount, 0)) sum_mbf_kh_ownamount, --
             sum(iif(kh_is_our = 1, kh_amount, 0)) sum_mbf_kh_amount, --
             max(iif(kh_is_our = 1, kh_deliqdayscount, 0)) max_mbf_kh_deliqdayscount,
             sum(iif(kh_is_our = 1 and kh_is_active = 1 and kh_repaystatus containing 'Акт. пр-ка: 0 дн', 1, 0)) act_mbf_non_delinq,
             sum(iif(kh_is_our = 1 and kh_is_active = 1 and f_strtonum(kh_repaystatus) > 0, 1, 0)) act_mbf_delinq,
             sum(iif(kh_is_our = 1 and kh_is_active = 1 and f_strtonum(kh_repaystatus) = 0, kh_unpaidamount, 0)) sum_mbf_kh_non_delinq_balance,
             sum(iif(kh_is_our = 1 and kh_is_active = 1 and f_strtonum(kh_repaystatus) > 0, kh_unpaidamount, 0)) sum_mbf_kh_delinq_balance,
             max(iif(kh_is_our = 1 and kh_is_active = 1, f_strtonum(kh_deliqdayscount), 0)) kib_mbf_act_max_delinq, --
             --<< MBulak
             -->> ne MBulak
             sum(iif(kh_is_our <> 1 and kh_deliqdayscount = 0, 1, 0)) kib_nombf_non_delinq, --
             sum(iif(kh_is_our <> 1 and kh_deliqdayscount > 0, 1, 0)) kib_nombf_delinq, --
             sum(iif(kh_is_our <> 1, kh_ownamount, 0)) sum_nombf_kh_ownamount, --
             sum(iif(kh_is_our <> 1, kh_amount, 0)) sum_nombf_kh_amount,
             max(iif(kh_is_our <> 1, kh_deliqdayscount, 0)) max_nombf_kh_deliqdayscount,
             sum(iif(kh_is_our <> 1 and kh_is_active = 1 and kh_repaystatus containing 'Акт. пр-ка: 0 дн', 1, 0)) act_nombf_non_delinq,
             sum(iif(kh_is_our <> 1 and kh_is_active = 1 and f_strtonum(kh_repaystatus) > 0, 1, 0)) act_nombf_delinq,
             sum(iif(kh_is_our <> 1 and kh_is_active = 1 and f_strtonum(kh_repaystatus) = 0, kh_unpaidamount, 0)) sum_nombf_kh_non_delinq_balance,
             sum(iif(kh_is_our <> 1 and kh_is_active = 1 and f_strtonum(kh_repaystatus) > 0, kh_unpaidamount, 0)) sum_nombf_kh_delinq_balance,
             max(iif(kh_is_our <> 1 and kh_is_active = 1, f_strtonum(kh_deliqdayscount), 0)) kib_nombf_act_max_delinq, --
             --<< ne MBulak
             min(kh_dateaward) min_kh_dateaward, list(distinct score_rate, ', '), list(distinct risk_rate, ', ')
      from aarep_kib_miniapp_cli_report(:region_id, :ma_id, :c_id, 1)
      into kf_client_name, kbm_checked_date, kh_create_date, monthly_repay, --
           kib_mbf_non_delinq, kib_mbf_delinq, sum_mbf_kh_ownamount, sum_mbf_kh_amount, max_mbf_kh_deliqdayscount,
           act_mbf_non_delinq, act_mbf_delinq, sum_mbf_kh_non_delinq_balance, sum_mbf_kh_delinq_balance,
           kib_mbf_act_max_delinq, --
           kib_nombf_non_delinq, kib_nombf_delinq, sum_nombf_kh_ownamount, sum_nombf_kh_amount,
           max_nombf_kh_deliqdayscount, act_nombf_non_delinq, act_nombf_delinq, sum_nombf_kh_non_delinq_balance,
           sum_nombf_kh_delinq_balance, kib_nombf_act_max_delinq, --
           min_kh_dateaward, score_rate, risk_rate;

      if (coalesce(kf_client_name, '') = '') then
        kf_client_name = c_name_fio;

      kbm_checked_date = coalesce(kbm_checked_date, kh_create_date);

      c_score = null;
      c_score = f_point(:c_id, :ma_date_app, 0);

      suspend;
    end
  end
end
