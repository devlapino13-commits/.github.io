CREATE PROCEDURE HP_KONTRACT_OPER_FOR_DELETE(
    KONT_ID DECIMAL(18,0))
RETURNS (
    OL_DATE DATE,
    OL_DOC_NO INTEGER,
    OL_MEMO VARCHAR(1024),
    OL_DATETIME TIMESTAMP,
    UN_NAME VARCHAR(30),
    DISB_OS DECIMAL(18,4),
    PAYED_OS DECIMAL(18,4),
    PAYED_PERC DECIMAL(18,4),
    PAYED_FINE DECIMAL(18,4),
    PAYED_KOMMIS DECIMAL(18,4),
    PAYED_TAX DECIMAL(18,4),
    SUM_OF_ALL DECIMAL(18,4),
    OL_ID DECIMAL(18,0),
    OL_PARENT_ID DECIMAL(18,0),
    OL_BALANCE_KIND VARCHAR(50),
    WRITEOFF_OS DECIMAL(18,4),
    WRITEOFF_PERC DECIMAL(18,4),
    WRITEOFF_FINE DECIMAL(18,4),
    ADVANCED_OS DECIMAL(18,4),
    RESTRUCT_PERC DECIMAL(18,4),
    IS_LOCKED INTEGER,
    KONT_COUNT INTEGER,
    OL_DOC_KIND VARCHAR(20),
    OL_PAY_TYPE VARCHAR(50),
    OL_FROM_REG_ID INTEGER,
    OL_REGION_NAME VARCHAR(100),
    OL_DOC_REGION_ID INTEGER,
    OL_DOC_REGION_NAME VARCHAR(100),
    WRITEOFF_ALL DECIMAL(18,4),
    KONTRACT_NO VARCHAR(20),
    OX_OPER_PLUS DECIMAL(18,4),
    OX_OPER_MINUS DECIMAL(18,4),
    OX_OPER_WRITEOFF DECIMAL(18,4),
    OL_OPER_TYPE D_INTEGER,
    OL_OPER_SUBTYPE D_INTEGER)
AS
declare variable t_os decimal(18,4) = 0;
declare variable t_perc decimal(18,4) = 0;
declare variable t_fine decimal(18,4) = 0;
declare variable t_kommis decimal(18,4) = 0;
declare variable un_id decimal(18,0);
declare variable k_id decimal(18,0);
declare variable oek_rd_id integer;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  kontract_no = '';
  select k_kontract_no
  from kontract
  where k_id = :kont_id
  into :kontract_no;

  for select distinct ok_ol_date, ok_from_ol_id
      from operkred, operlist
      where ok_from_k_id = :kont_id and
            ok_ol_is_active = 1 and
            ok_from_ol_id = ol_id and
--            ok_type <> 3 and
            ok_type <> 1 and
            ok_type <> 7 and
            not(ok_type = 4 and
            ok_sh_id = 0 and
            ol_parent_id is not null)
      union
      select distinct ol_date, oko_from_ol_id
      from operkroff, operlist
      where oko_from_k_id = :kont_id and
            operlist.is_delete = 0 and
            oko_type = 4 and
            oko_from_ol_id = ol_id
      union
      select distinct ox_ol_date, ox_from_ol_id
      from operextra
      where ox_from_rd_id = :oek_rd_id and
            ox_object_kind = 9 and
            ox_object_id = :kont_id and
            ox_ol_is_active = 1
      order by 1, 2
      into :ol_date, :ol_id
  do
  begin
    disb_os = 0;
    payed_os = 0;
    payed_perc = 0;
    payed_fine = 0;
    payed_kommis = 0;
    payed_tax = 0;
    writeoff_os = 0;
    writeoff_perc = 0;
    writeoff_fine = 0;
    advanced_os = 0;
    restruct_perc = 0;
    ox_oper_plus = 0;
    ox_oper_minus = 0;
    ox_oper_writeoff = 0;
    ol_balance_kind = '';
    select sum(iif(ok_type = 8, ok_os, 0.0)), sum(iif(ok_type = 4, -ok_os, 0.0)), sum(iif(ok_type = 4, -ok_perc, 0.0)),
           sum(iif(ok_type = 4, -ok_fine, 0.0)), sum(iif(ok_type = 4, -ok_kommis - ok_kommis_adv, 0.0)),
           sum(iif(ok_type = 4, -ok_tax, 0.0)), sum(iif(ok_type in (5, 6), -ok_os, 0.0)),
           sum(iif(ok_type in (5, 6), -ok_perc, 0.0)), sum(iif(ok_type in (5, 6), -ok_fine, 0.0)),
           sum(iif(ok_type = 9, -ok_os, 0.0)), sum(iif(ok_type in (2, 3), ok_sh_perc, 0.0))
    from operkred
    where ok_from_ol_id = :ol_id and
          ok_from_k_id = :kont_id
    into :disb_os, :payed_os, :payed_perc, :payed_fine, :payed_kommis, :payed_tax, :writeoff_os, :writeoff_perc,
         :writeoff_fine, :advanced_os, :restruct_perc;

    disb_os = coalesce(disb_os, 0);
    payed_os = coalesce(payed_os, 0);
    payed_perc = coalesce(payed_perc, 0);
    payed_fine = coalesce(payed_fine, 0);
    payed_kommis = coalesce(payed_kommis, 0);
    payed_tax = coalesce(payed_tax, 0);
    writeoff_os = coalesce(writeoff_os, 0);
    writeoff_perc = coalesce(writeoff_perc, 0);
    writeoff_fine = coalesce(writeoff_fine, 0);
    advanced_os = coalesce(advanced_os, 0);
    restruct_perc = coalesce(restruct_perc, 0);

    if (disb_os <> 0) then
    begin
      select sum(-ok_kommis)
      from operkred, operlist
      where ok_type = 4 and
            ok_from_k_id = :kont_id and
            ok_ol_date = :ol_date and
            ok_ol_is_active = 1 and
            ok_from_ol_id = ol_id and
            ol_parent_id is not null
      into :t_kommis;
      if (t_kommis is null) then
        t_kommis = 0;
      payed_kommis = payed_kommis + t_kommis;
    end
    ol_balance_kind = 'Баланс.';
    t_os = 0;
    t_perc = 0;
    t_fine = 0;
    select sum(-oko_os), sum(-oko_perc), sum(-oko_other)
    from operkroff
    where oko_from_ol_id = :ol_id and
          oko_from_k_id = :kont_id and
          oko_type = 4
    into :t_os, :t_perc, :t_fine;
    if (t_os is null) then
      t_os = 0;
    if (t_perc is null) then
      t_perc = 0;
    if (t_fine is null) then
      t_fine = 0;
    if ((t_os <> 0) or (t_perc <> 0) or (t_fine <> 0)) then
      ol_balance_kind = 'Внебаланс.';

    payed_os = payed_os + t_os;
    payed_perc = payed_perc + t_perc;
    payed_fine = payed_fine + t_fine;

    select sum(ox_amount_plus), sum(ox_amount_minus), sum(ox_amount_writeoff)
    from operextra
    where ox_from_rd_id = :oek_rd_id and
          ox_from_ol_id = :ol_id and
          ox_object_kind = 9 and
          ox_object_id = :kont_id
    into :ox_oper_plus, :ox_oper_minus, :ox_oper_writeoff;

    if (ox_oper_plus is null) then
      ox_oper_plus = 0;
    if (ox_oper_minus is null) then
      ox_oper_minus = 0;
    if (ox_oper_writeoff is null) then
      ox_oper_writeoff = 0;

    un_id = null;
    un_name = '';
    select
      ol_doc_no, ol_memo, ol_datetime, ol_user_id, ol_parent_id, ol_from_reg_id, reg_code || ' ' || reg_name,
      ol_oper_type, ol_oper_subtype
    from operlist, region
    where ol_id = :ol_id and
          ol_from_reg_id = reg_unicode
    into
      :ol_doc_no, :ol_memo, :ol_datetime, :un_id, :ol_parent_id, ol_from_reg_id, ol_region_name,
      :ol_oper_type, :ol_oper_subtype;

    ol_doc_region_id = ol_from_reg_id;
    ol_doc_region_name = ol_region_name;

    if (un_id is not null) then
    begin
      select un_name
      from username
      where un_id = :un_id
      into :un_name;
    end

    sum_of_all = disb_os + payed_os + payed_perc + payed_fine + payed_tax + advanced_os + ox_oper_plus + ox_oper_minus + payed_kommis;
    writeoff_all = writeoff_os + writeoff_perc + writeoff_fine + ox_oper_writeoff;

    is_locked = 0;
    if (ol_parent_id is not null) then
    begin
      is_locked = 1;
      select ol_from_reg_id, reg_code || ' ' || reg_name
      from operlist, region
      where ol_id = :ol_parent_id and
            ol_from_reg_id = reg_unicode
      into ol_doc_region_id, ol_doc_region_name;
    end

    kont_count = 0;
    for select distinct ok_from_k_id
        from operkred
        where ok_from_ol_id = :ol_id
        union
        select distinct oko_from_k_id
        from operkroff
        where oko_from_ol_id = :ol_id
        union
        select distinct ox_object_id
        from operextra
        where ox_from_rd_id = :oek_rd_id and
              ox_from_ol_id = :ol_id and
              ox_object_kind = 9
        into :k_id
    do
    begin
      kont_count = kont_count + 1;
    end
    if (kont_count > 1) then
      is_locked = 2;

    ol_doc_kind = 'OTHER';
    ol_pay_type = 'Безналичная';
    if (exists(select *
               from operbook, acctypelist, acc_list
               where ob_account_dt = al_id and
                     al_from_bg_id = atl_acc_mask and
                     ob_from_ol_id = iif(:ol_parent_id is null, :ol_id, :ol_parent_id) and
                     atl_from_at_id = 11 and
                     ob_summa <> 0)) then
    begin
      ol_doc_kind = 'CASH_REPAY';
      ol_pay_type = 'Наличная';
    end
    else
    if (exists(select *
               from operbook, acctypelist, acc_list
               where ob_account_kt = al_id and
                     al_from_bg_id = atl_acc_mask and
                     ob_from_ol_id = iif(:ol_parent_id is null, :ol_id, :ol_parent_id) and
                     atl_from_at_id = 11 and
                     ob_summa <> 0)) then
    begin
      ol_doc_kind = 'CASH_DISB';
      ol_pay_type = 'Наличная';
    end
    if (ol_parent_id is not null) then
    begin
      ol_pay_type = ol_pay_type || ', другой офис';
    end
    suspend;
  end
end
