CREATE PROCEDURE PRINT_NEW_SCHEDULE(
    PRINT_K_ID DECIMAL(18,0))
RETURNS (
    SH_FROM_K_ID DECIMAL(18,0),
    SH_NUMBER INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_KOMMIS DECIMAL(18,4),
    SH_PERC_TAX DECIMAL(18,4),
    SH_TOTAL DECIMAL(18,4),
    SH_TOTAL_OS_PERC DECIMAL(18,4),
    SH_BALANCE_BEFORE DECIMAL(18,4),
    SH_BALANCE_AFTER DECIMAL(18,4),
    SH_DAYS INTEGER,
    SH_REMAIN_OS DECIMAL(18,4),
    SH_DATE_FIRST D_DATE)
AS
declare variable SH_DD date;
declare variable CM_DATABASE_ID integer;
begin

  :cm_database_id = f_get_db_id();
  sh_kommis = 0;
  sh_perc_tax = 0;
  sh_from_k_id = print_k_id;
  sh_days = null;
  sh_date_first = null;

  select sum(ok_os), max(ok_ol_date)
    from operkred
    where ok_ol_is_active = 1
      and ok_from_k_id = :print_k_id
      and ok_from_c_id = 0
    into :sh_balance_before, :sh_dd;

  sh_balance_after = :sh_balance_before;

  for select ok_sh_id, ok_sh_date,
             sum(ok_os + ok_sh_os),
             sum(ok_perc + ok_sh_perc)
      from operkred
      where ok_from_k_id = :print_k_id
        and ok_ol_is_active = 1
        and ok_type < 7
        and ok_from_c_id = 0
      group by ok_sh_id, ok_sh_date
      having sum(ok_os + ok_sh_os) <> 0
         or sum(ok_perc + ok_sh_perc) <> 0
      order by ok_sh_date
      into :sh_number, :sh_date, :sh_os, :sh_perc
  do
  begin
    if (:sh_date_first is null) then
       sh_date_first = sh_date;
    sh_balance_after = :sh_balance_after - :sh_os;

    if (sh_days is not null) then
        sh_days = cast((:sh_date - :sh_dd) as integer);
    else
        sh_days = null;

    -- НСП
    sh_perc_tax = iif(:cm_database_id in (39801, 41701),
                      round(sh_perc / 100 * 2),
                      0);

    -- Общая без НСП
    sh_total_os_perc = sh_os + sh_perc;

    -- Полная сумма
    sh_total = sh_os + sh_perc + sh_perc_tax;

    -- остаток ОС (основного долга)
    sh_remain_os = coalesce(:sh_balance_before, 0) - coalesce(:sh_os, 0);

    if (sh_total <> 0) then
        suspend;

    sh_remain_os = 0; -- для итоговой строки остаток основного долга = 0
    sh_days = 0;
    sh_dd = sh_date;
    sh_balance_before = :sh_balance_before - :sh_os;
  end
end
