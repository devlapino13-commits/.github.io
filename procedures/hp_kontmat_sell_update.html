CREATE PROCEDURE HP_KONTMAT_SELL_UPDATE(
    K_ID DECIMAL(18,0))
AS
declare variable kml_id integer;
declare variable loan_amount decimal(18,4);
declare variable total_amount decimal(18,4) = 0;
declare variable total_perc decimal(18,4) = 0;
declare variable total_tax decimal(18,4) = 0;
declare variable koef_perc double precision;
declare variable koef_tax double precision;
declare variable amount_profit decimal(18,4) = 0;
declare variable amount_one_cost decimal(18,4) = 0;
declare variable kml_cost decimal(18,4) = 0;
declare variable kml_count decimal(18,4) = 0;
declare variable amount_cost_tax decimal(18,4) = 0;
begin
-- Процедура пересчитывает наценку

  select k_amount_approved
  from kontract
  where k_id = :k_id
  into :loan_amount;
  if (coalesce(loan_amount, 0) = 0) then
    exit;

  select sum(kml_total)
  from kontmat
  where kml_object_kind = 9 and
        kml_object_id = :k_id
  into :total_amount;
  if (coalesce(total_amount, 0) = 0) then
    exit;

  select sum(sh_perc), sum(sh_perc_tax)
  from fm_schedule_list(9, :k_id)
  where sh_id > 0
  into :total_perc, total_tax;
  if (coalesce(total_perc, 0) = 0) then
    exit;

  koef_perc = total_perc / total_amount;
  koef_tax = total_tax / total_amount;

  for select kml_id, kml_count, kml_cost
      from kontmat
      where kml_object_kind = 9 and
            kml_object_id = :k_id
      into :kml_id, :kml_count, :kml_cost
  do
  begin
    if (kml_count = 0) then
      kml_count = 1;
    amount_profit = round(kml_cost * koef_perc, 4);
    amount_one_cost = kml_cost + amount_profit;
    amount_cost_tax = round(kml_cost * koef_tax, 4);
    total_perc = total_perc - amount_profit * kml_count;
    total_tax = total_tax - amount_cost_tax * kml_count;

    update kontmat
    set kml_sell_profit = :amount_profit,
        kml_sell_nsp = :amount_cost_tax,
        kml_sell_cost = :amount_one_cost,
        kml_sell_total = (:amount_one_cost + :amount_cost_tax) * :kml_count
    where kml_id = :kml_id and
          (kml_sell_profit <> :amount_profit or
          kml_sell_nsp <> :amount_cost_tax or
          kml_sell_cost <> :amount_one_cost or
          kml_count <> :kml_count);

  end
  if ((total_perc <> 0.0) or (total_tax <> 0.0)) then
  begin
    update kontmat
    set kml_sell_profit = kml_sell_profit + round((:total_perc / :kml_count), 2),
        kml_sell_nsp = :amount_cost_tax + round((:total_tax / :kml_count), 2),
        kml_sell_cost = :amount_one_cost + round((:total_perc / :kml_count), 2),
        kml_sell_total = round(kml_sell_total + :total_perc)
    where kml_id = :kml_id;
  end
end
