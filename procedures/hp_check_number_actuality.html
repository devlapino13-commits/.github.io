CREATE PROCEDURE HP_CHECK_NUMBER_ACTUALITY(
    C_ID D_ID,
    CODE D_STR_10)
RETURNS (
    RES_TEXT D_STR_8100,
    SHOW_RES D_STR_10)
AS
declare variable CNC_CODE D_INTEGER;
declare variable INPUT_CODE D_INTEGER;
declare variable CNC_ID D_ID;
declare variable CNC_CREATE_DATETIME D_DATETIME;
declare variable DIFF_TIME D_FLOAT;
declare variable COST_DIFF_MINUTE D_FLOAT;
begin
  -- RFCRU-3522 процедура проверяет код
  res_text = '';
  show_res = 'dialog';
  cost_diff_minute = 60; -- константа для проверки разницы во времени (5 = 5 минут)

  if(code not similar to '[[:DIGIT:]]+') then
    res_text = 'Код должен быть целым положительным числом';

  if(res_text = '') then
  begin
    input_code = cast(code as d_integer);

    select cnc.cnc_code, cnc.cnc_id, cnc.cnc_create_datetime
    from check_number_code cnc
    where cnc.cnc_from_c_id = :c_id
      and cnc.cnc_is_active = 1
    order by cnc.cnc_create_datetime
    into cnc_code, cnc_id, cnc_create_datetime;

    cnc_id = coalesce(cnc_id, 0);

    if(cnc_id = 0) then
      res_text = 'Ошибка: Код не был отправлен';

    diff_time = (current_timestamp - cnc_create_datetime) * 24 * 60; -- разница во времени в минутах

    if(cost_diff_minute < diff_time) then
      res_text = 'Код устарел попробуйте отправит еще раз';
    else if (cnc_code = input_code) then
    begin
      res_text = '';

      update check_number_code cnc   -- код верный
         set cnc.cnc_is_sucsess = 1
      where cnc.cnc_id = :cnc_id;
    end
    else if(cnc_code <> input_code) then
      res_text = 'Код неверный';
  end

  suspend;
end
