CREATE PROCEDURE AAREP_SMS_SEND(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    BRANCH_NAME STR_100,
    K_KONTRACT_NO STR_20,
    CLI_FIO STR_100,
    K_DISB_DATE DATE,
    LSS_KIND_STR STR_100,
    SMS_TEXT D_STR_2048,
    LSS_SEND_STATUS_STR STR_100,
    DATE_SEND DATE)
AS
declare variable reg_id integer;
declare variable lss_kind smallint;
declare variable reg_id_prior integer;
declare variable lss_send_status smallint;
begin
  reg_id_prior = 0;
  for
  select distinct
         lss_from_reg_id,
         lss_kontract_no,
         lss_fio,
         lss_kind,
         lss_sms_text1||lss_sms_text2,
         msg_send_status, --lss_send_status,
         cast(msg_datetime as date)
  from listsmssend, fsrobot_msgbox
  where lss_cell_no <> ''
  and lss_cell_no = msg_mail_to
  and (msg_body = lss_sms_text1 or msg_body = lss_sms_text2)
  and msg_body is not null
  and msg_body <> ''
  and cast(msg_datetime as date) between :rep_date_begin and :rep_date_end
  into reg_id,
       k_kontract_no,
       cli_fio,
       lss_kind,
       sms_text,
       lss_send_status,
       date_send
  do
  begin

    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select r1.reg_code||' '||r1.reg_name,
             r2.reg_code||' '||r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :reg_id
      and r1.parent = r2.reg_unicode
      into reg_name, branch_name;
    end

    select k_disb_date from kontract
    where k_kontract_no = :k_kontract_no
    into k_disb_date;

    lss_send_status_str = case lss_send_status
                     when 0 then 'в очереди'
                     when 1 then 'доставлено'
                     else 'не доставлено'
                   end;
    lss_kind_str = '';
    if (lss_kind  = 1) then lss_kind_str = 'sms рассылка Указ Президента';
    else
    if (lss_kind  = 2) then lss_kind_str = 'sms рассылка СПИСАНИЕ %% и ПЕНИ';
    else
    if (lss_kind  = 3) then lss_kind_str = 'sms рассылка Указ Президента 51 клиент';
    else
    if (lss_kind  = 4) then lss_kind_str = 'sms рассылка СПИСАНИЕ %% и ПЕНИ 51 клиент';
    else
    if (lss_kind  = 5) then lss_kind_str = 'sms рассылка СПИСАНИЕ %% и ПЕНИ 8 клиентов';

    suspend;
  end
end
