CREATE PROCEDURE HP_DO_RESCHEDULE_CAN_DELINQ
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable db_id d_integer;
declare variable sh_date d_date;
begin
  -- Проверяет возможность реструктуризации просроченых займов

  res_is_error = 0;
  res_text = '';

  db_id = f_get_db_id();
  sh_date = null;

  select sh_date
  from temp_sh
  where sh_repay_mode = 1 and
        (sh_zad_os <> 0 or sh_zad_perc <> 0)
  order by sh_date
  rows 1
  into sh_date;

  if (sh_date is not null) then
  begin
    if (db_id = 94401) then
    begin
      if (sh_date < '20.03.2020') then
        res_text = 'Просроченные займы до 20.03.2020 не подлежат реструктуризации';
    end
    else
    if (db_id = 81001) then
    begin
      if (sh_date < '12.05.2020') then
        res_text = 'Просроченные займы до 12.05.2020 не подлежат реструктуризации';
    end
    else
    if (db_id in (39801, 41701)) then
    begin
      if (sh_date < '01.03.2020') then
        res_text = 'Просроченные займы до 01.03.2020 не подлежат реструктуризации';
    end
    else
      res_text = 'Просроченные займы не подлежат реструктуризации';

    if (res_text <> '') then
      res_is_error = 1;
  end

  suspend;
end
