CREATE PROCEDURE AAREP_KS_STAFF(
    REP_REGION INTEGER,
    REP_DATE_END DATE,
    REP_CURRENCY VARCHAR(3))
RETURNS (
    KF_OFFICER DECIMAL(18,0),
    KF_OFFICER_NAME VARCHAR(100),
    REG_UNICODE INTEGER,
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    K_KOLICH INTEGER,
    JL_NAME VARCHAR(100),
    PS_IS_ON_STAFF SMALLINT,
    PS_IS_ON_STAFF_STR VARCHAR(20),
    KE_TEACHER_ID DECIMAL(18,0),
    KE_TEACHER_NAME VARCHAR(100),
    KE_TEACHER_JL_NAME VARCHAR(100),
    KE_TEACHER_IS_ON_STAFF SMALLINT,
    KE_TEACHER_IS_ON_STAFF_STR VARCHAR(20),
    REP_ACTIVE_BALANCE DECIMAL(18,4),
    PS_CURR_STATUS VARCHAR(100))
AS
declare variable rep_active_balance_1 decimal(18,4);
declare variable parent_id integer;
declare variable reg_prior integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  reg_prior = 0;

  for select kal_new_kf_officer, c1.c_name_fio, r1.reg_unicode, r1.reg_code || ' ' || r1.reg_name, r1.parent, count(*),
             jl1.jl_name, hp1.ps_is_on_staff, ke_teacher_id, c2.c_name_fio, jl2.jl_name, hp2.ps_is_on_staff, parent_name

      from knaktlist
      left join hr_pstatus hp1 on (kal_new_kf_officer = hp1.ps_from_c_id and
            :rep_date_end between hp1.ps_date_begin and hp1.ps_date_end and
            hp1.ps_is_working = 1 and
            hp1.ps_status = 1)
      left join joblist jl1 on (hp1.ps_from_jl_id = jl1.jl_id)
      left join krexpert on (ke_from_p_id = kal_new_kf_officer and
            ke_from_reg_id = kal_new_reg_id and
            :rep_date_end between ke_date and ke_date_end)
      left join clients c2 on (ke_teacher_id = c2.c_id)
      left join hr_pstatus hp2 on (ke_teacher_id = hp2.ps_from_c_id and
            :rep_date_end between hp2.ps_date_begin and hp2.ps_date_end and
            hp2.ps_is_working = 1 and
            hp2.ps_status = 1)
      left join joblist jl2 on (hp2.ps_from_jl_id = jl2.jl_id), kontstate, region r1, region_list(:rep_region, 1, 3, 1), kontract, clients c1
      where (id_list = r1.reg_unicode) and
            kal_new_reg_id = r1.reg_unicode and
            r1.reg_status = 3 and
            :rep_date_end between kal_date and kal_date_end and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_new_kf_officer = c1.c_id and
            (hp1.ps_is_on_staff = 0 or
            (hp1.ps_is_on_staff is null) or
            (ke_teacher_id is not null and
            hp1.ps_is_on_staff = 1))
      group by kal_new_kf_officer, r1.reg_unicode, jl1.jl_name, hp1.ps_is_on_staff, c1.c_name_fio, r1.reg_code, r1.reg_name, r1.parent, ke_teacher_id, c2.c_name_fio, jl2.jl_name, hp2.ps_is_on_staff, parent_name
      order by parent_name, r1.reg_code, c1.c_name_fio

      into :kf_officer, :kf_officer_name, :reg_unicode, :reg_name, :parent_id, :k_kolich, :jl_name, :ps_is_on_staff,
           :ke_teacher_id, :ke_teacher_name, :ke_teacher_jl_name, :ke_teacher_is_on_staff, :parent_name

  do
  begin
    if (parent_id <> reg_prior) then
    begin
      select reg_name
      from region
      where reg_unicode = :parent_id
      into :parent_name;
      reg_prior = parent_id;
    end

    rep_active_balance = 0;
    rep_active_balance_1 = 0;
    ps_curr_status = '';
    for select sum(
               case
                 when :rep_currency = '000' then (case
                                                    when cr_rate < k_index_curr_rate then oks_balance * k_index_curr_rate /*Вычисляет активный портфель текущего сотрудника*/
                                                    else oks_balance * cr_rate
                                                  end)
                 when :rep_currency = '---' then oks_balance
                 when :rep_currency = k_index_curr_code then oks_balance
               end)
        from oksummary, kontract, opercurr
        where oks_from_k_id = k_id and
              k_index_curr_code = cr_from_cu_id and
              oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end and
              :rep_date_end between cr_date and cr_date_end and
              ((:rep_currency = '---') or
              (:rep_currency = '000') or
              (:rep_currency = k_index_curr_code)) and
              oks_region_id = :reg_unicode and
              oks_kc_id = :kf_officer and
              oks_balance > 0 and
              oks_close_state = 0
        into :rep_active_balance_1
    do
    begin
      rep_active_balance = rep_active_balance_1;
    end

    select sawt_name
    from hr_voctravel, hr_orders, sa_worktype /*Вытаскивает текущий статус по кадровым приказам КСника*/
    where hrv_from_ho_id = ho_id and
          ho_order_type = sawt_id and
          hrv_from_c_id = :kf_officer and
          hrv_status = 1 and
          :rep_date_end between hrv_date_begin and hrv_date_end_fact and
          hrv_status = 1
    order by hrv_date_begin desc
    rows 1
    into :ps_curr_status;

    ke_teacher_is_on_staff_str = null;
    if (ps_is_on_staff is not null) then
    begin
      if (ps_is_on_staff = 1) then
        ps_is_on_staff_str = 'Штатный';
      else
        ps_is_on_staff_str = 'Внештатный';
    end
    else
      ps_is_on_staff_str = 'Приказ не оформлен';

    if (ke_teacher_is_on_staff = 1) then
      ke_teacher_is_on_staff_str = 'Штатный';
    else
    if (ke_teacher_is_on_staff is not null) then
      ke_teacher_is_on_staff_str = 'Внештатный';

    suspend;
  end
end
