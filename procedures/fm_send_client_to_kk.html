CREATE PROCEDURE FM_SEND_CLIENT_TO_KK(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0),
    CLIENT_MEMO VARCHAR(255))
RETURNS (
    RES_TEXT D_STR_8100,
    SHOW_RES D_STR_10)
AS
declare variable db_id integer;
declare variable k_id decimal(18,0);
declare variable k_status integer;
declare variable acl_status integer;
declare variable rd_id integer;
declare variable total_status integer;
declare variable acl_id decimal(18,0);
declare variable k_disb_date date;
declare variable pr_id id_type;
declare variable result integer;
declare variable pr_code str_100;
declare variable region_id integer;
declare variable kk_preference smallint;
declare variable k_is_online smallint;
declare variable tmp_is_ok d_integer;
declare variable bsc_scoring_level d_integer;
declare variable c_id d_id;
declare variable mac_id d_id;
declare variable maci_id d_id;
declare variable maci_is_documented d_smallint;
declare variable maci_is_blocked d_smallint;
declare variable obm_memo d_str_255;
declare variable survey_json d_str_8100;
begin
  res_text = '';
  show_res = 'dialog';

  db_id = f_get_db_id();

  if (obj_kind = 'miniapp') then
  begin
    select k_id
    from kontract
    where k_from_ma_app_id = :obj_id
    order by k_disb_date desc
    rows 1
    into :k_id;
    if (k_id is not null) then
      obj_id = :k_id;

    select mac.mac_from_c_id
    from miniappclient mac
    where mac.mac_from_ma_id = :obj_id and
          mac.mac_amount > 0
    into :c_id;
  end
  else
  if (obj_kind = 'clients') then
  begin
    select distinct k_id
    from kontract, knclient
    where kc_from_k_id = k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = :obj_id and
          k_disb_date >= (current_date - 1)
    order by k_id desc -- Берем последний созданный
    rows 1
    into :k_id;

    c_id = :obj_id;
    if (k_id is not null) then
      obj_id = :k_id;

  end
  else if (obj_kind = 'kontract') then
  begin
    select kc.kc_from_c_id
    from knclient kc
    where kc.kc_from_k_id = :obj_id and
          kc.kc_amount_approved > 0
    into :c_id;
  end 


  for select k_id, k_status, k_disb_date, k_from_pr_id, kont.k_last_reg_id, kont.k_is_online
      from kontract kont
      where k_id = :obj_id
      into :k_id, :k_status, :k_disb_date, :pr_id, :region_id, :k_is_online
  do
  begin
    if (k_status = -2) then
    begin
      res_text = 'Заявка еще не переведена в договор';
      suspend;
      exit;
    end
    if (k_status = -1) then
    begin
      res_text = 'Заявка/Договор аннулирован. За причиной обратитесь к МНМ';
      suspend;
      exit;
    end
    if (k_status = 1) then
    begin
      res_text = 'Договор уже утвержден, нет необходимости повторной отправки';
      suspend;
      exit;
    end

    if (db_id = 81001 and (select is_oivo from hp_check_oivo_assignment(:c_id, :un_id)) = 1) then
    begin
      if(f_meet_passed(:c_id) <> 1) then
      begin
        res_text = 'Требуется верификация клиента по Qr-коду';
        suspend;
        exit;
      end
    end 

    if (k_status = 0) then
    begin
      acl_status = null;
      select acl_status
      from hp_actionlist_kontract_get(:k_id)
      into :acl_status;

      if (acl_status = 2) then
      begin
        res_text = iif(db_id in (39801, 41701), 'Сертификатор', 'Кредитный комитет') || ' уже одобрил договор, нет необходимости в повторной отправке';
        suspend;
        exit;
      end
      if (acl_status between 0 and 1) then
      begin
        res_text = 'Договор уже находится на ' || iif(db_id in (39801, 41701), 'сертификации', 'рассмотрении кредитного комитета');
        suspend;
        exit;
      end

      -- Проверка условий
      select result_is_ok, result_txt
      from fm_check_loan_konditions('Send_KK', :k_id, 0, :un_id)
      into tmp_is_ok, res_text;

      if (tmp_is_ok = 0) then
      begin
        show_res = 'window';
        suspend;
        exit;
      end

      -- Отправка на КК
      res_text = '';
      select res_message
      from hp_actionlist_object_insert('kontract', :k_id, 'actiongroup.check_loan_data', :un_id)
      into :res_text;

      if (coalesce(res_text, '') = '') then
      begin
        if (client_memo <> '') then
          execute procedure hp_objectprops_memo_set('Kontract', :k_id, 'Refdata', 'ObjectProps.Client_memo',
              :client_memo, :un_id);

        res_text = 'Заявка успешно отправлена на ' || iif(db_id in (39801, 41701), 'сертификацию', 'кредитный комитет');

        execute procedure hp_miniapp_for_penalty_update('Kontract', :k_id);

        if (db_id = 81001) then
        begin
          result = 0;

          select pr.pr_code
          from product pr
          where pr.pr_id = :pr_id
          into :pr_code;

          select coalesce(reg.reg_kk_preference, 0) -- предоодобрение КК
          from region reg
          where reg.reg_unicode = :region_id
          into :kk_preference;

          if (current_date < '20.07.2018') then
            kk_preference = 1;

          if ((kk_preference = 1) and
              (k_is_online = 0) and
              (pr_code not containing ('.zalog_car.')) and
              (pr_code not containing ('.education.')) and
              (pr_code not containing ('.reliable_emp.')) and
              (pr_code not containing ('.float_percent.'))) then
          begin
            if ((pr_code containing ('.PATENT.')) or (pr_code containing ('.PATENT PLUS.')) or (pr_code containing ('.SHANS.'))) then
              result = 1;

            if (result = 0) then
            begin
              -- если вся группа соcтоит из повторников
              if (not exists(select *
                             from knclient kncl
                             where kncl.kc_from_k_id = :k_id and
                                   kncl.kc_amount_approved > 0 and
                                   kncl.kc_circle <= 1)) then
              begin
                result = 1;
              end
            end
          end

          if ((pr_code containing ('.zalog_car.')) or
              (pr_code containing ('.education.')) or
              (pr_code containing ('.reliable_emp.')) and
              (pr_code containing ('.float_percent.')) or
              (k_is_online = 1)) then
            result = 0;

          if ((result = 0) and
              (kk_preference = 1) and
              (k_is_online = 0) and
              (pr_code not containing ('.zalog_car.')) and
              (pr_code not containing ('.education.')) and
              (pr_code not containing ('.reliable_emp.')) and
              (pr_code not containing ('.float_percent.'))) then
          begin
            total_status = null;

            select k_curr_status
            from hp_get_kontract_status(:k_id, iif(:k_disb_date > current_date, current_date, :k_disb_date))
            into :total_status;

            total_status = coalesce(total_status, 0);

            if (total_status in (4, 5)) then -- если 4 или 5 звезд автоматически утверждаю заявку КК
              result = 1;
          end

          -- RFCRU-3640 если результат скоринга положительный для продуктов "Универсальный" и "Ойва" автоматически утверждаю заявку КК
          if (result = 0 and
              (pr_code containing ('.universal.') or
              pr_code containing ('.oivo_pack.'))) then
          begin
            select min(bsc_scoring_level)
            from knclient, bki_scoring
            where bsc_from_kc_id = kc_id
              and kc_from_k_id = :k_id
            into bsc_scoring_level;
      
            if (bsc_scoring_level = 2 and f_send_to_kk_sofrcheck(:k_id) = 0) then
              result = 1;
          end 

          if (result = 1) then -- если 4 или 5 звезд автоматически утверждаю заявку КК
          begin
            select rd_id
            from hp_get_check_loan_data(:k_id)
            into:rd_id;

            for select ao.aco_id
                from actionorder ao
                where ao.aco_from_rd_id = :rd_id and
                      ao.aco_object_kind = 9 and
                      ao.aco_object_id = :k_id and
                      aco_status = 0
                into :acl_id
            do
            begin
              update actionlist
              set acl_status = 2,
                  acl_respons_c_id = 1000000000001 -- Sanjar Islamov
              where acl_parent_id = :acl_id and
                    acl_dont_show_in_order = 1;

              update actionlist
              set acl_status = 2,
                  acl_respons_c_id = 1000000000001 -- Sanjar Islamov
              where acl_id = :acl_id;

              res_text = 'Заем одобрен';
            end
          end

          if (f_get_const('documented_income_active') = 1) then
          begin
            for select maci.maci_from_mac_id
            from miniappclient mac
            join kontract k on k.k_from_ma_app_id = mac.mac_from_ma_id
            join miniappclient_income maci on maci.maci_from_mac_id = mac.mac_id
            where k.k_id = :k_id
              and mac.mac_amount > 0
              and maci.maci_is_documented = 1
              and maci.maci_is_blocked <> 1
              and coalesce(maci_survey, '{}') <> '{}'
            into :mac_id
            do
            begin
              execute procedure pdn_save_json(:mac_id, :un_id, null, 'send_kk');
            end
          end
        end
      end
      suspend;
      exit;
    end
  end

  res_text = 'Не найден договор в процессе';
  suspend;
  exit;

end
