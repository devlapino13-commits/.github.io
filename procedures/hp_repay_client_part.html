CREATE PROCEDURE HP_REPAY_CLIENT_PART
AS
declare variable client_koef double precision;
declare variable group_balance decimal(18,4) = 0;
declare variable pr_round_perc integer;
declare variable tmp_amount decimal(18,4) = 0;
declare variable s_rec_no integer;
declare variable client_balance decimal(18,4) = 0;
declare variable tax_perc decimal(18,4) = 0;
begin
  select client_koef, pr_round_perc, tax_perc
  from temp_params
  into :client_koef, pr_round_perc, tax_perc;
  if (client_koef = 1) then
    exit;

  select sum(zad_os), max(s_rec_no)
  from temp_vrepay
  into :group_balance, s_rec_no;
  client_balance = round(group_balance * client_koef, 0);

  update temp_vrepay
  set pay_os = round(zad_os * :client_koef, 0),
      pay_perc = round(calcing_perc * :client_koef, :pr_round_perc),
      pay_fine = round(zad_fine * :client_koef, :pr_round_perc),
      pay_kommis = round(zad_kommis, :pr_round_perc),
      pay_extra = round(zad_extra * :client_koef, :pr_round_perc),
      pay_tax = round(calcing_perc * :tax_perc * :client_koef / 100, :pr_round_perc) + round(zad_kommis * :tax_perc / 100, :pr_round_perc)
  where s_date is not null; -- добавил igusev 22.09.2018 #operkroff_pay

-- << добавил igusev 22.09.2018  #operkroff_pay
  update temp_vrepay
  set pay_os = round(zad_os, 0),
      pay_perc = round(calcing_perc, :pr_round_perc),
      pay_fine = 0,
      pay_kommis = 0,
      pay_extra = 0,
      pay_tax = 0
  where s_date is null;
-- << -----------------------------------------

  select sum(pay_os)
  from temp_vrepay
  into :tmp_amount;

  if (client_balance <> tmp_amount) then
    update temp_vrepay
    set pay_os = pay_os + (:client_balance - :tmp_amount)
    where s_rec_no = :s_rec_no;
end
