CREATE PROCEDURE FM_BLOBPDFPAGE_INS(
    BPD_ID D_INTEGER,
    BPP_BLOB D_BLOB_BINARY,
    PAGE_NO D_SMALLINT)
RETURNS (
    ERR_TEXT D_STR_255,
    BPD_STATUS D_SMALLINT)
AS
begin
  err_text = ''; -- на случай, если понадобится вернуть ошибку
  bpd_status = 0;

  insert into blobpdfpage (bpp_from_bpd_id, bpp_blob, bpp_page_no)
  values (:bpd_id, :bpp_blob, :page_no);

  -- Если сохраняется последняя страница то в bpd_status будет 1
  -- Тогда FsMobile сгенерирует pdf и вызовет fm_blobpdfdoc_set_pdf
  -- Если же bpd_status = 0, ждем остальные страницы

  select bpd_status
  from blobpdfdoc
  where bpd_id = :bpd_id
  into bpd_status;

  suspend;
end
