CREATE PROCEDURE AAREP_PSK_DISB_TEMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID ID_TYPE,
    PR_ID SMALLINT,
    PR_LOAN_TYPE SMALLINT,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_PERIOD DECIMAL(15,2),
    KC_FROM_C_ID ID_TYPE,
    KC_AMOUNT_APPROVED CURR_TYPE,
    K_PERCENT_EFFECT PERC_TYPE,
    PR_PRODUCT_KIND SMALLINT,
    IS_SECURED SMALLINT,
    ITEM_1 SMALLINT,
    ITEM_2 SMALLINT,
    ITEM_3 SMALLINT,
    PR_CODE VARCHAR(100),
    SECTION INTEGER,
    K_DISB_DATE DATE,
    IS_KONTRACT INTEGER,
    IS_POS SMALLINT)
AS
declare variable Z_NAME_FIO varchar(255);
declare variable I integer;
declare variable K integer;
declare variable L integer;
declare variable K_KREDLINE_END date;
declare variable K_PERIOD_DAY integer;
declare variable K_ID_PRIOR ID_TYPE;
declare variable KD_K_ORIG ID_TYPE;
declare variable KD_K_ORIG_PRIOR ID_TYPE;
declare variable KS_DATE_END date;
begin

  item_1 = 1; item_2 = 0; item_3 = 0; suspend;

  item_1 = 2; item_2 = 0; item_3 = 0; suspend;
  item_1 = 2; item_2 = 1; item_3 = 0; suspend;
  item_1 = 2; item_2 = 2; item_3 = 0; suspend;

  item_1 = 3; item_2 = 0; item_3 = 0; suspend;

  item_1 = 3; item_2 = 1; item_3 = 0; suspend;
  item_1 = 3; item_2 = 1; item_3 = 1; suspend;
  item_1 = 3; item_2 = 1; item_3 = 2; suspend;

  item_1 = 3; item_2 = 2; item_3 = 0; suspend;
  item_1 = 3; item_2 = 2; item_3 = 1; suspend;
  item_1 = 3; item_2 = 2; item_3 = 2; suspend;

  item_1 = 3; item_2 = 3; item_3 = 0; suspend;
  item_1 = 3; item_2 = 3; item_3 = 1; suspend;
  item_1 = 3; item_2 = 3; item_3 = 2; suspend;
  item_1 = 3; item_2 = 3; item_3 = 3; suspend;

  item_1 = 3; item_2 = 4; item_3 = 0; suspend;
  item_1 = 3; item_2 = 4; item_3 = 1; suspend;
  item_1 = 3; item_2 = 4; item_3 = 2; suspend;
  item_1 = 3; item_2 = 4; item_3 = 3; suspend;

  item_1 = 3; item_2 = 5; item_3 = 0; suspend;
  item_1 = 3; item_2 = 5; item_3 = 1; suspend;
  item_1 = 3; item_2 = 5; item_3 = 2; suspend;
  item_1 = 3; item_2 = 5; item_3 = 3; suspend;
  item_1 = 3; item_2 = 5; item_3 = 4; suspend;

  item_1 = 4; item_2 = 0; item_3 = 0; suspend;
  item_1 = 4; item_2 = 1; item_3 = 0; suspend;
  item_1 = 4; item_2 = 1; item_3 = 1; suspend;
  item_1 = 4; item_2 = 1; item_3 = 2; suspend;
  item_1 = 4; item_2 = 1; item_3 = 3; suspend;
  item_1 = 4; item_2 = 2; item_3 = 0; suspend;

  kd_k_orig_prior = 0;
  k_id_prior = 0;

  for
  select k_id, pr_id, pr_loan_type, k_kolich_client_approved, k_period, k_percent_effect_fifd, pr_product_kind, ks_date_begin,
         pr_code
  from kontstate, kontract, product
  where ks_ol_is_delete = 0
  and ks_state = 1
  and cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end
  and ks_from_k_id = k_id
  and k_from_pr_id = pr_id
--  and k_id = 101010025277700101
  order by k_id
  into k_id, pr_id, pr_loan_type, k_kolich_client_approved, k_period, k_percent_effect, pr_product_kind, k_disb_date,
       pr_code
  do
  begin
    select max(sh_date) from schedule
    where sh_from_k_id = :k_id
    into k_kredline_end;
    k_period_day = k_kredline_end - k_disb_date;

    if (pr_code containing '.patent plus.' or pr_code containing '.patent.') then is_pos = 1;
    else is_pos = 0;

--      section = 1 - обеспеченный
    select is_secured
    from aarep_get_secured_loan(:pr_code,:pr_loan_type,:pr_product_kind,:pr_id,:k_disb_date,:k_kolich_client_approved,:k_id)
    into section;

    if (section = 1) then is_secured = 1; else is_secured = 0;
    item_1 = 0;
    item_2 = 0;
    item_3 = 0;

    if (is_pos = 0) then --------------------------------------------------------------------------------
    begin
        if (pr_code containing '.zalog_car.' or pr_code containing '.zalog.') then item_1 = 1;
        else
        if (is_secured = 1) then item_1 = 2;
        else item_1 = 3;
    
        if (item_1 = 2) then
        begin
          if (k_period_day <= 365) then item_2 = 1;  -- меньше или равно 365; добавлено по письму Ёльнуры от 19.09.2015
          else item_2 = 2;                       -- свыше года 2,2
        end else
        if (item_1 = 3) then
        begin
    --    Њредыдущий заменен следующим от 19.05.2015 по письму ®льнуры
          item_2 = case 
                     when k_period_day <= 30 then 1
                     when k_period_day <= 60 then 2
                     when k_period_day <= 180 then 3
                     when k_period_day <= 365 then 4
                     else 5
                   end;
        end
    
        if (k_id_prior <> k_id) then
        begin
          k_id_prior = k_id;
          kd_k_orig = null;
          select kd_k_orig from kontdevides
          where kd_k_new = :k_id
          into kd_k_orig;
    --    если не амнистирован то нулл будет и вместо нулла присваиваем k_id
          kd_k_orig = coalesce(kd_k_orig, k_id);
        end 
        is_kontract = 0;
        if (kd_k_orig_prior <> kd_k_orig) then
        begin
          kd_k_orig_prior = kd_k_orig;
          is_kontract = 1;
        end
    
        for
        select kc_from_c_id, kc_amount_approved from knclient
        where kc_from_k_id = :k_id
        and kc_amount_approved > 0
        into kc_from_c_id, kc_amount_approved
        do
        begin
          if (item_1 > 2) then
          begin
              if (item_2 <= 2) then
              begin
                item_3 = case
                           when kc_amount_approved <= 30000.000 then 1
                           else 2
                         end;
              end else
              if (item_2 in (3,4)) then
              begin
                item_3 = case
                           when kc_amount_approved <= 30000.000 then 1
                           when kc_amount_approved <= 100000.000 then 2
                           else 3
                         end;
              end else
              if (item_2 > 4) then
              begin
                item_3 = case
                           when kc_amount_approved <= 30000.000 then 1
                           when kc_amount_approved <= 60000.000 then 2
                           when kc_amount_approved <= 100000.000 then 3
                           else 4
                         end;
              end 
          end
          suspend;
          is_kontract = 0; -- если в группе несколько клиентов то чтобы несколько раз не посчитал
        end
    end else
    begin
      item_1 = 4;
      if (k_period_day <= 365) then item_2 = 1;
      else item_2 = 2;
      is_kontract = 1;

        for
        select kc_from_c_id, kc_amount_approved from knclient
        where kc_from_k_id = :k_id
        and kc_amount_approved > 0
        into kc_from_c_id, kc_amount_approved
        do
        begin

          if (item_2 = 1) then
          begin
            item_3 = case
                       when kc_amount_approved <= 30000.000 then 1
                       when kc_amount_approved <= 100000.000 then 2
                       else 3
                     end;
          end

          suspend;
        end

    end 

  end

end
