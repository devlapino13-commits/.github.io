CREATE PROCEDURE MKR_FILL_INDIV_POKET(
    KONTRACT_NO VARCHAR(20),
    CLIENT_FIO STR_255,
    REPORT_DATE_END DATE,
    REPORT_CATEG SMALLINT)
AS
declare variable pocket_id id_type;
declare variable kc_from_k_id id_type;
declare variable kc_number id_type;
declare variable report_date_begin date;
declare variable k_disb_date date;
begin
--PURPOSE: Для МКР. формирование кредитного пакета в БД по одному человеку.

  /* Создаю покет по активным и закрытым кредитам ----------------------------*/
  if (report_categ is null) then
    report_categ = 0;
  kc_from_k_id = null;

  select knc.kc_from_k_id, kont.k_disb_date, knc.kc_number
  from knclient knc
  inner join kontract kont on kont.k_id = knc.kc_from_k_id
  inner join clients cl on cl.c_id = knc.kc_from_c_id
  where kont.k_kontract_no = trim(:kontract_no) and
        cl.c_name_fio = trim(:client_fio)
  and knc.kc_amount_approved > 0
  into :kc_from_k_id, :k_disb_date, :kc_number;

  if (kc_from_k_id is not null) then
  begin
    report_date_begin = k_disb_date;

    pocket_id = gen_id(gen_mkrpoket_id, 1);
    insert into mkr_poket (mp_id, mp_create_time, mp_date_begin, mp_date_end, mp_type, mp_categ)
    values (:pocket_id, cast('NOW' as timestamp), :report_date_begin, :report_date_end, 2, :report_categ);

    /*--------------------------------------------------------------------------*/
    insert into mkr_credit (mc_type, mc_from_mp_id, mc_id, mc_from_c_id, mc_name, mc_placeofbirth, mc_countrycode,
                            mc_pincode, mc_bankruptcystatus, mc_dateofbirth, mc_from_k_id, mc_accountno, mc_credittype,
                            mc_creditlineamount, mc_purposeofcredit, mc_currencyofcredit, mc_initialamountofcredit,
                            mc_disoutamountofcredit, mc_annualinterestrate, mc_dateofgrant, mc_duetimefirstcontract,
                            mc_duetimelastcontract, mc_lastpaymentdate, mc_monthlypaymentamount,
                            mc_daysmainsumisoverdue, mc_daysinterestisoverdue, mc_creditperiodinmonths,
                            mc_oiaforrepperiod, mc_oiaforlastthreemonths, mc_numberofprolongs, mc_creditclasscode,
                            mc_creditstatuscode, mc_provision)
    select cr_type, :pocket_id, id, knc_id, name, 'azerbaycan', --Po umolchaniyu
           countrycode, pincode, 0, --Po umolchaniyu
           dateofbirth, :kc_from_k_id, accountno, '001', --Po umolchaniyu
           0, --Po umolchaniyu
           purposeofcredit, currencyofcredit, initialamountofcredit, disoutamountofcredit, annualinterestrate,
           dateofgrant, duetimefirstcontract, duetimelastcontract, lastpaymentdate, monthlypaymentamount,
           daysmainsumisoverdue, daysinterestisoverdue, creditperiodinmonths, oiaforrepperiod, oiaforlastthreemonths,
           numberofprolongs, creditclasscode, creditstatuscode, 0 --Po umolchaniyu
    from mkr_fill_indiv_poket_temp(:report_date_end, :kc_from_k_id, :kc_number, :report_categ);
  end
  exit;
end
