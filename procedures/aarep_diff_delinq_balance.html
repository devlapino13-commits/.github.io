CREATE PROCEDURE AAREP_DIFF_DELINQ_BALANCE(
    REG_ID D_ID,
    IN_LO_ID D_ID,
    PAR_DAYS D_SMALLINT,
    REP_DATE_END D_DATE)
RETURNS (
    FIRST_DATE_CURR_MONTH D_DATE,
    LAST_DATE_PRV_MONTH D_DATE,
    LO_ID D_ID,
    K_LO_ID_END D_ID,
    K_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    K_DIFF_DELINQ_BALANCE D_CURRENCY,
    K_DELINQ_BALANCE_BEGIN D_CURRENCY,
    K_DELINQ_BALANCE_END D_CURRENCY,
    K_BALANCE_END D_CURRENCY,
    K_DELINQ_DATE_END D_DATE)
AS
begin
  if (reg_id < 10000) then
    reg_id = cast('1' || right('0' || reg_id, 4) || '00' as d_id);

  -- НАЧАЛО КОДА
  first_date_curr_month = rep_date_end - extract(day from rep_date_end);
  last_date_prv_month = first_date_curr_month - 1;

  k_diff_delinq_balance = 0;
  k_delinq_balance_begin = 0;
  k_delinq_balance_end = 0;

  -->>>>>>> Получаем список просрочек на конец предыдущего месяца
  for select oks_kc_id, oks_from_k_id, oks_balance
      from oksummary oks
      left join region_list(:reg_id, 1, -1, 1) on :reg_id is not null
      where :last_date_prv_month between oks_date_begin and oks_date_end and
            oks_delinq_date < (:first_date_curr_month - :par_days) and
            oks_close_state = 0 and
            (oks_kc_id = :in_lo_id and
            :reg_id is null or
            :in_lo_id is null and
             id_list = oks_region_id)
      into lo_id, k_id, k_delinq_balance_begin
  do
  begin
    --Проверяем, не числится ли в ОВР на начало периода
    if (exists(select *
               from kontract k
               inner join sb_akt sa on sa.sba_from_k_id = k_id and
                     :last_date_prv_month between sa.sba_date_begin and sa.sba_date_end and
                     sa.sba_is_delete = 0
               where k_id = :k_id)) then
      continue; -- Если числитя, игнорируем и переходим на другой займ

    k_lo_id_end = null;
    k_balance_end = 0;
    k_diff_delinq_balance = 0; -- Разница изменения баланса просрочки по одному займу
    select oks_balance, oks_kc_id, oks_delinq_date
    from oksummary
    where oks_from_k_id = :k_id and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_close_state = 0
    into k_balance_end, k_lo_id_end, k_delinq_date_end;

    if (k_lo_id_end is not null) then
    begin
      if (k_delinq_date_end >= (:rep_date_end - :par_days)) then
        k_balance_end = 0; -- Значит займ уже не в просрочке

      k_diff_delinq_balance = k_balance_end - k_delinq_balance_begin;

      if (k_lo_id_end <> lo_id) then -- Значит займ на конец месяца числится у другого ВИПС
        k_diff_delinq_balance = 0;
    end

    if (k_lo_id_end is null) then
    begin
      -- значит займ закрыт
      if (exists(select *
                 from kontract k
                 inner join operkred ok on k_id = ok_from_k_id and
                       k.k_fact_closed_ol_id = ok_from_ol_id and
                       ok_ol_is_active = 1 and
                       ok_type = 5 --займ закрыт через списания
                 where k_id = :k_id and
                       k_fact_closed_ol_id is not null)) then
        k_diff_delinq_balance = 0;
    end

    if (k_diff_delinq_balance <> 0) then
    begin
      --Проверяем, не числится ли в ОВР
      if (exists(select *
                 from kontract k
                 inner join sb_akt sa on sa.sba_from_k_id = k_id and
                       :last_date_prv_month between sa.sba_date_begin and sa.sba_date_end and
                       sa.sba_is_delete = 0
                 where k_id = :k_id)) then
        k_diff_delinq_balance = 0;
    end

    if (k_diff_delinq_balance <> 0) then
    begin
      select k_kontract_no
      from kontract k
      where k_id = :k_id
      into k_kontract_no;

      suspend;
    end
  end

  k_diff_delinq_balance = 0; --Важно! Не убирать!
  k_delinq_balance_begin = 0; --Важно! Не убирать!
  k_delinq_balance_end = 0; --Важно! Не убирать!
  k_balance_end = 0; --Важно! Не убирать!
  lo_id = null; --Важно! Не убирать!
  k_lo_id_end = null; --Важно! Не убирать!
  k_id = null; --Важно! Не убирать!
  k_kontract_no = null; --Важно! Не убирать!
  k_delinq_date_end = null; --Важно! Не убирать!

  -- Получаем список новых просрочек на конец месяца
  for select oks_kc_id, oks_from_k_id, oks_balance
      from oksummary
      left join region_list(:reg_id, 1, -1, 1) on :reg_id is not null
      where oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_delinq_date >= (:first_date_curr_month - :par_days) and
            oks_delinq_date < (:rep_date_end - :par_days) and
            (oks_kc_id = :in_lo_id and
            :reg_id is null or
            :in_lo_id is null and
            id_list = oks_region_id)
      into lo_id, k_id, k_delinq_balance_end
  do
  begin
    -- если на начало месяца займ был у этого ВИПС, то считаем как увеличение баланса просрочек
    if (exists(select *
               from oksummary
               where oks_from_k_id = :k_id and
                     oks_date_begin <= :last_date_prv_month and
                     oks_date_end >= :last_date_prv_month and
                     oks_kc_id = :lo_id)) then
    begin
      k_diff_delinq_balance = k_diff_delinq_balance + k_delinq_balance_end;

      if (k_diff_delinq_balance <> 0) then
      begin
        select k_kontract_no
        from kontract k
        where k_id = :k_id
        into k_kontract_no;

        suspend;
      end

      k_diff_delinq_balance = 0; --Важно! Не убирать!
    end
  end
  -- КОНЕЦ КОДА
end
