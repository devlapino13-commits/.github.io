                               HRV_DATE_END_FACT >= :RES_DATE_BEGIN and

                                     CPD_ADD_DATA_2, -- Рост портфеля без просрочек

          set CPR_ADDITIONAL_POINTS = 6,
              CPR_COMMENT = 'Доп. бал за 2 место: 6'
          where CPR_FROM_CPP_ID = :CPP_ID and
                CPR_ID = :CPR_ID;
        if (:ORDER_NO = 3) then
          update COMPETRESULT
          set CPR_ADDITIONAL_POINTS = 3,
              CPR_COMMENT = 'Доп. бал за 3 место: 3'
          where CPR_FROM_CPP_ID = :CPP_ID and
                CPR_ID = :CPR_ID;
      end
    end

    execute procedure HP_COMPETITION_RECALC_ORDER(:CPP_ID);
  end
  if (STAGE = 3) then -- если финальные итоги
  begin
    -- Удаляем существующие расчеты за этот период
    delete from COMPETRESULT
    where (CPR_FROM_CPP_ID = :CPP_ID);

    for select CPR_OBJECT_KIND, CPR_OBJECT_ID, avg(CPR_FINAL_POINTS)
        from COMPETRESULT, COMPETPERIOD
        where CPP_FROM_RD_ID = :RD_ID and
              CPP_ID = CPR_FROM_CPP_ID and
              CPP_STAGE = (:STAGE - 1) and
              CPP_DATE_END between :RES_DATE_BEGIN and :RES_DATE_END
        group by CPR_OBJECT_KIND, CPR_OBJECT_ID
        into :KIND, :RES_TEAM_ID, :CPR_FINAL_POINTS
    do
    begin
      CPR_GROUP_NAME = '';
      -- Получаем название Группы где последний раз была команда
      select CPR_GROUP_NAME
      from COMPETRESULT, COMPETPERIOD
      where CPR_FROM_CPP_ID = CPP_ID and
            CPP_FROM_RD_ID = :RD_ID and
            CPR_OBJECT_KIND = -1 and
            CPR_OBJECT_ID = :RES_TEAM_ID and
            CPR_GROUP_NAME <> '' and
            CPP_ORDERNO < :CPP_ORDERNO
      order by CPP_ORDERNO desc rows 1
      into :CPR_GROUP_NAME;
      if (CPR_GROUP_NAME is null) then
        CPR_GROUP_NAME = '';

      insert into COMPETRESULT (CPR_FROM_CPP_ID, CPR_GROUP_NAME, CPR_OBJECT_KIND, CPR_OBJECT_ID, CPR_CALCED_POINTS,
                                CPR_RESULT_KIND, CPR_FROM_UN_ID)
      values (:CPP_ID, :CPR_GROUP_NAME, :KIND, :RES_TEAM_ID, :CPR_FINAL_POINTS, 0, :UN_ID)
      returning CPR_ID
      into :CPR_ID;
    end
    execute procedure HP_COMPETITION_RECALC_ORDER(:CPP_ID);
  end
  suspend;
end
