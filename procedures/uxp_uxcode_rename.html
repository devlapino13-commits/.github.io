CREATE PROCEDURE UXP$UXCODE_RENAME(
    UX_UUID D_UUID,
    NEW_CODE D_STR_CODE,
    NEW_NAME D_STR_100,
    UX_UUID_CODE D_STR_CODE = null)
AS
declare variable ux_kind d_str_code;
declare variable ux_is_base d_bool;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  select uxf$kind(uu_kind, 'code'), uu_is_base
  from ux$uniq
  where uu_id = :ux_uuid
  into ux_kind, ux_is_base;

  ux_kind = uxf$code(ux_uuid, 'kind');

  if (ux_kind in ('uxform', 'uxcommand', 'uxreport')) then
  begin
    if (ux_is_base = 0) then
    begin
      select ufm_ux_uuid
      from ux$form
      where ufm_uuid = :ux_uuid
      into ux_uuid;
    end

    update ux$code
    set ux_scode = :new_code,
        ux_name = :new_name
    where ux_uuid = :ux_uuid;
  end else
  if (ux_kind = 'uxframe') then
  begin
    update ux$frame
    set ufr_code = :new_code,
        ufr_name = :new_name
    where ufr_uuid = :ux_uuid;
  end else
  if (ux_kind = 'uxalias') then
  begin
    update ux$alias
    set uxa_code = :new_code,
        uxa_name = :new_name
    where uxa_uuid = :ux_uuid;
  end else
    exception e_message f_format('Неопределенный UxCode, ux_kind: %1', ux_kind);

  if (row_count = 0) then
    exception e_message 'UxCode объект не найден';
end
