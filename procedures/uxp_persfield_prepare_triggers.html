CREATE PROCEDURE UXP$PERSFIELD_PREPARE_TRIGGERS(
    TABLE_NAME D_STR_CODE)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_SCRIPT D_STR_8100)
AS
declare variable field_name d_str_code;
declare variable key_field d_str_code;
declare variable key_count d_integer;
declare variable link_source d_str_code;
declare variable link_field d_str_code;
declare variable value_field d_str_255;
declare variable pif_owner_field d_str_code;
declare variable owner_key_field d_str_code;
begin
  res_is_error = 0;
  res_script = '';

  table_name = trim(lower(table_name));
  key_field = '';
  key_count = 0;

  if (not exists(select * from ux$persfield where pif_source = :table_name)) then
    res_script = 'Нет полей в таблице зарегистрированные как персональные данные ' || upper(table_name);
  else
  begin
    select key_fields, key_count
    from uxp$uxfield_db_keyfields(:table_name, 0)
    into key_field, key_count;
  
    if (key_field = '') then
      res_script = 'Не найдено ключевое поле таблицы: ' || upper(table_name);
    else
    if (key_count > 1) then
    begin
      if (table_name = 'addrclient') then
        key_field = 'adc_from_c_id';
      else
        res_script = 'Таблицы с составными первичными ключами не поддерживаются: ' || upper(table_name);
    end
  end

  if (res_script <> '') then
  begin
    res_is_error = 1;
    suspend;
    exit;
  end

  res_script = '

SET TERM ^ ;

CREATE OR ALTER TRIGGER ' || upper(table_name) || '_TRG_PIF_UPD FOR ' || upper(table_name) || '
ACTIVE AFTER INSERT OR UPDATE POSITION 888
AS
begin
  -- Триггер генерируется автоматический через процедуру UXP$PERSFIELD_PREPARE_TRIGGERS';

  for select pif_field_name, pif_link_source, pif_link_field, pif_owner_field
  from ux$persfield
  where pif_source = :table_name
  into field_name, link_source, link_field, owner_key_field
  do
  begin
    if (owner_key_field = '') then
      pif_owner_field = 'null';
    else
      pif_owner_field = 'new.' || owner_key_field;

    value_field = 'new.' || field_name;

    if (link_source <> '' and link_field <> '') then
      value_field = 'uxf$perslink(''' || link_source || ''', new.' || link_field || ')';

    if (link_field = '') then
      link_field = field_name;

    res_script = res_script || '

  if ((inserting and new.' || link_field || ' is not null) or (updating and new.' || link_field || ' is distinct from old.' || link_field || ')) then
    execute procedure hp_persvalue_update(''' || table_name || ''', ''' || field_name || ''', ' || pif_owner_field || ', ' || value_field || ');';
  end

  pif_owner_field = 'old.' || owner_key_field;

  res_script = res_script || '
end
^
SET TERM ; ^';


  res_script = res_script || '
SET TERM ^ ;

CREATE OR ALTER TRIGGER ' || upper(table_name) || '_TRG_PIF_DEL FOR ' || upper(table_name) || '
ACTIVE AFTER DELETE POSITION 888
AS
begin
  -- Триггер генерируется автоматический через процедуру UXP$PERSFIELD_PREPARE_TRIGGERS

  execute procedure hp_persvalue_delete(''' || table_name || ''', '''', ' || pif_owner_field || ');
end
^
SET TERM ; ^
';

  suspend;
end
