CREATE PROCEDURE HP_PARSE_JSON_AFM_CHECK(
    JSON_INFO D_BLOB_TEXT,
    EDC_CODE D_STR_CODE)
RETURNS (
    CATEGORY_ID D_SMALLINT,
    STATUS D_SMALLINT)
AS
declare variable I D_INTEGER;
declare variable PATH D_STR_100;
declare variable OBJECT D_STR_4096;
declare variable target_fio d_str_255;
declare variable first_name d_str_255;
declare variable last_name d_str_255;
declare variable middle_name d_str_255;
declare variable fio_full d_str_255;
declare variable CATEGORY_COUNT D_INTEGER;
begin
    if (coalesce(:edc_code, '') not in ('afm_name_check', 'afm_iin_check')) then
        exception e_default f_format('Unknown edc_code. Expected: afm_name_check/afm_iin_check, actual: %1', :edc_code);

    i = 0;
    path = 'data.sanctions';

    category_count = coalesce(f_json_blob_get_count(:json_info, :path), 0);

    if (edc_code = 'afm_name_check') then
    begin
        target_fio = f_json_blob_get_value(:json_info, 'data.search_parameter', '');

        if (coalesce(target_fio, '') = '') then
            exception e_message 'Не удалось получить ФИО клиента для сравнения';

        select res_text
        from hp_txt_clear(:target_fio)
        into :target_fio;
    end

    /*
        Пример объекта, который обрабатывается в цикле:

        {
            "id": 2731,
            "category_id": 9,
            "name": "лиц (ранее вовлеченных) в НОН",
            "status": "acting",
            "first_name": {
                "String": "АЗАМАТ",
                "Valid": true
            },
            "last_name": {
                "String": "ЖАКУПОВ",
                "Valid": true
            },
            "middle_name": {
                "String": "СЕРИКОВИЧ",
                "Valid": true
            },
            "iin": "920107351464",
            "added_date": null,
            "removed_date": null,
            "indication": false,
            "all_names": null
        }

        Таких объектов в ответе возвращается множество, по разным категориям
    */

    while(:i < :category_count) do
    begin
        object = f_json_blob_get_value(:json_info, f_format('%1[%2]', :path, :i), '');
        category_id = f_json_str_get_value(:object, 'category_id', '0');
        status = f_enum('afm_status', f_json_str_get_value(:object, 'status', '0'), 'id');

        -- RFCKZ-295
        if (edc_code = 'afm_name_check') then
        begin
            -- При проверке по ФИО 11 и 12 категорию пропускаем (эти категории проверяются по ИИН)
            if (:category_id in (11,12)) then
            begin
                i = i + 1;
                continue;
            end

            first_name = f_json_str_get_value(:object, 'first_name.String', '');
            last_name = f_json_str_get_value(:object, 'last_name.String', '');
            middle_name = f_json_str_get_value(:object, 'middle_name.String', '');

            /*
                С ФИО в ответе у АФМ происходит полная каша. В некоторых ответах ФИО корректно разбивается
                на части, каждая из которых находится на своем месте, т.е. имя в first_name, фамилия в last_name,
                отчество в middle_name.

                Но бывают случаи, когда полное ФИО находится только в одном месте, к примеру, в first_name:

                "first_name": {
                  "String": "МАХМУД  СУЛТАН ",
                  "Valid": true
                },
                "last_name": {
                  "String": "",
                  "Valid": true
                },
                "middle_name": {
                  "String": "",
                  "Valid": true

               Поэтому, прежде чем сравнивать ФИО, его нужно корректно достать. Чтобы сделать это
               конкатенируем строки следующим образом: last_name (Фамилия) + first_name (Имя) + middle_name (Отчество)
               Получившаяся в результате строка и будет итоговым ФИО, с которым будет сравниваться ФИО клиента
            */

            fio_full = last_name || ' '  || first_name || ' ' || middle_name;

            select res_text
            from hp_txt_clear(:fio_full)
            into :fio_full;

            -- Если ФИО различны, прололжаем
            if (f_levenstain_is_same(:target_fio, :fio_full) = 0) then
            begin
                i = i + 1;
                continue;
            end
        end

        i = i + 1;
        suspend;
    end
end
