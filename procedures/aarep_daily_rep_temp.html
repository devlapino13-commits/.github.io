CREATE PROCEDURE AAREP_DAILY_REP_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_OFFICE VARCHAR(100),
    AAREP_ACTIVE_PORT_GR DECIMAL(18,4),
    AAREP_ACTIVE_PORT_PAR DECIMAL(18,4),
    AAREP_ACTIVE_PORT_IND DECIMAL(18,4),
    AAREP_ACTIVE_PORT_PER DECIMAL(18,4),
    AAREP_ACTIVE_PORT_IPF DECIMAL(18,4),
    AAREP_ACTIVE_PORT_ALL DECIMAL(18,4),
    AAREP_ACTIVE_CLIENT_END INTEGER,
    AAREP_DELIQ_BALANCE_ALL DECIMAL(18,4),
    AAREP_DELIQ_BALANCE_7 INTEGER,
    AAREP_DELIQ_BALANCE_30 INTEGER)
AS
declare variable PR_LOAN_TYPE integer;
declare variable AAREP_K_ID decimal(18,0);
declare variable PARAL integer;
declare variable EMP integer;
declare variable AAREP_ACTIVE_PORT_END decimal(18,4);
declare variable DEL_DAYS integer;
declare variable REG_ID integer;
declare variable REG_ID_PRIOR integer;
declare variable PR_CODE varchar(100);
begin
  /* Активные на начало */
  aarep_active_port_gr = 0;
  aarep_active_port_par = 0;
  aarep_active_port_ind = 0;
  aarep_active_port_per = 0;
  aarep_active_port_ipf = 0;
  aarep_active_port_all = 0;
  aarep_active_client_end = 0;
  aarep_deliq_balance_all = 0;
  aarep_deliq_balance_7 = 0;
  aarep_deliq_balance_30 = 0;
  /* Активные на конец */
  /*
   for select aarep_k_id, aarep_k_balance, aarep_kolich_active,
   pr_loan_type, pr_is_employer, pr_category, reg_code||' '||reg_name
   from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 3),
   kontract, product, region, knaktlist
   where k_id = aarep_k_id
   and k_from_pr_id = pr_id
   and aarep_region_id = reg_unicode
   and k_from_kf_id = kal_from_kf_id
   and knaktlist.is_delete = 0
   and :report_date_end between kal_date and kal_date_end
   and kal_new_kf_officer <> 1000000000003
   into :aarep_k_id, :AAREP_ACTIVE_PORT_END, :AAREP_ACTIVE_CLIENT_END,
   :pr_loan_type, :emp, :paral, :aarep_office */

  reg_id_prior = 0;
  for select k_id, oks_balance, k_kolich_client_approved, pr_loan_type, pr_is_employer, pr_category, oks_region_id,
             pr_code
      from oksummary, kontract, product, region_list(:report_region_id, 1, 3, 1)
      where :report_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_region_id = id_list and
            oks_kc_id <> 1000000000003 and
            oks_from_k_id = k_id and
            k_from_pr_id = pr_id
            and oks_close_state = 0 
      order by oks_region_id
      into aarep_k_id, aarep_active_port_end, aarep_active_client_end, pr_loan_type, emp, paral, reg_id, pr_code
  do
  begin
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into aarep_office;
    end
    aarep_active_port_gr = 0;
    aarep_active_port_par = 0;
    aarep_active_port_ind = 0;
    aarep_active_port_per = 0;
    aarep_active_port_ipf = 0;
    aarep_active_port_all = 0;
    if ((pr_loan_type = 0) and
        (emp = 0) and
        (paral = 0)) then
      aarep_active_port_gr = aarep_active_port_end;
    if ((pr_loan_type = 1) and
        (emp = 0) and
        (paral = 0)) then
      aarep_active_port_ind = aarep_active_port_end;
    if (paral = 1) then
      aarep_active_port_par = aarep_active_port_end;
    if (emp = 1) then
      aarep_active_port_per = aarep_active_port_end;
    if (pr_code containing '.IPF.') then
      aarep_active_port_ipf = aarep_active_port_end; -- было pr_code containing '.MURABAHA.' заменен от 02.04.2018 Улан
    aarep_active_port_all = aarep_active_port_gr + aarep_active_port_ind + aarep_active_port_par + aarep_active_port_per;
    suspend;
  end
  aarep_active_port_gr = 0;
  aarep_active_port_par = 0;
  aarep_active_port_ind = 0;
  aarep_active_port_per = 0;
  aarep_active_port_ipf = 0;
  aarep_active_port_all = 0;
  aarep_deliq_balance_all = 0;
  aarep_deliq_balance_7 = 0;
  aarep_deliq_balance_30 = 0;
  aarep_active_client_end = 0;

  /* Просроченные на конец */
  /*
   for select aarep_k_id, aarep_k_balance, AAREP_K_DELIQ_DAYS, reg_code||' '||reg_name
   from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 5),
   region, kontract, knaktlist
   where aarep_region_id = reg_unicode
   and aarep_k_id = k_id
   and k_from_kf_id = kal_from_kf_id
   and knaktlist.is_delete = 0
   and :report_date_end between kal_date and kal_date_end
   and kal_new_kf_officer <> 1000000000003
   into :aarep_k_id, :AAREP_DELIQ_BALANCE_ALL, :DEL_DAYS, :aarep_office */

  reg_id_prior = 0;
  for select k_id, oks_balance, :report_date_end - oks_delinq_date, oks_region_id
      from oksummary, kontract, product, region_list(:report_region_id, 1, 3, 1)
      where :report_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_region_id = id_list and
            oks_kc_id <> 1000000000003 and
            oks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            oks_delinq_date <= :report_date_end
            and oks_close_state = 0 
      order by oks_region_id
      into :aarep_k_id, :aarep_deliq_balance_all, :del_days, reg_id
  do
  begin
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into aarep_office;
    end
    aarep_deliq_balance_7 = 0;
    aarep_deliq_balance_30 = 0;
    if (del_days > 7) then
      aarep_deliq_balance_7 = aarep_deliq_balance_all;
    if (del_days > 30) then
      aarep_deliq_balance_30 = aarep_deliq_balance_all;
    suspend;
  end
end
