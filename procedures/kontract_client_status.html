CREATE PROCEDURE KONTRACT_CLIENT_STATUS(
    KONT_ID DECIMAL(18,0),
    DD DATE)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_AMOUNT DECIMAL(18,4),
    C_CIRCLE INTEGER,
    C_STATUS_BEGIN VARCHAR(100),
    C_STATUS_AFTER VARCHAR(100),
    C_LOANS_ACTIVE INTEGER,
    C_LOANS_PROCESS INTEGER,
    C_LOANS_CLOSED INTEGER,
    C_ID DECIMAL(18,0))
AS
declare variable c_last_bonus_status integer;
declare variable c_curr_bonus_status integer;
declare variable db_id integer;
begin

  :db_id = f_get_db_id();
  for select c_id, c_name_fio, kc_amount_approved, kc_circle
      from knclient, clients
      where kc_from_k_id = :kont_id and
            kc_from_c_id = c_id and
            kc_amount_approved > 0
      into :c_id, :c_name_fio, :c_amount, :c_circle
  do
  begin
    select c_last_status, c_curr_status, c_last_status_str, c_curr_status_str
    from hp_get_client_status_str(:c_id, :dd)
    into :c_last_bonus_status, :c_curr_bonus_status, :c_status_begin, :c_status_after;


    c_loans_active = 0;
    select count(*)
    from knclient, kontstate
    where ks_from_k_id = kc_from_k_id and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999' and
          ks_state = 1 and
          kc_from_c_id = :c_id and
          kc_amount_approved > 0
    into :c_loans_active;
    if (c_loans_active is null) then
      c_loans_active = 0;

    c_loans_process = 0;
    select count(*)
    from knclient, kontstate
    where ks_from_k_id = kc_from_k_id and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999' and
          ks_state = 0 and
          kc_from_c_id = :c_id and
          kc_amount_approved > 0
    into :c_loans_process;
    if (c_loans_process is null) then
      c_loans_process = 0;

    c_loans_closed = 0;
    select count(*)
    from knclient, kontstate
    where ks_from_k_id = kc_from_k_id and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999' and
          ks_state = 2 and
          kc_from_c_id = :c_id and
          kc_amount_approved > 0
    into :c_loans_closed;
    if (c_loans_closed is null) then
      c_loans_closed = 0;

    suspend;
  end
end
