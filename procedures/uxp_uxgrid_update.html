CREATE PROCEDURE UXP$UXGRID_UPDATE(
    GRID_UUID D_UUID,
    UI_HIDE_ON_GROUP D_BOOL,
    UI_GROUP_PANEL D_BOOL,
    UI_FIND_PANEL D_SMALLINT,
    UI_HIDE_SUMMARY D_BOOL,
    UI_AUTOWIDTH D_SMALLINT,
    UI_DISABLE_EDIT D_BOOL,
    UI_HEADER_AUTOHEIGHT D_BOOL = null,
    UI_SHOW_INFO_PANEL D_BOOL = null,
    UI_FORMATTING D_STR_8100 = null,
    UI_HIDE_INPLACE_EDITOR D_BOOL = null,
    UI_SHOW_EXCEL_IMPORT D_BOOL = null,
    GRID_UUID_CODE D_STR_CODE = null,
    UI_TREE_CAN_MOVE D_STR_CODE = null,
    UI_ALLOW_APPEND D_BOOL = null,
    UI_KIND D_STR_CODE = null,
    UI_CALENDAR_FIELDS D_STR_4096 = null)
AS
declare variable pos d_integer;
begin
  if (coalesce(grid_uuid, '') = '') then
  begin
    if (coalesce(grid_uuid_code, '') <> '') then
    begin
      pos = position('__', grid_uuid_code);
      if (pos > 0) then
      begin
        grid_uuid_code = substring(grid_uuid_code from pos + 2);
        grid_uuid_code = replace(grid_uuid_code, '_', '-');
      end

      grid_uuid = uxf$uniq(grid_uuid_code);
    end else
      exception e_default 'Emptry grid uuid';
  end

  update ux$grid
  set ugr_hide_on_group = :ui_hide_on_group,
      ugr_group_panel = :ui_group_panel,
      ugr_find_panel = :ui_find_panel,
      ugr_hide_summary = :ui_hide_summary,
      ugr_autowidth = :ui_autowidth,
      ugr_disable_edit = coalesce(:ui_disable_edit, ugr_disable_edit),
      ugr_header_autoheight = coalesce(:ui_header_autoheight, ugr_header_autoheight),
      ugr_show_info_panel = coalesce(:ui_show_info_panel, ugr_show_info_panel),
      ugr_hide_inplace_editor = coalesce(:ui_hide_inplace_editor, ugr_hide_inplace_editor),
      ugr_formatting = coalesce(:ui_formatting, ugr_formatting),
      ugr_show_excel_import = coalesce(:ui_show_excel_import, ugr_show_excel_import),
      ugr_tree_can_move = coalesce(:ui_tree_can_move, ugr_tree_can_move),
      ugr_allow_append = coalesce(:ui_allow_append, ugr_allow_append),
      ugr_ui_kind = coalesce(:ui_kind, ugr_ui_kind),
      ugr_calendar_fields = coalesce(:ui_calendar_fields, ugr_calendar_fields)
  where ugr_uuid = :grid_uuid;

  delete from ux$gritem
  where ugi_ugr_uuid = :grid_uuid and
        ugi_bnd_id = -2;

  uxf$syncmode('off');

  update ux$band
  set bnd_index = null
  where bnd_grid_uuid = :grid_uuid and
        bnd_parent_id is null
  order by bnd_position;
end
