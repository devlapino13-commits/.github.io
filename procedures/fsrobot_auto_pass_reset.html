CREATE PROCEDURE FSROBOT_AUTO_PASS_RESET
AS
declare variable day_ integer;
declare variable month_ integer;
declare variable c_id id_type;
begin
  select extract(day from current_date), extract(month from current_date)
  from rdb$database
  into :day_, :month_;
  if ((day_ = 13 and
      month_ = 1) or (day_ = 3 and
      month_ = 7)) then
  begin
    for select ps.ps_from_c_id
        from hr_pstatus ps
        where current_date between ps_date_begin and ps_date_end and
              ps_status = 1 and
              ps_is_working = 1
        into :c_id
    do
    begin
      update username
      set un_password_new = 'СБРОШЕН',
          un_password_reset = '1000000000000'
      where (un_from_c_id = :c_id);
    end
  end
end
