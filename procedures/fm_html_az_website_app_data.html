CREATE PROCEDURE FM_HTML_AZ_WEBSITE_APP_DATA(
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable CL_FIO D_STR_100;
declare variable CL_BIRTH_DATE D_DATE;
declare variable CL_PHONE D_STR_20;
declare variable CL_LANGUAGE D_STR_10;
declare variable DATE_APP D_DATE;
declare variable AW_UN_ID D_ID;
declare variable AW_ID D_ID;
declare variable LANG D_STR_LANG;
declare variable AW_STATUS D_SMALLINT;
begin
  res_text = '';
  lang = f_get_lang_code(un_id, null);
  res_text = (select res_style
              from fm_html_styles) || '<body>';

  un_id = (select un_id
           from hp_test_un_id(:un_id));

  select aw.aw_cl_fio, aw.aw_birth_date, aw.aw_phone, aw.aw_language, aw.aw_date, aw.aw_from_un_id, aw.aw_status
  from az_website aw
  where aw.aw_id = :obj_id
  into cl_fio, cl_birth_date, cl_phone, cl_language, date_app, aw_un_id, aw_status;

  res_text = blob_append(res_text, '<h2>' || f_lang(:lang, 'Информация о заявке') || '</h2><br>' ||
                         '<b>' || f_lang(:lang, 'ФИО') || ':' || '</b>' || ' ' || cl_fio || '<br>' ||
                         '<b>' || f_lang(:lang, 'Дата рождения') || ':' || '</b>' || ' ' || f_datetostr(cl_birth_date) || '<br>' ||
                         '<b>' || f_lang(:lang, 'Язык общения') || ':' || '</b>' || ' ' || cl_language || '<br>' ||
                         '<b>' || f_lang(:lang, 'Дата заявки') || ':' || '</b>' || ' ' || f_datetostr(:date_app) || '<br>');

  if (aw_un_id is null) then
  begin
    res_text = blob_append(res_text, '<br>' || f_uri_sql(f_lang(:lang, 'Забрать заявку'), 'take_az_app', 'az_website', :obj_id, 'sql', 'Подтвердите действие', '', 0, 0, '', '', 'dialog', '') || '</br>');
  end
  else
  if (un_id = aw_un_id) then
  begin
    res_text = blob_append(res_text, '<br><b>' || f_lang(:lang, 'Номер телефона') || ':</b>' || f_uri_phone(cl_phone) || '</br>');

    if (aw_status = 0) then
    begin
      res_text = blob_append(res_text, '<br><table width="100%"><col width="50%"><col width="50%">');
      res_text = blob_append(res_text, f_html_table_row(f_uri_sql(f_lang(lang, 'Отказать в заявке'), 'decline_az_app', 'az_website', :obj_id, 'combo', 'Укажите причину отказа', '', 0, 0, '', '', 'dialog', 'cancel_loan'),
                                               f_uri_sql(f_lang(lang, 'Принять заявку'), 'accept_az_app', 'az_website', :obj_id, 'sql', f_lang(:lang, 'Подтвердите действие'), '', 0, 0, '', '', 'dialog', ''),
                                               '', '', 0, 0, -1, -1, 2, ''));
    end
  end

  res_text = blob_append(res_text, '</body>');
  suspend;
end
