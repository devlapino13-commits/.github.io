CREATE PROCEDURE AAREP_PRODUCT_VALIDATE_LIST(
    DISB_DATE DATE,
    CLIENT_KOLICH INTEGER,
    LOAN_AMOUNT DECIMAL(18,4),
    LOAN_REPAY INTEGER)
RETURNS (
    PR_ID INTEGER,
    PR_NAME VARCHAR(255),
    PR_MEMO VARCHAR(255),
    PR_CURRENCY VARCHAR(3),
    PR_INDEX_CURR_CODE VARCHAR(3))
AS
declare variable kol integer;
declare variable un_name varchar(100);
declare variable is_hq_user integer;
declare variable pr_code varchar(100);
begin
  pr_memo = '';
  if (client_kolich <= 0) then
    client_kolich = null;
  if (loan_amount <= 0) then
    loan_amount = null;
  if (loan_repay <= 0) then
    loan_repay = null;

  un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN');

  is_hq_user = 0;
  if (exists(select *
             from username
             where un_curr_reg_id = f_hq_id() and
                   un_name = :un_name)) then
    is_hq_user = 1;

  for select pr_id, iif(coalesce(pr_memo, '') <> '', pr.pr_memo, pr_name), pr_currency, pr_index_curr_code, pr_memo,
             pr_code
      from product pr
      where pr_date_begin <= :disb_date and
            pr_date_end >= :disb_date
      into :pr_id, :pr_name, :pr_currency, :pr_index_curr_code, pr_memo, pr_code
  do
  begin
    if (pr_code not containing '.EMP.' or is_hq_user = 1) then -- если продукт для сотрудников, он будет виден только пользователю из ГО
    begin
      if (exists(select *
                 from prodlogic
                 where prl_from_pr_id = :pr_id and
                       coalesce(prl_date_begin, '01.01.1900') <= :disb_date and
                       coalesce(prl_date_end, '31.12.9999') >= :disb_date and
                       coalesce(:client_kolich, coalesce(prl_kolich_min, 0)) >= coalesce(prl_kolich_min, 0) and
                       coalesce(:client_kolich, coalesce(prl_kolich_min, 0)) <= coalesce(prl_kolich_max, 99999999999) and
                       coalesce(:loan_amount, coalesce(prl_amount_gr_min, 0)) >= coalesce(prl_amount_gr_min, 0) and
                       coalesce(:loan_amount, coalesce(prl_amount_gr_min, 0)) <= coalesce(prl_amount_gr_max, 99999999999) and
                       coalesce(:loan_repay, coalesce(prl_repay_min, 0)) >= coalesce(prl_repay_min, 0) and
                       coalesce(:loan_repay, coalesce(prl_repay_min, 0)) <= coalesce(prl_repay_max, 99999999999))) then
      begin
        suspend;
      end
    end
  end
end
