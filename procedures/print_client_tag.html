CREATE PROCEDURE PRINT_CLIENT_TAG(
    AAREP_K_ID DECIMAL(18,0))
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_TYPE INTEGER,
    C_CITIZEN_STR VARCHAR(255),
    KC_STATUS INTEGER,
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KC_AMOUNT_APPROVED_EKV DECIMAL(18,4),
    KC_NUMBER INTEGER,
    KC_PATENT_NO VARCHAR(40),
    KC_PATENT_DATE DATE,
    CA_ADDRESS VARCHAR(255),
    C_FULL_ADDRESS VARCHAR(255),
    C_FULL_ADDRESS_INDEX VARCHAR(255),
    KC_CIRCLE INTEGER,
    C_BANK_ACCOUNT_NO VARCHAR(50),
    C_BANK_CODE VARCHAR(10),
    C_BANK_NAME VARCHAR(255),
    C_BANK_CORR_ACC VARCHAR(255),
    K_KOMMIS_AMOUNT DECIMAL(18,4),
    KC_KOMMIS INTEGER,
    KC_BONUS_TEXT VARCHAR(30),
    C_ADDRESS_FACT VARCHAR(255),
    C_ADDRESS_RODINA_YUR VARCHAR(255),
    C_ADDRESS_RODINA_FACT VARCHAR(255),
    KC_NSP_AMOUNT DECIMAL(18,4),
    KC_KOMMIS_AND_NSP DECIMAL(18,4),
    C_ALTERNAT_FIO VARCHAR(100),
    KC_LOAN_TARGET VARCHAR(255),
    C_CELL_NO VARCHAR(20),
    C_EMAIL VARCHAR(100),
    CONTACT_DATA VARCHAR(5000),
    CONTACT_DATA_RF VARCHAR(5000),
    C_PASS_WHO_CODE VARCHAR(20),
    C_TEL VARCHAR(100),
    KC_LOAN_TARGET_LANG VARCHAR(255),
    C_UNIQ_ACC VARCHAR(12),
    FACT_LOAN_TARGET VARCHAR(100),
    FOR_PURPOSE_USED VARCHAR(100),
    ANALIZ_STATUS VARCHAR(100),
    ANALIZ_DATE DATE,
    ANALIZ_LO_FIO VARCHAR(100))
AS
declare variable kc_from_c_id decimal(18,0);
declare variable k_kommis_perc decimal(18,7);
declare variable k_amount_approved decimal(18,4);
declare variable st_staff integer;
declare variable st_client integer;
declare variable k_disb_date date;
declare variable k_bonus integer;
declare variable c_citizen integer;
declare variable k_nsp_percent decimal(18,7);
declare variable k_kolich_client_approved integer;
declare variable k_nsp_amount decimal(18,4);
declare variable kc_nsp_amount_all decimal(18,4);
declare variable kol integer;
declare variable kc_kommis_all decimal(18,4);
declare variable cont_data_temp varchar(700);
declare variable kk integer;
declare variable cc_type integer;
declare variable acl_id id_type;
declare variable acl_fill_date date;
declare variable kc_id d_id;
declare variable obj_kind d_integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  k_nsp_percent = 0;
  select k_kommis_perc, k_amount_approved, k_disb_date, k_kolich_client_approved
  from kontract
  where k_id = :aarep_k_id
  into :k_kommis_perc, :k_amount_approved, :k_disb_date, :k_kolich_client_approved;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -2 -- нсп от доп.процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end
  k_nsp_amount = 0;
  kc_kommis_all = 0;
  kc_nsp_amount_all = 0;
  kol = 0;
  for select c_name_fio, c_birth_date, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire, c_type, c_citizen,
             kc_status, kc_amount_approved, kc_amount_approved * k_index_curr_rate, kc_number, kc_patent_no,
             kc_patent_date, adl_address_str, kc_circle, kc_from_c_id, c_name_eng, rd_name, rd_memo,
             (select cont_value
              from hp_contdata_get_cell_main(0, c_id)),
             (select cont_value
              from hp_contdata_get_email_main(0, c_id)), cl.c_pass_who_code, kc_id
      from clients cl, knclient, addrclient, addrlist, kontract, refdata
      where kc_from_k_id = :aarep_k_id and
            kc_from_c_id = c_id and
            k_id = :aarep_k_id and
            kc_amount_approved > 0 and
            adc_from_c_id = c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id and
            kc_loan_target = rd_id
      order by kc_number
      into :c_name_fio, :c_birth_date, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_type, :c_citizen,
           :kc_status, :kc_amount_approved, :kc_amount_approved_ekv, :kc_number, :kc_patent_no, :kc_patent_date,
           :c_full_address, :kc_circle, :kc_from_c_id, :c_alternat_fio, :kc_loan_target, :kc_loan_target_lang,
           :c_cell_no, :c_email, :c_pass_who_code, :kc_id
  do
  begin
    if (not((c_alternat_fio containing ('uulu')) or (c_alternat_fio containing ('kyzy')))) then
    begin
      kk = position(' ', :c_alternat_fio);
      if (kk > 0) then
      begin
        kk = position(' ', c_alternat_fio, kk + 1);
        if (kk > 0) then
          c_alternat_fio = substring(:c_alternat_fio from 1 for kk - 1);
      end
    end

    if (kc_loan_target = 'Не указан') then
      kc_loan_target = '';

    select rd_name
    from refdata
    where rd_id = :c_citizen
    into :c_citizen_str;

    kol = kol + 1;

    select c_last_bonus_status, c_last_bonus_status_str
    from hp_kontstate_client_status(:kc_from_c_id, :k_disb_date)
    into :k_bonus, :kc_bonus_text;

    ca_address = c_full_address;

    c_address_fact = '';
    c_address_rodina_yur = '';
    c_address_rodina_fact = '';

    select adl_address_str,
           left(iif(adc_phone is null or (adc_phone = ''), 'Номер не указан', adc_phone), 100)
    from addrclient, addrlist
    where adc_from_c_id = :kc_from_c_id and
          adc_address_type = 2 and
          adc_from_adl_id = adl_id
    into :c_address_fact, c_tel;
    if (c_address_fact is null) then
      c_address_fact = '';
    if (c_address_fact = 'Не указан') then
      c_address_fact = '';

    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :kc_from_c_id and
          adc_address_type = 4 and
          adc_from_adl_id = adl_id
    into :c_address_rodina_yur;
    if (c_address_rodina_yur is null) then
      c_address_rodina_yur = '';
    if (c_address_rodina_yur = 'Не указан') then
      c_address_rodina_yur = '';

    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :kc_from_c_id and
          adc_address_type = 5 and
          adc_from_adl_id = adl_id
    into :c_address_rodina_fact;
    if (c_address_rodina_fact is null) then
      c_address_rodina_fact = '';
    if (c_address_rodina_fact = 'Не указан') then
      c_address_rodina_fact = '';

    c_bank_account_no = '';
    c_bank_code = '';
    c_bank_name = '';

    select res
    from fhp_round_bank(:k_amount_approved * :k_kommis_perc / 100.00, 0)
    into k_kommis_amount;
    select res
    from fhp_round_bank(:k_kommis_amount * :k_nsp_percent / 100, 0)
    into k_nsp_amount;

    select res
    from fhp_round_bank(:kc_amount_approved * :k_kommis_perc / 100, 2)
    into kc_kommis;
    if (kol < k_kolich_client_approved) then
      kc_kommis_all = kc_kommis_all + kc_kommis;
    else
      kc_kommis = k_kommis_amount - kc_kommis_all;

    select res
    from fhp_round_bank(:kc_kommis * :k_nsp_percent / 100, 0)
    into kc_nsp_amount;
    if (kol < k_kolich_client_approved) then
      kc_nsp_amount_all = kc_nsp_amount_all + kc_nsp_amount;
    else
      kc_nsp_amount = k_nsp_amount - kc_nsp_amount_all;

    kc_kommis_and_nsp = kc_kommis + kc_nsp_amount;

    if (kc_from_c_id is not null) then
    begin
      select bc_account_no, bc_from_b_id, b_name, b_corr_acc
      from bankclient, banks
      where bc_from_c_id = :kc_from_c_id and
            bc_from_b_id = b_id and
            bc_is_main = 1 and
            bc_is_active = 1 rows 1
      into :c_bank_account_no, :c_bank_code, :c_bank_name, :c_bank_corr_acc;
    end

    contact_data_rf = '';
    contact_data = '';
    for select cc_type, contact_data
        from aarep_kontract_contacts(:kc_from_c_id)
        into cc_type, :cont_data_temp
    do
    begin
      if (cc_type = 0) then
        contact_data_rf = contact_data_rf || ascii_char(13) || cont_data_temp;
      if (cc_type = 1) then
        contact_data = contact_data || ascii_char(13) || cont_data_temp;
    end

    c_uniq_acc = 'XXX';
    select uniq_acc_str
    from hp_show_clientuniq_acc_no(:kc_from_c_id)
    into c_uniq_acc; --Лиц.счет добавлен 01.07.16

    if (c_uniq_acc = 'XXX') then
      c_uniq_acc = 'Не закреплен';

    if (c_full_address containing 'г.Санкт-Петербург') then
    begin
      if (c_full_address starting 'г') then
        c_full_address_index = '190000, ' || c_full_address;
      else
        c_full_address_index = '190000' || substring(c_full_address from 7);
    end
    else
      c_full_address_index = c_full_address;

    acl_id = null;

    fact_loan_target = '-'; -- фактическая цель займа
    for_purpose_used = '-'; -- целевое использование
    analiz_status = '-'; -- Статус проведения повторного анализа платежеспособности клиента
    analiz_lo_fio = '-';

    select al.acl_id, cast(al.acl_last_datetime as date), cl.c_name_fio, al.acl_object_kind
    from actionlist al
    inner join refdata rd on rd.rd_id = al.acl_from_rd_id
    inner join clients cl on cl.c_id = al.acl_respons_c_id
    where al.acl_date >= :k_disb_date + 59 and
          al.acl_status = 2 and
          ((al.acl_object_kind = 0 and al.acl_object_id = :kc_from_c_id) or
           (al.acl_object_kind = 8 and al.acl_object_id = :kc_id)) and
          rd.rd_code = 'Anketa.TargetUseLoan'
    order by al.acl_date
    rows 1
    into :acl_id, :acl_fill_date, :analiz_lo_fio, :obj_kind;

    analiz_date = null;

    if (acl_id is not null) then -- если есть заполненная анкета "Мониторинг целевого использования"
    begin
      select res_str_values
      from hp_objectprops_anketa_get(
        iif(:obj_kind = 0, 'Clients', 'KnClient'),
        iif(:obj_kind = 0, :kc_from_c_id, :kc_id),
        -1,
        'AnketaQuestions.LoanUseFact',
        :acl_fill_date,
        null)
      into :fact_loan_target;

      select cast(al.acl_last_datetime as date)
      from actionlist al
      inner join refdata rd on rd.rd_id = al.acl_from_rd_id
      inner join clients cl on cl.c_id = al.acl_respons_c_id
      where al.acl_parent_id = :acl_id and
            al.acl_status > -2 -- только заполненные
            and
            al.acl_object_kind = 0 and
            al.acl_object_id = :kc_from_c_id and
            rd.rd_code = 'Anketa.ClientAnaliz' rows 1
      into :analiz_date;

      if (analiz_date is null) then
        for_purpose_used = 'Да';
      else
      begin
        for_purpose_used = 'Нет';

        select res_str_values
        from hp_objectprops_anketa_get('Clients', :kc_from_c_id, -1, 'AnketaQuestions.Reanalysis', :analiz_date, null)
        into :analiz_status;
      end
    end

    suspend;
  end
end
