CREATE PROCEDURE A_TEST(
    REP_DATE DATE)
RETURNS (
    REG_ID INTEGER,
    REG_NAME VARCHAR(100),
    REG_DATE_OPEN DATE,
    REG_DATE_FIRST_DISB DATE,
    REG_LO_COUNT INTEGER,
    REG_HR_LO_COUNT INTEGER,
    REG_SSVZ_COUNT INTEGER,
    REG_REGION_NAME VARCHAR(100),
    REG_OKRUG_NAME VARCHAR(100),
    REG_MANAGER_FIO VARCHAR(100),
    REG_MANAGER_DATE_BEGIN DATE,
    REG_MANAGER_JOBNAME VARCHAR(100))
AS
declare variable need_suspend integer;
declare variable c_id decimal(18,0);
begin

  for select reg_unicode, reg_code || ' ' || reg_name, reg_date_open
      from region
      where reg_status = 3 and
            coalesce(reg_date_close, '31.12.2999') >= :rep_date
      into :reg_id, :reg_name, reg_date_open
  do
  begin
    need_suspend = 0;
    reg_lo_count = 0;
    reg_hr_lo_count = 0;
    reg_ssvz_count = 0;
    reg_manager_fio = '';
    reg_manager_date_begin = null;
    reg_manager_jobname = '';

    reg_date_first_disb = null;
    select min(ks_date_begin)
    from kontstate
    where ks_region_id = :reg_id and
          ks_state = 1 and
          ks_ol_is_delete = 0
    into :reg_date_first_disb;

    for select distinct oks_kc_id
        from oksummary
        where oks_region_id = :reg_id and
              oks_date_begin <= :rep_date and
              oks_date_end >= :rep_date and
              oks_balance > 0 and
              oks_kc_id > 1000000000004 and
              oks_close_state = 0 
        into :c_id
    do
    begin
      reg_lo_count = reg_lo_count + 1;
    end
    for select distinct ps_from_c_id
        from hr_pstatus, joblist, jobkind
        where ps_from_reg_id = :reg_id and
              ps_date_begin <= :rep_date and
              ps_date_end >= :rep_date and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')
        into :c_id
    do
    begin
      reg_hr_lo_count = reg_lo_count + 1;
    end
    for select distinct ps_from_c_id
        from hr_pstatus, joblist, jobkind
        where ps_from_reg_id = :reg_id and
              ps_date_begin <= :rep_date and
              ps_date_end >= :rep_date and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code = 'JobSystem.Region_KM'
        into :c_id
    do
    begin
      reg_ssvz_count = reg_ssvz_count + 1;
    end
    reg_region_name = '';
    reg_okrug_name = '';
    select aa_curr_level2_name, aa_curr_level3_name
    from aarep_get_office_categ(:reg_id, :rep_date)
    into :reg_region_name, reg_okrug_name;

    suspend;
    need_suspend = 0;
    reg_lo_count = 0;
    reg_hr_lo_count = 0;
    reg_ssvz_count = 0;
    reg_manager_fio = '';
    reg_manager_date_begin = null;
    reg_date_first_disb = null;
    reg_date_open = null;
    reg_region_name = '';
    reg_okrug_name = '';
    reg_manager_jobname = '';

    for select ps_from_c_id, c_name_fio, jl_name
        from hr_pstatus, joblist, jobkind, clients
        where ps_from_reg_id = :reg_id and
              ps_date_begin <= :rep_date and
              ps_date_end >= :rep_date and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code = 'JobSystem.Region_KM' and
              ps_from_c_id = c_id
        into :c_id, :reg_manager_fio, reg_manager_jobname
    do
    begin
      reg_manager_date_begin = null;
      select min(ps_date_begin)
      from hr_pstatus, joblist, jobkind
      where ps_from_reg_id = :reg_id and
            ps_date_begin <= :rep_date and
            ps_from_jl_id = jl_id and
            ps_from_c_id = :c_id and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.Region_KM'
      into :reg_manager_date_begin;
      suspend;
    end

  end
end
