CREATE PROCEDURE HP_DO_PEREPLAT_CORRECT(
    OL_DATE DATE,
    REG_NAME D_STR_CODE,
    DT_ACCOUNT D_STR_CODE,
    KT_ACCOUNT D_STR_CODE,
    K_KONTRACT_NO D_STR_CODE,
    OPER_KIND INTEGER,
    AMOUNT CURR_TYPE)
AS
declare variable ol_id id_type;
declare variable reg_id integer;
declare variable dt_acc d_str_code;
declare variable kt_acc d_str_code;
declare variable reg_code d_str_code;
begin
  select reg_unicode, reg_code
  from region
  where reg_name = :reg_name
  into :reg_id, reg_code;

  if (reg_id is null) then
    exception e_default 'Region not found: ' || coalesce(reg_name, '<null>');

  dt_acc = dt_account;
  dt_acc = replace(dt_acc, '/', '000000');
  dt_acc = left(dt_acc, 5) || right(substring(dt_acc from 6), 6);
  dt_acc = reg_code || '810' || dt_acc;

  kt_acc = kt_account;
  kt_acc = replace(kt_acc, '/', '000000');
  kt_acc = left(kt_acc, 5) || right(substring(kt_acc from 6), 6);
  kt_acc = reg_code || '810' || kt_acc;

  select ob_from_ol_id
  from operbook
  where ob_ol_is_active = 1 and
        ob_ol_reg_id = :reg_id and
        ob_account_dt = :dt_acc and
        ob_account_kt = :kt_acc and
        ob_ol_date = :ol_date and
        ob_summa = :amount
  into :ol_id;

  if (ol_id is null) then
    exception e_default 'OL_ID not found. Ol_date: ' || coalesce(ol_date, '<null>') || 'reg_name: ' || coalesce(reg_name, '<null>') || 'dt_account: ' || coalesce(dt_acc, '<null>') || 'kt_account: ' || coalesce(kt_acc, '<null>') || 'amount: ' || coalesce(amount, '<null>');

  execute procedure hp_do_pereplat_ol_id_link(:ol_id, :k_kontract_no, :oper_kind, :dt_account, :amount);
end
