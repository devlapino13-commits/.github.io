CREATE PROCEDURE MC_QR_PAYMENT_SUMMAR(
    MCL_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    CS_NAME D_STR_255,
    SUM_IQP_AMOUNT D_CURRENCY,
    COUNT_IQP_ID D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable db_id d_integer;
declare variable cs_id d_integer;
declare variable lang_code d_str_lang;
begin
  db_id = f_get_db_id();
  err_text = '';
  lang_code = f_get_lang_code(null, :mcl_id);

  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id --18331
  into c_id; --1090101387630

  if (c_id is null or db_id not in (39801, 41701)) then
  begin
    suspend;
    exit;
  end

  select cs.cs_name
  from clientshops cs
  where cs.cs_from_c_id = :c_id
  into cs_name;

  select count(p.iqp_id), sum(p.iqp_amount)
  from ipc_qr_payments p
  inner join clientshops cs on cs.cs_id = p.iqp_from_cs_id
  where cs.cs_from_c_id = :c_id and
        p.iqp_ipc_state = 2 and
        p.iqp_date between :rep_date_begin and :rep_date_end
  into count_iqp_id, sum_iqp_amount;

  if (sum_iqp_amount is null and
      count_iqp_id is null) then
  begin
    err_text = ''; --f_lang(lang_code, 'За указанный перод платежей нет');
    suspend;
    exit;
  end
  suspend;
end
