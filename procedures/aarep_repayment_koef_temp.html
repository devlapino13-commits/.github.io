CREATE PROCEDURE AAREP_REPAYMENT_KOEF_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    REP_DISB DECIMAL(18,4),
    REP_ACTIVE_BEGIN DECIMAL(18,4),
    REP_ACTIVE_END DECIMAL(18,4),
    REP_PLAN_OS DECIMAL(18,4),
    REP_REPAY_ALL DECIMAL(18,4),
    REP_REPAY_NORMAL DECIMAL(18,4),
    REP_REPAY_ADV DECIMAL(18,4),
    REP_REPAY_ADV_A DECIMAL(18,4),
    REP_REPAY_ADV_B DECIMAL(18,4),
    REP_REPAY_ADV_C DECIMAL(18,4),
    REP_REPAY_DELINQ DECIMAL(18,4),
    REP_WRITE_OFF DECIMAL(18,4),
    REP_OKRUG VARCHAR(100),
    REP_OFFICE VARCHAR(100),
    REP_DISB_GR DECIMAL(18,4),
    REP_DISB_IND DECIMAL(18,4),
    REP_ACTIVE_BEGIN_GR DECIMAL(18,4),
    REP_ACTIVE_BEGIN_IND DECIMAL(18,4),
    REP_ACTIVE_END_GR DECIMAL(18,4),
    REP_ACTIVE_END_IND DECIMAL(18,4),
    REP_REPAY_ALL_GR DECIMAL(18,4),
    REP_REPAY_ALL_IND DECIMAL(18,4),
    REP_REPAY_NORMAL_GR DECIMAL(18,4),
    REP_REPAY_NORMAL_IND DECIMAL(18,4),
    REP_REPAY_ADV_GR DECIMAL(18,4),
    REP_REPAY_ADV_GR_A DECIMAL(18,4),
    REP_REPAY_ADV_GR_B DECIMAL(18,4),
    REP_REPAY_ADV_GR_C DECIMAL(18,4),
    REP_REPAY_ADV_IND DECIMAL(18,4),
    REP_REPAY_ADV_IND_A DECIMAL(18,4),
    REP_REPAY_ADV_IND_B DECIMAL(18,4),
    REP_REPAY_ADV_IND_C DECIMAL(18,4),
    REP_REPAY_DELINQ_GR DECIMAL(18,4),
    REP_REPAY_DELINQ_IND DECIMAL(18,4),
    REP_WRITE_OFF_GR DECIMAL(18,4),
    REP_WRITE_OFF_IND DECIMAL(18,4),
    REP_PLAN_OS_GR DECIMAL(18,4),
    REP_PLAN_OS_IND DECIMAL(18,4),
    PR_LOAN_TYPE INTEGER,
    REP_KC_FIO VARCHAR(100))
AS
declare variable ol_date d_date;
declare variable sh_date d_date;
declare variable rep_os d_currency;
declare variable ok_type d_integer;
declare variable n d_integer;
declare variable reg_id d_integer;
declare variable reg_id_prior d_integer;
declare variable k_id d_id;
declare variable cr_rate d_currency;
begin
  reg_id_prior = 0;
  rep_plan_os = 0;
  rep_active_end = 0;
  rep_active_begin = 0;
  rep_disb = 0;
  rep_os = 0;
  rep_repay_all = 0;
  rep_repay_normal = 0;
  rep_repay_adv = 0;
  rep_repay_delinq = 0;
  rep_write_off = 0;

  rep_plan_os_gr = 0;
  rep_active_end_gr = 0;
  rep_active_begin_gr = 0;
  rep_disb_gr = 0;
  rep_repay_all_gr = 0;
  rep_repay_normal_gr = 0;
  rep_repay_adv_gr = 0;
  rep_repay_adv_gr_a = 0;
  rep_repay_adv_gr_b = 0;
  rep_repay_adv_gr_c = 0;
  rep_repay_delinq_gr = 0;
  rep_write_off_gr = 0;

  rep_plan_os_ind = 0;
  rep_active_end_ind = 0;
  rep_active_begin_ind = 0;
  rep_disb_ind = 0;
  rep_repay_all_ind = 0;
  rep_repay_normal_ind = 0;
  rep_repay_adv_ind = 0;
  rep_repay_adv_ind_a = 0;
  rep_repay_adv_ind_b = 0;
  rep_repay_adv_ind_c = 0;
  rep_repay_delinq_ind = 0;
  rep_write_off_ind = 0;
  for select id_list
      from region_list(:report_region_id, 1, 3, 1)
      into :reg_id
  do
  begin
    select aa_curr_office_name, aa_curr_level3_name
    from aarep_get_office_categ(:reg_id, :report_date_end)
    into :rep_office, rep_okrug;
    suspend;
  end

  for select pr_is_indiv, aa_curr_level2_name, aa_curr_office_name, aa_curr_kc_fio, sum(aa_repay_normonth_os),
             sum(aa_amount_active_end), sum(aa_amount_active_begin), sum(aa_amount_disb)
      from aarep_analitica_port_temp(:report_region_id, :report_date_begin, :report_date_end, '', 0, 0, 1, 1, 0, 0, 0)
      group by pr_is_indiv, aa_curr_level2_name, aa_curr_office_name, aa_curr_kc_fio
      into :pr_loan_type, :rep_okrug, :rep_office, :rep_kc_fio, :rep_plan_os, :rep_active_end, :rep_active_begin,
           :rep_disb
  do
  begin
    if (pr_loan_type = 0) then
    begin
      rep_plan_os_ind = 0;
      rep_active_end_ind = 0;
      rep_active_begin_ind = 0;
      rep_disb_ind = 0;
      rep_plan_os_gr = rep_plan_os;
      rep_active_end_gr = rep_active_end;
      rep_active_begin_gr = rep_active_begin;
      rep_disb_gr = rep_disb;
    end
    else
    begin
      rep_plan_os_gr = 0;
      rep_active_end_gr = 0;
      rep_active_begin_gr = 0;
      rep_disb_gr = 0;
      rep_plan_os_ind = rep_plan_os;
      rep_active_end_ind = rep_active_end;
      rep_active_begin_ind = rep_active_begin;
      rep_disb_ind = rep_disb;
    end
    suspend;
  end

  pr_loan_type = 0;
  rep_plan_os = 0;
  rep_active_end = 0;
  rep_active_begin = 0;
  rep_disb = 0;
  rep_os = 0;
  rep_repay_all = 0;
  rep_repay_normal = 0;
  rep_repay_adv = 0;
  rep_repay_delinq = 0;
  rep_write_off = 0;

  rep_plan_os_gr = 0;
  rep_active_end_gr = 0;
  rep_active_begin_gr = 0;
  rep_disb_gr = 0;
  rep_repay_all_gr = 0;
  rep_repay_normal_gr = 0;
  rep_repay_adv_gr = 0;
  rep_repay_adv_gr_a = 0;
  rep_repay_adv_gr_b = 0;
  rep_repay_adv_gr_c = 0;
  rep_repay_delinq_gr = 0;
  rep_write_off_gr = 0;

  rep_plan_os_ind = 0;
  rep_active_end_ind = 0;
  rep_active_begin_ind = 0;
  rep_disb_ind = 0;
  rep_repay_all_ind = 0;
  rep_repay_normal_ind = 0;
  rep_repay_adv_ind = 0;
  rep_repay_adv_ind_a = 0;
  rep_repay_adv_ind_b = 0;
  rep_repay_adv_ind_c = 0;
  rep_repay_delinq_ind = 0;
  rep_write_off_ind = 0;
  k_id = 0;

  for select ok_ol_date, ok_ol_reg_id, ok_sh_date, ok_type, pr_loan_type, sum(-cr_rate * ok_os), k_id, cr_rate
      from operkred, region_list(:report_region_id, 1, -1, 1), opercurr, kontract, product
      where ok_ol_is_active = 1 and
            ok_ol_reg_id = id_list and
            ok_type >= 4 and
            ok_type <= 6 and
            k_id = ok_from_k_id and
            k_index_curr_code = cr_from_cu_id and
            ok_ol_date between cr_date and cr_date_end and
            ok_ol_date between :report_date_begin and :report_date_end and
            k_from_pr_id = pr_id and
            ok_from_c_id = 0
      group by ok_ol_date, ok_ol_reg_id, ok_sh_date, ok_type, pr_loan_type, k_id, cr_rate
      into ol_date, reg_id, sh_date, ok_type, pr_loan_type, rep_os, k_id, cr_rate
  do
  begin
    if (reg_id_prior <> reg_id) then
    begin
      select aa_curr_office_name, aa_curr_level3_name
      from aarep_get_office_categ(:reg_id, :report_date_end)
      into :rep_office, rep_okrug;
      reg_id_prior = :reg_id;
    end
    rep_repay_all = 0;
    rep_repay_normal = 0;
    rep_repay_adv = 0;
    rep_repay_adv_a = 0;
    rep_repay_adv_b = 0;
    rep_repay_adv_c = 0;
    rep_repay_delinq = 0;
    rep_write_off = 0;

    rep_repay_all_gr = 0;
    rep_repay_normal_gr = 0;
    rep_repay_adv_gr = 0;
    rep_repay_adv_gr_a = 0;
    rep_repay_adv_gr_b = 0;
    rep_repay_adv_gr_c = 0;
    rep_repay_delinq_gr = 0;
    rep_write_off_gr = 0;

    rep_repay_all_ind = 0;
    rep_repay_normal_ind = 0;
    rep_repay_adv_ind = 0;
    rep_repay_adv_ind_a = 0;
    rep_repay_adv_ind_b = 0;
    rep_repay_adv_ind_c = 0;
    rep_repay_delinq_ind = 0;
    rep_write_off_ind = 0;

    if (ok_type <> 4) then
      rep_write_off = :rep_os;
    else
    begin
      n = extract(year from ol_date) * 12 + extract(month from ol_date);
      n = n - extract(year from sh_date) * 12 - extract(month from sh_date);
      if (n = 0) then
        rep_repay_normal = :rep_os;
      else
      if (n < 0) then
        begin
          rep_repay_adv = :rep_os;

          select hist_new, hist_no, hist_still
          from aarep_kont_next_history(:k_id, :pr_loan_type, :report_date_begin, :report_date_end, :sh_date, :ol_date, :rep_os, :cr_rate)
          into rep_repay_adv_a, rep_repay_adv_b, rep_repay_adv_c;
        end
      else
        rep_repay_delinq = :rep_os;
      rep_repay_all = :rep_os;
    end
    if (pr_loan_type = 0) then
    begin
      rep_repay_all_gr = rep_repay_all;
      rep_repay_normal_gr = rep_repay_normal;
      rep_repay_adv_gr = rep_repay_adv;
      rep_repay_adv_gr_a = rep_repay_adv_a;
      rep_repay_adv_gr_b = rep_repay_adv_b;
      rep_repay_adv_gr_c = rep_repay_adv_c;
      rep_repay_delinq_gr = rep_repay_delinq;
      rep_write_off_gr = rep_write_off;
    end
    else
    begin
      rep_repay_all_ind = rep_repay_all;
      rep_repay_normal_ind = rep_repay_normal;
      rep_repay_adv_ind = rep_repay_adv;
      rep_repay_adv_ind_a = rep_repay_adv_a;
      rep_repay_adv_ind_b = rep_repay_adv_b;
      rep_repay_adv_ind_c = rep_repay_adv_c;
      rep_repay_delinq_ind = rep_repay_delinq;
      rep_write_off_ind = rep_write_off;
    end
    suspend;
  end
end
