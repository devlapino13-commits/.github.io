CREATE PROCEDURE HP_E_HR_ORDERS_ONE_SAVE(
    HO_ID D_ID,
    HO_DATE D_DATE,
    HO_FROM_REG_ID D_INTEGER,
    HO_TEXT D_STR_LONG,
    HO_REASON D_STR_255,
    HO_REASON_DATE D_DATE,
    HO_FROM_UN_ID D_ID,
    HO_ORDER_TYPE D_INTEGER,
    HO_ORDER_SUBTYPE D_INTEGER,
    HO_ORDER_NO D_STR_20,
    HO_SIGNER_ID D_INTEGER,
    HO_ACCOUNTANT_ID ID_TYPE,
    HO_HELP_AMOUNT D_INTEGER,
    HO_CERT_MARRIAGE D_BOOL,
    HO_CERT_BIRTH_CHILD D_BOOL,
    HO_CERT_BIRTH_EMP D_BOOL,
    HO_CERT_DEATH D_BOOL,
    HRV_NUMBER D_INTEGER,
    HRV_FROM_C_ID ID_TYPE,
    HRV_DATE_BEGIN D_DATE,
    HRV_DATE_END D_DATE,
    HRV_DAYS D_INTEGER,
    HRV_DAY_BACK D_DATE,
    HRV_DIRECTION D_STR_100,
    HRV_STATUS D_INTEGER,
    HRV_CHANGER_ID D_ID,
    HRV_NOT_CLOSE_ACCESS D_BOOL,
    IN_HRA_ID D_INTEGER)
AS
declare variable res_saldo d_currency;
begin
  if (coalesce(hrv_from_c_id, 0) = 0) then
    exception e_message 'Сотрудник не указан';

  select res_saldo
  from hp_e_hr_vocation_check_days(:hrv_from_c_id, :ho_order_type, :hrv_date_begin, :hrv_days)
  into :res_saldo;

  if (res_saldo < 0) then
    exception e_message 'После оформления данного приказа, остаток отпускных дней на конец года станет: ' || res_saldo;

  insert into hr_orders (
      ho_id,
      ho_date,
      ho_from_reg_id,
      ho_text,
      ho_reason,
      ho_reason_date,
      ho_from_un_id,
      ho_order_type,
      ho_order_subtype,
      ho_order_no,
      ho_signer_id,
      ho_accountant_id,
      ho_help_amount,
      ho_cert_marriage,
      ho_cert_birth_child,
      ho_cert_birth_emp,
      ho_cert_death)
  values (
      :ho_id,
      :ho_date,
      :ho_from_reg_id,
      :ho_text,
      :ho_reason,
      :ho_reason_date,
      :ho_from_un_id,
      :ho_order_type,
      :ho_order_subtype,
      :ho_order_no,
      :ho_signer_id,
      :ho_accountant_id,
      :ho_help_amount,
      :ho_cert_marriage,
      :ho_cert_birth_child,
      :ho_cert_birth_emp,
      :ho_cert_death)
  returning ho_id
  into ho_id;

  if (in_hra_id is not null) then
  begin
    update hr_applicat
    set hra_from_ho_id = :ho_id
    where hra_id = :in_hra_id;
  end

  hrv_number = 0;

  insert into hr_voctravel (hrv_from_ho_id, hrv_number, hrv_from_c_id, hrv_date_begin, hrv_date_end, hrv_days,
                            hrv_day_back, hrv_direction, hrv_status,
                            hrv_changer_id, hrv_not_close_access)
  values (:ho_id, :hrv_number, :hrv_from_c_id, :hrv_date_begin, :hrv_date_end, :hrv_days, :hrv_day_back,
         :hrv_direction, :hrv_status, :hrv_changer_id, :hrv_not_close_access);
end
