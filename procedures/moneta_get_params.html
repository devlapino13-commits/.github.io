Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE MONETA_GET_PARAMS(
    REP_DATE D_DATE = current_date)
RETURNS (
    ERROR_TXT D_STR_100,
    MONETA_COMMISSION D_CURRENCY,
    RUB_COURSE D_CURRENCY)
AS
begin
  error_txt = '';

  moneta_commission = 0; --f_get_const('moneta_commission');

  if (moneta_commission is null) then
    error_txt = 'Комиссия "Монеты" не определена';

  rub_course = (select cr_rate
                from opercurr oc
                where :rep_date between oc.cr_date and oc.cr_date_end
                and oc.cr_from_cu_id = 810);

  if (rub_course is null) then
    error_txt = 'Курс рубля не определен';

  suspend;
end
