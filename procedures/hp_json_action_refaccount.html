CREATE PROCEDURE HP_JSON_ACTION_REFACCOUNT(
    ID D_STR_CODE,
    IN_CLASS_NAME D_STR_CODE,
    IN_COMMAND D_STR_CODE,
    IN_JSON D_STR_8100,
    UN_ID D_ID)
RETURNS (
    RES_ID D_STR_CODE,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable ja_json d_str_8100;
declare variable ja_name d_str_255;
declare variable reg_id d_id;
declare variable ab_id d_id;
begin
  if (coalesce(id, '') = '') then
    id = '0';

  res_id = id;
  res_error_text = '';

  if (coalesce(in_class_name, '') = '') then
    exit;

  if (in_command = 'doLoad') then -- загрузка новой (пустой) записи
  begin
    select un_curr_reg_id
    from username
    where un_id = :un_id
    into reg_id;

    ja_name = 'vRefAccount';
    ja_json = f_json_make_pair('acc_from_reg_id', reg_id, 'num');
    ja_json = ja_json || ',' || f_json_make_pair('acc_date_open', current_date, 'date');

    ab_id = f_json_str_get_value(in_json, 'ab_id', null);
    if (coalesce(ab_id, 0) <> 0) then
    begin
      ja_json = ja_json || ',' || f_json_make_pair('acc_from_ab_id', ab_id, 'num');
      ja_json = ja_json || ',' || f_json_make_pair('acc_from_ab_id_name', f_objname('ref_accbalance', ab_id), 'str');
    end
    ja_json = ja_json || ',' || f_json_make_pair('acc_non_resident', 0, 'num');
    ja_json = ja_json || ',' || f_json_make_pair('acc_from_cu_id', f_get_const('cm_default_currency'), 'str');

    ja_json = '{' || ja_json || '}';
    insert into temp_json_action (ja_name, ja_json)
    values (:ja_name, :ja_json);

    suspend;
  end

  if (in_command = 'doSave') then
  begin
    ja_name = 'vRefAccount';
    for select ja_json
        from temp_json_action
        where ja_name = :ja_name
        into :ja_json
    do
    begin
      for select
          f_json_str_get_value(:ja_json, 'acc_from_ab_id',null) as acc_from_ab_id,
          f_json_str_get_value(:ja_json, 'acc_from_sab_id',null) as acc_from_sab_id,
          f_json_str_get_value(:ja_json, 'acc_from_reg_id',null) as acc_from_reg_id,
          f_json_str_get_value(:ja_json, 'acc_from_cu_id','') as acc_from_cu_id,
          f_json_str_get_value(:ja_json, 'acc_from_inx_id',0) as acc_from_inx_id,
          f_json_str_get_value(:ja_json, 'acc_object_kind',0) as acc_object_kind,
          f_json_str_get_value(:ja_json, 'acc_object_id',0) as acc_object_id,
          f_json_str_get_value(:ja_json, 'acc_code','') as acc_code,
          f_json_str_get_value(:ja_json, 'acc_number',0) as acc_number,
          f_json_str_get_value(:ja_json, 'acc_is_currency',0) as acc_is_currency,
          f_json_str_get_value(:ja_json, 'acc_name','') as acc_name,
          f_json_str_get_value(:ja_json, 'acc_memo','') as acc_memo,
          f_json_str_get_value(:ja_json, 'acc_type',0) as acc_type,
          f_json_str_get_value(:ja_json, 'acc_out_balance',0) as acc_out_balance,
          f_json_str_get_value(:ja_json, 'acc_date_open',null) as acc_date_open,
          f_json_str_get_value(:ja_json, 'acc_date_close',null) as acc_date_close,
          f_json_str_get_value(:ja_json, 'acc_non_resident',0) as acc_non_resident
      from rdb$database as cursor cur
      do
      begin
        res_error_text = '';
        if (coalesce(cur.acc_name, '') = '') then
          res_error_text = 'Название счета не заполнено';
        else
        if (coalesce(cur.acc_from_ab_id, 0) = 0) then
          res_error_text = 'Балансовый счет не указан';
        else
        if (coalesce(cur.acc_from_sab_id, 0) = 0) then
          res_error_text = 'Системный счет не указан';
        else
        if (coalesce(cur.acc_from_reg_id, 0) = 0) then
          res_error_text = 'Офис счета не указан';
        else
        if (coalesce(cur.acc_name, '') = '') then
          res_error_text = 'Название счета не указано';
        else
        begin
          update or insert into ref_account (acc_id, acc_from_ab_id, acc_from_sab_id, acc_from_reg_id,
                                             acc_from_cu_id, acc_from_inx_id, acc_object_kind, acc_object_id, acc_code,
                                             acc_number, acc_is_currency, acc_name, acc_memo, acc_type, acc_out_balance,
                                             acc_date_open, acc_date_close, acc_non_resident)
          values (iif(:id = 0, null, :id), cur.acc_from_ab_id, cur.acc_from_sab_id, cur.acc_from_reg_id, cur.acc_from_cu_id,
                  cur.acc_from_inx_id, cur.acc_object_kind, cur.acc_object_id, cur.acc_code, cur.acc_number, cur.acc_is_currency, cur.acc_name,
                  cur.acc_memo, cur.acc_type, cur.acc_out_balance, cur.acc_date_open, cur.acc_date_close, cur.acc_non_resident)
          matching (acc_id)
          returning acc_id
          into res_id;
        end

        if (res_error_text <> '') then
        begin
          suspend;
          exit;
        end
      end
    end
    suspend;
  end
end
