CREATE PROCEDURE CM_KG_0222_SQL(
    MA_ID D_INTEGER NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable C_FAMILY D_STR_100;
declare variable FAMILY_TUNDUK D_STR_100;
begin
  res_info_text = 'Добавьте фото документа , подтверждающего факт. семейный статус';
  res_error_is_critical = 1;
  for select mac.mac_from_c_id
      from miniappclient mac
      where mac.mac_from_ma_id = :ma_id and
      mac.mac_amount >= 100000
      into :c_id
  do
  begin
    select f_get_rd_code(c.c_family)
    from clients c
    where c.c_id = :c_id
    into c_family;

    if (c_family = 'family.married') then
    begin
      select res_str_values --
      from hp_objectprops_anketa_get('Clients', :c_id, -1, 'AnketaQuestions.Family_tunduk', current_date, null)
      into :family_tunduk;

      if (family_tunduk = 'Не заполнен' and
          not exists(select *
                     from clientdocs cds
                     inner join passkind pk on pk.pk_id = cds.cds_from_pk_id and
                           pk.pk_is_delete = 0
                     inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id and
                           pdt.pdt_code in ('certificate_marriage', 'certificate_annulment', 'certificate_death', 'power_of_attorney', 'judicial_act')
                     where cds.cds_from_c_id = :c_id and
                           cds.cds_is_delete = 0 and
                           current_date between cds.cds_doc_date_begin and coalesce(cds.cds_doc_date_end, '31.12.9999'))) then
        suspend;
    end
  end
end
