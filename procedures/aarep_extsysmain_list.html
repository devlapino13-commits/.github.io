CREATE PROCEDURE AAREP_EXTSYSMAIN_LIST(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_ONLY_PAY INTEGER)
RETURNS (
    ESM_ID DECIMAL(18,0),
    ESM_TRANS_ID VARCHAR(20),
    ESM_PROJECT VARCHAR(20),
    ESM_ACTION VARCHAR(20),
    ESM_DATE DATE,
    ESM_ACCOUNT VARCHAR(20),
    ESM_OBJECT_KIND INTEGER,
    ESM_OBJECT_ID DECIMAL(18,0),
    ESM_DATETIME TIMESTAMP,
    ESM_RESULT_ID INTEGER,
    ESM_STATUS INTEGER,
    ESM_EXT_DATETIME TIMESTAMP,
    ESM_AMOUNT DECIMAL(18,4),
    ESM_STATUS_STR VARCHAR(100),
    ESM_OBJECT_NAME VARCHAR(255),
    OL_ID DECIMAL(18,0),
    REG_NAME VARCHAR(100),
    REG_ID INTEGER,
    ESM_FROM_K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20))
AS
begin
  for select esm_id, esm_trans_id, lower(esm_project), esm_action, esm_date, esm_account, esm_object_kind, esm_object_id,
             esm_datetime, esm_result_id, esm_status, esm_ext_datetime, esm_amount, esm_from_k_id
      from extsysmain, extsyspro
      where esm_date between :rep_date_begin and :rep_date_end and
            esm_project = esp_project
      order by esm_id desc
      into :esm_id, :esm_trans_id, :esm_project, :esm_action, :esm_date, :esm_account, :esm_object_kind, :esm_object_id,
           :esm_datetime, :esm_result_id, :esm_status, :esm_ext_datetime, :esm_amount,
           :esm_from_k_id
  do
  begin
    k_kontract_no = '';

    esm_status_str = (case
                        when esm_status = -3 then 'Ошибка пр обработке'
                        when esm_status = -2 then 'Тестовый платеж'
                        when esm_status = -1 then 'Системная ошибка'
                        when esm_status = 0 then 'В процессе'
                        when esm_status = 1 then 'Обработан'
                      end);

    esm_object_name = '';
    ol_id = null;
    reg_name = '';
    reg_id = 0;
    if ((rep_only_pay = 1 and
        esm_action = 'pay') or (rep_only_pay <> 1)) then
    begin
      if (esm_object_kind = 0) then
      begin
        select c_name_fio
        from clients
        where c_id = :esm_object_id
        into :esm_object_name;
      end
      if (esm_action = 'pay') then
      begin
        select eso_from_ol_id, reg_unicode, reg_code || ' ' || reg_name
        from extsysoper, operlist, region
        where eso_from_esm_id = :esm_id and
              eso_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              ol_from_reg_id = reg_unicode
        into :ol_id, :reg_id, :reg_name;
      end

      if (esm_from_k_id is not null) then
      begin
        select k_kontract_no
        from kontract
        where k_id = :esm_from_k_id
        into :k_kontract_no;
      end
      suspend;
    end
  end
end
