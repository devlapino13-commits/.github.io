declare function is_cyr_more (
    str d_str_long)
returns boolean
as
declare variable cyr_str d_str_long;
declare variable lat_str d_str_long;
begin
  cyr_str = str;
  lat_str = str;

  select out_string
  from hp_check_characters_str(:cyr_str, '[а-яА-ЯЁё0-9@- ]*')
  into cyr_str;

  select out_string
  from hp_check_characters_str(:lat_str, '[ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0-9@- ]*')
  into lat_str;

  return char_length(trim(cyr_str)) > char_length(trim(lat_str));
end

declare function del_twice_space (
    str d_str_255)
returns d_str_255
as
begin
  str = replace(str, ascii_char(9), ' ');
  str = replace(str, '\', '');
  str = replace(str, 'ʻ', '');
  str = f_replace_all(str, ' -', '-');
  str = f_replace_all(str, '- ', '-');
  str = f_replace_all(str, '  ', ' ');

  return trim(str);
end

declare procedure hp_my_get_passport_data (
    pk_id d_integer,
    country_citizen d_str_255,
    doc_number_in d_str_20,
    doc_who_in d_str_255,
    doc_department_code_in d_str_255)
returns (
    doc_country d_str_10,
    doc_country_text d_str_255,
    doc_type d_str_10,
    doc_type_text d_str_255,
    doc_serial d_str_10,
    doc_number d_str_20,
    doc_who d_str_255,
    doc_department_code d_str_255)
as
declare variable serial_length d_integer;
begin
  doc_number = doc_number_in;
  doc_who = upper(trim(doc_who_in));
  doc_department_code = doc_department_code_in;

  doc_type_text = null;

  select rd_memo
  from refdata
  where rd_main_code = 'CountryOKSM' and
        (rd_name = case :country_citizen
                     when 'Кыргызстан' then 'Киргизия'
                     when 'Республика Молдова' then 'Молдова, Республика'
                     else :country_citizen
                   end)
  into doc_country;

  if (doc_country is null) then
  begin
    doc_country = '999';
    doc_country_text = upper(country_citizen);
  end

  doc_type = case pk_id
               when 12 then '21' -- Паспорт гражданина РФ
               when 13 then '22.1' -- Загран. паспорт РФ
               else '31' -- Паспорт, выданный иностранным государством и признаваемый в Российской Федерации
             end;

  ---- серия и номер документа -----
  doc_number = replace(doc_number, ' ', '');
  doc_number = replace(doc_number, '-', '');

  serial_length = case pk_id
                    when 12 then 4 -- Паспорт гражданина РФ
                    when 13 then 2 -- Загран. паспорт РФ
                    when 17 then 2 -- Паспорт УЗ (зелёный)
                    when 330 then 2 -- Биометрический паспорт УЗ (бордовый)
                    when 15 then 1 -- Паспорт Тадж.
                    else 0
                  end;

  if (char_length(doc_number) > serial_length and
      serial_length > 0) then
  begin
    doc_serial = upper(left(doc_number, serial_length));
    doc_number = upper(substring(doc_number from serial_length + 1));
  end

  select trim(out_string)
  from hp_check_characters_str(:doc_who, '[a-zA-Zа-яА-ЯЁё0-9@- .,'';\/№():_+|"#?!;]*')
  into doc_who;

  if (coalesce(doc_who, '') = '') then
    doc_who = '-';

  if (pk_id = 12) then
  begin
    doc_department_code = replace(doc_department_code, ' ', ''); -- Код подразделения

    if (doc_department_code not similar to '[0-9]{3}\-[0-9]{3}' escape '\') then
    begin -- если Код подразделения не подходит по маске, то попытаемся его привести к нужному формату
      select out_string
      from hp_check_characters_str(:doc_department_code, '[0-9]*')
      into doc_department_code;

      if (char_length(doc_department_code) = 6) then -- если цифр достаточно
        doc_department_code = left(doc_department_code, 3) || '-' || substring(doc_department_code from 4);
      else
        doc_department_code = trim(iif(current_date < '18.02.2024', '999-999', null)); -- заглушка
    end
  end
  else
    doc_department_code = null;

  suspend;
end

begin
  if (bkt_format is null) then
    bkt_format = 0;

  if (xml_version_id is null) then
    select bxv_id
    from bki_xml_version
    where bxv_date_begin <= current_date
    order by bxv_date_begin desc
    rows 1

                   and bkp_id = bkl_from_bkp_id
                   and bkc_id = bod_from_bkc_id

                   and bod_object_kind = 0

  -- смена ДУЛ

  for select cds_id, cds_doc_no, cds_doc_date_begin, cds_doc_date_end, del_twice_space(cds_doc_issued_by), cds_doc_issued_code
  from hp_clientdocs_pass_list(:c_id, 1, current_date, '01.07.2014')
  where cds_doc_no <> :doc_number

                    where bkc_id = bod_from_bkc_id
                      and bkl_id = bkc_from_bkl_id
                      and bkp_id = bkl_from_bkp_id
                      and bkc_id = bod_from_bkc_id
                      and bod_kind = 1
                      and bod_object_kind = 0
                      and bod_object_id = :c_id
                      and bod_from_cds_id = cds_id
                      and bkp_from_bkk_id = :bkk_id)
  order by cds_id desc
  into hist_cds_id, hist_doc_number, hist_doc_date, hist_doc_end_date, hist_doc_who, hist_doc_department_code
  do
  begin
    if (hist_doc_end_date not between hist_doc_date and '31.12.2199') then
      hist_doc_end_date = null;

    if (hist_cds_id is not null) then
    begin
      select cds_from_pk_id
      from clientdocs cds
      where cds_id = :hist_cds_id
      into :old_pk_id;
    
      select pk_country_id
      from passkind

      hist_doc_number = f_my_fix_wrong_lat(hist_doc_number);
      old_country_citizen = f_my_fix_wrong_cyr((select rd_name from refdata where rd_id = :hist_citizen));

      hist_doc_who = del_twice_space(hist_doc_who);
      hist_doc_who = iif(cast(is_cyr_more(hist_doc_who) as boolean), cast(f_my_fix_wrong_cyr(hist_doc_who) as d_str_255), cast(f_my_fix_wrong_lat(hist_doc_who) as d_str_255));
    
      if (hist_doc_number = doc_number) then
       continue;

      my_obj_set_s(pers_change_jso, 'ch_title', 1);

      hist_jso = json.obj_create();
      json.arr_set_obj(hist_jsa, -1, hist_jso);
      my_obj_set_s(hist_jso, 'hist_name_sign', 0);
      my_obj_set_s(hist_jso, 'hist_doc_sign', 1);
      my_obj_set_s(hist_jso, 'hist_cds_id', hist_cds_id);

      my_obj_set_s(hist_jso, 'hist_doc_country', hist_doc_country);
      my_obj_set_s(hist_jso, 'hist_doc_country_text', hist_doc_country_text);
      my_obj_set_s(hist_jso, 'hist_doc_type', hist_doc_type);
      my_obj_set_s(hist_jso, 'hist_doc_type_text', hist_doc_type_text);
      my_obj_set_s(hist_jso, 'hist_doc_serial', hist_doc_serial);
      my_obj_set_s(hist_jso, 'hist_doc_number', hist_doc_number);
      my_obj_set_s(hist_jso, 'hist_doc_date', hist_doc_date);
      my_obj_set_s(hist_jso, 'hist_doc_who', hist_doc_who);
      my_obj_set_s(hist_jso, 'hist_doc_department_code', hist_doc_department_code);
      my_obj_set_s(hist_jso, 'hist_doc_end_date', hist_doc_end_date);

      if (hist_citizen <> c_citizen) then
      begin
        hist_fio = iif(hist_citizen = 13, json.obj_get_val(pers_change_jso, 'c_name_fio', c_name_fio),
                                          json.obj_get_val(pers_change_jso, 'c_name_eng', c_name_eng));

        if (curr_fio <> hist_fio) then
        begin
          select del_twice_space(name_last), del_twice_space(name_first), del_twice_space(name_middle)
          from hp_fio_clear_ru(:hist_fio)
          into hist_name_last, hist_name_first, hist_name_middle;

          my_obj_set_s(hist_jso, 'hist_name_sign', 1);
          my_obj_set_s(hist_jso, 'hist_name_last', hist_name_last);
          my_obj_set_s(hist_jso, 'hist_name_first', hist_name_first);
          my_obj_set_s(hist_jso, 'hist_name_middle', hist_name_middle);
          my_obj_set_s(hist_jso, 'hist_name_doc_date', doc_date);

    end
  end

  if (json.count_get(hist_jsa) = 0) then
  begin
    hist_jso = json.obj_create();
    json.arr_set_obj(hist_jsa, -1, hist_jso);
    my_obj_set_s(hist_jso, 'hist_doc_sign', 0);
    my_obj_set_s(hist_jso, 'hist_name_sign', 0);
  end
  ----------------------------------------------

  select rd_memo from refdata
  where rd_main_code = 'CountryOKSM'
    and (rd_name = case f_my_fix_wrong_cyr(:c_birth_country)
                      when 'Кыргызстан' then 'Киргизия'
                      when 'Кыргызская Республика' then 'Киргизия'
                      when 'Республика Кыргызстан' then 'Киргизия'
                      when 'Убекистан' then 'Узбекистан'
                      when 'Республика Узбекистан' then 'Узбекистан'
                      when 'Республика Таджикистан' then 'Таджикистан'
                      else f_my_fix_wrong_cyr(:c_birth_country)
                   end)
  into birth_country;

  birth_country = coalesce(birth_country, '999');

-- RFCRU-3680
--  birth_place = upper(iif(cast(is_cyr_more(birth_place) as boolean), cast(f_my_fix_wrong_cyr(birth_place) as d_str_255), cast(f_my_fix_wrong_lat(birth_place) as d_str_255)));
--
--  select del_twice_space(out_string)
--  from hp_check_characters_str(:birth_place, '[a-zA-Zа-яА-ЯЁё0-9@- .,'';\/№():_+|"#?!;]*')
--  into birth_place;
--
--  if (birth_place is null) then
--    birth_place = '-';

  my_obj_set_s(jso, 'birth_date', birth_date);
  my_obj_set_s(jso, 'birth_country', birth_country);
--  my_obj_set_s(jso, 'birth_place', birth_place);

  select inn_received, inn
  from inn_query_status(:c_id)
  into inn_received, inn_no;

  if (inn_received = 1) then
  begin
    my_obj_set_s(jso, 'inn_code', 1);
    my_obj_set_s(jso, 'inn_no', inn_no);
    my_obj_set_s(jso, 'inn_checked', 1);
    my_obj_set_s(jso, 'special_tax_mode', iif(json.obj_get_val(jso, 'deal_sign_credit', 1) = 0, 1, 0));
  end

  if (c_citizen = 13) then -- если РФ
  begin
    select cds_doc_no
    from clientdocs, passkind, passdoctype
    where cds_from_c_id = :c_id
      and cds_from_pk_id = pk_id
      and pk_from_pdt_id = pdt_id
      and pdt_code = 'snils'
      and cds_is_delete = 0
    order by cds_from_c_id desc, cds_doc_date_begin desc
    rows 1
    into snils_no;

    select out_string
    from hp_check_characters_str(:snils_no, '[0-9]+')
    into snils_no;

    if (char_length(coalesce(snils_no, '')) <> 11) then
      snils_no = null;
    else
    begin
      i = 1;
      c1 = 0;
      while (i < 10) do
      begin
        n = substring(snils_no from i for 1);
        c1 = c1 + n * (10 - i);
        i = i + 1;
      end

      if (lpad(right(mod(c1, 101), 2), 2, '0') = substring(snils_no from 10 for 2)) then
        my_obj_set_s(jso, 'snils_no', snils_no);
    end
  end

  a_fias = null;       -- Номер адреса в ФИАС
  a_okato = null;      -- Код населенного пункта по ОКАТО
  a_other_statement = null; --  Иной населенный пункт
  a_index = null;      -- Почтовый индекс
  a_street = null;     -- Улица
  a_house = null;      -- Дом
  a_domain = null;     -- Владение
  a_block = null;      -- Корпус
  a_build = null;      -- Строение
  a_apartment = null;  -- Квартира

  for select adc_address_type, replace(upper(adl_flat_no), ' ', ''), replace(upper(adl_depart_no), ' ', ''),
          replace(upper(adl_section_no), ' ', ''), replace(upper(adl_building_no), ' ', ''), replace(upper(adl_vladen_no), ' ', ''), adl_street_id,
          adl_from_pl_id, adl_post_index,
          substring(cast(adl_id as str_255) from 1 for 3),
          add_registration_date
  from addrclient
  inner join addrlist on adl_id = adc_from_adl_id
  left join addrdetails on add_adl_id = adc_from_adl_id
  where adc_from_c_id = :c_id
    and adc_address_type in (1, 2)
  order by adc_address_type
  into adc_type, a_apartment, a_house,
       a_block, a_build, a_domain, adl_street_id,
       adl_from_pl_id, a_index, republ_code,
       ar_reg_date
  do
  begin
    if (a_apartment = '0') then a_apartment = null;
    if (a_house = '0') then a_house = null;
    if (a_block = '0') then a_block = null;
    if (a_build = '0') then a_build = null;
    if (a_domain = '0') then a_domain = null;

    a_country = (select rd2.rd_memo
                from refdata rd1, refdata rd2
                where rd1.rd_str_id = :republ_code
                  and rd1.rd_main_code = 'Citizen'
                  and rd2.rd_main_code = 'CountryOKSM'
                  and rd2.rd_name = case rd1.rd_name
                                        when 'Кыргызстан' then 'Киргизия'
                                        when 'Республика Молдова' then 'Молдова, Республика'
                                        else rd1.rd_name
                                     end);

    a_country_text = '';
    if (coalesce(a_country, '') = '') then
    begin
      a_country = '999';
      a_country_text = (select upper(rd_name)
                        from refdata rd
                        where rd_str_id = :republ_code
                          and rd_main_code = 'Citizen');
  
      if (coalesce(a_country_text, '') = '') then
        a_country_text = '-';
     end 

    select upper(pl.pl_name), trim(pl.pl_post_index), trim(pl.pl_add_code)
    from places pl
    where pl.pl_id = :adl_from_pl_id
    into a_other_statement, a_index, a_okato;
  
    if (a_index not similar to '[0-9]{6}') then
      a_index = null;
  
    if (coalesce(a_okato, '') = '') then
      a_okato = '99999999999';
  
    if (a_okato <> '99999999999') then
      a_other_statement = null;

    select del_twice_space(upper(pl.pl_name)),
      trim(iif(trim(coalesce(pl_post_index, '')) not similar to '[0-9]{6}', :a_index, pl_post_index))  -- по умолчанию наибольший приоритет отдаю почтовому индексу, указанному на уровне улицы
    from places pl
    inner join placetype pt on pl.pl_status = pt.plt_id
    where pl.pl_id = :adl_street_id
    into a_street, a_index;
  
    select del_twice_space(out_string)
    from hp_check_characters_str(:a_street, '[а-яА-ЯЁё0-9- ]*')
    into a_street;

    if (adc_type = 1) then -- место регистрации
    begin
--      if (ar_reg_date is null) then -- дата регистрации. Если не нашли по новому, поищем по старому
      ar_reg_date = null; -- временно, пока Илья разбирается с письмом ЦБ (RFCRU-5035)
      a_reg_code = 1;
      begin
        select cds_doc_date_begin, cds_id
        from clientdocs cds, passkind pk, passdoctype pdt
        where pk_id = cds_from_pk_id
          and pdt_id = pk_from_pdt_id
          and cds_from_c_id = :c_id
          and cds_doc_date_begin <= :rep_date
          and coalesce(cds_delete_datetime, current_timestamp) >= cast(:rep_date as date)  -- Не аннулирован
          and cds_verify_status = 2  -- подтвержденный
          and pdt.pdt_code in ('registration_permanent', 'registration_fiz', 'registration_yur')
        order by cds_create_datetime desc
        rows 1
        into ar_reg_date, cds_id;

        select rd_tag
        from addresstype_registration
        inner join refdata rd on rd.rd_id = atr_from_rd_id
        where atr_from_cds_id = :cds_id
        order by atr_datetime desc
        rows 1
        into a_reg_code;
      end

      my_obj_set_s(jso, 'ar_reg_code', a_reg_code);
      my_obj_set_s(jso, 'ar_index', a_index);
      my_obj_set_s(jso, 'ar_country', a_country);
      my_obj_set_s(jso, 'ar_country_text', a_country_text);
      my_obj_set_s(jso, 'ar_fias', a_fias);
      my_obj_set_s(jso, 'ar_okato', a_okato);
      my_obj_set_s(jso, 'ar_other_statement', a_other_statement);
      my_obj_set_s(jso, 'ar_street', a_street);
      my_obj_set_s(jso, 'ar_house', a_house);
      my_obj_set_s(jso, 'ar_domain', a_domain);
      my_obj_set_s(jso, 'ar_block', a_block);
      my_obj_set_s(jso, 'ar_build', a_build);
      my_obj_set_s(jso, 'ar_apartment', a_apartment);
--      my_obj_set_s(jso, 'ar_reg_department_code', ar_reg_department_code);  временно, пока Илья разбирается с письмом ЦБ (RFCRU-5035)
--      my_obj_set_s(jso, 'ar_reg_place', ar_reg_place);   временно, пока Илья разбирается с письмом ЦБ (RFCRU-5035)

      if (ar_reg_date > coalesce(birth_date, '01.01.1900')) then
        my_obj_set_s(jso, 'ar_reg_date', ar_reg_date);
    end
    else
    begin
      -- проверка на совпадение
      if (coalesce(a_index, '')             = coalesce(json.obj_get_val(jso, 'ar_index'), '')
        and coalesce(a_country, '')         = coalesce(json.obj_get_val(jso, 'ar_country'), '')
        and coalesce(a_country_text, '')    = coalesce(json.obj_get_val(jso, 'ar_country_text'), '')
        and coalesce(a_fias, '')            = coalesce(json.obj_get_val(jso, 'ar_fias'), '')
        and coalesce(a_okato, '')           = coalesce(json.obj_get_val(jso, 'ar_okato'), '')
        and coalesce(a_other_statement, '') = coalesce(json.obj_get_val(jso, 'ar_other_statement'), '')
        and coalesce(a_street, '')          = coalesce(json.obj_get_val(jso, 'ar_street'), '')
        and coalesce(a_house, '')           = coalesce(json.obj_get_val(jso, 'ar_house'), '')
        and coalesce(a_domain, '')          = coalesce(json.obj_get_val(jso, 'ar_domain'), '')
        and coalesce(a_block, '')           = coalesce(json.obj_get_val(jso, 'ar_block'), '')
        and coalesce(a_build, '')           = coalesce(json.obj_get_val(jso, 'ar_build'), '')
        and coalesce(a_apartment, '')       = coalesce(json.obj_get_val(jso, 'ar_apartment'), ''))
      then
        my_obj_set_s(jso, 'af_sign', 0);
      else
      begin
        my_obj_set_s(jso, 'af_sign', 1);
        my_obj_set_s(jso, 'af_index', a_index);
        my_obj_set_s(jso, 'af_country', a_country);
        my_obj_set_s(jso, 'af_country_text', a_country_text);
        my_obj_set_s(jso, 'af_fias', a_fias);
        my_obj_set_s(jso, 'af_okato', a_okato);
        my_obj_set_s(jso, 'af_other_statement', a_other_statement);
        my_obj_set_s(jso, 'af_street', a_street);
        my_obj_set_s(jso, 'af_house', a_house);
        my_obj_set_s(jso, 'af_domain', a_domain);
        my_obj_set_s(jso, 'af_block', a_block);
        my_obj_set_s(jso, 'af_build', a_build);
        my_obj_set_s(jso, 'af_apartment', a_apartment);
      end

  contacts_phone_number = f_cont_info(0, c_id, 'cell', 'ok');

  select out_string
  from hp_check_characters_str(:contacts_phone_number, '[0-9]+')
  into contacts_phone_number;

  if (coalesce(contacts_phone_number, '') <> '') then
    my_obj_set_s(jso, 'contacts_phone_number', contacts_phone_number);
  else
  begin
    -- если на момент формирования события нет верифицированного номера, то подтягиваем адрес электронной почты
    contacts_email = f_cont_info('clients', c_id, 'email', 'ok');

    if (coalesce(contacts_email, '') <> '') then
      my_obj_set_s(jso, 'contacts_email', contacts_email);
    else
    begin
      -- если адреса электронной почты нет, то последний верифицированный номер
      contacts_phone_number = f_cont_info(0, c_id, 'cell', 'last');

      select out_string
      from hp_check_characters_str(:contacts_phone_number, '[0-9]+')
      into contacts_phone_number;
    
      if (coalesce(contacts_phone_number, '') <> '') then
        my_obj_set_s(jso, 'contacts_phone_number', contacts_phone_number);

    my_obj_set_s(jso, 'ogrn', :ogrn);  
    my_obj_set_s(jso, 'reg_date', :reg_date);
    my_obj_set_s(jso, 'special_tax_mode', 0);

    where is_judicial = 0
    into bankruptcy_publ_date, bankruptcy_link, bankruptcy_completion_date, bankruptcy_num, bankruptcy_stop_date;

      my_obj_set_s(jso, 'bankruptcy_num', coalesce(bankruptcy_num, '-'));
      my_obj_set_s(jso, 'bankruptcy_publ_date', bankruptcy_publ_date);

        my_obj_set_s(jso, 'bankruptcy_stage', 6);
        my_obj_set_s(jso, 'bankruptcy_end_fact', 0);
      end
      else if (bankruptcy_completion_date is not null) then
      begin
        my_obj_set_s(jso, 'bankruptcy_stage', 7);
        my_obj_set_s(jso, 'bankruptcy_end_fact', 1);
        my_obj_set_s(jso, 'settlement_exist', 1);
        my_obj_set_s(jso, 'bankruptcy_completion_date', bankruptcy_completion_date);
      end
      else if (bankruptcy_stop_date is not null) then
      begin
        my_obj_set_s(jso, 'bankruptcy_stage', 8);
        my_obj_set_s(jso, 'bankruptcy_end_fact', 0);
        my_obj_set_s(jso, 'settlement_exist', 0);
      end
    end
  end
