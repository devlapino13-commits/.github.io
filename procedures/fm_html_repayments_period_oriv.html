CREATE PROCEDURE FM_HTML_REPAYMENTS_PERIOD_ORIV(
    UN_ID DECIMAL(18,0),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    IS_KURATOR_ORIV D_SMALLINT)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable LO_ID decimal(18,0);
declare variable DB_ID integer;
declare variable K_ID D_ID;
declare variable TITLE D_STR_100;
declare variable TITLE_ADD D_STR_20;
declare variable IS_KURATOR_SB D_SMALLINT;
declare variable K_KONTRACT_NO D_STR_20;
declare variable REPAY_SUM decimal(18,2);
declare variable OL_DATE date;
declare variable C_ID D_ID;
declare variable REPAY_ONE_ORIV decimal(18,2);
declare variable REPAY_SUM_ALL decimal(18,2);
declare variable C_NAME_FIO D_STR_100;
declare variable SBL_ID D_ID;
declare variable REPAY_ACTIVE CURR_TYPE;
declare variable REPAY_CLOSE CURR_TYPE;
declare variable RN_NAME varchar(30);
declare variable SBL_NAME STR_100;
begin

  --сводный отчет
  db_id = f_get_db_id();
  un_id = (select un_id
           from hp_test_un_id(:un_id));

  select un_from_c_id
  from username
  where un_id = :un_id
  into :lo_id; --id sotrudnika
  if (lo_id is null) then
    exit;

  if (not exists(select *
                 from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.ViewAccessOriv')
                 where has_access = 1)) then
    exit; --если это не специалист отдела взыскания

  is_kurator_sb = 0;
  if (exists(select *
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.ViewAccessToAll')
             where has_access = 1)) then
    is_kurator_sb = 1;

  res_text = (select res_style
              from fm_html_styles) || '<body>';

  title = iif(db_id <> 94401, 'Погашения за период', 'Dövr üçün ödənişlər');
  title_add = iif(db_id <> 94401, ' (сводный)', ' birləşdirilmişdir');

  title = iif(is_kurator_sb = 1, title || title_add, title);

  res_text = BLOB_APPEND(res_text, f_format('<h1 class="green_text">%1</h1>', title));
  res_text = BLOB_APPEND(res_text, '<br /> дата начала ' || f_datetostr(rep_date_begin) || '<br /> дата конца ' || f_datetostr(rep_date_end));
  if (is_kurator_sb = 0) then
  begin
    repay_sum_all = 0;
    res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="30%">
       <col width="40%">
       <col width="30%">
       <tr>' || f_format('<th>%1</th><th>%2</th><th>%3</th></tr>', iif(db_id <> 94401, 'Дата', 'Tarix'), iif(db_id <> 94401, 'Номер договора', 'Müqavilənin sayı'), iif(db_id <> 94401, 'Сумма платежа', 'Ödəniş məbləği')));

    for select distinct k.k_id, k.k_kontract_no, sl.sbl_id
        from sb_akt sa
        inner join sb_list sl on sl.sbl_id = sa.sba_from_sbl_id
        inner join kontract k on k.k_id = sa.sba_from_k_id
        where sl.sbl_from_c_id = :lo_id and
              (sba_date_end = '31.12.9999' or
              iif(sba_date_end = '31.12.9999', current_date, sba_date_end) between :rep_date_begin and :rep_date_end) and
              sa.sba_is_delete = 0 and
              sba_status = 2 and
              sbl_type = 1

        into :k_id, k_kontract_no, sbl_id
    do
    begin
      repay_sum = 0;
      for select coalesce(-sum(ok.ok_os + ok.ok_perc + ok.ok_fine), 0) as repay_sum, ok.ok_ol_date as ol_date
          from operkred ok
          inner join sb_akt sba on sba.sba_from_k_id = ok.ok_from_k_id
          where ok.ok_from_k_id = :k_id and
                sba.sba_from_sbl_id = :sbl_id and
                ok.ok_type in (4, 9) and
                ok.ok_ol_is_active = 1 and
                ok.ok_ol_date between :rep_date_begin and :rep_date_end and
                ok.ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                ok_from_c_id = 0
          group by ok_ol_date
          having sum(ok_os + ok_perc + ok_fine) <> 0
          union all
          select coalesce(-sum(oko_os + oko_perc), 0) as repay_sum, okf.oko_ol_date as ol_date
          from operkroff okf
          inner join sb_akt sba on sba.sba_from_k_id = okf.oko_from_k_id
          where okf.oko_from_k_id = :k_id and
                sba.sba_from_sbl_id = :sbl_id and
                oko_type = 4 and
                oko_ol_is_active = 1 and
                oko_ol_date between :rep_date_begin and :rep_date_end and
                oko_ol_date between sba.sba_date_begin and sba.sba_date_end
                -- :rep_date_end between sba_date_begin and sba_date_end and
          group by oko_ol_date
          having sum(oko_os + oko_perc) <> 0
          into :repay_sum, ol_date
      do
      begin
        if (repay_sum > 0) then
        begin
          repay_sum_all = repay_sum_all + repay_sum;
          repay_sum = round(repay_sum, 0);
          res_text = BLOB_APPEND(res_text, f_format('<tr><td>%1</td><td align="center">%2</td><td>%3</td></tr>', f_datetostr(ol_date) || '</a>', k_kontract_no || '</a>', repay_sum));
        end
      end

    end
    res_text = BLOB_APPEND(res_text, f_format('<tr><td><b>%1</b></td><td align="center">%2</td><td><b>%3</b></td></tr>', 'Итого:' || '</a>', '' || '</a>', repay_sum_all));
  end

  if (is_kurator_sb = 1 and
      (is_kurator_oriv is null)) then
  begin
    rn_name = '';
    select rn_name
    from permissions, rolename
    where pm_object_kind = (select ukt_id
                            from hp_get_uniqtable_id('Username')) and
          pm_object_id = :un_id and
          pm_subject_kind = (select ukt_id
                             from hp_get_uniqtable_id('Rolename')) and
          pm_is_delete = 0 and
          current_date between coalesce(pm_date_begin, '01.01.2000') and coalesce(pm_date_end, '31.12.2999') and
          pm_subject_id = rn_id and
          rn_name = 'sb_kg'
    into rn_name;
    rn_name = coalesce(rn_name, '');

    res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="70%">
       <col width="30%">
       <tr>' || f_format('<th>%1</th><th>%2</th></tr>', iif(db_id <> 94401, 'ОРИВ', 'ORIV'), iif(db_id <> 94401, 'Сумма за период', 'Dövr üçün məbləğ')));
    repay_sum_all = 0;
    for select sb.sbl_from_c_id, cl.c_name_fio, sb.sbl_id, left(sbl_name, 100)
        from sb_list sb
        inner join clients cl on cl.c_id = sb.sbl_from_c_id
        where current_date between sb.sbl_date_begin and coalesce(sb.sbl_date_end, '31.12.9999') and
              sbl_type = 1 and
              (cast(:rn_name as varchar(30)) = '' or
              (cast(:rn_name as varchar(30)) = 'sb_kg' and
              sbl_name starting 'kg ')) 
        order by sbl_name
        into :c_id, c_name_fio, sbl_id, sbl_name
    do
    begin
      if (:db_id in (39801, 41701) and
          sbl_id = 55) then
        continue;

      if (rn_name = 'sb_kg') then
        c_name_fio = sbl_name;
      repay_one_oriv = 0;
      repay_active = 0;
      repay_close = 0;

      select coalesce(-sum(ok_os + ok_perc), 0)
      from sb_akt, operkred
      where sba_is_delete = 0 and
            sba_status = 2 and
            sba_from_sbl_id = :sbl_id and
            sba_from_k_id = ok_from_k_id and
            sba_from_sbl_id > 0 and
            ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            --            sba_date_begin <= :rep_date_end and
            --            sba_date_end >= :rep_date_begin and
            ok_ol_date between sba_date_begin and sba_date_end and
            --:rep_date_end between sba_date_begin and sba_date_end and
            ok_from_c_id = 0 and
            ok_type in (4, 9)
      into :repay_active;

      select coalesce(-sum(oko_os + oko_perc), 0)
      from sb_akt, operkroff
      where sba_is_delete = 0 and
            sba_status = 2 and
            sba_from_sbl_id = :sbl_id and
            sba_from_k_id = oko_from_k_id and
            sba_from_sbl_id > 0 and
            oko_ol_is_active = 1 and
            oko_ol_date between :rep_date_begin and :rep_date_end and
            --            :rep_date_end between sba_date_begin and sba_date_end and  -- заменил на ниже следующую от 14.4.21 Улан
            oko_ol_date between sba_date_begin and sba_date_end and
            oko_type = 4
      into :repay_close;

      repay_one_oriv = repay_active + repay_close;
      repay_sum_all = repay_sum_all + repay_one_oriv;

      if (repay_one_oriv > 0) then
        res_text = BLOB_APPEND(res_text, f_format('<tr><td>%1</td><td align="center">%2</td></tr>', c_name_fio || '</a>', :repay_one_oriv));
    end
    if (is_kurator_sb = 1) then
      res_text = BLOB_APPEND(res_text, f_format('<tr><td><b>%1</b></td><td align="center"><b>%2</b></td></tr>', 'Итого:' || '</a>', :repay_sum_all));
  end

  if (is_kurator_oriv = 1) then
  begin
    repay_sum_all = 0;
    res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="30%">
       <col width="40%">
       <col width="30%">
       <tr>' || f_format('<th>%1</th><th>%2</th><th>%3</th></tr>', iif(db_id <> 94401, 'Дата', 'Tarix'), iif(db_id <> 94401, 'Номер договора', 'Müqavilənin sayı'), iif(db_id <> 94401, 'Сумма платежа', 'Ödəniş məbləği')));

    for select distinct k.k_id, k.k_kontract_no, sl.sbl_id
        from sb_akt sa
        inner join sb_list sl on sl.sbl_id = sa.sba_from_sbl_id
        inner join kontract k on k.k_id = sa.sba_from_k_id
        where sl.sbl_from_c_id = :lo_id and
              (sba_date_end = '31.12.9999' or
              iif(sba_date_end = '31.12.9999', current_date, sba_date_end) between :rep_date_begin and :rep_date_end) and
              sa.sba_is_delete = 0 and
              sba_status = 2 and
              sbl_type = 1
        into :k_id, k_kontract_no, sbl_id
    do
    begin
      repay_sum = 0;
      for select coalesce(-sum(ok.ok_os + ok.ok_perc + ok.ok_fine), 0) as repay_sum, ok.ok_ol_date as ol_date
          from operkred ok
          inner join sb_akt sba on sba.sba_from_k_id = ok.ok_from_k_id
          where ok.ok_from_k_id = :k_id and
                sba.sba_from_sbl_id = :sbl_id and
                ok.ok_type in (4, 9) and
                ok.ok_ol_is_active = 1 and
                ok.ok_ol_date between :rep_date_begin and :rep_date_end and
                ok.ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                ok_from_c_id = 0
          group by ok_ol_date
          having sum(ok_os + ok_perc + ok_fine) <> 0
          union all
          select coalesce(-sum(oko_os + oko_perc + oko_other), 0) as repay_sum, okf.oko_ol_date as ol_date
          from operkroff okf
          inner join sb_akt sba on sba.sba_from_k_id = okf.oko_from_k_id
          where okf.oko_from_k_id = :k_id and
                sba.sba_from_sbl_id = :sbl_id and
                oko_type = 4 and
                oko_ol_is_active = 1 and
                oko_ol_date between :rep_date_begin and :rep_date_end and
                oko_ol_date between sba.sba_date_begin and sba.sba_date_end
                -- :rep_date_end between sba_date_begin and sba_date_end and
          group by oko_ol_date
          having sum(oko_os + oko_perc + oko_other) <> 0
          into :repay_sum, ol_date
      do
      begin
        if (repay_sum > 0) then
        begin
          repay_sum_all = repay_sum_all + repay_sum;
          repay_sum = round(repay_sum, 0);
          res_text = BLOB_APPEND(res_text, f_format('<tr><td>%1</td><td align="center">%2</td><td>%3</td></tr>', f_datetostr(ol_date) || '</a>', k_kontract_no || '</a>', repay_sum));
        end
      end
    end
    res_text = BLOB_APPEND(res_text, f_format('<tr><td><b>%1</b></td><td align="center">%2</td><td><b>%3</b></td></tr>', 'Итого:' || '</a>', '' || '</a>', repay_sum_all));
  end
  res_text = BLOB_APPEND(res_text, '</table></body>');
  suspend;
end
