CREATE PROCEDURE HP_OPERCASH_DEFAULT_INFO(
    OBJECT_ID D_ID,
    OPER_DATE D_DATE,
    OPER_KIND D_STR_CODE,
    IS_IPF D_BOOL,
    C_ID D_ID = null)
RETURNS (
    RES_DEF_AMOUNT D_CURRENCY,
    RES_ADD_AMOUNT D_CURRENCY,
    RES_KONT_COMMENT D_STR_255,
    RES_DEF_MEMO D_STR_LONG,
    K_CONS_KIND D_SMALLINT,
    IS_DEFERRED_PAYMENT D_BOOL)
AS
declare variable s_running_total d_currency;
declare variable tmp_amount d_currency;
declare variable al_id d_str_20;
declare variable reg_id d_integer;
declare variable db_id d_integer;
declare variable folder_no d_str_code;
declare variable kf_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable k_from_pr_id d_integer;
declare variable k_kommis_amount d_currency;
declare variable k_nsp_amount d_currency;
declare variable nsp_perc d_percent;
declare variable card_id d_integer;
declare variable card_no d_str_20;
declare variable card_acc d_integer;
declare variable rd_name d_str_255;
declare variable s_all d_currency;
declare variable amount_total d_currency;
declare variable temp_comment d_str_255;
declare variable kc_id d_id;
declare variable k_info d_str_100;
declare variable pr_code d_str_100;
declare variable res_text d_str_255;
begin
  res_def_amount = 0;
  res_add_amount = 0;
  res_kont_comment = '';
  res_def_memo = '';
  folder_no = '';

  if (object_id is null) then
    exit;

  if (is_ipf is null) then
    is_ipf = 0;

  oper_kind = f_enum_code('cashoperkind', oper_kind, 0);

  if (coalesce(oper_kind, '') = '') then
    Exit;

  select k_kontract_no,
         k_disb_date,
         k_cons_kind,
         k_from_pr_id
  from kontract
  where k_id = :object_id
  into k_kontract_no,
       k_disb_date,
       k_cons_kind,
       k_from_pr_id;

  k_info = k_kontract_no || ' от ' || f_datetostr(k_disb_date);

  if (k_cons_kind is null) then
    k_cons_kind = 0;

  if (oper_kind in ('plus.loan.repay', 'plus.loan.cl_repay')) then
  begin
    select res_text
    from hp_ipf_check_insurance_pay(:object_id)
    into res_text;

    if (res_text <> '') then
    begin
      res_def_amount = 0;
      res_add_amount = 0;
      res_def_memo = res_text;
      res_kont_comment = 'Запрет погашения займа пока не закроются обязательства по страхованию'; -- RFCKG-5390
    end else
    begin
      delete from temp_vrepay
      where 1 = 1;
  
      if (oper_kind <> 'plus.loan.cl_repay') then
        c_id = null;
  
      select res_amount, s_running_total
      from hp_repay_fill(:object_id, null, :oper_date, 0, 1, 1, 1, 1, 1)
      into :res_def_amount, -- сумма по на текущую дату
           :s_running_total; -- сумма для полного погашения
  
      if (oper_kind = 'plus.loan.cl_repay') then
      begin
        select sum(kc_amount_approved), sum(iif(kc_from_c_id = :c_id, kc_amount_approved, 0))
        from knclient
        where kc_from_k_id = :object_id and
              kc_amount_approved > 0 and
              kc_closed_ol_id is null
        into s_all, res_def_amount;
  
        if (s_all is null) then
          s_all = 0;
  
        if (res_def_amount is null) then
          res_def_amount = 0;
  
        s_running_total = round(s_running_total / s_all * res_def_amount);
        res_def_amount = s_running_total;
      end
  
      res_kont_comment = 'Рекомендуемая сумма: ' || f_numtostr(res_def_amount, 2);
      res_kont_comment = res_kont_comment || ', для полного закрытия займа: ' || f_numtostr(s_running_total, 2);
  
      delete from temp_vrepay
      where 1 = 1;
  
      select res_memo
      from hp_repay_make_memo(:object_id, :oper_date, 0, '', 1, 1, 1, 1)
      into :res_def_memo;
      if (res_def_memo is null) then
        res_def_memo = '';
    end
  end else
  if (oper_kind = 'plus.loan.card_kommis') then
  begin
    res_kont_comment = 'Отсутствует заявка на перевыпуск карты';

    select cac_card_id, cac_from_cca_id, rd_name
    from cardaktclient, refdata
    where cac_from_k_id = :object_id and
          cac_from_rd_id = rd_id and
          cac_status  = f_enum_id('cardaktclient', 'issued_delayed')
    order by cac_date desc
    rows 1
    into card_id, card_acc, rd_name;

    if (card_id is not null and card_acc is not null) then
    begin
      select cal_pan_short
      from cardlist
      where cal_card_id = :card_id
      into card_no;

      if (coalesce(card_no, '') <> '') then
      begin
        res_def_amount = 350;
        card_no = left(card_no, 6) || '******' || right(card_no, 4);
        res_kont_comment = 'Комиссия ' || f_numtostr(res_def_amount, 0) || ' сом за карту ' || card_no;
        res_def_memo = 'Комиссия за перевыпуск карты %1; карт.счет № %2; причина: %3';
        res_def_memo = f_format(res_def_memo, :card_no, :card_acc, rd_name);
      end
    end
  end else
  if (oper_kind = 'plus.loan.offpay') then
  begin
    delete from temp_vrepay
    where 1 = 1;

    select res_amount_saldo
    from aarep_operextra_saldo(9, :object_id, f_get_rd_id('OperExtraKind.LawClientExpense'), 'LoanRepay', :oper_date, 0)
    into s_running_total;

    select sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc + oko_other + oko_sh_other)
    from operkroff
    where oko_from_k_id = :object_id and
          oko_from_c_id = :c_id and
          oko_ol_is_active = 1
    into res_def_amount;

    if (res_def_amount is null) then
      res_def_amount = 0;

    if (s_running_total > 0) then
    begin
      res_def_memo = 'ВНИМАНИЕ, имеется долг по судебным взысканиям: ' || f_numtostr(s_running_total, 2);
      res_kont_comment = 'Запрет на прием внесистемого (' || f_numtostr(res_def_amount, 2) || ' сом) пока не закроется судебные.';
      s_running_total = 0;
    end else
    begin
      s_running_total = res_def_amount;
      res_kont_comment = 'Общий внесистемный долг: ' || f_numtostr(res_def_amount, 2);
  
      delete from temp_vrepay
      where 1 = 1;
  
      select res_memo
      from hp_repay_make_memo(:object_id, :oper_date, 0, '', 1, 1, 1, 1)
      into :res_def_memo;
      if (res_def_memo is null) then
        res_def_memo = '';
    end
  end else
  if (oper_kind = 'plus.loan.sud') then
  begin
    delete from temp_vrepay
    where 1 = 1;

    select res_amount_saldo
    from aarep_operextra_saldo(9, :object_id, f_get_rd_id('OperExtraKind.LawClientExpense'), 'LoanRepay', :oper_date, 0)
    into res_def_amount;

    s_running_total = res_def_amount;
    res_kont_comment = 'Долг по судебным издержкам: ' || f_numtostr(res_def_amount, 2);

    delete from temp_vrepay
    where 1 = 1;

    res_def_memo = 'Погашение долга по судебным издержкам, дог. ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date);
  end else
  if (oper_kind = 'minus.loan.overpay') then
  begin
    select -sum(ok_os)
    from operkred
    where ok_from_k_id = :object_id and
          ok_ol_is_active = 1 and
          ok_type = 9
    into res_def_amount;

    if (exists(select *
               from temp_minus_loan_overpay
               where k_id = :object_id)) then
      select -sum(ok_os + ok_perc + ok_tax)
      from operkred
      where ok_from_k_id = :object_id and
            ok_ol_is_active = 1 and
            ok_type = 9
      into res_def_amount; -- RFCKG-5143

    if (res_def_amount is null) then
      res_def_amount = 0;

    if (res_def_amount < 0) then
      res_def_amount = 0;

    res_kont_comment = 'Доступная сумма к возврату: ' || f_numtostr(res_def_amount, 2);
    res_def_memo = 'Возврат переплаты по договору займа ' || k_info;
  end else
  if (oper_kind = 'plus.loan.konsult') then
  begin
    select res_amount
    from hp_opercash_loan_other_amount(:object_id, 'CashTransactionIncom.Consultation')
    into :res_def_amount;

    if (res_def_amount is null) then
      res_def_amount = 0;

    res_def_memo = 'Оплата за консультационные услуги ';
    res_kont_comment = 'Необходимая сумма к оплате: ' || f_numtostr(res_def_amount, 2);

    if (k_cons_kind > 0) then -- оплата конс. услуг в рассрочку (RFCKG-4900)
    begin
      select kc_id
      from knclient
      where kc_from_k_id = :object_id and
            kc_amount_approved > 0
      rows 1
      into kc_id;

      select price
      from clt_get_price_for_consultation(:kc_id)
      into amount_total;

      select res_text
      from hp_consult_payment_info(:object_id)
      into temp_comment;

      res_def_memo = res_def_memo || 'в рассрочку ';
      res_kont_comment = 'Стоимость услуги ' || round(amount_total, 2) || ', остаток долга ' || round(res_def_amount, 2) || ' ';
      res_kont_comment = res_kont_comment || ascii_char(10) || temp_comment;
    end
  end else
  if (oper_kind = 'minus.loan.konsult') then
  begin
    res_def_amount = 0;
    res_def_memo = 'Возврат оплаты за консультационные услуги';
    res_kont_comment = 'Необходимая сумма к возврату: ' || f_numtostr(res_def_amount, 2);
  end else
  if (oper_kind = 'plus.loan.insurance') then
  begin
    select res_amount
    from hp_opercash_loan_other_amount(:object_id, 'CashTransactionIncom.Insurance')
    into :res_def_amount;

    if (res_def_amount is null) then
      res_def_amount = 0;

    select pr_code
    from product
    where pr_id = :k_from_pr_id
    into pr_code;

    if (pr_code containing '.ipf.') then
      res_def_memo = 'Оплата за страхование финансирования ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date);
    else
      res_def_memo = 'Оплата за страхование кредита ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date);

    res_kont_comment = 'Необходимая сумма к оплате: ' || f_numtostr(res_def_amount, 2);

    if (pr_code containing '.ipf.' and pr_code not containing '.ignore_insurance.') then
      is_deferred_payment = 1; -- отсроченный платеж (RFCKG-5390)
  end else
  if (oper_kind = 'plus.loan.partial') then
  begin
    select res_amount
    from hp_opercash_loan_other_amount(:object_id, 'CashTransactionIncom.PartialPrice')
    into :res_def_amount;

    res_def_memo = 'Частичная оплата по ИПФ, договор финансирования ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date);
    res_kont_comment = 'Необходимая сумма к оплате: ' || f_numtostr(res_def_amount, 2);
  end else
  if (oper_kind = 'minus.loan.disb') then
  begin
    db_id = f_get_db_id();

    select k_amount_approved,
           k_from_kf_id,
           k_disb_date,
           case
               when (:db_id in (39801, 41701) and not(pr_code containing '.IPF.')) then 'Выдача кредита согл. договора займа ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date) || ' г. Кр.папка № ' -- было pr_code containing '.MURABAHA.' изменен от 02.04.2018 Улан
               when (:db_id in (39801, 41701) and pr_code containing '.IPF.') then 'Выдача согл. договора финансирования ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date) || ' г. Фин.досье № ' -- было pr_code containing '.MURABAHA.' изменен от 02.04.2018 Улан
               when (:db_id = 81001) then 'Выдача займа согл. договора займа ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date) || ' г. Кр.папка № '
               when (:db_id = 94401) then f_datetostr(k_disb_date) || ' tarixli ' || k_kontract_no || ' nomreli muqavileye esasen kredit buraxilishi. Kredit papkasi № '
             end
    from kontract, product
    where k_id = :object_id and
          k_from_pr_id = pr_id
    into res_def_amount, kf_id, k_disb_date, res_def_memo;

    folder_no = '';

    select ka.kal_new_kf_name, ka.kal_new_reg_id
    from knaktlist ka
    where ka.kal_from_kf_id = :kf_id and
          ka.is_delete = 0 and
          ka.kal_date <= :k_disb_date and
          ka.kal_date_end >= :k_disb_date
    into folder_no, reg_id;

    res_def_memo = trim(res_def_memo) || ' ' || folder_no;

    res_kont_comment = 'Сумма к выдаче займа: ' || f_numtostr(res_def_amount, 2);

    select sh_kommis
    from schedule
    where sh_from_k_id = :object_id and
          sh_id = 0
    into res_add_amount;

    k_nsp_amount = 0;
    k_kommis_amount = 0;
    if (res_add_amount > 0) then
    begin
      k_kommis_amount = res_add_amount;

      select nsp_percent_for_extra_int
      from hp_get_kontract_nsp_acc(:reg_id, :oper_date)
      into :nsp_perc;

      if (nsp_perc <> 0.0) then
      begin
        k_nsp_amount = round(k_kommis_amount * nsp_perc / 100.0, 0);
        res_add_amount = k_kommis_amount + k_nsp_amount;
        res_kont_comment = res_kont_comment || ', приход комиссии с НсП: ' || f_numtostr(res_add_amount, 2);
      end else
        res_kont_comment = res_kont_comment || ', приход комиссии: ' || f_numtostr(res_add_amount, 2);
    end
    if (:db_id in (39801, 41701) and
        exists(select *
               from product
               where pr_id = :k_from_pr_id and
                     pr_code containing '.ipf.')) then
    begin
      -- в КР займы по ИПФ через РМК не выдаются (RFCKG-5347)
      res_def_amount = null;
      res_add_amount = null;
      res_kont_comment = null;
      res_def_memo = null;
      k_cons_kind = null;
    end
  end else
  if (oper_kind = 'minus.to_bank') then
  begin
    res_def_memo = 'Сдача наличности в банк';
    res_def_amount = 0;

    al_id = '';
    tmp_amount = 0;

    select csp_office_id, iif(:is_ipf = 0, csp_cls_al_id, csp_ipf_al_id)
    from cashplace
    where csp_id = :object_id
    into reg_id, al_id;

    if (al_id is not null) then
      tmp_amount = f_al_bal(al_id, oper_date, 'sum');

    res_kont_comment = 'Остаток по кассе: ' || f_numtostr(tmp_amount, 2);

    tmp_amount = 0;
    select sum(iif(csp_cls_al_id is not null, f_al_bal(csp_cls_al_id, :oper_date, 'sum'), 0) +
               iif(csp_ipf_al_id is not null, f_al_bal(csp_ipf_al_id, :oper_date, 'sum'), 0))
    from cashplace
    where csp_office_id = :reg_id
    into tmp_amount;

    if (tmp_amount is null) then
      tmp_amount = 0;

    if (tmp_amount <> 0) then
      res_kont_comment = res_kont_comment || ', всего по офису: ' || f_numtostr(tmp_amount, 2);
  end else
  if (oper_kind = 'plus.from_bank') then
  begin
    res_def_memo = 'Обналичивание с расчетного счета согл.чека № ';
    res_def_amount = 0;
  end

  suspend;
end
