CREATE PROCEDURE CB_AZ_MO_2020(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    BALANCE_BEGIN CURR_TYPE,
    REPAY_SUM_NOT_OS DECIMAL(15,4),
    DELINQ_OS_PERIOD DECIMAL(15,4),
    BALANCE_DELINQ7_END CURR_TYPE,
    LO_QTY INTEGER,
    AVG_DISB_QTY_LO DECIMAL(18,2),
    ZNACH_PAR7 DECIMAL(18,2),
    BONUS_FACTOR_1 DECIMAL(18,2),
    BONUS_FACTOR_2 DECIMAL(18,2),
    SUM_VOZNOG_OFFICE DECIMAL(15,4),
    BONUS_MO DECIMAL(15,2),
    DISB_LO INTEGER,
    WORK_DAYS_QTY INTEGER,
    MANAGER_NAME STR_100,
    LO_STAGE_MONTH_FULL DECIMAL(15,1),
    PS_DATE_BEGIN DATE,
    BONUS_MEMO VARCHAR(255))
AS
declare variable reg_id integer;
declare variable rep_lastday_prev_month date;
declare variable rep_first_day_month date;
declare variable manager_id id_type;
declare variable jk_system_code str_100;
declare variable mo_qty smallint;
declare variable rep_year_month integer;
begin
  rep_lastday_prev_month = rep_date_end - extract(day from rep_date_end); --Последний день пред.месяца
  rep_first_day_month = rep_lastday_prev_month + 1; --Последний день пред.месяца
  rep_year_month = extract(year from :rep_date_end) || right('0' || extract(month from :rep_date_end), 2);

  --считаем количество рабочих дней
  select count(scd_work_hours) work_days_qty
  from sa_calendays sc
  where sc.scd_date between :rep_first_day_month and :rep_date_end and
        scd_work_hours > 0
  into :work_days_qty;

  --Разработан по процедуре начисления бонусов для менеджеров офисов (МО)

  --Расчет бонус фактора 1.
  /*2.1. МО устанавливается бонус фактор (согласно таблицы №1, в зависимости от значения PAR7 филиала) в процентном выражении
  от суммы вознаграждения подотчетного филиала. PAR7 – это баланс кредитов с просроченными платежами более 7 дней.
    */

  for select oks_region_id, sum(iif(:rep_date_end - oks_delinq_date > 7, oks_balance, 0))
      from oksummary oks
      inner join region_list(:rep_region_id, 1, -1, 1) on oks.oks_region_id = id_list
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_close_state = 0
      group by oks_region_id
      into :reg_id, balance_delinq7_end --PAR7 на конец
  do
  begin
    balance_begin = null;
    select sum(oks_balance)
    from oksummary
    where oks_region_id = :reg_id and
          oks_close_state = 0 and
          :rep_lastday_prev_month between oks_date_begin and oks_date_end
    into balance_begin; --  Балан КП на начало
    balance_begin = coalesce(balance_begin, 0);

    --2.2.1  «Значение PAR7 в % от КП» = «Баланс PAR7 на конец отчетного месяца» / «Баланс КП на начало отчетного месяца» * 100%
    if (balance_begin = 0 and
        balance_delinq7_end > 0) then
      znach_par7 = 100.0;
    else
    if (balance_delinq7_end = 0) then
      znach_par7 = 0.0;
    else
      znach_par7 = cast(balance_delinq7_end as double precision) / balance_begin * 100; --

    select count(ks_from_k_id)
    from kontstate ks
    where ks.ks_region_id = :reg_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_first_day_month and :rep_date_end
    into disb_lo; --выдачи

    lo_qty = 0;
    with lo_list
    as (select ps_from_c_id,
               (select min(ps_date_begin)
                from hr_pstatus ps1, hr_orders ho
                where ps_status = 1 and
                      ps_from_ho_id = ho_id and
                      ho.ho_status = 1 and
                      ps_date_begin < ps_date_end and
                      ho_order_type = 1 and
                      ps1.ps_from_c_id = ps.ps_from_c_id) first_date_begin
        from hr_pstatus ps, joblist, jobkind, sa_tabel sa
        where ps_from_reg_id = :reg_id and
              ps_status = 1 and
              ps_is_working = 1 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code = 'JobSystem.LO_Standart' and
              :rep_date_end between ps_date_begin and ps_date_end and
              ps_date_begin <= dateadd(day, -15, :rep_date_end) and
              ps.ps_from_c_id = sa.sat_from_c_id and
              ps_from_reg_id = sa.sat_office_id and
              sa.sat_year_month = :rep_year_month and
              sa.sat_total_hours > 118 --8*15=120. берем 115 чтобы получить 15 дней
    )
    select count(*)
    from lo_list
    where first_date_begin <= :rep_lastday_prev_month
    into lo_qty;
    if (lo_qty is null) then
      lo_qty = 0.0;

    --2.2.2 «Среднее количество выдач на КС/СКС в рабочий день» = «Количество выдач филиала за месяц» / «Количество КС/СКС филиала» / «Количество рабочих дней филиала в отчетный месяц»
    if (lo_qty > 0) then
      avg_disb_qty_lo = round(cast(disb_lo as d_currency) / cast(lo_qty as d_currency) / cast(work_days_qty as d_currency), 2);
    else
      avg_disb_qty_lo = 0.0;

    -->> 2.2.3

    --  Суммы погашении процентов пеня и админ сбор
    repay_sum_not_os = null;
    select -sum(ok_perc + ok_fine + ok_kommis)
    from operkred
    where ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_ol_reg_id = :reg_id and
          ok_ol_date between :rep_first_day_month and :rep_date_end
    into repay_sum_not_os;
    repay_sum_not_os = coalesce(repay_sum_not_os, 0);

    --  Сумма просроченной задолженности по ОС за отчетный период
    delinq_os_period = null;
    select sum(ok_os + ok_sh_os)
    from oksummary, operkred
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_region_id = :reg_id and
          :rep_date_end >= oks_delinq_date and
          oks_from_k_id = ok_from_k_id and
          oks_close_state = 0 and
          ok_ol_is_active = 1 and
          ok_type < 7 and
          ok_sh_date between :rep_first_day_month and :rep_date_end and
          ok_ol_date <= :rep_date_end
    into delinq_os_period;
    delinq_os_period = coalesce(delinq_os_period, 0);

    --  «Сумма вознаграждения подотчетного филиала» = (все фактически полученные суммы доходов по кредитованию за отчетный период) * 1% - (сумма просроченной задолженности по ОС за отчетный период) * 5%
    sum_voznog_office = repay_sum_not_os / 100 - delinq_os_period / 100 * 5;
    --<<2.2.3

    bonus_factor_1 = case
                       when znach_par7 <= 1.0 then 1.20
                       when znach_par7 <= 2.5 then 1.00
                       when znach_par7 <= 4.0 then 0.90
                       when znach_par7 <= 7.0 then 0.60
                       when znach_par7 <= 12.0 then 0.30
                       else 0.20
                     end;

    if (:rep_date_end >= '01.11.2020' and
        znach_par7 > 12.0) then
      bonus_factor_1 = 0;

    bonus_factor_2 = case
                       when avg_disb_qty_lo > 2.0 then 1.30
                       when avg_disb_qty_lo >= 1.6 then 1.10
                       when avg_disb_qty_lo >= 1.2 then 1.00
                       when avg_disb_qty_lo >= 1.0 then 0.90
                       when avg_disb_qty_lo >= 0.85 then 0.60
                       when avg_disb_qty_lo >= 0.7 then 0.40
                       else 0.0
                     end;

    manager_id = null;
    jk_system_code = null;
    ps_date_begin = null;
    mo_qty = 0;
    manager_name = null;
    lo_stage_month_full = null;
    for select ps_from_c_id, jk_system_code, ps_date_begin
        from hr_pstatus ps, joblist, jobkind
        where :rep_date_end between ps_date_begin and ps_date_end and --Должен работать на момент начисления бонусов
              ps_status = 1 and
              ps_is_working = 1 and
              ps_from_reg_id = :reg_id and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code = 'JobSystem.Region_KM'
        order by ps_date_begin
        into manager_id, jk_system_code, ps_date_begin
    do
    begin
      if (mo_qty = 0 and
          ps_date_begin < rep_first_day_month + 15) then
      begin
        mo_qty = 1;

        select c_name_fio
        from clients
        where c_id = :manager_id
        into manager_name;

        lo_stage_month_full = null;
        select lo_stage_month_full
        from hp_get_lo_stage_full(:manager_id, :rep_date_end, :jk_system_code, 0)
        into lo_stage_month_full; --Стаж

        --  вытаскивает данные по выговорам и бонус коэффициент
        bonus_memo = null;
        select bonus_memo
        from hp_get_vigovor_koef(:manager_id, :rep_date_end)
        where bonus_koef < 100 or
              bonus_sum > 0
        into bonus_memo;
        bonus_memo = coalesce(bonus_memo, '');
        if (bonus_memo <> '') then
        begin
          mo_qty = 0;
          bonus_memo = left(manager_name || '-' || bonus_memo, 255);
        end
      end
    end

    --2.2.4>> «Сумма бонуса МО» = «Сумма вознаграждения подотчетного филиала» * «Бонус фактор МО 1» * «Бонус фактор МО 2»
    bonus_mo = sum_voznog_office * bonus_factor_1 * bonus_factor_2;

    manager_name = coalesce(manager_name, '');
    lo_stage_month_full = coalesce(lo_stage_month_full, 0);
    if (bonus_mo < 0 or (manager_name <> '' and
        (lo_stage_month_full < 3 or bonus_memo <> ''))) then
      bonus_mo = 0;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into :reg_name;

    suspend;
  end
end
