CREATE PROCEDURE HR_PSTATUS_ONE_PERSONAL(
    PER_ID DECIMAL(18,0))
RETURNS (
    PS_ID INTEGER,
    PS_FROM_HO_ID DECIMAL(18,0),
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    PS_PERIOD_MONTH INTEGER,
    PS_DATE_END_IN_ORDER DATE,
    PS_FROM_REG_ID INTEGER,
    PS_FROM_DEP_ID INTEGER,
    PS_FROM_JL_ID INTEGER,
    PS_IS_ON_STAFF SMALLINT,
    PS_IS_WORKING SMALLINT,
    PS_SALARY DECIMAL(18,4),
    PS_REG_NAME VARCHAR(100),
    PS_DEP_NAME VARCHAR(100),
    PS_JL_NAME VARCHAR(100),
    PS_TABEL_NO VARCHAR(10),
    PSA_AGREEMENT_NO VARCHAR(10),
    JL_ACC SMALLINT)
AS
begin
  ps_date_end = '31.12.9999';
  for select ps_id, ps_from_ho_id, ps_date_begin, ps_period_month, ps_date_end_in_order, ps_from_reg_id, ps_from_dep_id,
             ps_from_jl_id, ps_is_on_staff, ps_is_working, ps_salary, reg_name, dep_name, jl_name, ps_tabel_no,
             iif(ps_from_jl_id in (226, 227) and f_get_db_id() in (39801, 41701), 1, 0)
      from hr_pstatus
      left join region on (ps_from_reg_id = reg_unicode)
      left join depart on (ps_from_dep_id = dep_id)
      left join joblist on (ps_from_jl_id = jl_id)
      where ps_from_c_id = :per_id and
            ps_status = 1
      order by ps_date_begin desc, ps_from_ho_id desc, ps_id desc
      into :ps_id, :ps_from_ho_id, :ps_date_begin, :ps_period_month, :ps_date_end_in_order, :ps_from_reg_id,
           :ps_from_dep_id, :ps_from_jl_id, :ps_is_on_staff, :ps_is_working, :ps_salary, :ps_reg_name, :ps_dep_name,
           :ps_jl_name, :ps_tabel_no, :jl_acc
  do
  begin
    psa_agreement_no = null;

    select psa.psa_agreement_no
    from hr_pstatus_agreement psa
    where psa.psa_from_ho_id = :ps_from_ho_id and
          psa.psa_from_ps_id = :ps_id
    into :psa_agreement_no;

    suspend;
    ps_date_end = ps_date_begin - 1;
  end
end
