CREATE PROCEDURE HP_BKI_PARAMETERS_LIST(
    BEV_ID D_ID,
    WITH_VALUES D_BOOL)
RETURNS (
    BFV_ID D_ID,
    PARENT_ID D_ID,
    DB_FIELD D_STR_CODE,
    XML_NODE D_STR_CODE,
    PARENT_NAME D_STR_100,
    INDICATIOR_NAME D_STR_LONG,
    IS_AVAILABLE D_BOOL,
    VAR_VALUE_OLD D_STR_8100,
    VAR_VALUE_NEW D_STR_8100)
AS
declare variable xml_version_id d_integer;
declare variable bki_name d_str_255;
declare variable bki_code d_str_code;
declare variable event_code d_str_code;

begin
  if (bev_id is null) then
    exit;

  if (with_values is null) then
    with_values = 0;

  select xml_version_id
  from hp_bki_get_current_xml_ver
  into xml_version_id;

  select f_enum('bki_event', bev_event_id, 'code')
  from bki_events
  where bev_id = :bev_id
  into event_code;

  event_code = left(event_code, position('_', event_code) - 1);

  for select bfv_id, parent_id, bki_code, bki_name, db_field, xml_node
  from hp_bki_indicator_list('Event_' || :event_code, :xml_version_id)
  into bfv_id, parent_id, bki_code, bki_name, db_field, xml_node
  do
  begin
    is_available = 0;
    var_value_old = null;
    var_value_new = null;
    parent_name = null;
  
    indicatior_name = iif(db_field starting with '<', '', bki_code || '  ') || bki_name;

    if (with_values = 0 and db_field starting with '<') then  -- для дерева отметим узел если хотя бы один ребенок отмечен
    begin
        if (exists(select bfv_id
                from hp_bki_indicator_list(:bki_code, :xml_version_id), bki_send_data_corr
                where bsd_from_bev_id = :bev_id
                  and bsd_from_bfv_id = bfv_id))
      then
        is_available = 1;
    end
    else
      if (exists(
            select bsd_id
            from bki_send_data_corr, bki_xml_fields_ver, bki_xml_fields, bki_indicators
            where bsd_from_bev_id = :bev_id
              and bsd_from_bfv_id = :bfv_id
              and bfv_id = bsd_from_bfv_id
              and bxf_id = bfv_from_bxf_id
              and bki_code = bxf_from_bki_code
              and bfv_from_bxv_id = :xml_version_id
              and coalesce(bki_db_field, '') not starting with '<'))
      then
        is_available = 1;

    if (with_values = 1 and is_available = 1 and coalesce(db_field, '') not starting with '<') then
    begin
      select f_bki_data_text(bsd_from_bkd_id), f_bki_data_text(bsd_new_value_id)
      from bki_send_data_corr
      where bsd_from_bfv_id = :bfv_id
        and bsd_from_bev_id = :bev_id
      into var_value_old, var_value_new;
  
      if (row_count = 0) then
        is_available = 0;

      select bki_name
      from bki_xml_fields_ver, bki_xml_fields, bki_indicators
      where bfv_id = :parent_id
        and bxf_id = bfv_from_bxf_id
        and bki_code = bxf_from_bki_code
      into parent_name;
    end

    if (with_values = 0 or is_available = 1) then
      suspend;
  end 
end
