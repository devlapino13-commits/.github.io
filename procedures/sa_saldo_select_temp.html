CREATE PROCEDURE SA_SALDO_SELECT_TEMP(
    SA_REGION_ID INTEGER,
    SA_DD DATE)
RETURNS (
    SS_FROM_C_ID DECIMAL(18,0),
    SS_OFFICE_ID INTEGER,
    SS_SALDO DECIMAL(18,4))
AS
begin
  for select oss_from_c_id, oss_saldo
      from opersalsaldo
      where oss_date_begin <= :sa_dd and
            oss_date_end >= :sa_dd and
            oss_saldo <> 0
      into :ss_from_c_id, :ss_saldo
  do
  begin
    select rep_region_id
    from hp_get_employer_region(:ss_from_c_id, :sa_dd, :sa_region_id)
    into :ss_office_id;
    if (ss_office_id is not null) then
      suspend;
  end
