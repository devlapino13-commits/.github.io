CREATE PROCEDURE AAREP_REPORT_ON_ACTIVE_POINTS(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    K_FACT_DISB_DATE D_DATE,
    OFFICE_NAME D_STR_100,
    C_INN VARCHAR(20),
    C_NAME_FIO D_STR_255,
    K_KONTRACT_NO D_STR_20,
    KC_AMOUNT_APPROVED CURR_TYPE,
    PR_NAME D_STR_100,
    KC_CIRCLE D_INTEGER,
    K_PERCENT DECIMAL(9,0),
    OFFICER_NAME D_STR_255,
    K_PERIOD D_INTEGER,
    PR_TYPE D_STR_20,
    CLIENT_STATUS D_STR_20,
    CL_POINTS_DATE D_DATE,
    CL_POINTS D_STR_20,
    PR_MEMO D_STR_255,
    REGION_NAME D_STR_100,
    ACTIVE_KONTRACT_QTY D_INTEGER,
    OKS_BALANCE CURR_TYPE,
    AMOUNT_APPROVED CURR_TYPE)
AS
declare variable reg_id id_type;
declare variable c_id id_type;
declare variable pr_id id_type;
declare variable kf_id id_type;
begin
  client_status = '';
  for select k.k_kontract_no,
             k.k_period,
             k.k_fact_disb_date,
             k.k_percent,
             kc.kc_amount_approved,
             kc.kc_circle,
             kc.kc_from_c_id,
             k.k_last_reg_id,
             k.k_from_pr_id,
             k.k_from_kf_id
      from kontract k
      inner join knclient kc on kc.kc_from_k_id = k.k_id
      inner join kontstate ks on ks.ks_from_k_id = k.k_id
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(ks.ks_region_id as integer)
      where kc.kc_amount_approved > 0 and
            ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end
      into :k_kontract_no,
           :k_period,
           :k_fact_disb_date,
           :k_percent,
           :kc_amount_approved,
           :kc_circle,
           :c_id,
           :reg_id,
           :pr_id,
           :kf_id
  do
  begin

  select c.c_name_fio
  from knaktlist kal
  inner join clients c on c.c_id = kal.kal_new_kf_officer
  where kal.kal_from_kf_id = :kf_id and
  :k_fact_disb_date between kal.kal_date and kal.kal_date_end and
  kal.is_delete = 0
  into :officer_name;

    select iif(pr.pr_code containing ('.ipf.'), 'ипф', 'классика'),
               pr.pr_name,
               pr.pr_memo
    from product pr
    where pr.pr_id = :pr_id
    into :pr_type,
         :pr_name,
         :pr_memo;

    select sum(kc.kc_amount_approved)
    from knclient kc
    inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
    where kc.kc_from_c_id = :c_id and
          kc.kc_amount_approved > 0 and
          ks.ks_state = 1 and
          ks.ks_ol_is_delete = 0 and
          ks.ks_date_begin between :rep_date_begin and :rep_date_end and
          ks.ks_date_begin = :k_fact_disb_date
    into :amount_approved;

    if (:kc_circle <= 1) then
      client_status = 'Новый';
    else
      client_status = 'Повторный';

    office_name = null;
    region_name = null;

    select rg.reg_code || ' ' || rg.reg_name,
           rg2.reg_name
    from region rg
    inner join region rg2 on rg2.reg_unicode = rg.parent
    where rg.reg_unicode = :reg_id and
          rg2.reg_unicode > 0
    into :office_name,
         :region_name;

    c_name_fio = null;
    c_inn = null;

    select cl.c_name_fio,
           cl.c_inn
    from clients cl
    where cl.c_id = :c_id
    into :c_name_fio,
         :c_inn;

    cl_points = f_point(:c_id, :k_fact_disb_date, 0);
    cl_points_date = :k_fact_disb_date;

    active_kontract_qty = 0;

    select count(*)
    from kontstate ks
    inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
    where kc.kc_from_c_id = :c_id and
          kc.kc_amount_approved > 0 and
          ks.ks_state = 1 and
          :k_fact_disb_date between ks.ks_date_begin and ks.ks_date_end and
          ks.ks_ol_is_delete = 0
    into :active_kontract_qty;

    oks_balance = 0;

    select sum(oks.oks_balance * (kc.kc_amount_approved / kt.k_amount_approved))
    from oksummary oks
    inner join knclient kc on kc.kc_from_k_id = oks.oks_from_k_id
    inner join kontract kt on kc.kc_from_k_id = kt.k_id
    where kc.kc_from_c_id = :c_id and
          kc.kc_amount_approved > 0 and
          :k_fact_disb_date between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_close_state = 0
    into :oks_balance;
    suspend;
  end
end
