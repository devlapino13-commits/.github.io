CREATE PROCEDURE HP_EDOC_CLIENT_MSG(
    EDS_ID D_ID)
RETURNS (
    MSG_ID D_ID,
    MSG_TYPE D_INTEGER,
    MSG_MAIL_TO D_STR_255,
    MSG_BODY D_STR_8100,
    MSG_STATUS D_INTEGER,
    MSG_DATETIME D_DATETIME,
    MSG_SEND_STATUS D_INTEGER,
    MSG_TYPE_STR D_STR_CODE,
    MSG_STATUS_STR D_STR_CODE,
    MSG_SEND_STATUS_STR D_STR_CODE,
    EDM_STATUS D_SMALLINT)
AS
declare variable n d_integer;
declare variable str2 d_str_100;
declare variable phone d_str_100;
begin
  for select msg_id, msg_type, msg_mail_to, msg_body, msg_status, msg_datetime, msg_status, edm_status
      from edoc_msg, fsrobot_msgbox
      where edm_from_eds_id = :eds_id and
            edm_from_msg_id = msg_id and
            (coalesce(msg_mail_to, '') <> '' or
            msg_type = 2)
      order by msg_datetime, msg_id
      into msg_id, msg_type, msg_mail_to, msg_body, msg_status, msg_datetime, msg_send_status, edm_status
  do
  begin
    if (coalesce(msg_mail_to, '') = '') then
      msg_mail_to = 'КМП';
    msg_type_str = trim(case msg_type
                          when 0 then 'email'
                          when 1 then 'sms'
                          when 2 then 'push'
                          when 3 then 'sms'
                          when 4 then 'email'
                          else 'другое'
                        end);

    if (msg_status < 0) then
      msg_status_str = 'ошибка';
    else
    if (msg_status = 0) then
      msg_status_str = 'отправляется';
    else
      msg_status_str = 'отправлен';

    msg_send_status_str = msg_status_str;

    n = 0;
    while (n <= 9) do
    begin
      msg_body = replace(msg_body, n, '*');
      n = n + 1;
    end

    str2 = '';
    if (msg_type_str = 'email') then
    begin
      n = position('@', msg_mail_to);
      if (n > 0) then
      begin
        str2 = substring(msg_mail_to from n);
        msg_mail_to = left(msg_mail_to, n - 1);
      end
    end

    if (msg_type = 2) then
    begin
      phone = null;
      select clt_code
      from contlist
      inner join edoc_msg on edm_from_clt_id = clt_id
      where edm_from_eds_id = :eds_id and
            edm_from_msg_id = :msg_id
      into phone;

      if (coalesce(phone, '') <> '') then
        msg_mail_to = phone; -- RFCRU-2229
    end

    if (not exists(select *
                   from hp_permission_access_check(uxf$user(), 'PermActions', 0, 'finstart.see_phone_for_edoc')
                   where has_access = 1)) then
      msg_mail_to = left(msg_mail_to, char_length(msg_mail_to) - 5) || '***' || right(msg_mail_to, 2) || str2;

    suspend;
  end
end
