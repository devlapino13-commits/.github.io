CREATE PROCEDURE AAREP_BAL_CLIENT(
    REP_DATE D_DATE,
    REGION_ID D_ID)
RETURNS (
    REGION_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    KONTRACT_NO D_STR_100,
    DISB_DATE D_DATE,
    DISB_AMOUNT CURR_TYPE,
    BALANCE CURR_TYPE,
    PERC CURR_TYPE,
    DEL_DAYS INTEGER,
    CL_FIO D_STR_100,
    CL_NAME D_STR_100,
    CL_SNAME D_STR_100,
    CL_OTCHES D_STR_100,
    INN D_STR_100,
    PR_NAME D_STR_100,
    BIS_TYPE D_STR_255,
    FIN_TYPE D_STR_100,
    ZAD_BAL CURR_TYPE,
    ZAD_OS_CL CURR_TYPE,
    ZAD_PERC_CL CURR_TYPE,
    PENYA CURR_TYPE,
    REG_STAT_NAME D_STR_100)
AS
declare variable k_id d_id;
declare variable c_id d_id;
declare variable reg_id d_id;
declare variable oks_balance curr_type;
declare variable k_amount_approved curr_type;
declare variable kc_amount_approved_fact curr_type;
declare variable close_ol_date d_date;
declare variable zad_perc_no_delinq_kc curr_type;
declare variable bs_name d_str_100;
declare variable delinq_os curr_type;
declare variable delinq_perc curr_type;
declare variable fine curr_type;
declare variable zad_perc curr_type;
declare variable zad_os curr_type;
declare variable zad_os_no_delinq curr_type;
declare variable zad_perc_no_delinq curr_type;
declare variable zad_perc_delinq curr_type;
declare variable zad_os_delinq curr_type;
declare variable sh_fine_base curr_type;
declare variable zad_os_kc curr_type;
declare variable zad_perc_kc curr_type;
declare variable zad_os_no_delinq_kc curr_type;
declare variable k decimal(18,2);
declare variable dolya decimal(18,2);
begin
  for select oks.oks_from_k_id, oks.oks_region_id,
             iif(:rep_date - oks.oks_delinq_date <= 0, 0, :rep_date - oks.oks_delinq_date), oks_balance,
             reg.reg_code || ' ' || reg.reg_name, --
             reg2.reg_code || ' ' || reg2.reg_name, --
             reg.reg_stat_rep_region
      from oksummary oks
      inner join region_list(:region_id, 1, 3, 1) t on t.id_list = oks.oks_region_id
      inner join region reg on oks.oks_region_id = reg.reg_unicode
      inner join region reg2 on reg.parent = reg2.reg_unicode
      where :rep_date between oks.oks_date_begin and oks.oks_date_end and
            oks_close_state = 0
      into k_id, reg_id, del_days, oks_balance, office_name, region_name, reg_stat_name
  do
  begin
    kontract_no = '';
    pr_name = '';
    fin_type = '';
    select k.k_kontract_no, k_disb_date, k.k_amount_approved, pr.pr_name,
           iif(pr.pr_code containing '.IPF.', 'ИПФ', 'Классика')
    from kontract k
    inner join product pr on k.k_from_pr_id = pr_id
    where k.k_id = :k_id
    into kontract_no, disb_date, k_amount_approved, pr_name, fin_type;

    close_ol_date = null;
    kc_amount_approved_fact = null;

    for select kc.kc_from_c_id, c.c_name_fio, c.c_inn, round(kc_amount_approved, 2), ol_date,
               sum(iif(ol_id is null, round(kc_amount_approved, 2), 0)) over()
        from knclient kc
        inner join clients c on c.c_id = kc.kc_from_c_id
        left join operlist ol on ol.ol_id = kc.kc_closed_ol_id and
              ol.is_delete = 0 and
              ol.ol_date <= :rep_date
        where kc.kc_from_k_id = :k_id and
              kc.kc_amount_approved > 0
        into c_id, cl_fio, inn, disb_amount, close_ol_date, kc_amount_approved_fact
    do
    begin

      zad_perc_no_delinq_kc = 0;
      zad_perc = 0;
      zad_os_cl = 0;
      penya = 0;
      select sum(sum_fine_base_os), sum(iif(sh_date <= :rep_date, sh_calc_perc, 0)), sum(sh_calc_fine),
             sum(iif(sh_date <= :rep_date, zad_perc, 0)), sum(zad_os), sum(iif(sh_date >= :rep_date, zad_os, 0)),
             sum(iif(sh_date >= :rep_date, sh_calc_perc, 0)), sum(iif(sh_date < :rep_date, sh_calc_perc, 0)),
             sum(iif(sh_date < :rep_date, zad_os, 0)), sum(sh_fine_base)
      from aarep_make_zad_list(:k_id, :rep_date, 1)
      into delinq_os, delinq_perc, fine, zad_perc, zad_os, zad_os_no_delinq, zad_perc_no_delinq, zad_perc_delinq,
           zad_os_delinq, sh_fine_base; -- delinq_perc накопленный процент  delinq_os Задолж по ОС
      if (close_ol_date is null) then
      begin

        k = 1.0;
        if (kc_amount_approved_fact > 0) then
          k_amount_approved = kc_amount_approved_fact;

        k = disb_amount / k_amount_approved;

        balance = round(oks_balance * k, 2);
        dolya = 0;
        dolya = cast(disb_amount as double precision) / k_amount_approved;
        zad_os_kc = round(dolya * zad_os, 2);
        zad_os_no_delinq_kc = round(dolya * zad_os_no_delinq, 2);
        zad_os_cl = round(dolya * zad_os_delinq, 2);
        zad_perc_kc = round(dolya * zad_perc, 2);
        zad_perc_no_delinq_kc = round(dolya * zad_perc_no_delinq, 2);
        zad_perc_cl = round(dolya * zad_perc_delinq, 2);
        penya = round(dolya * fine, 2);
      end
      else
        balance = 0.0;

      cl_sname = '';
      cl_name = '';
      cl_otches = '';
      select left(txt1, 1) || lower(right(txt1, char_length(txt1) - 1)),
             left(txt2, 1) || lower(right(txt2, char_length(txt2) - 1)), iif(txt3 is not null and txt3 <> '',
             (left(txt3, 1) || lower(right(txt3, char_length(txt3) - 1))), '')
      from hp_fio_clear(:cl_fio)
      into :cl_sname, :cl_name, :cl_otches;

      cl_name = coalesce(cl_name, ''); -- Имя
      cl_sname = coalesce(cl_sname, ''); -- фамилия
      cl_otches = coalesce(cl_otches, ''); -- Отчетво
      if (cl_fio like '%уулу%') then
        cl_sname = cl_sname || ' уулу';

      if (cl_fio like '%кызы%') then
        cl_sname = cl_sname || ' кызы';

      if (cl_fio like '%uulu%') then
        cl_sname = cl_sname || ' uulu';

      if (cl_fio like '%kyzy%') then
        cl_sname = cl_sname || ' kyzy';
      bis_type = null;
      bs_name = null;
      select bt_name, bs_name, kc_amount_approved
      from knclient kc, bussubtype, bustype
      where kc_from_c_id = :c_id and
            kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            kc_busuness_subtype = bs_id and
            bs_from_bt_id = bt_id
      into bis_type, bs_name, disb_amount;
      if (bs_name is null) then
      begin
        bis_type = 'не указан';
        bs_name = 'не указан';
      end
      zad_bal = 0;
      if (del_days <> 0) then
      begin
        :zad_bal = :balance;
      end
      perc = 0;
      perc = zad_perc_no_delinq_kc + zad_perc_cl;
      suspend;
    end
  end
end
