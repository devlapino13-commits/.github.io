CREATE PROCEDURE OKT_ACTIVE_QUESTION_LIST
RETURNS (
    CP_ID INTEGER,
    CPC_NO INTEGER,
    COUNTER INTEGER,
    CPC_QESTION_DATETIME TIMESTAMP,
    CITIZEN VARCHAR(10),
    CP_FIO STR_100,
    CPC_QESTION_OPER STR_100,
    CALL_TYPE VARCHAR(10),
    CPC_BLOCKED_OPER_ID ID_TYPE,
    CPC_BLOCKED_OPER STR_100,
    CPC_QESTION_TEXT STR_255,
    HAS_ANSWER SMALLINT,
    IS_CLIENT SMALLINT)
AS
begin
  counter = 0;

  for select cp.cp_id, cpc.cpc_no, cpc.cpc_qestion_datetime,

             (
             case cp.cp_citizen
               when 0 then 'Не указ.'
               when 8 then 'КГ'
               when 9 then 'УЗ'
               when 13 then 'РФ'
               when 12 then 'ТЖ'
             end),
             cp.cp_fio, un.un_name as cpc_qestion_oper,

             (
             case cpc.cpc_call_type
               when 1 then 'Исходящ'
               when 2 then 'Входящ'
             end),
             cpc.cpc_blocked_oper_id, left(cpc.cpc_qestion_text, 150),
             (
             case cpc_not_answered
               when 0 then 1
               else 0
             end),
             (
             case
               when cp.cp_from_c_id is null then 0
               else 1
             end)
      from cc_person_calls cpc
      inner join cc_persons cp on cp.cp_id = cpc.cpc_from_cp_id
      inner join username un on un.un_id = cpc.cpc_qestion_oper_id
      where cpc.cpc_qestion_open = 1 and
            cpc.cpc_to_lawyer = 0 and
            cpc.cpc_no > 0
      order by cpc.cpc_qestion_datetime, cpc.cpc_from_cp_id, cpc.cpc_no
      into :cp_id, :cpc_no, :cpc_qestion_datetime, :citizen, :cp_fio, :cpc_qestion_oper, :call_type,
           :cpc_blocked_oper_id, :cpc_qestion_text, :has_answer, :is_client
  do
  begin
    cpc_blocked_oper = '';
    if (cpc_blocked_oper_id is not null) then
    begin
      select un.un_name
      from username un
      where un.un_id = :cpc_blocked_oper_id
      into :cpc_blocked_oper;
    end

    counter = counter + 1;

    cpc_qestion_text = replace(cpc_qestion_text, ascii_char(13), ' ');
    cpc_qestion_text = replace(cpc_qestion_text, ascii_char(10), ' ');
    cpc_qestion_text = trim(cpc_qestion_text);

    if (cpc_qestion_text <> '') then
    begin
      if (char_length(cpc_qestion_text) > 100) then
        cpc_qestion_text = left(cpc_qestion_text, 100) || '...';
    end

    suspend;
  end
end
