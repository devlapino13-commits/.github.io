CREATE PROCEDURE HP_CREATE_MONITORING(
    K_NO D_STR_20)
RETURNS (
    ACL_ID D_ID,
    RES_TXT D_STR_255)
AS
declare variable k_id d_id;
declare variable date_x d_date;
declare variable obj_kind d_integer;
declare variable k_disb_date d_date;
declare variable acl_date d_date;
declare variable acl_datetime d_datetime;
declare variable acl_last_datetime d_datetime;
declare variable is_ipf d_bool;
declare variable reg_id d_integer;
declare variable kc_c_id d_id;
declare variable kc_un_id d_id;
declare variable obj_id d_id;
declare variable rd_id d_integer;
declare variable kf_id d_id;
declare variable pr_code d_str_100;
declare variable kc_loan_target d_integer;
begin
  acl_id = 0;
  date_x = '03.03.2022'; -- Дата перехода на knclient
  rd_id = f_get_rd_id('Anketa.TargetUseLoan');
  res_txt = '';

  if (k_no similar to '[-]?[[:DIGIT:]]+') then
    k_id = (select k_id from kontract where k_id = :k_no);
  else
    k_id = (select k_id from kontract where k_kontract_no = :k_no);

  select
    k.k_id,
    k.k_disb_date,
    pr.pr_code,
    iif(pr.pr_code containing '.ipf.', 1, 0),
    kal.kal_new_kf_officer,
    kal.kal_new_reg_id,
    kal.kal_from_kf_id
  from kontract k
    inner join knaktlist kal
      on kal.kal_from_kf_id = k.k_from_kf_id
    inner join product pr
      on pr.pr_id = k.k_from_pr_id
  where k.k_id = :k_id
    and current_date between kal.kal_date and kal.kal_date_end
    and kal.is_delete = 0
  into
    :k_id,
    :k_disb_date,
    :pr_code,
    :is_ipf,
    :kc_c_id,
    :reg_id,
    :kf_id;
  -- Если папка за Виртуальным сотрудником то передаём её МО офиса
  if (:kc_c_id = 1000000000100 or
      :kc_c_id = 1000000000005) then
  begin
    kc_c_id = null;
    -- Получаем c_id МО офиса
    select ps.ps_from_c_id
    from hr_pstatus ps
    inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
    inner join jobkind jk on jk.jk_id = jl.jl_kind and
                             jk.jk_system_code = 'JobSystem.Region_KM'
    where ps.ps_from_reg_id = :reg_id and
          current_date between ps.ps_date_begin and ps.ps_date_end and
          ps.ps_is_working = 1
    rows 1
    into :kc_c_id;

    -- Если у офиса нет МО, передаём задачу КС у которого была папка до передачи на Виртуального сотрудника
    if (:kc_c_id is null) then
    begin
      select kal.kal_new_kf_officer
      from knaktlist kal
      where kal.kal_from_kf_id = :kf_id and
            kal.kal_new_kf_officer not in (1000000000100, 1000000000005) and
            kal.is_delete = 0
      rows 1
      into :kc_c_id;
    end 
  end 

  -- un_id сотрудника за которым на данный момент закреплен кредит
  select un.un_id
  from username un
  where un.un_from_c_id = :kc_c_id
  into kc_un_id;

  -- Дата записи actionlist
  if (k_disb_date < '14.02.2022') then
    acl_date = k_disb_date + iif(k_disb_date >= '22.12.2021' and is_ipf = 1, 1, 8);
  else
    acl_date = k_disb_date + 1;
  acl_datetime = cast(acl_date as timestamp);
  acl_last_datetime = acl_datetime;

  -- В зависимости от даты берет obj_kind clients или knclient
  obj_kind = iif(k_disb_date > date_x, f_objkind('knclient', 'id'), f_objkind('clients', 'id'));

  -- Если кредит групповой проходит по всем клиентам
  for select
    iif(:obj_kind = f_objkind('clients', 'id'), kc_from_c_id, kc_id),
    kc_loan_target
  from knclient
  where kc_from_k_id = :k_id
  into
    obj_id,
    kc_loan_target
  do
  begin
    -- RFCKG-4003
    if (pr_code containing '.ipf.' and f_get_rd_code(kc_loan_target) not in ('Loan_Target.BusinessGrowth', 'Loan_Target.BusinessCreation')) then
    begin
        res_txt = 'По данному контраку нельзя создать задачу на заполенние ЦМ, для более подробной информации обратитесь в операционный отдел';
        suspend;
        exit;
    end

    if (not exists(select * -- Проверка на наличие уже существующей записи
                   from actionlist acl
                   where acl.acl_from_rd_id = :rd_id and
                         acl.acl_date = :acl_date and
                         acl.acl_object_kind = :obj_kind and
                         acl.acl_object_id = :obj_id and
                         acl.acl_from_reg_id = :reg_id)) then


    begin
      -- Если записи нет, добавляем запись
      insert into actionlist (acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                              acl_respons_c_id, acl_datetime, acl_from_un_id, acl_result_rd_id, acl_last_datetime)
      values (:acl_date, :reg_id, :obj_kind, :obj_id, :rd_id, 1, :kc_c_id, :acl_datetime, :kc_un_id, 0,
              :acl_last_datetime)
      returning acl_id
      into :acl_id;

      res_txt = '';
    end
    else
    begin
      select acl.acl_id
      from actionlist acl
      where acl.acl_from_rd_id = :rd_id and
            acl.acl_date = :acl_date and
            acl.acl_object_kind = :obj_kind and
            acl.acl_object_id = :obj_id and
            acl.acl_from_reg_id = :reg_id
      into acl_id;
      res_txt = 'Задача уже существует ' || obj_id;
    end
    suspend;
  end
end
