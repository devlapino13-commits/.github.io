CREATE PROCEDURE HP_AFM_CHECK_STATUS(
    K_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable C_ID D_ID;
declare variable MA_ID D_ID;
declare variable APPLICATE_DATE D_DATE;
begin
  res_text = '';

  select kc.kc_from_c_id, kn.k_from_ma_app_id, kn.k_applicate_date
  from kontract kn
  inner join knclient kc on kc.kc_from_k_id = kn.k_id
  where kn.k_id = :k_id and
        kc.kc_amount_approved > 0
  into :c_id, :ma_id, :applicate_date;

  if (not
      exists(select 1
             from extinfolist eil
             where eil.eil_from_c_id = :c_id and
                   eil.eil_from_eic_id in (13, 14) and
                   eil.eil_object_kind = f_objkind('miniapp', 'id') and
                   eil.eil_object_id = :ma_id and
                   eil.eil_date = :applicate_date and
                   eil.eil_status = 2)) then
  begin
    res_text = 'Ошибка на стороне АФМ, переотправьте запрос';
    suspend;
    exit;
  end

  select coalesce(hp.res_text, '')
  from hp_afm_get_status('afm', :c_id, :ma_id) hp
  where hp.status <> 1
  into :res_text;

  suspend;
end
