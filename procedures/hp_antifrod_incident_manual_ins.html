CREATE PROCEDURE HP_ANTIFROD_INCIDENT_MANUAL_INS(
    K_ID D_ID,
    AIM_CHECK_DATETIME D_DATETIME,
    AIM_INCIDENT_STR D_STR_LONG,
    AIM_INT_INV_STR D_STR_LONG,
    AIM_RECOMENDATION D_STR_LONG,
    AIM_CREATE_UN_ID D_ID,
    AIM_REQ_STATUS D_INTEGER,
    AIM_KUSP_NUMBER D_STR_64,
    AIM_TKT_NUMBER D_STR_64,
    AIM_REQ_DATETIME D_DATETIME,
    AIM_IS_SENDFC D_BOOL)
AS
declare variable aim_req_status_str d_str_code;
begin
  aim_req_status = coalesce(aim_req_status, 1);

  aim_req_status_str = f_enum('aim_status', :aim_req_status, 'code');

  if (aim_req_status_str = 'pol' and (coalesce(trim(aim_kusp_number), '') = '' or
                                      coalesce(trim(aim_tkt_number), '') = '' or
                                      aim_req_datetime is null)) then
  begin
    exception e_message('Требуется заполнить "Номер в КУСП", "Регистрационный номер талона-уведомления", "Дата принятия заявки в прав. органах." при совершенном обращении в правоохранительные органы!');
  end

  insert into antifrod_incident_manual(aim_object_kind, aim_object_id, aim_create_un_id, aim_check_datetime, aim_incident_str, aim_int_inv_str, aim_recomendation, aim_req_status, aim_kusp_number, aim_tkt_number, aim_req_datetime, aim_is_sendfc)
  values (9, :k_id, :aim_create_un_id, :aim_check_datetime, :aim_incident_str, :aim_int_inv_str, :aim_recomendation, :aim_req_status, :aim_kusp_number, :aim_tkt_number, :aim_req_datetime, coalesce(:aim_is_sendfc,0));
end
