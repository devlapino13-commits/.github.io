CREATE PROCEDURE FM_RKO_CONFIRM(
    OBJ_KIND D_STR_100,
    OBJ_ID ID_TYPE,
    UN_ID ID_TYPE)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable opc_status d_smallint;
declare variable opc_status_str d_str_100;
declare variable opc_is_signed d_bool;
declare variable opc_doc_no d_integer;
declare variable opc_memo d_str_long;
declare variable opc_amount d_currency;
declare variable opc_from_c_id d_id;
declare variable opc_cashier_id d_id;
declare variable opc_manager_id d_id;
declare variable opc_from_edl_id d_id;
declare variable opc_from_ol_id d_id;
declare variable opc_amount_str d_str_20;
declare variable opc_amount_text d_str_255;
declare variable opc_datetime_str d_str_20;
declare variable opc_oper_kind_name d_str_100;
declare variable client_fio d_str_100;
declare variable cashier_fio d_str_100;
declare variable manager_fio d_str_100;
declare variable comment_text d_str_100;
declare variable lang d_str_lang;
begin
  -- возвращает данные Расходно-кассового ордера для его подтверждения контролером (RFCKG-4031)
  -- данные возвращаются в виде html для показа в МП

  lang = f_get_lang_code(un_id, null);
  comment_text = 'Подтверждение Расходно-кассового ордера контролером (менеджером офиса)';

  select opc_doc_no,
         f_datetimetostr(opc_datetime),
         f_enum_name('cashoperkind', opc_oper_kind, 0),
         opc_memo,
         opc_amount,
         opc_from_c_id,
         opc_cashier_id,
         opc_manager_id,
         opc_status,
         opc_is_signed,
         opc_from_edl_id,
         opc_from_ol_id
  from opercash
  where opc_is_delete = 0 and
        opc_sign = -1 and
        opc_id = :obj_id
  into opc_doc_no,
       opc_datetime_str,
       opc_oper_kind_name,
       opc_memo,
       opc_amount,
       opc_from_c_id,
       opc_cashier_id,
       opc_manager_id,
       opc_status,
       opc_is_signed,
       opc_from_edl_id,
       opc_from_ol_id;

  if (opc_amount is null or opc_status is null or opc_from_c_id is null) then
    exception e_message 'Расходно-кассовый ордер не найден';

  select f_numtostr(:opc_amount, 2),
         lower(replace(curr_str, '.', ''))
  from fhp_currencystr(:opc_amount, 2)
  into opc_amount_str,
       opc_amount_text;

  select c_name_fio
  from clients
  where c_id = :opc_from_c_id
  into client_fio;

  select c_name_fio
  from clients
  where c_id = :opc_cashier_id
  into cashier_fio;

  select c_name_fio
  from clients
  where c_id = :opc_manager_id
  into manager_fio;

  opc_status_str = '√';

  if (opc_from_ol_id is null) then
  begin
    opc_status_str = case opc_status
                       when -1 then 'Отменен'
                       when 0 then 'Новый'
                       when 1 then 'Подписан'
                     end;

    if (opc_status = 0 and opc_from_edl_id is not null) then
      opc_status_str = 'В ожидании';
  end

  if (opc_from_ol_id is not null and opc_status = 0) then
    opc_status_str = 'В ожидании';

  if (opc_is_signed = 1) then
    opc_status_str = opc_status_str || ', контролером подтвержден';
  else
    opc_status_str = opc_status_str || ', ожидает подтверждения контролера';

  res_text = (select res_style
              from fm_html_styles) || '<body>';

  res_text = blob_append(res_text, '<h3>РАСХОДНО-КАССОВЫЙ ОРДЕР № ' || opc_doc_no || ' от ' || opc_datetime_str) || '</h3>';
  res_text = blob_append(res_text, '<p>Тип операции: <b>' || opc_oper_kind_name || '</b></p>');
  res_text = blob_append(res_text, '<p>Назначение: <b>' || opc_memo || '</b></p>');
  res_text = blob_append(res_text, '<p>Сумма: <b>' || opc_amount_str || '</b> (' || opc_amount_text || ')</p>');
  res_text = blob_append(res_text, '<p>Клиент: <b>' || client_fio || '</b></p>');
  res_text = blob_append(res_text, '<p>Кассир: <b>' || cashier_fio || '</b></p>');
  res_text = blob_append(res_text, '<p>Контролер: <b>' || manager_fio || '</b></p>');
  res_text = blob_append(res_text, '<p>Статус документа: <b>' || opc_status_str || '</b> ');

  if (opc_is_signed = 0) then
    res_text = blob_append(res_text, f_uri_sql(f_lang(lang, 'Подтвердить') || '</p>', 'rko_confirm', 'opercash', obj_id, 'none', 'Подтверждаете?', comment_text, 0, 0, '', '', 'dialog', '') || '<br>');

  res_text = blob_append(res_text, '</body>');
  suspend;
end
