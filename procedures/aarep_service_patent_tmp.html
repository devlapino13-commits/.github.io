CREATE PROCEDURE AAREP_SERVICE_PATENT_TMP(
    SD_ID INTEGER,
    SD_DATE_BEGIN DATE,
    SD_DATE_END DATE,
    SD_IS_PATENT INTEGER)
RETURNS (
    RES_DD DATE)
AS
declare variable dd_min date;
declare variable dd_max date;
begin
  if (sd_is_patent = 1) then
  begin
    for select sdc_patent_date_begin, sdc_patent_date_end
        from servicedocs
        where sdc_patent_date_begin is not null and
              sdc_patent_date_end is not null and
              sdc_patent_date_end >= sdc_patent_date_begin and
              sdc_from_sd_id = :sd_id
        into :dd_min, :dd_max
    do
    begin
      res_dd = dd_min;
      while (res_dd <= :dd_max) do
      begin
        if ((res_dd >= sd_date_begin) and
            (res_dd <= :sd_date_end)) then
          suspend;
        res_dd = res_dd + 1;
      end
    end
  end
  else
  begin
    for select sdc_spolis_date_begin, sdc_spolis_date_end
        from servicedocs
        where sdc_spolis_date_begin is not null and
              sdc_spolis_date_end is not null and
              sdc_spolis_date_end >= sdc_spolis_date_begin and
              sdc_from_sd_id = :sd_id
        into :dd_min, :dd_max
    do
    begin
      res_dd = dd_min;
      while (res_dd <= :dd_max) do
      begin
        if ((res_dd >= sd_date_begin) and
            (res_dd <= :sd_date_end)) then
          suspend;
        res_dd = res_dd + 1;
      end
    end
  end
end
