CREATE PROCEDURE CB_KG_RM_2021_08(
    REP_DATE_END DATE)
RETURNS (
    REGION_NAME VARCHAR(100),
    KOL_KC INTEGER,
    BASIC_BONUS CURR_TYPE,
    KPI1 CURR_TYPE,
    KPI2 CURR_TYPE,
    KPI3 CURR_TYPE,
    BONUS_RM CURR_TYPE,
    BONUS_RM_COMMENT INTEGER,
    KOL_DISB D_INTEGER,
    REG_BALANCE_END DECIMAL(18,2),
    REG_PAR30_END DECIMAL(18,2),
    KOEF_PAR30 DECIMAL(15,4),
    PLAN_SPF INTEGER,
    CUMULATIVE_KPI CURR_TYPE,
    AAREP_BONUS_MEMO VARCHAR(50),
    KOL_KC_4 CURR_TYPE,
    DISB_44 CURR_TYPE)
AS
declare variable rep_date_begin date;
declare variable region_id integer;
declare variable rep_period integer;
declare variable c_id id_type;
declare variable bonus_koef curr_type;
declare variable bonus_memo varchar(255);
declare variable lo_id d_id;
declare variable kol_disb_tmp integer;
declare variable plan_spf_company integer;
begin
  basic_bonus = 0;
  rep_date_begin = rep_date_end - extract(day from rep_date_end);
  rep_period = extract(year from rep_date_end) || right((extract(month from rep_date_end) + 100), 2);

  :plan_spf_company = 0;

  for select prs_plan_spf
      from planregionspf
      where prs_period = :rep_period and
            prs_kind = 0
      into plan_spf
  do
  begin
    :plan_spf_company = :plan_spf_company + :plan_spf;
  end

  for select prs_region, prs_plan_spf
      from planregionspf
      where prs_period = :rep_period and
            prs_kind = 0
      into region_id, plan_spf
  do
  begin
    select reg_name
    from region
    where reg_unicode = :region_id
    into region_name;

    for select ps_from_c_id
        from hr_pstatus, region_list(:region_id, 1, 3, 1), joblist, jobsystem
        where ps_from_reg_id = id_list and
              :rep_date_end between ps_date_begin and ps_date_end and
              ps_is_working = 1 and
              ps_status = 1 and
              ps_from_jl_id = jl_id and
              js_id = jl_kind and
              js_memo = 'JobSystem.Region_Director'
        into c_id

    do
    begin
      if (exists (select *
                  from hr_voctravel hrv
                  inner join hr_orders ho on ho.ho_id = hrv.hrv_from_ho_id
                  where :rep_date_end between hrv.hrv_date_begin and hrv.hrv_date_end and
                        hrv.hrv_from_c_id = :c_id and
                        ho.ho_status = 1 and
                        ho.ho_order_type in (17,18) and --17 Отпуск по беременности и родам, 18 Отпуск без содерж.по уходу за ребенком
                        ho.ho_status = 1)) then
      continue;



      kol_kc = 0;
      select count(*)
      from aarep_personal_list(:region_id, :rep_date_end, 1)
      where jl_name in ('Специалист по финансированию', 'Старший специалист по финансированию') and
            ps_curr_status not in ('Отпуск без содерж.по уходу за ребенком', 'Отпуск по беременности и родам', 'Трудовой отпуск декретницы')
      into kol_kc;
      kol_kc = coalesce(kol_kc, 1);

      --  Базовый бонус РМ
      basic_bonus = round((:kol_kc * 450000) / :plan_spf_company, 0);

      select count(*) --
      from kontstate ks, region_list(:region_id, 1, -1, 1)
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and -- говорит о том что кредит активный
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            cast(ks.ks_region_id as integer) = id_list
      into kol_disb;

      :reg_balance_end = 0;
      :reg_par30_end = 0;
      :koef_par30 = 0;
      :kol_kc_4 = 0;
      :disb_44 = 0;
      for select ps_from_c_id
          from aarep_personal_list(:region_id, :rep_date_end, 1)
          where jl_name in ('Специалист по финансированию', 'Старший специалист по финансированию') and
                ps_curr_status not in ('Отпуск без содерж.по уходу за ребенком', 'Отпуск по беременности и родам', 'Трудовой отпуск декретницы') and
                (((cast(:rep_date_end as date) - ps_date_begin) >= 120) or
                (ho_text like '%Перевести%'))
          into lo_id
      do
      begin

        :kol_kc_4 = :kol_kc_4 + 1;
        select count(*) --
        from kontstate ks
        inner join kontract k on k.k_id = ks.ks_from_k_id and
              k.k_fact_disb_lo_id = :lo_id
        where ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1 and -- говорит о том что кредит активный
              ks.ks_date_begin between :rep_date_begin and :rep_date_end
        into kol_disb_tmp;
        if (kol_disb_tmp >= 44) then
          :disb_44 = :disb_44 + 1;
      end

      select sum(oks_balance), sum(iif(:rep_date_end - oks_delinq_date > 30, oks_balance, 0))
      from oksummary, region_list(:region_id, 1, 3, 1), kontract, product
      where cast(oks_region_id as integer) = id_list and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            oks_balance > 0 and
            oks_close_state = 0
      into reg_balance_end, reg_par30_end;

      koef_par30 = (:reg_par30_end / :reg_balance_end) * 100;
      --  KPI1  коэфф пар30
      kpi1 = case
               when koef_par30 <= 0.5 then 1.2
               when koef_par30 <= 1 then 1.0
               when koef_par30 <= 1.5 then 0.5
               else 0
             end;

      --  KPI2    коэфф штат
      kpi2 = plan_spf - kol_kc;
      kpi2 = case
               when kpi2 <= 0 then 1.0
               when kpi2 <= 2 then 0.5
               else 0
             end;
      --  KPI3    коэфф выдачи

      kpi3 = :disb_44 / :kol_kc_4;
      --   kpi3 = :kpi3 / :kol_kc_4;
      kpi3 = case
               when kpi3 < 0.49 then 0
               when kpi3 <= 0.59 then 0.5
               when kpi3 <= 0.75 then 0.75
               when kpi3 <= 0.89 then 1.0
               when kpi3 >= 0.9 then 1.2
             end;

      --  Совокупный KPI
      cumulative_kpi = kpi1 * 0.3 + kpi2 * 0.3 + kpi3 * 0.4;

      --  Бонус РМ к начислению
      if (basic_bonus < 0) then
        bonus_rm = 0;
      else
        bonus_rm = round(cast(basic_bonus as double precision) * (kpi1 * 0.3 + kpi2 * 0.3 + kpi3 * 0.4), 0);

      if (bonus_rm < 0) then
        bonus_rm = 0;
      ----------------------------------------------------------------------------------
      bonus_rm_comment = 0;
      bonus_memo = '';
      bonus_koef = 0;

      select bonus_koef, bonus_memo
      from hp_get_vigovor_koef(:c_id, :rep_date_end)
      where bonus_memo <> 'Нет замечаний и выговоров'
      into bonus_koef, bonus_memo;

      if (bonus_rm > 0 and
          bonus_memo <> '') then
      begin
        bonus_rm_comment = bonus_rm - 100;
        if (bonus_rm_comment < 0) then
          bonus_rm_comment = 0;
      end
      aarep_bonus_memo = bonus_memo;
      if (bonus_memo = '') then
        bonus_rm_comment = bonus_rm;
      suspend;
    end
  end
end
