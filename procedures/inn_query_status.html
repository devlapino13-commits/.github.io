CREATE PROCEDURE INN_QUERY_STATUS(
    C_ID D_ID)
RETURNS (
    CAN_QUERY_INN D_BOOL,
    QUERY_STATUS_STR D_STR_100,
    QUERY_STATUS_COLOR D_STR_20,
    INN_RECEIVED D_BOOL,
    INN D_STR_20,
    IS_FINISHED D_SMALLINT,
    INN_CLI_IS_EQUAL D_BOOL,
    IS_AUTO_QUERYS_FINISHED D_BOOL)
AS
declare variable ICM_INN_RECEIVED D_SMALLINT;
declare variable IS_CLOSE_TASK D_BOOL;
declare variable RD_ID D_INTEGER;
declare variable ACTIVE_TASK_MANUAL_CHECK D_BOOL;
declare variable COUNT_QUERY D_INTEGER;
declare variable COUNT_FINESHED_QUERY D_INTEGER;
declare variable ACL_STATUS D_INTEGER;
begin
    /*
        можно ли делать запрос по ИНН да/нет
        Нет:
        1)  если уже получен ИНН по новой схеме
        2)  eсли имеющийся запрос еще не обработан
    */
    can_query_inn = 0;
    is_close_task = 0;
    is_auto_querys_finished = 0;

    query_status_color = 'black';
    rd_id = f_get_rd_id('Actiongroup.manual_inn_check');
    inn_cli_is_equal = 0;

    select acl_status
    from hp_get_actionlist_task_state('ActionGroup.Manual_Inn_Check', 0, :c_id)
    into acl_status;

    --Если на текущую дату была закрыта задача на ручной поиск ИНН RFCRU-3264
    if(exists (select *
               from actionlist acl
               where acl.acl_from_rd_id = :rd_id
                 and acl.acl_status = 2 -- Подтвежден
                 and acl.acl_object_kind = 0
                 and cast(acl.acl_last_datetime as date) = current_date -- на текущую дату
                 and acl.acl_object_id = :c_id)) then
        is_close_task = 1;


    active_task_manual_check = 0;
    if (is_close_task = 0) then
    begin
        if(exists (select *
                   from actionorder ao
                   where ao.aco_from_rd_id = :rd_id
                     and ao.aco_object_kind = 0
                     and ao.aco_respons_c_id = :c_id)) then
            active_task_manual_check = 1;
    end

    select t.icm_is_finished,
           t.icm_inn_received,
           t.icm_inn
    from inn_check_main t
    where t.icm_from_c_id = :c_id
    into :is_finished,
         :icm_inn_received,
         :inn;

    inn_received = coalesce(icm_inn_received, 0);

     -- Проверка статуса запросов по вариативный поиск ИНН
     select count(*), --Кол-во запросов
            sum(case when icq.icq_status <> 0 then 1 else 0 end) --Кол-во завершенных запросов
     from inn_check_queries icq
     where icq.icq_from_c_id = :c_id
       and icq.icq_creation_stage = f_enum_id('inn_request_source', 'from_auto_query')
     into count_query,
          count_fineshed_query;

    if (is_finished is null) then
    begin
        is_finished = 0;
        icm_inn_received = 0;

        can_query_inn = 1;
        query_status_str = 'Запросов еще не было';
    end
    else if (icm_inn_received = 0) then
    begin
        if (is_finished = 0 or acl_status in (0, 1)) then -- Запросы еще не обработались ИЛИ идет поиск андеррайтерами вручную
        begin
            can_query_inn = 0;  -- < -- Важно!
            query_status_str = 'Запрос ИНН в процессе. Ожидайте';
        end
        else if(is_close_task = 1) then
        begin
            can_query_inn = 0;
            query_status_str = 'Сегодня уже была закрыта задача на ручной поиск ИНН';
            query_status_color = 'red';
        end
        else if (is_finished = 1) then
        begin
            if (active_task_manual_check = 0) then
            begin
                can_query_inn = 1;
                query_status_str = 'ИНН не найден. Приложите скриншоты с ИНН и отправьте на ручную обработку';
                query_status_color = 'red';
            end
            else
            begin
                can_query_inn = 0;
                query_status_str = 'Запрос на ручной поиск ИНН находится в обработке у КК';
                query_status_color = 'red';                
            end
        end
    end
    else if (icm_inn_received = 1) then
    begin
        can_query_inn = 0;
        query_status_str = 'Запрос обработан. ИНН получен';
        query_status_color = 'green';
    end

    if (icm_inn_received = 1) then
    begin
        if ((select cl.c_inn from clients cl where cl.c_id = :c_id) = :inn) then
            inn_cli_is_equal = 1;
    end

    --После того как все задачи по вариативному поиску обработались И ИНН не был найден
    if(count_query = count_fineshed_query and icm_inn_received = 0) then
        is_auto_querys_finished = 1;

    suspend;
end
