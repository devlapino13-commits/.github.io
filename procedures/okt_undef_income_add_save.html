CREATE PROCEDURE OKT_UNDEF_INCOME_ADD_SAVE(
    UN_ID D_ID,
    PHONE D_STR_20,
    MEMO D_STR_8100,
    K_ID D_ID,
    OBJECT_KIND D_STR_20,
    OBJECT_ID D_ID,
    PANT_ID D_ID,
    CLIENT_CELL_NO_ADD_NEW D_STR_100,
    CONTACT_CELL_NO_NEW D_STR_100)
AS
declare variable oui_id d_integer;
begin
  insert into okt_undef_income (oui_phone, oui_from_un_id, oui_memo)
  values (:phone, :un_id, :memo)
  returning oui_id
  into :oui_id;

  insert into okt_undef_income_add_data (oia_from_oui_id, oia_from_k_id, oia_object_kind, oia_object_id,
                                         oia_client_cell_no_add_new, oia_contact_cell_no_new, oia_from_pant_id)
  values (:oui_id, :k_id, f_objkind(:object_kind, 'id'), :object_id, :client_cell_no_add_new,
          :contact_cell_no_new, :pant_id);
end
