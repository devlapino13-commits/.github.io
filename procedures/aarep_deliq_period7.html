CREATE PROCEDURE AAREP_DELIQ_PERIOD7(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_CURRENCY VARCHAR(3))
RETURNS (
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    K_KOLICH_REPAY INTEGER,
    PR_NAME VARCHAR(50),
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    KAL_NEW_KF_NAME VARCHAR(50),
    DELIQ_OS_BEGIN DECIMAL(18,4),
    DELIQ_PERC_BEGIN DECIMAL(18,4),
    DELIQ_DAY_BEGIN INTEGER,
    DELIQ_BALANCE_BEGIN DECIMAL(18,4),
    DELIQ_OS_END DECIMAL(18,4),
    DELIQ_PERC_END DECIMAL(18,4),
    DELIQ_DAY_END INTEGER,
    DELIQ_BALANCE_END DECIMAL(18,4),
    PR_LOAN_TYPE_STR VARCHAR(10),
    DIFFERENCE_BALANCE DECIMAL(18,4))
AS
begin
  for select k_id, k_kontract_no, k_disb_date, k_amount_approved, k_circle, k_circle_internal, k_kolich_repay, pr_name,
             reg_name, parent_name, kal_new_kf_name, deliq_os_begin, deliq_perc_begin, deliq_day_begin,
             deliq_balance_begin, deliq_os_end, deliq_perc_end, deliq_day_end, deliq_balance_end, pr_loan_type_str
      from aarep_deliq_period(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_currency)
      where deliq_day_begin > 7 or
            (deliq_day_end > 7)
      into :k_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :k_circle, :k_circle_internal, :k_kolich_repay,
           :pr_name, :reg_name, :parent_name, :kal_new_kf_name, :deliq_os_begin, :deliq_perc_begin, :deliq_day_begin,
           :deliq_balance_begin, :deliq_os_end, :deliq_perc_end, :deliq_day_end, :deliq_balance_end, :pr_loan_type_str
  do
  begin
    if (:deliq_balance_begin is null) then
      deliq_balance_begin = 0;
    if (:deliq_balance_end is null) then
      deliq_balance_end = 0;

    if (:deliq_day_begin > 7 and
        :deliq_day_end <= 7) then
    begin
      deliq_os_end = 0;
      deliq_perc_end = 0;
      deliq_day_end = 0;
      deliq_balance_end = 0;
    end

    if (:deliq_day_begin <= 7 and
        :deliq_day_end > 7) then
    begin
      deliq_os_begin = 0;
      deliq_perc_begin = 0;
      deliq_day_begin = 0;
      deliq_balance_begin = 0;
    end

    difference_balance = :deliq_balance_end - :deliq_balance_begin;

    suspend;
  end
end
