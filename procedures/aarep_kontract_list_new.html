    REP_COLL_KOEF_PERC DECIMAL(18,7),
    K_PERCENT_EFFECT DECIMAL(18,7),
    K_PERCENT_OSTATOK DECIMAL(18,7))
AS
declare variable pr_is_secured integer;
declare variable pr_int_calc_type integer;
declare variable k_is_kredline integer;
declare variable perc_gr decimal(18,7);
declare variable perc_ind decimal(18,7);
declare variable pr_is_employer integer;
declare variable aarep_disb_ekv decimal(18,4);
declare variable curr_rate decimal(18,4);
declare variable k_percent_input decimal(18,7);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  k_percent_ostatok = 0;
  for select ak.aarep_k_id, k_from_kf_id, k_currency, k_applicate_date, k_applicate_no, k_kontract_no, k_kontract_date,
             k_from_pr_id, k_disb_date, k_amount_applicate, k_stage, k_period, k_kolich_lgota, k_percent,
             k_percent_lgota, k_kolich_repay, k_from_f_id, k_from_pt_id, k_circle, k_circle_internal, k_percent_per_day,
             k_memo, k_amount_approved, k_index_curr_rate, k_percent_for_days, k_kolich_client_applicate,
             k_kolich_client_approved, k_repay_kolich_closed, k_class, k_kommis_perc, k_from_cs_id, k_credit_target,
             k_is_kredline, k_kredline_end, k_percent_effect, kakt1.kal_new_kf_name, kakt1.kal_new_no, r1.reg_unicode,
             r1.reg_name, r2.reg_unicode, r2.reg_name, kakt1.kal_new_kf_officer, kakt1.kal_new_officer_code,
             c1.c_name_fio, kakt1.kal_date, kakt1.kal_date_end, kakt2.kal_date, kf_memo, ak.aarep_k_balance,
             ak.aarep_disb_ekv, ak.aarep_k_closed_date, ak.aarep_k_deliq_os, ak.aarep_k_deliq_perc,
             ak.aarep_k_deliq_days, ak.aarep_k_deliq_repays, ak.aarep_k_deliq_begin_date, ak.aarep_date_action,
             ak.aarep_plan_os, ak.aarep_plan_perc, ak.aarep_plan_all, ak.aarep_plan_repay_no, ak.aarep_plan_repay_date,
             ak.aarep_deposit, ak.aarep_zalog_name, ak.aarep_zalog, ak.aarep_zalog_plus, ak.aarep_zalog_minus,
             ak.aarep_coll_plan_os, ak.aarep_coll_fact_os, ak.aarep_coll_koef_perc, k_clients_leader_id, c2.c_name_fio,
             c2.c_pass_no, c2.c_pass_date, c2.c_pass_who, k_clients_cashier_id, c3.c_name_fio, c3.c_pass_no,
             c3.c_pass_date, c3.c_pass_who, pr_code, pr_name, pr_loan_type, pr_is_employer, pr_is_secured,
             pr_interest_method, pr_percent_for_days, class1.cs_id, class1.cs_name, class1.cs_llr_kind_text,
             class1.cs_is_for_group, class1.cs_is_for_indiv, class2.cs_id, class2.cs_name, class2.cs_llr_kind_text,
             lk1.lk_percent, k_percent_input
      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, :report_id) ak
      left join(llc_kont lk1 left join classific class2 on (lk1.lk_status = class2.cs_id)) on (ak.aarep_k_id = lk1.lk_from_k_id and
            lk1.lk_date <= ak.aarep_date_action and
            lk1.lk_date_end >= ak.aarep_date_action)
      left join(knaktlist kakt1 left join clients c1 on (kakt1.kal_new_kf_officer = c1.c_id) left join(region r1 left join region r2 on (r1.parent = r2.reg_unicode)) on (kakt1.kal_new_reg_id = r1.reg_unicode)) on (ak.aarep_kf_id = kakt1.kal_from_kf_id and
            kakt1.kal_date <= ak.aarep_date_action and
            kakt1.kal_date_end >= ak.aarep_date_action)

      left join(kontract left join clients c2 on (kontract.k_clients_leader_id = c2.c_id) left join clients c3 on (kontract.k_clients_cashier_id = c3.c_id) left join product on (kontract.k_from_pr_id = pr_id) left join kfolder on (kontract.k_from_kf_id = kf_id) left join classific class1 on (kontract.k_from_cs_id = class1.cs_id) left join knaktlist kakt2 on kontract.k_from_kf_id = kakt2.kal_from_kf_id and
            kakt2.kal_from_ka_id = 0 and
            kakt2.kal_number = 1) on (k_id = aarep_k_id)

      into :k_id, :k_from_kf_id, :k_currency, :k_applicate_date, :k_applicate_no, :k_kontract_no, :k_kontract_date,
           :k_from_pr_id, :k_disb_date, :k_amount_applicate, :k_stage, :k_period, :k_kolich_lgota, :k_percent,
           :k_percent_lgota, :k_kolich_repay, :k_from_f_id, :k_from_pt_id, :k_circle, :k_circle_internal,
           :k_percent_per_day, :k_memo, :k_amount_approved, :curr_rate, :k_percent_for_days, :k_kolich_client_applicate,
           :k_kolich_client_approved, :k_repay_kolich_closed, :k_risk_category, :k_kommis_perc, :k_from_cs_id,
           :k_credit_target, :k_is_kredline, :k_kredline_end, :k_percent_effect, :kf_name, :kf_number, :kf_region_id,
           :kf_region_name, :kf_branch_id, :kf_branch_name, :kf_officer_id, :kf_officer_code, :kf_officer_name,
           :kf_date_in_to_krs, :kf_date_out_from_krs, :kf_date_create, :kf_memo, :rep_active_balance, :aarep_disb_ekv,
           :rep_date_closed, :rep_deliq_os, :rep_deliq_perc, :rep_deliq_days_kolich, :rep_deliq_repays_str,
           :rep_deliq_begin_date, :rep_date_action, :rep_plan_os, :rep_plan_perc, :rep_plan_all, :rep_plan_repay_no,
           :rep_plan_repay_date, :rep_deposit, :rep_zalog_name, :rep_zalog, :rep_zalog_plus, :rep_zalog_minus,
           :rep_coll_plan_os, :rep_coll_fact_os, :rep_coll_koef_perc, :cl_leader_id, :cl_leader_name,
           :cl_leader_pass_no, :cl_leader_pass_date, :cl_leader_pass_who, :cl_cashier_id, :cl_cashier_name,
           :cl_cashier_pass_no, :cl_cashier_pass_date, :cl_cashier_pass_who, :pr_code, :pr_name, :pr_loan_type,
           :pr_is_employer, :pr_is_secured, :pr_int_calc_type, :pr_int_for_days, :llc_begin_status,
           :llc_begin_status_str, :llc_begin_reserve, :perc_gr, :perc_ind, :llc_current_status, :llc_current_status_str,
           :llc_current_reserve, :llc_current_percent, :k_percent_input
  do
  begin
    cl_leader_full_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_address_type = 1 and
          adc_from_c_id = :cl_leader_id and
          adc_from_adl_id = adl_id
    into :cl_leader_full_address;

    cl_cashier_full_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_address_type = 1 and
          adc_from_c_id = :cl_cashier_id and
          adc_from_adl_id = adl_id
    into :cl_cashier_full_address;

    k_percent_ostatok = 0;
    if (pr_loan_type = 0) then
      llc_begin_percent = :perc_gr;
    else
      llc_begin_percent = :perc_ind;
    if (report_currency = '000') then
    begin
      k_amount_approved = :k_amount_approved * :curr_rate;
    end
    if (report_id = 2) then
    begin
      k_disb_date = rep_date_action;
      k_amount_approved = rep_plan_os;
      rep_plan_os = 0;
    end
    if (report_id = 3) then
    begin
      if (k_percent_input = :k_percent) then
      begin
        select result_perc
        from aarep_float_to_annuitet(:k_percent, :k_kolich_repay)
        into :k_percent_ostatok;
      end
      else
      begin
        k_percent_ostatok = cast(:k_percent as decimal(15,4));
      end
    end
    k_disb_year = extract(year from k_disb_date);
    k_disb_month = extract(month from k_disb_date);

    pr_loan_type_str = '';
    if (pr_loan_type = 0) then
      pr_loan_type_str = 'Группа';
    if (pr_loan_type = 1) then
      pr_loan_type_str = 'Индив.';

    pr_is_employer_str = '';
    if (pr_is_employer = 0) then
      pr_is_employer_str = 'Для клиент.';
    if (pr_is_employer = 1) then
      pr_is_employer_str = 'Для сотрудн.';

    if (k_is_kredline = 0) then
      k_is_kredline_str = 'Нет';
    if (k_is_kredline = 1) then
      k_is_kredline_str = 'Да';

    pr_is_secured_str = '';
    if (pr_is_secured = 0) then
      pr_is_secured_str = 'Без залога';
    if (pr_is_secured = 1) then
      pr_is_secured_str = 'Залог';
    if (pr_is_secured = 2) then
      pr_is_secured_str = 'Поручители';
    if (pr_is_secured = 3) then
      pr_is_secured_str = 'Залог,поручит.';

    pr_int_calc_type_str = '';
    if (pr_int_calc_type = 0) then
      pr_int_calc_type_str = 'На сумму кредита';
    if (pr_int_calc_type = 1) then
      pr_int_calc_type_str = 'На остаток';
    if (pr_int_calc_type = 2) then
      pr_int_calc_type_str = 'Аннуитет';

    if (llc_current_status is null) then
      llc_current_status = llc_begin_status;
    if (llc_current_status_str is null) then
      llc_current_status_str = llc_begin_status_str;
    if (llc_current_reserve is null) then
      llc_current_reserve = llc_begin_reserve;
    if (llc_current_percent is null) then
      llc_current_percent = llc_begin_percent;

    k_circle_real = :k_circle + :k_circle_internal - 1;

    if (rep_plan_repay_no = k_kolich_repay) then
      rep_plan_repay_is_last = 'Посл.плат.';
    else
      rep_plan_repay_is_last = '';
    llc_current_amount = (:rep_active_balance * :llc_current_percent) / 100;
    llc_current_amount = cast(llc_current_amount as decimal(18,2));
    k_percent_real = :k_percent - :k_percent_lgota;
    suspend;
  end

end
