CREATE PROCEDURE AAREP_EDOC_LIST(
    REP_REGION_ID D_INTEGER = 1000000,
    REP_DATE_BEGIN D_DATE = current_date,
    REP_DATE_END D_DATE = current_date)
RETURNS (
    EDL_ID D_ID,
    REG_NAME D_STR_CODE,
    C_NAME_FIO D_STR_100,
    EDC_NAME D_STR_100,
    EDL_BLANK_DATE D_DATE,
    EDL_CREATE_DATETIME D_DATETIME,
    EDS_STATUS_DATETIME D_DATETIME,
    EDS_STATUS_STR D_STR_CODE,
    EDL_STATUS_STR D_STR_CODE,
    UN_FIO D_STR_100)
AS
declare variable C_ID D_ID;
declare variable REG_ID D_INTEGER;
declare variable REG_LST D_STR_8100;
declare variable IS_FREELANCER boolean;
begin
  reg_lst = '';
  select
    list(id_list, ';')
  from region_list(:rep_region_id, 1, 3, 1)
  into
    :reg_lst;

  for
    select
      edl_id,
      c_id,
      c_name_fio,
      edc_name,
      edl_blank_date,
      edl_create_datetime,
      eds_status_datetime,
      trim(
        case eds_status
          when 0 then 'не подписан'
          when 1 then 'подписан'
        else '' end),
      trim(
        case edl_status
          when 0 then 'не подписан'
          when 1 then 'подписан'
          when -1 then 'аннулирован'
        else '' end) || iif(coalesce(edl_date_end, current_date) < current_date, ', срок истёк', ''),
      un_fio,
      edc_code in (
        'probationer_kontract',
        'kochegar_kontract',
        'freelancer_kontract',
        'add_for_freelancers',
        'termination_freelancer',
        'probationer_kontract',
        'freelancer_akt')
    from edoc_list
      inner join edoc_cfg
        on edc_id = edl_from_edc_id
      inner join edoc_sign
        on eds_from_edl_id = edl_id
          and edc_code in (
            'prof_cust.agreement',
            'gpsa.agreement',
            'gpse.agreement',
            'gpsi.agreement',
            'tsp.agreement',
            'installment_goods.agreement',
            'auto.agreement',

            'act.reconciliation',  ---RFCKG-4747
            'probationer_kontract',
            'kochegar_kontract',
            'freelancer_kontract',
            'add_for_freelancers',
            'termination_freelancer',
            'probationer_kontract',
            'freelancer_akt')
      inner join clients
        on c_id = eds_from_c_id
      left join username
        on un_id = edl_create_un_id
    where edl_blank_date between :rep_date_begin and :rep_date_end
    order by c_name_fio, eds_status_datetime desc
  into
    :edl_id,
    :c_id,
    :c_name_fio,
    :edc_name,
    :edl_blank_date,
    :edl_create_datetime,
    :eds_status_datetime,
    :eds_status_str,
    :edl_status_str,
    :un_fio,
    :is_freelancer
  do
  begin
    if (is_freelancer) then
      reg_id = (
        select first 1 sd_from_reg_id
        from servicedog
        where sd_from_c_id = :c_id
        order by sd_id desc);
    else
      reg_id = (
        select first 1 ag_from_init_reg_id
        from agreements
        where ag_object_kind = 0
          and ag_object_id = :c_id
          order by ag_id desc);

    if (reg_lst containing reg_id) then
    begin
      reg_name = (
        select reg_code || ' ' || reg_name
        from region
        where reg_unicode = :reg_id);
      suspend;
    end
  end
end
