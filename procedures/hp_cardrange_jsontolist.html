CREATE PROCEDURE HP_CARDRANGE_JSONTOLIST(
    CARD_LIST D_STR_8100)
RETURNS (
    FIRST_CARD_ID D_INTEGER,
    LAST_CARD_ID D_INTEGER,
    CARD_COUNT D_INTEGER,
    IS_PINCOVER D_BOOL,
    ID D_INTEGER)
AS
declare variable n d_integer;
declare variable len_first d_integer;
declare variable len_last d_integer;
begin
  id = 0;
  n = f_json_str_get_count(:card_list, 'cards');
  while (id < n) do
  begin
    first_card_id = f_json_str_get_value(:card_list, 'cards[' || :id || '].b', null);
    last_card_id = f_json_str_get_value(:card_list, 'cards[' || :id || '].e', null);

    len_first = char_length(first_card_id);
    len_last = char_length(last_card_id);

    if (first_card_id is null or last_card_id is null) then
      exception e_default 'Элемент массива не найден.';
    if (len_first < 4 or len_last < 4) then
      exception e_default 'Слишком короткий номер.';

    card_count = cast(left(last_card_id, len_last - 1) as d_integer) -
      cast(left(first_card_id, len_first - 1) as d_integer) + 1;

    id = id + 1;

    is_pincover = (
    select
      cem_is_pincover
    from cardlist
      inner join cardembomain
        on cem_id = cal_from_cem_id
    where cal_card_id = :first_card_id);

    suspend;
  end
end
