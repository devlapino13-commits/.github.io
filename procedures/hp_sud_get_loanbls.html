CREATE PROCEDURE HP_SUD_GET_LOANBLS(
    SLD_ID_IN D_ID,
    SHOW_DELETED D_BOOL)
RETURNS (
    WHERE_FROM_STR D_STR_CODE,
    SLB_ID D_ID,
    BLS_ID D_ID,
    IS_DELETE D_BOOL,
    CREATOR D_STR_LONG,
    REMOVER D_STR_LONG,
    EDB_ID D_ID)
AS
declare variable object_kind d_str_code;
  declare variable object_id d_id;
begin
  if (show_deleted is null) then
    show_deleted = 0;

  for select slb_id, slb_from_bls_id, slb_is_delete,
         coalesce((select un_fio from username where un_id = slb_from_un_id)||' ', '') || f_datetimetostr(slb_datetime),
         coalesce((select un_fio from username where un_id = slb_delete_un_id)||' ', '') || f_datetimetostr(slb_delete_datetime),
         slb_object_kind, slb_object_id
  from sud_loanbls
  where slb_from_sld_id = :sld_id_in
    and (:show_deleted = 1 or slb_is_delete = 0)
  into slb_id, bls_id, is_delete,  creator, remover, object_kind, object_id
  do
  begin
    where_from_str = case f_objkind(object_kind, 'code')
                        when 'edoc_blank' then 'Электронный документ'
                        when 'clientdocs' then 'Документ клиента'
                        when 'sud_loandocs' then 'Прикреплен вручную'
                        else f_objkind(object_kind, 'code')
                     end;

    edb_id = null;

    if (bls_id is null and f_objkind(object_kind, 'code') = 'edoc_blank') then
      edb_id = object_id;

    suspend;
  end
end
