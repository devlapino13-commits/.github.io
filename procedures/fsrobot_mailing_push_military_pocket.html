CREATE PROCEDURE FSROBOT_MAILING_PUSH_MILITARY_POCKET(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable GOOGLE_ID D_STR_255;
declare variable TMP_KONTRACT_LIST D_STR_LONG;
declare variable C_NAME_FIO D_STR_100;
declare variable C_ID D_ID;
declare variable KF_ID D_ID;
declare variable MCP_ID D_INTEGER;
declare variable LO_C_ID D_ID;
begin
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID  - пользователь

  mcp_id = cast(rep_params as d_integer);

  for select mcq.mcq_from_c_id
  from military_check_query mcq
  where mcq.mcq_from_mcp_id = :mcp_id and
        mcq.mcq_status = 1 -- берем только призывников
  into c_id
  do
  begin
    tmp_kontract_list = '';
    lo_c_id = 0;
    google_id = '';
    kf_id = 0;

    select list(t.k_kontract_no, ', ')
    from (
        select k.k_kontract_no
        from knclient kc
        inner join kontstate ks on kc.kc_from_k_id = ks.ks_from_k_id
        inner join kontract k on k.k_id = ks.ks_from_k_id
        where ks.ks_ol_is_delete = 0
          and ks.ks_state = 1
          and current_date - 1 between ks.ks_date_begin and ks.ks_date_end
          and kc.kc_amount_approved > 0
          and kc.kc_from_c_id = :c_id
        order by k.k_id
    ) t
    into :tmp_kontract_list;

    if (coalesce(:tmp_kontract_list, '') <> '') then
    begin
      select c.c_name_fio
      from clients c
      where c.c_id = :c_id
      into c_name_fio;

      select k.k_from_kf_id
        from knclient kc
        inner join kontstate ks on kc.kc_from_k_id = ks.ks_from_k_id
        inner join kontract k on k.k_id = ks.ks_from_k_id
        where ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1 and
              current_date - 1 between ks.ks_date_begin and ks.ks_date_end and
              kc.kc_amount_approved > 0 and
              kc.kc_from_c_id = :c_id
        order by k.k_id desc
        rows 1
       into kf_id;

       select kal.kal_new_kf_officer
       from knaktlist kal
       where kal.kal_from_kf_id = :kf_id and
             kal.is_delete = 0 and
             kal_date_end >= '31.12.2999'
       into lo_c_id;

       if (coalesce(lo_c_id, 0) <> 0) then
       begin
         select un_google_id
         from username un
         where un.un_from_c_id = :lo_c_id
         into google_id;
       end

       if (coalesce(google_id, '') <> '') then
       begin
         res_maillist = google_id;
         res_subject = f_format('Клиент %1 призван на срочную воинскую службу', :c_name_fio);
         suspend;
       end
    end
  end
end
