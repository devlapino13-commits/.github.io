CREATE PROCEDURE HP_DO_KN_OPERBOOK_ONE_LOAN_DATE(
    K_ID ID_TYPE,
    OL_DATE D_DATE,
    OL_DATE_PRV D_DATE,
    LOAN_OL_ID D_ID)
AS
declare variable un_id id_type;
declare variable ol_id id_type;
declare variable disb_ol_id id_type;
declare variable oks_date_begin date;
declare variable doc_reg_id integer;
declare variable reg_id integer;
declare variable doc_ol_id decimal(18,0);
declare variable ol_is_active integer;
declare variable disb_ol_is_active integer;
declare variable kc_id id_type;
declare variable loan_reg_id integer;
declare variable disb_reg_id integer;
declare variable knakt_date date;
declare variable oek_rd_id integer;
declare variable kob_id d_id;
declare variable ol_oper_subtype d_integer;
declare variable is_fact d_smallint;
declare variable total_amount d_currency;
declare variable orig_reg_id d_integer;
declare variable orp_ol_date d_date;
declare variable is_disb d_bool;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  un_id = 1000000000001;

  select orp_ol_date
  from operrepeat, operlist ol
  where orp_from_k_id = :k_id and
        orp_ol_date >= :ol_date and
        orp_from_ol_id = ol.ol_id and
        ol.is_delete = 0
  rows 1
  into orp_ol_date;

  if (orp_ol_date is not null) then
    exception e_default 'Reverse OperRepeat found: ' || f_datetostr(orp_ol_date);

  if (exists(select *
  from operfraud, operlist ol
  where ofr_from_k_id = :k_id and
        ofr_storno_date >= :ol_date and
        ofr_storno_ol_id = ol.ol_id and
        ol.is_delete = 0)) then
   Exit;

  select kal_new_reg_id
  from knaktlist, kontract
  where kal_from_kf_id = k_from_kf_id and
        k_id = :k_id and
        kal_date <= :ol_date and
        kal_date_end >= :ol_date and
        knaktlist.is_delete = 0
  into :orig_reg_id;

  reg_id = orig_reg_id;

  if (loan_ol_id is null and
      ol_date <= cast('28.02.2019' as date)) then -- начиная с марта 2019 года создаем единый OL_ID
  begin
    for select distinct kob_from_ol_id
        from kn_operbook, knclient
        where kc_from_k_id = :k_id and
              kc_id = kob_from_kc_id and
              kob_ol_date > :ol_date_prv and
              kob_ol_is_active = 1
        into :ol_id
    do
    begin
      if (exists(select *
                 from operlist
                 where ol_id = :ol_id and
                       ol_oper_type = 2 and
                       ol_oper_subtype in (30, 31, 32))) then
      begin
        delete from operlist
        where ol_id = :ol_id;
      end
    end

    for select kc_id
        from knclient
        where kc_from_k_id = :k_id
        into :kc_id
    do
    begin
      delete from kn_operbook
      where kob_from_kc_id = :kc_id and
            kob_ol_date > :ol_date_prv and
            kob_ol_is_active = 1;
    end
  end

  --  обработка на уровне участников
  for select kc_id
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into :kc_id
  do
  begin
    if (loan_ol_id is null) then
    begin
      if (ol_date >= cast('01.03.2019' as date)) then
      begin
        for select distinct kob_from_ol_id, ol_oper_subtype
            from kn_operbook, operlist
            where kob_from_kc_id = :kc_id and
                  kob_from_ol_id = ol_id and
                  kob_ol_date between(:ol_date_prv + 1) and cast('31.12.9999' as date) and
                  kob_ol_is_active = 1 and
                  ol_oper_type = 2 and
                  ol_oper_subtype in (31, 32, 33, 34)
            into ol_id, ol_oper_subtype
        do
        begin
          if (ol_oper_subtype = 31) then
          begin
            -- проводки по смене резиденства
            delete from operlist
            where ol_id = :ol_id;
          end
          else
          begin
            -- начисления: процент, резерв и корректировка ВБС
            delete from kn_operbook
            where kob_from_ol_id = :ol_id and
                  kob_from_kc_id = :kc_id;
          end
        end

        for select kob_id
            from kn_operbook, operlist
            where kob_from_ol_id = ol_id and
                  not(ol_oper_type = 2 and
                  ol_oper_subtype in (32, 33, 34)) and
                  kob_from_kc_id = :kc_id and
                  kob_ol_date between(:ol_date_prv + 1) and cast('31.12.9999' as date) and
                  kob_ol_is_active = 1
            order by kob_from_ol_id desc, kob_id desc
            into kob_id
        do
        begin
          delete from kn_operbook
          where kob_id = :kob_id;
        end
      end

      -- доначисляем с последней операции
      knakt_date = null;
      for select oks_date_begin,
                 iif(oks_is_pmt = 1 or oks_is_month = 1 or oks_date_begin <= cast('28.02.2019' as date), 1, 0)
          from kn_oksummary
          where oks_from_kc_id = :kc_id and
                oks_date_begin between(:ol_date_prv + 1) and (:ol_date - 1)
          order by oks_from_kc_id, oks_date_begin
          into oks_date_begin, is_fact
      do
      begin
        if (knakt_date is null or
            knakt_date <> oks_date_begin) then
        begin
          select kal_new_reg_id
          from knaktlist, kontract
          where kal_from_kf_id = k_from_kf_id and
                k_id = :k_id and
                kal_date <= :oks_date_begin and
                kal_date_end >= :oks_date_begin and
                knaktlist.is_delete = 0
          into :reg_id;
        end

        execute procedure hp_do_kn_operbook_kont_perc(:kc_id, :oks_date_begin, :reg_id, 1, :un_id);

        if (is_fact = 1) then
        begin
          execute procedure hp_do_kn_operbook_kont_rsv(:kc_id, :oks_date_begin, :reg_id, 1, :un_id);
          execute procedure hp_do_kn_operbook_kont_vbs(:kc_id, :oks_date_begin, :reg_id, 1, :un_id);
        end

        knakt_date = oks_date_begin;
      end
    end else
    begin
      for select kob_id
          from kn_operbook, operlist
          where kob_from_ol_id = ol_id and
                ol_oper_type = 2 and
                ol_oper_subtype in (32, 33, 34) and
                kob_from_kc_id = :kc_id and
                kob_ol_date = :ol_date and
                kob_ol_is_active = 1
          order by kob_from_ol_id desc, kob_id desc
          into kob_id
      do
      begin
        delete from kn_operbook
        where kob_id = :kob_id;
      end
    end

    is_fact = 0;

    if (exists(select *
               from kn_oksummary
               where oks_from_kc_id = :kc_id and
                     oks_date_begin = :ol_date and
                     oks_is_pmt = 1 and
                     oks_close_state <> 2)) then
      is_fact = 1;

    disb_ol_id = null;

    is_disb = 0;
    for select distinct ok_from_ol_id, ok_ol_reg_id, ol_parent_id, ok_ol_is_active, iif(ok_os > 0, 1, 0)
        from kn_operkred, operlist
        where ok_from_kc_id = :kc_id and
              ok_type = 8 and
              ok_os <> 0 and
              ol_id = ok_from_ol_id and
              ok_ol_date = :ol_date and
              (:loan_ol_id is null or
              cast(ol_id as d_id) = :loan_ol_id) and
              ok_ol_is_active = 1
        order by ok_from_ol_id
        into ol_id, doc_reg_id, doc_ol_id, ol_is_active, is_disb
    do
    begin
      is_fact = 1;

      if (is_disb = 1) then
        disb_ol_id = ol_id;

      disb_reg_id = doc_reg_id;
      disb_ol_is_active = ol_is_active;

      if (doc_ol_id is null) then
        doc_ol_id = ol_id;
      else
      begin
        select ol_from_reg_id
        from operlist
        where ol_id = :doc_ol_id
        into :doc_reg_id;
      end

      execute procedure hp_do_kn_operbook_save(:doc_ol_id, :ol_id, :kc_id, 'disb', :doc_reg_id, :ol_date, :ol_is_active);

    end

    -- доп.процент
    for select distinct ok_from_ol_id, ok_ol_reg_id, ol_parent_id, ok_ol_is_active
        from kn_operkred, operlist
        where ok_from_kc_id = :kc_id and
              ok_kommis < 0 and
              ok_type = 4 and
              ol_id = ok_from_ol_id and
              ok_ol_date = :ol_date and
              (:loan_ol_id is null or
              cast(ol_id as d_id) = :loan_ol_id) and
              ok_ol_is_active = 1
        order by ok_from_ol_id
        into ol_id, doc_reg_id, doc_ol_id, ol_is_active
    do
    begin
      is_fact = 1;
      doc_ol_id = ol_id;

      execute procedure hp_do_kn_operbook_save(:doc_ol_id, :ol_id, :kc_id, 'kommis', :doc_reg_id, :ol_date,
          :ol_is_active);
    end

    -- погашения
    loan_reg_id = orig_reg_id;

    -- Начисляем проценты на начало дня если это не день выдачи
    if (disb_ol_id is null) then
      execute procedure hp_do_kn_operbook_kont_perc(:kc_id, :ol_date, :reg_id, 1, :un_id);

    -- Проводим все погашения за день
    for select distinct ok_from_ol_id, ok_ol_reg_id, ol_parent_id, ok_ol_is_active
        from kn_operkred, operlist
        where ok_from_kc_id = :kc_id and
              ok_ol_date = :ol_date and
              (:loan_ol_id is null or
              cast(ol_id as d_id) = :loan_ol_id) and
              ok_from_ol_id = ol_id and
              (ok_type in (4, 5, 6) or
              (ok_type = 9 and
              ok_os < 0)) and
              (ok_os <> 0 or
              ok_perc <> 0 or
              ok_fine <> 0 or
              ok_kommis_adv <> 0)
        order by ok_from_ol_id
        into :ol_id, :doc_reg_id, :doc_ol_id, :ol_is_active
    do
    begin
      if (doc_ol_id is null) then
        doc_ol_id = ol_id;
      else
      begin
        select ol_from_reg_id
        from operlist
        where ol_id = :doc_ol_id
        into :doc_reg_id;
      end

      if (exists(select *
                 from kn_operkred
                 where (ok_os <> 0 or
                       ok_perc <> 0 or
                       ok_fine <> 0) and
                       ok_type in (5, 6) and
                       ok_from_ol_id = :ol_id and
                       ok_ol_is_active = 1 and
                       ok_from_kc_id = :kc_id)) then
      begin
        is_fact = 1;
        execute procedure hp_do_kn_operbook_save(:doc_ol_id, :ol_id, :kc_id, 'writeoff', :doc_reg_id, :ol_date,
            :ol_is_active);
      end

      if (exists(select *
                 from kn_operkred
                 where (ok_os <> 0 or
                       ok_perc <> 0 or
                       ok_fine <> 0 or
                       ok_kommis_adv <> 0) and
                       ok_type in (4, 9) and
                       ok_from_ol_id = :ol_id and
                       ok_ol_is_active = 1 and
                       ok_from_kc_id = :kc_id)) then
      begin
        is_fact = 1;
        execute procedure hp_do_kn_operbook_save(:doc_ol_id, :ol_id, :kc_id, 'repay', :doc_reg_id, :ol_date,
            :ol_is_active);
      end
    end

    -- возврат переплаты
    for select distinct ok_from_ol_id, ok_ol_reg_id, ol_parent_id, ok_ol_is_active
        from kn_operkred, operlist
        where ok_from_kc_id = :kc_id and
              ok_ol_date = :ol_date and
              ok_ol_is_active = 1 and
              (:loan_ol_id is null or
              cast(ol_id as d_id) = :loan_ol_id) and
              ok_from_ol_id = ol_id and
              ok_type = 9 and
              ok_os > 0
        order by ok_from_ol_id, ok_from_kc_id
        into :ol_id, :doc_reg_id, :doc_ol_id, :ol_is_active
    do
    begin
      total_amount = 0;
      select sum(ok_os + ok_perc + ok_fine + ok_kommis)
      from kn_operkred
      where ok_from_kc_id = :kc_id and
            ok_ol_date = :ol_date and
            ok_from_ol_id = :ol_id and
            ok_type in (4, 9)
      into total_amount;
      if (total_amount is null) then
        total_amount = 0;

      if (total_amount <> 0) then
      begin
        is_fact = 1;

        if (doc_ol_id is null) then
          doc_ol_id = ol_id;
        else
        begin
          select ol_from_reg_id
          from operlist
          where ol_id = :doc_ol_id
          into :doc_reg_id;
        end

        execute procedure hp_do_kn_operbook_save(:doc_ol_id, :ol_id, :kc_id, 'pereplat', :doc_reg_id, :ol_date,
            :ol_is_active);
      end
    end
  end
  -- конец обработки на уровне участников

  -- все погашения по ранее списанным займам
  for select distinct oko_from_ol_id, kc_id, ol_from_reg_id, ol_parent_id, oko_ol_is_active
      from operkroff, operlist, knclient
      where (oko_os <> 0 or
            oko_perc <> 0) and
            oko_type = 4 and
            oko_from_ol_id = ol_id and
            oko_from_c_id = kc_from_c_id and
            oko_from_k_id = kc_from_k_id and
            kc_from_k_id = :k_id and
            oko_ol_date = :ol_date and
            (:loan_ol_id is null or
            cast(ol_id as d_id) = :loan_ol_id)
      order by oko_from_ol_id, kc_id
      into :ol_id, :kc_id, :doc_reg_id, :doc_ol_id, :ol_is_active
  do
  begin
    is_fact = 1;

    if (doc_ol_id is null) then
      doc_ol_id = ol_id;
    else
    begin
      select ol_from_reg_id
      from operlist
      where ol_id = :doc_ol_id
      into :doc_reg_id;
    end

    execute procedure hp_do_kn_operbook_save(:doc_ol_id, :ol_id, :kc_id, 'operkroff', :doc_reg_id, :ol_date,
        :ol_is_active);
  end

  -- все начисления по судебным искам
  for select distinct ox_from_ol_id, ox_from_reg_id, ol_parent_id, ox_ol_is_active
      from operextra, operlist
      where ox_from_rd_id = :oek_rd_id and
            ox_amount_plus <> 0 and
            ox_from_ol_id = ol_id and
            ox_object_kind = 9 and
            ox_object_id = :k_id and
            ox_ol_date = :ol_date and
            (:loan_ol_id is null or
            cast(ol_id as d_id) = :loan_ol_id)
      order by ox_from_ol_id
      into :ol_id, :doc_reg_id, :doc_ol_id, :ol_is_active
  do
  begin
    is_fact = 1;

    if (doc_ol_id is null) then
      doc_ol_id = ol_id;
    else
    begin
      select ol_from_reg_id
      from operlist
      where ol_id = :doc_ol_id
      into :doc_reg_id;
    end

    for select kc_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into :kc_id
    do
    begin
      execute procedure hp_do_kn_operbook_save(:doc_ol_id, :ol_id, :kc_id, 'operextra', :doc_reg_id, :ol_date,
          :ol_is_active);
    end
  end

  -- все погашения по судебным искам
  for select distinct ox_from_ol_id, ox_from_reg_id, ol_parent_id, ox_ol_is_active
      from operextra, operlist
      where ox_from_rd_id = :oek_rd_id and
            ox_amount_minus <> 0 and
            ox_from_ol_id = ol_id and
            ox_object_kind = 9 and
            ox_object_id = :k_id and
            ox_ol_date = :ol_date and
            (:loan_ol_id is null or
            cast(ol_id as d_id) = :loan_ol_id)
      order by ox_from_ol_id
      into :ol_id, :doc_reg_id, :doc_ol_id, :ol_is_active
  do
  begin
    is_fact = 1;

    if (doc_ol_id is null) then
      doc_ol_id = ol_id;
    else
    begin
      select ol_from_reg_id
      from operlist
      where ol_id = :doc_ol_id
      into :doc_reg_id;
    end

    for select kc_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into :kc_id
    do
    begin
      execute procedure hp_do_kn_operbook_save(:doc_ol_id, :ol_id, :kc_id, 'operextra', :doc_reg_id, :ol_date,
          :ol_is_active);
    end
  end

  -- все списания по судебным искам
  for select distinct ox_from_ol_id, ox_from_reg_id, ol_parent_id, ox_ol_is_active
      from operextra, operlist
      where ox_from_rd_id = :oek_rd_id and
            ox_amount_writeoff <> 0 and
            ox_from_ol_id = ol_id and
            ox_object_kind = 9 and
            ox_object_id = :k_id and
            ox_ol_date = :ol_date and
            (:loan_ol_id is null or
            cast(ol_id as d_id) = :loan_ol_id)
      order by ox_from_ol_id
      into :ol_id, :doc_reg_id, :doc_ol_id, :ol_is_active
  do
  begin
    is_fact = 1;

    if (doc_ol_id is null) then
      doc_ol_id = ol_id;
    else
    begin
      select ol_from_reg_id
      from operlist
      where ol_id = :doc_ol_id
      into :doc_reg_id;
    end

    for select kc_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into :kc_id
    do
    begin
      execute procedure hp_do_kn_operbook_save(:doc_ol_id, :ol_id, :kc_id, 'operextraoff', :doc_reg_id, :ol_date,
          :ol_is_active);
    end
  end

  ol_is_active = 1;
  if (disb_ol_id is not null) then
  begin
    loan_reg_id = disb_reg_id;
    ol_is_active = disb_ol_is_active;
  end

  if (is_fact = 1 or
      ol_date = f_date_month_end(ol_date)) then
  begin
    for select kc_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into :kc_id
    do
    begin
      -- Начисляем резерв на конец дня
      execute procedure hp_do_kn_operbook_kont_rsv(:kc_id, :ol_date, :loan_reg_id, :ol_is_active, :un_id);

      -- Начисляем ВБС на конец дня
      execute procedure hp_do_kn_operbook_kont_vbs(:kc_id, :ol_date, :loan_reg_id, :ol_is_active, :un_id);

      if (ol_date >= cast('30.04.2019' as date) and
          ol_date = f_date_month_end(ol_date)) then
      begin
        -- Устанавливаем флажок, что на конец месяца проводки по займу построены
        update kn_oksummary
        set oks_month_state = 1
        where oks_from_kc_id = :kc_id and
              oks_date_begin = :ol_date and
              oks_month_state <> 1;
      end
    end
  end
end
