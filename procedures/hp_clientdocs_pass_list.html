CREATE PROCEDURE HP_CLIENTDOCS_PASS_LIST(
    C_ID D_ID NOT NULL,
    ONLY_HIST D_BOOL NOT NULL = 0,
    REP_DATE D_DATE = null,
    BEGIN_DISB_DATE D_DATE = null)
RETURNS (
    CDS_ID D_INTEGER,
    CDS_FROM_PK_ID D_INTEGER,
    CDS_DOC_NO D_STR_100,
    CDS_DOC_DATE_BEGIN D_DATE,
    CDS_DOC_DATE_END D_DATE,
    CDS_DOC_ISSUED_BY D_STR_255,
    CDS_DOC_ISSUED_CODE D_STR_CODE,
    CDS_DOC_FOR_COMPANY D_STR_255,
    CDS_DOC_TERRITORY D_STR_255,
    DRL_ID D_ID,
    CURR_DOC_ADD_DATE D_DATE)
AS
declare variable CONST_NEW_ALGORITHM_DATE D_DATE;
declare variable CDS_INDEX_DOC_NO D_STR_100;
declare variable CDS_INDEX_DOC_NO_TMP D_STR_100;
declare variable CDS_DOC_DATE_BEGIN_TMP D_DATE;
declare variable CURR_CDS_ID D_INTEGER;
declare variable CURR_DOC_NO D_STR_100;
declare variable CURR_DOC_DATE_BEGIN D_DATE;
begin
  -- процедура возвращает данные паспортов клиента:
  --  Прикрепленные до даты смены алгоритма поиска: только те, которые были актуальны на даты выдач кредитов этому клиенту
  --  Прикрепленные после даты смены алгоритма поиска все верифицированные, у которых дата и номер отличаются от предыдущего,
  --  и если таких несколько в один день, то берем только последний на конец дня.

  --  only_hist   0 - вернет все паспорта, 1 - вернет без текущего
  --  begin_disb_date указывается, если надо ограничить выборку снизу по дате выдачи кредита

  const_new_algorithm_date = date '2022-01-01'; -- Дата смены алгоритма поиска

  if (rep_date is null) then
    rep_date = current_date;

  if (begin_disb_date is null) then
    begin_disb_date = date '1900-01-01';

  curr_cds_id = null;
  -- текущий паспорт
  select cds_id, cds_create_datetime, cds_doc_no, cds_doc_date_begin
  from clientdocs cds
  where cds.cds_from_c_id = :c_id
    and cds.cds_is_main_doc = 1
    and cds.cds_verify_status = 2
    and cds.cds_doc_date_begin <= :rep_date
    and cds_index_doc_no <> ''
  order by cds_id + 0 desc
  rows 1
  into curr_cds_id, curr_doc_add_date, curr_doc_no, curr_doc_date_begin;

  for
    with tab_old_variant_tmp as   -- по старому алгоритму (только актуальные паспорта на момент выдачи кредитов)
          (select distinct k_disb_date as date_tmp, max(cds_id) as cds_id
           from knclient kc, kontract k, kontstate ks, clientdocs cds
           where k_id = kc_from_k_id
             and ks_from_k_id = k_id
             and kc_from_c_id = :c_id
             and k_disb_date >= :begin_disb_date
             and :rep_date between ks.ks_date_begin and ks.ks_date_end
             and ks.ks_ol_is_delete = 0
             and ks.ks_state in (1,2)
             and cds.cds_from_c_id = kc_from_c_id
             and cds_doc_date_begin <= k_disb_date
             and cast(cds.cds_create_datetime as date) <= k_disb_date
             and cds.cds_is_main_doc = 1
             and cds.cds_verify_status = 2
             and cds_index_doc_no <> ''
             and cast(cds.cds_create_datetime as date) < :const_new_algorithm_date
           group by 1),
    tab_old_variant as  -- уберем дубли + фолбэк, если старая выборка пуста
      (
        /* обычный путь по k_disb_date */
        select max(t.date_tmp) as date_tmp, t.cds_id
        from tab_old_variant_tmp t
        group by t.cds_id

        union all

        /* ФОЛБЭК: если tab_old_variant_tmp пуст (актуально для сотрудников) */
        select coalesce(cds.cds_doc_date_begin, cast(cds.cds_create_datetime as date)) as date_tmp,
               max(cds.cds_id) as cds_id
        from clientdocs cds
        where cds.cds_from_c_id = :c_id
          and cds.cds_is_main_doc = 1
          and cds.cds_verify_status = 2
          and cds_index_doc_no <> ''
          and cast(cds.cds_create_datetime as date) < :const_new_algorithm_date
          and not exists (select 1 from tab_old_variant_tmp)
        group by coalesce(cds.cds_doc_date_begin, cast(cds.cds_create_datetime as date))
      ),

    /* === ИСПРАВЛЕНО: группируем по дате И по нормализованному номеру === */
    tab_new_variant as  -- по новому алгоритму (по дате начала документа + номеру)
      (
        select
          n.date_tmp,
          /* последний в рамках одного и того же номера в этот день */
          max(n.cds_id) as cds_id
        from (
          select
            coalesce(cds.cds_doc_date_begin, cast(cds.cds_create_datetime as date)) as date_tmp,
            /* нормализация номера: index_no -> no; убираем пробелы/тире/«№», lower; пустые -> NULL */
            nullif(
              lower(
                replace(replace(replace(coalesce(cds.cds_index_doc_no, cds.cds_doc_no), ' ', ''), '-', ''), '№', '')
              ),
              ''
            ) as num_norm,
            cds.cds_id
          from clientdocs cds
          where cds.cds_from_c_id = :c_id
            and cds.cds_is_main_doc = 1
            and cds.cds_verify_status = 2
            and cds_index_doc_no <> ''
            and cast(cds.cds_create_datetime as date) >= :const_new_algorithm_date
        ) n
        /* если номер пуст (редко), чтобы не склеить все в один день — уникализируем по cds_id */
        group by n.date_tmp, coalesce(n.num_norm, cast(n.cds_id as varchar(20)))
      ),
    /* === /ИСПРАВЛЕНО === */

    tab as  -- объединяем
      (select date_tmp, cds_id from tab_old_variant
       union all
       select date_tmp, cds_id from tab_new_variant)

    select :curr_cds_id from rdb$database
    where :only_hist = 0

    union

    select distinct cds_id
    from tab
    where cds_id is distinct from :curr_cds_id

    order by 1 desc
    into cds_id
  do
  begin
    select cds_from_pk_id, cds_doc_no, cds_doc_date_begin, cds_doc_date_end,
           cds_doc_issued_by,
           coalesce(cds_doc_issued_code, f_docrecog_value(cds_from_drl_id, 'authority_code', '')),
           cds_doc_for_company, cds_doc_territory, cds_index_doc_no, cds_from_drl_id
    from clientdocs cds
    where cds_id = :cds_id
    into cds_from_pk_id, cds_doc_no, cds_doc_date_begin, cds_doc_date_end,
         cds_doc_issued_by, cds_doc_issued_code, cds_doc_for_company, cds_doc_territory, cds_index_doc_no, drl_id;

    -- не включать запись, совпадающую с текущим паспортом по номеру и дате начала
    if (cds_id <> :curr_cds_id
        and cds_doc_no is not distinct from :curr_doc_no
        and cds_doc_date_begin is not distinct from :curr_doc_date_begin) then
      continue;

    -- дедуп: пропускаем полные дубликаты подряд по номеру+дате
    if (cds_index_doc_no is not distinct from cds_index_doc_no_tmp
        and cds_doc_date_begin is not distinct from cds_doc_date_begin_tmp) then
      continue;

    cds_index_doc_no_tmp = cds_index_doc_no;
    cds_doc_date_begin_tmp = cds_doc_date_begin;

    suspend;
  end
end
