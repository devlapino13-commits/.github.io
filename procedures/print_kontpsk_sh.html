CREATE PROCEDURE PRINT_KONTPSK_SH(
    KPS_ID D_ID)
RETURNS (
    SH_ID D_INTEGER,
    SH_DATE D_DATE,
    SH_OS D_CURRENCY,
    SH_PERC D_CURRENCY,
    SH_ALL D_CURRENCY,
    SH_TAX_PERC D_CURRENCY,
    SH_FINE D_CURRENCY)
AS
declare variable K_ID D_ID;
declare variable OL_ID D_ID;
declare variable OL_DATE D_DATE;
declare variable DB_ID D_ID;
begin
  db_id = f_get_db_id();

  select kps_from_k_id, ol_id, ol_date
  from kontpsk, operlist
  where kps_id = :kps_id and
        kps_from_ol_id = ol_id
  into k_id, ol_id, ol_date;

  if (db_id = 39801) then
  begin
    for select sh_id, zad_os, zad_perc, sh_date, sh_calc_fine
        from aarep_make_zad_list(:k_id, :ol_date, 1)
        where sh_date is not null
        into :sh_id, sh_os, sh_perc, :sh_date, :sh_fine
    do
    begin
      sh_tax_perc = (select res from fhp_round_bank((:sh_perc * 2 / 100.0), 0));
      sh_tax_perc = coalesce(:sh_tax_perc, 0);

      sh_all = :sh_os + :sh_perc + :sh_tax_perc + :sh_fine;

      if (sh_all <> 0) then
         suspend;
    end
  end
  else
  begin

    sh_fine = 0; 
    sh_tax_perc = 0;

    for select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_type < 7 and
              ok_ol_date <= :ol_date and
              ok_from_ol_id <= :ol_id
        group by ok_sh_id, ok_sh_date
        having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
        order by ok_sh_date
        into sh_id, sh_date, sh_os, sh_perc
    do
    begin
      if (db_id = 41701) then
      begin
        sh_tax_perc = (select res from fhp_round_bank((:sh_perc * 2 / 100.0), 0));
      end
      else
      begin
        sh_tax_perc = 0;
      end

      sh_all = sh_os + sh_perc + sh_tax_perc + sh_fine;
      suspend;
    end
  end
end
