CREATE PROCEDURE HP_HTML_GET_LOAN_STATUS_FPS(
    KC_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    RES_LINK D_STR_LONG)
AS
declare variable c_id d_id;
declare variable cell_no d_str_100;
declare variable clt_id d_integer;
declare variable fpb_id d_id;
declare variable fpb_name d_str_255;
declare variable fcb_status d_integer;
declare variable fcb_status_str d_str_255;
declare variable lang d_str_lang;
begin
  res_text = '';
  fcb_status_str = '';
  res_link = '';

  if (f_tmp_const('moneta_fps_active') <> 1) then
  begin
    suspend;
    exit;
  end

  select kc_from_c_id
  from kontract, knclient
  where kc_id = :kc_id and
        kc_from_k_id = k_id and
        k_disb_kind = 7
  into c_id;

  if (c_id is null) then-- Возвращаем пустой текст и выходим
  begin
    suspend;
    exit;
  end

  lang = f_get_lang_code(un_id, null);
  cell_no = f_cont(c_id, 'cell', 'ok', 'main');
  if (cell_no = '') then
  begin
    res_text = 'СБП не доступен: Основной номер отсутствует.';
    suspend;
    exit;
  end

  clt_id = f_cont_id(cell_no, 'add');
  cell_no = f_cont_format(cell_no, 'cell', 'secured');

  select fpb_id, fpb_name, fcb_status
  from fps_client_banks, fps_loan_banks, fps_banks
  where fbl_from_fcb_id = fcb_id and
        flb_from_kc_id = :kc_id and
        fcb_from_clt_id = :clt_id and
        fcb_from_c_id = :c_id and
        fcb_from_fpb_id = fpb_id
  into fpb_id, fpb_name, fcb_status;

  if (fpb_id is not null) then
  begin
    if (f_get_const('fps_manual_disb') <> 1) then
    begin
      if (fcb_status = 0) then
        fcb_status_str = 'На проверке';
      if (fcb_status = 1) then
        fcb_status_str = 'Требуется доп. проверка от КК';
      else if (fcb_status = 2) then
        fcb_status_str = 'Подтвержден';
      else if (fcb_status = -1) then
        fcb_status_str = 'Клиент не найден в этом банке';
      else if (fcb_status = -2) then
        fcb_status_str = 'Отказано, другой клиент';
  
      res_text = f_format('СБП: <b>%1</b>, %2<br><b>%3</b>', cell_no, fcb_status_str, fpb_name);
    end
    else
      res_text = f_format('СБП: <b>%1</b>, <b>%2</b>', cell_no, fpb_name);
    res_link = f_uri_sql(f_lang(lang, 'Изменить банк'), 'set_fps_bank',
      'KnClient', kc_id, 'combo', f_lang(lang, 'Выберите банк для выдачи займа'),
      '', 0, 0, fpb_id, '', 'dialog', '');
  end else
  begin
    res_text = f_format('СБП: %1<br>', cell_no);
    res_link = f_uri_sql(f_lang(lang, 'Выбрать банк'), 'set_fps_bank',
      'KnClient', kc_id, 'combo', f_lang(lang, 'Выберите банк для выдачи займа'),
      '', 0, 0, '', '', 'dialog', '');
  end

  suspend;
end
