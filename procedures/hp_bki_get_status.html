CREATE PROCEDURE HP_BKI_GET_STATUS(
    C_ID D_ID NOT NULL)
RETURNS (
    STATUS_ID D_SMALLINT,
    STATUS_TEXT D_STR_100)
AS
begin
  -- возвращает статус запроса в БКИ по "equifax_40" (RFCRU-4132)

  select max(eil_status)
  from extinfolist
  inner join extinfocfg on eic_id = eil_from_eic_id
  where eil_from_c_id = :c_id and
        eic_code in ('equifax_40', f_get_const('main_bki_config')) and -- на случай переключения на резерв
        datediff(day, eil_date, current_date) <= eic_info_actual_days
  into status_id;

  status_text = case
                  when status_id is null then 'Нет запроса в БКИ'
                  when status_id < 0 then 'Ошибка получения кредитной истории'
                  when status_id = 0 then 'Запрос в БКИ в очереди для отправки'
                  when status_id = 1 then 'Запрос в БКИ отправлен, в ожидании ответа'
                  when status_id = 2 then 'Запрос в БКИ отправлен, ответ получен'
                end;

  suspend;
end
