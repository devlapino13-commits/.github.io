CREATE PROCEDURE HP_EXPRESSAPP_FILL(
    REP_DATE_END D_DATE,
    REP_C_ID D_ID)
RETURNS (
    C_ID ID_TYPE,
    KOL_KONT_NO_DELINQ D_SMALLINT,
    KOL_KONT_FACT D_SMALLINT,
    ASP_STATUS D_STR_CODE,
    LGC_CALC D_INTEGER,
    MP_STATUS D_SMALLINT,
    BASE_K_ID ID_TYPE,
    CHECK_DOCPRLOAN SMALLINT,
    CS_INFO_STATE SMALLINT)
AS
declare variable delinq_day d_integer;
declare variable is_insert d_smallint;
declare variable pr_id d_integer;
declare variable exa_amount curr_type;
declare variable exa_period d_integer;
declare variable exa_lo_id id_type;
declare variable exa_from_reg_id integer;
declare variable exa_percent d_percent;
declare variable exa_repay_day d_smallint; /* погашение по 25м числам */
declare variable exa_from_un_id id_type;
begin
  -- вытаскиваем всех клиентво исключая дубликатов
  for
  select distinct kc_from_c_id, count(*), max(oks_from_k_id), max(oks_kc_id), max(oks_region_id)
  from oksummary, knclient
  where :rep_date_end between oks_date_begin and oks_date_end
  and :rep_date_end < oks_delinq_date -- исключаем те контракты которые были в просрочке
  and oks_from_k_id = kc_from_k_id
  and kc_amount_approved > 0
  and oks_close_state = 0
  and kc_from_c_id = :rep_c_id
  group by kc_from_c_id
  into c_id, kol_kont_no_delinq, base_k_id, exa_lo_id, exa_from_reg_id
  do
  begin
    -- проверка на действующего договора АСП
    asp_status = f_edoc_status('asp.agreement', :c_id, :rep_date_end);
    if (asp_status <> 'signed') then
    begin
      continue;
    end

    -- проверка на кол-во активных контрактов
    kol_kont_fact = null;
    select count(*) from kontstate, knclient
    where ks_ol_is_delete = 0
    and ks_state = 1
    and :rep_date_end between ks_date_begin and ks_date_end
    and ks_from_k_id = kc_from_k_id
    and kc_amount_approved > 0
    and kc_from_c_id = :c_id
    into kol_kont_fact;

    -- если кол-во актинвых контрактов не совпадает с кол-вом контраквто без просрочки
    -- то проверяется все контракты клинта на просрочку
    delinq_day = null;
    if (kol_kont_no_delinq <> kol_kont_fact) then
    begin
      /*
      -- проверяем каждого контракта на просрочку
      select max(:rep_date_end - oks_delinq_date) from knclient, oksummary
      where kc_from_c_id = :c_id
      and kc_amount_approved > 0
      and kc_from_k_id = oks_from_k_id
      and oks_close_state = 0
      and :rep_date_end between oks_date_begin and oks_date_end
      into delinq_day;
      */
      -- нет необходимости проверять на просроченные дни всех контрактов,
      -- так как в первом запросе возвращаются все непросроченные контракты, и если кол-во контрактов больше чем не просроченные
      -- то значит имеется просрочка, текущий момент был проверен.
      continue;
    end

    -- проверка имеется ли легитимные платежи
    lgc_calc = null;
    select sum(lgc_calc)
    from legitimcalc
    where lgc_from_c_id = :c_id
    and lgc_date <= current_date
    into lgc_calc;
    lgc_calc = coalesce(lgc_calc,0);
    if (lgc_calc <= 0) then
    begin
      continue;
    end

    -- проверка имеется ли МП у клиента 

    mp_status = f_fsm_client_status(c_id);

    if (mp_status = 0) then
    begin
      continue;
    end 

    -- проверка по документам имеется ли соответствии
    if (f_docprloan_valid(:c_id,139,0,:rep_date_end) = -1)
    then check_docprloan = 0; -- 0 - не подходит по документам;
    else check_docprloan = 1; -- 1 - соответсвует по документам


    -- после выдачи по документам данные менялись или нет
    cs_info_state = null;
    select cs_info_state from clientstate
    where cs_from_c_id = :c_id
    into cs_info_state; -- 0 - Данные менялись; 1 - Данные не изменились

    is_insert = 1;  -- если нужно заполнить таблицу expressapp то делаем = 1
    if (is_insert = 1) then
    begin
      -- вычисляем основной контракт если у клинета их несколько

      if (kol_kont_no_delinq > 1) then
      begin
        -- берем последний выданный
        select ks_from_k_id from kontstate, knclient
        where ks_ol_is_delete = 0
        and ks_state = 1
        and :rep_date_end between ks_date_end and ks_date_end
        and ks_from_k_id = kc_from_k_id
        and kc_amount_approved > 0
        and kc_from_c_id = :c_id
        order by ks_date_begin desc
        rows 1
        into base_k_id;
         -- вычисляем ВИПС и регион
        select kal_new_kf_officer, kal_new_reg_id
        from kontract, knaktlist
        where k_id = :base_k_id
        and k_from_kf_id = kal_from_kf_id
        and knaktlist.is_delete = 0
        and current_date between kal_date and kal_date_end
        into exa_lo_id, exa_from_reg_id;
      end 

      pr_id = 139; -- продукт доверие
      exa_amount = 25000; -- сумма выдачи
      exa_period = 9; -- период
      exa_percent = 8; -- процентная ставка
      exa_repay_day = 25; -- погашение по 25м числам
      exa_from_un_id = 1000000000000; -- FINSTART_ROBOT

      insert into expressapp (exa_id, exa_date, exa_from_c_id, exa_base_k_id, exa_from_exc_id, exa_final_status,
                              exa_wait_status, exa_from_ma_id, exa_from_k_id, exa_from_edl_id, exa_disb_date, exa_from_pr_id,
                              exa_amount, exa_period, exa_percent, exa_repay_day, exa_lo_id, exa_from_reg_id, exa_try_count,
                              exa_from_un_id, exa_datetime)
      values (null, current_date, :c_id, :base_k_id, 0, 0, 0,
              null, null, null, maxvalue(:rep_date_end + 3, current_date), :pr_id, :exa_amount, :exa_period,
              :exa_percent, :exa_repay_day, :exa_lo_id, :exa_from_reg_id, 0, :exa_from_un_id, current_timestamp);

    end

    suspend;
  end
end
