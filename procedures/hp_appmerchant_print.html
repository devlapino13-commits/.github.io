CREATE PROCEDURE HP_APPMERCHANT_PRINT(
    IQA_ID D_ID NOT NULL)
RETURNS (
    IQA_DATE D_DATE,
    CS_FULLNAME D_STR_LONG,
    CS_NAME D_STR_255,
    RD_NAME D_STR_255,
    IQS_NO D_INTEGER,
    IQS_BANK_DATA D_STR_100,
    IQS_MIN_TRANS_AMOUNT D_CURRENCY,
    IQS_MAX_TRANS_AMOUNT D_CURRENCY,
    COMPANY_NAME D_STR_LONG)
AS
declare variable c_id d_id;
declare variable tcp_data d_str_long;
declare variable rd_id_tsp d_integer;
begin
  company_name = f_company_name('short', 'rus');
  iqs_bank_data = 'БИК 507001, расчетный счет 1013890209990149';
  iqs_min_trans_amount = 1;
  iqs_max_trans_amount = 200000;
  rd_id_tsp = f_get_rd_id('AgreementKinds.tsp');

  for
    select
      iqa_date,
      iqs_tcp_data,
      iqs_shop_name,
      iqs_no,
      cs_from_c_id
    from ipc_qr_app
      inner join ipc_qr_app_shops
        on iqs_from_iqa_id = iqa_id
      inner join clientshops
        on cs_id = iqs_from_cs_id and
        cs_from_rd_id = :rd_id_tsp
    where iqa_id = :iqa_id
    order by iqs_no
  into
    :iqa_date,
    :cs_fullname,
    :cs_name,
    :iqs_no,
    :c_id
  do
  begin
    select first 1
      rd_memo
    from agreements
      inner join agreement_feac
        on afe_from_ag_id = ag_id
      inner join refdata
        on rd_id = afe_from_rd_id
    where ag_object_kind = 0
      and ag_object_id = :c_id
      and ag_from_rd_id = f_get_rd_id('AgreementKinds.tsp')
      and ag_status = 1
      and :iqa_date between coalesce(ag_date_begin, '01.01.1900') and coalesce(ag_date_end, '31.12.9999')
    order by ag_id desc, afe_from_rd_id
    into
      :rd_name;
    suspend;
  end
end
