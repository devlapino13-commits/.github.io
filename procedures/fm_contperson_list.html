CREATE PROCEDURE FM_CONTPERSON_LIST(
    C_ID DECIMAL(18,0))
RETURNS (
    ID INTEGER,
    NAME VARCHAR(100),
    RELATION VARCHAR(100),
    CELL_NO VARCHAR(255),
    CELL_NO_ADD VARCHAR(255),
    LANGUAGE_ID INTEGER,
    LANGUAGE VARCHAR(100),
    STATUS INTEGER,
    MEMO VARCHAR(255),
    ADDRESS VARCHAR(255),
    COMMENTS VARCHAR(255),
    VERIF_IGNORE SMALLINT,
    CREATE_DATE DATE,
    ADL_ID ID_TYPE,
    CELL_NO_ID D_INTEGER,
    CELL_NO_ADD_ID D_INTEGER)
AS
begin
  cell_no_id = 0;
  cell_no_add_id = 0;

  for select cc_id, coalesce(cc_name, ''), coalesce(cc_status, ''), cc_cell_no, cc_cell_no_add, cc_language,
             left(cc_check_memo, 255),
             case
               when cc_total_res = -2 then 4 -- Не подтвердили
               when cc_total_res = -1 then 3 -- Не дозвонились
               when cc_total_res = 1 then 2 -- Подтвердили
               when cc_in_process = 1 then 1 -- В процессе
               else 0
             end,
             cc_from_adl_id, cc_svz_comments, cc_verif_ignore, cc_create_date
      from contacts, fm_contperson_id(:c_id)
      where cc_id = list_cc_id
      into id, name, relation, cell_no, cell_no_add, language_id, memo, status, adl_id, comments, verif_ignore,
           create_date
  do
  begin
    if (name = '') then
      name = 'ФИО не указано';
    if (relation = '') then
      relation = 'Кем приходится?';
    memo = replace(memo, 'Примечание: ', '');
    language = '';
    if (language_id is not null) then
    begin
      select rd_name
      from refdata
      where rd_id = :language_id
      into language;
    end
    address = 'Адрес не заполнен.';
    if (adl_id is null) then
      adl_id = 0;
    if (adl_id <> 0 and
        right(adl_id, 9) <> '000000000') then
    begin
      select adl_address_str
      from addrlist
      where adl_id = :adl_id
      into :address;
    end

    if (f_get_const('fsm_phonecall_enabled') = 1) then
    begin
      if (exists(
        select *
        from hp_is_client_delinqed(:c_id)
        where is_delinq = 1)) then
      begin
        -- Когда клиент в просрочке (в РФ), запрещаем звонить
        relation = relation || coalesce(' (' || cell_no || ')', '');
        cell_no = '';
        cell_no_add = '';
      end
      else
      begin
        -- Нужны только для звонков через Asterisk
        cell_no_id = f_cont_id(cell_no, 'add');
        cell_no_add_id = f_cont_id(cell_no_add, 'add');
      end
    end

    suspend;
  end
end
