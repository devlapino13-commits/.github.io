CREATE PROCEDURE FM_HTML_CLIENTLINK_NO_LO(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable cl_fio str_100;
declare variable lo_id d_id;
declare variable reg_id d_integer;
declare variable office_address d_str_255;
declare variable def_text d_str_code;
declare variable emw_language d_str_100;
begin
  res_text = (select res_style from fm_html_styles) || '<body>';

  select un_from_c_id
  from username
  where un_id = :un_id
  into :lo_id;

  select ps_from_reg_id
  from hr_pstatus
  where ps_from_c_id = :lo_id and
        ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_is_working = 1 and
        ps_status = 1
  into :reg_id;

  office_address = '';
  if (reg_id is not null) then
  begin
    select adl_address_str
    from region, addrlist
    where reg_unicode = :reg_id and
          reg_address_fact = adl_id
    into office_address;
  end

  for select clk_id, clk_from_c_id, clk_link_name, clk_link_cell_no, clk_comment, clk_status, clk_add_datetime,
             clk_from_un_id
      from clientlink
      where clk_id = :obj_id as cursor clk
  do
  begin
    res_text = blob_append(res_text, 'Имя: <b>' || clk.clk_link_name || '</b>');
    select Emw_Language
    from extminiweb where emw_phone = clk.clk_link_cell_no
    into Emw_Language;
    Emw_Language = coalesce(Emw_Language,'');
    res_text = blob_append(res_text, '<br>Язык общения: <b>' || Emw_Language || '</b>');

    def_text = case clk.clk_status
                 when -2 then 'Отказ'
                 when -1 then 'Перезвонить'
                 when 0 then 'Новый контакт'
                 when 1 then 'Заинтересован'
                 when 2 then 'Получил займ'
               end;

    res_text = blob_append(res_text, '<br>Текущий статус: <b>' || def_text || '</b>');
    if (clk.clk_status in (-2, -1, 0, 1)) then
    begin
      res_text = blob_append(res_text, ' ' || f_uri_sql('Изменить',
        'set_clientlink_status', 'clientlink', :obj_id, 'combo',
        'Укажите новый статус', '', 0, 0, clk.clk_status, '', 'dialog', '') || '<br>');
    end

    res_text = blob_append(res_text, '<br>');
    res_text = blob_append(res_text, '<br>Комментарий:<br><b>' || coalesce(clk.clk_comment, 'Отсутствует') || '</b><br>');

    if (coalesce(clk.clk_comment, '') = '') then
      def_text = 'Добавить';
    else
      def_text = 'Изменить';
    res_text = blob_append(res_text, f_uri_sql(:def_text, 'set_clientlink_comment',
      'clientlink', :obj_id, 'text', 'Введите комментарий', '', 0, 255,
      clk.clk_comment, '', 'dialog', '') || '<br>');

    if (office_address <> '') then
      res_text = blob_append(res_text, '<br><br>' || f_uri_sql('Отправить адрес офиса по СМС',
        'send_clientlink_sms', 'clientlink', obj_id, 'none',
        'Подтвердите отправку sms на номер ' || clk.clk_link_cell_no ||', с текстом:',
        office_address, 0, 255, '', '', 'dialog', '') || '<br>');

    res_text = blob_append(res_text, '<br><br>');
    res_text = blob_append(res_text, 'Номер предоставлен клиентом:');

    select c_name_fio
    from clients
    where c_id = clk.clk_from_c_id
    into cl_fio;
    res_text = blob_append(res_text, '<br><b>' || f_uri_open(cl_fio, 'client', 'Clients', clk.clk_from_c_id, cl_fio, '', un_id) || '</b>'); -- добавленом гипер ссылка для килента
    res_text = blob_append(res_text, '<br>' || f_datetimetostr(clk.clk_add_datetime));
    res_text = blob_append(res_text, '<br><br>' || f_uri_sql('Забрать на обработку',
        'pick_up_clientlink', 'clientlink', obj_id, 'none',
        'Забрать клиента <b>' || clk.clk_link_name ||'</b>, для дальнейшей работы с клиентом.', '', 0, 255, '', '', 'dialog', '') || '<br>');
  end
  res_text = blob_append(res_text, '</body>');
  suspend;
end
