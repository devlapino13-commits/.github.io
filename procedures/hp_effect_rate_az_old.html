CREATE PROCEDURE HP_EFFECT_RATE_AZ_OLD(
    K_ID D_ID)
RETURNS (
    CALC_EFF DECIMAL(18,3))
AS
declare variable max_iter integer = 100;
declare variable calc_error double precision = 0.000001;
declare variable iter integer;
declare variable prc double precision;
declare variable l_prc double precision;
declare variable h_prc double precision;
declare variable npv double precision;
declare variable sh_amount decimal(18,4) = 0;
declare variable j integer;
begin
  -- Расчет полной стоимости кредита (Азербайджан)
  -- В Excel IRR, ВСД, умножается на 12 месяцев

  calc_eff = 0.0;
  if (not(exists(select *
                 from temp_schedule))) then
  begin
    suspend;
    exit;
  end

  l_prc = 0; -- 0%
  h_prc = 10; -- 1000%
  iter = 0;
  while (iter <= max_iter) do
  begin
    prc = (l_prc + h_prc) / 2;
    npv = 0;
    j = 0;
    for select sh_os
        from temp_schedule
        order by sh_date
        into sh_amount
    do
    begin
      npv = npv + sh_amount / power(1 + prc, j);
      j = j + 1;
    end
    if (abs(npv) < calc_error) then
      break;
    if (npv > 0) then
      l_prc = prc;
    else
      h_prc = prc;
    iter = iter + 1;
  end
  if (iter >= max_iter) then
    exception e_default 'Cannot calc effective rate for contract: ' || k_id;

  -- Возвращаем ставку, умноженную на 12 (годовая)
  calc_eff = round(prc * 12, 5) * 100;
  suspend;
end
