CREATE PROCEDURE HP_BKI_GET_SENDED_DATA_UX(
    REC_ID D_ID,
    PREV_REC_ID D_ID)
RETURNS (
    PARENT_NAME D_STR_255,
    XML_NAME D_STR_CODE,
    FIELD_NAME D_STR_LONG,
    VAL D_STR_2048,
    PREV_VAL D_STR_2048,
    STATE_VAL D_SMALLINT)
AS
declare variable bfv_id d_id;
  declare variable xml_version_id d_id;
  declare variable bkc_id d_id;
  declare variable prev_bkc_id d_id;
  declare variable bev_id d_id;
  declare variable prev_bev_id d_id;
  declare variable obj_kind d_smallint;
  declare variable obj_id d_id;
  declare variable prev_obj_id d_id;
begin
  select bxv_id
  from bki_xml_version
  where bxv_date_begin <= current_date
  order by bxv_date_begin desc
  rows 1
  into xml_version_id;

  bev_id = rec_id;
  prev_bev_id = prev_rec_id;
  bkc_id = (select bev_from_bkc_id from bki_events where bev_id = :bev_id);
  prev_bkc_id = (select bev_from_bkc_id from bki_events where bev_id = :bev_id);

  for with recursive tab as(
    select bfv_id, bfv_position, bki_code, bki_name, bki_db_field, bxf_xml_node, bxf_default_value, bxf_id, bfv_parent_id
    from bki_xml_fields_ver, bki_xml_fields, bki_indicators
    where bxf_id = bfv_from_bxf_id
      and bki_code = bxf_from_bki_code
      and bfv_from_bxv_id = :xml_version_id
    order by bfv_position
  ), t1 (id, pos, code, name, db, xml, def, xml_id, root, obj_kind, obj_id, prev_obj_id) as(
    select bfv_id, bfv_position, bki_code, bki_name, bki_db_field, bxf_xml_node, bxf_default_value, bxf_id, bki_code,
      iif(bki_code = 'Title', 1, 2),
      iif(bki_code = 'Title', :bkc_id, :bev_id),
      iif(bki_code = 'Title', :prev_bkc_id, :prev_bev_id)
    from tab
    where bfv_parent_id is null
    union all
    select bfv_id, bfv_position, bki_code, bki_name, bki_db_field, bxf_xml_node, bxf_default_value, bxf_id, root,
      obj_kind, obj_id, prev_obj_id
    from t1, tab
    where bfv_parent_id = id
  )
  select id, xml, f_format('%1 %2', code, name), root, obj_kind, obj_id, prev_obj_id
  from t1
  into bfv_id, xml_name, field_name, parent_name, obj_kind, obj_id, prev_obj_id
  do
  begin
    val = null;

    select list(f_bki_data_text(bsd_from_bkd_id))
    from bki_send_data
    where bsd_from_bfv_id = :bfv_id
      and bsd_obj_kind = :obj_kind
      and bsd_obj_id = :obj_id
    into val;

    if (val is null) then
      select list(f_bki_data_text(bsd_from_bkd_id))
      from bki_send_data_oper
      where bsd_from_bfv_id = :bfv_id
        and bsd_obj_kind = :obj_kind
        and bsd_obj_id = :obj_id
      into val;

    if (prev_obj_id is not null) then
    begin
      prev_val = null;

      select list(f_bki_data_text(bsd_from_bkd_id))
      from bki_send_data
      where bsd_from_bfv_id = :bfv_id
        and bsd_obj_kind = :obj_kind
        and bsd_obj_id = :prev_obj_id
      into prev_val;
  
      if (prev_val is null) then
        select list(f_bki_data_text(bsd_from_bkd_id))
        from bki_send_data_oper
        where bsd_from_bfv_id = :bfv_id
          and bsd_obj_kind = :obj_kind
          and bsd_obj_id = :prev_obj_id
        into prev_val;

      state_val = case
                    when val is not distinct from prev_val then 1 -- Не изменилось
                    when coalesce(val, '') = '' and coalesce(prev_val, '') <> '' then 2 -- Пропало
                    when coalesce(val, '') <> '' and coalesce(prev_val, '') = '' then 3 -- Новое
                    else 4                                                              -- Изменилось
                  end;
    end

    if (val is not null or prev_val is not null) then
      suspend;

  end
end
