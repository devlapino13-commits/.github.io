CREATE PROCEDURE HP_LLC_CHECK_TRANSFER_UNCLAS(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    LKC_FROM_C_ID DECIMAL(18,0),
    C_FIO VARCHAR(100),
    LKC_PERCENT_OLD_MAX DECIMAL(18,7),
    LKC_PERCENT DECIMAL(18,7),
    LKC_PERCENT_NEW DECIMAL(18,7),
    CUR_MAX_DELIQ_DAYS SMALLINT,
    MAX_DELIQ_DAYS SMALLINT,
    KONTRACT_NO_LIST VARCHAR(1024),
    C_INN VARCHAR(20),
    LKC_STATUS SMALLINT)
AS
begin
  /* Перевод классифицированных активов в категорию неклассифицированных осуществляется только при выполнении следующих условий:  
        - отсутствия просроченных сумм;  
        - улучшения финансового состояния заемщика; 
        - своевременной полной оплаты последних трех взносов, согласно графику погашения.  
    При погашении всей просроченной задолженности активы, классифицированные как «сомнительные» или «потери»,
        переводятся в категорию «субстандартные».  */

  for select lkc_from_c_id, lkc_percent, --
             max(lkc.lkc_deliq_days) as cur_max_deliq_days, --
             max((select max(ok_ol_date - ok_sh_date)
                  from operkred ok
                  where ok_from_k_id = lkc.lkc_from_k_id and
                        ok_type = 4 and
                        ok_ol_is_active = 1 and
                        ok_sh_date between dateadd(month, -3, :rep_date_end) and :rep_date_end and --3 месяца
                        ok_sh_date <= ok_ol_date and --
                        ok_ol_date <= :rep_date_end)) as max_delinq_days, --
             max((select max(lkc2.lkc_percent)
                  from llc_main lm2
                  inner join llc_kont_cli lkc2 on lkc2.lkc_from_lm_id = lm2.lm_id and
                        -- выборка только по индив. кредитам
                        lkc2.lkc_is_indiv = 1 and
                        lkc2.lkc_percent > 25.0 and
                        lkc2.lkc_from_c_id = lkc.lkc_from_c_id and
                        lkc2.lkc_from_k_id = lkc.lkc_from_k_id and
                        coalesce(lkc2.lkc_mark, false)
                  where lm2.lm_date_status < :rep_date_end)) lkc_percent_old_max, --
             list(k_kontract_no, '; ') kontract_no_list
      from llc_main lm
      inner join region_list(:rep_region_id, 1, -1, 1) on lm_from_reg_id = id_list
      inner join llc_kont_cli lkc on lkc_from_lm_id = lm_id and
            -- выборка только по индив. кредитам
            lkc_is_indiv = 1 and
            lkc_percent < 25.0
      inner join kontract k on lkc.lkc_from_k_id = k_id
      where lm_date_status = :rep_date_end
      group by 1, 2
      into lkc_from_c_id, lkc_percent, cur_max_deliq_days, max_deliq_days, lkc_percent_old_max, kontract_no_list
  do
  begin
    if (lkc_percent_old_max > lkc_percent) then
    begin
      if (coalesce(max_deliq_days, -1) < 0) then
        max_deliq_days = 0;
      if (cur_max_deliq_days > max_deliq_days) then
        max_deliq_days = cur_max_deliq_days;

      if (cur_max_deliq_days > 0) then
      begin
        lkc_percent_new = 25.0;
        lkc_status = 5;
      end
      else
        lkc_percent_new = lkc_percent;
      if (lkc_percent = 0 and
          cur_max_deliq_days = 0) then
      begin
        if (max_deliq_days = 0) then
        begin
          lkc_percent_new = 5.0;
          lkc_status = 4;
        end
        else
        begin
          lkc_percent_new = 25.0;
          lkc_status = 5;
        end
      end

      if (lkc_percent = 5 and
          cur_max_deliq_days = 0 and
          max_deliq_days > 0) then
        begin
          lkc_percent_new = 25.0;
          lkc_status = 5;
        end

      if (lkc_percent_new <> lkc_percent) then
      begin
        c_inn = '';
        select c_name_fio, c.c_inn
        from clients c
        where c_id = :lkc_from_c_id
        into c_fio, c_inn;

        suspend;
      end
    end
  end
end
