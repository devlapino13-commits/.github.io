CREATE PROCEDURE INN_KK_SAVE(
    ACL_ID D_ID,
    IS_FOUND D_BOOL,
    IS_RETRY D_BOOL = 0,
    NEW_MEMO D_STR_8100 = '')
RETURNS (
    RES_ACL_ID D_ID)
AS
declare variable em_id d_id;
declare variable c_id  d_id;
declare variable rd_id d_integer;
declare variable un_id d_id;
begin
  rd_id = f_get_rd_id('ActionGroup.Manual_Inn_Check');
  for
    select
      acl_object_id,
      acl_respons_c_id,
      acl_from_un_id
    from actionlist
    where acl_id = :acl_id
      and acl_from_rd_id = :rd_id
      and acl_status = 1
  into
    :c_id,
    :em_id,
    :un_id
  do
  begin
    if (is_found = 1) then
    begin
      res_acl_id = (
        select first 1
          acl_id
        from actionlist
        where acl_object_kind = 0
          and acl_object_id = :c_id
          and acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO'))
          and acl_status = 0
        order by acl_id desc);
      if (res_acl_id > 0) then
        update actionlist set
          acl_respons_c_id = :em_id,
          acl_status = 1
        where acl_id = :res_acl_id;
    end
    update actionlist set
      acl_status = iif(:is_retry = 0, 2, -1)
    where acl_id = :acl_id;
    if (is_retry = 1 and new_memo <> '') then
      execute procedure hp_objectprops_memo_set('0', :c_id, -1,
        :rd_id, :new_memo, :un_id);
  end

  res_acl_id = coalesce(res_acl_id, 0);
  suspend;
end
