CREATE PROCEDURE HP_MENULIST_REPORT_SYNC_POS(
    PARENT_ID D_INTEGER,
    PARENT_REC_ID D_INTEGER = 0,
    NEED_REPORT_POS_SYNC D_BOOL = 0)
RETURNS (
    REC_ID D_INTEGER,
    ML_ID D_INTEGER,
    ML_PARENT_ID D_INTEGER,
    ML_INDEX D_INTEGER,
    ML_POSITION D_INTEGER,
    RL_ORDER_NO D_INTEGER,
    ML_CAPTION D_STR_255,
    ML_REAL_POSITION D_INTEGER,
    ML_REAL_DIFFERENT D_BOOL,
    ML_REPORT_DIFFERENT D_BOOL)
AS
declare variable tmp_parent d_integer;
declare variable tmp_real_position d_integer;
declare variable flag_code d_str_code;
begin
  flag_code = 'menulist_moved';

  if (need_report_pos_sync is null) then
    need_report_pos_sync = 0;

  if (parent_rec_id is null) then
    parent_rec_id = 0;

  rec_id = parent_rec_id;

  tmp_real_position = -1;
  for select ml_id, ml_parent_id, ml_index, ml_position, coalesce(rl_order_no, ml_position), ml_caption
  from menulist
       left join reportlist on (ml_id = rl_id)
  where ml_parent_id is not distinct from :parent_id
  order by coalesce(rl_order_no, ml_position), abs(ml_id)
  into ml_id, ml_parent_id, ml_index, ml_position, rl_order_no, ml_caption
  do
  begin
    tmp_real_position = tmp_real_position + 1;
    ml_real_position = tmp_real_position;

    ml_real_different = 0;
    if (ml_real_position <> ml_position) then
    begin
      ml_real_different = 1;

      if (need_report_pos_sync = 1) then
      begin
        update menulist
        set ml_position = :ml_real_position
        where ml_id = :ml_id and
              ml_position is distinct from :ml_real_position;
      end
    end

    ml_report_different = 0;
    if (rl_order_no is not null and rl_order_no <> ml_real_position) then
      ml_report_different = 1;

    if (need_report_pos_sync = 1) then
    begin
      uxf$flag(flag_code, 'on');

      update reportlist
      set rl_order_no = :ml_real_position
      where rl_id = :ml_id and
            rl_order_no is distinct from :ml_real_position;

      uxf$flag(flag_code, 'off');
    end

    rec_id = rec_id + 1;

    suspend;

    tmp_parent = ml_id;
    for select rec_id, ml_id, ml_parent_id, ml_index, ml_position, rl_order_no, ml_caption, ml_real_position,
               ml_real_different, ml_report_different
    from hp_menulist_report_sync_pos(:tmp_parent, :rec_id, :need_report_pos_sync)
    into :rec_id, :ml_id, :ml_parent_id, :ml_index, :ml_position, :rl_order_no, :ml_caption, :ml_real_position,
         :ml_real_different, :ml_report_different
    do
      suspend;
  end
end
