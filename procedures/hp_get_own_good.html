CREATE PROCEDURE HP_GET_OWN_GOOD(
    ID D_ID,
    DT D_DATE = null)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    BILL_ID D_ID,
    BILL_LINE_NO D_INTEGER,
    REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    WP_ID D_ID,
    WP_NAME D_STR_100,
    MOTV_ID D_ID,
    MOTV_NAME D_STR_100,
    ML_ID D_ID,
    ML_NAME D_STR_255,
    ML_MODEL D_STR_255,
    YL_ID D_INTEGER,
    YL_CODE D_STR_20,
    QUANTITY D_CURRENCY,
    COST D_CURRENCY,
    TOTAL D_CURRENCY,
    QUANTITY_RESERVED D_CURRENCY,
    TOTAL_RESERVED D_CURRENCY)
AS
begin
  dt = coalesce(dt, current_date);
  for
    select
      c_id,
      c_name_fio,
      ob_bill_id,
      ob_bill_line_no,
      ob_from_ml_id
    from providerowngood
      inner join clients
        on c_id = pog_from_c_id
      inner join operbill
        on ob_from_own_c_id = pog_from_c_id
          and :dt between ob_date_begin and ob_date_end
          and ob_is_delete = 0
    where pog_from_c_id = :id
  into
    :c_id,
    :c_name_fio,
    :bill_id,
    :bill_line_no,
    :ml_id
  do
  begin
    select
      ml_name,
      ml_model,
      yl_id,
      yl_code
    from matlist
      left join yedizlist
        on yl_id = ml_from_yl_id
    where ml_id = :ml_id
    into
      :ml_name,
      :ml_model,
      :yl_id,
      :yl_code;

    for
      select
        om_workplace_id,
        om_matotvet_id,
        sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus),
        sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus)
      from opermat
        inner join operlist
          on ol_id = om_from_ol_id
            and is_delete = 0
      where om_bill_id = :bill_id
        and om_bill_line_no = :bill_line_no
      group by om_workplace_id, om_matotvet_id
      having sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) <> 0
        or sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) <> 0
    into
      :wp_id,
      :motv_id,
      :quantity,
      :total
    do
    begin
      cost = iif(quantity = 0, 0, round(total / quantity, 0));
      select
        quantity,
        total
      from hp_kontmat_own_reserved(
        :bill_id,
        :bill_line_no,
        :wp_id,
        :motv_id,
        :dt,
        :c_id,
        :ml_id
      )
      into
        :quantity_reserved,
        :total_reserved;

      if (quantity <> 0 or total <> 0) then
      begin
        select
          wp_name,
          reg_unicode,
          reg_name
        from workplace
          inner join region
            on reg_unicode = wp_from_reg_id
        where wp_id = :wp_id
        into
          :wp_name,
          :reg_id,
          :reg_name;
        motv_name = (select c_name_fio from clients where c_id = :motv_id);
        suspend;
      end
    end
  end
end
