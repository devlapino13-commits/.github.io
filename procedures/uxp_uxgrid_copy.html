CREATE PROCEDURE UXP$UXGRID_COPY(
    GRID_CODE D_STR_CODE,
    NEW_NAME D_STR_100,
    NEW_MEMO D_STR_255,
    IS_PRIVATE D_BOOL,
    UN_ID D_ID)
RETURNS (
    NEW_UUID D_UUID,
    NEW_CODE D_STR_CODE)
AS
declare variable ugr_uuid d_uuid;
declare variable share_uuid d_uuid;
begin
  ugr_uuid = char_to_uuid(grid_code);

  for select ugr_uuid, ugr_ufr_uuid, ugr_share_uuid, ugr_name, ugr_memo, ugr_is_private, ugr_default,
             ugr_hide_on_group, ugr_group_panel, ugr_find_panel, ugr_hide_summary,
             ugr_autowidth, ugr_header_autoheight, ugr_show_info_panel,
             ugr_hide_inplace_editor, ugr_show_excel_import, ugr_owner_user,
             ugr_inactive, ugr_excel, ugr_ds_code, ugr_group_field, ugr_formatting,
             ugr_tree_can_move, ugr_allow_append, ugr_ui_kind, ugr_calendar_fields
  from ux$grid
  where ugr_uuid = :ugr_uuid
  as cursor cur
  do
  begin
    insert into ux$grid (ugr_ufr_uuid, ugr_share_uuid, ugr_name, ugr_memo, ugr_is_private, ugr_hide_on_group,
                         ugr_group_panel, ugr_find_panel, ugr_hide_summary,
                         ugr_autowidth, ugr_header_autoheight, ugr_show_info_panel,
                         ugr_hide_inplace_editor, ugr_show_excel_import, ugr_owner_user, ugr_inactive,
                         ugr_excel, ugr_ds_code, ugr_group_field, ugr_formatting,
                         ugr_tree_can_move, ugr_allow_append, ugr_ui_kind, ugr_calendar_fields)
    values (cur.ugr_ufr_uuid, cur.ugr_share_uuid, :new_name, :new_memo, :is_private, cur.ugr_hide_on_group,
            cur.ugr_group_panel, cur.ugr_find_panel, cur.ugr_hide_summary,
            cur.ugr_autowidth, cur.ugr_header_autoheight, cur.ugr_show_info_panel,
            cur.ugr_hide_inplace_editor, cur.ugr_show_excel_import, cur.ugr_owner_user, cur.ugr_inactive,
            cur.ugr_excel, cur.ugr_ds_code, cur.ugr_group_field, cur.ugr_formatting,
            cur.ugr_tree_can_move, cur.ugr_allow_append, cur.ugr_ui_kind, cur.ugr_calendar_fields)
    returning ugr_uuid, ugr_share_uuid
    into :new_uuid, :share_uuid;

    new_code = uuid_to_char(new_uuid);

    insert into ux$gritem (ugi_ugr_uuid, ugi_uf_uuid, ugi_position, ugi_width, ugi_sizable,
                           ugi_order_index, ugi_group_index, ugi_fixed_kind, ugi_summary_kind,
                           ugi_header_align, ugi_value_align)
    select :new_uuid, ugi_uf_uuid, ugi_position, ugi_width, ugi_sizable, ugi_order_index,
           ugi_group_index, ugi_fixed_kind, ugi_summary_kind,
           ugi_header_align, ugi_value_align
    from ux$gritem
    where ugi_ugr_uuid = :ugr_uuid
    order by ugi_position;

    if (is_private = 1) then
    begin
      if (not exists(select * from ux$grshare where ugs_ugr_uuid = :share_uuid and ugs_un_id = :un_id)) then
      begin
        insert into ux$grshare (ugs_ugr_uuid, ugs_un_id)
        values (:share_uuid, :un_id);
      end
    end
  end

  if (new_uuid is null) then
    exception e_default f_format('report grid not found, uuid: %1', grid_code);

  suspend;
end
