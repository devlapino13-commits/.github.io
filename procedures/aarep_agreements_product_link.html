CREATE PROCEDURE AAREP_AGREEMENTS_PRODUCT_LINK(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    REP_AG_ID D_INTEGER,
    REP_AG_CAPTION D_STR_255,
    REP_PR_NAME D_STR_255)
AS
begin
  for
    select
      ag_id,
      ag_caption,
      coalesce(pr_memo, 'ВСЕ')
    from agreements
      inner join region_list(:rep_region_id, 1, -1, 1)
        on ag_from_reg_id = id_list
      inner join refdata
        on ag_from_rd_id = rd_id
          and rd_code in ('AgreementKinds.IPF_Sellers', 'AgreementKinds.tsp_and_ipf')
      left join prodagreement
        on pag_from_ag_id = ag_id
      left join product
        on pr_id = pag_from_pr_id
    where :rep_date_end between coalesce(ag_date_begin, '01.01.1900')
      and coalesce(ag_date_end, '31.12.2999')
      and ag_status = 1
    order by 2,3
  into
    :rep_ag_id,
    :rep_ag_caption,
    :rep_pr_name
  do suspend;
end
