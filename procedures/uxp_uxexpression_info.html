CREATE PROCEDURE UXP$UXEXPRESSION_INFO(
    UX_UUID D_UUID,
    KIND D_SMALLINT)
RETURNS (
    EXP_UX_UUID D_UUID,
    EXP_KIND D_SMALLINT,
    EXP_EXPRESSION D_STR_LONG,
    EXP_CALC_UFR_UUID D_UUID,
    EXP_MESSAGE D_STR_LONG,
    EXP_IS_SQL D_BOOL,
    EXP_ACTION D_SMALLINT,
    EXP_POSITION D_INTEGER,
    EXP_IS_BOOL D_BOOL,
    EXP_CALC_UFR_UUID_CODE D_STR_CODE,
    EXP_CALC_KIND D_SMALLINT)
AS
begin
  exp_calc_ufr_uuid_code = '';

  select exp_ux_uuid,
         exp_kind,
         exp_expression,
         exp_calc_ufr_uuid,
         exp_message,
         exp_is_sql,
         exp_action,
         exp_position,
         exp_is_bool,
         exp_calc_kind
  from ux$expression
  where exp_ux_uuid = :ux_uuid and
        exp_kind = :kind
  into :exp_ux_uuid,
       :exp_kind,
       :exp_expression,
       :exp_calc_ufr_uuid,
       :exp_message,
       :exp_is_sql,
       :exp_action,
       :exp_position,
       :exp_is_bool,
       :exp_calc_kind;

  if (row_count = 0) then
  begin
    exp_kind = kind;
    exp_ux_uuid = ux_uuid;
    exp_expression = '';
    exp_message = '';
    exp_position = 0;
    exp_is_bool = 1;
    exp_is_sql = 0;
    exp_action = 0;
    exp_calc_kind = 0;

    if (kind = 0) then
      exp_action = 0;
    else
    if (kind = 1) then
    begin
      exp_is_bool = 0;
      exp_action = 1;
    end else
    if (kind = 2) then
    begin
      exp_action = 2;
      exp_is_bool = 0;
    end else
    if (kind = 3) then
    begin
      exp_action = 7;
      exp_is_bool = 0;
    end
  end

  if (exp_calc_ufr_uuid is not null) then
    exp_calc_ufr_uuid_code = uuid_to_char(exp_calc_ufr_uuid);

  suspend;
end
