CREATE PROCEDURE HP_CLIENT_BANK_CARD(
    C_ID ID_TYPE)
RETURNS (
    TOKEN STR_100,
    TOKEN_DATETIME D_DATETIME,
    CARDNUMBER D_STR_100)
AS
begin
  token = null;

  select et.et_token, et.et_token_datetime, et.et_cardnumber
  from extsys_token et
  where et.et_from_c_id = :c_id and
        et.et_is_actual = 1
  into :token, :token_datetime, :cardnumber;

  suspend;
end
