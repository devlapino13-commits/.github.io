      select sum(osd_value)
      from opersalmain osm
      inner join opersaldetail osd on osd.osd_from_osm_id = osm.osm_id and
            osd.osd_from_sac_id = 19 and
            osd.osd_value <> 0
      where osm_ol_is_active = 1 and
            osm_ol_date between :first_day_of_rep_month and :rep_date_end and
            osm_from_c_id = :c_id
      group by osd.osd_from_sac_id
      into aarep_oklad;
      if (aarep_oklad is null) then
        aarep_oklad = 0;

      aarep_returnment_koef = 1.0;
      if (aarep_plan_all <> 0) then
        aarep_returnment_koef = round(aarep_fact_planned / aarep_plan_all, 4);
      else
        aarep_returnment_koef = 0.0;
      if (aarep_returnment_koef > 1.0) then
        aarep_returnment_koef = 1.0;

      aarep_bonus_plus = round((aarep_fact_all / 1000.0) * koef_bonus, 0);
      aarep_bonus_minus = round((aarep_delinq_os / 1000.0) * koef_penalty, 0);
      aarep_bonus_consult = round((aarep_consult_summa / 1000.0) * koef_consult, 0);
      aarep_insurance_sum = round((aarep_insurance_sum / 1000.0) * koef_insurance, 0);

      aarep_bonus_minus_sb = kol_transfer_sb * 50;

      aarep_bonus_total_fact = 0;
      aarep_bonus_total_to_pay = 0;
      aarep_bonus_total = aarep_bonus_plus + aarep_bonus_consult + aarep_insurance_sum - aarep_bonus_minus - aarep_bonus_minus_sb;
      fixed_bonus_gross = 0;

      if (aarep_oklad < 24611.11) then
        fixed_bonus_gross = 24611.11 - :aarep_oklad;

      aarep_bonus_total_fact = aarep_bonus_total - aarep_oklad - fixed_bonus_gross;

      aarep_bonus_total_to_pay = aarep_bonus_total_fact; -- бонус к начислению
      if (aarep_bonus_total_to_pay < 0) then
        aarep_bonus_total_to_pay = 0;
      aarep_bonus_ideal = round(((aarep_fact_delinq_perc + aarep_fact_kommis + aarep_month_plan_perc + aarep_fact_adv_perc + aarep_fact_fine + aarep_fact_kommis_ipf) / 1000.0) * koef_bonus, 0);

      aarep_day_bonus_plus = aarep_bonus_plus;
      aarep_day_bonus_consult = aarep_bonus_consult;
      aarep_day_bonus_minus = aarep_bonus_minus;
      aarep_day_bonus_total = aarep_bonus_total;
      aarep_day_returnment_koef = aarep_returnment_koef;

      aarep1_plan_all = 0;
      aarep1_fact_all = 0;
      aarep1_fact_planned = 0;
      aarep1_fact_other = 0;
      using_scores = 0;

      if ((extract(day from :rep_date_end) > 1) and
          (need_one_day = 1)) then
      begin
        select sum(aarep_plan_all), sum(aarep_fact_all), sum(aarep_fact_planned), sum(aarep_fact_other),
               sum(aarep_delinq_os), sum(aarep_fact_kommis), sum(aarep_consult_summa)
        from cb_kg_lo_210104_one_tmp(:reg_id, :c_id, :rep_date_end - 1)
        into :aarep1_plan_all, :aarep1_fact_all, :aarep1_fact_planned, :aarep1_fact_other, aarep1_delinq_os,
             aarep_fact_kommis, aarep1_consult_summa;

        if (aarep1_plan_all is null) then
          aarep1_plan_all = 0;
        if (aarep1_fact_all is null) then
          aarep1_fact_all = 0;
        if (aarep1_fact_planned is null) then
          aarep1_fact_planned = 0;
        if (aarep1_fact_other is null) then
          aarep1_fact_other = 0;
        if (aarep1_delinq_os is null) then
          aarep1_delinq_os = 0;

        aarep_day_returnment_koef = 1.0;
        if (aarep_plan_all - aarep1_plan_all <> 0) then
          aarep_day_returnment_koef = round((aarep_fact_planned - aarep1_fact_planned) / (aarep_plan_all - aarep1_plan_all), 4);
        else
          aarep_day_returnment_koef = 0.0;
        if (aarep_day_returnment_koef > 1.0) then
          aarep_day_returnment_koef = 1.0;
        if (aarep_day_returnment_koef < 0) then
          aarep_day_returnment_koef = 1.0;

        aarep_day_bonus_plus = round((aarep1_fact_all / 1000.0) * koef_bonus, 0);
        aarep_day_bonus_consult = round((aarep1_consult_summa / 1000.0) * koef_consult, 0);
        aarep_day_bonus_minus = round((aarep1_delinq_os / 1000.0) * koef_penalty, 0);
        aarep_day_bonus_total = aarep_day_bonus_plus + aarep_day_bonus_consult - aarep_day_bonus_minus;

        aarep_day_bonus_consult = aarep_bonus_consult - aarep_day_bonus_consult;
        aarep_day_bonus_plus = aarep_bonus_plus - aarep_day_bonus_plus;
        aarep_day_bonus_minus = aarep_bonus_minus - aarep_day_bonus_minus;
        aarep_day_bonus_total = aarep_bonus_total - aarep_day_bonus_total;
      end

      aarep_bonus_memo = '';
      last_exp = 0;
      total_exp = 0;
      days_calc = 0;

--      select last_exp_changed,
--             (last_exp_changed - last_dekr_changed), hired_date_changed
--      from hp_ks_exp_calc(:rep_date_end, :c_id)
--      into :last_exp, :total_exp, :hired_date;

      --RFCKG-1981
      select t.last_exp, t.pure_exp_last, t.ps_date_begin
      from hp_get_clean_exp(:rep_date_end, :c_id) t
      into last_exp, total_exp, hired_date;

      if (:last_exp < 3) then
      begin
        --        если дата когда устроился последний раз менее 3х месяцев то обнуляем бонус к начислению
        aarep_bonus_total_to_pay = 0;
        aarep_bonus_memo = 'Стаж менее 3х месяцев';
      end

      if (last_exp > 0) then
        select sum(sat.sat_days_calc)
        from sa_tabel sat
        where sat.sat_year_month = :year_month and
              sat.sat_from_c_id = :c_id and
              sat.sat_office_id = :reg_id and
              sat.sat_from_sawt_id = 0
        into :days_calc;
      else
        select res
        from hp_calc_worked_days(:year_month, :c_id, :rep_date_end, :hired_date)
        into :days_calc;

      fixed_bonus_gross_cacl = :fixed_bonus_gross;
      /*
      if (days_calc = 0) then
        fixed_bonus_gross_cacl = 0;
       */
      if (aarep_bonus_total_to_pay is null) then
        aarep_bonus_total_to_pay = 0;

      aarep_bonus_total_to_pay_hist = 0;
      aarep_bonus_total_to_pay = aarep_bonus_total_to_pay + fixed_bonus_gross_cacl * iif(:last_exp = 0, (days_calc / norma_days), 1);

      kontr_sum = 0;

      for select iif(pr.pr_code containing '.gr.', 'gr', 'ind') as kontr_type,
                 count(distinct(kc.kc_from_c_id)) as counted
          from kontstate ks
          inner join kontract k on k.k_id = ks.ks_from_k_id
          inner join product pr on k.k_from_pr_id = pr.pr_id and
                pr.pr_code containing '.ipf.'
          inner join knclient kc on kc.kc_from_k_id = k.k_id and
                kc.kc_amount_approved > 0
          where ks.ks_ol_is_delete = 0 and
                ks.ks_state = 1 and
                ks.ks_date_begin between :first_day_of_rep_month and :rep_date_end and
                ks.ks_region_id = :reg_id and
                ks.ks_kc_id = :c_id
          group by 1
          union all
          select iif(pr.pr_code containing '.gr.', 'gr', 'ind') as kontr_type,
                 count(distinct(kc.kc_from_c_id)) as counted
          from kontstate ks
          inner join kontract k on k.k_id = ks.ks_from_k_id
          inner join product pr on k.k_from_pr_id = pr.pr_id and
                pr.pr_code not containing '.ipf.'
          inner join knclient kc on kc.kc_from_k_id = k.k_id and
                kc.kc_amount_approved > 0
          where ks.ks_ol_is_delete = 0 and
                ks.ks_state = 1 and
                ks.ks_date_begin between :first_day_of_rep_month and :rep_date_end and
                ks.ks_region_id = :reg_id and
                ks.ks_kc_id = :c_id
          group by 1 as cursor kontr_tmp

      do
      begin
        kontr_sum = :kontr_sum + kontr_tmp.counted;
      end

      select result_deduction
      from hp_cacl_ks_fine_kp_gross(:total_exp, :days_calc, :norma_days, :aarep_par_30_coef, :kontr_sum, :rep_date_end)
      into :fine_kp_gross;

      aarep_bonus_total_to_pay = aarep_bonus_total_to_pay - fine_kp_gross;

      select bonus_fact, using_scores, dv_sum
      from hp_calc_rebuke_bonus(:aarep_bonus_total_to_pay, :c_id, current_date, :hist_date_time)
      into :aarep_bonus_total_to_pay_hist, :using_scores_hist, :dv_sum_hist;

      if (days_calc = 0) then
      begin
        if (aarep_bonus_total_to_pay_hist < 0) then
          aarep_bonus_total_to_pay_hist = 0;
        fixed_bonus_gross_cacl = 0;

        aarep_bonus_total_to_pay_hist = fixed_bonus_gross_cacl - fine_kp_gross + aarep_bonus_total_to_pay_hist;
      end 

      if (aarep_bonus_total_to_pay_hist < 0) then
        aarep_bonus_total_to_pay_hist = 0;


      select bonus_fact, using_scores, dv_sum
      from hp_calc_rebuke_bonus(:aarep_bonus_total_to_pay, :c_id, current_date)
      into :aarep_bonus_total_to_pay, :using_scores, :dv_sum;

      if (aarep_bonus_total_to_pay < 0) then
        aarep_bonus_total_to_pay = 0;

      if (rep_c_id <> 0) then
        suspend;
      else
      if ((aarep_port_balance_all <> 0) or (aarep_bonus_total <> 0)) then
        suspend;
    end
  end
