CREATE PROCEDURE HP_STOP_CREDIT_STATUS_CLIENT(
    C_ID D_ID,
    REP_DATETIME D_DATETIME)
RETURNS (
    EIL_ID D_ID,
    STATUS D_SMALLINT,
    QUERY_DATETIME D_DATETIME,
    ANSWER_DATETIME D_DATETIME)
AS
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable STATUS_TEMP D_STR_10;
declare variable EIL_STATUS D_SMALLINT;
begin
  -- Процедура вернет статус запроса проверки клиента на STOP_CREDIT на заданную дату/время (RFCKZ-370)
  -- Процедура создана на базе hp_stop_credit_query_status

  /*
  Возможные статусы (status):
      1 -- есть активный стоп кредит (то есть, для данной заявки существует запись в таблице extinfolist с соответствующим json, содержащим положительный ответ от сервиса),
           актуальность этой записи требуется проверять отдельно
      0 -- нет активного стоп кредита (то есть, для данной заявки сущесвтует запись в таблице extinfolist с соответствующим json, содержащим отрицательный ответ от сервиса),
           актуальность этой записи требуется проверять отдельно
     -1 -- запрос еще не создавался (нет записи в extinfolist для данной заявки)
     -2 -- запрос создан, но ответ еще не получен. Данный статус соответствует статусам (-1, 0, 1) поля extinfolist.eil_status
     -3 -- сервис ответил с ошибкой. Данный статус соответствует статусам (-2, -3) поля extinfolist.eil_status
  */

  select eil.eil_id,
         eil.eil_from_bls_id,
         eil.eil_create_datetime,
         eil.eil_status_datetime,
         eil.eil_status
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eic.eic_code = 'stop_credit' and
        eil.eil_from_c_id = :c_id and
        eil.eil_status_datetime <= :rep_datetime
  order by eil.eil_id desc
  rows 1
  into :eil_id,
       :bls_id,
       :query_datetime,
       :answer_datetime,
       :eil_status;

  -- Запрос не создавался (запись не существует)
  if (eil_status is null) then
  begin
    status = -1;
    suspend;
    exit;
  end

  -- Сервис ответил с ошибкой, запрос неуспешен
  if (eil_status in (-2, -3)) then
  begin
    status = -3;
    suspend;
    exit;
  end

  if (eil_status in (-1, 0, 1) or coalesce(:bls_id, 0) = 0) then
  begin
    status = -2; -- Запрос был отправлен, но ответ еще не получен
    suspend;
    exit;
  end

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into json_text;

  status_temp = left(f_json_str_get_value(:json_text, 'result', ''), 10);

  status = (case status_temp
              when 'true' then 1
              when 'false' then 0
              else -3 -- RFCKZ-23: Если в ответе значение не совпадает с ожидаемым, считаем, что сервис ответил с ошибкой
            end);

  suspend;
end
