CREATE PROCEDURE OKT_CHECK_ACTION(
    ACL_ID ID_TYPE)
RETURNS (
    IS_ACTIVE SMALLINT)
AS
declare variable rd_code str_255;
declare variable object_id id_type;
declare variable cc_verify_all_contacts integer;
declare variable do_continue smallint;
begin
  /*
    OKTELL
    процедура проверяет, активна задача или нет.
    IS_ACTIVE (1-активна; 0-не активна)
  */

  is_active = 0;

  object_id = null;

  select ao.aco_object_id, rd.rd_code
  from actionorder ao
  inner join refdata rd on rd.rd_id = ao.aco_from_rd_id
  where ao.aco_id = :acl_id
  into :object_id, :rd_code;

  if (object_id is not null) then
    is_active = 1; -- если задача активна

  if (is_active = 1) then
  begin
    if (rd_code = 'OKTELL.Projects.Verification') then
    begin
-- подлежит ли контакт верификации (1 - не подлежит)
      if ((select coalesce(cont.cc_verif_ignore, 0)
           from contacts cont
           where cont.cc_id = :object_id) = 1) then
      begin
        is_active = 0;
        suspend;
        exit;
      end

      select cm.cm_cc_verify_all_contacts
      from company cm
      into :cc_verify_all_contacts;

      cc_verify_all_contacts = coalesce(cc_verify_all_contacts, 1);

      do_continue = 0;

      if (cc_verify_all_contacts = 1) then
        do_continue = 1;
      else
      begin
        if (exists(select *
                   from hp_contacts_check_complete(18, :acl_id))) then
          do_continue = 1;
      end

      if (do_continue = 0) then
        is_active = 0;
    end
  end

  suspend;
end
