CREATE PROCEDURE HP_EDOC_SEND_PUSH_INTERN(
    OBJ_ID D_ID)
AS
declare variable reg_id d_id;
declare variable c_id d_id;
declare variable mail_to d_str_100;
declare variable reg_name d_str_255;
declare variable c_name_fio d_str_255;
declare variable cont_number d_str_100;
declare variable msg_subject d_str_255;
declare variable msg_text d_str_255;
declare variable sms_subject d_str_code;
declare variable sms_body d_str_255;
declare variable un_id d_id;
begin
  select srd.sd_from_reg_id, srd.sd_from_c_id
  from servicedog srd
  where srd.sd_id = :obj_id
  into reg_id, c_id;

  select c.c_name_fio
  from clients c
  where c.c_id = :c_id
  into c_name_fio;

  select reg.reg_name, reg.reg_email_manager
  from region reg
  where reg.reg_unicode = :reg_id
  into reg_name, mail_to;
  cont_number = f_cont_info('clients', :c_id, 'cell_main', 'ok');
  msg_subject = 'ДОГОВОР НА СТАЖИРОВКУ ПОДПИСАН ' || :c_name_fio || ',  офис ' || :reg_name;
  msg_text = msg_subject || ', ' || coalesce(cont_number, '');
  insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_from_un_id)
  values (4, 'edu_kg.mirm@mbulak.kg;' || coalesce(:mail_to, ''), :msg_subject, :msg_text, 1000000000000);

  select un.un_id
  from username un
  where un.un_from_c_id = :c_id
  into un_id;

  un_id = coalesce(un_id, -1);

  if (un_id = -1) then
  begin
    select result_un_id
    from hp_create_new_user(cast(:c_id as d_str_20), :c_id, :c_name_fio, :reg_id, 0, 20, 1000000000000)
    into un_id;
  end

  sms_subject = 'Академия Акылдуу';
  sms_body = 'Доступ к Академии Акылдуу https://academy.mbulak.kg?uid=' || :un_id;

  execute procedure hp_send_quick_message('sms', :cont_number , :sms_subject, :sms_body, '', -1, 1000000000000);
end
