CREATE PROCEDURE FM_DOCRECOGREQUEST_DETAIL(
    OBJECT_KIND D_STR_CODE,
    OBJECT_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable c_name_fio d_str_100;
declare variable dp_name d_str_100;
declare variable drm_name d_str_100;
declare variable drr_status d_integer;
declare variable drr_status_str d_str_100;
declare variable drr_create_datetime d_datetime;
declare variable drr_status_datetime d_datetime;
declare variable lang d_str_lang;
declare variable th_fmt d_str_100;
declare variable tr_fmt d_str_100;
declare variable tr_lnk_fmt d_str_100;
declare variable status_fmt d_str_100;
declare variable link d_str_255;
declare variable c_id d_id;
declare variable drl_id d_id;
begin
  res_text = (select res_style
              from fm_html_styles) || '<body>';

  lang = f_get_lang_code(un_id, null);
  tr_fmt = '<tr><td><b>%1</b></td><td>%2</td></tr>';
  tr_lnk_fmt = '<tr><td><b>%2</b></td><td>%1</td></tr>'; -- Реверсивный, чтобы f_format не ломался из-за %20 (пробелов)
  th_fmt = '<tr><th colspan="2">%1</th></tr>';
  status_fmt = '%1 %2';

  res_text = blob_append(res_text, '<table width="100%"><col width="30%"><col width="70%">');
  res_text = blob_append(res_text, f_format(th_fmt, f_lang(lang, 'Детали запроса на фото документа')));


  for select drr_from_drl_id, c_id, c_name_fio, dp_name, drm_name, drr_status,
    drr_create_datetime, drr_status_datetime
  from docrecogrequest
       inner join clients on (drr_from_c_id = c_id)
       inner join docprop on (drr_from_dp_id = dp_id)
       left join docrecogmain on (drr_from_drm_id = drm_id)
  where drr_id = :object_id
  into drl_id, c_id, c_name_fio, dp_name, drm_name, drr_status,
    drr_create_datetime, drr_status_datetime
  do
  begin
    drr_status_str = case drr_status
                   when -2 then f_format(status_fmt, f_emoji('cross mark'), f_lang(lang, 'Запрос аннулирован'))
                   when -1 then f_format(status_fmt, f_emoji('counterclockwise arrows button'), f_lang(lang, 'Ждем замены документа от клиента'))
                   when 0 then f_format(status_fmt, f_emoji('hourglass not done'), f_lang(lang, 'Ждем фото документа от клиента'))
                   when 1 then f_format(status_fmt, f_emoji('check mark button'), f_lang(lang, 'Подтвердите качество и корректность документа'))
                   when 2 then f_format(status_fmt, f_emoji('page facing up'), f_lang(lang, 'Документ готов'))
                 end;

    res_text = blob_append(res_text, f_format(tr_fmt, f_lang(lang, 'Клиент'), c_name_fio));
    res_text = blob_append(res_text, f_format(tr_fmt, f_lang(lang, 'Тип документа'), dp_name));
    if (drm_name is not null) then
      res_text = blob_append(res_text, f_format(tr_fmt, f_lang(lang, 'Документ'), drm_name));
    res_text = blob_append(res_text, f_format(tr_fmt, f_lang(lang, 'Статус'), drr_status_str));
    res_text = blob_append(res_text, f_format(tr_fmt, f_lang(lang, 'Время создания'), f_datetimetostr(drr_create_datetime)));
    res_text = blob_append(res_text, f_format(tr_fmt, f_lang(lang, 'Время изменения'), f_datetimetostr(drr_status_datetime)));
    res_text = blob_append(res_text, '</table><br>');


    res_text = blob_append(res_text, '<table width="100%"><col width="30%"><col width="70%">');
    res_text = blob_append(res_text, f_format(th_fmt, f_lang(lang, 'Действия')));

    if (drr_status = 1 and drl_id is not null) then
    begin
      link = f_uri_open(f_lang(lang, 'Верифицировать'),
        'doc_recog_add', object_kind, object_id,
        '', 'drl_id=' || drl_id, un_id);
      res_text = blob_append(res_text, f_format(tr_lnk_fmt, f_emoji('memo') || ' ' || f_lang(lang, 'Открыть документ и проверить/заполнить нужные поля'), link));
    end

    if (drr_status in (-1, 0)) then
    begin
      link = f_uri_sql(f_lang(lang, 'Аннулировать'), 'docrecogrequest_cancel',
        object_kind, object_id, 'none', f_lang(lang, 'Подтвердите аннулирование запроса'),
        null, null, null, null, null, 'dialog', '');

      res_text = blob_append(res_text, f_format(tr_lnk_fmt, f_emoji('cross mark') || ' ' || f_lang(lang, 'Удалить запрос документа'), link));
    end

    -- Общие действия
    link = f_uri_open(f_lang(lang, 'Клиент'), 'client', 'clients', c_id, left(c_name_fio, 50), '', un_id);
    res_text = blob_append(res_text, f_format(tr_lnk_fmt, f_emoji('person') || ' ' || f_lang(lang, 'Открыть окно клиента'), link));

    res_text = blob_append(res_text, '</table></body>');
  end
  suspend;
end
