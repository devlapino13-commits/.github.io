Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$TRANSTEMP_MOVE
AS
declare variable orig_text d_str_255;
declare variable is_delphi d_bool;
declare variable orig_is_delphi d_bool;
begin
  for select orig_text, is_delphi
  from ux$transtemp
  into :orig_text, is_delphi
  do
  begin
    select trm_is_delphi
    from ux$transmain
    where trm_orig_text = :orig_text
    into orig_is_delphi;

    if (row_count = 0) then
    begin
      insert into ux$transmain (trm_orig_text, trm_is_delphi)
      values (:orig_text, :is_delphi);

      when any do
        exit;
    end else
    if (coalesce(orig_is_delphi, 0) < is_delphi) then
    begin
      update ux$transmain
      set trm_is_delphi = :is_delphi
      where trm_orig_text = :orig_text;
    end
  end
end
