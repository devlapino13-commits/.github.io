CREATE PROCEDURE HP_EXPRESSAPP_FILL_ONE(
    C_ID ID_TYPE,
    REP_DATE D_DATE,
    NEED_INSERT D_SMALLINT = 0,
    CLI_ACTIVE D_INTEGER = -1)
RETURNS (
    RES D_SMALLINT,
    RES_TEXT STR_100)
AS
declare variable K_ID ID_TYPE;
declare variable ASP_STATUS D_STR_CODE;
declare variable IS_FIRST D_SMALLINT;
declare variable BASE_K_ID ID_TYPE;
declare variable LGC_CALC D_INTEGER;
declare variable PR_ID D_INTEGER;
declare variable EXA_AMOUNT CURR_TYPE;
declare variable EXA_PERIOD D_INTEGER;
declare variable EXA_LO_ID ID_TYPE;
declare variable EXA_FROM_REG_ID integer;
declare variable EXA_PERCENT D_PERCENT;
declare variable EXA_REPAY_DAY D_SMALLINT; /* погашение по 25м числам */
declare variable EXA_FROM_UN_ID ID_TYPE;
declare variable LOANDISBDATE D_DATE;
declare variable C_CURR_BALL D_INTEGER;
declare variable K_DISB_DATE D_DATE;
declare variable CLI_BALANCE CURR_TYPE;
declare variable DELINQ_DAY D_INTEGER;
declare variable MAX_DELINQ_DAY D_INTEGER;
declare variable POINT D_INTEGER;
declare variable MAX_BALANCE CURR_TYPE;
begin
  if (rep_date is null) then rep_date = current_date;
  res  = 1;
  res_text = '';

  -- исключаем из ЧС
  if (exists(select * from moratory
  where m_from_c_id = :c_id
  and m_is_active = 1
  and m_is_black = 2)) then
  begin
    res_text = 'Клиент в стоп листе.';
  end

-- если клиенту отказали
  if (res_text = '') then
  begin
    if (exists(select * from kontreslist, knclient
      where krl_datetime between :rep_date - 30 and :rep_date
      and krl_from_k_id = kc_from_k_id
      and kc_amount_approved > 0
      and kc_from_c_id = :c_id)) then
    begin
      res_text = 'Ранее клиенту отказали на уровне контракта.';
    end
  end 

  if (res_text = '') then
  begin
    if (exists(select * from kontreslist, miniappclient
      where krl_datetime between :rep_date - 30 and :rep_date
      and krl_ma_id = mac_from_ma_id
      and mac_amount > 0
      and mac_from_c_id = :c_id)) then
    begin
      res_text = 'Ранее клиенту отказали на уровен заявки.';
    end
  end 

  if (res_text = '') then
  begin
    if (exists(select *
      from actionlist, refdata, knclient
      where acl_from_rd_id = rd_id and
            acl_date between :rep_date - 30 and :rep_date and
            rd_main_code = 'actiongroup' and
            rd_str_id = 'underwriter' and
            acl_dont_show_in_order = 0 and
            acl_object_kind = 9 and
            acl_status = -2 and
            acl_object_id = kc_from_k_id and
            kc_amount_approved > 0 and
            acl_dont_show_in_order = 0 and
            kc_from_c_id = :c_id)) then
    begin
      res_text = 'Ранее клиенту отказали(андерайтеры)';
    end
  end

  -- сотрудников исключаем
  if (res_text = '' and
  exists(select * from hr_pstatus
  where ps_from_c_id = :c_id
  and :rep_date between ps_date_begin and ps_date_end
  and ps_status = 1
  and ps_is_working = 1)) then
  begin
    res_text = 'Клиент сотрудник';
  end

  -- проверка на действующего договора АСП
  if (res_text = '') then
  begin
    asp_status = f_edoc_status('asp.agreement', :c_id, :rep_date);
    if (asp_status <> 'signed') then
    begin
      res_text = 'нет действующего АСП';
    end
  end

  -- проверка имеется ли МП у клиента
  if (res_text = '' and f_fsm_client_status(c_id) = 0) then
  begin
    res_text = 'У клиента не имеются КМП';
  end 

  if (res_text = '') then
  begin
    if (cli_active = -1 or cli_active is null) then
    begin
      if (exists(select * from kontstate, knclient
      where ks_ol_is_delete = 0
      and ks_state = 1
      and :rep_date between ks_date_begin and ks_date_end
      and ks_from_k_id = kc_from_k_id
      and kc_amount_approved > 0
      and kc_from_c_id = :c_id)) then cli_active = 1;
      else cli_active = 0;
    end
    -- вычисляем основной контракт
    if (cli_active = 1) then
    begin
      select ks_from_k_id from kontstate, knclient
      where ks_ol_is_delete = 0
      and ks_state = 1
      and :rep_date between ks_date_begin and ks_date_end
      and ks_from_k_id = kc_from_k_id
      and kc_amount_approved > 0
      and kc_from_c_id = :c_id
      order by ks_date_begin desc
      rows 1
      into base_k_id;
    end else
    begin
      select ks_from_k_id from kontstate, knclient
      where ks_ol_is_delete = 0
      and ks_state = 1
      and ks_from_k_id = kc_from_k_id
      and kc_amount_approved > 0
      and kc_from_c_id = :c_id
      order by ks_date_begin desc
      rows 1
      into base_k_id;
    end 
  end

  cli_balance = null;
  delinq_day = null;
  lgc_calc = null;
  max_delinq_day = null;
  point = null;
  max_balance = null;

  if (res_text = '' and cli_active = 1) then
  begin
    --  проверка на просрочку
    select sum(oks_balance),
           max(:rep_date - oks_delinq_date)
    from knclient, oksummary
    where kc_from_c_id = :c_id
    and kc_amount_approved > 0
    and kc_from_k_id = oks_from_k_id
    and :rep_date between oks_date_begin and oks_date_end
    and oks_close_state = 0
    into cli_balance, delinq_day;

    cli_balance = coalesce(cli_balance, 0);
    delinq_day = coalesce(delinq_day,0);
    if (delinq_day < 0) then delinq_day = 0;

    if (delinq_day > 0 ) then -- если имеется просрочка то исключаем
    begin
      res_text = 'Имеется активная просрочка';
    end 

    if (res_text = '') then
    begin
      lgc_calc = null;
      select sum(lgc_calc)
      from legitimcalc
      where lgc_from_c_id = :c_id and
            lgc_from_k_id = :base_k_id and
            lgc_date <= :rep_date
      into lgc_calc;
      lgc_calc = coalesce(lgc_calc,0);
      if (lgc_calc < 2) then
      begin
        res_text = 'Не имеются легитимных платежей';
      end
   end

    if (res_text = '') then
    begin
      point = f_point(c_id, rep_date);

      max_balance = case
                      when point >= 24 then 150000
                      when point >= 18 then 120000
                      when point >= 15 then 100000
                      when point >= 12 then 80000
                      when point >= 9 then 60000
                      when point >= 6 then 50000
                      when point >= 3 then 40000
                      else 35000
                    end;
      if (max_balance - cli_balance < 35000) then
      begin
        res_text = 'Не соответсвует сумма по баллам';
      end
    end
  end

  if (res_text = '' and cli_active = 0) then
  begin
    -- если клиент не активный то по последннему циклу проверяем максимальныеп просроченные дни
    -- если превышает 30ти дней то исключаем
    max_delinq_day = null;
    select max(ok_ol_date - ok_sh_date) from operkred
    where ok_from_k_id = :base_k_id
    and ok_ol_is_active = 1
    and ok_ol_date - ok_sh_date > 0
    and ok_type in (4,5)
    and ok_os + ok_perc < 0
    into max_delinq_day;
    max_delinq_day = coalesce(max_delinq_day, 0);
    if (max_delinq_day > 30) then
    begin
      res_text = 'Последний контракт закрылся с просрочкой более 30 дней';
    end 
  end


  if (res_text = '') then
  begin
    -- вычисляем ВИПС и регион
    select kal_new_kf_officer, kal_new_reg_id, k_repay_day, k_disb_date
    from kontract, knaktlist
    where k_id = :base_k_id
    and k_from_kf_id = kal_from_kf_id
    and knaktlist.is_delete = 0
    and current_date between kal_date and kal_date_end
    into exa_lo_id, exa_from_reg_id, exa_repay_day, k_disb_date;
    
    exa_repay_day = coalesce(exa_repay_day, 0);
    if (exa_repay_day = 0) then exa_repay_day = extract(day from k_disb_date);

    if (exa_repay_day >= 25)
    then exa_repay_day = 1;
    else exa_repay_day = exa_repay_day + 2;

    if (exa_lo_id <= 1000000000004 and
        cli_active = 0 and
        max_delinq_day > 7) then  -- если находится на Виртуальном сотруднике то проверяем на просрочку более 7ми дней
    begin
      res_text = 'Папка на виртуальном сотруднике и имеет просрочку более 7ми дней';
    end
  end

  if (res_text = '' and need_insert = 1) then
  begin
    loandisbdate = current_date;
    pr_id = 146; -- продукт доверие
    exa_period = 12; -- период
--    exa_repay_day = 25; -- погашение по 25м числам
    exa_from_un_id = 1000000000000; -- FINSTART_ROBOT

    c_curr_ball = null;
    c_curr_ball = f_point(:c_id, :loandisbdate);
    if (c_curr_ball is null) then
      c_curr_ball = 0;
    exa_percent = 8; -- процентная ставка
    exa_amount = 35000; -- сумма выдачи
    select percent, amount_max
    from perc_amount_interval_by_point(:pr_id, :c_curr_ball)
    into exa_percent, exa_amount;

    insert into expressapp (exa_id, exa_date, exa_from_c_id, exa_base_k_id, exa_from_exc_id, exa_final_status,
                            exa_wait_status, exa_from_ma_id, exa_from_k_id, exa_from_edl_id, exa_disb_date, exa_from_pr_id,
                            exa_amount, exa_period, exa_percent, exa_repay_day, exa_lo_id, exa_from_reg_id, exa_try_count,
                            exa_from_un_id, exa_datetime, exa_from_exp_id, exa_group_id)
    values (null, current_date, :c_id, :base_k_id, 10, 0, 0,
            null, null, null, :loandisbdate, :pr_id, :exa_amount, :exa_period,
            :exa_percent, :exa_repay_day, :exa_lo_id, :exa_from_reg_id, 0, :exa_from_un_id, current_timestamp, 2, 5);
  end

  if (res_text <> '') then res = 0;
  suspend;
end
