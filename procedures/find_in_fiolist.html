CREATE PROCEDURE FIND_IN_FIOLIST(
    FIND_TXT VARCHAR(100))
RETURNS (
    FL_ID DECIMAL(18,0))
AS
declare variable full_text varchar(100);
declare variable txt1 varchar(100);
declare variable txt2 varchar(100);
declare variable txt3 varchar(100);
declare variable clear_txt varchar(100);
begin
  if (coalesce(find_txt, '') = '') then
    exit;

  fl_id = null;
  select full_text, txt1, txt2, txt3
  from hp_fio_clear(:find_txt)
  into :full_text, :txt1, :txt2, :txt3;
  clear_txt = trim(full_text);
  while (position(' ', clear_txt) > 0) do
    clear_txt = replace(clear_txt, ' ', '');
  if (char_length(clear_txt) >= 3) then
  begin
    if (txt3 <> '') then
    begin
-- поиск по трем словам
      for select fl_id
          from fiolist
          where (fl_name1 starting :txt1 and
                fl_name2 starting :txt2 and
                fl_name3 starting :txt3) or
                (fl_name1 starting :txt1 and
                fl_name2 starting :txt3 and
                fl_name3 starting :txt2) or
                (fl_name1 starting :txt2 and
                fl_name2 starting :txt1 and
                fl_name3 starting :txt3) or
                (fl_name1 starting :txt2 and
                fl_name2 starting :txt3 and
                fl_name3 starting :txt1) or
                (fl_name1 starting :txt3 and
                fl_name2 starting :txt1 and
                fl_name3 starting :txt2) or
                (fl_name1 starting :txt3 and
                fl_name2 starting :txt2 and
                fl_name3 starting :txt1)
          plan(fiolist index(fiolist_name1, fiolist_name2, fiolist_name3))
          into :fl_id
      do
      begin
        suspend;
      end
    end
    else
    begin
      if (txt2 <> '') then
      begin
-- поиск по двум словам
        for select fl_id
            from fiolist
            where (fl_name1 starting :txt1 and
                  fl_name2 starting :txt2) or
                  (fl_name1 starting :txt1 and
                  fl_name3 starting :txt2) or
                  (fl_name2 starting :txt1 and
                  fl_name1 starting :txt2) or
                  (fl_name2 starting :txt1 and
                  fl_name3 starting :txt2) or
                  (fl_name3 starting :txt1 and
                  fl_name1 starting :txt2) or
                  (fl_name3 starting :txt1 and
                  fl_name2 starting :txt2)
            plan(fiolist index(fiolist_name1, fiolist_name2, fiolist_name3))
            into :fl_id
        do
        begin
          suspend;
        end
      end
      else
      begin
-- поиск через одно слово
        for select fl_id
            from fiolist
            where fl_name1 starting :txt1 or
                  fl_name2 starting :txt1 or
                  fl_name3 starting :txt1
            plan(fiolist index(fiolist_name1, fiolist_name2, fiolist_name3))
            into :fl_id
        do
        begin
          suspend;
        end
      end
    end
  end
end
