CREATE PROCEDURE HP_INSERT_CLIENTDEVICEKONT(
    K_ID D_ID)
AS
declare variable c_id d_id;
declare variable cdv_id d_id;
declare variable el_id d_id;
declare variable cdv_update_datetime timestamp;
declare variable el_code d_str_code;
declare variable k_disb_date d_date;
begin

  select knc.kc_from_c_id, k.k_disb_date
  from knclient knc, kontract k
  where k.k_id = knc.kc_from_k_id and
        knc.kc_from_k_id = :k_id and
        knc.kc_amount_approved > 0 rows 1
  into :c_id, k_disb_date;

  if (c_id is null) then
  begin
    exit;
  end

  -- Перебираем элементы из enum_list с видом 'client_device'
  for select en.el_id, en.el_code
      from enum_list en
      where en.el_kind = 'client_device'
      order by en.el_id
      into :el_id, el_code
  do
  begin
    cdv_id = 0;
    cdv_update_datetime = null;

    if (el_code <> 'ip_address') then
    begin
      -- Находим последнее обновление clientdevice
      select cdv_id, cdv_update_datetime
      from clientdevice cdv
      where cdv_from_el_id = :el_id and
            cdv_from_c_id = :c_id and
            cdv.cdv_create_datetime < (cast(:k_disb_date as d_date)+ 1)
      order by cdv_update_datetime desc
      rows 1
      into :cdv_id, :cdv_update_datetime;

      -- Если запись найдена, вставляем или обновляем данные в clientdevicekont
      if (cdv_id > 0) then
      begin
        update or insert into clientdevicekont (cdk_from_cdv_id, cdk_from_k_id, cdk_update_datetime)
        values (:cdv_id, :k_id, :cdv_update_datetime)
        matching (cdk_from_cdv_id, cdk_from_k_id);
      end
    end
    else
    begin
      -- Обрабатываем ip_address
      for select cdv.cdv_id, cdv.cdv_update_datetime
          from clientdevice cdv
          where cdv_from_c_id = :c_id and
                cdv_from_el_id = f_enum('client_device', 'ip_address', 'id') and
                cdv.cdv_create_datetime < (cast(:k_disb_date as d_date)+ 1)
          order by cdv_update_datetime desc
          rows 5
          into :cdv_id, :cdv_update_datetime
      do
      begin
        if (cdv_id > 0) then
        begin
          --Если запись найдена, вставляем или обновляем данные в clientdevicekont
          update or insert into clientdevicekont (cdk_from_cdv_id, cdk_from_k_id, cdk_update_datetime)
          values (:cdv_id, :k_id, :cdv_update_datetime)
          matching (cdk_from_cdv_id, cdk_from_k_id);
        end

      end
    end
  end
end
