CREATE PROCEDURE OKT_SIRENA_LIST(
    REP_DATE D_DATE,
    PROJECT_CODE STR_100,
    SUBPROJECT_CODE STR_100,
    CALLED_OPER STR_100,
    K_FOLDER_NAME STR_100,
    KONTRACT_NO D_STR_CODE,
    CITIZENSHIP_CODE D_STR_100,
    DELINQ_DAYS_MIN D_INTEGER,
    DELINQ_DAYS_MAX D_INTEGER,
    MAX_REC_QTY D_INTEGER,
    CALL_CHECK D_INTEGER = 0,
    HIDE_BLOCKED_GROUPS D_SMALLINT = 0,
    IGNORE_COMMANDS D_SMALLINT = 0,
    FOR_DB_ID D_INTEGER = 81001)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    KF_NAME STR_100,
    REG_NAME STR_100,
    C_NAME_FIO STR_100,
    CITIZEN_STR STR_20,
    DELINQ_DAYS D_INTEGER,
    CL_TRY_QTY D_INTEGER,
    CL_CALL_QTY D_INTEGER,
    CL_LAST_CALL_DATE D_DATE,
    CL_LAST_TRY_DATE D_DATE,
    CALLED_OPER_STR STR_100,
    C_PLASTCARD_NO STR_20,
    PANT_ID ID_TYPE,
    CON_TRY_QTY D_INTEGER,
    CON_CALL_QTY D_INTEGER,
    RESULT_STR VARCHAR(1000),
    NUM_ID D_INTEGER,
    UN_ID ID_TYPE,
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    KC_ID ID_TYPE,
    CONT_CALL_IS_ACTIVE D_SMALLINT,
    CONT_CALL_IS_ACTIVE_REAL D_SMALLINT,
    NEXT_CALL_CHECK_OK D_INTEGER,
    IGNORE_LIST_CHECK_OK D_INTEGER,
    CAN_CALL D_INTEGER,
    COMMAND_ID D_INTEGER,
    REG_ID D_INTEGER,
    CLI_QTY_COUNTER D_INTEGER)
AS
declare variable KF_ID ID_TYPE;
declare variable IS_OK D_SMALLINT;
declare variable PANT_DATE_BEGIN D_DATE;
declare variable REC_QTY D_INTEGER;
declare variable CL_CALL_RESULT STR_100;
declare variable CL_MEMO varchar(1000);
declare variable CITIZENSHIP D_INTEGER;
declare variable IGNORE_UN_ID ID_TYPE;
declare variable PROJECT_CODE_ID D_INTEGER;
declare variable CELL_MAIN_ID D_INTEGER;
declare variable CALL_CENTER_PANTERA_COMMANDS D_BOOL;
declare variable UN_ID_OPER D_ID;
begin
  rec_qty = 0;
  citizenship_code = coalesce(citizenship_code, '');
  citizenship = null;
  if (f_get_db_id() <> :for_db_id) then
    exit;

  if (ignore_commands = 0) then
  begin
      if (f_get_const('call_center_pantera_commands') = 1) then
        call_center_pantera_commands = 1; -- 1 - перешли на командную работу
  end
  else
    call_center_pantera_commands = 0;

  select rd.rd_id
  from refdata rd
  where rd.rd_code = :project_code
  into :project_code_id;

  if ((call_center_pantera_commands = 1) and
     (called_oper <> 'OKTELL_LIST') and
     (called_oper <> '')) then
  begin
    select un.un_id
    from username un
    where un.un_name = :called_oper
    into :un_id_oper;
    
    select command_id
    from get_oper_command(:un_id_oper, :project_code_id, :rep_date)
    into :command_id;
    
    if (un_id_oper is null) then exit;
    if (command_id is null) then exit;
  end

  if (citizenship_code <> '') then
  begin
    select rd.rd_id
    from refdata rd
    where rd.rd_code = :citizenship_code
    into :citizenship;
  end

  citizenship = coalesce(citizenship, -1);
  max_rec_qty = coalesce(max_rec_qty, 0);

  select un_id
  from username un
  where un.un_name = 'IGUSEV' -- Исключаю свой логин, т.к. я являюсь виртуальным опертором
  into :ignore_un_id;

  for select k_id, kf_id, k_kontract_no, delinq_days, oks_kc_id, reg_id, cli_qty_counter
      from okt_sirena_list_tmp(:rep_date, :project_code, :subproject_code, :k_folder_name, :kontract_no, :delinq_days_min, :delinq_days_max, :called_oper)
      into :k_id, :kf_id, :k_kontract_no, :delinq_days, :kc_id, :reg_id, :cli_qty_counter
  do
  begin
    is_ok = 1;
    pant_id = 0;
    called_oper_str = '';
    pant_date_begin = null;

    un_id = null;

    if (is_ok = 1) then
    begin
      is_ok = 0;
      if (called_oper = '') then --  если это блокировка
      begin
        select pantk.pant_oper_called,
               pantk.pant_date_begin,
               pantk.pant_from_un_id,
               pantkc.pkc_from_pcm_id
        from pantera_kontracts pantk
        left join pantera_kontracts_com  pantkc on pantk.pant_id = pantkc.pkc_from_pant_id
        where pantk.pant_rd_project_type = :project_code_id and
              :rep_date between pantk.pant_date_begin and pantk.pant_date_end and
              pantk.pant_from_k_id = :k_id rows 1
        into :called_oper_str,
             :pant_date_begin,
             :un_id,
             :command_id;

        if (pant_date_begin = :rep_date) then
          is_ok = 1;
        else if (pant_date_begin is null) then
          is_ok = 1;

        if (un_id = :ignore_un_id) then
          is_ok = 0;
      end
      else
      if (called_oper <> 'OKTELL_LIST') then
      begin
        if (call_center_pantera_commands = 1) then -- 1= перешли на командную работу
        begin
            select pantk.pant_id,
                   pantk.pant_from_num_id,
                   pantk.pant_from_un_id,
                   pantkc.pkc_from_pcm_id
            from pantera_kontracts pantk
            left join pantera_kontracts_com  pantkc on pantk.pant_id = pantkc.pkc_from_pant_id
            where pantk.pant_rd_project_type = :project_code_id and
                  :rep_date between pantk.pant_date_begin and pantk.pant_date_end and
                  pantk.pant_from_k_id = :k_id and
                  pantkc.pkc_from_pcm_id = :command_id
            rows 1
            into :pant_id,
                 :num_id,
                 :un_id,
                 :command_id;
        end
        else
        begin
            select pantk.pant_id,
                   pantk.pant_from_num_id,
                   pantk.pant_from_un_id,
                   pantkc.pkc_from_pcm_id
            from pantera_kontracts pantk
            left join pantera_kontracts_com  pantkc on pantk.pant_id = pantkc.pkc_from_pant_id
            where pantk.pant_rd_project_type = :project_code_id and
                  :rep_date between pantk.pant_date_begin and pantk.pant_date_end and
                  pantk.pant_oper_called = :called_oper and
                  pantk.pant_from_k_id = :k_id
            rows 1
            into :pant_id,
                 :num_id,
                 :un_id,
                 :command_id;        
        end

        called_oper_str = called_oper;

        if (pant_id is null) then
          pant_id = 0;
        if (pant_id <> 0) then
          is_ok = 1;

        if (un_id = :ignore_un_id) then
          is_ok = 0;
      end
      else
      if (called_oper = 'OKTELL_LIST') then
      begin
        select pantk.pant_id,
               pantk.pant_from_num_id,
               pantk.pant_oper_called,
               pantk.pant_from_un_id,
               pantkc.pkc_from_pcm_id
        from pantera_kontracts pantk
        left join pantera_kontracts_com  pantkc on pantk.pant_id = pantkc.pkc_from_pant_id
        where pantk.pant_rd_project_type = :project_code_id and
              pantk.pant_date_end = '30.12.9999' and
              pantk.pant_oper_called <> '' and
              pantk.pant_from_k_id = :k_id rows 1
        into :pant_id,
             :num_id,
             :called_oper_str,
             :un_id,
             :command_id;

        if (pant_id is null) then
          pant_id = 0;
        if (pant_id <> 0) then
          is_ok = 1;

        if (un_id = :ignore_un_id) then
          is_ok = 0;
      end

      if (is_ok = 1) then
      begin
        select kal.kal_new_kf_name,
               reg.reg_name
        from knaktlist kal
        inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
        where kal.kal_from_kf_id = :kf_id and
              kal.is_delete = 0 and
              :rep_date between kal.kal_date and kal.kal_date_end rows 1
        into :kf_name,
             :reg_name;

        if (called_oper <> 'OKTELL_LIST') then
        begin
          for select cl.c_id, cl.c_name_fio, rd.rd_name,
                     f_cont_info('clients', kncl.kc_from_c_id, 'cell_main', 'last')
              from knclient kncl
              inner join clients cl on cl.c_id = kncl.kc_from_c_id
              inner join refdata rd on rd.rd_id = cl.c_citizen
              where kncl.kc_from_k_id = :k_id and
                    kncl.kc_amount_approved > 0 and
                    ((cl.c_citizen = :citizenship) or
                    (:citizenship = -1))
              into :c_id, :c_name_fio, :citizen_str, :c_plastcard_no
          do
          begin
            cont_call_is_active = null;

            cl_try_qty = 0;
            cl_call_qty = 0;
            cl_last_call_date = null;
            cl_last_try_date = null;
            con_try_qty = 0;
            con_call_qty = 0;

            if (call_check = 1) then -- если мы проверяем по возможности звонка
            begin
              next_call_check_ok = null;
              ignore_list_check_ok = null;
              can_call = 0;

              select rd.rd_id
              from refdata rd
              where rd.rd_code = 'ContactInfo.Cell_Main'
              into :cell_main_id;

              -- проверяю наступило ли время следующего звонка
              select can_call
              from hp_can_call_check(:k_id, 'Clients', :c_id, null)
              into :next_call_check_ok;

              -- проверяю по игнор-листу
              select can_call
              from okt_can_call(null, 'Clients', :c_id, null, null, :cell_main_id, null)
              into :ignore_list_check_ok;

              next_call_check_ok = coalesce(next_call_check_ok, 0);
              ignore_list_check_ok = coalesce(ignore_list_check_ok, 0);

              if ((next_call_check_ok = 1) and
                  (ignore_list_check_ok = 1)) then
                can_call = 1;
            end

            if (pant_id <> 0) then
            begin
              select pds.pds_try_qty, pds.pds_call_qty, pds.pds_last_call_date, pds.pds_last_try_date,
                     pds_cont_call_is_active
              from pantera_data_svod pds
              where pds.pds_from_pan_id = :pant_id and
                    pds.pds_from_c_id = :c_id and
                    coalesce(pds.pds_from_cc_id, 0) = 0 and
                    coalesce(pds.pds_from_poruch_id, 0) = 0 rows 1
              into :cl_try_qty, :cl_call_qty, :cl_last_call_date, :cl_last_try_date, :cont_call_is_active;

              cont_call_is_active_real = cont_call_is_active;

              if (cont_call_is_active is null) then
              begin
                cont_call_is_active_real = 0;
                cont_call_is_active = 1;
              end

              select sum(pds.pds_try_qty), sum(pds.pds_call_qty)
              from pantera_data_svod pds
              where pds.pds_from_pan_id = :pant_id and
                    pds.pds_from_c_id = :c_id and
                    ((coalesce(pds.pds_from_cc_id, 0) > 0) or
                    (coalesce(pds.pds_from_poruch_id, 0) > 0)) rows 1
              into :con_try_qty, :con_call_qty;

              cl_try_qty = coalesce(cl_try_qty, 0);
              cl_call_qty = coalesce(cl_call_qty, 0);
              con_try_qty = coalesce(con_try_qty, 0);
              con_call_qty = coalesce(con_call_qty, 0);

              result_str = '';
              cl_call_result = '';
              cl_memo = '';

              if (cl_try_qty > 0) then
              begin
                select rd.rd_name, left(pc.pcl_memo, 1000)
                from pantera_clients pc
                inner join refdata rd on rd.rd_id = pc.pcl_rd_res
                where pc.pcl_from_pan_id = :pant_id and
                      pc.pcl_from_c_id = :c_id
                order by pc.pcl_called_date desc
                rows 1
                into :cl_call_result, :cl_memo;

                cl_call_result = coalesce(cl_call_result, '');
                cl_memo = coalesce(cl_memo, '');

                result_str = 'Результат: ' || left(cl_call_result, 1000);

                if (cl_memo <> '') then
                  result_str = left(result_str || '. Примечание: ' || cl_memo, 1000);
              end
            end

            is_ok = 1;
            called_oper_str = coalesce(called_oper_str, '');
            if ((hide_blocked_groups = 1) and
                (called_oper = '') and
                (called_oper_str <> '')) then
              is_ok = 0;

            if (is_ok = 1) then
            begin
              suspend;
              rec_qty = rec_qty + 1;

              if ((rec_qty >= max_rec_qty) and
                  (max_rec_qty <> 0)) then
                exit;
            end
          end
        end
        else
        begin
          suspend;
        end
      end
    end
  end
end
