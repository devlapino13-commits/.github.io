CREATE PROCEDURE CM_KG_0016_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kol_day integer;
declare variable min_koef d_integer;
declare variable max_koef d_integer;
declare variable kol_day_min d_integer;
declare variable kol_day_max d_integer;
declare variable ma_refund d_integer;
begin
-- при утверждении графика:
--От даты выдачи до первого погашения не должно быть меньше 21 дней и не больше 45 дней (21 и 45 разрешаем).
  kol_day = null;
  select max(sh_date) - min(sh_date)
  from schedule
  where sh_from_k_id = :k_id
  and sh_id in (0,1)
  into kol_day;
  --min_koef = 0;
  --max_koef = 0;
  select /*(case maa.ma_refund when 1 then 28
                                when 2 then 59 -- 28 + 31
                                else 0 end) as min_koef,
         (case maa.ma_refund when 1 then 31
                                when 2 then 62 -- 31 + 31
                                else 0 end) as max_koef  */
         coalesce(maa.ma_refund,0)
  from miniapp maa
  inner join kontract kont on kont.k_from_ma_app_id = maa.ma_id
  where kont.k_id = :k_id
  and maa.ma_date_end = '31.12.9999'
  into :ma_refund; --:min_koef, max_koef;
  kol_day_min = 21; -- + min_koef;
  kol_day_max = 45; -- + max_koef;
  if (kol_day is null) then kol_day = 0;
  if (ma_refund = 0) then
  begin
      if  (not(kol_day between :kol_day_min and :kol_day_max) and
           k_id not in(109010020831300101, 109010020831500101, 109010020831700101, 109010020831900101)) then
      begin
        res_error_text = 'От даты выдачи до первого погашения не должно быть меньше '||kol_day_min||' дней и не больше '||kol_day_max||' дней ';
        res_error_is_critical = 1;
        suspend;
      end
  end
end
