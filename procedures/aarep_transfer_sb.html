CREATE PROCEDURE AAREP_TRANSFER_SB(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    OFFICER_NAME VARCHAR(100),
    OFFICE_FOLDER VARCHAR(60),
    SBA_DATE_BEGIN_LAST DATE,
    SBL_NAME_LAST VARCHAR(50),
    SBA_DATE_BEGIN DATE,
    SBL_NAME_CURR VARCHAR(50),
    TRANSFER_BALANCE CURR_TYPE,
    DELINQ_ZAD_OS CURR_TYPE,
    SH_CALC_PERC CURR_TYPE,
    SH_CALC_FINE CURR_TYPE,
    CITIZEN_STR VARCHAR(20),
    DELINQ_DAY_TRANSFER INTEGER)
AS
declare variable sba_from_k_id id_type;
declare variable zad_perc curr_type;
declare variable c_citizen integer;
declare variable kol integer;
begin

  for select sba_date_begin, sbl_name, sba_from_k_id
      from sb_akt, sb_list, kontstate
      where sba_is_delete = 0 and
            sba_date_begin between :rep_date_begin and :rep_date_end and
            sba_status = 2 and
            sba_from_sbl_id = sbl_id and
            sbl_type = 1 and
            sba_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            sba_date_begin between ks_date_begin and ks_date_end
      into sba_date_begin, sbl_name_curr, sba_from_k_id
  do
  begin
    k_kontract_no = null;
    officer_name = null;
    office_folder = null;
    select k_kontract_no, c_name_fio, reg_code || ' ' || reg_name
    from kontract, knaktlist, clients, region
    where k_id = :sba_from_k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal_new_kf_officer = c_id and
          kal_new_reg_id = reg_unicode
    into k_kontract_no, officer_name, office_folder;

    sbl_name_last = null;
    select sbl_name, sba_date_begin
    from sb_akt, sb_list
    where sba_is_delete = 0 and
          sba_status = 2 and
          sba_from_sbl_id = sbl_id and
          sba_from_k_id = :sba_from_k_id and
          :sba_date_begin - 1 between sba_date_begin and sba_date_end
    into sbl_name_last, sba_date_begin_last;

    select sum(zad_os), sum(zad_perc), sum(sh_calc_perc), sum(sh_calc_fine)
    from aarep_make_zad_list(:sba_from_k_id, :sba_date_begin, 1)
    where sh_repay_type = 1 and
          sh_date is not null
    into delinq_zad_os, zad_perc, sh_calc_perc, sh_calc_fine;
--if (zad_perc > sh_calc_perc) then sh_calc_perc = zad_perc;

    transfer_balance = (select oks_balance
                        from oksummary
                        where oks_from_k_id = :sba_from_k_id and
                              :sba_date_begin between oks_date_begin and oks_date_end and
                              oks_close_state = 0 and
                              oks_balance > 0);
    transfer_balance = coalesce(transfer_balance, 0);

--   определяет гражданство группы
    c_citizen = null;
    select c_citizen, count(*) kol
    from knclient, clients
    where kc_from_k_id = :sba_from_k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = c_id
    group by c_citizen
    order by kol desc, c_citizen
    rows 1
    into c_citizen, kol;

    if (c_citizen is null) then
      c_citizen = 0;
    citizen_str = null;
    select rd_name
    from refdata
    where rd_id = :c_citizen
    into citizen_str;
--   кол-во просроченных дней на дату передачи
    delinq_day_transfer = 0;
    select :sba_date_begin - oks_delinq_date
    from oksummary
    where oks_from_k_id = :sba_from_k_id and
          :sba_date_begin between oks_date_begin and oks_date_end
--     and oks_balance > 0
          and
          :sba_date_begin > oks_delinq_date and
          oks_close_state = 0
    into delinq_day_transfer;
    if (delinq_day_transfer is null) then
      delinq_day_transfer = 0;

    suspend;
  end

end
