CREATE PROCEDURE HP_ADDRCLIENT(
    C_ID DECIMAL(18,0))
RETURNS (
    ADC_FROM_C_ID TYPE OF COLUMN ADDRCLIENT.ADC_FROM_C_ID,
    ADC_ADDRESS_TYPE TYPE OF COLUMN ADDRCLIENT.ADC_ADDRESS_TYPE,
    ADC_FROM_ADL_ID TYPE OF COLUMN ADDRCLIENT.ADC_FROM_ADL_ID,
    ADC_PHONE TYPE OF COLUMN ADDRCLIENT.ADC_PHONE,
    ADC_MEMO VARCHAR(1000),
    ADC_DATE TYPE OF COLUMN ADDRCLIENT.ADC_DATE,
    ADC_IS_RENT TYPE OF COLUMN ADDRCLIENT.ADC_IS_RENT,
    ADC_ADDRESS_TYPE_STR VARCHAR(20),
    ADC_ADDRESS_STR VARCHAR(255),
    ADC_FROM_UN_ID DECIMAL(18,0),
    ADC_COMMENT VARCHAR(255),
    IS_HISTORY INTEGER)
AS
declare variable db_id decimal(18,0);
begin

  db_id = f_get_db_id();
  adc_address_type = 0;
  adc_from_c_id = :c_id;
  while (adc_address_type < 5) do
  begin
    adc_address_type = adc_address_type + 1;
    adc_from_adl_id = null;
    select adc_from_adl_id, adc_phone, adc_memo, adc_date, adc_is_rent, adc_from_un_id
    from addrclient
    where adc_from_c_id = :c_id and
          adc_address_type = :adc_address_type
    into :adc_from_adl_id, :adc_phone, :adc_memo, :adc_date, :adc_is_rent, :adc_from_un_id;
    if (adc_from_adl_id is null) then
    begin
      adc_from_adl_id = db_id * 1000000000;
      adc_phone = null;
      adc_memo = null;
      adc_date = current_date;
      adc_is_rent = 0;
      adc_from_un_id = null;
    end

    if (f_tmp_const('use_perstemp') = 1) then
    begin
      if (adc_address_type = 1) then
      begin
        select pte_adl_id_1
        from perstemp
        where pte_from_c_id = :c_id
        into adc_from_adl_id;
      end

      if (adc_address_type = 2) then
      begin
        select pte_adl_id_2
        from perstemp
        where pte_from_c_id = :c_id
        into adc_from_adl_id;
      end
    end

    select adl_address_str, ---, ADL_COMMENT
           iif(coalesce(adl.adl_comment, '') <> '', 'Ориентир:' || coalesce(adl_comment, '') || '; ', '') || coalesce(:adc_memo, '')
    from addrlist adl
    where adl_id = :adc_from_adl_id
    into :adc_address_str, adc_memo; --ADC_COMMENT;

    if (adc_address_str is null) then
      adc_address_str = '';
    --if (ADC_COMMENT is null) then ADC_COMMENT = '';
    adc_memo = coalesce(adc_memo, '');

    if (adc_address_type = 1) then
      adc_address_type_str = 'Юридически';
    if (adc_address_type = 2) then
      adc_address_type_str = 'Фактически';
    if (adc_address_type = 3) then
      adc_address_type_str = 'Место работы/бизнеса';
    if (db_id in (39801, 41701) and
        adc_address_type in (4, 5)) then
      continue;
    if (adc_address_type = 4) then
      adc_address_type_str = 'Юр.родина ';
    if (adc_address_type = 5) then
      adc_address_type_str = 'Факт.родина';
    if (exists(select *
               from addrclog
               where adcl_from_c_id = :c_id and
                     adcl_address_type = :adc_address_type)) then
      is_history = 1;
    else
      is_history = 0;
    suspend;
  end
end
