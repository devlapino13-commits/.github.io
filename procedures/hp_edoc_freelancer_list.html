CREATE PROCEDURE HP_EDOC_FREELANCER_LIST(
    SD_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDC_CODE D_STR_CODE,
    EDC_NAME D_STR_100)
AS
declare variable C_ID D_ID;
declare variable SD_CATEGORY D_STR_255;
declare variable EDOC_KIND D_STR_CODE;
begin
    -- RFCKG-3484
    select sd.sd_from_c_id, sd.sd_category
    from servicedog sd
    where sd.sd_id = :sd_id
    into c_id, sd_category;

    if (coalesce(:c_id, 0) > 0) then
    begin
        edoc_kind =
        case
            when :sd_category containing 'стажёр' then 'probationer_kontract'
            when :sd_category containing 'кочегар' then 'kochegar_kontract'
            else 'freelancer_kontract'
        end;

        if (f_edoc_status(:edoc_kind, :sd_id, current_date) <> 'signed') then
        begin
            select edc_code, edc_name
            from edoc_cfg
            where edc_code = :edoc_kind
            into :edc_code, :edc_name;

            suspend;
            exit;
        end

        if (edoc_kind = 'kochegar_kontract') then
        begin
            for select edc_code, edc_name
            from edoc_cfg
            where edc_code in ('add_for_freelancers', 'termination_freelancer')
            into :edc_code, :edc_name
            do
                suspend;

        end
    end
end
