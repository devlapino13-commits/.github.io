CREATE PROCEDURE HP_CONTDATA_DELETE(
    CDA_ID D_ID,
    CONT_DELETE_KIND D_STR_CODE,
    BASE_CDA_ID D_ID,
    UN_ID D_ID)
AS
begin
  if (un_id is not null) then
  begin
    update contdata
    set cda_from_un_id = :un_id
    where cda_id = :cda_id and
          coalesce(cda_from_un_id, -777) <> coalesce(:un_id, -777);
  end

  update or insert into contdelete (cdl_from_cda_id, cdl_cont_delete_kind, cdl_base_cda_id)
  values (:cda_id, f_enum('cont_delete_kind', :cont_delete_kind, 'id'), :base_cda_id)
  matching (cdl_from_cda_id);
end
