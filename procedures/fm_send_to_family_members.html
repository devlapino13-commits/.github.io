CREATE PROCEDURE FM_SEND_TO_FAMILY_MEMBERS(
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID,
    UN_ID D_ID,
    IS_PORUCH SMALLINT)
RETURNS (
    RES_TEXT D_BLOB_TEXT,
    SHOW_RES D_STR_10)
AS
declare variable db_id d_integer;
declare variable tmp_int d_integer;
declare variable cl_fio_prior d_str_100;
declare variable cl_fio d_str_100;
declare variable tmp_str_long d_str_2048;
begin
  res_text = '';
  show_res = 'window';

  db_id = f_get_db_id();

  tmp_int = 0;
if (f_objkind(obj_kind, 'code') = 'miniapp') then
  begin
    cl_fio_prior = '';
    for select res_client_fio,
               iif(res_is_critical = 1, iif(:db_id <> 94401, '<font color="red">критично</font>', '<font color="red">kritik</font>'), iif(res_is_critical = 2, iif(:db_id <> 94401, '<font color="red">очень критично</font>', '<font color="red">krtiki hədd</font>'), iif(:db_id <> 94401, 'сигнал', 'Siqnal'))) || ' -> ' || res_txt
        from hp_check_data_complete(:obj_kind, :obj_id)
        where res_is_critical > 0
        order by res_client_fio
        into :cl_fio, :tmp_str_long
    do
    begin
      tmp_int = tmp_int + 1;
      if (cl_fio <> cl_fio_prior) then
      begin
        res_text = BLOB_APPEND(res_text, iif(cl_fio_prior <> '', '<br><br>', '') || '<b>' || cl_fio || '</b>');
        cl_fio_prior = cl_fio;
      end
      res_text = BLOB_APPEND(res_text, '<br>' || tmp_str_long);
    end
  end

  if (tmp_int = 0) then
    res_text = '';
  else
    res_text = BLOB_APPEND(res_text, '</body>');

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  select send_kol
  from insert_family_members_miniapp(0, :obj_id, :un_id, :is_poruch)
  into tmp_int;

  if (tmp_int = 0) then
    res_text = iif(db_id <> 94401, 'Клиенты НЕ отправлены. Возможно они ранее уже отправлялись', 'Müştərilər göndərilmədi.Əvvəl göndərilib');
  else
    res_text = iif(db_id <> 94401, 'Данные успешно отправлены на проверку место работы. Количество отправленных клиентов: ', 'Məlumat iş yerinə yoxlanılması üçün uğurla göndərildi. Göndərilən müştəri sayı: ') || tmp_int;
  suspend;
end
