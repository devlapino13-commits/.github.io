CREATE PROCEDURE AAREP_ON_FINES(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(50),
    K_KF_NO VARCHAR(100),
    FINE_BALANCE_BEGIN DECIMAL(18,2),
    FINE_BALANCE_END DECIMAL(18,2),
    FINE_PAYED DECIMAL(18,2),
    FINE_WRITEOFF DECIMAL(18,2),
    FINE_NACH_PER DECIMAL(18,2))
AS
declare variable k_id id_type;
declare variable ok_type integer;
declare variable ok_fine_buff decimal(18,2);
begin
  fine_balance_begin = 0;
--Кредиты за период
  for select distinct ks_from_k_id
      from kontstate ks1
      where ks1.ks_ol_is_delete = 0 and
            ks1.ks_state = 1 and
            ks1.ks_date_begin < :rep_date_end and
            ks1.ks_date_end >= :rep_date_begin - 1
      into k_id
  do
  begin
--Баланс на начало дня
    with tabl1
    as (select sum(sh_calc_fine) sh_fine
        from aarep_make_zad_list_temp(:k_id, :rep_date_begin, 1)
        where sh_date is not null
        union
        select -sum(ok_fine) ok_fine
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_c_id = 0 and
              ok_type in (4, 5) and
              ok_fine <> 0 and
              ok_ol_date = :rep_date_begin --Чтобы получить точную сумму
              and
              ok_from_k_id = :k_id)
    select sum(sh_fine)
    from tabl1
    into fine_balance_begin;

    if (fine_balance_begin >= 0) then
    begin
      fine_balance_end = 0;
      fine_payed = 0;
      fine_writeoff = 0;
      ok_fine_buff = null;
      for select distinct ok_type, -sum(ok_fine) ok_fine
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_c_id = 0 and
                ok_type in (4, 5) and
                ok_fine <> 0 and
                ok_ol_date between :rep_date_begin and :rep_date_end and
                ok_from_k_id = :k_id
          group by ok_type
          having -sum(ok_fine) <> 0
          into ok_type, ok_fine_buff
      do
        if (ok_fine_buff is not null) then
        begin
          if (ok_type = 4) then
            fine_payed = ok_fine_buff;
          if (ok_type = 5) then
            fine_writeoff = ok_fine_buff;
        end
--Остаток на конец
      with tabl2
      as (select sum(sh_calc_fine) sh_fine2
          from aarep_make_zad_list_temp(:k_id, :rep_date_end, 1)
          where sh_date is not null
          union
          select -sum(ok_fine) ok_fine2
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_c_id = 0 and
                ok_type in (4, 5) and
                ok_fine <> 0 and
                ok_ol_date = :rep_date_end and
                ok_from_k_id = :k_id)
      select sum(sh_fine2)
      from tabl2
      into fine_balance_end;

      if (fine_balance_end is null) then
        fine_balance_end = 0;

--Вычисляем начисленные пении за период
      fine_nach_per = fine_balance_end + fine_payed + fine_writeoff - fine_balance_begin;

      if (fine_balance_end + fine_payed + fine_writeoff + fine_balance_begin <> 0) then
      begin
        select distinct k_kontract_no, kal_new_kf_name, reg_name
        from kontract, knaktlist kal, region
        where k_id = :k_id and
              k_from_kf_id = kal_from_kf_id and
              kal.is_delete = 0 and
              :rep_date_end between kal_date and kal_date_end and
              kal.kal_new_reg_id = reg_unicode
        into :k_kontract_no, :k_kf_no, :reg_name;
        suspend;
      end
    end
  end
end
