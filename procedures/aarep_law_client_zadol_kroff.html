CREATE PROCEDURE AAREP_LAW_CLIENT_ZADOL_KROFF(
    K_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    CZ_FROM_C_ID ID_TYPE,
    CZ_CL_OS CURR_TYPE,
    CZ_CL_PERC CURR_TYPE,
    CZ_CL_OTHER CURR_TYPE,
    CZ_CL_ALL_ZADOL CURR_TYPE)
AS
begin
  for select oko_from_c_id cz_from_c_id, sum(oko_os + oko_sh_os) cz_cl_os, sum(oko_perc + oko_sh_perc) cz_cl_perc,
             sum(oko_other + oko_sh_other) cz_cl_other
      from operkroff, operlist
      where oko_from_k_id = :k_id and
            oko_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            ol_date <= :rep_date --S.K. 15.01.2019г. ( <= вместо = )
      group by oko_from_c_id
      into cz_from_c_id, cz_cl_os, cz_cl_perc, cz_cl_other
  do
  begin
    cz_cl_all_zadol = cz_cl_os + cz_cl_perc + cz_cl_other;
    suspend;
  end
end
