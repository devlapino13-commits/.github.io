CREATE PROCEDURE HP_OPERCASH_SAVE(
    OPC_ID D_INTEGER,
    OPC_DATE D_DATE,
    OPC_OFFICE_ID D_INTEGER,
    OPC_FROM_CSP_ID D_INTEGER,
    OPC_OPER_KIND D_SMALLINT,
    OPC_SIGN D_SMALLINT,
    OPC_AMOUNT D_CURRENCY,
    OPC_AMOUNT$BOND D_STR_LONG,
    OPC_ADD_AMOUNT D_CURRENCY,
    OPC_ADD_AMOUNT$BOND D_STR_LONG,
    OPC_FROM_C_ID D_ID,
    OPC_MANAGER_ID D_ID,
    OPC_IS_IPF D_BOOL,
    OPC_CORR_AL_ID D_STR_20,
    OPC_FROM_K_ID D_ID,
    OPC_CORR_REG_ID D_INTEGER,
    OPC_CARD_NO D_STR_20,
    OPC_FROM_AG_ID D_INTEGER,
    OPC_MEMO D_STR_LONG,
    OPC_FROM_UN_ID D_ID,
    OPC_ADD_INFO D_STR_255,
    OPC_OTHER_CLIENT D_STR_255,
    OPC_OTHER_CLIENT_ID D_ID,
    CIL_FROM_BLS_ID D_ID)
AS
declare variable oper_code      d_str_code;
declare variable cil_id         d_integer;
declare variable add_oper_kind  d_integer;
declare variable add_amount     d_currency;
declare variable add_oper_sign  d_smallint;
declare variable add_memo       d_str_long;
declare variable cfl_id         d_integer;
declare variable al_id          d_str_20;
declare variable tmp_amount     d_currency;
declare variable pr_id          d_integer;
declare variable kf_id          d_id;
declare variable k_disb_date    d_date;
declare variable main_doc_no    d_integer;
declare variable child_doc_no   d_integer;
declare variable tmp_doc_no     d_integer;
declare variable tmp_region     d_str_100;
declare variable tmp_date       d_date;
declare variable tmp_message    d_str_long;
declare variable al_account     d_str_code;
declare variable al_id_go       d_str_20;
declare variable k_cons_kind    d_smallint;
declare variable insuranse_k_id d_id;
declare variable pr_code        d_str_100;
declare variable res_text       d_str_255;
begin
  if (opc_other_client_id is not null) then -- RFCKG-3927
  begin
      select c_name_fio
      from clients
      where c_id = :opc_other_client_id
      into :opc_other_client;

      if (f_asp_signed(:opc_other_client_id, current_date) <> 1) then
        exception e_message 'У вносителя нет активного ЭЦП';
  end

  if (opc_from_k_id is not null) then
  begin
    opc_is_ipf = 0;
    select iif(pr_code containing '.ipf.', 1, 0), k_from_kf_id, k_disb_date, k_from_pr_id, pr_code
    from kontract, product
    where k_id = :opc_from_k_id and
          k_from_pr_id = pr_id
    into opc_is_ipf, kf_id, k_disb_date, pr_id, pr_code;

    if (row_count = 0) then
      exception e_message 'Id займа не найдена';

    for select opc_doc_no, opc_date, opc_amount, reg_code || ' / ' || csp_cash_no
    from opercash, region, cashplace
    where opc_from_k_id = :opc_from_k_id and
          opc_is_delete = 0 and
          opc_from_ol_id is null and
          opc_office_id = reg_unicode and
          csp_id = opc_from_csp_id and
          opc_id <> coalesce(:opc_id, 0)
    rows 1
    into tmp_doc_no, tmp_date, tmp_amount, tmp_region
    do
    begin
      tmp_message = 'По займу есть операция в процессе: док. %1 от %2 на сумму %3 (касса: %4)';
      exception e_message f_format(tmp_message, tmp_doc_no, f_datetostr(tmp_date), round(tmp_amount, 0), tmp_region);
    end

    for select opc_doc_no, opc_date, opc_amount, reg_code || ' / ' || csp_cash_no
    from operlistappr op, opercash, region, cashplace
    where opc_from_k_id = :opc_from_k_id and
          opc_is_delete = 0 and
          op.olr_from_ol_id = opc_from_ol_id and
          op.olr_reg_state = 1 and
          op.olr_appr_status = 0 and
          opc_office_id = reg_unicode and
          csp_id = opc_from_csp_id and
          opc_id <> coalesce(:opc_id, 0)
    rows 1
    into tmp_doc_no, tmp_date, tmp_amount, tmp_region
    do
    begin
      tmp_message = 'По займу есть запрос на аннулирование: док. %1 от %2 на сумму %3 (касса: %4)';
      exception e_message f_format(tmp_message, tmp_doc_no, f_datetostr(tmp_date), round(tmp_amount, 0), tmp_region);
    end

    select kal_new_reg_id
    from knaktlist kn
    where kal_from_kf_id = :kf_id and
          kal_date <= :opc_date and
          kal_date_end >= :opc_date and
          kn.is_delete = 0
    into opc_corr_reg_id;
  end

  if (coalesce(opc_amount, 0) = 0) then
    exception e_message 'Сумма не указана';

  if (exists(select *
             from hp_is_day_closed(:opc_office_id, :opc_date, 1)
             where day_closed = 1)) then
    exception e_message 'Опер.день кассы закрыт: ' || f_datetostr(opc_date);

  select iif(:opc_is_ipf = 0, csp_cls_al_id, csp_ipf_al_id)
  from cashplace
  where csp_id = :opc_from_csp_id
  into al_id;

  if (al_id is null) then
  begin
    if (opc_is_ipf = 0) then
      exception e_message 'Отсутствует лицевой счет кассы "Классика"';
    else
      exception e_message 'Отсутствует лицевой счет кассы "ИПФ"';
  end

  oper_code = f_enum_code('cashoperkind', opc_oper_kind);

  if (oper_code = 'minus.loan.disb') then
  begin
    if (k_disb_date <> opc_date) then
      exception e_message 'Дата выдачи не совпадает, по договору займа: ' || f_datetostr(k_disb_date);

    tmp_message = '';
    select kf_client_fio || ': ' || kf_error_text
    from hp_check_loan_konditions(:opc_from_k_id, 0, 0, '.loan_disb.', :opc_from_un_id)
    where kf_error_is_critical = 1
    rows 1
    into tmp_message;

    if (coalesce(tmp_message, '') <> '') then
      exception e_message tmp_message;
  end
  
  if (oper_code = 'minus.loan.disb' and pr_code containing '.insur_base.') then     -- RFCKG-4831
  begin
    select lir.lir_kontract_insurance
    from loan_insurance_relation lir
    where lir.lir_kontract_base = :opc_from_k_id
    into insuranse_k_id;

    insuranse_k_id = coalesce(insuranse_k_id, 0);

    if (insuranse_k_id = 0) then
      exception e_message 'Cначала нужно оформить кредит на страхование';

    if (f_edoc_status('loan.agreement', :insuranse_k_id, current_date) <> 'signed') then
      exception e_message 'Cначала нужно оформить кредит на страхование';
  end

  if (oper_code in ('plus.loan.repay', 'plus.loan.cl_repay')) then
  begin
    select res_text
    from hp_ipf_check_insurance_pay(:opc_from_k_id)
    into res_text;

    if (res_text <> '') then
      exception e_message res_text; -- RFCKG-5390
  end

  if (oper_code = 'plus.loan.offpay') then
  begin
    tmp_amount = 0;
    select res_amount_saldo
    from aarep_operextra_saldo(9, :opc_from_k_id, f_get_rd_id('OperExtraKind.LawClientExpense'), 'LoanRepay', :opc_date, 0)
    into tmp_amount;

    if (tmp_amount > 0) then
      exception e_message 'Имеется долг по судебным взысканиям, сначала необходимо закрыть этот долг.';

    tmp_amount = 0;
    select sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc + oko_other + oko_sh_other)
    from operkroff
    where oko_from_k_id = :opc_from_k_id and
          oko_from_c_id = :opc_from_c_id and
          oko_ol_is_active = 1
    into tmp_amount;

    if (tmp_amount is null) then
      tmp_amount = 0;

    if (opc_amount > tmp_amount) then
      exception e_message 'Сумма погашения больше чем задолженность клиента: ' || round(tmp_amount, 0);

    -- Счет учета основного долга по ранее списанным займам
    select res_acc_writeoff
    from hp_make_kredit_acclist(:pr_id, :opc_corr_reg_id, 1)
    into :opc_corr_al_id;

    if (opc_corr_al_id = '') then
      exception e_message 'Не настроены счета погашений ранее списанных займов продукта';
  end

  if (oper_code = 'plus.loan.sud') then
  begin
    tmp_amount = 0;
    select res_amount_saldo
    from aarep_operextra_saldo(9, :opc_from_k_id, f_get_rd_id('OperExtraKind.LawClientExpense'), 'LoanRepay', :opc_date, 0)
    into tmp_amount;

    if (tmp_amount is null) then
      tmp_amount = 0;

    if (opc_amount > tmp_amount) then
      exception e_message 'Сумма погашения больше чем задолженность: ' || round(tmp_amount, 0);

    -- Счет учета судебной издержки
    opc_corr_al_id = f_cashacc(opc_corr_reg_id, opc_oper_kind, opc_is_ipf);

    if (opc_corr_al_id = '') then
      exception e_message 'Не настроены счета судебных взысканий';
  end

  if (oper_code = 'plus.loan.card_kommis') then
  begin
    -- Счет учета комиссии за выпуск карты
    opc_corr_reg_id = opc_office_id;
    opc_corr_al_id = f_cashacc(opc_corr_reg_id, opc_oper_kind, opc_is_ipf);

    if (opc_corr_al_id = '') then
      exception e_message 'Не настроены счета комиссии за выпуск карты';
  end

  if (oper_code = 'plus.loan.partial') then
  begin
    select res_amount
    from hp_opercash_loan_other_amount(:opc_from_k_id, 'CashTransactionIncom.PartialPrice')
    into :tmp_amount;

    if (tmp_amount = 0) then
      exception e_message 'Задолженность по частиной оплате отсутствует';
    else
    if (tmp_amount < opc_amount) then
      exception e_message 'Сумма больше чем задолженность по частиной оплате';

    select res_account
    from hp_make_kredit_acclist(:pr_id, :opc_office_id, 46)
    into :opc_corr_al_id;

    if (opc_corr_al_id = '') then
      exception e_message 'Не настроены счета кредитного продукта для частичной оплаты товара по ИФП';
  end

  if (oper_code = 'plus.loan.konsult') then
  begin
    select k_from_pr_id, k_cons_kind
    from kontract
    where k_id = :opc_from_k_id
    into pr_id, k_cons_kind;

    select res_account
    from hp_make_kredit_acclist(:pr_id, :opc_office_id, 47)
    into :opc_corr_al_id;

    if (opc_corr_al_id = '') then
      exception e_message 'Не настроены счета кредитного продукта для оплаты консультации';

    if (k_cons_kind is null) then
      k_cons_kind = 0;

    if (k_cons_kind = 0 and
        exists(select * from opercash
               where opc_from_k_id = :opc_from_k_id and
                     opc_is_delete = 0 and
                     opc_oper_kind = :opc_oper_kind)) then
      exception e_message 'Оплата за консультацию уже произведена';

    select res_amount
    from hp_opercash_loan_other_amount(:opc_from_k_id, 'CashTransactionIncom.Consultation')
    into :tmp_amount;

    if (tmp_amount = 0) then
      exception e_message 'Задолженность по оплате консультации отсутствует';
    else
    if (tmp_amount < opc_amount) then
      exception e_message 'Сумма больше чем задолженность по оплате консультации';
  end

  if (oper_code = 'plus.loan.insurance') then
  begin
    if ((pr_code not containing '.ipf.' or pr_code containing '.ignore_insurance.') and -- RFCKG-5390
         exists(select * from opercash
               where opc_from_k_id = :opc_from_k_id and
                     opc_is_delete = 0 and
                     opc_oper_kind = :opc_oper_kind)) then
      exception e_message 'Оплата за страхование уже произведена';

    select res_amount
    from hp_opercash_loan_other_amount(:opc_from_k_id, 'CashTransactionIncom.Insurance')
    into :tmp_amount;

    if (tmp_amount = 0) then
      exception e_message 'Задолженность по оплате за страхование отсутствует';
    else
    if (tmp_amount < opc_amount) then
      exception e_message 'Сумма оплаты превышает задолженность по страхованию';

    select ag_from_al_id
    from agreements
    inner join insurance_kontract on ik_from_ag_id = ag_id
    where ik_from_k_id = :opc_from_k_id
    into al_id_go; -- счет страхования в ГО

    if (pr_code containing '.ipf.' and
        pr_code not containing '.ignore_insurance.') then
      al_account = '11531/311'; -- RFCKG-5390
    else
      select al_account
      from acc_list
      where al_id = :al_id_go
      into al_account; -- короткий код счета (группа счетов/субсчет контрагента, например 21122/301)
  
    select al_id
    from acc_list
    where al_currency = '417' and
          al_from_reg_id = :opc_office_id and
          al_account = :al_account
    into opc_corr_al_id; -- счет страхования в офисе, где принимается платеж (RFCKG-4907)

    if (opc_corr_al_id = '') then
      exception e_message 'Не настроены счета кредитного продукта для оплаты страхования';
  end

  main_doc_no = null;
  child_doc_no = null;

  if (coalesce(opc_id, 0) <> 0) then
  begin
    select opc_from_cil_id, opc_doc_no
    from opercash
    where opc_id = :opc_id
    into cil_id, main_doc_no;

    select opc_doc_no
    from opercash
    where opc_parent_id = :opc_id and
          opc_sign <> 0
    rows 1
    into child_doc_no;

    update opercash
    set opc_is_delete = 1
    where opc_id = :opc_id;

    opc_id = null;
  end

  cil_id = null;
  if (oper_code in ('minus.office', 'minus.to_bank', 'plus.from_bank', 'minus.to_kassa')) then
  begin
    insert into cashinvlist (cil_from_csp_id, cil_date, cil_office_id, cil_oper_kind, cil_from_bls_id,
                             cil_inspector_id, cil_need_bonds, cil_inv_amount, cil_from_un_id)
    values (:opc_from_csp_id, :opc_date,
            iif(:oper_code = 'minus.office', :opc_corr_reg_id, :opc_office_id),
            :opc_oper_kind, :cil_from_bls_id,
            :opc_manager_id, 1, :opc_amount, :opc_from_un_id)
    returning (cil_id)
    into cil_id;

    execute procedure hp_cashbond_update(:cil_id, :opc_amount$bond);
  end

  insert into opercash (opc_date, opc_doc_no, opc_office_id, opc_from_csp_id, opc_oper_kind,
                        opc_sign, opc_amount, opc_from_c_id, opc_is_ipf, opc_corr_al_id,
                        opc_from_k_id, opc_corr_reg_id, opc_card_no, opc_from_ag_id,
                        opc_manager_id, opc_from_cil_id, opc_memo, opc_add_info, opc_other_client, opc_other_client_id, opc_from_un_id)
  values (:opc_date, :main_doc_no, :opc_office_id, :opc_from_csp_id, :opc_oper_kind,
          :opc_sign, :opc_amount, :opc_from_c_id, :opc_is_ipf,
          :opc_corr_al_id, :opc_from_k_id, :opc_corr_reg_id, :opc_card_no, :opc_from_ag_id,
          :opc_manager_id, :cil_id, :opc_memo, :opc_add_info, :opc_other_client, :opc_other_client_id, :opc_from_un_id)
  returning opc_id
  into opc_id;

  add_memo = '';
  if (oper_code = 'plus.from_bank') then
  begin
    if (opc_add_amount = opc_amount) then
    begin
      add_oper_kind = f_enum_id('cashoperkind', 'plus.from_bank'); -- если по чему больше по факту
      add_oper_sign = 1;
      add_memo = 'Акт обналичивания из банка';
    end
    if (opc_amount > opc_add_amount) then
    begin
      add_oper_kind = f_enum_id('cashoperkind', 'minus.saldo'); -- если по чему больше по факту
      add_oper_sign = -1;
      add_memo = 'Недостача при обналивании из банка';
    end
    else
    begin
      add_oper_kind = f_enum_id('cashoperkind', 'plus.saldo');
      add_oper_sign = 1;
      add_memo = 'Излишка при обналивании из банка';
    end

    add_amount = abs(opc_add_amount - opc_amount);

    insert into cashinvlist (cil_parent_id, cil_from_csp_id, cil_date, cil_oper_kind,
                             cil_inspector_id, cil_inv_amount, cil_from_un_id)
    values (:cil_id, :opc_from_csp_id, :opc_date, :add_oper_kind,
            :opc_manager_id, :opc_add_amount, :opc_from_un_id)
    returning (cil_id)
    into cfl_id;

    execute procedure hp_cashbond_update(:cfl_id, :opc_add_amount$bond);

    if (add_amount <> 0) then
    begin
      insert into opercash (opc_parent_id, opc_doc_no, opc_date, opc_from_csp_id, opc_oper_kind,
                            opc_sign, opc_amount, opc_from_c_id, opc_is_ipf, opc_corr_al_id,
                            opc_manager_id, opc_from_k_id, opc_corr_reg_id, opc_from_cil_id, opc_memo,
                            opc_other_client, opc_from_un_id)
      values (:opc_id, :child_doc_no, :opc_date, :opc_from_csp_id, :add_oper_kind,
              :add_oper_sign, :add_amount, :opc_from_c_id, :opc_is_ipf, :opc_add_info,
              :opc_manager_id, :opc_from_k_id, :opc_office_id, :cfl_id, :add_memo,
              :opc_other_client, :opc_from_un_id);
    end
  end

  add_memo = '';
  if (oper_code = 'minus.loan.disb' and opc_add_amount > 0) then
  begin
    add_oper_kind = f_enum_id('cashoperkind', 'plus.loan.kommis');
    add_oper_sign = 1;
    add_memo = 'Комиссия за кассовое обслуживание с НсП';
    add_amount = opc_add_amount;

    insert into opercash (opc_parent_id, opc_doc_no, opc_date, opc_from_csp_id, opc_oper_kind,
                          opc_sign, opc_amount, opc_from_c_id, opc_is_ipf, opc_corr_al_id,
                          opc_manager_id, opc_from_k_id, opc_corr_reg_id, opc_from_cil_id, opc_memo, opc_from_un_id)
    values (:opc_id, :child_doc_no, :opc_date, :opc_from_csp_id, :add_oper_kind,
            :add_oper_sign, :add_amount, :opc_from_c_id, :opc_is_ipf, null,
            :opc_manager_id, :opc_from_k_id, :opc_office_id, :cfl_id, :add_memo, :opc_from_un_id);
  end
end
