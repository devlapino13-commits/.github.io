CREATE PROCEDURE AAREP_ACCOUNT_OBORSALDO(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3),
    REPORT_ACCNAME_FILT VARCHAR(200) = null)
RETURNS (
    ACCOUNT VARCHAR(20),
    ACCOUNT_NAME VARCHAR(255),
    ACC_BALANCE_NO VARCHAR(20),
    ACC_BALANCE VARCHAR(255),
    ACC_GROUP VARCHAR(255),
    ACC_CLASS VARCHAR(255),
    CURR_CODE VARCHAR(3),
    BEG_DT DECIMAL(18,4),
    BEG_KT DECIMAL(18,4),
    OB_DT DECIMAL(18,4),
    OB_KT DECIMAL(18,4),
    END_DT DECIMAL(18,4),
    END_KT DECIMAL(18,4),
    LAST_OPER_DATE DATE,
    ACC_GROUP_NO VARCHAR(20),
    ACC_KIND INTEGER,
    ACC_KIND_NAME VARCHAR(255),
    ACCOUNT_TYPE INTEGER,
    ACCOUNT_IS_ERROR INTEGER,
    BEGIN_SALDO_X DECIMAL(18,4),
    END_SALDO_X DECIMAL(18,4),
    REG_NAME VARCHAR(100),
    ACC_BALANCE_NAME VARCHAR(255),
    ACC_DESCRIPTION D_STR_255)
AS
declare variable summa decimal(18,4);
declare variable acc_balance_no_prior varchar(20);
begin
  acc_balance_no_prior = '';
  acc_balance_name = '';
  for select account, account_name, acc_balance_no, acc_balance, acc_group, acc_class, curr_code, acc_group_no,
             acc_kind, acc_kind_name, account_type, sum(ob_dt), sum(ob_kt), sum(end_dt), sum(end_kt),
             max(last_oper_date)
      from aarep_account_oborsaldo_temp(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency)
      group by account, account_name, acc_balance_no, acc_balance, acc_group, acc_class, curr_code, acc_group_no, acc_kind, acc_kind_name, account_type
      order by curr_code, acc_balance_no, account
      into :account, :account_name, :acc_balance_no, :acc_balance, :acc_group, :acc_class, :curr_code, :acc_group_no,
           :acc_kind, :acc_kind_name, :account_type, :ob_dt, :ob_kt, :end_dt, :end_kt, :last_oper_date
  do
  begin
    if (acc_balance_no <> acc_balance_no_prior) then
    begin
      select bg_id || ' ' || bg_name
      from balgroup
      where bg_id = :acc_balance_no
      into :acc_balance_name;
    end
    reg_name = null;
    select
      reg_code || ' ' || reg_name,
      al_description
    from region, acc_list
    where al_id = :account and
          al_from_reg_id = reg_unicode
    into
      :reg_name,
      :acc_description;

    account_is_error = 0;
    beg_dt = 0;
    beg_kt = 0;
    summa = (-ob_dt + ob_kt + end_dt - end_kt);
    if (summa >= 0) then
      beg_dt = summa;
    else
      beg_kt = -1 * summa;
    if ((account_type = 0) and
        (end_dt - end_kt < 0)) then
      account_is_error = 1;
    if ((account_type = 1) and
        (end_dt - end_kt > 0)) then
      account_is_error = 1;
    begin_saldo_x = beg_kt - beg_dt;
    end_saldo_x = end_kt - end_dt;

    if (report_accname_filt is null or
      position(report_accname_filt, account_name) > 0)
    then suspend;
  end
end
