CREATE PROCEDURE FM_KONTMAT_REMOVE(
    KML_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    ERR_TEXT VARCHAR(255))
AS
begin
  err_text = '';
  if (coalesce(kml_id, 0) = 0) then
    err_text = 'Товар не найден';

  if (err_text = '') then
    if (not exists(select *
                   from kontmat_own kmo
                   where kmo.kmo_kml_id = :kml_id) and
        exists(select *
               from kontmat
               inner join operlist on ol_id = kml_bill_id and
                     is_delete = 0
               where kml_id = :kml_id)) then
      err_text = 'Операция запрещена. Сначала удалите бухгалтерскую проводку.';

  if (err_text = '') then
  begin
    delete from kontmat
    where kml_id = :kml_id;

    delete from kontmat_imei kmi
    where kmi.kmi_kml_id = :kml_id;
  end
  suspend;
end
