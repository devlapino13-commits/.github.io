CREATE PROCEDURE AAREP_DIGITIZATION_KMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ACL_DATE DATE,
    CLI_NAME STR_100,
    OTV_NAME STR_100,
    ACL_DATETIME TIMESTAMP,
    ACL_LAST_DATETIME TIMESTAMP,
    ACL_LIFETIME_IN_SECOND INTEGER,
    ACL_STATUS_STR D_STR_20,
    IN_MINUTE INTEGER,
    ACL_MEMO STR_255,
    PK_NAME STR_100,
    ACL_FROM_REG_ID D_INTEGER,
    REJECTION_REASON STR_255)
AS
declare variable ut_id            integer;
declare variable drl_from_c_id    id_type;
declare variable acl_respons_c_id id_type;
declare variable acl_status       integer;
declare variable drl_from_pk_id   integer;
declare variable acm_memo         varchar(4000);
declare variable json_key         varchar(255);
declare variable json_value       varchar(255);
declare variable acl_id           id_type;
declare variable acl_result_rd_id d_integer;
declare variable rd_id            d_integer;
declare variable rd1_id           d_integer;
declare variable acl_object_id    d_id;
begin
  ut_id = f_objkind('docrecoglist', 'id');
  rd_id = f_get_rd_id('ActionGroup.DocRecog_Client');
  rd1_id = f_get_rd_id('ActionGroup.DocRecog_Client_OiVO');
  for
  select acl_date,
         acl_respons_c_id, -- отцифровщик
         acl_datetime, -- Дата и время поступления
         acl_last_datetime, -- Дата и время решения
         acl_lifetime_in_second, -- Время рассмотрения (в секундах)
         acl_status, --Статус оцифровки
         datediff(minute, acl_datetime, acl_last_datetime), --Общее время (в минутах)
         acl_id,
         acl_from_reg_id,
         acl_result_rd_id,
         acl_object_id
  from actionlist
  where acl_last_datetime >= :rep_date_begin
  and acl_last_datetime < :rep_date_end + 1
  and acl_object_kind = :ut_id
--  and drl_status <> -1 -- не отображать черновики
  and acl_from_rd_id in (:rd_id, :rd1_id)
    into acl_date,
         acl_respons_c_id,
         acl_datetime,
         acl_last_datetime,
         acl_lifetime_in_second,
         acl_status,
         in_minute,
         acl_id,
         acl_from_reg_id,
         acl_result_rd_id,
         acl_object_id
  do
  begin
    drl_from_c_id = null;
    drl_from_pk_id = null;

    select drl_from_c_id, -- клиент
           drl_from_pk_id
    from docrecoglist
    where drl_from_c_id is not null
    and drl_id = :acl_object_id
    into :drl_from_c_id,
         :drl_from_pk_id;

    cli_name = null;
    select c_name_fio
    from clients
    where c_id = :drl_from_c_id
    into cli_name;

    otv_name = null;
    select c_name_fio from clients
    where c_id = :acl_respons_c_id
    into otv_name;

    acl_status_str = case acl_status
                       when -3 then 'Отмена заполнения'
                       when -2 then 'Отказан'
                       when -1 then 'На доработку'
                       when 0 then 'В ожидании'
                       when 1 then 'Принят на обработку'
                       when 2 then 'Подтвежден'
                       else 'Неверный статус'
                     end;

    pk_name = null;
    select pk_name from passkind
    where pk_id = :drl_from_pk_id
    into pk_name;

    acl_memo = '';
    acm_memo = '';
    json_key = null;
    json_value = null;
    select acm_memo
    from actionmemo
    where acm_from_acl_id = :acl_id
    into :acm_memo;
    if (coalesce(acm_memo, '') <> '') then
    begin
      for select json_key, json_value
          from fm_json_parse(:acm_memo, 0, 'verify')
          into :json_key, :json_value
      do
      begin
        if (json_key = 'comment') then
          acl_memo = json_value;
      end
    end
    rejection_reason = null;
    if (acl_status = -2) then
    begin
      select rd_name from refdata
      where rd_id = :acl_result_rd_id
      into rejection_reason;
    end
    suspend;
  end 

end
