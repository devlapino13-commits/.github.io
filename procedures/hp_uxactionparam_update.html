CREATE PROCEDURE HP_UXACTIONPARAM_UPDATE(
    UAP_UA_CODE D_STR_100,
    UAP_LINK_UFP_CODE D_STR_100,
    UAP_FROM_UFP_CODE D_STR_100,
    UAP_KIND D_SMALLINT,
    UAP_DEFAULT D_STR_255,
    UAP_USE_DEFAULT D_SMALLINT,
    UAP_LOCKED D_BOOL)
AS
begin
  if (uap_from_ufp_code = '') then
    uap_from_ufp_code = null;

  if (uap_use_default = -1) then
  begin
    uap_default = '';
    uap_from_ufp_code = null;
  end

  if (uap_from_ufp_code is null and trim(coalesce(uap_default, '')) = '') then
  begin
    -- Параметры не указаны, если есть запись, то удаляем
    delete from uxactionparam
    where uap_ua_code = :uap_ua_code and
          uap_kind = :uap_kind and
          uap_link_ufp_code = :uap_link_ufp_code;
  end else
  begin
    if (uap_use_default = 0) then
      if (uap_from_ufp_code is null and trim(coalesce(uap_use_default, '')) <> '') then
        uap_use_default = 1;

    update or insert into uxactionparam (uap_ua_code, uap_link_ufp_code, uap_from_ufp_code, uap_kind,
                                         uap_default, uap_use_default, uap_locked)
    values (:uap_ua_code, :uap_link_ufp_code, :uap_from_ufp_code, :uap_kind,
            :uap_default, :uap_use_default, :uap_locked)
    matching (uap_ua_code, uap_kind, uap_link_ufp_code);
  end
end
