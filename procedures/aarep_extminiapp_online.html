CREATE PROCEDURE AAREP_EXTMINIAPP_ONLINE(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    EMA_ID D_INTEGER,
    IS_KMP_KC D_STR_CODE,
    FIO STR_100,
    CELL_NO STR_100,
    IS_KRED D_STR_10,
    EMA_OFFICE STR_255,
    EMA_DATETIME D_DATETIME,
    EMA_BLOCK_DATETIME D_DATETIME,
    EMA_BLOCK_FIO STR_100,
    EMA_STATUS_STR STR_100,
    EMA_STATUS_DATETIME D_DATETIME,
    EMA_REFUSE D_STR_255,
    EMA_REFUSAL_FIO D_STR_100,
    MA_DATE_SAVE D_DATETIME,
    MA_STATUS D_STR_100,
    MA_REFUSE D_STR_255,
    MA_REFUSE_FIO D_STR_100,
    C_NAME_FIO STR_100,
    K_DISB_DATE D_DATE,
    K_KONTRACT_NO STR_20,
    C_CITIZEN_STR STR_100)
AS
declare variable ema_status d_integer;
declare variable tkt_id d_id;
declare variable ema_date d_date;
declare variable kf_id d_id;
declare variable ma_is_active d_integer;
declare variable k_id d_id;
declare variable c_id d_id;
declare variable ma_id d_id;
declare variable ma_refuse_reason d_str_255;
declare variable k_refuse_reason d_str_255;
declare variable k_state d_integer;
declare variable ema_refuse_code d_str_255;
declare variable k_kontract_date d_date;
begin
    for select ema.ema_id,  -- id заявки
               iif(f_source_str(tkt.tkt_source) = 'FsClient', 'КМП', 'КЦ'),  -- сточник КЦ/КМП
               cl.c_name_fio,  -- ФИО клиента
               reg_code || ' ' || reg_name,  -- Офис онлайн-заявки 
               ema.ema_datetime, -- Время создания онлайн-заявки 
               ema.ema_block_datetime, -- Время блокировки заявки
               s.c_name_fio,  -- Кто заблокировал заявку
               ema.ema_status,
               ema.ema_status_datetime, -- Время статуса онлайн-заявки
               ema.ema_from_tkt_id,
               ema.ema_date,
               r.rd_name, -- Причина отказа по онлайн-заявке
               cl.c_id,
               r.rd_code,
               c.rd_name
    from extminiapp ema
    inner join ticket tkt on tkt.tkt_id = ema.ema_from_tkt_id
    inner join region on reg_unicode = ema.ema_office_id
    inner join region_list(:rep_region_id, 1, -1, 1) on id_list = reg_unicode
    inner join clients cl on cl.c_id = ema.ema_from_c_id  -- клиент
    left join clients s on s.c_id = ema.ema_lo_id -- кред служащий который заблокир заявку
    inner join refdata r on r.rd_id = ema.ema_result_rd_id
    inner join refdata c on c.rd_id = cl.c_citizen
    where ema.ema_date between :rep_date_begin and :rep_date_end
    into ema_id,
         is_kmp_kc,
         fio,
         ema_office,
         ema_datetime,
         ema_block_datetime,
         ema_block_fio,
         ema_status,
         ema_status_datetime,
         tkt_id,
         ema_date,
         ema_refuse,
         c_id,
         ema_refuse_code,
         c_citizen_str
    do
    begin 
        ema_refusal_fio = ''; -- Кто отказал
        if (ema_status = -1) then
            ema_refusal_fio = ema_block_fio;

        -- Статус онлайн-заявки
        ema_status_str = (case ema_status
                            when -1 then 'Отказ на онлайн заявку'
                            when 0 then (case
                                            when ema_block_datetime is null then 'Не заблокировано'
                                            when ema_block_datetime is not null then 'Заблокировано'
                                         end)
                            when 1 then 'Онлайн заявка обработана'
                          end);


        -- Контактный номер
        cell_no = f_cont_info('clients', :c_id, 'cell_main', 'ok');

        -- Кредитовался клиент ранее ДА/НЕТ
        select iif(count(kc.kc_from_k_id) = 0, 'Нет', 'Да')
        from knclient kc
        inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
        where kc.kc_from_c_id = :c_id
        and kc.kc_amount_approved > 0
        and ks.ks_state = 1
        and ks.ks_ol_is_delete = 0
        and ks.ks_date_begin < :ema_date
        into is_kred;

        ma_id = null;
        ma_is_active = null;
        ma_date_save = null;
        ma_status = null;
        ma_refuse = null;
        ma_refuse_fio = null;
        k_id = null;
        k_kontract_no = null;
        k_disb_date = null;
        c_name_fio = null;
        kf_id = null;

        if ((ema_status = -1 and ema_refuse_code = 'website_miniapp.Refuse.MiniappCreateCancel') or (ema_status = 1)) then
        begin
            if (exists(select ma_id
                            from miniapp maa
                            where ma_from_tkt_id = :tkt_id)) then
            begin
                select ma_id
                from miniapp maa
                where ma_from_tkt_id = :tkt_id
                rows 1
                into :ma_id;
    
                select k_id
                from kontract
                where k_from_tkt_id = :tkt_id
                rows 1
                into k_id;
            end
            else
            begin
                select min(mac_from_ma_id),
                       ma_date_app
                from miniappclient
                inner join miniapp on ma_id = mac_from_ma_id
                where mac_from_c_id = :c_id
                and ma_date_app >= :ema_date
                and ma_date_end = '31.12.9999'
                and mac_amount > 0
                group by 2
                order by 2
                rows 1
                into ma_id,
                     k_kontract_date;
            end
    
            select k_id
            from kontract
            where k_from_ma_app_id = :ma_id
            rows 1
            into k_id;
    
            if (ma_id is not null and k_id is null) then
            begin
                select k_id
                from kontract
                where k_from_ma_app_id = :ma_id
                and  k_kontract_date >= :k_kontract_date
                rows 1
                into k_id;
            end

            select ma_is_active,
                   ma_date_save,
                   ma_from_kf_id
            from miniapp
            where ma_id = :ma_id
            and ma_date_end = '31.12.9999'
            rows 1
            into ma_is_active,
                 ma_date_save,
                 kf_id;


            select c.c_name_fio  -- ВИПС кредитной папки
            from knaktlist kal
            inner join clients c on c.c_id = kal.kal_new_kf_officer
            where kal.is_delete = 0
            and :rep_date_end between kal.kal_date and kal.kal_date_end
            and kal.kal_from_kf_id = :kf_id
            rows 1
            into :c_name_fio;
        end

        if (ma_is_active <= 0) then
        begin
            select krs.kr_text,
                   un.un_fio
            from kontreslist kl
            left join kontres krs on kr_id = krl_from_kr_id
            left join username un on un_id = kl.krl_from_un_id
            where kl.krl_ma_id = :ma_id
            rows 1
            into ma_refuse_reason,
                 ma_refuse_fio;
        end

        if (ma_is_active = 0) then
            ma_status = 'Отказ';
        if (ma_is_active = 1) then
            ma_status = 'В процессе';
        if (ma_is_active = -1) then
        begin
            ma_status = 'Отказ';
            ma_refuse = ma_refuse_reason;
        end

        if (k_id is not null) then
        begin
            select k_status,
                   iif(k_status < 0, '', k_kontract_no),
                   kr_text,
                   ks.ks_date_begin
            from kontract
            left join kontreslist on krl_from_k_id = k_id
            left join kontres on kr_id = krl_from_kr_id
            left join kontstate ks on ks_from_k_id = k_id
            left join clients on c_id = ks_kc_id
            where k_id = :k_id
            and current_date between ks_date_begin and ks_date_end
            and ks_ol_is_delete = 0
            rows 1
            into k_state,
                 k_kontract_no,
                 k_refuse_reason,
                 k_disb_date;
        end


        if (ma_id is not null and k_id is not null) then
        begin
            if (k_state = 1) then
                ma_status = 'Выдан';
            if (k_state = 0) then
                ma_status = 'В процессе';
            if (k_state = -1) then
            begin
                ma_status = 'Отказ';
                ma_refuse = k_refuse_reason;
            end 
            if (k_state = -2) then
            begin
                if (ma_is_active = 0) then
                ma_status = 'Отказ';
                if (ma_is_active = 1) then
                ma_status = 'В процессе';
                if (ma_is_active = -1) then
                begin
                    ma_status = 'Отказ';
                    ma_refuse = ma_refuse_reason;
                end
            end
        end
        suspend;
    end
end
