CREATE PROCEDURE AAREP_KN_OKS_DELINQ(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    LINE_CAPTION D_STR_100,
    STAGE1 D_CURRENCY,
    STAGE2 D_CURRENCY,
    STAGE3 D_CURRENCY)
AS
declare variable stage d_smallint;
declare variable delinq_days d_integer;
declare variable amount d_currency;
declare variable ro d_currency;
declare variable delinq_00_s1 d_currency;
declare variable delinq_00_s2 d_currency;
declare variable delinq_00_s3 d_currency;
declare variable delinq_01_30_s1 d_currency;
declare variable delinq_01_30_s2 d_currency;
declare variable delinq_01_30_s3 d_currency;
declare variable delinq_31_60_s1 d_currency;
declare variable delinq_31_60_s2 d_currency;
declare variable delinq_31_60_s3 d_currency;
declare variable delinq_61_90_s1 d_currency;
declare variable delinq_61_90_s2 d_currency;
declare variable delinq_61_90_s3 d_currency;
declare variable delinq_91_s1 d_currency;
declare variable delinq_91_s2 d_currency;
declare variable delinq_91_s3 d_currency;
declare variable total_s1 d_currency;
declare variable total_s2 d_currency;
declare variable total_s3 d_currency;
declare variable ro_s1 d_currency;
declare variable ro_s2 d_currency;
declare variable ro_s3 d_currency;
declare variable out_ro_s1 d_currency;
declare variable out_ro_s2 d_currency;
declare variable out_ro_s3 d_currency;
begin
  -- возвращает данные отчета "РО по корзинам по дням просрочки" (RFCKG-4597)

  delinq_00_s1 = 0;
  delinq_00_s2 = 0;
  delinq_00_s3 = 0;
  delinq_01_30_s1 = 0;
  delinq_01_30_s2 = 0;
  delinq_01_30_s3 = 0;
  delinq_31_60_s1 = 0;
  delinq_31_60_s2 = 0;
  delinq_31_60_s3 = 0;
  delinq_61_90_s1 = 0;
  delinq_61_90_s2 = 0;
  delinq_61_90_s3 = 0;
  delinq_91_s1 = 0;
  delinq_91_s2 = 0;
  delinq_91_s3 = 0;
  total_s1 = 0;
  total_s2 = 0;
  total_s3 = 0;
  ro_s1 = 0;
  ro_s2 = 0;
  ro_s3 = 0;
  out_ro_s1 = 0;
  out_ro_s2 = 0;
  out_ro_s3 = 0;

  for select oks_ecl_acc_kind,
             oks_delinq_days,
             oks_vbs_end,
             oks_ecl_vbs
      from aarep_kn_oksummary_rep(:rep_date_begin, :rep_date_end)
      into stage, -- корзина РО на конец периода (Стадия)
           delinq_days, -- дни просрочки
           amount, -- ВБС на конец (ОД+%)
           ro -- оценочный резерв под ожидаемые кредитные убытки (РО)
  do
  begin
    if (stage not between 1 and 3) then continue;

    delinq_days = coalesce(delinq_days, 0);
    amount = coalesce(amount, 0);
    ro = coalesce(ro, 0);

    if (stage = 1) then begin
      ro_s1 = ro_s1 + ro;
      if (delinq_days = 0)                    then delinq_00_s1     = delinq_00_s1    + amount;
      else if (delinq_days between 1 and 30)  then delinq_01_30_s1  = delinq_01_30_s1 + amount;
      else if (delinq_days between 31 and 60) then delinq_31_60_s1  = delinq_31_60_s1 + amount;
      else if (delinq_days between 61 and 90) then delinq_61_90_s1  = delinq_61_90_s1 + amount;
      else if (delinq_days > 90)              then delinq_91_s1     = delinq_91_s1    + amount;
    end
    else if (stage = 2) then begin
      ro_s2 = ro_s2 + ro;
      if (delinq_days = 0)                    then delinq_00_s2     = delinq_00_s2    + amount;
      else if (delinq_days between 1 and 30)  then delinq_01_30_s2  = delinq_01_30_s2 + amount;
      else if (delinq_days between 31 and 60) then delinq_31_60_s2  = delinq_31_60_s2 + amount;
      else if (delinq_days between 61 and 90) then delinq_61_90_s2  = delinq_61_90_s2 + amount;
      else if (delinq_days > 90)              then delinq_91_s2     = delinq_91_s2    + amount;
    end
    else if (stage = 3) then begin
      ro_s3 = ro_s3 + ro;
      if (delinq_days = 0)                    then delinq_00_s3     = delinq_00_s3    + amount;
      else if (delinq_days between 1 and 30)  then delinq_01_30_s3  = delinq_01_30_s3 + amount;
      else if (delinq_days between 31 and 60) then delinq_31_60_s3  = delinq_31_60_s3 + amount;
      else if (delinq_days between 61 and 90) then delinq_61_90_s3  = delinq_61_90_s3 + amount;
      else if (delinq_days > 90)              then delinq_91_s3     = delinq_91_s3    + amount;
    end
  end

  -- Всего займы выданные (сумма всех выданных займов)
  total_s1 = delinq_00_s1 + delinq_01_30_s1 + delinq_31_60_s1 + delinq_61_90_s1 + delinq_91_s1;
  total_s2 = delinq_00_s2 + delinq_01_30_s2 + delinq_31_60_s2 + delinq_61_90_s2 + delinq_91_s2;
  total_s3 = delinq_00_s3 + delinq_01_30_s3 + delinq_31_60_s3 + delinq_61_90_s3 + delinq_91_s3;

  -- Всего займы выданные, за вычетом оценочного резерва под ожидаемые кредитные убытки
  out_ro_s1 = total_s1 - ro_s1;
  out_ro_s2 = total_s2 - ro_s2;
  out_ro_s3 = total_s3 - ro_s3;

  --***--
  -- Далее отображение таблицы по строкам
  line_caption = 'Без просрочки';
  stage1 = delinq_00_s1;
  stage2 = delinq_00_s2;
  stage3 = delinq_00_s3;
  suspend;

  line_caption = 'Просрочка 1-30';
  stage1 = delinq_01_30_s1;
  stage2 = delinq_01_30_s2;
  stage3 = delinq_01_30_s3;
  suspend;

  line_caption = 'Просрочка 31-60';
  stage1 = delinq_31_60_s1;
  stage2 = delinq_31_60_s2;
  stage3 = delinq_31_60_s3;
  suspend;

  line_caption = 'Просрочка 61-90';
  stage1 = delinq_61_90_s1;
  stage2 = delinq_61_90_s2;
  stage3 = delinq_61_90_s3;
  suspend;

  line_caption = 'Просрочка 91+';
  stage1 = delinq_91_s1;
  stage2 = delinq_91_s2;
  stage3 = delinq_91_s3;
  suspend;

  line_caption = 'Всего займы выданные';
  stage1 = total_s1;
  stage2 = total_s2;
  stage3 = total_s3;
  suspend;

  line_caption = 'Оценочный резерв под ожидаемые кредитные убытки';
  stage1 = ro_s1;
  stage2 = ro_s2;
  stage3 = ro_s3;
  suspend;

  line_caption = 'Всего займы выданные, за вычетом оценочного резерва под ожидаемые кредитные убытки';
  stage1 = out_ro_s1;
  stage2 = out_ro_s2;
  stage3 = out_ro_s3;
  suspend;
end
