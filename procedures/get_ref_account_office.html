CREATE PROCEDURE GET_REF_ACCOUNT_OFFICE(
    SAB_FULL_NAME D_STR_255,
    SAB_INVERT_NAME D_STR_255,
    ACC_FROM_REG_ID D_ID,
    ACC_FROM_CU_ID D_STR_LANG,
    ACC_NUMBER D_ID,
    ACC_NON_RESIDENT SMALLINT = 0,
    ACC_CLIENT_TYPE SMALLINT = 0,
    ACC_DATE DATE = current_date)
RETURNS (
    ACC_ID D_ID)
AS
begin
  select acc_id
  from get_ref_account_any(:sab_full_name, :sab_invert_name, null, null, :acc_from_cu_id, :acc_from_reg_id, :acc_number, :acc_non_resident, :acc_client_type, :acc_date)
  into :acc_id;

  suspend;
end
