Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE FIND_REFDATA(
    FIND_TXT D_STR_255,
    FIND_TXT_LAT D_STR_255,
    FIND_TXT_CYR D_STR_255,
    MAIN_CODE D_STR_255,
    SHOW_ONLY_ACTIVE D_SMALLINT,
    SHOW_WITH_ZERO D_SMALLINT)
RETURNS (
    RD_ID D_INTEGER,
    RD_PARENT_ID D_INTEGER,
    RD_CODE D_STR_255,
    RD_MAIN_CODE D_STR_255,
    RD_NAME D_STR_255,
    RD_MEMO D_STR_255,
    RD_ORDERNO D_INTEGER,
    RD_IS_ACTIVE D_INTEGER,
    RD_IS_LOCKED D_INTEGER,
    RD_TAG D_INTEGER,
    RD_STR_ID D_STR_100,
    RD_CHILD_COUNT D_INTEGER)
AS
declare variable is_found d_smallint;
begin
  is_found = 0;
  if (find_txt is null) then
    find_txt = '';
  if (find_txt_lat is null) then
    find_txt_lat = '';
  if (find_txt_cyr is null) then
    find_txt_cyr = '';

  if (find_txt = '') then
    exit;

  if (find_txt_lat = '') then
  begin
    select result_txt
    from hp_convert_format(:find_txt, 'lat')
    into :find_txt_lat;
  end
  if (find_txt_cyr = '') then
  begin
    select result_txt
    from hp_convert_format(:find_txt, 'rus')
    into :find_txt_cyr;
  end

  for select rd_id, rd_parent_id, rd_code, rd_main_code, rd_name, rd_memo, rd_orderno, rd_is_active, rd_is_locked,
             rd_tag, rd_str_id
      from refdata
      where rd_main_code = :main_code and
            rd_is_active >= :show_only_active and
            rd_str_id starting :find_txt and
            rd_parent_id is not null
      into :rd_id, :rd_parent_id, :rd_code, :rd_main_code, :rd_name, :rd_memo, :rd_orderno, :rd_is_active,
           :rd_is_locked, :rd_tag, :rd_str_id
  do
  begin
    is_found = 1;
    rd_child_count = 0;
    select count(*)
    from refdata
    where rd_parent_id = :rd_id
    into :rd_child_count;
    if (rd_child_count is null) then
      rd_child_count = 0;
    suspend;
  end
-- Если результат найден по коду, прекращаем поиск
  if (is_found = 1) then
    exit;

  for select rd_id, rd_parent_id, rd_code, rd_main_code, rd_name, rd_memo, rd_orderno, rd_is_active, rd_is_locked,
             rd_tag, rd_str_id
      from refdata
      where (rd_main_code = :main_code or
            (rd_id = 0 and
            :show_with_zero = 1)) and
            rd_is_active >= :show_only_active and
            rd_parent_id is not null and
            (rd_name containing :find_txt or
            rd_name containing :find_txt_lat or
            rd_name containing :find_txt_cyr or
            rd_memo containing :find_txt or
            rd_memo containing :find_txt_lat or
            rd_memo containing :find_txt_cyr or
            rd_str_id containing :find_txt) rows 50 -- Если результат большой, показываем только первые 50
      into :rd_id, :rd_parent_id, :rd_code, :rd_main_code, :rd_name, :rd_memo, :rd_orderno, :rd_is_active,
           :rd_is_locked, :rd_tag, :rd_str_id
  do
  begin
    rd_child_count = 0;
    select count(*)
    from refdata
    where rd_parent_id = :rd_id
    into :rd_child_count;
    if (rd_child_count is null) then
      rd_child_count = 0;
    suspend;
  end
end
