CREATE PROCEDURE HP_CLIENTCOMREG_LIST(
    C_ID D_ID,
    SHOW_ALL D_BOOL NOT NULL,
    IS_ARCHIVED D_BOOL NOT NULL)
RETURNS (
    CCR_ID D_INTEGER,
    CCR_FROM_C_ID D_ID,
    CCR_LIMIT_38 D_BOOL,
    CCR_LIMIT_152 D_BOOL,
    CCR_LIMIT_230 D_BOOL,
    CCR_IS_ARCHIVED SMALLINT)
AS
begin
  c_id = coalesce(c_id, 0);
  if (show_all = 1) then
  begin
    for select ccr_id, ccr_from_c_id, ccr_limit_38, ccr_limit_152, ccr_limit_230,
               ccr_is_archived
        from clientcomreg
        where ccr_is_archived = :is_archived
        into ccr_id, ccr_from_c_id, ccr_limit_38, ccr_limit_152, ccr_limit_230,
             ccr_is_archived
    do
      suspend;
  end
  else
  begin
    if (c_id = 0) then
      exit;
    for select ccr_id, ccr_from_c_id, ccr_limit_38, ccr_limit_152, ccr_limit_230,
               ccr_is_archived
        from clientcomreg
        where ccr_is_archived = :is_archived
          and ccr_from_c_id = :c_id
        into ccr_id, ccr_from_c_id, ccr_limit_38, ccr_limit_152, ccr_limit_230,
             ccr_is_archived
    do
      suspend;
  end
end
