CREATE PROCEDURE RCIB_GET_LAST_ERRORS(
    REP_DATE DATE)
RETURNS (
    RCP_DATE_BEGIN DATE,
    RCP_DATE DATE,
    RCP_FILENAME VARCHAR(255),
    K_KONTRACT_NO VARCHAR(20),
    RCE_ERR_CODE INTEGER,
    RCE_ERR_TEXT VARCHAR(255))
AS
declare variable rcp_id decimal(18,0);
declare variable pos1 integer;
declare variable pos2 integer;
declare variable str1 varchar(255);
declare variable str2 varchar(255);
declare variable str3 varchar(255);
declare variable str4 varchar(255);
declare variable str5 varchar(255);
begin
  select rcp.rcp_date_begin, rcp.rcp_date, rcp.rcp_filename, rcp.rcp_id
  from rcib_pocket rcp
  where rcp.rcp_date <= :rep_date and
        rcp.rcp_sended = 1
  order by rcp.rcp_date desc, rcp.rcp_version desc
  rows 1

  into :rcp_date_begin, :rcp_date, :rcp_filename, :rcp_id;

  for select distinct kont.k_kontract_no, rke.rce_err_code, rke.rce_err_text
      from rcib_kont_errors rke
      inner join rcib_kontract rck on rck.rck_rec_no = rke.rce_from_rec_no and
            rck.rck_from_cp_id = rke.rce_pocket_id
      inner join kontract kont on kont.k_id = rck.rck_from_k_id and
            rke.rce_pocket_id = :rcp_id
      into :k_kontract_no, :rce_err_code, :rce_err_text
  do
  begin
    if (rce_err_code = 5020) then
    begin
      str1 = '';
      str2 = '';

      pos1 = position('&apos;' in rce_err_text) + 6;
      pos2 = position('&apos;:' in rce_err_text);

      str1 = substring(rce_err_text from pos1 for pos2 - pos1);
      if (str1 = 'index') then
        str1 = 'Почтовый индекс';
      if (str1 = 'birthplace') then
        str1 = 'Место рождения';
      if (str1 = 'docplace') then
        str1 = 'Место выдачи документа';

      pos1 = position('facet &apos;' in rce_err_text) + 12;
      pos2 = position('&apos;]' in rce_err_text);
      str2 = substring(rce_err_text from pos1 for pos2 - pos1);
      if (str2 = 'minLength') then
        str2 = 'Мин. длина';
      if (str2 = 'maxLength') then
        str2 = 'Макс. длина';
      str2 = '(' || str2 || ')';

      str3 = substring(rce_err_text from pos2 + 6);

      pos1 = position('&apos;' in str3) + 6;
      str4 = substring(str3 from pos1);

      pos2 = position('&apos;' in str4);

      str5 = substring(str4 from 1 for pos2 - 1);

      rce_err_text = str1 || ' ' || str2 || ' Значение = ' || str5;
    end
    suspend;
  end
end
