CREATE PROCEDURE HP_FPS_NAME_ACTION(
    ACTION_KIND D_STR_CODE,
    FCB_ID D_INTEGER,
    FCN_NAME D_STR_100,
    UN_ID D_ID)
RETURNS (
    FCB_STATUS D_INTEGER,
    FCB_STATUS_STR D_STR_100)
AS
declare variable c_id d_id;
declare variable min_distance d_integer;
begin
  -- Процедура для проверки и подтверждения имен получателей по СБП.
  -- Вызывается из Finstart в процессе выдачи по № телефона

  if (fcb_id is null) then
    exception e_default 'fcb_id is null';

  if (action_kind = 'verify') then
  begin
    -- Сохраняет статус проверки банка клиента по СБП.
    -- Если банк не возвращает PAM-фразу (fcn_name = ''), причины могут быть:
    -- 1. Неверно указан банк, где обслуживается клиент
    -- 2. Клиент не ключил получение денег по СБП (Сбер так любит делать)
    -- 3. В банке у клиента другой номер, не тот что подтвержден у нас
    -- Иначе в Client_name лежит ФИО из ответа банка

    fcb_status = null;
    fcb_status_str = 'ОШИБКА статуса';

    -- Логируем проверку
    insert into fps_client_banks_checks (fcc_from_fcb_id, fcc_from_un_id)
    values (:fcb_id, :un_id);

    if (fcn_name = '') then
    begin
      fcb_status = -1; -- PAM-фразы нет, см. выше
      fcb_status_str = 'Клиент НЕ НАЙДЕН в банке получателе';
    end
    else
    begin
      select fcb_from_c_id
      from fps_client_banks
      where fcb_id = :fcb_id
      into c_id;
  
      fcb_status = 1;
      fcb_status_str = 'ФИО ОТЛИЧАЕТСЯ, требуется подтверждение';

      if (exists(select * from moneta_client_fps_names(:c_id, 'list')
                 where client_names = :fcn_name))
      then
      begin
        fcb_status = 2;
        fcb_status_str = 'ФИО СОВПАДАЮТ';
      end
      else
      begin
        -- Если минимальное расстояние редактирования больше пяти, акцентируем
        select min(f_levenstain2(client_names, :fcn_name, 5))
        from moneta_client_fps_names(:c_id, 'list')
        into min_distance;
        min_distance = coalesce(min_distance, 255); -- Перестраховка
        if (min_distance > 5) then
        begin
          fcb_status = 1;
          fcb_status_str = 'ФИО ЗНАЧИТЕЛЬНО ОТЛИЧАЕТСЯ!!!';
        end
      end

      if (not exists(
      select * from fps_client_names
      where fcn_from_fcb_id = :fcb_id and
            fcn_name = :fcn_name)) then
      begin
        -- Имя добавляем даже если совпадает, на случай, если потом андеррайтеры изменят
        insert into fps_client_names (fcn_from_fcb_id, fcn_name, fcn_status)
        values (:fcb_id, :fcn_name, iif(:fcb_status = 2, 2, 0));
      end
      else
      begin
        -- Если имя есть, контрольно обновляем статус
        update fps_client_names
        set fcn_status = iif(:fcb_status = 2, 2, 0)
        where fcn_from_fcb_id = :fcb_id and
            fcn_name = :fcn_name and
            fcn_status <> iif(:fcb_status = 2, 2, 0);
      end
    end
  
    update fps_client_banks
    set fcb_status = :fcb_status
    where fcb_id = :fcb_id and
          fcb_status <> :fcb_status;

    suspend;
  end
  else if (action_kind = 'approve') then
  begin
    -- Ручное подтверждение имени клиента пользователем (fcb_status = 1)
    -- Если на этапе проверки (verify) оказалось что ФИО отличается,
    -- пользователь проверил, решил что имена похожие и можно перечислять
    update fps_client_names
    set fcn_status = 2,
        fcn_approve_un_id = :un_id
    where fcn_from_fcb_id = :fcb_id and
        fcn_name = :fcn_name and
        fcn_status = 0;
    -- fps_client_bank.fcb_status обновится через триггер
    fcb_status = 2;
    fcb_status_str = 'ФИО подтверждено ВРУЧНУЮ';
    suspend;
  end
  else
    exception e_default 'Invalid action_kind:' || coalesce(action_kind, 'null');

end
