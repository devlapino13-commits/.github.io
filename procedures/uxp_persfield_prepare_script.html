CREATE PROCEDURE UXP$PERSFIELD_PREPARE_SCRIPT(
    TABLE_NAME D_STR_CODE)
RETURNS (
    RES_SCRIPT D_BLOB_TEXT)
AS
declare variable res_is_error d_bool;
declare variable res_triggers d_str_8100;
begin
  select res_is_error, res_script
  from uxp$persfield_prepare_view(:table_name)
  into :res_is_error, :res_script;

  if (res_is_error = 1) then
  begin
    suspend;
    exit;
  end

  select res_is_error, res_script
  from uxp$persfield_prepare_triggers(:table_name)
  into :res_is_error, :res_triggers;

  res_script = res_script || res_triggers;

  suspend;
end
