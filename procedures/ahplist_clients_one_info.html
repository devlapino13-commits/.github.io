CREATE PROCEDURE AHPLIST_CLIENTS_ONE_INFO(
    CLIENT_ID DECIMAL(18,0))
RETURNS (
    C_ID DECIMAL(18,0),
    C_TYPE INTEGER,
    C_NAME_FIO VARCHAR(100),
    C_SEX SMALLINT,
    C_BIRTH_DATE DATE,
    C_NATION INTEGER,
    C_CITIZEN INTEGER,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_FROM_REG_ID INTEGER,
    C_EMAIL VARCHAR(100),
    C_FAMILY INTEGER,
    C_EDUCATION INTEGER,
    C_KOL_FAMILY INTEGER,
    C_KOL_CHILD INTEGER,
    C_MEMO VARCHAR(1024),
    C_REG_NO VARCHAR(20),
    C_OKPO VARCHAR(20),
    C_INN VARCHAR(20),
    C_ACCOUNT VARCHAR(20),
    C_ORG_FROM_B_ID VARCHAR(10),
    C_HIRED_DATE DATE,
    C_IS_RESIDENT SMALLINT,
    C_ARMY INTEGER,
    C_TABEL VARCHAR(10),
    C_PLASTCARD_NO VARCHAR(20),
    C_OTHERS VARCHAR(20),
    C_IS_PERSONAL SMALLINT,
    C_DOC_TYPE INTEGER,
    C_NATION_VIEW VARCHAR(100),
    C_CITIZEN_VIEW VARCHAR(100),
    C_FAMILY_VIEW VARCHAR(100),
    C_EDUCATION_VIEW VARCHAR(100),
    C_ARMY_VIEW VARCHAR(100),
    C_SEX_VIEW VARCHAR(10),
    C_BANK_NAME VARCHAR(255),
    C_IS_RESIDENT_TEXT VARCHAR(10),
    C_IS_PERSONAL_TEXT VARCHAR(10),
    C_PHOTO BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    C_COMMENT VARCHAR(1024),
    M_COMMENT_OFFICER VARCHAR(1024),
    M_COMMENT_MANAGER VARCHAR(1024),
    M_USER_FIO VARCHAR(100),
    M_DATETIME TIMESTAMP)
AS
declare variable ST_STAFF integer;
declare variable ST_CLIENT integer;
declare variable C_BLOB_PHOTO_ID integer;
declare variable C_IS_BLACK integer;
declare variable UN_FROM_C_ID ID_TYPE;
declare variable DB_ID D_ID;
begin
  db_id = f_get_db_id();

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select c_id, c_type, c_name_fio, c_sex, c_birth_date, c_nation, c_citizen, c_pass_no, c_pass_date, c_pass_who,
         c_pass_date_expire, c_from_reg_id,
         (select cont_value
          from hp_contdata_get_email_main(0, :client_id)), c_family, c_education, c_kol_family, c_kol_child, c_memo,
         c_reg_no, c_okpo, c_inn, c_account, c_org_from_b_id, c_hired_date, c_is_resident, c_army, c_tabel,
         (select cont_value
          from hp_contdata_get_cell_main(0, :client_id)), c_others, c_is_personal, c_doc_type, d1.rd_name, d2.rd_name,
         d3.rd_name, d4.rd_name, d5.rd_name, c_blob_photo_id
  from clients, refdata d1, refdata d2, refdata d3, refdata d4, refdata d5
  where (c_nation = d1.rd_id) and
        (c_citizen = d2.rd_id) and
        (c_family = d3.rd_id) and
        (c_education = d4.rd_id) and
        (c_army = d5.rd_id) and
        c_id = :client_id
  into :c_id, :c_type, :c_name_fio, :c_sex, :c_birth_date, :c_nation, :c_citizen, :c_pass_no, :c_pass_date, :c_pass_who,
       :c_pass_date_expire, :c_from_reg_id, :c_email, :c_family, :c_education, :c_kol_family, :c_kol_child, :c_memo,
       :c_reg_no, :c_okpo, :c_inn, :c_account, :c_org_from_b_id, :c_hired_date, :c_is_resident, :c_army, :c_tabel,
       :c_plastcard_no, :c_others, :c_is_personal, :c_doc_type, :c_nation_view, :c_citizen_view, :c_family_view,
       :c_education_view, :c_army_view, :c_blob_photo_id;

  select b_name
  from banks
  where b_id = :c_org_from_b_id
  into :c_bank_name;

  if (c_bank_name is null) then
    c_bank_name = '';
  if (c_blob_photo_id > 0) then
  begin
    select blx_blob
    from hp_get_blobex_blob(:c_blob_photo_id)
    into :c_photo;
  end
  else
  begin
    select c1.cb_blob
    from clientblob c1
    where c1.cb_from_c_id = :c_id and
          c1.cb_id = (select min(c2.cb_id)
                      from clientblob c2
                      where c2.cb_from_c_id = :c_id and
                            cb_type = 0)
    into :c_photo;
  end

  if (c_sex = 0) then
    c_sex_view = 'Женский';
  else
    c_sex_view = 'Мужской';

  if (c_is_resident = 1) then
    c_is_resident_text = 'Да';
  else
    c_is_resident_text = 'Нет';

  if (c_is_personal = 1) then
    c_is_personal_text = 'Сотрудник';
  else
    c_is_personal_text = 'Клиент';

  c_comment = '';
  m_comment_officer = '';
  m_comment_manager = '';

  m_user_fio = null;
  select c_comment, c_is_black, m_comment_officer, m_comment_manager, m_user_fio, m_datetime
  from hp_blacklist_search(:client_id, :c_name_fio, :c_birth_date, :c_pass_no, :c_inn, :c_plastcard_no, 1, 1, 0)
  where c_comment_id >= 1
  order by c_comment_id desc, c_is_black desc
  rows 1
  into :c_comment, c_is_black, m_comment_officer, m_comment_manager, m_user_fio, m_datetime;
  if (coalesce(c_comment, '') <> '') then
  begin
    if (c_is_black = 1) then
      c_comment = 'ОТКАЗАННЫЕ: ' || c_comment;
    if (c_is_black = 2) then
      c_comment = iif(db_id = 41701, 'Клиент с просрочкой: ', 'СТОП ЛИСТ: ') || c_comment;
  end
  if (m_user_fio is not null) then
  begin
    un_from_c_id = null;
    select un_from_c_id
    from username
    where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN')
    into :un_from_c_id;
    if (coalesce(un_from_c_id, 0) <> 0) then
    begin
      if (exists(select *
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :un_from_c_id and
                       current_date between ps_date_begin and ps_date_end and
                       ps_status = 1 and
                       ps_is_working = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id
                       --      and jk_system_code in ('JobSystem.LO_Standart','') -- по системным должностям не получится так как под JobSystem.LO_Standart Генеральный директор
                       and
                       jk_name in ('СВЗ', 'Кассир', 'Администратор'))) then
      begin
        m_user_fio = null;
        m_datetime = null;
      end
    end
  end
  suspend;
end
