CREATE PROCEDURE HP_DO_DISBURSEMENT_FPS(
    K_ID D_ID,
    PAYMENT_ID D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    ERROR_TEXT D_STR_8100)
AS
begin
  if (exists(select *
             from kontract
             inner join product on pr_id = k_from_pr_id
             where k_id = :k_id
             and pr_code containing '.zalog.')) then
  begin
    if (f_edoc_status('loan.collateral', :k_id, current_date) <> 'signed') then
    begin
      -- 16.07.2026, aschegolskiy. Добавил отправку ошибки на почту, чтобы
      -- убедиться что код НЕ срабатывает. Время хранения писем - 365 дней.
      -- Если за год не будет ни одного письма в fsrobot_msgbox,
      -- весь этот блок можно выпилить.
      in autonomous transaction do
        execute procedure hp_send_quick_message('email', 'ru.finstart@mbfplanet.com',
        'Попытка выдачи без подписанного договора залога.',
        f_format('Здравствуйте<br><br> Договор залога не подписан, k_id:%1<br><br>', k_id),
        'hp_do_disbursement_fps', 365, 1000000000000);

      error_text = 'Договор залога не подписан. НЕ повторяйте попытку выдачи, сообщите номер договора на ru.finstart@molbulak.com';
      suspend;
      exit;
    end
  end

  -- Контрольная проверка, в fps_locks обязательно должна быть запись
  if (not exists(select * from fps_locks where fpl_from_k_id = :k_id)) then
  begin
    error_text = 'Ошибка получения блокировки договора, обратитесь к разработчикам.';
    suspend;
    exit;
  end
  -- Сохраняем платежный идентификатор
  update fps_locks
  set fpl_payment_id = :payment_id
  where fpl_from_k_id = :k_id;

  select error_text
  from hp_do_disbursement_card(:k_id, :un_id)
  into :error_text;

  -- Разблокируем контракт
  execute procedure hp_fps_unlock_kontract(:k_id);

  suspend;
end
