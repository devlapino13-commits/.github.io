CREATE PROCEDURE AAREP_LOGINFO_DETAIL(
    LGI_ID D_ID)
RETURNS (
    LGD_ID D_ID,
    LGD_FIELD_NAME D_STR_CODE,
    LGD_FIELD_CAPTION D_STR_CODE,
    LGD_OLD_VALUE D_STR_LONG,
    LGD_NEW_VALUE D_STR_LONG,
    LGD_IS_CHANGED D_BOOL)
AS
declare variable lgf_id d_integer;
declare variable sql_curr_value d_str_long;
declare variable obj_kind d_integer;
declare variable table_name d_str_code;
declare variable key_field d_str_code;
declare variable obj_id d_id;
declare variable lgi_action d_smallint;
begin
  -- РћС‚РѕР±СЂР°Р¶Р°РµС‚ РґРµС‚Р°Р»Рё РїРѕР»РµР№ (СЃС‚Р°СЂРѕРµ Рё РЅРѕРІРѕРµ Р·РЅР°С‡РµРЅРёРµ) РІ Р»РѕРіР°С… РёР·РјРµРЅРµРЅРёР№

  select lgi_object_kind, lgi_object_id, lgi_action
  from loginfo
  where lgi_id = :lgi_id
  into obj_kind, obj_id, lgi_action;

  table_name = f_objkind(obj_kind, 'name');
  key_field = f_objkind(obj_kind, 'field');

  for select lgf_id, lgf_field_name, lgf_caption
  from logfield
  where lgf_table_name = :table_name and
        lgf_logging <> 0
  order by lgf_id
  into lgf_id, lgd_field_name, lgd_field_caption
  do
  begin
    lgd_new_value = null;
    lgd_old_value = null;
    lgd_is_changed = 0;
    lgd_id = null;

    select lgd_id, lgd_old_value
    from logdata
    where lgd_from_lgi_id = :lgi_id and
          lgd_from_lgf_id = :lgf_id
    into :lgd_id, :lgd_old_value;

    if (lgi_action <> -1) then
    begin
      if (lgd_id is not null) then
        lgd_is_changed = 1;
  
      select lgd_old_value
      from logdata
      where lgd_from_lgf_id = :lgf_id and
            lgd_from_lgi_id > :lgi_id
      order by lgd_id
      rows 1
      into lgd_new_value;
  
      if (row_count = 0) then
      begin
        sql_curr_value = f_format('select %1 from %2 where %3 = %4', lgd_field_name, table_name, key_field, obj_id);
  
        execute statement (sql_curr_value)
        into lgd_new_value;
      end
    end

    suspend;
  end
end
