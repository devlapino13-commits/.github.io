CREATE PROCEDURE AAREP_TRANSFER_BEELINE_ADDRESS(
    A_CLIENT_ID TYPE OF COLUMN CLIENTS.C_ID NOT NULL,
    A_IS_ADDRESS_RESIDENCE_PERMITS D_BOOL NOT NULL)
RETURNS (
    CLIENT_ID TYPE OF COLUMN CLIENTS.C_ID,
    ADDRESS_INDEX D_STR_10,
    ADDRESS_COUNTRY D_STR_10,
    ADDRESS_REGION D_STR_100,
    ADDRESS_DISTRICT D_STR_100,
    ADDRESS_LOCALITY_TYPE D_STR_100,
    ADDRESS_LOCALITY_NAME D_STR_100,
    ADDRESS_STREET_TYPE D_STR_100,
    ADDRESS_STREET_NAME D_STR_100,
    ADDRESS_HOUSE_TYPE D_STR_100,
    ADDRESS_HOUSE_NUMBER D_STR_100,
    ADDRESS_BUILDING_TYPE D_STR_100,
    ADDRESS_BUILDING_NUMBER D_STR_100,
    ADDRESS_ROOM_TYPE D_STR_100,
    ADDRESS_ROOM_NUMBER D_STR_100,
    ADDRESS_COMMENT D_STR_100,
    ADDRESS_FIAS_ID D_STR_100)
AS
begin
  for
    with gens as (
                select  '.'     as code, ''                        as name, 01 as id from rdb$database
      union all select  'стан.' as code, 'Станица'                 as name, 02 as id from rdb$database
      union all select  'хут.'  as code, 'Хутор'                   as name, 03 as id from rdb$database
      union all select  'аул'   as code, 'Аул'                     as name, 04 as id from rdb$database
      union all select  'пгт.'  as code, 'Поселок городского типа' as name, 05 as id from rdb$database
      union all select  'мкр.'  as code, 'Микрорайон'              as name, 06 as id from rdb$database
      union all select  'район' as code, 'Район'                   as name, 07 as id from rdb$database
      union all select  'п/р'   as code, 'Промышленный район'      as name, 08 as id from rdb$database
      union all select  'рп.'   as code, 'Рабочий поселок'         as name, 09 as id from rdb$database
      union all select  'г.'    as code, 'Город'                   as name, 10 as id from rdb$database
      union all select  'пос.'  as code, 'Поселок'                 as name, 11 as id from rdb$database
      union all select  'дер.'  as code, 'Деревня'                 as name, 12 as id from rdb$database
      union all select  'с.'    as code, 'Село'                    as name, 13 as id from rdb$database
    )
    , street as (
                select  ''     as code, ''           as name, 01 as id from rdb$database
      union all select  'вал'  as code, 'вал'        as name, 02 as id from rdb$database
      union all select  'ск-р' as code, 'Сквер'      as name, 03 as id from rdb$database
      union all select  'мкр.' as code, 'Микрорайон' as name, 04 as id from rdb$database
      union all select  'км'   as code, 'Километр'   as name, 05 as id from rdb$database
      union all select  'тр-т' as code, 'Тракт'      as name, 06 as id from rdb$database
      union all select  'лн.'  as code, 'Линия'      as name, 07 as id from rdb$database
      union all select  'алл.' as code, 'Аллея'      as name, 08 as id from rdb$database
      union all select  'ул.'  as code, 'Улица'      as name, 09 as id from rdb$database
      union all select  'пр-т' as code, 'Проспект'   as name, 10 as id from rdb$database
      union all select  'пер.' as code, 'Переулок'   as name, 11 as id from rdb$database
      union all select  'ш.'   as code, 'Шоссе'      as name, 12 as id from rdb$database
      union all select  'туп.' as code, 'Тупик'      as name, 13 as id from rdb$database
      union all select  'пр-д' as code, 'Проезд'     as name, 14 as id from rdb$database
      union all select  'пл.'  as code, 'Площадь'    as name, 15 as id from rdb$database
      union all select  'б-р'  as code, 'Бульвар'    as name, 16 as id from rdb$database
      union all select  'наб.' as code, 'Набережная' as name, 17 as id from rdb$database
      union all select  'кв-л' as code, 'Квартал'    as name, 18 as id from rdb$database
    )
    , building_type as(
                select  ''      as code, ''          as name, 01 as id from rdb$database
      union all select  'влад.' as code, 'Владение'  as name, 02 as id from rdb$database
      union all select  'зд.'   as code, 'Здание'    as name, 03 as id from rdb$database
      union all select  'пом.'  as code, 'Помещение' as name, 04 as id from rdb$database
      union all select  'лит.'  as code, 'Литера'    as name, 05 as id from rdb$database
      union all select  'Стр.'  as code, 'Строение'  as name, 06 as id from rdb$database
      union all select  'Корп.' as code, 'Корпус'    as name, 07 as id from rdb$database
    )
    , room_type as(
                select  ''     as code, ''         as name, 01 as id from rdb$database
      union all select  'Кв.'  as code, 'Квартира' as name, 02 as id from rdb$database
      union all select  'Ком.' as code, 'Комната'  as name, 03 as id from rdb$database

    )
    , house_type as(
                select  'вл.'  as code, 'вл.'      as name, 01 as id from rdb$database
      union all select  'д.'   as code, 'д.'       as name, 02 as id from rdb$database
      union all select  'двл.' as code, 'двл.'     as name, 03 as id from rdb$database
      union all select  'уч.'  as code, 'уч.'      as name, 04 as id from rdb$database
      union all select  'зд.'  as code, 'зд.'      as name, 05 as id from rdb$database
      union all select  'стр.' as code, 'строение' as name, 06 as id from rdb$database
    )
    select
        :a_client_id
        , addr.adl_post_index
        , (select first 1 bc_code from beeline_country where (bc_is_unknown = 1) or (bs_names_alternative containing ',' || addr.country || ',')  order by bc_is_unknown asc)
        , addr.region_name
        , addr.district

        -- Населенный пункт
        , (select first 1 code from gens where id = iif(addr.city is null, 01, 10))
        , coalesce(addr.city, addr.locality)

        -- Улица
        , coalesce((select first 1 code from street where lower(name)=lower(addr.street_plt)), '!!!!!!' || addr.street_plt)
        , addr.street

        -- Дом
        , iif(addr.adl_depart_no in ('0', ''), '!!!!!!', (select first 1 code from house_type where lower(name)=lower('д.')))
        , addr.adl_depart_no

        -- Строение
        , (select first 1 code from building_type where id = iif(addr.adl_building_no in ('0', ''), iif(addr.adl_section_no in ('0', ''), 01, 06), 07))
        , iif(addr.adl_building_no in ('0', ''), iif(addr.adl_section_no in ('0', ''), '', addr.adl_section_no), addr.adl_building_no)

        -- Помещение
        , (select first 1 code from room_type where id = iif(coalesce(addr.adl_flat_no, 0) = '0', 01, 02))
        , iif(addr.adl_flat_no in ('0', ''), '', addr.adl_flat_no)

        , null as address_reg_comment
        , null as address_reg_fias_id
      from
        hp_get_client_address(:a_client_id, :a_is_address_residence_permits + 1) as addr
      where
        (1 = 1)
      into
          :client_id
        , :address_index
        , :address_country
        , :address_region
        , :address_district
        , :address_locality_type
        , :address_locality_name
        , :address_street_type
        , :address_street_name
        , :address_house_type
        , :address_house_number
        , :address_building_type
        , :address_building_number
        , :address_room_type
        , :address_room_number
        , :address_comment
        , :address_fias_id
    do
      suspend;
end
