CREATE PROCEDURE CM_KG_0030_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable c_id id_type;
declare variable kc_amount_approved curr_type;
declare variable kc_circle integer;
declare variable ma_app_id integer;
declare variable kbm_id id_type;
declare variable kbm_id_old id_type;
declare variable max_sum curr_type;
declare variable min_sum curr_type;
declare variable max_delinq_day_cli integer;
declare variable max_delinq_day_cli_old integer;
declare variable kc_amount_approved_old curr_type;
declare variable k_id_old id_type;
declare variable kf_id id_type;
declare variable reg_id integer;
declare variable k_kolich_repay integer;
declare variable k_kolich_client_approved integer;
declare variable k_percent_per_day perc_type;
declare variable pr_product_kind integer;
declare variable k_circle integer;
declare variable k_kommis_perc perc_type;
declare variable k_amount_approved curr_type;
declare variable pr_name varchar(50);
declare variable pr_code varchar(100);
declare variable pr_code_old varchar(100);
declare variable percent_snijen perc_type;
declare variable percent_snijen_plan perc_type;
declare variable kol_kib_cli integer;
declare variable kh_clients_fio varchar(1024);
declare variable c_if varchar(100);
declare variable c_if_old varchar(100);
declare variable k_percent_per_day_old perc_type;
declare variable k_kommis_perc_old perc_type;
declare variable k_id_temp id_type;
declare variable close_date_old integer;
declare variable ks_date_begin_old date;
declare variable ks_date_end_old date;
declare variable oks_delinq_day integer;
declare variable new_cl integer;
declare variable poruch_qty integer;
declare variable k_percent_per_day_cl_old perc_type;
declare variable k_percent_per_day_cl_old1 perc_type;
declare variable kommis_snij decimal(15,2);
declare variable akcia varchar(20);
declare variable k_kredline_end_old date;
declare variable k_fact_closed_date_old date;
declare variable kc_circle_akc integer;
declare variable max_delinq_day_cli_akc integer;
declare variable k_id_old_akc id_type;
declare variable k_kredline_end_akc date;
declare variable k_fact_closed_date_akc date;
declare variable i integer;
begin
 
  for select k_disb_date, k_from_kf_id, k_last_reg_id,
             k_kolich_repay, k_kolich_client_approved, K_Percent_Per_Day,
             pr_product_kind, k_circle, k_kommis_perc,
             k_amount_approved, pr_name, pr_code, k_from_ma_app_id
    from kontract, product
    where k_id=:k_id
    and k_from_pr_id = pr_id
    and pr_code containing '.IND.'
    and pr_code containing '.BIRGE.'
    and pr_code not containing '.EC.'
    and pr_code not containing '.PAR.'
  into :kf_disb_date, :kf_id, :reg_id,
       :k_kolich_repay, :k_kolich_client_approved,
       K_Percent_Per_Day, :pr_product_kind, k_circle,
       k_kommis_perc, k_amount_approved, pr_name, pr_code, ma_app_id
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;
        
    select datediff(year, c_birth_date, :kf_disb_date)from zalog  zal, clients
    where zal.z_from_k_id = :k_id
    and zal.z_type = 0
    and zal.z_subtype = 1
    and zal.z_from_c_id = c_id
    rows 1
    into :poruch_qty;
    
    poruch_qty = coalesce(poruch_qty, 0);
    
    if (poruch_qty < 1) then
      begin
        res_error_text='Не указан поручитель!';
        res_error_is_critical=2;
        suspend;
      end
    
    if (poruch_qty not between 18 and 70) then
      begin
        res_error_text='Возраст поручителя должен быть от 18 до 70 лет!';
        res_error_is_critical=2;
        suspend;
      end
                             
    percent_snijen = round(K_Percent_Per_Day * 365, 1);
    percent_snijen_plan = 0;                    
    min_sum = 15000;
    max_sum = 55000;
        
    if (k_kolich_client_approved <> 1) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Количество клиентов в кредите должен быть 1';
      suspend;
    end
    if (k_amount_approved > max_sum) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Общая сумма для группы не может быть больше 55 000 сомов';
      suspend;
    end
    --2,2 срок
    if (k_kolich_repay not between 3 and 18) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Срок кредита должен быть от 3 до 18 мес.';
      suspend;
    end 
    if(iif(pr_code containing '.A.' and kf_disb_date between '01.08.16' and '31.08.16', 1, 0)=0) then
    begin
        --2,3
        if (percent_snijen > 43.5) then
        begin
          res_error_is_critical = 2;
          res_error_text = 'Процентная ставка не может быть больше 43,5%.';
          suspend;
        end  
    
        if (percent_snijen < 37) then
        begin
          res_error_is_critical = 2;
          res_error_text = 'Процентная ставка не может быть ниже 37%.';
          suspend;
        end
    
        --2,4 первый раз 43,5
        if (k_circle = 1 and percent_snijen <> 43.5) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Для нового кредита годовая процентная ставка должен быть 43,5%.';
          suspend;
        end 
    
        --2,5 первый доп% 3,9%
        if (k_circle = 1 and k_kommis_perc <> 3.9) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Для нового кредита дополнительная процентная ставка должна быть 3,9%.';
          suspend;
        end
    end
    --Проверка по клиенту
    if (k_kolich_client_approved = 1) then
    begin
      for select kncl.kc_from_c_id,
             kncl.kc_amount_approved,
             kncl.kc_circle,
             cl.c_name_fio
        from knclient kncl, clients cl
        where kncl.kc_amount_approved > 0
        and cl.c_id = kncl.kc_from_c_id
        and kncl.kc_from_k_id = :k_id
--        rows 1
      into :c_id,
           :kc_amount_approved,
           :kc_circle,
           :res_info_text
      do
      begin
              
       i=coalesce(i, 0)+1;
       if(i=2) then
          begin
            res_error_text='Количество участников более одного!';
            res_error_is_critical=2;
            suspend;
            exit;
          end
      end 
      if (kc_amount_approved <> :k_amount_approved) then
      begin  
        res_error_is_critical = 2;
        res_error_text = 'Внимание сумма клиента не совпадает в суммой кредита. Обратитесь в тех.поддержку!';
        suspend;
      end     
    
      if (kc_amount_approved not between min_sum and max_sum) then
      begin
        res_error_is_critical = 2;
        res_error_text = 'Сумма для клиента может быть от 15 000 до 55 000';
        suspend;
      end
      if (kc_circle > 1 and iif(pr_code containing '.A.' and kf_disb_date between '01.08.16' and '31.08.16', 1, 0)=0) then
      begin
      --2,4 4 Показатель максимальной просрочки по всем продуктам является определяющим преимущественным фактором --смотреть паспорт
        select kncl.kc_amount_approved,
               kncl.kc_from_k_id,
               ks_date_begin,
               round(K_Percent_Per_Day * 365, 1) ,
               kont.k_kredline_end,
               kont.k_fact_closed_date
        from kontstate ks
        inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
        inner join kontract kont on ks.ks_from_k_id = k_id
        where ks.ks_state = 1
        and ks.ks_ol_is_delete = 0
        and ks_date_begin < :kf_disb_date
        and kncl.kc_amount_approved > 0
        and kncl.kc_from_c_id = :c_id
        order by ks.ks_date_begin desc
        rows 1
        into :kc_amount_approved_old, :k_id_old, ks_date_begin_old, K_Percent_Per_Day_cl_old, k_kredline_end_old, :k_fact_closed_date_old;
        -- определяю макимальное число дней прссрочки на предыдущем цикле   
        max_delinq_day_cli = null;
        select max(ok.ok_ol_date - ok_sh_date)
        from operkred ok
        where ok.ok_ol_is_active = 1
        and ok.ok_ol_date <= :kf_disb_date
        and ok.ok_ol_date > ok_sh_date
        and ok_type in(4,5,9)
        and ok_from_k_id = :k_id_old
        and ok_os + ok_perc < 0
        and ok_from_c_id=0
        into max_delinq_day_cli;
        
        max_delinq_day_cli = coalesce(max_delinq_day_cli,0);
        if (k_circle > 1 and
          percent_snijen between 37 and 43.5) then
        begin
          --Проверяем пред. цикл контракта
          select round(K_Percent_Per_Day * 365,4),
                 k_kommis_perc,
                 k_id,
                 pr_code,
                 ks_date_end - ks_date_begin as close_date_old,
                 ks_date_begin,
                 ks_date_end
          from kontract, kontstate, product
          where k_from_kf_id = :kf_id
            and k_id = ks_from_k_id
            and ks_state = 1
            and ks_ol_is_delete = 0
            and ks_date_begin < :kf_disb_date  
            and k_from_pr_id = pr_id
            and pr_code containing '.IND.'
            and pr_code containing '.BIRGE.'
            and pr_code not containing '.EC.'
            and pr_code not containing '.PAR.'
            order by ks_date_begin desc
            rows 1
          into K_Percent_Per_Day_old, k_kommis_perc_old, k_id_temp, pr_code_old, close_date_old, ks_date_begin_old, ks_date_end_old;
          -- если предыдущий заем выдавался во время действия акции, тогда ...
          if (ks_date_begin_old between '18.03.2016' and '31.03.2016') then
          begin
            if (k_kommis_perc_old < 3.5) then
            begin
                k_kommis_perc_old = k_kommis_perc_old * 2;
            end
          end
          if (ks_date_end_old > kf_disb_date) then
            begin
              res_error_is_critical = 2;
              res_error_text = 'Внимание имеется активный кредит, проверка снижении %% и доп%% невозможна!';
              suspend;
            end else
          begin
            select max(ok_ol_date - ok_sh_date)
            from operkred
            where ok_ol_is_active = 1
            and ok_from_k_id = :k_id_temp
            and ok_type in (4,5,9)
            and ok_os + ok_perc < 0
            into oks_delinq_day;
          if (max_delinq_day_cli > oks_delinq_day) then oks_delinq_day = max_delinq_day_cli;
          if (pr_code_old containing '.A.') then
            begin
              K_Percent_Per_Day_old = null;
              k_kommis_perc_old = null;     k_id_temp = null;
              pr_code_old = null;           close_date_old = null;
              ks_date_begin_old = null;     ks_date_end_old = null;
    
              --Проверяем пред. цикл контракта до акции
              select round(K_Percent_Per_Day * 365,4),
                     k_kommis_perc,
                     k_id,
                     pr_code,
                     ks_date_end - ks_date_begin as close_date_old,
                     ks_date_begin,
                     ks_date_end
              from kontract, kontstate, product
              where k_from_kf_id = :kf_id
                and k_id = ks_from_k_id
                and ks_state = 1
                and ks_ol_is_delete = 0
                and ks_date_begin < :kf_disb_date  
                and k_from_pr_id = pr_id
                and pr_code containing '.IND.'
                and pr_code containing '.BIRGE.'
                and pr_code not containing '.EC.'
                and pr_code not containing '.PAR.'
                and pr_code not containing '.A.'
                order by ks_date_begin desc
                rows 1
              into K_Percent_Per_Day_old, k_kommis_perc_old, k_id_temp, pr_code_old, close_date_old, ks_date_begin_old, ks_date_end_old;
    
              select max(ok_ol_date - ok_sh_date)
              from operkred
              where ok_ol_is_active = 1
              and ok_from_k_id = :k_id_temp
              and ok_type in (4,5,9)
              and ok_os + ok_perc < 0
              into oks_delinq_day;
    
              if (max_delinq_day_cli > oks_delinq_day) then oks_delinq_day = max_delinq_day_cli;
            end
          if (percent_snijen_plan = 0) then
            percent_snijen_plan = case when oks_delinq_day > 30 then 200
                                       when oks_delinq_day between 16 and 30 then iif(:kf_disb_date < '27.06.16', 3, 2)--3.0
                                       when oks_delinq_day between 8 and 15 then iif(:kf_disb_date < '27.06.16', 2, 1)--2.0
                                       when oks_delinq_day between 1 and 7 then iif(:kf_disb_date < '27.06.16', 1, 0)--1.0
                                       else percent_snijen_plan
                                  end;
          --Если нет досрочного закрытия и просрочек то процент снижаемся на 0,2
          if (percent_snijen_plan = 0 and close_date_old >= 90 and oks_delinq_day <=0) then percent_snijen_plan = -0.2;
          percent_snijen_plan = K_Percent_Per_Day_old + percent_snijen_plan;
          if (percent_snijen_plan > 43.5) then percent_snijen_plan = 43.5;
          if (percent_snijen_plan < 37) then percent_snijen_plan = 37;
         
          --2,4 2 Не снижается если было досрочное погашение
          if (close_date_old < 90 and percent_snijen <> percent_snijen_plan) then
          begin
            res_error_is_critical = 2;
            res_error_text='Последный цикл кредита закрыт досрочно, годовая процентная ставка должно быть: '||percent_snijen_plan;
            suspend;
          end
        
          if (close_date_old >= 90 and percent_snijen <> percent_snijen_plan) then
          begin
            if (oks_delinq_day between 1 and 7) then
            begin
              res_error_is_critical = 2;
              res_error_text='В предыдущем цикле были просрочки до 7-дней, годовая процентная ставка должно быть: '||percent_snijen_plan;
              suspend;
            end
            
            if (oks_delinq_day between 8 and 15) then
            begin
              res_error_is_critical = 2;
              res_error_text='В предыдущем цикле были просрочки до 15-дней, годовая процентная ставка должно быть: '||percent_snijen_plan;
              suspend;
            end
            
            if (oks_delinq_day between 16 and 30) then
            begin
              res_error_is_critical = 2;
              res_error_text='В предыдущем цикле были просрочки до 30-дней, годовая процентная ставка должно быть: '||percent_snijen_plan;
              suspend;
            end
            
            if (oks_delinq_day > 30) then
            begin
                res_error_is_critical = 2;
                res_error_text='В предыдущем цикле были просрочки более 30-дней, годовая процентная ставка должно быть: '||percent_snijen_plan;
              suspend;
            end
            else
            begin
                res_error_is_critical = 2;
                res_error_text='Годовая процентная ставка должно быть: '||percent_snijen_plan;
              suspend;
            end
          end
          --Новые условия по снижению от 27 июня 16 года. приказ 62-06-16 о/д
          kommis_snij = case
                          when oks_delinq_day > 30 then 3.9
                          when oks_delinq_day between 16 and 30 then iif(:kf_disb_date < '27.06.16', 0.3, 0.2)--0.3
                          when oks_delinq_day between 8 and 15 then iif(:kf_disb_date < '27.06.16', 0.2, 0.1)--0.2
                          when oks_delinq_day between 1 and 7 then  iif(:kf_disb_date < '27.06.16', 0.1, 0)--0.1
                          else -0.1
                        end;
          k_kommis_perc_old = k_kommis_perc_old + kommis_snij;
        
          if (k_kommis_perc_old < 3.5) then k_kommis_perc_old = 3.5;
          if (k_kommis_perc_old > 3.9) then k_kommis_perc_old = 3.9;
      /*Новые усвловия которых нет в паспорте, условия поставлены по словам Алтын Шакировой:
        1. Доп.%% снижается если последний кредит был получен до 25 января 16 года и закрыт без просрочек, не досрочно 3,9-0,3
        2. Доп.%% ставится 3,9 если был закрыт досрочно или с просрочкой и дата выдачи до 25 января 16 года.
      */
      if (ks_date_begin_old < '25.01.16' and oks_delinq_day <= 0 and close_date_old >= 90) then
      k_kommis_perc_old = 3.8;
      if (ks_date_begin_old < '25.01.16' and oks_delinq_day > 0 or close_date_old < 90) then
      k_kommis_perc_old = 3.9; --Конец новых условии от 24.02.16
        
          /* Акция c 18 - 31 марта--
          1. Повторный клиент
          2. нет просрочек на послднем цикле
          */
          akcia = '';
          --k_fact_closed_date_old = coalesce(k_fact_closed_date_old, cast('31.12.9999' as date));
          if (:kf_disb_date between '18.03.2016' and '31.03.2016') then
          begin
           -- беру послдений закрытый займ клиента
            select kncl.kc_from_k_id, kont.k_kredline_end, ks.ks_date_begin
            from kontstate ks
            inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
            inner join kontract kont on ks.ks_from_k_id = k_id
            where ks.ks_state = 2
            and :kf_disb_date between ks.ks_date_begin and ks.ks_date_end
            and ks.ks_ol_is_delete = 0
            and kont.k_disb_date < :kf_disb_date
            and kncl.kc_amount_approved > 0
            and kncl.kc_from_c_id = :c_id
            order by kont.k_disb_date desc
            rows 1
            into :k_id_old_akc, :k_kredline_end_akc, k_fact_closed_date_akc;
            -- определяю макимальное число дней прссрочки на предыдущем цикле   
            max_delinq_day_cli_akc = null;
            select max(ok.ok_ol_date - ok_sh_date)
            from operkred ok
            where ok.ok_ol_is_active = 1
            and ok.ok_ol_date <= :kf_disb_date
            and ok.ok_ol_date > ok_sh_date
            and ok_type in(4,5,9)
            and ok_from_k_id = :k_id_old_akc
            and ok_os + ok_perc < 0
            and ok_from_c_id=0
            into max_delinq_day_cli_akc;
            max_delinq_day_cli_akc = coalesce(max_delinq_day_cli_akc, 0);
            k_fact_closed_date_akc = coalesce(k_fact_closed_date_akc, cast('31.12.9999' as date));
            if ((max_delinq_day_cli_akc <= 0) and
                (k_id_old_akc is not null) and
                (k_kredline_end_akc <= cast('31.05.2016' as date) or (k_fact_closed_date_akc < cast('18.03.2016' as date)))
                ) then
            begin
                k_kommis_perc_old = round(k_kommis_perc_old/2,1);
                akcia = ' (Акция)';
            end
          end

          if (k_kommis_perc <> k_kommis_perc_old and oks_delinq_day > 0) then
          begin
            res_error_is_critical = 2;
            res_error_text='В предыдущем цикле были просрочки: '||oks_delinq_day||' д., дополнительная процентная ставка должно быть: '||k_kommis_perc_old;
            suspend;
          end else
            if (k_kommis_perc <> k_kommis_perc_old) then
            begin
              res_error_is_critical = 1;
              res_error_text='Дополнительная процентная ставка должно быть: '||k_kommis_perc_old||akcia;
              suspend;
            end
            if (akcia = '') then
            begin
                if (k_kommis_perc not between 3.5 and 3.9) then
                begin
                  res_error_is_critical = 2;
                  res_error_text='Дополнительная процентная ставка должно быть: от 3.5% до 3.9%';
                  suspend;
                end
             end
             else
             begin
                if (k_kommis_perc not between 1.8 and 3.9) then
                begin
                  res_error_is_critical = 2;
                  res_error_text='Дополнительная процентная ставка должно быть: от 1.8% до 3.9%'||akcia;
                  suspend;
                end   
             end
          end
        end
      end
    end
  end
end
