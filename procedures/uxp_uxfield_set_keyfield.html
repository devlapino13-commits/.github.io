CREATE PROCEDURE UXP$UXFIELD_SET_KEYFIELD(
    UF_UUID D_UUID)
AS
declare variable uf_pk_field d_integer;
declare variable uf_is_param d_bool;
declare variable us_kind d_smallint;
declare variable us_is_table d_bool;
declare variable uf_field_name d_str_code;
declare variable uf_owner d_uuid;
declare variable tmp_field d_str_code;
declare variable keyfields d_str_100;
declare variable uf_source d_str_code;
declare variable us_is_view d_bool;
declare variable pk_found d_bool;
begin
  -- Назначает или удаляет признак ключевого поля

  select uf_pk_field, uf_owner, us_code, uf_field_name, uf_is_param, us_kind,
         us_is_table, us_is_view
  from ux$field, ux$source
  where uf_uuid = :uf_uuid and
        uf_owner = us_uuid
  into uf_pk_field, uf_owner, uf_source, uf_field_name, uf_is_param, us_kind,
       us_is_table, us_is_view;

  if (uf_is_param = 1) then
    exception e_message ('Параметр не может быть ключевым полем');

  if (us_kind = 0 and us_is_table = 1 and us_is_view = 0) then
    exception e_message ('По реальным таблицам ключевые поля меняются только на уровне БД.');

  tmp_field = '';
  keyfields = '';
  pk_found = 0;
  for select uf_field_name
  from ux$field
  where uf_owner = :uf_owner and
        uf_pk_field <> 0
  order by uf_pk_field
  into tmp_field
  do
  begin
    if (uf_pk_field <> 0 and tmp_field = uf_field_name) then
    begin
      pk_found = 1;
      Continue;
    end

    if (keyfields <> '') then
      keyfields = keyfields || ';';

    keyfields = keyfields || tmp_field;
  end

  if (uf_pk_field = 0) then
  begin
    if (keyfields <> '') then
      keyfields = keyfields || ';';

    keyfields = keyfields || uf_field_name;
  end

  if (keyfields = '' and pk_found = 0) then
  begin
    delete from ux$index
    where uix_source = :uf_source and
          uix_pk = 1;
  end else
  begin
    if (keyfields = '' and pk_found = 1) then
      keyfields = uf_field_name;

    update ux$index
    set uix_state = 1
    where uix_source = :uf_source and
          uix_fields = :keyfields;

    update or insert into ux$index (uix_source, uix_fields, uix_pk, uix_virtual, uix_state)
    values (:uf_source, :keyfields, 1, 1, 0)
    matching (uix_source, uix_pk);
  end

  update ux$source
  set us_keyfields = :keyfields
  where us_uuid = :uf_owner and
        us_keyfields is distinct from :keyfields;
end
