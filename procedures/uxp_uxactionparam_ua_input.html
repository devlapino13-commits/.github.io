CREATE PROCEDURE UXP$UXACTIONPARAM_UA_INPUT(
    UA_UUID D_UUID)
RETURNS (
    UAP_UA_UUID D_UUID,
    UAP_PARAM_NAME D_STR_CODE,
    UAP_OWNER_UFP_UUID D_UUID,
    UAP_OWNER_UFP_CODE D_STR_CODE,
    UAP_KIND D_SMALLINT,
    UAP_DEFAULT D_STR_255,
    UAP_USE_DEFAULT D_SMALLINT,
    UAP_UXA_UFP_CAPTION D_STR_255,
    UAP_LOCKED D_BOOL,
    RES_EDITOR D_STR_CODE,
    RES_STATE D_INTEGER,
    RES_READONLY D_BOOL,
    RES_UXA_UUID D_UUID,
    RES_CATEGORY_1 D_STR_255,
    UF_DEFAULT D_STR_100,
    UAP_EXEC_PARAM_ID D_SMALLINT,
    RES_EXPRESSION D_STR_LONG,
    RES_FRAME_NAME D_STR_100)
AS
declare variable uf_name d_str_code;
declare variable uf_pk_field d_smallint;
declare variable ufp_is_output d_bool;
declare variable uf_field_name d_str_code;
declare variable uf_is_param d_bool;
declare variable ufp_lookup d_uuid;
declare variable ufp_lookup_code d_str_code;
declare variable us_code d_str_code;
declare variable prv_lookup d_uuid;
declare variable ufr_detail_field d_str_code;
declare variable uap_uxa_ufp_uuid d_uuid;
declare variable ufm_uuid d_uuid;
declare variable res_source_field d_str_code;
declare variable uxa_kind d_integer;
declare variable ufp_public d_bool;
begin
  if (ua_uuid is null) then
    Exit;

  select uxa_uuid, uxa_kind
  from ux$action, ux$alias
  where ua_uuid = :ua_uuid and
        ua_uxa_uuid = uxa_uuid
  into res_uxa_uuid, uxa_kind;

  if (res_uxa_uuid is null) then
    exit;

  ufm_uuid = uxf$ver(res_uxa_uuid, 'draft');

  uap_ua_uuid = ua_uuid;
  res_state = 0;
  uap_exec_param_id = 0;

  ufp_is_output = 0;
  prv_lookup = null;
  for select distinct ufp_uuid, uf_name, uf_field_name, ufp_editor, uf_pk_field, uf_is_param,
             ufp_parent, ufp_public, uf_default, ufr_detail_field, ufr_name
  from ux$frparam, ux$field, ux$source fld, ux$frame, ux$source fr
  where ufr_owner = :ufm_uuid and
        ufp_ufr_uuid = ufr_uuid and
        ufp_uf_uuid = uf_uuid and
        uf_owner = fld.us_uuid and
        ufr_us_uuid = fr.us_uuid and
        ufp_parent is null
  order by ufr_is_mainframe desc, ufr_uuid, uf_is_param desc, uf_position
  into uap_uxa_ufp_uuid, uf_name, uf_field_name, res_editor, uf_pk_field, uf_is_param,
       ufp_lookup, ufp_public, uf_default, ufr_detail_field, res_frame_name
  do
  begin
    -- Если действие на добавление записи, отображаем все поля для передачи значений по умолчанию при вставки
    if (uxa_kind <> 2 and (not(uf_is_param = 1 or uf_pk_field <> 0))) then
      continue;

    if (uf_is_param = 1 and ufp_public <> 1) then
      continue;

    uap_param_name = uf_field_name;

    if (uf_field_name starting 'preblank$') then
      Continue;

    res_category_1 = '';
    if (ufp_lookup is not null) then
    begin
      if (uf_is_param = 1) then
      begin
        if (prv_lookup is distinct from ufp_lookup) then
        begin
          ufp_lookup_code = '';
          select uf_name
          from ux$frparam, ux$field
          where ufp_uuid = :ufp_lookup and
                ufp_uf_uuid = uf_uuid
          into ufp_lookup_code;

          us_code = '';
          select us_code
          from ux$frlookup, ux$source
          where ufl_ufp_uuid = :ufp_lookup and
                ufl_us_uuid = us_uuid
          into :us_code;
        end
        res_category_1 = 'Выборка ' || Upper(ufp_lookup_code) || ', параметры ' || upper(us_code);
      end
    end else
    begin
      if (uf_is_param = 1) then
        res_category_1 = ' Параметры';
      else
        res_category_1 = ' Поля';
    end

    prv_lookup = ufp_lookup;

    uap_kind = ufp_is_output;

    -- если в форме один фрейм, не показываем источник чтобы текст был покороче
    uap_uxa_ufp_caption = uf_field_name;

    if (uf_pk_field <> 0) then
      uap_uxa_ufp_caption = uap_uxa_ufp_caption || ' (pk' || uf_pk_field || ')';

    if (uap_uxa_ufp_caption not containing uf_name) then
      uap_uxa_ufp_caption = uap_uxa_ufp_caption || '   - ' || uf_name;

    select res_def_value, res_owner_ufp_uuid, res_owner_ufp_code, res_source_field,
           res_use_default, res_state, res_locked, res_readonly, res_exec_param_id
    from uxp$uxfrparam_value(:uap_uxa_ufp_uuid, :res_uxa_uuid, :ua_uuid)
    into :uap_default, uap_owner_ufp_uuid, uap_owner_ufp_code, res_source_field,
         :uap_use_default, :res_state, :uap_locked, :res_readonly, uap_exec_param_id;

    if (uap_owner_ufp_uuid is null and res_source_field <> '') then
      uap_owner_ufp_code = res_source_field;

    if (ufr_detail_field = uf_field_name) then
    begin
      uap_locked = 1;
      res_readonly = 1;
      res_state = -5;
    end

    res_expression = '';
    select exp_expression
    from ux$expression
    where exp_ux_uuid = :uap_uxa_ufp_uuid and
          exp_kind = 1
    into res_expression;

    if (uap_locked = 0 and res_expression <> '') then
    begin
      uap_locked = 1;
      res_readonly = 1;
      res_state = -6; -- Параметр с выражением
    end

    suspend;
  end
end
