CREATE PROCEDURE HP_ZALOGAUTO_SEND(
    ZA_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable lang d_str_lang;
declare variable za_from_ma_id d_integer;
declare variable za_from_k_id d_id;
declare variable za_from_c_id d_id;
declare variable za_orig_sts_id d_id;
declare variable za_orig_pts_id d_id;
declare variable zal_id d_integer;
declare variable zal_cost d_currency;
declare variable blob_count d_integer;
declare variable min_blob_count d_integer;
declare variable zap_id d_integer;
declare variable zap_status d_smallint;
declare variable zap_status_str d_str_code;
declare variable zal_webpage d_str_long;
declare variable zal_status d_smallint;
declare variable z_id d_integer;
begin
  -- Отправка автозалога на рассмотрение
  res_is_error = 0;
  res_text = '';
  min_blob_count = 6; -- Минимально необходимое количество фото залога

  lang = f_get_lang_code(un_id, null);

  select za_from_ma_id, za_from_k_id, za_from_c_id, za_orig_sts_id,
    za_orig_pts_id, zal_id, zal_cost, zal_webpage, za_from_z_id
  from zalogauto, zalogautolist
  where za_id = :za_id and
        za_from_zal_id = zal_id
  into za_from_ma_id, za_from_k_id, za_from_c_id, za_orig_sts_id, za_orig_pts_id,
    zal_id, zal_cost, zal_webpage, z_id;

  if (za_from_k_id is null and za_from_ma_id is null) then
    res_text = f_lang(lang, 'Обратитесь в техподдержку, заявка и контракт не указаны');

  if (za_from_c_id is null) then
    res_text = f_lang(lang, 'Не указан поручитель');

  if (za_orig_sts_id is null) then
    res_text = f_lang(lang, 'Не указан документ СТС (до оформления)');

  if (za_orig_pts_id is null) then
    res_text = f_lang(lang, 'Не указан документ ПТС (до оформления)');

  if (coalesce(zal_cost, 0) = 0) then
    res_text = f_lang(lang, 'Не указанa стоимость автозалога');

  if (z_id is not null) then
    res_text = f_lang(lang, 'Залог уже оформлен');

  blob_count = 0;
  select count(*)
  from zalogautoblob, blobstore
  where
    zab_from_zal_id = :zal_id and
    zab_from_bls_id = bls_id and
    bls_source_system = 1 -- Закрепленные в МП
  into blob_count;

  if (blob_count < min_blob_count) then
    res_text = f_format(
      f_lang(lang, 'Недостаточно фотографий залога: %1 (требуется минимум: %2)'),
      blob_count, min_blob_count);

  zap_id = null;
  select zap_id, zap_status
  from zalogautoprof
  where zap_from_za_id = :za_id
  into zap_id, zap_status;

  if (zap_id is null) then
    res_text = f_lang(lang, 'Не найдено профсуждение');
  else
  begin
    if (zap_status not in (-3, -1)) then
    begin
      zap_status_str =
        case zap_status
          when -3 then 'Новый (не отправлен)'
          when -2 then 'Отказ'
          when -1 then 'Доработка'
          when  0 then 'В очереди'
          when  1 then 'На рассмотрении'
          when  2 then 'Одобрен'
          else 'Ошибка (отсутствует)'
        end;
      res_text = f_format(
        f_lang(lang, 'Нельзя отправить на проверку. Неверный статус: %1'),
        zap_status_str);
    end
  end

  if (res_text = '') then
  begin
    if (coalesce(zal_webpage, '') = '') then
      zal_status = 2;
    else
      zal_status = 1;

    update zalogautolist
    set zal_status = :zal_status
    where zal_id = :zal_id;

    select res_is_error, res_text
    from hp_actionlist_zalogauto_add(:zap_id, :un_id)
    into :res_is_error, :res_text;
  end

  if (res_text <> '') then
    res_is_error = 1;

  suspend;
end
