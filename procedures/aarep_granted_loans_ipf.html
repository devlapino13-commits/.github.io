CREATE PROCEDURE AAREP_GRANTED_LOANS_IPF(
    REP_REGION_ID ID_TYPE,
    K_ID_IN ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME STR_100,
    K_KONTRACT_NO STR_100,
    CL_FIO_ALL STR_100,
    K_DISB_DATE DATE,
    PR_NAME STR_100,
    K_AMOUNT_APPROVED CURR_TYPE,
    KML_SELLER_FIO STR_100,
    KML_INN STR_20,
    KML_NAME STR_255,
    KML_COUNT CURR_TYPE,
    KML_TOTAL CURR_TYPE,
    KML_ACTUAL CURR_TYPE,
    KML_MODEL STR_255,
    KS_FIO STR_100,
    FOLDER_NAME STR_100,
    KML_ISSUE_DATETIME TIMESTAMP,
    DATE_PAYMENT D_DATE,
    FEG_CODE D_STR_10,
    KMI_IMEI D_STR_20)
AS
declare variable office_id id_type;
declare variable office_id_tmp id_type;
declare variable k_id id_type;
declare variable db_id id_type;
declare variable kf_id id_type;
declare variable kml_c_id id_type;
declare variable date_purchase date;
declare variable rd_id d_integer;
declare variable rd_id1 d_integer;
declare variable rd_id2 d_integer;
declare variable rd_id3 d_integer;
declare variable kml_bill_id id_type;
declare variable kml_bill_line_no integer;
declare variable kml_id integer;
declare variable feg_id d_integer;
begin
  /* Процедура разработана по запросу бухгалтерии.
  Данные получаем по выданным займам по исламскому принципу финансирования.
  необходимые поля:
  - офис(на конец отчета)
  - № договора/папки
  - ФИО клиента
  - дата выдачи
  - кр.продукт
  - сумма выдачи
  - наименование Поставщика/Контрагента
  - наименование товара/услуг
  - количество ед.
  - стоимость товара/услуг (Цена товара без наценки)
  - фактические выплаты в пользу контрагента

--Добавляем поля Модель товара,  ФИО СпФ, номер папки, время покупки товара -- 31/01/2019
--    KML_MODEL
--    KML_KC_FIO
--    KML_FOLDER_NAME
--    KML_ISSUE_DATETIME

  Отчет может работать на период по всем активным кредитам, либо можно
  получить на опеределнный контракт зная дату выдачи!
  */
  db_id = f_get_db_id();
  k_id_in = coalesce(k_id_in, 0);
  rep_date_begin = coalesce(:rep_date_begin, :rep_date_end);

  select rd_id from refdata
  where rd_code = 'AgreementKinds.IPF_Sellers'
  into :rd_id;

  select rd_id from refdata
  where rd_code = 'AgreementKinds.tsp_and_ipf'
  into :rd_id1;

  select rd_id from refdata
  where rd_code = 'AgreementKinds.IPF_Idjara'
  into :rd_id2;

  select rd_id from refdata
  where rd_code = 'AgreementKinds.IPF_WithCommission'
  into :rd_id3;

  for select office_id, k_id, k_kontract_no,  k_disb_date,  k_amount_approved,  pr_name,  ks_fio, -- КС, который выдал заем
                      kf_id
      from AAREP_GRANTED_LOANS_IPF_TMP(:rep_region_id,:k_id_in, :rep_date_begin, :rep_date_end)
      into :office_id, :k_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :pr_name, :ks_fio,
                      :kf_id
  do
  begin
    select kal.kal_new_kf_name
    from knaktlist kal
    where kal.kal_from_kf_id = :kf_id and
          :rep_date_end between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into :folder_name; --added by Aleksei on 31-01-2019

    if (coalesce(office_id_tmp, 0) <> office_id) then
    begin
      select reg_name
      from region
      where reg_unicode = :office_id
      into :office_name;
      office_id_tmp = office_id;
    end

    cl_fio_all = '';
    select list(c_name_fio, ', ')
    from knclient, clients
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = c_id
    into :cl_fio_all;

    for select
          ag.ag_object_id,
          c.c_name_fio,
          ml.ml_name,
          km.kml_count,
          km.kml_total,
          ml.ml_model,
          c.c_inn,
          km.kml_bill_id,
          km.kml_bill_line_no,
          km.kml_from_feg_id,
          km.kml_id
        from kontmat km
        inner join agreements ag on km.kml_from_ag_id = ag.ag_id
        inner join clients c on ag.ag_object_id = c.c_id
        inner join matlist ml on km.kml_from_ml_id = ml.ml_id
        where km.kml_object_kind in (9,11) and 
              km.kml_object_id = :k_id and
              km.kml_from_ml_id is not null and
              ag.ag_from_rd_id in (:rd_id, :rd_id1, :rd_id2, :rd_id3) and
              ag.ag_object_kind = 0
        into
          :kml_c_id,
          :kml_seller_fio,
          :kml_name,
          :kml_count,
          :kml_total,
          :kml_model,
          :kml_inn,
          :kml_bill_id,
          :kml_bill_line_no,
          :feg_id,
          :kml_id
    do
    begin
      kml_actual = 0;
      date_payment = null;
      kml_issue_datetime = null;
      feg_code = null;
      kmi_imei = null;

      select feg_code from foreign_econom_goods_ref
      where feg_id = :feg_id
      into :feg_code;

      select cast(kmi_imei as d_str_20) from kontmat_imei
      where kmi_kml_id = :kml_id
      into :kmi_imei;

      select first 1 ol.ol_datetime
      from opermat om
      inner join operlist ol on om.om_from_ol_id = ol.ol_id
      where om.om_bill_id = :kml_bill_id and
            om.om_bill_line_no = :kml_bill_line_no and
            om.om_oper_type = 5 and
            ol.is_delete = 0
      into :kml_issue_datetime;

      if (db_id in (39801, 41701)) then
      begin
        date_purchase = coalesce(cast(kml_issue_datetime as date), :rep_date_begin);

        date_payment = (select esm.esm_date
                        from kontpaytasks kpt
                        inner join extsysmain esm on esm.esm_id = kpt.kpt_from_esm_id and
                                                     esm.esm_status = 1
                        where kpt.kpr_from_k_id = :k_id and
                              kpt.kpt_status in (1,2)
                              order by kpt.kpt_datetime desc
                              rows 1);
        if (date_payment is null) then
          date_payment = (
            select first 1 ob_ol_date
            from operbook
            where ob_client_dt = :kml_c_id
              and ob_ol_date between :date_purchase and :rep_date_end
              and ob_short_acc_dt = '21199/99'
              and ob_detail containing :k_kontract_no
              and ob_ol_is_active = 1);

        if (date_payment is not null)
          then kml_actual = kml_total;
      end
      suspend;
    end
  end
end
