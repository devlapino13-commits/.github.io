CREATE PROCEDURE PRINT_AZ_SALARY(
    K_ID D_ID)
RETURNS (
    EMPLOYEE_POSITIONLABOURCONTRACT D_STR_255,
    EMPLOYEE_POSITION D_STR_100,
    EMPLOYEE_SALARY D_CURRENCY,
    EMPLOYEE_WORKPLACE D_STR_255,
    EMPLOYER_PHONE D_STR_100,
    EMPLOYER_LEGALADDRESS D_STR_255,
    EMPLOYER_NAME D_BLOB_TEXT)
AS
declare variable COUNTER D_INTEGER;
declare variable ACTIVE_COUNT D_INTEGER;
declare variable C_ID D_ID;
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_STR_8100;
begin
  employee_positionlabourcontract = '';
  employee_position = '';
  employee_salary = 0;
  employee_workplace = '';
  employer_phone = '';
  employer_legaladdress = '';
  employer_name = '';

  -- На момент написания в АЗ нет групповых кредитов
  select kc.kc_from_c_id
  from knclient kc
  where kc.kc_from_k_id = :k_id
  into :c_id;

  select eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id and
        eic.eic_code = 'salary'
  where eil.eil_from_c_id = :c_id and
        eil.eil_status = 2
  order by eil.eil_id desc
  rows 1
  into :bls_id;

  if (bls_id is not null) then
  begin
    select cast(blx_blob as d_blob_text)
    from hp_get_blobex_blob(:bls_id)
    into :json_text;

    counter = 0;
    active_count = f_json_str_get_count(json_text, 'Active');
    while (active_count > counter) do
    begin
      employee_positionlabourcontract = f_json_str_get_value(:json_text, 'Active[' || cast(counter as varchar(25)) || '].Employee.PositionLabourContract', '');
      employee_position = f_json_str_get_value(:json_text, 'Active[' || cast(counter as varchar(25)) || '].Employee.Position', '');
      employee_salary = f_json_str_get_value(:json_text, 'Active[' || cast(counter as varchar(25)) || '].Employee.Salary', 0);
      employee_workplace = f_json_str_get_value(:json_text, 'Active[' || cast(counter as varchar(25)) || '].Employee.WorkPlace', '');
      employer_phone = f_json_str_get_value(:json_text, 'Active[' || cast(counter as varchar(25)) || '].Employer.Phone', '');
      employer_legaladdress = f_json_str_get_value(:json_text, 'Active[' || cast(counter as varchar(25)) || '].Employer.LegalAddress', '');
      employer_name = f_json_str_get_value(:json_text, 'Active[' || cast(counter as varchar(25)) || '].Employer.Name', '');

      counter = counter + 1;
      suspend;
    end
  end
  else
    suspend;
end
