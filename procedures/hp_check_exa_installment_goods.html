CREATE PROCEDURE HP_CHECK_EXA_INSTALLMENT_GOODS(
    GROUP_ID D_INTEGER)
AS
declare variable enum_status d_smallint;
declare variable exa_id d_id;
declare variable mp_id d_id;
declare variable sql_request d_str_255;
begin
  enum_status = f_enum_id('mc_purchase_status', 'in_process', null);

  for select mp.mp_from_exa_id, mp.mp_id
      from mc_purchase mp
      inner join expressapp exa on exa.exa_id = mp.mp_from_exa_id
      inner join expresscfg exc on exc.exc_id = exa.exa_from_exc_id
      where exa.exa_from_exp_id = -2 and
            exc.exc_action_code = 'miniapp.create' and
            exa.exa_date = current_date and
            mp.mp_status = :enum_status and
            exa.exa_group_id = :group_id
      into exa_id, mp_id
  do
  begin
    sql_request = 'execute procedure hp_disb_installment_goods(' || :mp_id || ', ' || :exa_id || ')';

    execute statement sql_request
        with autonomous transaction;
  end
end
