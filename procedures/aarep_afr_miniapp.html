CREATE PROCEDURE AAREP_AFR_MINIAPP(
    REP_REG_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_64,
    KAL_NEW_KF_NAME D_STR_100,
    MA_ID D_INTEGER,
    MA_NO D_INTEGER,
    MA_DATE_APP D_DATE,
    MA_SUMMA_MINUS D_CURRENCY,
    MA_IS_DELETE D_INTEGER,
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    C_INN D_STR_20,
    C_AGE D_SMALLINT,
    IS_STOP_CREDIT_APP D_STR_10,
    IS_STOP_CREDIT_DISB D_STR_10,
    IS_BANKRUPT D_STR_10,
    IS_CHECKED D_STR_10,
    IS_MATCHED D_STR_CODE)
AS
declare variable REG_UNICODE D_INTEGER;
declare variable KF_ID D_ID;
declare variable MA_DATE_SAVE D_DATETIME;
declare variable DISB_OL_DATETIME D_DATETIME;
declare variable AC_STATUS D_STR_CODE;
begin
  -- данные отчета "Заявки на кредит" (RFCKZ-370)

  for select id_list
      from region_list(:rep_reg_id, 1, 3, 1)
      into reg_unicode
  do
  begin
    ma_id = null;
    ma_no = null;
    ma_date_app = null;
    ma_date_save = null;
    ma_summa_minus = null;
    ma_is_delete = null;
    kf_id = null;

    for select ma_id,
               ma_no,
               ma_date_app,
               ma_date_save,
               ma_summa_minus,
               iif(ma_is_active = 1, 0, 1),
               ma_from_kf_id
        from miniapp
        where ma_office_id = :reg_unicode and
              ma_date_app between :rep_date_begin and :rep_date_end and
              ma_date_end = '31.12.9999'
        into ma_id,
             ma_no,
             ma_date_app,
             ma_date_save,
             ma_summa_minus,
             ma_is_delete,
             kf_id
    do
    begin
      kal_new_kf_name = null;
      reg_name = null;

      select kal_new_kf_name
      from knaktlist
      where kal_from_kf_id = :kf_id and
            :ma_date_app between kal_date and kal_date_end
      into kal_new_kf_name;

      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_unicode
      into reg_name;

      for select mac_from_c_id
          from miniappclient
          where mac_from_ma_id = :ma_id and
                mac_amount > 0
          into c_id
      do
      begin
        c_name_fio = null;
        c_inn = null;
        c_age = null;
        is_stop_credit_app = null;
        is_stop_credit_disb = null;
        disb_ol_datetime = null;

        select c_name_fio,
               c_inn,
               f_age(c_birth_date, :ma_date_app)
        from clients
        where c_id = :c_id
        into c_name_fio,
             c_inn,
             c_age;

        select first 1
            f_enum('antifraud_status', ac.ac_status, 'code')
        from antifraud_check ac
        where ac.ac_inn = :c_inn
        order by ac.ac_id desc
        into :ac_status;

        is_checked = 'Нет';

        if (ac_status is not null) then
        begin
            is_checked = 'Да';
            is_matched =
                case
                    when :ac_status = 'not_found' then 'Нет'
                    when :ac_status = 'was_found' then 'Да'
                    when ac_status = 'in_process' then 'В обработке'
                    else 'Ошибка проверки'
                end;
        end

        is_bankrupt = iif(f_bankrupt(:c_id, :ma_date_save) = 1, 'Да', 'Нет');

        select iif(status = 1, 'Да', 'Нет')
        from hp_stop_credit_status_client(:c_id, :ma_date_save)
        into is_stop_credit_app;

        select ol_datetime
        from operlist
        inner join kontract on k_fact_disb_ol_id = ol_id
        where k_status = 1 and
              k_from_ma_app_id = :ma_id
        into disb_ol_datetime;

        if (disb_ol_datetime is not null) then
          select iif(status = 1, 'Да', 'Нет')
          from hp_stop_credit_status_client(:c_id, :disb_ol_datetime)
          into is_stop_credit_disb;

        suspend;
      end
    end
  end
end
