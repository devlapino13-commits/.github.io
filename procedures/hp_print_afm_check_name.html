CREATE PROCEDURE HP_PRINT_AFM_CHECK_NAME(
    K_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    REQUEST_ID D_STR_CODE,
    RESPONSE_DATETIME D_DATETIME,
    SEARCH_DATETIME D_DATETIME,
    C_NAME_FIO D_STR_100,
    RES_TEXT D_BLOB_TEXT,
    QR_CODE D_STR_LONG)
AS
declare variable C_ID D_ID;
declare variable MA_ID D_ID;
declare variable BLOB_ID D_ID;
declare variable JSON_INFO D_BLOB_TEXT;
begin
    select ac.ac_from_c_id, ac.ac_from_ma_id
    from afm_check ac
    join kontract k on ac.ac_from_ma_id = k.k_from_ma_app_id
    where k.k_id = :k_id
          and ac.ac_is_actual = 1
    into :c_id, :ma_id;

    select res_text
    from hp_print_afm_result(:c_id, :ma_id)
    into :res_text;

    select c_name_fio
    from clients
    where c_id = :c_id
    into :c_name_fio;

    select eil.eil_from_bls_id, eil.eil_create_datetime, eil.eil_status_datetime
    from extinfolist eil
    join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
                           and eic.eic_code = 'afm_name_check'
    where eil.eil_object_id = :ma_id
          and eil.eil_object_kind = f_objkind('miniapp', 'id')
          and eil.eil_status = 2
          and eil.eil_from_bls_id is not null
          and coalesce(eil.eil_from_bls_id, 0) > 0
    into :blob_id, :search_datetime, :response_datetime;

    in autonomous transaction do
        select blx_blob
        from hp_get_blobex_blob(:blob_id)
        into :json_info;

    :request_id = f_json_blob_get_value(:json_info, 'data.id', '');
    :qr_code = 'https://aml-system.kz/check-request/' || :request_id;
    suspend;
end
