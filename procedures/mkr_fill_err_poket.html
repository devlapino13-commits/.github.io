CREATE PROCEDURE MKR_FILL_ERR_POKET(
    POKET_ID DECIMAL(18,0),
    LP_NUMBER INTEGER)
AS
declare variable pocket_id decimal(18,0);
declare variable p_type smallint;
declare variable report_date_begin date;
declare variable report_date_end date;
declare variable mp_categ smallint;
begin
  /*
    CREATED BY IGUSEV 21.10.2012

    PURPOSE: Для МКР. формирование кредитоного пакета по логам с ошибками получеными от мкр.

    */
  select mp.mp_date_begin, mp.mp_date_end, mp_type, mp.mp_categ
  from mkr_poket mp
  where mp.mp_from_mp_id = :poket_id and
        mp.mp_lp_number = :lp_number
  into :report_date_begin, :report_date_end, :p_type, :mp_categ;

  /* Создаю покет по активным и закрытым кредитам ----------------------------*/
  pocket_id = gen_id(gen_mkrpoket_id, 1);
  insert into mkr_poket (mp_id, mp_create_time, mp_date_begin, mp_date_end, mp_type, mp_from_mp_id, mp_lp_number,
                         mp_categ)
  values (:pocket_id, cast('NOW' as timestamp), :report_date_begin, :report_date_end, :p_type, :poket_id, :lp_number,
          :mp_categ);

  /*--------------------------------------------------------------------------*/
  insert into mkr_credit (mc_type, mc_from_mp_id, mc_id, mc_from_c_id, mc_name, mc_placeofbirth, mc_countrycode,
                          mc_pincode, mc_bankruptcystatus, mc_dateofbirth, mc_from_k_id, mc_accountno, mc_credittype,
                          mc_creditlineamount, mc_purposeofcredit, mc_currencyofcredit, mc_initialamountofcredit,
                          mc_disoutamountofcredit, mc_annualinterestrate, mc_dateofgrant, mc_duetimefirstcontract,
                          mc_duetimelastcontract, mc_lastpaymentdate, mc_monthlypaymentamount, mc_daysmainsumisoverdue,
                          mc_daysinterestisoverdue, mc_creditperiodinmonths, mc_oiaforrepperiod,
                          mc_oiaforlastthreemonths, mc_numberofprolongs, mc_creditclasscode, mc_creditstatuscode,
                          mc_provision)
  select cr_type, :pocket_id, id, knc_id, name, placeofbirth, countrycode, pincode, bankruptcystatus, dateofbirth, k_id,
         accountno, credittype, creditlineamount, purposeofcredit, currencyofcredit, initialamountofcredit,
         disoutamountofcredit, annualinterestrate, dateofgrant, duetimefirstcontract, duetimelastcontract,
         lastpaymentdate, monthlypaymentamount, daysmainsumisoverdue, daysinterestisoverdue, creditperiodinmonths,
         oiaforrepperiod, oiaforlastthreemonths, numberofprolongs, creditclasscode, creditstatuscode, provision
  from mkr_prepare_credlist(:report_date_begin, :report_date_end, :mp_categ) t
  inner join mkr_err_log_credits elc on elc.lc_k_id = t.k_id and
        elc.lc_knc_number = t.kc_number
  where elc.lc_from_poket_id = :poket_id and
        elc.lc_from_lp_number = :lp_number;
  exit;
end
