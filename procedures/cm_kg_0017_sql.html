CREATE PROCEDURE CM_KG_0017_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kf_id id_type;
declare variable reg_id integer;
declare variable k_kolich_repay integer;
declare variable k_kolich_client_approved integer;
declare variable k_percent_per_day perc_type;
declare variable pr_product_kind integer;
declare variable k_circle integer;
declare variable k_kommis_perc perc_type;
declare variable k_amount_approved curr_type;
declare variable pr_name varchar(50);
declare variable pr_code varchar(100);
declare variable kolich_poruch integer;
declare variable re_k_id id_type;
declare variable re_kol_cli integer;
declare variable re_percent_day perc_type;
declare variable kd_ol_date date;
declare variable kd_ol_date1 date;
declare variable percent_snijen curr_type;
declare variable percent_snijen_plan curr_type;
declare variable re_k_id_cli_prior id_type;
declare variable max_delinq_day_cli integer;
declare variable c_id_delinq id_type;
declare variable max_kol_delinq integer;
declare variable c_id id_type;
declare variable re_k_id_cli id_type;
declare variable re_k_disb_date date;
declare variable re_k_close_date date;
declare variable is_repeat integer;
declare variable re_disb_date_prior date;
declare variable kol_delinq integer;
declare variable delinq_day_cli integer;
declare variable ma_app_id integer;
declare variable kbm_id curr_type;
declare variable kol_kib_cli integer;
declare variable plan_kommis_perc perc_type;
declare variable re_pr_code varchar(100);
declare variable c_curr_bonus_status integer;
declare variable re_kommis_perc perc_type;
declare variable kommis_snijen perc_type;
declare variable re_max_delinq_day integer;
declare variable re_kol_delinq integer;
begin
-- Условия выдачи кредитного продукта "ГРУППОВОЙ-ВЫГОДНЫЙ"
  for
  select k_disb_date, k_from_kf_id, k_last_reg_id, k_kolich_repay, k_kolich_client_approved, K_Percent_Per_Day,
          pr_product_kind, k_circle, k_kommis_perc, k_amount_approved, pr_name, pr_code,
          k_from_ma_app_id
  from kontract, product
  where k_id=:k_id
  and k_from_pr_id = pr_id
  and pr_code containing '.VYGOD.'
  into :kf_disb_date, :kf_id, :reg_id, :k_kolich_repay, :k_kolich_client_approved, K_Percent_Per_Day,
       :pr_product_kind, k_circle, k_kommis_perc, k_amount_approved, pr_name, pr_code,
       :ma_app_id
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;
    if (round(K_Percent_Per_Day * 365, 4) = 29.9000) then
    begin
--    если новый кредит то проверяем не брал ли в другом ФКУ
      if (k_circle <= 1) then
      begin
--      Для групп, переходящих из других ФКУ(в составе не менее 2х человек)
        kol_kib_cli = 0;
        for
        select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id
        and kc_amount_approved > 0
        and kc_circle <= 1
        into c_id
        do
        begin
          kbm_id = null;
          select kbm_id
          from hp_kibminiapp_last(:c_id, :ma_app_id)
          into kbm_id;

--        проверка клиента, имеется ли история в другом ФКУ
          if (kbm_id is not null) then
          begin
            if (exists(
                select *
                from kibhistory
                where kh_from_kbm_id = :kbm_id
                and kh_creditstatus not containing('Отказано')
                and kh_creditstatus not containing('Рассмотрение заявки')
                and kh_is_our = 0
                )) then
            begin
              kol_kib_cli = kol_kib_cli + 1;
            end
          end
        end
        if (kol_kib_cli >= 2) then plan_kommis_perc = 5.7;
        else plan_kommis_perc = 5.9;
      end else
      begin
--      определяется предыдущий контракт
        re_k_id = null;
        re_pr_code = null;
        select k_id, pr_code, k_kommis_perc
        from kontstate, kontract, product
        where ks_ol_is_delete = 0
        and ks_state = 1
        and ks_from_k_id = k_id
        and k_from_kf_id = :kf_id
        and k_disb_date < :kf_disb_date
        and k_from_pr_id = pr_id
        order by ks_date_begin desc
        rows 1
        into re_k_id, re_pr_code, re_kommis_perc;
        if (re_pr_code not containing '.VYGOD.') then
        begin
--        если предыдущий цикл не выгодный продукт то проверяем статус контракта
          C_CURR_BONUS_STATUS = null;
          select C_CURR_BONUS_STATUS from HP_KONTSTATE_LOAN_STATUS(:k_id, 0, 0, :kf_disb_date, null)
          into C_CURR_BONUS_STATUS;
          if (C_CURR_BONUS_STATUS is null) then C_CURR_BONUS_STATUS = 0;
          plan_kommis_perc = case
                                when C_CURR_BONUS_STATUS = 0 then 5.9
                                when C_CURR_BONUS_STATUS = 1 then 5.8
                                when C_CURR_BONUS_STATUS = 2 then 5.7
                                when C_CURR_BONUS_STATUS = 3 then 5.6
                               when C_CURR_BONUS_STATUS >= 4 then 5.5
                             end;
        end else
        begin
--        по продукту выгодный, прверка предыдущий цикл
--        вычисляем максисальный просроченных день в предыдущем цикле где брал ВЫГОДНЫЙ
          re_max_delinq_day = null;
          re_kol_delinq = null;
          select max(ok_ol_date - ok_sh_date), count(*)
          from operkred
          where ok_ol_is_active = 1
          and ok_from_k_id = :k_id
          and ok_type in (4,5,9)
          and ok_os + ok_perc < 0
          and ok_ol_date > ok_sh_date
          into re_max_delinq_day, re_kol_delinq;
          if (re_max_delinq_day is null) then re_max_delinq_day = 0;
          if (re_kol_delinq is null) then re_kol_delinq = 0;
          kommis_snijen = case
                            when re_max_delinq_day = 0 then 0.1
                            when re_max_delinq_day in (1,2) and re_kol_delinq = 1 then 0.0
                            when re_max_delinq_day in (1,2) and re_kol_delinq > 1 then -0.2
                            when re_max_delinq_day between 3 and 7 then -0.3
                            when re_max_delinq_day between 8 and 15 then -0.5
                            when re_max_delinq_day between 16 and 30 then -10
                            else -100
                          end;
          plan_kommis_perc = re_kommis_perc - kommis_snijen;
          if (plan_kommis_perc < 2.9) then plan_kommis_perc = 2.9;
          if (plan_kommis_perc > 5.9) then plan_kommis_perc = 5.9;
        end
      end
      if (plan_kommis_perc <> k_kommis_perc) then
      begin
        res_error_is_critical = 2;
        res_error_text='Доп. процент не правильно указан, посмотрите предыдущий цикл контракта!';
        suspend;
      end 
    end else
    begin
      res_error_is_critical = 2;
      res_error_text='Для данного кр.продукта % должен быть 29,9%';
      suspend;
    end 
  end
end
