Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_INSURANCE_TARIF_UPD(
    IT_ID D_ID,
    IT_IS_ACTIVE D_SMALLINT,
    IT_LAST_DATETIME D_DATETIME,
    IT_MIN_POINT D_INTEGER,
    IT_MAX_POINT D_INTEGER,
    IT_TARIF D_PERCENT,
    IAC_COMMIS D_CURRENCY,
    IT_TARIF_TYPE D_INTEGER)
AS
begin
    if (coalesce(it_tarif_type, 0) not in (0, 1)) then exception e_message('Неверный тип тарифа');

    update insurance_tarif
    set it_is_active = :it_is_active,
        it_last_datetime = current_timestamp,
        it_min_point = :it_min_point,
        it_max_point = :it_max_point,
        it_tarif = :it_tarif,
        it_tarif_type = :it_tarif_type
    where it_id = :it_id;

    update insur_ag_commis
    set iac_commis = :iac_commis,
        iac_datetime = current_date
    where iac_from_it_id = :it_id;
end
