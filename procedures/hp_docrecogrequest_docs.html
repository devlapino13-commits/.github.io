CREATE PROCEDURE HP_DOCRECOGREQUEST_DOCS(
    C_ID D_ID)
RETURNS (
    DRM_ID D_INTEGER)
AS
declare variable dp_id d_integer;
begin
  if (f_get_const('docrecogrequest_active') <> 1) then
    exit;

  for select drr_from_dp_id, drr_from_drm_id
  from docrecogrequest
       inner join docprop on (drr_from_dp_id = dp_id)
       left join docrecogmain on (drr_from_drm_id = drm_id)
  where drr_from_c_id = :c_id and
        drr_status in (0, -1) and
        drr_create_datetime >= (current_date - 10)
  into dp_id, drm_id
  do
  begin
    if (coalesce(drm_id, 0) > 0) then
      suspend;
    else
    begin
      for select drm_id
      from hp_docprop_docrecogmain_list('clients', :c_id, :dp_id)
      where drm_id > 0
      into :drm_id
      do
        suspend;
    end
  end
end
