CREATE PROCEDURE AAREP_CLIENT_LINKS(
    REP_C_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable c_cell_no varchar(100);
declare variable contact_fio varchar(255);
declare variable rel_name varchar(100);
begin
  -- Процедура для получения родственников и контактных лиц клиента (с телефонами)
  res_text = (select res_style
              from fm_html_styles) || '<body>';

  for select c_name_fio, rel_name, c_plastcard_no
      from hp_client_relation_list(:rep_c_id, 0)
      where coalesce(c_plastcard_no, '') <> ''
      union
      select coalesce(cc_name, ''), coalesce(cc_status, 'Контактное лицо'), cc_cell_no
      from contacts, fm_contperson_id(:rep_c_id)
      where cc_id = list_cc_id and
            coalesce(contacts.cc_cell_no, '') <> ''
      into contact_fio, rel_name, c_cell_no
  do
  begin
    c_cell_no = replace(c_cell_no, ' ', '');
    c_cell_no = replace(c_cell_no, '+', '');
    c_cell_no = replace(c_cell_no, '-', '');
    c_cell_no = replace(c_cell_no, '(', '');
    c_cell_no = replace(c_cell_no, ')', '');

    if (c_cell_no <> '') then
    begin
      res_text = BLOB_APPEND(res_text, rel_name || ': <b>' || contact_fio || '</b>, ' || f_uri_phone(c_cell_no, 0, rep_c_id, 1) || '<br>');
    end
  end
  if (res_text <> '') then
    res_text = BLOB_APPEND('<h1 class="green_text">Контактные лица для связи:</h1>', res_text);
  else
    res_text = '<h2>Нет контактных лиц для связи.</h2>';
  res_text = BLOB_APPEND(res_text, '</body>');
  suspend;
end
