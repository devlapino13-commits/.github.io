CREATE PROCEDURE UXP$UXFRITEM_ADD_AUX(
    UX_UUID D_UUID,
    ITEM_KIND D_STR_CODE,
    ITEM_CONTROL D_STR_CODE,
    CURRENT_UFI_UUID D_UUID)
RETURNS (
    UFI_UUID_STR D_STR_CODE)
AS
declare variable ufi_position d_integer;
declare variable ufi_parent_uuid d_uuid;
declare variable ufi_kind d_integer;
declare variable ufi_uuid d_uuid;
declare variable grid_uuid d_uuid;
declare variable ufi_control d_str_code;
begin
  -- Добавляет один AUX элемент или группу
  if (item_kind = 'group' and item_control in ('grid', 'tree')) then
  begin
    item_kind = item_control;
    item_control = '';
  end

  if (trim(coalesce(current_ufi_uuid, '')) <> '') then
  begin
    select ufi_position, ufi_parent_uuid, ufi_kind, ufi_control
    from ux$fritem
    where ufi_owner = :ux_uuid and
          ufi_uuid = :current_ufi_uuid
    into ufi_position, ufi_parent_uuid, ufi_kind, ufi_control;

    if (ufi_position is not null) then
    begin
      ufi_position = ufi_position + 1;
      if (ufi_kind = -1 and ufi_control not in ('grid', 'tree')) then
      begin
        ufi_parent_uuid = current_ufi_uuid;
        ufi_position = null;
      end
    end
  end

  if (item_kind in ('grid', 'tree')) then
  begin
    grid_uuid = null;
    if (coalesce(item_control, '') <> '') then
    begin
      select ugr_uuid
      from ux$grid
      where ugr_ufr_uuid = :ux_uuid and
            ugr_uuid = char_to_uuid(:item_control)
      into grid_uuid;

      if (row_count = 0) then
        exception e_message 'Не найден набор данных: ' || item_control;
    end else
    begin
      insert into ux$grid (ugr_ufr_uuid, ugr_is_private)
      values (:ux_uuid, 0)
      returning ugr_uuid
      into :grid_uuid;
    end

    if (not exists(select * from ux$fritem
    where ufi_uuid = :grid_uuid)) then
    begin
      insert into ux$fritem (ufi_uuid, ufi_owner, ufi_control, ufi_kind, ufi_parent_uuid, ufi_position)
      values (:grid_uuid, :ux_uuid, :item_kind, -1, :ufi_parent_uuid, :ufi_position)
      returning ufi_uuid
      into ufi_uuid;
    end else
      ufi_uuid = grid_uuid;
  end else
  if (item_kind = 'group') then
  begin
    insert into ux$fritem (ufi_owner, ufi_control, ufi_kind, ufi_parent_uuid, ufi_position)
    values (:ux_uuid, :item_control, -1, :ufi_parent_uuid, :ufi_position)
    returning ufi_uuid
    into ufi_uuid;
  end else
  if (item_kind = 'aux') then
  begin
    -- Другие AUX-элементы
    insert into ux$fritem (ufi_owner, ufi_control, ufi_kind, ufi_parent_uuid, ufi_position)
    values (:ux_uuid, :item_control, 2, :ufi_parent_uuid, :ufi_position)
  returning ufi_uuid
  into ufi_uuid;
  end else
    exception e_default f_format('Unknown item_kind: %1', item_kind);

  if (ufi_uuid is not null) then
  begin
    ufi_uuid_str = uuid_to_char(ufi_uuid);
    suspend;
  end
end
