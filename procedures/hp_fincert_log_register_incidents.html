CREATE PROCEDURE HP_FINCERT_LOG_REGISTER_INCIDENTS
AS
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable c_id d_id;
declare variable res_incident d_str_255;
declare variable check_datetime d_datetime;
declare variable aim_req_datetime d_datetime;
declare variable aim_kusp_number d_str_64;
declare variable aim_tkt_number d_str_64;
declare variable aim_status d_str_code;
begin
  for select hp.c_id, hp.object_kind, hp.object_id, hp.res_incident, hp.check_datetime,
             hp.aim_kusp_number, hp.aim_tkt_number, hp.aim_req_datetime, hp.aim_req_status_code
  from aarep_antifrod_incident_list(current_date -1, current_date -1) hp
  where hp.is_sendfc = 1
  into :c_id, :obj_kind, :obj_id, :res_incident, check_datetime,
       :aim_kusp_number, :aim_tkt_number, :aim_req_datetime, :aim_status
  do
  begin
    if (not
        exists(select 1
               from fincert_history_log fhl
               where fhl.fhl_from_c_id = :c_id and
                     fhl.fhl_obj_kind = :obj_kind and
                     fhl.fhl_obj_id = :obj_id)) then
    
    insert into fincert_history_log (fhl_from_c_id, fhl_obj_kind, fhl_obj_id, fhl_status, fhl_res_incident, fhl_incident_date_time,
                                     fhl_law_req_request, fhl_law_req_number, fhl_law_req_ticket, fhl_law_req_datetime_at)
    values (:c_id, :obj_kind, :obj_id, f_enum_id('fincert_status', 'pending'), :res_incident, :check_datetime,
            :aim_status, :aim_kusp_number, :aim_tkt_number, :aim_req_datetime);
  end
end
