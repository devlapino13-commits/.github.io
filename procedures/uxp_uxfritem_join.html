CREATE PROCEDURE UXP$UXFRITEM_JOIN(
    SELECTED_UFI_LIST D_STR_8100,
    PARENT_UUID_STR D_STR_CODE)
AS
declare variable new_group d_str_100;
declare variable ufi_code_str d_str_code;
declare variable ufi_position d_integer;
declare variable new_position d_integer;
declare variable ux_uuid d_uuid;
declare variable ufi_parent_uuid d_uuid;
declare variable ufi_uuid d_uuid;
begin
  if (coalesce(selected_ufi_list, '') = '') then
    Exit;

  if (coalesce(parent_uuid_str, '') = '') then
    Exit;

  ufi_parent_uuid = char_to_uuid(parent_uuid_str);

  select ufi_owner
  from ux$fritem
  where ufi_uuid = :ufi_parent_uuid
  into ux_uuid;

  -- Проверяем что все эелементы в одной ветке
  -- Заодно проверяем минимальную позицию чтобы вставить новую группу в эту позицию
  new_position = null;
  for select trim(res_str)
  from uxp$split_str(:selected_ufi_list, ',')
  into :ufi_code_str
  do
  begin
    ufi_uuid = char_to_uuid(ufi_code_str);

    select ufi_position
    from ux$fritem
    where ufi_uuid = :ufi_uuid and
          ufi_parent_uuid = :ufi_parent_uuid
    into ufi_position;

    if (ufi_position is null) then
      exception e_message 'Не все выдаленные элементы в одной группе.';

    if (new_position is null or new_position > ufi_position) then
      new_position = ufi_position;
  end

  select ufi_uuid_str
  from uxp$uxfritem_add_aux(:ux_uuid, 'group', 'group', :ufi_uuid)
  into new_group;

  if (coalesce(new_group, '') = '') then
    exception e_message 'Не удалось создать новую группу';

  update ux$fritem
  set ufi_position = :new_position
  where ufi_uuid = char_to_uuid(:new_group);

  new_position = 0;
  for select trim(res_str)
  from uxp$split_str(:selected_ufi_list, ',')
  into :ufi_code_str
  do
  begin
    execute procedure uxp$uxfritem_move(:ufi_code_str, :new_group, :new_position);
    new_position = new_position + 1;
  end
end
