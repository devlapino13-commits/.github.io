CREATE PROCEDURE AAREP_TABEL_CHECK_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    SAT_FROM_C_ID DECIMAL(18,0),
    SAT_YEAR INTEGER,
    SAT_MONTH INTEGER,
    SAWT_ID INTEGER,
    SAWT_NAME VARCHAR(100),
    SAT_TABEL_DAYS DECIMAL(18,4),
    SAT_ORDER_DAYS DECIMAL(18,4))
AS
declare variable rep_year integer;
declare variable sawt_include_holiday integer;
declare variable d_begin date;
declare variable d_end date;
declare variable dd date;
declare variable sat_year_month integer;
begin
  sat_order_days = 0;
  rep_year = extract(year from :rep_date_end);
  for select sat_from_c_id, sat_year_month, sawt_id, sawt_name,
             sum(sat_day_1 + sat_day_2 + sat_day_3 + sat_day_4 + sat_day_5 + sat_day_6 + sat_day_7 + sat_day_8 + sat_day_9 + sat_day_10 + sat_day_11 + sat_day_12 + sat_day_13 + sat_day_14 + sat_day_15 + sat_day_16 + sat_day_17 + sat_day_18 + sat_day_19 + sat_day_20 + sat_day_21 + sat_day_22 + sat_day_23 + sat_day_24 + sat_day_25 + sat_day_26 + sat_day_27 + sat_day_28 + sat_day_29 + sat_day_30 + sat_day_31)
      from sa_tabel, sa_worktype, region_list(:rep_region_id, 1, -1, 1)
      where sat_from_sawt_id = sawt_id and
            sawt_is_order = 0 and
            sat_is_ready = 1 and
            sat_office_id = id_list and
            sat_year_month between(:rep_year * 100) and (:rep_year * 100 + 99) and
            sawt_id > 0 and
            sawt_base_type = 4 and
            sawt_hours_real > 0
      group by sat_from_c_id, sat_year_month, sawt_id, sawt_name
      into :sat_from_c_id, :sat_year_month, :sawt_id, :sawt_name, :sat_tabel_days
  do
  begin
    sat_month = mod(sat_year_month, 100);
    sat_year = (sat_year_month - sat_month) / 100;

    sat_tabel_days = sat_tabel_days / 8;
    suspend;
  end

  sat_tabel_days = 0;
  sat_order_days = 1;
  for select hrv_from_c_id, hrv_date_begin, hrv_date_end, sawt_id, sawt_name, sawt_include_holiday
      from hr_voctravel, hr_orders, sa_worktype, region_list(:rep_region_id, 1, -1, 1)
      where hrv_from_ho_id = ho_id and
            ho_order_type = sawt_id and
            hrv_status = 1 and
            sawt_is_order = 0 and
            ho_from_reg_id = id_list and
            (extract(year from hrv_date_begin) = :rep_year or
            extract(year from hrv_date_end) = :rep_year) and
            sawt_base_type = 4 and
            sawt_hours_real > 0
      into :sat_from_c_id, :d_begin, :d_end, :sawt_id, :sawt_name, :sawt_include_holiday
  do
  begin
    dd = d_begin;
    while (dd <= :d_end) do
    begin
      sat_order_days = 1;
      if (exists(select *
                 from sa_calendar
                 where scal_month = extract(month from :dd) and
                       scal_day = extract(day from :dd) and
                       scal_active = 1)) then
        sat_order_days = 0;

      sat_year = extract(year from :dd);
      if (sat_year = :rep_year) then
      begin
        sat_month = extract(month from :dd);
        if (sawt_include_holiday = 1) then
          suspend;
        else
        begin
          if (extract(weekday from :dd) > 0 and
              extract(weekday from :dd) < 6) then
            suspend;
        end
      end
      dd = dd + 1;
    end
  end
end
