CREATE PROCEDURE HP_DO_EXPRESSAPP_ACTION_APPROVE(
    EXA_ID D_INTEGER)
RETURNS (
    WAIT_STATUS D_SMALLINT)
AS
declare variable EXA_FROM_K_ID ID_TYPE;
declare variable EXA_FROM_UN_ID ID_TYPE;
declare variable K_STATUS D_SMALLINT;
declare variable EXA_BASE_K_ID ID_TYPE;
declare variable K_APPROVED_MANAGER_LAST ID_TYPE;
declare variable KC_ID ID_TYPE;
begin
  wait_status = 0;
  select exa_from_k_id, exa_from_un_id, exa_base_k_id
  from expressapp
  where exa_id = :exa_id
  into exa_from_k_id, exa_from_un_id, exa_base_k_id;

  k_status = null;
  select k_status from kontract
  where k_id =:exa_from_k_id
  into k_status;

  if (k_status is null or k_status in (-2,-1)) then
  begin
    wait_status = -1;
  end else
  if (k_status = 0) then
  begin
    if (f_get_db_id() = 81001) then
    begin
      -- Расчет ПДН
      for select kc_id
      from knclient
      where kc_from_k_id = :exa_from_k_id and
            kc_amount_approved > 0
      into kc_id
      do
        execute procedure hp_knclient_pdn_update(:kc_id, 0, null, null, null, null, null, 'expressapp_approve');
    end

    select k_approved_manager from kontract
    where k_id = :exa_base_k_id
    into k_approved_manager_last;

    update kontract
    set k_status = 1,
        k_approved_manager = :k_approved_manager_last,
        k_user_id = :exa_from_un_id
    where k_id = :exa_from_k_id;
  end
--  если k_status = 1 то уже займ утвержден то тогда тоже продолжаем
  suspend;
end
