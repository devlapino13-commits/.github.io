CREATE PROCEDURE FM_EDOC_BLANK_LIST(
    UN_ID D_ID,
    EDS_ID D_ID)
RETURNS (
    EDB_ID D_ID,
    EDB_NAME D_STR_100,
    EDB_VISITED D_BOOL)
AS
begin
  for select edb_id, edf_print_name
  from edoc_blank, edoc_frx, edoc_sign
  where eds_id = :eds_id and
        eds_from_edl_id =  edb_from_edl_id and
        edb_from_edf_id = edf_id and
        (edb_from_c_id is null or edb_from_c_id = eds_from_c_id) -- выдаем общие документы или конкретного участника
  into edb_id, edb_name
  do
  begin
    edb_visited = 1;
    if (exists(
    select * from edoc_open
    where edo_from_eds_id = :eds_id and
          edo_from_edb_id = :edb_id and
          (edo_before_sign_datetime is not null or
          edo_after_sign_datetime is not null))) then edb_visited = 0;
    suspend;
  end
end
