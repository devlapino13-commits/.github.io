CREATE PROCEDURE HP_PRINT_BIOMETRY_AFC_INFO(
    MA_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME_FIO D_STR_255,
    C_IIN D_STR_20,
    C_ADDRESS_LEGAL D_STR_255,
    C_PHONE_NUMBER D_STR_20,
    DOC_CREATION_DATE D_STR_20,
    HEADER_DATE D_STR_20,
    DOC_SOURCE_NUMBER D_ID,
    LOAN_APP_DOC_SIGN_DATE D_STR_20,
    BIOMETRY_DATE D_STR_20,
    BIOMETRY_SOURCE D_STR_255,
    BIOMETRY_RESULT D_STR_255,
    ANTIFRAUD_DATE D_STR_20,
    ANTIFRAUD_RESULT D_STR_255)
AS
declare variable c_id d_id;
declare variable edl_id d_id;
begin
    if (ma_id is null) then
        exception e_message 'Не передан ID заявки';

    if (coalesce(add_params, '') <> '') then
    begin
        edl_id = f_json_str_get_value(add_params, 'edl_id', null);

        select f_datetostr(edl.edl_create_datetime)
        from edoc_list edl
        where edl.edl_id = :edl_id
        into :doc_creation_date;

        doc_source_number = edl_id;
        header_date = doc_creation_date;
    end

    select mac.mac_from_c_id
    from miniappclient mac
    where mac.mac_from_ma_id = :ma_id and
          mac.mac_amount > 0
    into :c_id;

    select c.c_name_fio, c.c_inn
    from clients c
    where c.c_id = :c_id
    into :c_name_fio, :c_iin;
    
    select first 1 adl.adl_address_str
    from addrclient adc
    inner join addrlist adl on adc.adc_from_adl_id = adl.adl_id
    where adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 1 --Юридический адрес
    order by adc.adc_date desc
    into :c_address_legal;

    c_phone_number = f_cont(:c_id, 'cell', 'ok', 'main');

    -- Получаем дату подписания заявки на получение микрокредита
    select first 1 f_datetostr(eds.eds_status_datetime)
    from edoc_list edl
    inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
    inner join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id
    where
        edc.edc_code = 'loan.application' and
        edl.edl_object_kind = f_objkind('miniapp', 'id') and
        edl.edl_object_id = :ma_id and
        eds.eds_from_c_id = :c_id and
        eds.eds_status = 1
    order by eds.eds_status_datetime desc
    into :loan_app_doc_sign_date;

    -- Получаем информацию по прохождению биометрии (имеется в виду сервис через который подписываются документы)
    biometry_date = loan_app_doc_sign_date; -- Считаем, что дата прохождения биометрии такая же, как и дата подписания документа
    biometry_source = 'через сервис ТОО "ПКБ"';
    biometry_result = 'подтверждена'; -- Если клиент подписал документ (а иначе эта процедура не вызовется), то он прошел биометрию

    select f_datetostr(antifraud_check_date), antifraud_check_result
    from hp_antifraud_print_result(f_objkind('miniapp', 'id'), :ma_id)
    into :antifraud_date, :antifraud_result;

    antifraud_date = coalesce(antifraud_date, '');
    antifraud_result = coalesce(antifraud_result, '');

    suspend;
end
