CREATE PROCEDURE SUBKONTO_LIST(
    ID INTEGER,
    SELF_INCLUDE INTEGER,
    IS_DOWN INTEGER)
RETURNS (
    ID_LIST INTEGER,
    ID_SUB_PARENT INTEGER,
    SUB_PARENT_NAME VARCHAR(50))
AS
begin
  for with recursive mc_list
      as (select sub_id, sub_parent
          from subkonto
          where sub_id = :id
          union all
          select s.sub_id, s.sub_parent
          from subkonto s
          join mc_list on ((:is_down = 1 and
                mc_list.sub_id = s.sub_parent) or (:is_down = 0 and
                mc_list.sub_parent = s.sub_id)))
      select sub_id, sub_parent
      from mc_list
      where ((:self_include = 0 and
            sub_id <> :id) or
            (:self_include = 1))
      into :id_list, :id_sub_parent
  do
  begin
    if (id_sub_parent is not null) then
    begin
      select distinct sub_name
      from subkonto
      where sub_id = :id_sub_parent rows 1
      into :sub_parent_name;
    end
    else
    begin
      id_sub_parent = 0;
      sub_parent_name = '';
    end
    suspend;
  end

end
