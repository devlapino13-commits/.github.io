CREATE PROCEDURE HP_GET_VIGOVOR_KOEF(
    REP_C_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    BONUS_KOEF DECIMAL(18,4),
    BONUS_SUM DECIMAL(18,4),
    BONUS_MEMO VARCHAR(255))
AS
declare variable rb_koef decimal(18,4);
declare variable rb_type integer;
declare variable rb_number varchar(20);
declare variable kol integer;
begin
  bonus_koef = 100;
  bonus_sum = 0;
  bonus_memo = '';
  kol = 0;
  for select rb_bonus_koef, rb_type, rb_number
      from rebuke
      where rb_from_c_id = :rep_c_id and
            rb_bonus_change = 1 and
            rb_period_start <= :rep_date and
            rb_period_end >= :rep_date and
            rb_is_delete = 0
      into :rb_koef, :rb_type, :rb_number
  do
  begin
    if (rb_koef <= 1) then
    begin
      bonus_koef = bonus_koef - rb_koef * 100;
    end
    if (rb_koef > 1) then
    begin
      bonus_sum = bonus_sum - rb_koef;
    end
    if (rb_type = 1) then
    begin
      if (bonus_memo = '') then
        bonus_memo = 'Выговор: ' || :rb_number;
      else
        bonus_memo = bonus_memo || ', выговор: ' || :rb_number;
    end
    if (rb_type = 2) then
    begin
      if (bonus_memo = '') then
        bonus_memo = 'Замечание: ' || :rb_number;
      else
        bonus_memo = bonus_memo || ', замечание: ' || :rb_number;
    end
    if (rb_type = 3) then
    begin
      if (bonus_memo = '') then
        bonus_memo = 'Устное замечание: ' || :rb_number;
      else
        bonus_memo = bonus_memo || ', устное замечание: ' || :rb_number;
    end
    kol = kol + 1;
    suspend;
  end
  if (kol = 0) then
  begin
    bonus_memo = 'Нет замечаний и выговоров';
    suspend;
  end
end
