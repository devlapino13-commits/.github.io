CREATE PROCEDURE CRIF_GET_KONTRACT_DATA_TMP(
    POCKET_ID INTEGER)
RETURNS (
    K_ID ID_TYPE,
    K_KONTRACT_NO D_STR_CODE,
    ORDER_MARK D_SMALLINT,
    PHASEOFCONTRACT D_STR_10,
    DAYS_PAST_DUE_DOMAIN D_STR_10,
    DAYS_PAST_DUE_DOMAIN_PREV D_STR_10,
    LOCAL_STATUS D_STR_CODE)
AS
declare variable K_ID_OLD ID_TYPE;
declare variable REP_DATE_BEGIN D_DATE;
declare variable REP_DATE_END D_DATE;
declare variable POCKET_TYPE D_STR_10;
declare variable DO_SUSPEND D_BOOL;
declare variable DISB_DATE D_DATE;
declare variable DATE_TYPE D_STR_20;
declare variable DAYS_PAST_DUE_DOMAIN_TMP D_STR_10;
begin
  k_id_old = 0;

  select cp.cp_date_begin, --
         cp.cp_date_end, --
         cp.cp_pocket_type
  from cib_pocket cp
  where cp.cp_id = :pocket_id
  into :rep_date_begin,
       :rep_date_end,
       :pocket_type;

  if ((rep_date_end is null) or (pocket_type is null)) then
    exit;

  for select ks_from_k_id,
             order_mark,
             phaseofcontract
      from (
            -- получаю список закрытых займов
            select ks_from_k_id,
                   order_mark, 'CL' as phaseofcontract
            from (select ks.ks_from_k_id,
                         0 as order_mark -- <-- важно!
                  from kontstate ks
                  where ks.ks_state = 2 and
                        ks.ks_date_begin between :rep_date_begin and :rep_date_end and
                        ks.ks_ol_is_delete = 0) closedkont
            union
            -- получаю список активных займов
            select ks_from_k_id,
                   order_mark,
                   'AC' as phaseofcontract
            from (select ks.ks_from_k_id, 2 as order_mark
                  from kontstate ks
                  where ks.ks_state = 1 and
                        :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
                        ks.ks_ol_is_delete = 0) activekont
            ---------------- Актуализаци информации в CC ----------------- >
            -- > Активные в КИБ, но закрытые в МБулак +
            -- > Активные в МБулак, которые в КИБ не были отпралены
            union
            select ks_from_k_id,
                   order_mark,
                   phaseofcontract
            from (select ks.ks_from_k_id,
                         0 as order_mark,
                         iif(ks.ks_state = 1, 'AC', 'CL') as phaseofcontract
                  from cib_actuality cke
                  inner join kontstate ks  on ks.ks_from_k_id = cke.k_id
                  where :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
                        ks.ks_ol_is_delete = 0) CheckedKont
      ) total
      order by ks_from_k_id,
               order_mark
      into :k_id,
           :order_mark,
           :phaseofcontract
  do
  begin
    if (k_id <> k_id_old) then
    begin
      k_id_old = k_id;



      -- если крежит в игнор листе, его не отправляем
      if (exists(select * from kib_ignore where ki_from_k_id = :k_id)) then
        continue;

      select kont.k_kontract_no,
             kont.k_disb_date
      from kontract kont
      where kont.k_id = :k_id
      into :k_kontract_no,
           :disb_date;

      do_suspend = 0;
    
      if (phaseofcontract = 'CL') then  -- если крежит был закрыт отпраляем его в любом случае
      begin
        days_past_due_domain = '1';
        local_status = 'closed';
        do_suspend = 1;
      end
      else -- если это активный заем
      begin
           days_past_due_domain = '';
        
           -- если у кредита изменился статус просрочки по отновшению к предыдущему дню, отправляем его
           for select date_type,
                      days_past_due_domain
           from crif_get_loan_categ(:k_id, :rep_date_end)
           into :date_type,
                :days_past_due_domain_tmp
           do
           begin
               if (date_type = 'rep_date') then
                   days_past_due_domain = days_past_due_domain_tmp;
               else
                   days_past_due_domain_prev = days_past_due_domain_tmp;
           end
           if (days_past_due_domain <> days_past_due_domain_prev) then
           begin
               local_status = 'chacnged_delinq_status';
               do_suspend = 1;
           end
           -- < ---------
           -- если это выдача займа
           if ((do_suspend = 0) and (disb_date between rep_date_begin and rep_date_end)) then
           begin
               local_status = 'disb';
               do_suspend = 1;
           end
           -- если на период пакета по кредиту была оплата, добавляем кредит в пакет
           if (do_suspend = 0) then
           begin
               if (exists(select *
                       from operkred ok
                       where ok.ok_type in (2, 4, 9) and
                             ok.ok_ol_is_active = 1 and
                             ok.ok_from_k_id = :k_id and
                             ok.ok_ol_date between :rep_date_begin and :rep_date_end)) then
               begin
                   local_status = 'payed';
                   do_suspend = 1;
               end
           end
           -- < --
      end

      if (do_suspend = 1) then
           suspend;
    end
  end
end
