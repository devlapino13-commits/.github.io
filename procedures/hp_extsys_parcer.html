CREATE PROCEDURE HP_EXTSYS_PARCER(
    PARAMS VARCHAR(8000),
    KEY_CODE VARCHAR(1000))
RETURNS (
    FIELD_CODE VARCHAR(1000),
    FIELD_VALUE VARCHAR(1000))
AS
declare variable lastpos integer;
declare variable nextpos integer;
declare variable pair varchar(1000);
declare variable t integer;
begin
-- Разбор и сохранение значений полей, поля при необходимости создаются
  if (key_code is null) then
    key_code = '';
  params = params || '&';

  if (key_code <> '') then
  begin
    field_code = key_code;
    field_value = '';
    lastpos = 0;
    nextpos = 0;
    lastpos = position(key_code || '=', params, 1);
    if (lastpos > 0) then
    begin
      lastpos = position('=', params, lastpos) + 1;
      nextpos = position('&', params, lastpos);
      if (nextpos > 0) then
        field_value = substring(params from lastpos for nextpos - lastpos);
    end
    suspend;
  end
  else
  begin
    lastpos = 1;
    nextpos = position('&', params, lastpos);
    while (nextpos > 1) do
    begin
      field_value = '';
      field_code = '';
      pair = substring(params from lastpos for nextpos - lastpos);
      t = position('=', pair);
      if (t > 0) then -- Разбор пары имя=значение
      begin
        field_code = left(pair, t - 1);
        field_value = substring(pair from t + 1);
        if (field_code <> '' or field_value <> '') then
          suspend;
      end
      lastpos = nextpos + 1;
      nextpos = position('&', params, lastpos);
    end
  end
end
