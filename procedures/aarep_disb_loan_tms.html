CREATE PROCEDURE AAREP_DISB_LOAN_TMS(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_OFFICE_NAME D_STR_100,
    RES_LO_FIO D_STR_100,
    RES_DISB_COUNT D_INTEGER,
    RES_DISB_AMOUNT D_CURRENCY,
    RES_CELL_NO D_STR_CODE)
AS
declare variable reg_id d_integer;
declare variable c_id d_id;
begin
  for select ks_region_id, ks_kc_id, count(*), sum(k_amount_approved)
  from kontstate, kontract
  where ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_date_begin between :rep_date_begin and :rep_date_end and
        ks_from_k_id = k_id and
        k_from_pr_id in (520, 521)
  group by 1, 2
  into reg_id, c_id, res_disb_count, res_disb_amount
  do
  begin
    res_cell_no = f_cont(c_id, 'cell', 'ok', 'main');

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into res_office_name;

    select c_name_fio
    from clients
    where c_id = :c_id
    into res_lo_fio;

    suspend;
  end
end
