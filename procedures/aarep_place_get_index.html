CREATE PROCEDURE AAREP_PLACE_GET_INDEX(
    PL_ID DECIMAL(18,0))
RETURNS (
    ADL_POST_INDEX INTEGER)
AS
declare variable adl_id decimal(18,0);
begin
  adl_post_index = 0;
  select pl_post_index
  from places
  where pl_id = :pl_id and
        pl_post_index is not null and
        pl_post_index <> ''
  into :adl_post_index;
  if (adl_post_index is null) then
    adl_post_index = 0;
  if (adl_post_index = 0) then
  begin
    adl_post_index = 0;
    select adl_id, adl_post_index
    from addrlist
    where adl_street_id = :pl_id and
          adl_post_index is not null and
          adl_post_index > 0
    order by adl_id desc
    rows 1
    into :adl_id, :adl_post_index;
    if (adl_post_index is null) then
      adl_post_index = 0;
  end
  suspend;
end
