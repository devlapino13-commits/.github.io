CREATE PROCEDURE HP_SUD_GET_REQUEST_STATUS(
    SLC_ID D_ID,
    KIND D_SMALLINT)
RETURNS (
    EIL_STATUS D_INTEGER,
    STATUS_STR D_STR_100,
    BLS_ID D_ID)
AS
begin
  select iif((sue_date + eic_info_actual_days) < current_date, -99, eil_status), eil_from_bls_id
  from sud_extinfo, extinfolist, extinfocfg
  where eil_id = sue_from_eil_id
    and eic_id = eil_from_eic_id
    and sue_slc_id = :slc_id
    and sue_kind = :kind
  order by sue_date desc, sue_id desc
  rows 1
  into eil_status, bls_id;

  if (eil_status is null) then
    exit;

  status_str = case eil_status
                 when -99 then 'Запрос устарел'
                 when -3 then 'Ошибка запроса подсудности'
                 when -2 then 'Ошибка запроса подсудности'
                 when -1 then 'Запрос создан'
                 when 0  then 'Запрос создан'
                 when 1  then 'Запрос отправлен. Ожидание ответа'
                 when 2  then 'Ответ получен'
                 else eil_status
               end;

  suspend;
end
