CREATE PROCEDURE HP_PHOTO_ON_DATE_DISB_AUTO_ADD(
    C_ID D_ID,
    BLS_ID D_ID,
    UN_ID D_ID)
AS
declare variable PK_ID D_INTEGER;
declare variable CDS_ID D_INTEGER;
begin
    /*
        Для КГ, если делается фото клиента, это же самое фото должно стать фото на день выдачи.
        Цель: чтобы ВИПС не приходилось делать 2-а фото по одному и тому же клиенту
    */
    if(f_get_db_id() not in (39801, 41701)) then exit;

    select pask.pk_id
    from passkind pask
    inner join passdoctype pdt on pdt.pdt_id = pask.pk_from_pdt_id
    where pdt.pdt_code = 'foto_on_date_disb'
    into :pk_id;

    for select cds.cds_id
         from clientdocs cds
         where cds.cds_from_pk_id = :pk_id
         and cds.cds_from_c_id = :c_id
         and cds.cds_doc_date_begin = current_date
         and cds.cds_is_delete = 0
    into :cds_id
    do
    begin
        update clientdocs
        set cds_doc_blob_1 = :bls_id,
            cds_from_un_id = :un_id,
            cds_datetime = current_timestamp,
            cds_create_datetime = current_timestamp,
            cds_create_un_id = :un_id
        where cds_id = :cds_id;
    end
    if (cds_id is null) then
    begin
        insert into clientdocs (cds_from_c_id, cds_from_pk_id, cds_doc_blob_1, cds_from_un_id, cds_verify_status, cds_create_un_id)
        values (:c_id, :pk_id, :bls_id, :un_id, 2, :un_id);
    end
end
