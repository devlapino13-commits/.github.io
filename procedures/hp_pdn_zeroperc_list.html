CREATE PROCEDURE HP_PDN_ZEROPERC_LIST(
    DISB_PERIOD D_INTEGER,
    PCG_ID D_INTEGER,
    SHOW_LAST_VALUE D_SMALLINT,
    SHOW_ALL_GROUP D_SMALLINT)
RETURNS (
    PZP_ID D_INTEGER,
    PZP_DISB_PERIOD D_INTEGER,
    PZP_DATE_BEGIN D_DATE,
    PZP_FROM_PCG_ID D_INTEGER,
    PZP_DAY_MIN D_INTEGER,
    PZP_DAY_MAX D_INTEGER,
    PZP_AMOUNT_MIN D_CURRENCY,
    PZP_AMOUNT_MAX D_CURRENCY,
    PZP_ZALOG_KIND D_SMALLINT,
    PZP_PORUCH_KIND D_SMALLINT,
    PZP_LINE_ID D_INTEGER,
    PZP_PERCENT D_PERCENT,
    PCG_NAME D_STR_100,
    PZP_STR_PERIOD D_STR_CODE,
    PZP_CHECK_STATE D_SMALLINT,
    PCG_GROUP_TYPE D_STR_8100)
AS
declare variable pcg_id_tmp d_integer;
declare variable tmp_period d_integer;
begin
  if (show_last_value is null) then
    show_last_value = 0;

  if (show_all_group is null) then
    show_all_group = 0;

  for select pcg_id, pcg_name, pcg_group_type
  from hp_pdn_credgroup_list
  where (:show_all_group = 1 or pcg_id = :pcg_id)
  order by pcg_id
  into :pcg_id_tmp, :pcg_name, :pcg_group_type
  do
  begin
    tmp_period = disb_period;
    if (show_last_value = 1) then
    begin
      select pzp_disb_period
      from pdn_zeroperc
      where pzp_from_pcg_id = :pcg_id_tmp and
            pzp_disb_period <= :disb_period
      order by pzp_disb_period desc
      rows 1
      into tmp_period;
    end

    for select pzp_id, pzp_disb_period, pzp_date_begin, pzp_from_pcg_id, pzp_day_min, pzp_day_max, pzp_amount_min, pzp_amount_max,
               pzp_zalog_kind, pzp_poruch_kind, pzp_line_id, pzp_percent, pzp_check_state
    from pdn_zeroperc
    where pzp_from_pcg_id = :pcg_id_tmp and
          pzp_disb_period = :tmp_period
    order by pzp_from_pcg_id, pzp_disb_period, pzp_day_min, pzp_amount_min, pzp_zalog_kind, pzp_poruch_kind, pzp_id
    into :pzp_id, :pzp_disb_period, :pzp_date_begin, :pzp_from_pcg_id, :pzp_day_min, :pzp_day_max, :pzp_amount_min, :pzp_amount_max,
         :pzp_zalog_kind, :pzp_poruch_kind, :pzp_line_id, :pzp_percent, :pzp_check_state
    do
    begin
      pzp_str_period = left(pzp_disb_period, 4) || '-' || right(pzp_disb_period, 2);
      suspend;
    end
  end
end
