            from (select distinct '1_operkred' as character_new, kc_number, kc_id, k_amount_approved,

                                  c_inn, iif(kc_non_resident = 1, 0, 1) as kc_non_resident
                  from operkred, knclient, clients, kontract
                  where ((ok_from_ol_id = :ol and
                        cast(:ol_id_other as decimal(18,0)) = 0) or
                        (ok_from_ol_id = :ol_id_other)) and
                        ok_from_k_id = kc_from_k_id and
                        kc_amount_approved > 0 and
                        kc_from_c_id = c_id and
                        k_id = kc_from_k_id and
                        ok_from_c_id = 0
                  union all
                  select distinct '2_operkroff' as character_new, kc_number, kc_id, k_amount_approved,
                                  kc_amount_approved, c_name_fio, c_pass_no, c_pass_date, c_pass_who, c_reg_no, c_okpo,
                                  c_inn, iif(kc_non_resident = 1, 0, 1) as kc_non_resident
                  from operkroff okf, knclient, clients, kontract
                  where ((okf.oko_from_ol_id = :ol and
                        cast(:ol_id_other as decimal(18,0)) = 0) or
                        (okf.oko_from_ol_id = :ol_id_other)) and

                                  kc_amount_approved, c_name_fio, c_pass_no, c_pass_date, c_pass_who, c_reg_no, c_okpo,
                                  c_inn, iif(kc_non_resident = 1, 0, 1) as kc_non_resident
                  from operextra oe, knclient, clients, kontract
                  where ((oe.ox_from_ol_id = :ol and
                        cast(:ol_id_other as decimal(18,0)) = 0) or
                        (oe.ox_from_ol_id = :ol_id_other)) and
                        oe.ox_object_kind = 9 and
                        oe.ox_object_id = kc_from_k_id and
                        kc_amount_approved > 0 and
                        kc_from_c_id = c_id and
                        k_id = kc_from_k_id)

            if (bl_corr_acc3_tmp = '91.01') then
              bl_summa3 = okf_summ;
            if (bl_corr_acc4_tmp = '91.01') then
              bl_summa4 = okf_summ;
          end

              bl_client_name = bl_doc_client_name || ' за ' || bl_client_name;
            sub_no = sub_no + 1;
            if (sub_no < last_line) then
            begin
              if (character_new starting('1_operkred')) then
              begin
                bl_summa1 = cast(bl_summa1_total as double precision) / k_amount;
                bl_summa1 = round(bl_summa1 * kc_amount);
                bl_summa2 = cast(bl_summa2_total as double precision) / k_amount;
                bl_summa2 = round(bl_summa2 * kc_amount);
                bl_summa3 = cast(bl_summa3_total as double precision) / k_amount;
                bl_summa3 = round(bl_summa3 * kc_amount);
                bl_summa4 = cast(bl_summa4_total as double precision) / k_amount;
                bl_summa4 = round(bl_summa4 * kc_amount);
              end
              bl_summa1_tmp = bl_summa1_tmp + bl_summa1;

              if ((bl_summa2_tmp + bl_summa2) > bl_summa2_total) then
              begin
                bl_summa2 = bl_summa2 - ((bl_summa2_tmp + bl_summa2) - bl_summa2_total);
              end
              bl_summa2_tmp = bl_summa2_tmp + bl_summa2;

              bl_summa3_tmp = bl_summa3_tmp + bl_summa3;
              bl_summa4_tmp = bl_summa4_tmp + bl_summa4;
            end
            else
            begin
              bl_summa1 = bl_summa1_total - bl_summa1_tmp;
              bl_summa2 = bl_summa2_total - bl_summa2_tmp;
              bl_summa3 = bl_summa3_total - bl_summa3_tmp;
              bl_summa4 = bl_summa4_total - bl_summa4_tmp;
            end

            bl_doc_no_str = reg_code || '-' || right('000000' || bl_doc_no, 6) || '/' || sub_no;

            if ((cm_database_id = 81001) and
                (ol_date >= '01.01.2018')) then
            begin
              bl_corr_acc1 = bl_corr_acc1_tmp;
              bl_corr_acc2 = bl_corr_acc2_tmp;
              bl_corr_acc3 = bl_corr_acc3_tmp;
              bl_corr_acc4 = bl_corr_acc4_tmp;
              bl_account = bl_account_tmp;

              if (bl_account = '50.01') then
                bl_account = '20202';
              if (bl_account = '51.01') then
                bl_account = '20501';
              if (bl_account starting '70.') then
                bl_account = '60306';

              c_is_resident = coalesce(c_is_resident, 0);

-- when '51.01' then '20501'
              bl_corr_acc1 = (case bl_corr_acc1

                                when '76.02' then '60323'
                                when '76.06' then '47422'

                                else bl_corr_acc1
                              end);

                                when '91.01' then '71801'
                                when '51.01' then '20501'
                                else bl_corr_acc2
                              end);

              bl_corr_acc3 = (case bl_corr_acc3

                                when '76.02' then '60323'
                                when '76.06' then '47422'
                                when '91.01' then '71801'
                                when '51.01' then '20501'
                                else bl_corr_acc3
                              end);

              bl_corr_acc4 = (case bl_corr_acc4
                                when '50.01' then '20202'
                                when '76.02' then '60323'
                                when '76.06' then '47422'
                                when '91.01' then '71801'
                                when '51.01' then '20501'
                                else bl_corr_acc4
                              end);
              if (c_is_resident = 1) then
              begin
                bl_corr_acc1 = (case bl_corr_acc1
                                  when '58.03' then '48801'
                                  when '76.09' then '48809'
                                  when '98.01' then '48809'
                                  when '51.01' then '20501'
                                  else bl_corr_acc1
                                end);

                bl_corr_acc2 = (case bl_corr_acc2
                                  when '58.03' then '48801'
                                  when '76.09' then '48809'
                                  when '98.01' then '48809'
                                  when '51.01' then '20501'
                                  else bl_corr_acc2
                                end);
