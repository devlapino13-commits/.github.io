CREATE PROCEDURE AAREP_DAILY_REPORT(
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    KOL_DISB_CONT_DAY INTEGER,
    KOL_DISB_CLI_DAY INTEGER,
    SUM_DISB_DAY CURR_TYPE,
    KOL_DISB_CONT_MONTH INTEGER,
    KOL_DISB_CLI_MONTH INTEGER,
    SUM_DISB_MONTH CURR_TYPE,
    KOL_CLI INTEGER,
    OKS_BALANCE CURR_TYPE,
    DELINQ_BALANCE1 CURR_TYPE,
    DELINQ_BALANCE7 CURR_TYPE,
    DELINQ_BALANCE30 CURR_TYPE,
    PAR1 DECIMAL(15,2),
    PAR7 DECIMAL(15,2),
    PAR30 DECIMAL(15,2))
AS
declare variable REP_DATE_BEGIN date;
declare variable REG_ID integer;
begin
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;
  reg_id = 1270100;
  for select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      union
      select 'Kursk'
      from rdb$database
      into reg_name
  do
  begin

    kol_disb_cont_day = null;
    kol_disb_cli_day = null;
    sum_disb_day = null;
    kol_disb_cont_day = null;
    kol_disb_cli_day = null;
    sum_disb_day = null;
--  выданные кредиты
    select
--         выданные на дату
           sum(iif(ks_date_begin = :rep_date_end, 1, 0)), -- кол-во выданных договоров
           sum(iif(ks_date_begin = :rep_date_end, k_kolich_client_approved, 0)), -- кол-во выданных договоров
           sum(iif(ks_date_begin = :rep_date_end, k_amount_approved, 0)), -- —умма выдачи
--         выданные с начало мес¤ца
           count(*), -- кол-во выданных договоров
           sum(k_kolich_client_approved), -- кол-во выданных договоров
           sum(k_amount_approved) -- —умма выдачи
    from kontstate, kontract
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_from_k_id = k_id and
          ks_region_id = :reg_id and
          (:reg_name <> 'Kursk' or
          :reg_name = k_from_pt_id)
    into kol_disb_cont_day, kol_disb_cli_day, sum_disb_day, kol_disb_cont_month, kol_disb_cli_month, sum_disb_month;
    kol_disb_cont_day = coalesce(kol_disb_cont_day, 0);
    kol_disb_cli_day = coalesce(kol_disb_cli_day, 0);
    sum_disb_day = coalesce(sum_disb_day, 0);
    kol_disb_cont_month = coalesce(kol_disb_cont_month, 0);
    kol_disb_cli_month = coalesce(kol_disb_cli_month, 0);
    sum_disb_month = coalesce(sum_disb_month, 0);

--  портфель и баланс просрочки
    kol_cli = null;
    oks_balance = null;
    delinq_balance1 = null;
    delinq_balance7 = null;
    delinq_balance30 = null;
    select sum(k_kolich_client_approved), sum(oks_balance),
           sum(iif(:rep_date_end - oks_delinq_date >= 0, oks_balance, 0)),
           sum(iif(:rep_date_end - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(:rep_date_end - oks_delinq_date > 30, oks_balance, 0))
    from oksummary, kontract
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_from_k_id = k_id and
          oks_region_id = :reg_id and
          (:reg_name <> 'Kursk' or
          :reg_name = k_from_pt_id) and
          :rep_date_end - oks_delinq_date <= 360
          and oks_close_state = 0 
    into kol_cli, oks_balance, delinq_balance1, delinq_balance7, delinq_balance30;

    kol_cli = coalesce(kol_cli, 0);
    oks_balance = coalesce(oks_balance, 0);
    delinq_balance1 = coalesce(delinq_balance1, 0);
    delinq_balance7 = coalesce(delinq_balance7, 0);
    delinq_balance30 = coalesce(delinq_balance30, 0);

    par1 = 0;
    par7 = 0;
    par30 = 0;
    if (oks_balance > 0) then
    begin
      par1 = cast(delinq_balance1 as double precision) / oks_balance * 100;
      par7 = cast(delinq_balance7 as double precision) / oks_balance * 100;
      par30 = cast(delinq_balance30 as double precision) / oks_balance * 100;
    end
    suspend;
  end
end
