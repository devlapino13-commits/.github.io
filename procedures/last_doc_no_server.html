CREATE PROCEDURE LAST_DOC_NO_SERVER(
    DD D_DATE,
    KIND D_INTEGER,
    REG_ID D_INTEGER,
    OL_ID D_ID,
    OPC_ID D_INTEGER = null)
RETURNS (
    NUM D_INTEGER)
AS
declare variable dn_code varchar(255);
declare variable dn_beg_type integer;
declare variable dn_region_kind integer;
declare variable max_n integer;
declare variable dn_id integer;
declare variable start_date d_date;
begin
  /* Procedure Text */
  select dn_admin_opertypes, dn_begin_type, dn_region_kind
  from docname
  where dn_id = :kind
  into :dn_code, :dn_beg_type, dn_region_kind;

  if (dn_region_kind is null) then
    dn_region_kind = 0;

  if (dn_region_kind <> 0) then
    exception e_default 'dn_region_kind always must be 0';

  if (ol_id is null) then
    ol_id = 0;

  if (opc_id is null) then
    opc_id = 0;

  start_date = dd;

-- 0 - нумерация не обнуляется
-- 1 - Обнуляется в начале года
-- 2 - обнуляется в начале месяца
-- 3 - обнуляется каждый день

  if (dn_beg_type = 0) then
    start_date = cast('01.01.1990' as date);
  else
  if (dn_beg_type = 1) then
  begin
    start_date = cast('01.01.' || extract(year from dd) as date);
    dd = cast('31.12.' || extract(year from dd) as date);
  end
  else
  if (dn_beg_type = 2) then
  begin
    start_date = dd - extract(day from dd) + 1;
    dd = dateadd(month, 1, start_date) - 1;
  end

  num = 0;

  if (dn_code is null) then
    dn_code = '';

  if (dn_code <> '') then
  begin
    for select dn_id
        from docname
        where dn_admin_opertypes = :dn_code
        into :dn_id
    do
    begin
      max_n = 0;

      if (dn_beg_type = 0) then
      begin
        select ol_doc_no
        from operlist
        where is_delete = 0 and
              ol_doc_kind = :dn_id and
              ol_doc_no is not null and
              ol_from_reg_id = :reg_id and
              ol_id <> :ol_id
        order by ol_from_reg_id desc, ol_doc_kind desc, ol_doc_no desc
        rows 1
        into :max_n;
      end else
      begin
        select ol_doc_no
        from operlist
        where is_delete = 0 and
              ol_doc_kind = :dn_id and
              ol_doc_no is not null and
              ol_from_reg_id = :reg_id and
              ol_id <> :ol_id and
              ol_date between :start_date and :dd
        order by ol_from_reg_id desc, ol_doc_kind desc, ol_doc_no desc
        rows 1
        into :max_n;
      end

      if (max_n > num) then
        num = max_n;
    end
  end else
    num = null;

  max_n = 0;
  if (kind in (0, 1)) then
  begin
    if (kind = 0) then
      dn_id = -1;
    else
      dn_id = 1;

    if (dn_beg_type = 0) then
    begin
      select opc_doc_no
      from opercash
      where opc_is_delete = 0 and
            opc_sign = :dn_id and
            opc_doc_no is not null and
            opc_office_id = :reg_id and
            opc_id <> :opc_id
      order by opc_office_id desc, opc_sign desc, opc_doc_no desc
      rows 1
      into :max_n;
    end else
    begin
      select opc_doc_no
      from opercash
      where opc_is_delete = 0 and
            opc_sign = :dn_id and
            opc_doc_no is not null and
            opc_office_id = :reg_id and
            opc_id <> :opc_id and
            opc_date between :start_date and :dd
      order by opc_office_id desc, opc_sign desc, opc_doc_no desc
      rows 1
      into :max_n;
    end

    if (max_n > num) then
      num = max_n;
  end

  if (num is not null) then
    num = num + 1;

  suspend;
end
