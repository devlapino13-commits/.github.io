CREATE PROCEDURE HP_EDOC_CONF_BLANK_LIST(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    BLN_ID D_INTEGER,
    BLN_NAME D_STR_255)
AS
declare variable obj_kind_code d_str_code;
declare variable ho_order_type d_integer;
declare variable doc_date d_date;
begin
  if (coalesce(obj_kind, '') = '') then
    obj_kind = 'hr_orders';

  obj_kind = f_objkind(obj_kind, 'id');
  obj_kind_code = f_objkind(obj_kind, 'code');

  if (obj_kind_code not in ('hr_orders', 'servicedog')) then
    exception e_default 'Incorrect object_kind: ' || coalesce(obj_kind, '<Null>');

  if (obj_kind_code = 'hr_orders') then
  begin
    select ho_order_type, ho_date
    from hr_orders
    where ho_id = :obj_id
    into ho_order_type, doc_date;

    if (row_count = 0) then
      Exit;
  end else
    ho_order_type = -1;

  for select distinct bln_id, bln_name
  from sa_blank, blankname, blankblob
  where sab_from_sawt_id = :ho_order_type and
        sab_from_bln_id = bln_id and
        blb_from_bln_id = bln_id and
        blb_frx_blob is not null and
        blb_date_begin <= :doc_date and
        blb_date_end >= :doc_date
  into bln_id, bln_name
  do
    suspend;
end
