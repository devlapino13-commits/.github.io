CREATE PROCEDURE FM_DPD_ANKETA_ACTION(
    UN_ID D_ID,
    ACTION_KIND D_STR_CODE,
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    ACTION_DATA D_STR_8100)
RETURNS (
    RES_CODE D_SMALLINT,
    RES_TEXT D_BLOB_TEXT)
AS
declare variable jso1 d_integer;  
declare variable jso2 d_integer;
declare variable jsa1 d_integer;
declare variable jso_tmp d_integer;
declare variable survey_json d_str_8100;
declare variable iterator d_integer;
declare variable monthly_count d_smallint;
declare variable monthly_info d_str_long;
declare variable income d_currency;
declare variable kc_id d_id;
declare variable kc_date d_date;
declare variable last_date d_date;
declare variable dtjob d_str_20;
declare variable dtlastm_idx d_integer;
declare variable first_date d_date;
declare variable i d_integer = 0;
declare variable found boolean = false;
declare variable cur_date d_date;
begin
    res_code = 1;
    res_text = '';
    survey_json = '{}';

    if (:obj_kind not in ('miniappclient_income')) then
    begin
        res_code = -1;
        res_text = 'Неверный obj_kind';

        suspend;
        exit;
    end

    if (action_kind = 'pdn_info') then
    begin
        select res_code, res_text
        from pdn_get_min_sufficient_income(:obj_kind, :obj_id)
        into res_code, res_text;

        suspend;
        exit;
    end

    if (action_kind = 'booklet') then
    begin
        select res_code, res_text
        from pdn_verified_avg_monthly_income(:obj_kind, :obj_id)
        into res_code, res_text;

        suspend;
        exit;
    end 

    if (action_kind = 'employer_info') then
    begin
        select kc_id,
               survey_json
        from pdn_info_action(:obj_kind, :obj_id)
        into :kc_id,
             :survey_json;
       
        select kc_date
        from hp_pdn_getinfo_add(:kc_id)
        into :kc_date;

        if (kc_date is null) then
        begin
            res_code = -1;
            res_text = 'Ошибка получения данных клиента';
            suspend;
            exit;
        end

        jso2 = json.parse(:survey_json);

        jso1 = json.obj_create();
        jsa1 = json.arr_create();

        json.obj_set_arr(jso1, 'gdoc_list', jsa1);

        jso_tmp = json.obj_create(); 
        json.arr_set_obj(jsa1, -1, jso_tmp); 
        json.obj_set_i(jso_tmp, 'id', 0); 
        json.obj_set_s(jso_tmp, 'name', 'Справка(и) о ЗП по форме по КНД 1175018 (бывшая 2-НДФЛ)');
    
        jso_tmp = json.obj_create(); 
        json.arr_set_obj(jsa1, -1, jso_tmp); 
        json.obj_set_i(jso_tmp, 'id', 1); 
        json.obj_set_s(jso_tmp, 'name', 'Справка(и) о ЗП по форме работодателя (в т.ч. ИП)');
    
        jso_tmp = json.obj_create(); 
        json.arr_set_obj(jsa1, -1, jso_tmp); 
        json.obj_set_i(jso_tmp, 'id', 2); 
        json.obj_set_s(jso_tmp, 'name', 'Банковская выписка по поступлениям по карте / счету');
    
        jsa1 = json.arr_create();
        json.obj_set_arr(jso1, 'dtlastm_list', jsa1);
    
        begin
          last_date = dateadd(-13 month to f_date_month_begin(:kc_date));
          i = 0;
          while (i < 14) do
          begin
            jso_tmp = json.obj_create();
            json.arr_set_obj(jsa1, -1, jso_tmp);
            json.obj_set_i(jso_tmp, 'id', :i);
            json.obj_set_s(jso_tmp, 'name', f_convert_datetostr(:last_date, ''));
        
            last_date = dateadd(1 month to :last_date);
            i = i + 1;
          end
        end

        json.obj_set_s(jso1, 'head',  'АНКЕТА РАСЧЕТА ДОКУМЕНТАЛЬНО ПОДТВЕРЖДЕННОГО ДОХОДА');
        json.obj_set_s(jso1, 'text1', '1.1 Какой из следующих видов документов может предоставить клиент?');
        json.obj_set_s(jso1, 'text2', '1.2 Укажите месяц последнего полученного дохода согласно документам, подтверждающим доход, полученным от клиента');
        json.obj_set_s(jso1, 'text3', '1.3 Укажите дату приема на работу');
        json.obj_set_s(jso1, 'text4', 'ВНИМАНИЕ! Заполнить только, если клиент работает у единственного работодателя менее 12 мес. но более 3-х полных мес., если дата начала работы находится в указанном диапазоне дат:');

        json.obj_set_i(jso1, 'gdoc', cast(coalesce(f_json_str_get_value(:survey_json, 'gdoc', '0'), '0') as integer));

        if (coalesce(f_json_str_get_value(:survey_json, 'dtlastm', ''), '') = '') then
            json.obj_set_i(jso1, 'dtlastm', 13);
        else
            json.obj_set_s(jso1, 'dtlastm', f_json_str_get_value(:survey_json, 'dtlastm', ''));

        json.obj_set_s(jso1, 'dtpdn_txt',  f_convert_datetostr(:kc_date, '.lower.'));
        json.obj_set_d(jso1, 'dtjobb', f_date_month_begin(dateadd(-1 year to f_date_month_begin(:kc_date))));
        json.obj_set_d(jso1, 'dtjobe', f_date_month_end(dateadd(-4 month to f_date_month_begin(:kc_date))));
        json.obj_set_s(jso1, 'dtjob', f_json_str_get_value(:survey_json, 'dtjob', null));
        json.obj_set_s(jso1, 'employer', f_json_str_get_value(:survey_json, 'employer', ''));
    
        res_text = json.obj_to_json(jso1);
        suspend;
        exit;
    end

    if (action_kind = 'monthly_info') then
    begin
        select kc_id, survey_json
        from pdn_info_action(:obj_kind, :obj_id)
        into :kc_id, :survey_json;
    
        select kc_date
        from hp_pdn_getinfo_add(:kc_id)
        into :kc_date;
    
        if (:kc_date is null) then
        begin
            res_code = -1;
            res_text = 'Не указан месяц расчета ПДН!';
            suspend;
            exit;
        end

        dtlastm_idx = cast(coalesce(f_json_str_get_value(:survey_json, 'dtlastm', '13'), '13') as integer);
        dtjob       = f_json_str_get_value(:survey_json, 'dtjob', '');
        last_date = dateadd(iif(dtlastm_idx > 11, -1, -13 + dtlastm_idx) month to f_date_month_begin(:kc_date));
        last_date = f_date_month_begin(:last_date);

        if (coalesce(dtjob, '') <> '') then
            first_date = f_date_month_begin(dateadd(1 month to f_date_month_begin(cast(:dtjob as d_date))));
        else
            first_date = f_date_month_begin(dateadd(iif(dtlastm_idx > 11, -12, -13) month to :kc_date));

        jso1 = json.obj_create();
        jsa1 = json.arr_create();
        json.obj_set_arr(jso1, 'data', jsa1);
    
        income = 0;
        cur_date = :last_date;
    
        while (:cur_date >= :first_date) do
        begin
            jso_tmp = json.obj_create();
    
            json.obj_set_s(jso_tmp, 'nmonth', f_convert_datetostr(:cur_date, ''));
            json.obj_set_i(jso_tmp, 'vmonth', extract(year from :cur_date)*100 + extract(month from :cur_date));
    
            iterator = 0;
            found = false;
            monthly_count = f_json_str_get_count(:survey_json, 'data');
    
            while (iterator < :monthly_count) do
            begin
                monthly_info = f_json_str_get_value(:survey_json, f_format('data[%1].vmonth', :iterator), '0');
    
                if (cast(:monthly_info as integer) = extract(year from :cur_date)*100 + extract(month from :cur_date)) then
                begin
                    json.obj_set_s(jso_tmp, 'sum1', f_json_str_get_value(:survey_json, f_format('data[%1].sum1', :iterator), '0'));
                    json.obj_set_s(jso_tmp, 'sum2', f_json_str_get_value(:survey_json, f_format('data[%1].sum2', :iterator), '0'));
                    json.obj_set_s(jso_tmp, 'sum3', f_json_str_get_value(:survey_json, f_format('data[%1].sum3', :iterator), '0'));
                    json.obj_set_s(jso_tmp, 'sum4', f_json_str_get_value(:survey_json, f_format('data[%1].sum4', :iterator), '0'));
                    json.obj_set_s(jso_tmp, 'sumt', f_json_str_get_value(:survey_json, f_format('data[%1].sumt', :iterator), '0'));
                    found = true;
                    break;
                end
                iterator = iterator + 1;
            end
    
            if (not found) then
            begin
                json.obj_set_s(jso_tmp, 'sum1', '0');
                json.obj_set_s(jso_tmp, 'sum2', '0');
                json.obj_set_s(jso_tmp, 'sum3', '0');
                json.obj_set_s(jso_tmp, 'sum4', '0');
                json.obj_set_s(jso_tmp, 'sumt', '0');
            end
    
            income = :income + cast(f_json_str_get_value(json.obj_to_json(jso_tmp), 'sumt', '0') as decimal(18,2));
            json.arr_set_obj(jsa1, -1, jso_tmp);
            cur_date = dateadd(-1 month to :cur_date);
        end
    
        json.obj_set_s(jso1, 'text1', f_format('Заполните сведения о доходах клиента. Наименование предприятия "%1"', f_json_str_get_value(:survey_json, 'employer', '')));
        json.obj_set_f(jso1, 'income', :income);
    
        res_text = json.obj_to_json(jso1);
    
        suspend;
        exit;
    end

    if (action_kind = 'final_info') then
    begin
        select survey_json
        from pdn_info_action(:obj_kind, :obj_id)
        into :res_text;

        suspend;
        exit;
    end

    if (action_kind = 'employer_save') then
    begin
        execute procedure pdn_save_json(:obj_id, :un_id, :action_data, :action_kind);

--        select res_code, res_text 
--        from dpd_anketa_check(:obj_kind, :obj_id, 1) 
--        into res_code, res_text;

        suspend;
        exit;
    end

    if (action_kind = 'monthly_save') then
    begin
        execute procedure pdn_save_json(:obj_id, :un_id, :action_data, :action_kind);

        select res_code, res_text 
        from dpd_anketa_check(:obj_kind, :obj_id, 2) 
        into res_code, res_text;

        suspend;
        exit;
    end

    if (action_kind = 'final_save') then
    begin
        select res_code, res_text 
        from dpd_anketa_check(:obj_kind, :obj_id, 0) 
        into res_code, res_text;

        suspend;
        exit;
    end

    res_code = -1;
    res_text = 'Unknown action_kind = ' || coalesce(action_kind, '<NULL>');

    suspend;
    exit;
end
