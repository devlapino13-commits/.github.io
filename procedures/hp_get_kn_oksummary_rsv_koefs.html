    end
    else
    begin
      if (oks_delinq_days <= 30) then
        oks_ecl_acc_kind = 1;
      else
      if (oks_delinq_days <= 90) then
        oks_ecl_acc_kind = 2;
    end
  end

  -- Если договор расторжен (дефолт), призначем как 3 категорию
  if (oks_date_stop_perc is not null and
      oks_date_stop_perc <= oks_date_begin) then
  begin
    oks_ecl_acc_kind = 3;
    oks_ecl_kind = 2;
    oks_ecl_percent = ecl_lgd_percent;
  end

  if (oks_ecl_acc_kind = 3 and
      oks_date_begin < oks_restruct_date and
      oks_date_begin >= oks_last_sh_date) then
    oks_ecl_acc_kind = 2;

  suspend;
end
