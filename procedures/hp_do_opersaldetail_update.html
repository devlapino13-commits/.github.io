CREATE PROCEDURE HP_DO_OPERSALDETAIL_UPDATE(
    OL_ID D_ID,
    C_ID D_ID,
    SAC_ID D_INTEGER,
    SAL_VALUE D_CURRENCY)
AS
declare variable osm_id d_id;
begin
  select osm_id
  from opersalmain
  where osm_from_ol_id = :ol_id and
        osm_from_c_id = :c_id rows 1
  into osm_id;

  if (osm_id is null) then
  begin
    insert into opersalmain (osm_from_ol_id, osm_from_c_id)
    values (:ol_id, :c_id)
    returning osm_id
    into :osm_id;
  end

  update or insert into opersaldetail (osd_from_osm_id, osd_from_sac_id, osd_value)
  values (:osm_id, :sac_id, :sal_value)
  matching (osd_from_osm_id, osd_from_sac_id);
end
