CREATE PROCEDURE HP_BREFORMAT_FIXBEM(
    OBJ_CODE D_STR_CODE NOT NULL,
    OBJ_ID D_ID NOT NULL,
    BLN_TYPE D_INTEGER NOT NULL)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable edb_id d_id;
declare variable edc_name d_str_100;
declare variable num d_integer;
begin
  edc_name =
    case bln_type
      when 1 then '"Заявка на получение кредита"'
      when 2 then '"Кредитный договор"'
      when 5 then '"Анкета ПТФ/ОД"'
    else '???' end;

  num = 0;
  for
    select
      edb_id
    from edoc_blank
      inner join blankname
        on bln_id = edb_from_bln_id
          and bln_from_blt_id = :bln_type
    where edb_from_edl_id = (
      select edl_id
      from edoc_list
        inner join edoc_cfg
          on edc_id = edl_from_edc_id
            and edc_code = :obj_code
      where edl_object_id = :obj_id
        and edl_object_kind = edc_object_kind
      order by edl_id desc
      rows 1)
  into
    :edb_id
  do
  begin
    update edoc_blank set
      edb_status = 0
    where edb_id = :edb_id
      and edb_status <> 0;
    num = num + row_count;
  end

  res_text = '';
  if (num = 0) then
  begin
    if (edb_id is null) then
      res_text = 'Бланк для электронного документа %1 не был сформирован.';
    else
      res_text = 'Эл.документ %1 уже находится в очереди на переформирование.';
  end else
    res_text = 'Эл.документ %1 был поставлен в очередь на переформирование.';
  res_text = f_format(res_text, edc_name);

  suspend;
end
