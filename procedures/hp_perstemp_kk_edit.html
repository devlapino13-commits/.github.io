    :c_ext_inn,
    :c_pincode,
    :c_snils,
    :c_doc_type,
    :c_pass_no,
    :c_pass_date,
    :c_pass_date_expire,
    :c_pass_who,
    :c_pass_who_code,
    :c_blob_pass1_id,
    :c_blob_pass2_id,
    :c_blob_photo_id,
    :c_birth_country,
    :c_birth_city,
    :res_add_info;
    
  if (pte_pass_unlimit is null) then
    pte_pass_unlimit = 0;

  c_pass_unlimit = 0;

  if (extract(year from c_pass_date_expire) >= 2999) then
    c_pass_unlimit = 1;

  c_adl_id_1 = null;
  c_adl_id_2 = null;
  c_pass_drl_id = null;

  select adc_from_adl_id
  from addrclient
  where adc_from_c_id = :in_c_id and
        adc_address_type = 1
  into c_adl_id_1;

  select adc_from_adl_id
  from addrclient
  where adc_from_c_id = :in_c_id and
        adc_address_type = 2
  into c_adl_id_2;

  pte_edit_kind = null;

  select res_add_info
  from hp_perstemp_state_info(:in_c_id, :init_un_id)
  into :client_info;

  if (client_info <> '') then
    res_add_info = res_add_info || ', инфо: ' || client_info;

  un_fio = '';
  select un_fio, un_from_c_id
  from username
  where un_id = :init_un_id
  into un_fio, un_c_id;

  jl_name = '';
  select jl_name
  from hr_pstatus, joblist
  where ps_from_c_id = :un_c_id and
        ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_is_working = 1 and
        ps_from_jl_id = jl_id
  into jl_name;

  if (exists(
      select *
      from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.ClientVerifyShowInitiator')
      where has_access = 1)) then
  begin
    if (un_fio <> '') then
    begin
      select short_fio
      from hp_fio_clear(:un_fio)
      into un_fio;

      res_add_info = res_add_info || ', инициатор: ' || un_fio;

      if (jl_name <> '') then
        res_add_info = res_add_info || ', ' || jl_name;
    end
  end else
  if (jl_name <> '') then
    res_add_info = res_add_info || ', инициатор: ' || jl_name;

  if (res_warning = 1) then
    res_add_info = 'ВНИМАНИЕ, в списке ТиЭ. ' || res_add_info;

  -- RFCRU-4498 Код ОКАТО для адресов

  select pl.pl_add_code
  from addrlist adl
  inner join places pl on pl.pl_id = adl.adl_from_pl_id
  where adl.adl_id = :pte_adl_id_1
  into res_okato_for_pte_adl_id_1;

  select pl.pl_add_code
  from addrlist adl
  inner join places pl on pl.pl_id = adl.adl_from_pl_id
  where adl.adl_id = :pte_adl_id_2
  into res_okato_for_pte_adl_id_2;

  select pl.pl_add_code
  from addrlist adl
  inner join places pl on pl.pl_id = adl.adl_from_pl_id
  where adl.adl_id = :c_adl_id_1
  into res_okato_for_c_adl_id_1;

  select pl.pl_add_code
  from addrlist adl
  inner join places pl on pl.pl_id = adl.adl_from_pl_id
  where adl.adl_id = :c_adl_id_2
  into res_okato_for_c_adl_id_2;

  select cds_id,
         cds_doc_blob_1,
         cds_doc_blob_2,
         left(pk_name || ', годен до ' || f_datetostr(cds_doc_date_end) || ', добавлен ' || f_datetimetostr(cds_create_datetime), 100)
  from clientdocs
  inner join passkind on pk_id = cds_from_pk_id
  inner join passdoctype on pdt_id = pk_from_pdt_id
  where cds_from_c_id = :in_c_id and
        pdt_code in ('registration_fiz', 'registration_permanent', 'registration_yur') -- RFCRU-5729
  order by iif(cds_doc_date_end < current_date or cds_is_delete = 1, 1, 0), cds_create_datetime desc
  rows 1
  into pte_cds_id, -- документ регистрации (RFCRU-5061)
       pte_reg1_id,
       pte_reg2_id,
       pte_reg_info;

  select atr_from_rd_id
  from addresstype_registration
  where atr_from_cds_id = :pte_cds_id
  order by atr_datetime desc
  rows 1
  into pte_reg_type; -- код адреса регистрации (RFCRU-5407)

  suspend;
end
