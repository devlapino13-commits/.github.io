CREATE PROCEDURE HP_ZALOGAUTOLIST_SET_BLOB(
    ZAL_ID D_INTEGER,
    ZAL_BLOB D_BLOB_BINARY,
    UN_ID D_ID)
AS
declare variable bls_id d_id;
declare variable blob_count d_integer;
begin

  execute procedure hp_insert_into_blobstore ('car_screenshot.jpg', 'Скриншот авто', :zal_blob,
    :un_id, 2, 0)
  returning_values :bls_id;

  select count(*) from zalogautoblob
  where zab_from_zal_id = :zal_id
  into blob_count;

  if (blob_count = 1) then
    update zalogautoblob
    set zab_from_bls_id = :bls_id
    where zab_from_zal_id = :zal_id;
  else
    insert into zalogautoblob (zab_from_zal_id, zab_from_bls_id)
    values (:zal_id, :bls_id);

  update zalogautolist
  set zal_status = 2
  where zal_id = :zal_id;
end
