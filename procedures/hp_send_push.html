CREATE PROCEDURE HP_SEND_PUSH
AS
declare variable c_id d_id;
declare variable edl_id d_id;
declare variable res_text d_str_255;
declare variable i d_integer;
begin
  i = 0;
  for select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_is_delete = 0 and
      mcl.mcl_from_c_id is not null and
      (f_edoc_status('persdata.consent', mcl.mcl_from_c_id, current_date) = 'empty') and
      (f_asp_signed(mcl.mcl_from_c_id, current_date) <> 0)
  rows 10
  into :c_id
  do
  begin
    if (not exists (
          select 1
          from hp_edoc_client_continfo('persdata.consent', :c_id, 1000000000000) hp
          where hp.res_text  = '')) then
      continue;

    select edl_id, res_text
    from hp_edoc_insert('persdata.consent', f_objkind('clients', 'id'), :c_id, 1000000000000)
    into :edl_id, :res_text;

    if (res_text <> '') then
    begin
      insert into err_log (erl_id, edl_text) -- на первое время, для быстрого анализа
      values (:edl_id, :res_text);
    end
    else
    i = i + 1;

    if (i = 5) then
      exit;
  end
end
