CREATE PROCEDURE HP_DOCPRLOAN_TEST_LIST(
    OBJ_ID D_ID,
    PR_ID D_INTEGER,
    CLIENT_KIND D_SMALLINT,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    TEST_KIND D_STR_CODE,
    ONLY_INVALID D_BOOL = 0)
RETURNS (
    RES_ID D_INTEGER,
    RES_OBJECT_NAME D_STR_255,
    RES_SET_ID D_INTEGER,
    RES_DOC_SET_NAME D_STR_100,
    RES_DOCPROP_NAME D_STR_255,
    RES_VALID D_SMALLINT,
    RES_C_ID D_ID,
    RES_K_ID D_ID)
AS
declare variable k_kontract_no d_str_code;
declare variable rep_date d_date;
begin
  if (test_kind not in ('clients', 'kontract', 'process', 'process_all', 'disb', 'disb_all')) then
    exit;

  if (only_invalid is null) then
    only_invalid = 0;

  res_k_id = null;
  res_id = 0;

  if (test_kind = 'clients') then
  begin
    select c_id, c_name_fio from clients
    where c_id = :obj_id
    into res_c_id, res_object_name;

    rep_date = rep_date_end;

    for select res_set_id, res_doc_set_name, res_docprop_name, res_valid
    from hp_docprloan_valid(:res_c_id, :pr_id, :client_kind, :rep_date, 1)
    into :res_set_id, :res_doc_set_name, :res_docprop_name, :res_valid
    do
    begin
      res_id = res_id + 1;
      suspend;
    end
  end

  if (test_kind = 'kontract') then
  begin
    select k_id, k_kontract_no, k_disb_date, k_from_pr_id
    from kontract
    where k_id = :obj_id
    into res_k_id, k_kontract_no, rep_date, pr_id;

    for select c_id, :k_kontract_no || ', ' || c_name_fio
    from clients, knclient
    where kc_from_k_id = :res_k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = c_id
    into res_c_id, res_object_name
    do
    begin
      if (only_invalid = 1 and f_docprloan_valid(:res_c_id, :pr_id, :client_kind, :rep_date) <> -1) then
        Continue;

      for select res_set_id, res_doc_set_name, res_docprop_name, res_valid
      from hp_docprloan_valid(:res_c_id, :pr_id, :client_kind, :rep_date, 1)
      into :res_set_id, :res_doc_set_name, :res_docprop_name, :res_valid
      do
      begin
        res_id = res_id + 1;
        suspend;
      end
    end
  end

  if (test_kind = 'process' or test_kind = 'process_all') then
  begin
    rep_date = rep_date_end;

    for select k_id, k_kontract_no
    from kontract
    where k_status in (0, 1) and
          k_disb_date = :rep_date and
          (:test_kind = 'process_all' or k_from_pr_id = :pr_id)
    into res_k_id, k_kontract_no
    do
    begin
      for select c_id, :k_kontract_no || ', ' || c_name_fio
      from clients, knclient
      where kc_from_k_id = :res_k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id
      into res_c_id, res_object_name
      do
      begin
        if (only_invalid = 1 and f_docprloan_valid(:res_c_id, :pr_id, :client_kind, :rep_date) <> -1) then
          Continue;

        for select res_set_id, res_doc_set_name, res_docprop_name, res_valid
        from hp_docprloan_valid(:res_c_id, :pr_id, :client_kind, :rep_date, 1)
        into :res_set_id, :res_doc_set_name, :res_docprop_name, :res_valid
        do
        begin
          res_id = res_id + 1;
          suspend;
        end
      end
    end
  end

  if (test_kind = 'disb' or test_kind = 'disb_all') then
  begin
    for select k_id, k_kontract_no, ks_date_begin
    from kontract, kontstate
    where ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_state = 1 and
          ks_ol_is_delete = 0 and
          ks_from_k_id = k_id and
          (:test_kind = 'disb_all' or k_from_pr_id = :pr_id)
    into res_k_id, k_kontract_no, rep_date
    do
    begin
      for select c_id, :k_kontract_no || ', ' || c_name_fio
      from clients, knclient
      where kc_from_k_id = :res_k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id
      into res_c_id, res_object_name
      do
      begin
        if (only_invalid = 1 and f_docprloan_valid(:res_c_id, :pr_id, :client_kind, :rep_date) <> -1) then
          Continue;

        for select res_set_id, res_doc_set_name, res_docprop_name, res_valid
        from hp_docprloan_valid(:res_c_id, :pr_id, :client_kind, :rep_date, 1)
        into :res_set_id, :res_doc_set_name, :res_docprop_name, :res_valid
        do
        begin
          res_id = res_id + 1;
          suspend;
        end
      end
    end
  end
end
