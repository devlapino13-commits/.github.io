CREATE PROCEDURE HP_CHECK_LIMIT_KONTRACT(
    K_ID D_ID NOT NULL)
RETURNS (
    ERROR_STATUS D_STR_100,
    ERROR_TEXT D_STR_255)
AS
declare variable c_name_fio d_str_100;
declare variable final_pdn d_currency;
declare variable k_amount_approved d_currency;
declare variable final_limit d_currency;
declare variable is_error d_smallint;
declare variable c_id d_id;
declare variable ks_date_begin d_date;
declare variable kol_day d_integer;
declare variable k_refund d_smallint;
declare variable k_id_last d_id;
declare variable pr_code_last d_str_100;
declare variable kc_id d_id;
begin
  --RFCRU-2874 автоматизация продукта Удобный 3

  -- Лимит проверяется в :
  -- ОТПРАВКА КК
  -- УТВЕРЖДЕНИЕ ЗАЙМА
  -- ВЫДАЧА
  error_status = 1;
  error_text = '';
  final_pdn = 0;
  for select kn.k_id, kn.k_amount_approved, knc.kc_from_c_id, kn.k_refund, kc_id
      from kontract kn
      inner join knclient knc on knc.kc_from_k_id = kn.k_id
      inner join product pr on pr.pr_id = kn.k_from_pr_id
      where kn.k_id = :k_id and
            knc.kc_amount_approved > 0 and
            pr.pr_code containing '.comfortable.'
      into :k_id, k_amount_approved, c_id, k_refund, kc_id
  do
  begin
    --проверяем условие пдн и суммы займа
    select kc_pdn_koef
    from knclient_pdn
    where kc_id = :kc_id and
          kcp_kind = 0
    into final_pdn;

    if (final_pdn <= 50) then
    begin
      if ((k_amount_approved = 49000) or (k_amount_approved = 100000)) then
        error_status = 0;
    end
    else
    begin
      if (k_amount_approved = 49000) then
        error_status = 0;
    end

    if (error_status = 1) then
    begin
       if (final_pdn > 50) then
        error_text = f_format('ПДН более 50%, сумма займа должна быть 49 000р');
      else
        error_text = f_format('ПДН не более 50%, сумма займа должна быть 49 000р или 100 000 р');
      --error_text = f_format('По займу "Удобный 3.0" сумма займа может быть 49 000 или 100 000');
      suspend;
    end
  end
end
