CREATE PROCEDURE FM_HTML_MAILMESSAGE_BODY(
    UN_ID DECIMAL(18,0),
    OBJECT_KIND VARCHAR(100),
    OBJECT_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable TMP_BODY blob sub_type 1 segment size 80;
declare variable RES_TYPE integer;
declare variable RES_SUBJECT varchar(255);
declare variable RES_DATETIME timestamp;
declare variable SENDER_UN_ID decimal(18,0);
declare variable RES_SENDER_FIO varchar(100);
declare variable RES_TYPE_STR varchar(20);
declare variable MAIL_TO varchar(2048);
declare variable DB_ID integer;
declare variable UN_NAME D_STR_100;
begin
  select un_name
  from username
  where un_id = :un_id
  into :un_name;

  if (:un_name = 'fmtest') then
  begin
    if (object_kind = 'fsrobot_mailmain_ceo') then
    begin
      select html_content
      from apple_test_data
      where report_type = 'CEO_MESSAGE'
      into :res_text;
      suspend;
    end
    else if (object_kind = 'fsrobot_mailmain_other') then
    begin
      select html_content
      from apple_test_data
      where report_type = 'OTHER_MESSAGE'
      into :res_text;
      suspend;
    end
    exit;
  end

  db_id = f_get_db_id();
  res_text = iif(db_id <> 94401, 'Сообщение не найдено', 'Mesaj tapılmadı');
  for select 0, fsm_subject, fsm_body, fsm_datetime, fsm_from_un_id, fsm_mail_to
      from fsrobot_mailmain
      where fsm_is_delete = 0 and
            fsm_id = :object_id and
            :object_kind = 'fsrobot_mailmain'
      union
      select msg_type, msg_subject, msg_body, msg_datetime, msg_from_un_id, ''
      from fsrobot_msgbox
      where msg_id = :object_id and
            :object_kind = 'fsrobot_msgbox' and
            msg_status = 1
      into :res_type, :res_subject, :tmp_body, :res_datetime, :sender_un_id, mail_to
  do
  begin
    select c_name_fio
    from username, clients
    where un_id = :sender_un_id and
          un_from_c_id = c_id
    into :res_sender_fio;

    if (tmp_body is null) then
      tmp_body = '';
    if (res_type = 2 and
        tmp_body <> '') then
    begin
      select title, t.body
      from fm_json_extract(:tmp_body) t
      into res_subject, :tmp_body;
    end

    res_type_str = (case res_type
                      when 0 then 'mail'
                      when 1 then 'sms'
                      when 2 then 'push'
                      when 3 then 'sms'
                      when 4 then 'email'
                    end);

    select name_surname
    from hp_fio_clear(:res_sender_fio)
    into :res_sender_fio;

    res_text = '<b>' || res_subject || f_format('</b><br><br><small>%1: <b>%2</b><br>', iif(db_id <> 94401, 'От кого', 'Kimdən'), res_sender_fio);
    if (sender_un_id = un_id and
        coalesce(mail_to, '') <> '') then
      res_text = BLOB_APPEND(res_text, f_format('%1: <b>%2</b><br>', iif(db_id <> 94401, 'Кому', 'Kimə'), mail_to));
    res_text = BLOB_APPEND(res_text, res_type_str || iif(db_id <> 94401, ' от ', ' vasitəsi ilə ') || f_datetimetostr(res_datetime) || '</small><br>');

    res_text = BLOB_APPEND(res_text, '<hr width="100%" color="#0b7fab">');
    res_text = BLOB_APPEND(res_text, tmp_body);
  end
  suspend;
end
