CREATE PROCEDURE HP_GET_HTML_LOAN_LIMIT(
    C_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable cli_balance d_currency;
declare variable secured_balance d_currency; /* Баланс залоговых займов клиента */
declare variable unsecured_balance d_currency; /* Баланс беззалоговых займов клиента */
declare variable l1_limit d_currency; /* Лимит Л1 из документа */
declare variable l2_limit d_currency; /* Лимит Л2 из документа */
declare variable amount_max d_currency;
declare variable available_limit d_currency;
declare variable client_point d_integer;
declare variable pr_id d_integer;
declare variable temp_limit d_currency;
begin
  -- RFCRU-5521
  cli_balance = 0;
  secured_balance = 0;
  unsecured_balance = 0;
  l1_limit = 0;
  l2_limit = 0;
  amount_max = 0;
  available_limit = 0;
  client_point = 0;
  pr_id = null;
  res_text = '';

  select coalesce(sum(oks.oks_balance), 0),                             -- Общий баланс
         coalesce(sum(iif(p.pr_is_secured = 1, oks.oks_balance, 0)), 0) -- Только залоговый баланс
  from knclient kc
  join oksummary oks on kc.kc_from_k_id = oks.oks_from_k_id
  left join miniapp ma on kc.kc_from_c_id = ma.ma_id
  left join product p on ma.ma_from_pr_id = p.pr_id
  where kc.kc_from_c_id = :c_id
    and kc.kc_amount_approved > 0
    and oks.oks_balance > 0
    and oks.oks_close_state = 0
    and current_date between oks.oks_date_begin and oks.oks_date_end
  into :cli_balance, :secured_balance;

  -- Вычисляем БЕЗЗАЛОГОВЫЙ баланс клиента
  unsecured_balance = cli_balance - secured_balance;

  -- Получаем баллы клиента
  client_point = f_point(:c_id, current_date, 0);

  l1_limit = case
    when :client_point between 0 and 6   then 100000
    when :client_point between 7 and 11  then 150000
    when :client_point between 12 and 17 then 200000
    when :client_point between 18 and 24 then 250000
    when :client_point between 25 and 35 then 300000
    when :client_point between 36 and 45 then 350000
    when :client_point >= 46             then 450000
    else 100000
  end;

  l2_limit = case
    when :client_point between 0 and 11  then 200000
    when :client_point between 12 and 17 then 250000
    when :client_point between 18 and 24 then 300000
    when :client_point between 25 and 35 then 350000
    when :client_point between 36 and 45 then 400000
    when :client_point >= 46             then 500000
    else 200000
  end;

  select first 1 ma.ma_from_pr_id
  from miniapp ma
  join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
  where mac.mac_from_c_id = :c_id
    and mac.mac_amount > 0
    and ma.ma_is_active = 1
    and ma.ma_from_pr_id is not null
  order by ma.ma_date_app desc
  into :pr_id;

  if (pr_id is not null) then
  begin
    begin
      if (secured_balance > 0) then
      begin
        temp_limit = l2_limit - secured_balance;
        if (l1_limit < temp_limit) then
          temp_limit = l1_limit;

        available_limit = temp_limit - unsecured_balance;
      end
      else
      begin
        available_limit = l1_limit - cli_balance;
      end

      if (available_limit < 0) then
        available_limit = 0;

    when any do
      begin
        available_limit = 0;
      end
    end
  end

  res_text =
      '<br><b>' ||
      'Общий баланс активных займов (ОД): ' || round(:cli_balance) || ' руб.<br>' ||
      'Доступный лимит: ' || round(:available_limit) || ' руб.' ||
      '</b></font><br>';

  suspend;
end
