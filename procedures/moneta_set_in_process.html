CREATE PROCEDURE MONETA_SET_IN_PROCESS(
    ESM_ID D_ID,
    ESM_FULL_SOURCE D_STR_8100)
AS
begin
  --   Процедура переводит выплату во временный статус, перед отправкой денег на
  --   карту/СБП/счет, по Монете.
  --   Процедура вызывается сразу перед отправкой на карту и при переводе через
  --   Систему Быстрых Платежей по номеру телефона, переводе на расчетный счет

  if (esm_id is null) then
    exception e_default 'ESM_ID is null';

  if (not exists(select *
                 from extsysmain esm
                 where esm.esm_id = :esm_id and
                       esm.esm_project = 'moneta_disb' and
                       esm.esm_action in ('CardDisb', 'PhoneDisb', 'AccDisb') and
                       esm.esm_status = 0)) then
  begin
    exception e_default f_format('Incorrect esm_id: %1', esm_id);
  end

  update extsysmain
  set esm_status = 999,
      esm_full_source = :esm_full_source
  where esm_id = :esm_id;
end
