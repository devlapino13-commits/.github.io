CREATE PROCEDURE AAREP_CONTROL_LEVELS_PER(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KD_OL_DATE D_DATE,
    K_KONTRACT_NO STR_100,
    K_FACT_DISB_DATE D_DATE,
    PR_NAME VARCHAR(100),
    OKS_DELINQ_DATE D_DATE,
    FLOAT_PERC_LEVEL_ZN VARCHAR(10),
    DELINQ_DATE INTEGER,
    PERC_PER_YEAR_BEFORE DECIMAL(18,7),
    PERC_PER_YEAR_AFTER DECIMAL(18,7),
    AAREP_PLAN_OS DECIMAL(18,4),
    AAREP_PLAN_PERC DECIMAL(18,4),
    REP_PLAN_OS_FACT DECIMAL(18,4),
    REP_PLAN_PERC_FACT DECIMAL(18,4),
    AMOUNT_OF_OC DECIMAL(18,4),
    AMOUNT_OF_PERC DECIMAL(18,4),
    KONT_ZAD_OS_2 DECIMAL(18,4),
    OKS_DELINQ_OS DECIMAL(18,4),
    OKS_DELINQ_OS_BEFORE DECIMAL(18,4),
    KONT_ZAD_PERC_2 DECIMAL(18,4),
    OKS_DELINQ_PERC DECIMAL(18,4),
    OKS_DELINQ_PERC_BEFORE DECIMAL(18,4))
AS
declare variable k_id id_type;
declare variable float_perc_level d_smallint;
declare variable date1 d_date;
declare variable count_d integer;
declare variable coun_dey integer;
declare variable kont_zad_os decimal(18,4);
declare variable kont_zad_perc decimal(18,4);
declare variable plan_oc decimal(18,4);
declare variable plan_perc decimal(18,4);
declare variable oks_pmt_os_norm decimal(18,4); /* Своевременные платежи (день в день) ОС */
declare variable oks_pmt_perc_norm decimal(18,4); /* Своевременные платежи (день в день)% */
begin
  float_perc_level_zn = '';
  aarep_plan_os = 0;
  aarep_plan_perc = 0;
  oks_delinq_date = null;
  perc_per_year_after = 0;
  perc_per_year_before = 0;
  rep_plan_os_fact = 0;
  rep_plan_perc_fact = 0;
  amount_of_oc = null;
  amount_of_perc = null;
  kont_zad_os_2 = 0;
  kont_zad_perc_2 = 0;
  delinq_date = 0;
  plan_oc = 0;
  plan_perc = 0;
  oks_pmt_os_norm = 0;
  oks_pmt_perc_norm = 0;
  --получаем список контрактов по которым произошли повышения ставки на дату отчета
  for select kd_from_k_id, kd_float_perc_level, kd_ol_date
      from kontdata, operlist
      where kd_float_perc_level <> 0 and
            ol_id = kd_from_ol_id and
            kd_ol_date between :rep_date_begin and :rep_date_end and
            ol_oper_type = 1 and
            ol_oper_subtype = 12 and
            kd_ol_is_delete = 0
      into :k_id, :float_perc_level, :kd_ol_date
  do
  begin
    if (float_perc_level = 1) then --определяем уровень повыешения ставки
      float_perc_level_zn = '№1';

    if (float_perc_level = 2) then
      float_perc_level_zn = '№2';

    if (float_perc_level > 2) then
      float_perc_level_zn = '№3';

    delinq_date=0;
    select --oks.oks_delinq_date, --получаем просроченные дни на момент отчета
           iif((cast(:kd_ol_date as date) - oks.oks_delinq_date < 0), 0, cast(:kd_ol_date as date) - oks.oks_delinq_date)
    from oksummary oks
    where :kd_ol_date - 1 between oks.oks_date_begin and oks.oks_date_end and
          cast(:kd_ol_date as date) - oks.oks_delinq_date >= 1 and
          oks.oks_from_k_id = :k_id
    order by oks_delinq_date asc
    rows 1
    into :delinq_date;
    --into :oks_delinq_date, :delinq_date;

    oks_delinq_date=null;
    select oks.oks_delinq_date -- плановая дата погашения
    from oksummary oks
    where :kd_ol_date  between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_from_k_id = :k_id
    order by oks_delinq_date asc
    rows 1
    into :oks_delinq_date;

    select kont.k_kontract_no, --№ контракта
           prod.pr_name, --кр. продукт
           k_fact_disb_date --дата выдачи
    from kontract kont
    inner join product prod on prod.pr_id = kont.k_from_pr_id
    inner join oksummary ok on ok.oks_from_k_id = kont.k_id
    where oks_date_begin between '01.01.1900' and :rep_date_end and
          kont.k_id = :k_id
    rows 1
    into :k_kontract_no, :pr_name, :k_fact_disb_date;

    select kd_percent_per_day * 365 as perc_per_year --получаем ставку после повышения
    from kontdata
    where kd_from_k_id = :k_id and
          :kd_ol_date between kd_ol_date and kd_date_end and
          kd_ol_is_delete = 0
    into :perc_per_year_after; --после изменений

    select kd_percent_per_day * 365 as perc_per_year --получаем ставку до повышения
    from kontdata
    where kd_from_k_id = :k_id and
          :kd_ol_date - 1 between kd_ol_date and kd_date_end and
          kd_ol_is_delete = 0
    into :perc_per_year_before; --до изменений

    if (aarep_plan_os = 0) then
      amount_of_oc = 0;
    else
      amount_of_oc = amount_of_oc;

    if (aarep_plan_perc = 0) then
      amount_of_perc = 0;
    else
      amount_of_perc = amount_of_perc;

    select sum(zad_os), sum(zad_perc)
    from aarep_make_zad_list(:k_id, :kd_ol_date, 1)
    where sh_date is not null
    into :kont_zad_os, :kont_zad_perc;

    select sum(oks.oks_delinq_os), sum(oks.oks_delinq_perc)
    from oksummary oks
    where oks.oks_from_k_id = :k_id and
          oks.oks_date_begin <= :kd_ol_date and
          oks.oks_date_end >= :kd_ol_date and
          oks.oks_close_state = 0
    into :oks_delinq_os, oks_delinq_perc;

    select sum(oks.oks_delinq_os), sum(oks.oks_delinq_perc)
    from oksummary oks
    where oks.oks_from_k_id = :k_id and
          oks.oks_date_begin <= (:oks_delinq_date - 1) and
          oks.oks_date_end >= (:oks_delinq_date - 1) and
          oks.oks_close_state = 0
    into :oks_delinq_os_before, oks_delinq_perc_before;

    select sum(zad_os), --задолженность по ОС
           sum(sh_calc_perc) --накопленные % на дату
    from aarep_make_zad_list(:k_id, :kd_ol_date, 1)
    into :kont_zad_os_2, :kont_zad_perc_2;

    select zad_os, zad_perc
    from aarep_make_zad_list(:k_id, :oks_delinq_date, 0)
    rows 1
    into :aarep_plan_os, :aarep_plan_perc;

    amount_of_oc = aarep_plan_os;
    amount_of_perc = aarep_plan_perc;

    select sum(oks_pmt_os_norm), sum(oks_pmt_perc_norm), sum(oks_pmt_os_delinq), sum(oks_pmt_perc_delinq),
           sum(oks_pmt_os_adv), sum(oks_pmt_perc_adv)
    from kn_oksummary, knclient, kontract, clients, knaktlist akt
    where oks_date_begin = :kd_ol_date - 1 and --внесла изменение -1
          oks_from_kc_id = kc_id and
          kc_from_k_id = k_id and
          kc_from_c_id = c_id and
          oks_from_k_id = :k_id and
          akt.kal_from_kf_id = k_from_kf_id and
          akt.kal_date <= :kd_ol_date and
          akt.kal_date_end >= :kd_ol_date and
          akt.is_delete = 0
    into :oks_pmt_os_norm, :oks_pmt_perc_norm, :rep_plan_os_fact, :rep_plan_perc_fact, :plan_oc, :plan_perc;

    rep_plan_os_fact = -(oks_pmt_os_norm + rep_plan_os_fact + plan_oc); --получаем фактическую сумму погашения по ОС
    rep_plan_perc_fact = -(oks_pmt_perc_norm + rep_plan_perc_fact + plan_perc); --получаем Фактическая сумму погашения %%

    if (rep_plan_os_fact is not null) then
      amount_of_oc = aarep_plan_os - rep_plan_os_fact;
    else
      amount_of_oc = aarep_plan_os;

    if (rep_plan_perc_fact is not null) then
      amount_of_perc = aarep_plan_perc - rep_plan_perc_fact;
    else
      amount_of_perc = aarep_plan_perc;

    if (rep_plan_os_fact is null) then
       rep_plan_os_fact = 0.00;
     else
       rep_plan_os_fact = rep_plan_os_fact;

    if (rep_plan_perc_fact is null) then
       rep_plan_perc_fact = 0.00;
     else
       rep_plan_perc_fact = rep_plan_perc_fact;
    suspend;
  end
end
