CREATE PROCEDURE AAREP_OKA_REPAYMENT_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    APL_DATE DATE,
    APL_REGION VARCHAR(100),
    APL_CLIENT_NAME VARCHAR(100),
    APL_MEMO VARCHAR(1024),
    APL_DATETIME TIMESTAMP,
    APL_USER_NAME VARCHAR(100),
    APL_AMOUNT DECIMAL(15,2),
    APL_NEW_KF_NAME VARCHAR(100),
    APL_OS DECIMAL(18,4),
    APL_PERC DECIMAL(18,4),
    APL_FINE DECIMAL(18,4),
    APL_KONTRACT_NO VARCHAR(20),
    IS_AMNIST INTEGER,
    K_KONTRACT_CATEGORY VARCHAR(50),
    KF_OFFICER_NAME VARCHAR(100))
AS
declare variable apl_ol_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
declare variable apl_k_id decimal(18,0);
declare variable kal_new_kf_officer curr_type;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ol_id, ol_date, reg_code || ' ' || reg_name, k_id, k_kontract_no, left(ol_doc_client_name, 100), ol_memo,
             ol_datetime, c_name_fio, kal_new_kf_name, sum(-1 * ok_os - ok_perc - ok_fine - ok_kommis - ok_kommis_adv),
             -sum(ok_os), -sum(ok_perc), -sum(ok_fine), kal_new_kf_officer

      from operkred, operlist, username, region, clients, knaktlist, kontract, region_list(:rep_region_id, 1, -1, 1)
      where ok_from_ol_id = ol_id and
            reg_unicode = id_list and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_type = 4 and
            ol_user_id = un_id and
            reg_unicode = ol_from_reg_id and
            un_from_c_id = c_id and
            ok_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            ok_ol_is_active = 1 and
            operlist.is_delete = 0 and
            knaktlist.is_delete = 0 and
            ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
            and
            ok_from_c_id = 0
      group by ol_id, ol_date, reg_code, reg_name, ol_doc_client_name, ol_memo, ol_datetime, kal_new_kf_name, c_name_fio, k_kontract_no, k_id, kal_new_kf_officer
      into :apl_ol_id, :apl_date, :apl_region, :apl_k_id, :apl_kontract_no, :apl_client_name, :apl_memo, :apl_datetime,
           :apl_user_name, :apl_new_kf_name, :apl_amount, :apl_os, :apl_perc, :apl_fine, :kal_new_kf_officer
  do
  begin
    if (exists(select 1
               from kontdevides
               where kd_k_new = :apl_k_id)) then
      is_amnist = 1;
    else
      is_amnist = 0;

    k_kontract_category = null;
    select kont_category
    from aarep_get_category_id(:apl_k_id, :apl_date - 1, null, null, null, null, null, null)
    into k_kontract_category;
    if (k_kontract_category is null) then
      k_kontract_category = 'не определен';

    kf_officer_name = (select c_name_fio
                       from clients
                       where c_id = :kal_new_kf_officer);

    suspend;
  end
end
