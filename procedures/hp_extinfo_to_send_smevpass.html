CREATE PROCEDURE HP_EXTINFO_TO_SEND_SMEVPASS(
    C_ID ID_TYPE,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    JSON_VALUE D_STR_8100)
AS
declare variable c_name_fio d_str_100;
declare variable last_name d_str_100;
declare variable first_name d_str_100;
declare variable middle_name d_str_100;
declare variable doc_serial d_str_10;
declare variable doc_number d_str_20;
declare variable pk_id d_integer;
declare variable doc_no d_str_code;
declare variable c_birth_date d_date;
begin
  if (object_kind <> f_objkind('clientdocs', 'id')) then
    exception e_default 'Wrong object_kind = ' || coalesce(object_kind, '<NULL>');

  select c_name_fio, c_birth_date
  from clients
  where c_id = :c_id
  into c_name_fio, c_birth_date;

  select name_last, name_first, name_middle
  from hp_fio_clear_ru(:c_name_fio)
  into last_name, first_name, middle_name;

  -- RFCRU-4749
  select sfr_last_name, sfr_first_name, sfr_patronymic
  from smevpass_fio_for_request
  where sfr_from_c_id = :c_id
    and sfr_is_delete = 0
  into last_name, first_name, middle_name;

  select cds_from_pk_id, cds_doc_no
  from clientdocs cds
  where cds_id = :object_id
  into pk_id, doc_no;

  select replace(pass_seria, ' ', ''), pass_number
  from split_passport_number(:pk_id, :doc_no)
  into doc_serial, doc_number;

  json_value = '{' ||
          f_json_make_pair('F_name', first_name, 'str',  1) ||
          f_json_make_pair('L_name', last_name, 'str',  1) ||
          iif(coalesce(middle_name, '') = '', '', f_json_make_pair('Patronymic', middle_name, 'str',  1)) ||
          f_json_make_pair('b_date', c_birth_date, 'date',  1) ||
          f_json_make_pair('doc_type', '21', 'str',  1) ||
          f_json_make_pair('doc_series', doc_serial, 'str',  1) ||
          f_json_make_pair('doc_num', doc_number, 'str') ||
      '}';

  suspend;
end
