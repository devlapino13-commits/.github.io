CREATE PROCEDURE AAREP_CARD_ISSUE_ACQUIRING(
    REP_DATE_END D_DATE = current_date)
RETURNS (
    RD_ID D_INTEGER,
    RD_NAME D_STR_255,
    NUMCARD D_INTEGER,
    NUMCARDACC D_INTEGER,
    NUMKONT D_INTEGER,
    AMOUNT D_ID,
    CASH_NUM D_INTEGER,
    CASH_AMOUNT D_ID,
    NCASH_NUM D_INTEGER,
    NCASH_AMOUNT D_ID)
AS
begin
  insert into temp_card (card_id, rd_id)
  select
    cal_card_id,
    rd_id
  from cardlist
    inner join cardplace
      on cpl_id = cal_last_cpl_id
    inner join territoriessts_link
      on tsl_reg_id = cpl_from_reg_id
    inner join refdata
      on rd_id = tsl_rd_id
  where cal_date_issue < :rep_date_end
    and exists(select * from cardlistloan where clo_card_id = cal_card_id);

  for
    with card as
    (
      select
        rd_id,
        count(*) as numcard
      from temp_card
      group by rd_id
    ), kont as
    (
      select
        a.rd_id,
        count(*) as numkont,
        sum(a.amount) as amount
      from (
        select
          rd_id,
          clo_from_k_id as k_id,
          avg(k_amount_approved) as amount
        from temp_card
          inner join cardlistloan
            on clo_card_id = card_id
          inner join kontract
            on k_id = clo_from_k_id
              and k_disb_date < :rep_date_end
         group by rd_id, clo_from_k_id) as a
      group by a.rd_id
    ), cash as
    (
      select
        rd_id,
        sum(cash_amount) as cash_amount,
        sum(ncash_amount) as ncash_amount,
        sum(cash_num) as cash_num,
        sum(ncash_num) as ncash_num
      from temp_card
        inner join (
            select
              cs_card_id as card_id,
              0 as cash_amount,
              cs_amount as ncash_amount,
              0 as cash_num,
              1 as ncash_num
            from cardsync
              inner join cardsyncopers
                on cso_from_cs_id = cs_id
              inner join operlist
                on ol_id = cso_from_ol_id
                  and ol_date < :rep_date_end
                  and is_delete = 0
            where cs_from_csc_id = 20
              and cs_status = 2
            union all
            select
              ied_from_card_id as card_id,
              iif(ied_trade_point_mcc in ('6010','6011'), ied_trans_amount, 0) as cash_amount,
              iif(ied_trade_point_mcc in ('6010','6011'), 0, ied_trans_amount) as ncash_amount,
              iif(ied_trade_point_mcc in ('6010','6011'), 1, 0) as cash_num,
              iif(ied_trade_point_mcc in ('6010','6011'), 0, 1) as ncash_num
            from ipc_exd_detail
              inner join operlist
                on ol_id = ied_from_ol_id
                  and ol_date < :rep_date_end
                  and is_delete = 0
            where ied_from_ol_id > 0
            union all
            select
              po_card_id as card_id,
              po_amount as cash_amount,
              0 as ncash_amount,
              1 as cash_num,
              0 as ncash_num
            from pvnopers
              inner join operlist
                on ol_id = po_from_ol_id
                  and ol_date < :rep_date_end
                  and is_delete = 0
            where po_from_ol_id > 0) as a
          on a.card_id = temp_card.card_id
      group by rd_id
    )
    select
      card.rd_id,
      numcard,
      numkont,
      amount/1000,
      cash_num,
      cash_amount/1000,
      ncash_num,
      ncash_amount/1000
    from card
      left join kont
        on kont.rd_id = card.rd_id
      left join cash
        on cash.rd_id = card.rd_id
  into
    :rd_id,
    :numcard,
    :numkont,
    :amount,
    :cash_num,
    :cash_amount,
    :ncash_num,
    :ncash_amount
  do
  begin
    rd_name = (select rd_name from refdata where rd_id = :rd_id);
    numcardacc = numcard;
    suspend;
  end
end
