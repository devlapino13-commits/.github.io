CREATE PROCEDURE MC_EDIT_CLIENT(
    MCL_ID D_ID,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE)
RETURNS (
    C_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable cell_no d_str_code;
declare variable leven integer;
declare variable clt_id d_integer;
begin
  select mcl_login, mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into cell_no, c_id;

  if (c_id is not null) then
  begin
    res_text = 'Ваши данные уже подтверждены';
    suspend;
    exit;
  end

  if (f_get_const('new_contdata') = 0) then -- если работаем по старой структуре. После переключения удалить код ниже.
  begin
    select cci_object_id
    from continfo, clients
    where cci_object_kind = 0 and
          cci_object_id = c_id and
          c_birth_date = :c_birth_date and
          c_name_fio = :c_name_fio and
          (ok_cell_main = :cell_no or
          ok_cell_add = :cell_no) rows 1
    into c_id;
  
    if (c_id is null) then
    begin
      for select cci_object_id, f_levenstain(c_name_fio, :c_name_fio, 1)
          from continfo, clients
          where cci_object_kind = 0 and
                cci_object_id = c_id and
                c_birth_date = :c_birth_date and
                (ok_cell_main = :cell_no or
                ok_cell_add = :cell_no)
          order by 2, 1 desc
          into c_id, leven
      do
      begin
        if (leven = 1) then
          break;
        c_id = null;
      end
    end
  end else
  begin
    clt_id = f_cont_id(cell_no, null);
    if (clt_id is not null) then
    begin
      select c_id
      from contdata, clients
      where cda_from_c_id = c_id and
            c_birth_date = :c_birth_date and
            c_name_fio = :c_name_fio and
            cda_from_clt_id = :clt_id and
            cda_cont_kind = f_enum('cont_kind', 'cell', 'id') and
            cda_status = 1 and
            cda_is_delete = 0
      order by cda_is_main desc
      rows 1
      into c_id;

      if (c_id is null) then
      begin
        select c_id
        from contdata, clients
        where cda_from_c_id = c_id and
              c_birth_date = :c_birth_date and
              cda_from_clt_id = :clt_id and
              cda_cont_kind = f_enum('cont_kind', 'cell', 'id') and
              cda_status = 1 and
              cda_is_delete = 0 and
              f_levenstain(c_name_fio, :c_name_fio, 1) = 1
        order by cda_is_main desc, c_id desc
        rows 1
        into c_id;
      end
    end
  end

  if (c_id is not null) then
  begin
    update mc_login
    set mcl_from_c_id = :c_id
    where mcl_id = :mcl_id;

    res_text = 'Ваши данные подтверждены';
  end
  else
  begin
    res_text = 'Мы не нашли ваши данные с таким номером телефона.';
  end
  suspend;
end
