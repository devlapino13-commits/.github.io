CREATE PROCEDURE HP_OPERCASH_HISTORY_FILL(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
AS
begin
  delete from opercash
  where opc_date between :rep_date_begin and :rep_date_end;

  insert into opercash (opc_date, opc_doc_no, opc_office_id, opc_from_csp_id, opc_oper_kind,
                        opc_sign, opc_amount, opc_from_c_id, opc_is_ipf, opc_from_k_id,
                        opc_memo, opc_is_online, opc_status, opc_from_ol_id, opc_is_delete,
                        opc_from_un_id, opc_datetime, opc_add_info)
  select ol_date, ol_doc_no, reg_id, coalesce(csp_id, 0), 0,
         ol_doc_sign, ob_summa_plus + ob_summa_minus, ol_document_client_id, ol_is_ipf, ol_main_k_id,
         ol_memo, 0, 1, ol_id, ol_is_delete,
         ol_user_id, ol_datetime, loan_count
  from hp_opercash_history_rep(:rep_region_id, :rep_date_begin, :rep_date_end)
  order by reg_id, ol_date, ol_id;
end
