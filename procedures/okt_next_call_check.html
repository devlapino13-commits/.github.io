CREATE PROCEDURE OKT_NEXT_CALL_CHECK(
    K_ID ID_TYPE,
    OBJ_KIND STR_100,
    OBJ_ID ID_TYPE,
    NEXT_CALL TIMESTAMP,
    ASKED_CALLBACK SMALLINT,
    REP_DATE DATE)
RETURNS (
    RES_ERROR STR_255,
    RES_NEXT_CALL DATE)
AS
declare variable c_id id_type;
declare variable poruch_id id_type;
declare variable cc_id id_type;
declare variable rep_date_begin date;
declare variable first_called_date date;
declare variable res_next_call_str varchar(10);
declare variable db_id integer;
declare variable k_fact_disb_date date;
begin
  c_id = null;
  poruch_id = null;
  cc_id = null;

  res_error = '';
  asked_callback = coalesce(asked_callback, 0);

  if (asked_callback = 1) then
  begin
    suspend;
    exit;
  end

  :db_id = f_get_db_id();
  if (:db_id <> 81001) then
  begin
    suspend;
    exit;
  end

  select kont.k_fact_disb_date
  from kontract kont
  where kont.k_id = :k_id
  into :k_fact_disb_date;

  if (k_fact_disb_date > '05.12.2016') then
  begin
    suspend;
    exit;
  end

  rep_date = coalesce(rep_date, current_date);
  rep_date_begin = rep_date - 6;

  if (obj_kind = 'Clients') then
  begin
-- если не найден среди участников группы, значит это поручитель
    if (exists(select *
               from knclient kncl
               where kncl.kc_from_k_id = :k_id and
                     kncl.kc_from_c_id = :obj_id and
                     kncl.kc_amount_approved > 0)) then
      c_id = :obj_id;
    else
      poruch_id = :obj_id;
  end
  else
  if (obj_kind = 'Contacts') then
    cc_id = :obj_id;

  if (c_id is not null) then
  begin
-- получаю первый успешный звонок за последние 7 дней
    select pcl.pcl_called_date
    from pantera_clients pcl
    inner join refdata rd on rd.rd_id = pcl.pcl_rd_res
    where pcl.pcl_called_date between :rep_date_begin and :rep_date and
          pcl.pcl_from_c_id = :c_id and
          rd.rd_tag > 0 -- только успешные звонки
    order by pcl_called_datetime
    rows 1
    into :first_called_date;
  end

  if (poruch_id is not null) then
  begin
-- звонки поручителям
    select pcon.pcon_called_date
    from pantera_contacts pcon
    inner join refdata rd on rd.rd_id = pcon.pcon_rd_res
    where pcon.pcon_called_date between :rep_date_begin and :rep_date and
          pcon.pcon_from_poruch_id = :poruch_id and
          rd.rd_tag > 0 -- только успешные звонки
    order by pcon_called_datetime
    rows 1
    into :first_called_date;
  end

  if (cc_id is not null) then
  begin
-- звонки контактным лицам
    select pcon.pcon_called_date
    from pantera_contacts pcon
    inner join refdata rd on rd.rd_id = pcon.pcon_rd_res
    where pcon.pcon_called_date between :rep_date_begin and :rep_date and
          pcon.pcon_from_cc_id = :cc_id and
          rd.rd_tag > 0 -- только успешные звонки
    order by pcon_called_datetime
    rows 1
    into :first_called_date;
  end

  if (first_called_date is null) then -- если это первый успешный звонок за 7 дней
    res_next_call = rep_date + 1;
  else -- если это второй успешный звонок за 7 дней
    res_next_call = first_called_date + 7;

  if (cast(next_call as date) < res_next_call) then
  begin
    res_next_call_str = replace(iif(extract(day from res_next_call) < 10, '0', '') || extract(day from res_next_call) || '.' || iif(extract(month from res_next_call) < 10, '0', '') || extract(month from res_next_call) || '.' || extract(year from res_next_call), ' ', '');

    res_error = 'Ближайший звонок не может быть совершен раньше ' || res_next_call_str;
  end

  suspend;
end
