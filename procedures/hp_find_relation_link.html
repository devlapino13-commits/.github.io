CREATE PROCEDURE HP_FIND_RELATION_LINK(
    C_ID D_ID)
RETURNS (
    REA_ID D_ID,
    C_NAME_FIO_LO D_STR_100,
    C_NAME_FIO_CL D_STR_100,
    REL_NAME D_STR_CODE,
    REA_DATE DATE)
AS
declare variable sba_from_k_id d_id;
declare variable rae_from_lo_id d_id;
begin

for select rea.rea_id, rea.rea_date, rea.rae_from_lo_id
      from relatives_active_employees rea
      where rea.rae_from_c_id = :c_id and
            rea.rea_is_delete = 0
      into :rea_id, rea_date,rae_from_lo_id
  do
  begin
    c_name_fio_cl = '';
    c_name_fio_lo = '';
    rel_name = '';


    select cl.c_name_fio
    from clients cl
    where cl.c_id =:c_id
    into: c_name_fio_cl;

    select c_name_fio, rel_name
    from hp_client_relation_list(:c_id, 1)
    where crel_other_c_id = :rae_from_lo_id
    into:c_name_fio_lo, rel_name;

    suspend;
  end
end
