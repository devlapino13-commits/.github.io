CREATE PROCEDURE GET_MC_AGREEMENT_URL(
    DOCUMENT D_STR_CODE NOT NULL,
    REP_DATE D_DATE NOT NULL,
    LANG D_STR_LANG NOT NULL = '')
RETURNS (
    URL D_STR_2048)
AS
declare variable DB_ID D_ID;
declare variable VER D_STR_20;
begin
  if (document not in ('mobile_app_use', 'personal_data_processing')) then
    exception e_message 'Документ: ['||:document||'] не из списка допустимых';

  db_id = f_get_db_id();

  if ((db_id in (39801, 41701)) and (LANG not in ('ru', 'kg'))) then
    exception e_message 'Язык: ['||lang||'] не определен';

  select t.ama_version
  from agreements_mc_adj t
  where t.ama_document = f_enum_id('mc_document', :document)
  and :rep_date between t.ama_date_begin and t.ama_date_end
  into :ver;

  if (ver is null) then
    exception e_message 'Версия документа не определена';

  if (db_id in (39801, 41701)) then
  begin
    if (document = 'mobile_app_use') then
        if (lang = 'ru') then
            url = 'https://mbulak.kg/ru/mp-reg-rules/v'||ver;
        else if (lang = 'kg') then
            url = 'https://mbulak.kg/mp-reg-rules/v'||ver;

    if (document = 'personal_data_processing') then
        if (lang = 'ru') then
            url = 'https://mbulak.kg/ru/mp-data-agree/v'||ver;
        else
            url = 'https://mbulak.kg/mp-data-agree/v'||ver;
  end
  else if (db_id = 81001) then
  begin
    if (document = 'mobile_app_use') then
        url = 'https://www.molbulak.ru/mc_registration_rules/v'||ver;
    else if (document = 'personal_data_processing') then
        url = 'https://www.molbulak.ru/docsmp/soglasie-na-obrabotku/v'||ver;
  end
  if (db_id = 94401) then
  begin
    if (document = 'mobile_app_use') then
        url = 'https://www.mbulak.az/mp-reg-rules/v'||ver;
    else if (document = 'personal_data_processing') then
        url = 'https://www.mbulak.az/mp-data-agree/v'||ver;
  end
  suspend;
end
