CREATE PROCEDURE AAREP_RATING_LO_TMP_OWN(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_REGION_ID INTEGER,
    AA_KC_ID DECIMAL(18,0),
    AA_TOTAL_BALANCE DECIMAL(18,4),
    AA_TOTAL_CLIENT INTEGER,
    AA_DELINQ_BALANCE DECIMAL(18,4),
    AA_DELINQ_CLIENT INTEGER,
    AA_DISB_CL_KOLICH INTEGER)
AS
declare variable aa_curr_kc_id decimal(18,0);
declare variable aa_disb_kc_id decimal(18,0);
declare variable aa_is_own integer;
declare variable aa_delinq_begin_date date;
declare variable dd_disb_date date;
declare variable curr_currency varchar(3);
declare variable kc_id_prior decimal(18,0);
declare variable dd date;
begin
  /* подсчет баланса кредита, просроченных и выдач для расчета рейтинга
  в расчете берет не офис действия, а офис где работает сотрудник
  Sanjari, 05.07.2012 */
  kc_id_prior = 0;
  aa_disb_cl_kolich = 0;
  select cm_default_currency
  from company
  into :curr_currency;

  for select dk_disb_lo_id, dk_balance_end, dk_client_kol, dk_delinq_balance, dk_delinq_cl_kolich, dk_disb_date,
             dk_date
      from aarep_rating_deliq_temp(:rep_region_id, :rep_date_end)
      into :aa_disb_kc_id, :aa_total_balance, :aa_total_client, :aa_delinq_balance, :aa_delinq_client, :dd_disb_date,
           :aa_delinq_begin_date
  do
  begin
    aa_curr_kc_id = aa_disb_kc_id;
    aa_kc_id = aa_curr_kc_id;
    if (aa_curr_kc_id <> aa_disb_kc_id) then
      aa_is_own = 0;
    else
      aa_is_own = 1;

    if (aa_kc_id <> kc_id_prior) then
    begin
      select kal_new_reg_id, kal_date
      from knaktlist
      where is_delete = 0 and
            kal_new_kf_officer = :aa_kc_id
      order by kal_date desc
      rows 1
      into :aa_curr_region_id, :dd;
      kc_id_prior = aa_kc_id;
    end
    if (curr_currency = '810') then
    begin
      if (dd_disb_date < '01.04.2012') then
      begin
        aa_delinq_balance = 0;
        aa_delinq_client = 0;
      end
      if (aa_is_own = 0) then
      begin
        aa_delinq_balance = 0;
        aa_delinq_client = 0;
      end
    end
    if (curr_currency = '417') then
    begin
      if (aa_is_own = 0) then
      begin
        aa_delinq_balance = 0;
        aa_delinq_client = 0;
      end
      if ((rep_date_end - aa_delinq_begin_date) <= 7) then
      begin
        aa_delinq_balance = 0;
        aa_delinq_client = 0;
      end
    end
    suspend;
  end

  aa_delinq_balance = 0;
  aa_delinq_client = 0;
  aa_total_balance = 0;
  aa_total_client = 0;

  kc_id_prior = 0;
  for select ks_kc_id, k_kolich_client_approved
      from kontract, kontstate, region_list(:rep_region_id, 1, -1, 1)
      where k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
            ks_state = 1 and
            ks_region_id = id_list
      into :aa_kc_id, :aa_disb_cl_kolich
  do
  begin
    if (aa_kc_id <> kc_id_prior) then
    begin
      select kal_new_reg_id, kal_date
      from knaktlist
      where is_delete = 0 and
            kal_new_kf_officer = :aa_kc_id
      order by kal_date desc
      rows 1
      into :aa_curr_region_id, :dd;
      kc_id_prior = aa_kc_id;
    end
    suspend;
  end
end
