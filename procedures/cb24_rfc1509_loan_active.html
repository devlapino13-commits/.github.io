CREATE PROCEDURE CB24_RFC1509_LOAN_ACTIVE(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    LOAN_ID D_ID,
    LOAN_UID D_STR_CODE,
    LOAN_DEAL_DATE D_DATE,
    LOAN_FACT_TRANSFER_DATE D_DATE,
    LOAN_SUM D_CURRENCY,
    DUE_ARREAR_PRINCIPAL D_CURRENCY,
    DUE_ARREAR_INTEREST D_CURRENCY,
    LOAN_CURR D_STR_10,
    LOAN_RATE D_PERCENT,
    LOAN_RATE_DISCOUNTED D_PERCENT,
    LOAN_PSK D_PERCENT,
    LOAN_AVG_PAYMENT D_CURRENCY,
    LOAN_DEAL_END_DATE D_DATE,
    LOAN_FACT_END_DATE D_DATE,
    LOAN_RESTRUCT_UID D_STR_CODE,
    LOAN_RESTRUCT_DATE_FIRST D_DATE,
    LOAN_RESTRUCT_DATE_LAST D_DATE,
    LOAN_RESTRUCT_END_DATE D_DATE,
    SUBJECT_TYPE D_STR_20,
    SUBJECT_ID D_ID,
    SUBJECT_PRIVATE_FULLNAME D_STR_100,
    SUBJECT_PRIVATE_BIRTH_DATE D_DATE,
    SUBJECT_PRIVATE_COUNTRY_CODE D_STR_20,
    SUBJECT_PRIVATE_DOC_SERIAL D_STR_CODE,
    SUBJECT_PRIVATE_DOC_NUMBER D_STR_CODE,
    SUBJECT_LEGAL_FULLNAME D_STR_255,
    SUBJECT_LEGAL_INN D_STR_20,
    SUBJECT_LEGAL_OGRN D_STR_20,
    RES_KONTRACT_NO D_STR_20,
    RES_DOC_ID D_ID,
    RES_PK_NAME D_STR_100,
    LOAN_IS_ONLINE D_BOOL,
    LOAN_IS_ZALOG D_BOOL,
    LOAN_IS_PROUCH D_BOOL)
AS
declare variable KC_DISB_PASSPORT_ID D_INTEGER;
declare variable PK_ID D_INTEGER;
declare variable CDS_DOC_NO D_STR_CODE;
declare variable K_ID D_ID;
declare variable DB_ID D_INTEGER;
declare variable KS_FROM_OL_ID D_ID;
declare variable SH_TOTAL_PERC D_CURRENCY;
declare variable MONTHS D_INTEGER;
declare variable K_AMOUNT_APPROVED D_CURRENCY;
declare variable DATE_SUD_RIZ D_DATE;
declare variable SAB_ID D_INTEGER;
declare variable PORUCH_SALDO D_CURRENCY;
begin
  subject_type = 'ФЛ';
  subject_legal_fullname = '';
  subject_legal_inn = '';
  subject_legal_ogrn = '';
  loan_curr = 'руб.';

  db_id = f_get_db_id();

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Поручительство'
  into sab_id;

  for select ks_from_k_id, ks_from_ol_id, ks_date_begin, k_kontract_no, k_kont_last_date, ks_date_end,
             round(k_percent_per_day * 365.000, 3), k_percent_effect_fifd, k_restruct_date, k_kredline_end,
             k_fix_percent_amount, k_amount_approved, k_fact_disb_online, iif((pr_code containing '.ZALOG_CAR.' or pr_code containing '.zalog.'), 1, 0)
  from kontstate, kontract, product, region_list(:rep_region_id, 1, 3, 1)
  where ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_date_begin <= :rep_date_end and
        ks_date_end >= :rep_date_end and
        ks_from_k_id = k_id and
        k_from_pr_id = pr_id and
        ks_region_id + 0 = id_list
  into k_id, ks_from_ol_id, loan_deal_date, res_kontract_no, loan_deal_end_date, loan_fact_end_date,
       loan_rate, loan_psk, loan_restruct_date_first, loan_restruct_end_date,
       sh_total_perc, k_amount_approved, loan_is_online, loan_is_zalog
  do
  begin
    loan_fact_transfer_date = loan_deal_date;

    if (loan_fact_end_date >= '31.12.2999') then
      loan_fact_end_date = null;
    else
      loan_fact_end_date = loan_fact_end_date + 1;

    loan_rate_discounted = loan_rate;
    select round(kd_percent_per_day * 365.000, 3)
    from kontdata
    where kd_from_k_id = :k_id and
          kd_from_ol_id = :ks_from_ol_id and
          kd_ol_date = :loan_deal_date + 1
    into loan_rate_discounted;

    loan_avg_payment = 0;
    months = ceil(cast(datediff(day, loan_deal_date, loan_deal_end_date) as d_currency) / (cast(365.0 as d_currency) / cast(12.0 as d_currency)));

    sh_total_perc = 0;
    select sum(sh_perc)
    from hp_get_base_schedule(:k_id)
    into sh_total_perc;

    if (sh_total_perc is null) then
      sh_total_perc = 0;

    if(months > 0) then
      loan_avg_payment = (k_amount_approved + sh_total_perc) / cast(months as d_currency);
    else loan_avg_payment = (k_amount_approved + sh_total_perc);

    if (loan_avg_payment is null) then
      loan_avg_payment = 0;

    loan_avg_payment = round(loan_avg_payment, 2);

    loan_restruct_uid = '';
    loan_restruct_date_last = null;

    if (loan_restruct_date_first is not null) then
    begin
      select restruct_date_last
      from aarep_get_restructuring(:k_id, :rep_date_end, :db_id)
      into loan_restruct_date_last;

      if (loan_restruct_date_last is null) then
        loan_restruct_date_last = loan_restruct_date_first;
    end else
      loan_restruct_end_date = null;

    for select kc_id, kc_disb_passport_id, kc_from_c_id, c_name_fio, c_birth_date, kc_amount_approved
    from knclient, clients
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = c_id
    into loan_id, kc_disb_passport_id, subject_id, subject_private_fullname, subject_private_birth_date, loan_sum
    do
    begin
      loan_uid = f_uuid_cbr(f_uuid('knclient', loan_id));

      if (loan_restruct_date_first is not null) then
        loan_restruct_uid = loan_uid;

      due_arrear_principal = 0;
      due_arrear_interest = 0;

      loan_is_prouch = 0;
      poruch_saldo = f_get_acc_amount(:loan_id, :sab_id, :rep_date_end, :rep_date_end, 'bal');

      if (poruch_saldo <> 0) then
        loan_is_prouch = 1;

      select oks_balance, oks_bal_perc
      from kn_oksummary
      where oks_from_kc_id = :loan_id and
            oks_date_begin <= :rep_date_end
      order by oks_from_kc_id desc, oks_date_begin desc
      rows 1
      into due_arrear_principal, due_arrear_interest;

      date_sud_riz = null;

--      select date_sud_riz, -- Реализация имущества завершена
--      from hp_get_kontur_ext_info(:subject_id, :loan_deal_date, :rep_date_end)
--      into date_sud_riz;


      if (date_sud_riz is not null) then
      begin
        if (loan_fact_end_date is null) then
          loan_fact_end_date = date_sud_riz;
        else
          loan_fact_end_date = minvalue(loan_fact_end_date, date_sud_riz);
      end

      res_doc_id = null;
      subject_private_doc_serial = '';
      subject_private_doc_number = '';
      subject_private_country_code = '';
      res_pk_name = '';

      if (kc_disb_passport_id is not null) then
      begin
        res_doc_id = kc_disb_passport_id;

        select cds_doc_no, pk_id, pk_name, left(pk_country, 2)
        from clientdocs, passkind
        where cds_id = :res_doc_id and
              cds_from_pk_id = pk_id
        into cds_doc_no, pk_id, res_pk_name, subject_private_country_code;

        select res_doc_seria, res_doc_number
        from hp_passport_serial_num(:cds_doc_no, :pk_id)
        into subject_private_doc_serial, subject_private_doc_number;
      end

      suspend;
    end
  end
end
