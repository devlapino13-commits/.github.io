CREATE PROCEDURE HP_SMSDIALOG_ACTION_BALANCE(
    C_ID DECIMAL(18,0))
RETURNS (
    SMS_TEXT VARCHAR(255),
    HTML_BLOB BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable uniq_acc varchar(50);
declare variable sh_date date;
declare variable sh_sum curr_type;
declare variable k_id id_type;
declare variable ks_date_end date;
declare variable okr_sum curr_type;
declare variable all_zad_sum integer;
declare variable txt1 varchar(100);
declare variable sh_date_txt varchar(20);
declare variable k_kontract_no varchar(50);
declare variable sh_date_tmp date;
declare variable sh_sum1 curr_type;
declare variable sh_sum2 curr_type;
declare variable has_termination smallint;
declare variable is_delinq integer;
declare variable txt1_ru str_255;
declare variable txt2 str_255;
declare variable has_problem_1 smallint;
declare variable delinq_sum curr_type;
declare variable full_debt curr_type;
declare variable c_fio str_100;
declare variable full_debt_total curr_type;
declare variable undelinq_sh_date date;
declare variable undelinq_sh_date_txt str_100;
declare variable delinq_days integer;
declare variable cm_name str_255;
begin
  select cm.cm_name
  from company cm
  into :cm_name;

  sms_text = '';
  if (c_id is null) then
  begin
    sms_text = 'Ваш номер не зарегистрирован в системе МолБулак';
    html_blob = '<b>Номер не зарегистрирован в системе МолБулак</b>';
    suspend;
    exit;
  end

  uniq_acc = -1;

  select uniq_acc
  from hp_show_clientuniq_acc_no(:c_id)
  into :uniq_acc;
  if (uniq_acc = -1) then
  begin
    -- присваиваю лицевой счет если он ранее не был присовен
    select uniq_acc_str
    from hp_fix_uniq_acc(:c_id)
    into :uniq_acc;
  end

  html_blob = '<b>Лицевой счет</b>...' || :uniq_acc || '<br>';

  c_fio = '';

  select cl.c_name_fio
  from clients cl
  where cl.c_id = :c_id
  into :c_fio;

  html_blob = BLOB_APPEND(html_blob, '<b>ФИО</b>..................' || :c_fio || '<br>');

  full_debt_total = 0;

  for select kc_from_k_id, ks_date_end
      from kontstate, knclient
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin <= current_date and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = :c_id
      order by ks_date_begin desc
      into :k_id, :ks_date_end
  do
  begin
    has_termination = 0;

    select k_kontract_no, iif(ol.ol_id is null, 0, 1)
    from kontract kont
    left join operlist ol on ol.ol_id = k_stopped_perc_ol_id and
          ol.is_delete = 0
    where k_id = :k_id
    into k_kontract_no, has_termination;

    sh_date = null;
    sh_sum = 0;
    sh_date_tmp = null;
    is_delinq = 0;

    delinq_sum = 0;
    full_debt = 0;

    undelinq_sh_date = null;

    if (ks_date_end > current_date) then --Если кредит активный
    begin
      for select sh_date, sum(zad_os + sh_calc_perc + sh_calc_fine), -- сумма для смс, на случай просроченной задолженности
                 sum(zad_os + zad_perc) -- сумма для смс, на случай если нет просроченной задолженности
          from aarep_make_zad_list(:k_id, current_date, 1)
          where sh_date is not null
          group by sh_date
          order by sh_date
          into sh_date_tmp, sh_sum1, sh_sum2
      do
      begin
        if (has_termination = 0) then -- если не расторгнут
        begin
          if (sh_date_tmp < current_date) then -- если просрочка (получаю всю просроченную сумму)
          begin
            if (sh_date is null) then
              sh_date = current_date;
            sh_sum = sh_sum + sh_sum1;
          end
          else
          begin
            if (sh_date is null) then -- если нет просрочки (получаю сумму ближайшего платежа)
            begin
              sh_date = sh_date_tmp;
              sh_sum = sh_sum + sh_sum2;
            end
          end
        end
        else
        begin
          if (sh_date is null) then
            sh_date = current_date;
          sh_sum = sh_sum + sh_sum1;
        end
        ---------

        if (sh_date_tmp < current_date) then -- если просрочка (получаю всю просроченную сумму)
        begin
          is_delinq = 1;
          delinq_sum = delinq_sum + sh_sum1;
        end
        else
        begin
          if (undelinq_sh_date is null) then
            undelinq_sh_date = sh_date_tmp;
        end

        full_debt = full_debt + sh_sum1;
      end
    end

    sh_date_txt = right(0 || extract(day from sh_date), 2) || '.' || right(0 || extract(month from sh_date), 2) || '.' || right(extract(year from sh_date), 2);

    sh_sum = coalesce(sh_sum, 0);

    okr_sum = null;
    select sum(oko_os + oko_perc + oko_other + oko_sh_os + oko_sh_perc + oko_sh_other)
    from operkroff okf, operlist ol
    where okf.oko_from_k_id = :k_id and
          okf.oko_from_c_id = :c_id and
          okf.oko_type in (1, 4) and
          okf.oko_from_ol_id = ol.ol_id and
          ol.is_delete = 0 and
          ol.ol_date <= current_date
    into okr_sum;

    okr_sum = coalesce(okr_sum, 0);

    all_zad_sum = cast((sh_sum + okr_sum) as integer);
    full_debt = full_debt + okr_sum;

    full_debt_total = full_debt_total + full_debt;

    if (all_zad_sum > 0) then
    begin
      html_blob = BLOB_APPEND(html_blob, '<br>');
      html_blob = BLOB_APPEND(html_blob, '<b>Договор №</b>....................................' || :k_kontract_no || '<br>');

      if (has_termination = 1) then
        html_blob = BLOB_APPEND(html_blob, '<b>договор расторгнут в связи с нарушением графика платежей' || '<br>');

      if (undelinq_sh_date is not null) then -- если есть непросроченные платежи
      begin
        undelinq_sh_date_txt = right(0 || extract(day from undelinq_sh_date), 2) || '.' || right(0 || extract(month from undelinq_sh_date), 2) || '.' || right(extract(year from undelinq_sh_date), 2);

        --html_blob = BLOB_APPEND(html_blob,'<b>Сумма ближайшего платежа</b>....'||'XXX'||'р. <br>');

        if (has_termination = 0) then -- если не расторгнут
        begin
          html_blob = BLOB_APPEND(html_blob, '<b>Дата ближайшего платежа</b>........' || :undelinq_sh_date_txt || '<br>');
          html_blob = BLOB_APPEND(html_blob, '<b>Сумма ближайшего ежемесячного платежа</b>...' || round(:sh_sum) || ' р. <br>');
        end
        else
          html_blob = BLOB_APPEND(html_blob, '<b>Дата платежа</b>........' || :undelinq_sh_date_txt || '<br>');
      end

      delinq_days = 0;

      if (is_delinq = 1) then
      begin
        select current_date - oks.oks_delinq_date
        from oksummary oks
        where oks.oks_from_k_id = :k_id and
              oks.oks_close_state = 0 and
              current_date between oks.oks_date_begin and oks.oks_date_end and
              current_date > oks.oks_delinq_date
        into :delinq_days;

        html_blob = BLOB_APPEND(html_blob, '<b>Продолжительность просрочки</b>...' || :delinq_days || ' дней/дня <br>');
      end

      html_blob = BLOB_APPEND(html_blob, '<b>Общая сумма задолженности</b>...' || round(full_debt) || ' р.<br>');

      if ((is_delinq = 0) and
          (okr_sum = 0)) then
      begin
        html_blob = BLOB_APPEND(html_blob, '<b>По договору нет активной просроки.</b><br>');
        html_blob = BLOB_APPEND(html_blob, '<b>В случае полного досрочного закрытия долга требуется заявление, иначе штраф</b>');
      end

      if (is_delinq = 1) then
      begin
        html_blob = BLOB_APPEND(html_blob, '<b>Заем просрочен.</b><br>');
      end

      --html_blob = BLOB_APPEND(html_blob,'<b>Сумма просрочки</b>........................'||round(:delinq_sum )||'р.<br>');

      if (txt1 is null) then
      begin
        /*1)    «Лиц.счет__. Долг на ДД/ММ/ГГг._____руб. ООО «МКК М Булак» тел. 88005559348»
          (высылается в следующих случаях:
          1) если договор расторгнут;
          2) если задолженность списана на «вне баланс»;
          3) если договор не расторгнут, но по нему есть активная просрочка);*/

        has_problem_1 = 0;

        if ((is_delinq = 1) or (coalesce(okr_sum, 0) <> 0) or (has_termination = 1)) then
        begin
          sms_text = 'Лиц.счет' || :uniq_acc || '. Долг на ' || :sh_date_txt || 'г.' || :all_zad_sum || 'руб. ' || :cm_name || ' тел. 88005559348';
          txt1 = 'Дог.№' || :k_kontract_no || ', долг на ' || :sh_date_txt || 'г.' || :all_zad_sum || 'руб.';
          txt1_ru = txt1;
          has_problem_1 = 1;
        end
        else
        /*2)    «Lits.schet___. Blijaishii platej ДД/ММ/ГГg. _____ rub.»
          (если клиент не имеет просроченной задолженности перед Компанией).*/
        begin
          sms_text = 'Lits.schet' || :uniq_acc || '. Blijaishee pogashenie ' || :sh_date_txt || 'g. ' || replace(:all_zad_sum, '.', ',') || ' rub.';
          txt1 = 'Dog. №' || :k_kontract_no || '. Blijaishee pogashenie ' || :sh_date_txt || '. ' || replace(:all_zad_sum, '.', ',') || ' rub.';
          txt1_ru = 'Дог.№' || :k_kontract_no || ', ближ. платеж ' || :sh_date_txt || '. ' || :all_zad_sum || ' руб.';
        end
      end
      else
      begin
        if ((is_delinq = 1) or (coalesce(okr_sum, 0) <> 0) or (has_termination = 1)) then
        begin
          txt2 = 'Дог.№' || :k_kontract_no || ', долг на ' || :sh_date_txt || 'г.' || :all_zad_sum || 'руб.';

          sms_text = 'Лиц.счет:' || :uniq_acc || '. ' || :txt1_ru || ' ' || :txt2 || ' ' || :cm_name || ' тел. 88005559348';
        end
        else
        begin
          if (has_problem_1 = 0) then
          begin
            txt2 = 'Dog. №' || :k_kontract_no || ' blijaishii pogashenie ' || :sh_date_txt || '. ' || replace(:all_zad_sum, '.', ',') || ' rub.';
            sms_text = 'Lits.schet' || :uniq_acc || '. ' || :txt1 || ' ' || :txt2;
          end
          else
          begin
            txt2 = 'Дог.№' || :k_kontract_no || ', ближ. платеж ' || :sh_date_txt || '. ' || :all_zad_sum || ' руб.';
            sms_text = 'Лиц.счет:' || :uniq_acc || '. ' || :txt1_ru || ' ' || :txt2 || ' ' || :cm_name || ' тел. 88005559348';
          end
        end
      end

      html_blob = BLOB_APPEND(html_blob, '<br>');
    end
  end

  if ((full_debt_total <> 0) and
      (coalesce(txt2, '') <> '')) then
    html_blob = BLOB_APPEND(html_blob, '<br><b>ИТОГО задолженность, по всем договорам</b>...' || round(full_debt_total) || ' р. <br>');

  if (sms_text = '') then
    sms_text = 'Активный заем не обнаружен';

  suspend;
end
