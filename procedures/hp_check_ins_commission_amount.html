CREATE PROCEDURE HP_CHECK_INS_COMMISSION_AMOUNT(
    K_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable k_amount_approved d_currency;
declare variable n_value d_integer;
declare variable pr_code d_str_255;
declare variable commission_amount d_currency;
declare variable mbulak_share d_currency;
declare variable comission d_currency;
declare variable mbulak d_currency;
declare variable k_disb_date d_date;
begin
    res_text = '';
    for select k.k_amount_approved, k.k_kredline_end - k.k_disb_date, pr.pr_code, k.k_disb_date
    from kontract k
    inner join product pr on pr.pr_id = k.k_from_pr_id and
        pr.pr_code not containing '.ignore_insurance.' -- RFCKG-5377
    inner join insurance_kontract ik on ik.ik_from_k_id = k.k_id
    where k.k_id = :k_id
    into k_amount_approved, n_value, pr_code, k_disb_date
    do
    begin
      select commission_amount, mbulak_share
      from calc_insurance_amount(:k_amount_approved, :n_value, :pr_code, :k_disb_date, :k_id)
      into commission_amount, mbulak_share;

      select coalesce(ik.ik_commission, 0), coalesce(ik.ik_mbulak_share, 0)
      from insurance_kontract ik
      where ik.ik_from_k_id = :k_id
      into comission, mbulak;

      if (comission <> commission_amount or mbulak_share <> mbulak) then
      begin
        res_text = 'Ставки по страхованию поменялись, надо заново пере отправить заявлению на страхование';
        suspend;
      end
    end
end
