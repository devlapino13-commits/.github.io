CREATE PROCEDURE AAREP_LOAN_PORTFOLIO(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    KAL_NEW_KF_NAME VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    C_NAME_FIO VARCHAR(255),
    K_DISB_DATE DATE,
    KREDLINE_END DATE,
    K_PERIOD DECIMAL(15,2),
    K_PERCENT_EFFECT DECIMAL(18,7),
    K_PERCENT DECIMAL(18,7),
    PR_LOAN_TYPE_STR VARCHAR(20),
    PR_IS_SECURED_STR VARCHAR(20),
    K_ADD_MONTH D_STR_20,
    KC_AMOUNT_APPROVED CURR_TYPE,
    BALANCE CURR_TYPE,
    LKC_STATUS_STR VARCHAR(50),
    LKC_LLC_SUM CURR_TYPE,
    LKC_DELIQ_DAYS INTEGER,
    BALANCE_PROS_CL CURR_TYPE,
    KNC_REP_DELIQ_OS_CL CURR_TYPE,
    KNC_REP_DELIQ_PERC_CL CURR_TYPE,
    AAREP_IS_RESTRUCTED INTEGER,
    OFFCE_NAME VARCHAR(50),
    BT_NAME VARCHAR(50),
    TYPE_FINANCING VARCHAR(20),
    K_PERCENT_PER_DAY DECIMAL(18,7),
    KNC_FINE CURR_TYPE,
    C_INN VARCHAR(20),
    K_DISB_KINDD D_STR_100,
    K_BY_RATING D_STR_CODE)
AS
declare variable pr_loan_type integer;
declare variable pr_is_secured integer;
declare variable lkc_from_k_id id_type;
declare variable close_ol_date d_date;
declare variable lkc_from_c_id id_type;
declare variable lkc_from_k_id_old d_id;
declare variable fine decimal(18,4);
declare variable share_k decimal(18,8);
declare variable oks_delinq_os curr_type;
declare variable oks_delinq_perc curr_type;
declare variable k_amount_approved curr_type;
declare variable oks_balance curr_type;
declare variable k_disb_kind d_integer;
begin
  lkc_from_k_id_old = 0;

  for select aal.lkc_from_c_id, aal.lkc_from_k_id, aal.k_kontract_no, aal.c_name_fio, aal.k_disb_date,
             aal.lkc_status_str, aal.lkc_llc_sum, aal.lkc_deliq_days, aal.offce_name, aal.bt_name, aal.type_financing,
             aal.c_inn
      from aarep_llc_clients_reports(:rep_region_id, :rep_date_end) aal
      into lkc_from_c_id, lkc_from_k_id, k_kontract_no, c_name_fio, k_disb_date, lkc_status_str, lkc_llc_sum,
           lkc_deliq_days, offce_name, bt_name, type_financing, c_inn
  do
  begin
    if (lkc_from_k_id <> lkc_from_k_id_old) then
    begin
      lkc_from_k_id_old = lkc_from_k_id;
      select knl.kal_new_kf_name, k.k_kredline_end, k.k_period, k.k_percent_effect,
             k.k_percent, pr.pr_loan_type, pr.pr_is_secured,
             case k.k_add_month
               when 0 then '0 мес.'
               when 1 then '1 мес.'
               when 2 then '2 мес.'
             end,
             k.k_percent_per_day, k.k_amount_approved, k.k_disb_kind,
             iif(k.k_by_rating = 1, 'По кредитному рейтингу', 'Стандартная выдача')
      from kontract k
      inner join knaktlist knl on knl.kal_from_kf_id = k.k_from_kf_id and
            knl.is_delete = 0 and
            :rep_date_end between knl.kal_date and knl.kal_date_end
      inner join product pr on pr.pr_id = k.k_from_pr_id
      where k.k_id = :lkc_from_k_id
      into kal_new_kf_name, kredline_end, k_period, k_percent_effect, k_percent, pr_loan_type,
           pr_is_secured, k_add_month, k_percent_per_day, k_amount_approved, k_disb_kind, k_by_rating;

      k_disb_kindd = case :k_disb_kind when 6 then 'На карту Мбулак'
                                     when 5 then 'На карту клиента'
                                     else 'Касса'
                                     end;

      select kd_percent_per_day
      from kontdata
      where kd_ol_is_delete = 0 and
            :rep_date_end between kd_ol_date and kd_date_end and
            kd_from_k_id = :lkc_from_k_id
      into k_percent_per_day;

      pr_loan_type_str = '';
      if (pr_loan_type = 0) then
        pr_loan_type_str = 'группа';
      if (pr_loan_type = 1) then
        pr_loan_type_str = 'индив.';

      pr_is_secured_str = '';
      if (pr_is_secured = 0) then
        pr_is_secured_str = 'без залога';
      if (pr_is_secured = 1) then
        pr_is_secured_str = 'залог';
      if (pr_is_secured = 2) then
        pr_is_secured_str = 'поручители';
      if (pr_is_secured = 3) then
        pr_is_secured_str = 'залог,поруч.';
      -- баланс
      oks_balance = null;
      select oks.oks_balance
      from oksummary oks
      where oks_from_k_id = :lkc_from_k_id and
            oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_close_state = 0
      into oks_balance;

      oks_delinq_os = null;
      oks_delinq_perc = null;
      fine = null;
      select sum(sum_fine_base_os), sum(iif(sh_date < :rep_date_end, sh_calc_perc, 0)), sum(sh_calc_fine)
      from aarep_make_zad_list(:lkc_from_k_id, :rep_date_end, 1)
      into oks_delinq_os, oks_delinq_perc, fine;

      if (exists(select *
                 from aarep_get_restructuring(:lkc_from_k_id, :rep_date_end, 41701)
                 where is_restruct = 1)) then
        aarep_is_restructed = 1;
      else
        aarep_is_restructed = 0;

      for select kc_amount_approved, ol_date
          from knclient kc
          left join operlist ol on ol.ol_id = kc.kc_closed_ol_id and
                ol.is_delete = 0 and
                ol.ol_date <= :rep_date_end
          where kc_from_k_id = :lkc_from_k_id and
                kc_amount_approved > 0
          into :kc_amount_approved, :close_ol_date
      do
      begin
        if (close_ol_date is not null) then
          k_amount_approved = k_amount_approved - kc_amount_approved;
      end
    end

    close_ol_date = null;
    kc_amount_approved = null;

    for select kc_amount_approved, ol_date
        from knclient kc
        left join operlist ol on ol.ol_id = kc.kc_closed_ol_id and
              ol.is_delete = 0 and
              ol.ol_date <= :rep_date_end
        where kc_from_k_id = :lkc_from_k_id and
              kc_amount_approved > 0 and
              kc.kc_from_c_id = :lkc_from_c_id
        into kc_amount_approved, close_ol_date
    do
    begin
      if (close_ol_date is null) then
      begin
        --доля
        share_k = 1.0;
        share_k = kc_amount_approved / k_amount_approved;

        balance = (oks_balance * share_k);
        knc_fine = round(fine * share_k);
        if (lkc_deliq_days > 0) then
        begin
          balance_pros_cl = round(oks_balance * share_k);
          knc_rep_deliq_os_cl = round(oks_delinq_os * share_k);
          knc_rep_deliq_perc_cl = round(oks_delinq_perc * share_k);
        end
        else
        begin
          balance_pros_cl = 0;
          knc_rep_deliq_os_cl = 0;
          knc_rep_deliq_perc_cl = 0;
        end
      end
      else
      begin
        balance = 0.0;
        balance_pros_cl = 0.0;
        knc_rep_deliq_os_cl = 0.0;
        knc_rep_deliq_perc_cl = 0.0;
        knc_fine = 0.0;
      end
      suspend;
    end
  end
end
