CREATE PROCEDURE HP_GET_KN_CLIENT_RESIDENT(
    KC_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    NON_RESIDENT SMALLINT,
    CLIENT_TYPE SMALLINT)
AS
begin
  select knr_non_resident
  from kn_resident
  where knr_from_kc_id = :kc_id and
        knr_date_begin <= :rep_date and
        knr_date_end >= :rep_date
  into non_resident;

  select iif(:non_resident is not null, :non_resident, kc_non_resident), kc_client_type
  from knclient
  where kc_id = :kc_id
  into :non_resident, client_type;
  if (non_resident is null) then
    non_resident = 0;
  if (client_type is null) then
    client_type = 3; -- Физ.лицо

  suspend;
end
