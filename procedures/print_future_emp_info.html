CREATE PROCEDURE PRINT_FUTURE_EMP_INFO(
    PERSONAL_ID ID_TYPE NOT NULL)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_PASS_WHO VARCHAR(255),
    C_PASS_WHO_CODE VARCHAR(20),
    C_PASS_DATE DATE,
    C_ADDRESS_YUR VARCHAR(255),
    OFFICE_NAME VARCHAR(50),
    OFFICE_LEGAL_ADDRESS VARCHAR(255),
    OFFICE_FACT_ADDRESS VARCHAR(255),
    OFFICE_NAME_CITY VARCHAR(255))
AS
declare variable reg_id id_type;
declare variable reg_address_legal id_type;
declare variable reg_address_fact id_type;
begin
  reg_id = 1000001;
  select c_name_fio, coalesce(c_pass_no, 'Не указан') as c_pass_no,
         coalesce(c_pass_who, 'Не указан') as c_pass_who_code,
         coalesce(c_pass_who_code, 'Не указан') as c_pass_who_code,
         coalesce(c_pass_date, cast('01.01.1099' as date)) as c_pass_date
  from clients cl
  where c_id = :personal_id
  into c_name_fio, c_pass_no, c_pass_who, c_pass_who_code, c_pass_date;

  select adl_address_str
  from addrclient acl, addrlist al
  where :personal_id = acl.adc_from_c_id and
        acl.adc_address_type = 1 and
        acl.adc_from_adl_id = al.adl_id
  into c_address_yur;

  c_address_yur = coalesce(c_address_yur, 'Не указан');

  select reg_name, reg_stat_rep_region, reg.reg_address_legal, reg_address_fact
  from region reg
  where reg_unicode = :reg_id
  into office_name, office_name_city, reg_address_legal, reg_address_fact;

  office_name_city = coalesce(office_name_city, 'г. Москва');

  select adl_address_str
  from addrlist adl1
  where adl1.adl_id = :reg_address_legal
  into :office_legal_address;

  office_legal_address = coalesce(office_legal_address, 'Не указан');

  select adl_address_str
  from addrlist adl2
  where adl2.adl_id = :reg_address_fact
  into :office_fact_address;

  office_fact_address = coalesce(office_fact_address, 'Не указан');

  suspend;
end
