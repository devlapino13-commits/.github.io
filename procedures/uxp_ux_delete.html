CREATE PROCEDURE UXP$UX_DELETE(
    UX_UUID D_UUID,
    FILL_DEPEND D_BOOL,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    RES_UUID_STR D_STR_CODE,
    RES_DELETED D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable form_count d_integer;
declare variable ufm_ux_uuid d_uuid;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  res_deleted = 1;
  res_text = 'Объект успешно удален';
  res_uuid_str = '';

  select ufm_ux_uuid
  from ux$form
  where ufm_uuid = :ux_uuid
  into ufm_ux_uuid;

  form_count = 0;
  if (ufm_ux_uuid is not null) then
  begin
    select count(*)
    from ux$form
    where ufm_ux_uuid = :ufm_ux_uuid
    into form_count;
  end else
  begin
    if (exists(select * from ux$frame
               where ufr_uuid = :ux_uuid)) then
    begin
      delete from ux$frame
      where ufr_uuid = :ux_uuid;

      res_text = 'Фрейм успешно удален';

      suspend;
      exit;
    end
  end

  if (exists(select * from ux$uniq
  where uu_id = :ux_uuid)) then
  begin
    if (fill_depend = 1) then
      execute procedure uxp$uxdepend_fill(:ux_uuid, 1);

    if (exists(select * from ux$depend
    where udp_uuid = :ux_uuid and
          (udp_lock_delete = 1 or udp_child_locked = 1))) then
    begin
      res_deleted = 0;
      res_uuid_str = uuid_to_char(ux_uuid);
      res_text = 'Имеются зависимые объекты, запрет на удаление.';
    end
  end

  if (res_deleted = 1) then
  begin
    delete from ux$uniq
    where uu_id = :ux_uuid;
  
    delete from ux$depend
    where udp_uuid = :ux_uuid;

    if (form_count = 1) then
    begin
      delete from ux$uniq
      where uu_id = :ufm_ux_uuid;
    end
  end

  suspend;
end
