CREATE PROCEDURE HP_EXTINFO_BMG_PROCESS(
    EIL_ID D_ID)
AS
declare variable result d_smallint;
declare variable answer_data d_str_10;
declare variable json_str d_str_255;
declare variable bls_id d_id;
declare variable eil_status d_smallint;
begin
    -- RFCKG-4142 Обработка ответа от БМГ
    select eil.eil_from_bls_id, eil.eil_status
    from extinfolist eil
    inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
    where eil.eil_id = :eil_id
        and eic.eic_code = 'bmg'
    into bls_id, eil_status;

    if (:eil_status = 2) then
    begin
        select cast(blx_blob as d_blob_text)
        from hp_get_blobex_blob(:bls_id)
        into json_str;

        if (row_count is null or coalesce(json_str, '') = '') then
            exit;

        result = cast(f_json_str_get_value(json_str, 'result', null) as d_smallint);
        answer_data = cast(f_json_str_get_value(json_str, 'data', null) as d_str_10);

        if (result = 0) then -- успешный запрос
        begin
            update bmg_clients bc
            set bc.bc_status = iif(:answer_data = 'true', 1, -2)
            where bc.bc_from_eil_id = :eil_id;
        end
    end
    else
    begin
        -- Обработка ошибок
        result =
            case :eil_status
                when -2 then -3 -- окончательная ошибка
                when -3 then -4 -- запрос отменен
                when -1 then -1 -- временная ошибка
                else 0 -- в процессе
            end;

        update bmg_clients bc
        set bc.bc_status = :result
        where bc.bc_from_eil_id = :eil_id;
    end
end
