CREATE PROCEDURE CRM_CRED_HISTORY(
    C_ID ID_TYPE)
RETURNS (
    JSON_VALUE D_BLOB_TEXT)
AS
declare variable C_NAME_FIO D_STR_100;
declare variable C_BIRTH_DATE date;
declare variable UNIC_ACC_STR STR_10;
declare variable ACTIVE_KONT_ID ID_TYPE;
declare variable ACTIVE_KONT_DELINQ D_SMALLINT;
declare variable IS_FIRST_ROW D_SMALLINT;
declare variable CLOSED_KONT_ID ID_TYPE;
declare variable CLOSED_KONT_DELINQ D_SMALLINT;
declare variable OFFICE_ID integer;
declare variable SVZ_ID ID_TYPE;
declare variable SERIES smallint;
declare variable CLIENT_STATUS smallint;
declare variable LAST_MA_ID ID_TYPE;
declare variable LAST_K_ID ID_TYPE;
declare variable LAST_KRL_ID ID_TYPE;
declare variable LAST_ID_STATUS STR_100;
declare variable LAST_DATE D_DATE;
declare variable LAST_ID_REFUSAL STR_255;
declare variable AMOUNT_DUE CURR_TYPE;
declare variable MATURITY_DATE date;
declare variable DELAY_KONT_ID ID_TYPE;
declare variable DELAY_COUNT integer;
declare variable AMOUNT_DEBT CURR_TYPE;
declare variable SERVICE_DEBT STR_255;
declare variable ACTIVE_NO STR_20;
declare variable CLOSED_NO STR_20;
declare variable ACT_KONT_NO STR_20;
declare variable DELINQ_KONT_NO STR_20;
declare variable FIRST_PAYMENT CURR_TYPE;
declare variable UPCOMING_PAYMENT CURR_TYPE;
declare variable NEXT_PAYMENT CURR_TYPE;
declare variable SH_ID smallint;
declare variable C_CTZN_NAME STR_100;
declare variable SPEC_UCH D_STR_20;
begin

    json_value = '{ ';

    for select c.c_name_fio,
               c.c_birth_date,
               rd_name
    from clients c
    inner join refdata on rd_id = c_citizen
    where c.c_id = :c_id
    into :c_name_fio, :c_birth_date, c_ctzn_name

    do
    begin 

        select uniq_acc_str
        from hp_show_clientuniq_acc_no (:c_id)
        into :unic_acc_str;

    end

    json_value = BLOB_APPEND(json_value, f_json_make_pair('fio', c_name_fio, 'str') || ', '
                            || f_json_make_pair('birth_date', c_birth_date, 'date') || ', '
                            || f_json_make_pair('citizen', c_ctzn_name, 'str') || ', '
                            || f_json_make_pair('personal_account', unic_acc_str, 'str') || ', '
                            || '"document_id": [{'
                            || '"active": [' );
    is_first_row = 1;
    for select ks.ks_from_k_id,
               coalesce(max(ok.ok_ol_date - ok.ok_sh_date), 0),
               k.k_kontract_no
    from kontstate ks
    inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
    left join operkred ok on ok.ok_from_k_id = ks.ks_from_k_id
    inner join kontract k on k.k_id = ks.ks_from_k_id
    where ks.ks_ol_is_delete = 0
      and ks.ks_state = 1
      and current_date between ks.ks_date_begin and ks.ks_date_end
      and kc.kc_from_c_id = :c_id
      and ok.ok_ol_is_active = 1
      and ok.ok_ol_date  <= current_date
      and ok.ok_type < 7
      and ok.ok_from_c_id=0
    group by 1,3
    into :active_kont_id, :active_kont_delinq, :active_no
    do
    begin
        if (is_first_row = 0) then
            json_value = BLOB_APPEND(json_value, ', ');
    
        json_value = BLOB_APPEND(json_value, '{ ' || f_json_make_pair('kontract_no', :active_no, 'str') || ', '
                                || f_json_make_pair('id', :active_kont_id, 'int') || ', '
                                || f_json_make_pair('count', active_kont_delinq, 'int')
                                || '} ');
        if (is_first_row = 1) then
            is_first_row = 0;      
    end
    json_value = BLOB_APPEND(json_value, '],'
                            || '"history": [' );
    is_first_row = 1;
    for select ks.ks_from_k_id,
               coalesce(max(ok.ok_ol_date - ok.ok_sh_date), 0),
               k.k_kontract_no
    from operkred ok
    inner join knclient kc on kc.kc_from_k_id = ok.ok_from_k_id
    inner join kontstate ks on kc.kc_from_k_id = ks.ks_from_k_id 
    inner join kontract k on k.k_id = ks.ks_from_k_id 
    where kc.kc_from_c_id = :c_id
      and ks.ks_ol_is_delete = 0
      and ks.ks_state = 2
      and current_date between ks.ks_date_begin and ks.ks_date_end
      and ok.ok_ol_is_active = 1
      and ok.ok_ol_date  <= current_date
      and ok.ok_type < 7
      and ok.ok_from_c_id=0
    group by 1,3
    into :closed_kont_id, :closed_kont_delinq, :closed_no
    do
    begin
        -- RFCRU-4630
        spec_uch = 'Не в спец учете';

        select 'В спец учете'
        from oksummary oks
        where oks_from_k_id = :closed_kont_id and
              current_date between oks.oks_date_begin and oks.oks_date_end and
              (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc + oks_out_other + oks_extra) > 0
        into spec_uch;

        if (is_first_row = 0) then
            json_value = BLOB_APPEND(json_value, ', ');
    
        json_value = BLOB_APPEND(json_value, '{ ' || f_json_make_pair('kontract_no', :closed_no, 'str') || ', '
                                || f_json_make_pair('id', :closed_kont_id, 'int') || ', '
                                || f_json_make_pair('count', closed_kont_delinq, 'int') || ', '
                                || f_json_make_pair('spec_uch', spec_uch, 'string')
                                || '} ');
        if (is_first_row = 1) then
            is_first_row = 0;      
    end

    json_value = BLOB_APPEND(json_value, ']}],');

    select iif(ma.ma_office_id <> k.k_last_reg_id, coalesce(k.k_last_reg_id, ma.ma_office_id),coalesce(k.k_last_reg_id, ma.ma_office_id)),
           iif(un.un_from_c_id <> k.k_fact_disb_lo_id, coalesce(k.k_fact_disb_lo_id, un.un_from_c_id), coalesce(k.k_fact_disb_lo_id, un.un_from_c_id))
    from miniappclient mac
    inner join miniapp ma on ma_id = mac_from_ma_id
    inner join username un on un.un_id = ma.ma_from_un_id
    left join kontract k on k.k_from_ma_app_id = ma.ma_id
    where mac.mac_from_c_id = :c_id
    order by ma.ma_id desc
    rows 1
    into :office_id, :svz_id;

    select k.k_circle
    from knclient kc
    inner join kontract k on k.k_id = kc.kc_from_k_id
    where kc.kc_from_c_id = :c_id
    order by k.k_circle desc
    rows 1
    into :series;

    client_status = f_point(:c_id, current_date);

     json_value = BLOB_APPEND(json_value, f_json_make_pair('office_id', office_id, 'int') || ', '
                            || f_json_make_pair('svz_id', svz_id, 'int') || ', '
                            || f_json_make_pair('series', coalesce(series, 0), 'int') || ', '
                            || f_json_make_pair ('client_status', coalesce(client_status, 0),  'int') || ', ');


    for select max(ma_id),
               max(ma_date_app)
    from miniappclient
    inner join miniapp on ma_id = mac_from_ma_id
    where mac_from_c_id = :c_id
      and mac_amount > 0
    into :last_ma_id, last_date
    do
    begin

        for select (case
                when ma.ma_is_active <= 0 or k.k_status <= -1 then 'Отказан'
                when ma.ma_is_active = 1 and k.k_status = 0 then 'В процессе'
                when ma.ma_is_active = 1 and k.k_status = 1 then 'Выдан'
                end) as status,
               k.k_id
        from miniapp ma
        left join kontract k on k.k_from_ma_app_id = ma.ma_id
        where ma.ma_id = :last_ma_id
          and ma_date_app = :last_date
        into last_id_status, last_k_id
        do
        begin
            last_id_status = trim(last_id_status);
            if(last_id_status = 'Отказан') then
            begin
    
                for select max(krl_id)
                from kontreslist
                where krl_from_k_id = :last_k_id
                  or krl_ma_id = :last_ma_id
                into last_krl_id
                do
                begin
                    select kr_text
                    from kontres
                    inner join kontreslist krl on krl_from_kr_id = kr_id
                    where krl.krl_id = :last_krl_id
                    into last_id_refusal;
                end
            end
            json_value = BLOB_APPEND(json_value, f_json_make_pair('last_date', last_date, 'date') || ', '
                                    || f_json_make_pair('last_id_status', last_id_status, 'string') || ', '
                                    || f_json_make_pair('last_id_refusal', coalesce(last_id_refusal, ''),  'string') || ', ');
        end
    end

    json_value = BLOB_APPEND(json_value, '"upcoming_payment" : [');

    is_first_row = 1;
    for select kc.kc_from_k_id,
               min(ok.ok_sh_date),
               k.k_kontract_no
    from knclient kc
    inner join oksummary oks on oks.oks_from_k_id = kc.kc_from_k_id
    inner join operkred ok on ok.ok_from_k_id = kc.kc_from_k_id
    inner join kontract k on k.k_id = oks.oks_from_k_id
    where kc.kc_from_c_id = :c_id
      and oks.oks_close_state = 0
      and current_date - oks.oks_delinq_date <= 0
      and current_date between oks.oks_date_begin and oks.oks_date_end
      and ok.ok_ol_is_active = 1
      and ok.ok_type < 7
      and ok.ok_sh_date >= current_date
    group by 1,3
    into :active_kont_id, :maturity_date, act_kont_no
    do
    begin

    select sum(ok_sh_os + ok_sh_perc)
    from operkred
    where ok_from_k_id = :active_kont_id
      and ok_ol_is_active = 1
      and ok_type < 7
      and ok_sh_date = :maturity_date
    into :amount_due;

    select ok_sh_os + ok_sh_perc
    from operkred
    where ok_from_k_id = :active_kont_id
      and ok_ol_is_active = 1
      and ok_sh_id = 1
      and ok_type = 1
    into :first_payment;

    if (is_first_row = 0) then
        json_value = BLOB_APPEND(json_value, ', ');

        json_value = BLOB_APPEND(json_value, '{ ' || f_json_make_pair('kontract_no', act_kont_no, 'str') || ', '
                                || f_json_make_pair('id', active_kont_id, 'bigint') || ', '
                                || f_json_make_pair('first_payment', first_payment, 'float(11,2)') || ', '
                                || f_json_make_pair('amount_due', amount_due, 'float(11,2)') || ', '
                                || f_json_make_pair('maturity_date', maturity_date, 'date')
                                || '} ');
        if (is_first_row = 1) then
            is_first_row = 0;      
    end 
    json_value = BLOB_APPEND(json_value, '],'

                            || '"delay_payment" : [');

    is_first_row = 1;

    for select kc.kc_from_k_id,
               current_date - oks.oks_delinq_date,
               k.k_kontract_no
    from knclient kc
    inner join oksummary oks on oks.oks_from_k_id = kc.kc_from_k_id
    inner join kontract k on k.k_id = oks.oks_from_k_id
    where kc.kc_from_c_id = :c_id
      and oks.oks_close_state = 0
      and current_date - oks.oks_delinq_date > 0
      and current_date between oks.oks_date_begin and oks.oks_date_end
    into :delay_kont_id, :delay_count, delinq_kont_no
    do
    begin

    select sum(sh_calc_fine + zad_os + zad_perc)
    from aarep_make_zad_list(:delay_kont_id, current_date, 0,0)
    into :amount_debt;

    select ok_sh_os + ok_sh_perc
    from operkred
    where ok_from_k_id = :delay_kont_id
      and ok_ol_is_active = 1
      and ok_type = 1
      and ok_sh_id = 1
    into :first_payment;

    sh_id = 0;

    select sh_calc_fine + zad_os + zad_perc, sh_id
    from aarep_make_zad_list(:delay_kont_id, current_date, 0,0)
    rows 1
    into :upcoming_payment, sh_id;

    sh_id = sh_id + 1;

    select coalesce(sh_calc_fine + zad_os + zad_perc, 0.0000)
    from aarep_make_zad_list(:delay_kont_id, current_date, 0,0)
    where sh_id = :sh_id
    into :next_payment;

     if (is_first_row = 0) then
        json_value = BLOB_APPEND(json_value, ', ');

        json_value = BLOB_APPEND(json_value, '{ ' || f_json_make_pair('kontract_no', delinq_kont_no, 'str') || ', '
                                || f_json_make_pair('id', delay_kont_id, 'bigint') || ', '
                                || f_json_make_pair('delay_count', delay_count, 'int') || ', '
                                || f_json_make_pair('amount_debt', amount_debt, 'float(11,2)') || ','
                                || f_json_make_pair('first_payment', first_payment, 'float(11,2)') || ','
                                || f_json_make_pair('upcoming_payment', upcoming_payment, 'float(11,2)') || ','
                                || f_json_make_pair('next_payment', next_payment, 'float(11,2)')
                                || '} ');
        if (is_first_row = 1) then
            is_first_row = 0;
    end
    json_value = BLOB_APPEND(json_value, '],'
                            || '"service" : [' );
    is_first_row = 1;
    for select sbl_name
    from sb_list
    inner join sb_akt on sba_from_sbl_id = sbl_id
    where sba_from_k_id = :delay_kont_id
      and sba_date_end > current_date
    into service_debt
    do
    begin

        if (is_first_row = 0) then
        json_value = BLOB_APPEND(json_value, ', ');

        json_value = BLOB_APPEND(json_value, '{ ' || f_json_make_pair('service_debt', coalesce(service_debt, ''), 'string') || '} ');

        if (is_first_row = 1) then
            is_first_row = 0;

    end

    json_value = BLOB_APPEND(json_value, ']}');

    suspend;
end
