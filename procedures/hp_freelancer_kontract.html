CREATE PROCEDURE HP_FREELANCER_KONTRACT(
    PAR1 D_ID,
    ADD_PARAMS D_STR_LONG)
RETURNS (
    NAME_FIO D_STR_100,
    PASS_NO VARCHAR(30),
    PASS_DATE D_DATE,
    PASS_WHO D_STR_255,
    FULL_ADRESS D_STR_255,
    INN D_STR_20,
    ADRESS_FACT D_STR_255,
    AGREEMENT_SUM D_CURRENCY,
    DATE_SIGN D_DATE,
    QR_CODE D_STR_LONG,
    AGREEMENT_SUM_MEMO VARCHAR(1024),
    BC_ACCOUNT_NO VARCHAR(50),
    B_ID D_STR_10,
    B_NAME D_STR_255)
AS
declare variable EDL_ID D_ID;
declare variable SIGN_STATUS D_SMALLINT;
declare variable SD_ID D_SMALLINT;
begin

  select c.c_name_fio, c.c_inn, cds.cds_doc_issued_by, cds.cds_index_doc_no, cds.cds_doc_date_begin
  from clients c
  left join clientdocs cds on c.c_id = cds.cds_from_c_id and
        cds.cds_is_delete = 0 and
        cds.cds_is_main_doc = 1 and
        current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end
  where c.c_id = :par1
  into :name_fio, :inn, :pass_who, :pass_no, :pass_date;

  select bc.bc_account_no, b.b_id, b.b_name
  from clients c
  left join bankclient bc on c.c_id = bc.bc_from_c_id
  left join banks b on bc.bc_from_b_id = b.b_id
  where c.c_id = :par1 and
        bc.bc_is_active = 1 and
        bc.bc_is_main = 1
  rows 1
  into :bc_account_no, :b_id, :b_name;

  select adl.adl_address_str
  from addrclient adc
  inner join addrlist adl on adc.adc_from_adl_id = adl.adl_id
  where adc.adc_from_c_id = :par1 and
        adc.adc_address_type = 1
  into :full_adress;

  select sd.sd_id, adl.adl_address_str
  from servicedog sd
  inner join region reg on reg.reg_unicode = sd.sd_from_reg_id
  inner join addrlist adl on adl.adl_id = reg.reg_address_fact
  where sd.sd_from_c_id = :par1
    --and sd.sd_kind = 0
  order by sd.sd_id desc
  rows 1
  into :sd_id, :adress_fact;

  select sum(ssh_amount_dogovor)
  from servicesh
  where ssh_date <= current_date and
        ssh_date_regist <= current_date and
        ssh_from_sd_id = :sd_id
  into :agreement_sum;

  select sd.sd_memo
  from servicedog sd
  where sd.sd_id = :sd_id
  into :agreement_sum_memo;

  select cast(eds.eds_status_datetime as date), eds.eds_status
  from edoc_list edl
  inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id and
        edc.edc_code = 'freelancer_kontract'
  inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
  where edl.edl_object_kind = 0 and
        edl.edl_object_id = :par1
  order by edl_status desc, edl_id desc
  rows 1
  into :date_sign, :sign_status;

  if (sign_status is null) then
    date_sign = null;

  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(:add_params, 'edl_id', null);
    qr_code = f_edoc_qr_by_edl(:edl_id, :par1);
  end

  suspend;
end
