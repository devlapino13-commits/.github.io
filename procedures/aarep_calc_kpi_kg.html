CREATE PROCEDURE AAREP_CALC_KPI_KG(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    K_KOLICH INTEGER,
    K_KC_ID DECIMAL(18,0),
    K_REGION_ID INTEGER,
    K_BALANCE DECIMAL(18,4),
    K_JOBNAME VARCHAR(255),
    K_IS_KC INTEGER,
    K_STAGE_BEGIN DATE,
    K_IS_WORKING VARCHAR(20),
    K_NEED_KPI INTEGER,
    K_BRANCH_CODE VARCHAR(20),
    K_REGION_NAME VARCHAR(100),
    K_KC_NAME VARCHAR(100),
    K_BRANCH_NAME VARCHAR(100))
AS
declare variable k_is_work integer;
begin
  for select sum(k_kolich), k_kc_id, k_region_id, sum(k_balance), k_jobname, k_is_kc, k_stage_begin, k_is_working,
             k_need_kpi, k_branch_code, k_region_name, k_kc_name, k_branch_name
      from aarep_calc_kpi_kg_tmp2(:rep_region_id, :rep_date_end)
      group by k_kc_id, k_region_id, k_jobname, k_is_kc, k_stage_begin, k_is_working, k_need_kpi, k_branch_code, k_region_name, k_kc_name, k_branch_name
      order by k_kc_id, k_region_id
      into :k_kolich, :k_kc_id, :k_region_id, :k_balance, :k_jobname, :k_is_kc, :k_stage_begin, :k_is_working,
           :k_need_kpi, :k_branch_code, :k_region_name, :k_kc_name, :k_branch_name
  do
  begin
    suspend;
  end
end
