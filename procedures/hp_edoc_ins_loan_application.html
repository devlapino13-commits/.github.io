CREATE PROCEDURE HP_EDOC_INS_LOAN_APPLICATION(
    MA_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable PR_CODE D_STR_100;
declare variable DB_ID D_INTEGER;
declare variable C_ID D_ID;
declare variable MA_DATE_APP D_DATE;
declare variable ADL_ADDRESS_STR D_STR_255;
declare variable LANG D_STR_LANG;
declare variable C_NAME_FIO D_STR_100;
declare variable C_PFT_RISK_KOEF D_INTEGER;
declare variable MAC_ID D_INTEGER;
declare variable MAC_AMOUNT D_CURRENCY;
declare variable MA_DATE_OPER D_DATE;
declare variable C_INN D_STR_20;
declare variable ICM_INN D_STR_20;
declare variable INN_CLI_IS_EQUAL D_BOOL;
declare variable INN_RECEIVED D_BOOL;
declare variable C_CITIZEN D_STR_20;
declare variable CA_OBJECT_ID D_ID;
declare variable KPD D_CURRENCY;
declare variable SERVICE_STATUS D_SMALLINT;
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable REQ_STATUS D_SMALLINT;
declare variable ACTUALITY_DATE D_DATE;
declare variable IP_STATUS D_SMALLINT;
declare variable GKB_EIL_STATUS D_SMALLINT;
declare variable GKB_DECISION D_SMALLINT;
declare variable GKB_HAS_RESULT D_SMALLINT;
declare variable GKB_SKIP_TEXT D_STR_255;
declare variable QUERY_COUNT_90 D_INTEGER;
begin
  -- Процедура создает электронный документ на заявку на займ.
  -- Готовит бланки и отправляет СМС.
  res_text = '';
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);

  if (db_id in (39801, 41701)) then
  begin
    if (f_tmp_const('edoc_ipf_inactive') = 1) then -- если по ИФП eDoc отключен
    begin
      select pr_code, ma_date_app
      from miniapp, product
      where ma_id = :ma_id and
            ma_from_pr_id = pr_id
      order by ma_id, ma_number
      rows 1
      into pr_code, ma_date_app;
  
      -- по заявкам ИФП пока не формируем эл.документ
      if (position('.ipf.', pr_code) > 0) then
      begin
        res_text = 'Электронные заявки для ИПФ пока не предусмотрены';
        suspend;
        exit;
      end
    end

    if (res_text = '') then
    begin
      select res_text
      from hp_miniapp_check_goods(:ma_id)
      into res_text;
    end
    
    --RFCKG-2952
    if (res_text = '' and db_id <> 39801) then
    begin
      if (not exists(select *
                     from miniappclient mac
                     inner join clp_miniappclient clm on clm.cmc_from_ma_id = mac.mac_from_ma_id and
                           clm.cmc_from_c_id = mac.mac_from_c_id
                     where mac.mac_from_ma_id = :ma_id and
                           mac.mac_amount > 0)) then

        res_text = 'Сначала заполните цель и отрасль финансирования';
    end 


    if (res_text = '') then
    begin
      select res_text
      from hp_miniapp_edoc_check_cond(:ma_id)
      into res_text;
    end
  end

  if (db_id = 81001 and res_text = '') then
  begin
    for select cl.c_id
        from miniappclient mac
        inner join clients cl on cl.c_id = mac.mac_from_c_id
        where mac.mac_from_ma_id = :ma_id and
              mac.mac_amount > 0
        into :c_id
    do
    begin
      if (not exists(
            select 1
            from actionlist acl
            where acl.acl_object_kind = 0 and
                  acl.acl_object_id = :c_id and
                  acl.acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify_OiVO'), f_get_rd_id('ActionGroup.Passport_verify')) and
                  acl.acl_status = 2 and
                  acl.acl_date > cast('01.04.2026' as d_date))) then
      begin
        res_text = 'Верификация пройдена до 01.04.2026, требуется пройти повторно';

        update clients cl   -- если последняя верификация клиента была до даты "01.04.2026", то сбрасываем верификацию  (RFCRU-5657)
        set cl.c_verify_status = 1
        where cl.c_id = :c_id and
              cl.c_verify_status = 2;

        leave;
      end
    end

    for select mac_id
    from miniapp ma
    inner join miniappclient mac on mac.mac_from_ma_id = ma_id
    where ma.ma_id = :ma_id
      and ma.ma_disb_kind = 7
      and ma.ma_is_active = 1
      and ma.ma_date_end = '31.12.9999'
    into :mac_id
    do
    begin
      if (not exists
        (
          select 1
          from fps_application_banks fab
          where fab.fab_from_mac_id = :mac_id
        )
      ) then
      begin
        res_text = 'Пожалуйста, сперва укажите банк';
        leave;
      end
    end
  end

  /*  RFCKZ-389
  if (db_id = 39801 and res_text = '') then
    select coalesce(hp.res_text, '')
    from miniapp ma, miniappclient mac, hp_afm_get_status('afm', mac.mac_from_c_id, mac.mac_from_ma_id) hp
    where ma.ma_id = :ma_id
        and ma.ma_id = mac.mac_from_ma_id
        and ma.ma_date_app = hp.answer_date
        and ma.ma_is_active = 1
        and ma.ma_date_end >= '31.12.2999'
        and mac.mac_amount > 0
        and hp.status <> 1
    into :res_text;
  */
  if (db_id = 39801 and res_text = '' and f_get_const('afm_is_active') = '0') then   -- RFCKZ-79
  begin
    -- Проверка адреса регистрации
    for select mac.mac_from_c_id, mac.mac_amount
        from miniappclient mac
        where mac.mac_from_ma_id = :ma_id and
              mac.mac_amount > 0
        into c_id, mac_amount
    do
    begin
      select eil.eil_from_bls_id
      from extinfolist eil
      inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
      where eil.eil_from_c_id = :c_id
        and eil.eil_status = 2
        and eic.eic_code = 'checkme_info'
      order by eil.eil_status_datetime desc
      rows 1
      into bls_id;

      if (coalesce(bls_id, 0) <> 0) then
      begin
        select blx_blob
        from hp_get_blobex_blob(:bls_id)
        into json_text;
      end

      for select service_status
      from hp_parse_json_checkme_service(:json_text)
      into :service_status
      do
      begin
        if (:service_status in (3, 4)) then
        begin
          if (not exists(select *
                       from clientdocs cds
                       where cds.cds_from_c_id = :c_id and
                             cds.cds_is_delete = 0 and
                             cds.cds_from_pk_id in (371, 372))) then
            res_text = 'Анкета ПОД/ФТ не заполнена';
        end
      end
    end
  end

    -- RFCKG-4573
  if (db_id = 41701) then
  begin
    for select mac.mac_from_c_id
        from miniappclient mac
        where mac.mac_from_ma_id = :ma_id and
              mac.mac_amount > 0
        into c_id
    do
    begin
      if ((f_edoc_status('personaldata.consent', :c_id, current_date)) <> 'signed') then
        res_text = 'Сначала подпишите согласие на обработку персональных данных';
    end
  end

  if (res_text = '') then
  begin
    -- Проверка адреса регистрации
    for select mac.mac_from_c_id, mac.mac_amount
        from miniappclient mac
        where mac.mac_from_ma_id = :ma_id and
              mac.mac_is_poruch <> -2
        into c_id, mac_amount
    do
    begin
      c_name_fio = '';
      c_pft_risk_koef = null;

      select c.c_name_fio, c.c_pft_risk_koef, c.c_inn
      from clients c
      where c.c_id = :c_id
      into c_name_fio, c_pft_risk_koef, c_inn;
  
      adl_address_str = null;
      select adl_address_str
      from addrclient
      inner join addrlist on adl_id = adc_from_adl_id
      where adc_from_c_id = :c_id
        and adc_address_type = 1
      into adl_address_str;
      if (coalesce(adl_address_str, 'Не указан') = 'Не указан') then
        res_text = f_lang(lang, 'Юр.адрес не заполнен, сначала заполните адрес клиента ' || :c_name_fio);

        --Значение уровня риска заполняется дефолтно значением = -1
        --Проверку должны проходить только клиенты (т.е. лица дающие согласие, поручители не проверяются)
      --if (db_id = 41701 and res_text = '' and coalesce(c_pft_risk_koef, -1) = -1 and coalesce(mac_amount,0) > 0) then
      if (db_id in (41701) and res_text = '' and coalesce(c_pft_risk_koef, -1) = -1 and coalesce(mac_amount,0) > 0) then -- передвинули отправку запроса АФМ
        res_text = 'Не указан уровень риска ПФТ/ОД, сначала укажите уровень риска клиента ' || :c_name_fio;
    end
  end

  if (db_id = 81001 and res_text = '') then
  begin
    select res_text
    from hp_check_client_miniapp(:c_id, :ma_id, :un_id)
    into res_text;
  end

  if (db_id = 81001 and res_text = '') then
  begin
    if (exists(select *
               from kontreslist
               where krl_ma_id = :ma_id) or exists(select *
                                                   from miniapp
                                                   where ma_id = :ma_id and
                                                         ma_is_active < 1)) then
      res_text = 'По данной заявке был отказ - для выдачи займа создайте новую';
  end

  /*if (db_id = 81001 and             -- RFCRU-3486
      res_text = '') then
  begin
    if (current_date >= '01.04.2025' and
        exists(select *
               from miniapp
               where ma_id = :ma_id and
                     ma_date_end = '31.12.9999' and
                     ma_refund = 1)) then
      res_text = 'Рефинансирование временно приостановлено';
  end*/

  if (db_id = 81001 and res_text = '') then
  begin
    select res_error_text
    from edoc_loan_app_blacklist_check(:c_id)
    into :res_text;
  end

  if (db_id = 81001 and res_text = '' and current_date >= '02.06.2025') then
  begin
    select err_text
    from hp_check_loan_period(:ma_id)
    into res_text;
  end

  if (db_id = 81001 and res_text = '') then
  begin
    select pr_code, ma_date_oper
    from miniapp, product
    where ma_id = :ma_id and
          ma_from_pr_id = pr_id and
          ma_is_active = 1 and
          ma_date_end >= '31.12.9999'
    into :pr_code, ma_date_oper;

    if (f_get_const('inn_ru_source') > 0 and
        (pr_code not containing '.zalog' or ma_date_oper >= '01.09.2025')) then --RFCRU-4016
    begin
        select inn_cli_is_equal,
               inn_received,
               coalesce(inn, '')
        from inn_query_status(:c_id)
        into :inn_cli_is_equal,
             :inn_received,
             :icm_inn;

        if (inn_received = 0) then
        begin
            res_text = 'У клиента отсутствует ИНН';
        end
        else if (inn_cli_is_equal = 0) then
        begin
          select icm.icm_inn
          from inn_check_main icm
          where icm.icm_from_c_id = :c_id
          into icm_inn;

          res_text = 'Найденный ИНН (' || icm_inn || ') отличается от ИНН в карточке физ. лица (' || coalesce(c_inn, '') || '). Необходимо пройти верификацию';
        end
    end

    if (pr_code containing '.universal.' or pr_code containing '.oivo_pack.') then
    begin
      for select mcl.mac_from_c_id
          from miniapp m
          inner join miniappclient mcl on mcl.mac_from_ma_id = m.ma_id
          where mcl.mac_amount > 0 and m.ma_id = :ma_id
          into :c_id
      do
      begin
        c_citizen = '';
        select coalesce(r.rd_code, '')
        from clients cl
        inner join refdata r on r.rd_id = cl.c_citizen
        where cl.c_id = :c_id
        into :c_citizen;

        if (c_citizen <> 'Citizen.RU') then
          res_text = 'Продукт только для граждан РФ, выберите другой продукт';
      end
    end

    if (pr_code containing '.business.' and res_text = '') then
    begin
      -- RFCRU-5314
      -- Проверяем можем ли мы отправить заявление на заем для продукта Бизнес

      select
        req_status,
        actuality_date,
        coalesce(ip_status, 0)
      from hp_kontur_ip_get_base_info('miniapp', :ma_id)
      into
        req_status,
        actuality_date,
        ip_status;

      -- Как видно из блока ниже, валиден только ip_status = 1 (Клиент является ИП)
      res_text =
        case
          when req_status is null then 'Сначала проверьте статус ИП'
          when current_date >= actuality_date then 'Сначала проверьте статус ИП'
          when req_status < 0 then 'Ошибка'
          when req_status = 0 then 'Сначала проверьте статус ИП'
          when req_status = 1 then 'Сначала проверьте статус ИП'
          when req_status = 2 and ip_status = 1 then ''
          when req_status = 2 and ip_status = 0 then 'Нет действующего статуса ИП, измените продукт'
          else 'Не предвиденная ошибка, просим обратиться в службу технической поддержки'
        end;
    end
  end 

  --RFCKG-4327
  if (res_text = '' and db_id = 39801) then
  begin
    select ca.ca_object_id
    from cfp_anketa ca
    inner join miniappclient mac on ca.ca_object_id = mac.mac_id
    where mac.mac_from_ma_id = :ma_id
    and ca.ca_object_kind = f_objkind('miniappclient', 'id')
    and ca.ca_from_c_id = mac.mac_from_c_id
    into :ca_object_id;

    if (coalesce(:ca_object_id, 0) = 0) then
      res_text = 'Сначала необходимо заполнить Анкету доходов/расходов';
    else
    begin
      select cakdn.cakdn_kdn
      from cfp_anketa_kdn cakdn
      where cakdn.cakdn_from_mac_id = :ca_object_id
      into :kpd;

      if (kpd > 0.5) then
        res_text = 'КДН клиента выше нормы';
    end

    if (res_text = '') then
    begin
      if (exists(
          select 1 from miniappclient mac
          where mac.mac_from_ma_id = :ma_id
            and mac.mac_amount > 0
            and not exists (
                select 1 from negativ_kib_history nkh
                inner join extinfolist eil on eil.eil_id = nkh.nkh_from_eil_id
                where nkh.nkh_from_c_id = mac.mac_from_c_id
                  and nkh.nkh_last = 1
                  and eil.eil_status = 2))) then
        res_text = 'Количество запросов отчета КИБ за последние 90 дней не проверено (запрос еще обрабатывается или не отправлен)';
    end
  end

  -- RFCKZ-877: проверка Fraud Score ГКБ перед созданием документа заявки.
  -- Запрос отправляет робот HP_GKB_FRAUD_ROBOT после прохождения всех
  -- обязательных стоп-факторов, здесь только читаем результат.
  if (res_text = '' and db_id = 39801
      and coalesce(f_get_const('gkb_fraud_is_active'), '0') = '1') then
  begin
    gkb_eil_status = null;
    gkb_decision   = null;
    gkb_has_result = 0;
    gkb_skip_text  = null;

    -- результат разбора по этой заявке
    select first 1 1, gpc.gpc_decision
    from gkb_pay_check gpc
    where gpc.gpc_object_kind = f_objkind('miniapp', 'id')
      and gpc.gpc_object_id = :ma_id
      and gpc.gpc_is_error = 0
    order by gpc.gpc_id desc
    into :gkb_has_result, :gkb_decision;

    if (gkb_has_result = 1) then
    begin
      -- 2 = автоотказ (C3, C4 и всё, что хуже)
      if (gkb_decision = 2) then
        res_text = f_lang(lang, 'Заявка отклонена. Клиент не прошел проверку по скоринговой модели Fraud Score ГКБ.');
    end
    else
    begin
      -- результата нет — смотрим состояние запроса
      select first 1 eil.eil_status
      from extinfolist eil
      inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
      where eic.eic_code = 'gkb_pay_checks_core'
        and eil.eil_object_kind = f_objkind('miniapp', 'id')
        and eil.eil_object_id = :ma_id
      order by eil.eil_id desc
      into :gkb_eil_status;

      if (gkb_eil_status is null) then
      begin
        -- запрос ещё не создан — показываем причину от робота
        select first 1 gfs.gfs_reason
        from gkb_fraud_skip gfs
        where gfs.gfs_from_ma_id = :ma_id
        into :gkb_skip_text;

        if (coalesce(gkb_skip_text, '') <> '') then
          res_text = gkb_skip_text;
        else
          res_text = f_lang(lang, 'Проверка Fraud Score ГКБ в процессе создания. Запрос будет отправлен в течение 3 минут, пожалуйста, подождите.');
      end
      else if (gkb_eil_status = 0) then
        res_text = f_lang(lang, 'Запрос Fraud Score ГКБ в процессе отправки. Пожалуйста, подождите.');
      else if (gkb_eil_status = 1) then
        res_text = f_lang(lang, 'Запрос Fraud Score ГКБ в обработке. Пожалуйста, подождите.');
      else if (gkb_eil_status = 2) then
        res_text = f_lang(lang, 'Запрос Fraud Score ГКБ в процессе обработки. Пожалуйста, подождите.');
      -- eil_status < 0 -> сервис недоступен (категория E), заявку не блокируем
    end
  end

  -- RFCRU-5644
  if (db_id = 81001 and res_text = '') then
  for
    select
      mac_from_c_id
    from miniappclient
    where mac_from_ma_id = :ma_id
      and mac_amount > 0
  into
      :c_id
  do
  begin
    res_text = (
      select
        err_txt
      from hp_separate_fio_checkvalid(-:c_id));
    if (res_text <> '') then
      break;
  end

  if (ma_id = 2945995) then
    res_text = ''; -- test RFCRU-6095

  if (res_text = '') then
  begin
    select edl_id, res_text
    from hp_edoc_insert('loan.application', f_objkind('miniapp', 'id'), :ma_id, :un_id)
    into edl_id, res_text;
  end
  /*  RFCKG-4801 теперь создаем бланк в пакете с эл. документ "заявка на получение кредита"
  if (res_text = '' and db_id in (39801, 41701)) then
  begin
    for select mac.mac_from_c_id, ma.ma_date_oper
        from miniappclient mac
        inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
        where mac.mac_from_ma_id = :ma_id and
              ma.ma_is_active = 1 and
              ma.ma_date_end >= '31.12.2999'
        into c_id, ma_date_oper
    do
    begin
      if (exists(select *
                 from clients c
                 inner join public_persons pp on pp.pp_from_c_id = c.c_id
                 where c.c_id = :c_id and
                       coalesce(c.c_is_public, 0) = 1 and
                       pp.pp_is_delete = 0 and
                       :ma_date_oper between pp.pp_date_begin and pp.pp_date_end)) then
      begin
        select edl_id, res_text
        from hp_edoc_insert('anketa_npdl.application', f_objkind('clients', 'id'), :c_id, :un_id)
        into edl_id, res_text;
      end
    end
  end
  */


  suspend;
end
