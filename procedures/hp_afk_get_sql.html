CREATE PROCEDURE HP_AFK_GET_SQL(
    F_MASTER D_STR_64,
    REC_EVENT D_STR_64)
RETURNS (
    TEXT_SQL D_STR_2048)
AS
declare variable v_fk_table d_str_64;
declare variable v_fk_field d_str_64;
declare variable v_ac_deleting d_str_64;
declare variable v_ac_updating d_str_64;
declare variable v_fk_master d_str_64;
declare variable v_order_ d_integer;
declare variable v_idx_name d_str_64;
declare variable n d_integer;
declare variable v_and d_str_10;
declare variable v_where d_str_2048;
declare variable v_set d_str_2048;
declare variable v_delim d_str_10;
declare variable v_messa d_str_2048;
begin

  for select distinct 1 order_, ta.fk_table, ta.ac_deleting,
                      ta.ac_updating, ta.idx_name
      from hp_afk_get_table(:f_master) ta
      where ((:rec_event = 'deleting' and
            ta.ac_deleting = 'NO ACTION') or
            (:rec_event = 'updating' and
            ta.ac_updating = 'NO ACTION'))
      union
      select distinct 2 order_, ta.fk_table, ta.ac_deleting,
                      ta.ac_updating, ta.idx_name
      from hp_afk_get_table(:f_master) ta
      where ((:rec_event = 'deleting' and
            ta.ac_deleting <> 'NO ACTION') or
            (:rec_event = 'updating' and
            ta.ac_updating <> 'NO ACTION'))
      order by 1
      into :v_order_, :v_fk_table, :v_ac_deleting, :v_ac_updating,
           :v_idx_name
  do
  begin

    if (rec_event = 'deleting') then
    begin
      if (v_ac_deleting = 'CASCADE') then --готово
      begin
        text_sql = ' ';
        v_and = '';
        n = 0;
        for select ta.fk_field
            from hp_afk_get_table(:f_master) ta
            where ta.idx_name = :v_idx_name and
                  (:v_idx_name != '' or
                  fk_table = :v_fk_table) and
                  ta.ac_deleting = :v_ac_deleting
            order by ta.field_position
            into :v_fk_field
        do
        begin
          n = n + 1;
          text_sql = text_sql || v_and || v_fk_field || ' = :o' || :n || '_id ';
          v_and = ' and ';
          if (v_idx_name = '') then
          begin
            text_sql = 'delete from ' || v_fk_table || ' where ' || text_sql;
            suspend;
            v_and = '';
            n = 0;
          end
        end

        text_sql = 'delete from ' || v_fk_table || ' where ' || text_sql;

      end

      else
      if (v_ac_deleting = 'NO ACTION') then
      begin
        text_sql = ' ';
        v_where = '';
        v_and = '';
        v_set = '';
        v_delim = '';
        v_messa = '';
        n = 0;
        for select ta.fk_field, ta.fk_master
            from hp_afk_get_table(:f_master) ta
            where ta.idx_name = :v_idx_name and
                  (:v_idx_name != '' or
                  fk_table = :v_fk_table) and
                  ta.ac_deleting = :v_ac_deleting
            order by ta.field_position
            into :v_fk_field, :v_fk_master
        do
        begin
          n = n + 1;
          v_where = v_where || v_and || v_fk_field || ' = :o' || :n || '_id ';
          v_set = v_set || v_delim || 'o' || :n || '_id d_str_64 = :o' || :n || '_id';
          v_and = ' and ';
          v_messa = v_messa || v_delim || '"' || v_fk_master || '" = '' || :o' || :n || '_id || ''';
          v_delim = ', ';
          if (v_idx_name = '') then
          begin
            text_sql = 'execute block(' || v_set || ')
as
begin
  if (exists(select 1 from ' || v_fk_table || ' where ' || v_where || ')) then
    exception e_default ''Referential integrity violation in a table "' || upper(v_fk_table) || '".
Foreign key references are present for the record.
Problematic key value is (' || v_messa || ').'';
end
';
            suspend;
            v_where = '';
            v_and = '';
            v_set = '';
            v_delim = '';
            v_messa = '';
            n = 0;
          end

        end

        text_sql = 'execute block(' || v_set || ')
as
begin
  if (exists(select 1 from ' || v_fk_table || ' where ' || v_where || ')) then
    exception e_default ''Referential integrity violation in a table "' || upper(v_fk_table) || '".
Foreign key references are present for the record.
Problematic key value is (' || v_messa || ').'';
end
';
      end
      else
      if (v_ac_deleting = 'SET NULL') then --готово
      begin
        text_sql = ' ';
        v_where = '';
        v_set = '';
        v_and = '';
        v_delim = '';
        n = 0;
        for select ta.fk_field
            from hp_afk_get_table(:f_master) ta
            where ta.idx_name = :v_idx_name and
                  (:v_idx_name != '' or
                  fk_table = :v_fk_table) and
                  ta.ac_deleting = :v_ac_deleting
            order by ta.field_position
            into :v_fk_field
        do
        begin
          n = n + 1;
          v_where = v_where || v_and || v_fk_field || ' = :o' || :n || '_id ';
          v_set = v_set || v_delim || v_fk_field || ' = null';
          v_and = ' and ';
          v_delim = ', ';
          if (v_idx_name = '') then
          begin
            text_sql = 'update ' || v_fk_table || ' set ' || v_set || ' where ' || v_where;
            suspend;
            v_where = '';
            v_set = '';
            v_and = '';
            v_delim = '';
            n = 0;

          end

        end

        text_sql = 'update ' || v_fk_table || ' set ' || v_set || ' where ' || v_where;
      end
    end
    else


    /****************************************************************************************************/
    if (rec_event = 'updating') then
    begin
      if (v_ac_updating = 'CASCADE') then
      begin
        text_sql = ' ';
        v_where = '';
        v_set = '';
        v_and = '';
        v_delim = '';
        n = 0;
        for select ta.fk_field
            from hp_afk_get_table(:f_master) ta
            where ta.idx_name = :v_idx_name and
                  (:v_idx_name != '' or
                  fk_table = :v_fk_table) and
                  ta.ac_updating = :v_ac_updating
            order by ta.field_position
            into :v_fk_field
        do
        begin
          n = n + 1;
          v_where = v_where || v_and || v_fk_field || ' = :o' || :n || '_id ';
          v_set = v_set || v_delim || v_fk_field || ' = :n' || :n || '_id';
          v_and = ' and ';
          v_delim = ', ';
          if (v_idx_name = '') then
          begin
            text_sql = 'update ' || v_fk_table || ' set ' || v_set || ' where ' || v_where;
            suspend;
            v_where = '';
            v_set = '';
            v_and = '';
            v_delim = '';
            n = 0;
          end
        end
        if (v_idx_name != '') then
          text_sql = 'update ' || v_fk_table || ' set ' || v_set || ' where ' || v_where;
      end
      else
      if (v_ac_updating = 'NO ACTION') then
      begin
        text_sql = ' ';
        v_where = '';
        v_and = '';
        v_set = '';
        v_delim = '';
        v_messa = '';
        n = 0;
        for select ta.fk_field, ta.fk_master
            from hp_afk_get_table(:f_master) ta
            where ta.idx_name = :v_idx_name and
                  (:v_idx_name != '' or
                  fk_table = :v_fk_table) and
                  ta.ac_updating = :v_ac_updating
            order by ta.field_position
            into :v_fk_field, :v_fk_master
        do
        begin
          n = n + 1;
          v_where = v_where || v_and || v_fk_field || ' = :o' || :n || '_id ';
          v_set = v_set || v_delim || 'o' || :n || '_id d_str_64 = :o' || :n || '_id';
          v_and = ' and ';
          v_messa = v_messa || v_delim || '"' || v_fk_master || '" = '' || :o' || :n || '_id || ''';
          v_delim = ', ';
          if (v_idx_name = '') then
          begin
            text_sql = 'execute block(' || v_set || ')
as
begin
  if (exists(select 1 from ' || v_fk_table || ' where ' || v_where || ')) then
    exception e_default ''Referential integrity violation in a table "' || upper(v_fk_table) || '".
Foreign key references are present for the record.
Problematic key value is (' || v_messa || ').'';
end
';
            v_and = '';
            v_delim = '';
            v_messa = '';
            v_set = '';
            v_where = '';
            n = 0;
            suspend;
          end
        end
        if (v_idx_name != '') then
          text_sql = 'execute block(' || v_set || ')
as
begin
  if (exists(select 1 from ' || v_fk_table || ' where ' || v_where || ')) then
    exception e_default ''Referential integrity violation in a table "' || upper(v_fk_table) || '".
Foreign key references are present for the record.
Problematic key value is (' || v_messa || ').'';
end
';
      end
      else
      if (v_ac_updating = 'SET NULL') then
      begin
        text_sql = ' ';
        v_where = '';
        v_set = '';
        v_and = '';
        v_delim = '';
        n = 0;
        for select ta.fk_field
            from hp_afk_get_table(:f_master) ta
            where ta.idx_name = :v_idx_name and
                  (:v_idx_name != '' or
                  fk_table = :v_fk_table) and
                  ta.ac_updating = :v_ac_updating
            order by ta.field_position
            into :v_fk_field
        do
        begin
          n = n + 1;
          v_where = v_where || v_and || v_fk_field || ' = :o' || :n || '_id ';
          v_set = v_set || v_delim || v_fk_field || ' = null';
          v_and = ' and ';
          v_delim = ', ';
        end

        text_sql = 'update ' || v_fk_table || ' set ' || v_set || ' where ' || v_where;
      end
    end
    if (v_idx_name != '') then
      begin
      text_sql = text_sql ||'--'||v_idx_name;
      suspend;
      end
  end

end
