CREATE PROCEDURE HP_INVOICEIPFD_PRINT(
    ID D_INTEGER)
RETURNS (
    OL_ID D_ID,
    NUMB D_INTEGER,
    KML_COUNT D_CURRENCY,
    KML_TOTAL D_CURRENCY,
    ML_NAME D_STR_255,
    ML_MODEL D_STR_255,
    YL_CODE D_STR_20,
    TXT_NAME D_STR_LONG,
    OM_NUMBER D_INTEGER)
AS
begin
  numb = 0;
  for select invd_ol_id,
             om_number,
             kml_count,
             kml_total,
             ml_name,
             ml_model,
             yl_code
      from invoiceipf_detail
      inner join opermat on om_from_ol_id = invd_ol_id
      inner join kontmat on kml_bill_id = om_bill_id and kml_bill_line_no = om_bill_line_no
      inner join matlist on ml_id = kml_from_ml_id
      inner join yedizlist on yl_id = ml_from_yl_id
      left join opermat_hide on omh_ol_id = om_from_ol_id and omh_number = om_number
      where invd_inv_id = :id and
            omh_ol_id is null
      into ol_id,
           om_number,
           kml_count,
           kml_total,
           ml_name,
           ml_model,
           yl_code
  do
  begin
    numb = numb + 1;
    txt_name = ml_name || ' ' || replace(ml_model, ml_name, '');

    suspend;
  end
end
