CREATE PROCEDURE HP_FIND_CLIENTS_TMP(
    FIND_C_ID DECIMAL(18,0),
    FIND_FIO VARCHAR(100),
    FIND_BIRTH_DATE DATE,
    FIND_CODE VARCHAR(100),
    FIND_PASS_DATE DATE,
    FIND_PASS_NO VARCHAR(100),
    FIND_CONT_INFO VARCHAR(100),
    SHOW_JUST_CRITICAL INTEGER,
    FIND_CITIZEN INTEGER = 0,
    IS_PHYS_FACE D_SMALLINT = -1)
RETURNS (
    C_ID DECIMAL(18,0),
    IS_CRITICAL INTEGER,
    RES_SAME_FIO INTEGER,
    RES_SAME_BIRTH_DATE INTEGER,
    RES_SAME_PASS_NO INTEGER,
    RES_SAME_PASS_DATE INTEGER,
    RES_SAME_CONTACT INTEGER,
    RES_SAME_CODE INTEGER)
AS
declare variable tmp_c_name_fio varchar(100); /* SHOW_JUST_CRITICAL */
declare variable tmp_birth_date date;
declare variable tmp_pass_no varchar(100);
declare variable tmp_pass_date date;
declare variable is_fio_ident integer;
declare variable txt1 varchar(100);
declare variable txt2 varchar(100);
declare variable txt3 varchar(100);
declare variable find_orig_fio varchar(100);
declare variable critical_found integer;
declare variable orig_txt1 varchar(100);
declare variable orig_txt2 varchar(100);
declare variable orig_txt3 varchar(100);
declare variable tmp_inn varchar(20);
declare variable tmp_pincode varchar(20);
declare variable tmp_c_name_fio2 varchar(100);
declare variable i integer;
declare variable tmp_fio_text varchar(255);
declare variable fio1 varchar(100);
declare variable fio2 varchar(100);
declare variable xfio1 varchar(100);
declare variable txt3_1 varchar(100);
declare variable txt3_2 varchar(100);
declare variable clt_id d_integer;
declare variable is_new_contdata d_smallint;
begin
  -- Процедура ищет совпадающих контрагентов
  -- Сначала ищем критические совпадения: кроме ФИО совпадает один из входящих параметров
  -- Если не найдены, ищем по ФИО

  c_id = 0;
  if (coalesce(find_c_id, 0) <= 0) then
    find_c_id = 0;

  is_new_contdata = f_get_const('new_contdata');
  is_new_contdata = 1; -- процедуру переводим на новый механизм, Sanjari, 21.12.2019

  -- проверяем входные параметры
  if (find_pass_no is null) then
    find_pass_no = '';
  if (find_fio is null) then
    find_fio = '';
  if (find_code is null) then
    find_code = '';
  if (find_cont_info is null) then
    find_cont_info = '';
  if (find_birth_date <= cast('01.01.1900' as date)) then
    find_birth_date = null;

  -- очишаем входные параметры
  find_fio = f_replace_all(find_fio, '  ', ' ');
  find_fio = trim(find_fio);
  find_pass_no = upper(f_replace_all(find_pass_no, ' ', ''));
  find_pass_no = trim(find_pass_no);
  find_cont_info = upper(f_replace_all(find_cont_info, ' ', ''));
  find_cont_info = trim(find_cont_info);
  find_code = trim(find_code);

  -- если указаны только нули, приравниваем как не заполненные
  if (right('000000' || find_code, 6) = '000000') then
    find_code = '';
  if (right('000000' || find_pass_no, 6) = '000000') then
    find_pass_no = '';
  if (right(find_cont_info, 6) = '000000') then
    find_cont_info = '';

  -- Если поиск только критичечких, но доп.поля пустые, завершаем работу
  if (show_just_critical = 1 and
      find_code = '' and
      find_pass_no = '' and
      find_birth_date is null and
      find_cont_info = '') then
    exit;

  -- первоначальное состояние ФИО
  find_orig_fio = find_fio;
  orig_txt1 = '';
  orig_txt2 = '';
  orig_txt3 = '';

  is_critical = 0;
  res_same_fio = -1;
  res_same_birth_date = 0;
  res_same_pass_no = 0;
  res_same_pass_date = 0;
  res_same_contact = 0;
  res_same_code = 0;
  critical_found = 0;

  -- Убираем лишние символы из ФИО
  if (find_fio <> '') then
  begin
    select full_text, txt1, txt2, txt3
    from hp_fio_clear(:find_fio)
    into :find_fio, :orig_txt1, :orig_txt2, :orig_txt3;
  end

  -- Поиск по параметрам кроме ФИО
  if (find_birth_date is not null or
      find_pass_no <> '' or
      find_cont_info <> '' or
      find_code <> '') then
  begin
    if (find_code = '') then
      find_code = 'AAAAAAAAAA';
    if (find_pass_no = '') then
      find_pass_no = 'AAAAAAAAAA';
    if (find_cont_info = '') then
      find_cont_info = 'AAAAAAAAAA';
    if (find_birth_date is null) then
      find_birth_date = '01.01.1900';

    -- Просматриваем список клиентов совпадающие по критериям:
    for
        -- Одинаковые ИНН
        select distinct c_id, c_birth_date, c_name_fio, c_alternat_fio, c_pass_no, c_pass_date, iif(:find_code = c_ext_inn, c_ext_inn, c_inn), c_pincode
        from clients
        where c_id <> :find_c_id and
              c_id <> 0 and
              :find_code in (c_inn, c_ext_inn) and
              (:is_phys_face = -1 or c_type = :is_phys_face)

        -- Одинаковый пин-код (для АЗ)
        union
        select c_id, c_birth_date, c_name_fio, c_alternat_fio, c_pass_no, c_pass_date, iif(:find_code = c_ext_inn, c_ext_inn, c_inn), c_pincode
        from clients
        where c_id <> :find_c_id and
              c_id <> 0 and
              c_pincode = :find_code and
              (:is_phys_face = -1 or c_type = :is_phys_face)

        -- Одинаковая ДР
        union
        select c_id, c_birth_date, c_name_fio, c_alternat_fio, c_pass_no, c_pass_date, iif(:find_code = c_ext_inn, c_ext_inn, c_inn), c_pincode
        from clients
        where c_id <> :find_c_id and
              c_id <> 0 and
              c_birth_date = :find_birth_date and
              c_birth_date <> '01.01.1900' and
              (:is_phys_face = -1 or c_type = :is_phys_face)

        -- Одинаковый номер паспорта
        union
        select c_id, c_birth_date, c_name_fio, c_alternat_fio, c_pass_no, c_pass_date, iif(:find_code = c_ext_inn, c_ext_inn, c_inn), c_pincode
        from clients
        where c_id <> :find_c_id and
              c_id <> 0 and
              c_index_c_pass_no = :find_pass_no and
              (:find_citizen = 0 or
              c_citizen = :find_citizen) and
              (:is_phys_face = -1 or c_type = :is_phys_face)

        -- Одинаковые контакты
        union
        select c_id, c_birth_date, c_name_fio, c_alternat_fio, c_pass_no, c_pass_date, iif(:find_code = c_ext_inn, c_ext_inn, c_inn), c_pincode
        from hp_find_client_by_continfo(:find_c_id, :find_cont_info, :is_phys_face)
        into :c_id, :tmp_birth_date, :tmp_c_name_fio, :tmp_c_name_fio2, :tmp_pass_no, :tmp_pass_date, :tmp_inn,
             :tmp_pincode
    do
    begin
      if (tmp_c_name_fio is null) then
        tmp_c_name_fio = '';
      if (tmp_c_name_fio2 is null) then
        tmp_c_name_fio2 = '';
      if (tmp_inn is null) then
        tmp_inn = '';
      if (tmp_pincode is null) then
        tmp_pincode = '';
      if (tmp_pass_no is null) then
        tmp_pass_no = '';

      tmp_c_name_fio = f_replace_all(tmp_c_name_fio, '  ', ' ');
      tmp_c_name_fio = trim(tmp_c_name_fio);
      tmp_c_name_fio2 = f_replace_all(tmp_c_name_fio2, '  ', ' ');
      tmp_c_name_fio2 = trim(tmp_c_name_fio2);
      tmp_pass_no = f_replace_all(tmp_pass_no, ' ', '');
      tmp_pass_no = trim(tmp_pass_no);
      tmp_inn = f_replace_all(tmp_inn, ' ', '');
      tmp_inn = trim(tmp_inn);
      tmp_pincode = f_replace_all(tmp_pincode, ' ', '');
      tmp_pincode = trim(tmp_pincode);

      is_critical = 0;
      res_same_birth_date = 0;
      res_same_pass_no = 0;
      res_same_pass_date = 0;
      res_same_contact = 0;
      res_same_code = 0;

      if (find_code = tmp_inn) then
        res_same_code = 1;
      if (find_code = tmp_pincode) then
        res_same_code = 1;
      if (find_pass_no = tmp_pass_no) then
        res_same_pass_no = 1;
      if (find_birth_date = tmp_birth_date) then
        res_same_birth_date = 1;
      if (find_pass_date = tmp_pass_date) then
        res_same_pass_date = 1;

      if (find_cont_info <> 'AAAAAAAAAA') then
      begin
        if (is_new_contdata = 0) then
        begin
          if (exists(select *
                     from objectprops, objectvalues, refdata
                     where opr_object_kind = 0 and
                           opr_object_id = :c_id and
                           opr_subject_kind = -1 and
                           opr_subject_id = rd_id and
                           rd_main_code = 'ContactInfo' and
                           rd_code <> 'ContactInfo' and
                           opr_id = ovs_from_opr_id and
                           ovs_value = :find_cont_info)) then
          begin
            res_same_contact = 1;
          end
        end else
        begin
          clt_id = f_cont_id(find_cont_info, 'null');
          if (clt_id is not null) then
          begin
            if (exists(
                      select * from contdata
                      where cda_from_c_id = :c_id and
                            cda_from_clt_id = :clt_id))
            then res_same_contact = 1;
          end
        end
      end

      res_same_fio = -1; -- ФИО не похоже
      is_fio_ident = 0;
      if (find_fio <> '') then
      begin
        -- Сравниваем найденные ФИО с исходным вариантом ФИО
        if (find_orig_fio = tmp_c_name_fio or
            find_orig_fio = tmp_c_name_fio2) then
        begin
          res_same_fio = 2;
          is_fio_ident = 1;
        end

        -- Если не равен, проверяем по частям
        i = 1;
        txt3_1 = '';
        txt3_2 = '';
        while (i <= 2) do
        begin
          if (i = 1) then
            tmp_fio_text = tmp_c_name_fio; -- Сначала на кириллице
          if (i = 2) then
            tmp_fio_text = tmp_c_name_fio2; -- Следующий раз не латинице

          -- проверка после очистки от лишних символов
          if (is_fio_ident = 0) then
          begin
            select full_text, txt1, txt2, txt3
            from hp_fio_clear(:tmp_fio_text)
            into :tmp_fio_text, txt1, txt2, txt3;

            if (i = 1) then
              txt3_1 = txt3;
            if (i = 2) then
              txt3_2 = txt3;

            if ((orig_txt3 <> '' and
                txt3 <> '') or
                (orig_txt3 = '' and
                txt3 = '')) then
            begin
              -- если оба с отчеством или оба без отчества
              fio1 = find_fio;
              fio2 = tmp_fio_text;
              xfio1 = orig_txt2 || ' ' || orig_txt1;
              if (orig_txt3 <> '') then
                xfio1 = xfio1 || ' ' || orig_txt3;
            end
            else
            begin
              -- проверяем без отчества
              fio1 = orig_txt1 || ' ' || orig_txt2;
              fio2 = txt1 || ' ' || txt2;
              xfio1 = orig_txt2 || ' ' || orig_txt1;
            end
            if (fio1 = fio2 or
                xfio1 = fio2) then
            begin
              res_same_fio = 2; -- ФИО одинаковое
              is_fio_ident = 1;
            end
          end
          -- если точных совпадений не найдено, ищем по Левенштейну (с ошибками и перестановкой местами)
          if (is_fio_ident = 0) then
          begin
            if (f_levenstain_is_same(:fio1, :fio2) = 1) then
            begin
              res_same_fio = 1; -- ФИО очень похоже
              is_fio_ident = 1;
            end
            else
            begin
              if (f_levenstain_is_same(:xfio1, :fio2) = 1) then
              begin
                res_same_fio = 1; -- ФИО очень похоже
                is_fio_ident = 1;
              end
            end
          end
          i = i + 1;
        end

        -- проверка по количеству слов через Левенштейна
        if (is_fio_ident = 0) then
        begin
          if (exists(select object_id
                     from hp_find_from_namelist(:c_id, 0, :find_fio))) then
          begin
            res_same_fio = 0; -- ФИО похоже
            is_fio_ident = 1;
            if (orig_txt3 <> '') then
            begin
              if (txt3_1 <> '' and
                  txt3_2 = '') then
              begin
                if (f_levenstain_is_same(:orig_txt3, :txt3_1) = 0) then
                begin
                  res_same_fio = -1;
                  is_fio_ident = 0;
                end
              end
              if (txt3_1 = '' and
                  txt3_2 <> '') then
              begin
                if (f_levenstain_is_same(:orig_txt3, :txt3_2) = 0) then
                begin
                  res_same_fio = -1;
                  is_fio_ident = 0;
                end
              end
              if (txt3_1 <> '' and
                  txt3_2 <> '') then
              begin
                if (f_levenstain_is_same(:orig_txt3, :txt3_1) = 0 and
                    f_levenstain_is_same(:orig_txt3, :txt3_2) = 0) then
                begin
                  res_same_fio = -1;
                  is_fio_ident = 0;
                end
              end
            end
          end
        end
      end

      if ((res_same_code = 1) or
          (res_same_pass_no = 1) or
          (res_same_birth_date = 1 and
          is_fio_ident = 1) or
          (res_same_contact = 1 and
          is_fio_ident = 1) or
          (res_same_pass_date = 1 and
          res_same_fio >= 1)) then
      begin
        is_critical = 1;
        critical_found = 1;
        suspend;
      end
      else
      begin
        if (res_same_contact = 1 and
            show_just_critical = 0) then
        begin
          suspend;
        end
      end
    end
  end -- Конец поиска другим параметрам кроме ФИО

  if (critical_found = 1) then
    exit; -- если найдены критические совпадения, завершаем работу
  if (show_just_critical = 1) then
    exit; -- Если искали только критических, завершаем работу

  -- если кроме ФИО ничего не указано, производим поиск только по ФИО
  if (find_fio <> '') then
  begin
    -- Список похожих по ФИО
    for select c_id, c_name_fio, c_alternat_fio, c_birth_date, c_index_c_pass_no, c_pass_date
        from clients
        where c_id <> :find_c_id and
              c_name_fio = :find_orig_fio and
              (:is_phys_face = -1 or c_type = :is_phys_face)
        union
        select object_id, c_name_fio, c_alternat_fio, c_birth_date, c_index_c_pass_no, c_pass_date
        from hp_find_from_namelist(0, 0, :find_fio), clients
        where object_id = c_id and
              c_id <> :find_c_id and
              object_id <> 0 and
              (:is_phys_face = -1 or c_type = :is_phys_face)
        into :c_id, :tmp_c_name_fio, :tmp_c_name_fio2, :tmp_birth_date, :tmp_pass_no, :tmp_pass_date
    do
    begin
      if (tmp_c_name_fio is null) then
        tmp_c_name_fio = '';
      if (tmp_c_name_fio2 is null) then
        tmp_c_name_fio2 = '';
      tmp_c_name_fio = f_replace_all(tmp_c_name_fio, '  ', ' ');
      tmp_c_name_fio = trim(tmp_c_name_fio);
      tmp_c_name_fio2 = f_replace_all(tmp_c_name_fio2, '  ', ' ');
      tmp_c_name_fio2 = trim(tmp_c_name_fio2);

      critical_found = 0;
      is_critical = 0;

      res_same_fio = 0; -- ФИО похоже
      res_same_birth_date = 0;
      res_same_pass_no = 0;
      res_same_pass_date = 0;
      res_same_contact = 0;
      res_same_code = 0;

      is_fio_ident = 1;
      txt3_1 = '';
      txt3_2 = '';
      -- Сравниваем найденные ФИО с исходным вариантом ФИО
      if (find_orig_fio = tmp_c_name_fio or
          find_orig_fio = tmp_c_name_fio2) then
        res_same_fio = 2;
      else
      begin
        -- Если не равен, проверяем по частям
        i = 1;
        while (i <= 2) do
        begin
          if (i = 1) then
            tmp_fio_text = tmp_c_name_fio; -- Сначала на кириллице
          if (i = 2) then
            tmp_fio_text = tmp_c_name_fio2; -- Следующий раз не латинице

          -- проверка после очистки от лишних символов
          select full_text, txt1, txt2, txt3
          from hp_fio_clear(:tmp_fio_text)
          into :tmp_fio_text, txt1, txt2, txt3;

          if (i = 1) then
            txt3_1 = txt3;
          if (i = 2) then
            txt3_2 = txt3;

          if ((orig_txt3 <> '' and
              txt3 <> '') or
              (orig_txt3 = '' and
              txt3 = '')) then
          begin
            -- если оба с отчеством или оба без отчества
            fio1 = find_fio;
            fio2 = tmp_fio_text;
            xfio1 = orig_txt2 || ' ' || orig_txt1;
            if (orig_txt3 <> '') then
              xfio1 = xfio1 || ' ' || orig_txt3;
          end
          else
          begin
            -- проверяем без отчества
            fio1 = orig_txt1 || ' ' || orig_txt2;
            fio2 = txt1 || ' ' || txt2;
            xfio1 = orig_txt2 || ' ' || orig_txt1;
          end
          if (fio1 = fio2 or
              xfio1 = fio2) then
          begin
            res_same_fio = 2; -- ФИО одинаковое
          end
          -- если точных совпадений не найдено, ищем по Левенштейну (с ошибками и перестановкой местами)
          if (f_levenstain_is_same(:fio1, :fio2) = 1) then
          begin
            res_same_fio = 1; -- ФИО очень похоже
          end
          else
          begin
            if (f_levenstain_is_same(:xfio1, :fio2) = 1) then
            begin
              res_same_fio = 1; -- ФИО очень похоже
            end
          end
          -- если найдены сходства, и в обеих вариантах есть отчетство, проверяем совпадение отчества
          i = i + 1;
        end

        if (orig_txt3 <> '' and
            is_fio_ident = 1) then
        begin
          if (txt3_1 <> '' and
              txt3_2 = '') then
          begin
            if (f_levenstain_is_same(:orig_txt3, :txt3_1) = 0) then
            begin
              res_same_fio = -1;
              is_fio_ident = 0;
            end
          end
          if (txt3_1 = '' and
              txt3_2 <> '') then
          begin
            if (f_levenstain_is_same(:orig_txt3, :txt3_2) = 0) then
            begin
              res_same_fio = -1;
              is_fio_ident = 0;
            end
          end
          if (txt3_1 <> '' and
              txt3_2 <> '') then
          begin
            if (f_levenstain_is_same(:orig_txt3, :txt3_1) = 0 and
                f_levenstain_is_same(:orig_txt3, :txt3_2) = 0) then
            begin
              res_same_fio = -1;
              is_fio_ident = 0;
            end
          end
        end

      end
      if (is_fio_ident = 1) then
        suspend;
    end -- Конец поиска по ФИО
  end

end
