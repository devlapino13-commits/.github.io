CREATE PROCEDURE AAREP_FLOAT_TO_ANNUITET(
    FLAT_PERC DECIMAL(18,7),
    PERIOD_REAL INTEGER)
RETURNS (
    RESULT_PERC DECIMAL(18,4))
AS
declare variable pow double precision;
declare variable annu double precision;
declare variable annu_need double precision;
declare variable a2 double precision;
declare variable min1 double precision;
declare variable min2 double precision;
declare variable perc double precision;
declare variable perc2 double precision;
declare variable amount double precision;
declare variable k integer;
declare variable annu_need_a2 double precision;
declare variable ext double precision;
begin
  /* Процедура переводит простую %% ставку на остаток (аннуитет) */
  result_perc = 0;
  amount = 100000;
  perc2 = 0;
  if (period_real > 1) then
  begin
    ext = 0.01;
    annu_need = cast(amount / period_real + amount * flat_perc / 100 as double precision);
    perc2 = 10000;
    perc = cast(perc2 / 100 as decimal(15,9));
    pow = power((1 + perc), period_real);
    annu = amount * cast((perc * pow) as double precision);
    if (pow <> 1) then
      annu = annu / (pow - 1);
    else
      annu = 0;
    a2 = cast(annu as decimal(15,0));
    if (perc2 = 0) then
      a2 = 0;
    k = 0;
    min2 = a2;
    annu_need_a2 = annu_need - a2;
    if (annu_need_a2 < 0) then
      annu_need_a2 = -1 * annu_need_a2;
    while ((annu_need_a2 > ext) and (k < 100000)) do
    begin
      k = k + 1;
      if (a2 > annu_need) then
        perc2 = perc2 - perc2 / 2;
      if (a2 < annu_need) then
        perc2 = perc2 + perc2 / 2;
      perc2 = cast(perc2 as decimal(15,7));
      perc = cast(perc2 / 100 as decimal(15,9));
      if (perc2 <> 0) then
      begin
        pow = power((1 + perc), period_real);
        annu = amount * cast((perc * pow) as double precision);
        if (pow <> 1) then
          annu = annu / (pow - 1);
        else
          annu = 0;
        a2 = cast(annu as decimal(15,0));
      end
      else
      begin
        a2 = 0;
      end
      annu_need_a2 = annu_need - a2;
      if (annu_need_a2 < 0) then
        annu_need_a2 = -1 * annu_need_a2;
      if (min2 > annu_need_a2) then
      begin
        min2 = annu_need_a2;
        min1 = perc2;
      end
    end
    if (k = 100000) then
    begin
      perc2 = min1;
    end
  end
  else
    perc2 = flat_perc;
  result_perc = perc2;
  suspend;
end
