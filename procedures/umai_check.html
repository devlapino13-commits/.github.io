CREATE PROCEDURE UMAI_CHECK(
    ESM_ID D_ID,
    FULL_SOURCE D_STR_8100,
    ANSWER D_STR_8100,
    ERROR_CODE D_BOOL)
AS
declare variable esa_id d_id;
declare variable esm_id_pay d_id;
begin
  error_code = coalesce(error_code, 0);

  update extsysmain em
  set em.esm_full_source = :full_source
  where em.esm_id  = :esm_id;

  insert into extsysanswer (esa_from_esm_id, esa_message)
  values (:esm_id, :answer)
  returning esa_id
  into :esa_id;

  if ((error_code = 0) and (f_json_str_get_value(:answer, 'StatusCode', null) = '1')) then
  begin
     insert into extsysmain (esm_project, esm_action, esm_date, esm_account, esm_object_kind,
                             esm_object_id, esm_status, esm_full_source, esm_amount, esm_from_k_id)
     select esm_project, 'pay', current_date, esm_account, esm_object_kind, esm_object_id,
            0, esm_full_source, esm_amount, esm_from_k_id
     from extsysmain
     where esm_id = :esm_id
     returning esm_id
     into :esm_id_pay;

     update finerefund fr
     set fr.fr_from_esm_id = :esm_id_pay
     where fr.fr_from_esm_id = :esm_id;

     update overpayment
     set ovp_pay_esm_id = :esm_id_pay
     where ovp_check_esm_id = :esm_id
       and ovp_status = 2;

     update nbkr_umai_transfer nut
     set nut.nut_from_esm_id_pay = :esm_id_pay
     where nut.nut_from_esm_id_check = :esm_id;

     update kommisrefund_cards kmr
     set kmr.kmr_pay_esm_id = :esm_id_pay
     where kmr.kmr_check_esm_id = :esm_id;

     update extsysmain em
     set em.esm_status = 1,
         em.esm_ext_datetime = current_timestamp
     where em.esm_id = :esm_id;
  end
  else
  begin
    update extsysmain em
    set em.esm_status = -1,
        em.esm_ext_datetime = current_timestamp
    where em.esm_id = :esm_id;

     update overpayment
     set ovp_status = 4
     where ovp_check_esm_id = :esm_id
       and ovp_status = 2;

    update nbkr_umai_transfer nut
    set nut.nut_status = -2
    where nut.nut_from_esm_id_check = :esm_id;

    update kommisrefund_cards kmr
    set kmr.kmr_status = -1
    where kmr.kmr_check_esm_id = :esm_id;

    update extsysanswer ea
    set ea.esa_status = -1
    where ea.esa_id = :esa_id;
  end
end
