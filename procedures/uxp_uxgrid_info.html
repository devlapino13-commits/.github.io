CREATE PROCEDURE UXP$UXGRID_INFO(
    GRID_UUID D_UUID,
    GRID_UUID_CODE D_STR_CODE = null)
RETURNS (
    RES_IS_UXFRITEM D_BOOL,
    RES_UUID D_UUID,
    RES_NAME D_STR_100,
    RES_MEMO D_STR_255,
    RES_HIDE_ON_GROUP D_BOOL,
    RES_GROUP_PANEL D_BOOL,
    RES_FIND_PANEL D_SMALLINT,
    RES_HIDE_SUMMARY D_BOOL,
    RES_DISABLE_EDIT D_BOOL,
    RES_AUTOWIDTH D_SMALLINT,
    RES_HEADER_AUTOHEIGHT D_BOOL,
    RES_SHOW_INFO_PANEL D_BOOL,
    RES_HIDE_INPLACE_EDITOR D_BOOL,
    RES_SHOW_EXCEL_IMPORT D_BOOL,
    RES_OWNER_USER D_STR_CODE,
    RES_HAS_EXCEL D_STR_CODE,
    RES_FORMATTING D_STR_8100,
    RES_TREE_CAN_MOVE D_BOOL,
    RES_ALLOW_APPEND D_BOOL,
    RES_UI_KIND D_STR_CODE,
    RES_CALENDAR_FIELDS D_STR_4096)
AS
declare variable ufi_control d_str_code;
declare variable ui_ufr_uuid d_uuid;
begin
  if (coalesce(grid_uuid, '') = '') then
    grid_uuid = uxf$uniq(grid_uuid_code);

  res_is_uxfritem = 1;
  select ugr_ufr_uuid
  from ux$grid
  where ugr_uuid = :grid_uuid
  into ui_ufr_uuid;

  if (row_count = 1) then
    res_is_uxfritem = 0;

  select ufi_control
  from ux$fritem
  where ufi_uuid = :grid_uuid
  into ufi_control;

  if (ui_ufr_uuid is not null and ufi_control not in ('grid', 'tree')) then
      exception e_default f_format('Grid_uuid control must be grid or tree, ufi_uuid: %1', uuid_to_char(grid_uuid));

  if (ui_ufr_uuid is null) then
    exception e_default 'Grid_uuid is not Grid items';

  res_uuid = grid_uuid;
  res_header_autoheight = 1;
  res_show_info_panel = 0;
  res_hide_inplace_editor = 0;
  res_show_excel_import = 0;
  res_owner_user = '';
  res_has_excel = '';
  res_formatting = '';
  res_tree_can_move = 0;
  res_allow_append = 0;

  if (res_is_uxfritem = 0) then
  begin
    for select ugr_name, ugr_memo, ugr_hide_on_group, ugr_group_panel,
               ugr_find_panel, ugr_hide_summary, ugr_disable_edit,
               ugr_autowidth, ugr_header_autoheight, ugr_show_info_panel, coalesce(ugr_formatting, ''),
               ugr_hide_inplace_editor, ugr_show_excel_import, ugr_owner_user,
               trim(iif(ugr_excel is null, 'Без Excel шаблона', 'Excel шаблона прикреплен')),
               ugr_tree_can_move, ugr_allow_append, ugr_ui_kind, ugr_calendar_fields
    from ux$grid
    where ugr_uuid = :grid_uuid
    into :res_name, :res_memo, :res_hide_on_group, :res_group_panel,
         :res_find_panel, :res_hide_summary, res_disable_edit,
         :res_autowidth, res_header_autoheight, res_show_info_panel, res_formatting,
         :res_hide_inplace_editor, :res_show_excel_import, :res_owner_user,
         :res_has_excel, res_tree_can_move, res_allow_append, :res_ui_kind, :res_calendar_fields
    do
      suspend;

    if (row_count = 0) then
      exception e_default 'Grid not found';

  end else
  begin
    res_memo = '';
    for select ufi_name, ufr_hide_on_group, ufr_group_panel, ufr_find_panel, ufr_hide_summary, ufr_grid_disable_edit,
               ufr_autowidth
        from ux$fritem, ux$frame
        where ufi_uuid = :grid_uuid and
              ufi_ufr_uuid = ufr_uuid and
              ufr_is_grid = 1
    into :res_name, :res_hide_on_group, :res_group_panel, :res_find_panel, :res_hide_summary, :res_disable_edit,
         :res_autowidth
    do
      suspend;

    if (row_count = 0) then
      exception e_default 'Grid not found';
  end
end
