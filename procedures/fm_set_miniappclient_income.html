CREATE PROCEDURE FM_SET_MINIAPPCLIENT_INCOME(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    MACI_IS_DOCUMENTED D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable clt_id d_integer;
declare variable clt_code d_str_code;
declare variable fcb_id d_integer;
declare variable obj_code d_str_code;
declare variable c_id d_id;
declare variable k_status d_smallint;
declare variable c_status d_smallint;
declare variable k_id d_id;
declare variable need_name d_bool;
declare variable check_count d_integer;
declare variable ma_id d_integer;
declare variable rd_id d_integer;
begin
  res_is_error = 0;
  res_text = 'Операция прошла успешно';

  obj_code = f_objkind(obj_kind, 'code');

  if (obj_code <> 'miniappclient_income') then
    exception e_default f_format('Incorrect obj_kind. miniappclient_income expected, actually: %1', obj_code);

  if (coalesce(obj_id, 0) > 0) then
  begin
    select m.ma_id
    from miniappclient mcl
    inner join miniapp m on m.ma_id = mcl.mac_from_ma_id
    where mcl.mac_id = :obj_id and
          mcl.mac_amount > 0 rows 1
    into :ma_id;

    select k_id, kc_from_c_id, k_status, c_verify_status
    from kontract k
    inner join knclient knc on knc.kc_from_k_id = k.k_id and
          knc.kc_amount_approved > 0
    inner join clients cl on cl.c_id = knc.kc_from_c_id
    where k.k_from_ma_app_id = :ma_id rows 1
    into k_id, c_id, k_status, c_status;

    select k_id, kc_from_c_id, k_status, c_verify_status
    from knclient, kontract, clients
    where kc_id = :obj_id and
          kc_from_k_id = k_id and
          kc_from_c_id = c_id
    into k_id, c_id, k_status, c_status;

    if (k_status = 1) then
    begin
      res_is_error = 1;
      res_text = 'Договор уже утвержден. Необходимо вернуть в процесс.';
      suspend;
      exit;
    end


    select rd_id
    from hp_get_check_loan_data(:k_id)
    into:rd_id;

    if (exists(select *
               from actionlist
               where acl_object_kind = 9 and
                     acl_object_id = :k_id and
                     acl_status = 1 and
                     acl_from_rd_id = :rd_id)) then
    begin
      res_is_error = 1;
      res_text = 'Договор находится на рассмотрении КК. Изменения запрещены.';
      suspend;
      exit;
    end
  end
  if (maci_is_documented is null) then
  begin
    res_is_error = 1;
    res_text = 'Вариант не выбран. Выберите один вариант из списка.';
    suspend;
    exit;
  end

  if (exists(select *
             from miniappclient_income maci
             where maci.maci_from_mac_id = :obj_id and
                   maci.maci_is_documented = :maci_is_documented)) then
  begin
    res_is_error = 1;
    res_text = 'Новое значение совпадает со старым значением, операция отменяется';
    suspend;
    exit;
  end

  update or insert into miniappclient_income (maci_from_mac_id, maci_is_documented, maci_datetime, maci_from_un_id)
  values (:obj_id, :maci_is_documented, current_timestamp, :un_id)
  matching (maci_from_mac_id);

  suspend;
end
