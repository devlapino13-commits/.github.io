CREATE PROCEDURE HP_RELATIONLINK_UPDATE(
    C_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0),
    IS_CANDIDATE INTEGER)
AS
declare variable c_id_tmp decimal(18,0);
declare variable rd_id_staff_staff integer;
declare variable rll_id integer;
declare variable rel_code varchar(50);
declare variable c_sex integer;
declare variable rd_id_candidate_staff integer;
declare variable rd_id_staff_old_staff integer;
declare variable new_rd_status integer;
begin
  if (is_candidate is null) then
    is_candidate = 0;

  select rd_id
  from refdata
  where rd_code = 'RelationLink.Staff_Staff'
  into :rd_id_staff_staff;

  select rd_id
  from refdata
  where rd_code = 'RelationLink.Candidate_Staff'
  into :rd_id_candidate_staff;

  select rd_id
  from refdata
  where rd_code = 'RelationLink.Staff_Old_Staff'
  into :rd_id_staff_old_staff;

  if ((rd_id_staff_staff is null) or (rd_id_candidate_staff is null) or (rd_id_staff_old_staff is null)) then
    exit;

  if (exists(select *
             from hr_pstatus
             where ps_from_c_id = :c_id and
                   ps_date_end >= '31.12.2999' and
                   ps_status = 1 and
                   ps_is_working = 1)) then
    is_candidate = 0; -- действующий сотрудник
  else
  if (exists(select *
             from hr_pstatus
             where ps_from_c_id = :c_id and
                   ps_date_end >= '31.12.2999' and
                   ps_status = 1 and
                   ps_is_working = 0)) then
    is_candidate = -1; -- бывший сотрудник
  else
  if (is_candidate <> 1) then
  begin
    update relationlink
    set rll_is_active = 0,
        rll_from_un_id = :un_id
    where (rll_client_1 = :c_id or
          rll_client_2 = :c_id) and
          rll_is_active = 1;
    exit; -- кандидат. Если в параметре не указан что кандидат,
-- значит он не имеет отношение к работе и завершаем работу
  end

  new_rd_status = (case
                     when (is_candidate = -1) then :rd_id_staff_old_staff
                     when is_candidate = 0 then :rd_id_staff_staff
                     when is_candidate = 1 then :rd_id_candidate_staff
                   end);

  update relationlink
  set rll_link_kind = :new_rd_status,
      rll_from_un_id = :un_id
  where (rll_client_1 = :c_id or
        rll_client_2 = :c_id) and
        rll_is_active = 1 and
        rll_link_kind <> :new_rd_status;

-- Если связанное лицо не сотрудник, аннулируем связь
  for select rll_id, rll_client_1
      from relationlink
      where rll_link_kind = :rd_id_staff_staff and
            rll_client_2 = :c_id and
            rll_is_active = 1
      union
      select rll_id, rll_client_2
      from relationlink
      where rll_link_kind = :rd_id_staff_staff and
            rll_client_1 = :c_id and
            rll_is_active = 1
      into :rll_id, :c_id_tmp
  do
  begin
    if (not(exists(select *
                   from hr_pstatus
                   where ps_from_c_id = :c_id_tmp and
                         ps_date_end >= '31.12.2999' and
                         ps_status = 1 and
                         ps_is_working = 1))) then
    begin
      update relationlink
      set rll_link_kind = :rd_id_staff_old_staff,
          rll_from_un_id = :un_id
      where rll_id = :rll_id;
    end
  end

-- просматриваем всех родственников, добавляем в список если тоже сотрудник
  select c_sex
  from clients
  where c_id = :c_id
  into :c_sex;

  for select rel_other_c_id, rel_code
      from hp_client_relation_tmp(:c_id, 4)
      into :c_id_tmp, :rel_code
  do
  begin
    if (c_id_tmp <> c_id) then
    begin
      rel_code = (select rel_code
                  from hp_get_relation_short_id(:rel_code, :c_sex));

      if (exists(select *
                 from hr_pstatus
                 where ps_from_c_id = :c_id_tmp and
                       ps_date_end >= '31.12.2999' and
                       ps_status = 1 and
                       ps_is_working = 1)) then
      begin
        update or insert into relationlink (rll_client_1, rll_client_2, rll_link_kind, rll_link_code, rll_date,
                                            rll_from_un_id, rll_is_active)
        values (:c_id, :c_id_tmp, :new_rd_status, :rel_code, current_date, :un_id, 1)
        matching (rll_client_1, rll_client_2);

        rel_code = (select rel_code
                    from hp_get_relation_reverse(:rel_code, :c_sex));

        update or insert into relationlink (rll_client_1, rll_client_2, rll_link_kind, rll_link_code, rll_date,
                                            rll_from_un_id, rll_is_active)
        values (:c_id_tmp, :c_id, :new_rd_status, :rel_code, current_date, :un_id, 1)
        matching (rll_client_1, rll_client_2);
      end
    end
  end
end
