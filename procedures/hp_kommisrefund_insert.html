CREATE PROCEDURE HP_KOMMISREFUND_INSERT
AS
declare variable amount d_integer;
declare variable amount_approved d_integer;
declare variable amount_1_4 d_currency;
declare variable amount_1_5 d_currency;
declare variable ok_phone d_str_20;
declare variable k_id id_type;
declare variable c_id id_type;
begin
  for select kc.kc_amount_approved, k.k_id, kc.kc_from_c_id
      from kontract k
      inner join kontstate ks on k.k_id = ks.ks_from_k_id
            and ks.ks_state = 1
            and ks.ks_ol_is_delete = 0
            and ks.ks_date_begin between '25.04.2022' and '21.09.2022'
      inner join knclient kc on k.k_id = kc.kc_from_k_id
            and kc.kc_amount_approved > 0
      inner join product pr on pr.pr_id = k.k_from_pr_id
            and pr.pr_code containing '.mbulak_cards.'
  where k.k_kommis_perc <> 1.4
  into :amount_approved, :k_id, :c_id
  do
  begin
    amount_1_5 = amount_approved * 1.5 / 100;
    amount_1_4 = amount_approved * 1.4 / 100;
    amount = amount_1_5 - amount_1_4;

    ok_phone = f_cont_info('clients', :c_id, 'cell_main', 'ok');

    if (ok_phone is null or ok_phone = '') then
    begin
      ok_phone = f_cont_info('clients', :c_id, 'cell_add', 'ok');
    end

    insert into kommisrefund_cards (kmr_id, kmr_from_k_id, kmr_from_c_id, kmr_amount, kmr_ok_cell_no)
    values (null, :k_id, :c_id, :amount, :ok_phone);
  end
end
