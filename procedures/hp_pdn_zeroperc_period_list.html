CREATE PROCEDURE HP_PDN_ZEROPERC_PERIOD_LIST
RETURNS (
    PERIOD_NO D_INTEGER)
AS
declare variable c_date d_date;
declare variable c_date_begin d_date;
begin
  c_date = current_date - extract(day from current_date) + 1;
  c_date_begin = '01.01.' || extract(year from c_date);

  if (datediff(month, c_date_begin, c_date) < 6) then
    c_date_begin = dateadd(month, -6, c_date);

  while (c_date >= c_date_begin) do
  begin
    period_no = extract(year from c_date) * 100 + extract(month from c_date);
    suspend;
    c_date = dateadd(month, -1, c_date);
  end

  for select distinct pzp_disb_period
  from pdn_zeroperc
  where pzp_disb_period < :period_no
  order by pzp_disb_period desc
  into period_no
  do
    suspend;
end
