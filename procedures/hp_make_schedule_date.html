CREATE PROCEDURE HP_MAKE_SCHEDULE_DATE(
    LOAN_AMOUNT DECIMAL(18,4),
    LOAN_REPAY_KOLICH INTEGER,
    LOAN_REPAY_DAY INTEGER,
    LOAN_DISB_DATE DATE,
    LOAN_PRODUCT_ID INTEGER,
    LOAN_KOMMIS_PERC DECIMAL(18,7),
    LOAN_KOMMIS_LGOTA DECIMAL(18,4),
    LOAN_GRACES VARCHAR(255),
    LOAN_GRACE_IS_MONTH INTEGER,
    LOAN_ADD_MONTH D_INTEGER = 0,
    REG_ID D_INTEGER = null)
AS
declare variable pr_percent_for_days integer;
declare variable pr_is_365 integer;
declare variable calc_kommis_amount decimal(18,4);
declare variable day_no integer;
declare variable first_dd date;
declare variable pr_week_or_month integer;
declare variable pr_wm_kolich integer;
declare variable pr_is_weekend_work integer;
declare variable pr_percent_method integer;
declare variable sh_date_prior date;
declare variable pr_round integer;
declare variable pr_round_perc integer;
declare variable pr_round_kommis integer;
declare variable cm_holiday_is_off integer;
declare variable isweekend integer;
declare variable go_sign integer;
declare variable orig_first_dd date;
declare variable total_grace integer;
declare variable sh_id integer;
declare variable sh_date date;
declare variable sh_days integer;
declare variable sh_os decimal(18,4);
declare variable sh_perc decimal(18,4);
declare variable sh_kommis decimal(18,4);
declare variable sh_grace integer;
declare variable db_id integer;
declare variable pr_code varchar(100);
declare variable is_mobile_office integer;
declare variable add_day integer;
declare variable min_days integer;
declare variable last_dd date;
declare variable is_new_schedule integer;
declare variable max_days d_integer;
declare variable another_schedule d_smallint;
declare variable first_weekend_weekday d_smallint;
declare variable second_weekend_weekday d_smallint;
begin
  db_id = f_get_db_id();
  if (loan_graces is null) then
    loan_graces = '';
  is_new_schedule = 0;
  if (loan_graces = 'new') then
    is_new_schedule = 1;
  min_days = 15;
  max_days = 45;

  total_grace = 0;
  if (loan_add_month is null) then
    loan_add_month = 0;

  if (loan_product_id is null) then
    exit;

  -- Тестовые данные вручную
  --  loan_disb_date='02.04.2016';     -- Дата выдачи
  --  loan_repay_day=0;                -- Предпочитаемая дата погашения
  --  loan_repay_kolich=12;            -- Количество погашений
  --  loan_product_id=173;             -- ID продукта
  --  loan_amount=40000;               -- Сумма кредита
  --  loan_percent=34.9;               -- Процентная ставка
  --  loan_kommis_perc=6.7;            -- Ставка доп.процента

  -- Либо можно заполнить через ID объекта
  --  obj_kind=9;
  --  obj_id=109010017709800201;

  -- Получаем необходимые настройки кредитного продукта
  select pr_percent_for_days, pr_is_360, pr_week_or_month, pr_wm_kolich, pr_is_weekend_work, pr_interest_method,
         pr_round, pr_round_perc, pr_round_admin, pr_code
  from product
  where pr_id = :loan_product_id
  into :pr_percent_for_days, pr_is_365, pr_week_or_month, pr_wm_kolich, pr_is_weekend_work, :pr_percent_method,
       :pr_round, :pr_round_perc, :pr_round_kommis, pr_code;

  if (loan_repay_day = -1 and
      pr_week_or_month <> 2) then
    loan_repay_day = 0;

  is_mobile_office = 0;
  if (db_id in (39801, 41701) and
      pr_code containing '.MOB.') then
    is_mobile_office = 1;

  pr_round = -pr_round;
  pr_round_perc = -pr_round_perc;
  pr_round_kommis = -pr_round_kommis;

  select cm_holiday_is_off --, cm_holiday_sat, cm_holiday_sun
  from company
  into cm_holiday_is_off; --, cm_holiday_sat, cm_holiday_sun;

  if (coalesce(pr_percent_for_days, 0) = 0) then
    exit;

  another_schedule = 0;

  --Проверяю страну и офис
  --Пока реализация требуется только для одного офиса, но в будущем нужно создать таблицу с тремя полями: region, first_weekend_weekday, second_weekend_weekday
  if (db_id in (39801, 41701) and
      reg_id = 1280400) then
    another_schedule = 1;

  if (another_schedule = 1) then
  begin
    --Офис Амана-Исламик Карасуу, обозначаю выходные дни
    if (reg_id = 1280400) then
    begin
      first_weekend_weekday = 0; --Воскресенье
    end
  end

  delete from temp_schedule
  where 1 = 1;

  -- Вычисляем сумму доп.процента с учетом предоставленных скидок
  if (db_id not in (39801, 41701) or loan_amount <= 11000000) then
    calc_kommis_amount = round(loan_amount * loan_kommis_perc / 100.0, :pr_round_kommis);
  else
    calc_kommis_amount = round(cast(loan_amount as d_integer) * loan_kommis_perc / 100.0, :pr_round_kommis);
  calc_kommis_amount = calc_kommis_amount - loan_kommis_lgota;
  if (calc_kommis_amount < 0) then
    calc_kommis_amount = 0.0;

  -- Запоминаем оринигальную дату первого погашения, влияет на сумму первого погашения
  if (pr_week_or_month = 2) then -- Погашения ежемесячные
    orig_first_dd = dateadd(month, :pr_wm_kolich, loan_disb_date);
  if (pr_week_or_month = 1) then -- Погашения еженедельные
    orig_first_dd = dateadd(week, :pr_wm_kolich, loan_disb_date);
  if (pr_week_or_month = 0) then -- Погашения еежедневные
    orig_first_dd = dateadd(day, :pr_wm_kolich, loan_disb_date);

  if (is_mobile_office = 1) then -- Продукты для мобильных офисов
  begin
    orig_first_dd = dateadd(week, 4, loan_disb_date); -- Первый платеж ставим через 4 недели
    if (loan_add_month = 1) then
      orig_first_dd = dateadd(day, 28, orig_first_dd);
    if (loan_add_month = 2) then
      orig_first_dd = dateadd(day, 63, orig_first_dd);

    first_dd = orig_first_dd;
  end
  else
    orig_first_dd = dateadd(month, loan_add_month, orig_first_dd);

  -- Определяем первую дату погашения
  if (is_mobile_office = 0) then
  begin
    if (pr_week_or_month = 2) then -- Погашения ежемесячные
    begin
      if (loan_repay_day = -1) then -- конец месяца
      begin
        first_dd = f_date_month_end(loan_disb_date);
        if (datediff(day, loan_disb_date, first_dd) < min_days) then
          first_dd = f_date_month_end(first_dd + 1);
      end
      if (loan_repay_day = 0) then
        first_dd = dateadd(month, :pr_wm_kolich + loan_add_month, loan_disb_date); -- согласно даты выдачи
      if (loan_repay_day > 0) then -- Указанный день месяца
      begin
        day_no = extract(day from loan_disb_date);
        if (day_no < loan_repay_day) then
        begin
          first_dd = dateadd(day, loan_repay_day - day_no, dateadd(month, :pr_wm_kolich - 1 + loan_add_month, loan_disb_date));
          if (loan_repay_day - day_no < min_days) then
            first_dd = dateadd(month, 1, first_dd);
        end
        else
        begin
          first_dd = dateadd(month, :pr_wm_kolich, loan_disb_date);
          day_no = extract(day from first_dd);
          first_dd = dateadd(day, loan_repay_day - day_no, first_dd);

          if (datediff(day, loan_disb_date, dateadd(month, 1, first_dd)) <= max_days) then
            first_dd = dateadd(month, 1, first_dd);

          if (datediff(day, loan_disb_date, first_dd) < min_days) then
            first_dd = dateadd(month, 1, first_dd);
          first_dd = dateadd(month, loan_add_month, first_dd);
        end
        if (first_dd < dateadd(month, 1 + loan_add_month, loan_disb_date) and
            db_id = 81001 and
            (loan_repay_kolich + loan_add_month) <> ((loan_repay_kolich + loan_add_month) / 12) * 12) then
          last_dd = dateadd(month, loan_repay_kolich + loan_add_month, loan_disb_date);
      end
    end
    if (pr_week_or_month = 1) then
      first_dd = dateadd(week, :pr_wm_kolich, loan_disb_date); -- Погашения еженедельные
    if (pr_week_or_month = 0) then
      first_dd = dateadd(day, :pr_wm_kolich, loan_disb_date); -- Погашения ежедневные
  end

  sh_id = 0;
  sh_date = loan_disb_date;
  sh_days = 0;
  sh_os = 0.0;
  sh_perc = 0.0;
  sh_kommis = calc_kommis_amount;
  sh_grace = 0;

  -- Вставляем первую запись в график погашений
  insert into temp_schedule (sh_id, sh_date, sh_days, sh_os, sh_perc, sh_kommis, sh_grace)
  values (:sh_id, :sh_date, :sh_days, :sh_os, :sh_perc, :sh_kommis, :sh_grace);
  sh_kommis = 0;

  if (db_id = 81001 and
      loan_disb_date >= '01.05.2017') then
  begin
    -- вычисляем другие даты в графике погашений
    sh_date_prior = loan_disb_date;
    total_grace = 0;
    while (sh_id < loan_repay_kolich) do
    begin
      sh_id = sh_id + 1;
      sh_grace = null;

      if (is_mobile_office = 0) then
      begin
        if (loan_repay_day = -1) then -- Погашения ежемесячные в конце месяца
        begin
          sh_date = dateadd(month, :pr_wm_kolich * (sh_id - 1), first_dd);
          sh_date = f_date_month_end(sh_date);
        end
        else
        begin
          if (pr_week_or_month = 2) then
            sh_date = dateadd(month, :pr_wm_kolich * (sh_id - 1), first_dd); -- Погашения ежемесячные
          if (pr_week_or_month = 1) then
            sh_date = dateadd(week, :pr_wm_kolich * (sh_id - 1), first_dd); -- Погашения еженедельные
          if (pr_week_or_month = 0) then
            sh_date = dateadd(day, :pr_wm_kolich * (sh_id - 1), first_dd); -- Погашения ежедневные
        end
        if (sh_id = loan_repay_kolich and
            pr_week_or_month = 2) then
        begin
          if ((db_id = 81001 and
              (loan_repay_kolich + loan_add_month) = ((loan_repay_kolich + loan_add_month) / 12) * 12) or (db_id <> 81001)) then
          begin
            sh_date = dateadd(year, ((loan_repay_kolich + loan_add_month) * pr_wm_kolich) / 12, loan_disb_date);
            if (pr_percent_method <> -1) then
              sh_date = sh_date - 1; -- если не ИФП, то предыдущая дата
          end
          if (last_dd is not null) then
            sh_date = last_dd;
        end

        if (extract(weekday from sh_date) in (0, 6)) then -- Погашение попадает в сб или вс
        begin
          if ((db_id = 81001 and
              (loan_repay_kolich + loan_add_month) = ((loan_repay_kolich + loan_add_month) / 12) * 12 and
              sh_id = loan_repay_kolich) or (db_id <> 81001 and
              sh_id = loan_repay_kolich)) then
          begin
            if (pr_percent_method <> -1) then
            begin
              if (extract(weekday from sh_date) = 0) then
                sh_date = dateadd(day, -2, sh_date);
              else
                sh_date = dateadd(day, -1, sh_date);
            end
          end
          else
          begin
            if (pr_is_weekend_work = 0) then -- перенос в ближайший рабочий день: суббота на пятницу, воскресенье на понедельник
              sh_date = iif(extract(weekday from sh_date) = 6, sh_date - 1, sh_date + 1);

            if (pr_is_weekend_work in (3, 4)) then -- перенос в понедельник, погашения в опер.дни сотрудника (4) неактуальна
            begin
              if (extract(weekday from sh_date) = 6) then
                sh_date = dateadd(day, 2, sh_date);
              else
                sh_date = dateadd(day, 1, sh_date);
            end
          end
        end

        if (cm_holiday_is_off = 0 and
            not(pr_is_weekend_work in (1, 2))) then
        begin
          if (sh_id <= loan_repay_kolich or pr_percent_method <> -1) then
          begin
            -- Проверяем на праздники
            isweekend = 0;
            if (exists(select *
                       from sa_calendar
                       where scal_month = extract(month from :sh_date) and
                             scal_day = extract(day from :sh_date) and
                             scal_active = 1)) then
              isweekend = 1;

            if (isweekend = 0) then
            begin
              -- Проверяем на календарь рабочих дней
              if (exists(select *
                         from sa_calendays
                         where scd_work_hours = 0 and
                               scd_date = :sh_date)) then
                isweekend = 1;
            end

            while (isweekend = 1) do
            -- двигаем вперед пока не нашли рабочий день
            begin
              go_sign = 1;
              if ((loan_repay_kolich + loan_add_month) = ((loan_repay_kolich + loan_add_month) / 12) * 12 and
                  loan_repay_kolich = sh_id) then
                go_sign = -1;

              isweekend = 0;
              sh_date = sh_date + go_sign;

              if (exists(select *
                         from sa_calendar
                         where scal_month = extract(month from :sh_date) and
                               scal_day = extract(day from :sh_date) and
                               scal_active = 1)) then
                isweekend = 1;

              if (isweekend = 0) then
              begin
                if (exists(select *
                           from sa_calendays
                           where scd_work_hours = 0 and
                                 scd_date = :sh_date)) then
                  isweekend = 1;
              end
              if (extract(weekday from sh_date) in (0, 6)) then
                isweekend = 1;
            end
          end
        end
      end
      else
      begin
        add_day = (sh_id / 3);
        if (loan_add_month = 0) then
          sh_date = dateadd(week, 4 * (sh_id - 1) + add_day, first_dd); -- Погашения по неделям
        if (loan_add_month = 1) then
          sh_date = dateadd(week, 4 * (sh_id - 1) + add_day + 4, first_dd); -- Погашения по неделям
        if (loan_add_month = 2) then
          sh_date = dateadd(week, 4 * (sh_id - 1) + add_day + 9, first_dd); -- Погашения по неделям
      end

      if (db_id = 81001 and
          pr_code containing '.EMP.' and
          pr_code containing '.reliable_emp.' and
          pr_code containing '.IND.') then
        sh_grace = 0;
      else
      begin
        sh_grace = 0;
        if (sh_id < loan_repay_kolich) then
        begin
          if (is_new_schedule = 1 and
              extract(month from sh_date) in (1, 2)) then
            sh_grace = 1;
          else
          begin
            if (loan_grace_is_month = 0) then
            begin
              if (position('[' || sh_id || ']', loan_graces) > 0) then
                sh_grace = 1;
            end
            else
            begin
              if (position('[' || extract(month from sh_date) || ']', loan_graces) > 0) then
                sh_grace = 1;
            end
          end
        end
      end

      sh_days = (select res_days
                 from fhp_agein360(:sh_date_prior, :sh_date, :pr_is_365));

      insert into temp_schedule (sh_id, sh_date, sh_days, sh_os, sh_perc, sh_kommis, sh_grace)
      values (:sh_id, :sh_date, :sh_days, :sh_os, :sh_perc, :sh_kommis, :sh_grace);

      sh_date_prior = sh_date;
      total_grace = total_grace + sh_grace;
    end -- даты вычислены
  end
  else
  begin
    -- вычисляем другие даты в графике погашений
    sh_date_prior = loan_disb_date;
    total_grace = 0;
    while (sh_id < loan_repay_kolich) do
    begin
      sh_id = sh_id + 1;
      sh_grace = null;

      if (is_mobile_office = 0) then
      begin
        if (pr_week_or_month = 2) then
          sh_date = dateadd(month, :pr_wm_kolich * (sh_id - 1), first_dd); -- Погашения ежемесячные
        if (pr_week_or_month = 1) then
          sh_date = dateadd(week, :pr_wm_kolich * (sh_id - 1), first_dd); -- Погашения еженедельные
        if (pr_week_or_month = 0) then
          sh_date = dateadd(day, :pr_wm_kolich * (sh_id - 1), first_dd); -- Погашения ежедневные

        if (sh_id = loan_repay_kolich) then
        begin
          if (pr_week_or_month = 2) then
          begin
            if ((loan_repay_kolich + loan_add_month) = ((loan_repay_kolich + loan_add_month) / 12) * 12) then
            begin
              sh_date = dateadd(year, ((loan_repay_kolich + loan_add_month) * pr_wm_kolich) / 12, loan_disb_date);
              if (pr_percent_method <> -1) then
                sh_date = sh_date - 1; -- если не ИФП, то предыдущая дата
            end
            else
              sh_date = dateadd(month, (loan_repay_kolich + loan_add_month) * pr_wm_kolich, loan_disb_date);
          end
        end

        if (extract(weekday from sh_date) in (0, 6) -- Погашение попадает в сб или вс
            and
            another_schedule = 0) then
        begin
          if (sh_id < loan_repay_kolich) then
          begin
            if (pr_is_weekend_work = 0) then -- перенос в ближайший рабочий день: суббота на пятницу, воскресенье на понедельник
              sh_date = iif(extract(weekday from sh_date) = 6, sh_date - 1, sh_date + 1);

            if (pr_is_weekend_work in (3, 4)) then -- перенос в понедельник, погашения в опер.дни сотрудника (4) неактуальна
            begin
              if (extract(weekday from sh_date) = 6) then
                sh_date = dateadd(day, 2, sh_date);
              else
                sh_date = dateadd(day, 1, sh_date);
            end
          end
          else
          begin
            if (pr_percent_method <> -1) then
            begin
              if (extract(weekday from sh_date) = 0) then
                sh_date = dateadd(day, -2, sh_date);
              else
                sh_date = dateadd(day, -1, sh_date);
            end
          end
        end

        if (another_schedule = 1) then
        begin
          if (sh_id < loan_repay_kolich) then
          begin
            if (extract(weekday from sh_date) = first_weekend_weekday) then
            begin
              sh_date = dateadd(day, 1, sh_date);
            end
          end
          else
          begin
            if (pr_percent_method <> -1) then
            begin
              if (extract(weekday from sh_date) = first_weekend_weekday) then
                sh_date = dateadd(day, -1, sh_date);
            end
          end
        end

        if (cm_holiday_is_off = 0 and
            not(pr_is_weekend_work in (1, 2))) then
        begin
          if (sh_id < loan_repay_kolich or pr_percent_method <> -1) then
          begin
            -- Проверяем на праздники
            isweekend = 0;
            if (exists(select *
                       from sa_calendar
                       where scal_month = extract(month from :sh_date) and
                             scal_day = extract(day from :sh_date) and
                             scal_active = 1)) then
              isweekend = 1;

            if (isweekend = 0 and
                another_schedule = 0) then
            begin
              -- Проверяем на календарь рабочих дней
              if (exists(select *
                         from sa_calendays
                         where scd_work_hours = 0 and
                               scd_date = :sh_date)) then
                isweekend = 1;
            end

            if (another_schedule = 1 and
                isweekend = 0) then
            begin
              isweekend = iif(extract(weekday from sh_date) <> first_weekend_weekday, 0, 1);
            end

            while (isweekend = 1) do
            -- двигаем вперед пока не нашли рабочий день
            begin
              go_sign = 1;
              if ((loan_repay_kolich + loan_add_month) = ((loan_repay_kolich + loan_add_month) / 12) * 12 and
                  loan_repay_kolich = sh_id) then
                go_sign = -1;

              isweekend = 0;
              sh_date = sh_date + go_sign;

              if (exists(select *
                         from sa_calendar
                         where scal_month = extract(month from :sh_date) and
                               scal_day = extract(day from :sh_date) and
                               scal_active = 1)) then
                isweekend = 1;

              if (isweekend = 0 and
                  another_schedule = 0) then
              begin
                if (exists(select *
                           from sa_calendays
                           where scd_work_hours = 0 and
                                 scd_date = :sh_date)) then
                  isweekend = 1;
              end
              if (extract(weekday from sh_date) in (0, 6) and
                  another_schedule = 0) then
                isweekend = 1;

              if (another_schedule = 1 and
                  isweekend = 0) then
              begin
                isweekend = iif(extract(weekday from sh_date) <> first_weekend_weekday, 0, 1);
              end
            end
          end
        end
      end
      else
      begin
        add_day = (sh_id / 3);
        sh_date = dateadd(week, 4 * (sh_id - 1) + add_day, first_dd); -- Погашения ежемесячные
      end

      sh_grace = 0;
      if (sh_id < loan_repay_kolich and
          db_id = 81001) then
      begin
        if (is_new_schedule = 1 and
            extract(month from sh_date) in (1, 2)) then
          sh_grace = 1;
        else
        begin
          if (loan_grace_is_month = 0) then
          begin
            if (position('[' || sh_id || ']', loan_graces) > 0) then
              sh_grace = 1;
          end
          else
          begin
            if (position('[' || extract(month from sh_date) || ']', loan_graces) > 0) then
              sh_grace = 1;
          end
        end
      end

      if (sh_grace is null) then
        sh_grace = 0;
      if (db_id in (39801, 41701) and
          pr_code containing '.jonokoi.') then
      begin
        if (sh_id = 1) then
          sh_grace = 1;
      end
      sh_days = (select res_days
                 from fhp_agein360(:sh_date_prior, :sh_date, :pr_is_365));

      insert into temp_schedule (sh_id, sh_date, sh_days, sh_os, sh_perc, sh_kommis, sh_grace)
      values (:sh_id, :sh_date, :sh_days, :sh_os, :sh_perc, :sh_kommis, :sh_grace);

      sh_date_prior = sh_date;
      total_grace = total_grace + sh_grace;
    end -- даты вычислены
  end
