CREATE PROCEDURE HP_VERSION_HISTORY_LOCAL(
    ITEM_NAME D_STR_100 = null)
RETURNS (
    LAST_NAME D_STR_100,
    LAST_DATE D_DATETIME,
    MAX_DATE D_DATETIME,
    LAST_USER D_STR_CODE,
    LAST_DB_ID D_INTEGER,
    LAST_BODY D_STR_CODE)
AS
declare variable add_hour d_integer;
begin
  last_db_id = f_get_db_id();
  add_hour = 0;

  if (last_db_id = 81001) then
    add_hour = -3;

  if (last_db_id = 94401) then
    add_hour = -2;

  if (last_db_id = 39801) then
    add_hour = -1;

  item_name = upper(trim(coalesce(item_name, '')));

  for select vsl_name, vsl_ver_datetime, vsl_max_datetime, vsl_username, vsl_body_hash
  from versionlog
  where (vsl_name = :item_name or :item_name = '')
  into :last_name, :last_date, :max_date, :last_user, :last_body
  do
  begin
    if (add_hour <> 0) then
    begin
      last_date = dateadd(hour, add_hour, last_date);
      max_date = dateadd(hour, add_hour, max_date);
    end

    suspend;
  end
end
