Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_KKT_GET_CHECK_INTERVAL(
    USER_DATETIME D_DATETIME)
RETURNS (
    CHECK_INTERVAL D_INTEGER,
    REP_PRINT_MODE D_INTEGER)
AS
begin
  -- Процедура возвращает интервал (в минутах) проверки наличия чеков, для фонового потока печати чеков.
  -- Вызывается из Finstart, uKktUtils
  -- Реализация через процедуру, т.к. есть идея менять интервал в зависимости от времени суток
  check_interval = f_get_const('kkt_check_interval');
  rep_print_mode = f_get_const('kkt_print_mode');
  suspend;
end
