CREATE PROCEDURE HP_GET_EMPLOYER_REGION(
    REP_C_ID DECIMAL(18,0),
    REP_DATE DATE,
    REP_IN_REGION INTEGER)
RETURNS (
    REP_REGION_ID INTEGER)
AS
begin
  rep_region_id = null;
  select ps_from_reg_id
  from hr_pstatus
  where ps_from_c_id = :rep_c_id and
        ps_date_begin <= :rep_date and
        ps_date_end >= :rep_date and
        ps_status = 1
  order by ps_from_c_id desc, ps_date_begin desc, ps_is_working desc
  rows 1
  into :rep_region_id;

  if ((rep_region_id is not null) and
      (rep_in_region is not null)) then
  begin
    if (not(exists(select *
                   from region_list(:rep_region_id, 1, -1, 0)
                   where id_list = :rep_in_region))) then
      rep_region_id = null;
  end
  suspend;
end
