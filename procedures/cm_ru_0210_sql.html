CREATE PROCEDURE CM_RU_0210_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable c_id_approved_manager d_id;
declare variable un_c_id d_id;
declare variable un_id_approved d_id;
begin

  if (un_id = 1010100001610) then exit; -- По письму Руслана от 1.12.23, прошу Доуловой Г. МНМ ОП 0119 открыть постоянный доступ к утверждению новых клиентов.
  if (un_id = 1010100000960) then exit; --Прошу на постоянной основе открыть Рыскелдиевой Айдане с 13 августа доступ для утверждения займов новым клиентам по своей команде. 

  -- письмо от 24.11.2022 13:17 -- Лаура
  --  Прошу открыть доступ к утверждению займов новым клиентам МНМ офиса Олимпийский и Новослободский на три дня: 24,25 и 26 ноября.
  --  Я взяла отгул в эти дни. Согласовано с Бахтияр Латихановичем.

  if (un_id in (1010100001610, 1010100001590, 1010100000960)) then
  begin
    if (current_date between '24.11.2022' and '26.11.2022') then
      exit;
    if (current_date between '09.12.2022' and '11.12.2022') then
      exit;
    if (current_date between '17.12.2022' and '09.01.2023') then
      exit;
    if (current_date between '10.01.2023' and '11.01.2023') then
      exit;
    if (current_date between '08.03.2023' and '08.03.2023') then
      exit;
    if (current_date between '17.03.2023' and '17.03.2023') then
      exit;
    if (current_date between '31.03.2023' and '01.04.2023') then
      exit;
    if (current_date between '03.04.2023' and '03.04.2023') then
      exit;
    if (current_date between '07.04.2023' and '09.04.2023') then
      exit;
    if (current_date between '14.04.2023' and '14.04.2023') then
      exit;
    if (current_date between '19.04.2023' and '19.04.2023') then
      exit;
    if (current_date between '21.04.2023' and '21.04.2023') then
      exit;
    if (current_date between '24.04.2023' and '15.05.2023') then
      exit;
    if (current_date between '16.05.2023' and '16.05.2023') then
      exit;
  end

  if (un_id in (1010100001610, 1010100000960)) then  -- Доулова Гулзат и Рыскелдиева Айдана Сабырбековна
  begin
    if (current_date < '01.02.2023') then
      exit;
    if (current_date in ('29.05.2023')) then
      exit;
    if (current_date = '28.06.2023') then
      exit;
    if (current_date between '03.07.2023' and '07.07.2023') then
      exit;
    if (current_date in ('21.09.2023','22.09.2023')) then
      exit;
    if (current_date between '03.10.2023' and '10.10.2023') then
      exit;
    if (current_date = '27.10.2023') then
      exit;
    if (current_date between '01.11.2023' and '03.11.2023') then
      exit;
    if (current_date between '17.11.2023' and '20.11.2023') then
      exit;
    if (current_date in ('01.12.2023')) then
      exit;

  end

  if (un_id in (1010100001590)) then
  begin
    if (current_date < '01.03.2023') then
      exit;
    if (extract(weekday from current_date) in (5, 6)) then
      exit; -- письмо от 5.7.23 тема письма: RE: Доступ утверждению новых клиентов_ОП 0124
  end

  if (un_id in (1010100000064, 1010100000469)) then
  begin
    if (current_date >= '17.05.2023') then
      exit; -- доступ открыт по задаче RFCRU-474
  end

  if (un_id = 1010100001610 and
      (current_date = '14.06.2023' or current_date = '21.06.2023' or current_date = '13.11.2023')) then
    exit;

  if ((un_id in (1010100001610, 1010100000960)) and
      (current_date between '24.07.2023 ' and '15.08.2023')) then
    exit;

  if ((un_id in (1010100001610, 1010100000960)) and
      (current_date between '17.08.2023 ' and '19.08.2023')) then
    exit;
  if (un_id in (1010100000960) and current_date = '15.09.2023') then
    exit;

  if (un_id in (1010100000960) and
      (current_date = '03.12.2023' or
       current_date between '04.01.2024' and '09.01.2024' or
       current_date between '02.02.2024' and '19.02.2024' or
       current_date between '19.02.2024' and '10.03.2024' or
       current_date between '11.03.2024' and '15.03.2024' or
       current_date between '09.07.2024' and '22.07.2024'
      )) then  --Рыскелдиева Айдана Сабырбековна
    exit;
  if (un_id = 1010100000960 and current_date <= '31.05.2024' and
  extract(weekday from current_date) = 0) then exit;

  if (un_id = 1010100000571 and current_date in ('24.06.2024')) then
    exit;

--  Прошу открыть доступ на утверждение  займам по новым клиентам ВВиПС Талантбек кызы Айзада, Маткаримова Дамира, Абылкасымова Рахат с
--11.03.2024 по 17.03.2024 в связи с моим отпуском.
-- письмо от 11 марта от
  if (un_id in (1010100001251, 1010100002504, 1010100002865) and
  current_date between '11.03.2024' and '17.03.2024') then exit;

  if (un_id in (1010100000974) and
  current_date between '28.06.2024' and '09.07.2024') then exit;



  if (un_id in (1010100002504, 1010100002865, 1010100003697, 1010100004325)) then
  begin
    if (current_date between '09.09.2024' and '22.09.2024') then exit;
  end

  --RFCRU-2573
  if (un_id in (1010100003069, 1010100000974, 1010100000571, 1010100001260, 1010100000802)) then
  begin
    if (current_date between '22.10.2024' and '22.10.2025') then exit;
  end 


  -- Если офис (Олимпийский или Новослободский) и пользователь НЕ Эркин Мирзаева то выдаем блок
  res_info_text = '';
  select un_from_c_id
  from username
  where un_name = 'EMIRZAEVA'
  into un_c_id;

  for select kc_from_c_id, k_approved_manager
      from kontract, knaktlist, knclient
      where k_id = :k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            current_date between kal_date and kal_date_end and
            kal_new_reg_id in (1011900, 1012500) and
            k_id = kc_from_k_id and
            kc_amount_approved > 0
      into c_id, c_id_approved_manager
  do
  begin

    if (not exists(select *
                   from kontstate, knclient
                   where ks_ol_is_delete = 0 and
                         ks_state = 1 and
                         ks_from_k_id = kc_from_k_id and
                         kc_amount_approved > 0 and
                         kc_from_c_id = :c_id)) then
    begin
      /*
      if (check_kind = '.loan_approve.' and
          not exists(select *
                     from username
                     where un_id = :un_id and
                           un_name = 'EMIRZAEVA')) then
      begin
        res_error_text = 'Не имеете права утверждать новым клиентам данного офиса.';
        res_error_is_critical = 2;
        suspend;
      end
      */
      select un_id
      from username
      where un_from_c_id = :c_id_approved_manager
      into un_id_approved;

      if (un_id_approved not in (1010100001610, 1010100001590, 1010100000960, 1010100000064, 1010100000469, 1010100001251, 1010100002504, 1010100002865,
      1010100000571, 1010100000974, 1010100003697, 1010100004325, 1010100003069, 1010100001260, 1010100000802)) then
      begin
        if (check_kind in ('.check_preview.', '.loan_disb.') and
            un_c_id <> c_id_approved_manager) then
        begin
          res_error_text = 'Клиент новый, должна утвердить менеджер офиса!';
          res_error_is_critical = 2;
          suspend;
        end
      end
    end
  end
end
