CREATE PROCEDURE CM_RU_0074_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable kf_id id_type;
declare variable kf_officer id_type;
declare variable is_svz smallint;
declare variable last_k_id id_type;
declare variable k_kredline_end date;
declare variable oks_balance id_type;
declare variable delinq_day integer;
declare variable ks_date_begin date;
declare variable ks_date_end date;
declare variable action_date date;
declare variable k_amount curr_type;
declare variable max_delinq_day integer;
declare variable ostatok curr_type;
declare variable c_id id_type;
begin
  for
  select ma_date_oper, ma_from_kf_id
  from miniapp, product
  where ma_id = :ma_id
  and ma_date_end = '31.12.9999'
  and ma_from_pr_id = pr_id
  and pr_code containing '.ACT.'
  into kf_disb_date, kf_id
  do
  begin
    is_svz = 0;
    for
    select kal_new_kf_officer from knaktlist
    where kal_from_kf_id = :kf_id
    and knaktlist.is_delete = 0
    order by kal_date desc
    rows 2 -- проверяем последние 2 СВЗ т.е. после уволнения могут дургому передать
    into kf_officer
    do
    begin
      if (exists(select * from hp_get_departed_svz(:kf_officer)
          where result = 1)) then is_svz = 1;
    end 
    if (is_svz = 0) then
    begin
      res_error_text = 'Выбранный продукт предназначен для клиентов уволенного СВЗ. Текущий состав клиентов не является группой уволенного СВЗ.';
      res_error_is_critical = 2;
      suspend;
      exit;
    end 
--  вычисляем последний цикл
    select k_id, k_kredline_end, ks_date_begin, ks_date_end, k_amount_approved
    from kontract, kontstate
    where k_from_kf_id = :kf_id
    and k_disb_date < :kf_disb_date
    and k_id = ks_from_k_id
    and ks_state = 1
    order by ks_date_begin desc
    rows 1
    into last_k_id, k_kredline_end, ks_date_begin, ks_date_end, k_amount;
    if (k_kredline_end - kf_disb_date < 90) then
    begin
      res_error_text = 'Доконца срока осталось менее 3х месяцев, не подлежит к акции';
      res_error_is_critical = 2;
      suspend;
      exit;
    end 
--  проверяем контракт закрылся или нет
--  при следующей выдаче контракт может закрыться
--  если контракт закрылся то датой проверки на просрочку и баланса будем брать дату до закрытия т.е. action_date
    if (current_date between ks_date_begin and ks_date_end) then action_date = current_date;
    else action_date = ks_date_end;
    oks_balance = null;
    delinq_day = null;
    select oks_balance, :action_date - oks_delinq_date
    from oksummary
    where :action_date between oks_date_begin and oks_date_end
    and oks_from_k_id = :last_k_id
    and oks_close_state = 0
    into oks_balance, delinq_day;
    if (delinq_day > 0) then
    begin
      res_error_text = 'Контракт просрочен и не подлежит к ации';
      res_error_is_critical = 2;
      suspend;
      exit;
    end 
--  проверяем максимальный просроченный платеж
    max_delinq_day = null;
    select max(ok_ol_date - ok_sh_date)
    from operkred
    where ok_from_k_id = :last_k_id
    and ok_type in (4,5)
    and ok_os + ok_perc < -100.0
    and ok_ol_is_active = 1
    into max_delinq_day;
    max_delinq_day = coalesce(max_delinq_day, 0);
    if (max_delinq_day > 30) then
    begin
      res_error_text = 'В контракте были просрочки более 30 дней';
      res_error_is_critical = 2;
      suspend;
      exit;
    end 
--  проверяем каждого клиента на баланc не менее 15 тыс.
    for
    select c_name_fio,
           c_pass_no,
           kc_amount_approved,
           c_id
    from knclient, clients
    where kc_from_k_id = :last_k_id
    and kc_amount_approved > 0
    and kc_from_c_id = c_id
    into res_info_text,
         kf_client_pass_no,
         kc_amount,
         c_id
    do
    begin
      ostatok = cast(kc_amount as double precision)/k_amount * oks_balance;
      if (ostatok < 15000.0) then
      begin
        res_error_text = 'У клиента баланс менее 15тыс. руб. не подлежит к акции';
        res_error_is_critical = 2;
        suspend;
        exit;
      end
      if (exists(
          select * from miniappclient
          where mac_from_ma_id = :ma_id
          and mac_from_c_id = :c_id
          and mac_amount > :kc_amount + 10000.0
           )) then
      begin
        res_error_text = 'выдаваемая сумма клиента превышает остатка задолженности более чем на 10тыс. руб.';
        res_error_is_critical = 2;
        suspend;
      end
    end
    suspend;
  end
end
