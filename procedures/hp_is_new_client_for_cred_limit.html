CREATE PROCEDURE HP_IS_NEW_CLIENT_FOR_CRED_LIMIT(
    C_ID D_ID)
RETURNS (
    IS_NEW_CLIENT D_BOOL)
AS
declare variable KS_STATE D_SMALLINT;
declare variable PR_CODE D_STR_100;
begin
  -- Клиент новый в случаях:
  --     1) Клиент не кредитовался в компании
  --     2) У клиента ТОЛЬКО активные кредиты по продуктам "Семена", "Товары в рассрочку"

  is_new_client = 1;

  for select ks_state,
             pr.pr_code
  from knclient kc
  inner join kontract k on k.k_id = kc.kc_from_k_id
  inner join product pr on pr.pr_id = k.k_from_pr_id
  inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
  where kc.kc_from_c_id = :c_id and
        kc.kc_amount_approved > 0 and
        ks.ks_ol_is_delete = 0 and
        current_date between ks.ks_date_begin and ks.ks_date_end and
        ks.ks_state in (1, 2)
  into :ks_state,
       :pr_code
  do
  begin
    if (pr_code containing '.sugarmill.' or
        pr_code containing '.installment_goods.') then
    begin
        if (ks_state = 2) then
        begin
            is_new_client = 0;
            break;
        end
    end
    else
    begin
        is_new_client = 0;
        break;   
    end
  end
  suspend;
end
