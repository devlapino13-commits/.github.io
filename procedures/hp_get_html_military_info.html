CREATE PROCEDURE HP_GET_HTML_MILITARY_INFO(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID)
RETURNS (
    RES_HTML D_STR_8100)
AS
declare variable c_id d_id;
declare variable status d_smallint;
declare variable status_text d_str_255;
declare variable query_date d_date;
declare variable answer d_date;
declare variable c_sex d_smallint;
declare variable age d_integer;
declare variable link d_str_long;
begin
  res_html = '';
  link = '';

  if (obj_kind = 9) then
  begin

    for select kc_from_c_id
    from knclient
    where kc_from_k_id = :obj_id
    and kc_amount_applicate > 0
    into :c_id
    do
    begin
       select status,
              query_date,
              answer
       from hp_military_query_status(:c_id)
       into :status,
            :query_date,
            :answer;
    end

  end

  if (obj_kind = 11) then
  begin
    for select mac_from_c_id
    from miniappclient
    where mac_from_ma_id = :obj_id and
          mac_amount > 0
    into c_id
    do
    begin
      select  status,
              query_date,
              answer
       from hp_military_query_status(:c_id)
       into :status,
            :query_date,
            :answer;
    end
  end

  select c.c_sex,
         f_age(c.c_birth_date, current_date)
  from clients c
  where c.c_id = :c_id
  into c_sex, age;

  if (coalesce(age, '') = '') then
    exception e_message('Не указана дата рождения клиента!');

  if (age not between 18 and 27 -- клиент младше 18 ИЛИ старше 27
      or (c_sex <> 1) ) then    -- клиент не мужчина
  begin
    suspend;
    exit;
  end

  status_text = (case status
                   when 0 then 'Уволен со срочной воинской службы <br>Дата увольнения: %1'
                   when 1 then 'Призван на срочную воинскую службу <br>Дата призыва: %1'
                   else ''
                 end);

  if(current_date <> query_date or
      coalesce(query_date, '') = '' or
      status = -2) then --Если текущий день отличается от даты проверки, у пользователя есть возможность запросить повторную проверку ИЛИ ответ на предыдущий запрос ошибка (-2)
    link = f_uri_sql(iif(coalesce(query_date, '') <> '', 'Проверить повторно', 'Проверить'), 'military_status', 'clients', :c_id, 'none', 'Подтверждаете', '', 0, 0, '', '', 'dialog', '');

  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="30%"><col width="30%">');
  res_html = blob_append(res_html, f_html_table_row('Призывник',
                                                    f_format(status_text, f_datetostr(answer)),
                                                    f_format('Дата проверки: %1<br>' || link, f_datetostr(query_date)),
                                                    -1, -1, -1, -1, -1, 3, ''));
  res_html = blob_append(res_html, '</table>');
  suspend;
end
