CREATE PROCEDURE AAREP_EDOC_CL_STATUS(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    REG_NAME D_STR_100,
    REG_NAME_P D_STR_100,
    EDS_STATUS_DATE DATE,
    STATUS VARCHAR(50),
    KC_FIO D_STR_100,
    CDS_DATETIME D_DATETIME,
    SEM_DATETIME D_DATETIME)
AS
declare variable c_id d_id;
declare variable ma_id d_id;
declare variable ma_date_app date;
declare variable k_id d_id;
declare variable k_fact_disb_date date;
declare variable kc_amount_approved d_currency;
declare variable k_status d_str_20;
declare variable z_disb_date date;
declare variable z_type d_smallint;
declare variable kml_disb_date date;
declare variable road d_str_100;
declare variable rel_c_id d_id;
declare variable r_ma_id d_id;
declare variable r_ma_date_app date;
declare variable edl_create_un_id d_id;
begin
  for select el.edl_object_id, el.edl_create_un_id, cast(eds_status_datetime as date), r.reg_name, cl.c_name_fio,
             rr.reg_name
      from edoc_list el, edoc_sign es, clients cl, region r, region rr, region_list(:rep_region_id, 1, 3, 1)
      where el.edl_object_kind = 0 and
            es.eds_from_edl_id = el.edl_id and
            el.edl_blank_status = 1 and
            cl.c_id = el.edl_object_id and
            rr.reg_unicode = r.parent and
            r.reg_unicode = el.edl_from_reg_id and
            el.edl_from_edc_id = f_edoc_kind('asp.agreement', 'id') and
            el.edl_status = 1 and
            el.edl_blank_date between :rep_date_begin and :rep_date_end and
            id_list = r.reg_unicode and
            coalesce(el.edl_date_end, '31.12.9999') >= :rep_date_end
      into :c_id, edl_create_un_id, eds_status_date, reg_name, c_name_fio, reg_name_p
  do
  begin
    select cdc.cds_datetime
    from clientdocs cdc
    where cdc.cds_from_c_id = :c_id
          and cdc.cds_is_delete = 0
          and cdc.cds_from_pk_id = 313
          and cdc.cds_doc_blob_1 is not null
          and cdc.cds_doc_blob_2 is not null
    order by cdc.cds_datetime desc
    rows 1
    into cds_datetime;

    sem_datetime = null;
    select cdc.cds_datetime
    from clientdocs cdc
    where cdc.cds_from_c_id = :c_id
          and cdc.cds_is_delete = 0
          and cdc.cds_from_pk_id = 336
          and cdc.cds_doc_blob_1 is not null
          and cdc.cds_doc_blob_2 is not null
    order by cdc.cds_datetime desc
    rows 1
    into sem_datetime;

    select un.un_fio
    from username un
    where un.un_id = :edl_create_un_id
    into kc_fio;

    status = '';
    ma_id = null;
    ma_date_app = null;

    k_id = null;
    k_status = null;
    k_fact_disb_date = null;
    kc_amount_approved = null;

    z_disb_date = null;
    z_type = null;

    kml_disb_date = null;

    rel_c_id = null;
    r_ma_id = null;
    r_ma_date_app = null;

    select m.ma_id, cast(ma_date_app as date)
    from miniappclient mcl
    inner join miniapp m on m.ma_id = mcl.mac_from_ma_id
    where m.ma_date_end = '31.12.9999' and
          mcl.mac_from_c_id = :c_id and
          m.ma_date_app >= :eds_status_date and
          m.ma_is_active = 1 and
          mcl.mac_amount > 0
    order by ma_date_app
    rows 1
    into :ma_id, ma_date_app; --берем первую заявку после подписания договора ЭЦП

    select k.k_fact_disb_date, z.z_type
    from zalog z
    inner join kontract k on k.k_id = z.z_from_k_id
    where z_from_c_id = :c_id and
          k.k_fact_disb_date >= :eds_status_date
    order by k_fact_disb_date
    rows 1
    into :z_disb_date, z_type; --ищем среди поручителей

    select k.k_fact_disb_date
    from kontmat kml
    inner join kontract k on k.k_id = kml.kml_object_id and
          kml.kml_object_kind = 9
    inner join agreements on (kml_from_ag_id = ag_id)
    where kml_from_ag_id = :c_id and
          k.k_fact_disb_date >= :eds_status_date
    order by k_fact_disb_date
    rows 1
    into :kml_disb_date; --ищем среди продавцов контрагентов

    select crel_other_c_id
    from hp_client_relation_list(:c_id, 0)
    where crel_from_rel_id in ('1', '2') rows 1
    into :rel_c_id;

    select m.ma_id, cast(ma_date_app as date)
    from miniappclient mcl
    inner join miniapp m on m.ma_id = mcl.mac_from_ma_id
    where m.ma_date_end = '31.12.9999' and
          mcl.mac_from_c_id = :rel_c_id and
          m.ma_date_app >= :eds_status_date and
          m.ma_is_active = 1 and
          mcl.mac_amount > 0
    order by ma_date_app
    rows 1
    into :r_ma_id, r_ma_date_app;

    road = null;

    ma_date_app = coalesce(ma_date_app, '31.12.9999');
    z_disb_date = coalesce(z_disb_date, '31.12.9999');
    kml_disb_date = coalesce(kml_disb_date, '31.12.9999');
    r_ma_date_app = coalesce(r_ma_date_app, '31.12.9999');

    if (r_ma_date_app <= ma_date_app and
        r_ma_date_app <= kml_disb_date and
        r_ma_date_app <= z_disb_date) then
      road = 'rel';
    if (z_disb_date <= ma_date_app and
        z_disb_date <= kml_disb_date and
        z_disb_date <= r_ma_date_app) then
      road = 'zalog';
    if (kml_disb_date <= ma_date_app and
        kml_disb_date <= z_disb_date and
        kml_disb_date <= r_ma_date_app) then
      road = 'kml';
    if (ma_date_app <= z_disb_date and
        ma_date_app <= kml_disb_date and
        ma_date_app <= r_ma_date_app) then
      road = 'miniapp';

    if (road = 'kml') then
    begin
      status = 'Контрагент-продовец';
    end

    if (road = 'rel') then
    begin
      status = 'Лицо дающее согласие';
    end

    if (road = 'zalog') then
    begin
      if (z_type = 0) then
        status = 'Поручитель';
      if (z_type = 1) then
        status = 'Залогодатель';
      if (z_type <> 0 and
          z_type <> 1) then
        status = 'error';
    end

    if (road = 'miniapp') then
    begin
      status = 'не получил кредит/финансирование';
      if (ma_id is not null) then
      begin
        select k.k_id, k.k_status, k.k_fact_disb_date, knc.kc_amount_approved
        from kontract k
        inner join knclient knc on knc.kc_from_k_id = k.k_id
        where k.k_from_ma_app_id = :ma_id
        order by k.k_status desc
        rows 1
        into :k_id, k_status, k_fact_disb_date, kc_amount_approved;

        if (k_status > 0) then
          status = 'Клиент';
      end
    end
    suspend;
  end
end
