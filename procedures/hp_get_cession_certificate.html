CREATE PROCEDURE HP_GET_CESSION_CERTIFICATE(
    K_ID D_ID)
RETURNS (
    C_NAME_FIO D_STR_255,
    C_PASS_NO D_STR_64,
    C_PASS_DATE D_DATE,
    C_PASS_WHO D_STR_255,
    C_PASS_WHO_CODE D_STR_20,
    K_KONTRACT_NO_OUT D_STR_20,
    K_KONTRACT_DATE D_DATE,
    ZADOLG_STATUS D_STR_20,
    ZADOLG_SUM NUMERIC(15,2),
    DT_CESSII D_DATE,
    CESSIONER_NAME D_STR_255,
    CESSIONER_INN D_STR_20,
    CESSIONER_ADDRESS D_STR_255,
    ERROR_MSG D_STR_255)
AS
declare variable C_ID bigint;
declare variable KC_ID bigint;
declare variable SCP_ID integer;
declare variable SAG_ID integer;
declare variable OKS_CURR_SH_OS numeric(15,2);
declare variable OKS_CURR_SH_PERC numeric(15,2);
begin
  -- RFCRU-5717
  error_msg = null;
  c_name_fio = null;
  c_pass_no = null;
  c_pass_date = null;
  c_pass_who = null;
  c_pass_who_code = null;
  k_kontract_no_out = null;
  k_kontract_date = null;
  zadolg_status = null;
  zadolg_sum = null;
  dt_cessii = null;
  cessioner_name = null;
  cessioner_inn = null;
  cessioner_address = null;

  -- шаг 1: найти договор
  select k.k_kontract_no, k.k_kontract_date
  from kontract k
  where k.k_id = :k_id
  into k_kontract_no_out, k_kontract_date;

  if (k_kontract_no_out is null) then
  begin
    error_msg = 'договор не найден';
    suspend;
    exit;
  end

  -- шаг 2: проверка, что договор в списке проданных + дата цессии + ссылка на цессионария
  select scp.scp_id, scp.scp_date, scp.scp_from_sag_id
  from sud_cession_packet scp, sud_cession_loans scl
  where scl.scl_from_scp_id = scp.scp_id
    and scl.scl_from_k_id = :k_id
    and scp.scp_status > 0
  into scp_id, dt_cessii, sag_id;

  if (scp_id is null) then
  begin
    error_msg = 'договор не находится в списке проданных';
    suspend;
    exit;
  end

  -- шаг 3: данные нового кредитора (цессионария)
  select sag.sag_fullname, sag.sag_contractor_inn, sag.sag_address
  from sud_agents sag
  where sag.sag_id = :sag_id
  into cessioner_name, cessioner_inn, cessioner_address;

  -- шаг 4: связка договор -> клиент
  select c.c_name_fio, c.c_pass_no, c.c_pass_who, c.c_pass_date, c.c_pass_who_code, kc.kc_id, kc.kc_from_c_id
  from knclient kc
  inner join clients c on c.c_id = kc.kc_from_c_id
  where kc.kc_from_k_id = :k_id
    and kc.kc_amount_approved > 0
  into c_name_fio, c_pass_no, c_pass_who, c_pass_date, c_pass_who_code, kc_id, c_id;

  -- шаг 5: задолженность через oksummary
  select oks.oks_curr_sh_os, oks.oks_curr_sh_perc
  from oksummary oks
  where oks.oks_from_k_id = :k_id
  order by oks.oks_date_begin desc
  rows 1
  into oks_curr_sh_os, oks_curr_sh_perc;

  zadolg_sum = coalesce(oks_curr_sh_os, 0) + coalesce(oks_curr_sh_perc, 0);
  zadolg_status = iif(zadolg_sum > 0, 'есть', 'отсутствует');

  suspend;
end
