CREATE PROCEDURE CM_RU_0129_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable KF_CLIENT_PASS_NO varchar(30);
declare variable C_ID D_ID;
declare variable RES_AMOUNT D_CURRENCY;
declare variable PR_CODE D_STR_100;
declare variable K_ADD_MONTH smallint;
begin
  if (k_id = 101010006745300601) then exit;
  select k.k_clients_cashier_id, pr.pr_code, k_add_month
  from kontract k
  inner join product pr on pr.pr_id = k.k_from_pr_id
  where k_id = :k_id
  into :c_id, :pr_code, :k_add_month;
  if (c_id is not null) then
    select sum(cast(res_amount as d_currency))
    from hp_kontrefund_k_list(:k_id)
    into :res_amount;
  res_amount = coalesce(res_amount, 0);
  for select kf_client_fio, kf_client_pass_no, kf_disb_date, kc_amount, kf_error_text, kf_error_is_critical
      from hp_check_loan_docx(:k_id)
      into :res_info_text, :kf_client_pass_no, :kf_disb_date, :kc_amount, :res_error_text, :res_error_is_critical
  do
  begin
    -- jira 1971
    if (kf_disb_date between '20.05.2020' and '15.06.2020') then
    begin
      if (res_error_text containing ' не приложен документ о подаче на патент и т.п.' and
          pr_code containing '.shans.') then
        res_error_is_critical = 0;
      if (res_error_text = 'Продукт "Экспресс", не приложен документ продлевающий пребывание в РФ') then
        res_error_is_critical = 0;
    end
    if (res_amount = kc_amount and
        k_add_month > 0) then
    begin
      if (res_error_text containing ' не приложен документ о подаче на патент и т.п.') then
        res_error_is_critical = 0;
      if (res_error_text containing 'Не приложен документ дающий право на работу в РФ') then
        res_error_is_critical = 0;
    end
    --Указом президента отменены ограничения по срокам подачи на патент. В этой связи прошу сделать не критичным блок по запросу документа о подаче на патент, даже если с момента въезда в РФ прошло более 20 дней. Zara
    if (res_error_text containing ' не приложен документ о подаче на патент и т.п.') then
      res_error_is_critical = 1;

    if (res_error_text containing 'Не приложен документ дающий право на работу в РФ') then
        res_error_is_critical = 0;

    suspend;
  end
end
