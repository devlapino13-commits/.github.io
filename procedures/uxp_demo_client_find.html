CREATE PROCEDURE UXP$DEMO_CLIENT_FIND(
    FIND_TEXT D_STR_100,
    CLIENT_TYPE D_BOOL,
    ONLY_STAFF D_BOOL)
RETURNS (
    CLI_ID D_INTEGER,
    CLI_TYPE D_SMALLINT,
    CLI_COUNTRY D_STR_CODE,
    CLI_NAME D_STR_100,
    CLI_DATE D_DATE,
    CLI_INN D_STR_CODE,
    CLI_ADDRESS D_STR_255,
    CLI_IS_STAFF D_BOOL)
AS
begin
  if (client_type = -1) then -- все виды контрагентов
    client_type = null;

  for select cli_id,
             cli_type,
             cli_country,
             cli_name,
             cli_date,
             cli_inn,
             cli_address,
             cli_is_staff
      from ux$demo_client
      where cli_type = coalesce(:client_type, cli_type) and
            cli_is_staff >= :only_staff and
            (cli_name containing :find_text or
            cli_inn containing :find_text or
            cli_address containing :find_text)
      into :cli_id,
           :cli_type,
           :cli_country,
           :cli_name,
           :cli_date,
           :cli_inn,
           :cli_address,
           :cli_is_staff
  do
  begin
    suspend;
  end
end
