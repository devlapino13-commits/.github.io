CREATE PROCEDURE AAREP_KONTRACT_ACTIVE_TEMP(
    REG_ID INTEGER,
    REP_DATE DATE,
    REPORT_CURRENCY VARCHAR(3),
    REPORT_REVERSE INTEGER)
RETURNS (
    AC_K_ID DECIMAL(18,0),
    AC_K_BALANCE DECIMAL(18,4),
    AC_LOAN_TYPE SMALLINT,
    AC_KF_ID DECIMAL(18,0),
    AC_REGION_ID INTEGER,
    AC_CLIENT_KOLICH INTEGER,
    AC_DISB DECIMAL(18,4))
AS
declare variable curr_prior varchar(3);
declare variable curr_code varchar(3);
declare variable curr_rate decimal(18,4);
declare variable k_curr_rate decimal(18,4);
declare variable cm_def_curr varchar(3);
declare variable tmp_sum decimal(18,4);
declare variable ok_type integer;
declare variable pr_set_rate_in_kont integer;
declare variable pr_id integer;
declare variable k_disb_curr_rate decimal(18,4);
declare variable pr_id_prior integer;
declare variable need_curr_rate integer;
begin
  curr_prior = '000';
  select cm_default_currency
  from company
  into :cm_def_curr;

  if (report_currency = '000') then
    need_curr_rate = 1;
  else
    need_curr_rate = 0;
  pr_id_prior = 0;

  for select kk_currency, kk_amount_approved, kk_id, kk_from_pr_id, kk_from_kf_id, kk_region_id, kk_kolich_client,
             kk_curr_rate, kk_disb_curr_rate, kk_balance
      from aarep_active_kont_id_in_period(:reg_id, null, :rep_date, 0, 0, :need_curr_rate, 0, 0)
      into :curr_code, :ac_disb, :ac_k_id, :pr_id, :ac_kf_id, :ac_region_id, :ac_client_kolich, :k_curr_rate,
           :k_disb_curr_rate, ac_k_balance
  do
  begin
    if (ac_k_balance > 0) then
    begin
      if (pr_id <> pr_id_prior) then
      begin
        select pr_loan_type
        from product
        where pr_id = :pr_id
        into :ac_loan_type;
        pr_id_prior = :pr_id;
      end

      if ((report_currency = '000') and
          (cm_def_curr <> curr_code)) then
      begin
        ac_k_balance = cast(:ac_k_balance * :k_curr_rate as decimal(15,2));
        ac_disb = cast(:ac_disb * :k_disb_curr_rate as decimal(15,2));
      end
      suspend;
    end
  end

end
