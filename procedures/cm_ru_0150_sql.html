CREATE PROCEDURE CM_RU_0150_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable err_txt d_str_long;
declare variable adr_type d_str_100;
declare variable adl_post_index d_str_100;
declare variable c_pass_who str_255;
declare variable error_txt d_str_long;
declare variable c_inn varchar(20);
declare variable result integer;
declare variable c_pass_who_code str_255;
declare variable c_birth_city str_255;
declare variable k_disb_date date;
declare variable doc_code str_100;
declare variable doc_no str_100;
declare variable adl_address_str d_str_255;
declare variable adc_address_type d_smallint;
begin
  res_error_is_critical = 2;
  for select kncl.kc_from_c_id, cl.c_name_fio, coalesce(cl.c_pass_who, ''), iif(cl.c_citizen = 13, cl.c_inn, ''),
             coalesce(cl.c_pass_who_code, ''), coalesce(cl.c_birth_city, ''), kont.k_disb_date
      from knclient kncl
      inner join clients cl on kncl.kc_from_c_id = cl.c_id
      inner join kontract kont on kont.k_id = kncl.kc_from_k_id
      where kncl.kc_from_k_id = :k_id and
            kncl.kc_amount_approved > 0
      into :c_id, :res_info_text, :c_pass_who, :c_inn, :c_pass_who_code, :c_birth_city, :k_disb_date
  do
  begin
    if (c_pass_who_code <> '') then
    begin
      if (c_pass_who_code not similar to('[-]?[a-zа-яё0-9-.\ ]+')) then
      begin
        res_error_is_critical = 2;
        res_error_text = 'Код подразделения для документа удостоверяющего личность указан не корретно. Сейчас:' || c_pass_who_code;
        suspend;
      end
    end
    if (c_birth_city = '') then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Мето рождения клиента не указано';
      suspend;
    end
    ----------- >
    error_txt = null;
    select error_txt
    from hp_check_characters_str(:c_pass_who, '[[:ALNUM:]а-яё@-+_.\/#,№!:;"''={}() ]+')
    into :error_txt;
    if (error_txt <> '') then
    begin
      res_error_text = left('Место выдачи документа указано не корректно. ' || error_txt, 255);
      suspend;
    end
    error_txt = null;
    select error_txt
    from hp_check_characters_str(:c_birth_city, '[[:ALNUM:]а-яё@-+_.\/#,№!:;"''={}() ]+')
    into :error_txt;
    if (error_txt <> '') then
    begin
      res_error_text = 'Место рождения указано не корректно. ' || error_txt;
      suspend;
    end
    for select adl.adl_address_str, adc.adc_address_type
        from addrclient adc
        inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
        where adc.adc_address_type in (1, 2) and
              adc.adc_from_c_id = :c_id
        into :adl_address_str, adc_address_type
    do
    begin
      error_txt = null;
      select error_txt
      from hp_check_characters_str(:adl_address_str, '[[:ALNUM:]а-яё@-+_.\/#,№!:;"''={}() ]+')
      into :error_txt;
      if (error_txt <> '') then
      begin
        if (adc_address_type = 1) then
          res_error_text = 'Фактический адрес указан не корректно. ' || error_txt;
        else
          res_error_text = 'Адрес регистрации указан не корректно. ' || error_txt;
        suspend;
      end
    end
    ------------ <
    if (c_inn <> '') then
    begin
      if (character_length(c_inn) <> 12) then
      begin
        res_error_text = 'ИНН (' || c_inn || ') указан не корректно. Допустимая длина 12 символов, а сейчас ' || character_length(c_inn) || ' симв.';
        suspend;
      end
    end
    result = 1;
    if (substring(trim(res_info_text) from 1 for 1) similar to '[а-яё]') then --Получаем 1й символ и проверяем на кириллицу
      :result = iif(res_info_text similar to '%[ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnoopqrstuvwxyz]+%', 1, 0); --кириллица
    else
      :result = iif(res_info_text similar to '%[а-яё]+%', 1, 0); --латынь
    if (result = 1) then
    begin
      res_error_text = 'ФИО указано не корреткно. В ФИО пристуствует как кириллица так и латиница';
      suspend;
    end
    for select trim(iif(adcl.adc_address_type = 1, 'Адрес регистрации', 'Адрес проживания')),
               coalesce(adl.adl_post_index, '')
        from addrclient adcl
        inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
        where adcl.adc_from_c_id = :c_id and
              adcl.adc_address_type in (1, 2)
        into :adr_type, :adl_post_index
    do
    begin
      err_txt = '';
      if (adl_post_index = '') then
        err_txt = 'почтовый индекс не указан';
      else
      if (character_length(adl_post_index) <> 6) then
        err_txt = 'длина почтового индекса должна быть 6 символов, сейчас: ' || character_length(adl_post_index) || ' символов. Почт. индекс: ' || adl_post_index;
      if (err_txt <> '') then
      begin
        res_error_text = adr_type || ': ' || err_txt;
        suspend;
      end
    end
    for select pdt_code, cd.cds_doc_no
        from clientdocs cd
        inner join passkind on cds_from_pk_id = pk_id
        inner join passdoctype on pdt_id = pk_from_pdt_id
        where cds_from_c_id = :c_id and
              cds_doc_blob_1 > 0 and
              :k_disb_date between cds_doc_date_begin and coalesce(cds_doc_date_end, '31.12.9999') and
              cd.cds_verify_status in (0, 1) and
              cd.cds_is_delete = 0
        into :doc_code, doc_no
    do
    begin
      if (doc_code = 'snils') then
      begin
        if (char_length(doc_no) <> 11) then
        begin
          res_error_is_critical = 2;
          res_error_text = 'Ошибка в СНИЛС, длина мене 11 цифр. № документа: ' || doc_no;
          suspend;
        end
        else
        begin
          if (not(doc_no similar to '[-]?[0-9]+')) then
          begin
            res_error_is_critical = 2;
            res_error_text = 'Ошибка в СНИЛС, допустимы только цифры, без доп. символов: ' || doc_no;
            suspend;
          end
        end
      end
    end
  end
end
