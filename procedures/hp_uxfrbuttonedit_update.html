CREATE PROCEDURE HP_UXFRBUTTONEDIT_UPDATE(
    UFP_CODE D_STR_100,
    UA_USR_TEXT_FIELD D_STR_CODE,
    USR_CODE D_STR_100,
    IS_DIALOG D_BOOL,
    UA_NAME D_STR_100)
AS
declare variable ube_frame_code d_str_code;
declare variable ube_frame_id_field d_str_code;
declare variable source_is_table d_bool;
begin
  if (trim(coalesce(ufp_code, '')) = '') then
    exception e_default f_format('ufp_code is empty');

  select ufp_ufr_code, ufp_origin_name
  from uxfrparam
  where ufp_code = :ufp_code
  into ube_frame_code, ube_frame_id_field;

  if (row_count = 0) then
    exception e_message f_format('Поле "%1" не найдено, возможно уже удалено.', ufp_code);

  if (not exists(select * from uxsearch
  where usr_code = :usr_code)) then
    exception e_default f_format('Поисковик не найден: %1', usr_code);

  select ufd_is_table
  from uxfrdata
  where ufd_ufr_code = :ube_frame_code and
        ufd_kind = 0
  into source_is_table;

  if (source_is_table = 0 and coalesce(ua_usr_text_field, '') = '') then
    exception e_message f_format('Источник - хранимая процедура. Заполнение текстового результата обязательно, поисковик: usr_code', usr_code);

  update or insert into uxaction (ua_ux_code, ua_uxa_code, ua_name, ua_usr_id_field, ua_usr_text_field)
  values (:ube_frame_code, :usr_code, :ua_name, :ube_frame_id_field, :ua_usr_text_field)
  matching (ua_ux_code, ua_uxa_code, ua_usr_id_field);

  if (is_dialog = 0) then
  begin
    update uxfrparam
    set ufp_editor = 'buttonedit'
    where ufp_code = :ufp_code and
          ufp_editor <> 'buttonedit';
  end else
  begin
    update uxfrparam
    set ufp_editor = 'dialogedit'
    where ufp_code = :ufp_code and
          ufp_editor <> 'dialogedit';
  end
end
