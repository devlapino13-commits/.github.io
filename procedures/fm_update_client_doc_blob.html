CREATE PROCEDURE FM_UPDATE_CLIENT_DOC_BLOB(
    CDS_ID INTEGER,
    BLS_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    BLS_FILENAME VARCHAR(100),
    BLS_BLOB_SIZE FLOAT,
    PAGE_NO SMALLINT,
    SRC_KIND SMALLINT,
    UN_ID DECIMAL(18,0))
AS
declare variable bls_id decimal(18,0);
declare variable c_id decimal(18,0);
declare variable pk_id integer;
declare variable need_photo2 integer;
declare variable bls_caption varchar(120);
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  uxf$set_user(un_id);
  bls_id = 0;
  select pk_name
  from clientdocs, passkind
  where cds_id = :cds_id and
        cds_from_pk_id = pk_id
  into bls_caption;
  bls_caption = 'Стр.' || :page_no || coalesce(': ' || bls_caption, '');

  -- Вставка блоба
  execute procedure hp_insert_into_blobstore (:bls_filename, :bls_caption, :bls_blob, :un_id, :src_kind, 1) -- FsMobile
  returning_values :bls_id;

  c_id = null;
  select cds_from_c_id, cds_from_pk_id
  from clientdocs
  where cds_is_main_doc = 1 and
        cds_id = :cds_id and
        cds_is_delete = 0 rows 1
  into :c_id, :pk_id;

  if (c_id is null) then
  begin
    -- к документу прикрепляем блобы
    if (page_no = 1) then
    begin
      update clientdocs
      set cds_doc_blob_1 = :bls_id,
          cds_verify_status = -3,
          cds_from_un_id = :un_id
      where cds_id = :cds_id;
    end
    else
    begin
      update clientdocs
      set cds_doc_blob_2 = :bls_id,
          cds_verify_status = -3,
          cds_from_un_id = :un_id
      where cds_id = :cds_id;
    end
  end
  else
  begin
    if (page_no = 1) then
    begin
      update clients
      set c_blob_pass1_id = :bls_id,
          c_from_un_id = :un_id
      where c_id = :c_id;
    end
    else
    begin
      update clients
      set c_blob_pass2_id = :bls_id,
          c_from_un_id = :un_id
      where c_id = :c_id;
    end

    select drm_has_page2
    from hp_get_drm_has_pages(:pk_id)
    into :need_photo2;

    if (exists(select *
               from clients
               where c_id = :c_id and
                     coalesce(c_name_fio, '') <> '' and
                     c_birth_date is not null and
                     coalesce(c_pass_no, '') <> '' and
                     coalesce(c_pass_date, '') <> '' and
                     coalesce(c_blob_photo_id, 0) > 0 and
                     coalesce(c_blob_pass1_id, 0) > 0 and
                     (coalesce(c_blob_pass2_id, 0) > 0 or
                     :need_photo2 = 0) and
                     c_verify_status <> 1)) then
    begin
      if (db_id in (39801, 41701)) then
      begin --Доп прооверки для КГ
        if (exists(select *
                   from hp_status_param_verif(:c_id)
                   where result = 0)) then
        begin
          update clients
          set c_verify_status = 1,
              c_from_un_id = :un_id
          where c_id = :c_id;
        end
      end
      else
      begin
        update clients
        set c_verify_status = 1,
            c_from_un_id = :un_id
        where c_id = :c_id;
      end
    end
  end
end
