CREATE PROCEDURE AAREP_KS_ACTIVE_STAGE(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(50),
    AA_CURR_BRANCH_NAME VARCHAR(50),
    AA_CURR_LEVEL_NAME VARCHAR(50),
    AA_CURR_KC_FIO VARCHAR(100),
    AA_AMOUNT_DISB DECIMAL(18,4),
    AA_AMOUNT_ACTIVE_BEGIN DECIMAL(18,4),
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_KOLICH_CL_ACTIVE_BEGIN INTEGER,
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    AA_KOLICH_CL_DISB INTEGER,
    AA_DELIQ_BALANCE DECIMAL(18,4),
    AA_DELIQ_BALANCE7 DECIMAL(18,4),
    AA_DELIQ_BALANCE30 DECIMAL(18,4),
    AA_DELIQ_CL_KOLICH INTEGER,
    AA_DELIQ_CL_KOLICH7 INTEGER,
    AA_DELIQ_CL_KOLICH30 INTEGER,
    AA_REPAY_WRITEOFF_OS DECIMAL(18,4),
    AA_REPAY_WRITEOFF_PERC DECIMAL(18,4),
    AA_REPAY_WRITEOFF_FINE DECIMAL(18,4),
    AA_REST_NORM_BAL DECIMAL(18,4),
    AA_REST_DELIQ_BAL DECIMAL(18,4),
    AA_REST_NORM_KOL INTEGER,
    AA_REST_DELIQ_KOL INTEGER,
    AA_KC_STAGE VARCHAR(100),
    KC_STAGE INTEGER)
AS
declare variable aa_curr_region_id integer;
declare variable aa_curr_kc_id decimal(18,0);
declare variable ps_date date;
declare variable kc_id_prior decimal(18,0);
begin
  kc_id_prior = 0;
  for select aa_curr_region_id, aa_curr_kc_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name,
             aa_curr_kc_fio, sum(aa_amount_disb), sum(aa_amount_active_begin), sum(aa_amount_active_end),
             sum(aa_kolich_cl_active_begin), sum(aa_kolich_cl_active_end), sum(aa_kolich_cl_disb),
             sum(aa_deliq_balance), sum(aa_deliq_balance7), sum(aa_deliq_balance30), sum(aa_deliq_cl_kolich),
             sum(aa_deliq_cl_kolich7), sum(aa_deliq_cl_kolich30), sum(aa_repay_writeoff_os),
             sum(aa_repay_writeoff_perc), sum(aa_repay_writeoff_fine), sum(aa_rest_norm_bal), sum(aa_rest_deliq_bal),
             sum(aa_rest_norm_kol), sum(aa_rest_deliq_kol)

      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 0, 1, 1, 0, 0, 0)
      group by aa_curr_region_id, aa_curr_kc_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_kc_fio
      into :aa_curr_region_id, :aa_curr_kc_id, :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_level_name,
           :aa_curr_kc_fio, :aa_amount_disb, :aa_amount_active_begin, :aa_amount_active_end, :aa_kolich_cl_active_begin,
           :aa_kolich_cl_active_end, :aa_kolich_cl_disb, :aa_deliq_balance, :aa_deliq_balance7, :aa_deliq_balance30,
           :aa_deliq_cl_kolich, :aa_deliq_cl_kolich7, :aa_deliq_cl_kolich30, :aa_repay_writeoff_os,
           :aa_repay_writeoff_perc, :aa_repay_writeoff_fine, :aa_rest_norm_bal, :aa_rest_deliq_bal, :aa_rest_norm_kol, :aa_rest_deliq_kol

  do
  begin

    if (kc_id_prior <> aa_curr_kc_id) then
    begin
      ps_date = '01.01.1900';
      select ps_date_begin
      from hr_pstatus, hr_orders, joblist, jobkind
      where ps_from_c_id = :aa_curr_kc_id and
            ps_from_ho_id = ho_id and
            (ps_date_begin is not null) and
            jl_id = ps_from_jl_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            jl_kind = jk_id
            /*and ps_from_reg_id=:AA_CURR_REGION_ID*/
      order by ps_date_begin
      rows 1
      into :ps_date;
      if ((ps_date is null) or (ps_date = '01.01.1900')) then
        aa_kc_stage = 'XX Стаж не определен';
      else
      begin
        kc_stage = datediff(month, :ps_date, :rep_date_end);/*между двумя датами определяет сколько месяцев*/

        /*   if (kc_stage<=3)
        then AA_KC_STAGE='00 Стаж КС до 3 месяцев';
        else
        begin
          if (kc_stage<12) then
          begin
            if (kc_stage<10)
            then AA_KC_STAGE='0'||:kc_stage||' Стаж КС '||:kc_stage||' мес.';
            else  AA_KC_STAGE=:kc_stage||' Стаж КС '||:kc_stage||' мес.';
          end else AA_KC_STAGE='12 Стаж КС более года';
        end   */

        aa_kc_stage = case
                        when kc_stage <= 3 then '1. Стаж КС до 3 месяцев'
                        when kc_stage between 4 and 6 then '2. Стаж КС 4-6 месяцев'
                        when kc_stage between 7 and 12 then '3. Стаж КС 7-12 месяцев'
                        when kc_stage between 13 and 18 then '4. Стаж КС 13-18 месяцев'
                        when kc_stage between 19 and 24 then '5. Стаж КС 19-24 месяцев'
                        when kc_stage between 25 and 36 then '6. Стаж КС 25-36 месяцев'
                        when kc_stage > 36 then '7. Стаж КС более 36 месяцев'
                      end;

      end
      kc_id_prior = aa_curr_kc_id;
    end

    suspend;
  end

end
