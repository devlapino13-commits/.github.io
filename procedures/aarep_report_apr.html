CREATE PROCEDURE AAREP_REPORT_APR(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    OFFICER_NAME STR_100,
    BALANCE_BEGIN CURR_TYPE,
    BALANCE_END CURR_TYPE,
    CHANGES CURR_TYPE,
    DEL_BAL_BEGIN CURR_TYPE,
    DEL_BAL_END CURR_TYPE,
    CHANGES_BALANCE CURR_TYPE,
    DEL_BAL_BEGIN_7 CURR_TYPE,
    DEL_BAL_END_7 CURR_TYPE,
    CHANGES_BALANCE_7 CURR_TYPE,
    DEL_BAL_BEGIN_30 CURR_TYPE,
    DEL_BAL_END_30 CURR_TYPE,
    CHANGES_BALANCE_30 CURR_TYPE,
    KOL_DISB INTEGER,
    SVZ_KOL_ACT INTEGER,
    WORK_DAYS_QTY INTEGER,
    KOL_DISB_AVG DECIMAL(4,2),
    FIRST_DATE_MONTH DATE,
    DISB_SUM DECIMAL(18,4),
    PARENT_NAME STR_255,
    OKS_KC_ID ID_TYPE,
    REG_UNICODE D_INTEGER)
AS
declare variable oks_region_id integer;
declare variable reg_id_prior integer;
declare variable kc_id_prior id_type;
declare variable dd integer;
declare variable sawt_name str_100;
declare variable ps_from_c_id id_type;
declare variable hrv_days integer;
declare variable jl_name varchar(100);
declare variable last_date_prv_month date;
begin
  first_date_month = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date);
  last_date_prv_month = first_date_month - 1;

  --считаем количество рабочих дней
  select count(scd_work_hours) work_days_qty
  from sa_calendays sc
  where sc.scd_date between :first_date_month and :rep_date_end and
        scd_work_hours > 0
  into :work_days_qty;

  reg_id_prior = -1;
  kc_id_prior = -1;

  for select reg_unicode, reg_code || ' ' || reg_name, parent_name
      from region, region_list(:rep_region_id, 1, 3, 1)
      where reg_unicode = id_list
      order by reg_name
      into reg_unicode, reg_name, parent_name
  do
  begin

    for with tabl1
        --Баланс КП(Баланс просроч. на начало и на конец)
        as (select oks_region_id, oks_kc_id, oks_balance as balance_begin,
                   cast(:last_date_prv_month as date) - oks_delinq_date as del_day_begin, 0 as balance_end, -1 as del_day_end

            from oksummary
            where :last_date_prv_month between oks_date_begin and oks_date_end and
                  oks_balance > 0 and
                  oks_close_state = 0 and
                  oks_region_id = :reg_unicode
            union all
            select oks_region_id, oks_kc_id, 0, -1, oks_balance, cast(:rep_date_end as date) - oks_delinq_date
            from oksummary
            where :rep_date_end between oks_date_begin and oks_date_end and
                  oks_balance > 0 and
                  oks_close_state = 0 and
                  oks_region_id = :reg_unicode)
        select oks_region_id, oks_kc_id, sum(balance_begin) as balance_begin, sum(balance_end) as balance_end,
               sum(iif(del_day_begin >= 0, balance_begin, 0)) as del_bal_begin,
               sum(iif(del_day_end >= 0, balance_end, 0)) as del_bal_end,
               sum(iif(del_day_begin > 7, balance_begin, 0)) as del_bal_begin,
               sum(iif(del_day_end > 7, balance_end, 0)) as del_bal_end,
               sum(iif(del_day_begin > 30, balance_begin, 0)) as del_bal_begin_30,
               sum(iif(del_day_end > 30, balance_end, 0)) as del_bal_end_30
        from tabl1
        group by 1, 2
        into oks_region_id, oks_kc_id, balance_begin, balance_end, del_bal_begin, del_bal_end,
             del_bal_begin_7, del_bal_end_7, del_bal_begin_30, del_bal_end_30

    do
    begin

      if (kc_id_prior <> oks_kc_id) then
      begin
        kc_id_prior = oks_kc_id;
        --ФИО сотрудника
        officer_name = null;
        select count(c_id), c_name_fio
        from clients
        where c_id = :oks_kc_id
        group by c_name_fio
        into svz_kol_act, officer_name;

        --В поле 12 указывается количество ВИПС в филиале, которые реально работали (декретников и больничных более 14 календарных дней не учитываем, отпускных учитываем)
        --если сотрудник не брал больничные итд  и реально работал то у него в отчете должно выходить 1
        --если есть декретники и больничные более 14 календарных дней то в отчете они должны выходить 0
        --если же сотрудник в больничном допусти 3 дня то должно выходить 1
        jl_name = '';
        select ps_from_c_id, jl_name
        from jobkind, joblist, hr_pstatus
        where jk_system_code = 'JobSystem.LO_Standart' and --SVZ
              jk_id = jl_kind and
              jl_id = ps_from_jl_id and
              ps_status = 1 and
              ps_is_working = 1 and
              :rep_date_end between ps_date_begin and ps_date_end and
              ps_from_c_id = :oks_kc_id
        into ps_from_c_id, jl_name;

        if (jl_name not in ('Kredit mütəxəssisi', 'Baş kredit mütəxəssisi', 'Aparıcı kredit mütəxəssisi', 'Kiçik kredit mütəxəssisi')) then
          svz_kol_act = 0;

        sawt_name = null;
        select sawt_name, hrv_days
        from hr_voctravel, hr_orders, sa_worktype
        where hrv_from_ho_id = ho_id and
              ho_order_type = sawt_id and
              hrv_from_c_id = :oks_kc_id and
              :rep_date_end between hrv_date_begin and hrv_date_end_fact
        order by hrv_date_begin desc
        rows 1
        into sawt_name, hrv_days;

        if (hrv_days > 14 and
            sawt_name in ('Декретный отпуск', 'Отпуск по уходу за ребенком', 'Оплачиваемый больнич.отпуск', 'Неоплачиваемый больнич.отпуск')) then
          svz_kol_act = 0;

             --Выданные кредиты
       --Сумма выданных кредитов 
        kol_disb = 0;
        disb_sum =0;
        select count(*), sum(k_amount_approved)
        from kontstate, kontract
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_date_begin between :first_date_month and :rep_date_end and
              ks_region_id = :reg_unicode and
              ks_kc_id = :oks_kc_id and
              ks_from_k_id = k_id
        into kol_disb, disb_sum;
        if (disb_sum is null) then
          disb_sum = 0;

     /*   --Выданные кредиты
        kol_disb = 0;
        select count(*)
        from kontstate
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_kc_id = :oks_kc_id and
              ks_date_begin between :first_date_month and :rep_date_end
        into kol_disb;   */

        -- Средняя выдача на 1 КС считается как = количество выдач/количество КС/количество рабочих дней, рабочих дней в январе в АЗ включая 28 января – 17.
        if (svz_kol_act > 0 and
            work_days_qty <> 0) then
          kol_disb_avg = round(cast(kol_disb as d_currency) / cast(svz_kol_act as d_currency) / cast(work_days_qty as d_currency), 2);
        else
          kol_disb_avg = 0.0;
      end
      --Изменение баланса
      changes = -(balance_begin - balance_end);
      changes_balance = -(del_bal_begin - del_bal_end);
      changes_balance_7 = -(del_bal_begin_7 - del_bal_end_7);
      changes_balance_30 = -(del_bal_begin_30 - del_bal_end_30);

      suspend;
    end
  end
end
