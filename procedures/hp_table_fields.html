CREATE PROCEDURE HP_TABLE_FIELDS(
    TABLE_NAME D_STR_CODE)
RETURNS (
    RES_POSITION D_INTEGER,
    RES_TABLE_NAME D_STR_CODE,
    RES_FIELD_NAME D_STR_CODE,
    RES_DOMAIN D_STR_CODE,
    RES_FIELD_SIZE D_INTEGER,
    RES_FIELD_TYPE D_STR_CODE,
    RES_COMMENT D_STR_4096,
    RES_IS_KEY_FIELD D_BOOL,
    RES_IS_FK_FIELD D_BOOL,
    RES_NOT_NULL D_BOOL,
    RES_DEFAULT D_STR_255,
    RES_FK_TABLE D_STR_CODE,
    RES_FK_FIELD D_STR_CODE,
    RES_IS_VIRTUAL D_BOOL,
    RES_CAPTION D_STR_CODE,
    UF_ID D_INTEGER)
AS
begin
  table_name = upper(trim(table_name));
  res_table_name = lower(table_name);
  res_is_virtual = 0;
  res_position = 0;

  uf_id = null;
  res_domain = '';

  for select lower(trim(r.rdb$field_name))
      from rdb$fields f, rdb$relation_fields r
      where f.rdb$field_name = r.rdb$field_source and
            r.rdb$system_flag = 0 and
            r.rdb$relation_name = :table_name
      order by r.rdb$field_position
      into res_field_name
  do
  begin
    res_caption = '';

    select lgf_caption
    from logfield
    where lgf_table_name = :table_name and
          lgf_field_name = :res_field_name
    into res_caption;

    for select res_position, res_domain, res_field_size, res_field_type, res_comment, res_is_key_field, res_is_fk_field,
               res_not_null, res_default, res_fk_table, res_fk_field
    from hp_uxfield_one(:table_name, :res_field_name)
    into :res_position, :res_domain, :res_field_size, :res_field_type, :res_comment, :res_is_key_field, res_is_fk_field,
         :res_not_null, :res_default, :res_fk_table, :res_fk_field
    do
      suspend;
  end
end
