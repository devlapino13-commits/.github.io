CREATE PROCEDURE AAREP_CASHBOOKPAGES_CHECK(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    CBP_DATE DATE,
    ERR_TEXT VARCHAR(200),
    CBP_FIRST_PAGE_NUM INTEGER,
    CBP_LAST_PAGE_NUM INTEGER,
    CBP_SALDO_BEGIN DECIMAL(18,4),
    CBP_SALDO_END DECIMAL(18,4),
    CJR_SALDO_BEGIN DECIMAL(18,4),
    CJR_SALDO_END DECIMAL(18,4))
AS
declare variable prev_last_page_num d_integer;
declare variable cbp_year d_smallint;
declare variable cbp_is_actual d_smallint;
declare variable eps_cash_date_begin d_date;
declare variable sab_id d_integer;
declare variable acc_id d_id;
declare variable al_id d_str_20;
begin
  -- Проверка кассовой книги с начала года по дату отчета

  cbp_year = extract(year from rep_date);
  cbp_date = cast('01.01.' || cbp_year as date);
  prev_last_page_num = null;
  eps_cash_date_begin = f_tmp_const('eps_cash_date_begin');
  while (cbp_date <= rep_date) do
  begin
    err_text = 'Инфо: Ошибок не найдено';

    -- Получаем значения кассовой книги (начальное сальдо, конечное сальдо)
    cbp_saldo_begin = null;
    cbp_saldo_end = null;
    cjr_saldo_begin = null;
    cjr_saldo_end = null;
    if (eps_cash_date_begin > cbp_date) then
      select ol_saldo_begin_dt - ol_saldo_begin_kt,
             ol_saldo_end_dt - ol_saldo_end_kt
      from aarep_cash_journal(:rep_region_id, :cbp_date)
      rows 1
      into cjr_saldo_begin, cjr_saldo_end;
    else
      select ol_saldo_begin_dt - ol_saldo_begin_kt,
             ol_saldo_end_dt - ol_saldo_end_kt
      from aarep_cash_journal_eps(:rep_region_id, :cbp_date)
      rows 1
      into cjr_saldo_begin, cjr_saldo_end;

    cbp_first_page_num = null;
    cbp_last_page_num = null;
    select cbp_first_page_num, cbp_last_page_num, cbp_saldo_begin,
           cbp_saldo_end, cbp_is_actual
    from cashbookpages
    where cbp_date = :cbp_date and
          cbp_from_reg_id = :rep_region_id and
          cbp_number = 0
    into :cbp_first_page_num, :cbp_last_page_num, :cbp_saldo_begin,
         :cbp_saldo_end, :cbp_is_actual;

    -- Проверки
    if (cjr_saldo_begin is null) then
    begin
      if (cbp_saldo_begin is not null) then
        err_text = 'Ошибка: По кассе не было операций, но кассовая книга ' || 'напечатана';
      else
        err_text = 'Инфо: Нет операций';
    end
    else
    begin
      if (cbp_saldo_begin is null) then
        err_text = 'Ошибка: Есть операции по кассе, но кассовая книга ' || 'не печаталась';
      else
      begin
        if (cbp_is_actual = 0) then
          err_text = 'Ошибка: После изменений в предыдущих днях актуальная ' || 'кассовая книга не печаталась';
        else
        begin
          -- Проверка что нумерация между днями не сбилась
          if (prev_last_page_num is not null) then
            if (cbp_first_page_num <> prev_last_page_num + 1) then
              err_text = 'Ошибка: Нумерация нарушена, последний номер ' || 'страницы за предыдущий день:' || prev_last_page_num || ', начальный номер страницы за текущий день:' || cbp_first_page_num;

          -- Проверка, что остатки совпадают
          if (cjr_saldo_begin <> cbp_saldo_begin) then
            err_text = 'Фактическое сальдо на начало не совпадает с сальдо ' || 'на начало в напечатанной кассовой книге.';

          if (cjr_saldo_end <> cbp_saldo_end) then
            err_text = 'Фактическое сальдо на конец не совпадает с сальдо ' || 'на конец в напечатанной кассовой книге.';
        end
      end
    end
    suspend;
    -- Перенос нумерации страниц
    if (cbp_last_page_num is not null) then
      prev_last_page_num = cbp_last_page_num;
    cbp_date = dateadd(day, 1, cbp_date);
  end

  -- если не наступила дата запуска по ЕПС, завершаем работу
  if (eps_cash_date_begin > rep_date) then
    exit;

  cbp_date = rep_date;
  cbp_first_page_num = null;
  cbp_last_page_num = null;
  err_text = 'Разница в остатке Минфин и ЕПС';

  for select sab_id
  from sys_accbalance
  where sab_full_code starting 'Деньги.Нал.Касса'
  into sab_id
  do
  begin
    for select acc_id
    from ref_account, region_list(:rep_region_id, 1, 3, 1)
    where acc_from_sab_id = :sab_id and
          acc_from_reg_id = id_list
    into acc_id
    do
    begin
      cbp_saldo_begin = 0;
      cbp_saldo_end = 0;
      cjr_saldo_begin = 0;
      cjr_saldo_end = 0;

      cjr_saldo_begin = f_acc_bal(acc_id, cbp_date - 1);
      cjr_saldo_end = f_acc_bal(acc_id, cbp_date);

      al_id = '';
      select al_id
      from acc_list
      where al_from_acc_id = :acc_id
      rows 1
      into al_id;

      if (al_id <> '') then
      begin
        cbp_saldo_begin = f_al_bal(al_id, cbp_date - 1);
        cbp_saldo_end = f_al_bal(al_id, cbp_date);
      end

      if (cjr_saldo_begin <> cbp_saldo_begin or cjr_saldo_end <> cbp_saldo_end) then
        suspend;
    end
  end
end
