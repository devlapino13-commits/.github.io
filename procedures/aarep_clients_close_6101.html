CREATE PROCEDURE AAREP_CLIENTS_CLOSE_6101(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    C_NAME_FIO STR_100,
    C_BIRTH_DATE D_DATE,
    CITIZEN STR_100,
    AAREP_K_DELIQ_DAYS D_INTEGER,
    KF_REGION_NAME STR_100,
    KF_OFFICER_NAME STR_100,
    COUNTER D_INTEGER,
    KF_FOLDER_NAME D_STR_100,
    C_POINT D_INTEGER,
    LAST_DATE_CLOSE D_DATE,
    IS_STOPLIST D_STR_10,
    IS_CESSION D_STR_10,
    DEPORT_DATE_BEGIN D_DATE,
    DEPORT_DATE_END D_DATE,
    DEPORT_STATUS D_STR_255)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable kf_id d_id;
declare variable ks_id d_id;
declare variable doc_no str_100;
declare variable doc_date d_date;
begin
  -- RFCRU-6101: Разработка отчета «Закрытые клиенты с закреплением и статусом нахождения в РФ»
  counter = 0;
  for
      select distinct kc.kc_from_c_id
      from kontstate ks
      join kontract k on k.k_id = ks_from_k_id
      join knclient kc on kc_from_k_id = k_id
      join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
      join region_list(:report_region_id,1,3,1) rl on rl.id_list = kal.kal_new_reg_id
      where kc_amount_approved > 0
            and kal.is_delete = 0
            and ks.ks_ol_is_delete = 0
            and ks.ks_state = 2
            and ks.ks_date_begin between :report_date_begin and :report_date_end
      into :c_id
  do
  begin
    select oks_date_end, oks_from_k_id --, oks_date_end - oks.oks_delinq_date
    from oksummary oks, knclient kc
    where kc_from_k_id = oks_from_k_id
          and kc_from_c_id = :c_id
    order by 1 desc
    rows 1
    into :last_date_close, :k_id; -- , :aarep_k_deliq_days;


-- Как определить число дней максимальной непрерывной неактивной просрочки
 
    select coalesce(max(ok.ok_ol_date - ok_sh_date), 0) 
    from operkred ok
    where ok.ok_ol_is_active = 1
         -- and ok.ok_ol_date <= :report_date_end
         and ok.ok_ol_date > ok_sh_date
         and ok_type < 7
         and ok_from_k_id = :k_id
         and ok_os + ok_perc < 0
         and ok_from_c_id = 0
    into :aarep_k_deliq_days;


/*В поле "Дни просрочки (макс)" необходимо отображать максимальные дни просрочки в цикле.
Например: клиент закрыт, но за последний цикл он сделал непрерывно 100 дней просрочки,
и вот 100 дней отображаются как "Дни просрочки (макс)". Если данную информацию сложно
добавлять в данный отчет, то поле можно убрать. Оно не критичное. */

    if (last_date_close > :report_date_end) then
        continue;

    if (aarep_k_deliq_days < 0) then
        aarep_k_deliq_days = 0;

    is_cession = iif(exists (select scl_id
                             from sud_cession_packet
                             join sud_cession_loans on scl_from_scp_id = scp_id
                             where scl_from_k_id = :k_id and
                                   scp_status > 0 and
                                   scp_date <= :report_date_end
                             ), 'да', 'нет');

/*
K_FACT_DISB_KF_NO
K_FACT_CLOSED_REG_ID -- Офис закрытия кредита
K_FACT_DISB_REG_ID -- Регион выдачи
K_FACT_DISB_LO_ID -- Выдавщий КС

kf_region_name  K_LAST_REG_ID -- Текущий офис, за которым числится кредитная папка
kf_folder_name  K_FACT_CLOSED_KF_NO -- Номер папки при закрытии
kf_officer_name K_FACT_CLOSED_LO_ID -- КС закрывщий кредит  */

      select r.reg_name, k.k_fact_closed_kf_no, c_name_fio
      from kontract k
      join region r on r.reg_unicode = k.k_last_reg_id
      join clients c on c.c_id = k.k_fact_closed_lo_id
      where k.k_id = :k_id
      into kf_region_name, kf_folder_name, kf_officer_name;

--    select reg_name, c_name_fio, kal.kal_new_kf_name
--    from knaktlist kal
--    join region on reg_unicode = kal_new_reg_id
--    join clients on c_id = kal_new_kf_officer
--    where kal_from_kf_id = :kf_id and
--          kal_date_end = '31.12.9999' and
--          kal.is_delete = 0
--    into kf_region_name, kf_officer_name, kf_folder_name;


      select c.c_name_fio, c.c_birth_date, rd.rd_name, f_point(c_id, :report_date_end)
      from clients c
      join refdata rd on rd.rd_id = c.c_citizen
      where c.c_id = :c_id
      into c_name_fio, c_birth_date, citizen, c_point;

      is_stoplist = iif(exists (
          select m.m_is_active
          from moratory m
          where m.m_from_c_id = :c_id and m.m_is_active = 1
                and :report_date_end between m.m_date_begin and m.m_date_end
                          ), 'да', 'нет');

      select cd.cds_doc_date_begin, cd.cds_doc_date_end
      from clientdocs cd
      join passkind pk on pk.pk_id = cd.cds_from_pk_id
      join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cd.cds_from_c_id = :c_id and
            pdt.pdt_code = 'deport_document' and
            cd.cds_is_delete = 0 and
            :report_date_end between cd.cds_doc_date_begin and cd.cds_doc_date_end
      order by cd.cds_doc_date_begin desc
      rows 1
      into :deport_date_begin, :deport_date_end;

      select iif(status_full = '', 'Нет', coalesce(status_full, 'Нет'))
      from hp_deprot_status(:c_id)
      into deport_status;

      counter = counter + 1;

    suspend;
  end
end
