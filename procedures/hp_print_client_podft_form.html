CREATE PROCEDURE HP_PRINT_CLIENT_PODFT_FORM(
    MA_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    SURNAME D_STR_100,
    NAME D_STR_100,
    PATRONYMIC D_STR_100,
    CITIZEN D_STR_255,
    C_BIRTH_DATE D_DATE,
    C_INN D_STR_20,
    C_BIRTH_CITY D_STR_255,
    C_PASS_NO D_STR_64,
    C_ADDRESS_REGISTR D_STR_255,
    C_ADDRESS_FACT D_STR_255,
    IS_PUBLIC_CL D_INTEGER,
    CELL_MAIN D_STR_100,
    IS_VERIFIED D_INTEGER,
    C_VERIFY_DATETIME D_DATE,
    C_NAME_FIO_BENEFIC D_STR_255,
    CITIZEN_BENEFIC D_STR_100,
    INN_BENEFIC D_STR_20,
    PASS_NO_BENEFIC D_STR_64,
    PASS_WHO_BENEFIC D_STR_255,
    PASS_DATE_BENEFIC D_STR_20,
    PASS_DATE_EXPIRE_BENEFIC D_STR_20,
    CELL_MAIN_BENEFIC D_STR_100,
    IS_IN_TERROR_LIST D_INTEGER,
    CI_DATETIME D_DATE,
    IS_PUBLIC_CL_BENEFIC D_STR_10,
    RISK_LEVEL D_INTEGER,
    UN_FIO D_STR_255,
    RISK_REASON D_STR_64,
    FORM_FILLED_DATE D_DATE,
    FORM_UPDATE_DATE D_DATE,
    C_IS_RESIDENT D_SMALLINT,
    C_PASS_DATE D_DATE,
    C_PASS_DATE_EXPIRE D_DATE,
    C_PASS_WHO D_STR_255,
    C_DOC_TYPE_STR D_STR_255,
    PIN_CODE D_STR_CODE)
AS
declare variable C_ID D_ID;
declare variable C_ID_BENEFIC D_ID;
declare variable CI_STATUS D_ID;
declare variable TMP_ID D_ID;
declare variable C_DOC_TYPE D_INTEGER;
declare variable EDL_ID D_ID;
declare variable EDL_BLANK_DATE D_DATE;
begin
  is_public_cl_benefic = 'Нет';

  if (not exists(select * from clients
                 where c_id = :c_id)) then
  begin
    tmp_id = f_json_str_get_value(add_params, 'c_id', null);
    if (tmp_id is not null) then
      c_id = tmp_id;
  end

  select fl.fl_name1, fl.fl_name2, fl.fl_name3,
         cl.c_birth_date, cl.c_birth_city, cl.c_pass_no,
         cl.c_inn, rd.rd_name, cl.c_verify_status,
         cl.c_verify_datetime, cl.c_is_resident,
         cl.c_pass_date, cl.c_pass_date_expire,
         cl.c_pass_who, cl.c_doc_type
  from clients cl
  inner join fiolist fl on fl.fl_id = cl.c_from_fl_id
  inner join refdata rd on rd.rd_id = cl.c_citizen
  where cl.c_id = :c_id
  into :surname, name, patronymic,
        c_birth_date, c_birth_city, c_pass_no,
        c_inn, citizen, is_verified,
        c_verify_datetime, c_is_resident,
        c_pass_date, c_pass_date_expire,
        c_pass_who, c_doc_type;

  select pk_name
  from passkind
  where pk_id = :c_doc_type
  into c_doc_type_str;

  cell_main = f_cont_info(0, :c_id, 'cell_main', 'ok');

  select coalesce(adl.adl_address_str, '')
  from addrclient adc
  inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
  where adc.adc_address_type = 1 and -- юридический
        adc.adc_from_c_id = :c_id
  into :c_address_fact;
  if (c_address_fact = 'Не указан') then
    c_address_fact = '';

  select coalesce(adl.adl_address_str, '')
  from addrclient adc
  inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
  where adc.adc_address_type = 2 and -- фактический
        adc.adc_from_c_id = :c_id
  into :c_address_registr;
  if (c_address_registr = 'Не указан') then
    c_address_registr = '';

  select cl.c_is_public, cl.c_pft_risk_koef
  from clients cl
  where cl.c_id = :c_id
  into :is_public_cl, risk_level;

  if (risk_level = 0) then
    select 2
    from clients cl
    where cl.c_id = :c_id
      and cl.c_is_public <> 0
    into :risk_level;


  ---- СВЕДЕНИЯ О БЕНЕФИЦИАРНОМ СОБСТВЕННИКЕ ----
  select cl.c_name_fio, rd.rd_name, cl.c_inn, cl.c_pass_no,
         cl.c_pass_who, f_datetostr(cl.c_pass_date), f_datetostr(cl.c_pass_date_expire), cb.cb_ben_from_c_id,
         iif(cl.c_is_public = 1, 'Да', 'Нет')
  from clientbenefic cb
  inner join clients cl on cl.c_id = cb.cb_from_c_id
  inner join refdata rd on rd.rd_id = cl.c_citizen
  where cb.cb_from_c_id = :c_id
  into :c_name_fio_benefic, citizen_benefic, inn_benefic, pass_no_benefic,
        pass_who_benefic, pass_date_benefic, pass_date_expire_benefic, c_id_benefic,
        is_public_cl_benefic;

  cell_main_benefic = f_cont_info(0, :c_id_benefic, 'cell_main', 'ok');

  select first 1 c.c_name_fio
  from miniappclient mac
  inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
  inner join clients c on c.c_id = ma.ma_manager_id
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1
  order by ma.ma_number desc
  into :un_fio;

  is_in_terror_list = 0;

  select iif(status > 1, 1, 0), answer_date
  from hp_afm_get_status('afm', :c_id, :ma_id)
  into :is_in_terror_list, :ci_datetime;

  risk_reason = '';

  if (:is_public_cl = 1) then
  risk_reason = 'Клиент является публичным должностным лицом';

  form_filled_date = current_date;
  form_update_date = current_date;

  -- пин-код электронной подписи клиента. Текст пустой если не подписан
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);

    if (edl_id is not null) then
    begin
     select edl.edl_blank_date
     from edoc_list edl
     where edl.edl_id = :edl_id
     into edl_blank_date;

     
     form_filled_date = edl_blank_date;
     form_update_date = edl_blank_date;

      select eds.eds_pin
      from edoc_sign eds
      where eds.eds_from_edl_id = :edl_id
        and eds.eds_from_c_id = :c_id
        and eds.eds_status = 1
      into :pin_code;
    end
  end

  suspend;
end
