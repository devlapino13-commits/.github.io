CREATE PROCEDURE HP_DAILY_SMS_DELINQ_DAYS(
    NUM_OF_DAYS D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable PREVIOUS_C_ID D_ID;
declare variable K_ID D_ID;
declare variable MSG_ID D_ID;
declare variable MSG_TEXT_1 D_STR_255;
declare variable MSG_TEXT_2 D_STR_255;
declare variable PHONE_NUMBER D_STR_20;
declare variable REP_AMOUNT decimal(18,0);
declare variable NSD_PERCENT D_INTEGER;
begin
    if (num_of_days not in (5,70)) then
        exit;

    -- RFCAZ-381
    previous_c_id = 0;
    msg_text_1 = 'Mehkemeye verilme barede!Hormetli musteri,“M Bulak BOKT” MMC-ne olan gecikdirilmis kredit borcunuzu odemediyiniz ucun isiniz Mehkemeye verilecek.';
    msg_text_2 = 'İcra haqqında qanuna esasen barenizde Mecburi icra tedbirleri gorulecek(olkeden getmek huququnun mehdudlasdirilmasi,emlak uzerine hebs/telebin satmaq yolu ile emlaka,resmi gelirlerinize yoneldilmesi,mehkeme qerarı esasında mecburi getirilme ve s.)';

    for select
        kc_from_c_id,
        kc_from_k_id
    from oksummary
    inner join knclient on oks_from_k_id = kc_from_k_id
        and kc_amount_approved > 0
    where current_date between oks_date_begin and oks_date_end
        and current_date - oks_delinq_date = :num_of_days
        and oks_close_state = 0
    order by kc_from_c_id
    into
        c_id,
        k_id
    do
    begin
        if (previous_c_id = c_id) then continue;
        else previous_c_id = c_id;

        msg_id = null;
        phone_number = null;
        phone_number = f_cont_info('clients', :c_id, 'cell', 'ok');

        if (num_of_days = 5 and phone_number is not null) then
        begin
            if (exists
                (
                    select ok_from_k_id
                    from operkred, kontstate
                    where ks_from_k_id = ok_from_k_id and
                    ks_ol_is_delete = 0 and
                    ks_date_begin <= current_date and
                    ks_date_end >= current_date and
                    ks_state = 1 and
                    ok_ol_is_active = 1 and
                    ok_sh_date = current_date - :num_of_days and
                    ok_type <> 8 and
                    ok_type <> 7 and
                    ok_from_c_id = 0 and
                    ok_from_k_id = :k_id
                    group by ok_from_k_id
                    having sum(ok_sh_os + ok_os + ok_sh_perc + ok_perc) > 0
                )
            ) then
            begin
                select coalesce(nsd_percent / 100, 0)
                from nspconfig_data
                where nsd_from_nsm_id = -1 and
                    (current_date + :num_of_days) between nsd_date_begin and nsd_date_end
                into nsd_percent;

                if (coalesce(nsd_percent, 0) = 0) then
                    nsd_percent = 0;

                select sum(round(zad_os + sh_calc_perc + sh_calc_perc * :nsd_percent + sh_calc_fine)) delinq_sum
                from aarep_make_zad_list_temp(:k_id, current_date, 1)
                where sh_date = current_date - :num_of_days
                into rep_amount;
    
                msg_text_2 = 'MBulaka odenishiniz '||:num_of_days||' gun gecikir. Bugune olan gecikme borcunuz '||:rep_amount||' AZN. Gecikmede olan her gune cerime hesablanir';
    
                insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id, msg_command_kind, msg_object_kind, msg_object_id)
                values (1, :phone_number, '', :msg_text_2, -1, 1000000000000, :num_of_days||'delinq_days', 0, :c_id)
                returning msg_id
                into :msg_id;
            end
        end

        if (num_of_days = 70 and phone_number is not null) then
        begin
            -- RFCAZ-385
            -- чтобы не засорять таблицу delinquent_loan_notify, беру только один msg_id
            insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id, msg_command_kind, msg_object_kind, msg_object_id)
            values (1, :phone_number, '', :msg_text_1, -1, 1000000000000, :num_of_days||'delinq_days', 0, :c_id);

            insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id, msg_command_kind, msg_object_kind, msg_object_id)
            values (1, :phone_number, '', :msg_text_2, -1, 1000000000000, :num_of_days||'delinq_days', 0, :c_id)
            returning msg_id
            into :msg_id;

        end

        insert into delinquent_loan_notify (dl_from_c_id, dl_from_k_id, dl_from_msg_id, dl_from_el_code)
        values (:c_id, :k_id, :msg_id, cast(:num_of_days as d_str_code));
    end
end
