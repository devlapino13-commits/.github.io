CREATE PROCEDURE MC_CES_DO_ASP_SIGN(
    MCL_ID D_ID)
RETURNS (
    DO_SIGN D_BOOL)
AS
declare variable C_ID D_ID;
declare variable C_VERIFY_STATUS D_SMALLINT;
begin
    if(f_get_db_id() not in (39801, 41701)
        or f_get_const('ces_active') = 0
    ) then
    begin
        do_sign = 1; -- обходим ОЭП
        suspend;
        exit;
    end

    select
        mcl_from_c_id,
        c_verify_status
    from mc_login
    inner join clients on mcl_from_c_id = c_id
    where mcl_id = :mcl_id
        and mcl_is_delete = 0
    into
        :c_id,
        :c_verify_status;

    do_sign = 0;
    if (c_verify_status <> 2) then
        do_sign = 0;

    if (f_asp_signed(:c_id, current_date) = 1) then
        do_sign = 1;

    suspend;
end
