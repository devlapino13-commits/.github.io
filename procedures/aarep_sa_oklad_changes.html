CREATE PROCEDURE AAREP_SA_OKLAD_CHANGES(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    SAO_FROM_C_ID DECIMAL(18,0),
    SAO_ID INTEGER,
    SAO_DATE DATE,
    SAO_SUMMA DECIMAL(18,4),
    SAO_MEMO VARCHAR(255),
    SAO_IS_NET INTEGER,
    SAO_CHILD_KOLICH INTEGER,
    SAO_IS_LGOTA INTEGER,
    SAO_FROM_UN_ID DECIMAL(18,0),
    SAO_DATETIME TIMESTAMP,
    SAO_USERNAME VARCHAR(30),
    SAO_PERSONAL_FIO VARCHAR(100),
    SAO_MOUNTAIN_PERC DECIMAL(18,7),
    SAO_IS_HOUR INTEGER,
    SAO_SF_LGOTA DECIMAL(18,7),
    SAO_PDNAL_LGOTA DECIMAL(18,4),
    SAO_DONATION DECIMAL(18,4),
    SAO_JL_NAME VARCHAR(100),
    SAO_REG_NAME VARCHAR(100),
    PS_IS_WORKING_STR VARCHAR(50),
    SAO_OKLAD_CHANGES DECIMAL(18,4),
    SAO_OKALD_CHANGED INTEGER)
AS
declare variable c_id decimal(18,0);
declare variable ps_is_working integer;
declare variable can_suspend integer;
declare variable ps_from_reg_id integer;
declare variable last_found integer;
begin
  for with tabs
      as (select distinct sao_from_c_id as c_id, 1 as changed
          from sa_oklad
          where sao_date between :rep_date_begin and :rep_date_end and
                sao_summa <> 0
          union
          select ps_from_c_id as c_id, 0 as changed
          from hr_pstatus
          where ps_status = 1 and
                ps_date_begin <= (:rep_date_begin - 1) and
                ps_date_end >= (:rep_date_begin - 1) and
                ps_is_working = 1)
      select c_id, max(changed)
      from tabs
      group by c_id
      into :c_id, sao_okald_changed
  do
  begin
    can_suspend = 0;
    sao_reg_name = '';
    sao_jl_name = '';
    ps_is_working = 0;
    ps_from_reg_id = 0;
    select jl_name, reg_code || ' ' || reg_name, ps_is_working, ps_from_reg_id
    from region, joblist, hr_pstatus
    where ps_from_reg_id = reg_unicode and
          ps_from_jl_id = jl_id and
          ps_status = 1 and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          ps_from_c_id = :c_id
    order by ps_is_working desc
    rows 1
    into :sao_jl_name, :sao_reg_name, ps_is_working, ps_from_reg_id;
    if (ps_is_working = 1) then
      ps_is_working_str = 'Работает';
    else
      ps_is_working_str = 'Уволен';
    if (ps_from_reg_id is null) then
      ps_from_reg_id = 0;

    if (rep_region_id <> 1000000) then
    begin
      if (exists(select *
                 from region_list(:ps_from_reg_id, 1, -1, 0)
                 where id_list = :rep_region_id)) then
        can_suspend = 1;
    end
    else
      can_suspend = 1;

    last_found = 0;
    if (can_suspend = 1) then
    begin
      for select sao_from_c_id, sao_id, sao_date, sao_summa, sao_memo, sao_is_net, sao_child_kolich, sao_is_lgota,
                 sao_from_un_id, sao_datetime, sao_username, sao_personal_fio, sao_mountain_perc, sao_is_hour,
                 sao_sf_lgota, sao_pdnal_lgota, sao_donation, sao_oklad_changes
          from aarep_sa_oklad_one(:c_id)
          order by sao_date desc
          into :sao_from_c_id, :sao_id, :sao_date, :sao_summa, :sao_memo, :sao_is_net, :sao_child_kolich, :sao_is_lgota,
               :sao_from_un_id, :sao_datetime, :sao_username, :sao_personal_fio, :sao_mountain_perc, :sao_is_hour,
               :sao_sf_lgota, :sao_pdnal_lgota, :sao_donation, :sao_oklad_changes
      do
      begin
        if (sao_date <= :rep_date_end) then
        begin
          if (last_found = 0) then
          begin
            if (sao_date < :rep_date_begin) then
              last_found = 1;
            suspend;
          end
        end
      end
    end
  end
end
