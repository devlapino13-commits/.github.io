CREATE PROCEDURE MC_EXTINFO_REQUEST(
    MCL_ID D_ID NOT NULL,
    CODE D_STR_CODE NOT NULL,
    EXTRA_DATA D_STR_4096 = '')
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255,
    EIL_ID D_ID)
AS
declare variable mcl_lang d_str_lang;
declare variable c_id d_id;
declare variable ma_id d_integer;
declare variable un_id d_id;
declare variable client_pass_pk_id d_integer;
begin
    /*
      Процедура используется в КМП для отправки запросов через extinfolist
      Статус запроса затем проверяется через процедуру - mc_extinfo_status
    */
    res_code = 'success';
    res_text = '';
    un_id = 1000000000000;
    mcl_lang = f_get_lang_code(0, mcl_id);

    -- Проверка паспорта через СМЭВ
    if (code = 'passport_smev') then
    begin
        
        c_id = (select mcl.mcl_from_c_id from mc_login mcl where mcl.mcl_id = :mcl_id);
        if (c_id is null) then
        begin
            res_code = 'error';
            res_text = f_lang(:mcl_lang, 'Не удалось загрузить данные профиля. Попробуйте войти в приложение заново');
            suspend;
            exit;
        end

        -- У клиента должен быть верифицированный внутренний паспорт РФ в базе для проверки по СМЭВ
        select first 1 cds_from_pk_id
        from clientdocs cds
        where cds.cds_from_c_id = :c_id
            and cds_is_main_doc = 1
            and cds_is_delete = 0
            and cds_verify_status = 2
            and cds_index_doc_no <> ''
        order by cds_id desc
        into client_pass_pk_id;
        

        -- passkind (pk_id = 12) - Внутренний паспорт РФ
        if (client_pass_pk_id is distinct from 12) then
        begin
            -- Если попали сюда, значит паспорта нет, либо прикреплен не внутренний паспорт РФ
            res_code = 'error';
            res_text = f_lang(mcl_lang, 'Не удалось начать проверку паспорта: у вас не прикреплен/не верифицирован внутренний паспорт РФ');
            suspend;
            exit;
        end

        begin
        execute procedure hp_extinfo_insert_smevpass(:c_id, 0, 1000000000000);

        -- Если возникла ошибка при отправке запроса
        when any do
            begin
                res_code = 'error';
                res_text = f_lang(mcl_lang, 'Что-то пошло не так. Попробуйте повторить попытку позже');
                suspend;
                exit;
            end
        end

        -- Вытаскиваем eil_id. Т.к. процедура отправки запроса по СМЭВ не возвращает eil_id, приходится делать так
        select first 1 eil.eil_id
        from extinfolist eil
        inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
        where eic.eic_code = :code
            and eil.eil_from_c_id = :c_id
            and eil.eil_date = current_date
        order by eil.eil_id desc
        into :eil_id;

    end

    -- RFCRU-5828
    -- Запросы для финальной проверки онлайн-заявки: банкротство (kontur),
    -- кредитная история (equifax_40), ССП и самозапрет (bki_ssp_40)
    else if (code in ('kontur', 'equifax_40', 'bki_ssp_40')) then
    begin

        c_id = (select mcl.mcl_from_c_id from mc_login mcl where mcl.mcl_id = :mcl_id);
        if (c_id is null) then
        begin
            res_code = 'error';
            res_text = f_lang(:mcl_lang, 'Не удалось загрузить данные профиля. Попробуйте войти в приложение заново');
            suspend;
            exit;
        end

        ma_id = f_json_str_get_value(extra_data, 'ma_id', 0);
        if (coalesce(:ma_id, 0) = 0) then
        begin
            res_code = 'error';
            res_text = f_lang(:mcl_lang, 'Не удалось начать проверку. Попробуйте ещё раз');
            suspend;
            exit;
        end

        -- действующий запрос (новый, в работе либо успешный) в пределах
        -- срока актуальности - повторно не отправляем, возвращаем его eil_id;
        -- запросы с ошибкой (eil_status < -1) не учитываем, чтобы можно
        -- было отправить новый запрос после сбоя сервиса
        select first 1 eil.eil_id
        from extinfolist eil
        inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
            and eic.eic_code = :code
        where eil.eil_from_c_id = :c_id
            and eil.eil_status >= -1
            and datediff(day, eil.eil_date, current_date) <= eic.eic_info_actual_days
        order by eil.eil_id desc
        into :eil_id;

        -- Если нет актуального запроса, тогда формируем новый
        if (eil_id is null) then
        begin
            -- отправляет клиентов заявки и документы смены ФИО
            if (code = 'kontur') then
                execute procedure hp_extinfo_insert_kontur('miniapp', :ma_id, 0, :un_id);
            else
                execute procedure hp_extinfo_insert_bki_cfg(:code, :c_id, f_objkind('miniapp', 'id'), :ma_id, 0, :un_id);

            -- Берём последний eil_id
            select first 1 eil.eil_id
            from extinfolist eil
            inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
                and eic.eic_code = :code
            where eil.eil_from_c_id = :c_id
            order by eil.eil_id desc
            into :eil_id;
        end

        -- Если даже после формирования у нас нет запроса, тогда пробрасываем ошибку
        if (eil_id is null) then
        begin
            res_code = 'error';
            res_text = f_lang(:mcl_lang, 'Не удалось начать проверку. Попробуйте ещё раз');
            suspend;
            exit;
        end

    end
    else
    begin
        res_code = 'error';
        res_text = f_lang(:mcl_lang, 'Не удалось начать проверку. Обновите приложение и попробуйте снова');
        suspend;
        exit;
    end

    suspend;

    -- Если возникла ошибка при отправке запроса
    when any do
    begin
        res_code = 'error';
        res_text = f_lang(mcl_lang, 'Что-то пошло не так. Попробуйте повторить попытку позже');
        suspend;
        exit;
    end
end
