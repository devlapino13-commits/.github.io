declare function getcatval(kind d_str_code, code d_str_code)
  returns d_str_255
  as
  begin
    return coalesce((select bct_name from bki_catalog where bct_kind = :kind and bct_code = :code), 'Неопределен');

declare procedure get_debt_from_json(jsa d_integer)
  returns(
    pos d_integer, 
    debt_sign d_bool,
    calc_date d_date,
    debt_date d_date,
    first_sum d_currency,
    debt_op_sum d_currency,
    debt_percent_sum d_currency,
    debt_sum d_currency
  )
  as
    declare variable cnt d_integer;
    declare variable calc_date_str d_str_255;
    declare variable debt_date_str d_str_255;
    declare variable jso d_integer;
  begin
    if (jsa is null) then
      exit;

    cnt = json.count_get(jsa);
    pos = 0;
    while (pos < cnt) do
    begin
      jso = json.arr_get_obj(jsa, pos);

      debt_sign = json.obj_get_val(jso, 'sign', 1);
      debt_op_sum  = replace(replace(json.obj_get_val(jso, 'opSum', 0), ',', '.'), '-', 0);
      debt_percent_sum  = replace(replace(json.obj_get_val(jso, 'percentSum', 0), ',', '.'), '-', 0);
      debt_sum  = replace(replace(json.obj_get_val(jso, 'sum', 0), ',', '.'), '-', 0);
      first_sum  = replace(replace(json.obj_get_val(jso, 'firstSum', 0), ',', '.'), '-', 0);
      calc_date_str = json.obj_get_val(jso, 'calcDate', null);
      debt_date_str = json.obj_get_val(jso, 'date', null);

      calc_date = iif(calc_date_str = '-', null, calc_date_str);
      debt_date = iif(debt_date_str = '-', null, debt_date_str);

      if (debt_sum = 0) then
        debt_date = null;

      suspend;
      pos = pos + 1;
    end 
  end

declare procedure get_contract_amount(jsa d_integer)
  returns(
    summ d_currency,
    currency d_str_code
  )
  as
    declare variable cnt d_integer;
    declare variable pos d_integer;
    declare variable jso d_integer;
  begin
    if (jsa is null) then
      exit;

    cnt = json.count_get(jsa);
    pos = 0;
    summ = 0;
    while (pos < cnt) do
    begin
      jso = json.arr_get_obj(jsa, pos);
      pos = pos + 1;

      summ = summ + cast(replace(replace(json.obj_get_val(jso, 'sum', 0), ',', '.'), '-', 0) as d_currency);
      currency = json.obj_get_val(jso, 'currency');
    end

    suspend;
  end

    num_str d_str_code)
  returns d_currency

      return 0;
  end

    return json.obj_get_arr(handle, path);
  else if (val_type = 'Object') then
  begin
    handle = json.obj_get_obj(handle, path);
    jsa = json.arr_create();
    json.arr_set_obj(jsa, -1, handle);

    return jsa;
  end
  else
    exception e_default path || ' is not Array/Object';
end

      from get_debt_from_json(:jsa_debt_overdue)
      order by calc_date desc, debt_sum desc
      rows 1
      into cred_sum_overdue, cred_sum_overdue_op, overdue_debt_date;

      select overdue_day, my_strtonum(last_payout_sum), last_payout_date
      from get_payment_from_json(:jsa_payments)
      order by last_payout_date desc
      rows 1
      into cred_day_overdue, last_payout_sum, last_payout_date;

      cred_active = case
        when stop_load_code = '3' and purchaser_commercial_type <> '3'  then 2 --'Договор продан (переуступка прав требований)'
        when stop_load_code = '3' and purchaser_commercial_type = '3'   then 5 --'Договор продан коллекторам (переуступка прав требований)'
        when stop_load_code not in ('', '3')                            then 8 --'Договор между Партнером и Бюро об оказании информационных услуг прекратил действие или расторгнут'
        when contract_end_reason = '10'                                 then 4 --'Договор рефинансирован'
        when contract_end_reason not in ('', '10')                      then 0 --'Договор закрыт'
        else 1 --'Договор активен'
      end;

      cred_active_str = case cred_active
        when 0 then  'Договор закрыт'
        when 1 then  'Договор активен'
        when 2 then  'Договор продан (переуступка прав требований)'
        when 4 then  'Договор рефинансирован'
        when 5 then  'Договор продан коллекторам (переуступка прав требований)'
        when 8 then  'Договор между Партнером и Бюро об оказании информационных услуг прекратил действие или расторгнут'
        else 'Неопределен'
      end;

      cred_active_order = case
        when cred_active = 2 then 2
        when cred_active = 5 then 3
        when cred_active in (0,8,4) then 4
        else 1
      end;
  
      cred_active_order_str = case cred_active_order
        when 1 then 'Активные договоры'
        when 2 then 'Проданные договоры'
        when 3 then 'Переданные коллекторам договоры'
        when 4 then 'Закрытые договоры'
      end;
  
      cred_max_overdue = 0;
      cred_max_day_overdue = 'Без просрочки';

      if (cred_no <= sumary_count) then
      begin
        jso_summary = json.arr_get_obj(jsa_summary, cred_no - 1);
        cred_max_overdue = my_strtonum(json.obj_get_val(jso_summary, 'cred_max_overdue', 0));
  
        cred_max_day_overdue = case
          when f_strtonum(json.obj_get_val(jso_summary, 'delay_more',  0)) > 0 then 'более 90'
          when f_strtonum(json.obj_get_val(jso_summary, 'delay90',     0)) > 0 then '61 - 90'
          when f_strtonum(json.obj_get_val(jso_summary, 'delay60',     0)) > 0 then '31 - 60'
          when f_strtonum(json.obj_get_val(jso_summary, 'delay30',     0)) > 0 then '6 - 30'
          when f_strtonum(json.obj_get_val(jso_summary, 'delay5',      0)) > 0 then 'до 6'
          else 'Без просрочки'
        end;
      end

      suspend;
    end
  end
end
