CREATE PROCEDURE HP_IS_USER_CAN_VIEW_PERSONAL(
    UN_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0))
RETURNS (
    CAN_VIEW INTEGER,
    MESSAGE_TXT VARCHAR(255))
AS
declare variable reg_id integer;
declare variable reg_all integer;
declare variable ps_is_working integer;
declare variable st_user_client_access integer;
declare variable st_user_staff_access integer;
declare variable un_name varchar(100);
declare variable tmp integer;
declare variable uniqtab_username integer;
declare variable uniqtab_region integer;
declare variable db_id d_integer;
declare variable crlf d_str_lang;
begin
  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Region')
  into :uniqtab_region;
  message_txt = '';
  db_id = f_get_db_id();
  crlf = _ascii x'0D0A';

  st_user_client_access = 0;
  st_user_staff_access = 0;

  un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN');

  select un_id
  from username
  where un_name = :un_name
  into :un_id;
  if (un_id is null) then
    un_id = 0;

  tmp = 0;
  select has_access
  from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Client_Data_Edit')
  into :tmp;
  if (tmp = 1) then
    st_user_client_access = 2;
  else
  begin
    select has_access
    from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Client_Detail_Readonly')
    into :tmp;
    if (tmp = 1) then
      st_user_client_access = 1;
  end

  tmp = 0;
  select has_access
  from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Staff_Data_Edit')
  into :tmp;
  if (tmp = 1) then
    st_user_staff_access = 2;
  else
  begin
    select has_access
    from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Staff_Detail_Readonly')
    into :tmp;
    if (tmp = 1) then
      st_user_staff_access = 1;
  end

  select has_access
  from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Superuser')
  into :tmp;
  if (tmp = 1) then
  begin
    st_user_client_access = 2;
    st_user_staff_access = 2;
  end

  can_view = 0;
  message_txt = 'У Вас нет прав на просмотр данных физ.лиц';
  select ps_from_reg_id, ps_is_working
  from hr_pstatus
  where ps_from_c_id = :c_id and
        ps_date_end >= '31.12.2999' and
        ps_status = 1
  order by ps_date_begin desc
  rows 1
  into :reg_id, :ps_is_working;
  if (ps_is_working = 1) then
  begin
    if (reg_id is null) then
    begin
      can_view = st_user_staff_access;
      message_txt = '';
    end
    if (reg_id is not null) then
    begin
      can_view = st_user_staff_access;
      message_txt = 'У Вас нет прав на просмотр данных сотрудников';

      for select distinct pm_subject_id
          from permissions
          where pm_subject_kind = :uniqtab_region and
                pm_object_kind = :uniqtab_username and
                pm_object_id = :un_id and
                pm_is_delete = 0 and
                (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                (coalesce(pm_date_end, '31.12.2999') >= current_date)
          into :reg_all
      do
      begin
        if (exists(select id_list
                   from region_orig_list(:reg_all, 1, -1, 1)
                   where id_list = :reg_id)) then
        begin
          message_txt = '';
        end
      end
      if (message_txt <> '') then
        can_view = 0;
    end
    if (st_user_staff_access = 0) then
    begin
      message_txt = 'У Вас нет прав на просмотр/редактирование данных сотрудников';
      can_view = 0;
    end
  end
  else
  begin
    if (un_id in (1010100000353)) then
    begin
      can_view = st_user_client_access;
      message_txt = '';
    end
    else
    begin
      can_view = 0;
      message_txt = 'У Вас нет прав на просмотр данных активных клиентов других офисов';
      reg_id = null;
      for select distinct k_last_reg_id
          from knclient, kontstate, kontract
          where kc_amount_approved > 0 and
                kc_from_k_id = ks_from_k_id and
                kc_from_c_id = :c_id and
                ks_date_begin <= current_date and
                ks_date_end >= current_date and
                ks_state = 1 and
                ks_from_k_id = k_id rows 1
          into :reg_id
      do
      begin
        if (st_user_client_access >= 1) then
        begin
          for select distinct pm_subject_id
              from permissions
              where pm_subject_kind = :uniqtab_region and
                    pm_object_kind = :uniqtab_username and
                    pm_object_id = :un_id and
                    pm_is_delete = 0 and
                    (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                    (coalesce(pm_date_end, '31.12.2999') >= current_date)
              into :reg_all
          do
          begin
            if (exists(select id_list
                       from region_orig_list(:reg_all, 1, -1, 1)
                       where id_list = :reg_id)) then
            begin
              can_view = st_user_client_access;
              message_txt = '';
            end
          end
          if (message_txt <> '') then
            can_view = 0;
        end
      end
      if (reg_id is null) then
      begin
        can_view = st_user_client_access;
        message_txt = '';
      end
    end
  end
  if (db_id = 81001 and not exists(select *
                 from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.CanEditTerrAndEx')
                 where has_access = 1)) then
  begin
    if (exists(select * from aarep_check_finpolis_one(:c_id))) then
    begin
      can_view = 0;
      message_txt = 'ВНИМАНИЕ,' || crlf ||
                    'Необходимо связаться с комплайнс менеджером ГО.' || crlf ||
                    'Дальнейшие операции с этим контрагентом запрещены.' || crlf ||
                    'Детали в меню "Проверка по спискам ТиЭ"' || crlf;
    end
  end

  suspend;
end
