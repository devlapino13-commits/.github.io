CREATE PROCEDURE HP_GET_UI_UXEXPRESSION(
    UA_CODE D_STR_100)
RETURNS (
    RES_FR_CODE D_STR_CODE,
    RES_IS_PARAM D_BOOL,
    RES_FIELD_NAME D_STR_CODE,
    RES_EXPRESSION D_STR_LONG,
    RES_MESSAGE D_STR_255)
AS
declare variable ux_code d_str_100;
declare variable ufp_code d_str_100;
declare variable ufp_name d_str_code;
declare variable required_str d_str_255;
declare variable ufp_required d_bool;
declare variable ufp_editor d_str_code;
begin
  select ux_code
  from hp_get_uxalias(:ua_code)
  into ux_code;

  for select ucf_ufr_code
  from uxcodeframe
  where ucf_ux_code  = :ux_code
  into res_fr_code
  do
  begin
    for select ufp_code, ufp_origin_name, ufp_name, ufp_required, ufp_is_param, ufp_editor
    from uxfrparam
    where ufp_ufr_code = :res_fr_code and
          ufp_in_use = 1 and
          ufp_is_delete = 0
    into ufp_code, res_field_name, ufp_name, ufp_required, res_is_param, ufp_editor
    do
    begin
      res_expression = '';
      res_message = '';

      if (ufp_editor <> 'mask') then
      begin
        select uex_expression, uex_message
        from uxexpression
        where uex_ufp_code = :ufp_code and
              uex_inactive = 0
        into res_expression, res_message;
      end

      if (ufp_required = 1) then
      begin
        required_str = res_field_name || ' is not null';
        if (res_expression = '') then
        begin
          res_expression = required_str;
          res_message = 'Поле "' || ufp_name || '" не заполнено';
        end else
          res_expression = '(' || required_str || ') and (' || res_expression || ')';
      end else
      if (res_expression <> '') then
      begin
        required_str = res_field_name || ' is null';
        res_expression = '(' || required_str || ') or (' || res_expression || ')';
      end

      if (res_expression <> '') then
        suspend;
    end

    res_is_param = 0;
    res_field_name = '';
    for select uex_expression, uex_message
    from uxexpression
    where uex_ux_code = :res_fr_code and
          uex_ufp_code is null and
          uex_inactive = 0
    order by uex_position
    into res_expression, res_message
    do
      suspend;
  end
end
