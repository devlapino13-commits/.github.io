CREATE PROCEDURE CM_KG_0172_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id id_type;
declare variable pr_code d_str_100;
declare variable k_amount_approved decimal(18,2);
declare variable active_balance decimal(18,2);
begin
  res_error_is_critical = 0;

  for select kc.kc_from_c_id, pr_code, k.k_amount_approved
      from knclient kc
      inner join kontract k on k.k_id = kc.kc_from_k_id
      inner join product pr on pr.pr_id=k.k_from_pr_id
      where kc.kc_from_k_id = :k_id
            and k.k_disb_kind = 6
            and kc.kc_amount_approved > 0
            and pr.pr_code not containing '.PAR.'
            and pr.pr_code not containing '.jonokoi_cards.'
      into c_id, pr_code, k_amount_approved
  do
  begin
    if (pr_code containing '.online_loan.') then
    begin
      select sum(oks.oks_balance)
      from knclient kc
      inner join oksummary oks on oks.oks_from_k_id = kc.kc_from_k_id and
            current_date between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_close_state = 0
      inner join kontract kt on kt.k_id = oks.oks_from_k_id
      inner join product pr on pr.pr_id = kt.k_from_pr_id and
            pr.pr_code containing '.online_loan.'
      where kc.kc_from_c_id = :c_id and
            kc.kc_amount_approved > 0
      into active_balance;

      if (k_amount_approved + coalesce(active_balance, 0) > 49999) then
      begin
        res_error_is_critical = 2;
        res_error_text = 'Совокупный баланс онлайн займа на карту не может превышать 49999 сом';
        suspend;
      end
      exit;
    end 

    if (exists(select *
               from kontstate ks
               inner join kontract k on k.k_id = ks.ks_from_k_id and
                     k.k_disb_kind = 6
               inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id and
                     kc.kc_from_c_id = :c_id
                     and kc.kc_amount_approved > 0
               where current_date between ks.ks_date_begin and ks.ks_date_end 
                     and ks.ks_state = 1
                     and ks.ks_ol_is_delete=0)) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'имеется активный кредит на карту М Булак!';
      suspend;
    end
  end
end
