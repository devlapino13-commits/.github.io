CREATE PROCEDURE HP_COM_PACKET_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    LIST_KIND D_STR_CODE)
RETURNS (
    CMP_ID D_INTEGER,
    CMP_STR_NO D_STR_CODE,
    CMP_DATE D_DATE,
    CMP_IS_IPF D_SMALLINT,
    CMP_MEMO D_STR_LONG,
    CMP_FROM_REG_ID D_INTEGER,
    CMP_STATUS D_SMALLINT,
    CMP_STATUS_STR D_STR_CODE,
    CMP_REGION_NAME D_STR_100,
    CMP_KIND D_STR_CODE,
    CMP_DECISION D_STR_CODE,
    CMP_FROM_BLN_ID D_INTEGER,
    COUNT_TOTAL D_INTEGER,
    COUNT_APPROVED D_SMALLINT,
    COUNT_DONE D_INTEGER)
AS
declare variable enum_approved d_integer;
declare variable enum_done d_integer;
declare variable only_actual d_smallint;
declare variable need_suspend d_smallint;
declare variable cmp_from_cmk_id d_integer;
declare variable cmp_from_cmd_id d_integer;
begin
  -- возвращает список заседания кредитного комитета
  -- list_kind может быть:
  --   full    -- полный список за период
  --   execute -- готовые к исполнению

  if (trim(coalesce(list_kind, '')) = '') then
    list_kind = 'full';

  enum_approved = f_enum('com_action_status', 'approved', 'id');
  enum_done = f_enum('com_action_status', 'done', 'id');

  if (position('.actual', list_kind) > 0) then
  begin
    only_actual = 1;
    list_kind = replace(list_kind, '.actual', '');
  end


  for select cmp_id,
             cmp_str_no,
             cmp_date,
             cmp_is_ipf,
             cmp_memo,
             cmp_from_reg_id,
             cmp_status,
             reg_code || ' ' || reg_name,
             cmp_from_cmk_id,
             cmp_from_cmd_id
      from com_packet, region
      where cmp_date between :rep_date_begin and :rep_date_end and
            cmp_from_reg_id = reg_unicode and
            (:list_kind = 'full' or (:list_kind = 'execute' and cmp_status in (0, 1)))
      order by cmp_date desc, cmp_id desc
      into :cmp_id,
           :cmp_str_no,
           :cmp_date,
           :cmp_is_ipf,
           :cmp_memo,
           :cmp_from_reg_id,
           :cmp_status,
           :cmp_region_name,
           :cmp_from_cmk_id,
           :cmp_from_cmd_id
  do
  begin
    cmp_kind = '';
    cmp_decision = '';
    cmp_from_bln_id = null;

    if (cmp_from_cmk_id is not null) then
    begin
      select cmk_code, cmk_from_bln_id
      from com_kind
      where cmk_id = :cmp_from_cmk_id
      into cmp_kind, cmp_from_bln_id;
    end

    if (cmp_from_cmd_id is not null) then
    begin
      select cmd_name
      from com_decision
      where cmd_id = :cmp_from_cmd_id
      into :cmp_decision;
    end

    count_total = 0;
    count_approved = 0;
    count_done = 0;

    cmp_status_str = '';
    if (cmp_status = 0) then
      cmp_status_str = 'В процессе';
    if (cmp_status = 1) then
      cmp_status_str = 'Завершен';

    select count(*),
           sum(iif(cml_last_status in (:enum_approved, :enum_done), 1, 0)),
           sum(iif(cml_last_status = :enum_done, 1, 0))
    from com_list
    where cml_last_cmp_id = :cmp_id
    into count_total, count_approved, count_done;

    if (count_total is null) then
      count_total = 0;
    if (count_approved is null) then
      count_approved = 0;
    if (count_done is null) then
      count_done = 0;

    need_suspend = 1;
    if (only_actual = 1 and
        cmp_status = 1 and
        count_approved = count_done)
    then
      need_suspend = 0;

    if (need_suspend = 1) then
      suspend;
  end
end
