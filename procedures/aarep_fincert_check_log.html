CREATE PROCEDURE AAREP_FINCERT_CHECK_LOG(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_C_ID D_ID)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    C_BIRTH_DATE D_DATE,
    ENTITY_NUM D_STR_255,
    BP_CHECK_KIND D_STR_CODE,
    CHECK_DATETIME D_DATETIME,
    RES D_BOOL,
    RES_FID D_STR_CODE,
    REG_NAME D_STR_255)
AS
declare function f_get_entity_num(object_kind d_integer, object_id d_id)
returns d_str_255
as
  declare variable res d_str_255;
begin
  res = f_format('(Unknown kind %1) %2', object_kind, object_id);

  if (object_kind = 11) then
  begin
    select ma_no
    from miniapp ma
    where ma_id = :object_id
    rows 1
    into res;

    select :res || ' / ' || k_kontract_no
    from kontract
    where k_from_ma_app_id = :object_id
    order by k_status desc, k_id desc
    rows 1
    into res;
  end

  return res;
end 

declare procedure hp_ma_list (reg_id d_integer, rep_date_begin d_date, rep_date_end d_date, rep_c_id d_id)
returns (rep_ma_id d_id)
as
begin
  if (rep_c_id is not null) then
    for select mac_from_ma_id
    from miniappclient
    where mac_from_c_id = :rep_c_id
    into rep_ma_id
    do
      suspend;
  else
    for select ma_id
    from miniapp, region_list(:reg_id, 1, -1, 1)
    where ma_office_id = id_list
      and ma_number = 0
      and ma_date_app between :rep_date_begin and :rep_date_end
    into rep_ma_id
    do
      suspend;
end

declare variable fcl_id d_id;
begin
  for select f_get_entity_num(fcl_object_kind, fcl_object_id),
      min(fcl_datetime),
      min(fcl_id)
  from hp_ma_list(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_c_id), fincert_check_log
  where fcl_object_kind = 11
    and fcl_object_id = rep_ma_id
  group by 1
  into entity_num, check_datetime, fcl_id
  do
  begin
    for select c_id,
        c_name_fio,
        c_birth_date,
        iif(fcl_fid is null, 0, 1),
        f_enum('bp_check_kind', fcl_tag, 'name'),
        iif(fcl_fid is null, '-', f_enum('fincert_file', fcl_fid, 'name')),
        reg_code || ' ' || reg_name
    from fincert_check_log, miniapp, miniappclient, clients, region
    where fcl_id = :fcl_id
      and ma_date_end = '31.12.9999'
      and fcl_object_kind = 11
      and fcl_object_id = ma_id
      and mac_from_ma_id = ma_id
      and c_id = mac_from_c_id
      and reg_unicode = ma_office_id
    into c_id, c_name_fio, c_birth_date, res, bp_check_kind, res_fid, reg_name
    do
      suspend;

  end 
end
