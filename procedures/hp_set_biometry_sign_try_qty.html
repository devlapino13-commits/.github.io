CREATE PROCEDURE HP_SET_BIOMETRY_SIGN_TRY_QTY(
    BIOMETRY_SIGN_TRY_VALUE D_INTEGER)
RETURNS (
    RES_TEXT_INFO D_STR_255,
    RES_TEXT_ERROR D_STR_255)
AS
begin
  -- RFCKG-3757 Отключение СМС при подписании Эл документов в кредитном/ИПФ процессе.
  res_text_info = '';
  res_text_error = '';

  if (biometry_sign_try_value <= 0) then
    res_text_error = 'Количество попыток должно быть больше 1';

  if (res_text_error = '') then
  begin
    update const_var cv
       set cv.cv_value = cast(:biometry_sign_try_value as d_str_code)
    where cv.cv_code = 'biometry_sign_try_qty';

    res_text_info = f_format('Количество попыток установлено на %1', biometry_sign_try_value);
  end

  suspend;
end
