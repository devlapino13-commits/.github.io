CREATE PROCEDURE AAREP_LLC_CALCED_PERC_CLI(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    AAREP_PERC_BEGIN D_CURRENCY,
    AAREP_PERC_CALCED D_CURRENCY,
    AAREP_PERC_PAYED D_CURRENCY,
    AAREP_PERC_WRITEOFF D_CURRENCY,
    AAREP_PERC_END D_CURRENCY,
    AAREP_BALANCE_END D_CURRENCY,
    AAREP_KONTRACT_NO D_STR_20,
    AAREP_IS_INDIV D_STR_10,
    AAREP_PRODUCT_NAME D_STR_CODE,
    AAREP_PRODUCT_KIND D_STR_20,
    REP_CURRENCY D_STR_LANG,
    OFFICE_ID D_INTEGER,
    OFFICE_NAME VARCHAR(60),
    BRANCH_NAME VARCHAR(60),
    REP_DELIQ_DAYS_KOLICH D_INTEGER,
    LLC_SUM_PERC D_CURRENCY,
    REG_STAT_REP_REGION D_STR_100,
    C_NAME_FIO D_STR_100,
    BS_NAME D_STR_CODE,
    BT_NAME D_STR_CODE,
    C_DOLYA D_PERCENT,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_STOP_PERC D_DATE,
    K_KREDLINE_END D_DATE,
    LKC_BALANCE D_CURRENCY,
    LKC_PERCENT D_PERCENT,
    LKC_STATUS D_SMALLINT,
    RESERVE_NAME D_STR_20,
    AAREP_DISB_DATE D_DATE,
    IS_RESTRUCT D_INTEGER,
    AAREP_DELAYED_PERC_BEGIN D_CURRENCY,
    AAREP_DELAYED_PERC_END D_CURRENCY,
    IS_MBH BOOLEAN,
    C_INN D_STR_20)
AS
declare variable def_curr d_str_lang;
declare variable aarep_k_id d_id;
declare variable c_id d_id;
declare variable k_index_curr_rate d_currency;
declare variable cr_rate d_currency;
declare variable kc_busuness_subtype d_integer;
declare variable aarep_perc_begin1 d_currency;
declare variable aarep_perc_calced1 d_currency;
declare variable aarep_perc_payed1 d_currency;
declare variable aarep_perc_writeoff1 d_currency;
declare variable aarep_perc_end1 d_currency;
declare variable aarep_balance_end1 d_currency;
declare variable llc_sum_perc1 d_currency;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  select cm_default_currency
  from company cm
  into def_curr;

  cr_rate = null;
  select cr_rate
  from opercurr
  where :rep_date_end between cr_date and cr_date_end
    and cr_from_cu_id = 840
  into :cr_rate;
  if (cr_rate is null) then
    cr_rate = 1;

  for
    select
      aarep_k_id,
      aarep_kontract_no,
      aarep_is_indiv,
      aarep_product_name,
      aarep_product_kind,
      rep_currency,
      k_stop_perc,
      aarep_disb_date,
      is_mbh,
      sum(aarep_perc_begin),
      sum(aarep_perc_calced),
      sum(aarep_perc_payed),
      sum(aarep_perc_writeoff),
      sum(aarep_perc_end),
      sum(aarep_balance_end),
      coalesce(sum(aarep_delayed_perc_begin), 0),
      coalesce(sum(aarep_delayed_perc_end), 0)
    from aarep_calc_perc_list(:rep_region_id, :rep_date_begin, :rep_date_end, '---')
    group by
      aarep_k_id,
      aarep_kontract_no,
      aarep_is_indiv,
      aarep_product_name,
      aarep_product_kind,
      rep_currency,
      k_stop_perc,
      aarep_disb_date,
      is_mbh
  into
    :aarep_k_id,
    :aarep_kontract_no,
    :aarep_is_indiv,
    :aarep_product_name,
    :aarep_product_kind,
    :rep_currency,
    :k_stop_perc,
    :aarep_disb_date,
    :is_mbh,
    :aarep_perc_begin,
    :aarep_perc_calced,
    :aarep_perc_payed,
    :aarep_perc_writeoff,
    :aarep_perc_end,
    :aarep_balance_end,
    :aarep_delayed_perc_begin,
    :aarep_delayed_perc_end
  do
  begin
    k_index_curr_rate = null;
    office_name = null;
    branch_name = null;
    k_kredline_end = null;

    select
      k_index_curr_rate,
      r1.reg_unicode,
      r1.reg_code || ' ' || r1.reg_name,
      r2.reg_code || ' ' || r2.reg_name,
      r1.reg_stat_rep_region,
      k_amount_approved,
      k_kredline_end
    from kontract
      inner join knaktlist
        on kal_from_kf_id = k_from_kf_id
          and :rep_date_end between kal_date and kal_date_end
          and is_delete = 0
      inner join region r1
        on r1.reg_unicode = kal_new_reg_id
      inner join region r2
        on r2.reg_unicode = r1.parent
    where k_id = :aarep_k_id
    into
      :k_index_curr_rate,
      :office_id,
      :office_name,
      :branch_name,
      :reg_stat_rep_region,
      :k_amount_approved,
      :k_kredline_end;

    k_index_curr_rate = coalesce(k_index_curr_rate, 1);
    rep_deliq_days_kolich = coalesce((
      select rep_delinq_days
      from aarep_kont_delinq_status(:aarep_k_id, :rep_date_end)), 0);

    if (rep_currency <> def_curr) then
      if (k_index_curr_rate > cr_rate) then
      begin
        aarep_perc_end = aarep_perc_end * k_index_curr_rate;
        aarep_balance_end = aarep_balance_end * k_index_curr_rate;
      end
      else
      begin
        aarep_perc_end = aarep_perc_end * cr_rate;
        aarep_balance_end = aarep_balance_end * cr_rate;
      end

    llc_sum_perc = 0;
    if (rep_deliq_days_kolich >= 90 or coalesce(k_stop_perc, '31.12.9999') <= :rep_date_end) then
    begin
      llc_sum_perc = aarep_perc_end;
    end
    aarep_perc_begin1 = aarep_perc_begin;
    aarep_perc_calced1 = aarep_perc_calced;
    aarep_perc_payed1 = aarep_perc_payed;
    aarep_perc_writeoff1 = aarep_perc_writeoff;
    aarep_perc_end1 = aarep_perc_end;
    aarep_balance_end1 = aarep_balance_end;
    llc_sum_perc1 = llc_sum_perc;
    lkc_balance = 0;

    -- добавил igusev 03.11.2018 Основание: Письмот от Эржана. Тема: Внести столбец -->
    if (db_id in (39801, 41701)) then
      is_restruct = (
        select is_restruct
        from aarep_get_restructuring(:aarep_k_id, :rep_date_end, :db_id));
    --
    for
      select
        c_name_fio,
        cast(kc_amount_approved / :k_amount_approved as double precision),
        kc_busuness_subtype,
        c_inn,
        c_id
      from knclient
        inner join clients
          on c_id = kc_from_c_id
      where kc_from_k_id = :aarep_k_id
        and kc_amount_approved > 0
    into
      :c_name_fio,
      :c_dolya,
      :kc_busuness_subtype,
      :c_inn,
      :c_id
    do
    begin
      aarep_perc_begin = aarep_perc_begin1 * c_dolya;
      aarep_perc_calced = aarep_perc_calced1 * c_dolya;
      aarep_perc_payed = aarep_perc_payed1 * c_dolya;
      aarep_perc_writeoff = aarep_perc_writeoff1 * c_dolya;
      aarep_perc_end = aarep_perc_end1 * c_dolya;
      aarep_balance_end = aarep_balance_end1 * c_dolya;
      llc_sum_perc = llc_sum_perc1 * c_dolya;

      bs_name = null;
      bt_name = null;
      select
        bs_name,
        bt_name
      from bussubtype
        inner join bustype
          on bt_id = bs_from_bt_id
      where bs_id = :kc_busuness_subtype
      into
        :bs_name,
        :bt_name;
      if (bs_name is null) then
        bs_name = 'Не указан';
      if (bt_name is null) then
        bt_name = 'Не указан';

      lkc_balance = 0;
      lkc_percent = 0;
      select
        lkc_balance,
        lkc_percent,
        lkc_status
      from llc_kont_cli
        inner join llc_main
          on lm_id = lkc_from_lm_id
            and lm_date_status between :rep_date_end - 4 and :rep_date_end + 4
            and lm_is_signed = 1
      where lkc_from_k_id = :aarep_k_id
        and lkc_from_c_id = :c_id
      into
        :lkc_balance,
        :lkc_percent,
        :lkc_status;

      if (rep_date_end >= '30.06.2023') then
      begin
        if (aarep_perc_end > aarep_delayed_perc_end) then
        begin
          if (rep_deliq_days_kolich >= 90) then
            lkc_percent = 100;
          llc_sum_perc = (aarep_perc_end - aarep_delayed_perc_end) * lkc_percent / 100;
        end else
        begin
          lkc_percent = 0;
          llc_sum_perc = 0;
        end
      end
      reserve_name = iif(lkc_status <= 4, 'Общий резерв', 'Специальный резерв');

      if (db_id in (39801, 41701) and k_stop_perc <= rep_date_end) then
      begin
        lkc_percent = 100;
        llc_sum_perc = aarep_perc_end - aarep_delayed_perc_end;
        if (llc_sum_perc < 0) then llc_sum_perc = 0;
      end

      suspend;
    end
  end
end
