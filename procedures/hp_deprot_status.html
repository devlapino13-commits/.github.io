CREATE PROCEDURE HP_DEPROT_STATUS(
    C_ID D_ID,
    RED_DATE D_DATE = current_date)
RETURNS (
    STATUS_FULL D_STR_255,
    STATUS D_SMALLINT,
    IS_HISTORY D_SMALLINT)
AS
declare variable deport_reason d_integer;
declare variable deport_start_date d_date;
declare variable deport_finish_date d_date;
declare variable deport_comment d_str_long;
declare variable doc_code d_str_100;
declare variable cds_id d_id;
begin
-- RFCRU-5875 вернет Статус нахождения клиента в РФ
-- status = 1 - подтверждено
-- status = 0 - не подтверждено

-- status_full = Причина с чч.мм.гг по чч.мм.гг (документами подтверждено/ не подтверждено)

  deport_start_date = current_date;
  deport_finish_date = current_date;
  deport_comment = '';
  status_full = '';
  status = 0;
  is_history = 0;

  if (exists( select *
              from deport_status_list dsl
              where dsl.dsl_from_c_id = :c_id and
                    dsl.dsl_record_status = 0)) then
  is_history = 1;

  select dsl.dsl_deport_reason,
         dsl.dsl_deport_start_date,
         dsl.dsl_deport_finish_date,
         dsl.dsl_deport_comment
  from deport_status_list dsl
  where dsl.dsl_from_c_id = :c_id and
          dsl.dsl_record_status = 2
  into deport_reason,
       deport_start_date,
       deport_finish_date,
       deport_comment;

  if (coalesce(deport_reason, -1) <> -1) then
  begin
    doc_code = (case f_enum_code('deport_reason', :deport_reason)
                  when 'deport' then 'deport_document'
                  when 'rkl' then 'depart_info_rkl'
                  when 'license' then 'depart_info_patent'
                  when 'other' then 'border_stamp'
                  else ''
                end);

    select cd.cds_id
    from clientdocs cd
    inner join passkind pk on pk.pk_id = cd.cds_from_pk_id
    inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
    where cd.cds_from_c_id = :c_id
      and pdt.pdt_code = :doc_code
      and cd.cds_is_delete = 0
      and cd.cds_doc_date_begin = :deport_start_date
      and cds_doc_date_end = :deport_finish_date
    order by cd.cds_doc_date_begin desc
    rows 1
    into :cds_id;

    if (coalesce(cds_id, 0) <> 0) then
      status = 1;

    status_full = f_format('%1 с %2 по %3 (документами %4)',
                                     f_enum_name('deport_reason', :deport_reason),
                                     f_datetostr(:deport_start_date),
                                     f_datetostr(:deport_finish_date),
                                     iif(status = 1, 'подтверждено', 'не подтверждено'));

  end

  suspend;
end
