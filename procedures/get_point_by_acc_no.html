CREATE PROCEDURE GET_POINT_BY_ACC_NO(
    C_BIRTH_DATE D_DATE NOT NULL,
    CL_UNIQ_ACC D_INTEGER NOT NULL,
    LANG D_STR_CODE = 'ru')
RETURNS (
    SCORE D_INTEGER NOT NULL,
    ERR_TEXT D_STR_100 NOT NULL)
AS
declare variable C_ID D_ID;
begin
    score = 0;
    err_text = '';

    for select c_id
    from hp_check_clientuniq_acc_no(:c_birth_date, :cl_uniq_acc)
    into c_id
    do
    begin
        if (c_id = -1) then
        begin
            err_text = f_lang(:lang, 'Клиент не найден');
            suspend;
            exit;
        end
        
        score = f_point(:c_id, current_date);
        suspend;
    end
end
