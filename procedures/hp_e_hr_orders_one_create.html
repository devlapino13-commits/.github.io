CREATE PROCEDURE HP_E_HR_ORDERS_ONE_CREATE(
    IN_OFFICE_ID D_INTEGER,
    IN_ORDER_TYPE D_INTEGER,
    IN_C_ID D_ID,
    IN_HRA_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    HO_ID D_ID,
    HO_DATE D_DATE,
    HO_FROM_REG_ID D_INTEGER,
    HO_TEXT D_STR_LONG,
    HO_REASON D_STR_255,
    HO_REASON_DATE D_DATE,
    HO_FROM_UN_ID D_ID,
    HO_ORDER_TYPE D_INTEGER,
    HO_ORDER_SUBTYPE D_INTEGER,
    HO_ORDER_NO D_STR_20,
    HO_STATUS D_INTEGER,
    HO_STATUS_USER_ID D_ID,
    HO_SIGNER_ID D_INTEGER,
    HO_ACCOUNTANT_ID ID_TYPE,
    HO_HELP_AMOUNT D_INTEGER,
    HO_CERT_MARRIAGE D_BOOL,
    HO_CERT_BIRTH_CHILD D_BOOL,
    HO_CERT_BIRTH_EMP D_BOOL,
    HO_CERT_DEATH D_BOOL,
    RES_CAN_EDIT D_BOOL,
    RES_IS_SIGNED D_BOOL,
    HRV_NUMBER D_INTEGER,
    HRV_FROM_C_ID ID_TYPE,
    HRV_DATE_BEGIN D_DATE,
    HRV_DATE_END D_DATE,
    HRV_DAYS D_INTEGER,
    HRV_DAY_BACK D_DATE,
    HRV_DIRECTION D_STR_100,
    HRV_STATUS D_INTEGER,
    HRV_CHANGER_ID D_ID,
    HRV_NOT_CLOSE_ACCESS D_BOOL,
    RES_ACTUAL_OFFICE D_STR_100,
    RES_ACTUAL_JOBLIST D_STR_255,
    RES_ACTUAL_REG_ID D_INTEGER,
    RES_HRA_ID D_INTEGER)
AS
declare variable last_ho_id d_id;
declare variable hra_date_begin d_date;
declare variable hra_date_end d_date;
declare variable hra_is_approved d_smallint;
declare variable hra_from_ho_id d_id;
declare variable head_office d_integer;
begin
  res_can_edit = 1;
  res_is_signed = 0;

  res_actual_office = '';
  res_actual_joblist = '';
  ho_order_no = '';
  ho_status = 0;
  hrv_days = 1;

  ho_id = 0;

  ho_date = current_date;
  res_hra_id = in_hra_id;

  ho_order_type = in_order_type;
  ho_order_subtype = 0;
  ho_help_amount = 0;
  ho_cert_marriage = 0;
  ho_cert_birth_child = 0;
  ho_cert_birth_emp = 0;
  ho_cert_death = 0;
  ho_accountant_id = 0;

  if (in_hra_id is not null) then
  begin
    select hra_from_c_id, hra_from_reg_id, hra_order_type, hra_date_begin, hra_date_end, hra_is_approved, hra_from_ho_id,
           hra_changer_id
    from hr_applicat
    where hra_id = :in_hra_id
    into :in_c_id, :ho_from_reg_id, :ho_order_type, hra_date_begin, hra_date_end, hra_is_approved, hra_from_ho_id,
         hrv_changer_id;

    if (hra_is_approved = 0) then
      exception e_message 'Предварительное заявление еще не утверждено';

    if (hra_from_ho_id is not null) then
      exception e_message 'По данной заявке уже оформлен приказ';
  end

  select ho_id, 
         coalesce(:ho_from_reg_id, ho_from_reg_id),
         ho_text,
         ho_reason,
         ho_reason_date,
         ho_order_type,
         ho_order_subtype
  from hr_orders, sa_worktype
  where ho_from_un_id = :un_id and
        ho_order_type = coalesce(:ho_order_type, ho_order_type) and
        ho_order_type = sawt_id and
        sawt_is_order = 0
  order by ho_date desc, ho_id desc
  rows 1
  into last_ho_id,
       ho_from_reg_id,
       ho_text,
       ho_reason,
       ho_reason_date,
       ho_order_type,
       ho_order_subtype;

  if (row_count = 0 and ho_from_reg_id is null) then
  begin
    if (exists(select * from region
               where reg_unicode = :in_office_id and
                     reg_status = 3)) then
      ho_from_reg_id = in_office_id;
  end

  ho_from_un_id = un_id;

  if (ho_signer_id is null) then
  begin
    head_office = f_hq_id();

    if (ho_from_reg_id = head_office) then
    begin
      select ho_signer_id
      from hr_orders
      where ho_signer_id is not null and
            ho_from_reg_id = :ho_from_reg_id
      order by ho_id desc
      rows 1
      into ho_signer_id;
    end else
    begin
      select ho_signer_id
      from hr_orders
      where ho_signer_id is not null and
            ho_from_reg_id <> :ho_from_reg_id
      order by ho_id desc
      rows 1
      into ho_signer_id;
    end
  end

  hrv_number = 0;
  hrv_date_begin = coalesce(hra_date_begin, current_date);

  if (in_c_id is not null) then
    hrv_from_c_id = in_c_id;

  if (hrv_from_c_id is not null) then
  begin
    select res_actual_reg_id, res_actual_office, res_actual_joblist
    from hp_e_hr_orders_pers_info(:hrv_from_c_id, :hrv_date_begin)
    into :res_actual_reg_id, :res_actual_office, :res_actual_joblist;
  end

  if (res_actual_reg_id is null) then
    res_actual_reg_id = ho_from_reg_id;

  if (hra_date_end is null) then
  begin
    if (last_ho_id is not null) then
    begin
      select hrv_not_close_access, hrv_date_begin, hrv_days
      from hr_voctravel
      where hrv_from_ho_id = :last_ho_id
      rows 1
      into hrv_not_close_access, hrv_date_begin, hrv_days;
  
      if (hrv_date_begin < current_date) then
        hrv_date_begin = current_date;
  
      if (coalesce(hrv_days, 0) = 0) then
        hrv_days = 1;
    end
  
    select date_end, next_work_date
    from hp_hr_next_date(:hrv_date_begin, :hrv_days, :ho_order_type, :res_actual_reg_id)
    into :hrv_date_end, :hrv_day_back;
  end else
  begin
    hrv_date_end = hra_date_end;
    select diff_days, day_back
    from hp_hr_diff_date(:hra_date_begin, :hra_date_end, :ho_order_type, :res_actual_reg_id)
    into :hrv_days, :hrv_day_back;
  end

  if (hrv_not_close_access is null) then
    hrv_not_close_access = 1;

  if (ho_order_type = 25) then -- Отстранение от работы
    ho_reason = 'Кызматтык иликт�©�©нун жыйынтыгы боюнча корутунду / Заключение по результатам служебного расследования'; -- RFCKG-4620

  suspend;
end
