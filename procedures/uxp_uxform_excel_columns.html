Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXFORM_EXCEL_COLUMNS(
    UFM_UUID D_UUID,
    ALL_FIELDS D_BOOL = 0)
RETURNS (
    FR_ID D_INTEGER,
    FR_UUID_STR D_STR_CODE,
    FR_MASTER_STR D_STR_CODE,
    FR_IS_GRID D_BOOL,
    FR_CAPTION D_STR_255,
    UF_IS_PARAM D_BOOL,
    UF_FIELD_NAME D_STR_CODE,
    UF_DATA_TYPE D_STR_CODE,
    UF_IN_UI D_BOOL,
    UGI_CAPTION D_STR_100,
    UGI_POSITION D_INTEGER,
    UGI_WIDTH D_INTEGER,
    UGI_GROUP_INDEX D_INTEGER,
    UGI_SUMMARY_KIND D_INTEGER,
    UGI_HINT D_STR_LONG)
AS
begin
  for select fr_id, fr_uuid_str, fr_master_str, fr_is_grid, fr_caption, uf_is_param, uf_field_name, uf_data_type,
             uf_in_ui, ugi_caption, ugi_position, ugi_width, ugi_group_index, ugi_summary_kind, ugi_hint
  from uxp$uxform_excel_fields(:ufm_uuid, :all_fields)
  order by fr_id, uf_is_param desc, uf_in_ui desc, ugi_position
  into :fr_id, :fr_uuid_str, :fr_master_str, :fr_is_grid, :fr_caption, :uf_is_param, :uf_field_name,
       :uf_data_type, :uf_in_ui, :ugi_caption, :ugi_position, :ugi_width, :ugi_group_index, :ugi_summary_kind, :ugi_hint
  do
    suspend;
end
