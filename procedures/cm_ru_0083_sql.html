CREATE PROCEDURE CM_RU_0083_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable MIN_SUM CURR_TYPE;
declare variable MAX_SUM CURR_TYPE;
declare variable K_KOLICH_REPAY integer;
declare variable PR_CODE varchar(100);
declare variable KC_FROM_C_ID ID_TYPE;
declare variable K_LAST_REG_ID ID_TYPE;
declare variable K_KOMMIS_PERC PERC_TYPE;
declare variable KOL_CLI_GROUP integer;
declare variable PR_ID integer;
declare variable K_PERCENT PERC_TYPE;
declare variable C_BIRTH_DATE date;
declare variable C_CITIZEN integer;
declare variable MIGR_DATE date;
declare variable CON_RF integer;
declare variable CON_ROD integer;
declare variable PATENT_ACTIVE integer;
declare variable PATENT_DATE date;
declare variable PATENT_DATE_END date;
begin
  for select K_AMOUNT_APPROVED, K_KOLICH_REPAY, PR_CODE, K_DISB_DATE, K_KOMMIS_PERC, K_KOLICH_CLIENT_APPROVED, PR_ID,
             K_PERCENT, K_LAST_REG_ID
      from KONTRACT, PRODUCT
      where K_FROM_PR_ID = PR_ID and
            K_ID = :K_ID and
            PR_CODE containing '.IND.' and
            PR_CODE containing '.PATENT PLUS.' and --Микрозаем на патент плюс
            PR_CODE not containing '.PAR.'
      into KC_AMOUNT, K_KOLICH_REPAY, PR_CODE, KF_DISB_DATE, K_KOMMIS_PERC, KOL_CLI_GROUP, PR_ID, K_PERCENT,
           K_LAST_REG_ID
  do
  begin
    --Проверка офиса, выдается только в нескольких офисах
    if (K_LAST_REG_ID not in (1460100, 1400200, 1500100, 1120300, iif(KF_DISB_DATE >= '21.05.18', 1130200, 0))) then
    begin
      res_error_text = 'Выдача займа по данному продукту запрешается  в указанном офисе!';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
    --Проверка по паспорту продукта, пункт 1.1.
    if (KOL_CLI_GROUP <> 1) then
    begin
      --'1.1. Продукт является индивидуальным микрозаймом!';
      res_error_text = 'Продукт является индивидуальным микрозаймом!';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
    select KC_FROM_C_ID, C_BIRTH_DATE, C_CITIZEN, C_NAME_FIO
    from KNCLIENT KC, CLIENTS
    where KC_AMOUNT_APPROVED > 0 and
          KC_FROM_K_ID = :K_ID and
          KC_FROM_C_ID = C_ID
    into :KC_FROM_C_ID, C_BIRTH_DATE, C_CITIZEN, :res_info_text;
    MIN_SUM = 7700; --Минимальная по продукту
    MAX_SUM = 30000; --Максимальная по продукту
    if (K_LAST_REG_ID = 1400200) then --4002 ВОРОНЕЖ
    begin
      MIN_SUM = 9800;
      MAX_SUM = 14700;
    end
    if (K_LAST_REG_ID = 1500100) then --5001 Ульяновск
    begin
      --'Сумма микрозайма в офисе Ульяновск от 7 700 до 16 008 руб.!';
      MIN_SUM = 7700;
      MAX_SUM = 16008;
    end
    if (K_LAST_REG_ID = 1460100) then --4601 Владимир
    begin
      --'Минимальная сумма займа по продукту «Микрозаем на патент плюс» для офиса Владимир = 12200!';
      MIN_SUM = iif(:kf_disb_date<'08.06.18', 12200, 10700);
      MAX_SUM = 30000;
    end
    if (K_LAST_REG_ID = 1130200) then  --ОП 13-02 Омск
    begin
      --'Приказ 145-05-18 от15.05.18 Микрозаем на патент Плюс в ОП 13-02 Омск.Сумма микрозайма должен быть от 10000 до 20000руб.';
      MIN_SUM = 10000;
      MAX_SUM = 20000;
    end
    if (K_LAST_REG_ID = 1120300) then --ОП № 12-03 г. Сургут
    begin
      --'Сумма микрозайма по продукту    12 750 – 17 650 рублей';
      MIN_SUM = 12750;
      MAX_SUM = 17650;
    end
    if (KC_AMOUNT < MIN_SUM) then
    begin
      res_error_text = 'Сумма микрозайма должен быть от: ' || :MIN_SUM || ' руб.!';
      res_error_is_critical = 1;
      suspend;
    end
    if (KC_AMOUNT > MAX_SUM) then
    begin
      res_error_text = 'Сумма микрозайма должен быть до: ' || :MAX_SUM || ' руб.!';
      res_error_is_critical = 1;
      suspend;
    end
    --Проверка по паспорту продукта, пункт 1.3.
    if (K_KOLICH_REPAY not between 3 and 9) then
    begin
      --'1.3. Срок микрозайма должен быть от 3 до 9 месяцев!';
      res_error_text = 'Срок микрозайма должен быть от 3 до 9 месяцев!';
      res_error_is_critical = 1;
      suspend;
    end
    --Проверка по паспорту продукта, пункт 1.6.
    if (exists(select distinct 1
               from SCHEDULE SCH
               where SCH.SH_FROM_K_ID = :K_ID and
                     SCH.SH_OS = 0 and
                     SCH.SH_ID > 1)) then
    begin
      --'1.6. Льготный период не соответствует условию паспорта продукта!';
      res_error_text = 'Клиент вправе воспользоваться льготным периодом в течение первого месяца займа!';
      res_error_is_critical = 1;
      suspend;
    end
    --Проверить по условиям паспорта продукта, пункты 1.7. и 1.8. невозможно!
    --Проверка по паспорту продукта, пункт 1.9.
    if (exists(select *
               from SCHEDULE SCH
               where SCH.SH_FROM_K_ID = :K_ID and
                     (SCH.SH_ID > 0 and
                     SCH.SH_OS + SH_PERC not between 2500 and 7000 and
                     SH_OS <> 0 or SH_OS = 0 and
                     SH_PERC > 7000))) then
    begin
      --'1.9.Сумма план.погашения не соответствует условию паспорта продукта!';
      res_error_text = 'Сумма план.погашения должна быть от 2500 р. до 7000 р.!';
      res_error_is_critical = 1;
      suspend;
    end
    --Проверка по паспорту продукта, пункт 2.1.
    if (C_BIRTH_DATE > dateadd(year, -18, :KF_DISB_DATE)) then
    begin
      --'2.1. Возраст клиента не позволяет получить кредит, смотрите пункт!';
      res_error_text = 'Возраст клиента не позволяет получить кредит (не достиг 18 лет)!';
      res_error_is_critical = 1;
      suspend;
    end
    --Проверка по паспорту продукта, пункт 2.1.
    if (C_CITIZEN not in (9, 12)) then
    begin
      --'2.1. Гражданство клиента не позволяет получить кредит, смотрите пункт!';
      res_error_text = 'Несоответствие гражданства клиента с условием продукта!';
      res_error_is_critical = 1;
      suspend;
    end
    select distinct CDS_DOC_DATE_BEGIN
    from CLIENTDOCS, PASSKIND, PASSDOCTYPE
    where CDS_FROM_PK_ID = PK_ID and
          PK_FROM_PDT_ID = PDT_ID and
          CDS_IS_DELETE = 0 and
          PDT_CODE = 'magration_card' and
          :KF_DISB_DATE between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END and
          CDS_FROM_C_ID = :KC_FROM_C_ID and
          coalesce(cds_doc_blob_1, 0) > 0
          rows 1
    into MIGR_DATE;
    if (MIGR_DATE is null) then
    begin
      --'2.1. Не добавлена миграционная карта!';
      res_error_text = 'Не добавлена миграционная карта!';
      res_error_is_critical = 1;
      suspend;
    end
    PATENT_DATE = null;
    PATENT_DATE_END = null;
    select distinct CDS_DOC_DATE_BEGIN, CDS_DOC_DATE_END
    from CLIENTDOCS, PASSKIND, PASSDOCTYPE
    where CDS_FROM_PK_ID = PK_ID and
          PK_FROM_PDT_ID = PDT_ID and
          CDS_IS_DELETE = 0 and
          PDT_CODE = 'work_patent' and
          CDS_FROM_C_ID = :KC_FROM_C_ID and
          coalesce(cds_doc_blob_1, 0) > 0
    order by CDS_DOC_DATE_BEGIN desc rows 1
    into PATENT_DATE, PATENT_DATE_END;
    if (PATENT_DATE is null) then
      PATENT_DATE = '01.01.01';
    if (PATENT_DATE_END is null) then
      PATENT_DATE_END = '01.01.01';
    if (PATENT_DATE > KF_DISB_DATE) then
    begin
      res_error_text = 'Дата выдачи патента больше даты выдачи кредита!';
      res_error_is_critical = 1;
      suspend;
    end
    --Пункт 2.3. Если займ берется на продление патента
    if (PATENT_DATE < KF_DISB_DATE and
        KF_DISB_DATE - MIGR_DATE > 20) then
    begin
      --пункт: 2.3.1
      if (KF_DISB_DATE > '05.03.18') then
      begin
        select count(*)
        from SA_CALENDAYS SC
        where SC.SCD_DATE between :KF_DISB_DATE and :PATENT_DATE_END and
              SC.SCD_WORK_HOURS > 0
        into PATENT_ACTIVE;
        if (coalesce(PATENT_ACTIVE, 0) < 15) then
        begin
          --'2.3.1. Срок окончание текущего патента менее одного месяца!';
          res_error_text = 'Срок окончание текущего патента менее одного месяца!';
          res_error_is_critical = 1;
          suspend;
        end
      end
      else
      if (dateadd(month, 1, KF_DISB_DATE) > PATENT_DATE_END) then
      begin
        --'2.3.1. Срок окончание текущего патента менее одного месяца!';
        res_error_text = 'Срок окончание текущего патента менее одного месяца!';
        res_error_is_critical = 1;
        suspend;
      end
      --пункт: 2.3.2 пока невозможно добавить на проверку(необходимо квитанции патента на последние 12 мес.)
      --пункт: 2.3.3
      if (not exists(select *
                     from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                     where CDS_FROM_C_ID = :KC_FROM_C_ID and
                           CDS_FROM_PK_ID = PK_ID and
                           PK_FROM_PDT_ID = PDT_ID and
                           CDS_IS_DELETE = 0 and
                           coalesce(cds_doc_blob_1, 0) > 0 and
                           PDT_CODE in ('certificate_rus_lang', -- Сертификат о знании русского языка
                                    'diplom_vuz', -- Диплом ВУЗ
                                    'diplom_suz', -- Диплом СУЗ
                                    'diplom_school' -- Аттестат
                                    ))) then

      begin
        res_error_text = 'Необходимо добавить один из необходимых документов(сертификат о знании рус.яз., аттестат или диплом)';
        res_error_is_critical = 1;
        suspend;
      end
    end
  end
end
