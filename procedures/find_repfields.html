CREATE PROCEDURE FIND_REPFIELDS(
    REP_ID INTEGER,
    TEXT VARCHAR(50))
RETURNS (
    RF_NUMBER INTEGER,
    RF_FIELD_NAME VARCHAR(50),
    RF_NAME_RUS VARCHAR(50),
    RF_HINT_RUS VARCHAR(256),
    RF_NAME_ENG VARCHAR(50),
    RF_HINT_ENG VARCHAR(256))
AS
begin
  text = upper(:text);
  for select rf_number, rf_field_name, rf_name_rus, rf_hint_rus, rf_name_eng, rf_hint_eng
      from repfields
      where rf_from_rl_id = :rep_id and
            ((upper(rf_field_name) like '%' || :text || '%') or
            (upper(rf_name_rus) like '%' || :text || '%') or
            (upper(rf_hint_rus) like '%' || :text || '%') or
            (upper(rf_name_eng) like '%' || :text || '%') or
            (upper(rf_hint_eng) like '%' || :text || '%'))
      into :rf_number, :rf_field_name, :rf_name_rus, :rf_hint_rus, :rf_name_eng, :rf_hint_eng
  do
  begin
    suspend;
  end
end
