CREATE PROCEDURE CM_KG_0089_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable husband_qty integer;
declare variable c_id id_type;
declare variable crel_other_c_id id_type;
declare variable pr_code d_str_100;
begin
  for select KC_FROM_C_ID, pr.pr_code
      from KNCLIENT kc
      inner join kontract kt on kt.k_id = kc.kc_from_k_id
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      where KC_FROM_K_ID = :K_ID and
            KC_AMOUNT_APPROVED > 0
      into C_ID, pr_code
  do
  begin
    if (exists(select count(CREL_FROM_C_ID)
               from CLIENTRELATION CL
               where :C_ID = CL.CREL_FROM_C_ID and
                     CREL_FROM_REL_ID in (1, 2) AND crel_is_active = 1 --
                     having count(CREL_FROM_C_ID) > 1)) then

    begin
      res_error_text = 'Супруг должен быть один. Аннулируйте дубликат!';
      res_error_is_critical = 2;
      suspend;
    end
  end
  for select zal.z_from_c_id
      from zalog zal
      where zal.z_from_k_id = :K_ID and
            zal.z_from_c_id is not null
      into C_ID
  do
  begin
    if (exists(select count(CREL_FROM_C_ID)
               from CLIENTRELATION CL
               where :C_ID = CL.CREL_FROM_C_ID and
                     CREL_FROM_REL_ID in (1, 2) AND crel_is_active = 1 --
                     having count(CREL_FROM_C_ID) > 1)) then

    begin
      res_error_text = '[Проблема у поручителя] Супруг должен быть один. Аннулируйте дубликат!';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
