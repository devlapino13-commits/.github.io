CREATE PROCEDURE FM_HTML_REPORT_SETTING_KMP(
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable lo_id decimal(18,0);
declare variable k_id decimal(18,0);
declare variable c_name_fio varchar(100);
declare variable c_id decimal(18,0);
declare variable db_id integer;
declare variable mcl_app_version d_smallint;
declare variable mcl_app_version_text varchar(50);
declare variable deliq_days d_str_20;
begin

  db_id = f_get_db_id();

  select un_from_c_id
  from username
  where un_id = :un_id
  into :lo_id; --id sotrudnika
  if (lo_id is null) then
    exit;

  res_text = (select res_style
              from fm_html_styles) || '<body>';
  res_text = BLOB_APPEND(res_text, f_format('<h1 class="green_text">%1</h1>', iif(db_id <> 94401, 'Список клиентов', 'Müştərilərin siyahısı')));
  res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="60%">
       <col width="20%">
       <col width="20%">
       <tr>' || f_format('<th>%1</th><th>%2</th><th>%3</th></tr>', iif(db_id <> 94401, 'Клиент', 'Müştəri'), iif(db_id <> 94401, ' Версия КМП', 'Versiya Nömrə'), iif(db_id <> 94401, 'Cтатус просрочки', 'Vaxtı keçmiş status')));

  for select oks_from_k_id,
             iif((cast(current_date as date) - oks_delinq_date <= 30), 'до 30 дней (вкл.)', 'более 30 дней')
      from oksummary
      where oks_kc_id = :lo_id and --id sotrudnika
            current_date between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_close_state = 0
      order by 2 asc
      into :k_id, deliq_days
  do
  begin
    for select distinct c_name_fio, c_id, f_fsm_client_status(kc_from_c_id)
        from knclient, clients
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = c_id
        into :c_name_fio, :c_id, :mcl_app_version
    do
    begin
      mcl_app_version_text = '';

      if (mcl_app_version = 0) then
        mcl_app_version_text = 'не установлен';

      else
      if (mcl_app_version = 1) then
        mcl_app_version_text = 'Требуется обновление';

      else
      if (mcl_app_version = 2) then
        mcl_app_version_text = 'установлен';

      res_text = blob_append(res_text, f_format('<tr><td>%1</td><td align="center">%2</td><td>%3</td></tr>',
        c_name_fio || ' ' || f_uri_phone(f_cont_info(0, c_id, 'cell', 'last'), 0, c_id) , mcl_app_version_text, deliq_days));
    end
  end

  res_text = blob_append(res_text, '</table></body>');
  suspend;
end
