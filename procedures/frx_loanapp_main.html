CREATE PROCEDURE FRX_LOANAPP_MAIN(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    MA_ID DECIMAL(18,0),
    FOLDER_NAME VARCHAR(100),
    PRODUCT_NAME VARCHAR(100),
    NO_LGOTA_1 INTEGER,
    NO_LGOTA_2 INTEGER,
    NO_LGOTA_ALL INTEGER,
    PERIOD INTEGER,
    REPAY_DAY VARCHAR(100),
    EDIT_DATETIME TIMESTAMP,
    DISB_DATE DATE,
    LO_FIO VARCHAR(100),
    LO_JL_NAME VARCHAR(100),
    REG_NAME VARCHAR(100),
    PRODUCT_COMMON_NAME VARCHAR(100),
    MA_NO INTEGER,
    CREATE_DATETIME TIMESTAMP,
    CREATE_DATE DATE,
    MA_PERCENT DECIMAL(18,7),
    MA_KOMMIS_PERC DECIMAL(18,7),
    FIRST_SH_DATE DATE,
    CLIENT_LIST VARCHAR(1000),
    MA_DATE_LAST_EDIT DATE,
    FIRST_PAY VARCHAR(10),
    JANUARY_PAY INTEGER,
    FEBRUARY_PAY INTEGER,
    IS_CASH INTEGER,
    IS_CASHK INTEGER,
    PR_CODE STR_100,
    K_KONTRACT_NO STR_20,
    MA_SUMMA_MINUS CURR_TYPE,
    MA_KOLICH_REPAY D_INTEGER,
    QR_CODE D_STR_CODE,
    HAS_LGOTA_PERIOD D_SMALLINT,
    EDOC_CREATE_UN_NAME D_STR_CODE,
    EDOC_CREATE_UN_FIO D_STR_100,
    EDOC_CREATE_DATETIME D_DATETIME,
    EDOC_CREATE_DATE D_DATE,
    EDOC_CREATE_UN_JOBLIST D_STR_100,
    MANAGER_FIO D_STR_255,
    MANAGER_JOBLIST D_STR_255,
    MA_SIGN_DATE D_DATE,
    APPR_UN_FIO D_STR_100,
    APPR_DATETIME D_DATETIME,
    APPR_STATUS D_STR_CODE,
    MANAGER_DOC_TYPE_STR D_STR_100,
    MANAGER_PASS_NO D_STR_CODE,
    MANAGER_PASS_DATE D_DATE,
    MANAGER_PASS_WHO D_STR_100,
    LO_DOC_TYPE_STR D_STR_100,
    LO_PASS_NO D_STR_CODE,
    LO_PASS_DATE D_DATE,
    LO_PASS_WHO D_STR_100,
    BANK_NAME D_STR_100,
    PHONE_NUMBER D_STR_20,
    EMAIL D_STR_CODE)
AS
declare variable SH_OBJ_ID decimal(18,0);
declare variable KF_ID decimal(18,0);
declare variable LO_ID decimal(18,0);
declare variable C_ID ID_TYPE;
declare variable C_NAME_FIO STR_100;
declare variable CLI_CELLNO STR_100;
declare variable K_ADD_MONTH D_INTEGER;
declare variable EDL_ID D_ID;
declare variable MA_MANAGER_ID D_ID;
declare variable UN_C_ID D_ID;
declare variable K_ID D_ID;
declare variable APPR_UN_ID D_ID;
declare variable DB_ID D_INTEGER;
declare variable MA_DISB_KIND integer;
declare variable EDS_STATUS D_SMALLINT;
declare variable EDS_STATUS_DATE D_DATE;
begin
  -- Процедура готовит данные для печати основной части кредитной заявки
  obj_kind = f_objkind(obj_kind, 'id');
  db_id = f_get_db_id();

  k_id = obj_id;
  sh_obj_id = obj_id;
  if (obj_kind = 9) then
  begin -- если передан кредит, то вычисляем заявку
    select
      k_from_ma_app_id,
      iif(coalesce(k_disb_kind, 1) = 1, 1, 0),
      k_kontract_no,
      iif(coalesce(k_kolich_lgota,0) = 0, 0, 1),
      k_add_month
    from kontract
    where k_id = :obj_id
    into
      :ma_id,
      :is_cashk,
      :k_kontract_no,
      :has_lgota_period,
      :k_add_month;
  end
  else
  begin
    ma_id = :obj_id;
    select
      k_id,
      k_kontract_no,
      iif(coalesce(k_kolich_lgota,0) = 0, 0, 1),
      k_add_month
    from kontract
    where k_from_ma_app_id = :ma_id
    order by k_disb_date desc
    rows 1
    into
      :k_id,
      :k_kontract_no,
      :has_lgota_period,
      :k_add_month;
    sh_obj_id = k_id;
  end
  has_lgota_period = iif(has_lgota_period = 1 or k_add_month > 0, 1, 0);

  select ma_from_kf_id, ma_kolich_repay, ma_repay_day, ma_date_oper, ma_date_save, pr_name, pr_common_name, ma_no,
         ma.ma_percent, ma.ma_kommis_perc, ma.ma_date_app, pr_code, ma_summa_minus, ma_kolich_repay, ma_manager_id
  from miniapp ma, product
  where ma_id = :ma_id and
        ma_date_end >= '31.12.2999' and
        ma_from_pr_id = pr_id
  into :kf_id, :period, :repay_day, disb_date, edit_datetime, product_name, product_common_name, :ma_no, :ma_percent,
       :ma_kommis_perc, :ma_date_last_edit, :pr_code, :ma_summa_minus, :ma_kolich_repay, ma_manager_id;

  select maa.ma_date_save, cast(maa.ma_date_save as date)
  from miniapp maa
  where maa.ma_id = :ma_id
  order by maa.ma_number
  rows 1
  into :create_datetime, :create_date;

  if (kf_id is null) then
    exit;

  manager_fio = '';
  manager_joblist = '';

  select c_name_fio
  from clients
  where c_id = :ma_manager_id
  into manager_fio;

  -- RFCKG-4226
  if (db_id in (39801, 41701)) then
  begin
      select c_pass_no, c_pass_date, c_pass_who, pk_name
      from clients
      inner join passkind on pk_id = c_doc_type
      where c_id = :ma_manager_id
      into manager_pass_no, manager_pass_date, manager_pass_who, manager_doc_type_str;
  end

  select jl_name
  from hr_pstatus, joblist
  where ps_from_c_id = :ma_manager_id and
        :create_date between ps_date_begin and ps_date_end and
        ps_is_working = 1 and
        ps_from_jl_id = jl_id
  rows 1
  into manager_joblist;

  if (repay_day = 0) then
    repay_day = extract(day from disb_date);
  if (repay_day = -1) then
    repay_day = 'Последний раб.день месяца';

  select kal_new_kf_name, reg_code || ' ' || reg_name, c_id, c_name_fio
  from knaktlist, region, clients
  where kal_from_kf_id = :kf_id and
        knaktlist.is_delete = 0 and
        kal_date <= :disb_date and
        kal_date_end >= :disb_date and
        kal_new_reg_id = reg_unicode and
        kal_new_kf_officer = c_id
  into :folder_name, :reg_name, :lo_id, :lo_fio;

  -- RFCKG-4226
  if (db_id in (39801, 41701)) then
  begin
      select c_pass_no, c_pass_date, c_pass_who, pk_name
      from clients
      inner join passkind on pk_id = c_doc_type
      where c_id = :lo_id
      into lo_pass_no, lo_pass_date, lo_pass_who, lo_doc_type_str;
  end

  select jl_name
  from joblist, hr_pstatus
  where ps_from_jl_id = jl_id and
        ps_from_c_id = :lo_id and
        ps_is_working = 1 and
        :create_date between ps_date_begin and ps_date_end
  into lo_jl_name;

  no_lgota_1 = 1;
  no_lgota_2 = 1;
  no_lgota_all = 1;

  if (not exists(select *
                 from schedule
                 where sh_from_k_id = :sh_obj_id)) then
  begin
    sh_obj_id = :ma_id * -1;
  end

  first_pay = 0;

  -- если первый платеж - льготный
  if (exists(select *
             from schedule
             where sh_from_k_id = :sh_obj_id and
                   sh_id = 1 and
                   sh_os = 0)) then
    first_pay = 1;

  if (exists(select *
             from schedule
             where sh_from_k_id = :sh_obj_id and
                   sh_id > 0 and
                   extract(month from sh_date) = 1 and
                   sh_os = 0)) then
    no_lgota_1 = 0;

  if (exists(select *
             from schedule
             where sh_from_k_id = :sh_obj_id and
                   sh_id > 0 and
                   extract(month from sh_date) = 2 and
                   sh_os = 0)) then
    no_lgota_2 = 0;

  if (exists(select *
             from schedule
             where sh_from_k_id = :sh_obj_id and
                   sh_id > 0 and
                   sh_os = 0)) then
    no_lgota_all = 0;

  select min(sh_date)
  from schedule
  where sh_from_k_id = :sh_obj_id and
        sh_id > 0
  into first_sh_date;

  ----------------------------
  client_list = '';

  select iif(coalesce(maa.ma_disb_kind, 1) = 1, 1, 0)
  from miniapp maa
  where maa.ma_id = :ma_id
  order by maa.ma_number desc
  rows 1
  into :is_cash;

  if (:db_id in (39801, 41701)) then
  begin
   select ma.ma_disb_kind
   from miniapp ma
   where ma.ma_id = :ma_id
   order by ma.ma_number desc
   rows 1
   into ma_disb_kind;
   :is_cash = case :ma_disb_kind
               when 1 then 1
               when 5 then 0
               when 6 then 2
              end;
   if (:ma_disb_kind not in (1,5,6)) then
        exception e_message 'Для бланка данный способ выдачи займа не допустим (disb_kind = '||:ma_disb_kind||')';
  end

  for select macl.mac_from_c_id, cl.c_name_fio
      from miniappclient macl
      inner join clients cl on cl.c_id = macl.mac_from_c_id
      where macl.mac_from_ma_id = :ma_id
      into :c_id, :c_name_fio
  do
  begin
    cli_cellno = null;

    select cont_value
    from hp_edoc_get_status('loan.application', 'miniapp', :ma_id, null)
    where c_id = :c_id
    into :cli_cellno;

    if (cli_cellno is null) then
      cli_cellno = 'верификация отсутсвует';

    client_list = iif(client_list = '', client_list, client_list || ', ') || c_name_fio || ' ' || cli_cellno;

    -- QR-код электронной подписи клиента. Текст пустой если не подписан
    qr_code = f_edoc_qr('loan.application', 'Miniapp', ma_id, c_id);

    --Определяем время подписания заявки
    select coalesce(cast(CLIENT_SIGN_DATETIME as date), current_date)
    from HP_EDOC_GET_STATUS('loan.application', 'miniapp', :MA_ID, null)
    where EDOC_STATUS = 1 and
          C_ID = :C_ID
    into :MA_SIGN_DATE;
    MA_SIGN_DATE = coalesce(MA_SIGN_DATE, current_date);

  end
  january_pay = 0;
  february_pay = 0;

  --RFCRU-5181 убрать ограничения по продуктам

    if (no_lgota_1 = 0) then
      january_pay = 1;
    if (no_lgota_2 = 0) then
      february_pay = 1;

  appr_un_fio = '';
  appr_datetime = null;
  appr_status = '';

  -- QR-код электронной подписи клиента. Текст пустой если не подписан
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);
    if (edl_id is not null) then
    begin
      select un_from_c_id, un_name, edl_create_datetime, cast(edl_create_datetime as date)
      from edoc_list, username
      where edl_id = :edl_id and
            edl_create_un_id = un_id
      into un_c_id, edoc_create_un_name, edoc_create_datetime, edoc_create_date;

      if (:pr_code containing '.express_online.' and :db_id in (39801, 41701)) then
      begin
        edoc_create_un_fio = '';
        select cl.c_name_fio
        from expressapp exa
        inner join clients cl on cl.c_id = exa.exa_lo_id
        where exa.exa_from_ma_id = :ma_id
        into edoc_create_un_fio;
      end
      else
      begin
        edoc_create_un_fio = '';
        edoc_create_un_joblist = '';
        select jl_name, c_name_fio
        from joblist, hr_pstatus, clients
        where ps_from_jl_id = jl_id and
            ps_from_c_id = c_id and
            ps_from_c_id = :un_c_id and
            ps_is_working = 1 and
            :create_date between ps_date_begin and ps_date_end
        into edoc_create_un_joblist, edoc_create_un_fio;
      end
    end
  end

  -- Блок для Анкеты ПОФТЭ, КГ
  if (db_id in (39801, 41701) and k_id is not null) then
  begin
    if (exists(select * from kontract where k_id = :k_id and k_status = 1)) then
    begin
      select kl_user_id, kl_datetime
      from kontlog
      where kl_from_k_id = :k_id and
            kl_state = 1
      order by kl_from_k_id desc, kl_id desc
      rows 1
      into appr_un_id, appr_datetime;

      if (row_count > 0) then
      begin
        if (pr_code containing '.express_online.') then
        begin
          appr_un_fio = manager_fio;
        end 
        else
        begin
          select c_name_fio
          from username, clients
          where un_id = :appr_un_id and
                un_from_c_id = c_id
          into appr_un_fio;
        end
        appr_status = 'ПРОВЕРЕНО';
      end
    end
  end

  if (db_id = 81001) then
  begin
    email = '';
    bank_name = '';
    phone_number = '';

    if (obj_kind = 9) then
    begin
      select ctl.clt_code, fpb.fpb_name
      from knclient kc
      inner join fps_loan_banks flb on flb.flb_from_kc_id = kc.kc_id
      inner join fps_client_banks fcb on flb.fbl_from_fcb_id = fcb.fcb_id
      inner join contlist ctl on fcb.fcb_from_clt_id = ctl.clt_id
      inner join fps_banks fpb on fcb.fcb_from_fpb_id = fpb.fpb_id
      where kc.kc_from_k_id = :obj_id
        and kc.kc_amount_approved > 0
      into phone_number, bank_name;

    end

    if (obj_kind = 11) then
    begin
      select ctl.clt_code, fpb.fpb_name
      from miniappclient mac
      inner join fps_application_banks fab on fab.fab_from_mac_id = mac.mac_id
      inner join fps_client_banks fcb on fab.fab_from_fcb_id = fcb.fcb_id
      inner join contlist ctl on fcb.fcb_from_clt_id = ctl.clt_id
      inner join fps_banks fpb on fcb.fcb_from_fpb_id = fpb.fpb_id
      where mac.mac_from_ma_id = :obj_id
      into phone_number, bank_name;

    end

    email = f_cont(:c_id, 'email', 'last');

  end

  suspend;
end
