CREATE PROCEDURE HP_LLC_CREATE_CLI(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    LM_ID ID_TYPE,
    LM_FROM_REG_ID INTEGER,
    AAREP_LLC_K_ID VARCHAR(20),
    LLC_K_ID DECIMAL(18,0),
    LLC_STATUS SMALLINT,
    LLC_PERCENT DECIMAL(18,7),
    LLC_NUMBER INTEGER,
    KC_FROM_K_ID ID_TYPE,
    KC_NUMBER INTEGER)
AS
declare variable kind integer;
declare variable delinq_kont id_type;
declare variable kk integer;
declare variable kc_has_loans smallint;
declare variable kc_has_family_loans smallint;
declare variable aarep_llc_status_kont smallint; /* Статус РППУ для контракта */
declare variable aarep_llc_percent_kont perc_type;
declare variable lkc_from_c_id id_type;
declare variable lkc_status smallint;
declare variable lkc_percent perc_type;
declare variable lkc_from_k_id id_type;
declare variable lkc_deliq_days integer;
declare variable lkc_is_indiv smallint;
declare variable lkc_from_reg_id integer;
declare variable lkc_from_lm_id id_type;
declare variable crel_other_c_id id_type;
declare variable suprug_name_fio varchar(100);
declare variable is_poruch smallint;
declare variable kk_str_poruch varchar(50);
declare variable is_suprug_kred smallint;
declare variable lkc_suprug_id id_type;
declare variable lkc_number integer;
declare variable lkc_suprug_memo varchar(255);
declare variable lkc_deliq_days_str varchar(255);
declare variable woff_sum_cli curr_type;
declare variable kc_closed_ol_id id_type;
declare variable kk_str varchar(255);
declare variable aarep_llc_percent decimal(18,7);
declare variable kc_balance decimal(18,4);
declare variable kc_from_c_id decimal(18,0);
declare variable aarep_llc_status_temp integer;
declare variable kol_rest integer;
declare variable aarep_memo varchar(255);
declare variable aarep_disb decimal(18,4);
declare variable aarep_llc_loan_type integer;
declare variable aarep_llc_k_balance decimal(18,4);
declare variable aarep_llc_k_deliq_days integer;
declare variable rep_year d_integer;
declare variable rep_month d_integer;
declare variable rep_mn d_integer;
declare variable rep_mnsv d_integer;
declare variable max_delinq_day integer;
declare variable lm_date d_date;
declare variable c_type d_bool;
declare variable lkc_mark boolean;
declare variable new_method boolean;
declare variable k_disb_date d_date;
declare variable k_percent d_currency;
declare variable lir_rate d_currency;
declare variable k_by_rating d_smallint;
begin
  -- Форс-мажор (пандемией «Коронавирус»)
  -- при расчете РППУ - не есть реструктуризация
  update refdata set rd_memo = '' where rd_id = 4075;

  -- Проставляет галочки в другом ФКУ
  rep_year = extract(year from :rep_date_end);
  rep_month = extract(month from :rep_date_end);
  rep_mn = 12 * (rep_year - 1) + rep_month;

  -- включительно до января 2017 в инфо КИБ было
  --   "30 дней до 59 дней просрочки"
  -- после
  --   "31 дней до 60 дней просрочки"
  rep_mnsv = 24193;

  -- у активных клиентов убирает галочки
  for select kc_from_k_id, kc_number
      from kontstate, knclient
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            kc_has_loans > 0 -- Клиент имеет кредит в другом ФКУ
      into kc_from_k_id, kc_number
  do
  begin
--    Обновляет поле kc_has_loans = 0 в таблице knclient
    update knclient
    set kc_has_loans = 0
    where kc_from_k_id = :kc_from_k_id and
          kc_number = :kc_number;
    suspend;
  end
--  и из таблицы kibtaken, активным клиентам проставляет галочки где на поле kc_has_loans
  /*
    for
    select distinct kc_from_k_id, kc_number
    from kontstate, knclient, kibtaken
    where ks_ol_is_delete = 0
    and ks_state = 1
    and :rep_date_end between ks_date_begin and ks_date_end
    and ks_from_k_id = kc_from_k_id
    and kc_amount_approved > 0
    and kt_from_c_id = kc_from_c_id
    and kt_year = :rep_year
    and kt_month = :rep_month
    and kc_has_loans = 0
    into kc_from_k_id, kc_number
    do
    begin
--    Обновляет поле kc_has_loans = 0 в таблице knclient
      update knclient set kc_has_loans = 1
      where kc_from_k_id = :kc_from_k_id
      and kc_number = :kc_number;
      suspend;
    end
    */
  for select kc_from_k_id, kc_number, kc_from_c_id
      from kontstate, knclient
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0
      into kc_from_k_id, kc_number, kc_from_c_id
  do
  begin
    if (exists(select *
               from kibtaken
               where kt_year = :rep_year and
                     kt_month = :rep_month and
                     (kt_from_c_id = :kc_from_c_id or
                     kt_from_c_id = (select c_original_id
                                     from clients
                                     where c_id = :kc_from_c_id and
                                           c_original_id > 0)))) then
    begin
--      Обновляет поле kc_has_loans = 0 в таблице knclient
      update knclient
      set kc_has_loans = 1
      where kc_from_k_id = :kc_from_k_id and
            kc_number = :kc_number;
      suspend;
    end
  end

-- далее начисление РППУ
  delinq_kont = 0;
  for select lm_id, lm_from_reg_id, lm_date
      from llc_main, region_list(:rep_region_id, 1, 3, 1)
      where lm_is_signed = 1 and
            lm_date_status between :rep_date_end - 4 and :rep_date_end + 4 and
            lm_from_reg_id = id_list
      into lm_id, lm_from_reg_id, lm_date
  do
  begin
    delete from llc_kont_cli
    where lkc_from_lm_id = :lm_id;
    for with tab1
        as (select aarep_k_id as aarep_llc_k_id, 0 as kind, -- Просрочкенные контракты вытаскивает
                   aarep_k_deliq_days as aarep_llc_k_deliq_days, k_amount_approved as aarep_disb,
                   aarep_k_balance as aarep_llc_k_balance,
                   iif(k_kolich_client_approved >= 3, 0, 1) as aarep_llc_loan_type,
                   k_disb_date, k_percent, k_by_rating
            from aarep_kontract_id(:lm_from_reg_id, :rep_date_end, :rep_date_end, 0, '000', 5), kontract
            where aarep_k_id = k_id
            union
            select ac_k_id as aarep_llc_k_id, 1 as kind, -- Все контракты вытаскивает
                   0 as aarep_llc_k_deliq_days, -- у всех контрактов просроченные делает 0
                   ac_disb as aarep_disb, ac_k_balance as aarep_llc_k_balance,
                   iif(ac_client_kolich >= 3, 0, 1) as aarep_llc_loan_type,
                   k_disb_date, k_percent, k_by_rating
            from aarep_kontract_active_temp(:lm_from_reg_id, :rep_date_end, '000', 0), kontract
            where ac_k_id = k_id
        )
        select
          aarep_llc_k_id,
          kind,
          aarep_llc_k_deliq_days,
          aarep_disb,
          aarep_llc_k_balance,
          aarep_llc_loan_type,
          k_disb_date,
          k_percent,
          coalesce(k_by_rating, 0)
        from tab1
        order by aarep_llc_k_id, kind
        into
          :aarep_llc_k_id,
          :kind,
          :aarep_llc_k_deliq_days,
          :aarep_disb,
          :aarep_llc_k_balance,
          :aarep_llc_loan_type,
          :k_disb_date,
          :k_percent,
          :k_by_rating
    do
    begin
      if (delinq_kont <> aarep_llc_k_id) then -- Если по контракту было просрочка то контракт без просрочки не будет брать
      begin
--        Добавлено от 06.05.2015, исключение клиентов у кого было списано доля или погашено
        woff_sum_cli = null; -- выданная сумма клиента у которого закрыта своя доля
        select sum(kc_amount_approved)
        from knclient, operlist
        where kc_from_k_id = :aarep_llc_k_id and
              kc_amount_approved > 0 and
              kc_closed_ol_id = ol_id and
              operlist.is_delete = 0 and
              ol_date_oper_day <= :rep_date_end
        into woff_sum_cli;
        if (woff_sum_cli is null) then
          woff_sum_cli = 0;
--        из выданной суммы исключаем долю клиента чтобы правильная пропорция было
        if (aarep_disb - woff_sum_cli <= 0) then
          woff_sum_cli = 0;
        aarep_disb = aarep_disb - woff_sum_cli;

        if (kind = 0) then
          delinq_kont = aarep_llc_k_id;
--          Вычисляет кол-во реструктуризации
        kol_rest = 0;
        /*
          select count(ol_id) from operlist, operrest
          where or_from_ol_id = ol_id
          and operlist.is_delete = 0
          and or_from_k_id = :aarep_llc_k_id
          and ol_date <= :rep_date_end
          and or_date_end >= :rep_date_end
          into :kol_rest;
          */
        select restruct_kol
        from aarep_get_restructuring(:aarep_llc_k_id, :rep_date_end, 41701)
        into kol_rest;

        if (kol_rest is null) then
          kol_rest = 0;

        aarep_memo = null;
        if (kol_rest > 0) then
          aarep_memo = 'Реструктурированный кредит';

        -- НБКР от 20 декабря 2023 года № 2023-П-12/80-4-(НПА)
        new_method = false;
        lir_rate = null;
        if (aarep_llc_loan_type = 1 and k_disb_date >= '12.01.2024') then
        begin
          lir_rate = (select first 1 lir_rate from llc_intrate where lir_date <= :k_disb_date order by lir_date desc);
          new_method = (k_by_rating = 1) or ((lir_rate > 0) and (k_percent > lir_rate)) ;
        end

        -- Групповые кредиты
        if (:aarep_llc_loan_type = 0) then
        begin
          aarep_llc_status_temp = case
                                    when ((:aarep_llc_k_deliq_days >= 90)) then 7 -- Потери
                                    when ((:aarep_llc_k_deliq_days between 60 and 89) or (kol_rest >= 3)) then 6 -- Сомнительные
                                    when (:aarep_llc_k_deliq_days between 30 and 59) then 5 -- Субстандартные
                                    when ((:aarep_llc_k_deliq_days between 4 and 29) or (kol_rest in (1,2))) then 4 -- Активы под наблюдением
                                    when ((:aarep_llc_k_deliq_days between 0 and 3) and (kol_rest = 0)) then 3  -- Удовлетворительные
                                  end;

          -- смерть клиента
          if (aarep_llc_status_temp < 4) then begin
            if (exists(select *
                       from operkred, operlist, refdata
                       where ok_from_k_id = :aarep_llc_k_id and
                             ok_ol_is_active = 1 and
                             ok_type = 5 and
                             ok_os + ok_perc + ok_fine <> 0 and
                             ok_ol_date <= :rep_date_end and
                             ok_from_ol_id = ol_id and
                             ol_from_rd_id = rd_id and
                             rd_code = 'Reason_WriteOff.IsClientDead')) then
            begin
              aarep_llc_status_temp = 4;
              aarep_memo = 'смерть одного из участников';
            end
          end

          -- выбытие участника из группы
          if (aarep_llc_status_temp < 4) then begin
            if (woff_sum_cli > 0) then
            begin
              aarep_llc_status_temp = 4;
              aarep_memo = 'Выбыл участник группы.';
            end
          end

          -- просрочка у отдельных участников ГСО более 7 дней в течении текущего цикла
          if (aarep_llc_status_temp < 4) then begin
            max_delinq_day = null;
            select max(ok_ol_date - ok_sh_date)
            from operkred
            where ok_ol_is_active = 1 and
                  ok_ol_date <= :rep_date_end and
                  ok_from_k_id = :aarep_llc_k_id and
                  ok_type in (4, 5, 6, 9) and
                  ok_ol_date > ok_sh_date and
                  ok_os + ok_perc < 0
            into max_delinq_day;
            if (max_delinq_day is null) then
              max_delinq_day = 0;
            if (max_delinq_day > 7) then
            begin
              aarep_llc_status_temp = 4;
              aarep_memo = 'Имеется просроченная выплата более 7 дней';
             end
          end
        end

        -- Индивидуальные кредиты
        if (:aarep_llc_loan_type = 1) then
        begin
          if (new_method)
          then aarep_llc_status_temp =
               case
                 when ((:aarep_llc_k_deliq_days >= 90))
                   then 7
                 when ((:aarep_llc_k_deliq_days between 60 and 89) or (kol_rest >= 3))
                   then 6
                 when (:aarep_llc_k_deliq_days between 30 and 59)
                   then 5
                 when ((:aarep_llc_k_deliq_days between 4 and 29) or (kol_rest in (1,2)))
                   then 4
                 when ((:aarep_llc_k_deliq_days between 0 and 3) and (kol_rest = 0))
                   then 3
               end;
          else aarep_llc_status_temp =
               case
                 when :aarep_llc_k_deliq_days > 180
                   then 7 -- Потери
                 when ((:aarep_llc_k_deliq_days between 91 and 180) or (kol_rest >=2 and :aarep_llc_k_deliq_days > 60))
                   then 6 -- Сомнительные
                 when ((:aarep_llc_k_deliq_days between 61 and 90) or ((kol_rest >=1 and :aarep_llc_k_deliq_days > 60) or (kol_rest >=2 and :aarep_llc_k_deliq_days = 0)))
                   then 5 -- Субстандартные
                 when ((:aarep_llc_k_deliq_days between 31 and 60) or (kol_rest >=1))
                   then 4 -- Активы под наблюдением
                 when (:aarep_llc_k_deliq_days <= 30 and (kol_rest = 0))
                   then 1 -- Стандартные
               end;
        end

        if (aarep_llc_status_temp < 4) then -- добавлено от 06.01.2016
        begin
          if (aarep_llc_k_id in (109010004283700501, 109010015491200101, 109010010117300201, 109010017747300101, 109010009799700301, 109010013042900201, -- добавлено по письму Айым от 12.01.2016 Улан
                             109010009261100201, 109010010353600301 -- добавлено по письму Айым от 12.02.2016 Улан
                             )) then
          begin
            aarep_llc_status_temp = 4;
            aarep_memo = 'смерть одного из участников';
          end
        end

        -- Вытаскивает из таблицы classific процент начисление РППУ
        select
               case :aarep_llc_loan_type
                 when 0 then cs_is_for_group
                 when 1 then cs_is_for_indiv
               end
        from classific
        where cs_id = :aarep_llc_status_temp
        into :aarep_llc_percent;
        aarep_llc_status_kont = aarep_llc_status_temp;
        aarep_llc_percent_kont = aarep_llc_percent;

        -- Деление на клиентов
        for select kc_from_c_id, cast(kc_amount_approved as double precision) / :aarep_disb * :aarep_llc_k_balance,
                   kc_has_loans,
                   case :aarep_llc_loan_type
                     when 1 then kc_has_family_loans
                     else 0
                   end,
                   kc_closed_ol_id
            from knclient
            where kc_from_k_id = :aarep_llc_k_id and
                  kc_amount_approved > 0
            into kc_from_c_id, kc_balance, kc_has_loans, kc_has_family_loans, kc_closed_ol_id
        do
        begin
          -- Является ли клиент юр лицом
          select c.c_type
          from clients c
          where c.c_id = :kc_from_c_id
          into :c_type;
          -- если была закрытия доли клиента то выполняет следующее
          if (woff_sum_cli <> 0) then
          begin
            if (exists(select *
                       from operlist
                       where ol_id = :kc_closed_ol_id and
                             operlist.is_delete = 0 and
                             ol_date_oper_day <= :rep_date_end)) then
            begin
              kc_balance = 0;
              kc_has_loans = 0;
              kc_has_family_loans = 0;
            end
          end

          if (kc_has_loans is null) then
            kc_has_loans = 0;
          if (kc_has_family_loans is null) then
            kc_has_family_loans = 0;
          aarep_llc_status_temp = aarep_llc_status_kont; -- для того чтоб не затереть первоначальный статус, т.е. на группу один раз присваивается
          aarep_llc_percent = aarep_llc_percent_kont;
          if (aarep_llc_status_temp < 4 and
              kc_has_loans = 1 and
              c_type = 0) then --Данное условие не включает в себя юр. лица (RFCKG-63)
          begin
            aarep_llc_status_temp = 4;
            aarep_llc_percent = 10;
          end

--1. по своим <30 Дней, в других ФКУ галочка =<30, у родственнков есть галочка (=<30) - то "Активы под наблюдением 10%"
--2. по своим <30 Дней, в других ФКУ галочка >30, у родственнков есть галочка (>30) - то "Субстандартный 25%"
--  Определяет максимальное кол-во просроченных дней для классификации РППУ по индив c другой ФКУ
          kk_str = null;
          if (aarep_llc_loan_type = 1 and
              kc_has_loans = 1) then
          begin
            kk = null;
            with tabl1
            as (select
                  case kt_status_pogash
                    when 'Платежи по графику' then 2
                    when 'до 30 дней просрочки' then 3
                    when '30 дней до 59 дней просрочки' then 4
                     when '31 дней до 60 дней просрочки' then 4
                    when '60 дней до 89 дней просрочки' then 5
                     when '61 дней до 90 дней просрочки' then 5
                    when '90 дней  до 119 дней просрочки' then 6
                     when '91 дней  до 120 дней просрочки' then 6
                    when '120 дней просрочки и более' then 7
                     when '121 дней просрочки и более' then 7
                  else 1 end as kk
                from kibtaken
                where (kt_from_c_id = :kc_from_c_id or
                      kt_from_c_id = (select c_original_id
                                      from clients
                                      where c_id = :kc_from_c_id and
                                            c_original_id > 0)) and
                      kt_year = :rep_year and
                      kt_month = :rep_month)
            select max(kk)
            from tabl1
            into kk;
            if (kk > 3 and aarep_llc_status_temp < 5 and not new_method) then
            begin
              aarep_llc_status_temp = 5;
              aarep_llc_percent = 25;
            end
            kk_str =
              case :kk
                when 2 then 'Платежи по графику'
                when 3 then 'до 30 дней просрочки'
                when 4 then iif (rep_mn > rep_mnsv, '31 дней до 60 дней просрочки', '30 дней до 59 дней просрочки')
                when 5 then iif (rep_mn > rep_mnsv, '61 дней до 90 дней просрочки', '60 дней до 89 дней просрочки')
                when 6 then iif (rep_mn > rep_mnsv, '91 дней  до 120 дней просрочки', '90 дней  до 119 дней просрочки')
                when 7 then iif (rep_mn > rep_mnsv, '121 дней просрочки и более', '120 дней просрочки и более')
              else '-- не выбран --' end;
          end

--          Разработка от января 2015, проверка супруг в другом ФКУ
          suprug_name_fio = '';
          kk_str_poruch = null;
          is_suprug_kred = 0;
          crel_other_c_id = null;

          if (aarep_llc_loan_type = 1) then
          begin
--          у клиента вытаскивает супругу

            crel_other_c_id = null;
            /*  -- заменен на нижеследующий
                 select CREL_OTHER_C_ID from clientrelation
                 where crel_from_c_id = :kc_from_c_id -- ошибка было такое :LKC_FROM_C_ID
                 and crel_from_rel_id in ('1','2')
                 and CREL_IS_ACTIVE = 1
                 rows 1
                 */
            select crel_other_c_id
            from hp_client_relation_list(:kc_from_c_id, 0)
            where crel_from_rel_id in ('1', '2') and
                  crel_is_active = 1 rows 1
            into crel_other_c_id;

          from tabl1, llc_kont_cli
          where lm_id = lkc_from_lm_id)
      select
        lkc_from_lm_id,
        lkc_from_k_id,
        max(lkc_status),
        max(lkc_percent)
      from tabl2
      where lkc_is_indiv = 0
      group by lkc_from_k_id, lkc_from_lm_id
      having max(lkc_status) >= 5
  into
    :lkc_from_lm_id,
    :llc_k_id,
    :llc_status,
    :llc_percent
  do
  begin
    for
      select
        lkc_number,
        lkc_status,
        lkc_percent,
        lkc_is_indiv
      from llc_kont_cli
      where lkc_from_lm_id = :lkc_from_lm_id
        and lkc_from_k_id = :llc_k_id
        and lkc_status < :llc_status
    into
      :llc_number,
      :lkc_status,
      :lkc_percent,
      :lkc_is_indiv
    do
    begin
      update llc_kont_cli set
        lkc_status = :llc_status,
        lkc_percent = :llc_percent
      where lkc_from_lm_id = :lkc_from_lm_id
        and lkc_from_k_id = :llc_k_id
        and lkc_number = :llc_number;
      suspend;
    end
  end

--  Присваиваем максимальный статус на все кредиты у клиента,
--  т.е. если у клиента несколько кредитов то на все кредиты присваивается максимальный статус
  for with tabl1
      as (select lm_id
          from llc_main
          where lm_date = :rep_date_end),
      tabl2
      as (select llc_kont_cli.*
          from tabl1, llc_kont_cli
          where lm_id = lkc_from_lm_id)
      select
        lkc_from_lm_id,
        lkc_from_c_id,
        max(lkc_status),
        max(lkc_percent)
      from tabl2
      where lkc_is_indiv = 0
      group by lkc_from_c_id, lkc_from_lm_id
      having max(lkc_status) >= 5
  into
    :lkc_from_lm_id,
    :lkc_from_c_id,
    :llc_status,
    :llc_percent
  do
  begin
    for
      select
        lkc_number,
        lkc_status,
        lkc_percent
      from llc_kont_cli
      where lkc_from_lm_id = :lkc_from_lm_id
        and lkc_from_c_id = :lkc_from_c_id
        and lkc_status < :llc_status
    into
      :llc_number,
      :lkc_status,
      :lkc_percent
    do
    begin
      update llc_kont_cli set
        lkc_status = :llc_status,
        lkc_percent = :llc_percent
      where lkc_from_lm_id = :lkc_from_lm_id
        and lkc_from_k_id = :llc_k_id
        and lkc_number = :llc_number;
      suspend;
    end
  end

-- На основе клиентов корректирует основной РППУ на уровне группы
  for
    select
      lkc_from_lm_id,
      lkc_from_k_id,
      max(lkc_status),
      sum(cast(lkc_balance as double precision) * lkc_percent / 100) / sum(lkc_balance) * 100,
      lkc_deliq_days,
      lkc_is_indiv,
      lkc_from_reg_id
    from llc_kont_cli, llc_main
    where lm_is_signed = 1
      and lm_date_status between :rep_date_end - 4 and :rep_date_end + 4
      and lkc_from_lm_id = lm_id
    group by lkc_from_lm_id, lkc_from_k_id, lkc_is_indiv, lkc_from_reg_id, lkc_deliq_days
    having sum(lkc_balance) <> 0
  into
    :lkc_from_lm_id,
    :lkc_from_k_id,
    :lkc_status,
    :lkc_percent,
    :lkc_deliq_days,
    :lkc_is_indiv,
    :lkc_from_reg_id
  do
  begin
    update llc_kont set
      lk_deliq_days = :lkc_deliq_days,
      lk_from_reg_id = :lkc_from_reg_id,
      lk_status = :lkc_status,
      lk_percent = :lkc_percent,
      lk_is_indiv = :lkc_is_indiv,
      lk_percent_orig = :lkc_percent
    where lk_from_lm_id = :lkc_from_lm_id
      and lk_from_k_id = :lkc_from_k_id;
  end

  --По запросу Назгуль от 01.03.21 Тема письма: Новые изменения в РППУ
  for
    select
      lkc_from_c_id,
      lkc_percent_new,
      lkc_status
    from hp_llc_check_transfer_unclas(:rep_region_id, :rep_date_end)
  into
    :lkc_from_c_id,
    :lkc_percent,
    :lkc_status
  do
  begin
    update llc_kont_cli set
      lkc_percent = :lkc_percent,
      lkc_status=:lkc_status
    where lkc_from_c_id = :lkc_from_c_id
      and lkc_is_indiv = 1
      and not lkc_mark
      and exists(select * from llc_main where lm_id = lkc_from_lm_id and lm_date_status = :rep_date_end);
  end
  --

  update refdata set
    rd_memo = 'Реструктуризация'
  where rd_id = 4075;
end
