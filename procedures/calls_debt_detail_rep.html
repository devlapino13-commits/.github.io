CREATE PROCEDURE CALLS_DEBT_DETAIL_REP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(255),
    KONTRACT_NO VARCHAR(255),
    DELIQ_START_DATE DATE,
    CLIENT_FIO VARCHAR(255),
    FIO VARCHAR(255),
    CONT_TYPE VARCHAR(20),
    RES VARCHAR(20),
    COMMENTS VARCHAR(255),
    PAY_DATE DATE,
    CALL_DATETIME TIMESTAMP,
    OPER_FIO VARCHAR(255))
AS
declare variable contacts_id integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /*
        CREATED BY IGUSEV 17.07.2012
        PURPOSE: Вытаскивает список всех обзвоненных должников и их контактных лиц за период
  */

  for select reg.reg_name, /* наименование офиса кред. папки*/
             kont.k_kontract_no, /* № кредитного договора*/
             ccdl_deliq_start_date, /* дата начала просрочки*/
             cl.c_name_fio,
             /* ФИО должника*/
             (
             case
               when cdbl.ccdl_from_contacts_id is null then cl.c_name_fio
               else cont.cc_name
             end) as fio,
             /* ФИО конт лица*/
             (
             case
               when cdbl.ccdl_from_contacts_id is null then 'Клиент'
               else 'Конт. лицо'
             end) as cont_type,
             coalesce(cdbl.ccdl_from_contacts_id, 0) as contacts_id,
             (
             case cdr.ccdr_result
               when -1 then 'Не дозвон.'
               when 0 then 'Отрицат.'
               when 1 then 'Положит.'
             end) as res,
             cdr.ccdr_comments, cdr.ccdr_pay_date, /* дата обещ. платежа*/
             cdr.ccdr_call_datetime, /* дата звонка*/
             cdr.ccdr_oper_fio /*фио оператора*/
      from callcenter_debtors_list cdbl
      inner join callcenter_debtors_result cdr on cdbl.ccdl_id = cdr.ccdr_from_ccdl_id
      inner join kontract kont on kont.k_id = cdbl.ccdl_from_k_id
      inner join region_list(:rep_region_id, 1, -1, 1) regl on regl.id_list = ccdl_kf_branch_id
      inner join region reg on reg.reg_unicode = regl.id_list
      inner join clients cl on cl.c_id = cdbl.ccdl_from_c_id
      left join contacts cont on cont.cc_id = cdbl.ccdl_from_contacts_id
      where cdbl.ccdl_create_date >= :rep_date_begin and
            cdbl.ccdl_create_date <= :rep_date_end + 1 and
            cdr.ccdr_call_datetime >= :rep_date_begin and
            cdr.ccdr_call_datetime <= :rep_date_end + 1 and
            cdr.ccdr_oper_fio not like '%Automatic%'
      order by reg.reg_name, k_kontract_no, cdbl.ccdl_deliq_start_date, cl.c_name_fio, contacts_id, cc_name
      into :reg_name, :kontract_no, :deliq_start_date, :client_fio, :fio, :cont_type, :contacts_id, :res, :comments,
           :pay_date, :call_datetime, :oper_fio
  do
  begin
    suspend;
  end
end
