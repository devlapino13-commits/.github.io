CREATE PROCEDURE AAREP_ACCRUED_INTEREST_1C(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AAREP_KONTRACT_NO VARCHAR(20),
    K_CLIENTS_CASHIER_ID VARCHAR(20),
    CL_CASHIER_NAME VARCHAR(100),
    REPORT_DATE DATE,
    AAREP_PERC_END CURR_TYPE,
    ACCOUNT_DT VARCHAR(20),
    ACCOUNT_KT VARCHAR(20),
    AAREP_OUT_PERC_END DECIMAL(18,4))
AS
declare variable aarep_k_id id_type;
declare variable c_id id_type;
declare variable pr_id integer;
declare variable reg_id integer;
begin
  report_date = :rep_date_end;
  for select aarep_k_id, aarep_kontract_no, c_id, k_clients_cashier_id, aarep_perc_end, -- начисленные проценты за период
             aarep_out_perc_end
      from aarep_calc_perc_list(:rep_region_id, :rep_date_begin, :rep_date_end, '---')
--  where aarep_perc_end <> 0
      into aarep_k_id, aarep_kontract_no, c_id, k_clients_cashier_id, aarep_perc_end, aarep_out_perc_end
  do
  begin
    select coalesce(c_name_fio, '')
    from clients
    where c_id = :c_id
    into cl_cashier_name;
--  Вытаскивает счета 3, 36 указанного контракта
    account_dt = '';
    account_kt = '';

    select kal_new_reg_id, k_from_pr_id
    from kontract, knaktlist
    where k_id = :aarep_k_id and
          knaktlist.is_delete = 0 and
          kal_from_kf_id = k_from_kf_id and
          current_date between kal_date and kal_date_end
    into :reg_id, :pr_id;

    select res_account
    from hp_make_kredit_acclist(:pr_id, :reg_id, 3) -- Полученный процент
    into :account_dt;

    select res_account
    from hp_make_kredit_acclist(:pr_id, :reg_id, 36) -- Процентный доход
    into :account_kt;

    suspend;
  end
end
