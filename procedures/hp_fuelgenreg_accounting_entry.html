CREATE PROCEDURE HP_FUELGENREG_ACCOUNTING_ENTRY(
    FG_ID D_INTEGER)
RETURNS (
    OL_ID D_ID,
    ERR_TEXT D_STR_255)
AS
declare variable un_id d_id;
declare variable curr_code d_str_lang;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable fg_dtwriteoff d_date;
declare variable fg_dtbeg d_date;
declare variable fg_dtend d_date;
declare variable fg_office_id d_integer;
declare variable reg_name d_str_code;
declare variable fg_booker_id d_id;
declare variable pet_id d_id;
declare variable pet_name d_str_255;
declare variable pet_label d_str_code;
declare variable pet_is_rent d_integer;
declare variable pet_norma_km d_integer;
declare variable is_car d_smallint;
declare variable fg_whours d_currency;
declare variable fg_wrate d_currency;
declare variable fg_usedup d_currency;
declare variable fg_usedup_sum d_currency;
begin
  err_text = '';
  curr_code = '417';

  for
    select
      fg_dtwriteoff,
      fg_dtbeg,
      fg_dtend,
      fg_office_id,
      reg_name,
      fg_booker_id,
      fg_pet_id,
      pet_name,
      pet_label,
      pet_is_rent,
      pet_norma_km,
      is_car,
      fg_whours,
      fg_wrate,
      fg_usedup,
      fg_usedup_sum
    from hp_fuelgenreg_view (null, null, null, :fg_id)
    where fg_status = 0
      and fg_driver_dtsign is not null
      and fg_manager_dtsign is not null
  into
    :fg_dtwriteoff,
    :fg_dtbeg,
    :fg_dtend,
    :fg_office_id,
    :reg_name,
    :fg_booker_id,
    :pet_id,
    :pet_name,
    :pet_label,
    :pet_is_rent,
    :pet_norma_km,
    :is_car,
    :fg_whours,
    :fg_wrate,
    :fg_usedup,
    :fg_usedup_sum
  do
  begin
    un_id = (
      select first 1
        un_id
      from username
      where un_from_c_id = :fg_booker_id);
    acc_dt = (
      select al_id
      from acc_list
      where al_from_reg_id = :fg_office_id
        and al_currency = :curr_code
        and al_account = iif(:is_car = 1, iif(:pet_is_rent = 1, '80308/3', '80308/2'), '80304/7'));
    acc_kt = (
      select al_id
      from acc_list
      where al_from_reg_id = :fg_office_id
        and al_currency = :curr_code
        and al_account = '11533/1');

    insert into operlist (
      ol_from_reg_id,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_memo,
      ol_user_id
     )
    values (
      :fg_office_id,
      8,
      5,
      :fg_dtwriteoff,
      :fg_id,
      17,
      f_format('Списание ГСМ %1 л. %2 %3 номер: %4',
        cast(:fg_usedup as decimal(15,2)), :reg_name, :pet_name, :pet_label),
      :un_id
    )
    returning ol_id
    into :ol_id;

    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_detail,
      ob_summa,
      ob_summa_val
    )
    values (
      :ol_id,
      :curr_code,
      1,
      :acc_dt,
      :acc_kt,
      f_format('Списание ГСМ, отработано: %1 %3, норма: %2',
        cast(:fg_whours as decimal(15, 2)), cast(:fg_wrate as decimal(15, 2)), iif(:is_car = 1, 'км.', 'ч.')),
      :fg_usedup_sum,
      :fg_usedup_sum
    );

    insert into petopers (
      peo_from_ol_id,
      peo_id,
      peo_oper_date,
      peo_from_reg_id,
      peo_bill_id,
      peo_bill_line_no,
      peo_oper_type,
      peo_from_c_id,
      peo_cost,
      peo_from_pen_id,
      peo_from_pet_id,
      peo_trans_liter,
      peo_trans_summa,
      peo_probeg_fact_liter,
      peo_probeg_fact_km,
      peo_kolich,
      peo_liter,
      peo_summa,
      peo_speedometr,
      peo_talon_liter
    )
    select first 1
      :ol_id,
      1,
      :fg_dtwriteoff,
      :fg_office_id,
      peo_bill_id,
      peo_bill_line_no,
      7,
      peo_from_c_id,
      peo_cost,
      peo_from_pen_id,
      :pet_id,
      -:fg_usedup,
      -:fg_usedup_sum,
      :fg_usedup,
      :fg_whours,
      0,
      0,
      0,
      0,
      0
    from aarep_trans_talon_saldo(:pet_id, :fg_dtend)
    where /*peo_from_reg_id = :fg_office_id and*/ peo_liter > 0
    order by peo_bill_id desc;
  end

  if (pet_id is null) then
    err_text = 'Данные для построения бухгалтерских проводок не определены.';

  suspend;
end
