CREATE PROCEDURE HP_SURVEY_HIST_ANSW(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    RD_ID D_INTEGER,
    QUES_ID D_INTEGER)
RETURNS (
    OPR_ID D_ID,
    OVS_ID D_INTEGER,
    OVS_DATE_BEGIN D_DATE,
    OVS_DATE_END D_DATE,
    OVS_DATETIME D_DATETIME,
    OVS_IS_ACTIVE D_BOOL,
    OVS_VALUE D_STR_2048,
    OVS_ADD_VALUE D_CURRENCY,
    OVR_STATE D_BOOL,
    OVR_VALUE D_STR_2048,
    RES_RD_ID D_INTEGER,
    RD_NAME D_STR_2048,
    UN_ID D_ID,
    UN_FIO D_STR_100,
    IS_ACTUAL D_BOOL)
AS
declare variable opr_actual_ovs_id d_integer;
declare variable opr_fact_ovs_id d_integer;
begin
  select
    opr_id,
    opr_actual_ovs_id,
    opr_fact_ovs_id
  from objectprops
  where opr_object_kind = :obj_kind
    and opr_object_id = :obj_id
    and opr_subject_kind = -1
    and opr_subject_id = :ques_id
  into
    :opr_id,
    :opr_actual_ovs_id,
    :opr_fact_ovs_id;

  for
    select
      ovs_id,
      ovs_date_begin,
      ovs_date_end,
      ovs_datetime,
      ovs_is_active,
      ovs_value,
      ovs_add_value,
      ovr_state,
      ovr_value,
      rd_id,
      iif(ovr_value <> '', rd_name || ': ' || ovr_value, rd_name),
      un_id,
      un_fio,
      iif(ovs_id = :opr_actual_ovs_id, 1, 0)
    from objectvalues
      inner join objectvalres
        on ovr_from_opr_id = ovs_from_opr_id
          and ovr_from_ovs_id = ovs_id
      inner join refdata
        on rd_id = ovr_result_rd_id
      inner join username
        on un_id = ovs_from_un_id
    where ovs_from_opr_id = :opr_id
      and ovs_is_active = 1
      and ovs_date_end >= ovs_date_begin
  into
    :ovs_id,
    :ovs_date_begin,
    :ovs_date_end,
    :ovs_datetime,
    :ovs_is_active,
    :ovs_value,
    :ovs_add_value,
    :ovr_state,
    :ovr_value,
    :res_rd_id,
    :rd_name,
    :un_id,
    :un_fio,
    :is_actual
  do
  begin
    suspend;
  end
end
