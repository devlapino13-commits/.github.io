CREATE PROCEDURE HP_FINPOLIS_639_CHECK
AS
declare variable fps_id integer;
declare variable c_id id_type;
declare variable found_pass_no integer;
declare variable found_inn_no integer;
declare variable found_client_fio integer;
declare variable found_birth_date integer;
declare variable do_insert integer;
begin
  -- процедура осуществляет проверку по перечню 639 и для клиентов по которым обнаружены совпадения в  FINPOLIS_639_CLI_LINK
  for select fps_id, c_id, found_pass_no, found_inn_no,
             found_client_fio, found_birth_date
      from aarep_check_finpolis_639
      into fps_id, c_id, found_pass_no, found_inn_no,
           found_client_fio, found_birth_date
  do
  begin
    do_insert = 0;

    if (found_pass_no = 1) then do_insert = 1;
    if (found_inn_no = 1) then do_insert = 1;
    if ((found_client_fio = 1) and (found_birth_date = 1)) then do_insert = 1;

    if (do_insert = 1) then
    begin
        if (not exists(select *
                       from finpolis_639_cli_link t
                       where t.fcl_from_c_id = :c_id and
                             t.fcl_from_fps_id = :fps_id)) then
        begin
          insert into finpolis_639_cli_link (fcl_from_c_id, fcl_from_fps_id)
          values (:c_id, :fps_id);
        end
    end
  end
end
