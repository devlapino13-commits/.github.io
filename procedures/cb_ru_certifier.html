CREATE PROCEDURE CB_RU_CERTIFIER(
    REP_DATE_END D_DATE)
RETURNS (
    CERTIFIER_FIO D_STR_100,
    DISB_SUM_3 D_CURRENCY,
    DELINQ_BALANCE30_3 D_CURRENCY,
    KOL_APPROVED_3 D_SMALLINT,
    DISB_SUM_6 D_CURRENCY,
    DELINQ_BALANCE30_6 D_CURRENCY,
    PAR30_3 D_CURRENCY,
    BONUS_3 D_CURRENCY,
    KOL_APPROVED_6 D_SMALLINT,
    PAR30_6 D_CURRENCY,
    BONUS_6 D_CURRENCY,
    BONUS_TOTAL D_CURRENCY)
AS
declare variable month1 d_smallint;
declare variable year1 d_smallint;
declare variable certifier_id d_id;
declare variable rep_date_begin_3 date;
declare variable rep_date_end_3 date;
declare variable rep_date_begin_6 date;
declare variable rep_date_end_6 date;
begin
  -- бонус 3 месяца
  month1 = extract(month from :rep_date_end) - 3;
  year1 = extract(year from :rep_date_end);
  if (month1 < 1) then
  begin
    month1 = 12 + month1;
    year1 = year1 - 1;
  end
  rep_date_begin_3 = '1.'||month1||'.'||year1;
  rep_date_end_3 = dateadd(month, 1,rep_date_begin_3) - 1;

  month1 = extract(month from :rep_date_end) - 6;
  year1 = extract(year from :rep_date_end);
  if (month1 < 1) then
  begin
    month1 = 12 + month1;
    year1 = year1 - 1;
  end
  rep_date_begin_6 = '1.'||month1||'.'||year1;
  rep_date_end_6 = dateadd(month, 1,rep_date_begin_6) - 1;

  for
  select c_id, c_name_fio from clients
  where c_id in (1010100059428) -- Эркин Мирзаева
  into certifier_id, certifier_fio
  do
  begin
    disb_sum_3 = null;
    delinq_balance30_3 = null;
    kol_approved_3 = null;
    select sum(k_amount_approved),
           sum((select oks_balance from oksummary
            where oks_from_k_id = k_id
            and :rep_date_end - oks_delinq_date > 30
            and :rep_date_end between oks_date_begin and oks_date_end
            and oks_close_state = 0)),
            count(*)
    from kontstate, kontract, knclient
    where ks_ol_is_delete = 0
    and ks_state = 1
    and ks_date_begin between :rep_date_begin_3 and :rep_date_end_3
    and ks_from_k_id = k_id
    and k_approved_manager = :certifier_id
    and ks_from_k_id = kc_from_k_id
    and kc_amount_approved > 0
    and kc_circle < 2
    into disb_sum_3, delinq_balance30_3, kol_approved_3;
    disb_sum_3 = coalesce(disb_sum_3, 0);
    delinq_balance30_3 = coalesce(delinq_balance30_3, 0);
    kol_approved_3 = coalesce(kol_approved_3, 0);
  
    if (disb_sum_3 <> 0) then
      par30_3 = round((cast(delinq_balance30_3 as double precision) / disb_sum_3 * 100), 2);
    else
      par30_3 = 0;
  
    bonus_3 = case
                when par30_3 < 2.0 then 1 * kol_approved_3 * 40
                when par30_3 < 2.5 then 0.75 * kol_approved_3 * 40
                when par30_3 < 3 then 0.5 * kol_approved_3 * 40
                when par30_3 < 3.5 then 0.25 * kol_approved_3 * 40
                else 0
              end;
  
    disb_sum_6 = null;
    delinq_balance30_6 = null;
    kol_approved_6 = null;
    select sum(k_amount_approved),
           sum((select oks_balance from oksummary
            where oks_from_k_id = k_id
            and :rep_date_end - oks_delinq_date > 30
            and :rep_date_end between oks_date_begin and oks_date_end
            and oks_close_state = 0)),
            count(*)
    from kontstate, kontract, knclient
    where ks_ol_is_delete = 0
    and ks_state = 1
    and ks_date_begin between :rep_date_begin_6 and :rep_date_end_6
    and ks_from_k_id = k_id
    and k_approved_manager = :certifier_id
    and ks_from_k_id = kc_from_k_id
    and kc_amount_approved > 0
    and kc_circle < 2
    into disb_sum_6, delinq_balance30_6, kol_approved_6;
    disb_sum_6 = coalesce(disb_sum_6, 0);
    delinq_balance30_6 = coalesce(delinq_balance30_6, 0);
    kol_approved_6 = coalesce(kol_approved_6, 0);
  
    if (disb_sum_6 <> 0) then
      par30_6 = round((cast(delinq_balance30_6 as double precision) / disb_sum_6 * 100), 2);
    else
      par30_6 = 0;
  
    bonus_6 = case
                when par30_6 < 3.5 then 1 * kol_approved_6 * 50
                when par30_6 < 4.0 then 0.75 * kol_approved_6 * 50
                when par30_6 < 4.5 then 0.5 * kol_approved_6 * 50
                when par30_6 < 5.0 then 0.25 * kol_approved_6 * 50
                else 0
              end;
  
    bonus_total = bonus_3 + bonus_6;
    suspend;
  end
end
