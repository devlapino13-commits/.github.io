Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE AAREP_KZ_RO_NALOG(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    OKS_FROM_KC_ID D_ID,
    PR_NAME D_STR_CODE,
    OKS_DISB_AMOUNT D_CURRENCY,
    R_RO D_CURRENCY,
    OKS_PMT_OS_ALL D_CURRENCY,
    OKS_PMT_PERC_ALL D_CURRENCY,
    OKS_ECL_VBS D_CURRENCY,
    OKS_ECL_VBS_BEGIN D_CURRENCY,
    REPAY_OS_PERC D_CURRENCY,
    OKS_BALANCE_BEGIN D_CURRENCY,
    OKS_BAL_PERC D_CURRENCY,
    REPAY_DISB_PERC D_CURRENCY,
    OKS_BALANCE D_CURRENCY,
    BAL_PERC D_CURRENCY,
    RO_RAZN D_CURRENCY,
    RO_DOHOD D_CURRENCY,
    RO_RASHOD D_CURRENCY,
    OKS_DISB_AMOUNT_BEGIN D_CURRENCY,
    OKS_BAL_PERC_BEGIN D_CURRENCY,
    REPAY_DISB_PERC_BEGIN D_CURRENCY)
AS
begin

  for
  select c_name_fio,
         k_kontract_no,
         oks_from_kc_id,
         pr_name,
         oks_disb_amount,
         oks_pmt_os_all,
         oks_pmt_perc_all,
         oks_ecl_vbs,
         oks_bal_perc,
         oks_balance

  from aarep_kn_oksummary_rep(:rep_date_begin, :rep_date_end)
  into c_name_fio, --Наименование заемщика → ФИО клиента и статус займа
       k_kontract_no, --№ договора → Номер займа / № договора
       oks_from_kc_id, --Идентификационный номер договора → Id займа
       pr_name, --Признак однородности → ФИО клиента и статус займа / Наименование продукта (Аннуитет/Дифференцированный → однородность)
       oks_disb_amount,
       oks_pmt_os_all,
       oks_pmt_perc_all,
       oks_ecl_vbs,
       oks_bal_perc,
       oks_balance
  do
  begin
    pr_name = 'Однородный'; -- все контракты будут однороднымии
    oks_ecl_vbs_begin = 0;
    select oks_ecl_vbs, oks_balance, oks_disb_amount, oks_bal_perc  from kn_oksummary
    where oks_date_begin between :rep_date_begin - 1 and :rep_date_begin - 1
    and oks_from_kc_id = :oks_from_kc_id
    into oks_ecl_vbs_begin, -- Сумма провизий на начало периода → Расход / Расходы по РО / РО на начало
         oks_balance_begin, --Задолженность на начало периода → Своевременные платежи (день в день) / Сумма выдачи
         oks_disb_amount_begin,
         oks_bal_perc_begin;

    --Задолженность на начало периода → Своевременные платежи (день в день) / Сумма выдачи
    repay_disb_perc_begin = - oks_disb_amount_begin + oks_bal_perc_begin;

    --Уменьшение права требования (исполнение заемщиком) → Всего платежи за период / Осн. сумма + Проценты
    repay_os_perc = oks_pmt_os_all + oks_pmt_perc_all;

    --Увеличение права требования (выдача, начисления) → Всего платежи за период / Всего (отрицательные значения → начисления)
    repay_disb_perc = - oks_disb_amount + oks_bal_perc;

    --Задолженность на конец года → ФИО клиента и статус займа / Сумма займа – Всего платежи
    bal_perc = oks_balance + oks_bal_perc;

    -- Доходы от снижения провизий → Расход / Расход по кор-ке до РО начисление по счету 6240.21
    ro_razn = oks_ecl_vbs - oks_ecl_vbs_begin;
    if (ro_razn < 0) then ro_dohod = -ro_razn;
    else ro_dohod = 0;

    -- Расходы по созданию провизий → Расход / Расходы по РО начисление по счету 7440.21
    if (ro_razn > 0) then ro_rashod = ro_razn;
    else ro_rashod = 0;


    suspend;
  end
end
