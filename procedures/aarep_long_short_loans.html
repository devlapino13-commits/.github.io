Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE AAREP_LONG_SHORT_LOANS(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    SEQ_NO D_INTEGER,
    K_KONTRACT_NO D_STR_CODE,
    C_NAME_FIO D_STR_255,
    C_INN D_STR_20,
    K_ID D_ID,
    REG_NAME D_STR_255,
    PR_NAME D_STR_100,
    K_PERIOD D_INTEGER,
    LOAN_TYPE D_STR_20,
    OKS_BALANCE_PREV CURR_TYPE,
    OKS_BAL_PERC_PREV CURR_TYPE,
    OKS_BALANCE_PREV_DELINQ CURR_TYPE,
    OKS_BAL_PERC_DELINQ_PREV CURR_TYPE,
    OKS_ECL_VBS_PREV CURR_TYPE,
    OKS_ECL_ACC_KIND_BEG D_SMALLINT,
    OKS_ACR_PERC CURR_TYPE,
    OKS_PMT_OS_ALL_SS CURR_TYPE,
    OKS_PMT_PERC_ALL_SS CURR_TYPE,
    OKS_PMT_OS_ALL_DELINQ CURR_TYPE,
    OKS_PMT_PERC_ALL_SS_DELINQ CURR_TYPE,
    TRANS_FINE CURR_TYPE,
    OKS_BALANCE_CUR CURR_TYPE,
    OKS_BAL_PERC_CUR CURR_TYPE,
    OKS_BALANCE_DELINQ_CUR CURR_TYPE,
    OKS_BAL_PERC_DELINQ_CUR CURR_TYPE,
    OKS_ECL_VBS_CUR CURR_TYPE)
AS
declare variable seq_counter integer = 0;
declare variable prev_month_end date;
begin

-- конец предыдущего месяца
prev_month_end = :rep_date_begin - 1;

for
select
    oks.oks_from_k_id,
    sum(iif(oks_date_begin = :prev_month_end and oks_close_state = 0 and oks_delinq_days = 0,oks_balance,0)),
    sum(iif(oks_date_begin = :prev_month_end and oks_delinq_days = 0, oks_bal_perc,0)),
    sum(iif(oks_date_begin = :prev_month_end and oks_close_state = 0 and oks_delinq_days > 0,oks_balance,0)),
    sum(iif(oks_date_begin = :prev_month_end and oks_delinq_days > 0, oks_bal_perc,0)),
    sum(iif(oks_date_begin = :prev_month_end and oks_close_state = 0, oks_ecl_vbs,0)),
    sum(iif(oks_date_begin = :prev_month_end, oks_ecl_acc_kind, 0)),
    sum(iif(oks_date_begin between :rep_date_begin and :rep_date_end, oks_acr_perc,0)),
    sum(iif(oks_date_begin between :rep_date_begin and :rep_date_end and oks_delinq_days = 0,
            oks_pmt_os_norm + oks_pmt_os_delinq + oks_pmt_os_adv,0)),
    sum(iif(oks_date_begin between :rep_date_begin and :rep_date_end and oks_delinq_days = 0,
            oks_pmt_perc_norm + oks_pmt_perc_delinq + oks_pmt_perc_adv,0)),
    sum(iif(oks_date_begin between :rep_date_begin and :rep_date_end and oks_delinq_days > 0,
            oks_pmt_os_norm + oks_pmt_os_delinq + oks_pmt_os_adv,0)),
    sum(iif(oks_date_begin between :rep_date_begin and :rep_date_end and oks_delinq_days > 0,
            oks_pmt_perc_norm + oks_pmt_perc_delinq + oks_pmt_perc_adv,0)),

    sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0 and oks_delinq_days = 0, oks_balance,0)),
    sum(iif(oks_date_begin = :rep_date_end and oks_delinq_days = 0, oks_bal_perc,0)),
    sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0 and oks_delinq_days > 0, oks_balance,0)),
    sum(iif(oks_date_begin = :rep_date_end and oks_delinq_days > 0, oks_bal_perc,0)),
    sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0, oks_ecl_vbs,0))

from kn_oksummary oks
where oks.oks_date_begin between :prev_month_end and :rep_date_end
group by oks.oks_from_k_id

into :k_id,
     :oks_balance_prev,
     :oks_bal_perc_prev,
     :oks_balance_prev_delinq,
     :oks_bal_perc_delinq_prev,
     :oks_ecl_vbs_prev,
     :oks_ecl_acc_kind_beg,
     :oks_acr_perc,
     :oks_pmt_os_all_ss,
     :oks_pmt_perc_all_ss,
     :oks_pmt_os_all_delinq,
     :oks_pmt_perc_all_ss_delinq,
     :oks_balance_cur,
     :oks_bal_perc_cur,
     :oks_balance_delinq_cur,
     :oks_bal_perc_delinq_cur,
     :oks_ecl_vbs_cur
do
begin

    select k.k_kontract_no, c.c_name_fio, c.c_inn, r.reg_code || ' ' || r.reg_name, pr.pr_name, k_period, iif(k.k_period <= 12,'краткосрочный','долгосрочный')
    from kontract k
    inner join knclient kc on kc.kc_from_k_id = k.k_id
    inner join clients c on c.c_id = kc.kc_from_c_id
    inner join knaktlist akt on akt.kal_from_kf_id = k.k_from_kf_id
      and akt.kal_date <= :rep_date_end
      and akt.kal_date_end >= :rep_date_end
      and akt.is_delete = 0
    inner join region r on r.reg_unicode = akt.kal_new_reg_id
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where k.k_id = :k_id
     into :k_kontract_no,
          :c_name_fio,
          :c_inn,
          :reg_name,
          :pr_name,
          :k_period,
          :loan_type;

    trans_fine = 0;

    select coalesce(sum(ok_fine), 0)
    from operkred
    where ok_ol_is_active = 1
      and ok_ol_date between :rep_date_begin and :rep_date_end
      and ok_type > 3
      and ok_type <> 7
      and ok_from_c_id = 0
      and ok_from_k_id = :k_id
    into :trans_fine;

    seq_counter = seq_counter + 1;
    seq_no = seq_counter;

    suspend;
end

end
