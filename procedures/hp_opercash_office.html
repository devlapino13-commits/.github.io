CREATE PROCEDURE HP_OPERCASH_OFFICE(
    OFFICE_ID D_INTEGER,
    CASH_ID D_INTEGER,
    REP_DATE_END D_DATE,
    DOC_STATUS D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    OPC_ID D_INTEGER,
    OPC_PARENT_ID D_INTEGER,
    OPC_DATE D_DATE,
    OPC_DOC_NO D_INTEGER,
    OPC_OFFICE_ID D_INTEGER,
    OPC_FROM_CSP_ID D_INTEGER,
    OPC_OPER_KIND D_SMALLINT,
    OPC_SIGN D_SMALLINT,
    OPC_AMOUNT D_CURRENCY,
    OPC_FROM_C_ID D_ID,
    OPC_IS_IPF D_BOOL,
    OPC_CORR_AL_ID D_STR_20,
    OPC_FROM_K_ID D_ID,
    OPC_CORR_REG_ID D_INTEGER,
    OPC_MEMO D_STR_LONG,
    OPC_IS_ONLINE D_BOOL,
    OPC_FROM_EDL_ID D_ID,
    OPC_STATUS D_SMALLINT,
    OPC_MANAGER_ID D_ID,
    OPC_IS_SIGNED D_BOOL,
    OPC_FROM_OL_ID D_ID,
    OPC_IS_DELETE D_BOOL,
    OPC_FROM_UN_ID D_ID,
    OPC_DATETIME D_DATETIME,
    RES_OFFICE_NAME D_STR_100,
    RES_CORR_OFFICE D_STR_100,
    RES_CLIENT_FIO D_STR_255,
    RES_AMOUNT_PLUS D_CURRENCY,
    RES_AMOUNT_MINUS D_CURRENCY,
    RES_CASH_ACCOUNT D_STR_20,
    RES_CORR_ACCOUNT D_STR_20,
    RES_STATUS_STR D_STR_255,
    RES_USERNAME D_STR_CODE,
    RES_OPER_KIND D_STR_100,
    RES_IS_CHILD D_BOOL,
    RES_CAN_EXECUTE D_BOOL,
    RES_OPER_CODE D_STR_CODE,
    RES_IS_CLOSEDAY D_BOOL,
    RES_BLANK_ID D_INTEGER,
    RES_MANAGER_FIO D_STR_100,
    RES_CAN_EDIT D_BOOL,
    RES_EDIT_KIND D_STR_CODE,
    RES_PASS1_BLS_ID D_ID)
AS
declare variable LAST_OPER_KIND D_INTEGER;
declare variable CSO_ACTIVATE_DATE D_DATE;
declare variable CSO_IS_ACTIVE D_BOOL;
declare variable OPC_OTHER_CLIENT D_STR_255;
declare variable PROCESS_TEXT D_STR_255;
begin
  if (doc_status is null) then
    doc_status = 0;

  if (cash_id is null) then
    cash_id = 0;

  select cso_is_active, cso_activate_date
  from cashoffice
  where cso_office_id = :office_id
  into cso_is_active, cso_activate_date;

  if (cso_is_active = 0 or cso_activate_date is null or (cso_activate_date - 1) > rep_date_end) then
    exception e_message '"Рабочее место кассира" не активирован для данного офиса';

  for select opc_id,
             opc_parent_id,
             opc_date,
             opc_doc_no,
             opc_office_id,
             opc_from_csp_id,
             opc_oper_kind,
             opc_sign,
             opc_amount,
             opc_from_c_id,
             opc_is_ipf,
             opc_corr_al_id,
             opc_from_k_id,
             opc_corr_reg_id,
             opc_memo,
             opc_is_online,
             opc_from_edl_id,
             opc_status,
             opc_manager_id,
             opc_is_signed,
             opc_from_ol_id,
             opc_is_delete,
             opc_from_un_id,
             opc_datetime,
             opc_other_client,
             r1.reg_code || ' ' || r1.reg_name,
             r2.reg_code || ' ' || r2.reg_name,
             cl.c_name_fio,
             iif(opc_is_ipf = 0, csp_cls_al_id, csp_ipf_al_id),
             s.un_name,
             mgr.c_name_fio
      from opercash
           inner join region r1 on (opc_office_id = r1.reg_unicode)
           inner join region r2 on (opc_corr_reg_id = r2.reg_unicode)
           left join clients cl on (opc_from_c_id = cl.c_id)
           left join clients mgr on (opc_manager_id = mgr.c_id)
           inner join cashplace on (opc_from_csp_id = csp_id)
           inner join username s on (opc_from_un_id = s.un_id)
      where opc_office_id = :office_id and
            opc_date = :rep_date_end and
            (:cash_id = 0 or opc_from_csp_id = :cash_id)
      order by opc_id desc
      into :opc_id,
           :opc_parent_id,
           :opc_date,
           :opc_doc_no,
           :opc_office_id,
           :opc_from_csp_id,
           :opc_oper_kind,
           :opc_sign,
           :opc_amount,
           :opc_from_c_id,
           :opc_is_ipf,
           :opc_corr_al_id,
           :opc_from_k_id,
           :opc_corr_reg_id,
           :opc_memo,
           :opc_is_online,
           :opc_from_edl_id,
           :opc_status,
           :opc_manager_id,
           :opc_is_signed,
           :opc_from_ol_id,
           :opc_is_delete,
           :opc_from_un_id,
           :opc_datetime,
           :opc_other_client,
           :res_office_name,
           :res_corr_office,
           :res_client_fio,
           :res_cash_account,
           :res_username,
           :res_manager_fio
  do
  begin
    if (opc_doc_no = 0) then
      opc_doc_no = null;

    if (coalesce(opc_other_client, '') <> '') then
      res_client_fio = trim(res_client_fio || ' через ' || opc_other_client);

    res_is_child = 0;
    if (opc_parent_id is not null) then
      res_is_child = 1;

    res_can_execute = 1;
    if (opc_parent_id is not null or opc_from_ol_id is not null or opc_is_delete = 1) then
      res_can_execute = 0;

    res_can_edit = 1;
    if (opc_parent_id is not null or (opc_from_ol_id is not null and opc_is_delete = 0)) then
      res_can_edit = 0;

    if (opc_status <> 0 and opc_is_online = 1) then
      res_can_edit = 0;

    if (res_corr_office = res_office_name) then
      res_corr_office = '';

    if (coalesce(res_cash_account, '') <> '') then
      res_cash_account = substring(res_cash_account from 8 for 5) || '/' || cast(right(res_cash_account, 6) as integer);

    res_corr_account = '';
    if (coalesce(opc_corr_al_id, '') <> '') then
      res_corr_account = substring(opc_corr_al_id from 8 for 5) || '/' || cast(right(opc_corr_al_id, 6) as integer);
    else
    if (opc_from_k_id is not null) then
      res_corr_account = (select k_kontract_no from kontract where k_id = :opc_from_k_id);

    res_amount_plus = 0;
    res_amount_minus = 0;
    res_is_closeday = 0;

    if (opc_oper_kind is distinct from last_oper_kind) then
    begin
      res_oper_kind = f_enum_name('cashoperkind', opc_oper_kind, 0);
      res_oper_code = f_enum_code('cashoperkind', opc_oper_kind, 0);
      last_oper_kind = opc_oper_kind;
    end

    res_pass1_bls_id = null;
    if (opc_from_c_id is not null and opc_from_ol_id is null and opc_is_delete = 0 and opc_sign = -1) then
    begin
      select c_blob_pass1_id
      from clients
      where c_id = :opc_from_c_id
      into res_pass1_bls_id;
    end

    res_blank_id = 77;
    if (opc_sign = -1) then
      res_blank_id = 80;

    if (res_oper_code = 'check.close') then
    begin
      res_can_edit = 0;
      res_is_closeday = 1;
      res_blank_id = 79;
    end

    if (opc_is_online = 0 and opc_from_ol_id is null) then
      res_blank_id = null;

    if (opc_sign = 1) then
      res_amount_plus = opc_amount;
    else
      res_amount_minus = opc_amount;

    res_status_str = '√';

    if (opc_from_ol_id is null or opc_is_delete = 1) then
    begin
      if (opc_is_delete = 1) then
        res_status_str = 'Удален';
      else
      begin
        res_status_str = case opc_status
          when -1 then 'Отменен'
          when 0 then 'Новый'
          when 1 then 'Подписан'
        end;

        if (opc_status = 0 and opc_from_edl_id is not null) then
          res_status_str = 'В ожидании';

        select cpr_error_text
        from cashprocess
        where cpr_from_opc_id = :opc_id
        into process_text;

        if (row_count = 1) then
        begin
          if (coalesce(process_text, '') = '') then
            res_status_str = 'В очереди проведения';
          else
            res_status_str = 'Ошибка: ' || process_text;
        end
      end
    end else
    if (opc_status = 0) then
      res_status_str = 'В ожидании';

    if (doc_status = 1) then
    begin
      if (opc_is_delete = 1 or opc_status = -1) then
        Continue;
    end else
    if (doc_status = -1) then
    begin
      if (not(opc_is_delete = 1 or opc_status = -1)) then
        Continue;
    end

    res_edit_kind = '';
    if (res_can_edit = 1) then
    begin
      if (res_oper_code = 'plus.from_bank') then
        res_edit_kind = 'from_bank';
      else
      if (res_oper_code = 'minus.loan.disb') then
        res_edit_kind = 'disb';
      else
      if (res_oper_code starting 'plus.loan.' or res_oper_code starting 'minus.loan.') then
        res_edit_kind = 'loan';
      else
      if (res_oper_code in ('minus.to_bank', 'minus.office', 'minus.to_kassa')) then
        res_edit_kind = 'inkassa';
      else
      if (res_oper_code in ('minus.cash', 'plus.cash')) then
        res_edit_kind = 'other';
    end

    if (res_edit_kind = '') then
      res_can_edit = 0;

    -- Остатки при активации "Рабочего места кассира"
    if (rep_date_end < cso_activate_date) then
      res_can_edit = 0;

    suspend;
  end
end
