CREATE PROCEDURE AAREP_PRO_ACTIVE_LOANS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    IS_IPF SMALLINT = 0)
RETURNS (
    REP_OBLAST VARCHAR(255),
    REP_BUS_NAME VARCHAR(255),
    REP_PLAN_OS DECIMAL(18,4),
    REP_FACT_OS DECIMAL(18,4),
    REP_BALANCE DECIMAL(18,4),
    REP_CL_KOLICH INTEGER,
    REP_MIN_DISB DECIMAL(18,4),
    REP_MAX_DISB DECIMAL(18,4),
    REP_K_PERIOD DECIMAL(18,4),
    REP_PERC_ANNUAL_KOEF DECIMAL(18,7),
    REP_PERC_EFFECT_KOEF DECIMAL(18,7),
    REP_DISB_AVG DECIMAL(18,4),
    REP_PLAN_OS1000 DECIMAL(18,4),
    REP_FACT_OS1000 DECIMAL(18,4),
    REP_BALANCE1000 DECIMAL(18,4),
    REP_MIN_DISB1000 DECIMAL(18,4),
    REP_MAX_DISB1000 DECIMAL(18,4),
    REP_DISB_AVG1000 DECIMAL(18,4),
    LKC_BALANCE CURR_TYPE)
AS
declare variable rep_perc_amount_annual double precision;
declare variable rep_perc_amount_effect double precision;
declare variable rep_disb_amount decimal(18,4);
begin
  for with rep_data
      as (select rep_oblast, rep_bus_name, sum(rep_plan_os) as rep_plan_os, sum(rep_fact_os) as rep_fact_os,
                 sum(rep_balance) as rep_balance, sum(kol_cli_fact) as rep_cl_kolich, -- было sum(REP_CL_KOLICH) as REP_CL_KOLICH
                 sum(rep_perc_amount_annual) as rep_perc_amount_annual,
                 sum(rep_perc_amount_effect) as rep_perc_amount_effect, sum(rep_disb_amount) as rep_disb_amount,
                 min(rep_min_disb) as rep_min_disb, max(rep_max_disb) as rep_max_disb,
                 sum(rep_k_period) as rep_k_period, sum(lkc_balance) as lkc_balance
          from aarep_pro_active_loans_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, 1, :is_ipf)
          group by rep_oblast, rep_bus_name)
      select rep_oblast, rep_bus_name, sum(rep_disb_amount), sum(rep_plan_os), sum(rep_fact_os), sum(rep_balance),
             sum(rep_cl_kolich), sum(rep_perc_amount_annual), sum(rep_perc_amount_effect), min(rep_min_disb),
             max(rep_max_disb), sum(rep_k_period), sum(lkc_balance)
      from rep_data
      group by rep_oblast, rep_bus_name
      union
      select ' По стране', rep_bus_name, sum(rep_disb_amount), sum(rep_plan_os), sum(rep_fact_os),
             sum(rep_balance), sum(rep_cl_kolich), sum(rep_perc_amount_annual), sum(rep_perc_amount_effect),
             min(rep_min_disb), max(rep_max_disb), sum(rep_k_period), sum(lkc_balance)
      from rep_data
      group by rep_bus_name
      union
      select rep_oblast, 'х Итого', sum(rep_disb_amount), sum(rep_plan_os), sum(rep_fact_os), sum(rep_balance),
             sum(rep_cl_kolich), sum(rep_perc_amount_annual), sum(rep_perc_amount_effect), min(rep_min_disb),
             max(rep_max_disb), sum(rep_k_period), sum(lkc_balance)
      from rep_data
      group by rep_oblast
      union
      select ' По стране', 'х Итого', sum(rep_disb_amount), sum(rep_plan_os), sum(rep_fact_os),
             sum(rep_balance), sum(rep_cl_kolich), sum(rep_perc_amount_annual), sum(rep_perc_amount_effect),
             min(rep_min_disb), max(rep_max_disb), sum(rep_k_period), sum(lkc_balance)
      from rep_data
      into :rep_oblast, :rep_bus_name, :rep_disb_amount, :rep_plan_os, :rep_fact_os, :rep_balance, :rep_cl_kolich,
           :rep_perc_amount_annual, :rep_perc_amount_effect, :rep_min_disb, :rep_max_disb, :rep_k_period, lkc_balance
  do
  begin
    if (rep_cl_kolich <> 0) then
      rep_k_period = cast(rep_k_period / rep_cl_kolich as decimal(15,1));
    else
      rep_k_period = 0;
    if (rep_min_disb = 999999999999) then
      rep_min_disb = 0;
    if (rep_balance <> 0) then
    begin
      rep_perc_annual_koef = cast(rep_perc_amount_annual / rep_balance as decimal(15,7));
      rep_perc_annual_koef = cast(rep_perc_annual_koef * 100 as decimal(15,2));
      rep_perc_effect_koef = cast(rep_perc_amount_effect / rep_balance as decimal(15,7));
      rep_perc_effect_koef = cast(rep_perc_effect_koef * 100 as decimal(15,2));
    end
    else
    begin
      rep_perc_annual_koef = 0;
      rep_perc_effect_koef = 0;
    end
    if (rep_cl_kolich <> 0) then
      rep_disb_avg = cast(rep_disb_amount / rep_cl_kolich as decimal(15,2));
    else
      rep_disb_avg = 0;

    rep_plan_os1000 = cast(rep_plan_os / 1000 as decimal(15,2));
    rep_fact_os1000 = cast(rep_fact_os / 1000 as decimal(15,2));
    rep_balance1000 = cast(rep_balance / 1000 as decimal(15,2));
    rep_min_disb1000 = cast(rep_min_disb / 1000 as decimal(15,2));
    rep_max_disb1000 = cast(rep_max_disb / 1000 as decimal(15,2));
    rep_disb_avg1000 = cast(rep_disb_avg / 1000 as decimal(15,2));
    suspend;
  end
end
