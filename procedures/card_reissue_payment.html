CREATE PROCEDURE CARD_REISSUE_PAYMENT(
    CAC_ID D_INTEGER,
    ACC_ENTRY BOOLEAN = false)
RETURNS (
    OL_ID D_ID)
AS
declare variable dt_curr d_date;
declare variable cs_id d_integer;
declare variable new_cs_id d_integer;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable c_id d_id;
declare variable un_id d_id;
declare variable reg_id d_integer;
declare variable amount d_currency;
declare variable pay_amount d_currency;
declare variable cst_packet_no d_str_10;
declare variable cst_ticket_no d_str_10;
declare variable non_cash boolean;
declare variable curr_code d_str_lang;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable edl_id d_id;
declare variable is_pincover d_bool;
declare variable reason d_str_255;
begin
  dt_curr = current_date;
  curr_code = '417';
  pay_amount = 350.00;

  for
    select
      cac_card_id,
      cac_from_cca_id,
      cac_from_c_id,
      cac_from_un_id,
      cac_from_reg_id,
      cac_from_cs_id,
      cac_amount,
      rd_name
    from cardaktclient
      inner join refdata
        on rd_id = cac_from_rd_id
    where cac_id = :cac_id
      and cac_status  = f_enum_id('cardaktclient', 'issued_delayed')
  into
    :card_id,
    :cca_id,
    :c_id,
    :un_id,
    :reg_id,
    :cs_id,
    :amount,
    :reason
  do
  begin
    non_cash = amount >= pay_amount;
    if (acc_entry) then
    begin
      -- бух.проводка по комиссии за перевыпуск
      acc_dt = (
        select al_id
        from acc_list
        where al_from_reg_id = :reg_id
          and al_currency = :curr_code
          and al_account = iif(:non_cash, '20001/1', '10001/1'));
      acc_kt = (
        select al_id
        from acc_list
        where al_from_reg_id = :reg_id
          and al_currency = :curr_code
          and al_account = '21114/15');
      insert into operlist (
        ol_from_reg_id,
        ol_oper_type,
        ol_oper_subtype,
        ol_date,
        ol_doc_no,
        ol_doc_kind,
        ol_memo,
        ol_user_id,
        ol_document_client
       )
      values (
        :reg_id,
        1,
        2,
        :dt_curr,
        :cac_id,
        1,
        f_format('Комиссия за перевыпуск карты ID %1; карт.счет № %2; причина: %3', :card_id, :cca_id, :reason),
        :un_id,
        :c_id
      )
      returning ol_id
      into :ol_id;

      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id,
        :curr_code,
        1,
        :acc_dt,
        :acc_kt,
        :pay_amount,
        :pay_amount
      );

      -- формирование эл.документа на подпись
      is_pincover = (
        select
          cem_is_pincover
        from cardlist
          inner join cardembomain
            on cem_id = cal_from_cem_id
        where cal_card_id = :card_id);

      edl_id = (
        select
          edl_id
        from hp_edoc_insert(
          iif(:is_pincover = 1, 'bankcard_pin.mbulak', 'bankcard.mbulak'),
          f_objkind('cardaktclient', 'id'),
          :cac_id,
          :un_id));

      update cardaktclient set
        cac_from_ol_id_com = :ol_id,
        cac_from_edl_id = :edl_id
      where cac_id = :cac_id;
    end

    if (non_cash and not acc_entry) then
    begin
      -- запрос на списание комиссии в МПЦ
      select
        packet_no,
        ticket_no
      from cardsynctrn_create
      into
        :cst_packet_no,
        :cst_ticket_no;

      insert into cardsync(
        cs_parent_id,
        cs_from_csc_id,
        cs_card_id,
        cs_from_cca_id,
        cs_object_kind,
        cs_object_id,
        cs_amount,
        cs_request_json,
        cs_office_id,
        cs_from_k_id,
        cs_from_c_id,
        cs_from_un_id
      )
      select
        cs_id,
        f_get_cardsynccfg('payment_for_card_reissue', 'id'),
        cs_card_id,
        cs_from_cca_id,
        cs_object_kind,
        cs_object_id,
        :pay_amount,
        f_format('{"card_acct": "%1","card_acct_ccy": "KGS","tran_type": "326",', cs_from_cca_id) ||
          f_format('"tran_ccy": "KGS","tran_amount": %1,"tran_pocket": "%2","tran_ticket": "%3","tran_comment": "%4","bank_c": "21","groupc": "01","payment_mode": 2,"tran_date_time": "%5"}',
            round(:pay_amount,2), :cst_packet_no, :cst_ticket_no, 'Комиссия за перевыпуск', f_datetimetostr(current_timestamp)),
        cs_office_id,
        cs_from_k_id,
        cs_from_c_id,
        cs_from_un_id
      from cardsync
      where cs_id = :cs_id
      returning
        cs_id
      into
        :new_cs_id;

      update cardsynctrn set
        cst_from_cs_id = :new_cs_id
      where cst_packet_no = :cst_packet_no
        and cst_ticket_no = :cst_ticket_no;
    end
    suspend;
  end
end
