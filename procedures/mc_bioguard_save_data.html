CREATE PROCEDURE MC_BIOGUARD_SAVE_DATA(
    SESSION_ID D_STR_100 NOT NULL,
    LANGUAGE_CODE D_STR_100,
    JSON_INFO D_BLOB_TEXT NOT NULL,
    IMG_FILE D_BLOB_BINARY NOT NULL)
AS
declare variable EIL_ID D_ID;
declare variable BR_ID D_ID;
declare variable C_ID D_ID;
declare variable IS_DOCUMENT_VALID D_SMALLINT;
declare variable IIN D_STR_20;
declare variable DOB D_DATE;
declare variable GENDER D_STR_10;
declare variable NATIONALITY D_STR_255;
declare variable NATIONALITY_KZ D_STR_255;
declare variable CITIZENSHIP D_STR_255;
declare variable CITIZENSHIP_KZ D_STR_255;
declare variable COUNTRY D_STR_255;
declare variable BIRTH_PLACE D_STR_255;
declare variable DOC_NUMBER D_STR_100;
declare variable DOC_ISSUE_DATE D_DATE;
declare variable DOC_END_DATE D_DATE;
declare variable DOC_ISSUE_ORG_NAME D_STR_255;
declare variable C_NAME_FIO D_STR_255;
declare variable NATION_ID D_ID;
declare variable CITIZEN_ID D_ID;
declare variable PHONE_NUMBER D_STR_20;
declare variable CLIENT_PHOTO_ID D_ID;
declare variable COUNTRY_CODE D_STR_100;
declare variable COUNTRY_TXT D_STR_255;
declare variable DISTRICT_CODE D_STR_20;
declare variable DISTRICT_TXT D_STR_255;
declare variable REGION_CODE D_STR_100;
declare variable REGION_TXT D_STR_255;
declare variable CITY_TXT D_STR_255;
declare variable STREET_TXT D_STR_255;
declare variable BUILDING_TXT D_STR_255;
declare variable FLAT_TXT D_STR_255;
declare variable COUNTRY_CODE_FACT D_STR_100;
declare variable COUNTRY_TXT_FACT D_STR_255;
declare variable DISTRICT_CODE_FACT D_STR_20;
declare variable DISTRICT_TXT_FACT D_STR_255;
declare variable REGION_CODE_FACT D_STR_100;
declare variable REGION_TXT_FACT D_STR_255;
declare variable CITY_TXT_FACT D_STR_255;
declare variable STREET_TXT_FACT D_STR_255;
declare variable BUILDING_TXT_FACT D_STR_255;
declare variable FLAT_TXT_FACT D_STR_255;
declare variable KZ_PASSPORT_CARD_PK_ID D_INTEGER;
declare variable ERROR_CODE D_STR_CODE;
declare variable PHOTO_1_BINARY D_BLOB_BINARY;
declare variable PHOTO_2_BINARY D_BLOB_BINARY;
declare variable DOCRECOG_JSON D_BLOB_TEXT;
declare variable CLIENT_PASS_DOC_NO D_STR_255;
declare variable IS_NEW_CLIENT D_BOOL;
declare variable SURNAME D_STR_100;
declare variable FIRSTNAME D_STR_100;
declare variable PATRONYMIC D_STR_100;
declare variable ENG_SURNAME D_STR_100;
declare variable ENG_FIRSTNAME D_STR_100;
begin
    language_code = coalesce(:language_code, 'ru');
    is_new_client = 0;

    -- Сохраняем ответ об биогарда
    select eil_id, br_id, client_photo_bls_id
    from hp_bioguard_save_response(:session_id, :json_info, :img_file)
    into :eil_id, :br_id, :client_photo_id;

    select first 1
        btd.btd_passport_photo1,
        btd.btd_passport_photo2,
        btd.btd_docrecog_json
    from bioguard_temp_data btd
    where btd.btd_session_id = :session_id
    order by btd.btd_create_datetime desc
    into
        :photo_1_binary,
        :photo_2_binary,
        :docrecog_json;

    -- Получаем номер паспорта, который прикрепил клиент (он распознается библиотекой)
    select field_value
    from hp_parse_docrecog_json(:docrecog_json)
    where field_code = 'number'
    into :client_pass_doc_no;

    select
        iin, dob, gender, nationality, nationality_kz, citizenship, citizenship_kz, country,
        birth_place, doc_number, doc_issue_date, doc_end_date, issue_org_name,
        c_name_fio, is_document_valid, phone_number,
        country_code, country_txt, district_code, district_txt, region_code, region_txt, city_txt, street_txt, building_txt, flat_txt,
        country_code_fact, country_txt_fact, district_code_fact, district_txt_fact, region_code_fact, region_txt_fact, city_txt_fact, street_txt_fact, building_txt_fact, flat_txt_fact,
        surname, firstname, patronymic, eng_surname, eng_firstname
    from hp_parse_json_bioguard(:json_info, :client_pass_doc_no)
    into
        :iin, :dob, :gender, :nationality, :nationality_kz, :citizenship, :citizenship_kz, :country,
        :birth_place, :doc_number, :doc_issue_date,  :doc_end_date, :doc_issue_org_name,
        :c_name_fio, :is_document_valid, :phone_number,
        :country_code, :country_txt, :district_code, :district_txt, :region_code, :region_txt, :city_txt, :street_txt, :building_txt, :flat_txt,
        :country_code_fact, :country_txt_fact, :district_code_fact, :district_txt_fact, :region_code_fact, :region_txt_fact, :city_txt_fact, :street_txt_fact, :building_txt_fact, :flat_txt_fact,
        :surname, :firstname, :patronymic, eng_surname, eng_firstname;

    --  Проверяем документ на валидность
    if (is_document_valid = 0) then
    begin
        -- Присваиваем ответу соответсвующий статус
        error_code = 'BioguardStatus.DOCUMENT_ERROR';
        execute procedure hp_bioguard_update_status(:br_id, :eil_id, :error_code);
        exit;

    end

    -- создание или поиск клиента
    :c_id = (select c.c_id from clients c where c.c_inn = :iin);
    if (coalesce(:c_id, 0) = 0) then
    begin
        nation_id = (select rd_id from hp_get_rd_id_by_name('Nation', :nationality, :nationality_kz));
        citizen_id = (select rd_id from hp_get_rd_id_by_name('Citizen', :citizenship, :citizenship_kz));
    
        -- RFCKG-4496
        select pkk.pk_id
        from passkind pkk
        inner join passdoctype pdt on pkk.pk_from_pdt_id = pdt.pdt_id
        where pdt.pdt_code = 'passport_card'
            and pkk.pk_country = 'kaz'
            and pdt.pdt_is_main_doc = 1
        into :kz_passport_card_pk_id;

        begin
            select c_id
            from fm_add_client(:c_name_fio, iif(:gender = '1', 1, 0), :dob, :country, :birth_place, :nation_id, :citizen_id, :kz_passport_card_pk_id, :doc_number, :doc_issue_date, :iin, 1000000000000, :surname, :firstname, :patronymic, :eng_surname, :eng_firstname, null)
            into :c_id;

            is_new_client = 1;

            when any do
                begin
                    -- Присваиваем ответу соответсвующий статус
                    error_code = 'BioguardStatus.SERVER_ERROR';
                    execute procedure hp_bioguard_update_status(:br_id, :eil_id, :error_code);
                    exit;
                end    
        end 
    end

    begin
        -- RFCKZ-497 Обновляем персональные данные клиента
        error_code = 'BioguardStatus.PERSONAL_DATA_ERROR';
        execute procedure hp_bioguard_save_pers_data(:c_id, :is_new_client, :c_name_fio, :gender, :dob, :country, :birth_place, :nationality, :nationality_kz, :citizenship, :citizenship_kz);

        -- RFCKZ-115 Сохраняем паспортные данные и верифицируем клиента
        error_code = 'BioguardStatus.PASSPORT_ERROR';
        execute procedure hp_bioguard_verify_client(:c_id, :session_id, :doc_number, :doc_issue_date, :doc_end_date, :doc_issue_org_name, :client_photo_id, :photo_1_binary, :photo_2_binary);
    
        -- RFCKG-4463 Добавляем номер телефона
        error_code = 'BioguardStatus.CONTINFO_ERROR';
        execute procedure hp_continfo_force_update(:c_id, 'cell', :phone_number, 2, 1000000000000, current_timestamp);

        -- RFCKG-4493 Добавляем адрес клиента
        error_code = 'BioguardStatus.LEGAL_ADDRESS_ERROR'; -- Юридический адрес
        execute procedure hp_bioguard_save_address(1, :c_id, :country_code, :country_txt, :district_code, :district_txt, :region_code, :region_txt, :city_txt, :street_txt, :building_txt, :flat_txt);

        error_code = 'BioguardStatus.FACT_ADDRESS_ERROR'; -- Фактический адрес
        execute procedure hp_bioguard_save_address(2, :c_id, :country_code_fact, :country_txt_fact, :district_code_fact, :district_txt_fact, :region_code_fact, :region_txt_fact, :city_txt_fact, :street_txt_fact, :building_txt_fact, :flat_txt_fact);


        when any do
            begin
                -- Сохраняем данный ответ, как ошибочный
                execute procedure hp_bioguard_update_status(:br_id, :eil_id, :error_code);
                exit;
    
            end    
    end

    -- RFCKZ-206
    -- Если во время всего процесса не было ошибок, тогда сохраняем результат ответа, как успешный
    execute procedure hp_bioguard_update_status(:br_id, :eil_id, 'BioguardStatus.SUCCESS', :c_id);
end
