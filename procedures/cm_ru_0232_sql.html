CREATE PROCEDURE CM_RU_0232_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable disb_date d_date;
declare variable kolich_repay d_integer;
declare variable c_id d_id;
declare variable point d_integer;
declare variable pr_id d_integer;
declare variable amount_max d_currency;
declare variable amount d_currency;
declare variable cli_balance d_currency;
declare variable ma_refund d_integer;
declare variable refund_balance d_currency;
declare variable limit_max d_currency;
declare variable refund_balance_one d_currency;
declare variable authorized_amount d_currency;
declare variable mds d_currency;
declare variable pr_name d_str_code;
declare variable pr_code str_100;
begin
  limit_max = 200000;
  -- проверка по продукту Залоговый
  for
  select ma_date_oper, ma_kolich_repay, pr_id, ma_refund, pr_code
  from miniapp, product
  where ma_id = :ma_id
  and ma_is_active = 1
  and ma_date_end = '31.12.9999'
  and ma_from_pr_id = pr_id
  and (pr_code containing '.selfemployed.' or
       pr_code containing '.business.')
  into disb_date, kolich_repay, pr_id, ma_refund, pr_code
  do
  begin
    if (pr_code containing '.selfemployed_plus.') then
      exit;

    -- если имеется активный займ то запрещено брать залоговый
    select mac_from_c_id, mac_amount from miniappclient
    where mac_from_ma_id = :ma_id
    and mac_amount > 0
    rows 1
    into c_id, amount;

    point = f_point_get(11, ma_id, c_id, disb_date);

    if (exists(select * from kontstate, knclient, kontract, product
    where ks_ol_is_delete = 0
    and ks_state = 1
    and :disb_date between ks_date_begin and ks_date_end
    and ks_from_k_id = kc_from_k_id
    and kc_amount_approved > 0
    and kc_from_c_id = :c_id
    and ks_from_k_id = k_id
    and k_from_pr_id = pr_id
    and (pr_code containing '.education.' or pr_code containing '.zalog_car.' /* or pr_code containing '.zalog.'*/)
    )) then
    begin
      res_error_text = 'Имеется активный займ по залоговому или по образованию, выдача запрещается!';
      res_error_is_critical = 2;
      suspend;
    end

    cli_balance = 0;
    select sum(oks_balance)
    from knclient, oksummary
    where kc_from_c_id  = :c_id
    and kc_amount_approved > 0
    and kc_from_k_id = oks_from_k_id
    and oks_balance > 0
    and oks_close_state = 0
    and :disb_date between oks_date_begin and oks_date_end
    into cli_balance;
    cli_balance = coalesce(cli_balance, 0);

    for
    select amount_max
    from perc_amount_interval_by_point(:pr_id, :point)
    into :amount_max
    do
    begin
      select mds
      from hp_max_amount_allowed(11, :ma_id, :amount_max)
      into mds;

      if ((cli_balance + amount > amount_max) and
          ma_refund = 0 and current_date < '23.01.2025') then
      begin
        amount_max = amount_max - cli_balance;
        res_error_text = f_format('Баланс ОС по активным: %1 р. Разрешенная сумма для выдачи: %2, Бал кл: %3', round(cli_balance), round(amount_max), point);
        res_error_is_critical = 2;
        suspend;
      end
      if ((amount > mds) and
          ma_refund = 0 and current_date >= '23.01.2025') then
      begin
        amount_max = amount_max - cli_balance;
        res_error_text = f_format('Баланс ОС по активным: %1 р. Разрешенная сумма для выдачи: %2, Бал кл: %3', round(cli_balance), round(mds), point);
        res_error_is_critical = 2;
        suspend;
      end

      if (ma_refund = 1) then
      begin
        refund_balance = 0;
        select sum(oks_balance)
        from knclient, oksummary, kontract, product
        where kc_from_c_id = :c_id and
              kc_amount_approved > 0 and
              kc_from_k_id = oks_from_k_id and
              k_id = oks_from_k_id and
              k_from_pr_id = pr_id and
              pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.|.shans.)%' and
              oks_balance > 0 and
              oks_close_state = 0 and
              :disb_date between oks_date_begin and oks_date_end
        into refund_balance;
        refund_balance = coalesce(refund_balance, 0);


        refund_balance_one = 0;
        select oks_balance
        from knclient, oksummary, kontract, product
        where kc_from_c_id = :c_id and
              kc_amount_approved > 0 and
              kc_from_k_id = oks_from_k_id and
              k_id = oks_from_k_id and
              k_from_pr_id = pr_id and
              pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.|.shans.)%' and
              oks_balance > 0 and
              oks_close_state = 0 and
              :disb_date between oks_date_begin and oks_date_end
        order by k_disb_date
        rows 1
        into refund_balance_one;
        refund_balance_one = coalesce(refund_balance_one, 0);

        limit_max = amount_max;
        if ((cli_balance + amount - refund_balance_one) > limit_max) then
        begin
          amount_max = amount_max - cli_balance + refund_balance_one;
--          authorized_amount = iif(limit_max - cli_balance < amount_max, limit_max - cli_balance, amount_max);
          authorized_amount = amount_max;

--          res_error_text = 'Общий баланс всех займов превысит 200 000 р., разрешенная сумма для рефинансирования ' ||round(authorized_amount)|| ' р.';
          res_error_text = f_format('Баланс ОС по активным: %1 р. Разрешенная сумма для выдачи: %2, Бал кл: %3', round(cli_balance), round(amount_max), point);
          res_error_is_critical = 2;
          suspend;
        end

        if (((cli_balance - refund_balance) + amount > (amount_max)) and (cli_balance + amount) <= limit_max) then
        begin
          amount_max = amount_max - cli_balance + refund_balance;
          res_error_text = f_format('Баланс ОС по активным: %1 р. Разрешенная сумма для выдачи: %2, Бал кл: %3', round(cli_balance), round(amount_max), point);
          res_error_is_critical = 2;
          suspend;
        end
      end
    end
  end

  -- Если у клиента имеется активный залоговый займ то запрещается другим выдачам
  for
  select ma_date_oper, ma_kolich_repay, mac_from_c_id from miniapp, product, miniappclient
  where ma_id = :ma_id
  and ma_is_active = 1
  and ma_date_end = '31.12.9999'
  and ma_from_pr_id = pr_id
  and (pr_code containing '.education.' or pr_code containing '.ZALOG_CAR.' /* or pr_code containing '.zalog.'
      /* or pr_code containing '.selfemployed.' */)
  and ma_id = mac_from_ma_id
  and mac_amount > 0
  into disb_date, kolich_repay, c_id
  do
  begin
    pr_name = null;

    select pr_name
    from kontstate, knclient, kontract, product
    where ks_ol_is_delete = 0
    and ks_state = 1
    and :disb_date between ks_date_begin and ks_date_end
    and ks_from_k_id = kc_from_k_id
    and kc_amount_approved > 0
    and kc_from_c_id = :c_id
    and ks_from_k_id = k_id
    and k_from_pr_id = pr_id
    and (pr_code containing '.selfemployed.' or
         pr_code containing '.business.')
    rows 1
    into pr_name;

    if (pr_name is not null) then
    begin
      res_error_text = 'Имеется активный ' || pr_name || ' займ, выдача запрещена!';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
