          select distinct rm_id, rm_date, rm_from_c_id, rl_corr_office, rm_status, rm_from_rk_id, rm_no, c_name_fio,
                          rk_text, reg_name
          from reqmain, clients, reqlist, reqsign, region_list(:report_region, 1, -1, 1), reqkind, region
          where rm_from_c_id = c_id and
                rm_id = rl_from_rm_id and
                (rl_corr_office = id_list) and
                rl_corr_office = reg_unicode and
                rm_from_rk_id = rk_id and
                rm_from_rk_id = 0 and
                rs_from_rm_id = rm_id and
                ((:report_status = -2) or
                (rm_status = :report_status)) and
                ((:report_no = -1) or
                (rm_no = :report_no)) and
                (rs_datetime >= :report_date_begin) and
                (rs_datetime <= :report_date_end)

          into :rm_id, :rm_date, :rm_from_c_id, :rm_from_reg_id, :rm_status, :rm_from_rk_id, :rm_no, :rm_c_name_fio,
               :rm_rk_text, :rm_region_name
      do
      begin
        if (rm_status = -1) then
          rm_status_text = 'Аннулирован';
        if (rm_status = 0) then
          rm_status_text = 'В процессе';
        if (rm_status = 1) then
          rm_status_text = 'Утвержден';
        if (rm_status = 2) then
          rm_status_text = 'Отказано';
        if (rm_status = 3) then
          rm_status_text = 'Исполнен';
        if (rm_status = 4) then
          rm_status_text = 'Проверен';
        if (rm_status = 5) then
          rm_status_text = 'Согласован';
        suspend;
      end
    end
  end
