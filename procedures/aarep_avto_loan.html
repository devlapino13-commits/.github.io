CREATE PROCEDURE AAREP_AVTO_LOAN(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KAL_NEW_KF_NAME STR_100,
    K_KONTRACT_NO STR_20,
    K_DISB_DATE DATE,
    KS_STATE STR_10,
    KS_DATE_BEGIN DATE)
AS
begin
  for select kal.kal_new_kf_name, k.k_kontract_no, k.k_disb_date,
             case ks.ks_state
               when 1 then 'Выдан'
               when 2 then 'Закрыт'
             end,
             ks.ks_date_begin
      from kontract k
      inner join product pr on k.k_from_pr_id = pr.pr_id
      inner join knclient kc on kc.kc_from_k_id = k.k_id
      inner join kontstate ks on ks.ks_from_k_id = k.k_id
      inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
      where (pr.pr_code containing '.ZALOG_CAR.' or pr.pr_code containing '.zalog.') and
            kc.kc_amount_approved > 0 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state in (1, 2) and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_date_begin between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into kal_new_kf_name, k_kontract_no, k_disb_date, ks_state, ks_date_begin
  do
    suspend;
end
