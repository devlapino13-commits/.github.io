CREATE PROCEDURE AAREP_ACT_BU_NEW(
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO STR_20,
    KC_ID ID_TYPE,
    DISB_DATE DATE,
    DISB_MFO STR_10,
    OKS_BALANCE CURR_TYPE,
    OKS_BAL_PERC CURR_TYPE,
    OKS_DIFF_ESP_BAL CURR_TYPE,
    DELINQ_DAY INTEGER,
    BN_POSLE_MFO_12 CURR_TYPE,
    OKS_ECL_VBS CURR_TYPE,
    OX_AMOUNT_ZADOL CURR_TYPE,
    OKS_LLR_OS CURR_TYPE,
    OKS_LLR_PERC CURR_TYPE,
    OKS_DIFF_ECL_LLR CURR_TYPE,
    OX_AMOUNT_RO CURR_TYPE,
    OKS_ECL_PERCENT PERC_TYPE,
    BN_DO_MFO_10 CURR_TYPE,
    OKS_KONT_LAST_DATE DATE,
    LAST_REPAY_DATE DATE,
    OKS_DELINQ_DATE DATE,
    OKS_DELINQ_DAYS INTEGER,
    OKS_DELINQ_DATE_365 DATE,
    LAST_REPAY_DATE_DEY INTEGER,
    PR_NAME VARCHAR(100),
    CLIENT_TYPE D_STR_20)
AS
declare variable k_id id_type;
declare variable k_id_prior id_type;
declare variable rep_date_prv date;
declare variable k_base_k_id id_type;
declare variable k_base_k_id_prior id_type;
declare variable rep_date_begin date;
declare variable oks_delinq_date_tmp date;
declare variable sum1 curr_type;
declare variable oks_date_begin date;
declare variable count_d integer;
declare variable coun_dey integer;
declare variable k_id_tmp id_type;
declare variable date1 date;
declare variable oks_llr_os_new curr_type;
declare variable oks_llr_perc_new curr_type;
declare variable oks_diff_ecl_llr_new curr_type;
declare variable pr_code d_str_100;
begin
  k_id_prior = 0;
  k_base_k_id_prior = 0;
  last_repay_date_dey = 0;
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;

  for with tabl1 as (
        select oks_from_k_id,
               oks_from_kc_id,
               oks_disb_date,
               oks_kont_last_date,
               oks_balance - oks_off_os as oks_balance,
               oks_bal_perc - oks_off_perc as oks_bal_perc,
               iif(oks_close_state = 0, oks_diff_esp_bal, iif(oks_close_state = 1 and (oks_off_os < 0 or oks_off_perc < 0), oks_bal_esp_end, 0)) as oks_diff_esp_bal,
               iif(oks_delinq_date is not null, oks_delinq_date, oks_delinq_date_prv) as oks_delinq_date_tmp,
               iif(oks_close_state = 0, oks_ecl_vbs, 0) - oks_off_os - oks_off_perc + iif(oks_close_state = 1 and (oks_off_os < 0 or oks_off_perc < 0), oks_bal_esp_end, 0) as oks_ecl_vbs,
               oks_llr_os - iif(oks_disb_date >= '28.02.2012', oks_off_os, 0) as oks_llr_os,
               oks_llr_perc - iif(oks_disb_date >= '28.02.2012', oks_off_perc, 0) as oks_llr_perc,
               iif(oks_is_month = 1 and oks_close_state = 0, oks_diff_ecl_llr, 0)
                 - iif(oks_disb_date < '28.02.2012', oks_off_os + oks_off_perc, 0)
                 + iif(oks_close_state = 1 and (oks_off_os < 0 or oks_off_perc < 0), oks_bal_esp_end, 0) oks_diff_ecl_llr,
               oks_ecl_percent,
               -iif(oks_disb_date < '28.02.2012', oks_off_os + oks_off_perc, 0) as bn_do_mfo_10,
               -iif(oks_disb_date >= '28.02.2012', oks_off_os + oks_off_perc, 0) as bn_posle_mfo_12,
               0 as ox_amount_zadol, --"Задолж.госпош.",
               0 as ox_amount_ro, --"Резерв.госпош.",
               oks_delinq_days,
               oks_date_begin,

               oks_llr_os - oks_off_os as oks_llr_os_new,
               oks_llr_perc - oks_off_perc as oks_llr_perc_new,
               iif(oks_is_month = 1 and oks_close_state = 0, oks_diff_ecl_llr, 0)
--                 - iif(oks_disb_date < '28.02.2012', oks_off_os + oks_off_perc, 0)
                 + iif(oks_close_state = 1 and (oks_off_os < 0 or oks_off_perc < 0), oks_bal_esp_end, 0) oks_diff_ecl_llr_new

        from kn_oksummary
        where oks_date_begin = :rep_date_end and
              (oks_close_state = 0 or
              (oks_close_state = 1 and
              (oks_off_os < 0 or
              oks_off_perc < 0)) or
              (oks_close_state = 2 and
              oks_bal_law > 0)) and
              oks_from_k_id not in (101010025010600201, 101010025779700201, 101010024808600201)
        union all
        select oks_from_k_id,
               oks_from_kc_id,
               oks_disb_date,
               oks_kont_last_date,
               null,
               null,
               null,
               null,
               null,
               null,
               null,
               null,
               null,
               null,
               null,
               iif(oks_is_month = 1, oks_bal_law, 0) + oks_off_law,-- as "Задолж.госпош.",
               iif(oks_is_month = 1, oks_reserv_law, 0) + oks_off_law,-- as "Резерв.госпош.",
               null,
               oks_date_begin,
               null, 
               null,
               null
        from kn_oksummary
        where :rep_date_end between oks_date_begin and oks_date_end
        and (oks_bal_law > 0 or oks_reserv_law > 0 or oks_off_law > 0)
        )
      select oks_from_k_id,
             oks_from_kc_id,
             oks_disb_date,
             oks_kont_last_date,
             max(oks_balance),
             max(oks_bal_perc),
             max(oks_diff_esp_bal),
             max(oks_delinq_date_tmp),
             max(oks_ecl_vbs),
             max(oks_llr_os),
             max(oks_llr_perc),
             max(oks_diff_ecl_llr),
             max(oks_ecl_percent),
             max(bn_do_mfo_10),
             max(bn_posle_mfo_12),
             max(ox_amount_zadol),
             max(ox_amount_ro),
             max(oks_delinq_days),
             oks_date_begin,

             max(oks_llr_os_new),
             max(oks_llr_perc_new),
             max(oks_diff_ecl_llr_new)
      from tabl1
      group by oks_from_k_id, oks_from_kc_id, oks_disb_date, oks_kont_last_date, oks_date_begin
      order by 1
      --rows 10
      into k_id,
           kc_id,
           disb_date,
           oks_kont_last_date,
           oks_balance,
           oks_bal_perc, -- 7. Задолженность по начисленным процентам, руб. коп.
           oks_diff_esp_bal, --  8. Корректировка до ВБС по методу ЭСП, руб. коп.
           oks_delinq_date_tmp,
           oks_ecl_vbs, -- 13. Сформированный РО, руб. коп.
           oks_llr_os, -- 15. Сумма сформированного РВПЗ по ОД, руб. коп.
           oks_llr_perc, -- 16. Сумма сформированного РВПЗ по %%, руб. коп.
           oks_diff_ecl_llr, -- 17. Сумма сформированной корректировки до РО, руб. коп.
           oks_ecl_percent, -- Ставка РО
           bn_do_mfo_10,
           bn_posle_mfo_12,
           ox_amount_zadol,
           ox_amount_ro,
           oks_delinq_days,
           oks_date_begin,

           oks_llr_os_new,
           oks_llr_perc_new,
           oks_diff_ecl_llr_new
  do
  begin
    if (disb_date  < '28.02.2012' and rep_date_end >= '31.10.2020') then
    begin
      oks_llr_os = oks_llr_os_new;
      oks_llr_perc = oks_llr_perc_new;
      oks_diff_ecl_llr = oks_diff_ecl_llr_new;
    end 

    if (k_id_prior <> k_id) then
    begin
      k_id_prior = k_id;
      select k_kontract_no, k_base_k_id, pr_name, pr_code
      from kontract
      inner join product on pr_id = k_from_pr_id
      where k_id = :k_id
      into k_kontract_no, k_base_k_id, pr_name, pr_code;

      if (k_base_k_id is not null) then
      begin
        --      последняя дата оплаты и кол-во дней просрочки определяются с учетом слэшей. Т.е. амнистированные договоры со слэшами рассматриваются как отдельные договоры.
        --      Это неверно. Договор со слэшем – часть исходного договора и определять максимальное кол-во дней просрочки и самую позднюю дату погашения нужно по всем созаемщикам группы,
        --      которые были в исходном договоре, последующая история амнистий и выделений игнорируется.

        if (coalesce(k_base_k_id_prior, 0) <> coalesce(k_base_k_id, 0)) then
        begin
          k_base_k_id_prior = k_base_k_id;

          last_repay_date = null;
          select ok_ol_date
          from operkred, kontract, oksummary
          where ok_ol_is_active = 1 and
                ok_type = 4 and
                ok_from_c_id = 0 and
                ok_ol_date <= :rep_date_end and
                ok_os + ok_perc + ok_fine < 0 and
                ok_from_k_id = k_id and
                k_base_k_id = :k_base_k_id and
                k_id = oks_from_k_id and
                :rep_date_begin between oks_date_begin and oks_date_end and
                oks_close_state = 0
          order by ok_ol_date desc
          rows 1
          into :last_repay_date;

          select min(iif(oks_delinq_date is not null, oks_delinq_date, oks_delinq_date_prv))
          from kn_oksummary, kontract
          where oks_date_begin = :rep_date_end and
                (oks_close_state = 0 or
                (oks_close_state = 1 and
                (oks_off_os < 0 or
                oks_off_perc < 0)) or
                (oks_close_state = 2 and
                oks_bal_law > 0)) and
                oks_from_k_id = k_id and
                k_base_k_id = :k_base_k_id
          into oks_delinq_date;

        end
      end
      else
      begin
        last_repay_date = null;
        select ok_ol_date
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_from_c_id = 0 and
              ok_ol_date <= :rep_date_end and
              ok_os + ok_perc + ok_fine < 0
        order by ok_ol_date desc
        rows 1
        into :last_repay_date;
        oks_delinq_date = oks_delinq_date_tmp;
      end

      delinq_day = rep_date_end - oks_delinq_date;
      if (delinq_day < 0) then
        delinq_day = null;
    end

    client_type = 'ФЛ';
    if (pr_code containing '.selfemployed.') then
      client_type = 'СЗ';

    --  Последняя дата оплаты» (столбец 5 по договору) < (дата отчета минус 180 дней
    bn_posle_mfo_12 = 0;
    if (disb_date >= '28.02.2012' and
        coalesce(delinq_day, 0) > 365 and
        coalesce(last_repay_date, disb_date) < rep_date_end - 180) then
    begin
      bn_posle_mfo_12 = oks_balance + oks_bal_perc;
    end

    --    bn_do_mfo_10 = 0; -- 10. с истекшим сроком исковой давности (займы "до реестра"), руб. коп. -- ПОКА НЕ БУДЕМ ФОРМИРОВАТЬ, ПОДТЯГИВАЕТСЯ ИЗ ПРИЛОЖЕНИЯ 1, ДОЛГО ГОТОВИТЬСЯ
    --    bn_do_mfo_11 = 0; -- 11. по которой получено судебное решение о невозможности взыскания, руб. коп.
    --    bn_posle_mfo_12 = 0; -- 12. по которой количество дней просрочки превысило 1 год (займы "после реестра")

    if (disb_date >= '28.02.2012') then
    begin
      disb_mfo = 'После';
      --      if (delinq_day > 365) then bn_posle_mfo_12 = oks_balance +  oks_bal_perc;
    end
    else
    begin
      disb_mfo = 'До';

      --      if (dateadd(year, -3,:rep_date_end) > oks_delinq_date) then bn_do_mfo_10 = oks_balance +  oks_bal_perc;
      --      bn_do_mfo_11 = oks_balance +  oks_bal_perc;
    end
    /*
    ox_amount_zadol = null;
    select sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff)
    from kn_operextra
    where ox_ol_is_active = 1
    and ox_from_kc_id = :kc_id
    and ox_ol_date <= :rep_date_end
    into ox_amount_zadol;
    if (ox_amount_zadol is null) then ox_amount_zadol = 0;

    ox_amount_ro = ox_amount_zadol * oks_ecl_percent / 100;  -- Сумма РО
    */

    -- Внесена дороботка по задаче на Jira BOARD-2664
    oks_delinq_date_365 = null;
    coun_dey = 0;
    date1 = null;
    count_d = 0;
    select kn.oks_date_begin, kn.oks_delinq_days
    from kn_oksummary kn
    where oks_date_begin <= :rep_date_end and
          kn.oks_from_k_id = :k_id and
          kn.oks_delinq_days >= 365
    order by oks_date_begin asc
    rows 1
    into :date1, count_d;

    if (count_d > 365) then
      coun_dey = count_d - 365;
    else
      coun_dey = count_d;

    oks_delinq_date_365 = date1 - coun_dey; --Дата достижения просрочкой возраста 365 дней (первый раз)

    if (oks_delinq_date_365 is null) then
    begin
      if (k_base_k_id is not null) then k_id_tmp = k_base_k_id;
      else k_id_tmp = k_id;

      oks_delinq_date_365 = null;
      coun_dey = 0;
      date1 = null;
      count_d = 0;

      select oks_date_begin, oks_date_begin - iif(oks_delinq_date is not null, oks_delinq_date, oks_delinq_date_prv)
      from kn_oksummary
      where oks_date_begin <= :rep_date_end and
            oks_from_k_id = :k_id_tmp and
            oks_date_begin - iif(oks_delinq_date is not null, oks_delinq_date, oks_delinq_date_prv) >= 365
      order by oks_date_begin asc
      rows 1
      into :date1, count_d;
  
      if (count_d > 365) then
        coun_dey = count_d - 365;
      else
        coun_dey = count_d;

      oks_delinq_date_365 = date1 - coun_dey;
    end 

    last_repay_date_dey = rep_date_end - last_repay_date; --кол-во дней с даты последнего погашения к дате акта

    oks_delinq_days = delinq_day; -- нужно брать просроченные дни закрытых кредитов тоже т.е. тут нужно вычислять по старому алгаритму, добавлено от 17.02.2021 Улан



    suspend;
  end
end
