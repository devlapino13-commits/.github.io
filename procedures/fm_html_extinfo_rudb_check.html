CREATE PROCEDURE FM_HTML_EXTINFO_RUDB_CHECK(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable lang d_str_lang;
declare variable eil_date d_date;
declare variable bls_id d_id;
declare variable c_name_fio d_str_255;
declare variable calc_status d_str_255;
declare variable c_birth_date d_date;
declare variable json d_blob_text;
declare variable c_inn d_str_100;
declare variable c_pass_no d_str_20;
declare variable c_pass_date d_date;
declare variable category d_str_64;
declare variable qty_kred d_integer;
declare variable qty_active d_integer;
declare variable sum_ostatok d_currency;
declare variable qty_delinq d_integer;
declare variable state_str d_str_20;
declare variable sum_delinq_ostatok d_currency;
declare variable ox_amount_plus d_currency;
declare variable max_delinq_day d_integer;
begin
  lang = f_get_lang_code(un_id, null);

  select eil_date, eil_from_bls_id
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id and
        eil_from_eic_id = eic_id and
        eic_code = 'check_in_ru_db' and
        eil_is_json = 1 and
        eil_status = 2
  order by eil_date desc, eil_id
  rows 1
  into eil_date, bls_id;

  c_name_fio = f_lang(lang, 'Клиент не найден');
  if (row_count = 0) then
    calc_status = f_lang(lang, 'Информация о проверке в РФ БД не получена');
  else
  begin
    if (coalesce(bls_id, 0) <= 0) then
      calc_status = f_lang(lang, 'Информация о проверке в РФ БД по клиенту отсутствует.');
    else
    begin
      calc_status = f_format('<br>%1: %2', f_lang(lang, 'Дата проверки'), f_datetostr(eil_date));
    end
  end

  select c_name_fio, c_inn, c_pass_date, c_pass_no, c_birth_date
  from clients
  where c_id = :c_id
  into c_name_fio, c_inn, c_pass_date, c_pass_no, c_birth_date;
  res_html = (select res_style
              from fm_html_styles) || f_format('<body><a name="top"></a><h2>%1</h2>%2<br>', c_name_fio, calc_status);

  select cast(res_blob as d_blob_text)
  from fm_blob_get(:bls_id)
  into :json;
  if (coalesce(:json, '') <> '' and
      :json <> '{}') then
  begin
    category = f_json_blob_get_value(:json, 'category', '');
    qty_kred = f_json_blob_get_value(:json, 'qty_kred', 0);
    qty_active = f_json_blob_get_value(:json, 'qty_active', 0);
    sum_ostatok = f_json_blob_get_value(:json, 'sum_ostatok', 0);
    qty_delinq = f_json_blob_get_value(:json, 'qty_delinq', 0);
    state_str = f_json_blob_get_value(:json, 'state_str', '');
    sum_delinq_ostatok = f_json_blob_get_value(:json, 'sum_delinq_ostatok', 0);
    ox_amount_plus = f_json_blob_get_value(:json, 'ox_amount_plus', 0);
    max_delinq_day = f_json_blob_get_value(:json, 'max_delinq_day', 0);
    res_html = blob_append(res_html, '<br><h1>' || f_lang(lang, 'Информация о проверке клиента в РФ БД') || '<br></h1>');
    res_html = blob_append(res_html, f_format('<h3>%1</h3><table width="100%"><col width="40%"><col width="60%">', ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Ф.И.О.') || '</b>', :c_name_fio, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Дата рождения') || '</b>', f_datetostr(:c_birth_date), '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, '№ паспорта') || '</b>', :c_pass_no, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Дата выдачи паспорта') || '</b>', f_datetostr(:c_pass_date), '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Категория') || '</b>', :category, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Всего получено') || '</b>', :qty_kred, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Из них активных') || '</b>', :qty_active, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Остаток суммы для закрытия по активному займу') || '</b>', round(:sum_ostatok, 2), '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'В просрочке') || '</b>', :qty_delinq, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Статус займа в просрочке') || '</b>', :state_str, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Остаток суммы для закрытия по займу в просрочке') || '</b>', round(:sum_delinq_ostatok, 2), '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Сумма задолженности судебных издержек') || '</b>', round(:ox_amount_plus, 2), '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Макс. кол-во просроченных дней') || '</b>', :max_delinq_day, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, '</table>');
  end
  else
  begin
    res_html = blob_append(res_html, '<br>' || f_lang(lang, 'Информация о клиенте не найдена') || '</br>');
    suspend;
    exit;
  end
  suspend;
end
