CREATE PROCEDURE FM_HTML_DEPORT_INFO(
    OBJ_ID D_ID,
    OBJ_KIND D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable lang d_str_lang;
declare variable documen_state d_str_20;
declare variable res_link d_str_2048;
declare variable obj_kind_str d_str_code;
declare variable deport_reason d_integer;
declare variable deport_start_date d_date;
declare variable deport_finish_date d_date;
declare variable deport_comment d_str_long;
declare variable is_edit d_bool;
declare variable is_empty d_bool;
declare variable doc_code d_str_100;
declare variable cds_id d_id;
begin
  lang = f_get_lang_code(un_id, null);
  documen_state = 'Не подтверждено';
  res_link = '';
  is_edit = 0;
  is_empty = 0;
  deport_reason = coalesce(f_enum_id('deport_reason', 'deport'), 1);
  deport_start_date = current_date;
  deport_finish_date = current_date;
  deport_comment = '';

  res_html = blob_append(res_html, (select res_style
                                    from fm_html_styles));

  res_html = blob_append(res_html, '<body>');

  obj_kind_str = f_objkind(obj_kind, 'code');

  if (coalesce(obj_kind_str, '') <> 'clients') then
  begin
    exception e_default(f_format('OBJ_KIND must be CLIENTS in FM_HTML_DEPORT_INFO obj_kind = %1, obj_id = %2, un_id = %3',
                                  coalesce(obj_kind, ''), coalesce(obj_id, ''), coalesce(un_id, '')));
  end
  if (coalesce(obj_id, 0) = 0) then
  begin
    exception e_default(f_format('OBJ_ID cannot be NULL in FM_HTML_DEPORT_INFO obj_kind = %1, obj_id = %2, un_id = %3',
                                  coalesce(obj_kind, ''), coalesce(obj_id, ''), coalesce(un_id, '')));
  end

  if (exists(select 1
             from deport_status_list dsl
             where dsl.dsl_from_c_id = :obj_id and
                   dsl.dsl_record_status = 1)) then
  begin
  -- Если есть запись в процессе редактирования
    is_edit = 1;

    select dsl.dsl_deport_reason,
           dsl.dsl_deport_start_date,
           dsl.dsl_deport_finish_date,
           dsl.dsl_deport_comment
    from deport_status_list dsl
    where dsl.dsl_from_c_id = :obj_id and
          dsl.dsl_record_status = 1
    into deport_reason,
         deport_start_date,
         deport_finish_date,
         deport_comment;
  end
  else if (exists(select 1
             from deport_status_list dsl
             where dsl.dsl_from_c_id = :obj_id and
                   dsl.dsl_record_status = 2)) then
  begin
  -- Если есть сохраненная (актуальная) запись

    select dsl.dsl_deport_reason,
           dsl.dsl_deport_start_date,
           dsl.dsl_deport_finish_date,
           dsl.dsl_deport_comment
    from deport_status_list dsl
    where dsl.dsl_from_c_id = :obj_id and
          dsl.dsl_record_status = 2
    into deport_reason,
         deport_start_date,
         deport_finish_date,
         deport_comment;

    select iif(status = 1, 'Подтверждено', 'Не подтверждено')
    from hp_deprot_status(:obj_id)
    into documen_state;
  end
  else
    is_empty = 1;

  res_html = blob_append(res_html, f_format('<h3 style="color: %1;">%2</h3>', iif(is_edit = 1, 'orange', 'green' ), iif(is_empty = 1, 'Нет сохраненных сведений', iif(is_edit = 1, 'Процесс редактирования', 'Изменения сохранены'))));
  res_html = blob_append(res_html, '<table width="100%"><col width="30%"><col width="30%"><col width="30%">');
  -- Причина выезда-------------------------------------------------------------
  res_link = f_uri_sql(f_lang(lang, 'Изменить'), 'change_deport_reason', obj_kind_str, obj_id, 'combo', f_lang(lang,'Хотите изменить причину выезда?'), '', 0, 0, deport_reason, '', 'dialog', 'null');

  res_html = blob_append(res_html, f_html_table_row(f_lang(lang, 'Причина выезда'), coalesce(f_enum_name('deport_reason', deport_reason), ''), res_link, '', -1, -1, -1, -1, 3, ''));

  -- Дата выезда из РФ----------------------------------------------------------
  res_link = f_uri_sql(f_lang(lang, 'Изменить'), 'change_deport_start_date', obj_kind_str, obj_id, 'date', f_lang(lang, 'Хотите изменить дату выезда?'), '', 4, 355, f_datetostr(deport_start_date), f_lang(lang, 'Укажите новую дату выезда'), 'dialog', 'null');

  res_html = blob_append(res_html, f_html_table_row(f_lang(lang, 'Дата выезда из РФ'), f_datetostr(deport_start_date), res_link, '', -1, -1, -1, -1, 3, ''));

  -- Дата возвращения в РФ------------------------------------------------------
  res_link = f_uri_sql(f_lang(lang, 'Изменить'), 'change_deport_finish_date', obj_kind_str, obj_id, 'date', f_lang(lang, 'Хотите изменить дату выезда?'), '', 4, 355, f_datetostr(deport_finish_date), f_lang(lang, 'Укажите новую дату выезда'), 'dialog', 'null');

  res_html = blob_append(res_html, f_html_table_row(f_lang(lang, 'Дата возвращения в РФ'), f_datetostr(deport_finish_date), res_link, '', -1, -1, -1, -1, 3, ''));

  -- Подтверждение документами--------------------------------------------------
  res_html = blob_append(res_html, f_html_table_row(f_lang(lang, 'Подтверждение документами'), documen_state, '', '', -1, -1, -1, -1, 3, ''));

  -- Комментарий----------------------------------------------------------------
  res_link = f_uri_sql(f_lang(lang, 'Изменить'), 'change_deport_comment', obj_kind_str, obj_id, 'text', f_lang(lang, 'Вы действительно хотите изменить комментарий?'), '', 4, 355, left(coalesce(deport_comment, ''), 100), '', 'dialog', 'null');

  res_html = blob_append(res_html, f_html_table_row(f_lang(lang, 'Комментарий'), coalesce(deport_comment, ''), res_link, '', -1, -1, -1, -1, 3, ''));

  res_html = blob_append(res_html, '</table>');

  if (is_edit = 1) then
  begin
    res_link = f_uri_sql('Сохранить изменения', 'save_deport_info', obj_kind_str, obj_id, 'none', 'Подтверждаете сохраненеие?', '', 0, 0, '', '', 'dialog', '');
    res_html = blob_append(res_html, f_format('<div style="padding: 10px;">%1</div>', res_link));

    res_link = f_uri_sql('Отменить изменения', 'cancel_deport_info', obj_kind_str, obj_id, 'none', 'Подтверждаете отмену?', '', 0, 0, '', '', 'dialog', '');
    res_html = blob_append(res_html, f_format('<div style="padding: 10px;">%1</div>', res_link));
  end

  res_html = blob_append(res_html, '</body>');
  suspend;
end
