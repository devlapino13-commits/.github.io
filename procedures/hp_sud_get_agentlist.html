CREATE PROCEDURE HP_SUD_GET_AGENTLIST(
    TAG D_INTEGER = 0,
    OBJECT_ID D_ID = null,
    PROC_TYPE D_SMALLINT = null)
RETURNS (
    SAG_ID D_ID,
    NAME D_STR_255,
    ADDRESS D_STR_255,
    OKTMO D_STR_20,
    CLIENT_DEFAULT D_STR_255)
AS
declare variable sag_kind d_integer;
begin
  -- Список контрагентов
  -- tag: 0 - все (по умолчанию), 1 - ответственные (те кто может вести дело), 2 - суды, 3 - коллекторы, 4 - контрагенты по ФССП, 5 - контрагенты по судам
  -- object_id для tag = 2 - smr_id, для tag = 3 - slc_id
  -- выход:
  -- client_default: для tag = 2 и 3 - клиенты для которых контрагенты определены внешним сервисом автоматически

  if (tag is null) then tag = 0;

  if (tag = 0) then
  begin
    for select sag_id, sag_name
    from sud_agents
    into sag_id, name
    do
      suspend;
  end

  else if (tag = 1) then
  begin
    for select sag_id, sag_name
    from sud_agents
    where sag_kind in (0, 1)
    into sag_id, name
    do
      suspend;
  end

  else if (tag = 2) then
  begin
    if (object_id is null and proc_type is null) then
      exit;

    if (proc_type is null) then
      proc_type = (select smd_proc_type from sud_maindocs where smd_id = :object_id);

    if (proc_type is null) then
      exception e_message 'Не выбран тип делопроизводства!';

    sag_kind = iif(proc_type = 0, 2, 3);

    -- определенные внешнием сервисом
    for select iif(:proc_type = 0, slc_auto_worldcourt, slc_auto_federalcourt) as sag_id,
               c_name_fio
    from sud_loanclient, knclient, clients
    where slc_from_smr_id = :object_id
      and kc_id = slc_from_kc_id
      and c_id = kc_from_c_id
    into sag_id, client_default
    do
    begin
      select sag_name, sag_address, sag_oktmo
      from sud_agents
      where sag_id = :sag_id
      into name, address, oktmo;

      suspend;
    end 

    for select sag_id, sag_name, sag_address, sag_oktmo, 'Из справочника'
    from sud_agents
    where sag_kind = :sag_kind
    into sag_id, name, address, oktmo, client_default
    do
      suspend;
  end

  else if (tag = 3) then
  begin
    if (object_id is not null) then
    begin
      select smc_auto_collect, c_name_fio
      from sud_loanclient, sud_maincollect, knclient, clients
      where smc_id = :object_id
        and smc_from_slc_id = slc_id
        and kc_id = slc_from_kc_id
        and c_id = kc_from_c_id
      into sag_id, client_default;

      select sag_name, sag_address, sag_oktmo
      from sud_agents
      where sag_id = :sag_id
      into name, address, oktmo;

      suspend;
    end

    for select sag_id, sag_name, sag_address, sag_oktmo, 'Из справочника'
    from sud_agents
    where sag_kind = 4
    into sag_id, name, address, oktmo, client_default
    do
      suspend;
  end

  else if (tag in (4, 5)) then
  begin
    for select sag_id, sag_name
    from sud_agents
    where sag_kind = 1
    into sag_id, name
    do
      suspend;
  end
end
