CREATE PROCEDURE HP_PRODUCTBLANK_CHECK_EDIT(
    PR_ID D_INTEGER,
    BLN_ID D_INTEGER,
    REP_DATE D_DATE,
    RES_CHECKED D_SMALLINT,
    UN_ID D_ID)
AS
declare variable prb_id d_integer;
declare variable prb_date_begin d_date;
begin
  if (res_checked = 1) then
  begin
    update productblank
    set prb_date_end = null
    where prb_from_pr_id = :pr_id and
          prb_from_bln_id = :bln_id and
          prb_date_begin <= (:rep_date - 1) and
          prb_date_end = :rep_date - 1
    order by prb_date_begin desc, prb_id desc
    rows 1;

    if (not exists(select * from productblank
                   where prb_from_pr_id = :pr_id and
                         prb_from_bln_id = :bln_id and
                         :rep_date between prb_date_begin and
                                           coalesce(prb_date_end, :rep_date))) then
    begin
      insert into productblank (prb_from_pr_id, prb_from_bln_id, prb_date_begin, prb_from_un_id)
      values (:pr_id, :bln_id, :rep_date, :un_id);
    end
  end else
  begin
    for select prb_id, prb_date_begin
    from productblank
    where prb_from_pr_id = :pr_id and
          prb_from_bln_id = :bln_id and
          :rep_date between prb_date_begin and coalesce(prb_date_end, :rep_date)
    into prb_id, prb_date_begin
    do
    begin
      if (prb_date_begin = rep_date) then
      begin
        delete from productblank
        where prb_id = :prb_id;
      end else
      begin
        update productblank
        set prb_date_end = (:rep_date - 1)
        where prb_id = :prb_id;
      end
    end
  end
end
