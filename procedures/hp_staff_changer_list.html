CREATE PROCEDURE HP_STAFF_CHANGER_LIST(
    UN_ID DECIMAL(18,0),
    WITH_HIMSELF INTEGER)
RETURNS (
    RES_C_ID DECIMAL(18,0))
AS
declare variable c_id decimal(18,0);
begin
-- Отображает список сотрудников, кого заменяет этот сотрудник
-- если второй параметр 1, то в списке будет отображаться и он сам.
  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  if (with_himself in (-1, 1)) then
  begin
    res_c_id = c_id;
    suspend;
  end

  if (with_himself = -1) then
    exit;

  for select hrv_from_c_id
      from hr_voctravel, hr_orders
      where hrv_date_begin <= current_date and
            hrv_date_end_fact >= current_date and
            hrv_status = 1 and
            hrv_from_ho_id = ho_id and
            ho_order_type <> 20 -- Выход из декрета
            and
            hrv_changer_id = :c_id
      into :res_c_id
  do
  begin
    suspend;
  end
end
