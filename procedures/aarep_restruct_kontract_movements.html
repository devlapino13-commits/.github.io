CREATE PROCEDURE AAREP_RESTRUCT_KONTRACT_MOVEMENTS(
    REP_DATE_END D_DATE,
    MOVEMENT_KIND D_SMALLINT)
RETURNS (
    CLIENT_FIO D_STR_255,
    KFOLDER_NO D_STR_100,
    KONTRACT_NO D_STR_20,
    MOVED_FROM D_STR_100,
    MOVED_TO D_STR_100,
    DAYS_OVEDUE_QTY D_INTEGER)
AS
declare variable k_id d_id;
declare variable k_from_kf_id d_id;
begin
    /*
      RFCKZ-705
      Процедура возвращает данные для отчета по движениям контрактов между отделами (юр. отделом и ОРиВ)

      movement_kind:
      1 - Движения контрактов из юр. отдела в ОРиВ
      2 - Движения контрактов из ОРиВ в юр. отдел
    */

    if (movement_kind = 1) then
    begin
        for
            select
                rbc.rbc_from_k_id,
                k.k_kontract_no,
                k.k_from_kf_id,
                c.c_name_fio
            from restruct_block_cancel rbc
            inner join kontract k on rbc.rbc_from_k_id = k.k_id
            inner join knclient kc on rbc.rbc_from_k_id = kc.kc_from_k_id and
                                      kc.kc_amount_approved > 0
            inner join clients c on kc.kc_from_c_id = c.c_id
            -- Передача кредита от юр. отдела в ОРиВ происходит при отмене испол. надписи (поле rbc_date_begin), поэтому для выборки берем это поле
            where rbc.rbc_date_begin = :rep_date_end and
                  rbc.rbc_moved_law_to_oriv = 1
            into
                :k_id,
                :kontract_no,
                :k_from_kf_id,
                :client_fio
        do
        begin
            select kal.kal_new_kf_name
            from knaktlist kal
            where :rep_date_end between kal.kal_date and kal.kal_date_end and
                  kal.kal_from_kf_id = :k_from_kf_id and
                  kal.is_delete = 0
            into :kfolder_no;

            select iif(cast(:rep_date_end as date) - oks.oks_delinq_date < 0, 0, cast(:rep_date_end as date) - oks.oks_delinq_date)
            from oksummary oks
            where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
                  oks.oks_from_k_id = :k_id and
                  oks.oks_close_state = 0
            into :days_ovedue_qty;


            moved_from = 'Юридический отдел';
            moved_to = 'ОРиВ';

            suspend;
        end
    end

    if (movement_kind = 2) then
    begin
        for
            select
                rbc.rbc_from_k_id,
                k.k_kontract_no,
                k.k_from_kf_id,
                c.c_name_fio
            from restruct_block_cancel rbc
            inner join kontract k on rbc.rbc_from_k_id = k.k_id
            inner join knclient kc on rbc.rbc_from_k_id = kc.kc_from_k_id and
                                      kc.kc_amount_approved > 0
            inner join clients c on kc.kc_from_c_id = c.c_id
            -- Передача кредита от ОРиВ в юр. отдел проводит робот по прошествии 30 дней с момента отмены испол. надписи,
            -- поле rbc_date_end содержит как раз дату эту дату, поэтому выборку делаем по нему
            where rbc.rbc_date_end = :rep_date_end and
                  rbc.rbc_moved_oriv_to_law = 1
            into
                :k_id,
                :kontract_no,
                :k_from_kf_id,
                :client_fio
        do
        begin
            select kal.kal_new_kf_name
            from knaktlist kal
            where :rep_date_end between kal.kal_date and kal.kal_date_end and
                  kal.kal_from_kf_id = :k_from_kf_id and
                  kal.is_delete = 0
            into :kfolder_no;

            select iif(cast(:rep_date_end as date) - oks.oks_delinq_date < 0, 0, cast(:rep_date_end as date) - oks.oks_delinq_date)
            from oksummary oks
            where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
                  oks.oks_from_k_id = :k_id and
                  oks.oks_close_state = 0
            into :days_ovedue_qty;


            moved_from = 'ОРиВ';
            moved_to = 'Юридический отдел';

            suspend;
        end

    end
end
