CREATE PROCEDURE FM_CLIENTLINK_ADD(
    UN_ID D_ID,
    C_ID D_ID,
    NAME D_STR_100,
    PHONE_NO D_STR_CODE,
    MCL_ID D_ID = 0)
RETURNS (
    RES_TEXT D_STR_255,
    NEED_REFRESH D_INTEGER)
AS
declare variable found_c_id d_id;
declare variable date_end d_date;
declare variable ovs_datetime d_datetime;
declare variable clt_id d_integer;
declare variable is_new_contdata d_smallint;
declare variable clk_from_un_id d_id;
declare variable clk_id d_id;
declare variable lo_id_partner id_type;
declare variable partner_fio str_100;
declare variable send_text str_255;
begin
  need_refresh = 0;

  un_id = (select un_id
           from hp_test_un_id(:un_id));

  is_new_contdata = f_get_const('new_contdata');
  is_new_contdata = 1; -- процедуру переводим на новый механизм, Sanjari, 21.12.2019

  select clk_from_c_id, clk_from_un_id, clk_id
  from clientlink
  where clk_link_cell_no = :phone_no
  rows 1
  into found_c_id, clk_from_un_id, clk_id;

  -- если ранее добавлен с самим партнером то меняем un_id,
  --  если un_id Випса то Випс берет на доработку а если FINSTART_ROBOT то кидается на общий очередь
  if (clk_from_un_id = 0 and un_id <> 0) then
  begin
    update clientlink set clk_from_un_id = :un_id
    where clk_id = :clk_id;
  end 

  if (found_c_id is not null) then
  begin
    if (found_c_id = c_id) then
      res_text = 'Номер уже существует в списке друзей этого клиента';
    else
      res_text = 'Номер уже существует в списке друзей другого клиента';

    Suspend;
    exit;
  end

  for select c_id
  from hp_find_client_by_continfo(0, :phone_no)
  into :found_c_id
  do
  begin
    date_end = null;
    res_text = '';
    ovs_datetime = null;

    if (is_new_contdata = 0) then
    begin
      select ovs_datetime
      from objectprops, objectvalues, refdata
      where opr_id = ovs_from_opr_id and
            opr_actual_ovs_id = ovs_id and
            opr_object_kind = 0 and
            opr_object_id = :found_c_id and
            opr_subject_kind = -1 and
            opr_subject_id = rd_id and
            ovs_value = :phone_no and
            hash(lower(ovs_value)) = hash(lower(:phone_no)) and
            rd_main_code = 'ContactInfo' and
            rd_main_code <> rd_code
      order by ovs_datetime desc
      rows 1
      into ovs_datetime;
    end else
    begin
      clt_id = f_cont_id(phone_no, 'null');
      if (clt_id is not null) then
      begin
        select cda_date
        from contdata
        where cda_from_c_id = :found_c_id and
              cda_from_clt_id = :clt_id and
              cda_is_delete = 0
        order by cda_id
        rows 1
        into ovs_datetime;
      end
    end

    if (ovs_datetime is not null) then
    begin
      if (datediff(day, cast(ovs_datetime as date), current_date) <= 30) then
        res_text = 'Номер принадлежит существующему клиенту';
    end

    if (res_text = '') then
    begin
      select ks_date_end
      from kontstate, knclient
      where kc_from_c_id = :found_c_id and
            kc_amount_approved > 0 and
            kc_from_k_id = ks_from_k_id and
            ks_state = 1 and
            ks_ol_is_delete = 0
      order by ks_date_end desc, ks_state desc, ks_ol_is_delete desc
      rows 1
      into date_end;
  
      if (date_end >= cast('31.12.2999' as date)) then
        res_text = 'Номер принадлежит активному клиенту';
      else
      if (date_end >= dateadd(month, -6, current_date)) then
        res_text = 'Клиент с таким номером недавно обслуживался в компании';
    end

    if (res_text <> '') then
    begin
      Suspend;
      exit;
    end
  end

  if (coalesce(mcl_id, 0) > 0) then
  begin
    select mcl.mcl_from_c_id
    from mc_login mcl
    where mcl.mcl_id = :mcl_id
    into :c_id;
  end 

  insert into clientlink (clk_from_c_id, clk_link_name, clk_link_cell_no, clk_comment, clk_status, clk_from_un_id, clk_from_mcl_id)
  values (:c_id, :name, :phone_no, '', 0, :un_id, :mcl_id);

  if (mcl_id <> 0) then
  begin
    select kal_new_kf_officer
    from knclient, kontstate, kontract, knaktlist
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          current_date between kal_date and kal_date_end
    order by kc_id desc rows 1
    into lo_id_partner;

    select c_name_fio
    from clients
    where c_id = :c_id
    into partner_fio;

    send_text = 'Ваш Партнер ' || partner_fio || ' добавил друга';
    execute procedure hp_send_push_to_staff(:lo_id_partner, 'Партнер добавил Друга', :send_text, 1000000000000, 'msg_to_staff', 7);
  end

  res_text = 'Успешно добавлен';
  need_refresh = 1;
  suspend;
end
