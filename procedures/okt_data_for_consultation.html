CREATE PROCEDURE OKT_DATA_FOR_CONSULTATION(
    CELL_NO D_STR_100,
    FIO D_STR_100,
    STEP D_SMALLINT = 1)
RETURNS (
    C_ID D_ID,
    C_BIRTH_DATE D_STR_100,
    C_NAME_FIO D_STR_100,
    DATA_TXT D_BLOB_TEXT)
AS
declare variable acc d_str_100;
declare variable kontract_no d_str_100;
declare variable disb_date d_str_100;
declare variable k_id d_id;
declare variable total_debt d_integer;
declare variable delinq_days d_integer;
declare variable svz_id d_id;
declare variable svz_fio d_str_100;
declare variable svz_cell_no d_str_100;
declare variable reg_name d_str_100;
begin
    data_txt = '';

    cell_no = trim(coalesce(cell_no, ''));
    fio = trim(coalesce(fio, ''));

    if ((cell_no = '') and (fio = '')) then
    begin
        suspend;
        exit;
    end

    if ((cell_no <> '') and (fio <> '')) then
    begin
        suspend;
        exit;
    end

    if (step = 0) then
    begin
        if (fio <> '') then
        begin
            for select cl.c_id,
                   f_datetostr(c_birth_date),
                   c_name_fio
            from clients cl
            where cl.c_name_fio = :fio    
            into :c_id, :c_birth_date, :c_name_fio
            do
            begin
                suspend;
            end
        end

        if (cell_no <> '') then
        begin
           for select c_id,
                 f_datetostr(c_birth_date) as c_birth_date,
                 c_name_fio
           from hp_find_client_by_continfo(null, :cell_no)
           into :c_id, :c_birth_date, :c_name_fio
           do
           begin
                suspend;
           end
        end
    end

    if (step = 1) then
    begin
        for select distinct c_id, c_birth_date, c_name_fio
        from okt_data_for_consultation(:cell_no, :fio, 0)
        into :c_id, :c_birth_date, :c_name_fio
        do
        begin
            data_txt = BLOB_APPEND(data_txt,'<b>ФИО:</b> '||:c_name_fio||'<br>');
            data_txt = BLOB_APPEND(data_txt,'<b>Дата рождения:</b> '||:c_birth_date||'<br>');
    
            select uniq_acc_str
            from hp_show_clientuniq_acc_no(:c_id)
            into :acc; -- № Лицевого счета
    
            data_txt = BLOB_APPEND(data_txt,'<b>Лицевой счет:</b> '||:acc||'<br>');
            data_txt = BLOB_APPEND(data_txt,'<br>');
    
            for select kont.k_kontract_no,
                       f_datetostr(kont.k_disb_date),
                       k_id,
                       kal.kal_new_kf_officer,
                       cl.c_name_fio,
                       coalesce(f_cont_info('clients', kal.kal_new_kf_officer, 'cell_main', 'ok'),''),
                       reg.reg_name
            from kontstate ks
            inner join kontract kont on kont.k_id = ks.ks_from_k_id
            inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
            inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
            inner join clients cl on cl.c_id = kal.kal_new_kf_officer
            inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
            where kncl.kc_amount_approved > 0
            and kncl.kc_from_c_id = :c_id
            and ks.ks_ol_is_delete = 0
            and ks.ks_state = 1
            and current_date between ks.ks_date_begin and ks.ks_date_end
            and current_date between kal.kal_date and kal.kal_date_end
            and kal.is_delete = 0
            into :kontract_no,
                 :disb_date,
                 :k_id,
                 :svz_id,
                 :svz_fio,
                 :svz_cell_no,
                 :reg_name
            do
            begin
                data_txt = BLOB_APPEND(data_txt,'<b>-------------------------------------</b>'||'<br>');
                data_txt = BLOB_APPEND(data_txt,'<b>№ договора:</b> '||kontract_no||'<br>');
                data_txt = BLOB_APPEND(data_txt,'<b>Офис кредитной папки:</b> '||reg_name||'<br>');
                data_txt = BLOB_APPEND(data_txt,'<b>Дата выдачи:</b> '||disb_date||'<br>');
                data_txt = BLOB_APPEND(data_txt,'<b>Рекомендуемая сумма платежа:</b> '||(select round(RECOM_PAY_SUM) from GET_RECOMMENDED_PAY_SUM(:k_id, current_date))||'<br>');
    
                delinq_days = 0;
    
                select iif((current_date - oks.oks_delinq_date<0), 0, current_date -  oks.oks_delinq_date) -- число дней активной просрочки
                from oksummary oks
                where current_date between oks.oks_date_begin and oks.oks_date_end
                and oks_close_state = 0
                and oks.oks_from_k_id = :k_id
                into :delinq_days;
                data_txt = BLOB_APPEND(data_txt,'<b>Дней активной просрочки:</b> '||:delinq_days||'<br>');
                data_txt = BLOB_APPEND(data_txt,'<b>ID ВИПС:</b> '||:svz_id||'<br>');
                data_txt = BLOB_APPEND(data_txt,'<b>ФИО ВИПС:</b> '||:svz_fio||'<br>');
                --data_txt = BLOB_APPEND(data_txt,'<b>Телефон ВИПС:</b> '||:svz_cell_no||'<br>');
                data_txt = BLOB_APPEND(data_txt,'<b>-------------------------------------</b>'||'<br>');
            end

            if (k_id is null) then
                data_txt = BLOB_APPEND(data_txt,'<b>Активный договор не найден</b><br>');
        end
    end
    suspend;
end
