CREATE PROCEDURE HP_SEND_CLIENT_TO_VERIFY(
    C_ID D_ID,
    UN_ID D_ID,
    ONLY_PHOTO_CHANGED D_SMALLINT = 0)
AS
declare variable RD_ID D_INTEGER;
declare variable ACL_ID D_ID;
declare variable UN_C_ID D_ID;
declare variable REG_ID D_INTEGER;
declare variable DB_ID D_INTEGER;
begin
  db_id = f_get_db_id();
  rd_id = f_get_rd_id('ActionGroup.Passport_Verify');
  if (rd_id is null) then
    exit;

  if (db_id = 81001) then  -- RFCRU-5800
  begin
    if (exists(select 1
               from mc_login mcl
               where mcl.mcl_id = rdb$get_context('USER_TRANSACTION', 'tmp_mcl_id') and
                     mcl.mcl_project = f_enum('mcl_project', 'oivo', 'id')))
    then
      exit;

    if (exists(select 1
               from actionlist acl
               where acl.acl_object_kind = 0 and
                     acl.acl_object_id = :c_id and
                     acl.acl_status in (0, 1, 2) and
                     acl.acl_from_rd_id = f_get_rd_id('ActionGroup.Passport_Verify_OiVO')))
    then
      exit;
  end

  select aco_id
  from actionorder
  where aco_object_kind = 0 and
        aco_object_id = :c_id and
        aco_from_rd_id = :rd_id rows 1
  into :acl_id;

  if (acl_id is not null) then
  begin
    delete from actionmemo
    where acm_from_acl_id = :acl_id;
    exit;
  end

  if (only_photo_changed is null) then
    only_photo_changed = 0;

  select un_from_c_id
  from username
  where un_id = :un_id
  into :un_c_id;

  if (un_c_id is not null) then
  begin
    select ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :un_c_id and
          ps_date_end >= '31.12.2999' and
          ps_status = 1
    order by ps_date_begin desc
    rows 1
    into :reg_id;
  end
  if (reg_id is null) then
  begin
    select cs_region_id
    from clientstate
    where cs_from_c_id = :c_id
    into :reg_id;
  end
  if (reg_id is null) then
  begin
    reg_id = f_hq_id();
  end

  -- если меняется только фото клиента и до этого был верифицирован, отмечаем это в acl_quota_used = 1
  insert into actionlist (acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                          acl_respons_c_id, acl_from_un_id, acl_dont_show_in_order, acl_quota_used)
  values (current_date, :reg_id, 0, :c_id, :rd_id, 0, null, :un_id, 0, :only_photo_changed);
end
