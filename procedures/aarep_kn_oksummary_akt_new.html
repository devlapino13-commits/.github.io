CREATE PROCEDURE AAREP_KN_OKSUMMARY_AKT_NEW(
    REP_DATE D_DATE)
RETURNS (
    REC_NO D_INTEGER,
    OKS_FROM_KC_ID D_ID,
    OKS_DATE_BEGIN D_DATE,
    C_NAME_FIO D_STR_100,
    C_ID_STR D_STR_CODE,
    K_KONTRACT_NO D_STR_CODE,
    K_PERIOD_DAY D_INTEGER,
    KC_AMOUNT_APPROVED D_CURRENCY,
    OKS_DISB_DATE D_DATE,
    OKS_PERCENT_NORM D_PERCENT,
    OKS_PERCENT_ESP D_PERCENT,
    OKS_CLOSE_STATE D_SMALLINT,
    OKS_DELINQ_DAYS D_INTEGER,
    OKS_DELINQ_DAYS_LLR D_INTEGER,
    OKS_DELINQ_DAYS_365 D_INTEGER,
    OKS_LAST_SH_DATE D_DATE,
    OKS_KONT_LAST_DATE D_DATE,
    OKS_BALANCE D_CURRENCY,
    OKS_REFUND_DELINQ_BEFORE D_INTEGER,
    OKS_REFUND_DELINQ_180 D_INTEGER,
    OKS_LAST_RESH_DATE D_DATE,
    OKS_LAST_RESH_DELINQ_BEFORE D_INTEGER,
    OKS_LAST_RESH_DELINQ_180 D_INTEGER,
    OKS_ZALOG_COST D_CURRENCY,
    OKS_LLR_BASE_PERCENT D_PERCENT,
    OKS_LLR_PERCENT D_PERCENT,
    OKS_PDN_KOEF D_PERCENT,
    OKS_BAL_PERC D_CURRENCY,
    OKS_BAL_KOMMIS D_CURRENCY,
    OKS_EAD_AMOUNT D_CURRENCY,
    OKS_BAL_ESP_END D_CURRENCY,
    OKS_DIFF_ESP_BAL D_CURRENCY,
    OKS_ECL_ACC_KIND D_INTEGER,
    OKS_ECL_PERCENT D_PERCENT,
    OKS_LLR_OS D_CURRENCY,
    OKS_LLR_PERC D_CURRENCY,
    OKS_ECL_VBS D_CURRENCY,
    OKS_DIFF_ECL_LLR D_CURRENCY,
    OKS_LLR_PERCENT_OUTER D_PERCENT,
    OKS_LLR_PERCENT_INNER D_PERCENT,
    OKS_LLR_PERCENT_NADZOR D_PERCENT,
    OKS_LLR_OS_NADZOR D_CURRENCY,
    OKS_LLR_PERC_NADZOR D_CURRENCY,
    OKS_ECL_VBS_NADZOR D_CURRENCY,
    OKS_LLR_GROUP D_INTEGER,
    PR_NAME D_STR_100,
    OKS_IS_REFUND_STR D_STR_CODE,
    OKS_IS_REFUND_LLR_STR D_STR_CODE,
    OKS_IS_RESCHEDULE_LLR_STR D_STR_CODE,
    OKS_IS_ZALOG_STR D_STR_CODE,
    OKS_IS_ZALOG_LLR_STR D_STR_CODE,
    OKS_ZALOG_IS_BANKRUPT_STR D_STR_CODE,
    OKS_CLIENT_TYPE_STR D_STR_CODE,
    MSP_SUBJECT_STR D_STR_CODE,
    OKS_IS_NOT_TARGETED_STR D_STR_CODE,
    OKS_NO_DOCUMENT_STR D_STR_CODE,
    OKS_CLIENT_IS_BANKRUPT_STR D_STR_CODE,
    OKS_DELINQ_DAYS_OWN_LLR D_INTEGER,
    OKS_PARAL_LOAN_COUNT D_INTEGER,
    OKS_PARAL_DELINQ_DAY D_INTEGER,
    OKS_DAYS_FROM_DISB D_INTEGER,
    OKS_LAST_RESH_DAYS D_INTEGER)
AS
declare variable oks_is_refund smallint;
declare variable oks_is_refund_llr smallint;
declare variable oks_is_reschedule_llr smallint;
declare variable oks_is_zalog smallint;
declare variable oks_is_zalog_llr smallint;
declare variable oks_client_is_bankrupt smallint;
declare variable oks_zalog_is_bankrupt smallint;
declare variable oks_is_not_targeted smallint;
declare variable oks_no_document smallint;
begin
  rec_no = 0;
  for select oks_from_kc_id,
             oks_date_begin, oks_close_state, kc_amount_approved,
             datediff(day, k_disb_date, k_kont_last_date), k_kontract_no, oks_from_kc_id, oks_disb_date,
             oks_kont_last_date, oks_last_sh_date, oks_balance, oks_bal_perc, oks_bal_kommis, oks_diff_esp_bal,
             oks_bal_esp_end, oks_ead_amount, oks_refund_delinq_180, oks_last_resh_date, oks_last_resh_delinq_180,
             oks_zalog_cost, oks_llr_group, oks_delinq_days, oks_last_resh_delinq_before, oks_refund_delinq_before,
             oks_delinq_days_own_llr, oks_paral_loan_count, oks_paral_delinq_day, oks_delinq_days_llr,
             oks_delinq_days_365, oks_llr_base_percent, oks_llr_percent_outer, oks_ecl_percent,
             oks_ecl_vbs, oks_llr_percent_inner, oks_llr_percent_nadzor, oks_llr_percent, oks_llr_os_nadzor,
             oks_llr_perc_nadzor, oks_ecl_vbs_nadzor, oks_llr_os, oks_llr_perc, oks_diff_ecl_llr, oks_ecl_acc_kind,
             pr_name, oks_percent_norm, oks_percent_esp, c_name_fio, 'n' || c_id,
             oks_is_refund, oks_is_refund_llr, oks_is_reschedule_llr, oks_is_zalog, oks_is_zalog_llr,
             oks_client_is_bankrupt, oks_zalog_is_bankrupt, oks_is_not_targeted, oks_no_document,
             datediff(day, k_disb_date, :rep_date)
      from kn_oksummary, knclient, kontract, clients, product
      where oks_from_kc_id = kc_id and
            oks_date_begin = :rep_date and
            oks_close_state = 0 and
            oks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            kc_from_c_id = c_id
      into oks_from_kc_id,
           oks_date_begin, oks_close_state, kc_amount_approved, k_period_day, k_kontract_no,
           oks_from_kc_id, oks_disb_date, oks_kont_last_date, oks_last_sh_date, oks_balance, oks_bal_perc,
           oks_bal_kommis, oks_diff_esp_bal, oks_bal_esp_end, oks_ead_amount, oks_refund_delinq_180, oks_last_resh_date,
           oks_last_resh_delinq_180, oks_zalog_cost, oks_llr_group, oks_delinq_days, oks_last_resh_delinq_before,
           oks_refund_delinq_before, oks_delinq_days_own_llr, oks_paral_loan_count, oks_paral_delinq_day,
           oks_delinq_days_llr, oks_delinq_days_365, oks_llr_base_percent, oks_llr_percent_outer,
           oks_ecl_percent, oks_ecl_vbs, oks_llr_percent_inner, oks_llr_percent_nadzor, oks_llr_percent,
           oks_llr_os_nadzor, oks_llr_perc_nadzor, oks_ecl_vbs_nadzor, oks_llr_os, oks_llr_perc, oks_diff_ecl_llr,
           oks_ecl_acc_kind, pr_name, oks_percent_norm, oks_percent_esp, c_name_fio, c_id_str,
           oks_is_refund, oks_is_refund_llr, oks_is_reschedule_llr, oks_is_zalog, oks_is_zalog_llr,
           oks_client_is_bankrupt, oks_zalog_is_bankrupt, oks_is_not_targeted, oks_no_document,
           oks_days_from_disb
  do
  begin
    rec_no = rec_no + 1;

    oks_last_resh_days = null;
    if (oks_last_resh_date is not null) then
      oks_last_resh_days = datediff(day, oks_last_resh_date, rep_date);

    oks_pdn_koef = 0;
    select kc_pdn_koef
    from knclient_pdn
    where kc_id = :oks_from_kc_id
      and kcp_kind = 0
    into oks_pdn_koef;

    if (oks_is_refund = 1) then
      oks_is_refund_str = 'Да';
    else
      oks_is_refund_str = 'Нет';

    if (oks_is_refund_llr = 1) then
      oks_is_refund_llr_str = 'Да';
    else
      oks_is_refund_llr_str = 'Нет';

    if (oks_is_reschedule_llr = 1) then
      oks_is_reschedule_llr_str = 'Да';
    else
      oks_is_reschedule_llr_str = 'Нет';

    if (oks_is_zalog = 1) then
      oks_is_zalog_str = 'Да';
    else
      oks_is_zalog_str = 'Нет';

    if (oks_is_zalog_llr = 1) then
      oks_is_zalog_llr_str = 'Да';
    else
      oks_is_zalog_llr_str = 'Нет';

    if (oks_zalog_is_bankrupt = 1) then
      oks_zalog_is_bankrupt_str = 'Да';
    else
      oks_zalog_is_bankrupt_str = 'Нет';

    if (oks_is_not_targeted = 1) then
      oks_is_not_targeted_str = 'Да';
    else
      oks_is_not_targeted_str = 'Нет';

    if (oks_no_document = 1) then
      oks_no_document_str = 'Да';
    else
      oks_no_document_str = 'Нет';

    if (oks_client_is_bankrupt = 1) then
      oks_client_is_bankrupt_str = 'Да';
    else
      oks_client_is_bankrupt_str = 'Нет';

    oks_client_type_str = 'ФЛ';
    msp_subject_str = 'Нет';

    suspend;
  end
end
