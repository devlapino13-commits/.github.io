CREATE PROCEDURE AAREP_CC_CHECK_RESULT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    C_NAME_FIO STR_100,
    K_KONTRACT_NO STR_100,
    K_DISB_DATE DATE,
    RESULT STR_100,
    CC_CHECK_MEMO STR_255,
    CC_CALL_USER STR_100,
    CC_CHECKED_DATE DATE,
    CC_CHECKED_DATETIME TIMESTAMP,
    KS_STATE STR_100,
    CC_NAME STR_100)
AS
begin
  for select reg.reg_name, cl.c_name_fio, kont.k_kontract_no,

             k_disb_date,
             (
             case cc_total_res
               when 1 then 'Подтверждена'
               when -1 then 'Не дозвонились'
               when -2 then 'Не подтверждена'
               else ''
             end) as result,
             cont.cc_check_memo,

             cont.cc_call_user, cont.cc_checked_date, cont.cc_checked_datetime,

             (
             case ks.ks_state
               when -1 then 'Аннулирован'
               when 0 then 'В процессе'
               when 1 then 'Выдан'
               when 2 then 'Закрыт'
             end) as ks_state,
             cont.cc_name
      from contacts cont
      inner join knclient kncl on kncl.kc_from_c_id = cont.cc_from_c_id
      inner join clients cl on cl.c_id = cont.cc_from_c_id
      inner join kontract kont on kont.k_id = kncl.kc_from_k_id
      inner join kontstate ks on ks.ks_from_k_id = kont.k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
      inner join region_list(:rep_region_id, 1, -1, 1) on id_list = kal.kal_new_reg_id
      inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
      where cont.cc_checked_date between :rep_date_begin and :rep_date_end and
            cont.cc_total_res <> 0 and
            kncl.kc_amount_approved > 0 and
            :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_ol_is_delete = 0 and
            kal.is_delete = 0 and
            kont.k_disb_date between :rep_date_begin and :rep_date_end + 15 and
            :rep_date_end between kal.kal_date and kal.kal_date_end
      order by cont.cc_checked_date
      into :reg_name, :c_name_fio, :k_kontract_no, :k_disb_date, :result, :cc_check_memo, :cc_call_user,
           :cc_checked_date, :cc_checked_datetime, :ks_state, :cc_name
  do
  begin
    suspend;
  end
end
