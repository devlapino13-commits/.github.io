CREATE PROCEDURE HP_EDOC_COM_ACTION(
    EDL_ID D_ID NOT NULL,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable CML_ID D_ID;
declare variable EDC_CODE D_STR_CODE;
declare variable NEED_EDOC_CANCEL D_SMALLINT;
begin
  res_is_error = 0;
  res_text = '';

  select edl_object_id, edc_code
  from edoc_list, edoc_cfg
  where edl_id = :edl_id and
        edl_object_kind = f_objkind('com_list', 'id') and
        edl_from_edc_id = edc_id
  into cml_id, edc_code;

  if (cml_id is null) then
  begin
    suspend;
    exit;
  end

  if (edc_code in ('loan.reschedule', 'loan.termination')) then
  begin
    select res_is_error, res_text, need_edoc_cancel
    from hp_com_action_execute(:cml_id, :un_id)
    into res_is_error, res_text, need_edoc_cancel;

    if (need_edoc_cancel = 1) then
    begin
      update edoc_sign
      set eds_status = 0
      where eds_from_edl_id = :edl_id and
            eds_status <> 0;
  
      update edoc_list
      set edl_status = -1
      where edl_id = :edl_id;
    end

    if (res_is_error = 0) then
    begin
      select res_is_error, res_text
      from hp_com_action_add(:cml_id, null, 'done', :un_id)
      into res_is_error, :res_text;
    end
  end

  if (edc_code = 'loan.reschedule.pdn') then
  begin
    select res_is_error, res_text
    from hp_com_action_add(:cml_id, null, 'signed_edoc_pdn', :un_id)
    into res_is_error, :res_text;
  end

  suspend;
end
