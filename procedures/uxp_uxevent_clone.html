Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXEVENT_CLONE(
    UX_UUID D_UUID)
AS
declare variable tmp_uuid d_uuid;
declare variable new_ux_uuid d_uuid;
declare variable evt_uuid d_uuid;
begin
  new_ux_uuid = uxf$pair(ux_uuid);

  for select evt_uuid, evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position
  from ux$event
  where evt_sender_uuid = :ux_uuid
  order by evt_position
  as cursor cur
  do
  begin
    if (new_ux_uuid is null) then
      exception e_message f_format('ux_uuid clone not found: %1', uuid_to_char(ux_uuid));

    tmp_uuid = uxf$pair(cur.evt_target_uuid);

    update or insert into ux$event (evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position)
    values (:new_ux_uuid, cur.evt_event_kind, :tmp_uuid, cur.evt_handler_kind, cur.evt_position)
    matching (evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind)
    returning (evt_uuid)
    into evt_uuid;

    execute procedure uxp$uxpairuuid_add(cur.evt_uuid, :evt_uuid);

    insert into ux$expression (exp_ux_uuid, exp_kind, exp_expression, exp_calc_expr, exp_calc_ufr_uuid, exp_message,
                               exp_is_sql, exp_action, exp_position, exp_is_bool, exp_calc_kind)
    select :evt_uuid, exp_kind, exp_expression, exp_calc_expr, uxf$pair(exp_calc_ufr_uuid), exp_message,
           exp_is_sql, exp_action, exp_position, exp_is_bool, exp_calc_kind
    from ux$expression
    where exp_ux_uuid = :cur.evt_uuid;
  end

  for select evt_uuid, evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position
  from ux$event
  where evt_target_uuid = :ux_uuid
  order by evt_sender_uuid, evt_position
  as cursor cur
  do
  begin
    if (new_ux_uuid is null) then
      exception e_message f_format('ux_uuid clone not found: %1', uuid_to_char(ux_uuid));

    tmp_uuid = uxf$pair(cur.evt_sender_uuid);

    update or insert into ux$event (evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position)
    values (:tmp_uuid, cur.evt_event_kind, :new_ux_uuid, cur.evt_handler_kind, cur.evt_position)
    matching (evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind)
    returning (evt_uuid)
    into evt_uuid;

    execute procedure uxp$uxpairuuid_add(cur.evt_uuid, :evt_uuid);

    insert into ux$expression (exp_ux_uuid, exp_kind, exp_expression, exp_calc_expr, exp_calc_ufr_uuid, exp_message,
                               exp_is_sql, exp_action, exp_position, exp_is_bool, exp_calc_kind)
    select :evt_uuid, exp_kind, exp_expression, exp_calc_expr, uxf$pair(exp_calc_ufr_uuid), exp_message,
           exp_is_sql, exp_action, exp_position, exp_is_bool, exp_calc_kind
    from ux$expression
    where exp_ux_uuid = :cur.evt_uuid;
  end
end
