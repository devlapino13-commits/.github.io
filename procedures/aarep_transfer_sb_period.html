CREATE PROCEDURE AAREP_TRANSFER_SB_PERIOD(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    SBA_DATE_BEGIN DATE,
    SBA_DATE_END DATE,
    SBA_MEMO VARCHAR(20),
    DATE_OPER DATE,
    KAL_NEW_KF_NAME VARCHAR(30),
    K_KONTRACT_NO VARCHAR(20),
    GR_BALANCE CURR_TYPE,
    GR_DELINQ_OS CURR_TYPE,
    GR_DELINQ_PERC CURR_TYPE,
    GR_FINE CURR_TYPE,
    C_NAME_FIO VARCHAR(100),
    CL_BALANCE CURR_TYPE,
    CL_DELINQ_OS CURR_TYPE,
    CL_DELINQ_PERC CURR_TYPE,
    CL_FINE CURR_TYPE,
    DQ_CURRENT_ADDRESS VARCHAR(255),
    DQ_CURRENT_PHONE VARCHAR(255),
    SH_DATE_MIN DATE,
    DELINQ_DAY INTEGER,
    DATE_RAPAY DATE,
    YUR_ADDRESS VARCHAR(255),
    FACT_ADDRESS VARCHAR(255),
    PHONE_NUMBERS_HOME VARCHAR(1024),
    REG_NAME VARCHAR(100),
    C_PLASTCARD_NO VARCHAR(20),
    KONTRACT_NAME VARCHAR(100),
    SBL_NAME VARCHAR(100))
AS
declare variable k_amount_approved curr_type;
declare variable dolya double precision;
declare variable c_id id_type;
declare variable cc_address varchar(255);
declare variable sba_from_k_id decimal(18,0);
declare variable sbl_from_c_id decimal(18,0);
declare variable sbl_from_c_id_prior id_type;
begin
-- operkred_fix

  sbl_from_c_id_prior = 0;
  for select sba_from_k_id, sba_date_begin, sba_date_end, sbl_from_c_id
      from sb_akt, sb_list
      where sba_is_delete = 0 and
            sba_status = 2 and
            sba_from_sbl_id = sbl_id and
            sbl_type = 0 and
            sbl_from_reg_id = :rep_region_id and
            sbl_id <> 0 and
            (sba_date_begin between :rep_date_begin and :rep_date_end or
            sba_date_end between :rep_date_begin - 1 and :rep_date_end - 1)
      order by sbl_from_c_id, sba_from_k_id, sba_date_begin, sba_date_end
      into sba_from_k_id, sba_date_begin, sba_date_end, sbl_from_c_id
  do
  begin
    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :rep_region_id
    into reg_name;

    if (sbl_from_c_id_prior <> sbl_from_c_id) then
    begin
      sbl_from_c_id_prior = sbl_from_c_id;
      sbl_name = null;
      select c_name_fio
      from clients
      where c_id = :sbl_from_c_id
      into sbl_name;
      sbl_name = 'ФИО СБ - ' || sbl_name;
    end
    if ((:sba_date_begin between :rep_date_begin and :rep_date_end) and
        (:sba_date_end between :rep_date_begin - 1 and :rep_date_end - 1)) then
    begin
      sba_memo = 'передан/отозван';
      date_oper = :sba_date_begin;
    end
    else
    begin
      if (:sba_date_begin between :rep_date_begin and :rep_date_end) then
      begin
        sba_memo = 'передан';
        date_oper = :sba_date_begin;
      end
      else
      begin
        sba_memo = 'отозван';
        date_oper = :sba_date_end + 1;
      end
    end

    select kal_new_kf_name, k_kontract_no, k_amount_approved
    from kontract, knaktlist
    where k_id = :sba_from_k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end
    into kal_new_kf_name, k_kontract_no, k_amount_approved;

    gr_balance = null;
    gr_delinq_os = null;
    gr_delinq_perc = null;
    gr_fine = null;
    select sum(zad_os), sum(iif(sh_repay_type = 1, zad_os, 0)),
           sum(iif(sh_repay_type = 1, iif(zad_perc >= sh_calc_perc, zad_perc, sh_calc_perc), 0)), sum(sh_calc_fine),
           min(iif(sh_repay_type = 1, sh_date, :rep_date_end))
    from aarep_make_zad_list(:sba_from_k_id, :rep_date_end, 1)
    where sh_date is not null
    into gr_balance, gr_delinq_os, gr_delinq_perc, gr_fine, sh_date_min;
    if (gr_balance is null) then
      gr_balance = 0;
    if (gr_delinq_os is null) then
      gr_delinq_os = 0;
    if (gr_delinq_perc is null) then
      gr_delinq_perc = 0;
    if (gr_fine is null) then
      gr_fine = 0;

    delinq_day = :rep_date_end - sh_date_min;
    date_rapay = null;
    select max(ok_ol_date)
    from operkred
    where ok_from_k_id = :sba_from_k_id and
          ok_type = 4 and
          ok_os + ok_perc + ok_fine < 0 and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date_end and
          ok_from_c_id = 0
    into date_rapay;

    if (sba_date_end >= '31.12.2999') then
      sba_date_end = null;

    for select c_id, c_name_fio,
               (select cont_value
                from hp_contdata_get_cell_main(0, c_id)),
               cast(kc_amount_approved as double precision) / :k_amount_approved
        from knclient, clients
        where kc_from_k_id = :sba_from_k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = c_id
        into c_id, c_name_fio, c_plastcard_no, dolya
    do
    begin
      dq_current_address = null;
      dq_current_phone = null;
      select dq_current_address, dq_current_phone
      from delinqrep
      where dq_from_k_id = :sba_from_k_id and
            dq_from_c_id = :c_id
      order by dq_id desc
      rows 1
      into dq_current_address, dq_current_phone;

      cl_balance = gr_balance * dolya;
      cl_delinq_os = gr_delinq_os * dolya;
      cl_delinq_perc = gr_delinq_perc * dolya;
      cl_fine = gr_fine * dolya;
--    Юр. и Факт адрес
      yur_address = null;
      fact_address = null;
      select max(iif(adc_address_type = 1, adl_address_str, null)),
             max(iif(adc_address_type = 2, adl_address_str, null))
      from addrclient, addrlist
      where adc_from_c_id = :c_id and
            adc_address_type in (1, 2) and
            adc_from_adl_id = adl_id
      into yur_address, fact_address;
      phone_numbers_home = '';
      for select coalesce(cc_name, '') || ' ' || contacts.cc_cell_no || ';'
          from contacts
          where cc_from_c_id = :c_id and
                contacts.cc_cell_no is not null and
                cc_type = 1
          into cc_address
      do
      begin
        phone_numbers_home = phone_numbers_home || cc_address || ' ';
      end

      kontract_name = '№ папки:' || kal_new_kf_name || '; № договора:' || k_kontract_no;

      suspend;
    end
  end

end
