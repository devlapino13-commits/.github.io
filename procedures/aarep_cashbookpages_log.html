CREATE PROCEDURE AAREP_CASHBOOKPAGES_LOG(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    CBP_NUMBER INTEGER,
    CBP_DATETIME TIMESTAMP,
    CBP_FIRST_PAGE_NUM INTEGER,
    CBP_LAST_PAGE_NUM INTEGER,
    CBP_FROM_UN_ID ID_TYPE,
    C_NAME_FIO VARCHAR(100),
    OPER_NAME VARCHAR(50),
    OPER_DESC VARCHAR(100))
AS
declare variable cbp_is_actual smallint;
declare variable cbp_changed_date date;
begin
  /*
  Возвращает историю печати кассовой книги
  */
  for select cbp_number, cbp_datetime, cbp_first_page_num, cbp_last_page_num, cbp_from_un_id, cbp_is_actual,
             cbp_changed_date
      from cashbookpages
      where cbp_date = :rep_date and
            cbp_from_reg_id = :rep_region_id
      order by cbp_number
      into :cbp_number, :cbp_datetime, :cbp_first_page_num, :cbp_last_page_num, :cbp_from_un_id, :cbp_is_actual,
           :cbp_changed_date
  do
  begin
    c_name_fio = '';
    select c_name_fio
    from username, clients
    where un_id = :cbp_from_un_id and
          un_from_c_id = c_id
    into c_name_fio;

    if (cbp_is_actual = 0) then
    begin
      oper_name = 'Сброс нумерации';
      oper_desc = 'Сброс из-за изменения нумерации за ' || right(100 + extract(day from cbp_changed_date), 2) || '.' || right(100 + extract(month from cbp_changed_date), 2) || '.' || extract(year from cbp_changed_date) || 'г.';
    end
    else
    begin
      oper_name = 'Печать';
      oper_desc = '';
    end
    suspend;
  end
end
