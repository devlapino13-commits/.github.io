CREATE PROCEDURE AAREP_BLOB_LIST(
    KEY_ID DECIMAL(18,0),
    BLC_ID INTEGER)
RETURNS (
    KEYFIELD1_VALUE DECIMAL(18,0),
    KEYFIELD2_VALUE DECIMAL(18,0),
    KEYFIELD3_VALUE DECIMAL(18,0),
    BLOB_BLS_ID DECIMAL(18,0),
    BLOB_CAPTION VARCHAR(100),
    BLOB_FILENAME VARCHAR(100),
    BLOB_SIZE FLOAT,
    BLOB_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    KEYFIELD1_NAME VARCHAR(100),
    KEYFIELD2_NAME VARCHAR(100),
    KEYFIELD3_NAME VARCHAR(100),
    BLC_TABLE_NAME VARCHAR(100),
    BLC_BLOB_FIELDNAME VARCHAR(100),
    BLC_LINK_FIELD VARCHAR(100))
AS
declare variable blc_keyfields varchar(100);
declare variable kol integer;
declare variable k integer;
declare variable kf1_name varchar(100);
declare variable kf2_name varchar(100);
declare variable kf3_name varchar(100);
declare variable sql_text varchar(1024);
begin
  blc_table_name = '';
  kol = 0;
  select blc_table_name, blc_keyfields, blc_blob_fieldname, blc_link_field
  from blobconfig
  where blc_id = :blc_id
  into :blc_table_name, :blc_keyfields, :blc_blob_fieldname, :blc_link_field;
  if (blc_table_name is null) then
    blc_table_name = '';
  if (blc_link_field is null) then
    blc_link_field = '';
  blc_keyfields = replace(blc_keyfields, ' ', '');
  blc_keyfields = upper(blc_keyfields);
  if (blc_table_name <> '') then
  begin
    keyfield1_name = '';
    keyfield2_name = '';
    keyfield3_name = '';
    k = position(';', blc_keyfields);
    if (k > 0) then
    begin
      keyfield1_name = left(blc_keyfields, k - 1);
      blc_keyfields = substring(blc_keyfields from k + 1 for 100);
      k = position(';', blc_keyfields);
      if (k > 0) then
      begin
        keyfield2_name = left(blc_keyfields, k - 1);
        blc_keyfields = substring(blc_keyfields from k + 1 for 100);
        if (blc_keyfields <> '') then
          keyfield3_name = blc_keyfields;
      end
      else
        keyfield2_name = blc_keyfields;
    end
    else
      keyfield1_name = blc_keyfields;
    kf1_name = keyfield1_name;
    if (keyfield2_name <> '') then
      kf2_name = keyfield2_name;
    else
      kf2_name = kf1_name;
    if (keyfield3_name <> '') then
      kf3_name = keyfield3_name;
    else
      kf3_name = kf2_name;
    sql_text = 'select ' || kf1_name || ', ' || kf2_name || ', ' || kf3_name || ', ' || blc_blob_fieldname || ' from ' || blc_table_name;
    sql_text = sql_text || ' where ' || blc_link_field || '=' || :key_id;
    for execute statement(sql_text)
            into :keyfield1_value, keyfield2_value, keyfield3_value, blob_bls_id
    do
    begin
      blob_filename = null;
      blob_caption = null;
      blob_blob = null;
      blob_size = 0;
      if (blob_bls_id <> 0) then
      begin
        select bls_filename, bls_caption, bls_blob_size
        from blobstore
        where bls_id = :blob_bls_id
        into :blob_filename, :blob_caption, :blob_size;
        if (blob_filename is not null) then
        begin
          select blx_blob
          from hp_get_blobex_blob(:blob_bls_id)
          into :blob_blob;
          kol = kol + 1;
        end
      end
      suspend;
    end
    if (kol = 0) then
    begin
      blob_bls_id = 0;
      blob_filename = '';
      blob_caption = 'Пусто‚ в процессе закачки';
      blob_blob = null;
      blob_size = 0;
      suspend;
    end
  end
end
