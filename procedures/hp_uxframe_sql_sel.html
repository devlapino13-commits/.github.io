CREATE PROCEDURE HP_UXFRAME_SQL_SEL(
    UX_CODE D_STR_CODE,
    FR_CODE D_STR_CODE,
    NEED_SECURE_MASK D_BOOL = 0)
RETURNS (
    SQL_TEXT D_STR_8100,
    SQL_IS_EXEC D_BOOL)
AS
declare variable field_name d_str_code;
declare variable field_list d_str_8100;
declare variable enter_sym d_str_10;
declare variable ufp_editor d_str_code;
declare variable ufd_source d_str_code;
declare variable ufd_is_table d_bool;
declare variable buttonedit_sql d_str_long;
declare variable master_field d_str_code;
declare variable detail_field d_str_code;
declare variable master_field_count d_integer;
declare variable detail_field_count d_integer;
declare variable master_detail_where d_str_long;
declare variable res_str d_str_code;
declare variable is_detail d_bool;
declare variable param_index d_integer;
declare variable ufi_master_ufr_code d_str_code;
declare variable md_linked d_bool;
declare variable ufr_show_kind d_smallint;
declare variable ufi_ua_code d_str_100;
declare variable ua_uxa_code d_str_code;
declare variable ua_usr_text_field d_str_code;
declare variable ufp_code d_str_100;
declare variable ux_kind d_str_code;
declare variable max_rows d_integer;
declare variable alias_letter d_str_10;
declare variable search_text_field d_str_code;
begin
  -- Приготовление Select sql для фрейма

  if (need_secure_mask is null) then
    need_secure_mask = 0;

  sql_text = '';
  field_list = '';
  enter_sym = ascii_char(13) || ascii_char(10);

  select ufd_source, ufd_is_table, ufr_show_kind
  from uxfrdata, uxframe
  where ufr_code = :fr_code and
        ufd_ufr_code = ufr_code and
        ufd_kind = 0
  into ufd_source, ufd_is_table, ufr_show_kind;

  if (ufr_show_kind = 0) then
  begin
    suspend;
    exit;
  end

  search_text_field = '';

  if (need_secure_mask = 1) then
  begin
    select usr_text_field
    from uxalias, uxsearch
    where uxa_ux_code = :ux_code and
          uxa_code = usr_code and
          usr_text_field <> ''
    rows 1
    into search_text_field;
  end

  alias_letter = '';

  sql_is_exec = 0;
  master_field = '';
  detail_field = '';
  is_detail = 0;
  master_field_count = 0;
  detail_field_count = 0;

  select ucf_master_field, ucf_detail_field, ucf_master_ufr_code, uxf$kind(ux_kind, 'code')
  from uxcodeframe, uxcode
  where ucf_ux_code = :ux_code and
        ucf_ufr_code = :fr_code and
        ux_code = ucf_ux_code
  into master_field, detail_field, ufi_master_ufr_code, ux_kind;

  if (row_count > 0) then
  begin
    if (ufi_master_ufr_code is not null) then
    begin
      is_detail = 1;
      if (coalesce(master_field, '') = '' or
          coalesce(detail_field, '') = '') then
        exception e_message f_format('Поля мастер-деталь не настроены, форма: %1, frame: %2', ux_code, fr_code);
  
      select count(*)
      from hp_split_str(:master_field, ';')
      into :master_field_count;
  
      select count(*)
      from hp_split_str(:detail_field, ';')
      into :detail_field_count;
  
      if (detail_field_count <> master_field_count) then
        exception e_message f_format('Количество полей мастер и деталь не совпадает, форма: %1, frame: %2', ux_code, fr_code);
    end
  end else
    exception e_default f_format('Frame not found. форма: %1, frame: %2', ux_code, fr_code);

  master_detail_where = '';
  master_field_count = 0;
  detail_field_count = 0;

  if (ufd_is_table = 1 and is_detail = 1) then
  begin
    for select res_str
    from hp_split_str(:detail_field, ';')
    into :res_str
    do
    begin
      detail_field_count = detail_field_count + 1;

      -- Если поле из мастер-деталь связки не является параметром, добавляем в WHERE
      if (exists(select * from uxfrparam
      where ufp_ufr_code = :fr_code and
            ufp_origin_name = :res_str and
            ufp_is_param = 0)) then
      begin
        if (master_detail_where <> '') then
          master_detail_where = master_detail_where || ' and ' || enter_sym;
  
        master_detail_where = master_detail_where || res_str || ' = :%' || detail_field_count;
      end
    end

    for select res_str
    from hp_split_str(:master_field, ';')
    into :res_str
    do
    begin
      master_field_count = master_field_count + 1;
      master_detail_where = replace(master_detail_where, '%'||master_field_count, res_str);
    end

    if (master_detail_where <> '') then
      master_detail_where = ' (' || master_detail_where || ') ';
  end

  for select ufp_code, ufp_origin_name, ufp_editor
  from uxfrparam
  where ufp_ufr_code = :fr_code and
        ufp_is_param = 0 and
        ufp_is_delete = 0 and
        ufp_in_use = 1
  order by ufp_index
  into ufp_code, field_name, ufp_editor
  do
  begin
    if (field_list <> '') then
      field_list = field_list || ', ' || enter_sym;

    if (ufp_editor in ('blob_bls', 'blob')) then
    begin
      field_list = field_list || field_name;
      field_list = field_list || ', ' || enter_sym;
      field_list = field_list || 'uxf$blob(' || field_name || ', 1) as ' || field_name || '$json';
      field_list = field_list || ', ' || enter_sym;
      field_list = field_list || 'uxf$blob(' || field_name || ', 0) as ' || field_name || '$text';

      if (ufp_editor = 'blob') then
      begin
        field_list = field_list || ', ' || enter_sym;
        field_list = field_list || 'cast(null as d_blob_binary) as ' || field_name || '$blob';
        field_list = field_list || ', ' || enter_sym;
        field_list = field_list || 'cast(null as d_str_255) as ' || field_name || '$file';
      end

      ufp_editor = 'blob_json';
    end else
    if (ufp_editor = 'blob_json') then
    begin
      field_list = field_list || field_name || ' as ' || field_name || '$json';
      field_list = field_list || ', ' || enter_sym;
      field_list = field_list || 'iif(coalesce(' || field_name || ', '''') <> '''', ''Файл(ы)'', ''(Пусто)'') as ' || field_name || '$text';
    end
    else
    if (need_secure_mask = 1 and field_name <> search_text_field) then
      field_list = field_list || 'uxf$mask(' || field_name || ', ''' || ufp_editor || ''') as ' || field_name;
    else
      field_list = field_list || field_name;

    ufi_ua_code = '';
    ua_uxa_code = '';
    ua_usr_text_field = '';
    buttonedit_sql = '';

    if (ufp_editor in ('buttonedit', 'dialogedit')) then
    begin
      select ua_code, ua_uxa_code, ua_usr_text_field
      from uxfrparam, uxaction
      where ufp_code = :ufp_code and
            ufp_ufr_code = ua_ux_code and
            ufp_origin_name = ua_usr_id_field and
            ua_usr_id_field = :field_name and
            ua_ux_code = :fr_code
      into ufi_ua_code, ua_uxa_code, ua_usr_text_field;

      if (coalesce(ufi_ua_code, '') = '') then
        exception e_default f_format('%1 action not found, ufp_code: %2', ufp_editor, ufp_code);

      if (ua_usr_text_field = '') then
      begin
        -- если поле нужно формировать динамически, получаем текст SQL
        select ukn_sql
        from uxalias, uxcode, uxkeyname
        where uxa_code = :ua_uxa_code and
              uxa_ux_code = ux_code and
              ux_table_name = ukn_table_name
        into buttonedit_sql;

        if (buttonedit_sql <> '') then
        begin
          alias_letter = 'a';
          buttonedit_sql = replace(buttonedit_sql, ':id', alias_letter || '.' || field_name);
          buttonedit_sql = ' (' || buttonedit_sql || ') as ' || field_name || '$text';
  
          field_list = field_list || ', ' || enter_sym || buttonedit_sql;
        end else
          exception e_default f_format('%1 SQL not found, fr_code: %1, field: %2', ufp_editor, fr_code, field_name);
      end else
        field_list = field_list || ', ' || enter_sym || ua_usr_text_field || ' as ' || field_name || '$text';
    end
  end

  if (field_list <> '') then
    sql_text = 'select ' || enter_sym || field_list || enter_sym || 'from ' || ufd_source;
  else
  begin
    if (ux_kind = 'uxcommand') then
    begin
      sql_text = 'execute procedure ' || ufd_source;
      sql_is_exec = 1;
    end
    else
      exception e_default f_format('Select SQL no fields, ux_code: %1, fr_code: %2', ux_code, fr_code);
  end

  field_list = '';

  if (ufd_is_table = 0) then
  begin
    for select ufp_origin_name
    from uxfrparam
    where ufp_ufr_code = :fr_code and
          ufp_is_param = 1 and
          ufp_is_delete = 0
    order by ufp_index
    into field_name
    do
    begin
      md_linked = 0;
      if (is_detail = 1) then
      begin
        detail_field_count = 0;
        param_index = 0;
        for select res_str
        from hp_split_str(:detail_field, ';')
        into :res_str
        do
        begin
          detail_field_count = detail_field_count + 1;
          if (field_name = res_str) then
          begin
            param_index = detail_field_count;
            break;
          end
        end

        if (param_index > 0) then
        begin
          master_field_count = 0;

          for select res_str
          from hp_split_str(:master_field, ';')
          into :res_str
          do
          begin
            master_field_count = master_field_count + 1;
            if (master_field_count = param_index) then
            begin
              field_name = res_str;
              md_linked = 1;
              break;
            end
          end

          if (md_linked = 0) then
            exception e_default f_format('Master field param not found, ux_code: %1, fr_code: %1', ux_code, fr_code);
        end
      end

      if (field_list <> '') then
        field_list = field_list || ', ';

      field_list = field_list || ':' || field_name;
    end

    if (field_list <> '') then
      sql_text = sql_text || '(' || field_list || ')';

    if (alias_letter <> '') then
      sql_text = sql_text || ' ' || alias_letter;
  end else
  begin
    if (alias_letter <> '') then
      sql_text = sql_text || ' ' || alias_letter;

    if (detail_field <> '') then
      detail_field = ';' || replace(detail_field, ' ', '') || ';';

    if (ufr_show_kind = 1) then
    begin
      for select ufp_origin_name
      from uxfrparam
      where ufp_ufr_code = :fr_code and
            ufp_pk_field <> 0
      order by ufp_index
      into field_name
      do
      begin
        if (position(';' || field_name || ';', detail_field) > 0) then
          Continue;
  
        if (field_list <> '') then
          field_list = field_list || ' and ' || enter_sym;
  
        field_list = field_list || field_name || ' = :' || field_name;
      end
    end

    if (field_list <> '') then
    begin
      sql_text = sql_text || enter_sym || ' where (' || field_list || ')';

      if (master_detail_where <> '') then
      sql_text = sql_text || enter_sym || master_detail_where;
    end else
      if (master_detail_where <> '') then
        sql_text = sql_text || enter_sym || ' where ' || master_detail_where;
  end

  -- При ufr_show_kind = 1 в интерфейсе будем проверять кол-во строк
  -- если строк более одной, выдаем ошибку настройки

  max_rows = 0;
  if (sql_text <> '') then
  begin
    if (ufr_show_kind = 1) then
      max_rows = 2;
    else
      if (ufr_show_kind = 2 and ufd_is_table = 1) then
        max_rows = 500; -- если источник просто таблица, не даем более 500 строк
  end

  if (max_rows <> 0) then
    sql_text = sql_text || ' rows ' || max_rows;

  suspend;
end
