CREATE PROCEDURE HP_FIND_CLIENTS_SPOUSES(
    REP_DATE_END DATE = current_date)
RETURNS (
    REG_NAME STR_100,
    K_FACT_DISB_DATE DATE,
    KAL_NEW_KF_NAME STR_100,
    K_KONTRACT_NO STR_20,
    C_NAME_FIO STR_100)
AS
declare variable c_id id_type;
declare variable c_sex smallint;
declare variable ks_from_k_id id_type;
begin
  for select cl.c_id, cl.c_name_fio, cl.c_sex, ks.ks_from_k_id
      from kontstate ks
      inner join knclient knc on knc.kc_from_k_id = ks.ks_from_k_id
      inner join clients cl on cl.c_id = knc.kc_from_c_id
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            knc.kc_amount_approved > 0 and
            :rep_date_end between ks.ks_date_begin and ks.ks_date_end
      into :c_id, c_name_fio, c_sex, ks_from_k_id
  do
  begin
    if (exists(select count(*)
               from hp_client_relation_list(:c_id, 0)
               where rel_name like '%супруг%' having count(*) > 1)) then
    begin
      :kal_new_kf_name = null;
      :k_kontract_no = null;
      :k_fact_disb_date = null;
      :reg_name = null;

      select kt.kal_new_kf_name, k.k_kontract_no, k.k_fact_disb_date, r.reg_name
      from kontract k, knaktlist kt, region r
      where k.k_id = :ks_from_k_id and
            k.k_from_kf_id = kt.kal_from_kf_id and
            r.reg_unicode = k.k_last_reg_id and
            :rep_date_end between kt.kal_date and kt.kal_date_end
      into :kal_new_kf_name, k_kontract_no, k_fact_disb_date, reg_name;

      suspend;
    end
  end
end
