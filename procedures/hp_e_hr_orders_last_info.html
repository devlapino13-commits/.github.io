CREATE PROCEDURE HP_E_HR_ORDERS_LAST_INFO(
    IN_C_ID D_ID,
    IN_DATE D_DATE)
RETURNS (
    PS_FROM_REG_ID D_INTEGER,
    PS_FROM_DEP_ID D_INTEGER,
    PS_FROM_JL_ID D_INTEGER,
    PS_IS_MOUNT D_BOOL,
    PS_TABEL_NO D_STR_10,
    PS_TRANSFER_OFFICE D_INTEGER,
    RES_WORK_KIND D_INTEGER,
    HM_MOUNT D_INTEGER,
    RES_FULL_INFO D_STR_LONG)
AS
begin
  res_full_info = '';

  select ps_from_reg_id, ps_from_dep_id, ps_from_jl_id, ps_is_mount, ps_tabel_no, ps_from_reg_id as ps_transfer_office,
  iif(ps_is_part = 1, 2, ps_is_on_staff) as res_work_kind
  from hr_pstatus
  where ps_from_c_id = :in_c_id and
        ps_date_begin <= :in_date and
        ps_status = 1 and
        ps_is_working = 1 and
        ps_date_end >= ps_date_begin
  order by ps_date_begin desc, ps_from_ho_id desc
  rows 1
  into :ps_from_reg_id,
       :ps_from_dep_id,
       :ps_from_jl_id,
       :ps_is_mount,
       :ps_tabel_no,
       :ps_transfer_office,
       :res_work_kind;

  select res_full_info
  from hp_e_hr_orders_pers_info(:in_c_id, :in_date)
  into res_full_info;

  if (ps_is_mount is null) then
    ps_is_mount = 0;

  if (res_work_kind is null) then
    res_work_kind = 0;

  hm_mount = 0;

  select hm.hm_mount
  from hr_pstatus ps
  inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id and
        ho.ho_order_type = 1 and
        ho.ho_status = 1
  left join hr_mount hm on hm.hm_ps_from_ho_id = ps.ps_from_ho_id and
        hm.hm_from_ps_id = ps.ps_id
  where ps.ps_from_c_id = :in_c_id and
        ps.ps_status = 1 and
        ps.ps_date_begin <= :in_date
  order by ps.ps_date_begin desc
  rows 1
  into :hm_mount;


  suspend;
end
