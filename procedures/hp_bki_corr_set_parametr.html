CREATE PROCEDURE HP_BKI_CORR_SET_PARAMETR(
    BEV_ID D_ID,
    BFV_ID D_ID,
    IS_AVAILABLE D_BOOL)
AS
declare variable bfv_id_temp d_id;
declare variable db_field d_str_100;
declare variable xml_version_id d_integer;
begin
  if (bev_id is null or bfv_id is null) then
    exit;

  if (is_available is null) then
    is_available = 0;

  select xml_version_id
  from hp_bki_get_current_xml_ver
  into xml_version_id;

  select bki_db_field
  from bki_xml_fields_ver, bki_xml_fields, bki_indicators
  where bxf_id = bfv_from_bxf_id
    and bki_code = bxf_from_bki_code
    and bfv_from_bxv_id = :xml_version_id
    and bfv_id = :bfv_id
  into db_field;

  if (is_available = 1) then
  begin
    if (not exists(select 1
      from bki_send_data_corr
      where bsd_from_bfv_id = :bfv_id
        and bsd_from_bev_id = :bev_id))
    then
    begin
      if (coalesce(db_field, '') not starts '<') then
        insert into bki_send_data_corr (bsd_from_bev_id, bsd_from_bfv_id)
        values (:bev_id, :bfv_id);
      else -- если объект или массив
        -- отметим детей, если нет ни одного отмеченного
        if (not exists(
          with recursive tab as(
            select bfv_id, bki_db_field, bfv_parent_id
            from bki_xml_fields_ver, bki_xml_fields, bki_indicators
            where bxf_id = bfv_from_bxf_id
              and bki_code = bxf_from_bki_code
              and bfv_from_bxv_id = :xml_version_id
            order by bfv_position
          ), t1 (id, db_field, parent_id) as(
            select bfv_id, bki_db_field, bfv_parent_id
            from tab
            where bfv_parent_id = :bfv_id
              and (bki_db_field starting with '<'
                or exists(select 1 from bki_send_data_corr where bsd_from_bfv_id = bfv_id and bsd_from_bev_id = :bev_id))
            union all
            select bfv_id, bki_db_field, bfv_parent_id
            from t1, tab
            where bfv_parent_id = id
              and (bki_db_field starting with '<'
                or exists(select 1 from bki_send_data_corr where bsd_from_bfv_id = bfv_id and bsd_from_bev_id = :bev_id))
          )
          select 1
          from t1
          where coalesce(db_field, '') not starting with '<'))
        then
          for select bfv_id
          from bki_xml_fields_ver
          where bfv_parent_id = :bfv_id
          order by bfv_position
          into bfv_id_temp
          do
            execute procedure hp_bki_corr_set_parametr(:bev_id, :bfv_id_temp, :is_available);

    end
  end
  else
  begin
    if (coalesce(db_field, '') not starting with '<') then
      delete from bki_send_data_corr
      where bsd_from_bfv_id = :bfv_id
        and bsd_from_bev_id = :bev_id;
    else
      -- удалим всех детей
      for select bfv_id
      from bki_xml_fields_ver
      where bfv_parent_id = :bfv_id
      into bfv_id_temp
      do
        execute procedure hp_bki_corr_set_parametr(:bev_id, :bfv_id_temp, :is_available);

  end
end
