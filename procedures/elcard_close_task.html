CREATE PROCEDURE ELCARD_CLOSE_TASK(
    ESM_ID D_ID)
AS
begin
  /*
   Процедура предназначена для закрытия задачи перед проверкой карты / отправкой денег на карту,
   чтобы если удалось отправить дееньги на карту и не удалось записать результат в БД, исключить ситуацию когда
   1) при проверке активности карты платежно поручиение уже создано и из-за глюков сервиса оно пытается создаться снова
   2) деньги могут повторно улететь на карту клиенту.

   Процедура вызывается сразу перед самой провркой активности карты/отправкой на карту.
  */

  if (esm_id is null) then
    exception e_default 'ESM_ID is null';

  if (not exists(select *
                 from extsysmain esm
                 where esm.esm_id = :esm_id and
                       esm.esm_project = 'elcard_disb' and
                       esm.esm_action in ('CardCheck', 'CardDisb') and
                       esm.esm_status = 0)) then
  begin
    exception e_default 'ESM_ID is not correct';
  end

  update extsysmain em
  set em.esm_status = -999
  where em.esm_id = :esm_id;
end
