CREATE PROCEDURE AAREP_ANTIFRAUD_MATCH_REPOR(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    INN D_STR_20,
    STATUS_DATE_TIME D_DATETIME,
    CATEGORIES D_STR_100)
AS
declare variable BLS_ID D_ID;
declare variable JSON_INFO D_BLOB_TEXT;
begin
  for select ac.ac_inn, eil.eil_from_bls_id, eil.eil_status_datetime
  from antifraud_check ac
  inner join extinfolist eil on eil.eil_id = ac.ac_from_eil_id
  where ac.ac_status = 2 and
        eil.eil_date between :rep_date_begin and :rep_date_end
  into :inn, :bls_id, :status_date_time
  do
  begin
    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into :json_info;

    if (f_json_str_get_count(:json_info, 'items[0].categories') > 0) then
      categories = f_json_str_get_value(:json_info, 'items[0].categories[0]', '');
    else
      categories = 'Пусто';

    suspend;
  end
end
