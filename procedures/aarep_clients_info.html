CREATE PROCEDURE AAREP_CLIENTS_INFO(
    REP_DATE_END DATE)
RETURNS (
    KS_FROM_K_ID DECIMAL(18,0),
    KS_DATE_BEGIN DATE,
    KS_DATE_END DATE,
    K_KONTRACT_NO VARCHAR(20),
    K_APPLICATE_DATE DATE,
    K_DISB_DATE DATE,
    K_KREDLINE_END DATE,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    PR_ID INTEGER,
    PR_LOAN_TYPE SMALLINT,
    PR_CODE VARCHAR(100),
    PR_NAME VARCHAR(50),
    KC_STATUS SMALLINT,
    KC_STATUS_STR VARCHAR(20),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_CREDIT_TARGET VARCHAR(255),
    K_CIRCLE INTEGER,
    K_PERIOD DECIMAL(15,2),
    K_KOLICH_REPAY INTEGER,
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_SEX SMALLINT,
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_FULL_ADDRESS VARCHAR(255),
    C_KOL_FAMILY INTEGER,
    C_KOL_CHILD INTEGER,
    C_NATION INTEGER,
    C_NATION_STR VARCHAR(100),
    C_CITIZEN INTEGER,
    C_CITIZEN_STR VARCHAR(100),
    C_EDUCATION INTEGER,
    C_EDUCATION_STR VARCHAR(100),
    C_FAMMILY INTEGER,
    C_FAMMILY_STR VARCHAR(100),
    KAL_NEW_KF_OFFICER DECIMAL(18,0),
    OFFICER VARCHAR(100),
    OFF_NATION_STR VARCHAR(100),
    OFF_BIRTH_DATE DATE,
    OFF_SEX SMALLINT,
    OFF_FAMMILY_STR VARCHAR(100),
    OFF_CITIZEN_STR VARCHAR(20),
    PS_IS_WORKING_TEXT VARCHAR(50),
    PS_IS_ON_STAFF_STR VARCHAR(20),
    AA_KC_STAGE VARCHAR(50),
    PS_DATE DATE,
    JL_NAME VARCHAR(100),
    KC_STAGE_MONTH INTEGER,
    KC_STAGE_YEARS INTEGER,
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KC_CIRCLE INTEGER,
    BT_ID INTEGER,
    BT_NAME VARCHAR(50),
    BS_ID INTEGER,
    BS_NAME VARCHAR(50),
    REG_ID INTEGER,
    REG_NAME VARCHAR(61),
    BRANCHE_ID INTEGER,
    BRANCHE_NAME VARCHAR(61),
    DISTRICT_ID INTEGER,
    DISTRICT_NAME VARCHAR(61),
    K_KONT_BALANCE DECIMAL(18,4),
    K_CLI_BALANCE DECIMAL(18,4),
    DELIQ_OS DECIMAL(18,4),
    DELIQ_PERC DECIMAL(18,4),
    DELIQ_DAY INTEGER,
    DELIQ_CLI_OS DECIMAL(18,4),
    DELIQ_CLI_PERC DECIMAL(18,4),
    STATUS_KONT VARCHAR(20),
    STATUS_DELINQ VARCHAR(20),
    KOL_CHECK_RESULT SMALLINT,
    DELIQ_1_3 INTEGER,
    DELIQ_4_10 INTEGER,
    DELIQ_10_ INTEGER,
    DELIQ_DAY_ALL_OLD INTEGER,
    DELIQ_DAY_ALL INTEGER,
    DELIQ_DAY_MAX INTEGER)
AS
declare variable old_k_id decimal(18,0);
declare variable ok_from_k_id decimal(18,0);
declare variable ok_sh_date date;
declare variable ok_os_deliq decimal(18,4);
declare variable ok_perc_deliq integer;
declare variable is_deliq smallint;
declare variable cli_dolya decimal(18,4);
declare variable ok_sh_date_min date;
declare variable deliq_day_old integer;
declare variable old_k_id_1 decimal(18,0);
declare variable old_officer_id decimal(18,0);
declare variable ps_is_working smallint;
declare variable ps_is_on_staff integer;
begin
-- oksummary_fix

  old_k_id = 0;
  old_k_id_1 = 0;
  k_cli_balance = 0;
  old_officer_id = 0;
  for select ks_from_k_id, ks_date_begin, ks_date_end, k_kontract_no, k_applicate_date, k_disb_date, k_kredline_end,
             k_kolich_client_approved, k_amount_approved, k_credit_target, k_circle, k_period, k_kolich_repay, pr_id,
             pr_loan_type, pr_code, pr_name, kc_status,
             case kc_status
               when 0 then 'Участник'
               when 1 then 'Лидер'
               when 2 then 'Казначей'
             end,
             cli.c_id, cli.c_name_fio, cli.c_sex, cli.c_birth_date, cli.c_pass_no, adl_address_str, cli.c_kol_family,
             cli.c_kol_child, d1.rd_id, d1.rd_name c_nation_str, d2.rd_id, d2.rd_name c_citizen_str, d3.rd_id,
             d3.rd_name c_education_str, d4.rd_id, d4.rd_name c_fammily_str, kal_new_kf_officer, off.c_name_fio officer,
             d5.rd_name off_nation_str, off.c_birth_date off_birth_date, off.c_sex off_sex, d6.rd_name off_fammily_str,
             d7.rd_name offcitizen_str, kc_amount_approved, kc_circle, bt_id, bt_name, bs_id, bs_name, reg1.reg_unicode,
             reg1.reg_code || ' ' || reg1.reg_name, reg2.reg_unicode, reg2.reg_code || ' ' || reg2.reg_name,
             reg3.reg_unicode, reg3.reg_code || ' ' || reg3.reg_name

      from kontstate, knclient, kontract, addrlist, addrclient, clients cli, refdata d1, refdata d2, refdata d3, refdata d4, knaktlist, clients off, product, bussubtype, bustype, refdata d5, refdata d6, refdata d7, region reg1
      left join region reg2 on (reg1.parent = reg2.reg_unicode)
      left join region reg3 on (reg2.parent = reg3.reg_unicode)
      where ks_state = 1 and
            ks_ol_is_delete = 0 and
            kc_from_k_id = ks_from_k_id and
            ks_from_k_id = k_id and
            kc_from_c_id = cli.c_id and
            kc_amount_approved > 0 and
            kc_amount_approved is not null and
            cli.c_nation = d1.rd_id and
            cli.c_citizen = d2.rd_id and
            cli.c_education = d3.rd_id and
            cli.c_family = d4.rd_id and
            kal_from_kf_id = k_from_kf_id and
            knaktlist.is_delete = 0 and
            ks_date_begin between kal_date and kal_date_end and
            kal_new_kf_officer = off.c_id and
            bs_from_bt_id = bt_id and
            kc_busuness_subtype = bs_id
            /*and 'now' between ks_date_begin and ks_date_end*/
            and
            reg1.reg_unicode = kal_new_reg_id and
            d5.rd_id = off.c_nation and
            d6.rd_id = off.c_family and
            d7.rd_id = off.c_citizen and
            k_from_pr_id = pr_id and
            adc_address_type = 1 and
            adc_from_c_id = cli.c_id and
            adc_from_adl_id = adl_id
      order by ks_from_k_id

      into :ks_from_k_id, :ks_date_begin, :ks_date_end, :k_kontract_no, :k_applicate_date, :k_disb_date,
           :k_kredline_end, :k_kolich_client_approved, :k_amount_approved, :k_credit_target, :k_circle, :k_period,
           :k_kolich_repay, :pr_id, :pr_loan_type, :pr_code, :pr_name, :kc_status, :kc_status_str, :c_id, :c_name_fio,
           :c_sex, :c_birth_date, :c_pass_no, :c_full_address, :c_kol_family, :c_kol_child, :c_nation, :c_nation_str,
           :c_citizen, :c_citizen_str, :c_education, :c_education_str, :c_fammily, :c_fammily_str, :kal_new_kf_officer,
           :officer, :off_nation_str, :off_birth_date, :off_sex, :off_fammily_str, :off_citizen_str,
           :kc_amount_approved, :kc_circle, :bt_id, :bt_name, :bs_id, :bs_name, :reg_id, :reg_name, :branche_id,
           :branche_name, :district_id, :district_name

  do
  begin
    kol_check_result = null;
    select count(*)
    from contacts /*вычисляет кол-во подтвержденных контактов*/
    where cc_from_c_id = :c_id and
          cc_check_result = 1
    into :kol_check_result;

    if (:rep_date_end between :ks_date_begin and :ks_date_end) then
    /*Если контракт активный делает следующее*/
    begin
      k_cli_balance = null;
      status_kont = 'Активный';
      /*Вычисления активного портфеля контракта :KS_FROM_K_ID*/
      if (old_k_id <> :ks_from_k_id) then
      /*Проверяет на повторность контракта, если контракт повторяеться то не будет повторно считать*/
      begin
        /*Вычисляет баланс контракта*/
        select oks_balance
        from oksummary
        where oks_from_k_id = :ks_from_k_id and
              oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end and
              oks_close_state = 0
        into :k_kont_balance;

        ok_sh_date = null;
        ok_os_deliq = null;
        ok_perc_deliq = null;

        deliq_os = 0;
        deliq_perc = 0;
        deliq_day = 0;
        deliq_day_max = 0;
        deliq_cli_os = null;
        deliq_cli_perc = null;

        status_delinq = '';
        ok_sh_date_min = :rep_date_end;
        is_deliq = 0;

        for /*Вычисляет просроченные контракты*/
            select ok_from_k_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
            from operkred
            where ok_ol_is_active = 1 and
                  ok_type < 7 and
                  ok_ol_date <= :rep_date_end and
                  ok_sh_date <= :rep_date_end and
                  ok_from_k_id = :ks_from_k_id and
                  ok_from_c_id = 0
            group by ok_from_k_id, ok_sh_date
            having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0
            into :ok_from_k_id, :ok_sh_date, :ok_os_deliq, :ok_perc_deliq
        do
        begin
          if (:ok_os_deliq + :ok_perc_deliq > 0) then
          begin
            is_deliq = 1;
            if (ok_sh_date_min > :ok_sh_date) then
              ok_sh_date_min = :ok_sh_date;
          end
          deliq_os = deliq_os + ok_os_deliq;
          deliq_perc = deliq_perc + ok_perc_deliq;
        end

        if (is_deliq = 1) then
        begin
          deliq_day = cast((:rep_date_end - ok_sh_date_min) as integer);
          status_delinq = 'Просрочка';
          deliq_day_max = deliq_day;
        end
      end
      cli_dolya = :kc_amount_approved / :k_amount_approved;/*коэффицент доли клиента*/
      k_cli_balance = cli_dolya * :k_kont_balance;/*Вычисляет долю портфеля клиента*/
      deliq_cli_os = cli_dolya * :deliq_os;
      deliq_cli_perc = cli_dolya * deliq_perc;
      old_k_id = :ks_from_k_id;
    end
    else
    begin
      k_cli_balance = null;
      k_kont_balance = null;
      deliq_os = null;
      deliq_perc = null;
      deliq_day = null;
      ok_sh_date_min = null;
      deliq_cli_os = null;
      deliq_cli_perc = null;
      status_kont = 'Закрытый';
      status_delinq = null;
    end

    if (old_k_id_1 <> :ks_from_k_id) then
    /*Вычисляет дни просроченных выплат*/
    begin
      deliq_day_old = 0;
      deliq_1_3 = 0;
      deliq_4_10 = 0;
      deliq_10_ = 0;
      deliq_day_all_old = 0;
      for select max(ok_ol_date) - ok_sh_date
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_k_id = :ks_from_k_id and
                ok_type in (4, 5) and
                ok_from_c_id = 0
          group by ok_sh_date
          having max(ok_ol_date) - ok_sh_date > 0
          into :deliq_day_old
      do
      begin
        if (:deliq_day_old between 1 and 3) then
          deliq_1_3 = deliq_1_3 + 1;
        if (:deliq_day_old between 4 and 10) then
          deliq_4_10 = deliq_4_10 + 1;
        if (:deliq_day_old > 10) then
          deliq_10_ = deliq_10_ + 1;
        deliq_day_all_old = deliq_day_all_old + :deliq_day_old;
        if (:deliq_day_max < :deliq_day_old) then
          deliq_day_max = :deliq_day_old;
      end
    end
    old_k_id_1 = :ks_from_k_id;

    deliq_day_all = deliq_day_all_old + deliq_day;

    /*Вычисляет параметры Кредитного служащего*/
    if (:old_officer_id <> :kal_new_kf_officer) then
    /*проверка на повторно не будет считать*/
    begin
      ps_is_working = null;
      ps_is_working_text = null;

      select ps_is_working
      from hr_pstatus, hr_orders, sa_worktype
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_c_id = :kal_new_kf_officer and
            ps_from_ho_id = ho_id and
            ho_order_type = sawt_id
      into :ps_is_working;
      if (ps_is_working = 1) then
        ps_is_working_text = 'Работает';
      else
      begin
        if (ps_is_working = 0) then
          ps_is_working_text = 'Уволен';
        else
          ps_is_working_text = 'Не оперделен';
      end
      ps_is_on_staff = null;
      jl_name = null;
      select ps_is_on_staff, jl_name
      from hr_pstatus, joblist
      where ps_from_c_id = :kal_new_kf_officer and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id
      into :ps_is_on_staff, :jl_name;
      if (ps_is_on_staff is not null) then
        if (ps_is_on_staff = 1) then
          ps_is_on_staff_str = 'Штатный';
        else
          ps_is_on_staff_str = 'Внештатный';
      else
        ps_is_on_staff_str = 'Приказ не оформлен';

      ps_date = null;
      kc_stage_month = null;
      kc_stage_years = null;
      aa_kc_stage = null;

      select ps_date_begin
      from hr_pstatus, joblist, jobkind /*Вычисляет стаж КС*/
      where ps_from_c_id = :kal_new_kf_officer and
            (ps_date_begin is not null) and
            jl_id = ps_from_jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')
      order by ps_date_begin
      rows 1
      into :ps_date;
      if (ps_date is null) then
        aa_kc_stage = 'XX Стаж не определен';
      else
      begin
        if (ps_is_working = 1) then
        begin
          /* kc_stage_month=f_ageinmonths(:ps_date, :rep_date_end);  /*между двумя датами определяет сколько месяцев*/
          kc_stage_month = datediff(month, :ps_date, :rep_date_end);
          kc_stage_years = kc_stage_month / 12;
          if (kc_stage_years < 1) then
            aa_kc_stage = '0. Стаж КС менее года';
          else
          if (kc_stage_years = 1) then
            aa_kc_stage = '1. Стаж КС год';
          else
          if (kc_stage_years > 1) then
            aa_kc_stage = '2. Стаж КС более года';
        end
      end
    end
    old_officer_id = :kal_new_kf_officer;

    suspend;
  end
end
