CREATE PROCEDURE AAREP_VIPS_KF_INFO(
    REP_REGION_ID INTEGER,
    REP_DATE DATE,
    LO_ID ID_TYPE)
RETURNS (
    OKS_REGION_ID ID_TYPE,
    OKS_KC_ID ID_TYPE,
    REG_NAME STR_100,
    KF_OFFICER_NAME STR_100,
    C_STAGE INTEGER,
    OKS_KC_BALANCE CURR_TYPE,
    NOT_OWN_CL_QTY INTEGER,
    IS_OWN_CL_QTY INTEGER,
    NOT_OWN_PAR30 PERC_TYPE,
    IS_OWN_PAR30 PERC_TYPE)
AS
declare variable oks_region_id_tmp id_type;
begin
  /* ФИО ВиПС
  Стаж работы
  Кол-во клиентов (своих)
  Кол-во клиентов (переданных)
  ПАР-30 (по своим)
  ПАР-30 (по переданным)
 */
  if (coalesce(lo_id, 0) > 1000000000003) then
  begin
    if (coalesce(rep_region_id, 0) = 0) then
      rep_region_id = 1000000;
  end
  else
    lo_id = 0;

  for select oks_region_id, oks_kc_id, --
             sum(oks_balance),
--       Переданные кредиты
             sum(iif(ks_kc_id <> oks_kc_id, k_kolich_client_approved, 0)),
             sum(iif(ks_kc_id <> oks_kc_id and :rep_date - oks_delinq_date > 30, oks_balance, 0)),
--       Собственные кредиты
             sum(iif(ks_kc_id = oks_kc_id, k_kolich_client_approved, 0)),
             sum(iif(ks_kc_id = oks_kc_id and :rep_date - oks_delinq_date > 30, oks_balance, 0))
      from oksummary, kontract k, kontstate ks, region_list(:rep_region_id, 1, 3, 1)
      where oks_region_id = id_list and
            :rep_date between oks_date_begin and oks_date_end and
--OKS_BALANCE > 0 and
            oks_kc_id > 1000000000003 and
            oks_kc_id = iif(:lo_id <> 0, :lo_id, oks_kc_id) and
            :rep_date - oks_delinq_date <= 360 and
            oks_from_k_id = k_id and
            oks_close_state = 0 and
            k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date between ks_date_begin and ks_date_end
      group by 1, 2
      into oks_region_id, oks_kc_id, oks_kc_balance, not_own_cl_qty, not_own_par30, is_own_cl_qty, is_own_par30
  do
  begin
    if (coalesce(oks_region_id_tmp, 0) <> oks_region_id) then
    begin
      select reg_name
      from region r
      where r.reg_unicode = :oks_region_id
      into reg_name;

      oks_region_id_tmp = oks_region_id;
    end

    select c_name_fio
    from clients
    where c_id = :oks_kc_id
    into :kf_officer_name;

    c_stage = 0;
    select ss1.c_stage
    from aarep_staj_ks(:oks_kc_id, :rep_date) ss1
    into :c_stage;

    suspend;
  end
end
