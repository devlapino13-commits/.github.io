CREATE PROCEDURE FM_LO_PENALTY(
    UN_ID ID_TYPE,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable counter d_integer;
declare variable c_name_fio str_100;
declare variable block_datetime d_datetime;
declare variable unblock_datetime d_datetime;
declare variable lp_amount curr_type;
declare variable lo_id id_type;
declare variable ema_id d_id;
declare variable ma_id d_integer;
declare variable ema_datetime d_datetime;
begin

    res_html = (select res_style
                from fm_html_styles) || '<body>';

    res_html = BLOB_APPEND(res_html, f_format('<h1 class="green_text">%1</h1>', 'Отчет по онлайн-заявкам'));
    res_html = BLOB_APPEND(res_html, '<table width="100%"><col width="5%"><col width="25%"><col width="25%"><col width="25%"><col width="20%">' || '<tr><th>№</th>' || '<th>ФИО клиента</th>' || '<th>Дата и время блок. заявки</th>' || '<th>Дата и время разблок. заявки</th>' || '<th>Сумма штрафа</th></tr>');
    counter = 0;

    select un_from_c_id
    from username
    where un_id = :un_id
    into lo_id;

    for select ema_id,
               lp_amount,
               c_name_fio,
               lp_datetime,
               ema_datetime
    from lo_penalty
    inner join extminiapp on ema_id = lp_from_ema_id
    left join clients on c_id = ema_from_c_id
    where lp_date between :rep_date_begin and :rep_date_end
    and lp_is_active = 1
    and lp_from_c_id = :lo_id
    order by lp_date
    into ema_id, lp_amount, c_name_fio, unblock_datetime, ema_datetime
    do
    begin
        counter = counter + 1;
        block_datetime = null;

        select emal_datetime
        from extminiapp_log
        where emal_from_ema_id = :ema_id
        and emal_datetime < :unblock_datetime
        order by 1 desc
        rows 1
        into block_datetime;

        if (block_datetime is null) then
            block_datetime = ema_datetime;

        res_html = BLOB_APPEND(res_html, '<tr><td align="center">' || counter || '</td>' || '<td align="center">' || c_name_fio || '</td>' || '<td align="center">' || f_datetimetostr(block_datetime) || '</td>' || '<td align="center">' || f_datetimetostr(unblock_datetime) || '</td>' || '<td align="right">' || f_numtostr(lp_amount, 2) || '</td></tr>');
    end
    res_html = BLOB_APPEND(res_html, '</table>');

    res_html = BLOB_APPEND(res_html, f_format('<h1 class="green_text">%1</h1>', 'Отчет по заявкам на заем'));
    res_html = BLOB_APPEND(res_html, '<table width="100%"><col width="5%"><col width="25%"><col width="25%"><col width="25%"><col width="20%">' || '<tr><th>№</th>' || '<th>ФИО клиента</th>' || '<th>Дата и время создания заявки</th>' || '<th>Дата и время просрочки заявки</th>' || '<th>Сумма штрафа</th></tr>');


    block_datetime = null;
    unblock_datetime = null;
    counter = 0;

    for select lp_from_ma_id,
               lp_amount,
               c_name_fio,
               lp_datetime
    from lo_penalty
    inner join miniappclient on mac_from_ma_id = lp_from_ma_id
    left join clients on c_id = mac_from_c_id
    where lp_date between :rep_date_begin and :rep_date_end
    and lp_is_active = 1
    and lp_from_c_id = :lo_id
    order by lp_date
    into ma_id, lp_amount, c_name_fio, unblock_datetime
    do
    begin
        counter = counter + 1;  
        select ma_date_save
        from miniapp
        where ma_id = :ma_id
        order by ma_id, ma_number
        rows 1
        into block_datetime;

    res_html = BLOB_APPEND(res_html, '<tr><td align="center">' || counter || '</td>' || '<td align="center">' || c_name_fio || '</td>' || '<td align="center">' || f_datetimetostr(block_datetime) || '</td>' || '<td align="center">' || f_datetimetostr(unblock_datetime) || '</td>' || '<td align="right">' || f_numtostr(lp_amount, 2) || '</td></tr>');
    end
    res_html = BLOB_APPEND(res_html, '</table></body>');
    suspend;
end
