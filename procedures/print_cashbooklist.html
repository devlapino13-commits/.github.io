CREATE PROCEDURE PRINT_CASHBOOKLIST(
    BOOK_ID D_INTEGER)
RETURNS (
    CBL_ID D_INTEGER,
    CBL_FROM_CBM_ID D_INTEGER,
    CBL_DOC_NO D_INTEGER,
    CBL_IS_IPF D_BOOL,
    CBL_FROM_OL_ID D_ID,
    CBL_CLIENT_FIO D_STR_100,
    CBL_AMOUNT_PLUS D_CURRENCY,
    CBL_AMOUNT_MINUS D_CURRENCY,
    CBL_CASH_ACC_NO D_INTEGER,
    CBL_CASH_ACCOUNT D_STR_20,
    CBL_CORR_ACCOUNT D_STR_255,
    CBL_DIGITAL_SIGNATURE D_SMALLINT)
AS
declare variable C_ID D_ID;
begin
  for select cbl_id,
             cbl_from_cbm_id,
             cbl_doc_no,
             cbl_is_ipf,
             cbl_from_ol_id,
             cbl_client_fio,
             cbl_amount_plus,
             cbl_amount_minus,
             cbl_cash_acc_no,
             cbl_cash_account,
             cbl_corr_account
      from cashbooklist
      where cbl_from_cbm_id = :book_id
      order by cbl_id
      into :cbl_id,
           :cbl_from_cbm_id,
           :cbl_doc_no,
           :cbl_is_ipf,
           :cbl_from_ol_id,
           :cbl_client_fio,
           :cbl_amount_plus,
           :cbl_amount_minus,
           :cbl_cash_acc_no,
           :cbl_cash_account,
           :cbl_corr_account
  do
  begin
    select opc.opc_is_online   -- RFCKG-4548
    from opercash opc
    where opc.opc_from_ol_id = :cbl_from_ol_id
    into :cbl_digital_signature;

    suspend;
  end
end
