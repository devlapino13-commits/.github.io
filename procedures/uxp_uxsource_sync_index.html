CREATE PROCEDURE UXP$UXSOURCE_SYNC_INDEX(
    US_UUID D_UUID,
    US_UUID_CODE D_STR_CODE = null)
AS
declare variable us_is_table d_bool;
declare variable index_name d_str_code;
declare variable us_source d_str_code;
declare variable us_kind d_smallint;
declare variable source_name d_str_code;
begin
  -- Сихронизирует индексы БД с таблицей uxIndex

  if (coalesce(us_uuid, '') = '') then
    us_uuid = uxf$uniq(us_uuid_code);

  select us_code, us_is_table, us_kind
  from ux$source
  where us_uuid = :us_uuid
  into us_source, us_is_table, us_kind;

  update ux$index
  set uix_state = -2
  where uix_source = :us_source and
        uix_virtual = 0;

  if (us_kind = 0 and us_is_table = 1) then
  begin
    for select i.rdb$index_name
    from rdb$indices i
    where rdb$relation_name = upper(:us_source) and
          (i.rdb$unique_flag = 1 or i.rdb$foreign_key is not null)
    into index_name
    do
    begin
      source_name = '';
      select source_name
      from uxp$index_load(:index_name)
      into :source_name;
  
      if (source_name = '') then
        exception e_default f_format('Index source not found: %1', index_name);
    end
  end

  delete from ux$index
  where uix_source = :us_source and
        uix_state = -2;

  update ux$index
  set uix_state = 0
  where uix_source = :us_source and
        uix_state <> 0;
end
