CREATE PROCEDURE CR_JSON_NEW_ACCOUNT(
    C_ID D_ID,
    SHOW_ADD_INFO SMALLINT = 0)
RETURNS (
    NEW_ACCOUNT D_STR_2048)
AS
declare variable cycle smallint;
declare variable stat_change smallint;
declare variable status smallint;
declare variable account_no varchar(20);
declare variable acc_prty smallint;
declare variable tranz_acct varchar(50);
declare variable sta_coment varchar(255);
declare variable auth_bonus d_currency;
declare variable ab_expirity date;
declare variable crd_expiry date;
declare variable atm_limit d_currency;
declare variable adjust_flag smallint;
declare variable pay_code integer;
declare variable pay_freq integer;
declare variable calcul_mode integer;
declare variable c_accnt_type d_str_lang;
declare variable cond_set d_str_lang;
declare variable card_acct d_str_code;
declare variable ccy d_str_lang;
declare variable min_bal d_currency;
declare variable crd d_currency;
declare variable non_reduce_bal d_currency;
begin
    :cycle = 0;
    :stat_change = 1;
    :status = 0;
    :account_no = null;
    :acc_prty = 1;
    :tranz_acct = null;
    :sta_coment = null;
    :auth_bonus = 0;
    :ab_expirity = null;
    :crd_expiry = null;
    :atm_limit = 0;
    :adjust_flag = 0;
    :pay_code = 0;
    :pay_freq = 0;
    :calcul_mode = 0;

    c_accnt_type = '00';
    ccy = 'KGS';

    select cca.cca_code, cca.cca_min_balance, cca.cc_condition_set /*cca.cca_product*/, cca.cca_credit_limit, cca.cca_blocked_amount
    from cardclientacc cca
    inner join cardclientid cci on cca.cca_from_cci_id = cci.cci_id
    where cci.cci_from_c_id = :c_id
    into :card_acct, :min_bal, :cond_set, :crd, :non_reduce_bal;

    new_account = '{' ||
    f_json_make_pair('c_accnt_type', :c_accnt_type, 'str', 1) ||
    f_json_make_pair('card_acct', :card_acct, 'str', 1) ||
    f_json_make_pair('ccy', :ccy, 'str', 1) ||
    f_json_make_pair('cycle', :cycle, 'num', 1) ||
    f_json_make_pair('min_bal', :min_bal, 'num', 1) ||
    f_json_make_pair('cond_set', :cond_set, 'str', 1) ||
    f_json_make_pair('crd', :crd, 'num', 1) ||
    f_json_make_pair('stat_change', :stat_change, 'num', 1) ||
    f_json_make_pair('non_reduce_bal', :non_reduce_bal, 'num', 1) ||
    f_json_make_pair('status', :status, 'num', iif(show_add_info = 1, 1, 0));

    if (:show_add_info = 1) then
        begin
        :new_account = :new_account ||
            f_json_make_pair('account_no', :account_no, 'str', 1) ||
            f_json_make_pair('acc_prty', :acc_prty, 'num', 1) ||
            f_json_make_pair('tranz_acct', :tranz_acct, 'str', 1) ||
            f_json_make_pair('sta_coment', :sta_coment, 'str', 1) ||
            f_json_make_pair('auth_bonus', :auth_bonus, 'num', 1) ||
            f_json_make_pair('ab_expirity', :ab_expirity, 'date', 1) ||
            f_json_make_pair('crd_expiry', :crd_expiry, 'date', 1) ||
            f_json_make_pair('atm_limit', :atm_limit, 'num', 1) ||
            f_json_make_pair('adjust_flag', :adjust_flag, 'num', 1) ||
            f_json_make_pair('pay_code', :pay_code, 'num', 1) ||
            f_json_make_pair('pay_freq', :pay_freq, 'num', 1) ||
            f_json_make_pair('calcul_mode', :calcul_mode, 'num');
        end

    :new_account = :new_account || '}';

    suspend;
end
