CREATE PROCEDURE HP_FACERECOGLIST_SET_RESULT(
    ID D_INTEGER,
    JSON_DATA D_STR_LONG)
AS
declare variable res d_integer;
declare variable koef d_percent;
declare variable mes d_str_255;
declare variable sec d_currency;
begin
  -- Процедура для сохранения результата сверки фотографий

  -- пример: json_data
  -- {"result":1,"koef":0.5,"message":"","second":5.2}
  --result 1 - успех; -1 - ошибка (message - детали ошибки)

  if (trim(coalesce(json_data, '')) = '') then
    exit;

  res = f_json_str_get_value(json_data, 'result', '-1');
  koef = f_json_str_get_value(json_data, 'koef', '-1');
  mes = f_json_str_get_value(json_data, 'message', '');
  sec = f_json_str_get_value(json_data, 'second', '-1');

  --if (res = -1) then
  --  exception e_default 'value for "result" not found';

  if (koef = -1) then
    exception e_default 'value for "koef" not found';

  if (sec = -1) then
    exception e_default 'value for "second" not found';

  update facerecoglist
  set frl_state = :res,
      frl_koef = :koef,
      frl_message = :mes,
      frl_check_seconds = :sec
  where frl_id = :id;
end
