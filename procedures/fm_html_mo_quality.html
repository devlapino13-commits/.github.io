CREATE PROCEDURE FM_HTML_MO_QUALITY(
    UN_ID DECIMAL(18,0),
    REP_DATE DATE,
    REG_ID INTEGER)
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable LO_FIO varchar(100);
declare variable LINE_STR varchar(1024);
declare variable DB_ID integer;
declare variable MANAGER_ID decimal(18,0);
declare variable RES_PERIOD_DATE date;
declare variable RES_QUAL_KOEF1 decimal(18,7);
declare variable RES_QUAL_KOEF2 decimal(18,7);
declare variable RES_QUAL_KOEF3 decimal(18,7);
declare variable RES_TOTAL1 integer;
declare variable RES_TOTAL2 integer;
declare variable RES_TOTAL3 integer;
declare variable RES_DELINQ1 integer;
declare variable RES_DELINQ2 integer;
declare variable RES_DELINQ3 integer;
declare variable RES_DELINQ1_ALL integer;
declare variable RES_DELINQ2_ALL integer;
declare variable RES_DELINQ3_ALL integer;
declare variable MONTH_NO integer;
declare variable PERIOD_STR varchar(100);
declare variable RES_TOTAL1_ALL integer;
declare variable RES_TOTAL2_ALL integer;
declare variable RES_TOTAL3_ALL integer;
declare variable REP_DATE_BEGIN date;
declare variable HTML_TEMPLATE D_BLOB_TEXT;
declare variable UN_NAME D_STR_CODE;
declare variable office_title d_str_255;
begin
-- Расчет показателей портфеля СВЗ или МО

  -- A.S. Коллеги, максимально используйте конкатенацию при сборке res_html, это предотвращает распухание базы.
  -- Т.к. каждое присваивание res_html рождает новый временный блоб.

  db_id = f_get_db_id();
  res_html = (select res_style from fm_html_styles) || '<body>';

  if (rep_date is null) then
    rep_date = current_date;
  rep_date_begin = dateadd(month, -3, rep_date);
  rep_date_begin = rep_date_begin - extract(day from rep_date_begin) + 1;

  -- RFCRU-4360 Будет убрано после тестирования
  select un_name
  from username
  where un_id = :un_id
  into :un_name;

  if (un_name = 'fmtest') then
  begin
    select html_content
    from apple_test_data
    where report_type = 'MO_QUALITY'
    into :html_template;

    if (html_template is not null) then
    begin
      line_str = 'дата : ' || right('0' || extract(day from rep_date), 2) || '.' || right('0' || extract(month from rep_date), 2) || '.' || extract(year from rep_date);
      line_str = '<font color="black">' || :line_str || '</font>';
      res_html = blob_append(res_html, line_str || '<br>');
      res_html = blob_append(res_html, html_template);
      suspend;
      exit;
    end
    else
    begin
      res_html = blob_append(res_html, '<font color="red">Тестовый шаблон не найден в таблице apple_test_data</font><br>');
      suspend;
      exit;
    end
  end

  select un_from_c_id, c_name_fio
  from clients, username
  where un_id = :un_id and
        un_from_c_id = c_id
  into manager_id, lo_fio;

  if (lo_fio is null) then
    lo_fio = '';

  office_title = (select reg_code || ' ' || reg_name from region where reg_unicode = :reg_id);

  if (not(exists(select *
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :manager_id and
                       ps_date_begin <= :rep_date and
                       ps_date_end >= :rep_date and
                       ps_is_working = 1 and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code = 'JobSystem.Region_KM'))) then
    manager_id = 0;

  res_html = BLOB_APPEND(res_html,
    f_format('<h1 class="green_text">%1</h1>', iif(db_id <> 94401, 'Качество принимаемых решений', 'Qəbul olunan qərarların keyfiyyəti')) ||
    '<b>' || lo_fio || '</b><br>');

  line_str = iif(db_id <> 94401, 'Дата', 'Tarix') || ': ' || iif(rep_date is not null, f_datetostr(rep_date), iif(db_id <> 94401, 'Не указан', 'Qeyd olunmayıb'));
  res_html = BLOB_APPEND(res_html, line_str || '<br>');

  line_str = iif(db_id <> 94401, 'Офис', 'Ofis') || ': ' || coalesce(:office_title, iif(db_id <> 94401, 'Не указан', 'Qeyd olunmayıb'));
  res_html = BLOB_APPEND(res_html, line_str || '<br>');

  if (manager_id = 0) then
  begin
    res_html = BLOB_APPEND(res_html, f_format('<br>%1.<br>', iif(db_id <> 94401, 'Данный отчет доступен только менеджерам', 'Hesabat yalnız menecerler üçün nəzərdə tutulub')));
    suspend;
    exit;
  end

  res_html = BLOB_APPEND(res_html, f_format('<br>%1 (%2).<br>',
    iif(db_id <> 94401, 'Отображает долю первых трех погашений', 'İlk üş ödənişin hissəsini göstərir'),
    iif(db_id <> 94401, 'своевременно или с опозданием не более 7 дней', 'vaxtında və ya 7 gündən gec olmayan ödənişlər')));

  if (db_id = 81001) then
    res_html = BLOB_APPEND(res_html, 'Данный процентный показатель будет использован при расчете рейтинга МО.<br>');
  res_html = BLOB_APPEND(res_html, '<br>');

  res_html = BLOB_APPEND(res_html, '<table width="100%"><col width="25%"><col width="25%"><col width="25%"><col width="25%"><tr>' ||
  f_format('<th>%1</th><th>1-%2<wbr>%3</th><th>2-%2<wbr>%3</th><th>3-%2<wbr>%3</th></tr>',
    iif(db_id <> 94401, 'Месяц', 'Ay'), trim(iif(db_id <> 94401, 'пога', 'Ödə')), trim(iif(db_id <> 94401, 'шение', 'niş'))));

  res_delinq1_all = 0;
  res_delinq2_all = 0;
  res_delinq3_all = 0;

  res_total1_all = 0;
  res_total2_all = 0;
  res_total3_all = 0;

  for select res_period_date, res_qual_koef1, res_qual_koef2, res_qual_koef3, res_total1, res_total2, res_total3,
             res_delinq1, res_delinq2, res_delinq3
      from aarep_disb_manager_quality(:reg_id, :rep_date_begin, :rep_date, :manager_id)
      into res_period_date, res_qual_koef1, res_qual_koef2, res_qual_koef3, res_total1, res_total2, res_total3,
           res_delinq1, res_delinq2, res_delinq3
  do
  begin
    res_delinq1_all = res_delinq1_all + res_delinq1;
    res_delinq2_all = res_delinq2_all + res_delinq2;
    res_delinq3_all = res_delinq3_all + res_delinq3;

    res_total1_all = res_total1_all + res_total1;
    res_total2_all = res_total2_all + res_total2;
    res_total3_all = res_total3_all + res_total3;

    period_str = extract(year from res_period_date) || ' ';
    month_no = extract(month from res_period_date);
    period_str = period_str ||
      (case month_no
         when 1 then iif(db_id <> 94401, 'Янв.', 'Yanv')
         when 2 then iif(db_id <> 94401, 'Фев.', 'Fevr')
         when 3 then iif(db_id <> 94401, 'Март', 'Mart')
         when 4 then iif(db_id <> 94401, 'Апр.', 'Aprl')
         when 5 then iif(db_id <> 94401, 'Май', 'May')
         when 6 then iif(db_id <> 94401, 'Июнь', 'Iyun')
         when 7 then iif(db_id <> 94401, 'Июль', 'Iyul')
         when 8 then iif(db_id <> 94401, 'Авг.', 'Avqs')
         when 9 then iif(db_id <> 94401, 'Сен.', 'Sent')
         when 10 then iif(db_id <> 94401, 'Окт.', 'Oktb')
         when 11 then iif(db_id <> 94401, 'Ноя.', 'Noyb')
         when 12 then iif(db_id <> 94401, 'Дек.', 'Dekb')
       end);

    res_html = BLOB_APPEND(res_html,
    f_html_table_row('<b>' || period_str || '<b>', '<b>' || round(res_qual_koef1, 1) || '%</b>', '<b>' || round(res_qual_koef2, 1) || '%</b>', '<b>' || round(res_qual_koef3, 1) || '%</b>', -1, 1, 1, 1) ||
    f_html_table_row(iif(db_id <> 94401, 'Кол-во', 'Sayı'), res_total1, res_total2, res_total3, 1, 1, 1, 1));
  end
  res_qual_koef1 = 100.0;
  res_qual_koef2 = 100.0;
  res_qual_koef3 = 100.0;
  if (res_total1_all <> 0) then
    res_qual_koef1 = (1.0 - cast(res_delinq1_all as double precision) / cast(res_total1_all as double precision)) * 100.0;
  if (res_total2_all <> 0) then
    res_qual_koef2 = (1.0 - cast(res_delinq2_all as double precision) / cast(res_total1_all as double precision)) * 100.0;
  if (res_total3_all <> 0) then
    res_qual_koef3 = (1.0 - cast(res_delinq3_all as double precision) / cast(res_total3_all as double precision)) * 100.0;

  res_html = BLOB_APPEND(res_html,
    f_html_table_row(f_format('<b>%1<b>', iif(db_id <> 94401, 'Всего', 'Ümumi')), '<b>' || round(res_qual_koef1, 1) || '%</b>', '<b>' || round(res_qual_koef2, 1) || '%</b>', '<b>' || round(res_qual_koef3, 1) || '%</b>', -1, 1, 1, 1) ||
    f_html_table_row(iif(db_id <> 94401, 'Кол-во', 'Sayı'), res_total1_all, res_total2_all, res_total3_all, 1, 1, 1, 1) ||
    '</table></body>');

  suspend;
end
