CREATE PROCEDURE HP_EXTMINIAPP_WEB_SELL_TMS(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255,
    EMA_ID D_ID,
    EMA_UUID_CODE D_STR_CODE)
AS
declare variable UN_C_ID D_ID;
declare variable UN_REG_ID D_INTEGER;
declare variable EMA_UUID D_UUID;
declare variable CELL_NO D_STR_CODE;
declare variable ASP_STATUS D_SMALLINT;
declare variable LINK_TEXT D_STR_255;
declare variable C_ALTERNAT_FIO D_STR_100;
declare variable FNAME D_STR_100;
declare variable C_ID D_ID;
declare variable OBJ_KIND_CODE D_STR_CODE;
declare variable WEB_LINK D_STR_255;
declare variable RD_CODE D_STR_CODE;
declare variable PR_ID D_INTEGER;
declare variable ALL_BALANCE_CLI D_CURRENCY;
declare variable DELINQ_DAY_CLI D_CURRENCY;
declare variable POINT D_INTEGER;
declare variable MAX_AMOUNT D_CURRENCY;
begin
  res_is_error = 1;
  res_text = '';

  select un_from_c_id
  from username
  where un_id = :un_id
  into un_c_id;

  if (un_c_id is null) then
    res_text = 'Карточка пользователя не найдена';
  else
  begin
    select ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :un_c_id and
          ps_date_begin <= current_date and
          ps_date_end >= current_date and
          ps_is_working = 1
    into un_reg_id;

    if (un_c_id is null) then
      res_text = 'Не найден кадровый приказ по пользователю';
    else
    begin
      if (not exists(select * from krexpert
                     where ke_from_p_id = :un_c_id and
                           ke_status = 1 and
                           ke_date <= current_date and
                           ke_date_end >= current_date and
                           ke_is_krs <= 2 and
                           ke_from_reg_id = :un_reg_id)) then
        res_text = 'У сотрудника нет кода кредитного специалиста';
    end
  end

  if (res_text = '') then
  begin
    obj_kind_code = f_objkind(obj_kind, 'name');
    if (obj_kind_code = 'clients') then
      c_id = obj_id;
    else
    if (obj_kind_code = 'kontract') then
    begin
      select kc_from_c_id
      from knclient
      where kc_from_k_id = :obj_id and
            kc_amount_approved > 0
      order by kc_status desc
      rows 1
      into c_id;
    end
    else
    if (obj_kind_code = 'miniapp') then
    begin
      select mac_from_c_id
      from miniappclient
      where mac_from_ma_id = :obj_id and
            mac_amount > 0
      order by mac_is_cashier desc
      rows 1
      into c_id;
    end

    if (c_id is null) then
      res_text = 'Id клиента не найден';
  end

  if (res_text = '') then
  begin
    -- 2 (online)    действующий договора есть и он валидный.
    asp_status = f_asp_signed(c_id, current_date); -- f_asp_valid(c_id, current_date);

    if (asp_status <> 1) then
      res_text = 'У клиента нет действующего договора АСП для онлайн оформления';
  end

  cell_no = '';
  if (res_text = '') then
  begin
    cell_no = f_cont_info('clients', c_id, 'cell_main', 'ok');
    if (cell_no = '') then
      res_text = 'У клиента нет подтвержденного основного номера';
  end

  if (res_text = '') then
  begin
    if (exists(select * from extminiapp
               where ema_from_c_id = :c_id and
                     ema_status = 0)) then
      res_text = 'У клиента уже имеется предварительная заявка в процессе';
  end

  if (res_text = '') then
  begin
    all_balance_cli = null;
    delinq_day_cli = null;
    select sum(cast(kc_amount_approved as double precision) /  k_amount_approved * oks_balance) cli_balance,
           max(current_date - oks_delinq_date)
    from knclient, oksummary, kontract
    where kc_from_c_id = :c_id and
      kc_amount_approved > 0 and
      kc_from_k_id = oks_from_k_id and
      current_date between oks_date_begin and oks_date_end and
      oks_close_state = 0 and
      oks_from_k_id = k_id
    into all_balance_cli, delinq_day_cli;
    all_balance_cli = coalesce(all_balance_cli, 0);
    delinq_day_cli = coalesce(delinq_day_cli, 0);

    if (delinq_day_cli >= 1) then
      res_text = 'У клиента имеется активная просрочка';

    if (res_text = '') then
    begin
      point = f_point(c_id, current_date);
      max_amount = case 
                     when point <= 5 then 150000
                     when point <= 11 then 200000
                     when point <= 17 then 250000
                     when point <= 23 then 250000
                     when point <= 35 then 250000
                     else 300000
                   end;
      if (all_balance_cli >= max_amount) then
        res_text = 'У клиента совокупный баланс превышает разрешенного лимита';
    end
  end

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  pr_id =  546 ; -- Продукт для 5-дневных офисов
  select rd_code
  from region, refdata
  where reg_unicode = :un_reg_id and
        reg_office_hours_id = rd_id
  into rd_code;

  if (rd_code not in ('office_hours_5', 'office_hours_5mob')) then
    pr_id = 545; -- Продукт для 6 b 7-дневных офисов

  insert into extminiapp (ema_date, ema_from_c_id, ema_amount, ema_period, ema_office_id, ema_lo_id, ema_from_un_id, ema_from_pr_id)
  values (current_date, :c_id, 0, 0, :un_reg_id, :un_c_id, :un_id, :pr_id)
  returning ema_id
  into :ema_id;

  insert into uuidlist (udl_object_kind, udl_object_id)
  values (f_objkind('extminiapp', 'id'), :ema_id)
  returning udl_uuid
  into :ema_uuid;

  select c_alternat_fio
  from clients
  where c_id = :c_id
  into c_alternat_fio;

  select fname
  from fio_separation(:c_alternat_fio)
  into fname;

  web_link = 'https://tms.club/mbulak/energylife';
  link_text = '%1, viberite tarif dlya pokupki v rassrochku: %2?uid=%3';
  link_text = f_format(link_text, fname, web_link, lower(uuid_to_char(ema_uuid)));

  execute procedure hp_send_quick_message('sms', :cell_no, '', :link_text, '', 365, un_id);

  ema_uuid_code = uuid_to_char(ema_uuid);

  res_is_error = 0;
  res_text = 'Клиенту отправлен СМС на номер ' || cell_no || ' для покупки "Энергия жизни" в рассрочку';
  suspend;
end
