CREATE PROCEDURE AAREP_CASH_OPERDAY_CLOSING(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    RES_IS_OVERLIMIT INTEGER,
    RES_SALDO_CASH DECIMAL(18,4),
    REG_CASH_LIMIT DECIMAL(18,4),
    REG_MEMO VARCHAR(255),
    ERR_LIST D_STR_100)
AS
declare variable s1 decimal(18,4);
declare variable al_id varchar(20);
declare variable res_acc_kind integer;
declare variable res_saldo decimal(18,4);
declare variable atl_acc_mask varchar(20);
declare variable db_id d_id;
begin
  res_saldo_cash = 0;
  res_is_overlimit = 0;
  reg_cash_limit = 0;
  reg_memo = '';
  db_id = f_get_db_id();

  if (exists(select *
             from region
             where reg_unicode = :report_region_id and
                   reg_status = 3)) then
  begin
  res_saldo_cash = 0;
  res_is_overlimit = 0;
    if (db_id in (39801, 41701)) then
    begin
      if (exists(select ks_from_k_id
                 from kontstate, kontract
                 where ks_ol_is_delete = 0 and
                       :report_date_end between ks_date_begin and ks_date_end and
                       ks_state = 0 and
                       ks_from_k_id = k_id and
                       ks_region_id = :report_region_id and
                       k_status = 1)) then


      begin
        res_is_overlimit = 2;
        reg_memo = 'Имеются утвержденные неактивные договора, необходимо выдать или перенести дату выдачи' ;
        err_list = 'block';
        suspend;
      end

      if (exists(select acl_id, k_id, c_id, project, reg_name, kf_name, k_kontract_no, seller_fio, kml_total,
                        error_text, 0 as for_pay, acl_date, is_expired
                 from epay_actions_list(:report_region_id)
                 where --acl_date = :REPORT_DATE_END and
                       error_text = '')) then
      begin
        res_is_overlimit = 3;
        reg_memo = 'Имеются неоплаченные платежи';
        err_list = 'ElSomCloseDay';
        suspend;
      end

      if (exists(select *
                 from cashcheckmain
                 where ccm_is_delete = 0 and
                       ccm_status <> 2 and -- статус не 'Подписан'
                       ccm_date = current_date and
                       ccm_from_reg_id = :report_region_id)) then
      begin
        reg_memo = 'Акт инвентаризации не подписан'; -- RFCKG-4261
        err_list = 'UnsignedInventoryAct';
        suspend;  
      end
    end
    res_is_overlimit = 0;
    reg_memo ='';
    err_list = '';

    select res_saldo_cash
    from aarep_money_cash_saldo(:report_region_id, :report_date_end)
    into :res_saldo_cash;

    select rlm_limit_cash_max
    from reglimits
    where rlm_from_reg_id = :report_region_id and
          rlm_date <= :report_date_end
    order by rlm_date desc, rlm_id desc
    rows 1
    into :reg_cash_limit;

    if (reg_cash_limit is null) then
      reg_cash_limit = 0;
    if (res_saldo_cash > reg_cash_limit) then
    begin
      res_is_overlimit = 1;
      reg_memo = 'Превышен допустимый остаток кассы';
      err_list = 'CashCloseDay';
      suspend;
    end

  end
end
