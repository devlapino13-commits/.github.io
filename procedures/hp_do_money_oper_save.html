CREATE PROCEDURE HP_DO_MONEY_OPER_SAVE(
    OPER_DATE DATE,
    REG_ID INTEGER,
    DOC_KIND INTEGER,
    OPER_SUBTYPE INTEGER,
    IS_ZALOG INTEGER,
    DOC_NO INTEGER,
    DOC_NO_STR D_STR_CODE,
    MEMO D_STR_2048,
    KONTRAGENT_TYPE INTEGER,
    CLIENT_ID ID_TYPE,
    BANK_ACC_NO_CLIENT D_STR_CODE,
    OFFICE_ID ID_TYPE,
    BANK_ACC_NO D_STR_CODE,
    DT_ACC D_STR_CODE,
    KT_ACC D_STR_CODE,
    CURR_ID D_STR_10,
    CURR_RATE CURR_TYPE,
    AMOUNT_VAL CURR_TYPE,
    PC_CODE D_STR_CODE,
    KP_CODE D_STR_CODE,
    OB_DETAIL D_STR_LONG,
    SAC_ID INTEGER,
    KONT_ID ID_TYPE,
    KONTRACT_NO D_STR_CODE,
    OPERATION_TYPE INTEGER,
    SUBKONTO_DT D_INTEGER,
    SUBKONTO_KT D_INTEGER,
    UN_ID ID_TYPE)
RETURNS (
    OL_ID ID_TYPE,
    OL_ID_STR D_STR_255,
    RES_TEXT D_STR_LONG)
AS
declare variable sac_month_minus curr_type;
declare variable sac_month_saldo curr_type;
declare variable amount curr_type;
declare variable ob_client_dt id_type;
declare variable ob_client_kt id_type;
declare variable sac_sign integer;
declare variable reg_salary integer;
declare variable ok_id id_type;
declare variable kf_id d_id;
declare variable k_reg_id d_integer;
declare variable db_id integer;
declare variable acc_from_1 d_str_code;
declare variable acc_from_2 d_str_code;
declare variable acc_from_3 d_str_code;
declare variable acc_from_4 d_str_code;
begin
-- Процедура сохранения денежной операции
  db_id = f_get_db_id();
  if (:kontract_no <> '' and :kontract_no is not null) then
    begin
      select k.k_from_kf_id
      from kontract k
      where k.k_kontract_no = :kontract_no and
        k.k_kontract_date = :oper_date
      into :kf_id;

      select kal.kal_new_reg_id
      from knaktlist kal
      where :oper_date between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0 and
          kal.kal_from_kf_id = :kf_id
      into :k_reg_id;

    end


  ol_id = 0;
  res_text = '';
  ol_id_str = '';

  if (is_zalog is null) then
    is_zalog = 0;

  if (subkonto_dt is null) then subkonto_dt = 0;
  if (subkonto_kt is null) then subkonto_kt = 0;

  if (coalesce(curr_rate, 0) = 0) then
    curr_rate = 1;
  amount = round(amount_val * curr_rate, 2);

  if (client_id = 0) then
    client_id = null;
  if (office_id = 0) then
    office_id = null;

  if (res_text = '' and
      oper_date is null) then
    res_text = 'Дата не указана';
  if (res_text = '' and
      coalesce(reg_id, 0) = 0) then
    res_text = 'Офис не указан';
  if (res_text = '' and
      coalesce(dt_acc, '') = '') then
    res_text = 'Дебет счет не указан';
  if (res_text = '' and
      coalesce(kt_acc, '') = '') then
    res_text = 'Кредит счет не указан';
  if (res_text = '' and
      coalesce(curr_id, '') = '') then
    res_text = 'Валюта не указана';
  if (res_text = '' and
      coalesce(doc_kind, -1) = -1) then
    res_text = 'Тип документа не указан';
  if (res_text = '' and
      coalesce(amount_val, 0) = 0) then
    res_text = 'Сумма не указана';
  if (res_text = '' and
      coalesce(memo, '') = '') then
    res_text = 'Назначение платежа не указано';
  if (res_text = '' and
      coalesce(memo, '') = '') then
    res_text = 'Назначение платежа не указано';
  if (res_text = '' and
      is_zalog = 1 and
      coalesce(kont_id, 0) = 0) then
    res_text = 'Договор займа не указан';
  if (res_text = '' and
      coalesce(client_id, 0) = 0 and
      :db_id in (39801, 41701)) then
    res_text = 'Cотрудник/контрагент не указан';
  if (res_text = '' and
      oper_subtype in (1, 2) and
      coalesce(kontract_no, '') <> '') then
  begin
    select error_text, ok_id
    from hp_operkontract_ins(-1, -1, :kontract_no, :operation_type, :amount)
    into :res_text, ok_id;
  end

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  sac_month_minus = -1;
  sac_month_saldo = -1;

  if (coalesce(sac_id, 0) > 0) then
  begin
    select sac_id
    from sa_config
    where sac_is_credit = 9
    into sac_month_minus;

    select sac_id
    from sa_config
    where sac_is_credit = 10
    into sac_month_saldo;
  end

-- Создаем запись для денежной операции

  ob_client_dt = null;
  ob_client_kt = null;

  sac_sign = 0;
  if (client_id is not null) then
  begin
    if (oper_subtype in (2, 7, 8)) then
    begin
      sac_sign = 1;
      ob_client_dt = client_id;
    end
    if (oper_subtype in (1, 6)) then
    begin
      sac_sign = -1;
      ob_client_kt = client_id;
    end

    if (oper_subtype in (3, 4, 5, 9)) then
    begin
      sac_sign = 0;
      ob_client_dt = client_id;
      ob_client_kt = client_id;
    end
  end

  ol_id = (select max(ob.ob_from_ol_id)
                 from operbook ob) + 1;
  if (:k_reg_id = :reg_id or :k_reg_id is null or db_id not in (39801, 41701)) then
      execute procedure hp_do_money_oper_save_tmp(:ol_id, :reg_id, :doc_no, :oper_date, :doc_no_str, :doc_kind, :memo, :un_id, :client_id, :bank_acc_no_client, :office_id, :bank_acc_no, :oper_subtype, :kontragent_type, :curr_id, :curr_rate, :ob_client_dt, :dt_acc, :amount, :amount_val, :pc_code, :kp_code, :kt_acc, :ob_client_kt, :ob_detail, :sac_id);
  else
    begin
      select lr.ln_account
      from loronostro lr
      where lr.ln_region_loro = :k_reg_id and
            lr.ln_region_nostro = 1060100
      into :acc_from_1;

      select lr.ln_account
      from loronostro lr
      where lr.ln_region_loro = 1060100 and
            lr.ln_region_nostro = :k_reg_id
      into :acc_from_2;

      select lr.ln_account
      from loronostro lr
      where lr.ln_region_loro = 1060100 and
            lr.ln_region_nostro = :reg_id
      into :acc_from_3;

      select lr.ln_account
      from loronostro lr
      where lr.ln_region_loro = :reg_id and
            lr.ln_region_nostro = 1060100
      into :acc_from_4;

      :memo = :memo || ' через другой офис';

      execute procedure hp_do_money_oper_save_tmp(:ol_id, :k_reg_id, :doc_no, :oper_date, :doc_no_str, :doc_kind, :memo, :un_id, :client_id, :bank_acc_no_client, :office_id, :bank_acc_no, :oper_subtype, :kontragent_type, :curr_id, :curr_rate, :ob_client_dt, :acc_from_4, :amount, :amount_val, :pc_code, :kp_code, :kt_acc, :ob_client_kt, :ob_detail, :sac_id);
      ol_id = ol_id + 1;
      execute procedure hp_do_money_oper_save_tmp(:ol_id, 1060100, :doc_no, :oper_date, :doc_no_str, :doc_kind, :memo, :un_id, :client_id, :bank_acc_no_client, :office_id, :bank_acc_no, :oper_subtype, :kontragent_type, :curr_id, :curr_rate, :ob_client_dt, :acc_from_2, :amount, :amount_val, :pc_code, :kp_code, :acc_from_3, :ob_client_kt, :ob_detail, :sac_id);
      ol_id = ol_id + 1;
      execute procedure hp_do_money_oper_save_tmp(:ol_id, :reg_id, :doc_no, :oper_date, :doc_no_str, :doc_kind, :memo, :un_id, :client_id, :bank_acc_no_client, :office_id, :bank_acc_no, :oper_subtype, :kontragent_type, :curr_id, :curr_rate, :ob_client_dt, :dt_acc, :amount, :amount_val, :pc_code, :kp_code, :acc_from_1, :ob_client_kt, :ob_detail, :sac_id);
    end
    
  ol_id_str = ol_id || '-' || doc_kind;

  if (ok_id is not null) then
  begin
    select error_text
    from hp_operkontract_ins(:ok_id, :ol_id, '', :operation_type, 0)
    into :res_text;

    res_text = '';
  end

  if (client_id is not null and
      sac_sign <> 0 and
      sac_id > 0) then
  begin
    reg_salary = null;

    select ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :client_id and
          ps_date_begin <= :oper_date and
          ps_date_end >= :oper_date and
          ps_status = 1
    order by ps_date_begin desc, ps_date_end_in_order desc, ps_is_working desc
    rows 1
    into reg_salary;

    if (reg_salary is null) then
    begin
      reg_salary = f_hq_id(db_id);
    end

    insert into opersalary (os_from_ol_id, os_from_reg_id, os_from_c_id, os_from_sac_id, os_value)
    values (:ol_id, :reg_salary, :client_id, :sac_id, :amount);

    if (coalesce(sac_month_minus, -1) <> -1) then
    begin
      insert into opersalary (os_from_ol_id, os_from_reg_id, os_from_c_id, os_from_sac_id, os_value)
      values (:ol_id, :reg_salary, :client_id, :sac_month_minus, :amount);
    end

    if (coalesce(sac_month_saldo, -1) <> -1) then
    begin
      insert into opersalary (os_from_ol_id, os_from_reg_id, os_from_c_id, os_from_sac_id, os_value)
      values (:ol_id, :reg_salary, :client_id, :sac_month_saldo, :amount);
    end
  end

  if (client_id is not null and
      sac_sign <> 0 and
      sac_id < 0) then
  begin
    insert into servicecalc (ssc_from_ol_id, ssc_id, ssc_from_sd_id, ssc_date, ssc_amount_dogovor, ssc_other_plus,
                             ssc_tax_client_sfond, ssc_tax_client_pdnal, ssc_tax_client_other, ssc_tax_company_sfond,
                             ssc_client_saldo, ssc_total_days, ssc_patent_days, ssc_spolis_days, ssc_kind,
                             ssc_disb_amount, ssc_zp_koef)
    values (:ol_id, 1, -:sac_id, :oper_date, 0, 0, 0, 0, 0, 0, :amount * cast(:sac_sign as curr_type) * -1.0000, -- ssc_client_saldo
            0, 0, 0, 1, :amount * cast(:sac_sign as curr_type), -- ssc_disb_amount
            1);
  end

  if (coalesce(is_zalog, 0) = 1 and
      coalesce(kont_id, 0) > 0) then
  begin
    insert into operkredzalog (okz_from_ol_id, okz_number, okz_from_k_id, okz_summa_plus, okz_summa_minus, okz_is_money,
                               okz_account)
    values (:ol_id, null, :kont_id, iif(:oper_subtype in (1, 6), :amount, 0.0), -- okz_summa_plus
            iif(:oper_subtype not in (1, 6), :amount, 0.0), -- okz_summa_minus
            1, -- okz_is_money
            iif(:oper_subtype in (1, 6), :kt_acc, :dt_acc)); -- okz_account
  end

  suspend;
end
