CREATE PROCEDURE AAREP_WRITEOFF_PLAN_RU(
    REP_REGION_ID INTEGER,
    REP_DATE DATE,
    BALANCE_DATE DATE,
    IS_ADVANCED SMALLINT)
RETURNS (
    REG_ID INTEGER,
    K_ID DECIMAL(18,0),
    REG_NAME STR_100,
    KF_NAME STR_100,
    KONTRACT_NO VARCHAR(20),
    ZAD_OS DECIMAL(18,4),
    ZAD_PERC DECIMAL(18,4),
    SH_CALC_PERC DECIMAL(18,4),
    SH_CALC_FINE DECIMAL(18,4),
    SH_DATE DATE,
    OK_OL_DATE DATE,
    WRITEOFF_PLAN_DATE DATE,
    MESS_STR STR_100,
    SH_ID INTEGER,
    PR_ID INTEGER,
    PR_NAME VARCHAR(50),
    K_DISB_DATE DATE,
    REG_CODE VARCHAR(10),
    K_ID_STR STR_100)
AS
declare variable reg_id_temp integer;
declare variable null_date date;
declare variable has_orig smallint;
declare variable has_new smallint;
declare variable k_orig decimal(18,0);
declare variable rep_date_temp date;
declare variable k_id_old decimal(18,0);
declare variable period integer;
declare variable wfi_id integer;
begin
  /*
    igusev

    IS_ADVANCED  - 1 - расширенный набор данных. (обрабатывает все просроченные платежи)
                 - 0 - обрабатывает платежи, по которым просрочка больше или равна 3 года на дату отчета  REP_DATE
    
    REP_DATE  - дата, по отношению к которой определяю, просрочен платеж или нет, подлежит списанию или нет.
    BALANCE_DATE - дата среза баланса.

---- Внимание!
         Если требуется произвести списание датой 31.12.2014, но фактически списание осуществляется 20.02.2015, то:

    REP_DATE = 31.12.2014
    BALANCE_DATE = 20.02.2015
    IS_ADVANCED = 0

    Алгоритм процедуры:

    Смежные контракты - это контракты, образовавшиеся в ходе амнистии + основной контракт, от которого они отделились.

    На примере группы 3-2-42-152.ВС. ВСК-3., по которой были амнистии.
    Для контракта 0301-002190-01, смежными контрактами будут являться все контракты, которые были амнистированы от него,
    т.е.: 0301-002190-01/2, 0301-002190-01/3, 0301-002190-01/4.
    Для контракта 0301-002190-01/3, смежными контрактами будет являться контакт, от которого он был амнистирован
    0301-002190-01 и контракты,  амнистированные от  0301-002190-01, т.е. 0301-002190-01/2, 0301-002190-01/4.

    На дату (BALANCE_DATE) берутся все просроченные кредиты.
    На дату (REP_DATE) нет смысла брать, т.к. даже если кредит был просрочен на 31.12.2014, на дату проверки баланса картина по нему может быть другой.
    Для каждого из этих кредитов, получается баланс на дату (BALANCE_DATE).

    Каждый активный платеж объединяется с аналогичными платежами по графику, в смежных контрактах,
    и рассматривается в дальнейшем как один общий платеж.

    Выявляется наиближайшая к (BALANCE_DATE) дата фактического платежа клиента по этому общему платежу,
    (т.е. в контрактах №...1,  №....2,  №...3 есть платеж по графику с датой 12.02.14, по этому платежу проверяется
    наличие оплаты клиентами по всем смежным контрактам, и выявляется самый последний (свежий) платеж)
    Это платеж будет считаться фактической датой оплаты для вех смежных контрактов.

    Если фактическая дата оплаты по платежу больше даты по графику, тогда
    Ориентировочная дата списания (для платежа) = фактическая дата платежа + 3 года 
    иначе ориентировочная дата списания (для платежа) = дата платежа по графику + 3 года
    
    Если Ориентировочная дата списания меньше даты отчета, тогда поле «Статус» = «Подлежит списанию»

    Таким образом, отсрочка по платежам связана в рамках всех этих контрактов.
    Например:  если по платежу, который должен быть уплачен 29.11.2011, заплатили 1 рубль 01.02.2015 в контракте 0301-002190-01/4,
    тогда для контрактов (0301-002190-01, 0301-002190-01/2, 0301-002190-01/3) у которых этот платеж в полной мере не закрыт
    (и последние оплаты была до 01.02.2015), дата списания будет передвинута на 01.02.2015 + 3 года.

----------  Важно!  --------------------------------------------------------

    1) Если разные номера погашений в смежных контрактах, но их даты одинаковые это не страшно, они будут объедены в один платеж , т.к.
    алгоритм отталкивается от даты платежа, а не от номера.
    Для контракта 1701-004771-01, как раз такой случай.

    2) Алгоритм не позволяет объединять погашения в смежных контрактах, где была реструктуризация и даты погашения в смежных контрактах отличаются.

    3)  Дата баланса важна потому, что когда происходит списание, программа должна отталкиваться от
    текущего баланса контрактов, а не от того который был когда-то.

    Пример контракты:
    0113-016085-03
    0302-046584-01
    0302-046584-01/2

    В этих контрактах на 31.12.2014 есть проблема с задвоением текущего платежа, но  после 31.12.2014 проблема устранена.
    Поэтому хоть списание и осуществляется на 31.12.2014, баланс должен браться на дату фактического списания (BALANCE_DATE)
  */

--if (REP_K_ID is null) then REP_K_ID = 0;
  k_id_old = 0;

  period = extract(year from :rep_date) * 100 + extract(month from :rep_date);

  if (balance_date is null) then
    balance_date = current_date;

  null_date = cast('01.01.1900' as date);
  reg_id_temp = 0;
  if (is_advanced is null) then
    is_advanced = 0;

-- беру только кредиты выданные до '28.02.2018'
  for select oks.oks_region_id, oks.oks_from_k_id
      from oksummary oks
      inner join region_list(:rep_region_id, 1, 3, 1) on oks.oks_region_id = id_list
      inner join kontract kont on kont.k_id = oks.oks_from_k_id
      where oks_close_state = 0 and
            :rep_date between oks_date_begin and oks_date_end and
            (((cast(:balance_date as date) - oks_delinq_date >= 0) /* and :IS_ADVANCED = 1 */ ) /*or  -- здесь я беру именно BALANCE_DATE, т.к. дата проыерки баланс будет отталкиваться от нее
      ((cast(:rep_date as date) - oks_delinq_date >= 1092) and :IS_ADVANCED = 0)*/ ) and
            oks.oks_close_state = 0 and
            kont.k_disb_date < '28.02.2012'
      order by oks_region_id, oks_from_k_id
      into reg_id, k_id
  do
  begin
--if ((K_ID = REP_K_ID) and (REP_K_ID <> 0)) then
--begin
    if (reg_id <> reg_id_temp) then
    begin
      select reg.reg_code, reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :reg_code, :reg_name;

      reg_id_temp = reg_id;
    end

    has_orig = 0;
    has_new = 0;
    k_orig = null;

    select kdev.kd_k_orig
    from kontdevides kdev
    where kdev.kd_k_new = :k_id
    into :k_orig;

    if (k_orig is not null) then
      has_orig = 1;
    if (exists(select *
               from kontdevides kdev
               where kdev.kd_k_orig = :k_id)) then
      has_new = 1;

    if (has_new = 1) then
      k_orig = k_id;

    rep_date_temp = dateadd(-3 year to rep_date);

    for select zad_os, zad_perc, sh_calc_perc, sh_calc_fine, sh_date, sh_id
        from aarep_make_zad_list(:k_id, :balance_date, 1)
        where ((sh_date <= :rep_date and
              :is_advanced = 1) or
              (sh_date <= :rep_date_temp and
              :is_advanced = 0))
        order by sh_date
        into :zad_os, :zad_perc, :sh_calc_perc, :sh_calc_fine, :sh_date, :sh_id
    do
    begin
      wfi_id = null;

-- проверка списка с платежами которые не следует списывать -------
      select wfi.wfi_id
      from writeoff_ignore wfi
      where wfi.wfi_period = :period and
            wfi.wfi_from_k_id = :k_id and
            wfi.wfi_sh_date = :sh_date rows 1
      into wfi_id;

      ok_ol_date = null;
      writeoff_plan_date = sh_date;

      if ((has_orig = 0) and
          (has_new = 0)) then
      begin
        select max(ok.ok_ol_date)
        from operkred ok
        where ok.ok_from_k_id = :k_id and
              ok.ok_from_c_id = 0 and
              ok.ok_type in (4, 9) and
              ok.ok_ol_is_active = 1 and
              ok.ok_sh_date = :sh_date and
              ok.ok_sh_date <= :balance_date
        into :ok_ol_date;
      end
      else
      begin
        /*
                select max(ok.ok_ol_date)
                from operkred ok
                where (ok.ok_from_k_id  = :k_id
                       or ok.ok_from_k_id in (select kdev.kd_k_new from kontdevides kdev where kdev.kd_k_orig = :k_orig)
                       or ok.ok_from_k_id in (select kdev.kd_k_orig from kontdevides kdev where kdev.kd_k_new = :k_id))
                and  ok.ok_from_c_id = 0
                and ok.ok_type in (4,9)
                and ok.ok_ol_is_active = 1
                and ok.ok_sh_date =  :SH_DATE
                and ok.ok_sh_date <= :balance_date
                into :ok_ol_date; */

        select max(ok.ok_ol_date)
        from (select :k_id as k_id
              from rdb$database
              union
              select kdev.kd_k_new
              from kontdevides kdev
              where kdev.kd_k_orig = :k_id
              union
              select kdev.kd_k_orig
              from kontdevides kdev
              where kdev.kd_k_new = :k_id) as a
        inner join operkred ok on ok.ok_from_k_id = a.k_id and
              ok.ok_from_c_id = 0 and
              ok.ok_type in (4, 9) and
              ok.ok_ol_is_active = 1 and
              ok.ok_sh_date <= current_date and
              ok.ok_sh_date = :sh_date
        into :ok_ol_date;
      end
      if (coalesce(ok_ol_date, null_date) > sh_date) then
        writeoff_plan_date = ok_ol_date;
      writeoff_plan_date = dateadd(3 year to writeoff_plan_date);
      mess_str = '';

      if (wfi_id is null) then
      begin
        if (writeoff_plan_date < rep_date) then
          mess_str = 'Подлежит списанию';
      end
      else
        mess_str = 'Списание отменено';

      if (k_id <> k_id_old) then
      begin
        select kal.kal_new_kf_name, kont.k_kontract_no, pr.pr_id, pr.pr_name
        from knaktlist kal
        inner join kontract kont on kont.k_from_kf_id = kal.kal_from_kf_id
        inner join product pr on pr.pr_id = kont.k_from_pr_id
        where kal.is_delete = 0 and
              :balance_date between kal_date and kal_date_end and
              kont.k_id = :k_id
        into kf_name, kontract_no, pr_id, pr_name;

        select ks.ks_date_begin
        from kontstate ks
        where ks.ks_from_k_id = :k_id and
              ks.ks_state = 1 and
              ks.ks_ol_is_delete = 0
        into :k_disb_date;
      end

      k_id_str = cast(k_id as str_100);
      suspend;

      k_id_old = k_id;
    end
--if (REP_K_ID<>0) then exit;
--end
  end
end
