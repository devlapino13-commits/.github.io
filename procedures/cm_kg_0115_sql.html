CREATE PROCEDURE CM_KG_0115_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare variable mac_from_c_id id_type;
declare variable disb_date d_date;
declare variable mac_amount d_currency;
declare variable k_id d_id;
declare variable oks_balance d_currency;
begin
  -- если это заявка по продукту фермер
  for select mac.mac_from_c_id, maa.ma_date_oper, mac.mac_amount
      from miniappclient mac
      inner join miniapp maa on maa.ma_id = mac.mac_from_ma_id
      inner join product pr on pr.pr_id = maa.ma_from_pr_id
      where mac.mac_from_ma_id = :ma_id and
            mac.mac_amount > 0 and
            maa.ma_is_active = 1 and
            maa.ma_date_end = '31.12.9999' and
            pr.pr_code like '%.farmer.%'
      into mac_from_c_id, disb_date, :mac_amount
  do
  begin
    -- и нет активного кредита по Фермеру, тогда блок
    k_id = null;
    select ks.ks_from_k_id
    from knclient
    inner join kontstate ks on kc_from_k_id = ks_from_k_id
    inner join kontract kont on kont.k_id = ks.ks_from_k_id
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    where kc_from_c_id = :mac_from_c_id and
          kc_amount_approved > 0 and
          pr.pr_code like '%.farmer.%' and
          ks_state = 1 and
          ks_ol_is_delete = 0 and
          :disb_date between ks.ks_date_begin and ks.ks_date_end
    order by ks_date_begin
    rows 1
    into :k_id;
    if (k_id is null) then
    begin
      select cl.c_name_fio --клиент
      from clients cl
      where cl.c_id = :mac_from_c_id
      into :res_info_text;
      res_error_is_critical = 1;
      res_error_text = 'Продукт Фермер доступен только для рефинансирования';
      suspend;
    end
    else
    begin
      oks_balance = 0;
      select oks.oks_balance
      from oksummary oks
      where oks.oks_from_k_id = :k_id and
            current_date between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_close_state = 0
      into :oks_balance;
      oks_balance = coalesce(oks_balance, 0);
      if (oks_balance + 25000 < mac_amount) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Сумма в заявке ' || round(mac_amount, 0) || ' больше чем долг по ОС ' || round(oks_balance, 0) || ' + 25000 сом, действующего займа';
        suspend;
      end
    end
  end
end
