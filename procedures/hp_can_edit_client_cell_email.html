CREATE PROCEDURE HP_CAN_EDIT_CLIENT_CELL_EMAIL(
    C_ID DECIMAL(18,0))
RETURNS (
    NEED_APPLICATION_BLOB INTEGER)
AS
declare variable ma_id integer;
declare variable ma_is_active integer;
declare variable ma_disb_date date;
declare variable k_status integer;
declare variable cm_database_id integer;
begin
  -- проверка возможности редаткирования телефона и/или email
  -- Если есть активный или утвержденный кредит, то требуем прикрепления скана заявления.

  -- Цель процедуры:
  --    Если у клиента есть активный или утвержденный кредит,
  --      то запрещаем (Can_Edit=0),
  --      в других случаях разрезаем (Can_Edit=1) менять номер телефона без прикрепления скана

  need_application_blob = -1;
  cm_database_id = f_get_db_id();

  if (not((cm_database_id = 81001) and
      (current_date >= '01.07.2014'))) then
  begin
    need_application_blob = 0;
    suspend;
    exit;
  end

  for select mac_from_ma_id
      from miniappclient
      where mac_from_c_id = :c_id and
            mac_amount > 0
      into :ma_id
  do
  begin
    ma_is_active = 0;
    -- проверяем, есть ли активные заявки клиента за последний месяц
    select ma_is_active, ma_date_oper
    from miniapp
    where ma_id = :ma_id and
          ma_date_end = '31.12.9999'
    into :ma_is_active, :ma_disb_date;
    if (ma_is_active is null) then
      ma_is_active = 0;
    if ((ma_is_active = 1) and
        (ma_disb_date < (current_date - 30))) then
      ma_is_active = 0;

    if (ma_is_active = 1) then
    begin
      k_status = 0;
      select k_status
      from kontract
      where k_from_ma_app_id = :ma_id rows 1
      into :k_status;
      if (k_status is null) then
        k_status = 0;

      if (k_status = 0) then
        need_application_blob = 0;
      if (k_status = 1) then
        need_application_blob = 1;
    end
  end

  if (need_application_blob = -1) then
  begin
    need_application_blob = 0;

    if (exists(select *
               from knclient, oksummary
               where kc_from_c_id = :c_id and
                     kc_amount_approved > 0 and
                     kc_from_k_id = oks_from_k_id and
                     oks_date_begin <= current_date and
                     oks_date_end >= current_date and
                     oks_balance > 0 and
                     oks_close_state = 0)) then
      need_application_blob = 1;
  end
  suspend;
end
