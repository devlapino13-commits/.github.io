CREATE PROCEDURE CB_KG_LO_2014_08(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_REGION_ID INTEGER,
    AA_CURR_KC_ID ID_TYPE,
    DELINQ_BALANCE CURR_TYPE,
    BEGIN_BALANCE CURR_TYPE,
    AA_JUST_CL_BEGIN INTEGER,
    AA_JUST_CL_END INTEGER,
    PAR7 DECIMAL(15,5),
    KOL_CLI_FIELD INTEGER,
    FACTOR_KOL_CLI CURR_TYPE,
    BONUS_3_1 CURR_TYPE,
    REPAY_DELINQ CURR_TYPE,
    BONUS_4_1 CURR_TYPE,
    REPAY_DELINQ_CLOSED CURR_TYPE,
    BONUS_4_2 CURR_TYPE,
    POGASH_VNESYSTEM CURR_TYPE,
    BONUS_5_1 CURR_TYPE,
    BONUS_TOTAL CURR_TYPE,
    JL_NAME VARCHAR(100),
    PS_IS_WORKING SMALLINT,
    BONUS_KOEF CURR_TYPE,
    BONUS_SUM CURR_TYPE,
    BONUS_MEMO VARCHAR(255),
    BONUS_END CURR_TYPE,
    AA_CURR_KC_FIO VARCHAR(100),
    SKS_MEMO VARCHAR(255),
    BONUS_SKS CURR_TYPE,
    REG_NAME_LAST VARCHAR(100),
    REG_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    OKRUG_NAME VARCHAR(100),
    REG_PAR8_180 DECIMAL(15,5))
AS
declare variable rep_date_begin date;
declare variable ps_is_on_staff smallint;
declare variable jk_system_code varchar(100);
declare variable sks_date_begin date;
declare variable date_half_month date;
declare variable sks_reg_id_last integer;
declare variable curr_reg_id_prior integer;
declare variable reg_kol_cli integer;
declare variable repay_delinq_temp curr_type;
declare variable ps_date_begin date;
declare variable ps_date_end date;
declare variable ps_date_begin_last date;
declare variable stage_date date;
begin
  date_half_month = :rep_date_end - extract(day from :rep_date_end) + 15; -- 15е число
  rep_date_begin = :rep_date_end - extract(day from :rep_date_end) + 1; -- Начало периода

  insert into temp_kc_reg_info (kc_region_id, kc_officer_id, kc_delinq7_01012014, kc_balance_begin, kc_kol_cli_begin,
                                kc_kol_cli, kc_repay_delinq, --KC_REPAY_DELINQ_CLOSED,
                                kc_balance, kc_delinq_8_180)
  select aa_curr_region_id, aa_curr_kc_id, sum(iif(aa_delinq_begin_date >= '01.01.2014', aa_deliq_balance7, 0)),
         sum(aa_amount_active_begin), sum(aa_just_cl_begin), sum(aa_just_cl_end),
         sum(iif(aa_delinq_begin_date < '01.01.2014', aa_repay_all_os + aa_repay_all_perc + aa_repay_all_fine, 0)),

         /*   sum(iif(AA_DELINQ_BEGIN_DATE < '01.01.2014'
                 and
                 exists(select * from kontstate
                        where ks_ol_is_delete = 0
                        and ks_state = 2
                        and cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end
                        and ks_from_k_id = AA_K_ID)
                 , Aa_Repay_All_Os + Aa_Repay_All_Perc + Aa_Repay_All_Fine, 0)),  */

         sum(aa_amount_active_end), sum(aa_deliq_balance7 - aa_deliq_balance180)

  from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 1, 1, 1, 0, 0, null)
  where not exists(select *
                   from sb_akt, sb_list -- Пункт 2.3
                   where sba_is_delete = 0 and
                         sba_status = 2 and
                         sba_from_sbl_id <> 0 and
                         :rep_date_end between sba_date_begin and sba_date_end and
                         sba_from_k_id = aa_k_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1)
  group by aa_curr_region_id, aa_curr_kc_id;

  insert into temp_sks_reg_info (reg_id, reg_par8_180, reg_kol_cli)
  select kc_region_id, sum(kc_delinq_8_180) / sum(kc_balance) * 100, sum(kc_kol_cli)
  from temp_kc_reg_info
  where kc_balance > 0
  group by kc_region_id;

  curr_reg_id_prior = 0;

  for select kc_region_id, kc_officer_id, kc_delinq7_01012014, kc_balance_begin, kc_kol_cli_begin, kc_kol_cli,
             kc_repay_delinq
      from temp_kc_reg_info
      into aa_curr_region_id, aa_curr_kc_id, delinq_balance, begin_balance, aa_just_cl_begin, aa_just_cl_end,
           repay_delinq
  do
  begin
    repay_delinq_closed = null;
    repay_delinq_temp = null;
    with tabl1
    as (select *
        from kontstate
        where ks_region_id = :aa_curr_region_id and
              ks_kc_id = :aa_curr_kc_id and
              ks_ol_is_delete = 0 and
              ks_state = 2 and
              cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
              not exists(select *
                         from sb_akt, sb_list -- Пункт 2.3
                         where sba_is_delete = 0 and
                               sba_status = 2 and
                               sba_from_sbl_id <> 0 and
                               :rep_date_end between sba_date_begin and sba_date_end and
                               sba_from_k_id = ks_from_k_id and
                               sba_from_sbl_id = sbl_id and
                               sbl_type = 1)),
    tabl2
    as (select tabl1.*
        from tabl1, oksummary
        where ks_from_k_id = oks_from_k_id and
              ks_date_begin - 1 between oks_date_begin and oks_date_end and
              oks_delinq_date < '01.01.2014' and
              oks_close_state = 0 and
              exists(select *
                     from operkred
                     where ok_ol_is_active = 1 and
                           ok_type = 4 and
                           ok_os + ok_perc + ok_fine < 0 and
                           ok_from_k_id = oks_from_k_id and
                           ok_from_ol_id = ks_from_ol_id and
                           ok_from_c_id = 0))
    select -sum(iif(ok_ol_date between '01.07.2014' and '30.09.2014', ok_os + ok_perc + ok_fine, 0)),
           -sum(iif(ok_ol_date between :rep_date_begin and :rep_date_end, ok_os + ok_perc + ok_fine, 0))
    from tabl2, operkred
    where ks_from_k_id = ok_from_k_id and
          ok_from_c_id = 0 and
          ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_os + ok_perc + ok_fine < 0 and
          (ok_ol_date between '01.07.2014' and '30.09.2014' or
          ok_ol_date between :rep_date_begin and :rep_date_end)
    into repay_delinq_closed, repay_delinq_temp;
    if (repay_delinq_closed is null) then
      repay_delinq_closed = 0;
    if (repay_delinq_temp is null) then
      repay_delinq_temp = 0;
    if (repay_delinq is null) then
      repay_delinq = 0;
    if (repay_delinq_temp > 0) then
      repay_delinq = repay_delinq_temp + repay_delinq;

    if (begin_balance > 0) then
      par7 = cast(delinq_balance as double precision) / begin_balance * 100;
    else
      par7 = 100;

-- Вычисления бонус фактора кол-во клиентов и пар, пункт 3.1
    kol_cli_field = case
                      when :aa_just_cl_end <= 200 then 1
                      when :aa_just_cl_end <= 250 then 2
                      when :aa_just_cl_end <= 300 then 3
                      when :aa_just_cl_end <= 350 then 4
                      when :aa_just_cl_end <= 400 then 5
                      else 6
                    end;

    if (kol_cli_field = 1) then
      factor_kol_cli = 0;
    else
    begin
      factor_kol_cli = case kol_cli_field
                         when 2 then case
                                       when par7 <= 0.1 then 23
                                       when par7 <= 0.5 then 19.55
                                       when par7 <= 1.0 then 17.25
                                       else 0
                                     end
                         when 3 then case
                                       when par7 <= 0.1 then 27
                                       when par7 <= 0.5 then 22.95
                                       when par7 <= 1.0 then 20.25
                                       else 0
                                     end
                         when 4 then case
                                       when par7 <= 0.1 then 32
                                       when par7 <= 0.5 then 27.20
                                       when par7 <= 1.0 then 24.00
                                       else 0
                                     end
                         when 5 then case
                                       when par7 <= 0.1 then 37
                                       when par7 <= 0.5 then 31.45
                                       when par7 <= 1.0 then 27.75
                                       else 0
                                     end
                         else case
                                when par7 <= 0.1 then 40
                                when par7 <= 0.5 then 34
                                when par7 <= 1.0 then 30
                                else 0
                              end
                       end;
    end
    bonus_3_1 = aa_just_cl_end * factor_kol_cli;

    if (aa_just_cl_end - aa_just_cl_begin >= 1) then
    begin
      bonus_4_1 = repay_delinq * 10 / 100;
    end
    else
    begin
      bonus_4_1 = repay_delinq * 5 / 100;
    end

    if (:rep_date_end <= '30.09.2014') then
      bonus_4_2 = repay_delinq_closed * 2.5 / 100;
    else
      bonus_4_2 = 0;

--  Пункт 5, Бонус за возврат суммы кредитов отнесенные н внесистемный учет
    select -sum(oko_os + oko_perc)
    from knaktlist, kontract k1, operlist ol1, operkroff oko1
    where knaktlist.is_delete = 0 and
          kal_new_kf_officer = :aa_curr_kc_id and
          ol1.ol_date between kal_date and kal_date_end and
          kal_from_kf_id = k_from_kf_id and
          k_id = oko_from_k_id and
          oko_type = 4 and
          oko_from_ol_id = ol1.ol_id and
          ol1.is_delete = 0 and
          ol1.ol_date_oper_day between :rep_date_begin and :rep_date_end and
          exists(select distinct oko2.oko_from_k_id
                 from operkroff oko2, operlist ol2
                 where oko2.oko_from_k_id = k1.k_id and
                       oko2.oko_type = 1 and
                       oko2.oko_from_ol_id = ol2.ol_id and
                       ol2.is_delete = 0 and
                       ol2.ol_date < '01.01.2014') and
          not exists(select *
                     from sb_akt, sb_list -- Пункт 2.3
                     where sba_is_delete = 0 and
                           sba_status = 2 and
                           sba_from_sbl_id <> 0 and
                           :rep_date_end between sba_date_begin and sba_date_end and
                           sba_from_k_id = k1.k_id and
                           sba_from_sbl_id = sbl_id and
                           sbl_type = 1)
    into pogash_vnesystem;
    if (pogash_vnesystem is null) then
      pogash_vnesystem = 0;
    bonus_5_1 = pogash_vnesystem * 7.5 / 100;

    bonus_total = bonus_3_1 + bonus_4_1 + bonus_4_2 + bonus_5_1;
    jl_name = null;
    ps_is_on_staff = null;
    ps_is_working = null;
    select jl_name, ps_is_on_staff, ps_is_working, jk_system_code, ps_date_begin
    from hr_pstatus, joblist, jobkind
    where ps_from_c_id = :aa_curr_kc_id and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id
    order by ps_is_working desc
    rows 1
    into jl_name, ps_is_on_staff, ps_is_working, jk_system_code, sks_date_begin;
    if (ps_is_working is null) then
      ps_is_working = 0;
    if (ps_is_working = 0) then
      bonus_total = 0; -- если сотрудник уволен то не начислять бонус
--  Разработан по приказу от 26.02.2015г.
--  если продолжительности стажа КС 6 более месяцев то имеет доступ к бонусам
    stage_date = current_date;
    if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
        bonus_total > 0) then
    begin
--    если продолжительность КСника менее 6 месяцев то проверяет следующее
      if (dateadd(month, 6, sks_date_begin) > :rep_date_end) then
      begin
--      вычисляет продолжительность работы КСника и возвращает самую первую дату когда был КСником
--      если среди них разрыв был то берет до разрыва
        for select ps_date_begin, ps_date_end
            from hr_pstatus, joblist, jobkind
            where ps_from_c_id = :aa_curr_kc_id and
                  ps_from_jl_id = jl_id and
                  jl_kind = jk_id and
                  jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') -- в проверку включает КС и СКС, если вдруг раньше был СКС и стал КСником то это в стаже включалось
                  and
                  ps_status = 1 and
                  ps_is_working = 1
            order by ps_date_begin desc
            into ps_date_begin, ps_date_end
        do
        begin
          if (ps_date_begin = sks_date_begin) then
          begin
            stage_date = ps_date_begin;
            ps_date_begin_last = ps_date_begin;
          end
          else
          begin
            if (ps_date_end = ps_date_begin_last - 1) then
            begin
              stage_date = ps_date_begin;
              ps_date_begin_last = ps_date_begin;
            end
          end
        end

        if (dateadd(month, 6, stage_date) > :rep_date_end) then
        begin
          bonus_total = 0;
        end

      end
    end

--  вытаскивает данные по выговорам и бонус коэффициент
    select bonus_koef, bonus_sum, bonus_memo
    from hp_get_vigovor_koef(:aa_curr_kc_id, :rep_date_end)
    into bonus_koef, bonus_sum, bonus_memo;
--  Расчет бонуса КС
    bonus_end = bonus_total * bonus_koef / 100 - bonus_sum;
    if (bonus_end < 0) then
      bonus_end = 0;

    select c_name_fio
    from clients
    where c_id = :aa_curr_kc_id
    into aa_curr_kc_fio;

    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
    from region r1, region r2, region r3
    where r1.reg_unicode = :aa_curr_region_id and
          r1.parent = r2.reg_unicode and
          r2.parent = r3.reg_unicode
    into reg_name, branch_name, okrug_name;

--  Если по системной должности СКС то дополнительный расчет бонуса для СКС
    if (curr_reg_id_prior <> aa_curr_region_id) then
    begin
      curr_reg_id_prior = aa_curr_region_id;
      reg_par8_180 = null;
      reg_kol_cli = null;
      select reg_par8_180, reg_kol_cli
      from temp_sks_reg_info
      where reg_id = :aa_curr_region_id
      into reg_par8_180, reg_kol_cli;
      if (reg_par8_180 is null) then
        reg_par8_180 = 0;
      if (reg_kol_cli is null) then
        reg_kol_cli = 0;
    end

    sks_memo = null;
    bonus_sks = 0;
    if (jk_system_code = 'JobSystem.LO_Senior') then
    begin
      if (sks_date_begin > date_half_month) then
      begin
        sks_reg_id_last = null;
        select ps_from_reg_id
        from hr_pstatus, joblist, jobkind
        where :date_half_month between ps_date_begin and ps_date_end and
              ps_status = 1 and
              ps_is_working = 1 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code = 'JobSystem.LO_Senior' -- Выбор
              and
              ps_from_c_id = :aa_curr_kc_id
        into sks_reg_id_last;
        if (sks_reg_id_last is not null) then
        begin
          reg_name_last = (select reg_code || ' ' || reg_name
                           from region
                           where reg_unicode = :aa_curr_region_id);
          sks_memo = 'СКС перевелся в офис ' || reg_name_last || ', от ' || sks_date_begin;
          aa_curr_region_id = sks_reg_id_last;
        end
      end

      if (ps_is_working = 1 and
          reg_kol_cli > 1500) then
      begin
        bonus_sks = case
                      when reg_par8_180 = 0 then iif(reg_kol_cli > 2000, 7500, 5500)
                      when reg_par8_180 <= 1 then iif(reg_kol_cli > 2000, 6000, 4400)
--                          when reg_par8_180 <= 1.5 then iif(reg_kol_cli > 2000, 4500, 3300)
                      else 0
                    end;
        if (:rep_date_end > '30.04.2014' and
            reg_par8_180 > 1) then
          bonus_sks = 0;

        bonus_end = bonus_sks * bonus_koef / 100 - bonus_sum + bonus_end;
        if (bonus_end < 0) then
          bonus_end = 0;
      end
    end

    suspend;
  end

end
