CREATE PROCEDURE AAREP_PLAN_FACT_DISB_VIPS(
    REP_REGION_ID D_ID,
    REP_DATE D_DATE,
    REP_KC_ID D_ID = null)
RETURNS (
    REG_ID D_ID,
    OKS_KC_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    OFFICE_NAME D_STR_100,
    REG_NAME D_STR_100,
    KC_OFFICER_NAME D_STR_100,
    LO_STAGE_MONTH D_INTEGER,
    PLAN_DISB_SUM CURR_TYPE,
    FACT_DISB_SUM CURR_TYPE,
    OKS_BALANCE D_CURRENCY,
    KOL_CL D_INTEGER)
AS
declare variable reg_id_tmp d_id;
declare variable rd_id d_id;
declare variable rep_period d_integer;
begin
  select rd.rd_id
  from refdata rd
  where rd_code = 'planindicators.disb_amount'
  into rd_id;

  rep_date_begin = cast('01.' || right('0' || extract(month from :rep_date - 1), 2) || '.' || extract(year from :rep_date - 1) as date);
  rep_period = 100 * extract(year from rep_date_begin) + extract(month from rep_date_begin);

  for select distinct pd_from_reg_id, pd_from_c_id, pd_data_1
      from hp_tmp_plandata_show(:rep_region_id, :rep_date_begin, :rd_id, :rep_period, 1, null)
      where pd_is_office = 2 and
            pd_data_1 > 0 and
            (pd_from_c_id = :rep_kc_id or
            :rep_kc_id is null)
      into reg_id, oks_kc_id, plan_disb_sum
  do
  begin
    lo_stage_month = 0;
    select lo_stage_month_h
    from hp_get_lo_stage(:oks_kc_id, :rep_date)
    into lo_stage_month;
    if (coalesce(lo_stage_month, -1) < 0) then
      lo_stage_month = 0;

    kc_officer_name = '';
    select c_name_fio
    from clients c
    where c_id = :oks_kc_id
    into kc_officer_name;

    fact_disb_sum = 0;
    select sum(k_amount_approved)
    from kontstate ks
    inner join kontract k on ks_from_k_id = k_id
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_date_begin and :rep_date and
          k.k_fact_disb_lo_id = :oks_kc_id
    into fact_disb_sum;

    oks_balance = 0;
    kol_cl = 0;
    select sum(oks_balance), sum(k_kolich_client_approved)
    from oksummary ok
    inner join kontract k on oks_from_k_id = k_id
    where :oks_kc_id = oks_kc_id and
          :rep_date between oks_date_begin and oks_date_end
          and oks_close_state = 0
    into :oks_balance, :kol_cl;

    if (coalesce(reg_id_tmp, 0) <> reg_id) then
    begin
      select r1.reg_name, r2.reg_name
      from region r1
      inner join region r2 on r1.parent = r2.reg_unicode
      where r1.reg_unicode = :reg_id
      into office_name, reg_name;

      reg_id_tmp = reg_id;
    end

    suspend;
  end
end
