CREATE PROCEDURE AAREP_EMPL_DOC_CHANGE(
    REP_DATE DATE)
RETURNS (
    REC_NO INTEGER,
    C_NAME_FIO VARCHAR(255),
    JL_NAME VARCHAR(100),
    REG_NAME VARCHAR(50),
    DOV_STR VARCHAR(50))
AS
declare variable dov_status integer;
declare variable ps_from_c_id id_type;
begin
  rec_no = 0;
  for select distinct hrp.ps_from_c_id, c_name_fio, reg_name, jl_name
      from hr_pstatus hrp, clientdocs cd, clients cl, region reg, joblist jl
      where hrp.ps_status = 1 and
            hrp.ps_is_working = 1 and
            :rep_date between hrp.ps_date_begin and ps_date_end and
            ps_from_c_id = cd.cds_from_c_id and
            cd.cds_doc_date_begin >= :rep_date - 3 and
            cd.cds_is_delete = 0 and
            cl.c_id = ps_from_c_id and
            hrp.ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id
      into ps_from_c_id, c_name_fio, reg_name, jl_name
  do
  begin
    rec_no = rec_no + 1;

    dov_status = 0;

    select iif(count(dvc_from_c_id) > 0, 1, 0)
    from dovclients, dovlist dl
    where dvc_from_c_id = :ps_from_c_id and
          dvc_from_dl_id = dl_id and
          dl.dl_status = 1 and
          :rep_date between dl.dl_date_begin and iif(dl.dl_date_end_fact is null, dl.dl_date_end, dl.dl_date_end_fact)
    into dov_status;

    dov_str = '';
    if (dov_status = 1) then
      dov_str = 'Есть доверенность';

    suspend;
  end
end
