CREATE PROCEDURE HP_WRITEOFF_ASP_CANCEL
AS
declare variable c_id d_id;
declare variable rep_date d_date;
declare variable cancel_date d_date;
declare variable asp_kind d_integer;
declare variable cell_app_kind d_integer;
declare variable push_body d_str_long;
begin
  -- RFCRU-4701
  -- процедура ищет клиентов у кого все договоры списаны а АСП активен
  -- и расторгает АСП с отправкой ПУШ
  asp_kind = f_edoc_kind('asp.agreement', 'id');
  cell_app_kind = f_edoc_kind('cell.application', 'id');
  rep_date = current_date - extract(day from current_date);  -- ищем только до конца предыдущего месяца

  for select kc_from_c_id
  from oksummary
  inner join knclient on kc_from_k_id = oks_from_k_id
  inner join edoc_list on edl_object_kind = 0 and edl_object_id = kc_from_c_id + 0
                      and edl_from_edc_id = :asp_kind
                      and coalesce(edl_date_end, current_date) >= current_date
                      and edl_status >= 0
  left join operkroff on oko_from_k_id = kc_from_k_id
                      and oko_ol_is_active = 1
                      and oko_type = 1
                      and oko_sh_os + oko_sh_perc + oko_other > 0
  where current_date between oks_date_begin and oks_date_end
    and (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc) > 0 -- без ГП
  group by 1
  having min(oks_close_state) = 1
    and min(oks_out_os + oks_out_perc + oks_out_other) > 0
    and max(oko_ol_date) <= :rep_date
  into c_id
  do
  begin
    -- удостоверимся что нет открытых заявок
    if (exists(select 1 from hp_clientstate_get(:c_id) where res_client_state in (0,1))) then
      continue;

    -- расторгаем АСП
    cancel_date = f_date_work_day(current_date + 1); -- дата расторжения - следующий рабочий день

    update edoc_list
    set edl_date_end = :cancel_date - 1
    where edl_object_kind = 0
      and edl_object_id = :c_id
      and edl_from_edc_id in (:asp_kind, :cell_app_kind)
      and coalesce(edl_date_end, :cancel_date) >= :cancel_date
      and edl_status >= 0;

    -- отправка ПУШ клиенту
    push_body = f_format('Уважаемый(ая) %1, уведомляем Вас о досрочном расторжении Договора на оказание информационных услуг заключенного с ООО «МКК М Булак» с %2.',
                         (select c_name_fio from clients where c_id = :c_id),
                         f_datetostr(cancel_date));

    f_send_push('clients', c_id, 'Расторжение договора АСП', push_body, 'hp_writeoff_asp_cancel', -1);
  end
end
