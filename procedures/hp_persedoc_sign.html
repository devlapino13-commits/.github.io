CREATE PROCEDURE HP_PERSEDOC_SIGN(
    EDL_ID D_ID)
AS
declare variable c_id d_id;
declare variable edl_create_datetime d_datetime;
declare variable pda_id d_id;
declare variable pva_id d_id;
declare variable edc_pers_info d_str_255;
declare variable pif_uuid d_uuid;
declare variable pin_id d_id;
begin
  if (f_get_db_id() <> 81001) then
    exit;

  for select eds_from_c_id, edc_pers_info, edl_create_datetime
  from edoc_sign, edoc_list, edoc_cfg
  where edl_id = :edl_id and
        eds_from_edl_id = edl_id and
        eds_status = 1 and
        edl_status = 1 and
        edl_from_edc_id = edc_id and
        coalesce(edc_pers_info, '') <> ''
  into c_id, edc_pers_info, edl_create_datetime
  do
  begin
    if (edc_pers_info = 'all') then
    begin
      for select pda_id
      from persinfo, persdata
      where pin_object_kind = 0 and
            pin_object_id = :c_id and
            pin_id = pda_from_pin_id
      into pda_id
      do
      begin
        for select pva_id
        from persvalue
        where pva_from_pda_id = :pda_id and
              pva_datetime <= :edl_create_datetime
        order by pva_datetime desc, pva_id desc
        rows 1
        into pva_id
        do
        begin
          if (not exists(select * from persedoc
                         where ped_pva_id = :pva_id and
                               ped_from_edl_id is not null)) then
          begin
            update or insert into persedoc (ped_pva_id, ped_from_edl_id)
            values (:pva_id, :edl_id)
            matching (ped_pva_id);
          end
        end
      end
    end else
    begin
      edc_pers_info = replace(edc_pers_info, ' ', '');
      edc_pers_info = replace(edc_pers_info, ',', '],[');
      edc_pers_info = '[' || edc_pers_info || ']';

      for select pin_id
      from persinfo
      where pin_object_kind = 0 and
            pin_object_id = :c_id
      into pin_id
      do
      begin
        for select pif_uuid
        from ux$persfield
        where :edc_pers_info containing '[' || pif_source || ']' or
              :edc_pers_info containing '[' || pif_field_name || ']'
        into :pif_uuid
        do
        begin
          for select pda_id
          from persdata
          where pda_from_pin_id = :pin_id and
                pda_from_pif_uuid = :pif_uuid
          into pda_id
          do
          begin
            for select pva_id
            from persvalue
            where pva_from_pda_id = :pda_id and
                  pva_datetime <= :edl_create_datetime
            order by pva_datetime desc, pva_id desc
            rows 1
            into pva_id
            do
            begin
              if (not exists(select * from persedoc
                             where ped_pva_id = :pva_id and
                                   ped_from_edl_id is not null)) then
              begin
                update or insert into persedoc (ped_pva_id, ped_from_edl_id)
                values (:pva_id, :edl_id)
                matching (ped_pva_id);
              end
            end
          end
        end
      end
    end
  end
end
