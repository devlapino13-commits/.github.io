CREATE PROCEDURE OKT_INFO_SAVE_2_MAIN(
    PROJECT STR_100,
    OBJ_ID ID_TYPE,
    OBJECT_KIND STR_100,
    OBJECT_ID ID_TYPE,
    UN_ID ID_TYPE,
    PARAMS D_BLOB_TEXT)
RETURNS (
    RES SMALLINT)
AS
declare variable KEY_ D_STR_64;
declare variable VALUE_ D_STR_2048;
declare variable EQUALITY_INDEX D_INTEGER;
declare variable SEMICOLON_INDEX D_INTEGER;
declare variable COMMA_INDEX D_INTEGER;
declare variable PARAM_STRING D_BLOB_TEXT;
declare variable PHONE_NUMBER D_STR_20;
declare variable PHONE_ID D_INTEGER;
declare variable NEW_DURATION D_STR_255;
declare variable NEW_FILES D_STR_2048;
declare variable OBJECT_KIND_ D_INTEGER;
declare variable OBJECT_ID_ ID_TYPE;
begin
    -- OKTELL.
    -- на данный момент, процедура предназначена для сохранения длительности диалога между оператором и абонентом (duration=...)
    -- в будущем, может использоваться для сохранения каких-либо дополнительных значений.
    if (project containing ('OKTELL.')) then
    begin
        -- DMIHAILECHKO -- RFCKG-2531 -->
        -- если параметров в params будет 42, цикл отработает 42 раза.
        -- цикл пройдётся по всем параметрам, в данном случае их тут два (duration и file)
        -- и возьмёт значение для каждого из них, главное соблюдать "=" и ";"
        -- каждый параметр это ключ1=значение1;
        while (char_length(trim(params)) > 0) do
        begin
            key_ = '';
            value_ = '';
            equality_index = 0;
            semicolon_index = 0;

            semicolon_index = position(';', params);
            if (semicolon_index = 0) then
                semicolon_index = char_length(params) + 1;
    
            param_string = trim(substring(params from 1 for semicolon_index - 1));
    
            equality_index = position('=', param_string);
            if (equality_index = 0) then
                exception e_default 'пропущен знак равенства' || param_string;
    
            key_ = trim(substring(param_string from 1 for equality_index - 1));
            value_ = trim(substring(param_string from equality_index + 1));
    
            if (key_ is null or char_length(key_) = 0) then
                exception e_default 'ключь не может быть пустым';
            if (value_ is null or char_length(value_) = 0) then
                exception e_default 'значение с ключом' || key_ || ' не может быть пустым ';

            -- обрезаю уже использованные ключ-значения и оставляю только с которыми ещё предстоит работа
            params = trim(substring(params from semicolon_index + 1));
    
            if (trim(lower(key_)) = 'phone_number') then
                phone_number = value_;
            if (trim(lower(key_)) = 'duration') then
                new_duration = value_;
            if (trim(lower(key_)) = 'file_path') then
                new_files = value_;

        end

        if (new_files is not null) then
        begin
            -- RFCKG-2832
            phone_id = null;
            select pop_id
            from pantera_oper_phones
            where pop_phone = :phone_number
            into :phone_id;

            if (phone_id is null) then
                insert into pantera_oper_phones (pop_phone)
                values (:phone_number)
                returning pop_id
                into :phone_id;

            -- RFCKG-2803
            if(object_kind = 'Clients') then
            begin
                :object_kind_ = f_objkind('pantera_clients', 'id');

                select first 1 pl.pcl_id
                from pantera_clients pl
                where pl.pcl_from_pan_id = :obj_id
                    and pl.pcl_from_c_id = :object_id
                order by pl.pcl_id desc
                into :object_id_;

                if(object_id_ is null) then
                    exit; -- RFCKG-2836
                    --exception e_default 'по данному obj_id ' || obj_id || ' и object_id ' || object_id || ' ничего не найдено';

                update pantera_clients pl
                set pl.pcl_duration = cast(:new_duration as integer),
                    pl.pcl_phone_id = :phone_id
                where pl.pcl_id = :object_id_;
            end
    
            if (object_kind = 'Contacts') then
            begin
                :object_kind_ = f_objkind('pantera_contacts', 'id');

                select first 1 pc.pcon_id
                from pantera_contacts pc
                where pc.pcon_from_pan_id = :obj_id and
                      pc.pcon_from_cc_id = :object_id
                order by pc.pcon_id desc
                into :object_id_;

                if(object_id_ is null) then
                    exit; -- RFCKG-2836
                    --exception e_default 'по данному obj_id ' || obj_id || ' и object_id ' || object_id || ' ничего не найдено';

                update pantera_contacts pc
                set pc.pcon_duration = cast(:new_duration as integer),
                    pc.pcon_phone_id = :phone_id
                where pc.pcon_id = :object_id_;
            end

            -- если файлов в new_files будет 42, цикл отработает 42 раза.
            value_ = '';
            while (char_length(trim(new_files)) > 0) do
            begin
                comma_index = 0;
                comma_index = position(',', new_files);
                if (comma_index = 0) then
                    comma_index = char_length(new_files) + 1;

                value_ = trim(substring(new_files from 1 for comma_index - 1));
                if (value_ is null or char_length(value_) = 0) then
                    exception e_default 'путь к файлу не может быть пустым';

                -- обрезаю уже использованные файлы и оставляю только с которыми ещё предстоит работа
                new_files = trim(substring(new_files from comma_index + 1));

                insert into okt_dialog_path (odp_dialog_path, odp_object_kind, odp_object_id)
                values (:value_, :object_kind_, :object_id_);
            end
        end

        suspend;
    end
end
