CREATE PROCEDURE HP_EXTMINIWEB_INS(
    FIO D_STR_100,
    BIRTH_DATE D_DATE,
    LANGUAGE D_STR_100,
    CITY D_STR_100,
    PHONE D_STR_20,
    PARTNER_UUID D_STR_CODE,
    SMS_CODE D_STR_10)
RETURNS (
    RES_TEXT D_STR_255,
    RES_CODE D_BOOL)
AS
declare variable emw_id d_id;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable pnr_from_c_id d_id;
declare variable lo_id_partner d_id;
declare variable last_reg_id d_integer;
declare variable city_partner d_str_100;
declare variable un_id_lo d_id;
declare variable uid d_str_code;
declare variable un_id_robot d_id;
declare variable res_text1 d_str_255;
declare variable send_text str_255;
declare variable partner_fio str_100;
begin
  select un_id from username
  where un_name = 'FINSTART_ROBOT'
  into un_id_robot;

  if (un_id_robot is null) then un_id_robot = 1000000000000;

--  uid_char = uuid_to_char(:partner_uuid);
  insert into extminiweb (emw_date, emw_datetime, emw_cl_fio, emw_birth_date, emw_language, emw_city, emw_phone,
                          emw_status, emw_from_un_id, emw_uuid, emw_sms_code)
  values (current_date, current_timestamp, :fio, :birth_date, :language, :city, :phone,
          0, :un_id_robot, char_to_uuid(:partner_uuid), :sms_code)
  returning emw_id
  into :emw_id;

  if (emw_id is not null) then
  begin
    res_text = 'Спасибо за обращение! В ближайшее время с вами свяжется наш специалист';
    res_code = 0;
  end else
  begin
    res_text = 'Ошибка!';
    res_code = 1;
  end

  -- добавление в друзья партнера
  if (res_code = 0) then
  begin
    uid = :partner_uuid;
    select obj_kind, -- partner
           obj_id    -- pnr_id
    from hp_uuid(:uid)
    into obj_kind, obj_id;

    if (obj_kind is null) then -- ругаться что ссылка не верная
    begin
      res_text = 'Ссылка не верная!';
      res_code = 1;
      suspend;
      exit;
    end 

    if (f_objkind(:obj_kind, 'code') <> 'partner') then -- ругаться что ссылка не верная
    begin
      res_text = 'Ссылка не верная!';
      res_code = 1;
      suspend;
      exit;
    end 

    -- находим партнера
    select pnr_from_c_id from partner
    where pnr_id = :obj_id
    into pnr_from_c_id;

    -- добавляем в друзья партнера
    -- нужно подставлять того пользователья который является ВИПСом партнера, если они находятся в одном городе
    --    , тем самым ВИПС может видеть друзей своего клиента(партнера)

    -- находим ВИПСА партнера и регион
    select kal_new_kf_officer, kal_new_reg_id
    from knclient, kontstate, kontract, knaktlist
    where kc_from_c_id = :pnr_from_c_id
    and kc_amount_approved > 0
    and kc_from_k_id = ks_from_k_id
    and ks_ol_is_delete = 0
    and ks_state = 1
    and ks_from_k_id = k_id
    and k_from_kf_id = kal_from_kf_id
    and knaktlist.is_delete = 0
    and current_date between kal_date and kal_date_end
    order by kc_id desc
    rows 1
    into lo_id_partner, last_reg_id;

    select c_name_fio
    from clients
    where c_id = :pnr_from_c_id
    into partner_fio;

    -- находим город партнера
    city_partner = null;
    select PL_NAME
    from region, addrlist, places
    where reg_address_fact = adl_id
    and adl_from_pl_id = pl_id
    and reg_unicode = :last_reg_id
    into city_partner;

    un_id_lo = null;
    if (city_partner = city) then -- если города совпадают
    begin
      -- находим un_id випса
      select un_id from username
      where un_from_c_id = :lo_id_partner
      into un_id_lo;
    end 
    if (un_id_lo is null) then un_id_lo = :un_id_robot;

    -- если un_id_lo = 1000000000000 то нужно кидать в общий катёл

    select res_text
    from fm_clientlink_add(:un_id_lo, :pnr_from_c_id, :fio, :phone)
    into :res_text1;

    if (res_text1 <> 'Успешно добавлен') then
    begin
      res_text = res_text1;
      -- если номер успешно не добавлено то убираем из сохраненных
      delete from extminiweb
      where emw_id = :emw_id;
    end 

    if (un_id_lo <> un_id_robot) then
    begin
      update extminiweb set emw_uuid = :un_id_lo
      where emw_id = :emw_id;
    end 

    -- если успешно добавлен и город друга совпадает с партнером, то отправляем пуш
    if (res_text1 = 'Успешно добавлен' and city_partner = city) then
    begin
      send_text = 'Друг ' || phone || ', ' || fio || ' вашего Партнера ' || partner_fio || ' отправил анкету. Зайдите в МП и свяжитесь с ним';
      execute procedure hp_send_push_to_staff(:lo_id_partner, 'У Партнера добавился Друг', :send_text, 1000000000000, 'msg_to_staff', 7);
    end
  end 

  suspend;
end
