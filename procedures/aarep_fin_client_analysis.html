CREATE PROCEDURE AAREP_FIN_CLIENT_ANALYSIS(
    REPORT_REGION_ID D_INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    C_NAME_FIO D_STR_255,
    KONTRACT_NO D_STR_20,
    K_FACT_DISB_DATE D_DATE,
    OFFICE_NAME D_STR_64,
    OFFICER_FIO_NAME D_STR_255,
    TYPE_FINANCING D_STR_20,
    PR_NAME D_STR_64,
    KC_AMOUNT DECIMAL(18,4),
    BT_NAME D_STR_64,
    BS_NAME D_STR_64,
    KC_LOAN_TARGET_STR D_STR_100,
    LOAN_PURPOSE D_STR_100,
    LOAN_PURPOSE_DETAIL D_STR_LONG,
    MAIN_INCOME DECIMAL(18,4),
    EXTRA_INCOME DECIMAL(18,4),
    EXPENSES_TOTAL DECIMAL(18,4),
    OTHER_DEBT_PAYMENTS DECIMAL(18,4),
    MONTHLY_PAYMENT DECIMAL(18,4),
    NET_PROFIT DECIMAL(18,4),
    KPD DECIMAL(18,4),
    KNC_REP_ACTIVE_BALANCE DECIMAL(18,4),
    K_BY_RATING D_STR_CODE)
AS
declare variable K_ID D_ID;
declare variable PR_ID D_ID;
declare variable KC_ID D_ID;
declare variable C_ID D_ID;
declare variable MA_ID D_ID;
declare variable MAC_ID D_ID;
declare variable KL_DATE D_DATE;
declare variable REP_DATE D_DATE;
declare variable COMMUNAL_EXPENSES decimal(18,4);
declare variable LIVING_EXPENSES decimal(18,4);
declare variable LEGAL_DOCS_EXPENSES decimal(18,4);
declare variable OTHER_BUSINESS_EXPENSES decimal(18,4);
declare variable AAREP_K_BALANCE decimal(18,4);
declare variable K_AMOUNT_APPROVED decimal(18,4);
declare variable BY_RATING D_SMALLINT;
begin

  for
  select aki.kk_id, aki.kk_balance,
         cl1.c_name_fio, kn.k_kontract_no, kn.k_fact_disb_date,
         reg.reg_name, cl2.c_name_fio, kc.kc_amount_approved,
         bt.bt_name, bs.bs_name, rd.rd_name, coalesce(kn.k_by_rating, 0),
         iif(kn.k_by_rating = 1, 'По кредитному рейтингу', 'Стандартная выдача'),
         kn.k_from_pr_id, kn.k_amount_approved,
         kn.k_from_ma_app_id, kc.kc_from_c_id, kc.kc_id
  from aarep_active_kont_id_in_period(:report_region_id, null, :report_date_end, 0, 0, 0, 0, 0) aki
  left join kontract kn on kn.k_id = aki.kk_id
  left join knclient kc on kc.kc_from_k_id = kn.k_id
  left join clients cl1 on cl1.c_id = kc.kc_from_c_id
  left join clients cl2 on cl2.c_id = kn.k_fact_disb_lo_id
  left join knaktlist kal on kal.kal_from_kf_id = kn.k_from_kf_id and
         :report_date_end between kal.kal_date and kal.kal_date_end and kal.is_delete = 0
  left join region reg on reg.reg_unicode = kal.kal_new_reg_id
  left join bussubtype bs on bs.bs_id = kc.kc_busuness_subtype
  left join bustype bt on bt.bt_id = bs.bs_from_bt_id
  left join refdata rd on rd.rd_id = kc.kc_loan_target
  into :k_id, aarep_k_balance,
        c_name_fio, kontract_no, k_fact_disb_date,
        office_name, officer_fio_name, kc_amount,
        bt_name, bs_name, kc_loan_target_str, by_rating,
        k_by_rating,
        pr_id, k_amount_approved,
        ma_id, c_id, kc_id
  do
  begin
    if (by_rating <> 1) then
    begin
        select rd.rd_name, mac.mac_id
        from miniappclient mac
        left join refdata rd on rd.rd_id = mac.mac_loan_target
        where mac.mac_from_c_id = :c_id and
              mac.mac_from_ma_id = :ma_id
        into :loan_purpose, mac_id;
    
        select loan_description
        from hp_get_loan_description(:c_id, :ma_id)
        into :loan_purpose_detail;
    
        if (exists(select *
                   from cfp_anketa ca
                   where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
                         ca.ca_object_id = :mac_id and
                         ca.ca_from_c_id = :c_id)) then
        begin
          select ca.ca_avr_income
          from cfp_anketa ca
          where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
                ca.ca_object_id = :mac_id and
                ca.ca_from_c_id = :c_id
          into main_income;
        
          select extra_amount, communal_expenses, living_expenses, legal_docs_expenses,
                 other_business_expenses, repayment_obligations, net_profit, amount_monthly_payments
          from get_cfp_kpd(:mac_id)
          into :extra_income, :communal_expenses, :living_expenses, :legal_docs_expenses,
               :other_business_expenses, :other_debt_payments, :net_profit, :monthly_payment;
        end
        else
        begin
          select max(kl.kl_datetime)
          from kontlog kl
          where kl.kl_from_k_id = :k_id and
                kl.kl_state = 1
          into kl_date;
    
          rep_date = current_date;
          if (not exists(select *
                         from actionlist
                         where acl_object_kind = f_objkind('MiniAppClient', 'id') and
                               acl_object_id = :mac_id and
                               acl_from_rd_id = f_get_rd_id('anketa.solvency'))) then
            rep_date = coalesce(:kl_date, current_date);
    
          select monthly_income, extra_amount, communal_expenses, living_expenses, legal_docs_expenses,
                 other_business_expenses, repayment_obligations, net_profit, amount_monthly_payments
          from hp_get_kpd(:kc_id, :rep_date)
          into main_income, extra_income, communal_expenses, living_expenses, legal_docs_expenses, other_business_expenses,
               other_debt_payments, net_profit, monthly_payment;
        end
        expenses_total = communal_expenses + living_expenses + legal_docs_expenses + other_business_expenses;
        if (monthly_payment <> 0) then
          kpd = (net_profit/monthly_payment*100);
      end
      else
      begin
        loan_purpose = '';
        loan_purpose_detail = '';
        main_income = null;
        extra_income = null;
        other_debt_payments = null;
        monthly_payment = null;
        net_profit = null;
        expenses_total = null;
        kpd = null;
      end

      select iif(pr.pr_code containing '.ipf.', 'ипф', 'классика') ,
             pr.pr_name
      from product pr
      where pr.pr_id = :pr_id
      into :type_financing, pr_name;

      if (kc_amount <> 0) then
        knc_rep_active_balance = (cast(aarep_k_balance as double precision) / k_amount_approved) * kc_amount;
      else
        knc_rep_active_balance = 0;

      suspend;
  end
end



/*
-- communal_expenses - ежемес расходы на жилье
-- living_expenses - ежемес семейные расходы
-- legal_docs_expenses - расходы в мес на патент и соц. полис
-- other_business_expenses - другие ежемес расходы по бизнесу

officer_fio_name  - кс выдавший кредит
kontract_no - номер договора
k_fact_disb_date - дата выдачи
c_name_fio - фио клиента
type_financing - тип финансирования
pr_name - название продукта
kc_amount - сумма кредита клиента
bt_name - тип бизнеса клиента
bs_name - субтип бизнеса
kc_loan_target_str - цель кредита
loan_purpose - цель в заявке
loan_purpose_detail - уточнение по цели
office_name - офис

main_income - основной ежемесячный доход
extra_income - сумма ежемес доп доходов
other_debt_payments - погашение других долговых обязательств
monthly_payment - сумма ежемесячной выплаты
net_profit - чистая прибыль
kpd - кпд

expenses_total - расходы семейные по бизнесу

1010100 - Балыкчи
*/
