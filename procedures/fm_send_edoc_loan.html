CREATE PROCEDURE FM_SEND_EDOC_LOAN(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    K_DISB_DATE DATE,
    UN_NAME VARCHAR(50),
    RES_TEXT VARCHAR(1024))
AS
declare variable cm_database_id integer;
declare variable k_id decimal(18,0);
declare variable ksr_id integer;
declare variable reg_id integer;
declare variable edoc_status d_str_code;
begin
  -- Процедура для формирования электронного документа (договор займа и прочие бланки).

  cm_database_id = f_get_db_id();

  res_text = '';

  if (obj_kind = 'Miniapp') then
  begin
    res_text = iif(cm_database_id <> 94401, 'Отмена. Электронные документы можно формировать только по договорам в процессе', 'Ləğv olunur. Elektron sənədlər ancaq prossesdə olan müqavilələrde yaradıla bilər');
    suspend;
    exit;
  end

  select distinct k_id, k_last_reg_id, k_disb_date
  from kontract, knclient
  where kc_from_k_id = k_id and
        k_status = 0 and
        kc_amount_approved > 0 and
        kc_from_c_id = :obj_id
  union
  select k_id, k_last_reg_id, k_disb_date
  from kontract
  where k_id = :obj_id and
        k_status = 0
  order by 3 desc
  rows 1
  into :k_id, reg_id, k_disb_date;

  if (k_id is null) then
  begin
    res_text = iif(cm_database_id <> 94401, 'Отмена. Не найден договор в процессе.', 'Ləğv olunur. Prossesdə olan müqavilə mövcud deyil');
    suspend;
    exit;
  end
  else
  begin
    obj_kind = 'Kontract';
    obj_id = :k_id;
  end

  if (k_disb_date < current_date) then
  begin
    res_text = iif(cm_database_id <> 94401, 'Отмена. Дата выдачи меньше сегодняшней даты.', 'Ləğv olunur.  Buraxılış tarixi bu günün tarixindən fərqlənir.');
    suspend;
    exit;
  end

  edoc_status = f_edoc_status('loan.agreement', obj_id, k_disb_date);

  -- Возможные варианта функции:
  -- inactive
  -- empty
  -- signed
  -- unsigned
  -- cancelled

  if (edoc_status = 'inactive') then
    res_text = 'Отмена. Проект не активен';

  if (edoc_status = 'signed') then
    res_text = 'Отмена. Электронные документы уже подписаны. Сначала аннулируйте их.';

  if (edoc_status = 'unsigned') then
    res_text = 'Отмена. Электронные документы уже сформированы. Сначала аннулируйте их.';

  if (res_text <> '') then
  begin
    Suspend;
    Exit;
  end

  -- Проверка прав пользователя
  ksr_id = 0;
  select ksr_id
  from aarep_kontsign_access(:k_id, :un_id, :reg_id, '')
  into :ksr_id;
  if (ksr_id = 0) then
  begin
    res_text = iif(cm_database_id <> 94401, 'Отмена. У вас нет прав на этот регион.', 'Ləğv olunur. Sizin bu reqion(filial) üzrə müqaviləye hüququnuz yoxdur.');
    suspend;
    exit;
  end

  -- Проверка условий

  select un_name
  from username
  where un_id = :un_id
  into un_name;

  begin
    rdb$set_context('USER_SESSION', 'FINSTART_USER_LOGIN', un_name);

    if (exists(select *
               from hp_check_loan_konditions(:k_id, 0, 0, 'Loan_Approve', :un_id)
               where kf_error_is_critical > 0)) then
    begin
      res_text = iif(cm_database_id <> 94401, 'Отмена. Имеются критические несоответствия. Устраните их изучив детали.', 'Ləğv olunur. Kritik uyğunsuzluqlar mövcuddur. Məlumatları dəqiqləşdirib səhvləri aradan qaldırın.');
      suspend;
      exit;
    end

    when any do
    begin
      rdb$set_context('USER_SESSION', 'FINSTART_USER_LOGIN', null);
      exception;
    end
  end


  if (res_text = '') then
  begin
    -- Формирование электронных документов.
    select res_text
    from hp_edoc_ins_loan_agreement(:obj_id, :un_id)
    into :res_text;

    if (res_text = '') then
      res_text = 'Успешно. Подготовка электронных документов в процессе, по готовности будет отправлен СМС';
  end

  suspend;
end
