CREATE PROCEDURE UXP$UXBUGTEST_LIST(
    KIND D_SMALLINT,
    UA_UUID D_UUID,
    USER_NAME D_STR_CODE)
RETURNS (
    BTS_UUID D_UUID,
    BTS_TEXT D_STR_4096,
    BTS_JIRA_ID D_STR_CODE,
    BTS_IS_DONE D_BOOL,
    BTS_CREATE_USER_NAME D_STR_CODE,
    BTS_CREATE_DATETIME D_DATETIME,
    BTS_CREATE_DB_NAME D_STR_CODE,
    BTS_STATE D_SMALLINT,
    BTS_STATE_USER_NAME D_STR_CODE,
    BTS_STATE_DATETIME D_DATETIME,
    BTS_KIND_STR D_STR_CODE,
    BTS_OPEN_UA_UUID D_UUID,
    RES_UA_CAPTION D_STR_255,
    RES_UA_OPEN_CAPTION D_STR_255,
    RES_UA_OPEN_UXA_KIND D_STR_CODE,
    RES_JIRA_LINK D_STR_LONG)
AS
declare variable jira_web d_str_long;
declare variable bts_kind d_smallint;
declare variable bts_ua_uuid d_uuid;
declare variable bts_db_id d_integer;
declare variable bts_db_is_actual d_bool;
declare variable is_poweruser d_bool;
declare variable un_id d_id;
begin
  jira_web = 'https://jira.mbulak.kg/browse/';

  if (kind = -2) then
    kind = null;

  if (trim(coalesce(user_name, '')) = '') then
  begin
    user_name = uxf$user_login(uxf$user());
    user_name = trim(lower(user_name));
  end

  un_id = uxf$user_id(user_name, false);

  is_poweruser = uxf$user_is('poweruser', un_id);

  for select bts_uuid, bts_is_done, bts_ua_uuid, bts_open_ua_uuid, bts_kind,
             bts_text, bts_jira_id, bts_db_id, bts_db_is_actual, bts_create_user_name, bts_create_datetime,
             bts_state, bts_state_user_name, bts_state_datetime
  from ux$bugtest
  where (:ua_uuid is not null or bts_kind = coalesce(:kind, bts_kind)) and
        (:ua_uuid is null or bts_open_ua_uuid = :ua_uuid)
  order by bts_create_datetime desc
  into :bts_uuid, :bts_is_done, :bts_ua_uuid, :bts_open_ua_uuid, :bts_kind,
       :bts_text, :bts_jira_id, :bts_db_id, :bts_db_is_actual, :bts_create_user_name,
       :bts_create_datetime, :bts_state, :bts_state_user_name, :bts_state_datetime
  do
  begin
    bts_state_user_name = trim(coalesce(bts_state_user_name, ''));
    bts_create_user_name = trim(coalesce(bts_create_user_name, ''));

    if (bts_create_user_name <> user_name) then
    begin
      if (bts_state_user_name <> '' and bts_state_user_name <> user_name) then
        Continue;
    end

    if (is_poweruser = 0 and bts_state_user_name = '') then
      Continue;

    bts_kind_str = case bts_kind
      when -1 then 'Баг, ошибка'
      when 0 then 'Тестирование'
      when 1 then 'Доработка'
    end;

    res_ua_caption = '';

    select ua_name || '    / ' || uxa_fcode
    from ux$action, ux$alias
    where ua_uuid = :bts_ua_uuid and
          ua_uxa_uuid = uxa_uuid
    into res_ua_caption;

    res_ua_open_caption = '';

    res_ua_open_uxa_kind = '';
    if (bts_ua_uuid <> bts_open_ua_uuid) then
    begin
      select ua_name || '    / ' || uxa_fcode, uxf$kind(uxf$code(uxa_owner, 'kind'), 'caption')
      from ux$action, ux$alias
      where ua_uuid = :bts_open_ua_uuid and
            ua_uxa_uuid = uxa_uuid
      into res_ua_open_caption, res_ua_open_uxa_kind;
    end

    if (coalesce(bts_jira_id, '') <> '') then
      res_jira_link = jira_web || bts_jira_id;

    bts_create_db_name = case bts_db_id
      when 41701 then 'KG'
      when 39801 then 'KZ'
      when 81001 then 'RU'
      when 94401 then 'AZ'
      else 'xx'
    end;

    if (bts_db_is_actual = 0) then
      bts_create_db_name = bts_create_db_name || ', last';

    suspend;
  end
end
