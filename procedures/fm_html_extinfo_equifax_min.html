CREATE PROCEDURE FM_HTML_EXTINFO_EQUIFAX_MIN(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable mb_active d_integer;
declare variable has_active_debt str_100;
declare variable overdue_debt_sum d_currency;
declare variable curr_debt_sum d_currency;
declare variable debt_days d_integer;
declare variable bls_id d_id;
declare variable c_name_fio str_100;
declare variable lang d_str_lang;
declare variable row_fmt str_255;
declare variable calc_status str_255;
declare variable eil_date d_date;
declare variable all_debt_sum d_currency;
declare variable cred_sum_debt d_currency;
declare variable cred_sum d_currency;
declare variable error_txt str_255;
declare variable all_sum_all d_str_8100;
declare variable cred_date d_date;
declare variable sell_cnt d_integer;
declare variable sell_sum d_currency;
declare variable cred_active_order d_integer;
declare variable score_level d_smallint;
declare variable score_text d_str_255;
declare variable jso d_integer;
declare variable overdue_cnt d_integer;
declare variable active_cnt d_integer;
declare variable cred_no d_integer;
declare variable color d_str_10;
declare variable cred_uid d_str_code;
declare variable cred_uid_trunc d_str_code;
declare variable kontract_no d_str_code;
declare variable ko_too_big d_boolean;
begin
  lang = f_get_lang_code(un_id, null);
  row_fmt = '<tr><td>%1</td><td><b>%2</b></td>';
  
  select c_name_fio
  from clients
  where c_id = :c_id
  into c_name_fio;
  
  res_html = (select replace(replace(res_style, '</style>', ''), '</head>', '') from fm_html_styles);

  res_html = BLOB_APPEND(res_html, '.overlay {top: 0;right: 0;bottom: 0;left: 0;z-index: 10;display: none;background-color: rgba(0, 0, 0, 0.65);position: fixed;}' ||
      '.overlay:target {display: block;}' ||
      '.popup {top: -100%;right: 0;left: 50%;font-size: 14px;z-index: 20;margin: 0;width: 85%;min-width: 120px;max-width: 400px;position: fixed;padding: 15px;border: 1px solid #383838;background: #fefefe;}' ||
      '.overlay:target+.popup {-webkit-transform: translate(-50%, 0);-ms-transform: translate(-50%, 0);-o-transform: translate(-50%, 0);transform: translate(-50%, 0);top: 20%;}' ||
      '.close {top: -10px;right: -10px;width: 20px;height: 20px;position: absolute;padding: 0;border: 2px solid #ccc;-webkit-border-radius: 15px;-moz-border-radius: 15px;-ms-border-radius: 15px;-o-border-radius: 15px;border-radius: 15px;' ||
      'background-color: rgba(61, 61, 61, 0.8);text-align: center;font: 13px/20px ''Tahoma'', Arial, sans-serif;font-weight: bold;-webkit-transition: all ease .8s;-moz-transition: all ease .8s;-ms-transition: all ease .8s;-o-transition: all ease .8s;transition: all ease .8s;}' ||
      '.close:before {color: rgba(255, 255, 255, 0.9);content: "X";text-shadow: 0 -1px rgba(0, 0, 0, 0.9);font-size: 12px;}' ||
      '</style></head>');


  /********************** Скоринг Бюро *********************************/
  select eil_from_bls_id, eil_date
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id
    and eil_from_eic_id = eic_id
    and eic_code = 'equifax_40'
    and eil_is_json = 1
    and eil_status = 2
  order by eil_date desc, eil_id
  rows 1
  into bls_id, eil_date;
  
  if (bls_id is null) then
    calc_status = f_lang(lang, 'Кредитная история не получена');
  else
    calc_status = f_format('<br>%1: %2', 'Кредитная история получена', f_datetostr(eil_date));

  res_html = BLOB_APPEND(res_html, f_format('<body><h1>Источник №1</h1><a name="top"></a><h2>%1</h2>%2<br><br>', c_name_fio, calc_status));

  if (bls_id is null) then
  begin
    suspend;
    exit;
  end 

  active_cnt = 0;
  debt_days = 0;
  overdue_cnt = 0;
  all_sum_all = '';
  mb_active = 0;
  all_debt_sum = 0;
  sell_cnt = 0;
  sell_sum = 0;
  ko_too_big = false;
  
  for select cred_no, cred_sum, cred_sum_debt, cred_day_overdue, cred_sum_overdue, cred_date, error_txt, score_level, score_text, cred_uid, cred_active_order
  from aarep_equifax_cred(:bls_id, 'equifax_40')
  where cred_active_order in (1,2) or coalesce(error_txt, '') <> ''
  order by cred_date desc
  into cred_no, cred_sum, cred_sum_debt, debt_days, overdue_debt_sum, cred_date, error_txt, score_level, score_text, cred_uid, cred_active_order
  do
  begin
    if (cred_active_order = 1) then
    begin
      active_cnt = active_cnt + 1;
      overdue_debt_sum = coalesce(overdue_debt_sum, 0.00);
      curr_debt_sum = cred_sum_debt - overdue_debt_sum;
      all_debt_sum = all_debt_sum + curr_debt_sum;
      cred_uid_trunc = '****' || right(cred_uid, 6);
      color = 'green';
    
      if (overdue_debt_sum > 0) then
      begin
        overdue_cnt = overdue_cnt + 1;
        color = 'red';
      end
  
      kontract_no = (select kontract_no from hp_bki_get_uid_info(:cred_uid));
  
      if (char_length(all_sum_all) > 7100) then
        ko_too_big = true;
      else
        all_sum_all = all_sum_all || trim(iif(all_sum_all = '', '', '<br>')) ||
              f_format('<font color = "%1">№%2:</font> %3: <font color = "green">%4;</font>',
                       color,
                       cred_no,
                       f_numtostr(round(cred_sum, 2), 2) || ' от ' || f_datetostr(cred_date),
                       f_numtostr(round(curr_debt_sum, 2), 2)) ||
              iif(overdue_debt_sum > 0,
                  f_format('<font color = "red"> %1 - %2 дней; </font>',
                          f_numtostr(round(overdue_debt_sum, 2), 2),
                          debt_days),
                  '') ||
              iif (kontract_no is not null,
                    kontract_no,
                    f_format('<a href="#win%1">%2</a><a href="#x" class="overlay" id="win%1"></a><div class="popup">%3<a class="close" href="#close"></a></div><br>',
                        active_cnt, cred_uid_trunc, cred_uid));
  
      if (exists(select kc_from_k_id
             from knclient
             inner join kontstate on ks_from_k_id = kc_from_k_id
             where kc_from_c_id = :c_id
             and ks_state = 1
             and ks_date_begin = :cred_date
             and ks_ol_is_delete = 0))
      then
        mb_active = mb_active + 1;
    end
    else if (cred_active_order = 2) then --RFCRU-3406
    begin
      sell_cnt = sell_cnt + 1;                      -- Если кредит продан, увеличиваем счетчик и добавляем его сумму
      sell_sum = sell_sum + cred_sum_debt;
    end
  end

  if (ko_too_big) then
    all_sum_all = all_sum_all || '<b>...</b><br>';

  has_active_debt = f_format('<font%1>%2</font>', iif(overdue_cnt > 0,' color = "#FF0000"', ''), overdue_cnt);

  all_debt_sum = coalesce(all_debt_sum, 0.00);

  if (coalesce(error_txt, '') <> '') then
  begin
    res_html = BLOB_APPEND(res_html, f_format('<h1 style="color: red;">%1</h1></body>', error_txt));
    suspend;
    exit;
  end

  res_html = BLOB_APPEND(res_html, f_format(
    '<table width="100%"><col width="60%"><col width="40%"><th>%1</th><th>%2</th>', 'Категория', 'Значение'));
  
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'Общее количество активных кредитов'), active_cnt));
  
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'в том числе в М Булак'),
    mb_active));
  
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'Наличие активных кредитов с просрочкой'),
    has_active_debt));
  
  res_html = BLOB_APPEND(res_html, '<tr><td>' || f_lang(lang, 'Сумма текущей задолженности, руб.') || '</td><td><b>' || all_sum_all || '</b></td>');
  
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'Сумма общей текущей задолженности по активным кредитам, руб.'),
    f_numtostr(round(all_debt_sum, 2), 2)));
  
  -- Добавление информации о проданных кредитах
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'Количество проданных кредитов'),
    sell_cnt));
  
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'Текущий баланс по активным проданным кредитам, руб.'),
    f_numtostr(round(sell_sum, 2), 2)));
  
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'Самозапрет'),
    iif(f_is_selfblock(:c_id, current_date) = 1, 'да', 'нет')));

   if (score_level is null) then
   begin
    select json.parse(cast(blx_blob as d_blob_text))
    from hp_get_blobex_blob(:bls_id)
    into jso;

    select score_level, res_txt
    from hp_bki_scoring_check(:jso, current_date)
    into score_level, score_text;
   end 

   res_html = BLOB_APPEND(res_html, f_format(row_fmt,
      f_lang(lang, 'Результат скоринга'),
        '<span style="color: '||
            (case score_level
                when -1 then 'black'
                when 0 then 'red'
                when 1 then 'blue'
                when 2 then 'green'
                else 'black'
            end)||
         ';">'|| score_text ||'</span>'));

   res_html = BLOB_APPEND(res_html,
      f_format('</table><a href="#top">%1</a><br>'));


  /********************** НБКИ ***************************/
  bls_id = null;

  select eil_from_bls_id, eil_date
  from extinfolist, extinfocfg eic
  where eil_from_c_id = :c_id
    and eil_from_eic_id = eic_id
    and eic_code = 'nbki_40'
    and eil_is_json = 1
    and eil_status = 2
    and eil_date + eic_info_actual_days >= current_date
  order by eil_date desc, eil_id
  rows 1
  into bls_id, eil_date;

  if (bls_id is null) then
  begin
    suspend;
    exit;
  end 

  calc_status = f_format('<br>%1: %2', 'Кредитная история получена', f_datetostr(eil_date));
  res_html = BLOB_APPEND(res_html, f_format('<h1>Источник №2</h1><a name="top"></a>%1<br><br>', calc_status));

  active_cnt = 0;
  debt_days = 0;
  overdue_cnt = 0;
  all_sum_all = '';
  mb_active = 0;
  all_debt_sum = 0;
  sell_cnt = 0;
  sell_sum = 0;
  ko_too_big = false;
  
  for select cred_no, cred_sum, cred_sum_debt, cred_day_overdue, cred_sum_overdue, cred_date, cred_uid, cred_active_order
  from aarep_ko_nbki_cred(:bls_id)
  where cred_active_order in (1,2)
  order by cred_date desc
  into cred_no, cred_sum, cred_sum_debt, debt_days, overdue_debt_sum, cred_date, cred_uid, cred_active_order
  do
  begin
    if (cred_active_order = 1) then
    begin
      active_cnt = active_cnt + 1;
      overdue_debt_sum = coalesce(overdue_debt_sum, 0.00);
      curr_debt_sum = cred_sum_debt - overdue_debt_sum;
      all_debt_sum = all_debt_sum + curr_debt_sum;
      cred_uid_trunc = '****' || right(cred_uid, 6);
      color = 'green';
    
      if (overdue_debt_sum > 0) then
      begin
        overdue_cnt = overdue_cnt + 1;
        color = 'red';
      end
  
      kontract_no = (select kontract_no from hp_bki_get_uid_info(:cred_uid));
  
      if (char_length(all_sum_all) > 7100) then
        ko_too_big = true;
      else
        all_sum_all = all_sum_all || trim(iif(all_sum_all = '', '', '<br>')) ||
              f_format('<font color = "%1">№%2:</font> %3: <font color = "green">%4;</font>',
                       color,
                       cred_no,
                       f_numtostr(round(cred_sum, 2), 2) || ' от ' || f_datetostr(cred_date),
                       f_numtostr(round(curr_debt_sum, 2), 2)) ||
              iif(overdue_debt_sum > 0,
                  f_format('<font color = "red"> %1 - %2 дней; </font>',
                          f_numtostr(round(overdue_debt_sum, 2), 2),
                          debt_days),
                  '') ||
              iif (kontract_no is not null,
                    kontract_no,
                    f_format('<a href="#win%1">%2</a><a href="#x" class="overlay" id="win%1"></a><div class="popup">%3<a class="close" href="#close"></a></div><br>',
                        active_cnt, cred_uid_trunc, cred_uid));
  
      if (exists(select kc_from_k_id
             from knclient
             inner join kontstate on ks_from_k_id = kc_from_k_id
             where kc_from_c_id = :c_id
             and ks_state = 1
             and ks_date_begin = :cred_date
             and ks_ol_is_delete = 0))
      then
        mb_active = mb_active + 1;
    end
    else if (cred_active_order = 2) then --RFCRU-3406
    begin
      sell_cnt = sell_cnt + 1;                      -- Если кредит продан, увеличиваем счетчик и добавляем его сумму
      sell_sum = sell_sum + cred_sum_debt;
    end
  end

  if (ko_too_big) then
    all_sum_all = all_sum_all || '<b>...</b><br>';

  has_active_debt = f_format('<font%1>%2</font>', iif(overdue_cnt > 0,' color = "#FF0000"', ''), overdue_cnt);

  all_debt_sum = coalesce(all_debt_sum, 0.00);

  res_html = BLOB_APPEND(res_html, f_format(
    '<table width="100%"><col width="60%"><col width="40%"><th>%1</th><th>%2</th>', 'Категория', 'Значение'));
  
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'Общее количество активных кредитов'), active_cnt));
  
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'в том числе в М Булак'),
    mb_active));
  
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'Наличие активных кредитов с просрочкой'),
    has_active_debt));
  
  res_html = BLOB_APPEND(res_html, '<tr><td>' || f_lang(lang, 'Сумма текущей задолженности, руб.') || '</td><td><b>' || all_sum_all || '</b></td>');
  
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'Сумма общей текущей задолженности по активным кредитам, руб.'),
    f_numtostr(round(all_debt_sum, 2), 2)));
  
  -- Добавление информации о проданных кредитах
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'Количество проданных кредитов'),
    sell_cnt));
  
  res_html = BLOB_APPEND(res_html, f_format(row_fmt,
    f_lang(lang, 'Текущий баланс по активным проданным кредитам, руб.'),
    f_numtostr(round(sell_sum, 2), 2)));
  
   res_html = BLOB_APPEND(res_html,
      f_format('</table><a href="#top">%1</a><br>'));
   res_html = BLOB_APPEND(res_html, '</body>');

   suspend;
end
