CREATE PROCEDURE OKT_SIRENA_CONTACTS_LIST(
    PANT_ID ID_TYPE,
    C_ID ID_TYPE)
RETURNS (
    CC_ID INTEGER,
    CC_STATUS STR_100,
    CC_TYPE STR_100,
    CC_NAME STR_100,
    CC_CELL_NO STR_100,
    OLD_CONTACT INTEGER,
    IS_VERIFIED SMALLINT,
    CALL_QTY INTEGER,
    LAST_CALL_DATE DATE,
    LAST_CALL_PROJECT STR_100,
    LAST_CALL_RESULT STR_255,
    LAST_CALL_MEMO VARCHAR(1000),
    LAST_NEXT_CALL DATE,
    LAST_OPER_CALLED STR_100,
    CAN_CALL INTEGER,
    REASON STR_255,
    CLIENT_CELL_NO STR_100,
    TOTAL_RESULT BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    NEXT_CALL_CHECK_OK INTEGER,
    IGNORE_LIST_CHECK_OK INTEGER)
AS
declare variable k_id id_type;
declare variable counter integer;
declare variable pcon_called_date_tmp date;
declare variable project_tmp str_100;
declare variable call_res_tmp str_255;
declare variable res_tag_tmp integer;
declare variable pcon_memo_tmp varchar(1000);
declare variable pcon_next_call_tmp date;
declare variable oper_called_tmp str_100;
declare variable cell_main_id integer;
begin
  client_cell_no = f_cont_info('clients', c_id, 'cell_main', 'last');

  select pantk.pant_from_k_id
  from pantera_kontracts pantk
  where pantk.pant_id = :pant_id
  into :k_id;

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ContactInfo.Cell_Main'
  into :cell_main_id;

  for select cont.cc_id, cont.cc_status, iif(cont.cc_type = 0, 'РФ', 'Родина'), cont.cc_name, cont.cc_cell_no,
             cont.cc_is_old_contact, iif(cc_total_res = 1, 1, 0)
      from contacts cont
      where cont.cc_from_c_id = :c_id and
            cont.cc_total_res >= 0
      order by cc_is_old_contact
      into cc_id, cc_status, cc_type, cc_name, cc_cell_no, old_contact, is_verified
  do
  begin
    total_result = '';

    last_call_date = null;
    last_call_project = null;
    last_call_result = null;
    last_call_memo = null;
    last_next_call = null;
    last_oper_called = null;

    if ((trim(coalesce(cc_name, '')) <> '') and
        (trim(coalesce(cc_cell_no, '')) <> '')) then
    begin
      counter = 0;
      call_qty = 0;

      for select pantc.pcon_called_date, rd.rd_name as project, rd2.rd_name as call_res, rd2.rd_tag as res_tag,
                 pantc.pcon_memo, pantc.pcon_next_call, pantk.pant_oper_called
          from pantera_contacts pantc
          inner join pantera_kontracts pantk on pantk.pant_id = pantc.pcon_from_pan_id
          inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
          inner join refdata rd2 on rd2.rd_id = pantc.pcon_rd_res
          where pantk.pant_from_k_id = :k_id and
                pantc.pcon_from_cc_id = :cc_id
          order by pantc.pcon_id desc
          into :pcon_called_date_tmp, :project_tmp, :call_res_tmp, :res_tag_tmp, :pcon_memo_tmp, :pcon_next_call_tmp,
               :oper_called_tmp
      do
      begin
        reason = '';
        can_call = 0;

        if (counter = 0) then
        begin
          last_call_date = :pcon_called_date_tmp;
          last_call_project = :project_tmp;
          last_call_result = :call_res_tmp;
          last_call_memo = :pcon_memo_tmp;
          last_next_call = :pcon_next_call_tmp;
          last_oper_called = :oper_called_tmp;

          counter = counter + 1;
        end

        last_call_result = coalesce(last_call_result, '');
        last_call_memo = coalesce(last_call_memo, '');
        last_call_project = coalesce(last_call_project, '');
        last_oper_called = coalesce(last_oper_called, '');

        if (:res_tag_tmp > 0) then
        begin
          call_qty = call_qty + 1;
        end
      end

-- проверяю наступило ли время следующего звонка
      select can_call
      from hp_can_call_check(:k_id, 'Contacts', :cc_id, null)
      into :next_call_check_ok;

-- проверяю по игнор-листу
      select can_call
      from okt_can_call(null, 'Contacts', :cc_id, null, null, :cell_main_id, null)
      into :ignore_list_check_ok;

      if ((next_call_check_ok = 1) and
          (ignore_list_check_ok = 1)) then
        can_call = 1;

      if (last_call_result <> '') then
      begin
        total_result = BLOB_APPEND(total_result,'Поcледний звонок. Результат: ' || last_call_result || '; ');
        total_result = BLOB_APPEND(total_result, 'Примечание: ' || iif(last_call_memo = '', 'Отсутствует', last_call_memo) || '; ');
        total_result = BLOB_APPEND(total_result, 'Проект: ' || last_call_project || '; ');
        total_result = BLOB_APPEND(total_result, 'Оператор: ' || last_oper_called || '; ');
      end

      total_result = coalesce(total_result, '');

      suspend;
    end
  end
end
