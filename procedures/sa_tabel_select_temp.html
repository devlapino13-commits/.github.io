CREATE PROCEDURE SA_TABEL_SELECT_TEMP(
    SA_REGION_ID INTEGER,
    SA_YEAR INTEGER,
    SA_MONTH INTEGER)
RETURNS (
    SAT_FROM_C_ID DECIMAL(18,0),
    SAT_OFFICE_ID INTEGER,
    SAT_TOTAL_NORMA INTEGER,
    SAT_TOTAL_WORKED INTEGER,
    SAT_TOTAL_OTPUSK INTEGER,
    SAT_TOTAL_BOLNICH INTEGER)
AS
declare variable sat_total_calc integer;
declare variable sat_base_type integer;
declare variable tnorm integer;
begin
  select sum(scd_work_hours)
  from sa_calendays
  where scd_year_month = (:sa_year * 100 + :sa_month)
  into :tnorm;
  if (tnorm is null) then
    tnorm = 0;

  for select sat_from_c_id, sat_office_id, sum(iif(sat_from_sawt_id = 0, :tnorm, 0)), sum(sat_total_calc),
             sawt_base_type
      from sa_tabel, sa_worktype, region_list(:sa_region_id, 1, -1, 1)
      where sawt_id = sat_from_sawt_id and
            sat_is_ready = 1 and
            sat_year_month = (:sa_year * 100 + :sa_month) and
            (sat_office_id = id_list)
      group by sat_from_c_id, sat_office_id, sawt_base_type
      order by sat_from_c_id, sat_office_id, sawt_base_type
      into :sat_from_c_id, :sat_office_id, :sat_total_norma, :sat_total_calc, :sat_base_type

  do
  begin
    sat_total_worked = 0;
    sat_total_otpusk = 0;
    sat_total_bolnich = 0;
    if (sat_base_type = 0) then
      sat_total_worked = sat_total_calc;
    if (sat_base_type = 4) then
      sat_total_otpusk = sat_total_calc;
    if (sat_base_type = 5) then
      sat_total_bolnich = sat_total_calc;
    suspend;
  end
end
