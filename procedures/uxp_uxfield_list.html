CREATE PROCEDURE UXP$UXFIELD_LIST(
    UX_UUID D_UUID,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    UF_UUID D_UUID,
    UF_OWNER D_UUID,
    UF_IS_PARAM D_BOOL,
    UF_PK_FIELD D_SMALLINT,
    UF_POSITION D_INTEGER,
    UF_DOMAIN D_STR_CODE,
    UF_DATA_TYPE D_STR_CODE,
    UF_TYPE D_SMALLINT,
    UF_NOT_NULL D_BOOL,
    UF_DEFAULT D_STR_100,
    UF_SECURED D_BOOL,
    UF_FIELD_NAME D_STR_100,
    UF_NAME D_STR_100,
    UF_EXPRESSION D_STR_4096,
    UF_STATE D_SMALLINT,
    RES_FIELD_KIND D_STR_10,
    RES_IS_FK D_BOOL,
    RES_FK_TABLES D_STR_255,
    RES_IS_VIRTUAL D_BOOL,
    UF_EDI_UUID D_UUID,
    RES_EDI_UUID_CODE D_STR_CODE,
    RES_EDI_KIND D_SMALLINT,
    RES_CATEGORY D_STR_CODE,
    RES_EXPR_KINDS D_SMALLINT)
AS
declare variable source_name d_str_code;
declare variable tmp_text d_str_code;
declare variable uix_virtual d_bool;
declare variable uf_readonly d_boolean;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  source_name = uxf$code(ux_uuid, 'code');
  uf_owner = ux_uuid;

  for select uf_uuid,
             uf_is_param,
             uf_pk_field,
             uf_position,
             uf_domain,
             uf_data_type,
             uf_type,
             uf_not_null,
             uf_readonly,
             uf_default,
             uf_secured,
             uf_field_name,
             uf_name,
             uf_edi_uuid,
             uf_state
      from ux$field
      where uf_owner = :ux_uuid
      order by uf_is_param desc, uf_position
      into :uf_uuid,
           :uf_is_param,
           :uf_pk_field,
           :uf_position,
           :uf_domain,
           :uf_data_type,
           :uf_type,
           :uf_not_null,
           :uf_readonly,
           :uf_default,
           :uf_secured,
           :uf_field_name,
           :uf_name,
           :uf_edi_uuid,
           :uf_state
  do
  begin
    res_is_fk = 0;
    res_fk_tables = '';

    if (uf_is_param = 1) then
      res_category = ' Параметры';
    else
      res_category = ' Поля';

    uf_expression = '';
    select exp_expression_all, exp_kinds
    from uxp$uxexpression_text(:uf_uuid)
    into :uf_expression, res_expr_kinds;

    res_is_virtual = iif(uf_type = 0, 0, 1);

    for select uix_fk_source, uix_virtual
    from ux$index
    where uix_source = :source_name and
          uix_fk = 1 and
          uix_fields || ';' containing :uf_field_name || ';'
    into tmp_text, uix_virtual
    do
    begin
      if (res_fk_tables <> '') then
        res_fk_tables = res_fk_tables || ', ';

      if (uix_virtual = 1) then
        tmp_text = tmp_text || ' [V]';

      res_fk_tables = res_fk_tables || tmp_text;
      res_is_fk = 1;
    end

    res_field_kind = '';
    if (res_is_fk = 0) then
    begin
      if (uf_pk_field <> 0) then
        res_field_kind = 'Pk' || uf_pk_field;
    end else
    begin
      if (uf_pk_field = 0) then
        res_field_kind = 'Fk';
      else
        res_field_kind = 'FP' || uf_pk_field;
    end

    if (uf_type = 1) then
    begin
      if (res_field_kind = '') then
        res_field_kind = 'V';
      else
        res_field_kind = 'V ' || res_field_kind;
    end else
    if (uf_type = -1) then
    begin
      if (res_field_kind = '') then
        res_field_kind = 'X';
      else
        res_field_kind = 'X ' || res_field_kind;
    end

    if (uf_readonly) then
    begin
      if (res_field_kind = '') then
        res_field_kind = 'RO';
      else
        res_field_kind = res_field_kind || ' ' || 'RO';
    end

    res_edi_uuid_code = '';
    res_edi_kind = 0;
    if (uf_edi_uuid is not null) then
    begin
      res_edi_uuid_code = uuid_to_char(uf_edi_uuid);
      res_edi_kind = 1;
    end

    suspend;
  end
end
