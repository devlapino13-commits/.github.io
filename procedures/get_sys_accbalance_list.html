CREATE PROCEDURE GET_SYS_ACCBALANCE_LIST(
    IS_FULL D_SMALLINT = 0)
RETURNS (
    SAB_ID D_ID,
    SAB_NAME D_STR_255)
AS
begin
  sab_id = -1;
  sab_name = '';
  suspend;

  for select sab_id, sab_full_code || '  -  ' || sab_name
      from sys_accbalance
      where :is_full = 1 or (:is_full = 0 and sab_create_kind <> 0)
      order by 2
      into :sab_id, sab_name
  do
    suspend;
end
