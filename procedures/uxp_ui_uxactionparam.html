CREATE PROCEDURE UXP$UI_UXACTIONPARAM(
    UA_UUID D_UUID,
    PARENT_UA_UUID D_UUID,
    VER_KIND D_STR_CODE = null,
    UA_UUID_CODE D_STR_CODE = null,
    PARENT_UA_UUID_CODE D_STR_CODE = null)
RETURNS (
    UAP_FR_UUID D_UUID,
    UAP_FIELD D_STR_CODE,
    UAP_IS_PARAM D_BOOL,
    UAP_SOURCE_FR_UUID D_UUID,
    UAP_SOURCE_FIELD D_STR_CODE,
    UAP_SOURCE_IS_PARAM D_BOOL,
    UAP_VALUE D_STR_255,
    UAP_IS_PARENT_UA D_BOOL,
    UAP_LOOKUP_CODE D_STR_100,
    UAP_PUBLIC D_BOOL,
    UAP_EXEC_PARAM_ID D_SMALLINT)
AS
declare variable uap_default d_str_255;
declare variable uap_use_default d_bool;
declare variable ufm_uuid d_uuid;
declare variable param_kind d_smallint;
declare variable uxa_uuid d_uuid;
declare variable fr_uuid d_uuid;
declare variable ufp_uuid d_uuid;
declare variable uxa_owner d_uuid;
declare variable ua_uxa_fcode d_str_100;
declare variable uf_edi_uuid d_uuid;
declare variable ua_owner_kind d_str_code;
declare variable uf_field_name d_str_code;
begin
  if (coalesce(ua_uuid, '') = '') then
    ua_uuid = uxf$uniq(ua_uuid_code);

  if (uuid_to_char(parent_ua_uuid) = '00000000-0000-0000-0000-000000000000') then
    parent_ua_uuid = null;
  else
  if (coalesce(parent_ua_uuid, '') = '' and coalesce(parent_ua_uuid_code, '') <> '') then
    parent_ua_uuid = uxf$uniq(parent_ua_uuid_code);

  select uxa_uuid, ua_uxa_owner, ua_owner_kind
  from uxp$uxalias_info(:ua_uuid)
  into uxa_uuid, :uxa_owner, :ua_owner_kind;

  ufm_uuid = uxf$ver(uxa_owner, ver_kind);

  for select ufr_uuid
  from ux$frame
  where ufr_owner = :ufm_uuid
  into fr_uuid
  do
  begin
    uap_fr_uuid = fr_uuid;

    for select ufp_uuid, ufp_public,
               iif(uf_is_param = 1 or
                  (ufp_parent is null and (uf_pk_field <> 0) and us_is_table = 1), 1, 0),
               uf_edi_uuid,
               uf_field_name
    from ux$frparam, ux$field, ux$source
    where ufp_ufr_uuid = :fr_uuid and
          ufp_uf_uuid = uf_uuid and
          uf_owner = us_uuid
    into ufp_uuid, uap_public, param_kind, uf_edi_uuid, uf_field_name
    do
    begin
      uap_is_parent_ua = 0;

      for select res_fr_uuid, res_field_name, res_is_param, res_def_value,
                 res_source_fr_uuid, res_source_field,
                 res_source_is_param, res_lookup_code, res_exec_param_id
      from uxp$uxfrparam_value(:ufp_uuid, :uxa_uuid, :ua_uuid, 1)
      into :uap_fr_uuid, :uap_field, :uap_is_param, :uap_value,
           :uap_source_fr_uuid, :uap_source_field,
           :uap_source_is_param, :uap_lookup_code, uap_exec_param_id
      do
      begin
        if (parent_ua_uuid is not null and uap_source_field <> '' and ua_owner_kind <> 'uxextform') then
        begin
          for select ua_uxa_fcode, ownp.ufp_ufr_uuid, ownf.uf_field_name, uap_use_default,
                     trim(coalesce(uap_default, '')), ownf.uf_is_param, 1
          from ux$actionparam, ux$action, ux$frparam ownp, ux$field ownf
          where uap_ua_uuid = :parent_ua_uuid and
                uap_param_name = :uap_source_field and
                uap_is_output = 0 and
                uap_ua_uuid = ua_uuid and
                uap_owner_ufp_uuid = ownp.ufp_uuid and
                ownp.ufp_uf_uuid = ownf.uf_uuid
          into ua_uxa_fcode, uap_source_fr_uuid, uap_source_field, uap_use_default,
               uap_default, uap_source_is_param, uap_is_parent_ua
          do
          begin
            select iif(uf_pk_field <> 0 or uf_is_param = 1, 1, 0)
            from uxp$uxactionparam_alias_ufp(:ua_uxa_fcode, :uap_field)
            where iif(uf_pk_field <> 0 or uf_is_param = 1, 1, 0) = :param_kind
            into uap_is_param;
          end

          if (uap_source_field is null) then
            uap_source_field = '';
  
          if (uap_use_default = 1 and uap_default = '') then
            uap_use_default = 0;
        end
  
        if (uap_source_field <> '' or uap_value <> '' or uap_exec_param_id > 0) then
          suspend;
      end
    end
  end
end
