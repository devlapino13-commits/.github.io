CREATE PROCEDURE HP_PRE_SUD_CLIENTS_KONT(
    REP_C_ID ID_TYPE)
RETURNS (
    K_ID ID_TYPE,
    K_KONTRACT_NO STR_20,
    DISB_DATE D_DATE,
    DELINQ_DAYS D_INTEGER,
    ZAD_ALL CURR_TYPE,
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    ZAD_FINE CURR_TYPE,
    ZAD_EXTRA CURR_TYPE)
AS
declare variable oek_rd_id d_integer;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  for select kc_from_k_id
      from knclient
      where kc_from_c_id = :rep_c_id and
            kc_amount_approved > 0
      into k_id
  do
  begin
    for select k_id, max(delinq_days), sum(zad_os), sum(zad_perc), sum(zad_fine), sum(zad_extra)
        from (select oks_from_k_id as k_id, current_date - oks_delinq_date as delinq_days, 0 zad_os, 0 zad_perc,
                     0 zad_fine, 0 zad_extra
              from oksummary
              where current_date between oks_date_begin and oks_date_end and
                    current_date - oks_delinq_date > 0 and
                    oks_close_state = 0 and
                    oks_from_k_id = :k_id
              union all
              select oko_from_k_id, 0 delinq_days, oko_os + oko_sh_os as zad_os, oko_perc + oko_sh_perc as zad_perc,
                     oko_other + oko_sh_other as zad_fine, 0 zad_extra
              from operkroff
              where oko_ol_is_active = 1 and
                    oko_ol_date <= current_date and
                    oko_from_k_id = :k_id
              union all
              select ox_object_id as k_id, 0 delinq_days, 0 zad_os, 0 zad_perc, 0 zad_fine,
                     ox_amount_plus - ox_amount_minus - ox_amount_writeoff as zad_extra
              from operextra
              where ox_ol_is_active = 1 and
                    ox_from_rd_id = :oek_rd_id and
                    ox_object_kind = f_objkind('kontract', 'id') and
                    ox_ol_date <= current_date and
                    ox_object_kind = 9 and
                    ox_object_id = :k_id)
        group by k_id
        having sum(zad_os) > 0 or sum(zad_perc) > 0 or sum(zad_fine) > 0 or sum(zad_extra) > 0 or max(delinq_days) > 0
        into k_id, delinq_days, zad_os, zad_perc, zad_fine, zad_extra
    do
    begin
      select k_kontract_no, k_disb_date
      from kontract
      where k_id = :k_id
      into k_kontract_no, disb_date;

      if (delinq_days > 0) then
      begin
        select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine)
        from aarep_make_zad_list(:k_id, current_date, 1)
        into zad_os, zad_perc, zad_fine;
      end
      else
        select current_date - oks_delinq_date
        from oksummary
        where oks_from_k_id = :k_id and
              oks_delinq_date < '31.12.9999'
        order by oks_date_begin desc rows 1
        into delinq_days;

      zad_all = zad_os + zad_perc + zad_fine + zad_extra;
      suspend;
    end
  end
end
