CREATE PROCEDURE AAREP_ACTIVE_NOT_TRANSFER_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_CURRENCY VARCHAR(3))
RETURNS (
    K_ID DECIMAL(18,0),
    K_CURRENCY VARCHAR(3),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_PERCENT DECIMAL(18,7),
    K_KOLICH_REPAY INTEGER,
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_PERCENT_CATEGORY DECIMAL(18,4),
    KF_OFFICER_NAME VARCHAR(100),
    KAL_NEW_KF_NAME VARCHAR(100),
    KAL_DATE DATE,
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    PR_CODE VARCHAR(100),
    PR_NAME VARCHAR(50),
    PR_LOAN_TYPE INTEGER,
    REP_ACTIVE_BALANCE DECIMAL(18,4),
    PR_LOAN_TYPE_STR VARCHAR(10),
    IS_TRANSFER INTEGER,
    PS_DATE_BEGIN DATE,
    C_STAGE INTEGER,
    C_STAGE_YEAR INTEGER,
    CR_RATE DECIMAL(18,4))
AS
declare variable kf_officer_id decimal(18,0);
declare variable kal_from_kf_id decimal(18,0);
declare variable kal_new_kf_officer decimal(18,0);
declare variable kal_new_reg_id integer;
declare variable parent_id integer;
declare variable reg_prior integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  reg_prior = 0;
  for select k_id, k_currency, k_kontract_no, k_disb_date, k_percent, k_kolich_repay, k_circle, k_circle_internal,
             k_amount_approved * k_index_curr_rate, k_kolich_client_approved, k_percent_category, c_id, c_name_fio,
             kal_from_kf_id, kal_new_kf_officer, kal_new_kf_name, kal_new_reg_id, kal_date, reg_code || ' ' || reg_name,
             pr_code, pr_name, pr_loan_type, region.parent,
             case
               when cr_rate < k_index_curr_rate then k_index_curr_rate
               else cr_rate
             end,
             sum(
             case
               when :rep_currency = '000' then (case
                                                  when cr_rate < k_index_curr_rate then ok_os * k_index_curr_rate
                                                  else ok_os * cr_rate
                                                end)
               when :rep_currency = '---' then ok_os
               when :rep_currency = k_index_curr_code then ok_os
             end)

      from operkred, operlist, kontract, opercurr, kontstate, knaktlist, region, region_list(:rep_region_id, 1, 3, 1), clients, product
      where ok_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            ok_from_k_id = k_id and
            k_index_curr_code = cr_from_cu_id and
            ol_date <= :rep_date_end and
            :rep_date_end between cr_date and cr_date_end and
            ok_type not in (1, 2, 3, 9, 10) and
            kontstate.ks_ol_is_delete = 0 and
            ks_from_k_id = k_id and
            cast(ks_date_begin as date) <= :rep_date_end and
            ks_date_end >= :rep_date_end and
            ks_state = 1 and
            kal_from_kf_id = k_from_kf_id and
            :rep_date_end between kal_date and kal_date_end and
            ((:rep_currency = '---') or
            (:rep_currency = '000') or
            (:rep_currency = k_index_curr_code)) and
            knaktlist.is_delete = 0 and
            (id_list = reg_unicode) and
            reg_unicode = kal_new_reg_id and
            c_id = kal_new_kf_officer and
            ok_os <> 0 and
            k_from_pr_id = pr_id and
            ok_from_c_id = 0
            /*and k_disb_date>=kal_date*/
            /*and k_disb_date<kal_date*/ /*Подключает не чистый портфель*/

      group by k_id, k_currency, k_kontract_no, k_disb_date, k_percent, k_kolich_repay, k_circle, k_circle_internal, k_amount_approved, k_index_curr_rate, k_kolich_client_approved, k_percent_category, c_id, c_name_fio, kal_from_kf_id, kal_new_kf_officer, kal_new_kf_name, kal_new_reg_id, kal_date, reg_code, reg_name, pr_code, pr_name, pr_loan_type, cr_rate, region.parent

      having sum(ok_os) <> 0

      into :k_id, :k_currency, :k_kontract_no, :k_disb_date, :k_percent, :k_kolich_repay, :k_circle, :k_circle_internal,
           :k_amount_approved, :k_kolich_client_approved, :k_percent_category, :kf_officer_id, :kf_officer_name,
           :kal_from_kf_id, :kal_new_kf_officer, :kal_new_kf_name, :kal_new_reg_id, :kal_date, :reg_name, :pr_code,
           :pr_name, :pr_loan_type, :cr_rate, :rep_active_balance, :parent_id

  do
  begin
    if (parent_id <> reg_prior) then
    begin
      select reg_name
      from region
      where reg_unicode = :parent_id
      into :parent_name;
      reg_prior = parent_id;
    end

    if (pr_loan_type = 0) then
      pr_loan_type_str = 'Группа';
    if (pr_loan_type = 1) then
      pr_loan_type_str = 'Индив.';

    /*  if (:k_disb_date>=:kal_date) then is_transfer=0; else is_transfer=1;*/

    if (exists(select kal1.kal_from_kf_id
               from knaktlist kal1
               where kal1.kal_from_kf_id = :kal_from_kf_id and
                     :k_disb_date between kal1.kal_date and kal1.kal_date_end and
                     kal1.kal_new_kf_officer = :kal_new_kf_officer
                     /*  and kal1.kal_new_reg_id=:kal_new_reg_id*/
                     and
                     kal1.is_delete = 0)) then
      is_transfer = 0;
    else
      is_transfer = 1;

    select ss1.ps_date_begin, ss1.c_stage, ss1.c_stage_year
    from aarep_staj_ks(:kf_officer_id, :rep_date_end) ss1
    into :ps_date_begin, :c_stage, :c_stage_year;

    suspend;

  end
end
