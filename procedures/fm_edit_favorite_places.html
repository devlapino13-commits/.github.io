CREATE PROCEDURE FM_EDIT_FAVORITE_PLACES(
    UN_ID DECIMAL(18,0),
    PL_ID DECIMAL(18,0),
    ACTION_KIND VARCHAR(100))
AS
begin
  if (action_kind = 'add') then
  begin
    if (exists(select *
               from userplacestat
               where ups_from_pl_id = :pl_id and
                     ups_from_un_id = :un_id)) then
      update userplacestat
      set ups_access_kolich = ups_access_kolich + 1000000
      where ups_from_pl_id = :pl_id and
            ups_from_un_id = :un_id and
            ups_access_kolich < 1000000;
    else
      insert into userplacestat (ups_from_un_id, ups_from_pl_id, ups_access_kolich)
      values (:un_id, :pl_id, 1000000);
  end

  if (action_kind = 'remove') then
  begin
    update userplacestat
    set ups_access_kolich = ups_access_kolich - 1000000
    where ups_from_pl_id = :pl_id and
          ups_from_un_id = :un_id and
          ups_access_kolich >= 1000000;

    delete from userplacestat
    where ups_from_pl_id = :pl_id and
          ups_from_un_id = :un_id and
          ups_access_kolich = 0;
  end

  if (action_kind = 'clear_all') then
  begin
    delete from userplacestat
    where ups_from_un_id = :un_id;
  end
end
