CREATE PROCEDURE HP_SHOW_CLIENTUNIQ_ACC_NO(
    C_ID DECIMAL(18,0))
RETURNS (
    UNIQ_BIRTH_DATE DATE,
    UNIQ_ACC INTEGER,
    UNIQ_ACC_STR VARCHAR(10))
AS
declare variable c_ext_id integer;
declare variable total integer;
declare variable db_inc integer;
begin
-- Входящий параметр: ID клиента.
-- Задача процедуры вычислить окончательный номер лицевого счета которого будем сообщать клиенту и принимать для операций
-- Если клиент не верифицирован, то порядковый номер счета минусовой и результат данной процедуры -1.

  uniq_acc_str = 'XXX';
  uniq_acc = -1;
  select cunq_birth_date, cunq_ext_id
  from clientuniq
  where cunq_from_c_id = :c_id and
        cunq_ext_id > 0
  into uniq_birth_date, c_ext_id;
  if (c_ext_id is not null) then
  begin

    db_inc = f_get_db_id();
    db_inc = (case db_inc
                when 41701 then 1
                when 81001 then 2
                when 94401 then 3
              end);

-- Формула расчета контрольного разряда: [День+Месяц+Год+(Порядковый номер * 97)] остаток от деления на 10 + КОД БД;
    total = extract(day from uniq_birth_date) + extract(month from uniq_birth_date) + extract(year from uniq_birth_date) + c_ext_id * 97 + db_inc;

    total = mod(total, 10);
    uniq_acc = c_ext_id * 10 + total;
    uniq_acc_str = uniq_acc;
  end
  suspend;
end
