CREATE PROCEDURE HP_EXTINFO_TO_SEND_FULL(
    PROJECT_CODE D_STR_CODE)
RETURNS (
    ID D_ID,
    JSON_DATA D_STR_8100,
    KIB_NAME D_STR_CODE)
AS
declare variable eic_id d_id;
declare variable delay_days d_integer;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable birthd d_str_10;
declare variable passn d_str_code;
declare variable socc d_str_code;
declare variable addr_count d_integer;
declare variable object_kind d_integer;
declare variable object_id d_id;
declare variable db_id d_id;
declare variable rows_to_suspend d_integer;
begin
  -- процедура готовит список запросов для получения данных из внешних источников,
  -- например БКИ и т.д.

  delay_days = 3;
  db_id = f_get_db_id();

  rows_to_suspend = 500;
  if (project_code = 'kontur') then
    rows_to_suspend = 10000;

  if (coalesce(project_code, '') <> '') then
  begin
    select eic_id
    from extinfocfg
    where eic_code = :project_code
    into eic_id;

    if (eic_id is null) then
      exception e_default 'Project code not found: ' || coalesce(project_code, '<NULL>');
  end

  for select eil_id, eil_from_c_id, eic_code, eil_object_kind, eil_object_id
      from extinfolist, extinfocfg
      where eil_from_eic_id = eic_id and
            eil_is_json = 1 and
            eil_status in (-1, 0) and -- новые запросы и запросы с временной ошибкой
            eil_date >= current_date - :delay_days and
            (:eic_id is null or
            eil_from_eic_id = :eic_id)
      order by iif (:db_id = 81001, -eil_id, eil_id)
      into id, c_id, kib_name, object_kind, object_id
  do
  begin
    if (kib_name = 'ishenim') then
    begin
      select cl.c_name_fio, f_datetostr(cl.c_birth_date), coalesce(cl.c_inn, ''), coalesce(cl.c_pass_no, '')
      from clients cl
      where cl.c_id = :c_id
      into :c_name_fio, :birthd, :socc, :passn;

      c_name_fio = replace(c_name_fio, 'ң', 'н');
      c_name_fio = replace(c_name_fio, '�©', 'о');
      c_name_fio = replace(c_name_fio, 'ү', 'у');

      json_data = '{' || f_json_make_pair('fullName', :c_name_fio, 'str') || ', ' || f_json_make_pair('internalPassport', :passn, 'str') || ', ' || f_json_make_pair('birthdate', :birthd, 'str') || ', ' || f_json_make_pair('socCode', :socc, 'str') || '}';
      suspend;
    end

    if (kib_name = 'moneta_doc') then
    begin
      json_data = (select json_data from hp_extinfo_to_send_moneta_doc(:object_kind, :object_id));
      suspend;
    end
    if (kib_name = 'DigiDebt_court') then
    begin
      json_data = (select json_data from hp_extinfo_to_send_digidebt_c(:object_kind, :object_id, :c_id));
      suspend;
    end

    if (kib_name = 'inn_nalog') then
    begin
      json_data = (select json_value from hp_extinfo_to_send_inn_nalog(:c_id));
      suspend;
    end

    if (kib_name = 'equifax_40') then
    begin
      json_data = (select json_value from hp_extinfo_to_send_equifax_40(:c_id, :object_kind, :object_id));
      suspend;
    end

    if (kib_name = 'nbki_40') then
    begin
      json_data = (select json_value from hp_extinfo_to_send_nbki_40(:c_id, :object_kind, :object_id));
      suspend;
    end

    if (kib_name = 'bki_ssp_40') then
    begin
      json_data = (select json_value from hp_extinfo_to_send_ssp_40(:c_id, :object_kind, :object_id));
      suspend;
    end

    if (kib_name = 'kontur') then
    begin
      json_data = (select json_value from hp_extinfo_to_send_kontur(:c_id, :object_kind, :object_id));
      suspend;
    end

    if (kib_name = 'passport_status') then
    begin
      json_data = (select json_value from hp_extinfo_to_send_tunduk_pass(:c_id));
      suspend;
    end

    if (kib_name = 'marital_status') then
    begin
      json_data = (select json_value from hp_extinfo_to_send_tunduk_marit(:c_id));
      suspend;
    end

    if (kib_name = 'address_asb') then
    begin
      json_data = (select json_value from hp_extinfo_to_send_tunduk_addr(:c_id));
      suspend;
    end

    if (kib_name in ('equifax', 'nbki', 'fssp', 'tunduk', 'fin', 'salary', 'pension', 'INN_entity', 'family_members', 'address_fact')) then
    begin
      addr_count = 0;
      select count(*)
      from addrclient
      where adc_from_c_id = :c_id and
            adc_address_type in (1, 2) and
            coalesce(adc_from_adl_id, 0) <> 0
      into addr_count;

      if (addr_count = 2) then
      begin
        if (kib_name = 'equifax') then
        begin
          json_data = (select json_value
                       from hp_extinfo_to_send_equifax(:c_id, :object_kind, :object_id));

          if ((select result
               from hp_extinfo_check_for_complete(:c_id, :object_kind, :object_id)) = 1) then
            suspend;
        end
        if (kib_name = 'nbki') then
        begin
          json_data = (select json_value
                       from hp_extinfo_to_send_nbki(:c_id, :object_kind, :object_id));
          suspend;
        end
      end

      if (kib_name = 'fssp') then
      begin
        json_data = (select json_value
                     from hp_extinfo_to_send_fssp(:c_id));
        suspend;
      end

      if (kib_name = 'tunduk') then
      begin
        json_data = (select json_value
                     from hp_extinfo_to_send_tunduk(:c_id, :object_kind, :object_id));
        suspend;
      end

      if (kib_name in ('fin', 'salary')) then
      begin
        if ((select coalesce(cl.c_pincode, '')
             from clients cl
             where cl.c_id = :c_id) <> '') then
        begin
          json_data = (select json_value
                       from hp_extinfo_to_send_chek_fin(:c_id));
          suspend;
        end
      end

      if (kib_name = 'pension') then
      begin
        if ((select coalesce(cl.c_pincode, '')
             from clients cl
             where cl.c_id = :c_id) <> '') then
        begin
          json_data = (select json_value
                       from hp_extinfo_to_send_chek_fin(:c_id));
          suspend;
        end
      end

      if (kib_name in ('INN_entity', 'family_members', 'address_fact')) then
      begin
        if ((select coalesce(cl.c_inn, '')
             from clients cl
             where cl.c_id = :c_id) <> '') then
        begin
          json_data = (select json_value
                       from hp_extinfo_to_send_chek_inn(:c_id));
          suspend;
        end
      end
    end
  end

  -- аннулируем запрос в связи с превышением времени ожилания (3 дня)
  begin
    update extinfolist
    set eil_status = -3
    where eil_is_json = 1 and
          eil_status in (-1, 0, 1) and
          eil_date < current_date - :delay_days;
    when any do
    -- глушим ошибки если были блокировки
    begin
    end
  end
  --  when any do
  --    exception e_default 'Check client :' || c_id;
end
