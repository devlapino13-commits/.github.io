CREATE PROCEDURE FINDKONTRACT_CHECK(
    CHECK_KIND D_STR_20,
    CHECK_TXT D_STR_100)
AS
declare variable str d_str_100;
declare variable i d_integer;
declare variable counter d_integer;
declare variable need_qty d_integer;
declare variable obj_name d_str_20;
begin
    check_txt = coalesce(check_txt, '');
    check_txt = trim(check_txt);

    if (check_txt = '') then
        exit;

    if (check_kind not in ('folder', 'loan')) then
        exit;

    if (check_kind = 'folder') then -- '1-4-33-524'
    begin
        obj_name = '№ группы';
        need_qty = 2;
    end
    if (check_kind = 'loan') then  -- 0104-1882506-01
    begin
        obj_name = '№ договора';
        need_qty = 2;

        if (character_length(check_txt) = 18) then -- если поиск по id контракта
            exit;
    end

    str = check_txt;
    str = replace(str, '-', '0');
    replace(str, '/', '0');

    if (str not similar to '[-]?[[:DIGIT:]]+') then
        exception e_message 'Недопустимые символы в '||obj_name;

    str = check_txt;

    counter = 0;
    i = -1;

    while (i <> 0) do
    begin
        i = position('-' in str);
        if (i <> 0) then
        begin
            counter = counter + 1;
            str = substring(str from i+1);
        end
    end

    if (counter < need_qty) then
        exception e_message 'Должно быть указано хотя бы '||need_qty||' знака "-" в '||obj_name;
end
