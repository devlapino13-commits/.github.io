CREATE PROCEDURE AAREP_SMS_BIRTHDAY(
    REP_DATE DATE)
RETURNS (
    PHONE VARCHAR(20),
    CLIENT_ID DECIMAL(18,0),
    C_ALTERNAT_FIO VARCHAR(100))
AS
declare variable rep_month integer;
declare variable rep_day integer;
declare variable no_leap_year integer;
declare variable db_id d_integer;
begin
  rep_month = extract(month from :rep_date);
  rep_day = extract(day from :rep_date);
  db_id = f_get_db_id();

  no_leap_year = 0;
  if (rep_month * 100 + rep_day = 301 and
      extract(month from dateadd(day, -1, :rep_date)) * 100 + extract(day from dateadd(day, -1, :rep_date)) = 228) then
    no_leap_year = 1; -- не високосный год

  for select c_id,
             (select cont_value
              from hp_contdata_get_cell_main(0, c_id)), c_alternat_fio
      from clients
      where c_birth_date is not null and
            (extract(month from c_birth_date) * 100 + extract(day from c_birth_date) = :rep_month * 100 + :rep_day
              or (:no_leap_year = 1
                  and extract(month from c_birth_date) * 100 + extract(day from c_birth_date) = 229))
      into :client_id, :phone, :c_alternat_fio
  do
  begin
    if (phone <> '') then
    begin
      if (exists(select *
                 from knclient, kontstate
                 where kc_from_c_id = :client_id and
                       kc_amount_approved > 0 and
                       ks_from_k_id = kc_from_k_id and
                       ks_ol_is_delete = 0 and
                       ks_date_begin <= :rep_date and
                       ks_date_end >= :rep_date and
                       ks_state = 1)) then
      begin
        if (db_id <> 81001 or f_clientcom_allowed(client_id, 1, 1, 1) = 1) then
        suspend;
      end
    end
  end
end
