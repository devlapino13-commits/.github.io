CREATE PROCEDURE HP_MONETA_DIFF_KONT_PAY(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    CARD_NUMBER STR_100,
    K_KONTRACT_NO STR_100,
    ESM_DATE DATE,
    AMOUNT CURR_TYPE,
    SVZ_FIO STR_100,
    K_AMOUNT_APPROVED CURR_TYPE,
    K_DISB_DATE DATE,
    KF_NAME VARCHAR(100),
    KC_AMOUNT_APPROVED CURR_TYPE,
    CL_FIO VARCHAR(100),
    KC_STATUS_TXT VARCHAR(10),
    C_PLASTCARD_NO VARCHAR(50),
    YUR_HOME_ADDRESS_NAME VARCHAR(255),
    WORK_ADDRESS_NAME VARCHAR(255))
AS
declare variable kf_id id_type;
declare variable k_id_old id_type;
declare variable reg_id_old integer;
declare variable kf_officer_old id_type;
declare variable k_id id_type;
declare variable reg_id integer;
declare variable kf_officer id_type;
declare variable kc_from_c_id id_type;
declare variable adc_address_str varchar(255);
declare variable adc_address_type_str varchar(20);
declare variable tmp_str d_str_100;
begin
  kf_officer_old = 0;
  reg_id_old = 0;
  k_id_old = 0;

  for select k_id, card_number, amount, esm_date, reg_id, kf_officer
      from hp_moneta_diff_kont_pay_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, 3)
      order by card_number, k_id
      into :k_id, :card_number, :amount, :esm_date, :reg_id, :kf_officer
  do
  begin
    if (reg_id <> reg_id_old) then
    begin
      select reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :reg_name;

      reg_id_old = reg_id;
    end

    if (kf_officer <> kf_officer_old) then
    begin
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :kf_officer
      into :svz_fio;

      kf_officer_old = kf_officer;
    end

    if (k_id <> k_id_old) then
    begin
      select kont.k_kontract_no, k_amount_approved, k_disb_date, k_from_kf_id
      from kontract kont
      where kont.k_id = :k_id
      into :k_kontract_no, k_amount_approved, k_disb_date, kf_id;

      kf_name = '';
      select kal.kal_new_kf_name
      from knaktlist kal
      where kal.kal_from_kf_id = :kf_id and
            :rep_date_end between kal_date and kal_date_end and
            kal.is_delete = 0
      into kf_name;

      kc_status_txt = '';
      for select kc_from_c_id, kc_amount_approved,
                 case kc_status
                   when 0 then 'Участник'
                   when 1 then 'Лидер'
                   when 2 then 'Казначей'
                 end
          from knclient kc
          where kc_from_k_id = :k_id and
                kc_amount_approved > 0
          into kc_from_c_id, kc_amount_approved, kc_status_txt
      do
      begin
        select c_name_fio
        from clients
        where c_id = :kc_from_c_id
        into cl_fio;

        c_plastcard_no = '';

        tmp_str = f_cont_info('clients', kc_from_c_id, 'cell_main', 'ok');
        if (tmp_str <> '') then c_plastcard_no = tmp_str;

        tmp_str = f_cont_info('clients', kc_from_c_id, 'cell_add', 'ok');
        if (tmp_str <> '' and tmp_str <> c_plastcard_no) then
        begin
          if (c_plastcard_no <> '')
          then c_plastcard_no = c_plastcard_no || ', ' || tmp_str;
          else c_plastcard_no = tmp_str;
        end

        adc_address_str = '';
        adc_address_type_str = '';
        yur_home_address_name = null;
        work_address_name = null;
        for select adc_address_str, adc_address_type_str
            from hp_addrclient(:kc_from_c_id)
            where adc_address_type_str = 'Юридически'
            into adc_address_str, adc_address_type_str
        do
        begin
          if (adc_address_type_str = 'Юридически') then
            yur_home_address_name = adc_address_str;
          else
          if (adc_address_type_str = 'Место работы/бизнеса') then
            work_address_name = adc_address_str;
        end

        if (coalesce(work_address_name, 'Не указан') = 'Не указан') then
          select ovs_value
          from hp_objectprops_get(0, :kc_from_c_id, -1, 'work_address_name', :rep_date_end)
          into work_address_name;

        work_address_name = replace(work_address_name, 'Значение:', '');

        suspend;
      end

      k_id_old = k_id;
    end
  end
end
