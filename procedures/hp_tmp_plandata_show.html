CREATE PROCEDURE HP_TMP_PLANDATA_SHOW(
    REP_REGION_ID INTEGER,
    REP_DATE DATE,
    REP_INDICATOR VARCHAR(255),
    REP_PERIOD_BEGIN INTEGER,
    REP_PERIOD_COUNT INTEGER,
    UN_ID DECIMAL(18,0),
    IS_MGR D_BOOL = 0)
RETURNS (
    PD_ID INTEGER,
    PD_FROM_PM_ID INTEGER,
    PD_FROM_REG_ID INTEGER,
    PD_FROM_C_ID DECIMAL(18,0),
    PD_NAME VARCHAR(255),
    PD_JOBNAME VARCHAR(255),
    PD_STAGE_BEGIN DATE,
    PD_DATE_END DATE,
    PD_PERIOD_BEGIN INTEGER,
    PD_DATA_0 DECIMAL(18,4),
    PD_DATA_1 DECIMAL(18,4),
    PD_DATA_2 DECIMAL(18,4),
    PD_DATA_3 DECIMAL(18,4),
    PD_DATA_4 DECIMAL(18,4),
    PD_DATA_5 DECIMAL(18,4),
    PD_DATA_6 DECIMAL(18,4),
    PD_DATA_7 DECIMAL(18,4),
    PD_DATA_8 DECIMAL(18,4),
    PD_DATA_9 DECIMAL(18,4),
    PD_DATA_10 DECIMAL(18,4),
    PD_DATA_11 DECIMAL(18,4),
    PD_DATA_12 DECIMAL(18,4),
    PD_PARENT_ID INTEGER,
    PD_IS_OFFICE INTEGER,
    PD_IS_READONLY INTEGER)
AS
begin
  delete from temp_plandata;

  execute procedure hp_plandata_fill(:rep_region_id, :rep_date, :rep_indicator, :rep_period_begin, :rep_period_count,
      :un_id, :is_mgr);

  for select pd_id, pd_from_pm_id, pd_from_reg_id, pd_from_c_id, pd_name, pd_jobname, pd_stage_begin, pd_date_end,
             pd_period_begin, pd_data_0, pd_data_1, pd_data_2, pd_data_3, pd_data_4, pd_data_5, pd_data_6, pd_data_7,
             pd_data_8, pd_data_9, pd_data_10, pd_data_11, pd_data_12, pd_parent_id, pd_is_office, pd_is_readonly
      from temp_plandata
      into :pd_id, :pd_from_pm_id, :pd_from_reg_id, :pd_from_c_id, :pd_name, :pd_jobname, :pd_stage_begin, :pd_date_end,
           :pd_period_begin, :pd_data_0, :pd_data_1, :pd_data_2, :pd_data_3, :pd_data_4, :pd_data_5, :pd_data_6,
           :pd_data_7, :pd_data_8, :pd_data_9, :pd_data_10, :pd_data_11, :pd_data_12, :pd_parent_id, :pd_is_office,
           :pd_is_readonly
  do
  begin
    suspend;
  end
end
