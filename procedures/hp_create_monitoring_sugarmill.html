CREATE PROCEDURE HP_CREATE_MONITORING_SUGARMILL
AS
declare variable kf_id d_id;
declare variable reg_id d_integer;
declare variable lo_id d_id;
declare variable k_id d_id;
declare variable k_disb_date d_date;
declare variable rep_date d_date;
declare variable rd_anketa d_integer;
declare variable datex d_date;
declare variable obj_kind1 d_integer;
declare variable obj_kind2 d_integer;
declare variable un_id d_id;
begin
  -- процедура должна запускаться каждый вечер и создавать задачи на заполнение анкеты по @мониторингу целевого использования кредита"
  -- Процедура должна запускаться после 21:30
  rd_anketa = f_get_rd_id('Anketa.TargetUseLoan');
  if (rd_anketa is null) then
    exit;

  rep_date = current_date;
  datex = '03.03.2022';
  obj_kind1 = f_objkind('Clients', 'id');
  obj_kind2 = f_objkind('KnClient', 'id');
  un_id = 1000000000001;

  for select k_id, k_disb_date, k_from_kf_id
      from kontstate
      inner join kontract on k_id = ks_from_k_id
      inner join product on pr_id = k_from_pr_id
      where ks_date_begin = :rep_date and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            pr_code containing '.sugarmill.'
      into :k_id, :k_disb_date, :kf_id
  do
  begin
    select kal_new_reg_id, kal_new_kf_officer
    from knaktlist
    where kal_from_kf_id = :kf_id and
          :rep_date between kal_date and kal_date_end and
          is_delete = 0
    into :reg_id, :lo_id;

    -- Создаем задачу завтрашним днем
    insert into actionlist (acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                            acl_respons_c_id, acl_from_un_id)
    select :rep_date + 1, :reg_id, iif(:k_disb_date >= :datex, :obj_kind2, :obj_kind1),
           iif(:k_disb_date >= :datex, kc_id, kc_from_c_id), :rd_anketa, 0, :lo_id, :un_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0;

    rep_date = current_date;
  end
end
