CREATE PROCEDURE HP_FT_ANKETA_LIST_CREATE(
    JSON_DATA D_STR_8100,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable ARR_COUNT D_INTEGER;
declare variable FAL_ID D_ID;
declare variable ARR_ITEM D_STR_LONG;
declare variable IDX D_INTEGER;
declare variable F_ID D_INTEGER;
declare variable F_VALUE D_BOOL;
declare variable F_COMMENT D_STR_255;
declare variable FAF_DEFAULT_VALUE D_BOOL;
declare variable FAF_NEED_COMMENT D_BOOL;
declare variable FAL_NEED_CHECK D_BOOL;
declare variable C_ID D_ID;
declare variable LANG D_STR_LANG;
declare variable FAF_NEED_CHECK D_BOOL;
declare variable FAF_FROM_FAK_ID D_INTEGER;
declare variable FAK_CODE D_STR_20;
declare variable FAK_ID D_SMALLINT;
declare variable NEED_SHOW_FT D_SMALLINT;
declare variable SELFEMPLOYED_START_DATE D_DATE;
declare variable EDL_ID D_ID;
declare variable EDL_STATUS D_SMALLINT;
declare variable EDL_OBJECT_ID D_ID;
declare variable IS_CORRECT boolean;
declare variable ERR_TXT D_STR_8100;
begin
  -- Процедура создает анкету-опросник, заполняет поля и создает электронный документ
  -- Вызывается из МП сотрудников

  res_is_error = 0;
  res_text = '';
  fal_need_check = 0;
  lang = f_get_lang_code(un_id, null);

  c_id = f_json_str_get_value(json_data, 'c_id', null);
  if (c_id is null) then
    exception e_default 'c_id not found';

  -- RFCRU-5644
  select
    is_correct,
    err_txt
  from hp_separate_fio_checkvalid(-:c_id)
  into
    :is_correct,
    :err_txt;
  if (not is_correct) then
    exception e_message :err_txt;

  --RFCRU-2543
  selfemployed_start_date = cast(f_get_const('selfemployed_start_date') as date);

  need_show_ft = 0;

  if (selfemployed_start_date <> '31.12.9999' and
      exists(select *
             from miniappclient mac
             inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id and
                    ma.ma_is_active = 1 and
                    ma.ma_date_end >= '31.12.9999'
             inner join product pr on pr.pr_id = ma.ma_from_pr_id and
                    pr.pr_code containing '.selfemployed.'
             where mac.mac_from_c_id = :c_id and
                   not exists(select *
                              from kontract kt
                              inner join kontstate ks on ks.ks_from_k_id = kt.k_id and
                                    ks.ks_state = 1 and
                                    ks.ks_ol_is_delete = 0 and
                                    ks.ks_date_begin <= current_date
                              where kt.k_from_ma_app_id = ma.ma_id)) and f_tmp_const('use_perstemp') <> 1) then
    need_show_ft = 1;

  if (f_meet_valid(c_id, 120) = 0 and (select is_oivo from hp_check_oivo_assignment(:c_id, :un_id)) <> 1) then
    res_text = f_lang(lang, 'Требуется верификация клиента по QR-коду.');
  else
  if (f_asp_valid(c_id, current_date) = 0) then
    res_text = f_lang(lang, 'Требуется подписание договора АСП с клиентом.');
  else
  if (f_ft_valid(c_id, current_date) = 1 and need_show_ft = 0) then
    res_text = f_lang(lang, 'Создание Анкеты-опросника не требуется');
  else
  begin
    insert into ft_anketa_list (fal_from_c_id, fal_create_un_id, fal_state, fal_from_fak_id)
    values (:c_id, :un_id, 0, 1)
    returning fal_id
    into :fal_id;

    arr_count = f_json_str_get_count(json_data, 'answers');

    idx = 0;
    fak_id = 1;

    while (idx < arr_count) do
    begin
      arr_item = f_json_str_get_value(json_data, 'answers[' || idx || ']', '');
      if (arr_item <> '') then
      begin
        f_id = f_json_str_get_value(arr_item, 'id', null);

        if (f_id is null) then
          exception e_default 'Question ID is null';

        f_value = f_json_str_get_value(arr_item, 'ans', null);

        if (f_value is null) then
          exception e_default 'Question ansver is null';

        f_comment = f_json_str_get_value(arr_item, 'text', '');

        faf_default_value = null;
        select faf_default_value, faf_need_comment, faf_need_check, faf_from_fak_id
        from ft_anketa_field
        where faf_id = :f_id
        into faf_default_value, faf_need_comment, faf_need_check, faf_from_fak_id;

        if (faf_from_fak_id > 1) then
            fak_id = faf_from_fak_id;

        if (row_count = 0) then
          exception e_default f_format('Question not found, Id: %1', f_id);

        if (faf_default_value <> f_value) then
        begin
          if (faf_need_comment = 1 and
              coalesce(f_comment, '') = '') then
            exception e_default f_format('Comment needed, question Id: %1', f_id);

          insert into ft_anketa_value (fav_from_fal_id, fav_from_faf_id, fav_comment)
          values (:fal_id, :f_id, :f_comment);

          if (faf_need_check = 1) then
            fal_need_check = 1;
        end
      end

      idx = idx + 1;
    end

    fak_code = '';
    select fak.fak_code
    from ft_anketa_kind fak
    where fak.fak_id = :faf_from_fak_id
    into :fak_code;

    if (fal_need_check = 1) then
    begin
      update ft_anketa_list
      set fal_need_check = 1
      where fal_id = :fal_id;
    end

    if (fak_id > 1) then
    begin
      update ft_anketa_list
      set fal_from_fak_id = :fak_id
      where fal_id = :fal_id;
    end

    for select edl.edl_id, edl.edl_status, edl.edl_object_id
        from edoc_list edl
        inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
        where eds.eds_from_c_id = :c_id and
              edl.edl_object_kind = f_objkind('ft_anketa_list', 'id') and
              edl.edl_from_edc_id = f_edoc_kind('ft.anketa', 'id')
        order by edl.edl_id
        into edl_id, edl_status, edl_object_id
    do
    begin
      if (edl_status = 1) then
        update edoc_list edl
        set edl.edl_date_end = current_date - 1
        where edl.edl_id = :edl_id;
      else
      if (edl_status = 0) then
        execute procedure hp_edoc_cancel('ft.anketa', f_objkind('ft_anketa_list', 'id'), :edl_object_id);
    end

    select res_is_error, res_text
    from hp_edoc_ins_ft_anketa(:fal_id, :un_id)
    into res_is_error, res_text;

    if (res_text = '') then
    begin
      if (fak_code = 'client') then
        res_text = f_lang(lang, 'Анкета-опросник сохранена и направлена клиенту на подпись.');
      else
      if (fak_code = 'zalog') then
        res_text = f_lang(lang, 'Анкета-опросник сохранена и направлена залогодателю на подпись.');

      if (fal_need_check = 1) then
        res_text = res_text || f_lang(lang, 'Далее требуется проверка, т.к. содержит нестандартные ответы.');
    end
  end
  suspend;
end
