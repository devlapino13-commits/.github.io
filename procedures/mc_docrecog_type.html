Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE MC_DOCRECOG_TYPE(
    MCL_ID D_ID,
    DOCRECOG_TYPE D_STR_255)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_100,
    DRM_NAME D_STR_255,
    DRM_CODE D_STR_255,
    DRM_API_CODE D_STR_255,
    DRM_IS_MANUAL D_BOOL,
    DRM_RECOG_2 D_BOOL,
    DRM_SHOW_FIELDS_CLIENT D_BOOL,
    DRM_HAS_PAGE1 D_BOOL,
    DRM_HAS_PAGE2 D_BOOL,
    DRM_USE_FRONT_CAMERA D_BOOL,
    DRM_CAN_LOAD_GALLERY D_BOOL,
    DRM_FIRST_ACTION D_STR_255,
    DRM_SECOND_ACTION D_STR_255)
AS
declare variable DOC_CODE D_STR_100;
declare variable C_ID D_ID; /* ID клиента, привязанного к данному mcl_id */
declare variable MAIN_DOC_DRM_CODE D_STR_255; /* drm_code основного документа клиента, с которым он проходил верификацию */
declare variable document_suffix d_str_code;
begin
    -- RFCRU-4623
    res_code = 'success';
    res_text = '';
    document_suffix = '';

    select mcl.mcl_from_c_id
    from mc_login mcl
    where mcl.mcl_id = :mcl_id
    into :c_id;

    if (:docrecog_type = 'change_phone.main_document') then
    begin
        /*
        Для docrecog_type = 'change_phone.main_document' процедура возвращает алгоритм распознавания (docrecogmain)
        для основного документа клиента, с которым он должен пройти верификацию для смены номера телефона

        Далее получаем алгоритм распознавания, который должен использоваться для верификации при смене номера телефона
        Для этого добавляем к основному DRM_CODE документа суффикс .phonechange. Именно он будет служить как признак, что
        документ был отправлен именно для смены номера телефона, и его не нужно отправлять на оцифровку и сохранять в карточку
        клиента
        */

        drm_first_action = 'draft_create';
        drm_second_action = 'client_passport_match';
        drm_use_front_camera = 0; -- Для основного документа всегда используется задняя камера
        doc_code = '';

        select drm.drm_code
        from clientdocs cds
        inner join passkind pk on cds.cds_from_pk_id = pk.pk_id
        inner join docrecogmain drm on pk.pk_from_drm_id = drm.drm_id
        where cds.cds_from_c_id = :c_id
            and cds.cds_is_main_doc = 1
            and cds.cds_is_delete = 0
            and cds.cds_verify_status = 2
        into :doc_code;

        document_suffix = '.phonechange';
    end
    else if (:docrecog_type = 'change_phone.selfie') then
    begin
        -- Для docrecog_type = 'change_phone.selfie' процедура возвращает инструкцию
        -- для прикрепления селфи пользователя при смене номера телефона без наличия доступа к старому

        drm_first_action = 'draft_create';
        drm_second_action = 'draft_update';
        drm_use_front_camera = 1;
        doc_code = 'selfie.phonechange';
    end
    else
    begin
        res_code = 'error';
        res_text = 'В приложении произошёл сбой, просим обратиться в службу технической поддержки';
    end


    select
        drm_name,
        res_drm_code || coalesce(:document_suffix, ''),
        drm_api_code,
        drm_is_manual,
        drm_recog_2,
        drm_show_fields_client,
        drm_has_page1,
        drm_has_page2,
        drm_can_load_gallery
    from hp_print_docrecog_fields(:doc_code)
    into
        :drm_name,
        :drm_code,
        :drm_api_code,
        :drm_is_manual,
        :drm_recog_2,
        :drm_show_fields_client,
        :drm_has_page1,
        :drm_has_page2,
        :drm_can_load_gallery;

    suspend;
end
