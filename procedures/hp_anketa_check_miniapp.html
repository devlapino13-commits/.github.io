CREATE PROCEDURE HP_ANKETA_CHECK_MINIAPP(
    MA_ID INTEGER)
RETURNS (
    KF_CLIENT_FIO VARCHAR(100),
    KF_CLIENT_PASS_NO VARCHAR(30),
    KF_DISB_DATE DATE,
    KC_AMOUNT DECIMAL(18,4),
    KF_ERROR_TEXT VARCHAR(255),
    KF_ERROR_IS_CRITICAL INTEGER)
AS
declare c_id id_type;
declare ma_id1 d_integer;
declare dt_attach d_date;
begin
  for
    select
      c_id,
      c_name_fio,
      c_pass_no,
      ma_date_oper,
      mac_amount
    from miniapp
      inner join product
        on pr_id = ma_from_pr_id
          and pr_code not containing '.ZALOG_CAR.'
      inner join miniappclient
        on mac_from_ma_id = ma_id
          and mac_amount > 0
      inner join clients
        on c_id = mac_from_c_id
    where ma_id = :ma_id
      and ma_date_end = '31.12.9999'
  into
    :c_id,
    :kf_client_fio,
    :kf_client_pass_no,
    :kf_disb_date,
    :kc_amount
  do
  begin
    if ((f_ft_valid(:c_id, :kf_disb_date) = 0) and
      not exists(
        select *
        from clientdocs
          inner join passkind
            on pk_id = cds_from_pk_id
          inner join passdoctype
            on pdt_id = pk_from_pdt_id
              and pdt_code = 'anketa_questionnaire'
        where cds_from_c_id = :c_id
          and cds_is_delete = 0
          and (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0) )) then
    begin
      kf_error_text = 'Для данного клиента требуется закрепить анкету - опросник';
      kf_error_is_critical = 0;
      suspend;
    end
  end
end
