CREATE PROCEDURE HP_STOP_PERC(
    K_ID D_ID,
    ADD_PARAMS D_STR_8100)
RETURNS (
    OL_ID D_ID,
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable DATE_STOP D_DATE;
declare variable IS_STOP_PERC D_BOOL;
declare variable IS_STOP_FINE D_BOOL;
declare variable DATE_STOP_PERC D_DATE;
declare variable DATE_STOP_FINE D_DATE;
declare variable K_STOP_PERC D_DATE;
declare variable K_STOP_FINE D_DATE;
begin
  date_stop = f_json_str_get_value(add_params, 'date_stop', null);
  is_stop_perc = f_json_str_get_value(add_params, 'is_stop_perc', 0);
  is_stop_fine = f_json_str_get_value(add_params, 'is_stop_fine', 0);

  if (is_stop_perc = 1) then date_stop_perc = date_stop;
    else date_stop_perc = null;
  if (is_stop_fine = 1) then date_stop_fine = date_stop;
    else date_stop_fine = null;

  if (is_stop_perc = 1 or is_stop_fine = 1) then
  begin
    k_stop_perc = null;
    k_stop_fine = null;
    select k_stop_perc, k_stop_fine from kontract
    where k_id = :k_id
    into k_stop_perc, k_stop_fine;

    if (is_stop_perc = 1 and k_stop_perc is not null and
      is_stop_fine = 1 and k_stop_fine is not null) then
    begin
      res_is_error = 1;
      res_text = 'По текущему контракту уже было расторжение.';
    end else
    begin
      execute procedure hp_kontract_stop_perc(:k_id, :date_stop_perc, :date_stop_fine);

      res_is_error = 0;
      res_text = '';

      ol_id = null;
      select ol_id from operkred, operlist
      where ok_from_k_id = :k_id
      and ok_ol_is_active = 1
      and ok_ol_date = :date_stop
      and ok_from_ol_id = ol_id
      and ol_memo starting 'Приостановление процентов договора'
      rows 1
      into ol_id;

      if (ol_id is null) then
      begin
        res_is_error = 1;
        res_text = 'Приостановление не произведено, проверьте дату приостановление!';
      end
    end
  end else
  begin
    res_is_error = 1;
    res_text = 'Не выбран приостановление процентов или пени!';
  end
  suspend;
end
