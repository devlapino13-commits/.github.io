CREATE PROCEDURE HP_PERSINFO_RESTORE_TMP(
    C_ID D_ID)
RETURNS (
    FIELD_NAME D_STR_CODE,
    PIF_KEY_ID D_ID,
    PIF_ACTIAL_DATA D_BOOL,
    PIF_UUID D_UUID,
    PIF_INDEXED D_BOOL,
    PIF_VALUE D_STR_255,
    PIF_USR_ID D_ID,
    PIF_DATETIME D_DATETIME,
    PIF_SIGNED D_BOOL)
AS
declare variable pif_source d_str_code;
declare variable adl_id d_id;
declare variable un_id d_id;
declare variable log_datetime d_datetime;
declare variable log_un_id d_id;
declare variable adcl_address_type d_integer;
declare variable is_main d_smallint;
declare variable cont_kind d_smallint;
declare variable cda_id d_id;
declare variable cda_is_delete d_bool;
declare variable first_datetime d_datetime;
begin
  pif_signed = 0;
  pif_key_id = c_id;

  pif_source = 'clients';
  pif_actial_data = 0;

  select c_datetime
  from clients
  where c_id = :c_id
  into first_datetime;

  for select cll_name_fio, cll_birth_date, cll_pass_no, cll_pass_date, cll_curr_user_id, cll_curr_datetime,
         cll_blob_pass1_id, cll_blob_pass2_id
  from clientlog
  where cll_from_c_id = :c_id
  order by cll_curr_datetime, cll_id
  as cursor cur
  do
  begin
    if (pif_datetime is null) then
    begin
      if (first_datetime > cur.cll_curr_datetime) then
      begin
        select min(coalesce(cds_create_datetime, cds_datetime))
        from clientdocs
        where cds_from_c_id = :c_id
        into first_datetime;
      end

      if (first_datetime <= cur.cll_curr_datetime) then
        pif_datetime = first_datetime;
      else
        pif_datetime = cur.cll_curr_datetime;
    end

    if (first_datetime is null or first_datetime > pif_datetime) then
      first_datetime = pif_datetime;

    if (pif_usr_id is null) then
      pif_usr_id = uxf$usr_by(1, cur.cll_curr_user_id);

    if (coalesce(cur.cll_name_fio, '') <> '') then
    begin
      field_name = 'c_name_fio';
      pif_uuid = null;
      select pif_uuid, pif_indexed
      from ux$persfield
      where pif_source = :pif_source and
            pif_field_name = :field_name and
            pif_inactive = 0
      into pif_uuid, pif_indexed;
  
      if (row_count = 1) then
      begin
        pif_value = cur.cll_name_fio;
        suspend;
      end
    end

    if (cur.cll_birth_date is not null) then
    begin
      field_name = 'c_birth_date';
      pif_uuid = null;
      select pif_uuid, pif_indexed
      from ux$persfield
      where pif_source = :pif_source and
            pif_field_name = :field_name and
            pif_inactive = 0
      into pif_uuid, pif_indexed;
  
      if (row_count = 1) then
      begin
        pif_value = f_datetostr(cur.cll_birth_date);
        suspend;
      end
    end

    if (coalesce(cur.cll_pass_no, '') <> '') then
    begin
      field_name = 'c_pass_no';
      pif_uuid = null;
      select pif_uuid, pif_indexed
      from ux$persfield
      where pif_source = :pif_source and
            pif_field_name = :field_name and
            pif_inactive = 0
      into pif_uuid, pif_indexed;
  
      if (row_count = 1) then
      begin
        pif_value = cur.cll_pass_no;
        suspend;
      end
    end

    if (cur.cll_pass_date is not null) then
    begin
      field_name = 'c_pass_date';
      pif_uuid = null;
      select pif_uuid, pif_indexed
      from ux$persfield
      where pif_source = :pif_source and
            pif_field_name = :field_name and
            pif_inactive = 0
      into pif_uuid, pif_indexed;
  
      if (row_count = 1) then
      begin
        pif_value = f_datetostr(cur.cll_pass_date);
        suspend;
      end
    end

    if (cur.cll_blob_pass1_id is not null) then
    begin
      field_name = 'c_blob_pass1_id';
      pif_uuid = null;
      select pif_uuid, pif_indexed
      from ux$persfield
      where pif_source = :pif_source and
            pif_field_name = :field_name and
            pif_inactive = 0
      into pif_uuid, pif_indexed;
  
      if (row_count = 1) then
      begin
        pif_value = cur.cll_blob_pass1_id;
        suspend;
      end
    end

    if (cur.cll_blob_pass2_id is not null) then
    begin
      field_name = 'c_blob_pass2_id';
      pif_uuid = null;
      select pif_uuid, pif_indexed
      from ux$persfield
      where pif_source = :pif_source and
            pif_field_name = :field_name and
            pif_inactive = 0
      into pif_uuid, pif_indexed;
  
      if (row_count = 1) then
      begin
        pif_value = cur.cll_blob_pass2_id;
        suspend;
      end
    end

    pif_datetime = cur.cll_curr_datetime;
    pif_usr_id = uxf$usr_by(1, cur.cll_curr_user_id);
  end

  if (pif_datetime is null) then
    pif_datetime = first_datetime;

  pif_actial_data = 1;
  for select c_name_fio,
             c_alternat_fio,
             c_name_eng,
             c_birth_date,
             uxf$perslink('refdata', c_citizen) as c_citizen,
             c_birth_country,
             c_birth_city,
             c_pass_no,
             c_pass_date,
             c_pass_date_expire,
             c_pass_who_code,
             c_pass_who,
             c_inn,
             c_blob_pass1_id,
             c_blob_pass2_id,
             c_from_un_id
  from clients
  where c_id = :c_id
  as cursor cur
  do
  begin
    if (pif_usr_id is null) then
      pif_usr_id = uxf$usr_by(1, cur.c_from_un_id);

    for select pif_uuid, pif_field_name, pif_indexed
    from ux$persfield
    where pif_source = :pif_source and
          pif_inactive = 0
    into pif_uuid, field_name, pif_indexed
    do
    begin
      pif_value = case field_name
        when 'c_name_fio' then cur.c_name_fio
        when 'c_alternat_fio' then cur.c_alternat_fio
        when 'c_name_eng' then cur.c_name_eng
        when 'c_birth_date' then f_datetostr(cur.c_birth_date)
        when 'c_citizen_str' then cur.c_citizen
        when 'c_birth_country' then cur.c_birth_country
        when 'c_birth_city' then cur.c_birth_city
        when 'c_pass_no' then cur.c_pass_no
        when 'c_pass_date' then f_datetostr(cur.c_pass_date)
        when 'c_pass_date_expire' then f_datetostr(cur.c_pass_date_expire)
        when 'c_pass_who_code' then cur.c_pass_who_code
        when 'c_pass_who' then cur.c_pass_who
        when 'c_inn' then cur.c_inn
        when 'c_blob_pass1_id' then cur.c_blob_pass1_id
        when 'c_blob_pass2_id' then cur.c_blob_pass2_id
        else '$UNDEFINED_VALUE$'
      end;
      if (trim(pif_value) = '$UNDEFINED_VALUE$') then
          exception e_message 'Undefined field: ' || field_name;

      pif_value = trim(pif_value);

      suspend;
    end
  end

  pif_source = 'addrclient';
  pif_actial_data = 0;

  for select adcl_from_adl_id, adcl_address_type, adcl_from_un_id, adcl_datetime
  from addrclog
  where adcl_from_c_id = :c_id and
        adcl_address_type in (1, 2)
  into adl_id, adcl_address_type, un_id, pif_datetime
  do
  begin
    if (adcl_address_type = 1) then
      field_name = 'address_legal';
    else
      field_name = 'address_fact';

    pif_uuid = null;
    select pif_uuid, pif_indexed
    from ux$persfield
    where pif_source = :pif_source and
          pif_field_name = :field_name and
          pif_inactive = 0
    into pif_uuid, pif_indexed;

    pif_usr_id = uxf$usr_by(1, un_id);

    for select adl_address_str, adl_from_un_id, adl_datetime
    from addrloglist
    where adl_id = :adl_id
    into :pif_value, :un_id, :pif_datetime
    do
    begin
      pif_usr_id = uxf$usr_by(1, un_id);

      suspend;
    end

    for select adl_address_str, adl_from_un_id, adl_datetime
    from addrlist
    where adl_id = :adl_id
    into :pif_value, :un_id, :pif_datetime
    do
    begin
      pif_usr_id = uxf$usr_by(1, un_id);

      suspend;
    end
  end

  for select adc_from_adl_id, adc_address_type, adc_from_un_id, adc_datetime
  from addrclient
  where adc_from_c_id = :c_id and
        adc_address_type in (1, 2)
  into adl_id, adcl_address_type, log_un_id, log_datetime
  do
  begin
    pif_actial_data = 0;

    if (adcl_address_type = 1) then
      field_name = 'address_legal';
    else
      field_name = 'address_fact';

    pif_uuid = null;
    select pif_uuid, pif_indexed
    from ux$persfield
    where pif_source = :pif_source and
          pif_field_name = :field_name and
          pif_inactive = 0
    into pif_uuid, pif_indexed;

    pif_usr_id = uxf$usr_by(1, un_id);

    for select adl_address_str, adl_from_un_id, adl_datetime
    from addrloglist
    where adl_id = :adl_id
    into :pif_value, :un_id, :pif_datetime
    do
    begin
      pif_usr_id = uxf$usr_by(1, un_id);

      suspend;
    end

    pif_actial_data = 1;
    for select adl_address_str, adl_from_un_id, adl_datetime
    from addrlist
    where adl_id = :adl_id
    into :pif_value, :un_id, :pif_datetime
    do
    begin
      if (log_datetime > pif_datetime) then
      begin
        pif_datetime = log_datetime;
        un_id = log_un_id;
      end

      pif_usr_id = uxf$usr_by(1, un_id);

      suspend;
    end
  end

  pif_source = 'contdata';

  for select el_id, el_code
  from enum_list
  where el_kind = 'cont_kind' and
        el_id <> 0
  into cont_kind, field_name
  do
  begin
    is_main = 0;
    if (field_name = 'cell_main') then
    begin
      cont_kind = 0;
      is_main = 1;
    end else
    if (field_name = 'cell_add') then
      cont_kind = 0;

    pif_uuid = null;
    select pif_uuid, pif_indexed
    from ux$persfield
    where pif_source = :pif_source and
          pif_field_name = :field_name and
          pif_inactive = 0
    into pif_uuid, pif_indexed;

    if (pif_uuid is not null) then
    begin
      pif_actial_data = 1;
      for select cda_id, clt_code, cda_is_delete, cda_date, iif(cda_status = 1, 1, 0), cda_from_un_id
      from contdata, contlist
      where cda_from_c_id = :c_id and
            cda_cont_kind = :cont_kind and
            cda_is_main = :is_main and
            cda_from_clt_id = clt_id
      order by cda_date desc, cda_id desc
      into :cda_id, :pif_value, :cda_is_delete, :pif_datetime, :pif_signed, :un_id
      do
      begin
        select clg_datetime
        from contlog
        where clg_from_cda_id = :cda_id and
              clg_datetime < (:pif_datetime + 1)
        order by clg_datetime desc
        rows 1
        into pif_datetime;
  
        pif_usr_id = uxf$usr_by(1, un_id);

        if (cda_is_delete = 1) then
          pif_value = '';
  
        suspend;
        pif_actial_data = 0;
      end
    end
  end
end
