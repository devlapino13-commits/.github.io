CREATE PROCEDURE FM_HTML_CLENTS_BL(
    MAL_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable db_id integer;
declare variable mal_from_m_id id_type;
declare variable mal_req_user id_type;
declare variable mal_req_comment varchar(512);
declare variable mal_req_datetime timestamp;
declare variable c_name_fio str_100;
declare variable c_name_fio_user str_100;
declare variable c_birth_date date;
declare variable lang d_str_lang;
declare variable c_id id_type;
declare variable k_kontract_no varchar(100);
declare variable k_id id_type;
declare variable ok_ol_date date;
declare variable delinq_days integer;
declare variable table_text d_str_8100;
declare variable loan_state d_str_100;
begin
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);
  res_text = (select res_style
              from fm_html_styles) || '<body>';

  select mo.mal_from_m_id, mo.mal_req_user, mo.mal_req_comment, mo.mal_req_datetime, cl.c_name_fio, cl.c_birth_date,
         cl.c_id
  from morapprlist mo, moratory m, clients cl
  where mo.mal_from_m_id = m.m_id and
        mo.mal_id = :mal_id and
        m.m_from_c_id = cl.c_id
  into :mal_from_m_id, :mal_req_user, :mal_req_comment, mal_req_datetime, :c_name_fio, :c_birth_date, :c_id;

  -- последний цикл клиента
  select k_id, k_kontract_no
  from knclient, kontstate, kontract
  where kc_from_c_id = :c_id and
        kc_amount_approved > 0 and
        kc_from_k_id = k_id and
        kc_from_k_id = ks_from_k_id and
        ks_ol_is_delete = 0
        --and ks_state = 1
  order by ks_date_begin desc
  rows 1
  into k_id, k_kontract_no;

  loan_state = '';

  select (case ks.ks_state
            when 1 then 'Активный'
            when 2 then 'Закрытый'
            else 'Неопределено'
          end)
  from kontstate ks
  where ks.ks_from_k_id = :k_id and
        current_date between ks.ks_date_begin and ks.ks_date_end and
        ks.ks_ol_is_delete = 0
  into :loan_state;
  loan_state = coalesce(loan_state, '');

  -- получаем платежи клиента + число дней просрочки на дату платежа
  table_text = '<table width="80%">
       <col width="35%">
       <col width="45%">
       <tr>
       <th>Дата</th>
       <th>Дней просрочки</th>
       </tr>';

  for select ok.ok_ol_date,
             (ok.ok_ol_date - min(ok.ok_sh_date)) as delinq_days
      from operkred ok
      where ok.ok_from_k_id = :k_id and
            ok.ok_ol_is_active = 1 and
            ok.ok_type in (4, 5)
      group by ok.ok_ol_date, ok.ok_from_ol_id
      order by ok.ok_ol_date, ok.ok_from_ol_id
      into :ok_ol_date, :delinq_days
  do
  begin
    delinq_days = iif(delinq_days < 0, 0, delinq_days);
    table_text = table_text || '<tr><td>' || f_datetostr(ok_ol_date) || '</td><td align="right">' || delinq_days || '</td></tr>';
  end
  table_text = table_text || '</table>';

  if (mal_from_m_id is null) then -- если поле m_from_c_id было пустым то
  begin
    select mo.mal_from_m_id, mo.mal_req_user, mo.mal_req_comment, mo.mal_req_datetime, m.m_client_fio,
           m.m_client_birth_date
    from morapprlist mo, moratory m
    where mo.mal_from_m_id = m.m_id and
          mo.mal_id = :mal_id
    into :mal_from_m_id, :mal_req_user, :mal_req_comment, mal_req_datetime, :c_name_fio, :c_birth_date;
  end

  select cl.c_name_fio
  from username u
  inner join clients cl on u.un_from_c_id = cl.c_id
  where u.un_id = :mal_req_user
  into c_name_fio_user;

  res_text = BLOB_APPEND(res_text, '<h1 class="green_text">' || c_name_fio || '</h1>' || -->>
  'Дата рождения <b>:' || f_datetostr(c_birth_date) || '</b><br>' || -->>
  'Автор запроса:<b>' || c_name_fio_user || '</b><br>' || --
  'Дата запроса:<b>' || mal_req_datetime || '</b><br>' || '<h1 class="green_text">' || --
  'Информация по последнему циклу' || '</h1>' || --
  '№ договора:<b>' || coalesce(k_kontract_no, '--') || '</b><br>' || --
  'Активная просрочка:<b>' || coalesce(delinq_days, '--') || ' дней </b><br>' || --
  'Статус кредита:<b>' || coalesce(loan_state, '--') || ' </b><br>' || '<h1 class="green_text">' || --
  'Платежи' || '</h1></b><br>');

  res_text = BLOB_APPEND(res_text, table_text || '<br>' || '<br>Комментарий сотрудника:<b>' || coalesce(mal_req_comment, 'нет комментария') || '</b></br>');

  if (mal_id is not null and
      db_id = 81001) then
  begin
    res_text = BLOB_APPEND(res_text, --
    f_uri_sql(f_lang(lang, 'Утвердить запрос'), 'go_out_bl_list', 'morapprlist', mal_id, 'text', --
    f_lang(lang, 'Комментарий для утверждения запроса'), '', 4, 256, '', '', 'dialog', '') || '&emsp;&emsp;' || --
    f_uri_sql(f_lang(lang, 'Отклонить запрос'), 'keep_bl_list', 'morapprlist', mal_id, 'text', --
    f_lang(lang, 'Комментарий для отклонения запроса'), '', 4, 256, '', '', 'dialog', '') || '<br>');
  end
  if (mal_id is not null and
      db_id in (39801, 41701)) then
  begin
--    if (un_id in (1090100000637, 1090100000184) or --
--        (exists(select *
--                from username un
--                inner join morapprlist mpl on mpl.mal_id = :mal_id
--                inner join aarep_get_reg_director(mpl.mal_from_reg_id) on un.un_from_c_id = c_id_reg_dir
--                where un.un_id = :un_id))) then
      /*Условие было убрано по запросу Чолпон задача в мантисе 223*/
      res_text = BLOB_APPEND(res_text, --
      f_uri_sql(f_lang(lang, 'Утвердить запрос'), 'go_out_bl_list', 'morapprlist', mal_id, 'text', --
      f_lang(lang, 'Комментарий для утверждения запроса'), '', 4, 256, '', '', 'dialog', '') || '&emsp;&emsp;' || --
      f_uri_sql(f_lang(lang, 'Отклонить запрос'), 'keep_bl_list', 'morapprlist', mal_id, 'text', --
      f_lang(lang, 'Комментарий для отклонения запроса'), '', 4, 256, '', '', 'dialog', '') || '<br>');
  end

  res_text = BLOB_APPEND(res_text, '</body>');
  suspend;
end
