CREATE PROCEDURE AAREP_PURCHASE_CUSTOMER
RETURNS (
    NAME D_STR_255,
    INN D_STR_20,
    ADR1 D_STR_2048,
    ADR2 D_STR_255,
    OKPO D_STR_20)
AS
declare variable reg_unicode d_integer;
declare variable pl_id id_type;
declare variable pl_name d_str_100;
declare variable adl_id id_type;
begin
  reg_unicode = (select reg_unicode from region where reg_code = '001');                                                                                                    
  select
    reg_kont_company,
    reg_inn,
    reg_from_pl_id,
    reg_address_legal,
    reg_okpo                      
  from region
  where reg_unicode = :reg_unicode          
  into
    name,
    inn,
    pl_id,
    adl_id,
    okpo;
             
  adr1 = '';
  while (pl_id <> 417000000001) do
  begin
    select
      pl_name,
      parent
    from places
    where pl_id = :pl_id
    into
      pl_name,
      pl_id;
    if (row_count = 1) then adr1 = pl_name || iif(adr1='','',', ') || adr1;
    else leave;
  end
     
  adr2 = '';
  select
    iif(pl_id is null, '', 'ул.' || pl_name) ||
    iif(adl_depart_no in ('0', ''), '', ' д.' || adl_depart_no ||
      iif(adl_depart_pref='', '', '/' || adl_depart_pref)) ||
    iif(adl_flat_no in ('0', ''), '', ' кв.' || adl_flat_no)
  from addrlist
    left join places
      on adl_street_id = pl_id
  where adl_id = :adl_id
  into
    adr2;
    
  suspend;
end
