CREATE PROCEDURE FM_HTML_LAST3_HISTORY(
    C_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable k_id decimal(18,0);
declare variable ks_date_begin date;
declare variable kc_amount decimal(18,4) = 0;
declare variable client_count integer;
declare variable cl_circle integer;
declare variable k_kolich_repay integer;
declare variable pr_name varchar(255);
declare variable ks_date_end date;
declare variable ks_state_str varchar(100);
declare variable disb_region varchar(100);
declare variable disb_lo varchar(100);
declare variable closed_region varchar(100);
declare variable closed_lo varchar(100);
declare variable line_count integer;
declare variable kf_id decimal(18,0);
declare variable pr_category integer;
declare variable c_name_fio varchar(100);
declare variable k_kontract_no varchar(100);
declare variable max_delinq_days integer;
declare variable max_delinq_count integer;
declare variable db_id integer;
declare variable kc_loan_target d_str_100;
begin

  :db_id = f_get_db_id();

  res_text = (select res_style
              from fm_html_styles) || '<body>';
  res_text = BLOB_APPEND(res_text, f_format('<h1 class="green_text">%1</h1>', iif(db_id <> 94401, 'История клиента', 'Müştərinin tarixçəsi')));

  select c_name_fio
  from clients
  where c_id = :c_id
  into :c_name_fio;
  res_text = BLOB_APPEND(res_text, f_format('<h2>%1</h2>%2<br><br>', c_name_fio, iif(:db_id <> 94401, 'последние три цикла', 'son üç dövr')));

  res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="40%">
       <col width="60%">
       <tr>' || f_format('<th>%1</th><th>%2</th></tr>', iif(db_id <> 94401, 'Условия', 'Şərtlər'),
                                                        iif(db_id <> 94401, 'Значения', 'Məlumat')));

  line_count = 0;
  for select ks_from_k_id, k_from_kf_id, k_kontract_no, ks_date_begin, kc_amount_approved, k_kolich_client_approved,
             coalesce(kc_circle, 1), k_kolich_repay, pr_name, ks_date_end, reg_code || ' ' || reg_name, c_name_fio,
             pr_category, rd_name
      from kontstate, kontract, knclient, product, region, clients,refdata
      where ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_from_k_id = k_id and
            kc_from_k_id = k_id and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            k_from_pr_id = pr_id and
            ks_region_id = reg_unicode and
            ks_kc_id = c_id and
            rd_id = kc_loan_target
      order by ks_date_begin desc
      into k_id, kf_id, k_kontract_no, ks_date_begin, kc_amount, client_count, cl_circle, k_kolich_repay, pr_name,
           ks_date_end, disb_region, disb_lo, pr_category,kc_loan_target
  do
  begin
    if (line_count = 3) then
      leave;

    max_delinq_days = 0;
    select max(ok_ol_date - ok_sh_date)
    from operkred
    where ok_type in (4, 5, 6) and
          ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date >= ok_sh_date and
          (ok_os + ok_perc) < 0
    into :max_delinq_days;
    if (max_delinq_days is null) then
      max_delinq_days = 0;

    max_delinq_count = 0;
    select count(distinct ok_sh_date)
    from operkred
    where ok_type in (4, 5, 6) and
          ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date > ok_sh_date and
          (ok_os + ok_perc) < 0
    into :max_delinq_count;
    if (max_delinq_count is null) then
      max_delinq_count = 0;

    closed_region = '';
    closed_lo = '';
    ks_state_str = iif(db_id <> 94401, 'Активный', 'Aktiv');
    if (ks_date_end < '31.12.2999') then
    begin
      ks_state_str = iif(db_id <> 94401, 'Закрыт', 'Bağlı');
      select reg_code || ' ' || reg_name, c_name_fio
      from kontstate, region, clients
      where ks_from_k_id = :k_id and
            ks_state = 2 and
            ks_ol_is_delete = 0 and
            ks_region_id = reg_unicode and
            ks_kc_id = c_id
      into closed_region, closed_lo;
    end
    else
    begin
      select reg_code || ' ' || reg_name, c_name_fio
      from knaktlist, region, clients
      where kal_from_kf_id = :kf_id and
            kal_date <= current_date and
            kal_date_end >= current_date and
            knaktlist.is_delete = 0 and
            kal_new_reg_id = reg_unicode and
            kal_new_kf_officer = c_id
      into closed_region, closed_lo;
    end
    if (cl_circle <= 0) then
      cl_circle = 1;

    res_text = BLOB_APPEND(res_text, f_format('<tr><td class="filled" align="left"><b>%1</b></td><td class="filled" align="left"><b>%2</b></td></tr>', iif(db_id <> 94401, 'Цикл клиента', 'Müştərinin dövrü'), cl_circle) ||
                           f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', iif(db_id <> 94401, 'Статус займа', 'Kreditin statusu'), ks_state_str) ||
                           f_format('<tr><td align="left">%1 №</td><td align="left">%2</td></tr>', iif(db_id <> 94401, 'Договор', 'Müqavilə'), k_kontract_no) ||
                           f_format('<tr><td align="left">%1</td><td align="left">', iif(db_id <> 94401, 'Дата выдачи', 'Verilmə tarixi')) || right('0' || extract(day from ks_date_begin), 2) || '.' || right('0' || extract(month from ks_date_begin), 2) || '.' || extract(year from ks_date_begin) || '</td></tr>'||
                           f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', iif(db_id <> 94401, 'Цель', 'Məqsəd'), kc_loan_target));
    if (ks_date_end < '31.12.2999') then
      res_text = BLOB_APPEND(res_text, f_format('<tr><td align="left">%1</td><td align="left">', iif(db_id <> 94401, 'Дата закрытия', 'Bağlanma tarixi')) || right('0' || extract(day from ks_date_end), 2) || '.' || right('0' || extract(month from ks_date_end), 2) || '.' || extract(year from ks_date_end) || '</td></tr>');

    res_text = BLOB_APPEND(res_text, f_format('<tr><td align="left"><b>%1</b></td><td align="left"><b>%2</b></td></tr>', iif(db_id <> 94401, 'Макс.дни просрочки', 'Maksimal gecikmə günü'),max_delinq_days) ||
                           f_format('<tr><td align="left"><b>%1</b></td><td align="left"><b>%2</b></td></tr>', iif(db_id <> 94401, 'Кол-во платежей совершенные с опозданием', 'Gecikmə ilə ödənilən ödəniş sayı'), max_delinq_count) ||
                           f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', iif(db_id <> 94401, 'Сумма клиента', 'Müştərinin məbləği'), round(kc_amount)) ||
                           f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', iif(db_id <> 94401, 'Кол-во участников', 'İştirakçıların sayı'), client_count) ||
                           f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', iif(db_id <> 94401, 'Срок кредита', 'Kreditin müddəti'), k_kolich_repay) ||
                           f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', iif(db_id <> 94401, 'Продукт', 'Məhsul'), pr_name));

    if (ks_date_end >= '31.12.2999') then
    begin
      if (disb_region <> closed_region or disb_lo <> closed_lo) then
      begin
        res_text = BLOB_APPEND(res_text, f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr><tr><td align="left">%3</td><td align="left">%4</td></tr>', iif(db_id <> 94401, 'Место выдачи', 'Kr. verən filial'),
                                                                                                                                                              disb_region,
                                                                                                                                                              iif(db_id <> 94401, 'Выдавший спец.', 'Kr.verən mütəx.'),
                                                                                                                                                              disb_lo));
      end
    end
    if (ks_date_end < '31.12.2999') then
    begin
      res_text = BLOB_APPEND(res_text, f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr><tr><td align="left">%3</td><td align="left">%4</td></tr>', iif(db_id <> 94401, 'Место закрытия', 'Bağlanma filialı'),
                                                                                                                                                closed_region,
                                                                                                                                                iif(db_id <> 94401, 'СВЗ на момент закрытия', 'Bağlanma tarixinə kr. mütəx.'),
                                                                                                                                                closed_lo));
    end
    else
    begin
      res_text = BLOB_APPEND(res_text, f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr><tr><td align="left">%3</td><td align="left">%4</td></tr>', iif(db_id <> 94401, 'Текущий офис', 'Cari filial'),
                                                                                                                                                            closed_region,
                                                                                                                                                            iif(db_id <> 94401, 'Текущий СВЗ', 'Cari mütəx.'),
                                                                                                                                                            closed_lo));
    end

    if (pr_category = 0) then
      line_count = line_count + 1;
  end
  res_text = BLOB_APPEND(res_text, '</table>');
  res_text = BLOB_APPEND(res_text, '</body>');
  suspend;

end
