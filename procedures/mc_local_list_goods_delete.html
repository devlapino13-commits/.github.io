CREATE PROCEDURE MC_LOCAL_LIST_GOODS_DELETE(
    MCL_ID BIGINT,
    GOODS_ID D_ID)
AS
declare variable cs_id d_id;
begin
if (f_get_db_id() not in (39801, 41701)) then exit;
  select cs.cs_id
  from mc_login mcl
  inner join clientshops cs on cs.cs_from_c_id = mcl.mcl_from_c_id
  where cs.cs_is_delete = 0 and
        mcl.mcl_is_delete = 0 and mcl.mcl_id = :mcl_id
  into cs_id;

  if (exists(select *
             from gi_local_list gll
             where gll.gll_id = :goods_id and
                   gll.gll_from_cs_id = :cs_id) and
      f_get_db_id() in (39801, 41701)) then
    delete from gi_local_list gll
    where gll.gll_id = :goods_id;
end
