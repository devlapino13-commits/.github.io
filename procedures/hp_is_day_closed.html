CREATE PROCEDURE HP_IS_DAY_CLOSED(
    REG_ID INTEGER,
    REG_DATE DATE,
    REG_IS_CASH INTEGER)
RETURNS (
    DAY_CLOSED INTEGER,
    DAY_MEMO VARCHAR(255))
AS
begin
  /* Procedure Text */
  day_closed = 0;
  day_memo = '';
  if (reg_is_cash = 1) then
  begin
    select cd_is_cash_closed, cd_open_memo
    from closeday
    where cd_from_reg_id = :reg_id and
          cd_date = :reg_date and
          cd_is_cash_closed <> -1
    order by cd_number desc
    rows 1
    into :day_closed, day_memo;
  end
  else
  begin
    select cd_is_closed, cd_open_memo
    from closeday
    where cd_from_reg_id = :reg_id and
          cd_date = :reg_date and
          cd_is_closed <> -1
    order by cd_number desc
    rows 1
    into :day_closed, day_memo;
  end
  if (day_closed is null) then
    day_closed = 0;
  suspend;
end
