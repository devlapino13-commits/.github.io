CREATE PROCEDURE FM_LOYAL_CLIENT_LIST(
    DATE_END D_DATE,
    FIND_TEXT D_STR_255,
    UN_ID D_ID,
    REG_ID D_INTEGER)
RETURNS (
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID,
    TEXT_1 VARCHAR(512),
    TEXT_2 VARCHAR(512),
    TEXT_3 VARCHAR(512))
AS
declare variable C_ID D_ID;
declare variable AG_DATE_BEGIN D_DATE;
declare variable AG_STATUS D_SMALLINT;
declare variable LINK_COUNT D_INTEGER;
declare variable LINK_DISB D_INTEGER;
declare variable LINK_CALCED D_CURRENCY;
declare variable LINK_BALANCE D_CURRENCY;
declare variable RD_ID D_INTEGER;
declare variable LO_ID D_ID;
declare variable MGR_ID D_ID;
declare variable JK_SYSTEM_CODE D_STR_100;
declare variable LML_ID D_INTEGER;
begin
  if (find_text is null) then
    find_text = '';
  if (date_end is null) then
    date_end = current_date;

  select un_from_c_id
  from username
  where un_id = :un_id
  into :lo_id;

  if (lo_id is null) then
    exit;

  obj_kind = f_objkind('agreements', 'id');
  rd_id = f_get_rd_id('AgreementKinds.loyal_agreement');

  if (rd_id is null) then
    exit;

  select jk.jk_system_code
  from hr_pstatus hps
  inner join joblist jl on jl.jl_id = hps.ps_from_jl_id
  inner join jobkind jk on jk.jk_id = jl.jl_kind
  where hps.ps_from_c_id = :lo_id and
        hps.ps_is_working = 1 and
        current_date between hps.ps_date_begin and hps.ps_date_end
  into :jk_system_code;

  if (jk_system_code <> 'JobSystem.Region_KM') then
  begin
    for select distinct kc_from_c_id
        from kontstate, knclient, knaktlist, kontract
        where ks_ol_is_delete = 0
        and ks_state = 1
        and :date_end >= ks_date_begin
        and dateadd(month, -24, :date_end) <= ks_date_end
        and ks_from_k_id = kc_from_k_id
        and kc_amount_approved > 0
        and ks_from_k_id = k_id
        and k_from_kf_id = kal_from_kf_id
        and knaktlist.is_delete = 0
        and :date_end between kal_date and kal_date_end
        and kal_new_kf_officer = :lo_id
        into c_id
    do
    begin
      select c_name_fio
      from clients
      where c_id = :c_id
      into text_1;

      if (find_text <> '') then
        if (position(find_text, text_1) = 0) then
          continue;

      obj_id = null;
      text_2 = '';
      text_3 = '';

      select ag_id, ag_date_begin, ag_status
      from agreements
      where ag_from_rd_id = :rd_id and
            ag_object_kind = f_objkind('clients', 'id') and
            ag_object_id = :c_id and
            ag_status <> -1
      order by ag_date_begin desc 
      rows 1
      into obj_id, ag_date_begin, ag_status;

      if (obj_id is not null) then
      begin
        if (ag_status = 1) then
          text_1 = text_1 || '<br>' || 'Участвует с ' || f_datetostr(ag_date_begin);
        else
          text_1 = text_1 || '<br>' || 'Приглашение не подтверждено';

        link_count = 0;
        link_disb = 0;
        link_calced = 0;
        link_balance = 0;

        select count(*)
        from clientlink
        where clk_from_c_id = :c_id
        into link_count;

        select count(*), sum(iif(ogr_sign = 1, ogr_amount, 0.0)), sum(ogr_amount * ogr_sign * f_loyal(ogr_from_ol_id, :date_end) + ogr_tax_fl * ogr_sign * f_loyal(ogr_from_ol_id, :date_end))
        from operagr
        where ogr_from_ag_id = :obj_id and
              ogr_ol_is_active = 1
        into link_disb, link_calced, link_balance;
        if (link_disb is null) then
          link_disb = 0;
        if (link_calced is null) then
          link_calced = 0;
        if (link_balance is null) then
          link_balance = 0;

        text_2 = link_count || '<br>' || link_disb;
        text_3 = cast(link_calced as decimal(15,0)) || '<br>' || cast(link_balance as decimal(15,0));

        suspend;
      end
    end
  end
  if (jk_system_code = 'JobSystem.Region_KM') then
  begin
    mgr_id = lo_id;

    select lml_id
    from loanmgrlist
    where lml_from_reg_id = :reg_id and
          iif(lml_from_c_id is not null, lml_from_c_id = :mgr_id, lml_from_c_id is null) and
          :date_end between lml_date_begin and lml_date_end
    order by 1 desc rows 1
    into lml_id;

    for select c_id
        from hr_pstatus
        inner join joblist on ps_from_jl_id = jl_id
        inner join jobkind on jl_kind = jk_id
        inner join clients on c_id = ps_from_c_id
        inner join region_list(:reg_id, 1, 3, 1) on id_list = ps_from_reg_id
        where :date_end between ps_date_begin and ps_date_end and
              ps_is_working = 1 and
              ps_status = 1 and
              f_lo_mgr_id(null, :date_end, c_id, ps_from_reg_id) = :lml_id
        into lo_id
    do
    begin
      for select distinct kc_from_c_id
          from oksummary, knclient
          where :date_end between oks_date_begin and oks_date_end and
                oks_from_k_id = kc_from_k_id and
                kc_amount_approved > 0 and
                oks_kc_id = :lo_id and
                oks_close_state = 0
          into c_id
      do
      begin
        select c_name_fio
        from clients
        where c_id = :c_id
        into text_1;

        if (find_text <> '') then
          if (position(find_text, text_1) = 0) then
            continue;

        obj_id = null;
        text_2 = '';
        text_3 = '';

        select ag_id, ag_date_begin, ag_status
        from agreements
        where ag_from_rd_id = :rd_id and
              ag_object_kind = f_objkind('clients', 'id') and
              ag_object_id = :c_id and
              ag_status <> -1
        into obj_id, ag_date_begin, ag_status;

        if (obj_id is not null) then
        begin
          if (ag_status = 1) then
            text_1 = text_1 || '<br>' || 'Участвует с ' || f_datetostr(ag_date_begin);
          else
            text_1 = text_1 || '<br>' || 'Приглашение не подтверждено';

          link_count = 0;
          link_disb = 0;
          link_calced = 0;
          link_balance = 0;

          select count(*)
          from clientlink
          where clk_from_c_id = :c_id
          into link_count;

          select count(*), sum(iif(ogr_sign = 1, ogr_amount, 0.0)), sum(ogr_amount * ogr_sign * f_loyal(ogr_from_ol_id, :date_end) + ogr_tax_fl * ogr_sign * f_loyal(ogr_from_ol_id, :date_end))
          from operagr
          where ogr_from_ag_id = :obj_id and
                ogr_ol_is_active = 1
          into link_disb, link_calced, link_balance;
          if (link_disb is null) then
            link_disb = 0;
          if (link_calced is null) then
            link_calced = 0;
          if (link_balance is null) then
            link_balance = 0;

          text_2 = link_count || '<br>' || link_disb;
          text_3 = cast(link_calced as decimal(15,0)) || '<br>' || cast(link_balance as decimal(15,0));

          suspend;
        end
      end
    end
  end
end
