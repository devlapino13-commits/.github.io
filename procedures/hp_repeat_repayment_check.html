CREATE PROCEDURE HP_REPEAT_REPAYMENT_CHECK(
    K_ID D_ID,
    START_DATE D_DATE,
    REVERSE_DATE D_DATE,
    LOAN_REVERSE D_SMALLINT,
    BOOK_REVERSE D_SMALLINT,
    SQL_TEXT D_STR_8100,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable db_id d_integer;
declare variable operbook_close_date d_date;
declare variable k_disb_date d_date;
begin
  -- Процедура проверяет возможность запуска процедуры HP_REPEAT_REPAYMENT
  res_text = '';

  if (res_text = '' and coalesce(loan_reverse, 0) = 0) then
    res_text = 'Действие с записями по займам не указано';

  if (res_text = '' and coalesce(book_reverse, 0) = 0) then
    res_text = 'Действие бух.проводками не указано';

  if (res_text = '' and k_id is null) then
    res_text = 'ID займа не указан';

  if (res_text = '' and start_date is null) then
    res_text = 'Дата начала не указана';

  if (res_text = '' and trim(coalesce(sql_text, '')) = '') then
    res_text = 'Корректировочный SQL скрипт не указан. Если не требуется, пропишите слово EMPTY';

  if (res_text = '' and un_id is null) then
    res_text = 'Пользователь не указан';

  if (res_text = '' and exists(select * from temp_operrepeat
  where opr_from_k_id = :k_id)) then
    res_text = f_format('Первичные данные уже занесены в лог, сначала очистите TEMP_OPERREPEAT: %1', k_id);

  db_id = f_get_db_id();

  if (res_text = '' and db_id <> 81001) then
    res_text = 'Данная опция пока доступна только для РФ';

  operbook_close_date = f_get_const('kn_operbook_close_date');

  if (res_text = '' and reverse_date <= operbook_close_date) then
    res_text = f_format('Дата сторнирования не должна быть в закрытом периоде. Опердень закрыт: %1',
                        f_datetostr(operbook_close_date));

  if (res_text = '' and start_date > reverse_date) then
    res_text = 'Дата проведения должна быть больше чем дата начала';

  if (res_text = '') then
  begin
    select k_disb_date
    from kontract
    where k_id = :k_id
    into k_disb_date;
  
    if (start_date <= k_disb_date) then
      res_text = 'Дата начала должна быть позже даты выдачи займа';
  end

  suspend;
end
