CREATE PROCEDURE HP_INSERT_COMPET_HR_LIST_CHECK(
    C_ID D_ID,
    REG_ID D_INTEGER,
    DATE_BEGIN DATE)
RETURNS (
    ANSWER VARCHAR(20))
AS
begin
  answer = '';
  if (exists(select *
             from compet_hr_list chl
             where chl.chl_c_id = :c_id and
                   chl.chl_reg_id = :reg_id) or exists(select *
                                                       from hr_pstatus ps
                                                       where ps.ps_from_c_id = :c_id and
                                                             iif(ps.ps_date_end is null, cast('31.12.2999' as date), ps.ps_date_end) > :date_begin and
                                                             ps.ps_from_reg_id = :reg_id and
                                                             ps.ps_status = 1)) then
  begin
    answer = 'not aloud';
    suspend;
  end
  else
  begin
    answer = 'aloud';
    suspend;
  end
end
