CREATE PROCEDURE HP_CARDOPER_ANNUL_ADD(
    OL_ID D_ID,
    UN_ID D_ID)
RETURNS (
    CON_ID D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable res_ol_id d_id;
declare variable res_ol_date d_date;
declare variable res_ol_datetime d_datetime;
declare variable res_reg_id d_integer;
declare variable res_k_id d_id;
declare variable res_k_disb_kind d_integer;
declare variable res_c_id d_id;
declare variable res_cca_id d_integer;
declare variable res_summa d_currency;
declare variable res_summaperc d_currency;
declare variable res_is_delete d_smallint;
begin
  err_text = '';
  con_id = 0;

  if (exists(select * from cardoper_annul where con_ol_id = :ol_id))
    then err_text = 'Выбранная операция уже есть в списке.';

  if (err_text = '') then
  begin
    select
      ol_id,
      ol_date,
      ol_datetime,
      ol_from_reg_id,
      ol_document_client,
      is_delete
    from operlist
    where ol_id = :ol_id
    into
      :res_ol_id,
      :res_ol_date,
      :res_ol_datetime,
      :res_reg_id,
      :res_c_id,
      :res_is_delete;
    if (res_ol_id > 0) then
    begin
      if (res_is_delete = 0) then
      begin
        select
          ok_from_k_id,
          abs(sum(ok_os + ok_perc + ok_tax)),
          abs(sum(ok_perc + ok_tax))
        from operkred
        where ok_from_ol_id = :res_ol_id
          and ok_type = 4
        group by ok_from_k_id
       into
         :res_k_id,
         :res_summa,
         :res_summaperc;

        if (res_k_id > 0) then
        begin
          res_k_disb_kind = (
            select
              k_disb_kind
            from kontract
            where k_id = :res_k_id);
          if (res_k_disb_kind = 6) then
          begin
            res_cca_id = (
              select
                cca_id
              from cardclientid
                inner join cardclientacc
                  on cca_from_cci_id = cci_id
              where cci_from_c_id = :res_c_id);
            if (exists(
              select * from operkred
                inner join operlist
                  on ol_id = ok_from_ol_id
                    and ol_datetime > :res_ol_datetime
              where ok_from_k_id = :res_k_id
                and ok_from_ol_id <> :res_ol_id
                and ok_ol_date >= :res_ol_date
                and ok_type = 4
                and ok_ol_is_active = 1)) then
            err_text = 'Это не последняя операция погашения.';
          end else
            err_text = 'Данная операция не является карточной.';
        end else
          err_text = 'Это не кредитная операция погашения.';
      end else
        err_text = 'Нельзя выбрать аннулированную операцию.';
    end else
      err_text = 'Операция не определена.';
  end

  if (err_text = '') then
    insert into cardoper_annul(
      con_ol_id,
      con_reg_id,
      con_k_id,
      con_c_id,
      con_cca_id,
      con_summa,
      con_summaperc,
      con_un_id
    ) values (
      :res_ol_id,
      :res_reg_id,
      :res_k_id,
      :res_c_id,
      :res_cca_id,
      coalesce(:res_summa, 0),
      coalesce(:res_summaperc, 0),
      :un_id
    ) returning con_id
    into :con_id;

  suspend;
end
