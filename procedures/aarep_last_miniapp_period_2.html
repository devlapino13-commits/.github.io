CREATE PROCEDURE AAREP_LAST_MINIAPP_PERIOD_2(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    REGION_NAME VARCHAR(50),
    REGION_ID INTEGER,
    MA_DATE_OPER DATE,
    PLAN_OS DECIMAL(18,4),
    MA_PLAN_SUM DECIMAL(18,4),
    MA_PLAN_SALDO DECIMAL(18,4))
AS
begin
  for select reg_code || ' ' || reg_name, reg_id, date_action, sum(plan_os), sum(ma_plan_sum),
             sum(plan_os) - sum(ma_plan_sum)
      from aarep_last_miniapp_period_1(:report_region_id, :report_date_begin, :report_date_end), region, region_list(:report_region_id, 1, -1, 1)
      where reg_id = reg_unicode and
            (id_list = reg_unicode)
      group by reg_code, reg_name, reg_id, date_action
      into :region_name, :region_id, :ma_date_oper, :plan_os, :ma_plan_sum, :ma_plan_saldo
  do
  begin
    suspend;
  end
  suspend;
end
