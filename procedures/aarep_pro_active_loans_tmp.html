CREATE PROCEDURE AAREP_PRO_ACTIVE_LOANS_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    BT_ID INTEGER,
    IS_IPF SMALLINT = 0)
RETURNS (
    REP_OBLAST VARCHAR(255),
    REP_BUS_NAME VARCHAR(255),
    REP_PLAN_OS DECIMAL(18,4),
    REP_FACT_OS DECIMAL(18,4),
    REP_BALANCE DECIMAL(18,4),
    REP_CL_KOLICH INTEGER,
    REP_PERC_AMOUNT_ANNUAL DOUBLE PRECISION,
    REP_PERC_AMOUNT_EFFECT DOUBLE PRECISION,
    REP_MIN_DISB DECIMAL(18,4),
    REP_MAX_DISB DECIMAL(18,4),
    REP_K_PERIOD DECIMAL(18,4),
    REP_DISB_AMOUNT DECIMAL(18,4),
    REP_BALANCE_BEGIN DECIMAL(18,4),
    REP_DISB_CL_AMOUNT DECIMAL(18,4),
    REP_DISB_CL_KOLICH INTEGER,
    REP_PERC_DISB_ANNUAL DECIMAL(18,4),
    KOL_CLI_FACT INTEGER,
    LKC_BALANCE CURR_TYPE,
    REP_CL_TYPE D_INTEGER)
AS
declare variable k_id_prior decimal(18,0);
declare variable ol_date date;
declare variable sh_date date;
declare variable ok_os decimal(18,4);
declare variable sh_os decimal(18,4);
declare variable rep_k_plan_os decimal(18,4);
declare variable rep_k_fact_os integer;
declare variable rep_k_disb_amount decimal(18,4);
declare variable n decimal(18,7);
declare variable k_balance decimal(18,4);
declare variable ok_type integer;
declare variable k_perc_annu decimal(18,7);
declare variable p_perc_eff decimal(18,7);
declare variable k_index_curr_rate decimal(18,4);
declare variable def_curr varchar(3);
declare variable k_index_curr_code varchar(3);
declare variable pr_set_rate_in_kont integer;
declare variable curr_rate decimal(18,4);
declare variable ks_date_end date;
declare variable k_id decimal(18,0);
declare variable k_balance_begin decimal(18,4);
declare variable k_disb_amount decimal(18,4);
declare variable ks_date_begin date;
declare variable k_percent decimal(18,7);
declare variable cm_is_client_mode integer;
declare variable c_id id_type;
declare variable k_disb_date date;
declare variable dd_close_tmp date;
begin
-- operkred_fix
  select cm_default_currency, cm_is_client_mode
  from company
  into def_curr, cm_is_client_mode;

  if (cm_is_client_mode = 0) then
  begin
    k_id_prior = 0;
    rep_perc_amount_annual = 0;
    rep_perc_amount_effect = 0;
    rep_min_disb = 999999999999;
    rep_max_disb = 0;
    rep_plan_os = 0;
    rep_fact_os = 0;
    rep_disb_amount = 0;
    rep_balance = 0;
    rep_cl_kolich = 0;
    rep_k_period = 0;
    kol_cli_fact = 0;

    for select reg_stat_rep_region, bsl_name
        from region_list(:rep_region_id, 1, -1, 1), busstandartlist, region
        where reg_unicode = id_list and
              bsl_from_bsm_id = :bt_id -- нбкр про
              and
              bsl_id > 0 and
              reg_status = 3
        into :rep_oblast, :rep_bus_name
    do
    begin
      if (rep_oblast = 'г.ош') then
        rep_oblast = 'ошская область';
      suspend;
    end

    for select k_id, k_amount_approved * k_index_curr_rate, reg_stat_rep_region, bsl_name,
               kc_amount_approved * k_index_curr_rate, kd_percent_per_day * 365 as k_percent_begin_annual,
               k_percent_effect, k_period, k_index_curr_rate, k_index_curr_code, pr_set_rate_in_kont, ks_date_begin,
               ks_date_end, k_percent_per_day * 365.0, k_disb_date, kc_from_c_id, c_type
        from kontstate, region_list(:rep_region_id, 1, -1, 1), knclient, clients,  kontract, busstandartlist, busstandartlink, region, knaktlist, product, kontdata
        where cast(ks_date_begin as date) <= :rep_date_end and
              ks_date_end >= (:rep_date_begin - 1) and
              ks_state = 1 and
              ks_date_end >= cast(ks_date_begin as date) and
              kal_new_reg_id = reg_unicode and
              kal_from_kf_id = k_from_kf_id and
              kal_date <= :rep_date_end and
              kal_date_end >= :rep_date_end and
              knaktlist.is_delete = 0 and
              ks_region_id = id_list and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_from_k_id = kc_from_k_id and
              kc_amount_approved > 0 and
              ks_from_k_id = k_id and kc_from_c_id = c_id and
              bsl_from_bsm_id = :bt_id -- нбкр про
              and
              bslk_from_bsm_id = bsl_from_bsm_id and
              bslk_from_bsl_id = bsl_id and
              k_from_pr_id = pr_id and
              bslk_from_bs_id = kc_busuness_subtype and
              kd_from_k_id = k_id -- добавленные строки, процент берет из таблицы kontdata
              and
              :rep_date_end between kd_ol_date and kd_date_end and
              kd_ol_is_delete = 0 and
              (:is_ipf = 0 or
              (:is_ipf = 1 and
              pr_code containing '.ipf.'))
        order by k_id
        into k_id, rep_k_disb_amount, rep_oblast, rep_bus_name, rep_disb_amount, k_perc_annu, p_perc_eff, rep_k_period,
             k_index_curr_rate, k_index_curr_code, pr_set_rate_in_kont, ks_date_begin, ks_date_end, k_percent,
             k_disb_date, c_id, rep_cl_type
    do
    begin
      k_perc_annu = coalesce(f_get_perc_averaged(k_id), k_perc_annu);
      rep_disb_cl_kolich = 0;
      rep_disb_cl_amount = 0;
      rep_perc_disb_annual = 0;
      if (k_id <> k_id_prior) then
      begin
        k_balance = 0;
        rep_balance_begin = 0;
        rep_k_plan_os = 0;
        rep_k_fact_os = 0;
        k_balance_begin = 0;
        for select ok_ol_date, ok_sh_date, ok_os, ok_sh_os, ok_type
            from operkred
            where ok_ol_is_active = 1 and
                  ok_from_k_id = :k_id and
                  ok_ol_date <= :rep_date_end and
                  ok_from_c_id = 0
            into ol_date, sh_date, ok_os, sh_os, ok_type
        do
        begin
          if ((ol_date < :rep_date_begin) and
              (ok_type in (4, 5, 6, 8))) then
            k_balance_begin = k_balance_begin + :ok_os;
          if ((ks_date_end >= :rep_date_end) and
              (ol_date <= :rep_date_end) and
              (ok_type in (4, 5, 6, 8))) then
            k_balance = k_balance + :ok_os;
          if ((ol_date <= :rep_date_begin) and
              (sh_date between :rep_date_begin and :rep_date_end)) then
            rep_k_plan_os = rep_k_plan_os + :sh_os;
          if ((ol_date between :rep_date_begin and :rep_date_end) and
              (ok_type = 4)) then
            rep_k_fact_os = rep_k_fact_os - :ok_os;
        end
        if (k_balance > 0) then
        begin
          rep_cl_kolich = 1;
        end
        else
        begin
          rep_cl_kolich = 0;
          rep_k_period = 0;
        end
        k_id_prior = k_id;
        if (k_index_curr_code <> def_curr) then
        begin
          if (rep_cl_kolich = 1) then
          begin
            select cr_rate
            from opercurr
            where cr_date <= :rep_date_end and
                  cr_date_end >= :rep_date_end and
                  cr_from_cu_id = :k_index_curr_code
            into :curr_rate;
            if (pr_set_rate_in_kont = 1) then
            begin
              if (curr_rate < k_index_curr_rate) then
                curr_rate = k_index_curr_rate;
            end
            k_balance = cast(k_balance * curr_rate as decimal(15,2));
          end

          select cr_rate
          from opercurr
          where cr_date <= :rep_date_begin and
                cr_date_end >= :rep_date_begin and
                cr_from_cu_id = :k_index_curr_code
          into :curr_rate;
          if (pr_set_rate_in_kont = 1) then
          begin
            if (curr_rate < k_index_curr_rate) then
              curr_rate = k_index_curr_rate;
          end
          k_balance_begin = cast(k_balance_begin * curr_rate as decimal(15,2));
          rep_k_plan_os = cast(rep_k_plan_os * curr_rate as decimal(15,2));
          rep_k_fact_os = cast(rep_k_fact_os * curr_rate as decimal(15,2));
        end
      end

      if (rep_cl_kolich = 1) then
      begin
        kol_cli_fact = 1;
        for select kc_date_close_os
            from knclient
            where ((kc_date_disb between '01.01.1900' and (:k_disb_date - 1)) or
                  (kc_date_disb = :k_disb_date and
                  kc_from_k_id < :k_id)) and
                  kc_date_close_os between :k_disb_date and '31.12.9999' and
                  kc_from_c_id = :c_id
            into :dd_close_tmp
        do
        begin
          if (dd_close_tmp >= :rep_date_end) then
          begin
            kol_cli_fact = 0;
          end
        end
      end
      else
        kol_cli_fact = 0;

      n = cast(rep_disb_amount / rep_k_disb_amount as decimal(15,7));
      rep_balance = cast(k_balance * n as decimal(15,2));
      rep_balance_begin = cast(k_balance_begin * n as decimal(15,2));
      if (ks_date_begin between :rep_date_begin and :rep_date_end) then
      begin
        rep_disb_cl_amount = rep_disb_amount;
        rep_disb_cl_kolich = 1;
        rep_perc_disb_annual = rep_disb_amount / 100.0;
        rep_perc_disb_annual = rep_perc_disb_annual * k_perc_annu;
      end
      rep_plan_os = cast(rep_k_plan_os * n as decimal(15,2));
      rep_fact_os = cast(rep_k_fact_os * n as decimal(15,2));
      if (rep_cl_kolich = 1) then
      begin
        rep_perc_amount_annual = cast(rep_balance as double precision) / 100.0;
        rep_perc_amount_annual = rep_perc_amount_annual * k_perc_annu;
        rep_perc_amount_effect = cast(rep_balance as double precision) / 100.0;
        rep_perc_amount_effect = rep_perc_amount_effect * p_perc_eff;
        rep_min_disb = rep_disb_amount;
        rep_max_disb = rep_disb_amount;
      end
      else
      begin
        rep_perc_amount_annual = 0;
        rep_perc_amount_effect = 0;
        rep_min_disb = 999999999999;
        rep_max_disb = 0;
        rep_disb_amount = 0;
        rep_balance = 0;
        rep_k_period = 0;
      end
      if (rep_oblast = 'г.ош') then
        rep_oblast = 'ошская область';

      lkc_balance = null;
      select lkc_balance
      from llc_main, llc_kont_cli
      where lm_date_status between :rep_date_end - 4 and :rep_date_end + 4 and
            lm_id = lkc_from_lm_id and
            lkc_from_k_id = :k_id and
            lkc_from_c_id = :c_id
      into lkc_balance;

      suspend;
    end
    exit;
  end

  k_id_prior = 0;
  rep_perc_amount_annual = 0;
  rep_perc_amount_effect = 0;
  rep_min_disb = 999999999999;
  rep_max_disb = 0;
  rep_plan_os = 0;
  rep_fact_os = 0;
  rep_disb_amount = 0;
  rep_balance = 0;
  rep_cl_kolich = 0;
  rep_k_period = 0;
  kol_cli_fact = 0;

  for select reg_stat_rep_region, bsl_name
      from region_list(:rep_region_id, 1, -1, 1), busstandartlist, region
      where reg_unicode = id_list and
            bsl_from_bsm_id = :bt_id -- нбкр про
            and
            bsl_id > 0 and
            reg_status = 3
      into :rep_oblast, :rep_bus_name
  do
  begin
    if (rep_oblast = 'г.ош') then
      rep_oblast = 'ошская область';
    suspend;
  end

  for select k_id, k_amount_approved * k_index_curr_rate, reg_stat_rep_region, bsl_name,
             kc_amount_approved * k_index_curr_rate, kd_percent_per_day * 365 as k_percent_begin_annual,
             k_percent_effect, k_period, k_index_curr_rate, k_index_curr_code, pr_set_rate_in_kont, ks_date_begin,
             ks_date_end, k_percent_per_day * 365.0, kc_from_c_id, k_disb_date, c_type
      from kontstate, region_list(:rep_region_id, 1, -1, 1), knclient, clients, kontract, busstandartlist, busstandartlink, region, knaktlist, product, kontdata
      where cast(ks_date_begin as date) <= :rep_date_end and
            ks_date_end >= (:rep_date_begin - 1) and
            ks_state = 1 and
            ks_date_end >= cast(ks_date_begin as date) and
            kal_new_reg_id = reg_unicode and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            knaktlist.is_delete = 0 and
            ks_region_id = id_list and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            ks_from_k_id = k_id and kc_from_c_id = c_id and
            bsl_from_bsm_id = :bt_id -- нбкр про
            and
            bslk_from_bsm_id = bsl_from_bsm_id and
            bslk_from_bsl_id = bsl_id and
            k_from_pr_id = pr_id and
            bslk_from_bs_id = kc_busuness_subtype and
            kd_from_k_id = k_id -- добавленные строки, процент берет из таблицы kontdata
            and
            :rep_date_end between kd_ol_date and kd_date_end and
            kd_ol_is_delete = 0 and
            (:is_ipf = 0 or
            (:is_ipf = 1 and
            pr_code containing '.ipf.'))
      order by k_id
      into k_id, rep_k_disb_amount, rep_oblast, rep_bus_name, rep_disb_amount, k_perc_annu, p_perc_eff, rep_k_period,
           k_index_curr_rate, k_index_curr_code, pr_set_rate_in_kont, ks_date_begin, ks_date_end, k_percent, c_id,
           k_disb_date, rep_cl_type
  do
  begin                                                     
    k_perc_annu = coalesce(f_get_perc_averaged(k_id), k_perc_annu);
    rep_disb_cl_kolich = 0;
    rep_disb_cl_amount = 0;
    rep_perc_disb_annual = 0;
    if (k_id <> k_id_prior) then
    begin
      k_balance = 0;
      rep_balance_begin = 0;
      rep_k_plan_os = 0;
      rep_k_fact_os = 0;
      k_balance_begin = 0;
      for select ok_ol_date, ok_sh_date, ok_os, ok_sh_os, ok_type
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_k_id = :k_id and
                ok_ol_date <= :rep_date_end and
                ok_from_c_id = :c_id
          into ol_date, sh_date, ok_os, sh_os, ok_type
      do
      begin
        if ((ol_date < :rep_date_begin) and
            (ok_type in (4, 5, 6, 8))) then
          k_balance_begin = k_balance_begin + :ok_os;
        if ((ks_date_end >= :rep_date_end) and
            (ol_date <= :rep_date_end) and
            (ok_type in (4, 5, 6, 8))) then
          k_balance = k_balance + :ok_os;
        if ((ol_date <= :rep_date_begin) and
            (sh_date between :rep_date_begin and :rep_date_end)) then
          rep_k_plan_os = rep_k_plan_os + :sh_os;
        if ((ol_date between :rep_date_begin and :rep_date_end) and
            (ok_type = 4)) then
          rep_k_fact_os = rep_k_fact_os - :ok_os;
      end

      if (k_balance > 0) then
      begin
        rep_cl_kolich = 1;
      end
      else
      begin
        rep_cl_kolich = 0;
        rep_k_period = 0;
      end
      k_id_prior = k_id;
      if (k_index_curr_code <> def_curr) then
      begin
        if (rep_cl_kolich = 1) then
        begin
          select cr_rate
          from opercurr
          where cr_date <= :rep_date_end and
                cr_date_end >= :rep_date_end and
                cr_from_cu_id = :k_index_curr_code
          into :curr_rate;
          if (pr_set_rate_in_kont = 1) then
          begin
            if (curr_rate < k_index_curr_rate) then
              curr_rate = k_index_curr_rate;
          end
          k_balance = cast(k_balance * curr_rate as decimal(15,2));
        end

        select cr_rate
        from opercurr
        where cr_date <= :rep_date_begin and
              cr_date_end >= :rep_date_begin and
              cr_from_cu_id = :k_index_curr_code
        into :curr_rate;
        if (pr_set_rate_in_kont = 1) then
        begin
          if (curr_rate < k_index_curr_rate) then
            curr_rate = k_index_curr_rate;
        end
        k_balance_begin = cast(k_balance_begin * curr_rate as decimal(15,2));
        rep_k_plan_os = cast(rep_k_plan_os * curr_rate as decimal(15,2));
        rep_k_fact_os = cast(rep_k_fact_os * curr_rate as decimal(15,2));
      end
    end

    if (rep_cl_kolich = 1) then
    begin
      kol_cli_fact = 1;
      for select kc_date_close_os
          from knclient
          where ((kc_date_disb between '01.01.1900' and (:k_disb_date - 1)) or
                (kc_date_disb = :k_disb_date and
                kc_from_k_id < :k_id)) and
                kc_date_close_os between :k_disb_date and '31.12.9999' and
                kc_from_c_id = :c_id
          into :dd_close_tmp
      do
      begin
        if (dd_close_tmp >= :rep_date_end) then
        begin
          kol_cli_fact = 0;
        end
      end
    end
    else
      kol_cli_fact = 0;

--        n=cast(rep_disb_amount/rep_k_disb_amount as decimal(15,7));
    n = 1;
    rep_balance = cast(k_balance * n as decimal(15,2));
    rep_balance_begin = cast(k_balance_begin * n as decimal(15,2));
    if (ks_date_begin between :rep_date_begin and :rep_date_end) then
    begin
      rep_disb_cl_amount = rep_disb_amount;
      rep_disb_cl_kolich = 1;
      rep_perc_disb_annual = rep_disb_amount / 100.0;
      rep_perc_disb_annual = rep_perc_disb_annual * k_perc_annu;
    end
    rep_plan_os = cast(rep_k_plan_os * n as decimal(15,2));
    rep_fact_os = cast(rep_k_fact_os * n as decimal(15,2));
    if (rep_cl_kolich = 1) then
    begin
      rep_perc_amount_annual = cast(rep_balance as double precision) / 100.0;
      rep_perc_amount_annual = rep_perc_amount_annual * k_perc_annu;
      rep_perc_amount_effect = cast(rep_balance as double precision) / 100.0;
      rep_perc_amount_effect = rep_perc_amount_effect * p_perc_eff;
      rep_min_disb = rep_disb_amount;
      rep_max_disb = rep_disb_amount;
    end
    else
    begin
      rep_perc_amount_annual = 0;
      rep_perc_amount_effect = 0;
      rep_min_disb = 999999999999;
      rep_max_disb = 0;
      rep_disb_amount = 0;
      rep_balance = 0;
      rep_k_period = 0;
    end
    if (rep_oblast = 'г.ош') then
      rep_oblast = 'ошская область';

    lkc_balance = null;
    select lkc_balance
    from llc_main, llc_kont_cli
    where lm_date_status between :rep_date_end - 4 and :rep_date_end + 4 and
          lm_id = lkc_from_lm_id and
          lkc_from_k_id = :k_id and
          lkc_from_c_id = :c_id
    into lkc_balance;

    suspend;
  end
