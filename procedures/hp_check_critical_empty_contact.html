CREATE PROCEDURE HP_CHECK_CRITICAL_EMPTY_CONTACT(
    REP_OBJECT_KIND VARCHAR(100),
    REP_OBJECT_ID DECIMAL(18,0),
    REP_FILTER VARCHAR(100),
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_COUNT INTEGER,
    RES_ERROR_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable cm_database_id integer;
begin
-- Выдает количество критичный контактов которые не заполнены
  if(rep_object_id = 109010183200400101) then exit;
  if(rep_object_id = 109010199564600101) then exit;
  if(rep_object_id = 109010196226600101) then exit; --от чолпон 21,10,22
  if(rep_object_id = 109010209366000101) then exit; --от чолпон 02,11,22
  if(rep_object_id = 109010205351400101) then exit; --от Чолпон 11.11.22
  :cm_database_id = f_get_db_id();
--  'staff_critical' для сотрудников
--  'client_critical' для клиентов
  res_error_text = '';
  select count(*)
  from aarep_object_continfo(:rep_object_kind, :rep_object_id, 0, :rep_filter, :un_id)
  into res_count;
  if (res_count is null) then
    res_count = 0;

  if (res_count > 0) then
  begin
    if (rep_filter = 'staff_critical') then
    begin
      res_error_text = 'Ваши контактные данные не подтверждены.' || ascii_char(13) || ascii_char(10) || 'Вам необходимо пройти верификацию.' || ascii_char(13) || ascii_char(10) || 'Убедительная просьба, не откладывать этот процесс.' || ascii_char(13) || ascii_char(10) || 'Иначе, для Вас доступ на некоторые полезные сервисы и доступы будут ограничены.' || ascii_char(13) || ascii_char(10) || '' || ascii_char(13) || ascii_char(10) || 'В следующем окне нажмите кнопку "Инструкция" для ознакомления этапами верификации контактных данных.';
    end
    if (rep_filter = 'client_critical') then
    begin
      res_error_text = 'Контактные данные клиентов не подтверждены.' || ascii_char(13) || ascii_char(10) || 'Необходимо провести верификацию.' || ascii_char(13) || ascii_char(10) || 'Без верификации невозможно выдать кредит.' || ascii_char(13) || ascii_char(10) || '' || ascii_char(13) || ascii_char(10) || 'В следующем окне нажмите кнопку "Инструкция" для ознакомления этапами верификации контактных данных.';
    end
    if (res_error_text = '') then
      res_error_text = 'Некоторые контактные данные не подтверждены.';
  end
  suspend;
end
