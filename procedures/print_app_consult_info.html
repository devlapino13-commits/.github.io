CREATE PROCEDURE PRINT_APP_CONSULT_INFO(
    ARC_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_PASS_NO VARCHAR(30),
    C_PASS_WHO D_STR_255,
    C_PASS_DATE D_INTEGER,
    AMOUNT D_CURRENCY,
    DOC_NO_TMP D_INTEGER,
    OL_DATE_TMP DATE,
    QR_CODE D_STR_LONG)
AS
declare variable CDS_DATETIME timestamp;
declare variable C_ID bigint;
declare variable EDL_ID integer;
begin

  select c.c_name_fio, cds.cds_doc_no, cds.cds_doc_issued_by, extract(year from cds.cds_doc_date_begin),
         cds.cds_datetime, arci.arci_amount, c.c_id
  from app_return_consult arc
  inner join app_return_consult_info arci on arci.arci_id = arc.arc_id
  inner join clients c on arc.arc_from_c_id = c.c_id
  inner join clientdocs cds on cds.cds_from_c_id = arc.arc_from_c_id and
        current_date between cds.cds_doc_date_begin and coalesce(cds.cds_doc_date_end, '31.12.2999') and
        cds.cds_is_delete = 0
  inner join passkind pk on pk.pk_id = cds.cds_from_pk_id and
        pk.pk_passport_issuers containing 'PassportIssuers'
  where arc.arc_id = :arc_id
  order by cds.cds_datetime desc rows 1
  into :c_name_fio, :c_pass_no, :c_pass_who, :c_pass_date, :cds_datetime, :amount, :c_id;

  select ol.ol_date, ol.ol_doc_no
  from operkontract ok
  inner join operlist ol on ol.ol_id = ok.ok_from_ol_id and
        ol.is_delete = 0 and
        ol.ol_document_client = :c_id
  order by 1 desc rows 1
  into :ol_date_tmp, doc_no_tmp;

  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(:add_params, 'edl_id', null);
    qr_code = f_edoc_qr_by_edl(:edl_id, :c_id);
  end

  suspend;
end
