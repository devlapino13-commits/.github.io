CREATE PROCEDURE CB_AZ_MO_2026(
    REP_REGION_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME STR_100,
    BALANCE_BEGIN DECIMAL(18,2),
    REPAY_SUM_NOT_OS DECIMAL(18,2),
    DELINQ_OS_PERIOD DECIMAL(18,2),
    BALANCE_DELINQ30_END DECIMAL(18,2),
    PLAN_PORTFOLIO DECIMAL(18,2),
    FACT_PORTFOLIO DECIMAL(18,2),
    ZNACH_PAR30 DECIMAL(18,2),
    PLAN_PERC_DONE DECIMAL(18,2),
    BONUS_FACTOR_1 DECIMAL(18,1),
    BONUS_FACTOR_2 DECIMAL(18,1),
    SUM_VOZNOG_OFFICE DECIMAL(18,2),
    BONUS_MO DECIMAL(18,2),
    MANAGER_NAME STR_100)
AS
declare variable REG_ID D_ID;
declare variable REP_LASTDAY_PREV_MONTH D_DATE;
declare variable REP_FIRST_DAY_MONTH D_DATE;
declare variable MANAGER_ID D_ID;
declare variable MO_QTY D_SMALLINT;
declare variable LO_STAGE_MONTH_FULL D_CURRENCY;
declare variable PS_DATE_BEGIN D_DATE;
declare variable V_BONUS_MEMO STR_255;
begin
  rep_lastday_prev_month = rep_date_end - extract(day from rep_date_end);
  rep_first_day_month = rep_lastday_prev_month + 1;

  for select oks_region_id,
      sum(iif((:rep_date_end - oks_delinq_date > 30) and (:rep_date_end - oks_delinq_date <= 360), oks_balance, 0)),
      sum(iif((:rep_date_end - oks_delinq_date <= 360), oks_balance, 0))
  from oksummary
  inner join region_list(:rep_region_id, 1, -1, 1) on oks_region_id = id_list
  where :rep_date_end between oks_date_begin and oks_date_end and
        oks_balance > 0 and
        oks_close_state = 0
  group by oks_region_id
  into :reg_id, :balance_delinq30_end, :fact_portfolio
  do
  begin
    manager_name = null;
    manager_id = null;
    v_bonus_memo = '';
    lo_stage_month_full = 0;
    mo_qty = 0;
    plan_portfolio = 0;
    balance_begin = 0;

    -- 2.1 кп на начало
    select coalesce(sum(oks_balance), 0)
    from oksummary
    where oks_region_id = :reg_id and
          :rep_lastday_prev_month between oks_date_begin and oks_date_end
    into :balance_begin;

    -- 2.2 par30-360
    if (balance_begin > 0) then
      znach_par30 = round(cast(balance_delinq30_end as double precision) / balance_begin * 100, 2);
    else
      znach_par30 = iif(balance_delinq30_end > 0, 100, 0);

    bonus_factor_1 =
      case
        when znach_par30 < 0.8  then 1.20
        when znach_par30 <= 1.0 then 1.10
        when znach_par30 <= 1.5 then 1.00
        when znach_par30 <= 2.0 then 0.70
        when znach_par30 <= 2.5 then 0.60
        when znach_par30 <= 4.0 then 0.40
        else 0.10
      end;

    -- 2.3 планирование
    select coalesce(sum(o.po_calc), 0)
    from planoper o
    inner join planmain m on o.po_from_pm_id = m.pm_id
    where m.pm_from_reg_id = :reg_id and
         (o.po_period = cast(extract(year from :rep_date_end) || right('0' || extract(month from :rep_date_end), 2) as integer)
          or o.po_date = :rep_lastday_prev_month)
    into :plan_portfolio;

    if (plan_portfolio > 0) then
      plan_perc_done = round((fact_portfolio / plan_portfolio) * 100, 2);
    else
      plan_perc_done = 0;

    bonus_factor_2 =
      case
        when plan_perc_done >= 103  then 1.20
        when plan_perc_done >= 101  then 1.10
        when plan_perc_done >= 100  then 1.00
        when plan_perc_done >= 99.5 then 0.80
        when plan_perc_done >= 99.0 then 0.70
        when plan_perc_done >= 98.5 then 0.55
        else 0.40
      end;

    -- 2.5 финансы
    select coalesce(-sum(ok_perc + ok_fine + ok_kommis), 0) 
    from operkred
    where ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_ol_reg_id = :reg_id and
          ok_ol_date between :rep_first_day_month and :rep_date_end
    into :repay_sum_not_os;

    select coalesce(sum(ok_os + ok_sh_os), 0) 
    from oksummary oks
    inner join operkred ok on oks.oks_from_k_id = ok.ok_from_k_id
    where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_region_id = :reg_id and
          :rep_date_end >= oks.oks_delinq_date and
          oks.oks_close_state = 0 and
          ok.ok_ol_is_active = 1 and
          ok.ok_type < 7 and
          ok.ok_sh_date between :rep_first_day_month and :rep_date_end and
          ok.ok_ol_date <= :rep_date_end
    into :delinq_os_period;

    sum_voznog_office = (repay_sum_not_os / 100) - (delinq_os_period / 100 * 5);

    -- 3. кадры: поиск менеджера
    for select ps.ps_from_c_id, ps.ps_date_begin
    from hr_pstatus ps
    inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
    inner join jobkind jk on jl.jl_kind = jk.jk_id
    where :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
          ps.ps_status = 1 and
          ps.ps_is_working = 1 and
          ps.ps_from_reg_id = :reg_id and
          jk.jk_name in ('руководитель филиала', 'kredit meneceri') and
          jk.jk_system_code = 'jobsystem.region_km'
    order by ps.ps_date_begin
    into :manager_id, :ps_date_begin
    do
    begin
      if (mo_qty = 0 and ps_date_begin < :rep_first_day_month + 15) then
      begin
        mo_qty = 1;
        
        select c_name_fio
        from clients 
        where c_id = :manager_id
        into :manager_name;
        
        select lo_stage_month_full
        from hp_get_lo_stage_full(:manager_id, :rep_date_end, 'jobsystem.region_km', 0)
        into :lo_stage_month_full;
        
        select bonus_memo
        from hp_get_vigovor_koef(:manager_id, :rep_date_end)
        where bonus_koef < 100 or bonus_sum > 0
        into :v_bonus_memo;
      end
    end

    -- 4. итоговый расчет
    bonus_mo = round(:sum_voznog_office * :bonus_factor_1 * :bonus_factor_2, 2);
    
    if (bonus_mo < 0 or (manager_name is not null and (coalesce(lo_stage_month_full, 0) < 1 or coalesce(v_bonus_memo, '') <> ''))) then
      bonus_mo = 0;

    select reg_code || ' ' || reg_name
    from region 
    where reg_unicode = :reg_id 
    into :reg_name;
    
    suspend;
  end
end
