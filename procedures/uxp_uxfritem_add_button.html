CREATE PROCEDURE UXP$UXFRITEM_ADD_BUTTON(
    UX_UUID D_UUID,
    CURRENT_UFI_UUID D_UUID,
    UA_UUID D_UUID)
RETURNS (
    UFI_UUID_STR D_STR_CODE)
AS
declare variable ufi_position d_integer;
declare variable ufi_parent_uuid d_uuid;
declare variable ufi_kind d_integer;
declare variable ufi_uuid d_uuid;
declare variable ua_name d_str_100;
declare variable ua_hint d_str_255;
begin
  -- Добавляет один AUX элемент или группу

  if (trim(coalesce(current_ufi_uuid, '')) <> '') then
  begin
    select ufi_position, ufi_parent_uuid, ufi_kind
    from ux$fritem
    where ufi_owner = :ux_uuid and
          ufi_uuid = :current_ufi_uuid
    into ufi_position, ufi_parent_uuid, ufi_kind;

    if (ufi_position is not null) then
    begin
      ufi_position = ufi_position + 1;
      if (ufi_kind = -1) then
      begin
        ufi_parent_uuid = current_ufi_uuid;
        ufi_position = null;
      end
    end
  end

  select ua_name, ua_hint
  from ux$action
  where ua_uuid = :ua_uuid
  into ua_name, ua_hint;

  if (coalesce(ua_hint, '') = '') then
    ua_hint = ua_name;

  insert into ux$fritem (ufi_owner, ufi_control, ufi_kind, ufi_parent_uuid, ufi_position, ufi_name, ufi_hint)
  values (:ux_uuid, 'button', 2, :ufi_parent_uuid, :ufi_position, :ua_name, :ua_hint)
  returning ufi_uuid
  into ufi_uuid;

  insert into ux$frbutton (ufb_ufi_uuid, ufb_ua_uuid)
  values (:ufi_uuid, :ua_uuid);

  ufi_uuid_str = uuid_to_char(ufi_uuid);
  suspend;
end
