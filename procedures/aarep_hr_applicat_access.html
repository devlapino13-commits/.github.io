CREATE PROCEDURE AAREP_HR_APPLICAT_ACCESS(
    USER_ID DECIMAL(18,0),
    REG_ID INTEGER)
RETURNS (
    KSR_ID INTEGER)
AS
declare variable un_approve integer;
begin
  ksr_id = 0;

  select has_access
  from hp_permission_access_check(:user_id, 'PermActions', 0, 'FinStart.User_Admin_Approve')
  into :un_approve;
  if (un_approve = 0) then
    exit;

  select distinct ksr_id
  from kontsignreg, kontsignmen, region_list(:reg_id, 1, -1, 0), username
  where (ksr_from_reg_id = id_list) and
        ksm_from_c_id = un_from_c_id and
        un_id = :user_id and
        ksr_is_active = 1 and
        ksr_from_ksm_id = ksm_id and
        coalesce(ksr_date_begin, '01.01.1990') <= current_date and
        coalesce(ksr_date_end, '31.12.9999') >= current_date rows 1
  into :ksr_id;
  if (ksr_id is null) then
    ksr_id = 0;
  if (un_approve is null) then
    un_approve = 0;
  if ((ksr_id <> 0) and
      (un_approve <> 0)) then
    suspend;
end
