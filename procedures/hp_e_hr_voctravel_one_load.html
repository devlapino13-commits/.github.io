CREATE PROCEDURE HP_E_HR_VOCTRAVEL_ONE_LOAD(
    IN_ORDER_ID D_ID)
RETURNS (
    HRV_FROM_HO_ID D_ID,
    RES_CAN_EDIT D_BOOL,
    RES_IS_SIGNED D_BOOL,
    HRV_NUMBER D_INTEGER,
    HRV_FROM_C_ID ID_TYPE,
    HRV_DATE_BEGIN D_DATE,
    HRV_DATE_END D_DATE,
    HRV_DAYS D_INTEGER,
    HRV_DAY_BACK D_DATE,
    HRV_DIRECTION D_STR_100,
    HRV_STATUS D_INTEGER,
    HRV_CHANGER_ID D_ID,
    HRV_NOT_CLOSE_ACCESS D_BOOL,
    HRV_DATE_END_FACT D_DATE,
    RES_ACTUAL_OFFICE D_STR_100,
    RES_ACTUAL_JOBLIST D_STR_255,
    RES_ORDER_TYPE D_INTEGER,
    RES_ACTUAL_REG_ID D_INTEGER)
AS
begin
  select ho_order_type
  from hr_orders
  where ho_id = :in_order_id
  into res_order_type;

  for select hrv_from_ho_id, hrv_number, hrv_from_c_id, hrv_date_begin, hrv_date_end, hrv_days, hrv_day_back,
             hrv_direction, hrv_status, hrv_date_end_fact, hrv_changer_id, hrv_not_close_access
  from hr_voctravel
  where hrv_from_ho_id = :in_order_id
  into :hrv_from_ho_id, :hrv_number, :hrv_from_c_id, :hrv_date_begin, :hrv_date_end, :hrv_days, :hrv_day_back,
       :hrv_direction, :hrv_status, :hrv_date_end_fact, :hrv_changer_id, :hrv_not_close_access
  do
  begin
    res_can_edit = 1;
    res_is_signed = 0;
  
    res_actual_office = '';
    res_actual_joblist = '';
    if (hrv_from_c_id is not null) then
    begin
      select res_actual_reg_id, res_actual_office, res_actual_joblist
      from hp_e_hr_orders_pers_info(:hrv_from_c_id, :hrv_date_begin)
      into :res_actual_reg_id, :res_actual_office, :res_actual_joblist;
    end

    if (hrv_not_close_access is null) then
      hrv_not_close_access = 0;
  
    suspend;
  end
end
