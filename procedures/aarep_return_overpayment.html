CREATE PROCEDURE AAREP_RETURN_OVERPAYMENT(
    K_ID ID_TYPE,
    C_ID ID_TYPE)
RETURNS (
    RES_K_NO STR_20,
    RES_K_DATE DATE,
    RES_C_NAME STR_100,
    RES_ADDRESS STR_255,
    RES_CELL STR_100,
    RES_CM_NAME STR_255,
    EDOC_DATE_CREATE D_DATE)
AS
begin
  select
    k_kontract_no,
    k_kontract_date
  from kontract
  where k_id = :k_id
  into :res_k_no, :res_k_date;

  select
    c_name_fio
  from clients
  where c_id = :c_id
  into :res_c_name;

  select
    adl_address_str
  from addrclient
    inner join addrlist
      on adl_id = adc_from_adl_id
  where adc_from_c_id = :c_id
    and adc_address_type = 1
  into :res_address;

  select cast(edl_create_datetime as date)
  from edoc_list
  inner join edoc_cfg on edc_id = edl_from_edc_id
  where edl_object_kind = 9 and
        edl_object_id = :k_id and
        edc_code = 'loan.agreement' and
        edl_status <> -1
  order by edl_create_datetime desc rows 1
  into edoc_date_create;

  res_cell = f_cont_info('clients', c_id, 'cell_main', 'last');

  select
    cm_name
  from company
  into :res_cm_name;

  suspend;
end
