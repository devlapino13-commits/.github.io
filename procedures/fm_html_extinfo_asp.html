CREATE PROCEDURE FM_HTML_EXTINFO_ASP(
    EIL_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable LANG D_STR_LANG;
declare variable C_ID D_ID;
declare variable EIL_DATETIME D_DATETIME;
declare variable C_NAME_FIO D_STR_255;
declare variable CALC_STATUS D_STR_2048;
declare variable JSON_DOCUMENT D_BLOB_TEXT;
declare variable IIN D_STR_20;
declare variable SDDFAMILYMEMBERS D_CURRENCY;
declare variable RESULT D_STR_255;
begin
  -- RFCRU-3940 Справка АСП
  -- отображение полученных справок от сервиса 
  lang = f_get_lang_code(un_id, null);

  for select eil.eil_status_datetime, res_blob, eil_from_c_id
  from extinfolist eil, extinfocfg eic, fm_blob_get(eil_from_bls_id)
  where eil_id = :eil_id and
        eil_from_eic_id = eic_id and
        eil_is_json = 1 and
        eil_status = 2
  rows 1
  into eil_datetime, :json_document, c_id
  do
  begin
    c_name_fio = f_lang(lang, 'Клиент не найден');
    calc_status = '';

    if (row_count = 0) then
      calc_status = f_lang(lang, 'Справка АСП не получена');
    else
    begin
      if (coalesce(:json_document, 0) = 0) then
        calc_status = f_lang(lang, 'Информация о доходах по клиенту отсутствует.');
    end
  
    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;
    res_html = (select res_style
                from fm_html_styles) || f_format('<body><a name="top"></a><h2>%1</h2>%2<br>', c_name_fio, calc_status);

    res_html = BLOB_APPEND(res_html, f_format('<h1>%1</h1>', 'Справка о АСП'));
    result = f_json_str_get_value(:json_document, 'state', '');

    sddFamilyMembers = 0;
    if (result = 'ASP') then
    begin
      sddFamilyMembers = cast(f_json_str_get_value(:json_document, 'sddFamilyMembers', '') as d_currency);
    end
    iin = f_json_str_get_value(:json_document, 'iin', '');

    res_html = BLOB_APPEND(res_html, '<table><table width="100%"><col width="40%"><col width="60%">');
    res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Ф.И.О заявителя') || '</b>', :c_name_fio, '', '', -1, -1, -1, -1, 2, ''));
    res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'ИИН') || '</b>', :iin, '', '', -1, -1, -1, -1, 2, ''));
    res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Средняя сумма выплат') || '</b>', :sddFamilyMembers, '', '', -1, -1, -1, -1, 2, ''));
    res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Дата и время получения ответа Компанией') || '</b>', f_datetimetostr(eil_datetime), '', '', -1, -1, -1, -1, 2, ''));
    res_html = BLOB_APPEND(res_html, '</table>');
    res_html = BLOB_APPEND(res_html, '<br>');
  end

  res_html = BLOB_APPEND(res_html, '</body>');
  suspend;
end
