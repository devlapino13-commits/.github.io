CREATE PROCEDURE HP_CORRECT_ONE_PERCENT(
    K_ID DECIMAL(18,0),
    OL_DATE DATE,
    OL_PERC_AMOUNT DECIMAL(18,4),
    OK_WITH_TAX INTEGER)
AS
declare variable ok_sh_id integer;
declare variable ok_sh_date date;
declare variable ok_sh_perc decimal(18,4);
declare variable kv integer;
declare variable ok_number integer;
declare variable ol_id decimal(18,0);
declare variable ok_tax0 decimal(18,4);
declare variable ok_tax1 decimal(18,4);
begin
  /* Procedure Text */
  if (ol_perc_amount > 0) then
  begin
    ok_number = 0;
    ok_tax0 = 0;
    if (ok_with_tax = 1) then
    begin
      ok_tax0 = cast(:ol_perc_amount / 9 as decimal(15,0));
    end
    select kv
    from get_kv(1010100, 5)
    into :ol_id;
    insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                          ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                          ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                          ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month)
    values (:ol_id, 1010100, 1, :ol_date, null, '', 6, 'IMPORT: Процент по кредиту', null, 0, null,
            1000000000001, null, null, 0, null, null, null, 1, :ol_date, 0, null, 0, null, null);

    for select ok_sh_id, ok_sh_date, sum(ok_perc + ok_sh_perc + ok_tax)
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :k_id and
              ok_from_c_id = 0
        group by ok_sh_id, ok_sh_date
        having sum(ok_perc + ok_sh_perc + ok_tax) <> 0
        order by ok_sh_id, ok_sh_date
        into :ok_sh_id, :ok_sh_date, :ok_sh_perc
    do
    begin
      if ((ol_perc_amount + ok_tax0) > 0) then
      begin
        ok_number = :ok_number + 1;
        if (ol_perc_amount + ok_tax0 <= :ok_sh_perc) then
        begin
          insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                                ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
          values (:ol_id, :ok_number, :k_id, :ok_sh_id, :ok_sh_date, 4, 0, -1 * :ol_perc_amount, 0, 1, 0, 0, 0, 0,
                  -1 * :ok_tax0);
          ol_perc_amount = 0;
          ok_tax0 = 0;
        end
        else
        begin
          ok_tax1 = cast(:ok_sh_perc * 0.1 as decimal(15,0));
          insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                                ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
          values (:ol_id, :ok_number, :k_id, :ok_sh_id, :ok_sh_date, 4, 0, -1 * (:ok_sh_perc - :ok_tax1), 0, 1, 0, 0, 0,
                  0, -1 * :ok_tax1);
          ol_perc_amount = :ol_perc_amount - (:ok_sh_perc - :ok_tax1);
          ok_tax0 = :ok_tax0 - :ok_tax1;
        end
      end
    end
  end
end
