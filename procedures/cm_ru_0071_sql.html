CREATE PROCEDURE CM_RU_0071_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable kc_from_c_id id_type;
declare variable re_k_id id_type;
declare variable reg_id integer;
declare variable reg_parent integer;
declare variable parent_name varchar(255);
declare variable re_disb_date date;
declare variable re_kredline_end date;
declare variable par_re_disb_date date;
declare variable kc_circle integer;
declare variable cl_par_max_delinq_day integer;
declare variable cl_par_sum_delinq_day integer;
declare variable max_delinq_day integer;
declare variable sum_delinq_day integer;
declare variable min_sum decimal(18,0);
declare variable max_sum decimal(18,0);
declare variable max_kc_amount decimal(18,0);
declare variable close_month3 integer;
declare variable close_month6 integer;
declare variable close_month3_c3 integer;
declare variable pr_code varchar(100);
begin
  for select kc_from_c_id,
             kc_amount_approved,
             k_disb_date,
             k_last_reg_id,
             pr_code
      from knclient kc, kontract, product
      where kc_amount_approved > 0
      and kc_from_k_id = :k_id
      and kc_from_k_id = k_id
      and k_from_pr_id = pr_id
      and pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.|.LOYAL.)%'
      and pr_code not containing '.PATENT.'
      and pr_code not containing '.PILOT.'
  into kc_from_c_id,
       kc_amount,
       kf_disb_date,
       reg_id,
       pr_code
  do
  begin                                      --1590000
    if (res_info_text is null) then --Чтобы каждому клиенту не работал скрипт по определению региона
    select distinct r.parent, r2.reg_name
    from region r, region r2
    where r.is_delete = 0
    and r.parent = r2.reg_unicode
    and r.reg_unicode = :reg_id
    into reg_parent, parent_name;
    min_sum = 15000;
    max_sum = 25000;
    res_info_text = '';
    select c_name_fio
    from clients
    where c_id = :kc_from_c_id
    into res_info_text;
    select count(*) kc_circle
    from knclient, kontstate
    where kc_from_c_id = :kc_from_c_id
      and kc_amount_approved > 0
      and kc_from_k_id = ks_from_k_id
      and ks_ol_is_delete = 0
      and ks_state = 1
      and ks_date_begin < :kf_disb_date
    into kc_circle;
    kc_circle = coalesce(kc_circle+1, 1);
    if (kc_circle > 1) then
    begin
      --проверяется предыдущий цикл клиента на просрочку
      re_k_id = null;
      select kc_from_k_id,
             ks_date_begin,
             k_kredline_end
      from knclient, kontstate, kontract, product
      where kc_from_c_id = :kc_from_c_id
      and kc_amount_approved > 0
      and kc_from_k_id = ks_from_k_id
      and ks_ol_is_delete = 0
      and ks_state = 1
      and ks_from_k_id <> :k_id
      and ks_date_begin < :kf_disb_date
      and ks_from_k_id = k_id
      and k_from_pr_id = pr_id
      and pr_code not containing '.PAR.'
      order by ks_date_begin desc
      rows 1
      into re_k_id,
           re_disb_date,
           re_kredline_end;
        
      if (re_k_id is null) then
      begin
        kc_circle = 1;
      end else
      begin
        select
          max(iif(dateadd(month, 3, ks_date_begin)-10<=iif(ks_date_end >= :kf_disb_date, :kf_disb_date, ks_date_end+1),1,0)
              ) as close_month3,
          max(iif(dateadd(month, 6, ks_date_begin)-10<=iif(ks_date_end >= :kf_disb_date, :kf_disb_date, ks_date_end+1),1,0)
              ) as close_month6,
          max(kc_amount_approved) as max_kc_amount,
          max(iif(kc_circle+3 >= :kc_circle,
                  iif(dateadd(month, 3, ks_date_begin)-10<=iif(ks_date_end >= :kf_disb_date,
                      :kf_disb_date, ks_date_end+1),1,0),0)) as close_month3_c3
        from knclient, kontstate, kontract, product
          where kc_amount_approved > 0
          and kc_from_c_id = :kc_from_c_id
          and kc_from_k_id = ks_from_k_id
          and ks_ol_is_delete = 0
          and ks_state = 1
          and ks_from_k_id <> :k_id
          and ks_date_begin < :kf_disb_date
          and ks_from_k_id = k_id
          and k_from_pr_id = pr_id
          and pr_code not containing '.PAR.'
        into close_month3, close_month6, max_kc_amount, close_month3_c3;
        close_month3 = coalesce(close_month3, 0);
        close_month6 = coalesce(close_month6, 0);
        --вычисляется макс кол-во просроченных дней
        max_delinq_day = 0;
        sum_delinq_day = 0;
        with tabl1 as (
            select distinct ok_ol_date,
                   ok_ol_date - ok_sh_date as delinq_day
            from operkred
            where ok_ol_is_active = 1
            and ok_from_k_id = :k_id
            and ok_ol_date > ok_sh_date
            and ok_os + ok_perc < 0
            and ok_type in (4,5,9)
            )
        select max(delinq_day), sum(delinq_day) from tabl1
        into max_delinq_day, sum_delinq_day;
        if (max_delinq_day is null) then max_delinq_day = 0;
        if (sum_delinq_day is null) then sum_delinq_day = 0;
    
        --проверяется предыдущий параллельный цикл клиента на просрочку
        re_k_id = null;
        select kc_from_k_id, ks_date_begin
          from knclient, kontstate
        where kc_amount_approved > 0
          and kc_from_c_id = :kc_from_c_id 
          and kc_from_k_id <> :k_id
          and kc_from_k_id = ks_from_k_id
          and ks_ol_is_delete = 0
          and ks_state = 1
          and ks_date_begin between :re_disb_date and :kf_disb_date
          order by ks_date_begin desc
          rows 1
        into re_k_id, par_re_disb_date;
    
        --вычисляется макс кол-во просроченных дней
        cl_par_max_delinq_day = 0;
        cl_par_sum_delinq_day = 0;
        with tabl1 as (
            select distinct ok_ol_date,
                   ok_ol_date - ok_sh_date as delinq_day
            from operkred
            where ok_ol_is_active = 1
            and ok_from_k_id = :k_id
            and ok_ol_date > ok_sh_date
            and ok_os + ok_perc < 0
            and ok_type in (4,5,9)
            )
        select max(delinq_day), sum(delinq_day) from tabl1
        into cl_par_max_delinq_day, cl_par_sum_delinq_day;
        if (cl_par_max_delinq_day is null) then cl_par_max_delinq_day = 0;
        if (cl_par_sum_delinq_day is null) then cl_par_sum_delinq_day = 0;
    
        if (max_delinq_day < cl_par_max_delinq_day and re_disb_date <= par_re_disb_date) then
          max_delinq_day = cl_par_max_delinq_day;
    
        if (sum_delinq_day < cl_par_sum_delinq_day and re_disb_date <= par_re_disb_date) then
          sum_delinq_day = cl_par_sum_delinq_day;
      end
    end 
    if (kc_amount > max_sum) then
    begin
      if ((kf_disb_date <= '22.08.16'
           and reg_parent in(1010004,    --Уральский регион
                             1420001,    --Северный регион
                             1590000))     --Центральный регион
           or
           (kf_disb_date > '22.08.16'
            and reg_parent in(1010004,    --Уральский регион
                              1590000)     --Центральный регион
            and pr_code not containing '.kazan.'
                        )) then
      begin                                                             
        if (kc_circle > 1) then
        begin
          if (kc_amount > 40000) then
          begin
            if (max_kc_amount >= 40000 and close_month3_c3 = 1 and sum_delinq_day < 15 and kc_circle > 3) then
            begin
              max_sum = 60000;
            end
            else
            begin     
              max_sum = 40000;
              if (close_month3_c3 = 0) then
              begin 
                res_error_text = 'Клиент пользовался микрозаймами менее 3-мес. в последних 3х займах и не может получить более: '||max_sum;
                res_error_is_critical = 1;
                suspend;
              end
              if (sum_delinq_day > 14) then
              begin 
                res_error_text = 'Суммарная просрочка превышает 14-дней ('||sum_delinq_day||' д.) и не может получить: '||kc_amount;
                res_error_is_critical = 1;
                suspend;
              end
              if (max_kc_amount < max_sum
                  and pr_code not containing '.kazan.' ) then
              begin
                res_error_text = parent_name||'. Клиент ранее получал максимально '||:max_kc_amount||' р., и может получить до '||:max_sum||' руб.';
                res_error_is_critical = 1;
                suspend;
              end
            end
          end
          if (kc_amount between 25001 and 40000) then
          begin
            if (close_month3_c3 = 1) then
            begin
              max_sum = 40000;
            end   else
            begin
              max_sum = 25000;
              begin 
                res_error_text = 'Клиент пользовался микрозаймами менее 3-мес. в последних 3х займах и не может получить более: '||max_sum;
                res_error_is_critical = 1;
                suspend;
              end
            end
          end
        end
        if (kc_circle <= 1 or max_delinq_day > 7) then max_sum = 25000;
      end
      if (reg_parent in(1010005 --Сибирский регион
                        )) then
      begin                                               
        if (kc_circle > 1) then
        begin
            if (close_month3 = 1) then
            begin
              max_sum = 60000;
            end else
            begin
              max_sum = 30000;
              if (kc_amount > max_sum) then
              begin 
                res_error_text = 'Клиент пользовался микрозаймами менее 3-мес. и не может получить более: '||max_sum;
                res_error_is_critical = 1;
                suspend;
              end
            end
        end
        if (kc_circle <= 1 or max_delinq_day > 7) then max_sum = 30000;
      end
      if ((kf_disb_date <= '22.08.16' and reg_parent in(1010000) --Московский регион
           or(kf_disb_date > '22.08.16' and reg_parent in(1010000, 1010001, 1010002, --Московский регион
                                                          1420001)    --Северный регион
                        ))) then
      begin                                               
        if (kc_circle > 1) then
        begin                                           
          if (kc_amount > 40000) then
          begin
            if(close_month6 = 1) then
            begin
              max_sum = 60000;
            end else
            begin
              max_sum = 45000;
                               
              if (kc_amount > max_sum) then
              begin
                res_error_text = 'Клиент пользовался микрозаймами менее 6-мес. и не может получить более: '||max_sum;
                res_error_is_critical = 1;
                suspend;
              end
            end
            if(sum_delinq_day < 15) then
            begin
              max_sum = 60000;
            end else
            begin
              max_sum = 45000;      
              if (kc_amount > max_sum) then
              begin
                res_error_text = 'Суммарная просрочка превышает 14-дней ('||sum_delinq_day||' д.) и не может получить: '||kc_amount;
                res_error_is_critical = 1;
                suspend;
              end
            end
          end
                                               
          if (kc_amount between 30001 and 45000) then
          begin
            if(close_month3 = 1) then
            begin
              max_sum = 45000;
            end else
            begin
              max_sum = 30000;
                                    
              if (kc_amount > max_sum) then
              begin
                res_error_text = 'Клиент пользовался микрозаймами менее 3-мес. и не может получить более: '||max_sum;
                res_error_is_critical = 1;
                suspend;
              end
            end
          end
        end
        if (kc_circle <= 1 or max_delinq_day > 7) then max_sum = 30000;
      end
      if (reg_parent in(1010003)) then --Дальневосточный регион
      begin                                                
        if (kc_circle > 1 and max_kc_amount between 30001 and 60000) then
        begin 
            if (close_month3 = 1) then
            begin
              max_sum = 60000;
            end   else
            begin
              max_sum = 35000;
                               
              if (kc_amount > max_sum) then
              begin 
                res_error_text = 'Клиент пользовался микрозаймами менее 3-мес. и не может получить более: '||max_sum;
                res_error_is_critical = 1;
                suspend;
              end
            end
        end
        if (kc_circle <= 1 or max_delinq_day > 7) then max_sum = 35000;
      end   
    end
                   
    if (kc_amount > max_sum) then
    begin         
      if (kc_circle <= 1) then
      begin
        res_error_text = parent_name||'. Клиент новый и не может получить более: '||max_sum;
        res_error_is_critical = 1;
        suspend;
      end       
      if (max_delinq_day > 7) then
      begin
        res_error_text = parent_name||'. Клиент допустил просрочку более 7 дней ('||max_delinq_day||'д.) и может получить до: '||:max_sum;
        res_error_is_critical = 1;
        suspend;
      end
       
    end
  end 
end
