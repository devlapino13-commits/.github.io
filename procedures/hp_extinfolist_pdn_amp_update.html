CREATE PROCEDURE HP_EXTINFOLIST_PDN_AMP_UPDATE(
    KCP_ID D_ID,
    KC_ID D_ID,
    CALC_DATE D_DATE,
    JSO D_INTEGER)
AS
declare variable uid d_str_code;
declare variable kbki_ogrn d_str_20;
declare variable bki_ogrn d_str_20;
declare variable kpa_calc_date d_date;
declare variable kpa_sum d_currency;
declare variable currency d_str_10;
declare variable kpa_id d_id;
declare variable kbki_no d_integer;
declare variable kbki_count d_integer;
declare variable bki_no d_integer;
declare variable bki_count d_integer;
declare variable loan_no d_integer;
declare variable loan_count d_integer;
declare variable our_uid d_str_code;
declare variable kbki_jsa d_integer;
declare variable kbki_jso d_integer;
declare variable bki_jsa d_integer;
declare variable bki_jso d_integer;
declare variable loan_jsa d_integer;
declare variable loan_jso d_integer;
declare variable kbki_addinfo_jso d_integer;
declare variable our_kc_id d_id;

declare function f_my_get_bki_id (ogrn d_str_20)
returns d_id
as
  declare variable res d_id;
begin
  res = (select bct_id from bki_catalog where bct_kind = 'bki_ogrn' and bct_code = :ogrn);
  if (res is null) then
    insert into bki_catalog (bct_kind, bct_code, bct_name)
    values ('bki_ogrn', :ogrn, :ogrn)
    returning bct_id
    into res;

  return res;
end

begin
  -- процедура обновления информации по ССП СКИ по методу "одного окна" из полученного от сервиса БКИ json

  our_uid = f_uuid_cbr(f_uuid('knclient', kc_id));

  if (calc_date is null) then calc_date = current_date;

  delete from knclient_pdn_amplist
  where kpa_from_kcp_id = :kcp_id;

  -- разбор json
  kbki_no = 0;
  kbki_count = 0;
  kbki_jsa = json.obj_get_arr(jso, 'КБКИ');
  kbki_addinfo_jso = json.obj_create();

  if (kbki_jsa is not null) then
    kbki_count = json.count_get(kbki_jsa);

  while (kbki_no < kbki_count) do
  begin
    kbki_no = kbki_no + 1;
    kbki_jso = json.arr_get_obj(kbki_jsa, kbki_no - 1);
    kbki_ogrn = json.obj_get_val(kbki_jso, 'ОГРН', '');

    if (json.contains(kbki_jso, 'СубъектНеНайден') = 1) then
       json.obj_set_s(kbki_addinfo_jso, kbki_ogrn, 'СубъектНеНайден');
    else if (json.contains(kbki_jso, 'ОбязательствНет') = 1) then
       json.obj_set_s(kbki_addinfo_jso, kbki_ogrn, 'ОбязательствНет');
    else if (json.contains(kbki_jso, 'Ошибка') = 1) then
       json.obj_set_s(kbki_addinfo_jso, kbki_ogrn, 'Ошибка');

    if (json.count_get(kbki_addinfo_jso) > 0) then
      update knclient_pdn
      set kcp_kbki_addinfo = json.obj_to_json(:kbki_addinfo_jso)
      where kcp_id = :kcp_id;

    bki_no = 0;
    bki_count = 0;
    bki_jsa = json.obj_get_arr(kbki_jso, 'Обязательства.БКИ');
  
    if (bki_jsa is not null) then
      bki_count = json.count_get(bki_jsa);

    while (bki_no < bki_count) do
    begin
      bki_no = bki_no + 1;
      bki_jso = json.arr_get_obj(bki_jsa, bki_no - 1);
      bki_ogrn = json.obj_get_val(bki_jso, 'ОГРН', '');
  
      loan_no = 0;
      loan_count = 0;
      loan_jsa = json.obj_get_arr(bki_jso, 'Договор');
    
      if (loan_jsa is not null) then
        loan_count = json.count_get(loan_jsa);

      while (loan_no < loan_count) do
      begin
        loan_no = loan_no + 1;
        loan_jso = json.arr_get_obj(loan_jsa, loan_no - 1);

        uid           = json.obj_get_val(loan_jso, 'УИД', '');
        kpa_calc_date = json.obj_get_val(loan_jso, 'СреднемесячныйПлатеж.ДатаРасчета', '01.01.1900');
        kpa_sum       = json.obj_get_val(loan_jso, 'СреднемесячныйПлатеж.value', 0);
        currency      = json.obj_get_val(loan_jso, 'СреднемесячныйПлатеж.Валюта', '');

        if (our_uid = uid) then -- если делаем пересчет после выдачи, то этот же не учитываем.
          continue;

        if (currency not in ('RUB', 'RUR')) then
        begin
          if (currency = 'USD') then
            select round(:kpa_sum * curr_rate, 2)
            from get_curr_rate('840', :calc_date)
            into kpa_sum;
          else
            exception e_default 'Обнаружен валютный займ в КБКИ';
        end

        insert into knclient_pdn_amplist(kpa_from_kcp_id, kpa_uid, kpa_kbki, kpa_bki, kpa_calc_date, kpa_sum, kpa_currency)
        values (:kcp_id, :uid, f_my_get_bki_id(:kbki_ogrn), f_my_get_bki_id(:bki_ogrn), :kpa_calc_date, :kpa_sum, :currency);
      end
    end
  end
  json.destroy(kbki_addinfo_jso);

  ----------------
  for select distinct kpa_uid
  from knclient_pdn_amplist
  where kpa_from_kcp_id = :kcp_id
  into uid
  do
  begin
    our_kc_id = (select obj_id from hp_find_cbr_uuid(:uid) where obj_kind = 8);

    if (our_kc_id is null) then
    begin
      kpa_id = null;

      with t1 as (
        select kpa_id, kpa_calc_date,
          case bct_code
            when '1247700058319' then 1    -- СБ
            when '1047820008895' then 1    -- Эквифакс
            when '1057746710713' then 2    -- НБКИ
            when '1047796788819' then 3    -- ОКБ
            when '1057747734934' then 4    -- Русск стандарт
            else 99
          end as priority
        from knclient_pdn_amplist, bki_catalog
        where bct_id = kpa_kbki
          and kpa_from_kcp_id = :kcp_id
          and kpa_uid = :uid)
      select kpa_id
      from t1
      order by kpa_calc_date desc, priority
      rows 1
      into kpa_id;
  
      update knclient_pdn_amplist
      set kpa_is_actual = 1
      where kpa_id = :kpa_id;
    end
    else
      update knclient_pdn_amplist
      set kpa_from_kc_id = :our_kc_id
      where kpa_from_kcp_id = :kcp_id
        and kpa_uid = :uid;
  end
end
