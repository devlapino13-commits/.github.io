CREATE PROCEDURE HP_MORATORY_FIND_CLIENT(
    CL_FIO D_STR_100)
RETURNS (
    C_NAME_FIO D_STR_100,
    PASS_NO VARCHAR(30),
    BIRTH_DATE D_DATE,
    M_COMMENT_OFFICER D_STR_2048,
    REQ_STATUS D_STR_100,
    STATUS D_STR_20,
    M_OFFICER_NAME D_STR_100,
    REG_NAME VARCHAR(50),
    REG_PARENT_NAME VARCHAR(50),
    REQ_DATE D_DATETIME,
    REQ_DATE_CLOSE D_DATETIME,
    ACTION_AUTHOR D_STR_100,
    APPR_COMMENT D_STR_100)
AS
declare variable m_id d_id;
declare variable mal_cnt d_integer;
declare variable m_num d_integer;
declare variable reg_id_tmp d_integer;
declare variable reg_id d_integer;
declare variable un_id d_id;
declare variable un_id_tmp d_id;
declare variable un_id_tmp2 d_id;
declare variable appr_un_id d_id;
declare variable appr_un_id_temp d_id;
declare variable c_id d_id;
declare variable m_comment_officer_temp d_str_100;
begin
  un_id = 0;
  appr_un_id = 0;
  reg_id = 0;

  select c.c_id
  from clients c
  where c.c_name_fio = :cl_fio
  into c_id;

  for select m.m_client_fio, m.m_id, m.m_number, m.m_from_reg_id, m.m_client_pass_no, m.m_client_birth_date,
             m.m_comment_officer, m.m_user_create
      from moratory m
      where (m.m_client_fio = :cl_fio or
            m.m_from_c_id = :c_id)
      into :c_name_fio, :m_id, m_num, :reg_id_tmp, :pass_no, :birth_date, :m_comment_officer, un_id_tmp
  do
  begin
    if (reg_id <> reg_id_tmp) then
    begin
      reg_id = reg_id_tmp;
      select reg.reg_name, reg2.reg_name
      from region reg
      inner join region reg2 on reg.parent = reg2.reg_unicode
      where reg.reg_unicode = :reg_id
      into :reg_name, :reg_parent_name;
    end

    mal_cnt = 1;
    while (:mal_cnt >= 0) do
    begin
      req_status = case mal_cnt
                     when 1 then 'Добавить в черный список'
                     when 0 then 'Удалить из черного списка'
                   end;
      reg_id = 0;
      status = null;
      un_id_tmp2 = null;
      appr_un_id_temp = null;
      req_date_close = null;
      req_date = null;

      with zpr
      as (select *
          from morapprlist mal
          where mal.mal_from_m_id = :m_id and
                mal.mal_from_m_number = :m_num)

      select
             case zpr.mal_appr_status
               when 0 then 'В процессе'
               when 1 then 'Утвержден'
               when 2 then 'Отклонен'
             end,
             zpr.mal_req_user, zpr.mal_appr_user, zpr.mal_appr_datetime, zpr.mal_req_datetime, zpr.mal_appr_comment
      from zpr
      where zpr.mal_req_state = :mal_cnt
      into :status, :un_id_tmp2, :appr_un_id_temp, :req_date_close, :req_date, :appr_comment;

      un_id_tmp2 = coalesce(:un_id_tmp2, :un_id_tmp);

      if (un_id <> un_id_tmp2) then
      begin
        un_id = un_id_tmp2;
        select un.un_fio
        from username un
        where un.un_id = :un_id
        into :m_officer_name;
      end

      if (un_id = 1000000000001) then
        m_officer_name = 'Finstart_Robot';

      if (appr_un_id_temp <> appr_un_id) then
      begin
        appr_un_id = appr_un_id_temp;
        select un.un_fio
        from username un
        where un.un_id = :appr_un_id
        into :action_author;
      end

      status = coalesce(status, 'запрос не найден');
      if (status = 'запрос не найден') then
      begin
        action_author = null;
        un_id = 0;
        appr_un_id = 0;
      end
      suspend;
      :mal_cnt = :mal_cnt - 1;
      m_comment_officer = null;
    end
  end
end
