CREATE PROCEDURE AAREP_REVISION_OPERKR_OPERBOOK(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OL_DATE DATE,
    OK_FROM_OL_ID DECIMAL(18,0),
    OK_TYPE INTEGER,
    OK_OS DECIMAL(18,4),
    OK_PERC DECIMAL(18,4),
    OK_FINE DECIMAL(18,4),
    OK_KOMMIS DECIMAL(18,4),
    OB_OS DECIMAL(18,4),
    OB_PERC DECIMAL(18,4),
    OB_FINE DECIMAL(18,4),
    OB_KOMMIS DECIMAL(18,4))
AS
declare variable ob_account_dt varchar(20);
declare variable ob_account_kt varchar(20);
declare variable ob_summa_val decimal(18,4);
begin
  for select ok_ol_date, ok_from_ol_id, ok_type, sum(ok_os), sum(ok_perc), sum(
             case
               when k_index_curr_rate > cr_rate then k_index_curr_rate * ok_fine
               else cr_rate * ok_fine
             end),
             sum(ok_kommis)
      from operkred, kontract, opercurr
      where ok_ol_is_active = 1 and
            ok_type in (4, 5, 8, 9) and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            k_id = ok_from_k_id and
            k_index_curr_code = cr_from_cu_id and
            ok_ol_date between cr_date and cr_date_end /* and ok_from_k_id=109010001343200101 */
            and
            ok_from_c_id = 0
      group by ok_ol_date, ok_from_ol_id, ok_type, k_index_curr_rate, cr_rate
      into :ol_date, :ok_from_ol_id, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_kommis
  do
  begin
    if (ok_type = 5) then
      ok_fine = 0;
    ob_os = 0;
    ob_perc = 0;
    ob_fine = 0;
    ob_kommis = 0;
    for select ob_account_dt, ob_account_kt, sum(ob_summa_val)
        from operbook
        where ob_from_ol_id = :ok_from_ol_id
        group by ob_account_dt, ob_account_kt
        into :ob_account_dt, :ob_account_kt, :ob_summa_val

    do
    begin
      if (:ob_account_kt like '_______109_1%') then
        ob_os = :ob_summa_val + ob_os;
      if (:ob_account_kt like '_______211410000__') then
        ob_perc = :ob_summa_val + ob_perc;
      if (:ob_account_kt like '_______60631%') then
        ob_fine = :ob_summa_val + ob_fine;
      if (:ob_account_kt like '_______211410001__') then
        ob_kommis = :ob_summa_val + ob_kommis;
      if (:ob_account_dt like '_______109_1%') then
        ob_os = -1 * :ob_summa_val + ob_os;
      if ((:ob_account_dt like '_______60521%') and
          (:ob_account_kt like '_______11437%')) then
        ob_perc = :ob_summa_val + ob_perc;
      if (:ok_type = 9) then
      begin
        /*ob_os=0;*/
        ob_perc = 0;
        ob_fine = 0;
        ob_kommis = 0;
        if (:ob_account_kt like '_______21199%') then
          ob_os = :ob_summa_val;
      end
    end

    suspend;
  end
end
