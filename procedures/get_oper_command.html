CREATE PROCEDURE GET_OPER_COMMAND(
    UN_ID D_ID,
    PROJECT_ID_IN D_INTEGER,
    REP_DATE D_DATE = current_date)
RETURNS (
    COMMAND_ID D_ID,
    COMMAND_NAME D_STR_100,
    PROJECT_ID D_INTEGER)
AS
begin
  for select pcm.pcm_id,
             pcm.pcm_name,
             pcm.pcm_project_id
  from pantera_commands pcm
  inner join pantera_commands_operators pco on pco.pco_from_pcm_id = pcm.pcm_id
  where pco.pco_from_un_id = :un_id
  and :rep_date between pco.pco_date_begin and pco.pco_date_end
  and coalesce(pcm.pcm_date_end, cast('31.12.9999' as date)) >= :rep_date
  into :command_id,
       :command_name,
       :project_id
  do
    if (un_id = 1010100000278) then
    begin
        if (project_id = project_id_in) then
            suspend;
    end
    else
        suspend;
end
