CREATE PROCEDURE HP_PAYSYS_MAKE_OPERBOOK(
    PSM_ID D_ID)
RETURNS (
    OL_ID D_ID,
    OPER_KIND D_INTEGER,
    RES_TXT STR_255)
AS
declare variable reg_id d_integer;
declare variable def_curr d_str_code;
declare variable db_id d_integer;
declare variable psm_date d_date;
declare variable psm_memo d_str_long;
declare variable psm_oper_kind d_integer;
declare variable psm_from_un_id d_id;
declare variable dst_c_id d_id;
declare variable item_count d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable ol_parent_id d_id;
declare variable psi_code str_20;
declare variable k_disb_date date;
declare variable k_kontract_no str_20;
declare variable doc_no integer;
declare variable find_str str_20;
declare variable psp_id integer;
declare variable al_number integer;
begin
  res_txt = '';
  db_id = f_get_db_id();

  if (db_id <> 94401) then
  begin
    if (db_id in (39801, 41701)) then
    begin
      ol_id = (select ol_id from hp_paysys_make_operbook_kg(:psm_id));
      oper_kind = 1;
      suspend;
    end
    exit;
  end

  def_curr = f_get_const('cm_default_currency');
  select psm_date, psm_memo, psm_oper_kind, psm_from_un_id, psm_from_c_destin
  from paysys_main
  where psm_id = :psm_id
  into :psm_date, psm_memo, psm_oper_kind, psm_from_un_id, dst_c_id;


  select pp.psp_id
  from PAYSYS_LIST pl
  inner join paysys_main pm on pl.psl_from_psm_id = pm.psm_id
  inner join paysys_item pim on pim.psi_id = pl.psl_from_psi_id
  inner join paysys_project pp on pp.psp_id = pim.psi_from_psp_id
  where pm.psm_id = :psm_id
  rows 1
  into :psp_id;

  if (psp_id = 1) then -- Millicard
  begin
    if (psm_oper_kind = 0) then al_number = 1;
    else if (psm_oper_kind = 1) then al_number = 2;
  end
  else if (psp_id = 2) then  -- Millicard Business
  begin
    if (psm_oper_kind = 0) then al_number = 3;
    else if  (psm_oper_kind = 1) then al_number = 4;
  end

  if (psm_date is null) then
    exit;
  if (psm_memo is null) then
    psm_memo = '';

-- S.K.
  find_str = 'Закрепление БСО';
  if (psm_memo containing :find_str) then
  begin
    psi_code = substring(psm_memo from 19 for 16);
    select first 1 k_disb_date, k_kontract_no
    from knclient
    inner join kontract on k_id = kc_from_k_id and
          k_status = 1
    inner join kontstate on ks_from_k_id = k_id and
          ks_ol_is_delete = 0 and
          ks_state = 0 and
          :psm_date between ks_date_begin and ks_date_end
    where kc_from_c_id = :dst_c_id and
          kc_amount_approved > 0
    into :k_disb_date, :k_kontract_no;
    psm_memo = iif(:k_disb_date is null, '???', f_datetostr(:k_disb_date)) || ' tarixli ' || coalesce(:k_kontract_no, '???') || ' nömrəli müqaviləyə əsasən ' || :psi_code || ' nömrəli kartın verilməsi';
  end
--

-- расходная часть операции
  select psl_object_id, count(*)
  from paysys_list
  where psl_from_psm_id = :psm_id and
        psl_sign = -1 and
        psl_object_kind = 1
  group by 1
  into reg_id, item_count;

  if (item_count is null) then
    item_count = 0;
  if (reg_id is not null) then
  begin
    oper_kind = -1;
    if (psi_code is not null) then
      doc_no = (select coalesce(max(ol_doc_no), 0) + 1
                from operlist
                where ol_date = :psm_date and
                      is_delete = 0 and
                      ol_doc_kind = 0 and
                      ol_from_reg_id = :reg_id
--and ol_memo like '%kartin%'
      );
    insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_user_id, ol_document_client,
                          ol_oper_subtype, ol_parent_id, ol_doc_no)
    values (:reg_id, 2, :psm_date, 0, :psm_memo, :psm_from_un_id, :dst_c_id, /*:src_c_id,*/ 2, :ol_parent_id, :doc_no)
    returning ol_id
    into :ol_id;
    if (ol_parent_id is null) then
      ol_parent_id = ol_id;

    select al_id
    from acc_list
    where al_from_bg_id = '99990' and
          al_number = 1 and
          al_from_reg_id = :reg_id
    into :acc_dt;
    if (acc_dt is null) then
      exception e_default 'Account not found: 99990/1 for region ' || reg_id;

    select al_id
    from acc_list al
    where al_from_bg_id = '99287' and
          al.al_number = :al_number and -- 0 - на первый лицевой счет, 1 (аннулирование) на второй лицевой счет
          al_from_reg_id = :reg_id
    into :acc_kt;
    if (acc_kt is null) then
      exception e_default 'Account not found: 99287/' || :al_number || ' for region ' || reg_id;

    insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_summa, ob_summa_val,
                          ob_account_kt)
    values (:ol_id, :def_curr, 1, :acc_dt, :item_count, :item_count, :acc_kt);

    suspend;
  end

-- Приходная часть операции
  ol_id = null;
  reg_id = null;
  item_count = 0;
  select psl_object_id, count(*)
  from paysys_list
  where psl_from_psm_id = :psm_id and
        psl_sign = 1 and
        psl_object_kind = 1
  group by 1
  into reg_id, item_count;

  if (item_count is null) then
    item_count = 0;
  if (reg_id is not null) then
  begin
    oper_kind = 1;
    insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_user_id, ol_document_client,
                          ol_oper_subtype, ol_parent_id)
    values (:reg_id, 2, :psm_date, 1, :psm_memo, :psm_from_un_id, :dst_c_id, 1, :ol_parent_id)
    returning ol_id
    into :ol_id;
    if (ol_parent_id is null) then
      ol_parent_id = ol_id;

    select al_id
    from acc_list
    where al_from_bg_id = '99287' and
          al_number = :al_number and -- 0 - на первый лицевой счет, 1 (аннулирование) на второй лицевой счет
          al_from_reg_id = :reg_id
    into :acc_dt;
    if (acc_dt is null) then
      exception e_default 'Account not found: 99287/' || :al_number || ' for region ' || reg_id;

    select al_id
    from acc_list
    where al_from_bg_id = '99990' and
          al_number = 1 and
          al_from_reg_id = :reg_id
    into :acc_kt;
    if (acc_kt is null) then
      exception e_default 'Account not found: 99990/1 for region ' || reg_id;

    insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_summa, ob_summa_val,
                          ob_account_kt)
    values (:ol_id, :def_curr, 1, :acc_dt, :item_count, :item_count, :acc_kt);
    suspend;
  end
  if (ol_parent_id is not null) then
  begin
    update paysys_main
    set psm_from_ol_id = :ol_parent_id
    where psm_id = :psm_id;
  end
end
