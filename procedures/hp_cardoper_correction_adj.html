CREATE PROCEDURE HP_CARDOPER_CORRECTION_ADJ(
    COC_ID D_INTEGER)
AS
declare variable c_id d_id;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable cca_code d_str_code;
declare variable cs_request_json d_str_8100;
declare variable comment_txt varchar(140);
declare variable cst_packet_no varchar(8);
declare variable cst_ticket_no varchar(7);
declare variable k_id d_id;
declare variable un_id d_id;
declare variable coc_amount d_currency;
declare variable coc_date d_date;
declare variable office_id d_integer;
declare variable cs_id d_integer;
declare variable coc_type d_smallint;
begin
    /*
         Процедура отвечает за формирование операции для МПЦ
         "Пополнение карт счета в случае непревиденного некорректного списания",
         по факту погашения займа
    */

    select coc.coc_amount,
           coc.coc_card_id,
           coc.coc_date,
           coc.coc_from_un_id,
           coc.coc_type
    from cardoper_correction coc
    where coc.coc_id = :coc_id
    into :coc_amount,
         :card_id,
         :coc_date,
         :un_id,
         :coc_type;

    if (coc_type <> 2) then
        exception e_message 'Ошибка! Операция не "пополнение"';
    
    comment_txt = 'Пополнение на сумму = '||round(coc_amount, 2)||' сом';

    select packet_no,
           ticket_no
    from cardsynctrn_create
    into :cst_packet_no,
         :cst_ticket_no;

    if (cst_packet_no is null) then
        exception e_message '№ пакета не опредлен';
    
    if (cst_ticket_no is null) then
        exception e_message '№ тикета не опредлен';

    -- данные тяну из последней операции установки кредитного лимита, по факту выдачи
    select cs.cs_from_cca_id,
           cs.cs_office_id,
           cs.cs_from_k_id,
           cs.cs_from_c_id,
           caa.cca_code
    from cardsync cs
    inner join cardclientacc caa on caa.cca_id = cs.cs_from_cca_id
    where cs.cs_card_id = :card_id
    and cs.cs_from_csc_id = f_get_cardsynccfg('card_disb', 'id')
    and cs.cs_status = 2
    and cs.cs_date < :coc_date
    order by cs.cs_id desc
    rows 1
    into :cca_id,
         :office_id,
         :k_id, 
         :c_id,
         :cca_code;

        cs_request_json = f_format('{
        "card_acct": "%1",
        "card_acct_ccy": "KGS",
        "tran_type": "11C",
        "tran_ccy": "KGS",
        "tran_amount": %2,
        "tran_pocket": "%3",
        "tran_ticket": "%4",
        "tran_comment": "%5",
        "bank_c": "21",
        "groupc": "01",
        "tran_date_time": "' || f_datetimetostr(current_timestamp) || '"
    }',
    :cca_code,
    round(:coc_amount,2),
    :cst_packet_no,
    :cst_ticket_no,
    :comment_txt
    );

    insert into cardsync(
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_amount,
      cs_request_json,
      cs_office_id,
      cs_from_un_id,
      cs_from_k_id,
      cs_from_c_id
    )
    values(
      f_get_cardsynccfg('cardoper_correction_adjunction', 'id'),
      :card_id,
      :cca_id,
      f_objkind('cardoper_correction', 'id'),
      :coc_id,
      :coc_amount,
      :cs_request_json,
      :office_id,
      :un_id,
      :k_id,
      :c_id
    )
    returning cs_id
    into :cs_id;

    update cardsynctrn cst
    set cst.cst_from_cs_id = :cs_id
    where cst.cst_packet_no = :cst_packet_no and
          cst.cst_ticket_no = :cst_ticket_no;

    update cardoper_correction coc
    set coc.coc_from_cs_id = :cs_id
    where coc.coc_id = :coc_id;
end
