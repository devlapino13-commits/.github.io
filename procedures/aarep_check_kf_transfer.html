CREATE PROCEDURE AAREP_CHECK_KF_TRANSFER(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID ID_TYPE,
    CURRENT_REG_NAME VARCHAR(255),
    K_KONTRACT_NO VARCHAR(50),
    KF_NAME VARCHAR(50),
    DISB_DATE DATE,
    CLOSE_DATE DATE,
    DISB_KS_FIO VARCHAR(100),
    CURRENT_KS_FIO VARCHAR(100),
    DISB_REG_NAME VARCHAR(255),
    KAL_DATE_END DATE,
    MEMO_TYPE SMALLINT,
    MEMO_STR VARCHAR(1000),
    IS_WORKING VARCHAR(100),
    KONT_STATUS VARCHAR(50),
    DELINQ_DAYS INTEGER)
AS
declare variable trans_qty integer;
declare variable disb_reg_id integer;
declare variable dismissal_date date;
declare variable is_suspend smallint;
declare variable kf_id id_type;
declare variable c_id id_type;
declare variable kf_receive_date date;
declare variable new_reg_id integer;
declare variable working smallint;
declare variable str_temp varchar(255);
declare variable reg_temp varchar(100);
declare variable return_date date;
declare variable close_date_tmp date;
begin
-- вытаскиваю все контракты выданные за период
  for select reg.reg_name, kont.k_id, kont.k_from_kf_id, ks.ks_date_begin, kont.k_kontract_no
      from kontract kont
      inner join kontstate ks on kont.k_id = ks.ks_from_k_id
      inner join region reg on reg.reg_unicode = kont.k_last_reg_id
      inner join region_list(:rep_region_id, 1, -1, 1) rl on rl.id_list = kont.k_last_reg_id
      where ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0
-- and  kont.k_id =...
      order by reg.reg_name
      into :current_reg_name, :k_id, :kf_id, :disb_date, :k_kontract_no
  do
  begin
    memo_type = 0;
    close_date_tmp = null;
    close_date = null;

-- определяю дату закрытия кредита
    select ks.ks_date_begin
    from kontstate ks
    where ks.ks_from_k_id = :k_id and
          ks.ks_ol_is_delete = 0 and
          ks.ks_state = 2
    into :close_date_tmp;

-- если кредит еще не закрыт
    if (close_date_tmp is null) then
    begin
      kont_status = 'Активный';
      close_date_tmp = :rep_date_end;
    end
    else
    begin
      kont_status = 'Закрытый';
      close_date = close_date_tmp;
    end

    if (close_date_tmp > rep_date_end) then
      close_date_tmp = :rep_date_end;

    trans_qty = null;
-- определяю, были ли передачи папки с момента выдачи займа и до закрытия контракта
    select count(*)
    from knaktlist kal
    where kal.kal_from_kf_id = :kf_id and
          kal.is_delete = 0 and
          kal_date_end >= :disb_date -- <<  меня интресуют передачи папки после выдачи контракта
          and
          kal_date <= :close_date_tmp -- << т.к. меня не интресуют передачи папки после закрытия контракта
    into :trans_qty;

    if (trans_qty is null) then
      trans_qty = 0;

-- Выполнение пункта 1.
    /*
    1.      Что передаваемая папка была выдана именно этим СВЗ, от которого она была передана. 
            Напишите плиз более детально, можно на том же примере который вы описывали.
            Ведь в любом случае если кредит числится за каким-либо СВЗ когда выдается, то и передается в первый раз от этого СВЗ или могут быть другие ситуации? ДРУГИХ ПРИЧИН НЕТ, т.е. здесь главное чтобы, ему до этого группа не была передана
            
            СВЗ ИВАНОВ А. САМ ПОДГОТОВИЛ И ВЫДАЛ ЗАЕМ ГРУППЕ 100-1-1-1 (Т.Е. ЕМУ ЭТА ПАПКА НЕ БЫЛА ПЕРЕДАНА), И ОТ НЕГО ПОТОМ ЭТУ ПАПКУ (ПО КАКИМ ЛИБО ПРИЧИНАМ) ПЕРЕДАЛИ СВЗ СИДОРОВУ
    */
    if (trans_qty > 1) then -- если передачи были
    begin
      kal_date_end = null;
-- Рассматриваю первую передачу папки от СВЗ, контый выдал контракт
      for select kal.kal_new_kf_officer, kal.kal_date_end, -- после этой даты кредитная папка числится уже за другим СВЗ,
                 cl.c_name_fio, kal.kal_new_reg_id, reg.reg_name
          from knaktlist kal
          inner join clients cl on cl.c_id = kal.kal_new_kf_officer
          inner join region reg on reg.reg_unicode = kal_new_reg_id
          where kal.kal_from_kf_id = :kf_id and
                kal.is_delete = 0 and
                kal.kal_new_kf_officer <> 1000000000000 -- виртуального сотрудника не рассматриваю
                and
                kal_date_end >= :disb_date -- <<  меня интресуют передачи папки после выдачи контракта
                and
                kal_date <= :close_date_tmp -- << т.к. меня не интресуют передачи папки после закрытия контракта
                and
                kal.kal_date_end <> cast('31.12.9999' as date) -- меня не интересует последняя запись, т.к. она не отражает передачу папки
          order by kal.kal_date
          rows 1
          into :c_id, :kal_date_end, -- дата первой передачи кредитной папки
               :disb_ks_fio, :disb_reg_id, :disb_reg_name -- офис где выдавался займ
      do
      begin
        memo_type = 0;
        memo_str = '';
        /*
                теперь нужно проверить, на kal_date_end СВЗ еще работал в Мофисе где выдавался займ или был уволен
                т.к. на момент передачи папки сотрдуник еще по любому должен рбаотать, следовательно проверяю
                1) Если СВЗ с момента передачи папки проработал еще 7 дней в офисе, где выдавался займ, считаю, что он передал папку и остался работать    MEMO_TYPE = 1
                2) если сотрдуник с момента передачи папки  не проработал  7 дней и уволился, счиатаю, что он ушел  MEMO_TYPE = 2
                3) если выполнился пункт 2) и сотрудник затем вернулся в тот же офис,в течении 2 месяцев с даты увольнения, но кредитная папка ему не была обратно передана,   MEMO_TYPE = 3
            */
-- Проверка по пункту (1)
        dismissal_date = null;

        select hps.ps_date_begin, hps.ps_from_reg_id, hps.ps_is_working
        from hr_pstatus hps
        where hps.ps_from_c_id = :c_id and
              hps.ps_status = 1 and
              hps.ps_date_begin between :kal_date_end and :kal_date_end + 7 and
              ((hps.ps_is_working = 0) or
              (hps.ps_from_reg_id <> :disb_reg_id)) -- ищу либо увольнение, либо перевод в другой офис
        order by ps_date_begin
        rows 1
        into :dismissal_date, :new_reg_id, :working; -- дата увольнения или перевода в другой офис

        is_suspend = 0;
        return_date = null;
        kf_receive_date = null;
-- Проверка по пункту (1)
        if (dismissal_date is null) then
        begin
          memo_type = 1; -- если увольнения или перевода в друг офис, в рамках 7 дней не выявлено

-- получаю первое ближайшее увольнение или перевод в другой офис псоле 7 дней с момента передачи папки
          working = 2;
          select hps.ps_date_begin, hps.ps_from_reg_id, hps.ps_is_working
          from hr_pstatus hps
          where hps.ps_from_c_id = :c_id and
                hps.ps_status = 1 and
                hps.ps_date_begin > :kal_date_end + 7 and
                ((hps.ps_is_working = 0) or
                (hps.ps_from_reg_id <> :disb_reg_id)) -- ищу либо увольнение, либо перевод в другой офис
          order by ps_date_begin
          rows 1
          into :dismissal_date, :new_reg_id, :working; -- дата увольнения или перевода в другой офис

          if (working is null) then
            working = 2;

          str_temp = '';

          if ((dismissal_date is not null) and
              (working = 1)) then
          begin
            select reg.reg_name
            from region reg
            where reg.reg_unicode = :new_reg_id
            into :reg_temp;

            str_temp = 'А перевод в другой офис (' || reg_temp || '), сотоялся (' || dismissal_date || ')';
          end
          if ((dismissal_date is not null) and
              (working = 0)) then
          begin
            str_temp = 'А увольнение состоялось (' || dismissal_date || ')';
          end
          if (str_temp <> '') then
            memo_type = 2;

          memo_str = :disb_ks_fio || ' передал папку ' || cast(:kal_date_end as date) || ', он  не уволился и не переводился в другой офис, в течение 7 дней после передачи. ' || str_temp;

          is_suspend = 1;
        end
        else
        begin
-- по умолчанию считаю что выполнился пункт (2)
          memo_type = 4;
-- проверка по пункту (3)

          return_date = null;

          select hps.ps_date_begin
          from hr_pstatus hps
          where hps.ps_from_c_id = :c_id and
                hps.ps_status = 1 and
                hps.ps_date_begin > :kal_date_end and
                hps.ps_date_begin <= :kal_date_end + 60 and
                hps.ps_from_reg_id = :disb_reg_id and
                hps.ps_is_working = 1 and
                hps.ps_from_jl_id = 11 /*11 - СВЗ. Важно знать что сотрудник вренлся на должность СВЗ */
          order by ps_date_begin
          rows 1
          into :return_date;

          if (return_date is not null) then
          begin
            memo_type = 3;
            /*
                          т.к. сотрудник вренулся в тот же офис, где и выдавал займ, проверяю
                          вернули ли ему папку.
                        */
            kf_receive_date = null; -- дата обратного возвращения кредитной папки

            select kal.kal_date
            from knaktlist kal
            where kal.kal_from_kf_id = :kf_id and
                  kal.is_delete = 0 and
                  kal.kal_new_kf_officer = :c_id and
                  kal_date <= :close_date_tmp -- << т.к. меня не интресуют передачи папки после закрытия контракта
                  and
                  kal.kal_date >= :return_date
            order by kal.kal_date
            rows 1
            into :kf_receive_date;

            if (kf_receive_date is null) then
            begin
              str_temp = '';
              if (working = 1) then
              begin
                select reg.reg_name
                from region reg
                where reg.reg_unicode = :new_reg_id
                into :reg_temp;

                str_temp = 'перевелся в течении 7 дней после передачи, в другой офис (' || reg_temp || '), дата перевода (' || dismissal_date || ') ';
              end
              if (working = 0) then
              begin
                str_temp = 'уволился в течении 7 дней после передачи, дата увольнения (' || dismissal_date || ') ';
              end

              memo_str = :disb_ks_fio || ' передал папку ' || cast(:kal_date_end as date) || ', и ' || str_temp || ', затем вренулся в офис (' || disb_reg_name || '), в течении 60 дней , с момента передачи папки (Дата возврата ' || return_date || '), но кредитная папка к нему не вернулась';
              is_suspend = 1;
            end
          end
        end

        if (is_suspend = 1) then
        begin
          select kal_new_kf_name, cl.c_name_fio
          from knaktlist kal
          inner join clients cl on cl.c_id = kal.kal_new_kf_officer
          where kal.is_delete = 0 and
                kal_from_kf_id = :kf_id and
                kal_date_end = '31.12.9999'
          into :kf_name, :current_ks_fio;

          delinq_days = null;

          select iif(oks_delinq_date > cast(:kal_date_end as date), 0, cast(:kal_date_end as date) - oks_delinq_date)
          from oksummary oks
          where :kal_date_end between oks.oks_date_begin and oks_date_end and
                oks.oks_delinq_date < oks.oks_date_begin and
                oks_from_k_id = :k_id and
                oks_close_state = 0
          into :delinq_days;

          if (delinq_days is null) then
            delinq_days = 0; -- число дней просрочки на момент передачи

-- проверяю работает ли СВЗ выдавший займ на данный момент в офисе где заем был выдан
          if (exists(select *
                     from hr_pstatus hps
                     where hps.ps_from_c_id = :c_id and
                           hps.ps_status = 1 and
                           hps.ps_date_end = cast('31.12.9999' as date) and
                           hps.ps_from_reg_id = :disb_reg_id and
                           hps.ps_is_working = 1 and
                           hps.ps_from_jl_id = 11 /*11 - СВЗ*/ )) then
          begin
            is_working = 'Работает';
          end
          else
          begin
            is_working = 'Не работает';
          end
          suspend;
        end
      end
    end
  end
end
