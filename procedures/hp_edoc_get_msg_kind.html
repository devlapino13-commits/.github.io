CREATE PROCEDURE HP_EDOC_GET_MSG_KIND(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    EDC_ID D_INTEGER,
    EDS_ID D_ID = null)
RETURNS (
    MSG_KIND D_SMALLINT)
AS
declare variable c_id d_id;
declare variable edc_msg_kind d_smallint;
declare variable db_id d_integer;
declare variable pr_code d_str_100;
declare variable use_push d_bool;
begin
  -- Процедура возвращает тип эл. документа в зависимости (описание в edoc_cfg.edc_msg_kind)
  -- Ранее называлась HP_EDOC_GET_SIGN_KIND
  msg_kind = 0;

  db_id = f_get_db_id();

  select edc.edc_msg_kind
  from edoc_cfg edc
  where edc.edc_id = :edc_id
  into edc_msg_kind;

  --Для consulting.agreement создание edoc идёт сначала по kc_id, далее отправка смс проверяется по c_id
  --доп инфа в hp_html_get_loan_status, hp_edoc_ins_ca, fm_html_edoc_info
  if (db_id in (39801, 41701) and f_edoc_kind('consulting.agreement', 'id') = :edc_id and f_objkind(:obj_kind, 'name') = 'knclient') then
  begin
    obj_kind = 'knclient';
    obj_id = (select kc.kc_id
              from knclient kc
              where kc.kc_id = :obj_id);
  end 

  if (edc_msg_kind = 2) then
  begin
    -- Приоритет - отправка PUSH, проверяем необходимые условия
    select eds_from_c_id
    from edoc_sign
    where eds_id = :eds_id
    into c_id;

    use_push = 1;

    -- Проверяем, что у пользователя есть КМП, FIREBASE ID,
    -- и с даты последнего логина не прошло более 90 дней
    if (not exists(
        select *
        from mc_login
        where mcl_from_c_id = :c_id
          and mcl_is_delete = 0
          and datediff(day, mcl_last_login_datetime, current_date) <= 90
          and coalesce(mcl_google_id, '') <> '')) then
      use_push = 0;

    -- Если ранее высылали PUSH, возможно не дошло, поэтому переключаемся на SMS
    -- Правило такое, сначала 2 PUSH, затем переходим на SMS

    if (use_push = 1 and
        db_id in (39801, 41701, 81001) and
        exists(select edm_from_eds_id
               from edoc_msg
               left join fsrobot_msgbox on msg_id = edm_from_msg_id
               where edm_from_eds_id = :eds_id and
                     msg_type = 2
               group by edm_from_eds_id
               having count(*) > 1)) then
          use_push = 0;


    if (use_push = 0) then
    begin
     if (db_id not in (39801, 41701)) then
     begin
        msg_kind = 1;
        suspend;
        exit;
     end
     else
        msg_kind = 1;
    end
  end

  --Первым делом нужно проверить edc_msg_kind, если msg_kind <> 4, то возвращаем результат полученный в edoc_cfg
  if (edc_msg_kind not in (4, 5)) then
  begin
    msg_kind = edc_msg_kind;
    suspend;
    exit;
  end

  obj_kind = f_objkind(:obj_kind, 'code');

  --Проверяем наличие фото с ГРС, также проверяем продукт
  if (obj_kind in ('miniapp', 'kontract')) then
  begin
    --Перезаписываю переменную obj_id и привожу её в конечном итоге к ma_id
    if (obj_kind = 'kontract') then
    begin
      select kt.k_from_ma_app_id
      from kontract kt
      where kt.k_id = :obj_id
      into obj_id;
    end

    for select mac.mac_from_c_id, pr.pr_code
        from miniapp ma
        inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
        inner join product pr on pr.pr_id = ma.ma_from_pr_id
        where ma.ma_id = :obj_id and
              ma.ma_is_active = 1 and
              ma.ma_date_end = '31.12.9999'
        into c_id, pr_code
    do
    begin
      --Проверяем pr_code для установки msg_kind (проверка нужна для экспресс займоов и товаров в рассрочку)
      if (db_id in (39801, 41701)) then
      begin
        if (pr_code containing '.online_loan.') then
        begin
          msg_kind = 3;
          suspend;
          exit;
        end 

        if (pr_code containing '.express_online.') then
        begin
          --Если в продукте есть код сах завода или товаров в рассрочку, то тип эл. документа без СМС и PUSH
          --Иначе для стандартных экспресс займов тип эл. документа с СМС и PUSH
          if (pr_code containing '.sugarmill.' or pr_code containing '.installment_goods.') then
          begin
            msg_kind = 3;
            suspend;
            exit;
          end
          else
          begin
            msg_kind = 0;
            suspend;
            exit;
          end
        end

        if (pr_code containing '.ISTISNAA.') then
        begin
          msg_kind = 0;
          suspend;
          exit;
        end
        -- RFCKG-3065  При создании документа отправлять PUSH
        if (pr_code containing '.tms.') then
        begin
          msg_kind = 0;
          suspend;
          exit;
        end
        -- RFCKG-3065  end
      end

      --Провеярем активна ли константа по подписанию эл. документов через биометрию
      --Проверка стоит внизу чтобы не сломать логику для товаров в рассрочку
      if (f_get_const('biometry_sign_is_active') = 0) then
      begin
        suspend;
        exit;
      end

      if (not exists(select *
                     from extinfolist eil
                     inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                     where eil.eil_from_c_id = :c_id and
                           eic.eic_code = 'Tunduk' and
                           eil.eil_status = 2)) then
      begin
        suspend;
        exit;
      end
    end
  end
  else
  if (obj_kind = 'clients') then
  begin
    if (f_get_const('biometry_sign_is_active') = 0) then
    begin
      suspend;
      exit;
    end

    --Проверяем сотрудник или нет, если сотрудник то действуем по механизму с отправкой СМС
    if (exists(select *
               from hr_pstatus ps
               where ps.ps_from_c_id = :obj_id and
                     current_date between ps.ps_date_begin and ps.ps_date_end and
                     ps.ps_is_working = 1 and
                     ps.ps_status = 1)) then
    begin
      suspend;
      exit;
    end

    if (not exists(select *
                   from extinfolist eil
                   inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                   where eil.eil_from_c_id = :obj_id and
                         eic.eic_code = 'Tunduk' and
                         eil.eil_status = 2)) then
    begin
      suspend;
      exit;
    end
  end

  --Проверяем отправлено ли СМС, если отправлено СМС то msg_kind = 0
  for select eds.eds_id
      from edoc_list edl
      inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
      where edl.edl_object_kind = f_objkind(:obj_kind, 'id') and
            edl.edl_object_id = :obj_id and
            edl.edl_status >= 0 and
            current_date between edl.edl_blank_date and coalesce(edl.edl_date_end, current_date) and
            edl.edl_from_edc_id = :edc_id
      into eds_id
  do
  begin
    if (exists(select *
               from edoc_msg edm
               where edm.edm_from_eds_id = :eds_id)) then
    begin
      suspend;
      exit;
    end
  end

  --Если все проверки пройдены то возвращаем тип эл. документа с подписью через биометрию
  msg_kind = edc_msg_kind;
  suspend;
end
