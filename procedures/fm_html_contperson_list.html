CREATE PROCEDURE FM_HTML_CONTPERSON_LIST(
    UN_ID D_ID,
    C_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable name d_str_100;
declare variable relation d_str_100;
declare variable cell_no d_str_255;
declare variable cell_no_add d_str_255;
declare variable language_id d_integer;
declare variable language d_str_100;
declare variable status d_str_20;
declare variable memo d_str_255;
declare variable address d_str_255;
declare variable comments d_str_255;
declare variable create_date d_date;
declare variable adl_id d_id;
declare variable is_old d_integer;
begin
  res_text = (select res_style
              from fm_html_styles) || '<body>';
  is_old = 0;
  while (is_old < 2) do
  begin
    res_text = blob_append(res_text, '<h1>' || iif(is_old = 0, 'Действующие контакты:', 'Предыдущие контакты:') || '</h1>');
    for select coalesce(cc_name, ''), coalesce(cc_status, ''), cc_cell_no, cc_cell_no_add,
               cc_language,
               left(coalesce(cc_check_memo,''), 255),
               case cc_total_res
                 when 1 then 'Подтвержден'
                 when -1 then 'Не дозвонились'
                 when -2 then 'Не подтверждена'
                 else ''
               end,
               cc_from_adl_id, coalesce(cc_svz_comments, ''), cc_create_date
        from contacts
        where cc_from_c_id = :c_id and
              cc_is_old_contact = :is_old
        order by cc_date desc
        rows 20
        into name, relation, cell_no, cell_no_add, language_id, memo, status, adl_id, comments, create_date
    do
    begin
      if (name = '') then
        name = 'ФИО не указано';
      if (relation = '') then
        relation = 'Не указано';
      memo = replace(memo, 'Примечание: ', '');
      language = '';
      if (language_id is not null) then
      begin
        select rd_name
        from refdata
        where rd_id = :language_id
        into language;
      end
      address = 'Адрес не заполнен.';
      if (adl_id is null) then
        adl_id = 0;
      if (adl_id <> 0 and
          right(adl_id, 9) <> '000000000') then
      begin
        select adl_address_str
        from addrlist
        where adl_id = :adl_id
        into :address;
      end
  
      res_text = blob_append(res_text,
        '<b>' || name || '</b><br>' ||
        'Кем приходится: ' || relation || '<br>' ||
        'Дата заполнения: ' || f_datetostr(create_date) || '<br>' ||
        'Язык: ' || language || '<br>' ||
        'Статус: ' || status || '<br>' ||
        'Адрес: ' || address || '<br>' ||
        'Примечание специалиста: ' || memo || '<br>' ||
        'Примечание оператора: ' || comments || '<br>' ||
        '№ телефона: ' || f_uri_phone(cell_no, 0, c_id, 1) || '<br>' ||
        iif(cell_no_add <> '', 'Доп. № телефона: ' || f_uri_phone(cell_no_add, 0, c_id, 1) || '<br>', '') ||
        '<br><hr width="100%">');
  
    end
    is_old = is_old + 1;
  end
  res_text = blob_append(res_text, '</body>');
  suspend;
end
