CREATE PROCEDURE HP_REPAY_AZREPAYKONT
RETURNS (
    ARK_FROM_K_ID ID_TYPE,
    ARK_LOAN_ID ID_TYPE,
    ARK_SUMM CURR_TYPE,
    OL_ID DECIMAL(18,0),
    ERROR_TEXT STR_255,
    OKS_BALANCE CURR_TYPE,
    MESSAGE_TEXT STR_100)
AS
declare variable rep_date date;
declare variable k_last_reg_id integer;
declare variable kc_from_c_id id_type;
declare variable repay_dt_account_no str_20;
declare variable is_finish smallint;
begin
  rep_date = '19.04.2019';
  for select ark_from_k_id, ark_loan_id, ark_summ
      from azrepaykont tt
      where ark_kol_cli = 1 and
            ark_is_finish = 0 and
            ark_summ > 0 and
            tt.ark_ks_state = 1
      into ark_from_k_id, ark_loan_id, ark_summ
  do
  begin
    is_finish = 0;
    oks_balance = null;
    select oks_balance
    from oksummary
    where oks_from_k_id = :ark_from_k_id and
          :rep_date between oks_date_begin and oks_date_end and
          oks_close_state = 0
    into oks_balance;
    oks_balance = coalesce(oks_balance, 0);
    error_text = '';
    message_text = '';
    ol_id = null;
    if (ark_summ > oks_balance) then
    begin
      message_text = 'Внимание! Баланс контракта меньше суммы погашения. Село переплатой';
    end

--begin
    select k_last_reg_id
    from kontract
    where k_id = :ark_from_k_id
    into k_last_reg_id;

    repay_dt_account_no = null;
    select al_id
    from acc_list
    where al_from_reg_id = :k_last_reg_id and
          al_account = '25159/1'
    into repay_dt_account_no;

    if (repay_dt_account_no is null) then
      error_text = 'По текущему офису не создан счет "25159/1"';
    else
    begin
      select kc_from_c_id
      from knclient
      where kc_from_k_id = :ark_from_k_id and
            kc_amount_approved > 0
      into kc_from_c_id;

      error_text = 'Не сработала процедура hp_do_repayment';
      select ol_id, error_text
      from hp_do_repayment(:ark_from_k_id, 0, :rep_date, :ark_summ, 1, -- :repay_to_os,
      0, -- :repay_to_perc,
      0, -- :repay_to_fine,
      0, -- :repay_to_kommis,
      1, -- :need_save,
      0, -- :is_full_repay,
      0, -- :is_writeoff,
      :k_last_reg_id, -- :repay_fact_office_id, -- головной офис в рф
      3, -- :repay_dt_account_kind, -- внутренний бух.счет европлата
      :repay_dt_account_no, -- :repay_dt_account_no,
      1, -- :repay_curr_rate,
      1000000000000, -- :repay_user_id,
      :kc_from_c_id, -- :repay_document_client,
      'Ödəyici “AR Prezidentinin 28.02.2019-cu il tarixli Fərmanına əsasən Dövlət tərəfindən ödənilmişdir', -- :repay_doc_client_name_reason,
      null, -- :repay_manager_c_id,
      0, -- :repay_from_rd_id,
      0)
      into ol_id, error_text;
      if (error_text = '') then
        is_finish = 1;
    end
--end

    update azrepaykont tt
    set ark_error_text = :error_text,
        ark_repay_ol_id = :ol_id,
        ark_is_finish = :is_finish,
        tt.akr_message_text = :message_text
    where ark_from_k_id = :ark_from_k_id and
          ark_loan_id = :ark_loan_id;

    suspend;
  end
end
