CREATE PROCEDURE CB_RU_LO_2012_12_NEW(
    REP_REGION_ID INTEGER,
    REP_BONUS_ID INTEGER,
    REP_USER_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    CB_PERSONAL_ID ID_TYPE,
    CB_REGION_ID INTEGER,
    CB_DISB_KOLICH_CLI INTEGER,
    CB_CLOSE_KOLICH_CLI INTEGER,
    CB_ACTIV_KOLICH_CLI INTEGER,
    CB_SUM_VZYSKAN CURR_TYPE,
    CB_DELINQ_BALANCE CURR_TYPE,
    CB_ACTIV_BALANCE CURR_TYPE,
    CB_DISB_KOLICH_CLI_3 INTEGER,
    CB_PERSONAL_STAGE DATE,
    CB_PERSONAL_STAGE_MONTH DECIMAL(15,1),
    PAR7 DECIMAL(15,2),
    CB_IS_GET INTEGER,
    CB_BONUS_TOTAL CURR_TYPE,
    CB_IS_WORKING INTEGER,
    CB_MEMO VARCHAR(255),
    CB_PORTFEL_C CURR_TYPE,
    CB_PORTFEL_D CURR_TYPE)
AS
declare variable rep_date_begin date;
declare variable rep_date_end_2 date;
declare variable rep_date_begin_3 date;
declare variable cb_disb_kolich_cli_3_all integer;
declare variable is_hr smallint;
begin

  rep_date_begin = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date); -- Начало периода
  rep_date_end_2 = rep_date_begin - 1; -- Конец предыдущего месяца
  if (:rep_date_end between '02.01.2013' and '31.01.2013') then -- Для того чтобы если за январь2013 считать то за пред-й портфель брал 25.12.2012
  begin
    rep_date_end_2 = '25.12.2012';
    rep_date_begin = rep_date_end_2 + 1;
  end
  rep_date_begin_3 = :rep_date_begin - 32;
  rep_date_begin_3 = cast('01.' || extract(month from :rep_date_begin_3) || '.' || extract(year from :rep_date_begin_3) as date); -- Начало 3 месяц назад

  for with tabl1
      as (select distinct kal_new_kf_officer, kal_new_reg_id
          from kontstate, kontract, knaktlist
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                :rep_date_end between ks_date_begin and ks_date_end and
                ks_from_k_id = k_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                :rep_date_end between kal_date and kal_date_end
          union
          select distinct ks_kc_id, ks_region_id
          from kontstate
          where ks_ol_is_delete = 0 and
                cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
                ks_state in (1, 2))

      select kal_new_kf_officer, kal_new_reg_id
      from tabl1, region_list(:rep_region_id, 1, 3, 1)
      where kal_new_reg_id = id_list
      into cb_personal_id, cb_region_id
  do
  begin
--  Вычисляет кол-во выдачи данного СВЗ в данном офиса
    select coalesce(sum(k_kolich_client_approved), 0)
    from kontstate, kontract
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :cb_personal_id and
          ks_region_id = :cb_region_id and
          ks_from_k_id = k_id
    into cb_disb_kolich_cli;
--  Вычисляет кол-во закрытых данного свз в данном регионе
    select sum(
           case
             when ((select kont_category
                    from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 'Категория 1' and (k_disb_date < '01.12.2012' or (coalesce(ks_date_begin, '01.01.1990') >= k_kredline_end))) -- если кредит выдан от 01.12.2012 и после, и закрыт досрочно то их не включает
               then k_kolich_client_approved
             else 0
           end)
    from kontstate, kontract
    where ks_ol_is_delete = 0 and
          ks_state = 2 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :cb_personal_id and
          ks_region_id = :cb_region_id and
          ks_from_k_id = k_id
    into cb_close_kolich_cli;
    if (cb_close_kolich_cli is null) then
      cb_close_kolich_cli = 0;
--  Вычисляет кол-во активных клиентво по категории 1 и 2
    select sum(
           case
             when (((select kont_category
                     from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) in ('Категория 1', 'Категория 2')) and (coalesce((select rep_delinq_days
                                                                                                                                                                              from aarep_kont_delinq_status(k_id, :rep_date_end)), 0) <= 30)) -- кол-во клиентов катег 1 и 2, которые просроч. дни < 30 (пункт 3.1.1)
               then k_kolich_client_approved
             else 0
           end)
    from kontstate, kontract, knaktlist
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal_new_kf_officer = :cb_personal_id and
          kal_new_reg_id = :cb_region_id
    into :cb_activ_kolich_cli;
    if (cb_activ_kolich_cli is null) then
      cb_activ_kolich_cli = 0;
--  Сумма взыскании по Категории 2
    select coalesce(-sum(ok_os + ok_perc + ok_fine), 0)
    from operkred, kontract, knaktlist
    where ok_ol_is_active = 1 and
          ok_ol_date between :rep_date_begin and :rep_date_end and
          ok_from_c_id = 0 and
          ok_type = 4 and
          ok_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          ok_ol_date between kal_date and kal_date_end and
          kal_new_kf_officer = :cb_personal_id and
          kal_new_reg_id = :cb_region_id and
          (select kont_category
           from aarep_get_category_id(k_id, ok_ol_date, null, null, null, null, null, null)) = 'Категория 2'
    into cb_sum_vzyskan;
--  Просроченный баланс категории 1 от 8 до 180 дней
    select sum(
           case
             when (coalesce((select rep_delinq_days
                             from aarep_kont_delinq_status(k_id, :rep_date_end)), 0) between 8 and 180) then (select sum(ok_os)
                                                                                                              from operkred
                                                                                                              where ok_from_k_id = k_id and
                                                                                                                    ok_from_c_id = 0 and
                                                                                                                    ok_type in (4, 5, 6, 8) and
                                                                                                                    ok_ol_is_active = 1 and
                                                                                                                    ok_ol_date <= :rep_date_end)
             else 0
           end)
    from kontstate, kontract, knaktlist
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal_new_kf_officer = :cb_personal_id and
          kal_new_reg_id = :cb_region_id and
          (select kont_category
           from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 'Категория 1'
    into cb_delinq_balance;
    if (cb_delinq_balance is null) then
      cb_delinq_balance = 0;
--  Расчет баланса КС на конец предыдущего месяца по Категории 1; B - общий портфель катег.1 на предыдущую отчетную дату
    select sum(
           case(select kont_category
                from aarep_get_category_id(k_id, :rep_date_end_2, null, null, null, null, null, null))
             when 'Категория 1' then coalesce((select sum(ok_os)
                                                        from operkred
                                                        where ok_from_k_id = k_id and
                                                              ok_from_c_id = 0 and
                                                              ok_type in (4, 5, 6, 8) and
                                                              ok_ol_is_active = 1 and
                                                              ok_ol_date <= :rep_date_end_2), 0)
             else 0
           end)
    from kontstate, kontract, knaktlist
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end_2 between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date_end_2 between kal_date and kal_date_end and
          kal_new_kf_officer = :cb_personal_id and
          kal_new_reg_id = :cb_region_id
    into :cb_activ_balance;
    if (:cb_activ_balance is null) then
      cb_activ_balance = 0;
-- С - Портфель займов, переданный в категорию 1 от другого СВЗ в течение отчетного месяца
    cb_portfel_c = null;
    select sum(
           case(select kont_category
                from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null))
             when 'Категория 1' then coalesce((select sum(ok_os)
                                                        from operkred
                                                        where ok_from_k_id = k_id and
                                                              ok_from_c_id = 0 and
                                                              ok_type in (4, 5, 6, 8) and
                                                              ok_ol_is_active = 1 and
                                                              ok_ol_date <= :rep_date_end), 0)
             else 0
           end)
    from kontstate, kontract, knaktlist kal1, knaktlist kal2
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal1.kal_from_kf_id and
          kal1.is_delete = 0 and
          :rep_date_end between kal1.kal_date and kal1.kal_date_end and
          kal1.kal_new_kf_officer = :cb_personal_id and
          kal1.kal_new_reg_id = :cb_region_id and
          k_from_kf_id = kal2.kal_from_kf_id and
          kal2.is_delete = 0 and
          :rep_date_begin between kal2.kal_date and kal2.kal_date_end and
          kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer
    into cb_portfel_c;
    if (cb_portfel_c is null) then
      cb_portfel_c = 0;

-- D - Портфель займов, переданный в категорию 1 от данного СВЗ другому СВЗ в течение отчетного месяца
    cb_portfel_d = null;
    select sum(
           case(select kont_category
                from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null))
             when 'Категория 1' then coalesce((select sum(ok_os)
                                                        from operkred
                                                        where ok_from_k_id = k_id and
                                                              ok_from_c_id = 0 and
                                                              ok_type in (4, 5, 6, 8) and
                                                              ok_ol_is_active = 1 and
                                                              ok_ol_date <= :rep_date_end), 0)
             else 0
           end)
    from kontstate, kontract, knaktlist kal1, knaktlist kal2
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal1.kal_from_kf_id and
          kal1.is_delete = 0 and
          :rep_date_end between kal1.kal_date and kal1.kal_date_end
--   and kal1.kal_new_kf_officer = :cb_personal_id
          and
          k_from_kf_id = kal2.kal_from_kf_id and
          kal2.is_delete = 0 and
          :rep_date_begin between kal2.kal_date and kal2.kal_date_end and
          kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer and
          kal2.kal_new_kf_officer = :cb_personal_id and
          kal2.kal_new_reg_id = :cb_region_id
    into cb_portfel_d;
    if (cb_portfel_d is null) then
      cb_portfel_d = 0;

-- расчет новых клиентов за последние 3 месяца
    if (:rep_date_end > '31.01.2013') then -- С февраля будет действовать то что новых клиентов больше 18ти, нужно вопрос поднять новых клиентов только по своему офису считать ?
    begin
      cb_disb_kolich_cli_3_all = null;
      select sum(
             case
               when kc1.kc_circle <= 1 then 1
               else case
                      when (select coalesce(max(ks2.ks_date_begin), '01.01.1990') -- если цикл свыше одного и из предыдущих контрактов нет то будет считать как новый
                            from kontstate ks2, knclient kc2
                            where ks2.ks_ol_is_delete = 0 and
                                  ks2.ks_state = 2 and
                                  ks2.ks_date_begin < ks1.ks_date_begin and
                                  ks2.ks_from_k_id = kc2.kc_from_k_id and
                                  kc2.kc_amount_approved > 0 and
                                  kc2.kc_from_c_id = kc1.kc_from_c_id) < ks1.ks_date_begin - 180 -- клиента будет считать новым если клиент до этого не брал более 6месяцев
                        then 1
                      else 0
                    end
             end)
      from kontstate ks1, knclient kc1
      where ks1.ks_ol_is_delete = 0 and
            ks1.ks_state = 1 and
            cast(ks1.ks_date_begin as date) between :rep_date_begin_3 and :rep_date_end and
            ks1.ks_kc_id = :cb_personal_id
--     and ks1.ks_region_id = :cb_region_id -- Нужно уточнить про новых клиентов, если отдельно по офисам разбить то это нужно включить
            and
            kc1.kc_from_k_id = ks1.ks_from_k_id and
            kc1.kc_amount_approved > 0
      into :cb_disb_kolich_cli_3_all;
      if (:cb_disb_kolich_cli_3_all is null) then
        cb_disb_kolich_cli_3_all = 0;
    end
    else
    begin
      cb_disb_kolich_cli_3_all = 18;
    end

-- Раздельно по офисам, для отображения
    if (:rep_date_end > '31.01.2013') then -- С февраля будет действовать то что новых клиентов больше 18ти, нужно вопрос поднять новых клиентов только по своему офису считать ?
    begin
      cb_disb_kolich_cli_3 = null;
      select sum(
             case
               when kc1.kc_circle <= 1 then 1
               else case
                      when (select coalesce(max(ks2.ks_date_begin), '01.01.1990') -- если цикл свыше одного и из предыдущих контрактов нет то будет считать как новый
                            from kontstate ks2, knclient kc2
                            where ks2.ks_ol_is_delete = 0 and
                                  ks2.ks_state = 2 and
                                  ks2.ks_date_begin < ks1.ks_date_begin and
                                  ks2.ks_from_k_id = kc2.kc_from_k_id and
                                  kc2.kc_amount_approved > 0 and
                                  kc2.kc_from_c_id = kc1.kc_from_c_id) < ks1.ks_date_begin - 180 -- клиента будет считать новым если клиент до этого не брал более 6месяцев
                        then 1
                      else 0
                    end
             end)
      from kontstate ks1, knclient kc1
      where ks1.ks_ol_is_delete = 0 and
            ks1.ks_state = 1 and
            cast(ks1.ks_date_begin as date) between :rep_date_begin_3 and :rep_date_end and
            ks1.ks_kc_id = :cb_personal_id and
            ks1.ks_region_id = :cb_region_id -- Нужно уточнить про новых клиентов, если отдельно по офисам разбить то это нужно включить
            and
            kc1.kc_from_k_id = ks1.ks_from_k_id and
            kc1.kc_amount_approved > 0
      into :cb_disb_kolich_cli_3;
      if (:cb_disb_kolich_cli_3 is null) then
        cb_disb_kolich_cli_3 = 0;
    end
    else
    begin
      cb_disb_kolich_cli_3 = 18;
    end

-- Вычисляет стаж КС     .
    cb_personal_stage = null;
    cb_is_working = null;

    is_hr = 1; -- 0 - стаж считает с первой выдачи; 1 - стаж берет из трудового договора
    if (is_hr = 0) then
    begin
      select cb_personal_stage, cb_is_working
      from cb_personal_data(:cb_region_id, :rep_date_end, :cb_personal_id)
      into cb_personal_stage, cb_is_working;
    end
    else
    begin
--    Дата поступления на работу
      select ps_date_begin
      from hr_pstatus --, hr_orders , joblist
      where ps_from_c_id = :cb_personal_id and
            ps_date_begin is not null and
            ps_status = 1
      order by ps_date_begin
      rows 1
      into :cb_personal_stage;
--    Cтатус на конец даты,  0 - не работает; 1 - работает;
      select ps_is_working
      from hr_pstatus
      where ps_from_c_id = :cb_personal_id and
            ps_date_begin is not null and
            ps_status = 1 and
            :rep_date_end between ps_date_begin and ps_date_end
      into cb_is_working;
    end

    if (cb_personal_stage is null) then
      cb_personal_stage = :rep_date_end;
    if (cb_is_working is null) then
      cb_is_working = 0;
    cb_personal_stage_month = datediff(day, :cb_personal_stage, :rep_date_end) / 30;

-- Расчет PAR7
    /*   if (cb_delinq_balance = 0 and cb_activ_balance = 0) then
   begin
     par7 = 0;
   end else
   begin
     if (cb_activ_balance > 0) then
     begin
       par7 = cb_delinq_balance / cb_activ_balance * 100;
     end else
     begin
       par7 = 100;
     end 
   end */
-- По положению от 27.02.2013
    if (cb_delinq_balance = 0) then
    begin
      par7 = 0;
    end
    else
    begin
      if (cb_activ_balance + cb_portfel_c - cb_portfel_d > 0) then
      begin
        par7 = cb_delinq_balance / (cb_activ_balance + cb_portfel_c - cb_portfel_d) * 100;
      end
      else
      begin
        par7 = 100;
      end
    end

-- Разработано по пункту 3.1.1

    if (par7 <= 7.0 and
        cb_is_working = 1) then
    begin
      if (cb_disb_kolich_cli_3_all >= 18) then
      begin
        if ((cb_personal_stage_month >= 1 and
            cb_personal_stage_month < 6) or (cb_personal_stage_month >= 6 and
            cb_personal_stage_month < 9 and
            cb_activ_kolich_cli >= 50) or (cb_personal_stage_month >= 9 and
            cb_personal_stage_month < 12 and
            cb_activ_kolich_cli >= 75) or (cb_personal_stage_month >= 12 and
            cb_personal_stage_month < 15 and
            cb_activ_kolich_cli >= 100) or (cb_personal_stage_month >= 15 and
            cb_personal_stage_month < 18 and
            cb_activ_kolich_cli >= 125) or (cb_personal_stage_month >= 18 and
            cb_activ_kolich_cli >= 150)) then
        begin
          cb_is_get = 1;
          cb_memo = '';
        end
        else
        begin
          cb_is_get = 0;
          cb_memo = 'Бонус не положено из-за стажа';
        end
      end
      else
      begin
        if ((cb_personal_stage_month >= 18 and
            cb_activ_kolich_cli >= 150) or (cb_personal_stage_month >= 1 and
            cb_personal_stage_month < 3)) then -- кол-во новых кли. > 18 не действует у кого стаж от 1-3 мес.
        begin
          cb_is_get = 1;
          cb_memo = '';
        end
        else
        begin
          cb_is_get = 0;
          cb_memo = 'Бонус не положено из-за кол-во новых клиентов за последние 3 месяца.';
        end
      end
    end
    else
    begin
      cb_is_get = 0;
      if (cb_is_working <> 1) then
        cb_memo = 'Сотрудник в данном офисе не работает.';
      else
        cb_memo = 'Бонус не положено из-за PAR > 7.';
    end

    cb_bonus_total = 0;
    if (cb_is_get = 1) then
    begin
      cb_bonus_total = (cb_disb_kolich_cli * 500) + (cb_close_kolich_cli * 1000) + (cb_sum_vzyskan * 8 / 100);
    end

    suspend;
  end
end
