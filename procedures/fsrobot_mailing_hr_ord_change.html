CREATE PROCEDURE FSROBOT_MAILING_HR_ORD_CHANGE(
    HO_ID ID_TYPE,
    OLD_JL_ID D_INTEGER,
    NEW_JL_ID D_INTEGER,
    OLD_DATE D_DATE,
    NEW_DATE D_DATE,
    C_ID ID_TYPE)
AS
declare variable email_text d_str_long;
declare variable ho_order_no str_20;
declare variable ho_order_type d_integer;
declare variable ho_cause_id d_integer;
declare variable ps_date_begin d_date;
declare variable jl_name str_100;
declare variable c_name_fio str_100;
declare variable old_jl_name str_100;
begin
  select iif(:old_jl_id <> :new_jl_id, coalesce(ho_order_no, 'б/н'), ho_order_no), ho_order_type, ho_cause_id,
         ps_date_begin, jl_name, c_name_fio
  from hr_orders
  inner join hr_pstatus on ps_from_ho_id = ho_id
  inner join joblist on jl_id = ps_from_jl_id
  inner join clients on c_id = ps_from_c_id
  where ho_id = :ho_id and
        ho_status = 1 and
        ps_from_c_id = :c_id rows 1
  into ho_order_no, ho_order_type, ho_cause_id, ps_date_begin, jl_name, c_name_fio;

  if (not exists(select *
                 from refdata
                 where rd_code = 'OrderChangeTermContract.WorkCondition' and
                       rd_id = :ho_cause_id)) then
  begin
    email_text = 'Внесено изменение в приказ № ' || ho_order_no || ' от ' || f_datetostr(ps_date_begin) || '<br>По сотруднику <b>' || jl_name || ', ' || c_name_fio || '</b><br>';

    if (old_jl_id <> new_jl_id) then
    begin
      select jl_name
      from joblist
      where jl_id = :old_jl_id
      into old_jl_name;

      email_text = email_text || '<br>Старая должность: <b>' || old_jl_name || '</b><br>Новая должность: <b>' || jl_name || '</b><br>';
    end

    if (old_date <> new_date) then
    begin
      email_text = email_text || '<br>Старая дата начала действия: <b>' || f_datetostr(old_date) || '</b><br>Новая дата начала действия: <b>' || f_datetostr(new_date) || '</b>';
    end

    if (email_text is null) then
      exit;

    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                msg_command_kind, msg_object_kind, msg_object_id)
    values (0, 'ru.staff@molbulak.com', 'Внесено изменение в приказ', :email_text, 7,
            1000000000000, '', 0, 0);
  end
end
