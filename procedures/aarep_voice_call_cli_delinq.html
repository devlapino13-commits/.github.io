CREATE PROCEDURE AAREP_VOICE_CALL_CLI_DELINQ(
    REP_DATE_END DATE)
AS
declare variable phone varchar(80);
declare variable delinq_day integer;
declare variable c_ammount decimal(18,4);
declare variable k_kontract_no varchar(20);
declare variable c_id d_id;
begin

  insert into voice_call_result (vc_phone_no, vc_amount, vc_script_no, vc_call_result, vc_last_call, vc_call_count,
                                 vc_date_save, vc_kontract_no)
  select vc_phone_no, vc_amount, vc_script_no, vc_call_result, vc_last_call, vc_call_count, current_date,
         vc_kontract_no
  from voice_call;

  delete from voice_call
  where 1 = 1;

  insert into voice_call (vc_phone_no, vc_amount, vc_script_no, vc_call_result, vc_last_call, vc_call_count)
  values ('996772681079', 123, 27, null, null, 0);
  insert into voice_call (vc_phone_no, vc_amount, vc_script_no, vc_call_result, vc_last_call, vc_call_count)
  values ('996772681079', 123, 57, null, null, 0);
  insert into voice_call (vc_phone_no, vc_amount, vc_script_no, vc_call_result, vc_last_call, vc_call_count)
  values ('996772681079', 123, 90, null, null, 0);
  --  INSERT INTO VOICE_CALL (VC_PHONE_NO, VC_AMOUNT, VC_SCRIPT_NO, VC_CALL_RESULT, VC_LAST_CALL, VC_CALL_COUNT) VALUES ('996772110011', 123, 5, NULL, NULL, 0);

  for with tabl1
      as (select oks_region_id, oks_from_k_id, cast(:rep_date_end as date) - oks_delinq_date as delinq_day
          from oksummary
          where :rep_date_end between oks_date_begin and oks_date_end and
                cast(:rep_date_end as date) - oks_delinq_date in (27, 57, 90) and
                oks_close_state = 0)
      select delinq_day,
             (select sum(zad_os + zad_perc + sh_calc_fine)
              from aarep_make_zad_list(k_id, :rep_date_end, 0)
              where sh_date is not null and
                    sh_repay_type = 1), k_kontract_no, kc_from_c_id
      from tabl1, knclient, kontract
      where oks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            oks_from_k_id = k_id
      into :delinq_day, :c_ammount, :k_kontract_no, c_id
  do
  begin
    phone = f_cont_info('clients', c_id, 'cell_main', 'ok');
    if (char_length(phone) <> 11) then
      continue;

    insert into voice_call (vc_phone_no, vc_amount, vc_script_no, vc_call_count, vc_kontract_no)
    values (:phone, :c_ammount, :delinq_day, 0, :k_kontract_no);
  end
end
