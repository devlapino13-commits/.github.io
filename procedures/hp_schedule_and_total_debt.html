CREATE PROCEDURE HP_SCHEDULE_AND_TOTAL_DEBT(
    K_ID ID_TYPE)
RETURNS (
    AMOUNTTOBEPAID DECIMAL(18,2),
    TOTALDEBT DECIMAL(18,2))
AS
declare variable SH_DATE D_DATE;
declare variable DO_STOP D_INTEGER;
declare variable AMOUNTTOBEPAID_TMP D_CURRENCY;
declare variable TOTALDEBT_TMP D_CURRENCY;
begin
  do_stop = 0;  -- эта переменная нужна для того чтобы остановиться на первом ближайщем платеже в цикле
  amounttobepaid = 0; -- сумма ближайщего платежа по контракту
  totaldebt = 0; -- общий долг по контракту

  for select sum(zad_os + zad_perc + sh_calc_fine + zad_kommis),
             sum(apr.zad_os + apr.sh_calc_fine + apr.zad_kommis + apr.sh_add_perc + sh_calc_perc), apr.sh_date
      from aarep_make_zad_list(:k_id, current_date, 1) apr
      group by sh_date
      into amounttobepaid_tmp, totaldebt_tmp, sh_date
  do
  begin
    totaldebt = totaldebt + totaldebt_tmp;
    if (sh_date < current_date) then
      amounttobepaid = amounttobepaid + amounttobepaid_tmp;

    if (sh_date >= current_date and
        do_stop = 0) then
    begin
      amounttobepaid = amounttobepaid + amounttobepaid_tmp;
      do_stop = 1;
    end
  end
  if (exists(select *
             from oksummary oks
             where oks.oks_from_k_id = :k_id and
                   oks.oks_close_state = 1)) then
    amounttobepaid = totaldebt; -- для спец учёта сумма общего долга равна сумме ближайшего платежа
  suspend;
end
