CREATE PROCEDURE HP_GET_LIGRES_CONTENT(
    BLS_ID D_ID)
RETURNS (
    IP_NUM D_STR_100,
    DEBT_TYPE D_STR_255,
    IP_START D_DATE,
    IP_END_DATE D_DATE,
    DEBT_SUM D_CURRENCY,
    SUBJECT_NAME D_STR_LONG,
    FSSP_DEP D_STR_255,
    FSSP_ADR D_STR_LONG,
    END_SECTION D_STR_255,
    END_REASON D_STR_LONG,
    IS_CLOSED D_SMALLINT,
    FIO_PRISTAV D_STR_255,
    CLAIMANT_INN D_STR_20)
AS
declare variable ip_start_str d_str_255;
declare variable ip_end_date_str d_str_255;
declare variable debt_sum_str d_str_255;
declare variable arr_result d_integer;
declare variable arr_response d_integer;
declare variable obj_resp d_integer;
declare variable jso d_integer;
begin
  if (bls_id is null) then exit;
  select iif(blx_blob is null, null, json.parse(blx_blob))
  from hp_get_blobex_blob(:bls_id)
  into jso;

  arr_result = json.obj_get_arr(jso, 'result');
  if (json.count_get(arr_result) = 0) then
    exit;

  arr_response = json.obj_get_arr(
                   json.arr_get_obj(
                     json.arr_get_arr(arr_result, 0),
                     0
                   ),
                   'response'
                 );

  for select jso
  from hp_bki_json_get_list(:arr_response, 1)
  into obj_resp
  do
  begin
    is_closed = 0;

    ip_num = json.obj_get_val(obj_resp, 'ip.number', '');
    debt_type = json.obj_get_val(obj_resp, 'subject.type', '');
    ip_start_str = json.obj_get_val(obj_resp, 'ip.initiated', '');
    ip_end_date_str = json.obj_get_val(obj_resp, 'ending.date', '');
  
    if (ip_start_str = '') then
      ip_start = null;
    else if (position(' ' in ip_start_str) > 0) then
      ip_start = cast(substring(ip_start_str from 1 for position(' ' in ip_start_str) - 1) as d_date);
    else
      ip_start = cast(ip_start_str as d_date);
  
    if (ip_end_date_str = '') then
      ip_end_date = null;
    else if (position(' ' in ip_end_date_str) > 0) then
      ip_end_date = cast(substring(ip_end_date_str from 1 for position(' ' in ip_end_date_str) - 1) as d_date);
    else
      ip_end_date = cast(ip_end_date_str as d_date);
  
    debt_sum_str = json.obj_get_val(obj_resp, 'subject.debt', '0');
    debt_sum = iif(debt_sum_str = '', 0, debt_sum_str);

    subject_name = json.obj_get_val(obj_resp, 'subject.name', '');
    fssp_dep = json.obj_get_val(obj_resp, 'department.name', '');
    fssp_adr = json.obj_get_val(obj_resp, 'department.address', '');
    end_section = json.obj_get_val(obj_resp, 'ending.section', '');
    end_reason = json.obj_get_val(obj_resp, 'ending.reason', '');
    fio_pristav = json.obj_get_val(obj_resp, 'bailiff.lfs', '');
    claimant_inn = json.obj_get_val(obj_resp, 'claimant.inn', '');
  
    if (coalesce(end_section, '') containing 'ст. 46') then
      is_closed = 1;
    else
      is_closed = 0;

    suspend;
  end
end
