CREATE PROCEDURE FM_HTML_APP_RETURN_CONSULT(
    ARC_ID D_INTEGER)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable db_id d_integer;
declare variable c_fio d_str_100;
declare variable amount decimal(18,0);
declare variable oper_no d_integer;
declare variable return_reason varchar(30);
declare variable pay_date date;
declare variable k_no d_str_20;
declare variable confirm_txt d_str_20;
declare variable sign_type d_smallint;
declare variable message_conf varchar(50);
begin
  res_text = (select res_style
              from fm_html_styles) || '<body>';

  :db_id = f_get_db_id();

  select c_name_fio, amount, doc_no_tmp, ol_date_tmp,
    case arci.arci_return_reason
        when 0 then 'ошибочно внесенной суммой'
        when 1 then 'отказом от получения услуг'
    end,
    coalesce(k.k_kontract_no, ''),
    arci.arci_sign_type
  from print_app_consult_info(:arc_id)
  inner join app_return_consult_info arci on arci.arci_id = :arc_id
  inner join app_return_consult arc on arc.arc_id = arci.arci_id
  inner join kontract k on arc.arc_from_k_id = k.k_id
  into :c_fio, :amount, :oper_no, :pay_date, :return_reason, :k_no, :sign_type;

  if (sign_type = 0) then
    begin
        confirm_txt = 'Отправить СМС';
        message_conf = 'Отправить ссылку на подписание через СМС?';
    end
  if (sign_type = 1) then
    begin
        confirm_txt = 'Утвердить';
        message_conf = 'Утвердить заявление?';
    end

  res_text = BLOB_APPEND(res_text, '<h1 class="green_text">' || :k_no || '</h1>');

  res_text = BLOB_APPEND(res_text, 'Я, <b>' || :c_fio || '</b><br>' ||
    'Прошу вернуть внесенную мной сумму за консультационные услуги по квитанции на взнос наличными №<b>' ||
    :oper_no || '</b> от <b>' || :pay_date || '</b> в размере <b>' || :amount ||
    '</b> сом в связи с <b>' || :return_reason || '</b>.<br>');


  res_text = BLOB_APPEND(res_text, '<h1>' || f_uri_sql(f_lang('ru', :confirm_txt), 'confirm_return_consultation', 'app_return_consult', :arc_id, 'none', f_lang('ru', :message_conf),'',0,0,'','','dialog','')
        || '</h1>');

  res_text = BLOB_APPEND(res_text, '<h1>' || f_uri_sql(f_lang('ru', 'Запретить'), 'decline_return_consultation', 'app_return_consult', :arc_id, 'none', f_lang('ru', 'Запертить возврат?'),'',0,0,'','','dialog','')
        || '</h1>');

  res_text = BLOB_APPEND(res_text, '</body>');

  suspend;
end
