CREATE PROCEDURE AAREP_ACT_OVERPAYMENT(
    REP_DATE_END DATE)
RETURNS (
    ACC_ID D_ID,
    ACC_CODE D_STR_CODE,
    ACC_OBJECT_ID D_ID,
    KOBS_BAL_SUM CURR_TYPE,
    KOBS_DATE DATE,
    K_KONTRACT_NO STR_20,
    LOAN_STATUS STR_10,
    CLIENT_NAME_FIO STR_100,
    C_NAME_FIO STR_100,
    CLIENT_TYPE D_STR_20)
AS
declare variable kobs_dt_sum curr_type;
declare variable kobs_kt_sum curr_type;
declare variable k_id id_type;
declare variable c_id id_type;
declare variable pr_code d_str_100;
begin
  for select acc_id, acc_code, acc_object_id, -kobs_bal_sum, kobs_date, kobs_dt_sum, kobs_kt_sum
      from ref_account, kn_obsummary
      where acc_from_sab_id = 53 and
            kobs_acc_id = acc_id and
            :rep_date_end between kobs_date and kobs_date_end and
            kobs_bal_sum <> 0 and
            acc_object_kind + 0 = 8 and
            acc_object_id >= 9
            --  rows 1000
      into acc_id, acc_code, acc_object_id, kobs_bal_sum, kobs_date, kobs_dt_sum, kobs_kt_sum
  do
  begin
    if (kobs_dt_sum = 0 and
        kobs_kt_sum = 0) then
    begin
      select kobs_date
      from kn_obsummary
      where :kobs_date - 1 between kobs_date and kobs_date_end and
            kobs_bal_sum <> 0 and
            kobs_acc_id = :acc_id
      into kobs_date;
    end

    if (kobs_date < '01.01.2018') then
    begin
      kobs_date = null; -- текущее поле не должно возвращать null, еслин null то переплаты не должно быть
      select max(ok_ol_date)
      from kn_operkred
      where ok_from_kc_id = :acc_object_id and
            ok_ol_is_active = 1 and
            ok_ol_date <= :rep_date_end and
            ok_type = 9 and
            ok_os <> 0
      into kobs_date;
    end

    select k_kontract_no, k_id, kc_from_c_id, p.pr_code
    from knclient, kontract k, product p
    where kc_id = :acc_object_id and
          p.pr_id = k.k_from_pr_id and
          kc_from_k_id = k_id
    into k_kontract_no, k_id, c_id, pr_code;

    client_type = 'ФЛ';
    if (pr_code containing '.selfemployed.') then
      client_type = 'СЗ';

    --    acc_code = f_fmt_acc(acc_id);

    select iif(ks_state = 1, 'Активный', 'Закрытый')
    from kontstate
    where ks_from_k_id = :k_id and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_ol_is_delete = 0
    into loan_status;

    client_name_fio = null;
    select c_name_fio
    from clients c
    where c_id = :c_id rows 1
    into client_name_fio;

    c_name_fio = 'n' || :c_id;

    suspend;
  end
end
