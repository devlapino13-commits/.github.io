CREATE PROCEDURE AAREP_HR_STATISTIC(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PERIOD_STR VARCHAR(50),
    JOB_TITLE VARCHAR(50),
    IS_HIRING INTEGER,
    IS_DISMISSED INTEGER,
    IS_TRANSFER INTEGER,
    IS_REBUKE INTEGER,
    IS_VACATION INTEGER,
    KOL_TRAININGS INTEGER)
AS
declare variable mm integer;
declare variable yy integer;
declare variable date_begin date;
declare variable date_end date;
declare variable k integer;
begin
  date_begin = rep_date_begin;
  date_end = dateadd(month, 1, date_begin - extract(day from date_begin) + 1) - 1; -- вычисляется конец месяца
  k = 0;
  while (date_begin <= rep_date_end) do
  begin
    date_end = dateadd(month, 1, date_begin - extract(day from date_begin) + 1) - 1; -- вычисляется конец месяца
    if (date_end > rep_date_end) then
      date_end = rep_date_end;

    if (extract(day from date_begin) = 1 and
        date_end = dateadd(month, 1, date_begin - extract(day from date_begin) + 1) - 1) then
    begin
      mm = extract(month from date_begin);
      yy = extract(year from date_begin);
      period_str = case mm
                     when 1 then 'Январь'
                     when 2 then 'Февраль'
                     when 3 then 'Март'
                     when 4 then 'Апрель'
                     when 5 then 'Май'
                     when 6 then 'Июнь'
                     when 7 then 'Июль'
                     when 8 then 'Август'
                     when 9 then 'Сентябрь'
                     when 10 then 'Октябрь'
                     when 11 then 'Ноябрь'
                     else 'Декабрь'
                   end;
      period_str = period_str || ' ' || yy;
    end
    else
    begin
      period_str = right((extract(day from date_begin) + 100), 2) || '.' || right((extract(month from date_begin) + 100), 2) || '.' || extract(year from date_begin) || ' - ' || right((extract(day from date_end) + 100), 2) || '.' || right((extract(month from date_end) + 100), 2) || '.' || extract(year from date_end);
    end
    k = k + 1;
    if (k = 10) then
      k = 90;
    if (k = 100) then
      k = 900;
    period_str = k || '. ' || period_str;
    for select job_title, sum(is_hiring), sum(is_dismissed), sum(is_transfer), sum(is_rebuke), sum(is_vacation),
               sum(kol_trainings)
        from aarep_hr_statistic_temp(:rep_region_id, :date_begin, :date_end)
        group by job_title
        into :job_title, :is_hiring, :is_dismissed, :is_transfer, :is_rebuke, :is_vacation, :kol_trainings
    do
    begin
      suspend;
    end
    date_begin = date_end + 1;
  end
end
