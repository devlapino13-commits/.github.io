CREATE PROCEDURE AAREP_PAYMENTS_VIA_MONETA(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_REGION_ID D_ID)
RETURNS (
    OFFICE VARCHAR(50),
    PAY_DATE D_DATE,
    PAY_AMOUNT DECIMAL(18,4),
    KONTRACT_NO D_STR_20,
    KREDIT_FOLDER D_STR_100,
    CLIENT_FIO D_STR_100,
    FIELD_VALUE VARCHAR(500),
    WORKER_FIO D_STR_100)
AS
declare variable esm_full_source blob sub_type 1 segment size 80;
declare variable fc varchar(1000);
declare variable fv varchar(1000);
declare variable v_card varchar(1000);
declare variable v_phone varchar(1000);
declare variable v_fio varchar(1000);
declare variable v_bank varchar(1000);
begin
  -- Процедура для отчета по погашениям через Монету с деталями
  -- RFCRU4431
  for select r.reg_name,
             k.k_kontract_no, --Номер контракта
             exsm.esm_date, --Дата Платежа
             exsm.esm_amount, --Сумма платежа
             exsm.esm_full_source,
             cl.c_name_fio, --ФИО клиента
             lo.c_name_fio, --ФИО кс
             kal.kal_new_kf_name --Номер кредитной папки
      from extsysmain exsm
      left join clients cl on cl.c_id = exsm.esm_object_id
      left join kontract k on k.k_id = exsm.esm_from_k_id
      join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
      left join region r on r.reg_unicode = kal.kal_new_reg_id
      join region_list(:rep_region_id, 1, 3, 1) on kal.kal_new_reg_id + 0 = id_list
      left join clients lo on lo.c_id = kal.kal_new_kf_officer --витыскиваю фио кс
      where exsm.esm_action = 'pay' and --Оплата
            exsm.esm_status = 1 and
            exsm.esm_date between :rep_date_begin and :rep_date_end and
            exsm.esm_project = 'moneta' and --через монету
            kal.is_delete = 0 and
            :rep_date_end between kal.kal_date and kal.kal_date_end
      into :office,
           :kontract_no,
           :pay_date,
           :pay_amount,
           :esm_full_source,
           :client_fio,
           :worker_fio,
           :kredit_folder
  do
  begin
    v_card = null;
    v_phone = null;
    v_fio = null;
    v_bank = null;
    for select field_code,
               field_value
        from hp_extsys_parcer(:esm_full_source, '')
        into :fc,
             :fv
    do
    begin
      if (fc = 'cardnumber') then
        v_card = fv;
      else if (fc = 'fioplat') then
        v_fio = fv;
      else if (fc = 'sbpphone') then
        v_phone = fv;
      else if (fc = 'sbpbank') then
        v_bank = fv;
    end

    if (v_card is not null) then
      field_value = v_card;
    else
      field_value = coalesce(v_fio, '') || ' | ' || coalesce(v_phone, '') || ' | ' || coalesce(v_bank, '');

    suspend;
  end
end
