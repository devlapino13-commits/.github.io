CREATE PROCEDURE HP_INCOME_EXPENSES_SAVE(
    K_ID ID_TYPE)
AS
declare variable KC_ID ID_TYPE;
declare variable C_ID ID_TYPE;
declare variable OVS_VALUE STR_100;
declare variable KPD decimal(15,2);
declare variable I integer;
declare variable ERROR_COD STR_100;
declare variable NET_PROFIT decimal(15,2);
declare variable LIVING_EXPENSES decimal(15,2);
declare variable MONTHLY_INCOME decimal(15,2);
--declare variable UNFORESEEN_EXPENSES decimal(15,2);
declare variable LEGAL_DOCS_EXPENSES decimal(15,2);
declare variable OTHER_BUSINESS_EXPENSES decimal(15,2);
declare variable EXTRA_AMOUNT decimal(15,2);
declare variable AMOUNT_MONTHLY_PAYMENTS decimal(15,2);
declare variable REPAYMENT_OBLIGATIONS decimal(15,2);
begin
  for select knc.kc_id, knc.kc_from_c_id
      from knclient knc
      where knc.kc_from_k_id = :k_id and
            knc.kc_amount_approved > 0
      into :kc_id, :c_id
  do
  begin

------------------------------------------------------------------------------------------------
    ovs_value = null;
    net_profit = null;
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'net_profit', current_date)
    into :ovs_value; -- Чистая прибыль
    i = 1;
    OVS_VALUE = replace(OVS_VALUE, 'Значение: ', '');
    OVS_VALUE = replace(OVS_VALUE, 'Значение', '');
    if (OVS_VALUE <> '') then
    begin
      net_profit = cast(OVS_VALUE as decimal);
      execute procedure hp_objectprops_set(8, :kc_id, -1, 'ClientIncome.NetProfit', current_date,
          net_profit, null, 1000000000000);
    end
    ----------------------------------------------------------------------------------------------------
    ovs_value = null;
    living_expenses = null;
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'living_expenses', current_date)
    into :ovs_value; -- Ежемесячные семейные расходы
    i = 2;
    OVS_VALUE = replace(OVS_VALUE, 'Значение: ', '');
    OVS_VALUE = replace(OVS_VALUE, 'Значение', '');

    if (OVS_VALUE <> '') then
    begin
      living_expenses = cast(OVS_VALUE as decimal);
      execute procedure hp_objectprops_set(8, :kc_id, -1, 'ClientIncome.LivingExpenses', current_date,
          living_expenses, null, 1000000000000);
    end

    ---------------------------------------------------------------------------------------------------
    ovs_value = null;
    monthly_income = null;
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'monthly_income', current_date)
    into :ovs_value; -- Ежемесячный основной доход
    i = 3;
    OVS_VALUE = replace(OVS_VALUE, 'Значение: ', '');
    OVS_VALUE = replace(OVS_VALUE, 'Значение', '');

    if (OVS_VALUE <> '') then
    begin
      monthly_income = cast(OVS_VALUE as decimal);
      execute procedure hp_objectprops_set(8, :kc_id, -1, 'ClientIncome.MonthlyIncome', current_date,
          monthly_income, null, 1000000000000);
    end

    ---------------------------------------------------------------------------------------------------
    /*
    ovs_value = null;
    unforeseen_expenses = null;

    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'AnketaQuestions.unforeseen_expenses', current_date)
    into :ovs_value; -- Ежемесячные накопления на непредвиденные будущие расходы
    i = 4;
    OVS_VALUE = replace(OVS_VALUE, 'Значение: ', '');
    OVS_VALUE = replace(OVS_VALUE, 'Значение', '');

    if (OVS_VALUE <> '') then
    begin
      unforeseen_expenses = cast(OVS_VALUE as decimal);
      execute procedure hp_objectprops_set(8, :kc_id, -1, 'ClientIncome.UnforeseenExpenses', current_date,
          unforeseen_expenses, null, 1000000000000);
    end
    */
    ---------------------------------------------------------------------------------------------------

    ovs_value = null;
    legal_docs_expenses = null;

    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'AnketaQuestions.legal_docs_expenses', current_date)
    into :ovs_value; -- Расходы на патент и соц. полис (в месяц) 
    i = 5;
    OVS_VALUE = replace(OVS_VALUE, 'Значение: ', '');
    OVS_VALUE = replace(OVS_VALUE, 'Значение', '');

    if (OVS_VALUE <> '') then
    begin
      legal_docs_expenses = cast(OVS_VALUE as decimal);
      execute procedure hp_objectprops_set(8, :kc_id, -1, 'ClientIncome.LegalDocsExpenses', current_date,
          legal_docs_expenses, null, 1000000000000);
    end

    ---------------------------------------------------------------------------------------------------

    ovs_value = null;
    Other_business_expenses = null;

    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'AnketaQuestions.Other_business_expenses', current_date)
    into :ovs_value; -- Другие ежемесячные расходы по бизнесу
    i = 6;
    OVS_VALUE = replace(OVS_VALUE, 'Значение: ', '');
    OVS_VALUE = replace(OVS_VALUE, 'Значение', '');

    if (OVS_VALUE <> '') then
    begin
      Other_business_expenses = cast(OVS_VALUE as decimal);
      execute procedure hp_objectprops_set(8, :kc_id, -1, 'ClientIncome.OtherBusinessExpenses', current_date,
          Other_business_expenses, null, 1000000000000);
    end

    ---------------------------------------------------------------------------------------------------
    ovs_value = null;
    extra_amount = null;

    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'extra_amount', current_date)
    into :ovs_value; -- Другие ежемесячные расходы по бизнесу
    i = 7;
    OVS_VALUE = replace(OVS_VALUE, 'Значение: ', '');
    OVS_VALUE = replace(OVS_VALUE, 'Значение', '');

    if (OVS_VALUE <> '') then
    begin
      extra_amount = cast(OVS_VALUE as decimal);
      execute procedure hp_objectprops_set(8, :kc_id, -1, 'ClientIncome.ExtraAmount', current_date,
          extra_amount, null, 1000000000000);
    end
    ---------------------------------------------------------------------------------------------------
    ovs_value = null;
    amount_monthly_payments = null;

    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'amount_monthly_payments', current_date)
    into :ovs_value; -- сумма ежемесчной выплаты
    i = 8;
    OVS_VALUE = replace(OVS_VALUE, 'Значение: ', '');
    OVS_VALUE = replace(OVS_VALUE, 'Значение', '');

    if (OVS_VALUE <> '') then
    begin
      amount_monthly_payments = cast(OVS_VALUE as decimal);
      execute procedure hp_objectprops_set(8, :kc_id, -1, 'ClientIncome.AmountMonthlyPayments', current_date,
          amount_monthly_payments, null, 1000000000000);
    end

    ---------------------------------------------------------------------------------------------------

    ovs_value = null;
    repayment_obligations = null;

    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'AnketaQuestions.repayment_obligations', current_date)
    into :ovs_value; -- Погашение других долговых обязательств
    i = 9;
    OVS_VALUE = replace(OVS_VALUE, 'Значение: ', '');
    OVS_VALUE = replace(OVS_VALUE, 'Значение', '');

    if (OVS_VALUE <> '') then
    begin
      repayment_obligations = cast(OVS_VALUE as decimal);
      execute procedure hp_objectprops_set(8, :kc_id, -1, 'ClientIncome.RepaymentObligations', current_date,
          repayment_obligations, null, 1000000000000);
    end
    ---------------------------------------------------------------------------------------------------
    if ((:net_profit is not null) and (amount_monthly_payments is not null) and (amount_monthly_payments>0)) then
    begin
      kpd = (net_profit*100)/amount_monthly_payments;                -- считаем КПД продукта
      execute procedure hp_objectprops_set(8, :kc_id, -1, 'ClientIncome.KPD', current_date,
          kpd, null, 1000000000000);
    end
    when any do               -- ловим ошибку связанную с вводом некорректных символов
      begin
        if (i = 1) then error_cod = 'net_profit';
        if (i = 2) then error_cod = 'living_expenses';
        if (i = 3) then error_cod = 'monthly_income';
        --if (i = 4) then error_cod = 'AnketaQuestions.unforeseen_expenses';
        if (i = 5) then error_cod = 'AnketaQuestions.legal_docs_expenses';
        if (i = 6) then error_cod = 'AnketaQuestions.Other_business_expenses';
        if (i = 7) then error_cod = 'extra_amount';
        if (i = 8) then error_cod = 'amount_monthly_payments';
        if (i = 9) then error_cod = 'AnketaQuestions.repayment_obligations';

        exception e_default 'Error: c_id = '||:c_id||' answer for question ' ||error_cod|| ' cant convert to int';
      end
  end
end
