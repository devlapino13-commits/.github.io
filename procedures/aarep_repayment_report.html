CREATE PROCEDURE AAREP_REPAYMENT_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_PROJECT VARCHAR(100))
RETURNS (
    AA_K_ID DECIMAL(18,0),
    AAREP_BRANCH_NAME VARCHAR(100),
    AAREP_OFFICE_NAME VARCHAR(100),
    AAREP_LO_NAME VARCHAR(100),
    AAREP_LOAN_KIND VARCHAR(100),
    AAREP_LOAN_NAME VARCHAR(100),
    AAREP_ACTIVE_BALANCE_B DECIMAL(18,4),
    AAREP_ACTIVE_KOLICH_CLIENT_B INTEGER,
    AAREP_ACTIVE_BALANCE_E DECIMAL(18,4),
    AAREP_ACTIVE_KOLICH_CLIENT_E INTEGER,
    AAREP_DISB_AMOUNT DECIMAL(18,4),
    AAREP_DISB_KOLICH_CLIENT INTEGER,
    AAREP_TRANS_OS DECIMAL(18,4),
    AAREP_TRANS_PERC DECIMAL(18,4),
    AAREP_TRANS_FINE DECIMAL(18,4),
    AAREP_TRANS_KOMMIS DECIMAL(18,4),
    AAREP_WRITEOFF_OS DECIMAL(18,4),
    AAREP_WRITEOFF_PERC DECIMAL(18,4),
    AAREP_WRITEOFF_FINE DECIMAL(18,4),
    AAREP_DELIQ_OS DECIMAL(18,4),
    AAREP_DELIQ_PERC DECIMAL(18,4),
    AAREP_DELIQ_BALANCE DECIMAL(18,4),
    AAREP_DELIQ_KOLICH_CLIENT INTEGER,
    AAREP_DELIQ_BALANCE7 DECIMAL(18,4),
    AAREP_DELIQ_KOLICH_CLIENT7 INTEGER,
    AAREP_DELIQ_BALANCE30 DECIMAL(18,4),
    AAREP_DELIQ_KOLICH_CLIENT30 INTEGER,
    AAREP_DELIQ_BALANCE60 DECIMAL(18,4),
    AAREP_DELIQ_KOLICH_CLIENT60 INTEGER,
    AAREP_DELIQ_BALANCE90 DECIMAL(18,4),
    AAREP_DELIQ_KOLICH_CLIENT90 INTEGER,
    AAREP_DELIQ_BALANCE180 DECIMAL(18,4),
    AAREP_DELIQ_KOLICH_CLIENT180 INTEGER,
    AAREP_DELIQ_BALANCE360 DECIMAL(18,4),
    AAREP_DELIQ_KOLICH_CLIENT360 INTEGER,
    AAREP_COLL_PLAN_OS DECIMAL(18,4),
    AAREP_COLL_FACT_OS DECIMAL(18,4),
    AA_LOAN_AMOUNT DECIMAL(18,4),
    AA_DISB_DATE DATE,
    AA_KONTRACT_NO VARCHAR(20),
    AA_LEADER_NAME VARCHAR(100),
    AA_CIRCLE INTEGER,
    AA_PERIOD INTEGER,
    AA_CASHIER_NAME VARCHAR(100),
    AA_LEADER_ADDRESS VARCHAR(255),
    AA_REPAY_TOTAL_OS DECIMAL(18,4),
    AA_KF_NAME VARCHAR(100),
    AA_K_FINANCED INTEGER)
AS
declare variable disb_cl decimal(18,4);
declare variable st_staff integer;
declare variable st_client integer;
declare variable c_id decimal(18,0);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  if (rep_project is null) then
    rep_project = '';
  for select k_financed, aa_k_id, aa_loan_amount, k_disb_date, k_kontract_no, c1.c_name_fio, c2.c_name_fio,
             k_circle + k_circle_internal - 1, k_kolich_repay, c1.c_id, aa_branch_name, aa_office_name, aa_lo_name,
             aa_loan_kind, aa_loan_name, aa_kf_name,

             sum(aa_active_balance_b), sum(aa_active_kolich_client_b), sum(aa_active_balance_e),
             sum(aa_active_kolich_client_e), sum(aa_disb_amount), sum(aa_disb_kolich_client), sum(aa_trans_os),
             sum(aa_trans_perc), sum(aa_trans_fine), sum(aa_trans_kommis), sum(aa_writeoff_os), sum(aa_writeoff_perc),
             sum(aa_writeoff_fine),

             sum(aa_deliq_os), sum(aa_deliq_perc),

             sum(aa_deliq_balance), sum(aa_deliq_kolich_client),

             sum(aa_deliq_balance7), sum(aa_deliq_kolich_client7),

             sum(aa_deliq_balance30), sum(aa_deliq_kolich_client30),

             sum(aa_deliq_balance60), sum(aa_deliq_kolich_client60),

             sum(aa_deliq_balance90), sum(aa_deliq_kolich_client90),

             sum(aa_deliq_balance180), sum(aa_deliq_kolich_client180),

             sum(aa_deliq_balance360), sum(aa_deliq_kolich_client360),

             sum(aa_coll_plan_os), /* coll_plan_os */
             sum(aa_coll_fact_os) /* coll_fact_os */

      from (kontract
      left join clients c1 on (k_clients_leader_id = c1.c_id)
      left join clients c2 on (k_clients_cashier_id = c2.c_id)), aarep_repayment_reptmp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_project)
      where aa_k_id = k_id
      group by k_financed, aa_k_id, aa_loan_amount, k_disb_date, k_kontract_no, c1.c_name_fio, c2.c_name_fio, k_circle, k_circle_internal, k_kolich_repay, c1.c_id, aa_branch_name, aa_office_name, aa_lo_name, aa_loan_kind, aa_loan_name, aa_kf_name
      into :aa_k_financed, :aa_k_id, :aa_loan_amount, :aa_disb_date, :aa_kontract_no, :aa_leader_name, :aa_cashier_name,
           :aa_circle, :aa_period, :c_id, :aarep_branch_name, :aarep_office_name, :aarep_lo_name, :aarep_loan_kind,
           :aarep_loan_name, :aa_kf_name, :aarep_active_balance_b, :aarep_active_kolich_client_b,
           :aarep_active_balance_e, :aarep_active_kolich_client_e, :aarep_disb_amount, :aarep_disb_kolich_client,
           :aarep_trans_os, :aarep_trans_perc, :aarep_trans_fine, :aarep_trans_kommis, :aarep_writeoff_os,
           :aarep_writeoff_perc, :aarep_writeoff_fine,

           :aarep_deliq_os, :aarep_deliq_perc,

           :aarep_deliq_balance, :aarep_deliq_kolich_client,

           :aarep_deliq_balance7, :aarep_deliq_kolich_client7,

           :aarep_deliq_balance30, :aarep_deliq_kolich_client30,

           :aarep_deliq_balance60, :aarep_deliq_kolich_client60,

           :aarep_deliq_balance90, :aarep_deliq_kolich_client90,

           :aarep_deliq_balance180, :aarep_deliq_kolich_client180,

           :aarep_deliq_balance360, :aarep_deliq_kolich_client360,

           :aarep_coll_plan_os, :aarep_coll_fact_os
  do
  begin
    aa_leader_address = '';
    select adl_address_str
    from addrlist, addrclient
    where adc_address_type = 1 and
          adc_address_type = 1 and
          adc_from_c_id = :c_id and
          adc_from_adl_id = adl_id
    into :aa_leader_address;

    aa_repay_total_os = 0;
    aa_repay_total_os = :aa_loan_amount - :aarep_active_balance_e;
    suspend;
  end

end
