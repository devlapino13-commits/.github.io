CREATE PROCEDURE AAREP_CALC_PERC_KN_COMPARE(
    K_ID D_ID,
    REP_DATE D_DATE,
    KEEP_MINUS_SALDO D_BOOL = 0,
    KEEP_SALDO_ON_CLOSE D_BOOL = 0)
RETURNS (
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    OKS_BALANCE D_CURRENCY,
    OKS_BALANCE_FOR_PERC D_CURRENCY,
    ORIG_PERCENT D_PERCENT,
    ORIG_ACR_PERC D_CURRENCY,
    ORIG_BAL_PERC D_CURRENCY,
    DEBUG_ACR_PERC D_CURRENCY,
    DEBUG_BAL_PERC D_CURRENCY,
    DIFF_ACR_PERC D_CURRENCY,
    DIFF_BAL_PERC D_CURRENCY,
    DEBUG_PAYED D_CURRENCY,
    DEBUG_WRITEOFF D_CURRENCY,
    DEBUG_PERC_BASE D_CURRENCY)
AS
declare variable orig_keep_minus_saldo d_bool;
begin
  if (keep_minus_saldo is null) then
    keep_minus_saldo = 0;

  orig_keep_minus_saldo = keep_minus_saldo;

  rep_date_begin = null;
  for select oks_date_begin, max(oks_percent_norm_prv), sum(oks_balance), sum(oks_bal_to_perc_prv), sum(oks_acr_perc), sum(oks_bal_perc)
  from kn_oksummary
  where oks_from_k_id = :k_id and
        oks_date_begin <= :rep_date
  group by 1
  order by 1
  into rep_date_end, orig_percent, oks_balance, oks_balance_for_perc, orig_acr_perc, orig_bal_perc
  do
  begin
    if (rep_date_end <> rep_date)
    then keep_minus_saldo = 1;
    else keep_minus_saldo = orig_keep_minus_saldo;

    debug_acr_perc = 0;
    debug_bal_perc = 0;
    debug_payed = 0;
    debug_writeoff = 0;
    orig_percent = cast(orig_percent as double precision)/ 365.0;

    if (rep_date_begin is null) then
      rep_date_begin = rep_date_end;

    select result_calced, result_perc_end, result_payed, result_writeoff, result_perc_base
    from aarep_calc_perc_one_cls(:k_id, :rep_date_begin, :rep_date_end, 0, :keep_minus_saldo, :keep_saldo_on_close)
    into debug_acr_perc, debug_bal_perc, debug_payed, debug_writeoff, debug_perc_base;

    diff_acr_perc = debug_acr_perc - orig_acr_perc;
    diff_bal_perc = debug_bal_perc - orig_bal_perc;

    if (diff_acr_perc = 0) then
      diff_acr_perc = null;

    if (diff_bal_perc = 0) then
      diff_bal_perc = null;

    suspend;

    rep_date_begin = rep_date_end + 1;
  end
end
