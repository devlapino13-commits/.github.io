CREATE PROCEDURE UXP$GRANT_PERM_TO_ROLE(
    ROLE_NAME D_STR_CODE)
AS
declare variable object_name d_str_code;
declare variable access_kind d_str_code;
declare variable object_kind d_str_code;
declare variable res_text d_str_long;
declare variable full_access_list d_str_8100;
begin
  role_name = trim(role_name);
  access_kind = 'ALL';

  select list('[' || uro_object_name || ']')
  from ux$readonly
  into full_access_list;

  if (role_name in ('SEL', 'APR')) then
    access_kind = 'select';

--  res_text = 'revoke ALL on ALL from ' || role_name || ';';
--  execute statement (res_text);

  for select trim(res_str)
  from uxp$split_str('table, view, procedure, function, package, sequence, exception',',')
  into object_kind
  do
  begin
    res_text = 'grant alter any ' || object_kind || ' to role ' || role_name || ';';
    execute statement (res_text);
    when any do
    begin
    end
  end

  object_kind = '';
  for select t.rdb$relation_name
  from rdb$relations t
  where coalesce(t.rdb$system_flag, 0) = 0
  into object_name
  do
  begin
    object_name = trim(object_name);

    -- список таблиц доступных в режиме чтения
    if (object_name starting 'TEMP' or full_access_list containing '[' || object_name || ']') then
      res_text = 'grant all on ' || object_name || ' to ' || role_name || ';';
    else
      res_text = 'grant ' || access_kind || ' on ' || object_name || ' to ' || role_name || ';';

    execute statement (res_text);
    when any do
    begin
    end
  end

  access_kind = 'execute';

  object_kind = 'procedure';
  for select p.rdb$procedure_name
  from rdb$procedures p
  where coalesce(p.rdb$system_flag, 0) = 0
  into object_name
  do
  begin
    object_name = trim(object_name);

    res_text = 'grant ' || access_kind || ' on ' || object_kind || ' ' || object_name || ' to ' || role_name || ';';
    execute statement (res_text);
    when any do
    begin
    end
  end

  object_kind = 'function';
  for select f.rdb$function_name
  from rdb$functions f
  where coalesce(f.rdb$system_flag, 0) = 0
  into object_name
  do
  begin
    object_name = trim(object_name);

    res_text = 'grant ' || access_kind || ' on ' || object_kind || ' ' || object_name || ' to ' || role_name || ';';
    execute statement (res_text);
    when any do
    begin
    end
  end

  object_kind = 'package';
  for select pk.rdb$package_name
  from rdb$packages pk
  where coalesce(pk.rdb$system_flag, 0) = 0
  into object_name
  do
  begin
    object_name = trim(object_name);

    res_text = 'grant ' || access_kind || ' on ' || object_kind || ' ' || object_name || ' to ' || role_name || ';';
    execute statement (res_text);
    when any do
    begin
    end
  end

  access_kind = 'usage';
  object_kind = 'sequence';
  for select g.rdb$generator_name
  from rdb$generators g
  where coalesce(g.rdb$system_flag, 0) = 0
  into object_name
  do
  begin
    object_name = trim(object_name);

    res_text = 'grant ' || access_kind || ' on ' || object_kind || ' ' || object_name || ' to ' || role_name || ';';
    execute statement (res_text);
    when any do
    begin
    end
  end

  access_kind = 'usage';
  object_kind = 'exception';
  for select e.rdb$exception_name
  from rdb$exceptions e
  where coalesce(e.rdb$system_flag, 0) = 0
  into object_name
  do
  begin
    object_name = trim(object_name);

    res_text = 'grant ' || access_kind || ' on ' || object_kind || ' ' || object_name || ' to ' || role_name || ';';
    execute statement (res_text);
    when any do
    begin
    end
  end
end
