CREATE PROCEDURE AAREP_KNAKT_TRANSFER_LIST(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_FROM_KF_ID DECIMAL(18,0),
    K_CURRENCY VARCHAR(3),
    K_APPLICATE_DATE DATE,
    K_APPLICATE_NO INTEGER,
    K_KONTRACT_NO VARCHAR(20),
    K_KONTRACT_DATE DATE,
    K_FROM_PR_ID INTEGER,
    K_DISB_DATE DATE,
    K_AMOUNT_APPLICATE DECIMAL(18,4),
    K_STAGE INTEGER,
    K_PERIOD DECIMAL(18,2),
    K_KOLICH_LGOTA INTEGER,
    K_PERCENT DECIMAL(18,7),
    K_PERCENT_LGOTA DECIMAL(18,7),
    K_KOLICH_REPAY INTEGER,
    K_FROM_F_ID VARCHAR(20),
    K_FROM_PT_ID VARCHAR(20),
    K_STATUS INTEGER,
    K_CLOSE_DATE DATE,
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    K_PERCENT_PER_DAY DECIMAL(18,7),
    K_MEMO VARCHAR(1024),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_PERCENT_FOR_DAYS INTEGER,
    K_KOLICH_CLIENT_APPLICATE INTEGER,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_REPAY_KOLICH_CLOSED INTEGER,
    K_CLASS VARCHAR(20),
    K_KOMMIS_PERC DECIMAL(18,7),
    K_FROM_CS_ID INTEGER,
    K_CLIENTS_LEADER_ID DECIMAL(18,0),
    K_CLIENTS_CASHIER_ID DECIMAL(18,0),
    K_CREDIT_TARGET VARCHAR(255),
    K_FROM_PAC_ID INTEGER,
    K_IS_KREDLINE INTEGER,
    K_KREDLINE_END DATE,
    K_STOP_PERC DATE,
    K_STOP_FINE DATE,
    K_INDEX_CURR_CODE VARCHAR(3),
    K_INDEX_CURR_RATE DECIMAL(18,4),
    K_FROM_MA_REG_ID INTEGER,
    K_FROM_MA_APP_ID INTEGER,
    K_PERCENT_INPUT DECIMAL(18,7),
    K_PERCENT_CATEGORY DECIMAL(18,7),
    K_PERCENT_EFFECT DECIMAL(18,7),
    F_1 DECIMAL(18,4),
    K_REGION_ID INTEGER,
    K_REGION_NAME VARCHAR(100),
    K_LOAN_OFFICER_FIO VARCHAR(100),
    KAL_DATE2 DATE)
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select k_id, k_from_kf_id, k_currency, k_applicate_date, k_applicate_no, k_kontract_no, k_kontract_date,
             k_from_pr_id, k_disb_date, k_amount_applicate, k_stage, k_period, k_kolich_lgota, k_percent,
             k_percent_lgota, k_kolich_repay, k_from_f_id, k_from_pt_id, k_status, k_close_date, k_circle,
             k_circle_internal, k_percent_per_day, k_memo, k_amount_approved, k_percent_for_days,
             k_kolich_client_applicate, k_kolich_client_approved, k_repay_kolich_closed, k_class, k_kommis_perc,
             k_from_cs_id, k_clients_leader_id, k_clients_cashier_id, k_credit_target, k_from_pac_id, k_is_kredline,
             k_kredline_end, k_stop_perc, k_stop_fine, k_index_curr_code, k_index_curr_rate, k_from_ma_reg_id,
             k_from_ma_app_id, k_percent_input, k_percent_category, k_percent_effect, sum1 * cr_rate, k_region1,
             reg_code || ' ' || reg_name, c_name_fio, kont_knakt.kal_date1
      from kont_knakt(:report_date_begin, :report_date_end, :report_region_id), hp_get_balance(k_id1, :report_date_end), opercurr, kontract, region, clients
      where :report_date_end >= cr_date and
            :report_date_end <= cr_date_end and
            cr_from_cu_id = k_currency1 and
            k_id1 = kontract.k_id and
            c_id = k_kc_id and
            reg_unicode = k_region1
      into :k_id, :k_from_kf_id, :k_currency, :k_applicate_date, :k_applicate_no, :k_kontract_no, :k_kontract_date,
           :k_from_pr_id, :k_disb_date, :k_amount_applicate, :k_stage, :k_period, :k_kolich_lgota, :k_percent,
           :k_percent_lgota, :k_kolich_repay, :k_from_f_id, :k_from_pt_id, :k_status, :k_close_date, :k_circle,
           :k_circle_internal, :k_percent_per_day, :k_memo, :k_amount_approved, :k_percent_for_days,
           :k_kolich_client_applicate, :k_kolich_client_approved, :k_repay_kolich_closed, :k_class, :k_kommis_perc,
           :k_from_cs_id, :k_clients_leader_id, :k_clients_cashier_id, :k_credit_target, :k_from_pac_id, :k_is_kredline,
           :k_kredline_end, :k_stop_perc, :k_stop_fine, :k_index_curr_code, :k_index_curr_rate, :k_from_ma_reg_id,
           :k_from_ma_app_id, :k_percent_input, :k_percent_category, :k_percent_effect, :f_1, :k_region_id,
           :k_region_name, :k_loan_officer_fio, :kal_date2
  do
  begin
    suspend;
  end
end
