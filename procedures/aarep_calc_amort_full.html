CREATE PROCEDURE AAREP_CALC_AMORT_FULL(
    REG_ID INTEGER,
    DD DATE,
    IS_FULL INTEGER)
RETURNS (
    PCA_BILL_ID DECIMAL(18,0),
    PCA_BILL_LINE_NO INTEGER,
    PCA_START_YEAR INTEGER,
    PCA_START_MONTH INTEGER,
    PCA_YEAR_NO INTEGER,
    PCA_MONTH_NO INTEGER,
    PCA_AGE INTEGER,
    PCA_TOTAL_OS DECIMAL(18,4),
    PCA_TOTAL_AMORT DECIMAL(18,4),
    PCA_LIFETIME INTEGER,
    PCA_WP_ID DECIMAL(18,0),
    PCA_MO_ID DECIMAL(18,0),
    PCA_ML_ID DECIMAL(18,0),
    PCA_MC_ID DECIMAL(18,0),
    PCA_INVENT_STR VARCHAR(20),
    PCA_CURR_OS DECIMAL(18,4),
    PCA_CURR_AMORT DECIMAL(18,4),
    PCA_CURR_AMORT_OTHER DECIMAL(18,4),
    PCA_LIFETIME_SALDO INTEGER,
    PCA_BEGIN_OS DECIMAL(18,4),
    PCA_BEGIN_AMORT DECIMAL(18,4),
    PCA_COST DECIMAL(18,4),
    PCA_AMORT_ACC VARCHAR(20),
    PCA_REGION_ID INTEGER)
AS
declare variable year_no integer;
declare variable month_no integer;
declare variable bill_id_prior decimal(18,0);
declare variable bill_line_no_prior integer;
declare variable bill_region_prior integer;
begin
  /* Procedure Text */
  bill_id_prior = 0;
  bill_line_no_prior = 0;
  year_no = extract(year from dd);
  month_no = extract(month from dd);
  for select mbal_summa, mbal_bill_id, mbal_bill_line_no, pca_start_year, pca_start_month, pca_year_no, pca_month_no,
             mbal_wp_id, mbal_mo_id, mbal_ml_id, mbal_mc_id, mbal_invent_str, mbal_region_id, pca_age, mc_acc_amort,
             sum(pca_total_os), sum(pca_total_amort), sum(pca_lifetime), sum(pca_curr_os), sum(pca_curr_amort),
             sum(pca_curr_amort_other), sum(pca_lifetime_saldo) - 1
      from matlist_balance_list(:reg_id, 0, :dd, 1, 1)
      left join aarep_calc_amort_temp(:reg_id, :year_no, :month_no) on (pca_bill_id = mbal_bill_id and
            pca_bill_line_no = mbal_bill_line_no and
            pca_region_id = mbal_region_id), matcateg
      where mc_id = mbal_mc_id
      group by mbal_summa, mbal_bill_id, mbal_bill_line_no, pca_start_year, pca_start_month, pca_year_no, pca_month_no, mbal_wp_id, mbal_mo_id, mbal_ml_id, mbal_mc_id, mbal_invent_str, mbal_region_id, pca_age, mc_acc_amort
      order by mbal_bill_id, mbal_bill_line_no, 13, 1 desc
      into :pca_cost, :pca_bill_id, :pca_bill_line_no, :pca_start_year, :pca_start_month, :pca_year_no, :pca_month_no,
           :pca_wp_id, :pca_mo_id, :pca_ml_id, :pca_mc_id, :pca_invent_str, :pca_region_id, :pca_age, :pca_amort_acc,
           :pca_total_os, :pca_total_amort, :pca_lifetime, :pca_curr_os, :pca_curr_amort, pca_curr_amort_other,
           :pca_lifetime_saldo
  do
  begin
    if (pca_start_year is null) then
      pca_start_year = 0;
    if (pca_start_month is null) then
      pca_start_month = 0;
    if (pca_year_no is null) then
      pca_year_no = 0;
    if (pca_month_no is null) then
      pca_month_no = 0;
    if (pca_age is null) then
      pca_age = 0;
    if (pca_total_os is null) then
      pca_total_os = 0;
    if (pca_total_amort is null) then
      pca_total_amort = 0;
    if (pca_curr_os is null) then
      pca_curr_os = 0;
    if (pca_curr_amort is null) then
      pca_curr_amort = 0;
    if (pca_curr_amort_other is null) then
      pca_curr_amort_other = 0;
    if (pca_lifetime is null) then
      pca_lifetime = 0;
    if (pca_lifetime_saldo is null) then
      pca_lifetime_saldo = 0;

    pca_begin_os = pca_total_os - pca_curr_os;
    pca_begin_amort = pca_total_amort - pca_curr_amort - pca_curr_amort_other;
    if (pca_begin_os < 0) then
    begin
      pca_total_os = pca_curr_os;
      pca_begin_os = 0;
    end
    if (pca_begin_amort < 0) then
    begin
      pca_total_amort = pca_curr_amort;
      pca_begin_amort = 0;
    end
    if ((pca_cost <> 0) or (pca_total_os <> 0) or (pca_curr_os <> 0) or (pca_curr_amort <> 0) or (pca_curr_amort_other <> 0)) then
    begin
      if (pca_cost = 0) then
      begin
        if ((pca_total_os > 0) and
            (pca_total_amort > 0)) then
        begin
          pca_curr_os = pca_curr_os - pca_total_os;
          pca_curr_amort_other = pca_curr_amort_other - pca_total_amort;
          pca_total_os = 0;
          pca_total_amort = 0;
        end
      end
      if ((pca_begin_os <> 0) or (pca_total_os <> 0)) then
      begin
        if (is_full = 0) then
        begin
          if ((pca_curr_os <> 0) or (pca_curr_amort <> 0)) then
          begin
            if ((bill_id_prior <> pca_bill_id) or (bill_line_no_prior <> pca_bill_line_no) or (bill_region_prior <> pca_region_id)) then
            begin
              suspend;
              bill_id_prior = :pca_bill_id;
              bill_line_no_prior = :pca_bill_line_no;
              bill_region_prior = :pca_region_id;
            end
          end
        end
        else
        begin
          if ((pca_curr_os <> 0) or (pca_total_os <> 0) or (pca_total_amort <> 0) or (pca_curr_os <> 0) or (pca_curr_amort <> 0) or (pca_curr_amort_other <> 0)) then
          begin
            if ((bill_id_prior <> pca_bill_id) or (bill_line_no_prior <> pca_bill_line_no) or (bill_region_prior <> pca_region_id)) then
            begin
              suspend;
              bill_id_prior = :pca_bill_id;
              bill_line_no_prior = :pca_bill_line_no;
              bill_region_prior = :pca_region_id;
            end
          end
        end
      end
    end
  end
end
