CREATE PROCEDURE AAREP_CALC_PERC_ONE_201302(
    KONT_ID DECIMAL(18,0),
    DATE_BEGIN DATE,
    DATE_END DATE,
    PERC_PER_DAY DECIMAL(18,7),
    CALC_METHOD INTEGER,
    DISB_AMOUNT DECIMAL(18,4),
    INCLUDE_LAST_DAY_PERC INTEGER,
    K_STOP_PERC DATE)
RETURNS (
    RESULT_BALANCE_BEGIN DECIMAL(18,4),
    RESULT_PERC_BEGIN DECIMAL(18,4),
    RESULT_CALCED DECIMAL(18,4),
    RESULT_PAYED DECIMAL(18,4),
    RESULT_WRITEOFF DECIMAL(18,4),
    RESULT_PERC_END DECIMAL(18,4),
    RESULT_BALANCE_END DECIMAL(18,4),
    RESULT_CALCED_ALL DECIMAL(18,4),
    RESULT_PAYED_ALL DECIMAL(18,4),
    RESULT_WRITEOFF_ALL DECIMAL(18,4),
    RESULT_DISB_AMOUNT DECIMAL(18,4),
    RESULT_DISB_DATE DATE,
    RESULT_CALCED_TAX DECIMAL(18,4),
    RESULT_PAYED_TAX DECIMAL(18,4),
    RESULT_WRITEOFF_TAX DECIMAL(18,4),
    RESULT_PERC_END_TAX DECIMAL(18,4),
    RESULT_CALCED_ALL_TAX DECIMAL(18,4),
    RESULT_PAYED_ALL_TAX DECIMAL(18,4),
    RESULT_WRITEOFF_ALL_TAX DECIMAL(18,4),
    RESULT_PERC_BEGIN_TAX DECIMAL(18,4),
    KONT_LAST_DATE DATE)
AS
declare variable dd date;
declare variable include_tmp integer;
begin
  /* Procedure Text */
  include_tmp = include_last_day_perc;
  if (include_last_day_perc = 2) then
    include_last_day_perc = 1;
  for select result_balance_begin, result_perc_begin, result_calced, result_payed, result_writeoff, result_perc_end,
             result_balance_end, result_calced_all, result_payed_all, result_writeoff_all, result_disb_amount,
             result_disb_date, result_calced_tax, result_payed_tax, result_writeoff_tax, result_perc_end_tax,
             result_calced_all_tax, result_payed_all_tax, result_writeoff_all_tax
      from aarep_cp_one_temp_201302(:kont_id, :date_begin, :date_end, :perc_per_day, :calc_method, :disb_amount, :include_last_day_perc, :k_stop_perc)
      into :result_balance_begin, :result_perc_begin, :result_calced, :result_payed, :result_writeoff, :result_perc_end,
           :result_balance_end, :result_calced_all, :result_payed_all, :result_writeoff_all, :result_disb_amount,
           :result_disb_date, :result_calced_tax, :result_payed_tax, :result_writeoff_tax, :result_perc_end_tax,
           :result_calced_all_tax, :result_payed_all_tax, :result_writeoff_all_tax
  do
  begin
    if (include_tmp = 1) then
    begin
      result_perc_begin = 0;
      dd = :date_begin - 1;
      select result_perc_end, result_perc_end_tax, result_balance_end
      from aarep_cp_one_temp_201302(:kont_id, :dd - 1, :dd, :perc_per_day, :calc_method, :disb_amount, :include_last_day_perc, :k_stop_perc)
      into :result_perc_begin, :result_perc_begin_tax, :result_balance_begin;
      if ((result_perc_begin > 0) and
          (result_balance_begin <= 0)) then
        result_perc_begin = 0;
      if ((result_perc_begin_tax > 0) and
          (result_balance_begin <= 0)) then
        result_perc_begin_tax = 0;
      if ((result_perc_end > 0) and
          (result_balance_end <= 0)) then
        result_perc_end = 0;
      if ((result_perc_end_tax > 0) and
          (result_balance_end <= 0)) then
        result_perc_end_tax = 0;

      result_perc_begin = cast(result_perc_begin as decimal(18,2));
      result_perc_end = cast(result_perc_end as decimal(18,2));
      result_calced = result_perc_end - result_perc_begin + (result_payed + result_writeoff);
      result_calced_all = result_perc_end + result_payed_all + result_writeoff_all;

      result_perc_begin_tax = cast(result_perc_begin_tax as decimal(18,2));
      result_perc_end_tax = cast(result_perc_end_tax as decimal(18,2));
      result_calced_tax = result_perc_end_tax - result_perc_begin_tax + (result_payed_tax + result_writeoff_tax);
      result_calced_all_tax = result_perc_end_tax + result_payed_all_tax + result_writeoff_all_tax;
    end
    if ((result_perc_end > 0) and
        (result_balance_end <= 0)) then
      result_perc_end = 0;
    if ((result_perc_end_tax > 0) and
        (result_balance_end <= 0)) then
      result_perc_end_tax = 0;

    result_perc_end = cast(result_perc_end as decimal(18,2));
    result_calced = result_perc_end - result_perc_begin + (result_payed + result_writeoff);
    result_calced_all = result_perc_end + result_payed_all + result_writeoff_all;

    result_perc_end_tax = cast(result_perc_end_tax as decimal(18,2));
    result_calced_tax = result_perc_end_tax - result_perc_begin_tax + (result_payed_tax + result_writeoff_tax);
    result_calced_all_tax = result_perc_end_tax + result_payed_all_tax + result_writeoff_all_tax;

    suspend;
  end
end
