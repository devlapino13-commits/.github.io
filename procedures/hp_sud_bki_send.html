CREATE PROCEDURE HP_SUD_BKI_SEND(
    SUB_ID D_ID)
RETURNS (
    ERROR_TEXT D_STR_CODE)
AS
declare variable k_id d_id;
declare variable sml_decision_date d_date;
declare variable sub_status d_smallint;
begin
  error_text = '';

  select smr_from_k_id, sml_decision_date, sub_status
  from sud_bki, sud_loanclient, sud_mainreq, sud_mainlist
  where sub_id = :sub_id
    and slc_id = sub_from_slc_id
    and smr_id = slc_from_smr_id
    and sml_id = smr_id
  into k_id, sml_decision_date, sub_status;

  if (error_text = '' and sub_status > 0) then
    error_text = 'Уже отправлено в модуль БКИ';

  if (error_text = '' and k_id is null) then
    error_text = 'Подходящая запись в модуле ДРПВ не найдена';

  if (error_text = '' and sml_decision_date is null) then
    error_text = 'Дата решения не выставлена';

  if (error_text = '') then
  begin
    execute procedure hp_bki_add_to_buffer('.hp_sud_bki_send.', 'kontract', :k_id, :sml_decision_date);

    update sud_bki
    set sub_status = 1,
        sub_from_bev_id = null
    where sub_id = :sub_id;
  end

  suspend;
end
