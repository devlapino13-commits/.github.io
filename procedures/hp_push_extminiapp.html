CREATE PROCEDURE HP_PUSH_EXTMINIAPP(
    REG_ID INTEGER,
    LO_ID ID_TYPE = null,
    C_ID ID_TYPE = null)
AS
declare variable LO_ID2 ID_TYPE;
declare variable SENT_TEXT STR_255;
declare variable CLIENT_FIO STR_100;
declare variable LO_FIO D_STR_100;
begin
  client_fio = null;
  if (c_id is not null) then
  begin
    select '. Клиент:'||c_name_fio from clients
    where c_id = :c_id
    into client_fio;
  end 
  client_fio = coalesce(client_fio,'');
  -------- при подаче онлайн-заявки направляю пуш-уведомление КМ -------------->
  for select ps_from_c_id
  from hr_pstatus, joblist, jobkind
  where current_date between ps_date_begin and ps_date_end
  and ps_is_working = 1
  and ps_status = 1
  and ps_from_reg_id = :reg_id
  and ps_from_jl_id = jl_id
  and jl_kind = jk_id
  and jk_system_code = 'JobSystem.Region_KM'
  into lo_id2
  do
  begin
    sent_text = lo_id2||'; Подана онлайн-заявка'||client_fio;
    if (lo_id is not null) then
    begin
        select cl.c_name_fio
        from clients cl
        where cl.c_id = :lo_id
        into :lo_fio;
        sent_text = sent_text||'. За ВИПС: '||lo_fio;
    end
    execute procedure fsrobot_insert_event('DoMailing', 'push_sent_app', left(:sent_text, 255), 1000000000000);
  end
  -----------------------------------------------------------------------------
  ------- при подаче онлайн-заявки направляю пуш-уведомление ВИПС ------------->
  if (lo_id is null) then -- если заявка не села на конкретного ВИПС-а, отправляем уведомление всем ВСИП-ам офиса
  begin
    if (f_get_db_id() <> 81001 or coalesce(rdb$get_context('USER_TRANSACTION', 'extminiapp_no_push'), 0) = 0) then
    begin
      for select ps_from_c_id
      from hr_pstatus, joblist, jobkind
      where current_date between ps_date_begin and ps_date_end
      and ps_is_working = 1
      and ps_status = 1
      and ps_from_reg_id = :reg_id
      and ps_from_jl_id = jl_id
      and jl_kind = jk_id
      and jk_system_code = 'JobSystem.LO_Standart'
      into lo_id2
      do
      begin
        sent_text = lo_id2||'; Подана онлайн-заявка, нужно обработать'||client_fio;
        execute procedure fsrobot_insert_event('DoMailing', 'push_sent_app', :sent_text, 1000000000000);
      end
    end
  end else -- если села на конкретного
  begin
    sent_text = lo_id||'; За Вами закреплена новая онлайн-заявка, нужно обработать'||client_fio;
    execute procedure fsrobot_insert_event('DoMailing', 'push_sent_app', :sent_text, 1000000000000);
  end
end
