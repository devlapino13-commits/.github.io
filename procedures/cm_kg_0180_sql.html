CREATE PROCEDURE CM_KG_0180_SQL(
    K_ID ID_TYPE NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable k_obj_kind d_integer;
declare variable prod_str d_str_100;
declare variable pr_code d_str_100;
begin
  k_obj_kind = f_objkind('kontract', 'id');
  res_error_is_critical = 1;
  res_error_text = 'Удобрение не соответствует продукту. Должен быть: ';

  for select (case when pr_code containing '.superfosfat.' then 'Суперфосфат (Каз)'
                   when pr_code containing '.tashkura.' then 'Суперфосфат из фосфоритов Ташкура'
                   when pr_code containing '.ammofos.' then 'Аммофос'
                   when pr_code containing '.azofoska.' then 'Нитроаммофоска (азофоска)'
                 --  when pr_code containing '.selitra.' then 'Селитра аммиачная (Каз)'
                 --  when pr_code containing '.selitrauz.' then 'Селитра аммиачная (Узбекистан)'
                   when pr_code containing '.Afu.' then 'Азотно-фосфорное удобрение (АФУ)'
                   when pr_code containing '.diammofoska.' then 'Удобрение азотно-фосфорно-калийное (диаммофоска)'
                   when pr_code containing '.selitrarf.' then 'Селитра аммиачная (РФ)'
                --   when pr_code containing '.karbamid.' then 'Карбамид (Узбекистан)'
               end), pr.pr_code
      from kontract k
      inner join product pr on pr.pr_id = k.k_from_pr_id and
            (pr.pr_code containing '.superfosfat.' or
             pr.pr_code containing '.tashkura.' or
             pr.pr_code containing '.ammofos.' or
             pr.pr_code containing '.azofoska.' or
          --   pr.pr_code containing '.selitra.' or
          --   pr.pr_code containing '.selitrauz.' or
             pr.pr_code containing '.Afu.' or
             pr.pr_code containing '.diammofoska.' or
             pr.pr_code containing '.selitrarf.')
           --  pr.pr_code containing '.karbamid.')
      where k.k_id = :k_id
      into prod_str, pr_code
  do
  begin
    prod_str = trim(prod_str);

    if (not exists(select *
                     from kontmat kml
                     inner join matlist ml on ml.ml_id = kml.kml_from_ml_id
                     where ml.ml_name containing :prod_str and
                           kml.kml_object_kind = :k_obj_kind and
                           kml.kml_object_id = :k_id)) then
      begin
        res_error_text = res_error_text || prod_str;
        suspend;
      end
  end
end
