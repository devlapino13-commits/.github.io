CREATE PROCEDURE HP_ACTIONLIST_ZALOGAUTO_ADD(
    ZAP_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_ACL_ID D_ID,
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable kf_id d_id;
declare variable reg_id d_integer;
declare variable k_id d_id;
declare variable ma_id d_integer;
begin
  res_is_error = 0;
  res_text = '';

  select za_from_k_id, za_from_ma_id
  from zalogautoprof, zalogauto
  where zap_id = :zap_id and
        zap_from_za_id = za_id
  into k_id, ma_id;

  if (k_id is not null) then
  begin
    select k_from_kf_id from kontract
    where k_id = :k_id
    into kf_id;
  end
  else if (ma_id is not null) then
  begin
    select ma_from_kf_id
    from miniapp
    where ma_id = :ma_id
    order by ma_id desc, ma_number desc
    rows 1
    into kf_id;
  end

  select kal_new_reg_id
  from knaktlist
  where kal_from_kf_id = :kf_id and
        kal_date <= current_date and
        kal_date_end >= current_date and
        is_delete = 0
  into reg_id;


  select res_acl_id
  from hp_actionlist_insert(null, :reg_id, 'zalogautoprof', :zap_id, 'ActionGroup.ZalogAutoProf', 0, :un_id)
  into :res_acl_id;

  suspend;
end
