CREATE PROCEDURE FM_HTML_STAFF_REPAY(
    UN_ID DECIMAL(18,0),
    DATE_BEGIN DATE,
    DATE_END DATE,
    REG_ID INTEGER)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable str varchar(1024);
declare variable count_plan integer;
declare variable count_fact integer;
declare variable count_part integer;
declare variable total_plan integer;
declare variable total_fact integer;
declare variable total_part integer;
declare variable cm_database_id integer;
begin
  /*  select cm_database_id from company
  into :cm_database_id; */
  :cm_database_id = f_get_db_id();

-- Стили и заголовок таблицы
  res_text = (select res_style
              from fm_html_styles) || '<body>';
  res_text = BLOB_APPEND(res_text, f_format('<h1>%1</h1>', iif(cm_database_id <> 94401, 'Статистика по погашениям', 'Ödənişlər üzrə statistika')));

  if (coalesce(:reg_id, 0) = 0) then
  begin
    res_text = blob_append(res_text, 'Регион не указан, обратитесь к тех. поддержке');
    suspend;
    exit;
  end

  if (date_begin = date_end) then
  begin
    res_text = BLOB_APPEND(res_text, iif(cm_database_id <> 94401, 'Данные на ', 'Cari tarixə məlumat ') || right('0' || extract(day from date_end), 2) || '.' || right('0' || extract(month from date_end), 2) || '.' || extract(year from date_end));
  end
  else
  begin
    res_text = BLOB_APPEND(res_text, iif(cm_database_id <> 94401, 'Данные за период ', 'Carı dövr üzrə məlumat') || right('0' || extract(day from date_begin), 2) || '.' || right('0' || extract(month from date_begin), 2) || '.' || extract(year from date_begin) || ' - ' || right('0' || extract(day from date_end), 2) || '.' || right('0' || extract(month from date_end), 2) || '.' || extract(year from date_end));
  end
  res_text = BLOB_APPEND(res_text, f_format('<h3>(%1)</small></h2><br>', iif(cm_database_id <> 94401, 'данные на уровне группы', 'qrup üzrə məlumat')));

  total_plan = 0;
  total_fact = 0;
  total_part = 0;

  res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="60%">
       <col width="10%">
       <col width="10%">
       <col width="10%">
       <col width="10%">
       <tr>' || f_format('<th>%1</th><th>%2</th><th>%3<wbr>%4</th>', iif(cm_database_id <> 94401, 'ФИО сотрудника', 'Əməkdaşın ASA'),
                                                                     iif(cm_database_id <> 94401, 'По плану', 'Plan üzrə'),
                                                                     trim(iif(cm_database_id <> 94401, 'Опла', 'Ödəni')),
                                                                     trim(iif(cm_database_id <> 94401, 'чено', 'lib'))) ||
                f_format('<th>%1<wbr>%2</th><th>%3<wbr>%4</th>', trim(iif(cm_database_id <> 94401, 'Час', 'səi')),
                                                                 trim(iif(cm_database_id <> 94401, 'тично', 'səi')),
                                                                 trim(iif(cm_database_id <> 94401, 'Выпол', 'İcra')),
                                                                 trim(iif(cm_database_id <> 94401, 'нен', 'olunan'))) || '</tr>');
  for select '<tr>' || '<td align="left">' || c_name_fio || '</td>' || '<td align="right">' || count_plan || '</td>' || '<td align="right">' || count_fact || '</td>' || '<td align="right">' || count_part || '</td>' || '<td align="right">' || iif(count_plan > 0, round(count_fact * 100.0 / count_plan, 1), 100) || '%</td>' || '</tr>',
             count_plan, count_fact, count_part
      from aarep_staff_repay_planfact(:un_id, :date_begin, :date_end, :reg_id)
      into :str, count_plan, count_fact, count_part
  do
  begin
    total_plan = total_plan + count_plan;
    total_fact = total_fact + count_fact;
    total_part = total_part + count_part;

    res_text = BLOB_APPEND(res_text, str);
  end
  str = '<tr>' || f_format('<td align="left"><b>%1</b></td>', iif(cm_database_id <> 94401, 'Всего', 'Cəmi')) || '<td align="right"><b>' || total_plan || '</b></td>' || '<td align="right"><b>' || total_fact || '</b></td>' || '<td align="right"><b>' || total_part || '</b></td>' || '<td align="right"><b>' || iif(total_plan > 0, round(total_fact * 100.0 / total_plan, 1), 100) || '%</b></td>' || '</tr>';
  res_text = BLOB_APPEND(res_text, str);

  res_text = BLOB_APPEND(res_text, '</table></body>');
  suspend;
end
