CREATE PROCEDURE HP_SQLLISUSERS_ADD(
    UN_ID DECIMAL(18,0),
    SL_ID INTEGER,
    C_ID DECIMAL(18,0))
RETURNS (
    RES_ID INTEGER)
AS
declare variable c_un_id decimal(18,0);
begin
  select un_id
  from username
  where un_from_c_id = :c_id rows 1
  into :c_un_id;
  if (c_un_id is not null) then
  begin
    if (not(exists(select *
                   from sqllistusers
                   where slu_from_sl_id = :sl_id and
                         slu_from_un_id = :c_un_id))) then
    begin
      if (exists(select *
                 from hp_un_is_active(:c_un_id)
                 where un_is_active = 1)) then
      begin
        insert into sqllistusers (slu_from_sl_id, slu_from_un_id, slu_execute_count, slu_shared_un_id)
        values (:sl_id, :c_un_id, 0, :un_id);
        res_id = 1; -- Доступ передан
      end
    end
    else
      res_id = 2; -- Доступ уже есть
  end
  else
    res_id = 3; -- У сотрудника нет учетной записи
  suspend;
end
