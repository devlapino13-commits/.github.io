CREATE PROCEDURE HP_KONTMAT_VIEW(
    K_ID D_ID)
RETURNS (
    KML_ID D_INTEGER,
    KML_OBJECT_KIND D_INTEGER,
    KML_OBJECT_ID ID_TYPE,
    KML_FROM_ML_ID ID_TYPE,
    KML_COUNT D_CURRENCY,
    KML_COST D_CURRENCY,
    KML_TOTAL D_CURRENCY,
    KML_GUARANTEE_MONTH D_CURRENCY,
    KML_FROM_AG_ID D_INTEGER,
    KML_BILL_ID ID_TYPE,
    KML_BILL_LINE_NO D_INTEGER,
    KML_IS_RELATIVE D_INTEGER,
    KML_IS_USED_STR D_STR_10,
    AG_CAPTION D_STR_255,
    ML_NAME D_STR_255,
    ML_MODEL D_STR_255,
    KML_IS_OWN D_BOOL,
    KMI_IMEI D_ID)
AS
declare variable c_id d_id;
begin
  for
    select
      kml_id,
      kml_object_kind,
      kml_object_id,
      kml_from_ml_id,
      kml_count,
      kml_cost,
      kml_total,
      kml_guarantee_month,
      kml_from_ag_id,
      kml_bill_id,
      kml_bill_line_no,
      kml_is_relative,
      iif(kml_is_used = 1,'б/у', 'новый'),
      ag_caption,
      ml_name,
      ml_model,
      kmo_c_id
    from kontmat
      inner join matlist
        on ml_id = kml_from_ml_id
      left join agreements
        on ag_id = kml_from_ag_id
      left join kontmat_own
        on kmo_kml_id = kml_id
    where kml_object_kind = 9
      and kml_object_id = :k_id
  into
    :kml_id,
    :kml_object_kind,
    :kml_object_id,
    :kml_from_ml_id,
    :kml_count,
    :kml_cost,
    :kml_total,
    :kml_guarantee_month,
    :kml_from_ag_id,
    :kml_bill_id,
    :kml_bill_line_no,
    :kml_is_relative,
    :kml_is_used_str,
    :ag_caption,
    :ml_name,
    :ml_model,
    :c_id
  do
  begin
    kml_is_own = iif(c_id > 0, 1, 0);
    kmi_imei = 0;
    if (c_id > 0) then
      ag_caption = (select c_name_fio from clients where c_id = :c_id);

    select kmi.kmi_imei from kontmat_imei kmi
    where kmi.kmi_kml_id = :kml_id
    into  kmi_imei;
    suspend;
  end
end
