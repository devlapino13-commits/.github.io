CREATE PROCEDURE UXP$UXACTIONPARAM_UFP(
    UA_UUID D_UUID,
    UA_UUID_CODE D_STR_CODE = null)
RETURNS (
    ID D_STR_CODE,
    TEXT D_STR_255,
    SOURCE_FIELD D_STR_100)
AS
declare variable fr_uuid d_uuid;
declare variable ua_owner d_uuid;
declare variable uf_is_param d_integer;
declare variable ufd_source d_str_code;
declare variable uf_name d_str_100;
declare variable ufr_is_mainframe d_bool;
declare variable ux_kind d_str_code;
begin
  if (coalesce(ua_uuid, '') = '') then
    ua_uuid = uxf$uniq(ua_uuid_code);

  id = null;
  text = '';
  suspend;

  if (coalesce(ua_uuid, '') = '') then
    exit;

  select ua_owner
  from ux$action
  where ua_uuid = :ua_uuid
  into ua_owner;

  if (ua_owner is null) then
    ua_owner = ua_uuid;

  ux_kind = uxf$code(ua_owner, 'kind');

  if (ux_kind = 'uxframe') then
  begin
    select ufr_owner
    from ux$frame
    where ufr_uuid = :ua_owner
    into ua_owner;
  end

  for select ufr_uuid, us_code, ufr_is_mainframe
  from ux$frame, ux$source
  where ufr_owner = :ua_owner and
        ufr_us_uuid = us_uuid
  into fr_uuid, ufd_source, ufr_is_mainframe
  do
  begin
    for select uuid_to_char(ufp_uuid), uf_name, uf_field_name, uf_is_param, us_code || '$' || uf_field_name
    from ux$frparam, ux$field, ux$source
    where ufp_ufr_uuid = :fr_uuid and
          ufp_uf_uuid = uf_uuid and
          uf_owner = us_uuid and
          ufp_parent is null
    order by uf_is_param desc, uf_position
    into id, uf_name, text, uf_is_param, source_field
    do
    begin
      if (uf_is_param = 1) then
        text = text || '  /P';
      else
        text = text || '  /F';

      if (text not containing uf_name) then
        text = text || '   - ' || uf_name;

      if (ufr_is_mainframe = 0) then
        text = text || '   (' || ufd_source || ')';

      suspend;
    end
  end
end
