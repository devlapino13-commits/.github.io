CREATE PROCEDURE PRINT_KONT_DOPINFO(
    PRINT_K_ID DECIMAL(18,0))
RETURNS (
    KDI_USE_DAYS INTEGER,
    KDI_PAYED_OS DECIMAL(18,4),
    KDI_PAYED_PERC DECIMAL(18,4),
    KDI_PAYED_FINE DECIMAL(18,4),
    KDI_PAYED_TOTAL DECIMAL(18,4),
    KDI_SH_OS DECIMAL(18,4),
    KDI_SH_PERC DECIMAL(18,4),
    KDI_STOP_PERC DATE,
    KDI_ZAD_OS_NOW DECIMAL(18,4),
    KDI_CALC_PERC_NOW DECIMAL(18,4),
    KDI_CALC_TAX_NOW DECIMAL(18,4),
    KDI_SUM_TOTAL_NOW DECIMAL(18,4),
    KDI_PERC_RES_CALCED DECIMAL(18,4),
    KDI_TOTAL_RES_CALCED DECIMAL(18,4),
    KDI_ZAD_PERC_NOW DECIMAL(18,4),
    KDI_CALC_FINE_NOW DECIMAL(18,4),
    KDI_SUM_TOTAL_NOW_F DECIMAL(18,4),
    KDI_OKF_ZAD_OS DECIMAL(18,4),
    KDI_OKF_ZAD_PERC DECIMAL(18,4),
    KDI_OKF_FINE DECIMAL(18,4),
    KDI_STOP_KONTRACT D_STR_20)
AS
declare variable k_disb_date date;
declare variable kdi_writeoff_perc decimal(18,4) = 0;
begin
  /* 08.07.2013 Гусев Игорь
  DKI_USE_DAYS - дата пользования кредитом, рассчитывается как разница между датой приостановления начисления % и датой выдачи кредита (источник информации А. Жунусов)*/
  select datediff(day, kont.k_disb_date, kont.k_stop_perc), kont.k_stop_perc, k_disb_date
  from kontract kont
  where kont.k_id = :print_k_id
  into :kdi_use_days, :kdi_stop_perc, k_disb_date;

  select -sum(ok_os), -sum(iif(ok_type = 4, ok_perc, 0)), -sum(ok_fine), -sum(iif(ok_type = 5, ok_perc, 0))
  from operkred ok
  where ok_type in (4, 5) and
        ok_ol_is_active = 1 and
        ok_from_k_id = :print_k_id and
        ok_from_c_id = 0
  into :kdi_payed_os, :kdi_payed_perc, :kdi_payed_fine, kdi_writeoff_perc;
  if (kdi_writeoff_perc is null) then
    kdi_writeoff_perc = 0;

  kdi_payed_total = coalesce(kdi_payed_os, 0) + coalesce(kdi_payed_perc, 0) + coalesce(kdi_payed_fine, 0);

  select sum(ok_sh_os), sum(ok_sh_perc)
  from operkred
  where ok_type = 1 and
        ok_ol_is_active = 1 and
        ok_from_k_id = :print_k_id and
        ok_from_c_id = 0
  into :kdi_sh_os, :kdi_sh_perc;


  select f_datetostr(date_stop_kontract)
  from hp_find_stop_kontract_date(:print_k_id)
  into:kdi_stop_kontract;


  select sum(zad_os), sum(zad_perc), sum(sh_calc_perc), sum(sh_calc_fine)
  from aarep_make_zad_list(:print_k_id, current_date, 1)
  where sh_date is not null
  into :kdi_zad_os_now, :kdi_zad_perc_now, :kdi_calc_perc_now, :kdi_calc_fine_now;

  kdi_zad_os_now = coalesce(kdi_zad_os_now, 0);
  kdi_zad_perc_now = coalesce(kdi_zad_perc_now, 0);
  kdi_calc_perc_now = coalesce(kdi_calc_perc_now, 0);
  kdi_calc_fine_now = coalesce(kdi_calc_fine_now, 0);

  select sum(okf.oko_os + okf.oko_sh_os), sum(okf.oko_perc + okf.oko_sh_perc), sum(okf.oko_other + okf.oko_sh_other)
  from operkroff okf
  inner join operlist ol on ol.ol_id = okf.oko_from_ol_id
  where okf.oko_from_k_id = :print_k_id and
        ol.is_delete = 0
  into :kdi_okf_zad_os, kdi_okf_zad_perc, kdi_okf_fine;

  kdi_okf_zad_os = coalesce(kdi_okf_zad_os, 0);
  kdi_okf_zad_perc = coalesce(kdi_okf_zad_perc, 0);
  kdi_okf_fine = coalesce(kdi_okf_fine, 0);

  kdi_zad_os_now = kdi_zad_os_now + kdi_okf_zad_os;
  kdi_calc_perc_now = kdi_calc_perc_now + kdi_okf_zad_perc;
  kdi_calc_tax_now = kdi_calc_perc_now * 0.02; -- RFCKG-5624 -- (Убрал по запросу от юристов(JIRA (RFCKG-439) Внести изменения в шаблон "Уведомление о невыполнении обязательств (Мурабаха))
  kdi_calc_fine_now = kdi_calc_fine_now + kdi_okf_fine;

  if ((abs(kdi_calc_perc_now - kdi_zad_perc_now)) <= 1) then
    kdi_calc_perc_now = kdi_zad_perc_now;

  kdi_sum_total_now = kdi_zad_os_now + kdi_calc_perc_now;
  kdi_perc_res_calced = kdi_calc_perc_now + kdi_payed_perc + kdi_writeoff_perc;
  kdi_total_res_calced = kdi_sh_os + kdi_perc_res_calced;
  kdi_sum_total_now_f = kdi_sum_total_now + kdi_calc_fine_now + kdi_calc_tax_now;

  suspend;
end
