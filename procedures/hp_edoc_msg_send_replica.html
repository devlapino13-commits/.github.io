CREATE PROCEDURE HP_EDOC_MSG_SEND_REPLICA(
    EDL_ID D_ID)
AS
declare variable email_text d_str_8100;
declare variable c_id d_id;
declare variable mail_tmp d_str_8100;
declare variable doc_name d_str_255;
declare variable edl_object_kind d_integer;
declare variable edl_object_id d_id;
declare variable mail_to d_str_long;
declare variable msg_type d_integer;
declare variable edc_state d_integer;
declare variable edc_name d_str_255;
declare variable eds_id d_id;
declare variable esc_message d_str_2048;
declare variable msg_subject d_str_long;
declare variable un_id d_id;
declare variable cont_info d_str_100;
declare variable c_name d_str_100;
declare variable edl_blank_date d_date;
declare variable msg_id d_id;
begin
  -- Процедура используется для контроля заполнения документов и комплектакцию
  -- Отправляет программистам дубликаты СМС клиентов через email

  if (f_tmp_const('edoc_send_replica_active') <> 1) then
    exit;

  select edc_state, edc_name, edl_object_kind, edl_object_id, edl_create_un_id, edl_blank_date
  from edoc_list, edoc_cfg
  where edl_id = :edl_id and
        edl_from_edc_id = edc_id
  into edc_state, edc_name, edl_object_kind, edl_object_id, un_id, edl_blank_date;

  -- Если не боевой режим и тестирование с регионами, завершаем работу
  if (edc_state not in (2, 3)) then
    exit;

  msg_type = 4;
  mail_to = 'finstart@mbfplanet.com';

  -- получаем офис и номер документа
  select doc_name
  from hp_edoc_object_info(:edl_object_kind, :edl_object_id, :un_id, :edl_blank_date)
  into :doc_name;

  email_text = '';
  mail_tmp = '';

  if (edc_state = 3) then -- боевой режим
  begin
    msg_subject = 'Электронный документ - Боевой режим - НЕ ПОДПИСЫВАТЬ';

    -- собираем текст СМС по всем подписантам
    for select eds_id, eds_from_c_id
    from edoc_sign
    where eds_from_edl_id = :edl_id
    into eds_id, c_id
    do
    begin
      c_name = '<Без ФИО>';
      if (c_id is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :c_id
        into c_name;
      end else c_name = 'Контролер (подписан)';
  
      -- Получаем номер и текст сообщения отправленные текущему подписанту EDS_ID
      for select distinct msg_mail_to, msg_body
      from edoc_msg, fsrobot_msgbox
      where edm_from_eds_id = :eds_id and
            edm_from_msg_id = msg_id
      into :cont_info, esc_message
      do
      begin
        if (mail_tmp <> '') then
          mail_tmp = mail_tmp || '<br><br>';
    
        mail_tmp = mail_tmp ||
                  'Получатель: <b>' || c_name || '</b><br>' ||
                  'Телефон: <b>' || cont_info || '</b><br>' ||
                  'Сообщение: <br>' || esc_message || '<br>';
      end
    end

    if (mail_tmp <> '') then
    begin
      email_text = 'Электронный документ: ' || edc_name;
    
      if (doc_name <> '') then
        email_text = email_text || ', № ' || doc_name;
    
      email_text = email_text || '<br>' ||
                                'Проверяемый проект: ' || edc_name ||'<br>' ||
                                'Id электронного документа: ' || edl_id ||'<br>' ||
                                'Клиентам отправлен СМС, НЕ ПОДПИСЫВАТЬ по их ссылкам!<br><br>';
    
      email_text = email_text || mail_tmp || '<br>';
    end
  end else
  begin
    msg_subject = 'Электронный документ - дубликат - НЕ ПОДПИСЫВАТЬ';
    for select distinct edm_from_msg_id
    from edoc_sign, edoc_msg
    where eds_from_edl_id = :edl_id and
          eds_id = edm_from_eds_id
    into :msg_id
    do
    begin
      insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days,
                                  msg_from_un_id, msg_command_kind, msg_object_kind, msg_object_id)
      select :msg_type, :mail_to, :msg_subject, msg_body, 30, msg_from_un_id,
             msg_command_kind, msg_object_kind, msg_object_id
      from fsrobot_msgbox
      where msg_id = :msg_id;
    end
  end
end
