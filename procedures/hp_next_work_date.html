CREATE PROCEDURE HP_NEXT_WORK_DATE(
    DATE_BEGIN D_DATE,
    PLUS_DAYS D_INTEGER,
    WEEK_WORKDAYS D_INTEGER = 0)
RETURNS (
    DATE_END D_DATE,
    NEXT_WORK_DATE D_DATE)
AS
declare variable counter d_integer;
declare variable scd_work_hours d_smallint;
declare variable weekday_id d_smallint;
begin
  counter = 1;
  date_end = date_begin;

  if (coalesce(week_workdays, 0) = 0) then
    week_workdays = 5;

  if (coalesce(plus_days, 0) = 0) then
    plus_days = 1;

  if (week_workdays = -1) then
    date_end = dateadd(day, plus_days - 1, date_begin);
  else
  begin
    while (:counter < :plus_days) do
    begin
      date_end = date_end + 1;
      scd_work_hours = null;
  
      select scd_work_hours
      from sa_calendays
      where scd_date = :date_end
      into scd_work_hours;
  
      weekday_id = extract(weekday from date_end);
      if (scd_work_hours = 0 and weekday_id not in (0, 6)) then
        continue;

      if (scd_work_hours = 0 and weekday_id in (0, 6) and week_workdays = 7) then
      begin
        if (exists(select *
                   from sa_calendar
                   where scal_month = extract(month from :date_end) and
                         scal_day = extract(day from :date_end) and
                             scal_active = 1)) then
          continue;
      end
  
      if (scd_work_hours is null) then
      begin
        if (week_workdays <= 5) then
        begin
          if (weekday_id in (0, 6)) then
            scd_work_hours = 0;
        end else
        if (week_workdays = 6) then
        begin
          if (weekday_id = 0) then
            scd_work_hours = 0;
        end
      end
  
      if (scd_work_hours = 0) then
      begin
        if (week_workdays <= 5) then
          continue;
  
        if (week_workdays = 6 and weekday_id = 0) then
          continue;
      end
  
      counter = counter + 1;
    end
  end

  counter = 0;
  plus_days = 1;
  week_workdays = 5;
  next_work_date = date_end;
  while (:counter < plus_days) do
  begin
    next_work_date = next_work_date + 1;
    scd_work_hours = null;

    select scd_work_hours
    from sa_calendays
    where scd_date = :next_work_date
    into scd_work_hours;

    if (scd_work_hours is null) then
    begin
      if (week_workdays <= 5) then
      begin
        if (extract(weekday from next_work_date) in (0, 6)) then
          scd_work_hours = 0;
      end else
      if (week_workdays = 6) then
      begin
        if (extract(weekday from next_work_date) = 0) then
          scd_work_hours = 0;
      end
    end

    if (scd_work_hours = 0) then
      continue;

    counter = counter + 1;
  end

  suspend;
end
