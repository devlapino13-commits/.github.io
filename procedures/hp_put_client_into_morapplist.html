CREATE PROCEDURE HP_PUT_CLIENT_INTO_MORAPPLIST(
    C_ID DECIMAL(18,0),
    REQ_COMMENT VARCHAR(512),
    REG_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    MAL_ID INTEGER,
    MAL_REQ_DATETIME TIMESTAMP)
AS
declare variable m_id decimal(18,0);
declare variable m_number integer;
declare variable c_name_fio d_str_100;
declare variable tmp_c_name_fio d_str_100;
declare variable c_birth_date d_date;
begin
  mal_id = 0;

  select mal_req_datetime
  from morapprlist, moratory
  where mal_appr_status = 0 and
        mal_req_state = 0 and
        mal_from_m_id = m_id and
        mal_from_m_number = m_number and
        m_from_c_id = :c_id
  order by mal_req_datetime desc
  rows 1
  into mal_req_datetime;
  if (mal_req_datetime is not null) then
  begin
    mal_id = -1;
    suspend;
    exit;
  end

  c_name_fio = '';

  select c_name_fio, c_birth_date
  from clients
  where c_id = :c_id
  into c_name_fio, c_birth_date;

  for select m_id, m_number
      from moratory
      where m_from_c_id = :c_id and
            m_is_active = 1 and
            m_is_black in (1, 2) -- отказники + Стоп-лист
      union all
      -- Чтобы поймать импортированных из других стран
      select m_id, m_number
      from moratory
      where cast(m_from_c_id as id_type) is null -- чтобы отключить индекс
            and
            m_is_active = 1 and
            m_is_black in (1, 2) -- Отказники + Стоп-лист
            and
            m_client_fio = :c_name_fio
      into m_id, m_number
  do
  begin
    if (m_id is not null) then
    begin
      insert into morapprlist (mal_id, mal_from_m_id, mal_from_m_number, mal_req_state, mal_req_user, mal_req_comment,
                               mal_from_reg_id)
      values (-1, :m_id, :m_number, 0, :un_id, :req_comment, :reg_id)
      returning mal_id
      into mal_id;
    end
  end

  if (m_id is null) then
  begin
    -- Нечеткий поиск. На случай, если совпадение с черным списком по ФИО было частичным

    select full_text
    from hp_fio_clear(:c_name_fio)
    into tmp_c_name_fio;

    for select m_id, m_number
        from moratory
        where cast(m_from_c_id as id_type) is null -- чтобы отключить индекс
              and
              m_is_active = 1 and
              m_is_black in (1, 2) and -- Отказники + Стоп-лист
              m_client_birth_date = :c_birth_date and
              f_levenstain_is_same(m_client_fio, :tmp_c_name_fio) = 1 and
              coalesce(m_client_fio, '') <> ''
        into m_id, m_number
    do
    begin
      if (m_id is not null) then
      begin
        insert into morapprlist (mal_id, mal_from_m_id, mal_from_m_number, mal_req_state, mal_req_user, mal_req_comment,
                                 mal_from_reg_id)
        values (-1, :m_id, :m_number, 0, :un_id, :req_comment, :reg_id)
        returning mal_id
        into :mal_id;
      end
    end
  end

  suspend;
end
