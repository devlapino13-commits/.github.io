CREATE PROCEDURE REP_DEALING_PROBLEM_CUSTOMERS(
    REP_REGION_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    K_KONTRACT_NO VARCHAR(50),
    KF_NAME VARCHAR(100),
    DELINQ_SH_OS CURR_TYPE,
    DELINQ_SH_OS_CL CURR_TYPE,
    DELINQ_DAYS INTEGER,
    TEL_NO VARCHAR(20),
    OFFICE_NAME VARCHAR(50))
AS
declare variable k_id id_type;
declare variable k_amount_approved curr_type;
declare variable kc_amount_approved curr_type;
begin
  for select oks_from_k_id, cast(:rep_date_end as date) - oks_delinq_date as deliq_days
      from oksummary oks, region_list(:rep_region_id, 1, -1, 1)
      where oks_balance > 0 and
            :rep_date_end between oks_date_begin and oks_date_end and
            cast(:rep_date_end as date) - oks_delinq_date > 7 and
            (oks_region_id = id_list)
            and oks_close_state = 0
      order by 2 desc
      into k_id, delinq_days
  do
  begin
    select k_kontract_no, k.k_amount_approved, kal_new_kf_name, reg_name
    from kontract k, knaktlist kal, region reg
    where k_id = :k_id and
          k_from_kf_id = kal_from_kf_id and
          kal.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal.kal_new_reg_id = reg_unicode
    into k_kontract_no, k_amount_approved, kf_name, office_name;

    select sum(ok_os + ok_sh_os) sh_os
    from operkred ok
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_from_c_id = 0 and
          ok_type < 7 and
          ok_ol_date <= :rep_date_end and
          ok_sh_date < :rep_date_end
    into delinq_sh_os;

    if (delinq_sh_os > 0) then
    begin
      for select cl.c_name_fio,
                 (select cont_value
                  from hp_contdata_get_cell_main(0, c_id)), kc_amount_approved
          from clients cl, knclient
          where c_id = kc_from_c_id and
                kc_amount_approved > 0 and
                kc_from_k_id = :k_id
          into c_name_fio, tel_no, kc_amount_approved
      do
      begin
        delinq_sh_os_cl = round(kc_amount_approved / k_amount_approved * delinq_sh_os);

        if (tel_no is null) then
          tel_no = '';
        suspend;
      end
    end
--    suspend;
  end
end
