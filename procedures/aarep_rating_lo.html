CREATE PROCEDURE AAREP_RATING_LO(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_REGION_ID INTEGER,
    AA_KC_ID DECIMAL(18,0),
    AA_TOTAL_BALANCE DECIMAL(18,4),
    AA_TOTAL_CLIENT INTEGER,
    AA_DELINQ_BALANCE DECIMAL(18,4),
    AA_DELINQ_CLIENT INTEGER,
    AA_DISB_CL_KOLICH INTEGER,
    AA_REG_DIR_FIO VARCHAR(100),
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_LEVEL2_NAME VARCHAR(100),
    AA_CURR_LEVEL3_NAME VARCHAR(100),
    AA_LO_FIO VARCHAR(100),
    AA_LO_JOBNAME VARCHAR(100),
    AA_LO_STATUS VARCHAR(100),
    AA_IS_LO VARCHAR(20),
    AA_CAN_SHOW INTEGER)
AS
declare variable reg_prior integer;
declare variable lo_prior decimal(18,0);
declare variable ps_is_working integer;
declare variable ps_is_on_staff integer;
declare variable jl_kind varchar(100);
declare variable reg_level integer;
declare variable orig_level_name varchar(100);
declare variable rep_access_kind integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable go_office integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select cm_rating_show_kind
  from company
  into :rep_access_kind;
  if (rep_access_kind is null) then
    rep_access_kind = 1;
-- вычисляет ID Головного офиса
   go_office = f_hq_id();


  reg_prior = 0;
  lo_prior = 0;
  aa_lo_status = '';
  aa_lo_jobname = '';
  aa_can_show = 0;
  if (rep_access_kind = 1) then
  begin
    select reg_report_level
    from region
    where reg_unicode = :rep_region_id and
          reg_unicode <> :go_office -- Добавленная строка чтоб исключить головного офиса
    into :reg_level;
    if (reg_level <= 2) then
    begin
      select r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :rep_region_id and
            r1.parent = r2.reg_unicode
      into :orig_level_name;
    end
    else
      orig_level_name = '';
  end
  for select aa_curr_region_id, aa_kc_id, sum(aa_total_balance), sum(aa_total_client), sum(aa_delinq_balance),
             sum(aa_delinq_client), sum(aa_disb_cl_kolich)
      from aarep_rating_lo_tmp_own(1000000, :rep_date_begin, :rep_date_end)
      where aa_curr_region_id <> :go_office -- Добавленная строка чтоб исключить головного офиса
      group by aa_curr_region_id, aa_kc_id
      into :aa_curr_region_id, :aa_kc_id, :aa_total_balance, :aa_total_client, :aa_delinq_balance, :aa_delinq_client,
           :aa_disb_cl_kolich
  do
  begin
    if (aa_total_balance is null) then
      aa_total_balance = 0;
    if (aa_total_client is null) then
      aa_total_client = 0;
    if (aa_delinq_balance is null) then
      aa_delinq_balance = 0;
    if (aa_delinq_client is null) then
      aa_delinq_client = 0;
    if (aa_disb_cl_kolich is null) then
      aa_disb_cl_kolich = 0;
    if (reg_prior <> aa_curr_region_id) then
    begin
      select rc_value_01, aa_curr_office_name, aa_curr_level2_name, aa_curr_level3_name
      from aarep_get_office_categ(:aa_curr_region_id, :rep_date_end)
      where rc_reg_id <> :go_office -- Добавленная строка чтоб исключить головного офиса

      into :aa_reg_dir_fio, :aa_curr_office_name, :aa_curr_level2_name, :aa_curr_level3_name;
      reg_prior = aa_curr_region_id;
    end
    if (lo_prior <> aa_kc_id) then
    begin
      select c_name_fio
      from clients
      where c_id = :aa_kc_id
      into :aa_lo_fio;
      lo_prior = :aa_kc_id;

      aa_lo_jobname = 'Нет приказа';
      ps_is_working = 0;
      ps_is_on_staff = 0;
      jl_kind = 0;

      select jl_name, ps_is_working, ps_is_on_staff, jk_system_code
      from hr_pstatus, joblist, jobkind
      where ps_from_c_id = :aa_kc_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id
      order by ps_date_begin desc, ps_is_working desc
      rows 1
      into :aa_lo_jobname, :ps_is_working, :ps_is_on_staff, :jl_kind;

      if (ps_is_working = 1) then
      begin
        if (ps_is_on_staff = 0) then
          aa_lo_status = 'Работает, внештатный';
        else
          aa_lo_status = 'Работает';
      end
      else
      begin
        aa_lo_status = 'Неработает';
      end
      if (exists(select *
                 from hr_voctravel, hr_orders, sa_worktype
                 where hrv_from_ho_id = ho_id and
                       ho_order_type = sawt_id and
                       hrv_from_c_id = :aa_kc_id and
                       hrv_date_begin <= :rep_date_end and
                       hrv_date_end_fact >= :rep_date_end and
                       hrv_status = 1 and
                       sawt_base_type = 5 and
                       sawt_include_holiday = 1 and
                       sawt_is_paid = 0 /* дикретный отпуск */
                       rows 1)) then
        aa_lo_status = 'Декретный отпуск';

      if (jl_kind in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
        aa_is_lo = 'Специалист';
      else
        aa_is_lo = 'Другое';
    end
    if ((aa_disb_cl_kolich <> 0) or (aa_total_balance <> 0)) then
    begin
      if (rep_access_kind = 0) then
        aa_can_show = 1;
      else
      if (rep_access_kind = 1) then
      begin
        if (reg_level = 0) then
        begin
          if (orig_level_name = aa_curr_level2_name) then
            aa_can_show = 1;
          else
            aa_can_show = 0;
        end
        else
        begin
          if (reg_level in (1, 2)) then
          begin
            if (orig_level_name = aa_curr_level3_name) then
              aa_can_show = 1;
            else
              aa_can_show = 0;
          end
          else
            aa_can_show = 1;
        end
      end
      else
      begin
        if (exists(select *
                   from region_list(:rep_region_id, 1, -1, 1)
                   where id_list = :aa_curr_region_id)) then
          aa_can_show = 1;
        else
          aa_can_show = 0;
      end
      suspend;
    end
  end
end
