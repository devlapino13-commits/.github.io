CREATE PROCEDURE HP_KONT_VAC_INFO(
    K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    OV_OS D_CURRENCY,
    OV_PERC D_CURRENCY,
    LAST_OS_SH_ID D_INTEGER,
    VAC_DATE_BEGIN D_DATE,
    VAC_DATE_END D_DATE,
    VAC_MAX_PART_LIMIT D_CURRENCY)
AS
declare variable vac_begin_days d_integer;
declare variable vac_days d_integer;
declare variable ol_date d_date;
begin
  if (rep_date is null) then
    rep_date = current_date;

  select min(ol_date), sum(ov_os), sum(ov_perc), sum(ov_last_os_sh_id),
         sum(ov_vac_begin_days), sum(ov_vac_days), sum(ov_max_part_limit)
  from opervac, operlist ol
  where ov_from_k_id = :k_id and
        ov_from_ol_id = ol_id and
        ol.is_delete = 0 and
        ol.ol_date <= :rep_date
  into ol_date, ov_os, ov_perc, last_os_sh_id, vac_begin_days, vac_days, vac_max_part_limit;

  if (ov_os is null) then
    ov_os = 0;

  if (last_os_sh_id = 0) then
    last_os_sh_id = null;

  if (vac_begin_days is null) then
    vac_begin_days = 0;

  if (vac_days is null) then
    vac_days = 0;

  if (vac_max_part_limit is null) then
    vac_max_part_limit = 0;

  if (ol_date is not null) then
  begin
    vac_date_begin = dateadd(day, vac_begin_days, ol_date);
    vac_date_end = dateadd(day, vac_days, vac_date_begin);
    suspend;
  end
end
