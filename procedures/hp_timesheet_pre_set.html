CREATE PROCEDURE HP_TIMESHEET_PRE_SET(
    REG_ID D_INTEGER,
    INP_YEAR D_INTEGER,
    INP_MONTH D_INTEGER)
RETURNS (
    WORKING_DAYS_IN_MONTH D_INTEGER,
    LAST_DAY D_INTEGER,
    WEEKEND_DAYS D_STR_100,
    IS_NEW D_SMALLINT,
    MONTH_IS_CLOSED D_SMALLINT,
    NORMA_WORKING_DAYS D_INTEGER,
    PRE_WEEKEND_DAY D_STR_100,
    REMOVE_WEEKEND D_INTEGER)
AS
declare variable date_begin d_date;
declare variable date_end d_date;
declare variable date_tmp d_date;
declare variable weekday_tmp d_integer;
declare variable year_month d_str_10;
declare variable holidays d_str_100;
begin
  date_begin = '01.' || :inp_month || '.' || :inp_year;
  date_end = f_date_month_end(:date_begin);
  last_day = extract(day from :date_end);
  year_month = :inp_year || iif(:inp_month < 10, '0' || :inp_month, :inp_month);
  weekend_days = '';
  pre_weekend_day = '';
  working_days_in_month = 7;
  remove_weekend = 1;

  select sowd.sowd_days
  from sa_office_working_days sowd
  where sowd.sowd_from_reg_id = :reg_id
    and :date_end between sowd.sowd_date_begin and sowd.sowd_date_end
  into :working_days_in_month;

  select
    list(scal_day, '.') || '.'
  from sa_calendar
  where scal_month = :inp_month
    and scal_active = 1
  into
    :holidays;

  weekday_tmp = -1;

  select list(extract(day from scd_date), '.') || '.'
  from (select scd_date
        from sa_calendays
        where scd_year_month = :year_month and
              scd_work_hours = 7
        order by scd_date ascending)
  into :pre_weekend_day;

  if (working_days_in_month = 5) then
  begin
    select list(extract(day from scd_date), '.') || '.'
    from (select scd_date
          from sa_calendays
          where scd_year_month = :year_month and
                scd_work_hours = 0
          order by scd_date ascending)
    into :weekend_days;
  end

  date_tmp = :date_begin;

  if (working_days_in_month = 6 or (working_days_in_month = 5 and
      weekend_days = '')) then
  begin
    while (:date_tmp <= :date_end) do
    begin
      weekday_tmp = extract(weekday from cast(:date_tmp as date));
      if (working_days_in_month = 5 and
          weekday_tmp in (6, 0)) then
        weekend_days = weekend_days || extract(day from :date_tmp) || '.';
      if (working_days_in_month = 6 and
          weekday_tmp = 0) then
        weekend_days = weekend_days || extract(day from :date_tmp) || '.';
      date_tmp = :date_tmp + 1;
    end
    weekend_days = coalesce(weekend_days, '') || coalesce(holidays, '');
  end

  if (working_days_in_month = 7) then
    weekend_days = holidays;

  if (weekend_days is null) then
    weekend_days = '';
  if (pre_weekend_day is null) then
    pre_weekend_day = '';

  if (exists(
    select *
    from sa_tabelo
    where sat_year_month = :year_month
      and sat_office_id = :reg_id)) then
    is_new = 0;
  else
    is_new = 1;

  month_is_closed = 0;
  select is_closed
  from hp_sa_tabel_month_status(:date_begin)
  into :month_is_closed;

  norma_working_days = 0;
  select count(*)
  from sa_calendays
  where scd_year_month = :year_month and
        scd_work_hours > 0
  into :norma_working_days;

  suspend;
end
