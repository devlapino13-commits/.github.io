CREATE PROCEDURE FM_ZALOGAUTO_EDIT(
    ZA_ID D_INTEGER,
    ZA_OBJECT_KIND D_STR_CODE,
    ZA_OBJECT_ID D_ID,
    ZA_FROM_C_ID D_ID,
    ZAL_COST D_CURRENCY,
    ZA_ORIG_STS_ID D_INTEGER,
    ZA_ORIG_PTS_ID D_INTEGER,
    ZAL_WEBPAGE D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
begin
  zal_webpage = coalesce(zal_webpage, '');
  select res_is_error, res_text
  from hp_zalogauto_edit(:za_id, :za_object_kind, :za_object_id,
    :za_from_c_id, :zal_cost, :za_orig_sts_id, :za_orig_pts_id, null, null,
    :zal_webpage, :un_id)
  into :res_is_error, :res_text;
  suspend;
end
