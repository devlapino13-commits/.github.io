CREATE PROCEDURE AAREP_DELINQ_KONTRACTS_FOR_DR(
    REP_REGION_ID ID_TYPE,
    REP_DATE DATE,
    REP_STEP SMALLINT = 2)
RETURNS (
    REG_ID ID_TYPE,
    OKS_KC_ID ID_TYPE,
    R_DATE DATE,
    OFFICE_NAME VARCHAR(50),
    REG_NAME VARCHAR(50),
    KC_OFFICER_NAME VARCHAR(100),
    LO_STAGE_MONTH INTEGER,
    OKS_BALANCE CURR_TYPE,
    KOL_CL INTEGER,
    BALANCE_PAR30 CURR_TYPE,
    PAR30 CURR_TYPE,
    KOL_CL_PAR30 INTEGER,
    NT_BALANCE CURR_TYPE,
    NT_KOL_CL INTEGER,
    NT_BALANCE_PAR30 CURR_TYPE,
    NT_PAR30 CURR_TYPE,
    NT_KOL_CL_PAR30 INTEGER,
    PR_NAME STR_255,
    CELL_NO STR_100)
AS
declare variable oks_kc_id_tmp id_type;
declare variable reg_id_tmp id_type;
begin
  if (coalesce(rep_step, 0) <> 1) then
    rep_step = 2;

  if (rep_step = 1) then
  begin
    if (extract(day from rep_date) = 1) then
      rep_date = rep_date - 1; --Если отченый день 1-й день месяца, отчет готовится за пред.месяц.
    for with table_date
        as (select :rep_date as rdate
            from rdb$database
            where extract(day from :rep_date + 1) = 1
            union
            select cast('25.' || right('0' || extract(month from :rep_date), 2) || '.' || extract(year from :rep_date) as date)
            from rdb$database
            union
            select cast('20.' || right('0' || extract(month from :rep_date), 2) || '.' || extract(year from :rep_date) as date)
            from rdb$database
            union
            select cast('15.' || right('0' || extract(month from :rep_date), 2) || '.' || extract(year from :rep_date) as date)
            from rdb$database
            union
            select cast('10.' || right('0' || extract(month from :rep_date), 2) || '.' || extract(year from :rep_date) as date)
            from rdb$database
            union
            select cast('05.' || right('0' || extract(month from :rep_date), 2) || '.' || extract(year from :rep_date) as date)
            from rdb$database
            --            union
            --            select cast('01.' || right('0' || extract(month from :rep_date), 2) || '.' || extract(year from :rep_date) as date)
            --            from rdb$database
        )
        select distinct rdate
        from table_date
        where rdate <= :rep_date
        into :r_date
    do
    begin
      -- для всех активных контрактов на дату, вычисляю задолженность
      for select oks_region_id, oks.oks_kc_id, pr_name, --поля для группировки

                 sum(iif(oks_kc_id = k.k_fact_disb_lo_id, oks_balance, 0)) nt_balance, --Баланс по собств. выдачам
                 sum(iif(oks_kc_id = k.k_fact_disb_lo_id, k.k_kolich_client_approved, 0)) nt_kol_cl, --Общее количество клиентов Випс
                 sum(iif(oks_kc_id = k.k_fact_disb_lo_id and :r_date - oks_delinq_date > 30, oks_balance, 0)) nt_balance_par30, --Общий Баланс PAR 30
                 sum(iif(oks_kc_id = k.k_fact_disb_lo_id and :r_date - oks_delinq_date > 30, k.k_kolich_client_approved, 0)) nt_par30_kol_cl, --Баланс PAR 30 по собств. выдачам

                 sum(oks_balance) oks_balance, --Общий баланс Випс
                 sum(k.k_kolich_client_approved), --Общее количество клиентов Випс
                 sum(iif(:r_date - oks_delinq_date > 30, oks_balance, 0)) oks_balance, --Общий Баланс PAR 30
                 sum(iif(:r_date - oks_delinq_date > 30, k.k_kolich_client_approved, 0)) --общее количество клиентов ВИпс

          from region_list(:rep_region_id, 1, -1, 1)
          inner join oksummary oks on (id_list = oks_region_id and
                oks_close_state = 0 and
                :r_date between oks_date_begin and oks_date_end)
          inner join kontract k on oks_from_k_id = k_id
          inner join product pr on (k_from_pr_id = pr.pr_id)
          where :r_date - oks_delinq_date < 361 --361 и более не включаем в отчет
          group by 1, 2, 3
          into :reg_id, oks_kc_id, pr_name, --
               nt_balance, nt_kol_cl, nt_balance_par30, nt_kol_cl_par30, --
               oks_balance, kol_cl, balance_par30, kol_cl_par30
      do
      begin
        --nt_ - not transferred не переданные (свои клиенты, портфель)
        par30 = 0; --% PAR30
        if (balance_par30 > 0) then
          par30 = balance_par30 / oks_balance * 100;
        nt_par30 = 0; --% PAR30
        if (nt_balance_par30 > 0) then
          nt_par30 = nt_balance_par30 / nt_balance * 100;

        suspend;
      end
    end
  end

  if (rep_step = 2) then
  begin
    for select reg_id, oks_kc_id, r_date, oks_balance, kol_cl,
               balance_par30, par30, kol_cl_par30, nt_balance, nt_kol_cl, nt_balance_par30, nt_par30, nt_kol_cl_par30,
               pr_name
        from aarep_delinq_kontracts_for_dr(:rep_region_id, :rep_date, 1)
        order by reg_id, oks_kc_id, pr_name, r_date desc
        into reg_id, oks_kc_id, r_date, oks_balance, kol_cl,
             balance_par30, par30, kol_cl_par30, nt_balance, nt_kol_cl, nt_balance_par30, nt_par30, nt_kol_cl_par30,
             pr_name
    do
    begin
      if (coalesce(oks_kc_id_tmp, 0) <> oks_kc_id) then
      begin
        lo_stage_month = 0;
        select lo_stage_month_h
        from hp_get_lo_stage(:oks_kc_id, :rep_date)
        into lo_stage_month;
        if (coalesce(lo_stage_month, -1) < 0) then
          lo_stage_month = 0;

        select c_name_fio
        from clients c
        where c_id = :oks_kc_id
        into kc_officer_name;

        cell_no = f_cont_info('clients', oks_kc_id, 'cell_main', 'ok');
        if (cell_no = '') then
          cell_no = null;

        oks_kc_id_tmp = oks_kc_id;
      end

      if (coalesce(reg_id_tmp, 0) <> reg_id) then
      begin
        select r1.reg_name, r2.reg_name
        from region r1
        inner join region r2 on r1.parent = r2.reg_unicode
        where r1.reg_unicode = :reg_id
        into office_name, reg_name;

        reg_id_tmp = reg_id;
      end
      suspend;
    end
  end
end
