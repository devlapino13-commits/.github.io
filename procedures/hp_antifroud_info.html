CREATE PROCEDURE HP_ANTIFROUD_INFO(
    DT_BEGIN D_DATE = null,
    DT_END D_DATE = null)
RETURNS (
    REG_NAME D_STR_CODE,
    NUM_LOAN D_INTEGER,
    NUM_AVTO D_INTEGER,
    NUM_MANUAL D_INTEGER,
    NUM_ERR1 D_INTEGER,
    NUM_ERR2 D_INTEGER,
    NUM_ERR3 D_INTEGER,
    PERC_MANUAL D_INTEGER)
AS
declare variable num d_integer;
declare variable inum_loan d_integer;
declare variable inum_avto d_integer;
declare variable inum_manual d_integer;
declare variable inum_err1 d_integer;
declare variable inum_err2 d_integer;
declare variable inum_err3 d_integer;
begin
  dt_begin = coalesce(dt_begin, current_date);
  dt_end = coalesce(dt_end, current_date);

  num = 0;
  inum_loan = 0;
  inum_avto = 0;
  inum_manual = 0;
  inum_err1 = 0;
  inum_err2 = 0;
  inum_err3 = 0;
  for
    select
      reg_name as "Офис",
      count(*) as "Всего займов",
      sum(iif(vwp_acl_id > 0, 0, 1)) as "Антифрод авто.",
      sum(iif(vwp_acl_id > 0, 1, 0)) as "Андеррайтеры",
      sum(iif((vwp_acl_id > 0) and (vwp_json_answer like '{%}') and
        (f_json_str_get_value(vwp_json_answer, 'info.liveness_result', -1) <> 0), 1, 0)) as "Ошибок антифр.",
      sum(iif((vwp_acl_id > 0) and (vwp_json_answer like '{%}') and
        (f_json_str_get_value(vwp_json_answer, 'info.liveness_result', -1) = 0), 1, 0)) as "Ошибок сверки",
      sum(iif(vwp_json_answer not like '{%}', 1, 0)) as "Ошибок обработки"
    from verify_webcam_photo
      inner join kontract
        on k_id = vwp_k_id
      inner join knaktlist
        on kal_from_kf_id = k_from_kf_id
          and cast(vwp_datetime as d_date) between kal_date and kal_date_end
          and is_delete = 0
      inner join region
        on reg_unicode = kal_new_reg_id
    where (vwp_datetime >= :dt_begin) and (vwp_datetime < :dt_end + 1)
    group by reg_name
    order by 1
  into
    :reg_name,
    :num_loan,
    :num_avto,
    :num_manual,
    :num_err1,
    :num_err2,
    :num_err3
  do
  begin
    num = num + 1;
    inum_loan = inum_loan + num_loan;
    inum_avto = inum_avto + num_avto;
    inum_manual = inum_manual + num_manual;
    inum_err1 = inum_err1 + num_err1;
    inum_err2 = inum_err2 + num_err2;
    inum_err3 = inum_err3 + num_err3;
    perc_manual = round(100 * num_manual / num_loan, 0);
    suspend;
  end
  if (num > 0) then
  begin
    reg_name = 'ИТОГ (' || f_datetostr(dt_begin) || '-' || f_datetostr(dt_end) || ')';
    num_loan = inum_loan;
    num_avto = inum_avto;
    num_manual = inum_manual;
    num_err1 = inum_err1;
    num_err2 = inum_err2;
    num_err3 = inum_err3;
    perc_manual = round(100 * inum_manual / inum_loan, 0);
    suspend;
  end
end
