CREATE PROCEDURE FM_LOANS_IN_PROCESS(
    UN_ID D_ID,
    REG_ID D_INTEGER)
RETURNS (
    K_ID D_ID,
    MA_ID D_INTEGER,
    KF_NAME D_STR_100,
    LOAN_STATUS D_STR_100,
    DISB_DATE D_DATE,
    MA_PLAN_DISB_TIME D_DATETIME,
    PR_ID D_INTEGER)
AS
declare variable c_id d_id;
declare variable kf_id d_id;
declare variable ks_state d_integer;
declare variable k_status d_integer;
declare variable res_object_status d_integer;
declare variable ma_id_tmp d_integer;
declare variable is_manager d_smallint;
declare variable rd_code d_str_100;
declare variable regions_str d_str_4096;
begin
  select un_from_c_id
  from username
  where un_id = :un_id
  into c_id;

  if (c_id is null) then
    exit;

  is_manager = (select has_access
    from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.CanManageLoans'));

  if (not exists(select ps_from_reg_id
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :c_id and
                       ps_date_begin <= current_date and
                       ps_date_end >= current_date and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       (jk_system_code = 'JobSystem.Region_KM' or :is_manager = 1) and
                       ps_is_working = 1)) then
    reg_id = -1;

  regions_str = null;

  if (reg_id <> -1) then
  begin
    regions_str = '.';

    for select id_list
        from region_list(:reg_id, 1, 3, 1)
        into :reg_id
    do
        regions_str = regions_str || :reg_id || '.' ;
  end
  else
    regions_str = '.-1.';

  for select value_out
  from str_with_separator_parser(:regions_str, '.')
  into :reg_id
  do
  begin
      for select res_c_id
          from hp_staff_changer_list(:un_id, 1)
          where :reg_id = -1
          union
          select res_c_id
          from hp_staff_changer_list(:un_id, -1)
          where :reg_id <> -1
          into :c_id
      do
      begin
        ma_id = 0;
        for select k_id, ks_state, k_status, kf_id, disb_date, ma_id_tmp, pr_id
            from fm_loans_in_process_tmp(:reg_id, :c_id)
            into :k_id, ks_state, k_status, kf_id, disb_date, ma_id_tmp, :pr_id
        do
        begin
          ma_plan_disb_time = null;
          select ma_plan_disb_time
          from miniapp
          where ma_id = :ma_id_tmp and
                ma_date_end >= '31.12.2999' and
                ma_date_app <= current_date
          into :ma_plan_disb_time;
    
          loan_status = '';
          if (ks_state = 1) then
            loan_status = 'Выдан';
          else
          begin
            if (k_status = -1) then
              loan_status = 'Отказано';
            if (k_status = 0) then
            begin
              loan_status = 'В процессе';
              res_object_status = null;
    
              select rd_code
              from hp_get_check_loan_data(:k_id)
              into:rd_code;
    
              select ovs_add_value
              from hp_objectprops_get('Kontract', :k_id, -1, :rd_code, null)
              into :res_object_status;
              if (res_object_status is not null) then
              begin
                loan_status = (case res_object_status
                                 when -2 then 'КК: Отказан'
                                 when -1 then 'КК: Доработка'
                                 when 0 then 'КК: В очереди'
                                 when 1 then 'КК: На рассмотр.'
                                 when 2 then 'КК: Подтвержден'
                               end);
              end
            end
            if (k_status = 1) then
            begin
              loan_status = 'Утвержден';
            end
          end
          kf_name = '';
          select kal_new_kf_name
          from knaktlist
          where kal_date_end >= '31.12.2999' and
                knaktlist.is_delete = 0 and
                kal_from_kf_id = :kf_id
          into :kf_name;
    
          suspend;
        end
    
        ks_state = 0;
        k_id = 0;
        loan_status = '';
        pr_id = 0;
        for select ma_id, k_status, kf_id, kf_name, disb_date, ma_plan_disb_time, pr_id
            from fm_loans_in_process_tmp2(:reg_id, :c_id)
            into :ma_id, k_status, kf_id, kf_name, :disb_date, ma_plan_disb_time, :pr_id
        do
        begin
          if (not(exists(select *
                         from kontract
                         where k_from_ma_app_id = :ma_id and
                               k_status >= -1))) then
          begin
            loan_status = '';
            if (k_status = -1) then
              loan_status = 'Отказана';
            if (k_status = 0) then
              loan_status = 'Аннулирована';
            if (k_status = 1) then
              loan_status = 'Заявка';
            suspend;
          end
        end
      end
  end
end
