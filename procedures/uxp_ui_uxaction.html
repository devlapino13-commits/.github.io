CREATE PROCEDURE UXP$UI_UXACTION(
    UA_UUID D_UUID,
    VER_KIND D_STR_CODE = null,
    UA_UUID_CODE D_STR_CODE = null)
RETURNS (
    UA_ACTION_KIND D_STR_CODE,
    UA_DISABLE_ON_EDIT D_BOOL,
    UA_READONLY D_BOOL,
    UXA_UUID D_UUID,
    UXA_CODE D_STR_CODE,
    UX_CAPTION D_STR_100,
    UX_KIND D_INTEGER,
    UX_WIDTH D_INTEGER,
    UX_HEIGHT D_INTEGER,
    UX_DIALOGEDIT_UA_UUID D_UUID,
    UX_DIALOGEDIT_FIELD D_STR_CODE,
    UX_DIALOGEDIT_IS_PARAM D_BOOL,
    UX_DIALOGEDIT_FIND_FIELD D_STR_CODE,
    UX_SOURCE_ROWS D_SMALLINT,
    UX_SOURCE_REFRESH D_BOOL,
    UX_SOURCE_LOCATE_FIELDS D_STR_255,
    UX_FORM_INVISIBLE D_BOOL,
    UX_SPLIT_PARAMS D_BOOL,
    UX_VER_KIND D_STR_CODE,
    UX_PUBLIC_PARAM_SQL D_STR_8100,
    UX_REPORT_KIND D_STR_CODE,
    UX_SPLIT_PARAM_WIDTH D_INTEGER,
    US_ASYNC_MODE D_BOOL,
    UX_REPORT_IMPORT_KIND D_SMALLINT,
    UX_REPORT_IMPORT_ID D_INTEGER,
    US_USE_REPORT_DB D_BOOL,
    US_ALLOW_SWITCH_DB D_BOOL,
    UFM_WEB_HELP D_STR_LONG,
    UX_DIALOGEDIT_MULTI_DELIMITER D_STR_CODE,
    UXA_MENU_KIND D_SMALLINT,
    UX_COMMIT_DIALOG D_BOOL,
    UFM_DONT_PREPARE_ON_SHOW D_BOOL,
    UFM_UPDATES_KIND D_SMALLINT,
    UA_AUTO_REFRESH_SECONDS D_INTEGER,
    UX_LIFETIME_SECS D_INTEGER,
    UA_HAS_OUTPUTS D_BOOL,
    UA_ORIG_READONLY D_BOOL,
    UX_CLOSE_ON_SAVE D_BOOL,
    UX_IS_PREBLANKS D_BOOL,
    UA_TRIGGER_CODE D_STR_CODE,
    UA_PREBLANK_SAVE D_STR_CODE,
    UX_IS_DOC D_BOOL,
    UX_REPOST_UA_UUID D_UUID,
    UX_DOC_FRAME_UUID D_UUID,
    UFM_EXCEL D_BLOB_BINARY)
AS
declare variable mainframe_uuid d_uuid;
declare variable dialog_uxa_uuid d_uuid;
declare variable ux_open_kind d_smallint; /* Нужно ли открывать окно перед выполнением команды */
declare variable ufp_uuid d_uuid;
declare variable ufr_code d_str_100;
declare variable uf_field_name d_str_code;
declare variable unlinked_fields d_str_255;
declare variable ufr_is_grid d_bool;
declare variable ufr_can_update d_bool;
declare variable ufr_can_insert d_bool;
declare variable ufr_can_delete d_bool;
declare variable ufr_uuid d_uuid;
declare variable pk_count d_integer;
declare variable ufr_name d_str_100;
declare variable edit_frames d_bool;
declare variable is_linked d_bool;
declare variable uxa_kind d_smallint;
declare variable ua_usr_id_field d_str_code;
declare variable uxa_owner d_uuid;
declare variable ux_code d_str_100;
declare variable ufr_us_uuid d_uuid;
declare variable mainframe_code d_str_100;
declare variable us_is_table d_bool;
declare variable ufm_uuid d_uuid;
declare variable ux_kind_code d_str_code;
declare variable frame_count d_integer;
declare variable urt_kind d_integer;
declare variable ua_usr_text_field d_str_code;
declare variable ufm_lock_change_db d_bool;
declare variable ua_owner_kind d_str_code;
declare variable need_public_params d_bool;
declare variable main_us_uuid d_uuid;
declare variable ufr_add_table_fields d_str_code;
declare variable uap_param_name d_str_code;
declare variable uap_owner_ufp_uuid d_uuid;
declare variable tmp_field d_str_code;
declare variable ua_safe_mode d_bool;
declare variable mm_uuid d_uuid;
declare variable main_ua_uuid d_uuid;
begin
  if (coalesce(ua_uuid, '') = '') then
    ua_uuid = uxf$uniq(ua_uuid_code);

  frame_count = 0;
  ux_split_params = 0;
  ux_form_invisible = 0;
  ux_source_rows = -1; -- если не команда, а Update или Delete SQL тоже поправить
  ux_source_refresh = 0; -- если не команда, а Update или Delete SQL тоже поправить
  ux_source_locate_fields = '';
  ua_auto_refresh_seconds = 0;
  ux_lifetime_secs = 0;
  ux_is_preblanks = 0;
  ua_trigger_code = '';

  select ua_uxa_owner, ua_uxa_owner_kind, uxa_uuid, ua_name, ua_owner_kind,
         ua_disable_on_edit, ua_readonly, ua_safe_mode, uxa_kind, ua_usr_id_field,
         uxa_menu_kind, ua_auto_refresh_seconds, ua_trigger_code, ua_preblank_save
  from uxp$uxalias_info(:ua_uuid)
  into :uxa_owner, :ux_kind, :uxa_uuid, :ux_caption, ua_owner_kind,
       :ua_disable_on_edit, :ua_readonly, :ua_safe_mode, uxa_kind, ua_usr_id_field,
       uxa_menu_kind, ua_auto_refresh_seconds, ua_trigger_code, ua_preblank_save;

  if (uxa_owner is null) then
    exception e_message f_format('UxAction not found: %1', coalesce(uuid_to_char(ua_uuid), 'null'));

  mm_uuid = uxf$mainmenu();

  select ua_uuid
  from ux$action
  where ua_owner = :mm_uuid and
        ua_uxa_uuid = :uxa_uuid and
        ua_usr_id_field = ''
  into main_ua_uuid;

  ux_caption = uxf$lang(ux_caption);
  ufm_uuid = uxf$ver(uxa_owner, ver_kind);
  uxa_code = uxf$code(uxa_uuid, 'fcode');

  ux_kind_code = uxf$kind(ux_kind, 'code');

  select ufm_async_mode, ufm_updates_kind, ufm_use_report_db, ufm_web_help, ufm_lock_change_db,
         ufm_dont_prepare_on_show, ufm_source_refresh, ufm_lifetime_secs, ufm_close_on_save,
         iif(ux_fcode = 'fm_ux$preblanks', 1, 0),
         trim(iif(ux_release_uuid = ufm_uuid, 'release', 'draft'))
  from ux$form, ux$code
  where ufm_uuid = :ufm_uuid and
        ufm_ux_uuid = ux_uuid
  into us_async_mode, ufm_updates_kind, us_use_report_db, ufm_web_help, ufm_lock_change_db,
       ufm_dont_prepare_on_show, ux_source_refresh, ux_lifetime_secs, ux_close_on_save,
       ux_is_preblanks,
       ux_ver_kind;

  ufm_excel = null;
  select urt_excel
  from ux$report
  where urt_uuid = :ufm_uuid
  into ufm_excel;

  if (ux_is_preblanks = 0) then
    ua_preblank_save = '';

  ufm_web_help = trim(coalesce(ufm_web_help, ''));

  us_allow_switch_db = 1;
  if (ufm_lock_change_db = 1) then
    us_allow_switch_db = 0;

  ux_report_kind = '';
  ux_report_import_kind = 0;
  ux_report_import_id = 0;

  if (ux_kind_code = 'uxcommand') then
  begin
    ua_action_kind = 'command';
    ufm_updates_kind = 1; -- Не кэшированные изменения данных
    ux_close_on_save = 1;
  end
  else
  if (ux_kind_code = 'uxreport') then
  begin
    ua_action_kind = 'report';

    select urt_kind, urt_import_kind, urt_import_id
    from ux$report
    where urt_uuid = :ufm_uuid
    into urt_kind, ux_report_import_kind, ux_report_import_id;

    if (urt_kind = 3) then
      urt_kind = 2;

    ux_report_kind = trim(case urt_kind
      when 0 then 'grid'
      when 1 then 'excel'
      when 2 then 'fastreport'
      when 3 then 'blank'
    end);

    if (ux_report_kind = '') then
      exception e_default f_format('Unknown Report kind: %1', ux_report_kind);

    if (ua_action_kind = 'report') then
      ux_split_params = 1;     -- Только для табличных отчетов параметры показываем в отдельном окне
  end
  else
  begin
    ua_action_kind = trim(case
      when ua_usr_id_field <> '' then 'search'
      when uxa_kind = 2 then 'insert'
      else 'open'
    end);
  end

  if (ua_action_kind <> 'open') then
    ua_auto_refresh_seconds = 0;

  if (ua_action_kind = 'search') then
    ux_close_on_save = 0;

  if (ufm_dont_prepare_on_show = 0 and ux_kind_code in ('uxreport', 'uxcommand') and ux_report_import_kind <> 2) then
    ufm_dont_prepare_on_show = 1;

  if (ufm_dont_prepare_on_show = 1) then
  begin
    if (ua_action_kind in ('search', 'insert')) then
      ufm_dont_prepare_on_show = 0;
    else
    begin
      if (not exists(select * from ux$frame
                     where ufr_owner = :ufm_uuid and
                           ufr_is_grid = 1)) then
        ufm_dont_prepare_on_show = 0;
    end
  end

  if (ux_split_params = 1) then
    ux_split_param_width = 300; -- Ширина отдельного окна ввода параметров

  ux_code = uxf$code(uxa_owner, 'fcode');

  if (ua_action_kind is null) then
    exception e_default f_format('Unknown action_kind, ua_uuid: %1', uuid_to_char(ua_uuid));

  ux_is_doc = 0;

  if (ua_safe_mode = 0) then
  begin
    ux_doc_frame_uuid = uxf$docframe(ufm_uuid);

    if (ux_doc_frame_uuid is not null) then
      ux_is_doc = 1;
  end

  select ufr_uuid, ufr_can_insert, ufr_is_grid, uxf$code(ufr_uuid, 'fcode')
  from ux$frame, ux$source
  where ufr_owner = :ufm_uuid and
        ufr_is_mainframe = 1 and
        ufr_us_uuid = us_uuid
  into mainframe_uuid, ufr_can_insert, ufr_is_grid, mainframe_code;

  if (ua_safe_mode = 0 and ux_is_doc = 1 and ux_doc_frame_uuid is null) then
    ux_doc_frame_uuid = mainframe_uuid; -- в режиме теста это всегда главный фрейм

  if (ux_is_doc = 1) then
  begin
    select ua_uuid
    from ux$action
    where ua_uxa_fcode = 'afm_ust$docrepost.find' and -- Кодовое имя алиаса формы "Мои документы для перепроведения"
          ua_inactive = 0 and
          ua_usr_id_field = '' and
          ua_owner = uxf$mainmenu()
    into ux_repost_ua_uuid;
  end

  if (ufr_is_grid = 0) then
    us_async_mode = 0;

  if (ufr_is_grid = 1) then
    ux_lifetime_secs = 0;

  if (ua_action_kind in ('command', 'insert')) then
  begin
    us_async_mode = 0;
    us_use_report_db = 0;
    us_allow_switch_db = 0;
  end

  ux_commit_dialog = 0;
  if (ua_action_kind = 'command') then
  begin
    select cmd_open_kind, cmd_source_rows, cmd_commit_dialog
    from ux$command
    where cmd_uuid = :ufm_uuid
    into :ux_open_kind, :ux_source_rows, ux_commit_dialog;

    if (ux_open_kind in (0, 2)) then
    begin
      ux_form_invisible = 1;
      unlinked_fields = '';

      for select ufp_uuid, uf_field_name
      from ux$frparam, ux$field
      where ufp_ufr_uuid = :mainframe_uuid and
            ufp_uf_uuid = uf_uuid and
            uf_is_param = 1 and
            ufp_editor <> 'dialogedit' and
            ufp_public = 1 and
            uf_edi_uuid is null
      into ufp_uuid, uf_field_name
      do
      begin
        is_linked = 0;
        select iif(res_state <> 0 and (res_source_field <> '' or res_def_value <> ''), 1, 0)
        from uxp$uxfrparam_value(:ufp_uuid, :uxa_uuid, :ua_uuid)
        into :is_linked;

        if (is_linked = 0) then
        begin
          if (not exists(select * from ux$actionparam
          where uap_ua_uuid = :ua_uuid and
                uap_param_name = :uf_field_name and
                uap_is_output = 0)) then
          begin
            if (char_length(unlinked_fields || ', ' || uf_field_name) <= 255) then
            begin
              if (unlinked_fields <> '') then
                unlinked_fields = unlinked_fields || ', ';

              unlinked_fields = unlinked_fields || uf_field_name;
            end

            ux_form_invisible = 0;
          end
        end
      end

      if (ux_open_kind = 2 and ux_form_invisible = 0) then
        exception e_message f_format('Окно команды %1 невидимое.', upper(ux_code)) || ascii_char(13) ||
                                     'Имеются параметры без связи:' || ascii_char(13) ||
                                     upper(unlinked_fields);
    end

    if (ux_form_invisible = 0) then
      ux_commit_dialog = 0;
  end else
  begin
    if (ua_action_kind = 'insert') then
    begin
      ux_source_refresh = 1;

      if (ufr_can_insert = 0) then
        exception e_message f_format('Фрейм не поддерживает создание записи, fr_code: %1', mainframe_code);
    end
  end

  edit_frames = 0;
  for select ufr_uuid, uxf$code(ufr_uuid, 'fcode'), ufr_name, ufr_is_grid,
             ufr_can_update, ufr_can_insert, ufr_can_delete, ufr_us_uuid, us_is_table, ufr_add_table_fields
  from ux$frame, ux$source
  where ufr_owner = :ufm_uuid and
        ufr_us_uuid = us_uuid
  into ufr_uuid, ufr_code, ufr_name, ufr_is_grid,
       ufr_can_update, ufr_can_insert, ufr_can_delete, ufr_us_uuid, us_is_table, ufr_add_table_fields
  do
  begin
    frame_count = frame_count + 1;
    unlinked_fields = '';
    main_us_uuid = null;

    if (coalesce(ufr_add_table_fields, '') <> '') then
    begin
      select us_uuid
      from ux$source
      where us_code = :ufr_add_table_fields
      into main_us_uuid;
    end

    if (main_us_uuid is null) then
      main_us_uuid = ufr_us_uuid;

    if (ufr_can_update = 1 or ufr_can_insert = 1 or ufr_can_delete = 1) then
    begin
      edit_frames = 1;
      pk_count = 0;
      for select ufp_uuid, uf_field_name
      from ux$frparam, ux$field
      where ufp_ufr_uuid = :ufr_uuid and
            ufp_uf_uuid = uf_uuid and
            uf_pk_field <> 0
      into ufp_uuid, uf_field_name
      do
      begin
        pk_count = pk_count + 1;

        if (ufr_can_update = 1) then
        begin
          if (exists(select * from ux$frdata
          where ufd_ufr_uuid = :ufr_uuid and
                ufd_kind = 1 and
                ufd_us_uuid not in (:ufr_us_uuid, :main_us_uuid))) then
          begin
            if (not exists(select * from ux$framesql
            where ufs_ufp_uuid = :ufp_uuid and
                  ufs_sql_kind = 1)) then
              exception e_message f_format('Фрейм %1 (%2), ключевое поле %3 не прикреплен к Update SQL полям.',
                                           upper(ufr_code), ufr_name, upper(uf_field_name));
          end
        end

        if (ufr_can_delete = 1) then
        begin
          if (exists(select * from ux$frdata
          where ufd_ufr_uuid = :ufr_uuid and
                ufd_kind = 3 and
                ufd_us_uuid not in (:ufr_us_uuid, :main_us_uuid))) then
          begin
            if (not exists(select * from ux$framesql
            where ufs_ufp_uuid = :ufp_uuid and
                  ufs_sql_kind = 3)) then
              exception e_message f_format('Фрейм %1 (%2), ключевое поле %3 не прикреплен к Delete SQL полям.',
                                           upper(ufr_code), ufr_name, upper(uf_field_name));
          end
        end
      end

      if (pk_count = 0 and us_is_table = 1) then
        exception e_message f_format('Фрейм %1 (%2) обозначен', upper(ufr_code), ufr_name) || ascii_char(13) ||
                                     'как модифицируемый, но ключевые поля не указаны';
    end

    -- если фрейм однострочный, и не связан как дочернее,
    -- то проверить наличие привязки параметров по ключевым полям
    if (ufr_is_grid = 0 and ua_action_kind <> 'insert') then
    begin
      for select ufp_uuid, uf_field_name
      from ux$frparam, ux$field
      where ufp_ufr_uuid = :ufr_uuid and
            ufp_uf_uuid = uf_uuid and
            uf_is_param = 0 and
            uf_pk_field <> 0 and
            ufp_parent is null and
            ufp_public = 1
      into ufp_uuid, uf_field_name
      do
      begin
        if (ua_owner_kind <> 'uxextform') then
        begin
          if (not exists(select * from ux$actionparam
          where uap_ua_uuid = :ua_uuid and
                uap_param_name = :uf_field_name and
                uap_is_output = 0 and
                uap_owner_ufp_uuid is not null)) then
          begin
            if (not exists(select * from ux$actionparam
            where uap_ua_uuid = :main_ua_uuid and
                  uap_param_name = :uf_field_name and
                  uap_is_output = 0 and
                  uap_exec_param_id <> 0)) then
            begin
              if (char_length(unlinked_fields || ', ' || uf_field_name) <= 255) then
              begin
                if (unlinked_fields <> '') then
                  unlinked_fields = unlinked_fields || ', ';
                unlinked_fields = unlinked_fields || uf_field_name;
              end
            end
          end
        end else
        begin
          if (not exists(select * from ux$actionparam
          where uap_ua_uuid = :ua_uuid and
                uap_param_name = :uf_field_name and
                uap_is_output = 0 and
                coalesce(uap_owner_ext_field, '') <> '')) then
          begin
            if (char_length(unlinked_fields || ', ' || uf_field_name) <= 255) then
            begin
              if (unlinked_fields <> '') then
                unlinked_fields = unlinked_fields || ', ';
              unlinked_fields = unlinked_fields || uf_field_name;
            end
          end
        end
      end
    end

    if (unlinked_fields <> '') then
      exception e_message f_format('Фрейм %1 (%2)', upper(ufr_code), ufr_name) || ascii_char(13) ||
                                   'является однострочным. ключевые поля без связи:' || ascii_char(13) ||
                                   upper(unlinked_fields);
  end

  ua_orig_readonly = ua_readonly;

  if (edit_frames = 0 and ua_action_kind <> 'command') then
    ua_readonly = 1;

  -- Вызывается как поисковик, превращаем в безопасный режим
  if (ua_readonly = 0 and ua_action_kind = 'search' and ua_safe_mode = 0) then
    ua_safe_mode = 1;

  if (ua_action_kind = 'report') then
  begin
    ua_readonly = 1;
    ua_safe_mode = 1;

    if (ux_report_kind = 'excel' and ufm_excel is null and frame_count > 1) then
    begin
      select ugr_excel
      from ux$frame, ux$grid
      where ufr_owner = :ufm_uuid and
            ufr_uuid = ugr_ufr_uuid and
            ugr_excel is not null
      order by ufr_is_mainframe desc, ugr_default desc
      rows 1
      into ufm_excel;
    end
  end

--  if (us_async_mode = 1 and frame_count > 1) then
--  us_async_mode = 0;

  if (ua_readonly = 0) then
  begin
    us_use_report_db = 0;
    us_allow_switch_db = 0;
  end

  if (us_use_report_db = 1) then
  begin
    if (uxf$rep_db() = '') then
    begin
      us_use_report_db = 0;
      us_allow_switch_db = 0;
    end
  end

--  select list(uf_field_name || '=' || uap_param_name)
--  from ux$actionparam, ux$frparam, ux$field
--  where uap_ua_uuid = :ua_uuid and
--        uap_owner_ufp_uuid = ufp_uuid and
--        uap_is_output = 0 and
--        ufp_uf_uuid = uf_uuid and
--        uf_pk_field <> 0
--  into ux_source_locate_fields;

  for select uap_param_name, uap_owner_ufp_uuid
  from ux$actionparam
  where uap_ua_uuid = :ua_uuid and
        uap_owner_ufp_uuid is not null and
        uap_is_output = 0
  into uap_param_name, uap_owner_ufp_uuid
  do
  begin
    if (not exists(select * from ux$frame, ux$frparam, ux$field
                   where ufr_owner = :ufm_uuid and
                         ufr_uuid = ufp_ufr_uuid and
                         ufp_uf_uuid = uf_uuid and
                         uf_field_name = :uap_param_name)) then
      Continue;

    tmp_field = '';
    select uf_field_name
    from ux$frparam, ux$field
    where ufp_uuid = :uap_owner_ufp_uuid and
          ufp_uf_uuid = uf_uuid and
          uf_pk_field <> 0
    into tmp_field;

    if (tmp_field = '') then
      Continue;

    if (ux_source_locate_fields <> '') then
      ux_source_locate_fields = ux_source_locate_fields || ',';

    ux_source_locate_fields = ux_source_locate_fields || tmp_field || '=' || uap_param_name;
  end

  ua_has_outputs = 0;
  if (exists(select * from ux$actionparam
  where uap_ua_uuid = :ua_uuid and
        uap_is_output = 1)) then
    ua_has_outputs = 1;

  if (ux_source_locate_fields is null) then
    ux_source_locate_fields = '';

  ux_width = 0;
  ux_height = 0;

  select ufi_width, ufi_height
  from ux$fritem
  where ufi_owner = :ufm_uuid and
        ufi_parent_uuid is null
  into :ux_width, :ux_height;

  if (ua_action_kind = 'report' and ux_height = 0) then
    ux_height = 500;

  ux_dialogedit_is_param = 0;
  ux_dialogedit_ua_uuid = null;
  ux_dialogedit_field = '';
  ux_dialogedit_find_field = '';
  ua_usr_text_field = '';
  dialog_uxa_uuid = null;

  -- Диалоговое окно НЕ открывается для полей если это открыть или действие.
  -- Открыааем если Insert и поле, либо это параметр
  -- Если их несколько, берем только первую

  select ua_uxa_uuid, usr_id_field, ua_usr_id_field, uf_is_param, ua_usr_text_field
  from ux$action, ux$frparam, ux$search, ux$field
  where ua_owner = :mainframe_uuid and
        ufp_ufr_uuid = ua_owner and
        ufp_uf_uuid = uf_uuid and
        uf_field_name = ua_usr_id_field and
        ufp_editor = 'dialogedit' and
        ua_uxa_uuid = usr_uuid
  order by uf_is_param desc, uf_position
  rows 1
  into dialog_uxa_uuid, ux_dialogedit_find_field, ux_dialogedit_field, ux_dialogedit_is_param, ua_usr_text_field;

  ux_dialogedit_multi_delimiter = '';
  if (ua_usr_text_field containing '$') then
    ux_dialogedit_multi_delimiter = ua_usr_text_field;

  if (dialog_uxa_uuid is not null) then
  begin
    if (ua_action_kind = 'insert' or ux_dialogedit_is_param = 1) then
    begin
      select ua_uuid
      from ux$action
      where ua_owner = :mainframe_uuid and
            ua_usr_id_field = :ux_dialogedit_field and
            ua_uxa_uuid = :dialog_uxa_uuid
      into ux_dialogedit_ua_uuid;
    end else
    begin
      ux_dialogedit_is_param = 0;
      ux_dialogedit_ua_uuid = null;
      ux_dialogedit_field = '';
    end
  end

  ux_public_param_sql = '';
  need_public_params = 0;
  if (ux_split_params = 1) then
    need_public_params = 1;
  else
  if (frame_count > 1) then
  begin
    for select uf_field_name
    from ux$frame, ux$frparam, ux$field
    where ufr_owner = :ufm_uuid and
--          ufr_master_ufr_uuid is null and
          ufr_uuid = ufp_ufr_uuid and
          ufp_uf_uuid = uf_uuid and
          uf_is_param = 1 and
          ufr_detail_field <> uf_field_name
    group by 1
    having count(*) > 1
    into uf_field_name
    do
    begin
      if (exists(select * from ux$frame, ux$fritem, ux$field
                 where ufr_owner = :ufm_uuid and
                       ufi_hidden = 0 and
                       ufi_ufr_uuid = ufr_uuid and
                       ufi_uf_uuid = uf_uuid and
                       uf_field_name = :uf_field_name and
                       uf_is_param = 1)) then
      begin
        need_public_params = 1;
        break;
      end
    end
  end

  if (need_public_params = 1) then
  begin
    select sql_text
    from uxp$uxframe_sql_param(:mainframe_uuid, 1)
    into ux_public_param_sql;
  end

  suspend;
end
