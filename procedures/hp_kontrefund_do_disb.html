CREATE PROCEDURE HP_KONTREFUND_DO_DISB(
    JSON_DATA D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_LONG)
AS
declare variable repay_count d_integer;
declare variable disb_k_id d_id;
declare variable disb_amount d_currency;
declare variable repay_k_id d_id;
declare variable repay_amount d_currency;
declare variable repay_no d_integer;
declare variable ol_repay_amount d_currency;
declare variable loan_ol_id d_id;
declare variable error_text d_str_255;
declare variable disb_date d_date;
declare variable tmp_amount d_currency;
declare variable kf_id d_id;
declare variable reg_id d_id;
declare variable ol_id d_id;
declare variable ma_id d_integer;
declare variable manager_id d_id;
declare variable k_approved_manager d_id;
declare variable cash_account d_str_code;
declare variable cash_bal_acc d_str_code;
declare variable db_id d_id;
declare variable cashier_id d_id;
declare variable krd_id d_id;
declare variable res_disb_ol_ids d_str_255;
declare variable disb_ol_id d_id;
declare variable pos d_integer;
declare variable repays d_str_long;
declare variable repay_ol_id d_id;
declare variable tmp_item d_str_255;
declare variable k_kommis_perc d_percent;
declare variable pr_code d_str_100;
begin
  -- json_data - список займов (repay), которых нужно закрыть и выдать новую disb_k_id
  -- json_data:
  -- {"disb_k_id":"111111111", "repay":[{"k_id":2222222, "amount":10000}, {"k_id":33333333, "amount":15000}]}

  res_is_error = 1;
  res_text = '';

  disb_k_id = f_json_str_get_value(json_data, 'disb_k_id', null);
  if (disb_k_id is null) then
  begin
    res_text = 'Займ для выдачи не указан';
    suspend;
    exit;
  end

  select k_disb_date, k_amount_approved, k_from_kf_id, k_from_ma_app_id, k_approved_manager, k_kommis_perc, pr.pr_code
  from kontract kont
  inner join product pr on pr.pr_id = kont.k_from_pr_id
  where k_id = :disb_k_id
  into disb_date, disb_amount, kf_id, ma_id, k_approved_manager, k_kommis_perc, :pr_code;

  if (coalesce(k_kommis_perc, 0) > 0) then
  begin
    res_text = 'При рефинансировании через МП доп.процент должен быть 0. Сейчас указан ' || coalesce(k_kommis_perc, 0) ||'%';
    suspend;
    exit;
  end

  if (ma_id is not null) then
  begin
    select ma_manager_id
    from miniapp
    where ma_id = :ma_id
    order by ma_id desc, ma_number desc
    rows 1
    into manager_id;
  end

  if (manager_id is null) then
    manager_id = k_approved_manager;

  repay_count = f_json_str_get_count(json_data, 'repay');
  if (repay_count = 0) then
  begin
    res_text = 'Кредиты для погашения не указаны';
    suspend;
    exit;
  end

--  if (repay_count > 1) then
--  begin
--    res_text = 'Необходимо выбрать только один займ для закрытия';
--    suspend;
--    exit;
--  end

  repay_no = 0;
  tmp_amount = 0;
  while (repay_no < repay_count) do
  begin
    repay_k_id = f_json_str_get_value(json_data, 'repay[' || repay_no || '].k_id', null);
    repay_amount = f_json_str_get_value(json_data, 'repay[' || repay_no || '].amount', 0);

    if (repay_k_id is null or repay_amount = 0) then
    begin
      res_text = 'Кредит для погашения или сумма не указаны';
      suspend;
      exit;
    end
    else
      tmp_amount = tmp_amount + repay_amount;
    repay_no = repay_no + 1;
  end

  if (tmp_amount <> disb_amount) then
  begin
    res_text = 'Сумма погашений отличается от суммы займа.';
    suspend;
    exit;
  end

  select k.kal_new_reg_id
  from knaktlist k
  where k.kal_from_kf_id = :kf_id and
        k.is_delete = 0 and
        :disb_date between k.kal_date and k.kal_date_end
  into :reg_id;

  if (reg_id is null) then
  begin
    res_text = 'Офис выдачи не определен';
    suspend;
    exit;
  end

  cash_account = '';
  cash_bal_acc = '10023';
  db_id = f_get_db_id();

  cash_bal_acc = case db_id
                   when 41701 then '10023'
                   when 81001 then '50-01'
                   when 94401 then '100-1'
                 end;

  select al_id
  from acc_list
  where al_from_reg_id = :reg_id and
        al_number = iif(:pr_code containing '.ipf.', 101, 1) and
        al_from_bg_id = :cash_bal_acc and
        al_currency = f_get_const('cm_default_currency')
  into cash_account;

  if (coalesce(cash_account, '') = '') then
  begin
    res_text = 'Счет кассы не найден';
    suspend;
    exit;
  end

  select left(list(kf_client_fio || ': ' || kf_error_text), 1024)
  from hp_edoc_sign_check_kont(:disb_k_id)
  where kf_error_is_critical > 0
  into :res_text;

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  select left(list(kf_client_fio || ': ' || kf_error_text), 1024)
  from hp_check_loan_konditions(:disb_k_id, 0, 0, 'loan_disb', :un_id)
  where kf_error_is_critical > 0
  into :res_text;

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  repays = '';
  repay_no = 0;
  while (repay_no < repay_count) do
  begin
    tmp_item = '{"k_id":%1,"ol_id":%2,"amount":%3}';
    repay_k_id = f_json_str_get_value(json_data, 'repay[' || repay_no || '].k_id', null);
    repay_amount = f_json_str_get_value(json_data, 'repay[' || repay_no || '].amount', 0);

    cashier_id = null;
    select kc_from_c_id
    from knclient
    where kc_from_k_id = :repay_k_id and
          kc_amount_approved > 0 and
          kc_status = 2
    rows 1
    into cashier_id;

    select ol_id, loan_ol_id, ol_repay_amount, error_text
    from hp_do_repayment(:repay_k_id, 0, :disb_date, :repay_amount, 1, 1, 1, 1, 1, 1, 0, :reg_id, 1, :cash_account, 1, :un_id, :cashier_id, null, :manager_id, null, 0, 0)
    into :ol_id, :loan_ol_id, :ol_repay_amount, :error_text;

    if (error_text = '' and
        ol_repay_amount <> repay_amount) then
      error_text = 'Сумма для закрытия займа отличается';

    if (error_text = '') then
    begin
      if (not exists(select *
                     from kontstate
                     where ks_from_k_id = :repay_k_id and
                           ks_state = 2 and
                           ks_ol_is_delete = 0 and
                           ks_from_ol_id = :loan_ol_id and
                           ks_ol_is_delete = 0)) then
        error_text = 'Кредит не закрылся. Обратитесь к техподдержке.';
    end

    if (error_text = '') then
    begin
      if (repays <> '') then
        repays = repays || ',';
      repays = repays || f_format(tmp_item, repay_k_id, ol_id, repay_amount);
    end

    res_text = error_text;

    if (res_text <> '') then
    begin
      suspend;
      exit;
    end

    repay_no = repay_no + 1;
  end

  if (ol_id is not null) then
  begin
    cashier_id = null;
    select kc_from_c_id
    from knclient
    where kc_from_k_id = :disb_k_id and
          kc_amount_approved > 0 and
          kc_status = 2
    rows 1
    into cashier_id;

    select res_ol_ids, res_text
    from hp_do_disbursement(:un_id, :disb_k_id, :reg_id, 1, :cash_account, :reg_id, 1, 0, :cashier_id, :manager_id)
    into res_disb_ol_ids, :res_text;

    if (res_text <> '') then
    begin
      suspend;
      exit;
    end
  end

  pos = position('-', res_disb_ol_ids);
  if (pos > 0) then
    disb_ol_id = left(res_disb_ol_ids, pos - 1);
  insert into kontrefund_disb (krd_from_k_id, krd_amount, krd_from_ol_id, krd_status, krd_create_un_id)
  values (:disb_k_id, :disb_amount, :disb_ol_id, 0, :un_id)
  returning krd_id
  into :krd_id;

  if (repays <> '') then
  begin
    repays = '{"pay_list":[' || repays || ']';
    repay_count = f_json_str_get_count(repays, 'pay_list');
    repay_no = 0;
    while (repay_no < repay_count) do
    begin
      repay_k_id = f_json_str_get_value(repays, 'pay_list[' || repay_no || '].k_id', null);
      repay_ol_id = f_json_str_get_value(repays, 'pay_list[' || repay_no || '].ol_id', null);
      repay_amount = f_json_str_get_value(repays, 'pay_list[' || repay_no || '].amount', 0);
  
      insert into kontrefund_pay (krp_from_krd_id, krp_from_k_id, krp_amount, krp_from_ol_id)
      values (:krd_id, :repay_k_id, :repay_amount, :repay_ol_id);
  
      repay_no = repay_no + 1;
    end
  end

  res_is_error = 0;
  res_text = 'Займ успешно выдан выдан через рефинансирование';
  suspend;
end
