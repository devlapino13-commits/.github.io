CREATE PROCEDURE CB_KG_KM_2017_02_IGNOR_KONTR(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    DELINQ_BALANCE_7 CURR_TYPE,
    BALANCE_BEGIN CURR_TYPE,
    ZNACH_PAR7 CURR_TYPE,
    BONUS_FACTOR CURR_TYPE,
    KOL_SSVZ INTEGER,
    FIO_SSVZ VARCHAR(1024),
    REG_NAME VARCHAR(100),
    BONUS_KM INTEGER,
    SUM_VOZNOG_OFFICE CURR_TYPE,
    REPAY_SUM_NOT_OS CURR_TYPE,
    DELINQ_OS_PERIOD CURR_TYPE,
    REPAY_KOMMIS_IPF CURR_TYPE,
    PLAN_SPF INTEGER,
    FACT_SPF INTEGER)
AS
declare variable rep_date_begin date;
declare variable reg_id integer;
declare variable rep_period integer;
declare variable diff_spf integer;
begin

  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;
  rep_period = extract(year from rep_date_end) || right((extract(month from rep_date_end) + 100), 2);

  for select oks_region_id, sum(iif(:rep_date_end - oks_delinq_date > 7, oks_balance, 0))
      from oksummary oks, region_list(:rep_region_id, 1, 3, 1)
      where oks_region_id = id_list and
            --oks_from_k_id <> ign_from_k_id and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_close_state = 0 and
            not exists(select * from IGNOR_CONTRACT_BONUS_KM_KG  where ign_from_k_id = oks_from_k_id)
      group by oks_region_id
      into reg_id, delinq_balance_7
  do
  begin
    --  Балан КП и баланс PAR7 на начало
    balance_begin = null;
    select sum(oks_balance)
    from oksummary
    where oks_region_id = :reg_id and
          :rep_date_begin - 1 between oks_date_begin and oks_date_end and
          oks_close_state = 0 and
          not exists(select * from IGNOR_CONTRACT_BONUS_KM_KG  where ign_from_k_id = oks_from_k_id)
    into balance_begin;
    balance_begin = coalesce(balance_begin, 0);
    --  Значение PAR7* в % от КП
    if (balance_begin = 0 and
        delinq_balance_7 > 0) then
      znach_par7 = 100;
    else
    if (delinq_balance_7 = 0) then
      znach_par7 = 0;
    else
      znach_par7 = cast(delinq_balance_7 as double precision) / balance_begin * 100;

    --  Суммы погашении процентов пеня и админ сбор
    repay_sum_not_os = null;
    select -sum(ok_perc + ok_fine + ok_kommis)
    from operkred
    where ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_ol_reg_id = :reg_id and
          ok_ol_date between :rep_date_begin and :rep_date_end and
          not exists(select * from IGNOR_CONTRACT_BONUS_KM_KG i where i.ign_from_k_id = ok_from_k_id)
    into repay_sum_not_os;
    repay_sum_not_os = coalesce(repay_sum_not_os, 0);

    --  Комиссия за выдачу ИПФ
    repay_kommis_ipf = null;
    select sum(ok_amount)
    from refdata, operkontract, operlist
    where rd_code = 'CashTransactionIncom.IPFCommission' and
          rd_id = ok_operation_type and
          ok_from_ol_id = ol_id and
          operlist.is_delete = 0 and
          ol_from_reg_id = :reg_id and
          ol_date between :rep_date_begin and :rep_date_end and
          not exists(select * from IGNOR_CONTRACT_BONUS_KM_KG i where i.ign_from_k_id = OK_FROM_K_ID)
    into repay_kommis_ipf;
    repay_kommis_ipf = coalesce(repay_kommis_ipf, 0);

    --  Сумма просроченной задолженности по ОС за отчетный период
    delinq_os_period = null;
    select sum(ok_os + ok_sh_os)
    from oksummary, operkred
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_region_id = :reg_id and
          :rep_date_end >= oks_delinq_date and
          oks_from_k_id = ok_from_k_id and
          oks_close_state = 0 and
          ok_ol_is_active = 1 and
          ok_type < 7 and
          ok_sh_date between :rep_date_begin and :rep_date_end and
          ok_ol_date <= :rep_date_end and
          not exists(select * from IGNOR_CONTRACT_BONUS_KM_KG  where ign_from_k_id = oks_from_k_id)
    into delinq_os_period;
    delinq_os_period = coalesce(delinq_os_period, 0);
    --  Сумма вознаграждения подотчетного офиса
    sum_voznog_office = (repay_sum_not_os + repay_kommis_ipf) * 9 / 100 - delinq_os_period * 40 / 100;

    --  Бонус фактор
    if (rep_date_end <= '31.03.2019') then
    begin
      bonus_factor = case
                       when znach_par7 = 0.0 then 1.20
                       when znach_par7 <= 0.5 then 1.00
                       when znach_par7 <= 1.0 then 0.90
                       when znach_par7 <= 1.5 then 0.50
                       else 0
                     end;
    end
    else
    begin
      bonus_factor = case
                       when znach_par7 <= 0.1 then 1.20
                       when znach_par7 <= 0.5 then 1.00
                       when znach_par7 <= 1.0 then 0.90
                       when znach_par7 <= 1.5 then 0.50
                       else 0
                     end;
    end
    kol_ssvz = 0;
    fio_ssvz = '';
    select count(*), list(c_name_fio, '; ')
    from hr_pstatus, joblist, jobkind, clients
    where ps_status = 1 and
          ps_is_working = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code = 'JobSystem.Region_KM' and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_reg_id = :reg_id and
          ps_from_c_id = c_id and
          (ps_date_begin <= dateadd(month, -3, :rep_date_end) or
          (select min(hr1.ps_date_begin)
           from hr_pstatus hr1
           where hr1.ps_is_working = 1 and
                 hr1.ps_status = 1 and
                 hr1.ps_from_c_id = c_id) <= dateadd(month, -3, :rep_date_end)) and
          coalesce((select sawt_id
                    from hr_voctravel, hr_orders, sa_worktype
                    where hrv_from_ho_id = ho_id and
                          ho_order_type = sawt_id and
                          hrv_from_c_id = ps_from_c_id and
                          :rep_date_end between hrv_date_begin and hrv_date_end_fact
                    order by hrv_date_begin desc
                    rows 1), 0) not in (16, 17, 18, 8) -- Декретников не включает
    into kol_ssvz, fio_ssvz;
    if (kol_ssvz is null) then
      kol_ssvz = 0;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;

    --  План СПФ
    plan_spf = null;
    select prs_plan_spf
    from planregionspf
    where prs_period = :rep_period and
          prs_kind = 1 and
          prs_region = :reg_id
    into plan_spf;
    plan_spf = coalesce(plan_spf, 0);

    --  Факт СПФ
    fact_spf = null;
    select count(distinct oks_kc_id)
    from oksummary
    where oks_region_id = :reg_id and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_close_state = 0 and
          not exists(select * from IGNOR_CONTRACT_BONUS_KM_KG  where ign_from_k_id = oks_from_k_id)
    into fact_spf;
    fact_spf = coalesce(fact_spf, 0);

    diff_spf = plan_spf - fact_spf;
    if (diff_spf < 0) then
      diff_spf = 0;

    bonus_km = sum_voznog_office * 0.15 * bonus_factor;

    if (bonus_km < 0) then
      bonus_km = 0;

    --  штраф 10% с  бонуса к начислению за каждого недостающего СПФ относительно планового показателя количества СПФ на конец месяца.
    if (bonus_km > 0 and
        diff_spf > 0) then
    begin
      if (diff_spf > 10) then
        diff_spf = 10;
      if (rep_period <> 202004) then
        bonus_km = bonus_km - bonus_km * diff_spf * 10 / 100;
    end

    suspend;
  end
end
