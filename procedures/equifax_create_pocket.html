CREATE PROCEDURE EQUIFAX_CREATE_POCKET(
    INPUT_DATE DATE,
    POCKET_TYPE VARCHAR(50))
RETURNS (
    POCKET_ID INTEGER,
    POCKET_NAME STR_255,
    POCKET_DATE DATE)
AS
declare variable y varchar(4);
declare variable m varchar(2);
declare variable d varchar(2);
declare variable v varchar(1);
declare variable cp_counter integer;
declare variable cib_id integer;
declare variable pocket_type_tmp d_str_100;
declare variable cp_date_begin d_date;
begin
  /*
  POCKET_TYPE
    .d.kont.    - ежедневный по кредитам
    .d.miniapp. - ежедневный по заявкам
    .m.kont.    - ежемесячный по кредитам
    .m.miniapp. - ежемесячный по заявкам
  */
  -- '01.07.2014' -- дата запуска проекта по выгрузке кредитных историй
  -- '01.03.2015' -- дата запуска проекта по выгрузке ФИЧ-файла

  pocket_date = input_date;
  cp_date_begin = pocket_date;


  --  pocket_type_tmp может быть либо по кредиту (.kont.) илбо по заявке (.miniapp.), на этом все
  -- удаляю лишние слова
  if (pocket_type containing ('.miniapp.')) then
    pocket_type_tmp = '.miniapp.';
  else if (pocket_type containing ('.kont.')) then
    pocket_type_tmp = '.kont.';

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'CIB.EQUIFAX'
  into cib_id;

  cp_counter = 0;

  select cp.cp_counter
  from cib_pocket cp
  where cp.cp_date_end = :pocket_date and
        cp.cp_pocket_type containing (:pocket_type_tmp) and
        cp.cp_cib_id = :cib_id and
        cp.cp_pocket_status >= 0
  order by cp.cp_date_end desc, cp.cp_id desc
  rows 1
  into :cp_counter;

  cp_counter = cp_counter + 1;

  if (pocket_type containing ('.miniapp.')) then
    y = cast(extract(year from pocket_date) as varchar(4));
  else
  if (pocket_type containing ('.kont.')) then
  begin
    y = cast(extract(year from pocket_date) as varchar(4));
    y = (case y when '2014' then 'E'
                when '2015' then 'F'
                when '2016' then 'G'
                when '2017' then 'H'
                when '2018' then 'I'
                when '2019' then 'J'
                when '2020' then 'K'
                when '2021' then 'L'
                when '2022' then 'M'
                when '2023' then 'N'
                when '2024' then 'O'
                when '2025' then 'P'
                when '2025' then 'Q'
                when '2026' then 'R'
                when '2027' then 'S'
                end);
  end

  m = cast(extract(month from :pocket_date) as varchar(2));
  d = cast(extract(day from :pocket_date) as varchar(2));

  if (pocket_type containing ('.kont.')) then
  begin
    m = (case m when '10' then 'A'
                when '11' then 'B'
                when '12' then 'C'
                else m end);

    d = (case d when '10' then 'A'
                when '11' then 'B'
                when '12' then 'C'
                when '13' then 'D'
                when '14' then 'E'
                when '15' then 'F'
                when '16' then 'G'
                when '17' then 'H'
                when '18' then 'I'
                when '19' then 'J'
                when '20' then 'K'
                when '21' then 'L'
                when '22' then 'M'
                when '23' then 'N'
                when '24' then 'O'
                when '25' then 'P'
                when '26' then 'Q'
                when '27' then 'R'
                when '28' then 'S'
                when '29' then 'T'
                when '30' then 'U'
                when '31' then 'V'
                else d end);

    if (cp_counter < 10) then
      v = cast(cp_counter as varchar(1));

    v  = (case cp_counter when '10' then 'A'
                          when '11' then 'B'
                          when '12' then 'C'
                          when '13' then 'D'
                          when '14' then 'E'
                          when '15' then 'F'
                          when '16' then 'G'
                          when '17' then 'H'
                          when '18' then 'I'
                          when '19' then 'J'
                          when '20' then 'K'
                          when '21' then 'L'
                          when '22' then 'M'
                          when '23' then 'N'
                          when '24' then 'O'
                          when '25' then 'P'
                          else v end);

    pocket_name = '47L' || 'U' || y || m || d || v; -- формирую название файла

    --
    if (pocket_type containing '.monthly.') then
    begin
       cp_date_begin = (select cast('01.' || cast(extract(month from cast(:pocket_date as date)) as varchar(2)) || '.' || cast(extract(year from cast(:pocket_date as date)) as varchar(4)) as date)
                     from rdb$database);
    end
  end
  else -- пакет по заявкам
  begin
    if (char_length(m) < 2) then
      m = '0' || m;
    if (char_length(d) < 2) then
      d = '0' || d;

    pocket_name = '47L' || '_' || y || m || d || '_' ||(case
                                                          when char_length(cp_counter) = 4 then cp_counter
                                                          when char_length(cp_counter) = 3 then '0' || cp_counter
                                                          when char_length(cp_counter) = 2 then '00' || cp_counter
                                                          when char_length(cp_counter) = 1 then '000' || cp_counter
                                                        end);
  end
  insert into cib_pocket (cp_counter, cp_date_begin, cp_date_end, cp_pocket_name, cp_pocket_type, cp_pocket_status,
                          cp_pocket_kib_status, cp_cib_id)
  values (:cp_counter, :cp_date_begin, :pocket_date, :pocket_name, :pocket_type, 0, null, f_get_rd_id('CIB.EQUIFAX'))
  returning cp_id
  into :pocket_id;

  suspend;
end
