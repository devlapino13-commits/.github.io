CREATE PROCEDURE AAREP_ACCOUNT_OBORSAL_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_WITH_ACCOUNT INTEGER,
    REPORT_WITH_CLIENT INTEGER,
    REPORT_WITH_SUBKONTO INTEGER,
    REPORT_WITH_CURRENCY INTEGER)
RETURNS (
    OB_BALANCE VARCHAR(20),
    OB_BALANCE_NAME VARCHAR(255),
    OB_ACCOUNT VARCHAR(20),
    OB_CURRENCY VARCHAR(3),
    OB_CLIENT_ID DECIMAL(18,0),
    OB_SUBKONTO_ID INTEGER,
    OB_END_DT DECIMAL(18,4),
    OB_END_KT DECIMAL(18,4),
    OB_OBOROT_DT DECIMAL(18,4),
    OB_OBOROT_KT DECIMAL(18,4),
    OB_END_DT_EKV DECIMAL(18,4),
    OB_END_KT_EKV DECIMAL(18,4),
    OB_OBOROT_DT_EKV DECIMAL(18,4),
    OB_OBOROT_KT_EKV DECIMAL(18,4),
    OB_ACCOUNT_NAME VARCHAR(255))
AS
declare variable curr_currency varchar(3);
declare variable ob_end decimal(18,4);
declare variable ob_end_ekv decimal(18,4);
begin
  ob_end_dt = 0;
  ob_end_kt = 0;
  ob_oborot_dt = 0;
  ob_oborot_kt = 0;
  ob_end_dt_ekv = 0;
  ob_end_kt_ekv = 0;
  ob_oborot_dt_ekv = 0;
  ob_oborot_kt_ekv = 0;
  select cm_default_currency
  from company
  into :curr_currency;
  report_with_subkonto = 0;

  ob_subkonto_id = 0;
  ob_client_id = null;

  if (report_with_client = 1) then
  begin
    for select al_from_bg_id, bg_name, al_name, ob_account_dt, ob_currency, ob_client_dt, sum(ob_summa_val),
               sum(ob_summa)
        from acc_list, balgroup, operbook, operlist, region_list(:report_region_id, 1, -1, 1)
        where ob_from_ol_id = ol_id and
              ob_ol_is_active = 1 and
              ob_account_dt = al_id and
              al_from_bg_id = bg_id and
              ob_ol_date between :report_date_begin and :report_date_end and
              ob_ol_reg_id = id_list
        group by al_from_bg_id, bg_name, al_name, ob_account_dt, ob_currency, ob_client_dt
        into :ob_balance, :ob_balance_name, :ob_account_name, :ob_account, :ob_currency, :ob_client_id, :ob_oborot_dt,
             :ob_oborot_dt_ekv
    do
    begin
      if (ob_currency = curr_currency) then
      begin
        ob_end_dt = 0;
        ob_end_kt = 0;
        ob_oborot_dt = 0;
        ob_oborot_kt = 0;
      end
      if (report_with_currency = 0) then
      begin
        ob_currency = '000';
        ob_end_dt = 0;
        ob_end_kt = 0;
        ob_oborot_dt = 0;
        ob_oborot_kt = 0;
      end
      if (report_with_account = 0) then
      begin
        ob_account = :ob_balance;
        ob_account_name = :ob_balance_name;
      end
      suspend;
    end

    ob_end_dt = 0;
    ob_end_kt = 0;
    ob_oborot_dt = 0;
    ob_oborot_kt = 0;
    ob_end_dt_ekv = 0;
    ob_end_kt_ekv = 0;
    ob_oborot_dt_ekv = 0;
    ob_oborot_kt_ekv = 0;

    for select al_from_bg_id, bg_name, al_name, ob_account_kt, ob_currency, ob_client_kt, sum(ob_summa_val),
               sum(ob_summa)
        from acc_list, balgroup, operbook, operlist, region_list(:report_region_id, 1, -1, 1)
        where ob_from_ol_id = ol_id and
              ob_ol_is_active = 1 and
              ob_account_kt = al_id and
              al_from_bg_id = bg_id and
              ob_ol_date between :report_date_begin and :report_date_end and
              ob_ol_reg_id = id_list
        group by al_from_bg_id, bg_name, al_name, ob_account_kt, ob_currency, ob_client_kt
        into :ob_balance, :ob_balance_name, :ob_account_name, :ob_account, :ob_currency, :ob_client_id, :ob_oborot_kt,
             :ob_oborot_kt_ekv
    do
    begin
      if (ob_currency = curr_currency) then
      begin
        ob_end_dt = 0;
        ob_end_kt = 0;
        ob_oborot_dt = 0;
        ob_oborot_kt = 0;
      end
      if (report_with_currency = 0) then
      begin
        ob_currency = '000';
        ob_end_dt = 0;
        ob_end_kt = 0;
        ob_oborot_dt = 0;
        ob_oborot_kt = 0;
      end
      if (report_with_account = 0) then
      begin
        ob_account = :ob_balance;
        ob_account_name = :ob_balance_name;
      end
      suspend;
    end

    ob_end_dt = 0;
    ob_end_kt = 0;
    ob_oborot_dt = 0;
    ob_oborot_kt = 0;
    ob_end_dt_ekv = 0;
    ob_end_kt_ekv = 0;
    ob_oborot_dt_ekv = 0;
    ob_oborot_kt_ekv = 0;

    for select al_from_bg_id, bg_name, al_name, ob_account_dt, ob_currency, ob_client_dt, sum(ob_summa_val),
               sum(ob_summa)
        from acc_list, balgroup, operbook, operlist, region_list(:report_region_id, 1, -1, 1)
        where ob_from_ol_id = ol_id and
              ob_ol_is_active = 1 and
              ob_account_dt = al_id and
              al_from_bg_id = bg_id and
              ob_ol_date between '01.01.1900' and :report_date_end and
              ob_ol_reg_id = id_list
        group by al_from_bg_id, bg_name, al_name, ob_account_dt, ob_currency, ob_client_dt
        into :ob_balance, :ob_balance_name, :ob_account_name, :ob_account, :ob_currency, :ob_client_id, :ob_end_dt,
             :ob_end_dt_ekv
    do
    begin
      if (ob_currency = curr_currency) then
      begin
        ob_end_dt = 0;
        ob_end_kt = 0;
        ob_oborot_dt = 0;
        ob_oborot_kt = 0;
      end
      if (report_with_currency = 0) then
      begin
        ob_currency = '000';
        ob_end_dt = 0;
        ob_end_kt = 0;
        ob_oborot_dt = 0;
        ob_oborot_kt = 0;
      end
      if (report_with_account = 0) then
      begin
        ob_account = :ob_balance;
        ob_account_name = :ob_balance_name;
      end
      suspend;
    end

    ob_end_dt = 0;
    ob_end_kt = 0;
    ob_oborot_dt = 0;
    ob_oborot_kt = 0;
    ob_end_dt_ekv = 0;
    ob_end_kt_ekv = 0;
    ob_oborot_dt_ekv = 0;
    ob_oborot_kt_ekv = 0;

    for select al_from_bg_id, bg_name, al_name, ob_account_kt, ob_currency, ob_client_kt, sum(ob_summa_val),
               sum(ob_summa)
        from acc_list, balgroup, operbook, operlist, region_list(:report_region_id, 1, -1, 1)
        where ob_from_ol_id = ol_id and
              ob_ol_is_active = 1 and
              ob_account_kt = al_id and
              al_from_bg_id = bg_id and
              ob_ol_date between '01.01.1900' and :report_date_end and
              ob_ol_reg_id = id_list
        group by al_from_bg_id, bg_name, al_name, ob_account_kt, ob_currency, ob_client_kt
        into :ob_balance, :ob_balance_name, :ob_account_name, :ob_account, :ob_currency, :ob_client_id, :ob_end_kt,
             :ob_end_kt_ekv
    do
    begin
      if (ob_currency = curr_currency) then
      begin
        ob_end_dt = 0;
        ob_end_kt = 0;
        ob_oborot_dt = 0;
        ob_oborot_kt = 0;
      end
      if (report_with_currency = 0) then
      begin
        ob_currency = '000';
        ob_end_dt = 0;
        ob_end_kt = 0;
        ob_oborot_dt = 0;
        ob_oborot_kt = 0;
      end
      if (report_with_account = 0) then
      begin
        ob_account = :ob_balance;
        ob_account_name = :ob_balance_name;
      end
      suspend;
    end
  end
  else
  begin -- без клиентов, чисто оборотка
    ob_client_id = null;
    ob_subkonto_id = 0;
    for select distinct obs_from_al_id
    from obsummary, acc_list, region_list(:report_region_id, 1, -1, 1)
    where obs_from_al_id = al_id and
          obs_date between :report_date_begin and :report_date_end and
          al_from_reg_id = id_list
    into :ob_account
    do
    begin
      select dt_ekv, kt_ekv, dt_val, kt_val
      from hp_al_obor(:ob_account, :report_date_begin, :report_date_end)
      into :ob_oborot_dt_ekv, :ob_oborot_kt_ekv, :ob_oborot_dt, :ob_oborot_kt;

      select al_from_bg_id, bg_name, al_name, al_currency
      from acc_list, balgroup
      where al_id = :ob_account and
            al_from_bg_id = bg_id
      into :ob_balance, :ob_balance_name, :ob_account_name, :ob_currency;

      if (ob_currency = curr_currency) then
      begin
        ob_end_dt = 0;
        ob_end_kt = 0;
        ob_oborot_dt = 0;
        ob_oborot_kt = 0;
      end
      if (report_with_currency = 0) then
      begin
        ob_currency = '000';
        ob_end_dt = 0;
        ob_end_kt = 0;
        ob_oborot_dt = 0;
        ob_oborot_kt = 0;
      end
      if (report_with_account = 0) then
      begin
        ob_account = :ob_balance;
        ob_account_name = :ob_balance_name;
      end
      suspend;
    end

    ob_end_dt = 0;
    ob_end_kt = 0;
    ob_oborot_dt = 0;
    ob_oborot_kt = 0;
    ob_end_dt_ekv = 0;
    ob_end_kt_ekv = 0;
    ob_oborot_dt_ekv = 0;
    ob_oborot_kt_ekv = 0;

    for select account, currency, summa_val, summa_ekv
        from aarep_account_saldo_temp(:report_region_id, :report_date_end)
        into :ob_account, :ob_currency, :ob_end, :ob_end_ekv
    do
    begin
      ob_end_dt = 0;
      ob_end_kt = 0;
      ob_end_dt_ekv = 0;
      ob_end_kt_ekv = 0;
      select al_from_bg_id, bg_name, al_name
      from acc_list, balgroup
      where al_id = :ob_account and
            al_from_bg_id = bg_id
      into :ob_balance, :ob_balance_name, :ob_account_name;
      if (ob_end >= 0) then
        ob_end_dt = ob_end;
      else
        ob_end_kt = ob_end;

      if (ob_end_ekv >= 0) then
        ob_end_dt_ekv = ob_end_ekv;
      else
        ob_end_kt_ekv = ob_end_ekv;

      if (ob_currency = curr_currency) then
      begin
        ob_end_dt = 0;
        ob_end_kt = 0;
        ob_oborot_dt = 0;
        ob_oborot_kt = 0;
      end
      if (report_with_currency = 0) then
      begin
        ob_currency = '000';
