CREATE PROCEDURE FIND_IN_BLACK_LIST(
    FIND_C_ID DECIMAL(18,0),
    FIND_FIO VARCHAR(100),
    FIND_IN_MORATORY INTEGER,
    FIND_IN_REJECTED INTEGER,
    FIND_IN_BLACKLIST INTEGER)
RETURNS (
    M_ID DECIMAL(18,0),
    M_NUMBER INTEGER,
    M_FROM_C_ID DECIMAL(18,0),
    M_LAST_K_ID DECIMAL(18,0),
    M_IS_BLACK INTEGER,
    M_DATE_BEGIN DATE,
    M_DATE_END DATE,
    M_MONTH INTEGER,
    M_IS_ACTIVE SMALLINT,
    M_COMMENT_GROUP VARCHAR(2048),
    M_COMMENT_OFFICER VARCHAR(2048),
    M_COMMENT_MANAGER VARCHAR(2048),
    M_CLIENT_FIO VARCHAR(100),
    M_CLIENT_PASS_NO VARCHAR(30),
    M_CLIENT_ADDRESS VARCHAR(255),
    M_CLIENT_BIRTH_DATE DATE,
    M_OUR_CLIENTS INTEGER,
    M_FROM_REG_ID INTEGER,
    M_NO INTEGER,
    M_DATETIME TIMESTAMP,
    M_DATETIME_EDIT TIMESTAMP,
    M_USER_CREATE DECIMAL(18,0),
    M_USER_EDIT DECIMAL(18,0),
    M_DB_CODE VARCHAR(3),
    M_ORIGINAL_ID DECIMAL(18,0),
    M_FROM_FL_ID DECIMAL(18,0),
    M_FROM_MA_ID INTEGER,
    M_STATUS_STR VARCHAR(100),
    C_COMMENT_ID INTEGER)
AS
begin
  if (find_c_id is null) then
    find_c_id = 0;
  m_from_ma_id = null;

  for select m_id, m_number, c_id, c_name_fio, c_comment_id
      from hp_blacklist_search(:find_c_id, :find_fio, null, null, null, null, :find_in_blacklist, :find_in_rejected, :find_in_moratory)
      into m_id, m_number, m_from_c_id, :find_fio, c_comment_id
  do
  begin
    for select m_from_c_id, m_last_k_id, m_is_black, m_date_begin, m_date_end, m_month, m_is_active, m_comment_group,
               m_comment_officer, m_comment_manager, m_client_fio, m_client_pass_no, m_client_address,
               m_client_birth_date, m_our_clients, m_from_reg_id, m_no, m_datetime, m_datetime_edit, m_user_create,
               m_user_edit, m_db_code, m_original_id, m_from_fl_id
        from moratory
        where m_id = :m_id and
              m_number = :m_number
        into :m_from_c_id, :m_last_k_id, :m_is_black, :m_date_begin, :m_date_end, :m_month, :m_is_active,
             :m_comment_group, :m_comment_officer, :m_comment_manager, :m_client_fio, :m_client_pass_no,
             :m_client_address, :m_client_birth_date, :m_our_clients, :m_from_reg_id, :m_no, :m_datetime,
             :m_datetime_edit, :m_user_create, :m_user_edit, :m_db_code, :m_original_id, :m_from_fl_id
    do
    begin
      if (m_is_black = 0) then
      begin
        if (exists(select *
                   from mormotive, morstatus
                   where mm_from_m_id = :m_id and
                         mm_number = :m_number and
                         mm_from_ms_id = ms_id and
                         ms_is_force = 1)) then
          m_is_black = 2;
      end
      if (m_is_black = 0) then
        m_status_str = 'Мораторий';
      if (m_is_black = 1) then
        m_status_str = 'Ранее отказано';
      if (m_is_black = 2) then
        m_status_str = 'Стоп лист';

      if ((find_in_moratory = 1 and
          m_is_black = 0) or (find_in_rejected = 1 and
          m_is_black = 1) or (find_in_blacklist = 1 and
          m_is_black = 2)) then
      begin
        suspend;
      end
    end
  end
end
