CREATE PROCEDURE AAREP_REPORT_FOR_TSB_1(
    REP_DATE_END DATE)
RETURNS (
    MEMO_RESIDENT VARCHAR(255),
    CLI_BALANCE_DO_28022012 CURR_TYPE,
    CLI_PERC_SALDO_DO_28022012 CURR_TYPE,
    CLI_BALANCE_POSLE_28022012 CURR_TYPE,
    CLI_PERC_SALDO_POSLE_28022012 CURR_TYPE,
    CLI_KOMMIS_SALDO_DO_28022012 CURR_TYPE,
    CLI_KOMMIS_SALDO_POSLE_28022012 CURR_TYPE,
    CLI_RPPU_BAL_DO_28022012 CURR_TYPE,
    CLI_RPPU_BAL_POSLE_28022012 CURR_TYPE,
    CLI_RPPU_PERC_POSLE_28022012 CURR_TYPE,
    MIN_PERCENT_YEAR CURR_TYPE,
    MAX_PERCENT_YEAR CURR_TYPE,
    MIN_PERCENT_EFFECT CURR_TYPE,
    MAX_PERCENT_EFFECT CURR_TYPE,
    MIN_K_PERIOD_DAY INTEGER,
    MAX_K_PERIOD_DAY INTEGER)
AS
declare variable res_client_is_resident smallint;
declare variable loan_kind smallint;
declare variable loan_kind_memo varchar(50);
begin

  for select loan_kind, res_client_is_resident, sum(iif(k_disb_date < '28.02.2012', cli_balance, 0)),
             sum(iif(k_disb_date < '28.02.2012', cli_perc_saldo, 0)),
             sum(iif(k_disb_date >= '28.02.2012', cli_balance, 0)),
             sum(iif(k_disb_date >= '28.02.2012', cli_perc_saldo, 0)),
             sum(iif(k_disb_date < '28.02.2012', cli_kommis_saldo, 0)),
             sum(iif(k_disb_date >= '28.02.2012', cli_kommis_saldo, 0)), sum(cli_rppu_bal_do_28022012),
             sum(cli_rppu_bal_posle_28022012), sum(cli_rppu_perc_posle_28022012), min(percent_year) min_percent_year,
             max(percent_year) max_percent_year, min(percent_effect) min_percent_effect,
             max(percent_effect) max_percent_effect, min(k_period_day) min_k_period_day,
             max(k_period_day) max_k_period_day
      from aarep_report_for_tsb_1_tmp(:rep_date_end)
      group by 1, 2
      order by 1, 2 desc
      into :loan_kind, :res_client_is_resident, :cli_balance_do_28022012, :cli_perc_saldo_do_28022012,
           :cli_balance_posle_28022012, :cli_perc_saldo_posle_28022012, :cli_kommis_saldo_do_28022012,
           :cli_kommis_saldo_posle_28022012, :cli_rppu_bal_do_28022012, :cli_rppu_bal_posle_28022012,
           :cli_rppu_perc_posle_28022012, :min_percent_year, :max_percent_year, :min_percent_effect,
           :max_percent_effect, :min_k_period_day, :max_k_period_day
  do
  begin

    loan_kind_memo = case loan_kind
                       when 1 then 'сотрудникам'
                       when 2 then 'будущим сотрудникам'
                       when 3 then 'физическим лицам'
                     end;

--    memo_resident = iif(res_client_is_resident = 1, 'микрозаймы, выданные физическим лицам', 'микрозаймы, выданные физическим лицам – нерезидентам');
    memo_resident = iif(res_client_is_resident = 1, 'микрозаймы, выданные ' || :loan_kind_memo, 'микрозаймы, выданные ' || :loan_kind_memo || ' – нерезидентам');

    suspend;
  end
end
