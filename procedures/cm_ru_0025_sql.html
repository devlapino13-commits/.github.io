CREATE PROCEDURE CM_RU_0025_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kol_day integer;
declare variable disb_date date;
declare variable first_repay_date date;
declare variable min_days integer;
declare variable max_days integer;
begin
  --От даты выдачи до первого погашения не должно быть меньше min_days дней и не больше max_days дней.
  min_days = 15;
  max_days = 45;
  kol_day = null;
  select min(sh_date),
         max(sh_date)
  from schedule
  where sh_from_k_id = :k_id
  and sh_id in (0,1)
  into disb_date, first_repay_date;
  kol_day = first_repay_date - disb_date;
  if (kol_day is null) then
  begin
    res_error_text = 'График погашения не построен.';
    res_error_is_critical = 2;
    suspend;
    Exit;
  end
   if (exists
          (select *
          from kontract
          where k_id = :k_id
          and (k_add_month>0 or k_refund>0))) then exit;

  if (not(kol_day between min_days and max_days)) then
  begin
    -- Если платежи попадают между 9-12 января или 1-11 мая, НЕ выводим сообщение о несоответствии по максимальным дням между платежами
    if (not((extract(month from first_repay_date) = 1 and extract(day from first_repay_date) between 9 and 12) or
        (extract(month from first_repay_date) = 5 and extract(day from first_repay_date) between 1 and 11)))  then
    begin
      res_error_text = 'От даты выдачи до первого погашения должно быть не меньше ' || min_days || ' дней, и не больше ' || max_days || ' дней';
      res_error_is_critical = 1;
      suspend;
    end
  end
end
