CREATE PROCEDURE MC_OFFICE_INFO(
    MCL_ID D_ID)
RETURNS (
    OFFICE_CITY_ID D_ID,
    REG_ID D_INTEGER,
    RES_TEXT D_STR_LONG,
    RES_CODE D_SMALLINT)
AS
declare variable C_ID ID_TYPE;
declare variable SVZ_ID ID_TYPE;
declare variable SVZ_FIO STR_100;
declare variable OFFICE_NAME STR_100;
declare variable ADDRESS STR_255;
declare variable K_ID decimal(18,0);
declare variable LANG_CODE STR_10;
begin

  mcl_id = coalesce(mcl_id, 0);
  res_text = null;
  reg_id = null;
  res_code = 0;

  select mcl_from_c_id, mcl.mcl_lang_code
  from mc_login mcl
  where mcl_id = :mcl_id
  into :c_id, lang_code;

  if (c_id is not null) then
  begin
    select k_id, kal_new_kf_officer, c_name_fio, kal_new_reg_id
    from knclient
    inner join kontract on k_id = kc_from_k_id
    inner join knaktlist on kal_from_kf_id = k_from_kf_id
    inner join clients on c_id = kal_new_kf_officer
    where kc_from_c_id = :c_id and
          kal_date_end = '31.12.9999'
    order by kc_id desc
    rows 1
    into k_id, svz_id, svz_fio, reg_id;

    if (k_id is not null) then
    begin
      select office_city_id, office_name, address
      from mc_office_list
      where office_id = :reg_id
      into office_city_id, office_name, address;

      if (lang_code = 'kg') then
      begin
        res_text = 'Сиз номер ' || substring(reg_id from 2 for 2) || '-' || substring(reg_id from 4 for 2) || ', "' || office_name || '" офисинде катталгансыз';

        if ((select ps_is_working
             from hr_pstatus
             where ps_from_c_id = :svz_id and
                   ps_date_end = '31.12.9999') = 1) then
          res_text = res_text || ', сиздин адис ' || svz_fio || ', байланыш номери: ' || f_cont_info('clients', svz_id, 'cell_main', 'ok') || '. Бирок сиз башка кеңсе тандай аласыз.';
      end
      if (lang_code = 'ru') then
      begin
        res_text = 'Вы обслуживаетесь в офисе №' || substring(reg_id from 2 for 2) || '-' || substring(reg_id from 4 for 2) || ', "' || office_name;

        if ((select ps_is_working
             from hr_pstatus
             where ps_from_c_id = :svz_id and
                   ps_date_end = '31.12.9999') = 1) then
          res_text = res_text || ', у специалиста ' || svz_fio || ', тел.: ' || f_cont_info('clients', svz_id, 'cell_main', 'ok') || '. Но Вы можете выбрать другой офис.';

      end
      if (lang_code = 'az') then
      begin
        res_text = 'Sizə xidmət göstərən ofis №' || substring(reg_id from 2 for 2) || '-' || substring(reg_id from 4 for 2) || ', "' || office_name;

        if ((select ps_is_working
             from hr_pstatus
             where ps_from_c_id = :svz_id and
                   ps_date_end = '31.12.9999') = 1) then
          res_text = res_text || ', mütəxəssis ' || svz_fio || ', tel.: ' || f_cont_info('clients', svz_id, 'cell_main', 'ok') || '. Lakin siz başqa ofis seçə bilərsiniz, bunun üçün «ofisi dəyişmək» düyməsini sıxın və başqa ofisi seçin.';
      end
    end
  end
  suspend;
end
