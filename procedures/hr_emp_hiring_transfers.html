CREATE PROCEDURE HR_EMP_HIRING_TRANSFERS(
    ORDER_ID DECIMAL(18,0))
RETURNS (
    PS_DATE_BEGIN DATE,
    HO_ORDER_NO VARCHAR(100),
    HO_REASON VARCHAR(500),
    JOB_NAME VARCHAR(300),
    REG_NAME VARCHAR(300))
AS
begin
  for
    select
        ps.ps_date_begin,
        ho.ho_order_no,
        ho.ho_reason,
        jl.jl_name as job_name,
        r.reg_name
    from hr_pstatus ps_main
    inner join hr_pstatus ps on ps.ps_from_c_id = ps_main.ps_from_c_id
    inner join hr_orders ho on ps.ps_from_ho_id = ho.ho_id
    inner join sa_worktype swt on ho.ho_order_type = swt.sawt_id
    inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
    inner join region r on ps.ps_from_reg_id = r.reg_unicode
    where ps_main.ps_from_ho_id = :order_id
    and ps.ps_status = 1
    and ho.ho_status = 1
    and ps.ps_date_begin < ps.ps_date_end
    and swt.sawt_id <> 6
    order by ps.ps_date_begin
    into
        :ps_date_begin,
        :ho_order_no,
        :ho_reason,
        :job_name,
        :reg_name
  do
    suspend;
end
