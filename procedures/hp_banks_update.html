CREATE PROCEDURE HP_BANKS_UPDATE(
    STATUS D_SMALLINT,
    JSON_BLOB D_BLOB_TEXT,
    ERROR_TEXT D_STR_LONG)
AS
declare variable bul_error_id d_id;
declare variable jso d_integer;
declare variable jsa d_integer;
declare variable base_path d_str_255;
declare variable arr_length d_integer;
declare variable acc_arr_length d_integer;
declare variable i d_integer;
declare variable j d_integer;
declare variable bic d_str_10;
declare variable adress_str d_str_255;
declare variable account d_str_255;
declare variable account_date_in d_date;
declare variable v_account_date_in d_date;
declare variable name d_str_255;
declare variable regulation_account_type d_str_20;
declare variable account_status d_str_20;
begin
  -- RFCRU-5254 Интеграция с ЦБ, справочник БИК. 
  if (f_get_db_id() <> 81001) then
    exit;

  base_path = 'data';

  if (status = 1) then
  begin
     -- Делаем удаление всех записей в таблице banks
     update banks b
        set b.is_delete = 1;

    -- если status = 1, то обрабатываем json (подробно в ТЗ)  json_blob - обязательно
     i = 0;
     jso = json.parse(json_blob);
     if (jso is not null) then
       jsa = json.obj_get_arr(jso, base_path);

     if (jsa is not null) then
       arr_length = json.count_get(jsa);
     else
       arr_length = -1;

     while (i < arr_length) do
     begin
       j = 0;
       -- BIC – БИК
       bic = json.obj_get_val(jso, f_format('%1[%2].BIC', base_path, i), '');

       -- name – Наименование
       name = json.obj_get_val(jso, f_format('%1[%2].ParticipantInfo.NameP', base_path, i), '');

       -- address – Адрес
       adress_str = f_format('%1, %2, %3',
                             json.obj_get_val(jso, f_format('%1[%2].ParticipantInfo.Ind', base_path, i), ''),
                             json.obj_get_val(jso, f_format('%1[%2].ParticipantInfo.Nnp', base_path, i), ''),
                             json.obj_get_val(jso, f_format('%1[%2].ParticipantInfo.Adr', base_path, i), ''));

       -- acc – Кор.счет (если нет – поле не передавать)--------------------------Начало
       jsa = json.obj_get_arr(jso, f_format('%1[%2].Accounts', base_path, i));
       if (jsa is not null) then
         acc_arr_length = json.count_get(jsa);
       else
         acc_arr_length = -1;

       account = null;
       account_date_in = null;
       v_account_date_in = null;

       while (j < acc_arr_length) do
       begin
         regulation_account_type = null;
         account_status = null;

         v_account_date_in = cast(json.obj_get_val(jso, f_format('%1[%2].Accounts[%3].DateIn', base_path, i, j), null) as d_date);

         regulation_account_type = json.obj_get_val(jso, f_format('%1[%2].Accounts[%3].RegulationAccountType', base_path, i, j), '');
         account_status = json.obj_get_val(jso, f_format('%1[%2].Accounts[%3].AccountStatus', base_path, i, j), '');

         if (regulation_account_type = 'CRSA' and           -- берем аккаунты где RegulationAccountType="CRSA" (тип корреспондентский счет)
             account_status = 'ACAC' and                     -- и AccountStatus="ACAC" (действующий)
             (account_date_in is null or                    -- Если несколько, то смотрим по дате, при прочих равных берем первый.
              account_date_in < v_account_date_in)) then
         begin
           account = json.obj_get_val(jso, f_format('%1[%2].Accounts[%3].Account', base_path, i, j), '');
           account_date_in = v_account_date_in;
         end

         j = j + 1;
       end
       -- acc – Кор.счет (если нет – поле не передавать)--------------------------_Конец

       update or insert into banks (b_id, b_name, b_address, is_delete, b_corr_acc, b_date_change)
       values (:bic, :name, :adress_str, 0, :account, current_date)
       matching (b_id);
       i = i + 1;
     end
  end 
  else if (status = -1) then
    bul_error_id = f_memo_id(error_text);
  else
    exception e_default 'Unknown status = ' || coalesce(status, '<NULL>');


  -- обязательно записываем факт обновления в  banks_update_log
  insert into banks_update_log (bul_status, bul_datetime, bul_error_id)
  values (:status, cast(current_timestamp as d_datetime), :bul_error_id);
end
