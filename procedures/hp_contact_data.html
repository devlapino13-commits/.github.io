CREATE PROCEDURE HP_CONTACT_DATA(
    OBJECT_ID ID_TYPE,
    OBJECT_KIND INTEGER)
RETURNS (
    CC_ID INTEGER,
    CC_FROM_C_ID DECIMAL(15,2),
    CC_NAME STR_100,
    CC_STATUS STR_100,
    CC_ADDRESS STR_255,
    CC_CELL_NO STR_255,
    CC_CHECK_RESULT INTEGER,
    CC_IN_PROCESS INTEGER,
    CC_TRY_KOL INTEGER,
    CC_CALL_USER STR_255,
    CC_TIME_BEGIN TIMESTAMP,
    CC_TIME_END TIMESTAMP,
    CC_FROM_UN_ID DECIMAL(15,2),
    CC_CHECK_MEMO STR_255,
    CC_TYPE INTEGER,
    CC_FROM_ADL_ID DECIMAL(15,2),
    CC_ADL_ADRESS STR_255,
    CC_LANGUAGE INTEGER,
    CC_BIRTH_DATE DATE,
    CC_IS_OLD_CONTACT SMALLINT,
    CC_CELL_NO_ADD STR_100,
    CC_SVZ_COMMENTS STR_255,
    CC_VERIF_IGNORE SMALLINT,
    CC_TOTAL_RES INTEGER,
    C_NAME_FIO STR_255,
    CC_BLOCKED_USER STR_100)
AS
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  for select cont.cc_id, cont.cc_from_c_id, cont.cc_name, cont.cc_status, cont.cc_address, cont.cc_cell_no,
             cont.cc_check_result, cont.cc_in_process, cont.cc_try_kol, cont.cc_call_user, cont.cc_time_begin,
             cont.cc_time_end, cont.cc_from_un_id, cont.cc_check_memo, cont.cc_type, cont.cc_from_adl_id,
             cont.cc_language, cont.cc_birth_date, cont.cc_is_old_contact, cont.cc_cell_no_add, cont.cc_svz_comments,
             cont.cc_verif_ignore, cont.cc_total_res, cc_blocked_user
      from contacts cont
      where ((cc_from_c_id = :object_id) and
            (:object_kind = 0) and (cont.cc_is_delete = 0)) or
            ((cc_id = :object_id) and
            (:object_kind = 13))
      order by cc_is_old_contact, cc_id
      into :cc_id, :cc_from_c_id, :cc_name, :cc_status, :cc_address, :cc_cell_no, :cc_check_result, :cc_in_process,
           :cc_try_kol, :cc_call_user, :cc_time_begin, :cc_time_end, :cc_from_un_id, :cc_check_memo, :cc_type,
           :cc_from_adl_id, :cc_language, :cc_birth_date, :cc_is_old_contact, :cc_cell_no_add, :cc_svz_comments,
           :cc_verif_ignore, :cc_total_res, :cc_blocked_user
  do
  begin
    cc_adl_adress = '';

    select adl_address_str
    from addrlist adl
    where adl.adl_id = :cc_from_adl_id
    into :cc_adl_adress;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :cc_from_c_id
    into :c_name_fio;

--    if (db_id = 81001) then  -- Скрываем для РФ базы
--    begin
--      cc_name = '';
--    end
    suspend;
  end
end
