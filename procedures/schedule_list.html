CREATE PROCEDURE SCHEDULE_LIST(
    K_ID DECIMAL(18,0),
    SUMMA DECIMAL(18,4))
RETURNS (
    SH_FROM_K_ID DECIMAL(18,0),
    SH_ID INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_KOMMIS DECIMAL(18,4),
    SH_ALL DECIMAL(18,4),
    SH_PERC_TAX DECIMAL(18,4))
AS
declare variable rep_kol integer;
declare variable sum_os decimal(18,4);
declare variable total_amount decimal(18,4);
begin
  sum_os = 0;
  for select k_amount_approved, k_kolich_repay, sh_from_k_id, sh_id, sh_date, sh_os, sh_perc, sh_kommis, sh_perc_tax
      from schedule, kontract
      where k_id = :k_id and
            sh_from_k_id = :k_id
      order by sh_id, sh_date
      into :total_amount, :rep_kol, :sh_from_k_id, :sh_id, :sh_date, :sh_os, :sh_perc, :sh_kommis, :sh_perc_tax
  do
  begin
    if (total_amount <> summa) then
    begin
      if (total_amount <> 0) then
      begin
        sh_os = sh_os / total_amount * summa;
        sh_os = cast(sh_os as decimal(18,2));
        sh_perc = sh_perc / total_amount * summa;
        sh_perc = cast(sh_perc as decimal(18,2));
        sh_kommis = sh_kommis / total_amount * summa;
        sh_kommis = cast(sh_kommis as decimal(18,2));
        sh_perc_tax = cast(sh_perc_tax as decimal(18,2));
        if (sh_id = rep_kol) then
        begin
          sh_os = summa - sum_os;
        end
        else
        begin
          sum_os = sum_os + sh_os;
        end
      end
    end
    sh_all = sh_os + sh_perc + sh_kommis + sh_perc_tax;
    suspend;
  end
end
