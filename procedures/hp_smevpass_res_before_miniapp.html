CREATE PROCEDURE HP_SMEVPASS_RES_BEFORE_MINIAPP(
    C_ID D_ID)
RETURNS (
    ERROR_TEXT D_STR_255)
AS
declare variable BLS_ID D_ID;
declare variable JSON_FULL_STR D_STR_8100;
declare variable EIL_STATUS D_INTEGER;
declare variable JSO D_INTEGER;
declare variable STATUS_DESC D_STR_255;
declare variable STATUS_CODE D_INTEGER;
declare variable INVALIDITY_REASON_DESC D_STR_255;
declare variable INVALIDITY_REASON_CODE D_INTEGER;
declare variable INVALIDITY_SINCE D_DATE;
declare variable CDS_ID D_ID;
begin
  error_text = '';

  select cds_id
  from clientdocs cds
  where cds.cds_from_c_id = :c_id
    and cds_is_main_doc = 1
    and cds_is_delete = 0
    and cds_verify_status = 2
    and cds_index_doc_no <> ''
  order by cds_id desc rows 1
  into cds_id;

  if (row_count = 0) then
  begin
    error_text = 'Нет верифицированного паспорта';
    suspend;
    exit;
  end

  select eil_from_bls_id, eil_status
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eil.eil_from_c_id = :c_id and
        eic.eic_code = 'passport_smev' and
        eil.eil_object_kind = f_objkind('clientdocs', 'id') and
        eil.eil_object_id = :cds_id
  order by eil.eil_status desc, eil.eil_date desc, eil.eil_id desc
  rows 1
  into bls_id, eil_status;

  if (row_count = 0) then
  begin
    error_text = 'Требуется проверка паспорта через СМЭВ';
    suspend;
    exit;
  end

  if (eil_status < 0) then
  begin
    error_text = 'Ошибка проверки паспорта по СМЭВ';
    suspend;
    exit;
  end

  if (eil_status < 2) then
  begin
    error_text = 'Дождитесь результата проверки паспорта через СМЭВ';
    suspend;
    exit;
  end

  select cast(blx_blob as d_blob_text)
  from hp_get_blobex_blob(:bls_id)
  into :json_full_str;

  if (json_full_str is null) then
  begin
    error_text = 'Ошибка проверки паспорта по СМЭВ: json пустой';
    suspend;
    exit;
  end

  -- разбор json
  jso = json.parse(json_full_str);

  status_desc = json.obj_get_val(jso, 'status_desc');
  status_code = json.obj_get_val(jso, 'status_code');
  invalidity_reason_desc = json.obj_get_val(jso, 'invalidity_reason_desc');

  if (status_code <> 300) then
  begin
    error_text = :status_desc || ': ' || :invalidity_reason_desc;
  end 

  suspend;
end

  /*
«Требуется проверка паспорта через СМЭВ» - это если паспорт еще не отправлен на проверку
«Дождитесь результата проверка паспорта через СМЭВ» - если проверка начата, но не закончена
«Результаты проверки через СМЭВ: паспорт недействителен» - если паспорт проверен и действительность не подтверждена
«Результаты проверки через СМЭВ: паспорт не найден» - если паспорт проверен и не найден
