      inner join refdata d5 on c4.c_army = d5.rd_id
      inner join bustype on bt_id = knclient.kc_business_type
      inner join bussubtype on bs_id = knclient.kc_busuness_subtype
      into :k_id, :k_from_kf_id, :k_currency, :k_applicate_date, :k_applicate_no, :k_kontract_no, :k_kontract_date,
           :k_from_pr_id, :k_disb_date, :k_amount_applicate, :k_stage, :k_period, :k_kolich_lgota, :k_percent,
           :k_percent_lgota, :k_kolich_repay, :k_from_f_id, :k_from_pt_id, :k_circle, :k_circle_internal,
           :k_percent_per_day, :k_memo, :k_amount_approved, :k_percent_for_days, :k_kolich_client_applicate,
           :k_kolich_client_approved, :k_repay_kolich_closed, :k_risk_category, :k_kommis_perc, :k_from_cs_id,
           :k_credit_target, :k_is_kredline, :k_kredline_end, :kf_name, :kf_number, :kf_region_id, :kf_region_name,
           :kf_branch_id, :kf_branch_name, :kf_officer_id, :kf_officer_code, :kf_officer_name, :kf_date_in_to_krs,
           :kf_date_out_from_krs, :kf_date_create, :kf_memo, :rep_active_balance, :rep_date_closed, :rep_deliq_os,
           :rep_deliq_perc, :rep_deliq_days_kolich, :rep_deliq_repays_str, :rep_deliq_begin_date, :rep_date_action,
           :rep_plan_os, :rep_plan_perc, :rep_plan_all, :rep_plan_repay_no, :rep_plan_repay_date,

           :cl_leader_id, :cl_leader_name, :cl_leader_pass_no, :cl_leader_pass_date, :cl_leader_pass_who,
           :cl_cashier_id, :cl_cashier_name, :cl_cashier_pass_no, :cl_cashier_pass_date, :cl_cashier_pass_who, :pr_code,
           :pr_name, :pr_loan_type, :pr_is_employer, :pr_is_secured, :pr_int_calc_type, :pr_int_for_days,
           :llc_begin_status_str, :llc_beg_class, :llc_current_class, :llc_begin_reserve, :perc_gr, :perc_ind,
           :llc_current_status_str, :llc_current_reserve, :llc_current_percent, :knc_id, :c_type, :knc_name_fio, :c_sex,
           :knc_birth_date, :knc_pass_no, :knc_pass_date, :knc_pass_who, :knc_pass_date_expire, :knc_from_reg_id,
           :knc_kol_family, :knc_kol_child, :knc_memo, :knc_reg_no, :knc_okpo, :knc_inn, :knc_account,
           :knc_org_from_b_id, :knc_hired_date, :knc_is_resident, :knc_others, :knc_doc_type, :knc_str_nation,
           :knc_str_citizen, :knc_str_family, :knc_str_education, :knc_str_army,

           :knc_number, :knc_amount_applicate, :knc_from_pt_id, :knc_business_type, :kc_is_new_business,
           :kc_is_registered, :knc_business_name, :knc_monthly_oborot, :knc_kapital, :knc_month_profit,
           :knc_amount_approved, :curr_rate, :knc_work_kolich, :knc_work_kolich_family, :kc_is_business_full_time,
           :knc_business_subtype, :bs_kind_fir, :kc_is_new_added, :kc_status, :knc_circle, :k_percent_input

  do
  begin
    cl_leader_full_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_address_type = 1 and
          adc_from_c_id = :cl_leader_id and
          adc_from_adl_id = adl_id
    into :cl_leader_full_address;

    cl_cashier_full_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_address_type = 1 and
          adc_from_c_id = :cl_cashier_id and
          adc_from_adl_id = adl_id
    into :cl_cashier_full_address;

    knc_full_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_address_type = 1 and
          adc_from_c_id = :knc_id and
          adc_from_adl_id = adl_id
    into :knc_full_address;

    knc_email = '';
    knc_plastcard_no = '';
    if (coalesce(knc_id, 0) <> 0) then
    begin
      select cont_value
      from hp_contdata_get_cell_main(0, :knc_id)
      into :knc_plastcard_no;

      select cont_value
      from hp_contdata_get_email_main(0, :knc_id)
      into :knc_email;
    end

    if (pr_loan_type = 0) then
      llc_begin_percent = :perc_gr;
    else
      llc_begin_percent = :perc_ind;
    if (report_currency = '000') then
    begin
      knc_amount_approved = :knc_amount_approved * :curr_rate;
      k_amount_approved = :k_amount_approved * :curr_rate;
    end
    if (report_id = 2) then
    begin
      k_disb_date = rep_date_action;
      k_amount_approved = rep_plan_os;
      rep_plan_os = 0;
    end
    if (report_id = 3) then
    begin
      if (k_percent_input = :k_percent) then
      begin
        select result_perc
        from aarep_float_to_annuitet(:k_percent, :k_kolich_repay)
        into :k_percent_ostatok;
      end
      else
      begin
        k_percent_ostatok = cast(:k_percent as decimal(15,4));
      end
    end

    k_disb_year = extract(year from k_disb_date);
    k_disb_month = extract(month from k_disb_date);

    if (k_is_kredline = 0) then
      k_is_kredline_str = 'Нет';
    if (k_is_kredline = 1) then
      k_is_kredline_str = 'Да';

    pr_loan_type_str = '';
    if (pr_loan_type = 0) then
      pr_loan_type_str = 'Группа';
    if (pr_loan_type = 1) then
      pr_loan_type_str = 'Индив.';

    pr_is_employer_str = '';
    if (pr_is_employer = 0) then
      pr_is_employer_str = 'Для клиент.';
    if (pr_is_employer = 1) then
      pr_is_employer_str = 'Для сотрудн.';

    pr_is_secured_str = '';
    if (pr_is_secured = 0) then
      pr_is_secured_str = 'Без залога';
    if (pr_is_secured = 1) then
      pr_is_secured_str = 'Залог';
    if (pr_is_secured = 2) then
      pr_is_secured_str = 'Поручители';
    if (pr_is_secured = 3) then
      pr_is_secured_str = 'Залог,поруч.';

    pr_int_calc_type_str = '';
    if (pr_int_calc_type = 0) then
      pr_int_calc_type_str = 'На сумму кредита';
    if (pr_int_calc_type = 1) then
      pr_int_calc_type_str = 'На остаток';
    if (pr_int_calc_type = 2) then
      pr_int_calc_type_str = 'Аннуитет';

    if (llc_current_status_str is null) then
      llc_current_status_str = llc_begin_status_str;
    if (llc_current_reserve is null) then
      llc_current_reserve = llc_begin_reserve;
    if (llc_current_percent is null) then
      llc_current_percent = llc_begin_percent;
    if (llc_current_class is null) then
      llc_current_class = llc_beg_class;

    llc_current_amount = (:rep_active_balance * :llc_current_percent) / 100;

    k_circle_real = :k_circle + :k_circle_internal - 1;
    k_percent_real = :k_percent - :k_percent_lgota;

    if (c_type = 0) then
      knc_type_str = 'Физ.лицо';
    else
      knc_type_str = 'Юр.лицо';

    if (c_sex = 0) then
      knc_sex_str = 'Женщина';
    if (c_sex = 1) then
      knc_sex_str = 'Мужчина';
    if (c_sex = 2) then
      knc_sex_str = 'Юр.лицо';

    if (knc_birth_date is not null) then
    begin
      knc_birth_year = extract(year from knc_birth_date);
      knc_birth_month = extract(month from knc_birth_date);
      knc_birth_day = extract(day from knc_birth_date);
      knc_age_year = datediff(month, knc_birth_date, :report_date_end) / 12;
    end
    else
    begin
      knc_birth_year = 0;
      knc_birth_month = 0;
      knc_birth_day = 0;
      knc_age_year = 0;
    end

    if (knc_pass_date_expire is not null) then
    begin
      if (knc_pass_date_expire < :report_date_end) then
        knc_pass_date_expired_str = 'Просрочен';
      else
      begin
        if (datediff(day, :report_date_end, knc_pass_date_expire) <= 30) then
          knc_pass_date_expired_str = 'Истекает';
        else
          knc_pass_date_expired_str = 'Действителен';
      end
    end
    else
      knc_pass_date_expired_str = 'Не указан';

    if (kc_status = 0) then
      knc_status_str = 'Участник';
    if (kc_status = 1) then
      knc_status_str = 'Лидер';
    if (kc_status = 2) then
      knc_status_str = 'Казначей';

    if (kc_is_new_business = 1) then
      knc_is_new_buss_str = 'Да';
    else
      knc_is_new_buss_str = 'Нет';

    if (kc_is_registered = 1) then
      knc_is_regist_str = 'Да';
    else
      knc_is_regist_str = 'Нет';

    if (kc_is_business_full_time = 1) then
      knc_is_full_time_str = 'Да';
    else
      knc_is_full_time_str = 'Нет';

    if (kc_is_new_added = 1) then
      knc_is_new_add_str = 'Да';
    else
      knc_is_new_add_str = 'Нет';
    knc_rep_llr_cl_amount = 0;
    if (knc_amount_approved <> 0) then
    begin
      /* aschegolskiy, 03.08.12 Вычисление knc_rep_active приводило к ошибке Integer overflow, при сумме кредита в 9.4 млн.*/
      /* Информация об ошибке: http://www.firebirdfaq.org/faq207/ */
      /* knc_rep_active_balance=(rep_active_balance/k_amount_Approved)*knc_amount_Approved;*/
      knc_rep_active_balance = (cast(rep_active_balance as double precision) / k_amount_approved) * knc_amount_approved;
      knc_rep_llr_cl_amount = (knc_rep_active_balance * :llc_current_percent) / 100;
      knc_rep_deliq_os = (rep_deliq_os / k_amount_approved) * knc_amount_approved;
      knc_rep_deliq_perc = (rep_deliq_perc / k_amount_approved) * knc_amount_approved;
      knc_rep_plan_os = (rep_plan_os / k_amount_approved) * knc_amount_approved;
      knc_rep_plan_perc = (rep_plan_perc / k_amount_approved) * knc_amount_approved;
      knc_rep_active_balance = cast(knc_rep_active_balance as decimal(18,2));
      knc_rep_deliq_os = cast(knc_rep_deliq_os as decimal(18,2));
      knc_rep_deliq_perc = cast(knc_rep_deliq_perc as decimal(18,2));
      knc_rep_plan_os = cast(knc_rep_plan_os as decimal(18,2));
      knc_rep_plan_perc = cast(knc_rep_plan_perc as decimal(18,2));
      knc_rep_plan_all = knc_rep_plan_os + knc_rep_plan_perc;
    end
    else
    begin
      knc_rep_active_balance = 0;
      knc_rep_deliq_os = 0;
      knc_rep_deliq_perc = 0;
      knc_rep_plan_os = 0;
      knc_rep_plan_perc = 0;
      knc_rep_active_balance = 0;
      knc_rep_deliq_os = 0;
      knc_rep_deliq_perc = 0;
      knc_rep_plan_os = 0;
      knc_rep_plan_perc = 0;
      knc_rep_plan_all = 0;
    end

    if (rep_plan_repay_no = k_kolich_repay) then
      rep_plan_repay_is_last = 'Посл.плат.';
    else
      rep_plan_repay_is_last = '';

    if (knc_circle = 1) then
      knc_new_or_repeat = 'Новый';
    else
      knc_new_or_repeat = 'Повторный';

    if (bs_kind_fir = 1) then
      knc_business_kind = 'Сельское хозяйство / растениеводство';
    if (bs_kind_fir = 2) then
      knc_business_kind = 'Сельское хозяйство / животноводство';
    if (bs_kind_fir = 3) then
      knc_business_kind = 'Производство';
    if (bs_kind_fir = 4) then
      knc_business_kind = 'Торговля';
    if (bs_kind_fir = 5) then
      knc_business_kind = 'Приобретение техники и оборудования';
    if (bs_kind_fir = 6) then
      knc_business_kind = 'Ипотека';
    if (bs_kind_fir = 7) then
      knc_business_kind = 'Потребительские кредиты';
    if (bs_kind_fir = 8) then
      knc_business_kind = 'Образование / Здоровье / Непредвиденные случаи';
    if (bs_kind_fir = 9) then
      knc_business_kind = 'Сфера услуг';
    if (bs_kind_fir = 10) then
      knc_business_kind = 'Прочее';

    suspend;
  end

end
