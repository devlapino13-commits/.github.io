CREATE PROCEDURE AAREP_LLC_CLIENTS(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OFFCE_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(50),
    OKRUG_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(20),
    K_PERCENT D_PERCENT,
    LIR_RATE D_PERCENT,
    C_NAME_FIO VARCHAR(100),
    C_INN D_STR_20,
    LKC_DELIQ_DAYS INTEGER,
    LKC_BALANCE CURR_TYPE,
    LKC_STATUS SMALLINT,
    LKC_STATUS_STR VARCHAR(50),
    LKC_PERCENT PERC_TYPE,
    LKC_IS_INDIV SMALLINT,
    LKC_MEMO VARCHAR(255),
    LKC_HAS_LOANS SMALLINT,
    LKC_HAS_FAMILY_LOANS SMALLINT,
    LKC_HAS_MEMO VARCHAR(255),
    LKC_LLC_SUM CURR_TYPE,
    PR_NAME VARCHAR(60),
    LLC_RESERV_KIND VARCHAR(50),
    REG_STAT_REP_REGION VARCHAR(255),
    BT_NAME VARCHAR(50),
    BS_NAME VARCHAR(50),
    KC_AMOUNT_APPROVED CURR_TYPE,
    K_DISB_DATE DATE,
    LM_FROM_REG_ID INTEGER,
    LM_DATE_STATUS DATE,
    LKC_IS_SUPRUG_KRED SMALLINT,
    SUPRUG_NAME_FIO VARCHAR(100),
    LKC_SUPRUG_MEMO VARCHAR(255),
    LKC_IS_SUPRUG_HIS_KRED SMALLINT,
    IS_IPF INTEGER,
    TYPE_FINANCING VARCHAR(20),
    MARK D_STR_CODE,
    CR_RATING D_STR_CODE,
    DETAILED_PURPOSE_DESCRIPTION_1 D_STR_255,
    DETAILED_PURPOSE_DESCRIPTION_2 D_STR_LONG,
    INCOME_ACTIVITIES_LEVEL_1 D_STR_LONG,
    INCOME_ACTIVITIES_LEVEL_2 D_STR_LONG,
    INCOME_ACTIVITIES_LEVEL_3 D_STR_LONG,
    INCOME_ACTIVITIES_LEVEL_4 D_STR_LONG,
    CP_INCOME_PREFIX_RU D_STR_255,
    INCOME_ACTIVITIES_LAST D_STR_LONG,
    PR_LOAN_TYPE_STR D_STR_20,
    K_PERIOD D_CURRENCY,
    KNC_SEX_STR D_STR_20,
    KNC_TYPE_STR D_STR_10,
    RAZBIVKA_SUMM D_STR_CODE,
    RAZBIVKA_MONTH D_STR_CODE,
    RAZBIVKA_SROK D_STR_CODE,
    PRIZNAK_PROSROCHKI D_STR_20,
    K_KREDLINE_END D_DATE)
AS
declare variable LM_ID decimal(18,0);
declare variable LKC_FROM_C_ID ID_TYPE;
declare variable LKC_FROM_K_ID ID_TYPE;
declare variable REP_DATE_LAST date;
declare variable DD date;
declare variable LKC_SUPRUG_ID ID_TYPE;
declare variable DB_ID D_INTEGER;
declare variable MA_ID D_ID;
declare variable MAC_ID D_ID;
declare variable CA_ID D_INTEGER;
declare variable CQ_BASIC_INCOM D_SMALLINT;
declare variable CQ_NAME_RU D_STR_255;
declare variable CAA_VALUE D_INTEGER;
declare variable CA_FROM_CP_ID D_INTEGER;
declare variable TEMP_VALUE D_STR_LONG;
declare variable CP_NAME D_STR_255;
declare variable HIERARCHY_LEVEL D_INTEGER;
begin
  db_id = f_get_db_id();
  -- Если дата отчета не последний день месяца, то отчет будет готовиться на последний день предыдущего месяца
  rep_date_last = :rep_date_end - extract(day from :rep_date_end);
  dd = :rep_date_last + 33;
  dd = :dd - extract(day from :dd);
  if (:rep_date_end < :dd) then
    rep_date_end = rep_date_last;

  for select lm_id, lm_from_reg_id, lm_date_status
      from llc_main, region_list(:rep_region_id, 1, 3, 1)
      where lm_is_signed = 1 and
            lm_date_status between :rep_date_end - 4 and :rep_date_end + 4 and
            lm_from_reg_id = id_list
      into lm_id, lm_from_reg_id, lm_date_status
  do
  begin
    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name,
           r1.reg_stat_rep_region
    from region r1, region r2, region r3
    where r1.reg_unicode = :lm_from_reg_id and
          r1.parent = r2.reg_unicode and
          r2.parent = r3.reg_unicode
    into offce_name, branch_name, okrug_name, reg_stat_rep_region;

    for select k_kontract_no, k_percent, k_from_ma_app_id, c_name_fio, c_inn, lkc_deliq_days, lkc_balance, lkc_status, iif(cs_id = 1 and :rep_date_end >= '01.01.2024', 'Нормальные', cs_name) as lkc_status_str,
               cs_llr_kind_text as llc_reserv_kind, lkc_percent, lkc_is_indiv, lkc_memo, lkc_has_loans,
               lkc_has_family_loans, lkc_has_memo,
               cast(lkc_balance as double precision) * lkc_percent / 100 as lkc_llc_sum, pr_id || ' ' || pr_name,
               lkc_from_c_id, lkc_from_k_id, k_disb_date, lkc_is_suprug_kred, lkc_suprug_id, lkc_suprug_memo,
               lkc_is_suprug_his_kred, iif(pr.pr_code containing ('.IPF.'), 1, 0) as is_ipf,
               lkc_rate,
               iif(lkc_is_indiv = 1, iif(lkc_mark, 'специальная', 'общая'), ''),
               iif(coalesce(k_by_rating, 0) = 0, 'стандартная выдача', 'по кредитному рейтингу'),
               iif(pr_loan_type = 0, 'группа', 'индив.'), iif(c_sex = 0, 'женщина', iif(c_sex = 1, 'мужчина', 'юр.лицо')),
               iif(c_type = 0, 'физ.лицо', 'юр.лицо')
        from llc_kont_cli, clients, kontract, classific, product pr
        where lkc_from_lm_id = :lm_id and
              c_id = lkc_from_c_id and
              k_id = lkc_from_k_id and
              cs_id = lkc_status and
              pr_id = k_from_pr_id
        into k_kontract_no, k_percent, ma_id,  c_name_fio, c_inn, lkc_deliq_days, lkc_balance, lkc_status, lkc_status_str, llc_reserv_kind,
             lkc_percent, lkc_is_indiv, lkc_memo, lkc_has_loans, lkc_has_family_loans, lkc_has_memo, lkc_llc_sum,
             pr_name, lkc_from_c_id, lkc_from_k_id, k_disb_date, lkc_is_suprug_kred, lkc_suprug_id, lkc_suprug_memo,
             lkc_is_suprug_his_kred, is_ipf, lir_rate, mark, cr_rating, pr_loan_type_str, knc_sex_str, knc_type_str
    do
    begin
--     Вытаскивает тип и субтип бизнеса клиента
      bt_name = null;
      bs_name = null;
      select bt_name, bs_name, kc_amount_approved
      from knclient kc, bussubtype, bustype
      where kc_from_c_id = :lkc_from_c_id and
            kc_from_k_id = :lkc_from_k_id and
            kc_amount_approved > 0 and
            kc_busuness_subtype = bs_id and
            bs_from_bt_id = bt_id
      into bt_name, bs_name, kc_amount_approved;
      if (bs_name is null) then
      begin
        bt_name = 'Не указан';
        bs_name = 'Не указан';
      end

      suprug_name_fio = (select c_name_fio
                         from clients
                         where c_id = :lkc_suprug_id);

      type_financing = iif(is_ipf = 1, 'ИПФ', 'Классика');
      ---------------------------RFCKG-3884-------------------------------------------------
      if (db_id in (39801, 41701)) then
      begin
        select mac_id
        from miniappclient
        where mac_from_ma_id = :ma_id
            and mac_from_c_id= :lkc_from_c_id
        into :mac_id;

        detailed_purpose_description_1 = '';
        detailed_purpose_description_2 = '';
        select clpq.cq_name, clpc.cc_name
        from clp_miniappclient clpm
        inner join clp_questions clpq on clpm.cmc_questions_id = clpq.cq_id
        inner join clp_clarification clpc on clpm.cmc_clarification_id = clpc.cc_id
        where clpm.cmc_from_c_id = :lkc_from_c_id and
              clpm.cmc_from_ma_id =:ma_id
        into :detailed_purpose_description_1,
             :detailed_purpose_description_2;
        
        if (detailed_purpose_description_1 = '') then
        begin
            for select iif(coalesce(min(ovr_value), '') = '', list(rd_name, ', '), min(ovr_value))
            from objectprops
            inner join objectvalues on ovs_from_opr_id = opr_id
            inner join objectvalres on ovr_from_opr_id = opr_id and ovr_from_ovs_id = ovs_id
            inner join refdata on rd_id = ovr_result_rd_id
            where opr_object_id = :lkc_from_c_id
                and opr_object_kind = 0
                and opr_subject_kind = -1
                and :k_disb_date between ovs_date_begin and ovs_date_end
                and opr_subject_id = f_get_rd_id('anketaquestions.descr_of_loan')
            group by opr_subject_id
            into :temp_value
            do
            begin
                if (detailed_purpose_description_1 = '') then
                    detailed_purpose_description_1 = temp_value;
                else
                    detailed_purpose_description_1 = detailed_purpose_description_1 || ', ' || temp_value;
            end
        end

        ca_id = null;
        ca_from_cp_id = null;
        select ca.ca_id, ca.ca_from_cp_id
            from cfp_anketa ca
            where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
                  ca.ca_object_id = :mac_id and
                  ca.ca_from_c_id = :lkc_from_c_id and
                  ca.ca_from_acl_id is null
            into :ca_id, :ca_from_cp_id;
        
        income_activities_level_1 = '';
        income_activities_level_2 = '';
        income_activities_level_3 = '';
        income_activities_level_4 = '';
        cp_income_prefix_ru = '';
        income_activities_last = '';
        if ((:ca_id is not null) and (:ca_from_cp_id is not null)) then
        begin
            select cp.cp_income_prefix_ru
            from cfp_positions cp
            where cp.cp_id = :ca_from_cp_id
            into :cp_income_prefix_ru;
        
            for with recursive cp_hierarchy as (
                select cp.cp_id, cp.cp_name_ru, cp.cp_parenet_id, 1 as level
                from cfp_positions cp
                where cp.cp_id = :ca_from_cp_id
                
                union all
        
                select cp.cp_id, cp.cp_name_ru, cp.cp_parenet_id, ch.level + 1
                from cfp_positions cp
                inner join cp_hierarchy ch on cp.cp_id = ch.cp_parenet_id
            )
            select cp_name_ru, level
            from cp_hierarchy
            order by level desc
            into :cp_name, :hierarchy_level
            do
            begin
                if (hierarchy_level = 1) then
                    income_activities_level_4 = :cp_name;
                else if (hierarchy_level = 2) then
                    income_activities_level_3 = :cp_name;
                else if (hierarchy_level = 3) then
                    income_activities_level_2 = :cp_name;
                else if (hierarchy_level = 4) then
                    income_activities_level_1 = :cp_name;
            end
        
            for select cq.cq_name_ru, caa.caa_value, cq.cq_basic_incom
            from cfp_anketa_answers caa
            inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
              cq.cq_is_delete = 0 and
              cq.cq_basic_incom = 1
            where caa.caa_from_ca_id = :ca_id
            into :cq_name_ru, :caa_value, :cq_basic_incom
        
            do
            begin
              if (coalesce(caa_value, 0) >= 0 and
                  cq_basic_incom = 1) then
              begin
                  if (income_activities_last = '') then
                    income_activities_last = cq_name_ru || ': ' || :caa_value;
                  else
                    income_activities_last = :income_activities_last || ', ' || :cq_name_ru || ': ' || :caa_value;
              end
            end
        end
        if (:income_activities_level_1 = '') then
        begin
            select coalesce(res_str_values, '')
            from hp_objectprops_anketa_get('Clients', :lkc_from_c_id, -1, 'work_place_name', :k_disb_date, null)
            into :income_activities_level_1;
        end
        
        if (:income_activities_level_1 = '') then
        begin
            select ca.ca_income_txt_ru from cfp_anketa ca
            where ca.ca_from_c_id = :lkc_from_c_id
            and ca.ca_object_kind = f_objkind('miniappclient', 'id')
            and ca.ca_object_id = :mac_id
            into :income_activities_level_1;
            :income_activities_last = '';
        end

        ---------------------------  RFCKG-4249  -------------------------------------------------
        select iif((cast(:rep_date_end as date) - oks.oks_delinq_date < 0), 'Не просрочен', cast(:rep_date_end as date) -  oks.oks_delinq_date),
               oks.oks_last_sh_date
        from oksummary oks
        where oks_close_state = 0 and
             :rep_date_end between oks_date_begin and oks_date_end and
             oks_from_k_id = :lkc_from_k_id
        into :priznak_prosrochki,
             :k_kredline_end;

        k_period = round(datediff(month, k_disb_date, k_kredline_end) +
                                 (extract(day from k_kredline_end) - extract(day from k_disb_date)) / 30.41, 1);

        razbivka_summ = (case
                         when kc_amount_approved <= 10000 then '(1) до 10 000 сом'
                         when kc_amount_approved between 10001 and 25000 then '(2) 10 001 - 25 000 сом'
                         when kc_amount_approved between 25001 and 50000 then '(3) 25 001 - 50 000 сом'
                         when kc_amount_approved between 50001 and 300000 then '(4) 50 001 - 300 000 сом'
                         else '(5) Свыше 300 000'
                       end);


        razbivka_month = (case
                          when k_period <= 3.4 then '(1) 0 - 3 мес'
                          when k_period between 3.5 and 6.4 then '(2) 3 - 6 мес'
                          when k_period between 6.5 and 12.4 then '(3) 6 - 12 мес'
                          when k_period between 12.5 and 36.4 then '(4) 12 - 36 мес'
                          else '(5) Свыше 36 мес'
                        end);

        razbivka_srok = (case
                           when k_period <= 12.4 then '(1) Краткосрочный'
                         when k_period between 12.5 and 36.4 then '(2) Среднесрочный'
                         else '(3) Долгосрочный'
                       end);
      end
      -------------------------------------------------------------------------------------------------
      suspend;
    end
  end
end
