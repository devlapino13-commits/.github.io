CREATE PROCEDURE HP_EDOC_BLANK_SET_SCAN(
    EDB_ID D_ID,
    EDB_BLOB D_BLOB_BINARY,
    UN_ID D_ID = null,
    BLS_KIND D_SMALLINT = 2,
    SOURCE_SYSTEM D_SMALLINT = 0)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable edl_status d_integer; /* EDB_STATUS */
declare variable edf_print_name d_str_255;
declare variable bls_id d_id;
begin
  res_is_error = 1;
  res_text = '';
  select edl_status, edf_print_name
  from edoc_blank_scan, edoc_blank, edoc_list, edoc_frx
  where edbs_from_edb_id = :edb_id and
        edb_id = edbs_from_edb_id and
        edb_from_edl_id = edl_id and
        edb_from_edf_id = edf_id
  into edl_status, edf_print_name;

  if (edl_status is null) then
    res_text = 'Запись не найдена';

  if (res_text = '') then
  begin
    execute procedure hp_insert_into_blobstore ('edoc_scan_' || :edb_id || '.pdf', :edf_print_name, :edb_blob, :un_id, :bls_kind, :source_system)
    returning_values :bls_id;

    update edoc_blank_scan
    set edbs_from_bls_id = :bls_id,
        edbs_status_un_id = :un_id
    where edbs_from_edb_id = :edb_id;

    res_is_error = 0;
    res_text = 'Файл успешно прикреплен';
  end

  suspend;
end
