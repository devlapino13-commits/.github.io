CREATE PROCEDURE UXP$UXFRCOMBO_ENUM_VALUES(
    ENUM_CODE D_STR_CODE,
    USE_ENUM_CODE D_BOOL)
RETURNS (
    COMBO_VALUES D_STR_8100,
    VALUES_COUNT D_INTEGER)
AS
declare variable arr_item d_str_255;
begin
  combo_values = '';
  values_count = 0;

  if (use_enum_code is null) then
    use_enum_code = 0;

  if (enum_code = 'user_language') then
  begin
    for select '{"id":' || coalesce(el_id, 'Null') || ',' ||
               '"text":' || f_json_escape(coalesce(el_name, ''), 'str') || ',' ||
               '"image":"' || trim(coalesce(uuid_to_char(el_img_uuid), '')) || '"' || '}'
    from enum_list
    where el_kind = :enum_code
    order by el_kind, el_id
    into :arr_item
    do
    begin
      if (coalesce(arr_item, '') = '') then
        Continue;
  
      if (combo_values <> '') then
        combo_values = combo_values || ',';
  
      combo_values = combo_values || arr_item;
      values_count = values_count + 1;
    end
  end else
  if (use_enum_code = 0) then
  begin
    for select '{"id":' || coalesce(el_id, 'Null') || ',' ||
               '"text":' || f_json_escape(coalesce(uxf$lang(el_name), ''), 'str') || ',' ||
               '"image":"' || trim(coalesce(uuid_to_char(el_img_uuid), '')) || '"' || '}'
    from enum_list
    where el_kind = :enum_code
    order by el_kind, el_id
    into :arr_item
    do
    begin
      if (coalesce(arr_item, '') = '') then
        Continue;
  
      if (combo_values <> '') then
        combo_values = combo_values || ',';
  
      combo_values = combo_values || arr_item;
      values_count = values_count + 1;
    end
  end else
  begin
    for select '{"id":' || iif(coalesce(el_code, '') = '' or el_code = 'null', 'null', f_json_escape(coalesce(el_code, ''), 'str')) || ',' ||
               '"text":' || f_json_escape(coalesce(uxf$lang(el_name), ''), 'str') || ',' ||
               '"image":"' || trim(coalesce(uuid_to_char(el_img_uuid), '')) || '"' || '}'
    from enum_list
    where el_kind = :enum_code
    order by el_kind, el_id
    into :arr_item
    do
    begin
      if (coalesce(arr_item, '') = '') then
        Continue;
  
      if (combo_values <> '') then
        combo_values = combo_values || ',';
  
      combo_values = combo_values || arr_item;
      values_count = values_count + 1;
    end
  end

  if (combo_values <> '') then
    combo_values = '[' || combo_values || ']';

  suspend;
end
