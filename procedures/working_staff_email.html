CREATE PROCEDURE WORKING_STAFF_EMAIL(
    REP_DATE DATE)
RETURNS (
    EMAIL VARCHAR(255),
    C_ID D_ID)
AS
declare variable email_tmp varchar(255);
begin
  email = '';
  for select ps_from_c_id,
             (select cont_value
              from hp_contdata_get_email_valid(0, :c_id))
      from hr_pstatus ps
      where :rep_date between ps_date_begin and ps_date_end and
            ps_status = 1 and
            ps_is_working = 1
      order by 2
      into :c_id, :email_tmp
  do
  begin
    if (:email_tmp <> '' and
        :email <> :email_tmp) then
    begin
      :email = :email_tmp;
      suspend;
    end
  end
end
