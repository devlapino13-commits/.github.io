CREATE PROCEDURE HP_PDN_REF_RECALC_FILL(
    REP_DATE_ONE D_DATE,
    REP_DATE_TWO D_DATE)
AS
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_fact_disb_date d_date;
declare variable disb_days d_integer;
declare variable k_fact_closed_date d_date;
declare variable k_kolich_client_approved d_integer;
declare variable ref_cnt_list d_str_20;
declare variable ref_our d_str_long;
declare variable ref_our_active d_str_long;
declare variable ref_our_cnt d_smallint;
declare variable ref_our_cnt_active d_smallint;
declare variable ref_ext d_str_long;
declare variable ref_ext_active d_str_long;
declare variable ref_ext_cnt d_smallint;
declare variable ref_ext_cnt_active d_smallint;
declare variable pdn_status d_str_code;
declare variable pdn_status_id d_smallint;
declare variable pdn_date d_date;
declare variable pdn_days d_integer;
declare variable ssp_date d_date;
declare variable ssp_note d_str_255;
declare variable ssp_note_id d_smallint;
declare variable ki_date d_date;
declare variable ki_note d_str_255;
declare variable ki_note_id d_smallint;
declare variable is_selected d_bool;
declare variable prr_datetime d_datetime;
declare variable ma_id d_integer;
declare variable c_id d_id;
declare variable kc_id d_id;
declare variable pdr_kind d_smallint;
declare variable ref_cnt_active d_smallint;
declare variable txt d_str_255;
declare variable cnt d_integer;
begin
  -- ПДН - расчет в связи с неисполнением цели по рефинансированию (RFCRU-4309)
  -- процедура очищяет и заполняет таблицу pdn_ref_recalc

  pdr_kind = 2; -- пересчет ПДН через 60 дней (enum_list -> pdn_calc_source -> ref_60)
  prr_datetime = current_timestamp; -- форма актуальна на дату/время

  cnt = rep_date_two - rep_date_one;

  if (cnt is null or cnt < 0) then
    exception e_message 'Период указан неверно';

  if (cnt > 100) then
    exception e_message 'Период более 100 дней';

  delete from pdn_ref_recalc;

  for select k_id,
             k_kontract_no,
             k_fact_disb_date,
             k_fact_closed_date,
             k_kolich_client_approved,
             k_from_ma_app_id,
             kc_id,
             kc_from_c_id
      from kontract
      inner join kontstate on ks_from_k_id = k_id
      inner join knclient on kc_from_k_id = k_id
      where ks_ol_is_delete = 0 and
            ks_date_begin between :rep_date_one and :rep_date_two and
            ks_state = 1 and
            k_refund = 1 and
            kc_amount_approved > 0
      order by k_fact_disb_date
      into k_id,
           k_kontract_no, -- номер договора
           k_fact_disb_date, -- дата выдачи займа
           k_fact_closed_date, -- дата закрытия займа
           k_kolich_client_approved,
           ma_id,
           kc_id,
           c_id
  do
  begin
    if (k_kolich_client_approved > 1) then
      exception e_message 'В займе с рефами более одного клиента';

    ref_our = null;
    ref_our_active = null;
    ref_ext = null;
    ref_ext_active = null;

    ref_our_cnt = null;
    ref_our_cnt_active = null;
    ref_ext_cnt = null;
    ref_ext_cnt_active = null;

    ref_cnt_list = null;
    ref_cnt_active = null;

    pdn_status = null;
    pdn_status_id = null;
    pdn_date = null;
    pdn_days = null;

    ssp_date = null;
    ssp_note = null;
    ssp_note_id = null;

    ki_date = null;
    ki_note = null;
    ki_note_id = null;

    disb_days = current_date - k_fact_disb_date; -- прошло дней с даты выдачи
    is_selected = 0;

    select max(eil_date)
    from extinfolist
    inner join extinfocfg on eic_id = eil_from_eic_id
    where eic_code = 'bki_ssp_40' and -- запрос ССП
          eil_status = 2 and -- запрос успешный
          eil_from_c_id = :c_id and
          eil_date > :k_fact_disb_date
    into ssp_date; -- дата последнего успешного запроса ССП после выдачи

    select max(eil_date)
    from extinfolist
    inner join extinfocfg on eic_id = eil_from_eic_id
    where eic_code = 'equifax_40' and -- запрос КИ
          eil_status = 2 and -- запрос успешный
          eil_from_c_id = :c_id and
          eil_date > :k_fact_disb_date
    into ki_date; -- дата последнего успешного запроса КИ после выдачи

    select list(k_kontract_no, ', '),
           count(*)
    from kontrefinance
    left join kontract on k_id = kr_from_k_id
    where kr_from_ma_id = :ma_id and
          k_id is not null
    into ref_our,
         ref_our_cnt; -- внутренние рефы всего

    select list(kr_loan_uuid, ', '),
           count(*)
    from kontrefinance
    where kr_from_ma_id = :ma_id and
          kr_from_k_id is null
    into ref_ext,
         ref_ext_cnt; -- внешние рефы всего

    select list(k_kontract_no, ', '),
           count(*)
    from kontrefinance
    inner join kontract on k_id = kr_from_k_id
    inner join oksummary on oks_from_k_id = k_id
    where kr_from_ma_id = :ma_id and
          current_date between oks_date_begin and oks_date_end
    into ref_our_active,
         ref_our_cnt_active; -- активные внутренние рефы

    if (ssp_date is null) then -- нет успешного запроса ССП после выдачи займа
    begin
      ref_ext_active = ref_ext; -- активные внешние рефы повторяются из колонки внешние рефы всего
      ref_ext_cnt_active = ref_ext_cnt;
    end
    else
    begin
      in autonomous transaction do
        select list(kr_loan_uuid, ', '),
               count(*)
        from kontrefinance
        inner join hp_bki_get_list_for_refinace(:c_id, :k_fact_disb_date + 1, 1) on uuid = kr_loan_uuid
        where kr_from_ma_id = :ma_id and
              kr_from_k_id is null
        into ref_ext_active,
             ref_ext_cnt_active; -- активные внешние рефы
    end

    ref_cnt_active = ref_our_cnt_active + ref_ext_cnt_active;
    ref_cnt_list = ref_our_cnt || '/' || ref_our_cnt_active || ' - ' || ref_ext_cnt || '/' || ref_ext_cnt_active; -- кол-во рефов (внутренних-внешних)

    select kc_date,
           kc_date - :k_fact_disb_date
    from knclient_pdn
    where kc_id = :kc_id and
          kcp_kind = :pdr_kind
    into pdn_date, -- дата пересчета ПДН
         pdn_days; -- кол-во дней, через которое ПДН пересчитан с даты выдачи

    if (pdn_date is not null) then
      pdn_status_id = 1; -- статус по перерасчету ПДН: ПДН рассчитан

    if (k_fact_closed_date is not null) then
      pdn_status_id = 2; -- Закрыт

    /* pdn_status_id:
   1 - ПДН расcчитан.                Условия статуса: pdn_date не пустой при pdr_kind = 2
   2 - Закрыт.                       Условия статуса: контракт с рефами закрыт
   3 - Рефы закрыты.                 Условия статуса: контракт, чьи рефы закрыты
   4 - Кандидат.                     Условия статуса: контракт не закрыт, имеет активные рефы, нет успешного запроса ССП или КИ
   5 - Кандидат: КИ получена.        Условия статуса: контракт не закрыт, имеет активные рефы, есть успешные запросы ССП и КИ за последние 2 дня включая текущий, КИ получена.
   6 - Кандидат: клиент не имеет КИ. Условия статуса: контракт не закрыт, имеет активные рефы, есть успешные запросы ССП и КИ за последние 2 дня включая текущий, КИ получена пустая.
   7 - Кандидат: КИ устарела.        Условия статуса: контракт не закрыт, имеет активные рефы, есть успешный запрос КИ, но не за текущий или предыдущий день.

   Доп. статус "Обновите ССП/КИ" - контракт имеет активный внешний реф + pdn_status_id in(4, 7).
   Доп. статус отображается после основного статуса.

   Действия:
   1) Отправить запрос ССП. Не запускается, если за за сегодня или вчера есть другой успешный запрос ССП. Не запускается, если отсутствуют активные внешние рефы.
   2) Отправить запрос ССП/КИ. Запрос ССП может быть пропущен, если за сегодня или вчера есть другой успешный запрос ССП. Точно также запрос КИ может быть пропущен.
   3) Пересчитать ПДН. Запускается только для pdn_status_id in(5, 6): "Кандидат: КИ получена" и "Кандидат: клиент не имеет КИ".
                       Не запускается если нет актуальных ССП и КИ (нету успешных запросов ССП/КИ за последние 2 дня, включая текущий).
   */

    if (pdn_status_id is null) then
    begin
      begin
        in autonomous transaction do
          select trim(ssp_result) || ' ' || f_datetimetostr(ssp_datetime),
                 ssp_result_id,
                 trim(ki_result) || ' ' || f_datetimetostr(ki_datetime),
                 ki_result_id
          from hp_pdn_get_request_status(:c_id, :k_fact_disb_date + 1)
          into ssp_note, -- комментарий относительно последнего запроса ССП после выдачи займа
               ssp_note_id,
               ki_note, -- комментарий относительно последнего запроса КИ после выдачи займа
               ki_note_id;

        when any do
        begin
          txt = 'При получении результатов запросов ССП/КИ произошла ошибка (hp_pdn_get_recalclist)';
          txt = txt || ascii_char(10) || 'c_id ' || c_id || ', k_fact_disb_date ' || f_datetostr(k_fact_disb_date);
          exception e_message txt;
        end
      end

      if (pdn_status_id is null and ref_cnt_active = 0) then
        pdn_status_id = 3; -- Рефы закрыты

      if (pdn_status_id is null and k_fact_closed_date is null and ref_cnt_active > 0 and (ssp_date is null or ki_date is null)) then
        pdn_status_id = 4; -- Кандидат

      if (pdn_status_id is null and k_fact_closed_date is null and ref_cnt_active > 0 and ki_date < current_date - 1) then
        pdn_status_id = 7; -- Кандидат: КИ устарела

      if (pdn_status_id is null and k_fact_closed_date is null and ref_cnt_active > 0 and ssp_date >= current_date - 1 and ki_date >= current_date - 1 and ki_note_id = 9) then
        pdn_status_id = 5; -- Кандидат: КИ получена

      if (pdn_status_id is null and k_fact_closed_date is null and ref_cnt_active > 0 and ssp_date >= current_date - 1 and ki_date >= current_date - 1 and ki_note_id = 5) then
        pdn_status_id = 6; -- Кандидат: клиент не имеет КИ
    end

    pdn_status = case
                   when pdn_status_id = 1 then 'ПДН рассчитан'
                   when pdn_status_id = 2 then 'Закрыт'
                   when pdn_status_id = 3 then 'Рефы закрыты'
                   when pdn_status_id = 4 then 'Кандидат'
                   when pdn_status_id = 5 then 'Кандидат: КИ получена'
                   when pdn_status_id = 6 then 'Кандидат: клиент не имеет КИ'
                   when pdn_status_id = 7 then 'Кандидат: КИ устарела'
                   else 'Не известно'
                 end;

    if (ref_ext_cnt_active > 0 and pdn_status_id in (4, 7)) then
      pdn_status = trim(pdn_status) || ', обновите ССП/КИ';

    insert into pdn_ref_recalc (prr_k_id, prr_k_kontract_no, prr_k_fact_disb_date, prr_disb_days, prr_k_fact_closed_date, prr_ref_cnt_list, prr_ref_our, prr_ref_our_active, prr_ref_our_cnt, prr_ref_our_cnt_active, prr_ref_ext, prr_ref_ext_active, prr_ref_ext_cnt, prr_ref_ext_cnt_active, prr_pdn_status, prr_pdn_status_id, prr_pdn_date, prr_pdn_days, prr_ssp_date, prr_ssp_note, prr_ssp_note_id, prr_ki_date, prr_ki_note, prr_ki_note_id, prr_is_selected, prr_datetime, prr_ma_id, prr_kc_id, prr_c_id)
    values (:k_id, :k_kontract_no, :k_fact_disb_date, :disb_days, :k_fact_closed_date, :ref_cnt_list, :ref_our, :ref_our_active, :ref_our_cnt, :ref_our_cnt_active, :ref_ext, :ref_ext_active, :ref_ext_cnt, :ref_ext_cnt_active, :pdn_status, :pdn_status_id, :pdn_date, :pdn_days, :ssp_date, :ssp_note, :ssp_note_id, :ki_date, :ki_note, :ki_note_id, :is_selected, :prr_datetime, :ma_id, :kc_id, :c_id);
  end
end
