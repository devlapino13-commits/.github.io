CREATE PROCEDURE HP_ACTIONLIST_GET_LOANS_STATES(
    C_ID ID_TYPE)
RETURNS (
    RES_OBJECT_DETAIL BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable opr_id d_integer;
declare variable res str_20;
declare variable ovs_datetime d_datetime;
declare variable k_kontract_no str_20;
declare variable res_comment d_str_long;
declare variable counter d_integer;
declare variable ovs_value d_integer;
begin

  res_object_detail = '';
  counter = 0;

  for select opr_id, ovs_datetime, k_kontract_no, ovs_value,
             case ovs_value
               when -1 then 'Доработка'
               when -2 then 'Отказан'
               when 2 then 'Одобрен'
             end
      from objectprops
      inner join objectvalues on ovs_from_opr_id = opr_id
      inner join username on un_id = ovs_from_un_id
      inner join knclient on kc_from_k_id = opr_object_id and
            opr_object_kind = 9
      inner join kontract on k_id = kc_from_k_id
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            opr_subject_id = (select rd_id
                              from hp_get_check_loan_data(k_id)) and
            ovs_value in (2, -1, -2)
      order by ovs_datetime desc
      rows 5
      into opr_id, ovs_datetime, k_kontract_no, ovs_value, res
  do
  begin
    counter = counter + 1;
    res_comment = '';

    if (ovs_value in (-1, -2)) then
    begin
      select f_get_verify_memo(obm_memo, 'reason')
      from objectmemo
      where obm_from_opr_id = :opr_id and
            obm_memo not containing ('Для сведения:') and
            obm_datetime = :ovs_datetime
      order by obm_id
      rows 1
      into res_comment;
    end

    res_object_detail = blob_append(res_object_detail, counter);
    res_object_detail = blob_append(res_object_detail, '. ' || k_kontract_no || ' - ');
    res_object_detail = blob_append(res_object_detail, '<b>' || res || '</b><br>');
    res_object_detail = blob_append(res_object_detail, f_datetimetostr(ovs_datetime) || '<br>');
    res_object_detail = blob_append(res_object_detail, '<i>' || res_comment || '</i><br><br>');

  end
  suspend;
end
