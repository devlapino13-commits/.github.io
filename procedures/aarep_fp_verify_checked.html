CREATE PROCEDURE AAREP_FP_VERIFY_CHECKED(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_ID D_INTEGER,
    RES_REG_NAME D_STR_CODE,
    RES_DATE_CHECK D_DATE,
    RES_DATE_CONF D_DATETIME,
    RES_NAME1 D_STR_100,
    RES_BIRTHDAY1 D_DATE,
    RES_NAME2 D_STR_100,
    RES_BIRTHDAY2 D_DATE,
    RES_CATEG D_STR_255,
    RES_ACT D_STR_20,
    RES_REASON D_STR_255)
AS
begin
  for
    select
      fpv_id,
      reg_name,
      fpv_date_check,
      fpv_confirm_date,
      c_name_fio,
      c_birth_date,
      fpv_fio,
      fpv_birthdate,
      rd_name,
      iif(fpv_is_confirm = 1, 'разрешение', 'отказ'),
      fpv_reason
    from fp_verify
      inner join clients
        on c_id = fpv_c_id
      inner join region_list(:rep_region_id, 1, -1, 1)
        on id_list = fpv_reg_id
      inner join region
        on reg_unicode = fpv_reg_id
      left join refdata
        on rd_id = fpv_terr_categ_id
    where cast(fpv_date_check as date) between :rep_date_begin and :rep_date_end
      and fpv_is_confirm <> 0
    order by fpv_date_check
  into
    :res_id,
    :res_reg_name,
    :res_date_check,
    :res_date_conf,
    :res_name1,
    :res_birthday1,
    :res_name2,
    :res_birthday2,
    :res_categ,
    :res_act,
    :res_reason
  do
  begin
    suspend;
  end
end
