CREATE PROCEDURE HP_KK_SELFEMPLOYED_VIEW(
    ACL_ID D_ID,
    C_ID D_ID)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    C_NAME_FIO D_STR_100,
    KSE_ID D_INTEGER,
    KSE_FROM_ACL_ID D_ID,
    KSE_FROM_UN_ID D_ID,
    KSE_FROM_K_ID D_ID,
    KSE_FROM_C_ID D_ID,
    KSE_DATETIME D_DATETIME,
    KSE_OGRNIP_NO D_STR_20,
    KSE_CLEARANCE_CERTIFICATE D_BOOL,
    KSE_REPUTATION D_BOOL,
    KSE_LICENSE_INFO D_STR_LONG)
AS
begin
  for
    select
      k_kontract_no,
      c_name_fio,
      kse_id,
      acl_id,
      kse_from_un_id,
      k_id,
      c_id,
      kse_datetime,
      kse_ogrnip_no,
      coalesce(kse_clearance_certificate, 0),
      coalesce(kse_reputation, 0),
      kse_license_info
    from actionlist
      inner join kontract
        on k_id = acl_object_id
      inner join knclient
        on kc_from_k_id = k_id
          and kc_from_c_id = :c_id
      inner join clients
        on c_id = kc_from_c_id
      left join kk_selfemployed
        on kse_from_acl_id = acl_id
          and kse_from_c_id = c_id
    where acl_id = :acl_id
      and acl_object_kind = 9
  into
    :k_kontract_no,
    :c_name_fio,
    :kse_id,
    :kse_from_acl_id,
    :kse_from_un_id,
    :kse_from_k_id,
    :kse_from_c_id,
    :kse_datetime,
    :kse_ogrnip_no,
    :kse_clearance_certificate,
    :kse_reputation,
    :kse_license_info
  do
  begin
    suspend;
  end
end
