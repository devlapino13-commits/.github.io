CREATE PROCEDURE HP_KN_OPERBOOK_FIX_ONE_71006(
    REP_DATE D_DATE,
    REP_OFFICE_ID D_INTEGER)
RETURNS (
    OL_ID D_ID,
    KOB_ID D_ID,
    ACC_CODE D_STR_20,
    ACC_NON_RESIDENT D_BOOL,
    OL_DATE_MIN D_DATE,
    OL_DATE_MAX D_DATE,
    BAL_SUM_1 D_CURRENCY,
    BAL_SUM_2 D_CURRENCY)
AS
declare variable acc_71006 d_id;
declare variable acc_71005 d_id;
declare variable detail_str d_str_255;
declare variable kob_detail_id d_id;
declare variable is_finish d_bool;
declare variable ol_date d_date;
begin
  detail_str = 'Кор-ка остатка ВБС';

  select obd_id
  from kn_obdetail
  where obd_detail = :detail_str
  into :kob_detail_id;

  if (kob_detail_id is null) then
  begin
    insert into kn_obdetail (obd_detail)
    values (:detail_str)
    returning obd_id
    into :kob_detail_id;
  end

  for select acc_id, acc_pair_acc_id, acc_code, acc_non_resident
  from ref_account
  where acc_from_sab_id = 47 and
        acc_from_reg_id = :rep_office_id and
        acc_type = 0
  into acc_71006, acc_71005, acc_code, acc_non_resident
  do
  begin
    is_finish = 1;

    ol_date_min = null;
    ol_date_max = null;
    ol_date = rep_date;

    update kn_operbook
    set kob_acc_dt = :acc_71006
    where kob_ol_date >= :ol_date and
          kob_acc_dt = :acc_71005
    order by kob_ol_date, kob_id;

    if (row_count > 0) then
      is_finish = 0;

    update kn_operbook
    set kob_acc_kt = :acc_71006
    where kob_ol_date >= :ol_date and
          kob_acc_kt = :acc_71005
    order by kob_ol_date, kob_id;

    if (row_count > 0) then
      is_finish = 0;

    while (is_finish = 0) do
    begin
      update kn_operbook
      set kob_summa = 0
      where kob_ol_date >= :ol_date and
            kob_acc_dt = :acc_71006 and
            kob_acc_kt = :acc_71006
      order by kob_ol_date, kob_id;

      ol_date_min = null;
      ol_date_max = null;
      kob_id = null;
      bal_sum_2 = 0;
      bal_sum_1 = 0;

      select kobs_date, kobs_bal_sum
      from kn_obsummary
      where kobs_acc_id = :acc_71006 and
            kobs_date >= :ol_date and
            kobs_bal_sum < 0
      order by 1
      rows 1
      into ol_date_min, bal_sum_1;
  
      if (ol_date_min is not null) then
      begin
        select kobs_date, kobs_bal_sum
        from kn_obsummary
        where kobs_acc_id = :acc_71006 and
              kobs_date > :ol_date_min and
              kobs_bal_sum > 0
        order by 1
        rows 1
        into ol_date_max, bal_sum_2;
  
        if (ol_date_max is null) then
        begin
          ol_date_max = '31.12.2100';
          is_finish = 1;
        end else
          ol_date = ol_date_max + 1;

        update kn_operbook
        set kob_acc_dt = :acc_71005
        where kob_ol_date between :ol_date_min + 1 and :ol_date_max and
              kob_acc_dt = :acc_71006
        order by kob_ol_date, kob_id;
    
        update kn_operbook
        set kob_acc_kt = :acc_71005
        where kob_ol_date between :ol_date_min + 1 and :ol_date_max and
              kob_acc_kt = :acc_71006
        order by kob_ol_date, kob_id;

        select ol_id
        from hp_get_kn_operbook_ol(:ol_date_min, :rep_office_id, 'calc_vbs')
        into :ol_id;

        insert into kn_operbook (kob_from_ol_id, kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa,
                                 kob_summa_val, kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id,
                                 kob_ol_is_active, kob_detail_id)
        values (:ol_id, '810', 1, :acc_71006, :acc_71005, -:bal_sum_1, 0,
                0, 0, :ol_date_min, :rep_office_id, 1, :kob_detail_id)
        returning kob_id
        into kob_id;
  
        suspend;

        if (bal_sum_2 > 0) then
        begin
          select ol_id
          from hp_get_kn_operbook_ol(:ol_date_max, :rep_office_id, 'calc_vbs')
          into :ol_id;
  
          insert into kn_operbook (kob_from_ol_id, kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa,
                                   kob_summa_val, kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id,
                                   kob_ol_is_active, kob_detail_id)
          values (:ol_id, '810', 1, :acc_71006, :acc_71005, :bal_sum_2, 0,
                  0, 0, :ol_date_max, :rep_office_id, 1, :kob_detail_id)
          returning kob_id
          into kob_id;

          suspend;
        end
      end else
        is_finish = 1;
    end
  end
end
