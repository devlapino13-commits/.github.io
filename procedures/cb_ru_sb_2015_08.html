CREATE PROCEDURE CB_RU_SB_2015_08(
    REP_DATE_END DATE)
RETURNS (
    A_COEFF CURR_TYPE,
    B_COEFF INTEGER,
    C_COEFF CURR_TYPE,
    SUM_VZYSKAN CURR_TYPE,
    SUM_BONUS_BEFORE_60 CURR_TYPE,
    SUM_BONUS_VZYSKAN CURR_TYPE,
    SUM_BONUS_TOTAL CURR_TYPE,
    KOL_CLI_BEGIN INTEGER,
    KOL_CLI_PEREDAN INTEGER,
    KOL_CLI_OTOZVAN INTEGER)
AS
declare variable rep_date1 date;
declare variable rep_date15 date;
declare variable rep_date16 date;
declare variable rep_date31 date;
declare variable a_coeff_1 curr_type;
declare variable b_coeff_1 integer;
declare variable sum_vzyskan_1 curr_type;
declare variable a_coeff_2 curr_type;
declare variable b_coeff_2 integer;
declare variable sum_vzyskan_2 curr_type;
declare variable kol_cli_begin_1 integer;
declare variable kol_cli_peredan_1 integer;
declare variable kol_cli_otozvan_1 integer;
declare variable kol_cli_begin_2 integer;
declare variable kol_cli_peredan_2 integer;
declare variable kol_cli_otozvan_2 integer;
declare variable kol_cli_coeff integer;
begin
  rep_date1 = rep_date_end - extract(day from :rep_date_end) + 1;
  rep_date15 = rep_date1 + 14;
  rep_date16 = rep_date1 + 15;
  rep_date31 = rep_date1 + 32;
  rep_date31 = rep_date31 - extract(day from :rep_date31);

  a_coeff_1 = null;
  b_coeff_1 = null;
  sum_vzyskan_1 = null;
  kol_cli_begin_1 = null;
  kol_cli_peredan_1 = null;
  kol_cli_otozvan_1 = null;
  select sum(iif(delinq_day_begin <= 60, perc_repay + fine_repay, 0)), -- A - сумма погашенных процентов и пени
         sum(kol_cli_before_60), -- B - Кол-во клиентов вставших в график или внесших погашения от 4 тыс руб.
         sum(iif(delinq_day_begin > 60 or is_woff = 1, os_repay + perc_repay + fine_repay, 0)), -- общая сумма взыскание с просрочки более 60 дней
         sum(kol_cli_begin), sum(kol_cli_peredan), sum(kol_cli_otozvan)
  from cb_ru_sb_2015_08_temp(:rep_date1, :rep_date15)
  into a_coeff_1, b_coeff_1, sum_vzyskan_1, kol_cli_begin_1, kol_cli_peredan_1, kol_cli_otozvan_1;
  if (a_coeff_1 is null) then
    a_coeff_1 = 0;
  if (b_coeff_1 is null) then
    b_coeff_1 = 0;
  if (sum_vzyskan_1 is null) then
    sum_vzyskan_1 = 0;

  if (kol_cli_begin_1 is null) then
    kol_cli_begin_1 = 0;
  if (kol_cli_peredan_1 is null) then
    kol_cli_peredan_1 = 0;
  if (kol_cli_otozvan_1 is null) then
    kol_cli_otozvan_1 = 0;

  a_coeff_2 = null;
  b_coeff_2 = null;
  sum_vzyskan_2 = null;
  kol_cli_begin_2 = null;
  kol_cli_peredan_2 = null;
  kol_cli_otozvan_2 = null;
  select sum(iif(delinq_day_begin <= 60, perc_repay + fine_repay, 0)), -- A - сумма погашенных процентов и пени
         sum(kol_cli_before_60), -- B - Кол-во клиентов вставших в график или внесших погашения от 4 тыс руб.
         sum(iif(delinq_day_begin > 60 or is_woff = 1, os_repay + perc_repay + fine_repay, 0)), -- общая сумма взыскание с просрочки более 60 дней
         sum(kol_cli_begin), sum(kol_cli_peredan), sum(kol_cli_otozvan)
  from cb_ru_sb_2015_08_temp(:rep_date16, :rep_date31)
  into a_coeff_2, b_coeff_2, sum_vzyskan_2, kol_cli_begin_2, kol_cli_peredan_2, kol_cli_otozvan_2;
  if (a_coeff_2 is null) then
    a_coeff_2 = 0;
  if (b_coeff_2 is null) then
    b_coeff_2 = 0;
  if (sum_vzyskan_2 is null) then
    sum_vzyskan_2 = 0;

  if (kol_cli_begin_2 is null) then
    kol_cli_begin_2 = 0;
  if (kol_cli_peredan_2 is null) then
    kol_cli_peredan_2 = 0;
  if (kol_cli_otozvan_2 is null) then
    kol_cli_otozvan_2 = 0;

  a_coeff = a_coeff_1 + a_coeff_2;
  b_coeff = b_coeff_1 + b_coeff_2;

  kol_cli_begin = kol_cli_begin_1 + kol_cli_begin_2;
  kol_cli_peredan = kol_cli_peredan_1 + kol_cli_peredan_2;
  kol_cli_otozvan = kol_cli_otozvan_1 + kol_cli_otozvan_2;

  if (:rep_date_end < cast('01.09.2015' as date)) then
  begin
    kol_cli_coeff = kol_cli_begin + kol_cli_peredan - kol_cli_otozvan;
  end
  else
  if (:rep_date_end < cast('01.06.2016' as date)) then
  begin
    kol_cli_coeff = 400;
  end
  else
  if (:rep_date_end < cast('01.06.2024' as date)) then
  begin
    kol_cli_coeff = kol_cli_begin + kol_cli_peredan; -- - kol_cli_otozvan;
    if (kol_cli_coeff > 400) then
      kol_cli_coeff = 400;
  end
  else
  begin
    kol_cli_coeff = kol_cli_begin + kol_cli_peredan; -- - kol_cli_otozvan;
    if (kol_cli_coeff > 200) then
      kol_cli_coeff = 200;
  end

  --  C_Coeff = cast((B_Coeff_1 + B_Coeff_2) as double precision) / 400;
  if (kol_cli_coeff > 0) then
    c_coeff = cast((b_coeff_1 + b_coeff_2) as double precision) / kol_cli_coeff;
  sum_vzyskan = sum_vzyskan_1 + sum_vzyskan_2;
  if (c_coeff > 0.22) then
  begin
    --    sum_bonus_before_60 = 0.3 * (A_Coeff / B_Coeff) * 400 * (C_Coeff - 0.3);
    sum_bonus_before_60 = 0.3 * (a_coeff / b_coeff) * kol_cli_coeff * (c_coeff - 0.22);
  end
  else
    sum_bonus_before_60 = 0;
  if (:rep_date_end < cast('01.06.2024' as date)) then
  begin
    if (sum_vzyskan > 200000) then
    begin
      sum_bonus_vzyskan = (sum_vzyskan - 200000) * 0.05;
    end
    else
      sum_bonus_vzyskan = 0;
  end
  else
  begin
    if (sum_vzyskan > 100000) then
    begin
      sum_bonus_vzyskan = (sum_vzyskan - 100000) * 0.05;
    end
    else
      sum_bonus_vzyskan = 0;
  end

  sum_bonus_total = sum_bonus_before_60 + sum_bonus_vzyskan;
  suspend;
end
