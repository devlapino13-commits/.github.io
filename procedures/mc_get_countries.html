Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE MC_GET_COUNTRIES
RETURNS (
    ID D_INTEGER,
    CODE D_STR_LANG,
    NAME D_STR_255)
AS
declare variable db_id d_integer;
declare variable tmp_rd_code d_str_255;
begin
-- На первом месте страна БД
  db_id = f_get_db_id();
  tmp_rd_code = case db_id
                  when 41701 then 'citizen.kg'
                  when 94401 then 'citizen.az'
                  else 'citizen.ru'
                end;

  for select rd_id, replace(rd_code, 'citizen.', ''), rd_name
      from refdata
      where rd_code = :tmp_rd_code and
            rd_is_active = 1
      union all
      select rd_id, replace(rd_code, 'citizen.', ''), rd_name
      from refdata
      where rd_main_code = 'citizen' and
            rd_is_active = 1 and
            rd_code <> rd_main_code and
            coalesce(rd_str_id, '') <> '' and
            rd_code <> :tmp_rd_code
            and rd_code in ('citizen.az', 'citizen.kg', 'citizen.ru', 'citizen.tj', 'citizen.uz')
      into id, code, name
  do
    suspend;
end
