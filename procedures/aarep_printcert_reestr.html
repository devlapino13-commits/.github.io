CREATE PROCEDURE AAREP_PRINTCERT_REESTR(
    PAR1 INTEGER)
RETURNS (
    LINE_NO INTEGER,
    K_MEMO VARCHAR(255),
    K_STOP_PERC DATE,
    C_NAME_FIO VARCHAR(100),
    C_ADDRESS_STR VARCHAR(255))
AS
declare variable PLP_IDX varchar(10);
declare variable PLS_IDX varchar(10);
declare variable IDX_POS integer;
declare variable ADDR_LEN integer;
begin
  line_no = 0;
  for select k_stop_perc, c_name_fio, adl_address_str, coalesce(plp.pl_post_index, ''),
             coalesce(pls.pl_post_index, '')
      from kontract, knclient, clients, printcert, addrclient, addrlist, places pls, places plp
      where prc_pocket_id = :par1 and
            prc_from_k_id = k_id and
            kc_from_k_id = k_id and
            kc_from_c_id = c_id and
            adc_from_c_id = c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id and
            kc_amount_approved > 0 and
            pls.pl_id = adl_street_id and
            plp.pl_id = adl_from_pl_id
      order by prc_id, kc_number
      into :k_stop_perc, c_name_fio, c_address_str, plp_idx, pls_idx
  do
  begin
    /* Добавление индекса к адресу */
-- Приоритет имеет индекс улицы
    if (pls_idx = '') then
      pls_idx = plp_idx;
    if (pls_idx <> '') then
    begin
-- Определение позиции, в которой заканчивается почтовый индекс
      idx_pos = 1;
      addr_len = char_length(c_address_str);
      while (substring(c_address_str from idx_pos for 1) similar to '[[:DIGIT:], ]' and idx_pos < addr_len) do
      begin
        idx_pos = idx_pos + 1;
      end
      idx_pos = idx_pos - 1;
      if (idx_pos = 0) then -- Индекса в адресе не было
        c_address_str = pls_idx || ', ' || c_address_str;
      else
        c_address_str = pls_idx || ', ' || right(c_address_str, addr_len - idx_pos);
    end
    line_no = line_no + 1;
    suspend;
  end
end
