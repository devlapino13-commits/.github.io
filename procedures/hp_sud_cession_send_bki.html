CREATE PROCEDURE HP_SUD_CESSION_SEND_BKI(
    IN_SCP_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable k_id d_id;
declare variable cession_date d_date;
begin
  res_text = '';

  select scp_date
  from sud_cession_packet
  where scp_id = :in_scp_id
    and scp_status = 0
  into cession_date;

  if (cession_date is null) then
    res_text = 'Не найден черновик для отправки в БКИ.';

  if (res_text = '') then
  begin
    for select scl_from_k_id
    from sud_cession_loans
    where scl_from_scp_id = :in_scp_id
    into k_id
    do
      execute procedure hp_bki_add_to_buffer('.hp_sud_cession_send_bki.', 9, k_id, cession_date);

    update sud_cession_packet
    set scp_status = 1,
        scp_bki_event_datetime = localtimestamp
    where scp_id = :in_scp_id;
  end 

  suspend;
end
