    KONT_KATEG_RESH D_STR_10,
    CRED_FULL_COST_CATEG DECIMAL(15,3),

    KOL_RESH_NOT_KK D_SMALLINT,

    FIRST_RESH_DATE_KK D_DATE,
    KOL_RESH_KK D_SMALLINT,
    RESH_KIND_STR D_STR_10,
    KOL_RESH_NOT_KK_STR D_SMALLINT)
AS
declare variable OKS_IS_REFUND D_SMALLINT;
declare variable OKS_IS_REFUND_LLR D_SMALLINT;
declare variable OKS_IS_RESCHEDULE_LLR D_SMALLINT;
declare variable OKS_IS_ZALOG D_SMALLINT;
declare variable OKS_IS_ZALOG_LLR D_SMALLINT;
declare variable OKS_CLIENT_IS_BANKRUPT D_SMALLINT;
declare variable OKS_ZALOG_IS_BANKRUPT D_SMALLINT;
declare variable OKS_IS_NOT_TARGETED D_SMALLINT;
declare variable OKS_NO_DOCUMENT D_SMALLINT;
declare variable C_ID D_ID;
declare variable K_ID D_ID;
declare variable SAB_ID_ZALOG_COST D_ID;
declare variable PR_CODE D_STR_100;
declare variable K_KREDLINE_END D_DATE;
declare variable K_PERIOD_DAY_RESH D_INTEGER;
declare variable IS_REF D_SMALLINT;
declare variable KC_CLIENT_TYPE D_INTEGER;
declare variable KC_IS_MSP D_SMALLINT;
declare variable OKS_RESTRUCT_DATE D_DATE;
begin
  rec_no = 0;
  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Залог'
  into :sab_id_zalog_cost;
  for select oks_from_kc_id, k_id, oks_date_begin, oks_close_state, kc_amount_approved,
             datediff(day, k_disb_date, k_kont_last_date), k_kontract_no, oks_from_kc_id, oks_disb_date,
             oks_kont_last_date, oks_last_sh_date, oks_balance, oks_bal_perc, oks_bal_kommis, oks_diff_esp_bal,
             oks_bal_esp_end, oks_ead_amount, oks_refund_delinq_180, oks_last_resh_date, oks_last_resh_delinq_180,
             oks_llr_group, oks_delinq_days, oks_last_resh_delinq_before, oks_refund_delinq_before,
             oks_delinq_days_own_llr, oks_paral_loan_count, oks_paral_delinq_day, oks_delinq_days_llr,
             oks_delinq_days_365, oks_llr_base_percent, oks_llr_percent_outer, oks_ecl_percent, oks_ecl_vbs,
             oks_llr_percent_inner, oks_llr_percent_nadzor, oks_llr_percent, oks_llr_os_nadzor, oks_llr_perc_nadzor,
             oks_ecl_vbs_nadzor, oks_llr_os, oks_llr_perc, oks_diff_ecl_llr, oks_ecl_acc_kind, pr_name,
             oks_percent_norm, oks_percent_esp, c_name_fio, c_id, oks_is_refund, oks_is_refund_llr,
             oks_is_reschedule_llr, oks_is_zalog, oks_is_zalog_llr, oks_client_is_bankrupt, oks_zalog_is_bankrupt,
             oks_is_not_targeted, oks_no_document, datediff(day, k_disb_date, :rep_date), k_fact_disb_online,
             oks_delinq_last_365, oks_delinq_field_30_1, oks_delinq_field_30_2, p.pr_code,
             cast(k_percent_effect_fifd as decimal(15,3)) as cred_full_cost, k_kredline_end,
             kc_client_type, kc_is_msp, oks_kol_resh, oks_restruct_date
      from kn_oksummary, knclient, kontract, clients, product p
      where oks_from_kc_id = kc_id and
            oks_date_begin = :rep_date and
            oks_close_state = 0 and
            oks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            kc_from_c_id = c_id and
            (oks_balance > 0 or
            oks_bal_perc > 0)
      into oks_from_kc_id, k_id, oks_date_begin, oks_close_state, kc_amount_approved, k_period_day, k_kontract_no,
           oks_from_kc_id, oks_disb_date, oks_kont_last_date, oks_last_sh_date, oks_balance, oks_bal_perc,
           oks_bal_kommis, oks_diff_esp_bal, oks_bal_esp_end, oks_ead_amount, oks_refund_delinq_180, oks_last_resh_date,
           oks_last_resh_delinq_180, oks_llr_group, oks_delinq_days, oks_last_resh_delinq_before,
           oks_refund_delinq_before, oks_delinq_days_own_llr, oks_paral_loan_count, oks_paral_delinq_day,
           oks_delinq_days_llr, oks_delinq_days_365, oks_llr_base_percent, oks_llr_percent_outer, oks_ecl_percent,
           oks_ecl_vbs, oks_llr_percent_inner, oks_llr_percent_nadzor, oks_llr_percent, oks_llr_os_nadzor,
           oks_llr_perc_nadzor, oks_ecl_vbs_nadzor, oks_llr_os, oks_llr_perc, oks_diff_ecl_llr, oks_ecl_acc_kind,
           pr_name, oks_percent_norm, oks_percent_esp, c_name_fio, c_id, oks_is_refund, oks_is_refund_llr,
           oks_is_reschedule_llr, oks_is_zalog, oks_is_zalog_llr, oks_client_is_bankrupt, oks_zalog_is_bankrupt,
           oks_is_not_targeted, oks_no_document, oks_days_from_disb, k_fact_disb_online, oks_delinq_last_365,
           oks_delinq_field_30_1, oks_delinq_field_30_2, pr_code, cred_full_cost, k_kredline_end,
           kc_client_type, kc_is_msp, oks_kol_resh, oks_restruct_date
  do
  begin
    c_id_str = 'n' || c_id;
    rec_no = rec_no + 1;

    kol_resh_not_kk = null;
    first_resh_date_not_kk = null;
    first_resh_date_kk = null;
    kol_resh_kk = null;
    resh_kind_str = null;
    oks_last_resh_date = null;

    kol_resh_not_kk_str = null;-- 'Нет';
    if (oks_restruct_date is not null) then
    begin
      select kol_resh_not_kk,
             first_resh_date_not_kk,
             first_resh_date_kk,
             kol_resh_kk,
             resh_kind_str,
             last_resh_date_not_kk
      from hp_get_reschedule(:oks_from_kc_id, :rep_date)
      into :kol_resh_not_kk,
           :first_resh_date_not_kk,
           :first_resh_date_kk,
           :kol_resh_kk,
           :resh_kind_str,
           :oks_last_resh_date;

      if (kol_resh_not_kk >= 3) then kol_resh_not_kk_str = 'Да';
    end

    oks_last_resh_days = null;
    if (oks_last_resh_date is not null) then
      oks_last_resh_days = datediff(day, oks_last_resh_date, rep_date);

    oks_reserv_percent = maxvalue(oks_ecl_percent, oks_llr_percent_outer); -- Ставка резерва, новая, общая

    if (pr_code containing '.selfemployed.') then
    begin
      client_type = 'СЗ';
    end else
    begin
      kc_is_msp = coalesce(kc_is_msp, 0);

      if (kc_client_type = 1) then
      begin
        client_type = 'ЮЛ';
      end else
      if (kc_client_type = 2) then
      begin
        if (kc_is_msp = 1) then client_type = 'ИП-МСП';
          else client_type = 'ИП-неМСП';
      end else
      begin
        client_type = 'ФЛ';
      end
    end

    oks_pdn_koef = 0;
    select kc_pdn_koef
    from knclient_pdn
    where kc_id = :oks_from_kc_id and
          kcp_kind = 0
    into oks_pdn_koef;

    is_ref = 0;
    if (:rep_date >= '01.02.2025') then
      is_ref = f_get_is_ref(:k_id, :rep_date);

    if ((oks_is_refund = 1) or (is_ref = 1)) then
      oks_is_refund_str = 'Да';
    else
      oks_is_refund_str = 'Нет';

    if ((oks_is_refund_llr = 1) /*or (is_ref = 1)*/) then
      oks_is_refund_llr_str = 'Да';
    else
      oks_is_refund_llr_str = 'Нет';

    if (oks_is_reschedule_llr = 1) then
      oks_is_reschedule_llr_str = 'Да';
    else
      oks_is_reschedule_llr_str = 'Нет';

    if (oks_is_zalog = 1) then
      oks_is_zalog_str = 'Да';
    else
      oks_is_zalog_str = 'Нет';

    if (oks_is_zalog_llr = 1) then
      oks_is_zalog_llr_str = 'Да';
    else
      oks_is_zalog_llr_str = 'Нет';

    if (oks_zalog_is_bankrupt = 1) then
      oks_zalog_is_bankrupt_str = 'Да';
    else
      oks_zalog_is_bankrupt_str = 'Нет';

    if (oks_is_not_targeted = 1) then
      oks_is_not_targeted_str = 'Да';
    else
      oks_is_not_targeted_str = 'Нет';

    if (oks_no_document = 1) then
      oks_no_document_str = 'Да';
    else
      oks_no_document_str = 'Нет';

    if (oks_client_is_bankrupt = 1) then
      oks_client_is_bankrupt_str = 'Да';
    else
      oks_client_is_bankrupt_str = 'Нет';

    oks_client_type_str = 'ФЛ';
    msp_subject_str = 'Нет';

    -- Получение информации о банкротстве клиента
    brt_status = '';
    brt_status_date = null;
    if (oks_client_is_bankrupt = 1) then
    begin
      select status, status_date
      from bankrupt, hp_get_kontur_info(brt_from_eil_id)
      where brt_from_c_id = :c_id and
            :rep_date between brt_date_begin and brt_date_end
      into brt_status, brt_status_date;
    end

    -- Залогодатель
    brt_zalog_fio = '';
    c_id = null;
    select z_from_c_id
    from zalog
    where z_from_k_id = :k_id and
          :rep_date between coalesce(z_date_begin, '01.01.1900') and coalesce(z_date_end, '31.12.9999') and
          z_type = 1 and
          z_from_c_id is not null
    order by z_id desc
    rows 1
    into c_id;

    if (c_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :c_id
      into brt_zalog_fio;
    end

    -- Получение информации о банкротстве залогодателя
    brt_zalog_status = '';
    brt_zalog_status_date = null;
    if (oks_zalog_is_bankrupt = 1) then
    begin
      select status, status_date
      from bankrupt, hp_get_kontur_info(brt_from_eil_id)
      where brt_from_c_id = :c_id and
            :rep_date between brt_date_begin and brt_date_end
      into brt_zalog_status, brt_zalog_status_date;
    end
    oks_zalog_cost = 0;
    oks_zalog_cost = -f_get_acc_amount(:oks_from_kc_id, :sab_id_zalog_cost, :rep_date, :rep_date, 'bal'); -- Залоговая стоимоть

    cred_full_cost_resh = null;
    kont_kateg_resh = null;
    if (oks_last_resh_date is not null) then
    begin
      /*
      select kps_psk from kontpsk, operlist
      where kps_from_k_id = :k_id
      and kps_from_ol_id = ol_id
      and operlist.is_delete = 0
      and ol_date = :oks_last_resh_date
      into cred_full_cost_resh;
      */
      if (cred_full_cost_resh is null) then
      begin
        select effect_rate
        from hp_effect_rate_to_date(:k_id, :oks_last_resh_date, 1)
        into :cred_full_cost_resh;
      end

      k_period_day_resh = k_kredline_end - oks_disb_date;

      if (pr_code containing '.zalog_car.' or pr_code containing '.zalog.') then
      begin
        kont_kateg_resh = '2.1';
      end
      else
      if (cred_full_cost_resh is not null and
          coalesce(cred_full_cost_resh, 0) = 0) then
      begin
        kont_kateg_resh = '2.6.1';
      end
      else
      if (k_period_day_resh <= 30) then
      begin
        if (kc_amount_approved <= 30000) then
          kont_kateg_resh = '2.3.1.1';
        else
          kont_kateg_resh = '2.3.1.2';
      end
      else
      if (k_period_day_resh <= 60) then
      begin
        if (kc_amount_approved <= 30000) then
          kont_kateg_resh = '2.3.2.1';
        else
          kont_kateg_resh = '2.3.2.2';
      end
      else
      if (k_period_day_resh <= 180) then
      begin
        if (kc_amount_approved <= 30000) then
          kont_kateg_resh = '2.3.3.1';
        else
        if (kc_amount_approved <= 100000) then
          kont_kateg_resh = '2.3.3.2';
        else
          kont_kateg_resh = '2.3.3.3';
      end
      else
      if (k_period_day_resh <= 365) then
      begin
        if (kc_amount_approved <= 100000) then
          kont_kateg_resh = '2.3.4.1';
        else
          kont_kateg_resh = '2.3.4.2';
      end
      else
      if (k_period_day_resh <= 10000000) then
      begin
        if (kc_amount_approved <= 100000) then
          kont_kateg_resh = '2.3.5.1';
        else
          kont_kateg_resh = '2.3.5.2';
      end
    end

    if (pr_code containing '.zalog_car.' or pr_code containing '.zalog.') then
    begin
      kont_kateg = '2.1';
    end
    else
    if (cred_full_cost is not null and
        coalesce(cred_full_cost, 0) = 0) then
    begin
      kont_kateg = '2.6.1';
    end
    else
    if (k_period_day <= 30) then
    begin
      if (kc_amount_approved <= 30000) then
        kont_kateg = '2.3.1.1';
      else
        kont_kateg = '2.3.1.2';
    end
    else
    if (k_period_day <= 60) then
    begin
      if (kc_amount_approved <= 30000) then
        kont_kateg = '2.3.2.1';
      else
        kont_kateg = '2.3.2.2';
    end
    else
    if (k_period_day <= 180) then
    begin
      if (kc_amount_approved <= 30000) then
        kont_kateg = '2.3.3.1';
      else
      if (kc_amount_approved <= 100000) then
        kont_kateg = '2.3.3.2';
      else
        kont_kateg = '2.3.3.3';
    end
    else
    if (k_period_day <= 365) then
    begin
      if (kc_amount_approved <= 100000) then
        kont_kateg = '2.3.4.1';
      else
        kont_kateg = '2.3.4.2';
    end
    else
    if (k_period_day <= 10000000) then
    begin
      if (kc_amount_approved <= 100000) then
        kont_kateg = '2.3.5.1';
      else
        kont_kateg = '2.3.5.2';
    end

    if (oks_last_resh_date is null) then
    begin
      cred_full_cost_categ = cred_full_cost;
    end
    else
    if (kont_kateg = kont_kateg_resh) then
    begin
      cred_full_cost_categ = cred_full_cost;
    end
    else
      cred_full_cost_categ = cred_full_cost_resh;

    if (client_type in ('СЗ','ИП-МСП','ИП-неМСП')) then
    begin
      kont_kateg = null;
      kont_kateg_resh = null;
      cred_full_cost_categ = null;
    end 

    suspend;
  end
end
