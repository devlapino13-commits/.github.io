CREATE PROCEDURE AAREP_MATURITY_REPORT(
    REG_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    YEAR_NO INTEGER,
    MONTH_NO INTEGER,
    DIFF_NO INTEGER,
    IS_NEW SMALLINT,
    PLAN_OS DECIMAL(18,4),
    LLR_BALANCE DECIMAL(18,4),
    K_BALANCE DECIMAL(18,4),
    LLR_MINUS DECIMAL(18,4),
    REPAY_NAME VARCHAR(50),
    PLAN_PERC DECIMAL(18,4),
    MONTH_END DATE)
AS
declare variable tmp_llc decimal(18,4);
declare variable tmp_balance decimal(18,4);
declare variable k_balance_tmp_0 decimal(18,4);
declare variable k_balance_tmp_1 decimal(18,4);
declare variable llr_balance_0 decimal(18,4);
declare variable llr_balance_1 decimal(18,4);
begin
  for select rep_year, rep_month, rep_line, iif(k_circle <= 1, 0, 1), sum(sh_os), sum(sh_perc), sum(kont_llc_amount),
             sum(kont_balance), sum(kont_llc_minus)
      from aarep_maturity_rep_tmp(:rep_date)
      group by rep_year, rep_month, rep_line, 4
      order by 4, rep_line
      into :year_no, :month_no, :diff_no, :is_new, :plan_os, :plan_perc, :tmp_llc, :tmp_balance, :llr_minus
  do
  begin
    if (diff_no = -1) then
    begin
      if (is_new = 0) then
      begin
        repay_name = 'Показатели новых на начало';
        k_balance_tmp_0 = tmp_balance;
        llr_balance_0 = tmp_llc;
      end
      if (is_new = 1) then
      begin
        repay_name = 'Показатели повторных на начало';
        k_balance_tmp_1 = tmp_balance;
        llr_balance_1 = tmp_llc;
      end
      k_balance = tmp_balance;
      llr_balance = tmp_llc;
      year_no = null;
      month_no = null;
    end
    else
    begin
      k_balance = k_balance - plan_os;
      llr_balance = llr_balance - llr_minus;
    end

    if (diff_no = 0) then
      repay_name = 'Погашение просроченных';
    if (diff_no > 0) then
      repay_name = 'Через ' || diff_no || ' мес.';

    if (month_no < 12) then
      month_end = cast(('01.' ||(month_no + 1) || '.' || year_no) as date) - 1;
    else
      month_end = cast(('31.12.' || year_no) as date);
    suspend;
  end
  repay_name = 'Всего';
  k_balance = k_balance_tmp_0 + k_balance_tmp_1;
  llr_balance = llr_balance_0 + llr_balance_1;
  year_no = null;
  month_no = null;
  plan_os = null;
  plan_perc = null;
  diff_no = -1;
  month_end = null;
  suspend;
end
