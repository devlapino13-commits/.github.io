CREATE PROCEDURE HP_CHECK_DOUBLE_CREDIT(
    K_ID DECIMAL(18,0),
    K_IS_DISB INTEGER)
RETURNS (
    RES_NO INTEGER,
    KF_CLIENT_FIO VARCHAR(100),
    KF_NAME VARCHAR(100),
    KF_OFFICE VARCHAR(100),
    KF_KC VARCHAR(100),
    KF_DISB_DATE DATE,
    KF_CLIENT_PASS_NO VARCHAR(30))
AS
declare variable pr_can_disb_paral integer;
declare variable k_disb_date date;
declare variable c_id decimal(18,0);
declare variable kk_id decimal(18,0);
declare variable res_balance decimal(18,4);
declare variable res_last_date date;
declare variable is_found integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable db_id integer;
declare variable k_refund integer;
declare variable pr_code varchar(100);
declare variable pr_code_last varchar(100);
declare variable kc_amount_approved decimal(18,4);
begin
-- knclient_fix
-- Внимание, у этих клиентов имеется активный кредит. Выдача кредита отменяется.  
  if (K_ID = 101010054000300201) then exit;

  db_id = f_get_db_id();

  if (db_id = 94401) then
    exit; -- RFCAZ-285

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  select k_disb_date, pr_can_disb_paral, coalesce(k_refund, 0), pr_code
  from product, kontract
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into :k_disb_date, :pr_can_disb_paral, :k_refund, :pr_code;

  if (pr_can_disb_paral is null) then
    pr_can_disb_paral = 0;

  if (pr_can_disb_paral = 0 and
      k_refund = 0 and
      pr_code not containing '.EXPRESS.' and
      pr_code not containing '.food.' and
      pr_code not containing '.farmer.' and
      pr_code not containing '.float_percent.' and
      pr_code not containing '.express_online.' and
      pr_code not containing '.support.' and
      pr_code not containing '.summer_plus.' and
      pr_code not containing '.comfortable.' and
      pr_code not containing '.zalog.' and
      pr_code not containing '.selfemployed.' and
      pr_code not containing '.loyal.' and
      pr_code not containing '.reliable_client.' and
      pr_code not containing '.shans.' and
      pr_code not containing '.privilege_plus.'  and
      pr_code not containing '.universal.'       ---RFCRU-4991
      ) then
  begin
    res_no = 0;
    for select kc_from_c_id, c_name_fio, c_index_c_pass_no, kc_amount_approved
        from knclient, clients
        where kc_from_k_id = :k_id and
              kc_amount_approved is not null and
              kc_amount_approved <> 0 and
              kc_from_c_id = c_id
        into :c_id, :kf_client_fio, :kf_client_pass_no, kc_amount_approved
    do
    begin
      is_found = 0;
      for select c_id
          from clients
--      where (c_index_c_name_fio starting with :kf_client_fio or :kf_client_fio like c_index_c_name_fio||'%')
          where (c_name_fio starting with :kf_client_fio or
                :kf_client_fio like c_name_fio || '%') and
                c_index_c_pass_no = :kf_client_pass_no
          into :c_id
      do
      begin
        if (is_found = 0) then
        begin
          if (pr_can_disb_paral = 0) then
          begin
            for select distinct reg_code || ' ' || reg_name, kc.c_name_fio, kal_new_kf_name, k_id, ks_date_begin,
                                pr_code
                from kontstate, knclient, knaktlist, kontract, region, clients kc, product pr
                where ks_from_k_id <> :k_id and
                      kc_from_c_id = :c_id and
                      kc_from_k_id = ks_from_k_id and
                      cast(ks_date_begin as date) <= :k_disb_date and
                      ks_date_end >= :k_disb_date and
                      ks_state = 1 and
                      ks_ol_is_delete = 0 and
                      k_from_pr_id = pr_id and
                      pr_category = 0 and
                      k_id = ks_from_k_id and
                      kal_from_kf_id = k_from_kf_id and
                      kal_date <= :k_disb_date and
                      kal_date_end >= :k_disb_date and
                      knaktlist.is_delete = 0 and
                      kal_new_reg_id = reg_unicode and
                      kal_new_kf_officer = kc.c_id and
                      kc_amount_approved is not null and
                      kc_amount_approved <> 0 and
                      pr.pr_code not containing ('.EXPRESS.') and
                      pr.pr_code not containing ('.FOOD.') and
                      pr.pr_code not containing ('.float_percent.') and
                      pr_code not containing '.summer_plus.' and
                      pr_code not containing '.comfortable.' and
                      pr_code not containing '.zalog.' and
                      pr_code not containing '.shans.'
                into :kf_office, :kf_kc, :kf_name, :kk_id, :kf_disb_date, :pr_code_last
            do
              if (not(:kc_amount_approved = 15000 and
                  :pr_code containing '.BARAKAT.' and
                  :pr_code_last containing '.patent.')) then --Условие добавлено по запросу Зары 10.04.18
              begin
                res_balance = 0;
                is_found = 1;
                if (k_is_disb = 0) then
                begin
                  res_last_date = null;
                  res_balance = 0;
                  select ok_sh_date, sum(ok_os + ok_sh_os)
                  from operkred
                  where ok_from_k_id = :kk_id and
                        ok_ol_is_active = 1 and
                        ok_type < 8 and
                        ok_from_c_id = 0
                  group by ok_sh_date
                  having sum(ok_os + ok_sh_os) <> 0
                  order by ok_sh_date desc
                  rows 1
                  into :res_last_date, :res_balance;
                  if (k_disb_date <> res_last_date) then
                  begin
                    res_no = res_no + 1;
                    if (not(db_id = 94401 and
                        res_no = 1 and
                        kc_amount_approved + res_balance < 3000)) then
                      suspend;
                  end
                end
                else
                begin
                  res_balance = 0;
                  select sum(ok_os + ok_sh_os)
                  from operkred
                  where ok_from_k_id = :kk_id and
                        ok_ol_is_active = 1 and
                        ok_type < 8 and
                        ok_from_c_id = 0 having sum(ok_os + ok_sh_os) > 0
                  into :res_balance;
                  if (res_balance > 0) then
                  begin
                    res_no = res_no + 1;
                    if (not(db_id = 94401 and
                        res_no = 1 and
                        kc_amount_approved + res_balance < 3000)) then
                      suspend;
                  end
                end
              end
          end
          if (pr_can_disb_paral = 2) then
          begin
            for select distinct reg_code || ' ' || reg_name, kc.c_name_fio, kal_new_kf_name, k_id, ks_date_begin
                from kontstate, knclient, knaktlist, kontract, region, clients kc, product pr
                where ks_from_k_id <> :k_id and
                      kc_from_c_id = :c_id and
                      kc_from_k_id = ks_from_k_id and
                      cast(ks_date_begin as date) <= :k_disb_date and
                      ks_date_end >= :k_disb_date and
                      ks_state <> 0 and
                      k_from_pr_id = pr_id and
                      pr_category = 0 and
                      ks_ol_is_delete = 0 and
                      k_id = ks_from_k_id and
                      kal_from_kf_id = k_from_kf_id and
                      kal_date <= :k_disb_date and
                      kal_date_end >= :k_disb_date and
                      knaktlist.is_delete = 0 and
                      kal_new_reg_id = reg_unicode and
                      kal_new_kf_officer = kc.c_id and
                      kc_amount_approved is not null and
                      kc_amount_approved <> 0 and
                      pr.pr_code not containing ('.EXPRESS.')  and
                      pr.pr_code not containing ('.float_percent.')
                order by ks_date_begin desc
                rows 2
                into :kf_office, :kf_kc, :kf_name, :kk_id, :kf_disb_date
            do
            begin
              /* Поиск погашений с просрочками более 7 дней */
              if (exists(select *
                         from operkred
                         where ok_ol_is_active = 1 and
                               ok_from_k_id = :kk_id and
                               ok_type = 4 and
                               ok_from_c_id = 0 and
                               (ok_os + ok_perc + ok_fine) < 0 and
                               (ok_ol_date - ok_sh_date) > 7)) then
              begin
                res_balance = 0;
                is_found = 1;
                res_no = res_no + 1;
                if (not(db_id = 94401 and
                    res_no = 1 and
                    kc_amount_approved + res_balance < 3000)) then
                  suspend;
              end
              if (is_found = 0) then
              begin
                /* поиск активных просрочек */
                for select reg_code || ' ' || reg_name, kc.c_name_fio, kal_new_kf_name, k_id, ks_date_begin
                    from kontstate, knclient, knaktlist, kontract, region, clients kc
                    where ks_from_k_id <> :k_id and
                          kc_from_c_id = :c_id and
                          kc_from_k_id = ks_from_k_id and
                          cast(ks_date_begin as date) <= :k_disb_date and
                          ks_date_end >= :k_disb_date and
                          ks_state = 1 and
                          ks_ol_is_delete = 0 and
                          k_id = ks_from_k_id and
                          kal_from_kf_id = k_from_kf_id and
                          kal_date <= :k_disb_date and
                          kal_date_end >= :k_disb_date and
                          knaktlist.is_delete = 0 and
                          kal_new_reg_id = reg_unicode and
                          kal_new_kf_officer = kc.c_id and
                          kc_amount_approved is not null and
                          kc_amount_approved <> 0
                    into :kf_office, :kf_kc, :kf_name, :kk_id, :kf_disb_date
                do
                begin
                  if (exists(select ok_sh_date, sum(ok_os + ok_sh_os)
                             from operkred
                             where ok_from_k_id = :kk_id and
                                   ok_ol_is_active = 1 and
                                   ok_type < 8 and
                                   ok_ol_date <= :k_disb_date and
                                   ok_sh_date <= :k_disb_date and
                                   ok_from_c_id = 0
                             group by ok_sh_date
                             having sum(ok_os + ok_sh_os) > 0)) then
                  begin
                    res_balance = 0;
                    is_found = 1;
                    res_no = res_no + 1;
                    if (not(db_id = 94401 and
                        res_no = 1 and
                        kc_amount_approved + res_balance < 3000)) then
                      suspend;
                  end
                end
              end
            end
          end
        end /* is_found=0 */
      end
    end
  end
end
