CREATE PROCEDURE PROC_CALC_AMORT_FULL(
    REG_ID INTEGER,
    YEAR_NO INTEGER,
    MONTH_NO INTEGER,
    DD DATE)

    PCA_BILL_ID DECIMAL(18,0),
    PCA_BILL_LINE_NO INTEGER,
    PCA_START_YEAR INTEGER,
    PCA_START_MONTH INTEGER,
    PCA_YEAR_NO INTEGER,

    PCA_KOEF INTEGER,
    PCA_AGE INTEGER,
    PCA_TOTAL_OS DECIMAL(18,4),
    PCA_TOTAL_AMORT DECIMAL(18,4),
    PCA_LIFETIME INTEGER,
    PCA_CALC_TYPE SMALLINT,

    PCA_INVENT_STR VARCHAR(20),
    PCA_REGION_ID INTEGER)
AS
begin
  /* Procedure Text */
  for select pca_bill_id, pca_bill_line_no, pca_start_year, pca_start_month, pca_year_no, pca_month_no, pca_new_amort,

             mbal_ml_id, mbal_mc_id, mbal_invent_str, mbal_region_id
      from proc_calc_amort(:reg_id, :year_no, :month_no), matlist_balance_list(:reg_id, 0, :dd, 0, 1)
      where pca_bill_id = mbal_bill_id and

      into :pca_bill_id, :pca_bill_line_no, :pca_start_year, :pca_start_month, :pca_year_no, :pca_month_no,
           :pca_new_amort, :pca_koef, :pca_age, :pca_total_os, :pca_total_amort, :pca_lifetime, :pca_calc_type,
           :pca_wp_id, :pca_mo_id, :pca_ml_id, :pca_mc_id, :pca_invent_str, :pca_region_id
