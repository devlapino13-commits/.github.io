CREATE PROCEDURE HP_EDOC_AFTER_SIGN_ACTION(
    EDL_ID D_ID NOT NULL)
AS
declare variable EDL_STATUS D_SMALLINT;
declare variable EDL_OBJECT_KIND D_INTEGER;
declare variable EDL_OBJECT_ID D_ID;
declare variable EDOC_KIND D_STR_CODE;
declare variable OBJ_KIND D_STR_CODE;
declare variable DB_ID D_INTEGER;
declare variable REG_ID D_INTEGER;
declare variable UN_ID D_ID;
declare variable C_ID D_ID;
declare variable EDS_ID D_ID;
declare variable MSG_ID D_INTEGER;
declare variable CLT_ID D_INTEGER;
declare variable CDA_ID D_ID;
declare variable EDL_CREATE_UN_ID D_ID;
declare variable ERR_TEXT D_STR_255;
declare variable ARC_ID D_INTEGER;
declare variable EDC_MSG_KIND D_SMALLINT;
declare variable OVS_ID D_INTEGER;
declare variable OPR_ID ID_TYPE;
declare variable MAC_AMOUNT CURR_TYPE;
declare variable LANG D_STR_LANG;
declare variable EDF_CODE D_STR_CODE;
declare variable MA_AMOUNT D_CURRENCY;
declare variable MA_ID D_INTEGER;
declare variable FINCERT_RES D_SMALLINT;
declare variable PR_CODE D_STR_100;
declare variable EDC_NAME D_STR_100;
begin
  -- Процедура выполняет действия после подписания электронного документа.
  -- Вызывается из триггера eDoc_list after update

  select edl_status, edl_object_kind, edl_object_id, edc_code, edc_msg_kind, edl_create_un_id, edc_name
  from edoc_list, edoc_cfg
  where edl_id = :edl_id and
        edl_from_edc_id = edc_id
  into edl_status, edl_object_kind, edl_object_id, edoc_kind, edc_msg_kind, edl_create_un_id, edc_name;

  if (row_count = 0) then
    exception e_default f_format('eDoc_List record not found, edl_id = %1', edl_id);

  if (edl_status <> 1) then
    exception e_default f_format('eDoc_List not signed, edl_id = %1', edl_id);

  db_id = f_get_db_id();
  obj_kind = f_objkind(edl_object_kind, 'name');

  if (obj_kind = 'agreements') then
  begin
    update agreements
    set ag_status = 1
    where ag_id = :edl_object_id and
          ag_status <> 1;
  end
  
  if (obj_kind = 'kontract') then
  begin
    select pr.pr_code
    from kontract k
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where k.k_id = :edl_object_id
    into pr_code;
  end

  if (db_id = 81001 ) then
  begin
    -- RFCRU-5337 Записываем состояние перс данных клиента, для того, чтобы иметь возможность удалить их
    -- Примечание: Если по одному клиенту создались дубликаты, то они просто не будут обработаны должным образом
    for select eds.eds_from_c_id
    from edoc_sign eds
    where eds.eds_from_edl_id = :edl_id
        and eds.eds_from_c_id is not null
        and eds.eds_status = 1
    into :c_id
    do
    begin
      if (coalesce(c_id, 0) > 0 and edoc_kind = 'asp.agreement') then
        execute procedure hp_clientstate_order_create(:c_id, 'csr_edoc_asp_signed', 1);

      if (coalesce(c_id, 0) > 0 and (lower(edc_name) containing 'согласие' or lower(edc_name) containing 'соглашение')) then
        execute procedure hp_clientstate_order_create(:c_id, 'csr_edoc_agreement', 1);
    end

    -- RFCRU-4632 обновляем статутс, только у документа, который был создан f_enum_id('mc_phone_change', 'created')
    if (edoc_kind = 'cell.application.no_access') then
      update mc_phone_change
      set mpc_status = f_enum_id('mc_phone_change', 'signed')
      where mpc_from_edl_id = :edl_id
        and mpc_status = f_enum_id('mc_phone_change', 'created');
  end

  -- После подписание eDoc_List, обновляем статус контактных данных
  for select edm_from_eds_id, edm_from_msg_id, edm_from_clt_id, edm_from_cda_id, eds_from_c_id
      from edoc_sign, edoc_msg
      where eds_from_edl_id = :edl_id and
            eds_status = 1 and
            edm_from_eds_id = eds_id and
            edm_status = 1 and
            eds_from_c_id is not null
      into eds_id, msg_id, clt_id, cda_id, c_id
  do
  begin
    if (edoc_kind in ('asp.agreement', 'cell.application', 'asp.agreement_employee')) then
    begin
      execute procedure hp_contveri_sign_sync('edoc_sign', :eds_id, :msg_id, 'edoc');
      if (db_id in (39801, 41701)) then
      begin
        if (exists(select *
                   from cardclientid
                   where cci_from_c_id = :edl_object_id)) then
          err_text = (select err_text
                      from hp_cardclient_update_customer(:edl_object_id, null, 0, :edl_create_un_id));
      end
    end
    execute procedure hp_contdata_date_update(:eds_id, :cda_id, :clt_id);

    if (edoc_kind in ('cell.application.kmp')) then
    begin
      cda_id = 0;
      select cda_id
      from contdata cda
      inner join contlist clt on clt.clt_id = cda.cda_from_clt_id
      where cda_from_c_id = :c_id and
            cda_cont_kind = f_enum('cont_kind', 'cell', 'id') and
            cda_is_delete = 0 and
            cda_status = 0 and
            cda_is_main = 1
      order by cda_id desc
      rows 1
      into :cda_id;

      if (cda_id > 0) then
        -- верифицирует номер
        execute procedure hp_contveri_set_ok(:cda_id, 1000000000000, current_timestamp);

      opr_id = 0;
      ovs_id = 0;
      select opr_id, ovs_id
      from objectvalues, objectprops
      where opr_object_kind = f_objkind('clients', 'id') and
            opr_object_id = :c_id and
            opr_subject_kind = f_objkind('refdata', 'id') and
            opr_subject_id = f_get_rd_id('ContactInfo.Cell_Main') and
            ovs_from_opr_id = opr_id and
            ovs_id = opr_actual_ovs_id
      into :opr_id, ovs_id;

      if (opr_id > 0) then
        update objectvalues
        set ovs_add_value = 2
        where (ovs_from_opr_id = :opr_id) and
              (ovs_id = :ovs_id);

    end
  end

  if (edoc_kind = 'probationer_kontract' and
      db_id in (39801, 41701)) then
  begin
    execute procedure hp_edoc_send_push_intern(:edl_object_id);
  end
  if (edoc_kind = 'loan.application') then -- Отключен до исключения MINIAPP_VERIFY
  begin
    if (db_id = 81001) then
    begin
      execute procedure hp_extinfo_insert_bki('miniapp', :edl_object_id, 1, edl_create_un_id);
      execute procedure hp_extinfo_insert_kontur('miniapp', :edl_object_id, 1, edl_create_un_id);
      execute procedure hp_extinfo_insert('ligres_fssp_cli_check', 'miniapp', :edl_object_id, 0, :un_id);
      if (f_get_const('selfemployed_active ') <> 1) then
        execute procedure hp_extinfo_insert('inn_nalog', 'miniapp', :edl_object_id, 0, edl_create_un_id);
    end

    -- RFCKG-5208: создание инфо-листа для ИПФ/STANDART
    if (db_id = 41701) then
      execute procedure hp_edoc_infolist_create(:edl_object_id, :edl_create_un_id);

    -- Отправляем участников на проверку КК
    select ma_office_id, ma_from_un_id
    from miniapp
    where ma_id = :edl_object_id and
          ma_is_active = 1
    order by ma_number desc
    rows 1
    into reg_id, un_id;

    for select mac_from_c_id, mac_amount
        from miniappclient
        where mac_from_ma_id = :edl_object_id and
              mac_amount > 0
        into :c_id, mac_amount
    do
    begin
      execute procedure hp_actionlist_create_check_info(:c_id, :reg_id, 0, :un_id);

      if (db_id = 81001) then
      begin
        execute procedure hp_extinfo_insert_smevpass(:c_id, 1, :un_id);

        lang = f_get_lang_code(null, c_id);
        execute procedure mc_send_push('clients', c_id,
          f_lang(lang, 'Создана кредитная заявка'),
          f_format(f_lang(lang, 'Создана кредитная заявка. Ваша сумма: %1 %2'), round(mac_amount, 2), 'руб.'));
      end
    end
  end

  if (edoc_kind in ('bankcard.mbulak', 'bankcard_pin.mbulak')) then
  begin
    -- установка статуса "Акт подписан"
    update cardaktclient
    set cac_status = 1
    where cac_id = :edl_object_id and
          cac_status in (f_enum_id('cardaktclient', 'created'), f_enum_id('cardaktclient', 'issued_delayed'));
  end
  if (obj_kind = 'cardblockopers') then
  begin
    update cardblockopers
    set cbo_status = 2
    where cbo_id = :edl_object_id;
  end
  if (obj_kind = 'cardpinchange') then
  begin
    if (edc_msg_kind <> 3) then
      update cardpinchange
      set cph_status = 2
      where cph_id = :edl_object_id;
  end

  if (obj_kind = 'app_return_consult') then
  begin
    select arci.arci_id, arc.arc_status_un_id
    from app_return_consult_info arci
    inner join app_return_consult arc on arci.arci_id = arc.arc_id
    where arci.arci_from_edl_id = :edl_id
    into :arc_id, :un_id;

    if (:arc_id is not null) then
      update app_return_consult
      set arc_status = f_enum('app_return_consult', 'approved', 'id'),
          arc_status_un_id = :un_id
      where (arc_id = :arc_id);
  end

  if (db_id in (39801, 41701) and
      edoc_kind = 'loan.agreement') then
  begin
  if (db_id = 41701) then
  begin
    select res_text
    from hp_auto_disb_ipf(:edl_object_id)
    into err_text;

    if (err_text <> '') then
      exception e_message(err_text);
   end
   
   if (db_id = 41701 and
       edoc_kind = 'loan.agreement' and
       pr_code containing '.insur_base.') then  -- RFCKG-4831
   begin
      -- если это основной займ то создаем заем страхование
      execute procedure hp_insert_insurance_kontract(:edl_object_id, :edl_create_un_id);
   end
   
   if (db_id = 39801) then
      execute procedure cnp_request_ins(:edl_object_id, :edl_create_un_id);
  end

  if (db_id = 39801 and
      edoc_kind = 'loan_disb_consent') then
  begin
    execute procedure cnp_request_ins(:edl_object_id, :edl_create_un_id);
  end
  
  if (db_id = 39801 and
      edoc_kind = 'restruct_application') then
  begin
   execute procedure hp_extinfo_insert('restruct_schedule', 'kontract', :edl_object_id, 1, :edl_create_un_id);
  end

  if (db_id = 39801 and
      edoc_kind = 'add_loan_consent') then
  begin
   execute procedure hp_do_kont_for_restruct_save(:edl_object_id);
  end

  if (db_id in (39801, 41701) and
      edoc_kind = 'idjara_goods.contragent_act') then
  begin
    execute procedure hp_create_oper_idjara_goods(:edl_object_id);
  end 

  if (edoc_kind = 'overpayment.phone') then
    err_text = (select err_text
                from hp_overpayment_setstatus(:edl_object_id, 2, :edl_create_un_id));

  if (edoc_kind = 'opercash.book') then
    execute procedure hp_opercash_do_edoc_sign(:edl_id);

  if (edoc_kind = 'act_TMC.agreement') then
    execute procedure hp_transfertmc_sign(abs(:edl_object_id));

  if (edoc_kind = 'fuelgenreg.agreement') then
    execute procedure hp_fuelgenreg_sign(:edl_object_id);

  if (edoc_kind = 'accesss.cashroom') then
    err_text = (select err_text from hp_accesscashroom_sign(:edl_object_id));

  if (edoc_kind = 'cash.inventory_act') then
    err_text = (select err_text from hp_cashcheck_sign(:edl_object_id));

  if (edoc_kind = 'hr.general_doc') then
    update signdoc set
      sgn_status = 3
    where sgn_id = :edl_object_id;

  -- отправляем push уведомление если подписан вручную (имеются сканы)
  execute procedure hp_edoc_send_push(:edl_id);

  if (db_id = 81001 and edoc_kind = 'loan.application') then
  begin
    -- если это заявка, запишем в буфер для отправки в БКИ
    execute procedure hp_bki_add_to_buffer('.hp_edoc_after_sign_action.', 'miniapp', :edl_object_id, current_date);
  end

  if (edoc_kind = 'inventory.inner_act') then
  begin
    update invmember imem
    set imem.imem_sign_status = 1,
        imem.imem_sign_type = 1
    where imem.imem_id = :edl_object_id;
  end

  -- после подписание клиента на измения услоия займа
  -- отказ в выдаче
  if (edoc_kind = 'loan.cancel') then
  begin
    select edl_create_un_id
    from edoc_list
    where edl_id = :edl_id
    into :un_id;
    execute procedure hp_cancel_kontract(:edl_object_id, :un_id);
  end

  if (db_id = 81001 and edoc_kind in ('loan.agreement', 'loan.application', 'ad.agreement', 'pdata.agreement')) then
  begin
    -- Если подписаны бланки:
    -- Согласие на обработку персональных данных клиента (pdata.agreement, ФЗ-152)
    -- Согласие на получение рекламных сообщений и инфо (ad.agreement, ФЗ-38)
    -- Снимаем ограничения по соответствующим ФЗ

    for select eds_from_c_id, edf_code
        from edoc_blank, edoc_frx, edoc_sign
        where edb_from_edl_id = :edl_id
        and edb_from_edf_id = edf_id
        and eds_from_edl_id = :edl_id
        and (edb_from_c_id = eds_from_c_id or edb_from_c_id is null) -- Общий или персональный бланк
        and edb_sign_status = 1
        and edf_code in ('pdata.agreement', 'ad.agreement')
    into c_id, edf_code
    do
    begin
      if (edf_code = 'ad.agreement') then
        execute procedure hp_clientcomreg_set(:c_id, null, 'remove', 1, 0, 0, 'edoc_signed');
      else if (edf_code = 'pdata.agreement') then
        execute procedure hp_clientcomreg_set(:c_id, null, 'remove', 0, 1, 0, 'edoc_signed');
    end
  end

  if (db_id = 81001 and edoc_kind = 'loan.agreement' and pr_code containing '.business.') then
  begin
    -- RFCRU-5444 по ТЗ из RFCRU-5160
    update knclient kc
    set kc.kc_is_msp = coalesce((select is_msp from hp_kontur_ip_get_la_info('kontract', :edl_object_id)), 0)
    where kc_from_k_id = :edl_object_id;
  end

  if (db_id = 81001 and edoc_kind = 'loan.pdn') then
    execute procedure hp_send_edoc_pack_ru(:edl_object_id, 'loan.agreement_send');

  if (db_id = 81001 and edoc_kind = 'overpay_active.app') then
    update overpay_main
    set opm_status = 2
    where opm_id = :edl_object_id;

  -- RFCKZ-494
  if (db_id = 39801 and edoc_kind = 'overpay_active') then
    update overpay_main
    set opm_status = 2
    where opm_id = :edl_object_id;

  if (db_id = 41701 and edoc_kind = 'insurance.application') then
    update insurance_kontract ik
    set ik.ik_enum_status = f_enum_id('insurance_kontract', 'edoc_signed')
    where ik.ik_from_k_id = :edl_object_id;

  if (db_id = 39801) then
    begin
       if (edoc_kind = 'loan.application') then
         begin
           select mac.mac_from_c_id, mac.mac_amount
           from miniappclient mac
           where mac.mac_from_ma_id = :edl_object_id
                and mac.mac_amount > 0
           into :c_id, :ma_amount;

           -- pmi_status = 1 (при подписании документа заявки на займ)
           -- pmi_rep_type = 2 (используем 2 только в первом запросе(после подписания заявки на займ)) RFCKG-4065
           if (c_id is not null) then
           begin
             insert into pkb_miniapp_info(pmi_from_ma_id, pmi_from_c_id, pmi_create_datetime, pmi_status, pmi_is_sent, pmi_sum, pmi_rep_type, pmi_project)
             values (:edl_object_id, :c_id, current_timestamp, 1, 0, :ma_amount, 2, f_get_rd_id('cib.pkb'));

             insert into pkb_miniapp_info(pmi_from_ma_id, pmi_from_c_id, pmi_create_datetime, pmi_status, pmi_is_sent, pmi_sum, pmi_rep_type, pmi_project)
             values (:edl_object_id, :c_id, current_timestamp, 1, 0, :ma_amount, 2, f_get_rd_id('cib.gkb'));
           end
            -- RFCKZ-659: Таймер 58 минут — старт после подписания заявки клиентом
           execute procedure hp_ma_timer_kz_start(:edl_object_id);
         end
       else if (edoc_kind = 'pers_data_consent') then
         begin
            --RFCKG-4485
            execute procedure hp_extinfo_insert('pkb_available_reports', f_objkind('miniapp', 'id'), :edl_object_id, 1, 1000000000000);

            /*
                RFCKG-4392
                Когда клиент подписывает согласие на обработку персональных данных, отправляем запрос к сервису стоп-кредит
            */
            execute procedure hp_extinfo_insert('stop_credit', f_objkind('miniapp', 'id'), :edl_object_id, 1, 1000000000000);

            -- RFCKZ-996 автозапрос статуса СУСН
            err_text = (select res_text
                        from hp_susn_query(:edl_object_id, :edl_create_un_id));
        end
    end 

end
