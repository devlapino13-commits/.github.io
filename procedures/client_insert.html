CREATE PROCEDURE CLIENT_INSERT(
    C_FROM_REG_ID INTEGER,
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_INN VARCHAR(20),
    C_CITIZEN INTEGER,
    C_DOC_TYPE INTEGER,
    C_FROM_UN_ID DECIMAL(18,0))
RETURNS (
    C_ID DECIMAL(18,0))
AS
declare variable n integer;
declare variable reg integer;
declare variable cr_dd_nation integer;
declare variable cr_dd_family integer;
declare variable cr_dd_education integer;
declare variable cr_dd_army integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select result
  from get_kv_id(2)
  into :c_id;

  select cr_dd_nation, cr_dd_family, cr_dd_education, cr_dd_army
  from curruserdata
  where cr_from_u_id = :c_from_un_id rows 1
  into :cr_dd_nation, :cr_dd_family, :cr_dd_education, :cr_dd_army;

  insert into clients (c_id, c_type, c_name_fio, c_sex, c_birth_date, c_nation, c_citizen, c_pass_no, c_from_reg_id,
                       c_family, c_education, c_kol_family, c_inn, c_is_resident, c_army, c_doc_type, c_from_un_id,
                       c_kol_child)
  values (:c_id, 0, :c_name_fio, 0, :c_birth_date, :cr_dd_nation, :c_citizen, :c_pass_no, :c_from_reg_id, :cr_dd_family,
          :cr_dd_education, 1, :c_inn, 1, :cr_dd_army, :c_doc_type, :c_from_un_id, 0);
end
