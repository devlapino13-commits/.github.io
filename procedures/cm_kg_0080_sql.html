CREATE PROCEDURE CM_KG_0080_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable MA_OFFICE_ID integer;
declare variable PR_CODE STR_100;
declare variable C_ID ID_TYPE;
declare variable MAC_AMOUNT CURR_TYPE;
declare variable C_NAME_FIO D_STR_100;
declare variable MA_DATE_OPER D_DATE;
declare variable MA_SUMMA_MINUS CURR_TYPE;
begin

  for select ma_office_id, pr_code, mac.mac_amount, mac.mac_from_c_id, ma.ma_date_oper, ma.ma_summa_minus
      from miniapp ma
      inner join product pr on pr.pr_id = ma.ma_from_pr_id
      inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
      where ma_id = :ma_id and
            ma_is_active = 1 and
            ma_date_end = '31.12.9999' and
            mac.mac_amount > 0
      into :ma_office_id, :pr_code, :mac_amount, :c_id, :ma_date_oper, :ma_summa_minus
  do
  begin
    /* if (ma_office_id in (1130200) and

    begin
      res_error_text = 'В текущем офисе приостановлены выдачи.';
      res_error_is_critical = 2;

        pr_code not like '%.IND.%' and
        ma_summa_minus > 300000 and
        ma_date_oper < '12.06.2022') then
    begin
      res_error_text = 'Сумма займа не может превышать 300000 сомов.';
      res_error_is_critical = 2;
      suspend;
    end
    else
    if ((ma_office_id = 1040800 and
        mac_amount > 500000 and
        ma_date_oper >= '12.06.2022')) then
    begin
      -- RFCKG-3798 увеличил сумму до 500тыс.

      c_name_fio = null;
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into c_name_fio;

      res_error_text = 'Сумма займа не может превышать 300000 сомов. ФИО клиента ' || c_name_fio;
      res_error_is_critical = 2;
      suspend;
    end
    -- убрала ma_office_id 1040800 основание письмо от Улана письмо от 15.01.2021 вр 16:20
/*    if (ma_office_id = 1130200) then
