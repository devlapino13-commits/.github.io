CREATE PROCEDURE AAREP_GET_CERT_PERIOD_TMP(
    REP_OFFICE_ID INTEGER,
    REP_DATE DATE,
    REP_DELTA INTEGER,
    REP_MA_ID INTEGER)
RETURNS (
    REP_TIME_VALUE DECIMAL(18,4),
    REP_GROUP_COUNT INTEGER,
    REP_MEMO VARCHAR(255),
    REP_SELECTED INTEGER)
AS
declare variable ma_id integer;
declare variable ma_from_kf_id decimal(18,0);
declare variable ma_time integer;
declare variable tmp_hour integer;
declare variable tmp_min integer;
declare variable ma_plan_disb_time timestamp;
begin
-- Возвращает время сертификаций в офисе на определенную дату

  rep_selected = 0;

  rep_time_value = 0;
  rep_group_count = 0;
  rep_memo = '';
  suspend;

  rep_time_value = 7.0;
  while (rep_time_value <= 21.00) do
  begin
    suspend;
    rep_time_value = rep_time_value + cast(rep_delta as decimal(15,2)) / 100;
    if (mod((rep_time_value * 100), 100) >= 60) then
      rep_time_value = round(rep_time_value);
  end
  rep_time_value = 0;

  rep_group_count = 1;

  for select ma_id, ma_from_kf_id, ma_plan_disb_time
      from kontract, kontstate, miniapp
      where ks_region_id = :rep_office_id and
            ks_from_k_id = k_id and
            ks_state = 0 and
            ks_date_begin <= ks_date_end and
            ks_date_end = '31.12.9999' and
            ks_ol_is_delete = 0 and
            k_status >= 0 and
            k_disb_date = :rep_date and
            k_from_ma_app_id = ma_id and
            ma_date_oper <> :rep_date and
            ma_date_end = '31.12.9999' and
            ma_is_active = 1
      union
      select ma_id, ma_from_kf_id, ma_plan_disb_time
      from miniapp
      where ma_office_id = :rep_office_id and
            ma_date_oper = :rep_date and
            ma_date_end = '31.12.9999' and
            ma_is_active = 1
      into :ma_id, :ma_from_kf_id, :ma_plan_disb_time
  do
  begin
    rep_selected = 0;
    tmp_hour = 0;
    tmp_min = 0;
    rep_time_value = 0;
    if (ma_plan_disb_time is not null) then
    begin
      if (cast(ma_plan_disb_time as date) <> cast(rep_date as date)) then
        ma_plan_disb_time = cast(rep_date as date);
      tmp_hour = extract(hour from ma_plan_disb_time);
      tmp_min = extract(minute from ma_plan_disb_time);
      if (tmp_hour >= 7 and
          tmp_hour <= 21) then
      begin
        tmp_min = tmp_min / rep_delta;
        tmp_min = tmp_min * rep_delta;
        rep_time_value = tmp_hour + cast(tmp_min as decimal(15,2)) / 100;
      end
    end

    select kal_new_kf_name
    from knaktlist
    where kal_from_kf_id = :ma_from_kf_id and
          kal_date_end >= '31.12.2999' and
          is_delete = 0
    into :rep_memo;

    if (ma_id = rep_ma_id) then
      rep_selected = 1;
    suspend;
  end
  rep_group_count = 0;

end
