CREATE PROCEDURE AAREP_ACTIVE_KONT_ID_IN_PERIOD(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_NEED_ORDER_BY_OFFICE INTEGER,
    REP_NEED_JUST_DELINQ INTEGER,
    REP_NEED_CURR_RATE INTEGER,
    REP_NEED_DELINQ_OS INTEGER,
    REP_NEED_DELINQ_FINE INTEGER)
RETURNS (
    KK_ID DECIMAL(18,0),
    KK_REGION_ID INTEGER,
    KK_KC_ID DECIMAL(18,0),
    KK_CLOSE_DATE DATE,
    KK_DELINQ_DATE DATE,
    KK_DELINQ_DAYS INTEGER,
    KK_BALANCE DECIMAL(18,4),
    KK_IS_DELINQ INTEGER,
    KK_DELINQ_BALANCE DECIMAL(18,4),
    KK_DELINQ_OS DECIMAL(18,4),
    KK_DELINQ_PERC DECIMAL(18,4),
    KK_DELINQ_FINE DECIMAL(18,4),
    KK_CURR_RATE DECIMAL(18,4),
    KK_CURRENCY VARCHAR(3),
    KK_FROM_PR_ID INTEGER,
    KK_DISB_DATE DATE,
    KK_DISB_REG_ID INTEGER,
    KK_DISB_KC_ID DECIMAL(18,0),
    KK_AMOUNT_APPROVED DECIMAL(18,4),
    KK_DISB_CURR_RATE DECIMAL(18,4),
    KK_FROM_KF_ID DECIMAL(18,0),
    KK_KOLICH_CLIENT INTEGER,
    KK_KONTRACT_NO VARCHAR(20),
    KK_PERCENT_PER_DAY DECIMAL(18,7),
    KK_STOP_PERC DATE,
    KK_KREDLINE_END DATE,
    KK_ZERO_BALANCE_DATE DATE)
AS
declare variable def_curr varchar(3);
declare variable k_index_curr_rate decimal(18,4);
declare variable pr_id_prior integer;
declare variable pr_set_rate_in_kont integer;
begin
-- Вытаскивает список всех активных контрактов на конечную дату и закрытых за период
-- Если Rep_Date_end пусто, то вытащит только активных на конецную дату

  if (rep_need_just_delinq = 1) then
    rep_date_begin = null; -- чтоб закрытых кредитов не вытаскивал

  kk_delinq_os = 0;
  kk_delinq_perc = 0;
  kk_delinq_fine = 0;
  kk_curr_rate = 1;

  if (rep_need_curr_rate = 1) then
  begin
    select cm_default_currency
    from company
    into :def_curr;
  end
  pr_id_prior = 0;

  kk_close_date = null;
  kk_delinq_balance = 0;
  if (rep_need_order_by_office = 0) then -- сортировка по офису не требуется, быстрый Fetch
  begin
    if (rep_region_id = 1000000) then -- Если отчет берется по всей стране, фильтр по регионам не требуется
    begin
      for select oks_from_k_id, oks_region_id, oks_kc_id, oks_delinq_date, oks_balance, k_index_curr_code, k_from_pr_id,
                 k_index_curr_rate, k_fact_disb_date, k_fact_disb_reg_id, k_fact_disb_lo_id, k_amount_approved,
                 k_from_kf_id, k_kolich_client_approved, k_kredline_end, k_kontract_no, k_percent_per_day, k_stop_perc,
                 k_zero_balance_date
          from oksummary, kontract
          where oks_date_begin between '01.01.1900' and :rep_date_end and
                oks_date_end between(:rep_date_end) and '31.12.9999' and
                oks_from_k_id = k_id and
                oks_close_state = 0
--          order by oks_region_id, oks_kc_id
          into :kk_id, kk_region_id, :kk_kc_id, :kk_delinq_date, :kk_balance, :kk_currency, :kk_from_pr_id,
               :k_index_curr_rate, :kk_disb_date, :kk_disb_reg_id, :kk_disb_kc_id, :kk_amount_approved, :kk_from_kf_id,
               :kk_kolich_client, :kk_kredline_end, :kk_kontract_no, :kk_percent_per_day, :kk_stop_perc,
               :kk_zero_balance_date
      do
      begin
        kk_disb_curr_rate = :k_index_curr_rate;
        kk_delinq_days = 0;
        kk_delinq_balance = 0;
        if (kk_delinq_date > :rep_date_end) then
        begin
          kk_delinq_date = null;
          kk_delinq_days = 0;
          kk_delinq_balance = 0;
          kk_is_delinq = 0;
        end
        else
        begin
          kk_delinq_days = datediff(day from :kk_delinq_date to :rep_date_end);
          kk_delinq_balance = kk_balance;
          kk_is_delinq = 1;
        end
        if (rep_need_curr_rate = 1) then
        begin
          kk_curr_rate = 1;
          if (kk_currency <> def_curr) then
          begin
            select cr_rate
            from opercurr
            where cr_date <= :rep_date_end and
                  cr_date_end >= :rep_date_end and
                  cr_from_cu_id = :kk_currency
            into :kk_curr_rate;

            if (kk_curr_rate < k_index_curr_rate) then
            begin
              if (kk_from_pr_id <> pr_id_prior) then
              begin
                select pr_set_rate_in_kont
                from product
                where pr_id = :kk_from_pr_id
                into :pr_set_rate_in_kont;

                pr_id_prior = kk_from_pr_id;
              end
              if (pr_set_rate_in_kont = 1) then
                kk_curr_rate = :k_index_curr_rate;
            end
          end
        end
        kk_delinq_os = 0;
        kk_delinq_perc = 0;
        kk_delinq_fine = 0;

        if ((rep_need_just_delinq = 0) or (rep_need_just_delinq = 1 and
            kk_delinq_date is not null)) then
        begin
          if (rep_need_delinq_os = 1 and
              kk_delinq_date is not null) then -- требуется просроченные ОС и проценты
          begin
            select sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
            from operkred
            where ok_from_k_id = :kk_id and
                  ok_sh_date between :kk_delinq_date and :rep_date_end and
                  ok_type < 7 and
                  ok_ol_is_active = 1 and
                  ok_ol_date <= :rep_date_end and
                  ok_from_c_id = 0
            into :kk_delinq_os, :kk_delinq_perc;
            if (kk_delinq_os + kk_delinq_perc < 0.01) then
            begin
              kk_delinq_os = 0;
              kk_delinq_perc = 0;
            end
          end
          if (rep_need_delinq_fine = 1 and
              kk_delinq_date is not null) then -- требуется просроченные ОС и проценты
          begin
            select sum(sh_calc_fine)
            from aarep_make_zad_list(:kk_id, :rep_date_end, 0)
            where sh_date is not null
            into :kk_delinq_fine;
          end
          suspend;
        end
      end
      kk_delinq_date = null;
      kk_delinq_balance = 0;
      kk_balance = 0;
      kk_delinq_days = 0;
      kk_curr_rate = 1;
      kk_is_delinq = 0;
      if (rep_date_begin is not null) then
      begin
        for select ks_from_k_id, ks_region_id, ks_kc_id, ks_date_begin, k_index_curr_code, k_from_pr_id,

                   k_kolich_client_approved, k_kredline_end, k_kontract_no, k_percent_per_day, k_stop_perc,
                   k_zero_balance_date
            from kontstate, kontract

                  ks_date_begin between :rep_date_begin and :rep_date_end and
                  ks_from_k_id = k_id
--              order by ks_region_id, ks_kc_id
            into kk_id, kk_region_id, :kk_kc_id, :kk_close_date, :kk_currency, :kk_from_pr_id, :kk_disb_date,

                 :kk_kredline_end, :kk_kontract_no, :kk_percent_per_day, :kk_stop_perc, :kk_zero_balance_date
        do
        begin
          kk_disb_curr_rate = :k_index_curr_rate;
          suspend;
        end
      end
    end
    else -- получить список по ограниченным офисам
    begin
      for select oks_from_k_id, oks_region_id, oks_kc_id, oks_delinq_date, oks_balance, k_index_curr_code, k_from_pr_id,

--          order by oks_region_id, oks_kc_id

                   k_kolich_client_approved, k_kredline_end, k_kontract_no, k_percent_per_day, k_stop_perc,

                 k_index_curr_rate, k_fact_disb_date, k_fact_disb_reg_id, k_fact_disb_lo_id, k_amount_approved,
                 k_from_kf_id, k_kolich_client_approved, k_kredline_end, k_kontract_no, k_percent_per_day, k_stop_perc,
                 k_zero_balance_date
          from oksummary, region_list(:rep_region_id, 1, -1, 1), kontract
          where oks_date_begin between '01.01.1900' and :rep_date_end and
                oks_date_end between(:rep_date_end) and '31.12.9999' and
                oks_region_id = id_list and
                oks_from_k_id = k_id and
                oks_close_state = 0
          order by oks_region_id, oks_kc_id
          into :kk_id, kk_region_id, :kk_kc_id, :kk_delinq_date, :kk_balance, :kk_currency, :kk_from_pr_id,
               :k_index_curr_rate, :kk_disb_date, :kk_disb_reg_id, :kk_disb_kc_id, :kk_amount_approved, :kk_from_kf_id,
               :kk_kolich_client, :kk_kredline_end, :kk_kontract_no, :kk_percent_per_day, :kk_stop_perc,
               :kk_zero_balance_date
      do
      begin
        kk_disb_curr_rate = :k_index_curr_rate;
        kk_delinq_days = 0;
        kk_delinq_balance = 0;
        if (kk_delinq_date > :rep_date_end) then
        begin
          kk_delinq_date = null;
          kk_delinq_days = 0;
          kk_delinq_balance = 0;
          kk_is_delinq = 0;
        end
        else
        begin
          kk_delinq_days = datediff(day from :kk_delinq_date to :rep_date_end);
          kk_delinq_balance = kk_balance;
          kk_is_delinq = 1;
        end
        if (rep_need_curr_rate = 1) then
        begin
          kk_curr_rate = 1;
          if (kk_currency <> def_curr) then
          begin
            select cr_rate
            from opercurr
            where cr_date <= :rep_date_end and
                  cr_date_end >= :rep_date_end and
                  cr_from_cu_id = :kk_currency
            into :kk_curr_rate;

            if (kk_curr_rate < k_index_curr_rate) then
            begin
              if (kk_from_pr_id <> pr_id_prior) then
              begin
                select pr_set_rate_in_kont
                from product
                where pr_id = :kk_from_pr_id
                into :pr_set_rate_in_kont;

                pr_id_prior = kk_from_pr_id;
              end
              if (pr_set_rate_in_kont = 1) then
                kk_curr_rate = :k_index_curr_rate;
            end
          end
        end
        kk_delinq_os = 0;
        kk_delinq_perc = 0;
        kk_delinq_fine = 0;

        if ((rep_need_just_delinq = 0) or (rep_need_just_delinq = 1 and
            kk_delinq_date is not null)) then
        begin
          if (rep_need_delinq_os = 1 and
              kk_delinq_date is not null) then -- требуется просроченные ОС и проценты
          begin
            select sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
            from operkred
            where ok_from_k_id = :kk_id and
                  ok_sh_date between :kk_delinq_date and :rep_date_end and
                  ok_type < 7 and
                  ok_ol_is_active = 1 and
                  ok_ol_date <= :rep_date_end and
                  ok_from_c_id = 0
            into :kk_delinq_os, :kk_delinq_perc;
            if (kk_delinq_os + kk_delinq_perc < 0.01) then
            begin
              kk_delinq_os = 0;
              kk_delinq_perc = 0;
            end
          end
          if (rep_need_delinq_fine = 1 and
              kk_delinq_date is not null) then -- требуется просроченные ОС и проценты
          begin
            select sum(sh_calc_fine)
            from aarep_make_zad_list(:kk_id, :rep_date_end, 0)
            where sh_date is not null
            into :kk_delinq_fine;
          end
          suspend;
        end
      end
      kk_delinq_date = null;
      kk_delinq_balance = 0;
      kk_balance = 0;
      kk_delinq_days = 0;
      kk_curr_rate = 1;
      kk_is_delinq = 0;
      if (rep_date_begin is not null) then
      begin
        for select ks_from_k_id, ks_region_id, ks_kc_id, ks_date_begin, k_index_curr_code, k_from_pr_id,
                   k_fact_disb_date, k_fact_disb_reg_id, k_fact_disb_lo_id, k_amount_approved, k_from_kf_id,
                   k_kolich_client_approved, k_kredline_end, k_kontract_no, k_percent_per_day, k_stop_perc,
                   k_zero_balance_date
            from region_list(:rep_region_id, 1, -1, 1), kontstate, kontract
            where ks_region_id = id_list and
                  ks_state = 2 and
                  ks_ol_is_delete = 0 and
                  ks_date_begin between :rep_date_begin and :rep_date_end and
                  ks_from_k_id = k_id
            order by ks_region_id, ks_kc_id
            into kk_id, kk_region_id, :kk_kc_id, :kk_close_date, :kk_currency, :kk_from_pr_id, :kk_disb_date,
                 :kk_disb_reg_id, :kk_disb_kc_id, :kk_amount_approved, :kk_from_kf_id, :kk_kolich_client,
                 :kk_kredline_end, :kk_kontract_no, :kk_percent_per_day, :kk_stop_perc, :kk_zero_balance_date
        do
        begin
          kk_disb_curr_rate = :k_index_curr_rate;
          suspend;
        end
      end
    end -- конец первого блока, без сортировки офиса
  end
end
