CREATE PROCEDURE AAREP_ACCOUNT_SALDO(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    ACCOUNT VARCHAR(20),
    ACCOUNT_NAME VARCHAR(255),
    ACC_BALANCE VARCHAR(255),
    ACC_NUMBER INTEGER,
    BALANCE_NAME VARCHAR(255),
    ACC_GROUP VARCHAR(255),
    ACC_CLASS VARCHAR(255),
    CURR_CODE VARCHAR(3),
    END_DT DECIMAL(18,4),
    END_KT DECIMAL(18,4),
    LAST_OPER_DATE DATE,
    ACC_GROUP_NO VARCHAR(20),
    ACC_KIND INTEGER,
    ACC_KIND_NAME VARCHAR(255),
    ACCOUNT_TYPE INTEGER,
    ACCOUNT_IS_ERROR INTEGER,
    END_DT_CURR DECIMAL(18,4),
    END_KT_CURR DECIMAL(18,4),
    CURR_INFO VARCHAR(3))
AS
declare variable curr_rate decimal(18,4);
declare variable curr_prior varchar(3);
declare variable summa decimal(18,4);
declare variable summa_ekv decimal(18,4);
declare variable acc_kind_temp integer;
declare variable summa_curr decimal(18,4);
begin

  for select account, al_name, al_number, al_from_bg_id, b1.bg_name, b2.bg_name, b2.parent, currency, sum(summa_val),
             sum(summa_ekv), max(max_date), b2.bg_id, b2.bg_kind, al_type
      from aarep_account_saldo_temp(:report_region_id, :report_date_end), acc_list, balgroup b1, balgroup b2
      where account = al_id and
            al_from_bg_id = b1.bg_id and
            b1.parent = b2.bg_id and
            ((currency = :report_currency) or
            (:report_currency = '---') or
            (:report_currency = '000'))
      group by account, al_name, al_number, al_from_bg_id, b1.bg_name, b2.bg_name, b2.parent, currency, b2.bg_id, b2.bg_kind, al_type
      having sum(summa_val) <> 0 or sum(summa_ekv) <> 0
      into :account, :account_name, :acc_number, :acc_balance, :balance_name, :acc_group, :acc_class, :curr_code,
           :summa, :summa_ekv, :last_oper_date, :acc_group_no, :acc_kind_temp, :account_type
  do
  begin
    curr_info = curr_code;
    if (summa is null) then
      summa = 0;
    if (summa_ekv is null) then
      summa_ekv = 0;
    if (report_currency <> '000') then
    begin
      summa_ekv = :summa;
    end

    acc_kind = 0;
    acc_kind_name = 'Не указан';
    if (acc_kind_temp = 1) then
    begin
      acc_kind = 1;
      acc_kind_name = 'Активы';
    end
    if ((acc_kind_temp = 2) or (acc_kind_temp = 3)) then
    begin
      acc_kind = 2;
      acc_kind_name = 'Обязательства и капитал';
    end
    if ((acc_kind_temp = 4) or (acc_kind_temp = 5)) then
    begin
      acc_kind = 3;
      acc_kind_name = 'Доходы и расходы';
    end
    if (report_currency = '000') then
    begin
      curr_code = '000';
    end

    if (summa is null) then
      summa = 0;
    end_dt = 0;
    end_kt = 0;
    if (summa_ekv >= 0) then
    begin
      end_dt = summa_ekv;
      end_kt = 0;
    end
    else
    begin
      end_dt = 0;
      end_kt = -1 * summa_ekv;
    end

    end_dt_curr = 0;
    end_kt_curr = 0;
    if (summa >= 0) then
    begin
      end_dt_curr = :summa;
      end_kt_curr = 0;
    end
    else
    begin
      end_dt_curr = 0;
      end_kt_curr = -1 * :summa;
    end

    account_is_error = 0;
    if ((account_type = 0) and
        (end_dt - end_kt < 0)) then
      account_is_error = 1;
    if ((account_type = 1) and
        (end_dt - end_kt > 0)) then
      account_is_error = 1;
    suspend;
  end
end
