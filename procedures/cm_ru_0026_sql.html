CREATE PROCEDURE CM_RU_0026_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable K_DISB_DATE date;
declare variable K_KREDLINE_END date;
declare variable MAX_KREDLINE_END date;
declare variable MAX_KREDLINE_END_STR varchar(20);
declare variable KOL_YEAR D_SMALLINT;
declare variable PR_CODE D_STR_100;
begin
  select kont.k_disb_date, k_kredline_end, p.pr_code
  from kontract kont
  inner join product p on p.pr_id = kont.k_from_pr_id
  where k_id = :k_id and
        not exists(select *
                   from product
                   where k_from_pr_id = pr_id and
                         (pr_code containing '.ZALOG_CAR.' or
                         pr_code containing '.zalog.' or
                         pr_code containing '.privilege.' or
                         pr_code containing '.privilege_plus.'or
                         pr_code containing '.refinance.' or
                         pr_code containing '.refinance_oivo.' or
                         pr_code containing '.refinance_plus.' or
                         pr_code containing '.switch_plus.' or
                         pr_code containing '.b_bonus.'  or
                         pr_code containing '.b_refinance.' ))
  into :k_disb_date, :k_kredline_end, pr_code;
  kol_year = 1;

  if (k_disb_date >= '10.07.2025') then
  begin
    if (pr_code containing '.reliable_client.' or pr_code containing '.universal.' or pr_code containing '.oivo.' or pr_code containing '.selfemployed.' or pr_code containing '.par.' or
        pr_code containing '.business.' or pr_code containing '.plus_100.' or pr_code containing '.plus_100_b.') then
      kol_year = 2;
  end
  max_kredline_end = dateadd(year, kol_year, k_disb_date) - 1; --dateadd(364 day to k_disb_date);

  if (max_kredline_end < k_kredline_end) then
  begin
    max_kredline_end_str = replace(iif(extract(day from max_kredline_end) < 10, '0', '') || --
    extract(day from max_kredline_end) || '.' || iif(extract(month from max_kredline_end) < 10, '0', '') || --
    extract(month from max_kredline_end) || '.' || extract(year from max_kredline_end), ' ', '');
    res_error_text = 'Дата окончания срока займа не должна превышать ' || max_kredline_end_str;
    res_error_is_critical = 2;
    suspend;
  end
end
