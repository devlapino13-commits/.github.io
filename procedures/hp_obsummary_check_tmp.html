CREATE PROCEDURE HP_OBSUMMARY_CHECK_TMP(
    REP_REGION INTEGER,
    REP_DATE DATE)
RETURNS (
    ACCOUNT VARCHAR(20),
    SUMMA_VAL DECIMAL(18,4),
    SUMMA_EKV DECIMAL(18,4))
AS
begin
  for select account, sum(summa_val), sum(summa_ekv)
      from aarep_account_saldo_obs(:rep_region, :rep_date)
      group by account
      having sum(summa_val) <> 0 or sum(summa_ekv) <> 0
      into :account, :summa_val, :summa_ekv
  do
  begin
    suspend;
  end
  for select obs_from_al_id, sum(obs_dt_val - obs_kt_val), sum(obs_dt_ekv - obs_kt_ekv)
      from obsummary, region_list(:rep_region, 1, -1, 1), acc_list
      where obs_date <= :rep_date and
            obs_from_al_id = al_id and
            al_from_reg_id = id_list
      group by obs_from_al_id
      into :account, :summa_val, :summa_ekv
  do
  begin
    summa_val = 0 - summa_val;
    summa_ekv = 0 - summa_ekv;
    suspend;
  end
end
