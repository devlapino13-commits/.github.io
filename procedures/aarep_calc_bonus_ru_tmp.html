CREATE PROCEDURE AAREP_CALC_BONUS_RU_TMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    LAST_REG INTEGER,
    KC_ID DECIMAL(18,0),
    KC_DISB_AMOUNT DECIMAL(18,4),
    KC_DISB_AMOUNT_NEW DECIMAL(18,4),
    KC_DISB_KOL INTEGER,
    KC_DISB_KOL_NEW INTEGER)
AS
declare variable reg_id integer;
declare variable ol_date date;
declare variable ok_os decimal(18,4);
declare variable kc_prior decimal(18,0);
declare variable kc_id1 decimal(18,0);
declare variable k_circle decimal(18,4);
declare variable k_kolich integer;
begin
  kc_prior = 0;
  last_reg = 0;
  kc_disb_amount = 0;
  kc_disb_amount_new = 0;
  kc_disb_kol = 0;
  kc_disb_kol_new = 0;
  for select kal_new_kf_officer, kal_new_reg_id, ok_ol_date, ok_os, k_circle, k_kolich_client_approved
      from operkred, kontract, knaktlist
      where ok_ol_is_active = 1 and
            knaktlist.is_delete = 0 and
            ok_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            ok_ol_date between kal_date and kal_date_end and
            ok_type = 8 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_from_c_id = 0
      order by kal_new_kf_officer, ok_ol_date
      into :kc_id, :reg_id, :ol_date, :ok_os, :k_circle, :k_kolich
  do
  begin
    if ((kc_id = kc_prior) or (kc_prior = 0)) then
    begin
      if (k_circle > 1) then
        kc_disb_amount = kc_disb_amount + :ok_os;
      else
        kc_disb_amount_new = kc_disb_amount_new + :ok_os;
      if (k_circle > 1) then
        kc_disb_kol = kc_disb_kol + :k_kolich;
      else
        kc_disb_kol_new = kc_disb_kol_new + :k_kolich;
      ok_os = 0;
    end
    else
    begin
      kc_id1 = :kc_id;
      kc_id = kc_prior;
      suspend;
      kc_id = kc_id1;
      kc_disb_amount = 0;
      kc_disb_amount_new = 0;
      kc_disb_kol = 0;
      kc_disb_kol_new = 0;
      if (k_circle > 1) then
        kc_disb_kol = :k_kolich;
      else
        kc_disb_kol_new = :k_kolich;
      if (k_circle > 1) then
        kc_disb_amount = :ok_os;
      else
        kc_disb_amount_new = :ok_os;
      ok_os = 0;
    end
    last_reg = reg_id;
    kc_prior = :kc_id;
  end
  suspend;
end
