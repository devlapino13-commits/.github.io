CREATE PROCEDURE UXP$UXFRITEM_REMOVE(
    UFI_UUID_STR D_STR_CODE)
AS
declare variable tmp_uuid_str d_str_code;
declare variable ufi_uuid d_uuid;
begin
  if (coalesce(ufi_uuid_str, '') = '') then
    exit;

  ufi_uuid = char_to_uuid(ufi_uuid_str);

  for select uuid_to_char(ufi_uuid)
  from ux$fritem
  where ufi_parent_uuid = :ufi_uuid
  into tmp_uuid_str
  do
    execute procedure uxp$uxfritem_remove(:tmp_uuid_str);

  delete from ux$fritem
  where ufi_uuid = :ufi_uuid and
        ufi_parent_uuid is not null;
end
