CREATE PROCEDURE OKT_SAVE_CALL_INFO_API(
    PANT_ID D_ID,
    OBJECT_KIND D_STR_100,
    OBJECT_ID D_ID,
    SUBJECT_KIND D_STR_100,
    UN_ID D_ID,
    CALL_RESULT D_INTEGER,
    MEMO_STR D_STR_LONG,
    NEXT_CALL_DATE D_DATE,
    ASKED_CALLBACK D_BOOL,
    CONT_CALL_IS_ACTIVE D_BOOL,
    OTL_ID D_INTEGER,
    PNC_ID D_ID)
AS
declare variable json_data d_str_4096;
declare variable next_call_date_str d_str_20;
declare variable memo_str_fixed d_str_long;
begin
    if (memo_str is null) then
        memo_str_fixed = '';
    else
        memo_str_fixed = memo_str;

    if (next_call_date is null) then
        next_call_date_str = null;
    else
        next_call_date_str = f_datetostr(:next_call_date);

    json_data = '{' ||
        f_json_make_pair('pant_id', pant_id, 'int', 1) ||
        f_json_make_pair('object_kind', object_kind, 'str', 1) ||
        f_json_make_pair('object_id', object_id, 'int', 1) ||
        f_json_make_pair('subject_kind', subject_kind, 'str', 1) ||
        f_json_make_pair('un_id', un_id, 'int', 1) ||
        f_json_make_pair('call_result', call_result, 'int', 1) ||
        f_json_make_pair('memo_str', memo_str_fixed, 'str', 1) ||
        f_json_make_pair('next_call_date', next_call_date_str, 'str', 1) ||
        f_json_make_pair('asked_callback', asked_callback, 'bool', 1) ||
        f_json_make_pair('cont_call_is_active', cont_call_is_active, 'bool', 1) ||
        f_json_make_pair('otl_id', otl_id, 'int', 1) ||
        f_json_make_pair('pnc_id', pnc_id, 'int', 0) ||
        '}';

    insert into okt_call_save_task (ocs_project_id, ocst_json)
    values (1, :json_data);
end
