CREATE PROCEDURE HP_EDOC_SIGN_DRAFT(
    EDL_ID D_ID)
AS
declare variable eds_id d_id;
begin
  -- Процедура подписывает электронный документ в тестовой базе
  -- Используется для обхода ожидания подписания эл.документа

  -- Если боевая база, завершаем работу без действий
  if (exists(select * from hp_db_is_actual)) then
    exit;

  for select eds_id
  from edoc_sign
  where eds_from_edl_id = :edl_id
  into eds_id
  do
  begin
    -- Сообщение подтверждаем, чтобы номер телефона подтвердился когда
    -- подписывается договор АСП
    update edoc_msg
    set edm_status = 1
    where edm_from_eds_id = :eds_id
      and edm_status <> 1;

    -- Подписание документа
    update edoc_sign
    set eds_status = 1,
        eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'paper', 'id')
    where eds_id = :eds_id;
  end
end
