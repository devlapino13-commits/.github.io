CREATE PROCEDURE FHP_AGEIN360(
    DATE_BEGIN DATE,
    DATE_END DATE,
    IS_360 INTEGER)
RETURNS (
    RES_DAYS INTEGER)
AS
declare variable eo1 integer;
declare variable ye1 integer;
declare variable mo1 integer;
declare variable da1 integer;
declare variable ye2 integer;
declare variable mo2 integer;
declare variable da2 integer;
declare variable mm integer;
declare variable yy integer;
declare variable kol integer;
declare variable d integer;
begin
  res_days = 0;
  if (date_begin < date_end) then
  begin
    if (is_360 = 0) then
      res_days = date_end - date_begin;
    if (is_360 = 1) then
    /* American method, False in Excel */
    begin
      da1 = extract(day from date_begin);
      mo1 = extract(month from date_begin);
      ye1 = extract(year from date_begin);

      da2 = extract(day from date_end);
      mo2 = extract(month from date_end);
      ye2 = extract(year from date_end);

      mm = extract(month from date_begin);
      yy = extract(year from date_begin);
      if (mm = 12) then
      begin
        yy = yy + 1;
        mm = 1;
      end
      else
        mm = mm + 1;
      eo1 = extract(day from (cast('01.' || mm || '.' || yy as date) - 1));

      if (da1 = eo1) then
        da1 = 30;
      if (da2 = 31) then
      begin
        if (da1 < eo1) then
        begin
          date_end = date_end + 1;
          da2 = extract(day from date_end);
          mo2 = extract(month from date_end);
          ye2 = extract(year from date_end);
        end
        else
          da2 = 30;
      end
      kol = ((ye2 * 12 + mo2) - (ye1 * 12 + mo1)) * 30;
      res_days = kol + (da2 - da1);
    end
    if (is_360 = 2) then
    /* Eutopean method, True */
    begin
      da1 = extract(day from date_begin);
      mo1 = extract(month from date_begin);
      ye1 = extract(year from date_begin);

      da2 = extract(day from date_end);
      mo2 = extract(month from date_end);
      ye2 = extract(year from date_end);
      if (da1 = 31) then
        da1 = 30;
      if (da2 = 31) then
        da2 = 30;
      kol = ((ye2 * 12 + mo2) - (ye1 * 12 + mo1)) * 30;
      res_days = kol + (da2 - da1);
    end
    if (is_360 = 3) then
    /* User method, month=30 days */
    begin
      da1 = extract(day from date_begin);
      mo1 = extract(month from date_begin);
      ye1 = extract(year from date_begin);

      da2 = extract(day from date_end);
      mo2 = extract(month from date_end);
      ye2 = extract(year from date_end);

      mm = extract(month from date_begin);
      yy = extract(year from date_begin);
      if (mm = 12) then
      begin
        yy = yy + 1;
        mm = 1;
      end
      else
        mm = mm + 1;
      d = extract(day from (cast('01.' || mm || '.' || yy as date) - 1));
      if (da1 = d) then
        da1 = 30;
      if (da1 = 31) then
        da1 = 30;

      mm = extract(month from date_end);
      yy = extract(year from date_end);
      if (mm = 12) then
      begin
        yy = yy + 1;
        mm = 1;
      end
      else
        mm = mm + 1;
      d = extract(day from (cast('01.' || mm || '.' || yy as date) - 1));
      if (da2 = d) then
        da2 = 30;

      kol = ((ye2 * 12 + mo2) - (ye1 * 12 + mo1)) * 30;
      res_days = kol + (da2 - da1);
    end
  end
  suspend;
end
