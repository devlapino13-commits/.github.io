CREATE PROCEDURE AAREP_ISSUANCE_PLAN(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    OFFICER_NAME VARCHAR(100),
    SUM_PLAN_KL CURR_TYPE,
    SUM_OSTALOS_KL CURR_TYPE,
    SUM_PLAN_IPF CURR_TYPE,
    SUM_OSTALOS_IPF CURR_TYPE)
AS
declare variable rep_date_begin date;
declare variable rep_date_end date;
declare variable reg_id integer;
declare variable reg_id_prior integer;
declare variable sum_ahead_kl decimal(18,4);
declare variable sum_fact_kl decimal(18,4);
declare variable sum_ahead_ipf decimal(18,4);
declare variable sum_fact_ipf decimal(18,4);
declare variable c_id decimal(18,0);
begin
  rep_date_begin = rep_date - extract(day from rep_date) + 1;
  rep_date_end = dateadd(month, 1, rep_date_begin) - 1;
  reg_id_prior = -1;
  for select c_id, sum(iif(is_ipf <> 1, sum_plan, 0)), sum(iif(is_ipf <> 1, sum_ahead, 0)),
             sum(iif(is_ipf <> 1, sum_fact, 0)), sum(iif(is_ipf = 1, sum_plan, 0)), sum(iif(is_ipf = 1, sum_ahead, 0)),
             sum(iif(is_ipf = 1, sum_fact, 0))
      from aarep_issuance_plan_temp(:rep_date), region_list(:rep_region_id, 1, 3, 1)
      where c_id > 1000000000004 and
            reg_id = id_list
      group by c_id
      into c_id, sum_plan_kl, sum_ahead_kl, sum_fact_kl, sum_plan_ipf, sum_ahead_ipf, sum_fact_ipf
  do
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into officer_name;

    reg_id = 0;
    select ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :c_id and
          ps_status = 1 and
          :rep_date_end between ps_date_begin and ps_date_end
    into reg_id;

    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      reg_name = null;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;
    end

--    un_id = null;
--    select un_id from username
--    where un_from_c_id = :c_id
--    into un_id;

    sum_ostalos_kl = sum_plan_kl + sum_ahead_kl - sum_fact_kl;
    sum_ostalos_ipf = sum_plan_ipf + sum_ahead_ipf - sum_fact_ipf;
    suspend;
  end
end
