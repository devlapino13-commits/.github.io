CREATE PROCEDURE AAREP_SIRENA_TRANSFER_LIST(
    REP_PREOJECT STR_100,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    CC_PROJECT STR_100,
    K_KONTRACT_NO STR_100,
    TRANSFER_TYPE STR_100,
    PT_DATE DATE,
    PROJECT STR_100,
    OUT_OPER_CALLED STR_100,
    INPUT_OPER_CALLED STR_100,
    DELINQ_DAYS INTEGER,
    DELINQ_OS CURR_TYPE,
    DELINQ_PERC CURR_TYPE,
    DELINQ_FINE CURR_TYPE,
    REASON STR_255,
    TRANSFER_CATEGORY STR_100)
AS
declare variable out_pant_id id_type;
declare variable input_pant_id id_type;
declare variable k_id id_type;
declare variable kf_officer id_type;
declare variable project_id integer;
begin
  for select proj.rd_name, pt.pt_out_pant_id, pt.pt_input_pant_id, pt.pt_delinq_os, pt.pt_delinq_perc,
             pt.pt_delinq_fine, pt.pt_delinq_days, rd.rd_name as reason, pt.pt_from_k_id, pt.pt_date,
             pt.pt_project_rd_id
      from pantera_transfer pt
      inner join refdata rd on pt.pt_reason_rd_id = rd.rd_id
      inner join refdata proj on proj.rd_id = pt.pt_project_rd_id
      where pt.pt_date between :rep_date_begin and :rep_date_end and
            proj.rd_code containing (:rep_preoject) --in ('OKTELL.Projects.Sirena', 'OKTELL.Projects.Sirena1', 'OKTELL.Projects.Sirena2')
      into :cc_project, :out_pant_id, :input_pant_id, :delinq_os, :delinq_perc, :delinq_fine, :delinq_days, :reason,
           :k_id, :pt_date, :project_id
  do
  begin
    out_oper_called = null;
    input_oper_called = null;

    select pantk.pant_oper_called
    from pantera_kontracts pantk
    where pantk.pant_id = :out_pant_id
    into :out_oper_called;

    select pantk.pant_oper_called
    from pantera_kontracts pantk
    where pantk.pant_id = :input_pant_id
    into :input_oper_called;

    out_oper_called = coalesce(out_oper_called, '');
    input_oper_called = coalesce(input_oper_called, '');

    if (out_oper_called = 'IGUSEV') then
      out_oper_called = 'Виртуальный оператор';
    if (input_oper_called = 'IGUSEV') then
      input_oper_called = 'Виртуальный оператор';

    transfer_type = 'Передача';

    if ((out_oper_called <> '') and
        (input_oper_called = '')) then
      transfer_type = 'Разблокировка';
    if ((out_oper_called = '') and
        (input_oper_called <> '')) then
      transfer_type = 'Блокировка';

    transfer_category = '';

    if ((transfer_type = 'Блокировка') and
        (rep_preoject = 'OKTELL.Projects.Pantera')) then
    begin
      transfer_category = 'Первичная';
      if (exists(select *
                 from pantera_kontracts pantk
                 where pantk.pant_id <> :input_pant_id and
                       pantk.pant_from_k_id = :k_id and
                       pantk.pant_date_begin <= :pt_date and
                       coalesce(pantk.pant_is_dismiss, 0) = 0 -- важно, чтобы не поймать увольнения
                       and
                       pantk.pant_rd_project_type = :project_id and
                       pantk.pant_oper_called <> :input_oper_called)) then
      begin
        transfer_category = 'Вторичная';
      end
    end

    select kont.k_kontract_no, kal_new_kf_officer
    from kontract kont
    inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
    where kont.k_id = :k_id and
          :pt_date between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into :k_kontract_no, kf_officer;

    if (kf_officer = 1000000000004) then
      project = 'ВС Сирена';
    else
      project = 'СВЗ';

    suspend;
  end
end
