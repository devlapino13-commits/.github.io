CREATE PROCEDURE AAREP_EXTMINIAPP_ONLY_FROM_APP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    EMA_ID D_INTEGER,
    FIO STR_100,
    CELL_NO STR_100,
    MCL_LOGIN STR_100,
    KC_CIRCLE INTEGER,
    EMA_OFFICE STR_255,
    EMA_DATETIME D_DATETIME,
    EMA_BLOCK_DATETIME D_DATETIME,
    SVZ_FIO STR_100,
    CURR_STATE STR_100,
    CURR_STATE_DATETIME D_DATETIME,
    REFUSE_REASON STR_255,
    FACT_DISB_SVZ_FIO STR_100,
    K_KONTRACT_NO STR_20)
AS
declare variable TKT_ID D_ID;
declare variable C_ID ID_TYPE;
declare variable EMA_DATE date;
declare variable EMA_STATUS D_SMALLINT;
declare variable EMA_STATUS_DATETIME D_DATETIME;
declare variable EMA_REFUSE STR_255;
declare variable EMA_REFUSE_CODE STR_255;
declare variable MA_ID D_INTEGER;
declare variable MA_REFUSE_REASON STR_255;
declare variable MA_IS_ACTIVE D_SMALLINT;
declare variable MA_STATE_DATETIME D_DATETIME;
declare variable MA_SVZ STR_100;
declare variable K_ID ID_TYPE;
declare variable K_STATE D_SMALLINT;
declare variable K_KONTRACT_DATE date;
declare variable K_REFUSE_REASON STR_255;
declare variable K_STATE_DATETIME D_DATETIME;
declare variable K_SVZ STR_100;
begin

    for select ema_id, c.c_name_fio, ema_from_c_id, reg_code || ' ' || reg_name,
               ema_datetime, ema_block_datetime, s.c_name_fio,
               ema_status, ema_status_datetime, rd_code, ema_from_tkt_id, ema_date, rd_name, left(mcl_login, 11)
    from extminiapp 
    inner join region on reg_unicode = ema_office_id
    inner join region_list(:rep_region_id, 1, -1, 1) on id_list = reg_unicode
    inner join clients c on c.c_id = ema_from_c_id
    left join clients s on s.c_id = ema_lo_id
    inner join refdata on rd_id = ema_result_rd_id
    inner join mc_login on mcl_id = ema_from_mcl_id
    where ema_date between :rep_date_begin and :rep_date_end
    and ema_from_mcl_id is not null
    into ema_id, fio, c_id, ema_office, ema_datetime, ema_block_datetime, svz_fio,
         ema_status, ema_status_datetime, ema_refuse_code, tkt_id, ema_date, ema_refuse, mcl_login
    do
    begin 
        cell_no = f_cont_info('clients', :c_id, 'cell_main', 'ok');

        select max(kc_circle)
        from knclient 
        where kc_from_c_id = :c_id
        into kc_circle;

        ma_id = null;
        ma_refuse_reason = null;
        ma_is_active = null;
        ma_state_datetime = null;
        k_id = null;
        k_refuse_reason = null;
        k_state = null;
        curr_state = null;
        curr_state_datetime = null;
        refuse_reason = null;
        fact_disb_svz_fio = null;
        k_kontract_no = null;
        k_state_datetime = null;
        ma_svz = null;
        k_svz = null;

         if ( (ema_status = -1 and ema_refuse_code = 'website_miniapp.Refuse.MiniappCreateCancel') or (ema_status = 1) ) then
        begin
            if (exists(select ma_id
                        from miniapp maa
                        where ma_from_tkt_id = :tkt_id)) then
            begin
                select ma_id
                from miniapp
                where ma_from_tkt_id = :tkt_id
                rows 1
                into ma_id;

                select k_id
                from kontract
                where k_from_tkt_id = :tkt_id
                rows 1
                into k_id;
            end
            else
            begin
                select min(mac_from_ma_id), ma_date_app
                from miniappclient
                inner join miniapp on ma_id = mac_from_ma_id
                where mac_from_c_id = :c_id
                and ma_date_app >= :ema_date
                and ma_date_end = '31.12.9999'
                and mac_amount > 0
                group by 2
                order by 2
                rows 1
                into ma_id, k_kontract_date;
            end

            select k_id
            from kontract
            where k_from_ma_app_id = :ma_id
            rows 1
            into k_id;

            if (ma_id is not null and k_id is null) then
            begin
                select k_id
                from kontract
                where k_from_ma_app_id = :ma_id
                and  k_kontract_date >= :k_kontract_date
                rows 1
                into k_id;
            end

            select ma_is_active, kr_text, ma_date_save, un_fio
            from miniapp
            left join kontreslist on krl_ma_id = ma_id
            left join kontres krs on kr_id = krl_from_kr_id
            left join username on un_id = ma_from_un_id
            where ma_id = :ma_id
            and ma_date_end = '31.12.9999'
            rows 1
            into ma_is_active, ma_refuse_reason, ma_state_datetime, ma_svz;

            select k_status, iif(k_status < 0, '', k_kontract_no), kr_text, c_name_fio
            from kontract
            left join kontreslist on krl_from_k_id = k_id
            left join kontres on kr_id = krl_from_kr_id
            left join kontstate on ks_from_k_id = k_id
            left join clients on c_id = ks_kc_id
            where k_id = :k_id
            and current_date between ks_date_begin and ks_date_end
            and ks_ol_is_delete = 0
            rows 1
            into k_state, k_kontract_no,  k_refuse_reason, k_svz;

            if (k_id is not null) then
            begin
                select ol_datetime
                from operkred
                inner join operlist on ok_from_ol_id = ol_id
                where ok_from_k_id = :k_id
                and ok_type = 8
                and ok_ol_is_active = 1
                into :k_state_datetime;
            end
        end

        curr_state = '';
        refuse_reason = '';

         if (ma_id is null and k_id is null) then
        begin
            if (ema_status = 0) then
            begin
                if (ema_block_datetime is null) then
                    curr_state = 'Не заблокировано ВИПС';
                 if (ema_block_datetime is not null) then
                    curr_state = 'Заблокировано ВИПС';
            end
            if (ema_status = -1) then
            begin
                curr_state = 'Отказ';
                refuse_reason = ema_refuse;
            end
        end

        if (ma_id is not null and k_id is null) then
        begin
            if (ma_is_active = 0) then
                curr_state = 'Отказ';
            if (ma_is_active = 1) then
                curr_state = 'В процессе';
            if (ma_is_active = -1) then
            begin
                curr_state = 'Отказ';
                refuse_reason = ma_refuse_reason;
            end
        end

        if (ma_id is not null and k_id is not null) then
        begin
            if (k_state = 1) then
                curr_state = 'Выдан';
            if (k_state = 0) then
                curr_state = 'В процессе';
            if (k_state = -1) then
            begin
                curr_state = 'Отказ';
                refuse_reason = k_refuse_reason;
            end 
            if (k_state = -2) then
            begin
                if (ma_is_active = 0) then
                curr_state = 'Отказ';
                if (ma_is_active = 1) then
                curr_state = 'В процессе';
                if (ma_is_active = -1) then
                begin
                    curr_state = 'Отказ';
                    refuse_reason = ma_refuse_reason;
                end
            end
        end 

        if (ma_svz is not null) then
            fact_disb_svz_fio = ma_svz;
        if (k_svz is not null) then
            fact_disb_svz_fio = k_svz;

        if (ema_datetime is not null) then
            curr_state_datetime = ema_datetime;
        if (ema_status_datetime is not null) then
            curr_state_datetime = ema_status_datetime;
        if (ma_state_datetime is not null) then
            curr_state_datetime = ma_state_datetime;
        if (k_state_datetime is not null) then
            curr_state_datetime = k_state_datetime;

        suspend;

    end
end
