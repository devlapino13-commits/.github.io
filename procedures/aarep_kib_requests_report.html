CREATE PROCEDURE AAREP_KIB_REQUESTS_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OKRUG_NAME STR_100,
    REGION_NAME STR_100,
    OFFICE_NAME STR_100,
    KF_NO STR_100,
    MA_NO INTEGER,
    MA_ID INTEGER,
    CL_TYPE STR_100,
    MA_KONT_STAT_STR STR_100,
    PR_NAME STR_100)
AS
declare variable st_staff integer;
declare variable st_client integer;
declare variable k_status integer;
declare variable ma_kont_id id_type;
declare variable ma_kont_state integer;
declare variable reg_unicode integer;
declare variable is_active smallint;
declare variable ma_id_prior integer;
declare variable ma_number_prior integer;
declare variable ma_number integer;
declare variable dd date;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  dd = cast('01.01.2999' as date);
  ma_id_prior = 0;

  for select kaa.kbm_ma_id,
             (
             case macl.mac_is_poruch
               when 0 then 'Клиент'
               else 'Поручитель'
             end) as cl_type
      from kibminiapp kaa
      inner join miniappclient macl on macl.mac_from_ma_id = kaa.kbm_ma_id and
            macl.mac_from_c_id = kaa.kbm_from_c_id
      where kaa.kbm_pr_id_datetime >= :rep_date_begin and
            kaa.kbm_pr_id_datetime < :rep_date_end + 1
      order by kaa.kbm_ma_id
      into :ma_id, :cl_type
  do
  begin
    if (ma_id <> ma_id_prior) then
    begin
      okrug_name = '';

      select reg3.reg_name as okrug_name, reg2.reg_name as region_name, reg.reg_name as office_name, maa.ma_no,
             maa.ma_is_active
      from miniapp maa
      inner join region_list(:rep_region_id, 1, -1, 1) rl on id_list = maa.ma_office_id
      inner join region reg on id_list = reg.reg_unicode
      inner join region reg2 on reg.parent = reg2.reg_unicode
      inner join region reg3 on reg2.parent = reg3.reg_unicode
      where maa.ma_id = :ma_id and
            maa.ma_date_end = '31.12.9999'
      into :okrug_name, :region_name, :office_name, :ma_no, is_active;

      if (okrug_name <> '') then
      begin
        ma_kont_id = null;
        k_status = null;
        ma_kont_state = null;
        kf_no = null;
        pr_name = '';

        if (is_active = 1) then
        begin
          for select k_id, k_status, kal_new_kf_name, pr.pr_name
              from kontract kont
              inner join knaktlist kal on kal_from_kf_id = k_from_kf_id
              inner join product pr on pr.pr_id = kont.k_from_pr_id
              where k_from_ma_app_id = :ma_id and
                    kal.is_delete = 0 and
                    kal_date <= :rep_date_end and
                    kal_date_end >= :rep_date_end
              order by kont.k_fact_disb_date desc
              rows 1
              into :ma_kont_id, :k_status, kf_no, :pr_name
          do
          begin
            select distinct ks_state
            from kontstate
            where ks_from_k_id = :ma_kont_id and
                  ks_ol_is_delete = 0 and
                  ks_date_end >= :dd
            into :ma_kont_state;
          end

          if (ma_kont_state is null) then
            ma_kont_state = -1;
          if ((k_status = -1) and
              (ma_kont_state = 0)) then
            ma_kont_state = -2;
          if (ma_kont_state = 0 and
              k_status = -2) then
            ma_kont_state = -1;

          if (ma_kont_state = -2) then
            ma_kont_stat_str = 'Аннулирован';
          if (ma_kont_state = -1) then
            ma_kont_stat_str = 'Заявка';
          if (ma_kont_state = 0) then
            ma_kont_stat_str = 'В процессе';
          if (ma_kont_state = 1) then
            ma_kont_stat_str = 'Активный';
          if (ma_kont_state = 2) then
            ma_kont_stat_str = 'Закрытый';
        end
        else
          ma_kont_stat_str = 'Аннулирован';
      end
      ma_id_prior = ma_id;
    end
    suspend;
  end
end
