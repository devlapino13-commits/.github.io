CREATE PROCEDURE HP_OBJECTPROPS_MEMO_SET(
    OBJECT_KIND VARCHAR(100),
    OBJECT_ID VARCHAR(255),
    SUBJECT_KIND VARCHAR(100),
    SUBJECT_ID VARCHAR(255),
    SET_MEMO BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    UN_ID DECIMAL(18,0))
AS
declare variable uniq_object_id decimal(18,0);
declare variable uniq_subject_id decimal(18,0);
declare variable uniq_object_kind integer;
declare variable uniq_subject_kind integer;
declare variable ovs_from_opr_id decimal(18,0);
begin
-- Процедура устанавливает новое Примечание для пары Object/Subject

  select opr_object_kind, opr_object_id, opr_subject_kind, opr_subject_id
  from hp_get_propsparam(:object_kind, :object_id, :subject_kind, :subject_id)
  into :uniq_object_kind, :uniq_object_id, :uniq_subject_kind, :uniq_subject_id;

  if (set_memo is null) then
    set_memo = '';
  if (set_memo = '') then
    exit;

  select opr_id
  from objectprops
  where opr_object_kind = :uniq_object_kind and
        opr_subject_kind = :uniq_subject_kind and
        opr_object_id = :uniq_object_id and
        opr_subject_id = :uniq_subject_id
  into :ovs_from_opr_id;

  if (ovs_from_opr_id is null) then
  begin
    update or insert into objectprops (opr_object_kind, opr_object_id, opr_subject_kind, opr_subject_id)
    values (:uniq_object_kind, :uniq_object_id, :uniq_subject_kind, :uniq_subject_id)
    matching (opr_object_kind, opr_object_id, opr_subject_kind, opr_subject_id)
    returning (opr_id)
    into :ovs_from_opr_id;

    insert into objectvalues (ovs_from_opr_id, ovs_date_begin, ovs_value, ovs_add_value, ovs_from_un_id)
    values (:ovs_from_opr_id, current_date, '0', 0, :un_id);
  end
  insert into objectmemo (obm_from_opr_id, obm_memo, obm_from_un_id)
  values (:ovs_from_opr_id, :set_memo, :un_id);
end
