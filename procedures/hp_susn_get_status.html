CREATE PROCEDURE HP_SUSN_GET_STATUS(
    C_ID D_ID,
    MA_ID D_ID)
RETURNS (
    RES_STATUS D_SMALLINT,
    IS_SUSN D_SMALLINT,
    SUSN_NAMES D_STR_2048,
    TRY_COUNT D_INTEGER,
    MA_ID_OUT D_ID,
    IS_APPLICABLE D_SMALLINT)
AS
declare variable EIC_ID D_ID;
declare variable OBJ_KIND_MA D_INTEGER;
declare variable EIL_STATUS D_SMALLINT;
declare variable BLS_ID D_ID;
declare variable JSON_TMP blob sub_type 1 segment size 80;
declare variable STATUS_COUNT D_INTEGER;
declare variable I D_INTEGER;
declare variable CUR_NAME D_STR_LONG;
declare variable CUR_DEND D_STR_100;
declare variable CUR_ISO D_STR_100;
declare variable MA_CREATED timestamp; /* RFCKZ-996 */
declare variable SUSN_START D_STR_100; /* RFCKZ-996 */
begin
  -- RFCKZ-996
  res_status = 0;
  is_susn = 0;
  susn_names = '';
  try_count = 0;
  bls_id = null;
  json_tmp = null;
  obj_kind_ma = f_objkind('miniapp', 'id');

  eic_id = (select eic.eic_id
            from extinfocfg eic
            where eic.eic_code = 'susn');

  -- заявка не передана - берем последнюю действующую заявку клиента
  if (ma_id is null) then
    select first 1 ma.ma_id
    from miniapp ma
    inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
                                and mac.mac_amount > 0
    where mac.mac_from_c_id = :c_id
      and ma.ma_date_end >= current_date
    order by ma.ma_date_oper desc, ma.ma_id desc
    into ma_id;

  ma_id_out = ma_id;

  /* --- RFCKZ-996: общий выключатель и отсечка по дате запуска -----------
     susn_active         = 1  -> проверка СУСН включена
     susn_start_datetime      -> заявки, созданные раньше этого момента,
                                 под задачу не подпадают. Формат
                                 'YYYY-MM-DD HH:MM:SS'.
                                 Пусто = ограничения по времени нет.

     Момент создания заявки берём как min(ma_date_save) по всем версиям
     заявки: miniapp версионируется, и последующее редактирование
     меняет ma_date_save у новой версии, а минимум остаётся моментом
     первичного создания.

     Когда проверка неприменима, отдаём res_status = 2 и is_susn = 0.
     Это состояние «как до RFCKZ-996»: блоки пропускают, предел КДН 0.5,
     предупреждений нет. Вывод статуса скрывается по is_applicable = 0.   */
  is_applicable = 1;

  if (coalesce(f_get_const('susn_active'), '0') <> '1') then
    is_applicable = 0;

  if (is_applicable = 1 and ma_id is not null) then
  begin
    susn_start = coalesce(f_get_const('susn_start_datetime'), '');

    if (susn_start <> '') then
    begin
      ma_created = null;

      select min(ma.ma_date_save)
      from miniapp ma
      where ma.ma_id = :ma_id
      into ma_created;

      if (ma_created is null or ma_created < cast(:susn_start as timestamp)) then
        is_applicable = 0;
    end
  end

  if (is_applicable = 0) then
  begin
    res_status = 2;
    is_susn = 0;
    susn_names = '';
    try_count = 0;
    suspend;
    exit;
  end

  if (ma_id is null or eic_id is null) then
  begin
    suspend;
    exit;
  end

  select first 1 eil.eil_status, eil.eil_from_bls_id
  from extinfolist eil
  where eil.eil_from_eic_id = :eic_id
    and eil.eil_object_kind = :obj_kind_ma
    and eil.eil_object_id = :ma_id
  order by eil.eil_id desc
  into eil_status, bls_id;

  select count(*)
  from extinfolist eil
  where eil.eil_from_eic_id = :eic_id
    and eil.eil_object_kind = :obj_kind_ma
    and eil.eil_object_id = :ma_id
  into try_count;

  if (eil_status is null) then
  begin
    res_status = 0;
    suspend;
    exit;
  end

  if (eil_status in (-1, 0, 1)) then
  begin
    res_status = 1;
    suspend;
    exit;
  end

  if (eil_status <> 2 or bls_id is null) then
  begin
    res_status = -1;
    suspend;
    exit;
  end

  res_status = 2;

  select b.blx_blob
  from hp_get_blobex_blob(:bls_id) b
  into json_tmp;

  status_count = coalesce(f_json_blob_get_count(json_tmp, 'result.Status'), 0);

  if (status_count > 0) then
  begin
    status_count = minvalue(status_count, 10);

    -- текущая дата в ISO для сравнения со statusDend
    cur_iso = lpad(cast(extract(year  from current_date) as varchar(4)), 4, '0') || '-'
           || lpad(cast(extract(month from current_date) as varchar(2)), 2, '0') || '-'
           || lpad(cast(extract(day   from current_date) as varchar(2)), 2, '0');

    i = 0;
    while (i < status_count) do
    begin
      -- статус с истекшим сроком действия не учитываем
      cur_dend = left(coalesce(f_json_blob_get_value(json_tmp,
                   'result.Status[' || cast(:i as varchar(10)) || '].statusDend', null), ''), 10);

      if (cur_dend = '' or cur_dend >= :cur_iso) then
      begin
        is_susn = 1;

        cur_name = f_json_blob_get_value(json_tmp,
                     'result.Status[' || cast(:i as varchar(10)) || '].statusNameRu', null);

        if (coalesce(cur_name, '') <> '') then
          susn_names = susn_names || iif(susn_names <> '', ', ', '') || cur_name;
      end

      i = i + 1;
    end
  end

  suspend;
end
