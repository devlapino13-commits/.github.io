CREATE PROCEDURE HP_CHECK_PHOTO_KIND(
    BLC_ID INTEGER,
    UN_ID DECIMAL(18,0),
    KEY_1 DECIMAL(18,0),
    KEY_2 DECIMAL(18,0))
RETURNS (
    BLC_CAN_PHOTO INTEGER,
    BLC_CAN_SCAN INTEGER,
    BLC_CAN_FILES INTEGER)
AS
declare variable s_name varchar(100);
declare variable is_poweruser integer;
declare variable blc_table_name varchar(100);
declare variable is_recorevy boolean;
begin
  is_poweruser = 0;
  is_recorevy  = exists(select * from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.RecoveryLostDocs') where has_access = 1)
    and exists(select * from clients_doc_loss where c_id = :key_1);

-- если пользователь один из нижеуказанных лиц, то может закреплять доки через файлы
  s_name = 'FinStart.Superuser';
  if (is_poweruser = 0) then
    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, :s_name)
               where has_access = 1)) then
      is_poweruser = 1;

  s_name = 'FinStart.Programmer';
  if (is_poweruser = 0) then
    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, :s_name)
               where has_access = 1)) then
      is_poweruser = 1;

  s_name = 'FinStart.Permission_Administrator';
  if (is_poweruser = 0) then
    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, :s_name)
               where has_access = 1)) then
      is_poweruser = 1;

  s_name = 'FinStart.Permission_Operator';
  if (is_poweruser = 0) then
    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, :s_name)
               where has_access = 1)) then
      is_poweruser = 1;

  s_name = 'FinStart.Permission_Staff_Data_Edit'; -- кадровики
  if (is_poweruser = 0) then
    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, :s_name)
               where has_access = 1)) then
      is_poweruser = 1;

  for select blc_can_photo, blc_can_scan, blc_can_files, blc_table_name
      from blobconfig
      where blc_id = :blc_id
      into :blc_can_photo, blc_can_scan, blc_can_files, blc_table_name
  do
  begin
    if (blc_table_name = 'clientdocs') then
    begin
      blc_can_files = 0;
      select pk_can_load
      from clientdocs, passkind
      where cds_from_c_id = :key_1 and
            cds_id = :key_2 and
            cds_from_pk_id = pk_id
      into :blc_can_files;
      blc_can_files = coalesce(blc_can_files, 0);
    end
    -- если сотрудник отдела кадров или тех.поддержки, даем возможность прикреплять файлы
    if (blc_can_files = 0 and
        is_poweruser = 1) then
      blc_can_files = 1;

    if (is_recorevy) then
    begin
      --blc_can_photo = 0;
      --blc_can_scan = 0;
      blc_can_files = 1;
    end
    suspend;
  end
end
