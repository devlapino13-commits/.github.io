CREATE PROCEDURE FM_HTML_ANTIFRAUD_CLIENT_CHECK(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable IS_WORKING D_BOOL;
declare variable C_INN D_STR_20;
declare variable AC_ID D_STR_CODE;
declare variable AC_STATUS D_STR_CODE;
declare variable AC_STATUS_STR D_STR_255;
declare variable AC_DATE D_DATE;
declare variable NEED_CHECK D_BOOL;
begin
    res_text = '';

    -- Если, механизм антифрода выключен - тогда ничего не отображаем на стороне МП
    :is_working = (select is_working from antifraud_api_working_status);
    if (coalesce(:is_working, 0) = 0) then
    begin
        suspend;
        exit;
    end

    res_text = blob_append(res_text, '<br><b>Проверка на антифрод</b><br>');
    :c_inn = (select c.c_inn from clients c where c.c_id = :c_id);

    -- Получаем информацию о последней проверке
    select first 1
        f_enum('antifraud_status', ac.ac_status, 'code'),
        ac.ac_date
    from antifraud_check ac
    where ac.ac_inn = :c_inn
    order by ac.ac_id desc
    into
        :ac_status,
        :ac_date;

    ac_status_str =
        case :ac_status
            when 'not_found' then 'В списке не найден'
            when 'was_found' then 'В списке найден'
            when 'client_error' then 'При попытке проверки в АЦ возникла ошибка.'
            when 'server_error' then 'При попытке проверки в АЦ возникла ошибка.'
            when 'format_error' then 'При попытке проверки в АЦ возникла ошибка.'
            when 'in_process' then 'Запрос в обработке'
        end;

    if (coalesce(:ac_status, '') = '') then
    begin
        res_text = blob_append(res_text, 'Проверка не проводилась<br>');
    end
    else
    begin
        res_text = blob_append(res_text, f_format('Последняя проверка: %1<br>', f_datetostr(:ac_date)));
        res_text = blob_append(res_text, f_format('Статус: %1<br>', :ac_status_str));
    end

    -- Определяем нужно ли показывать кнопку "Проверить"
    need_check = f_need_antifraud_check(:c_id);
    
    if (:need_check = 1) then
    begin
        -- отправить запрос
        res_text = blob_append(res_text, 
            f_uri_sql
            (
                'Проверить',
                'start_antifraud_check', 
                '',
                0,
                'none',
                'Запустить проверку антифрода?',
                '',
                0,
                0,
                '',
                '',
                'dialog',
                :c_inn,
                0
            ) || '<br>');
    end

    suspend;
end
