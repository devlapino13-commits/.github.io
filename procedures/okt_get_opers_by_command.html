CREATE PROCEDURE OKT_GET_OPERS_BY_COMMAND(
    COMMAND_ID D_INTEGER,
    REP_DATE D_DATE = current_date)
RETURNS (
    UN_ID D_ID,
    UN_LOGIN D_STR_100)
AS
begin
  for select pco.pco_from_un_id,
             un.un_name
  from pantera_commands pcm
  inner join pantera_commands_operators pco on pco.pco_from_pcm_id = pcm.pcm_id
  inner join username un on un.un_id = pco.pco_from_un_id
  where :rep_date between pco.pco_date_begin and pco.pco_date_end
  and coalesce(pcm.pcm_date_end, '31.12.9999') >= :rep_date
  and pco.pco_from_pcm_id = :command_id
  into :un_id,
       :un_login
  do
    suspend;
end
