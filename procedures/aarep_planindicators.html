CREATE PROCEDURE AAREP_PLANINDICATORS(
    REP_DATE_BEGIN DATE)
RETURNS (
    RD_NAME STR_255,
    REG_NAME STR_100,
    SUM_PLAN DECIMAL(18,2),
    SUM_FACT DECIMAL(18,2),
    SUM_OST DECIMAL(18,2))
AS
declare variable UN_ID decimal(18,0);
declare variable DT_BEG date;
declare variable DT_END date;
declare variable YR_CALC integer;
declare variable MN_CALC integer;
declare variable RD_ID integer;
declare variable IS_IPF STR_255;
declare variable C_ID ID_TYPE;
declare variable K_ID ID_TYPE;
declare variable K_AMOUNT_APPROVED CURR_TYPE;
declare variable SUM_AHEAD CURR_TYPE;
declare variable REG_ID integer;
declare variable JK_SYSTEM_CODE varchar(100);
declare variable IS_KM smallint;
declare variable RD_CODE STR_255; /* declare variable sum_fact curr_type; */
begin
  rep_date_begin = coalesce(rep_date_begin, current_date);
  dt_beg = dateadd(1 - extract(day from rep_date_begin) day to rep_date_begin);
  dt_end = dateadd(month, 1, dt_beg) - 1;
  yr_calc = extract(year from rep_date_begin);
  mn_calc = extract(month from rep_date_begin);
  for select rd_id, rd_code, rd_name
      from refdata
      where rd_code in ('PlanIndicators.Disb_amount_bonus', 'PlanIndicators.LO_Count', 'PlanIndicators.BalancePar31_360')
      order by rd_id
      into :rd_id, :rd_code, :rd_name
  do
  begin
    for select reg_id, reg_name
        from okt_office_list

        into :reg_id, reg_name
    do
    begin

      sum_plan = 0;
      sum_fact = 0;
      sum_ahead = 0;
      un_id = null;
      select u.un_id
      from hr_pstatus h
      inner join username u on u.un_from_c_id = h.ps_from_c_id
      inner join joblist jl on jl.jl_id = h.ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where h.ps_is_working = 1 and
            h.ps_status = 1 and
            :rep_date_begin between h.ps_date_begin and h.ps_date_end and
            jk.jk_system_code containing 'JobSystem.Region_KM' and
            h.ps_from_reg_id = :reg_id rows 1
      into :un_id;
      if (un_id is null) then
        continue;

      c_id = (select un_from_c_id
              from username
              where un_id = :un_id);
      if (c_id is null) then
      begin
        suspend;
        exit;
      end

      is_km = 1;
      jk_system_code = 'JobSystem.Region_KM';

      is_ipf = iif(rd_code in ('PlanIndicators.Disb_Amount', 'PlanIndicators.Disb_amount_bonus'), 0, 1);
      if (rd_code <> 'PlanIndicators.BalancePar31_360') then
        sum_plan = coalesce((select sum(pd_data_1)
                             from hp_tmp_plandata_show(:reg_id, :rep_date_begin, :rd_id, 100 * :yr_calc + :mn_calc, 1, :un_id) -- было  1000000, :dt_calc, :rd_id, 100 * :yr_calc + :mn_calc, 1, :un_id)
                             where pd_from_c_id = :c_id or
                                   (:is_km = 1 and
                                   pd_from_c_id = 0)), 0);

      else
        sum_plan = coalesce((select sum(pd_data_1)
                             from hp_tmp_plandata_show(:reg_id, null, :rd_id, 100 * :yr_calc + :mn_calc, 0, 0)
                             where pd_is_office = 1), 0);

      if (sum_plan = 0) then
        continue;
      if (rd_code in ('PlanIndicators.Disb_Amount', 'PlanIndicators.Disb_Amount_IPF', 'PlanIndicators.Disb_amount_bonus')) then
      begin
        sum_fact = 0;
        sum_ahead = 0;
        for select k_id, k_amount_approved
            from kontract
            inner join product on pr_id = k_from_pr_id and
                  iif(pr_code containing '.ipf.', 1, 0) = :is_ipf
            inner join kontstate on ks_from_k_id = k_id and
                  ks_ol_is_delete = 0 and
                  ks_state = 1 and
                  ks_date_begin between :dt_beg and :dt_end and
                  (ks_kc_id = :c_id or (:is_km = 1 and
                  ks_region_id = :reg_id))
            into :k_id, :k_amount_approved
        do
        begin
          sum_fact = sum_fact + k_amount_approved;
        end
        if (rd_code in ('PlanIndicators.Disb_Amount', 'PlanIndicators.Disb_Amount_IPF')) then
        begin
          select -sum(ok_os)
          from operkred, kontract, knaktlist, product
          where ok_ol_is_active = 1 and
                ok_ol_date between :dt_beg and :dt_end and
                ok_os < 0 and
                ok_type in (4, 9) and
                ok_sh_date > :dt_end and
                ok_from_k_id = k_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                ok_ol_date between kal_date and kal_date_end and
                k_from_pr_id = pr_id and
                ((:is_ipf = 1 and
                pr_code containing '.ipf.') or
                (:is_ipf = 0 and
                pr_code not containing '.ipf.')) and
                ((:is_km = 1 and
                kal_new_reg_id = :reg_id) or
                kal_new_kf_officer = :c_id)
          into sum_ahead;

          if (sum_ahead is null) then
            sum_ahead = 0;
          sum_plan = sum_plan + sum_ahead;
          sum_ost = sum_plan + sum_ahead - sum_fact;
        end
        if (rd_code in ('PlanIndicators.Disb_amount_bonus')) then
        begin
          sum_ost = sum_plan - sum_fact;
          sum_ost = iif(sum_ost > 0, sum_ost, 0);
        end
      sum_fact = round(sum_fact/1000, 2);
      sum_plan = round(sum_plan/1000, 2);
      sum_ost = round(sum_ost/1000, 2);
      end

      if (rd_code in ('PlanIndicators.LO_Count')) then
      begin
        select --count(h.ps_from_c_id) --общее количество СВЗ   -- заменил на ниже следующее
--               sum(iif(jk_system_code = 'JobSystem.LO_Standart',1,0.5))
               sum(iif(jk_name = 'СВЗ 0,5',0.5,1))
        from hr_pstatus h
        inner join region r on r.reg_unicode = h.ps_from_reg_id
        inner join joblist jl on jl.jl_id = h.ps_from_jl_id
        inner join jobkind jk on jk.jk_id = jl.jl_kind
        where :rep_date_begin between h.ps_date_begin and h.ps_date_end and
              h.ps_is_working = 1 and
              jk.jk_system_code = 'JobSystem.LO_Standart' and
--              jk.jk_system_code in ('JobSystem.LO_Standart','JobSystem.LO_Standart05') and
              r.reg_unicode = :reg_id and
              ps_from_c_id not in (select hrv.hrv_from_c_id -- ИСКЛЮЧАЕМ
                                   from hr_voctravel hrv, sa_worktype swt, hr_orders hr --  •    Отпуск без содержания по уходу за ребенком
                                   where hrv.hrv_date_begin <= hrv.hrv_date_end_fact and --  •    Приказ_отпуск_по_уходу_за_ребенком
                                         hrv.hrv_status = 1 and --  •    Декретный отпуск.
                                         ho_order_type = sawt_id and
                                         hrv_from_ho_id = ho_id and --     sawt_base_type =5 чтобы исключить 21-ю позицию в кг и  аз
                                         :rep_date_begin between hrv.hrv_date_begin and hrv.hrv_date_end_fact and
                                         swt.sawt_id in (16, 17, 18, 21) and
                                         swt.sawt_base_type = 5)
        into :sum_fact;
        sum_ost = sum_plan - sum_fact; --     sum_fact_1 = sum_plan - sum_fact;
        sum_ost = iif(sum_ost > 0, sum_ost, 0);
      sum_fact = round(sum_fact, 0);
      sum_plan = round(sum_plan, 0);
      sum_ost = round(sum_ost, 0);
      end

      if (rd_code in ('PlanIndicators.BalancePar31_360')) then
      begin
        select sum(oks_balance) oks_balance_par30
        from oksummary oks
        where :rep_date_begin between oks.oks_date_begin and oks.oks_date_end and
              oks.oks_region_id = :reg_id and
              :rep_date_begin - oks_delinq_date between 31 and 360
              and oks_close_state = 0
        into :sum_fact;
        sum_ost = sum_fact - sum_plan; -- sum_fact_1 = sum_plan - sum_fact;
        sum_ost = iif(sum_ost > 0, sum_ost, 0);
      sum_fact = round(sum_fact/1000, 2);
      sum_plan = round(sum_plan/1000, 2);
      sum_ost = round(sum_ost/1000, 2);
      end

      suspend;
    end
  end
end
