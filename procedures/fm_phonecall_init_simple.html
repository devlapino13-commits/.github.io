CREATE PROCEDURE FM_PHONECALL_INIT_SIMPLE(
    PHONE_NO D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    ERR_TEXT D_STR_255,
    PRV_PHONE_NO D_STR_CODE)
AS
declare variable pnc_src_c_id d_id;
declare variable pnc_src_phone_id d_integer;
declare variable is_delinq d_bool;
declare variable c_id bigint;
declare variable phone_id bigint;
begin
  err_text = '';

  select un_from_c_id
  from username
  where un_id = :un_id
  into pnc_src_c_id;

  if (not exists(select * from hp_db_is_actual)) then
  begin
    err_text = 'Вы подключены к тестовой базе. Звонки недоступны.';
    suspend;
    exit;
  end

  phone_id = f_cont_id(phone_no, 'null');
  if (phone_id is not null) then
  begin
    -- Ищем клиента по подтвержденному номеру телефона
    -- Приоритет у основного номера, который был добавлен последним
    select cda_from_c_id
    from contdata
    where cda_from_clt_id = :phone_id
      and cda_is_delete = 0
      and cda_cont_kind = 0
      and cda_status = 1
    order by cda_is_main desc, cda_id desc
    rows 1
    into c_id;
  end
  else
    phone_id = f_cont_id(phone_no, 'add');

  -- Находим номер телефона пользователя
  pnc_src_phone_id = f_cont_id(f_cont_info('clients', pnc_src_c_id, 'cell_main', 'ok'), 'null');

  if (coalesce(pnc_src_phone_id, 0) = 0) then
  begin
    err_text = 'У вас нет подтвержденного основного номера телефона, звонок невозможен';
    suspend;
    exit;
  end

  is_delinq = 0;
  if (c_id is not null) then
  begin
    select is_delinq
    from hp_is_client_delinqed(:c_id)
    into is_delinq;

    if (is_delinq = 1) then
    begin
      err_text = 'Клиент в просрочке, звонки запрещены.';
      suspend;
      exit;
    end
  end

  -- RFCRU-4112
  select coalesce(max(is_delinq), 0)
  from contacts, hp_is_client_delinqed(cc_from_c_id)
  where cc_cell_no = :phone_no
  into is_delinq;

  if (is_delinq = 1) then
  begin
    err_text = 'Контакт клиента в просрочке, звонки запрещены.';
    suspend;
    exit;
  end

  -- № телефона провайдера генерируется случайным образом
  -- У Oivo и MBulak разные пулы номеров
  if (f_is_employee_oivo(pnc_src_c_id, current_date) = 1) then
    prv_phone_no = '74952258057';
  else
    prv_phone_no = decode(round(rand() * 7),
      0, '74952216801',
      1, '74952216802',
      2, '74952217205',
      3, '74952217319',
      4, '74952217829',
      5, '74952217841',
      6, '74952258063',
      7, '74952258089');

  c_id = coalesce(c_id, 0); -- Чтобы обойти not null constraint в pnc_dst_c_id

  insert into phonecalls (
    pnc_src_phone_id, pnc_dst_phone_id, pnc_prv_phone_id, pnc_src_c_id,
    pnc_dst_c_id, pnc_direction, pnc_object_kind, pnc_object_id, pnc_is_delinq,
    pnc_is_contperson)
  values (:pnc_src_phone_id, :phone_id, f_cont_id(:prv_phone_no, 'add'),
    :pnc_src_c_id, :c_id, 1, 0, :c_id, :is_delinq, 0);
  suspend;
end
