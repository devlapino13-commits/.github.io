CREATE PROCEDURE AAREP_ANTIFROD_INCIDENT_LIST(
    DATE_BEGIN D_DATE,
    DATE_END D_DATE)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    BIRTH_DATE D_DATE,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID,
    KONTRACT_NO D_STR_CODE,
    CHECK_DATETIME D_DATETIME,
    RES_INCIDENT D_STR_LONG,
    RES_INT_INV D_STR_LONG,
    RES_RECOMENDATION D_STR_LONG,
    AIM_ID D_ID,
    AIM_REQ_STATUS D_STR_CODE,
    AIM_KUSP_NUMBER D_STR_64,
    AIM_TKT_NUMBER D_STR_64,
    AIM_REQ_DATETIME D_DATETIME,
    AIM_REQ_STATUS_CODE D_STR_CODE,
    IS_SENDFC D_BOOL)
AS
begin
  for
  -- Антифрод
    select afl_datetime, 9, afl_from_k_id,
      'Антифрод. Высокий уровень риска', 'Отказ в заключении договора микрозайма', '', null, null, null, null, null, f_enum('aim_status', 1, 'code'),
      1 as is_sendfc
    from antifrod_loan
    where afl_risk_level = 2
      and afl_datetime between :date_begin and :date_end + 1
  union all
  -- ФинЦЕРТ
    select fcl_datetime, fcl_object_kind, fcl_object_id,
      'Совпадение со списком Финцерт: ' || f_enum('fincert_file', fcl_fid, 'name'),
      'Отказ в заключении договора микрозайма', '', null, null, null, null, null, f_enum('aim_status', 1, 'code'),
      1 as is_sendfc
    from fincert_check_log
    where fcl_fid >= 0
      and fcl_datetime between :date_begin and :date_end + 1
  union all
  -- ручное добавление инцидента
    select aim_check_datetime, aim_object_kind, aim_object_id, aim_incident_str, aim_int_inv_str, aim_recomendation, aim_id, f_enum('aim_status', aim_req_status, 'name'), aim_kusp_number, aim_tkt_number, aim_req_datetime, f_enum('aim_status', aim_req_status, 'code'),
      aim_is_sendfc as is_sendfc
    from antifrod_incident_manual
    where aim_is_delete = 0
      and aim_create_datetime between :date_begin and :date_end + 1
  order by 1
  into check_datetime, object_kind, object_id, res_incident, res_int_inv, res_recomendation, aim_id, aim_req_status, aim_kusp_number, aim_tkt_number, aim_req_datetime, aim_req_status_code, is_sendfc
  do
  begin
    if (object_kind = 9) then
    begin
      for select c_id, c_name_fio, c_birth_date, k_kontract_no
      from kontract, knclient, clients
      where kc_from_k_id = k_id
        and c_id = kc_from_c_id
        and k_id = :object_id
      into c_id, c_name_fio, birth_date, kontract_no
      do
        suspend;
    end 
    else if (object_kind = 11) then
    begin
      for select c_id, c_name_fio, c_birth_date, ma_no
      from miniapp, miniappclient, clients
      where mac_from_ma_id = ma_id
        and c_id = mac_from_c_id
        and ma_id = :object_id
        and ma_number = 0
      into c_id, c_name_fio, birth_date, kontract_no
      do
        suspend;
    end
    else
      exception e_default 'Unknown object_kind = ' || coalesce(object_kind, '<NULL>');
  end
end
