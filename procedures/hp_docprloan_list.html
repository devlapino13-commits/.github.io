CREATE PROCEDURE HP_DOCPRLOAN_LIST(
    ONLY_ACTIVE D_BOOL = null)
RETURNS (
    PR_ID D_INTEGER,
    PR_NAME D_STR_100,
    PR_MEMO D_STR_255,
    PR_IS_DELETE D_BOOL,
    PR_DATE_BEGIN D_DATE,
    PR_DATE_END D_DATE,
    DPL_COUNT_STR D_STR_CODE)
AS
declare variable dpl_count d_integer;
begin
  if (only_active is null) then
    only_active = 0;

  for select pr_id, pr_name,
             pr_common_name ||
             iif(coalesce(pr_name, '') <> coalesce(pr_memo, '') and coalesce(pr_memo, '') <> '', ', ' || pr_memo, ''),
             iif(pr_date_end >= current_date, 0, 1),
             pr_date_begin, pr_date_end
  from product
  order by 4, pr_date_begin desc, pr_name
  into pr_id, pr_name,
       pr_memo, pr_is_delete,
       pr_date_begin, pr_date_end
  do
  begin
    if (only_active = 1 and pr_is_delete = 1) then
      Continue;

    if (pr_date_end >= '31.12.2999') then
      pr_date_end = null;

    dpl_count_str = '';

    dpl_count = 0;
    select count(*)
    from docprloan, docprop
    where dpl_from_pr_id = :pr_id and    
          dpl_client_kind = 0 and -- Условия для клиента
          dpl_is_delete = 0 and
          dpl_from_dp_id = dp_id and
          dp_is_base = 0
    into dpl_count;

    dpl_count_str = dpl_count;

    dpl_count = 0;
    select count(*)
    from docprloan, docprop
    where dpl_from_pr_id = :pr_id and    
          dpl_client_kind = 1 and -- Условия дя залогодателя
          dpl_is_delete = 0 and
          dpl_from_dp_id = dp_id and
          dp_is_base = 0
    into dpl_count;

    if (dpl_count <> 0) then
      dpl_count_str = dpl_count_str || ' / ' || dpl_count;

    dpl_count = 0;
    select count(*)
    from docprloan, docprop
    where dpl_from_pr_id = :pr_id and    
          dpl_client_kind = 2 and -- Условия дя поручителя
          dpl_is_delete = 0 and
          dpl_from_dp_id = dp_id and
          dp_is_base = 0
    into dpl_count;

    if (dpl_count <> 0) then
      dpl_count_str = dpl_count_str || ' / ' || dpl_count;

    if (dpl_count_str = '0') then
      dpl_count_str = '';

    suspend;
  end
end
