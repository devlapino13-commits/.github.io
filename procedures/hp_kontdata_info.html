CREATE PROCEDURE HP_KONTDATA_INFO(
    KONTRACT_NO VARCHAR(20),
    REP_DATE_END DATE,
    REP_PERCENT_PER_DAY PERC_TYPE)
RETURNS (
    K_ID ID_TYPE,
    PR_PERCENT_FOR_DAYS INTEGER,
    REG_NAME VARCHAR(100),
    KD_PERCENT_PER_DAY PERC_TYPE,
    KD_PERCENT PERC_TYPE,
    KD_FINE_PERCENT PERC_TYPE,
    K_MEMO VARCHAR(250),
    K INTEGER,
    KAL_NEW_REG_ID INTEGER)
AS
declare variable k_from_pr_id integer;
declare variable k_from_kf_id id_type;
begin
  k = 0;
  for select k_id, k_from_pr_id, k_from_kf_id
      from kontract
      where k_kontract_no = :kontract_no
      into k_id, k_from_pr_id, k_from_kf_id
  do
  begin
    pr_percent_for_days = 0;
    select pr_percent_for_days
    from product
    where pr_id = :k_from_pr_id
    into pr_percent_for_days;

    select kal_new_reg_id
    from knaktlist
    where kal_from_kf_id = :k_from_kf_id and
          :rep_date_end between kal_date and kal_date_end and
          knaktlist.is_delete = 0
    into kal_new_reg_id;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :kal_new_reg_id
    into reg_name;

    select kd_percent_per_day, kd_percent, kd_fine_percent
    from kontdata
    where kd_from_k_id = :k_id and
          kd_ol_is_delete = 0 and
          :rep_date_end between kd_ol_date and kd_date_end
    into kd_percent_per_day, kd_percent, kd_fine_percent;
    k = k + 1;
  end

  if (k = 0) then
  begin
    k_memo = 'Контракт не найден!';
  end
  else
  if (k > 1) then
  begin
    k_id = null;
    pr_percent_for_days = null;
    reg_name = null;
    kd_percent_per_day = null;
    kd_percent = null;
    k_memo = 'Под этим номером существуют более одного контракта!';
  end
  else
  if (not exists(select *
                 from kontstate
                 where ks_ol_is_delete = 0 and
                       ks_state = 1 and
                       :rep_date_end between ks_date_begin and ks_date_end and
                       ks_from_k_id = :k_id)) then
  begin
    k_memo = 'Статус контракта не активный!';
    k = 0;
  end
  else
  if (kd_percent_per_day = coalesce(rep_percent_per_day, 0)) then
  begin
    k_memo = 'Процентная ставка уже снижена!';
    k = 0;
  end
  else
  if (kd_percent_per_day < coalesce(rep_percent_per_day, 0)) then
  begin
    k_memo = 'Процентная ставка УВЕЛИЧЕНО!';
    k = 999;
  end
  else
    k_memo = 'Успешно импортирован!';

  suspend;
end
