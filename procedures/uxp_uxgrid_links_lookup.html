CREATE PROCEDURE UXP$UXGRID_LINKS_LOOKUP(
    GRID_UUID D_UUID)
RETURNS (
    ID D_STR_CODE,
    TEXT D_STR_255,
    KIND D_STR_CODE)
AS
declare variable ufr_uuid d_uuid;
declare variable ufm_uuid d_uuid;
begin
  select ugr_ufr_uuid, ufr_owner
  from ux$grid, ux$frame
  where ugr_uuid = :grid_uuid and
        ugr_ufr_uuid = ufr_uuid
  into ufr_uuid, ufm_uuid;

  kind = 'field';

  for select uf_field_name, uf_name
  from ux$field, ux$frame, ux$frparam
  where ufr_uuid = :ufr_uuid and
        ufp_ufr_uuid = ufr_uuid and
        uf_uuid = ufp_uf_uuid and
        uf_state = 0 and
        uf_is_param = 0 and
        ufp_parent is null
  order by iif(ufr_us_uuid = uf_owner, 0, 1), uf_position
  into id, text
  do
  begin
    text = uxf$join_str (:id, :text);
    suspend;
  end

  kind = 'form';
  for select uxa_fcode, uxa_name
  from ux$action, ux$alias, ux$code ux, ux$uniq
  where ua_owner  in (:ufr_uuid, :ufm_uuid) and
        ua_uxa_uuid = uxa_uuid and
        uxa_owner = ux_uuid and
        ux_uuid = uu_id and
        ua_inactive = 0 and
        ua_usr_id_field = ''
  into id, text
  do
  begin
    text = uxf$join_str (:id, :text);
    suspend;
  end
end
