CREATE PROCEDURE AAREP_KONTRESLIST_REP_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KRL_DATETIME TYPE OF COLUMN KONTRESLIST.KRL_DATETIME,
    KRL_DATE_OLD TYPE OF COLUMN KONTRESLIST.KRL_DATE_OLD,
    KRL_DATE_NEW TYPE OF COLUMN KONTRESLIST.KRL_DATE_NEW,
    KRL_FOLDER_NAME VARCHAR(100),
    KRL_REGION_NAME VARCHAR(100),
    KRL_USER_NAME VARCHAR(100),
    KRL_AMOUNT DECIMAL(18,4),
    KRL_TYPE VARCHAR(20),
    KR_TEXT VARCHAR(255),
    KR_KIND_TEXT VARCHAR(50),
    KR_IS_CERTIF_TEXT VARCHAR(50),
    KRL_OFFICER_NAME VARCHAR(100))
AS
declare variable ma_id_prior integer;
declare variable ma_number integer;
declare variable kr_kind integer;
declare variable kr_is_cercificated integer;
declare variable krl_id integer;
declare variable krl_from_kr_id integer;
declare variable krl_from_un_id decimal(18,0);
declare variable krl_from_k_id decimal(18,0);
declare variable krl_ma_reg_id integer;
declare variable krl_ma_id integer;
declare variable kal_new_kf_officer id_type;
begin
  krl_type = 'Догов.в процессе';
  for select krl_id, krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_id, krl_datetime, krl_date_old, krl_date_new,
             kal_new_kf_name, kal_new_kf_officer, reg_code || ' ' || reg_name, un_name, k_amount_approved, kr_kind,
             kr_is_cercificated, kr_text
      from kontreslist, knaktlist, kontract, region_list(:rep_region_id, 1, -1, 1), region, username, kontres
      where krl_date_old between :rep_date_begin and :rep_date_end and
            krl_from_k_id = k_id and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= krl_date_old and
            kal_date_end >= krl_date_old and
            knaktlist.is_delete = 0 and
            kal_new_reg_id = id_list and
            kal_new_reg_id = reg_unicode and
            krl_from_un_id = un_id and
            krl_from_kr_id = kr_id
      into :krl_id, :krl_from_kr_id, :krl_from_un_id, :krl_from_k_id, :krl_ma_id, :krl_datetime, :krl_date_old,
           :krl_date_new, :krl_folder_name, :kal_new_kf_officer, :krl_region_name, :krl_user_name, :krl_amount,
           :kr_kind, :kr_is_cercificated, :kr_text

  do
  begin
    if (kr_kind = 0) then
      kr_kind_text = 'Перенос даты';
    if (kr_kind = 1) then
      kr_kind_text = 'Отказ';
    if (kr_kind = 2) then
      kr_kind_text = 'Аннулирование';
    if (kr_is_cercificated = 0) then
      kr_is_certif_text = 'До сертификации';
    else
      kr_is_certif_text = 'После сертификации';

    select coalesce(c_name_fio, 'не определен')
    from clients
    where c_id = :kal_new_kf_officer
    into krl_officer_name;

    suspend;
  end
  ma_id_prior = 0;
  krl_type = 'Предв.заявка';
  for select distinct krl_id, krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_id, krl_datetime, krl_date_old,
                      krl_date_new, kal_new_kf_name, kal_new_kf_officer, reg_code || ' ' || reg_name, un_name,
                      ma_summa_minus, ma_number, kr_kind, kr_is_cercificated, kr_text
      from kontreslist, knaktlist, miniapp, region_list(:rep_region_id, 1, -1, 1), region, username, kontres
      where krl_date_old between :rep_date_begin and :rep_date_end and
            krl_ma_id = ma_id and
            kal_from_kf_id = ma_from_kf_id and
            kal_date <= krl_date_old and
            kal_date_end >= krl_date_old and
            knaktlist.is_delete = 0 and
            kal_new_reg_id = id_list and
            kal_new_reg_id = reg_unicode and
            krl_from_un_id = un_id and
            ma_date_save <= krl_datetime and
            krl_from_kr_id = kr_id
      order by krl_ma_reg_id, krl_ma_id, ma_number desc
      into :krl_id, :krl_from_kr_id, :krl_from_un_id, :krl_from_k_id, :krl_ma_id, :krl_datetime, :krl_date_old,
           :krl_date_new, :krl_folder_name, :kal_new_kf_officer, :krl_region_name, :krl_user_name, :krl_amount,
           :ma_number, :kr_kind, :kr_is_cercificated, :kr_text
  do
  begin
    if (kr_kind = 0) then
      kr_kind_text = 'Перенос даты';
    if (kr_kind = 1) then
      kr_kind_text = 'Отказ';
    if (kr_kind = 2) then
      kr_kind_text = 'Аннулирование';
    if (kr_is_cercificated = 0) then
      kr_is_certif_text = 'До сертификации';
    else
      kr_is_certif_text = 'После сертификации';

    if (ma_id_prior <> krl_ma_id) then
    begin
      select coalesce(c_name_fio, 'не определен')
      from clients
      where c_id = :kal_new_kf_officer
      into krl_officer_name;

      suspend;
    end
    ma_id_prior = :krl_ma_id;
  end
end
