CREATE PROCEDURE AAREP_DOSROCH_CLOSED(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    KF_NAME VARCHAR(50),
    K_FROM_KF_ID DECIMAL(18,0),
    KF_REGION_NAME VARCHAR(50),
    K_CLOSE_DATE DATE,
    K_ID DECIMAL(18,0),
    K_KREDLINE_END DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    NEW_KONT_AMOUNT_APPROVED DECIMAL(18,4),
    NEW_KONT_DISB_DATE DATE,
    NEW_KONT_ID DECIMAL(18,0),
    LAST_PAY_SUMM DECIMAL(18,4),
    K_KONTRACT_NO VARCHAR(50),
    NEW_KONTRACT_NO VARCHAR(50),
    CASHIER_FIO VARCHAR(100))
AS
declare variable reg_prior integer;
declare variable parent_id integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable close_ol_date date;
declare variable close_ol_id decimal(18,0);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  reg_prior = 0;
  for select distinct reg_name, k_id, kont.k_kontract_no, kal_new_kf_name, k_kredline_end, ks_date_begin, k_from_kf_id,
                      kont.k_amount_approved, ks.ks_from_ol_id
      from kontstate ks, kontract kont, operkred ok, knaktlist, region, region_list(:report_region_id, 1, -1, 1)
      where ks_from_k_id = k_id and
            ks_state = 2 and
            ks_date_begin < k_kredline_end and
            ok_from_k_id = k_id and
            kal_from_kf_id = k_from_kf_id and
            :report_date_begin >= kal_date and
            :report_date_end <= kal_date_end and
            (id_list = reg_unicode) and
            kal_new_reg_id = reg_unicode and
            ks_date_begin between :report_date_begin and :report_date_end and
            knaktlist.is_delete = 0 and
            ks.ks_ol_is_delete = 0 and
            ok_from_c_id = 0
      into :kf_region_name, :k_id, :k_kontract_no, :kf_name, :k_kredline_end, :k_close_date, :k_from_kf_id,
           :k_amount_approved, :close_ol_id
  do
  begin
    cashier_fio = '';
    select cl.c_name_fio
    from knclient kncl
    inner join clients cl on cl.c_id = kncl.kc_from_c_id
    where kncl.kc_from_k_id = :k_id and
          kncl.kc_status = 2 and
          kncl.kc_amount_approved > 0 rows 1
    into :cashier_fio;

    select ol.ol_date
    from operlist ol
    where ol.ol_id = :close_ol_id
    into :close_ol_date;

    last_pay_summ = 0;

    select sum(-ok.ok_os + -ok.ok_perc + -ok.ok_fine)
    from operkred ok
    where ok.ok_from_k_id = :k_id and
          ok.ok_ol_date = :close_ol_date and
          ok.ok_ol_is_active = 1 and
          ok.ok_type in (4, 9) and
          ok_from_c_id = 0
    into :last_pay_summ;

    new_kont_id = null;
    new_kont_disb_date = null;
    new_kont_amount_approved = null;
    new_kontract_no = null;

    select k_id, kont.k_kontract_no, ks.ks_date_begin, kont.k_amount_approved
    from kontract kont
    inner join kontstate ks on ks.ks_from_k_id = kont.k_id and
          k_from_kf_id = :k_from_kf_id and
          ks.ks_ol_is_delete = 0 and
          ks.ks_date_begin >= :k_close_date and
          ks.ks_state = 1 and
          kont.k_id <> :k_id
    order by ks_date_begin
    rows 1
    into :new_kont_id, :new_kontract_no, :new_kont_disb_date, :new_kont_amount_approved;

    if (new_kont_amount_approved is null) then
      new_kont_amount_approved = 0;

    suspend;
  end
end
