CREATE PROCEDURE HP_DO_KN_OKSUMMARY_ADD_LAW(
    REP_DATE D_DATE)
AS
declare variable kc_id d_id;
declare variable oks_date_begin d_date;
declare variable prv_month_end d_date;
begin
  prv_month_end = rep_date - extract(day from rep_date);

  for select distinct ox_from_kc_id
      from kn_operextra
      into :kc_id
  do
  begin
    -- kn_oksummary
    oks_date_begin = null;
    select oks_date_begin
    from kn_oksummary
    where oks_date_begin = :prv_month_end and
          oks_date_end = '31.12.9999' and
          oks_from_kc_id = :kc_id and
          oks_bal_law <> 0
    into :oks_date_begin;

    if (oks_date_begin < rep_date) then
      execute procedure hp_kn_oksummary_insert(:kc_id, :rep_date);
  end
end
