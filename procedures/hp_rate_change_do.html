CREATE PROCEDURE HP_RATE_CHANGE_DO(
    CHANGE_DATE D_DATE,
    OL_MEMO D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable k_kontract_no d_str_20;
declare variable k_fact_disb_date d_date;
declare variable trc_k_id d_id;
declare variable trc_perc_fresh d_percent;
declare variable txt d_str_255;
begin
  -- используется в форме "Снижение процентной ставки" (RFCRU-4306)

  if (change_date is null) then
    exception e_message 'Не указана дата изменения';

  if (ol_memo is null or trim(ol_memo) = '') then
    exception e_message 'Не указан комментарий (номер и дата протокола)';

  for select trc_k_id,
             trc_perc_fresh
      from temp_rate_change
      into trc_k_id,
           trc_perc_fresh
  do
  begin
    k_kontract_no = null;
    k_fact_disb_date = null;
    txt = null;

    select k_kontract_no,
           k_fact_disb_date
    from kontract
    where k_id = :trc_k_id
    into k_kontract_no,
         k_fact_disb_date;

    if (change_date <= k_fact_disb_date) then
      exception e_message 'Дата изменения раньше или равна дате выдачи договора ' || k_kontract_no || ' от ' || f_datetostr(k_fact_disb_date);

    begin
      select res_text
      from hp_do_reschedule_new_perc(:trc_k_id, :change_date, :trc_perc_fresh, :ol_memo, :un_id)
      into txt;

      update temp_rate_change
      set trc_note = :txt
      where trc_k_id = :trc_k_id;

      when any do
      begin
        res_is_error = 1;
        res_text = 'При изменении ставок произошла ошибка (hp_rate_change_do)';
        suspend;
        exit;
      end
    end
  end

  res_is_error = 0;
  res_text = 'Изменение ставок завершено.' || ascii_char(10) || 'Результаты приведены в колонке "Примечание"';

  suspend;
end
