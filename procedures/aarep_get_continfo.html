CREATE PROCEDURE AAREP_GET_CONTINFO(
    OBJECT_KIND VARCHAR(100),
    OBJECT_ID DECIMAL(18,0))
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_PLASTCARD_NO VARCHAR(100),
    C_EMAIL VARCHAR(100),
    C_SKYPE VARCHAR(100),
    C_CELL_IS_COMMITED INTEGER,
    C_CELL_NO_DATE DATE,
    C_CELL_IS_ABSENT INTEGER,
    C_EMAIL_VERIFIED INTEGER,
    C_EMAIL_VERIDY_DATETIME TIMESTAMP)
AS
declare variable c_id decimal(18,0);
declare variable cell_ok d_str_100;
declare variable email_ok d_str_100;
begin
  if (object_kind = 'clients' or object_kind = '0') then
  begin
    object_kind = 0;
    c_id = object_id;
  end
  if (object_kind = 'username') then
  begin
    select un_from_c_id
    from username
    where un_id = :object_id
    into :c_id;
    object_kind = '0';
  end
  if (c_id is null) then
    exit;

  c_name_fio = '';
  c_plastcard_no = '';
  c_email = '';
  c_skype = '';
  c_cell_is_commited = 0;
  c_cell_no_date = null;
  c_cell_is_absent = 0;
  c_email_verified = 0;
  c_email_veridy_datetime = null;

  select c_name_fio,
         f_cont_info('clients', c_id, 'cell_main', 'last'),
         f_cont_info('clients', c_id, 'email', 'last'),
         f_cont_info('clients', c_id, 'skype', 'last'),
         f_cont_info('clients', c_id, 'cell_main', 'ok'),
         f_cont_info('clients', c_id, 'email', 'ok')
  from clients
  where c_id = :c_id
  into c_name_fio, c_plastcard_no, c_email, c_skype, cell_ok, email_ok;

  c_cell_is_commited = 0;
  c_email_verified = 0;
  c_cell_is_absent = 0;

  if (c_plastcard_no = cell_ok) then c_cell_is_commited = 1;
  if (c_email = email_ok) then c_email_verified = 1;
  if (c_plastcard_no = '') then c_cell_is_absent = 1;

  if (f_get_const('new_contdata') = 0) then
  begin
    select res_date
    from hp_get_client_continfo(:c_id, 'ContactInfo.Cell_main')
    into c_cell_no_date;

    select res_datetime
    from hp_get_client_continfo(:c_id, 'ContactInfo.Email')
    into c_email_veridy_datetime;
  end else
  begin
    select cda_date
    from contdata
    where cda_from_c_id = :c_id and
          cda_cont_kind = 'cell' and
          cda_from_clt_id = f_cont_id(:c_plastcard_no, 'null') and
          cda_is_delete = 0 and
          cda_is_main = 1
    into c_cell_no_date;

    select cda_date
    from contdata, contveri
    where cda_from_c_id = :c_id and
          cda_cont_kind = 'email' and
          cda_from_clt_id = f_cont_id(:c_email, 'null') and
          cda_is_delete = 0 and
          cvr_from_cda_id = cda_id and
          cvr_status = 1
    order by cvr_datetime desc
    rows 1
    into c_email_veridy_datetime;
  end

  suspend;
end
