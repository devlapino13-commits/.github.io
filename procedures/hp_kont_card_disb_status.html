CREATE PROCEDURE HP_KONT_CARD_DISB_STATUS(
    K_ID D_ID)
RETURNS (
    STATUS D_INTEGER,
    DESCRIPTION D_STR_8100,
    CLIENT_TOTAL D_INTEGER,
    CLIENT_DONE D_INTEGER,
    CLIENT_ERROR D_INTEGER,
    CLIENT_CARDS D_SMALLINT)
AS
declare variable ESM_STATUS D_INTEGER;
declare variable ESM_ID D_ID;
declare variable C_ID D_ID;
declare variable C_NAME_FIO D_STR_100;
declare variable ERR_CODE D_STR_255;
declare variable ESM_ACTION D_STR_CODE;
declare variable ESM_FOUND D_SMALLINT;
declare variable DB_ID D_INTEGER;
declare variable PAY_PROJECT D_STR_100;
declare variable ESM_ACCOUNT D_STR_CODE;
declare variable K_DISB_KIND D_SMALLINT;
begin
  status = 2; -- нет строк на исполнение ("перевести деньги на карту" была создана)
  description = '';

  client_total = 0; -- всего клиентов
  client_done = 0;  -- скольким успешно отправлены
  client_error = 0; -- сколько с ошибками
  client_cards = 0; -- сколько с картами

  db_id = f_get_db_id();

  pay_project = (case db_id when 41701 then 'elcard_disb'
                            when 39801 then 'elcard_disb'
                 end);

  select k_disb_kind from kontract
  where k_id = :k_id
  into k_disb_kind;

  for select kncl.kc_from_c_id, cl.c_name_fio
      from knclient kncl
      inner join clients cl on cl.c_id = kncl.kc_from_c_id
      where kncl.kc_amount_approved > 0 and
            kncl.kc_from_k_id = :k_id
      into :c_id, :c_name_fio
  do
  begin
    esm_found = 0; -- возможно нет записи в ExtSysMain
    client_total = client_total + 1;

    if (db_id = 81001 and k_disb_kind = 7) then
    begin
      if ((select coalesce(fpb_code, '')
      from hp_fps_get_loan_bank(:k_id, :c_id)) <> '') then
          client_cards = client_cards + 1;
    end
    else
    begin
      if ((select coalesce(token, '')
           from hp_client_bank_card(:c_id)) <> '') then
          client_cards = client_cards + 1;
    end

    -- ищу задачу связанную с переводом денег на карту
    for select em.esm_status, em.esm_id, em.esm_action, em.esm_account
    from extsysmain em
    where em.esm_from_k_id = :k_id and
          em.esm_object_kind = f_objkind('clients', 'id') and
          em.esm_object_id = :c_id and
          (
            (:db_id = 81001 and em.esm_project in ('moneta_disb', 'wirebank_disb')) or
            (:db_id <> 81001 and em.esm_project = :pay_project)
          )
    order by esm_id desc
    rows 1
    into :esm_status, :esm_id, :esm_action, :esm_account
    do
    begin
      esm_found = 1; -- Запись найден
      status = minvalue(status, esm_status); -- как обший статус берем наименьшое значение среди всех клиентов

      if ((esm_status = 1) and (esm_action in ('CardDisb', 'PhoneDisb'))) then
         client_done = client_done + 1;

      if (esm_status <> -1) then -- если не ошибка, выходим из этого for select
        Break;

      err_code = '';
      client_error = client_error + 1;

      select left(trim(replace(ea.esa_message, 'error:', '')), 255)
      from extsysanswer ea
      where ea.esa_from_esm_id = :esm_id and
            ea.esa_message like '%error:%'
      into :err_code;

      if (description <> '') then
        description = description || ascii_char(10) || ascii_char(13); -- перенос строки

      if (db_id in (39801, 41701)) then
      begin
        description = description || '[' || c_name_fio || '] ' ||
          'карта: ' || esm_account || '; ' ||
          'код ошибки: ' || err_code || '; ' ||
          trim(iif(esm_action = 'CardDisb', 'Ошибка на этапе выдачи:', 'Ошибка на этапе проверки:'));
        description = description || ' ' || (case err_code
                                         when '100' then 'Произошла недокументированная ошибка' ||
                                                         ' (' ||
                                                         (select first 1
                                                            f_json_str_get_value(replace(esa_message, 'answer:', ''), 'message', '')
                                                          from extsysanswer where esa_from_esm_id = :esm_id
                                                            and esa_message starting 'answer:'
                                                         ) || ')'
                                         when '101' then 'Произошла внутренняя ошибка'
                                         when '102' then 'Сервис временно недоступен'
                                         when '103' then 'Ошибка соединения с провайдером'
                                         when '104' then 'Доступ запрещен'
                                         when '201' then 'Абонент не найден'
                                         when '202' then 'Недостаточно средств на счете'
                                         when '203' then 'Превышена сумма платежа'
                                         when '204' then 'Услуга временно недоступна'
                                         when '205' then 'Неверный формат суммы'
                                         when '301' then 'Неверный идентификатор платежного поручения'
                                         else 'Техническая ошибка'
                                       end) || '; ';
      end
      else
      begin
        description = description || '[' || c_name_fio || '] ' ||
          trim(iif(esm_action = 'CardDisb', 'Ошибка на этапе выдачи:', 'Ошибка на этапе проверки:')) || ' ' || :err_code;
      end
    end

    if (esm_found = 0) then
    begin
      if (description <> '') then
        description = description || ascii_char(10) || ascii_char(13); -- перенос строки

      description = description || '[' || c_name_fio || '] Нет записи об оплате';
    end
  end

  if (status = 2) then -- если по всем клиентам отсутствует запись в ExtSysMain, убираем описание
    description = '';

  suspend;
end
