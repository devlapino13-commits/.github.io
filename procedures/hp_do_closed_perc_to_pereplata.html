CREATE PROCEDURE HP_DO_CLOSED_PERC_TO_PEREPLATA(
    K_ID D_ID)
AS
declare variable rep_date d_date;
declare variable db_id d_integer;
declare variable start_date d_date;
declare variable closed_date d_date;
declare variable perc_saldo d_currency;
declare variable reg_id d_integer;
declare variable ol_memo d_str_long;
declare variable un_id d_id;
declare variable ol_id d_id;
declare variable sh_id d_integer;
declare variable sh_date d_date;
declare variable oks_bal_perc d_currency;
declare variable closed_ol_id d_id;
declare variable main_ol_id d_id;
declare variable al_id d_str_code;
declare variable acc_id d_id;
declare variable opl_acc_kind d_integer;
declare variable amount d_currency;
declare variable dt_sys_acc d_str_code;
declare variable kt_sys_acc d_str_code;
declare variable ob_detail d_str_255;
declare variable loan_acc_group d_str_code;
declare variable kc_id d_id;
declare variable dt_acc_id d_id;
declare variable kt_acc_id d_id;
declare variable non_resident d_smallint;
declare variable client_type d_smallint;
declare variable acc_calced d_str_code;
declare variable acc_profit d_str_code;
declare variable acc_payed d_str_code;
declare variable acc_pereplata d_str_code;
declare variable ac_type d_integer;
declare variable pr_id d_integer;
declare variable main_dt_acc d_str_code;
declare variable def_curr d_str_code;
declare variable acc_dt d_str_code;
declare variable acc_kt d_str_code;
declare variable oper_ol_id d_id;
declare variable disb_date d_date;
declare variable ok_number d_integer;
declare variable kn_perc_saldo d_currency;
declare variable is_done d_bool;
begin
  un_id = 1000000000001;
  loan_acc_group = 'Микрозаймы'; -- в будущем нужно прикрепить название раздела системного счета в таблицу продуктов.

  db_id = f_get_db_id();
  start_date = f_tmp_const('new_classic_perc_start_date');

  def_curr = '';
  if (db_id = 81001) then
    def_curr = '810';
  if (db_id = 41701) then
    def_curr = '417';
  if (db_id = 39801) then
    def_curr = '398';
  if (db_id = 94401) then
    def_curr = '944';

  select ks_date_begin
  from kontstate
  where ks_from_k_id = :k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0
  into disb_date;

  select ks_date_begin, ks_region_id, ks_from_ol_id
  from kontstate
  where ks_from_k_id = :k_id and
        ks_state = 2 and
        ks_ol_is_delete = 0
  into closed_date, reg_id, closed_ol_id;
  rep_date = closed_date;

  if (row_count = 0) then
  begin
    update temp_calcperc_fix
    set cpf_status = -5 -- Активный займ
    where cpf_from_k_id = :k_id and
          cpf_status = 0;

    exit;
  end

  select result_perc_end
  from aarep_calc_perc_one_cls(:k_id, :closed_date, :closed_date, 0, 1, 1)
  into :perc_saldo;

  if (perc_saldo >= 0) then
  begin
    update temp_calcperc_fix
    set cpf_status = 1
    where cpf_from_k_id = :k_id;

    exit;
  end

  if (db_id = 81001) then
  begin
    select iif(ol_parent_id is null, ol_id, ol_parent_id)
    from operlist
    where ol_id = :closed_ol_id
    into main_ol_id;

    select opl_acc_kind, opl_from_al_id, opl_from_acc_id
    from operloan
    where opl_from_ol_id = :main_ol_id
    into opl_acc_kind, al_id, acc_id;

    if (al_id is not null and acc_id is null) then
    begin
      if (al_id = '010181070-01000001') then
      begin
        select kob_acc_dt
        from kn_operbook, ref_account
        where kob_from_ol_id = :main_ol_id and
              kob_acc_dt = acc_id and
              acc_code like '4742_8100000%'
        order by kob_id
        rows 1
        into acc_id;
      end

      if (acc_id is null) then
        exception e_default f_format('Dt acc_id not found, k_id: %1, al_id: %2', k_id, al_id);
    end

    if (al_id is null) then
      exception e_default f_format('Dt al_id not found, k_id: %1', k_id);
  end

  select k_kontract_no || ' от ' || f_datetostr(k_disb_date), k_from_pr_id
  from kontract
  where k_id = :k_id
  into ol_memo, pr_id;

  ol_memo = 'Корректировка накопленных процентов по договору ' || ol_memo;

  if (closed_date < start_date and disb_date < start_date) then
  begin
    rep_date = start_date;

    insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind,
                          ol_memo, ol_parent_id, is_delete, ol_user_id)
    values (:reg_id, 1, 13, :start_date, 9, :ol_memo, null, 0, :un_id)
    returning ol_id
    into :ol_id;
  
    update or insert into operloan (opl_from_ol_id, opl_acc_kind, opl_from_al_id, opl_from_acc_id)
    values (:ol_id, :opl_acc_kind, :al_id, :acc_id)
    matching (opl_from_ol_id);
  
    oper_ol_id = ol_id;
    main_ol_id = ol_id;
  end else
    oper_ol_id = closed_ol_id;

  ac_type = 2; -- Начисленные проценты
  select res_account
  from hp_make_kredit_acclist(:pr_id, :reg_id, :ac_type)
  into :acc_calced;

  ac_type = 36; -- Доход процент
  select res_account
  from hp_make_kredit_acclist(:pr_id, :reg_id, :ac_type)
  into :acc_profit;

  ac_type = 3; -- Проценты полученные
  select res_account
  from hp_make_kredit_acclist(:pr_id, :reg_id, :ac_type)
  into :acc_payed;

  ac_type = 8; -- Переплата
  select res_account
  from hp_make_kredit_acclist(:pr_id, :reg_id, :ac_type)
  into :acc_pereplata;

  select ok_sh_id, ok_sh_date
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1
  order by ok_ol_is_active desc, ok_sh_date desc, ok_sh_id desc
  rows 1
  into sh_id, sh_date;

  if (sh_date is null) then
    exception e_default f_format('sh_date not found, k_id: %1', k_id);

  oks_bal_perc = 0;

  if (db_id = 81001) then
  begin
    select sum(oks_bal_perc)
    from kn_oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin = :rep_date
    into oks_bal_perc;
  
    if (oks_bal_perc is null) then
      oks_bal_perc = 0;

    if (oks_bal_perc <> 0) then
      exception e_default f_format('oks_bal_perc <> 0, k_id: %1', k_id);
  end

  insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                        ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                        ok_ol_is_active)
  values (:oper_ol_id, -1, :k_id, :sh_id, :sh_date, 4, 0, -:perc_saldo, 0,
          2, 0, 0, 0, 0, 0, :rep_date, :reg_id,
          1);

  insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                        ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                        ok_ol_is_active)
  values (:oper_ol_id, -1, :k_id, :sh_id, :sh_date, 3, 0, 0, 0,
          2, 0, 0, :perc_saldo, 0, 0, :rep_date, :reg_id,
          1);

  insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                        ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                        ok_ol_is_active)
  values (:oper_ol_id, -1, :k_id, :sh_id, :sh_date, 9, :perc_saldo, 0, 0,
          2, 0, 0, 0, 0, 0, :rep_date, :reg_id, 1)
  returning ok_number
  into ok_number;

  select ob_account_dt
  from operbook
  where ob_from_ol_id = :closed_ol_id
  order by ob_from_ol_id, ob_number
  rows 1
  into main_dt_acc;

  if (main_dt_acc is null) then
    exception e_default f_format('Main_Dt account not found, k_id: %1', k_id);

  if (db_id = 81001) then
  begin
    is_done = 0;

    if (closed_date >= start_date) then
    begin
      update kn_oksummary
      set oks_need_recalc = 1,
          oks_is_fix_bal_perc = 1,
          oks_bal_perc_fixed = 0
      where oks_from_k_id = :k_id and
            oks_date_begin = :rep_date;
  
      execute procedure hp_do_kn_operbook_one_online(:k_id, :rep_date, null, :rep_date - 1);
    end else
    begin
      is_done = 1;

      for select ok_from_kc_id, ok_os
      from kn_operkred
      where ok_from_ol_id = :oper_ol_id and
            ok_number = :ok_number and
            ok_type = 9
      into kc_id, kn_perc_saldo
      do
      begin
        is_done = 1;
        update kn_oksummary
        set oks_need_recalc = 1,
            oks_is_fix_bal_perc = 1,
            oks_bal_perc_fixed = 0
        where oks_from_kc_id = :kc_id and
              oks_date_begin = :rep_date;
    
        select non_resident, client_type
        from hp_get_kn_client_resident(:kc_id, :rep_date)
        into non_resident, client_type;
      
        amount = kn_perc_saldo;
        dt_sys_acc = 'Процент.Начислено';
        kt_sys_acc = 'Процент.Доход';
        ob_detail = 'Сторнирование излишне начисленного дохода по займу';
      
        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :dt_sys_acc, :kc_id, :non_resident, :client_type, :rep_date)
        into dt_acc_id;
      
        select acc_id
        from get_ref_account_office(:loan_acc_group, :kt_sys_acc, :reg_id, '', -1, :non_resident, :client_type, :rep_date)
        into kt_acc_id;
      
        execute procedure hp_do_kn_operbook_ins(:main_ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :rep_date, :reg_id, 1, :ob_detail, 0);
  
        amount = kn_perc_saldo;
        dt_sys_acc = 'Процент.Погашено';
        kt_sys_acc = 'Процент.Начислено';
        ob_detail = 'Сторнирование закрытия излишне начисленных процентов на излишне полученные';
    
        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :dt_sys_acc, :kc_id, :non_resident, :client_type, :rep_date)
        into dt_acc_id;
    
        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :rep_date)
        into kt_acc_id;
  
        execute procedure hp_do_kn_operbook_ins(:main_ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :rep_date, :reg_id, 1, :ob_detail, 0);
    
    
        amount = kn_perc_saldo;
        kt_acc_id = dt_acc_id;
        dt_acc_id = acc_id;
        ob_detail = 'Сторнирование излишне полученных процентов';
    
        execute procedure hp_do_kn_operbook_ins(:main_ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :rep_date, :reg_id, 1, :ob_detail, 0);
    
        amount = -kn_perc_saldo;
        dt_acc_id = acc_id;
        kt_sys_acc = 'Переплата.Погашено';
        ob_detail = 'Отражение переплаты на сумму излишне начисленных и полученных процентов';
    
        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :rep_date)
        into kt_acc_id;
    
        execute procedure hp_do_kn_operbook_ins(:main_ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :rep_date, :reg_id, 1, :ob_detail, 0);
      end
      if (is_done = 0) then
       exception e_default f_format('Kn_Operkred not found, k_id: %1', k_id);
    end

    if (closed_date < start_date and disb_date < start_date) then
    begin
      acc_dt = acc_calced;
      acc_kt = acc_profit;
      amount = perc_saldo;
      ob_detail = 'Сторнирование излишне начисленного дохода по займу';
  
      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                            ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                            ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
      values (:ol_id, -1, :def_curr, 1, null, :acc_dt, :amount,
              :amount, null, null, 0, :acc_kt, null, :ob_detail,
              :start_date, 1, :reg_id);
    end

    acc_dt = main_dt_acc;
    acc_kt = acc_payed;
    amount = perc_saldo;
    ob_detail = 'Сторнирование излишне полученных процентов';

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:oper_ol_id, -1, :def_curr, 1, null, :acc_dt, :amount,
            :amount, null, null, 0, :acc_kt, null, :ob_detail,
            :rep_date, 1, :reg_id);

    acc_dt = main_dt_acc;
    acc_kt = acc_pereplata;
    amount = -perc_saldo;
    ob_detail = 'Отражение переплаты на сумму излишне начисленных и полученных процентов';

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:oper_ol_id, -1, :def_curr, 1, null, :acc_dt, :amount,
            :amount, null, null, 0, :acc_kt, null, :ob_detail,
            :rep_date, 1, :reg_id);
  end

  if (db_id = 94401) then
  begin
    acc_dt = acc_profit;
    acc_kt = acc_payed;
    amount = -perc_saldo;
    ob_detail = 'Сторнирование излишне начисленного дохода по займу';

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:oper_ol_id, -1, :def_curr, 1, null, :acc_dt, :amount,
            :amount, null, null, 0, :acc_kt, null, :ob_detail,
            :rep_date, 1, :reg_id);


    acc_dt = acc_payed;
    acc_kt = acc_pereplata;
    amount = -perc_saldo;
    ob_detail = 'Отражение переплаты на сумму излишне начисленных и полученных процентов';

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:oper_ol_id, -1, :def_curr, 1, null, :acc_dt, :amount,
            :amount, null, null, 0, :acc_kt, null, :ob_detail,
            :rep_date, 1, :reg_id);
  end

  if (db_id in (39801, 41701)) then
  begin
    acc_dt = acc_calced;
    acc_kt = acc_profit;
    amount = perc_saldo;
    ob_detail = 'Сторнирование излишне начисленного дохода по займу';

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:oper_ol_id, -1, :def_curr, 1, null, :acc_dt, :amount,
            :amount, null, null, 0, :acc_kt, null, :ob_detail,
            :rep_date, 1, :reg_id);

    acc_dt = acc_pereplata;
    acc_kt = acc_payed;
    amount = perc_saldo;
    ob_detail = 'Отражение переплаты на сумму излишне полученных процентов';

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:oper_ol_id, -1, :def_curr, 1, null, :acc_dt, :amount,
            :amount, null, null, 0, :acc_kt, null, :ob_detail,
            :rep_date, 1, :reg_id);

    acc_dt = acc_payed;
    acc_kt = acc_calced;
    amount = perc_saldo;
    ob_detail = 'Сторнирование излишне полученных и начисленных процентов';

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:oper_ol_id, -1, :def_curr, 1, null, :acc_dt, :amount,
            :amount, null, null, 0, :acc_kt, null, :ob_detail,
            :rep_date, 1, :reg_id);
  end

  update temp_calcperc_fix
  set cpf_status = 1
  where cpf_from_k_id = :k_id;
end
