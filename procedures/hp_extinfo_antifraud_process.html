CREATE PROCEDURE HP_EXTINFO_ANTIFRAUD_PROCESS(
    EIL_ID D_ID)
AS
declare variable EIL_OBJECT_ID D_ID;
declare variable EIL_FROM_BLS_ID D_ID;
declare variable EIL_STATUS D_INTEGER;
declare variable JSON_INFO D_BLOB_TEXT;
declare variable TOTAL D_INTEGER;
declare variable MSG_TITLE D_STR_255;
declare variable MSG_BODY D_STR_2048;
declare variable INN D_STR_20;
declare variable C_ID D_ID;
declare variable C_PASS_NO D_STR_64;
declare variable C_PASS_WHO D_STR_255;
declare variable C_PASS_DATE D_DATE;
declare variable C_BIRTH_DATE D_DATE;
declare variable C_NAME_FIO D_STR_255;
declare variable CATEGORIES D_STR_255;
declare variable categories_count d_smallint;
declare variable i d_smallint;
declare variable temp_object d_str_long;
declare variable temp_category d_str_100;
begin
    select
        eil.eil_object_id,
        eil.eil_from_bls_id,
        eil.eil_status
    from extinfolist eil
    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
        and eic.eic_code = 'antifraud_client_check'
    where eil.eil_id = :eil_id
        and eil.eil_status in (2, -1, -2)
    into
        :eil_object_id,
        :eil_from_bls_id,
        :eil_status;

    if (:eil_status = 2) then
    begin
        :json_info = coalesce((select blx_blob from hp_get_blobex_blob(:eil_from_bls_id)), '{}');
        :total = coalesce(f_json_blob_get_value(:json_info, 'total', null), -3);

        -- Если json был как-то изменён сервисом Антирод-Центр, тогда возвращаем -1
        -- чтобы уведомить клиента, об ошибке
        :total =
        case
            when :total < 0 then -3
            when :total = 0 then 1
            when :total > 0 then 2
        end;

        update antifraud_check ac
        set ac.ac_status = :total
        where ac.ac_id = :eil_object_id;

        if (:total = 2) then  -- RFCKZ-448
        begin
          select ac.ac_inn
          from antifraud_check ac
          where ac.ac_id = :eil_object_id
          into :inn;

          i = 0;
          categories = '';
          categories_count = f_json_str_get_count(:json_info, 'items[0].categories');

          if (categories_count > 0) then
          begin
            while (i < categories_count) do
            begin
              temp_object = f_json_blob_get_value(:json_info, f_format('items[0].categories[%1]', :i), '');

              if (coalesce(temp_object, '') <> '' and trim(temp_object) starting '{' and right(trim(temp_object), 1) = '}') then
                temp_category = left(f_json_str_get_value(temp_object, 'category', ''), 100);

              if (coalesce(temp_category, '') = '') then
                temp_category = left(temp_object, 100);

              categories = categories || coalesce(temp_category, '') || ' ';
              i = i + 1;
            end
          end
          else
            categories = 'Нет данных';

          -- отправляем письмо Азамату
          :msg_title = 'Найдено совпадение со списком Антифрод центра.';
          :msg_body =
              'Дата и время события: ' || cast(current_timestamp as d_str_64) ||
              '<br> Категория: ' || :categories ||
              '<br> ИИН: ' || :inn;

          select cl.c_id, cl.c_pass_no, cl.c_pass_who,
                 cl.c_pass_date, cl.c_birth_date, cl.c_name_fio
          from clients cl
          where cl.c_inn = :inn
          into :c_id, c_pass_no, c_pass_who,
                c_pass_date, c_birth_date, c_name_fio;

          if (:c_id is not null) then -- если клиент уже есть в бд, то добавляем инцормацию о клиенте
          begin
            :msg_body = :msg_body ||
                '<br> ФИО: ' || :c_name_fio ||
                '<br> Паспортные данные: ' || 'номер паспорта: ' || coalesce(:c_pass_no, 'Нет данных') ||
                                              ', кем выдан: ' || coalesce(:c_pass_who, 'Нет данных') ||
                                              ', дата выдачи: ' || coalesce(cast(:c_pass_date as str_10), 'Нет данных') ||
                '<br> Дата рождения: ' || coalesce(cast(:c_birth_date as d_str_10), 'Нет данных');

          end

          -- RFCKZ-576
          execute procedure hp_send_message('email', 1010100002016, 'antifraud_found@mbulak.kz', :msg_title, :msg_body, '', 1, 1000000000000);
        end 

    end
    else if (:eil_status < 0) then
    begin
        -- отправляем письмо Азамату
        :msg_title = 'Сбой сервиса Антифрод РК';
        :msg_body =
            'Сообщаем, что в работе сервиса Антифрод РК зафиксирован технический сбой.<br>'||
            'Для корректного продолжения проверки необходимо:<br>'||
            '  1.Загрузить актуальный список АЦ в Базу.<br>'||
            '  2.Переключить проверку на режим работы по Базе.';

        -- RFCKZ-576
        execute procedure hp_send_message('email', 1010100002016, 'antifraud_found@mbulak.kz', :msg_title, :msg_body, '', 1, 1000000000000);

        update antifraud_check ac
        set ac.ac_status = :eil_status
        where ac.ac_id = :eil_object_id;

        if (:eil_status = -2) then
        begin
          insert into antifraud_error_log(ael_from_eil_id)
          values (:eil_id);
        end
    end
end
