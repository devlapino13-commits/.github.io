CREATE PROCEDURE HP_PERSVALUE_RESTORE(
    PIF_UUID D_UUID NOT NULL,
    PIF_INDEXED D_BOOL NOT NULL,
    TABLE_KEY_ID D_ID NOT NULL,
    FIELD_NEW_VALUE D_STR_255,
    VALUE_SIGNED D_BOOL,
    MODIFY_DATETIME D_DATETIME,
    COMMON_USR_ID D_ID)
AS
declare variable pda_id d_id;
declare variable pva_id d_id;
declare variable curr_value d_str_255;
declare variable pda_value_id d_id;
declare variable pva_is_null d_bool;
declare variable pva_is_delete d_bool;
declare variable pin_id d_id;
begin
  if (f_get_db_id() <> 81001) then
    exit;

  select pin_id
  from persinfo
  where pin_object_kind = 0 and
        pin_object_id = :table_key_id
  into pin_id;

  if (row_count = 0) then
  begin
    insert into persinfo (pin_object_kind, pin_object_id)
    values (0, :table_key_id)
    returning pin_id
    into pin_id;
  end

  select pda_id, pda_value_id
  from persdata
  where pda_from_pif_uuid = :pif_uuid and
        pda_from_pin_id = :pin_id
  into :pda_id, pda_value_id;

  if (pda_id is null) then
  begin
    insert into persdata (pda_from_pif_uuid, pda_from_pin_id)
    values (:pif_uuid, :pin_id)
    returning pda_id
    into :pda_id;
  end

  if (modify_datetime is null) then
    modify_datetime = cast('NOW' as timestamp);

  if (pda_value_id is not null) then
  begin
    pva_is_delete = 0;

    select pva_is_null, pva_is_delete
    from persvalue v
    where pva_id = :pda_value_id
    into pva_is_null, pva_is_delete;

    if (pva_is_delete = 0 and pva_is_null = 1 and field_new_value is null) then
      exit;

    if (pif_indexed = 0) then
    begin
      select pvs_value
      from persvalstr
      where pvs_pva_id = :pda_value_id
      into curr_value;
    end else
    begin
      select pvx_value
      from persvalidx
      where pvx_pva_id = :pda_value_id
      into curr_value;
    end

    if (pva_is_delete = 0 and curr_value is not distinct from field_new_value) then
      exit;
  end

  insert into persvalue (pva_from_pda_id, pva_is_null, pva_datetime, pva_usr_id)
  values (:pda_id, iif(:field_new_value is null, 1, 0), :modify_datetime, :common_usr_id)
  returning pva_id
  into :pva_id;

  if (field_new_value is not null) then
  begin
    if (pif_indexed = 0) then
    begin
      insert into persvalstr (pvs_pva_id, pvs_value)
      values (:pva_id, :field_new_value);
    end else
    begin
      insert into persvalidx (pvx_pva_id, pvx_value)
      values (:pva_id, :field_new_value);
    end

    if (value_signed = 1) then
    begin
      update or insert into persedoc (ped_pva_id)
      values (:pva_id)
      matching (ped_pva_id);
    end
  end
end
