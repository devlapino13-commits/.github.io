CREATE PROCEDURE PRINT_CALC_ZADOL(
    K_ID ID_TYPE)
RETURNS (
    K_AMOUNT_APPROVED CURR_TYPE,
    K_KOLICH_REPAY D_INTEGER,
    K_PERCENT PERC_TYPE,
    K_FACT_DISB_DATE D_DATE,
    SH_DATE D_DATE,
    K_STOP_PERC D_DATE,
    K_KOLICH_DAY_REAL INTEGER,
    DELINQ_DATE D_DATE,
    SH_PERC CURR_TYPE,
    DISB_AMOUNT CURR_TYPE,
    STR_MONTH_PL D_STR_255,
    ZAD_PERC CURR_TYPE,
    ZAD_OS CURR_TYPE,
    SH_OS CURR_TYPE,
    STR_CALC_PROC D_STR_255,
    STR_OS D_STR_100,
    MONTH_PL DECIMAL(18,4))
AS
declare variable ol_memo d_str_long;
declare variable k_amount curr_type;
declare variable k_stop_dd d_date;
declare variable sh_date_fakt d_date;
declare variable k_kredline_end d_date;
declare variable k_disb_date d_date;
declare variable k_percent_per_day perc_type;
declare variable pl_gr2 d_integer;
declare variable pl_gr1 d_integer;
declare variable sh_date_last d_date;
declare variable k_perc decimal(18,4);
declare variable dt d_date;
declare variable sh_date_ok d_date;
declare variable disb_amount_last curr_type;
begin
  k_amount_approved = 0;
  k_percent = 0;
  k_fact_disb_date = null;
  k_kolich_repay = 0;
  k_stop_perc = null;
  k_disb_date = null;
  k_kredline_end = null;
  sh_date = null;
  sh_os = 0;
  sh_perc = 0;
  str_month_pl = '';
  month_pl = 0;
  str_calc_proc = '';
  sh_date_last = null;
  delinq_date = null;
  sh_date_ok = null;
  ol_memo = '';
  str_os = '';
  for select k_amount_approved, k_percent_per_day * 365, k_percent_per_day, k_fact_disb_date, k_kolich_repay,
             k_stop_perc, k_disb_date, k_kredline_end
      from kontract kont, product
      where k_id = :k_id and --101010038660000301
            k_from_pr_id = pr_id
      into :k_amount_approved, :k_percent, :k_percent_per_day, :k_fact_disb_date, :k_kolich_repay, :k_stop_perc,
           :k_disb_date, k_kredline_end

  do
  begin
    k_kolich_day_real = 0;
    if ((k_disb_date is not null) and
        (k_kredline_end is not null)) then
      k_kolich_day_real = datediff(day, k_disb_date, k_kredline_end);
    month_pl = :k_amount_approved * (:k_percent / 100 / 12 + (:k_percent / 100 / 12 / (power((1 + :k_percent / 100 / 12), (:k_kolich_repay)) - 1)));
    month_pl=round(:month_pl,0) ;
    str_month_pl = cast(round(:k_amount_approved, 0) as varchar(20)) || '*(' || cast(:k_percent / 100 as varchar(20)) || '/12+(' || cast(:k_percent / 100 as varchar(20)) || '/12/((1+' || cast(:k_percent / 100 as varchar(20)) || '/12)^(' || cast(:k_kolich_repay as varchar(20)) || ')-1)))=' || cast(round(month_pl) as varchar(20));

    select max(oks_delinq_date) delinq_date
    from oksummary
    where :k_stop_perc between oks_date_begin and oks_date_end and
          oks_from_k_id = :k_id and
          oks_close_state = 0
    into delinq_date;
    suspend;
    k_perc = round(:k_percent, 2);
    k_percent = null;
    --month_pl = 0;
    str_month_pl = '';
    k_kolich_day_real = 0;
    k_kolich_repay = 0;
    k_fact_disb_date = null;
    k_stop_dd = k_stop_perc;
    k_stop_perc = null;
    disb_amount = k_amount_approved;
    disb_amount_last = k_amount_approved;
    k_amount = k_amount_approved;
    k_amount_approved = 0;
    delinq_date = null;

  end
  --контрольные даты
  for select distinct sh_date
      from (select sh_date --график
            from schedule
            where sh_from_k_id = :k_id
            union
            select sh_date -- платежи
            from aarep_make_zad_list(:k_id, :k_stop_dd, 1, 0)
            union
            select ok_ol_date --перенос
            from operkred, operlist
            where ok_ol_is_active = 1 and
                  ok_from_k_id = :k_id and
                  ok_type = 2 and
                  ok_from_ol_id = ol_id)
      order by 1

      into :dt
  do
  begin
    pl_gr1 = 0;
    pl_gr2 = 0;
    select sh_date, :disb_amount_last - sh_os, sh_os, sh_perc
    from schedule
    where sh_from_k_id = :k_id and
          sh_date <= :dt and
          (sh_date > :sh_date_last or
          :sh_date_last is null)
    into :sh_date, :disb_amount, :sh_os, :sh_perc;
    begin
      if (sh_perc > 0) then
        str_os = cast(round(month_pl, 0) as varchar(20)) || '-' || cast(round(:sh_perc, 0) as varchar(20));
      --sh_date_last = sh_date;
      if ((:sh_os = 0) and
          (:disb_amount <> :k_amount)) then
        str_calc_proc = 'Льготный период';
      else
      if (:disb_amount <> :k_amount) then
        str_calc_proc = '(' || cast(round(:disb_amount_last, 0) as varchar(20)) || '*' || cast(:k_perc / 100 as varchar(20)) || '\365)*' || cast(datediff(day, :sh_date_last, :sh_date) as varchar(20));
      -- suspend;

    end

    --перенос платежа
    for select ok_ol_date, ol_memo, max(ok_sh_os), max(ok_sh_perc)
        from operkred, operlist
        where ok_ol_is_active = 1 and
              ok_from_k_id = :k_id and
              ok_type = 2 and
              ok_from_ol_id = ol_id and
              ok_ol_date <= :dt and
              (ok_ol_date > :sh_date_last or
              :sh_date_last is null)
        group by 1, 2
        into :sh_date_ok, :ol_memo, :sh_os, :sh_perc
    do
    begin
      sh_date = sh_date_ok;
      str_calc_proc = :ol_memo;
      :sh_date_last = sh_date;
      suspend;
    end
    if (row_count = 0) then
      pl_gr1 = 1;
    ----платежи не по графику
    zad_perc = 0;
    zad_os = 0;
    for select zad_os, zad_perc, sh_date
        from aarep_make_zad_list(:k_id, :k_stop_dd, 1, 0)
        where sh_date <= :dt and
              (sh_date > :sh_date_last or
              :sh_date_last is null)
        order by sh_date
        into :zad_os, :zad_perc, :sh_date_fakt
    do
    begin
      sh_date = sh_date_fakt;
      --sh_date_last = sh_date_fakt;
      disb_amount = disb_amount_last - :sh_os;
      if ((:sh_os = 0) and
          (:disb_amount <> :k_amount)) then
        str_calc_proc = 'Льготный период';
      else
        str_calc_proc = '(' || cast(round(:disb_amount_last, 0) as varchar(20)) || '*' || cast(:k_perc / 100 as varchar(20)) || '\365)*' || cast(datediff(day, :sh_date_last, :sh_date) as varchar(20));
      suspend;

    end
    if (row_count = 0) then
      pl_gr2 = 1;
    if ((pl_gr1 = 1) and
        (pl_gr2 = 1)) then
      suspend;
    :sh_date_last = sh_date;
    :disb_amount_last = :disb_amount;
  end

end
