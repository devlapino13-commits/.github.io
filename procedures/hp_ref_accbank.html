CREATE PROCEDURE HP_REF_ACCBANK
RETURNS (
    ACB_ID D_ID,
    ACB_BANK_NAME D_STR_255,
    ACB_REG_NO D_STR_20,
    ACB_BIK D_STR_20,
    ACB_BANK_ACCOUNT D_STR_20,
    ACB_ACC_BALANCE D_STR_20,
    ACB_ACC_KIND D_STR_20,
    ACB_CHECKED D_SMALLINT,
    RES_ACC_CODE D_STR_CODE,
    RES_ACC_NAME D_STR_255,
    RES_ACC_MEMO D_STR_255)
AS
begin
  for select acb_id,
             acb_bank_name,
             acb_reg_no,
             acb_bik,
             acb_bank_account,
             acb_acc_balance,
             acb_acc_kind,
             acb_checked,
             acc_name,
             acc_memo
      from ref_accbank, ref_account
      where acb_id = acc_id
      into :acb_id,
           :acb_bank_name,
           :acb_reg_no,
           :acb_bik,
           :acb_bank_account,
           :acb_acc_balance,
           :acb_acc_kind,
           :acb_checked,
           :res_acc_name,
           :res_acc_memo
  do
  begin
    res_acc_code = f_fmt_acc(acb_id);
    suspend;
  end
end
