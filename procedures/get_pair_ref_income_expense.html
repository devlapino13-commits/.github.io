CREATE PROCEDURE GET_PAIR_REF_INCOME_EXPENSE
RETURNS (
    INX_ID D_ID,
    INX_CODE D_STR_CODE,
    INX_NAME D_STR_2048)
AS
begin
  for select e.inx_id, e.inx_code, e.inx_code || ' ' || e.inx_name || '  / ' || parent.inx_name
      from ref_income_expense e, ref_income_expense parent
      where e.inx_parent_id = parent.inx_id and
            e.inx_level = 5
      into :inx_id, :inx_code, :inx_name
  do
    suspend;
end
