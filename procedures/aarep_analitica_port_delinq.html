CREATE PROCEDURE AAREP_ANALITICA_PORT_DELINQ(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_WITH_LO INTEGER)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_KC_FIO VARCHAR(100),
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_DELIQ_BALANCE DECIMAL(18,4),
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    AA_DELIQ_CL_KOLICH INTEGER,
    AA_DELIQ_BEG_BALANCE DECIMAL(18,4),
    AA_DELIQ_BEG_CL_KOLICH INTEGER,
    AA_DELIQ_NEW_BALANCE DECIMAL(18,4),
    AA_DELIQ_NEW_CL_KOLICH INTEGER,
    AA_DELIQ_OLD_BALANCE DECIMAL(18,4),
    AA_DELIQ_OLD_CL_KOLICH INTEGER,
    AA_CATEGORY VARCHAR(255),
    AA_PAR_END DECIMAL(18,7),
    AA_DELIQ_BAL_BEG_CHANGE DECIMAL(18,4),
    AA_DELIQ_BAL_ALL_CHANGE DECIMAL(18,4),
    AA_DELIQ_RISE_PERC DECIMAL(18,7),
    AA_DELIQ_BEGIN_CHANGE_PERC DECIMAL(18,7),
    AA_DELIQ_NEW_PERC DECIMAL(18,7))
AS
declare variable del_no_change integer;
declare variable del_down integer;
declare variable del_no_old_change integer;
declare variable del_no_new integer;
declare variable del_no_old integer;
begin
  if (rep_with_lo = 0) then
  begin
    aa_curr_kc_fio = 'Все сотрудники';
    for select aa_curr_office_name, aa_curr_branch_name, sum(aa_amount_active_end), sum(aa_kolich_cl_active_end),
               sum(aa_deliq_balance1), sum(aa_deliq_cl_kolich1), sum(aa_deliq_beg_balance), sum(aa_deliq_beg_cl_kolich),
               sum(aa_deliq_balance1 * aa_deliq_is_new), sum(aa_deliq_cl_kolich1 * aa_deliq_is_new),
               sum(aa_deliq_balance1 * (1 - aa_deliq_is_new)), sum(aa_deliq_cl_kolich1 * (1 - aa_deliq_is_new))
        from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 0, 0, 1, 0, 0, 0)
        group by aa_curr_office_name, aa_curr_branch_name
        into :aa_curr_office_name, :aa_curr_branch_name, :aa_amount_active_end, :aa_kolich_cl_active_end,
             :aa_deliq_balance, :aa_deliq_cl_kolich, :aa_deliq_beg_balance, :aa_deliq_beg_cl_kolich,
             :aa_deliq_new_balance, :aa_deliq_new_cl_kolich, :aa_deliq_old_balance, :aa_deliq_old_cl_kolich
    do
    begin
      del_no_change = 0;
      del_down = 0;
      del_no_old_change = 0;
      del_no_new = 0;
      del_no_old = 0;
      if ((aa_deliq_beg_balance = aa_deliq_balance) and
          (aa_deliq_balance > 0)) then
        del_no_change = 1;
      if (aa_deliq_beg_balance > aa_deliq_old_balance) then
        del_down = 1;
      if ((aa_deliq_beg_balance = aa_deliq_old_balance) and
          (aa_deliq_old_balance > 0)) then
        del_no_old_change = 1;
      if (aa_deliq_new_balance = 0) then
        del_no_new = 1;
      if (aa_deliq_beg_balance = 0) then
        del_no_old = 1;

      if ((aa_deliq_beg_balance <> 0) or (aa_deliq_balance <> 0)) then
      begin
        aa_category = '';
        if ((del_down = 1) and
            (del_no_new = 1)) then
        begin
          if (aa_category <> '') then
            aa_category = aa_category || ' ' || 'Категория 1';
          else
            aa_category = 'Категория 1';
        end
        if ((del_down = 1) and
            (del_no_new = 0)) then
        begin
          if (aa_category <> '') then
            aa_category = aa_category || ' ' || 'Категория 2';
          else
            aa_category = 'Категория 2';
        end
        if ((del_no_old = 1) and
            (del_no_new = 0)) then
        begin
          if (aa_category <> '') then
            aa_category = aa_category || ' ' || 'Категория 3';
          else
            aa_category = 'Категория 3';
        end
        if (del_no_change = 1) then
        begin
          if (aa_category <> '') then
            aa_category = aa_category || ' ' || 'Категория 4';
          else
            aa_category = 'Категория 4';
        end
        if ((del_no_old_change = 1) and
            (del_no_new = 0)) then
        begin
          if (aa_category <> '') then
            aa_category = aa_category || ' ' || 'Категория 5';
          else
            aa_category = 'Категория 5';
        end
        if (aa_category = '') then
          aa_category = 'Категория не определена';
      end
      else
        aa_category = 'Нет просрочек';

      if (aa_amount_active_end > 0) then
        aa_par_end = cast((aa_deliq_balance / aa_amount_active_end * 100) as decimal(15,2));
      else
        aa_par_end = 0;
      aa_deliq_bal_beg_change = aa_deliq_old_balance - aa_deliq_beg_balance;
      aa_deliq_bal_all_change = aa_deliq_balance - aa_deliq_beg_balance;
      if (aa_deliq_beg_balance > 0) then
        aa_deliq_rise_perc = cast((aa_deliq_bal_all_change / aa_deliq_beg_balance * 100) as decimal(15,2));
      else
        aa_deliq_rise_perc = 0;

      if (aa_deliq_beg_balance > 0) then
        aa_deliq_begin_change_perc = cast((aa_deliq_bal_beg_change / aa_deliq_beg_balance * 100) as decimal(15,2));
      else
        aa_deliq_begin_change_perc = 0;

      if (aa_deliq_balance > 0) then
        aa_deliq_new_perc = cast((aa_deliq_new_balance / aa_deliq_balance * 100) as decimal(15,2));
      else
        aa_deliq_new_perc = 0;
      suspend;
    end
  end
  else
  begin
    for select aa_curr_office_name, aa_curr_branch_name, aa_curr_kc_fio, sum(aa_amount_active_end),
               sum(aa_kolich_cl_active_end), sum(aa_deliq_balance1), sum(aa_deliq_cl_kolich1),
               sum(aa_deliq_beg_balance), sum(aa_deliq_beg_cl_kolich), sum(aa_deliq_balance1 * aa_deliq_is_new),
               sum(aa_deliq_cl_kolich1 * aa_deliq_is_new), sum(aa_deliq_balance1 * (1 - aa_deliq_is_new)),
               sum(aa_deliq_cl_kolich1 * (1 - aa_deliq_is_new))
        from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 0, 0, 1, 0, 0, 0)
        group by aa_curr_office_name, aa_curr_branch_name, aa_curr_kc_fio
        into :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_kc_fio, :aa_amount_active_end,
             :aa_kolich_cl_active_end, :aa_deliq_balance, :aa_deliq_cl_kolich, :aa_deliq_beg_balance,
             :aa_deliq_beg_cl_kolich, :aa_deliq_new_balance, :aa_deliq_new_cl_kolich, :aa_deliq_old_balance,
             :aa_deliq_old_cl_kolich
    do
    begin
      del_no_change = 0;
      del_down = 0;
      del_no_old_change = 0;
      del_no_new = 0;
      del_no_old = 0;
      if ((aa_deliq_beg_balance = aa_deliq_balance) and
          (aa_deliq_balance > 0)) then
        del_no_change = 1;
      if (aa_deliq_beg_balance > aa_deliq_old_balance) then
        del_down = 1;
      if ((aa_deliq_beg_balance = aa_deliq_old_balance) and
          (aa_deliq_old_balance > 0)) then
        del_no_old_change = 1;
      if (aa_deliq_new_balance = 0) then
        del_no_new = 1;
      if (aa_deliq_beg_balance = 0) then
        del_no_old = 1;

      if ((aa_deliq_beg_balance <> 0) or (aa_deliq_balance <> 0)) then
      begin
        aa_category = '';
        if ((del_down = 1) and
            (del_no_new = 1)) then
        begin
          if (aa_category <> '') then
            aa_category = aa_category || ' ' || 'Категория 1';
          else
            aa_category = 'Категория 1';
        end
        if ((del_down = 1) and
            (del_no_new = 0)) then
        begin
          if (aa_category <> '') then
            aa_category = aa_category || ' ' || 'Категория 2';
          else
            aa_category = 'Категория 2';
        end
        if ((del_no_old = 1) and
            (del_no_new = 0)) then
        begin
          if (aa_category <> '') then
            aa_category = aa_category || ' ' || 'Категория 3';
          else
            aa_category = 'Категория 3';
        end
        if (del_no_change = 1) then
        begin
          if (aa_category <> '') then
            aa_category = aa_category || ' ' || 'Категория 4';
          else
            aa_category = 'Категория 4';
        end
        if ((del_no_old_change = 1) and
            (del_no_new = 0)) then
        begin
          if (aa_category <> '') then
            aa_category = aa_category || ' ' || 'Категория 5';
          else
            aa_category = 'Категория 5';
        end
        if (aa_category = '') then
          aa_category = 'Категория не определена';
      end
      else
        aa_category = 'Нет просрочек';

      if (aa_amount_active_end > 0) then
        aa_par_end = cast((aa_deliq_balance / aa_amount_active_end * 100) as decimal(15,2));
      else
        aa_par_end = 0;
      aa_deliq_bal_beg_change = aa_deliq_old_balance - aa_deliq_beg_balance;
      aa_deliq_bal_all_change = aa_deliq_balance - aa_deliq_beg_balance;
      if (aa_deliq_beg_balance > 0) then
        aa_deliq_rise_perc = cast((aa_deliq_bal_all_change / aa_deliq_beg_balance * 100) as decimal(15,2));
      else
        aa_deliq_rise_perc = 0;

      if (aa_deliq_beg_balance > 0) then
        aa_deliq_begin_change_perc = cast((aa_deliq_bal_beg_change / aa_deliq_beg_balance * 100) as decimal(15,2));
      else
        aa_deliq_begin_change_perc = 0;

      if (aa_deliq_balance > 0) then
        aa_deliq_new_perc = cast((aa_deliq_new_balance / aa_deliq_balance * 100) as decimal(15,2));
      else
        aa_deliq_new_perc = 0;

      suspend;
    end
  end

end
