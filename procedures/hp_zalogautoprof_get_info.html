CREATE PROCEDURE HP_ZALOGAUTOPROF_GET_INFO(
    ACL_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT,
    RES_IMAGE_LIST D_STR_8100,
    K_ID D_ID,
    C_ID D_ID,
    ZA_ORIG_STS_ID D_ID,
    ZA_ORIG_PTS_ID D_ID,
    ZA_FROM_C_ID D_ID,
    ZAP_ID D_INTEGER,
    ZA_ORIG_STS_DRL_ID D_ID,
    ZA_ORIG_PTS_DRL_ID D_ID,
    IS_READONLY D_SMALLINT)
AS
declare variable tmp_str d_str_8100;
declare variable z_name_fio d_str_100;
declare variable z_birth_date d_date;
declare variable drl_id d_id;
declare variable zap_date d_date;
declare variable zap_no d_str_code;
declare variable doc_image d_str_8100;
declare variable cds_id d_id;
declare variable pk_id d_integer;
declare variable doc_item d_str_long;
declare variable rd_id d_integer;
declare variable za_from_z_id d_integer;
declare variable zap_avg_cost d_currency;
declare variable zap_zalog_cost d_currency;
begin
  res_html = '';
  res_image_list = '';
  doc_image = '';
  doc_item = '';
  is_readonly = 0;

  select rd_id
  from refdata
  where rd_code = 'ActionGroup.ZalogAutoProf'
  into rd_id;

  select acl_object_id
  from actionlist
  where acl_id = :acl_id and
        acl_from_rd_id = :rd_id
  into zap_id;

  if (zap_id is null) then
    zap_id = acl_id;

  for select zap_date, zap_no, za_from_k_id, za_from_c_id, za_orig_sts_id, za_orig_pts_id, za_from_z_id,
             zap_avg_cost, zap_zalog_cost
  from zalogautoprof, zalogauto
  where zap_id = :zap_id and
        zap_from_za_id = za_id
  into zap_date, zap_no, k_id, za_from_c_id, za_orig_sts_id, za_orig_pts_id, za_from_z_id,
       zap_avg_cost, zap_zalog_cost
  do
  begin
    if (zap_no is null) then
      zap_no = '';

    for select k_kontract_no, k_disb_date, pr_name, k_amount_approved, k_period,
           c_id, c_name_fio, c_birth_date, c_doc_type
    from knclient, kontract, clients, product
    where k_id = :k_id and
          kc_from_k_id = k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = c_id and
          k_from_pr_id = pr_id
    rows 1
    as cursor c
    do
    begin
      c_id = c.c_id;

      cds_id = null;
      select cds_id
      from clientdocs
      where cds_from_c_id = :c_id and
            cds_from_pk_id = c.c_doc_type and
            cds_is_delete = 0
      order by cds_create_datetime desc, cds_id desc
      rows 1
      into cds_id;

      if (cds_id is not null) then
      begin
        doc_image = f_docs_item(cds_id, un_id);
        if (res_image_list <> '') then
          res_image_list = res_image_list || ',';
        res_image_list = res_image_list || doc_image;
      end

      za_orig_sts_drl_id = 0;
      if (za_orig_sts_id is not null) then
      begin
        doc_image = f_docs_item(za_orig_sts_id, un_id);
        if (res_image_list <> '') then
          res_image_list = res_image_list || ',';
        res_image_list = res_image_list || doc_image;

        select coalesce(cds_from_drl_id, 0)
        from clientdocs
        where cds_id = :za_orig_sts_id
        into za_orig_sts_drl_id;
      end

      za_orig_pts_drl_id = 0;
      if (za_orig_pts_id is not null) then
      begin
        doc_image = f_docs_item(za_orig_pts_id, un_id);
        if (res_image_list <> '') then
          res_image_list = res_image_list || ',';
        res_image_list = res_image_list || doc_image;

        select coalesce(cds_from_drl_id, 0)
        from clientdocs
        where cds_id = :za_orig_pts_id
        into za_orig_pts_drl_id;
      end

      doc_item = f_json_make_pair('caption', 'Клиент', 'str');
      res_image_list = f_json_str_set_value('{' || doc_item || '}', 'items', '['||res_image_list||']');

      tmp_str = '';
      tmp_str = tmp_str || f_format('Профсуждение <b>%1</b> от <b>%2</b> г.', zap_no, f_datetostr(zap_date)) || '<br>';
      tmp_str = tmp_str || f_format('Договор <b>%1</b> от <b>%2</b> г.', c.k_kontract_no, f_datetostr(c.k_disb_date)) || '<br>';
      tmp_str = tmp_str || f_format('Продукт: <b>%1</b>', c.pr_name) || '<br>';
      tmp_str = tmp_str || f_format('Сумма займа: <b>%1</b>', round(c.k_amount_approved, 0)) || '<br>';
      tmp_str = tmp_str || f_format('Срок займа: <b>%1</b> мес.', c.k_period) || '<br><br>';
      tmp_str = tmp_str || f_format('Среднерыночная стоимость <b>%1</b>', f_numtostr(zap_avg_cost, 2)) || '<br>';
      tmp_str = tmp_str || f_format('Залоговая стоимость <b>%1</b>', f_numtostr(zap_zalog_cost, 2)) || '<br><br>';

      tmp_str = tmp_str || f_format('Клиент <b>%1</b>, д.р. <b>%2</b> г.', c.c_name_fio, f_datetostr(c.c_birth_date)) || '<br>';

      if (c.c_id = za_from_c_id)
      then tmp_str = tmp_str || 'Залогодатель - сам клиент' || '<br>';
      else
      begin
        select c_name_fio, c_birth_date, c_doc_type
        from clients
        where c_id = :za_from_c_id
        into z_name_fio, z_birth_date, pk_id;
        tmp_str = tmp_str || f_format('Залогодатель <b>%1</b>, д.р. <b>%2</b> г.', z_name_fio, f_datetostr(z_birth_date)) || '<br>';

        cds_id = null;
        select cds_id
        from clientdocs
        where cds_from_c_id = :za_from_c_id and
              cds_from_pk_id = :pk_id and
              cds_is_delete = 0
        order by cds_create_datetime desc, cds_id desc
        rows 1
        into cds_id;
  
        if (cds_id is not null) then
        begin
          doc_image = f_docs_item(cds_id, un_id);
          doc_image = '{"caption":"Залогодатель", "items":[' || doc_image || ']}';

          if (res_image_list <> '') then
            res_image_list = res_image_list || ',';
          res_image_list = res_image_list || doc_image;
        end
      end

      res_html = tmp_str || '<br>';
      tmp_str = '';

      if (za_orig_sts_id is not null) then
      begin
        drl_id = null;
        select cds_from_drl_id
        from clientdocs
        where cds_id = :za_orig_sts_id
        into drl_id;

        if (drl_id is not null) then
        begin
          tmp_str = '';

          with tab as (select drf_name, coalesce(drv_final_value, '') as drv_final_value
          from docrecogvalue, docrecogfield
          where drv_from_drl_id = :drl_id and
                drv_from_drf_id = drf_id and
                drf_inactive = 0
          order by drf_position, drf_id)
          select list('<br>' || drf_name || ': <b>' || drv_final_value || '</b>')
          from tab
          into :tmp_str;

          if (tmp_str is null) then
            tmp_str = '';
        end

        if (tmp_str = '')
        then tmp_str = 'СТС отсутствует';
        else tmp_str = 'СТС: ' || tmp_str;

        res_html = BLOB_APPEND(res_html, tmp_str, '<br><br>');
      end

      if (za_orig_pts_id is not null) then
      begin
        drl_id = null;
        select cds_from_drl_id
        from clientdocs
        where cds_id = :za_orig_pts_id
        into drl_id;

        if (drl_id is not null) then
        begin
          tmp_str = '';
          with tab as (select drf_name, coalesce(drv_final_value, '') as drv_final_value
          from docrecogvalue, docrecogfield
          where drv_from_drl_id = :drl_id and
                drv_from_drf_id = drf_id and
                drf_inactive = 0
          order by drf_position, drf_id)
          select list('<br>' || drf_name || ': <b>' || drv_final_value || '</b>')
          from tab
          into :tmp_str;

          if (tmp_str is null) then
            tmp_str = '';
        end

        if (tmp_str = '')
        then tmp_str = 'ПТС отсутствует';
        else tmp_str = 'ПТС:' || tmp_str;

        res_html = BLOB_APPEND(res_html, tmp_str, '<br><br>');
      end
    end

    if (za_from_z_id is null) then
    begin
      -- инфо для формирования записи в таблице Zalog
      tmp_str = 'Данные для оформления:<br>';
      for select z_name, z_document, z_registry_no, z_registry_date, z_registry_agency, z_memo1, z_memo2
      from hp_zalogauto_fields(:zap_id)
      as cursor z
      do
      begin
        tmp_str = tmp_str || f_format('Название <b>%1</b>', z.z_name) || '<br>';
        tmp_str = tmp_str || f_format('Документы <b>%1</b>', z.z_document) || '<br>';
        tmp_str = tmp_str || f_format('Рег.номер <b>%1</b>', z.z_registry_no) || '<br>';
        tmp_str = tmp_str || f_format('Дата рег. <b>%1</b>', f_datetostr(z.z_registry_date)) || '<br>';
        tmp_str = tmp_str || f_format('Департ.рег. <b>%1</b>', z.z_registry_agency) || '<br>';
        tmp_str = tmp_str || f_format('Краткое примеч. <b>%1</b>', z.z_memo1) || '<br>';
        tmp_str = tmp_str || f_format('Полное примеч. <b>%1</b>', z.z_memo2) || '<br>';
      end
      res_html = BLOB_APPEND(res_html, tmp_str, '<br><br>');
    end
  end

  res_image_list = '[' || res_image_list || ']';
  res_html = BLOB_APPEND((select res_style from fm_html_styles), '<body>', res_html, '</body>');


  suspend;
end
