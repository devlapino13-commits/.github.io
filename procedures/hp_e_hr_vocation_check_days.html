CREATE PROCEDURE HP_E_HR_VOCATION_CHECK_DAYS(
    C_ID D_ID,
    ORDER_TYPE D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    KOL_DAYS D_INTEGER)
RETURNS (
    RES_SALDO D_CURRENCY)
AS
declare variable rep_region_id d_integer;
declare variable rep_year d_integer;
declare variable db_id d_integer;
declare variable date_buf_str d_str_10;
declare variable buf_year d_str_10;
declare variable buf_month d_integer;
declare variable year_end d_date;
declare variable buf_date_end d_date;
declare variable payed_tmp d_integer;
declare variable sat_voc_payed d_currency;
declare variable rep_month d_integer;
declare variable i d_integer;
declare variable calc_tmp d_currency;
declare variable sat_voc_calc d_currency;
declare variable hired_date d_date;
begin
  res_saldo = 0;

  db_id = f_get_db_id();

  if (db_id in (39801, 41701)) then
  begin
    if (exists(select *
               from sa_worktype
               where sawt_id = :order_type and
                     sawt_base_type = 4 and /* Отпускные */
                     sawt_is_paid = 1)) /* Оплачиваемые */
    then
    begin
      rep_year = extract(year from rep_date_begin);
      select ps_from_reg_id, ps_date_begin
      from hr_pstatus
      where ps_from_c_id = :c_id and
            ps_date_begin <= :rep_date_begin and
            ps_date_end >= :rep_date_begin and
            ps_status = 1
      rows 1
      into :rep_region_id, :hired_date;

      select max(st.sat_year_month)
      from sa_tabel st
      where st.sat_from_c_id = :c_id and
            st.sat_year_month >= 201301
      into :date_buf_str;

      if (date_buf_str is null) then
        date_buf_str = extract(year from hired_date) * 100 + extract(month from hired_date);

      buf_year = substring(date_buf_str from 1 for 4);
      buf_month = cast(substring(date_buf_str from 5 for 2) as integer);
      year_end = cast('31.12.' || :buf_year as date);

      sat_voc_payed = 0;
      if (buf_month < 12) then
      begin
        buf_month = buf_month + 1;
        buf_date_end = '01.' || buf_month || '.' || buf_year;

        for select coalesce(hrv_days, 0)
            from hr_voctravel, sa_worktype swt, hr_orders
            where ((hrv_date_begin > :buf_date_end and
                  hrv_date_begin <= :year_end) or
                  (hrv_date_end > :buf_date_end and
                  hrv_date_end <= :year_end)) and
                  hrv_from_c_id = :c_id and
                  ho_order_type = sawt_id and
                  hrv_from_ho_id = ho_id and
                  ho_status <> -1 and -- Не аннулированные (в процессе и утвержденные)
                  hrv_status >= 0 and
                  swt.sawt_base_type = 4 and /* Трудовой отпуск */
                  sawt_is_paid = 1 /* опалчиваемы отпуск */
            into payed_tmp
        do
          sat_voc_payed = sat_voc_payed + payed_tmp;
      end

-- закоментировал строку ниже и добавил две строки после нее igusev 11.11.14 --
--rep_month = 11;
      if (buf_month = 12) then
        rep_month = 12;
      else
        rep_month = buf_month - 1;

      if (rep_month = 0) then
        rep_month = 1;
-------------------------------------------------------------------------------

      res_saldo = 0;
      select sat_vocation_end
      from aarep_sa_tabel_one_voc(:rep_region_id, :c_id, :rep_year, :rep_month)
      into res_saldo;

      sat_voc_calc = 0;
      if (rep_month < 12) then
      begin
        i = rep_month + 1;
        while (i <= 12) do
        begin
          for select sat_vocation_curr
              from aarep_sa_tabel_one_voc(:rep_region_id, :c_id, :rep_year, :i)
              into :calc_tmp
          do
          begin
            sat_voc_calc = sat_voc_calc + calc_tmp;
          end
          i = i + 1;
        end
      end
      res_saldo = res_saldo + sat_voc_calc - sat_voc_payed - kol_days;
      if ((res_saldo < 0) and
          (res_saldo > -0.50)) then
        res_saldo = 0;
    end
  end
  else
    res_saldo = 36;
  suspend;
end
