CREATE PROCEDURE HP_FUELGENREG_PRINT(
    OBJ_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    FG_ID D_INTEGER,
    FG_DTWRITEOFF D_DATE,
    FG_DTBEG D_DATE,
    FG_DTEND D_DATE,
    FG_OFFICE_ID D_INTEGER,
    FG_PET_ID D_INTEGER,
    FG_START D_CURRENCY,
    FG_START_SUM D_CURRENCY,
    FG_FILLUP D_CURRENCY,
    FG_FILLUP_SUM D_CURRENCY,
    FG_WHOURS D_CURRENCY,
    FG_WRATE D_CURRENCY,
    FG_USEDUP D_CURRENCY,
    FG_USEDUP_SUM D_CURRENCY,
    FG_END D_CURRENCY,
    FG_END_SUM D_CURRENCY,
    FG_DRIVER_ID D_ID,
    FG_MANAGER_ID D_ID,
    FG_BOOKER_ID D_ID,
    FG_DRIVER_DTSIGN D_DATETIME,
    FG_MANAGER_DTSIGN D_DATETIME,
    FG_BOOKER_DTSIGN D_DATETIME,
    FG_DRIVER_EDL_ID D_ID,
    FG_MANAGER_EDL_ID D_ID,
    FG_BLS_ID D_ID,
    FG_STATUS D_INTEGER,
    PEN_NAME D_STR_255,
    PET_NAME D_STR_255,
    PET_LABEL D_STR_CODE,
    PET_IS_RENT D_INTEGER,
    PET_NORMA_KM D_INTEGER,
    PED_EXPENSE_SUMMER D_CURRENCY,
    PED_EXPENSE_SUMMER_MOUNT D_CURRENCY,
    PED_EXPENSE_WINTER D_CURRENCY,
    PED_EXPENSE_WINTER_MOUNT D_CURRENCY,
    REG_NAME D_STR_CODE,
    DRIVER_FIO D_STR_100,
    MANAGER_FIO D_STR_100,
    BOOKER_FIO D_STR_100,
    DRIVER_JT D_STR_100,
    MANAGER_JT D_STR_100,
    BOOKER_JT D_STR_100,
    DRIVER_LOGIN D_STR_CODE,
    MANAGER_LOGIN D_STR_CODE,
    BOOKER_LOGIN D_STR_CODE,
    DRIVER_QR D_STR_LONG,
    MANAGER_QR D_STR_LONG,
    IS_CAR D_SMALLINT,
    STATE D_STR_20)
AS
begin
  for
    select
      fg_id,
      fg_dtwriteoff,
      fg_dtbeg,
      fg_dtend,
      fg_office_id,
      fg_pet_id,
      fg_start,
      fg_start_sum,
      fg_fillup,
      fg_fillup_sum,
      fg_whours,
      fg_wrate,
      fg_usedup,
      fg_usedup_sum,
      fg_end,
      fg_end_sum,
      fg_driver_id,
      fg_manager_id,
      fg_booker_id,
      fg_driver_dtsign,
      fg_manager_dtsign,
      fg_booker_dtsign,
      fg_driver_edl_id,
      fg_manager_edl_id,
      fg_bls_id,
      fg_status,
      pet_name,
      pet_label,
      pet_is_rent,
      pet_norma_km,
      reg_name,
      driver_fio,
      manager_fio,
      booker_fio,
      is_car,
      state
    from hp_fuelgenreg_view (
      1000000,
      current_date,
      current_date,
      :obj_id)
  into
    :fg_id,
    :fg_dtwriteoff,
    :fg_dtbeg,
    :fg_dtend,
    :fg_office_id,
    :fg_pet_id,
    :fg_start,
    :fg_start_sum,
    :fg_fillup,
    :fg_fillup_sum,
    :fg_whours,
    :fg_wrate,
    :fg_usedup,
    :fg_usedup_sum,
    :fg_end,
    :fg_end_sum,
    :fg_driver_id,
    :fg_manager_id,
    :fg_booker_id,
    :fg_driver_dtsign,
    :fg_manager_dtsign,
    :fg_booker_dtsign,
    :fg_driver_edl_id,
    :fg_manager_edl_id,
    :fg_bls_id,
    :fg_status,
    :pet_name,
    :pet_label,
    :pet_is_rent,
    :pet_norma_km,
    :reg_name,
    :driver_fio,
    :manager_fio,
    :booker_fio,
    :is_car,
    :state
  do
  begin
    if (fg_driver_dtsign is not null) then
      driver_login = (
        select first 1 un_name
        from username
        where un_from_c_id = :fg_driver_id
        order by un_id desc);
    if (fg_manager_dtsign is not null) then
      manager_login = (
        select first 1 un_name
        from username
        where un_from_c_id = :fg_manager_id
        order by un_id desc);
    if (fg_booker_dtsign is not null) then
      booker_login = (
        select first 1 un_name
        from username
        where un_from_c_id = :fg_booker_id
        order by un_id desc);

    driver_jt = (
      select jl_name
      from hr_pstatus
        inner join joblist
          on jl_id = ps_from_jl_id
      where ps_from_c_id = :fg_driver_id
        and :fg_dtwriteoff between ps_date_begin and ps_date_end
        and ps_is_working = 1);
    manager_jt = (
      select jl_name
      from hr_pstatus
        inner join joblist
          on jl_id = ps_from_jl_id
      where ps_from_c_id = :fg_manager_id
        and :fg_dtwriteoff between ps_date_begin and ps_date_end
        and ps_is_working = 1);
    booker_jt = (
      select jl_name
      from hr_pstatus
        inner join joblist
          on jl_id = ps_from_jl_id
      where ps_from_c_id = :fg_booker_id
        and :fg_dtwriteoff between ps_date_begin and ps_date_end
        and ps_is_working = 1);

    driver_qr = f_edoc_qr_by_edl(:fg_driver_edl_id, :fg_driver_id);
    manager_qr = f_edoc_qr_by_edl(:fg_manager_edl_id, :fg_manager_id);

    select first 1
      ped_expense_summer,
      ped_expense_summer_mount,
      ped_expense_winter,
      ped_expense_winter_mount
    from petdata
    where ped_from_pet_id = :fg_pet_id
      and ped_date <= :fg_dtend
    order by ped_date desc, ped_id desc
    into
      :ped_expense_summer,
      :ped_expense_summer_mount,
      :ped_expense_winter,
      :ped_expense_winter_mount;

    pen_name = (
      select first 1
        peo_petrol_name
      from aarep_trans_talon_saldo(:fg_pet_id, :fg_dtbeg)
      where peo_from_reg_id = :fg_office_id and peo_liter > 0
      order by peo_oper_date desc);

    suspend;
  end
end
