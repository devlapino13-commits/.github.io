CREATE PROCEDURE HP_CALC_REPAYMENT_OBLIGATIONS(
    MA_ID D_ID,
    ACL_ID D_ID = 0)
RETURNS (
    REPAYMENT_OBLIGATIONS DECIMAL(15,2),
    DATA_SOURCE D_INTEGER,
    TEXT_ERROR D_STR_255,
    KBM_ID D_ID,
    KBM_REL_ID D_ID,
    SAS_EIL_ID D_ID,
    CROC_ID D_ID)
AS
declare variable ca_id d_id;
declare variable k_id_tmp d_id;
declare variable ma_date d_date;
declare variable sum_debt d_currency;
declare variable k_id d_id;
declare variable mac_id d_id;
declare variable c_id d_id;
declare variable ishenim_kib_rep_available d_bool;
declare variable sas_kib_rep_available d_bool;
declare variable pr_code d_str_100;
declare variable base_ma_id d_integer;
declare variable base_k_id d_id;
declare variable sas_bls_id d_id;
begin
  -- RFCKG-4535 расчет repayment_obligations (Погашение других долговых обязательств)
  text_error = '';
  ishenim_kib_rep_available = 0;
  sas_kib_rep_available = 0;

  select first 1
         pr.pr_code,
         ma.ma_date_oper
  from miniapp ma
  inner join product pr on pr.pr_id = ma.ma_from_pr_id
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.2999'
  order by ma.ma_number desc
  into pr_code,
       ma_date;

  select mac.mac_id,
         mac.mac_from_c_id
  from miniappclient mac
  where mac.mac_from_ma_id = :ma_id and
        mac.mac_amount > 0 rows 1
  into mac_id,
       c_id;

  select k.k_id
  from kontract k
  where k.k_from_ma_app_id = :ma_id
  into k_id;

  k_id = coalesce(k_id, 0);

  select ca.ca_id
  from cfp_anketa ca
  where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
        ca.ca_object_id = :mac_id and
        ca.ca_from_c_id = :c_id and
        (
          coalesce(:acl_id, 0) = 0
          or ca.ca_from_acl_id = :acl_id
        ) and
        (
           coalesce(:acl_id, 0) <> 0
           or ca.ca_from_acl_id is null
         )
        rows 1
  into ca_id;

  ca_id = coalesce(:ca_id, 0);
  ------------------------------------------------------------------------------------------------
  -------------------------- Расчет repayment_obligations  ---------------------------------------

  repayment_obligations = 0;
  ------------------------------------------------------------------------- Считаем Долг клиента в МБулак
  sum_debt = 0;
  for select ks.ks_from_k_id
  from knclient kc
  inner join kontstate ks on kc.kc_from_k_id = ks.ks_from_k_id and
        ks.ks_state = 1 and
        :ma_date between ks.ks_date_begin and ks.ks_date_end
  where kc.kc_from_c_id = :c_id and
        kc.kc_amount_approved > 0 and
        kc.kc_from_k_id <> :k_id
  into k_id_tmp
  do
  begin
    select sum(zad_os + zad_perc + sh_calc_perc + sh_calc_fine) / coalesce(count(*), 1)
    from aarep_make_zad_list(:k_id_tmp, :ma_date, 1)
    into sum_debt;
    repayment_obligations = repayment_obligations + :sum_debt;
  end

  if (pr_code containing '.insur_kontrakt.') then  -- RFCKG-4831 Берем данные КИБ из основного кредита
  begin
    select lir.lir_kontract_base
    from kontract k
    inner join loan_insurance_relation lir on k.k_id = lir.lir_kontract_insurance
    where k.k_from_ma_app_id = :ma_id
    into base_k_id;

    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :base_k_id
    into base_ma_id;

    if(coalesce(base_ma_id, 0) <> 0) then
      ma_id = base_ma_id;
    else
      text_error = 'Не удалось найти основной контракт при расчете КПД';
  end

  ----------------------------------------------------------------------- Проверяем доступность Save&Sound

  select first 1
            eil.eil_id,
            eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
  where ((eil.eil_date between :ma_date - 1 and :ma_date)
        or
        (eil.eil_object_kind = f_objkind('miniapp', 'id') and eil.eil_object_id = :ma_id)) and  -- RFCKG-5396
        eil.eil_from_c_id = :c_id and
        eic.eic_code = 'ss_cib_report' and
        eil.eil_status = 2
  order by eil.eil_status_datetime desc
  into sas_eil_id,
       sas_bls_id;

  sas_eil_id = coalesce(sas_eil_id, 0);

  if (sas_eil_id <> 0) then  -- Если по клиенту есть данные в КИБ
    sas_kib_rep_available = 1;

  ----------------------------------------------------------------------- Проверяем доступность Ishenim

  select kbm_id
  from hp_kibminiapp_last(:c_id, :ma_id)
  where kbm_is_checked = 1
  into kbm_id;

  kbm_id = coalesce(kbm_id, 0);

  if (kbm_id <> 0) then
    ishenim_kib_rep_available = 1;

  ----------------------------------------------------------------------- Считаем Долг клиента по Отчету КИБ без учета МБулак
  if (ishenim_kib_rep_available = 1) then
  begin
    -- 1.3. Расчет значения должен производиться по формуле, аналогичной формуле, используемой в КБ Ишеним.
    data_source = f_get_rd_id('CfpAnketaCalcSource.Ishenim');
    -- RFCKG-3810, RFCKG-4084
    repayment_obligations = repayment_obligations + coalesce((select sum(iif(kh.kh_credittype = 'GROUP_LOAN',
                                                                     ((kh.kh_unpaidamount + kh.kh_unpaidpercenamount) / kh.kh_amount * kh.kh_ownamount) * (select curr_rate
                                                                                                                                                           from hp_convert_kib_curr(kh.kh_currency)),
                                                                     (kh.kh_unpaidamount + kh.kh_unpaidpercenamount) * (select curr_rate
                                                                                                                        from hp_convert_kib_curr(kh.kh_currency))) / iif(datediff(month, :ma_date, kh.kh_datestop) > 0, datediff(month, :ma_date, kh.kh_datestop) + iif(extract(day from kbm.kbm_checked_datetime) < extract(day from kh.kh_datestop), 1, 0), 1))
                                                              from kibhistory kh
                                                              inner join kibminiapp kbm on kh.kh_from_kbm_id = kbm.kbm_id
                                                              where kh.kh_from_kbm_id = :kbm_id and
                                                                    kh.kh_is_our = 0 and
                                                                    kh.kh_is_debtor = 1 and
                                                                    kh.kh_creditstatus not containing 'settled'), 0);
  end
  else if (sas_kib_rep_available = 1) then
  begin
    -- RFCKG-4923 -- 1.1. В случае, если по техническим причинам КИБ Ишеним недоступен, то  в Анкете анализа клиента в 7 поле робот должен автоматически подтягивать числовые значение из отчета Сейфенд саунд.
    data_source = f_get_rd_id('CfpAnketaCalcSource.SafeAndSound');
    -- Для Клиента
    sum_debt = 0;
    for select (outstanding_inst + interest_outstanding_inst) * (select curr_rate
                                                                 from hp_convert_kib_curr(currency_inst)) -- остаток по ОС  + проценты * курс валюты займа
               / iif(datediff(month, :ma_date, end_date_inst) > 0,
                      datediff(month, :ma_date, end_date_inst) + iif(extract(day from eil_datetime) < extract(day from end_date_inst), 1, 0),
                       1)
    from print_report_safe_and_sound(:ma_id, :c_id)
    where is_our_inst = 0 and
          loan_status_inst = 'Открытый' and
          role_str_inst = 'Заемщик'
    into sum_debt
    do
    begin
      repayment_obligations = repayment_obligations + coalesce(sum_debt, 0);
    end
  end
  else
  begin
    text_error = 'Не получен отчет от КИБ при расчете КПД';
  end

  repayment_obligations = coalesce(repayment_obligations, 0);

  if (text_error = '') then
  begin
    insert into cfp_repayment_oblig_calc (croc_from_ca_id, croc_kbm_id, croc_sas_eil_id, croc_other_debt_repayment, croc_source)
    values (:ca_id, iif(:kbm_id = 0, null, :kbm_id), iif(:sas_eil_id = 0, null, :sas_eil_id) , :repayment_obligations, :data_source)
    returning croc_id
    into croc_id;
  end

  suspend;
end
