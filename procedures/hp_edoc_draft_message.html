CREATE PROCEDURE HP_EDOC_DRAFT_MESSAGE(
    EDOC_KIND D_STR_CODE,
    OBJ_ID D_ID,
    C_ID D_ID)
RETURNS (
    MSG_TEXT D_STR_8100)
AS
declare variable edc_state d_smallint;
declare variable obj_kind d_integer;
declare variable edl_id d_id;
declare variable eds_id d_id;
declare variable pin_code d_str_code;
declare variable eds_uuid d_str_code;
declare variable params d_str_8100;
begin
  -- Процедура возвращает ссылку и пинкод для подписания вместо клиента
  -- Алгоритм работает только с проектами в тестовом режиме. Если боевой режим, возвращает пустую строку

  msg_text = '';
  edoc_kind = f_edoc_kind(edoc_kind, 'id');

  select edc_state, edc_object_kind
  from edoc_cfg
  where edc_id = :edoc_kind
  into edc_state, obj_kind;

  if (edc_state <> 3) then
  begin
    select edl_id
    from edoc_list
    where edl_from_edc_id = :edoc_kind and
          edl_object_kind = :obj_kind and
          edl_object_id = :obj_id and
          edl_status = 0
    order by edl_id desc
    rows 1
    into edl_id;
  
    if (edl_id is not null) then
    begin
      select eds_id, edm_hash_pin
      from edoc_sign, edoc_msg
      where eds_from_edl_id = :edl_id and
            eds_from_c_id = :c_id and
            eds_status = 0 and
            edm_from_eds_id = eds_id and
            edm_status = 0
      order by edm_from_eds_id desc, edm_from_msg_id desc
      rows 1
      into eds_id, pin_code;

      if (eds_id is not null and coalesce(:pin_code, '') <> '') then
      begin
        if (char_length(pin_code) <= 6) then
        begin
          eds_uuid = f_uuid('edoc_sign', eds_id);
          msg_text = 'Тестовый режим, подпишите вместо клиента нажав %1.<br>Пин-код: <b>%2</b>';

          params = '';
          params = case f_get_db_id()
            when 81001 then 'https://edoc.molbulak.ru/'
            when 41701 then 'https://edoc.molbulak.kg/'
            when 94401 then 'https://edoc.molbulak.com/'
          end;
          params = trim(params);
          params = params || eds_uuid;
          params = f_format('<a href="%1">ссылку</a>', params);
          msg_text = f_format(msg_text, params, pin_code);
        end
      end
    end
  end
  suspend;
end
