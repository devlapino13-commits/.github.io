CREATE PROCEDURE HP_GET_KONTRACT_SUD_INFO(
    K_ID D_ID,
    C_ID D_ID)
RETURNS (
    LOAN_MESSAGE D_STR_2048)
AS
declare variable DELINQ_DAYS_CNT D_INTEGER;
declare variable JSON_TEMPLATE D_STR_2048;
declare variable SMR_ID D_ID;
declare variable EDS_ID D_ID;
begin
    if (f_get_db_id() <> 81001) then
    begin
        -- Для других стран (не РФ) возвращаем пустой JSON
        loan_message = '{}';
        suspend;
        exit;
    end

    -- RFCRU-5274: Если клиент банкрот со статусом "Реализация имущества завершена", то отображаем соответствующее сообщение
    if (exists(select *
               from bankrupt brt, hp_get_kontur_info(brt.brt_from_eil_id) ki
               where brt.brt_from_c_id = :c_id and
                     current_date between brt.brt_date_begin and brt.brt_date_end and
                     ki.status = 'Реализация имущества завершена')) then
    begin
        json_template = '{"is_critical":%1,"content":[{"type":"text","display":"%2"}]}';
        loan_message = f_format(json_template, '0', '*Вы освобождены  от исполнения обязательств в связи с вашим банкротством');
        suspend;
        exit;
    end  

    select first 1 smr.smr_id
    from sud_mainreq smr
    where smr.smr_from_k_id = :k_id and
          smr.smr_status = 2
    order by smr.smr_date desc
    into :smr_id;

    select first 1 eds.eds_id
    from edoc_list edl
    inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
    inner join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id
    where edl.edl_object_kind = f_objkind('sud_mainreq', 'id') and
         edl.edl_object_id = :smr_id and
         edc.edc_code = 'pretrial_claim'
    order by edl.edl_create_datetime desc
    into :eds_id;

    if (coalesce(eds_id, 0) <> 0) then
    begin
        json_template = '{"is_critical":1,"content":[{"type":"text","display":"%2"},{"type":"link","value":"%1","display":"Претензия"},{"type":"text","display":"%3"}]}';

        loan_message = f_format(:json_template,
                                trim('https://edoc.molbulak.ru/') || f_uuid('edoc_sign', :eds_id),
                                'Вам направлена досудебная Претензия, ознакомиться с текстом можно по ссылке: ',
                                'Тел: +7(906)7365444. В случае не оплаты задолженности дело будет передано в суд.'
                                );

        suspend;
        exit;
    end


    -- Получаем количество дней просрочки по кредиту
    select iif(current_date - oks.oks_delinq_date < 0, 0, current_date - oks.oks_delinq_date)
    from oksummary oks
    where (current_date between oks.oks_date_begin and oks.oks_date_end) and
          oks.oks_from_k_id = :k_id and
          oks_close_state = 0
    into :delinq_days_cnt;
    
    json_template = '{"is_critical":%1,"content":[{"type":"text","display":"%2"}]}';

    if (delinq_days_cnt = 0) then
        loan_message = f_format(:json_template, '0', 'Оплачивайте вовремя или заранее, чтобы сохранить низкую ставку по займу.');
    else if (delinq_days_cnt between 1 and 30) then
        loan_message = f_format(:json_template, '1', 'Срочно оплатите просроченный долг, чтобы не испортить кредитную историю и сохранить доступ к кредитованию.');
    else if (delinq_days_cnt > 30) then
        loan_message = f_format(:json_template, '1', 'Срочно закройте просроченный долг. Ваша кредитная история ухудшается, вы можете потерять доступ к кредитованию.');
    else
        loan_message = '{}';

    suspend;
end
