CREATE PROCEDURE DUPLICATED_IMEI_ACTIONS(
    KMI_IMEI D_ID NOT NULL,
    K_ID D_ID NOT NULL,
    EXISTING_KML_ID D_INTEGER NOT NULL,
    UN_ID D_ID NOT NULL)
AS
declare variable k_last_reg_id d_integer;
declare variable kontragent_fio d_str_100;
declare variable kf_officer_fio d_str_100;
declare variable reg_name d_str_code;
declare variable email_list d_str_100;
declare variable kf_officer_id d_id;
begin
  select k.k_last_reg_id, c.c_name_fio, kal.kal_new_kf_officer, c2.c_name_fio, reg.reg_name
  from kontract k
  inner join region reg on reg.reg_unicode = k.k_last_reg_id
  inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
  inner join kontmat km on km.kml_object_id = k.k_id
  inner join kontmat_imei kmi on km.kml_id = kmi.kmi_kml_id
  inner join agreements ag on ag.ag_id = km.kml_from_ag_id
  inner join clients c on c.c_id = ag.ag_object_id
  inner join clients c2 on c2.c_id = kal.kal_new_kf_officer
  where km.kml_object_kind = 9
    and km.kml_status = 1
    and kal.is_delete = 0
    and current_date between kal.kal_date and kal.kal_date_end
    and ag.ag_object_kind = f_objkind('clients', 'id')
    and kmi.kmi_imei = :kmi_imei
    and k.k_id = :k_id
  into: k_last_reg_id, kontragent_fio, kf_officer_id, kf_officer_fio, reg_name;

  --письмо ОВШК
  email_list = 'imei_mirm@mbulak.kg';
  execute procedure hp_send_quick_message('email', email_list, 'Совпадение IMEI кода при оформлении заявки',
    'Название Офиса: ' || :reg_name ||
    ', фамилия Контрагента: ' || kontragent_fio ||
    ', ФИО ВИПС: ' || kf_officer_fio ||
    ', Совпадающий IMEI- код: ' || kmi_imei,
     '', null, :un_id);

  insert into duplicated_imei(di_from_k_id, di_last_reg_id, di_kf_officer, di_from_kml_id)
  values (:k_id, :k_last_reg_id, :kf_officer_id, :existing_kml_id);
end
