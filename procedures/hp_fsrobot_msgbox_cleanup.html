CREATE PROCEDURE HP_FSROBOT_MSGBOX_CLEANUP
AS
declare variable msg_id d_id;
declare variable eds_id d_id;
declare variable msg_type d_smallint;
begin
  -- Очистка старых сообщений в fsrobot_msgbox
  for select msg_id, msg_type
  from fsrobot_msgbox
  where
    dateadd(day, msg_lifetime_days, msg_datetime) < current_timestamp and
    msg_lifetime_days >= 0
  into msg_id, msg_type
  do
  begin
    -- Проверяем связку с эл. документами
    eds_id = null;
    select edm_from_eds_id
    from edoc_msg
    where edm_from_msg_id = :msg_id
    into eds_id;

    -- Если есть связка, письма удаляем, остальное не трогаем
    if (msg_type = 0 and eds_id is not null) then
    begin
      delete from edoc_msg
      where edm_from_msg_id = :msg_id
        and edm_from_eds_id = :eds_id;
    end

    if (eds_id is null or msg_type = 0) then
    begin
      delete from fsrobot_msgbox
      where msg_id = :msg_id;
    end
  end
end
