CREATE PROCEDURE HP_PERSTEMP_STATE_INFO(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    CS_STAFF_STATE D_SMALLINT,
    CS_CLIENT_STATE D_SMALLINT,
    RES_ADD_INFO D_STR_255)
AS
declare variable un_c_id d_id;
begin
  select cs_staff_state, cs_client_state
  from clientstate
  where cs_from_c_id = :c_id
  into cs_staff_state, cs_client_state;

  if (cs_staff_state <> 1) then
  begin
    if (un_id is null) then
    begin
      un_id = uxf$user();
      when any do
        un_id = null;
    end

    if (un_id is not null) then
    begin
      select un_from_c_id
      from username
      where un_id = :un_id
      into un_c_id;

      if (un_c_id is not null) then
      begin
        if (exists(select *
        from hr_pstatus, joblist, jobkind
        where ps_from_c_id = :un_c_id and
              current_date between ps_date_begin and ps_date_end and
              ps_status = 1 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              ps_is_working = 1 and
              jk_system_code in ('JobSystem.HR_Department', 'JobSystem.HR_Service'))) then
          cs_staff_state = 2; -- Будущий сотрудник
      end
    end
  end

  res_add_info = '';
  if (cs_staff_state = 0) then
    res_add_info = res_add_info || 'бывший сотрудник';
  else
  if (cs_staff_state = 1) then
    res_add_info = res_add_info || 'сотрудник';
  else
  if (cs_staff_state = 2) then
    res_add_info = res_add_info || 'отправлен отделом кадров';

  if (res_add_info <> '') then
    res_add_info = res_add_info || ', ';

  if (cs_client_state = 0) then
    res_add_info = res_add_info || 'кредитная заявка';
  else
  if (cs_client_state = 1) then
    res_add_info = res_add_info || 'активный клиент';
  else
  if (cs_client_state = 2) then
    res_add_info = res_add_info || 'бывший клиент';
  else
  if (cs_client_state = -1) then
    res_add_info = res_add_info || 'без кредитной истории';

  suspend;
end
