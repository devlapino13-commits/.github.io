CREATE PROCEDURE AAREP_OPERKRED_LOYAL_REPAY(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    K_KONTRACT_NO D_STR_CODE,
    K_DISB_DATE D_DATE,
    PR_NAME D_STR_255,
    SH_ID D_INTEGER,
    SH_DATE D_DATE,
    PAY_OS D_CURRENCY,
    PAY_PERC D_CURRENCY,
    PAY_DATE D_DATE,
    LO_FIO D_STR_100,
    REG_NAME D_STR_100)
AS
declare variable k_id d_id;
declare variable kf_id d_id;
begin
  -- погашения по займам клиентов, привлеченные через агентов
  for select ks_from_k_id, ks_date_begin, k_kontract_no, pr_name, k_from_kf_id
  from kontstate, operagr, kontract, product
  where ogr_ol_date <= :rep_date_end and
        ogr_ol_is_active = 1 and
        ogr_sign = 1 and
        ogr_amount > 0 and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_from_ol_id = ogr_from_ol_id and
        ks_from_k_id = k_id and
        k_from_pr_id = pr_id
  into k_id, k_disb_date, k_kontract_no, pr_name, kf_id
  do
  begin
    lo_fio = '';
    reg_name = '';

    for select ok_ol_date, ok_sh_id, ok_sh_date, sum(-ok_os), sum(-ok_perc)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date between :rep_date_begin and :rep_date_end and
          ok_type = 4 and
          ok_sh_id in (1, 2) and
          (ok_os <> 0 or ok_perc <> 0)
    group by 1, 2, 3
    into pay_date, sh_id, sh_date, pay_os, pay_perc
    do
    begin
      if (lo_fio = '') then
      begin
        select c_name_fio, reg_code || ' '  || reg_name
        from knaktlist kal, clients, region
        where kal.kal_from_kf_id = :kf_id and
              kal.is_delete = 0 and
              :rep_date_end between kal.kal_date and kal.kal_date_end and
              kal.kal_new_kf_officer = c_id and
              kal.kal_new_reg_id = reg_unicode
        into lo_fio, reg_name;
      end

     suspend;
   end
  end
end
