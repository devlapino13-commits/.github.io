CREATE PROCEDURE MONETA_CREDITS_IN_PROCESS
RETURNS (
    ESM_ID D_ID,
    MNT_ID D_STR_10)
AS
declare variable k_id d_id;
begin
  -- Процедура предоставляет список выплат по Монете:
  -- карту/СБП/расчетный счет,
  -- по которым процесс выплаты находится в процессе.
  -- Вызывается из сервиса выдач по Монете.

  for select esm_id,
             esm_from_k_id
      from extsysmain
      where esm_project = 'moneta_disb' and
            esm_action in ('PhoneDisb', 'CardDisb', 'AccDisb') and
            esm_status = 999
      into esm_id,
           k_id
  do
  begin
    select iif(pr_code containing '.oivo_pack.', f_get_const('moneta_oivo_acc'), f_get_const('moneta_acc'))
    from kontract
    inner join product on pr_id = k_from_pr_id
    where k_id = :k_id
    into mnt_id;
    
    suspend;
   end
end
