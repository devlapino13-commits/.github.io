CREATE PROCEDURE AAREP_OPERLIST_COPY(
    OL_ORIG_ID DECIMAL(18,0),
    NEW_DD DATE,
    UN_ID DECIMAL(18,0))
AS
declare variable reg integer;
declare variable orig_kind integer;
declare variable orig_reg integer;
declare variable doc_no integer;
declare variable new_id decimal(18,0);
declare variable nn decimal(18,0);
declare variable new_main decimal(18,0);
begin
  reg = cast(left(:ol_orig_id, 7) as integer);
  select kv
  from get_kv(:reg, 5)
  into :new_id;
  new_main = :new_id;

  select ol_doc_kind, ol_from_reg_id
  from operlist
  where ol_id = :ol_orig_id
  into :orig_kind, :orig_reg;

  select num
  from last_doc_number(:new_dd, :orig_kind, :orig_reg)
  into :doc_no;

  insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                        ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                        ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                        ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month, ol_parent_id, ol_manager_c_id)
  select :new_id, ol_from_reg_id, ol_oper_type, :new_dd, :doc_no, ol_doc_no_str, ol_doc_kind, ol_memo, ol_status,
         is_delete, ol_datetime, :un_id, ol_document_client, ol_from_mo_id, ol_is_mo_modified, ol_document_office,
         ol_acc_no_client, ol_acc_no_office, ol_oper_subtype, ol_pocket, ol_doc_client_name, ol_is_client, ol_year,
         ol_month, ol_parent_id, ol_manager_c_id
  from operlist
  where ol_id = :ol_orig_id;

  insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                        ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                        ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
  select :new_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa,
         ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
         ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id
  from operbook
  where ob_from_ol_id = :ol_orig_id;

  if (f_get_const('operbook_kind') = 1) then
  begin
    insert into kn_operbook (kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                             kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id, kob_ol_is_active,
                             kob_detail_id, kob_client_dt, kob_client_kt, kob_from_ol_id, kob_from_kc_id)
    select kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val, kob_summa_tax_dt,
           kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id, kob_ol_is_active, kob_detail_id, kob_client_dt, kob_client_kt,
           :new_id, kob_from_kc_id
    from kn_operbook
    where kob_from_ol_id = :ol_orig_id;
  end

  for select ol_id
      from operlist
      where ol_parent_id = :ol_orig_id
      into :nn
  do
  begin
    select kv
    from get_kv(:reg, 5)
    into :new_id;

    insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                          ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                          ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                          ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month, ol_parent_id, ol_manager_c_id)
    select :new_id, ol_from_reg_id, ol_oper_type, :new_dd, :doc_no, ol_doc_no_str, ol_doc_kind, ol_memo, ol_status,
           is_delete, ol_datetime, :un_id, ol_document_client, ol_from_mo_id, ol_is_mo_modified, ol_document_office,
           ol_acc_no_client, ol_acc_no_office, ol_oper_subtype, ol_pocket, ol_doc_client_name, ol_is_client, ol_year,
           ol_month, :new_main, ol_manager_c_id
    from operlist
    where ol_id = :nn;

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                          ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                          ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date,
                          ob_ol_is_active, ob_ol_reg_id)
    select :new_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa,
           ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
           ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id
    from operbook
    where ob_from_ol_id = :nn;

    if (f_get_const('operbook_kind') = 1) then
    begin
      insert into kn_operbook (kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                               kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id, kob_ol_is_active,
                               kob_detail_id, kob_client_dt, kob_client_kt, kob_from_ol_id, kob_from_kc_id)
      select kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val, kob_summa_tax_dt,
             kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id, kob_ol_is_active, kob_detail_id, kob_client_dt, kob_client_kt,
             :new_id, kob_from_kc_id
      from kn_operbook
      where kob_from_ol_id = :nn;
    end
  end
end
