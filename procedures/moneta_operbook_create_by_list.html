CREATE PROCEDURE MONETA_OPERBOOK_CREATE_BY_LIST
AS
declare variable esm_id d_id;
declare variable base_esm_id d_id;
declare variable esm_project str_20;
begin
   -- процедура предназначется для запуска роботом fsrobot_sh_main

   for select em.esm_id, em.esm_object_id, em.esm_project
   from extsysmain em
   where em.esm_project in ('moneta_disb', 'wirebank_disb')
   and em.esm_action = 'OperbookCreate'
   and em.esm_status = 0
   into :esm_id, :base_esm_id, :esm_project
   do
   begin
     if (esm_project = 'moneta_disb') then
     begin
       -- строю проводку по комисии за выдачу чз Монету
       execute procedure moneta_operbook_create(:base_esm_id);

       update extsysmain em
       set em.esm_status = 1
       where em.esm_id = :esm_id;
     end
     if (esm_project = 'wirebank_disb') then
     begin
       -- строю проводку по комисии за выдачу чз Wirebank(ТКБ)
       execute procedure wirebank_operbook_create(:base_esm_id);

       update extsysmain em
       set em.esm_status = 1
       where em.esm_id = :esm_id;
     end
   end
end
