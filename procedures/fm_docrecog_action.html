CREATE PROCEDURE FM_DOCRECOG_ACTION(
    UN_ID D_ID,
    MCL_ID D_ID,
    ACTION_KIND D_STR_CODE,
    DOC_DATA D_BLOB_TEXT)
RETURNS (
    RES_DATA D_BLOB_TEXT,
    RES_ID D_INTEGER,
    RES_TEXT D_STR_LONG)
AS
declare variable drm_id d_integer;
declare variable array_size d_integer;
declare variable tmp_json d_str_8100;
declare variable tmp_json_arr d_str_8100;
declare variable item_no d_integer;
declare variable drm_code d_str_code;
declare variable field_code d_str_code;
declare variable field_value d_str_255;
declare variable field_accepted d_smallint;
declare variable pk_id d_integer;
declare variable c_id d_id;
declare variable drl_id d_id;
declare variable drf_id d_integer;
declare variable drv_id d_id;
declare variable cds_id d_integer;
declare variable is_passport d_integer;
declare variable is_main d_integer;
declare variable cds_date_begin d_date;
declare variable cds_date_end d_date;
declare variable retry_text d_str_long;
declare variable unaccepted_count d_integer;
declare variable accepted_count d_integer;
declare variable tmp_date d_date;
declare variable bls_1 d_id;
declare variable bls_2 d_id;
declare variable acl_id d_id;
declare variable acl_status d_smallint;
declare variable rd_id d_integer;
declare variable drm_is_manual d_bool;
declare variable pk_passport_issuers d_str_255;
declare variable drm_show_fields_staff d_smallint;
declare variable drm_has_page1 d_smallint;
declare variable drm_has_page2 d_smallint;
declare variable drm_name d_str_255;
declare variable c_name_fio d_str_100;
declare variable pdt_code d_str_100;
declare variable lang d_str_lang;
declare variable i d_integer;
declare variable counter d_integer;
declare variable db_id d_id;
declare variable inn_value d_str_20;
declare variable grs_active d_smallint;
declare variable drr_id d_integer;
declare variable doc_no_phone_change d_str_255;
declare variable cli_passport_doc_no d_str_255;
declare variable drl_project d_smallint;
begin
  -- Значения action_kind
  -- draft_create - Создается черновик документа/клиента (МП, КМП)
  -- draft_create_fs - Создается черновик документа (Finstart)
  -- draft_create_web - Создается черновик документа на сайте (Edoc Service)
  -- draft_update - Обновляется черновик документа/клиента (КМП)
  -- draft_load - Загружается черновик документа/сохраненный документ (Finstart, МП)
  -- final_save - Сохраняется чистовик (МП, Финстарт)
  -- actionlist_update - Сохранение результатов оцифровки (Finstart)
  -- draft_delete - Удаление черновика (Finstart)
  -- add_clientdoc - Добавление документа клиента (когда в дубликатах выбран существующий)
  -- draft_refuse - Отправка (запроса) документа на доработку, с указанием причины (МП)
  -- client_passport_match - Сличение паспорта клиента, с которым он был верифицирован с
  --                         тем, который он приложил для смены номера телефона (КМП, RFCRU-4628)

  db_id = f_get_db_id();
  if (un_id is not null) then
    uxf$set_user(un_id);

  res_id = 0;
  res_text = '';
  tmp_json = '';
  grs_active = f_grs_active(:un_id);
  un_id = coalesce(un_id, 0);
  mcl_id = coalesce(mcl_id, 0);
  if (un_id <> 0) then
  begin
    lang = f_get_lang_code(un_id, null);
    uxf$set_user(un_id);
  end
  else
    lang = f_get_lang_code(null, mcl_id);

  if (action_kind in ('draft_update', 'draft_load', 'final_save', 'draft_delete', 'client_passport_match')) then
  begin
    drl_id = f_json_blob_get_value(doc_data, 'drl_id', null);
    if (drl_id is null) then
    begin
      res_id = 1;
      res_text = f_format(f_lang(lang, 'Не найдено поле %1'), 'drl_id');
      suspend;
      exit;
    end
  end

  if (action_kind in ('draft_create', 'draft_create_fs', 'draft_create_web', 'draft_load')) then
  begin
    /*
    {"c_id": 1010100106656, "drm_code":"kgz.id.type1",
      "fields":[
        {"code":"name", "value":"ИВАН", "is_accepted":"0"},
        {"code":"surname", "value":"ИВАНОВ", "is_accepted":"0"},
        {"code":"birth_date", "value":"01.01.1990", "is_accepted":"0"},
        {"code":"issue_date", "value":"01.01.2010", "is_accepted":"0"}
      ]
    }
    */
    if (action_kind = 'draft_load') then
    begin
      -- Получаем уже заполненные поля из docrecogmain
      select drl_from_drm_id, drl_from_c_id, pk_id, pdt_is_main_doc,
             drl_from_bls_1, drl_from_bls_2, coalesce(pk_passport_issuers, ''),
             drm_is_manual, drl_is_main_doc, drm_name
      from docrecogmain, docrecoglist, passkind, passdoctype
      where drl_id = :drl_id and
            pk_id = drl_from_pk_id and
            pk_from_pdt_id = pdt_id and
            drl_from_drm_id = drm_id
      into drm_id, c_id, pk_id, is_passport, bls_1, bls_2, pk_passport_issuers,
           drm_is_manual, is_main, drm_name;
    end

    if (action_kind in ('draft_create', 'draft_create_web')) then
    begin
      drm_code = f_json_blob_get_value(doc_data, 'drm_code', null);

      if (db_id = 81001 and drm_code containing '.phonechange') then
      begin
        drl_project = f_enum_id('docrecog_project', 'phone_change_no_access');

        if (drm_code <> 'selfie.phonechange') then
          -- Обрезаем у drm_code суффикс .phonechange, т.к. он служит только как признак проекта
          -- Но если это селфи, то не трогаем, т.к. этот документ было решено оставить
          drm_code = substring(drm_code from 1 for position('.phonechange' in drm_code) - 1);
      end

      select drm_id, drm_is_manual, drm_show_fields_staff
      from docrecogmain
      where drm_code = :drm_code
      into drm_id, drm_is_manual, drm_show_fields_staff;
      if (drm_id is null) then
      begin
        res_id = 1;
        res_text = f_format(f_lang(lang, 'Не найден документ с кодом: %1'), coalesce(drm_code, '<null>'));
        suspend;
        exit;
      end
    end

    if (action_kind = 'draft_create_fs') then
    begin
      drm_id = f_json_blob_get_value(doc_data, 'drm_id', null);
      if (drm_id is null) then
      begin
        res_id = 1;
        res_text = f_format(f_lang(lang, 'Не найден документ с id: %1'), coalesce(drm_id, '<null>'));
        suspend;
        exit;
      end

      select drm_name
      from docrecogmain
      where drm_id = :drm_id
      into drm_name;
    end

    if (action_kind in ('draft_create', 'draft_create_fs', 'draft_create_web')) then
    begin
      -- Вычисление pk_id
      select pk_id, pdt_is_main_doc, coalesce(pk_passport_issuers, ''),
             pdt_code
      from passkind, passdoctype
      where pk_from_drm_id = :drm_id and
            pk_from_pdt_id = pdt_id and
            pk_is_delete = 0
      into pk_id, is_passport, pk_passport_issuers, pdt_code;

      if (mcl_id <> 0) then
      begin
        -- Для МП клиентского c_id берем из пользователя
        select mcl_from_c_id
        from mc_login
        where mcl_id = :mcl_id
        into c_id;
        c_id = coalesce(c_id, 0);
        res_data = coalesce(:res_data, '');
      end
      else
        c_id = f_json_blob_get_value(doc_data, 'c_id', null);

      if (c_id is null) then
      begin
        res_id = 1;
        res_text = f_lang(lang, 'Не передан id клиента');
        suspend;
        exit;
      end

      if (c_id = 0) then
        c_id = null;

      if (pdt_code = 'anketa_questionnaire') then
      begin
        if (f_ft_valid(c_id, current_date) = 1) then
          res_text = 'По данному клиенту анкету-опросник закреплять не требуется.';
        else
        if (f_asp_valid(c_id, current_date) = 0) then
          res_text = 'До закрепления анкеты, необходимо подписать договор АСП.';
        if (res_text <> '') then
        begin
          res_id = 1;
          suspend;
          exit;
        end
      end

      is_main = 0;
      if (is_passport = 1) then
      begin
        if (c_id is null) then
          is_main = 1;
        else
        begin
          -- Вычисляем значение поля is_main, на основании существующих документов
          select cds_doc_date_begin, cds_doc_date_end
          from clientdocs
          where cds_from_c_id = :c_id and
                cds_is_main_doc = 1 and
                cds_is_delete = 0
          order by cds_doc_date_end desc
          rows 1
          into cds_date_begin, cds_date_end;

          if (cds_date_begin is null or cds_date_end < dateadd(month, 2, current_date)) then
            is_main = 1; -- Либо нет паспорта, либо срок истекает, предлагаем назначить основным
        end
      end

      if (drl_project is null) then
        drl_project = f_enum_id('docrecog_project', 'default');

      -- Вставка распознаваемого документа
      insert into docrecoglist (drl_from_drm_id, drl_from_pk_id, drl_from_c_id,
                                drl_from_un_id, drl_from_mcl_id, drl_status,
                                drl_is_main_doc, drl_project)
      values (:drm_id, :pk_id, :c_id, :un_id, :mcl_id, -1, :is_main, :drl_project)
      returning drl_id
      into :drl_id;

      -- Разбор полей и вставка в поля распознаваемого документа
      -- Если массив пустой (ничего не распозналось/нет распознавания),
      -- или отсутствует (draft_create_fs) блок будет пропущен
      array_size = f_json_blob_get_count(doc_data, 'fields');
      item_no = 0;
      while (item_no < array_size) do
      begin
        tmp_json = f_json_blob_get_value(doc_data, 'fields[' || item_no || ']', null);
        field_code = f_json_str_get_value(tmp_json, 'code', '');
        field_value = left(trim(f_json_str_get_value(tmp_json, 'value', '')), 255);
        field_accepted = f_json_str_get_value(tmp_json, 'is_accepted', '');

        drf_id = null;
        select drf_id
        from docrecogfield
        where drf_from_drm_id = :drm_id and
              drf_api_code = :field_code
        into drf_id;
        if (drf_id is not null) then
        begin
          insert into docrecogvalue (drv_from_drl_id, drv_from_drf_id,
                                     drv_is_accepted, drv_recog_value,
                                     drv_final_value)
          values (:drl_id, :drf_id, :field_accepted, :field_value, :field_value);
        end
        else
        begin
          -- Пока игнорируем прилетевшие check-поля. Не сохраняем и не ругаемся.
          if (field_code not starting 'check_') then
          begin
            res_id = 1;
            res_text = f_format(f_lang(lang, 'Не найдено поле с кодом: %1'), coalesce(drm_code, '<null>') || '.' || coalesce(field_code, '<null>'));
            suspend;
            exit;
          end
        end
        item_no = item_no + 1;
      end

      -- Досоздаем требуемые поля.
      -- Т.е. когда поле не пришло с клиента, но нам нужно чтобы оно было создано
      for select drf_id
          from docrecogfield
          left join docrecogvalue on drv_from_drf_id = drf_id and
                drv_from_drl_id = :drl_id
          where drf_inactive = 0 and
                drf_from_drm_id = :drm_id and
                drv_id is null
          into drf_id
      do
      begin
        insert into docrecogvalue (drv_from_drl_id, drv_from_drf_id,
                                   drv_is_accepted, drv_recog_value,
                                   drv_final_value)
        values (:drl_id, :drf_id, 0, '', '');
      end

      -- Обновляем drv_is_accepted_calc для всех полей документа
      execute procedure hp_docrecog_calc_accepted(:drl_id);

      -- Поля из клиента заполнены и добавлены пустые значения для тех, которые не получены
      -- Обновляем вычисляемые поля
      execute procedure fm_docrecog_calc_field(:drl_id, 1);
      if (db_id = 41701 and
          grs_active = 1 and
          is_passport = 1 and
          action_kind <> 'draft_create_fs') then
      begin
        field_value = f_docrecog_value(drl_id, 'inn', null);
        if (char_length(field_value) <> 14) then
        begin
          res_id = 1;
          res_text = f_lang(lang, 'ИНН указан не верно, прошу переснять паспорт клиента!');
          suspend;
          exit;
        end
        field_value = f_docrecog_value(drl_id, 'doc_no', null);
        if (char_length(field_value) <> 9) then
        begin
          res_id = 1;
          res_text = f_lang(lang, 'Номер документа указан не верно, прошу переснять паспорт клиента!');
          suspend;
          exit;
        end

        if (coalesce(mcl_id, 0) <> 0) then
          select res_id, res_text
          from mc_docrecog_check(:mcl_id, f_docrecog_value(:drl_id, 'inn', null), f_docrecog_value(:drl_id, 'doc_no', null))
          into res_id, res_text;
        if (res_id = 1) then
        begin
          suspend;
          exit;
        end
      end
    end

    tmp_json = '';
    -- Формируем сначала искусственное поле "Основной документ",
    -- отображаем только когда клиент уже создан и добавляется новый паспорт
    if (is_passport = 1 and
        c_id is not null) then
    begin
      tmp_json = tmp_json || iif(tmp_json = '', '', ',') || '{' || f_json_make_pair('name', f_lang(lang, 'Основной документ'), 'str') || ',' || f_json_make_pair('is_accepted', 1, 'num') || ',' || f_json_make_pair('not_null', 1, 'num') || ',' || f_json_make_pair('position', 0, 'num') || ',' || f_json_make_pair('data_type', 3, 'num') || ',' || -- Чекбокс
      f_json_make_pair('drv_id', -1, 'num') || ',' || -- Назначаем запись -1
      f_json_make_pair('value', is_main, 'num') || '}';
    end

    -- Переменные для подсчета неправильно распознаных полей, используются в draft_create
    unaccepted_count = 0;
    accepted_count = 0;

    -- Формируем данные по документу, и используемым полям
    for select drf_code, drf_name, drf_not_null, drf_data_type, drv_id,
               drv_final_value, drf_is_extra, drf_edit_mask, drf_edit_mask_hint,
               minvalue(drv_is_accepted, drv_is_accepted_calc) as is_accepted,
               drf_position
        from docrecogfield, docrecogvalue
        where drf_from_drm_id = :drm_id and
              drv_from_drl_id = :drl_id and
              drv_from_drf_id = drf_id and
              drf_inactive = 0 and
              ((:c_id is null and
              drf_invisible_on_client = 0 and
              iif(:db_id = 41701 and
              :grs_active = 1, drf_code in ('doc_no', 'inn', 'authority'), true)) or
              (:c_id is not null and
              drf_invisible_on_doc = 0))
        order by drf_position as cursor cur_drv
    do
    begin
      tmp_json_arr = '';
      if (cur_drv.drf_data_type = 4) then
      begin
        select res_json_extra
        from fm_docrecog_field_extra(cur_drv.drf_code, :pk_id, :un_id)
        into tmp_json_arr;
      end

      if (cur_drv.drf_is_extra = 0) then
      begin
        if (cur_drv.is_accepted = 1) then
          accepted_count = accepted_count + 1;
        else
          unaccepted_count = unaccepted_count + 1;
      end

      if (cur_drv.drf_data_type = 5 and
          pk_passport_issuers <> '') then
      begin
        tmp_json_arr = f_json_make_pair('ref_name', pk_passport_issuers, 'str');
        -- Ищем парный drv_id, для сохранения кода подразделения
        drv_id = null;
        select drv_id
        from docrecogvalue, docrecogfield
        where drf_from_drm_id = :drm_id and
              drf_code = 'authority_code' and
              drv_from_drl_id = :drl_id and
              drv_from_drf_id = drf_id
        into drv_id;

        if (drv_id is not null) then
          tmp_json_arr = tmp_json_arr || ',' || f_json_make_pair('link_drv_id', drv_id, 'num');
      end

      tmp_json = tmp_json || iif(tmp_json = '', '', ',') || '{' || f_json_make_pair('name', cur_drv.drf_name, 'str') || ',' || f_json_make_pair('is_accepted', cur_drv.is_accepted, 'num') || ',' || f_json_make_pair('not_null', cur_drv.drf_not_null, 'num') || ',' || f_json_make_pair('position', cur_drv.drf_position, 'num') || ',' || f_json_make_pair('data_type', cur_drv.drf_data_type, 'num') || ',' || f_json_make_pair('drv_id', cur_drv.drv_id, 'num') || ',' || f_json_make_pair('value', cur_drv.drv_final_value, 'str') || iif(cur_drv.drf_edit_mask <> '', ',' || f_json_make_pair('edit_mask', cur_drv.drf_edit_mask, 'str'), '') || iif(cur_drv.drf_edit_mask_hint <> '', ',' || f_json_make_pair('edit_mask_hint', cur_drv.drf_edit_mask_hint, 'str'), '') || iif(cur_drv.drf_data_type = 4 and tmp_json_arr <> '', ',' || f_json_make_pair('items', tmp_json_arr, 'array'), '') || iif(cur_drv.drf_data_type = 5 and tmp_json_arr <> '', ',' || tmp_json_arr, '') || '}';
    end

    if (action_kind = 'draft_create') then
    begin
      -- Если много ошибок, показываем сообщение. Только клиенту, только паспорт
      retry_text = '';
      if (is_passport = 1 and
          drm_is_manual = 0 and
          mcl_id <> 0) then
      begin
        if (accepted_count = 0) then
          retry_text = f_format(',"retry_text":"%1"', f_lang(lang, 'Нет качественно распознанных полей. Рекомендуем вернутся назад и переснять заново.'));
        else
        if (cast(unaccepted_count as float) / (accepted_count + unaccepted_count) >= 0.75) then -- т.е. проблемных более 75% от общего количества
          retry_text = f_format(',"retry_text":"%1 %2 %3. %4."', f_format(f_lang(lang, 'Некачественно распознано %1 полей. Рекомендуем вернутся назад и переснять заново.'), unaccepted_count));
      end

      -- Формируем ответ
      if (drm_show_fields_staff = 0 and
          un_id <> 0) then
      begin
        -- Если дошли сюда, значит ошибок нет, т.е. res_id = 0, res_text = ''
        res_data = '{' || f_json_make_pair('drl_id', drl_id, 'num') || ',"fields":[]}';
        res_text = f_lang(lang, 'Документ сохранен. Фото загружаются. После загрузки, документ будет отправлен на оцифровку.');
      end
      else if ((select count(*) from docrecogfield where drf_from_drm_id = :drm_id and drf_inactive = 0) = 0 and
                drl_project <> f_enum_id('docrecog_project', 'phone_change_no_access')) then
      begin
        -- RFCKZ-217
        -- Если в документе нет полей, пропускаем экран редактирования полей,
        -- но показываем, сохраняем документ и сообщаем о завершении редактирования
        select res_data, res_id, res_text
        from fm_docrecog_action(:un_id, :mcl_id, 'final_save', '{' || f_json_make_pair('drl_id', :drl_id, 'num') || ',"fields":[]}')
        into res_data, res_id, res_text;

        if (res_id = 0) then
            res_text = f_lang(lang, 'Документ сохранен. Фото загружаются.');
      end
      else
        res_data = '{' || f_json_make_pair('drl_id', drl_id, 'num') || retry_text || ',"fields":[' || tmp_json || ']}';
    end

    if (action_kind = 'draft_load') then
    begin
      tmp_json_arr = '';
      for select rd_id, rd_name
          from refdata
          where rd_main_code = 'DocrecogRefusal' and
                rd_is_active = 1 and
                rd_code <> rd_main_code
          order by rd_id, rd_name as cursor cur_rd
      do
      begin
        tmp_json_arr = tmp_json_arr || iif(tmp_json_arr = '', '', ',') || '{' || f_json_make_pair('id', cur_rd.rd_id, 'num') || ',' || f_json_make_pair('name', cur_rd.rd_name, 'str') || '}';
      end

      c_name_fio = 'Новый клиент';
      select c_name_fio
      from clients
      where c_id = :c_id
      into c_name_fio;

      res_data = '{' || f_json_make_pair('drl_id', drl_id, 'num') || ',' || f_json_make_pair('c_id', coalesce(c_id, 0), 'num') || ',' || f_json_make_pair('client_name', coalesce(c_name_fio, 'NULL'), 'str') || ',' || f_json_make_pair('drm_name', drm_name, 'str') || ',' || f_json_make_pair('drl_bls_id_1', bls_1, 'num') || ',' || f_json_make_pair('drl_bls_id_2', bls_2, 'num') || ',' || f_json_make_pair('is_manual', drm_is_manual, 'num') || ',' || '"fields":[' || tmp_json || '],' || '"refuse_reasons":[' || tmp_json_arr || ']' || '}';
    end

    if (action_kind = 'draft_create_fs') then
    begin
      select drm_has_page1, drm_has_page2
      from hp_get_drm_has_pages(:pk_id)
      into :drm_has_page1, :drm_has_page2;

      res_data = '{' || f_json_make_pair('pk_id', pk_id, 'num') || ',' || f_json_make_pair('drm_name', drm_name, 'str') || ',' || f_json_make_pair('drl_id', drl_id, 'num') || ',' || f_json_make_pair('has_page1', drm_has_page1, 'num') || ',' || f_json_make_pair('has_page2', drm_has_page2, 'num') || ',' || '"fields":[' || tmp_json || ']' || '}';
    end

    if (action_kind = 'draft_create_web') then
    begin
      res_data = '{' || f_json_make_pair('drl_id', drl_id, 'num') || '}';
    end
    suspend;
  end

  if (action_kind in ('draft_update', 'final_save')) then
  begin
    -- Разбор массива fields и обновление полей распознаваемого документа
    array_size = f_json_blob_get_count(doc_data, 'fields');
    item_no = 0;
    while (item_no < array_size) do
    begin
      tmp_json = f_json_blob_get_value(doc_data, 'fields[' || item_no || ']', null);
      drv_id = f_json_str_get_value(tmp_json, 'drv_id', null);
      field_value = trim(f_json_str_get_value(tmp_json, 'value', ''));

      if (drv_id is not null) then
      begin
        if (drv_id <> -1) then -- не системное поле
        begin
          if (db_id = 41701 and
              exists(select *
                     from docrecogvalue drv
                     inner join docrecogfield drf on drf.drf_id = drv.drv_from_drf_id and
                           drf.drf_code in ('doc_no', 'inn')
                     where drv.drv_id = :drv_id)) then
          begin
            counter = 0;
            i = 0;
            select drv.drv_final_value
            from docrecogvalue drv
            where drv.drv_id = :drv_id
            into inn_value;
            while (i < char_length(inn_value)) do
            begin
              i = i + 1;
              if (substring(inn_value from i for 1) <> substring(:field_value from i for 1)) then
                counter = counter + 1;
            end
            if (counter > 1) then
            begin
              res_id = 1;
              res_text = f_lang(lang, 'Можно менять только 1 символ');
              suspend;
              exit;
            end
          end
          update docrecogvalue
          set drv_final_value = :field_value
          where drv_id = :drv_id and
                coalesce(drv_final_value collate unicode, '###') <> coalesce(:field_value collate unicode, '###') and
                drv_from_drl_id = :drl_id; -- контроль, на всякий случай
        end
        else
        begin
          -- обновляем поле is_main_doc (drv_id = -1)
          update docrecoglist
          set drl_is_main_doc = :field_value
          where drl_id = :drl_id and
                coalesce(drl_is_main_doc, -1) <> :field_value;
        end
      end
      item_no = item_no + 1;
    end

    if (mcl_id <> 0) then
    begin
      select mcl_from_c_id
      from mc_login
      where mcl_id = :mcl_id
      into c_id;
      c_id = coalesce(c_id, 0);
      res_data = coalesce(:res_data, '');
    end
    else
      c_id = f_json_blob_get_value(doc_data, 'c_id', null);

    if (c_id = 0) then
      c_id = null;

    if (c_id is not null) then
    begin
      -- Проверка ДР только для паспортов
      select pdt_is_main_doc, drm_code
      from docrecoglist, docrecogmain, passkind, passdoctype
      where drl_id = :drl_id and
            drl_from_drm_id = drm_id and
            pk_id = drl_from_pk_id and
            pk_from_pdt_id = pdt_id
      into is_passport, drm_code;

      if (is_passport = 1 or drm_code = 'rus.residence.old') then
      begin
        field_value = f_docrecog_value(drl_id, 'birth_date', null);
        if (field_value is not null) then
        begin
          tmp_date = field_value;

          if (exists(select *
                     from clients
                     where c_id = :c_id and
                           c_birth_date is not null and
                           c_birth_date <> :tmp_date)) then
          begin
            res_id = 1;
            res_text = f_lang(lang, 'У клиента дата рождения другая. Возможно это чужой документ.');
            suspend;
            exit;
          end

          when any do
          begin
            res_id = 1;
            res_text = f_lang(lang, 'Неправильная дата рождения');
            suspend;
            exit;
          end
        end
      end
    end

    -- заполнение автовычисляемых полей, которые не видны пользователю
    execute procedure fm_docrecog_calc_field(:drl_id, 0);

    -- Проверка полей, на соответствие маске в т.ч. calc- полей, поэтому тут
    select res_id, res_text
    from fm_docrecog_check_mask(:drl_id, :lang)
    into res_id, res_text;
    if (res_id = 1) then
    begin
      suspend;
      exit;
    end

    if (action_kind = 'final_save') then
    begin
      update docrecoglist
      set drl_status = iif(drl_from_bls_1 is not null or drl_from_bls_2 is not null, 1, 0),
          drl_from_un_id = :un_id
      where drl_id = :drl_id
      returning drl_from_c_id
      into c_id;

      if (c_id is not null) then
      begin
        -- Сохранение в client_docs
        begin
          select cds_id, err_text
          from fm_docrecog_add_clientdoc(:drl_id, :un_id)
          into :cds_id, :res_text;
          when gdscode lock_conflict, gdscode update_conflict, gdscode deadlock do
            res_text = 'Клиент редактируется. Попробуйте немного позже.' || ' ID-транзакции: ' || current_transaction;
        end
        -- RFCRU-2770
        select res_text
        from hp_add_pledger_item(:un_id, :c_id, :drl_id)
        into :res_text;

        if (res_text = '') then
        begin
          res_data = '{' || f_json_make_pair('cds_id', cds_id, 'num') || ',' || f_json_make_pair('drl_id', drl_id, 'num') || '}';
        end
        else
        begin
          res_id = 1;
          suspend;
          exit;
        end
      end
      else
      begin
        -- создание клиента, в JSON передаем C_ID
        res_text = '';
        select c_id, res_text
        from fm_docrecog_add_client(:drl_id, 0, :un_id)
        into :c_id, :res_text;

        if (res_text = '') then
        begin
          if (c_id <> 0) then -- клиент создан
          begin
            res_data = '{' || f_json_make_pair('c_id', c_id, 'num') || ',' || f_json_make_pair('status', 'ok', 'str') || '}';
          end
          else
          begin
            res_data = '{"c_id":0, "status":"duplicate"}';
          end
        end
        else
        begin
          res_id = 1;
          res_data = '{' || f_json_make_pair('c_id', c_id, 'num') || ',' || f_json_make_pair('status', 'error', 'str') || '}';
        end
      end

      -- Если передан drr_id, подтверждаем запрос
      drr_id = f_json_str_get_value(doc_data, 'drr_id', null);
      if (drr_id is not null) then
      begin
        select res_id, res_text
        from fm_docrecogrequest_set_status(:drr_id, 2, :un_id)
        into res_id, res_text;
      end
    end
    suspend;
  end

  if (action_kind = 'draft_refuse') then
  begin
    -- Отказ оцифровщика МП. Отправка на доработку.
    rd_id = f_json_str_get_value(doc_data, 'rd_id', null);
    drr_id = f_json_str_get_value(doc_data, 'drr_id', null);

    select res_id, res_text
    from fm_docrecogrequest_set_status(:drr_id, -1, :un_id, :rd_id)
    into res_id, res_text;
    suspend;
  end

  if (action_kind = 'actionlist_update') then
  begin
    acl_id = f_json_str_get_value(doc_data, 'acl_id', null);
    acl_status = f_json_str_get_value(doc_data, 'acl_status', null);
    rd_id = f_json_str_get_value(doc_data, 'acl_result_rd_id', null);
    c_id = f_json_str_get_value(doc_data, 'c_id', null);
    res_text = case
                 when acl_id is null then f_format(f_lang(lang, 'Не найдено поле %1'), 'acl_id')
                 when acl_status is null then f_format(f_lang(lang, 'Не найдено поле %1'), 'acl_status')
                 when acl_status = 2 and coalesce(c_id, 0) = 0 then f_format(f_lang(lang, 'Не найдено поле %1'), 'c_id')
                 when acl_status = -2 and rd_id is null then f_lang(lang, 'При отказе не заполнено примечание')
                 else ''
               end;
    res_text = trim(res_text);
    if (res_text <> '') then
    begin
      res_id = 1;
      suspend;
      exit;
    end

    update actionlist
    set acl_status = :acl_status,
        acl_result_rd_id = iif(:acl_status = -2, :rd_id, null),
        acl_respons_c_id = iif(:acl_status = 0, null, acl_respons_c_id),
        acl_from_un_id = :un_id
    where acl_id = :acl_id and
          coalesce(acl_status, -777) <> :acl_status
    returning acl_object_id
    into drl_id;

    if (acl_status = 2) then
    begin
      -- Если подтверждается, и паспорт в mc_login прописываем клиента
      select drl_from_mcl_id, drl_is_main_doc
      from docrecoglist
      where drl_id = :drl_id
      into mcl_id, is_main;

      if (coalesce(mcl_id, 0) <> 0 and
          is_main = 1) then
      begin
        update mc_login
        set mcl_from_c_id = :c_id
        where mcl_id = :mcl_id and
              coalesce(mcl_from_c_id, 0) = 0;

        update docrecoglist
        set drl_from_c_id = :c_id
        where drl_id = :drl_id and
              coalesce(drl_from_c_id, 0) = 0;

        -- Новому клиенту прописываем номер телефона из mc_login,
        -- Если по маске проходит
        -- Маска для всех стран: '(79|992|996|998|994)[[:DIGIT:]]{9}'
        select mcl_login
        from mc_login
        where mcl_id = :mcl_id
        into field_value;
        if (field_value similar to '(79|996)[[:DIGIT:]]{9}') then
        begin
          execute procedure hp_continfo_update(:c_id, 'ContactInfo.Cell_Main',
              :field_value, 0, :un_id);
          when any do
          begin
            -- Если будут проблемы с номером (mc_login) оцифровщик их не решит,
            -- Добавляем № телефона как комментарий
            execute procedure hp_objectprops_memo_set(0, :c_id, 'Refdata',
                'ObjectProps.Client_memo',
                'Номер телефона клиента:' || field_value,
                :un_id);
          end
        end
      end
    end
    suspend;
  end

  if (action_kind = 'draft_delete') then
  begin
    -- Очистка, при отмене сохранения создаваемого документа
    -- Если были закреплены блобы, чистим сначала их
    select drl_from_bls_1, drl_from_bls_2
    from docrecoglist
    where drl_id = :drl_id
    into bls_1, bls_2;
    if (bls_1 is not null or (bls_2 is not null)) then
    begin
      delete from blobstore
      where bls_id in (:bls_1, :bls_2);
      when any do
      begin
        -- Может случиться если blx перелкючился, в этом случае оставляем как есть
      end
    end

    -- Затем удаляем сам документ
    delete from docrecoglist
    where drl_id = :drl_id and
          drl_status <= 0;
  end

  if (action_kind = 'add_clientdoc') then
  begin
    drl_id = f_json_blob_get_value(doc_data, 'drl_id', null);
    c_id = f_json_blob_get_value(doc_data, 'c_id', null);
    res_text = case
                 when c_id is null then f_format(f_lang(lang, 'Не найдено поле %1'), 'c_id')
                 when drl_id is null then f_format(f_lang(lang, 'Не найдено поле %1'), 'drl_id')
                 else ''
               end;
    res_text = trim(res_text);
    if (res_text <> '') then
    begin
      res_id = 1;
      suspend;
      exit;
    end

    update docrecoglist
    set drl_from_c_id = :c_id
    where drl_id = :drl_id and
          drl_from_c_id is null;

    select err_text
    from fm_docrecog_add_clientdoc(:drl_id, :un_id)
    into res_text;
    if (res_text <> '') then
      res_id = 1;
    suspend;
  end

  if (action_kind = 'client_passport_match') then
  begin
    c_id = (select mcl.mcl_from_c_id from mc_login mcl where mcl.mcl_id = :mcl_id);

    res_text =
      case
        when coalesce(mcl_id, 0) <= 0 then f_lang(lang, 'Не найдено поле ') || 'mcl_id'
        when coalesce(c_id, 0) = 0 then f_lang(lang, 'Клиент не найден')
        else ''
      end;

    if (coalesce(res_text, '') <> '') then
    begin
      res_id = 1;
      res_text = trim(res_text);
      suspend;
      exit;
    end

    -- Получаем номер документа, распознанный с приложенного для смены номера телефона паспорта
    select first 1 drv.drv_recog_value
    from docrecogvalue drv
    inner join docrecogfield drf on drv.drv_from_drf_id = drf.drf_id
    where drv.drv_from_drl_id = :drl_id
      and drv.drv_is_accepted = 1
      and drf.drf_code = 'doc_no'
    order by drv.drv_id
    into :doc_no_phone_change;

    if (coalesce(doc_no_phone_change, '') = '') then
    begin
      res_id = 1;
      res_text = f_lang(lang, 'Не удалось обнаружить номер документа, пожалуйста, переснимите паспорт и попробуйте снова');
      suspend;
      exit;
    end

    -- Ищем верифицированный паспорт клиента
    select cds.cds_doc_no
    from clientdocs cds
    where cds.cds_from_c_id = :c_id
      and cds.cds_is_main_doc = 1
      and cds.cds_is_delete = 0
      and cds.cds_verify_status = 2
    into :cli_passport_doc_no;

    if (coalesce(cli_passport_doc_no, '') = '') then
    begin
      res_id = 1;
      res_text = f_lang(lang, 'У вас нет верифицированного паспорта');
      suspend;
      exit;
    end

    select drm.drm_code
    from docrecoglist drl
    inner join docrecogmain drm on drl.drl_from_drm_id = drm.drm_id
    where drl.drl_id = :drl_id
    into :drm_code;

    -- Удаляем у кода документа .phonechange
    if (drm_code containing '.phonechange') then
        drm_code = substring(drm_code from 1 for position('.phonechange' in drm_code) - 1);

    -- Удаляем пробелы для корректного сравнения
    doc_no_phone_change = f_replace_all(doc_no_phone_change, ' ', '');
    cli_passport_doc_no = f_replace_all(cli_passport_doc_no, ' ', '');

    if (drm_code = 'kgz.id.type2' and doc_no_phone_change starting 'JD') then
        doc_no_phone_change = replace(doc_no_phone_change, 'JD', 'ID');

    if (doc_no_phone_change <> cli_passport_doc_no) then
    begin
      res_id = 1;
      res_text = f_lang(lang, 'Ваш паспорт не прошел проверку. Пожалуйста, используйте тот паспорт, с которым вы проходили верификацию');
    end

    suspend;
  end
end
