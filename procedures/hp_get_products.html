CREATE PROCEDURE HP_GET_PRODUCTS(
    REP_DATE D_DATE)
RETURNS (
    PR_ID D_INTEGER,
    PR_NAME D_STR_255)
AS
declare variable db_id d_integer;
declare variable pr_name_orig d_str_255; /* Оригинальное имя продукта (product.pr_name) */
declare variable pr_kontract_format d_str_255;
begin
    db_id = f_get_db_id();

    if (db_id = 41701) then -- Список продуктов для КР
    begin
        -- Берем все актуальные продукты (365)
        for select pr.pr_id, pr.pr_name, pr.pr_kontract_format
            from product pr
            where pr.pr_active = 1 and
                  :rep_date between pr.pr_date_begin and pr.pr_date_end and
                  pr.pr_code containing '.365.'
            union
            -- Берем все продукты-исключения, даже если они не подходят под верхние условия
            select pr.pr_id, pr.pr_name, pr.pr_kontract_format
            from product pr
            inner join mc_products_extra mpe on mpe.mpe_from_pr_id = pr.pr_id
            into :pr_id, :pr_name_orig, pr_kontract_format
        do
        begin
            pr_name = f_format('ID: %1, %2 (%3)', pr_id, pr_name_orig, pr_kontract_format);

            suspend;
        end
    end
    else -- Список продуктов для других стран
    begin
        for select pr.pr_id, pr.pr_name
            from product pr
            where pr_active = 1 and
                  :rep_date between pr.pr_date_begin and pr.pr_date_end
            union
            -- Берем все продукты-исключения, даже если они не подходят под верхние условия
            select mpe.mpe_from_pr_id, pr.pr_name
            from mc_products_extra mpe
            inner join product pr on mpe.mpe_from_pr_id = pr.pr_id
            into :pr_id, :pr_name_orig
        do
        begin
            pr_name = f_format('ID: %1, %2', :pr_id, :pr_name_orig);

            suspend;
        end

    end

end
