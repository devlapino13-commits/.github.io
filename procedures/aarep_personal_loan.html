CREATE PROCEDURE AAREP_PERSONAL_LOAN(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    AAREP_K_ID DECIMAL(18,0),
    AAREP_DATE_ACTION DATE,
    AAREP_PLAN_OS DECIMAL(18,4),
    AAREP_PLAN_PERC DECIMAL(18,4),
    AAREP_PLAN_ALL DECIMAL(18,4),
    AAREP_PLAN_REPAY_NO INTEGER,
    AAREP_PLAN_REPAY_DATE DATE,
    AAREP_REGION_ID INTEGER,
    AAREP_CURRENCY VARCHAR(3),
    AAREP_NEED_CLOSE INTEGER,
    AAREP_KOLICH_REPAY INTEGER,
    AAREP_PERSONAL_ID DECIMAL(18,0),
    AAREP_ORIGINAL_ID DECIMAL(18,0),
    AAREP_PERSONAL_FIO VARCHAR(100),
    AAREP_ORIGINAL_FIO VARCHAR(100),
    AAREP_ORDER_TEXT VARCHAR(255),
    AAREP_ORDER_DATE DATE,
    AAREP_ZP_REGION_ID INTEGER,
    AAREP_ZP_BRANCH_ID INTEGER,
    AAREP_MEMO VARCHAR(255),
    AAREP_KONTRACT_NO VARCHAR(20),
    AAREP_DISB_DATE DATE,
    AAREP_FROM_SAC_ID INTEGER,
    AAREP_ACC_BALANCE VARCHAR(20),
    AAREP_ACC_PERCENT VARCHAR(20),
    AAREP_ACC_HQ_BRANCH VARCHAR(20),
    AAREP_ACC_BRANCH_HQ VARCHAR(20),
    AAREP_ACC_SALARY VARCHAR(20))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */
  aarep_k_id = 0;
  aarep_plan_perc = 0;
  aarep_plan_os = 0;
  aarep_plan_perc = 0;
  aarep_plan_repay_no = 0;
  aarep_plan_all = 0;
  select sac_id
  from sa_config
  where sac_is_credit = 1 rows 1
  into :aarep_from_sac_id;
  if (aarep_from_sac_id is null) then
    aarep_from_sac_id = -1;

  for select distinct k_currency, ks_from_k_id, kal_new_reg_id, k_kolich_repay, kc_from_c_id, c_name_fio, k_kontract_no,
                      k_disb_date
      from kontstate, kontract, knaktlist, region_list(:report_region_id, 1, -1, 1), product, knclient, clients
      where ks_ol_is_delete = 0 and
            k_from_pr_id = pr_id and
            kc_amount_approved <> 0 and
            k_id = kc_from_k_id and
            kc_from_c_id = c_id and
            pr_is_employer = 1 and
            ((ks_date_end >= :report_date_begin and
            cast(ks_date_begin as date) <= :report_date_end and
            ks_state = 1) or
            (ks_date_begin = :report_date_begin and
            ks_state = 2)) and
            kal_from_kf_id = k_from_kf_id and
            ks_from_k_id = k_id and
            kal_date <= :report_date_end and
            kal_date_end >= :report_date_end and
            (kal_new_reg_id = id_list)
      into :aarep_currency, :aarep_k_id, :aarep_region_id, :aarep_kolich_repay, :aarep_personal_id, :aarep_personal_fio,
           :aarep_kontract_no, :aarep_disb_date
  do
  begin
    select c_original_id
    from clients
    where c_id = :aarep_personal_id
    into :aarep_original_id;
    if (aarep_original_id is null) then
      aarep_original_id = :aarep_personal_id;

    select c_name_fio
    from clients
    where c_id = :aarep_original_id
    into :aarep_original_fio;

    aarep_order_text = null;
    aarep_order_date = null;
    aarep_zp_region_id = null;
    aarep_zp_branch_id = null;
    aarep_memo = '';
    select ps_text, ps_date_begin, ps_from_reg_id, ps_branch_id
    from hr_personal_last_position(:aarep_original_id, :report_date_end)
    into :aarep_order_text, :aarep_order_date, :aarep_zp_region_id, :aarep_zp_branch_id;
    if (aarep_zp_branch_id is null) then
    begin
      aarep_memo = 'Приказы по сотруднику не указаны';
    end
    else
      aarep_memo = 'Погашение кредита за счет ЗП, договор № ' || :aarep_kontract_no;

    aarep_acc_balance = null;
    aarep_acc_percent = null;
    aarep_acc_hq_branch = null;
    aarep_acc_branch_hq = null;
    aarep_acc_salary = null;

    select acc_balance, acc_percent
    from hp_kontract_account_list(:aarep_k_id, :report_date_end)
    into :aarep_acc_balance, :aarep_acc_percent;

    if (aarep_zp_branch_id <> aarep_region_id) then
    begin
      select ln_account
      from loronostro
      where ln_region_loro = :aarep_region_id and
            ln_region_nostro = :aarep_zp_branch_id
      into :aarep_acc_hq_branch;

      select ln_account
      from loronostro
      where ln_region_loro = :aarep_zp_branch_id and
            ln_region_nostro = :aarep_region_id
      into :aarep_acc_branch_hq;
    end
    select al_id
    from acc_list, acctypelist
    where al_from_reg_id = :aarep_zp_branch_id and
          al_from_bg_id like atl_acc_mask || '%' and
          atl_from_at_id = 37
    order by al_id
    rows 1
    into :aarep_acc_salary;

    aarep_need_close = 0;
    for select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_from_k_id = :aarep_k_id and
              ok_type < 8 /* Кроме выдач */
              and
              ok_ol_date <= :report_date_end and
              ok_sh_date >= :report_date_begin and
              ok_sh_date <= :report_date_end and
              ok_ol_is_active = 1 and
              ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
              and
              ok_from_c_id = 0
        group by ok_sh_id, ok_sh_date
        having(sum(ok_os + ok_sh_os) <> 0) or (sum(ok_perc + ok_sh_perc) <> 0)
        order by ok_sh_date
        into :aarep_plan_repay_no, :aarep_plan_repay_date, :aarep_plan_os, :aarep_plan_perc
    do
    begin
      aarep_date_action = :aarep_plan_repay_date;
      if (aarep_plan_os is null) then
        aarep_plan_os = 0;
      if (aarep_plan_perc is null) then
        aarep_plan_perc = 0;
      aarep_plan_all = aarep_plan_os + aarep_plan_perc;
      if (aarep_plan_repay_no >= aarep_kolich_repay) then
        aarep_need_close = 1;
      suspend;
    end
  end
end
