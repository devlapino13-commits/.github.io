CREATE PROCEDURE FIND_HRORDERS_TO_APPROVE(
    REGION_ID INTEGER)
RETURNS (
    ORDER_ID DECIMAL(18,0),
    ORDER_DATE DATE,
    ORDER_REG_NAME VARCHAR(50),
    ORDER_TEXT VARCHAR(1024),
    ORDER_REASON VARCHAR(255),
    ORDER_TYPE_ID INTEGER,
    ORDER_NO VARCHAR(20),
    ORDER_REG_ID INTEGER,
    PS_STATUS INTEGER)
AS
begin
  for select distinct ho_id, ho_date, region.reg_name, ho_text, ho_reason, ho_order_type, ho_order_no, ho_from_reg_id,
                      ho_status
      from region_list(:region_id, 1, -1, 1), hr_orders, region
      where ho_from_reg_id = id_list and
            ho_from_reg_id = reg_unicode and
            ho_status = 0
      into :order_id, :order_date, :order_reg_name, :order_text, :order_reason, :order_type_id, :order_no,
           :order_reg_id, ps_status
  do
  begin
    suspend;
  end
end
