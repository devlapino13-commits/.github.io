Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXUNIQ_INSERT(
    NEW_UX_UUID D_UUID,
    UX_KIND D_STR_CODE,
    UX_DATETIME D_DATETIME = null)
RETURNS (
    UX_UUID D_UUID)
AS
begin
  ux_kind = uxf$kind(ux_kind, 'id');

  ux_uuid = new_ux_uuid;
  if (ux_uuid is null) then
  begin
    insert into ux$uniq (uu_kind, uu_datetime)
    values (:ux_kind, :ux_datetime)
    returning uu_id
    into :ux_uuid;
  end else
  begin
    if (not exists(select * from ux$uniq
    where uu_id = :ux_uuid)) then
    begin
      insert into ux$uniq (uu_id, uu_kind, uu_datetime)
      values (:ux_uuid, :ux_kind, :ux_datetime);
    end
  end

  suspend;
end
