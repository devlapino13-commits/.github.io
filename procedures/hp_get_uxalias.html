Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_GET_UXALIAS(
    UA_CODE D_STR_100)
RETURNS (
    UA_UX_CODE D_STR_CODE,
    UX_CODE D_STR_CODE,
    UX_KIND D_INTEGER,
    UXA_CODE D_STR_100,
    UXA_NAME D_STR_100,
    UA_NAME D_STR_100,
    UA_DISABLE_ON_EDIT D_BOOL,
    UA_READONLY D_BOOL,
    UA_SAFE_MODE D_BOOL,
    UXA_KIND D_SMALLINT,
    UA_USR_ID_FIELD D_STR_CODE)
AS
declare variable not_found d_bool;
begin
  -- Возвращает информацию об объекте через код действия (UxAction.Ua_Code)
  -- При тестировании возможно действия не созданы и не прикреплен меню
  -- Поэтому случай с fmMainmenu обрабатываем дополнительно

  if (trim(coalesce(ua_code, '')) = '') then
    Exit;

  not_found = 0;
  ua_usr_id_field = '';
  uxa_kind = 0;

  select ua_ux_code, ua_uxa_code, uxa_kind, uxa_ux_code, ux_kind, uxa_name, ua_name,
         ua_usr_id_field, ua_disable_on_edit, ua_readonly, ua_safe_mode
  from uxaction, uxalias, uxcode
  where ua_code = :ua_code and
        ua_uxa_code = uxa_code and
        uxa_ux_code = ux_code
  into ua_ux_code, uxa_code, uxa_kind, ux_code, ux_kind, uxa_name, ua_name,
       ua_usr_id_field, ua_disable_on_edit, ua_readonly, ua_safe_mode;

  if (row_count = 0) then
  begin
    if (ua_code starting 'fm_Mainmenu#') then
    begin
      ua_disable_on_edit = 1;
      ua_readonly = 0;
      ua_safe_mode = 0;

      ua_ux_code = 'fm_Mainmenu';
      uxa_code = replace(ua_code, 'fm_Mainmenu#', '');

      select uxa_ux_code, ux_kind, uxa_name
      from uxalias, uxcode
      where uxa_code = :uxa_code and
            uxa_ux_code = ux_code
      into ux_code, ux_kind, uxa_name;

      ua_name = 'Главное меню - ' || uxa_name;

      if (row_count = 0) then
      not_found = 1;
    end else
      not_found = 1;
  end

  if (not_found = 1) then
    exception e_default f_format('Unknown UxAction: %1', ua_code);

  suspend;
end
