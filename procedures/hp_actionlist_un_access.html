CREATE PROCEDURE HP_ACTIONLIST_UN_ACCESS(
    SERVICE_KIND STR_255,
    UN_ID ID_TYPE)
RETURNS (
    RD_ID INTEGER,
    RD_CODE STR_255,
    RD_NAME STR_255,
    RD_STR_ID STR_100,
    RD_MEMO STR_255)
AS
declare variable is_working_day d_smallint;
declare variable db_id          d_integer;
begin
  -- Процедура возвращает список доступных RD_ID пользователя

  -- Service_kind - имя службы для которой проверяется доступы
  -- Service_kind будет закрепляться в свойстве RD_Str_ID элемента ActionGroup
  -- Текущие варианты значений Service_kind:
  -- UnderWriter, Digitising

  -- в будущем могут быть CallCenter_Calling, LO_Calling, Visit_Monitoring, Visit_Inspection и т.д.
  if (coalesce(service_kind, '') = '') then
  begin
    suspend;
    exit;
  end

  db_id = f_get_db_id();
  is_working_day = 0;

  if (db_id in (39801, 41701)) then
  begin
    if (exists(select *
               from sa_calendays
               where scd_date = current_date and
                     scd_work_hours > 0)) then
      is_working_day = 1;
  end

  for select rd_id, rd_code, rd_name, rd_str_id, rd_memo
      from refdata
      where rd_main_code = 'ActionGroup' and
            rd_main_code <> rd_code and
            (:service_kind = 'Service_All' or
            rd_str_id = :service_kind) and
            rd_is_active = 1 and
            (rd_str_id starting 'SimpleVerify_' or -- Стандартное окно верификации данных
            rd_str_id in ('UnderWriter', --  Кредитный комитет
                      'UnderWriter_Special', --  Кредитный комитет по СПЕЦ. продуктам
                      'Digitising', --  Оцифровка документов
                      'Passport_Verify', --  Верификация данных
                      'Passport_Verify_OiVO', --  Верификация данных по Ойво
                      'DocRecog_Client', --  Оцифровка документа клиента
                      'DocRecog_Client_OiVO', --  Оцифровка документа клиента по Ойво
                      'Check_Info', --  Проверка информации
                      'Check_Info_Error', --  Проверка информации c ошибками
                      'ZalogAutoProf',
                      'UnderWriter_PostCheck', -- Пост-проверка | нерейтинговые
                      'UnderWriter_PostCheckR', -- Пост-проверка | рейтинговые
                      'Photo_Verify', -- Сверка фото
                      'Photo_Verify_Issue', -- Сверка фото на момент выдачи
                      'Manual_Inn_Check'  -- Проверка ИНН клиента
                      ))
            and
            rd_code <> 'ActionGroup.Check_Client_Data' -- Кроме уровня клиента
      into :rd_id, :rd_code, :rd_name, :rd_str_id, :rd_memo
  do
  begin
    /* Закомментировал igusev согласно задаче RFCKG-1517
    if (db_id in (39801, 41701) and
        rd_str_id = 'UnderWriter_PostCheck' and
        is_working_day = 1 and
        current_time between '10:30:00' and '16:30:00') then
      continue;
    */

    if (rd_str_id = 'SimpleVerify_Phone_Number_Verify' and f_get_const('check_contact_number') <> 1) then
        continue;

    if (exists(select has_access
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'ActionGroup.' || :rd_str_id)
               where has_access = 1)) then
      suspend;
  end
end
