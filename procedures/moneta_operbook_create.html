CREATE PROCEDURE MONETA_OPERBOOK_CREATE(
    ESM_ID ID_TYPE)
AS
declare variable esm_date date;
declare variable k_kontract_no str_100;
declare variable id_no id_type;
declare variable ol_id id_type;
declare variable amount curr_type;
declare variable ob_memo varchar(1024);
declare variable acc_dt varchar(20);
declare variable acc_kt varchar(20);
declare variable esa_message d_blob_text;
declare variable doc_no integer;
declare variable acc_dt_id d_id;
declare variable acc_kt_id d_id;
declare variable k_id d_id;
declare variable is_oivo d_bool;
declare variable hq_reg_id d_integer;
begin
  esa_message = '';
  hq_reg_id = f_hq_id();

  select ea.esa_message, em.esm_date, em.esm_from_k_id
  from extsysanswer ea
  inner join extsysmain em on em.esm_id = ea.esa_from_esm_id
  where ea.esa_from_esm_id = :esm_id and
        ea.esa_message starting 'answer:'
  into :esa_message, :esm_date, k_id;

  k_kontract_no = '';
  is_oivo = 0;
  if (k_id is not null) then
  begin
    select k_kontract_no, iif(pr_code containing '.oivo_pack.', 1, 0)
    from kontract, product
    where k_id = :k_id
      and k_from_pr_id = pr_id
    into k_kontract_no, is_oivo;
  end else
  begin
    select epl_object_id
    from epay_trans, epay_list
    where ept_from_esm_id = :esm_id and
          epl_id = ept_from_epl_id
    into k_kontract_no;
  end


  esa_message = substring(esa_message from 8); -- убираем "answer:"
  amount = - cast(f_json_str_get_value(esa_message, 'attrs.sourceamountfee', 0) as curr_type);

  if (amount <= 0) then
    exit;

  select kv
  from get_kv(:hq_reg_id, 5)
  into :id_no;

  select num
  from last_doc_number(:esm_date, 9, :hq_reg_id)
  into :doc_no;

  ol_id = :id_no;

  if (k_id is not null)
  then ob_memo = 'Ком. за выдачу займа ' || k_kontract_no || ' со счета ' || iif(is_oivo = 1, f_get_const('moneta_oivo_acc'), f_get_const('moneta_acc'));
  else ob_memo = 'Ком. за выплату по партнерской программе ' || k_kontract_no || ' со счета ' || f_get_const('moneta_acc');

  insert into operlist (ol_id, ol_doc_no, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status,
                        is_delete, ol_user_id)
  values (:ol_id, :doc_no, :hq_reg_id, 2, :esm_date, 9, 'Комиссия за выдачу через Монета.РУ', 0, 0, 1000000000000);

  select al_id
  from acc_list
  where al_from_reg_id = :hq_reg_id and
        al_account = '91-02/3'
  into acc_dt;

  -- Счет комиссии берем отдельно, не из acc_list т.к. на 91-02/3 много привязанных счетов ЕПС
  acc_dt_id = f_get_acc_id('71702810005310310101'); -- Комиссия по Монете.РУ

  -- 51-01/18 Монета
  -- 51-01/24 Oivo
  select al_id, al_from_acc_id
  from acc_list
  where al_from_reg_id = :hq_reg_id and
        al_account = iif(:is_oivo = 1, '51-01/24', '51-01/18')
  into :acc_kt, acc_kt_id;

  insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                        ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                        ob_from_sac_id)
  values (:ol_id, 1, '810', 1, null, :acc_dt, :amount, :amount, null, null, null, :acc_kt, null, :ob_memo, null);

  if (acc_dt_id is not null and acc_kt_id is not null) then
  begin
    execute procedure hp_do_kn_operbook_ins(:ol_id, null, :acc_dt_id, :acc_kt_id, null, null, :amount, 0, :amount,
        :esm_date, :hq_reg_id, 1, 'Комиссия за выдачу через Монета.РУ');
  end

  insert into extsysanswer (esa_from_esm_id, esa_message)
  values (:esm_id, 'ol_id = ' || :ol_id);
end
