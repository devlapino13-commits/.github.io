CREATE PROCEDURE HP_DO_KN_OPEREXTRA_OFF(
    REP_DATE D_DATE)
AS
declare variable k_id d_id;
declare variable rd_id d_integer;
declare variable off_amount d_currency;
declare variable rec_count d_integer;
declare variable ol_id d_id;
declare variable total_sum d_currency;
declare variable total_tax d_currency;
begin
  -- процедура списывает госпошлины по истечению срока давности
  -- список KC_ID займа предоставляет бухгалтерия
  -- После чего этот список загружается в таблицу Temp_KC_ID и запускается данная процедура.

  -- после списанич БУ, запустить SQL ниже чтобы списались НУ.

--insert into kn_operbook (kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
--                         kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id, kob_ol_is_active,
--                         kob_detail_id, kob_client_dt, kob_client_kt, kob_from_ol_id, kob_from_kc_id)
--select kob_currency, kob_curr_rate, 7009628, kob_acc_kt, 0, 0, kob_summa,
--       kob_summa, kob_ol_date, kob_ol_reg_id, kob_ol_is_active, 7433, kob_client_dt, kob_client_kt,
--       kob_from_ol_id, kob_from_kc_id
--from kn_operbook
--where kob_from_ol_id = 1010100023741617 and
--      kob_detail_id = 7432




  total_sum = 0;
  total_tax = 0;
  rec_count = 0;

  rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');
  select ol_id
  from operlist
  where ol_from_reg_id = 1010100 and
        ol_date = :rep_date and
        ol_oper_type = 2 and
        ol_oper_subtype = 11 and
        ol_memo = 'Получено судебное решение о безнадежности взыскания'
  rows 1
  into ol_id;

  if (ol_id is null) then
  begin
    insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_doc_no,
                          ol_doc_no_str, ol_memo, ol_parent_id, is_delete, ol_user_id)
    values (1010100, 2, 11, :rep_date, 9, 0, '', 'Получено судебное решение о безнадежности взыскания',
            null, 0, 1000000000001)
    returning ol_id
    into :ol_id;
  end

  update company set cm_loan_history_action = 2
  where 1 = 1;

  for select distinct kc_from_k_id
      from temp_kc_id, knclient kc
      where temp_kc_id.kc_id = kc.kc_id
      into k_id
  do
  begin
    off_amount = 0;
    select sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff)
    from operextra
    where ox_object_kind = 9 and
          ox_object_id = :k_id and
          ox_ol_is_active = 1 and
          ox_ol_date <= :rep_date
    into off_amount;

    if (off_amount is null) then
      off_amount = 0;

    if (off_amount = 0) then
      Continue;

    insert into operextra (ox_from_ol_id, ox_ol_date, ox_ol_is_active, ox_object_kind, ox_object_id, ox_from_rd_id,
                           ox_amount_plus, ox_amount_minus, ox_detail, ox_from_reg_id, ox_amount_writeoff)
    values (:ol_id, :rep_date, 1, 9, :k_id, :rd_id,
            0, 0, 'Списано по истечению срока давности', 1010100, :off_amount);

    total_sum = total_sum + off_amount;
    total_tax = total_tax + off_amount;

    execute procedure hp_do_kn_operbook_one_loan_date(:k_id, :rep_date, :rep_date, :ol_id);

    rec_count = rec_count + 1;
--    if (rec_count > 1000) then
--      Break;
  end

  update company set cm_loan_history_action = 0
  where 1 = 1;
end
