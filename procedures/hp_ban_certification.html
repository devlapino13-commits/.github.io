CREATE PROCEDURE HP_BAN_CERTIFICATION(
    UN_ID D_ID,
    REG_ID D_INTEGER,
    K_ID D_ID = null)
RETURNS (
    IS_BAN D_BOOL)
AS
declare variable db_id d_integer;
declare variable k_disb_date d_date;
begin
  db_id = f_get_db_id();
  is_ban = 0;

  if (db_id in (39801, 41701)) then
  begin
    -- KG
    if (k_id is null) then k_disb_date = current_date;
    else k_disb_date = (select k_disb_date from kontract where k_id = :k_id);
    if (exists(
      select *
      from username
        inner join hr_pstatus
          on ps_from_c_id = un_from_c_id
            and :k_disb_date between ps_date_begin and ps_date_end
            and ps_is_working = 1
        inner join joblist
          on jl_id = ps_from_jl_id
        inner join jobkind
          on jk_id = jl_kind
           and jk_system_code = 'JobSystem.Region_KM'
      where un_id = :un_id)) then
    begin
      -- Кредитный менеджер
      if (exists(
            select *
            from underwritingcertif usf
            where ucf_from_reg_id = :reg_id and
            current_date between coalesce(usf.ucf_date_begin, '01.01.1900') and
                                 coalesce(usf.ucf_date_end, '31.12.9999') and
            usf.ucf_system = f_enum_id('access_sytem', 'undervriter_certif') and
            ucf_active = 1))
      then is_ban = 1; -- Доступ к сертификации закрыт
    end
  end
  suspend;
end
