CREATE PROCEDURE HP_FPS_RESEND_KK(
    K_ID D_ID,
    UN_ID D_ID)
AS
declare variable acl_id d_id;
declare variable new_memo d_str_255;
declare variable rd_id d_integer;
begin
  -- Если выдача по СБП и было успешное подтверждение КК
  -- и требуется подтверждение ФИО через КК, то переотправляем на КК
  if (f_get_const('fps_manual_disb') = 1) then
    exit;

  select rd_id
  from hp_get_check_loan_data(:k_id)
  into :rd_id;

  --rd_id = f_get_rd_id('ActionGroup.Check_Loan_Data');
  for select acl_id
      from fps_loan_banks, fps_client_banks, knclient, kontract, actionlist
      where k_id = :k_id and
            flb_from_kc_id = kc_id and
            fbl_from_fcb_id = fcb_id and
            fcb_status = 1 and
            kc_from_k_id = k_id and
            k_disb_kind = 7 -- СБП
            and
            k_status = 0 and
            acl_from_rd_id = :rd_id and
            acl_status = 2 and
            acl_object_kind = 9 and
            acl_object_id = k_id
      into acl_id
  do
  begin
    update actionlist
    set acl_status = 0,
        acl_from_un_id = :un_id
    where acl_id = :acl_id;

    new_memo = 'Повторно отправлен на КК в связи с заменой данных по СБП.';
    execute procedure hp_objectprops_memo_set(9, k_id, -1, rd_id, new_memo, un_id);
  end
end
