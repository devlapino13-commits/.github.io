CREATE PROCEDURE HP_JSON_KONTRACT_DOCS(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_DATA D_BLOB_TEXT)
AS
declare variable C_ID D_ID;
declare variable CAPTION D_STR_100;
declare variable JSON_ITEM D_STR_LONG;
declare variable JSON_LIST D_BLOB_TEXT; /* d_str_8100; */
declare variable CAN_EXPORT D_SMALLINT;
declare variable CAN_EXPORT_PASS D_SMALLINT;
declare variable KL_DATE D_DATE;
declare variable CDS_ID D_ID;
declare variable MA_ID D_ID;
declare variable C_KIND D_SMALLINT;
declare variable BLS_ID D_ID;
declare variable EDB_ID D_ID;
declare variable PDT_CODE D_STR_CODE;
declare variable BLN_ID D_INTEGER;
declare variable KC_ID D_ID;
declare variable DOC_CAPTION D_STR_LONG;
declare variable PR_CODE D_STR_100;
declare variable CAPTION_ASP D_STR_100;
declare variable DB_ID D_INTEGER;
declare variable BLANK_FIRST_DATE date;
declare variable EXA_ID D_INTEGER;
declare variable ZAP_ID D_INTEGER;
declare variable STATUS D_INTEGER;
declare variable VWP_TYPE D_BOOL;
declare variable KC_STATUS D_SMALLINT;
declare variable MP_ID D_ID;
declare variable V_DELIMITER D_STR_10;
declare variable CAPTION_SPEC D_STR_LONG;
declare variable BY_RATING D_SMALLINT;
declare variable EDC_CODE D_STR_CODE;
declare variable EDL_STATUS D_SMALLINT;
declare variable K_DISB_DATE D_DATE;
declare variable K_DISB_KIND D_INTEGER;
declare variable USE_CARDST boolean;
declare variable PK_NAME D_STR_100;
declare variable CDS_DATETIME D_DATE;
declare variable CAPTION_CONSENT D_STR_100;
declare variable OKS_DELINQ_DATE D_DATE;
declare variable SELF_PROHIBITION_STATUS D_INTEGER;
declare variable SELF_PROHIBITION_QUERY_QTY D_INTEGER;
declare variable SELF_PROHIBITION_QUERY_DATE D_DATE;
declare variable SELF_PROHIBITION_DOC D_SMALLINT;
declare variable MAIN_K_ID D_ID;
declare variable MAIN_MA_ID D_INTEGER;
declare variable KIB_CHANGE_ACTUALITI_DATE D_DATE;
declare variable ca_id d_integer;
declare variable ca_date d_date;
begin
  -- Процедура возвращает документы для электронного досье по займу (КГ)
  -- Договор:
  -- Заявка
  -- Клиент1
  --   + Электронные документы клиента
  -- +Клиент2
  --   + Электронные документы клиента
  -- +Лицо дающее согласие1
  --   + Электронные документы лица

  kib_change_actualiti_date = f_get_const('kib_change_actualiti_date');

  db_id = f_get_db_id();
  res_data = '';
  v_delimiter = ''; --разделитель для непустого json

  -- Права пользователя выгружать и печатать документы
  select has_access
  from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.KontDocsExport')
  into can_export;

  -- Права пользователя выгружать и печатать паспорт
  select has_access
  from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.KontDocsPassExport')
  into can_export_pass;

  select max(kl.kl_datetime)
  from kontlog kl
  where kl.kl_from_k_id = :k_id and
        kl.kl_state = 1
  into :kl_date;

  select k_from_ma_app_id, pr_code, k_by_rating, k_disb_date, k_disb_kind
  from kontract kont
  inner join product pr on pr_id = k_from_pr_id
  where k_id = :k_id
  into ma_id, pr_code, by_rating, k_disb_date, k_disb_kind;
  use_cardst = (db_id in (39801, 41701)) and (k_disb_kind = 6) and (k_disb_date >= f_get_const('card_change_date'));

  --Предложение по займу для старых экспрессов
  if (db_id in (39801, 41701) and
      pr_code containing '.express_online.' and
      pr_code not containing '.sugarmill.' and
      pr_code not containing '.installment_goods.') then
  begin
    select exa.exa_id
    from expressapp exa
    where exa.exa_id > 0 and
          exa.exa_from_k_id = :k_id
    into exa_id;

    json_item = '';
    json_list = '';
    select edb_id, edf_print_name
    from edoc_list, edoc_cfg, edoc_blank, edoc_frx
    where edl_object_kind = f_objkind('expressapp', 'id') and
          edl_object_id = :exa_id and
          edl_status = 1 and
          edl_from_edc_id = edc_id and
          edc_code = 'loan.express' and
          edb_from_edl_id = edl_id and
          edb_from_edf_id = edf_id and
          edb_from_bls_id > 0
    order by edb_from_c_id
    into edb_id, caption;

    edb_id = coalesce(edb_id, 0);
    caption = '';

    json_item = f_image_item('edoc_blank', edb_id, caption, un_id, 0, can_export, can_export);
    json_list = json_list || iif(json_list <> '', ',', '') || json_item;

    if (json_list <> '') then
    begin
    res_data = BLOB_APPEND(res_data, v_delimiter ||
      '{' ||
      f_json_make_pair('caption', 'Предложение по займу' || f_objname('expressapp', :exa_id) , 'str') || ',' ||
      '"auto_load":0, "blobs":[' || json_list || ']' ||
      '}');
    v_delimiter = ',';
    end
  end

  if (db_id in (39801, 41701) and pr_code containing '.installment_goods.') then
  begin
    select mp.mp_id
    from expressapp exa
    inner join mc_purchase mp on mp.mp_from_exa_id = exa.exa_id
    where exa.exa_id > 0 and
          exa.exa_from_k_id = :k_id and
          exa.exa_from_exp_id = -2  --Пакет по товарам в рассрочку
    into mp_id;

    json_item = '';
    json_list = '';
    select edb_id, edf_print_name
    from edoc_list, edoc_cfg, edoc_blank, edoc_frx
    where edl_object_kind = f_objkind('mc_purchase', 'id') and
          edl_object_id = :mp_id and
          edl_status = 1 and
          edl_from_edc_id = edc_id and
          edc_code = 'mc_purchase.agreement_autosign' and
          edb_from_edl_id = edl_id and
          edb_from_edf_id = edf_id and
          edb_from_bls_id > 0
    order by edb_from_c_id
    into edb_id, caption;

    edb_id = coalesce(edb_id, 0);
    caption = '';

    json_item = f_image_item('edoc_blank', edb_id, caption, un_id, 0, can_export, can_export);
    json_list = json_list || iif(json_list <> '', ',', '') || json_item;

    if (json_list <> '') then
    begin
    res_data = BLOB_APPEND(res_data, v_delimiter ||
      '{' ||
      f_json_make_pair('caption', 'Расписка по товарам в рассрочку ' || f_objname('mc_purchase', :mp_id), 'str') || ',' ||
      '"auto_load":0, "blobs":[' || json_list || ']' ||
      '}');
    v_delimiter = ',';
    end
  end  

  -- Заявка
  json_item = '';
  json_list = '';
  for select edb_id, edf_print_name
  from edoc_list, edoc_cfg, edoc_blank, edoc_frx
  where edl_object_kind = 11
    and edl_object_id = :ma_id
    and edl_status = 1
    and edl_from_edc_id = edc_id
    and edc_code in ('loan.application', 'loan.application_autosign', 'loan.spouses_consent')
    and edb_from_edl_id = edl_id
    and edb_from_edf_id = edf_id
  order by edb_from_c_id
    into edb_id, caption
  do
  begin
    json_item = f_image_item('edoc_blank', edb_id, caption, un_id, 0, can_export, can_export);
    json_list = json_list || iif(json_list <> '', ',', '') || json_item;
  end
  if (json_list <> '') then
  begin
    res_data = BLOB_APPEND(res_data, v_delimiter ||
      '{' ||
      f_json_make_pair('caption', 'Заявка на займ №' || f_objname('miniapp', ma_id) , 'str') || ',' ||
      '"auto_load":0, "blobs":[' || json_list || ']' ||
      '}');
    v_delimiter = ',';
  end

  -- Договор
  json_item = '';
  json_list = '';
  for select edb_id, edf_print_name
  from edoc_list, edoc_cfg, edoc_blank, edoc_frx
  where edl_object_kind = 9
    and edl_object_id = :k_id
    and edl_status = 1
    and edl_from_edc_id = edc_id
    and edc_code in ('loan.agreement', 'loan.agreement_autosign', 'ipf.chdp.agreement')
    and edb_from_edl_id = edl_id
    and edb_from_edf_id = edf_id
    and edb_from_bls_id > 0
    into edb_id, caption
  do
  begin
    json_item = f_image_item('edoc_blank', edb_id, caption, un_id, 0, can_export, can_export);
    json_list = json_list || iif(json_list <> '', ',', '') || json_item;
  end

  -- Бланки для контракта.
  -- Добавляются в его вкладки
  -- 1635 Решение по заявке Классика
  -- 1636 Решение по заявке ИПФ
  -- 1730 Гарантийное письмо для сах. завода
  -- 1766 Решение по заявке Классика (выдачи по кредитному рейтингу)
  -- 1767 Решение по заявке ИПФ (выдачи по кредитному рейтингу)
  for select bln_id, bln_name, min(blb.blb_date_begin)
  from blankname bln
  inner join blankblob blb on bln.bln_id = blb.blb_from_bln_id
  where bln.bln_id in (1635, 1636, 1730, 1766, 1767)
  group by 1,2
  into bln_id, doc_caption, blank_first_date
  do
  begin

    if (blank_first_date > kl_date) then continue;

    if ((bln_id in (1635, 1766)) and (pr_code containing '.ipf.')) then continue;
    if ((bln_id in (1636, 1767)) and (pr_code not containing '.ipf.')) then continue;

    if (db_id in (39801, 41701)) then
    begin
      if (pr_code containing '.express_online.' and pr_code not containing '.sugarmill.' and pr_code not containing '.installment_goods.') then
        continue;

      if (pr_code not containing '.sugarmill.' and bln_id = 1730) then
        continue;

      if (coalesce(by_rating, 0) = 0 and bln_id in (1766, 1767)) then
        continue;

      if (coalesce(by_rating, 0) = 1 and bln_id in (1635, 1636)) then
        continue;
    end 

    -- Есть доп. объект "blank"
      json_item =
        f_format(
          '{"id_kind":"blankname","id":%1,"detail":"%2 %3","caption":"%2","file_name":"%4.pdf","can_save":%5,"can_print":%5,',
          bln_id, doc_caption, f_datetostr(:k_disb_date), k_id || bln_id, can_export) ||
       f_format(
          '"blank":{"obj_kind":9,"obj_id":%1,"reg_id":0,"date_begin":"%2","date_end":"%2","manager_id":0}}',
          k_id, f_datetostr(:k_disb_date));
    
      json_list = json_list || iif(json_list <> '', ',', '') || json_item;

  end

  res_data = BLOB_APPEND(res_data, v_delimiter ||
    '{' ||
    f_json_make_pair('caption', 'Договор №' || f_objname('kontract', k_id) , 'str') || ',' ||
    '"auto_load":0, "blobs":[' || json_list || ']' ||
    '}');

  v_delimiter = ',';

  -- RFCKZ-805
  -- Фильтр edb_from_bls_id > 0 не ставим: бланк попадает в blobstore только
  -- вторым проходом edoc_list_au0, до этого он лежит в edb_blob
  if (db_id = 39801 and
      f_get_const('kid_active') = 1) then
  begin
    json_item = '';
    json_list = '';
    for select edb_id, edf_print_name
    from edoc_list, edoc_cfg, edoc_blank, edoc_frx
    where edl_object_kind = 9
      and edl_object_id = :k_id
      and edl_status = 1
      and edl_from_edc_id = edc_id
      and edc_code = 'loan.kid'
      and edb_from_edl_id = edl_id
      and edb_from_edf_id = edf_id
    order by edb_from_c_id
      into edb_id, caption
    do
    begin
      json_item = f_image_item('edoc_blank', edb_id, caption, un_id, 0, can_export, can_export);
      json_list = json_list || iif(json_list <> '', ',', '') || json_item;
    end
    if (json_list <> '') then
    begin
      res_data = BLOB_APPEND(res_data, v_delimiter ||
        '{' ||
        f_json_make_pair('caption', 'КИД и проект договора', 'str') || ',' ||
        '"auto_load":0, "blobs":[' || json_list || ']' ||
        '}');
      v_delimiter = ',';
    end
  end

  -- RFCKZ-122
  -- Пакет с согласиями в МП
  -- 1001 - Согласие на сбор и обработку перс. данных
  -- 1002 - Согласие для ГКБ и ПКБ на получение перс. данных от АО ЦРТР
  -- 1003 - Согласие субъекта кредитной истории на выдачу кред. отчета получателю из КБ
  -- 1004 - Согласие субъекта кредитной истории на предоставление инфо о нем в КБ
  if (db_id = 39801 and
      exists(select * from edoc_list edl
             where edl.edl_object_kind = 11 and
                   edl.edl_object_id = :ma_id and
                   edl.edl_status = 1)) then
  begin
    json_list = '';
    for select edb.edb_id, bln.bln_id, bln.bln_name, min(blb.blb_date_begin), edb.edb_from_c_id
    from edoc_list edl
    inner join edoc_blank edb on edb.edb_from_edl_id = edl.edl_id
    inner join blankname bln on bln.bln_id = edb.edb_from_bln_id
    inner join blankblob blb on bln.bln_id = blb.blb_from_bln_id
    where edl.edl_object_kind = 11
      and edl.edl_object_id = :ma_id
      and edl.edl_status = 1
      and edb.edb_from_bln_id in (1001, 1002, 1003, 1004)
    group by edb.edb_id, bln.bln_id, bln.bln_name, edb.edb_from_c_id
    into edb_id, bln_id, caption, blank_first_date, c_id
    do
    begin
      if (blank_first_date > kl_date) then continue;
      json_item = f_format('{"caption": "%1", "blobs": [%2]}',
                           caption,
                           f_image_item('edoc_blank', edb_id, caption, un_id, 0, can_export, can_export));
      json_list = json_list || iif(json_list <> '', ',', '') || json_item;
    end
    if (json_list <> '') then
    begin
      res_data = BLOB_APPEND(res_data, v_delimiter ||
        '{' ||
        f_json_make_pair('caption', 'Согласия', 'str') || ',' ||
        '"auto_load":0, "items":[' || json_list || ']' ||
        '}');
      v_delimiter = ',';
    end
  end

  caption_spec = '';

  if (db_id in (39801, 41701)) then
  begin
    for select edb.edb_id, edf.edf_print_name
        from edoc_list edl
        inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id and
             edc.edc_code = 'loan.specification'
        inner join kontract_istinaa_prepay kip on kip.kip_id = edl.edl_object_id and
                                                  kip.kip_from_k_id = :k_id
        inner join edoc_frx edf on edf.edf_from_edc_id = edc.edc_id
        inner join edoc_blank edb on edb.edb_from_edl_id = edl.edl_id
        where edl.edl_object_kind = f_objkind('kontract_istinaa_prepay', 'id') and
              edl.edl_status = 1
        order by edl_create_datetime desc
        rows 1
      into
        :edb_id,
        :caption_spec
    do
    begin
      json_list = f_image_item('edoc_blank', edb_id, caption_spec, un_id, 0, can_export, can_export);

      res_data = blob_append(res_data, v_delimiter ||
        '{' ||
        f_json_make_pair('caption', 'Спецификация/Техническое задание' , 'str') || ',' ||
        '"auto_load":0, "blobs":[' || json_list || ']' ||
        '}');
    end

    for select edb.edb_id, edf.edf_print_name, edc_code, edl_status
        from edoc_list edl
        inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id and
             edc.edc_code in ('act_from_builder.agreement', 'act_from_company.agreement',
                              'idjara_goods.client_notif', 'idjara_goods.contragent_notif')
        inner join kontmatag kma on kma.kma_id = edl.edl_object_id and
                                    kma.kma_from_k_id = :k_id
        inner join edoc_frx edf on edf.edf_from_edc_id = edc.edc_id
        inner join edoc_blank edb on edb.edb_from_edl_id = edl.edl_id
        where edl.edl_object_kind = f_objkind('kontmatag', 'id')
        order by edl_create_datetime desc
      into
        :edb_id,
        :caption_spec,
        :edc_code,
        :edl_status
    do
    begin
      --Условие добавлено чтобы не ломать предыдущую логику
      if (edc_code not in ('idjara_goods.client_notif', 'idjara_goods.contragent_notif') and edl_status <> 1) then
        continue;

      json_list = f_image_item('edoc_blank', edb_id, caption_spec, un_id, 0, can_export, can_export);

      res_data = blob_append(res_data, v_delimiter ||
        '{' ||
        f_json_make_pair('caption', :caption_spec , 'str') || ',' ||
        '"auto_load":0, "blobs":[' || json_list || ']' ||
        '}');
    end

    if (:pr_code containing '.insur_base.' or :pr_code containing  '.insur_kontrakt.') then     -- RFCKG-4831
    begin
      select lir.lir_kontract_base
      from loan_insurance_relation lir
      where lir.lir_kontract_insurance = :k_id
      into main_k_id;

      if (coalesce(main_k_id, 0) <> 0) then
      begin
        for select eib.eib_from_bls_id
            from extinfolist eil
            inner join extinfolistbls eib on eib.eib_from_eil_id = eil.eil_id
            inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
            inner join insurance_kontract on ik_from_k_id = eil.eil_object_id
            where eil.eil_object_id = :main_k_id and
                  eil.eil_object_kind = f_objkind('kontract', 'id') and
                  eil.eil_is_json = 1 and
                  eil.eil_status = 2 and
                  eib.eib_from_bls_id > 0 and
                  eic.eic_code = 'policy_info' and
                  ik_enum_status = f_enum_id('insurance_kontract', 'insurance_policy_obtained')
            into :bls_id
        do
        begin
            json_item = f_image_item('blobstore', :bls_id, 'Страховой сертификат', un_id);
    
            res_data = BLOB_APPEND(res_data, v_delimiter ||
            '{' || f_json_make_pair('caption', 'Страховой сертификат', 'str') || ',' || '"auto_load":0, "blobs":[' || json_item || ']' || '}');
        end
      end
    end
    else
    begin
      -- RFCKG-4059
      for select eib.eib_from_bls_id
      from extinfolist eil
      inner join extinfolistbls eib on eib.eib_from_eil_id = eil.eil_id
      inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
      inner join insurance_kontract on ik_from_k_id = eil.eil_object_id
      where eil.eil_object_id = :k_id and
        eil.eil_object_kind = f_objkind('kontract', 'id') and
        eil.eil_is_json = 1 and
        eil.eil_status = 2 and
        eib.eib_from_bls_id > 0 and
        eic.eic_code = 'policy_info' and
        ik_enum_status = f_enum_id('insurance_kontract', 'insurance_policy_obtained') -- RFCKG-4659
      into :bls_id
      do
      begin
        json_item = f_image_item('blobstore', :bls_id, 'Страховой сертификат', un_id);

        res_data = BLOB_APPEND(res_data, v_delimiter ||
        '{' || f_json_make_pair('caption', 'Страховой сертификат', 'str') || ',' || '"auto_load":0, "blobs":[' || json_item || ']' || '}');
      end
    end
  end
  
  -- Профсуждения по автозаймам (РФ)
  if (db_id = 81001) then
  begin
    bln_id = 1650;
    json_list = '';
    for select zap_id, f_format('%1 от %2', zap_no, f_datetostr(zap_date))
    from zalogautoprof, zalogauto
    where za_from_k_id = :k_id
      and zap_from_za_id = za_id
      and zap_status = 2
    into zap_id, doc_caption
    do
    begin
        json_item =
          f_format(
            '{"caption":"%2","blobs":[{"id_kind":"blankname","id":%1,"detail":"%2","caption":"%2","file_name":"zalogauto.pdf","can_save":%3,"can_print":%3,"blank":{"obj_id":%4}}]}',
            bln_id, doc_caption, can_export, zap_id);
        json_list = json_list || iif(json_list <> '', ',', '') || json_item;
    end
    if (json_list <> '') then
    begin
      res_data = BLOB_APPEND(res_data, v_delimiter ||
        '{' ||
        f_json_make_pair('caption', 'Профсуждения по переоценке авто', 'str') || ',' ||
        '"auto_load":0, "items":[' || json_list || ']' ||
        '}');
      v_delimiter = ',';
    end
  end
