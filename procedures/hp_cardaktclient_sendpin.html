CREATE PROCEDURE HP_CARDAKTCLIENT_SENDPIN(
    CAC_ID D_INTEGER NOT NULL,
    CSC_CODE D_STR_100 NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    ERR_CODE D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable card_id d_integer;
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable cca_id d_integer;
declare variable cac_status d_integer;
declare variable date_prev d_datetime;
declare variable retry_period d_integer;
declare variable cell_no d_str_20;
declare variable cs_request_json d_str_8100;
declare variable obj_kind d_integer;
declare variable status_prev d_integer;
begin
  err_code = 0;
  err_text = '';
  obj_kind = f_objkind('cardaktclient', 'id');

  if (csc_code not in ('send_pin', 'resend_pin')) then
  begin
    err_code = 1;
    err_text = 'Недопустимый вх.параметр процедуры';
  end

  if (err_text = '') then
  begin
    select
      cac_card_id,
      cac_from_reg_id,
      cac_from_c_id,
      cac_from_k_id,
      cac_from_cca_id,
      cac_status
    from cardaktclient
    where cac_id = :cac_id
    into
      :card_id,
      :reg_id,
      :c_id,
      :k_id,
      :cca_id,
      :cac_status;

    if (row_count = 0) then
    begin
      err_code = 2;
      err_text = 'Акт выдачи карты №' || cac_id || ' не найден';
    end

    if (err_text = '' and cac_status < 4) then
    begin
      err_code = 3;
      err_text = f_format('Текущий статус акта выдачи карты №%1 не соответствует данной операции.', cac_id);
    end

    if (err_text = '' and csc_code = 'resend_pin') then
    begin
      if (exists(
        select *
        from cardsync
        where cs_object_id = :cac_id
          and cs_object_kind = :obj_kind
          and cs_answer_json is null )) then
      begin
        err_code = 4;
        err_text = 'Предыдущий запрос на переотправку PIN-кода еще не обработан.';
      end

      if (err_text = '') then
      begin
        retry_period = 10; -- разрешенный период повтора (мин.)
        select first 1
          cs_datetime,
          cs_status
        from cardsync
        where cs_card_id = :card_id
          and cs_from_csc_id in (f_get_cardsynccfg_id('send_pin'), f_get_cardsynccfg_id('resend_pin'))
        order by cs_id desc
        into
          :date_prev,
          :status_prev;

        if (status_prev = -1) then
        begin
          err_code = 5;
          err_text = 'Временная ошибка отправки PIN-кода. Запрос будет повторен.';
        end else
        begin
          if (date_prev is not null and datediff(minute, date_prev, current_timestamp) < retry_period ) then
          begin
            err_code = 6;
            err_text = f_format('Задачу на переотправку PIN-кода клиенту, можно инициировать раз в %1 мин.', retry_period);
          end
        end
      end
    end
  end

  if (err_text = '') then
  begin
    -- новая задача по отправке/переотправке PIN-кода карты клиенту
    cell_no = coalesce(f_cont_info('clients', :c_id, 'cell_main', 'ok'), '');
    if (cell_no <> '') then
    begin
      cs_request_json = f_format('{"card_id":%1, "phone":"%2"}', card_id, cell_no);
      insert into cardsync(
        cs_from_csc_id,
        cs_card_id,
        cs_from_cca_id,
        cs_object_kind,
        cs_object_id,
        cs_request_json,
        cs_office_id,
        cs_from_k_id,
        cs_from_c_id,
        cs_from_un_id
        )
      values(
        f_get_cardsynccfg_id(:csc_code),
        :card_id,
        :cca_id,
        :obj_kind,
        :cac_id,
        :cs_request_json,
        :reg_id,
        :k_id,
        :c_id,
        :un_id
        );
      if (csc_code = 'resend_pin') then
        err_text = 'Запрос на переотправку PIN-кода сгенерирован.';
    end else
    begin
      err_code = 7;
      err_text = 'Верифицированный номер телефона клиента не определен.';
    end
  end

  suspend;
end
