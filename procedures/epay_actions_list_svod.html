CREATE PROCEDURE EPAY_ACTIONS_LIST_SVOD
RETURNS (
    PROJECT D_STR_100,
    PR_TYPE D_STR_20,
    ACC D_STR_20,
    PAY_AMOUNT D_CURRENCY,
    ACC_AMOUNT D_CURRENCY)
AS
declare variable al_id d_str_20;
begin
  for
    select
      a.project,
      a.pr_type,
      a.acc,
      coalesce(b.pay_amount, 0)
    from (
      select
        rd_str_id as project,
        'ИПФ' as pr_type,
        case rd_str_id
          when 'elsom' then '10101/100'
          when 'mbank' then '10101/1104'
        end as acc
      from refdata
      where rd_main_code = 'EPayProjects'
        and rd_is_active = 1
        and rd_parent_id > 0
      union
      select
        rd_str_id as project,
        'Классика' as pr_type,
        case rd_str_id
          when 'elsom' then '10150/100'
          when 'mbank' then '10150/1105'
        end as acc
    from refdata
    where rd_main_code = 'EPayProjects'
      and rd_is_active = 1
      and rd_parent_id > 0
    ) as a
      left join (
        select
          project,
          pr_type,
          sum(kml_total) as pay_amount
        from epay_actions_list(1000000)
        group by project, pr_type) as b
     on a.project = b.project
        and a.pr_type = b.pr_type
  into
    :project,
    :pr_type,
    :acc,
    :pay_amount
  do
  begin
    al_id = (
      select
        al_id
      from acc_list
      where al_account = :acc
        and al_from_reg_id = 1090100
        and al_currency = '417');
    acc_amount = f_al_bal(al_id, current_date);
    suspend;
  end
end
