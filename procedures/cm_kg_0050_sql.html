CREATE PROCEDURE CM_KG_0050_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable PR_ID integer;
declare variable K_AMOUNT CURR_TYPE;
declare variable DB_ID integer;
declare variable MIN_CL_AMOUNT CURR_TYPE;
declare variable MAX_CL_AMOUNT CURR_TYPE;
declare variable PERC_YEAR PERC_TYPE;
declare variable KOMMIS_PERC PERC_TYPE;
declare variable KOL_CLI integer;
declare variable KOL_REPAY integer;
declare variable C_ID_LIST varchar(255);
declare variable PERC_MIN PERC_TYPE;
declare variable PERC_MAX PERC_TYPE;
declare variable PERC_DEFAULT PERC_TYPE;
declare variable KOMMIS_PERC_MIN PERC_TYPE;
declare variable KOMMIS_PERC_MAX PERC_TYPE;
declare variable KOMMIS_PERC_DEFAULT PERC_TYPE;
declare variable PR_CODE STR_100;
declare variable K_ADD_MONTH smallint;
declare variable K_REFUND smallint;
declare variable K_DISB_KIND D_INTEGER;
begin
  if (k_id = 109010327922900101) then exit; -- RFCKG-4029

  select k_from_pr_id, k_disb_date, k_amount_approved, k_percent, k_kommis_perc, k_kolich_client_approved,
         k_kolich_repay, pr_code, k.k_add_month, k.k_refund, k.k_disb_kind
  from kontract k
  inner join product p on p.pr_id = k.k_from_pr_id
  where k_id = :k_id and
        pr_id > 0
  into pr_id, kf_disb_date, k_amount, perc_year, kommis_perc, kol_cli, kol_repay, pr_code, k_add_month, k_refund, k_disb_kind;
  --  для КГ базы, логика мин макс суммы из продукта можно вытаскивать

  if (pr_code containing '.express_online.') then exit; -- express займ
  if (pr_code containing '.online_loan.') then exit;
  if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
    exit;

  -- #cmd_50_min_max ->  
  c_id_list = '';
  select min(kncl.kc_amount_approved), max(kncl.kc_amount_approved), list(kc_from_c_id, ';')
  from knclient kncl
  where kncl.kc_from_k_id = :k_id and
        kncl.kc_amount_approved > 0
  into :min_cl_amount, :max_cl_amount, :c_id_list;
  -- < --
  min_cl_amount = coalesce(min_cl_amount, -1);
  max_cl_amount = coalesce(max_cl_amount, -1);
  select perc_min, perc_max, perc_default, kommis_perc_min, kommis_perc_max, kommis_perc_default
  from hp_get_percent(:pr_id, :kf_disb_date, :kol_cli, :k_amount, :kol_repay, :min_cl_amount, :max_cl_amount, :c_id_list, :k_add_month, :k_refund, :k_disb_kind)
  into :perc_min, :perc_max, :perc_default, :kommis_perc_min, :kommis_perc_max, :kommis_perc_default;

  perc_year = round(coalesce(perc_year, -1), 2);
  perc_min = round(coalesce(perc_min, -1), 2);
  perc_max = round(coalesce(perc_max, -1), 2);
  perc_default = round(coalesce(perc_default, -1), 2);
  kommis_perc_min = coalesce(kommis_perc_min, -1);
  kommis_perc_max = coalesce(kommis_perc_max, -1);
  kommis_perc_default = coalesce(kommis_perc_default, -1);

  -- RFCKG-4254
  if (perc_default = 100.0) then
  begin
    res_error_text = 'Процент/наценка установлен/а некорректно. Пожалуйста, перевыберите Продукт.';
    res_error_is_critical = 2;
    suspend;
    exit;
  end

  if (perc_year not between perc_min and perc_max and
      round(perc_year, 1) <> round(perc_default, 1)) then
  begin
    res_error_text = 'Процент не правильно установлено, предварительно должно быть:' || round(perc_default, 2);
    res_error_is_critical = 2;
    suspend;
  end
  if (kommis_perc not between kommis_perc_min and kommis_perc_max and
      round(kommis_perc, 1) <> round(kommis_perc_default, 1)) then
  begin
    res_error_text = 'Доп. процент не правильно установлено, предварительно должно быть:' || round(kommis_perc_default, 2);
    res_error_is_critical = 2;
    suspend;
  end
  if (perc_default not between perc_min and perc_max --
      or --
      kommis_perc_default not between kommis_perc_min and kommis_perc_max) then
  begin
    res_error_text = 'Настройки продукта неправильные, обратитесь к тех.поддержке БД';
    res_error_is_critical = 2;
    suspend;
  end
end
