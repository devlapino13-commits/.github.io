CREATE PROCEDURE CLI_PASSPORT_ACTUAL_CHECK(
    C_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    AGE_CHECK INTEGER,
    PASS_DATE DATE,
    C_PASS_DATE_EXPIRE DATE,
    IS_PROBLEM SMALLINT)
AS
declare variable c_birth_date date;
declare variable V_EXPIRE D_DATE;
declare variable C_CITIZEN D_INTEGER;
begin
  /* На вход C_ID, REP_DATE (считаем что это дата выдачи)

На выходе
Поля AGE_CHECK возвращает 0 или 20 или 45,  (0 только в случае ошибки)
         PASS_DATE  -дата выдачи паспорта
         IS_PROBLEM (0 или 1) 0 – все ОК, 1 – проблема и должен сработать блок

1 запись процедурой должна вернуться в любом случае.
Если передели не корректный C_ID и clients не найден, возвращаешь AGE_CHECK = 0, PASS_DATE   = null, IS_PROBLEM = 1
 */
  select c_birth_date, c.c_pass_date
  from clients c
  where c_id = :c_id and
        c.c_citizen = 13 --Rossiya
  into c_birth_date, pass_date;

  if (c_birth_date is not null and
      pass_date <= rep_date) then
  begin
    is_problem = 0;
    -- должен заменить паспорт когда исполнится 45. Паспорт действительный до даты рождения
    c_pass_date_expire = dateadd(year, 45, c_birth_date);
    --< На основании письма от Светланы 27.05.2020  поставлено задачи на jira 1990, в связи с этим было произведено изменение
    if (rep_date between '01.02.2020' and '14.07.2020' and
        c_pass_date_expire between '01.02.2020' and '14.07.2020') then
      c_pass_date_expire = '15.07.2020'; -->

    --if (pass_date < c_pass_date_expire and
    --    c_pass_date_expire < rep_date) then

    if (pass_date < c_pass_date_expire) then
    begin
      v_expire = c_pass_date_expire;
      if (c_citizen = 13) then
        v_expire = dateadd(90 day to v_expire);
      if (v_expire < rep_date) then
        begin
          age_check = 45;
          is_problem = 1;
        end
    end 
    else
    begin
      c_pass_date_expire = dateadd(year, 20, c_birth_date);

      if (rep_date between '01.02.2020' and '14.07.2020' and
          c_pass_date_expire between '01.02.2020' and '14.07.2020') then
        c_pass_date_expire = '15.07.2020';

      if (pass_date < c_pass_date_expire and
          c_pass_date_expire < rep_date) then
      begin
        age_check = 20;
        is_problem = 1;
      end
    end
  end
  else
  begin
    age_check = 0;
    is_problem = 1;
  end

  suspend;
end
