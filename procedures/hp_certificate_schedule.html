CREATE PROCEDURE HP_CERTIFICATE_SCHEDULE(
    UN_ID DECIMAL(18,0),
    REP_REGION_ID INTEGER,
    REP_DATE DATE,
    IS_READONLY INTEGER)
RETURNS (
    RES_TIME TIMESTAMP,
    RES_TEXT VARCHAR(255),
    RES_STATUS VARCHAR(255),
    RES_CAN_RESERV INTEGER,
    RES_OBJ_ID DECIMAL(18,0))
AS
declare variable delta_minute integer;
declare variable res_time_end timestamp;
declare variable kf_name varchar(100);
declare variable ma_id integer;
declare variable has_record integer;
declare variable k_id decimal(18,0);
declare variable timezone integer;
declare variable curr_datetime timestamp;
declare variable db_id integer;
declare variable fact_datetime timestamp;
declare variable ma_is_active integer;
declare variable c_id decimal(18,0);
declare variable ma_date_save timestamp;
begin

-- Скрипт для заполнения случайное время, для теста
--update miniapp
--set ma_plan_disb_time=dateadd(minute, (7+mod(ma_id/10, 14))*60+mod(ma_id, 4)*15 ,cast('11.11.2016' as timestamp))
--where ma_date_app<=current_date
--and ma_date_end>=current_date
--and ma_date_oper='11.11.2016'
--and ma_office_id=1010800
--and ma_plan_disb_time is null

  res_obj_id = 0;
  res_text = '';
  res_can_reserv = 0;
  res_status = '';

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  if (c_id is null) then
  begin
    res_text = 'Сотрудник не определен';
    suspend;
    exit;
  end

--  if (exists(select * from hr_pstatus, joblist, jobkind
--      where ps_from_c_id = :c_id
--      and ps_date_begin <= current_date
--      and ps_date_end >= current_date
--      and ps_is_working = 1
--      and ps_status = 1
--      and ps_from_jl_id = jl_id
--      and jl_kind = jk_id
--      and jk_system_code in ('JobSystem.Region_KM')))
--  then is_manager = 1;

  if (rep_region_id is null) then
  begin
    res_text = 'Офис заявки не определен';
    suspend;
    exit;
  end

  delta_minute = 15;
  timezone = 0;

  :db_id = f_get_db_id();
  if (db_id = 81001) then
  begin
    delta_minute = 10;

    timezone = 3;
    select coalesce(reg_timezone, 0)
    from region
    where reg_unicode = :rep_region_id
    into :timezone;
    if (timezone <> 0) then
      timezone = timezone - 3;
  end
  curr_datetime = dateadd(hour, timezone, current_timestamp);

--  curr_plan_disb_time=current_date;
--  curr_plan_disb_time=dateadd(minute, 600, curr_plan_disb_time);

  res_time = rep_date;
  res_time = dateadd(minute, 420 - delta_minute, res_time); -- 6:45
  while (extract(hour from res_time) < 22) do
  begin
    res_text = '';
    res_can_reserv = 0;
    res_status = '';
    res_obj_id = 0;

    res_time = dateadd(minute, delta_minute, res_time);
    res_time_end = dateadd(minute, delta_minute, res_time);
    kf_name = '';
    ma_id = 0;
    ma_is_active = 1;
    fact_datetime = null;

    has_record = 0;
    ma_id = 0;
    for select kal_new_kf_name, ma_id, ma_is_active, ma_date_save
        from knaktlist, miniapp
        where ma_office_id = :rep_region_id and
              ma_date_oper = :rep_date and
              ma_date_end >= '31.12.2999' and
              ma_from_kf_id = kal_from_kf_id and
              ma_plan_disb_time >= :res_time and
              ma_plan_disb_time < :res_time_end and
              kal_date <= :rep_date and
              kal_date_end >= :rep_date and
              knaktlist.is_delete = 0
        into kf_name, ma_id, ma_is_active, ma_date_save
    do
    begin
      res_text = '';
      res_can_reserv = 0;
      res_status = '';
      res_obj_id = 0;
      fact_datetime = null;
      k_id = null;

      has_record = 1;
      res_text = kf_name;

      if (ma_is_active = 1) then
      begin
        res_status = 'Занято';
        select k_id
        from kontract
        where k_from_ma_app_id = :ma_id and
              k_disb_date = :rep_date rows 1
        into k_id;
        if (k_id is not null) then
        begin
          select ol_datetime
          from kontstate, operlist
          where ks_ol_is_delete = 0 and
                ks_from_k_id = :k_id and
                ks_state = 1 and
                ks_from_ol_id = ol_id
          into :fact_datetime;
          if (fact_datetime is not null) then
          begin
            fact_datetime = dateadd(hour, timezone, fact_datetime);
            res_status = 'Выдан ' || right('00' || extract(hour from fact_datetime), 2) || ':' || right('00' || extract(minute from fact_datetime), 2);

          end
        end
        if (res_time_end < curr_datetime and
            fact_datetime is null) then
        begin
          res_status = 'Истёк';
        end
      end
      else
      begin
        res_status = iif(ma_is_active = 0, 'Отменен ', 'Отказан ') || right('00' || extract(hour from ma_date_save), 2) || ':' || right('00' || extract(minute from ma_date_save), 2);
--        if (is_readonly=0) then has_record=0;
      end

      res_obj_id = ma_id;

      if (has_record = 1) then
        suspend;

-- Если время по графику еще не истек, и произведена выдача либо аннулирование заявки,
-- то даем возможность выбрать эту строку
      if (res_time_end > curr_datetime and
          is_readonly = 0) then
      begin
        if (fact_datetime is not null and
            fact_datetime < res_time) then
          has_record = 0;
        if (ma_is_active = 0) then
          has_record = 0;
      end
    end
    if (has_record = 0) then
    begin
      if (res_time_end > curr_datetime and
          is_readonly = 0) then
        res_can_reserv = 1;
      else
        res_can_reserv = 0;
      res_obj_id = 0;
      res_text = '';
      res_status = '';
      suspend;
    end
  end

  if (is_readonly = 0) then
    exit;

-- Список групп, без времени сертификации
  res_time = null;
  res_text = '';
  res_can_reserv = 0;
  res_obj_id = 0;

  for select kal_new_kf_name, ma_id, ma_is_active, ma_date_save
      from knaktlist, miniapp
      where ma_office_id = :rep_region_id and
            ma_date_oper = :rep_date and
            ma_date_end >= '31.12.2999' and
            ma_from_kf_id = kal_from_kf_id and
            ma_plan_disb_time is null and
            kal_date <= :rep_date and
            kal_date_end >= :rep_date and
            knaktlist.is_delete = 0
      into kf_name, ma_id, ma_is_active, ma_date_save
  do
  begin
    res_obj_id = ma_id;
    res_status = 'Не назначено';
    fact_datetime = null;
    k_id = null;
    res_text = kf_name;

    if (ma_is_active = 1) then
    begin
      select k_id
      from kontract
      where k_from_ma_app_id = :ma_id and
            k_disb_date = :rep_date rows 1
      into k_id;
      if (k_id is not null) then
      begin
        select ol_datetime
        from kontstate, operlist
        where ks_ol_is_delete = 0 and
              ks_from_k_id = :k_id and
              ks_state = 1 and
              ks_from_ol_id = ol_id
        into :fact_datetime;
        if (fact_datetime is not null) then
        begin
          fact_datetime = dateadd(hour, timezone, fact_datetime);
          res_status = 'Выдан ' || right('00' || extract(hour from fact_datetime), 2) || ':' || right('00' || extract(minute from fact_datetime), 2);

        end
      end
    end
    if (ma_is_active <= 0) then
    begin
      res_status = iif(ma_is_active = 0, 'Отменен ', 'Отказан ') || right('00' || extract(hour from ma_date_save), 2) || ':' || right('00' || extract(minute from ma_date_save), 2);
    end

    suspend;
  end

end
