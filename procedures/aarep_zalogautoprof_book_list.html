CREATE PROCEDURE AAREP_ZALOGAUTOPROF_BOOK_LIST(
    K_ID D_ID)
RETURNS (
    RES_KOB_ID D_INTEGER,
    RES_OL_ID D_ID,
    RES_ZALOG_ACC D_STR_CODE,
    RES_LINE_NO D_INTEGER,
    RES_OL_DATE D_DATE,
    RES_CORR_ACC D_STR_CODE,
    RES_DT D_CURRENCY,
    RES_KT D_CURRENCY,
    RES_SALDO D_CURRENCY,
    RES_OL_MEMO D_STR_LONG,
    RES_DETAIL D_STR_LONG)
AS
declare variable sab_id d_integer;
declare variable kc_id d_id;
declare variable acc_id d_id;
declare variable tmp_amount d_currency;
declare variable kob_acc_dt d_id;
declare variable kob_acc_kt d_id;
begin
  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Залог'
  into :sab_id;

  for select kc_id
  from knclient
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into kc_id
  do
  begin
    for select acc_id
    from ref_account
    where acc_from_sab_id = :sab_id and
          acc_object_kind = 8 and
          acc_object_id = :kc_id
    into acc_id
    do
    begin
      res_zalog_acc = f_fmt_acc(acc_id);
      res_line_no = 0;
      res_saldo = 0;

      for select kob_id, kob_ol_date, kob_from_ol_id, kob_summa, kob_acc_dt, kob_acc_kt, ol_memo, obd_detail
      from kn_operbook, operlist, kn_obdetail
      where kob_ol_is_active = 1 and
            (kob_acc_dt = :acc_id or kob_acc_kt = :acc_id) and
            kob_summa <> 0 and
            kob_from_ol_id = ol_id and
            kob_detail_id = obd_id
      order by kob_ol_date, kob_from_ol_id, kob_id
      into res_kob_id, res_ol_date, res_ol_id, tmp_amount, kob_acc_dt, kob_acc_kt, res_ol_memo, res_detail
      do
      begin
        res_line_no = res_line_no + 1;
        res_dt = 0;
        res_kt = 0;

        if (kob_acc_dt = acc_id) then
        begin
          res_corr_acc = f_fmt_acc(kob_acc_kt);
          res_dt = tmp_amount;
        end
        else
        begin
          res_corr_acc = f_fmt_acc(kob_acc_dt);
          res_kt = tmp_amount;
        end

        res_saldo = res_saldo + res_dt - res_kt;

        suspend;
      end
    end
  end
end
