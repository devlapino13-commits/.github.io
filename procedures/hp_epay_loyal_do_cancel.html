CREATE PROCEDURE HP_EPAY_LOYAL_DO_CANCEL(
    AG_ID D_INTEGER,
    EPD_ID D_INTEGER,
    REP_DATE D_DATE,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable epr_id d_integer;
declare variable c_id d_id;
begin
  res_text = '';
  res_is_error = 0;

  select epr_id
  from epay_project
  where epr_code = 'epay_loyal'
  into :epr_id;

  if (coalesce(epd_id, 0) > 0) then
  begin
    delete from epay_doc
    where epd_id = :epd_id;

    res_text = 'Подтверждение отменено';
    suspend;
    exit;
  end

  select ag_object_id
  from agreements
  where ag_id = :ag_id and
        ag_status = 1 and
        :rep_date between ag_date_begin and coalesce(ag_date_end, :rep_date)
  into c_id;

  if (c_id is null) then
  begin
    res_is_error = 1;
    res_text = 'Не найден клиент с действующим соглашением';
  end

  if (res_text = '') then
  begin
    insert into epay_doc (epd_from_epr_id, epd_from_c_id, epd_object_kind, epd_object_id, epd_date_end, epd_amount,
                          epd_status, epd_from_epl_id, epd_memo_id, epd_from_un_id)
    values (:epr_id, :c_id, f_objkind('agreements', 'id'), :ag_id, :rep_date, 0, -1, null, 0, :un_id);

    res_text = 'Отказ успешно установлен';
  end

  suspend;
end
