Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_UXFRAME_PARAMS(
    ITEM_CODE D_STR_100,
    ADD_LOOKUP_PARAMS D_BOOL = 0)
RETURNS (
    UFP_CODE D_STR_100,
    UFP_INDEX D_INTEGER,
    UFP_IN_USE D_BOOL,
    UFP_ORIGIN_NAME D_STR_CODE,
    UFP_NAME D_STR_CODE,
    UFP_PK_FIELD D_SMALLINT,
    UFP_REQUIRED D_BOOL,
    UFP_LOCKED D_SMALLINT,
    UFP_DEF_VALUE D_STR_255,
    UFD_SOURCE D_STR_CODE,
    UFP_IS_PARAM D_BOOL,
    UFP_IS_DELETE D_INTEGER,
    UFP_IN_UI D_SMALLINT,
    UFP_DB_DEF_VALUE D_STR_100,
    UFP_DB_REQUIRED D_BOOL,
    UFP_FK_TABLES D_STR_LONG,
    UFP_IS_FK D_BOOL,
    UFP_EDITOR_INFO D_STR_LONG,
    UFP_EDITOR D_STR_CODE,
    UFP_ACCESS_KIND D_SMALLINT,
    RES_CAPTION D_STR_100,
    RES_DATA_TYPE D_STR_CODE,
    RES_UCM_CODE D_STR_100,
    RES_UCM_IS_OWN D_BOOL,
    RES_BUTTONEDIT_UXA D_STR_100,
    RES_STATE D_SMALLINT,
    RES_FIELD_KIND D_SMALLINT,
    RES_INDEX_TXT D_STR_CODE,
    UEX_EXPRESSION D_STR_LONG,
    UEX_MESSAGE D_STR_255)
AS
declare variable item_no d_integer;
declare variable ufd_is_table d_blob_binary;
declare variable field_size d_str_code;
declare variable field_domain d_str_code;
declare variable tmp_editor_info d_str_long;
declare variable ucm_source d_str_code;
declare variable ucm_field d_str_code;
declare variable enter_sym d_str_lang;
declare variable ube_frame_text_field d_str_code;
declare variable ube_find_text_field d_str_code;
declare variable uxa_name d_str_100;
declare variable ux_table_name d_str_code;
declare variable res_editor_warning d_bool;
declare variable ufr_code d_str_code;
declare variable is_lookup d_bool;
declare variable parent_code d_str_100;
declare variable tmp_ufp_code d_str_100;
declare variable ube_find_id_field d_str_code;
declare variable orig_source d_str_code;
declare variable orig_index_txt d_str_code;
declare variable rec_id d_integer;
begin
  -- Список параметров для добавления в uxFrItem.
  if (coalesce(item_code, '') = '') then
    Exit;

  ufp_editor_info = '';
  enter_sym = ascii_char(13) || ascii_char(10);
  orig_source = '';

  is_lookup = null;
  if (exists(select * from uxframe
             where ufr_code = :item_code)) then
  begin
    -- Передан фрейм
    ufr_code = item_code;
    parent_code = null;

    select ufd_source, ufd_is_table
    from uxfrdata
    where ufd_ufr_code = :ufr_code and
          ufd_kind = 0
    into ufd_source, ufd_is_table;

    is_lookup = 0;
  end else
  begin
    -- Передан UFP_Code от Lookup
    select ufp_ufr_code, ulp_source, iif(ulp_table_name = ulp_source, 1, 0)
    from uxfrparam, uxfrlookup, uxlookup
    where ufp_code = :item_code and
          ufl_ufp_code = ufp_code and
          ufl_ulp_code = ulp_code
    into ufr_code, ufd_source, ufd_is_table;

    if (row_count > 0) then
     is_lookup = 1;

    parent_code = item_code;
  end

  if (is_lookup is null) then
    Exit;

  orig_source = ufd_source;

  for select ufp_code, ufp_in_use, ufp_is_param, ufp_index, ufp_origin_name, ufp_name,
             ufp_editor, ufp_pk_field, ufp_required, ufp_is_delete, ufp_state, ufp_access_kind
  from uxfrparam
  where ufp_ufr_code = :ufr_code and
        ufp_parent_code is not distinct from :parent_code
  order by ufp_is_param desc, ufp_index
  into :ufp_code, :ufp_in_use, :ufp_is_param, :ufp_index, :ufp_origin_name, :ufp_name,
       :ufp_editor, :ufp_pk_field, :ufp_required, :ufp_is_delete, :res_state, ufp_access_kind
  do
  begin
    select res_def_value, res_locked
    from hp_uxparam_final_value(:ufp_code, '', '')
    into :ufp_def_value, :ufp_locked;

    res_caption = ufp_origin_name;
    ufd_source = orig_source;
    res_index_txt = ufp_index;

    res_field_kind = 0;

    if (ufp_is_param = 1) then
    begin
      res_field_kind = -1;
      if (ufp_locked = 1) then
        res_field_kind = -2;
    end
    else
    begin
      res_field_kind = ufp_pk_field;
      if (ufp_locked = 1) then
      begin
        res_field_kind = -3;
        if (ufp_pk_field <> 0) then
          res_field_kind = -3 - ufp_pk_field;
       end
    end

    if (res_state = 0) then
      if (ufp_is_delete = 1) then
        res_state = -3; -- удаленное поле

    if (ufp_editor is null) then
      ufp_editor = '';

    res_ucm_code = '';
    res_ucm_is_own = 0;
    ufp_fk_tables = '';
    ufp_is_fk = 0;
    res_buttonedit_uxa = '';

    ufp_editor_info = '';
    tmp_editor_info = '';
    uxa_name = '';
    ube_frame_text_field = '';
    ube_find_text_field = '';
    ux_table_name = '';
    res_editor_warning = 0;

    ufp_in_ui = 0;

    ufp_db_def_value = '';
    ufp_db_required = 0;
    res_data_type = '';

    select 'Редактор: ' || editor_name || ', ' || editor_memo
    from hp_uxfritem_controls('editor')
    where editor_id = :ufp_editor
    into ufp_editor_info;

    if (ufp_editor in ('buttonedit', 'dialogedit')) then
    begin
      select ua_usr_text_field, ua_uxa_code, usr_text_field, usr_id_field, uxa_name, ux_table_name
      from uxfrparam, uxaction, uxsearch, uxalias, uxcode
      where ufp_code = :ufp_code and
            ufp_ufr_code = ua_ux_code and
            ufp_origin_name = ua_usr_id_field and
            ua_uxa_code = usr_code and
            usr_code = uxa_code and
            uxa_ux_code = ux_code
      into :ube_frame_text_field, :res_buttonedit_uxa, :ube_find_text_field, :ube_find_id_field, :uxa_name, :ux_table_name;

      if (ufp_editor = 'buttonedit') then
        ufp_editor_info = ufp_editor_info || enter_sym ||
                          f_format('Открывает окно: %1 (%2)', res_buttonedit_uxa, uxa_name);
      else
        ufp_editor_info = ufp_editor_info || enter_sym ||
                          f_format('Открывает диалоговое окно: %1 (%2)', res_buttonedit_uxa, uxa_name);

      ufp_editor_info = ufp_editor_info || enter_sym || f_format('Возвращает поле: %1', upper(ube_find_id_field));
      if (ube_find_text_field <> '') then
        ufp_editor_info = ufp_editor_info || ', поле текста: ' || upper(ube_find_text_field);

      if (ube_frame_text_field <> '') then
      begin
        ufp_editor_info = ufp_editor_info || enter_sym ||
                          'Текст вставляется в поле: ' || upper(ube_frame_text_field);
      end else
      begin
        tmp_editor_info = '';
        select ukn_sql
        from uxkeyname
        where ukn_table_name = :ux_table_name
        into tmp_editor_info;

        if (tmp_editor_info <> '') then
        begin
          ufp_editor_info = ufp_editor_info || enter_sym ||
                            'SQL получения текста: ' || tmp_editor_info;
        end else
        begin
          res_editor_warning = 1;
          ufp_editor_info = ufp_editor_info || enter_sym ||
                            f_format('ВНИМАНИЕ, в таблице %1 отсутствует SQL получения текста', upper(ux_table_name));
        end
      end
    end

    if (ufp_editor = 'lookup') then
    begin
      select ulp_name,
             f_format('select %1 as id, %2 as text from %3',
                       iif(trim(coalesce(ulp_id_field, '')) = '', '<НЕ УКАЗАН>', ulp_id_field),
                       iif(trim(coalesce(ulp_text_field, '')) = '', '<НЕ УКАЗАН>', ulp_text_field),
                       iif(trim(coalesce(ulp_source, '')) = '', '<НЕ УКАЗАН>', ulp_source)),
             iif(trim(coalesce(ulp_id_field, ''))= '' or
                 trim(coalesce(ulp_text_field, ''))= '' or
                 trim(coalesce(ulp_source, ''))= '', 1, 0)
      from uxfrlookup, uxlookup
      where ufl_ufp_code = :ufp_code and
            ufl_ulp_code = ulp_code
      into :uxa_name, :tmp_editor_info, res_editor_warning;

      if (row_count > 0) then
      begin
        ufp_editor_info = ufp_editor_info || enter_sym || f_format('Название: %1', uxa_name);
        ufp_editor_info = ufp_editor_info || enter_sym || f_format('SQL списка: %1', tmp_editor_info);

        if (res_editor_warning = 1) then
          ufp_editor_info = ufp_editor_info || enter_sym || f_format('ВНИМАНИЕ, в SQL не все поля заполнены.');
      end
      else
      begin
        res_editor_warning = 1;
        ufp_editor_info = ufp_editor_info || enter_sym || 'ВНИМАНИЕ, настройки не найдены.';
      end
    end

    if (ufp_editor = 'combobox') then
    begin
      select ufc_ucm_code, ucm_name, ucm_source, ucm_field
      from uxfrcombobox, uxcombomain
      where ufc_ufp_code = :ufp_code and
            ufc_ucm_code = ucm_code
      into res_ucm_code, tmp_editor_info, ucm_source, ucm_field;

      if (ucm_source = ufd_source and ucm_field = ufp_origin_name) then
        res_ucm_is_own = 1; -- Свой список, значит доступно для редактирования

      ufp_editor_info = ufp_editor_info || enter_sym ||
                        f_format('Название: %1, создан для %2', tmp_editor_info, upper(ucm_source || '.' || ucm_field));

      tmp_editor_info = '';
      select right(list(ucl_id || '-' || ucl_text, ', '), 1000)
      from uxcombolist
      where ucl_from_ucm_code = :res_ucm_code
      into tmp_editor_info;

      if (tmp_editor_info is null) then
        tmp_editor_info = '';

      if (tmp_editor_info <> '') then
      begin
        if (char_length(ufp_editor_info || tmp_editor_info) > 1000) then
          tmp_editor_info = left(tmp_editor_info, 1000 - char_length(ufp_editor_info));

        ufp_editor_info = ufp_editor_info || enter_sym || 'Значения: ' || tmp_editor_info;
      end else
      begin
        res_editor_warning = 1;
        ufp_editor_info = ufp_editor_info || enter_sym || 'ВНИМАНИЕ, значения не заполнены.';
      end
    end

    uex_expression = '';
    uex_message = '';

    if (exists(select * from uxfritem
    where ufi_ux_code = :ufr_code and
          ufi_ufp_code = :ufp_code)) then
      ufp_in_ui = 1;

    select uex_expression, uex_message
    from uxexpression
    where uex_ufp_code = :ufp_code
    into uex_expression, uex_message;

    if (uex_expression <> '') then
    begin
      if (ufp_editor_info <> '') then
        ufp_editor_info = ufp_editor_info || enter_sym;

      if (ufp_editor = 'mask') then
      begin
        ufp_editor_info = ufp_editor_info || 'Маска ввода: ' || uex_expression || enter_sym;
        ufp_editor_info = ufp_editor_info || 'Текст несоответствия: ' || uex_message;
      end else
      begin
        ufp_editor_info = ufp_editor_info || 'Логическое выражение: ' || uex_expression || enter_sym;
        ufp_editor_info = ufp_editor_info || 'Текст несоответствия: ' || uex_message;
        ufp_in_ui = 2;
      end
    end

    select list(trim(fk_table || iif(fk_is_virtual = 0, ' (fk)', '')), ', ')
    from hp_uxfrparam_fk_list(:ufd_source, :ufp_origin_name)
    into ufp_fk_tables;

    if (ufp_fk_tables is null) then
      ufp_fk_tables = '';

    if (ufp_fk_tables <> '') then
      ufp_is_fk = 1;

    select field_type, field_size, field_default, field_required, field_domain
    from hp_get_field_type(:ufd_source, :ufp_origin_name, :ufd_is_table)
    into res_data_type, field_size, ufp_db_def_value, ufp_db_required, field_domain;

    if (ufp_db_def_value is null) then
      ufp_db_def_value = '';

    if (field_domain not starting 'rdb$') then
      res_data_type = field_domain;
    else
      if (res_data_type = 'string') then
        res_data_type = res_data_type || '['  || field_size || ']';

    if (ufp_is_param = 1) then
      res_caption = res_caption || '   (параметр)';

    if (res_editor_warning = 1) then
      ufp_in_ui = -1;

    item_no = item_no + 1;
    suspend;

    if (ufp_editor = 'lookup' and add_lookup_params = 1) then
    begin
      uex_expression = '';
      uex_message = '';

      orig_index_txt = res_index_txt;
      tmp_ufp_code = ufp_code;
      rec_id = 0;
      ufp_access_kind = 0;
      for select ufp_code, ufp_index, ufp_in_use, ufp_origin_name, ufp_name,
                 ufp_pk_field, ufp_required, ufp_locked, ufp_def_value, ufd_source, ufp_is_param,
                 ufp_is_delete, ufp_in_ui, ufp_db_def_value, ufp_db_required, ufp_fk_tables,
                 ufp_is_fk, ufp_editor_info, ufp_editor, res_caption, res_data_type,
                 res_ucm_code, res_ucm_is_own, res_buttonedit_uxa, res_state, res_field_kind
          from hp_uxframe_params(:tmp_ufp_code, 0)
          into :ufp_code, :ufp_index, :ufp_in_use, :ufp_origin_name, :ufp_name,
               :ufp_pk_field, :ufp_required, :ufp_locked, :ufp_def_value, :ufd_source, :ufp_is_param,
               :ufp_is_delete, :ufp_in_ui, :ufp_db_def_value, :ufp_db_required, :ufp_fk_tables,
               :ufp_is_fk, :ufp_editor_info, :ufp_editor, :res_caption, :res_data_type,
               :res_ucm_code, :res_ucm_is_own, :res_buttonedit_uxa, :res_state, res_field_kind
      do
      begin
        res_caption = '   ' || res_caption;
        res_index_txt = orig_index_txt || '-' || rec_id;
        suspend;
        rec_id = rec_id + 1;
      end
    end
  end
end
