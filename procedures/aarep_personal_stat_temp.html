CREATE PROCEDURE AAREP_PERSONAL_STAT_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    REG_CODE_NAME VARCHAR(50),
    JL_CATEGORY VARCHAR(50),
    JL_IS_FRONT_TEXT VARCHAR(50),
    C_SEX_TEXT VARCHAR(50),
    JL_IS_KRED_PERSON VARCHAR(50),
    KOL_ROTATION_MINUS INTEGER,
    KOL_ROTATION_PLUS INTEGER,
    KOL_GONE INTEGER,
    KOL_NEW INTEGER,
    KOL_BEGIN INTEGER,
    KOL_END INTEGER,
    JL_NAME VARCHAR(100))
AS
declare variable jk_system_code varchar(100);
declare variable jl_is_front integer;
declare variable c_sex smallint;
declare variable ho_order_type integer;
declare variable ps_date_begin date;
declare variable ps_date_end date;
declare variable is_ok smallint;
declare variable jl_name_temp varchar(100);
begin
  kol_begin = 0;
  kol_new = 0;
  kol_gone = 0;
  kol_end = 0;
  kol_rotation_plus = 0;
  kol_rotation_minus = 0;
  jl_name_temp = '';

  for select r1.reg_code || ' ' || r1.reg_name, jl_category, jk_system_code, ho_order_type, ps_date_begin, ps_date_end,
             jl_name, c_sex
      from hr_pstatus, hr_orders, region_list(:report_region_id, 1, -1, 1), joblist, region r1, clients, jobkind
      where ho_id = ps_from_ho_id and
            (ps_from_reg_id = id_list) and
            ps_date_begin < :report_date_begin and
            ps_date_end >= :report_date_begin - 1 -- igusev было до 26.09.2014 ps_date_end>:report_date_begin
            and
            ps_is_working = 1 and
            ps_from_reg_id = r1.reg_unicode and
            ps_from_jl_id = jl_id and
            c_id = ps_from_c_id and
            jl_kind = jk_id
      order by c_id, ps_date_begin
      into :reg_code_name, :jl_category, :jk_system_code, :ho_order_type, :ps_date_begin, :ps_date_end, :jl_name,
           :c_sex
  do
  begin
-- igusev 26.09.14 добавил на основании письма о проблеме от Ирины Мехтиевоы от 24.09.14 -- >>
    is_ok = 0;

    if (ps_date_end > report_date_begin) then
      is_ok = 1;
    else
    if ((ps_date_end = report_date_begin - 1) and
        (ho_order_type = 3) and
        (jl_name <> jl_name_temp)) then
      is_ok = 1;

    jl_name_temp = jl_name;
    if (is_ok = 1) then
    begin
-- << -----------------------------------
      if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.Region_KM', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
        jl_is_kred_person = 'Кредит.персонал';
      else
        jl_is_kred_person = 'Админ.персонал';

      if (jl_is_front = 1) then
        jl_is_front_text = 'Фронт офис';
      else
        jl_is_front_text = 'Бэк офис';

      if (c_sex = 1) then
        c_sex_text = 'Мужчина';
      else
        c_sex_text = 'Женщина';

      kol_begin = 1;
      suspend;
    end
  end

  for select r1.reg_code || ' ' || r1.reg_name, jl_category, ho_order_type, ps_date_begin, ps_date_end, jl_name, c_sex,
             jk_system_code
      from hr_pstatus, hr_orders, region_list(:report_region_id, 1, -1, 1), joblist, region r1, clients, jobkind
      where ho_id = ps_from_ho_id and
            (ps_from_reg_id = id_list) and
            ps_date_begin >= :report_date_begin and
            ps_date_begin <= :report_date_end and
            ps_from_reg_id = r1.reg_unicode and
            ps_from_jl_id = jl_id and
            ho_order_type <> 4 and
            ho_order_type <> 5 and
            c_id = ps_from_c_id and
            jl_kind = jk_id
      into :reg_code_name, :jl_category, :ho_order_type, :ps_date_begin, :ps_date_end, :jl_name, :c_sex,
           :jk_system_code
  do
  begin
    if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.Region_KM', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
      jl_is_kred_person = 'Кредит.персонал';
    else
      jl_is_kred_person = 'Админ.персонал';

    if (jl_is_front = 1) then
      jl_is_front_text = 'Фронт офис';
    else
      jl_is_front_text = 'Бэк офис';

    if (c_sex = 1) then
      c_sex_text = 'Мужчина';
    else
      c_sex_text = 'Женщина';

    kol_begin = 0;
    kol_new = 0;
    kol_gone = 0;
    kol_rotation_plus = 0;
    kol_rotation_minus = 0;

    if (ho_order_type = 1) then
      kol_new = 1;/* Найм нового */
    if (ho_order_type = 2) then
      kol_rotation_plus = 0;/* Приход из другого офиса */
    if (ho_order_type = 3) then
      kol_rotation_plus = 0;/* Приход изменение статуса */
    if (ho_order_type = 6) then
      kol_gone = 1;/* Увольнение */
    suspend;
  end
  kol_begin = 0;
  kol_new = 0;
  kol_gone = 0;
  kol_rotation_plus = 0;
  kol_rotation_minus = 0;

  for select r1.reg_code || ' ' || r1.reg_name, jl_category, ho_order_type, ps_date_begin, ps_date_end, jl_name, c_sex,
             jk_system_code
      from hr_pstatus, hr_orders, region_list(:report_region_id, 1, -1, 1), joblist, region r1, clients, jobkind
      where ho_id = ps_from_ho_id and
            (ps_from_reg_id = id_list) and
            ps_date_begin <= :report_date_end and
            ps_date_end >= :report_date_end and
            ps_is_working = 1 and
            ps_from_reg_id = r1.reg_unicode and
            ps_from_jl_id = jl_id and
            c_id = ps_from_c_id and
            jk_id = jl_kind -- данной привязки не было
      into :reg_code_name, :jl_category, :ho_order_type, :ps_date_begin, :ps_date_end, :jl_name, :c_sex,
           :jk_system_code
  do
  begin
    if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.Region_KM', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
      jl_is_kred_person = 'Кредит.персонал';
    else
      jl_is_kred_person = 'Админ.персонал';

    if (jl_is_front = 1) then
      jl_is_front_text = 'Фронт офис';
    else
      jl_is_front_text = 'Бэк офис';

    if (c_sex = 1) then
      c_sex_text = 'Мужчина';
    else
      c_sex_text = 'Женщина';

    kol_end = 1;
    suspend;
  end
end
