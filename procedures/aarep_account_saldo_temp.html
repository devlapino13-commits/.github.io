CREATE PROCEDURE AAREP_ACCOUNT_SALDO_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    ACCOUNT VARCHAR(20),
    CURRENCY VARCHAR(3),
    SUMMA_VAL DECIMAL(18,4),
    SUMMA_EKV DECIMAL(18,4),
    MAX_DATE DATE)
AS
begin
-- передает остаток по лицевым счетам
-- Sanjari, 25.08.2013
  for select al_id, al_currency
      from acc_list, region_list(:report_region_id, 1, 3, 1)
      where al_from_reg_id = id_list
      into :account, :currency
  do
  begin
    summa_ekv = f_al_bal(account, report_date_end);
    summa_val = f_al_bal(account, report_date_end, 'val');

    select obs_date
    from obsummary
    where obs_from_al_id = :account and
          obs_date <= :report_date_end
    order by obs_from_al_id desc, obs_date desc
    rows 1
    into :max_date;

    if ((summa_val <> 0) or (summa_ekv <> 0)) then
      suspend;
  end
end
