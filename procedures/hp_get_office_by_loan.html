CREATE PROCEDURE HP_GET_OFFICE_BY_LOAN(
    KONT_NO VARCHAR(20),
    KF_NAME VARCHAR(20),
    CL_FIO VARCHAR(100))
RETURNS (
    REG_ID DECIMAL(18,0))
AS
declare variable kf_id decimal(18,0);
declare variable c_id decimal(18,0);
declare variable cl_fio_id decimal(18,0);
begin
  if (coalesce(:kont_no, '') <> '') then
  begin
    select k_last_reg_id
    from kontract
    where k_kontract_no starting :kont_no rows 1
    into :reg_id;
    if (reg_id is not null) then
      suspend;
  end
  else
  begin
    if (coalesce(:kf_name, '') <> '') then
    begin
      select kal_new_reg_id
      from knaktlist
      where kal_new_kf_name starting :kf_name and
            is_delete = 0 and
            kal_date_end >= '31.12.2999' rows 1
      into :reg_id;
      if (reg_id is null) then
      begin
        select kal_from_kf_id
        from knaktlist
        where kal_new_kf_name = :kf_name and
              is_delete = 0 rows 1
        into :kf_id;
        if (kf_id is not null) then
        begin
          select kal_new_reg_id
          from knaktlist
          where kal_from_kf_id = :kf_id and
                is_delete = 0 and
                kal_date_end >= '31.12.2999' rows 1
          into :reg_id;
        end
      end
      if (reg_id is not null) then
        suspend;
    end
    else
    begin
      if (coalesce(cl_fio, '') <> '') then
      begin
        cl_fio_id = -777;
        if (cl_fio similar to '[-]?[[:DIGIT:]]+') then
          cl_fio_id = cl_fio;

        for select c_id
            from clients
            where (c_name_fio starting :cl_fio or
                  c_id = :cl_fio_id)
            into :c_id
        do
        begin
          select oks_region_id
          from oksummary, knclient, kontract
          where oks_from_k_id = k_id and
                kc_from_k_id = k_id and
                kc_amount_approved > 0 and
                oks_date_end >= current_date and
                oks_date_begin <= current_date and
                kc_from_c_id = :c_id and
                oks_close_state = 0 
          order by k_disb_date desc
          rows 1
          into :reg_id;
          if (reg_id is not null) then
            suspend;
        end
      end
    end
  end
end
