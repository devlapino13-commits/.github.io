CREATE PROCEDURE HP_SIGNMEMBER_CHOICE(
    KIND D_SMALLINT)
RETURNS (
    ID D_ID,
    CODE D_STR_10,
    NAME D_STR_100)
AS
begin
  if (kind = 0) then -- сотрудник
  begin
    for
      select
        c_id,
        c_name_fio
      from clients
        inner join hr_pstatus
          on ps_from_c_id = c_id
            and current_date between ps_date_begin and ps_date_end
            and ps_status = 1
            and ps_is_working = 1
      order by c_name_fio
    into
      :id,
      :name
    do
    begin
      suspend;
    end
    exit;
  end

  if (kind = 1) then -- должность
  begin
    for
      select
        jl_id,
        jl_name
      from joblist
      order by jl_name
    into
      :id,
      :name
    do
    begin
      suspend;
    end
    exit;
  end

  if (kind = 2) then -- отдел
  begin
    for
      select
        dep_id,
        dep_name
      from depart
      order by dep_name
    into
      :id,
      :name
    do
    begin
      suspend;
    end
    exit;
  end

  if (kind = 3) then -- region
  begin
    for
      select
        reg_unicode,
        reg_code,
        reg_name
      from region
      where is_delete = 0
        and reg_name <> ''
      order by reg_name
    into
      :id,
      :code,
      :name
    do
    begin
      suspend;
    end
    exit;
  end
end
