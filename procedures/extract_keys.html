CREATE PROCEDURE EXTRACT_KEYS(
    BLANK_LIST D_STR_LONG)
RETURNS (
    KEYS D_STR_255)
AS
declare variable PART D_STR_LONG;
declare variable POS D_INTEGER;
declare variable KEY_ONLY D_STR_20;
declare variable COLON_POS D_INTEGER;
begin
    keys = '';

    /*
        -- случай 1 --
        blank_list = '1:abc, 2:qwe, 3:xyz'
        keys = '1, 2, 3'

        -- случай 2 --
        blank_list = '1, 2, 3'
        keys = '1, 2, 3'

        -- случай 3 --
        blank_list = '1'
        keys = '1'
    */

    while (blank_list is not null and char_length(blank_list) > 0) do
    begin
        -- ищем запятую
        pos = position(',' in blank_list);

        if (pos > 0) then
        begin
            part = substring(blank_list from 1 for pos - 1);
            blank_list = substring(blank_list from pos + 1);
        end
        else
        begin
            part = blank_list;
            blank_list = null;
        end

        -- ищем двоеточие
        colon_pos = position(':' in part);

        if (colon_pos > 0) then
            key_only = substring(part from 1 for colon_pos - 1);
        else
            key_only = trim(part); -- если двоеточия нет, берём всю часть

        -- добавляем в результат
        if (keys = '') then
            keys = key_only;
        else
            keys = keys || ',' || key_only;
    end

    suspend;
end
