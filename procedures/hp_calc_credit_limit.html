CREATE PROCEDURE HP_CALC_CREDIT_LIMIT(
    C_ID D_ID,
    CALC_TYPE D_STR_100)
RETURNS (
    RES_TEXT D_STR_100,
    CREDIT_LIMIT D_INTEGER)
AS
declare variable kbm_id d_id;
declare variable age d_integer;
declare variable gender d_bool;
declare variable family_status d_bool;
declare variable kib_mbf_delinq d_integer;
declare variable max_mbf_kh_deliqdayscount d_integer;
declare variable act_mbf_non_delinq d_integer;
declare variable kib_nombf_non_delinq d_integer;
declare variable kib_nombf_delinq d_integer;
declare variable act_nombf_non_delinq d_integer;
declare variable const d_float;
declare variable score d_float;
declare variable max_delinq d_integer;
declare variable birthday d_date;
declare variable creditstatus d_str_255;
declare variable active_delinq d_integer;
declare variable max_sum d_integer;
declare variable balance d_currency;
declare variable active_balance d_currency;
declare variable k_id d_id;
declare variable miniapp_balance d_integer;
declare variable is_new_client d_bool;
begin
  const = 0.0231272;
  kib_mbf_delinq = 0;
  max_mbf_kh_deliqdayscount = 0;
  act_mbf_non_delinq = 0;
  kib_nombf_non_delinq = 0;
  kib_nombf_delinq = 0;
  act_nombf_non_delinq = 0;
  res_text = '';
  is_new_client = 1;
  active_balance = 0;

  -- опредляем это новый клиент или повторник, в контектсе расчета кредитного лимита
  is_new_client = (select is_new_client from hp_is_new_client_for_cred_limit(:c_id));

  -- Если клиент повторный, высчитываем кредитный лимит через баллы и активный баланс клиента
  -- По документу от 05.2023 "Товары в рассрочку" тоже расчитываются по баллам для повторных клиентов
  if (is_new_client = 0) then
  begin
    --RFCKG 2247
    select max_amount
    from hp_get_max_amount_by_point(:c_id, current_date, 0)
    into max_sum;

    -- Подсчёт активного баланса по клиенту
    for select ks.ks_from_k_id
        from kontstate ks
        inner join knclient kc on ks.ks_from_k_id = kc.kc_from_k_id
        inner join clients cl on kc.kc_from_c_id = cl.c_id
        where kc.kc_from_c_id = :c_id and
              kc.kc_amount_approved > 0 and
              current_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_state = 1 and
              ks.ks_ol_is_delete = 0
        into :k_id
    do
    begin
      for select sum(ok_os)
          from operkred
          where ok_from_k_id = :k_id and
                ok_ol_is_active = 1 and
                ok_type <= 8 and
                ok_ol_date <= current_date and
                ok_from_c_id = 0
          into :balance
      do
      begin
        active_balance = active_balance + balance;
      end
    end
    -- Кредитный лимит
    credit_limit = max_sum - active_balance;
  end
  else
  begin
    -- Берем последнюю кредитную историю клиента
    select kbm_id
    from mc_online_loan_get_kib_id(:c_id)
    into kbm_id;

    if (:kbm_id is null) then
    begin
      res_text = 'Данные КИБ неактуальны, отправьте повторный запрос в КИБ';
      suspend;
      exit;
    end

    -- Если клиент не имеет кредитной истории, устанавливаем лимит в 15000
    if (not exists(select *
                   from kibhistory kh
                   where kh.kh_from_kbm_id = :kbm_id and
                         kh.kh_is_debtor = 1 and
                         kh.kh_creditstatus not starting 'Отказано')) then
    begin
      res_text = 'У клиента нет кредитной истории';
      credit_limit = case calc_type
                       when 'kant_factory' then 40000
                       when 'installment_goods' then 15000
                       else 0
                     end;
      suspend;
      exit;
    end
    -- Если у клиента есть активная просрочка, кредитный лимит не устанавливается
    if (calc_type = 'kant_factory') then
    begin
      if (exists(select *
                 from kibhistory kh
                 where kh.kh_from_kbm_id = :kbm_id and
                       kh.kh_active_delinq_days_count > 0 and
                       kh.kh_is_debtor = 1 and
                       kh.kh_creditstatus not starting 'Отказано' and
                       kh.kh_is_our = 0)) then
      begin
        res_text = 'У клиента имеется активная просрочка в другом ФКУ';
        credit_limit = null;
        suspend;
        exit;
      end
    end
    else
    if (calc_type = 'installment_goods') then
      if (exists(select *
                 from kibhistory kh
                 where kh.kh_from_kbm_id = :kbm_id and
                       kh.kh_active_delinq_days_count > 0 and
                       kh.kh_is_debtor = 1 and
                       kh.kh_creditstatus not starting 'Отказано')) then
      begin
        res_text = 'У клиента имеется активная просрочка';
        credit_limit = null;
        suspend;
        exit;
      end

    select c.c_birth_date, c.c_sex
    from clients c
    where c.c_id = :c_id
    into :birthday, :gender;

    -- Семейное положение из анкеты
    select iif(rd.rd_code = 'Family.unmarried', 1, 0)
    from objectprops opr
    inner join objectvalues ovs on ovs.ovs_from_opr_id = opr.opr_id
    inner join objectvalres ovr on ovr.ovr_from_opr_id = opr.opr_id and
          ovr.ovr_from_ovs_id = ovs.ovs_id
    inner join refdata rd on rd.rd_id = ovr.ovr_result_rd_id
    where opr_object_kind = f_objkind('clients', 'id') and
          opr_object_id = :c_id and
          opr_subject_kind = -1 and
          opr_subject_id = f_get_rd_id('AnketaQuestions.Family') and
          current_date between ovs.ovs_date_begin and ovs.ovs_date_end
    into :family_status;

    if (family_status is null) then
    begin
      select iif(rd.rd_code = 'Family.unmarried', 1, 0)
      from clients c
      inner join refdata rd on rd.rd_id = c.c_family
      where c.c_id = :c_id
      into :family_status;
    end

    -- Если у клиента не заполнено семейное положение, по умолчанию не замужем
    if (family_status is null) then
      family_status = 1;

    -- Возраст клиента (до месяца)
    age = datediff(month, birthday, current_date) / 12;
    -- Если день рождения клиента < текушего дня отнимаем 1 год (расчитываем только полные года)
    if (extract(month from birthday) = extract(month from current_date) and
        extract(day from birthday) < extract(day from current_date)) then
      age = age - 1; -- Получаем возраст клиента (полные года до дня)

    -- Максимальная просрочка МБФ
    select coalesce(max(ok.ok_ol_date - ok_sh_date), 0)
    from operkred ok
    inner join knclient kc on kc.kc_from_k_id = ok.ok_from_k_id
    where kc.kc_from_c_id = :c_id and
          ok.ok_ol_is_active = 1 and
          ok.ok_ol_date <= current_date and
          ok.ok_ol_date > ok_sh_date and
          ok_type < 7 and
          ok_os + ok_perc < 0 and
          ok_from_c_id = 0
    into max_mbf_kh_deliqdayscount;

    -- Подсчёт активных кредитов без активной просрочки МБФ
    select count(*)
    from kontstate ks
    inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
    inner join oksummary oks on oks.oks_from_k_id = kc.kc_from_k_id
    where kc.kc_from_c_id = :c_id and
          ks.ks_state = 1 and
          ks.ks_ol_is_delete = 0 and
          current_date between ks.ks_date_begin and ks.ks_date_end and
          current_date between oks.oks_date_begin and oks.oks_date_end and
          current_date - oks.oks_delinq_date <= 0 and
          oks_close_state = 0
    into act_mbf_non_delinq;

    -- Подсчёт кредитов с просрочкой МБФ
    select count(*)
    from (select kc.kc_from_k_id, max(ok.ok_ol_date - ok_sh_date) as days_delinq
          from operkred ok
          inner join knclient kc on kc.kc_from_k_id = ok.ok_from_k_id
          where kc.kc_from_c_id = :c_id and
                ok.ok_ol_is_active = 1 and
                ok.ok_ol_date <= current_date and
                ok.ok_ol_date > ok_sh_date and
                ok_type < 7 and
                ok_os + ok_perc < 0 and
                ok_from_c_id = 0
          group by kc.kc_from_k_id)
    into kib_mbf_delinq;

    for select kh.kh_max_delinq_days_count,
               kh.kh_creditstatus,
               kh.kh_active_delinq_days_count
        from kibhistory kh
        where kh.kh_from_kbm_id = :kbm_id and
              kh.kh_is_our = 0 and
              kh.kh_is_debtor = 1 and
              kh.kh_creditstatus not starting 'Отказано'
        into max_delinq,
             creditstatus,
             active_delinq
    do
    begin
      -- Подсчёт активных кредитов без активной просрочки Др.ФКУ
      if ((creditstatus in ('Выдан', 'Реструктуризирован', 'Своевременный платеж', 'Начисление процентов', 'Granted_And_Activated', 'Restructured', 'Open_Under_Legal_Proceedings') or (creditstatus containing ('Просрочен'))) and
          active_delinq = 0) then
        act_nombf_non_delinq = act_nombf_non_delinq + 1;
      -- Общее кол-во займов с просрочкой за всю КИ Др.ФКУ
      if (max_delinq > 0) then
        kib_nombf_delinq = kib_nombf_delinq + 1;
      -- Общее кол-во займов без просрочки за всю КИ Др.ФКУ
      if (max_delinq = 0) then
        kib_nombf_non_delinq = kib_nombf_non_delinq + 1;

    end
    -- Расчёт скор-балла
    -- Формулу расчёта предоставили финансисты
    -- Константы на подобии 0.0015496 были предоставлены в файле с формулой расчёта
    score = age * 0.0015496 + gender * 0.0091520 + family_status * 0.0156304 + kib_mbf_delinq * 0.1585543 + max_mbf_kh_deliqdayscount * 0.0016714 + act_mbf_non_delinq * 0.3228265 + kib_nombf_non_delinq * (-0.0081193) + kib_nombf_delinq * 0.0177375 + act_nombf_non_delinq * 0.0157984 + const;
    -- Кредитный лимит
    if (calc_type = 'kant_factory') then
      credit_limit = case
                       when score < 0.3 then 100000
                       when score between 0.3 and 0.8 then 50000
                       else 0
                     end;
    else
    if (calc_type = 'installment_goods') then
      credit_limit = case
                       when score < 0.3 then 35000
                       when score between 0.3 and 0.8 then 20000
                       else 0
                     end;
  end
  select coalesce(ma_sum, 0)
  from get_client_miniapp_sum(:c_id)
  into :miniapp_balance;

  credit_limit = credit_limit - :miniapp_balance;
  credit_limit = iif(credit_limit < 0, 0, credit_limit);
  if (calc_type = 'kant_factory') then
    credit_limit = iif(credit_limit < 15000, 0, credit_limit);
  suspend;
end
