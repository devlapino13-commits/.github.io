CREATE PROCEDURE HP_ROLETAB_ANALYS
RETURNS (
    RN_NAME VARCHAR(30),
    TL_NAME VARCHAR(255),
    TL_MEMO VARCHAR(255),
    TL_GROUP_STR VARCHAR(100))
AS
declare variable tl_group_no integer;
begin
  for select rn_name, tl_name, tl_memo, tl_group_no
      from rolename, roletab, tablist
      where rn_id = rt_from_rn_id and
            rt_can_update = 0 and
            rt_can_insert = 0 and
            rt_can_delete = 0 and
            rt_from_ds_id = tl_id
      into :rn_name, tl_name, tl_memo, tl_group_no
  do
  begin
    if (tl_group_no = 1) then
      tl_group_str = 'Офисы';
    if (tl_group_no = 2) then
      tl_group_str = 'Бух.счета';
    if (tl_group_no = 3) then
      tl_group_str = 'Пользователи';
    if (tl_group_no = 4) then
      tl_group_str = 'Кред.продукты';
    if (tl_group_no = 5) then
      tl_group_str = 'Клиенты';
    if (tl_group_no = 6) then
      tl_group_str = 'Контракты';
    if (tl_group_no = 7) then
      tl_group_str = 'Кредитная папка';
    if (tl_group_no = 8) then
      tl_group_str = 'Приказы по кадрам';
    if (tl_group_no = 9) then
      tl_group_str = 'Операции';
    if (tl_group_no = 10) then
      tl_group_str = 'Шаблоны операций';
    if (tl_group_no = 11) then
      tl_group_str = 'Бланки';
    if (tl_group_no = 12) then
      tl_group_str = 'Отчеты';

    suspend;
  end
end
