CREATE PROCEDURE HP_CONTDATA_DOUBLE_LIST(
    SHOW_MIN_DOUBLE D_SMALLINT,
    SHOW_MAX_LINE D_SMALLINT)
RETURNS (
    REC_NO D_INTEGER,
    CLT_ID D_INTEGER,
    CLT_CELL_NO D_STR_255,
    CLT_ACTUAL_CDA_ID D_ID,
    CLT_ACTUAL_C_ID D_ID,
    CLT_ACTUAL_C_NAME D_STR_255,
    FOUND_COUNT D_INTEGER,
    FOUND_COUNT_ACTIVE D_INTEGER,
    FOUND_COUNT_STOPPED D_INTEGER,
    FOUND_REAL_ACTIVE D_SMALLINT,
    FOUND_CLIENT_LIST D_STR_8100)
AS
declare variable c_id d_id;
declare variable db_id d_id;
declare variable tmp_count_active d_integer;
declare variable tmp_count_stopped d_integer;
declare variable c_name_fio d_str_255;
begin
  if (show_max_line is null) then
    show_max_line = 50;

  if (show_min_double is null) then
    show_min_double = 1;

  rec_no = 0;
  db_id = f_get_db_id();

  for select clt_id, clt_code
  from contlist
  where clt_id <> 0
  into clt_id, clt_cell_no
  do
  begin
    select count(*)
    from contdata
    where cda_is_main = 1 and
          cda_is_delete = 0 and
          cda_status = 1 and
          cda_cont_kind = 0 and
          cda_from_clt_id = :clt_id
    into found_count;

    if (found_count in (0, 1)) then
      Continue;

    found_count_active = 0;
    found_count_stopped = 0;
    found_real_active = 0;
    found_client_list = '';

    c_id = null;
    for select cda_from_c_id, c_name_fio || ' ' || f_datetostr(c_birth_date)
    from contdata, clients
    where cda_is_main = 1 and
          cda_is_delete = 0 and
          cda_status = 1 and
          cda_cont_kind = 0 and
          cda_from_clt_id = :clt_id and
          cda_from_c_id = c_id
    into c_id, c_name_fio
    do
    begin
      tmp_count_active = 0;
      tmp_count_stopped = 0;

      if (exists(select *
      from knclient, kontstate
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            kc_from_k_id = ks_from_k_id and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_date_end >= '31.12.2999')) then tmp_count_active = 1;

       if (db_id = 81001) then
       begin
        if (exists(select *
        from knclient, kontstate, kontract
        where kc_from_c_id = :c_id and
              kc_amount_approved > 0 and
              kc_from_k_id = ks_from_k_id and
              ks_state = 1 and
              ks_ol_is_delete = 0 and
              ks_date_end >= '31.12.2999' and
              ks_from_k_id = k_id and
              k_stop_perc is not null)) then tmp_count_stopped = 1;
       end else
       begin
        if (exists(select *
        from knclient, oksummary
        where kc_from_c_id = :c_id and
              kc_amount_approved > 0 and
              kc_from_k_id = oks_from_k_id and
              oks_date_end >= '31.12.2999' and
              oks_delinq_date < (current_date - 360) and
              oks_close_state = 0)) then tmp_count_stopped = 1;
       end
       found_count_active = found_count_active + tmp_count_active;
       found_count_stopped = found_count_stopped + tmp_count_stopped;

       if (tmp_count_active = 1) then
       begin
         if (found_client_list <> '') then
           found_client_list = found_client_list || ', ';
         found_client_list = found_client_list || c_name_fio;
       end
    end
    found_real_active = found_count_active - found_count_stopped;

    if (found_real_active >= show_min_double) then
    begin
      rec_no = rec_no + 1;
      suspend;
      if (show_max_line > 0 and rec_no = show_max_line) then
        Exit;
    end
  end
end
