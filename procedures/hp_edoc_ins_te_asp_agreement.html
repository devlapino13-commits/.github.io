CREATE PROCEDURE HP_EDOC_INS_TE_ASP_AGREEMENT(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable is_active_fm_user d_bool;
declare variable client_un_id d_id;
declare variable db_id d_id;
declare variable rd_id d_id;
declare variable ver_status d_smallint;
declare variable is_check_number d_smallint;
begin
  -- Процедура создает договор на оказание информационных услуг (Договор АСП)
  -- Готовит бланки и отправляет СМС.
  res_text = '';
  res_is_error = 0;
  db_id = f_get_db_id();
  is_check_number = f_get_const('check_contact_number');

  for select c.c_name_fio, c.c_birth_date, c.c_pass_no, c.c_pass_who, c.c_pass_date
      from clients c
      where c.c_id = :c_id as cursor cur_c
  do
  begin
    if (coalesce(cur_c.c_name_fio, '') = '') then
      res_text = left(res_text || 'Не заполнено поле ФИО клиента. ', 255);
    if (cur_c.c_birth_date is null) then
      res_text = left(res_text || 'Не заполнено поле "Дата рождения". ', 255);
    if (db_id in (39801, 41701) and
        cur_c.c_birth_date > dateadd(year, -18, current_date)) then
      res_text = left(res_text || 'Клиенту еще не исполнилось 18 лет. ', 255);
    if (coalesce(cur_c.c_pass_no, '') = '') then
      res_text = left(res_text || 'Не заполнено поле "Номер документа удостоверяющего личность". ', 255);
    if (coalesce(cur_c.c_pass_who, '') = '') then
      res_text = left(res_text || 'Не заполнено поле "Орган выдавший документ". ', 255);
    if (cur_c.c_pass_date is null) then
      res_text = left(res_text || 'Не заполнено поле "Дата выдачи документа". ', 255);
  end

  if (res_text = '') then
  begin
    -- QR верификация требуется для всех, кроме сотрудников и действующих пользователей МП сотрудников
    if (f_is_employee(c_id, current_date) = 0 and
        f_meet_valid(c_id, 180) = 0) then
    begin
      is_active_fm_user = 0;

      select un_id
      from username
      where un_from_c_id = :c_id
      into client_un_id;

      if (client_un_id is not null) then
      begin
        if (exists(select *
                   from hp_permission_access_check(:client_un_id, 'PermActions', 0, 'FsMobile.Login')
                   where has_access = 1)) then
          is_active_fm_user = 1;
      end

      if (is_active_fm_user = 0) then
        res_text = f_lang(f_get_lang_code(un_id, null), 'Требуется верификация клиента по QR-коду.');
    end
  end
  if (db_id in (39801, 41701) and
      :is_check_number = 1) then
  begin
    select rd.rd_id
    from refdata rd
    where rd.rd_code = 'ActionGroup.SimpleVerify_Phone_Number_Verify'
    into :rd_id;

    select acl.acl_status
    from actionlist acl
    where acl.acl_object_kind = f_objkind('clients', 'id') and
          acl.acl_object_id = :c_id and
          acl.acl_from_rd_id = :rd_id
    order by acl.acl_datetime desc
    rows 1
    into ver_status;

    if (ver_status <> 2) then
      res_text = 'Номер ещё не проверен, отправка ЭЦП невозможна';
  end

  if (res_text = '') then
  begin
    select edl_id, res_text
    from hp_edoc_insert('asp.agreement', f_objkind('clients', 'id'), :c_id, :un_id)
    into edl_id, res_text;
  end

  if (res_text <> '') then
    res_is_error = 1;

  suspend;
end
