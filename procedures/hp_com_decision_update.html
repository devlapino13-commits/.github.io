CREATE PROCEDURE HP_COM_DECISION_UPDATE(
    CMK_ID D_INTEGER,
    NEW_CMD_ID D_INTEGER,
    NEW_CMD_CODE D_STR_CODE,
    NEW_CMD_NAME D_STR_100,
    NEW_CMD_MEMO D_STR_LONG,
    NEW_CMD_INACTIVE D_BOOL,
    NEW_CMDK_SELECTED D_BOOL)
RETURNS (
    CMD_ID D_INTEGER,
    CMD_CODE D_STR_CODE,
    CMD_NAME D_STR_100,
    CMD_MEMO D_STR_LONG,
    CMD_INACTIVE D_BOOL,
    CMDK_SELECTED D_BOOL)
AS
begin
  if (new_cmd_id is null) then
  begin
    insert into com_decision (cmd_code, cmd_name, cmd_memo, cmd_inactive)
    values (:new_cmd_code, :new_cmd_name, :new_cmd_memo, coalesce(:new_cmd_inactive, 0))
    returning cmd_id, cmd_code, cmd_name, cmd_memo, cmd_inactive
    into :cmd_id, :cmd_code, :cmd_name, :cmd_memo, :cmd_inactive;

    new_cmdk_selected = 1;
  end else
  begin
    update com_decision
    set cmd_id = :new_cmd_id,
        cmd_code = :new_cmd_code,
        cmd_name = :new_cmd_name,
        cmd_memo = :new_cmd_memo,
        cmd_inactive = :new_cmd_inactive
    where cmd_id = :new_cmd_id
    returning cmd_id, cmd_code, cmd_name, cmd_memo, cmd_inactive
    into cmd_id, cmd_code, cmd_name, cmd_memo, cmd_inactive;
  end

  cmdk_selected = coalesce(new_cmdk_selected, 0);
  if (cmdk_selected = 0) then
  begin
    delete from com_kind_dec
    where cmkd_from_cmk_id = :cmk_id and
          cmkd_from_cmd_id = :cmd_id;
  end
  else
  begin
    update or insert into com_kind_dec (cmkd_from_cmk_id, cmkd_from_cmd_id)
    values (:cmk_id, :cmd_id)
    matching (cmkd_from_cmk_id, cmkd_from_cmd_id);
  end

  suspend;
end
