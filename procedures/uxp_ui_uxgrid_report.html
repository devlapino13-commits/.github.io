CREATE PROCEDURE UXP$UI_UXGRID_REPORT(
    UFM_UUID D_UUID,
    UN_ID D_ID,
    UFM_UUID_CODE D_STR_CODE = null)
RETURNS (
    UGR_UUID D_UUID,
    UGR_NAME D_STR_100,
    UGR_MEMO D_STR_255,
    UGR_IS_PRIVATE D_BOOL,
    UGR_DEFAULT D_BOOL,
    UGR_CAN_SAVE D_BOOL,
    UGR_UUID_CODE D_STR_CODE,
    UGR_STATE_IMAGE D_INTEGER,
    UGR_DISABLE_EDIT D_BOOL,
    UGR_FILTER_UUID_CODE D_STR_CODE)
AS
declare variable ufr_uuid d_uuid;
declare variable ugs_owner_un_id d_id;
declare variable ugr_share_uuid d_uuid;
declare variable ugr_owner_user d_str_code;
declare variable un_name d_str_code;
begin
  if (coalesce(ufm_uuid, '') = '') then
    ufm_uuid = uxf$uniq(ufm_uuid_code);

  ugr_state_image = 0;

  select ufr_uuid
  from ux$frame
  where ufr_owner = :ufm_uuid and
        ufr_is_mainframe = 1
  into ufr_uuid;

  if (row_count = 0) then
    exception e_default f_format('ufm_uuid not found: ', coalesce(uuid_to_char(ufm_uuid), '<Null>'));

  un_name = uxf$user_login(un_id);

  for select ugr_uuid,
             ugr_share_uuid,
             ugr_name,
             ugr_memo,
             ugr_is_private,
             ugr_default,
             ugr_disable_edit,
             ugr_owner_user
      from ux$grid
      where ugr_ufr_uuid = :ufr_uuid and
            ugr_inactive = 0
      into :ugr_uuid,
           :ugr_share_uuid,
           :ugr_name,
           :ugr_memo,
           :ugr_is_private,
           :ugr_default,
           :ugr_disable_edit,
           :ugr_owner_user
  do
  begin
    ugr_can_save = 0;
    ugr_state_image = 0; -- Общедоступный шаблон

    if (ugr_is_private = 1) then
    begin
      ugs_owner_un_id = null;

      if (ugr_owner_user <> '' and ugr_owner_user = un_name) then
      begin
        ugr_can_save = 1;
        ugr_state_image = 2; -- Свой частный шаблон
      end else
      begin
        select ugs_owner_un_id
        from ux$grshare
        where ugs_ugr_uuid = :ugr_share_uuid and
              ugs_un_id = :un_id
        into ugs_owner_un_id;
  
        if (row_count = 0) then
          Continue;
  
        ugr_state_image = 1; -- Shared, чужой частный шаблон
  
        if (coalesce(ugs_owner_un_id, un_id) = un_id) then
        begin
          ugr_can_save = 1;
          ugr_state_image = 2; -- Свой частный шаблон
        end
      end
    end else
    begin
      if (ugr_owner_user <> '' and ugr_owner_user = un_name) then
      begin
        ugr_can_save = 1;
      end else
      begin
        if (exists(select * from ux$grshare
        where ugs_ugr_uuid = :ugr_share_uuid and
              ugs_un_id = :un_id)) then
        begin
          ugr_can_save = 1;
          ugr_state_image = 2; -- Свой частный шаблон
        end
      end
    end

    ugr_filter_uuid_code = '';
    select uuid_to_char(ufl_uuid)
    from ux$filter
    where ufl_owner = :ugr_uuid and
          ufl_kind = -1
    rows 1
    into ugr_filter_uuid_code;

    ugr_uuid_code = uuid_to_char(ugr_uuid);

    suspend;
  end
end
