CREATE PROCEDURE HP_GET_USER_ROLES(
    UN_ID ID_TYPE)
RETURNS (
    LST_ROLES STR_255)
AS
begin
  select
    left(list(rn_name, ';'),255)
  from permissions
  inner join rolename on rn_id = pm_subject_id
  where pm_object_kind = 0
    and pm_object_id = :un_id
    and pm_subject_kind = 1
    and current_date between coalesce(pm_date_begin, '01.01.2000')
                         and coalesce(pm_date_end, '31.12.2999')
    and pm_is_delete = 0
  into lst_roles;
  lst_roles = lower(coalesce(lst_roles, ''));
  suspend;
end
