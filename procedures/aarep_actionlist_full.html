CREATE PROCEDURE AAREP_ACTIONLIST_FULL(
    REP_ACL_ID ID_TYPE,
    REP_OBJECT_KIND_STR STR_100,
    REP_OBJECT_ID ID_TYPE,
    REP_ACTION_CODE STR_255,
    REP_ONLY_REJECTED INTEGER,
    REP_RESTRICT STR_255 = 'no')
RETURNS (
    REC_ID DECIMAL(18,0),
    REC_PARENT_ID DECIMAL(18,0),
    ACL_ID DECIMAL(18,0),
    ACL_PARENT_ID DECIMAL(18,0),
    ACL_DATE DATE,
    ACL_FROM_REG_ID INTEGER,
    ACL_OBJECT_KIND INTEGER,
    ACL_OBJECT_ID DECIMAL(18,0),
    ACL_FROM_RD_ID INTEGER,
    ACL_RESPONS_C_ID DECIMAL(18,0),
    RES_OBJECT_NAME VARCHAR(255),
    RES_RD_NAME VARCHAR(255),
    RES_RD_MEMO VARCHAR(255),
    RES_RD_ORDERNO INTEGER,
    RES_OPR_ID DECIMAL(18,0),
    RES_OBJECT_STATUS INTEGER,
    RES_OBJECT_DETAIL BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_OBJECT_MEMO BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_OBJECT_BLOB_IDS D_STR_2048,
    RDL_MAIN_RD_ID INTEGER,
    RES_HAS_BLOB INTEGER,
    RES_HAS_MEMO INTEGER,
    RES_OBJECT_BLOB_NAMES BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    ACL_RESPONS_C_FIO VARCHAR(100),
    KONT_STAR_STATUS INTEGER,
    RES_HAS_BLOCK D_BOOL,
    PROOF_INCOME D_INTEGER,
    DATA_ENTRY D_BOOL,
    SELF_EMPLOYED D_BOOL,
    SOLE_TRAIDER D_BOOL)
AS
declare variable rd_id                    integer;
declare variable lev_no                   integer;
declare variable rep_object_kind          integer;
declare variable k_id                     decimal(18,0);
declare variable pr_name                  varchar(255);
declare variable k_kolich_repay           integer;
declare variable k_amount_approved        decimal(18,4) = 0;
declare variable kc_circle                integer;
declare variable kf_id                    decimal(18,0);
declare variable reg_id                   integer;
declare variable lo_id                    decimal(18,0);
declare variable balance_all              decimal(18,4) = 0;
declare variable balance_par7             decimal(18,4) = 0;
declare variable par7                     decimal(18,7);
declare variable par_level                integer;
declare variable color_str                varchar(20);
declare variable k_disb_date              date;
declare variable ma_id                    decimal(18,0);
declare variable app_date                 date;
declare variable prior_status             integer;
declare variable app_date_old             date;
declare variable acl_dont_show_in_order   integer;
declare variable tmp_str                  varchar(255);
declare variable need_detail              integer;
declare variable refund                   integer;
declare variable pr_id                    integer;
declare variable client_count             integer;
declare variable all_cl_amount            curr_type;
declare variable min_cl_amount            curr_type;
declare variable max_cl_amount            curr_type;
declare variable c_id_list                str_255;
declare variable c_id                     id_type;
declare variable kc_amount_approved       curr_type;
declare variable base_percent             perc_type;
declare variable current_percent          perc_type;
declare variable perc_discount            perc_type;
declare variable db_id                    integer;
declare variable cli_status               integer;
declare variable c_stage                  integer;
declare variable res_verif_date           date;
declare variable rd_code                  varchar(255);
declare variable adl_check                str_255;
declare variable adl_send_date            str_100;
declare variable adl_send_kf_name         str_100;
declare variable is_online                d_smallint;
declare variable asp_sign_date_str        d_str_100;
declare variable asp_sign_date_end_str    d_str_100;
declare variable cell_no                  d_str_20;
declare variable pr_code                  d_str_100;
declare variable k_disb_kind              d_smallint;
declare variable is_poruch                d_bool;
declare variable is_doc                   d_bool;
declare variable k_kolich_client_approved d_integer;
declare variable rd_pd_id                 d_integer;
declare variable rd_dia_name              d_str_255;
declare variable acl_main_rd_id           d_integer;
declare variable obm_memo                 d_blob_text;
declare variable acl_status               d_integer;
declare variable bsc_scoring_level        d_integer;
declare variable bsc_scoring_level_txt    d_str_code;
declare variable bsc_scoring_level_color  d_str_code;
declare variable kontref_list             d_str_2048;
declare variable kr_loan_uuid             d_str_code;
declare variable ref_our_loan             d_smallint;
begin
-- Основая процедура для отображения окна кредитного комитета (андеррайтинг)

-- В качестве параметра необходимо передавать ACL_ID из ActionList либо RD_Code из RefData: ActionGroup
-- Данная процедура обращается в процедуры hp_actionlist_rd_status - hp_actionlist_get_1.
-- процедура hp_actionlist_get_1 получает в качестве параметра поле RD_Str_ID из строки справочника в RefData

  res_object_blob_names = '';
  if (rep_only_rejected is null) then
    rep_only_rejected = 0;
  res_has_blob = 0;
  res_has_memo = 0;
  res_has_block = 0;
  if (rep_acl_id is null) then
    rep_acl_id = 0;

  db_id = f_get_db_id();
  rd_dia_name = (select vcm_caption from verifycomment where vcm_code = '24');
  rd_pd_id = f_get_rd_id('ActionList.PersonalDetails');

  if (rep_acl_id = 0) then -- если конкретная строка в ActionList не указана, находим по RD_Code
  begin
    if (coalesce(rep_action_code, '') = '') then
      exit;

    if (rep_action_code similar to '[-]?[[:DIGIT:]]+') then
      rd_id = rep_action_code;
    else
    begin
      select rd_id
      from refdata
      where rd_code = :rep_action_code
      into :rd_id;
    end

    rep_object_kind = f_objkind(rep_object_kind_str, 'id');

    select acl_id, acl_status
    from actionlist
    where acl_object_kind = :rep_object_kind and
          acl_object_id = :rep_object_id and
          acl_from_rd_id = :rd_id
    order by acl_id desc
    rows 1
    into :rep_acl_id, :acl_status;

    if (rep_acl_id is null) then
      rep_acl_id = 0;
  end
  if (rep_acl_id = 0) then
    exit; -- Если не нашли, завершаем работу

  lev_no = 0;
-- Получаем детали заявки (объекта)
  for select acl_id, acl_parent_id, acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id,
             acl_respons_c_id, acl_dont_show_in_order
      from actionlist
      where (acl_id = :rep_acl_id or
            acl_parent_id = :rep_acl_id)
      order by acl_id
      into :acl_id, :acl_parent_id, :acl_date, :acl_from_reg_id, :acl_object_kind, :acl_object_id, :acl_from_rd_id,
           :acl_respons_c_id, :acl_dont_show_in_order
  do
  begin
    need_detail = 0;
    if (acl_parent_id is null) then
      acl_main_rd_id = acl_from_rd_id;

    acl_respons_c_fio = '';
    if (acl_respons_c_id is not null) then
    begin
      if (db_id = 81001) then
      begin
        select unr_fio_replace from  username_replace
        where unr_from_c_id = :acl_respons_c_id
        and unr_actions in (1,3)
        into acl_respons_c_fio;
      end
      if (acl_respons_c_fio = '') then
      begin
        select c_name_fio
        from clients
        where c_id = :acl_respons_c_id
        into :acl_respons_c_fio;
      end
    end

    rec_id = acl_id;
    rec_parent_id = acl_parent_id;

    res_object_detail = '';
    res_object_memo = '';
    res_object_blob_ids = '';
    res_opr_id = null;
    rdl_main_rd_id = acl_from_rd_id;

    -- значения, решения кредитного комитета
    select res_rd_name, res_rd_memo, res_rd_orderno, res_object_name, res_object_status, res_opr_id, res_object_detail,
           res_object_memo, res_object_blob_ids, res_object_blob_names, res_has_block
    from hp_actionlist_rd_status(:rep_acl_id, :acl_object_kind, :acl_object_id, :acl_from_rd_id)
    into :res_rd_name, :res_rd_memo, :res_rd_orderno, :res_object_name, :res_object_status, :res_opr_id,
         :res_object_detail, :res_object_memo, :res_object_blob_ids, :res_object_blob_names, :res_has_block;

    if (rep_only_rejected = 1) then
    begin
      res_object_detail = '';
      res_object_memo = '';
    end

    res_rd_orderno = 0;
    res_has_blob = 0;
    if (res_object_blob_ids <> '') then
      res_has_blob = 1;
    res_has_memo = 0;
    if (res_object_memo <> '') then
      res_has_memo = 1;

    if (res_object_detail <> '') then
    begin
      res_object_detail = BLOB_APPEND(res_object_detail, '<br>');
      res_object_detail = BLOB_APPEND(res_object_detail, res_object_blob_ids || '<br>');
    end

    if (acl_object_kind = 9) then -- Договор
    begin
      pr_name = '';
      select pr_name, k_kolich_repay, k_amount_approved, k_from_kf_id, k_disb_date, k_id, 
             k_from_ma_app_id, k_applicate_date, k_refund, pr_id, k_is_online, k_disb_kind,
             iif(pr_code containing '.selfemployed.', 1, 0), iif(pr_code containing '.business.', 1, 0)
      from kontract kont, product pr
      where k_id = :acl_object_id and
            k_from_pr_id = pr_id
      into pr_name, k_kolich_repay, k_amount_approved, kf_id, k_disb_date, k_id, 
           ma_id, app_date, refund, pr_id, is_online, k_disb_kind, self_employed, sole_traider;

      if (pr_name <> '') then
      begin
        if (app_date is not null) then
        begin
          if (app_date > k_disb_date) then
            app_date = null;
        end
        if (app_date is null) then
        begin
          if (ma_id is not null) then
          begin
            select min(ma_date_app)
            from miniapp
            where ma_id = :ma_id
            into :app_date;
          end
        end
        if (app_date is not null) then
        begin
          app_date_old = app_date;
          select min(ma_date_app)
          from miniapp
          where ma_from_kf_id = :kf_id and
                ma_date_app >= dateadd(day, -10, :app_date)
          into :app_date_old;
        end

        -- Блок СБП
        if (db_id = 81001 and f_get_const('fps_manual_disb') <> 1) then
        begin
          res_object_detail = BLOB_APPEND(res_object_detail, f_format('Тип выдачи: <b>%1</b><br>',
            trim(decode(k_disb_kind, 1, 'Наличными', 2, 'Безналичными', 3, 'Другой счет', 5, 'Банковская карта', 7, 'СБП по номеру телефона', 8, 'Расчетный счет', 'Неизвестно'))));
          for select f_format('<b>Выдача через СБП:</b> ФИО: %1, Банк: %2, Тел: %3<br>',
            iif(fcb_status = -1, '"Не найдено в банке получателе"', coalesce('<font color="red">Ручное подтверждение: ' || fcn_name || '</font>', '"Подтверждено алгоритмом"')),
            coalesce(fpb_name_full, fpb_name),
            f_cont_code(fcb_from_clt_id))
          from knclient
          inner join fps_loan_banks on flb_from_kc_id = kc_id
          inner join fps_client_banks on fbl_from_fcb_id = fcb_id
          inner join fps_banks on fcb_from_fpb_id = fpb_id
          left join fps_client_names on fcn_from_fcb_id = fcb_id
          where
            kc_from_k_id = :acl_object_id and
            kc_amount_approved > 0
          into tmp_str do
            res_object_detail = BLOB_APPEND(res_object_detail, tmp_str);
        end

        res_object_detail = BLOB_APPEND(res_object_detail, 'Кредитный продукт:   <b>' || pr_name || ' ' || trim(iif(is_online = 1, '[ОНЛАЙН]', ''))||'</b>');
        if (refund = 1) then
          res_object_detail = BLOB_APPEND(res_object_detail, ' <font color="red">(Перекредитование)</font><br>');
        else
          res_object_detail = BLOB_APPEND(res_object_detail, '<br>');
        res_object_detail = BLOB_APPEND(res_object_detail, 'Сумма кредита:       <b>' || round(k_amount_approved, 0) || '</b><br>');
        res_object_detail = BLOB_APPEND(res_object_detail, 'Количество платежей: <b>' || round(k_kolich_repay, 0) || '</b><br>');
        res_object_detail = BLOB_APPEND(res_object_detail, 'Дата выдачи: <b>' || iif(k_disb_date is not null, right('0' || extract(day from k_disb_date), 2) || '.' || right('0' || extract(month from k_disb_date), 2) || '.' || extract(year from k_disb_date), 'Не указана') || '</b><br>');

        res_object_detail = BLOB_APPEND(res_object_detail, 'Дата заявки: <b>' || iif(app_date is not null, right('0' || extract(day from app_date), 2) || '.' || right('0' || extract(month from app_date), 2) || '.' || extract(year from app_date), 'Не указана') || '</b><br>');
        if (app_date_old < app_date) then
        begin
          res_object_detail = BLOB_APPEND(res_object_detail, 'Дата первичной заявки: <b>' || iif(app_date_old is not null, right('0' || extract(day from app_date_old), 2) || '.' || right('0' || extract(month from app_date_old), 2) || '.' || extract(year from app_date_old), 'Не указана') || '</b><br>');
        end

        tmp_str = '';
        select obm_memo
        from hp_objectprops_memo_get('Kontract', :acl_object_id, 'Refdata', 'ObjectProps.Client_memo', 1)
        rows 1
        into :tmp_str;
        if (tmp_str is null) then
          tmp_str = '';
        if (tmp_str <> '') then
          res_object_detail = BLOB_APPEND(res_object_detail, 'Примечание для КК:   <b>' || tmp_str || '</b><br>');

        select kal_new_reg_id, kal_new_kf_officer
        from knaktlist
        where kal_from_kf_id = :kf_id and
              current_date between kal_date and kal_date_end and
              knaktlist.is_delete = 0
        into :reg_id, :lo_id;

        balance_all = 0;
        balance_par7 = 0;
        par7 = 0;
        select sum(oks_balance), sum(iif(oks_delinq_date < (current_date - 8), oks_balance, 0.0))
        from oksummary
        where oks_region_id = :reg_id and
              oks_date_end >= '31.12.2999' and
              oks_date_begin <= (current_date - 1) and
              oks_kc_id = :lo_id and
              oks_close_state = 0 
        into balance_all, balance_par7;
        if (balance_all is null) then
          balance_all = 0;
        if (balance_par7 is null) then
          balance_par7 = 0;
        if (balance_all <> 0) then
        begin
          par7 = balance_par7 / balance_all;
          par7 = round(par7 * 100.0, 2);
        end
        color_str = 'black';
        par_level = 1;
        if (par7 > 3) then
        begin
          par_level = 2;
          color_str = '"green"';
        end
        if (par7 > 7) then
        begin
          par_level = 3;
          color_str = '"#FF9900"';
        end
        if (par7 > 12) then
        begin
          par_level = 4;
          color_str = '"red"';
        end
        res_object_detail = BLOB_APPEND(res_object_detail, '<font color=' || :color_str || '>');
        res_object_detail = BLOB_APPEND(res_object_detail, 'Уровень PAR7 СВЗ:    <b>' || par_level || '</font></b><br>');

        c_stage = 0;
        select datediff(month, c_hired_date, current_date)
        from clients
        where c_id = :lo_id
        into c_stage;
        c_stage = coalesce(c_stage, 0);

        res_object_detail = BLOB_APPEND(res_object_detail, 'Стаж СВЗ:    <b>' || c_stage || '</font></b><br>');

        if (db_id = 81001) then
        begin
          select min(bsc_scoring_level)
          from knclient, bki_scoring
          where bsc_from_kc_id = kc_id
            and kc_from_k_id = :k_id
          into bsc_scoring_level;
  
          bsc_scoring_level_txt = case coalesce(bsc_scoring_level, -1)
                                    when 0 then 'Отрицательный'
                                    when 1 then 'Средний'
                                    when 2 then 'Положительный'
                                    else 'Без оценки'
                                  end;
  
          bsc_scoring_level_color = case coalesce(bsc_scoring_level, -1)
                                    when 0 then '"red"'
                                    when 1 then '"blue"'
                                    when 2 then '"green"'
                                    else 'black'
                                  end;

          res_object_detail = BLOB_APPEND(res_object_detail, 'Результат скоринга:    <b><font color=' || :bsc_scoring_level_color || '>' || bsc_scoring_level_txt || '</font></b><br>');

          --RFCRU-4120
          if (refund = 1) then
          begin
            kontref_list = '';

            for select kr.kr_loan_uuid, iif(kr.kr_from_k_id is not null, 1, 0)
            from kontrefinance kr
            where kr_from_ma_id = :ma_id
            into :kr_loan_uuid, ref_our_loan
            do
            begin
              if (ref_our_loan = 1) then
                kontref_list = kontref_list || '<font color="green">' || kr_loan_uuid || '; </font>';
              else
                kontref_list = kontref_list || '<font color="red">' || kr_loan_uuid || '; </font>';
            end

            if (kontref_list <> '') then
              res_object_detail = blob_append(res_object_detail, ' <font color="red">Рефинансируемые займы: <b>' || kontref_list || ' </b></font><br>');
          end
        end

        if (res_opr_id is not null) then
        begin
          prior_status = null;
          res_object_detail = BLOB_APPEND(res_object_detail, '<br>');
          select ovs_add_value
          from objectvalues
          where ovs_from_opr_id = :res_opr_id and
                ovs_is_active = 1 and
                not(ovs_add_value in (0, 1))
          order by ovs_from_opr_id desc, ovs_id desc
          rows 1
          into :prior_status;
          if (prior_status is not null and
              res_object_status <> prior_status) then
          begin
            if (prior_status = -2) then
            begin
              res_object_detail = BLOB_APPEND(res_object_detail, '<font color="red">');
              res_object_detail = BLOB_APPEND(res_object_detail, '<b>Ранее был отказан кредитным комитетом.</font></b><br>');
            end
            if (prior_status = -1) then
            begin
              res_object_detail = BLOB_APPEND(res_object_detail, '<font color="#FF9900">');
              res_object_detail = BLOB_APPEND(res_object_detail, '<b>Ранее был отправлен на доработку.</font></b><br>');
            end
            if (prior_status = 2) then
            begin
              res_object_detail = BLOB_APPEND(res_object_detail, '<font color="green">');
              res_object_detail = BLOB_APPEND(res_object_detail, '<b>Ранее был утвержден кредитным комитетом.</font></b><br>');
            end
          end
          if (prior_status is null) then
          begin
            res_object_detail = BLOB_APPEND(res_object_detail, '<font color="green">');
            res_object_detail = BLOB_APPEND(res_object_detail, '<b>Первый раз на рассмотрении КК.</font></b><br>');
          end
        end

        if (db_id = 81001) then
        begin

          kont_star_status = null;

          select k_curr_status
          from hp_get_kontract_status(:acl_object_id, iif(:k_disb_date > current_date, current_date, :k_disb_date))
          into :kont_star_status;

          kont_star_status = coalesce(kont_star_status, 0);

          color_str = '"red"';
          res_object_detail = BLOB_APPEND(res_object_detail, '<font color=' || :color_str || '>');
          res_object_detail = BLOB_APPEND(res_object_detail, 'Количество баллов:    <b>' || kont_star_status || '</b><br>');

          client_count = 0;
          all_cl_amount = 0;
          min_cl_amount = 1000000;
          max_cl_amount = 0;
          c_id_list = '';

          for select kncl.kc_from_c_id, kncl.kc_amount_approved
              from knclient kncl
              where kncl.kc_from_k_id = :acl_object_id and
                    kncl.kc_amount_approved > 0
              into :c_id, kc_amount_approved
          do
          begin
            client_count = client_count + 1;
            all_cl_amount = all_cl_amount + kc_amount_approved;

            if (min_cl_amount > kc_amount_approved) then
              min_cl_amount = kc_amount_approved;
            if (max_cl_amount < kc_amount_approved) then
              max_cl_amount = kc_amount_approved;
            c_id_list = c_id_list || c_id || ';';
          end
          if (c_id_list <> '') then
          c_id_list = substring(c_id_list from 1 for character_length(c_id_list) - 1); -- чтобы убрать лишнюю ';'

          base_percent = null;

-- получаю максимальную базовую % ставку из настроек продукта
          select plog.prl_percent_max
          from prodlogic plog
          where plog.prl_from_pr_id = :pr_id and
                :k_disb_date between plog.prl_date_begin and plog.prl_date_end and
                :max_cl_amount between coalesce(plog.prl_amount_cl_min, 0) and coalesce(plog.prl_amount_cl_max, 999999) and
                :client_count between coalesce(plog.prl_kolich_min, 0) and coalesce(plog.prl_kolich_max, 999999) and
                :k_kolich_repay between coalesce(plog.prl_repay_min, 0) and coalesce(plog.prl_repay_max, 999999) rows 1
          into :base_percent;

          base_percent = coalesce(base_percent, 0);
          current_percent = 0;

          /*
-- получаю текущаю процентную ставку по займу
            select PERC_DEFAULT
            from HP_GET_PERCENT(:pr_id, :k_disb_date, :CLIENT_COUNT, :ALL_CL_AMOUNT, :k_kolich_repay, :MIN_CL_AMOUNT, :MAX_CL_AMOUNT, :C_ID_LIST)
            into :current_percent;
            */

-- получаю текущаю процентную ставку по займу
          select kd.kd_percent
          from kontdata kd
          where kd.kd_from_k_id = :acl_object_id and
                :k_disb_date between kd.kd_ol_date and kd.kd_date_end
          into :current_percent;

          current_percent = coalesce(current_percent, 0);
          perc_discount = base_percent - current_percent;

          if (perc_discount = 0) then
            res_object_detail = BLOB_APPEND(res_object_detail, 'Процент скидки:    <b>0</b><br>');
          else
            res_object_detail = BLOB_APPEND(res_object_detail, 'Процент скидки:    <b>' || perc_discount || '</b><br>');
          res_object_detail = BLOB_APPEND(res_object_detail, '</font>');
        end

        if (db_id in (39801, 41701)) then
        begin

          kont_star_status = null;

          select k_curr_status
          from hp_get_kontract_status(:acl_object_id, iif(:k_disb_date > current_date, current_date, :k_disb_date))
          into :kont_star_status;

          kont_star_status = coalesce(kont_star_status, 0);

          color_str = '"red"';
          res_object_detail = BLOB_APPEND(res_object_detail, '<font color=' || :color_str || '>');
          res_object_detail = BLOB_APPEND(res_object_detail, 'Количество баллов:    <b>' || kont_star_status || '</b><br>');

          client_count = 0;
          all_cl_amount = 0;
          min_cl_amount = 1000000;
          max_cl_amount = 0;
          c_id_list = '';
          select pr.pr_code
          from product pr
          where pr.pr_id = :pr_id
          into :pr_code;

          if (:pr_code containing('.ipf.')) then
          begin
              perc_discount = (case when kont_star_status between 0 and 3 then 0
                                    when kont_star_status between 4 and 6 then 5
                                    when kont_star_status between 7 and 9 then 9
                                    when kont_star_status between 10 and 12 then 14
                                    when kont_star_status between 13 and 15 then 18
                                    when kont_star_status between 16 and 19 then 23
                                    when kont_star_status between 20 and 24 then 27
                                    when kont_star_status between 25 and 30 then 32
                                    when kont_star_status between 31 and 36 then 37
                                    when kont_star_status between 37 and 42 then 42
                                    when kont_star_status between 43 and 48 then 47
                                    when kont_star_status >48 then 50 end);
              if (perc_discount = 0) then
                res_object_detail = BLOB_APPEND(res_object_detail, 'Процент скидки:    <b>0</b><br>');
              else
                res_object_detail = BLOB_APPEND(res_object_detail, 'Процент скидки:    <b>' || round(perc_discount,2) || '</b><br>');
              res_object_detail = BLOB_APPEND(res_object_detail, '</font>');
          end
        end
      end
      if (acl_dont_show_in_order = 0 or rep_only_rejected = 0 or res_object_status < 0) then
      begin
        if (acl_from_rd_id <> 5791 or rep_only_rejected = 0) then
          suspend;
      end

      need_detail = 1;
    end -- end Договор

    if (acl_object_kind = 0 and
        acl_parent_id is not null) then -- Клиент
    begin
      k_id = null;
      select acl_object_id
      from actionlist
      where acl_id = :acl_parent_id and
            acl_object_kind = 9 rows 1
      into :k_id;
      if (k_id is not null) then
      begin
        pr_name = '';
        is_poruch = 0;
        select pr_name, k_kolich_repay, kc_amount_approved, kc_circle, k_disb_date, k_is_online, 0, k_kolich_client_approved
        from knclient, product, kontract
        where kc_from_k_id = :k_id and
              kc_from_c_id = :acl_object_id and
              kc_amount_approved > 0 and
              kc_from_k_id = k_id and
              k_from_pr_id = pr_id
        union
        select pr_name, 0, 0, 0, k_disb_date, k_is_online, 1, k_kolich_client_approved
        from zalog, product, kontract
        where z_from_k_id = k_id and
              k_from_pr_id = pr_id and
              z_from_k_id = :k_id and
              z_from_c_id = :acl_object_id and
              z_type = 0 and
              :db_id = 94401 -- поручитель (RFCAZ-175)
        into pr_name, k_kolich_repay, k_amount_approved, kc_circle, k_disb_date, is_online, is_poruch, k_kolich_client_approved;
        if (pr_name is null) then
          pr_name = '';

        if (pr_name <> '') then
        begin
          if (kc_circle is null or kc_circle <= 0) then
            kc_circle = 1;
          res_object_detail = BLOB_APPEND(res_object_detail, 'Кредитный продукт:   <b>' || pr_name || ' ' || trim(iif(is_online = 1, '[ОНЛАЙН]', ''))||'</b><br>');
          res_object_detail = BLOB_APPEND(res_object_detail, 'Сумма клиента:       <b>' || round(k_amount_approved, 0) || '</b><br>');
          res_object_detail = BLOB_APPEND(res_object_detail, 'Количество платежей: <b>' || round(k_kolich_repay, 0) || '</b><br>');

          if (db_id = 81001) then
          begin
            cli_status = null;

            select c_curr_status
            from hp_get_client_status(:acl_object_id, iif(:k_disb_date > current_date, current_date, :k_disb_date))
            into :cli_status;

            cli_status = coalesce(cli_status, 0);

            res_object_detail = BLOB_APPEND(res_object_detail, 'Статус:        <b>' || round(cli_status, 0) || '</b><br>');

            if ((ma_id is null) and (k_id is not null)) then
            begin
                select kont.k_from_ma_app_id
                from kontract kont
                where kont.k_id = :k_id
                into :ma_id;
            end

            res_verif_date = null;

            select client_sign_datetime
            from hp_edoc_get_status('loan.application', 'miniapp', :ma_id, null)
            where c_id = :acl_object_id
            into :res_verif_date;

            if (res_verif_date is not null) then
              res_object_detail = BLOB_APPEND(res_object_detail, 'Наличие подписанной заявки от <b>' || f_datetostr(res_verif_date)||'</b><br>');
            else
              res_object_detail = BLOB_APPEND(res_object_detail, 'Заявка не подписана<br>');

            asp_sign_date_str = '';
            asp_sign_date_end_str = '';

            select f_datetostr(eds_status_datetime),
                   f_datetostr(edl_date_end)
            from hp_edoc_client_sign(:acl_object_id)
            where edl_status = 1
            and  edl_from_edc_id = f_edoc_kind('asp.agreement','id')
            order by eds_status_datetime desc
            rows 1
            into :asp_sign_date_str, :asp_sign_date_end_str;

            res_object_detail = BLOB_APPEND(res_object_detail, 'Дата подписания АСП: <b>' || iif(:asp_sign_date_str = '', '[Не подписывался]', asp_sign_date_str)||'</b><br>');
            res_object_detail = BLOB_APPEND(res_object_detail, 'Дата расторжения АСП: <b>' || iif(:asp_sign_date_end_str = '', '---', asp_sign_date_end_str)||'</b><br>');

            is_doc = coalesce((
              select
                maci_is_documented
              from miniappclient
                inner join miniappclient_income
                  on maci_from_mac_id = mac_id
              where mac_from_ma_id = :ma_id
                and mac_from_c_id = :acl_object_id), 0);
            res_object_detail = BLOB_APPEND(res_object_detail, 'Доход подтвержден документально: <b>' || iif(:is_doc = 1, 'ДА', 'НЕТ')||'</b><br>');
          end
          else
          if (is_poruch = 1) then
            res_rd_name = replace(res_rd_name, 'клиентам', 'поручителю');
          else
            res_object_detail = BLOB_APPEND(res_object_detail, 'Цикл:        <b>' || round(kc_circle, 0) || '</b><br>');

          adl_check = '';

          for select f_datetimetostr(t.cfc_datetime), kal.kal_new_kf_name
              from contacts_for_check t
              inner join miniapp maa on t.cfk_from_ma_id = maa.ma_id
              inner join knaktlist kal on kal.kal_from_kf_id = maa.ma_from_kf_id
              where t.cfc_from_c_id = :acl_object_id and
                    maa.ma_date_end = '31.12.9999' rows 1
              into :adl_send_date, :adl_send_kf_name
          do
          begin
            if (not exists(select *
                           from kontstate ks
                           inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
                           where ks.ks_ol_is_delete = 0 and
                                 ks.ks_state = 1 and
                                 kncl.kc_amount_approved > 0 and
                                 kncl.kc_from_c_id = :acl_object_id)) then
            begin
              adl_check = '"Верификация на ногах": последнее письмо в КГ ушло: ' || adl_send_date || '; в рамках группы: ' || adl_send_kf_name || '<br>';
            end
          end
          res_object_detail = BLOB_APPEND(res_object_detail, coalesce(adl_check, ''));

          tmp_str = '';
          select obm_memo
          from hp_objectprops_memo_get('Clients', :acl_object_id, 'Refdata', 'ObjectProps.Client_memo', 1)
          rows 1
          into :tmp_str;
          if (tmp_str is null) then
            tmp_str = '';
          if (tmp_str <> '') then
            res_object_detail = BLOB_APPEND(res_object_detail, 'Примечание СВЗ:   <b>' || tmp_str || '</b><br>');

          -- S.K.
          if (db_id in (39801, 41701)) then
          begin
            cell_no = coalesce((select res_value from hp_get_client_continfo(:acl_object_id, 'ContactInfo.Cell_Main')),'');
            if (cell_no <> '') then
              res_object_detail = BLOB_APPEND(res_object_detail, '<br><a href=sip:' || cell_no || '>Позвонить клиенту</a>' || '<br>');
          end

          if (acl_dont_show_in_order = 0 or rep_only_rejected = 0 or res_object_status < 0) then
          begin
            if (acl_from_rd_id <> 5791 or rep_only_rejected = 0) then
              suspend;
          end

          need_detail = 1;

        end
      end
    end -- end Client
    res_object_detail = res_object_detail||''; --закрываем блоб читая его.

-- Готовым данные в HTML формате
    if (need_detail = 1) then
    begin
      for select acg_from_rd_id, rd_code
          from actiongroup
          inner join refdata on rd_id = acg_from_rd_id
          where acg_group_rd_id = :rdl_main_rd_id and
                acg_parent_id is null and
                acg_is_delete = 0
          order by acg_orderno
          into :acl_from_rd_id, rd_code
      do
      begin
        if (db_id = 81001 and
            rep_restrict = 'yes') then
        begin
          if (rd_code in ('ActionList.CheckingSignatureSample', 'ActionList.FMS_FSSP_Bases', 'ActionList.CIB', 'ActionList.ContactRelation', 'ActionList.AdditionalFiles')) then
            continue;
        end

        lev_no = lev_no - 1;
        rec_id = lev_no;
        res_object_detail = '';
        res_object_memo = '';
        res_object_blob_ids = '';
        res_opr_id = null;

        select res_rd_name, res_rd_memo, res_rd_orderno, res_object_name, res_object_status, res_opr_id,
               res_object_detail, res_object_memo, res_object_blob_ids, res_object_blob_names, res_has_block
        from hp_actionlist_rd_status(:rep_acl_id, :acl_object_kind, :acl_object_id, :acl_from_rd_id)
        into :res_rd_name, :res_rd_memo, :res_rd_orderno, :res_object_name, :res_object_status, :res_opr_id,
             :res_object_detail, :res_object_memo, :res_object_blob_ids, res_object_blob_names, :res_has_block;

        res_has_blob = 0;
        if (res_object_blob_ids <> '') then
          res_has_blob = 1;
        res_has_memo = 0;
        if (res_object_memo <> '') then
          res_has_memo = 1;

        proof_income = 0;
        data_entry = 0;
        if (rep_only_rejected = 0 or res_object_status < 0) then
        begin
          if (acl_from_rd_id <> 5791 or rep_only_rejected = 0) then
          begin
            -- RFCRU-2289
            if (acl_from_rd_id = rd_pd_id and db_id = 81001
              and f_get_const('documented_income_active') = '1') then
            begin
              if (exists(
                select *
                from kk_income
                  inner join kk_income_cib
                    on kic_from_ki_id = ki_id
                where ki_from_acl_id = :acl_parent_id
                  and ki_from_k_id = :k_id))
              then data_entry = 1;

              if (is_doc = 0) then
              begin
                obm_memo = (
                  select first 1
                    obm_memo
                  from hp_objectprops_get('Kontract', :k_id, -1, :acl_main_rd_id, null)
                    inner join objectmemo
                      on obm_from_opr_id = ovs_from_opr_id
                  order by obm_datetime desc);
                if (obm_memo containing rd_dia_name) then
                  proof_income = -1;
              end else
                proof_income = 1;
            end
            suspend;
          end
        end
      end
    end
  end -- end of Actionlist
end
