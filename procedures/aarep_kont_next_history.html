CREATE PROCEDURE AAREP_KONT_NEXT_HISTORY(
    REP_K_ID D_ID,
    REP_PR_LOAN_TYPE D_SMALLINT,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_SH_DATE D_DATE,
    REP_OL_DATE D_DATE,
    REP_OS D_CURRENCY,
    REP_CR_RATE D_CURRENCY)
RETURNS (
    HIST_NEW D_CURRENCY,
    HIST_NO D_CURRENCY,
    HIST_STILL D_CURRENCY)
AS
declare variable tmp_state d_integer;
declare variable tmp_state_date d_date;
declare variable c_id d_id;
declare variable c_os d_currency;
declare variable cnt d_integer;
declare variable n d_integer;
begin
  -- Процедура возвращает полученный rep_os (сумма погашения) согласно последующей истории (RFCKG-2086)
  -- Если контракт групповой и закрытый, то сумма досрочки считается по каждому клиенту отдельно.

  -- Входящие параметры:
  -- rep_k_id - контракт, по которому было погашение
  -- rep_date_begin - начало отчетного периода
  -- rep_date_end - конец отчетного периода
  -- rep_os - сумма погашения

  -- Исходящие параметры:
  -- hist_new - rep_k_id закрыт, клиент получил новый кредит (есть новая история)
  -- hist_no - rep_k_id закрыт, клиент не получил новый кредит (последующей истории нет)
  -- hist_still - rep_k_id остается активным на rep_date_end (история сохраняется за rep_k_id)

  hist_new = 0;
  hist_no = 0;
  hist_still = 0;

  select ks_state,
         ks_date_begin
  from kontract
  inner join kontstate on ks_from_k_id = k_id
  where k_id = :rep_k_id and
        ks_ol_is_delete = 0 and
        :rep_date_end between ks_date_begin and ks_date_end
  into tmp_state, -- состояние контракта на конец отчетного периода
       tmp_state_date; -- дата начала состояния

  if (tmp_state = 1) then -- контракт активный
  begin
    hist_still = :rep_os; -- вся сумма идет в "частично досрочные"
    suspend;
    exit;
  end

  if (tmp_state = 2) then -- контракт закрыт
  begin

    if (rep_pr_loan_type = 1) then -- контракт индивидуальный
    begin
      select kc_from_c_id
      from knclient
      where kc_amount_approved > 0 and
            kc_from_k_id = :rep_k_id
      into c_id;

      select count(*)
      from kontstate
      inner join knclient on kc_from_k_id = ks_from_k_id
      where kc_amount_approved > 0 and
            kc_from_c_id = :c_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :tmp_state_date and :rep_date_end
      into cnt; -- количество контрактов клиента, полученных после закрытия rep_k_id и до конца отчетного периода

      if (cnt > 0) then
        hist_new = :rep_os;
      else
        hist_no = :rep_os;
    end

    if (rep_pr_loan_type = 0) then -- контракт групповой
    begin
      for select kc_from_c_id
          from knclient
          where kc_amount_approved > 0 and
                kc_from_k_id = :rep_k_id
          into c_id
      do
      begin
        cnt = 0;
        c_os = 0;

        select sum(-ok_os * :rep_cr_rate)
        from operkred
        inner join operlist on ol_id = ok_from_ol_id
        where ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_sh_date > ok_ol_date and
              ok_sh_date = :rep_sh_date and
              ok_ol_date = :rep_ol_date and
              ok_from_k_id = :rep_k_id and
              ol_document_client = :c_id
        into c_os; -- сумма досрочки клиента группового контракта в дату rep_ol_date

        if (c_os > 0) then
        begin
          select count(*)
          from kontstate
          inner join knclient on kc_from_k_id = ks_from_k_id
          where kc_amount_approved > 0 and
                kc_from_c_id = :c_id and
                ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_begin between :tmp_state_date and :rep_date_end
          into cnt; -- количество контрактов клиента, полученных после закрытия rep_k_id и до конца отчетного периода

          if (cnt > 0) then
            hist_new = hist_new + c_os;
          else
            hist_no = hist_no + c_os;
        end
      end
    end

  end

  suspend;
end
