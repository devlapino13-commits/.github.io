CREATE PROCEDURE FM_HTML_SPOUSES_ACTIVE_BALANCE(
    OBJECT_ID D_ID)
RETURNS (
    SPOUSE_DATA D_BLOB_TEXT)
AS
declare variable spouse_fio d_str_100;
declare variable balance d_currency;
declare variable calc_perc d_currency;
declare variable nsp d_currency;
declare variable active_balance d_currency;
declare variable k_id d_id;
declare variable c_id d_id;
declare variable c_id_spouse d_id;
declare variable counter d_smallint;
declare variable rel_type d_str_20;
begin
  spouse_data = '';
  active_balance = 0;

  for select mac.mac_from_c_id
      from miniappclient mac
      inner join miniapp ma on mac.mac_from_ma_id = ma.ma_id
      where mac.mac_from_ma_id = :object_id and
            mac.mac_is_poruch = 0 and
            ma.ma_is_active = 1 and
            ma.ma_date_end = '31.12.9999'
      into :c_id
  do
  begin
    for select crt.rel_other_c_id, crt.rel_code
        from hp_client_relation_tmp(:c_id, 2) crt
        into :c_id_spouse, :rel_type
    do
    begin
      if ((rel_type = '') or (rel_type = 'x')) then
      begin
        rel_type = 0;
      end
      if (cast(rel_type as d_integer) <> 2) then
        continue;

      spouse_fio = '';

      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id_spouse
      into :spouse_fio;

      for select ks.ks_from_k_id
          from kontstate ks
          inner join knclient kc on ks.ks_from_k_id = kc.kc_from_k_id
          inner join clients cl on kc.kc_from_c_id = cl.c_id
          where kc.kc_from_c_id = :c_id_spouse and
                kc.kc_amount_approved > 0 and
                current_date between ks.ks_date_begin and ks.ks_date_end and
                ks.ks_state = 1 and
                ks.ks_ol_is_delete = 0
          into :k_id
      do
      begin
        for select sum(zl.zad_os + zl.sh_calc_perc), zl.sh_calc_perc
            from aarep_make_zad_list(:k_id, current_date, 1, 0) zl
            group by zl.sh_calc_perc
            into :balance, :calc_perc
        do
        begin
          nsp = calc_perc * 0.02;
          active_balance = active_balance + balance + round(nsp, 0);
        end
      end
      if ((spouse_fio <> '') or (spouse_fio is not null)) then
      begin
        spouse_data = BLOB_APPEND(spouse_data, 'ФИО супруга(и):<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || :spouse_fio || '</b><br>');
        spouse_data = BLOB_APPEND(spouse_data, 'Активный баланс супруга(и) в МБулаке:<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || round(coalesce(:active_balance, 0), 0) || '</b><br>');
      end
    end
  end
  spouse_data = coalesce(spouse_data, '');
  suspend;
end
