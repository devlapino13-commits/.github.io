CREATE PROCEDURE RCIB_PREPARE_CREDIT_LIST_TEMP(
    REP_DATE DATE,
    BASE_RCP_ID ID_TYPE,
    REP_K_ID ID_TYPE,
    REP_C_ID ID_TYPE,
    STEP SMALLINT)
RETURNS (
    K_KONTRACT_NO VARCHAR(50),
    K_ID ID_TYPE,
    K_AMOUNT_APPROVED DECIMAL(15,2),
    C_ID ID_TYPE,
    CRED_SUM DECIMAL(15,2),
    KC_NUMBER INTEGER,
    CRED_PERCENT DECIMAL(3,2),
    CRED_JOIN SMALLINT,
    CRED_STATUS SMALLINT,
    FACT_CLOSE_DATE DATE,
    CRED_ENDDATE DATE,
    CRED_DATE DATE,
    CRED_CURRENCY VARCHAR(3),
    K_DEVIDE_DATE DATE,
    K_ORIG_ID ID_TYPE,
    CRED_NO_PREV VARCHAR(50),
    REP_DATE_BEGIN DATE,
    TYPE_ SMALLINT,
    K_FROM_PR_ID ID_TYPE,
    IS_OBESP SMALLINT,
    IS_PORUCH SMALLINT)
AS
declare variable active_date date;
declare variable k_id_old id_type;
declare variable c_id_tmp id_type;
declare variable pr_code str_100;
begin
  rep_k_id = coalesce(rep_k_id, 0);
  rep_c_id = coalesce(rep_c_id, 0);
  if (base_rcp_id = 0) then
    base_rcp_id = null;

  if (step = 1) then
  begin
    active_date = cast('01.07.2014' as date); -- дата запуска проекта

    if (base_rcp_id is null) then
    begin
      if (rep_k_id = 0) then
      begin
        select max(rcp_date) + 1
        from rcib_pocket rcp
        where rcp.rcp_sended = 1 and
              rcp.rcp_is_kredit = 1 and
              rcp.rcp_type = 1 -- групповой
              and
              rcp.rcp_date <= :rep_date
        into :rep_date_begin;
      end
      else
      begin
        rep_date_begin = '01.02.1900';
      end
    end
    else --  если пакет по ошибкам
    begin
      select rcp.rcp_date_begin, rcp.rcp_date
      from rcib_pocket rcp
      where rcp.rcp_id = :base_rcp_id and
            rcp.rcp_is_kredit = 1 -- << -- кредитный пакет
      into :rep_date_begin, :rep_date;
    end

    if (rep_date_begin is null) then
      rep_date_begin = active_date;

    if (rep_k_id = 0) then
    begin
-- если это очередной кредитный пакет
      if (base_rcp_id is null) then
      begin
        for select distinct k_id, kc_from_c_id, k_kontract_no, k_disb_date, k_amount_approved, k_percent,
                            (
                            case
                              when k_kolich_client_approved > 1 then 1
                              else 0
                            end) as cred_join,
                            kncl.kc_amount_approved, kc_number, k_kredline_end, k_currency, cred_status,
                            fact_close_date, k_from_pr_id
            from (select 1 as cred_status, k_id, k_kontract_no, k_disb_date, k_amount_approved, k_percent,
                         k_kredline_end, k_currency, k_kolich_client_approved, null as fact_close_date, k_from_pr_id
                  from kontract kont
                  inner join kontstate ks on ks.ks_from_k_id = kont.k_id
                  where ks.ks_state = 1 and
                        ks.ks_ol_is_delete = 0 and
                        :rep_date between ks_date_begin and ks_date_end and
                        ks_date_begin >= :active_date
--and k_id =  101050000000900501 -- помни
                  union all
                  select 2 as cred_status, k_id, k_kontract_no, k_disb_date, k_amount_approved, k_percent,
                         k_kredline_end, k_currency, k_kolich_client_approved, ks_date_begin as fact_close_date,
                         k_from_pr_id
                  from kontract kont
                  inner join kontstate ks on ks.ks_from_k_id = kont.k_id
                  where ks.ks_state = 2 and
                        ks.ks_ol_is_delete = 0 and
                        ks_date_begin between :rep_date_begin and :rep_date and
                        kont.k_disb_date >= :active_date
--and k_id =  101050000000900501 -- помни
            ) a
            inner join knclient kncl on kncl.kc_from_k_id = k_id
            inner join clients cl on cl.c_id = kncl.kc_from_c_id
            where kncl.kc_amount_approved > 0 and
                  cl.c_type = 0
            order by k_id, kc_number
--rows 3000 --10005 -- << Помни что нужно убрать!
            into :k_id, :c_id, :k_kontract_no, :cred_date, :k_amount_approved, :cred_percent, :cred_join, :cred_sum,
                 :kc_number, :cred_enddate, :cred_currency, :cred_status, :fact_close_date, :k_from_pr_id
        do
        begin
          suspend;
        end
      end
      else -- если пакет по ошибкам
      begin
        for select distinct k_id, kc_from_c_id, k_kontract_no, k_disb_date, k_amount_approved, k_percent,
                            (
                            case
                              when k_kolich_client_approved > 1 then 1
                              else 0
                            end) as cred_join,
                            kncl.kc_amount_approved, kc_number, k_kredline_end, k_currency, cred_status,
                            fact_close_date, k_from_pr_id
            from (select distinct 1 as cred_status, k_id, k_kontract_no, k_disb_date, k_amount_approved, k_percent,
                                  k_kredline_end, k_currency, k_kolich_client_approved, null as fact_close_date,
                                  k_from_pr_id
                  from kontract kont
                  inner join kontstate ks on ks.ks_from_k_id = kont.k_id
                  inner join rcib_kontract rck on rck.rck_from_k_id = kont.k_id
                  inner join rcib_kont_errors rce on rck.rck_from_cp_id = rce.rce_pocket_id and
                        rck.rck_rec_no = rce.rce_from_rec_no
                  where ks.ks_state = 1 and
                        ks.ks_ol_is_delete = 0 and
                        :rep_date between ks_date_begin and ks_date_end and
                        rce.rce_pocket_id = :base_rcp_id
                  union all
                  select distinct 2 as cred_status, k_id, k_kontract_no, k_disb_date, k_amount_approved, k_percent,
                                  k_kredline_end, k_currency, k_kolich_client_approved,
                                  ks_date_begin as fact_close_date, k_from_pr_id
                  from kontract kont
                  inner join kontstate ks on ks.ks_from_k_id = kont.k_id
                  inner join rcib_kontract rck on rck.rck_from_k_id = kont.k_id
                  inner join rcib_kont_errors rce on rck.rck_from_cp_id = rce.rce_pocket_id and
                        rck.rck_rec_no = rce.rce_from_rec_no
                  where ks.ks_state = 2 and
                        ks.ks_ol_is_delete = 0 and
                        ks_date_begin between :rep_date_begin and :rep_date and
                        rce.rce_pocket_id = :base_rcp_id) a
            inner join knclient kncl on kncl.kc_from_k_id = k_id
            inner join clients cl on cl.c_id = kncl.kc_from_c_id
            where kncl.kc_amount_approved > 0 and
                  cl.c_type = 0
            order by k_id, kc_number
            into :k_id, :c_id, :k_kontract_no, :cred_date, :k_amount_approved, :cred_percent, :cred_join, :cred_sum,
                 :kc_number, :cred_enddate, :cred_currency, :cred_status, :fact_close_date, :k_from_pr_id
        do
        begin
          suspend;
        end
      end
    end
    else -- if K_ID <> 0
    begin
      for select distinct k_id, kc_from_c_id, k_kontract_no, k_disb_date, k_amount_approved, k_percent,
                          (
                          case
                            when k_kolich_client_approved > 1 then 1
                            else 0
                          end) as cred_join,
                          kncl.kc_amount_approved, kc_number, k_kredline_end, k_currency, cred_status, fact_close_date,
                          k_from_pr_id
          from (select 1 as cred_status, k_id, k_kontract_no, k_disb_date, k_amount_approved, k_percent, k_kredline_end,
                       k_currency, k_kolich_client_approved, null as fact_close_date, k_from_pr_id
                from kontract kont
                inner join kontstate ks on ks.ks_from_k_id = kont.k_id
                where ks.ks_state = 1 and
                      ks.ks_ol_is_delete = 0 and
                      :rep_date between ks_date_begin and ks_date_end and
                      kont.k_id = :rep_k_id
                union all
                select 2 as cred_status, k_id, k_kontract_no, k_disb_date, k_amount_approved, k_percent, k_kredline_end,
                       k_currency, k_kolich_client_approved, ks_date_begin as fact_close_date, k_from_pr_id
                from kontract kont
                inner join kontstate ks on ks.ks_from_k_id = kont.k_id
                where ks.ks_state = 2 and
                      ks.ks_ol_is_delete = 0 and
                      ks_date_begin between :rep_date_begin and :rep_date and
                      kont.k_disb_date >= :active_date and
                      kont.k_id = :rep_k_id) a
          inner join knclient kncl on kncl.kc_from_k_id = k_id
          inner join clients cl on cl.c_id = kncl.kc_from_c_id
          where kncl.kc_amount_approved > 0 and
                cl.c_type = 0 and
                ((kncl.kc_from_c_id = :rep_c_id) or
                (:rep_c_id = 0))
          order by k_id, kc_number
--rows 10005 -- << Помни что нужно убрать!
          into :k_id, :c_id, :k_kontract_no, :cred_date, :k_amount_approved, :cred_percent, :cred_join, :cred_sum,
               :kc_number, :cred_enddate, :cred_currency, :cred_status, :fact_close_date, :k_from_pr_id
      do
      begin
        suspend;
      end
    end
  end

-------

  if (step = 2) then
  begin
    k_id_old = 0;
    k_orig_id = 0;
    for select k_kontract_no, k_id, k_amount_approved, c_id, cred_sum, kc_number, cred_percent, cred_join, cred_status,
               fact_close_date, cred_enddate, cred_date, cred_currency, k_devide_date, k_orig_id, cred_no_prev,
               rep_date_begin, k_from_pr_id
        from rcib_prepare_credit_list_temp(:rep_date, :base_rcp_id, :rep_k_id, :rep_c_id, 1)
        into :k_kontract_no, :k_id, :k_amount_approved, :c_id, :cred_sum, :kc_number, :cred_percent, :cred_join,
             :cred_status, :fact_close_date, :cred_enddate, :cred_date, :cred_currency, :k_devide_date, :k_orig_id,
             :cred_no_prev, :rep_date_begin, :k_from_pr_id
    do
    begin
      if (k_id_old <> k_id) then
      begin
        k_id_old = :k_id;
        k_orig_id = 0;
        k_devide_date = null;

        cred_no_prev = '';
-- проверяю была ли амнистия
        select kd.kd_k_orig, kd.kd_date
        from kontdevides kd
        where kd.kd_k_new = :k_id
        into :k_orig_id, :k_devide_date;
      end

      k_orig_id = coalesce(k_orig_id, 0);
      cred_no_prev = coalesce(cred_no_prev, '');

      if (k_orig_id > 0) then -- если контракт амнистированный
      begin
-- проверяю с каким номером контракт отправлялся в киб первый раз,
-- чтобы с тем же номером и передавать в дальнейшем
        select rck.rck_cred_no
        from rcib_kontract rck
        where rck.rck_from_k_id = :k_orig_id and
              rck.rck_from_c_id = :c_id and
              rck.rck_from_cp_id > 10
        order by rck.rck_from_cp_id
        rows 1
        into :cred_no_prev;

        suspend;

-- если дата разделения до 18.05.2015 тогда передаю контракт дважды, с номером до разделения и после
        if (k_devide_date < cast('18.05.2015' as date)) then
        begin
          k_orig_id = 0;
          k_devide_date = null;
          cred_no_prev = '-';
          suspend;
        end
      end
      else
      begin
        suspend;
      end
    end
  end

  if (step = 3) then
  begin
    k_id_old = 0;

    for select k_kontract_no, k_id, k_amount_approved, c_id, cred_sum, kc_number, cred_percent, cred_join, cred_status,
               fact_close_date, cred_enddate, cred_date, cred_currency, k_devide_date, k_orig_id, cred_no_prev,
               rep_date_begin, k_from_pr_id
        from rcib_prepare_credit_list_temp(:rep_date, :base_rcp_id, :rep_k_id, :rep_c_id, 2)
--where K_id = 101050000000900501  -- Помни!!!
        into :k_kontract_no, :k_id, :k_amount_approved, :c_id_tmp, :cred_sum, :kc_number, :cred_percent, :cred_join,
             :cred_status, :fact_close_date, :cred_enddate, :cred_date, :cred_currency, :k_devide_date, :k_orig_id,
             :cred_no_prev, :rep_date_begin, :k_from_pr_id
    do
    begin
      if (k_id <> k_id_old) then
      begin
        k_id_old = k_id;

--            select pr.pr_code
--            from product pr where pr.pr_id = :k_from_pr_id
--            into :pr_code;

--            is_obesp = 1;
        /*
            1   Групповой микрозаем
                01.07.14 - 14.12.14    обеспечен      Приказ №123-06-14 от 25.06.14
                15.12.14 - 14.01.15    не обеспечен   Приказ №249-12-14 от 08.12.14
                15.01.15 - 30.06.15    обеспечен      Приказ №274-12-14 от 31.12.14
                01.07.15 - наст.время  не обеспечен   Приказ №26-06-15 от 25.06.15

            */
--            if (pr_code containing('.GR.') and((CRED_DATE between '15.12.2014' and '14.01.2015') or
--                                               (CRED_DATE >= '01.07.2015'))) then is_obesp = 0;
        /*
            2   Микрозаем 1+1
                15.12.14 - 14.01.15    не обеспечен   Приказ №250-12-14 от 08.12.14
                15.01.15 - 30.06.15    обеспечен      Приказ №274-12-14 от 31.12.14
                01.07.15 - наст.время  не обеспечен   Приказ №26-06-15 от 25.06.15
            */
--            if (pr_code containing('.IND.ONE.') and((CRED_DATE between '15.12.2014' and '14.01.2015') or
--                                                    (CRED_DATE >= '01.07.2015'))) then is_obesp = 0;
        /*
            3.  Индивидуальный с поручительством
                04.08.14 - 30.06.15    обеспечен      Приказ №157-08-14 от 01.08.14
                30.06.2015             отменен        Приказ №26-06-15 от 25.06.15

            4,5. Семья, Семья Плюс
                05.10.15 - 21.10.15    не обеспечен   Приказ №23-09-15 от 30.09.15
                                                      Приказ №18-10-15 от 26.10.15
                22.10.15 - наст.время  обеспечен      Приказ №11/1-10-15 от 19.10.15
            */
--            if (((pr_code containing ('.GR.FAMILY.')) or (pr_code containing ('.GR.FAMILY PLUS.')))
--                and(CRED_DATE between '05.10.2015' and '21.10.2015')) then is_obesp = 0;
        /*
            6   Баракат
                05.10.15  - наст.время не обеспечен   Приказ №23-09-15 от 30.09.15
            7   Баракат Плюс
                05.10.15  - наст.время не обеспечен   Приказ №23-09-15 от 30.09.15
            */

--            if (((pr_code containing ('.IND.BARAKAT.')) or (pr_code containing ('.IND.BARAKAT PLUS.')))
--                and(CRED_DATE >= '05.10.2015')) then is_obesp = 0;
        /*
            8   Будущий сотрудник
                01.07.14 - наст.время  обеспечен      Приказ №59-03-14 от 14.03.14

            9    Микрозаем сотруднику
                 05.06.15 - наст.время обеспечен      Приказ № 02-06-15 от 01.06.15
            */
        select is_secured
        from aarep_get_secured_loan(null, null, null, null, null, null, :k_id)
        into is_obesp;
      end

      if (is_obesp = 0) then -- если кредит не обчеспечен участники группы друг за друга не отвечают
      begin
        type_ = 0;
        is_poruch = 0;
        c_id = :c_id_tmp;

        suspend;
      end
      else
      begin
-- :type_ = 3, говорит о том что участник группы выступает в качестве поручителя для остальных участников группы
        for select kc_from_c_id, iif(kncl.kc_from_c_id = :c_id_tmp, 0, 3), iif(kncl.kc_from_c_id = :c_id_tmp, 0, 1)
            from knclient kncl
            where kncl.kc_from_k_id = :k_id and
                  kncl.kc_amount_approved > 0
            into :c_id, :type_, :is_poruch
        do
        begin
          suspend;
        end
      end
    end
  end
end
