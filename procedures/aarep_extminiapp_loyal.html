CREATE PROCEDURE AAREP_EXTMINIAPP_LOYAL(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    EMA_ID D_INTEGER,
    PARTNER_FIO STR_100,
    PARTNER_CELL_NO STR_100,
    PARTNER_LO_FIO STR_100,
    FRIEND_FIO STR_255,
    FRIEND_CELL_NO STR_100,
    LANG STR_20,
    SAVE_FR_DATETIME D_DATETIME,
    FRIEND_SOURCE STR_100,
    OFFICE STR_255,
    CITY_NAME STR_100,
    CLK_STATUS STR_100,
    EMA_DATETIME D_DATETIME,
    EMA_BLOCK_DATETIME D_DATETIME,
    EMA_LO_FIO STR_100,
    CURR_STATE STR_100,
    CURR_STATE_DATETIME D_DATETIME,
    REFUSE_REASON STR_255,
    KF_LO_FIO STR_100,
    KF_NAME STR_20,
    FACT_DISB_SVZ_FIO STR_100,
    K_KONTRACT_NO STR_20,
    FACT_DISB_DATE D_DATE)
AS
declare variable c_id id_type;
declare variable p_c_id id_type;
declare variable ema_status d_smallint;
declare variable ema_status_datetime d_datetime;
declare variable ema_refuse str_255;
declare variable ema_refuse_code str_255;
declare variable ma_id d_integer;
declare variable ma_refuse_reason str_255;
declare variable ma_is_active d_smallint;
declare variable ma_state_datetime d_datetime;
declare variable k_id id_type;
declare variable k_state d_smallint;
declare variable k_refuse_reason str_255;
declare variable k_state_datetime d_datetime;
declare variable kf_id id_type;
declare variable clk_fio str_100;
declare variable partner_city str_100;
declare variable partner_reg_id d_integer;
declare variable rep_city str_100;
begin
  select pl_name
  from region
  inner join addrlist on adl_id = reg_address_fact
  inner join places on pl_id = adl_from_pl_id
  where reg_unicode = :rep_region_id
  into rep_city;

  for select clk_fio, friend_cell_no, save_fr_datetime, friend_fio, partner_fio, c_id, p_c_id, partner_cell_no, lang,
             city_name, clk_status, friend_source
      from (select emw_cl_fio as clk_fio, emw_phone as friend_cell_no, emw_datetime as save_fr_datetime,
                   f.c_name_fio as friend_fio, p.c_name_fio as partner_fio, f.c_id as c_id, p.c_id as p_c_id,
                   f_cont_info('clients', p.c_id, 'cell_main', 'ok') as partner_cell_no, emw_language as lang,
                   emw_city as city_name,
                   case clk_status
                     when 0 then 'Новый'
                     when -1 then 'Перезвонить'
                     when 1 then 'Заинтересован'
                     when 2 then 'Заем выдан'
                     when -2 then 'Отказался'
                   end as clk_status,
                   'Друг отправил анкету' as friend_source
            from extminiweb
            inner join clientlink on clk_link_cell_no = emw_phone
            left join clients f on f.c_id = (select c_id
                                             from hp_find_client_by_continfo(0, clk_link_cell_no) rows 1)
            inner join clients p on p.c_id = clk_from_c_id
            where emw_date between :rep_date_begin and :rep_date_end
            union all
            select clk_link_name as clk_fio, clk_link_cell_no as friend_cell_no, clk_add_datetime as save_fr_datetime,
                   f.c_name_fio as friend_fio, p.c_name_fio as partner_fio, f.c_id as c_id, p.c_id as p_c_id,
                   f_cont_info('clients', p.c_id, 'cell_main', 'ok') as partner_cell_no, null as lang,
                   null as city_name,
                   case clk_status
                     when 0 then 'Новый'
                     when -1 then 'Перезвонить'
                     when 1 then 'Заинтересован'
                     when 2 then 'Заем выдан'
                     when -2 then 'Отказался'
                   end as clk_status,
                   'Партнер сохранил в КМП' as friend_source
            from clientlink
            left join clients f on f.c_id = (select c_id
                                             from hp_find_client_by_continfo(0, clk_link_cell_no) rows 1)
            inner join clients p on p.c_id = clk_from_c_id
            where coalesce(clk_from_mcl_id, 0) <> 0 and
                  cast(clk_add_datetime as date) between :rep_date_begin and :rep_date_end)

      into clk_fio, friend_cell_no, save_fr_datetime, friend_fio, partner_fio, c_id, p_c_id, partner_cell_no, lang,
           city_name, clk_status, friend_source
  do
  begin
    friend_fio = coalesce(friend_fio, '') || '(' || clk_fio || ')';
    partner_city = null;
    office = null;
    partner_lo_fio = null;

    select pl_name, reg_unicode, reg_code || ' ' || reg_name, c_name_fio
    from knclient
    inner join kontract on k_id = kc_from_k_id
    inner join knaktlist kal on kal_from_kf_id = k_from_kf_id
    inner join region on reg_unicode = kal_new_reg_id
    inner join addrlist on adl_id = reg_address_fact
    inner join places on pl_id = adl_from_pl_id
    inner join clients on c_id = kal_new_kf_officer
    where kc_from_c_id = :p_c_id and
          kc_amount_approved > 0 and
          kal_date_end = '31.12.9999' and
          k_status > 0 and
          kal.is_delete = 0
    order by k_fact_disb_date desc rows 1
    into partner_city, partner_reg_id, office, partner_lo_fio;

    if (friend_source = 'Друг отправил анкету' and
        partner_city <> city_name) then
      friend_source = 'От Партнера из другого города';

    ema_id = null;
    ema_datetime = null;
    ema_block_datetime = null;
    ema_lo_fio = null;
    ema_status = null;
    ema_status_datetime = null;
    ema_refuse = null;
    ema_refuse_code = null;
    curr_state = null;
    curr_state_datetime = null;
    refuse_reason = null;

    select ema_id, ema_datetime, ema_block_datetime, c_name_fio, ema_status, ema_status_datetime, rd_name, rd_code
    from extminiapp
    inner join region on reg_unicode = ema_office_id
    inner join refdata on rd_id = ema_result_rd_id
    inner join clients on c_id = ema_lo_id
    where ema_from_c_id = :c_id and
          ema_date >= cast(:save_fr_datetime as date)
    order by ema_id desc rows 1
    into ema_id, ema_datetime, ema_block_datetime, ema_lo_fio, ema_status, ema_status_datetime, ema_refuse,
         ema_refuse_code;

    if (ema_block_datetime is not null and
        ema_status = 0) then
    begin
      curr_state = 'Заблокировано ВИПС';
      curr_state_datetime = ema_block_datetime;
    end

    if (ema_block_datetime is null and
        ema_id is not null and
        ema_status = 0) then
    begin
      curr_state = 'Не заблокировано ВИПС';
      curr_state_datetime = ema_datetime;
    end

    if (ema_status = -1 and
        ema_refuse_code <> 'website_miniapp.Refuse.MiniappCreateCancel') then
    begin
      curr_state = 'Отказ на уровне онлайн-заявки';
      refuse_reason = ema_refuse;
      curr_state_datetime = ema_status_datetime;
    end

    ma_id = null;
    ma_is_active = null;
    kf_id = null;
    ma_state_datetime = null;
    ma_refuse_reason = null;

    select ma_id, ma_is_active, ma_from_kf_id, ma_date_save, kr_text
    from miniapp
    inner join miniappclient on mac_from_ma_id = ma_id
    left join kontreslist on krl_ma_id = ma_id
    left join kontres krs on kr_id = krl_from_kr_id
    where mac_from_c_id = :c_id and
          ma_date_end = '31.12.9999' and
          ma_date_app >= cast(:save_fr_datetime as date)
    order by ma_id desc rows 1
    into ma_id, ma_is_active, kf_id, ma_state_datetime, ma_refuse_reason;

    k_id = null;
    k_kontract_no = null;
    k_state = null;
    k_refuse_reason = null;
    fact_disb_svz_fio = null;

    select k_id, k_status, iif(k_status < 0, '', k_kontract_no), kr_text, c_name_fio
    from kontract
    left join kontreslist on krl_from_k_id = k_id
    left join kontres on kr_id = krl_from_kr_id
    left join clients on c_id = k_fact_disb_lo_id
    where k_from_ma_app_id = :ma_id
    order by k_id desc rows 1
    into k_id, k_state, k_kontract_no, k_refuse_reason, fact_disb_svz_fio;

    k_state_datetime = null;
    fact_disb_date = null;

    if (k_id is not null) then
    begin
      select ol_datetime, ol_date
      from operkred
      inner join operlist on ok_from_ol_id = ol_id
      where ok_from_k_id = :k_id and
            ok_type = 8 and
            ok_ol_is_active = 1
      into k_state_datetime, fact_disb_date;
    end

    kf_lo_fio = null;
    kf_name = null;

    select c_name_fio, kal_new_kf_name
    from knaktlist
    inner join clients on c_id = kal_new_kf_officer
    where kal_from_kf_id = :kf_id and
          kal_date_end = '31.12.9999' and
          is_delete = 0 rows 1
    into kf_lo_fio, kf_name;

    if (ma_id is not null and
        k_id is null) then
    begin
      if (ma_is_active = 0) then
        curr_state = 'Отказ';
      if (ma_is_active = 1) then
        curr_state = 'В процессе';
      if (ma_is_active = -1) then
      begin
        curr_state = 'Отказ';
        refuse_reason = ma_refuse_reason;
      end
    end

    if (ma_id is not null and
        k_id is not null) then
    begin
      if (k_state = 1) then
        curr_state = 'Выдан';
      if (k_state = 0) then
        curr_state = 'В процессе';
      if (k_state = -1) then
      begin
        curr_state = 'Отказ';
        refuse_reason = k_refuse_reason;
      end
      if (k_state = -2) then
      begin
        if (ma_is_active = 0) then
          curr_state = 'Отказ';
        if (ma_is_active = 1) then
          curr_state = 'В процессе';
        if (ma_is_active = -1) then
        begin
          curr_state = 'Отказ';
          refuse_reason = ma_refuse_reason;
        end
      end
    end

    if (ma_state_datetime is not null) then
      curr_state_datetime = ma_state_datetime;
    if (k_state_datetime is not null) then
      curr_state_datetime = k_state_datetime;

    if (rep_region_id = 1000000) then
      suspend;
    if (rep_region_id <> 1000000 and
        ((rep_region_id = partner_reg_id and
        friend_source <> 'От Партнера из другого города') or iif(friend_source = 'От Партнера из другого города', rep_city = coalesce(city_name, ''), null))) then
      suspend;
  end
end
