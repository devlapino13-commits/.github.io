CREATE PROCEDURE HP_CESSION_WRITEOFF_FRM(
    SCP_ID ID_TYPE,
    REP_DATE D_DATE,
    UN_ID D_ID)
AS
declare variable k_id id_type;
declare variable ol_id d_id;
declare variable kc_status d_smallint;
begin
    for select scl.scl_from_k_id
    from sud_cession_loans scl
    where scl.scl_from_scp_id = :scp_id
    into :k_id
    do
    begin
        for select new_ol_id, kc_status
        from hp_cession_writeoff(:k_id, :rep_date, :un_id)
        into :ol_id,
             :kc_status
        do
        begin
            if (kc_status = 2) then
            begin
                update sud_cession_loans scl
                set scl.scl_from_ol_id = :ol_id,
                    scl.scl_ol_is_active = 1,
                    scl.scl_ol_date = :rep_date
                where scl.scl_from_scp_id = :scp_id
                  and scl.scl_from_k_id = :k_id;
            end 
        end
    end
end
