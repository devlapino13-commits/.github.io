CREATE PROCEDURE UXP$UXALIAS_LIST(
    UX_UUID D_UUID,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    RES_ID D_STR_CODE,
    RES_UUID D_UUID,
    RES_CAPTION D_STR_255,
    RES_SHORT_CODE D_STR_CODE,
    RES_CODE D_STR_100,
    RES_ALIAS_KIND D_SMALLINT,
    RES_MENU_COUNT D_INTEGER,
    RES_SEARCH_USED D_INTEGER,
    RES_CAN_SEARCH D_BOOL,
    UXA_UUID D_UUID,
    UXA_NAME D_STR_100,
    UXA_KIND D_SMALLINT,
    RES_UUID_STR D_STR_CODE,
    RES_IMAGE D_SMALLINT,
    RES_PARENT_ID D_STR_CODE,
    UXA_MENU_KIND D_SMALLINT,
    RES_SEARCH_INFO D_STR_255,
    UXA_NEED_ACCESS D_BOOL)
AS
declare variable ux_kind d_str_code;
declare variable is_grid d_bool;
declare variable us_main_table d_str_code;
declare variable us_lookup_id d_str_code;
declare variable us_lookup_text d_str_255;
begin
  -- Список алиасов для показа действий

  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  res_uuid = ux_uuid;
  res_id = lower(uuid_to_char(res_uuid));
  res_parent_id = null;
  res_search_used = 0;
  uxa_uuid = null;
  uxa_menu_kind = null;
  uxa_need_access = 0;

  select count(*)
  from ux$action
  where ua_owner = :ux_uuid
  into res_menu_count;

  is_grid = 0;
  select ufr_is_grid, us_main_table
  from ux$frame, ux$source
  where ufr_owner = :ux_uuid and
        ufr_is_mainframe = 1 and
        ufr_us_uuid = us_uuid
  into is_grid, us_main_table;

  if (us_main_table is null) then
    us_main_table = '';

  res_search_info = '';
  if (us_main_table = '') then
    res_search_info = 'У источника данных не указана главная таблица';
  else
  begin
    select us_lookup_id, us_lookup_text
    from ux$source
    where us_code = :us_main_table
    into us_lookup_id, us_lookup_text;

    if (us_lookup_id is null) then
      us_lookup_id = '';
  
    if (us_lookup_text is null) then
      us_lookup_text = '';

    if (us_lookup_id = '' and us_lookup_text = '') then
      res_search_info = upper(us_main_table) || ': Не указаны Id и поле получения текста';
    else
    if (us_lookup_id = '') then
      res_search_info = upper(us_main_table) || ': Не указано Id поле';
    if (us_lookup_text = '') then
      res_search_info = upper(us_main_table) || ': Не указано поле получения текста';
  end

  res_can_search = 0;
  if (is_grid = 1) then
    res_can_search = 1;

  select ux_fcode, ux_name, ux_scode, uxf$kind(uu_kind, 'code')
  from ux$form, ux$uniq, ux$code
  where ufm_uuid = :ux_uuid and
        ufm_ux_uuid = ux_uuid and
        ux_uuid = uu_id
  into res_code, res_caption, res_short_code, ux_kind;

  res_uuid_str = lower(uuid_to_char(res_uuid));

  res_alias_kind = 0;
  res_image = 19;
  uxa_name = '';

  if (ux_kind <> 'uxcommand') then
  begin
    res_caption = res_caption || '  [базовый]';
  
    suspend;

    res_parent_id = res_id;
  end

  for select uxa_uuid, uxa_fcode, uxa_name, uxa_kind, uxa_code, uxa_menu_kind, uxa_need_access
  from ux$alias, ux$form
  where ufm_uuid = :ux_uuid and
        ufm_ux_uuid = uxa_owner
  into res_uuid, res_code, uxa_name, uxa_kind, res_short_code, uxa_menu_kind, uxa_need_access
  do
  begin
    res_uuid_str = lower(uuid_to_char(res_uuid));
    res_id = lower(uuid_to_char(res_uuid));

    uxa_uuid = res_uuid;
    res_alias_kind = 1;
    res_image = 20;

    if (exists(select * from ux$search
    where usr_uuid = :res_uuid)) then
    begin
      res_alias_kind = 2;
      res_image = 4;
    end
    else
      if (uxa_kind = 2) then
      begin
        res_alias_kind = 3;
        res_image = 48;
      end

    res_menu_count = 0;

    select count(*)
    from ux$action
    where ua_uxa_uuid = :res_uuid and
          ua_usr_id_field <> ''
    into res_search_used;
  
    res_caption = uxa_name;
    if (res_short_code <> '') then
      res_caption = res_caption || ' (' || res_short_code || ')';

    if (ux_kind <> 'uxcommand') then
    begin
      select count(*)
      from ux$aliasaction
      where uaa_uxa_uuid = :res_uuid
      into res_menu_count;
    end

    suspend;
  end
end
