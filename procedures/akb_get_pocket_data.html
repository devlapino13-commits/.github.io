CREATE PROCEDURE AKB_GET_POCKET_DATA(
    MP_ID D_ID)
RETURNS (
    POCKET_ID D_ID,
    CATEGORY_NAME D_STR_20,
    CR_TYPE SMALLINT,
    KNC_ID DECIMAL(18,0),
    BANKID D_INTEGER,
    BANKNAME D_STR_100,
    ID VARCHAR(255),
    NAME VARCHAR(255),
    PLACEOFBIRTH VARCHAR(255),
    PINCODE VARCHAR(255),
    COUNTRYCODE VARCHAR(3),
    BANKRUPTCYSTATUS SMALLINT,
    DATEOFBIRTH DATE,
    K_ID DECIMAL(18,0),
    ACCOUNTNO VARCHAR(255),
    CREDITTYPE VARCHAR(3),
    CREDITLINEAMOUNT DECIMAL(18,4),
    PURPOSEOFCREDIT VARCHAR(3),
    CURRENCYOFCREDIT VARCHAR(3),
    INITIALAMOUNTOFCREDIT DECIMAL(18,4),
    DISOUTAMOUNTOFCREDIT DECIMAL(18,4),
    ANNUALINTERESTRATE DECIMAL(18,7),
    DATEOFGRANT DATE,
    DUETIMEFIRSTCONTRACT DATE,
    DUETIMELASTCONTRACT DATE,
    LASTPAYMENTDATE DATE,
    MONTHLYPAYMENTAMOUNT DECIMAL(18,4),
    DAYSMAINSUMISOVERDUE INTEGER,
    DAYSINTERESTISOVERDUE INTEGER,
    OIAFORREPPERIOD DECIMAL(18,4),
    OIAFORLASTTHREEMONTHS DECIMAL(18,4),
    NUMBEROFPROLONGS INTEGER,
    CREDITCLASSCODE VARCHAR(3),
    CREDITSTATUSCODE VARCHAR(3),
    PROVISION DECIMAL(18,4),
    CREDITPERIODINMONTHS INTEGER,
    KC_NUMBER INTEGER,
    GUARANTORID D_STR_100,
    GUARANTTORNAME D_STR_100,
    GUARANTORCOUNTRYCODE VARCHAR(3),
    GUARANTORDATEOFBIRTH D_DATE,
    GUARANTORPLACEOFBIRTH D_STR_100,
    GUARANTORPINCODE D_STR_20,
    COLLATERALTYPECODE VARCHAR(3),
    ANYINFOTODISTING D_STR_2048,
    MARKETVALUE D_CURRENCY)
AS
declare variable pocket_date_begin d_date;
declare variable pocket_date_end d_date;
declare variable pocket_date_begin_str d_str_20;
declare variable pocket_month d_smallint;
declare variable pocket_categ d_integer;
begin
  if (f_get_db_id() <> 94401) then
    exit;

  if (f_get_const('akb_pocket_send') = 0) then
    exit;

  bankid = 2180;
  bankname = 'Mol Bulak Bank Olmayan Kredit Teshkilati MMC';

  for select mp.mp_id, mc_type, mc_id, mc_from_c_id, mc_name, mc_placeofbirth, mc_countrycode, mc_pincode,
             mc_bankruptcystatus, mc_dateofbirth, mc_from_k_id, mc_accountno, mc_credittype, mc_creditlineamount,
             mc_purposeofcredit, mc_currencyofcredit, mc_initialamountofcredit, mc_disoutamountofcredit,
             mc_annualinterestrate, mc_dateofgrant, mc_duetimefirstcontract, mc_duetimelastcontract, mc_lastpaymentdate,
             mc_monthlypaymentamount, mc_daysmainsumisoverdue, mc_daysinterestisoverdue, mc_creditperiodinmonths,
             mc_oiaforrepperiod, mc_oiaforlastthreemonths, mc_numberofprolongs, mc_creditclasscode, mc_creditstatuscode,
             mc_provision, mp.mp_categ
      from mkr_poket mp
      inner join mkr_credit mc on mp.mp_id = mc.mc_from_mp_id
      where mp.mp_is_sended in (-2, 0) and
            mp.mp_id = :mp_id
      into pocket_id, cr_type, id, knc_id, name, placeofbirth, countrycode, pincode, bankruptcystatus, dateofbirth,
           k_id, accountno, credittype, creditlineamount, purposeofcredit, currencyofcredit, initialamountofcredit,
           disoutamountofcredit, annualinterestrate, dateofgrant, duetimefirstcontract, duetimelastcontract,
           lastpaymentdate, monthlypaymentamount, daysmainsumisoverdue, daysinterestisoverdue, creditperiodinmonths,
           oiaforrepperiod, oiaforlastthreemonths, numberofprolongs, creditclasscode, creditstatuscode, provision, :pocket_categ
  do
  begin
    category_name = case
                      when pocket_categ = -1 then 'Внесистемный учет'
                      when pocket_categ = 1 then 'Активные'
                      when pocket_categ = 2 then 'Закрытые'
                    end;

    collateraltypecode = '';
    anyinfotodisting = '';
    marketvalue = null;
    select mzl.collateraltypecode, mzl.anyinfotodisting, mzl.marketvalue
    from mkr_zalog_list(:k_id) mzl
    into collateraltypecode, anyinfotodisting, marketvalue;

    guarantorid = '';
    guaranttorname = '';
    guarantorcountrycode = '';
    guarantordateofbirth = null;
    guarantorplaceofbirth = '';
    guarantorpincode = '';

    if (exists(select *
               from kontract k
               inner join miniapp ma on k.k_from_ma_app_id = ma.ma_id and
                     ma.ma_is_active = 1 and
                     ma.ma_date_end = '31.12.9999'
               inner join miniappclient mac on ma.ma_id = mac.mac_from_ma_id
               where k.k_id = :k_id and
                     mac.mac_is_poruch = 1)) then
    begin
      for select mgl.id, mgl.name, mgl.countrycode, mgl.dateofbirth, mgl.placeofbirth, mgl.pincode
          from mkr_guarantors_list(:k_id) mgl
          into guarantorid, guaranttorname, guarantorcountrycode, guarantordateofbirth, guarantorplaceofbirth,
               guarantorpincode
      do
      begin
        guarantorid = coalesce(guarantorid, '');
        guaranttorname = coalesce(guaranttorname, '');
        guarantorcountrycode = coalesce(guarantorcountrycode, '');
        guarantordateofbirth = coalesce(guarantordateofbirth, null);
        guarantorpincode = coalesce(guarantorpincode, '');
      suspend;
      end
    end
    if (guarantorid = '') then
    suspend;
  end
end
