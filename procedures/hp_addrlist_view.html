CREATE PROCEDURE HP_ADDRLIST_VIEW(
    ADDRESS_ID D_ID)
RETURNS (
    ADL_ID D_ID,
    ADL_FROM_PL_ID D_ID,
    ADL_DEPART_NO D_STR_20,
    ADL_DEPART_PREF D_STR_20,
    ADL_FLAT_NO D_STR_20,
    ADL_BUILDING_NO D_STR_20,
    ADL_SECTION_NO D_STR_20,
    ADL_CROSS_STREET_ID D_ID,
    ADL_ADDRESS_STR D_STR_255,
    ADL_COMMENT D_STR_255,
    ADL_POST_INDEX D_INTEGER,
    ADL_IS_EDITED D_INTEGER,
    ADL_STREET_ID D_ID,
    ADL_FROM_UN_ID D_ID,
    ADL_IS_WITHOUT_STREET D_INTEGER)
AS
begin
  for
    select
      adl_id,
      adl_from_pl_id,
      adl_depart_no,
      adl_depart_pref,
      adl_flat_no,
      adl_building_no,
      adl_section_no,
      adl_cross_street_id,
      adl_address_str,
      adl_comment,
      adl_post_index,
      adl_is_edited,
      adl_street_id,
      adl_from_un_id,
      adl_is_without_street
    from addrlist
    where adl_id = :address_id
  into
    :adl_id,
    :adl_from_pl_id,
    :adl_depart_no,
    :adl_depart_pref,
    :adl_flat_no,
    :adl_building_no,
    :adl_section_no,
    :adl_cross_street_id,
    :adl_address_str,
    :adl_comment,
    :adl_post_index,
    :adl_is_edited,
    :adl_street_id,
    :adl_from_un_id,
    :adl_is_without_street
  do
  begin
    suspend;
  end
end
