CREATE PROCEDURE AAREP_CALC_BONUS_KG_KS_20120601(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_KC_ID DECIMAL(18,0),
    AA_CURR_KC_FIO VARCHAR(100),
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_CURR_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_CURR_DELIQ_BALANCE DECIMAL(18,4),
    AA_LO_JOBNAME VARCHAR(100),
    PS_IS_ON_STAFF_STR VARCHAR(100),
    PS_IS_WORKING_STR VARCHAR(100),
    PAR DECIMAL(18,4),
    AA_AMOUNT_DISB DECIMAL(18,4),
    AA_KOLICH_CL_DISB INTEGER,
    DISB_AVG DECIMAL(18,4),
    NEW_CLI_CLOSED INTEGER,
    OLD_CLI_CLOSED INTEGER,
    NEW_CLI_CLOS_WOFF INTEGER,
    OLD_CLI_CLOS_WOFF INTEGER,
    NEW_CLI_CLOS_SB INTEGER,
    OLD_CLI_CLOS_SB INTEGER,
    OLD_CLI_DISB INTEGER,
    BONUS_ORIG DECIMAL(18,4),
    BONUS_TOTAL DECIMAL(18,4),
    JL_NAME VARCHAR(100),
    PS_IS_ON_STAFF INTEGER,
    PS_IS_WORKING INTEGER,
    PS_FROM_REG_ID INTEGER,
    PAR_BEZ DECIMAL(15,5))
AS
declare variable fact1 decimal(18,4);
declare variable fact2 decimal(18,4);
declare variable fact decimal(18,4);
begin
  for select aa_curr_kc_id, aa_curr_kc_fio, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name,
             sum(aa_curr_amount_active_end), sum(aa_curr_deliq_balance) /*,
             sum(AA_AMOUNT_DISB),
             sum(AA_KOLICH_CL_DISB) */

      from aarep_cb_kg_ks_20120601_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, null, null)
      group by aa_curr_kc_id, aa_curr_kc_fio, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name
      into :aa_curr_kc_id, :aa_curr_kc_fio, :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_level_name,

           :aa_curr_amount_active_end, :aa_curr_deliq_balance /*,
             :AA_AMOUNT_DISB,
             :AA_KOLICH_CL_DISB */
  do
  begin
    par = 0;
    if (:aa_curr_amount_active_end <> 0) then
    begin
      par_bez = :aa_curr_deliq_balance / :aa_curr_amount_active_end * 100;
      par = cast(:aa_curr_deliq_balance / :aa_curr_amount_active_end * 100 as decimal(15,1));
    end
    /*Вычисляет кол-во выдач, сумма выдачи и кол-во выдач повторным клиентам*/
    aa_amount_disb = null;
    select count(*), sum(
           case
             when (kc_circle > 1 and kc_amount_approved >= 10000.0) then 1
             else 0
           end),
           sum(kc_amount_approved)
    from kontstate, knclient, kontract, product
    where ks_ol_is_delete = 0 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_state = 1 and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          ks_kc_id = :aa_curr_kc_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 -- Не включать параллельные кредиты в расчеты, по приказу №23/1-02-13о/д от 07.02.2013
    into :aa_kolich_cl_disb, :old_cli_disb, aa_amount_disb;
    if (aa_amount_disb is null) then
      aa_amount_disb = 0;
--  Вычисление средней суммы выдачи
    disb_avg = 0;
    if (aa_kolich_cl_disb <> 0) then
      disb_avg = cast(:aa_amount_disb / aa_kolich_cl_disb as decimal(15,2));

    fact1 = 0;
    if (disb_avg < 36000.00) then
      fact1 = 20;
    /*fact2=0;*/
    fact2 = case
              when par between 1.0 and 1.5 then 25
              when par between 1.6 and 2.0 then 50
              when par > 2.0 then 100
              else 0
            end;
    fact = fact1 + fact2;

    /*Вычисляет кол-во закрытых новых и повторных клиентов*/
    select sum(
           case
             when kc_circle <= 1 then 1
             else 0
           end),
           sum(
           case
             when kc_circle > 1 then 1
             else 0
           end)
    from kontstate, knclient, kontract, product
    where ks_from_k_id = kc_from_k_id and
          ks_ol_is_delete = 0 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_state = 2 and
          kc_amount_approved >= 10000.0 -- Исправлено по приказу №141-10-12о/д от 25 октября, было kc_amount_approved > 0
          and
          ks_kc_id = :aa_curr_kc_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 -- Не включать параллельные кредиты в расчеты, по приказу №23/1-02-13о/д от 07.02.2013
    into :new_cli_closed, :old_cli_closed;
    if (:new_cli_closed is null) then
      new_cli_closed = 0;
    if (:old_cli_closed is null) then
      old_cli_closed = 0;

    /*Вычисляет кол-во закрытых новых и повтоных со списанием*/
    with tab1
    as (select distinct ks_from_k_id as ks_from_k_id
        from operkred, kontstate
        where ok_from_ol_id = ks_from_ol_id and
              ks_ol_is_delete = 0 and
              cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
              ks_state = 2 and
              ok_type = 5 and
              ok_os <> 0 and
              ks_kc_id = :aa_curr_kc_id and
              ok_from_c_id = 0 and
              (not exists(select *
                          from sb_akt, sb_list
                          where sba_from_k_id = ks_from_k_id /*Не включает переданные в СБ*/
                                and
                                :rep_date_end between sba_date_begin and sba_date_end and
                                sba_is_delete = 0 and
                                sba_from_sbl_id <> 0 and
                                sba_status = 2 and
                                sba_from_sbl_id = sbl_id and
                                sbl_type = 1)))
    select sum(
           case
             when kc_circle <= 1 then 1
             else 0
           end),
           sum(
           case
             when kc_circle > 1 then 1
             else 0
           end)
    from knclient, tab1, kontract, product
    where tab1.ks_from_k_id = kc_from_k_id and
          kc_amount_approved >= 10000.0 -- Исправлено по приказу №141-10-12о/д от 25 октября, было kc_amount_approved > 0
          and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 -- Не включать параллельные кредиты в расчеты, по приказу №23/1-02-13о/д от 07.02.2013
    into :new_cli_clos_woff, :old_cli_clos_woff;
    if (:new_cli_clos_woff is null) then
      new_cli_clos_woff = 0;
    if (:old_cli_clos_woff is null) then
      old_cli_clos_woff = 0;

    /*Вычисляет кол-во закрытых новых и повтоных клиентов переданные в СБ*/
    select sum(
           case
             when kc_circle <= 1 then 1
             else 0
           end),
           sum(
           case
             when kc_circle > 1 then 1
             else 0
           end)
    from kontstate, knclient, kontract k2, product
    where ks_from_k_id = kc_from_k_id and
          k2.k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_state = 2 and
          kc_amount_approved >= 10000.0 -- Исправлено по приказу №141-10-12о/д от 25 октября, было kc_amount_approved > 0
          and
          ks_kc_id = :aa_curr_kc_id and
          k2.k_is_problem = 1 and
          k2.k_from_pr_id = pr_id and
          pr_product_kind <> 2 -- Не включать параллельные кредиты в расчеты, по приказу №23/1-02-13о/д от 07.02.2013
          and
          exists(select *
                 from sb_akt, sb_list
                 where sba_from_k_id = k2.k_id /*Выборка только по СБ*/
                       and
                       sba_date_begin <= :rep_date_end and
                       sba_date_end >= :rep_date_end and
                       sba_is_delete = 0 and
                       sba_from_sbl_id <> 0 and
                       sba_status = 2 and
                       sba_from_sbl_id = sbl_id and
                       sbl_type = 1)
    into :new_cli_clos_sb, :old_cli_clos_sb;
    if (:new_cli_clos_sb is null) then
      new_cli_clos_sb = 0;
    if (:old_cli_clos_sb is null) then
      old_cli_clos_sb = 0;

    /*Вычисляет кол-во выдач для повторных клиентов*/
    /*  select count(*)
      from knaktlist, kontract, kontstate, knclient
      where ks_date_begin between kal_date and kal_date_end
      and knaktlist.is_delete=0
      and kal_from_kf_id=k_from_kf_id
      and ks_from_k_id=k_id
      and ks_ol_is_delete=0
      and ks_date_begin between :rep_date_begin and :rep_date_end
      and ks_state=1
      and k_id=kc_from_k_id
      and kc_amount_approved>0
      and kc_circle>1
      and kal_new_kf_officer=:AA_CURR_KC_ID
      into :old_cli_disb;  */

    if (:old_cli_disb is null) then
      old_cli_disb = 0;

    if (:new_cli_closed > :new_cli_clos_woff) then
      new_cli_closed = :new_cli_closed - :new_cli_clos_woff;
    else
      new_cli_closed = 0;
    if (:old_cli_closed > :old_cli_clos_woff) then
      old_cli_closed = :old_cli_closed - :old_cli_clos_woff;
    else
      old_cli_closed = 0;

    if (:new_cli_closed > :new_cli_clos_sb) then
      new_cli_closed = :new_cli_closed - :new_cli_clos_sb;
    else
      new_cli_closed = 0;
    if (:old_cli_closed > :old_cli_clos_sb) then
      old_cli_closed = :old_cli_closed - :old_cli_clos_sb;
    else
      old_cli_closed = 0;

    bonus_orig = :new_cli_closed * 600 + (:old_cli_closed + :old_cli_disb) * 300;

    if (fact < 100) then
      bonus_total = bonus_orig - bonus_orig * fact / 100;
    else
      bonus_total = 0;

    select jl_name, ps_is_on_staff, ps_is_working, ps_from_reg_id
    from hr_pstatus, joblist
    where ps_from_c_id = :aa_curr_kc_id and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_jl_id = jl_id
    order by ps_is_working desc
    rows 1
    into :jl_name, :ps_is_on_staff, :ps_is_working, :ps_from_reg_id;

    if (ps_is_working is null or ps_is_working = 0) then
      bonus_total = 0;
    if (ps_is_on_staff is not null) then
    begin
      if (ps_is_on_staff = 1) then
        ps_is_on_staff_str = 'Штатный';
      else
        ps_is_on_staff_str = 'Внештатный';
      if (ps_is_working = 0) then
      begin
        ps_is_on_staff_str = ps_is_on_staff_str || ', ' || 'уволен';
        ps_is_working_str = 'Уволен';
      end
      else
      if (ps_is_working = 1) then
        ps_is_working_str = 'Работает';
    end
    else
      ps_is_on_staff_str = 'Приказ не оформлен';

    if (ps_from_reg_id is not null) then
    begin
      select aa_curr_office_name, aa_curr_branch_name
      from aarep_get_office_categ(:ps_from_reg_id, :rep_date_end)
      into :aa_curr_office_name, :aa_curr_branch_name;
    end

    suspend;
  end

end
