CREATE PROCEDURE AAREP_ACCOUNT_OBOROT_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    ACCOUNT VARCHAR(20),
    CURRENCY VARCHAR(3),
    OB_DT DECIMAL(18,4),
    OB_KT DECIMAL(18,4),
    OB_DT_CURR DECIMAL(18,4),
    OB_KT_CURR DECIMAL(18,4),
    MAX_DATE DATE)
AS
begin
-- выдает обороты по лицевым счетам по регионам
-- Sanjari, 19.07.2012

  for select distinct al_currency, obs_from_al_id
  from obsummary, acc_list, region_list(:report_region_id, 1, -1, 1)
  where obs_from_al_id = al_id and
        al_from_reg_id = id_list and
        obs_date between :report_date_begin and :report_date_end
  into :currency, :account
  do
  begin
    select dt_ekv, kt_ekv, dt_val, kt_val, max_date
    from hp_al_obor(:account, :report_date_begin, :report_date_end)
    into :ob_dt, :ob_kt, :ob_dt_curr, :ob_kt_curr, max_date;

    suspend;
  end
end
