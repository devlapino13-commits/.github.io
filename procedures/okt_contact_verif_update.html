CREATE PROCEDURE OKT_CONTACT_VERIF_UPDATE(
    ACL_ID ID_TYPE)
AS
declare variable rd_anketa_code str_255;
declare variable acl_object_id id_type;
declare variable acl_object_kind integer;
declare variable acl_status integer;
declare variable acl_result_rd_id integer;
declare variable acl_from_un_id id_type;
declare variable rd_result_code str_255;
declare variable fiook smallint;
declare variable fiook_str str_255;
declare variable clientknown smallint;
declare variable clientknown_str str_255;
declare variable statusok smallint;
declare variable statusok_str str_255;
declare variable addressok smallint;
declare variable addressok_str str_255;
declare variable addcellno_str str_255;
declare variable check_result smallint;
declare variable memo_str blob sub_type 1 segment size 80;
declare variable fiook_name str_255;
declare variable clientknown_name str_255;
declare variable statusok_name str_255;
declare variable addressok_name str_255;
declare variable addcellno_name str_255;
declare variable birthdate_str str_255;
declare variable birthdate date;
begin
  select rd.rd_code, al.acl_object_id, al.acl_object_kind, al.acl_status, al.acl_result_rd_id, al.acl_from_un_id
  from actionlist al
  inner join refdata rd on rd.rd_id = al.acl_from_rd_id
  where al.acl_id = :acl_id
  into :rd_anketa_code, :acl_object_id, :acl_object_kind, :acl_status, :acl_result_rd_id, :acl_from_un_id;

  if (rd_anketa_code = 'OKTELL.Projects.Verification') then
  begin
    if (acl_status = 2) then -- Анкета обработана
    begin
      fiook = 0;
      fiook_str = '';
      memo_str = '';

      select res_has_code, iif(res_str_values in ('Да', 'Нет'), '', res_str_values)
      from hp_objectprops_anketa_get(:acl_object_kind, :acl_object_id, -1, 'AnketaQuestions.OKTELL.FioOK', null, 'Results.Yes')
      into :fiook, :fiook_str;

      fiook = coalesce(fiook, 0);
      fiook_str = coalesce(fiook_str, '');

      if (fiook_str <> '') then
      begin
        select rd.rd_name
        from refdata rd
        where rd.rd_code = 'AnketaQuestions.OKTELL.FioOK'
        into :fiook_name;

        memo_str = blob_append(memo_str, fiook_name || ': ' || fiook_str || '; ');
      end
----

      clientknown = 0;
      clientknown_str = '';

      select res_has_code, iif(res_str_values in ('Да', 'Нет'), '', res_str_values)
      from hp_objectprops_anketa_get(:acl_object_kind, :acl_object_id, -1, 'AnketaQuestions.OKTELL.ClientKnown', null, 'Results.Yes')
      into :clientknown, clientknown_str;

      clientknown = coalesce(clientknown, 0);
      clientknown_str = coalesce(clientknown_str, '');

      if (clientknown_str <> '') then
      begin
        select rd.rd_name
        from refdata rd
        where rd.rd_code = 'AnketaQuestions.OKTELL.ClientKnown'
        into :clientknown_name;

        memo_str = blob_append(memo_str, clientknown_name || ': ' || clientknown_str || '; ');
      end
---

      statusok = 0;
      statusok_str = '';

      select res_has_code, iif(res_str_values in ('Да', 'Нет'), '', res_str_values)
      from hp_objectprops_anketa_get(:acl_object_kind, :acl_object_id, -1, 'AnketaQuestions.OKTELL.StatusOK', null, 'Results.Yes')
      into :statusok, :statusok_str;

      statusok = coalesce(statusok, 0);
      statusok_str = coalesce(statusok_str, '');

      if (statusok_str <> '') then
      begin
        select rd.rd_name
        from refdata rd
        where rd.rd_code = 'AnketaQuestions.OKTELL.StatusOK'
        into :statusok_name;

        memo_str = blob_append(memo_str, statusok_name || ': ' || statusok_str || '; ');
      end
---

      addressok = 0;
      addressok_str = '';

      select res_has_code, iif(res_str_values in ('Да', 'Нет'), '', res_str_values)
      from hp_objectprops_anketa_get(:acl_object_kind, :acl_object_id, -1, 'AnketaQuestions.OKTELL.AddressOK', null, 'Results.Yes')
      into :addressok, addressok_str;

      addressok = coalesce(addressok, 0);
      addressok_str = coalesce(addressok_str, '');

      if (addressok_str <> '') then
      begin
        select rd.rd_name
        from refdata rd
        where rd.rd_code = 'AnketaQuestions.OKTELL.AddressOK'
        into :addressok_name;

        memo_str = blob_append(memo_str, addressok_name || ': ' || addressok_str || '; ');
      end
---

      addcellno_str = '';

      select iif(res_str_values = 'Значение', '', res_str_values)
      from hp_objectprops_anketa_get(:acl_object_kind, :acl_object_id, -1, 'AnketaQuestions.OKTELL.AddCellNo', null, 'Results.StrValue')
      into :addcellno_str;

      addcellno_str = coalesce(addcellno_str, '');
---

      birthdate_str = '';
      birthdate = null;

      select res_str_values
      from hp_objectprops_anketa_get(:acl_object_kind, :acl_object_id, -1, 'AnketaQuestions.OKTELL.BirthDate', null, 'Results.StrValue')
      into :birthdate_str;

      birthdate_str = coalesce(birthdate_str, '');

      if (birthdate_str <> '') then
      begin
        birthdate = cast(birthdate_str as date);
        when sqlcode -413 do
-- convert_error
        begin
        end
      end
----
      check_result = -2; -- не подтверждена

      if ((fiook + clientknown + statusok + addressok) = 4) then
        check_result = 1; -- подтверждена

      update contacts cont
      set cont.cc_check_result = :check_result,
          cont.cc_name_ok = :fiook,
          cont.cc_client_ok = :clientknown,
          cont.cc_status_ok = :statusok,
          cont.cc_address_ok = :addressok,
          cont.cc_call_user = (select un_name
                               from username
                               where un_id = :acl_from_un_id),
          cont.cc_check_memo = substring(:memo_str from 1 for 255),
          cont.cc_birth_date = iif(:birthdate is null, cont.cc_birth_date, :birthdate),
          cont.cc_cell_no_add = iif(:addcellno_str = '', cont.cc_cell_no_add, :addcellno_str)
      where cont.cc_id = :acl_object_id;
    end
    else
    if (acl_status = -2) then
    begin
      select rd.rd_code, rd.rd_name
      from refdata rd
      where rd.rd_id = :acl_result_rd_id
      into :rd_result_code, :memo_str;

      if (rd_result_code = 'ActionStatus.OKTELL.CantCalled') then -- если до абонента дозвониться не удалось
      begin
        update contacts cont
        set cont.cc_check_result = -1,
            cont.cc_total_res = -1,
            cont.cc_try_kol = 5,
            cont.cc_check_memo = :memo_str
        where cont.cc_id = :acl_object_id;
      end
      else
      /* if (rd_result_code <> 'ActionStatus.OKTELL.Canceled') then */ -- если абонент бросил трубку или номер не существует
      begin
        update contacts cont
        set cont.cc_check_result = -2,
            cont.cc_total_res = -2,
            cont.cc_check_memo = :memo_str
        where cont.cc_id = :acl_object_id and
              cont.cc_check_result = 0;
      end
    end
  end
end
