CREATE PROCEDURE AAREP_ROUTING_SHEET_DP(
    OFFICE_ID INTEGER,
    KS_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    TRAVEL_DATE DATE,
    DEPARTURE_PLACE STR_255,
    C_ADDR STR_255,
    ARRIVAL_PLACE STR_255)
AS
declare variable reg_address str_255;
declare variable k_id id_type;
declare variable address_str str_255;
declare variable pos integer;
declare variable counter integer;
declare variable c_qty integer;
declare variable kont_disb_date date;
begin
  /*    igusev 18.12.2014
        процедура предназначена для детальной части (Detail data) "Маршрутного листа"
  */
  select adl.adl_address_str
  from region reg
  inner join addrlist adl on adl.adl_id = reg.reg_address_fact
  where reg.reg_unicode = :office_id
  into :reg_address;

  -- вытаскиваю все кредиты, выданные за период по СВЗ из процедуры AAREP_ROUTING_SHEET
  for select kont.k_id, ks.ks_date_begin
      from kontstate ks
      inner join kontract kont on kont.k_id = ks.ks_from_k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
      where kont.k_last_reg_id = :office_id and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            kal.is_delete = 0 and
            kal_new_kf_officer = :ks_id --  <<
            and
            ks.ks_date_begin between kal.kal_date and kal.kal_date_end
      order by kont.k_id
      into :k_id, :kont_disb_date
  do
  begin
    travel_date = :kont_disb_date; -- дата поездки
    departure_place = reg_address; -- место отправления
    counter = 0;
    arrival_place = ''; -- место прибытия
    c_qty = 0;
    select count(*)
    from knclient kncl
    where kncl.kc_amount_approved > 0 and
          kncl.kc_from_k_id = :k_id
    into :c_qty;

    for select adl.adl_address_str
        from knclient kncl
        inner join clients cl on kncl.kc_from_c_id = cl.c_id
        inner join addrclient adcl on adcl.adc_from_c_id = cl.c_id
        inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
        where kncl.kc_amount_approved > 0 and
              kncl.kc_from_k_id = :k_id and
              adcl.adc_address_type = 2 -- адрес фактич. прожив
        into :address_str
    do
    begin
      counter = counter + 1;

      if (counter > 1) then
      begin
        departure_place = '';
        travel_date = null;
      end

      if (counter = c_qty) then
        arrival_place = reg_address;
      else
        arrival_place = '';

      -- из адреса беру все до номера дома, все что после номера дома вырезаю.
      -- Номер дома, квартира, корпус - не отображаю!
      c_addr = '';
      pos = position(', д.' in address_str);

      if (pos > 0) then
      begin
        c_addr = substring(address_str from 1 for pos - 1);
      end
      else
        c_addr = :address_str;

      suspend;
    end
  end
end
