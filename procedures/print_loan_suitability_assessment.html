CREATE PROCEDURE PRINT_LOAN_SUITABILITY_ASSESSMENT(
    K_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME_FIO D_STR_255,
    C_IIN D_STR_20,
    MA_NO D_INTEGER,
    MA_DATE_APP D_DATE,
    ASSESSMENT_DATE D_DATE,
    REQUESTED_AMOUNT D_CURRENCY,
    LOAN_TERM_MONTHS D_INTEGER,
    MONTHLY_PAYMENT D_CURRENCY,
    INTEREST_RATE D_PERCENT,
    IS_PRIMARY_CONTACT D_SMALLINT,
    INCOME_SOURCE_WAGE D_SMALLINT,
    INCOME_SOURCE_SELF_EMPLOYED D_SMALLINT,
    INCOME_SOURCE_RENT D_SMALLINT,
    INCOME_SOURCE_ENTREPRENEUR D_SMALLINT,
    INCOME_SOURCE_PENSION D_SMALLINT,
    INCOME_SOURCE_RELATIVES D_SMALLINT,
    INCOME_SOURCE_OTHER D_SMALLINT,
    AVERAGE_MONTHLY_INCOME D_CURRENCY,
    KDN_VALUE D_CURRENCY,
    KDD_VALUE D_CURRENCY,
    NEW_LOAN_PAYMENT D_CURRENCY,
    EXISTING_OBLIGATIONS_PAYMENT D_CURRENCY,
    CREDIT_BUREAU_CHECKED D_SMALLINT,
    OVERDUE_CHECKED D_SMALLINT,
    ACTIVE_BANK_LOANS_COUNT D_INTEGER,
    ACTIVE_MFO_LOANS_COUNT D_INTEGER,
    OVERDUE_BANK_LOANS_30_COUNT D_INTEGER,
    OVERDUE_MFO_LOANS_1_COUNT D_INTEGER,
    BANKRUPTCY_CHECKED D_SMALLINT,
    SELF_BAN_CHECKED D_SMALLINT,
    ANTIFRAUD_CHECKED D_SMALLINT,
    MILITARY_CHECKED D_SMALLINT,
    IMPACT_ASSESSMENT_DONE D_SMALLINT,
    IMPACT_RESULT D_SMALLINT,
    SOURCE_BORROWER D_SMALLINT,
    SOURCE_CREDIT_BUREAU D_SMALLINT,
    SOURCE_ANTIFRAUD D_SMALLINT,
    SOURCE_GIS D_SMALLINT,
    SOURCE_INTERNAL_DB D_SMALLINT,
    SOURCE_OTHER D_SMALLINT,
    INFO_VERIFICATION_RESULT D_SMALLINT,
    CREDIT_RISK_IDENTIFIED D_SMALLINT,
    BEHAVIORAL_RISK_IDENTIFIED D_SMALLINT,
    FRAUD_RISK_IDENTIFIED D_SMALLINT,
    ADDITIONAL_CHECK_DONE D_SMALLINT,
    FINAL_VERDICT D_SMALLINT,
    FINAL_DECISION D_SMALLINT)
AS
declare variable VAR_MA_ID D_ID;
declare variable VAR_C_ID D_ID;
declare variable VAR_MAC_ID D_ID;
declare variable VAR_CIRCLE D_INTEGER;
declare variable VAR_CA_ID D_ID;
declare variable VAR_KDN_LIMIT D_CURRENCY;
declare variable VAR_IS_GAMBLER D_SMALLINT;
declare variable VAR_IS_ASP D_SMALLINT;
declare variable VAR_MAX_OVERDUE_DAYS D_INTEGER;
declare variable VAR_IS_BANKRUPT D_SMALLINT;
declare variable VAR_IS_STOP_CREDIT D_SMALLINT;
declare variable VAR_STOP_FACTOR D_SMALLINT;
declare variable VAR_ANTIFRAUD_DATE D_DATE;
declare variable VAR_ANTIFRAUD_RESULT D_STR_100;
declare variable VAR_HAS_REFUSAL D_SMALLINT;
declare variable TMP_ID D_ID;
begin
    -- ================================================================
    -- 1. Базовые данные: контракт -> заявка -> клиент
    -- ================================================================
    select
        K.K_FROM_MA_APP_ID,
        KC.KC_FROM_C_ID,
        MA.MA_NO,
        MA.MA_DATE_APP,
        MA.MA_SUMMA_MINUS,
        MA.MA_KOLICH_REPAY,
        MA.MA_PERCENT,
        MAC.MAC_ID
    from KONTRACT K
        join KNCLIENT KC      on KC.KC_FROM_K_ID = K.K_ID and KC.KC_AMOUNT_APPROVED > 0
        join MINIAPP MA       on MA.MA_ID = K.K_FROM_MA_APP_ID and MA.MA_DATE_END = '31.12.9999'
        join MINIAPPCLIENT MAC on MAC.MAC_FROM_MA_ID = MA.MA_ID and MAC.MAC_FROM_C_ID = KC.KC_FROM_C_ID and MAC.MAC_AMOUNT > 0
    where K.K_ID = :K_ID
    into
        :VAR_MA_ID, :VAR_C_ID, :MA_NO, :MA_DATE_APP,
        :REQUESTED_AMOUNT, :LOAN_TERM_MONTHS, :INTEREST_RATE, :VAR_MAC_ID;

    -- Если основной поиск не дал результата (нестандартный кейс: например, кредит
    -- проведён по созаёмщику/поручителю с иным MAC_AMOUNT), пробуем взять переопределения
    -- из ADD_PARAMS.
    if (coalesce(ADD_PARAMS, '') <> '') then
    begin
        if (VAR_MA_ID is null) then
        begin
            tmp_id = f_json_str_get_value(add_params, 'ma_id', null);
            if (tmp_id is not null) then
                VAR_MA_ID = tmp_id;
        end

        if (VAR_C_ID is null) then
        begin
            tmp_id = f_json_str_get_value(add_params, 'c_id', null);
            if (tmp_id is not null) then
                VAR_C_ID = tmp_id;
        end

        if (VAR_MAC_ID is null) then
        begin
            tmp_id = f_json_str_get_value(add_params, 'mac_id', null);
            if (tmp_id is not null) then
                VAR_MAC_ID = tmp_id;
        end

        -- Данные по заявке (номер, дата, сумма/срок/ставка), если не получены основным join'ом
        if (VAR_MA_ID is not null and (MA_NO is null or REQUESTED_AMOUNT is null)) then
        begin
            select MA.MA_NO, MA.MA_DATE_APP, MA.MA_SUMMA_MINUS, MA.MA_KOLICH_REPAY, MA.MA_PERCENT
            from MINIAPP MA
            where MA.MA_ID = :VAR_MA_ID
            into :MA_NO, :MA_DATE_APP, :REQUESTED_AMOUNT, :LOAN_TERM_MONTHS, :INTEREST_RATE;
        end
    end

    if (VAR_MA_ID is null or VAR_C_ID is null or VAR_MAC_ID is null) then
        exception E_MESSAGE 'Данные по контракту, заявке или клиенту не найдены.';

    -- ================================================================
    -- 1. Общие данные / способ проведения оценки (1.2)
    -- ================================================================
    select C_NAME_FIO, C_INN from CLIENTS where C_ID = :VAR_C_ID
    into :C_NAME_FIO, :C_IIN;

    ASSESSMENT_DATE = current_date;

    select SH_ALL_WITH_TAX
    from FM_SCHEDULE_LIST(11, :VAR_MA_ID)
    where SH_ID = 1
    into :MONTHLY_PAYMENT;
    MONTHLY_PAYMENT = coalesce(MONTHLY_PAYMENT, 0);

    -- Первичное / повторное — цикл клиента по текущему контракту (проверенный паттерн, без +1)
    select KC.KC_CIRCLE
    from KNCLIENT KC
    where KC.KC_FROM_K_ID = :K_ID
      and KC.KC_FROM_C_ID = :VAR_C_ID
      and KC.KC_AMOUNT_APPROVED > 0
    into :VAR_CIRCLE;
    IS_PRIMARY_CONTACT = iif(coalesce(VAR_CIRCLE, 1) > 1, 0, 1);

    -- ================================================================
    -- 2. Анализ платежеспособности
    -- ================================================================
    -- Последняя анкета
    select CA_ID, CA_AVR_INCOME
    from CFP_ANKETA
    where CA_OBJECT_KIND = f_objkind('miniappclient', 'id')
      and CA_OBJECT_ID = :VAR_MAC_ID
      and CA_FROM_C_ID = :VAR_C_ID
    order by CA_DATE desc
    rows 1
    into :VAR_CA_ID, :AVERAGE_MONTHLY_INCOME;

    -- Источники дохода. Реализованы: ЗП, пенсия/пособие (+АСП, т.к. отдельной галочки под АСП в
    -- документе нет — см. ТЗ п.2 "пенсия/пособие: пенсионер=Да и/или доход пенсии/АСП>0").
    -- Самозанятость/аренда/предпринимательский/помощь родственников — RFCKZ-924 (пока всегда 0).
    INCOME_SOURCE_WAGE          = 0;
    INCOME_SOURCE_PENSION       = 0;
    INCOME_SOURCE_SELF_EMPLOYED = 0;
    INCOME_SOURCE_RENT          = 0;
    INCOME_SOURCE_ENTREPRENEUR  = 0;
    INCOME_SOURCE_RELATIVES     = 0;
    INCOME_SOURCE_OTHER         = 0;
    VAR_IS_ASP = 0;

    if (VAR_CA_ID is not null) then
    begin
        select
            max(case when CQ.CQ_CODE_NAME in ('contract_salary', 'salary_statement') then iif(coalesce(CAA.CAA_VALUE, 0) <> 0, 1, 0) end),
            max(case when CQ.CQ_CODE_NAME = 'cl_income_pension' then iif(coalesce(CAA.CAA_VALUE, 0) <> 0, 1, 0) end),
            max(case when CQ.CQ_CODE_NAME = 'cl_income_asp' then iif(coalesce(CAA.CAA_VALUE, 0) <> 0, 1, 0) end)
        from CFP_ANKETA_ANSWERS CAA
            join CFP_QUESTIONS CQ on CQ.CQ_ID = CAA.CAA_FROM_CQ_ID
        where CAA.CAA_FROM_CA_ID = :VAR_CA_ID
        into :INCOME_SOURCE_WAGE, :INCOME_SOURCE_PENSION, :VAR_IS_ASP;
    end

    INCOME_SOURCE_WAGE     = coalesce(INCOME_SOURCE_WAGE, 0);
    INCOME_SOURCE_PENSION  = coalesce(INCOME_SOURCE_PENSION, 0);
    VAR_IS_ASP             = coalesce(VAR_IS_ASP, 0);
    AVERAGE_MONTHLY_INCOME = coalesce(AVERAGE_MONTHLY_INCOME, 0);

    -- "пенсия/пособие" = пенсионер и/или доход пенсии/АСП > 0
    INCOME_SOURCE_PENSION = iif(INCOME_SOURCE_PENSION <> 0 or VAR_IS_ASP <> 0, 1, 0);

    -- КДН и КДД (последняя запись)
    select CAKDN_KDN, CAKDN_KDD
    from CFP_ANKETA_KDN
    where CAKDN_FROM_MAC_ID = :VAR_MAC_ID
      and CAKDN_FROM_C_ID   = :VAR_C_ID
    order by CAKDN_CHANGE_DATETIME desc
    rows 1
    into :KDN_VALUE, :KDD_VALUE;
    KDN_VALUE = coalesce(KDN_VALUE, 0);
    KDD_VALUE = coalesce(KDD_VALUE, 0);

    NEW_LOAN_PAYMENT = MONTHLY_PAYMENT;

    -- ================================================================
    -- 3. Кредитная история
    -- ================================================================
    -- Готовые агрегаты распарсенного отчёта ПКБ: платежи по обязательствам, макс. дни просрочки,
    -- банкротство, стоп-кредит, игроман, действующие/просроченные займы по банку и МФО.
    EXISTING_OBLIGATIONS_PAYMENT = 0;
    VAR_MAX_OVERDUE_DAYS = 0;
    VAR_IS_BANKRUPT = 0;
    VAR_IS_STOP_CREDIT = 0;
    VAR_IS_GAMBLER = 0;
    ACTIVE_BANK_LOANS_COUNT = 0;
    ACTIVE_MFO_LOANS_COUNT = 0;
    OVERDUE_BANK_LOANS_30_COUNT = 0;
    OVERDUE_MFO_LOANS_1_COUNT = 0;

    select
        coalesce(PRP_TOTAL_MONTHLY_INS_AMOUNT, 0),
        coalesce(PRP_MAX_OVERDUE_DAYS, 0),
        coalesce(PRP_IS_BANKRUPTCY, 0),
        coalesce(PRP_IS_STOP_CREDIT, 0),
        coalesce(PRP_IS_GAMBLING, 0),
        coalesce(PRP_ACTIVE_BANK_COUNT, 0),
        coalesce(PRP_ACTIVE_MFO_COUNT, 0),
        coalesce(PRP_OVERDUE_COUNT_BANK, 0),
        coalesce(PRP_OVERDUE_COUNT_MFO, 0)
    from PKB_REPORT_PARSED
    where PRP_FROM_MA_ID = :VAR_MA_ID and PRP_FROM_C_ID = :VAR_C_ID
    order by PRP_ID desc
    rows 1
    into :EXISTING_OBLIGATIONS_PAYMENT, :VAR_MAX_OVERDUE_DAYS,
         :VAR_IS_BANKRUPT, :VAR_IS_STOP_CREDIT, :VAR_IS_GAMBLER,
         :ACTIVE_BANK_LOANS_COUNT, :ACTIVE_MFO_LOANS_COUNT,
         :OVERDUE_BANK_LOANS_30_COUNT, :OVERDUE_MFO_LOANS_1_COUNT;

    -- Кредитное бюро / просрочка проверено — получен отчёт pkb_cib_report (EIL_STATUS=2)
    select iif(count(*) > 0, 1, 0)
    from EXTINFOLIST EIL
    join EXTINFOCFG EIC on EIC.EIC_ID = EIL.EIL_FROM_EIC_ID
    where EIL.EIL_OBJECT_KIND = f_objkind('miniapp', 'id')
      and EIL.EIL_OBJECT_ID = :VAR_MA_ID
      and EIL.EIL_FROM_C_ID = :VAR_C_ID
      and EIL.EIL_STATUS = 2
      and EIC.EIC_CODE = 'pkb_cib_report'
    into :CREDIT_BUREAU_CHECKED;
    OVERDUE_CHECKED = CREDIT_BUREAU_CHECKED;

    -- Банкротство проверено (ТЗ 3.3): источник — тот же отчёт ПКБ (TazalauBankruptInfo /
    -- PublicSources.Bankruptcy), а не EIC_CODE='bankruptcy' (тот код — проверка ПОСЛЕ выдачи
    -- займа и на момент формирования Листа физически ещё не существует).
    BANKRUPTCY_CHECKED = CREDIT_BUREAU_CHECKED;

    -- Флаги проверок (EIL_STATUS = 2)
    select iif(count(*) > 0, 1, 0)
    from EXTINFOLIST EIL join EXTINFOCFG EIC on EIC.EIC_ID = EIL.EIL_FROM_EIC_ID
    where EIL.EIL_OBJECT_KIND = f_objkind('miniapp', 'id') and EIL.EIL_OBJECT_ID = :VAR_MA_ID
      and EIL.EIL_FROM_C_ID = :VAR_C_ID and EIL.EIL_STATUS = 2 and EIC.EIC_CODE = 'stop_credit'
    into :SELF_BAN_CHECKED;

    select ANTIFRAUD_CHECK_DATE, ANTIFRAUD_CHECK_RESULT
    from HP_ANTIFRAUD_PRINT_RESULT(f_objkind('kontract', 'id'), :K_ID)
    into :VAR_ANTIFRAUD_DATE, :VAR_ANTIFRAUD_RESULT;
    ANTIFRAUD_CHECKED = iif(VAR_ANTIFRAUD_DATE is not null, 1, 0);

    select iif(count(*) > 0, 1, 0)
    from EXTINFOLIST EIL join EXTINFOCFG EIC on EIC.EIC_ID = EIL.EIL_FROM_EIC_ID
    where EIL.EIL_OBJECT_KIND = f_objkind('miniapp', 'id') and EIL.EIL_OBJECT_ID = :VAR_MA_ID
      and EIL.EIL_FROM_C_ID = :VAR_C_ID and EIL.EIL_STATUS = 2
      and EIC.EIC_CODE in ('military_status', 'military_pocket')
    into :MILITARY_CHECKED;

    -- ================================================================
    -- 4. Оценка влияния
    -- ================================================================
    select iif(count(*) > 0, 1, 0)
    from MINIAPP MA
    join MINIAPPCLIENT MAC2 on MAC2.MAC_FROM_MA_ID = MA.MA_ID
    where MAC2.MAC_FROM_C_ID = :VAR_C_ID
      and MA.MA_IS_ACTIVE < 1
      and MA.MA_DATE_APP >= dateadd(month, -6, current_date)
    into :VAR_HAS_REFUSAL;

    IMPACT_ASSESSMENT_DONE = iif(
        AVERAGE_MONTHLY_INCOME > 0
        and (EXISTING_OBLIGATIONS_PAYMENT > 0 or ACTIVE_BANK_LOANS_COUNT > 0 or ACTIVE_MFO_LOANS_COUNT > 0)
        and NEW_LOAN_PAYMENT > 0
        and VAR_HAS_REFUSAL = 0,
        1, 0);

    VAR_KDN_LIMIT = iif(VAR_IS_GAMBLER = 1 or VAR_IS_ASP = 1, 0.25, 0.5);

    if (IMPACT_ASSESSMENT_DONE = 1) then
    begin
        if (KDN_VALUE > VAR_KDN_LIMIT) then
            IMPACT_RESULT = 3;
        else
            IMPACT_RESULT = 1;
    end
    else
        IMPACT_RESULT = 2;

    -- ================================================================
    -- 5. Достоверность сведений
    -- ================================================================
    SOURCE_CREDIT_BUREAU = 1;
    SOURCE_INTERNAL_DB = 1;
    SOURCE_BORROWER = iif(AVERAGE_MONTHLY_INCOME > 0, 1, 0);
    SOURCE_ANTIFRAUD = ANTIFRAUD_CHECKED;
    SOURCE_GIS = iif(BANKRUPTCY_CHECKED = 1 or SELF_BAN_CHECKED = 1 or MILITARY_CHECKED = 1, 1, 0);
    SOURCE_OTHER = 0;
    INFO_VERIFICATION_RESULT = 1;

    -- ================================================================
    -- 6. Оценка рисков (1 = не выявлен, 2 = выявлен)
    -- ================================================================
    CREDIT_RISK_IDENTIFIED = iif(
        KDN_VALUE > VAR_KDN_LIMIT
        or VAR_MAX_OVERDUE_DAYS > 60
        or VAR_IS_BANKRUPT = 1
        or VAR_IS_STOP_CREDIT = 1,
        2, 1);

    BEHAVIORAL_RISK_IDENTIFIED = 1;                                     -- пока всегда «не выявлен»
    FRAUD_RISK_IDENTIFIED = iif(VAR_ANTIFRAUD_RESULT = 'информация обнаружена', 2, 1);
    ADDITIONAL_CHECK_DONE = 0;                                          -- пока всегда «Нет»

    -- ================================================================
    -- 7. Итоговое заключение
    -- ================================================================
    VAR_STOP_FACTOR = iif(
        KDN_VALUE > VAR_KDN_LIMIT
        or VAR_IS_BANKRUPT = 1
        or VAR_IS_STOP_CREDIT = 1
        or VAR_MAX_OVERDUE_DAYS > 60
        or FRAUD_RISK_IDENTIFIED = 2
        or CREDIT_RISK_IDENTIFIED = 2,
        1, 0);

    if (VAR_STOP_FACTOR = 1) then
    begin
        FINAL_VERDICT = 2;
        FINAL_DECISION = 2;
    end
    else if (IMPACT_RESULT = 2 or INFO_VERIFICATION_RESULT = 2 or ADDITIONAL_CHECK_DONE = 0) then
    begin
        FINAL_VERDICT = 3;
        FINAL_DECISION = 3;
    end
    else
    begin
        FINAL_VERDICT = 1;
        FINAL_DECISION = 1;
    end

    suspend;
end
