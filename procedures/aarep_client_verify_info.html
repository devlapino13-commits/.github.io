CREATE PROCEDURE AAREP_CLIENT_VERIFY_INFO(
    ACL_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    IS_READONLY INTEGER)
RETURNS (
    RES_C_ID DECIMAL(18,0),
    RES_ACL_ID DECIMAL(18,0),
    CV_STATUS_STR VARCHAR(255),
    RES_ACL_MEMO VARCHAR(255),
    RES_ONLY_PHOTO D_SMALLINT)
AS
declare variable acm_memo   varchar(4000);
declare variable acl_status integer;
declare variable json_key   varchar(255);
declare variable json_value varchar(255);
begin
  cv_status_str = '';
  res_acl_memo = '';
  res_only_photo = 0;

  if (acl_id > 0) then
  begin
    select acl_object_id, acl_status, acl_quota_used
    from actionlist
    where acl_id = :acl_id
    into :c_id, acl_status, res_only_photo;
  end
  else
  begin
    if (c_id > 0) then
    begin
      select acl_id, acl_status, acl_quota_used
      from actionlist
      where acl_object_kind = 0 and
            acl_object_id = :c_id and
            acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO'))
      order by acl_id desc
      rows 1
      into :acl_id, acl_status, res_only_photo;
    end
  end
  if (coalesce(acl_id, 0) = 0 and
      coalesce(c_id, 0) = 0) then
    exit;
  if (is_readonly = 0 and
      (not acl_status in (0, 1))) then
    exit;

  res_c_id = c_id;
  res_acl_id = acl_id;

  select acm_memo
  from actionmemo
  where acm_from_acl_id = :acl_id
  into :acm_memo;

  if (coalesce(acm_memo, '') <> '') then
  begin
    for select json_key, json_value
        from fm_json_parse(:acm_memo, 0, 'verify')
        into :json_key, :json_value
    do
    begin
      if (json_key = 'comment') then
        res_acl_memo = json_value;
      else
      begin
        if (json_value = '1') then
          cv_status_str = cv_status_str || upper(json_key) || ';';
      end
    end -- Конец персинга
  end
  suspend;
end
