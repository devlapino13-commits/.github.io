CREATE PROCEDURE CM_RU_0035_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_CLIENT_PASS_NO varchar(30);
declare variable KC_FROM_C_ID ID_TYPE;
declare variable C_CITIZEN integer;
declare variable RD_CODE varchar(255);
begin
  for
  select mac_from_c_id, mac_amount
  from miniappclient
  where mac_from_ma_id = :ma_id
  and mac_amount > 0
  into kc_from_c_id, kc_amount
  do
  begin
    select c_name_fio,
           c_citizen,
           c_pass_no
--           ,c_id
    from clients
    where c_id = :kc_from_c_id
    into res_info_text,
         c_citizen,
         KF_CLIENT_PASS_NO;
    select rd_code from refdata
    where rd_id = :c_citizen
    into rd_code;
    if (rd_code not in ('Citizen.RU','Citizen.KG','Citizen.TJ','Citizen.UZ','Citizen.KZ')) then
    begin
      res_error_text = 'Клиент не является гражданином след. республик: РФ, КГ, ТЖ, УЗ и КЗ';
      res_error_is_critical = 1;
      suspend;
    end
  end
end
