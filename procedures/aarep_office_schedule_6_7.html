CREATE PROCEDURE AAREP_OFFICE_SCHEDULE_6_7(
    REP_REGION_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_100,
    OFFICER_NAME D_STR_100,
    RD_NAME D_STR_20,
    STATUS D_STR_20)
AS
declare variable office_id d_id;
declare variable year_month_integer d_integer;
declare variable reg_office_hours_id d_id;
declare variable rd_id d_id;
begin
  rd_name = 0;
  status = 0;
  officer_name = '';
  reg_name = '';
  year_month_integer = extract(year from :rep_date_end) || iif(extract(month from :rep_date_end) < 10, '0' || extract(month from :rep_date_end), extract(month from :rep_date_end));
  for select iif(stl.stl_lock = 1, 'заблокирован', 'разблокирован'), stl.stl_office_id
      from sa_tabelo_lock stl
      inner join region_list(:rep_region_id, 1, 3, 1) r on r.id_list = stl.stl_office_id
      where stl.stl_year_month = :year_month_integer
      into :status, :office_id
  do
  begin
    select reg.reg_code || ' ' || reg.reg_name, reg2.reg_name, reg.reg_office_hours_id
    from region reg
    inner join region reg2 on reg2.reg_unicode = reg.parent
    where reg.reg_unicode = :office_id and
          current_date between reg.reg_date_open and coalesce(reg.reg_date_close, '31.12.9999') and
          reg2.reg_unicode > 0
    into :officer_name, :reg_name, :reg_office_hours_id;
    select rd.rd_name, sum(rd.rd_id)
    from refdata rd
    where rd.rd_id = :reg_office_hours_id and
          rd.rd_id not in (5277, 5276)
    group by rd.rd_name
    into :rd_name, rd_id;
    suspend;
  end
end
