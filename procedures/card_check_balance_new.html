CREATE PROCEDURE CARD_CHECK_BALANCE_NEW(
    OBJ_KIND D_STR_20,
    OBJ_ID D_ID,
    ON_CLOSING BOOLEAN = false)
RETURNS (
    CS_ID D_INTEGER)
AS
begin
  insert into cardsync(
    cs_from_csc_id,
    cs_card_id,
    cs_from_cca_id,
    cs_object_kind,
    cs_object_id,
    cs_request_json,
    cs_office_id,
    cs_from_k_id,
    cs_from_c_id,
    cs_from_un_id
    )
  select first 1
    f_get_cardsynccfg_id(iif(:on_closing, 'check_balance_on_closing', 'check_balance')),
    cs_card_id,
    cs_from_cca_id,
    9,
    cs_from_k_id,
    f_format('{"card_id": %1}', cs_card_id),
    cs_office_id,
    cs_from_k_id,
    cs_from_c_id,
    cs_from_un_id
  from cardsync
  where (:obj_kind = 'cardsync' and cs_id = :obj_id)
    or (:obj_kind = 'kontract' and cs_from_k_id = :obj_id)
    or (:obj_kind = 'clients' and cs_from_c_id = :obj_id)
    or (:obj_kind = 'cardlist' and cs_card_id = :obj_id)
    or (:obj_kind = 'cardacc' and cs_from_cca_id = :obj_id)
  order by cs_id desc
  returning cs_id
  into :cs_id;

  insert into cardsync_balance (
    csb_cs_id,
    csb_date,
    csb_cca_id,
    csb_loan_balance,
    csb_closing_check
  )
  select
    cs_id,
    cs_date,
    cs_from_cca_id,
    (select amount from hp_balance_cardloans('cardacc', cs_from_cca_id, cs_date)),
    cs_from_csc_id = 23
  from cardsync
  where cs_id = :cs_id;

  suspend;
end
