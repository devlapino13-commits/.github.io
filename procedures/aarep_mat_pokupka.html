CREATE PROCEDURE AAREP_MAT_POKUPKA(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    REP_ML_NAME VARCHAR(255),
    REP_INVENT_STR VARCHAR(50),
    REP_OPER_DATE DATE,
    REP_REGION_NAME VARCHAR(100),
    REP_ACCOUNT VARCHAR(20),
    REP_KOLICH_PLUS DECIMAL(18,4),
    REP_KOLICH_MINUS DECIMAL(18,4),
    REP_SUMMA_PLUS DECIMAL(18,4),
    REP_SUMMA_MINUS DECIMAL(18,4),
    REP_MATOTVET VARCHAR(100),
    REP_WORKPLACE VARCHAR(100),
    REP_CATEGORY VARCHAR(100),
    REP_CATEGORY_PARENT VARCHAR(100),
    REP_CATEGORY_TYPE VARCHAR(100),
    REP_ACCOUNT_NAME VARCHAR(100),
    REP_ACC_BALANCE VARCHAR(20),
    REP_MATOTVET_INFO VARCHAR(100),
    REP_WORKPLACE_INFO VARCHAR(100),
    REP_ACCOUNT_INFO VARCHAR(20),
    REP_BALANCE_NAME VARCHAR(100),
    REP_OPER_KIND VARCHAR(50),
    REP_MODERN DECIMAL(18,4),
    REP_WRITEOFF DECIMAL(18,4),
    OM_BILL_ID DECIMAL(18,0),
    OM_BILL_LINE_NO INTEGER,
    REP_IS_MODERN INTEGER)
AS
declare variable rep_cat_type integer;
declare variable ot_type integer;
declare variable om_matotvet_id_prior decimal(18,0);
declare variable om_matotvet_id decimal(18,0);
declare variable om_workplace_id_prior decimal(18,0);
declare variable om_workplace_id decimal(18,0);
declare variable ol_from_reg_id_prior integer;
declare variable ol_from_reg_id integer;
declare variable ob_from_mc_id_prior decimal(18,0);
declare variable ob_from_mc_id decimal(18,0);
declare variable ob_from_ml_id decimal(18,0);
declare variable om_workplace_info decimal(18,0);
declare variable om_matotvet_info decimal(18,0);
declare variable rep_account_prior varchar(20);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  rep_kolich_plus = 0;
  rep_kolich_minus = 0;
  rep_summa_plus = 0;
  rep_summa_minus = 0;

  om_matotvet_id_prior = 0;
  om_workplace_id_prior = 0;
  ol_from_reg_id_prior = 0;
  ob_from_mc_id_prior = 0;

  /* Приобретенные или списанные за период */
  for select ob_from_ml_id, ob_invent_str, ol_date_oper_day, ol_from_reg_id, om_account, om_kolich_plus,
             om_kolich_minus, om_summa_plus, om_summa_minus, om_account_info, om_oper_type, om_bill_id, ob_bill_line_no,
             om_workplace_info, om_matotvet_info, om_matotvet_id, om_workplace_id, ob_from_mc_id
      from opermat, operlist, operbill, region_list(:report_region_id, 1, -1, 1)
      where om_from_ol_id = ol_id and
            om_oper_type in (1, 3, 6, 7) and
            om_main_line = 1 and
            operlist.is_delete = 0 and
            ol_date_oper_day between :report_date_begin and :report_date_end and
            ol_document_office is null and
            id_list = ol_from_reg_id and
            ob_date_begin <= ol_date and
            ob_date_end >= ol_date and
            ob_bill_id = om_bill_id and
            ob_bill_line_no = om_bill_line_no and
            ob_is_delete = 0
      into :ob_from_ml_id, :rep_invent_str, :rep_oper_date, :ol_from_reg_id, :rep_account, :rep_kolich_plus,
           :rep_kolich_minus, :rep_summa_plus, :rep_summa_minus, :rep_account_info, :ot_type, :om_bill_id,
           :om_bill_line_no, :om_workplace_info, :om_matotvet_info, :om_matotvet_id, :om_workplace_id, :ob_from_mc_id
  do
  begin
    if (om_matotvet_id_prior <> om_matotvet_id) then
    begin
      select c_name_fio
      from clients
      where c_id = :om_matotvet_id
      into :rep_matotvet;
      om_matotvet_id_prior = om_matotvet_id;
    end
    if (om_workplace_id_prior <> om_workplace_id) then
    begin
      select wp_name
      from workplace
      where wp_id = :om_workplace_id
      into :rep_workplace;
      om_workplace_id_prior = om_workplace_id;
    end

    if (ol_from_reg_id_prior <> ol_from_reg_id) then
    begin
      select reg_name
      from region
      where reg_unicode = :ol_from_reg_id
      into :rep_region_name;
      ol_from_reg_id_prior = ol_from_reg_id;
    end
    if (ob_from_mc_id_prior <> ob_from_mc_id) then
    begin
      select m1.mc_name, m2.mc_name, m1.mc_status
      from matcateg m1, matcateg m2
      where m1.mc_id = :ob_from_mc_id and
            m1.parent = m2.mc_id
      into :rep_category, :rep_category_parent, rep_cat_type;
      ob_from_mc_id_prior = ob_from_mc_id;
    end
    if ((rep_account_prior is null) or (rep_account_prior <> rep_account)) then
    begin
      select al_name, al_from_bg_id
      from acc_list
      where al_id = :rep_account
      into :rep_account_name, :rep_acc_balance;

      select bg_name
      from balgroup
      where bg_id = :rep_acc_balance
      into :rep_balance_name;

      rep_account_prior = rep_account;
    end

    rep_matotvet_info = null;
    if (om_matotvet_info is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :om_matotvet_info
      into :rep_matotvet_info;
    end

    select ml_name
    from matlist
    where ml_id = :ob_from_ml_id
    into :rep_ml_name;

    rep_workplace_info = null;
    if (om_workplace_info is not null) then
    begin
      select wp_name
      from workplace
      where wp_id = :om_workplace_info
      into :rep_workplace_info;
    end

    if (ot_type = 1) then
      rep_is_modern = 0;
    else
      rep_is_modern = 1;

    if (rep_cat_type = 0) then
      rep_category_type = 'Основные средства';
    if (rep_cat_type = 1) then
      rep_category_type = 'МБП';
    if (rep_cat_type = 2) then
      rep_category_type = 'Материалы, расходники';
    if (rep_cat_type = 3) then
      rep_category_type = 'Нематеральные активы';

    rep_modern = 0;
    rep_writeoff = 0;

    if (ot_type = 1) then
      rep_oper_kind = 'Покупка, приобретение';
    if (ot_type = 2) then
      rep_oper_kind = 'Приход от производства';
    if (ot_type = 3) then
      rep_oper_kind = 'Перемещение внутри офиса / Выбитие';
    if (ot_type = 4) then
      rep_oper_kind = 'Внутренние списания из баланса';
    if (ot_type = 5) then
      rep_oper_kind = 'Продажа, Передача филиалам (Выбытие)';
    if (ot_type = 6) then
      rep_oper_kind = 'Модернизация, реконструкция';
    if (ot_type = 7) then
      rep_oper_kind = 'Разделение цельного / Объединение';
    if (ot_type = 8) then
      rep_oper_kind = 'Комплектация / Декомплектация';
    if (ot_type = 9) then
      rep_oper_kind = 'Переоценка (доход, убыток)';
    if (ot_type = 10) then
      rep_oper_kind = 'Переоценка за счет амортизации';
    suspend;
  end
end
