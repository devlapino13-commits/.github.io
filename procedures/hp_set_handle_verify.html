CREATE PROCEDURE HP_SET_HANDLE_VERIFY(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable rd_code d_str_255;
declare variable rd_name d_str_255;
declare variable rd_id d_integer;
declare variable db_id d_integer;
declare variable acl_id d_id;
declare variable reg_id d_integer;
declare variable c_verify_status d_integer;
declare variable respons_c_id d_id;
begin
  -- RFCKG-5256 Ручная верификация для иностранных граждан в окне Список клиентов, контрагентов
  db_id = f_get_db_id();
  rd_id = f_get_rd_id('ActionGroup.Passport_Verify');
  res_text = '';

  if (db_id <> 41701) then
    exit;

  select rd.rd_code,
         rd.rd_name,
         c.c_verify_status,
         c.c_from_reg_id
  from clients c
  inner join refdata rd on rd.rd_id = c.c_citizen
  where c.c_id = :c_id
  into rd_code,
       rd_name,
       c_verify_status,
       reg_id;

  if (coalesce(rd_code, '') = '' or coalesce(c_id, 0) = 0 or coalesce(un_id, 0) = 0) then
    exception e_default(f_format('Ошибка при ручной верификации клиента' || ascii_char(13) || 'c_id: %1' || ascii_char(13) || 'rd_code: %2' || ascii_char(13) || 'un_id: %3', coalesce(c_id, 'null'), coalesce(rd_code, 'null'), coalesce(un_id, 'null')));

  if (rd_code = 'Citizen.KG') then
    res_text = f_format('Ручная верификация невозможна для граждан %1.', :rd_name);

  if (c_verify_status = 2) then
    res_text = 'Клиент уже верифицирован.';


  select un.un_from_c_id
  from username un
  where un.un_id = :un_id
  into respons_c_id;

  if (res_text = '') then
  begin
    if (reg_id is null) then
    begin
      select ps_from_reg_id
      from hr_pstatus
      where ps_from_c_id = :c_id and
            ps_date_end >= '31.12.2999' and
            ps_status = 1
      order by ps_date_begin desc rows 1
      into :reg_id;
    end
    if (reg_id is null) then
    begin
      select cs_region_id
      from clientstate
      where cs_from_c_id = :c_id
      into :reg_id;
    end
    if (reg_id is null) then
    begin
      reg_id = f_hq_id();
    end

    insert into actionlist (acl_date,
                            acl_from_reg_id,
                            acl_object_kind,
                            acl_object_id,
                            acl_from_rd_id,
                            acl_status,
                            acl_respons_c_id,
                            acl_from_un_id,
                            acl_dont_show_in_order,
                            acl_quota_used)
    values (current_date, :reg_id, 0, :c_id, :rd_id, 0, :respons_c_id, :un_id, 0, 0)
    returning acl_id
    into :acl_id;

    update actionlist acl
    set acl.acl_status = 2
    where acl.acl_id = :acl_id;
  end
  suspend;
end
