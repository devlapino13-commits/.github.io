CREATE PROCEDURE HP_CASHINVENTORY_UPDATE(
    TM_CIL_ID D_INTEGER,
    TM_CASH_ID D_INTEGER,
    TM_DATE D_DATE,
    TM_OPER_KIND D_INTEGER,
    TM_MANAGER_ID D_ID,
    TM_AMOUNT D_CURRENCY,
    TM_FACT_AMOUNT D_CURRENCY,
    TM_FACT_AMOUNT$BOND D_STR_LONG,
    TM_AKT_PERSON_ID D_ID,
    TM_AKT_AUDIT_ID D_ID,
    TM_AKT_MEMBER_ID D_ID,
    UN_ID D_ID)
AS
declare variable cashier_id d_id;
declare variable res_cil_id d_integer;
declare variable cfl_id d_integer;
declare variable saldo d_currency;
begin
  if (un_id is null) then
    un_id = uxf$user();

  select csp_cashier_id
  from cashplace
  where csp_id = :tm_cash_id
  into cashier_id;

  if (cashier_id is null) then
    exception e_message 'Кассир не определен';

  if (tm_manager_id is null) then
    exception e_message 'Менеджер/контролер не указан';

  if (tm_akt_person_id is null) then
    tm_akt_person_id = cashier_id;

  saldo = f_cash_saldo(tm_cash_id, tm_date, 0);

  if (tm_amount <> round(saldo)) then
    exception e_message 'Изменился остаток в кассе, переоткройте окно.';

  if (coalesce(tm_cil_id, 0) <> 0) then
  begin
    res_cil_id = tm_cil_id;

    update cashinvlist
    set cil_inspector_id = :tm_manager_id,
        cil_inv_amount = :tm_amount,
        cil_from_un_id = :un_id
    where cil_id = :tm_cil_id;
  end else
  begin
    insert into cashinvlist (cil_from_csp_id, cil_date, cil_oper_kind,
                             cil_cashier_id, cil_inspector_id, cil_inv_amount, cil_from_un_id)
    values (:tm_cash_id, :tm_date, :tm_oper_kind,
            :cashier_id, :tm_manager_id, :tm_amount, :un_id)
    returning cil_id
    into :res_cil_id;
  end

  if (tm_akt_person_id is not null) then
    execute procedure hp_cashinvmembers_update(:res_cil_id, '-1', :tm_akt_person_id, 0, :un_id);

  if (tm_akt_audit_id is not null) then
    execute procedure hp_cashinvmembers_update(:res_cil_id, '2', :tm_akt_audit_id, 0, :un_id);

  if (tm_akt_member_id is not null) then
    execute procedure hp_cashinvmembers_update(:res_cil_id, '3', :tm_akt_member_id, 0, :un_id);

  if (tm_fact_amount is null) then
    tm_fact_amount = 0;

  if (tm_fact_amount$bond is null) then
    tm_fact_amount$bond = '';

  update or insert into cashinvlist (cil_parent_id, cil_from_csp_id, cil_date, cil_oper_kind,
                           cil_cashier_id, cil_inspector_id, cil_inv_amount, cil_is_delete, cil_from_un_id)
  values (:res_cil_id, :tm_cash_id, :tm_date, :tm_oper_kind,
          :cashier_id, :tm_manager_id, :tm_fact_amount, 0, :un_id)
  matching (cil_parent_id, cil_is_delete)
  returning cil_id
  into :cfl_id;

  execute procedure hp_cashbond_update(:cfl_id, :tm_fact_amount$bond);
  execute procedure hp_cashinvmembers_update(:cfl_id, '-1', :tm_akt_person_id, 1, :un_id);

  execute procedure hp_cashinventory_book_save(:res_cil_id, :tm_amount - :saldo, :un_id);
end
