CREATE PROCEDURE CM_RU_0043_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_client_pass_no varchar(30);
declare variable kf_id id_type;
declare variable kf_reg_id integer;
declare variable ps_from_c_id id_type;
declare variable kal_new_kf_officer id_type;
declare variable CREL_OTHER_C_ID id_type;
declare variable ps_fio varchar(100);
begin
--  проверка клиенты не является ли родственником сотрдуника офиса
--  вычисляем офис папки
  kf_id = round(:k_id / 100000,0);
  select kal_new_reg_id, kal.kal_new_kf_officer
  from knaktlist kal
  where current_date between kal_date and kal_date_end
  and kal.is_delete = 0
  and kal_from_kf_id = :kf_id
  into kf_reg_id, kal_new_kf_officer;
--  kf_reg_id = 1010300;
-- вытаскиваем всех сотрудников данного офиса
  for
  with tabl1 as (
      select ps_from_c_id
      from hr_pstatus, joblist jl, jobkind, refdata rd
      where current_date between ps_date_begin and ps_date_end
      and ps_from_reg_id = :kf_reg_id
      and ps_is_working = 1
      and ps_status = 1
      and ps_from_jl_id = jl_id
      and jl_kind = jk_id
      and jk_system_id = rd_id
      and rd.rd_main_code containing 'JobSystem'
      and rd.rd_code in('JobSystem.Region_KM'
--                        ,'JobSystem.Specialist_KOBD'
--                        ,'JobSystem.Specialist_SB'
                        ,'JobSystem.LO_Senior'
                        ,'JobSystem.LO_Standart'
                        )
      ),
  tabl2 as (
      select ps_from_c_id, crel_other_c_id
      from tabl1
        left join HP_CLIENT_RELATION_LIST(ps_from_c_id,0) on (1=1) and REL_NAME <> 'Дальн.родств.'
      union
      select :kal_new_kf_officer, kc_from_c_id from knclient
      where kc_from_k_id  = :k_id
      and kc_amount_approved > 0
      )
  select crel_other_c_id,
         max(ps_from_c_id) -- для того чтобы узнать КСника
  from tabl2
  where crel_other_c_id is not null
  group by crel_other_c_id
  having count(*) >= 2 and min(ps_from_c_id) = :kal_new_kf_officer
  into crel_other_c_id, ps_from_c_id
  do
  begin
    select c_name_fio, c_pass_no from clients
    where c_id = :crel_other_c_id
    into res_info_text, KF_CLIENT_PASS_NO;
    select c_name_fio from clients
    where c_id = :ps_from_c_id
    into ps_fio;
    res_error_text = 'Клиент является родственником сотрудника '||ps_fio;
    res_error_is_critical = 1;
    suspend;
  end
end
