CREATE PROCEDURE KIB_CHECK_LIST_TMP(
    STEP_QUERY D_BOOL,
    IS_EMPLOYEE D_BOOL = 0,
    IS_ONLINE D_BOOL = 0)
RETURNS (
    KBM_ID ID_TYPE)
AS
declare variable RD_ID D_INTEGER;
declare variable STR_MIDDLE D_STR_20;
declare variable STR_END D_STR_10;
declare variable KBM_IS_ONLINE D_BOOL;
begin
  is_employee = coalesce(is_employee, 0);
  step_query = coalesce(step_query, 0);
  rd_id = null;

  str_middle = '---';
  if (step_query = 1) then
    str_middle = 'PersonRequestId';
  else
  if (step_query = 2) then
    str_middle = 'CreditHistory';

  str_end = 'Cli';
  if (is_employee = 1) then
    str_end = 'Emp';

  select rd_id
  from refdata rd
  where rd.rd_code = 'KIBMiniappQueries.' || :str_middle || :str_end
  into :rd_id;

  if (rd_id is not null) then
  begin
    for select kbm.kbm_id
        from kibminiapp kbm
        where kbm.kbm_is_checked = 0 and
              kbm.kbm_ver_info = 0 and
              kbm.kbm_from_rd_id = :rd_id
        into :kbm_id
    do
    begin
      kbm_is_online = 0;
      if (exists (select *
           from kibminiapp_online kbo
           where kbo.kbm_id = :kbm_id)) then
      kbm_is_online = 1;

      if (is_online <> kbm_is_online) then
        continue;

      suspend;
    end
  end
end
