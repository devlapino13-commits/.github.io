CREATE PROCEDURE AHPLIST_BOOK_OBOROT(
    REGION_ID INTEGER,
    DATE_1 DATE,
    DATE_2 DATE,
    IS_NOMINAL SMALLINT,
    IS_OBOROT SMALLINT)
RETURNS (
    OBREP_ID VARCHAR(20),
    OBREP_CURRENCY VARCHAR(3),
    OBREP_BEGIN_DT DECIMAL(18,4),
    OBREP_BEGIN_KT DECIMAL(18,4),
    OBREP_OBOROT_DT DECIMAL(18,4),
    OBREP_OBOROT_KT DECIMAL(18,4),
    OBREP_END_DT DECIMAL(18,4),
    OBREP_END_KT DECIMAL(18,4),
    OBREP_LAST_DATE DATE,
    OBREP_BALANCE_ACC VARCHAR(20))
AS
declare variable dt decimal(18,4);
declare variable kt decimal(18,4);
begin
  /* Procedure Text */
  if (is_oborot = 0) then
    date_1 = date_2 + 1;

  for select obr_id, obr_currency, bg_id, sum(obr_begin_dt), sum(obr_begin_kt), sum(obr_oborot_dt), sum(obr_oborot_kt),
             max(obr_last_date)
      from operbook_oborot(:date_1, :date_2, :region_id, :is_nominal, :is_oborot), balgroup, acc_list
      where obr_id = al_id and
            al_from_bg_id = bg_id
      group by obr_id, obr_currency, bg_id
      having(sum(obr_begin_dt) - sum(obr_begin_kt) <> 0) or (sum(obr_oborot_dt) - sum(obr_oborot_kt) <> 0)
      into :obrep_id, :obrep_currency, :obrep_balance_acc, :dt, :kt, :obrep_oborot_dt, :obrep_oborot_kt,
           :obrep_last_date
  do
  begin
    if (dt >= kt) then
    begin
      obrep_begin_dt = dt - kt;
      obrep_begin_kt = 0;
    end
    else
    begin
      obrep_begin_dt = 0;
      obrep_begin_kt = kt - dt;
    end
    if (obrep_begin_dt + obrep_oborot_dt >= obrep_begin_kt + obrep_oborot_kt) then
    begin
      obrep_end_dt = obrep_begin_dt + obrep_oborot_dt - (obrep_begin_kt + obrep_oborot_kt);
      obrep_end_kt = 0;
    end
    else
    begin
      obrep_end_dt = 0;
      obrep_end_kt = obrep_begin_kt + obrep_oborot_kt - (obrep_begin_dt + obrep_oborot_dt);
    end
    if (is_oborot = 0) then
    begin
      obrep_begin_dt = 0;
      obrep_begin_kt = 0;
    end
    suspend;
  end
end
