CREATE PROCEDURE HP_GET_KN_OPERBOOK_OL(
    OL_DATE D_DATE,
    REG_ID D_ID,
    OPER_KIND D_STR_CODE)
RETURNS (
    OL_ID D_ID)
AS
declare variable ol_oper_type d_integer;
declare variable ol_oper_subtype d_integer;
declare variable ol_memo d_str_long;
declare variable un_id d_id;
begin
  if (oper_kind is null) then
    oper_kind = '';

  if (oper_kind = 'calc_perc') then
  begin
    ol_oper_type = 2;
    ol_oper_subtype = 32;
    ol_memo = 'Начисление процентов по займам';
  end
  if (oper_kind = 'calc_reserv') then
  begin
    ol_oper_type = 2;
    ol_oper_subtype = 33;
    ol_memo = 'Начисление резерва по займам';
  end
  if (oper_kind = 'calc_vbs') then
  begin
    ol_oper_type = 2;
    ol_oper_subtype = 34;
    ol_memo = 'Корректировка ВБС по займам';
  end

  if (ol_oper_subtype is null) then
    exception e_default 'Unknown oper subtype for ' || oper_kind;

  select ol_id
  from operlist
  where ol_date = :ol_date and
        ol_from_reg_id = :reg_id and
        ol_oper_type = :ol_oper_type and
        ol_oper_subtype = :ol_oper_subtype and
        is_delete = 0
  --  order by ol_id -- из-за этой строки процесс сильно затормозился, отключен Sanjari, 21.12.2019
  rows 1
  into ol_id;

  if (ol_id is null) then
  begin
    select un_id
    from username
    where un_name = 'FinStart_Robot'
    into un_id;

    insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_memo, ol_parent_id,
                          is_delete, ol_user_id)
    values (:reg_id, :ol_oper_type, :ol_oper_subtype, :ol_date, 9, :ol_memo, null, 0, :un_id)
    returning ol_id
    into :ol_id;
  end
  suspend;
end
