CREATE PROCEDURE PRINT_HRZAYAV(
    ORDER_ID DECIMAL(18,0))
RETURNS (
    LINE_NO INTEGER,
    HRZ_FROM_HO_ID DECIMAL(18,0),
    HRZ_NUMBER INTEGER,
    HRZ_FROM_C_ID DECIMAL(18,0),
    HRZ_DATE_BEGIN DATE,
    HRZ_DATE_END DATE,
    HRZ_DAYS INTEGER,
    HRZ_DAYS_DIFF INTEGER,
    HRZ_DAY_BACK DATE,
    HRZ_DAY_BACK_DIFF DATE,
    HRZ_DIRECTION VARCHAR(100),
    HRZ_CLASS VARCHAR(20),
    C_NAME_FIO VARCHAR(100),
    C_SEX_STR VARCHAR(20),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_EMAIL VARCHAR(100),
    C_KOL_FAMILY INTEGER,
    C_KOL_CHILD INTEGER,
    C_MEMO VARCHAR(1024),
    C_REG_NO VARCHAR(20),
    C_OKPO VARCHAR(20),
    C_INN VARCHAR(20),
    C_ACCOUNT VARCHAR(50),
    C_ORG_FROM_B_ID VARCHAR(10),
    C_HIRED_DATE DATE,
    C_RESIDENT_STR VARCHAR(20),
    C_CELL_PHONE VARCHAR(20),
    C_OTHERS VARCHAR(20),
    C_FULL_ADDRESS VARCHAR(255),
    C_ADDRESS_FACT VARCHAR(255),
    C_STR_NATION VARCHAR(100),
    C_STR_CITIZEN VARCHAR(100),
    C_STR_FAMILY VARCHAR(100),
    C_STR_EDUCATION VARCHAR(100),
    C_STR_ARMY VARCHAR(100),
    C_PASS_KIND VARCHAR(100),
    C_PINCODE VARCHAR(20),
    HRZ_POSITION VARCHAR(300),
    HRZ_POSITION_NAME VARCHAR(300),
    HRZ_COMPANY_NAME VARCHAR(255),
    HRZ_KONT_COMPANY VARCHAR(255),
    HRZ_DATE_DOGOVOR DATE,
    HRZ_CHANGER_FIO VARCHAR(100),
    HRZ_CHANGER_POSITION VARCHAR(300),
    HRZ_CHANGER_C_PASS_NO VARCHAR(30),
    HRZ_CHANGER_C_PASS_DATE DATE,
    HRZ_CHANGER_C_PASS_DATE_EXPIRE DATE,
    HRZ_CHANGER_ADRESS VARCHAR(255),
    HRZ_CHANGER_C_PASS_WHO VARCHAR(255),
    HRZ_DATE_DOGOVOR_LAST DATE,
    HRZ_HO_DATE_DOGOVOR_LAST DATE,
    PSA_AGREEMENT_NO VARCHAR(10),
    HRZ_TABEL_NO VARCHAR(10),
    HRZ_REG_NAME VARCHAR(100),
    REG_NAME_CODE VARCHAR(100),
    HRZ_POSITION_KG D_STR_255,
    HRZ_UNUSED_VAC_YEAR D_SMALLINT,
    HRZ_NURSING_TIME D_STR_CODE,
    HRZ_REASON D_STR_255,
    HRZ_REASON_DATE D_DATE,
    HRZ_HO_SIGNER D_STR_255)
AS
declare variable ho_order_type d_integer;
declare variable ho_order_subtype d_integer;
declare variable ho_signer_id d_integer;
declare variable c_sex smallint;
declare variable c_is_resident smallint;
declare variable st_staff integer;
declare variable st_client integer;
declare variable hrv_changer_id id_type;
declare variable holidays integer;
declare variable hrv_date_begin date;
declare variable hrv_date_end date;
declare variable i d_smallint;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  line_no = 0;
  for select hrv_from_ho_id, hrv_number, hrv_from_c_id, hrv_date_begin, hrv_date_end, hrv_days, hrv_day_back,
             hrv_direction, hrv_class, c1.c_name_fio, c1.c_sex, c1.c_birth_date, c1.c_pass_no, c1.c_pass_date,
             c1.c_pass_who, c1.c_pass_date_expire,
             (select cont_value
              from hp_contdata_get_email_main(0, c1.c_id)), c1.c_kol_family, c1.c_kol_child, c1.c_memo, c1.c_reg_no,
             c1.c_okpo, c1.c_inn, c1.c_account, c1.c_org_from_b_id, c1.c_hired_date, c1.c_is_resident,
             (select cont_value
              from hp_contdata_get_cell_main(0, c1.c_id)), c1.c_others, adl_address_str, d1.rd_name, d2.rd_name,
             d3.rd_name, d4.rd_name, d5.rd_name, pk_name, coalesce(hrv_changer_id, 0), c_pincode, hrv_date_begin, hrv_date_end

      from hr_voctravel, clients c1, refdata d1, refdata d2, refdata d3, refdata d4, refdata d5, passkind, addrlist, addrclient
      where hrv_from_ho_id = :order_id and
            hrv_from_c_id = c1.c_id and
            adc_from_c_id = c1.c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id and
            c_nation = d1.rd_id and
            c_citizen = d2.rd_id and
            c_family = d3.rd_id and
            c_education = d4.rd_id and
            c_army = d5.rd_id and
            c_doc_type = pk_id

      into :hrz_from_ho_id, :hrz_number, :hrz_from_c_id, :hrz_date_begin, :hrz_date_end, :hrz_days, :hrz_day_back,
           :hrz_direction, :hrz_class,

           :c_name_fio, :c_sex, :c_birth_date, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_email,
           :c_kol_family, :c_kol_child, :c_memo, :c_reg_no, :c_okpo, :c_inn, :c_account, :c_org_from_b_id,
           :c_hired_date, :c_is_resident, :c_cell_phone, :c_others, :c_full_address,

           :c_str_nation, :c_str_citizen, :c_str_family, :c_str_education, :c_str_army, :c_pass_kind, hrv_changer_id,
           c_pincode, :hrv_date_begin, :hrv_date_end
  do
  begin
    hrz_changer_fio = 'Не указан';
    hrz_changer_c_pass_no = null;
    hrz_changer_c_pass_date = null;
    hrz_changer_c_pass_date_expire = null;
    hrz_changer_adress = '';
    hrz_changer_position = '';

    select ho_order_type, ho_order_subtype
    from hr_orders
    where ho_id = :order_id 
    into ho_order_type, ho_order_subtype;
  
    if (ho_order_type = 21) then -- компенсация за неиспользованный отпуск (RFCKG-4273)
      hrz_unused_vac_year = extract(year from hrz_date_begin);

    if (ho_order_type = 23 or
        ho_order_type = 22) then -- перерыв на кормление ребенка (RFCKG-4273, RFCKG-4330)
    begin
       select lower(left(saws_name, 50))
       from sa_worksubtype
       where saws_from_sawt_id = :ho_order_type and
             saws_id = :ho_order_subtype
       into hrz_nursing_time;
    end

    if (hrv_changer_id > 0) then --Данные заменяющего сотрудника
    begin
      select c_name_fio, cl.c_pass_no, cl.c_pass_date, cl.c_pass_date_expire, cl.c_pass_who
      from clients cl
      where c_id = :hrv_changer_id
      into hrz_changer_fio, hrz_changer_c_pass_no, hrz_changer_c_pass_date, hrz_changer_c_pass_date_expire,
           hrz_changer_c_pass_who;

      select al.adl_address_str
      from addrclient ac, addrlist al
      where ac.adc_from_c_id = :hrv_changer_id and
            ac.adc_address_type = 1 and
            ac.adc_from_adl_id = al.adl_id
      into hrz_changer_adress;

      select jl_name || ' (' || reg_name || ', ' || dep_name || ')', jl_name_origin
      from hr_pstatus, depart, joblist, region
      where ps_from_c_id = :hrv_changer_id and
            ps_date_begin <= :hrz_date_begin and
            ps_date_end >= :hrz_date_begin and
            ps_from_reg_id = reg_unicode and
            ps_from_dep_id = dep_id and
            ps_from_jl_id = jl_id
      into :hrz_changer_position, :hrz_position_kg;

      hrz_position_kg = coalesce(hrz_position_kg, '');
    end

    if (c_sex = 0) then
      c_sex_str = 'Женский';
    if (c_sex = 1) then
      c_sex_str = 'Мужской';
    if (c_sex = 2) then
      c_sex_str = 'Юр.лицо';

    if (c_is_resident = 0) then
      c_resident_str = 'Не резидент';
    if (c_is_resident = 1) then
      c_resident_str = 'Резидент';

    hrz_position = '';
    hrz_position_name = '';
    select jl_name, jl_name || ' (' || reg_name || ', ' || dep_name || ')', reg_company_name, reg_kont_company,
           reg_name, reg_name || ' ' || reg_code, jl_name_origin
    from hr_pstatus, depart, joblist, region
    where ps_from_c_id = :hrz_from_c_id and
          ps_date_begin <= :hrz_date_begin and
          ps_date_end >= :hrz_date_begin and
          ps_from_reg_id = reg_unicode and
          ps_from_dep_id = dep_id and
          ps_from_jl_id = jl_id
    into :hrz_position_name, :hrz_position, :hrz_company_name, hrz_kont_company, hrz_reg_name, reg_name_code, hrz_position_kg;

    hrz_position_kg = coalesce(hrz_position_kg, '');

    select ps_date_begin
    from hr_pstatus, sa_worktype, hr_orders
    where ps_from_c_id = :hrz_from_c_id and
          ps_from_ho_id = ho_id and
          ho_order_type = sawt_id and
          sawt_base_type = 1
    order by ps_date_begin
    rows 1
    into :hrz_date_dogovor;

    -- RFCKG-4333
    if (db_id in (39801, 41701)) then
    begin
        select ho_reason, ho_reason_date, ho_signer_id
        from hr_orders
        where ho_id = :hrz_from_ho_id
        into :hrz_reason, hrz_reason_date, ho_signer_id;

        hrz_reason = coalesce(hrz_reason, '');

        select ksr_other_2
        from kontsignreg
        where ksr_id = :ho_signer_id
        into hrz_ho_signer;

        if (ho_order_type = 25) then -- Отстранение от работы (RFCKG-4620, RFCKG-4590)
          hrz_reason = 'Кызматтык иликт�©�©нун жыйынтыгы боюнча корутунду /' || ascii_char(10) || 'Заключение по результатам служебного расследования от ';
    end

    hrz_date_dogovor_last = null;
    hrz_tabel_no = null;
    select ps_date_begin, ps_tabel_no, ho_date
    from hr_pstatus, hr_orders
    where ps_status = 1 and
          ps_from_c_id = :hrz_from_c_id and
          ps_from_ho_id = ho_id and
          ho_order_type = 1
    order by ps_date_begin desc
    rows 1
    into hrz_date_dogovor_last, hrz_tabel_no, hrz_ho_date_dogovor_last;

    psa_agreement_no = null;
    select psa_agreement_no
    from hr_pstatus hp, hr_pstatus_agreement hpa
    where hp.ps_from_c_id = :hrz_from_c_id and
          hp.ps_from_ho_id = hpa.psa_from_ho_id and
          psa_agreement_no is not null rows 1
    into psa_agreement_no;

    c_address_fact = '';

    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :hrz_from_c_id and
          adc_address_type = 2 and
          adc_from_adl_id = adl_id
    into :c_address_fact;
    if (c_address_fact is null) then
      c_address_fact = '';
    if (c_address_fact = 'Не указан') then
      c_address_fact = '';

    line_no = line_no + 1;

    select count(*)
    from sa_calendays s
    where s.scd_date >= :hrv_date_begin and
          s.scd_date <= :hrv_date_end and
          extract(weekday from s.scd_date) between 1 and 5 and
          s.scd_work_hours = 0
    into :holidays; -- праздничные дни
    :hrz_days_diff = :hrz_days - :holidays;

    i = 0;
    :hrz_day_back_diff = :hrz_day_back;
    while (i = 0) do
    begin
      if (exists(select *
                 from sa_calendays s
                 where s.scd_date = :hrz_day_back_diff and
                       s.scd_work_hours = 0)) then
        :hrz_day_back_diff = :hrz_day_back_diff + 1; -- выход на работу с учетом праздников и выходных

      else
        i = 1;
    end
    suspend;
  end
end
