CREATE PROCEDURE HP_ACTIVE_TOTAL_BALANCE_KG(
    K_ID D_ID,
    C_ID D_ID)
RETURNS (
    ACTIVE_TOTAL_BALANCE D_CURRENCY)
AS
declare variable ma_id d_integer;
declare variable k_disb_date d_date;
declare variable k_id_active d_id;
declare variable crel_other_c_id d_id;
declare variable oc_rate d_currency;
declare variable cur varchar(3);
declare variable mbf_dept d_currency;
declare variable mbf_dept_tmp d_currency;
declare variable kh_unpaidamount_all_tmp d_currency;
declare variable kh_currency d_str_code;
declare variable kh_unpaidamount_all d_currency;
declare variable kh_unpaidamount_suprug_tmp d_currency;
declare variable kh_unpaidamount_suprug d_currency;
declare variable mbf_dept_suprug_tmp d_currency;
declare variable mbf_dept_suprug d_currency;
begin
  -- возвращает активный совокупный долг клиента и супруга/супруги по КИБу и МБФ

  if (f_get_db_id() not in (39801, 41701)) then
    exception e_message 'Ошибка: hp_active_total_balance_kg используется только в KG';

  select k_from_ma_app_id,
         k_disb_date
  from kontract
  where k_id = :k_id
  into ma_id,
       k_disb_date;

  active_total_balance = 0;
  kh_unpaidamount_all = 0;

  for select sum(coalesce(kh_unpaidamount, 0) + coalesce(kh_unpaidpercenamount, 0)) kh_unpaidamount_all,
             kh_currency
      from hp_kibminiapp_last(:c_id, :ma_id)
      inner join kibhistory on kh_from_kbm_id = kbm_id
      where kh_creditstatus not like '%Отказано%' and
            kh_creditstatus not like '%Рассмотрение заявки%' and
            kh_creditstatus not like 'Погашен%' and
            kh_creditstatus not like 'FULLY_PAID%' and
            :k_disb_date between kh_dateaward and kh_datestop and -- нельзя учитывать сумму кредита, отраженную в отчете КИБ, но закрытого у нас на момент утверждения кредита
            kh_is_debtor = 1 and -- не поручитель
            kh_is_our = 0 -- не МБФ
      group by kh_currency
      into kh_unpaidamount_all_tmp, -- сумма долга в других ФКУ
           kh_currency
  do
  begin
    oc_rate = 1;
    cur = '417';

    if (kh_currency in ('EUR', 'Евро')) then
      cur = '978';
    else if (kh_currency in ('RUB', 'Российский рубль')) then
      cur = '810';
    else if (kh_currency in ('USD', 'Доллар США')) then
      cur = '840';

    if (cur <> 417) then -- если не  Сомы, тогда получаем курс и умножаем на него
    begin
      select cr_rate
      from opercurr
      where current_date between cr_date and cr_date_end and
            cr_from_cu_id = :cur
      into oc_rate;
      kh_unpaidamount_all_tmp = kh_unpaidamount_all_tmp * oc_rate;
    end
    kh_unpaidamount_all = kh_unpaidamount_all + kh_unpaidamount_all_tmp;
  end

  -- получаю активный долг в МБФ
  mbf_dept = 0;
  k_id_active = null;

  for select ks.ks_from_k_id
      from kontstate ks
      inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
      where ks.ks_state = 1 and
            :k_disb_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_ol_is_delete = 0 and
            kncl.kc_from_c_id = :c_id and
            kncl.kc_amount_approved > 0 and
            ks.ks_from_k_id <> :k_id
      into :k_id_active
  do
  begin
    mbf_dept_tmp = 0;
    select sum(zad_os + sh_calc_perc)
    from aarep_make_zad_list(:k_id_active, :k_disb_date, 1)
    into mbf_dept_tmp;
    mbf_dept_tmp = coalesce(mbf_dept_tmp, 0);
    mbf_dept = mbf_dept + mbf_dept_tmp;
  end

  -- вычисляю сумму долга по кибу + в мбф по супругу клиента
  crel_other_c_id = null;
  select rel_other_c_id
  from hp_client_relation_tmp(:c_id, 1)
  where rel_code in (trim('1'), trim('2')) and
        rel_is_active = 1
  into crel_other_c_id;

  if (crel_other_c_id is not null) then
  begin
    kh_unpaidamount_suprug = 0;
    for select sum(coalesce(kh_unpaidamount, 0) + coalesce(kh_unpaidpercenamount, 0)) kh_unpaidamount_all,
               kh_currency
        from hp_kibminiapp_last(:crel_other_c_id, :ma_id)
        inner join kibhistory on kh_from_kbm_id = kbm_id
        where kh_creditstatus not like '%Отказано%' and
              kh_creditstatus not like '%Рассмотрение заявки%' and
              kh_creditstatus not like 'Погашен%' and
              kh_creditstatus not like 'FULLY_PAID%' and
              :k_disb_date between kh_dateaward and kh_datestop and -- нельзя учитывать сумму кредита, отраженную в отчете КИБ, но закрытого у нас на момент утверждения кредита
              kh_is_debtor = 1 and -- заёмщик (не поручитель)
              kh_is_our = 0 -- не МБФ
        group by kh_currency
        into kh_unpaidamount_suprug_tmp, -- сумма долга в других ФКУ
             kh_currency
    do
    begin
      kh_unpaidamount_suprug_tmp = coalesce(kh_unpaidamount_suprug_tmp, 0);
      oc_rate = 1;
      cur = '417';

      if (kh_currency in ('EUR', 'Евро')) then
        cur = '978';
      else if (kh_currency in ('RUB', 'Российский рубль')) then
        cur = '810';
      else if (kh_currency in ('USD', 'Доллар США')) then
        cur = '840';

      if (cur <> 417) then -- если не  Сомы, тогда получаем курс и умножаем на него
      begin
        select oc.cr_rate
        from opercurr oc
        where current_date between oc.cr_date and oc.cr_date_end and
              oc.cr_from_cu_id = :cur
        into :oc_rate;
        kh_unpaidamount_suprug_tmp = kh_unpaidamount_suprug_tmp * oc_rate;
      end
      kh_unpaidamount_suprug = kh_unpaidamount_suprug + kh_unpaidamount_suprug_tmp;
    end

    -- получаю активный долг в МБФ
    mbf_dept_suprug = 0;
    k_id_active = null;
    for select ks.ks_from_k_id
        from kontstate ks
        inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
        where ks.ks_state = 1 and
              :k_disb_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              kncl.kc_from_c_id = :crel_other_c_id and
              ks.ks_from_k_id <> :k_id
        into :k_id_active
    do
    begin
      mbf_dept_suprug_tmp = 0;
      select sum(zad_os + sh_calc_perc)
      from aarep_make_zad_list(:k_id_active, :k_disb_date, 1)
      into :mbf_dept_suprug_tmp;
      mbf_dept_suprug_tmp = coalesce(mbf_dept_suprug_tmp, 0);
      mbf_dept_suprug = mbf_dept_suprug + mbf_dept_suprug_tmp;
    end
  end
  kh_unpaidamount_suprug = coalesce(kh_unpaidamount_suprug, 0);
  mbf_dept_suprug = coalesce(mbf_dept_suprug, 0);

  -- итого активный совокупный баланс (сумма с другими ФКУ + сумма МБФ)
  active_total_balance = kh_unpaidamount_all + kh_unpaidamount_suprug + mbf_dept + mbf_dept_suprug;

  suspend;
end
