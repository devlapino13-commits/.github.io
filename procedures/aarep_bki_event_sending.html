CREATE PROCEDURE AAREP_BKI_EVENT_SENDING(
    REP_DATE_BEGIN D_DATE NOT NULL,
    REP_DATE_END D_DATE NOT NULL,
    LIMIT_DAYS D_INTEGER NOT NULL,
    DATES_ONLY D_BOOL NOT NULL)
RETURNS (
    BKI_NAME D_STR_100,
    EVENT_NAME D_STR_CODE,
    KONTRACT_NO D_STR_20,
    EVENT_DATE D_DATE,
    EVENT_REG_DATE D_DATE,
    STATUS_DATE D_DATE,
    DIFF_DAYS_EVENT D_INTEGER,
    DIFF_DAYS_REG D_INTEGER,
    OBJECT_KIND D_ID,
    OBJECT_ID D_ID,
    ERROR_REASON D_STR_2048)
AS
declare variable e_send_date d_date;
declare variable e_status_date d_date;
declare variable e_bkl_status d_integer;
declare variable e_bkl_id d_id;
declare variable e_bkp_status d_integer;
declare variable send_date d_date;
declare variable event_code d_str_code;
declare variable project_date d_date;
declare variable k_id d_id;
declare variable ma_id d_id;
declare variable bkk_id d_id;
declare variable ma_kind d_integer;
declare variable k_kind d_integer;
begin
  project_date = (select date_begin from bki_kind where bkk_code = 'ru.equifax');

  for
    -- выдача
    select 'Выдача', '2.2_disb', 9, k_id, k_disb_date, cast(ol_datetime as date)
    from kontstate ks, kontract k, operlist ol
    where k.k_id = ks.ks_from_k_id
      and ol_id = k_fact_disb_ol_id
      and ks.ks_ol_is_delete = 0
      and ks.ks_state = 1
      and ks.ks_date_begin between :rep_date_begin and :rep_date_end
  union all
    -- погашение
    select distinct 'Погашение', '2.3_change_deal_info_30days', 9, ok.ok_from_k_id, ok.ok_ol_date, cast(ol_datetime as date)
    from operkred ok, kontract k, operlist ol
    where ok.ok_type = 4
      and ok.ok_ol_is_active = 1
      and ok.ok_ol_date between :rep_date_begin and :rep_date_end
      and k_id = ok_from_k_id
      and ol_id = ok_from_ol_id
      and k_disb_date > :project_date
  union all
    -- погашение списанных
    select distinct 'Погашение', '2.3_change_deal_info_30days', 9, okf.oko_from_k_id, okf.oko_ol_date, cast(ol_datetime as date)
    from operkroff okf, kontract k, operlist ol
    where okf.oko_type = 4
      and okf.oko_ol_is_active = 1
      and okf.oko_ol_date between :rep_date_begin and :rep_date_end
      and k_id = oko_from_k_id
      and ol_id = oko_from_ol_id
      and k_disb_date > :project_date
  union all
    -- списание
    select distinct 'Списание', '2.1_change_contract_conditions', 9, oko_from_k_id, oko_ol_date, cast(ol_datetime as date)
    from operkroff oko, kontstate, oksummary, kontract, operlist
    where oko_ol_date between :rep_date_begin and :rep_date_end
      and oko_type = 1
      and oko_ol_is_active = 1
      and ks_from_k_id = oko_from_k_id
      and k_id = oko_from_k_id
      and ol_id = oko_from_ol_id
      and oko_ol_date between ks_date_begin and ks_date_end
      and ks_state = 2
      and oks_from_k_id = oko_from_k_id
      and oko_ol_date between oks_date_begin and oks_date_end
      and oko_sh_os + oko_sh_perc + oko_other > 0
      and k_disb_date > :project_date
  union all
    -- залоги
    select distinct 'Залог', '2.4_guarantees_change', 9, k_id, coalesce(z_date_begin, k_disb_date), coalesce(z_date_begin, k_disb_date)
    from zalog z, kontract, kontstate ks
    where z_from_k_id = k_id
      and ks_from_k_id = k_id
      and ks_ol_is_delete = 0
      and ks_state = 1
      and ks_date_begin = k_disb_date
      and k_status > 0
      and (k_disb_date between :rep_date_begin and :rep_date_end
          or z.z_date_begin between :rep_date_begin and :rep_date_end)
  union all
    select distinct 'Залог', '2.4_guarantees_change', 9, k_id, z_date_end, z_date_end
    from zalog z, kontract
    where z_from_k_id = k_id
      and k_status > 0
      and z.z_date_end between :rep_date_begin and :rep_date_end
  union all
    select distinct 'Суд', '2.6_change_sud_info', 9, lak_from_k_id, lam_date_begin, cast(lam_datetime as date)
    from lawkredit lk, lawmain lam, kontract
    where lk.lak_from_lam_id = lam.lam_id
      and k_id = lak_from_k_id
      and k_disb_date > :project_date
      and lam.lam_is_active = 1
      and lam.lam_date_begin is not null
      and lam_datetime between :rep_date_begin and :rep_date_end
  into event_name, event_code, object_kind, object_id, event_date, event_reg_date
  do
  begin
    -- стоплист
    if (object_kind = 9 and exists(select bsl_id from bki_stoplist_loans bsl where bsl.bsl_from_k_id = :object_id)) then
      continue;
    -- банкортства
    if (object_kind = 9 and exists(select bbt_date from bki_bankrupt where bbt_from_k_id = :object_id and bbt_date <= :event_date)) then
      continue;

    -- Цессии
    if (object_kind = 9
      and exists(select * from sud_cession_packet, sud_cession_loans
              where scl_from_scp_id = scp_id
                and scl_from_k_id = :object_id
                and scp_status > 0
                and scp_date >= :event_date))
    then
      continue;

    -- сво
    if (object_kind = 9 and object_id in (101010088872901101, 101010088872901301)) then
      continue;

    -- гос. пошлина
    if (object_kind = 9 and
        event_code = '2.3_change_deal_info_30days' and
        not exists(select *
                   from oksummary oks
                   where oks_from_k_id = :object_id and
                         :event_date - 1 between oks.oks_date_begin and oks.oks_date_end and
                         (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc) > 0)) then
      continue;

    k_id = null;
    ma_id = null;
    ma_kind = f_objkind('miniapp', 'id');
    k_kind = f_objkind('kontract', 'id');

    if (object_kind = f_objkind('miniapp', 'id')) then
      select k_id
      from kontract k
      where k.k_from_ma_app_id = :object_id
        and k.k_status >= 0
      order by 1 desc
      rows 1
      into k_id;
    else
    begin
      k_id = object_id;
      ma_id = null;
  
      select k_from_ma_app_id
      from kontract
      where k_id = :k_id
      into ma_id;
    end

    if (event_code = '2.3_change_deal_info_30days') then
    begin
      if (not exists(select * from oksummary oks
                  where oks.oks_from_k_id = :k_id
                    and :event_date between oks_date_begin and oks_date_end))
      then
      begin
        event_code = '2.5_close';
        event_name = 'Закрытие';
      end
    end

    for select bkk_id, bkk_name
    from bki_kind
    where bkk_inactive = 0
    into bkk_id, bki_name
    do
    begin
      if (bkk_id = 2 and not exists(select nkl_id from nbki_kontract_list where nkl_from_k_id = :k_id)) then
        continue;

      status_date = null;
      send_date = null;
      diff_days_event = null;
      diff_days_reg = null;
  
      with t_objects(object_kind, object_id) as(
          select :ma_kind, :ma_id
          from rdb$database
        union
          select :k_kind, k_id
          from kontract k
          where k_from_ma_app_id = :ma_id
      )
      select bkp_datetime, bkp_send_datetime
      from t_objects, bki_loans, bki_packet, bki_clients, bki_events
      where bkp_id = bkl_from_bkp_id
        and bkc_from_bkl_id = bkl_id
        and bev_from_bkc_id = bkc_id
        and bkl_object_kind = object_kind
        and bkl_object_id = object_id
        and bkp_status = 3
        and bkp_from_bkk_id = :bkk_id
        and bev_event_date = :event_date
        and bev_event_id = f_enum('bki_event', :event_code, 'id')
        and bev_status = 3
      order by bkp_datetime       -- дата первой принятой отправки с даты регистрации события
      rows 1
      into status_date, send_date;

      if (bkk_id = 2) then
         send_date = status_date;

      if (send_date is not null) then
      begin
        diff_days_event = send_date - event_date;
        diff_days_reg = send_date - event_reg_date;
      end
  
      if (diff_days_event is not null and diff_days_event <= limit_days) then   -- в дату события или позволительная разница
        continue;
  
      if (dates_only = 0) then
      begin
        kontract_no = iif(object_kind = 9, (select k_kontract_no from kontract where k_id = :object_id), object_id);
    
        -- выясняем почему не отправлено ранее
        error_reason = '';
    
        -- если дата регистрации события в БД отличается от даты события
        if (event_reg_date <> event_date) then
          error_reason = error_reason || 'Событие зарегистрировано позже самого события. ';
    
        -- если дата отправки отличается от даты статуса (пакет обработан БКИ позже чем отправлен)
        if (status_date > send_date) then
          error_reason = error_reason || f_format('Обработано (%1) позже чем отправлено (%2). ', f_datetostr(status_date), f_datetostr(send_date));
    
        -- если отправка была на следующий день после даты регистрации события, то всё ок, иначе надо дальше проверять
        if (status_date is null or cast(coalesce(send_date, status_date) as date) - event_reg_date > 1) then
        begin
          -- смотрим была ли попытка отправки
          e_status_date = null;
          e_send_date = null;
          e_bkl_status = null;
          e_bkl_id = null;
          e_bkp_status = null;
    
          -- ищем первый пакет на дату события
          with t_objects(object_kind, object_id) as(
              select :ma_kind, :ma_id
              from rdb$database
            union
              select :k_kind, k_id
              from kontract k
              where k_from_ma_app_id = :ma_id
          )
          select bkp_datetime, bkp_send_datetime, bev_status, bkl_id, bkp_status
          from t_objects, bki_loans, bki_packet, bki_clients, bki_events
          where bkp_id = bkl_from_bkp_id
            and bkc_from_bkl_id = bkl_id
            and bev_from_bkc_id = bkc_id
            and bkl_object_kind = object_kind
            and bkl_object_id = object_id
            and bkp_from_bkk_id = :bkk_id
            and bkp_from_bkt_id < 5
            and bev_event_date = :event_date
            and bev_event_id = f_enum('bki_event', :event_code, 'id')
            and coalesce(bkp_send_datetime, bkp_datetime) between :event_reg_date and coalesce(:send_date, :status_date, current_date) + 1  -- после регистрации в БД и до корректной отправки
          order by bkp_datetime desc
          rows 1
          into e_status_date, e_send_date, e_bkl_status, e_bkl_id, e_bkp_status;
    
          -- если на дату события не было попытки отправить
          if (coalesce(e_send_date, e_status_date) is null) then
          begin
            if (event_reg_date < current_date - 1) then
              error_reason = error_reason || 'Событие не попало в модуль БКИ. ';
          end
          else
          begin
            if (e_bkp_status < 2) then
              error_reason = error_reason || 'Технический сбой. ';
            else if (e_bkp_status = 2) then
              error_reason = error_reason || 'Отправлено. Не получен ответ от БКИ. ';
            else if (e_bkl_status = -3) then
              error_reason = error_reason || 'Событие не существовало на момент отправки. ';
            else if (e_bkl_status = -2) then
              error_reason = error_reason || 'Дублирование информации (внутренняя проверка). ';
            else if (e_bkl_status = -1) then
              error_reason = error_reason || 'Ошибка БКИ: ' ||
                          coalesce((select list(f_memo_text(bke_error_id))
                          from bki_clients, bki_events, bki_errors
                          where bkc_from_bkl_id = :e_bkl_id
                            and bev_from_bkc_id = bkc_id
                            and bke_from_bev_id = bev_id),'') || ' ';
          end
    
          -- если отправки не было, проверим буфер
          if (status_date is null) then
          begin
            if (exists(select bkb_id
                      from bki_buffer
                      where bkb_object_kind = :object_kind
                        and bkb_object_id = :object_id
                        and bkb_date = :event_date
                        and bkb_from_bkk_id = 1
                        and bkb_type = 0
            )) then
            begin
              error_reason = error_reason || 'Отправка запланирована.';
              error_reason = replace(error_reason, 'Событие не попало в модуль БКИ. ', '');
            end
          end
        end
      end
      suspend;
    end
  end
end
