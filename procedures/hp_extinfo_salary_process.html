CREATE PROCEDURE HP_EXTINFO_SALARY_PROCESS(
    EIL_ID D_ID)
AS
declare variable BLS_ID D_ID;
declare variable EIL_STATUS D_SMALLINT;
declare variable JSON_STR D_BLOB_TEXT;
declare variable ERROR_STATE D_STR_255;
declare variable C_ID D_ID;
declare variable UN_ID D_ID;
declare variable CONFIG D_STR_CODE;
declare variable COUNTER D_INTEGER;
declare variable SUMMA D_CURRENCY;
declare variable MAC_ID D_ID;
declare variable PAYMENT_LIST_LENGTH D_INTEGER;
declare variable PAYMENT_TMP D_CURRENCY;
declare variable I D_INTEGER;
begin
-- RFCKG-4255 обработка ответа
  select eil.eil_from_bls_id,
         eil.eil_status
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eil.eil_id = :eil_id and
        eic.eic_code = 'salary_check'
  into bls_id,
       eil_status;

  select icq.icq_from_c_id,
         icq.icq_from_un_id,
         icq.icq_config,
         mac.mac_id
  from income_check_querys icq
  inner join miniappclient mac on mac.mac_id = icq.icq_from_mac_id
  inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
  where icq.icq_from_eil_id = :eil_id
  order by icq.icq_create_datetime desc
  rows 1
  into c_id,
       un_id,
       config,
       mac_id;

  if (eil_status = 2) then
  begin

    select cast(blx_blob as d_blob_text)
    from hp_get_blobex_blob(:bls_id)
    into json_str;

    if (row_count is null or coalesce(json_str, '') = '') then
      exit;

    error_state = cast(f_json_blob_get_value(json_str, 'baseResponse.responseCode', '') as d_str_255);

    if (error_state in ('PROVIDER_ERROR', 'INTERNAL_ERROR', 'EMPTY_REPORT', 'INACTIVE', 'INVALID', 'NOT_FOUND', 'ERROR_TV_NOTRETRIEVED')) then
    begin

      select count(icq.icq_id)
      from income_check_querys icq
      where icq.icq_status = -1 and
            icq.icq_from_c_id = :c_id
      into counter;

      update income_check_querys
      set icq_status = -1
      where icq_from_eil_id = :eil_id;

      if (counter < 1) then    -- Если первый запрос вернул ошибку то автоматически отправляем второй запрос (Пункт 3)
        execute procedure hp_income_new_query(:mac_id, :un_id, :config);

    end
    else if (error_state in ('PAYMENTS_NOT_FOUND', 'MISSING_INPUT_PARAMETERS')) then
    begin
      update income_check_querys
        set icq_status = -2
      where icq_from_eil_id = :eil_id;
    end
    else if (error_state = 'FOUND') then       -- тут условия успешногоответа
    begin
      -- Узнаем среднее значение ЗП
      i = 0;
      payment_tmp = 0;
      summa = 0;
      payment_list_length = f_json_blob_get_count(json_str, 'baseResponse.deductionDetailed') - 1;

      if (payment_list_length >= 0) then -- если это массив
      begin
        while (i <= payment_list_length) do
        begin
          payment_tmp = cast(f_json_blob_get_value(json_str, f_format('baseResponse.deductionDetailed[%1].paymentSum', i), 0) as d_currency);

          summa = summa + payment_tmp;
          i = i + 1;
        end

        if (i > 0) then
          summa = cast((summa / i) as d_currency);
      end
      else
        summa = cast(f_json_blob_get_value(json_str, 'baseResponse.deductionDetailed.paymentSum', 0) as d_currency);

      -- Записываем результат
      if (summa <= 0) then
      begin
        update income_check_querys
           set icq_status = -2
        where icq_from_eil_id = :eil_id;
      end
      else
      begin
        update income_check_querys icq
           set icq.icq_status = 2,
             icq.icq_payment_sum = :summa
        where icq_from_eil_id = :eil_id;
      end
    end
  end

  if (eil_status in (-2, -3)) then
  begin
    update income_check_querys icq
       set icq.icq_status = -2
    where icq_from_eil_id = :eil_id;
  end
end
