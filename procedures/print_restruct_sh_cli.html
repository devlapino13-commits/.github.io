CREATE PROCEDURE PRINT_RESTRUCT_SH_CLI(
    PRINT_K_ID DECIMAL(18,0))
RETURNS (
    C_NAME_FIO VARCHAR(255),
    SH_FROM_K_ID ID_TYPE,
    SH_NUMBER INTEGER,
    SH_DATE DATE,
    SHCL_OS DECIMAL(18,4),
    SHCL_PERC DECIMAL(18,4),
    SHCL_PERC_TAX DECIMAL(18,4),
    SHCL_TOTAL DECIMAL(18,4),
    IS_HEADER SMALLINT,
    SHCL_FULL_TOTAL DECIMAL(18,4))
AS
declare variable sh_dd date;
declare variable k_amount_approved decimal(18,4);
declare variable kc_amount_approved decimal(18,4);
declare variable prop_perc decimal(18,7);
declare variable sh_balance_before_main decimal(18,4);
declare variable sh_balance_after_main decimal(18,4);
declare variable c_name_fio_temp varchar(255);
declare variable c_id decimal(18,0);
declare variable shcl_os_temp decimal(18,4);
declare variable shcl_perc_temp decimal(18,4);
declare variable shcl_perc_tax_temp decimal(18,4);
declare variable sh_id integer;
declare variable cm_is_client_mode integer;
begin
-- operkred_fix

  select cm_is_client_mode
  from company
  into cm_is_client_mode;

  if (cm_is_client_mode = 0) then
  begin
    sh_from_k_id = print_k_id;

    k_amount_approved = 0;
    select kont.k_amount_approved
    from kontract kont
    where kont.k_id = :print_k_id
    into :k_amount_approved;

    c_name_fio_temp = '';
    for select cl.c_name_fio, kncl.kc_amount_approved, cl.c_id
        from knclient kncl
        inner join clients cl on cl.c_id = kncl.kc_from_c_id
        where kncl.kc_from_k_id = :print_k_id and
              kncl.kc_amount_approved > 0
        into :c_name_fio, :kc_amount_approved, :c_id
    do
    begin
      shcl_full_total = 0;
      prop_perc = 0;
      shcl_os_temp = 0;
      shcl_perc_temp = 0;
      shcl_perc_tax_temp = 0;

      for select ok_sh_id, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
          from operkred
          where ok_from_k_id = :print_k_id and
                ok_ol_is_active = 1 and
                ok_type < 7 and
                ok_from_c_id = 0
          group by ok_sh_id
          having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
          order by ok_sh_id
          into :sh_id, :shcl_os_temp, :shcl_perc_temp
      do
      begin
        prop_perc = kc_amount_approved * 100 / k_amount_approved;
        shcl_os_temp = round(shcl_os_temp * prop_perc / 100, 2);
        shcl_perc_temp = round(shcl_perc_temp * prop_perc / 100, 2);
        shcl_perc_tax_temp = round(shcl_perc_tax_temp * prop_perc / 100, 2);
        shcl_full_total = shcl_full_total + shcl_os_temp + shcl_perc_temp + shcl_perc_tax_temp;
      end
      if (shcl_full_total <> 0) then
      begin
        for select sh_number, sh_date, shcl_os, shcl_perc, shcl_perc_tax, shcl_total, is_header
            from print_restruct_sh_cli_tmp(:print_k_id, :c_id)
            into :sh_number, :sh_date, :shcl_os, :shcl_perc, :shcl_perc_tax, :shcl_total, :is_header
        do
        begin
          suspend;
        end
      end
    end
    exit;
  end

  sh_from_k_id = print_k_id;

  k_amount_approved = 0;
  select kont.k_amount_approved
  from kontract kont
  where kont.k_id = :print_k_id
  into :k_amount_approved;

  c_name_fio_temp = '';
  for select cl.c_name_fio, kncl.kc_amount_approved, cl.c_id
      from knclient kncl
      inner join clients cl on cl.c_id = kncl.kc_from_c_id
      where kncl.kc_from_k_id = :print_k_id and
            kncl.kc_amount_approved > 0
      into :c_name_fio, :kc_amount_approved, :c_id
  do
  begin
    shcl_full_total = 0;
    prop_perc = 0;
    shcl_os_temp = 0;
    shcl_perc_temp = 0;
    shcl_perc_tax_temp = 0;

    for select ok_sh_id, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_from_k_id = :print_k_id and
              ok_ol_is_active = 1 and
              ok_type < 7 and
              ok_from_c_id = :c_id
        group by ok_sh_id
        having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
        order by ok_sh_id desc
        rows 1
        into :sh_id, :shcl_os_temp, :shcl_perc_temp
    do
    begin
      prop_perc = kc_amount_approved * 100 / k_amount_approved;
--        SHCL_OS_TEMP = round(SHCL_OS_TEMP * PROP_PERC/100,2);
--        SHCL_PERC_TEMP = round(SHCL_PERC_TEMP * PROP_PERC/100,2);
--        SHCL_PERC_TAX_TEMP = round(SHCL_PERC_TAX_TEMP * PROP_PERC/100,2);
      shcl_full_total = shcl_full_total + shcl_os_temp + shcl_perc_temp + shcl_perc_tax_temp;
    end

    if (shcl_full_total <> 0) then
    begin
      for select sh_number, sh_date, shcl_os, shcl_perc, shcl_perc_tax, shcl_total, is_header
          from print_restruct_sh_cli_tmp(:print_k_id, :c_id)
          into :sh_number, :sh_date, :shcl_os, :shcl_perc, :shcl_perc_tax, :shcl_total, :is_header
      do
      begin
        suspend;
      end
    end
  end

end
