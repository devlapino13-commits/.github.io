CREATE PROCEDURE UXP$UXFRLOOKUP_SOURCES_TMP(
    UFP_UUID D_UUID)
RETURNS (
    ID D_STR_CODE,
    TEXT D_STR_255,
    LINKED D_BOOL)
AS
declare variable uf_source d_str_code;
declare variable field_name d_str_code;
declare variable source_name d_str_code;
begin
  select us_code, uf_field_name
  from ux$frparam, ux$field, ux$source
  where ufp_uuid = :ufp_uuid and
        ufp_uf_uuid = uf_uuid and
        uf_owner = us_uuid
  into uf_source, field_name;

  linked = 1;
  for select us_code, iif(us_code = us_name, us_code, us_code || '    - ' || us_name)
  from ux$index, ux$source
  where uix_source = :uf_source and
        uix_fields = :field_name and
        uix_fk_source = us_code and
        coalesce(us_lookup_text, '') <> ''
  into id, text
  do
  begin
    suspend;

    source_name = id;

    for select us_code, iif(us_code = us_name, us_code, us_code || '    - ' || us_name)
    from ux$source
    where us_main_table = :source_name and
          coalesce(us_lookup_text, '') <> '' and
          us_kind <> -1 -- кроме SQL
    into id, text
    do
      suspend;
  end

  linked = 0;

  for select us_code, iif(us_code = us_name, us_code, us_code || '    - ' || us_name)
  from ux$source
  where us_is_table = 0 and    
        us_kind <> -1 and
        coalesce(us_lookup_id, '') <> '' and
        coalesce(us_lookup_text, '') <> '' and
        (us_code starting 'usp$' or us_code not containing '$')
  into :id, :text
  do
    suspend;
end
