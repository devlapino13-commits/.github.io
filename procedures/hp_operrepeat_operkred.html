CREATE PROCEDURE HP_OPERREPEAT_OPERKRED(
    ID D_ID)
RETURNS (
    REC_UNIQ D_STR_CODE,
    OK_OL_DATE D_DATE,
    OK_SH_ID D_INTEGER,
    OK_SH_DATE D_DATE,
    OK_SH_OS D_CURRENCY,
    OK_SH_PERC D_CURRENCY,
    OK_TYPE D_STR_CODE,
    OK_TOTAL D_CURRENCY,
    OK_OS D_CURRENCY,
    OK_PERC D_CURRENCY,
    OK_FINE D_CURRENCY,
    OK_KOMMIS D_CURRENCY,
    OK_KOMMIS_ADV D_CURRENCY,
    OK_TAX D_CURRENCY,
    OK_FROM_OL_ID D_STR_CODE,
    OK_NUMBER D_INTEGER)
AS
declare variable res_uniq d_str_long;
begin
  for select ok_from_ol_id, ok_number, ok_sh_id, ok_sh_date, ot_name, ok_os, ok_perc, ok_fine,
             ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date
      from operkred, kroper
      where ok_from_k_id = :id and
            ok_ol_is_active = 1 and
            ok_type = ot_id
      order by ok_ol_date, ok_from_ol_id, ok_number
      into :ok_from_ol_id, :ok_number, :ok_sh_id, :ok_sh_date, :ok_type, :ok_os, :ok_perc, :ok_fine,
           :ok_kommis, :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax, :ok_ol_date
  do
  begin
    ok_total = ok_os + ok_perc + ok_fine + ok_kommis + ok_kommis_adv + ok_tax;

    res_uniq = ok_from_ol_id || ok_number || ok_sh_id || ok_sh_date || ok_type || ok_os || ok_perc ||
               ok_fine || ok_kommis || ok_sh_os || ok_sh_perc || ok_kommis_adv || ok_tax || ok_ol_date;

    rec_uniq = f_md5(res_uniq);
    suspend;
  end
end
