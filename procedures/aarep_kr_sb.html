CREATE PROCEDURE AAREP_KR_SB(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_K_ID ID_TYPE,
    AA_CURR_LEVEL_NAME STR_100,
    AA_CURR_BRANCH_NAME STR_100,
    AA_CURR_OFFICE_NAME STR_100,
    AA_KF_NAME STR_100,
    AA_KONTRACT_NO STR_100,
    AA_SB_TEAM_NAME STR_100,
    AA_SB_NAME STR_100,
    SBA_DATE_BEGIN DATE,
    AA_KOLICH_CL_ACTIVE_BEGIN INTEGER,
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    AA_KOLICH_CLOSED INTEGER,
    AA_CURR_KC_FIO STR_100,
    K_CASHIER_FIO STR_100,
    AAREP_LEADER_FIO STR_100,
    AA_LOAN_AMOUNT CURR_TYPE,
    SBA_DISB_DATE DATE,
    AA_PRODUCT_NAME STR_100,
    AA_LOAN_CURRENCY VARCHAR(30),
    AA_LOAN_KIND VARCHAR(30),
    AA_DELIQ_REPAY_KOLICH INTEGER,
    AA_DELIQ_DAYS INTEGER,
    AA_DELIQ_BALANCE CURR_TYPE,
    AA_DELIQ_OS_ALL CURR_TYPE,
    AA_DELIQ_PERC_ALL CURR_TYPE,
    OKO_BALANCE_OTHER CURR_TYPE,
    OKO_BALANCE_TOTAL CURR_TYPE,
    OKO_WRITEOFF_OTHER_PER CURR_TYPE,
    AA_REPAY_WRITEOFF_FINE CURR_TYPE,
    AA_REPAY_WRITEOFF_PERC CURR_TYPE,
    AA_REPAY_WRITEOFF_OS CURR_TYPE,
    AA_REPAY_ALL_OS CURR_TYPE,
    AA_REPAY_ALL_PERC CURR_TYPE,
    AA_REPAY_ALL_KOMMIS CURR_TYPE,
    AA_REPAY_ALL_KOMMIS_ADV CURR_TYPE,
    AA_REPAY_ALL_FINE CURR_TYPE,
    OKO_PAYED_OTHER CURR_TYPE,
    AA_REPAY_ALL CURR_TYPE,
    AA_STATUS_TEXT STR_100,
    AA_AMOUNT_ACTIVE_END CURR_TYPE,
    AA_AMOUNT_ACTIVE_BEGIN CURR_TYPE,
    AA_CURR_KC_SEX STR_20,
    AA_DELIQ_BEG_CL_KOLICH CURR_TYPE,
    AA_DELIQ_BEG_BALANCE CURR_TYPE,
    SBL_TYPE_STR STR_100,
    OKO_PAYED_OS_PER CURR_TYPE,
    OKO_PAYED_PERC_PER CURR_TYPE,
    OKO_PAYED_OTHER_PER CURR_TYPE,
    OKO_PAYED_TOTAL_PER CURR_TYPE,
    TYPE_STR VARCHAR(20),
    PR_IS_INDIV INTEGER)
AS
declare variable emp integer;
declare variable paral integer;
declare variable pr_code varchar(100);
declare variable client_kolich integer;
begin
  for select aa_k_id, aa_curr_level_name, aa_curr_branch_name, aa_curr_office_name, null sba_kf_name, aa_kontract_no,
             aa_sb_team_name, aa_sb_name, sba_date_begin, aa_kolich_cl_active_begin, aa_kolich_cl_active_end,
             aa_kolich_closed, aa_curr_kc_fio, k_cashier_fio, null aarep_leader_fio, aa_loan_amount,
             dd_disb_date sba_disb_date, aa_product_name, aa_loan_currency, aa_loan_kind, aa_deliq_repay_kolich,
             aa_deliq_days, aa_deliq_balance, aa_deliq_os_all, aa_deliq_perc_all, 0 oko_balance_other, --Прочих задолженностей в активнем кредите нет
             aa_deliq_os_all + aa_deliq_perc_all oko_balance_total, null oko_writeoff_other_per, aa_repay_writeoff_fine,
             aa_repay_writeoff_perc, aa_repay_writeoff_os, aa_repay_all_os, aa_repay_all_perc, aa_repay_all_kommis,
             aa_repay_all_kommis_adv, aa_repay_all_fine, 0 oko_payed_other,

                          (aa_repay_all_os + aa_repay_all_perc + aa_repay_all_kommis + aa_repay_all_kommis_adv + aa_repay_all_fine) aa_repay_all,
             aa_status_text, aa_amount_active_end, aa_amount_active_begin, aa_curr_kc_sex, aa_deliq_beg_cl_kolich,
             aa_deliq_beg_balance, aa_repay_fact_period_os oko_payed_os_per, 0 oko_payed_perc_per,
             0 oko_payed_other_per, 0 oko_payed_total_per, 'Коллекторы' sbl_type_str, --в аналитике выборка только по коллекторам
             'Системный учет'
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, 'PROBLEM2', 0, 0, 1, 1, 0, 0, 0)
      union all
      select sba_from_k_id, aa_curr_branch_name, aa_curr_branch_name, sba_region_name, sba_kf_name, sba_kontract_no,
             sbl_team_name, sbl_name, sba_date_begin, aarep_clients_kol, aarep_clients_kol, aarep_clients_kol,
             sba_lo_fio, aarep_cashier_fio, aarep_leader_fio, sba_amount_approved, sba_disb_date, null aa_product_name, --Инфо берем после этого блока здесь нулл
             null aa_loan_currency, --Инфо берем после этого блока здесь нулл
             null aa_loan_kind, --Инфо берем после этого блока здесь нулл
             1 aa_deliq_repay_kolich, --Внесистемный учет не рассматривает количество погашении по количеству, только один
             null aa_deliq_days, --Не учитывает просроченные дни в внесистем учете
             oko_balance_os, oko_balance_os, oko_balance_perc, oko_balance_other, oko_balance_total,
             oko_writeoff_other_per, 0 aa_repay_writeoff_fine, --в внесимтемном учете не списывают пеню
             oko_writeoff_perc_per aa_repay_writeoff_perc, oko_writeoff_os_per aa_repay_writeoff_os, oko_payed_os,
             oko_payed_perc, 0 aa_repay_all_kommis, --Это сумма в OKO_PAYED_OTHER
             0 aa_repay_all_kommis_adv, --Это сумма в OKO_PAYED_OTHER
             0 aa_repay_all_fine, --Это сумма в OKO_PAYED_OTHER
             oko_payed_other, oko_payed_total, 'Закрытый кредит' aa_status_text, --В внесистемном учете только закрытые контракты
             oko_balance_os aa_amount_active_end, oko_balance_os + oko_payed_os aa_amount_active_begin, aa_curr_kc_sex,
             aarep_clients_kol aa_deliq_beg_cl_kolich, oko_balance_os + oko_payed_os aa_deliq_beg_balance,
             oko_payed_os_per, oko_payed_perc_per, oko_payed_other_per, oko_payed_total_per, sbl_type_str,
             'Внесистемный учет'
      from aarep_kroff_sb(:rep_region_id, :rep_date_begin, :rep_date_end)
      into aa_k_id, aa_curr_level_name, aa_curr_branch_name, aa_curr_office_name, aa_kf_name, aa_kontract_no,
           aa_sb_team_name, aa_sb_name, sba_date_begin, aa_kolich_cl_active_begin, aa_kolich_cl_active_end,
           aa_kolich_closed, aa_curr_kc_fio, k_cashier_fio, aarep_leader_fio, aa_loan_amount, sba_disb_date,
           aa_product_name, aa_loan_currency, aa_loan_kind, aa_deliq_repay_kolich, aa_deliq_days, aa_deliq_balance,
           aa_deliq_os_all, aa_deliq_perc_all, oko_balance_other, oko_balance_total, oko_writeoff_other_per,
           aa_repay_writeoff_fine, aa_repay_writeoff_perc, aa_repay_writeoff_os, aa_repay_all_os, aa_repay_all_perc,
           aa_repay_all_kommis, aa_repay_all_kommis_adv, aa_repay_all_fine, oko_payed_other, aa_repay_all,
           aa_status_text, aa_amount_active_end, aa_amount_active_begin, aa_curr_kc_sex, aa_deliq_beg_cl_kolich,
           aa_deliq_beg_balance, oko_payed_os_per, oko_payed_perc_per, oko_payed_other_per, oko_payed_total_per,
           sbl_type_str, type_str
  do
  begin
    if (aa_kf_name is null) then
    begin
      select kal.kal_new_kf_name
      from kontract, knaktlist kal
      where k_id = :aa_k_id and
            k_from_kf_id = kal.kal_from_kf_id and
            kal.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end
      into aa_kf_name;
    end

    if (aarep_leader_fio is null) then
    begin
      select c_name_fio
      from kontract, clients cl
      where k_id = :aa_k_id and
            k_clients_leader_id = c_id
      into aarep_leader_fio;
    end

    if (aa_product_name is null) then
    begin
      select pr_name, pr_loan_type, pr_is_employer, pr_category, pr_code, kt.k_kolich_client_approved
      from kontract kt, product pr
      where k_id = :aa_k_id and
            kt.k_from_pr_id = pr.pr_id
      into aa_product_name, :pr_is_indiv, :emp, :paral, pr_code, client_kolich;

      aa_loan_kind = 'A Групповой';

      if ((pr_is_indiv = 0) and
          (emp = 0) and
          (paral = 0) and
          (client_kolich <> 2)) then
      begin
        aa_loan_kind = 'A Групповой';
      end
      if ((pr_is_indiv = 1) and
          (emp = 0) and
          (paral = 0) and
          (pr_code like '%.ONE.%' or client_kolich = 2)) then
      begin
        aa_loan_kind = 'E Парный';
      end
      if (paral = 1) then
      begin
        aa_loan_kind = 'B Параллельный';
      end
      if ((pr_is_indiv = 1) and
          (emp = 0) and
          (paral = 0) and
          (client_kolich < 2) and
          (not(pr_code like '%.ONE.%' or client_kolich = 2))) then
      begin
        if (pr_code containing ('.SME.')) then
          aa_loan_kind = 'F SME Кредиты';
        else
          aa_loan_kind = 'C Индивидуальный';
      end
      if ((pr_is_indiv = 1) and
          (emp = 1) and
          (paral = 0)) then
      begin
        aa_loan_kind = 'D Сотрудники';
      end
    end

    if (aa_loan_currency is null) then
    begin
      select iif(kt.k_currency = kt.k_index_curr_code, 'Местная валюта', 'Иностранная валюта')
      from kontract kt
      where kt.k_id = :aa_k_id
      into aa_loan_currency;
    end

    suspend;
  end
end
