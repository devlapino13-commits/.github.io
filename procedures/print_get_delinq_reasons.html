CREATE PROCEDURE PRINT_GET_DELINQ_REASONS(
    KC_ID D_ID)
RETURNS (
    ANALIZ_DATE_HIS D_DATE,
    DELINQ_REASON_HIS D_STR_LONG)
AS
declare variable RD_ID D_INTEGER;
declare variable ACL_DATE D_DATE;
declare variable RES_STR_VALUES D_STR_255;
declare variable UN_ID D_ID;
declare variable C_ID D_ID;
declare variable OBJ_KIND D_STR_100;
declare variable OBJ_ID D_ID;
begin
    rd_id = f_get_rd_id('anketa.delinq.1');
    un_id = 1000000000001;

    select kc_from_c_id
    from knclient
    where kc_id = :kc_id
    into :c_id;

    obj_kind = 'KnClient';
    obj_id = kc_id;
    if (not exists
        (
            select *
            from actionlist
            where acl_object_kind = f_objkind(:obj_kind, 'id')
              and acl_object_id = :obj_id
              and acl_from_rd_id = :rd_id
        )
    ) then
    begin
        obj_kind = 'Clients';
        obj_id = c_id;
    end


    for select cast(acl_last_datetime as date) -- RFCKG-4267
    from actionlist
    inner join clients on c_id = acl_respons_c_id
    where acl_object_kind = f_objkind(:obj_kind, 'id')
      and acl_object_id = :obj_id
      and acl_date < current_date
      and acl_status = 2
      and acl_from_un_id <> :un_id
      and acl_from_rd_id = :rd_id
    order by acl_date desc
    into :acl_date
    do
    begin
        select coalesce(res_str_values, '')
        from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'delinqreport.reason', :acl_date, null)
        into :delinq_reason_his;

        analiz_date_his = :acl_date;
    
        if (delinq_reason_his <> '' and delinq_reason_his <> '0') then
        begin
            analiz_date_his = :acl_date;
            suspend;  -- возвращаем результат для каждой анкеты
        end
    end
end
