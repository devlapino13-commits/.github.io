CREATE PROCEDURE UXP$UXALIAS_CALLS(
    UX_UUID D_UUID,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    UXA_UUID D_UUID,
    UXA_OWNER D_UUID,
    UXA_CODE D_STR_CODE,
    UXA_FCODE D_STR_100,
    UXA_NAME D_STR_100,
    UXA_KIND D_SMALLINT,
    UXA_ID D_INTEGER,
    UXA_ML_ID D_INTEGER,
    UXA_ML_CODE D_STR_CODE,
    UXA_FS_INACTIVE D_BOOL,
    UXA_NEED_ACCESS D_BOOL,
    RES_UUID_STR D_STR_CODE,
    RES_IMAGE D_INTEGER,
    RES_MAINMENU D_STR_8100)
AS
declare variable mm_uuid d_uuid;
declare variable tmp_ua_uuid d_uuid;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  mm_uuid = uxf$mainmenu();

  for select ufm_ux_uuid
  from ux$frame, ux$form
  where ufr_uuid = :ux_uuid and
        ufr_owner = ufm_uuid
  union
  select ufm_ux_uuid
  from ux$form
  where ufm_uuid = :ux_uuid
  into uxa_owner
  do
  begin
    for select uxa_uuid, uxa_code, uxa_fcode, uxa_name, uxa_kind, uxa_id, uxa_fs_inactive, uxa_need_access
    from ux$alias
    where uxa_owner = :uxa_owner
    into :uxa_uuid, :uxa_code, :uxa_fcode, :uxa_name, :uxa_kind, :uxa_id, :uxa_fs_inactive, uxa_need_access
    do
    begin
      res_uuid_str = uuid_to_char(uxa_uuid);

      uxa_ml_id = null;
      tmp_ua_uuid = null;
      uxa_ml_code = '';

      select ua_uuid
      from ux$action
      where ua_owner = :mm_uuid and
            ua_uxa_uuid = :uxa_uuid
      rows 1
      into tmp_ua_uuid;

      if (tmp_ua_uuid is not null) then
      begin
        select item_id, item_code
        from uxp$actionmenu(:tmp_ua_uuid)
        into uxa_ml_id, uxa_ml_code;
      end

      res_image = 20;
  
      if (exists(select * from ux$search
      where usr_uuid = :uxa_uuid)) then
        res_image = 4;
      else
      if (uxa_kind = 2) then
        res_image = 48;

      res_mainmenu = uxf$menupath(uxa_ml_id);

      suspend;
    end
  end
end
