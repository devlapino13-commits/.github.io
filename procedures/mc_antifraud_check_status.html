CREATE PROCEDURE MC_ANTIFRAUD_CHECK_STATUS(
    INN D_STR_20 NOT NULL)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255)
AS
begin
    res_code = 'in_process';
    res_text = 'Запрос в обработке';

    select first 1 f_enum('antifraud_status', ac.ac_status, 'code')
    from antifraud_check ac
    where ac.ac_inn = :inn
        and ac.ac_date = current_date
    order by ac.ac_id desc
    into :res_code;

    res_text =
        case :res_code
            when 'not_found' then ''
            when 'was_found' then 'По результатам проверки вам отказано в регистрации.'
            when 'client_error' then 'При попытке проверки в АЦ возникла ошибка. Свяжитесь со своим специалистом.'
            when 'server_error' then 'При попытке проверки в АЦ возникла ошибка. Свяжитесь со своим специалистом.'
            when 'format_error' then 'Возникла ошибка при обработке ответа. Свяжитесь со своим специалистом.'
            when 'in_process' then 'Запрос в обработке.'
        end;
    suspend;
end
