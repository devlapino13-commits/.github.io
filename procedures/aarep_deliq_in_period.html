CREATE PROCEDURE AAREP_DELIQ_IN_PERIOD(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_CURRENCY VARCHAR(3))
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    K_KOLICH_REPAY INTEGER,
    PR_NAME VARCHAR(50),
    REG_NAME VARCHAR(60),
    PR_LOAN_TYPE_STR VARCHAR(10),
    IS_DELIQ_STR VARCHAR(10),
    REP_DELIQ_OS DECIMAL(18,4),
    REP_DELIQ_PERC DECIMAL(18,4),
    DELINQ_DAY INTEGER,
    REP_DELIQ_BEGIN_DATE DATE,
    REP_DELIQ_END_DATE DATE,
    REP_ACTIVE_BALANCE DECIMAL(18,4),
    KF_OFFICER_NAME VARCHAR(100),
    AAREP_DISB_LO VARCHAR(100),
    MAX_DELINQ_DAY INTEGER,
    BRANCH_NAME VARCHAR(60),
    OKRUG_NAME VARCHAR(60))
AS
declare variable pr_loan_type smallint;
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate decimal(18,4);
declare variable kal_new_kf_officer decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
declare variable cr_rate curr_type;
declare variable ks_kc_id id_type;
declare variable region_id integer;
declare variable k_from_kf_id curr_type;
declare variable is_delinq smallint;
declare variable region_id_prior integer;
declare variable k_id decimal(18,0);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  cr_rate = null;
  select cr_rate
  from opercurr
  where cr_from_cu_id = '840' and
        :rep_date_end between cr_date and cr_date_end
  into cr_rate;
  if (cr_rate is null) then
    cr_rate = 1;
  if (:rep_currency is null) then
    rep_currency = '000';
  region_id_prior = 0;
  for select oks_from_k_id, oks_delinq_date, oks_date_end, delinq_day, region_id, max_delinq_day, oks_balance,
             is_delinq
      from aarep_deliq_in_period_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
      order by region_id
      into :k_id, :rep_deliq_begin_date, :rep_deliq_end_date, :delinq_day, :region_id, :max_delinq_day,
           :rep_active_balance, :is_delinq
  do
  begin
    select k_kontract_no, k_disb_date, k_amount_approved, k_circle, k_circle_internal, k_kolich_repay, pr_name,
           pr_loan_type, k_index_curr_code, k_index_curr_rate, k_from_kf_id
    from kontract, product
    where k_from_pr_id = pr_id and
          k_id = :k_id
    into k_kontract_no, k_disb_date, k_amount_approved, k_circle, k_circle_internal, k_kolich_repay, pr_name,
         pr_loan_type, k_index_curr_code, k_index_curr_rate, k_from_kf_id;
--  Выдавший КС
    select ks_kc_id
    from kontstate
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_from_k_id = :k_id
    into ks_kc_id;
    select c_name_fio
    from clients
    where c_id = :ks_kc_id
    into :aarep_disb_lo;
--  Текущий КС
    select kal_new_kf_officer
    from knaktlist
    where kal_from_kf_id = :k_from_kf_id and
          :rep_date_end between kal_date and kal_date_end and
          knaktlist.is_delete = 0
    into :kal_new_kf_officer;
    select c_name_fio
    from clients
    where c_id = :kal_new_kf_officer
    into :kf_officer_name;
    rep_deliq_os = 0;
    rep_deliq_perc = 0;
    is_deliq_str = '';
    if (is_delinq = 1) then
    begin
      rep_deliq_os = null;
      rep_deliq_perc = null;
--    Вычисляет задолженность по ОС и по процентам
      with tabl1
      as (select ok_sh_date, sum(ok_os + ok_sh_os) zad_os, sum(ok_perc + ok_sh_perc) zad_perc
          from operkred
          where ok_from_k_id = :k_id and
                ok_ol_is_active = 1 and
                ok_ol_date <= :rep_date_end and
                ok_type in (1, 2, 3, 4, 5, 6, 9) and
                ok_from_c_id = 0
          group by ok_sh_date
          having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) <> 0)
      select sum(zad_os), sum(zad_perc)
      from tabl1
      where ok_sh_date <= :rep_date_end
      into rep_deliq_os, rep_deliq_perc;
      if (rep_deliq_os is null) then
        rep_deliq_os = 0;
      if (rep_deliq_perc is null) then
        rep_deliq_perc = 0;
      is_deliq_str = 'Просрочен';

      if (k_index_curr_code <> '417') then
      begin
        if (cr_rate > k_index_curr_rate) then
          rep_active_balance = rep_active_balance * cr_rate;
        else
          rep_active_balance = rep_active_balance * k_index_curr_rate;
      end
    end
    if (pr_loan_type = 0) then
      pr_loan_type_str = 'Групп.';
    else
    if (pr_loan_type = 1) then
      pr_loan_type_str = 'Индив.';

    if (region_id_prior <> region_id) then
    begin
      region_id_prior = region_id;
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
      from region r1, region r2, region r3
      where r1.reg_unicode = :region_id and
            r1.parent = r2.reg_unicode and
            r2.parent = r3.reg_unicode
      into reg_name, branch_name, okrug_name;
    end
    suspend;
  end
end
