CREATE PROCEDURE MAKE_NEW_KONTRACT(
    MA_ID INTEGER,
    MA_NUMBER INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    KF_ID DECIMAL(18,0),
    K_ID DECIMAL(18,0),
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    K_KONTRACT_NO VARCHAR(20))
AS
declare variable K_DISB_DATE date;
declare variable K_DISB_KIND D_INTEGER;
declare variable KC_ID D_ID;
declare variable FPB_ID D_INTEGER;
declare variable DB_ID D_INTEGER;
declare variable ERROR_TEXT D_STR_255;
declare variable ERROR_STATUS D_SMALLINT;
begin
  kf_id = 0;
  k_id = 0;
  k_circle = 0;
  k_circle_internal = 0;
  k_kontract_no = '';
  k_disb_date = null;
  db_id = f_get_db_id();

  select kf_id, k_id, k_circle, k_circle_internal
  from insert_new_kontract(0, :ma_id, :ma_number, :un_id)
  into :kf_id, :k_id, :k_circle, :k_circle_internal;

  select k_kontract_no, k_disb_date, k_disb_kind
  from kontract
  where k_id = :k_id
  into :k_kontract_no, :k_disb_date, :k_disb_kind;

  execute procedure fp_kont_fill_anketa(:k_id, :k_disb_date, :un_id);

  if (db_id = 81001) then
  begin
    -- Расчет ПДН
    for select kc_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into kc_id
    do
    begin
      execute procedure hp_knclient_pdn_update(:kc_id, 0, null, null, null, null, null, 'make_new_kontract');

      execute procedure hp_check_external_overdue(:kc_id);
      execute procedure hp_antifrod_check(:k_id);
    end

    select error_status, error_text
    from hp_check_limit_kontract(:k_id)
    into :error_status, error_text;

    if (error_status = 1) then
    begin
      exception e_message(:error_text);
    end

    -- RFCRU-5012
    if (k_disb_kind = 7) then
    begin
      for select kc.kc_id, fcb.fcb_from_fpb_id
      from kontract k
      inner join knclient kc on kc.kc_from_k_id = k.k_id
        and kc.kc_amount_approved > 0
      inner join miniappclient mac on mac.mac_from_ma_id = k.k_from_ma_app_id
        and mac.mac_from_c_id = kc.kc_from_c_id
      left join fps_application_banks fab on fab.fab_from_mac_id = mac.mac_id
      left join fps_client_banks fcb on fab.fab_from_fcb_id = fcb.fcb_id
      where k.k_id = :k_id
        and kc.kc_amount_approved > 0
        and k.k_from_ma_app_id = mac.mac_from_ma_id
      into :kc_id, :fpb_id
      do
      begin
        error_status = 0;
        error_text = '';

        select res_is_error, res_text
        from fm_set_fps_loan_banks('knclient', :kc_id, :fpb_id, :un_id)
        into error_status, error_text;

        if (error_status = 1) then
          exception e_message(:error_text);

      end
    end
  end

  if (db_id in (39801, 41701)) then
  begin
    update kontmat
    set kml_object_kind = f_objkind('kontract', 'id'),
        kml_object_id = :k_id
    where kml_object_kind = f_objkind('miniapp', 'id') and
          kml_object_id = :ma_id;

    execute procedure hp_kontmat_sell_update(:k_id);
  end


  if (db_id = 39801) then
  begin
    -- RFCKG-4587
    if (f_get_const('stop_credit_is_active') = 1) then
        execute procedure hp_extinfo_insert('stop_credit', f_objkind('kontract', 'id'), :k_id, 0);

    -- RFCKZ-128
    execute procedure hp_afm_send_compliance_mail(f_objkind('kontract', 'id'), :k_id, 'pdl');
  end

  -- RFCKG-4732 Информационный лист
  if (db_id = 41701 and f_get_const('new_loan_conditions') = '1') then
  begin
    if (exists(
      select 1
      from knclient kncl
      where kncl.kc_from_k_id = :k_id
        and kncl.kc_amount_approved > 0
        and kncl.kc_loan_target = f_get_rd_id('Loan_Target.Consumer')
    )) then
    begin
      select res_text
      from hp_edoc_insert('loan.application_new_conditions', f_objkind('kontract', 'id'), :k_id, :un_id)
      into :error_text;

      if (error_text is not null and error_text <> '') then
        exception e_message(:error_text);
    end
  end

  suspend;
end
