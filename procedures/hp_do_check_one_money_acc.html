CREATE PROCEDURE HP_DO_CHECK_ONE_MONEY_ACC(
    ACC_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    OL_ID D_ID,
    OL_DATE D_DATE,
    AL_PLUS D_CURRENCY,
    AL_MINUS D_CURRENCY,
    ACC_PLUS D_CURRENCY,
    ACC_MINUS D_CURRENCY)
AS
declare variable tmp_ol_id d_id;
declare variable tmp_plus d_currency;
declare variable tmp_minus d_currency;
declare variable al_id d_str_code;
begin
  select al_id
  from acc_list
  where al_from_acc_id = :acc_id
  into :al_id;

  if (al_id is null) then
    exception e_default 'Account not linked';

  al_plus = 0;
  acc_plus = 0;
  al_minus = 0;
  acc_minus = 0;

  for select ol_id, ol_date
      from operlist, operbook
      where ob_ol_is_active = 1 and
            ob_ol_date between :rep_date_begin and :rep_date_end and
            ob_from_ol_id = ol_id and
            ob_account_dt = :al_id and
            cast(ol_parent_id as d_id) is null
      into ol_id, ol_date
  do
  begin
    al_plus = 0;
    acc_plus = 0;
    for select ol_id
        from operlist
        where ol_id = :ol_id
        union
        select ol_id
        from operlist
        where ol_parent_id = :ol_id
        into tmp_ol_id
    do
    begin
      tmp_plus = 0;
      select sum(ob_summa)
      from operbook
      where ob_from_ol_id = :tmp_ol_id and
            ob_account_dt = :al_id
      into tmp_plus;
      if (tmp_plus is null) then
        tmp_plus = 0;

      al_plus = al_plus + tmp_plus;

      tmp_plus = 0;
      select sum(kob_summa)
      from kn_operbook
      where kob_from_ol_id = :tmp_ol_id and
            kob_acc_dt = :acc_id
      into tmp_plus;
      if (tmp_plus is null) then
        tmp_plus = 0;

      acc_plus = acc_plus + tmp_plus;
    end

    if (al_plus <> acc_plus) then
      suspend;
  end

  al_plus = 0;
  al_minus = 0;
  acc_plus = 0;
  acc_minus = 0;
  for select ol_id, ol_date
      from operlist, operbook
      where ob_ol_is_active = 1 and
            ob_ol_date between :rep_date_begin and :rep_date_end and
            ob_from_ol_id = ol_id and
            ob_account_kt = :al_id and
            cast(ol_parent_id as d_id) is null
      into ol_id, ol_date
  do
  begin
    al_plus = 0;
    al_minus = 0;
    acc_plus = 0;
    acc_minus = 0;
    for select ol_id
        from operlist
        where ol_id = :ol_id
        union
        select ol_id
        from operlist
        where ol_parent_id = :ol_id
        into tmp_ol_id
    do
    begin
      tmp_minus = 0;
      select sum(ob_summa)
      from operbook
      where ob_from_ol_id = :tmp_ol_id and
            ob_account_kt = :al_id
      into tmp_minus;
      if (tmp_minus is null) then
        tmp_minus = 0;

      al_minus = al_minus + tmp_minus;

      tmp_minus = 0;
      select sum(kob_summa)
      from kn_operbook
      where kob_from_ol_id = :tmp_ol_id and
            kob_acc_kt = :acc_id
      into tmp_minus;
      if (tmp_minus is null) then
        tmp_minus = 0;

      acc_minus = acc_minus + tmp_minus;
    end

    if (al_minus <> acc_minus) then
      suspend;
  end
end
