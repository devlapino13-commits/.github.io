CREATE PROCEDURE UXP$UXALIAS_AUTO_OPEN(
    UA_UUID D_UUID,
    UA_UUID_CODE D_STR_CODE = null)
RETURNS (
    RES_ID D_BOOL)
AS
declare variable ufm_uuid d_uuid;
declare variable is_locked d_bool;
declare variable uxa_uuid d_uuid;
declare variable res_frame_count d_smallint;
declare variable res_total_params d_smallint;
declare variable res_locked_params d_smallint;
declare variable ufr_uuid d_uuid;
declare variable ufp_uuid d_uuid;
declare variable uf_uuid d_uuid;
declare variable uf_field_name d_str_code;
begin
  res_frame_count = 0;
  res_total_params = 0;
  res_locked_params = 0;
  res_id = 0;

  if (coalesce(ua_uuid, '') = '') then
    ua_uuid = uxf$uniq(ua_uuid_code);

  select uxa_uuid, coalesce(ux_release_uuid, ux_draft_uuid)
  from ux$action, ux$alias, ux$code
  where ua_uuid = :ua_uuid and
        ua_uxa_uuid = uxa_uuid and
        uxa_owner = ux_uuid
  into uxa_uuid, ufm_uuid;
        
  for select ufr_uuid
  from ux$frame
  where ufr_owner = :ufm_uuid
  into ufr_uuid
  do
    res_frame_count = res_frame_count + 1;

  if (res_frame_count = 1) then
  begin
    for select ufp_uuid, uf_uuid, uf_field_name
    from ux$frparam, ux$field
    where ufp_ufr_uuid = :ufr_uuid and
          ufp_uf_uuid = uf_uuid and
          uf_is_param = 1
    into ufp_uuid, uf_uuid, uf_field_name
    do
    begin
      if (not exists(select * from ux$fritem
      where ufi_ufr_uuid = :ufr_uuid and
            ufi_uf_uuid = :uf_uuid and
            ufi_readonly = 0 and
            ufi_hidden = 0)) then
        Continue;

      res_total_params = res_total_params + 1;

      is_locked = 0;
      if (exists(select * from ux$paramdef
                 where upd_ufp_uuid = :ufp_uuid and
                       coalesce(upd_uxa_uuid, :uxa_uuid) = :uxa_uuid and
                       upd_locked = 1)) then
        is_locked = 1;

      if (is_locked = 0) then
      begin
        if (exists(select *
                   from ux$actionparam
                   where uap_ua_uuid = :ua_uuid and
                         uap_param_name = :uf_field_name and
                         uap_is_output = 0 and
                         uap_locked = 1)) then
        is_locked = 1;
      end

      res_locked_params = res_locked_params + is_locked;
    end
  end

  if (res_frame_count = 1 and res_total_params = res_locked_params) then
    res_id = 1;

  suspend;
end
