CREATE PROCEDURE UXP$TRANSLATE_INFO
RETURNS (
    RES_LANG_CODE D_STR_10,
    RES_TOTAL D_INTEGER,
    RES_NEW D_INTEGER,
    RES_TRANSLATED D_INTEGER,
    RES_UNCHECKED D_INTEGER,
    RES_CHECKED D_INTEGER,
    RES_ERROR D_INTEGER,
    RES_ENUM D_STR_100,
    RES_ENUM_ERROR D_BOOL)
AS
begin
  select count(*)
  from ux$transmain
  into res_total;

  for select distinct trl_lang_code
  from ux$translate
  union
  select el_code
  from enum_list
  where el_kind = 'user_language' and
        el_id <> 0
  into res_lang_code
  do
  begin
    res_enum = 'Отсутствует в справочнике';
    res_enum_error = 1;

    select el_name
    from enum_list
    where el_kind = 'user_language' and
          el_code = :res_lang_code
    into res_enum;

    if (row_count = 1) then
      res_enum_error = 0;

    res_new = 0;
    res_translated = 0;
    res_checked = 0;
    res_error = 0;
    res_unchecked = 0;
    select sum(iif(trl_from_trm_id is null, 1, 0)),
           sum(iif(trl_from_trm_id is not null, 1, 0)),
           sum(iif(trl_from_trm_id is not null and trl_checked = 1, 1, 0)),
           sum(iif(trl_from_trm_id is not null and trl_checked = 0, 1, 0)),
           sum(iif(trl_from_trm_id is not null and trl_checked is null, 1, 0))
    from ux$transmain
         left join ux$translate on (trl_from_trm_id = trm_id and trl_lang_code = :res_lang_code)
    into res_new, res_translated, res_checked, res_error, res_unchecked;

    if (res_new is null) then
      res_new = 0;
    if (res_translated is null) then
      res_translated = 0;
    if (res_checked is null) then
      res_checked = 0;
    if (res_error is null) then
      res_error = 0;
    if (res_unchecked is null) then
      res_unchecked = 0;

    suspend;
  end
end
