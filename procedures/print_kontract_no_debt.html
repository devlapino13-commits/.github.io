CREATE PROCEDURE PRINT_KONTRACT_NO_DEBT(
    K_ID ID_TYPE)
RETURNS (
    C_NAME_FIO STR_100,
    C_PASS_NO VARCHAR(30),
    C_PASS_WHO STR_255,
    C_PASS_DATE DATE,
    C_PASS_WHO_CODE STR_20,
    K_KONTRACT_NO STR_20,
    K_KONTRACT_DATE DATE,
    EDOC_DATE_CREATE D_DATE)
AS
declare variable counter smallint;
declare variable oko_debt curr_type;
declare variable ox_debt curr_type;
declare variable ks_state d_integer;
begin
  -- Если по договору была допущена непрерывная просрочка  от 120 дней.
  if (rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN') not in ('SDANILOVA', 'EASYAMOLOVA', 'ZGITINOVA', 'LABDURAKHMANOVA') and
  exists(select * from operkred
  where ok_ol_is_active = 1
  and ok_from_k_id = :k_id
  and ok_type in (4,5)
  and ok_ol_date - ok_sh_date >= 120
  and ok_os + ok_perc < 0)) then
  begin
    exception e_message 'За справкой незамедлительно обратитесь в Операционный отдел';
  end

  select cast(edl_create_datetime as date)
  from edoc_list
  inner join edoc_cfg on edc_id = edl_from_edc_id
  where edl_object_kind = 9 and
        edl_object_id = :k_id and
        edc_code = 'loan.agreement' and
        edl_status <> -1
  order by edl_create_datetime desc rows 1
  into edoc_date_create;

  counter = 0;
  select coalesce(sum(oko_os + oko_perc + oko_sh_os + oko_sh_perc), 0),
         coalesce(sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff), 0),
         ks_state
  from kontstate
  left join operkroff on ks_from_k_id = oko_from_k_id and oko_ol_is_active = 1
  left join operextra on ks_from_k_Id = ox_object_id and ox_object_kind = 9 and ox_ol_is_active = 1
  where ks_from_k_id = :k_id
        and ks_ol_is_delete = 0
        and ks_state = 2
        and current_date between ks_date_begin and ks_date_end
  group by 3
  order by 3
  rows 1
  into :oko_debt, :ox_debt, ks_state;
  counter = row_count;

  if (counter = 1 and
      oko_debt <= 0 and
      ox_debt <= 0 and
      ks_state = 2) then
  begin
    for select c_name_fio,
               c_pass_no,
               c_pass_who,
               c_pass_date,
               c_pass_who_code,
               trim(trailing '/' from left(k_kontract_no, 15)),
               k_kontract_date
        from knclient
        inner join kontract k on kc_from_k_id = k.k_id
        inner join clients on c_id = kc_from_c_id
        where k.k_id = :k_id and
              kc_amount_approved > 0
        into c_name_fio,
             c_pass_no,
             c_pass_who,
             c_pass_date,
             c_pass_who_code,
             k_kontract_no,
             k_kontract_date
    do
    begin
      suspend;
    end
  end
end
