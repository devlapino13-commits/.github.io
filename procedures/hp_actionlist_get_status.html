CREATE PROCEDURE HP_ACTIONLIST_GET_STATUS(
    ACL_OBJECT_KIND_STR VARCHAR(100),
    ACL_OBJECT_ID DECIMAL(18,0),
    ACL_ACTION_CODE VARCHAR(100))
RETURNS (
    ACL_ID DECIMAL(18,0),
    ACL_STATUS INTEGER,
    ACL_DATETIME TIMESTAMP)
AS
declare variable acl_object_kind integer;
declare variable acl_from_rd_id integer;
declare variable ovs_add_value decimal(18,4) = 0;
begin
-- получаем значение (состояние) ActionList
  acl_object_kind = f_objkind(acl_object_kind_str, 'id');

  select rd_id
  from refdata
  where rd_code = :acl_action_code
  into :acl_from_rd_id;

  select acl_id, acl_datetime
  from actionlist
  where acl_object_kind = :acl_object_kind and
        acl_object_id = :acl_object_id and
        acl_from_rd_id = :acl_from_rd_id
  order by acl_id desc
  rows 1
  into :acl_id, :acl_datetime;

  ovs_add_value = 0;
  select ovs_add_value
  from hp_objectprops_get(:acl_object_kind, :acl_object_id, 'RefData', :acl_from_rd_id, null)
  into :ovs_add_value;
  acl_status = round(ovs_add_value);
  if (acl_status is null) then
    acl_status = 0;
  suspend;
end
