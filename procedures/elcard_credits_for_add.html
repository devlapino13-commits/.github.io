CREATE PROCEDURE ELCARD_CREDITS_FOR_ADD
RETURNS (
    ESM_ID ID_TYPE,
    CARD_ID STR_100,
    DISB_AMOUNT CURR_TYPE,
    K_ID D_ID)
AS
declare variable esm_card_id str_100;
declare variable c_id id_type;
begin
  -- Процедура предоставляет список платежей, по которым должна пройти проверка, на возможность перевода
  for select em.esm_id, em.esm_account, em.esm_amount, em.esm_object_id, em.esm_from_k_id
      from extsysmain em
      where em.esm_project = 'elcard_disb' and
            em.esm_action = 'CardDisb' and
            em.esm_status = 0
      into :esm_id, :esm_card_id, :disb_amount, :c_id, :k_id
  do
  begin
    card_id = (select token
             from hp_client_bank_card(:c_id));

    if (esm_card_id <> card_id) then
        Continue; -- если на момент выдачи была одна карта, а на момент првоерки карта вдруг измнеилась, такой перевод дальше не пускаем

    suspend;
  end
end
