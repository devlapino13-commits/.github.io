CREATE PROCEDURE AAREP_MC_ONLINE_LOAN_PUSH
RETURNS (
    C_ID D_ID,
    CLIENT_FIO D_STR_255,
    CLIENT_POINTS D_INTEGER,
    CLIENT_LAST_KIB_DATE_CHECK D_DATETIME,
    REGION_NAME D_STR_255,
    OFFICE_NAME D_STR_255,
    EMPOLOYEE_FIO D_STR_255)
AS
declare variable K_ID D_ID;
begin
    for select c_id
    from mc_send_online_loan_push_tmp
    into c_id
    do
    begin
        select ks.ks_from_k_id
        from knclient kc
        inner join kontstate ks on kc.kc_from_k_id = ks.ks_from_k_id
            and ks.ks_ol_is_delete = 0
            and ks.ks_state = 1
        where kc.kc_from_c_id = :c_id
            and kc.kc_amount_approved > 0
        order by ks.ks_date_begin desc
        rows 1
        into k_id;

        select c.c_name_fio
        from clients c
        where c.c_id = :c_id
        into client_fio;

        select
            c.c_name_fio, 
            reg1.reg_name,
            reg2.reg_name
        from kontract k
        inner join knaktlist kal on k.k_from_kf_id = kal.kal_from_kf_id
            and current_date between kal.kal_date and kal.kal_date_end
            and kal.is_delete = 0
        inner join clients c on kal.kal_new_kf_officer = c.c_id
        inner join region reg1 on kal.kal_new_reg_id = reg1.reg_unicode
        inner join region reg2 on reg1.parent = reg2.reg_unicode
        where k_id = :k_id
        into
            empoloyee_fio,
            office_name,
            region_name;

        client_points = f_point(:c_id, current_date, 0);
        client_last_kib_date_check = (select kbm_datetime from hp_kibminiapp_last(:c_id, null));

        suspend;    
    end
end
