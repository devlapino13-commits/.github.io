CREATE PROCEDURE HP_MY_EDOC_LIST(
    INFO_KIND D_SMALLINT)
RETURNS (
    EDL_ID D_ID,
    EDL_BLANK_DATE D_DATE,
    EDL_CREATE_DATETIME D_DATETIME,
    EDL_BLANK_STATUS D_INTEGER,
    EDL_STATUS D_INTEGER,
    EDL_OBJECT_KIND D_INTEGER,
    EDL_OBJECT_ID D_ID,
    EDS_STATUS D_INTEGER,
    EDS_SIGN_KIND D_STR_CODE,
    EDC_CODE D_STR_CODE,
    EDC_NAME D_STR_100,
    RES_CAPTION D_STR_255,
    RES_BLANK_NAMES D_STR_LONG,
    RES_STATUS_KIND D_BOOL,
    RES_CAN_SIGN D_BOOL,
    RES_EDOC_ID D_ID)
AS
declare variable eds_enum_edoc_sign_kind d_integer;
declare variable c_id d_id;
begin
  c_id = uxf$usr_cid(0);

  if (c_id is null) then
    exit;

  if (info_kind is null) then
    info_kind = 0;

  for select e.edl_id, e.edl_blank_date, e.edl_create_datetime, e.edl_blank_status, e.edl_status,
             e.edl_object_kind, e.edl_object_id,
             s.eds_enum_edoc_sign_kind, s.eds_status,
             c.edc_code, c.edc_name
  from edoc_sign s
       inner join edoc_list e on s.eds_from_edl_id = e.edl_id
       inner join edoc_cfg c on c.edc_id = e.edl_from_edc_id
  where s.eds_from_c_id = :c_id
  order by eds_id desc
  into edl_id, edl_blank_date, edl_create_datetime, edl_blank_status, edl_status,
       edl_object_kind, edl_object_id,
       eds_enum_edoc_sign_kind, eds_status,
       edc_code, edc_name
  do
  begin
    if (info_kind = 0) then -- Сегодняшие и в ожидании подписания
    begin
      if (not (edl_status = 0 or edl_blank_date >= (current_date - 1))) then
        Continue;
    end else
    if (info_kind = 1) then -- Все подписанные
    begin
      if (edl_status <> 1) then
        Continue;
    end else
    if (info_kind = 2) then -- Аннулированные
    begin
      if (edl_status <> -1) then
        Continue;
    end

    eds_sign_kind = f_enum_name('edoc_sign_kind', eds_enum_edoc_sign_kind);

    begin
      res_caption = f_objname(edl_object_kind, edl_object_id);
      when any do
        res_caption = 'edc_name';
    end

    res_status_kind = case edl_status
      when -1 then 0
      when 0 then null
      when 1 then 1
    end;

    res_blank_names = '';
    select left(list(distinct edf_print_name), 1020)
    from edoc_blank, edoc_frx
    where edb_from_edl_id = :edl_id and
          edb_from_edf_id = edf_id
    into res_blank_names;

    res_can_sign = 0;
    if (eds_status = 0 and edl_status = 0 and edl_blank_status = 1) then
      res_can_sign = 1;

    res_edoc_id = edl_id;

    suspend;
  end
end
