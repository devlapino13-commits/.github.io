CREATE PROCEDURE AAREP_CP_ONE_NEW_TEMP2_201306(
    KONT_ID DECIMAL(18,0),
    DATE_END DATE,
    IS_CALC_PERC_TO_DELIQ INTEGER)
RETURNS (
    PDD DATE,
    POK_T INTEGER,
    PSH_DD DATE,
    PBALANCE DECIMAL(18,4),
    PPAYED_PERC DECIMAL(18,4),
    PDD_ORIG DATE,
    PPAYED_TAX DECIMAL(18,4))
AS
declare variable psh_dd2 date;
declare variable psh_dd_prior date;
declare variable pbalance_all decimal(18,4);
declare variable pbalance_orig decimal(18,4);
declare variable pdd_orig2 date;
declare variable psh_dd_orig date;
declare variable last_sh_dd date;
begin
  psh_dd_prior = null;
  pbalance_all = 0;
  if (is_calc_perc_to_deliq is null) then
    is_calc_perc_to_deliq = 0;
  if (is_calc_perc_to_deliq = 0) then
  begin /* Не начисляем процент на просроченную часть */
    psh_dd2 = null;
    for select ok_ol_date, ok_type, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc), sum(ok_tax)
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :kont_id and
              ok_type <= 7 and
              ok_from_c_id = 0
        group by ok_ol_date, ok_type, ok_sh_date
        having sum(ok_os + ok_sh_os) <> 0 or (sum(ok_perc) <> 0) or (sum(ok_tax) <> 0)
        order by 3, 1
        into :pdd, :pok_t, :psh_dd, :pbalance, :ppayed_perc, :ppayed_tax
    do
    begin
      pbalance_orig = :pbalance;
      pdd_orig2 = :pdd;
      psh_dd_orig = :psh_dd;
      if ((pok_t <> 3) and
          (pdd <= :date_end)) then -- 16.10.2012 - эксперимент, включаем OK_Type=3
--      if (pdd<=:date_end) then
      begin
        pdd_orig = :pdd;
        if (pdd > psh_dd) then
          pdd = :psh_dd;
        if (pbalance is null) then
          pbalance = 0;
        if (ppayed_perc is null) then
          ppayed_perc = 0;
        if (ppayed_tax is null) then
          ppayed_tax = 0;
        suspend;
      end

      if ((pok_t <> 3) and
          (psh_dd >= :date_end)) then -- 16.10.2012 - эксперимент, включаем OK_Type=3
--      if (psh_dd>=:date_end) then
      begin
        if (psh_dd2 is null) then
        begin
          psh_dd2 = :psh_dd;
          pdd = :date_end;
          pdd_orig = :date_end;
          pbalance = 0;
          ppayed_perc = 0;
          ppayed_tax = 0;
          pok_t = 4;
          suspend;
        end
      end
      pdd = :pdd_orig2;

      if ((pdd <= :date_end) and
          (psh_dd <= :date_end)) then
      begin
        last_sh_dd = psh_dd_orig;
        if ((psh_dd = :psh_dd_prior) or (psh_dd_prior is null)) then
        begin
          pbalance = :pbalance_orig;
          pbalance_all = pbalance_all - pbalance;
          pdd = :psh_dd;
          pdd_orig = :psh_dd;
          pbalance = 0;
          ppayed_perc = 0;
          ppayed_tax = 0;
          pok_t = 0;
        end
        else
        begin
          if (pbalance_all < 0) then
          begin
            pbalance = :pbalance_all;
            psh_dd = :psh_dd_prior;
            pdd = :psh_dd_prior;
            pdd_orig = :psh_dd_prior;
            ppayed_perc = 0;
            ppayed_tax = 0;
            pok_t = 0;
            suspend;
          end
          pbalance_all = 0;
          pbalance = :pbalance_orig;
          pbalance_all = pbalance_all - pbalance;
          psh_dd = :psh_dd_orig;
          pdd = :psh_dd;
          pdd_orig = :psh_dd;
          pbalance = 0;
          ppayed_perc = 0;
          ppayed_tax = 0;
          pok_t = 0;
        end
      end

      psh_dd_prior = :psh_dd;
    end

    if (pbalance_all < 0) then
    begin
      pbalance = :pbalance_all;
      psh_dd = :last_sh_dd;
      pdd = :psh_dd;
      pdd_orig = :psh_dd;
      ppayed_perc = 0;
      ppayed_tax = 0;
      pok_t = 0;
      suspend;
    end

  end
  else
  begin /* Начисляем процент на просроченную часть тоже */
    for select ok_ol_date, ok_type, sum(ok_os + ok_sh_os), sum(ok_perc), sum(ok_tax)
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :kont_id and
              (ok_type >= 2 and
              ok_type <> 3 and
              ok_type <= 8) -- 16.10.2012 - эксперимент, включаем OK_Type=3
--    and (OK_Type>=2 and OK_Type<=8)
              and
              ok_ol_date <= :date_end and
              ok_from_c_id = 0
        group by ok_ol_date, ok_type
        having sum(ok_os + ok_sh_os) <> 0 or (sum(ok_perc) <> 0) or (sum(ok_tax) <> 0)
        order by 1
        into :pdd, :pok_t, :pbalance, :ppayed_perc, :ppayed_tax
    do
    begin
      psh_dd = :pdd;
      pdd_orig = :pdd;
      if (pbalance is null) then
        pbalance = 0;
      if (ppayed_perc is null) then
        ppayed_perc = 0;
      if (ppayed_tax is null) then
        ppayed_tax = 0;
      suspend;
    end
    psh_dd = :date_end;
    pdd = :date_end;
    pdd_orig = :date_end;
    pbalance = 0;
    ppayed_perc = 0;
    ppayed_tax = 0;
    pok_t = 4;
    suspend;
  end

end
