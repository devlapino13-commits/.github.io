CREATE PROCEDURE UXP$UXFRAME_UPDATE(
    UFR_UUID D_UUID,
    UX_NAME D_STR_100,
    UFR_CODE D_STR_CODE,
    UFR_US_CODE D_STR_CODE,
    UFR_IS_GRID D_BOOL,
    UFR_TREE_PARENT_FIELD D_STR_CODE,
    UFR_TREE_IMAGE_FIELD D_STR_CODE,
    UFR_CAN_UPDATE D_BOOL,
    UFR_CAN_INSERT D_BOOL,
    UFR_CAN_DELETE D_BOOL,
    US_UPDATE D_STR_CODE,
    US_INSERT D_STR_CODE,
    US_DELETE D_STR_CODE,
    UFR_HIDE_ON_GROUP D_BOOL,
    UFR_GROUP_PANEL D_BOOL,
    UFR_FIND_PANEL D_SMALLINT,
    UFR_HIDE_SUMMARY D_BOOL,
    UFR_AUTOWIDTH D_SMALLINT,
    UFR_SQL_WHERE D_STR_LONG,
    UFR_SQL_ORDERBY D_STR_LONG,
    UFR_REFRESH_PARAMS D_STR_LONG = null,
    UFR_TREE_POSITION_FIELD D_STR_LONG = null,
    UFR_TREE_CAN_DRAG_FIELD D_STR_LONG = null,
    UFR_TREE_CAN_DROP_FIELD D_STR_LONG = null,
    UFR_ADD_TABLE_FIELDS D_STR_CODE = null)
AS
declare variable old_us_uuid d_uuid;
declare variable new_us_uuid d_uuid;
declare variable ufm_us_uuid d_uuid;
declare variable field_list d_str_long;
declare variable diff_count d_integer;
declare variable uf_field_name d_str_code;
declare variable field_count d_integer;
begin
  if (ufr_refresh_params is null) then
    ufr_refresh_params = '';

  if (ufr_tree_position_field is null) then
    ufr_tree_position_field = '';

  if (ufr_tree_can_drag_field is null) then
    ufr_tree_can_drag_field = '';

  if (ufr_tree_can_drop_field is null) then
    ufr_tree_can_drop_field = '';

  if (ufr_add_table_fields is null) then
    ufr_add_table_fields = '';

  select ufr_us_uuid, ux_us_uuid
  from ux$frame, ux$form, ux$code
  where ufr_uuid = :ufr_uuid and
        ufm_uuid = ufr_owner and
        ufm_ux_uuid = ux_uuid
  into old_us_uuid, ufm_us_uuid;

  if (coalesce(ufr_us_code, '') <> '') then
    new_us_uuid = uxf$uniq(:ufr_us_code);
  else
    new_us_uuid = ufm_us_uuid;

  diff_count = 0;
  field_count = 0;
  field_list = '';

  if (old_us_uuid is distinct from new_us_uuid) then
  begin
    for select uf_field_name
    from ux$field, ux$fritem
    where uf_owner = :old_us_uuid and
          ufi_uf_uuid = uf_uuid and
          ufi_ufr_uuid = :ufr_uuid
    into uf_field_name
    do
    begin
      field_count = field_count + 1;

      if (not exists(select * from ux$field
      where uf_owner = :new_us_uuid and
            uf_field_name = :uf_field_name)) then
      begin
        if (field_list <> '') then
          field_list = field_list || ', ';

        field_list = field_list || uf_field_name;
        diff_count = diff_count + 1;
      end

      if (diff_count = 10) then
      begin
        field_list = field_list || ' и т.д.';
        Break;
      end
    end

    if (field_list <> '') then
      exception e_message 'Сначала удалите поля из интерфейса: ' || field_list;
  end

  update ux$frame
  set ufr_code = :ufr_code,
      ufr_name = :ux_name,
      ufr_us_uuid = :new_us_uuid,
      ufr_is_grid = :ufr_is_grid,
      ufr_can_update = :ufr_can_update,
      ufr_can_insert = :ufr_can_insert,
      ufr_can_delete = :ufr_can_delete,
      ufr_tree_parent_field = :ufr_tree_parent_field,
      ufr_tree_image_field = :ufr_tree_image_field,
      ufr_hide_on_group = :ufr_hide_on_group,
      ufr_group_panel = :ufr_group_panel,
      ufr_find_panel = :ufr_find_panel,
      ufr_hide_summary = :ufr_hide_summary,
      ufr_autowidth = :ufr_autowidth,
      ufr_sql_where = :ufr_sql_where,
      ufr_sql_orderby = :ufr_sql_orderby,
      ufr_refresh_params = :ufr_refresh_params,
      ufr_tree_position_field = :ufr_tree_position_field,
      ufr_tree_can_drag_field = :ufr_tree_can_drag_field,
      ufr_tree_can_drop_field = :ufr_tree_can_drop_field,
      ufr_add_table_fields = :ufr_add_table_fields
  where ufr_uuid = :ufr_uuid;

  if (ufr_can_update = 1) then
  begin
    update or insert into ux$frdata (ufd_ufr_uuid, ufd_kind, ufd_us_uuid)
    values (:ufr_uuid, 1, uxf$uniq(:us_update))
    matching (ufd_ufr_uuid, ufd_kind);
  end

  if (ufr_can_insert = 1) then
  begin
    update or insert into ux$frdata (ufd_ufr_uuid, ufd_kind, ufd_us_uuid)
    values (:ufr_uuid, 2, uxf$uniq(:us_insert))
    matching (ufd_ufr_uuid, ufd_kind);
  end

  if (ufr_can_delete = 1) then
  begin
    update or insert into ux$frdata (ufd_ufr_uuid, ufd_kind, ufd_us_uuid)
    values (:ufr_uuid, 3, uxf$uniq(:us_delete))
    matching (ufd_ufr_uuid, ufd_kind);
  end
end
