CREATE PROCEDURE AAREP_CL_DOSROCH_CLOSED(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    LINE_NO D_INTEGER,
    C_ID D_ID,
    REG_NAME D_STR_CODE,
    KAL_NEW_KF_OFFICER_FIO D_STR_100,
    DOSROCH_DAYS D_INTEGER,
    K_CLOSED_KONTRACT_NO D_STR_CODE,
    K_CLOSED_KF_NO D_STR_100,
    K_CLOSED_AMOUNT_GR D_CURRENCY,
    K_CLOSED_AMOUNT D_CURRENCY,
    LAST_PAY_SUMM D_CURRENCY,
    K_CLOSED_PR_NAME D_STR_CODE,
    K_PLAN_CLOSE_DATE D_DATE,
    K_FACT_CLOSE_DATE D_DATE,
    K_NEXT_KONTRACT_NO D_STR_CODE,
    K_NEXT_KF_NO D_STR_100,
    K_NEXT_AMOUNT D_CURRENCY,
    K_NEXT_PR_NAME D_STR_CODE,
    K_NEXT_DISB_DATE D_DATE)
AS
declare variable kf_id d_id;
declare variable k_id d_id;
declare variable ol_id d_id;
declare variable reg_id d_id;
declare variable reg_id_tmp d_id;
declare variable kal_new_kf_officer d_id;
begin
  for select ks_from_k_id, ks_from_ol_id, ks_date_begin, --
             k_kont_last_date, k_from_kf_id, k_kontract_no, k.k_amount_approved, k.k_fact_closed_kf_no, --
             kal.kal_new_reg_id, kal.kal_new_kf_officer, --
             pr_name
      from kontstate ks
--
      inner join kontract k on ks_from_k_id = k_id
      inner join product p on k_from_pr_id = pr_id
      inner join knaktlist kal on k_from_kf_id = kal_from_kf_id
      inner join region_list(:rep_region_id, 1, -1, 1) on kal.kal_new_reg_id = id_list

      where ks_state = 2 and
            ks_ol_is_delete = 0 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_date_begin < k_kont_last_date and
            pr_category = 0 and
            :rep_date_end between kal_date and kal_date_end and
            kal.is_delete = 0
      order by kal_new_reg_id, k_from_kf_id, ks_date_begin desc
      into :k_id, :ol_id, :k_fact_close_date, k_plan_close_date, :kf_id, k_closed_kontract_no, k_closed_amount_gr,
           k_closed_kf_no, reg_id, kal_new_kf_officer, k_closed_pr_name
  do
  begin
    if (reg_id <> coalesce(reg_id_tmp, 0)) then
    begin
      reg_id_tmp = reg_id;
      select r.reg_name
      from region r
      where r.reg_unicode = :reg_id
      into reg_name;
    end

    kal_new_kf_officer_fio = null;
    select c_name_fio
    from clients c
    where c_id = :kal_new_kf_officer
    into kal_new_kf_officer_fio;

    dosroch_days = datediff(day, :k_fact_close_date, :k_plan_close_date);

    last_pay_summ = 0;
    select -sum(ok.ok_os + ok.ok_perc + ok.ok_fine)
    from operkred ok
    where ok.ok_from_k_id = :k_id and
          ok.ok_ol_date = :k_fact_close_date and
          ok.ok_ol_is_active = 1 and
          ok.ok_type in (4, 9) and
          ok_from_c_id = 0
    into :last_pay_summ;

    for select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into :c_id, k_closed_amount
    do
    begin
      last_pay_summ = last_pay_summ * :k_closed_amount / :k_closed_amount_gr;

      k_next_kontract_no = '';
      k_next_kf_no = '';
      k_next_disb_date = null;
      k_next_pr_name = null;
      k_next_amount = null;
      select k.k_fact_disb_kf_no, k_kontract_no, ks_date_begin, pr_name, kc_amount_approved
      from kontstate, knclient, kontract k, product
      where ks_from_k_id = kc_from_k_id and
            kc_from_c_id = :c_id and
            k_from_pr_id = pr_id and
            pr_category = 0 and
            ks_from_k_id = k_id and
            kc_amount_approved > 0 and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_date_begin >= :k_fact_close_date
      order by ks_date_begin
      rows 1
      into k_next_kf_no, k_next_kontract_no, k_next_disb_date, k_next_pr_name, k_next_amount;

      line_no = coalesce(line_no, 0) + 1;

      suspend;
    end
  end
end
