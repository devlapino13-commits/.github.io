CREATE PROCEDURE AAREP_POST_INSPECTION(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    ADD_FILTER D_BOOL)
RETURNS (
    ID D_ID,
    K_ID D_ID,
    C_ID D_ID,
    STATUS D_SMALLINT,
    DT_DISB D_DATE,
    DT_ROBOT D_DATETIME,
    DT_ARRIVAL D_DATETIME,
    DT_START D_DATETIME,
    DT_END D_DATETIME,
    TM_PROCESS D_INTEGER,
    TM_TOTAL D_INTEGER,
    K_NO D_STR_20,
    REG_NAME D_STR_CODE,
    PR_NAME D_STR_CODE,
    EMP_NAME D_STR_100,
    ST_NAME D_STR_20,
    KF_NAME D_STR_100,
    KC_NAME D_STR_100,
    MN_NAME D_STR_100,
    REASON D_STR_255,
    NOTE D_STR_2048,
    AMOUNT D_CURRENCY,
    NUM_CLI D_INTEGER,
    LST_CLI D_STR_2048,
    LST_POINT D_STR_100,
    LST_CIRCLE D_STR_100,
    FIRST_STATUS_REASON D_STR_255,
    PARENT_NAME D_STR_100,
    OFFICER_NAME D_STR_100,
    BY_RATING D_STR_10)
AS
declare variable SYM_DELIM D_STR_LANG;
declare variable RD_ID D_INTEGER;
declare variable RDR_ID D_INTEGER;
declare variable RD_ID1 D_INTEGER;
declare variable RD_ID2 D_INTEGER;
declare variable REG_ID D_INTEGER;
declare variable REG_LST D_STR_8100;
declare variable CM_NOTE D_STR_4096;
declare variable P1 D_INTEGER;
declare variable P2 D_INTEGER;
declare variable P3 D_INTEGER;
declare variable OPR_ID D_ID;
declare variable FIRST_REFUSED D_SMALLINT;
begin
  sym_delim = ascii_char(13) || ascii_char(10);
  rd_id = f_get_rd_id('ActionGroup.Check_Loan_Data_PostCheck');
  rdr_id = f_get_rd_id('ActionGroup.Check_Loan_Data_PostCheckR');

  reg_lst = '';
  select
    list(id_list, ';')
  from region_list(:rep_region_id, 1, 3, 1)
  into
    :reg_lst;

  for
    select
      acl_id,
      acl_object_id,
      acl_from_reg_id,
      acl_respons_c_id,
      c_name_fio,
      acl_status,
      case acl_status
        when -3 then 'Отмена'
        when -2 then 'Отказан'
        when -1 then 'На доработку'
        when  0 then 'В ожидании'
        when  1 then 'Принят на обработку'
        when  2 then 'Подтвежден'
      end,
      acl_datetime,
      acl_last_datetime,
      acl_from_rd_id
    from actionlist
      left join clients
        on c_id = acl_respons_c_id
    where acl_date between :rep_date_begin and :rep_date_end
      and acl_from_rd_id in (:rd_id, :rdr_id)
      --and acl_status <> 0
  into
    :id,
    :k_id,
    :reg_id,
    :c_id,
    :emp_name,
    :status,
    :st_name,
    :dt_arrival,
    :dt_end,
    :rd_id2
  do
  begin
    if (status not in (-2,-1, 1, 2) and add_filter = 1)
      then continue;

    if (reg_lst not containing reg_id)
      then continue;

    select c.c_name_fio
    from kontract k
    inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
    inner join clients c on kal.kal_new_kf_officer = c.c_id
    where k.k_id = :k_id and
          kal.is_delete = 0 and
          :rep_date_end between kal.kal_date and kal.kal_date_end
    into :officer_name;

    select opr_id
    from objectprops
    inner join objectvalues on ovs_from_opr_id = opr_id
    inner join username on un_id = ovs_from_un_id
    where opr_object_kind = 9 and
          opr_object_id = :k_id and
          ovs_date_end = '31.12.9999'
    rows 1
    into opr_id;

    select
           case ovs_value
             when '-1' then 1
             when '-2' then 2
             else 0
           end
    from objectvalues
    inner join username on un_id = ovs_from_un_id
    where ovs_from_opr_id = :opr_id and
          ovs_value in ('-1', '-2', '2') -- первичное присвоение статуса
          rows 1
    into first_refused;

    if (first_refused in (1, 2)) then
    begin
      select f_get_verify_memo(obm_memo, 'reason')
      from objectmemo
      where obm_from_opr_id = :opr_id and
            obm_memo not containing ('Для сведения:')
      order by obm_id rows 1
      into first_status_reason;
    end

    select rd_id
    from hp_get_check_loan_data(:k_id)
    into:rd_id1;

    dt_robot = (
      select first 1
        acl_last_datetime
     from actionlist
     where acl_object_kind = 9
       and acl_object_id = :k_id
       and acl_from_rd_id = :rd_id1
       and acl_respons_c_id = 1000000000000
       and acl_id < :id
       and acl_status = 2
     order by acl_id desc);

    dt_start = (
      select first 1
        ovs_datetime
      from objectprops
        inner join objectvalues
          on ovs_from_opr_id = opr_id
            and ovs_value = 1
      where opr_object_kind = 9 and
            opr_object_id = :k_id and
            opr_subject_kind = -1 and
            opr_subject_id = :rd_id2
      order by ovs_id);
    if (dt_start is null or status = 1)
      then dt_end = null;

    select
      k_kontract_no,
      k_fact_disb_date,
      k_amount_approved,
      k_kolich_client_approved,
      reg_name,
      pr_name,
      k_fact_disb_kf_no,
      c1.c_name_fio,
      c2.c_name_fio,
      parent_name,
      iif(coalesce(k_by_rating, 0) = 0, 'нет', 'да')
    from kontract
      inner join region
        on reg_unicode = k_last_reg_id
      inner join region_list(:reg_id, 1,3,1)
        on id_list = reg_unicode
      inner join product
        on pr_id = k_from_pr_id
      left join clients as c1
        on c1.c_id = k_fact_disb_lo_id
      left join clients as c2
        on c2.c_id = k_approved_manager
    where k_id = :k_id
    into
      :k_no,
      :dt_disb,
      :amount,
      :num_cli,
      :reg_name,
      :pr_name,
      :kf_name,
      :kc_name,
      :mn_name,
      :parent_name,
      :by_rating;
    select
      list(c_name_fio, ', '),
      list(f_point(kc_from_c_id, :dt_disb), ', '),
      list(kc_circle, ', ')
    from knclient
      inner join clients
        on c_id = kc_from_c_id
    where kc_from_k_id = :k_id
      and kc_amount_approved > 0
    into
      :lst_cli,
      :lst_point,
      :lst_circle;

    cm_note = (
      select first 1
        replace(f_get_verify_memo(obm_memo, 'reason_memo'), :sym_delim, ', ')
      from objectprops
        inner join objectmemo
          on obm_from_opr_id = opr_id
      where opr_object_kind = 9
        and opr_object_id = :k_id
        and opr_subject_kind = -1
        and opr_subject_id = :rd_id2
      order by obm_from_opr_id, obm_datetime desc);

    reason = '';
    note = '';
    if (cm_note <> '') then
    begin
      p1 = position(':', cm_note);
      p2 = position(']', cm_note);
      p3 = position('],', cm_note);

      if (p1 > 0 and p2 > p1) then
        reason = substring(cm_note from p1 + 2 for p2 - p1 -2);
      if (p3 > 0) then
        note = right(cm_note, char_length(cm_note) - p3 - 2);
    end

    tm_process = datediff(minute, dt_arrival, dt_start);
    tm_total = datediff(minute, dt_arrival, dt_end);

    suspend;
  end
end
