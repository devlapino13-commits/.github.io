CREATE PROCEDURE FSROBOT_MILLION_CREDIT(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0),
    CL_NAME STR_100)
AS
declare variable KONT_QTY integer;
begin
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID  - пользователь

-- СМС передаче клиента в Департамент розыска и принудительного взыскания для судебного разбирательства
--C_ID       = substring(rep_params from 1 for position(';' in rep_params) - 1);

--Res_MailList=:rep_email;

  select count(*)
  from kontstate, knclient
  where ks_from_k_id = kc_from_k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        kc_amount_approved > 0
  into :kont_qty;

  res_subject = 'Миллионный кредит ' || cast(kont_qty as str_100);

  res_body = 'Счетчик миллионного кредита:' || cast(kont_qty as str_100) || '<br>';
  res_body = BLOB_APPEND(res_body, 'Остается еще ' || cast((1000000 - :kont_qty) as str_100) || ' выдач.');

  res_maillist = 'btolbaev@mbfplanet.com, bahtiyar@molbulakfinance.kg, a.samidinova@molbulak.com, o.karimov@molbulak.com';

  suspend;
end
