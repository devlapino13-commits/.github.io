CREATE PROCEDURE HP_CASHPLACE_LIST(
    OFFICE_ID D_INTEGER)
RETURNS (
    CSP_ID D_INTEGER,
    CSP_OFFICE_ID D_INTEGER,
    CSP_KIND D_SMALLINT,
    CSP_CASH_NO D_INTEGER,
    CSP_IS_MAIN D_BOOL,
    CSP_CLS_AL_ID D_STR_20,
    CSP_IPF_AL_ID D_STR_20,
    CSP_CASHIER_ID D_ID,
    CSP_DATE_OPEN D_DATE,
    CSP_DATE_CLOSE D_DATE,
    CSP_FROM_UN_ID D_ID,
    CASHIER_FIO D_STR_100,
    CASHIER_CELLS D_STR_100)
AS
declare variable cell_main d_str_code;
declare variable cell_add d_str_code;
begin
  for select csp_id,
             csp_office_id,
             csp_kind,
             csp_cash_no,
             csp_is_main,
             csp_cls_al_id,
             csp_ipf_al_id,
             csp_cashier_id,
             csp_date_open,
             csp_date_close,
             csp_from_un_id,
             c_name_fio
      from cashplace
           left join clients on (csp_cashier_id = c_id)
      where csp_office_id = :office_id
      order by csp_kind, csp_cash_no
      into :csp_id,
           :csp_office_id,
           :csp_kind,
           :csp_cash_no,
           :csp_is_main,
           :csp_cls_al_id,
           :csp_ipf_al_id,
           :csp_cashier_id,
           :csp_date_open,
           :csp_date_close,
           :csp_from_un_id,
           :cashier_fio
  do
  begin
    cashier_cells = '';
    if (coalesce(cashier_fio, '') = '') then
      cashier_fio = '-';
    else
    begin
      cell_main = f_cont(csp_cashier_id, 'cell', 'last', 'main');
      cell_add = f_cont(csp_cashier_id, 'cell', 'last', 'add');
  
      if (cell_main <> '') then
        cashier_cells = cell_main;
  
      if (cell_add <> '' and cell_add <> cell_main) then
      begin
        if (cashier_cells = '') then
          cashier_cells = cell_add;
        else
          cashier_cells = cashier_cells || ', ' || cell_add;
      end
    end

    suspend;
  end
end
