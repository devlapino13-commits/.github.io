CREATE PROCEDURE HP_INVOICEIPF_CREATE(
    DT_SUPPLY D_DATE NOT NULL,
    UN_ID D_ID NOT NULL,
    REGP_ID D_INTEGER = null,
    SUPPLY_ID D_ID = null,
    NOTE D_STR_LONG = null)
RETURNS (
    INV_ID D_INTEGER,
    INV_COUNT D_INTEGER,
    INV_AMOUNT D_CURRENCY)
AS
declare variable rd_id d_integer;
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable ol_id d_id;
declare variable reg_lst d_str_8100;
declare variable ol_lst d_str_8100;
declare variable amount d_currency;
begin
  regp_id = coalesce(regp_id, 1000000);
  rd_id = f_get_rd_id('AgreementKinds.IPF_Sellers');
  inv_id = 0;
  inv_count = 0;
  inv_amount = 0;

  for
    select
      ob_client_kt,
      list(ob_from_ol_id, ','),
      sum(ob_summa)
    from operbook
    where ob_ol_date = :dt_supply
      and ob_short_acc_kt = '21199/99'
      and ob_ol_is_active = 1
      and ob_client_kt = coalesce(:supply_id, ob_client_kt)
      and exists (
        select *
        from opermat
          inner join kontmat
            on kml_bill_id = om_bill_id
              and kml_bill_line_no = om_bill_line_no
          inner join agreements
            on ag_id = kml_from_ag_id
              and ag_object_id = ob_client_kt
              and ag_from_rd_id = :rd_id
              and ag_tax_percent = 0.5
        where om_from_ol_id = ob_from_ol_id)
      and not exists(
        select *
        from invoiceipf_detail
          inner join invoiceipf
            on inv_id = invd_inv_id
              and inv_status >= 0
        where invd_ol_id = ob_from_ol_id)
    group by ob_client_kt
  into
    :c_id,
    :ol_lst,
    :amount
  do
  begin
    reg_id = (
      select first 1
        ag_from_init_reg_id
      from agreements
      where ag_object_id = :c_id
        and ag_from_rd_id = :rd_id
        and ag_tax_percent = 0.5
        and ag_from_init_reg_id is not null
      order by ag_id desc);

    if (not (reg_lst containing reg_id))
      then continue;

    insert into invoiceipf(
      inv_date_supply,
      inv_suppler_id,
      inv_reg_id,
      inv_amount,
      inv_note,
      inv_un_id)
    values (
      :dt_supply,
      :c_id,
      :reg_id,
      :amount,
      :note,
      :un_id)
    returning
      inv_id
    into
      :inv_id;

    for
      select
        res_str
      from hp_split_str(:ol_lst, ',')
    into
      :ol_id
    do
    begin
      if (not exists(
        select * from invoiceipf_detail
        where invd_inv_id = :inv_id
        and invd_ol_id = :ol_id))
      then
        insert into invoiceipf_detail(invd_inv_id, invd_ol_id)
        values(:inv_id, :ol_id);
    end

    inv_count = inv_count + 1;
    inv_amount = inv_amount + amount;
  end

  suspend;
end
