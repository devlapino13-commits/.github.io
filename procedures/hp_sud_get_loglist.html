CREATE PROCEDURE HP_SUD_GET_LOGLIST(
    REQUEST_ID D_ID,
    LAST_ON_TOP D_BOOL,
    UN_ID D_ID)
RETURNS (
    SUL_ID D_ID,
    SUL_IS_NEW D_BOOL,
    SLC_ID D_ID,
    CLIENT_NAME D_STR_100,
    STAGE_STR D_STR_CODE,
    STATUS_STR D_STR_CODE,
    UN_FIO D_STR_100,
    SUL_DATETIME D_DATETIME)
AS
declare variable kontract_no d_str_code;
declare variable sul_object_kind d_integer;
declare variable sul_status d_integer;
begin
  if (last_on_top is null) then
    last_on_top = 1;

  sul_is_new = 1;

  select k_kontract_no
  from sud_mainreq, kontract
  where smr_id = :request_id and
        smr_from_k_id = k_id
  into kontract_no;

  if (last_on_top = 1) then
  begin
    for select sul_id, sul_from_slc_id, sul_object_kind, sul_status, un_fio, sul_datetime
    from sud_mainlog, username
    where sul_from_smr_id = :request_id and
          sul_from_un_id = un_id
    order by sul_datetime desc, sul_id desc
    into :sul_id, :slc_id, :sul_object_kind, :sul_status, :un_fio, :sul_datetime
    do
    begin
      client_name = '';
      if (slc_id is null)
      then client_name = kontract_no;
      else
      begin
        select c_name_fio
        from sud_loanclient, knclient, clients
        where slc_id = :slc_id and
              slc_from_kc_id = kc_id and
              kc_from_c_id = c_id
        into client_name;
      end
  
      stage_str = '';
      status_str = '';
  
      select stage_str, status_str
      from hp_sud_get_stage_status(:sul_object_kind, :sul_status)
      into :stage_str, :status_str;
  
      suspend;
    end
  end else
  begin
    for select sul_id, sul_from_slc_id, sul_object_kind, sul_status, un_fio, sul_datetime
    from sud_mainlog, username
    where sul_from_smr_id = :request_id and
          sul_from_un_id = un_id
    order by sul_datetime, sul_id
    into :sul_id, :slc_id, :sul_object_kind, :sul_status, :un_fio, :sul_datetime
    do
    begin
      client_name = '';
      if (slc_id is null)
      then client_name = kontract_no;
      else
      begin
        select c_name_fio
        from sud_loanclient, knclient, clients
        where slc_id = :slc_id and
              slc_from_kc_id = kc_id and
              kc_from_c_id = c_id
        into client_name;
      end
  
      stage_str = '';
      status_str = '';
  
      select stage_str, status_str
      from hp_sud_get_stage_status(:sul_object_kind, :sul_status)
      into :stage_str, :status_str;
  
      suspend;
    end
  end
end
