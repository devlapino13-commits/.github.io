CREATE PROCEDURE AAREP_BUDOPER_FACT_PERIOD(
    REP_REGION_ID INTEGER,
    REP_BIT_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    BOI_FROM_BIT_ID INTEGER,
    BOI_FACT DECIMAL(18,4))
AS
declare variable ba_sign integer;
declare variable ba_dt_acc varchar(20);
declare variable ba_dt_no integer;
declare variable ba_kt_acc varchar(20);
declare variable ba_kt_no integer;
declare variable ba_kind integer;
declare variable ba_id integer;
declare variable tmp_amount decimal(18,4);
declare variable al_id varchar(20);
declare variable reg_id integer;
declare variable tmp_one decimal(18,4);
begin
  boi_fact = 0;
  boi_from_bit_id = :rep_bit_id;
  /*
    0 Оборот
    1 Остаток по Дт
    2 Остаток по Кт
    3 Сводный остаток (Дт-Кт)  */
  for select ba_id, ba_sign, ba_kind, ba_dt_acc, ba_dt_no, ba_kt_acc, ba_kt_no
      from budaccount, buditem
      where ba_from_bit_id = :rep_bit_id and
            bit_id = :rep_bit_id and
            bit_has_child = 0
      into :ba_id, :ba_sign, :ba_kind, :ba_dt_acc, :ba_dt_no, :ba_kt_acc, :ba_kt_no
  do
  begin
    if (ba_dt_acc = '') then
      ba_dt_acc = null;
    if (ba_kt_acc = '') then
      ba_kt_acc = null;

    if (ba_dt_no <= 0) then
      ba_dt_no = null;
    if (ba_kt_no <= 0) then
      ba_kt_no = null;
    tmp_amount = 0;
    if (ba_kind = 0) then
    /* Дт и Кт оборот */
    begin
      if ((ba_dt_acc is not null) or (ba_kt_acc is not null)) then
      begin
        if ((ba_dt_acc is not null) and
            (ba_kt_acc is not null)) then
        /* Есть Дт и Кт счет */
        begin
          if ((ba_dt_no is not null) and
              (ba_kt_no is not null)) then
          /* Есть Кт и Кт лиц.счет */
          begin
            select sum(ob_summa)
            from operbook, region_list(:rep_region_id, 1, -1, 1)
            where ob_ol_is_active = 1 and
                  ob_short_acc_dt = :ba_dt_acc || '/' || :ba_dt_no and
                  ob_short_acc_kt = :ba_kt_acc || '/' || :ba_kt_no and
                  ob_ol_reg_id = id_list and
                  ob_ol_date between :rep_date_begin and :rep_date_end
            into :tmp_amount;
          end
          if ((ba_dt_no is not null) and
              (ba_kt_no is null)) then
          /* Есть Дт, Нет Кт */
          begin
            select sum(ob_summa)
            from operbook, region_list(:rep_region_id, 1, -1, 1)
            where ob_ol_is_active = 1 and
                  ob_short_acc_dt = :ba_dt_acc || '/' || :ba_dt_no and
                  ob_short_acc_kt starting :ba_kt_acc and
                  ob_ol_reg_id = id_list and
                  ob_ol_date between :rep_date_begin and :rep_date_end
            into :tmp_amount;
          end
          if ((ba_dt_no is null) and
              (ba_kt_no is not null)) then
          /* Нет Дт, Есть Кт */
          begin
            select sum(ob_summa)
            from operbook, region_list(:rep_region_id, 1, -1, 1)
            where ob_ol_is_active = 1 and
                  ob_short_acc_dt starting :ba_dt_acc and
                  ob_short_acc_kt = :ba_kt_acc || '/' || :ba_kt_no and
                  ob_ol_reg_id = id_list and
                  ob_ol_date between :rep_date_begin and :rep_date_end
            into :tmp_amount;
          end
        end
        if (ba_kt_acc is null) then
        begin
          if (ba_dt_no is not null) then
          begin
            select sum(ob_summa)
            from operbook, region_list(:rep_region_id, 1, -1, 1)
            where ob_ol_is_active = 1 and
                  ob_ol_reg_id = id_list and
                  ob_ol_date between :rep_date_begin and :rep_date_end and
                  ob_short_acc_dt = :ba_dt_acc || '/' || :ba_dt_no
            into :tmp_amount;
          end
          if (ba_dt_no is null) then
          begin
            select sum(ob_summa)
            from operbook, region_list(:rep_region_id, 1, -1, 1)
            where ob_ol_is_active = 1 and
                  ob_ol_reg_id = id_list and
                  ob_ol_date between :rep_date_begin and :rep_date_end and
                  ob_short_acc_dt starting :ba_dt_acc
            into :tmp_amount;
          end
        end
        if (ba_dt_acc is null) then
        begin
          if (ba_kt_no is not null) then
          begin
            select sum(ob_summa)
            from operbook, region_list(:rep_region_id, 1, -1, 1)
            where ob_ol_is_active = 1 and
                  ob_ol_reg_id = id_list and
                  ob_ol_date between :rep_date_begin and :rep_date_end and
                  ob_short_acc_kt = :ba_kt_acc || '/' || :ba_kt_no
            into :tmp_amount;
          end
          if (ba_kt_no is null) then
          begin
            select sum(ob_summa)
            from operbook, region_list(:rep_region_id, 1, -1, 1)
            where ob_ol_is_active = 1 and
                  ob_ol_reg_id = id_list and
                  ob_ol_date between :rep_date_begin and :rep_date_end and
                  ob_short_acc_kt starting :ba_kt_acc
            into :tmp_amount;
          end
        end
      end
    end

    if ((ba_kind >= 1) and
        (ba_kind <= 3)) then
    /* Остатки */
    begin
      if (ba_dt_acc is not null) then
      begin
        tmp_amount = 0;
        for select al_id, al_from_reg_id
            from acc_list, region_list(:rep_region_id, 1, -1, 1)
            where al_from_bg_id = :ba_dt_acc and
                  al_from_reg_id = id_list and
                  ((:ba_dt_no is null) or
                  (:ba_dt_no is not null and
                  al_number = :ba_dt_no))
            into :al_id, :reg_id
        do
        begin
          tmp_one = f_al_bal(al_id, rep_date_end);
          tmp_amount = tmp_amount + tmp_one;
        end
        if (tmp_amount <> 0) then
        begin
          if (ba_kind = 1) then
          /* Дт остаток */
          begin
            if (tmp_amount < 0) then
              tmp_amount = 0;
          end
          if (ba_kind = 2) then
          /* Кт остаток */
          begin
            if (tmp_amount > 0) then
              tmp_amount = 0;
            else
              tmp_amount = 0 - tmp_amount;
          end
        end
      end
    end

    if (ba_kind = 4) then
    /* Чистый оборот Дт - Кт */
    begin
      if (ba_dt_acc is not null) then
      begin
        tmp_amount = 0;
        for select al_id
            from acc_list, region_list(:rep_region_id, 1, -1, 1)
            where al_from_bg_id = :ba_dt_acc and
                  al_from_reg_id = id_list and
                  ((:ba_dt_no is null) or
                  (:ba_dt_no is not null and
                  al_number = :ba_dt_no))
            into :al_id
        do
        begin
          select dt_ekv - kt_ekv
          from hp_al_obor(:al_id, :rep_date_begin, :rep_date_end)
          into :tmp_one;

          tmp_amount = tmp_amount + tmp_one;
        end
      end
    end
    if (tmp_amount is null) then
      tmp_amount = 0;
    boi_fact = boi_fact + tmp_amount * ba_sign;
  end
  suspend;
end
