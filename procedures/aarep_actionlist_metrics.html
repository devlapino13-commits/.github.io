CREATE PROCEDURE AAREP_ACTIONLIST_METRICS(
    K_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT,
    RES_CRITICAL_LEVEL D_SMALLINT)
AS
declare variable criteria d_str_255;
declare variable res_text d_str_long;
declare variable critical_level d_smallint;
declare variable jso d_integer;
declare variable data_jso d_integer;
declare variable table_jso d_integer;
declare variable lines_jsa d_integer;
declare variable cells_jsa d_integer;
declare variable line_html d_str_8100;
declare variable rep_knd_datetime d_datetime;

declare function get_color (
critical_level d_smallint)
returns d_str_20
as
begin
return case critical_level
when 0 then '#006600' -- зеленый
when 1 then '#999900' -- желтый
else '#990000' -- красный
end;
end


begin
  res_html = '';
  res_critical_level = 0;

  select rep_knd_datetime, jso
  from hp_antifrod_get_res(:k_id)
  into rep_knd_datetime, jso;

  if (rep_knd_datetime is not null) then
    res_html = blob_append(res_html, f_format('<br><h3>Дата проверки: %1</h3>', f_datetimetostr(rep_knd_datetime)));

  for select json.obj_get_obj(:jso, 'criteria_data.' || val)
  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'criteria_arr'), 0)
  into data_jso
  do
  begin
    criteria = json.obj_get_val(data_jso, 'criteria');
    critical_level = json.obj_get_val(data_jso, 'critical_level');
    res_text = json.obj_get_val(data_jso, 'res_text');
    table_jso = json.obj_get_obj(data_jso, 'add');
    lines_jsa = iif(table_jso is null, null, json.obj_get_arr(table_jso, 'lines'));

    res_html = blob_append(res_html, f_format('<br><br><h3>%1 &nbsp; &nbsp; <font color = "%2">%3</font></h3>', criteria, get_color(critical_level), res_text));

    if (lines_jsa is not null and json.count_get(lines_jsa) > 0) then
    begin
      res_critical_level = maxvalue(critical_level, res_critical_level);

      -- создадим таблицу
      line_html = '<table width="100%">';

      -- заголовок
      line_html = line_html || (select list('<col width="'|| val ||'%">', '') from hp_bki_json_get_list(json.obj_get_arr(:table_jso, 'width'), 0));
      line_html = line_html || '<tr>' || (select list('<th>'|| val ||'</th>', '') from hp_bki_json_get_list(json.obj_get_arr(:table_jso, 'header'), 0)) || '</tr>';

      -- строки
      for select jso
      from hp_bki_json_get_list(:lines_jsa, 2)
      into cells_jsa
      do
        line_html = line_html || '<tr>' || (select list('<td>'|| val ||'</td>', '') from hp_bki_json_get_list(:cells_jsa, 0)) || '</tr>';

      res_html = blob_append(res_html, line_html || '</table>');
    end
  end

  res_html = blob_append(
    f_format('<body><h1 style="color:#999900">%2</h1>',
        get_color(res_critical_level),
        iif(res_critical_level > 0, 'Выявлен стоп-фактор', 'Не выявлено стоп-факторов')),
    res_html);

  res_html = blob_append((select res_style from fm_html_styles),res_html);
  res_html = blob_append(res_html, '</body>');
  suspend;
end
