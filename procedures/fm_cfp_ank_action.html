CREATE PROCEDURE FM_CFP_ANK_ACTION(
    MAC_ID D_INTEGER,
    UN_ID D_ID,
    DO_ACTION D_STR_CODE,
    DATA_JSON D_BLOB_TEXT)
RETURNS (
    RES_HTML D_BLOB_TEXT,
    ERR_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable ca_id d_id;
declare variable c_name_fio d_str_255;
declare variable cq_name_ru d_str_255;
declare variable caa_value d_integer;
declare variable cp_id d_id;
declare variable counter d_integer;
declare variable quest_id d_integer;
declare variable obj_count d_integer;
declare variable answ_value d_integer;
declare variable name_ru d_str_255;
declare variable type_name d_str_100;
declare variable cq_name d_str_255;
declare variable cp_parent_id d_id;
declare variable cp_parent_tmp d_id;
declare variable ca_object_kind d_id;
declare variable pos_txt_ru d_str_long;
declare variable pos_txt_kg d_str_long;
declare variable avr_income d_integer;
declare variable txt_ru d_str_long;
declare variable txt_kg d_str_long;
declare variable cq_name_kg d_str_255;
declare variable kpd_err d_str_255;
declare variable ma_id d_id;
declare variable net_profit d_integer;
declare variable cq_basic_incom d_smallint;
declare variable ex_profit d_integer;
declare variable cq_code_name d_str_100;
declare variable expense d_integer;
declare variable kbm_id d_id;
declare variable repayment_obligations d_integer;
declare variable lang d_str_code;
declare variable k_id d_id;
declare variable k_id_tmp d_id;
declare variable sum_debt decimal(15,0);
declare variable cw_id d_id;
declare variable have_spouse_salary decimal(15,2);
declare variable crel_other_c_id d_id;
declare variable is_byudjet d_smallint;
declare variable cp_full_name_ru d_str_255;
declare variable salary_certificate d_integer;
declare variable cq_id d_integer;
declare variable prozh_min d_currency;
declare variable db_id d_integer;
declare variable number_of_children d_integer;
declare variable const_kdn_max_value d_float;
declare variable is_gambler d_bool;
declare variable json_document d_blob_text;
declare variable job_title d_str_255;
declare variable salary_income d_currency;
declare variable pension_income d_currency;
declare variable asp_income d_currency;
declare variable salary_statement d_currency;
declare variable is_overdue_90 d_smallint;
declare variable is_asp d_smallint;
declare variable all_income d_currency;
declare variable acl_id d_id;
declare variable is_susn d_smallint; /* RFCKZ-996 */
begin
  err_text = '';
  res_html = '';
  job_title = '';
  ca_object_kind = f_objkind('miniappclient', 'id');
  db_id = f_get_db_id();
  const_kdn_max_value = 0.5;
  is_overdue_90 = 0;
  is_asp = 0;
  all_income = 0;

  if(coalesce(mac_id, 0) < 0) then    -- RFCKG-5565
  begin
    acl_id = mac_id * -1;

    select acl.acl_object_id
    from actionlist acl
    where acl.acl_id = :acl_id
    into mac_id;
  end

  select prozh_min
  from hp_velichina_prozh_min      -- RFCKG-4324 Величина Прожиточного Минимума
  into prozh_min;

  select mac.mac_from_c_id, mac.mac_from_ma_id
  from miniappclient mac
  where mac.mac_id = :mac_id and
        mac.mac_amount > 0
  into c_id, ma_id;

  select k.k_id
  from kontract k
  where k.k_from_ma_app_id = :ma_id
  into k_id_tmp;

  k_id_tmp = coalesce(k_id_tmp, 0);

  select ca.ca_id                             -- RFCKG-5565
  from cfp_anketa ca
  where ca.ca_from_c_id = :c_id and
        ca.ca_object_kind = :ca_object_kind and
        ca.ca_object_id = :mac_id and
        (
          coalesce(:acl_id, 0) = 0
          or ca.ca_from_acl_id = :acl_id
        ) and
        (
            coalesce(:acl_id, 0) <> 0
            or ca.ca_from_acl_id is null
         )
  into ca_id;

  if (db_id = 41701) then
  begin
  select cq.cq_id
  from cfp_questions cq
  where cq.cq_code_name = 'salary_certificate' and
        cq.cq_is_delete = 0
  into cq_id;
  end
  if (db_id = 39801) then
  begin
  select case rd.rd_name
               when 'Вопрос не задан' then 0
               else coalesce(rd.rd_name, 0)
             end
  from objectvalues, objectprops, objectvalres, refdata rd
  where opr_object_kind = 0 and
      opr_object_id = :c_id and
      opr_subject_kind = '-1' and
      opr_subject_id in (9266) and -- ответ на вопрос Количество детей в анкете.
      ovs_from_opr_id = opr_id and
      ovr_result_rd_id = rd_id and
      ovs_id = opr_actual_ovs_id and
      ovr_from_opr_id = ovs_from_opr_id and
      ovr_from_ovs_id = ovs_id
  rows 1
  into number_of_children;

  select kji.kji_job_title
  from kib_job_info kji
  where kji.kji_from_c_id = :c_id and
        kji.kji_from_ma_id = :ma_id
        rows 1
  into job_title;
  if (coalesce(job_title,'') = '') then
   job_title = 'Самозанятый';

    select nkh.nkh_is_gambler, nkh.nkh_is_overdue_90
    from negativ_kib_history nkh
    where nkh.nkh_from_c_id = :c_id and
        nkh.nkh_last = 1
    order by nkh.nkh_from_eil_id desc
    rows 1
    into is_gambler, is_overdue_90;

    is_asp = f_is_asp_client(:c_id);

    is_susn = 0;                                  -- RFCKZ-996
    select is_susn
    from hp_susn_get_status(:c_id, :ma_id)
    into is_susn;

    if ((is_gambler = 1)
      or (is_asp = 1)
      or (is_overdue_90 = 1)
      or (is_susn = 1)) then
        const_kdn_max_value = 0.25;

  end
  if (do_action = 'preview') then
  begin
    -- логика для preview
    -- res_html = 'ФИО: Иванов Иван Иванович <br/> do_action = ' || do_action;
    select c.c_name_fio, c.c_blank_lang
    from clients c
    where c.c_id = :c_id
    into c_name_fio, lang;

    res_html = (select res_style
                from fm_html_styles) || f_format('<body><table width="100%"><col width="40%"><col width="60%">');
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(:lang, 'ФИО клиента') || '</b>', :c_name_fio, '', '', -1, -1, -1, -1, 2, ''));
    cp_id = f_json_blob_get_value(:data_json, 'job_id', 0);
    if (db_id = 39801) then
     cp_id = 1;
    name_ru = '';
    type_name = '';

    select iif(:lang = 'kg', cp.cp_name_kg, cp.cp_name_ru), cp.cp_parenet_id,
           f_lang(:lang, f_enum_name('cfp_positions', cp.cp_type))
    from cfp_positions cp
    where cp.cp_id = :cp_id
    into name_ru, cp_parent_id, type_name;

    if (db_id <> 39801) then -- RFCKG-4324
      res_html = blob_append(res_html, f_html_table_row('<b>' || :type_name || '</b>', :name_ru, '', '', -1, -1, -1, -1, 2, ''));

    if (db_id = 39801) then -- RFCKG-4324 
      res_html = blob_append(res_html, f_html_table_row('<b>Род занятий</b>',
                                                        job_title,
                                                        '', '', -1, -1, -1, -1, 2, ''));

    while (coalesce(cp_parent_id, 0) > 0) do
    begin
      select iif(:lang = 'kg', cp.cp_name_kg, cp.cp_name_ru), cp.cp_parenet_id,
             f_lang(:lang, f_enum_name('cfp_positions', cp.cp_type)), cp.cp_full_name_ru
      from cfp_positions cp
      where cp.cp_id = :cp_parent_id
      into name_ru, cp_parent_tmp, type_name, cp_full_name_ru;

      if (db_id = 41701 and
          cp_full_name_ru = 'Бюджетная сфера' and
          type_name = 'Вид деятельности') then
           is_byudjet = 1;

      if (db_id <> 39801) then -- RFCKG-4324
        res_html = blob_append(res_html, f_html_table_row('<b>' || :type_name || '</b>', :name_ru, '', '', -1, -1, -1, -1, 2, ''));

      cp_parent_id = cp_parent_tmp;
    end
    net_profit = 0;
    expense = 0;
    ex_profit = 0;
    have_spouse_salary = 0;
    obj_count = f_json_blob_get_count(:data_json, 'question_list');
    counter = 0;
    while (counter < obj_count) do
    begin
      quest_id = f_json_blob_get_value(:data_json, 'question_list[' || counter || '].id', 0);
      answ_value = f_json_blob_get_value(:data_json, 'question_list[' || counter || '].value', 0);

      select iif(:lang = 'kg', cq.cq_name_kg, cq.cq_name_ru), cq.cq_basic_incom, cq.cq_code_name
      from cfp_questions cq
      where cq.cq_id = :quest_id
      into cq_name, cq_basic_incom, cq_code_name;

      if (db_id = 39801 and
          cq_code_name = 'contract_salary') then
      begin
       select icq.icq_payment_sum
              from income_check_querys icq
              where icq.icq_from_mac_id = :mac_id and
                    icq.icq_from_c_id = :c_id and
                    icq.icq_status = 2 and
                    icq.icq_config = 'salary_check'
              order by icq.icq_create_datetime desc
              rows 1
              into :salary_income;
       if (:salary_income > 0) then
        answ_value = :salary_income;
      end

      if (db_id = 39801 and
          cq_code_name = 'salary_statement' and
          not exists(select * from clientdocs cds
                           inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
                           inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
                           where cds.cds_from_c_id = :c_id and
                                 cds.cds_is_delete = 0 and
                                 pdt.pdt_code = 'card_statement' and -- выписка по карте
                                 current_date between cds.cds_doc_date_begin and
                                 coalesce(cds.cds_doc_date_end, '31.12.2099'))) then
      begin
        answ_value = 0;
      end

      if (db_id = 39801 and
          cq_code_name = 'cl_income_asp') then
      begin
       select icq.icq_payment_sum
              from income_check_querys icq
              where icq.icq_from_mac_id = :mac_id and
                    icq.icq_from_c_id = :c_id and
                    icq.icq_status = 2 and
                    icq.icq_config = 'asp_check'
              order by icq.icq_create_datetime desc
              rows 1
              into :asp_income;
       if (:asp_income > 0) then
         answ_value = :asp_income;
      end

      res_html = blob_append(res_html, f_html_table_row('<b>' || coalesce(:cq_name, '') || '</b>', coalesce(:answ_value, ''), '', '', -1, -1, -1, -1, 2, ''));
      if (cq_code_name = 'cl_add_income_spouse_salary' and
          coalesce(answ_value, 0) > 0) then
        have_spouse_salary = 1;
      if (cq_code_name in ('monthly_housing_expenses', 'monthly_family_expenses', 'monthly_tax_expenses', 'monthly_other_expenses')) then
      begin
        if (expense = 0) then
          expense = :answ_value;
        else
          expense = expense + :answ_value;
      end
      if (coalesce(answ_value, 0) = 0 and
          cq_basic_incom = 1 and
          db_id <> 39801) then
      begin
        exception e_message 'Не заполнен поле - ' || :cq_name;
        suspend;
        exit;
      end
      if (cq_basic_incom = 1) then
      begin
        if (net_profit = 0) then
          net_profit = answ_value;
        else if (db_id = 41701) then
          net_profit = net_profit * :answ_value;
        else if (db_id = 39801) then
          net_profit = net_profit + :answ_value;
      end
      if (db_id = 41701 and
          cq_code_name in ('cl_add_income_spouse_salary', 'cl_add_income_household', 'cl_add_income_service')) then
        ex_profit = ex_profit + :answ_value;

      if(db_id = 39801) then
      begin
        if (cq_code_name = 'contract_salary') then
          salary_income = answ_value;
        if (cq_code_name = 'cl_income_pension') then
          pension_income = answ_value;
        if (cq_code_name = 'cl_income_asp') then
          asp_income = answ_value;
        -- RFCKZ-103
        if (cq_code_name = 'salary_statement') then
          salary_statement = answ_value;
      end

      counter = counter + 1;
      when any do
      begin
        if (db_id <> 39801) then
        begin
          exception e_message 'Ошибка. Значение основных доходов неправильно указаны. Слишком большие значения';
          suspend;
          exit;
        end
      end
    end
    if (db_id = 39801) then
    begin
      all_income = coalesce(salary_income, 0) + coalesce(salary_statement, 0) + coalesce(pension_income, 0) + coalesce(asp_income, 0);

      res_html = blob_append(res_html, f_html_table_row('<b>Оценка платежеспособности</b>',
                                                        iif(all_income >= (prozh_min + coalesce(number_of_children, 0) * 0.5 * prozh_min),  'Платежеспособен', '<span style="color: red;">Неплатежеспособен. <br> КДН не может быть рассчитан</span>'),
                                                        '', '', -1, -1, -1, -1, 2, '')); -- RFCKG-4324 минимальная величина платёжеспособности клиента
    end
   if (db_id = 41701) then
   begin
    if (is_byudjet = 1 and
        coalesce(:ca_id, 0) > 0 and
        exists(select * from extinfolist eil
               where eil.eil_from_eic_id = 10 and
                     eil.eil_from_c_id = :c_id and
                     eil.eil_is_json = 1 and
                     eil.eil_status = 2 and
                     coalesce(eil.eil_from_un_id, 0) = 0)) then
    begin
      select iif(:lang = 'kg', cq.cq_name_kg, cq.cq_name_ru),
             caa.caa_value
      from cfp_anketa_answers caa
      inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
                                     cq.cq_code_name = 'salary_certificate'
      where caa.caa_from_ca_id = :ca_id
      into cq_name, salary_certificate;

      if (coalesce(salary_certificate, 0) > 0) then
      begin
        res_html = blob_append(res_html, f_html_table_row('<b>' || :cq_name || '</b>', :salary_certificate, '', '', -1, -1, -1, -1, 2, ''));
        net_profit = coalesce(net_profit, 0) + salary_certificate;
      end
    end
    repayment_obligations = 0;
    if (have_spouse_salary = 1) then
    begin
      k_id_tmp = 0;
      kbm_id = 0;
      sum_debt = 0;
      select crel_other_c_id
      from hp_client_relation_list(:c_id, 0)
      where crel_from_rel_id in ('1', '2') rows 1
      into :crel_other_c_id;
      for select ks.ks_from_k_id
          from knclient kc
          inner join kontstate ks on kc.kc_from_k_id = ks.ks_from_k_id and
                ks.ks_state = 1 and
                current_date between ks.ks_date_begin and ks.ks_date_end
          where kc.kc_from_c_id = :crel_other_c_id and
                kc.kc_amount_approved > 0 and
                kc.kc_from_k_id <> :k_id
          into k_id_tmp
      do
      begin
        select sum(zad_os + zad_perc + sh_calc_perc + sh_calc_fine) / coalesce(count(*), 1)
        from aarep_make_zad_list(:k_id_tmp, current_date, 1)
        into sum_debt;
        repayment_obligations = repayment_obligations + :sum_debt;
      end
      --RFCKG-5479
      select kbm_id
      from hp_kibminiapp_last(:c_id, :ma_id)
      into :kbm_id;
      if (coalesce(kbm_id, 0) > 0) then
        repayment_obligations = repayment_obligations + coalesce((select sum((kh.kh_unpaidamount + kh.kh_unpaidpercenamount) / iif(datediff(month, current_date, kh.kh_datestop) > 0, datediff(month, current_date, kh.kh_datestop), 1))
                                                                  from kibhistory kh
                                                                  where kh.kh_from_kbm_id = :kbm_id and
                                                                        kh.kh_is_our = 0 and
                                                                        kh.kh_is_debtor = 1 and
                                                                        kh.kh_creditstatus not containing 'settled'), 0);
    end
    kbm_id = 0;
    sum_debt = 0;
    for select ks.ks_from_k_id
        from knclient kc
        inner join kontstate ks on kc.kc_from_k_id = ks.ks_from_k_id and
              ks.ks_state = 1 and
              current_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_date_begin < current_date
        where kc.kc_from_c_id = :c_id and
              kc.kc_amount_approved > 0 and
              kc.kc_from_k_id <> :k_id_tmp
        into k_id
    do
    begin
      select sum(zad_os + zad_perc + sh_calc_perc + sh_calc_fine) / coalesce(count(*), 1)
      from aarep_make_zad_list(:k_id, current_date, 1)
      into sum_debt;
      repayment_obligations = repayment_obligations + :sum_debt;
    end
    --RFCKG-5479
    select kbm_id
    from hp_kibminiapp_last(:c_id, :ma_id)
    into :kbm_id;
    if (coalesce(kbm_id, 0) > 0) then
    begin
      repayment_obligations = repayment_obligations + coalesce((select sum((kh.kh_unpaidamount + kh.kh_unpaidpercenamount) / iif(datediff(month, current_date, kh.kh_datestop) > 0, datediff(month, current_date, kh.kh_datestop), 1))
                                                                from kibhistory kh
                                                                where kh.kh_from_kbm_id = :kbm_id and
                                                                      kh.kh_is_our = 0 and
                                                                      kh.kh_is_debtor = 1 and
                                                                      kh.kh_creditstatus not containing 'settled'), 0);

      net_profit = net_profit + ex_profit - expense - repayment_obligations;
      res_html = blob_append(res_html, f_html_table_row(iif(:lang = 'kg', '<b>Айлык таза киреше: </b>', '<b>Чистая прибыль в месяц: </b>'), :net_profit, '', '', -1, -1, -1, -1, 2, ''));
    end
    else
      res_html = blob_append(res_html, f_html_table_row(iif(:lang = 'kg', '<b>Айлык таза киреше: </b>', '<b>Чистая прибыль в месяц: </b>'), iif(:lang = 'kg', 'КИБ тарабынан маалымат жетишсиз', 'отчет КИБ еще не получен'), '', '', -1, -1, -1, -1, 2, ''));
  end
    res_html = blob_append(res_html, '</table><br><br><br>');
  end

  if (do_action = 'save') then
  begin
    -- RFCKG-5428
    if (coalesce(c_id, -1) = -1) then
      exception e_message 'Не удалось сохранить анкету, т.к. заявка была изменена. Пожалуйста, вернитесь в окно заявки и заполните анкету заново';

    cp_id = f_json_blob_get_value(:data_json, 'job_id', 0);

    if (db_id = 39801) then
    begin
      cp_id = 1;
      -- RFCKZ-871 После сохранения система автоматически должна отправить запрос в сервис расчёта КДН и КДД
      execute procedure hp_kdn_score_query(:mac_id, :un_id);
    end

    select cw_id
    from cib_working_status
    into :cw_id;

    if (coalesce(ca_id, 0) > 0 and
        coalesce(data_json, '') <> '' and
        cp_id > 0) then
        begin
        if (db_id = 41701 and exists(select * from extinfolist eil
                     where eil.eil_from_eic_id = 10 and
                           eil.eil_from_c_id = :c_id and
                           eil.eil_is_json = 1 and
                           eil.eil_status = 2 and
                           coalesce(eil.eil_from_un_id, 0) = 0)) then
           delete from cfp_anketa_answers caa
           where caa.caa_from_ca_id = :ca_id and
                 caa.caa_from_cq_id <> :cq_id;
        else
          delete from cfp_anketa_answers caa
        where caa.caa_from_ca_id = :ca_id;

        end


    if (coalesce(ca_id, 0) > 0) then
    begin
      update cfp_anketa
      set ca_from_un_id = :un_id,
          ca_date = current_date,
          ca_datetime = current_timestamp,
          ca_from_cp_id = :cp_id,
          ca_from_cw_id = :cw_id
      where ca_id = :ca_id;
    end
    else
    begin
      insert into cfp_anketa (ca_object_kind, ca_object_id, ca_from_c_id, ca_date, ca_datetime, ca_from_un_id,
                              ca_from_cp_id, ca_from_cw_id, ca_from_acl_id)
      values (:ca_object_kind, :mac_id, :c_id, current_date, current_timestamp, :un_id, :cp_id, :cw_id, :acl_id)
      returning ca_id
      into :ca_id;
    end

    obj_count = f_json_blob_get_count(:data_json, 'question_list');
    counter = 0;
    while (counter < obj_count) do
    begin
      quest_id = f_json_blob_get_value(:data_json, 'question_list[' || counter || '].id', 0);
      answ_value = f_json_blob_get_value(:data_json, 'question_list[' || counter || '].value', 0);

      if (db_id = 39801 and
          quest_id = 1 and
          exists(select * from clientdocs cds
                 inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
                 inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
                 where cds.cds_from_c_id = :c_id and
                       cds.cds_is_delete = 0 and
                       pdt.pdt_code = 'card_statement' and -- выписка по карте
                       current_date between cds.cds_doc_date_begin and coalesce(cds.cds_doc_date_end, '31.12.2099'))) then
      begin
        select icq.icq_payment_sum
        from income_check_querys icq
        where icq.icq_from_mac_id = :mac_id and
              icq.icq_from_c_id = :c_id and
              icq.icq_status = 2 and
              icq.icq_config = 'salary_check'
        order by icq.icq_create_datetime desc
        rows 1
        into :salary_income;

        if (:salary_income > 0) then
         answ_value = :salary_income;
      end

      if (db_id = 39801 and
          quest_id = 4 and
          not exists(select * from clientdocs cds
                           inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
                           inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
                           where cds.cds_from_c_id = :c_id and
                                 cds.cds_is_delete = 0 and
                                 pdt.pdt_code = 'card_statement' and -- выписка по карте
                                 current_date between cds.cds_doc_date_begin and
                                 coalesce(cds.cds_doc_date_end, '31.12.2099'))) then
      begin
        answ_value = 0;
      end

      if (:quest_id <> :cq_id) then
        insert into cfp_anketa_answers (caa_from_ca_id, caa_from_cq_id, caa_value)
        values (:ca_id, :quest_id, :answ_value);
      else
        update or insert into cfp_anketa_answers (caa_from_ca_id, caa_from_cq_id, caa_value)
        values (:ca_id, :quest_id, :answ_value)
        matching(caa_from_ca_id, caa_from_cq_id);
      counter = counter + 1;
    end

    select cp.cp_full_name_ru || ' ' || coalesce(cp.cp_income_prefix_ru, ''),
           cp.cp_full_name_kg || ' ' || coalesce(cp.cp_income_prefix_kg, '')
    from cfp_positions cp
    where cp.cp_id = :cp_id
    into pos_txt_ru, pos_txt_kg;
    txt_ru = '';
    txt_kg = '';
    avr_income = 1;
    cq_basic_incom = 0;
    for select cq.cq_name_ru, cq.cq_name_kg, caa.caa_value, cq.cq_basic_incom
        from cfp_anketa_answers caa
        inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
              cq.cq_is_delete = 0 and
              cq.cq_basic_incom = 1
        where caa.caa_from_ca_id = :ca_id
        into cq_name_ru, cq_name_kg, caa_value, cq_basic_incom
    do
    begin
      if (db_id = 41701 and coalesce(caa_value, 0) <= 0 and
          cq_basic_incom = 1) then
      begin
        exception e_message 'Не заполнен поле - ' || :cq_name_ru;
        suspend;
        exit;
      end
      if (txt_ru = '') then
        txt_ru = cq_name_ru || ': ' || :caa_value;
      else
        txt_ru = :txt_ru || ', ' || :cq_name_ru || ': ' || :caa_value;
      if (txt_kg = '') then
        txt_kg = cq_name_kg || ': ' || :caa_value;
      else
        txt_kg = :txt_kg || ', ' || :cq_name_kg || ': ' || :caa_value;
      if (db_id = 41701) then
        avr_income = avr_income * :caa_value;
      if (db_id = 39801) then
        avr_income = avr_income + :caa_value;
      when any do
      begin
        exception e_message 'Ошибка. Значение основных доходов неправильно указаны. Слишком большие значения';
        suspend;
        exit;
      end
    end

    update cfp_anketa ca
    set ca.ca_income_txt_ru = coalesce(:pos_txt_ru, '') || ': ' || :txt_ru,
        ca.ca_income_txt_kg = coalesce(:pos_txt_kg, '') || ': ' || :txt_kg,
        ca.ca_avr_income = :avr_income
    where ca.ca_id = :ca_id;

    if (db_id = 41701 and coalesce(:acl_id, 0) <> 0) then
    begin
      update actionlist acl
         set acl.acl_status = 2
      where acl.acl_id = :acl_id;
    end

    if (db_id <> 39801 and coalesce(:acl_id, 0) = 0) then
    begin
      select kpd_err
      from save_cfp_kpd(:mac_id)
      into :kpd_err;

      if (kpd_err <> '') then
      begin
        exception e_message kpd_err;
        suspend;
        exit;
      end
    end

    if (exists(select *
               from kontract k
               inner join kontstate ks on ks.ks_from_k_id = k.k_id and
                                          ks.ks_ol_is_delete = 0 and
                                          ks.ks_state = 1 and
                                          current_date >= ks.ks_date_begin
               where k.k_from_ma_app_id = :ma_id)
        and db_id <> 39801
        and coalesce(:acl_id, 0) = 0) then
    begin

      select kpd_err
      from get_cfp_kpd(:mac_id)
      into err_text;
      if (err_text <> '') then
      begin
        exception e_message err_text;
        suspend;
        exit;
      end
    end
  end
  suspend;
end
