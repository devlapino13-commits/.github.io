CREATE PROCEDURE MC_PURCHASE_JOURNAL(
    MCL_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    MP_ID D_ID,
    MP_AMOUNT D_CURRENCY,
    MP_DATETIME D_DATETIME,
    MP_CATEG D_STR_255,
    MP_RECEIPT_NO D_STR_20,
    MP_STATUS D_INTEGER,
    MP_CL_NAME D_STR_100,
    EDOC_STATUS D_INTEGER,
    PAY_STATUS D_INTEGER)
AS
declare variable c_id d_id;
declare variable k_id d_id;
begin
  select mc.mcl_from_c_id
  from mc_login mc
  where mc.mcl_id = :mcl_id
  into :c_id;
  edoc_status = 1;
  pay_status = 1;

  -- процедура отображает операции куонтрагента за указанный период (требуется доработка)
  for select mp.mp_id, mp.mp_amount, mp.mp_datetime, feg.feg_name, mp.mp_receipt_number, mp.mp_status,
             coalesce(c.c_name_fio, ''), exa.exa_from_k_id
      from mc_purchase mp
      left join mc_login mcl on mcl.mcl_id = mp.mp_from_mcl_id
      left join clients c on c.c_id = mcl.mcl_from_c_id
      inner join foreign_econom_goods_ref feg on feg.feg_id = mp.mp_from_feg_id
      inner join clientshops cs on cs.cs_id = mp.mp_from_cs_id
      left join expressapp exa on exa.exa_id = mp.mp_from_exa_id
      where cs.cs_from_c_id = :c_id and
            mp.mp_date between :rep_date_begin and :rep_date_end and
            (mp.mp_is_finished = 1 or
            (mp.mp_is_finished = 0 and
            mp.mp_status = 0))
      order by mp.mp_datetime desc
      into :mp_id, :mp_amount, :mp_datetime, :mp_categ, :mp_receipt_no, :mp_status, :mp_cl_name, :k_id
  do
  begin
    if (coalesce(:k_id, 0) = 0) then
    begin
      edoc_status = 1;
      pay_status = 1;
    end
    else
    begin
      select
             case coalesce(edl.edl_status, -2)
               when -2 then -1
               when -1 then 0
               when 0 then 1
               when 1 then 2
             end
      from edoc_list edl
      where edl.edl_object_kind = f_objkind('kontract', 'id') and
            edl.edl_object_id = :k_id and
            edl.edl_blank_status = 1
      into edoc_status;
    end

    select
           case coalesce(esm.esm_status, -4)
             when -4 then 1
             when 1 then 2
             when 0 then 1
             when -3 then -1
             when -1 then -1
           end
    from extsysmain esm
    where esm.esm_project = 'elcard_disb' and
          esm.esm_action = 'CardCheck' and
          esm.esm_object_kind = 0 and
          esm.esm_object_id = :c_id and
          esm.esm_from_k_id = :k_id
    into :pay_status;

    if (mp_status in (0, -1)) then
      pay_status = 0;

    suspend;
  end
end
