CREATE PROCEDURE AAREP_ANALIT_COMPARE_DAYS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RC_VALUE_01 VARCHAR(100),
    RC_VALUE_02 VARCHAR(100),
    RC_VALUE_03 VARCHAR(100),
    RC_VALUE_04 VARCHAR(100),
    RC_VALUE_05 VARCHAR(100),
    RC_VALUE_06 VARCHAR(100),
    RC_VALUE_07 VARCHAR(100),
    RC_VALUE_08 VARCHAR(100),
    RC_VALUE_09 VARCHAR(100),
    RC_VALUE_10 VARCHAR(100),
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AAREP_ACTIVE_BALANCE_E DECIMAL(18,4),
    AAREP_ACTIVE_KOLICH_CLIENT_E INTEGER,
    AAREP_DELIQ_BALANCE DECIMAL(18,4),
    AAREP_DELIQ_KOLICH_CLIENT INTEGER,
    AAREP_DELIQ_BALANCE7 DECIMAL(18,4),
    AAREP_DELIQ_KOLICH_CLIENT7 INTEGER,
    AAREP_DELIQ_BALANCE30 DECIMAL(18,4),
    AAREP_DELIQ_KOLICH_CLIENT30 INTEGER)
AS
declare variable dd date;
begin
  dd = :rep_date_begin;
  while (dd <= :rep_date_end) do
  begin
    for select rc_value_01, rc_value_02, rc_value_03, rc_value_04, rc_value_05, rc_value_06, rc_value_07, rc_value_08,
               rc_value_09, rc_value_10, aa_curr_office_name,

               sum(aa_amount_active_end), sum(aa_kolich_cl_active_end),

               sum(aa_deliq_balance), sum(aa_deliq_cl_kolich),

               sum(aa_deliq_balance7), sum(aa_deliq_cl_kolich7),

               sum(aa_deliq_balance30), sum(aa_deliq_cl_kolich30)

        from aarep_analitica_port_temp(:rep_region_id, :dd, :dd, '', 0, 0, 0, 0, 0, 0, 0)
        group by rc_value_01, rc_value_02, rc_value_03, rc_value_04, rc_value_05, rc_value_06, rc_value_07, rc_value_08, rc_value_09, rc_value_10, aa_curr_office_name
        into :rc_value_01, :rc_value_02, :rc_value_03, :rc_value_04, :rc_value_05, :rc_value_06, :rc_value_07,
             :rc_value_08, :rc_value_09, :rc_value_10, :aa_curr_office_name,

             :aarep_active_balance_e, :aarep_active_kolich_client_e,

             :aarep_deliq_balance, :aarep_deliq_kolich_client,

             :aarep_deliq_balance7, :aarep_deliq_kolich_client7,

             :aarep_deliq_balance30, :aarep_deliq_kolich_client30

    do
    begin
      suspend;
    end
    dd = dd + 1;
  end

end
