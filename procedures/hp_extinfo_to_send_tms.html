CREATE PROCEDURE HP_EXTINFO_TO_SEND_TMS(
    EXA_ID D_INTEGER)
RETURNS (
    JSON_VALUE D_BLOB_TEXT)
AS
declare variable INN D_STR_100;
declare variable C_ID D_ID;
declare variable K_ID D_ID;
declare variable UID D_STR_CODE;
declare variable EMA_ID D_ID;
declare variable PHONE D_STR_CODE;
declare variable STATUS D_STR_CODE;
begin
  select exa_from_c_id, exa_from_k_id from expressapp
  where exa_id = :exa_id
  into c_id, k_id;

  ema_id = null;
  select ema_id from kontract, extminiapp
  where k_id = :k_id
  and k_from_tkt_id = ema_from_tkt_id
  into ema_id;

  if (ema_id is not null) then
    uid = f_uuid('extminiapp', ema_id);
  else uid = 'error';

  phone = null;
  select ok_cell_main from continfo
  where cci_object_id = :c_id
  and cci_object_kind = 0
  into phone;
  if (phone is not null) then
  begin
    if (position('+', phone , 1) = 0) then phone = '+'||phone;
  end else phone = 'error';

  if (uid <> 'error' and phone <> 'error') then status = 'success';
  else status = 'cancelled';

  json_value = '{' ||f_json_make_pair('uid', :uid, 'str', 1)
                   ||f_json_make_pair('status', :status, 'str', 1)
                   ||f_json_make_pair('phone', :phone, 'str', 0)
                   ||'}';
  suspend;
end
