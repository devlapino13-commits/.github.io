CREATE PROCEDURE AAREP_INSTALLMENT_GOODS_DISB(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    K_KONTRACT_NO D_STR_20,
    REG_NAME D_STR_64,
    KC_AMOUNT_APPROVED CURR_TYPE,
    KS_DATE_BEGIN D_DATE,
    FEG_NAME D_STR_100,
    IS_NEW_CLIENT D_STR_10)
AS
declare variable MA_ID D_INTEGER;
declare variable K_ID ID_TYPE;
declare variable C_ID ID_TYPE;
declare variable KS_REGION_ID D_INTEGER;
declare variable KC_CIRCLE D_INTEGER;
begin
  for select kn.k_from_ma_app_id,
             kn.k_id,
             kn.k_kontract_no,
             ks.ks_region_id,
             ks.ks_date_begin,
             kc.kc_from_c_id,
             kc.kc_circle,
             kc.kc_amount_approved
      from kontract kn
      inner join kontstate ks on ks.ks_from_k_id = kn.k_id
      inner join product pr on pr.pr_id = kn.k_from_pr_id
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            ks.ks_date_begin = :rep_date_end and
            kc.kc_amount_approved > 0 and
            pr.pr_code like '%.installment_goods.%'
      into :ma_id,
           :k_id,
           :k_kontract_no,
           :ks_region_id,
           :ks_date_begin,
           :c_id,
           :kc_circle,
           :kc_amount_approved
  do
  begin
    select cl.c_name_fio,
           cl.c_inn
    from clients cl
    where cl.c_id = :c_id
    into :c_name_fio,
         :c_inn;

    select reg.reg_name
    from region reg
    where reg.reg_unicode = :ks_region_id
    into :reg_name;

    select feg.feg_name
    from foreign_econom_goods_ref feg
    inner join mc_purchase mp on mp.mp_from_feg_id = feg.feg_id
    where mp.mp_from_ma_id = :ma_id
    into :feg_name;

    if (kc_circle > 1) then
      is_new_client = 'Повторный';
    else
      is_new_client = 'Новый';

    suspend;
  end
end
