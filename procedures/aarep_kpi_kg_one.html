CREATE PROCEDURE AAREP_KPI_KG_ONE(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_LO_ID ID_TYPE)
RETURNS (
    KOL_CLI_BEGIN INTEGER,
    KOL_CLI_END INTEGER,
    BALANCE_BEGIN CURR_TYPE,
    BALANCE_END CURR_TYPE,
    KOL_DISB_RE INTEGER,
    KOL_CLOSED INTEGER,
    KOL_CLOSED_WOFF INTEGER,
    KOL_DISB_ALL INTEGER,
    PLAN_OS CURR_TYPE,
    FACT_OS CURR_TYPE,
    DELINQ_BAL_1 CURR_TYPE,
    PAR1 CURR_TYPE,
    PLAN_CLI INTEGER,
    FACT_CLI INTEGER,
    MARK_CLI_PRIROST DECIMAL(15,2),
    PLAN_KP_PERC_MIN CURR_TYPE,
    PLAN_KP_PERC_MAX CURR_TYPE,
    PRIROST_KP CURR_TYPE,
    PRIROST_KP_PERC CURR_TYPE,
    MARK_KP DECIMAL(15,2),
    PERC_UDERJANIE_CLI DECIMAL(15,2),
    MARK_CLI_UDERJ DECIMAL(15,2),
    PLAN_DISB_CLI INTEGER,
    MARK_CLI_DISB DECIMAL(15,2),
    VOZVRAT_PERC DECIMAL(15,2),
    MARK_VOZVRAT DECIMAL(15,2),
    MARK_PAR DECIMAL(15,2),
    MARK_AVG DECIMAL(15,2))
AS
declare variable REG_ID integer;
declare variable C_ID ID_TYPE;
declare variable KOL_DISB integer;
declare variable KOL_CLOSE integer;
begin
--  прирост клиентов
--  выданные минус закрытые кредиты
--  расчет выданных и закрытых кредитов

--  Рост КП
--  Портфель на конец минус портфель на начало
--  Расчет портфеля на конец
--  Расчет PAR>1
  balance_end = null;
  kol_cli_end = null;
  delinq_bal_1 = null;
  select sum(oks_balance), sum(k_kolich_client_approved),
         sum(iif(:rep_date_end - oks_delinq_date >= 1, oks_balance, 0))
  from oksummary, kontract
  where :rep_date_end between oks_date_begin and oks_date_end and
        oks_kc_id = :rep_lo_id and
        oks_balance > 0 and
        oks_from_k_id = k_id
        and oks_close_state = 0 
  into balance_end, kol_cli_end, delinq_bal_1;
  balance_end = coalesce(balance_end, 0);
  kol_cli_end = coalesce(kol_cli_end, 0);
  delinq_bal_1 = coalesce(delinq_bal_1, 0);

--  Расчет портфеля на начало
  balance_begin = null;
  select sum(oks_balance), sum(k_kolich_client_approved)
  from oksummary, kontract
  where :rep_date_begin - 1 between oks_date_begin and oks_date_end and
        oks_kc_id = :rep_lo_id and
        oks_balance > 0 and
        oks_from_k_id = k_id
        and oks_close_state = 0 
  into balance_begin, kol_cli_begin;
  balance_begin = coalesce(balance_begin, 0);
  kol_cli_begin = coalesce(kol_cli_begin, 0);

--  Удержание клиентов
--    4. Кол-во выданных повторных кредитов (за период) / (Кол-во закрытых кредитов (за период) - Кол-во списанных кредитов (за период))
  kol_disb_re = null;
  kol_closed = null;
  kol_closed_woff = null;
  kol_disb_all = null;
  select sum(iif(ks_state = 1 and kc_circle > 1, 1, 0)), -- выданные повторные
         sum(iif(ks_state = 2, 1, 0)), -- закрытые все
         sum(iif(ks_state = 2 -- закрытые со списанием
         and exists(select *
                    from operkred
                    where ok_from_k_id = ks_from_k_id and
                          ok_ol_is_active = 1 and
                          ok_from_ol_id = ks_from_ol_id and
                          ok_type = 5), 1, 0)), sum(iif(ks_state = 1, 1, 0)) -- кол-во выдач
  from kontstate, knclient
  where ks_ol_is_delete = 0 and
        ks_date_begin between :rep_date_begin and :rep_date_end and
        ks_state > 0 and
        ks_kc_id = :rep_lo_id and
        ks_from_k_id = kc_from_k_id and
        kc_amount_approved > 0
  into kol_disb_re, kol_closed, kol_closed_woff, kol_disb_all;
  kol_disb_re = coalesce(kol_disb_re, 0);
  kol_closed = coalesce(kol_closed, 0);
  kol_closed_woff = coalesce(kol_closed_woff, 0);
  kol_disb_all = coalesce(kol_disb_all, 0);

--  Возвратность кредита
  plan_os = null;
  fact_os = null;
  with tabl1
  as (select oks_from_k_id as k_id
      from oksummary
      where :rep_date_begin - 1 between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_kc_id = :rep_lo_id
            and oks_close_state = 0 ),
  tabl2
  as (select ok_from_k_id as k_id2, sum(ok_os + ok_sh_os) as plan_os
      from operkred, tabl1
      where ok_from_k_id = k_id and
            ok_ol_is_active = 1 and
            ok_ol_date < :rep_date_begin and
            ok_sh_date between :rep_date_begin and :rep_date_end and
            ok_type <> 8
      group by ok_from_k_id
      having sum(ok_os + ok_sh_os) <> 0),
  tabl3
  as (select tabl2.*,
             (select -sum(ok_os)
              from operkred
              where ok_from_k_id = k_id2 and
                    ok_ol_is_active = 1 and
                    ok_type = 4 and
                    ok_ol_date between :rep_date_begin and :rep_date_end and
                    ok_sh_date between :rep_date_begin and :rep_date_end) as fact_os
      from tabl2)
  select sum(plan_os) as plan_os, sum(iif(plan_os < fact_os, plan_os, fact_os)) as fact_os
  from tabl3
  into plan_os, fact_os;
  plan_os = coalesce(plan_os, 0);
  fact_os = coalesce(fact_os, 0);

  if (balance_end > 0) then
    par1 = cast(delinq_bal_1 as double precision) / balance_end * 100;
  else
    par1 = 0;

  plan_cli = case
               when kol_cli_begin <= 100 then 20
               when kol_cli_begin <= 150 then 17
               when kol_cli_begin <= 200 then 15
               when kol_cli_begin <= 250 then 12
               when kol_cli_begin <= 300 then 10
               when kol_cli_begin <= 350 then 8
               when kol_cli_begin <= 400 then 6
               else 4
             end;
  fact_cli = kol_cli_end - kol_cli_begin;

  if (fact_cli >= plan_cli) then
    mark_cli_prirost = 5;
  else
  if (fact_cli <= 0) then
    mark_cli_prirost = 0;
  else
    mark_cli_prirost = cast(fact_cli as double precision) / plan_cli * 5;

  if (balance_begin <= 5000000.0000) then
  begin
    plan_kp_perc_min = 3.6;
    plan_kp_perc_max = 9.0;
  end
  else
  if (balance_begin <= 8000000.0000) then
  begin
    plan_kp_perc_min = 1.8;
    plan_kp_perc_max = 3.6;
  end
  else
  if (balance_begin <= 17000000.0000) then
  begin
    plan_kp_perc_min = 0.0;
    plan_kp_perc_max = 1.8;
  end
  else
  begin
    plan_kp_perc_min = -1.0;
    plan_kp_perc_max = 0.0;
  end

  prirost_kp = balance_end - balance_begin;
  if (balance_begin > 0) then
    prirost_kp_perc = cast(prirost_kp as double precision) / balance_begin * 100;
  else
    prirost_kp_perc = 0;

  if (prirost_kp_perc >= plan_kp_perc_max) then
    mark_kp = 5;
  else
  if (prirost_kp_perc < plan_kp_perc_min) then
    mark_kp = 0;
  else
    mark_kp = cast(5 as double precision) * ((prirost_kp_perc - plan_kp_perc_min) / (plan_kp_perc_max - plan_kp_perc_min));

  if (kol_closed - kol_closed_woff > 0) then
    perc_uderjanie_cli = cast(kol_disb_re as double precision) / (kol_closed - kol_closed_woff) * 100;
  else
    perc_uderjanie_cli = 0;

  if (perc_uderjanie_cli >= 85.00) then
    mark_cli_uderj = 5;
  else
  if (perc_uderjanie_cli < 50.00) then
    mark_cli_uderj = 0;
  else
    mark_cli_uderj = cast(5 as double precision) * (perc_uderjanie_cli - 50.00) / 35;

  plan_disb_cli = iif(kol_cli_begin < 200, 25, 40);

  if (kol_disb_all >= plan_disb_cli) then
    mark_cli_disb = 5;
  else
    mark_cli_disb = cast(5 as double precision) * kol_disb_all / plan_disb_cli;

  if (plan_os > 0) then
    vozvrat_perc = fact_os / plan_os * 100;
  else
    vozvrat_perc = 0;

  if (vozvrat_perc >= 85.00) then
    mark_vozvrat = cast(5 as double precision) * (vozvrat_perc - 85.00) / 15.00;
  else
    mark_vozvrat = 0;

  if (balance_end > 0) then
  begin
    if (par1 <= 1.00) then
    begin
      mark_par = 5 - 2 * par1;
    end
    else
    if (par1 <= 2.00) then
    begin
      mark_par = 3 - 3 * (par1 - 1);
    end
    else
    begin
      mark_par = 0;
    end
  end
  else
  begin
    mark_par = 0;
  end

  mark_avg = (mark_par + mark_vozvrat + mark_cli_disb + mark_cli_uderj + mark_kp + mark_cli_prirost) / 6;

  suspend;
end
