CREATE PROCEDURE HP_BKI_GET_SURVEY_CONTENT(
    C_ID D_ID,
    REP_DATE D_DATE = null)
RETURNS (
    STR_NO D_INTEGER,
    QUESTION_STR D_STR_255,
    ANSWER_STR D_STR_255)
AS
declare variable bls_id d_id;
declare variable active_cnt d_integer;
declare variable cred_sum_debt d_currency;
declare variable last_summ d_currency;
declare variable last_active_summ d_currency;
declare variable last_year d_integer;
declare variable last_active_year d_integer;
declare variable last_is_our d_bool;
declare variable last_active_is_our d_bool;
begin
  if (rep_date is null) then
    rep_date = current_date;

  select eil_from_bls_id
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id
    and eil_from_eic_id = eic_id
    and eic_code = 'equifax_40'
    and eil_status = 2
    and eil_date between (:rep_date - eic_info_actual_days) and :rep_date
  order by eil_date desc, eil_id
  rows 1
  into bls_id;

  if (bls_id is null) then
  begin
    question_str = 'Кредитная история не получена';
    suspend;
    exit;
  end

  active_cnt = 0;

  for select cred_sum_debt, cred_sum, extract (year from cred_date),
      iif(exists(select 1 from hp_find_cbr_uuid(cred_uid) where obj_id is not null), 1, 0)
  from aarep_equifax_cred(:bls_id, 'equifax_40')
  order by cred_date
  into cred_sum_debt, last_summ, last_year, last_is_our
  do
  begin
    if (cred_sum_debt > 0) then
    begin
      active_cnt = active_cnt + 1;
      last_active_summ = last_summ;
      last_active_year = last_year;
      last_active_is_our = last_is_our;
    end 
  end

  str_no = 1;
  question_str = 'Сколько у вас активных займов сейчас?';
  answer_str = active_cnt;
  suspend;

  str_no = 2;
  question_str = 'На какую сумму был получен последний займ?';
  answer_str = f_numtostr(coalesce(last_active_summ, last_summ), 2);
  suspend;

  str_no = 3;
  question_str = 'В каком году был получен последний займ?';
  answer_str = coalesce(last_active_year, last_year);
  suspend;

  str_no = 4;
  question_str = 'Ваш последний заем был получен в ООО "МКК М Булак" или в другой организации';
  answer_str = iif(coalesce(last_active_is_our, last_is_our) = 1, 'да, в М Булак', 'нет, не в М Булак');
  suspend;
end
