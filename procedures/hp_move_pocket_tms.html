CREATE PROCEDURE HP_MOVE_POCKET_TMS(
    REP_DATE DATE,
    REG_ID INTEGER,
    WP_REG_ID INTEGER,
    WP_ID DECIMAL(18,0),
    WP_TO DECIMAL(18,0),
    MO_FROM DECIMAL(18,0),
    MO_TO DECIMAL(18,0),
    OL_MEMO VARCHAR(1024),
    UN_ID DECIMAL(18,0))
RETURNS (
    OL_ID_1 DECIMAL(18,0))
AS
declare variable wp_name varchar(100);
declare variable reg_loronostro_head integer;
declare variable operlist_count integer;
declare variable doc_no integer;
declare variable om_account varchar(20);
declare variable om_amount_tms decimal(18,4);
declare variable om_amount_amort decimal(18,4);
declare variable acc_dt varchar(20);
declare variable acc_kt varchar(20);
declare variable ol_id decimal(18,0);
declare variable line_no integer;
declare variable ob_id integer;
declare variable cm_default_currency varchar(3);
declare variable wp_id_new decimal(18,0);
declare variable key_id integer;
declare variable corr_reg integer;
declare variable ol_id_3 decimal(18,0);
declare variable ol_id_2 decimal(18,0);
begin
  -- пакетный перенос ТМЦ из одного место другому, или из одного сотрудника другому
  -- при этом офисы могут быть разные
  -- Sanjar, 30/06/2013

  select cm_default_currency
  from company
  into :cm_default_currency;

  /*  REG_ID -- отправляющий
  WP_REG_ID -- принимающий

  WP_ID - откуда передается (рабочее место)
  WP_TO - куда передается (рабочее место)

  */
  if (reg_id <> wp_reg_id) then
  begin
    select reg_loronostro_head
    from region
    where reg_unicode = :reg_id
    into :reg_loronostro_head;

    if ((reg_id <> reg_loronostro_head) and
        (wp_reg_id <> reg_loronostro_head)) then
      operlist_count = 3; -- постредник ГО (БРУ)
    else
      operlist_count = 2; -- один из них ГО (БРУ)
    if (operlist_count = 2) then
    begin
      select kv
      from get_kv(1090100, 5)
      into :ol_id_1;
      ol_id_2 = 0;
      select kv
      from get_kv(1090100, 5)
      into :ol_id_3;
    end
    else
    begin
      select kv
      from get_kv(1090100, 5)
      into :ol_id_1;
      select kv
      from get_kv(1090100, 5)
      into :ol_id_2;
      select kv
      from get_kv(1090100, 5)
      into :ol_id_3;
    end
  end
  else
  begin
    operlist_count = 1;
    select kv
    from get_kv(1090100, 5)
    into :ol_id_1;
    ol_id_2 = 0;
    ol_id_3 = 0;
  end
  doc_no = 0;
  select num
  from last_doc_number(:rep_date, 10, :reg_id)
  into :doc_no;

  if (reg_id <> wp_reg_id) then
    corr_reg = wp_reg_id;
  else
    corr_reg = null;
  insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                        ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                        ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                        ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_parent_id)
  values (:ol_id_1, :reg_id, 3, :rep_date, :doc_no, '', 10, :ol_memo, 0, 0, null, :un_id, null, null, 0, :corr_reg,
          null, null, 5, :rep_date, 0, null, 0, null);

  if (ol_id_2 <> 0) then
  begin
    insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                          ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                          ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                          ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_parent_id)
    values (:ol_id_2, :reg_loronostro_head, 3, :rep_date, :doc_no, '', 10, :ol_memo, 0, 0, null, :un_id, null, null, 0,
            :reg_id, null, null, 5, :rep_date, 0, null, 0, :ol_id_1);
  end
  if (reg_id <> wp_reg_id) then
    corr_reg = reg_id;
  else
    corr_reg = null;

  if (ol_id_3 <> 0) then
  begin
    insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                          ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                          ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                          ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_parent_id)
    values (:ol_id_3, :wp_reg_id, 3, :rep_date, :doc_no, '', 10, :ol_memo, 0, 0, null, :un_id, null, null, 0, :corr_reg,
            null, null, 5, :rep_date, 0, null, 0, :ol_id_1);
  end
  ob_id = 0;

  for select om_account, sum(om_amount_tms), sum(om_amount_amort)
      from hp_move_tms_detail(:rep_date, :ol_id_1, :ol_id_3, :reg_id, :wp_reg_id, :wp_id, :wp_to, :mo_from, :mo_to)
      group by om_account
      into :om_account, :om_amount_tms, :om_amount_amort
  do
  begin
    if (om_amount_tms <> 0) then -- Передача стоимости
    begin
      ob_id = ob_id + 1;
      acc_kt = :om_account;
      select reg_code || right(:om_account, 14)
      from region
      where reg_unicode = :wp_reg_id
      into :acc_dt;

      line_no = 0;
      for select res_acc_dt, res_acc_kt
          from aarep_loronostro_link(:reg_id, :acc_dt, :acc_kt)
          into :acc_dt, :acc_kt
      do
      begin
        line_no = line_no + 1;
        if (line_no = 1) then
          ol_id = :ol_id_1;
        if (line_no = 2) then
        begin
          if (operlist_count = 2) then
            ol_id = :ol_id_3;
          else
            ol_id = :ol_id_2;
        end
        if (line_no = 3) then
          ol_id = :ol_id_3;

        insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                              ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                              ob_client_kt, ob_detail, ob_from_sac_id)
        values (:ol_id, :ob_id, :cm_default_currency, 1, null, :acc_dt, :om_amount_tms, :om_amount_tms, null, null,
                null, :acc_kt, null, 'Выбытие ТМЦ с рабочего места по офисам', null);
      end
    end

    if (om_amount_amort <> 0) then -- Передача износа
    begin
      ob_id = ob_id + 1;
      acc_dt = :om_account;
      select reg_code || right(:om_account, 14)
      from region
      where reg_unicode = :wp_reg_id
      into :acc_kt;

      line_no = 0;
      for select res_acc_dt, res_acc_kt
          from aarep_loronostro_link(:reg_id, :acc_dt, :acc_kt)
          into :acc_dt, :acc_kt
      do
      begin
        line_no = line_no + 1;
        if (line_no = 1) then
          ol_id = :ol_id_1;
        if (line_no = 2) then
        begin
          if (operlist_count = 2) then
            ol_id = :ol_id_3;
          else
            ol_id = :ol_id_2;
        end
        if (line_no = 3) then
          ol_id = :ol_id_3;

        insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                              ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                              ob_client_kt, ob_detail, ob_from_sac_id)
        values (:ol_id, :ob_id, :cm_default_currency, 1, null, :acc_dt, :om_amount_amort, :om_amount_amort, null, null,
                null, :acc_kt, null, 'Выбытие ТМЦ с рабочего места по офисам', null);
      end
    end
  end
  suspend;
end
