CREATE PROCEDURE HP_GC_ANALIZ_DETAIL_POS_VIEW(
    IGA_ID D_INTEGER)
RETURNS (
    IGDP_ID D_ID,
    IGDP_REC_NO D_INTEGER,
    IGDP_OBJECT_KIND D_INTEGER,
    IGDP_OBJECT_ID D_ID,
    IGDP_FROM_C_ID D_ID,
    C_NAME_FIO D_STR_100,
    IGDP_FROM_AG_ID D_INTEGER,
    IGDP_TERMINAL_ID D_INTEGER,
    TRM_INVENT_STR D_STR_CODE,
    CS_NAME D_STR_255,
    REG_NAME D_STR_CODE,
    IGDP_TYPE D_SMALLINT,
    IGDP_FROM_IB_ID D_INTEGER,
    BANK_NAME D_STR_255,
    IGDP_IS_OUR_CARD D_BOOL,
    IGDP_CURRENCY D_STR_LANG,
    IGDP_AMOUNT D_CURRENCY,
    IGDP_TSP_COMMIS_PERC D_PERCENT,
    IGDP_TSP_COMMIS_AMOUNT D_CURRENCY,
    IGDP_IPC_EMITENT_COMMIS_PERC D_PERCENT,
    IGDP_IPC_EMITENT_COMMIS_AMOUNT D_CURRENCY,
    IGDP_TRANS_NO D_STR_10,
    IGDP_TRANS_DATE D_DATE,
    IGDP_INET_ACQUIRING D_SMALLINT,
    IGDP_IS_ERROR D_BOOL,
    IGDP_ERROR_ID D_INTEGER,
    ERR_NAME D_STR_255)
AS
declare variable object_id d_id;
begin
  for
    select
      igdp_id,
      igdp_rec_no,
      igdp_object_kind,
      igdp_object_id,
      igdp_from_c_id,
      igdp_from_ag_id,
      igdp_terminal_id,
      igdp_type,
      igdp_from_ib_id,
      igdp_is_our_card,
      igdp_currency,
      igdp_amount,
      igdp_tsp_commis_perc,
      igdp_tsp_commis_amount,
      igdp_ipc_emitent_commis_perc,
      igdp_ipc_emitent_commis_amount,
      igdp_trans_no,
      igdp_trans_date,
      igdp_inet_acquiring,
      igdp_is_error,
      igdp_error_id
    from ipc_gc_analiz_detail_pos
    where igdp_from_iga_id = :iga_id
  into
    :igdp_id,
    :igdp_rec_no,
    :igdp_object_kind,
    :igdp_object_id,
    :igdp_from_c_id,
    :igdp_from_ag_id,
    :igdp_terminal_id,
    :igdp_type,
    :igdp_from_ib_id,
    :igdp_is_our_card,
    :igdp_currency,
    :igdp_amount,
    :igdp_tsp_commis_perc,
    :igdp_tsp_commis_amount,
    :igdp_ipc_emitent_commis_perc,
    :igdp_ipc_emitent_commis_amount,
    :igdp_trans_no,
    :igdp_trans_date,
    :igdp_inet_acquiring,
    :igdp_is_error,
    :igdp_error_id
  do
  begin
    c_name_fio = null;
    trm_invent_str = null;
    cs_name = null;
    reg_name = null;
    bank_name = null;
    err_name = null;

    if (igdp_from_c_id > 0) then
    begin
      if (igdp_type = 1) then
        c_name_fio = (select c_name_fio from clients where c_id = :igdp_from_c_id);
      else
        c_name_fio = 'М Булак';
    end

    if (igdp_terminal_id > 0) then
    begin
      select
        trm_invent_str,
        ito_object_id
      from ipc_terminals
        inner join ipc_terminal_ownership
          on ito_trm_id = trm_id
            and :igdp_trans_date between ito_date_begin and ito_date_end
      where trm_id = :igdp_terminal_id
      into
        :trm_invent_str,
        :object_id;

      if (igdp_type = 1) then
        cs_name = (select cs_name from clientshops where cs_id = :object_id);
      else
        reg_name = (select reg_name from region where reg_unicode = :object_id);
    end

    if (igdp_from_ib_id > 0) then
      bank_name = (select ib_ipc_bank_name from ipc_banks where ib_id = :igdp_from_ib_id);

    if (igdp_error_id > 0) then
      err_name = (select rd_name from refdata where rd_id = :igdp_error_id);

    if (coalesce(igdp_terminal_id, 0) = 0) then
      trm_invent_str =
        case igdp_inet_acquiring
          when 1 then 'TS210001'
          when 2 then 'IA210001'
        end;

    suspend;
  end
end
