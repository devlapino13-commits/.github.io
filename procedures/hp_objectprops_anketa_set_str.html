CREATE PROCEDURE HP_OBJECTPROPS_ANKETA_SET_STR(
    ACL_OBJECT_KIND D_INTEGER,
    ACL_OBJECT_ID D_ID,
    QUESTION_CODE D_STR_100,
    STR_VALUE D_STR_2048,
    UN_ID D_ID)
AS
declare variable question_id d_integer;
declare variable rd_str_value d_integer;
declare variable opr_id d_id;
declare variable ovs_id d_integer;
declare variable ovs_id1 d_integer;
begin
    question_id = f_get_rd_id(:question_code);
    rd_str_value = f_get_rd_id('Results.StrValue');

    execute procedure hp_objectprops_set(
        :acl_object_kind,
        :acl_object_id,
        -1,
        :question_id,
        current_date,
        '',
        0,
        :un_id);

    select
        opr_id,
        opr_actual_ovs_id
    from objectprops
    where opr_object_kind = :acl_object_kind
      and opr_object_id = :acl_object_id
      and opr_subject_kind = -1
      and opr_subject_id = :question_id
    into
        :opr_id,
        :ovs_id;

    ovs_id1 = (
        select max(ovs_id)
        from objectvalues
        where ovs_from_opr_id = :opr_id
    );

    delete from objectvalres
    where ovr_from_opr_id = :opr_id
      and ovr_from_ovs_id = :ovs_id1;

    insert into objectvalres (
        ovr_from_opr_id,
        ovr_from_ovs_id,
        ovr_result_rd_id,
        ovr_state,
        ovr_value
    )
    values (
        :opr_id,
        :ovs_id1,
        :rd_str_value,
        1,
        :str_value
    );
end
