CREATE PROCEDURE MC_CREATE_QR_CODE(
    GPS_LAT D_DOUBLE,
    GPS_LONG D_DOUBLE,
    PHONE_UNIQ_ID D_STR_CODE,
    MCL_ID D_ID)
RETURNS (
    QR_CODE D_STR_CODE,
    LABEL_HEADER D_STR_255,
    LABEL_FOOTER D_STR_255,
    ERR_TEXT D_STR_255)
AS
declare variable qrl_id d_id;
declare variable c_id d_id;
begin
  -- Процедура генерирует QR-код из МП клиента для прохождения QR-верификации

  err_text = '';
  qr_code = '';

  if (coalesce(gps_lat, 0) = 0 or
      coalesce(gps_long, 0) = 0 or
      coalesce(phone_uniq_id, '') = '') then
  begin
    err_text = 'Не удалось сгенерировать QR-код';
    suspend;
    exit;
  end

  if (mcl_id is null) then
    exception e_default 'Неверный вызов'; -- данная процедура должна вызываться только из МП клиента

  -- Защита от регистрации клиентов с одного телефона (Антифрод)
  -- Письмо от Зары: Пт 08.10.2021 12:28
  -- При попытке сгенерировать код для QR-верификации, будет проверка, был ли за
  -- последний месяц вход под другим логином, но с таким же идентификатором устройства.
  -- RFCRU-3259, дополнение, кроме логина должен отличаться c_id
  -- RFCRU-3766
  -- RFCRU-6057
  if (exists(
    select *
    from mc_login m1, mc_loginid, mc_login m2
    where m1.mcl_id = :mcl_id
    and mci_from_mcl_id <> m1.mcl_id
    and mci_dev_id = m1.mcl_dev_id
    and mci_last_datetime > dateadd(day, -30, current_date)
    and mci_from_mcl_id = m2.mcl_id
    and m2.mcl_is_delete = 0
    and m1.mcl_from_c_id is distinct from m2.mcl_from_c_id
    and m1.mcl_project = m2.mcl_project)) then
  begin
    err_text = 'QR-верификация на данном устройстве недоступна, так как с него уже была проведена верификация для другого пользователя';
    suspend;
    exit;
  end

  -- Обработка дедлока
  begin
  -- Аннулируем другие если еще не подтверждены
  update qr_list
  set qrl_is_delete = 1
  where qrl_from_mcl_id = :mcl_id and
        qrl_state = 0 and
        qrl_is_delete = 0;

  insert into qr_list (qrl_from_mcl_id, qrl_gps_lat, qrl_gps_long, qrl_phone_id)
  values (:mcl_id, :gps_lat, :gps_long, :phone_uniq_id)
  returning qrl_id, qrl_from_c_id
  into :qrl_id, :c_id;

  label_header = '';
  label_footer = f_lang(null, 'Покажите экран телефона нашему сотруднику');

  if (c_id is not null) then
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into label_header;
  end else
  begin
    select mcl_login
    from mc_login
    where mcl_id = :mcl_id
    into label_header;
  end

  qr_code = f_uuid('qr_list', qrl_id);
  when sqlcode -913 do
    begin
    -- Если ошибка дедлока
        err_text = 'Не удалось сгенерировать QR-код, повторите попытку через пару минут';
    end

    when any do
    begin
        -- Обработка других ошибок
        err_text = 'Не удалось сгенерировать QR-код';
    end  
  end
  suspend;
end
