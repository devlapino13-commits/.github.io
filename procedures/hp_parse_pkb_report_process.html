CREATE PROCEDURE HP_PARSE_PKB_REPORT_PROCESS(
    EIL_ID D_ID)
AS
declare variable BLS_ID D_ID;
declare variable C_ID D_ID;
declare variable MA_ID D_ID;
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable IDX D_INTEGER;
declare variable BASEPATH D_STR_255;
declare variable ARR_LENGTH D_INTEGER;

-- Переменные для действующих контрактов (ExistingContracts)
declare variable TEMP_OVERDUE_MAX D_STR_10;
declare variable TEMP_OVERDUE_MAX_INT D_INTEGER;

-- Новые переменные для закрытых контрактов (TerminatedContracts)
declare variable TE_TEMP_OVERDUE_MAX D_STR_10;
declare variable TE_TEMP_OVERDUE_MAX_INT D_INTEGER;

declare variable TYPE_OF_LINK D_STR_10;
declare variable TYPE_OF_FOUNDING D_STR_100;
declare variable TMPSTR D_STR_255;
declare variable OUTSTANDING_AMOUNT D_STR_20;
declare variable OVERDUE_MAX D_INTEGER;
declare variable OUT_AMOUNT_SUM D_CURRENCY;
declare variable ACTIVE_KONTRACT_COUNT D_INTEGER;
declare variable PORUCH_KONT_QTY D_INTEGER;
declare variable OVERDUE_7_30 D_INTEGER;
declare variable OVERDUE_31_60 D_INTEGER;
declare variable OVERDUE_61_90 D_INTEGER;
declare variable OVERDUE_91_360 D_INTEGER;
declare variable OVERDUE_360 D_INTEGER;
declare variable NUMBER_OF_CONTRACTS D_INTEGER;
declare variable JSON_SUM_INFO D_BLOB_TEXT;
declare variable JSON_PUBLIC_SOURCES D_BLOB_TEXT;
declare variable IS_GAMBLING D_SMALLINT;
declare variable IS_STOP_CREDIT D_SMALLINT;
declare variable IS_BANKRUPTCY D_SMALLINT;
declare variable IS_TAX_DEBT D_SMALLINT;
declare variable IS_FAKE_BUSINESS D_SMALLINT;
declare variable IS_BAD_SUPPLIER D_SMALLINT;
declare variable IS_WANTED_PERSON D_SMALLINT;
declare variable IS_WANTED_ALIMENT D_SMALLINT;
declare variable IS_BLACK_LIST D_SMALLINT;
declare variable MONTHLY_INS_AMOUNT D_STR_20;
declare variable CNT_OVERDUE_10D D_INTEGER;
declare variable CNT_OVERDUE_60D D_INTEGER;
declare variable TE_CR_HISTORY_COUNT D_INTEGER;
declare variable TOTAL_MONTHLY_INS_AMOUNT D_CURRENCY;
declare variable MONTHLY_INS_AMOUNT_INT D_CURRENCY;
declare variable CREDITOR_TYPE_ID D_STR_10;
declare variable CONTRACT_PHASE_ID D_STR_10;
declare variable OVERDUE_CUR_STR D_STR_20;
declare variable OVERDUE_CUR_INT D_INTEGER;
declare variable ACTIVE_BANK_COUNT D_INTEGER;
declare variable ACTIVE_MFO_COUNT D_INTEGER;
declare variable OVERDUE_COUNT_BANK D_INTEGER;
declare variable OVERDUE_COUNT_MFO D_INTEGER;
begin
  overdue_max = 0;
  out_amount_sum = 0;
  poruch_kont_qty = 0;
  active_kontract_count = 0;
  overdue_7_30 = 0;
  overdue_31_60 = 0;
  overdue_61_90 = 0;
  overdue_91_360 = 0;
  overdue_360 = 0;
  cnt_overdue_10d = 0;
  cnt_overdue_60d = 0;
  total_monthly_ins_amount = 0;
  monthly_ins_amount_int = 0;
  active_bank_count = 0;
  active_mfo_count = 0;
  overdue_count_bank = 0;
  overdue_count_mfo = 0;

  select eil.eil_from_bls_id, eil.eil_from_c_id, eil.eil_object_id
  from extinfolist eil
  where eil.eil_id = :eil_id
  into :bls_id, :c_id, :ma_id;

  select cast(blx_blob as d_blob_text)
  from hp_get_blobex_blob(:bls_id)
  into :json_text;

  -- 1. Сводная информация
  arr_length = f_json_blob_get_count(json_text, 'Report.Root.SummaryInformation.ExistingContracts.SubjectRole');
  json_sum_info = f_json_blob_get_value(json_text, 'Report.Root.SummaryInformation.ExistingContracts', '');
  idx = 0;
  while(idx < arr_length) do
  begin
    basepath = 'SubjectRole[' || idx || ']';
    if (f_json_str_get_value(json_sum_info, basepath || '.id', '') = '1') then
    begin
      tmpstr = f_clean_value(f_json_str_get_value(json_sum_info, basepath || '.TotalOutstandingDebt.value', ''));
      out_amount_sum = case when tmpstr <> '' then cast(tmpstr as decimal(18,2)) else 0 end;

      tmpstr = f_clean_value(f_json_str_get_value(json_sum_info, basepath || '.NumberOfContracts.value', ''));
      if (position('(' in tmpstr) > 0) then
        tmpstr = substring(tmpstr from 1 for position('(' in tmpstr) - 1);
      number_of_contracts = case when tmpstr <> '' then cast(tmpstr as d_integer) else 0 end;

      leave;
    end
    idx = idx + 1;
  end

  -- 2. Действующие контракты (ExistingContracts)
  arr_length = f_json_blob_get_count(json_text, 'Report.Root.ExistingContracts.Contract');
  active_kontract_count = iif(arr_length < 0, 0, arr_length);
  idx = 0;
  while (idx < arr_length) do
  begin
    basepath = 'Report.Root.ExistingContracts.Contract[' || :idx || ']';

    temp_overdue_max  = f_json_blob_get_value(json_text, basepath || '.NumberOfOverdueInstalmentsMax.value', '0');
    type_of_founding  = f_json_blob_get_value(json_text, basepath || '.TypeOfFounding.id', '');
    type_of_link      = f_json_blob_get_value(json_text, basepath || '.InterconnectedSubjects.TypeOfLink.value', '');
    monthly_ins_amount = f_json_blob_get_value(json_text, basepath || '.MonthlyInstalmentAmount.value', '');
    creditor_type_id  = f_json_blob_get_value(json_text, basepath || '.CreditorType.id', '');
    contract_phase_id = f_json_blob_get_value(json_text, basepath || '.ContractPhase.id', '');
    overdue_cur_str   = f_json_blob_get_value(json_text, basepath || '.NumberOfOverdueInstalments.value', '0');

    -- Приведение строк к числу для корректного числового сравнения
    temp_overdue_max_int = iif(temp_overdue_max <> '', cast(temp_overdue_max as integer), 0);
    overdue_cur_int     = iif(overdue_cur_str <> '', cast(overdue_cur_str as integer), 0);

    if (temp_overdue_max_int >= 10) then 
      cnt_overdue_10d = cnt_overdue_10d + 1;
    if (temp_overdue_max_int >= 60) then
      cnt_overdue_60d = cnt_overdue_60d + 1;

    -- Разделение по банкам / МФО
    if (contract_phase_id in ('4', '1', '') or contract_phase_id is null) then
    begin
      if (creditor_type_id = '1') then
      begin
        active_bank_count = active_bank_count + 1;
        if (overdue_cur_int > 30) then 
          overdue_count_bank = overdue_count_bank + 1;
      end
      else if (creditor_type_id = '2') then
      begin
        active_mfo_count = active_mfo_count + 1;
        if (overdue_cur_int > 1) then 
          overdue_count_mfo = overdue_count_mfo + 1;
      end
    end

    monthly_ins_amount_int = case when monthly_ins_amount <> '' then cast(f_clean_value(monthly_ins_amount) as decimal(15,2)) else 0 end;
    total_monthly_ins_amount = total_monthly_ins_amount + monthly_ins_amount_int;

    -- Расчет максимума ПО ДЕЙСТВУЮ�©ИМ КОНТРАКТАМ (числовой maxvalue)
    overdue_max = maxvalue(temp_overdue_max_int, overdue_max);

    if (type_of_link containing 'Поручитель') then
      poruch_kont_qty = poruch_kont_qty + 1;

    idx = idx + 1;
  end

  -- 3. Закрытые контракты (TerminatedContracts) — полностью обособленный блок
  overdue_7_30  = 0;
  overdue_31_60 = 0;
  overdue_61_90 = 0;
  overdue_91_360 = 0;
  overdue_360 = 0;
  te_temp_overdue_max_int = 0;
  te_cr_history_count = 0;

  arr_length = f_json_blob_get_count(json_text, 'Report.Root.TerminatedContracts.Contract');
  te_cr_history_count = iif(arr_length < 0, 0, arr_length);
  if (arr_length > 0) then
  begin
    idx = 0;
    while (idx < arr_length) do
    begin
      -- Используем строго локальные переменные блока TerminatedContracts
      te_temp_overdue_max = f_json_blob_get_value(json_text, 'Report.Root.TerminatedContracts.Contract[' || idx || '].NumberOfOverdueInstalmentsMax.value', '');
      te_temp_overdue_max_int = iif(te_temp_overdue_max <> '', cast(te_temp_overdue_max as integer), 0);

      if (te_temp_overdue_max_int between 7 and 30) then overdue_7_30 = overdue_7_30 + 1;
      else if (te_temp_overdue_max_int between 31 and 60) then overdue_31_60 = overdue_31_60 + 1;
      else if (te_temp_overdue_max_int between 61 and 90) then overdue_61_90 = overdue_61_90 + 1;
      else if (te_temp_overdue_max_int between 91 and 360) then overdue_91_360 = overdue_91_360 + 1;
      else if (te_temp_overdue_max_int > 360) then overdue_360 = overdue_360 + 1;

      idx = idx + 1;
    end
  end

  -- 4. Публичные источники
  json_public_sources = f_json_blob_get_value(json_text, 'Report.Root.PublicSources', '');
  -- информация из доп источников
  is_gambling = iif(f_json_blob_get_value(json_text, 'Report.Root.Gamblers.GamblerFeature', '') = 'true', 1, 0);
  is_stop_credit = iif(f_json_blob_get_value(json_text, 'Report.Root.StopCredit', '') = 'true', 1, 0);
  is_bankruptcy = iif(f_json_blob_get_count(json_text, 'Report.Root.NegativeData.NegativeStatus') >= 0
    and f_json_blob_get_value(json_text, 'Report.Root.NegativeData.NegativeStatus[0].NegativeStatusOfClient.NegativeStatusOfClient.id', '') in ('17','18'), 1, 0);
  is_tax_debt = iif(f_json_blob_get_value(json_public_sources, 'Areears.Status.id', '') = '1', 1, 0);
  is_fake_business = iif(f_json_blob_get_value(json_public_sources, 'FalseBusi.Status.id', '') = '1', 1, 0);
  is_bad_supplier = iif(f_json_blob_get_value(json_public_sources, 'RNUGosZAKUP.Status', '') = '1', 1, 0);
  is_black_list = iif(f_json_blob_get_value(json_public_sources, 'L150o10.Status.id', '') = '1', 1, 0);
  is_wanted_person = iif(f_json_blob_get_value(json_public_sources, 'QamqorList.Status.id', '') = '1', 1, 0);
  is_wanted_aliment = iif(f_json_blob_get_value(json_public_sources, 'QamqorAlimony.Status.id', '') = '1', 1, 0);

  -- 5. Запись результатов
  insert into pkb_report_parsed (prp_from_ma_id, prp_from_c_id, prp_from_eil_id, prp_max_overdue_days,
                                 prp_out_amount_total, prp_active_contract_count, prp_poruch_kont_qty,
                                 prp_overdue_7_30_count, prp_overdue_31_60_count, prp_overdue_61_90_count,
                                 prp_overdue_91_360_count, prp_overdue_over_360_count,
                                 prp_is_gambling, prp_is_stop_credit, prp_is_bankruptcy, prp_is_tax_debt,
                                 prp_is_fake_business, prp_is_bad_supplier, prp_is_black_list, prp_is_wanted_person,
                                 prp_is_wanted_aliment, prp_cnt_overdue_10d, prp_cnt_overdue_60d, prp_total_monthly_ins_amount,
                                 prp_te_cr_history_count, prp_overdue_count_bank, prp_overdue_count_mfo,
                                 prp_active_bank_count, prp_active_mfo_count)
  values (:ma_id, :c_id, :eil_id, :overdue_max, :out_amount_sum,
          :active_kontract_count, :poruch_kont_qty, :overdue_7_30, :overdue_31_60,
          :overdue_61_90, :overdue_91_360, :overdue_360,
          :is_gambling, :is_stop_credit, :is_bankruptcy, :is_tax_debt,
          :is_fake_business, :is_bad_supplier, :is_black_list, :is_wanted_person, :is_wanted_aliment,
          :cnt_overdue_10d, :cnt_overdue_60d, :total_monthly_ins_amount,
          :te_cr_history_count, :overdue_count_bank, :overdue_count_mfo,
          :active_bank_count, :active_mfo_count);
  /*                     вызывается в HP_REFRESH_EXTINFO_DATA (включил робота) а то очень долго обрабатыватся отчет
  if (not exists(select *
                from kib_job_info kji
                where kji.kji_from_eil_id = :eil_id)) then
    execute procedure hp_refresh_pkb_data(:eil_id); */

end
