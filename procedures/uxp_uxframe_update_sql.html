CREATE PROCEDURE UXP$UXFRAME_UPDATE_SQL(
    UFR_UUID D_UUID,
    UFR_SQL_TEXT D_STR_8100,
    UFR_JSON_DATA D_BLOB_TEXT = null,
    UFR_USE_SQL D_BOOL = 0,
    UFR_UUID_CODE D_STR_CODE = null)
AS
declare variable us_uuid d_uuid;
declare variable fb_domain d_str_code;
declare variable fb_type d_str_code;
declare variable origin_uf_uuid d_uuid;
declare variable usf_source d_str_code;
declare variable origin_us_uuid d_uuid;
declare variable is_param d_bool;
declare variable fb_db_domain d_str_code;
declare variable uf_name d_str_code;
declare variable orig_us_uuid d_uuid;
declare variable is_sql d_bool;
declare variable need_new_source d_bool;
declare variable res_kind d_str_code;
declare variable res_source_code d_str_code;
declare variable res_sql_select d_str_8100;
declare variable res_sql_from d_str_long;
declare variable res_sql_where d_str_long;
declare variable res_sql_orderby d_str_long;
declare variable source_params d_str_long;
declare variable param_str d_str_255;
declare variable param_index d_integer;
declare variable uf_uuid d_uuid;
declare variable uf_field_name d_str_code;
declare variable ufp_uuid d_uuid;
begin
  if (coalesce(ufr_uuid, '') = '') then
    ufr_uuid = uxf$uniq(ufr_uuid_code);

  if (ufr_use_sql is null) then
    ufr_use_sql = 0;

  if (coalesce(ufr_json_data, '') <> '') then
    execute procedure uxp$uxsqlfields_fill(:ufr_json_data);

  select res_kind, res_source_code, res_sql_select, res_sql_from, res_sql_where, res_sql_orderby
  from uxp$sql_parcer(:ufr_sql_text)
  into :res_kind, :res_source_code, :res_sql_select, :res_sql_from, :res_sql_where, :res_sql_orderby;

  if (res_kind = 'table') then
  begin
    -- Если источник одна и это таблица, при наличии параметром пропускаем как SQL текст
    if (exists(select * from ux$sqlfields
    where usf_is_field = 0 and             
          usf_source = :res_source_code)) then
      res_kind = 'sql';
  end

  if (res_kind <> 'sql' and ufr_use_sql = 1 and res_sql_select starting 'select *') then
    res_kind = 'sql';

  if (res_kind <> 'sql') then
  begin
    execute procedure uxp$uxsource_create(:res_source_code, '', null, 0, '', 1);

    us_uuid = uxf$uniq(res_source_code);

    update ux$frame
    set ufr_us_uuid = :us_uuid,
        ufr_sql_where = :res_sql_where,
        ufr_sql_orderby = :res_sql_orderby
    where ufr_uuid = :ufr_uuid and
          ufr_us_uuid is distinct from :us_uuid;

    source_params = '';
    if (res_sql_from <> '') then
    begin
      select source_params
      from uxp$sql_source_parcer(:res_sql_from)
      into :source_params;
    end

    if (source_params <> '') then
    begin
      for select uf_uuid
      from ux$field
      where uf_owner = :us_uuid and
            uf_is_param = 1
      into uf_uuid
      do
      begin
        for select ufp_uuid
        from ux$frparam
        where ufp_ufr_uuid = :ufr_uuid and
              ufp_uf_uuid = :uf_uuid
        into ufp_uuid
        do
        begin
          delete from ux$paramdef
          where upd_ufp_uuid = :ufp_uuid and
                upd_uxa_uuid is null;
        end
      end

      param_index = -1;
      for select res_str
      from uxp$split_str(:source_params, ',')
      into :param_str
      do
      begin
        param_index = param_index + 1;
        if (left(param_str, 1) <> ':') then
        begin
          uf_uuid = null;
          select uf_uuid
          from ux$field
          where uf_owner = :us_uuid and
                uf_is_param = 1 and
                uf_position = :param_index
          into uf_uuid;

          for select ufp_uuid
          from ux$frparam
          where ufp_ufr_uuid = :ufr_uuid and
                ufp_uf_uuid = :uf_uuid
          into ufp_uuid
          do
            execute procedure uxp$uxparamdef_update(:ufp_uuid, null, :param_str, 1);
        end
      end
    end
  end else -- if res_kind <> 'sql'
  begin
    select new_sql
    from uxp$sql_lowercase(:res_sql_select)
    into :res_sql_select;

    select res_uuid
    from uxp$uxsource_sql_to_uuid(:res_sql_select)
    into :us_uuid;
  
    select ufr_us_uuid, iif(coalesce(us_sql, '') <> '', 1, 0)
    from ux$frame, ux$source
    where ufr_uuid = :ufr_uuid and
          ufr_us_uuid = us_uuid
    into orig_us_uuid, is_sql;
  
    if (orig_us_uuid = us_uuid) then
    begin
      if (exists(select * from ux$field
                 where uf_owner = :us_uuid)) then
        exit;
    end
  
    need_new_source = 1;
  
    if (not exists(select * from ux$source
    where us_uuid = :us_uuid)) then
    begin
      -- парсинг SQL и создание источника
      if (not exists(select * from ux$sqlfields)) then
        exception e_default 'Sql fields are empty';
  
      if (is_sql = 1) then
      begin
        if (not exists(select * from ux$frame
        where ufr_us_uuid = :orig_us_uuid and
              ufr_uuid <> :ufr_uuid)) then
        begin
          if (exists(select * from ux$code, ux$form, ux$frame
          where ux_us_uuid = :orig_us_uuid and
                ux_uuid = ufm_ux_uuid and
                ufr_owner = ufm_uuid and
                ufr_uuid <> :ufr_uuid)) then
            need_new_source = 0;
        end
      end
  
      if (need_new_source = 1) then
      begin
        insert into ux$source (us_uuid, us_code, us_is_table, us_kind, us_sql)
        values (:us_uuid, uuid_to_char(:us_uuid), 0, -1, :res_sql_select);
      end else
      begin
        if (orig_us_uuid <> us_uuid) then
        begin
          update ux$source
          set us_uuid = :us_uuid
          where us_uuid = :orig_us_uuid;

          update ux$uniq
          set uu_id = :us_uuid
          where uu_id = :orig_us_uuid;
        end
      end
  
      for select distinct usf_source
      from ux$sqlfields
      where coalesce(usf_source, '') <> ''
      into usf_source
      do
      begin
        execute procedure uxp$uxsource_create(:usf_source, '', null, 0, '', 1);
  
        if (res_sql_from <> '') then
        begin
          source_params = '';
          select source_params
          from uxp$sql_source_parcer(:res_sql_from)
          where source_code = :usf_source
          into :source_params;
  
          if (source_params <> '') then
          begin
            param_index = -1;

            for select res_str
            from uxp$split_str(:source_params, ',')
            into :param_str
            do
            begin
              param_index = param_index + 1;

              uf_field_name = '';

              select uf_field_name
              from ux$source, ux$field
              where us_code = :usf_source and
                    uf_owner = us_uuid and
                    uf_is_param = 1 and
                    uf_position = :param_index
              into uf_field_name;

              if (left(param_str, 1) = ':') then
              begin
                update ux$sqlfields
                set usf_origin = :uf_field_name,
                    usf_source = :usf_source
                where usf_name = trim(substring(:param_str from 2)) and
                      usf_is_field = 0 and
                      usf_source = '';
              end else
              begin
                update ux$sqlfields
                set usf_index = usf_index + 1
                where usf_is_field = 0 and
                      usf_index >= :param_index
                order by usf_index desc;

                insert into ux$sqlfields (usf_is_field, usf_index, usf_name, usf_origin, usf_source, usf_param_value)
                values (0, :param_index, :uf_field_name, :uf_field_name, :usf_source, :param_str);
              end
            end
          end
        end
      end
  
      if (need_new_source = 0) then
      begin
        update ux$field
        set uf_state = -1
        where uf_owner = :us_uuid and
              uf_type = 0;
      end
  
      for select usf_is_field, usf_index, usf_name, usf_origin, usf_source, usf_type, usf_size, usf_origin_uf_uuid,
                 usf_db_domain, usf_data_type, usf_param_value
      from ux$sqlfields
      order by usf_is_field, usf_index
      as cursor cur
      do
      begin
        is_param = 1 - cur.usf_is_field;
        uf_name = cur.usf_name;
  
        origin_us_uuid = null;
        origin_uf_uuid = null;
        fb_domain = '';
        fb_db_domain = '';
        fb_type = '';
  
        if (cur.usf_type is not null) then
        begin
          select res_fb_domain, res_fb_type
          from uxp$datatype(cur.usf_type, cur.usf_size, :is_param)
          into :fb_db_domain, :fb_type;
        end
  
        fb_domain = fb_db_domain;
  
        if (coalesce(cur.usf_source, '') <> '') then
        begin
          select us_uuid
          from ux$source
          where us_code = cur.usf_source
          into origin_us_uuid;
  
          if (origin_us_uuid is null) then
            exception e_default f_format('Source not found: %1', usf_source);
  
          select uf_uuid, uf_domain, uf_data_type, iif(uf_field_name <> uf_name, uf_name, :uf_name)
          from ux$field
          where uf_owner = :origin_us_uuid and
                uf_field_name = cur.usf_origin
          into origin_uf_uuid, :fb_domain, :fb_type, uf_name;
        end
  
        update or insert into ux$field (uf_owner, uf_field_name, uf_name, uf_origin_uf_uuid, uf_is_param, uf_position,
                              uf_domain, uf_data_type, uf_default, uf_state)
        values (:us_uuid, cur.usf_name, :uf_name, :origin_uf_uuid, :is_param, cur.usf_index,
                :fb_domain, :fb_type, coalesce(cur.usf_param_value, ''), 0)
        matching (uf_owner, uf_field_name);
      end
  
      delete from ux$field
      where uf_owner = :us_uuid and
            uf_state = -1;
    end
  
    if (need_new_source = 1) then
    begin
      update ux$frame
      set ufr_us_uuid = :us_uuid,
          ufr_sql_where = '',
          ufr_sql_orderby = ''
      where ufr_uuid = :ufr_uuid;
    end
  end
end
