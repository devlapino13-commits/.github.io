CREATE PROCEDURE UXP$UXEVENT_ITEMS_TMP(
    UX_UUID D_UUID,
    LIST_KIND D_STR_CODE)
RETURNS (
    UFI_UUID D_UUID,
    UFI_CONTROL D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_HIDDEN D_BOOL,
    UFI_READONLY D_BOOL,
    IMAGE_INDEX D_INTEGER,
    RES_CAPTION D_STR_255,
    RES_COMMENT D_STR_LONG,
    UF_FIELD_NAME D_STR_CODE,
    UF_SOURCE D_STR_CODE,
    UFI_UUID_STR D_STR_CODE,
    UFI_PARENT_UUID_STR D_STR_CODE,
    UFI_EVENT_CONTROL D_STR_CODE,
    UFI_EVENT_STATUS D_SMALLINT,
    UFI_UFR_UUID D_UUID)
AS
declare variable ufr_uuid d_uuid;
declare variable ufp_uuid d_uuid;
declare variable uf_uuid d_uuid;
declare variable table_root d_uuid;
declare variable ufi_parent_uuid d_uuid;
begin
  if (coalesce(list_kind, '') = '') then
    list_kind = 'sender';

  if (list_kind not in ('sender', 'target')) then
    exception e_message f_format('Undefined list_kind: %1', list_kind);

  for select ufi_uuid, ufi_parent_uuid, ufi_control, ufi_position, ufi_hidden, ufi_readonly,
             image_detail, res_caption, res_comment, uf_field_name, uf_source,
             ufi_uuid_str, ufi_parent_uuid_str, ufi_event_control, ufi_ufr_uuid
      from uxp$uxfritem_list(:ux_uuid, -1, null, 'event')
      into ufi_uuid, ufi_parent_uuid, ufi_control, ufi_position, ufi_hidden, ufi_readonly,
           image_index, res_caption, res_comment, uf_field_name, uf_source,
           ufi_uuid_str, ufi_parent_uuid_str, ufi_event_control, ufi_ufr_uuid
  do
  begin
    select evn_status
    from uxp$uxevent_item_status(:ufi_uuid, :ufi_event_control, :list_kind)
    into :ufi_event_status;

    suspend;
  end

  table_root = f_md5_uuid('table_root');

  ufi_uuid = table_root;
  ufi_parent_uuid = null;
  res_caption = 'Поля и параметры';
  res_comment = 'Поля и параметры в БД';
  uf_source = '';
  ufi_control = 'root';
  ufi_event_control = 'root';
  ufi_position = 0;
  ufi_hidden = 0;
  ufi_readonly = 0;
  image_index = 84; -- Группа таблиц
  uf_field_name = '';
  ufi_uuid_str = uuid_to_char(ufi_uuid);
  ufi_parent_uuid_str = '';
  ufi_event_status = -1;
  ufi_ufr_uuid = null;
  suspend;

  for select fr_uuid, master_fr_uuid, fr_code, res_caption, fr_source
  from uxp$uxframe_tree(:ux_uuid)
  union
  select ufr_uuid, null, ufr_fcode, ufr_name, us_code
  from ux$frame, ux$source
  where ufr_uuid = :ux_uuid and
        ufr_us_uuid = us_uuid
  into :ufr_uuid, :ufi_parent_uuid, :res_comment, :res_caption, :uf_source
  do
  begin
    ufi_ufr_uuid = ufr_uuid;

    if (ufi_parent_uuid is null) then
      ufi_parent_uuid = table_root;

    ufi_uuid = ufr_uuid;
    ufi_control = 'frame';
    ufi_event_control = 'source';
    ufi_position = 0;
    ufi_hidden = 0;
    ufi_readonly = 1;
    image_index = 21; -- Таблица

    if (exists(select * from ux$frame
    where ufr_uuid = :ufr_uuid and
          (ufr_can_update = 1 or ufr_can_insert = 1 or ufr_can_delete = 1))) then
      ufi_readonly = 0;

    uf_field_name = '';

    ufi_uuid_str = uuid_to_char(ufi_uuid);
    ufi_parent_uuid_str = uuid_to_char(ufi_parent_uuid);

    select evn_status
    from uxp$uxevent_item_status(:ufi_uuid, :ufi_event_control, :list_kind)
    into :ufi_event_status;

    suspend;

    ufi_readonly = 0;
    for select ufp_uuid, ufp_parent, ufp_editor, uf_uuid, uf_field_name, uf_name, uf_position,
               iif(uf_is_param = 1, 'param', 'field')
    from ux$frparam, ux$field
    where ufp_ufr_uuid = :ufr_uuid and
          ufp_uf_uuid = uf_uuid
    order by uf_is_param desc, uf_position
    into :ufp_uuid, :ufi_parent_uuid, :ufi_control, uf_uuid, uf_field_name, res_caption, ufi_position,
         ufi_event_control
    do
    begin
      ufi_uuid = ufp_uuid;
      if (ufi_parent_uuid is null) then
        ufi_parent_uuid = ufr_uuid;

      select evn_status
      from uxp$uxevent_item_status(:ufi_uuid, :ufi_event_control, :list_kind)
      into :ufi_event_status;

      image_index = 106; -- параметр

      if (ufi_event_control = 'field') then
        image_index = 7; -- поле

      if (res_caption not containing uf_field_name) then
        res_caption = res_caption || '  (' || uf_field_name || ')';

      res_comment = 'Поле источника';
      ufi_hidden = 0;

      if (not exists(select *
      from ux$fritem
      where ufi_ufr_uuid = :ufr_uuid and   
            ufi_uf_uuid = :uf_uuid and
            ufi_hidden = 0)) then
        ufi_hidden = 1;

      ufi_uuid_str = uuid_to_char(ufi_uuid);
      ufi_parent_uuid_str = uuid_to_char(ufi_parent_uuid);

      suspend;
    end
  end
end
