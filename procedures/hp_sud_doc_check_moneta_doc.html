CREATE PROCEDURE HP_SUD_DOC_CHECK_MONETA_DOC
RETURNS (
    RES_STR D_STR_10)
AS
declare variable doc_kind_id d_integer;
declare variable request_id d_id;
declare variable bls_id d_id;
declare variable sld_id d_id;
declare variable eil_status d_integer;
declare variable doc_kind d_str_code;
begin
  -- для всех запросов на стадии сбора документов
  -- если квитанция о выдаче не прикреплена, то проверить есть ли она в ответах от внешнего сервиса
  -- и если есть, то прикрепить

  res_str = 'ok';
  doc_kind = 'transfer_receipt';
  select scd_id
  from sud_cfgdockind
  where scd_code = :doc_kind
  into :doc_kind_id;

  if (doc_kind_id is null) then
    exception e_default f_format('Unknown doc_kind: %1', coalesce(doc_kind, '<Null>'));

  for select smd_id
  from sud_maindocs
  where smd_status = 0
  into request_id
  do
  begin
    -- найдем пустой документ квитанции
    sld_id = (select sld_id from sud_loandocs
              where sld_from_smd_id = :request_id
                and sld_doc_kind = :doc_kind_id
                and sld_status <> -2
                and not exists(select slb_id from sud_loanbls
                                where slb_from_sld_id = sld_id
                                  and slb_is_delete = 0));

    -- если нет незаполненной квитанции, то выход
    if (sld_id is null) then
      continue;
  
    bls_id = null;
    eil_status = null;

    select eil.eil_from_bls_id, eil.eil_status
    from sud_mainreq, extinfocfg eic, extinfolist eil
    where smr_id = :request_id
      and eic.eic_code = 'moneta_doc'
      and eil.eil_from_eic_id = eic_id
      and eil.eil_object_kind = 9
      and eil.eil_object_id = smr_from_k_id
    order by eil.eil_date
    rows 1
    into bls_id, eil_status;
  
    if (coalesce(eil_status, 0) <> 2 or bls_id is null) then
      continue;

    if (bls_id is not null) then
    begin
      insert into sud_loanbls (slb_from_sld_id, slb_from_bls_id, slb_from_un_id)
      values (:sld_id, :bls_id, 1000000000000);

      update sud_loandocs
      set sld_kind = 0,
          sld_status = 1,
          sld_from_un_id = 1000000000000
      where sld_id = :sld_id;

    end
  end 
  suspend;
end
