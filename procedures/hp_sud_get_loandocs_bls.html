CREATE PROCEDURE HP_SUD_GET_LOANDOCS_BLS(
    REQUEST_ID D_ID NOT NULL,
    SLD_ID D_ID NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    SLB_ID D_ID,
    DOC_KIND D_STR_100,
    DOC_NAME D_STR_100,
    DOC_MEMO D_STR_LONG,
    BLS_ID D_ID,
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    DOC_IS_DELETE D_BOOL,
    DOC_DATETIME D_DATETIME,
    DOC_UN_FIO D_STR_100)
AS
declare variable doc_un_id d_id;
declare variable obj_kind_str d_str_code;
begin
  for select slb_id,
             scd_name,
             slb_from_bls_id,
             slb_object_kind,
             slb_object_id,
             slb_is_delete,
             iif(slb_is_delete = 0, slb_datetime, slb_delete_datetime),
             iif(slb_is_delete = 0, slb_from_un_id, slb_delete_un_id)
  from sud_loanbls, sud_loandocs, sud_cfgdockind
  where slb_from_sld_id = sld_id and
        sld_doc_kind = scd_id and
        sld_from_smd_id = :request_id and
        (slb_is_delete = 0 or :sld_id = 0) and
        (:sld_id = 0 or slb_from_sld_id = :sld_id)
  order by slb_is_delete, slb_from_sld_id, slb_id
  into :slb_id, :doc_kind, :bls_id,
       :obj_kind, :obj_id,
       :doc_is_delete, :doc_datetime, :doc_un_id
  do
  begin
    doc_un_fio = '';
    select un_fio
    from username
    where un_id = :doc_un_id
    into doc_un_fio;

    doc_name = '';
    doc_memo = '';

    obj_kind_str = f_objkind(obj_kind, 'code');
    if (obj_kind_str = 'edoc_blank') then
    begin
      select edf_print_name, 'Эл.документ от ' || f_datetostr(edl_blank_date)
      from edoc_blank, edoc_list, edoc_frx
      where edb_id = :obj_id and
            edb_from_edl_id = edl_id and
            edb_from_edf_id = edf_id
      into doc_name, doc_memo;
    end else
    if (obj_kind_str = 'clientdocs') then
    begin
      select pk_name,
             trim(iif(trim(coalesce(cds_doc_no, '')) <> '', '№ ' || cds_doc_no, '') ||
             iif(cds_doc_date_begin is not null, ' от ' || f_datetostr(cds_doc_date_begin), '') ||
             iif(cds_doc_date_end is not null, ' по ' || f_datetostr(cds_doc_date_end), ''))
      from clientdocs, passkind
      where cds_id = :obj_id and
            cds_from_pk_id = pk_id
      into doc_name, doc_memo;
    end else
      exception e_default f_format('obj_kind not defined: %1', obj_kind_str);

    suspend;
  end
end
