CREATE PROCEDURE HP_PETTRANS_SAVE(
    PET_ID D_INTEGER,
    PET_NAME D_STR_255,
    PET_LABEL D_STR_CODE,
    PET_IS_RENT D_BOOL,
    PET_NORMA_KM D_INTEGER,
    PET_NORMA_YEDIZ D_INTEGER,
    PET_BILL_ID D_ID,
    PET_BILL_LINE_NO D_INTEGER,
    PED_ID D_INTEGER,
    PED_DATE D_DATE,
    PED_REGION_PLACE_ID D_INTEGER,
    PED_FROM_C_ID D_ID,
    PED_ACC_BALANCE D_STR_20,
    PED_ACC_NUMBER D_INTEGER,
    PED_EXPENSE_SUMMER D_CURRENCY,
    PED_EXPENSE_SUMMER_MOUNT D_CURRENCY,
    PED_EXPENSE_WINTER D_CURRENCY,
    PED_EXPENSE_WINTER_MOUNT D_CURRENCY,
    PED_IS_ACTIVE D_INTEGER)
AS
declare variable tmp_ped_id d_integer;
declare variable tmp_ped_date d_date;
begin
  select
    max(ped_id),
    max(ped_date)
  from petdata
  where ped_from_pet_id = :pet_id
  into
    tmp_ped_id,
    tmp_ped_date;
  if (ped_date < tmp_ped_date) then
    exception e_message 'Дата не должна быть меньше чем ' || f_datetostr(tmp_ped_date);

  update or insert into pettrans (
    pet_id,
    pet_name,
    pet_label,
    pet_is_rent,
    pet_norma_km,
    pet_norma_yediz,
    pet_bill_id,
    pet_bill_line_no)
  values (
    :pet_id,
    :pet_name,
    :pet_label,
    :pet_is_rent,
    :pet_norma_km,
    :pet_norma_yediz,
    :pet_bill_id,
    :pet_bill_line_no)
   matching (pet_id)
   returning pet_id, pet_is_rent
  into
    :pet_id,
    :pet_is_rent;

  update petdata set
    ped_is_active = 0
  where ped_from_pet_id = :pet_id
    and ped_id < :ped_id
    and ped_is_active = 1;

  update or insert into petdata (
    ped_from_pet_id,
    ped_id,
    ped_date,
    ped_region_place_id,
    ped_from_c_id,
    ped_acc_balance,
    ped_acc_number,
    ped_expense_summer,
    ped_expense_summer_mount,
    ped_expense_winter,
    ped_expense_winter_mount,
    ped_is_active)
  values (
    :pet_id,
    iif(:ped_date = coalesce(:tmp_ped_date, :ped_date) , :ped_id, :ped_id + 1),
    :ped_date,
    :ped_region_place_id,
    :ped_from_c_id,
    :ped_acc_balance,
    :ped_acc_number,
    :ped_expense_summer,
    :ped_expense_summer_mount,
    :ped_expense_winter,
    :ped_expense_winter_mount,
    :ped_is_active)
   --matching (ped_from_pet_id, ped_id);
   matching (ped_from_pet_id, ped_date);
end
