CREATE PROCEDURE FM_REPORT_CMD_MRU(
    M_KIND D_STR_255,
    M_OBJ_KIND D_STR_CODE,
    M_OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    M_ID VARCHAR(50),
    M_NAME VARCHAR(255))
AS
declare variable n integer;
declare variable def_dd d_date;
declare variable db_id d_integer;
declare variable reg_id d_integer;
declare variable lang d_str_lang;
declare variable json_combo_list d_str_8100;
declare variable item_count d_integer;
declare variable item_no d_integer;
declare variable cmcp_param_name d_str_code;
declare variable edit_kind d_str_code;
begin
  -- Процедура возвращает значения для выпадающего списка для диалога, вызывающего fm_remport_cmd
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);

  if (m_kind = 'disb_date_change') then
  begin
    -- передаем возможные даты переноса даты выдачи
    def_dd = current_date;
    n = 0;
    while (n <= 10) do
    begin
      m_name = f_datetostr(def_dd);
      m_id = m_name;
      suspend;
      n = n + 1;
      def_dd = dateadd(day, 1, def_dd);
    end
  end
  -- Языки общения с клиентом
  if (m_kind = 'set_nation') then
  begin
    for select rd_id, rd_name
    from refdata
    where rd_main_code = 'Nation'
      and rd_parent_id is not null
      and rd_is_active = 1
    order by rd_orderno
    into m_id, m_name
    do
      suspend;
  end

  -- Тип оплаты конс. услуг -- RFCKG-4901
  if (m_kind = 'set_cons_kind') then
  begin
    for select el_id, el_name
    from enum_list
    where el_kind = 'consultation_kind'
    order by el_id
    into m_id, m_name
    do
      suspend;
  end

  if (m_kind = 'change_disb_kind') then
  begin
    m_id = 1;
    m_name = 'Наличными';
    suspend;

    -- После запуска БС СОФР, в РФ выдачи на карту запрещены
    if (f_can_disb_on_card(m_obj_kind, m_obj_id, db_id) = 1) then
    begin
      m_id = -1;
      m_name = 'Банковская карта онлайн';
      suspend;
  
      m_id = 5;
      m_name = 'Банковская карта офлайн';
      suspend;
    end

    if (f_tmp_const('moneta_fps_active') = 1) then
    begin
      m_id = 7;
      m_name = 'СБП по номеру телефона';
      suspend;
    end
  end

  if (m_kind = 'change_deport_reason') then    -- RFCRU-5875
  begin
    for select el.el_id,
               el.el_name
    from enum_list el
    where el.el_kind = 'deport_reason'
    into m_id,
         m_name
    do suspend;
  end

  -- Уровни риска ПФТ/ОД (Кыргызстан)
  if (m_kind = 'set_risk_koef') then
  begin
    m_id = 0;
    m_name = f_lang(lang, 'Низкий');
    suspend;

    m_id = 2;
    m_name = f_lang(lang, 'Высокий');
    suspend;
  end

  --Комментарии коллекторов по клиенту
  if (m_kind = 'oriv_comment') then
  begin
    m_id = 0;
    m_name = 'Должник найден - готов платить';
    suspend;

    m_id = 1;
    m_name = 'Погашает';
    suspend;

    m_id = 2;
    m_name = 'Необходимо подать в суд';
    suspend;

    m_id = 3;
    m_name = 'Потерял трудоспособность (смертельно боленинвалидность)';
    suspend;

    m_id = 4;
    m_name = 'Поручитель готов платить';
    suspend;

    m_id = 5;
    m_name = 'Поручитель погашает';
    suspend;
  end

  -- Причины отказа, в получении займа
  if (m_kind = 'cancel_loan') then
  begin
    for select kr_id, kr_text
    from kontres
    where kr_is_active = 1
    into m_id, m_name
    do
      suspend;
  end

  if (m_kind = 'decline_az_app') then
  begin
    for select kr_id, kr_text
    from kontres
    where kr_is_active = 1
    into m_id, m_name
    do
      suspend;
  end

  if (m_kind = 'extminiapp_refuse') then
  begin
    for select rd.rd_id, rd.rd_name
    from refdata rd
    where rd.rd_main_code = 'website_miniapp.Refuse'
      and rd.rd_parent_id is not null
      and rd.rd_is_active = 1
    into m_id, m_name
    do
      suspend;
  end

  if (m_kind = 'set_clientlink_status') then
  begin
    m_id = 1; m_name = f_lang(lang, 'Заинтересован');
    suspend;

    m_id = -1; m_name = f_lang(lang, 'Перезвонить');
    suspend;

    m_id = -2; m_name = f_lang(lang, 'Отказ');
    suspend;
  end

  if (m_kind = 'transfer_office') then
  begin
    for select office_id,office_name
    from fm_extminiapp_office_list
    into m_id,m_name
    do
      suspend;
  end

  if (m_kind = 'transfer_svz') then
  begin
    for select c_id, c_name_fio
    from hr_pstatus, joblist, jobkind, clients, extminiapp
    where ema_id = :m_obj_id and
         ps_from_reg_id = ema_office_id and
         ps_from_c_id = c_id and
         ps_date_begin <= current_date and
         ps_date_end >= current_date and
         ps_is_working = 1 and
         ps_status = 1 and
         ps_from_jl_id = jl_id and
         jl_kind = jk_id and
         jk_system_code in ('JobSystem.LO_Senior', 'JobSystem.LO_Standart')
    into m_id,m_name
    do
      suspend;
  end

  if (m_kind = 'set_birth_country') then
  begin
    for select rd_name,rd_name
    from fm_get_ref('citizen')
    into m_id, m_name
    do
      suspend;
  end

  if (m_kind = 'set_blank_lang') then
  begin
    -- Языки бланков
    if (db_id in (39801, 41701)) then
    begin
      m_id = 'ru';
      m_name = f_lang(lang, 'Русский');
      suspend;

      m_id = 'kg';
      m_name = f_lang(lang, 'Кыргызский');
      suspend;
    end
    else
      exception e_default 'Not implemented for db_id: ' || db_id;
  end

  if (m_kind = 'com_list_create') then
  begin
    for select cmc_id, cmc_name
        from hp_com_cfg_list(:m_obj_kind, :m_obj_id, :un_id)
        into m_id, m_name
    do
      suspend;
  end

  if (m_kind = 'asp_sign_sms') then
  begin
    -- Возвращаем список подписантов
    select ps_from_reg_id
    from hr_pstatus, username
    where un_id = :un_id and
      ps_from_c_id = un_from_c_id and
      ps_date_begin <= current_date and
      ps_date_end >= current_date and
      ps_status = 1 and
      ps_is_working = 1
    into :reg_id;

    if (reg_id is null) then
    begin
      -- Не нашли регион в приказах, используем ГО
      reg_id = f_hq_id(db_id);
    end
    if (db_id not in (39801, 41701)) then
    begin
      for select distinct c_id, c_name_fio
      from region r, kontsignreg, clients, kontsignmen
      where reg_unicode = :reg_id and
            ksr_from_reg_id in (reg_unicode, r.parent) and
            ksm_id = ksr_from_ksm_id and
            ksm_from_c_id = c_id and
            ksm_is_active = 1 and
            ksr_is_active = 1 and
            coalesce(ksr_date_begin, '01.01.1990') <= current_date and
            coalesce(ksr_date_end, '31.12.9999') >= current_date
      into :m_id, :m_name
      do
        suspend;
    end
    else
    begin
      for select distinct c_id, c_name_fio
      from region r, kontsignreg, clients, kontsignmen
      where reg_unicode = :reg_id and
            ksr_from_reg_id in (reg_unicode, r.parent) and
            ksm_id = ksr_from_ksm_id and
            ksm_from_c_id = c_id and
            ksm_is_active = 1 and
            ksr_is_active = 1 and
            coalesce(ksr_date_begin, '01.01.1990') <= current_date and
            coalesce(ksr_date_end, '31.12.9999') >= current_date
      union
         select c.c_id, c.c_name_fio
         from clients c
         where c.c_id = 1090100181509   -- Добавили Мамрасуловой Чинары
      into :m_id, :m_name
      do
        suspend;
    end
  end

  if (m_kind starting 'cml_set_rest_') then
  begin
    -- Выпадающие списки параметров запроса в кредитный комитет
    cmcp_param_name = replace(m_kind, 'cml_set_rest_', '');
    select cmcp_combo_list, cmcp_edit_kind
    from com_cfg_params, com_list
    where cml_id = :m_obj_id and
          cml_from_cmc_id = cmcp_from_cmc_id and
          cmcp_param_name = :cmcp_param_name
    into json_combo_list, edit_kind;

    if (edit_kind = 'refdata') then
    begin
      for select id, name
      from hp_com_list_refdata_combo(:json_combo_list)
      into m_id, m_name
      do
        suspend;
    end
    else
    if (edit_kind = 'combo' and json_combo_list <> '') then
    begin
      item_count = f_json_str_get_count(json_combo_list, 'list');
      item_no = 0;
      while (item_no < item_count) do
      begin
        m_id = f_json_str_get_value(json_combo_list, 'list[' || item_no || '].id', '');
        m_name = f_json_str_get_value(json_combo_list, 'list[' || item_no || '].name', '');
        suspend;

        item_no = item_no + 1;
      end
    end
  end

  if (m_kind = 'set_fps_bank') then
  begin
    for with curr_bank as (
    select fpb_id, fpb_name || '*' as fpb_name, -999 as fpb_order_no
    from fps_banks, fps_client_banks, fps_loan_banks
    where flb_from_kc_id = :m_obj_id
      and fbl_from_fcb_id  = fcb_id
      and fcb_from_fpb_id = fpb_id
      and fpb_status = 1
    ),
    other_banks as (
    select fpb_id, fpb_name, fpb_order_no from fps_banks
    where fpb_status = 1
      and fpb_id not in (select fpb_id from curr_bank)
      and fpb_name_full is not null -- Если нет полного имени, будет проблема с договором
    ),
    all_banks as (
    select fpb_id, fpb_name, fpb_order_no from curr_bank
    union all
    select fpb_id, fpb_name, fpb_order_no from other_banks
    )
    select fpb_id, fpb_name from all_banks
    order by fpb_order_no
    into m_id, m_name do
      suspend;
  end

  if (db_id = 81001 and m_kind starting 'extminiapp_refuse_') then
  begin
    for select rd.rd_id, rd.rd_name
    from refdata rd
    where rd_main_code = 'website_miniapp.refuse'
      and rd_str_id = :m_kind
      and rd_parent_id is not null
      and rd_is_active = 1
    order by rd_orderno
    into m_id, m_name
    do
      suspend;
  end

  if (m_kind = 'set_bki_survey_res') then
  begin
    m_id = 1; m_name = f_lang(lang, 'Положительный');
    suspend;
    m_id = -1; m_name = f_lang(lang, 'Отрицательный');
    suspend;
  end
  -- Доход подтвержден документально
  if (m_kind = 'set_maci_is_documented') then
  begin
    m_id = 0; m_name = 'нет';
    suspend;
    m_id = 1; m_name = 'да';
    suspend;
  end
end
