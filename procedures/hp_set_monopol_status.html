CREATE PROCEDURE HP_SET_MONOPOL_STATUS(
    IS_MONOPOL INTEGER)
AS
begin
  if (is_monopol not in (0, 1)) then
    exception e_default 'Неверное значение режима монопольного режима(is_monopol)';

  if (f_get_db_id() in (39801, 41701)) then
    post_event iif(is_monopol = 0, 'continue_operations', 'suspend_operations');

  update company
  set cm_is_monopol = :is_monopol;

  if (is_monopol = 1) then
  begin
    delete from mon$attachments
    where mon$attachment_id <> current_connection and mon$system_flag = 0;
  end
end
