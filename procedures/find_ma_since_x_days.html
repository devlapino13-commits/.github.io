CREATE PROCEDURE FIND_MA_SINCE_X_DAYS(
    LO_ID ID_TYPE,
    LO_REGION_ID INTEGER,
    CLIENT_ID ID_TYPE,
    REP_DAYS INTEGER)
RETURNS (
    LOANMINIAPPID INTEGER)
AS
declare variable db_id d_id;
begin
  db_id = f_get_db_id();
  -- -- Ищем действующ. заявку за послед. 15 дней
  for select ma_id
      from knaktlist, miniappclient ma, miniapp, product pr
      where knaktlist.is_delete = 0 and
            kal_new_kf_officer = :lo_id and
            kal_new_reg_id = :lo_region_id and
            kal_date_end >= '31.12.2999' and
            ma_from_kf_id = kal_from_kf_id and
            pr.pr_id = ma_from_pr_id and
            (pr.pr_code not containing '.fertilizer.' or :db_id not in (39801, 41701))
            -- and ma.mac_amount>0 -- Клиент может быть поруч.
            and
            ma_is_active >= 1 and
            ma.mac_from_ma_id = ma_id and
            ma_date_end >= '31.12.2999' and
            ma.mac_from_c_id = :client_id and
            ma_date_app >= (current_date - :rep_days)
      order by ma_date_app desc, ma_id desc
      rows 1
      into :loanminiappid
  do
  begin
    suspend;
  end
end
