CREATE PROCEDURE HP_EDOC_GET_BLANK_INFO(
    EDF_ID D_INTEGER,
    PR_ID D_INTEGER,
    BLANK_DATE D_DATE,
    LANG_CODE D_STR_CODE,
    BLOB_DATE D_DATE = null)
RETURNS (
    BLN_ID D_INTEGER,
    BLS_ID D_ID,
    ERR_TEXT D_STR_255,
    FOUND_COUNT D_INTEGER,
    FOUND_BLANKS D_STR_LONG)
AS
declare variable pr_code d_str_100;
declare variable edc_code d_str_code;
declare variable edf_add_code d_str_code;
declare variable edf_prepare_kind d_integer;
declare variable def_lang_code d_str_lang;
declare variable edf_from_blt_id d_integer;
declare variable edf_from_bln_id d_integer;
begin
  bln_id = null;
  bls_id = null;
  err_text = '';
  found_blanks = '';
  edf_prepare_kind = null;
  def_lang_code = f_get_const('blank_default_lang');

  if (blob_date is null) then
    blob_date = blank_date;

  for select edf_id, edf_from_bls_id, edf_add_code, edf_prepare_kind, edf_from_blt_id, edc_code, edf_from_bln_id
  from edoc_frx, edoc_cfg
  where edf_id = :edf_id and
        :blank_date between edf_date_begin and coalesce(edf_date_end, :blank_date) and
        edf_prepare_kind <> 0 and
        edf_from_edc_id = edc_id
  into edf_id, bls_id, edf_add_code, edf_prepare_kind, edf_from_blt_id, edc_code, edf_from_bln_id
  do
  begin
    bln_id = null;
    if (edf_prepare_kind <> 2) then -- если НЕ готовится через BLS_ID, убираем BLS_ID
    begin
      bls_id = null;

      if (edf_prepare_kind = 3) then
      begin
        -- тут пишем код для определения бланка согласно значения edf_add_code
        if (edc_code = 'loan.agreement' and edf_add_code = 'loan.agreement') then
        begin
          select pr_code
          from product
          where pr_id = :pr_id
          into pr_code;
  
          bln_id = 1453;
          if (position('.zalog_car.', pr_code) > 0) then bln_id = 1478;
          if (position('.education.', pr_code) > 0) then bln_id = 1503;
        end
      end

      if (edf_prepare_kind = 1) then
      begin
        found_count = 0;

        if (coalesce(edf_from_blt_id, 0) <> 0) then
        begin
          for select bln_id
          from blankname
          where bln_from_blt_id = :edf_from_blt_id and
                bln_lang = :lang_code
          into bln_id
          do
          begin
            if (exists(
            select * from blankblob
            where blb_from_bln_id = :bln_id and
                  :blob_date between blb_date_begin and blb_date_end and
                  coalesce(blb_frx_blob, blb_blob) is not null))
            then
            begin
              found_count = found_count + 1;
              if (found_blanks = '') then
                found_blanks = found_blanks || ', ';
              found_blanks = found_blanks || bln_id;
            end
          end
  
          -- если язык бланка не дефолтный и не найден бланк, ищем на дефолтном
          if (found_count = 0 and def_lang_code <> lang_code) then
          begin
            for select bln_id
            from blankname
            where bln_from_blt_id = :edf_from_blt_id and
                  bln_lang = :def_lang_code
            into bln_id
            do
            begin
              if (exists(
              select * from blankblob
              where blb_from_bln_id = :bln_id and
                    :blob_date between blb_date_begin and blb_date_end and
                    coalesce(blb_frx_blob, blb_blob) is not null)) then
              begin
                found_count = found_count + 1;
                if (found_blanks = '') then
                  found_blanks = found_blanks || ', ';
                found_blanks = found_blanks || bln_id;
              end
            end
          end
        end else
        begin
          if (edf_from_bln_id is not null) then
          begin
            bln_id = edf_from_bln_id;
            found_count = 1;
            found_blanks = bln_id;
          end
        end

        if (def_lang_code = lang_code) then
        begin
          if (err_text = '' and bln_id is null) then
            err_text = 'Blank not found.';

          if (err_text = '' and bln_id is not null and found_count = 0) then
            err_text = 'Blank blob not found.';
        end

        if (err_text = '' and found_count > 1) then
          err_text = 'Multiple blank found.';
      end

      if (edf_prepare_kind in (4, 5)) then
      begin
        if (err_text = '' and pr_id is null) then
          err_text = 'Product ID not found';

        found_count = 0;
        for select prb_from_bln_id
        from productblank, blankname
        where prb_from_pr_id = :pr_id and
              prb_from_bln_id = bln_id and
              bln_from_blt_id = :edf_from_blt_id and
              bln_lang = :lang_code and
              :blank_date between prb_date_begin and coalesce(prb_date_end, :blank_date)
        into bln_id
        do
        begin
          if (exists(
          select * from blankblob
          where blb_from_bln_id = :bln_id and
                :blob_date between blb_date_begin and blb_date_end and
                coalesce(blb_frx_blob, blb_blob) is not null)) then
          begin
            found_count = found_count + 1;
            if (found_blanks = '') then
              found_blanks = found_blanks || ', ';
            found_blanks = found_blanks || bln_id;
          end
        end

        -- если язык бланка не дефолтный и не найден бланк, ищем на дефолтном
        if (found_count = 0 and def_lang_code <> lang_code) then
        begin
          for select prb_from_bln_id
          from productblank, blankname
          where prb_from_pr_id = :pr_id and
                prb_from_bln_id = bln_id and
                bln_from_blt_id = :edf_from_blt_id and
                bln_lang = :def_lang_code and
                :blank_date between prb_date_begin and coalesce(prb_date_end, :blank_date)
          into bln_id
          do
          begin
            if (exists(
            select * from blankblob
            where blb_from_bln_id = :bln_id and
                  :blob_date between blb_date_begin and blb_date_end and
                  coalesce(blb_frx_blob, blb_blob) is not null)) then
            begin
              found_count = found_count + 1;
              if (found_blanks = '') then
                found_blanks = found_blanks || ', ';
              found_blanks = found_blanks || bln_id;
            end
          end
        end

        if (edf_prepare_kind = 4) then
        begin
          if (err_text = '' and bln_id is null) then
            err_text = 'Blank not found.';

          if (err_text = '' and bln_id is not null and found_count = 0) then
            err_text = 'Blank blob not found.';
        end

        if (err_text = '' and found_count > 1) then
          err_text = 'Multiple blank found.';
      end
    end
  end
  if (err_text = '' and edf_prepare_kind is null) then
    err_text = 'eDoc_Frx not found.';

  if (err_text = '' and
      not(edf_prepare_kind in (2, 5)) and
      bln_id is null) then
    err_text = 'Blank not found.';

  if (err_text = '' and bln_id is not null) then
  begin
    if (not exists(
    select * from blankblob
    where blb_from_bln_id = :bln_id and
          :blob_date between blb_date_begin and blb_date_end and
          coalesce(blb_frx_blob, blb_blob) is not null))
    then err_text = 'Blank frx blob found.';
  end

  if (err_text <> '') then
    err_text = err_text || ' ' || f_format('pr_id: %1, edf_id: %2', pr_id, edf_id);

  suspend;
end
