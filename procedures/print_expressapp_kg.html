CREATE PROCEDURE PRINT_EXPRESSAPP_KG(
    EXA_ID D_INTEGER)
RETURNS (
    LGOT_PERC1 D_PERCENT,
    LGOT_PERC2 D_PERCENT,
    BASE_PERC D_PERCENT,
    COMMISSION D_CURRENCY,
    FULL_COST D_CURRENCY,
    OFFICE_ADDR D_STR_255,
    CL_FIO D_STR_100,
    EXA_AMOUNT D_CURRENCY,
    EXA_DATE_END D_DATE)
AS
declare variable reg_id d_integer;
declare variable exa_date d_date;
begin
  lgot_perc2 = 0;
  base_perc = 0;
  commission = 0;

  select round(exa.exa_percent, 0), exa.exa_amount, exa.exa_from_reg_id, cl.c_name_fio, exa_date
  from expressapp exa
  inner join clients cl on cl.c_id = exa.exa_from_c_id
  where exa.exa_id = :exa_id
  into lgot_perc1, exa_amount, reg_id, cl_fio, exa_date;

  exa_date_end = exa_date + 3;

  lgot_perc2 = round((lgot_perc1 + 5.0), 0); --Льготная процентнтая ставка вторая
  base_perc = round((lgot_perc2 + 5.0),0); -- Базовая процентная ставка
  commission = round((exa_amount * 1.5 / 100),0); -- комиссия за кассовое обслуживание

  select sum(sh_os + sh_perc)
  from print_exa_schedule(:exa_id)
  into full_cost;

  -- адрес офиса
  select adl.adl_address_str
  from region rg
  inner join addrlist adl on adl.adl_id = rg.reg_address_fact
  where reg_unicode = :reg_id
  into office_addr;
  suspend;
end
