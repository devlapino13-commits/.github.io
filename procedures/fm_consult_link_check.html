CREATE PROCEDURE FM_CONSULT_LINK_CHECK(
    K_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable fio d_str_100;
declare variable kc_id d_id;
begin
  res_text = '';
  for select kc.kc_from_c_id, c.c_name_fio, kc.kc_id
      from clients c
      inner join knclient kc on kc.kc_from_c_id = c.c_id
      inner join kontract k on k.k_id = kc.kc_from_k_id
      where k.k_id = :k_id and
            kc.kc_amount_approved > 0
      into :c_id, :fio, :kc_id
  do
  begin
    if (not exists(select cvs.cks_visit_qty
                   from clt_kc_visit_svod cvs
                   where cvs.cks_from_kc_id = :kc_id)) then
      res_text = res_text || :fio || ', ';

   end
     if (res_text <> '') then
  begin
    res_text = left(res_text, char_length(res_text) - 2);
    res_text = 'Клиент(ы) не заходил(и) на портал для получения информ. услуги: ' || res_text;
  end

      suspend;
    end
