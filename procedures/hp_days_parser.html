CREATE PROCEDURE HP_DAYS_PARSER(
    DATE_LIST D_STR_100)
RETURNS (
    DAY_OUT D_INTEGER)
AS
begin
  day_out = 0;
  if (date_list = '') then
  begin
    suspend;
    exit;
  end
  while (date_list <> '') do
  begin
    day_out = cast(left(:date_list, position('.' in :date_list) - 1) as integer);
    date_list = overlay(:date_list placing '' from 1 for position('.' in :date_list));
    suspend;
  end
end
