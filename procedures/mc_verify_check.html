CREATE PROCEDURE MC_VERIFY_CHECK(
    MCL_ID D_ID)
RETURNS (
    C_ID D_ID,
    FIO D_STR_100,
    IS_VERIFY D_SMALLINT,
    USER_PATH D_STR_100,
    PANEL_TITLE D_STR_100,
    PANEL_TEXT D_STR_255,
    CAN_BIOMETRY_SIGN D_BOOL,
    INN D_STR_20,
    HAS_GRS_PHOTO D_BOOL,
    C_PASS_NO D_STR_100,
    CARD_ID D_ID,
    CARD_PAN D_STR_10)
AS
declare variable PHONE_NUMBER D_STR_20;
declare variable C_ID_BY_LOGIN D_ID;
declare variable ESP_EXIST D_INTEGER;
declare variable BLS_ID D_ID;
declare variable CLT_ID D_ID;
declare variable LANG_CODE D_STR_LANG;
declare variable EXTINFOLIST_STATUS D_INTEGER;
declare variable STATUS_FACERECOG D_INTEGER;
declare variable GRS_BLS_ID D_ID;
declare variable DB_ID D_INTEGER;
begin
  db_id = f_get_db_id();

  user_path = 'camera';
  panel_title = '';
  panel_text = '';
  is_verify = 0;
  can_biometry_sign = 0;
  has_grs_photo = 0;

  if (db_id = 81001) then
    can_biometry_sign = 1;

  select mf.mcf_status
  from mc_facerecog mf
  where mf.mcf_from_mcl_id = :mcl_id
  order by mf.mcf_create_datetime desc
  rows 1
  into status_facerecog;

  select e.eil_status
  from extinfolist e
  where e.eil_from_mcl_id = :mcl_id
  order by e.eil_create_datetime desc
  rows 1
  into :extinfolist_status;

  if (extinfolist_status in (0, 1)) then
  begin
    user_path = 'guest';
    panel_title = f_lang(lang_code, 'Авторизация в процессе');
    panel_text = f_lang(lang_code, 'Проверка данных займет некоторое время');
    suspend;
    exit;
  end

  select mcl.mcl_login, mcl.mcl_lang_code, mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :phone_number, :lang_code, :c_id_by_login;

  select cl.clt_id
  from contlist cl
  where cl.clt_code = :phone_number
  into clt_id;

  select cd.cda_from_c_id, cda_status
  from contdata cd
  where cd.cda_from_clt_id = :clt_id and
        cd.cda_is_delete = 0 and
        cd.cda_is_main = 1
  order by cd.cda_date desc
  rows 1
  into c_id, is_verify;

  -- 29.04.24 добавил igusev т.к. расхождение между c_id_by_login и c_id мешало тестирование.
  -- по сути, если c_id_by_login получен он должен являться основным
  if (c_id_by_login is not null) then
    c_id = c_id_by_login;

  if (c_id is not null or :c_id_by_login is not null) then
  begin
    grs_bls_id = (select bls_id
                  from get_grs_photo(iif(:c_id_by_login is null, :c_id, :c_id_by_login), 0, null, 0));
    if (coalesce(grs_bls_id, 0) > 0) then
      has_grs_photo = 1;
  end

  if (c_id is null and
      c_id_by_login is null) then
    esp_exist = 0;
  else
    esp_exist = f_asp_valid(iif(:c_id_by_login is null, :c_id, :c_id_by_login), current_date);

  if (f_asp_signed(coalesce(:c_id, :c_id_by_login), current_date) = 1) then
    esp_exist = 1;

  select cl.c_name_fio, cl.c_inn, cl.c_pass_no
  from clients cl
  where cl.c_id = coalesce(:c_id, :c_id_by_login)
  into :fio, :inn, :c_pass_no;

  select cal_card_id, card_pan_hiden_short
  from get_active_mbulak_card(coalesce(:c_id, :c_id_by_login))
  into card_id, card_pan;

  if (c_id_by_login is not null) then
  begin
      c_id = c_id_by_login;
      can_biometry_sign = esp_exist;
      user_path = 'main';
      suspend;
      exit;
  end

  if (c_id is not null or c_id_by_login is not null) then
  begin
    if (exists(select cda_status
               from contdata cd
               where (cd.cda_from_c_id = :c_id or
                     cd.cda_from_c_id = :c_id_by_login) and
                     cd.cda_is_delete = 0 and
                     cd.cda_is_main = 1)) then
      is_verify = 1;

    if (status_facerecog = 0 or extinfolist_status in (0, 1)) then
    begin
      user_path = 'camera';
      panel_title = f_lang(lang_code, 'Авторизация в процессе');
      panel_text = f_lang(lang_code, 'Проверка данных займет некоторое время');
      suspend;
      exit;
    end
    else
    begin
      select eib.eib_from_bls_id
      from extinfolistbls eib
      inner join extinfolist eil on eil.eil_id = eib.eib_from_eil_id and
            ((eil.eil_from_c_id = :c_id or eil.eil_from_c_id = :c_id_by_login) or eil.eil_from_mcl_id = :mcl_id) and
            eil.eil_from_eic_id = 2 and
            eil.eil_status = 2
      where eib.eib_kind = f_enum_id('extinfo_bls_kind', 'tunduk_client_photo') and
            eib.eib_from_bls_id <> 0
      order by eil.eil_create_datetime desc
      rows 1
      into bls_id;

      if (status_facerecog is null and
          :bls_id is not null) then
      begin
        user_path = 'camera';
        suspend;
        exit;
      end

      can_biometry_sign = 0; --биометрическая подпись запрещена

      if (bls_id is not null and
          esp_exist >= 1) then
        can_biometry_sign = 1; --биометрическая подпись разрешена

      bls_id = coalesce(:bls_id, 0);

      if (:bls_id <= 0) then
      begin
        is_verify = 0;
        suspend;
        exit;
      end
      if (esp_exist = 0) then
      begin
        user_path = 'guest';
        panel_title = f_lang(lang_code, 'Для получения займа вам вам необходимо подойти в ближайший офис');
        panel_text = f_lang(lang_code, 'Адреса и режим работы можно посмотреть по кнопке "На карте"');
        suspend;
        exit;
      end
      else
      begin
        if (status_facerecog = 1 and
            extinfolist_status = 2) then
        begin
          user_path = 'main';
          panel_title = '';
          panel_text = '';
        end
      end
    end
  end
  else
  begin
    suspend;
    exit;
  end
  suspend;
end
