CREATE PROCEDURE EDOC_LOAN_APP_BLACKLIST_CHECK(
    C_ID D_ID)
RETURNS (
    RES_ERROR_TEXT D_STR_255)
AS
declare variable par_c_fio d_str_100;
declare variable par_c_birth_date d_date;
declare variable par_c_pass_no d_str_20;
declare variable par_c_inn d_str_20;
declare variable par_c_cell_no d_str_20;
declare variable m_date_begin d_date;
declare variable m_date_end d_date;
begin
    if (c_id in (1010101181000, 1010100491679, 1010100758675, 1010100206410, 1010101299224)) then exit;
    -- RFCRU-3451, RFCRU-5104
   res_error_text = '';

   select c_name_fio,
          c_birth_date,
          coalesce(c_pass_no, ''),
          coalesce(c_inn, ''),
          f_cont_info('clients', :c_id, 'cell_main', 'ok')
   from clients cl
   where cl.c_id = :c_id
   into par_c_fio,
        par_c_birth_date,
        par_c_pass_no,
        par_c_inn,
        par_c_cell_no;

   select coalesce(left(c_comment || ', ' || bl_comment, 255), '')
   from hp_blacklist_search(:c_id, :par_c_fio, :par_c_birth_date, :par_c_pass_no, :par_c_inn, :par_c_cell_no, 1, 0, 0)
   where c_comment_id = 1 rows 1
   into :res_error_text;

   if (res_error_text <> '') then
   begin
        res_error_text = 'Клиент в стоп-листе, сначала выведите его из стоп-листа';
   end
   else
   begin
      select coalesce(left(c_comment || ', ' || bl_comment, 255), ''),
             coalesce(m_date_begin, '01.01.1900'),
             coalesce(m_date_end, '31.12.2999')
      from hp_blacklist_search(:c_id, :par_c_fio, :par_c_birth_date, :par_c_pass_no, :par_c_inn, :par_c_cell_no, 0, 1, 0)
      where c_comment_id = 1 rows 1
      into res_error_text,
           m_date_begin,
           m_date_end;

      if ((res_error_text <> '') and (current_date between m_date_begin and m_date_end)) then
        res_error_text = 'Клиент в отказниках, сначала выведите его из отказников';
   end

   suspend;
end
