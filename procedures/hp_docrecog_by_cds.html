CREATE PROCEDURE HP_DOCRECOG_BY_CDS(
    CDS_ID INTEGER)
RETURNS (
    DRL_ID BIGINT)
AS
declare variable drv_value d_str_255;
begin
  -- Создает документ в DocRecogList, по переданному cds_id (ClientDocs)
  -- Не создаем если: есть созданный DocRecogList, pk_from_drm_id is null
  for select
    cds_from_c_id, cds_from_pk_id, cds_doc_no, cds_doc_date_begin,
    cds_doc_date_end, cds_doc_issued_by, cds_doc_for_company,
    cds_doc_territory, cds_doc_blob_1, cds_doc_blob_2, cds_datetime,
    cds_is_delete, cds_memo_1, cds_memo_2, cds_memo_3, cds_memo_4, cds_memo_5,
    cds_is_main_doc, cds_create_datetime, cds_create_un_id,
    pk_from_drm_id,
    c_name_eng, c_alternat_fio, c_name_fio, c_birth_date, c_inn,
    c_birth_city, c_birth_country, c_sex
  from clientdocs, passkind, clients
  where
    cds_id = :cds_id and
    cds_from_c_id = c_id and
    cds_from_pk_id = pk_id and
    cds_from_drl_id is null and
    pk_from_drm_id is not null
  as cursor cur_cds
  do
  begin
    insert into docrecoglist(
      drl_from_drm_id, drl_from_pk_id, drl_from_c_id, drl_from_bls_1,
      drl_from_bls_2, drl_from_un_id, drl_from_mcl_id, drl_create_datetime,
      drl_saved_datetime, drl_blob1_datetime, drl_blob2_datetime,
      drl_status, drl_is_main_doc, drl_client_created)
    values (
      cur_cds.pk_from_drm_id, cur_cds.cds_from_pk_id, cur_cds.cds_from_c_id,
      cur_cds.cds_doc_blob_1, cur_cds.cds_doc_blob_2,
      cur_cds.cds_create_un_id, 0, cur_cds.cds_create_datetime,
      cur_cds.cds_datetime, cur_cds.cds_datetime, cur_cds.cds_datetime,
      iif(cur_cds.cds_doc_blob_1 is not null or cur_cds.cds_doc_blob_2 is not null, 1, 0),
      cur_cds.cds_is_main_doc, 0)
    returning drl_id
    into drl_id;

    -- Создаем поля
    for select
      drf_id, drf_code
    from docrecogfield
    where drf_from_drm_id = cur_cds.pk_from_drm_id and
      drf_inactive = 0
    as cursor cur_drf
    do
    begin
      drv_value = null;

      if (cur_drf.drf_code = 'doc_no') then
        drv_value = cur_cds.cds_doc_no;
      if (cur_drf.drf_code = 'date_begin') then
        drv_value = f_datetostr(cur_cds.cds_doc_date_begin);
      if (cur_drf.drf_code = 'date_end') then
        drv_value = f_datetostr(cur_cds.cds_doc_date_end);
      if (cur_drf.drf_code = 'authority') then
        drv_value = cur_cds.cds_doc_issued_by;
      if (cur_drf.drf_code = 'for_company') then
        drv_value = cur_cds.cds_doc_for_company;
      if (cur_drf.drf_code = 'territory') then
        drv_value = cur_cds.cds_doc_territory;
      if (cur_drf.drf_code = 'memo_1') then
        drv_value = cur_cds.cds_memo_1;
      if (cur_drf.drf_code = 'memo_2') then
        drv_value = cur_cds.cds_memo_1;
      if (cur_drf.drf_code = 'memo_3') then
        drv_value = cur_cds.cds_memo_1;
      if (cur_drf.drf_code = 'memo_4') then
        drv_value = cur_cds.cds_memo_1;
      if (cur_drf.drf_code = 'memo_5') then
        drv_value = cur_cds.cds_memo_1;

      -- ToDo, даты нужно форматировать
      if (cur_cds.cds_is_main_doc = 1 and cur_cds.cds_is_delete = 0) then
      begin
        -- Поля, которые берем с клиента только для текущего паспорта
        if (cur_drf.drf_code = 'calc_name_full') then
          drv_value = cur_cds.c_name_fio;
        if (cur_drf.drf_code = 'calc_name_full_eng') then
          drv_value = cur_cds.c_name_eng;
        if (cur_drf.drf_code = 'calc_name_full_alt') then
          drv_value = cur_cds.c_alternat_fio;
        if (cur_drf.drf_code = 'birth_date') then
          drv_value = f_datetostr(cur_cds.c_birth_date);
        if (cur_drf.drf_code = 'birth_place') then
          drv_value = cur_cds.c_birth_city;
        if (cur_drf.drf_code = 'birth_country') then
          drv_value = cur_cds.c_birth_country;
        if (cur_drf.drf_code = 'calc_gender') then
          drv_value = cur_cds.c_sex;
        if (cur_drf.drf_code = 'inn') then
          drv_value = cur_cds.c_inn;
      end

      if (drv_value is not null) then
      begin
        -- null не вставляем, т.к. функция f_docrecog_value обработает это при чтении
        insert into docrecogvalue (
          drv_from_drl_id, drv_from_drf_id, drv_recog_value,
          drv_final_value, drv_is_accepted, drv_is_accepted_calc)
        values (:drl_id, cur_drf.drf_id, '', :drv_value, 1, 1);
      end
    end
  end
  suspend;
end
