  -- ѕровер¤ем, есть ли за¤вка на досрочное погашение
  aa_date_repay = null;
  select aa_date_repay
  from apadvanced
  where aa_is_deleted = 0 and
        aa_from_k_id = :k_id and
        (:c_id = 0 or
        aa_from_c_id = :c_id) and
        aa_date_app <= :rep_date and
        (aa_date_repay + (iif(extract(weekday from aa_date_repay) = 0, 1, iif(extract(weekday from aa_date_repay) = 6, 2, 0)))) >= (:rep_date - 30)
  order by aa_date_app
  rows 1
  into :aa_date_repay;

  if (aa_date_repay is not null and
      aa_date_repay <= :rep_date) then
    need_kommis = 0;

-- если за¤влени¤ нет, берем проценты за 30 дней
  if (aa_date_repay is null) then
    aa_date_repay = :rep_date + 30;

-- если дата в за¤вке и фактическое плюс-минус 1 день
  if (abs(datediff(day, :rep_date, :aa_date_repay)) <= 1) then
    need_kommis = 0;

-- если дата погашени¤ в за¤вке попадает на выходные
  if (extract(weekday from aa_date_repay) in (0, 6)) then
  begin
-- если дата фактического погашени¤ в за¤вке попадает на выходные или в пт или в пн
    if (extract(weekday from rep_date) in (0, 1, 5, 6)) then
    begin
-- если между ними не более 2 дней, то не считаем комиссию.
      if (abs(datediff(day, :rep_date, :aa_date_repay)) <= 2) then
        need_kommis = 0;
    end
  end

  if (need_kommis = 1) then
  begin
-- ≈сли до конца срока кредита осталось меньше 30 дней, могут погашать без комиссии
    select ok_sh_date
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_from_c_id = 0
    group by ok_sh_date
    having sum(ok_sh_os + ok_os) > 0
    order by ok_sh_date desc
    rows 1
    into :last_date;

    if (last_date is null) then
    begin
      select sh_date
      from schedule
      where sh_from_k_id = :k_id
      order by sh_from_k_id desc, sh_date desc
      rows 1
      into :last_date;
    end
--    if ((rep_date>=last_date) and (datediff(day, :rep_date, :last_date)<=30))
--    then need_kommis=0;
    if ((rep_date <= last_date) and
        (datediff(day, :rep_date, :last_date) <= 30)) then
      aa_date_repay = last_date;
    if (rep_date >= last_date) then
      need_kommis = 0;
  end

  if (need_kommis = 1) then
  begin
-- ѕровер¤ем, есть ли за¤вка на следующий цикл у группы или участника группы
--ARAKHMANOV проверяем, утверждена ли заявка на следующий цикл у группы или участника группы 03.11.2022
    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              (:c_id = 0 or
              kc_from_c_id = :c_id)
        into :kc_id
    do
   begin
      if (exists(select *
                 from knclient kc
                 inner join kontract k on kc.kc_from_k_id = k.k_id
                 inner join miniapp ma on k.k_from_ma_app_id = ma.ma_id
                 where kc.kc_from_c_id = :kc_id and
                       k.k_status = 1 and
                       kc.kc_amount_approved > 0 and
                       ma_date_end >= '31.12.2999' and
                       ma_date_oper >= :rep_date and
                       ma_date_app <= :rep_date )) then
        need_kommis = 0;
    end
  end

  if (need_kommis = 1) then
  begin
-- если вышеуказанные услови¤ не соблюдались
-- вычисл¤ем сумму комиссии
    base_amount = 0;
    select oks_balance, kd_percent_per_day
    from oksummary, kontdata
    where oks_from_k_id = :k_id and                                                         
          oks_date_begin <= :rep_date and
          oks_date_end >= :rep_date and
          kd_from_k_id = :k_id
--    and kd_date_end>='31.12.2999'
          and
          kd_ol_is_delete = 0 and
          :rep_date between kd_ol_date and kd_date_end and
          oks_close_state = 0
    into :base_amount, k_percent_per_day;

    if (c_id > 0) then
    begin
      select k_amount_approved, kc_amount_approved
      from knclient, kontract
      where kc_from_c_id = :c_id and
            kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            k_id = :k_id
      into :gr_amount, :tmp;
      base_amount = round(cast(base_amount as double precision) * (tmp / cast(gr_amount as double precision)), 0);
    end
    kol_day = datediff(day, rep_date, aa_date_repay);
    if (kol_day < 0) then
      kol_day = 0;
    res_kommis_amount = round((cast(base_amount as double precision) / 100.0 * kol_day) * k_percent_per_day, 0);
  end
  suspend;
