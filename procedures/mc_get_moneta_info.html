CREATE PROCEDURE MC_GET_MONETA_INFO(
    C_ID D_ID,
    K_ID D_ID)
RETURNS (
    MNT_DESCRIPTION D_STR_255,
    MNT_ID D_ID)
AS
declare variable CUR_NAME D_STR_20;
declare variable K_KONTRACT_NO D_STR_20;
declare variable DB_ID D_INTEGER;
declare variable C_NAME_FIO varchar(100);
begin
  -- Процедура возвращает счет Монеты и Назначение платежа  Монеты для КМП
  -- Связанная процедура для сайта HP_EDOC_WEB_GET_CL_INFO
  db_id = f_get_db_id();
  mnt_description = '';

  select short_fio
  from clients, hp_fio_clear(c_name_fio)
  where c_id = :c_id
  into c_name_fio;

  cur_name = (select cm_currency_name from company);

  if (db_id = 81001) then
    cur_name = left(cur_name, 3) || '.'; -- рубль -> руб.

  select k_kontract_no,
         iif(pr_code containing '.oivo_pack.', f_get_const('moneta_oivo_acc'), f_get_const('moneta_acc'))
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into k_kontract_no,
       mnt_id;

  if (coalesce(k_kontract_no, '') = '') then
    mnt_description = f_format('Погашение займа, %1, без НДС', c_name_fio);
  else
    mnt_description = f_format('Погашение займа по договору №%1, %2, без НДС',
      k_kontract_no, c_name_fio);
  suspend;
end
