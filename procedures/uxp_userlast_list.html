CREATE PROCEDURE UXP$USERLAST_LIST(
    UN_ID D_ID,
    TIMEZONE D_DOUBLE = null)
RETURNS (
    UA_UUID D_UUID,
    UA_LAST_TIME D_DATETIME,
    UA_CAPTION D_STR_255,
    UA_MAINMENU D_STR_4096,
    UA_VISUAL_TIME D_STR_CODE,
    UA_UUID_CODE D_STR_CODE)
AS
declare variable ml_id d_integer;
begin
  if (timezone is null) then
    timezone = 0;

  for select usl_ua_uuid, usl_open_time, ua_name
  from ux$userlast, ux$action
  where usl_un_id = :un_id and
        usl_ua_uuid = ua_uuid
  order by usl_open_time desc
  into :ua_uuid, :ua_last_time, :ua_caption
  do
  begin
    ua_uuid_code = uuid_to_char(ua_uuid);
    ua_visual_time = f_datetimetostr(dateadd(hour, timezone, ua_last_time));
    ua_visual_time = right(ua_visual_time, 5) || ' ' || left(ua_visual_time, 10);

    ua_mainmenu = '';
    ml_id = null;

    select item_id
    from uxp$actionmenu(:ua_uuid)
    into ml_id;

    ua_mainmenu = uxf$menupath(ml_id);

    suspend;
  end
end
