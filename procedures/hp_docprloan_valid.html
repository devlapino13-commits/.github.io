CREATE PROCEDURE HP_DOCPRLOAN_VALID(
    C_ID D_ID,
    PRODUCT_ID D_INTEGER,
    CLIENT_KIND D_SMALLINT,
    REP_DATE D_DATE,
    FULL_CHECK D_BOOL)
RETURNS (
    RES_SET_ID D_INTEGER,
    RES_DOC_SET_NAME D_STR_100,
    RES_DOCPROP_NAME D_STR_255,
    RES_VALID D_SMALLINT)
AS
declare variable dp_id d_integer;
declare variable pk_id d_integer;
declare variable citizen_kind d_integer;
declare variable valid_kind d_integer;
declare variable citizen_code d_str_10;
declare variable db_id d_integer;
declare variable doc_found d_bool;
declare variable dpl_date_kind d_integer;
declare variable dpl_date_days d_integer;
declare variable need_doc_count d_integer;
declare variable found_count d_integer;
declare variable group_no d_integer;
declare variable lgota_date_begin d_date;
declare variable lgota_date_end d_date;
begin
  res_set_id = 0;
  res_valid = 0;
  citizen_kind = null;
  res_docprop_name = '';

  if (full_check is null) then
    full_check = 0;

  -- Свойство документа по которому документ считается паспортом.
  select dp_id
  from docprop
  where dp_code = 'person.passport'
  into dp_id;

  if (dp_id is null) then
  begin
    res_docprop_name = 'Свойство "Паспорт" не найден';
    suspend;
    exit;
  end

  -- Среди действующих на дату документов выбираем по пориоритету:
  -- Сначала паспорт текущей страны
  -- если такого нет, выбираем последнюю по дате прикрепления
  select dpk_citizen_kind
  from clientdocs, docproppk
  where dpk_from_dp_id = :dp_id and
        dpk_is_delete = 0 and
        cds_from_pk_id = dpk_from_pk_id and
        cds_from_c_id = :c_id and
        cds_verify_status = 2 and
        coalesce(cds_create_datetime, cds_datetime) < (:rep_date + 1) and
        (cds_delete_datetime is null or cds_delete_datetime > (:rep_date + 1)) and
        cds_doc_date_begin <= :rep_date and
        coalesce(cds_doc_date_end, :rep_date) >= :rep_date
  order by dpk_citizen_kind, coalesce(cds_create_datetime, cds_datetime) desc, cds_doc_date_end desc
  rows 1
  into citizen_kind;

  -- Если нет прикрепленного действующего паспорта,
  -- смотрим поле "Гражданство" и по нему вычисляем тип гражданства
  if (citizen_kind is null) then
  begin
    db_id = f_get_db_id();

    select replace(rd_code, 'citizen.', '')
    from clients, refdata
    where c_id = :c_id and
          c_citizen = rd_id
    into citizen_code;

    if ((db_id = 81001 and citizen_code = 'ru') or
        (db_id in (39801, 41701) and citizen_code = 'kg') or
        (db_id = 94401 and citizen_code = 'az'))
    then citizen_kind = 0;
    else
      if ((db_id = 81001 and citizen_code = 'kg') or
          (db_id in (39801, 41701) and citizen_code = 'ru'))
      then citizen_kind = 1;
      else citizen_kind = 2;
  end

  if (citizen_kind is null) then
  begin
    res_docprop_name = 'Вид гражданства не определен';
    suspend;
    exit;
  end

  res_valid = 1;

  -- Проверяем наличие документа для каждого свойства в комлекте документов
  for select 0
  from rdb$database
  union
  select distinct dpl_set_id
  from docprloan
  where dpl_from_pr_id = :product_id and
        dpl_client_kind = :client_kind and
        dpl_is_delete = 0 and
        dpl_set_id <> 0
  into res_set_id
  do
  begin
    if (res_set_id = 0)
    then res_doc_set_name = 'Документы';
    else res_doc_set_name = 'Документы ' || res_set_id;

    doc_found = 1;

    for select dp_id, dp_name
    from docprop
    where dp_is_base = 1
    union
    select dpl_from_dp_id, dp_name
    from docprloan, docprop
    where dpl_from_pr_id = :product_id and
          dpl_client_kind = :client_kind and
          dpl_set_id = :res_set_id and
          dpl_is_delete = 0 and
          dpl_from_dp_id = dp_id
    into dp_id, res_docprop_name
    do
    begin
      res_valid = 0;
      dpl_date_kind = 0;
      dpl_date_days = 0;

      select dpl_date_kind, dpl_date_days
      from docprloan
      where dpl_from_pr_id = :product_id and
            dpl_from_dp_id = :dp_id and
            dpl_client_kind = :client_kind and
            dpl_set_id = :res_set_id and
            dpl_is_delete = 0
      into dpl_date_kind, dpl_date_days;

      if (dpl_date_kind = -1) then
        Continue;

      -- типы документов соответствующие свойству
      for select 0 from rdb$database
      union
      select dpk_group_no
      from docproppk
      where dpk_from_dp_id = :dp_id and
            dpk_citizen_kind = :citizen_kind and
            dpk_group_no <> 0 and
            dpk_is_delete = 0
      into group_no
      do
      begin
        need_doc_count = 1;
        if (group_no <> 0) then
        begin
          select count(*)
          from docproppk
          where dpk_from_dp_id = :dp_id and
                dpk_citizen_kind = :citizen_kind and
                dpk_group_no = :group_no and
                dpk_is_delete = 0
          into need_doc_count;
        end

        found_count = 0;
        for select dpk_from_pk_id, dpk_valid_kind
        from docproppk
        where dpk_from_dp_id = :dp_id and
              dpk_citizen_kind = :citizen_kind and
              dpk_group_no = :group_no and
              dpk_is_delete = 0
        into pk_id, valid_kind
        do
        begin
          lgota_date_begin = null;
          lgota_date_end = null;

          select dpg_date_begin, dpg_date_end
          from docpklgota
          where dpg_from_pk_id = :pk_id and
                dpg_citizen_kind = :citizen_kind and
                dpg_is_delete = 0 and
                coalesce(dpg_date_begin, cast('01.01.1900' as date)) <= :rep_date and
                coalesce(dpg_date_end, cast('01.01.1900' as date)) >= :rep_date
          into lgota_date_begin, lgota_date_end;

          if (lgota_date_begin is null) then
            lgota_date_begin = cast('01.01.1900' as date);
          if (lgota_date_end is null) then
            lgota_date_end = cast('01.01.1900' as date);

          -- Проверяем наличие
          if (exists(select *
          from clientdocs
          where cds_from_pk_id = :pk_id and
                cds_from_c_id = :c_id and
--                cds_verify_status = 2 and
                coalesce(cds_create_datetime, cds_datetime) < (:rep_date + 1) and
                (cds_delete_datetime is null or cds_delete_datetime > (:rep_date + 1)) and
                cds_doc_date_begin <= :rep_date and
                (:valid_kind = 1 or  -- проверка только наличия по дате документа
                (:valid_kind = 0 and -- проверка включая срок действия
                  (coalesce(cds_doc_date_end, :rep_date) >= :rep_date or
                   coalesce(cds_doc_date_end, :rep_date) between :lgota_date_begin and :lgota_date_end))) and
                (:dpl_date_kind = 0 or
                (:dpl_date_kind = 1 and datediff(day, cds_doc_date_begin, :rep_date) <= :dpl_date_days) or
                (:dpl_date_kind = 2 and (datediff(day, :rep_date, cds_doc_date_end) >= :dpl_date_days or
                                         cds_doc_date_end between :lgota_date_begin and :lgota_date_end))))) then
            found_count = found_count + 1;

          if (found_count >= need_doc_count) then
          begin
            res_valid = 1;
            break;
          end
        end

        if (found_count >= need_doc_count) then
        begin
          res_valid = 1;
          break;
        end
      end

      -- Если по указанному свойству нет прикрепленных документов, завершаем не проверяя остальных
      if (res_valid = 0) then
        doc_found = 0;

      suspend;
    end

    if (full_check = 0 and doc_found = 1) then
      Break;
  end
end
