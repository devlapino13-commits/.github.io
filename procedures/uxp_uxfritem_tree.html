CREATE PROCEDURE UXP$UXFRITEM_TREE(
    UX_UUID D_UUID,
    PARENT_UUID D_UUID,
    FULL_UI_ITEMS D_SMALLINT,
    PARENT_IS_READONLY D_BOOL)
RETURNS (
    RES_UFI_UUID D_UUID,
    RES_PARENT_UFI D_UUID,
    RES_UFI_READONLY D_BOOL,
    RES_EVENT_CONTROL D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_NAME D_STR_100,
    UFI_CAPTION_ALIGN D_SMALLINT,
    UFI_HINT D_STR_LONG,
    UFI_H_ALIGN D_SMALLINT,
    UFI_V_ALIGN D_SMALLINT)
AS
declare variable tmp_parent d_uuid;
declare variable tmp_readonly d_bool;
declare variable ufi_kind d_integer;
declare variable ufi_control d_str_code;
declare variable ufi_ufr_uuid d_uuid;
declare variable tmp_ux_uuid d_uuid;
declare variable is_grid_column d_bool;
declare variable prv_parent_uuid d_uuid;
begin
  if (full_ui_items is null) then
    full_ui_items = 0;

  if (parent_is_readonly is null) then
    parent_is_readonly = 0;

  if (full_ui_items = 0) then
    parent_is_readonly = 0;

  res_event_control = '';
  res_parent_ufi = parent_uuid;

  -- Если вызывается из ВНЕ рекурсии, возвращаем саму ветку
  if (parent_uuid is null) then
  begin
    if (full_ui_items = -1 and uxf$code(ux_uuid, 'kind') = 'uxframe') then
    begin
      for select ufi_uuid, ufi_readonly, ufi_kind, ufi_control,
                 ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_h_align, ufi_v_align
      from ux$fritem, ux$frame
      where ufr_uuid = :ux_uuid and
            ufi_owner = ufr_owner and
            ufi_ufr_uuid = ufr_uuid and
            ufi_kind = 1
      into res_ufi_uuid, tmp_readonly, ufi_kind, ufi_control,
           ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_h_align, ufi_v_align
      do
      begin
        parent_is_readonly = maxvalue(tmp_readonly, parent_is_readonly);
        res_event_control = ufi_control;
        Suspend;
  
        res_parent_ufi = res_ufi_uuid;
      end
    end else
    begin
      for select ufi_uuid, ufi_readonly, ufi_kind, ufi_control,
                 ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_h_align, ufi_v_align
      from ux$fritem
      where ufi_uuid = :ux_uuid
      into res_ufi_uuid, tmp_readonly, ufi_kind, ufi_control,
           ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_h_align, ufi_v_align
      do
      begin
        parent_is_readonly = maxvalue(tmp_readonly, parent_is_readonly);
        res_event_control = ufi_control;
        Suspend;
  
        res_parent_ufi = res_ufi_uuid;
      end
    end

    if (row_count = 0) then
      exit;
  end

  for select ufi_uuid, ufi_readonly, ufi_kind, ufi_control, ufi_ufr_uuid,
             ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_h_align, ufi_v_align
  from ux$fritem
  where ufi_parent_uuid = :ux_uuid
  order by ufi_position
  into res_ufi_uuid, tmp_readonly, ufi_kind, ufi_control, ufi_ufr_uuid,
       ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_h_align, ufi_v_align
  do
  begin
    res_ufi_readonly = maxvalue(tmp_readonly, parent_is_readonly);

    if (full_ui_items = 1 and ufi_control = 'frame') then
      res_ufi_uuid = ufi_ufr_uuid;

    res_event_control = ufi_control;
    if (ufi_kind = 0) then
      res_event_control = 'editor';

    suspend;

    if (full_ui_items = 0 and ufi_control = 'frame') then
      Continue;

    is_grid_column = 0;
    if (full_ui_items = 1 and ufi_control in ('grid', 'tree')) then
      Continue;

    if (ufi_control in ('grid', 'tree')) then
      is_grid_column = 1;

    tmp_ux_uuid = res_ufi_uuid;
    tmp_parent = res_ufi_uuid;
    tmp_readonly = res_ufi_readonly;

    if (full_ui_items in (-1, 1) and ufi_control = 'frame') then
      tmp_ux_uuid = ufi_ufr_uuid;

    prv_parent_uuid = res_parent_ufi;
    for select res_ufi_uuid, res_parent_ufi, res_ufi_readonly, res_event_control,
               ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_h_align, ufi_v_align
    from uxp$uxfritem_tree(:tmp_ux_uuid, :tmp_parent, :full_ui_items, :tmp_readonly)
    into :res_ufi_uuid, :res_parent_ufi, :res_ufi_readonly, :res_event_control,
         ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_h_align, ufi_v_align
    do
    begin
      if (is_grid_column = 1) then
      begin
        if (res_event_control = 'editor') then
          res_event_control = 'column';
        else
          res_event_control = 'band';
      end

      suspend;
    end

    res_parent_ufi = prv_parent_uuid;
  end
end
