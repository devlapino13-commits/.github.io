CREATE PROCEDURE HP_SUD_GET_CLIENTSLIST_UX(
    PARAM_JSON D_STR_255)
RETURNS (
    SMC_ID D_ID,
    FIO D_STR_100,
    SUM_OS D_CURRENCY,
    SUM_PERC D_CURRENCY,
    SUM_FINE D_CURRENCY,
    SUM_OTHER D_CURRENCY,
    SUM_ALL D_CURRENCY,
    SUM_DATE D_DATE,
    SLC_ID D_ID,
    ADDRESS D_STR_LONG,
    REQUEST_STATUS D_STR_100,
    SLC_LAST_SUL_ID D_ID,
    STAGE_STR D_STR_100,
    STAGE_DATE D_DATETIME,
    STAGE_USER D_STR_100,
    COLLECTOR_NAME D_STR_255,
    AGENT_NAME D_STR_255,
    COLLECT_STATUS D_STR_CODE,
    COLLECT_BEGIN_DATE D_DATE,
    INITORDER_NUMBER D_STR_CODE,
    COLLECT_SEND_METHOD D_STR_255,
    INITORDER_DATE D_DATE,
    COLLECT_MARSHAL D_STR_255,
    COLLECT_MARSHAL_DEPART D_STR_LONG,
    COLLECT_ROSP D_STR_LONG,
    COLLECTOR_ID D_ID,
    AGENT_ID D_ID,
    DOC_ID D_ID,
    DOC_DATE D_DATE,
    STAGE_CODE D_STR_CODE,
    SMC_NO D_INTEGER,
    IS_READY_REPEAT D_BOOL)
AS
declare variable slc_auto_worldcourt d_id;
declare variable slc_auto_federalcourt d_id;
declare variable smc_auto_collect d_id;
declare variable proc_type d_str_code;
declare variable object_kind d_integer;
declare variable smc_agent_id d_id;
declare variable smc_from_un_id d_id;
declare variable kind d_smallint;
declare variable smc_status d_smallint;
declare variable smr_id d_id;
declare variable req_stage d_str_code;
declare variable param d_str_10;
declare variable is_show d_bool;
begin
  if (coalesce(param_json, '') = '') then
    exit;

  is_show = 1;
  smr_id = f_json_str_get_value(param_json, 'request_id', null);
  req_stage = f_json_str_get_value(param_json, 'stage_code', null);
  param = f_json_str_get_value(param_json, 'param', null);

  for select slc_id, c_name_fio, slc_sum_date, slc_sum_os, slc_sum_perc, slc_sum_fine, slc_sum_other, slc_sum_all, slc_address,
        slc_auto_worldcourt, slc_auto_federalcourt, slc_last_sul_id
  from sud_mainreq, sud_loanclient, knclient, clients
  where smr_id = :smr_id
    and kc_id = slc_from_kc_id
    and c_id = kc_from_c_id
    and slc_from_smr_id = smr_id
  into slc_id, fio, sum_date, sum_os, sum_perc, sum_fine, sum_other, sum_all, address,
      slc_auto_worldcourt, slc_auto_federalcourt, slc_last_sul_id
  do
  begin
    is_show = 1;
    request_status = null;
    stage_date = null;
    object_kind = null;
    kind = null;
    stage_user = null;
    stage_code = req_stage;

    if (req_stage containing '|D') then
    begin
      proc_type = (select f_enum('sud_proc_type', smd_proc_type, 'code', 0) from sud_maindocs where smd_id = :smr_id);

      if (proc_type is not null) then
      begin
        kind =  case proc_type
                  when 'court_order' then 1
                  when 'petition' then 2
                  else -1
                end;
  
        if (kind = -1) then
          exception e_default 'Unknown proc_type = ' || coalesce(proc_type, '<NULL>');

        if ((proc_type = 'court_order' and slc_auto_worldcourt is not null)
          or (proc_type = 'petition' and slc_auto_federalcourt is not null))
        then
          request_status = 'Подсудность определена';
      end
    end
    ------------------------------------------
    else if (req_stage containing '|C') then
    begin
      stage_code = '|J5|';

      select sul_datetime, sul_object_kind
      from sud_mainlog
      where sul_id = :slc_last_sul_id
      into stage_date, object_kind;

      if (object_kind = f_objkind('sud_maincollect', 'id')) then
      begin
        for select f_enum('sud_collect_status', smc_collect_status, 'name', 0),
               smc_begin_date, smc_number, smc_send_method, smc_initorder_date, smc_marshal, 
               smc_marshal_depart, smc_rosp, smc_status, smc_from_sag_id,
               smc_from_un_id, smc_agent_id, smc_doc_id, smc_doc_date, smc_agent_id, smc_auto_collect, smc_no, smc_id
        from sud_maincollect 
        where smc_from_slc_id = :slc_id
        into collect_status, collect_begin_date, initorder_number, collect_send_method, initorder_date, collect_marshal,
             collect_marshal_depart, collect_rosp, smc_status, collector_id, smc_from_un_id,
             agent_id, doc_id, doc_date, smc_agent_id, smc_auto_collect, smc_no, smc_id
        do
        begin
          stage_code = '|C0|';
          is_ready_repeat = 0;
          request_status = '';
    
          if (smc_status = 0) then
            stage_code = iif(collector_id is null or coalesce(collect_send_method, '') = '', '|C1|', '|C2|');
          else if (smc_status = 1) then
          begin
            if (smc_agent_id is not null) then
              stage_code = '|C6|';
            else
              stage_code = iif(coalesce(initorder_number, '') = '', '|C3|', '|C4|');
          end
          else if (smc_status = 2) then
          begin
            stage_code = '|C5|';

            if ((select min(smc_status) = 2 and not max(smc_collect_status = f_enum('sud_collect_status', 'executed', 'id'))
                  from sud_maincollect where smc_from_slc_id = :slc_id)
              and (collect_status = f_enum('sud_collect_status', 'rejection', 'name')
                or dateadd(month, 6, doc_date) < current_date))
            then
              is_ready_repeat = 1;
          end

          if (param not in ('A0', 'C0') and ('|C0|' || req_stage) not containing stage_code) then
            continue;

          collector_name = (select sag_name from sud_agents where sag_id = :collector_id);
          agent_name = (select sag_name from sud_agents where sag_id = :smc_agent_id);
          stage_user = (select un_fio from username where un_id = :smc_from_un_id);
          kind = 3;
  
          if (smc_auto_collect is not null)
          then
            request_status = 'Подсудность ФССП определена';
          else if (kind is not null) then
             select status_str
             from hp_sud_get_request_status(:slc_id, :kind)
             into request_status;

          stage_str = (select list(distinct name) from hp_sud_get_stages where :stage_code containing code);

          if (smc_status = 2 and doc_id is null) then
             stage_code = stage_code || '|C7|';

          suspend;
          is_show = 0; -- RFCRU-4741
        end
      end
      else
        if (param containing '|C') then
          continue;
    end

    if (is_show = 1) then
    begin
      if (request_status is null and kind is not null) then
         select status_str
         from hp_sud_get_request_status(:slc_id, :kind)
         into request_status;
  
      stage_str = (select list(distinct name) from hp_sud_get_stages where :stage_code containing code);

      suspend;
    end
  end
end
