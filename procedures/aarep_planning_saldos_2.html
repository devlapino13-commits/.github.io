CREATE PROCEDURE AAREP_PLANNING_SALDOS_2(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_AUTO_PERIOD INTEGER,
    REP_AUTO_KOEF DECIMAL(18,7),
    REP_INCLUDE_FIRST_DAY INTEGER)
RETURNS (
    MA_BRANCH_ID INTEGER,
    MA_BRANCH_NAME VARCHAR(100),
    MA_REGION_ID INTEGER,
    MA_REGION_NAME VARCHAR(100),
    MA_SALDO_0 DECIMAL(18,4),
    MA_SALDO_1 DECIMAL(18,4),
    MA_SALDO_2 DECIMAL(18,4),
    MA_SALDO_3 DECIMAL(18,4),
    MA_SALDO_4 DECIMAL(18,4),
    MA_SALDO_5 DECIMAL(18,4),
    MA_SALDO_6 DECIMAL(18,4),
    MA_SALDO_7 DECIMAL(18,4),
    MA_SALDO_8 DECIMAL(18,4),
    MA_SALDO_9 DECIMAL(18,4),
    MA_SALDO_10 DECIMAL(18,4),
    MA_SALDO_11 DECIMAL(18,4),
    MA_SALDO_12 DECIMAL(18,4),
    MA_SALDO_13 DECIMAL(18,4),
    MA_SALDO_14 DECIMAL(18,4),
    MA_DATE_OPER DATE)
AS
declare variable ma_saldo_cash decimal(18,4);
declare variable ma_saldo_bank decimal(18,4);
declare variable ma_admin_minus_process decimal(18,4);
declare variable ma_summa_miniappnew decimal(18,4);
declare variable ma_summa_minus_new decimal(18,4);
declare variable ma_summa_minus_repeat decimal(18,4);
declare variable ma_saldo_end decimal(18,4);
declare variable ma_transfer_minus decimal(18,4);
declare variable ma_transfer_plus decimal(18,4);
declare variable ma_admin_minus decimal(18,4);
declare variable ma_admin_plus decimal(18,4);
declare variable ma_summa_minus decimal(18,4);
declare variable ma_summa_plus_admin decimal(18,4);
declare variable ma_summa_plus_perc decimal(18,4);
declare variable ma_summa_plus_os decimal(18,4);
declare variable reg_prior integer;
begin

  ma_date_oper = :rep_date_begin - 1 + :rep_include_first_day;
  ma_summa_plus_os = 0;
  ma_summa_plus_perc = 0;
  ma_summa_plus_admin = 0;
  ma_summa_minus = 0;
  ma_admin_plus = 0;
  ma_admin_minus = 0;
  ma_transfer_plus = 0;
  ma_transfer_minus = 0;
  ma_summa_minus_repeat = 0;
  ma_summa_minus_new = 0;
  ma_summa_miniappnew = 0;
  ma_admin_minus_process = 0;

  ma_saldo_0 = 0;
  ma_saldo_1 = 0;
  ma_saldo_2 = 0;
  ma_saldo_3 = 0;
  ma_saldo_4 = 0;
  ma_saldo_5 = 0;
  ma_saldo_6 = 0;
  ma_saldo_7 = 0;
  ma_saldo_8 = 0;
  ma_saldo_9 = 0;
  ma_saldo_10 = 0;
  ma_saldo_11 = 0;
  ma_saldo_12 = 0;
  ma_saldo_13 = 0;
  ma_saldo_14 = 0;
  for select r1.reg_unicode, r1.parent, r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
      from region_list(:rep_region_id, 1, 3, 1), region r1, region r2
      where (id_list = r1.reg_unicode) and
            r1.parent = r2.reg_unicode and
            r1.reg_status = 3
      order by r1.parent, r1.reg_unicode
      into :ma_region_id, :ma_branch_id, :ma_region_name, :ma_branch_name
  do
  begin
    suspend;
  end
  ma_saldo_0 = 0;
  ma_saldo_1 = 0;
  ma_saldo_2 = 0;
  ma_saldo_3 = 0;
  ma_saldo_4 = 0;
  ma_saldo_5 = 0;
  ma_saldo_6 = 0;
  ma_saldo_7 = 0;
  ma_saldo_8 = 0;
  ma_saldo_9 = 0;
  ma_saldo_10 = 0;
  ma_saldo_11 = 0;
  ma_saldo_12 = 0;
  ma_saldo_13 = 0;
  ma_saldo_14 = 0;

  for select res_region_id, r1.parent, r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name,
             sum(res_saldo_cash), sum(res_saldo_bank)
      from aarep_money_saldo_region(:rep_region_id, :rep_date_begin - 1 + :rep_include_first_day), region r1, region r2
      where res_region_id = r1.reg_unicode and
            r1.parent = r2.reg_unicode
      group by res_region_id, r1.parent, r1.reg_code, r1.reg_name, r2.reg_code, r2.reg_name
      into :ma_region_id, :ma_branch_id, :ma_region_name, :ma_branch_name, :ma_saldo_cash, :ma_saldo_bank
  do
  begin
    ma_saldo_end = :ma_saldo_cash + :ma_saldo_bank;
    ma_saldo_0 = :ma_saldo_end;
    suspend;
  end
  reg_prior = 0;

  for select ma_region_id, ma_date_oper, sum(ma_summa_plus_os), sum(ma_summa_plus_perc), sum(ma_summa_plus_admin),
             sum(ma_summa_minus), sum(ma_admin_plus), sum(ma_admin_minus), sum(ma_transfer_plus),
             sum(ma_transfer_minus), sum(ma_summa_minus_repeat), sum(ma_summa_minus_new), sum(ma_summa_miniappnew),
             sum(ma_admin_minus_process), r1.parent, r1.reg_code || ' ' || r1.reg_name,
             r2.reg_code || ' ' || r2.reg_name
      from aarep_planning_report_temp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_auto_period, :rep_auto_koef, :rep_include_first_day), region r1, region r2
      where ma_region_id = r1.reg_unicode and
            r1.parent = r2.reg_unicode
      group by ma_region_id, ma_date_oper, r1.parent, r1.reg_code, r1.reg_name, r2.reg_code, r2.reg_name
      order by ma_region_id, ma_date_oper
      into :ma_region_id, :ma_date_oper, :ma_summa_plus_os, :ma_summa_plus_perc, :ma_summa_plus_admin, :ma_summa_minus,
           :ma_admin_plus, :ma_admin_minus, :ma_transfer_plus, :ma_transfer_minus, :ma_summa_minus_repeat,
           :ma_summa_minus_new, :ma_summa_miniappnew, :ma_admin_minus_process, :ma_branch_id, :ma_region_name,
           :ma_branch_name
  do
  begin
    ma_saldo_0 = 0;
    ma_saldo_1 = 0;
    ma_saldo_2 = 0;
    ma_saldo_3 = 0;
    ma_saldo_4 = 0;
    ma_saldo_5 = 0;
    ma_saldo_6 = 0;
    ma_saldo_7 = 0;
    ma_saldo_8 = 0;
    ma_saldo_9 = 0;
    ma_saldo_10 = 0;
    ma_saldo_11 = 0;
    ma_saldo_12 = 0;
    ma_saldo_13 = 0;
    ma_saldo_14 = 0;
    if (reg_prior <> ma_region_id) then
    begin
      ma_saldo_end = 0;
      reg_prior = ma_region_id;
    end
    ma_saldo_end = :ma_summa_plus_os + :ma_summa_plus_perc + :ma_summa_plus_admin - :ma_summa_minus + :ma_admin_plus - :ma_admin_minus - :ma_admin_minus_process + :ma_transfer_plus - :ma_transfer_minus - :ma_summa_minus_repeat - :ma_summa_minus_new - :ma_summa_miniappnew;

    if ((:ma_date_oper - :rep_date_begin) = 0) then
      ma_saldo_1 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 1) then
      ma_saldo_2 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 2) then
      ma_saldo_3 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 3) then
      ma_saldo_4 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 4) then
      ma_saldo_5 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 5) then
      ma_saldo_6 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 6) then
      ma_saldo_7 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 7) then
      ma_saldo_8 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 8) then
      ma_saldo_9 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 9) then
      ma_saldo_10 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 10) then
      ma_saldo_11 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 11) then
      ma_saldo_12 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 12) then
      ma_saldo_13 = :ma_saldo_end;
    if ((:ma_date_oper - :rep_date_begin) = 13) then
      ma_saldo_14 = :ma_saldo_end;
    suspend;
  end
end
