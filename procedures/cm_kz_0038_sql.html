CREATE PROCEDURE CM_KZ_0038_SQL(
    K_ID D_ID,
    CHECK_KIND D_STR_255,
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable DECISION_STATUS D_SMALLINT;
declare variable DECISION D_SMALLINT;
declare variable RATING D_STR_CODE;
declare variable GPC_ID D_ID;
declare variable MA_ID D_ID;
declare variable OBJ_KIND D_INTEGER;
declare variable OBJ_ID D_ID;
begin
  res_info_text = 'Проверка Fraud Score ГКБ';
  res_error_text = '';
  res_error_is_critical = 0;

  if (coalesce(f_get_const('gkb_fraud_is_active'), '0') <> '1') then
    exit;

  -- Ищем результат сначала по договору, потом по его заявке
  obj_kind = f_objkind('kontract', 'id');
  obj_id = :k_id;

  select decision_status, decision, rating, gpc_id
  from hp_gkb_fraud_query_status(:obj_kind, :obj_id)
  into :decision_status, :decision, :rating, :gpc_id;

  if (decision_status = -1) then
  begin
    -- по договору запроса не было — берём по заявке
    select kn.k_from_ma_app_id from kontract kn where kn.k_id = :k_id into :ma_id;
    if (ma_id is not null) then
    begin
      obj_kind = f_objkind('miniapp', 'id');
      obj_id = :ma_id;
      select decision_status, decision, rating, gpc_id
      from hp_gkb_fraud_query_status(:obj_kind, :obj_id)
      into :decision_status, :decision, :rating, :gpc_id;
    end
  end

  -- Категория E (недоступность) — пропускаем
  if (decision_status = -3) then
    exit;

  -- Запрос в работе — просим подождать
  if (decision_status = -2) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Проверка Fraud Score ГКБ не завершена. Пожалуйста, подождите.';
    suspend;
    exit;
  end

  -- Запроса не было вовсе — на утверждении это трактуем как недоступность (E),
  -- фиксируем и пропускаем (на заявке автоотказ уже отработал бы).
  if (decision_status = -1) then
    exit;

  -- Результат есть
  if (decision = 2) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Отказ по результатам проверки Fraud Score.';
    suspend;
  end
  else if (decision = 1) then
  begin
    -- critical = 1 — «мягкий» блок. Обходится правом
    -- FinStart.Access_To_Approve_NonStandart_Loans (в HP_APPROVE_LOAN такому
    -- пользователю critical_level ставится в 1, и условие 1 > 1 не срабатывает).
    --
    -- Практика на сегодня: вызов в HP_APPROVE_LOAN передаёт имя права без
    -- префикса FinStart., поэтому has_access всегда 0 и правом никто не
    -- пользуется. Значит утверждение из МП блокируется, а из ФС проходит —
    -- там проверка условий пропускается целиком (if (is_finstart = 0)).
    -- Отсюда и текст «может быть утверждена через ФС».
    -- Правку вызова договорились не вносить.
    res_error_is_critical = 1;
    res_error_text = 'Клиент отнесен к категории повышенного риска мошенничества ' ||
                     '(Fraud Score: ' || upper(rating) || '). Выдача возможна, ' ||
                     'рекомендуется визуальная проверка клиента и снижение ' ||
                     'суммы кредита при необходимости. Заявка может быть ' ||
                     'утверждена через ФС';
    suspend;
  end
  -- decision = 0 → pass
end
