CREATE PROCEDURE UXP$UXFILTER_UPDATE(
    UFL_UUID_CODE D_STR_CODE,
    UFL_OWNER_CODE D_STR_CODE,
    UFL_NAME D_STR_100,
    UFL_MEMO D_STR_255,
    UFL_FILTER_TEXT D_STR_4096,
    UFL_FILTER_CAPTION D_STR_4096,
    UFL_FILTER_BLOB D_BLOB_BINARY,
    UN_ID D_ID)
RETURNS (
    NEW_UFL_UUID_CODE D_STR_CODE)
AS
declare variable un_name d_str_code;
declare variable ufl_uuid d_uuid;
declare variable ufl_owner d_uuid;
declare variable new_ufl_uuid d_uuid;
begin
  if (un_id is null) then
    un_id = uxf$user();

  un_name = uxf$user_login(un_id);

  if (coalesce(ufl_uuid_code, '') <> '') then
    ufl_uuid = char_to_uuid(ufl_uuid_code);

  if (coalesce(ufl_owner_code, '') <> '') then
    ufl_owner = char_to_uuid(ufl_owner_code);
  else
    exception e_default 'Empty ufl_owner_code';

  if (ufl_uuid is not null) then
  begin
    update ux$filter
    set ufl_name = iif(coalesce(:ufl_name, '') = '', ufl_name, :ufl_name),
        ufl_memo = iif(coalesce(:ufl_memo, '') = '', ufl_memo, :ufl_memo),
        ufl_user = :un_name,
        ufl_filter_text = :ufl_filter_text,
        ufl_filter_caption = :ufl_filter_caption,
        ufl_filter_blob = :ufl_filter_blob
    where ufl_uuid = :ufl_uuid;

    new_ufl_uuid = ufl_uuid;
  end else
  begin
    insert into ux$filter (ufl_owner, ufl_name, ufl_memo, ufl_user, ufl_filter_text,  ufl_filter_caption, ufl_filter_blob)
    values (:ufl_owner, :ufl_name, :ufl_memo, :un_name, :ufl_filter_text, :ufl_filter_caption, :ufl_filter_blob)
    returning ufl_uuid
    into new_ufl_uuid;
  end

  new_ufl_uuid_code = uuid_to_char(new_ufl_uuid);

  suspend;
end
