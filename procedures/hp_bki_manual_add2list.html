CREATE PROCEDURE HP_BKI_MANUAL_ADD2LIST(
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID,
    EVENT_DATE D_DATE,
    BKK_ID D_ID,
    EVENT_CODE D_STR_CODE = null,
    THIS_EVENT_ONLY D_BOOL = 0)
AS
declare variable bmm_id d_id;
declare variable bml_id d_id;
declare variable bml_allow_events d_str_long;
begin
  select bmm_id
  from bki_manual_main
  where bmm_status = 0
  into bmm_id;

  if (bmm_id is null) then
    insert into bki_manual_main (bmm_from_bkk_id)
    values (:bkk_id)
    returning bmm_id
    into bmm_id;

  if (coalesce(this_event_only, 0) = 0) then
    event_code = null;

  select bml_id, bml_allow_events
  from bki_manual_list
  where bml_from_bmm_id = :bmm_id
    and bml_object_kind = :object_kind
    and bml_object_id = :object_id
    and bml_date = :event_date
  into bml_id, bml_allow_events;

  if (coalesce(this_event_only, 0) = 0) then
    bml_allow_events = null;

  if (coalesce(event_code, '') <> ''
    and coalesce(bml_allow_events, '') not containing event_code)
  then
     bml_allow_events = trim(iif(coalesce(bml_allow_events, '') = '', '', bml_allow_events || ',')) || event_code;

  if (bml_id is null) then
    insert into bki_manual_list (bml_from_bmm_id, bml_object_kind, bml_object_id, bml_date, bml_allow_events)
    values (:bmm_id, :object_kind, :object_id, :event_date, :bml_allow_events);
  else
    update bki_manual_list
    set bml_allow_events = :bml_allow_events
    where bml_id = :bml_id;
end
