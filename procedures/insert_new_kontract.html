CREATE PROCEDURE INSERT_NEW_KONTRACT(
    MA_REG_ID INTEGER,
    MA_ID INTEGER,
    MA_NUMBER INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    KF_ID DECIMAL(18,0),
    K_ID DECIMAL(18,0),
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER)
AS
declare variable cr_fund varchar(20);
declare variable cr_project varchar(20);
declare variable pr_currency varchar(3);
declare variable ma_date_app date;
declare variable ma_no integer;
declare variable ma_date_oper date;
declare variable ma_from_pr_id integer;
declare variable ma_summa_minus decimal(18,4);
declare variable ma_kolich_repay integer;
declare variable ma_percent decimal(18,7);
declare variable pr_percent_for_days integer;
declare variable ma_kolich_client integer;
declare variable pr_from_pac_id integer;
declare variable pr_index_curr_code varchar(3);
declare variable ma_repay_day integer;
declare variable res_id integer;
declare variable ma_refund integer;
declare variable add_info varchar(255);
declare variable ma_prepayment_summa curr_type;
declare variable k_from_pr_id integer;
declare variable ma_disb_kind integer;
declare variable is_online d_smallint;
declare variable by_rating d_smallint;
declare variable k_rating_source d_integer;
begin
  ma_reg_id = 0;
  select ma_from_kf_id, ma_date_oper, ma_number, ma_from_pr_id, ma_repay_day
  from miniapp
  where ma_id = :ma_id and
        ma_date_end >= '31.12.2999' rows 1
  into :kf_id, :ma_date_oper, :ma_number, :ma_from_pr_id, :ma_repay_day;

  select k_id, k_from_kf_id, k_circle, k_circle_internal
  from kontract
  where k_from_ma_app_id = :ma_id and
        k_status = -2
  order by k_disb_date desc
  rows 1
  into :k_id, :kf_id, :k_circle, :k_circle_internal;

  by_rating = 0;
  k_rating_source = 0;
  select by_rating, k_rating_source
  from hp_check_rating_result('miniapp', :ma_id)
  into by_rating, k_rating_source;
    
  if (k_id is not null) then
  begin
    update kontract set
      k_status = 0,
      k_repay_day = :ma_repay_day,
      k_by_rating = :by_rating,
      k_rating_source = :k_rating_source
    where k_id = :k_id;

    delete from kontmat km
    where km.kml_object_kind = 9 and
          km.kml_object_id = :k_id and
          coalesce(km.kml_total, 0) > 0;

    delete from kontmat km
    where km.kml_object_kind = 9 and
          km.kml_object_id = :k_id and
          coalesce(km.kml_total, 0) = 0;

    execute procedure hp_miniapp_kontract_sync(11, :ma_id);

    select res_id
    from hp_make_schedule(:un_id, 9, :k_id, 0, null, -2)
    into :res_id;

    suspend;
    exit;
  end

  k_id = -1;
--  select k_id, k_circle, k_circle_internal, k_from_pr_id
--  from kontract, kontstate
--  where k_status <= 0 and
--        k_from_kf_id = :kf_id and
--        ks_from_k_id = k_id and
--        ks_ol_is_delete = 0 and
--        ks_state = 0 and
--        ks_date_end >= cast('01.01.2999' as date) and
--        k_from_ma_app_id <> :ma_id
--  order by k_id desc
--  rows 1
--  into :k_id, :k_circle, :k_circle_internal, k_from_pr_id;
  if ((coalesce(k_id, -1) <> -1) or (not exists(select *
                                                from product
                                                where pr_id = :ma_from_pr_id and
                                                      pr_code containing '.PATENT PLUS.') and
      exists(select *
             from product
             where pr_id = :k_from_pr_id and
                   pr_code containing '.PATENT.'))) then
  begin
    /* существует свободный договор в процессе */
    delete from schedule
    where sh_from_k_id = :k_id;

    delete from knclient
    where kc_from_k_id = :k_id;

-- удаление поручителей
    delete from zalog
    where z_from_k_id = :k_id;

    /* редактирование договора под заявку */
    select cr_fund, cr_project
    from curruserdata
    where cr_from_u_id = :un_id
    into :cr_fund, :cr_project;

    if (cr_fund is null) then
    begin
      select f_code
      from fund
      where f_is_active = 1
      into :cr_fund;
    end

    if (cr_project is null) then
    begin
      select pt_id
      from projects
      where pt_is_active = 1
      into :cr_project;
    end

    select pr_currency, ma_date_app, ma_no, ma_date_oper, ma_from_pr_id, ma_date_oper, ma_summa_minus, ma_kolich_repay,
           ma_percent, ma_kolich_repay, ma_percent, pr_percent_for_days, ma_summa_minus, ma_kolich_client,
           pr_from_pac_id, pr_index_curr_code, ma_repay_day, coalesce(ma_refund, 0), coalesce(ma_add_info, ''),
           maa.ma_prepayment_summa, ma_disb_kind, maa.ma_is_online
    from miniapp maa, product
    where ma_from_pr_id = pr_id and
          ma_id = :ma_id and
          ma_number = :ma_number
    into :pr_currency, :ma_date_app, :ma_no, :ma_date_oper, :ma_from_pr_id, :ma_date_oper, :ma_summa_minus,
         :ma_kolich_repay, :ma_percent, :ma_kolich_repay, :ma_percent, :pr_percent_for_days, :ma_summa_minus,
         :ma_kolich_client, :pr_from_pac_id, :pr_index_curr_code, :ma_repay_day, :ma_refund, :add_info,
         :ma_prepayment_summa, :ma_disb_kind, :is_online;

    update kontract kont
    set k_currency = :pr_currency,
        k_applicate_date = :ma_date_app,
        k_applicate_no = :ma_no,
        k_kontract_date = :ma_date_oper,
        k_from_pr_id = :ma_from_pr_id,
        k_disb_date = :ma_date_oper,
        k_amount_applicate = :ma_summa_minus,
        k_kolich_lgota = 0,
        k_percent = :ma_percent,
        k_percent_lgota = 0,
        k_kolich_repay = :ma_kolich_repay,
        k_from_f_id = :cr_fund,
        k_from_pt_id = :cr_project,
        k_status = 0,
        k_amount_approved = :ma_summa_minus,
        k_percent_for_days = :pr_percent_for_days,
        k_kolich_client_applicate = :ma_kolich_client,
        k_kolich_client_approved = :ma_kolich_client,
        k_repay_kolich_closed = 0,
        k_kommis_perc = 0,
        k_from_cs_id = null,
        k_clients_leader_id = null,
        k_clients_cashier_id = null,
        k_credit_target = null,
        k_from_pac_id = :pr_from_pac_id,
        k_is_kredline = 0,
        k_kredline_end = null,
        k_kont_last_date = null,
        k_stop_perc = null,
        k_stop_fine = null,
        k_index_curr_code = :pr_index_curr_code,
        k_index_curr_rate = 1,
        k_from_ma_reg_id = :ma_reg_id,
        k_from_ma_app_id = :ma_id,
        k_percent_effect = 0,
        k_percent_effect_fifd = 0,
        k_user_id = :un_id,
        k_percent_flat_annual = 0,
        k_percent_begin_annual = 0,
        k_financed = 0,
        k_repay_day = :ma_repay_day,
        k_refund = :ma_refund,
        k_add_info = :add_info,
        k_prepayment_summa = :ma_prepayment_summa,
        k_disb_kind = :ma_disb_kind,
        k_is_online = :is_online,
        k_by_rating = :by_rating,
        k_rating_source = :k_rating_source
    where k_id = :k_id;
  end
  else
  begin
    /* нужно создать новую заявку */
    k_circle = 0;
    select max(k_circle)
    from kontract
    where k_from_kf_id = :kf_id
    into :k_circle;
    if (k_circle is null) then
      k_circle = 0;
    k_circle = k_circle + 1;
    k_circle_internal = 1;
    k_id = (kf_id * 1000 + k_circle) * 100 + 1;

    select cr_fund, cr_project
    from curruserdata
    where cr_from_u_id = :un_id
    into :cr_fund, :cr_project;

    if (cr_fund is null) then
    begin
      select f_code
      from fund
      where f_is_active = 1 rows 1
      into :cr_fund;
    end

    if (cr_project is null) then
    begin
      select pt_id
      from projects
      where pt_is_active = 1 rows 1
      into :cr_project;
    end

    insert into kontract (k_id, k_from_kf_id, k_currency, k_applicate_date, k_applicate_no, k_kontract_no,
                          k_kontract_date, k_from_pr_id, k_disb_date, k_amount_applicate, k_stage, k_period,
                          k_kolich_lgota, k_percent, k_percent_lgota, k_kolich_repay, k_status, k_circle,
                          k_circle_internal, k_percent_per_day, k_amount_approved, k_percent_for_days,
                          k_kolich_client_applicate, k_kolich_client_approved, k_from_pac_id, k_index_curr_code,
                          k_index_curr_rate, k_from_ma_reg_id, k_from_ma_app_id, k_percent_input, k_percent_category,
                          k_percent_effect, k_percent_effect_fifd, k_user_id, k_kommis_perc, k_from_f_id, k_from_pt_id,
                          k_repay_day, k_refund, k_add_info, k_prepayment_summa, k_disb_kind, k_is_online, k_by_rating, k_rating_source)
    select :k_id, :kf_id, pr_currency, ma_date_app, ma_no, '', ma_date_oper, ma_from_pr_id, ma_date_oper,
           ma_summa_minus, 1, ma_kolich_repay, 0, ma_percent, 0, ma_kolich_repay, 0, :k_circle, :k_circle_internal,
           cast(ma_percent / pr_percent_for_days as decimal(15,7)), ma_summa_minus, pr_percent_for_days,
           ma_kolich_client, ma_kolich_client, pr_from_pac_id, pr_index_curr_code, 1, :ma_reg_id, :ma_id, 0, 0, 0, 0,
           :un_id, 0, :cr_fund, :cr_project, ma_repay_day, coalesce(ma_refund, 0), coalesce(ma_add_info, ''),
           ma_prepayment_summa, ma_disb_kind, ma_is_online, :by_rating, :k_rating_source
    from miniapp, product
    where ma_from_pr_id = pr_id and
          ma_id = :ma_id and
          ma_number = :ma_number;
  end

  execute procedure hp_miniapp_kontract_sync(11, :ma_id);

  update schedule
  set sh_from_k_id = :k_id
  where sh_from_k_id = -:ma_id;

  if (not(exists(select *
                 from schedule
                 where sh_from_k_id = :k_id))) then
  begin
    select res_id
    from hp_make_schedule(:un_id, 9, :k_id, 0, null, 0)
    into :res_id;
  end
  else
  begin
    select res_id
    from hp_make_schedule(:un_id, 9, :k_id, 0, null, -2)
    into :res_id;
  end
  suspend;
end
