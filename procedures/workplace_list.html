CREATE PROCEDURE WORKPLACE_LIST(
    ID DECIMAL(18,0),
    SELF_INCLUDE INTEGER,
    KIND INTEGER,
    IS_DOWN INTEGER)
RETURNS (
    ID_LIST DECIMAL(18,0),
    ID_PARENT DECIMAL(18,0),
    PARENT_NAME VARCHAR(100))
AS
begin
  for with recursive mc_list
      as (select wp_id, parent, wp_status
          from workplace
          where wp_id = :id
          union all
          select w.wp_id, w.parent, w.wp_status
          from workplace w
          join mc_list on ((:is_down = 1 and
                mc_list.wp_id = w.parent) or (:is_down = 0 and
                mc_list.parent = w.wp_id)))
      select wp_id, parent
      from mc_list
      where ((:self_include = 0 and
            wp_id <> :id) or
            (:self_include = 1)) and
            ((:kind = -1) or
            (:kind <> -1 and
            wp_status = :kind))
      into :id_list, :id_parent
  do
  begin
    if (id_parent is not null) then
    begin
      select distinct wp_name
      from workplace
      where wp_id = :id_parent rows 1
      into :parent_name;
    end
    else
    begin
      id_parent = null;
      parent_name = '';
    end
    suspend;
  end

end
