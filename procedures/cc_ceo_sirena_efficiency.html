CREATE PROCEDURE CC_CEO_SIRENA_EFFICIENCY(
    REP_DATE DATE,
    STEP INTEGER,
    REP_PERIOD VARCHAR(10))
RETURNS (
    CALLED_OPER_STR STR_255,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    ON_DELINQ_QTY INTEGER,
    IS_D_PERIOD_BLOCKED INTEGER,
    IS_D_PAYED_AFTER_CALL INTEGER,
    D_PAYED_AFTER_CALL CURR_TYPE,
    D_DELINQ_SUM CURR_TYPE,
    D_EFFECTIVE PERC_TYPE,
    ON_SIRENA_QTY INTEGER,
    IS_S_PERIOD_BLOCKED INTEGER,
    IS_S_PAYED_AFTER_CALL INTEGER,
    S_PAYED_AFTER_CALL CURR_TYPE,
    S_DELINQ_SUM CURR_TYPE,
    S_EFFECTIVE PERC_TYPE,
    D_LESS_30 INTEGER,
    D_30_AND_MORE INTEGER,
    S_LESS_30 INTEGER,
    S_30_AND_MORE INTEGER,
    IS_S_TO_SB INTEGER,
    IS_D_TO_SB INTEGER)
AS
declare variable delinq_days integer;
declare variable is_ok smallint;
declare variable ok_summ curr_type;
declare variable zad_sum curr_type;
declare variable sba_date_begin date;
declare variable i integer;
declare variable max_i integer;
declare variable proj_type str_100;
declare variable k_id id_type;
declare variable pant_date_begin date;
declare variable rep_date_tmp date;
begin
  if (step = 1) then
  begin
    rep_date_end = rep_date;

    if (rep_period = 'day') then
      rep_date_begin = rep_date_end;
    if (rep_period = 'month') then
      rep_date_begin = (select cast('01.' || cast(extract(month from cast(:rep_date as date)) as varchar(2)) || '.' || cast(extract(year from cast(:rep_date as date)) as varchar(4)) as date)
                        from rdb$database);

    /*
      for select K_ID,
             DELINQ_DAYS,
             'sirena' as PROJ_TYPE
      from CC_DELINQ_LIST_TMP(:REP_DATE_END, 1, null, null, null, 1, 999)
      union all
      select K_ID,
             DELINQ_DAYS,
             'delinq' as PROJ_TYPE
      from CC_DELINQ_LIST_TMP(:REP_DATE_END, 2, null, null, null, 7, 999)
      into :K_ID,
           :DELINQ_DAYS,
           :PROJ_TYPE
      */

-- получаю общий список активных ПРОСРОЧЕННЫХ договоров, которые, на дату конца отчета, висят на операторах КЦ по проектам "Сирена", "Звонки должникам"
    for select pantk.pant_oper_called, pantk.pant_date_begin, pantk.pant_from_k_id,
               cast(:rep_date_end as date) - oks_delinq_date as delinq_days,
               iif(oks_kc_id = 1000000000004, 'sirena', 'delinq')
        from pantera_kontracts pantk
        inner join oksummary oks on oks.oks_from_k_id = pantk.pant_from_k_id
        inner join refdata rd on pantk.pant_rd_project_type = rd.rd_id
        where --pantk.pant_project_type = 4
              rd.rd_code in ('OKTELL.Projects.Sirena', 'OKTELL.Projects.Sirena1', 'OKTELL.Projects.Sirena2') and
              pantk.pant_date_end >= pant_date_begin -- исключаю передачи в один и тот же день
              and
              :rep_date_end between pant_date_begin and pant_date_end and
              :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
              cast(:rep_date_end as date) - oks_delinq_date >= 1 -- кредит должне быть в просрочке
              and
              oks_delinq_date < :rep_date and
              oks.oks_close_state = 0
        into :called_oper_str, :pant_date_begin, :k_id, :delinq_days, :proj_type
    do
    begin
      proj_type = coalesce(proj_type, '');

      if (((proj_type = 'delinq') and
          (delinq_days >= 7)) or (proj_type = 'sirena')) then
      begin
        is_ok = 1;
        is_s_to_sb = 0;
        is_d_to_sb = 0;
        on_sirena_qty = 0;
        on_delinq_qty = 0;
        is_d_period_blocked = 0;
        is_s_period_blocked = 0;
        ok_summ = 0;
        is_s_payed_after_call = 0;
        is_d_payed_after_call = 0;
        zad_sum = 0;
        d_delinq_sum = 0;
        s_delinq_sum = 0;
        sba_date_begin = null;
        s_less_30 = 0;
        d_less_30 = 0;
        s_30_and_more = 0;
        d_30_and_more = 0;

-- проверяю на дату конца отчета, висит ли контракт за сб/коллекторами
        select sba.sba_date_begin
        from sb_akt sba
        where sba.sba_from_k_id = :k_id and
              sba.sba_from_sbl_id <> 0 and
              sba.sba_is_delete = 0 and
              :rep_date_end between sba.sba_date_begin and sba.sba_date_end
        into :sba_date_begin;

        if (sba_date_begin is not null) then -- если висит за сб/коллекторами
        begin
          is_ok = 0;

          if (sba_date_begin between rep_date_begin and rep_date_end) then -- если на СБ договор был передан за отчетный период
          begin
            if (proj_type = 'sirena') then
              is_s_to_sb = 1;
            else
              is_d_to_sb = 1;
            suspend;
          end
        end

        if (is_ok = 1) then -- если на сб/коллекторах не висит
        begin
          is_ok = 0;

          if (proj_type = 'sirena') then
          begin
            on_sirena_qty = 1;

            if (delinq_days < 30) then
              s_less_30 = 1;
            if (delinq_days >= 30) then
              s_30_and_more = 1;
          end
          else
          begin
            on_delinq_qty = 1;

            if (delinq_days < 30) then
              d_less_30 = 1;
            if (delinq_days >= 30) then
              d_30_and_more = 1;
          end

          if (pant_date_begin between :rep_date_begin and :rep_date_end) then
          begin
            if (proj_type = 'sirena') then
              is_s_period_blocked = 1;
            else
              is_d_period_blocked = 1;
          end

          select sum(zad_os + sh_calc_perc + sh_calc_fine)
          from aarep_make_zad_list(:k_id, :rep_date_end, 1)
          where sh_date is not null
          into :zad_sum;

          zad_sum = coalesce(zad_sum, 0);

          if (zad_sum > 0) then
          begin
            if (proj_type = 'sirena') then
              s_delinq_sum = zad_sum;
            else
              d_delinq_sum = zad_sum;
          end

          suspend;
        end
      end
    end

--  Расчет учетных сумм погашения клентов

    is_s_to_sb = 0;
    is_d_to_sb = 0;
    on_sirena_qty = 0;
    on_delinq_qty = 0;
    called_oper_str = null;
    pant_date_begin = null;
    is_d_period_blocked = 0;
    is_s_period_blocked = 0;
    ok_summ = 0;
    is_s_payed_after_call = 1; -- <<
    is_d_payed_after_call = 1; -- <<
    zad_sum = 0;
    d_delinq_sum = 0;
    s_delinq_sum = 0;
    sba_date_begin = null;
    s_less_30 = 0;
    d_less_30 = 0;
    s_30_and_more = 0;
    d_30_and_more = 0;
    s_payed_after_call = 0;
    d_payed_after_call = 0;

    for select k_id, oper_called, payed_sum, 'sirena'
        from aarep_sirena_bonus_rep(:rep_date_begin, :rep_date_end, 3, 8) -- Сирена
        union all
        select k_id, oper_called, payed_sum, 'delinq'
        from aarep_sirena_bonus_rep(:rep_date_begin, :rep_date_end, 4, 8) -- Просрочки
        into k_id, called_oper_str, ok_summ, proj_type
    do
    begin
      is_s_payed_after_call = 0;
      is_d_payed_after_call = 0;
      d_payed_after_call = 0;
      s_payed_after_call = 0;

      if (proj_type = 'sirena') then
      begin
        is_s_payed_after_call = 1;
        s_payed_after_call = ok_summ;
      end
      else
      begin
        is_d_payed_after_call = 1;
        d_payed_after_call = ok_summ;
      end

      suspend;
    end
  end
  if (step = 2) then -- на этом шаге формирую цикл, т.к. в отчет должны быть видны 7 последних результатов
  begin

    i = 0;
    rep_date_tmp = rep_date;

    max_i = 1;
    if (rep_period = 'day') then
      max_i = 7;

    while (i < max_i) do
    begin
      rep_date_tmp = rep_date - (i);

      for select called_oper_str, rep_date_begin, rep_date_end, sum(on_sirena_qty), sum(on_delinq_qty),
                 sum(is_d_period_blocked), sum(is_s_period_blocked), sum(is_s_payed_after_call),
                 sum(is_d_payed_after_call), sum(s_payed_after_call), sum(d_payed_after_call), sum(d_delinq_sum),
                 sum(s_delinq_sum), sum(is_s_to_sb), sum(is_d_to_sb), sum(s_less_30), sum(d_less_30),
                 sum(s_30_and_more), sum(d_30_and_more)
          from cc_ceo_sirena_efficiency(:rep_date_tmp, 1, :rep_period)
          group by called_oper_str, rep_date_begin, rep_date_end
          order by called_oper_str
          into called_oper_str, rep_date_begin, rep_date_end, on_sirena_qty, on_delinq_qty, is_d_period_blocked,
               is_s_period_blocked, is_s_payed_after_call, is_d_payed_after_call, s_payed_after_call,
               d_payed_after_call, d_delinq_sum, s_delinq_sum, is_s_to_sb, is_d_to_sb, s_less_30, d_less_30,
               s_30_and_more, d_30_and_more
      do
      begin
        d_effective = 0;
        s_effective = 0;

        if (on_delinq_qty > 0) then
        begin
          if (d_delinq_sum = 0) then
            d_effective = 100;
          else
            d_effective = is_d_payed_after_call * 100 / on_delinq_qty; --(D_PAYED_AFTER_CALL / D_DELINQ_SUM) * 100;
        end

        if (on_sirena_qty > 0) then
        begin
          if (s_delinq_sum = 0) then
            s_effective = 100;
          else
            s_effective = is_s_payed_after_call * 100 / on_sirena_qty; -- (S_PAYED_AFTER_CALL / S_DELINQ_SUM) * 100;
        end

        suspend;
      end

      i = i + 1;
    end
  end
  if (step = 3) then -- этот шаг, исключительно для сортировки
  begin
    for select called_oper_str, rep_date_begin, rep_date_end, on_sirena_qty, on_delinq_qty, is_d_period_blocked,
               is_s_period_blocked, is_s_payed_after_call, is_d_payed_after_call, s_payed_after_call,
               d_payed_after_call, d_delinq_sum, s_delinq_sum, is_s_to_sb, is_d_to_sb, s_less_30, d_less_30,
               s_30_and_more, d_30_and_more, d_effective, s_effective
        from cc_ceo_sirena_efficiency(:rep_date, 2, :rep_period)
        order by called_oper_str, rep_date_begin desc
        into called_oper_str, rep_date_begin, rep_date_end, on_sirena_qty, on_delinq_qty, is_d_period_blocked,
             is_s_period_blocked, is_s_payed_after_call, is_d_payed_after_call, s_payed_after_call, d_payed_after_call,
             d_delinq_sum, s_delinq_sum, is_s_to_sb, is_d_to_sb, s_less_30, d_less_30, s_30_and_more, d_30_and_more,
             d_effective, s_effective
    do
    begin
      suspend;
    end
  end
end
