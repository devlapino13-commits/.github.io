CREATE PROCEDURE HP_FILL_KN_OKSUMMARY_ONE(
    REP_K_ID DECIMAL(18,0),
    REP_OL_ID DECIMAL(18,0),
    NEED_CLOSE_LOAN INTEGER)
AS
declare variable end_of_current_month date;
declare variable ok_from_ol_id id_type;
declare variable ok_number integer;
declare variable ok_sh_id integer;
declare variable ok_type integer;
declare variable ok_os curr_type;
declare variable ok_perc curr_type;
declare variable ok_fine curr_type;
declare variable ok_pay_type integer;
declare variable ok_kommis curr_type;
declare variable ok_sh_os curr_type;
declare variable ok_sh_perc curr_type;
declare variable ok_kommis_adv curr_type;
declare variable ok_tax curr_type;
declare variable ok_ol_reg_id integer;
declare variable ok_ol_is_active smallint;
declare variable kc_id id_type;
declare variable has_sh_changes smallint;
declare variable res_ok_os curr_type;
declare variable res_ok_perc curr_type;
declare variable res_ok_fine curr_type;
declare variable res_ok_kommis curr_type;
declare variable res_ok_sh_os curr_type;
declare variable res_ok_sh_perc curr_type;
declare variable res_ok_kommis_adv curr_type;
declare variable res_ok_tax curr_type;
declare variable gr_disb_line integer;
declare variable cl_number integer;
declare variable kc_amount curr_type;
declare variable tmp_amount curr_type;
declare variable ol_id id_type;
declare variable gr_resh_sum curr_type;
declare variable diff_disb curr_type;
declare variable ok_sh_date date;
declare variable ok_ol_date date;
declare variable oks_date_begin d_date;
begin
  end_of_current_month = current_date - extract(day from current_date) + 1;
  end_of_current_month = dateadd(month, 1, end_of_current_month) - 1;

  has_sh_changes = 0;
  gr_disb_line = 0;
  cl_number = 0;
  gr_resh_sum = 0;
  for select ok_from_ol_id, ok_number, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine, ok_pay_type, ok_kommis,
             ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active
      from operkred
      where ok_from_k_id = :rep_k_id and
            ok_from_ol_id = :rep_ol_id
--      and ok_ol_date < '12.10.2015'
--      and (ok_ol_date < '12.10.2015' or (ok_ol_date = '12.10.2015' and ok_number <7))
      order by ok_ol_date, ok_from_ol_id, ok_number
      into :ok_from_ol_id, :ok_number, :ok_sh_id, :ok_sh_date, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_pay_type,
           :ok_kommis, :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax, :ok_ol_date, :ok_ol_reg_id, :ok_ol_is_active
  do
  begin
    if (ok_type in (2, 3)) then
      gr_resh_sum = gr_resh_sum + ok_sh_os;

-- распределение суммы по участникам
    for select kc_id, res_ok_os, res_ok_perc, res_ok_fine, res_ok_kommis, res_ok_sh_os, res_ok_sh_perc,
               res_ok_kommis_adv, res_ok_tax
        from hp_devide_kn_operkred(:rep_k_id, :ok_sh_id, :ok_sh_date, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_kommis, :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax, :ok_ol_is_active, :ok_ol_date)
        into kc_id, res_ok_os, res_ok_perc, res_ok_fine, res_ok_kommis, res_ok_sh_os, res_ok_sh_perc, res_ok_kommis_adv,
             res_ok_tax
    do
    begin
      insert into kn_operkred (ok_from_ol_id, ok_number, ok_from_kc_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                               ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                               ok_ol_reg_id, ok_ol_is_active, ok_is_actual)
      values (:ok_from_ol_id, :ok_number, :kc_id, :ok_sh_id, :ok_sh_date, :ok_type, :res_ok_os, :res_ok_perc,
              :res_ok_fine, :ok_pay_type, :res_ok_kommis, :res_ok_sh_os, :res_ok_sh_perc, :res_ok_kommis_adv,
              :res_ok_tax, :ok_ol_date, :ok_ol_reg_id, :ok_ol_is_active, 0);

      if (ok_ol_date = '16.02.2013' and
          ok_type = 1) then -- при амнистировании были корректировки
      begin
        if (exists(select *
                   from operlist
                   where ol_id = :rep_ol_id and
                         ol_memo starting 'Корректировка первоначального графика после амнистирование')) then
        begin
          diff_disb = 0;
          select sum(ok_os - ok_sh_os)
          from kn_operkred
          where ok_from_kc_id = :kc_id and
                ok_ol_is_active = 1 and
                ok_type in (1, 8)
          into :diff_disb;
          if (diff_disb is null) then
            diff_disb = 0;

          if (diff_disb <> 0) then
          begin
            update kn_operkred
            set ok_sh_os = ok_sh_os + :diff_disb
            where ok_from_kc_id = :kc_id and
                  ok_sh_id = :ok_sh_id and
                  ok_type = 1 and
                  ok_sh_os <> 0 and
                  ok_ol_date < :ok_ol_date and
                  ok_ol_is_active = 1 rows 1;
          end
        end
      end
    end
-- конец распределения

    if (ok_type = 8) then
      if (gr_disb_line < ok_number) then
        gr_disb_line = ok_number;

    if (ok_type = 1) then
      if (cl_number < ok_number) then
        cl_number = ok_number;

    if (ok_type in (2, 3) and
        ok_sh_os <> 0) then
      has_sh_changes = 1;
  end

  if (gr_disb_line < cl_number) then
  begin
    for select kc_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :rep_k_id and
              kc_amount_approved > 0
        into :kc_id, kc_amount
    do
    begin
      tmp_amount = 0;
      select sum(ok_sh_os), max(ok_from_ol_id), max(ok_number)
      from kn_operkred
      where ok_from_kc_id = :kc_id and
            ok_ol_is_active = 1 and
            ok_type = 1
      into :tmp_amount, ol_id, ok_number;
      if (tmp_amount is null) then
        tmp_amount = 0;
      if (ok_number is null) then
        ok_number = 0;

      if (tmp_amount <> kc_amount and
          ok_number > 0) then
      begin
        update kn_operkred
        set ok_sh_os = ok_sh_os - (:tmp_amount - :kc_amount)
        where ok_from_ol_id = :ol_id and
              ok_number = :ok_number and
              ok_from_kc_id = :kc_id;
      end
    end
  end

  if ((has_sh_changes = 1 and
      gr_resh_sum = 0) or (need_close_loan = 1)) then -- имеется изменения в графике по ос
    execute procedure hp_kn_oksummary_fix_one(:rep_k_id, :rep_ol_id, :need_close_loan);

  kc_id = null;

--  При вставке сразу идет ok_is_actual = 1, поэтому код ниже закомментирован
--  Договор для проверки: 101010024356500101
--  Если вставлять по старому, в последнем погашении появляется отрицательный баланс

  for select kc_id
      from knclient
      where kc_from_k_id = :rep_k_id and
            kc_amount_approved > 0
      into :kc_id
  do
  begin
    update kn_operkred
    set ok_is_actual = 1
    where ok_from_kc_id = :kc_id and
          ok_from_ol_id = :rep_ol_id and
          ok_is_actual = 0;
  end

  if (need_close_loan = 1) then
  begin
    for select oks_from_kc_id, oks_date_begin
        from kn_oksummary
        where oks_from_k_id = :rep_k_id and
              oks_date_begin >= :ok_ol_date
        order by oks_from_kc_id, oks_date_begin
        into :kc_id, oks_date_begin
    do
    begin
      update kn_oksummary
      set oks_close_state = iif(oks_date_begin = :ok_ol_date, 1, 2)
      where oks_from_k_id = :kc_id and
            oks_date_begin = :oks_date_begin;
    end
  end
end
