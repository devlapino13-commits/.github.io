CREATE PROCEDURE MC_MAGAZINE_LIST
RETURNS (
    MAGAZINE_ID D_INTEGER,
    MAGAZINE_NAME D_STR_100,
    ADDRESS D_STR_255,
    LONGITUDE PERC_TYPE,
    LATITUDE PERC_TYPE,
    MAGAZINE_CITY_ID D_ID,
    WORK_HOURS D_STR_100,
    PHONE D_STR_100)
AS
begin
  for select ch.cs_id, ch.cs_name, coalesce(ch.cs_longitude, 0), coalesce(ch.cs_latitude, 0), adr.adl_address_str,
             pl.pl_id, ch.cs_hours, ch.cs_phone
      from clientshops ch
      inner join addrlist adr on adr.adl_id = ch.cs_from_adl_id
      inner join places pl on pl.pl_id = adr.adl_from_pl_id
      where ch.cs_is_delete = 0
      into magazine_id, magazine_name, longitude, latitude, address, magazine_city_id, work_hours, phone
  do
  begin
    suspend;
  end
end
