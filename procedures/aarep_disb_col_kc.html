CREATE PROCEDURE AAREP_DISB_COL_KC(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_KOLICH_ALL INTEGER,
    K_REGION_ID INTEGER,
    K_KC_ID DECIMAL(18,0),
    K_KOLICH_NEW INTEGER,
    K_KOLICH_REP INTEGER,
    K_AMOUNT DECIMAL(18,4))
AS
declare variable k_circle integer;
begin
  for select k_id, k_kolich_client_approved, k_amount_approved, ks_region_id, ks_kc_id, k_circle
      from kontract, kontstate
      where k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end
      into :k_id, :k_kolich_all, :k_amount, :k_region_id, :k_kc_id, :k_circle
  do
  begin
    k_kolich_new = 0;
    k_kolich_rep = 0;

    if (k_circle = 1) then
      k_kolich_new = :k_kolich_all;
    else
      k_kolich_rep = :k_kolich_all;
    /*
    if (k_amount is null) then k_amount=0;
    select sum(kc_circle) from knclient
    where kc_from_k_id=:k_id
    and kc_circle=1
    and kc_amount_approved is not null
    and kc_amount_approved>0
    into :K_KOLICH_NEW;
    if (k_kolich_new is null) then k_kolich_new=0;
    k_kolich_rep=K_KOLICH_ALL-k_kolich_new;
    */
    suspend;
  end
end
