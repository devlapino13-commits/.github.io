CREATE PROCEDURE HP_PVNOPERS_WRITEOFF(
    PO_ID D_INTEGER NOT NULL)
AS
declare variable po_amount d_currency;
declare variable po_status d_integer;
declare variable card_id d_integer;
declare variable po_oper_type d_smallint;
declare variable po_status_code d_str_code;
declare variable cca_code d_str_code;
declare variable cst_packet_no d_str_10;
declare variable cst_ticket_no d_integer;
declare variable cs_request_json d_str_8100;
declare variable comment_txt varchar(140);
declare variable cca_id d_integer;
declare variable cs_id d_integer;
declare variable po_office_id d_integer;
declare variable un_id d_id;
begin
    if (f_get_const('cardprjclose') = 1) then
      exit;
    select po.po_amount,
           po.po_status,
           po.po_card_id,
           po.po_oper_type,
           po.po_office_id,
           po.po_un_id
    from pvnopers po
    where po.po_id = :po_id
    into :po_amount,
         :po_status,
         :card_id,
         :po_oper_type,
         :po_office_id,
         :un_id;
    
    po_status_code = f_enum('pvnopers_status', :po_status, 'code');
    
    if (po_status_code = 'writeoff_check') then
        exception e_message 'Запрос на списание с карты уже отпрален, ожидается ответ';

    if (po_status_code in ('writeoff_check_ok', 'operation_ok')) then
        exception e_message 'Списание с карты ранее ранее уже было осущестлвено успешно';

    select caa.cca_code, caa.cca_id
    from cardlist cal
    inner join cardclientacc caa on caa.cca_id = cal.cal_from_cca_id
    where cal.cal_card_id = :card_id
    and cal_status = f_enum('cardlist_status', 'transfer_to_client', 'id')
    into :cca_code, :cca_id;

    select packet_no,
           ticket_no
    from cardsynctrn_create
    into :cst_packet_no,
         :cst_ticket_no;

    comment_txt = '';
    if (po_oper_type = 0) then
        comment_txt = 'Обналичивание с карты';
    else if (po_oper_type = 1) then
        comment_txt = 'Погашение кредита';

    -- payment_mode = 3 - моментальное выполнение с учетом ОТБ (баланса), т.е не может уходить в минус

    cs_request_json = f_format('
    {
        "card_acct": "%1",
        "card_id": '||:card_id||',
        "card_acct_ccy": "KGS",
        "tran_type": "207",
        "tran_ccy": "KGS",
        "tran_amount": %2,
        "tran_pocket": "%3",
        "tran_ticket": "%4",
        "tran_comment": "%5",
        "bank_c": "21",
        "groupc": "01",
        "payment_mode": 3,
        "tran_date_time": "' || f_datetimetostr(current_timestamp) || '"
    }',
    :cca_code,
    :po_amount,
    :cst_packet_no,
    :cst_ticket_no,
    :comment_txt
    );

    insert into cardsync(
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_amount,
      cs_request_json,
      cs_office_id,
      cs_from_un_id
    )
    values(
      f_get_cardsynccfg('pvn_write_off', 'id'),
      :card_id,
      :cca_id,
      f_objkind('pvnopers', 'id'),
      :po_id,
      round(:po_amount,2),
      :cs_request_json,
      :po_office_id,
      :un_id
    )
    returning cs_id
    into :cs_id;

    update pvnopers po
    set po.po_status = f_enum('pvnopers_status', 'writeoff_check', 'id'),
        po.po_card_writeoff_cs_id = :cs_id
    where po.po_id = :po_id;

    update cardsynctrn cst
    set cst.cst_from_cs_id = :cs_id
    where cst.cst_packet_no = :cst_packet_no and
          cst.cst_ticket_no = :cst_ticket_no;
end
