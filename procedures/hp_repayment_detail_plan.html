CREATE PROCEDURE HP_REPAYMENT_DETAIL_PLAN(
    K_ID D_ID,
    REP_DATE D_DATE,
    K_KONTRACT_NO D_STR_CODE = null)
RETURNS (
    OKS_DATE D_DATE,
    OKS_ORIG_OS D_CURRENCY,
    OKS_ORIG_PERC D_CURRENCY,
    OKS_ORIG_ALL D_CURRENCY,
    OKS_MOVED_OS D_CURRENCY,
    OKS_SH_OS D_CURRENCY,
    OKS_SH_PERC D_CURRENCY,
    OKS_SH_ALL D_CURRENCY,
    OKS_ZAD_OS D_CURRENCY,
    OKS_ZAD_PERC D_CURRENCY,
    OKS_INFO D_STR_CODE)
AS
declare variable ok_sh_id d_integer;
declare variable ol_id d_id;
declare variable k_stop_perc d_date;
declare variable oks_moved_date d_date;
declare variable oks_pmt_os d_currency;
declare variable oks_pmt_perc d_currency;
declare variable db_id d_integer;
begin

  db_id = f_get_db_id();
  if (k_id is null and
      k_kontract_no <> '') then
  begin
    select k_id
    from kontract
    where k_kontract_no = :k_kontract_no
    into k_id;
  end

  select k_stop_perc
  from kontract
  where k_id = :k_id
  into k_stop_perc;

  oks_pmt_os = 0;
  oks_pmt_perc = 0;

  if (db_id = 81001) then
  begin
    select -sum(oko.oko_os), -sum(oko.oko_perc)
    from operkroff oko
    inner join operlist ol on ol.ol_id = oko.oko_from_ol_id
    where oko.oko_from_k_id = :k_id and
          oko.oko_type = 4 and
          oko.oko_ol_is_active = 1 and
          oko.oko_ol_date <= :rep_date
    into oks_pmt_os, oks_pmt_perc;
  end

  -- изменено по RFCRU-5666
  for select s_date
  from hp_kontract_schedule_to_date(:k_id, :rep_date)
  union
  select k_stop_perc
  from kontract
  where k_id = :k_id
    and k_stop_perc is not null
  into oks_date
  do
  begin
    oks_orig_os = 0;
    oks_orig_perc = 0;

    select sh_os, sh_perc
    from schedule
    where sh_from_k_id = :k_id and
          sh_date = :oks_date
    into oks_orig_os, oks_orig_perc;

    oks_orig_all = oks_orig_os + oks_orig_perc;

    oks_sh_os = 0;
    oks_sh_perc = 0;

    select sum(ok_sh_os), sum(ok_sh_perc)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date and
          ok_sh_date = :oks_date and
          ok_type <= 3
    into oks_sh_os, oks_sh_perc;

    if (oks_sh_os is null) then
      oks_sh_os = 0;

    if (oks_sh_perc is null) then
      oks_sh_perc = 0;

    oks_sh_all = oks_sh_os + oks_sh_perc;

    oks_moved_os = 0;

    oks_moved_os = oks_sh_os - oks_orig_os;

    oks_zad_os = 0;
    oks_zad_perc = 0;

    select sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date and
          ok_sh_date = :oks_date and
          ok_type <= 4
    into oks_zad_os, oks_zad_perc;

    if (oks_zad_os is null) then
      oks_zad_os = 0;

    if (oks_zad_perc is null) then
      oks_zad_perc = 0;

    if (oks_zad_os > 0 and
        oks_pmt_os > 0) then
    begin
      if (oks_pmt_os > oks_zad_os) then
      begin
        oks_pmt_os = oks_pmt_os - oks_zad_os;
        oks_zad_os = 0;
      end
      else
      begin
        oks_zad_os = oks_zad_os - oks_pmt_os;
        oks_pmt_os = 0;
      end
    end

    if (oks_zad_perc > 0 and
        oks_pmt_perc > 0) then
    begin
      if (oks_pmt_perc > oks_zad_perc) then
      begin
        oks_pmt_perc = oks_pmt_perc - oks_zad_perc;
        oks_zad_perc = 0;
      end
      else
      begin
        oks_zad_perc = oks_zad_perc - oks_pmt_perc;
        oks_pmt_perc = 0;
      end
    end

    oks_info = '';
    oks_moved_date = null;
    ok_sh_id = null;
    ol_id = null;

    select ok_sh_id, ok_from_ol_id
    from operkred, operlist
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_sh_date = :oks_date and
          ok_from_ol_id = ol_id and
          ol_oper_type = 1 and
          ol_oper_subtype = 9 and
          (ok_sh_os < 0 or
          ok_sh_perc < 0)
    rows 1
    into ok_sh_id, ol_id;

    if (ok_sh_id is not null) then
    begin
      select ok_sh_date
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_sh_id = :ok_sh_id and
            ok_from_ol_id = :ol_id and
            ok_sh_date > :oks_date
      into oks_moved_date;

      if (oks_moved_date is not null) then
        oks_info = f_datetostr(oks_moved_date);
    end

    if (k_stop_perc = oks_date) then
    begin
      if (oks_info <> '') then
        oks_info = oks_info || ', договор расторгнут';
      else
        oks_info = 'Договор расторгнут';
    end

    -- убираем пустые строки
    if (oks_orig_os = 0
      and oks_orig_perc = 0
      and oks_moved_os = 0
      and oks_sh_os = 0
      and oks_sh_perc = 0
      and oks_zad_os = 0
      and oks_zad_perc = 0
      and oks_info = '')
    then
      continue;

    suspend;
  end
end
