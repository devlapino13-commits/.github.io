CREATE PROCEDURE HP_TERMINATION_NEW_GRAPH(
    K_ID D_ID,
    ADD_PARAMS D_STR_8100)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable DATE_STOP D_DATE;
declare variable SH_ID D_SMALLINT;
declare variable SH_DATE D_DATE;
declare variable SH_OS D_CURRENCY;
declare variable SH_PERC D_CURRENCY;
declare variable S_PERC D_CURRENCY;
declare variable NEW_OL_ID D_ID;
declare variable BAL D_CURRENCY;
declare variable S_PERC_ALL D_CURRENCY;
begin
  date_stop = f_json_str_get_value(add_params, 'date_stop', null);
  --date_stop = current_date; -----------------------------------------------------------Нужно убрать

  res_text = '';
  res_is_error = 0;
  if (date_stop is null) then
  begin
    res_text = 'Выберите дату.';
    res_is_error = 1;
    suspend;
    exit;
  end 

  delete from temp_operkred3
  where ok_from_k_id = :k_id;

  insert into temp_operkred3 (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                        ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                        ok_ol_is_active)
  select ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine, ok_pay_type,
         ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active
  from operkred
  where ok_from_k_id = :k_id
  and ok_ol_is_active = 1;

  if (exists(select * from temp_operkred3
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_ol_date > :date_stop)) then
  begin
    res_text = 'По договору имеются записи более поздней датой';
    res_is_error = 1;
    suspend;
    exit;
  end
  if (exists(select * from kontract
  where k_id =:k_id and
  (k_kredline_end < :date_stop or
  k_stop_perc is not null))) then
  begin
    res_text = 'По текущему контракту уже было расторжение.';
    res_is_error = 1;
    suspend;
    exit;
  end 

  new_ol_id = 1;
  bal = 0;
  s_perc_all = 0;
  for select sh_id, sh_date, zad_os, zad_perc, coalesce(sh_calc_perc, 0)
      from aarep_make_zad_list(:k_id, :date_stop, 1)
      where (zad_perc <> 0 or
            zad_os <> 0 or
            coalesce(sh_calc_perc, 0) <> 0) and
            sh_date > :date_stop
      into sh_id, sh_date, sh_os, sh_perc, s_perc
  do
  begin
    insert into temp_operkred3(ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                          ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_is_active)
    values (:new_ol_id, -1, :k_id, :sh_id, :sh_date, 3, 0, 0, 0, 3, 0, -:sh_os, -:sh_perc, 0, 0, :date_stop, 1);
    bal = bal + sh_os;
    s_perc_all = s_perc_all + s_perc;
  end
  insert into temp_operkred3 (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                          ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_is_active)
  values (:new_ol_id, -1, :k_id, :sh_id, :date_stop, 3, 0, 0, 0, 3, 0, :bal, :s_perc_all, 0, 0, :date_stop, 1);

  delete from temp_sh
  where 1=1;
  insert into temp_sh (sh_id, sh_date, sh_new_os, sh_new_perc)
  select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
  from temp_operkred3
  where ok_type < 7
  and ok_ol_is_active = 1
  and ok_from_k_id = :k_id
  group by 1, 2
  having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc)  <> 0;

  suspend;
end
