CREATE PROCEDURE FM_SURVEY_ACTION(
    UN_ID DECIMAL(18,0),
    ACTION_KIND VARCHAR(255),
    PARAMS BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
RETURNS (
    RES_ID INTEGER,
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable sva_id integer;
declare variable sva_caption varchar(255);
declare variable sva_memo varchar(255);
declare variable sva_codes varchar(255);
declare variable svq_id integer;
declare variable svq_caption varchar(255);
declare variable svq_memo varchar(255);
declare variable svq_time_start_str varchar(20);
declare variable svq_time_finish_str varchar(20);
declare variable state varchar(20);
declare variable result_count integer;
declare variable json_tmp blob sub_type 1 segment size 80;
declare variable json_key varchar(255);
declare variable json_value varchar(255);
declare variable set_delta integer;
declare variable set_time timestamp;
declare variable time_action_kind varchar(20);
declare variable v_delimiter d_str_10;
begin
  res_id = 1;
  res_text = '';

  for select json_key, json_value
      from fm_json_parse(:params)
      into :json_key, :json_value
  do
  begin
-- survey_update, survey_remove, survey_answer_list, survey_answer_update
-- survey_time_edit
    if (json_key = 'svq_id') then
      svq_id = json_value;

-- survey_list
    if (json_key = 'state') then
      state = json_value;

-- survey_update
    if (json_key = 'svq_caption') then
      svq_caption = json_value;
    if (json_key = 'svq_memo') then
      svq_memo = json_value;

-- survey_answer_add
    if (json_key = 'svq_id') then
      svq_id = json_value;
    if (json_key = 'sva_codes') then
      sva_codes = json_value;
    if (json_key = 'sva_caption') then
      sva_caption = json_value;
    if (json_key = 'sva_memo') then
      sva_memo = json_value;

-- survey_answer_remove
    if (json_key = 'sva_id') then
      sva_id = json_value;

-- survey_time_edit
    if (json_key = 'time_action_kind') then
      time_action_kind = json_value;
    if (json_key = 'set_delta') then
      set_delta = json_value;
    if (json_key = 'set_time') then
      set_time = json_value;
  end

  if (action_kind = 'survey_list') then
  begin
    json_tmp = '';
    v_delimiter = '';
    for select svq_id, svq_caption, svq_memo, result_count, svq_time_start_str, svq_time_finish_str
        from hp_survey_ques_list(0, '', :state)
        into svq_id, svq_caption, svq_memo, result_count, svq_time_start_str, svq_time_finish_str
    do
    begin
      json_tmp = blob_append(json_tmp, v_delimiter || '{' || '"svq_id":"' || svq_id || '", ' || '"svq_caption":"' || coalesce(svq_caption, '') || '",' || '"svq_memo":"' || coalesce(svq_memo, '') || '",' || '"result_count":"' || result_count || '",' || '"svq_time_start":"' || svq_time_start_str || '",' || '"svq_time_finish":"' || svq_time_finish_str || '"' || '}');
      v_delimiter = ',';
    end
    res_text = '{"list":[' || json_tmp || ']}';
    suspend;
    exit;
  end
  else
  if (action_kind = 'survey_update') then
  begin
    if (svq_id = 0) then
      svq_id = null;
    update or insert into survey_ques (svq_id, svq_caption, svq_memo)
    values (:svq_id, :svq_caption, :svq_memo)
    returning svq_id
    into :svq_id;
    res_text = '{"svq_id":"' || svq_id || '"}';
    suspend;
    exit;
  end
  else
  if (action_kind = 'survey_answer_list') then
  begin
    json_tmp = '';
    v_delimiter = '';
    for select sva_id, sva_caption, sva_memo, sva_codes
        from survey_answer
        where sva_from_svq_id = :svq_id
        into sva_id, sva_caption, sva_memo, sva_codes
    do
    begin
      sva_codes = substring(sva_codes from 2 for char_length(sva_codes) - 2);
      sva_codes = replace(sva_codes, '~', ' / ');
      json_tmp = blob_append(json_tmp, v_delimiter || '{' || '"sva_id":"' || sva_id || '", ' || '"sva_caption":"' || coalesce(sva_caption, '') || '",' || '"sva_memo":"' || coalesce(sva_memo, '') || '",' || '"sva_codes":"' || coalesce(sva_codes, '') || '"' || '}');
      v_delimiter = ',';
    end
    res_text = '{"list":[' || json_tmp || ']}';
    suspend;
    exit;
  end
  else
  if (action_kind = 'survey_answer_update') then
  begin
    if (sva_id = 0) then
      sva_id = null;
    update or insert into survey_answer (sva_id, sva_from_svq_id, sva_codes, sva_caption, sva_memo)
    values (:sva_id, :svq_id, :sva_codes, :sva_caption, :sva_memo)
    returning sva_id, sva_codes
    into :sva_id, :sva_codes;
    sva_codes = substring(sva_codes from 2 for char_length(sva_codes) - 2);
    sva_codes = replace(sva_codes, '~', ' / ');
    res_text = '{"sva_id":"' || sva_id || '",' || '"sva_codes":"' || sva_codes || '"}';
    suspend;
    exit;
  end
  else
  if (action_kind = 'survey_answer_remove') then
  begin
    delete from survey_answer
    where sva_id = :sva_id;
    res_text = '';
    suspend;
    exit;
  end
  else
  if (action_kind = 'survey_remove') then
  begin
    delete from survey_ques
    where svq_id = :svq_id;
    res_text = '';
    suspend;
    exit;
  end
  else
  if (action_kind = 'survey_time_edit') then
  begin
    select res_id, res_text
    from hp_survey_time_edit(:svq_id, :time_action_kind, :set_delta, :set_time)
    into :res_id, :res_text;
    suspend;
    exit;
  end
-- Неверная команда
  res_id = 0;
  res_text = 'Неверная команда';
  suspend;
  exit;
end
