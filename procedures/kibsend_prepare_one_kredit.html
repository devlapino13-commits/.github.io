CREATE PROCEDURE KIBSEND_PREPARE_ONE_KREDIT(
    KONT_ID DECIMAL(18,0))
RETURNS (
    BANK_CODE INTEGER,
    FIL_CODE INTEGER,
    UNIQ_CRED_NUM DECIMAL(18,0),
    CREDTYPE_CODE VARCHAR(20),
    CREDSTATEX_CODE VARCHAR(20),
    UNIQ_K_ID DECIMAL(18,0),
    POCKET_ID INTEGER,
    REPAYSTATUS INTEGER,
    GUARANTEE VARCHAR(100),
    COUNTOFDELINQUENCY INTEGER,
    UNPAIDAMOUNT DECIMAL(18,4),
    UNPAIDPERCENTAMOUNT INTEGER,
    STAT INTEGER)
AS
declare variable pr_kind integer;
declare variable dd_start date;
declare variable max_day integer;
declare variable last_date date;
declare variable date_close date;
declare variable kol integer;
declare variable dd date;
begin
  repaystatus = -1;
  countofdelinquency = 0;
  guarantee = '';
  unpaidpercentamount = 0;
  unpaidamount = 0;
  stat = 0;/* Договора в процессе тип - 0 */
  credstatex_code = '1-0';/* Рассмотрение заявки */

  pocket_id = gen_id(gen_kibpocket_id, 1);
  insert into kibpocket (kp_id, kp_status, kp_date, kp_memo, kp_create_time, kp_check_status)
  values (:pocket_id, :stat, cast('NOW' as date), 'Договора в процессе', cast('NOW' as timestamp), 1);

  for select distinct cm_kib_bank_code, reg_kib_code, k_id, pr_loan_type
      from kontract, knaktlist, region, company, product
      where k_from_kf_id = kal_from_kf_id and
            kal_date <= k_disb_date and
            kal_date_end >= k_disb_date and
            kal_new_reg_id = reg_unicode and
            k_id = :kont_id and
            k_from_pr_id = pr_id and
            pr_send_to_kib = 1 and
            pr_is_employer <> 1 and
            k_amount_approved <> 0
      order by k_id
      into :bank_code, :fil_code, :uniq_cred_num, :pr_kind
  do
  begin
    uniq_k_id = :uniq_cred_num;
    if (pr_kind = 1) then
      credtype_code = '7';
    if (pr_kind = 0) then
      credtype_code = '3';
    suspend;
  end
end
