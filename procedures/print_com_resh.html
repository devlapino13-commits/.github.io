CREATE PROCEDURE PRINT_COM_RESH(
    CML_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    KC_PDN_KOEF CURR_TYPE,
    QR_CODE D_STR_LONG,
    C_ID D_ID,
    K_ID D_ID,
    EDL_BLANK_DATE D_DATE,
    EDOC_DATE_CREATE D_DATE,
    K_KONTRACT_DATE D_DATE)
AS
declare variable edl_id d_id;
declare variable kcp_id d_id;
declare variable db_id d_integer;
begin
  -- Печать информации по Договору/клиенту в КК (com_list)
  db_id = f_get_db_id();
  select cml_object_id, cml_from_kcp_id
  from com_list
  where cml_id = :cml_id
  into k_id, kcp_id;

  select c_id, k_disb_date, k_kontract_no, c_name_fio, k_kontract_date
  from kontract, knclient, clients
  where k_id = :k_id
    and kc_from_k_id = k_id
    and kc_amount_approved > 0
    and kc_from_c_id = c_id
  into c_id, k_disb_date, k_kontract_no, c_name_fio, k_kontract_date;

  -- QR-код электронной подписи клиента. Текст пустой если не подписан
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);
    qr_code = f_edoc_qr_by_edl(edl_id, c_id);
    select edl_blank_date
    from edoc_list
    where edl_id = :edl_id
    into edl_blank_date;
  end

  select cast(edl_create_datetime as date)
  from edoc_list
  inner join edoc_cfg on edc_id = edl_from_edc_id
  where edl_object_kind = 9 and
        edl_object_id = :k_id and
        edc_code = 'loan.agreement' and
        edl_status <> -1
  order by edl_create_datetime desc rows 1
  into edoc_date_create;

  kc_pdn_koef = 0;
  if (db_id = 81001) then
  begin
    select coalesce(round(kc_pdn_koef, 2), 0)
    from knclient_pdn
    where kcp_id = :kcp_id
    into kc_pdn_koef;
    if (kc_pdn_koef = 0) then
      exception e_default 'Не рассчитан ПДН по кредитному коммитету. cml_id' || coalesce(cml_id, 'Null');
  end
  suspend;
end
