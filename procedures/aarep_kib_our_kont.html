CREATE PROCEDURE AAREP_KIB_OUR_KONT(
    REP_DATE_END DATE)
RETURNS (
    KS_FROM_K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_INDEX_CURR_CODE VARCHAR(10),
    K_INDEX_CURR_RATE DECIMAL(18,4),
    K_KONT_BALANCE DECIMAL(18,4),
    K_DISB_DATE DATE,
    K_PERIOD DECIMAL(15,2),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    PR_NAME VARCHAR(50),
    K_CLIENTS_CASHIER_STR VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    PR_LOAN_TYPE_STR VARCHAR(10),
    LK_STATUS SMALLINT,
    LK_PERCENT DECIMAL(18,7),
    LK_DATE DATE)
AS
declare variable cr_from_cu_id varchar(3);
declare variable cr_rate decimal(18,4);
declare variable pr_loan_type smallint;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select cr_from_cu_id, cr_rate
  from opercurr
  where :rep_date_end between cr_date and cr_date_end and
        cr_from_cu_id = '840'
  into :cr_from_cu_id, :cr_rate;

  for select distinct ks_from_k_id, k_kontract_no, k_amount_approved, k_index_curr_code, k_index_curr_rate, k_disb_date,
                      k_period, k_kolich_client_approved, k_circle, k_circle_internal, pr_name, pr_loan_type,
                      c_name_fio, c_pass_no

      from kontstate, knclient, kibminiapp, kibhistory, kontract, product, clients
      where :rep_date_end between ks_date_begin and ks_date_end and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            kc_from_k_id = ks_from_k_id and
            kc_amount_approved <> 0 and
            kbm_from_c_id = kc_from_c_id and
            kh_from_kbm_id = kbm_id and
            kh_is_our = 0 and
            k_id = ks_from_k_id and
            k_from_pr_id = pr_id and
            k_clients_cashier_id = c_id

      into :ks_from_k_id, :k_kontract_no, :k_amount_approved, :k_index_curr_code, :k_index_curr_rate, :k_disb_date,
           :k_period, :k_kolich_client_approved, :k_circle, :k_circle_internal, :pr_name, :pr_loan_type,
           :k_clients_cashier_str, :c_pass_no
  do
  begin
    k_kont_balance = 0;
    select oks_balance
    from oksummary
    where oks_from_k_id = :ks_from_k_id and
          oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_close_state = 0
    into :k_kont_balance;
    if (k_kont_balance < 0) then
      k_kont_balance = 0;

    if (:k_index_curr_code = :cr_from_cu_id) then
    begin
      if (:k_index_curr_rate > :cr_rate) then
        k_kont_balance = :k_kont_balance * :k_index_curr_rate;
      else
        k_kont_balance = :k_kont_balance * :cr_rate;
    end

    if (:pr_loan_type = 0) then
      pr_loan_type_str = 'Групп.';
    else
      pr_loan_type_str = 'Индив.';

    lk_status = null;
    lk_percent = null;

    select lk_status, lk_percent, lk_date
    from llc_kont
    where lk_from_k_id = :ks_from_k_id and
          :rep_date_end between lk_date and lk_date_end
    order by lk_date desc
    rows 1
    into :lk_status, :lk_percent, lk_date;

    suspend;
  end
end
