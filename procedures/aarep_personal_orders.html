AS
declare variable sawt_id d_integer;
declare variable ps_is_on_staff smallint;
declare variable ps_is_working smallint;
declare variable jl_is_front integer;
declare variable c_sex smallint;
declare variable c_is_resident smallint;
declare variable st float;
declare variable hrc_is_forced integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable db_id integer;
declare variable reg_hr_region_id d_integer;
declare variable jk_system_code d_str_100;
declare variable jk_id d_integer;
declare variable ho_id id_type;
begin
  /* CHANGED BY IGUSEV 01.02.2013
     в where добавлено условие   and ((coalesce(HO_STATUS,0) > -1) or (HO_STATUS = -1  and HO_STATUS_DATETIME>:REPORT_DATE_END) )
  */
  db_id = f_get_db_id();

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  for select ps_date_begin, ps_date_end, ps_period_month, ps_date_end_in_order, ps_is_on_staff, ps_is_working,
             ps_salary, ps_transfer_office, ps_tabel_no, ho_date, ho_from_reg_id, ho_text, ho_reason, ho_save_time,
             ho_order_no, jl_name, jl_order_no, dep_name, dep_order_no, r1.reg_name, r1.reg_code || ' ' || r1.reg_name,
             r2.reg_name, jl_is_front, jl_category, c_id, c_name_fio, c_sex, c_birth_date, d1.rd_name, d2.rd_name,
             c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire,
             (select cont_value
              from hp_contdata_get_email_main(0, c_id)), d3.rd_name, d4.rd_name, c_kol_family, c_kol_child, c_memo,
             c_reg_no, c_okpo, c_inn, c_account, c_org_from_b_id, c_hired_date, c_is_resident, d5.rd_name,
             (select cont_value
              from hp_contdata_get_cell_main(0, c_id)), c_others, adl_address_str, sawt_name, hrc_name, hrc_is_forced,
             jk_name, jk_system_code, r3.reg_code || ' ' || r3.reg_name, c_pincode, c_id, r1.reg_hr_region, sawt_id,
             jk_id, ho_id, d2.rd_str_id, jl_gni_code
      from hr_pstatus, hr_orders, region_list(:report_region_id, 1, -1, 1), joblist, depart, region r1, region r2, clients, region r3, refdata d1, refdata d2, refdata d3, refdata d4, refdata d5, sa_worktype, hr_class, jobkind, addrlist, addrclient
      where ho_id = ps_from_ho_id and
            (ps_from_reg_id = id_list) and
            iif(:db_id <> 81001, iif(:db_id in (39801, 41701) and
            ho_order_type = 6, ps_date_begin - 1 >= :report_date_begin, ps_date_begin >= :report_date_begin), ho_date >= :report_date_begin) and
            iif(:db_id <> 81001, iif(:db_id in (39801, 41701) and
            ho_order_type = 6, ps_date_begin - 1 <= :report_date_end, ps_date_begin <= :report_date_end), ho_date <= :report_date_end) and
            ps_from_reg_id = r1.reg_unicode and
            ps_from_dep_id = dep_id and
            ps_from_jl_id = jl_id and
            ho_from_reg_id = r2.reg_unicode and
            ps_class = hrc_id and
            ((:report_list_kind = -1) or
            ((:report_list_kind = 0) and
            (ho_order_type = 6)) or
            ((:report_list_kind = 1) and
            (ho_order_type = 1)))

            and
            ps_from_c_id = c_id and
            c_nation = d1.rd_id and
            c_citizen = d2.rd_id and
            c_family = d3.rd_id and
            c_education = d4.rd_id and
            c_army = d5.rd_id and
            ho_order_type = sawt_id and
            jl_kind = jk_id and
            adc_address_type = 1 and
            adc_from_c_id = c_id and
            adc_from_adl_id = adl_id and
            r1.parent = r3.reg_unicode

            and
            ((coalesce(ho_status, 0) > -1) or
            ((ho_status = -1 and
            ho_status_datetime > :report_date_end) and
            (:db_id <> 81001))) -- igusev 09.10.2018 добавил and (db_id <> 81001) т.к. условия "or (HO_STATUS = -1  and HO_STATUS_DATETIME>:REPORT_DATE_END)" не должно быть для РФ. Пистмо Ларисы "RE: Не корректно формируется отчет" от 09.10.2018
      order by jl_order_no, c_name_fio
      into :ps_date_begin, :ps_date_end, :ps_period_month, :ps_date_end_in_order, :ps_is_on_staff, :ps_is_working,
           :ps_salary, :ps_transfer_office, :ps_tabel_no, :ho_date, :ho_from_reg_id, :ho_text, :ho_reason,
           :ho_save_time, :ho_order_no, :jl_name, :jl_order_no, :dep_name, :dep_order_no, :reg_name, :reg_code_name,
           :ho_reg_name, :jl_is_front, :jl_category, :c_id, :c_name_fio, :c_sex, :c_birth_date, :c_nation, :c_citizen,
           :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_email, :c_family, :c_education, :c_kol_family,
           :c_kol_child, :c_memo, :c_reg_no, :c_okpo, :c_inn, :c_account, :c_org_from_b_id, :c_hired_date,
           :c_is_resident, :c_army, :c_plastcard_no, :c_others, :c_full_address, :ho_order_kind, :hrc_name,
           :hrc_is_forced, :jk_name, :jk_system_code, :ps_branch, c_pincode, c_id, :reg_hr_region_id, :sawt_id, jk_id,
           :ho_id, :c_citizen_code, :jl_gni_code
  do
  begin

    if (db_id in (39801, 41701) and
        report_list_kind = 0) then
    begin
      ps_date_end_in_order = ps_date_begin - 1; -- по просьбе Чинары Мамрасуловой BOARD-5142
    end

    reg_hr_region = '';

    if (reg_hr_region_id is not null) then
    begin
      select rd.rd_name
      from refdata rd
      where rd.rd_id = :reg_hr_region_id
      into :reg_hr_region;
    end

    if (ps_is_on_staff = 0) then
      ps_is_on_staff_text = 'Вне штатный';
    else
      ps_is_on_staff_text = 'Штатный';

    if (ps_is_working = 1) then
      ps_is_working_text = 'Работает';
    else
    begin
      if (ps_transfer_office is null) then
        ps_is_working_text = 'Уволен';
      else
        ps_is_working_text = 'Переведен';
    end

    if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.Region_KM', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
      jl_is_kred_person = 'Кредит.персонал';
    else
      jl_is_kred_person = 'Админ.персонал';

    if (jl_is_front = 1) then
      jl_is_front_text = 'Фронт офис';
    else
      jl_is_front_text = 'Бэк офис';

    if (c_sex = 1) then
      c_sex_text = 'Мужчина';
    else
      c_sex_text = 'Женщина';

    if (c_is_resident = 1) then
      c_is_resident_text = 'Резидент';
    else
      c_is_resident_text = 'Не резидент';
      
    if (c_birth_date is not null) then
    begin
      c_age = cast(datediff(month, c_birth_date, ps_date_end_in_order) / 12 as integer);
    end
    else
      c_age = 0;

    if (db_id in (94401, 81001)) then
    begin
      if (c_hired_date is not null) then
        c_stage = datediff(month, c_hired_date, report_date_end);
      else
        c_stage = 0;
      st = c_stage / 12;
      c_stage_year = cast(st as integer);
      if (c_stage_year > (c_stage / 12)) then
        c_stage_year = c_stage_year - 1;
    end

    jl_kred_kind = '';
    if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner')) then
      jl_kred_kind = 'Кредит.сотрудник';

    if (hrc_is_forced = 0) then
      hrc_is_force_txt = 'По желанию работника';
    else
      hrc_is_force_txt = 'По желанию работодателя';

    if (db_id in (39801, 41701) and
        :report_list_kind = 0) then ----Стаж в месяцах, годах.Неправильно считалось для повторных  уволенных сотрудников.
    begin
      select css.c_hired_date, css.c_stage, css.c_stage_year
      from aarep_work_experience(:ps_date_begin, :c_id) css
      into w_start_date, w_experience_month, w_experience_year;
    end

    if (db_id in (39801, 41701) and
        :report_list_kind = 1) then ------ Дата найма с таблицы clients поле c_hired_date тянется у повторных нанятых сотрудников тянется неправильно.
    begin ------Поэтому вытягиваю с таблицы hr_pstatus.
      select ps.ps_date_begin
      from hr_orders ho
      inner join hr_pstatus ps on ps.ps_from_ho_id = ho.ho_id
      where ps.ps_from_c_id = :c_id and
            ho.ho_order_type = 1 and
            ps.ps_is_working = 1 and
            ps.ps_status = 1 and
            ps.ps_date_begin between :report_date_begin and :report_date_end rows 1
      into date_start;

      select count(ho.ho_order_type)
      from hr_orders ho
      inner join hr_pstatus ps on ps.ps_from_ho_id = ho.ho_id
      where ps.ps_from_c_id = :c_id and
            ho.ho_order_type = 1 and
            ho.ho_status = 1
      into :hiring;
    end

    --Dobavil po prosbe Svetlany 10.03.16 mrahmanov
    date_hired_last = null;
    select ps_date_begin
    from hr_pstatus, hr_orders
    where ps_status = 1 and
          ps_from_c_id = :c_id and
          ps_from_ho_id = ho_id and
          ho_order_type = 1
    order by ps_date_begin desc
    rows 1
    into date_hired_last;

    notice_date = null;
    notice_period = null;
    select cd.cds_doc_date_begin, cd.cds_doc_date_end
    from clientdocs cd
    where cd.cds_from_c_id = :c_id and
          cd.cds_is_delete = 0 and
          cd.cds_from_pk_id = iif(:report_list_kind = 1, 89, 90) and
          cd.cds_doc_date_begin >= iif(:report_list_kind = 1, :date_hired_last, :ho_date)
    order by cd.cds_datetime desc
    rows 1
    into notice_date, notice_period;

    jk_perevod = '0';
    if ((:jk_system_code is not null) and
        (:sawt_id in (2, 3))) then
      select
             case
               when :jk_id in (5, 27) and (jk_id not in (5, 27) or jk_id = 0) then 'перевод на СВЗ'
               when (:jk_id not in (5, 27) or :jk_id = 0) and jk_id in (5, 27) and sawt_id <> 2 then 'перевод с СВЗ'
               --  when :jk_system_code = 'JobSystem.LO_Standart' and jk_system_code = 'JobSystem.LO_Standart' and sawt_id = 2 then 'перевод с СВЗ'
               when :jk_id not in (5, 27) and jk_id not in (5, 27) and :jk_id <> jk_id then 'перевод на иную должность'
               else '0'
             end
      from hr_pstatus, hr_orders, joblist, jobkind, sa_worktype
      where ho_id = ps_from_ho_id and
            iif(:db_id <> 81001, ps_date_begin >= :report_date_begin, ho_date >= :report_date_begin) and
            iif(:db_id <> 81001, ps_date_begin <= :report_date_end, ho_date <= :report_date_end) and
            --ho_save_time < :ho_save_time and
            iif(ps_date_begin = :ps_date_begin, ho_save_time < :ho_save_time, ps_date_begin <= :ps_date_begin) and
            ho_id <> :ho_id and
            ps_from_jl_id = jl_id and
            ((:report_list_kind = -1) or
            ((:report_list_kind = 0) and
            (ho_order_type = 6)) or
            ((:report_list_kind = 1) and
            (ho_order_type = 1))) and
            ((coalesce(ho_status, 0) > -1) or
            ((ho_status = -1 and
            ho_status_datetime > :report_date_end) and
            (:db_id <> 81001))) and

            ps_from_c_id = :c_id and
            jl_kind = jk_id and
            ho_order_type = sawt_id --and

      order by ps_date_begin desc, ho_save_time desc
      rows 1
      into :jk_perevod;

    select rd.rd_str_id
    from client_gni_category cgc
    inner join refdata rd on rd.rd_id = cgc.cgc_from_category
    where cgc.cgc_from_c_id = :c_id and
          cgc.cgc_is_delete = 0 and
          cgc.cgc_date_end = '31.12.2999'
    into :cgc_from_category;

    suspend;
  end
end
