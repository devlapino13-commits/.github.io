CREATE PROCEDURE AAREP_PERSONAL_STATISTIC(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    REG_CODE_NAME VARCHAR(50),
    JL_CATEGORY VARCHAR(50),
    JL_IS_FRONT_TEXT VARCHAR(50),
    C_SEX_TEXT VARCHAR(50),
    JL_IS_KRED_PERSON VARCHAR(50),
    KOL_ROTATION_MINUS INTEGER,
    KOL_ROTATION_PLUS INTEGER,
    KOL_GONE INTEGER,
    KOL_NEW INTEGER,
    KOL_BEGIN INTEGER,
    KOL_END INTEGER)
AS
begin
  for select reg_code_name, jl_category, jl_is_front_text, c_sex_text, jl_is_kred_person, sum(kol_end),
             sum(kol_rotation_plus), sum(kol_gone), sum(kol_new), sum(kol_begin)
      from aarep_personal_stat_temp(:report_region_id, :report_date_begin, :report_date_end)
      group by reg_code_name, jl_category, jl_is_front_text, c_sex_text, jl_is_kred_person
      order by reg_code_name, jl_category
      into :reg_code_name, :jl_category, :jl_is_front_text, :c_sex_text, :jl_is_kred_person, :kol_end,
           :kol_rotation_plus, :kol_gone, :kol_new, :kol_begin
  do
  begin
    kol_rotation_minus = kol_begin + kol_new + kol_rotation_plus - kol_gone - kol_end;
    suspend;
  end
end
