CREATE PROCEDURE AAREP_REQUEST_PRINT(
    REPORT_RM_ID DECIMAL(18,0))
RETURNS (
    RM_ID DECIMAL(18,0),
    RM_DATE DATE,
    RM_FROM_C_ID DECIMAL(18,0),
    RM_FROM_REG_ID INTEGER,
    RM_FROM_RK_ID INTEGER,
    RM_NO INTEGER,
    RM_C_NAME_FIO VARCHAR(100),
    RM_STATUS_TEXT VARCHAR(20),
    RM_RK_TEXT VARCHAR(255),
    RM_REGION_NAME VARCHAR(100),
    RL_NUMBER INTEGER,
    RL_TEXT VARCHAR(255),
    RL_DATE DATE,
    RL_KOLICH DECIMAL(18,4),
    RL_SUMMA_MINUS DECIMAL(18,4),
    RL_CORR_OFFICE VARCHAR(100),
    RL_YEDIZ VARCHAR(20),
    RL_CURRENCY VARCHAR(3))
AS
declare variable rm_status integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  rm_status = -2;
  select distinct rs_status
  from reqsign
  where rs_from_rm_id = :report_rm_id
  order by rs_number desc
  rows 1
  into :rm_status;
  if (rm_status is null) then
    rm_status = -2;
  if (rm_status = -2) then
    rm_status = 0;
  if (rm_status > -2) then
  begin
    if (rm_status = -1) then
      rm_status_text = 'Аннулирован';
    if (rm_status = 0) then
      rm_status_text = 'В процессе';
    if (rm_status = 1) then
      rm_status_text = 'Утвержден';
    if (rm_status = 2) then
      rm_status_text = 'Отказано';
    if (rm_status = 3) then
      rm_status_text = 'Исполнен';
    if (rm_status = 4) then
      rm_status_text = 'Проверен';
    if (rm_status = 5) then
      rm_status_text = 'Согласован';
  end
  for select distinct rm_id, rm_date, rm_from_c_id, rm_from_reg_id, rm_from_rk_id, rm_no, c_name_fio, rk_text,
                      r1.reg_name, rl_number, rl_text, rl_date, rl_kolich, rl_summa_minus - rl_summa_plus, r2.reg_name,
                      yl_code, rl_from_cu_id
      from reqmain, clients, reqlist, reqsign, reqkind, region r1, region r2, yedizlist
      where rm_from_c_id = c_id and
            rm_id = rl_from_rm_id and
            rm_from_reg_id = r1.reg_unicode and
            rl_corr_office = r2.reg_unicode and
            rm_from_rk_id = rk_id and
            rm_id = :report_rm_id and
            yl_id = rl_from_yl_code and
            rs_from_rm_id = rm_id
      into :rm_id, :rm_date, :rm_from_c_id, :rm_from_reg_id, :rm_from_rk_id, :rm_no, :rm_c_name_fio, :rm_rk_text,
           :rm_region_name, :rl_number, :rl_text, :rl_date, :rl_kolich, :rl_summa_minus, :rl_corr_office, :rl_yediz,
           :rl_currency
  do
  begin
    suspend;
  end
end
