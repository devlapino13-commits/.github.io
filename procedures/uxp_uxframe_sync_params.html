CREATE PROCEDURE UXP$UXFRAME_SYNC_PARAMS(
    UFR_UUID D_UUID,
    SQL_KIND D_SMALLINT,
    UFR_UUID_CODE D_STR_CODE = null)
AS
declare variable us_uuid d_uuid;
declare variable select_us_uuid d_uuid;
declare variable uf_uuid d_uuid;
declare variable uf_is_param d_bool;
declare variable uf_pk_field d_smallint;
declare variable us_is_table d_bool;
declare variable uf_not_null d_bool;
declare variable uf_default d_str_100;
declare variable uf_field_name d_str_code;
declare variable ufp_uf_uuid d_uuid;
declare variable ufp_uuid d_uuid;
declare variable main_us_uuid d_uuid;
declare variable uf_readonly d_boolean;
begin
  if (uxf$syncmode() = 1) then
    exit;

  if (coalesce(ufr_uuid, '') = '') then
    ufr_uuid = uxf$uniq(ufr_uuid_code);

  if (sql_kind = -1) then
  begin
    while (sql_kind < 3) do
    begin
      sql_kind = sql_kind + 1;
      execute procedure uxp$uxframe_sync_params(:ufr_uuid, :sql_kind);
    end

    exit;
  end

  select ufr_us_uuid
  from ux$frame
  where ufr_uuid = :ufr_uuid
  into select_us_uuid;

  if (sql_kind <> 0) then
  begin
    select ufd_us_uuid
    from ux$frdata
    where ufd_ufr_uuid = :ufr_uuid and
          ufd_kind = :sql_kind and
          ufd_us_uuid is distinct from :select_us_uuid
    into :us_uuid;

    if (us_uuid is null) then
    begin
      delete from ux$framesql
      where ufs_ufr_uuid = :ufr_uuid and
            ufs_sql_kind = :sql_kind;

      exit;
    end
  end

  main_us_uuid = null;

  if (sql_kind = 0) then
  begin
    us_uuid = select_us_uuid;

    select t.us_uuid
    from ux$frame, ux$source t
    where ufr_uuid = :ufr_uuid and
          ufr_add_table_fields = t.us_code
    into main_us_uuid;
  end

  if (main_us_uuid is null) then
    main_us_uuid = us_uuid;

  for select uf_uuid, uf_field_name, uf_is_param, uf_pk_field, us_is_table,
             uf_not_null, uf_readonly, uf_default
  from ux$field, ux$source
  where us_uuid in (:us_uuid, :main_us_uuid) and
        uf_owner = us_uuid
  order by iif(us_uuid = :us_uuid, 1, 2) desc, uf_position
  into :uf_uuid, :uf_field_name, :uf_is_param, :uf_pk_field, :us_is_table,
       :uf_not_null, :uf_readonly, :uf_default
  do
  begin
    if (sql_kind = 0) then
    begin
      if (not exists(select * from ux$frparam
      where ufp_ufr_uuid = :ufr_uuid and
            ufp_uf_uuid = :uf_uuid and
            ufp_parent is null)) then
      begin
        ufp_uf_uuid = null;
        ufp_uuid = null;

        select ufp_uuid, ufp_uf_uuid
        from ux$frparam, ux$field
        where ufp_ufr_uuid = :ufr_uuid and
              ufp_uf_uuid = uf_uuid and
              ufp_parent is null and
              uf_field_name = :uf_field_name
        rows 1
        into ufp_uuid, ufp_uf_uuid;

        if (ufp_uf_uuid is null) then
        begin
          insert into ux$frparam (ufp_ufr_uuid, ufp_uf_uuid, ufp_not_null)
          values (:ufr_uuid, :uf_uuid, :uf_not_null)
          returning ufp_uuid
          into ufp_uuid;
  
          if (coalesce(uf_default, '') <> '') then
            execute procedure uxp$uxparamdef_update(:ufp_uuid, null, :uf_default, 0);
        end else
        begin
          update ux$frparam
          set ufp_uf_uuid = :uf_uuid
          where ufp_uuid = :ufp_uuid;
        end
      end
    end else
    begin
      if (us_is_table = 0 and uf_is_param = 0) then
        Continue;

      if (us_is_table = 1 and uf_pk_field = 0 and sql_kind = 3) then
        Continue;

      if (us_is_table = 1 and uf_readonly) then
        Continue;

      if (not exists(select * from ux$framesql
      where ufs_ufr_uuid = :ufr_uuid and
            ufs_sql_kind = :sql_kind and
            ufs_uf_uuid = :uf_uuid)) then
      begin
        insert into ux$framesql (ufs_ufr_uuid, ufs_sql_kind, ufs_uf_uuid)
        values (:ufr_uuid, :sql_kind, :uf_uuid);
      end
    end
  end

  if (sql_kind = 0) then
  begin
    for select ufp_uf_uuid
    from ux$frparam, ux$field
    where ufp_ufr_uuid = :ufr_uuid and
          ufp_parent is null and
          ufp_uf_uuid = uf_uuid and
          uf_owner not in (:us_uuid, :main_us_uuid)
    into uf_uuid
    do
    begin
      delete from ux$frparam
      where ufp_ufr_uuid = :ufr_uuid and
            ufp_uf_uuid = :uf_uuid and
            ufp_parent is null;
    end
  end else
  begin
    for select ufs_uf_uuid
    from ux$framesql, ux$field
    where ufs_ufr_uuid = :ufr_uuid and
          ufs_sql_kind = :sql_kind and
          ufs_uf_uuid = uf_uuid and
          (not(uf_owner in (:us_uuid, :main_us_uuid)) or uf_readonly)
    into uf_uuid
    do
    begin
      delete from ux$framesql
      where ufs_ufr_uuid = :ufr_uuid and
            ufs_sql_kind = :sql_kind and
            ufs_uf_uuid = :uf_uuid;
    end
  end
end
