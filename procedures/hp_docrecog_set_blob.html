CREATE PROCEDURE HP_DOCRECOG_SET_BLOB(
    DRL_ID D_ID,
    BLS_BLOB D_BLOB_BINARY,
    BLS_FILENAME D_STR_100,
    PAGE_NO D_SMALLINT,
    BLS_KIND D_SMALLINT,
    BLS_SRC_SYSTEM D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    BLS_ID BIGINT)
AS
declare variable bls_caption d_str_100;
declare variable drl_status d_smallint;
declare variable pk_id d_integer;
declare variable cds_id d_id;
declare variable c_id d_id;
declare variable is_main_doc d_smallint;
declare variable need_photo2 d_integer;
declare variable acl_id d_id;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  bls_id = 0;
  select pk_id, pk_name, drl_from_c_id, coalesce(drl_is_main_doc, 0)
  from docrecoglist, passkind
  where drl_id = :drl_id and
        drl_from_pk_id = pk_id
  into pk_id, bls_caption, c_id, is_main_doc;

  bls_caption = 'Стр.' || :page_no || coalesce(': ' || bls_caption, '');

  select drm_has_page2
  from hp_get_drm_has_pages(:pk_id)
  into :need_photo2;

  -- Вставка блоба
  execute procedure hp_insert_into_blobstore (:bls_filename, :bls_caption, :bls_blob, :un_id, :bls_kind, :bls_src_system)
  returning_values :bls_id;

  cds_id = null;
  select cds_id
  from clientdocs
  where cds_from_c_id = :c_id and
        cds_from_drl_id = :drl_id
  order by cds_id desc
  rows 1
  into cds_id;

  -- к документу прикрепляем блобы
  if (page_no = 1) then
  begin
    update docrecoglist
    set drl_from_bls_1 = :bls_id,
        drl_from_un_id = :un_id,
        drl_status = iif(drl_from_bls_2 is not null or :need_photo2 = 0, 1, 0)
    where drl_id = :drl_id
    returning drl_status
    into drl_status;

    if (cds_id is not null) then
    begin
      if (is_main_doc = 0) then
      begin
        update clientdocs
        set cds_doc_blob_1 = :bls_id,
            cds_verify_status = -3,
            cds_from_un_id = :un_id
        where cds_id = :cds_id;
      end
      else
      begin
        update clients
        set c_blob_pass1_id = :bls_id,
            c_from_un_id = :un_id
        where c_id = :c_id;
      end
    end
  end
  else
  begin
    update docrecoglist
    set drl_from_bls_2 = :bls_id,
        drl_from_un_id = :un_id,
        drl_status = iif(drl_from_bls_1 is not null, 1, 0)
    where drl_id = :drl_id
    returning drl_status
    into drl_status;

    if (cds_id is not null) then
    begin
      if (is_main_doc = 0) then
      begin
        update clientdocs
        set cds_doc_blob_2 = :bls_id,
            cds_verify_status = -3,
            cds_from_un_id = :un_id
        where cds_id = :cds_id;
      end
      else
      begin
        update clients
        set c_blob_pass2_id = :bls_id,
            c_from_un_id = :un_id
        where c_id = :c_id;
      end
    end
  end

  if (drl_status = 1 and
      is_main_doc = 1) then
  begin
    -- Отправляем документ на верификацию.
    if (exists(select *
               from clients
               where c_id = :c_id and
                     coalesce(c_name_fio, '') <> '' and
                     c_birth_date is not null and
                     coalesce(c_pass_no, '') <> '' and
                     coalesce(c_pass_date, '') <> '' and
                     coalesce(c_blob_photo_id, 0) > 0 and
                     coalesce(c_blob_pass1_id, 0) > 0 and
                     (coalesce(c_blob_pass2_id, 0) > 0 or
                     :need_photo2 = 0) and
                     c_verify_status <> 1)) then
    begin
      if (db_id in (39801, 41701)) then
      begin --Доп прооверки для КГ
        if (exists(select *
                   from hp_status_param_verif(:c_id)
                   where result = 0)) then
        begin
          update clients
          set c_verify_status = 1,
              c_from_un_id = :un_id
          where c_id = :c_id;
        end

      end
      else
      begin
        update clients
        set c_verify_status = 1,
            c_from_un_id = :un_id
        where c_id = :c_id;
      end
    end
  end

  -- Если поля не показываем сотрудникам, то ставим на оцифровку
  if (exists(select *
             from docrecoglist, docrecogmain
             where drl_id = :drl_id and
                   drl_from_drm_id = drm_id and
                   drm_show_fields_staff = 0 and
                   drl_from_bls_1 is not null and
                   (drl_from_bls_2 is not null or
                   :need_photo2 = 0))) then
  begin
    select res_acl_id
    from hp_actionlist_insert(null, 1010100, 'DocRecogList', :drl_id, 'ActionGroup.DocRecog_Client', 0, :un_id)
    into acl_id;

    update docrecoglist
    set drl_from_acl_id = :acl_id
    where drl_id = :drl_id;
  end

  suspend;
end
