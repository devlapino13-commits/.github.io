CREATE PROCEDURE HP_BIOMET_CONSENT(
    MA_ID D_ID)
RETURNS (
    C_NAME_FIO D_STR_255,
    C_BIRTH_DATE D_DATE,
    KC_INN D_STR_20,
    C_PASS_NO D_STR_CODE,
    C_PASS_DATE D_DATE)
AS
begin
    for select c.c_name_fio,
               c.c_birth_date,
               coalesce(c.c_inn, '-'),
               c.c_pass_no,
               c.c_pass_date
    from miniapp ma
    join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
    join clients c on c.c_id = mac.mac_from_c_id
    where mac.mac_amount > 0
      and ma.ma_is_active = 1
      and ma.ma_date_end = '31.12.9999'
      and ma.ma_id = :ma_id
    into :c_name_fio,
         :c_birth_date,
         :kc_inn,
         :c_pass_no,
         :c_pass_date
    do
        suspend;
end
