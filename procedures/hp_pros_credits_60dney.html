CREATE PROCEDURE HP_PROS_CREDITS_60DNEY(
    REG_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    PARENT_NAME D_STR_255,
    REG_NAME D_STR_255,
    GROUP_NAME D_STR_100,
    KONTRACT_NO D_STR_100,
    LO_FIO D_STR_255,
    KC_FIO D_STR_255,
    KC_TEAM_NAME D_STR_255,
    KOLVO_PROS D_INTEGER,
    K_AMOUNT_APPROVED D_CURRENCY,
    BALANCE_OS D_CURRENCY,
    PERC_BALANCE D_CURRENCY,
    DATE_OF_LAST_REPAYMENT D_DATE,
    ZAD_BALANCE D_CURRENCY,
    C_NAME_FIO D_STR_255,
    VOZRAST D_INTEGER,
    K_CLOSE_DATE D_DATE)
AS
declare variable k_id d_id;
declare variable lo_id d_id;
declare variable new_kc_id d_id;
declare variable sba_date_begin d_date;
declare variable c_id d_id;
begin
  for select oks.oks_from_k_id, kc.kc_from_c_id,
             (cast(:rep_date_end as date) - oks.oks_delinq_date) as kolvo_pros
      from oksummary oks
      inner join region_list(:reg_id, 1, 3, 1) rl on rl.id_list = oks.oks_region_id
      inner join knclient kc on kc.kc_from_k_id = oks.oks_from_k_id and
            kc.kc_amount_approved > 0
      where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
            cast(:rep_date_end as date) - oks.oks_delinq_date > 60 and
            oks_close_state = 0
      into k_id, c_id, kolvo_pros
  do
  begin
    select k.k_kontract_no, k.k_fact_disb_lo_id, kal.kal_new_reg_id, kal.kal_new_kf_officer, kal.kal_new_kf_name,
           k.k_amount_approved, k.k_kredline_end
    from kontract k
    inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id and
          kal.is_delete = 0 and
          :rep_date_end between kal.kal_date and kal.kal_date_end
    where k.k_id = :k_id
    into kontract_no, lo_id, reg_id, new_kc_id, group_name, k_amount_approved, k_close_date;

    select c.c_name_fio, datediff(year, c.c_birth_date, :rep_date_end)
    from clients c
    where c.c_id = :c_id
    into c_name_fio, vozrast;

    select c.c_name_fio
    from clients c
    where c.c_id = :lo_id
    into lo_fio;

    select c.c_name_fio
    from clients c
    where c.c_id = :new_kc_id
    into kc_fio;

    select reg1.reg_name, reg2.reg_name
    from region reg1
    inner join region reg2 on reg2.reg_unicode = reg1.parent
    where reg1.reg_unicode = :reg_id and
          reg1.reg_unicode <> 0 and
          reg2.reg_unicode <> 0
    into reg_name, parent_name;

    select ok.ok_ol_date
    from operkred ok
    where ok.ok_type = 4 and
          ok.ok_ol_is_active = 1 and
          ok.ok_from_k_id = :k_id
    order by ok.ok_ol_date desc
    rows 1
    into date_of_last_repayment;

    select sum(amzl.zad_os), sum(amzl.zad_perc),
           sum(amzl.zad_os + amzl.sh_calc_perc + round(amzl.sh_calc_perc * 0.02, 0))
    from aarep_make_zad_list(:k_id, :rep_date_end, 1) amzl
    into balance_os, perc_balance, zad_balance;
    kc_team_name = '';
    if (exists(select *
               from sb_akt sa
               where sa.sba_from_k_id = :k_id and
                     sa.sba_is_delete = 0 and
                     :rep_date_end between sa.sba_date_begin and sa.sba_date_end)) then
    begin

      select sl.sbl_name
      from sb_akt sa
      inner join sb_list sl on sl.sbl_id = sa.sba_from_sbl_id
      where sa.sba_from_k_id = :k_id and
            sa.sba_is_delete = 0 and
            :rep_date_end between sa.sba_date_begin and sa.sba_date_end
      into kc_team_name;
      if (kc_team_name = 'ОВР') then
        kc_team_name = 'Колл. центр';
    end
    suspend;
  end
end
