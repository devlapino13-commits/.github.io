CREATE PROCEDURE HP_EDOC_CLIENT_SIGN(
    C_ID D_ID)
RETURNS (
    EDS_ID D_ID,
    EDS_FROM_EDL_ID D_ID,
    EDS_STATUS D_INTEGER,
    EDS_ENUM_EDOC_SIGN_KIND D_INTEGER,
    EDS_STATUS_DATETIME D_DATETIME,
    EDL_OBJECT_KIND D_INTEGER,
    EDL_OBJECT_ID D_ID,
    EDL_ENUM_EDOC_KIND D_INTEGER,
    EDL_STATUS D_INTEGER,
    EDL_CREATE_UN_ID D_ID,
    EDL_CREATE_DATETIME D_DATETIME,
    EDL_MANAGER_C_ID D_ID,
    EDS_STATUS_STR D_STR_CODE,
    EDL_STATUS_STR D_STR_CODE,
    EDOC_SIGN_KIND D_STR_CODE,
    EDOC_KIND D_STR_100,
    OBJ_INFO D_STR_255,
    EDL_FROM_EDC_ID D_INTEGER,
    EDL_BLANK_STATUS D_SMALLINT,
    EDL_BLANK_STATUS_STR D_STR_CODE,
    EDL_DATE_END D_DATE)
AS
declare variable OBJ_KIND_STR D_STR_CODE;
begin
  -- документы и соглашения подписанные клиентов
  for select eds_id, eds_from_edl_id, eds_status, eds_enum_edoc_sign_kind, eds_status_datetime,
             edl_object_kind, edl_object_id, edl_from_edc_id, edl_status, edl_create_un_id,
             edl_create_datetime, edl_manager_c_id, edc_name, edl_blank_status, edl_date_end
  from edoc_sign, edoc_list, edoc_cfg
  where eds_from_c_id = :c_id and
        eds_from_edl_id = edl_id and
        edl_from_edc_id = edc_id
  order by edl_create_datetime desc, eds_id desc, eds_status_datetime
  into :eds_id, :eds_from_edl_id, :eds_status, :eds_enum_edoc_sign_kind, :eds_status_datetime,
       :edl_object_kind, :edl_object_id, :edl_from_edc_id, :edl_status, :edl_create_un_id,
       :edl_create_datetime, :edl_manager_c_id, edoc_kind, edl_blank_status, edl_date_end
  do
  begin
    edl_enum_edoc_kind = edl_from_edc_id;

    eds_status_str = trim(case eds_status
      when 0  then 'не подписан'
      when 1  then 'подписан'
    end);

    edl_status_str = trim(case edl_status
      when 0 then 'не подписан'
      when 1 then 'подписан'
      when -1 then 'аннулирован'
    end);
    if (coalesce(edl_date_end, current_date) < current_date) then
      edl_status_str = edl_status_str || ', срок истёк';

    edl_blank_status_str = trim(case edl_blank_status
      when 0  then 'В очереди'
      when 1  then 'Готов'
      when -1  then 'Ошибка'
    end);

    if (eds_status = 1)
    then edoc_sign_kind = f_enum('edoc_sign_kind', eds_enum_edoc_sign_kind, 'name');
    else edoc_sign_kind = '';

    obj_info = '';
    obj_kind_str = f_objkind(edl_object_kind, 'name');

    if (obj_kind_str = 'Kontract') then
    begin
      select f_format('№ %1 от %2 г.', k_kontract_no, f_datetostr(k_kontract_date))
      from kontract
      where k_id = :edl_object_id
      into obj_info;
    end else
    if (obj_kind_str = 'Miniapp') then
    begin
      select f_format('№ %1 от %2 г.', ma_no, f_datetostr(ma_date_app))
      from miniapp
      where ma_id = :edl_object_id
      order by ma_id desc, ma_number desc
      rows 1
      into obj_info;
    end else
    if (obj_kind_str = 'Agreements') then
    begin
      select f_format('№ %1 от %2 г.', coalesce(ag_agreement_no, 'б/н'),
                      f_datetostr(coalesce(ag_agreement_date, ag_date_begin)))
      from agreements, refdata
      where ag_id = :edl_object_id and
            ag_from_rd_id = rd_id
      into obj_info;
    end else
    if (obj_kind_str = 'Clients') then
    begin
      select c_name_fio
      from clients
      where c_id = :edl_object_id
      into obj_info;
    end

    if (obj_info <> '')
    then obj_info =  edoc_kind || ', ' ||obj_info;
    else obj_info = edoc_kind;

    Suspend;
  end
end
