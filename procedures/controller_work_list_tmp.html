CREATE PROCEDURE CONTROLLER_WORK_LIST_TMP(
    UN_ID ID_TYPE,
    LIST_TYPE SMALLINT,
    KF_NAME_SHOW SMALLINT)
RETURNS (
    CONTR_ID INTEGER,
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    BLOCKED_DATE DATE,
    KF_COUNT INTEGER,
    KF_NAME STR_100,
    STATE VARCHAR(10),
    K_DISB_DATE DATE,
    MA_APP_ID ID_TYPE)
AS
declare variable k_id_tmp id_type;
declare variable kf_id_tmp id_type;
declare variable kf_id id_type;
declare variable rep_date date;
declare variable k_status integer;
declare variable new_schema_date date;
declare variable contrller_type str_255;
declare variable dw integer;
begin
  new_schema_date = '20.07.2015'; --'09.06.2099';  -- начало работы по новой схеме для контролеров
  rep_date = current_date;
  kf_name = '';
  kf_name_show = coalesce(kf_name_show, 0);
  kf_count = 0;
  k_id_tmp = 0;
  kf_id_tmp = 0;

  if (list_type in (1, 4)) then
    contrller_type = 'ControllerType.Specialist_Visitor';
  if (list_type = 2) then
    contrller_type = 'ControllerType.LO';
  if (list_type = 3) then
    contrller_type = 'ControllerType.Controller';

  dw = extract(weekday from rep_date);

  for select contr.contr_id, contr.contr_from_k_id, contr.contr_from_c_id,

             contr.contr_blocked_date, kont.k_from_kf_id, kont.k_status,

             kont.k_disb_date, kont.k_from_ma_app_id
      from controller contr
      inner join kontract kont on kont.k_id = contr.contr_from_k_id
      inner join refdata rd_type on rd_type.rd_id = contr.contr_rd_type
      inner join refdata rd_res on rd_res.rd_id = contr.contr_check_result
      where rd_res.rd_code = 'ControllerCheckStatus.NotChecked' and
            contr.contr_blocked_date is not null and
            contr.contr_from_un_id = :un_id and
            contr.contr_is_delete = 0 and
            rd_type.rd_code = :contrller_type and
            kont.k_status >= 0
      order by k_from_kf_id, contr.contr_from_k_id, contr.contr_from_c_id
      into :contr_id, :k_id, :c_id, :blocked_date, :kf_id, :k_status, :k_disb_date, :ma_app_id
  do
  begin
    if (kf_id <> :kf_id_tmp) then
    begin
      kf_count = kf_count + 1;
      kf_id_tmp = kf_id;

      if (kf_name_show = 1) then
      begin
        kf_name = '';
        kf_name = (select kal.kal_new_kf_name
                   from knaktlist kal
                   where kal.kal_from_kf_id = :kf_id and
                         :rep_date between kal.kal_date and kal.kal_date_end and
                         kal.is_delete = 0);
      end
    end

    if (list_type = 3) then
    begin
      if (k_id <> k_id_tmp) then
      begin
        k_id_tmp = k_id;
        state = '';

        if (current_date >= new_schema_date) then
        begin
          select (case ks_state
                    when 0 then 'В проц.'
                    when 1 then 'Выдан'
                  end)
          from kontstate ks
          where ks.ks_from_k_id = :k_id and
                :rep_date between ks.ks_date_begin and ks.ks_date_end and
                ks.ks_ol_is_delete = 0 and
                ks.ks_state >= 0
          into :state;
        end
        else
        begin
          select 'В проц.'
          from kontstate ks
          where ks.ks_from_k_id = :k_id and
                :rep_date between ks.ks_date_begin and ks.ks_date_end and
                ks.ks_ol_is_delete = 0 and
                ks.ks_state = 0
          into :state;
        end

        state = coalesce(state, '');

        if ((state <> '') or (dw in (0, 6))) then
        begin
          for select kncl.kc_from_c_id
              from knclient kncl
              where kncl.kc_from_k_id = :k_id and
                    kncl.kc_amount_approved > 0
              into :c_id
          do
          begin
            suspend;
          end
        end
      end
    end
    else
    if (list_type = 4) then
    begin
      if (k_status = 1) then -- для СВО в режиме - проверка выданных займов без ВО, показываю только выданные займы
      begin
        state = 'Выдан';
        suspend;
      end
    end
    else
    begin
      if (k_status = 0) then
      begin
        state = 'В проц.';
        suspend; --  для НЕ КОНТРОЛЕРОВ показываю только не утвержденные займы
      end
    end
  end
end
