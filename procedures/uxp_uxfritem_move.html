CREATE PROCEDURE UXP$UXFRITEM_MOVE(
    UFI_UUID_STR D_STR_CODE,
    NEW_PARENT_UUID_STR D_STR_CODE,
    NEW_POSITION D_INTEGER)
AS
declare variable new_parent_uuid d_uuid;
declare variable ufi_uuid d_uuid;
declare variable ufi_position d_integer;
declare variable ufi_owner d_uuid;
declare variable ufi_parent_uuid d_uuid;
begin
  new_parent_uuid = char_to_uuid(:new_parent_uuid_str);
  ufi_uuid = char_to_uuid(ufi_uuid_str);

  -- Если это НЕ поле, запрет на добавление внутрь таблицы или дерева
  if (exists(select * from ux$fritem
  where ufi_uuid = :new_parent_uuid and
        ufi_control in ('grid', 'tree'))) then
  begin
    if (not exists(select * from ux$fritem, ux$field
    where ufi_uuid = :ufi_uuid and
          ufi_uf_uuid = uf_uuid and
          uf_is_param = 0)) then
      exit;
  end

  if (new_parent_uuid is not null) then
  begin
    select ufi_parent_uuid
    from ux$fritem
    where ufi_uuid = :new_parent_uuid and
          ufi_kind = 1
    into ufi_parent_uuid;

    if (ufi_parent_uuid is not null) then
      new_parent_uuid = ufi_parent_uuid;
  end

  if (new_position >= 1000) then
  begin
    new_position = null;

    select max(ufi_position)
    from ux$fritem
    where ufi_parent_uuid = :new_parent_uuid
    into new_position;

    if (new_position is null) then
      new_position = 0;
  end

  select ufi_position, ufi_owner, ufi_parent_uuid
  from ux$fritem
  where ufi_uuid = :ufi_uuid
  into ufi_position, ufi_owner, ufi_parent_uuid;

  if (ufi_parent_uuid is distinct from :new_parent_uuid) then
    ufi_position = null;

  if (new_parent_uuid is not null) then
  begin
    if (not exists(select * from ux$fritem
                   where ufi_uuid = :new_parent_uuid and
                         ufi_owner is not distinct from :ufi_owner)) then
      exit;  -- Разные владельцы, не перемещаем
  end

  if (ufi_position is null) then
  begin
    -- Перемещается в разные ветки, управляется на уровне триггера
    update ux$fritem
    set ufi_parent_uuid = :new_parent_uuid,
        ufi_position = :new_position
    where ufi_uuid = :ufi_uuid;
  end else
  begin
    -- Перемещается внутри одной ветки, управляется тут на уровне процедуры
    update ux$fritem
    set ufi_position = :new_position
    where ufi_uuid = :ufi_uuid;
  end
end
