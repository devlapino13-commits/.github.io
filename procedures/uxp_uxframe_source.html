CREATE PROCEDURE UXP$UXFRAME_SOURCE(
    UX_UUID D_UUID,
    CURR_UFR_UUID D_UUID)
RETURNS (
    US_UUID D_UUID,
    UFR_UUID D_UUID,
    UF_IS_PARAM D_BOOL)
AS
declare variable ux_kind d_str_code;
declare variable ua_owner_kind d_str_code;
declare variable ua_owner d_uuid;
begin
  if (uuid_to_char(curr_ufr_uuid) = '00000000-0000-0000-0000-000000000000') then
    curr_ufr_uuid = null;

  uf_is_param = 0;
  if (curr_ufr_uuid is not null) then
  begin
    ufr_uuid = curr_ufr_uuid;

    select ufr_us_uuid
    from ux$frame
    where ufr_uuid = :curr_ufr_uuid
    into us_uuid;
  end else
  begin
    ux_kind = uxf$code(ux_uuid, 'kind');
    if (ux_kind = 'uxevent') then
    begin
      select evt_sender_uuid
      from ux$event
      where evt_uuid = :ux_uuid
      into ux_uuid;

      ux_kind = uxf$code(ux_uuid, 'kind');
    end

    if (ux_kind = 'uxfield') then
    begin
      select uf_owner, uf_is_param
      from ux$field
      where uf_uuid = :ux_uuid
      into us_uuid, uf_is_param;
    end else
    if (ux_kind = 'uxfrparam') then
    begin
      select ufr_us_uuid, ufr_uuid, uf_is_param
      from ux$frparam, ux$frame, ux$field
      where ufp_uuid = :ux_uuid and
            ufp_ufr_uuid = ufr_uuid and
            ufp_uf_uuid = uf_uuid
      into us_uuid, ufr_uuid, uf_is_param;
    end else
    if (ux_kind = 'uxfritem') then
    begin
      select ufr_us_uuid, ufr_uuid, uf_is_param
      from ux$fritem
           inner join ux$frame on (ufi_owner = ufr_uuid)
           left join ux$field on (ufi_uf_uuid = uf_uuid)
      where ufi_uuid = :ux_uuid
      into us_uuid, ufr_uuid, uf_is_param;

      if (uf_is_param is null) then
        uf_is_param = 0;
    end else
    if (ux_kind = 'uxgritem') then
    begin
      select ufr_us_uuid, ufr_uuid, uf_is_param
      from ux$gritem
           inner join ux$grid on (ugi_ugr_uuid = ugr_uuid)
           inner join ux$frame on (ugr_ufr_uuid = ufr_uuid)
           left join ux$field on (ugi_uf_uuid = uf_uuid)
      where ugi_uuid = :ux_uuid
      into us_uuid, ufr_uuid, uf_is_param;

      if (uf_is_param is null) then
        uf_is_param = 0;
    end else
    if (ux_kind = 'uxframe') then
    begin
      select ufr_us_uuid, ufr_uuid
      from ux$frame
      where ufr_uuid = :ux_uuid
      into us_uuid, ufr_uuid;
    end else
    if (ux_kind = 'uxgrid') then
    begin
      select ufr_us_uuid, ufr_uuid
      from ux$grid, ux$frame
      where ugr_uuid = :ux_uuid and
            ugr_ufr_uuid = ufr_uuid
      into us_uuid, ufr_uuid;
    end else
    if (ux_kind = 'uxcommand') then
    begin
      select ufr_us_uuid, ufr_uuid
      from ux$frame
      where ufr_owner = :ux_uuid
      into us_uuid, ufr_uuid;
    end else
    if (ux_kind = 'uxaction') then
    begin
      select ua_owner, uxf$kind(uu_kind, 'code')
      from ux$action, ux$uniq
      where ua_uuid = :ux_uuid and
            ua_owner = uu_id
      into ua_owner, ua_owner_kind;

      if (ua_owner_kind = 'uxframe') then
      begin
        select ufr_us_uuid, ufr_uuid
        from ux$frame
        where ufr_uuid = :ua_owner
        into us_uuid, ufr_uuid;
      end else
      begin
        select ufr_us_uuid, ufr_uuid
        from ux$frame
        where ufr_owner = :ua_owner and
              ufr_is_mainframe = 1
        into us_uuid, ufr_uuid;
      end
    end else
      exception e_default 'Undefined ux_kind: ' || coalesce(ux_kind, '<Null>');
  end

  suspend;
end
