CREATE PROCEDURE MC_ACTION_UNBLOCK(
    MCL_ID ID_TYPE,
    ACTION_ID ID_TYPE,
    REASON STR_255)
RETURNS (
    RES_CODE STR_100,
    ERR_TEXT STR_255)
AS
declare variable rd_id integer;
declare variable c_id id_type;
begin
  res_code = '';
  err_text = '';

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'mc_address_verify'
  into :rd_id;

  select t.mcl_from_c_id
  from mc_login t
  where t.mcl_id = :mcl_id
  into :c_id;

  if (c_id is null) then
  begin
    res_code = 'user_not_exists';
    err_text = 'Пользвоатель не оределен';
    suspend;
    exit;
  end

  if (not exists(select ao.aco_id
                 from actionorder ao
                 where ao.aco_respons_c_id = :c_id and
                       ao.aco_from_rd_id = :rd_id and
                       ao.aco_id = :action_id)) then
  begin
    res_code = 'action_not_blocked';
    err_text = 'Разблокировать не удалось, т.к. заявка за пользователем не заблокирована';
  end
  else
  begin
    update actionlist al
    set al.acl_status = 0
    where al.acl_id = :action_id;
  end

  suspend;
end
