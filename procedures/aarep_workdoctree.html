CREATE PROCEDURE AAREP_WORKDOCTREE(
    UN_ID DECIMAL(18,0),
    FIND_TXT VARCHAR(255),
    INCLUDE_ANNUL INTEGER)
RETURNS (
    WDT_ID INTEGER,
    WDT_PARENT INTEGER,
    WDT_LB INTEGER,
    WDT_RB INTEGER,
    WDT_NAME VARCHAR(255),
    WDT_MEMO VARCHAR(1024),
    WDT_KOLICH INTEGER,
    WDT_NAME_KOL VARCHAR(255),
    WDT_HAS_NEW INTEGER,
    WDT_SORT INTEGER)
AS
declare variable wk integer;
declare variable wn integer;
begin
  wk = 0;
  wn = 0;
  for select wdt_id, wdt_parent, wdt_lb, wdt_rb, left(wdt_name, 200), wdt_memo, wdt_sort, sum(wdt_kolich),
             sum(wdt_has_new)
      from aarep_workdoctree_tmp(:un_id, :find_txt, :include_annul)
      group by wdt_id, wdt_parent, wdt_lb, wdt_rb, wdt_name, wdt_memo, wdt_sort
      into :wdt_id, :wdt_parent, :wdt_lb, :wdt_rb, :wdt_name, :wdt_memo, :wdt_sort, :wdt_kolich, :wdt_has_new
  do
  begin
    if (wdt_kolich is null) then
      wdt_kolich = 0;
    if (wdt_has_new is null) then
      wdt_has_new = 0;
    if (wdt_kolich <> 0) then
      wdt_name_kol = wdt_name || ' (' || :wdt_kolich || ')';
    else
      wdt_name_kol = wdt_name;

    if (wdt_has_new <> 0) then
      wdt_name_kol = wdt_name_kol || ' [' || :wdt_has_new || ']';
    wk = wk + wdt_kolich;
    wn = wn + wdt_has_new;
    suspend;
  end
  wdt_id = -1;
  wdt_sort = -1;
  wdt_parent = null;
  wdt_lb = 0;
  wdt_rb = 0;
  wdt_name = ' Полный перечень';
  wdt_memo = 'Полный перечень документов без деления на категории';
  wdt_kolich = wk;
  wdt_has_new = wn;
  if (wdt_kolich <> 0) then
    wdt_name_kol = wdt_name || ' (' || :wdt_kolich || ')';
  else
    wdt_name_kol = wdt_name;

  if (wdt_has_new <> 0) then
    wdt_name_kol = wdt_name_kol || ' [' || :wdt_has_new || ']';

  suspend;
end
