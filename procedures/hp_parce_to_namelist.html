CREATE PROCEDURE HP_PARCE_TO_NAMELIST(
    OBJECT_KIND VARCHAR(100),
    OBJECT_ID DECIMAL(18,0),
    NAME_TXT VARCHAR(1024))
AS
declare variable obj_kind integer;
declare variable res_text varchar(1024);
declare variable res_count integer;
declare variable n integer;
declare variable res_word varchar(100);
declare variable nl_id integer;
declare variable full_added varchar(400);
begin
-- Процедура делит входящий текст по словам и сохраняет в таблице NameList и NameLinks

  obj_kind = f_objkind(object_kind, 'id');

-- Очищаем от лишних символов
  select res_text
  from hp_txt_clear(:name_txt)
  into res_text;

-- Удаляем ранее созданные связки слов с объектом
  delete from namelinks
  where nlk_object_kind = :object_kind and
        nlk_object_id = :object_id;

-- Если текст пуст, завершаем работу
  if (res_text = '') then
    exit;
  full_added = '';
-- Делим текст по словам и обрабатываем каждую
  while (res_text <> '') do
  begin
    n = position(' ', res_text);
    if (n > 0) then
    begin
      res_word = left(res_text, n - 1);
      res_text = substring(res_text from n + 1);
    end
    else
    begin
      res_word = res_text;
      res_text = '';
    end

    res_word = f_metaphone(:res_word);

    if (res_word <> '') then
    begin
      nl_id = null;
      select nl_id
      from namelist
      where nl_name = :res_word
      into :nl_id;
      if (nl_id is null) then
      begin
        insert into namelist (nl_name)
        values (:res_word)
        returning nl_id
        into :nl_id;
      end
      if (position('[' || nl_id || ']', full_added) = 0) then
      begin
        full_added = full_added || '[' || nl_id || ']';
        insert into namelinks (nlk_object_kind, nlk_object_id, nlk_from_nl_id)
        values (:obj_kind, :object_id, :nl_id);
      end
    end
  end
end
