CREATE PROCEDURE HP_COMPETITION_FOR_MO(
    UN_ID D_ID)
RETURNS (
    CPP_ID D_INTEGER,
    CPP_PERIOD_NAME D_STR_255)
AS
declare variable cpp_date_end d_date;
begin
  for select cpp.cpp_id, cpp.cpp_period_name, cpp_date_end
      from competperiod cpp
      where cpp.cpp_status = 1 and -- подтвежденные
            cpp.cpp_from_rd_id = f_get_rd_id('Competitions.Olimpic_2024_spf')
      order by cpp_date_end
      into :cpp_id, cpp_period_name, cpp_date_end
  do
  begin
    cpp_period_name = cpp_period_name || ' ' || 'на' || ' <b>' || extract(day from cpp_date_end) || ' ' ||(case
                                                                                                               when extract(month from cpp_date_end) = 1 then 'янв.'
                                                                                                               when extract(month from cpp_date_end) = 2 then 'фев.'
                                                                                                               when extract(month from cpp_date_end) = 3 then 'март'
                                                                                                               when extract(month from cpp_date_end) = 4 then 'апр.'
                                                                                                               when extract(month from cpp_date_end) = 5 then 'май'
                                                                                                               when extract(month from cpp_date_end) = 6 then 'июня.'
                                                                                                               when extract(month from cpp_date_end) = 7 then 'июля'
                                                                                                               when extract(month from cpp_date_end) = 8 then 'авг.'
                                                                                                               when extract(month from cpp_date_end) = 9 then 'сен.'
                                                                                                               when extract(month from cpp_date_end) = 10 then 'окт.'
                                                                                                               when extract(month from cpp_date_end) = 11 then 'ноя.'
                                                                                                               when extract(month from cpp_date_end) = 12 then 'дек.'
                                                                                                             end);

    suspend;
  end
end
