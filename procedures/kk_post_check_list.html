CREATE PROCEDURE KK_POST_CHECK_LIST(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_UN_ID ID_TYPE)
RETURNS (
    ACL_ID ID_TYPE,
    K_ID ID_TYPE,
    ACL_DATE DATE,
    K_KONTRACT_NO STR_100,
    IS_BLOKED INTEGER,
    KS_STATE INTEGER,
    K_LAST_REG_ID INTEGER,
    MARK STR_255)
AS
declare un_id id_type;
declare do_continue integer;
declare check_date date;
declare disb_old_id id_type;
begin
  for select acl_id, acl_date, acl_object_id, 'no'
      from actionlist
      where acl_date between :rep_date_begin and :rep_date_end and
            acl_object_kind = 9 and
            acl_status = 2 and
            acl_respons_c_id = 1000000000001 -- Sanjar Islamov
      union
      select acl_id, acl_date, acl_object_id, 'yes'
      from actionlist
      inner join actionmark on atm_from_acl_id = acl_id
      where acl_date between :rep_date_begin and :rep_date_end and
            acl_object_kind = 9 and
            acl_status = 2
      into :acl_id, :acl_date, :k_id, :mark
  do
  begin
    do_continue = 1;
    un_id = null;

    select kpc.kpc_checked_un_id, kpc.kpc_check_date
    from kk_post_check kpc
    where kpc.kpc_from_acl_id = :acl_id
    into :un_id, :check_date;

-- если проверка уже произведена, в списке эти заявки не отображаю
    if (un_id is not null) then
    begin
      if (un_id <> rep_un_id) then
        do_continue = 0;

      if (check_date is not null) then
        do_continue = 0;
    end

    if (do_continue = 1) then
    begin
      is_bloked = 0;

      if ((un_id = rep_un_id) and
          (check_date is null)) then
        is_bloked = 1;

      disb_old_id = null;

      ks_state = 0;

      select kont.k_kontract_no, kont.k_fact_disb_ol_id, kont.k_last_reg_id
      from kontract kont
      where kont.k_id = :k_id
      into :k_kontract_no, disb_old_id, :k_last_reg_id;

      if (disb_old_id is not null) then
      begin
        if (exists(select *
                   from operlist ol
                   where ol.ol_id = :disb_old_id and
                         ol.is_delete = 0)) then
          ks_state = 1;
      end

      suspend;
    end
  end
end
