CREATE PROCEDURE HP_WMA_BASIC_DATA_FILL(
    ESM_ID ID_TYPE,
    PARAMS VARCHAR(8000) = '')
RETURNS (
    ERROR_TEXT STR_255)
AS
declare variable ESM_DATE date;
declare variable WMA_PHONE_NO STR_100;
declare variable WMAT_NAME STR_100;
declare variable WMAT_BIRTHDAY date;
declare variable WMAT_CITY STR_100;
declare variable WMAT_GENDER_STR STR_10;
declare variable WMAT_CITIZENSHIP_STR STR_100;
declare variable WMAT_LANGUAGE_STR STR_100;
declare variable WMAT_MEETING_POINT_STR STR_100;
declare variable WMAT_MEETING_ADDRESS STR_255;
declare variable WMA_ROW_SOURCE integer;
declare variable ESM_FULL_SOURCE varchar(8000);
declare variable WMAT_GENDER smallint;
declare variable WMAT_CITIZENSHIP integer;
declare variable WMAT_LANGUAGE integer;
declare variable WMAT_MEETING_POINT integer;
declare variable WMAT_CITY_ID ID_TYPE;
declare variable WMA_ID integer;
declare variable DO_ACTION STR_100;
declare variable WMA_ID_ACTIVE integer;
declare variable START_WORKING_DATETIME timestamp;
declare variable WMAT_ID integer;
declare variable WMAT_LOAN_PURPOSE_STR STR_255;
declare variable WMAT_LOAN_PURPOSE integer;
declare variable WMA_DATE date;
begin
  /*
    command=save_cl_info
    &phone_no=79096667258
    &sms_code=5647
    &cl_name=УМАРОВ ХАЙИТБЕК АБДУМАНОБ УГЛИ
    &cl_birthday=13.03.1994
    &cl_city=Москва
    &cl_gender=Мужской
    &cl_citizenship=Узбекистан
    &cl_loan_term=
    &cl_loan_amount=
    &cl_language=Узбекский
    &cl_next_call_type=В течении 10 минут
    &cl_next_call_value=
    &meeting_point=Дома
    &meeting_address=Одинцова ул калхозная 118
    &cl_kolich_repay=0
    &data_source=www.molbulak.ru&
  */
  params = trim(coalesce(params, ''));

  select em.esm_date,
         em.esm_full_source
  from extsysmain em
  where em.esm_id = :esm_id
  into :esm_date,
       :esm_full_source;

  if (params = '') then
    params = esm_full_source;

  wma_phone_no = (select field_value
                  from hp_extsys_parcer(:params, 'phone_no'));


  /* 21.05.2019 igusev.
    Логика, предоставленная Санжаром при устном обсуждении на квартире у Толика

    Если поступила заявка с сайта МБФ, по номеру телефона ищу первую попавшуюся заявку с сайта, по которой статус тикета - "активный"
    Если
        нашел -> проверяю, начал ли КЦ работу по данной заявке, проверяю по полю start_working_datetime (дата первой блокировки)
            если
                начал    -> Сервис, должен вернть сообщение пользователю6 Вы ранее подавали заявку и она в обработке, позвоните в Колл-Центр
                не начал -> Update текущей заявки с сайта
         не нашел ->  Создаю новую заявку
  */


  wma_id = null;
  error_text = '';
  do_action = '';
  wma_date = null;

  /* 22.01.2020 igusev отключил данный блок т.к. Зуура просит, чтобы каждая поданная заявка со стороны клиента в КЦ рождала новую заявку.
  тема письма: "RE: не все заявки с сайта попадают в админку"
  так же выявлено что после выдчи займа тикет не закрывается, и как следствие клиент не может подать новую заявку

  -- ищу по номер телефона заявку с сайта, по которой не закрыта задача по тикету
  select wma.wma_id, wma.wma_start_working_datetime
  from cc_website_miniapp wma
  inner join ticket t on t.tkt_id = wma.wma_from_tkt_id
  where wma.wma_phone_no = :wma_phone_no
  and t.tkt_is_closed = 0
  order by  wma.wma_id desc 
  rows 1
  into :wma_id, :start_working_datetime;

  if (wma_id is null) then
    do_action = 'create_new';
  else
  begin
    if (start_working_datetime is not null) then
    begin
        error_text = 'Вы ранее подавали заявку и она в обработке. Пожалуйста свяжитесь с Колл-Центром по номеру (8 800 555 71 71)';
        suspend;
        exit;
    end
  end
  */

  -- 27.01.2020 igusev отключение выше-идущего блока привело к поялению дубликатов, что мешает работе КЦ. Дуюликаты сыплются из за проеблем на сторне сайта МБФ и нужно что-то делать чтобы новые задачи не ражделись по одной и той же заявке
  -- для решения проблемы добавил логику описанную ниже
  select wma.wma_id,
         wma.wma_start_working_datetime,
         wma.wma_date
  from cc_website_miniapp wma
  where wma.wma_phone_no = :wma_phone_no
  order by  wma.wma_id desc
  rows 1
  into :wma_id,
       :start_working_datetime,
       :wma_date;

  /*
    1) если заявка от данного лица вообще не поступала, тогда:
        создаю новую
    2) если поступила, тогда:
        а)если поступала сегодня, тогда:
            аа)если была в работе, тогда:
                возвращаю ошибку: Вы ранее подавали заявку и она в обработке. Пожалуйста свяжитесь с Колл-Центром по номеру (8 800 555 71 71)
                и заявку не регистрирую

            бб)если еще не была в работе, тогда произвожу update ранее прилетевшей заявки, т.е. обновляю набор полей по ней из только что обновленного набора данных, который прилетел с сайта МБФ
        б)если не поступила сегодня, тогда:
            регистрирую новую заявку
   */


  if (wma_id is null) then
    do_action = 'create_new';
  else
  begin
    if (wma_date = current_date) then
    begin
        if (start_working_datetime is not null) then
        begin
            error_text = 'Вы ранее подавали заявку и она в обработке. Пожалуйста свяжитесь с Колл-Центром по номеру (8 800 555 71 71)';
            suspend;
            exit;
        end
    end
    else
        do_action = 'create_new';
  end


  wmat_name = (select field_value
              from hp_extsys_parcer(:params, 'cl_name'));

  begin
    wmat_birthday = (select field_value
                    from hp_extsys_parcer(:params, 'cl_birthday'));
    when sqlcode -413 do
    begin
      wmat_birthday = null;
    end
  end

  wmat_city = left((select field_value
                   from hp_extsys_parcer(:params, 'cl_city')), 100);
  wmat_gender_str = left((select field_value
                         from hp_extsys_parcer(:params, 'cl_gender')), 10);
  wmat_citizenship_str = left((select field_value
                              from hp_extsys_parcer(:params, 'cl_citizenship')), 100);
  wmat_language_str = left((select field_value
                           from hp_extsys_parcer(:params, 'cl_language')), 100);
  wmat_meeting_point_str = left((select field_value
                                from hp_extsys_parcer(:params, 'meeting_point')), 100);
  wmat_meeting_address = left((select field_value
                              from hp_extsys_parcer(:params, 'meeting_address')), 255);

  wmat_loan_purpose_str = left((select field_value
                              from hp_extsys_parcer(:params, 'loan_purpose')), 255);


  wmat_gender = (case wmat_gender_str
                  when 'женский' then 0
                  when 'мужской' then 1
                  else null
                end);

  wmat_language = (case wmat_language_str
                    when 'Кыргызский' then f_get_rd_id('Language.KG')
                    when 'Русский'    then f_get_rd_id('Language.RU')
                    when 'Таджикский' then f_get_rd_id('Language.TJ')
                    when 'Узбекский'  then f_get_rd_id('Language.UZ')
                    else 0
                  end);

  wmat_citizenship = (case wmat_citizenship_str
                       when 'Кыргызстан'           then f_get_rd_id('Citizen.KG')
                       when 'Российская Федерация' then f_get_rd_id('Citizen.RU')
                       when 'Таджикистан'          then f_get_rd_id('Citizen.TJ')
                       when 'Узбекистан'           then f_get_rd_id('Citizen.UZ')
                       else 0
                     end);

  wmat_meeting_point = (case wmat_meeting_point_str
                         when 'Дома'      then f_get_rd_id('website_miniapp.meeting_point.home')
                         when 'На работе' then f_get_rd_id('website_miniapp.meeting_point.work')
                         when 'В офисе'   then f_get_rd_id('website_miniapp.meeting_point.office')
                         else 0
                       end);

   wmat_loan_purpose = (case wmat_loan_purpose_str
                         when 'На документы'      then f_get_rd_id('website_miniapp.loanpurpose.docs')
                         when 'Потребительский'   then f_get_rd_id('website_miniapp.loanpurpose.consumer')
                         when 'На покупку авто'   then f_get_rd_id('website_miniapp.loanpurpose.buycar')
                         when 'На оплату обучения' then f_get_rd_id('website_miniapp.LoanPurpose.Education')
                         else 0
                       end);

  wma_row_source = f_get_rd_id('website_miniapp_source.website_mbf');

  wmat_city = coalesce(wmat_city, '');
  if (wmat_city <> '') then
  begin
      wmat_city_id = (select pl_id
                     from okt_office_city_get(:wmat_city));
  end

  if (do_action = 'create_new') then
  begin
      insert into cc_website_miniapp (wma_date, wma_from_esm_id, wma_phone_no, wma_row_source)
      values (:esm_date, :esm_id, :wma_phone_no, :wma_row_source)
      returning wma_id
      into :wma_id;
  end

  insert into cc_website_miniapp_tmp (wmat_from_wma_id, wmat_name, wmat_birthday, wmat_city, wmat_gender,
                                      wmat_citizenship, wmat_language, wmat_meeting_point, wmat_meeting_address,
                                      wmat_city_id, wmat_loan_purpose)
  values (:wma_id, :wmat_name, :wmat_birthday, :wmat_city, :wmat_gender,
          :wmat_citizenship, :wmat_language, :wmat_meeting_point, :wmat_meeting_address,
          :wmat_city_id, :wmat_loan_purpose)
  returning wmat_id
  into :wmat_id;

  update cc_website_miniapp wma
  set wma.wma_last_wmat_id = :wmat_id
  where wma.wma_id = :wma_id;

  suspend;
end
