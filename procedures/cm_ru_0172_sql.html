CREATE PROCEDURE CM_RU_0172_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kf_client_pass_no varchar(30);
declare variable c_id id_type;
declare variable pr_id integer;
declare variable amount_min d_integer;
declare variable amount_max d_integer;
declare variable point_cl d_integer;
declare variable is_refund d_smallint;
declare variable active_loan_balance d_currency;
declare variable pr_code d_str_100;
declare variable final_pdn d_currency;
declare variable kc_id d_id;
declare variable mds d_currency;
declare variable perc_pdn d_currency;
declare variable refund_active_loan_balance d_currency;
begin
  for select c_id, c_name_fio, c_pass_no, ma_date_oper, mac_amount, ma_from_pr_id, ma_refund, pr_code
      from miniapp
      inner join product on pr_id = ma_from_pr_id and
            (pr_code containing '.reliable_client.' or pr_code containing '.reliable_client_plus.' or pr_code containing '.loyal.' or pr_code containing '.shans.' or pr_code containing '.express.' or pr_code containing '.universal.' or pr_code containing '.privilege.' or pr_code containing '.privilege_plus.' or pr_code containing '.oivo.' or pr_code containing '.refinance.' or pr_code containing '.refinance_oivo.')
      inner join miniappclient on mac_from_ma_id = ma_id and
            mac_amount > 0
      inner join clients on c_id = mac_from_c_id
      where ma_id = :ma_id and
            ma_date_end = '31.12.9999'
      into :c_id, :res_info_text, :kf_client_pass_no, :kf_disb_date, :kc_amount, :pr_id, is_refund, pr_code
  do
  begin
    if (kf_disb_date < cast('01.11.2020' as date)) then
      exit;

    point_cl = f_point_get(11, ma_id,c_id, kf_disb_date);

    if (not(/*pr_code containing '.loyal.' or */pr_code containing '.shans.' or pr_code containing '.express.')) then
    begin
      select amount_min, amount_max, point_cl
      from reliable_client_amount_interval(:c_id, :pr_id, :kf_disb_date)
      into :amount_min, :amount_max, :point_cl;
      if (kc_amount > amount_max) then
      begin
        res_error_text = 'У клиента ' || point_cl || ' балла, допустимая минимальная сумма ' || amount_min || ', максимальная ' || amount_max || ' рублей';
        res_error_is_critical = 1;
        suspend;
      end
    end

    if (kf_disb_date < cast('26.06.2023' as date)) then
    begin
      if (pr_code containing '.reliable_client.') then
      begin
        amount_max = case
                       when point_cl <= 2 then 35000
                       when point_cl <= 5 then 40000
                       when point_cl <= 8 then 50000
                       when point_cl <= 11 then 60000
                       when point_cl <= 14 then 80000
                       when point_cl <= 17 then 100000
                       when point_cl <= 23 then 120000
                       else 150000
                     end;
      end
      else
      if (pr_code containing '.reliable_client_plus.') then
      begin
        amount_max = case
                       when point_cl <= 2 then 55000
                       when point_cl <= 5 then 60000
                       when point_cl <= 8 then 70000
                       when point_cl <= 11 then 80000
                       when point_cl <= 17 then 100000
                       when point_cl <= 23 then 120000
                       else 150000
                     end;
      end
    end
    else
    if (kf_disb_date < cast('04.07.2024' as date)) then
    begin
      if (pr_code containing '.reliable_client.') then
      begin
        amount_max = case
                       when point_cl <= 2 then 55000
                       when point_cl <= 5 then 65000
                       when point_cl <= 8 then 75000
                       when point_cl <= 11 then 85000
                       when point_cl <= 14 then 100000
                       when point_cl <= 17 then 130000
                       when point_cl <= 23 then 150000
                       else 200000
                     end;
      end
      else
      if (pr_code containing '.reliable_client_plus.') then
      begin
        amount_max = case
                       when point_cl <= 2 then 75000
                       when point_cl <= 5 then 80000
                       when point_cl <= 8 then 90000
                       when point_cl <= 11 then 100000
                       when point_cl <= 17 then 130000
                       when point_cl <= 23 then 150000
                       else 200000
                     end;
      end
    end
    else
    begin
      if (pr_code containing '.reliable_client.' or pr_code containing '.loyal.' or pr_code containing '.shans.' or pr_code containing '.express.' or pr_code containing '.universal.' or pr_code containing '.privilege.' or pr_code containing '.oivo.') then
      begin
        final_pdn = 0;
        /* --   на стадии заявки ПДН не проверяем, коммент от Ю. Каштановой, задача 1900
            if (check_kind containing '.create_loan.' or
            check_kind containing '.save_application.') then
            begin
              select coalesce(kc_pdn_koef, 0)
              from knclient_pdn
              where kc_from_c_id = :c_id
              order by kc_id desc
              rows 1
              into :final_pdn;
            end
            final_pdn = 0;  -- временно, Улан потом переделает.
            */
        perc_pdn = 80;
        if (pr_code containing '.privilege.') then
          perc_pdn = 50;

        if (kf_disb_date < cast('15.08.2024' as date)) then
        begin
          if (final_pdn < perc_pdn) then
          begin
            amount_max = case
                           when point_cl <= 2 then 75000
                           when point_cl <= 5 then 90000
                           when point_cl <= 8 then 100000
                           when point_cl <= 11 then 120000
                           when point_cl <= 17 then 150000
                           when point_cl <= 23 then 200000
                           else 200000
                         end;
          end
          else
          begin
            amount_max = case
                           when point_cl <= 11 then 75000
                           when point_cl <= 17 then 90000
                           when point_cl <= 23 then 100000
                           else iif(kf_disb_date < '17.07.2024', 130000, 200000)
                         end;
          end
        end
        else
        if (kf_disb_date < cast('10.07.2025' as date)) then
        begin
          if (final_pdn < perc_pdn) then
          begin
            amount_max = case
                           when point_cl <= 3 then 80000
                           when point_cl <= 6 then 90000
                           when point_cl <= 9 then 110000
                           when point_cl <= 12 then 120000
                           when point_cl <= 18 then 150000
                           else 200000
                         end;
          end
          else
          begin
            amount_max = case
                           when point_cl <= 6 then 55000
                           when point_cl <= 9 then 75000
                           when point_cl <= 18 then 85000
                           else 100000
                         end;
          end
        end
        if (kf_disb_date < cast('10.02.2026' as date)) then
        begin
          amount_max = case
                         when point_cl <= 9 then 100000
                         when point_cl <= 15 then 150000
                         when point_cl <= 35 then 250000
                         when point_cl <= 45 then 300000
                         else 400000
                       end;
        end
        else   --RFCRU-4949
        begin
          amount_max = case
                         when point_cl <= 6 then 100000
                         when point_cl <= 11 then 150000
                         when point_cl <= 17 then 200000
                         when point_cl <= 24 then 250000
                         when point_cl <= 35 then 300000
                         when point_cl <= 45 then 350000
                         else 450000
                       end;
        end
        if (pr_code containing '.privilege.') then
          amount_max = amount_max + 100000;
      end
      else
      if (pr_code containing '.reliable_client_plus.') then
      begin
        amount_max = case
                       when point_cl <= 2 then 85000
                       when point_cl <= 5 then 100000
                       when point_cl <= 11 then 120000
                       when point_cl <= 17 then 150000
                       else 200000
                     end;
      end
    end

    select mds
    from hp_max_amount_allowed(11, :ma_id, :amount_max)
    into mds;


    active_loan_balance = 0;
    if (is_refund = 0) then
    begin
      select coalesce(sum(oks_balance), 0)
      from oksummary, knclient
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            oks_from_k_id = kc_from_k_id and
            :kf_disb_date between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into active_loan_balance;
    end
    else
    begin
      select sum(oks_balance)                               -- RFCRU-3486
      from oksummary, knclient, kontract, product
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            oks_from_k_id = kc_from_k_id and
            :kf_disb_date between oks_date_begin and oks_date_end and
            oks_close_state = 0 and
            oks_from_k_id = k_id and
            k_from_pr_id = pr_id
      into active_loan_balance;

      select  coalesce(sum(oks_balance), 0)                  -- RFCRU-3486
      from oksummary, knclient kc
      inner join kontrefinance kr on kr.kr_from_k_id = kc.kc_from_k_id
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            oks_from_k_id = kc_from_k_id and
            current_date between oks_date_begin and oks_date_end and
            oks_close_state = 0 and
            kr.kr_from_ma_id = :ma_id and
            kr.kr_from_k_id = kc_from_k_id
      into refund_active_loan_balance;
    end

    refund_active_loan_balance = coalesce(refund_active_loan_balance, 0);

    /*if ((amount_max < active_loan_balance + kc_amount) and
        is_refund = 1) then
    begin
      if ((pr_code containing '.reliable_client.' or pr_code containing '.loyal.' or pr_code containing '.shans.') and
          1 <> 1) then
      begin
        res_error_text = f_format('Превышена макс. допустимая сумма, тек. баланс: %1, балл: %2, ПДН: %3', f_numtostr(active_loan_balance, 0), point_cl, f_numtostr(final_pdn, 2));
      end
      else
      begin
        res_error_text = f_format('Превышена макс. допустимая сумма, тек. баланс: %1, балл: %2.', f_numtostr(active_loan_balance, 0), point_cl);
      end
      res_error_is_critical = 2;
      suspend;
    end */

    if (current_date < cast('23.01.2025' as date) and
        (amount_max < active_loan_balance + kc_amount)) then
    begin
      res_error_text = f_format('Превышена макс. допустимая сумма, тек. баланс: %1, балл: %2.', f_numtostr(active_loan_balance, 0), point_cl);
      res_error_is_critical = 2;
      suspend;
    end
    if (current_date >= cast('23.01.2025' as date) and
        (mds < (kc_amount - refund_active_loan_balance))) then
    begin
      res_error_text = f_format('Превышена макс. допустимая сумма, тек. баланс: %1, балл: %2.', f_numtostr(active_loan_balance, 0), point_cl);
      res_error_is_critical = 2;
      suspend;
    end

  end
end
