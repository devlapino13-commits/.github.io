CREATE PROCEDURE HP_GET_KNCLIENT(
    K_ID D_ID)
RETURNS (
    KC_FROM_K_ID D_ID,
    KC_NUMBER D_INTEGER,
    KC_FROM_C_ID D_ID,
    KC_AMOUNT_APPLICATE D_CURRENCY,
    KC_FROM_PT_ID VARCHAR(20),
    KC_BUSINESS_TYPE D_INTEGER,
    KC_IS_NEW_BUSINESS D_SMALLINT,
    KC_IS_REGISTERED D_SMALLINT,
    KC_BUSINESS_NAME D_STR_255,
    KC_MONTHLY_OBOROT D_CURRENCY,
    KC_KAPITAL D_CURRENCY,
    KC_MONTH_PROFIT D_CURRENCY,
    KC_AMOUNT_APPROVED D_CURRENCY,
    KC_WORK_KOLICH D_INTEGER,
    KC_WORK_KOLICH_FAMILY D_INTEGER,
    KC_IS_BUSINESS_FULL_TIME D_SMALLINT,
    KC_BUSUNESS_SUBTYPE D_INTEGER,
    KC_IS_NEW_ADDED D_SMALLINT,
    KC_STATUS D_SMALLINT,
    KC_CIRCLE D_INTEGER,
    KC_PATENT_NO VARCHAR(40),
    KC_PATENT_DATE DATE,
    KC_AGENT_ID D_ID,
    KC_HAS_LOANS D_INTEGER,
    KC_HAS_FAMILY_LOANS D_INTEGER,
    CLIENT_NAME D_STR_100,
    CLIENT_PASS VARCHAR(30),
    CLIENT_INN VARCHAR(20),
    C_TYPE D_INTEGER,
    C_IS_PERSONAL D_SMALLINT,
    C_SEX D_SMALLINT,
    KC_KOMMIS_LGOTA D_CURRENCY,
    KC_LOAN_TARGET D_INTEGER,
    C_HAS_PHOTO D_STR_20,
    C_HAS_PASS D_STR_20,
    BS_NAME VARCHAR(50))
AS
begin
  for select kc_from_k_id, kc_number, kc_from_c_id, kc_amount_applicate, kc_from_pt_id, kc_business_type,
             kc_is_new_business, kc_is_registered, kc_business_name, kc_monthly_oborot, kc_kapital, kc_month_profit,
             kc_amount_approved, kc_work_kolich, kc_work_kolich_family, kc_is_business_full_time, kc_busuness_subtype,
             kc_is_new_added, kc_status, kc_circle, kc_patent_no, kc_patent_date, kc_agent_id, kc_has_loans,
             kc_has_family_loans, c_name_fio as client_name, c_pass_no as client_pass, c_inn as client_inn, c_type,
             c_is_personal, c_sex, kc_kommis_lgota, kc_loan_target,
             iif(c_blob_photo_id > 0, 'Есть Фото клиента', 'Нет фото клиента') as c_has_photo,
             iif(c_blob_pass1_id > 0, 'Есть копия паспорта', 'Нет копии паспорта') as c_has_pass,
             bs.bs_name
      from knclient kc
      inner join clients c on kc.kc_from_c_id = c.c_id
      inner join bussubtype bs on bs.bs_id = kc.kc_busuness_subtype and
        bs.bs_is_delete = 0
      where kc_from_k_id = :k_id
      into kc_from_k_id, kc_number, kc_from_c_id, kc_amount_applicate, kc_from_pt_id, kc_business_type,
           kc_is_new_business, kc_is_registered, kc_business_name, kc_monthly_oborot, kc_kapital, kc_month_profit,
           kc_amount_approved, kc_work_kolich, kc_work_kolich_family, kc_is_business_full_time, kc_busuness_subtype,
           kc_is_new_added, kc_status, kc_circle, kc_patent_no, kc_patent_date, kc_agent_id, kc_has_loans,
           kc_has_family_loans, client_name, client_pass, client_inn, c_type, c_is_personal, c_sex,
           kc_kommis_lgota, kc_loan_target, c_has_photo, c_has_pass, bs_name
  do
  begin
    suspend;
  end
end
