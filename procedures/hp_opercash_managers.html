CREATE PROCEDURE HP_OPERCASH_MANAGERS(
    OFFICE_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    ID D_ID,
    TEXT D_STR_255)
AS
declare variable chief_id d_id;
declare variable chief_fio d_str_100;
begin
  if (f_get_db_id() in (39801, 41701)) then
  begin
    select un_from_c_id, un_fio
    from username
    where un_name = 'kabdymalikova'
    into chief_id, chief_fio;
  end

  for select ps_from_c_id, c_name_fio
  from hr_pstatus, clients, joblist, jobkind
  where ps_from_reg_id = :office_id and
        ps_is_working = 1 and
        ps_status = 1 and
        ps_from_c_id = c_id and
        ps_from_jl_id = jl_id and
        ps_date_begin <= :rep_date_end and
        ps_date_end >= :rep_date_end and
        jl_kind = jk_id and
        jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior',
                           'JobSystem.Accountant_Chief', 'JobSystem.Accountant_Officer', 'JobSystem.Accountant_AlterChief')
  order by ps_date_begin
  into id, text
  do
  begin
    if (id = chief_id) then
      chief_id = null;

    suspend;
  end

  if (chief_id is not null) then
  begin
    id = chief_id;
    text = chief_fio;
    suspend;
  end

  if (f_get_db_id() in (39801, 41701) and office_id = 1030600) then   -- задача RFCKG-3441
    for select un_from_c_id, un_fio
    from username
    where un_name = 'mimankanova'
    into id, text
    do
      suspend;

end
