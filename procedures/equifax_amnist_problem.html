CREATE PROCEDURE EQUIFAX_AMNIST_PROBLEM(
    STEP D_INTEGER)
RETURNS (
    K_ID D_ID,
    K_ORIG_ID D_ID,
    K_KONTRACT_NO D_STR_100,
    KONTRACT_NO_ORIG D_STR_100,
    C_ID D_ID,
    KC_NUMBER D_INTEGER,
    IS_PORUCH D_ID,
    CRED_NO D_STR_100,
    CRED_TYPE D_INTEGER,
    PASS_NO D_STR_100,
    K_DISB_DATE D_DATE,
    K_KREDLINE_END D_DATE,
    RES_STR D_STR_255)
AS
declare variable C_ID_TMP D_ID;
declare variable IS_OBESP D_INTEGER;
declare variable cred_no_tmp d_str_100;
begin
    /*
    Процедура предназначена для формирвоание файла для Эквифакс на удаление всех займов, связанных с амнистиями.
    */

    if (step = 1) then
    begin
        is_obesp = 1;
        for select kont.k_id,
                   kont.k_base_k_id,
                   kont.k_kontract_no,
                   kont.k_disb_date,
                   kont.k_kredline_end
        from kontract kont
        inner join kontstate ks on kont.k_id  = ks.ks_from_k_id
        where kont.k_base_k_id is not null
        and ks.ks_state = 1
        and ks.ks_date_begin >= '01.07.2014'
        and ks.ks_ol_is_delete = 0
        into :k_id, :k_orig_id, :k_kontract_no, :k_disb_date, :k_kredline_end
        do
        begin
            kontract_no_orig = null;
    
            select kont.k_kontract_no
            from kontract kont
            where kont.k_id = :k_orig_id
            into :kontract_no_orig;
    
    
            for select kc_from_c_id,
                       kc_number
              from knclient kncl
              inner join clients cl on cl.c_id = kncl.kc_from_c_id
              where kncl.kc_from_k_id = :k_id and
                    kncl.kc_amount_approved > 0 and
                    cl.c_type = 0
            into :c_id_tmp, :kc_number
            do
            begin
                if (is_obesp = 0) then -- если кредит не обеспечен участники группы друг за друга не отвечают
                begin
                  is_poruch = 0;
                  c_id = :c_id_tmp;
        
                  suspend;
                end
                else
                begin
                  -- :type_ = 3, говорит о том что участник группы выступает в качестве поручителя для остальных участников группы
                  for select kc_from_c_id,
                             iif(kncl.kc_from_c_id = :c_id_tmp, 0, 1)
                      from knclient kncl
                      where kncl.kc_from_k_id = :k_id and
                            kncl.kc_amount_approved > 0
                      into :c_id, :is_poruch
                  do
                  begin
                    suspend;
                  end
        
                  /* добавил igusev 24.01.2020 т.к. была проблема:
                  Из данного группового займа чз амнистию один участник отделяется.
                  И после отделения информация о том что он поручительством связан с другими участниками по кредиту из которого он отделился больше не отправляется.
                  Т.е. мы имеем заем который живет своей жизнью.
                  И если этот амнистированный договор закроется, информация о том, что участники группы из базового кредита теперь не имеют активного договора поручительства не передавалась
        
                  Сейчас этот момент поправил. И если договор связан амнистией и обеспечен, тогда все участники группы будут продолжать передаваться как поручители друг другу вплоть до закрытия займа. 
        
                  Мое письмо. Тема: RE: Эквифакс Пт 24.01.2020 15:13
                  */
                  if (k_orig_id is not null) then
                  begin
                        for select kc_from_c_id
                        from kontract kont
                        inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
                        where kncl.kc_amount_approved > 0
                        and kont.k_base_k_id = :k_orig_id
                        and kont.k_id <> :k_id
                        into :c_id
                        do
                        begin
                            is_poruch = 1;
                            suspend;
                        end
                  end
                end
            end
       end
   end

        into :K_ID, K_ORIG_ID, K_KONTRACT_NO, KONTRACT_NO_ORIG, C_ID, KC_NUMBER, IS_PORUCH, k_disb_date, k_kredline_end
        do
        begin
          if (is_poruch = 0) then -- рессчитываю только для заемщиков
          begin
            cred_type = '19';
    
            if (kontract_no_orig is null) then
                cred_no = k_kontract_no || '(' || cast(kc_number as varchar(10)) || ')';

            suspend;

          else
          begin
             cred_type = '90';
            suspend;
          end
        end

        into :K_ID, K_ORIG_ID, K_KONTRACT_NO, KONTRACT_NO_ORIG, C_ID, KC_NUMBER, IS_PORUCH, CRED_NO, CRED_TYPE, k_disb_date, k_kredline_end
        do
        begin
            cred_no_tmp = cred_no;
            if ((K_ID <> K_ORIG_ID) and (KC_NUMBER <> 1) and (IS_PORUCH = 0)) then
            begin

            end

            suspend;
        end

        from EQUIFAX_AMNIST_PROBLEM(3)
        order by K_ORIG_ID, C_ID
