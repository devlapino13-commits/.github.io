CREATE PROCEDURE AAREP_ACT_KONTRACTS_MASTER(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    C_ID ID_TYPE,
    SVZ_FIO STR_100,
    REG_NAME STR_100,
    JL_NAME STR_100)
AS
declare variable reg_id_old integer;
declare variable reg_id integer;
begin
  reg_id_old = 0;

  for select distinct oks.oks_kc_id, id_list
      from oksummary oks
      inner join region_list(:rep_region_id, 1, -1, 1) on id_list = oks.oks_region_id
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks.oks_close_state = 0 and
            oks.oks_kc_id > 1000000000004
      into :c_id, :reg_id
  do
  begin
    select cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into :svz_fio;

    select jl.jl_name
    from hr_pstatus hps
    inner join joblist jl on jl.jl_id = ps_from_jl_id
    inner join jobkind jk on jk.jk_id = jl.jl_kind
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          hps.ps_from_c_id = :c_id rows 1
    into :jl_name;

    if (reg_id <> reg_id_old) then
    begin
      reg_id_old = reg_id;

      select reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :reg_name;
    end

    suspend;
  end
end
