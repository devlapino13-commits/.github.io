CREATE PROCEDURE PRINT_PAYMENT_MON_OPERLIST(
    OBJ_ID D_ID)
RETURNS (
    OL_DATE D_DATE,
    OL_DOC_NO D_STR_10,
    OL_MEMO D_STR_LONG,
    OL_DATETIME D_DATETIME,
    UN_NAME D_STR_CODE,
    SUM_OF_OK_OS DECIMAL(18,2),
    SUM_OF_OK_PERC DECIMAL(18,2),
    SUM_OF_OK_FINE DECIMAL(18,2),
    SUM_OF_OK_TAX DECIMAL(18,2),
    OK_PAY_TYPE D_INTEGER,
    SUM_OF_OK_KOMMIS DECIMAL(18,2),
    OK_TYPE D_INTEGER,
    OK_PAY_TYPE_TEXT D_STR_20,
    OK_TYPE_TEXT D_STR_100,
    SUM_OF_ALL DECIMAL(18,2),
    OL_ID D_STR_20,
    OL_DATE_OPER_DAY D_DATE,
    OK_SH_DATE D_DATE,
    OK_SH_NUMBER D_INTEGER,
    SUM_OF_DEPO DECIMAL(18,2),
    KOL_DELIQ D_INTEGER,
    OL_CLIENT_NAME D_STR_LONG,
    KOL_DELINQ_STR D_INTEGER)
AS
declare variable rep_date d_date;
begin
  rep_date = current_date;

  for select t.ol_date, t.ol_doc_no, t.ol_memo, t.ol_datetime, t.un_name, t.sum_of_ok_os, t.sum_of_ok_perc,
             t.sum_of_ok_fine, t.sum_of_ok_tax, t.ok_pay_type, t.sum_of_ok_kommis, t.ok_type, t.ok_pay_type_text,
             t.ok_type_text, t.sum_of_all, t.ol_id, t.ol_date_oper_day, t.ok_sh_date, t.ok_sh_number, t.sum_of_depo,
             t.kol_deliq, t.ol_client_name, t.kol_delinq_str
      from kontract_full_oper_list_detail(:obj_id, :rep_date) t
      into :ol_date, :ol_doc_no, :ol_memo, :ol_datetime, :un_name, :sum_of_ok_os, :sum_of_ok_perc, :sum_of_ok_fine,
           :sum_of_ok_tax, :ok_pay_type, :sum_of_ok_kommis, :ok_type, :ok_pay_type_text, :ok_type_text, :sum_of_all,
           :ol_id, :ol_date_oper_day, :ok_sh_date, :ok_sh_number, :sum_of_depo, :kol_deliq, :ol_client_name,
           :kol_delinq_str
  do
  begin
    suspend;
  end
end
