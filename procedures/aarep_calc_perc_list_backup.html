CREATE PROCEDURE AAREP_CALC_PERC_LIST_BACKUP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_K_ID DECIMAL(18,0),
    AAREP_BALANCE_BEGIN DECIMAL(18,4),
    AAREP_PERC_BEGIN DECIMAL(18,4),
    AAREP_PERC_CALCED DECIMAL(18,4),
    AAREP_PERC_PAYED DECIMAL(18,4),
    AAREP_PERC_WRITEOFF DECIMAL(18,4),
    AAREP_PERC_END DECIMAL(18,4),
    AAREP_BALANCE_END DECIMAL(18,4),
    AAREP_KONTRACT_NO VARCHAR(20),
    AAREP_CLIENT_NAME VARCHAR(100),
    AAREP_IS_INDIV VARCHAR(10),
    AAREP_PRODUCT_NAME VARCHAR(50),
    AAREP_DISB_DATE DATE,
    AAREP_DISB_AMOUNT DECIMAL(18,4),
    AAREP_CALCED_ALL DECIMAL(18,4),
    AAREP_PAYED_ALL DECIMAL(18,4),
    AAREP_WRITEOFF_ALL DECIMAL(18,4),
    AAREP_PERCENT_PER_DAY DECIMAL(18,7),
    REP_DATE_END DATE,
    REP_MEMO VARCHAR(255),
    REP_CURRENCY VARCHAR(3),
    K_CLIENTS_CASHIER_ID VARCHAR(20),
    C_ID DECIMAL(18,0),
    K_STOP_PERC DATE)
AS
declare variable aarep_loan_t integer;
declare variable pr_calc_perc_tax integer;
declare variable aarep_calc_method integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable pr_id integer;
declare variable pr_id_prior integer;
declare variable db_id integer;
declare variable k_kredline_end date;
declare variable k_zero_balance_date date;
declare variable disb_amount decimal(18,4) = 0;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */

  :db_id = f_get_db_id();

  rep_date_end = report_date_end;
  rep_memo = 'Проценты по кредиту';
  rep_currency = '';
  pr_id_prior = 0;

  for select kk_id, k_amount_approved, k_percent_per_day, k_kontract_no, k_stop_perc, k_clients_cashier_id, k_currency,
             kk_from_pr_id, k_kredline_end, k_zero_balance_date
      from kontract, aarep_active_kont_id_in_period(:report_region_id, :report_date_begin, :rep_date_end, 0, 0, 0, 0, 0)
      where k_id = kk_id
      into :aarep_k_id, :disb_amount, :aarep_percent_per_day, :aarep_kontract_no, :k_stop_perc, :c_id, :rep_currency,
           :pr_id, :k_kredline_end, :k_zero_balance_date
  do
  begin
    if (pr_id <> pr_id_prior) then
    begin
      select pr_interest_method, pr_loan_type, pr_name, pr_calc_perc_tax
      from product
      where pr_id = :pr_id
      into :aarep_calc_method, :aarep_loan_t, :aarep_product_name, :pr_calc_perc_tax;
      pr_id_prior = pr_id;
      if (aarep_loan_t = 1) then
        aarep_is_indiv = 'Индив.';
      else
        aarep_is_indiv = 'Группа';
    end

    aarep_client_name = '';
    if (c_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :c_id
      into :aarep_client_name;
    end
    aarep_calced_all = 0;

    for select result_balance_begin, result_perc_begin, result_calced, result_payed, result_writeoff, result_perc_end,
               result_balance_end, result_calced_all, result_payed_all, result_writeoff_all, result_disb_amount,
               result_disb_date
        from aarep_calc_perc_one_201302(:aarep_k_id, :report_date_begin, :report_date_end, :aarep_percent_per_day, :aarep_calc_method, :disb_amount, 2, :k_stop_perc)
        into :aarep_balance_begin, :aarep_perc_begin, :aarep_perc_calced, :aarep_perc_payed, :aarep_perc_writeoff,
             :aarep_perc_end, :aarep_balance_end, :aarep_calced_all, :aarep_payed_all, :aarep_writeoff_all,
             :aarep_disb_amount, aarep_disb_date
    do
    begin
      k_clients_cashier_id = 'N' || cast(c_id as varchar(15));
      suspend;
    end

    /*
    for select result_balance_begin, result_perc_begin, result_calced, result_payed, result_writeoff, result_perc_end, result_balance_end,
    result_calced_all, result_payed_all, result_writeoff_all,
    result_disb_amount, result_disb_date
    from aarep_calc_perc_one(:aarep_k_id, :report_date_begin, :report_date_end, 1, :db_id, :k_kredline_end, :k_stop_perc, :k_zero_balance_date)
    into :aarep_balance_begin, :aarep_perc_begin, :aarep_perc_calced, :aarep_perc_payed, :aarep_perc_writeoff, :aarep_perc_end, :aarep_balance_end,
    :aarep_calced_all, :aarep_payed_all, :aarep_writeoff_all,
    :aarep_disb_amount, aarep_disb_date
    do
    begin
      K_CLIENTS_CASHIER_ID='N'||cast(c_id as varchar(15));
      Suspend;
    end
    */
  end

end
