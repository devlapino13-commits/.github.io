CREATE PROCEDURE AAREP_TAXAMORT_TEMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OL_DATE DATE,
    OAM_FROM_OL_ID DECIMAL(18,0),
    OAM_BILL_ID DECIMAL(18,0),
    OAM_BILL_LINE_NO INTEGER,
    OB_FROM_ML_ID DECIMAL(18,0),
    OB_FROM_MC_ID DECIMAL(18,0),
    OB_INVENT_STR VARCHAR(50),
    ML_NAME VARCHAR(255),
    MC_NAME VARCHAR(100),
    MC_STATUS INTEGER,
    OM_COST_SALDO_BEGIN DECIMAL(18,4),
    OM_COST DECIMAL(18,4),
    OM_COST_IN DECIMAL(18,4),
    OM_COST_OUT DECIMAL(18,4),
    OM_COST_SALDO_END DECIMAL(18,4),
    OM_IS_OS INTEGER,
    OAM_YEAR INTEGER,
    OAM_MONTH INTEGER,
    TAXGROUP INTEGER,
    OL_DATE_OUT DATE,
    KOEF_PERIOD DECIMAL(18,7),
    TAX_BASE_CURR_YEAR DECIMAL(18,4),
    TAX_BASE_NEXT_YEAR DECIMAL(18,4),
    TAX_PERCENT DECIMAL(18,7),
    TAX_IS_OS_STR VARCHAR(255),
    OM_TAX_AMORT_ALL DECIMAL(18,4))
AS
declare variable ol_date_temp date;
declare variable report_os_cost decimal(18,4);
begin
  report_os_cost = 10000;
  taxgroup = 1;
  om_cost_saldo_begin = 0;
  om_cost_saldo_end = 0;
  om_cost_in = 0;
  om_cost_out = 0;
  for select ol_date, oam_from_ol_id, oam_bill_id, oam_bill_line_no, ob_from_ml_id, ob_from_mc_id, ob_invent_str,
             ml_name, mc_name, mc_status, mc_tax_categ, mc_tax_percent
      from operamormain, operlist, operbill, matlist, matcateg
      where ol_date <= :rep_date_end and
            oam_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            ob_bill_id = oam_bill_id and
            ob_bill_line_no = oam_bill_line_no and
            ob_is_delete = 0 and
            oam_is_delete = 0 and
            ob_date_begin <= :rep_date_end and
            ob_date_end >= :rep_date_end and
            ml_id = ob_from_ml_id and
            mc_id = ob_from_mc_id
      into :ol_date, :oam_from_ol_id, :oam_bill_id, :oam_bill_line_no, :ob_from_ml_id, :ob_from_mc_id, :ob_invent_str,
           :ml_name, :mc_name, :mc_status, :taxgroup, :tax_percent
  do
  begin
    koef_period = 0;
    om_cost = 0;
    om_cost_saldo_begin = 0;
    om_cost_saldo_end = 0;
    om_cost_in = 0;
    om_cost_out = 0;
    select om_summa_plus
    from opermat, operlist
    where om_bill_id = om_from_ol_id and
          om_from_ol_id = ol_id and
          operlist.is_delete = 0 and
          om_bill_id = :oam_bill_id and
          om_bill_line_no = :oam_bill_line_no
    into :om_cost;
    if (om_cost < :report_os_cost) then
    begin
      om_is_os = 0;
      tax_is_os_str = 'Налог.аморт: МБП';
    end
    else
    begin
      om_is_os = 1;
      tax_is_os_str = 'Налог.аморт: Осн.средст.';
    end

    select sum(om_summa_plus - om_summa_minus), max(ol_date)
    from opermat, operlist
    where om_bill_id = :oam_bill_id and
          om_bill_line_no = :oam_bill_line_no and
          om_from_ol_id = ol_id and
          operlist.is_delete = 0 and
          operlist.ol_date < :rep_date_begin
    into :om_cost_saldo_begin, :ol_date_temp;
    if (om_cost_saldo_begin is null) then
      om_cost_saldo_begin = 0;

    if ((ol_date >= :rep_date_begin) and
        (ol_date <= :rep_date_end)) then
      om_cost_in = :om_cost;

    om_cost_saldo_end = 0;
    select sum(om_summa_plus - om_summa_minus), max(ol_date)
    from opermat, operlist
    where om_bill_id = :oam_bill_id and
          om_bill_line_no = :oam_bill_line_no and
          om_from_ol_id = ol_id and
          operlist.is_delete = 0 and
          operlist.ol_date <= :rep_date_end
    into :om_cost_saldo_end, :ol_date_temp;

    om_tax_amort_all = 0;

    if ((ol_date_temp >= :rep_date_begin) and
        (ol_date_temp <= :rep_date_end)) then
    begin
      if (om_cost_saldo_end > (om_cost_in + om_cost_saldo_begin)) then
        om_cost_in = om_cost_saldo_end - (om_cost_in + om_cost_saldo_begin);
      else
        om_cost_out = (om_cost_in + om_cost_saldo_begin) - om_cost_saldo_end;
    end
    if (ol_date_temp < :rep_date_begin) then
      om_cost_saldo_begin = om_cost_saldo_end;

    if ((om_cost_out <> 0) or (om_cost_in <> 0)) then
      ol_date_out = :ol_date_temp;
    else
      ol_date_out = null;

    tax_base_curr_year = 0;
    tax_base_next_year = 0;
    oam_year = extract(year from ol_date);
    oam_month = extract(month from ol_date);
    if ((ol_date >= :rep_date_begin) and
        (ol_date <= :rep_date_end)) then
    begin
      if (om_is_os = 1) then
      begin
        koef_period = 12 - oam_month + 0.5;
        tax_base_curr_year = cast(((om_cost_saldo_end / 12) * koef_period) as decimal(15,2));
      end
      else
      begin
        koef_period = 1;
        tax_base_curr_year = om_cost_saldo_end;
      end
      tax_base_next_year = om_cost_saldo_end - tax_base_curr_year;
    end
    else
    begin
      if ((ol_date >= dateadd(-1 year to :rep_date_begin)) and
          (ol_date <= dateadd(-1 year to :rep_date_end))) then
      begin
        if (om_is_os = 1) then
        begin
          koef_period = oam_month - 0.5;
          tax_base_curr_year = cast(((om_cost_saldo_end / 12) * koef_period) as decimal(15,2));
        end
        else
        begin
          koef_period = 1;
          tax_base_curr_year = om_cost_saldo_end;
          tax_base_curr_year = om_cost_saldo_end - tax_base_curr_year;
        end
      end
    end
    if (om_is_os = 0) then
      tax_percent = 100;
    om_tax_amort_all = cast((tax_base_curr_year / 100) * tax_percent as decimal(15,2));

    if ((om_cost_saldo_begin <> 0) or (om_cost_saldo_end <> 0) or (om_cost_in <> 0) or (om_cost_out <> 0)) then
      suspend;
  end
end
