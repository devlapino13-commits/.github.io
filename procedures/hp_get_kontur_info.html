CREATE PROCEDURE HP_GET_KONTUR_INFO(
    EIL_ID D_ID)
RETURNS (
    EIL_DATE D_DATE,
    C_ID D_ID,
    STATUS D_STR_CODE,
    STATUS_DATE D_DATE,
    NEED_CHECK D_BOOL,
    IS_BANKRUPT D_BOOL,
    IS_FINISHED D_BOOL,
    IS_JUDICIAL D_BOOL,
    MESSAGE_ID D_STR_CODE,
    PUB_DATE D_DATE,
    DELO_NO D_STR_255)
AS
declare variable json_str d_str_8100;
declare variable json_extra d_str_8100;
begin
  select cast(blx_blob as d_blob_text), eil_from_c_id, eil_status_datetime
  from extinfolist, extinfocfg, hp_get_blobex_blob(eil_from_bls_id)
  where eic_code in ('kontur', 'kontur_b2')
    and eil_from_eic_id = eic_id
    and eil_status = 2
    and eil_id = :eil_id
    and eil_from_bls_id is not null
  into json_str, c_id, eil_date;

  if (row_count is null or coalesce(json_str, '') = '') then
    exception e_default f_format('Invalid eil_id: %1. No data found', eil_id);

  is_bankrupt = 0;
  need_check = 0;
  is_finished = 0;
  is_judicial = 1;

  message_id = f_json_str_get_value(json_str, 'messageID', null);
  pub_date = left(f_json_str_get_value(json_str, 'pubDate', null), 10); -- 2022-02-18T00:00:00
  json_extra = f_json_str_get_value(json_str, 'extrajudicialBankruptcy', null);
  delo_no = f_json_str_get_value(json_str, 'caseNumber', null);
  if (json_extra is not null) then
  begin
    is_judicial = 0;
    status = f_json_str_get_value(json_extra, 'status', '');
    status_date = left(f_json_str_get_value(json_extra, 'endDate', null), 10); -- 2022-02-18T00:00:00
    if (status_date is null) then
      status_date = left(f_json_str_get_value(json_str, 'lastMessageDate', null), 10); -- 2024-04-02T00:00:00

    if (status in (
        'Возбуждена процедура внесудебного банкротства',
        'Прекращена процедура внесудебного банкротства',
        'Завершена процедура внесудебного банкротства'))
    then
      is_bankrupt = 1;
  end
  else
  begin
    status = f_json_str_get_value(json_str, 'stage', '');
    status_date = left(f_json_str_get_value(json_str, 'stageDate', null), 10);   --  "2023-11-16T00:00:00"
    if (status in (
        'Реструктуризация долгов',
        'Реализация имущества',
        'Реализация имущества завершена',
        'Конкурсное производство завершено',
        'Производство по делу прекращено',
        'Реструктуризация долгов завершена'
        ))
    then
      is_bankrupt = 1;
    else if (status = 'Не удалось определить стадию') then
      need_check = 1;
  end
  if (status in (
      'Реализация имущества завершена',
      'Конкурсное производство завершено',
      'Завершена процедура внесудебного банкротства'))
  then
    is_finished = 1;

  suspend;
end
