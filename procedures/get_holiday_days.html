CREATE PROCEDURE GET_HOLIDAY_DAYS(
    REP_YEAR D_SMALLINT = null)
RETURNS (
    HOLIDAY_DAY D_DATE)
AS
declare variable DATE_BEGIN D_DATE;
declare variable DATE_END D_DATE;
begin
  -- процедура возвращает даты не рабочих дней (выходных + праздничных) по 5-дневному графику
  -- входящие параметры: календарный год - если не передан, то берется текущий год

  if (rep_year is null) then
    rep_year = extract(year from current_date);

  date_begin = '01.01.' || rep_year;
  date_end = '31.12.' || rep_year;

  -- перебираем не рабочие дни отчетного года из sa_calendays (должен быть заполнен до этого)
  -- sa_calendays заполняется кадровиками в начале года и с расчетом 5 рабочих дней в неделю
  for select scd_date
      from sa_calendays
      where scd_date between :date_begin and :date_end and -- дни отчетного года
            scd_work_hours = 0 -- не рабочие дни
      into holiday_day
  do
    suspend;

end
