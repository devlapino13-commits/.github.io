CREATE PROCEDURE HP_PROD_CDM_IGNORE_SHOW(
    PR_ID D_INTEGER,
    PRL_ID D_INTEGER,
    SHOW_CODE D_STR_CODE)
RETURNS (
    CDM_ID D_INTEGER,
    CDM_MEMO D_STR_255,
    CDM_IS_ACTIVE D_INTEGER,
    CDM_OBJECT_KIND D_STR_CODE,
    CHECK_KIND D_STR_255)
AS
declare variable cdm_check_kind d_str_255;
begin
  -- RFCKG-3811 процедура отображает данные для модуля
  -- show_code отражает какую информацию нужно показать
  -- show_code = 'all' показывает все блоки
  -- show_code = 'byProduct' показывает игнорируемые продуктом
  if (show_code = 'all') then
  begin
    for select cdm.cdm_id,
               cdm.cdm_memo,
               cdm.cdm_is_active,
               cdm.cdm_object_kind,
               cdm.cdm_check_kind
        from conditionmain cdm
        order by cdm_id
        into cdm_id,
             cdm_memo,
             cdm_is_active,
             cdm_object_kind,
             cdm_check_kind
    do
    begin
      check_kind = '';

      if (exists(select *
                 from product_condition_ignore pci
                 where pci.pci_from_prl_id = :prl_id and
                       pci.pci_from_pr_id = :pr_id and
                       pci.pci_is_delete = 0 and
                       pci.pci_from_cdm_id = :cdm_id)) then
        cdm_is_active = 0;

      if (cdm_check_kind similar to '%(.loan.)%') then
        check_kind = check_kind || ' в кредите';

      if (cdm_check_kind similar to '%(.check_preview.)%') then
        check_kind = check_kind || ' при просмотре';

      if (cdm_check_kind similar to '%(.loan_approve.)%') then
        check_kind = check_kind || ' при утверждении займа';

      if (cdm_check_kind similar to '%(.send_kk.)%') then
        check_kind = check_kind || ' при отправке на кредитный коммитет';

      if (cdm_check_kind similar to '%(.loan_disb.)%') then
        check_kind = check_kind || ' при выдаче';

      if (cdm_check_kind similar to '%(.miniapp.)%') then
        check_kind = check_kind || ' на уровне кредитной заявки';

      if (cdm_check_kind similar to '%(.create_loan.)%') then
        check_kind = check_kind || ' при сохранении кредитной заявки';

      if (cdm_check_kind similar to '%(.create_loan.)%') then
        check_kind = check_kind || ' при сохранении заявки';
      suspend;
    end
  end

  if (show_code = 'byProduct') then
  begin
    cdm_is_active = 0;

    for select cdm.cdm_id,
               cdm.cdm_memo,
               cdm.cdm_object_kind,
               cdm.cdm_check_kind
        from product_condition_ignore pci
        inner join conditionmain cdm on cdm_id = pci.pci_from_cdm_id
        where pci.pci_from_prl_id = :prl_id and
              pci.pci_from_pr_id = :pr_id and
              pci.pci_is_delete = 0
        order by cdm_id
        into cdm_id,
             cdm_memo,
             cdm_object_kind,
             cdm_check_kind
    do
    begin
      if (cdm_check_kind similar to '%(.loan.)%') then
        check_kind = check_kind || ' в кредите';

      if (cdm_check_kind similar to '%(.check_preview.)%') then
        check_kind = check_kind || ' при просмотре';

      if (cdm_check_kind similar to '%(.loan_approve.)%') then
        check_kind = check_kind || ' при утверждении займа';

      if (cdm_check_kind similar to '%(.send_kk.)%') then
        check_kind = check_kind || ' при отправке на кредитный коммитет';

      if (cdm_check_kind similar to '%(.loan_disb.)%') then
        check_kind = check_kind || ' при выдаче';

      if (cdm_check_kind similar to '%(.miniapp.)%') then
        check_kind = check_kind || ' на уровне кредитной заявки';

      if (cdm_check_kind similar to '%(.create_loan.)%') then
        check_kind = check_kind || ' при сохранении кредитной заявки';

      if (cdm_check_kind similar to '%(.create_loan.)%') then
        check_kind = check_kind || ' при сохранении заявки';
      suspend;
    end
  end
end
