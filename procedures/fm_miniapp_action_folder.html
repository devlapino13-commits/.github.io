CREATE PROCEDURE FM_MINIAPP_ACTION_FOLDER(
    FOLDER_ID D_ID,
    LO_ID D_ID,
    LO_REGION_ID D_INTEGER,
    LANG D_STR_LANG,
    DB_ID D_INTEGER)
RETURNS (
    MA_RESULT_ID D_INTEGER,
    MA_RESULT_TXT D_BLOB_TEXT)
AS
declare variable folder_name varchar(255);
declare variable txt_large d_str_8100;
declare variable prod_name d_str_255;
declare variable client_count d_smallint;
declare variable c_id d_id;
begin
  ma_result_id = 1;
  ma_result_txt = '';

  -- Если папка уже выбрана, то возвращаем только ее
  if (coalesce(folder_id, 0) > 0) then
  begin
    select kal_new_kf_name
    from knaktlist
    where kal_date < kal_date_end and
          kal_date_end >= '31.12.2999' and
          kal_from_kf_id = :folder_id and
          knaktlist.is_delete = 0
    into folder_name;

    ma_result_txt = '{"folders":[{"id":"' || folder_id || '", "name":"' || folder_name || '"}]}';
    suspend;
    exit;
  end

  txt_large = '';

  if (db_id in (39801, 81001, 94401)) then
  begin
    -- Если кредит клиента индивидуальный
    --   То возвращаем его последнюю папку, в которой все выдачи были только этому клиенту (индивидуальные)
    --   Иначе действуем по-старому

    select count(*), min(mac_from_c_id)
    from temp_miniapp_client
    where mac_client_kind <> 'poruch'
    into client_count, c_id;

    if (client_count = 1) then
    begin
      -- Берем все папки клиента у этого СВЗ
      -- Сортируем по дате последней выдачи по убыванию
      -- Для каждой папки проверяем, если по ней не было выдачи групповых, возвращаем ее

      -- Т.к. папки групповых кредитов нам не нужны, сразу отсеиваем
      -- Берем только папки текущего КС в текущем офисе
      for select kal_from_kf_id as folder_id, kal_new_kf_name as folder_name
      from knaktlist, knclient, kontract, product
      where knaktlist.is_delete = 0 and
            kal_new_kf_officer = :lo_id and
            kal_new_reg_id = :lo_region_id and
            kal_date_end >= '31.12.2999' and
            kc_amount_approved > 0 and
            k_from_kf_id = kal_from_kf_id and
            k_status >= 1 and
            k_id = kc_from_k_id and
            kc_from_c_id = :c_id and
            k_from_pr_id = pr_id and
            pr_loan_type = 1
      group by folder_id, folder_name
      order by max(k_disb_date) desc
      into folder_id, folder_name
      do
      begin
        -- Проверяем, если по данной папке не было выдач другим клиентам, устраивает
        if (not exists(
            select * from kontract, knclient
            where
              k_from_kf_id = :folder_id and
              k_status >= 1 and
              k_id = kc_from_k_id and
              kc_from_c_id <> :c_id and
              kc_amount_approved >= 0 )) then
        begin
          ma_result_txt = '{"folders":[{"id":"' || folder_id || '", "name":"' || folder_name || '"}]}';
          suspend;
          exit;
        end
      end

      -- Если не нашли ничего подходящего, разрешаем создать новую папку
      ma_result_txt = f_format('{"folders":[{"id":"0", "name":"%1"}]}', f_lang(lang, 'Новая папка'));
      suspend;
      exit;
    end
  end

  -- Старый вариант (актуален в КР и везде для бывших групповых)
  -- Получаем все папки для
  for with tab
      as (
      -- выданных кредитов
      select distinct kal_from_kf_id as folder_id, kal_new_kf_name as folder_name, kc_from_c_id as c_id
      from knaktlist, knclient, kontract, temp_miniapp_client
      where knaktlist.is_delete = 0 and
            kal_new_kf_officer = :lo_id and
            kal_new_reg_id = :lo_region_id and
            kal_date_end >= '31.12.2999' and
            kc_amount_approved > 0 and
            k_from_kf_id = kal_from_kf_id and
            k_status >= 1 and
            k_id = kc_from_k_id and
            kc_from_c_id = mac_from_c_id and
            mac_client_kind not in ('person', 'consumer', 'poruch')
      union
      -- недавних заявок
      select distinct kal_from_kf_id as folder_id, kal_new_kf_name as folder_name, tc.mac_from_c_id as c_id
      from knaktlist, miniappclient ma, miniapp, temp_miniapp_client tc
      where knaktlist.is_delete = 0 and
            kal_new_kf_officer = :lo_id and
            kal_new_reg_id = :lo_region_id and
            kal_date_end >= '31.12.2999' and
            ma_from_kf_id = kal_from_kf_id and
            ma.mac_amount > 0 and
            (ma_is_active >= 1 or
            (ma_date_oper >= current_date - 35 and
            ma_is_active <= 0)) and
            ma.mac_from_ma_id = ma_id and
            ma_date_end >= '31.12.2999' and
            ma.mac_from_c_id = tc.mac_from_c_id and
            tc.mac_client_kind not in ('person', 'consumer', 'poruch'))

      select folder_id, folder_name
      from tab
      group by folder_id, folder_name
      order by count(*) desc
      into folder_id, folder_name
  do
  begin
    -- Добавляем имена продуктов для найденных папок клиента
    prod_name = '';
    select pr_name
    from kontract, kontstate, product
    where k_from_kf_id = :folder_id and
          k_from_pr_id = pr_id and
          k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1
    order by ks_date_begin desc, ks_from_k_id desc
    rows 1
    into prod_name;
    if (prod_name is null) then
      prod_name = '';

    if (prod_name <> '') then
    begin
      prod_name = replace(prod_name, '"', '\"');
      folder_name = folder_name || ' (' || prod_name || ')';
    end

    txt_large = txt_large || iif(txt_large <> '', ',', '') || '{"id":"' || folder_id || '", "name":"' || folder_name || '"}';
  end

  txt_large = txt_large || iif(txt_large <> '', ',', '') || f_format('{"id":"0", "name":"%1"}', f_lang(lang, 'Новая папка'));
  txt_large = f_format('{"id":"-1", "name":"%1"}', trim(f_lang(lang, 'Не указана'))) || iif(txt_large <> '', ',', '') || txt_large;
  ma_result_txt = '{"folders":[' || txt_large || ']}';

  suspend;
end
