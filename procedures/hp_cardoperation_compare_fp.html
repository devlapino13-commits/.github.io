CREATE PROCEDURE HP_CARDOPERATION_COMPARE_FP(
    CCA_ID D_INTEGER,
    DT_BEGIN D_DATE,
    DT_END D_DATE)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    CREDIT_LIMIT_PROV D_CURRENCY,
    CREDIT_LIMIT_OPER D_CURRENCY,
    AMOUNT_PROV D_CURRENCY,
    AMOUNT_OPER D_CURRENCY,
    AMOUNT_PERC_PROV D_CURRENCY,
    AMOUNT_PERC_OPER D_CURRENCY,
    AMOUNT_OVERP_PROV D_CURRENCY,
    AMOUNT_OVERP_OPER D_CURRENCY)
AS
begin
  for
    select
      c_id,
      c_name_fio
    from cardclientacc
      inner join cardclientid
        on cci_id = cca_from_cci_id
      inner join clients
        on c_id = cci_from_c_id
    where cca_id = :cca_id
      and cca_status = 0
  into
    :c_id,
    :c_name_fio
  do
  begin
    select
      coalesce(sum(iif(ok_type = 8, ok_os, 0)), 0),
      coalesce(-sum(iif((ok_type in (4, 5)
        or (ok_type = 9 and ok_os < 0 and ok_pay_type = 3)) and ok_ol_date between :dt_begin and :dt_end, ok_os + ok_perc + ok_tax, 0)), 0),
      coalesce(-sum(iif((ok_type = 4
        or (ok_type = 9 and ok_os < 0 and ok_pay_type = 3)) and ok_ol_date between :dt_begin and :dt_end, iif(ok_type = 9, ok_os, 0) + ok_perc + ok_tax, 0)), 0),
      coalesce(-sum(iif(ok_type = 9 and ok_os < 0 and ok_ol_date between :dt_begin and :dt_end, ok_os, 0)), 0)
    from operkred
      inner join (
          select
            clo_from_k_id
          from cardlist
            inner join cardlistloan
              on clo_card_id = cal_card_id
          where cal_from_cca_id = :cca_id
          group by clo_from_k_id) as a
        on ok_from_k_id = a.clo_from_k_id
          and ok_type in (8, 4, 5, 9)
          and ok_ol_is_active = 1
          and ok_ol_date <= :dt_end
    into
      :credit_limit_prov,
      :amount_prov,
      :amount_perc_prov,
      :amount_overp_prov;

    select
      coalesce(sum(iif(a.cs_from_csc_id = 7, a.cs_amount, 0)), 0),
      coalesce(sum(iif((a.cs_from_csc_id = 12 or
        (a.cs_from_csc_id = 11
          and not exists(
            select *
            from cardsync as b
              inner join cardsyncopers
                on cso_from_cs_id = b.cs_id
              inner join operlist as o
                on o.ol_id = cso_from_ol_id
                  and o.is_delete = 0
            where b.cs_parent_id = a.cs_id))) and a.cs_date between :dt_begin and :dt_end and is_delete = 0 , a.cs_amount, 0)), 0),
      coalesce(sum(iif(a.cs_from_csc_id = 11 and a.cs_date between :dt_begin and :dt_end and is_delete = 0, a.cs_amount, 0)), 0),
      coalesce(sum(iif(a.cs_from_csc_id = 24 and a.cs_date between :dt_begin and :dt_end, a.cs_amount, 0)), 0)
    from cardsync as a
      left join cardsyncopers
        on cso_from_cs_id = a.cs_id
      left join operlist
        on ol_id = cso_from_ol_id
    where a.cs_from_cca_id = :cca_id
      and a.cs_date <= :dt_end
      and a.cs_from_csc_id in (7, 11, 12, 24)
      and a.cs_status = 2
    into
      :credit_limit_oper,
      :amount_oper,
      :amount_perc_oper,
      :amount_overp_oper;

    suspend;
  end
end
