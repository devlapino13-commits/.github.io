    C_ID ID_TYPE,
    ADD_PARAMS D_STR_LONG = null)

    FIRST_TIME SMALLINT,
    ANK_FILLED_FIRST_TIME_TXT STR_100,
    ANK_UPDATE_DATA_TXT STR_100,
    C_NAME_FIO STR_100,
    RES_NON_RESIDENT INTEGER,
    C_BIRTH_DATE DATE,
    C_BIRTH_CITY STR_100,
    C_NATION_STR STR_20,
    CL_SEX SMALLINT,
    C_CITIZEN_STR STR_20,
    C_FAMILY_STR D_STR_100,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_INN VARCHAR(20),
    C_ADDRESS_YUR STR_255,
    C_ADDRESS_FACT STR_255,
    LAST_CELL_MAIN STR_20,
    C_EMAIL VARCHAR(100),
    C_IS_PUBLIC INTEGER,
    C_IS_BENEFIC_OWNER INTEGER,
    REGISTR_TYPE STR_100,
    REGISTR_DOC_NO STR_100,
    REGISTR_DATE_BEGIN DATE,
    REGISTR_DATE_END DATE,
    REGISTR_ISSUED_BY STR_255,
    REGISTR_MEMO_1 STR_255,
    REGISTR_MEMO_2 STR_255,
    PANT_TYPE STR_100,
    PANT_DOC_NO STR_100,
    PANT_DATE_BEGIN DATE,
    PANT_DATE_END DATE,
    PANT_ISSUED_BY STR_255,
    PANT_MEMO_1 STR_255,
    PANT_MEMO_2 STR_255,
    MAC_LOAN_TARGET_ORIGINAL STR_255,
    C_PFT_RISK_KOEF INTEGER,
    C_PFT_RISK_MEMO VARCHAR(1024),
    PRINT_DATE DATE,
    CREATE_DATE DATE,
    LO_FIO STR_100,
    LO_ID ID_TYPE,
    LO_JL_NAME STR_100,
    MALE_TXT STR_10,
    FEMALE_TXT STR_20,
    ANK_HIGH_RISK_TXT STR_20,
    ANK_LOW_RISK_TXT STR_20,
    C_ORG_TYPE_LONG STR_255,
    C_NAME_FULL STR_255,
    C_ALTERNAT_FIO STR_100,
    C_FORM_STATUS_SHORT STR_20,
    C_OKPO STR_20,
    C_DATETIME TIMESTAMP,
    QR_CODE D_STR_CODE,
    MAC_LOAN_TARGET STR_255,
    KC_LOAN_TARGET_ORIGINAL STR_255,
    C_AUTHORIZED_CAPITAL D_STR_255,
    C_BRANCHES D_STR_255,
    C_DOC_INFORM D_STR_255,
    C_GOALS D_STR_255,
    C_PASS_WHO_CODE D_STR_255,
    REG_NAME VARCHAR(50),
    PL_NAME D_STR_100,
    PANT_INFOLIC1 D_STR_2048,
    PANT_INFOLIC2 D_STR_2048,
    PANT_INFOLIC3 D_STR_2048,
    KML_BANK_NAME VARCHAR(255),
    KML_BANK_NO VARCHAR(50),
    KML_BANK_BIK ID_TYPE,
    KML_POLIS_TYPE STR_100,
    KML_POLIS_DOC_NO STR_100,
    KML_POLIS_DATE_BEGIN DATE,
    KML_POLIS_DATE_END DATE,

    HR_REG_NAME VARCHAR(50),

    KSR_KONT_TITUL VARCHAR(1024),

    PLACE_NAME D_STR_255,
    LAST_YEAR_DATE D_STR_100,
    DOC_NO_LEGAL D_STR_100,
    DOC_DATE_LEGAL D_DATE,
    DOC_ISSUED_BY_LEGAL D_STR_255,
    PK_NAME D_STR_255,
    EDL_BLANK_DATE D_DATE,
    C_TYPE SMALLINT,
    PANT_MEMO3 D_STR_255)
AS
begin
  select first_time, ank_filled_first_time_txt, ank_update_data_txt, c_name_fio, res_non_resident, c_birth_date,
         c_birth_city, c_nation_str, cl_sex, c_citizen_str, c_family_str, c_pass_no, c_pass_date, c_pass_who,
         c_pass_date_expire, c_inn, c_address_yur, c_address_fact, last_cell_main, c_email, c_is_public,
         c_is_benefic_owner, registr_type, registr_doc_no, registr_date_begin, registr_date_end, registr_issued_by,
         registr_memo_1, registr_memo_2, pant_type, pant_doc_no, pant_date_begin, pant_date_end, pant_issued_by,
         pant_memo_1, pant_memo_2, mac_loan_target_original, c_pft_risk_koef, c_pft_risk_memo, print_date, create_date,
         lo_fio, lo_id, lo_jl_name, male_txt, female_txt, ank_high_risk_txt, ank_low_risk_txt, c_org_type_long,
         c_name_full, c_alternat_fio, c_form_status_short, c_okpo, c_datetime, qr_code, mac_loan_target,
         kc_loan_target_original, c_authorized_capital, c_branches, c_doc_inform, c_goals, c_pass_who_code, reg_name,
         pl_name, pant_infolic1, pant_infolic2, pant_infolic3, kml_bank_name, kml_bank_no, kml_bank_bik, kml_polis_type,
         kml_polis_doc_no, kml_polis_date_begin, kml_polis_date_end, kml_polis_issued_by, hr_reg_name, ver_fio, sign_date,
         ksr_kont_titul, edl_create_datetime, edl_id, edl_create_date, place_name, last_year_date, doc_no_legal,
         doc_date_legal, doc_issued_by_legal, pk_name, edl_blank_date, c_type, pant_memo3
  from print_client_cli(:c_id, :add_params)
  into :first_time, :ank_filled_first_time_txt, :ank_update_data_txt, :c_name_fio, :res_non_resident, :c_birth_date,
       :c_birth_city, :c_nation_str, :cl_sex, :c_citizen_str, :c_family_str, :c_pass_no, :c_pass_date, :c_pass_who,
       :c_pass_date_expire, :c_inn, :c_address_yur, :c_address_fact, :last_cell_main, :c_email, :c_is_public,
       :c_is_benefic_owner, :registr_type, :registr_doc_no, :registr_date_begin, :registr_date_end, :registr_issued_by,
       :registr_memo_1, :registr_memo_2, :pant_type, :pant_doc_no, :pant_date_begin, :pant_date_end, :pant_issued_by,
       :pant_memo_1, :pant_memo_2, :mac_loan_target_original, :c_pft_risk_koef, :c_pft_risk_memo, :print_date,
       :create_date, :lo_fio, :lo_id, :lo_jl_name, :male_txt, :female_txt, :ank_high_risk_txt, :ank_low_risk_txt,
       :c_org_type_long, :c_name_full, :c_alternat_fio, :c_form_status_short, :c_okpo, :c_datetime, :qr_code,
       :mac_loan_target, :kc_loan_target_original, :c_authorized_capital, :c_branches, :c_doc_inform, :c_goals,
       :c_pass_who_code, :reg_name, :pl_name, :pant_infolic1, :pant_infolic2, :pant_infolic3, :kml_bank_name,
       :kml_bank_no, :kml_bank_bik, :kml_polis_type, :kml_polis_doc_no, :kml_polis_date_begin, :kml_polis_date_end,
       :kml_polis_issued_by, :hr_reg_name, :ver_fio, :sign_date, :ksr_kont_titul, :edl_create_datetime, :edl_id,
       :edl_create_date, :place_name, :last_year_date, :doc_no_legal, :doc_date_legal, :doc_issued_by_legal, :pk_name,
       :edl_blank_date, :c_type, :pant_memo3;

  -- Использовать данные из PersTemp, если нет измененных персональных данных, то тогда остается из Clients
  if (f_tmp_const('use_perstemp') = '1') then
  begin
    select pte_name_fio, pte_sex, pte_pass_no, pte_pass_date, pte_pass_who,
           pte_pass_date_expire, pte_inn, pte_birth_date, pte_birth_city,
           pte_alternat_fio, pte_pass_who_code
    from perstemp
    where pte_from_c_id = :c_id and
          pte_data_state = 2
    into c_name_fio, cl_sex, c_pass_no, c_pass_date, c_pass_who,
         c_pass_date_expire, c_inn, c_birth_date, c_birth_city,
         c_alternat_fio, c_pass_who_code;
  end

  suspend;
end
