CREATE PROCEDURE FM_HTML_DELINQ_FULL_INFO(
    UN_ID DECIMAL(18,0),
    DATE_END DATE,
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0))
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable ANKETA_CODE varchar(255);
declare variable ANKETA_ID D_INTEGER;
declare variable K_ID decimal(18,0);
declare variable STR varchar(512);
declare variable RES_HTML_TMP blob sub_type 1 segment size 80;
declare variable RES_DATE date;
declare variable RES_DATE_TMP date;
declare variable REPAY_AMOUNT decimal(18,4) = 0;
declare variable OKS_DELINQ_DATE date;
declare variable OKS_DELINQ_DATE_TMP date;
declare variable CL_ID decimal(18,0);
declare variable CM_DATABASE_ID integer;
declare variable OB_KIND D_INTEGER;
declare variable OB_ID D_ID;
declare variable ROUND_VALUE D_SMALLINT;
declare variable UN_FROM_C_ID D_ID;
declare variable KAL_NEW_KF_OFFICER D_ID;
declare variable OA_LO_DENIED_DATETIME D_DATETIME;
declare variable OA_LAST_LO_ID D_ID;
declare variable ACCEPT_APP_STR D_STR_LONG;
declare variable DECLINE_APP_STR D_STR_LONG;
declare variable RES_TEXT D_BLOB_TEXT;
declare variable KS_DATE_BEGIN D_DATE;
declare variable OA_IS_FIRED_LO D_SMALLINT;
declare variable CAN_ACCEPT D_SMALLINT;
declare variable EXP D_INTEGER;
declare variable KS_REGION_ID D_INTEGER;
declare variable PS_FROM_REG_ID D_INTEGER;
declare variable TMP_STR D_STR_LONG;
declare variable MAC_ID D_ID;
declare variable C_NAME_FIO D_STR_100;
declare variable LANG D_STR_LANG;
declare variable MA_ID D_ID;
declare variable KDN D_CURRENCY;
begin
  cm_database_id = f_get_db_id();
  lang = f_get_lang_code(:un_id, null);
  round_value = iif(cm_database_id = 94401, 2, 0);

  select un.un_from_c_id
  from username un
  where un.un_id = :un_id
  into un_from_c_id;

  /* Procedure Text */
  res_html = (select res_style from fm_html_styles) || '<body>';

  if (f_objkind(:obj_kind, 'code') = 'kontract') then
  begin
    -- RFCKZ-558
    if (cm_database_id = 39801) then
    begin
      select cak.cakdn_kdn
      from kontract k
      inner join miniappclient mac on k.k_from_ma_app_id = mac.mac_from_ma_id
        and mac.mac_amount > 0
      inner join cfp_anketa_kdn cak on cak.cakdn_from_mac_id = mac.mac_id
      where k.k_id = :obj_id
      into :kdn;


      res_html = blob_append(res_html, '<h1>Проверка клиента</h1>');

      res_html_tmp = (select res_html from hp_get_html_client_info(9, :obj_id));
      res_html = blob_append(res_html, res_html_tmp);

      res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="60%">');
      res_html = blob_append(res_html, f_html_table_row('КДН', f_format('<span style="color: %1;">%2</span>', iif(kdn > 0.5, 'red', 'black'), coalesce(kdn, 'Нет данных')), '', '', -1, -1, -1, -1, 2, ''));
      res_html = blob_append(res_html, '</table>');
    end

    if (exists
      (
        select *
        from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.LoanTargetChange')
        where has_access = 1
      )
    ) then
    begin
      res_html = blob_append(res_html, '<table width="100%"> <col width="50%"> <col width="50%"> <tr>');

      for select mac.mac_id, cl.c_name_fio
      from kontract kt
      inner join miniappclient mac on mac.mac_from_ma_id = kt.k_from_ma_app_id
        and mac.mac_amount > 0
      inner join clients cl on cl.c_id = mac.mac_from_c_id
      where kt.k_id = :obj_id
      into mac_id, c_name_fio
      do
      begin
        tmp_str = f_uri_open('Изменить цель и отрасль фин-я', 'clp_anketa', f_objkind('miniappclient', 'id'), :mac_id, null, null, :un_id);
        res_html = blob_append(res_html, f_html_table_row(:c_name_fio, :tmp_str, '', '', -1, -1, -1, -1, 2, ''));
      end
      res_html = blob_append(res_html, '</table><br>');
    end
  end

  -- RFCKG-4050
  if (cm_database_id = 41701
      and f_objkind(:obj_kind, 'code') = 'kontract'
      and exists
      (
        select *
        from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.edit_cfp_anketa')
        where has_access = 1
      )
  ) then
  begin
    res_html = blob_append(res_html, f_format('<h2>%1</h2>', 'Анализ доходов/расходов'));
    res_html = blob_append(res_html, '<table width="100%">
           <col width="50%">
           <col width="50%">
           <tr>');

    for select
      mac.mac_id,
      mac.mac_from_c_id
    from kontstate ks
    inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
      and kc.kc_amount_approved > 0
    inner join kontract k on k.k_id = ks.ks_from_k_id
    inner join miniappclient mac on k.k_from_ma_app_id = mac.mac_from_ma_id
      and mac.mac_amount > 0
    where ks.ks_from_k_id = :obj_id
      and ks.ks_ol_is_delete = 0
      and ks.ks_state = 1
      and current_date between ks.ks_date_begin and ks.ks_date_end
    into
      mac_id,
      cl_id
    do
    begin
      if(exists
        (
          select * /* 'Не заполнен ' ||*/
          from cfp_anketa ca
          where ca.ca_object_kind = f_objkind('miniappclient', 'id')
            and ca.ca_object_id = :mac_id
            and ca.ca_from_c_id = :cl_id
        )
      ) then
        -- RFCKG-3249
        res_html = blob_append(res_html, f_html_table_row(:c_name_fio, 'Заполнен<br>' || f_uri_html(f_lang(lang, 'Просмотр'), 'cfp_anketa_info', f_lang(lang, 'Анализ доходов/расходов'), 'miniappclient', :mac_id, current_date, current_date, ''), f_uri_open(f_lang(lang, 'Изменить'), 'cli_financial_profile', f_objkind('miniappclient', 'id'), :mac_id, null, null, :un_id), '', -1, -1, -1, -1, 3, ''));
      else
        res_html = blob_append(res_html, f_html_table_row(:c_name_fio, 'Не заполнен', f_uri_open(f_lang(lang, 'Заполнить'), 'cli_financial_profile', f_objkind('miniappclient', 'id'), :mac_id, null, null, un_id), '', -1, -1, -1, -1, 3, ''));

    end
    res_html = blob_append(res_html, '</table><br>');
  end

  anketa_code = 'anketa.delinq.1';
  anketa_id = f_get_rd_id(anketa_code);

  if (obj_kind = 'clients') then
  begin
    res_date = null;
    oks_delinq_date = null;
    repay_amount = 0;
    k_id = null;
    for select oks_from_k_id, oks_date_begin, oks_delinq_date
        from oksummary, knclient
        where oks_close_state = 0 and
              oks_date_begin <= :date_end and
              oks_date_end >= :date_end and
              oks_balance > 0
--    and oks_delinq_date<:date_end
              and
              oks_from_k_id = kc_from_k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = :obj_id
        into :k_id, res_date_tmp, oks_delinq_date_tmp
    do
    begin
      if (res_date is null) then
        res_date = res_date_tmp;
      if (oks_delinq_date is null) then
        oks_delinq_date = oks_delinq_date_tmp;

      if (res_date <= res_date_tmp) then
      begin
        res_date = maxvalue(res_date, res_date_tmp);
        select sum(-ok_os - ok_perc - ok_fine - ok_tax)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_type in (4, 9) and
              ok_ol_date = :res_date
        into :repay_amount;
        if (repay_amount is null) then
          repay_amount = 0;

        oks_delinq_date = minvalue(oks_delinq_date, oks_delinq_date_tmp);
      end
    end
  end
  else
  begin
    res_date = null;
    oks_delinq_date = null;
    repay_amount = 0;
    k_id = null;
    for select oks_from_k_id, oks_date_begin, oks_delinq_date
        from oksummary
        where oks_close_state = 0 and
              oks_date_begin <= :date_end and
              oks_date_end >= :date_end and
              oks_balance > 0
--    and oks_delinq_date<:date_end
              and
              oks_from_k_id = :obj_id
        into :k_id, res_date_tmp, oks_delinq_date_tmp
    do
    begin
      if (res_date is null) then
        res_date = res_date_tmp;
      if (oks_delinq_date is null) then
        oks_delinq_date = oks_delinq_date_tmp;

      if (res_date <= res_date_tmp) then
      begin
        res_date = maxvalue(res_date, res_date_tmp);
        select sum(-ok_os - ok_perc - ok_fine - ok_tax)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_type in (4, 9) and
              ok_ol_date = :res_date
        into :repay_amount;
        if (repay_amount is null) then
          repay_amount = 0;

        oks_delinq_date = minvalue(oks_delinq_date, oks_delinq_date_tmp);
      end
    end
  end
  if (k_id is not null) then
  begin
    if (cm_database_id in (39801, 41701)) then
    begin
      select oa.oa_lo_denied_datetime, kal.kal_new_kf_officer, oa.oa_last_lo_id, oa.oa_is_fired_lo, ks.ks_date_begin, ks.ks_region_id
      from kontract kt
      inner join kontstate ks on ks.ks_from_k_id = kt.k_id and
            ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_date_begin <= current_date
      inner join knaktlist kal on kal.kal_from_kf_id = kt.k_from_kf_id and
            kal.is_delete = 0 and
            current_date between kal.kal_date and kal.kal_date_end
      inner join expressapp exa on exa.exa_from_k_id = kt.k_id
      inner join online_app oa on oa.oa_from_exa_id = exa.exa_id
      where kt.k_id = :k_id
      into oa_lo_denied_datetime, kal_new_kf_officer, oa_last_lo_id, oa_is_fired_lo, ks_date_begin, ks_region_id;

      can_accept = 0;

      --RFCKG-2559
      if (kal_new_kf_officer = 1000000000102 and
          extract(day from :ks_date_begin) not between 29 and 31 and
          :ks_date_begin <= dateadd(month, 1, cast('01.' || extract(month from :ks_date_begin) || '.' || extract(year from :ks_date_begin ) as date)) - 1) then
        can_accept = 1;

      if (kal_new_kf_officer = 1000000000102 and
          extract(day from :ks_date_begin) between 29 and 31 and
          :ks_date_begin <= dateadd(month, 2, cast('01.' || extract(month from :ks_date_begin) || '.' || extract(year from :ks_date_begin ) as date)) - 1) then
        can_accept = 1;

      select ps_from_reg_id
      from hr_pstatus ps
      inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where ps.ps_from_c_id = :un_from_c_id and
            current_date between ps.ps_date_begin and ps.ps_date_end and
            ps.ps_is_working = 1 and
            ps.ps_status = 1 and
            jk.jk_system_code in ('jobsystem.lo_standart', 'jobsystem.lo_senior')
      into ps_from_reg_id;

      if (coalesce(ps_from_reg_id, 0) <> ks_region_id) then
        can_accept = 0;

      if (can_accept = 1) then
      begin
        --Дополнительные условия для того чтобы взять кредит в свой портфель
        can_accept = 0;

        if (oa_is_fired_lo = 1) then
        begin
          select ovs_value
          from hp_objectprops_get('clients', :un_from_c_id, 'refdata', 'lo_pure_exp', current_date)
          into exp;
        
          if (exp between 4 and 12) then
            can_accept = 1;
        end 
        else
        if (coalesce(oa_is_fired_lo, 0) = 0 and
            (oa_lo_denied_datetime is null and
             un_from_c_id = oa_last_lo_id and
             kal_new_kf_officer = 1000000000102)) then
          can_accept = 1;

        if (can_accept = 1) then
        begin
          accept_app_str = f_uri_sql('Принять', 'accept_online_app_kg', 'kontract', k_id,
          'none', 'Подтвердите что вы хотите принять этот кредит на свой баланс?',
          '', null, null, null, null, 'dialog', null);
        
          decline_app_str = f_uri_sql('Отказаться', 'decline_online_app_kg', 'kontract', k_id,
          'none', 'Вы действительно хотите отказаться от этого кредита?',
          '', null, null, null, null, 'dialog', null);
        
          --Использую переменную res_html_tmp так как она всё равно перезапишется ниже
          res_html_tmp = f_html_table_row(:accept_app_str, :decline_app_str, '', '', 0, 0, 0, 0, 2);
          res_html = BLOB_APPEND(res_html, '<table width="100%"> <col width="50%"><col width="50%"> ', res_html_tmp, '</table><br>');
        end 
      end
    end 

    if (coalesce(oks_delinq_date, '31.12.2999') < date_end) then
    begin
      res_html_tmp = f_format('<font color="red">%1 ', iif(cm_database_id <> 94401, 'Просрочка от', 'Gecikmə')) || coalesce(right('0' || extract(day from oks_delinq_date), 2) || '.' || right('0' || extract(month from oks_delinq_date), 2) || '.' || extract(year from oks_delinq_date), 'хх.хх.хххх') || ' г.</font><br>';
    end
    else
    begin
      res_html_tmp = f_format('<font color="blue">%1</font><br>', iif(cm_database_id <> 94401, 'Нет просрочки', 'Gecikmə yox'));
    end

    res_html_tmp = BLOB_APPEND(res_html_tmp, f_format('<font color="#02B316">%1 ', iif(cm_database_id <> 94401, 'Последний раз платили', 'Sonuncu ödəniş')) || coalesce(right('0' || extract(day from res_date), 2) || '.' || right('0' || extract(month from res_date), 2) || '.' || extract(year from res_date), 'хх.хх.хххх') || f_format(' %1, %2: %3</font>', iif(cm_database_id <> 94401, 'г.', 'il.'), iif(cm_database_id <> 94401, 'сумма', 'məbləğ'), round(repay_amount, :round_value)));

    if (obj_kind = 'clients') then
      cl_id = :obj_id;
    else
      cl_id = 0;
    select res_text
    from hp_html_delinq_anketa(:un_id, :date_end, :k_id, :cl_id)
    into :res_text;
    res_html = BLOB_APPEND(res_html, res_html_tmp, '<br><br>', coalesce(res_text, ''));

    res_html_tmp = '';
    res_date = null;
    if (cm_database_id in (39801, 41701)) then
    begin
      ob_kind = f_objkind('KnClient', 'id');
      for
        select kc_id
        from knclient
        where kc_from_k_id = :k_id
          and exists(
            select *
            from actionlist
            where acl_object_kind = :ob_kind
              and acl_object_id = kc_id
              and acl_from_rd_id = :anketa_id)
      into
        :ob_id
      do
      begin
        for
          select ques_text || ': <b>' || res_text || '</b>', res_date
          from hp_anketa_print(:ob_kind, :ob_id, :anketa_code, null)
        into
          :str,
          :res_date_tmp
        do
        begin
          if (str is null) then
            str = '';
          if (res_date is null) then
            res_date = res_date_tmp;
          if (res_date_tmp is not null) then
            res_date = maxvalue(res_date, res_date_tmp);
          res_html_tmp = BLOB_APPEND(res_html_tmp, str, '<br>');
        end
      end
    end else begin
      if (cl_id <> 0) then
      begin
        ob_kind = f_objkind('Clients', 'id');
        ob_id = cl_id;
        for
          select ques_text || ': <b>' || res_text || '</b>', res_date
          from hp_anketa_print(:ob_kind, :ob_id, :anketa_code, null)
        into
          :str,
          :res_date_tmp
        do
        begin
          if (str is null) then
            str = '';
          if (res_date is null) then
            res_date = res_date_tmp;
          if (res_date_tmp is not null) then
            res_date = maxvalue(res_date, res_date_tmp);
          res_html_tmp = BLOB_APPEND(res_html_tmp, str, '<br>');
        end
      end
    end
    if (res_html_tmp <> '') then
      res_html = BLOB_APPEND(res_html, f_format('<br><font color="#02B316"><b>%1 ',
        iif(cm_database_id <> 94401, 'Инфо от', 'Məlumat tarixi')) || coalesce(right('0' || extract(day from res_date), 2) || '.' || right('0' || extract(month from res_date), 2) || '.' || extract(year from res_date), 'хх.хх.хххх') || '</b></font><br>' || res_html_tmp);

    if (obj_kind = 'kontract') then
    begin
      str = '';
      select first 1 obm_memo
      from hp_objectprops_memo_get('Kontract', :k_id, 'Refdata', 'ObjectProps.Kontract_memo', 1)
      into :str;

      if (str is null) then
        str = '';
      res_html = BLOB_APPEND(res_html, f_format('<br>%1:<br><b>%2</b><br>', iif(cm_database_id <> 94401, 'Примечание специалиста', 'Mütəxəssis qeydi'), :str));
    end

    -- RFCKZ-558
    if(cm_database_id = 39801) then
    begin
      select first 1 kt.k_from_ma_app_id, mac.mac_from_c_id
      from kontract kt
      inner join miniappclient mac on mac.mac_from_ma_id = kt.k_from_ma_app_id
        and mac.mac_amount > 0
      where kt.k_id = :k_id
      into :ma_id, :cl_id;

      res_html = BLOB_APPEND(res_html, '<br><b>' || f_uri_html( 'Отчет КИБ', 'kib_report_kz', 'Отчет КИБ', 'miniapp', :ma_id, date_end, date_end, '{"c_id":' || cl_id || '}' ) || '</b><br>');
    end
  end

  res_html = BLOB_APPEND(res_html, '</body>');
  suspend;
end
