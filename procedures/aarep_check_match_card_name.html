CREATE PROCEDURE AAREP_CHECK_MATCH_CARD_NAME(
    REG_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    K_DISB_DATE DATE,
    KC_AMOUNT DECIMAL(18,4),
    M_FIO VARCHAR(255),
    CARD_INIC VARCHAR(100),
    ENCRYPT_CARD_NUM VARCHAR(30),
    CARD_NO VARCHAR(20),
    K_KONTRACT_NO VARCHAR(20),
    IS_ERR VARCHAR(20),
    KF_ERROR_TEXT VARCHAR(255),
    CL_FIO VARCHAR(100))
AS
declare variable k_id decimal(18,0);
begin
  for select distinct reg_name, ks_date_begin, coalesce(kc_amount, k_amount_approved) as kc_amount, c_name_fio,
                      cdm.kf_client_fio, cdm.kf_client_pass_no, k.k_kontract_no, esm.esm_account,
                      coalesce(kf_error_text, '') as kf_error_text, k.k_id
      from region_list(:reg_id, 1, -1, 1)
      inner join region r on id_list = r.reg_unicode
      inner join kontstate ks on ks_region_id = id_list and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end
      inner join kontract k on ks_from_k_id = k_id
      inner join clients c on k.k_approved_manager = c_id
      inner join extsysmain esm on esm.esm_from_k_id = k.k_id and
            esm.esm_action = 'CardDisb' and
            esm.esm_status = 1
      left join check_match_card_name(ks_from_k_id) cdm on kf_error_text <> ''
      into reg_name, k_disb_date, kc_amount, m_fio, card_inic, encrypt_card_num, k_kontract_no, card_no, kf_error_text, k_id

  do
  begin
    if (kf_error_text <> '') then
      is_err = 'Да';
    else
      is_err = 'Нет';

    for select c.c_name_fio
        from knclient kc
        inner join clients c on kc_from_c_id = c_id
        where :k_id = kc_from_k_id and
              kc_amount_approved > 0
        into :cl_fio
    do
    begin
      suspend;
    end
  end
end
