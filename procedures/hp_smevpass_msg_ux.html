CREATE PROCEDURE HP_SMEVPASS_MSG_UX(
    IN_SBC_ID D_ID)
RETURNS (
    SPM_ID D_ID,
    SPM_KIND D_SMALLINT,
    MSG_STATUS D_SMALLINT,
    MSG_DATETIME D_DATETIME,
    MSG_BODY D_STR_8100)
AS
begin
  for select spm_id, spm_kind, msg_body, msg_datetime, msg_status
  from smevpass_msg, edoc_list, edoc_sign, hp_edoc_client_msg(eds_id)
  where spm_from_sbc_id = :in_sbc_id
    and edl_object_kind = f_objkind('smevpass_msg', 'id')
    and edl_object_id = spm_id
    and eds_from_edl_id = edl_id
  into spm_id, spm_kind, msg_body, msg_datetime, msg_status
  do
    suspend;
end
