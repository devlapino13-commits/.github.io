CREATE PROCEDURE HP_COM_LIST_BLOB(
    CML_ID D_ID,
    UN_ID D_ID)
RETURNS (
    CMB_ID D_INTEGER,
    BLS_ID D_ID,
    BLS_CAPTION D_STR_100,
    BLS_DATETIME D_DATETIME,
    BLS_FROM_UN_FIO D_STR_100)
AS
declare variable bls_from_un_id d_id;
begin
  -- список прикрепленных файлов на запрос КК
  for select cmb_id, cmb_from_bls_id, bls_caption, bls_datetime, bls_from_un_id
  from com_blob, blobstore
  where cmb_from_cml_id = :cml_id and
        cmb_is_delete = 0 and
        cmb_from_bls_id = bls_id
  into cmb_id, bls_id, bls_caption, bls_datetime, bls_from_un_id
  do
  begin
    bls_from_un_fio = '';
    if (bls_from_un_id is not null) then
    begin
      select c_name_fio
      from username, clients
      where un_id = :un_id and
            un_from_c_id = c_id
      into bls_from_un_fio;
    end

    suspend;
  end
end
