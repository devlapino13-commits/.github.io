CREATE PROCEDURE CLIENTSTATUS_CORRECT_WITH_LOG(
    K_ID ID_TYPE)
AS
declare variable c_id id_type;
declare variable cs_star_total integer;
declare variable polez_platej_total integer;
declare variable ol_id decimal(18,0);
declare variable rep_date d_date;
declare variable old_status d_integer;
declare variable old_legitim_pmt d_integer;
begin
  for select ks_from_ol_id, ks_date_begin
      from kontstate
      where ks_ol_is_delete = 0 and
            ks_state = 2 and
            ks_from_k_id = :k_id
      into ol_id, rep_date
  do
  begin
    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id
    do
    begin
      old_status = 0;
      old_legitim_pmt = 0;
      select cs_star_total, cs_polez_platej
      from clientstatus
      where cs_from_k_id = :k_id and
            cs_from_c_id = :c_id and
            cs_date_begin <= :rep_date and
            cs_ol_is_delete = 0
      order by cs_date_begin desc
      rows 1
      into old_status, old_legitim_pmt;
  
      delete from clientstatus
      where cs_from_k_id = :k_id and
            cs_from_c_id = :c_id and
            cs_date_begin >= :rep_date;

      select c_curr_status, polez_platej_total
      from hp_get_client_status(:c_id, :rep_date)
      into cs_star_total, polez_platej_total;

      update or insert into clientstatus (cs_from_c_id, cs_from_k_id, cs_from_ol_id, cs_ol_is_delete, cs_date_begin,
                                          cs_date_end, cs_star_total, cs_polez_platej)
      values (:c_id, :k_id, :ol_id, 0, :rep_date, '31.12.9999', :cs_star_total, :polez_platej_total)
      matching (cs_from_c_id, cs_from_k_id, cs_from_ol_id);
    end
  end
end
