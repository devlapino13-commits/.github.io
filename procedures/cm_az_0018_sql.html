CREATE PROCEDURE CM_AZ_0018_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable pr_id integer;
declare variable disb_date date;
declare variable k_amount curr_type;
declare variable db_id integer;
declare variable min_cl_amount curr_type;
declare variable max_cl_amount curr_type;
declare variable perc_year perc_type;
declare variable kommis_perc perc_type;
declare variable kol_cli integer;
declare variable kol_repay integer;
declare variable c_id_list varchar(255);
declare variable perc_min perc_type;
declare variable perc_max perc_type;
declare variable perc_default perc_type;
declare variable kommis_perc_min perc_type;
declare variable kommis_perc_max perc_type;
declare variable kommis_perc_default perc_type;
begin
  select cm_database_id from company
  into db_id;
  for
  select k_from_pr_id,
         k_disb_date,
         k_amount_approved,
--         round(k_percent_per_day * 365,1),
         k_percent,
         k_kommis_perc,
         k_kolich_client_approved,
         k_kolich_repay
  from kontract
  where k_id = :k_id 
  and k_id <>109010021273300201
  into pr_id,
       disb_date,
       k_amount,
       perc_year,
       kommis_perc,
       kol_cli,
       kol_repay
  do
  begin
    min_cl_amount = -1;
    max_cl_amount = -1;
    if (db_id in (39801, 41701)) then
    begin
--  для КГ базы, логика мин макс суммы из продукта можно вытаскивать
        select prl_amount_cl_min, prl_amount_cl_max
        from prodlogic
        where prl_from_pr_id = :pr_id
        and :disb_date between prl_date_begin and prl_date_end
        rows 1
        into min_cl_amount, max_cl_amount;
    end
    c_id_list = '';
    select list(kc_from_c_id,';') from knclient
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    into c_id_list;
    perc_min = null;
    perc_max = null;
    perc_default = null;
    kommis_perc_min = null;
    kommis_perc_max = null;
    kommis_perc_default = null;
    select perc_min, perc_max, perc_default, kommis_perc_min, kommis_perc_max, kommis_perc_default
    from HP_GET_PERCENT(:pr_id, :disb_date, :kol_cli, :k_amount, :kol_repay, :min_cl_amount, :max_cl_amount, :c_id_list)
    into :perc_min, :perc_max, :perc_default, :kommis_perc_min, :kommis_perc_max, :kommis_perc_default;
    perc_min = coalesce(perc_min,-1);
    perc_max = coalesce(perc_max,-1);
    perc_default = coalesce(perc_default,-1);
    kommis_perc_min = coalesce(kommis_perc_min,-1);
    kommis_perc_max = coalesce(kommis_perc_max,-1);
    kommis_perc_default = coalesce(kommis_perc_default,-1);
    if (perc_year not between perc_min and perc_max
        and perc_year <> round(perc_default,1)) then
    begin
      res_error_text = 'Процент не правильно установлено, предварительно должно быть:'||round(perc_default,1);
      res_error_is_critical = 2;
      suspend;
    end 
    if (kommis_perc not between kommis_perc_min and kommis_perc_max
        and kommis_perc <> round(kommis_perc_default,1)) then
    begin
      res_error_text = 'Доп. процент не правильно установлено, предварительно должно быть:'||round(kommis_perc_default,1);
      res_error_is_critical = 2;
      suspend;
    end
  end
end
