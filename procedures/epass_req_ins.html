CREATE PROCEDURE EPASS_REQ_INS(
    INN D_STR_20,
    PASS D_STR_20,
    UN_ID D_ID,
    C_ID D_ID = null)
RETURNS (
    RES_TEXT D_STR_100)
AS
declare variable er_id d_integer;
declare variable er_timestamp d_datetime;
declare variable inn_tmp d_str_20;
begin
  if (c_id is not null) then
  begin
    select inn
    from fm_get_client_inn(:c_id)
    into :inn_tmp;

    if (:inn <> :inn_tmp and
        :inn_tmp <> '') then
    begin
      res_text = 'У выбранного клиента введенный ИНН не соотвертствует сохраненному.';
      suspend;
      exit;
    end
  end

  select er.er_id, er.er_timestamp
  from epass_req er
  where er.er_inn = :inn and
        er.er_status = 2
  order by 1 desc
  rows 1
  into :er_id, :er_timestamp;

  if (:er_id is not null and
      datediff(day, :er_timestamp, current_timestamp) < 1) then
  begin
    if (exists(select *
               from epass_response erp
               where erp.erp_from_er_id = :er_id)) then
    begin
      res_text = 'С момента прошлого запроса по указанному ИНН прошло менее суток.';
      suspend;
      exit;
    end
    update epass_req
    set er_status = -1
    where (er_id = :er_id);
  end

  if (not exists(select *
                 from epass_req er
                 where er.er_inn = :inn and
                       er.er_status in (0, 1) and
                       datediff(second, er.er_timestamp, current_timestamp) < f_get_const('epass_pass_lifetime'))) then
  begin
    insert into epass_req (er_inn, er_password, er_from_un_id, er_from_c_id)
    values (:inn, :pass, :un_id, :c_id);
    res_text = 'Запрос успешно сформирован.';
  end
  else
    res_text = 'По указанному ИНН запрос находится в обработке.';

  suspend;
end
