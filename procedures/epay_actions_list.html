CREATE PROCEDURE EPAY_ACTIONS_LIST(
    REGION_ID D_INTEGER)
RETURNS (
    ACL_ID D_ID,
    K_ID D_ID,
    C_ID D_ID,
    PROJECT D_STR_100,
    REG_NAME D_STR_100,
    KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    SELLER_FIO D_STR_100,
    KML_TOTAL D_CURRENCY,
    ERROR_TEXT D_STR_255,
    ACL_DATE D_DATE,
    IS_EXPIRED D_STR_100,
    PR_TYPE D_STR_20)
AS
declare variable rd_id d_integer;
declare variable rd_id1 d_integer;
declare variable rd_id2 d_integer;
declare variable rd_id3 d_integer;
declare variable acl_master d_id;
declare variable kf_id d_id;
declare variable reg_id d_integer;
declare variable reg_lst d_str_8100;
declare variable pr_ipf boolean;
declare variable pr_goods boolean;
declare variable pr_sugar boolean;
declare variable rd_id4 d_integer;
declare variable rd_id5 d_integer;
begin
  reg_lst = coalesce((
    select list(id_list, ';')
    from region_list(:region_id, 1, 3, 1)), '');

  rd_id = f_get_rd_id('EPayProjects');
  rd_id1 = f_get_rd_id('AgreementKinds.IPF_Sellers');
  rd_id2 = f_get_rd_id('AgreementKinds.TSP_and_IPF');
  rd_id3 = f_get_rd_id('ClientShops.outlets');
  rd_id4 = f_get_rd_id('AgreementKinds.IPF_Idjara');
  rd_id5 = f_get_rd_id('AgreementKinds.IPF_WithCommission');

  for
    select
      aco_id,
      k_id,
      k_kontract_no,
      k_from_kf_id,
      k_last_reg_id,
      reg_name,
      pr_code containing '.ipf.',
      pr_code containing '.installment_goods.',
      pr_code containing '.sugarmill.'
    from actionorder
      inner join kontract
        on k_id = aco_object_id
      inner join product
        on pr_id = k_from_pr_id
      inner join region
        on reg_unicode = k_last_reg_id
    where aco_from_rd_id = :rd_id
      and aco_object_kind = 9
  into
    :acl_master,
    :k_id,
    :k_kontract_no,
    :kf_id,
    :reg_id,
    :reg_name,
    :pr_ipf,
    :pr_goods,
    :pr_sugar
  do
  begin
    if (reg_lst not containing reg_id)
      then continue;

    pr_type = iif(pr_ipf, 'ИПФ', 'Классика');
    kf_name = (
      select
        kal_new_kf_name
      from knaktlist
      where kal_from_kf_id = :kf_id
        and current_date between kal_date and kal_date_end
        and is_delete = 0);

    for
      select
        acl_id,
        acl_date,
        acl_object_id,
        c_name_fio,
        rd_str_id
      from actionlist
        inner join refdata
          on rd_id = acl_from_rd_id
        inner join clients
          on c_id = acl_object_id
      where acl_parent_id = :acl_master
        and acl_object_kind = 0
        and acl_status not in (-2, 2) -- т.к. нет смысла показывать записи по задачам, где оплата прошла успешно.
    into
      :acl_id,
      :acl_date,
      :c_id,
      :seller_fio,
      :project
    do
    begin
      -- товары в рассрочку
      if (pr_goods) then
        kml_total = coalesce((
          select
            mp_amount * (1 - cs_discount / 100)
          from kontract
            inner join mc_purchase
              on mp_from_ma_id = k_from_ma_app_id
            inner join clientshops
              on cs_id = mp_from_cs_id
                and cs_from_rd_id = :rd_id3
                and cs_is_delete = 0
          where k_id = :k_id), 0);
      else
      -- сах.завод
      if (pr_sugar) then
        kml_total = coalesce((
          select
            spt_amount
          from sugarpaymenttask
          where spt_k_id = :k_id), 0);
      else
      -- ИПФ
      if (pr_ipf) then
        kml_total = coalesce((
          select
            sum(kml_total)-coalesce(sum(kml_commis),0)
          from kontmat
            inner join agreements
              on ag_id = kml_from_ag_id
                and ag_object_id = :c_id
                and ag_object_kind = 0
                and ag_from_rd_id in (:rd_id1, :rd_id2, :rd_id4, :rd_id5)
                and ag_status = 1
          where kml_object_kind = 9
            and kml_object_id = :k_id
            and kml_status = 1), 0);



      error_text = coalesce((
        select
          kpt_error_text
        from kontpaytasks kpt
        where kpr_from_acl_id = :acl_id
        order by kpt.kpt_id desc
        rows 1), '');

      if (error_text containing 'HTTP/1.1 500') then
        error_text = 'Временная ошибка сервиса KICB';
      if (error_text containing 'rule.exceedMaxLimitAmountDestination') then
        error_text = 'Превышен лимит по сумме за 1 транзакцию';
      if (error_text containing 'Customer status does not allow the transaction') then
        error_text = 'Пользователь заблокирован в системе ЭЛСОМ';
      if (error_text containing 'rule.exceedMonthlyLimitAmountDestination') then
        error_text = 'Превышен месячный лимит';

      if (current_date - acl_date >= 3) then
        is_expired = 'Просрочен';
      else
        is_expired = '';

      suspend;
    end
  end
end
