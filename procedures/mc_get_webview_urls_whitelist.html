CREATE PROCEDURE MC_GET_WEBVIEW_URLS_WHITELIST
RETURNS (
    INTERNAL_URLS D_STR_255)
AS
declare variable db_id d_integer;
declare variable temp_urls d_str_long;
begin
    -- RFCAZ-612
    db_id = f_get_db_id();

    -- Общие хоста
    temp_urls = 'mbulak://; fsc://; ';

    -- Домены только для КГ
    if (db_id = 41701) then
        temp_urls = temp_urls || 'mbulak.kg; pay.mbulak.kg; ';

    -- Домены только для КЗ
    if (db_id = 39801) then
        temp_urls = temp_urls || 'processing.kz; test.processing.kz; fcb-bio-guard-widget.1cb.kz; mbulak.kz; cdn.mbulak.kz; ';

    -- Домены только для РФ
    if (db_id = 81001) then
        temp_urls = temp_urls || 'moneta.ru; moneta.com; payanyway.ru; payanyway.com; cbrpay.ru; nspk.ru; wirebank.ru; molbulak.ru; mbulak.ru; edoc.mbulak; demo.moneta.ru; ';

    -- Домены только для АЗ
    if (db_id = 94401) then
        temp_urls = temp_urls || 'mbulak.az; ';

    -- Делим и возвращаем каждый домен/хост по отдельности
    for select res_str
    from hp_split_str(iif(char_length(:temp_urls) > 2, left(:temp_urls, char_length(:temp_urls) - 2), ''), '; ')
    into :internal_urls
    do
        suspend;
end
