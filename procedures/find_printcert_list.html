CREATE PROCEDURE FIND_PRINTCERT_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PRC_ID TYPE OF COLUMN PRINTCERT.PRC_ID,
    PRC_DATE TYPE OF COLUMN PRINTCERT.PRC_DATE,
    PRC_KIND TYPE OF COLUMN PRINTCERT.PRC_KIND,
    PRC_NO TYPE OF COLUMN PRINTCERT.PRC_NO,
    PRC_FROM_REG_ID TYPE OF COLUMN PRINTCERT.PRC_FROM_REG_ID,
    PRC_FROM_K_ID TYPE OF COLUMN PRINTCERT.PRC_FROM_K_ID,
    PRC_FROM_C_ID TYPE OF COLUMN PRINTCERT.PRC_FROM_C_ID,
    PRC_FROM_UN_ID TYPE OF COLUMN PRINTCERT.PRC_FROM_UN_ID,
    PRC_DATETIME TYPE OF COLUMN PRINTCERT.PRC_DATETIME,
    PRC_IS_ACTIVE TYPE OF COLUMN PRINTCERT.PRC_IS_ACTIVE,
    PRC_FROM_BLN_ID TYPE OF COLUMN PRINTCERT.PRC_FROM_BLN_ID,
    PRC_SIGN_BY_C_ID TYPE OF COLUMN PRINTCERT.PRC_SIGN_BY_C_ID,
    PRC_STOP_CALC_PERC TYPE OF COLUMN PRINTCERT.PRC_STOP_CALC_PERC,
    PRC_STOP_CALC_FINE TYPE OF COLUMN PRINTCERT.PRC_STOP_CALC_FINE,
    PRC_POCKET_ID TYPE OF COLUMN PRINTCERT.PRC_POCKET_ID,
    PRC_FOR_WHOM TYPE OF COLUMN PRINTCERT.PRC_FOR_WHOM,
    K_KONTRACT_NO VARCHAR(20),
    C_NAME_FIO VARCHAR(100),
    REG_NAME VARCHAR(100),
    PRC_CLIENT_KOL INTEGER,
    BLANK_IS_READONLY INTEGER,
    BLANK_SHOW_MODE INTEGER)
AS
begin
  for select prc_id, prc_date, prc_kind, prc_no, prc_from_reg_id, prc_from_k_id, prc_from_c_id, prc_from_un_id,
             prc_datetime, prc_is_active, prc_from_bln_id, prc_sign_by_c_id, prc_stop_calc_perc, prc_stop_calc_fine,
             prc_pocket_id, prc_for_whom, reg_code || ' ' || reg_name
      from printcert, region_list(:rep_region_id, 1, -1, 1), region
      where prc_from_reg_id = id_list and
            prc_date between :rep_date_begin and :rep_date_end and
            prc_from_reg_id = reg_unicode
      into :prc_id, :prc_date, :prc_kind, :prc_no, :prc_from_reg_id, :prc_from_k_id, :prc_from_c_id, :prc_from_un_id,
           :prc_datetime, :prc_is_active, :prc_from_bln_id, :prc_sign_by_c_id, :prc_stop_calc_perc, :prc_stop_calc_fine,
           :prc_pocket_id, :prc_for_whom, :reg_name
  do
  begin
    k_kontract_no = '';
    c_name_fio = 'Группа';
    prc_client_kol = 0;
    if (prc_from_k_id is not null) then
    begin
      select k_kontract_no, k_kolich_client_approved
      from kontract
      where k_id = :prc_from_k_id
      into :k_kontract_no, prc_client_kol;
    end
    if (prc_from_c_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :prc_from_c_id
      into :c_name_fio;
      prc_client_kol = 1;
    end
    /*
    blank_is_readonly = (select bln_is_read_only
                         from blankname
                         where bln_id = :prc_from_bln_id);
    */
    select
      bln_is_read_only,
      iif(blb_frx_status = 1 and blb_frx_blob is not null, 0, bln_show_mode)
    from blankname
      inner JOIN blankblob
        on blb_from_bln_id = bln_id
          and current_date between blb_date_begin and blb_date_end
    where bln_id = :prc_from_bln_id
    into
      :blank_is_readonly,
      :blank_show_mode;

    suspend;
  end
end
