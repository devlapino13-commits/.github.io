CREATE PROCEDURE HP_TABEL_AUTO_UPDATE(
    REG_ID D_INTEGER,
    YEAR_INP D_INTEGER,
    MONTH_INP D_INTEGER,
    C_ID D_ID = 0,
    CUSTOM_WEEKEND_LIST D_STR_100 = null)
AS
declare variable first_date d_date;
declare variable last_date d_date;
declare variable hrv_date_begin d_date;
declare variable hrv_date_end d_date;
declare variable ho_order_type d_integer;
declare variable year_month d_str_10;
declare variable weekend_list d_str_100;
declare variable is_custom_weekend d_bool;
begin
  first_date = '01.' || month_inp || '.' || year_inp; -- начало отчетного месяца
  last_date = f_date_month_end(first_date); -- конец отчетного месяца
  year_month = year_inp * 100 + month_inp; -- отчетный месяц в формате yyyymm

  select list(scal_day, '.') || '.'
  from sa_calendar
  where scal_active = 1 and
        scal_month = :month_inp
  into weekend_list; -- праздничные дни

  is_custom_weekend = 0;
  if (custom_weekend_list <> '') then
  begin
    weekend_list = custom_weekend_list; -- отредактированные не рабочие дни
    is_custom_weekend = 1;
  end

  if (:c_id > 0) then
  begin
    delete from sa_tabelo
    where sat_year_month = :year_month and
          sat_office_id = :reg_id and
          sat_from_c_id = :c_id;

    -- строка "Отработано"
    for select iif(ps_date_begin < :first_date, :first_date, ps_date_begin),
               iif(ps_date_end > :last_date, :last_date, ps_date_end)
        from hr_pstatus
        where ps_status = 1 and -- не аннулированные
              ps_from_reg_id = :reg_id and -- фильтр по офису
              ps_from_c_id = :c_id and -- фильтр по сотруднику
              ps_is_working = 1 and -- работающие в штате офиса
              (ps_date_begin between :first_date and :last_date or -- начало работы в отчетном месяце
              :first_date between ps_date_begin and ps_date_end) -- или начало отчетного месяца внутри периода работы
        into hrv_date_begin,
             hrv_date_end
    do
      execute procedure hp_insert_tabel_by_dates(:year_month, :c_id, 0, :reg_id, :hrv_date_begin, :hrv_date_end, :weekend_list, is_custom_weekend);

    -- строки отпусков: трудовой; неоплачиваемый; по беременности и родам; по уходу за ребенком без содержания; декретный
    for select ho.ho_order_type,
               iif(hrv.hrv_date_begin < :first_date, :first_date, hrv.hrv_date_begin),
               iif(hrv.hrv_date_end_fact > :last_date, :last_date, hrv.hrv_date_end_fact)
        from hr_orders ho
        inner join hr_voctravel hrv on ho.ho_id = hrv.hrv_from_ho_id
        where ho.ho_status = 1 and -- утвержденные приказы
              ho.ho_order_type in (7, 8, 17, 18, 24) and -- нужные виды отпусков
              hrv.hrv_from_c_id = :c_id and -- фильтр по сотруднику
              (hrv.hrv_date_begin between :first_date and :last_date or -- начало отпуска в отчетном месяце
              :first_date between hrv.hrv_date_begin and hrv.hrv_date_end) -- или начало отчетного месяца внутри периода отпуска
        into ho_order_type, -- тип приказа
             hrv_date_begin, -- начало отпуска
             hrv_date_end -- конец отпуска
    do
      execute procedure hp_insert_tabel_by_dates(:year_month, :c_id, :ho_order_type, :reg_id, :hrv_date_begin, :hrv_date_end, :weekend_list, is_custom_weekend);
  end
  else
  begin
    delete from sa_tabelo
    where sat_year_month = :year_month and
          sat_office_id = :reg_id;

    -- сотрудники офиса, строка "Отработано"
    for select ps_from_c_id,
               iif(ps_date_begin < :first_date, :first_date, ps_date_begin),
               iif(ps_date_end > :last_date, :last_date, ps_date_end)
        from hr_pstatus
        where ps_status = 1 and -- не аннулированные
              ps_from_reg_id = :reg_id and -- фильтр по офису
              ps_is_working = 1 and -- работающие в штате офиса
              (ps_date_begin between :first_date and :last_date or -- начало работы в отчетном месяце
              :first_date between ps_date_begin and ps_date_end) -- или начало отчетного месяца внутри периода работы
        into :c_id,
             hrv_date_begin,
             hrv_date_end
    do
    begin
      execute procedure hp_insert_tabel_by_dates(:year_month, :c_id, 0, :reg_id, :hrv_date_begin, :hrv_date_end, :weekend_list, is_custom_weekend);

      -- строки отпусков: трудовой; неоплачиваемый; по беременности и родам; по уходу за ребенком без содержания; декретный
      for select ho.ho_order_type,
                 iif(hrv.hrv_date_begin < :first_date, :first_date, hrv.hrv_date_begin),
                 iif(hrv.hrv_date_end_fact > :last_date, :last_date, hrv.hrv_date_end_fact)
          from hr_orders ho
          inner join hr_voctravel hrv on ho.ho_id = hrv.hrv_from_ho_id
          where ho.ho_status = 1 and -- утвержденные приказы
                ho.ho_order_type in (7, 8, 17, 18, 24) and -- нужные виды отпусков
                hrv.hrv_from_c_id = :c_id and -- фильтр по сотруднику
                (hrv.hrv_date_begin between :first_date and :last_date or -- начало отпуска в отчетном месяце
                :first_date between hrv.hrv_date_begin and hrv.hrv_date_end) -- или начало отчетного месяца внутри периода отпуска
          into ho_order_type, -- тип приказа
               hrv_date_begin, -- начало отпуска
               hrv_date_end -- конец отпуска
      do
        execute procedure hp_insert_tabel_by_dates(:year_month, :c_id, :ho_order_type, :reg_id, :hrv_date_begin, :hrv_date_end, :weekend_list, is_custom_weekend);
    end
  end
end
