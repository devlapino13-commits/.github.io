CREATE PROCEDURE UXP$PREBLANKS_SAVE(
    PREBLANK$ACTION_ID D_STR_CODE,
    PREBLANK$KIND D_STR_CODE,
    PREBLANK$CODE D_STR_CODE,
    IN_OBJ_KIND D_STR_CODE,
    IN_OBJ_ID D_STR_CODE,
    REP_REGION_ID D_INTEGER = null)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable kind_id d_integer;
declare variable save_kind d_str_code;
declare variable ua_preblank_save d_integer;
declare variable obj_kind d_str_code;
declare variable blank_count d_integer;
declare variable oml_subject d_str_long;
declare variable oml_memo d_str_8100;
declare variable oml_is_delinq d_bool;
declare variable c_id d_id;
declare variable oml_id d_integer;
declare variable client_address d_str_255;
declare variable k_id d_id;
declare variable delinq_days d_integer;
declare variable un_id d_id;
declare variable bls_id d_id;
declare variable bln_name d_str_255;
declare variable oml_regist_no d_str_code;
declare variable kind_id_str d_str_code;
declare variable edoc_code d_str_code;
declare variable pre_oml_id d_integer;
begin
  res_is_error = 1;
  res_text = '';

  un_id = uxf$user();

  preblank$code = trim(coalesce(preblank$code, ''));

  kind_id = f_enum('uxaction_preblank_kind', preblank$kind, 'id');
  if (kind_id is null) then
    kind_id = 0;

  if (preblank$action_id is null) then
    preblank$action_id = '';

  select ua_preblank_save
  from ux$action
  where ua_uuid = char_to_uuid(:preblank$action_id)
  into ua_preblank_save;

  if (ua_preblank_save = 0) then
    res_text = 'Отсутствует конфигурация сохранения';
  else
  begin
    save_kind = f_enum('uxaction_preblank_save', ua_preblank_save, 'code');
    if (save_kind = 'edoc') then
    begin
      kind_id_str = f_enum('uxaction_preblank_kind', preblank$kind, 'code');

      edoc_code = '';
      if (kind_id_str = 'hr') then
        edoc_code = 'staff.orders';

      if (edoc_code = '') then
        res_text = 'Настройки эл.документа не найдены, обратитесь к разработчикам';
      else
      begin
        select res_text
        from hp_edoc_insert(:edoc_code, 'hr_orders', :in_obj_id, :un_id)
        into :res_text;
      end

      if (res_text = '') then
        res_is_error = 0;
    end else
    if (save_kind = 'outmailing') then
    begin
      obj_kind = f_objkind(in_obj_kind, 'name');

      select count(*), max(bln_name), list(bln_name), max(coalesce(prb_from_oml_id, 0))
      from ux$preblanks, blankname
      where prb_kind = :kind_id and
            prb_code = :preblank$code and
            prb_prepared = 1 and
            prb_selected = 1 and
            prb_blank_blob is not null and
            prb_blank_bln_id = bln_id
      into blank_count, oml_subject, oml_memo, pre_oml_id;

      if (pre_oml_id = 0) then
        pre_oml_id = null;

      if (oml_subject is null) then
        oml_subject = '';

      if (oml_memo is null) then
        oml_memo = '';

      if (blank_count > 0) then
      begin
        if (blank_count = 1) then
          oml_memo = '';
        else
          oml_subject = 'Уведомления';

        oml_is_delinq = 1;
  
        if (obj_kind = 'kontract') then
        begin
          oml_is_delinq = 0;
          k_id = in_obj_id;

          delinq_days = 0;
          select datediff(day, oks_delinq_date, current_date), iif(:rep_region_id is null, oks_region_id, :rep_region_id)
          from oksummary
          where oks_from_k_id = :k_id and
                oks_date_begin <= current_date and
                oks_date_end >= current_date
          into delinq_days, rep_region_id;

          if (delinq_days > 0) then
          begin
            oml_is_delinq = 1;

            for select kc_from_c_id
            from knclient
            where kc_from_k_id = :k_id and
                  kc_amount_approved > 0
            into c_id
            do
            begin
              client_address = '';
              select adl_address_str
              from addrclient, addrlist
              where adc_from_c_id = :c_id and
                    adc_address_type = 1 and
                    adc_from_adl_id = adl_id
              into client_address;

              oml_id = null;
              if (pre_oml_id is not null) then
              begin
                update outmailing
                set oml_is_out = 1,
                    oml_group_kind = f_enum('outmailing_group_kind', 'office', 'id'),
                    oml_from_reg_id = :rep_region_id,
                    oml_from_c_id = :c_id,
                    oml_subject = :oml_subject,
                    oml_memo = :oml_memo,
                    oml_is_delinq = :oml_is_delinq
                where oml_id = :pre_oml_id
                returning oml_id, oml_regist_no
                into :oml_id, oml_regist_no;

                pre_oml_id = null;
              end

              if (oml_id is null) then
              begin
                insert into outmailing (oml_is_out, oml_date, oml_group_kind,
                                        oml_from_reg_id, oml_from_c_id, oml_subject, oml_memo, oml_is_delinq)
                values (1, current_date, f_enum('outmailing_group_kind', 'office', 'id'), :rep_region_id,
                        :c_id, :oml_subject, :oml_memo, :oml_is_delinq)
                returning oml_id, oml_regist_no
                into :oml_id, oml_regist_no;
              end

              insert into outmailkont (omk_from_oml_id, omk_from_k_id, omk_is_delinq)
              values (:oml_id, :in_obj_id, :oml_is_delinq);
  
              insert into outmailaddr (oma_from_oml_id, oma_kind, oma_address)
              values (:oml_id, f_enum('outmailing_address_kind', 'client', 'id'), :client_address);

              for select prb_blank_bls_id, bln_name
              from ux$preblanks, blankname
              where prb_kind = :kind_id and
                    prb_code = :preblank$code and
                    prb_prepared = 1 and
                    prb_selected = 1 and
                    prb_blank_bls_id is not null and
                    prb_blank_bln_id = bln_id
              into bls_id, bln_name
              do
              begin
                insert into outmailfile (omf_from_oml_id, omf_name, omf_from_bls_id, omf_from_un_id)
                values (:oml_id, :bln_name, :bls_id, :un_id);
              end
            end

            delete from outmailnum
            where 1 = 1;

            res_is_error = 0;
            res_text = 'Бланки успешно сохранены, исх.№ ' || oml_regist_no;
          end else
          begin
            res_text = 'Уведомления доступны только по просроченным займам';
          end
        end else
        begin
          res_text = 'Неопределенный объект: ' || obj_kind;
        end
      end else
      begin
        res_text = 'Не найдены приготовленные бланки для сохранения';
      end
    end else
    begin
      res_text = 'Неопреденный метод сохранения: ' || save_kind;
    end
  end

  suspend;
end
