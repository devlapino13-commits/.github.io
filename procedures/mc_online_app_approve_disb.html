CREATE PROCEDURE MC_ONLINE_APP_APPROVE_DISB(
    MCL_ID D_ID,
    ONLINE_APP_ID D_INTEGER)
RETURNS (
    SUCCESS_HEADER D_STR_CODE,
    SUCCESS_TEXT D_STR_255,
    ERROR_TEXT D_STR_255)
AS
declare variable EXA_ID D_ID;
declare variable MA_ID D_ID;
declare variable K_ID D_ID;
declare variable MA_EDL_ID D_ID;
declare variable K_EDL_ID D_ID;
declare variable RESULT D_SMALLINT;
declare variable C_ID D_ID;
declare variable CARD_ID D_ID;
declare variable AMOUNT D_INTEGER;
declare variable APP_JSON D_STR_8100;
declare variable APP_AMOUNT D_INTEGER;
declare variable OA_ID D_INTEGER;
declare variable OA_DATE date;
declare variable ERROR_TEXT_TMP D_STR_255;
declare variable MCL_LANG_CODE D_STR_LANG;
begin
  error_text = '';

  select mcl.mcl_from_c_id, mcl.mcl_lang_code
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into c_id,
       mcl_lang_code;

  if (coalesce(:c_id, -999) <> -999) then
    select cal_card_id
    from get_active_mbulak_card(:c_id)
    into card_id;

  if (coalesce(:card_id, -999) = -999) then
  begin
    success_header = '';
    success_text = '';
    error_text = f_lang(:mcl_lang_code, 'Отсутствует активная карта Мбулак');
  end

  select amount
  from mc_get_loan_limit(:mcl_id)
  into amount;

  select oa.oa_json,
         oa.oa_date
  from online_app oa
  where oa.oa_from_mcl_id = :mcl_id and
        oa.oa_status >= 0
  order by oa.oa_datetime desc
  rows 1
  into app_json,
       oa_date;

  if (oa_date <> current_date) then
  begin
    success_header = '';
    success_text = '';
    error_text = f_lang(:mcl_lang_code, 'Заявка была содана не сегодня. Она не может быть взята в обработку');
  end

  app_amount = f_json_blob_get_value(:app_json, 'amount', 0);

  if (coalesce(:amount, 0) = 0) then
  begin
    success_header = '';
    success_text = '';
    error_text = f_lang(:mcl_lang_code, 'Выдача невозможна. Запрашиваемая сумма (') || :app_amount || f_lang(:mcl_lang_code, ' сом) превышает доступный кредитный лимит ') || :amount || f_lang(:mcl_lang_code, ' сом).');
  end
  else
  begin
    if (:app_amount > :amount) then
    begin
      success_header = '';
      success_text = '';
      error_text = f_lang(:mcl_lang_code, 'Выдача невозможна. Запрашиваемая сумма (') || :app_amount || f_lang(:mcl_lang_code, ' сом) превышает доступный кредитный лимит ') || :amount || f_lang(:mcl_lang_code, ' сом).');
    end
  end

  if (error_text = '') then
  begin
    select oa.oa_from_exa_id,
           oa.oa_id
    from online_app oa
    where oa.oa_id = :online_app_id
    into exa_id,
         oa_id;

    select exa.exa_from_ma_id, exa.exa_from_k_id
    from expressapp exa
    where exa.exa_id = :exa_id
    into ma_id, k_id;

    select edl.edl_id
    from edoc_list edl
    where edl.edl_object_kind = f_objkind('miniapp', 'id') and
          edl.edl_object_id = :ma_id and
          edl.edl_from_edc_id = f_edoc_kind('loan.application', 'id') and
          edl.edl_status = 0
    into ma_edl_id;

    select edl.edl_id
    from edoc_list edl
    where edl.edl_object_kind = f_objkind('kontract', 'id') and
          edl.edl_object_id = :k_id and
          edl.edl_from_edc_id = f_edoc_kind('loan.agreement', 'id') and
          edl.edl_status = 0
    into k_edl_id;

    update edoc_sign eds
    set eds.eds_status = 1,
        eds.eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'mobile', 'id')
    where eds.eds_from_edl_id in (:ma_edl_id, :k_edl_id);

    execute procedure hp_online_loan_do_process(:online_app_id, :exa_id, 1);

    select exa.exa_final_status
    from expressapp exa
    where exa.exa_id = :exa_id
    into result;

    if (result = 2) then --Всё прошло успешно
    begin
      success_header = f_lang(:mcl_lang_code, 'Вы успешно оформили кредит');
      success_text = f_lang(:mcl_lang_code, 'Деньги будут отправлены на вашу карту Элкарт. Это может занять до 5 минут.');
      error_text = '';
    end
    else
    begin
      success_header = '';
      success_text = '';
      error_text = f_lang(:mcl_lang_code, 'Произошла ошибка. Просьба обратиться в ближайший офис компании.');

      select error_text
      from mc_online_app_cancel(:mcl_id, :online_app_id)
      into :error_text_tmp;
    end

    update online_app oa
    set oa.oa_status = 3
    where oa.oa_id = :oa_id;
  end
  else
  begin
     select error_text
     from mc_online_app_cancel(:mcl_id, :online_app_id)
     into :error_text_tmp;
  end
  suspend;
end
