CREATE PROCEDURE HP_CLIENTSHOPS_VIEW(
    RD_CODES D_BLOB_TEXT,
    ID D_ID = null)
RETURNS (
    CS_ID D_ID,
    CS_FROM_C_ID D_ID,
    C_NAME_FIO D_STR_100,
    CS_NAME D_STR_255,
    CS_NAME_LAT D_STR_255,
    CS_FROM_ADL_ID D_ID,
    ADL_ADDRESS_STR D_STR_255,
    CS_LATITUDE D_GEO,
    CS_LONGITUDE D_GEO,
    CS_CONTACT_FIO D_STR_100,
    CS_PHONE D_STR_20,
    PHONE_ENC D_STR_20,
    CS_HOURS D_STR_20,
    CS_WEEKEND_HOURS D_STR_20,
    CS_DATETIME D_DATETIME,
    CS_FROM_UN_ID D_ID,
    UN_FIO D_STR_100,
    CS_QR_DATA D_STR_2048,
    QR_ASSIGNED D_BOOL,
    CS_IPC_ID D_ID,
    TRM_BINDING D_BOOL,
    CS_IS_DELETE D_BOOL,
    OFFICE_NAME D_STR_CODE,
    REG_NAME D_STR_CODE,
    IQA_ID D_ID,
    CS_FROM_RD_ID D_INTEGER,
    CS_FROM_AL_ID D_STR_20,
    CS_FROM_REG_ID D_INTEGER,
    CS_DISCOUNT D_PERCENT,
    AL_ACCOUNT VARCHAR(32),
    INVENT_NUMBER D_STR_100)
AS
declare variable phone_len d_integer;
declare variable obj_kind d_integer;
declare variable reg_id d_integer;
declare variable dt_curr d_date;
declare variable rd_list d_blob_text;
declare variable rd_code d_str_255;
declare variable rd_id d_integer;
declare variable rd_id_tsp d_integer;
declare variable rd_id_ag_tsp d_integer;
begin
  rd_list = '.';
  if (:rd_codes not containing ';') then
    rd_codes = ';' || :rd_codes || ';';
  for select value_out
      from str_with_separator_parser(:rd_codes, ';')
      into :rd_code
  do
  begin
    rd_id = f_get_rd_id(:rd_code);
    rd_list = blob_append(rd_list, :rd_id, '.');
    if (:rd_code = 'ClientShops.tsp') then
      rd_id_ag_tsp = f_get_rd_id('AgreementKinds.tsp');
    rd_id_tsp = f_get_rd_id('ClientShops.tsp');
  end

  obj_kind = f_objkind('ClientShops', 'id');
  dt_curr = current_date;

  for select cs_id, cs_from_c_id, c_name_fio, cs_name, cs_name_lat, cs_from_adl_id, cs_latitude, cs_longitude,
             adl_address_str, cs_contact_fio, cs_phone, cs_hours, cs_weekend_hours, cs_datetime, cs_from_un_id, un_fio,
             cs_qr_data, cs_ipc_id, cs_is_delete, cs_from_rd_id, cs_from_al_id, cs_from_reg_id, cs_discount,
             al_account
      from clientshops
      inner join clients on c_id = cs_from_c_id
      inner join username on un_id = cs_from_un_id
      left join addrlist on adl_id = cs_from_adl_id
      left join acc_list on cs_from_al_id = al_id
      where cs_id = coalesce(:id, cs_id) and
            (:rd_list containing '.' || cs_from_rd_id || '.' or
            cs_from_rd_id is null)
      into :cs_id, :cs_from_c_id, :c_name_fio, :cs_name, :cs_name_lat, :cs_from_adl_id, :cs_latitude, :cs_longitude,
           :adl_address_str, :cs_contact_fio, :cs_phone, :cs_hours, :cs_weekend_hours, :cs_datetime, :cs_from_un_id,
           :un_fio, :cs_qr_data, :cs_ipc_id, :cs_is_delete, :cs_from_rd_id, :cs_from_al_id, :cs_from_reg_id,
           :cs_discount, :al_account
  do
  begin
    if (cs_id = 0) then
      reg_id = f_hq_id();
    else
    begin
      if (cs_from_rd_id = rd_id_tsp) then
        reg_id = (select ag_from_init_reg_id
                  from agreements
                  where ag_object_kind = 0 and
                        ag_object_id = :cs_from_c_id and
                        ag_from_rd_id = :rd_id_ag_tsp and
                        ag_status = 1 and
                        :dt_curr between coalesce(ag_date_begin, '01.01.1900') and coalesce(ag_date_end, '31.12.9999') and
                        ag_from_init_reg_id > 0
                  order by ag_id desc
                  rows 1);
      else
        reg_id = :cs_from_reg_id;
    end

    if (reg_id > 0) then
    begin
      select r1.reg_name, r2.reg_name
      from region as r1
      left join region as r2 on r2.reg_unicode = r1.parent
      where r1.reg_unicode = :reg_id
      into :office_name, :reg_name;
    end
    else
    begin
      office_name = null;
      reg_name = null;
    end

    iqa_id = (select iqa_id
              from ipc_qr_app_shops
              inner join ipc_qr_app on iqa_id = iqs_from_iqa_id and
                    iqa_status >= 0
              where iqs_from_cs_id = :cs_id);

    phone_len = char_length(cs_phone);
    phone_enc = iif(phone_len >= 4, overlay(cs_phone placing '****' from phone_len - 3), cs_phone);
    qr_assigned = iif(cs_qr_data is null, 0, 1);

    invent_number = (
      select trm_invent_str
      from ipc_terminal_ownership
        inner join ipc_terminals itm
          on itm.trm_id = ito_trm_id
            and trm_actual = 1
      where ito_object_id = :cs_id
        and ito_object_kind = :obj_kind
        and :dt_curr between ito_date_begin and ito_date_end
        and ito_status > -1);
    trm_binding = iif(invent_number is null, 0, 1);

    office_name = coalesce(office_name, '???');
    reg_name = coalesce(reg_name, '???');

    suspend;
  end
end
