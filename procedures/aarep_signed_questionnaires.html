CREATE PROCEDURE AAREP_SIGNED_QUESTIONNAIRES(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    RES_REP_DATE D_DATE,
    RES_REG_NAME VARCHAR(50),
    RES_LOAN_OFFICER D_STR_100,
    RES_KF_NAME D_STR_100,
    RES_KONTRACT_NO D_STR_20,
    RES_C_NAME_FIO D_STR_100)
AS
declare variable C_ID ID_TYPE;
declare variable PK_ID D_INTEGER;
declare variable IS_AUTO D_INTEGER;
declare variable K_DISB_DATE D_DATE;
declare variable REG_ID_GO D_INTEGER;
declare variable REG_ID D_INTEGER;
declare variable K_ID D_ID;
begin
  res_rep_date = rep_date_end;
  select
    pk_id
  from passkind
    inner join passdoctype
      on pdt_id = pk_from_pdt_id
        and pdt_code = 'anketa_questionnaire'
  into
    :pk_id;


  reg_id_go = f_hq_id();

  res_reg_name = (select reg_name from region where reg_unicode = :rep_region_id);
  for
    with cl as (
      select
        -- если в один день клиенту выдано два кредита
        -- берем только одну запись
        min(kc_id) as kc1_id,
        kc_from_c_id as c1_id,
        ks_region_id as reg_id
      from kontstate
        inner join knclient
          on kc_from_k_id = ks_from_k_id
            and kc_amount_approved > 0
      where ks_region_id in (:rep_region_id, :reg_id_go)
        and ks_date_end >= ks_date_begin
        and ks_date_begin = :rep_date_end
        and ks_state = 1
        and ks_ol_is_delete = 0
      group by kc_from_c_id, ks_region_id
    )

    select
      c0.c_name_fio,
      kal_new_kf_name,
      k_kontract_no,
      k_disb_date,
      iif((pr_code containing '.zalog_car.' or pr_code containing '.zalog.'),1,0),
      cl.c1_id,
      c1.c_name_fio,
      cl.reg_id,
      k_id
    from knclient
      inner join kontract
        on k_id = kc_from_k_id
      inner join product
        on pr_id = k_from_pr_id
      inner join knaktlist
        on kal_from_kf_id = k_from_kf_id
          and knaktlist.is_delete = 0
          and :rep_date_end between kal_date and kal_date_end
      inner join clients as c0
        on c0.c_id = kal_new_kf_officer
      inner join cl
        on cl.kc1_id = kc_id
      inner join clients as c1
        on c1.c_id = cl.c1_id
  order by kal_new_kf_officer
  into
    res_loan_officer,
    res_kf_name,
    res_kontract_no,
    k_disb_date,
    is_auto,
    c_id,
    res_c_name_fio,
    reg_id,
    k_id
  do
  begin
    -- Отчет отправляется если:
    -- 1. микрозаем под залог (автозайм), возвращаем клиента и залогодателя
    -- ИЛИ
    -- 2. в доках клиента есть прикрепленная анкета-опросник
    --    где в период с даты ее прикрепления & -до даты текущей выдачи
    --    не было выдано ни одного займа данному клиенту

    -- Если кредит числится в ГО, проверяем место работы сотрудника
    -- Если совпадает с rep_region_id, то тоже возвращаем

    if (is_auto = 1) then
    begin
      if (f_ft_valid(:c_id, :k_disb_date-1) = 0) then
        suspend;
      for select z_from_c_id, c_name_fio
      from zalog left join clients on z_from_c_id = c_id
      where z_from_k_id = :k_id
        and z_subtype = 4
        and z_from_c_id <> :c_id -- Если клиент-залогодатель, или уже вернули или не требуется
      into c_id, res_c_name_fio do
        suspend;
      continue;
    end

    if (exists(
      select *
      from clientdocs
      where cds_from_c_id = :c_id
        and cds_from_pk_id = :pk_id
        and cds_is_delete = 0
        and (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0)
        and not exists(
          select *
          from knclient
            inner join kontstate
              on ks_from_k_id = kc_from_k_id
                and ks_date_begin between cds_doc_date_begin and (:rep_date_end - 1)
                and ks_state = 1
                and ks_ol_is_delete = 0
          where kc_from_c_id = :c_id
            and kc_amount_approved > 0)))
    then
    begin
      if (reg_id = rep_region_id) then
        suspend;
      else if (reg_id = reg_id_go and exists (select *
        from hr_pstatus, clients
        where
          ps_from_reg_id = :rep_region_id and
          ps_is_working = 1 and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_c_id = :c_id and
          ps_status = 1)) then
        suspend;
    end
  end
end
