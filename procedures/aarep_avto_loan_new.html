CREATE PROCEDURE AAREP_AVTO_LOAN_NEW(
    REP_DATE DATE)
RETURNS (
    KAL_NEW_KF_NAME STR_100,
    K_KONTRACT_NO STR_20,
    K_DISB_DATE DATE,
    KS_STATE STR_100,
    KS_DATE_BEGIN DATE,
    CAR_OWNER STR_20)
AS
begin
  for
    select
      a.kal_new_kf_name,
      a.k_kontract_no,
      a.k_disb_date,
      a.ks_state,
      a.ks_date_begin,
      a.car_owner
    from (
      select * from aarep_avto_loan_tmp(:rep_date,  0)
      union
      select * from aarep_avto_loan_tmp(:rep_date,  1)
      union
      select * from aarep_avto_loan_tmp(:rep_date,  2)
      union
      select * from aarep_avto_loan_tmp(:rep_date,  3)
    ) as a
    order by a.ks_state
  into
    :kal_new_kf_name,
    :k_kontract_no,
    :k_disb_date,
    :ks_state,
    :ks_date_begin,
    :car_owner
  do suspend;
end
