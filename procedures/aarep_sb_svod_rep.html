CREATE PROCEDURE AAREP_SB_SVOD_REP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    SBL_ID DECIMAL(18,0),
    SBL_NAME VARCHAR(255),
    SENDED_IN_SB INTEGER,
    CLOSED_SB INTEGER,
    SENDED_IN_SB_VAL DECIMAL(18,4),
    CLOSED_SB_VAL DECIMAL(18,4),
    BACKED_TO_SCH INTEGER,
    BACKED_TO_SCH_VAL DECIMAL(18,4),
    NOT_BACKED_TO_SCH INTEGER,
    NOT_BACKED_TO_SCH_VAL DECIMAL(18,4),
    SB_BEFOR_PER INTEGER,
    SB_BEFOR_PER_VAL DECIMAL(18,4),
    BACK_TO_SVZ INTEGER,
    BACK_TO_SVZ_VAL DECIMAL(18,4))
AS
declare variable k_id decimal(18,0);
declare variable sended_in_sb_val_tmp decimal(18,4);
declare variable closed_sb_val_tmp decimal(18,4);
declare variable sba_date_begin date;
declare variable sba_date_end date;
declare variable payed_summ decimal(18,4);
declare variable result smallint;
declare variable sb_befor_per_val_tmp decimal(18,4);
declare variable from_sb_to_svz_val_tmp decimal(18,4);
declare variable k_id_temp decimal(18,0);
declare variable sb_number integer;
declare variable back_to_svz_val_tmp decimal(18,4);
begin
  /*описание к полям отчета исходя из требований к отчету
  Расшифровка к полям:
1)    Передано в СБ (кол-во) - отражает количеств кредитов, которые были переданы СБ за отчетный период, причем если кредит на конечную дату отчета уже не у СБ, он тем не менее отразится в этом поле. Если один и тот же кредит два раза был передан данному СБ, в этом поле он отразится только один раз.
2)    Передано в СБ (сумма) - отражает сумму задолженности (ОС+%+пеня) на дату конца отчета, по кредитам из пункта (1)
3)    Закрыто СБ (кол-во) – отображает количество закрытых кредитов , которые были  закрыты данным СБ  за указанный период.
4)    Закрыто СБ (сумма) – отражает общую сумму, которая была внесена по кредитам из пункта (3)  (ОС+%+пеня)
5)    Возвращенных в стандарт. учет (кол-во) – Отображает количество  контрактов из
      указанного периода в отчете,  которые были возвращены уполномоченному лицу из кредитного персонала(СВЗ,ССВЗ или др.), если кредит возвращался от данного СБ несколько раз, за указанный период, в данном поле, он отразится только один раз.
6)    Возвращенных в стандарт. учет (сумма) – сумму задолженности (ОС+%%+Пении) контрактов из пункта 5) на дату конца отчета.
7)    Число активных до периода (кол-во) – Количество активных кредитов числящихся за этим СБ, до начала указанного периода в отчете.
8)    Число активных до периода (сумма) – сумму задолженности (ОС+%%+Пении)  контрактов, из пункта 7) на дату конца отчета
9)    Не возвращенных в график активн. (кол-во) – отражает количество контрактов, активных на дату конца отчета,  где контракт ни разу не вышел из просрочки, за указанный период.
10)    Не возвращенных в график активн. (сумма) – отражает сумму, которую группа заплатила по контрактам из пункта 9)
11)    Возвращенных в график активн. (кол-во)  – отражает количество контрактов, активных на дату конца отчета,  где хотя-бы один раз баланс кредита был возвращен из просрочки данным СБ.
12)    Возвращенных в график активн. (сумма)  -  – отражает сумму, которую группа заплатила по контрактам из пункта 11)
  */

  for select sbl.sbl_id, sbl.sbl_name
      from sb_list sbl
      where sbl_type = 0 and
            sbl.sbl_id > 0 and
            (sbl.sbl_date_end >= :rep_date_begin or
            (sbl.sbl_date_end is null))
      order by sbl.sbl_name
      into :sbl_id, :sbl_name
  do
  begin
-- переданные кредиты в СБ
    sended_in_sb = 0;
    sended_in_sb_val = 0;

    for select distinct sba_from_k_id
        from sb_akt sba
        where sba.sba_date_begin between :rep_date_begin and :rep_date_end -- кредит был у СБ за указанный период. На конец периода кредитам может уже не быть у СБ
              and
              sba.sba_date_begin <= sba.sba_date_end and
              sba_is_delete = 0 and
              sba.sba_from_sbl_id = :sbl_id
        into :k_id
    do
    begin
      sended_in_sb = sended_in_sb + 1;

      select sum(zad_os + sh_calc_perc + sh_calc_fine)
      from aarep_make_zad_list(:k_id, :rep_date_end, 1)
      where sh_date is not null
      into :sended_in_sb_val_tmp;

      sended_in_sb_val = sended_in_sb_val + coalesce(sended_in_sb_val_tmp, 0);
    end

-- закрытые кредиты за период
    closed_sb = 0;
    closed_sb_val = 0;

    for select distinct ks.ks_from_k_id
        from kontstate ks
        inner join sb_akt sba on sba.sba_from_k_id = ks.ks_from_k_id
        where sba.sba_date_end >= :rep_date_begin -- на начало периода в отчете, контракт уже у СБ
              and
              sba.sba_date_begin <= sba.sba_date_end and
              ks.ks_date_begin <= sba.sba_date_end -- контракт закрылся, нахоясь на руках у СБ
              and
              sba_is_delete = 0 and
              sba.sba_from_sbl_id = :sbl_id and
              ks.ks_date_begin between :rep_date_begin and :rep_date_end and
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 2
        into :k_id
    do
    begin
      closed_sb = closed_sb + 1;

      select -sum(ok.ok_os + ok.ok_perc + ok.ok_fine)
      from operkred ok
      where ok.ok_ol_is_active = 1 and
            ok.ok_from_k_id = :k_id and
            ok.ok_type in (4, 9) and
            ok.ok_ol_date between :rep_date_begin and :rep_date_end and
            ok.ok_from_c_id = 0
      into :closed_sb_val_tmp;

      closed_sb_val = closed_sb_val + coalesce(closed_sb_val_tmp, 0);
    end

-- кол. активн. договоров, которые были на руках у данного СБ за период отчета

    backed_to_sch = 0;
    payed_summ = 0;
    backed_to_sch_val = 0;
    not_backed_to_sch = 0;
    not_backed_to_sch_val = 0;

    for select ks.ks_from_k_id, sba.sba_date_begin, sba.sba_date_end
        from kontstate ks
        inner join sb_akt sba on sba.sba_from_k_id = ks.ks_from_k_id
        where sba.sba_date_begin <= :rep_date_end -- две строки позволяют получить отрезок в рамках которого контракт числился за СБ
              and
              sba.sba_date_end >= :rep_date_begin

              and
              sba.sba_date_begin <= sba.sba_date_end and
              sba_is_delete = 0 and
              sba.sba_from_sbl_id = :sbl_id and
              :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1
        into :k_id, :sba_date_begin, :sba_date_end
    do
    begin
-- если за перид пока контракт был у СБ, были приняты погашения которые вернули график,
-- в состояние без просрочки, хотя бы один раз за период указанный в отчете, result = 1,
--  если график выровнять не удалось или не было принято погашения, result = 0

      select iif(sum(delinq) > 0, 1, 0) as result, coalesce(sum(total_sum), 0) as payed_summ
      from (select ok_from_k_id, oks_delinq_date, ok_ol_date, ok_sh_date, -sum(ok_os + ok_sh_os) os,
                   -sum(ok_perc + ok_sh_perc) perc, -sum(ok_fine) fine, -sum(ok_tax) tax,
                   -sum((ok_os + ok_sh_os) + (ok_perc + ok_sh_perc) + ok_fine + ok_tax) total_sum,
                   iif(oks_delinq_date >= ok_ol_date, 1, 0) delinq
            from operkred ok, oksummary
            where ok_ol_is_active = 1 and
                  ok_type in (4, 9) and
                  ok_ol_date between :rep_date_begin and :rep_date_end and
                  ok_from_k_id = oks_from_k_id and
                  ok_ol_date between oks_date_begin and oks_date_end and
                  ok_from_k_id = :k_id and
                  ok.ok_ol_date between :sba_date_begin and :sba_date_end and
                  ok.ok_from_c_id = 0 and
                  oks_close_state = 0
            group by ok_from_k_id, oks_delinq_date, ok_ol_date, ok_sh_date) table1
      into :result, :payed_summ;

      if (payed_summ > 0) then
      begin
        if (result = 1) then
        begin
          backed_to_sch = backed_to_sch + 1;
          backed_to_sch_val = backed_to_sch_val + payed_summ;
        end
        else
        begin
          not_backed_to_sch = not_backed_to_sch + 1;
          not_backed_to_sch_val = not_backed_to_sch_val + payed_summ;
        end
      end
      else
      begin
        not_backed_to_sch = not_backed_to_sch + 1;
      end
    end

-- активные кредиты до начала периода числящиеся за этим СБ

    sb_befor_per = 0;
    sb_befor_per_val = 0;

    for select ks.ks_from_k_id, sba.sba_date_begin, sba.sba_date_end
        from kontstate ks
        inner join sb_akt sba on sba.sba_from_k_id = ks.ks_from_k_id
        where sba.sba_date_end >= :rep_date_begin /* <<-- !!! кредиты должны числиться и на дату начала отчета*/
              and
              sba.sba_date_begin < :rep_date_begin -- клиенты были переданы СБ до даты начала отчета
              and
              sba.sba_date_begin <= sba.sba_date_end -- отсеиваю лишние записи
              and
              sba_is_delete = 0 and
              sba.sba_from_sbl_id = :sbl_id and
              :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1
        into :k_id, :sba_date_begin, :sba_date_end
    do
    begin
      sb_befor_per = sb_befor_per + 1;

      select sum(zad_os + sh_calc_perc + sh_calc_fine)
      from aarep_make_zad_list(:k_id, :rep_date_end, 1)
      where sh_date is not null
      into :sb_befor_per_val_tmp;

      sb_befor_per_val = sb_befor_per_val + coalesce(sb_befor_per_val_tmp, 0);
    end

    /*
      Вытаскиваем кредиты за период, которые были переданы СВЗ от СБ,
      вне завсисимости был ли передан кредит обратно к СБ или нет на конец периода.

      1. вытаскиваю все кредиты за период котрые были отправлены к СВЗ
      2. вытаскиваю предыдущую запись по ним за период, если она соответствует данному СБ, тогда + 1
      */

    k_id_temp = 0;
    back_to_svz = 0;
    back_to_svz_val_tmp = 0;
    back_to_svz_val = 0;

    for select sba.sba_from_k_id, sba.sba_number
        from sb_akt sba
        where sba.sba_date_begin <= sba.sba_date_end -- отсеиваю лишние записи
              and
              sba.sba_from_sbl_id = 0 and
              sba.sba_is_delete = 0 and
              sba.sba_date_begin between :rep_date_begin and :rep_date_end -- передача был в рамках периода
        order by sba.sba_from_k_id, sba.sba_date_begin
        into :k_id, :sb_number
    do
    begin
      if (k_id <> k_id_temp) then
      begin
        if (sb_number <> 1) then
        begin
          if (exists(select *
                     from sb_akt sba
                     where sba.sba_from_k_id = :k_id and
                           sba.sba_number < :sb_number and
                           sba.sba_from_sbl_id = :sbl_id and
                           sba.sba_is_delete = 0 and
                           sba.sba_date_begin <= sba.sba_date_end and
                           sba.sba_date_begin between :rep_date_begin and :rep_date_end)) then
          begin
            back_to_svz = back_to_svz + 1;

            select sum(zad_os + sh_calc_perc + sh_calc_fine)
            from aarep_make_zad_list(:k_id, :rep_date_end, 1)
            where sh_date is not null
            into :back_to_svz_val_tmp;

            back_to_svz_val = back_to_svz_val + coalesce(back_to_svz_val_tmp, 0);
          end
        end
        k_id_temp = k_id;
      end
    end
    suspend;
  end

end
