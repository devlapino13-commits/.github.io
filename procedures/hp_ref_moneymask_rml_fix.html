CREATE PROCEDURE HP_REF_MONEYMASK_RML_FIX(
    RMM_ID D_ID)
AS
declare variable rmm_from_rml_resident d_integer;
declare variable rmm_from_rml_noresident d_integer;
declare variable rmm_law_rml_resident d_integer;
declare variable rmm_law_rml_noresident d_integer;
declare variable rml_id d_integer;
declare variable rml_is_resident d_bool;
declare variable rml_is_law d_bool;
begin
  rmm_from_rml_resident = 0;
  rmm_from_rml_noresident = 0;
  rmm_law_rml_resident = 0;
  rmm_law_rml_noresident = 0;

  for select rml_id, rml_is_resident, rml_is_law
  from ref_moneylist
  where rml_from_rmm_id = :rmm_id and
        rml_inactive = 0
  into rml_id, rml_is_resident, rml_is_law
  do
  begin
    if (rml_is_resident = 1) then
    begin
      if (rml_is_law = 0)
      then rmm_from_rml_resident = rml_id;
      else rmm_law_rml_resident = rml_id;
    end else
    begin
      if (rml_is_law = 0)
      then rmm_from_rml_noresident = rml_id;
      else rmm_law_rml_noresident = rml_id;
    end
  end

  update ref_moneymask
  set rmm_from_rml_resident = :rmm_from_rml_resident,
      rmm_from_rml_noresident = :rmm_from_rml_noresident,
      rmm_law_rml_resident = :rmm_law_rml_resident,
      rmm_law_rml_noresident = :rmm_law_rml_noresident,
      rmm_order_no = iif(rmm_order_no = -1, 1, rmm_order_no)
  where rmm_id = :rmm_id;

end
