  if (rep_date = f_date_month_end(rep_date)) then
    oks_is_month = 1;

  res_vbs_final = 0;
  res_vbs_stopped = 0;
  res_oks_bal_esp_norm = 0;

  select ks_date_begin
  from kontstate
  where ks_from_k_id = :k_id and
        ks_date_begin <= :rep_date and
        ks_ol_is_delete = 0 and
        ks_date_end >= '31.12.2999' and
        ks_state = 2
  into :close_date;

  if (close_date is null)
  then
    oks_close_state = 0;
  else
    if (rep_date = close_date) then
      oks_close_state = 1;
    else
      oks_close_state = 2;

  amount_begin = 0;
  amount_final = 0;
  amount_zad_os = 0;
  amount_zad_perc = 0;

  res_sh_count = 0;
  res_str_vbs_dates = '';
  res_str_vbs_normal = '';
  oks_plan_sh_date = null;
  oks_delinq_date = null;

  oks_bal_zad_os = 0;
  oks_bal_zad_perc = 0;
  res_sh_perc = 0;
  max_delinq_days = 0;
  res_bal_moved_os = 0; -- пересенный платеж ОС, просроченный который стал не просроченным
  res_resh_delinq = 0; -- пересенный платеж через реструктуризацию

  oks_disb_amount = 0;
  oks_pmt_os_norm = 0;
  oks_pmt_perc_norm = 0;
  oks_pmt_kommis = 0;
  oks_pmt_other_adv = 0;
  oks_pmt_os_delinq = 0;
  oks_pmt_perc_delinq = 0;
  oks_pmt_os_adv = 0;
  oks_pmt_perc_adv = 0;
  oks_pmt_is_adv = 0;
  oks_rvs_perc = 0;
  oks_off_os = 0;
  oks_off_perc = 0;
  oks_is_sh = 0;
  oks_balance = 0;
  oks_last_resh_delinq_before = 0;
  tmp_resh_delinq = 0;
  oks_is_reschedule = 0;

  for select or_ol_date, or_delinq_day_before,
             coalesce(or_delinq_os_before - or_delinq_os_after + or_delinq_perc_before - or_delinq_perc_after, 0),
             or_from_rd_id
  from operrest
  where or_from_k_id = :k_id and
        or_llr_depends = 1 and
        or_ol_is_active = 1 and
        or_ol_date <= :rep_date
  order by or_ol_date
  into tmp_resh_date, oks_last_resh_delinq_before, tmp_resh_delinq, or_from_rd_id
  do
  begin
    if (oks_restruct_date is null) then
      oks_restruct_date = tmp_resh_date;

    if (tmp_resh_date = rep_date and tmp_resh_delinq > 0) then
      res_resh_delinq = tmp_resh_delinq;

    oks_last_resh_date = tmp_resh_date;
  end

  if (rep_date >= date_20260501) then
  begin
    if (oks_last_resh_date is not null) then
    begin
      if (exists(select * from refdata
      where rd_id = :or_from_rd_id
      and rd_code in ('Reason_Reschedule.8790', 'Reason_Reschedule.10086') -- Заключение контракта на участие в СВО
      )) then
      begin
        oks_last_resh_date = null;
        oks_last_resh_delinq_before = 0;
      end
    end
  end

  if (rep_date >= date_20221231 and kc_id = 1822725) then -- Первый займ с кредитным каникулом
  begin
    oks_restruct_date = '17.11.2022';
    res_resh_delinq = 0;

    if (oks_last_resh_date is null or oks_last_resh_date < oks_restruct_date) then
    oks_last_resh_date = oks_restruct_date;
  end

  oks_delinq_days_all = 0;
  oks_refund_delinq_180 = 0;
  delinq_array_len = datediff(day, oks_disb_date, rep_date) + 1;
  delinq_array = lpad('', delinq_array_len, '0'); -- массив просроченных дней

  oks_actual_last_date = null;
  for select ok_sh_date,
         sum(ok_os), -- oks_balance
         max(iif((ok_type = 4 and (ok_os < 0 or ok_perc < 0)) or (ok_type < 7 and ok_type <> 4 and (ok_os <> 0 or ok_perc <> 0)), ok_ol_date, ok_sh_date)), -- last_ol_date
         max(iif(ok_type = 2 and ol_oper_type = 1 and ol_oper_subtype in (4, 9) and ok_sh_date < ok_ol_date and
                (ok_sh_os < 0 or ok_sh_perc < 0), ok_ol_date, null)), -- last_moved_ol_date
         max(iif(ok_sh_date = :rep_date and ok_type <= 3, 1, 0)), -- oks_is_sh
         sum(iif(ok_ol_date = :rep_date and ok_type = 8, -ok_os, 0)), -- oks_disb_amount
         sum(iif(ok_ol_date = :rep_date and ok_ol_date = ok_sh_date and ok_type = 4, ok_os, 0)), -- oks_pmt_os_norm
         sum(iif(ok_ol_date = :rep_date and ok_ol_date = ok_sh_date and ok_type = 4, ok_perc, 0)), -- oks_pmt_perc_norm
         sum(iif(ok_ol_date = :rep_date and ok_type = 4, ok_kommis, 0)), -- oks_pmt_kommis
         sum(iif(ok_ol_date = :rep_date and ok_type = 4, ok_kommis_adv, 0)), -- oks_pmt_other_adv
         sum(iif(ok_ol_date = :rep_date and ok_type = 4 and ok_sh_date < ok_ol_date, ok_os, 0)), -- oks_pmt_os_delinq
         sum(iif(ok_ol_date = :rep_date and ok_type = 4 and ok_sh_date < ok_ol_date, ok_perc, 0)), -- oks_pmt_perc_delinq
         sum(iif(ok_ol_date = :rep_date and ok_type = 4 and ok_sh_date > ok_ol_date, ok_os, 0)), -- oks_pmt_os_adv
         sum(iif(ok_ol_date = :rep_date and ok_type = 4 and ok_sh_date > ok_ol_date, ok_perc, 0)), -- oks_pmt_perc_adv
         max(iif(ok_ol_date = :rep_date and ok_sh_date > ok_ol_date and ok_type <= 6 and ((abs(ok_os + ok_perc) > 0) or (abs(ok_sh_os + ok_sh_perc) > 0)), 1, 0)), -- oks_pmt_is_adv
         sum(iif(ok_ol_date = :rep_date and ok_type = 7, ok_perc, 0)), -- oks_rvs_perc
         sum(iif(ok_ol_date = :rep_date and ok_type in (5, 6), ok_os, 0)), -- oks_off_os
         sum(iif(ok_ol_date = :rep_date and ok_type in (5, 6), ok_perc, 0)), -- oks_off_perc
         max(iif(ok_ol_date = :rep_date, 1, 0)), -- oks_is_pmt

         list(iif(ok_sh_date >= '01.03.2020' and ok_sh_date > :rep_date and ok_type = 2 and
                 ol_oper_type = 1 and ol_oper_subtype = 9 and ok_sh_os > 0, ok_sh_id, 0)), -- moved_sh_id_list
         sum(iif((ok_type < 7 and ok_ol_date < :rep_date) or ok_type = 1, ok_os + ok_perc + ok_sh_os + ok_sh_perc, 0.0)), -- amount_begin
         sum(iif(ok_type < 7, ok_os + ok_sh_os, 0)), -- amount_zad_os
         sum(iif(ok_type < 7, ok_perc + ok_sh_perc, 0)), -- amount_zad_perc
         sum(iif(ok_type < 7, ok_sh_perc, 0)), -- sh_perc
         sum(iif(ok_type = 2 and (ok_sh_os <> 0 or ok_sh_perc <> 0), 1, 0)), -- has_max_restruc
         sum(iif(ok_type = 2 and (ok_sh_os <> 0 or ok_sh_perc <> 0) and ok_ol_date = :rep_date, 1, 0)), -- has_curr_restruc
         max(iif(ok_type in (4, 5, 6) and (ok_os < 0 or ok_perc < 0) and ok_ol_date > ok_sh_date,
             datediff(day, ok_sh_date, ok_ol_date), 0)) -- tmp_delinq_days
  from kn_operkred, operlist
  where ok_from_kc_id = :kc_id and
        ok_ol_is_active = 1 and
        ok_ol_date <= :rep_date and
        ok_from_ol_id = ol_id and
        ok_type <= 8
  group by ok_sh_date
  order by ok_sh_date desc
  into ok_sh_date, tmp_balance, last_ol_date, last_moved_ol_date, tmp_is_sh,
       tmp_disb_amount, tmp_pmt_os_norm, tmp_pmt_perc_norm, tmp_pmt_kommis, tmp_pmt_other_adv,
       tmp_pmt_os_delinq, tmp_pmt_perc_delinq, tmp_pmt_os_adv, tmp_pmt_perc_adv, tmp_pmt_is_adv,
       tmp_rvs_perc, tmp_off_os, tmp_off_perc, tmp_is_pmt,

       moved_sh_id_list, amount_begin, amount_zad_os, amount_zad_perc, sh_perc,
       has_max_restruc, has_curr_restruc, tmp_delinq_days
  do
  begin
    if (oks_actual_last_date is null) then
    begin
      if (amount_zad_os + amount_zad_perc >= 0.01) then
        oks_actual_last_date = ok_sh_date;
    end

    if (rep_date >= date_20221130 and last_moved_ol_date > last_ol_date) then
      last_ol_date = last_moved_ol_date;

    if (ok_sh_date < last_ol_date or amount_zad_os + amount_zad_perc > 0.01) then
    begin
      if (amount_zad_os + amount_zad_perc > 0.01) then
        last_ol_date = rep_date + 1;

      if (ok_sh_date < last_ol_date) then
        delinq_array = overlay(delinq_array
                               placing lpad('', datediff(day, ok_sh_date, last_ol_date), '1')
                               from datediff(day, oks_disb_date, ok_sh_date) + 1);
    end

    -- Фактические платежи
    oks_balance = oks_balance + coalesce(tmp_balance, 0);
    oks_disb_amount = oks_disb_amount + coalesce(tmp_disb_amount, 0);
    oks_pmt_os_norm = oks_pmt_os_norm + coalesce(tmp_pmt_os_norm, 0);
    oks_pmt_perc_norm = oks_pmt_perc_norm + coalesce(tmp_pmt_perc_norm, 0);
    oks_pmt_kommis = oks_pmt_kommis + coalesce(tmp_pmt_kommis, 0);
    oks_pmt_other_adv = oks_pmt_other_adv + coalesce(tmp_pmt_other_adv, 0);
    oks_pmt_os_delinq = oks_pmt_os_delinq + coalesce(tmp_pmt_os_delinq, 0);
    oks_pmt_perc_delinq = oks_pmt_perc_delinq + coalesce(tmp_pmt_perc_delinq, 0);
    oks_pmt_os_adv = oks_pmt_os_adv + coalesce(tmp_pmt_os_adv, 0);
    oks_pmt_perc_adv = oks_pmt_perc_adv + coalesce(tmp_pmt_perc_adv, 0);
    oks_rvs_perc = oks_rvs_perc + coalesce(tmp_rvs_perc, 0);
    oks_off_os = oks_off_os + coalesce(tmp_off_os, 0);
    oks_off_perc = oks_off_perc + coalesce(tmp_off_perc, 0);

    if (tmp_pmt_is_adv = 1) then
      oks_pmt_is_adv = 1;

    if (tmp_is_pmt = 0 and oks_date_stop_perc = rep_date) then
      tmp_is_pmt = 1;

    if (tmp_is_pmt = 1) then
      oks_is_pmt = 1;

    -- Вычисляем задолженность по графику, дата просрочки и плановую
    if (tmp_is_sh = 1) then oks_is_sh = 1;

    if (moved_sh_id_list is null) then
      moved_sh_id_list = '';

    if (moved_sh_id is null) then
      moved_sh_id = 0;

    if (tmp_resh_delinq is null) then
      tmp_resh_delinq = 0;

    amount_final = amount_zad_os + amount_zad_perc;
    if (tmp_delinq_days > max_delinq_days) then
      max_delinq_days = tmp_delinq_days;

    if (ok_sh_date = rep_date) then
      res_sh_perc = sh_perc;

    if (has_max_restruc > 0) then
    begin
      if (oks_last_sh_date is null or oks_last_sh_date < ok_sh_date) then
        oks_last_sh_date = ok_sh_date;
    end
    if (has_curr_restruc > 0) then
    begin
      if (res_prolong_date_curr is null or res_prolong_date_curr < ok_sh_date) then
        res_prolong_date_curr = ok_sh_date;
    end

    if (ok_sh_date <= rep_date) then
    begin
      if (amount_final > 0 and (oks_delinq_date is null or oks_delinq_date > ok_sh_date)) then
        oks_delinq_date = ok_sh_date; -- дата просрочки

      oks_bal_zad_os = oks_bal_zad_os + amount_zad_os;
      oks_bal_zad_perc = oks_bal_zad_perc + amount_zad_perc;
    end else
    begin
      -- если не просроченный платеж был перенесен из-за выходных,
      -- то проверяем был ли он первоначально до даты отчета
      -- если да, считаем его как перенесенный и на него не будет начисляться процент как и просроченным
      if (moved_sh_id_list <> '' and rep_date >= '31.03.2020') then
      begin
        for select res_str
        from hp_split_str(:moved_sh_id_list, ',')
        into :moved_sh_id
        do
        begin
          if (moved_sh_id = 0) then
            Continue;

          tmp_moved_os = 0;
          tmp_zad_os = 0;
          select sum(ok_os + iif(ok_type <> 3, ok_sh_os, 0)),
                 sum(iif(ok_sh_date > :rep_date and ok_type = 2 and ol_oper_type = 1 and ol_oper_subtype = 9 and ok_sh_os > 0, ok_sh_os, 0))
          from kn_operkred, operlist
          where ok_from_kc_id = :kc_id and
                ok_sh_id = :moved_sh_id and
                ok_sh_date = :ok_sh_date and
                ok_ol_date <= :rep_date and
                ok_ol_is_active = 1 and
                ok_from_ol_id = ol_id and
                ok_type < 7
          into tmp_zad_os, tmp_moved_os;
          if (tmp_zad_os is null) then
            tmp_zad_os = 0;
          if (tmp_moved_os is null) then
            tmp_moved_os = 0;
          if (tmp_moved_os > tmp_zad_os) then
            tmp_moved_os = tmp_zad_os;

          res_bal_moved_os = res_bal_moved_os + tmp_moved_os;
        end
      end
    end

    if (ok_sh_date >= rep_date) then
    begin
      res_oks_bal_esp_norm = res_oks_bal_esp_norm + amount_begin / power(1.0 + cast(oks_percent_esp as double precision) / 100.0, cast(datediff(day, rep_date, ok_sh_date) as double precision) / 365.0);

      res_vbs_final = res_vbs_final + amount_final / power(1.0 + cast(oks_percent_esp as double precision) / 100.0, cast(datediff(day, rep_date, ok_sh_date) as double precision) / 365.0);
      res_vbs_stopped = res_vbs_stopped + amount_zad_os;

      if (amount_final <> 0) then
      begin
        if (oks_plan_sh_date is null or oks_plan_sh_date > ok_sh_date) then
          oks_plan_sh_date = ok_sh_date; -- ближайшая дата погашения

        if (res_ead_last_sh_date is null or res_ead_last_sh_date < ok_sh_date) then
          res_ead_last_sh_date = ok_sh_date;

        res_sh_count = res_sh_count + 1;

        -- Внимание, значения в массиве должны быть отсортированы по возрастанию SH_DATE
        res_str_vbs_dates = f_datetostr(ok_sh_date) || res_str_vbs_dates;
        if (amount_final >= 0) then
          res_str_vbs_normal = lpad(amount_final, 20, '0') || res_str_vbs_normal;
        else
          res_str_vbs_normal = '-' || lpad(-amount_final, 19, '0') || res_str_vbs_normal;
      end
    end
  end

  if (rep_date >= date_20221130 and right(delinq_array, 1) = '1') then
    delinq_array = left(delinq_array, char_length(delinq_array) - 1) || '0';

  if (oks_last_sh_date is null or oks_last_sh_date < oks_kont_last_date) then
    oks_last_sh_date = oks_kont_last_date;

  if (res_prolong_date_curr is null or res_prolong_date_curr < oks_kont_last_date) then
    res_prolong_date_curr = oks_kont_last_date;

  if (oks_restruct_date is not null and
      rep_date >= oks_restruct_date and
      oks_last_sh_date > oks_kont_last_date) then
    oks_is_reschedule = 1;

  oks_delinq_days = 0;
  oks_delinq_days_own_llr = 0;
  oks_delinq_days_llr = 0;
  oks_delinq_days_365 = 0;
  oks_delinq_last_365 = 0;
  tmp_delinq_day_begin = 0;
  tmp_delinq_day_end = 0;
  oks_paral_loan_count = 0;
  oks_paral_delinq_day = 0;

  oks_delinq_field_30_1 = 0;
  oks_delinq_field_30_2 = 0;

  if (oks_delinq_date is not null) then
  begin
    oks_delinq_days = datediff(day, oks_delinq_date, rep_date);

    if (rep_date >= date_20230531) then
    begin
      pos = delinq_array_len;

      tmp_delinq_day_begin = 1;

      tmp_delinq_day_end = datediff(day, oks_disb_date, minvalue(oks_delinq_date, coalesce(oks_last_resh_date, oks_delinq_date)));
      if (tmp_delinq_day_end <= 0) then
        tmp_delinq_day_end = 0;

      while (pos > 0) do
      begin
        if (substring(delinq_array from pos for 1) = '1') then
        begin
          if (pos between tmp_delinq_day_begin and tmp_delinq_day_end) then
            oks_delinq_field_30_2 = oks_delinq_field_30_2 + 1;
        end
    
        pos = pos - 1;
      end

      if (oks_last_resh_date < oks_delinq_date) then
        oks_delinq_field_30_2 = oks_delinq_field_30_2 - oks_last_resh_delinq_before;

      if (oks_delinq_field_30_2 < 0) then
        oks_delinq_field_30_2 = 0;
    end

    if (oks_last_resh_date is not null and rep_date >= date_20230531 and oks_last_resh_date <= oks_delinq_date) then
    begin
      pos = delinq_array_len;

      tmp_delinq_day_begin = datediff(day, oks_disb_date, oks_last_resh_date) + 1;
      if (tmp_delinq_day_begin <= 0) then
        tmp_delinq_day_begin = 0;

      tmp_delinq_day_end = datediff(day, oks_disb_date, oks_delinq_date);
      if (tmp_delinq_day_end <= 0) then
        tmp_delinq_day_end = 0;

      while (pos > 0) do
      begin
        if (substring(delinq_array from pos for 1) = '1') then
        begin
          if (pos between tmp_delinq_day_begin and tmp_delinq_day_end) then
            oks_delinq_field_30_1 = oks_delinq_field_30_1 + 1;
        end
    
        pos = pos - 1;
      end
    end

    tmp_delinq_day_begin = 0;
    tmp_delinq_day_end = 0;

    max_delinq_days = maxvalue(max_delinq_days, datediff(day, oks_delinq_date, rep_date));

    oks_delinq_days_own_llr = oks_delinq_days + oks_refund_delinq_before + oks_last_resh_delinq_before;
    if (rep_date <= date_20211101) then
      oks_delinq_days_own_llr = oks_delinq_days_own_llr + 1;
  
    oks_delinq_days_llr = oks_delinq_days_own_llr;

    tmp_delinq_day_end = datediff(day, oks_disb_date, oks_delinq_date);
    if (rep_date < date_20221130) then
      tmp_delinq_day_end = datediff(day, oks_disb_date, oks_delinq_date) - 1;

    if (rep_date <= date_20211101) then
    begin
      tmp_delinq_day_begin = datediff(day, oks_disb_date, (oks_delinq_date - 365)) + 1;
      if (tmp_delinq_day_begin <= 0) then
        tmp_delinq_day_begin = 1;
    end else
    if (rep_date < date_20221130) then
    begin
      tmp_delinq_day_begin = datediff(day, oks_disb_date, (oks_delinq_date - 365));
      if (tmp_delinq_day_begin < 0) then
        tmp_delinq_day_begin = 0;
    end else
    begin
      tmp_delinq_day_begin = datediff(day, oks_disb_date, (oks_delinq_date - 365)) + 1;
      if (tmp_delinq_day_begin < 0) then
        tmp_delinq_day_begin = 0;
    end
  end

  oks_last_resh_delinq_180 = 0;
  oks_is_reschedule_llr = 0;
  if (oks_last_resh_date is not null) then
  begin
    oks_is_reschedule_llr = 1;
    oks_last_resh_begin = datediff(day, oks_disb_date, oks_last_resh_date) + 2; -- со след.дня реструктуризаци

    if (rep_date >= date_20221130) then
      oks_last_resh_begin = datediff(day, oks_disb_date, oks_last_resh_date) + 1; -- со дня реструктуризаци

    oks_last_resh_end = oks_last_resh_begin + 179;
  end

  if (rep_date >= date_20221130) then
  begin
    str_last_365 = right(delinq_array, 366);
    oks_delinq_last_365 = char_length(replace(str_last_365, '0', ''));
  end

  if (rep_date >= date_20221231) then
  begin
    oks_delinq_last_365 = oks_delinq_last_365 - oks_delinq_days;
    if (oks_delinq_last_365 < 0) then
      oks_delinq_last_365 = 0;
  end

  pos = delinq_array_len;
  delinq_180 = 0;
  oks_delinq_start_180 = null;

  while (pos > 0) do
  begin
    if (substring(delinq_array from pos for 1) = '1') then
    begin
      if (oks_is_refund = 1 and pos <= refund_day_180) then
        oks_refund_delinq_180 = oks_refund_delinq_180 + 1;

      if (oks_delinq_date is not null and
          pos between tmp_delinq_day_begin and tmp_delinq_day_end) then
        oks_delinq_days_365 = oks_delinq_days_365 + 1;

      if (oks_last_resh_date is not null and
          pos between oks_last_resh_begin and oks_last_resh_end) then
        oks_last_resh_delinq_180 = oks_last_resh_delinq_180 + 1;

      oks_delinq_days_all  = oks_delinq_days_all + 1;

      delinq_180 = delinq_180 + 1;

      if (delinq_180 = 180) then
        oks_delinq_start_180 = dateadd(day, -(delinq_array_len - pos), rep_date);
    end

    pos = pos - 1;
  end

  if (oks_delinq_start_180 is not null) then
  begin
    if (oks_delinq_days > 180 or rep_date < date_20221231) then
      oks_delinq_start_180 = null;

    if (oks_delinq_start_180 <= dateadd(year, -1, rep_date)) then
      oks_delinq_start_180 = null;
  end

  if (oks_is_refund_llr = 1 and oks_refund_delinq_180 = 0 and refund_day_180 = 180) then
    oks_is_refund_llr = 0;

  -- ниже условие добавлено по уточнению от Елены Грабовской, если после выдачи еще не прошли 180 дней то будет считаться рефинансированным
  if (oks_is_refund = 1 and oks_is_refund_llr = 0 and rep_day - 1 <= 180) then
    oks_is_refund_llr = 1;

  debtoverdue_days_ref = 0;
  if (rep_date >= date_20260601 and oks_is_refund_llr = 1 and db_id = 81001) then
  begin
    select max(iif(:kc_date < '21.05.2026', kpl_creddayoverdue, :kc_date - kpl_debtoverdue_date))
    from knclient_pdnlist
    where kpl_from_kcp_id = :kcp_id
    and kpl_exclusion_reason = 1
    into debtoverdue_days_ref;
    if (debtoverdue_days_ref is null) then debtoverdue_days_ref = 0;

    if (oks_refund_delinq_before < debtoverdue_days_ref) then oks_refund_delinq_before = debtoverdue_days_ref;
  end

  if (oks_is_reschedule_llr = 1 and
      oks_last_resh_delinq_180 = 0 and
      dateadd(day, 180, oks_last_resh_date) <= rep_date) then
    oks_is_reschedule_llr = 0;

  if (:rep_date >=:date_20260501 and prv_is_reschedule_llr = 1 and prv_kol_resh = 0) then
  begin
    -- ниже расчет производится только один раз в начале апреля 2026г.
    prv_kol_resh = null;
    with tabl1 as (
      select distinct oks_last_resh_date, 1 schet from kn_oksummary
      where oks_from_kc_id = :kc_id
      and oks_last_resh_date is not null
      and oks_date_begin <= :rep_date
      union
      select or_ol_date, -1
      from operrest, refdata
      where or_from_k_id = :k_id
      and or_ol_is_active = 1
      and or_ol_date <= :rep_date
      and or_from_rd_id = rd_id
      and rd_code in ('Reason_Reschedule.8790', 'Reason_Reschedule.10086')) -- исключаем СВО
    select sum(schet)
    from tabl1
    into prv_kol_resh;

    if (coalesce(prv_kol_resh, 0) = 0) then prv_kol_resh = 1;
  end 

  if (oks_is_reschedule_llr = 1 and coalesce(prv_last_resh_date, '01.01.1900') <>  coalesce(oks_last_resh_date, '01.01.1900')) then
  begin
    oks_kol_resh = prv_kol_resh + 1;
  end else
    oks_kol_resh = prv_kol_resh;

  if (oks_is_reschedule_llr = 0) then
    oks_delinq_field_30_1 = 0;

  if (oks_is_refund_llr = 0) then
    oks_delinq_field_30_2 = 0;

  oks_delinq_days_own_llr = oks_delinq_days_own_llr + oks_delinq_field_30_1 + oks_delinq_field_30_2;

  if (rep_date >= date_20260601 and oks_is_refund_llr = 1 and oks_delinq_days > 0 and db_id = 81001) then
    oks_delinq_days_own_llr = oks_delinq_days_own_llr + oks_refund_delinq_before;

  oks_delinq_days_llr = oks_delinq_days_own_llr;

  oks_paral_loan_count = 0;
  oks_paral_delinq_day = 0;
  select loan_count, max_delinq_day
  from hp_kn_oksummary_max_delinq(:kc_id, :c_id, :rep_date)
  into :oks_paral_loan_count, :oks_paral_delinq_day;

  oks_delinq_days_llr = maxvalue(oks_delinq_days_llr, oks_paral_delinq_day);

  if (rep_date >= date_20260601 and oks_is_refund_llr = 1 and oks_delinq_days_llr < oks_delinq_days_own_llr and db_id = 81001) then
    oks_delinq_days_llr = oks_delinq_days_own_llr;

  -- Предварительный остаток накопленных линейных процентов (без начисленных)
  oks_bal_perc = prv_bal_perc + oks_pmt_perc_norm + oks_pmt_perc_delinq + oks_pmt_perc_adv + oks_off_perc;

  oks_client_is_bankrupt = f_bankrupt(c_id, rep_date);
  if (oks_client_is_bankrupt = -1) then
    oks_client_is_bankrupt = 0;

  oks_zalog_is_bankrupt = 0;
  oks_is_not_targeted = 0;
  oks_no_document = 0;

  if (res_vbs_final is null) then
    res_vbs_final = 0;
  if (res_vbs_stopped is null) then
    res_vbs_stopped = 0;
  if (res_oks_bal_esp_norm is null) then
    res_oks_bal_esp_norm = 0;

  if (oks_is_sh = 0 and oks_is_month = 0) then
    oks_is_pmt = 1;

  -- Платежи по ранее списанным займам
  select sum(oko_os), sum(oko_perc)
  from operkroff
  where oko_from_k_id = :k_id and
        oko_from_c_id = :c_id and
        oko_type = 4 and
        oko_ol_date = :rep_date and
        oko_ol_is_active = 1
  into oks_pmt_os_off, oks_pmt_perc_off;

  if (oks_pmt_os_off is null) then oks_pmt_os_off = 0;
  if (oks_pmt_perc_off is null) then oks_pmt_perc_off = 0;
  if (oks_pmt_os_off <> 0 or oks_pmt_perc_off <> 0) then
    oks_is_pmt = 1;

  -- Платежи по прочим долгам, например судебные иски
  select sum(ox_amount_plus), sum(ox_amount_minus), sum(ox_amount_writeoff)
  from kn_operextra
  where ox_from_kc_id = :kc_id and
        ox_ol_date = :rep_date and
        ox_ol_is_active = 1
  into oks_acr_law, oks_pmt_law, oks_off_law;

  if (oks_acr_law is null) then oks_acr_law = 0;
  if (oks_pmt_law is null) then oks_pmt_law = 0;
  if (oks_off_law is null) then oks_off_law = 0;

  if (oks_pmt_law <> 0 or oks_pmt_law <> 0) then
    oks_is_pmt = 1;

  -- Код из триггера
  oks_days = datediff(day, prv_date, rep_date);

  if (prv_last_sh_date is null) then
    prv_last_sh_date = oks_kont_last_date;

  if (oks_is_month = 1 and
     (oks_bal_perc_fixed = 0 or
      oks_bal_kommis_fixed = 0)) then
  begin
    kcp_period = extract(year from rep_date) * 100 + extract(month from rep_date);

    select kcp_perc_saldo, kcp_kommis_saldo
    from kn_kontcalcperc
    where kcp_from_kc_id = :kc_id and
          kcp_period = :kcp_period
    into oks_bal_perc_fixed, oks_bal_kommis_fixed;

    if (row_count <> 0 or
        kcp_period <= 201808) then
    begin
      oks_is_fix_bal_perc = 1;
      oks_is_fix_bal_kommis = 1;
    end

    if (oks_bal_perc_fixed is null) then
      oks_bal_perc_fixed = 0;
    if (oks_bal_kommis_fixed is null) then
      oks_bal_kommis_fixed = 0;
  end

  -- Код из Autocalc
  sh_changed = 0;

  max_days_for_calc_perc_to_dlq = 90; -- начисляем максимум 90 дней.
  oks_orig_days = oks_days;

  -- если первая строка, устанавливаем сумму договора в необходимые поля
  if (rep_date = prv_date) then
  begin
    prv_bal_to_perc = -oks_disb_amount;
    prv_vbs = -oks_disb_amount;
    prv_bal_esp = -oks_disb_amount;
    oks_kommis_amount = -oks_pmt_kommis;
  end

  oks_pmt_all_ex_adv = oks_pmt_os_norm + oks_pmt_perc_norm +
                       oks_pmt_os_delinq + oks_pmt_perc_delinq +
                       oks_pmt_kommis + oks_off_os + oks_off_perc;

  if (oks_disb_amount > 0) then
    oks_pmt_all_ex_adv = oks_pmt_all_ex_adv - oks_disb_amount;

  oks_pmt_all_adv = oks_pmt_os_adv + oks_pmt_perc_adv + oks_pmt_other_adv;
  oks_pmt_all = oks_pmt_all_ex_adv + oks_pmt_all_adv;

  -- Сальдо по госпошлине
  oks_bal_law = prv_bal_law + oks_acr_law - oks_pmt_law - oks_off_law;

  -- Предварительный остаток накопленных линейных процентов (без начисленных)
  oks_bal_perc = prv_bal_perc + oks_pmt_perc_norm + oks_pmt_perc_delinq + oks_pmt_perc_adv + oks_off_perc;

  -- выдан с минусом, сторно операция
  if (kc_id = 989891 and
      rep_date >= '23.06.2020') then
  begin
    oks_bal_zad_os = 0;
    oks_vbs_end = 0;
  end

  oks_max_prolong = datediff(day, oks_kont_last_date, oks_last_sh_date);
  oks_day_prolong = datediff(day, oks_kont_last_date, res_prolong_date_curr);

  oks_is_reschedule = 0;
  if (oks_restruct_date is not null and
      cast(rep_date as date) >= cast(oks_restruct_date as date)) then
    oks_is_reschedule = iif(cast(oks_last_sh_date as date) > cast(oks_kont_last_date as date), 1, 0);

  -- вычисление даты просрочки если имеется
  if (oks_bal_zad_os + oks_bal_zad_perc <= 0) then
  begin
    -- если по графику нет долга, но после срока договора висит накопленный процент,
    -- то просрочу считаем с последней даты платежа
    oks_delinq_date = null;

    if (rep_date > oks_last_sh_date and
        oks_bal_perc > 0 and
        oks_close_state = 0) then
      oks_delinq_date = oks_last_sh_date;
  end

  last_date = oks_last_sh_date;
  days_for_calc_perc = oks_days;

  stop_days = 60;
  if (rep_date >= '30.06.2014') then
  begin
    if (oks_last_sh_date <= '02.05.2014') then
      stop_days = 60;
    else
      stop_days = 61;

    if (rep_date >= '01.01.2018') then
      stop_days = max_days_for_calc_perc_to_dlq + 1;
  end

  if (oks_delinq_date >= '01.11.2017') then
    stop_days = max_days_for_calc_perc_to_dlq + 1;

  if (oks_last_sh_date >= '01.01.2013') then
    last_date = dateadd(day, stop_days, last_date);

  if (oks_delinq_date is not null) then
    last_date = minvalue(last_date, dateadd(day, stop_days, oks_delinq_date));

  if (last_date > oks_last_sh_date) then
    last_date = oks_last_sh_date;

  if (oks_date_stop_perc is null) then
    date_stop_perc = last_date;
  else
    date_stop_perc = oks_date_stop_perc;

  if (date_stop_perc > oks_last_sh_date) then
    date_stop_perc = oks_last_sh_date;

  adv_kommis_date = oks_last_sh_date;

  -- комиссию начисляем до окончания первоначального графика
  if (rep_date >= '01.04.2020' and
      oks_kont_last_date < adv_kommis_date) then
    adv_kommis_date = oks_kont_last_date;

  if (oks_date_stop_perc is not null and
      cast(oks_date_stop_perc as date) < cast(adv_kommis_date as date)) then
  begin
    if (cast(oks_date_stop_perc as date) > cast('31.08.2018' as date)) then
      adv_kommis_date = cast(oks_date_stop_perc as date);
    else
      adv_kommis_date = cast('31.08.2018' as date);
  end

  if (oks_close_state = 1 and
      rep_date < adv_kommis_date) then
    adv_kommis_date = rep_date;

  if (oks_close_state = 1 and
      prv_bal_kommis <> 0) then
    adv_kommis_date = rep_date;

  if (rep_date > date_stop_perc) then
    days_for_calc_perc = date_stop_perc - prv_date;

  if (days_for_calc_perc < 0) then
    days_for_calc_perc = 0;

  oks_days = days_for_calc_perc;

  if (oks_is_fix_bal_perc = 1) then
  begin
    -- если конец месяца, то сумму начисленного процента корректируем на зафиксированный накопленный процент
    oks_acr_perc = oks_bal_perc_fixed - prv_bal_perc - oks_pmt_perc_norm - oks_pmt_perc_delinq - oks_pmt_perc_adv - oks_off_perc;
  end
  else
  begin
    -- процент по номинальной ставке, округляем до целого.
    oks_acr_perc = round((cast(prv_bal_to_perc as double precision) * prv_percent_norm / 100.0) * oks_days / 365.0, 0);

    if (oks_is_sh = 1) then
    begin
      if (abs(oks_acr_perc - (maxvalue(res_sh_perc, oks_bal_zad_perc) - prv_bal_perc)) <= 2.0) then
        oks_acr_perc = maxvalue(res_sh_perc, oks_bal_zad_perc) - prv_bal_perc;
    end
    else
    begin
      if (oks_is_pmt = 1) then
      begin
        if (abs(oks_acr_perc + prv_bal_perc + oks_pmt_perc_norm + oks_pmt_perc_delinq + oks_pmt_perc_adv + oks_off_perc) <= 2.0) then
          oks_acr_perc = -(prv_bal_perc + oks_pmt_perc_norm + oks_pmt_perc_delinq + oks_pmt_perc_adv + oks_off_perc);
      end
    end
    if (abs(oks_acr_perc) <= 1) then
      oks_acr_perc = 0;

    if (oks_close_state = 2) then
    begin
      if (prv_bal_perc = 0 and
          oks_pmt_perc_norm = 0 and
          oks_pmt_perc_delinq = 0 and
          oks_pmt_perc_adv = 0 and
          oks_pmt_perc_off = 0 and
          coalesce(oks_bal_perc_fixed, 0) = 0) then
        oks_acr_perc = 0;
    end

    if (oks_close_state = 1 and
        oks_is_fix_bal_perc = 0) then
    begin
      oks_acr_perc = -prv_bal_perc - oks_pmt_perc_norm - oks_pmt_perc_delinq - oks_pmt_perc_adv - oks_off_perc;

      -- Для реализации продуктов easy_start
      if (rep_date >= cast('21.04.2026' as date)) then
        oks_acr_perc = oks_acr_perc - oks_rvs_perc;
    end

    oks_acr_perc = oks_acr_perc + oks_rvs_perc; -- увеличиваем или уменьшаем начисленные проценты на суммы сторнирования
  end

  -- Определяем количество дней для начисления процентов по ЭСП
  days_for_calc_vbs_perc = oks_days;

  if (rep_date >= date_stop_perc) then
    days_for_calc_vbs_perc = date_stop_perc - prv_date;
  if (days_for_calc_vbs_perc < 0) then
    days_for_calc_vbs_perc = 0;

  if (oks_delinq_date is not null and
      oks_days > 0) then
  begin
    tmp_day = 0;
    if (rep_date >= '01.04.2019' and
        oks_close_state = 0 and
        oks_delinq_days >= max_days_for_calc_perc_to_dlq) then
      tmp_day = 1;

    if (oks_delinq_days > (max_days_for_calc_perc_to_dlq + tmp_day)) then
    begin
      if (datediff(day, oks_delinq_date, prv_date) <= (max_days_for_calc_perc_to_dlq + tmp_day)) then
        days_for_calc_vbs_perc = (max_days_for_calc_perc_to_dlq + tmp_day) - datediff(day, oks_delinq_date, prv_date);
      else
        days_for_calc_vbs_perc = 0;
    end
  end

  -- срочный баланс на конец дня
  if (oks_is_flat = 0) then
  begin
    oks_bal_to_perc = oks_balance - oks_bal_zad_os - res_bal_moved_os;
    if (k_is_vacated = 1) then
    begin
      ov_os = 0;

      select sum(ov_os)
      from opervac, operlist ol
      where ov_from_k_id = :k_id and
            ov_type = 1 and
            ov_from_ol_id = ol.ol_id and
            ol.ol_date <= :rep_date and
            ol.is_delete = 0
      into ov_os;

      if (ov_os is null) then
        ov_os = 0;

      oks_bal_to_perc = oks_bal_to_perc - ov_os;
      if (oks_bal_to_perc < 0) then
       oks_bal_to_perc = 0;
    end
  end
  else
    oks_bal_to_perc = prv_bal_to_perc;


  -- расчет доп.процента за дни
  kommis = 0;
  if (oks_is_fix_bal_kommis = 1) then
  begin
    -- если конец месяца, то сумму начисленного доп.процента корректируем на зафиксированный накопленный доп.процент
    kommis = prv_bal_kommis - oks_pmt_kommis - oks_bal_kommis_fixed;
  end
  else
  if (rep_date <> prv_date and
      prv_bal_kommis <> 0) then
  begin
    if (prv_bal_kommis > 0) then
    begin
      total_days = datediff(day, prv_date, prv_last_sh_date);
      if (total_days < 0) then
        total_days = 0;

      if (total_days <> 0) then
      begin
        kommis = prv_bal_kommis / cast(total_days as curr_type);
        kommis = round(kommis * cast(oks_orig_days as curr_type), 2);
        if (kommis > prv_bal_kommis) then
          kommis = prv_bal_kommis;
        else
        if ((prv_bal_kommis - kommis) <= 5) then
          kommis = prv_bal_kommis;
      end
      else
        kommis = prv_bal_kommis;
    end
    else
      kommis = prv_bal_kommis;
  end

  if (rep_date <= '31.12.2011' and
      oks_pmt_kommis <> 0) then
    kommis = -oks_pmt_kommis;

  if (prv_bal_kommis = 0 and
      oks_pmt_kommis = 0 and
      coalesce(oks_bal_kommis_fixed, 0) = 0 and
      oks_close_state = 2) then
    kommis = 0;

  oks_acr_kommis = kommis;

  -- если кредит закрыт досрочно, доначисляем остаток доп.процент
  oks_acr_kommis_adv = 0;
  if (rep_date = adv_kommis_date) then
    oks_acr_kommis_adv = prv_bal_kommis - kommis;

  -- фактически баланс на конец дня
  oks_bal_kommis = prv_bal_kommis - oks_pmt_kommis - oks_acr_kommis_adv - oks_acr_kommis;

  estimate_bal_end = prv_vbs + oks_acr_perc + oks_acr_kommis + oks_acr_kommis_adv + oks_pmt_all - oks_pmt_other_adv;

  -- если ожидаемая сумма на конец периода между 0 и -5 руб, поправляем процент чтобы сумма обнулилась
  if (rep_date >= '01.01.2018' and
      oks_is_fix_bal_perc = 0 and
      oks_close_state = 0) then
  begin
    if (estimate_bal_end < 0 and
        estimate_bal_end >= -5) then
      oks_acr_perc = oks_acr_perc - estimate_bal_end;
  end

  -- Накопленные проценты с учетом начисленных
  oks_bal_perc = oks_bal_perc + oks_acr_perc;

  oks_zalog_is_bankrupt = 0;
  oks_is_not_targeted = 0;
  oks_is_zalog = 0;
  oks_is_zalog_llr = 0;
  oks_zalog_cost = 0;
  zalog_c_id = null;

  if (pr_code containing '.zalog_car.' or pr_code containing '.zalog.') then
  begin
    select z_from_c_id
    from zalog
    where z_from_k_id = :k_id and
          :rep_date between coalesce(z_date_begin, '01.01.1900') and coalesce(z_date_end, '31.12.9999') and
          z_type = 1 and
          z_from_c_id is not null
    order by z_id desc
    rows 1
    into zalog_c_id;

    if (zalog_c_id is not null) then
      oks_zalog_is_bankrupt = f_bankrupt(c_id, rep_date);
    if (oks_zalog_is_bankrupt = -1) then
      oks_zalog_is_bankrupt = 0;

    sab_id = 90; -- sab_full_code = 'Микрозаймы.Внебаланс.Залог'
    oks_is_zalog = 1;

    oks_zalog_cost = -f_get_acc_amount(:kc_id, :sab_id, :rep_date, :rep_date, 'bal');  -- Залоговая стоимоть

    if (rep_date < date_20221231) then
    begin
      if (oks_zalog_cost >= oks_balance and oks_zalog_is_bankrupt = 0) then
        oks_is_zalog_llr = 1;
    end else
    if (oks_zalog_cost >= (oks_balance + oks_bal_perc) and oks_zalog_is_bankrupt = 0) then
      oks_is_zalog_llr = 1;
  end

  if (oks_llr_percent is null) then
    oks_llr_percent = 0;

  if (oks_llr_base_percent is null) then
    oks_llr_base_percent = 0;

  -- если РВПЗ зафиксирован, процентную ставку не меняем
  if (oks_is_fix_llr_os = 0 and oks_is_fix_llr_perc = 0) then
  begin
    if (rep_date < '01.09.2020') then
    begin
      tmp_is_resh = oks_is_reschedule;

      oks_llr_group = oks_llr_kind;
      if (tmp_is_resh = 1) then
        oks_llr_group = oks_llr_group + 3;

      select oks_llr_percent
      from hp_kn_oksummary_llr_percent(:rep_date, :oks_disb_date, :oks_llr_group, :oks_delinq_days, :oks_is_reschedule)
      into oks_llr_percent;
      oks_llr_base_percent = oks_llr_percent;
    end else
    begin
      tmp_is_resh = 0;
      if (oks_is_reschedule_llr = 1 or oks_is_refund_llr = 1) then
        tmp_is_resh = 1;
  
      oks_llr_kind = 3;
      if (oks_is_zalog_llr = 1)
      then oks_llr_kind = 2;
      else
        if (kc_amount_approved <= 30000 and tmp_period_day <= 30)
        then oks_llr_kind = 1;

      oks_llr_group = oks_llr_kind;

    oks_llr_os_nadzor = round(cast(oks_balance as double precision) * oks_llr_percent_nadzor / 100.0, 2);
    oks_llr_perc_nadzor = round(cast(oks_bal_perc as double precision) * oks_llr_percent_nadzor / 100.0, 2);
    oks_ecl_vbs_nadzor = oks_ecl_vbs - oks_llr_os_nadzor - oks_llr_perc_nadzor;
  end

  -- до 1 февраля 2019 года РО считаем по старому на ВБС 4
  if (rep_date <= '31.01.2019') then
  begin
    oks_ead_future_perc = 0;
    oks_ead_future_vbs = 0;
    oks_ead_amount = oks_bal_esp_end;
  end

  -- резерв по госпошлинам
  oks_reserv_law = round(cast(oks_bal_law as double precision) * oks_ecl_percent / 100.0, 2);

  -- если займ выдан до вступления в реестр, недоначисляем резерв на 1 руб при 100%-ой ставке
  if ((oks_ecl_percent = 100.0 or
      oks_close_state <> 0) and
      oks_bal_law > 0) then
  begin
    if (oks_bal_law >= 1) then
      oks_reserv_law = oks_bal_law - 1;
    else
      oks_reserv_law = oks_bal_law - 0.01;
  end

  oks_reserv_percent = maxvalue(oks_ecl_percent, oks_llr_percent_outer);  -- Ставка резерва, новая, общая

  if (db_id = 39801 and rep_date >= cast('31.10.2025' as date)) then
    oks_reserv_percent = oks_ecl_percent;  -- для КЗ берем только ставку РО

  if (rep_date >= date_20240331) then
  begin
    if (oks_is_fix_ecl_vbs = 0) then
    begin
      oks_ecl_vbs_fixed = 0;
  
      oks_ecl_vbs = round(cast(oks_ead_amount as double precision) * oks_reserv_percent / 100.0, 2);  -- РО
  
      if (oks_close_state = 1 and
          oks_diff_esp_bal <= 0) then
        oks_ecl_vbs = prv_diff_ecl_llr;
    end
    else
      oks_ecl_vbs = oks_ecl_vbs_fixed;

    -- Резерв НУ ОД
    if (oks_is_fix_llr_os = 0) then
    begin
      oks_llr_os_fixed = 0;
      oks_llr_os = round(cast(oks_balance as double precision) * oks_reserv_percent / 100.0, 2);
    end
    else
      oks_llr_os = oks_llr_os_fixed;

    -- Резерв НУ %%
    if (oks_is_fix_llr_perc = 0) then
    begin
      oks_llr_perc_fixed = 0;
      oks_llr_perc = round(cast(oks_bal_perc as double precision) * oks_reserv_percent / 100.0, 2);
    end
    else
      oks_llr_perc = oks_llr_perc_fixed;

    oks_llr_os_nadzor = round(cast(oks_balance as double precision) * oks_reserv_percent / 100.0, 2);
    oks_llr_perc_nadzor = round(cast(oks_bal_perc as double precision) * oks_reserv_percent / 100.0, 2);
    oks_ecl_vbs_nadzor = oks_ecl_vbs - oks_llr_os_nadzor - oks_llr_perc_nadzor;

    -- резерв по госпошлинам
    oks_reserv_law = round(cast(oks_bal_law as double precision) * oks_reserv_percent / 100.0, 2);

    -- если займ выдан до вступления в реестр, недоначисляем резерв на 1 руб при 100%-ой ставке
    if ((oks_reserv_percent = 100.0 or
        oks_close_state <> 0) and
        oks_bal_law > 0) then
    begin
      if (oks_bal_law >= 1) then
        oks_reserv_law = oks_bal_law - 1;
      else
        oks_reserv_law = oks_bal_law - 0.01;
    end
  end

  oks_diff_ecl_llr = oks_ecl_vbs - oks_llr_os - oks_llr_perc;

  if (oks_actual_last_date is null) then
    oks_actual_last_date = prv_last_sh_date;

  suspend;
end
