CREATE PROCEDURE MC_CITY_OFFICE_LIST(
    MCL_ID ID_TYPE = null)
RETURNS (
    ID ID_TYPE,
    NAME STR_255,
    IS_DEFAULT D_SMALLINT)
AS
begin
   is_default = 0;

   for select distinct pl.pl_id, pl.pl_name
   from places pl
   inner join addrlist adl on adl.adl_from_pl_id = pl.pl_id
   inner join region reg on reg.reg_address_fact = adl.adl_id
   where reg.reg_date_close is null and
         reg.is_delete = 0 and
         reg.reg_status = 3
   order by pl.pl_name
   into id, name
   do
    suspend;

   /* в данной процедуре нужно добавить условие при котором is_default vj;tn ,snm  = 1

      По умолчанию is_default  =0. Чз mcl_id получаю c_id и если клиента был активный или закрытый заем в течении 3 месяцев по отношению к текущей дате, тогда получаю самый последний заем (ближайший к текущей дате), из займа получаю офис, город и для этой записи is_default = 1.
    
      Для Тилека is_default  = 1 значит, что в контроле со списком городов,  город с is_default = 1 должен быть автоматом выставлен.
      Записей с is_default  = 1 не может быть более 1-ой, но может быть что записей с is_default  = 1 нет вообще.
   */
end
