CREATE PROCEDURE HP_CONTVERI_UPDATE(
    CV_OBJECT_KIND VARCHAR(100),
    CV_OBJECT_ID DECIMAL(18,0),
    CV_FROM_RD_ID VARCHAR(255),
    CV_CONTACT_VALUE VARCHAR(255),
    CV_PASSWORD VARCHAR(255),
    CV_ACTION VARCHAR(100),
    CV_ACTION_TEXT VARCHAR(255),
    CV_FROM_UN_ID DECIMAL(18,0))
RETURNS (
    CV_ERROR_TEXT VARCHAR(255))
AS
declare variable db_id d_integer;
declare variable edoc_status d_str_code;
declare variable rd_code d_str_code;
declare variable need_asp d_smallint;
declare variable edc_id d_integer;
declare variable clt_id d_integer;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable is_hr d_bool;
declare variable asp_const d_smallint;
begin
 asp_const = f_get_const('asp_const');
  -- Процедура выполняет все манипуляции с верификацией контактных данных: регистрация, подтверждение
  -- Если все нормально прошло, то результат CV_ERROR_TEXT=''

  -- В процедуре обрабатываеются только клиенты (Clients).
  -- Если нужны другие типы, используйте новые процедуры:
  -- hp_extsyspin_send для отправки кода
  -- hp_extsyspin_verify для проверки кода

  if (f_objkind(cv_object_kind, 'id') <> 0) then
    exception e_default 'Invalid object_kind. Use hp_extsyspin_send/hp_extsyspin_verify';

  if (coalesce(cv_action, '') = '') then
    exception e_default 'cv_action is null';

  if (cv_action_text is null) then
    cv_action_text = '';

  cv_error_text = '';

  db_id = f_get_db_id();
  if (cv_from_un_id is not null) then
    uxf$set_user(cv_from_un_id);

  if (cv_action in ('error_on_send', 'cancel', 'verify_income')) then
    exception e_default 'Found disabled action: ' || cv_action;

  -- если входящая команда не учтена в этой процедуре, вызываем ошибку
  if (cv_action not in ('register', 'verify_password', 'verify_manually')) then
    cv_error_text = 'Не распознана команда для выполнения: ' || cv_action;

  -- отключаем возможность подтверждения через сотрудника
  if (cv_action in ('verify_password', 'verify_manually')) then
  begin
    if (db_id = 94401)
    then cv_error_text = f_format('Müştəri SMS-dəki linki izləyərək nömrəni müstəqil olaraq təsdiqləməlidir və ya SMS-dən %1 nömrəsinə kod göndərməlidir', f_get_const('company_cell'));
    else cv_error_text = f_format('Клиент должен самостоятельно подтвердить номер, перейдя по ссылке в смс, или отправить код из смс на номер %1', f_get_const('company_cell'));
  end

  if (cv_error_text = '' and coalesce(cv_contact_value, '') = '') then
    cv_error_text = 'Контактная информация не заполнена';

  -- если тип объекта передан через название, определяем числовой код
  cv_object_kind = f_objkind(cv_object_kind, 'id');

  -- Если была ошибка, завершаем работу
  if (cv_error_text <> '') then
  begin
    suspend;
    exit;
  end


  begin
    rd_code = f_get_rd_code(cv_from_rd_id);
    when any do
    begin
      rd_code = cv_from_rd_id;
    end
  end

  if (rd_code = 'ContactInfo.Cell_Main') then
  begin
    for select c_id, c_name_fio
        from hp_find_client_by_continfo(null, :cv_contact_value)
        where c_id <> :cv_object_id
        into :c_id, :c_name_fio
    do
    begin
      if (exists(select *
                 from clientstate
                 where cs_from_c_id = :c_id and
                       cs_staff_state = 1)) then
      begin
        cv_error_text = 'Данный номер закреплен за сотрудником' || ' ' || :c_name_fio || ' ' || '(' || :cv_contact_value || ') ' || 'обратитесь в кадровый отдел за уточнением.';
        suspend;
        exit;
      end
    end
  end

  if (rd_code = 'ContactInfo.Cell_Add') then
  begin
    cv_error_text = 'Верификация доп. номера не требуется';
    suspend;
    exit;
  end

  if (db_id in (39801, 41701, 81001) and rd_code = 'ContactInfo.Cell_Main') then
  begin
    -- KG: Если клиент активный или повторный, требуется заявление на смену номера
    -- данное заявление подписывается онлайн и вручную
    -- RU: Если у клиента есть подтвержденный основной номер, тогда нужно
    -- заявление на смену номера
    if (db_id in (39801, 41701) and exists(select * from clientstate
                                 where cs_from_c_id = :cv_object_id
                                   and cs_client_state in (1, 2))
    or (db_id = 81001 and exists(select * from contdata
                                where cda_from_c_id  = :cv_object_id
                                  and cda_cont_kind = 0
                                  and cda_from_clt_id <> 0
                                  and cda_status = 1
                                  and cda_is_main = 1))) then
    begin
      if (f_edoc_state2('cell.application', current_date, cv_from_un_id) = 'active') then
      begin
        select res_text
        from hp_edoc_ins_cell_application(:cv_object_id, :cv_from_un_id)
        into cv_error_text;
  
        suspend;
        exit;
      end
    end

    if (f_edoc_state2('asp.agreement', current_date, cv_from_un_id) = 'active' and asp_const = 0) then
    begin
      edoc_status = f_edoc_status('asp.agreement', cv_object_id, current_date);
  
      need_asp = 0;
      if (edoc_status in ('empty', 'unsigned', 'cancelled')) then
         need_asp = 1;

      if (db_id in (39801, 41701) and edoc_status = 'signed') then
      begin
        select edc_id
        from edoc_cfg
        where edc_code = 'asp.agreement'
        into edc_id;

        clt_id = f_cont_id(cv_contact_value, 'add');

        -- если на указанном номере нет договора АСП, формируем новую
        if (not exists(select * from edoc_list, edoc_sign, edoc_msg
        where edl_from_edc_id = :edc_id and
              edl_object_kind = f_objkind('clients', 'id') and
              edl_object_id = :cv_object_id and
              current_date between edl_blank_date and coalesce(edl_date_end, current_date) and
              edl_status = 1 and
              edl_id = eds_from_edl_id and
              eds_from_c_id = :cv_object_id and
              edm_from_eds_id = eds_id and
              edm_from_clt_id = :clt_id)) then
          need_asp = 1;
      end

      if (need_asp = 1) then
      begin
        if (db_id in (39801, 41701)) then
          is_hr = (select has_access from hp_permission_access_check(:cv_from_un_id, 'PermActions', 0, 'FinStart.CanEditCellNo'));

        if (db_id in (39801, 41701) and is_hr = 0) then
        begin
          if (edoc_status = 'signed') then -- Аннулируем, т.к. номер изменился
            execute procedure hp_edoc_cancel('asp.agreement', 'clients', :cv_object_id);

          cv_error_text = 'Запрещено. Используйте ссылку для подписания договора ЭЦП';
          suspend;
          exit;
        end
        else
        begin
          execute procedure hp_edoc_cancel('asp.agreement', 'clients', :cv_object_id);
  
          select res_text
          from hp_edoc_ins_asp_agreement(:cv_object_id, :cv_from_un_id)
          where res_is_error = 1
          into cv_error_text;
    
          suspend;
          exit;
        end
      end
    end
  end

  select res_text
  from hp_contveri_send(:cv_object_id, :rd_code, :cv_contact_value, 0, :cv_from_un_id)
  where res_is_error = 1
  into :cv_error_text;

  suspend;
end
