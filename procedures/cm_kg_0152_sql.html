CREATE PROCEDURE CM_KG_0152_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable dt d_date;
declare variable pr_code d_str_100;
declare variable ma_id d_id;
declare variable c_id d_id;
declare variable temp_descr_of_loan d_str_255;
declare variable loan_description d_str_255;
begin
  if (exists(select *
             from expressapp exa
             where exa.exa_from_k_id = :k_id)) then
    exit;

  select pr.pr_code, kt.k_from_ma_app_id
  from kontract kt
  inner join product pr on pr.pr_id = kt.k_from_pr_id
  where kt.k_id = :k_id
  into pr_code, ma_id;

  if (pr_code containing '.express_similar.') then
    exit;

  dt = (select max(kl_datetime)
        from kontlog
        where kl_from_k_id = :k_id and
              kl_state = 1);
  dt = coalesce(dt, current_date);
  if (f_get_const('clp_active') = 1) then
  begin
    select mac.mac_from_c_id
    from miniappclient mac
    where mac.mac_from_ma_id = :ma_id and
          mac.mac_amount > 0 and
          mac.mac_is_cashier = 1
    into c_id;
    temp_descr_of_loan = '';
    loan_description = '';
    -- генерирую новое значение для descr_of_loan
    select clpq.cq_name || '; ' || clpc.cc_name
    from clp_miniappclient clpm
    inner join clp_questions clpq on clpm.cmc_questions_id = clpq.cq_id
    inner join clp_clarification clpc on clpm.cmc_clarification_id = clpc.cc_id
    where clpm.cmc_from_c_id = :c_id and
          clpm.cmc_from_ma_id = :ma_id
    into :loan_description;

    loan_description = coalesce(loan_description, '');

    if (:pr_code containing '.ipf.') then
    begin
      for select ml.ml_name
          from kontmat km
          inner join matlist ml on ml.ml_id = km.kml_from_ml_id
          where km.kml_object_kind = 9 and
                km.kml_object_id = :k_id
          into temp_descr_of_loan
      do
        :temp_descr_of_loan = :temp_descr_of_loan || '; ' || :temp_descr_of_loan;

      :loan_description = :loan_description || '; ' || :temp_descr_of_loan;
    end
    if (coalesce(:loan_description, '') = '') then
    begin
      res_error_text = 'Подробное описание по цели не заполнено!';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
  end
  else
  if (exists(select *
             from knclient
             inner join hp_objectprops_anketa_get('Clients', kc_from_c_id, -1, 'AnketaQuestions.descr_of_loan', :dt, null) on res_str_values is null or res_str_values = 'значение'
             where kc_from_k_id = :k_id and
                   kc_amount_approved > 0)) then
  begin
    res_error_text = 'Подробное описание по цели не заполнено!';
    res_error_is_critical = 2;
    suspend;
    exit;
  end
end
