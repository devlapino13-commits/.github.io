CREATE PROCEDURE CM_RU_0236_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable edl_blank_date d_date;
declare variable selfemployed_start_date d_date;
declare variable business_start_date d_date;
declare variable product_start_date d_date;
declare variable pr_code d_str_100;
begin
    selfemployed_start_date = cast(f_get_const('selfemployed_start_date') as date);
    business_start_date = cast(f_get_const('business_start_date') as date);

    if (f_tmp_const('use_perstemp') = 1) then
    exit;

    for select mac.mac_from_c_id, pr.pr_code
    from miniapp ma
    inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
    inner join product pr on pr.pr_id = ma.ma_from_pr_id
    where ma.ma_id = :ma_id and
          ma.ma_is_active = 1 and
          ma.ma_date_end >= '31.12.9999' and
          mac.mac_amount > 0 and
          (pr.pr_code containing '.selfemployed.' or
          pr.pr_code containing '.business.')
    into :c_id, :pr_code
    do
    begin
        product_start_date = cast('31.12.9999' as date);
        
        if (pr_code containing '.selfemployed.') then
            product_start_date = selfemployed_start_date;
        else if (pr_code containing '.business.') then
            product_start_date = business_start_date;
        
        if (product_start_date = cast('31.12.9999' as date)) then
            exit;
        
        edl_blank_date = null;
        select edl.edl_blank_date
        from edoc_sign eds
        inner join edoc_list edl on edl.edl_id = eds.eds_from_edl_id and
            edl.edl_object_kind = f_objkind('ft_anketa_list', 'id') and
            edl.edl_status = 1
        where eds.eds_from_c_id = :c_id
        order by edl.edl_id desc
        rows 1
        into edl_blank_date;
        
        if (coalesce(edl_blank_date, cast('01.01.1900' as date)) < product_start_date) then
        begin
            res_info_text = '';
            select cl.c_name_fio
            from clients cl
            where cl.c_id = :c_id
            into res_info_text;
            
            res_error_text = 'У клиента анкета-опросник старого образца, требуется подписать анкету-опросник нового образца';
            res_error_is_critical = 1;
            suspend;
        end
    end
end
