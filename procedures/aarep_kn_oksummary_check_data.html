CREATE PROCEDURE AAREP_KN_OKSUMMARY_CHECK_DATA
RETURNS (
    REP_LINE INTEGER,
    REP_NAME D_STR_255,
    REP_DATE DATE,
    REP_AMOUNT CURR_TYPE)
AS
declare variable start_month date;
begin
  rep_date = '31.12.2017';
  start_month = rep_date - extract(day from rep_date) + 1;

  while (rep_date <= '30.09.2018') do
  begin
    rep_line = 1;
    rep_name = 'Займы, где просроченный ОД меньше баланса';
    rep_amount = 0;
    select count(*)
    from kn_oksummary
    where oks_date_begin = :rep_date and
          oks_bal_zad_os > oks_balance
    into rep_amount;
    if (rep_amount is null) then
      rep_amount = 0;
    suspend;

    rep_line = 2;
    rep_name = 'Договора, где баланс участников с разными знаками';
    rep_amount = 0;
    with tab
    as (select oks_from_k_id, sum(oks_balance) as sum_balance, min(oks_balance) as min_balance
        from kn_oksummary
        where oks_date_begin = :rep_date
        group by 1)
    select count(*)
    from tab
    where (sum_balance > 0 and
          min_balance <= 0) or
          (sum_balance = 0 and
          min_balance < 0)
    into rep_amount;
    if (rep_amount is null) then
      rep_amount = 0;
    suspend;

    rep_line = 3;
    rep_name = 'Договора, где просроченные дни участников разные';
    rep_amount = 0;
    with tab
    as (select oks_from_k_id, max(coalesce(oks_delinq_date, '01.01.1990')) as max_delinq_date,
               min(coalesce(oks_delinq_date, '01.01.1990')) as min_delinq_date
        from kn_oksummary
        where oks_date_begin = :rep_date
        group by 1)
    select count(*)
    from tab
    where max_delinq_date <> min_delinq_date
    into rep_amount;
    if (rep_amount is null) then
      rep_amount = 0;
    suspend;

    rep_line = 4;
    rep_name = 'Займы, где на момент закрытия линейный ВБС < 0';
    rep_amount = 0;
    select count(*)
    from kn_oksummary
    where oks_date_begin between :start_month and :rep_date and
          oks_close_state = 1 and
          oks_vbs_end < 0
    into rep_amount;
    if (rep_amount is null) then
      rep_amount = 0;
    suspend;

    rep_line = 5;
    rep_name = 'Займы, где на момент закрытия ВБС ЕПС < 0';
    rep_amount = 0;
    select count(*)
    from kn_oksummary
    where oks_date_begin between :start_month and :rep_date and
          oks_close_state = 1 and
          oks_bal_esp_end < 0
    into rep_amount;
    if (rep_amount is null) then
      rep_amount = 0;
    suspend;

    rep_line = 6;
    rep_name = 'Займы, где на момент закрытия ВБС ЕПС < линейный ВБС';
    rep_amount = 0;
    select count(*)
    from kn_oksummary
    where oks_date_begin between :start_month and :rep_date and
          oks_close_state = 1 and
          oks_bal_esp_end < oks_vbs_end
    into rep_amount;
    if (rep_amount is null) then
      rep_amount = 0;
    suspend;

    rep_line = 7;
    rep_name = 'Займы, где РО больше ВБС ЕПС';
    rep_amount = 0;
    select count(*)
    from kn_oksummary
    where oks_date_begin = :rep_date and
          oks_ecl_vbs > oks_bal_esp_end
    into rep_amount;
    if (rep_amount is null) then
      rep_amount = 0;
    suspend;

    rep_line = 8;
    rep_name = 'Займы, где РО больше ВБС ЕПС более чем 1 руб.';
    rep_amount = 0;
    select count(*)
    from kn_oksummary
    where oks_date_begin = :rep_date and
          oks_ecl_vbs > oks_bal_esp_end + 1
    into rep_amount;
    if (rep_amount is null) then
      rep_amount = 0;
    suspend;

    rep_date = dateadd(month, 1, dateadd(day, 1, rep_date)) - 1;
    start_month = rep_date - extract(day from rep_date) + 1;
  end
end
