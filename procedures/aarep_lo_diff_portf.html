CREATE PROCEDURE AAREP_LO_DIFF_PORTF(
    RES_LO_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    LAST_DATE_PRV_MONTH DATE,
    REP_DATE_BEGIN DATE,
    K_ID D_ID,
    K_KONTRACT_NO D_STR_100,
    K_DIFF_BALANCE D_CURRENCY,
    K_DIFF_BALANCE_PAR30 D_CURRENCY,
    K_BALANCE_BEGIN D_CURRENCY,
    K_DELINQ_DAYS_BEGIN D_CURRENCY,
    K_BALANCE_END D_CURRENCY,
    K_DELINQ_DAYS_END D_CURRENCY,
    CLOSED_WRITOFF D_SMALLINT,
    TRANSFERRED_OTHER D_SMALLINT,
    TRANSF_BALANCE D_CURRENCY,
    ACCEPTED_BALANCE D_CURRENCY,
    KAL_DATE DATE,
    KAL_PRIOR_LO_ID D_ID,
    KAL_NEW_KF_OFFICER D_ID,
    KAL_DATE_DEL_DAYS D_INTEGER,
    KAL_DATE_BALANCE D_CURRENCY,
    K_FACT_DISB_LO_ID D_ID,
    K_BALANCE_PAR30_BEGIN D_CURRENCY,
    K_BALANCE_PAR30_END D_CURRENCY,
    K_BALANCE_0_30_BEGIN D_CURRENCY,
    K_BALANCE_0_30_END D_CURRENCY,
    K_DIFF_BALANCE_0_30 D_CURRENCY,
    K_WRITEOFF_SUM D_CURRENCY)
AS
declare variable kf_id d_id;
begin
  last_date_prv_month = rep_date_end - extract(day from rep_date_end);
  rep_date_begin = last_date_prv_month + 1;

  --2. Прирост портяеля без просрочки и с просрочкой 0-30 дней по итогам отчетоного месяца
  k_balance_begin = null;
  k_delinq_days_begin = null;
  k_balance_end = null;
  k_delinq_days_end = null;
  transf_balance = null;
  for with t_kont_info
      as (select oks_from_k_id, --
                 oks_balance as k_balance_begin, --
                 :last_date_prv_month - oks_delinq_date as k_delinq_days_begin, --
                 0 as k_balance_end, --
                 0 as k_delinq_days_end
          from oksummary
          where :last_date_prv_month between oks_date_begin and oks_date_end and
                oks_kc_id = :res_lo_id and
                oks_close_state = 0
          union all
          select oks_from_k_id, --
                 0 as k_balance_begin, --
                 0 as k_delinq_days_begin, --
                 oks_balance as k_balance_end, --
                 :rep_date_end - oks_delinq_date as k_delinq_days_end
          from oksummary
          where :rep_date_end between oks_date_begin and oks_date_end and
                oks_kc_id = :res_lo_id and
                oks_close_state = 0)
      select oks_from_k_id, --
             sum(k_balance_begin) as k_balance_begin, --
             sum(k_delinq_days_begin) as k_delinq_days_begin, --
             sum(k_balance_end) as k_balance_end, --
             sum(k_delinq_days_end) as k_delinq_days_end
      from t_kont_info
      group by 1
      into k_id, k_balance_begin, k_delinq_days_begin, k_balance_end, k_delinq_days_end
  do
  begin
    --  if (k_id = 109010012226700801) then
    --  k_id = 109010012226700801;
    k_diff_balance = 0;
    k_diff_balance_0_30 = 0;
    k_balance_0_30_begin = 0;
    k_balance_0_30_end = 0;
    k_balance_par30_begin = 0;
    k_balance_par30_end = 0;
    k_diff_balance_par30 = 0;
    k_balance_begin = coalesce(k_balance_begin, 0);
    k_balance_end = coalesce(k_balance_end, 0);
    closed_writoff = 0;
    transferred_other = 0;
    kal_date_balance = 0;
    kal_date_del_days = 0;
    accepted_balance = 0; --Принятый баланс
    transf_balance = 0; --Переденный баланс

    k_kontract_no = null;
    k_fact_disb_lo_id = null;
    select k_kontract_no, k.k_from_kf_id, k_fact_disb_lo_id
    from kontract k
    where k_id = :k_id
    into k_kontract_no, kf_id, k_fact_disb_lo_id;

    kal_date = null;
    kal_prior_lo_id = null;
    kal_new_kf_officer = null;
    select kal_date, kal_prior_lo_id, kal_new_kf_officer
    from knaktlist kal
    where kal_from_kf_id = :kf_id and
          kal.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal_prior_lo_id <> kal.kal_new_kf_officer
          --   если сотрудник был уволен и была передача то такой портфел будет минусоваться
          --                and not exists(select *
          --                           from hr_pstatus
          --                           where ps_from_c_id = kal_prior_lo_id and
          --                                 :rep_date_end between ps_date_begin and ps_date_end and
          --                                 ps_status = 1 and
          --                                 ps_is_working = 1))
    into kal_date, kal_prior_lo_id, kal_new_kf_officer;

    if (kal_date between :rep_date_begin and :rep_date_end) then
    begin
      kal_date_balance = null;
      kal_date_del_days = null;
      select oks_balance, --
             :rep_date_end - oks_delinq_date as k_delinq_days_end
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_from_k_id = :k_id and
            oks_close_state = 0
      into kal_date_balance, kal_date_del_days;

      if (coalesce(kal_date_del_days, -1) < 0) then
        kal_date_del_days = 0;
      kal_date_balance = coalesce(kal_date_balance, 0);

      if (res_lo_id <> kal_prior_lo_id and
          res_lo_id <> k_fact_disb_lo_id) then
        accepted_balance = kal_date_balance;

      if (res_lo_id <> kal_new_kf_officer) then
        transf_balance = kal_date_balance;
    end

    if (coalesce(k_delinq_days_begin, -1) < 0) then
      k_delinq_days_begin = 0;
    if (coalesce(k_delinq_days_end, -1) < 0) then
      k_delinq_days_end = 0;

    if (k_delinq_days_begin > 30) then
    begin
      k_balance_par30_begin = k_balance_begin;
    end

    if (k_delinq_days_end > 30) then
    begin
      k_balance_par30_end = k_balance_end;
    end

    if (k_delinq_days_begin <= 30) then
    begin
      k_balance_0_30_begin = k_balance_begin;
    end

    if (k_delinq_days_end <= 30 and
        accepted_balance = 0) then
    begin
      k_balance_0_30_end = k_balance_end;
    end

    --k_diff_balance 2. Прирост портфеля без просрочек выше 30 дней
    if ((k_delinq_days_begin <= 30 or k_delinq_days_end <= 30) and
        accepted_balance = 0) then
    begin
      k_diff_balance_0_30 = k_balance_0_30_end - k_balance_0_30_begin;
    end

    k_writeoff_sum = 0;
    if (k_delinq_days_begin > 30) then
    begin
      select -sum(ok_os)
      from operkred ok
      where :k_id = ok_from_k_id and
            ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_type = 5 --списания
      into k_writeoff_sum;
      k_writeoff_sum = coalesce(k_writeoff_sum, 0);

      if (k_writeoff_sum > 0) then
      begin
        k_balance_par30_end = k_balance_begin;
        closed_writoff = 1;
      end

      if (exists(select *
                 from sb_akt sa
                 where sa.sba_from_k_id = :k_id and
                       :rep_date_end between sa.sba_date_begin and sa.sba_date_end and
                       sa.sba_is_delete = 0) --
          or res_lo_id <> kal_new_kf_officer) then
      begin
        transferred_other = 1;
        k_balance_par30_end = k_balance_begin;
      end
    end

    if ((k_delinq_days_begin > 30 or k_delinq_days_end > 30) and
        transf_balance = 0 and
        accepted_balance = 0) then
    begin
      k_diff_balance_par30 = k_balance_par30_end - k_balance_par30_begin; --
    end

    k_diff_balance = k_balance_end - k_balance_begin;

    suspend;
  end
end
