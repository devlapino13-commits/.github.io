CREATE PROCEDURE HP_CARDLIST_JSON_CHECK(
    CARDPLACE_ID D_INTEGER NOT NULL,
    CARD_LIST D_STR_8100 NOT NULL,
    CARD_COUNT D_INTEGER NOT NULL,
    IS_CARD D_BOOL = 1)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable real_count d_integer;
declare variable first_card d_integer;
declare variable last_card d_integer;
declare variable fact_count d_integer;
declare variable item_count d_integer;
declare variable item_id d_integer;
declare variable first_card_id d_integer;
declare variable last_card_id d_integer;
declare variable total_card d_integer;
declare variable array_card_count d_integer;
declare variable first_len d_integer;
declare variable last_len d_integer;
begin
  -- Процедура проверяет наличие необходимого количества
  -- карты в указанном месте и диапазоне номеров
  res_text = '';
  if (is_card = 0) then
  begin
    suspend;
    exit;
  end

  -- Проверка на пересечение диапазонов в списке карт
  res_text = (select err_text from hp_cardrange_crossingcheck(:card_list));
  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  select count(*)
  from cardlist
  where cal_last_cpl_id = :cardplace_id
  into real_count;

  if (res_text = '' and real_count = 0) then
    res_text = 'В указанном местонахождении - карты отсутствуют.';

  if (card_list = '') then
    res_text = 'Не указан список карт.';

  if (res_text = '') then
  begin
    item_count = f_json_str_get_count(card_list, 'cards');
    item_id = 0;
    total_card = 0;
    while (res_text = '' and item_id < item_count) do
    begin
      first_card = f_json_str_get_value(card_list, 'cards[' || item_id || '].b', null);
      first_len = char_length(first_card);

      last_card = f_json_str_get_value(card_list, 'cards[' || item_id || '].e', null);
      last_len = char_length(last_card);

      if (first_card is null or last_card is null) then
      begin
        res_text = 'Элемент массива не найден.';
        suspend;
        exit;
      end
      if (first_len < 4 or last_len < 4) then
      begin
        res_text = f_format('Номер карты [%1] должен содержать более 3-х знаков.', iif(first_len < 4, first_card, last_card));
        suspend;
        exit;
      end

      array_card_count = cast(left(last_card, last_len - 1) as integer) -
                         cast(left(first_card, first_len - 1) as integer) + 1;

      fact_count = 0;
      for select first_card_id, last_card_id, card_count
      from hp_get_cardrange_list(:cardplace_id, :first_card, :last_card)
      into :first_card_id, :last_card_id, :fact_count
      do
      begin
        if (res_text = '' and first_card <> first_card_id) then
          res_text = f_format('Первая карта %1 отсутствует в указанном месте.', first_card);

        if (res_text = '' and last_card <> last_card_id) then
          res_text = f_format('Последняя карта %1 отсутствует в указанном месте.', last_card);

        if (res_text = '' and array_card_count <> fact_count) then
          res_text = f_format('В указанном диапазоне номеров [%1-%2] количество не совпадает [%3].', first_card, last_card, fact_count);
      end
  
      if (res_text = '' and fact_count = 0) then
        res_text = f_format('В указанном диапазоне номеров [%1-%2] карты отсутствуют.', first_card, last_card);

      if (res_text = '') then
        total_card = total_card + fact_count;

      item_id = item_id + 1;
    end

    if (res_text = '' and total_card <> card_count) then
      res_text = f_format('Списочное количество карт [%1] и введеное [%2] не совпадает.', total_card, card_count) ;
  end

  suspend;
end
