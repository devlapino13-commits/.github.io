CREATE PROCEDURE MC_CREATE_GRS_TASK(
    MCL_ID D_ID)
RETURNS (
    ID D_ID,
    JSON_DATA D_STR_8100)
AS
declare variable eic_id d_id;
declare variable object_kind d_integer;
declare variable drl_id d_id;
declare variable db_is_actual d_smallint;
declare variable object_id d_id;
begin
  object_id = mcl_id;

  db_is_actual = 0;
  if (exists(select *
             from hp_db_is_actual)) then
    db_is_actual = 1;

  select eic.eic_id
  from extinfocfg eic
  where eic.eic_code = 'Tunduk'
  into eic_id;

  object_kind = f_objkind('mc_login', 'id');

  --  if (db_is_actual = 1) then        --При создании задачи по ГРС нам еще не известно c_id
  --  begin
  --    object_kind = f_objkind('clients', 'id');
  --    object_id = (select ml.mcl_from_c_id
  --                 from mc_login ml
  --                 where ml.mcl_id = :mcl_id);
  --  end

  select json_value, drl_id
  from hp_extinfo_to_send_tunduk(:object_id, :object_kind, :object_id)
  into :json_data, :drl_id;

  select eil.eil_id
  from extinfolist eil
  where eil.eil_from_mcl_id = :mcl_id and
        eil.eil_status in (0, 2)
  order by eil.eil_id desc
  rows 1
  into id;

  if ((id is null)) then
  begin
    insert into extinfolist (eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_is_json, eil_from_eip_id,
                             eil_from_un_id, eil_from_mcl_id)
    values (:eic_id, current_date, :object_kind, :object_id, 1, null, 1000000000000, :mcl_id)
    returning eil_id
    into :id;
  end
  suspend;
end
