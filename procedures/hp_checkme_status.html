CREATE PROCEDURE HP_CHECKME_STATUS(
    C_ID D_ID)
RETURNS (
    SERVICE_CODE D_STR_CODE,
    SERVICE_STATUS D_SMALLINT,
    RES_TEXT D_STR_255,
    QUERY_DATE D_DATE,
    ANSWER_DATE D_DATE)
AS
begin
    service_code = '';
    service_status = -1;
    res_text = 'Необходимо отправить запрос в ПОД/ФТ';
    query_date = '01.01.1900';
    answer_date = '01.01.1900';

    select first 1
        cast(eil.eil_create_datetime as d_date),
        cast(eil.eil_status_datetime as d_date)
    from extinfolist eil
    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
        and eic.eic_code = 'checkme_info'
    where eil.eil_from_c_id = :c_id
        and eil.eil_status = 2
    order by eil.eil_status_datetime desc
    into
        query_date,
        answer_date;

    if (answer_date = '01.01.1900') then
    begin
        suspend;
        exit;

    end

    for select
        cc.cc_code,
        ci.ci_status,
        f_enum('checkme_info', ci.ci_status, 'name', 0)
    from checkme_info ci
    inner join checkme_cfg cc on ci.ci_from_cc_id = cc.cc_id
    where ci.ci_from_c_id = :c_id
        and ci.ci_is_actual = 1
    into
        service_code,
        service_status,
        res_text
    do
        suspend;

end
