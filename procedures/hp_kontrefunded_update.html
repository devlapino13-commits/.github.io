CREATE PROCEDURE HP_KONTREFUNDED_UPDATE(
    K_ID D_ID)
AS
declare variable kont_count d_integer;
declare variable delinq_days d_integer;
declare variable advanced_days d_integer;
declare variable kc_id d_id;
begin
  select count(*), max(res_refunded_delinq_day), max(res_refunded_adv_day)
  from hp_kontract_refund_list(:k_id, 0)
  into :kont_count, :delinq_days, :advanced_days;

  if (kont_count is null) then
    kont_count = 0;

  if (delinq_days is null) then
    delinq_days = 0;

  if (advanced_days is null) then
    advanced_days = 0;

  if (kont_count = 0) then
  begin
    delinq_days = 0;
    advanced_days = 0;

    delete from kontrefunded
    where krf_from_k_id = :k_id;

    update knclient
    set kc_refunded_count = :kont_count,
        kc_refunded_delinq_day = :delinq_days,
        kc_refunded_advanced_day = :advanced_days
    where kc_from_k_id = :k_id and
          (coalesce(kc_refunded_count, -1) <> :kont_count or
           coalesce(kc_refunded_delinq_day, -1) <> :delinq_days or
           coalesce(kc_refunded_advanced_day, -1) <> :advanced_days);
  end else
  begin
    update or insert into kontrefunded (krf_from_k_id, krf_count, krf_max_delinq_days, krf_max_advanced_days)
    values (:k_id, :kont_count, :delinq_days, :advanced_days)
    matching (krf_from_k_id);

    for select kc_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into kc_id
    do
    begin
      kont_count = 0;
      delinq_days = 0;
      advanced_days = 0;
  
      select count(*), max(res_refunded_delinq_day), max(res_refunded_adv_day)
      from hp_kontract_refund_list(:k_id, :kc_id)
      into :kont_count, :delinq_days, :advanced_days;
  
      if (kont_count is null) then
        kont_count = 0;
    
      if (delinq_days is null) then
        delinq_days = 0;
    
      if (advanced_days is null) then
        advanced_days = 0;
  
      update knclient
      set kc_refunded_count = :kont_count,
          kc_refunded_delinq_day = :delinq_days,
          kc_refunded_advanced_day = :advanced_days
      where kc_id = :kc_id and
            (coalesce(kc_refunded_count, -1) <> :kont_count or
             coalesce(kc_refunded_delinq_day, -1) <> :delinq_days or
             coalesce(kc_refunded_advanced_day, -1) <> :advanced_days);
    end
  end
end
