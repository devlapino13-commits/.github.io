CREATE PROCEDURE AAREP_ANALITICA_PORT_OWNS_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_PROJECT VARCHAR(100),
    REP_PERIOD_KIND INTEGER)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_CURR_KC_FIO VARCHAR(100),
    AA_KOLICH_OWN INTEGER,
    AA_AMOUNT_OWN DECIMAL(18,4),
    AA_KOLICH_OTHER INTEGER,
    AA_AMOUNT_OTHER DECIMAL(18,4))
AS
declare variable kol integer;
declare variable amount decimal(18,4);
declare variable aa_is_own integer;
begin
  for select aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_kc_fio, aa_kolich_cl_active_end,
             aa_amount_active_end, aa_is_own
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_project, :rep_period_kind, 0, 0, 0, 0, 0, 0)
      into :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_level_name, :aa_curr_kc_fio, :kol, :amount, :aa_is_own
  do
  begin
    aa_kolich_own = 0;
    aa_amount_own = 0;
    aa_kolich_other = 0;
    aa_amount_other = 0;

    if (aa_is_own = 1) then
    begin
      aa_kolich_own = :kol;
      aa_amount_own = :amount;
    end
    else
    begin
      aa_kolich_other = :kol;
      aa_amount_other = :amount;
    end
    suspend;
  end

end
