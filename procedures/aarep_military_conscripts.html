CREATE PROCEDURE AAREP_MILITARY_CONSCRIPTS(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    FIO STR_100,
    FILIAL STR_100,
    CONTRACT_NUMBER STR_100,
    CONTRACT_DATE D_DATE,
    SERVICE_BEGIN D_DATE,
    SERVICE_END D_DATE,
    VIPS_FIO STR_100)
AS
declare variable C_ID D_ID;
declare variable K_ID D_ID;
declare variable REG_ID D_ID;
declare variable KF_ID D_ID;
begin
  for select mcq.mcq_from_c_id,
             mcq.mcq_answer_date
  from military_check_pocket mcp
  inner join military_check_query mcq on mcq.mcq_from_mcp_id = mcp.mcp_id
  where mcp.mcp_create_date between :rep_date_begin and :rep_date_end and
        mcp.mcp_status = 2 and
        mcq.mcq_received = 1 and
        mcq.mcq_status = 1
  into :c_id, :service_begin
  do
  begin
    for select
          c.c_name_fio,
          k.k_kontract_no,
          k.k_fact_disb_date,
          k.k_last_reg_id,
          k.k_from_kf_id
    from knclient kc
    inner join kontract k on k.k_id = kc.kc_from_k_id
    inner join kontstate ks on ks.ks_from_k_id = k.k_id
    inner join clients c on c.c_id = kc.kc_from_c_id
    where kc.kc_from_c_id = :c_id and
          ks.ks_date_begin <= :rep_date_end and
          ks.ks_date_end >= :rep_date_begin and
          ks.ks_state = 1 and
          ks.ks_ol_is_delete = 0 and
          kc.kc_amount_approved > 0
    into
        :fio,
        :contract_number,
        :contract_date,
        :reg_id,
        :kf_id
    do
    begin
      service_end = dateadd(month, 14, :service_begin);
      select reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :filial;
    
      select c.c_name_fio
      from knaktlist kal
      inner join clients c on c.c_id = kal.kal_new_kf_officer
      where kal.kal_from_kf_id = :kf_id and
            kal.is_delete = 0 and
            :rep_date_end between kal.kal_date and kal.kal_date_end
      into :vips_fio;

      suspend;
    end
  end
end
