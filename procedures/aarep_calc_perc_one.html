CREATE PROCEDURE AAREP_CALC_PERC_ONE(
    KONT_ID DECIMAL(18,0),
    DATE_BEGIN DATE,
    DATE_END DATE,
    INCLUDE_LAST_DAY_PERC INTEGER,
    DB_ID INTEGER,
    K_KREDLINE_END DATE,
    K_STOP_PERC DATE,
    DD_0 DATE)
RETURNS (
    RESULT_BALANCE_BEGIN DECIMAL(18,4),
    RESULT_PERC_BEGIN DECIMAL(18,4),
    RESULT_CALCED DECIMAL(18,4),
    RESULT_PAYED DECIMAL(18,4),
    RESULT_WRITEOFF DECIMAL(18,4),
    RESULT_PERC_END DECIMAL(18,4),
    RESULT_BALANCE_END DECIMAL(18,4),
    RESULT_CALCED_ALL DECIMAL(18,4),
    RESULT_PAYED_ALL DECIMAL(18,4),
    RESULT_WRITEOFF_ALL DECIMAL(18,4),
    RESULT_DISB_AMOUNT DECIMAL(18,4),
    RESULT_DISB_DATE DATE,
    RESULT_OUT_PERC_END DECIMAL(18,4),
    RESULT_OUT_PERC_BEGIN DECIMAL(18,4),
    RESULT_OUT_CALCED DECIMAL(18,4),
    DD_CHANGE_BEGIN DATE,
    DD_CHANGE_END DATE,
    DD_CHANGE_DAYS INTEGER)
AS
declare variable dd date;
declare variable stop_days integer;
declare variable include_last_day_perc_new integer;
declare variable rep_period_begin integer;
declare variable rep_period_end integer;
declare variable result_perc_begin1 decimal(18,4) = 0;
declare variable result_calced1 decimal(18,4) = 0;
declare variable result_payed1 decimal(18,4) = 0;
declare variable result_writeoff1 decimal(18,4) = 0;
declare variable result_perc_end1 decimal(18,4) = 0;
declare variable kont_last_date date;
declare variable result_out_perc_begin1 decimal(18,4) = 0;
declare variable result_out_calced1 decimal(18,4) = 0;
declare variable result_out_perc_end1 decimal(18,4) = 0;
declare variable k_orig decimal(18,0);
begin
  /* Procedure Text */
  -- если include_last_day_perc нужен параметр 1, но при этом не нужно начислять на начало периода, то include_last_day_perc_new=2
  include_last_day_perc_new = include_last_day_perc;
  if (include_last_day_perc_new = 2) then
    include_last_day_perc_new = 1;

  rep_period_begin = 0;
  rep_period_end = 0;

  result_balance_begin = 0;
  result_perc_begin = 0;
  result_calced = 0;
  result_payed = 0;
  result_writeoff = 0;
  result_perc_end = 0;

  result_out_perc_begin = 0;
  result_out_calced = 0;
  result_out_perc_end = 0;

  if (extract(day from :date_begin) = 1) then
    rep_period_begin = extract(year from :date_begin) * 100 + extract(month from :date_begin);
  if (extract(day from (:date_end + 1)) = 1) then
    rep_period_end = extract(year from :date_end) * 100 + extract(month from :date_end);

  if (rep_period_end > 0) then
  begin
    if (not(exists(select *
                   from kontcalcperc
                   where kcp_from_k_id = :kont_id and
                         kcp_period >= :rep_period_end))) then
    begin
      k_orig = null;
      select kd_k_orig
      from kontdevides
      where kd_k_new = :kont_id
      into :k_orig;
      if (k_orig is not null) then
      begin
        if (not(exists(select *
                       from kontcalcperc
                       where kcp_from_k_id = :k_orig and
                             kcp_period >= :rep_period_end))) then
          rep_period_end = 0;
      end
      else
        rep_period_end = 0;
    end
  end
  if ((rep_period_end = 0) and
      (rep_period_begin > 0)) then
  begin
    if (not(exists(select *
                   from kontcalcperc
                   where kcp_from_k_id = :kont_id and
                         kcp_period >= :rep_period_begin))) then
    begin
      k_orig = null;
      select kd_k_orig
      from kontdevides
      where kd_k_new = :kont_id
      into :k_orig;
      if (k_orig is not null) then
      begin
        if (not(exists(select *
                       from kontcalcperc
                       where kcp_from_k_id = :k_orig and
                             kcp_period >= :rep_period_begin))) then
          rep_period_begin = 0;
      end
      else
        rep_period_begin = 0;
    end
  end

  if ((rep_period_begin > 0) and
      (rep_period_end > 0)) then -- периоды ровно совпадают на начало и конец месяца
  begin
    result_perc_begin1 = 0;
    result_calced1 = 0;
    result_payed1 = 0;
    result_writeoff1 = 0;
    result_perc_end1 = 0;

    result_out_perc_begin1 = 0;
    result_out_calced1 = 0;
    result_out_perc_end1 = 0;

    if (rep_period_begin = rep_period_end) then
    begin
      select kcp_perc_begin_saldo, kcp_perc_calced, kcp_perc_payed, kcp_perc_writeoff, kcp_perc_saldo,
             kcp_out_perc_begin_saldo, kcp_out_perc_calced, kcp_out_perc_saldo
      from kontcalcperc
      where kcp_period = :rep_period_end and
            kcp_from_k_id = :kont_id
      into :result_perc_begin1, :result_calced1, :result_payed1, :result_writeoff1, :result_perc_end1,
           :result_out_perc_begin1, :result_out_calced1, :result_out_perc_end1;
      if (result_payed1 is null) then
        result_payed1 = 0;
      if (result_writeoff1 is null) then
        result_writeoff1 = 0;
    end
    else
    begin
      select sum(iif(kcp_period = :rep_period_begin, kcp_perc_begin_saldo, 0)), sum(kcp_perc_calced),
             sum(kcp_perc_payed), sum(kcp_perc_writeoff), sum(iif(kcp_period = :rep_period_end, kcp_perc_saldo, 0)),
             sum(iif(kcp_period = :rep_period_begin, kcp_out_perc_begin_saldo, 0)), sum(kcp_out_perc_calced),
             sum(iif(kcp_period = :rep_period_end, kcp_out_perc_saldo, 0))
      from kontcalcperc
      where kcp_period between :rep_period_begin and :rep_period_end and
            kcp_from_k_id = :kont_id
      into :result_perc_begin1, :result_calced1, :result_payed1, :result_writeoff1, :result_perc_end1,
           :result_out_perc_begin1, :result_out_calced1, :result_out_perc_end1;

      if (result_perc_begin1 is null) then
        result_perc_begin1 = 0;
      if (result_calced1 is null) then
        result_calced1 = 0;
      if (result_payed1 is null) then
        result_payed1 = 0;
      if (result_writeoff1 is null) then
        result_writeoff1 = 0;
      if (result_perc_end1 is null) then
        result_perc_end1 = 0;

      if (result_out_perc_begin1 is null) then
        result_out_perc_begin1 = 0;
      if (result_out_calced1 is null) then
        result_out_calced1 = 0;
      if (result_out_perc_end1 is null) then
        result_out_perc_end1 = 0;
    end
    result_perc_begin = result_perc_begin1;
    result_calced = result_calced1;
    result_payed = result_payed1;
    result_writeoff = result_writeoff1;
    result_perc_end = result_perc_end1;

    result_out_perc_begin = result_out_perc_begin1;
    result_out_calced = result_out_calced1;
    result_out_perc_end = result_out_perc_end1;

    select oks_balance
    from oksummary
    where oks_date_begin <= :date_end and
          oks_date_end >= :date_end and
          oks_from_k_id = :kont_id and
          oks_close_state = 0
    into :result_balance_end;
    if (result_balance_end is null) then
      result_balance_end = 0;

    suspend;
    exit;
  end

  if ((rep_period_begin > 0) and
      (rep_period_end = 0)) then -- начальный период совпадает
  begin
    result_perc_begin1 = 0;
    select kcp_perc_begin_saldo, kcp_out_perc_begin_saldo
    from kontcalcperc
    where kcp_period = :rep_period_begin and
          kcp_from_k_id = :kont_id
    into :result_perc_begin1, :result_out_perc_begin1;
  end

  if ((rep_period_begin = 0) and
      (rep_period_end > 0)) then -- конец периода попадает на конец месяца
  begin
    result_perc_end1 = 0;
    result_payed1 = 0;
    result_writeoff1 = 0;

    result_out_perc_end1 = 0;

    select kcp_perc_saldo, kcp_out_perc_saldo
    from kontcalcperc
    where kcp_period = :rep_period_end and
          kcp_from_k_id = :kont_id
    into :result_perc_end1, :result_out_perc_end1;

    select sum(iif(ok_type = 4, -ok_perc, 0)), sum(iif(ok_type = 5, -ok_perc, 0))
    from operkred
    where ok_ol_is_active = 1 and
          ok_ol_date between :date_begin and :date_end and
          ok_from_k_id = :kont_id and
          ok_from_c_id = 0
    into :result_payed1, result_writeoff1;
    if (result_payed1 is null) then
      result_payed1 = 0;
    if (result_writeoff1 is null) then
      result_writeoff1 = 0;

    select oks_balance
    from oksummary
    where oks_date_begin <= :date_end and
          oks_date_end >= :date_end and
          oks_from_k_id = :kont_id and
          oks_close_state = 0
    into :result_balance_end;
    if (result_balance_end is null) then
      result_balance_end = 0;
  end

  if (db_id is null) then
  begin

    :db_id = f_get_db_id();

    select k_stop_perc
    from kontract
    where k_id = :kont_id
    into :k_stop_perc;

    if (k_stop_perc > dd) then
      k_stop_perc = null;

    select ok_sh_date
    from operkred
    where ok_from_k_id = :kont_id and
          ok_ol_is_active = 1 and
          ok_from_c_id = 0
    order by ok_from_k_id desc, ok_sh_date desc
    rows 1
    into :kont_last_date;

    dd_0 = null;
    select oks_date_begin
    from oksummary
    where oks_from_k_id = :kont_id and
          oks_balance <= 0 and
          oks_close_state = 0 rows 1
    into :dd_0;
  end
  else
    kont_last_date = :k_kredline_end;

  stop_days = 0;
  if (db_id = 81001) then
  begin
    if (date_end >= '30.06.2014') then
    begin
      if (kont_last_date <= '02.05.2014') then
        stop_days = 60;
      else
        stop_days = 61;
    end
    else
      stop_days = 60;
  end
  if (db_id in (39801, 41701)) then
    stop_days = 0;
  if (db_id = 94401) then
    stop_days = 0;

  if (kont_last_date >= '01.01.2013') then
  begin
    if (k_stop_perc is null) then
      k_stop_perc = kont_last_date + stop_days;
    else
    begin
      if (k_stop_perc > (kont_last_date + stop_days)) then
        k_stop_perc = kont_last_date + stop_days;
    end
  end
  else
  begin
    if (k_stop_perc is null) then
      k_stop_perc = kont_last_date;
    else
    begin
      if (k_stop_perc > kont_last_date) then
        k_stop_perc = kont_last_date;
    end
  end
  if (dd_0 is not null) then
  begin
    if (dd_0 < k_stop_perc) then
      k_stop_perc = :dd_0;
  end

  if (rep_period_end = 0) then
  begin
    select result_calced, result_payed, result_writeoff, result_perc_end, result_balance_end, result_calced_all,
           result_payed_all, result_writeoff_all, result_disb_amount, result_disb_date, result_out_calced,
           result_out_perc_end, dd_change_begin, dd_change_end, dd_change_days
    from aarep_calc_perc_one_temp(:kont_id, :date_begin, :date_end, :include_last_day_perc_new, :k_stop_perc, :db_id)
    into :result_calced, :result_payed, :result_writeoff, :result_perc_end, :result_balance_end, :result_calced_all,
         :result_payed_all, :result_writeoff_all, :result_disb_amount, :result_disb_date, :result_out_calced,
         :result_out_perc_end, :dd_change_begin, :dd_change_end, :dd_change_days;

    if (result_payed is null) then
      result_payed = 0;
    if (result_writeoff is null) then
      result_writeoff = 0;
  end
  else
  begin
    result_payed = result_payed1;
    result_writeoff = result_writeoff1;
    result_perc_end = result_perc_end1;
    result_out_perc_end = result_out_perc_end1;
  end
  if (include_last_day_perc = 1) then
  begin
    if (rep_period_begin = 0) then
    begin
      result_perc_begin = 0;
      result_out_perc_begin = 0;
      result_balance_begin = 0;
      dd = :date_begin - 1;
      select result_perc_end, result_out_perc_end, result_balance_end
      from aarep_calc_perc_one_temp(:kont_id, :dd - 1, :dd, :include_last_day_perc, :k_stop_perc, :db_id)
      into :result_perc_begin, :result_out_perc_begin, :result_balance_begin;
    end
    else
    begin
      result_perc_begin = :result_perc_begin1;
      result_out_perc_begin = :result_out_perc_begin1;
    end

    if (((date_end < '01.01.2013') and
        (db_id = 81001)) or
        ((date_end < '01.06.2013') and
        (db_id in (39801, 41701) or
        db_id = 41702)) or
        ((date_end < '01.06.2013') and
        (db_id = 94401))) then
    begin
      -- проценты обнуляются если дата отчета раньше чем 2013 год
      if ((result_perc_begin > 0) and
          (result_balance_begin <= 0)) then
        result_perc_begin = 0;
      if ((result_perc_end > 0) and
          (result_balance_end <= 0)) then
        result_perc_end = 0;

      if ((result_out_perc_begin > 0) and
          (result_balance_begin <= 0)) then
        result_out_perc_begin = 0;
      if ((result_out_perc_end > 0) and
          (result_balance_end <= 0)) then
        result_out_perc_end = 0;
    end
    result_perc_begin = cast(result_perc_begin as decimal(18,2));
    result_perc_end = cast(result_perc_end as decimal(18,2));
    result_calced = result_perc_end - result_perc_begin + (result_payed + result_writeoff);
    result_calced_all = result_perc_end + result_payed_all + result_writeoff_all;

    result_out_perc_begin = cast(result_out_perc_begin as decimal(18,2));
    result_out_perc_end = cast(result_out_perc_end as decimal(18,2));
  end
  result_out_calced = (result_perc_end + result_out_perc_end) - (result_perc_begin + result_out_perc_begin) + (result_payed + result_writeoff) - result_calced;

  --  if (result_out_perc_begin<0) then result_out_perc_begin=0;
  --  if (result_out_perc_end<0) then result_out_perc_end=0;
  --  if (result_out_calced<0) then result_out_calced=0;
  suspend;
end
