CREATE PROCEDURE HP_DEVIDED_KONT_RESULT(
    REP_DATE_END DATE,
    REP_K_ID ID_TYPE)
RETURNS (
    RES_DELIQ_DAYS INTEGER,
    RES_IS_RESTRUCT INTEGER,
    RES_SH_CLOSE_DATE DATE,
    RESTRUCT_DATE_LAST DATE,
    IS_SECURED INTEGER,
    DELINQ_DAY_REST_DATE INTEGER,
    LK_DELIQ_DAYS INTEGER)
AS
declare variable k_id id_type;
declare variable res_deliq_days_tmp integer;
declare variable is_restruct_tmp integer;
declare variable sh_close_date_tmp date;
declare variable restruct_date_last_tmp date;
declare variable pr_id integer;
declare variable k_disb_date date;
declare variable k_kolich_client_approved integer;
declare variable pr_id_old integer;
declare variable pr_code varchar(100);
declare variable pr_loan_type smallint;
declare variable pr_product_kind integer;
declare variable is_secured_tmp integer;
declare variable delinq_day_rest_date_tmp integer;
declare variable lk_deliq_days_tmp integer;
begin
  if ((rep_date_end is null) or (rep_k_id is null)) then
    exit;

  res_deliq_days = 0;
  res_is_restruct = 0;
  res_sh_close_date = '01.01.1900';
  restruct_date_last = '01.01.1900';
  is_secured = 0;
  delinq_day_rest_date = -777;
  lk_deliq_days = 0;
  pr_id_old = 0;

  for select kont.k_id, kont.k_from_pr_id, k_disb_date, k_kolich_client_approved
      from kontract kont
      where ((kont.k_base_k_id = :rep_k_id) or
            (kont.k_id = :rep_k_id))
      into :k_id, :pr_id, :k_disb_date, :k_kolich_client_approved
  do
  begin
----------- >>
    res_deliq_days_tmp = 0;

    select :rep_date_end - oks_delinq_date
    from oksummary oks
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_close_state = 0 and
          oks.oks_from_k_id = :k_id
    into res_deliq_days_tmp;

    res_deliq_days_tmp = coalesce(res_deliq_days_tmp, 0);
    if (res_deliq_days_tmp < 0) then
      res_deliq_days_tmp = 0;

    if (res_deliq_days_tmp > res_deliq_days) then
      res_deliq_days = res_deliq_days_tmp;
----------- <<
----------- >>
    lk_deliq_days_tmp = 0;

    select lk_deliq_days
    from llc_main, llc_kont
    where lm_is_signed = 1 and
          lm_id = lk_from_lm_id and
          :rep_date_end between lk_date and lk_date_end and
          lk_from_k_id = :k_id
    into lk_deliq_days_tmp;

    lk_deliq_days_tmp = coalesce(lk_deliq_days_tmp, 0);
    if (lk_deliq_days_tmp > lk_deliq_days) then
      lk_deliq_days = lk_deliq_days_tmp;
--------- <<
--------- >>
    is_restruct_tmp = 0;
    restruct_date_last_tmp = '01.01.1900';

    select is_restruct, restruct_date_last
    from aarep_get_restructuring(:k_id, :rep_date_end, 81001)
    into :is_restruct_tmp, :restruct_date_last_tmp;

    is_restruct_tmp = coalesce(is_restruct_tmp, 0);
    if (is_restruct_tmp > res_is_restruct) then
      res_is_restruct = is_restruct_tmp;

    if (:restruct_date_last_tmp > :restruct_date_last) then
      restruct_date_last = :restruct_date_last_tmp;
--------- <<
--------- >>
    delinq_day_rest_date_tmp = -777;

    if (is_restruct_tmp = 1) then
    begin
      restruct_date_last_tmp = restruct_date_last_tmp - 1;

      select :restruct_date_last_tmp - oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            :restruct_date_last_tmp between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into delinq_day_rest_date_tmp;

      if (delinq_day_rest_date_tmp < 0) then
        delinq_day_rest_date_tmp = null;
      if (delinq_day_rest_date_tmp > 0) then
        delinq_day_rest_date_tmp = delinq_day_rest_date_tmp + 1;
      restruct_date_last_tmp = restruct_date_last_tmp + 1;

      if (delinq_day_rest_date_tmp > delinq_day_rest_date) then
        delinq_day_rest_date = delinq_day_rest_date_tmp;
    end
--------- <<

    select max(sh_date)
    from schedule
    where sh_from_k_id = :k_id
    into :sh_close_date_tmp;

    if (sh_close_date_tmp > res_sh_close_date) then
      res_sh_close_date = sh_close_date_tmp;

    if (pr_id <> pr_id_old) then
    begin
      select pr.pr_code, pr.pr_loan_type, pr.pr_product_kind, pr.pr_id
      from product pr
      where pr.pr_id = :pr_id
      into pr_code, pr_loan_type, pr_product_kind, pr_id;

      pr_id_old = pr_id;
    end

    is_secured_tmp = null;
    select is_secured
    from aarep_get_secured_loan(:pr_code, :pr_loan_type, :pr_product_kind, :pr_id, :k_disb_date, :k_kolich_client_approved, :k_id)
    into is_secured_tmp;

    is_secured_tmp = coalesce(is_secured_tmp, 0);

    if (is_secured_tmp > is_secured) then
      is_secured = is_secured_tmp;
  end

  res_deliq_days = coalesce(res_deliq_days, 0);
  res_is_restruct = coalesce(res_is_restruct, 0);
  if (res_sh_close_date = '01.01.1900') then
    res_sh_close_date = null;
  if (:restruct_date_last = '01.01.1900') then
    restruct_date_last = null;
  if (delinq_day_rest_date = -777) then
    delinq_day_rest_date = null;
  is_secured = coalesce(is_secured, 0);

  suspend;
end
