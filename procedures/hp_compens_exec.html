CREATE PROCEDURE HP_COMPENS_EXEC(
    OPER_TYPE D_STR_100,
    KC_ID D_ID,
    REP_DATE D_DATE,
    TAG D_SMALLINT = 1)
RETURNS (
    MSG_TXT D_STR_255)
AS
declare variable kcc_id d_integer;
declare variable k_id d_id;
declare variable k_kontract_no d_str_100;
declare variable c_inn d_str_100;
begin
  /*
    oper_type:
    1) add_in_list - добавить в список
    2) do_payment - посадить платеж от робота
    3) find_in_list - поиск в списке
    4) delete_from_list - удаление из списка
  */
  msg_txt = '';

  /* Procedure Text */
  if (kc_id is null) then
  begin
    msg_txt = 'kc_id не предан';
    suspend;
    exit;
  end

  -- если клиент уже учавствует в программе субсидирования от правительства, получаю kcc_id
  select kcc.kcc_id, kcc.kcc_from_k_id
  from knclient_compens kcc
  where kcc.kcc_from_kc_id = :kc_id
  and kcc.kcc_date_end > kcc.kcc_date_begin
  into :kcc_id, :k_id;

  if (oper_type = 'add_in_list') then
  begin
    if (kcc_id is not null) then
    begin
        msg_txt = 'Клиент уже находится в списке';
        suspend;
        exit;            
    end
    else
    begin
        select kon.k_kontract_no, cl.c_inn, kon.k_id
        from knclient kncl
        inner join kontract kon on kon.k_id = kncl.kc_from_k_id
        inner join clients cl on cl.c_id = kncl.kc_from_c_id
        where kncl.kc_id = :kc_id
        into :k_kontract_no, :c_inn, :k_id;
    
        insert into knclient_compens (kcc_kontract_no, kcc_inn, kcc_tag)
        values (:k_kontract_no, :c_inn, :tag);

        msg_txt = 'Клиент, в список, был успешно добавлен';
        suspend;
        exit;
    end
  end

  if (oper_type = 'do_payment') then
  begin
      if (kcc_id is not null) then
      begin
          if (exists(
              select *
              from operkred ok
              inner join operlist ol on ok.ok_from_ol_id = ol.ol_id
              where ok.ok_from_k_id = :k_id
              and ok.ok_ol_date >= :rep_date
              and ok.ok_ol_is_active = 1)) then
          begin
               msg_txt = 'Ошибка! Есть операция более поздней датой';
               suspend;
               exit;
          end

          execute procedure fsrobot_opercompens(:rep_date, :k_id);

          if (exists(select *
                 from compens_robot_log t
                 inner join operlist ol on t.crl_from_ol_id = ol.ol_id
                 where ol.ol_date = :rep_date
                 and ol.is_delete = 0
                 and t.crl_from_kcc_id = :kcc_id)) then
          begin
               msg_txt = 'Платеж успешно посажен';
               suspend;
               exit;  
          end
          else
          begin
               msg_txt = 'Не удалось посадить платеж';
               suspend;
               exit;   
          end
      end
      else
      begin
        msg_txt = 'Клиент не найдет в списке';
        suspend;
        exit;
      end
  end

  if (oper_type = 'find_in_list') then
  begin
    if (kcc_id is not null) then
    begin
        msg_txt = 'Клиент найдет в списке';
        suspend;
        exit;            
    end
    else
    begin
        msg_txt = 'Клиент не найдет в списке';
        suspend;
        exit;     
    end
  end

  if (oper_type = 'delete_from_list') then
  begin
    if (kcc_id is not null) then
    begin
        update knclient_compens kcc
        set kcc.kcc_date_end = kcc.kcc_date_begin - 1,
            kcc.kcc_tag = kcc.kcc_tag * 10 + :tag
        where kcc.kcc_id = :kcc_id
        and kcc.kcc_date_end > kcc.kcc_date_begin;
    end
    else
    begin
        msg_txt = 'Клиент не найдет в списке';
        suspend;
        exit;     
    end
  end
end
