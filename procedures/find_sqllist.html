CREATE PROCEDURE FIND_SQLLIST(
    UN_ID DECIMAL(18,0))
RETURNS (
    SL_ID TYPE OF COLUMN SQLLIST.SL_ID,
    SL_NAME TYPE OF COLUMN SQLLIST.SL_NAME,
    SL_FROM_UN_ID TYPE OF COLUMN SQLLIST.SL_FROM_UN_ID,
    SL_SAVE_TIME TYPE OF COLUMN SQLLIST.SL_SAVE_TIME,
    SL_CODE TYPE OF COLUMN SQLLIST.SL_CODE,
    SL_MEMO TYPE OF COLUMN SQLLIST.SL_MEMO,
    SLU_EXECUTE_COUNT INTEGER,
    SLU_LAST_EXECUTE_TIME TIMESTAMP,
    UN_NAME VARCHAR(30))
AS
declare variable full_access integer;
begin
  full_access = 0;
  select has_access
  from hp_permission_access_check(:un_id, 'Menulist', 0, 'N148')
  into :full_access;

  if (full_access = 1) then
  begin
    for select sl_id, sl_name, sl_from_un_id, sl_save_time, sl_code, sl_memo, un_name
        from username, sqllist
        where un_id = sl_from_un_id
        into :sl_id, :sl_name, :sl_from_un_id, :sl_save_time, :sl_code, :sl_memo, :un_name
    do
    begin
      select slu_execute_count, slu_last_execute_time
      from sqllistusers
      where slu_from_sl_id = :sl_id and
            slu_from_un_id = :un_id
      into :slu_execute_count, :slu_last_execute_time;
      if (slu_execute_count is null) then
        slu_execute_count = 0;
      suspend;
    end
  end
  else
  begin
    for select sl_id, sl_name, sl_from_un_id, sl_save_time, sl_code, sl_memo, un_name, slu_execute_count,
               slu_last_execute_time
        from username, sqllist, sqllistusers
        where un_id = sl_from_un_id and
              sl_id = slu_from_sl_id and
              slu_from_un_id = :un_id
        into :sl_id, :sl_name, :sl_from_un_id, :sl_save_time, :sl_code, :sl_memo, :un_name, :slu_execute_count,
             :slu_last_execute_time
    do
    begin
      if (slu_execute_count is null) then
        slu_execute_count = 0;
      suspend;
    end
  end
end
