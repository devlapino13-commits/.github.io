CREATE PROCEDURE PRINT_KONT_ZADOL(
    PRINT_K_ID DECIMAL(18,0))
RETURNS (
    KZ_OS DECIMAL(18,4),
    KZ_PERC DECIMAL(18,4),
    KZ_FINE DECIMAL(18,4),
    KZ_ALL DECIMAL(18,4),
    KZ_DATE DATE,
    KZ_OS_PERC DECIMAL(18,4),
    KZ_DEL_DAYS INTEGER,
    KZ_TAX DECIMAL(18,4),
    KZ_TAX_CUR DECIMAL(18,4),
    KZ_START_ZADOL DATE,
    KZ_START_ZADOL_SUM DECIMAL(15,2),
    KZ_TAX_NSP CURR_TYPE,
    CZ_CURR_DEL_DAY_OSPERC INTEGER,
    MAX_DELINQ_DAYS INTEGER)
AS
declare variable sh_del_date date;
declare variable sh_min_date date;
declare variable rep_date date;
declare variable sh_date date;
declare variable sh_calc_fine decimal(18,4);
declare variable sh_calc_perc decimal(18,4);
declare variable zad_os decimal(18,4);
declare variable k_disb_date date;
declare variable k_nsp_percent decimal(18,7);
declare variable sh_perc decimal(18,4);
declare variable sh_kommis decimal(18,4);
declare variable initial_tax decimal(18,4);
declare variable payed_tax decimal(18,4);
declare variable sh_perc_temp decimal(18,4);
declare variable sh_kommis_temp decimal(18,4);
declare variable sh_perc_cur decimal(18,4);
declare variable sh_kommis_cur decimal(18,4);
declare variable initial_tax_cur decimal(18,4);
declare variable sh_os_temp decimal(18,4);
begin
  kz_os = 0;
  kz_perc = 0;
  kz_fine = 0;
  kz_all = 0;
  sh_min_date = null;
  rep_date = cast('23.05.2013' as date);

  if (current_date > '26.05.2013') then
  begin
--  Определение начало недели
    rep_date = current_date - iif(extract(weekday from current_date) = 0, 7, extract(weekday from current_date)) + 1;
  end
  for select sh_date, sh_calc_fine, sh_calc_perc, zad_os
      from aarep_make_zad_list(:print_k_id, :rep_date, 1)
      where sh_date is not null
      into :sh_date, :sh_calc_fine, :sh_calc_perc, :zad_os
  do
  begin
--Просроченные дни (пении не учитываются)
    if (sh_del_date is null and
        sh_date < rep_date and
        (zad_os <> 0 or sh_calc_perc <> 0)) then
    begin
      sh_del_date = :sh_date;
      cz_curr_del_day_osperc = current_date - sh_del_date;
    end
    cz_curr_del_day_osperc = coalesce(cz_curr_del_day_osperc, 0);

--Просроченные дни, пении учитываются
    if ((sh_min_date is null) and
        (sh_date < :rep_date)) then
    begin
      if ((zad_os <> 0) or (sh_calc_perc <> 0) or (sh_calc_fine <> 0)) then
      begin
        sh_min_date = :sh_date;
        kz_start_zadol = :sh_date;
      end
    end
    kz_os = kz_os + zad_os;
    kz_fine = kz_fine + sh_calc_fine;
    if (sh_calc_perc <> 0) then
      kz_perc = kz_perc + sh_calc_perc;
  end

  select k_disb_date
  from kontract kont
  where kont.k_id = :print_k_id
  into :k_disb_date;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end

  sh_perc = 0;
  sh_kommis = 0;
  sh_perc_cur = 0;
  sh_kommis_cur = 0;
  kz_start_zadol_sum = 0;
  for select coalesce(sh_os, 0), coalesce(sh_perc, 0), coalesce(sh_kommis, 0), sh_date
      from schedule
      where sh_from_k_id = :print_k_id
      into :sh_os_temp, :sh_perc_temp, :sh_kommis_temp, :sh_date
  do
  begin
    sh_perc = sh_perc + sh_perc_temp;
    sh_kommis = sh_kommis + sh_kommis_temp;

    if (sh_date <= rep_date) then
    begin
      sh_perc_cur = sh_perc_cur + sh_perc_temp;
      sh_kommis_cur = sh_kommis_cur + sh_kommis_temp;
    end

    if (sh_date = kz_start_zadol) then
    begin
      kz_start_zadol_sum = sh_os_temp + sh_perc_temp;
    end
  end
  initial_tax = cast(((sh_kommis + sh_perc) * k_nsp_percent / 100) as decimal(18,2));
  initial_tax_cur = cast(((sh_kommis_cur + sh_perc_cur) * k_nsp_percent / 100) as decimal(18,2));

  select sum(ok.ok_tax)
  from operkred ok
  where ok.ok_from_k_id = :print_k_id and
        ok.ok_ol_is_active = 1 and
        ok_from_c_id = 0
  into :payed_tax;

  payed_tax = coalesce(payed_tax, 0);
  kz_tax = initial_tax + payed_tax;

  kz_tax_cur = initial_tax_cur + payed_tax;

  kz_tax_nsp = round(kz_perc * k_nsp_percent / 100);

  kz_all = kz_all + (kz_os + kz_perc + kz_fine);
  kz_os_perc = kz_os + kz_perc;
  kz_date = :rep_date;

--Максимальная просрочка
  select max(iif(oks_date_end > current_date, current_date, oks_date_end + 1) - oks_delinq_date)
  from oksummary oks
  where oks_close_state = 0 and
        oks_from_k_id = :print_k_id and
        oks_delinq_date < oks_date_end
  into max_delinq_days;

  if (sh_min_date is not null) then
    kz_del_days = (rep_date - sh_min_date);
  else
    kz_del_days = 0;
  suspend;

end
