CREATE PROCEDURE CM_KG_0237_SQL(
    MA_ID D_INTEGER NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable C_INN D_STR_20;
declare variable BLOB_ID D_ID;
declare variable DATE_CREATE D_DATE;
declare variable DATE_BEGIN D_DATE;
declare variable DATE_END D_DATE;
begin
    -- RFCKG-3692
    for select
        c.c_id,
        c.c_inn
    from miniappclient mac
    inner join clients c on mac.mac_from_c_id = c.c_id
    where mac.mac_from_ma_id = :ma_id
        and mac.mac_amount > 0
    into
        :c_id,
        :c_inn
    do
    begin
        with cte_temp_table as
        (
            select
                cds.cds_doc_blob_1 as blob_id,
                cast(cds.cds_create_datetime as d_date) as date_create,
                cds.cds_doc_date_begin as date_begin,
                coalesce(cds.cds_doc_date_end, '31.12.2999') as date_end
            from clients c
            inner join clientdocs cds on cds.cds_from_c_id = c.c_id
                and cds.cds_is_delete = 0
                and cds.cds_is_main_doc = 1
            inner join passkind psk on cds.cds_from_pk_id = psk.pk_id
            inner join passdoctype pdt on psk.pk_from_pdt_id = pdt.pdt_id
                and pdt.pdt_code starting 'passport'
                and pdt.pdt_is_main_doc = 1
            where c.c_id = :c_id
                and c.c_verify_status = 2

            union all
        
            select
                erp.erp_pdf_blob as blob_id,
                cast(er.er_timestamp as date) as date_create,
                cast(f_json_str_get_value(erp.erp_json, 'date_of_issue', '01.01.1900') as date) as date_begin,
                cast(f_json_str_get_value(erp.erp_json, 'date_of_expiry', '01.01.1900') as date) as date_end
            from epass_req er
            inner join epass_response erp on erp.erp_from_er_id = er.er_id
            where er.er_inn = :c_inn
                and er.er_status = 2
        )
        select first 1
            t.blob_id,
            t.date_create,
            t.date_begin,
            t.date_end
        from cte_temp_table t
        order by t.date_create desc
        into
            :blob_id,
            :date_create,
            :date_begin,
            :date_end;

        if (date_create < date_begin
            or current_date not between date_begin and date_end
            or coalesce(blob_id, 0) = 0
        ) then
        begin
            res_error_is_critical = 2;
            res_error_text = 'Нужно прикрепить актуальное фото паспорта (237 блок)';
            suspend;
            exit;
        end
    end
end
