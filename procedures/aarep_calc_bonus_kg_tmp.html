CREATE PROCEDURE AAREP_CALC_BONUS_KG_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_REGION_ID INTEGER,
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_DELIQ_BALANCE DECIMAL(18,4),
    KM_ID DECIMAL(18,0),
    AA_APPROVE_KOL INTEGER,
    AA_ANNUL_KOL INTEGER,
    AA_AMOUNT_DISB DECIMAL(18,4))
AS
declare variable aa_deliq_days integer;
declare variable k_id_prior decimal(18,0);
declare variable kl_datetime timestamp;
declare variable aa_k_id decimal(18,0);
begin
  aa_approve_kol = 0;
  aa_annul_kol = 0;
  km_id = 0;
  for select aa_curr_region_id, aa_amount_active_end, aa_amount_disb, aa_deliq_balance, aa_deliq_days
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 0, 1, 0, 0, 0, 0)
      into :aa_curr_region_id, :aa_amount_active_end, :aa_amount_disb, :aa_deliq_balance, :aa_deliq_days
  do
  begin
    if ((aa_deliq_days < 8) or (aa_deliq_days > 180)) then
      aa_deliq_balance = 0;
    suspend;
  end
  k_id_prior = 0;
  aa_amount_active_end = 0;
  aa_deliq_balance = 0;
  aa_deliq_days = 0;
  aa_amount_disb = 0;

  k_id_prior = 0;
  aa_approve_kol = 1;
  for select kl_from_k_id, kl_datetime, un_from_c_id
      from kontlog, kontstate, username, kontract, product
      where kl_state = 1 and
            kl_from_k_id = ks_from_k_id and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            kl_user_id = un_id and
            k_id = ks_from_k_id and
            k_from_pr_id = pr_id and
            pr_category = 0
      order by kl_from_k_id, kl_datetime
      into :aa_k_id, :kl_datetime, :km_id
  do
  begin
    if (k_id_prior <> aa_k_id) then
    begin
      aa_curr_region_id = null;
      select ps_from_reg_id
      from hr_pstatus
      where ps_from_c_id = :km_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_is_working = 1 rows 1
      into :aa_curr_region_id;
      if (aa_curr_region_id is not null) then
      begin
        suspend;
      end
    end
    k_id_prior = aa_k_id;
  end

  k_id_prior = 0;
  aa_approve_kol = 0;
  for select kl_from_k_id, kl_datetime, un_from_c_id
      from kontlog, username
      where kl_user_id = un_id and
            kl_state = -1 and
            kl_datetime >= :rep_date_begin and
            kl_datetime <= :rep_date_end
      order by kl_from_k_id, kl_datetime
      into :aa_k_id, :kl_datetime, :km_id
  do
  begin
    if (k_id_prior <> aa_k_id) then
    begin
      if (not(exists(select *
                     from kontlog
                     where kl_from_k_id = :aa_k_id and
                           kl_state = 1))) then
      begin
        aa_curr_region_id = null;
        select ps_from_reg_id
        from hr_pstatus
        where ps_from_c_id = :km_id and
              ps_date_begin <= :rep_date_end and
              ps_date_end >= :rep_date_end and
              ps_is_working = 1 rows 1
        into :aa_curr_region_id;
        if (aa_curr_region_id is not null) then
        begin
          aa_annul_kol = 1;
          suspend;
        end
      end
    end
    k_id_prior = aa_k_id;
  end

end
