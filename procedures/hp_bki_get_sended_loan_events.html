CREATE PROCEDURE HP_BKI_GET_SENDED_LOAN_EVENTS(
    BKK_ID D_ID,
    FIND_TEXT D_STR_100,
    SHOW_DELETED D_BOOL = 0,
    IN_BEV_ID D_ID = null)
RETURNS (
    IS_DELETE D_BOOL,
    BKP_DATE D_DATE,
    BKP_CODE D_STR_CODE,
    BKP_STATUS D_INTEGER,
    BKP_DATETIME D_DATETIME,
    BKP_SEND_DATETIME D_DATETIME,
    EVENT_CODE D_STR_CODE,
    EVENT_DATE D_DATE,
    EVENT_TIME D_TIME,
    ACTION_CODE D_STR_CODE,
    TAG_TXT D_STR_CODE,
    REC_NO D_INTEGER,
    ERROR_CODE D_STR_CODE,
    ERROR_TEXT D_STR_LONG,
    FIXED_DATETIME D_DATETIME,
    BEV_STATUS D_INTEGER,
    CRED_STATUS D_STR_CODE,
    UID D_STR_CODE,
    MEMO D_STR_LONG,
    BEV_ID D_ID,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID,
    DEVIDE_MODE D_SMALLINT,
    CLIENT_FIO D_STR_255,
    PORUCH_FIO D_STR_255,
    OBJECT_TYPE D_STR_100,
    KONTRACT_NO D_STR_20,
    BKT_FORMAT D_INTEGER,
    BEV_HASH D_STR_64)
AS
declare variable ma_kind d_integer;
declare variable k_kind d_integer;
declare variable bkt_action_code d_str_code;
declare variable find_id d_id;
declare variable tag d_integer;
declare variable uid_k_id d_id;
declare variable poruch_c_id d_id;
declare variable kd_date d_date;
declare variable devide_poruch_type d_smallint;
begin
  if (in_bev_id is not null) then
    select bkl_object_id
    from bki_events, bki_clients, bki_loans
    where bev_id = :in_bev_id
      and bkc_id = bev_from_bkc_id
      and bkl_id = bkc_from_bkl_id
    into find_text;

  find_text = trim(coalesce(find_text, ''));

  if (find_text = '') then
    exit;

  ma_kind = f_objkind('miniapp', 'id');
  k_kind = f_objkind('kontract', 'id');

  if (char_length(find_text) = 38) then
    select object_id
    from hp_bki_get_uid_info(:find_text)
    into find_id;
  else
  begin
    find_id = find_text;

    when any do
      find_id = 0;
  end 

  select :k_kind, k_id, f_format('Договор №%1 от %2 (Id: %3)', k_kontract_no, k_disb_date, k_id)
  from kontract
  where k_kontract_no = :find_text
     or k_id = :find_id
  into object_kind, object_id, memo;

  if (object_kind is null) then
  begin
    select :ma_kind, ma_id, 'Заявка'
    from miniapp
    where ma_id = :find_id
    rows 1
    into object_kind, object_id, memo;

    select :memo || '. ' || coalesce(list(f_format('Договор №%1 от %2 (Id: %3)', k_kontract_no, k_disb_date, k_id)), '')
    from kontract
    where k_from_ma_app_id = :object_id
      and k_status >=0
    into memo;
  end

  if (object_kind is null) then
    exit;

  if (show_deleted is null) then show_deleted = 0;

  for with t_objects(object_kind, object_id) as(
      select :ma_kind, :object_id
      from rdb$database
      where :object_kind = :ma_kind
    union
      select :k_kind, k_id
      from kontract k
      where :object_kind = :ma_kind
        and k_from_ma_app_id = :object_id
    union
      select 9, k2.k_id
      from kontract k1, kontract k2
      where :object_kind = :k_kind
        and k1.k_id = :object_id
        and k2.k_from_ma_app_id = k1.k_from_ma_app_id
    union
      select :ma_kind, k_from_ma_app_id
      from kontract
      where :object_kind = :k_kind
        and k_id = :object_id
  )
  select iif(bev_delete_bev_id is null, 0, 1), bkp_date, bkp_code, bkp_status, bkp_send_datetime, bkp_datetime, f_enum('bki_packet_action', bkt_action, 'code'),
         bev_status, f_enum('bki_event', bev_event_id, 'code', 0), bev_event_date, bev_event_time, bev_tag, bev_loan_status, f_bki_data_text(bkc_uid_bkd_id),
         bev_rec_no, bke_error_code, f_memo_text(bke_error_id), bke_fixed_datetime, bev_id, bkt_format, bev_hash
  from t_objects
  inner join bki_loans on bkl_object_kind = object_kind and bkl_object_id = object_id
  inner join bki_packet  on bkp_id = bkl_from_bkp_id
  inner join bki_packet_type on bkt_id = bkp_from_bkt_id
  inner join bki_clients on bkc_from_bkl_id = bkl_id
  left join bki_events on bev_from_bkc_id = bkc_id
  left join bki_errors on bke_from_bev_id = bev_id
  where bkp_from_bkk_id = :bkk_id
  order by bkp_id, bev_rec_no
  into is_delete, bkp_date, bkp_code, bkp_status, bkp_send_datetime, bkp_datetime, bkt_action_code,
    bev_status, event_code, event_date, event_time, tag, cred_status, uid,
    rec_no, error_code, error_text, fixed_datetime, bev_id, bkt_format, bev_hash
  do
  begin
    if (show_deleted = 0 and is_delete = 1) then
      continue;

    object_type = null;
    kontract_no = null;
    uid_k_id = null;
    kd_date = null;

    if (coalesce(uid, '') <> '') then
    begin
      select object_type, kontract_no, object_id, poruch_c_id, client_fio, poruch_fio
      from hp_bki_get_uid_info(:uid)
      into object_type, kontract_no, uid_k_id, poruch_c_id, client_fio, poruch_fio;
  
      select 1, kd_date
      from kontdevides
      where kd_k_new = :uid_k_id
      into devide_poruch_type, kd_date;

      if (kd_date is null and poruch_c_id is not null) then
      begin
        select 2, kd_date
        from kontdevides
        where kd_k_orig = :uid_k_id
          and kd_c_id = :poruch_c_id
        into devide_poruch_type, kd_date;
      end
    end

    devide_mode = case
                    when kd_date is null then 0                                  -- не отделялся
                    when devide_poruch_type = 1 and kd_date > event_date then 1  -- до отделения
                    when devide_poruch_type = 1 and kd_date <= event_date then 2 -- после отделения
                    when devide_poruch_type = 2 and kd_date > event_date then 3  -- поручитель до отделения
                    when devide_poruch_type = 2 and kd_date <= event_date then 4 -- поручитель после отделения
                  end;

    action_code = case bkt_action_code
                    when 'default'  then 'B'
                    when 'recreate' then 'C9'
                    when 'delete'   then 'C.2'
                    when 'reset'    then '3.2+C.1'
                    when 'correction' then 'C.1'
                    else 'error'
                  end;

    tag_txt = case tag
                when 1  then 'operkred'
                when 2  then 'kontdata'
                when 3  then 'operkroff'
                when 4  then 'kontstate'
                when 5  then 'bki_manual'
                when 6  then 'bki_errors'
                when 7  then 'miniapp'
                when 8  then 'lawmain'
                when 9  then 'lawkredit'
                when 10 then 'hp_bki_get_packet'
                when 11 then 'hp_bki_get_loan_list_equifax'
                when 12 then 'hp_bki_packet_set_status'
                when 13 then 'operextra'
                when 14 then 'hp_edoc_after_sign_action'
                when 15 then 'zalog'
                when 16 then 'robot'
                when 17 then 'konteffrate'
                when 18 then 'kontpsk'
                when 19 then 'bki_bankrupt'
                else tag
              end;

      if (uid = find_text) then
        uid = '**' || uid;
    suspend;
  end

  -- проверим буфер и покажем следующую плановую отправку
  is_delete = 0;
  bkp_date = null;
  bkp_code = null;
  bkp_status = 10;
  bkp_datetime = null;
  bkp_send_datetime = null;
  event_code = null;
  event_date = null;
  action_code = null;
  tag_txt = null;
  rec_no = null;
  error_code = null;
  error_text = null;
  fixed_datetime = null;
  bev_status = null;
  cred_status = null;
  uid = null;
  memo = null;
  bev_id = null;
  devide_mode = null;
  client_fio = null;
  poruch_fio = null;
  object_type = null;
  kontract_no = null;

  for with t_objects(object_kind, object_id) as(
      select :ma_kind, :object_id
      from rdb$database
      where :object_kind = :ma_kind
    union
      select :k_kind, k_id
      from kontract k
      where :object_kind = :ma_kind
        and k_from_ma_app_id = :object_id
    union
      select 9, k2.k_id
      from kontract k1, kontract k2
      where :object_kind = :k_kind
        and k1.k_id = :object_id
        and k2.k_from_ma_app_id = k1.k_from_ma_app_id
    union
      select :ma_kind, k_from_ma_app_id
      from kontract
      where :object_kind = :k_kind
        and k_id = :object_id
  )
  select bkb_date
  from t_objects, bki_buffer
  where bkb_object_kind = object_kind
    and bkb_object_id = object_id
  order by 1
  into event_date
  do
    suspend;

end
