CREATE PROCEDURE CM_KG_0078_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kml_total curr_type;
declare variable kml_total_tmp curr_type;
declare variable ol_date date;
declare variable k_from_ma_app_id integer;
declare variable ag_id d_integer;
begin
  if (K_ID = 109010239773700101) then exit; ---RFCKG-613
  if (k_id = 109010248810200101) then exit; --RFCKG-857

  select K_AMOUNT_APPROVED + coalesce(KONT.K_PREPAYMENT_SUMMA, 0), K_DISB_DATE, KONT.K_FROM_MA_APP_ID
  from KONTRACT KONT, PRODUCT
  where K_ID = :K_ID and
        K_FROM_PR_ID = PR_ID and
        PR_CODE containing '.MURABAHA.' and
        PR_CODE not containing '.IDJARA.' and
        PR_CODE not containing '.KARDHASAN.' and
        PR_CODE not containing '.ISTISNAA.'
  into KC_AMOUNT, KF_DISB_DATE, K_FROM_MA_APP_ID;
  if (KC_AMOUNT > 0) then
  begin
    for select OL.OL_DATE, km.kml_from_ag_id, KM.KML_TOTAL
        from KONTMAT KM, MATLIST ML, OPERMAT OM, OPERLIST OL
        where KML_OBJECT_KIND = 9 and
              KML_OBJECT_ID = :K_ID and
              KML_FROM_ML_ID = ML_ID and
              KML_BILL_ID = OM_BILL_ID and
              KML_BILL_LINE_NO = OM_BILL_LINE_NO and
              OM_OPER_TYPE = 1 and
              OM_FROM_OL_ID = OL_ID and
              OL.IS_DELETE = 0
        into OL_DATE, ag_id, KML_TOTAL_TMP
    do
    begin
      KML_TOTAL = coalesce(KML_TOTAL, 0) + KML_TOTAL_TMP;

      if (not exists(
        select *
        from agreement_esign_access
        where aea_from_ag_id = :ag_id
          and aea_sign))
      then continue;

      if (exists(select *
                 from MINIAPP MA
                 where MA.MA_ID = :K_FROM_MA_APP_ID and
                       MA.MA_IS_ACTIVE = 1 and
                       :KF_DISB_DATE between MA.MA_DATE_APP and MA_DATE_END and
                       MA.MA_DATE_APP > :OL_DATE)) then
      begin
        res_error_text = 'Дата покупки меньше от даты заявки!';
        res_error_is_critical = 1;
        suspend;
      end
      if (OL_DATE > KF_DISB_DATE) then
      begin
        res_error_text = 'Дата покупки больше от даты выдачи!';
        res_error_is_critical = 1;
        suspend;
      end
    end
    if (KML_TOTAL = 0) then
    begin
      res_error_text = 'Товар(ы) для данного клиента не найдены! Необходимо оприходовать в склад!';
      res_error_is_critical = 2;
      suspend;
    end
    else
    if (KC_AMOUNT <> KML_TOTAL) then
    begin
      res_error_text = 'Сумма товар(а,ов) отличается от суммы финансирования. Проверьте стоить и/или сумму контракта';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
