CREATE PROCEDURE CM_RU_0089_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable ma_from_kf_id id_type;
declare variable reg_id integer;
declare variable re_k_id id_type;
declare variable c_id id_type;
declare variable re_kc_amount curr_type;
declare variable kc_from_k_id_prior id_type;
declare variable kc_from_k_id id_type;
declare variable re_pr_code varchar(100);
declare variable max_delinq_day integer;
begin
  for
  select k_from_kf_id,
         k_disb_date
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_code containing '.KAZAN.'
  into ma_from_kf_id,
       kf_disb_date
  do
  begin
--  Вычисляем офис папки, на дату выдачи
    select kal_new_reg_id from knaktlist
    where kal_from_kf_id = :ma_from_kf_id
    and knaktlist.is_delete = 0
    and :kf_disb_date between kal_date and kal_date_end
    into reg_id;
    if (reg_id <> 1280100) then
    begin
      res_error_text = 'Выбранный продукт акция действует только для офиса Казань';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
--  вычисляем предыдущий цикл
    re_k_id = null;
--    select re_k_id from hp_get_last_circle(:k_id,0,null,null)
--    into re_k_id;
    select k_id from kontstate, kontract
    where ks_ol_is_delete = 0
    and ks_state = 1
    and ks_from_k_id = k_id
    and k_from_kf_id = :ma_from_kf_id
    and ks_date_begin < :kf_disb_date
    order by ks_date_begin desc
    rows 1
    into re_k_id;
    re_k_id = coalesce(re_k_id,0);
    if (re_k_id > 0) then
    begin
--    проверка на просроченные дни
      if (exists(select * from oksummary
      where oks_from_k_id = :re_k_id
      and current_date between oks_date_begin and oks_date_end
      and oks_close_state = 0
      and current_date > oks_delinq_date)) then
      begin
         res_error_text = 'Имеется текущая просроченная задолженность. Акция не действует';
         res_error_is_critical = 2;
         suspend;
         exit;
      end
--    если предыдущий цикл брал по акции то проверяются на след. условии
      select pr_code from kontract, product
      where k_id = :re_k_id
      and k_from_pr_id = pr_id
      into re_pr_code;
      if (re_pr_code containing '.KAZAN.') then
      begin
--      вычисляем максимальный просроченный день
        max_delinq_day = null;
        select max(ok_ol_date - ok_sh_date) from operkred
        where ok_ol_is_active = 1
        and ok_from_k_id = :re_k_id
        and ok_type in (4,5)
        and ok_ol_date > ok_sh_date
        into max_delinq_day;
        max_delinq_day = coalesce(max_delinq_day,0);
        if (max_delinq_day > 7) then
        begin
          res_error_text = 'В предыдущем цикле были просрочки более 7ми дней, не могут брать по данному продукту';
          res_error_is_critical = 2;
          suspend;
        end
      end
      if (kf_disb_date > '31.03.2017'
          and re_pr_code not containing '.KAZAN.') then
      begin
        res_error_text = 'Выбранный продукт не активен, выберите другой продукт.';
        res_error_is_critical = 2;
        suspend;
      end
    end
--  проверка каждого клиента
    kc_from_k_id_prior = 0;
    for
    select kc_from_c_id,
           kc_amount_approved
    from knclient
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    into c_id,
         kc_amount
    do
    begin
--    сумма предыдущего цикла
      re_kc_amount = null;
      /*
      select kc_amount_approved
      from knclient
      where kc_from_c_id = :c_id
      and kc_from_k_id = :re_k_id
      and kc_amount_approved > 0
      into re_kc_amount;
      */
      kc_from_k_id = null;
      select kc_amount_approved,
             kc_from_k_id
      from kontstate, knclient, kontract, product
      where ks_ol_is_delete = 0
      and ks_state = 1
      and ks_date_begin < :kf_disb_date
      and ks_from_k_id = kc_from_k_id
      and kc_amount_approved > 0
      and kc_from_c_id = :c_id
      and ks_from_k_id = k_id
      and k_from_pr_id = pr_id
      and pr_product_kind <> 2
      order by ks_date_begin desc
      rows 1
      into re_kc_amount, kc_from_k_id;
      re_kc_amount = coalesce(re_kc_amount, 15000);
      if (kc_amount > re_kc_amount + 10000) then
      begin
        res_error_text = 'Сумма не должно превышать более чем 10тыс. от предыдущего цикла';
        res_error_is_critical = 2;
        suspend;
      end 
--    проверяем на просрочку предыдущий цикл клиента,
--    если в клиенте предыдущий цикл не совпадает с группой предыдущего цикла
      /*
      kc_from_k_id = null;
      select kc_from_k_id
      from kontstate, knclient
      where ks_ol_is_delete = 0
      and ks_state = 1
      and ks_date_begin < :kf_disb_date
      and kc_from_c_id = :c_id
      and kc_from_k_id = ks_from_k_id
      order by ks_date_begin desc
      rows 1
      into kc_from_k_id;
      */
      if (kc_from_k_id is not null
          and kc_from_k_id <> re_k_id
          and coalesce(kc_from_k_id_prior, 0) <> kc_from_k_id
          ) then
      begin
--      проверка на просрочку
        if (exists(select * from oksummary
        where oks_from_k_id = :kc_from_k_id
        and current_date between oks_date_begin and oks_date_end
        and oks_close_state = 0
        and current_date > oks_delinq_date)) then
        begin
           res_error_text = 'Имеется текущая просроченная задолженность. Акция не действует';
           res_error_is_critical = 2;
           suspend;
        end
      end 
      kc_from_k_id_prior = kc_from_k_id;
    end
  end
end
