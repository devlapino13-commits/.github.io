CREATE PROCEDURE PRINT_KK_MEMBER(
    CMP_ID D_INTEGER)
RETURNS (
    MEMBER_FIO D_STR_100,
    MEMBER_JOBNAME D_STR_100,
    MEMBER_OFFICE D_STR_100)
AS
declare variable cmp_date d_date;
declare variable staff_c_id d_id;
begin
  select cmp_date
  from com_packet
  where cmp_id = :cmp_id
  into cmp_date; -- Дата протокола

  for select cms_from_c_id
  from com_staff
  where cms_from_cmp_id = :cmp_id and
        cms_status = 0 -- Печатаем только участников
  order by cms_id
  into staff_c_id
  do
  begin
    select c_name_fio
    from clients
    where c_id = :staff_c_id
    into member_fio;

    member_office = '';
    member_jobname = '';

    select reg_code || ' ' || reg_name, jl_name
    from region, joblist, hr_pstatus
    where ps_from_c_id =: staff_c_id and
          ps_date_begin <= :cmp_date and
          ps_date_end >= :cmp_date and
          ps_from_jl_id = jl_id and
          ps_from_reg_id = reg_unicode
    into member_office, member_jobname;

    suspend;
  end
end
