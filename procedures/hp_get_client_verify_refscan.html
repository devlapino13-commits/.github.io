CREATE PROCEDURE HP_GET_CLIENT_VERIFY_REFSCAN(
    C_ID ID_TYPE)
RETURNS (
    DOC_ID INTEGER,
    DOC_NAME STR_100,
    DOC_MAIN INTEGER,
    REF_SCAN1 INTEGER,
    REF_SCAN2 INTEGER)
AS
begin
  for select pdt_id, pdt_name, cds_is_main_doc, max(cds_doc_blob_1), max(cds_doc_blob_2)
      from clientdocs
      inner join passkind on pk_id = cds_from_pk_id
      inner join passdoctype on pdt_id = pk_from_pdt_id and
            (pdt_code starting 'passport' or pdt_code starting 'registration')
      where cds_from_c_id = :c_id and
            cds_is_delete = 0
      group by pdt_id, pdt_name, cds_is_main_doc
      order by cds_is_main_doc desc, pdt_name
      into :doc_id, :doc_name, :doc_main, :ref_scan1, :ref_scan2
  do
    suspend;
end
