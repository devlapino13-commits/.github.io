CREATE PROCEDURE CM_KG_0231_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable spouse_id d_id;
declare variable cnt d_integer;
declare variable kf_id d_id;
declare variable kal_new_reg_id d_integer;
declare variable kal_new_kf_officer d_id;
declare variable current_reg_id d_integer;
declare variable current_kf_officer d_id;
declare variable reg_name d_str_code;
declare variable c_name_fio d_str_100;
begin
  res_error_is_critical = 0; -- блок информационный (RFCKG-2369)

  select count(*)
  from miniappclient
  where mac_from_ma_id = :ma_id and
        mac_amount > 0
  into cnt; -- кол-во клиентов в заявке

  if (cnt <> 1) then
    exit; -- групповые займы не проверяем и не блокируем

  select mac_from_c_id
  from miniappclient
  where mac_from_ma_id = :ma_id and
        mac_amount > 0
  into c_id; -- клиент

  with t as (select crel_other_c_id spouse_id
             from clientrelation
             where crel_from_c_id = :c_id and
                   crel_from_rel_id in ('1', '2') and
                   crel_is_active = 1
             union
             select crel_from_c_id spouse_id
             from clientrelation
             where crel_other_c_id = :c_id and
                   crel_from_rel_id in ('1', '2') and
                   crel_is_active = 1)
  select count(*)
  from t
  into cnt;

  if (cnt = 0) then
    exit; -- если у клиента не указан муж/жена, то не блокируем

  if (cnt > 1) then
  begin
    res_error_text = 'У клиента более одного супруга, проверьте родственные связи';
    suspend;
    exit;
  end

  select crel_other_c_id spouse_id
  from clientrelation
  where crel_from_c_id = :c_id and
        crel_from_rel_id in ('1', '2') and
        crel_is_active = 1
  union
  select crel_from_c_id spouse_id
  from clientrelation
  where crel_other_c_id = :c_id and
        crel_from_rel_id in ('1', '2') and
        crel_is_active = 1
  into spouse_id; -- муж/жена клиента

  select k_from_kf_id
  from kontract
  inner join kontstate on ks_from_k_id = k_id
  inner join knclient on kc_from_k_id = k_id
  where ks_ol_is_delete = 0 and
        ks_state = 1 and
        current_date between ks_date_begin and ks_date_end and
        kc_amount_approved > 0 and
        kc_from_c_id = :spouse_id
  order by ks_date_begin desc rows 1
  into kf_id;

  if (kf_id is null) then
    exit; -- если у мужа/жены клиента нет папки c активным договором, то не блокируем

  select kal_new_reg_id,
         kal_new_kf_officer
  from knaktlist
  where current_date between kal_date and kal_date_end and
        kal_from_kf_id = :kf_id and
        is_delete = 0
  into kal_new_reg_id,
       kal_new_kf_officer;

  select kal_new_reg_id,
         kal_new_kf_officer
  from kontract
  inner join knaktlist on kal_from_kf_id = k_from_kf_id
  where k_from_ma_app_id = :ma_id and
        is_delete = 0 and
        current_date between kal_date and kal_date_end
  into current_reg_id,
       current_kf_officer;

  if (current_reg_id = kal_new_reg_id and
      current_kf_officer = kal_new_kf_officer) then
    exit; -- если текущий офис и СПФ тот же, который обслуживает мужа/жену текущего клиента, то не блокируем

  select reg_name
  from region
  where reg_unicode = :kal_new_reg_id
  into reg_name;

  select c_name_fio
  from clients
  where c_id = :kal_new_kf_officer
  into c_name_fio;

  res_error_text = 'Супруг/а данного клиента уже обслуживается у другого СПФ - ' || c_name_fio;
  res_error_text = res_error_text || ', офис - ' || reg_name || '. Направьте клиента к этому СПФ.';

  suspend;
end
