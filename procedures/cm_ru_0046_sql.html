CREATE PROCEDURE CM_RU_0046_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable K_AMOUNT_APPROVED CURR_TYPE;
declare variable K_KOLICH_REPAY integer;
declare variable PR_CODE varchar(20);
declare variable KC_FROM_C_ID ID_TYPE;
declare variable K_KOMMIS_PERC PERC_TYPE;
declare variable KOL_CLI_GROUP integer;
declare variable PR_ID integer;
declare variable K_PERCENT PERC_TYPE;
declare variable c_birth_date date;
declare variable C_CITIZEN integer;
declare variable MIGR_DATE date;
declare variable CON_RF integer;
declare variable CON_ROD integer;
declare variable PATENT_DATE date;
begin
  for
  select k_amount_approved,
         k_kolich_repay,
         pr_code,
         k_disb_date,
         k_kommis_perc,
         k_kolich_client_approved,
         pr_id,
         k_percent
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_code containing '.IND.'
  and pr_code containing '.PATENT.' --Микрозаем на патент
  into k_amount_approved,
       k_kolich_repay,
       pr_code,
       kf_disb_date,
       k_kommis_perc,
       kol_cli_group,
       pr_id,
       k_percent
  do
  begin
  --Проверка по паспорту продукта, пункт 1.1.
  if (kol_cli_group <> 1) then
    begin
--      res_error_text = 'Количество клиентов не соответствует условию паспорта продукта: 1.1. Продукт является индивидуальным микрозаймом!';
      res_error_text = 'Количество клиентов не соответствует условию паспорта продукта!';
      res_error_is_critical = 1;
      suspend;
    end else
    begin
      --Проверка по паспорту продукта, пункт 1.2.
      if (k_amount_approved not between 15000 and 30000) then
        begin
--          res_error_text = 'Сумма выдачи не соответствует условию паспорта продукта: 1.2. Сумма микрозайма: от 15 000 до 30 000 руб.!';
          res_error_text = 'Сумма выдачи должен быть от 15 000 до 30 000 руб.!';
          res_error_is_critical = 1;
          suspend;
        end
      
      --Проверка по паспорту продукта, пункт 1.3.
      if (k_kolich_repay not between 3 and 9) then
        begin
--          res_error_text = 'Срок микрозайма не соответствует условию паспорта продукта: 1.3. Сроки микрозайма: от 3 до 9 месяцев!';
          res_error_text = 'Срок микрозайма должен быть от 3 до 9 месяцев!';
          res_error_is_critical = 1;
          suspend;
        end
      
      --Проверка по паспорту продукта, пункт 1.4.
      /*
      if (k_percent <> 9.900000) then
        begin
          res_error_text = 'Процентная ставка не соответствует условию паспорта продукта: 1.4. Процентная ставка: 9,9% в месяц!';
          res_error_is_critical = 1;
          suspend;
        end
      */
      --Проверка по паспорту продукта, пункт 1.5.
      if (k_kommis_perc <> 0) then
        begin
--          res_error_text = 'Дополнительный процент не соответствует условию паспорта продукта: 1.5. Дополнительный процент: не взимается!';
          res_error_text = 'Дополнительный процент не соответствует условию продукта!';
          res_error_is_critical = 1;
          suspend;
        end
      
      --Проверка по паспорту продукта, пункт 1.6.
      if (exists (select * from schedule sch where sch.sh_from_k_id = :k_id and sch.sh_os = 0 and sch.sh_id > 1)) then
        begin
--          res_error_text = 'Льготный период не соответствует условию паспорта продукта: 1.6. Льготный период: клиент вправе воспользоваться льготным периодом в течение первого месяца займа!';
          res_error_text = 'Клиент вправе воспользоваться льготным периодом в течение первого месяца займа!';
          res_error_is_critical = 1;
          suspend;
        end
      
      --Проверить по условиям паспорта продукта, пункты 1.7. и 1.8. невозможно!
    
      --Проверка по паспорту продукта, пункт 1.9.
      if (exists (select * from schedule sch where sch.sh_from_k_id = :k_id
                  and (sch.sh_id > 0 and sch.sh_os + sh_perc not between 3000 and 7000 and sh_os <> 0
                  or sh_os = 0 and sh_perc > 7000))) then
        begin
--          res_error_text = 'Сумма план.погашения не соответствует условию паспорта продукта, смотрите пункт: 1.9.!';
          res_error_text = 'Сумма план.погашения должна быть от 3000 р. до 7000 р.';
          res_error_is_critical = 1;
          suspend;
        end
      select kc_from_c_id,
             c_birth_date,
             c_citizen
      from knclient kc, clients
      where kc_amount_approved > 0
      and kc_from_k_id = :k_id
      and kc_from_c_id = c_id
      into :kc_from_c_id, c_birth_date, c_citizen;
  /*    --Проверка по паспорту продукта, пункт 2.1.
      if (exists (select distinct 1 from kontstate ks, knclient where ks_state = 1 and ks_ol_is_delete = 0
                  and :kf_disb_date between ks_date_begin and ks_date_end and ks_from_k_id = kc_from_k_id
                  and kc_amount_approved > 0 and kc_from_c_id = :kc_from_c_id)) then
        begin
          res_error_text = 'У клиента есть активный кредит, смотрите пункт: 2.1. в паспорте продукта!';
          res_error_is_critical = 1;
          suspend;
        end*/
    
      --Проверка по паспорту продукта, пункт 2.1.
      if (c_birth_date > dateadd(year, -21, :kf_disb_date)) then
        begin
--          res_error_text = 'Возраст клиента не позволяет получить кредит, смотрите пункт: 2.1. в паспорте продукта!';
          res_error_text = 'Возраст клиента не позволяет получить кредит!';
          res_error_is_critical = 1;
          suspend;
        end
    
      --Проверка по паспорту продукта, пункт 2.1.
      if (c_citizen not in(9,12)) then
        begin
--          res_error_text = 'Гражданство клиента не позволяет получить кредит, смотрите пункт: 2.1. в паспорте продукта!';
          res_error_text = 'Несоответствие гражданства клиента с условием продукта';
          res_error_is_critical = 1;
          suspend;
        end
    
      select distinct CDS_DOC_DATE_BEGIN
      from CLIENTDOCS, PASSKIND, PASSDOCTYPE
      where cds_from_pk_id = pk_id
      and pk_from_pdt_id = pdt_id
      and cds_is_delete = 0
      and pdt_code = 'magration_card'
      and :kf_disb_date between CDS_DOC_DATE_BEGIN and cds_doc_date_end
      and CDS_FROM_C_ID = :kc_from_c_id
      rows 1
      into migr_date;
    
      if (migr_date is null) then
        begin
--          res_error_text = 'Условия по проверке миграционной карты не позволяет получить кредит, смотрите пункт: 2.1. в паспорте продукта!';
          res_error_text = 'Нет миграционной карта либо не указан срок!';
          res_error_is_critical = 1;
          suspend;
        end
      select sum(iif(cc_type = 0, 1,0))con_rf,
             sum(iif(cc_type = 1, 1,0))con_rod
      from contacts
      where cc_from_c_id = :kc_from_c_id
      --and cc_check_result = 1
      and cc_is_old_contact = 0
      and cc_cell_no is not null
      into con_rf, con_rod;
      con_rf = coalesce(con_rf,0);
      con_rod = coalesce(con_rod,0);
      if (con_rf < 2 or con_rod < 2) then
        begin
--          res_error_text = 'Условия по проверке контактных данных не позволяет получить кредит, смотрите пункт: 2.1. в паспорте продукта!';           ---
          res_error_text = 'Необходимо указать 2 контакта на родине и 2 контакта в РФ';           ---
          res_error_is_critical = 1;
          suspend;
        end 
      select distinct CDS_DOC_DATE_BEGIN
      from CLIENTDOCS, PASSKIND, PASSDOCTYPE
      where cds_from_pk_id = pk_id
      and pk_from_pdt_id = pdt_id
      and cds_is_delete = 0
      and pdt_code = 'work_patent'
      and CDS_FROM_C_ID = :kc_from_c_id  
      order by CDS_DOC_DATE_BEGIN desc
      rows 1
      into patent_date;
      if (patent_date > kf_disb_date) then
        begin
          res_error_text = 'Дата выдачи патента больше даты выдачи кредита!';
          res_error_is_critical = 1;
          suspend;
        end    
      --пункт: 2.2.1
      if (kf_disb_date-migr_date > 10 and patent_date is null) then
        begin
--          res_error_text = 'Условия по проверке миграционной карты не позволяет получить кредит, смотрите пункт: 2.2.1. в паспорте продукта!';
          res_error_text = 'Срок миграционной карты менее 10-дней';
          res_error_is_critical = 1;
          suspend;
        end
      --пункт: 2.3.1
      if (datediff(month, patent_date, kf_disb_date) < 1) then
        begin
--          res_error_text = 'Срок окончание текущего патента менее одного месяца, смотрите пункт: 2.3.1. в паспорте продукта!!';
          res_error_text = 'Срок окончание текущего патента менее одного месяца';
          res_error_is_critical = 1;
          suspend;
        end
      --пункт: 2.3.2 пока невозможно сделать
      --пункт: 2.3.3
      if (not exists (
            select distinct 1
            from CLIENTDOCS, PASSKIND, PASSDOCTYPE
            where CDS_FROM_C_ID = :kc_from_c_id
            and cds_from_pk_id = pk_id
            and pk_from_pdt_id = pdt_id
            and cds_is_delete = 0
            and pdt_code in (                      
                             'certificate_rus_lang', -- Сертификат о знании русского языка
                             'diplom_vuz', -- Диплом ВУЗ
                             'diplom_suz', -- Диплом СУЗ
                             'diplom_school' -- Аттестат
                      ))) then
      
        begin
--          res_error_text = 'Нет необходимых документов, смотрите пункт: 2.3.3. в паспорте продукта!!';
          res_error_text = 'Необходимо указать один из документов: - Сертификат о знании русс.яз., Диплом или Аттестат!';
          res_error_is_critical = 0;
          suspend;
        end
    end
  end 
end
