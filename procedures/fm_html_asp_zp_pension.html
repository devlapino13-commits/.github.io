CREATE PROCEDURE FM_HTML_ASP_ZP_PENSION(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable lang d_str_lang;
declare variable c_id d_id;
declare variable c_name_fio d_str_255;
declare variable k_from_ma_app_id d_id;
declare variable mac_id d_id;
declare variable eil_id d_id;
declare variable section_title d_str_255;
declare variable table_row_text d_blob_text;
begin
    lang = f_get_lang_code(un_id, null);
    res_html = (select res_style
                from fm_html_styles);

    if (f_objkind(obj_kind, 'code') not in ('kontract')) then
    begin
        res_html = f_lang(lang, 'Не удалось получить информацию о справках');
        suspend;
        exit;
    end

    for
        select kc.kc_from_c_id, c.c_name_fio, k.k_from_ma_app_id
        from knclient kc
        inner join clients c on kc.kc_from_c_id = c.c_id
        inner join kontract k on kc.kc_from_k_id = k.k_id
        where kc.kc_from_k_id = :obj_id and
              kc.kc_amount_approved > 0
        into :c_id, :c_name_fio, :k_from_ma_app_id
    do
    begin
        select mac.mac_id
        from miniappclient mac
        where mac.mac_from_ma_id = :k_from_ma_app_id and
              mac.mac_from_c_id = :c_id and
              mac.mac_amount > 0
        into :mac_id;
    
        if (coalesce(c_id, 0) = 0 or coalesce(mac_id, 0) = 0) then
        begin
            res_html = f_lang(lang, 'Не удалось получить информацию о справках');
            suspend;
            exit;
        end
    
        -- Секция "Справка о доходах"
        section_title = f_lang(lang, 'Справка о доходах');
    
        select first 1 icq.icq_from_eil_id
        from income_check_querys icq
        inner join extinfolist eil on icq.icq_from_eil_id = eil.eil_id
        where icq.icq_from_c_id = :c_id and
              -- Заметил, что иногда в таблице income_check_querys ссылка icq_from_mac_id ведет на несуществующую запись в miniappclient
              -- поэтому для надежности, поставил условие еще и по icq_from_ma_id
              (icq.icq_from_mac_id = :mac_id or icq.icq_from_ma_id = :k_from_ma_app_id)  and
              icq.icq_status > 0 and
              icq.icq_config = 'salary_check' and
              eil.eil_status = 2 and
              eil.eil_from_bls_id is not null
        order by icq.icq_id desc
        into :eil_id;
    
        if (coalesce(eil_id, 0) <> 0) then
            table_row_text = '<b>' || f_uri_html(c_name_fio, 'html_salary_check', section_title, 'extinfolist', eil_id, current_date, current_date) || '</b>';
        else
            table_row_text = '<b>' || f_lang(lang, 'Справка о доходах не найдена') || '</b>';
    
        res_html = BLOB_APPEND(res_html, f_format('<h1>%1</h1>', section_title));
        res_html = BLOB_APPEND(res_html, '<table><table width="100%"><col width="100%">');
        res_html = BLOB_APPEND(res_html, f_html_table_row(table_row_text, '.', '', '', -1, -1, -1, -1, 1, ''));
        res_html = BLOB_APPEND(res_html, '</table>');
        res_html = BLOB_APPEND(res_html, '<br>');
    
        eil_id = null;
    
        -- Секция "Справка о пенсии"
        section_title = f_lang(lang, 'Справка о пенсии');
    
        select first 1 icq.icq_from_eil_id
        from income_check_querys icq
        inner join extinfolist eil on icq.icq_from_eil_id = eil.eil_id
        where icq.icq_from_c_id = :c_id and
              (icq.icq_from_mac_id = :mac_id or icq.icq_from_ma_id = :k_from_ma_app_id)  and
              icq.icq_status > 0 and
              icq.icq_config = 'pension_check' and
              eil.eil_status = 2 and
              eil.eil_from_bls_id is not null
        order by icq.icq_id desc
        into :eil_id;
    
        if (coalesce(eil_id, 0) <> 0) then
            table_row_text = '<b>' || f_uri_html(c_name_fio, 'html_pension_check', section_title, 'extinfolist', eil_id, current_date, current_date) || '</b>';
        else
            table_row_text = '<b>' || f_lang(lang, 'Справка о пенсии не найдена') || '</b>';
    
        res_html = BLOB_APPEND(res_html, f_format('<h1>%1</h1>', section_title));
        res_html = BLOB_APPEND(res_html, '<table><table width="100%"><col width="100%">');
        res_html = BLOB_APPEND(res_html, f_html_table_row(table_row_text, '.', '', '', -1, -1, -1, -1, 1, ''));
        res_html = BLOB_APPEND(res_html, '</table>');
        res_html = BLOB_APPEND(res_html, '<br>');
    
        eil_id = null;
    
        -- Секция "Справка АСП"
        section_title = f_lang(lang, 'Справка АСП');
    
        select first 1 icq.icq_from_eil_id
        from income_check_querys icq
        inner join extinfolist eil on icq.icq_from_eil_id = eil.eil_id
        where icq.icq_from_c_id = :c_id and
              (icq.icq_from_mac_id = :mac_id or icq.icq_from_ma_id = :k_from_ma_app_id)  and
              icq.icq_status > 0 and
              icq.icq_config = 'asp_check' and
              eil.eil_status = 2 and
              eil.eil_from_bls_id is not null
        order by icq.icq_id desc
        into :eil_id;
    
        if (coalesce(eil_id, 0) <> 0) then
            table_row_text = '<b>' || f_uri_html(c_name_fio, 'html_asp_check', section_title, 'extinfolist', eil_id, current_date, current_date) || '</b>';
        else
            table_row_text = '<b>' || f_lang(lang, 'Справка АСП не найдена') || '</b>';
    
        res_html = BLOB_APPEND(res_html, f_format('<h1>%1</h1>', section_title));
        res_html = BLOB_APPEND(res_html, '<table><table width="100%"><col width="100%">');
        res_html = BLOB_APPEND(res_html, f_html_table_row(table_row_text, '.', '', '', -1, -1, -1, -1, 1, ''));
        res_html = BLOB_APPEND(res_html, '</table>');
        res_html = BLOB_APPEND(res_html, '<br>');
    
        c_id = null;
        suspend;
    end
end
