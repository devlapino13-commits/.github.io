CREATE PROCEDURE LAST_DOC_NUMBER(
    DD DATE,
    KIND INTEGER,
    REG_ID INTEGER)
RETURNS (
    NUM INTEGER)
AS
declare variable dn_code varchar(255);
declare variable dn_beg_type integer;
declare variable max_n integer;
declare variable dd_begin d_date;
declare variable tmp_dd d_date;
begin
  select dn_admin_opertypes, dn_begin_type
  from docname
  where dn_id = :kind
  into :dn_code, :dn_beg_type;

  dd_begin = dd;
  if (dn_beg_type = 1) then -- Обнуляется раз в год
    dd_begin = cast('01.01.' || extract(year from :dd) as date);

  if (dn_beg_type = 2) then -- Обнуляется раз в Месяц
    dd_begin = cast('01.' || right('0' || extract(month from :dd), 2) || '.' || extract(year from :dd) as date);

  if (dn_code is null) then
    dn_code = '';

  max_n = 0;
  if (dn_code <> '') then
  begin
    num = 0;
    if (dn_beg_type = 0) then  -- Нумерация не обнуляется
    begin
      select ol_doc_no
      from operlist, docname
      where dn_admin_opertypes = :dn_code and
            ol_doc_kind = dn_id and
            is_delete = 0 and
            ol_doc_no is not null and
            ol_from_reg_id = :reg_id
      order by ol_doc_no desc
      rows 1
      into :max_n;
    end else
    begin
      tmp_dd = dd;
      while (max_n = 0 and tmp_dd >= dd_begin) do
      begin
        select ol_doc_no
        from operlist, docname
        where dn_admin_opertypes = :dn_code and
              ol_doc_kind = dn_id and
              is_delete = 0 and
              ol_doc_no is not null and
              ol_date = :tmp_dd and
              ol_from_reg_id = :reg_id
        order by ol_date desc, ol_from_reg_id desc, ol_doc_no desc
        rows 1
        into :max_n;
        if (max_n = 0) then tmp_dd = tmp_dd - 1;
      end
    end
  end
  num = coalesce(max_n, 0) + 1;
  suspend;
end
