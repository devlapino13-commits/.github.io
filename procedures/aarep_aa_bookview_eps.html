CREATE PROCEDURE AAREP_AA_BOOKVIEW_EPS(
    REGION_ID D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    DEL_STATUS D_SMALLINT,
    FILTER_JSON D_STR_8100,
    UN_ID D_ID)
RETURNS (
    KOB_CURRENCY D_STR_LANG,
    KOB_CURR_RATE D_CURRENCY,
    KOB_CLIENT_DT D_ID,
    KOB_ACC_DT D_ID,
    KOB_SUMMA_VAL D_CURRENCY,
    KOB_SUMMA D_CURRENCY,
    KOB_ACC_KT D_ID,
    KOB_CLIENT_KT D_ID,
    KOB_DETAIL D_STR_LONG,
    OL_ID D_STR_CODE,
    OL_FROM_REG_ID D_INTEGER,
    OL_DATE D_DATE,
    OL_DOC_NO D_INTEGER,
    OL_DOC_NO_STR D_STR_CODE,
    OL_MEMO D_STR_LONG,
    IS_DELETE D_SMALLINT,
    OL_DATETIME D_DATETIME,
    OL_DOC_CLIENT_NAME D_STR_255,
    UN_NAME D_STR_CODE,
    DN_NAME D_STR_255,
    MO_NAME D_STR_100,
    KOB_ID D_ID,
    KOB_CLIENT_DT_NAME D_STR_255,
    KOB_CLIENT_KT_NAME D_STR_255,
    IS_DELETE_NAME D_STR_CODE,
    OL_CALC_MEMO D_STR_LONG,
    KOB_ACC_DT_CODE D_STR_CODE,
    KOB_ACC_KT_CODE D_STR_CODE,
    OL_REGION_NAME D_STR_100,
    OL_BRANCH_NAME D_STR_100,
    KOB_ACC_DT_NAME D_STR_255,
    KOB_ACC_KT_NAME D_STR_255,
    OL_OPER_TYPE D_INTEGER,
    OL_DOC_KIND D_INTEGER,
    OL_IS_CLIENT D_SMALLINT,
    OL_POCKET D_INTEGER,
    OL_PARENT_ID D_ID,
    OL_OPER_SUBTYPE D_INTEGER,
    CAN_DELETE D_SMALLINT,
    KOB_SHORT_ACC_DT D_STR_CODE,
    KOB_SHORT_ACC_KT D_STR_CODE,
    KOB_SUMMA_TAX_DT CURR_TYPE,
    KOB_SUMMA_TAX_KT CURR_TYPE)
AS
declare variable kob_acc_dt_prior d_id;
declare variable kob_acc_kt_prior d_id;
declare variable ol_user_id decimal(18,0);
declare variable ol_user_id_prior decimal(18,0);
declare variable ol_doc_kind_prior integer;
declare variable ol_pocket_prior integer;
declare variable ol_from_reg_id_prior integer;
declare variable is_active smallint;
declare variable ol_document_client decimal(18,0);
declare variable sab_code_dt d_str_255;
declare variable sab_code_kt d_str_255;
declare variable memo_text d_str_255;
declare variable client_text d_str_255;
declare variable acc_dt d_str_code;
declare variable acc_kt d_str_code;
declare variable ab_id_dt d_id;
declare variable ab_id_kt d_id;
declare variable ab_code_dt d_str_code;
declare variable ab_code_kt d_str_code;
declare variable sab_id_dt d_id;
declare variable sab_id_kt d_id;
declare variable sab_filter_code_dt d_str_255;
declare variable sab_filter_code_kt d_str_255;
declare variable acc_filter_kind d_smallint;
declare variable kc_id d_id;
begin
  is_active = del_status;

  ol_branch_name = '';
  ol_user_id_prior = 0;
  kob_acc_dt_prior = -1;
  kob_acc_kt_prior = -1;

  kob_acc_dt_code = '';
  kob_acc_kt_code = '';

  ol_pocket_prior = -1;
  ol_doc_kind_prior = -1;
  ol_from_reg_id_prior = 0;

  acc_filter_kind = f_json_str_get_value(filter_json, 'acc_filter_kind', '0');
  kc_id = f_json_str_get_value(filter_json, 'kc_id', '0');

  acc_dt = f_json_str_get_value(filter_json, 'acc_mask_dt', '');
  acc_dt = replace(acc_dt, '.', '');
  if (acc_dt = '********************') then
    acc_dt = '';
  acc_dt = replace(acc_dt, '*', '_');

  if (acc_filter_kind = 1) then
  begin
    acc_kt = f_json_str_get_value(filter_json, 'acc_mask_kt', '');
    acc_kt = replace(acc_kt, '.', '');
    if (acc_kt = '********************') then
      acc_kt = '';
    acc_kt = replace(acc_kt, '*', '_');
  end

  if (acc_dt is null) then
    acc_dt = '';
  if (acc_kt is null) then
    acc_kt = '';

  ab_id_dt = f_json_str_get_value(filter_json, 'ab_id_dt', -1);
  ab_code_dt = '';
  if (ab_id_dt <> -1) then
  begin
    select ab_code
    from ref_accbalance
    where ab_id = :ab_id_dt
    into ab_code_dt;
  end

  ab_code_kt = '';
  if (acc_filter_kind = 1) then
  begin
    ab_id_kt = f_json_str_get_value(filter_json, 'ab_id_kt', -1);
    if (ab_id_kt <> -1) then
    begin
      select ab_code
      from ref_accbalance
      where ab_id = :ab_id_kt
      into ab_code_kt;
    end
  end

  sab_id_dt = f_json_str_get_value(filter_json, 'sab_id_dt', -1);
  sab_filter_code_dt = '';
  if (sab_id_dt <> -1) then
  begin
    select sab_full_code
    from sys_accbalance
    where sab_id = :sab_id_dt
    into sab_filter_code_dt;
  end

  sab_filter_code_kt = '';
  if (acc_filter_kind = 1) then
  begin
    sab_id_kt = f_json_str_get_value(filter_json, 'sab_id_kt', -1);
    if (sab_id_kt <> -1) then
    begin
      select sab_full_code
      from sys_accbalance
      where sab_id = :sab_id_kt
      into sab_filter_code_kt;
    end
  end

  client_text = f_json_str_get_value(filter_json, 'client_text', '');
  memo_text = f_json_str_get_value(filter_json, 'memo_text', '');

  for select ol_id, operlist.is_delete, ol_oper_type, ol_oper_subtype, ol_is_client, ol_pocket, ol_doc_kind,
             kob_currency, kob_curr_rate, kob_client_dt, kob_acc_dt, kob_summa_val, kob_summa, kob_acc_kt,
             kob_client_kt, obd_detail, ol_from_reg_id, ol_date, ol_doc_no, ol_doc_no_str, ol_memo, ol_datetime,
             ol_doc_client_name, ol_document_client, ol_user_id, ol_parent_id, kob_id, ol_doc_kind, ol_pocket,
             sdt.sab_full_code, skt.sab_full_code, dt.acc_add_code, kt.acc_add_code, kob_summa_tax_dt,
             kob_summa_tax_kt
      from operlist, region_list(:region_id, 1, 3, 1), kn_operbook, ref_account dt, ref_account kt, kn_obdetail, sys_accbalance sdt, sys_accbalance skt
      where kob_ol_date between :date_begin and :date_end and
            kob_acc_dt = dt.acc_id and
            kob_acc_kt = kt.acc_id and
            dt.acc_from_sab_id = sdt.sab_id and
            kt.acc_from_sab_id = skt.sab_id and
            kob_from_ol_id = ol_id and
            kob_ol_reg_id = id_list and
            kob_detail_id = obd_id and
            (:is_active <= -1 or
            kob_ol_is_active = :is_active) and
            (:acc_dt = '' or
            dt.acc_code like :acc_dt or
            (:acc_filter_kind = 0 and
            kt.acc_code like :acc_dt)) and
            (:acc_kt = '' or
            kt.acc_code like :acc_kt) and
            (:ab_code_dt = '' or
            dt.acc_ab_code starting :ab_code_dt or
            (:acc_filter_kind = 0 and
            kt.acc_ab_code starting :ab_code_dt)) and
            (:ab_code_kt = '' or
            kt.acc_ab_code starting :ab_code_kt) and
            (:kc_id = 0 or
            kob_from_kc_id = :kc_id)
      order by kob_ol_date, kob_from_ol_id, kob_id
      into :ol_id, :is_delete, :ol_oper_type, :ol_oper_subtype, :ol_is_client, :ol_pocket, :ol_doc_kind, :kob_currency,
           :kob_curr_rate, :kob_client_dt, :kob_acc_dt, :kob_summa_val, :kob_summa, :kob_acc_kt, :kob_client_kt,
           :kob_detail, :ol_from_reg_id, :ol_date, :ol_doc_no, :ol_doc_no_str, :ol_memo, :ol_datetime,
           :ol_doc_client_name, :ol_document_client, :ol_user_id, :ol_parent_id, :kob_id, :ol_doc_kind, :ol_pocket,
           :sab_code_dt, :sab_code_kt, :kob_short_acc_dt, :kob_short_acc_kt, :kob_summa_tax_dt, :kob_summa_tax_kt
  do
  begin
    can_delete = 1;

    if (ol_oper_type = 2 and
        ol_oper_subtype in (31, 32, 33, 34)) then
      can_delete = 0;

    if (kob_client_dt is null and
        kob_client_kt is null and
        ol_document_client is not null) then
    begin
      if (sab_code_dt starting 'Деньги.') then
        kob_client_kt = ol_document_client;

      if (sab_code_kt starting 'Деньги.') then
        kob_client_dt = ol_document_client;
    end

    if (acc_filter_kind = 0) then
    begin
      if (sab_filter_code_dt <> '') then
        if (sab_code_dt not starting sab_filter_code_dt and
            sab_code_kt not starting sab_filter_code_dt) then
          continue;
    end
    else
    begin
      if (sab_filter_code_dt <> '') then
        if (sab_code_dt not starting sab_filter_code_dt) then
          continue;

      if (sab_filter_code_kt <> '') then
        if (sab_code_kt not starting sab_filter_code_kt) then
          continue;
    end

    kob_client_dt_name = '';
    kob_client_kt_name = '';
    if (ol_doc_client_name is null) then
      ol_doc_client_name = '';

    if (kob_detail is null) then
      kob_detail = '';
    if (ol_memo is null) then
      ol_memo = '';

    if (memo_text <> '') then
    begin
      if (position(memo_text, ol_memo) = 0 and
          position(memo_text, kob_detail) = 0) then
        continue;
    end

    if (kob_client_dt is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :kob_client_dt
      into :kob_client_dt_name;
    end
    if (kob_client_kt is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :kob_client_kt
      into :kob_client_kt_name;
    end

    if (kob_client_dt_name = '' and
        kob_client_kt_name = '' and
        ol_doc_client_name <> '') then
    begin
      if (sab_code_dt starting 'Деньги.') then
        kob_client_dt_name = ol_doc_client_name;

      if (sab_code_kt starting 'Деньги.') then
        kob_client_kt_name = ol_doc_client_name;
    end

    if (client_text <> '') then
    begin
      if (position(client_text, kob_client_dt_name) = 0 and
          position(client_text, kob_client_kt_name) = 0) then
        continue;
    end

    if (ol_doc_kind <> ol_doc_kind_prior) then
    begin
      select dn_name
      from docname
      where dn_id = :ol_doc_kind
      into :dn_name;
      ol_doc_kind_prior = :ol_doc_kind;
    end

    if (ol_pocket <> ol_pocket_prior) then
    begin
      select mo_name
      from mopocket
      where mo_id = :ol_pocket
      into :mo_name;
      ol_pocket_prior = :ol_pocket;
    end

    if (ol_from_reg_id_prior <> ol_from_reg_id) then
    begin
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :ol_from_reg_id and
            r1.parent = r2.reg_unicode
      into :ol_region_name, ol_branch_name;
      ol_from_reg_id_prior = ol_from_reg_id;
    end

    if (ol_user_id <> ol_user_id_prior) then
    begin
      select un_name
      from username
      where un_id = :ol_user_id
      into :un_name;
      ol_user_id_prior = ol_user_id;
    end

    if (kob_acc_dt_prior <> :kob_acc_dt) then
    begin
      kob_acc_dt_code = f_fmt_acc(kob_acc_dt);
      select acc_name
      from ref_account
      where acc_id = :kob_acc_dt
      into :kob_acc_dt_name;
    end
    if (kob_acc_kt_prior <> :kob_acc_kt) then
    begin
      kob_acc_kt_code = f_fmt_acc(kob_acc_kt);
      select acc_name
      from ref_account
      where acc_id = :kob_acc_kt
      into :kob_acc_kt_name;
    end

    is_delete_name = null;
    if (is_delete = 1) then
      is_delete_name = 'Аннулирован';
    if (is_delete_name is null) then
      is_delete_name = '';
    if (kob_detail <> '' and
        ol_memo <> '') then
    begin
      if (kob_detail <> ol_memo) then
        ol_calc_memo = left(:kob_detail || ' ' || :ol_memo, 1024);
      else
        ol_calc_memo = :ol_memo;
    end
    else
    begin
      if (kob_detail <> '') then
        ol_calc_memo = :kob_detail;
      else
        ol_calc_memo = :ol_memo;
    end
    suspend;
  end
end
