CREATE PROCEDURE MATLIST_BALANCE_MATOTVET(
    REGION_ID D_INTEGER,
    MATOTVET_ID D_ID,
    DD D_DATE,
    IS_OWN D_BOOL = 0)
RETURNS (
    MBAL_BILL_ID D_ID,
    MBAL_BILL_LINE_NO D_INTEGER,
    MBAL_WP_ID D_ID,
    MBAL_MO_ID D_ID,
    MBAL_KOLICH D_CURRENCY,
    MBAL_SUMMA D_CURRENCY,
    MBAL_BILL_COUNT D_INTEGER,
    MBAL_ACCOUNT D_STR_20,
    MBAL_IS_SKLAD D_SMALLINT,
    MBAL_WP_NAME D_STR_100,
    MBAL_REGION_ID D_INTEGER,
    ML_NAME D_STR_255,
    YL_CODE D_STR_20,
    MBAL_ML_ID D_ID,
    MBAL_INVENT_NO D_INTEGER,
    MBAL_INVENT_STR D_STR_CODE,
    MBAL_MC_ID D_ID,
    MBAL_FULL_DETAIL D_STR_2048)
AS
declare variable wp d_id;
begin
  for
    select
      om_bill_id,
      om_bill_line_no,
      om_workplace_id,
      om_matotvet_id,
      om_account,
      wp_status,
      wp_name,
      wp_from_reg_id,
      sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) as kol_balance,
      sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) as sum_balance
    from opermat
      inner join operlist
        on ol_id = om_from_ol_id
          and ol_date_oper_day <= :dd
          and is_delete = 0
      inner join workplace
        on wp_id = om_workplace_id
          and wp_from_reg_id = :region_id
    where om_matotvet_id = :matotvet_id
    group by
      om_bill_id,
      om_bill_line_no,
      om_workplace_id,
      om_matotvet_id,
      om_account,
      wp_status,
      wp_name,
      wp_from_reg_id
    having sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) <> 0
      or sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) <> 0
    order by om_workplace_id, om_matotvet_id, om_bill_id, om_bill_line_no
  into
    :mbal_bill_id,
    :mbal_bill_line_no,
    :mbal_wp_id,
    :mbal_mo_id,
    :mbal_account,
    :wp,
    :mbal_wp_name,
    :mbal_region_id,
    :mbal_kolich,
    :mbal_summa
  do
  begin
    if (mbal_account is null) then
      mbal_account = '';
    if (mbal_kolich is null) then
      mbal_kolich = 0;
    if (mbal_summa is null) then
      mbal_summa = 0;
    if (wp = 6) then
      mbal_is_sklad = 1;
    else
      mbal_is_sklad = 0;
    mbal_bill_count = 1;
    ml_name = '';
    yl_code = '';
    select
      ob_invent_no,
      ob_invent_str,
      ml_id,
      ml_name,
      yl_code,
      ob_from_mc_id,
      ob_full_detail
    from operbill
      inner join matlist
        on ml_id = ob_from_ml_id
      inner join yedizlist
        on yl_id = ml_from_yl_id
    where ob_bill_id = :mbal_bill_id
      and ob_bill_line_no = :mbal_bill_line_no
      and :dd between ob_date_begin and ob_date_end
      and ob_is_delete = 0
      and ((:is_own = 0 and ob_invent_no > 0) or ob_from_own_c_id > 0)
    into
      :mbal_invent_no,
      :mbal_invent_str,
      :mbal_ml_id,
      :ml_name,
      :yl_code,
      :mbal_mc_id,
      :mbal_full_detail;
    if (ml_name is null) then
      ml_name = '';
    if (ml_name <> '') then
      suspend;
  end
end
