CREATE PROCEDURE AAREP_AZ_ONLINE_KMP(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    CL_FIO D_STR_100,
    PHONE_NUMBER D_STR_20,
    EMA_DATE D_DATE,
    EMA_DATETIME D_DATETIME,
    REG_NAME D_STR_100,
    UN_FIO D_STR_100,
    EMA_BLOCK_DATETIME D_DATETIME,
    EMA_STATUS_STR D_STR_CODE,
    DECLINE_REASON D_STR_255,
    MA_NO D_ID)
AS
declare variable EMA_FROM_TKT_ID D_ID;
declare variable EMA_STATUS D_INTEGER;
declare variable MA_ID D_ID;
declare variable EMA_LO_ID D_ID;
begin
  for select cl.c_name_fio,  -- ФИО клиента
             ema.ema_date,
             ema.ema_datetime, -- Время создания онлайн-заявки
             reg_code || ' ' || reg_name,  -- Офис онлайн-заявки
             ema.ema_block_datetime, -- Время блокировки заявки
             ema.ema_status,
             r.rd_name, -- Причина отказа по онлайн-заявке
             ema.ema_from_tkt_id,
             ema.ema_lo_id,
             mcl.mcl_login
      from extminiapp ema
      inner join ticket tkt on tkt.tkt_id = ema.ema_from_tkt_id
      inner join region on reg_unicode = ema.ema_office_id
      inner join region_list(:rep_region_id, 1, -1, 1) on id_list = reg_unicode
      inner join clients cl on cl.c_id = ema.ema_from_c_id  -- клиент
      inner join refdata r on r.rd_id = ema.ema_result_rd_id
      inner join mc_login mcl on mcl.mcl_id = ema.ema_from_mcl_id
      where ema.ema_date between :rep_date_begin and :rep_date_end
        and f_source_str(tkt.tkt_source) = 'FsClient'
        and mcl.mcl_is_delete = 0
      into :cl_fio,
           :ema_date,
           :ema_datetime,
           :reg_name,
           :ema_block_datetime,
           :ema_status,
           :decline_reason,
           :ema_from_tkt_id,
           :ema_lo_id,
           :phone_number
  do
  begin
    ma_no = null;
    ma_id = null;

    select ma_id, ma_no
    from miniapp maa
    where ma_from_tkt_id = :ema_from_tkt_id
    rows 1
    into :ma_id, ma_no;
        
    -- Статус онлайн-заявки
    ema_status_str = '';
    ema_status_str = (case ema_status
                        when -1 then 'Заявка отклонена'
                        when 0 then (case
                                        when ema_block_datetime is null then  'Заявка в процессе'
                                        when ema_block_datetime is not null then 'Заблокировано'
                                     end)
                        when 1 then 'Заявка одобрена'
                      end);

    
    un_fio = null;
    select c.c_name_fio
    from clients c
    where c.c_id = :ema_lo_id
    into un_fio;

    if (:decline_reason is null or :decline_reason = 'Не заполнен') then
    begin
        select first 1 kr.kr_text
        from kontreslist krl
        join kontres kr on kr.kr_id = krl.krl_from_kr_id
        where krl.krl_ma_id = :ma_id
        order by krl.krl_datetime desc
        into decline_reason;
    end

    suspend;
  end
end
