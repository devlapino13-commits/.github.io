CREATE PROCEDURE AAREP_POST_INSPECTION_MAILOUT(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    ID D_ID,
    K_ID D_ID,
    C_ID D_ID,
    STATUS D_SMALLINT,
    DT_DISB D_DATE,
    DT_ROBOT D_DATETIME,
    DT_ARRIVAL D_DATETIME,
    DT_START D_DATETIME,
    DT_END D_DATETIME,
    TM_PROCESS D_INTEGER,
    TM_TOTAL D_INTEGER,
    K_NO D_STR_20,
    REG_NAME D_STR_CODE,
    OFF_NAME D_STR_CODE,
    PR_NAME D_STR_CODE,
    EMP_NAME D_STR_100,
    ST_NAME D_STR_20,
    KF_NAME D_STR_100,
    KC_NAME D_STR_100,
    MN_NAME D_STR_100,
    REASON D_STR_255,
    NOTE D_STR_2048,
    AMOUNT D_CURRENCY,
    NUM_CLI D_INTEGER,
    LST_CLI D_STR_2048,
    LST_POINT D_STR_100,
    LST_CIRCLE D_STR_100)
AS
declare variable sym_delim d_str_lang;
declare variable rd_id d_integer;
declare variable rd_id1 d_integer;
declare variable reg_id d_integer;
declare variable reg_lst d_str_8100;
declare variable cm_note d_str_4096;
declare variable p1 d_integer;
declare variable p2 d_integer;
declare variable p3 d_integer;
begin
  sym_delim = ascii_char(13) || ascii_char(10);
  rd_id = f_get_rd_id('ActionGroup.Check_Loan_Data_PostCheck');
  --rd_id1 = f_get_rd_id('ActionGroup.Check_Loan_Data');

  reg_lst = '';
  select
    list(id_list, ';')
  from region_list(:rep_region_id, 1, 3, 1)
  into
    :reg_lst;

  for
    select
      acl_id,
      acl_object_id,
      acl_from_reg_id,
      acl_respons_c_id,
      c_name_fio,
      acl_status,
      case acl_status
        when -2 then 'Отказан'
        when -1 then 'На доработку'
      end,
      acl_datetime,
      acl_last_datetime
    from actionlist
      inner join clients
        on c_id = acl_respons_c_id
    where acl_from_rd_id = :rd_id
      and acl_last_datetime >= :rep_date_begin and acl_last_datetime < (:rep_date_end + 1)
      and acl_status in (-2, -1)
    order by acl_last_datetime
  into
    :id,
    :k_id,
    :reg_id,
    :c_id,
    :emp_name,
    :status,
    :st_name,
    :dt_arrival,
    :dt_end
  do
  begin
    if (reg_lst not containing reg_id)
      then continue;

    select rd_id
    from hp_get_check_loan_data(:k_id)
    into:rd_id1;

    dt_robot = (
      select first 1
        acl_last_datetime
     from actionlist
     where acl_object_kind = 9
       and acl_object_id = :k_id
       and acl_from_rd_id = :rd_id1
       and acl_respons_c_id = 1000000000000
       and acl_id < :id
       and acl_status = 2
     order by acl_id desc);

    dt_start = (
      select first 1
        ovs_datetime
      from objectprops
        inner join objectvalues
          on ovs_from_opr_id = opr_id
            and ovs_value = 1
      where opr_object_kind = 9 and
            opr_object_id = :k_id and
            opr_subject_kind = -1 and
            opr_subject_id = :rd_id
      order by ovs_id);
    if (dt_start is null or status = 1)
      then dt_end = null;

    select
      k_kontract_no,
      k_fact_disb_date,
      k_amount_approved,
      k_kolich_client_approved,
      r1.reg_name,
      r2.reg_name,
      pr_name,
      k_fact_disb_kf_no,
      c1.c_name_fio,
      c2.c_name_fio
    from kontract
      inner join region as r1
        on r1.reg_unicode = k_last_reg_id
      inner join region as r2
        on r2.reg_unicode = r1.parent
      inner join product
        on pr_id = k_from_pr_id
      inner join clients as c1
        on c1.c_id = k_fact_disb_lo_id
      inner join clients as c2
        on c2.c_id = k_approved_manager
    where k_id = :k_id
    into
      :k_no,
      :dt_disb,
      :amount,
      :num_cli,
      :off_name,
      :reg_name,
      :pr_name,
      :kf_name,
      :kc_name,
      :mn_name;

    select
      list(c_name_fio, ', '),
      list(f_point(kc_from_c_id, :dt_disb), ', '),
      list(kc_circle, ', ')
    from knclient
      inner join clients
        on c_id = kc_from_c_id
    where kc_from_k_id = :k_id
      and kc_amount_approved > 0
    into
      :lst_cli,
      :lst_point,
      :lst_circle;

    cm_note = (
      select first 1
        replace(f_get_verify_memo(obm_memo, 'reason_memo'), :sym_delim, ', ')
      from objectprops
        inner join objectmemo
          on obm_from_opr_id = opr_id
      where opr_object_kind = 9
        and opr_object_id = :k_id
        and opr_subject_kind = -1
        and opr_subject_id = :rd_id
      order by obm_from_opr_id, obm_datetime desc);

    reason = '';
    note = '';
    if (cm_note <> '') then
    begin
      p1 = position(':', cm_note);
      p2 = position(']', cm_note);
      p3 = position('],', cm_note);

      if (p1 > 0 and p2 > p1) then
        reason = substring(cm_note from p1 + 2 for p2 - p1 -2);
      if (p3 > 0) then
        note = right(cm_note, char_length(cm_note) - p3 - 2);
    end

    tm_process = datediff(minute, dt_arrival, dt_start);
    tm_total = datediff(minute, dt_arrival, dt_end);

    suspend;
  end
end
