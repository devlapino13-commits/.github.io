CREATE PROCEDURE HP_MAKE_OPERBOOK_ACCPOSIT(
    ACCOUNT_DT VARCHAR(20),
    ACCOUNT_KT VARCHAR(20))
RETURNS (
    LINE_NO INTEGER,
    RES_ACCOUNT_DT VARCHAR(20),
    RES_ACCOUNT_KT VARCHAR(20),
    CM_DEFAULT_CURRENCY VARCHAR(3))
AS
declare variable reg_code varchar(20);
declare variable bg_id varchar(20);
declare variable bg_number integer;
declare variable foreign_curr varchar(3);
declare variable acc_tmp varchar(20);
declare variable currency_kt varchar(3);
declare variable currency_dt varchar(3);
begin
  select cm_default_currency
  from company
  into :cm_default_currency;

  line_no = 1;
  res_account_dt = '';
  res_account_kt = '';
  reg_code = left(account_dt, 4);

  currency_dt = substring(account_dt from 5 for 3);
  currency_kt = substring(account_kt from 5 for 3);
  foreign_curr = '';
  if (currency_dt = cm_default_currency) then
    foreign_curr = currency_kt;
  if (currency_kt = cm_default_currency) then
    foreign_curr = currency_dt;
  if (foreign_curr = '') then
  begin
    suspend;
    exit;
  end

  bg_id = null;
  line_no = 1;
  select ap_position_bg, ap_position_number
  from accposit
  where ap_from_cl_id = :foreign_curr and
        ap_date_begin <= current_date
  order by ap_date_begin desc
  rows 1
  into :bg_id, bg_number;
  if (currency_dt = foreign_curr) then
  begin
    res_account_dt = account_dt;
    res_account_kt = '';
    if (bg_id is not null) then
      res_account_kt = reg_code || foreign_curr || bg_id || right('0000000' || bg_number, 6);
  end
  else
  begin
    res_account_kt = account_kt;
    res_account_dt = '';
    if (bg_id is not null) then
      res_account_dt = reg_code || foreign_curr || bg_id || right('0000000' || bg_number, 6);
  end
  suspend;

  bg_id = null;
  line_no = 2;
  select ap_position_bg, ap_position_number
  from accposit
  where ap_from_cl_id = :cm_default_currency and
        ap_date_begin <= current_date
  order by ap_date_begin desc
  rows 1
  into :bg_id, bg_number;
  if (currency_dt = cm_default_currency) then
  begin
    res_account_dt = account_dt;
    res_account_kt = '';
    if (bg_id is not null) then
      res_account_kt = reg_code || cm_default_currency || bg_id || right('0000000' || bg_number, 6);
  end
  else
  begin
    res_account_kt = account_kt;
    res_account_dt = '';
    if (bg_id is not null) then
      res_account_dt = reg_code || cm_default_currency || bg_id || right('0000000' || bg_number, 6);
  end
  suspend;
end
