CREATE PROCEDURE HP_CHECK_IDENTICAL_ADDRESS(
    C_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    ORIG_FIO VARCHAR(100),
    ORIG_ADDRESS VARCHAR(255),
    OTHER_FIO VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_KF_NAME VARCHAR(100),
    K_KF_OFFICER VARCHAR(100),
    K_KF_REGION VARCHAR(100),
    PROS_TEXT VARCHAR(50),
    PROS_AMOUNT DECIMAL(18,4))
AS
declare variable k_id decimal(18,0);
declare variable pros_date date;
declare variable txt varchar(20);
declare variable st_staff integer;
declare variable st_client integer;
declare variable adl_id decimal(18,0);
declare variable found_c_id decimal(18,0);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  for select c_name_fio, adl_address_str, adl_id
      from clients, addrclient, addrlist
      where c_id = :c_id and
            adc_from_c_id = c_id and
            adc_from_adl_id = adl_id and
            adc_address_type = 2
      into :orig_fio, :orig_address, :adl_id
  do
  begin
    for select res_c_id
    from hp_identical_client_address(:adl_id, :c_id, '2')
    into :found_c_id
    do
    begin
      select c_name_fio from clients where c_id = :found_c_id
      into other_fio;

--      проверка клиента на просрчку
      for select oks_from_k_id, oks_balance, oks_delinq_date
          from oksummary, knclient
          where :rep_date between oks_date_begin and oks_date_end and
                :rep_date > oks_delinq_date and
                oks_balance > 0 and
                oks_close_state = 0 and
                oks_from_k_id = kc_from_k_id and
                kc_amount_approved > 0 and
                kc_from_c_id = :found_c_id
          into k_id, pros_amount, pros_date
      do
      begin
        select k_kontract_no, k_disb_date, kal_new_kf_name, c_name_fio, reg_name
        from knaktlist, kontract, clients, region
        where kal_from_kf_id = k_from_kf_id and
              k_id = :k_id and
              :rep_date between kal_date and kal_date_end and
              knaktlist.is_delete = 0 and
              kal_new_kf_officer = c_id and
              kal_new_reg_id = reg_unicode
        into :k_kontract_no, :k_disb_date, :k_kf_name, :k_kf_officer, :k_kf_region;

        txt = :pros_date;
        pros_text = 'Просроч. с ' || right(:txt, 2) || '.' || substring(:txt from 6 for 2) || '.' || left(:txt, 4);
        pros_text = :pros_text || ' баланс просрочки ' || cast(:pros_amount as decimal(15,2));

        suspend;
      end
    end
  end
end
