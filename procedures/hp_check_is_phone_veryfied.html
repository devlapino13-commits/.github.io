CREATE PROCEDURE HP_CHECK_IS_PHONE_VERYFIED(
    ACL_ID D_ID)
RETURNS (
    RES_STR D_STR_20)
AS
declare variable rd_id d_integer;
declare variable object_kind d_integer;
declare variable object_id d_id;
begin

  res_str = 'not_veryfied';

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ActionGroup.SimpleVerify_Phone_Number_Verify'
  into :rd_id;

  select acl.acl_object_kind, acl.acl_object_id
  from actionlist acl
  where acl.acl_id = :acl_id
  into :object_kind, :object_id;

  if (exists(select *
             from objectprops opr
             inner join objectmemo obm on opr.opr_id = obm.obm_from_opr_id
             where opr.opr_object_kind = :object_kind and
                   opr.opr_object_id = :object_id and
                   opr.opr_subject_id = :rd_id)) then
    res_str = 'veryfied';
  suspend;
end
