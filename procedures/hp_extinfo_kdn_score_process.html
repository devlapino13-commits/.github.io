CREATE PROCEDURE HP_EXTINFO_KDN_SCORE_PROCESS(
    EIL_ID D_ID)
AS
declare variable BLS_ID D_ID;
declare variable EIL_STATUS D_SMALLINT;
declare variable JSO D_INTEGER;
declare variable kdn d_currency;
declare variable kdd d_currency;
declare variable all_income d_currency;
declare variable overdue_payment d_currency;
declare variable ma_sh_payment d_currency;
declare variable object_kind d_integer;
declare variable object_id d_id;
declare variable mac_id d_id;
declare variable c_id d_id;
declare variable un_id d_id;
declare variable ma_id d_id;
begin
  -- RFCKZ-870 Обработка ответа от сервиса расчёта КДН и КДД
  select eil.eil_from_bls_id,
         eil.eil_status,
         eil.eil_object_kind,
         eil.eil_object_id,
         eil.eil_from_un_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eil.eil_id = :eil_id
    and eic.eic_code = 'kdn_score'
  into bls_id,
       eil_status,
       object_kind,
       object_id,
       un_id;

  if (bls_id is not null) then
    select iif(blx_blob is null, null, json.parse(blx_blob))
    from hp_get_blobex_blob(:bls_id)
    into jso;

  if (:eil_status = 2 and
      jso is not null) then
  begin
    if (f_objkind(:object_kind, 'code') = 'miniappclient') then
    begin
      select mac.mac_id,
             mac.mac_from_ma_id,
             mac.mac_from_c_id
      from miniappclient mac
      where mac.mac_id = :object_id
      into mac_id,
           ma_id,
           c_id;

      select sum(sh_all_with_tax) / max(sh_id)
      from fm_schedule_list(11, :ma_id)
      where sh_id <> 0
      into :ma_sh_payment;

      ma_sh_payment = coalesce(:ma_sh_payment, 0);
    end

    kdn = json.obj_get_val(jso, 'kdn_req_response.applicationReport.kdn_score_total');
    kdd = json.obj_get_val(jso, 'kdn_req_response.kdd_req_response.kdd_score_total');
    all_income = json.obj_get_val(jso, 'kdn_req_response.applicationReport.income');
    overdue_payment = json.obj_get_val(jso, 'kdn_req_response.kdd_req_response.debt_amount');


    if (kdn is not null and
        kdd is not null and
        all_income is not null and
        overdue_payment is not null) then
    begin
      update or insert into cfp_anketa_kdn (cakdn_kdn, cakdn_kdd, cakdn_from_mac_id, cakdn_from_c_id, cakdn_from_un_id, cakdn_net_profit, cakdn_other_dept, cakdn_monthly_payment)
      values (:kdn, :kdd, :mac_id, :c_id, :un_id, :all_income, :overdue_payment, :ma_sh_payment)
      matching (cakdn_from_mac_id, cakdn_from_c_id);
    end
  end

end
