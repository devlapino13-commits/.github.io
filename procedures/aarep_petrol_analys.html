CREATE PROCEDURE AAREP_PETROL_ANALYS(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_TRANS_DATE DATE)
RETURNS (
    PET_ID INTEGER,
    PET_NAME VARCHAR(255),
    PET_LABEL VARCHAR(50),
    PET_IS_RENT INTEGER,
    PROBEG_LITR DECIMAL(18,4),
    PROBEG_KM DECIMAL(18,4),
    PROBEG_SUMMA DECIMAL(18,4),
    PET_REGION VARCHAR(100),
    PET_FIO VARCHAR(100),
    PET_JOBNAME VARCHAR(255),
    PET_IS_DRIVER VARCHAR(20))
AS
declare variable c_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select pet_id, pet_name, pet_label, pet_is_rent
      from pettrans
      where pet_norma_km = 100
      order by pet_id
      into :pet_id, :pet_name, :pet_label, :pet_is_rent
  do
  begin
    probeg_litr = 0;
    probeg_km = 0;
    probeg_summa = 0;
    select reg_code || ' ' || reg_name, c_name_fio, c_id
    from region, petdata, clients
    where ped_region_place_id = reg_unicode and
          c_id = ped_from_c_id and
          ped_from_pet_id = :pet_id and
          ped_date <= :rep_trans_date
    order by ped_date desc
    rows 1
    into :pet_region, :pet_fio, :c_id;

    pet_jobname = '';
    select ps_text
    from hr_personal_last_position(:c_id, :rep_trans_date)
    where upper(ps_text) like '%ВОДИТЕЛ%'
    into :pet_jobname;
    if (pet_jobname is null) then
      pet_jobname = '';
    pet_is_driver = 0;
    if (pet_jobname <> '') then
      pet_is_driver = 1;
    pet_jobname = '';
    select ps_text
    from hr_personal_last_position(:c_id, :rep_trans_date)
    into :pet_jobname;
    if (pet_jobname is null) then
      pet_jobname = '';
    if (pet_jobname = '') then
    begin
      pet_is_driver = 1;
      pet_jobname = 'Арендованный водитель (на оказание услуг)';
    end

    select sum(peo_probeg_fact_liter), sum(peo_probeg_fact_km), sum(-1 * peo_trans_summa)
    from petopers, operlist
    where peo_from_ol_id = ol_id and
          operlist.is_delete = 0 and
          peo_probeg_fact_liter <> 0 and
          ol_date_oper_day between :rep_date_begin and :rep_date_end and
          peo_from_pet_id = :pet_id
    into :probeg_litr, :probeg_km, :probeg_summa;
    if (probeg_litr is null) then
      probeg_litr = 0;
    if (probeg_km is null) then
      probeg_km = 0;
    if (probeg_summa is null) then
      probeg_summa = 0;
    if (probeg_km <> 0) then
      suspend;
  end
end
