CREATE PROCEDURE HP_FT_CHECKED_DAYS_INFO(
    FIL_ID D_ID,
    FIC_DATE D_DATE)
RETURNS (
    TXT D_STR_8100)
AS
declare variable c_id d_id;
declare variable fib_id d_id;
declare variable fil_date_open d_date;
declare variable fil_date_close d_date;
declare variable fisl_date_begin d_date;
declare variable fisl_date_end d_date;
declare variable fisl_for_check_days_txt d_str_long;
declare variable not_in_stop_list d_str_8100;
declare variable fid_check_date_tmp d_str_8100;
declare variable txt_not_in_stop_list d_str_8100;
declare variable in_stop_list d_str_8100;
declare variable txt_in_stop_list d_str_8100;
declare variable fib_kind d_smallint;
begin
  select fil_from_fib_id,
         fil_date_open,
         coalesce(fil_date_close, cast('31.12.9999' as date)),
         fil_from_c_id,
         fib_kind
  from ft_infolist, ft_infoblank
  where fil_id = :fil_id
    and fil_from_fib_id = fib_id
  into fib_id, fil_date_open, fil_date_close, c_id, fib_kind;
  if (fil_date_close < fic_date) then fic_date = fil_date_close;

  select fisl_date_begin, fisl_date_end, fisl_for_check_days_txt
  from ft_info_stoplist sl
  where sl.fisl_from_c_id = :c_id
  into fisl_date_begin, fisl_date_end, fisl_for_check_days_txt;

  fisl_date_begin = coalesce(fisl_date_begin, '31.12.9999');
  fisl_date_end = coalesce(fisl_date_end, '31.12.9999');
  fisl_for_check_days_txt = coalesce(fisl_for_check_days_txt, '');

  not_in_stop_list = '';

  for select f_datetostr(fid_check_date)
      from ft_infodate, ft_infoblank
      where fid_from_fib_id = fib_id and
            fid_check_date < :fisl_date_begin and
            fid_check_date > :fil_date_open and
            fid_check_date <= :fic_date and
            fib_kind = :fib_kind
      order by fid_check_date
      into :fid_check_date_tmp
  do
  begin
    not_in_stop_list = not_in_stop_list || fid_check_date_tmp || ', ';
  end
  if (not_in_stop_list <> '') then
    not_in_stop_list = substring(not_in_stop_list from 1 for character_length(not_in_stop_list) - 2);

  txt_not_in_stop_list = '';

  if (fil_date_open not between :fisl_date_begin and :fisl_date_end) then
    txt_not_in_stop_list = 'ДАТА ПРОВЕРКИ: ' || f_datetostr(fil_date_open) || ', ';

  if (position('ДАТА ПРОВЕРКИ' in txt_not_in_stop_list) > 0) then
    txt_not_in_stop_list = txt_not_in_stop_list || not_in_stop_list;
  else
  if (not_in_stop_list <> '') then
    txt_not_in_stop_list = 'ДАТА ПРОВЕРКИ: ' || not_in_stop_list;

  if (position('ДАТА ПРОВЕРКИ' in txt_not_in_stop_list) > 0) then
    txt_not_in_stop_list = txt_not_in_stop_list || ascii_char(13) || ascii_char(10) || 'РЕЗУЛЬТАТ ПРОВЕРКИ:  КЛИЕНТ ОТСУТСТВУЕТ В ПЕРЕЧНЕ РОСФИНМОНИТОРИНГА ТЕРРОРИСТОВ/ЭКСТРЕМИСТОВ И В ПЕРЕЧНЕ ЛИЦ, В ОТНОШЕНИИ КОТОРЫХ ПРИНЯТЫ РЕШЕНИЯ О ПРИМЕНЕНИИ МЕР ПО ЗАМОРАЖИВАНИЮ/БЛОКИРОВАНИЮ ДЕНЕЖНЫХ СРЕДСТВ И ИНОГО ИМУ�©ЕСТВА '||trim(iif(fib_id >= 7, ' И ПЕРЕЧНЕ ОНН', ''))|| ascii_char(13) || ascii_char(10);

  in_stop_list = '';

  for select f_datetostr(fid_check_date)
      from ft_infodate, ft_infoblank
      where fid_from_fib_id = fib_id and
            fid_check_date between :fisl_date_begin and :fisl_date_end and
            fid_check_date > :fil_date_open and
            fid_check_date <= :fic_date and
            fib_kind = :fib_kind
      order by fid_check_date
      into :fid_check_date_tmp
  do
  begin
    in_stop_list = in_stop_list || fid_check_date_tmp || ', ';
  end
  if (in_stop_list <> '') then
    in_stop_list = substring(in_stop_list from 1 for character_length(in_stop_list) - 2);

  txt_in_stop_list = '';

  if (fil_date_open between :fisl_date_begin and :fisl_date_end) then
    txt_in_stop_list = 'ДАТА ПРОВЕРКИ: ' || f_datetostr(fil_date_open) || ', ';

  if (position('ДАТА ПРОВЕРКИ' in txt_in_stop_list) > 0) then
    txt_in_stop_list = txt_in_stop_list || in_stop_list;
  else
  if (in_stop_list <> '') then
    txt_in_stop_list = 'ДАТА ПРОВЕРКИ: ' || in_stop_list;

  if (position('ДАТА ПРОВЕРКИ' in txt_in_stop_list) > 0) then
  begin
    txt_in_stop_list = txt_in_stop_list || ascii_char(13) || ascii_char(10) || fisl_for_check_days_txt;
  end

  txt = txt_not_in_stop_list || txt_in_stop_list;

  suspend;
end
