CREATE PROCEDURE HP_INSERT_FP_EXCLUDES(
    C_ID D_ID,
    REASON D_STR_255,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable c_name_fio d_str_255;
declare variable fp_name d_str_255;
declare variable fp_gr d_date;
declare variable cnt d_integer;
begin
    res_text = '';

    -- проверка на пустой id
    if (coalesce(c_id, 0) = 0) then
    begin
        res_text = 'Укажите ID клиента';
        suspend;
        exit;
    end

    -- имя клиента
    select first 1 c_name_fio
    from clients
    where c_id = :c_id
    into :c_name_fio;

    if (c_name_fio is null) then
        c_name_fio = 'ID = ' || c_id;

    cnt = 0;

    -- ЕДИНСТВЕННЫЙ вызов aarep_check_finpolis_tmp
    for
        select cf.fp_name, cf.fp_gr
        from aarep_check_finpolis_tmp(:c_id) cf
        into :fp_name, :fp_gr
    do
    begin
        -- попытка восстановить удалённую запись
        update finpolis_excludes fe
           set fe.fpe_is_delete       = 0,
               fe.fpe_change_un_id    = :un_id,
               fe.fpe_change_datetime = current_timestamp
         where fe.fpe_from_c_id = :c_id
           and fe.fpe_nameu     = :fp_name
           and fe.fpe_gr        = :fp_gr
           and fe.fpe_is_delete = 1;

        -- если не восстановили — пробуем вставить
        if (row_count = 0) then
        begin
            insert into finpolis_excludes (
                fpe_from_c_id,
                fpe_nameu,
                fpe_gr,
                fpe_reason,
                fpe_create_un_id,
                fpe_change_un_id
            )
            select
                :c_id,
                :fp_name,
                :fp_gr,
                :reason,
                :un_id,
                :un_id
            from rdb$database
            where not exists (
                select 1
                from finpolis_excludes fe
                where fe.fpe_from_c_id = :c_id
                  and fe.fpe_nameu     = :fp_name
                  and fe.fpe_gr        = :fp_gr
            );
        end

        cnt = cnt + 1;
    end

    -- если совпадений не было
    if (cnt = 0) then
    begin
        if (exists (
            select 1
            from finpolis_excludes fe
            where fe.fpe_from_c_id = :c_id
              and fe.fpe_is_delete = 0
        )) then
            res_text = 'Исключения по клиенту ' || :c_name_fio || ' уже актуальны.';
        else
            res_text = 'Совпадения по перечню ЭТ по клиенту ' || :c_name_fio || ' не найдены.';

        suspend;
    end
end
