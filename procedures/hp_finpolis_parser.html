CREATE PROCEDURE HP_FINPOLIS_PARSER(
    A_FP_ID D_ID NOT NULL,
    A_IS_PARSED D_BOOL NOT NULL,
    A_IS_FUZZY D_BOOL NOT NULL,
    A_NAMEU D_STR_255 = '')
AS
declare variable n integer;
declare variable txt_parsed d_str_2048;
begin
  if ((a_is_parsed = 1) or (a_is_fuzzy = 1)) then
  begin
    if (a_is_parsed = 1) then
      delete from finpolisparced
      where fpp_from_fp_id = :a_fp_id;

    if (a_is_fuzzy = 1) then
      delete from finpolisnames
      where fpn_from_fp_id = :a_fp_id;

    if (a_nameu > '') then
    begin
      a_nameu = replace(:a_nameu, '(', ';');
      a_nameu = replace(:a_nameu, ')', ';');
      a_nameu = f_replace_all(:a_nameu, '  ', ' ');
      a_nameu = replace(:a_nameu, ' ;', ';');
      a_nameu = f_replace_all(:a_nameu, '  ', ' ');
      a_nameu = upper(:a_nameu);

      while (a_nameu > '') do
      begin
        n = position(';', a_nameu) - 1;
        if (a_nameu <> ';') then
        begin
          if (n >= 0) then
          begin
            if (n > 250) then
              n = 250;
            txt_parsed = left(:a_nameu, n);
            a_nameu = substring(:a_nameu from n + 2 for 255);
            a_nameu = trim(:a_nameu);
          end
          else
          begin
            if (:a_nameu > '') then
            begin
              if (char_length(:a_nameu) <= 249) then
                txt_parsed = :a_nameu;
              else
                txt_parsed = left(:a_nameu, 249);
              a_nameu = '';
            end
          end

          txt_parsed = trim(:txt_parsed);
          if (char_length(:a_nameu) > 250) then
            a_nameu = left(:a_nameu, 250);

          if (txt_parsed > '') then
          begin
            if (a_is_parsed = 1) then
              if (not(exists(select fpp_from_fp_id
                             from finpolisparced
                             where fpp_fio = :txt_parsed and
                                   fpp_from_fp_id = :a_fp_id))) then
                insert into finpolisparced (fpp_fio, fpp_from_fp_id)
                values (:txt_parsed, :a_fp_id);

            if (a_is_fuzzy = 1) then
            begin

              select res_text
              from hp_txt_clear(:txt_parsed)
              into :txt_parsed; -- очистка от слов уулу угли кызы кизи

              if (:txt_parsed starting 'ЧП ') then
                :txt_parsed = substring(:txt_parsed from 4);

              if (:txt_parsed starting 'ИП ') then
                :txt_parsed = substring(:txt_parsed from 4);

              if (not(exists(select fpn_from_fp_id
                             from finpolisnames
                             where fpn_fio = :txt_parsed and
                                   fpn_from_fp_id = :a_fp_id))) then
                insert into finpolisnames (fpn_fio, fpn_from_fp_id, fpn_metaphone)
                values (:txt_parsed, :a_fp_id, (select out_text
                                                from hp_simple_metaphone(:txt_parsed)));
            end
          end
        end
        else
          a_nameu = '';
      end
    end
  end
end
