CREATE PROCEDURE AAREP_LIST_GROUP_PAR8_180_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OKS_REGION_ID INTEGER,
    OKS_KC_ID ID_TYPE,
    OKS_FROM_K_ID ID_TYPE,
    OKS_BALANCE CURR_TYPE,
    DELINQ_BALANCE8_180 CURR_TYPE,
    DELINQ_DAY INTEGER,
    K_KONTRACT_NO VARCHAR(20),
    DATE_ADOPTION DATE,
    DELINQ_DAY_TRANSFER INTEGER,
    IS_OWN INTEGER)
AS
declare variable oks_delinq_date date;
declare variable k_from_kf_id id_type;
declare variable disb_date date;
declare variable disb_lo_id id_type;
begin

  for select oks_region_id, oks_kc_id, oks_from_k_id, oks_balance, oks_delinq_date
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_region_id = id_list and
            oks_close_state = 0 and
            not exists(select *
                       from sb_akt, sb_list -- Пункт 2.3
                       where sba_is_delete = 0 and
                             sba_from_sbl_id <> 0 and
                             sba_status = 2 and
                             :rep_date_end between sba_date_begin and sba_date_end and
                             sba_from_k_id = oks_from_k_id and
                             sba_from_sbl_id = sbl_id and
                             sbl_type = 1)
      into oks_region_id, oks_kc_id, oks_from_k_id, oks_balance, oks_delinq_date
  do
  begin
    delinq_day = 0;
    if (:rep_date_end > :oks_delinq_date) then
      delinq_day = :rep_date_end - :oks_delinq_date;

    select k_kontract_no, k_from_kf_id
    from kontract
    where k_id = :oks_from_k_id
    into k_kontract_no, k_from_kf_id;

    disb_date = null;
    disb_lo_id = null;
    select ks_date_begin, ks_kc_id
    from kontstate
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = :oks_from_k_id
    into disb_date, disb_lo_id;
    is_own = 1;
    date_adoption = null; -- Дата принятия
    delinq_day_transfer = null; -- кол-во просроченных дней на дату передачи
    if (oks_kc_id <> disb_lo_id) then
    begin
      select kal_date
      from knaktlist
      where knaktlist.is_delete = 0 and
            kal_from_kf_id = :k_from_kf_id and
            :rep_date_end between kal_date and kal_date_end
      into date_adoption;

      select :date_adoption - oks_delinq_date
      from oksummary
      where oks_from_k_id = :oks_from_k_id and
            oks_balance > 0 and
            oks_close_state = 0 and
            :date_adoption between oks_date_begin and oks_date_end
--      and :date_adoption - oks_delinq_date > 7
      into delinq_day_transfer;
      if (delinq_day_transfer < 0) then
        delinq_day_transfer = 0;
      if (delinq_day_transfer > 7) then
        is_own = 0;
    end
    delinq_balance8_180 = 0;
    if (delinq_day between 8 and 180 and
        is_own = 1) then
    begin
      delinq_balance8_180 = oks_balance;
    end

    suspend;
  end
end
