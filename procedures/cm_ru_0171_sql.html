CREATE PROCEDURE CM_RU_0171_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable PR_CODE D_STR_100;
declare variable KC_FROM_C_ID D_ID;
declare variable COUNT_ACTIV_CREDITS D_SMALLINT;
declare variable COUNT_ACTIV_PAR D_SMALLINT;
declare variable CL_POINTS D_SMALLINT;
declare variable COUNT_RELIABLE_CLIENT_PLUS D_SMALLINT;
declare variable CL_ALL_BALANCE_OS D_CURRENCY;
declare variable CL_ALL_OS D_CURRENCY;
declare variable LIMIT_AMOUNT_MAX D_CURRENCY;
declare variable HAVE_DELINQ D_SMALLINT;
declare variable K_REFUND D_SMALLINT;
declare variable IS_OK D_INTEGER;
declare variable COUNT_ADVANTAGEOUS_SIDE D_INTEGER;
declare variable COUNT_ACTIV_NON_PARAL D_INTEGER;
declare variable COUNT_USPEH D_SMALLINT;
declare variable COUNT_PAR D_SMALLINT;
declare variable COUNT_SHANS D_SMALLINT;
declare variable COUNT_EXPRES D_SMALLINT;
declare variable COUNT_BLAGO D_SMALLINT;
declare variable COUNT_RELIABLE_CLIENT D_SMALLINT;
declare variable COUNT_ACTIV_EXPRESS_ONLINE D_SMALLINT;
declare variable COUNT_ACTIV_SUMMER_PLUS D_SMALLINT;
declare variable COUNT_ACTIV_COMFORTABLE D_SMALLINT;
declare variable KC_ID D_ID;
declare variable COUNT_ACTIV_ZALOG D_SMALLINT;
declare variable COUNT_ACTIV_SELFEMPLOYED D_SMALLINT;
declare variable COUNT_ACTIV_BUSINESS D_SMALLINT;
declare variable NOT_CHECK_PDN D_BOOL;
declare variable COUNT_UNIVERSAL D_SMALLINT;
declare variable COUNT_PRIVILEGE D_SMALLINT;
declare variable COUNT_PRIVILEGE_PLUS D_SMALLINT;
declare variable COUNT_OIVO D_SMALLINT;
declare variable COUNT_PLUS_60 D_SMALLINT;
declare variable COUNT_B_BONUS D_SMALLINT;
declare variable COUNT_SWITCH_PLUS D_SMALLINT;
begin
  select k_disb_date, pr_code, k_refund
  from kontract k
  inner join product pr on k_from_pr_id = pr_id
  where k_id = :k_id
  into :kf_disb_date, pr_code, k_refund;
  if (kf_disb_date < '01.11.20' or pr_code containing '.ZALOG_CAR.' or pr_code containing '.reliable_emp.' or pr_code containing '.zalog.' or
      pr_code containing '.selfemployed.' or
      pr_code containing '.business.') then
    exit;
  for select c_name_fio, kc_from_c_id, kc_amount_approved
      from knclient knc
      inner join clients c on kc_from_c_id = c_id
      where knc.kc_from_k_id = :k_id and
            knc.kc_amount_approved > 0
      into res_info_text, kc_from_c_id, kc_amount
  do
  begin
    select
           --
           --           sum(1) as count_activ_credits,
           sum(iif(pr_code not containing '.express_online.', 1, 0)) count_activ_credits,
           sum(iif(pr_code not containing '.PAR.', 1, 0)) count_activ_non_paral, --
           sum(iif(pr_code containing '.PAR.', 1, 0)) count_activ_par, --
           sum(iif(pr_code containing '.advantageous_side.', 1, 0)) count_advantageous_side, --
           sum(iif(pr_code not containing '.ZAL.' and pr_code not containing '.express_online.', oks_balance, 0)) cl_all_balance_os, --
           sum(iif(pr_code containing '.reliable_client_plus.', 1, 0)) count_reliable_client_plus, -- надежный клиент плюс
           sum(iif(oks.oks_delinq_date < :kf_disb_date, 1, 0)) have_delinq, --
           sum(iif(pr_code containing '.uspeh.', 1, 0)) count_uspeh, -- успех
           sum(iif(pr_code containing '.PAR.LOYAL.', 1, 0)) count_par, -- паралельный
           sum(iif(pr_code containing '.shans.', 1, 0)) count_shans, -- шанс
           sum(iif(pr_code containing '.express.', 1, 0)) count_expres, -- экспресс
           sum(iif(pr_code containing '.blago.', 1, 0)) count_blago, -- благо
           sum(iif(pr_code containing '.reliable_client.', 1, 0)) count_reliable_client, -- надежный клиент
           sum(iif(pr_code containing '.express_online.', 1, 0)) count_activ_express_online,
           sum(iif(pr_code containing '.summer_plus.', 1, 0)) count_activ_summer_plus,
           sum(iif(pr_code containing '.comfortable.', 1, 0)) count_activ_comfortable,
           sum(iif(pr_code containing '.zalog.', 1, 0)) count_activ_zalog,
           sum(iif(pr_code containing '.selfemployed.', 1, 0)) count_activ_selfemployed,
           sum(iif(pr_code containing '.business.', 1, 0)) count_activ_business,
           sum(iif(pr_code containing '.universal.', 1, 0)) count_universal,

           sum(iif(pr_code containing '.privilege.', 1, 0)) count_privilege,
           sum(iif(pr_code containing '.privilege_plus.', 1, 0)) count_privilege_plus,
           sum(iif(pr_code containing '.oivo.', 1, 0)) count_oivo,
           sum(iif(pr_code containing '.plus_60.', 1, 0)) count_plus_60,
           sum(iif(pr_code containing '.b_bonus.', 1, 0)) count_b_bonus,
           sum(iif(pr_code containing '.switch_plus.', 1, 0)) count_switch_plus
    from knclient kc
    -->>
    inner join kontstate ks on kc_from_k_id = ks_from_k_id and
          ks.ks_ol_is_delete = 0 and
          ks.ks_state = 1 and
          :kf_disb_date between ks_date_begin and ks_date_end
    inner join kontract k on ks_from_k_id = k_id
    inner join product pr on k.k_from_pr_id = pr_id
    inner join oksummary oks on ks_from_k_id = oks_from_k_id and
          :kf_disb_date between oks_date_begin and oks_date_end and
          oks.oks_close_state = 0
          --<<
    where kc_from_c_id = :kc_from_c_id
    into count_activ_credits, count_activ_non_paral, count_activ_par, count_advantageous_side, cl_all_balance_os,
         count_reliable_client_plus, have_delinq, count_uspeh, count_par, count_shans, count_expres, count_blago,
         count_reliable_client, count_activ_express_online, count_activ_summer_plus, count_activ_comfortable, count_activ_zalog,
         count_activ_selfemployed, count_activ_business, count_universal, count_privilege, count_privilege_plus, count_oivo,
         count_plus_60, count_b_bonus, count_switch_plus;
    count_activ_credits = coalesce(count_activ_credits, 0);
    count_activ_non_paral = coalesce(count_activ_non_paral, 0);
    count_activ_par = coalesce(count_activ_par, 0);
    cl_all_balance_os = coalesce(cl_all_balance_os, 0);
    count_reliable_client_plus = coalesce(count_reliable_client_plus, 0);
    have_delinq = coalesce(have_delinq, 0);
    count_advantageous_side = coalesce(count_advantageous_side, 0);
    count_uspeh = coalesce(count_uspeh, 0);
    count_par = coalesce(count_par, 0);
    count_shans = coalesce(count_shans, 0);
    count_expres = coalesce(count_expres, 0);
    count_blago = coalesce(count_blago, 0);
    count_reliable_client = coalesce(count_reliable_client, 0);
    count_activ_express_online = coalesce(count_activ_express_online, 0);
    count_activ_summer_plus = coalesce(count_activ_summer_plus, 0);
    count_activ_comfortable = coalesce(count_activ_comfortable, 0);
    count_activ_par = count_activ_par - count_activ_express_online;
    count_activ_zalog = coalesce(count_activ_zalog, 0);
    count_activ_selfemployed = coalesce(count_activ_selfemployed, 0);
    count_activ_business = coalesce(count_activ_business, 0);
    count_universal = coalesce(count_universal, 0);
    count_privilege = coalesce(count_privilege, 0);
    count_privilege_plus = coalesce(count_privilege_plus, 0);
    count_oivo = coalesce(count_oivo, 0);
    count_plus_60 = coalesce(count_plus_60, 0);
    count_b_bonus = coalesce(count_b_bonus, 0);
    if (have_delinq > 0) then
    begin
      /*    MRAKHMANOV
        2)    Клиент не может получить заем при наличии непогашенной просроченной
        задолженности по другому активному займу клиента в Компании, за исключением,
        когда самим продуктом предусмотрена такая возможность.   */
      res_error_is_critical = 1;
      res_error_text = 'У клиента активная просрочка по другому займу в Компании, выдача нового займа запрещена';
      suspend;
    end

    -- для продукта "Доверие" не проверяем ограничения по макс.сумме кол-во активных займов
    if (pr_code containing '.express_online.' or pr_code containing '.support.') then
      continue;

    -- Для продукта Лето Плюс проверка делается отдельно
    if (pr_code containing '.summer_plus.') then
      continue;
    if (pr_code containing '.comfortable.') then
      continue;
    if (pr_code containing '.plus_60.') then
      continue;
    if (pr_code containing '.plus_100.') then
      continue;
    if (pr_code containing '.plus_100_b.') then
      continue;
    if (pr_code containing '.b_bonus.') then
      continue;
    if (pr_code containing '.switch_plus.') then
      continue;
    is_ok = 0;
    if (count_activ_credits = 0) then
      is_ok = 1;
    if (k_refund = 1) then
      is_ok = 1;

    if (is_ok = 0 and
        count_activ_summer_plus = 1) then
      is_ok = 1;

    if (is_ok = 0 and
        count_activ_comfortable = 1) then
      is_ok = 1;

    if (is_ok = 0) then
    begin
      if (count_activ_credits - count_activ_comfortable = 1 and
          pr_code containing '.par.' and
          count_activ_par = 0 and
          count_advantageous_side = 0) then
        is_ok = 1;
    end
    if (is_ok = 0) then
    begin
      if (count_activ_credits = 1 and
          pr_code containing '.par.' and
          count_activ_par = 0 and
          count_advantageous_side = 1) then
        is_ok = 1;
    end
    if (is_ok = 0) then
    begin
      if (count_activ_credits - count_activ_comfortable = 1 and
          not(pr_code containing '.par.') and
          count_activ_par = 1 and
          count_advantageous_side = 0) then
        is_ok = 1;
    end
    if (is_ok = 0) then
    begin
      if (count_activ_credits = 1 and
          not(pr_code containing '.par.') and
          count_activ_par = 0 and
          count_advantageous_side = 1) then
        is_ok = 1;
    end
    if (is_ok = 0) then
    begin
      if (count_activ_zalog = 1 and
      count_activ_credits = 1) then
        is_ok = 1;
    end 
    if (is_ok = 0) then
    begin
      if (count_activ_selfemployed = 1 and
      count_activ_credits = 1) then
        is_ok = 1;
    end
    if (is_ok = 0) then
    begin
      if (count_activ_business = 1 and
      count_activ_credits = 1) then
        is_ok = 1;
    end
    if (is_ok = 0) then
    begin
      if (count_activ_credits = count_activ_comfortable) then
        is_ok = 1;
    end 


    -- count_advantageous_side
    if (is_ok = 0 and
        pr_code not containing '.privilege.' and
        pr_code not containing '.privilege_plus.') then
    begin
      /*    MRAKHMANOV
        3)    У клиента может быть не более двух активных займов в Компании.
        При этом:
        3.1.   клиент не может иметь два активных займа в Компании, за исключением случаев,
            когда второй активный заем  выдан по продукту, имеющему статус параллельного займа
            (заем считается параллельным, когда его выдача допускается при наличии другого
            активного займа и без обязанности полного закрытия текущего активного займа).   */
      if (not((pr_code containing '.loyal.' and (count_activ_par >= 1 or count_activ_par = 0)) or
              (pr_code containing '.reliable_client.' and (count_reliable_client >= 0 or count_activ_selfemployed >=1 or count_activ_business >= 1)) or
              (pr_code containing '.shans.' and count_shans = 0) or
              (pr_code containing '.universal.' and (count_universal >= 0 or count_activ_selfemployed >=1 or count_activ_business >= 1)) or
              (pr_code containing '.oivo.' and (count_oivo >= 0 or count_activ_selfemployed >=1 or count_activ_business >= 1)) or
              (pr_code containing '.express.' and count_expres = 0) -- RFCRU-4953
              ))
      then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Клиент не может иметь два активных займа в Компании';
        suspend;
      end
    end
    /*   MRAKHMANOV
       3.2.    сумма основного долга одного клиента по всем активным займам в Компании не должна превышать суммы,
       предусмотренной для его категории согласно накопленным баллам по программе лояльности,
       за исключением продукта «Микрозаем под залог».   */
    limit_amount_max = 0;
    cl_all_os = cl_all_balance_os + kc_amount;
    cl_points = f_point_get(9, k_id, kc_from_c_id, kf_disb_date);

    --Для клиентов с активным займом по продукту «Шанс», «Экспресс», «Благо», «Выгодная сторона» или «Надежный клиент», либо планирующих получить заем «Надежный клиент» при наличии «Параллельного»
    if (count_shans > 0 or count_expres > 0 or count_blago > 0 or count_advantageous_side > 0 or count_reliable_client > 0 or (pr_code containing '.reliable_client.' and
        count_par > 0) or count_privilege > 0 or count_privilege_plus > 0) then
    begin
      if (kf_disb_date < cast('26.06.2023' as date)) then
      begin
        limit_amount_max = case
                             when cl_points between 0 and 2 then 35000
                             when cl_points between 3 and 5 then 40000
                             when cl_points between 6 and 8 then 50000
                             when cl_points between 9 and 11 then 60000
                             when cl_points between 12 and 14 then 80000
                             when cl_points between 15 and 17 then 100000
                             when cl_points between 18 and 23 then 120000
                             when cl_points >= 24 then 150000
                           end;
      end
      else
      if (kf_disb_date < cast('04.07.2024' as date)) then
      begin
        limit_amount_max = case
                             when cl_points <= 2 then 55000
                             when cl_points <= 5 then 65000
                             when cl_points <= 8 then 75000
                             when cl_points <= 11 then 85000
                             when cl_points <= 14 then 100000
                             when cl_points <= 17 then 130000
                             when cl_points <= 23 then 150000
                             else 200000
                           end;
      end else
      begin
        select kc_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 rows 1
        into kc_id;

        -- RFCRU-2617 Если стоит галка "документально подтвержденный доход", то лимиты по ПДН не считаем до КК
        not_check_pdn = 0;

        if (check_kind not containing '.loan_disb.') then
          select 1
          from kontract, miniappclient, miniappclient_income
          where k_id = :k_id
            and mac_from_ma_id = k_from_ma_app_id
            and maci_from_mac_id = mac_id
            and maci_is_documented = 1
          rows 1
          into not_check_pdn;

        if (not_check_pdn = 0 and k_refund = 0) then
        begin
          for select error_is_critical, error_text
          from knclient_pdn_limit_check(:kc_id)
          into res_error_is_critical, res_error_text
          do
            suspend;
        end
      end
    end

      --Для клиентов с активным займом по продукту «Успех» или «Надежный клиент плюс», либо планирующих получить заем «Надежный клиент плюс» при наличии «Параллельного»
      if ((count_uspeh > 0) or (count_reliable_client_plus > 0) or (pr_code containing '.reliable_client_plus.' and
          count_par > 0)) then
      begin
        if (kf_disb_date < cast('26.06.2023' as date)) then
        begin
          limit_amount_max = case
                               when cl_points between 0 and 2 then 55000
                               when cl_points between 3 and 5 then 60000
                               when cl_points between 6 and 8 then 70000
                               when cl_points between 9 and 11 then 80000
                               when cl_points between 12 and 17 then 100000
                               when cl_points between 18 and 23 then 120000
                               when cl_points >= 24 then 150000
                             end;
        end
        else
        if (kf_disb_date < cast('04.07.2024' as date)) then
        begin
          limit_amount_max = case
                               when cl_points <= 2 then 75000
                               when cl_points <= 5 then 80000
                               when cl_points <= 8 then 90000
                               when cl_points <= 11 then 100000
                               when cl_points <= 17 then 130000
                               when cl_points <= 23 then 150000
                               else 200000
                             end;
        end
        else
        begin
          limit_amount_max = case
                               when cl_points <= 2 then 85000
                               when cl_points <= 5 then 100000
                               when cl_points <= 11 then 120000
                               when cl_points <= 17 then 150000
                               else 200000
                             end;
        end
      end

    if (pr_code containing '.reliable_client_plus.' and kf_disb_date >= cast('15.08.2024' as date)) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Выдачи по НК+ приостановлены, выберите другой продукт!';
      suspend;
    end 


    if (cl_all_os > limit_amount_max and
        limit_amount_max > 0 and
        k_refund = 0) then --k_refund = 0 добавил по письму Зары от 22.02.21 Музаффар
    begin
      limit_amount_max = limit_amount_max - cl_all_balance_os;
      if (limit_amount_max < 0) then
        limit_amount_max = 0;
      res_error_is_critical = 1;
      res_error_text = f_format('Баланс ОС по активным: %1 р. Разрешенная сумма для выдачи: %2, ПДН: , Балл: %3', round(cl_all_balance_os), round(limit_amount_max), cl_points);
      suspend;
    end
    /*  MRAKHMANOV
        Текущий активный заем по продукту «Благо» или «Успех» может быть перекредитован если:
            1)    У клиента нет активной просрочки на дату перекредитования;    */
    if (k_refund = 1 and
        have_delinq > 0 and
        (pr_code containing '.blago.' or pr_code containing '.uspeh.')) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'У клиента есть активная просрочка на дату перекредитования';
      suspend;
    end
  end
end
