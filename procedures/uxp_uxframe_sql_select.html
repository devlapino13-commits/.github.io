CREATE PROCEDURE UXP$UXFRAME_SQL_SELECT(
    FR_UUID D_UUID,
    FR_UUID_CODE D_STR_CODE = null)
RETURNS (
    SQL_TEXT D_STR_8100,
    SQL_IS_EXEC D_BOOL)
AS
declare variable field_name d_str_code;
declare variable field_list d_str_8100;
declare variable enter_sym d_str_10;
declare variable ufp_editor d_str_code;
declare variable fr_source d_str_code;
declare variable is_table d_bool;
declare variable buttonedit_sql d_str_long;
declare variable master_field d_str_code;
declare variable detail_field d_str_code;
declare variable master_field_count d_integer;
declare variable detail_field_count d_integer;
declare variable master_detail_where d_str_long;
declare variable res_str d_str_code;
declare variable is_detail d_bool;
declare variable param_index d_integer;
declare variable md_linked d_bool;
declare variable ufr_is_grid d_smallint;
declare variable ua_uxa_uuid d_uuid;
declare variable ua_usr_text_field d_str_code;
declare variable ux_kind d_str_code;
declare variable max_rows d_integer;
declare variable alias_letter d_str_10;
declare variable ux_code_msg d_str_100;
declare variable fr_code_msg d_str_100;
declare variable error_text d_str_255;
declare variable tmp_us_uuid d_uuid;
declare variable f_temp d_str_255;
declare variable ux_uuid d_uuid;
declare variable ufr_sql_where d_str_long;
declare variable ufr_sql_orderby d_str_long;
declare variable where_str d_str_code;
declare variable us_sql d_str_8100;
declare variable replaces_params d_str_long;
declare variable uf_domain d_str_code;
declare variable uf_select_name d_str_long;
declare variable edi_uuid d_uuid;
declare variable sys_uuid d_uuid;
declare variable edi_field_name d_str_code;
declare variable uf_type d_smallint;
declare variable ufr_add_table_fields d_str_code;
declare variable main_table d_str_code;
declare variable frame_keyfields d_str_code;
declare variable main_keyfields d_str_code;
declare variable fr_us_uuid d_uuid;
declare variable main_us_uuid d_uuid;
declare variable need_alias d_bool;
declare variable frame_alias d_str_code;
declare variable main_alias d_str_code;
declare variable uf_owner d_uuid;
declare variable join_sql d_str_8100;
begin
  -- Приготовление Select sql для фрейма

  if (coalesce(fr_uuid, '') = '') then
    fr_uuid = uxf$uniq(fr_uuid_code);

  select ufr_owner
  from ux$frame
  where ufr_uuid = :fr_uuid
  into ux_uuid;

  ux_code_msg = uxf$code(ux_uuid, 'fcode');
  fr_code_msg = uxf$code(fr_uuid, 'fcode');
  sys_uuid = uxf$sys_id();

  sql_text = '';
  field_list = '';
  enter_sym = ascii_char(13) || ascii_char(10);

  sql_is_exec = 0;
  master_field = '';
  detail_field = '';
  is_detail = 0;
  master_field_count = 0;
  detail_field_count = 0;
  replaces_params = '';


  select us_uuid, us_code, us_is_table, us_keyfields,
         ufr_is_grid, ufr_master_field, ufr_detail_field, iif(ufr_master_ufr_uuid is null, 0, 1),
         ufr_sql_where, ufr_sql_orderby,
         us_sql, ufr_add_table_fields
  from ux$frame, ux$source
  where ufr_uuid = :fr_uuid and
        ufr_us_uuid = us_uuid
  into fr_us_uuid, fr_source, is_table, frame_keyfields,
       ufr_is_grid, master_field, detail_field, is_detail,
       ufr_sql_where, ufr_sql_orderby,
       us_sql, ufr_add_table_fields;

  if (row_count = 0) then
    exception e_default f_format('Frame not found in the form. Form: %1, frame: %2', ux_code_msg, fr_code_msg);

  main_table = '';
  main_keyfields = '';
  main_us_uuid = null;
  need_alias = 0;
  frame_alias = 'a';
  main_alias = 't';

  if (frame_keyfields is null) then
    frame_keyfields = '';

  if (ufr_add_table_fields <> '') then
  begin
    select us_uuid, us_code, us_keyfields
    from ux$source
    where us_code = :ufr_add_table_fields and
          coalesce(us_keyfields, '') <> ''
    into main_us_uuid, main_table, main_keyfields;

    if (row_count = 0) then
      ufr_add_table_fields = '';

    if (coalesce(main_table, '') = '' or coalesce(frame_keyfields, '') = '' or coalesce(main_keyfields, '') = '') then
      ufr_add_table_fields = '';

    if (ufr_add_table_fields <> '') then
    begin
      if (exists(select *
                 from ux$field f, ux$field m
                 where f.uf_owner = :fr_us_uuid and
                       m.uf_owner = :main_us_uuid and
                       f.uf_is_param = 0 and
                       m.uf_is_param = 0 and
                       f.uf_field_name = m.uf_field_name)) then
        need_alias = 1;
    end
  end

  us_sql = trim(coalesce(us_sql, ''));

  if (us_sql <> '') then
  begin
    sql_text = us_sql;
    sql_is_exec = 0;
    suspend;
    exit;
  end

  ux_kind = uxf$code(ux_uuid, 'kind');
  alias_letter = '';

  if (is_detail = 1) then
  begin
    if (coalesce(master_field, '') = '' or
        coalesce(detail_field, '') = '') then
      exception e_message f_format('Поля мастер-деталь не настроены, форма: %1, фрейм: %2',
                                   ux_code_msg, fr_code_msg);

    select count(*)
    from uxp$split_str(:master_field, ';')
    into :master_field_count;

    select count(*)
    from uxp$split_str(:detail_field, ';')
    into :detail_field_count;

    if (detail_field_count <> master_field_count) then
      exception e_message f_format('Количество полей мастер и деталь не совпадает, форма: %1, фрейм: %2',
                                    ux_code_msg, fr_code_msg);
  end

  master_detail_where = '';
  master_field_count = 0;
  detail_field_count = 0;

  if (is_table = 1 and is_detail = 1) then
  begin
    for select res_str
    from uxp$split_str(:detail_field, ';')
    into :res_str
    do
    begin
      detail_field_count = detail_field_count + 1;

      -- Если поле из мастер-деталь связки не является параметром, добавляем в WHERE
      if (exists(select * from ux$frparam, ux$field
      where ufp_ufr_uuid = :fr_uuid and
            ufp_uf_uuid = uf_uuid and
            uf_field_name = :res_str and
            uf_is_param = 0)) then
      begin
        if (master_detail_where <> '') then
          master_detail_where = master_detail_where || ' and ' || enter_sym;
  
        if (need_alias = 1) then
          res_str = frame_alias || '.' || res_str;

        master_detail_where = master_detail_where || res_str || ' = :%' || detail_field_count;
      end
    end

    for select res_str
    from uxp$split_str(:master_field, ';')
    into :res_str
    do
    begin
      if (need_alias = 1) then
        res_str = frame_alias || '.' || res_str;

      master_field_count = master_field_count + 1;
      master_detail_where = replace(master_detail_where, '%'||master_field_count, res_str);
    end

    if (master_detail_where <> '') then
      master_detail_where = ' (' || master_detail_where || ') ';
  end

  for select uf_field_name, uf_select_name, ufp_editor, uf_edi_uuid, uf_type, uf_owner
  from ux$frparam, ux$field
  where ufp_ufr_uuid = :fr_uuid and
        ufp_uf_uuid = uf_uuid and
        uf_is_param = 0
  order by uf_position
  into field_name, uf_select_name, ufp_editor, edi_uuid, uf_type, uf_owner
  do
  begin
    edi_field_name = '';

    if (field_list <> '') then
      field_list = field_list || ', ' || enter_sym;

    if (uf_type = -1) then
    begin
      field_list = field_list || uf_select_name;
      Continue;
    end

    if (edi_uuid is not null) then
    begin
      select ufp_editor, uf_field_name
      from ux$frparam, ux$field
      where ufp_uuid = :edi_uuid and
            ufp_uf_uuid = uf_uuid
      into ufp_editor, edi_field_name;






    end

    if (ufp_editor in ('curredit', 'currfixed')) then
    begin
      if (not exists(select * from ux$frparam, ux$field
                     where ufp_ufr_uuid = :fr_uuid and
                           ufp_uf_uuid = uf_uuid and
                           uf_is_param = 0 and
                           uf_field_name = (:field_name || '$bond'))) then
      begin
        if (ufp_editor = 'currfixed') then
          exception e_default f_format('%1 - поле %3$bond отсутствует. Frame: %2, field: %3', ufp_editor, fr_code_msg, field_name);

        if (need_alias = 1) then
        begin
          if (uf_owner = fr_us_uuid) then
            field_list = field_list || frame_alias || '.';
          else
            field_list = field_list || main_alias || '.';
        end

        field_list = field_list || field_name;
        field_list = field_list || ', ' || enter_sym;
        field_list = field_list || 'cast(null as d_str_long) as ' || field_name || '$bond';
        Continue;
      end
    end

    if (ufp_editor in ('blob_bls', 'blob')) then
    begin
      if (need_alias = 1) then
      begin
        if (uf_owner = fr_us_uuid) then
          field_list = field_list || frame_alias || '.';
        else
          field_list = field_list || main_alias || '.';
      end

      field_list = field_list || field_name;
      field_list = field_list || ', ' || enter_sym;
      field_list = field_list || 'uxf$blob(' || field_name || ', 1) as ' || field_name || '$json';
      field_list = field_list || ', ' || enter_sym;
      field_list = field_list || 'uxf$blob(' || field_name || ', 0) as ' || field_name || '$text';

      if (ufp_editor = 'blob') then
      begin
        field_list = field_list || ', ' || enter_sym;
        field_list = field_list || 'cast(null as d_blob_binary) as ' || field_name || '$blob';
        field_list = field_list || ', ' || enter_sym;
        field_list = field_list || 'cast(null as d_str_255) as ' || field_name || '$file';
      end

      ufp_editor = 'blob_json';
    end else
    if (ufp_editor = 'blob_edoc') then
    begin
      if (need_alias = 1) then
      begin
        if (uf_owner = fr_us_uuid) then
          field_list = field_list || frame_alias || '.';
        else
          field_list = field_list || main_alias || '.';
      end

      field_list = field_list || field_name;
      field_list = field_list || ', ' || enter_sym;
      field_list = field_list || 'uxf$edocs(' || field_name || ', null) as ' || field_name || '$json';
      field_list = field_list || ', ' || enter_sym;
      field_list = field_list || 'cast(' || field_name || ' as d_str_code) as ' || field_name || '$text';

      ufp_editor = 'blob_json';
    end else
    if (ufp_editor = 'blob_json') then
    begin
      if (need_alias = 1) then
      begin
        if (uf_owner = fr_us_uuid) then
          field_list = field_list || frame_alias || '.';
        else
          field_list = field_list || main_alias || '.';
      end

      field_list = field_list || field_name || ' as ' || field_name || '$json';
      field_list = field_list || ', ' || enter_sym;
      field_list = field_list || 'iif(coalesce(' || field_name || ', '''') <> '''', ''Файл(ы)'', ''(Пусто)'') as ' || field_name || '$text';
    end else
    begin
      if (need_alias = 1 and uf_select_name = field_name) then
      begin
        if (uf_owner = fr_us_uuid) then
          field_list = field_list || frame_alias || '.';
        else
          field_list = field_list || main_alias || '.';
      end

      field_list = field_list || uf_select_name;
    end

    ua_uxa_uuid = null;
    ua_usr_text_field = '';
    buttonedit_sql = '';

    if (ufp_editor in ('buttonedit', 'dialogedit')) then
    begin
      if (edi_uuid is null) then
      begin
        select ua_uxa_uuid, ua_usr_text_field
        from ux$action
        where ua_owner = :fr_uuid and
              ua_usr_id_field = :field_name
        into ua_uxa_uuid, ua_usr_text_field;
      end else
      begin
        select ua_uxa_uuid
        from ux$action
        where ua_owner = :sys_uuid and
              ua_usr_id_field = :edi_field_name
        into ua_uxa_uuid;
      end

      if (row_count = 0) then
        exception e_default f_format('%1 action not found. Frame: %2, field: %3', ufp_editor, fr_code_msg, field_name);

      if (ua_usr_text_field = '' and not(ua_usr_text_field containing '$')) then
      begin
        select ux_us_uuid
        from ux$alias, ux$code
        where uxa_uuid = :ua_uxa_uuid and
              uxa_owner = ux_uuid
        into tmp_us_uuid;

        -- если поле нужно формировать динамически, получаем текст SQL
        select keyname_sql, error_text
        from uxp$uxsource_sql_prepare(:tmp_us_uuid, :field_name, uuid_to_char(:ua_uxa_uuid))
        into :buttonedit_sql, error_text;

        if (buttonedit_sql <> '') then
        begin
          alias_letter = '';

          if (need_alias = 1) then
          begin
            if (uf_owner = fr_us_uuid) then
              alias_letter = frame_alias || '.';
            else
              alias_letter = main_alias || '.';
          end

          buttonedit_sql = replace(buttonedit_sql, ':id', alias_letter || field_name);
          buttonedit_sql = ' (' || buttonedit_sql || ') as ' || field_name || '$text';
  
          field_list = field_list || ', ' || enter_sym || buttonedit_sql;
        end else
          exception e_default f_format('%1 SQL not found, fr_code: %2, field: %3, error: %4',
                                       ufp_editor, fr_code_msg, field_name, error_text);
      end else
      begin
        field_list = field_list || ', ' || enter_sym;

        if (need_alias = 1 and uf_select_name = field_name) then
        begin
          if (uf_owner = fr_us_uuid) then
            field_list = field_list || frame_alias || '.';
          else
            field_list = field_list || main_alias || '.';
        end

        if (ua_usr_text_field containing '$') then
          field_list = field_list || field_name || ' as ' || field_name || '$text';
        else
          field_list = field_list || ua_usr_text_field || ' as ' || field_name || '$text';
      end
    end
  end

  join_sql = '';
  if (field_list <> '') then
  begin
    sql_text = 'select [$$$]' || enter_sym || field_list || enter_sym || 'from ' || fr_source;

    if (need_alias = 1) then
    begin
      join_sql = frame_alias || enter_sym;
      join_sql = join_sql || '  join ' || main_table || ' ' || main_alias || ' on (';
      join_sql = join_sql || frame_alias || '.' || frame_keyfields || ' = ' || main_alias || '.' || main_keyfields || ')';
    end else
    if (ufr_add_table_fields <> '') then
    begin
      join_sql = enter_sym;
      join_sql = join_sql || '  join ' || main_table || ' on (';
      join_sql = join_sql || frame_keyfields || ' = ' || main_keyfields || ')';
    end
  end
  else
  begin
    if (ux_kind = 'uxcommand') then
    begin
      sql_text = 'execute procedure ' || fr_source;
      sql_is_exec = 1;
    end
    else
      exception e_default f_format('Select SQL no fields, ux_code: %1, fr_code: %2', ux_code_msg, fr_code_msg);
  end

  field_list = '';

  f_temp = 'coalesce(uxf$set_blob(:%1$blob, :%1$file), :%1)';

  if (is_table = 0) then
  begin
    for select uf_field_name, ufp_editor, uf_domain, uf_select_name
    from ux$frparam, ux$field
    where ufp_ufr_uuid = :fr_uuid and
          ufp_uf_uuid = uf_uuid and
          uf_is_param = 1 and
          uf_type = 0 and
--          uf_type <> -1 and
          ufp_parent is null
    order by uf_position
    into field_name, ufp_editor, uf_domain, uf_select_name
    do
    begin
      md_linked = 0;
      if (is_detail = 1) then
      begin
        detail_field_count = 0;
        param_index = 0;
        for select res_str
        from uxp$split_str(:detail_field, ';')
        into :res_str
        do
        begin
          detail_field_count = detail_field_count + 1;
          if (field_name = res_str) then
          begin
            param_index = detail_field_count;
            break;
          end
        end

        if (param_index > 0) then
        begin
          master_field_count = 0;

          for select res_str
          from uxp$split_str(:master_field, ';')
          into :res_str
          do
          begin
            master_field_count = master_field_count + 1;
            if (master_field_count = param_index) then
            begin
              replaces_params = replaces_params || f_format('cast(:%1 as %2) as %3,', res_str, uf_domain, field_name);

              field_name = res_str;

              md_linked = 1;
              break;
            end
          end

          if (md_linked = 0) then
            exception e_default f_format('Master field param not found, ux_code: %1, fr_code: %1',
                                          ux_code_msg, fr_code_msg);
        end
      end

      if (field_list <> '') then
        field_list = field_list || ', ';

      if (ufp_editor = 'blob') then
        field_list = field_list || f_format(f_temp, field_name);
      else
        field_list = field_list || ':' || field_name;
    end

    if (field_list <> '') then
      sql_text = sql_text || '(' || field_list || ')';

    if (join_sql <> '') then
      sql_text = sql_text || ' ' || join_sql;
    else
    if (alias_letter <> '') then
      sql_text = sql_text || ' ' || alias_letter;

    where_str = ' where ';
    if (sql_is_exec = 0 and ufr_sql_where <> '') then
    begin
      sql_text = sql_text || enter_sym || where_str || ' (' || ufr_sql_where || ')';
      where_str = ' and ';
    end
  end else
  begin
    if (alias_letter <> '') then
      sql_text = sql_text || ' ' || alias_letter;

    if (detail_field <> '') then
      detail_field = ';' || replace(detail_field, ' ', '') || ';';

    if (ufr_is_grid = 0) then
    begin
      for select uf_field_name
      from ux$frparam, ux$field
      where ufp_ufr_uuid = :fr_uuid and
            ufp_uf_uuid = uf_uuid and
            uf_pk_field <> 0 and
            uf_type <> -1 and
            ufp_parent is null
      order by uf_position
      into field_name
      do
      begin
        if (position(';' || field_name || ';', detail_field) > 0) then
          Continue;
  
        if (field_list <> '') then
          field_list = field_list || ' and ' || enter_sym;
  
        field_list = field_list || field_name || ' = :' || field_name;
      end
    end

    where_str = ' where ';
    if (sql_is_exec = 0 and ufr_sql_where <> '') then
    begin
      sql_text = sql_text || enter_sym || where_str || ' (' || ufr_sql_where || ')';
      where_str = ' and ';
    end

    if (field_list <> '') then
    begin
      sql_text = sql_text || enter_sym || where_str || ' (' || field_list || ')';
      where_str = ' and ';
    end

    if (master_detail_where <> '') then
      sql_text = sql_text || enter_sym || where_str || master_detail_where;
  end

  if (sql_is_exec = 0 and sql_text <> '' and ufr_sql_orderby <> '') then
    sql_text = sql_text || enter_sym || ' order by ' || ufr_sql_orderby;

  -- При ufr_is_grid = 0 в интерфейсе будем проверять кол-во строк
  -- если строк более одной, выдаем ошибку настройки

  max_rows = 0;
  if (sql_text <> '' and sql_is_exec = 0) then
  begin
    if (ufr_is_grid = 0) then
      max_rows = 2;
    else
    begin
      if (ufr_is_grid = 1 and is_table = 1) then
        max_rows = 500; -- если источник просто таблица, не даем более 500 строк

      if (coalesce(ufr_sql_where, '') <> '') then
        max_rows = 10000;
    end
  end

  if (max_rows <> 0) then
    sql_text = sql_text || ' rows ' || max_rows;

  sql_text = replace(sql_text, '[$$$]', replaces_params);

  suspend;
end
