CREATE PROCEDURE HP_FSROBOT_DELINQ_PUSH_AND_SMS
AS
declare variable K_ID D_ID;
declare variable K_KONTRACT_NO D_STR_20;
declare variable MCL_ID D_ID;
declare variable CELL D_STR_100;
declare variable PUSH_MESSAGE D_STR_LONG;
declare variable V_ERR D_STR_255;
declare variable V_RES D_STR_10;
declare variable DELAY_DAYS D_INTEGER;
begin
    if (f_get_db_id() <> 39801) then
    exit;

    --Реализовано для КЗ
    for select k.k_id,
               k.k_kontract_no,
               current_date - oks.oks_delinq_date
    from oksummary oks
    inner join kontract k on oks.oks_from_k_id = k.k_id
    where oks.oks_close_state = 0
    and current_date between oks.oks_date_begin and oks.oks_date_end
    and current_date - oks.oks_delinq_date between 5 and 15
    into :k_id,
         :k_kontract_no,
         :delay_days
    do
    begin
        for select mcl.mcl_id,
                   cc.cc_cell_no
        from knclient kc
        inner join clients c on kc.kc_from_c_id = c.c_id
        inner join mc_login mcl on mcl.mcl_from_c_id = kc.kc_from_c_id
        inner join contacts cc on c.c_id = cc.cc_from_c_id
        where kc.kc_from_k_id = :k_id
        and kc.kc_amount_approved > 0
        and mcl.mcl_is_delete = 0
        and cc.cc_is_old_contact = 0

        into :mcl_id,
             :cell

        do
        begin
            push_message = f_format(
                        'Уважаемый клиент, у вас имеется просрочка в количестве %2 дней по контракту №%1. Просьба срочно оплатить просроченный платеж через доступные платёжные системы',
                        k_kontract_no,
                        delay_days
                        );

            if (delay_days = 5) then                --Отправка PUSH и SMS уведомлений на 5 день просрочки
            begin
                execute procedure mc_send_push('mc_login', :mcl_id, 'Уведомление о просрочке', :push_message, :delay_days);
                execute procedure crm_send_sms(:cell, :push_message)
                returning_values :v_err, :v_res;
            end
            if (delay_days between 6 and 15) then   --Отправка PUSH уведомлений с 6 по 15 день просрочки
            begin
                execute procedure mc_send_push('mc_login', :mcl_id, 'Уведомление о просрочке', :push_message, :delay_days);

            end
        end
    end
end
