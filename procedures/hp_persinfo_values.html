CREATE PROCEDURE HP_PERSINFO_VALUES(
    C_ID D_ID)
RETURNS (
    PIF_SOURCE D_STR_CODE,
    PIF_FIELD_NAME D_STR_CODE,
    PIF_CAPTION D_STR_100,
    RES_LOG_COUNT D_INTEGER,
    RES_SIGNED_VALUE D_STR_255,
    RES_ACTUAL_VALUE D_STR_255,
    RES_IS_CHANGED D_BOOL,
    RES_ACTUAL_DATETIME D_DATETIME,
    RES_DATA_CHANGED D_BOOL)
AS
declare variable pin_id d_id;
declare variable pif_uuid d_uuid;
declare variable pda_id d_id;
begin
  for select pin_id
  from persinfo
  where pin_object_kind = 0 and
        pin_object_id = :c_id
  into pin_id
  do
  begin
    for select pif_uuid, pif_source, pif_field_name, pif_caption
    from ux$persfield
    where pif_inactive = 0
    order by pif_position
    into pif_uuid, pif_source, pif_field_name, pif_caption
    do
    begin
      for select pda_id, iif(pda_signed_id is distinct from pda_value_id, 1, 0),
                 pif$id_val(p.pda_signed_id),
                 pif$id_val(p.pda_value_id),
                 pda_data_changed
      from persdata p
      where pda_from_pin_id = :pin_id and
            pda_from_pif_uuid = :pif_uuid
      into pda_id, res_is_changed, res_signed_value, res_actual_value, res_data_changed
      do
      begin
        res_log_count = 0;
        res_actual_datetime = null;

        select count(*), max(pva_datetime)
        from persvalue
        where pva_from_pda_id = :pda_id
        into res_log_count, res_actual_datetime;

        suspend;
      end
    end
  end
end
