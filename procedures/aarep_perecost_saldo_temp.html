CREATE PROCEDURE AAREP_PERECOST_SALDO_TEMP(
    REGION_ID INTEGER,
    REPORT_DATE DATE)
RETURNS (
    OBREP_ID VARCHAR(20),
    OBREP_CURRENCY VARCHAR(3),
    OBREP_SALDO DECIMAL(18,4),
    OBREP_SALDO_EKV DECIMAL(18,4),
    OBREP_REGION_ID INTEGER,
    OBREP_CLIENT DECIMAL(18,0),
    OBREP_CLIENT_NAME VARCHAR(100),
    AC_AMOUNT DECIMAL(18,4))
AS
declare variable s1 decimal(18,4);
declare variable s2 decimal(18,4);
declare variable with_kontragent integer;
declare variable report_currency varchar(3);
begin
  with_kontragent = 0;
  select cm_recost_with_kontragent, cm_default_currency
  from company
  into :with_kontragent, :report_currency;
  ac_amount = 0;
  obrep_client_name = '';
  obrep_client = null;
  obrep_saldo = 0;
  obrep_saldo_ekv = 0;

  for select ac_region_id, curr_code, ac_no, sum(amount_diff)
      from aarep_kontract_active_curr_temp(:region_id, :report_date)
      group by ac_region_id, curr_code, ac_no
      into :obrep_region_id, :obrep_currency, :obrep_id, :ac_amount
  do
  begin
    if (ac_amount is null) then
      ac_amount = 0;
    if (ac_amount <> 0) then
      suspend;
  end
  ac_amount = 0;
  if ((with_kontragent is null) or (with_kontragent = 0)) then
  begin
    obrep_client_name = '';
    obrep_client = null;

    for select al_id, al_currency, al_from_reg_id
        from acc_list, region_list(:region_id, 1, -1, 1), balgroup
        where al_from_bg_id = bg_id and
              bg_is_recost = 1 and
              al_from_reg_id = id_list and
              al_currency <> :report_currency
        into :obrep_id, :obrep_currency, :obrep_region_id
    do
    begin
      obrep_saldo_ekv = f_al_bal(obrep_id, report_date);
      obrep_saldo = f_al_bal(obrep_id, report_date, 'val');

      obrep_saldo_ekv = cast(obrep_saldo_ekv as decimal(18,2));
      if ((obrep_saldo_ekv <> 0) or (obrep_saldo <> 0)) then
        suspend;
    end
  end
  else
  begin
    for select distinct ob_client_dt, ob_account_dt, al_currency, ol_from_reg_id
        from operbook, operlist, acc_list, balgroup, region_list(:region_id, 1, -1, 1)
        where ob_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              ob_currency <> :report_currency and
              ol_date <= :report_date and
              (ol_from_reg_id = id_list) and
              al_id = ob_account_dt and
              bg_id = al_from_bg_id and
              bg_is_recost = 1
        union
        select distinct ob_client_kt, ob_account_kt, al_currency, ol_from_reg_id
        from operbook, operlist, acc_list, balgroup, region_list(:region_id, 1, -1, 1)
        where ob_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              ob_currency <> :report_currency and
              ol_date <= :report_date and
              (ol_from_reg_id = id_list) and
              al_id = ob_account_kt and
              bg_id = al_from_bg_id and
              bg_is_recost = 1
        into :obrep_client, :obrep_id, :obrep_currency, :obrep_region_id
    do
    begin
      obrep_client_name = '';
      if (obrep_client is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :obrep_client
        into :obrep_client_name;
      end
      s1 = 0;
      s2 = 0;
      if (obrep_client is null) then
      begin
        select sum(ob_summa), sum(ob_summa_val)
        from operbook, operlist
        where ob_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              ob_account_dt = :obrep_id and
              (ob_client_dt is null) and
              ol_date <= :report_date and
              ol_from_reg_id = :obrep_region_id
        into :s1, :s2;
        if (s1 is null) then
          s1 = 0;
        if (s2 is null) then
          s2 = 0;
        obrep_saldo = s2;
        obrep_saldo_ekv = s1;

        select sum(ob_summa), sum(ob_summa_val)
        from operbook, operlist
        where ob_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              ob_account_kt = :obrep_id and
              (ob_client_kt is null) and
              ol_date <= :report_date and
              ol_from_reg_id = :obrep_region_id
        into :s1, :s2;
        if (s1 is null) then
          s1 = 0;
        if (s2 is null) then
          s2 = 0;
        obrep_saldo = obrep_saldo - s2;
        obrep_saldo_ekv = obrep_saldo_ekv - s1;
        obrep_saldo_ekv = cast(obrep_saldo_ekv as decimal(18,2));
        if ((obrep_saldo_ekv <> 0) or (obrep_saldo <> 0)) then
          suspend;
      end
      else
      begin
        select sum(ob_summa), sum(ob_summa_val)
        from operbook, operlist
        where ob_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              ob_account_dt = :obrep_id and
              ob_client_dt = :obrep_client and
              ol_date <= :report_date and
              ol_from_reg_id = :obrep_region_id
        into :s1, :s2;
        if (s1 is null) then
          s1 = 0;
        if (s2 is null) then
          s2 = 0;
        obrep_saldo = s2;
        obrep_saldo_ekv = s1;

        select sum(ob_summa), sum(ob_summa_val)
        from operbook, operlist
        where ob_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              ob_account_kt = :obrep_id and
              ob_client_kt = :obrep_client and
              ol_date <= :report_date and
              ol_from_reg_id = :obrep_region_id
        into :s1, :s2;
        if (s1 is null) then
          s1 = 0;
        if (s2 is null) then
          s2 = 0;
        obrep_saldo = obrep_saldo - s2;
        obrep_saldo_ekv = obrep_saldo_ekv - s1;
        obrep_saldo_ekv = cast(obrep_saldo_ekv as decimal(18,2));
        if ((obrep_saldo_ekv <> 0) or (obrep_saldo <> 0)) then
          suspend;
      end
    end
  end
end
