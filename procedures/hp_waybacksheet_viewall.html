CREATE PROCEDURE HP_WAYBACKSHEET_VIEWALL(
    DTBEGIN D_DATE,
    DTEND D_DATE,
    UN_ID D_ID = null)
RETURNS (
    WBS_ID D_INTEGER,
    WBS_SIGNER1 D_ID,
    SIGNER1_NAME D_STR_100,
    WBS_SIGNER1_DATE D_DATETIME,
    WBS_SIGNER2 D_ID,
    SIGNER2_NAME D_STR_100,
    WBS_SIGNER2_DATE D_DATETIME,
    WBS_SIGNER3 D_ID,
    SIGNER3_NAME D_STR_100,
    WBS_SIGNER3_DATE D_DATETIME,
    WBS_SIGNER4 D_ID,
    SIGNER4_NAME D_STR_100,
    WBS_SIGNER4_DATE D_DATETIME,
    WBS_SIGNER5 D_ID,
    SIGNER5_NAME D_STR_100,
    WBS_SIGNER5_DATE D_DATETIME,
    WBS_SIGNER6 D_ID,
    SIGNER6_NAME D_STR_100,
    WBS_SIGNER6_DATE D_DATETIME,
    WBS_SIGNER7 D_ID,
    SIGNER7_NAME D_STR_100,
    WBS_SIGNER7_DATE D_DATETIME,
    WBS_SIGNER8 D_ID,
    SIGNER8_NAME D_STR_100,
    WBS_SIGNER8_DATE D_DATETIME,
    WBS_COUNT_SIGNS D_INTEGER,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    HO_ID D_ID,
    HO_DATE D_DATE,
    HO_ORDER_NO D_STR_20,
    HO_FROM_REG_ID D_INTEGER,
    HO_STATUS D_INTEGER,
    OFFICE_NAME D_STR_CODE,
    JL_NAME D_STR_100,
    LAST_WORK_DATE D_DATE,
    IS_SIGN D_SMALLINT)
AS
declare variable lst_reg_id d_str_8100;
begin
  if (un_id > 0) then
    lst_reg_id = (select list(r_list, ' ') from user_reg_list(:un_id));

  for
    select
      wbs_id,
      wbs_signer1,
      wbs_signer1_date,
      wbs_signer2,
      wbs_signer2_date,
      wbs_signer3,
      wbs_signer3_date,
      wbs_signer4,
      wbs_signer4_date,
      wbs_signer5,
      wbs_signer5_date,
      wbs_signer6,
      wbs_signer6_date,
      wbs_signer7,
      wbs_signer7_date,
      wbs_signer8,
      wbs_signer8_date,
      wbs_count_signs,
      ho_id,
      ho_date,
      ho_order_no,
      ho_from_reg_id,
      ho_status
    from waybacksheet
      inner join hr_orders
        on ho_id = wbs_ho_id
          and ho_status = 1
    where wbs_ho_date between :dtbegin and :dtend
    order by ho_date desc
  into
    :wbs_id,
    :wbs_signer1,
    :wbs_signer1_date,
    :wbs_signer2,
    :wbs_signer2_date,
    :wbs_signer3,
    :wbs_signer3_date,
    :wbs_signer4,
    :wbs_signer4_date,
    :wbs_signer5,
    :wbs_signer5_date,
    :wbs_signer6,
    :wbs_signer6_date,
    :wbs_signer7,
    :wbs_signer7_date,
    :wbs_signer8,
    :wbs_signer8_date,
    :wbs_count_signs,
    :ho_id,
    :ho_date,
    :ho_order_no,
    :ho_from_reg_id,
    :ho_status
  do
  begin
    if (un_id is null or lst_reg_id containing ho_from_reg_id) then
    begin
      select first 1
        c_id,
        c_name_fio,
        jl_name,
        ps_date_begin - 1
      from hr_pstatus
        inner join joblist
          on jl_id = ps_from_jl_id
        inner join clients
          on c_id = ps_from_c_id
      where ps_from_ho_id = :ho_id
      order by ps_from_ho_id, ps_id desc
      into
        :c_id,
        :c_name_fio,
        :jl_name,
        :last_work_date;

      signer1_name =
        (select un_fio from username where un_id = :wbs_signer1);
      signer2_name =
        (select un_fio from username where un_id = :wbs_signer2);
      signer3_name =
        (select un_fio from username where un_id = :wbs_signer3);
      signer4_name =
        (select un_fio from username where un_id = :wbs_signer4);
      signer5_name =
        (select un_fio from username where un_id = :wbs_signer5);
      signer6_name =
        (select un_fio from username where un_id = :wbs_signer6);
      signer7_name =
        (select un_fio from username where un_id = :wbs_signer7);
      signer8_name =
        (select un_fio from username where un_id = :wbs_signer8);

      office_name =
        (select reg_name from region where reg_unicode = :ho_from_reg_id);

      if (:ho_date < cast('27.04.2026' as d_date)) then   -- Подписант номер 8 стал активным начиная с '27.04.2026'
        is_sign = iif(wbs_count_signs >= 7, 1, 0);
      else
        is_sign = iif(wbs_count_signs = 8, 1, 0);

      suspend;
    end
  end
end
