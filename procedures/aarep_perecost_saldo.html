CREATE PROCEDURE AAREP_PERECOST_SALDO(
    REGION_ID INTEGER,
    REPORT_DATE DATE)
RETURNS (
    OBREP_ID VARCHAR(20),
    OBREP_CURRENCY VARCHAR(3),
    OBREP_SALDO DECIMAL(18,4),
    OBREP_SALDO_EKV DECIMAL(18,4),
    OBREP_REGION_ID INTEGER,
    OBREP_CLIENT DECIMAL(18,0),
    OBREP_CLIENT_NAME VARCHAR(100),
    OBREP_DIFF_AMOUNT DECIMAL(18,4))
AS
begin
  for select obrep_id, obrep_currency, sum(obrep_saldo), sum(obrep_saldo_ekv), obrep_region_id,
             obrep_client, obrep_client_name, sum(ac_amount)
      from aarep_perecost_saldo_temp(:region_id, :report_date)
      group by obrep_id, obrep_currency, obrep_region_id, obrep_client, obrep_client_name
      into :obrep_id, :obrep_currency, :obrep_saldo, :obrep_saldo_ekv, :obrep_region_id, :obrep_client,
           :obrep_client_name, :obrep_diff_amount
  do
  begin
    if (obrep_diff_amount is null) then
      obrep_diff_amount = 0;
    suspend;
  end
end
