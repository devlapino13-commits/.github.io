CREATE PROCEDURE UXP$UXGRID_LIST(
    UFM_UUID D_UUID,
    UFM_UUID_CODE D_STR_CODE = null)
RETURNS (
    UGR_UUID D_UUID,
    UGR_UFR_UUID D_UUID,
    UGR_NAME D_STR_100,
    UGR_MEMO D_STR_255,
    UGR_IS_PRIVATE D_BOOL,
    UGR_DEFAULT D_BOOL,
    UGR_HIDE_ON_GROUP D_BOOL,
    UGR_GROUP_PANEL D_BOOL,
    UGR_FIND_PANEL D_SMALLINT,
    UGR_HIDE_SUMMARY D_BOOL,
    UGR_INACTIVE D_BOOL,
    UGR_UUID_CODE D_STR_CODE,
    UGR_AUTOWIDTH D_SMALLINT,
    UGR_HEADER_AUTOHEIGHT D_BOOL,
    UGR_SHOW_INFO_PANEL D_BOOL,
    UGR_HIDE_INPLACE_EDITOR D_BOOL,
    UGR_SHOW_EXCEL_IMPORT D_BOOL,
    UGR_ALLOW_APPEND D_BOOL)
AS
declare variable ux_uuid d_uuid;
begin
  if (coalesce(ufm_uuid, '') = '') then
    ufm_uuid = uxf$uniq(ufm_uuid_code);

  select ufr_uuid
  from ux$frame
  where ufr_owner = :ufm_uuid and
        ufr_is_mainframe = 1
  into ux_uuid;

  if (row_count = 0) then
    exit;

  for select ugr_uuid,
             ugr_ufr_uuid,
             ugr_name,
             ugr_memo,
             ugr_is_private,
             ugr_default,
             ugr_hide_on_group,
             ugr_group_panel,
             ugr_find_panel,
             ugr_hide_summary,
             ugr_inactive,
             ugr_autowidth,
             ugr_header_autoheight,
             ugr_show_info_panel,
             ugr_hide_inplace_editor,
             ugr_show_excel_import,
             ugr_allow_append
      from ux$grid
      where ugr_ufr_uuid = :ux_uuid
      into :ugr_uuid,
           :ugr_ufr_uuid,
           :ugr_name,
           :ugr_memo,
           :ugr_is_private,
           :ugr_default,
           :ugr_hide_on_group,
           :ugr_group_panel,
           :ugr_find_panel,
           :ugr_hide_summary,
           :ugr_inactive,
           :ugr_autowidth,
           :ugr_header_autoheight,
           :ugr_show_info_panel,
           :ugr_hide_inplace_editor,
           :ugr_show_excel_import,
           :ugr_allow_append
  do
  begin
    ugr_uuid_code = uuid_to_char(ugr_uuid);
    suspend;
  end
end
