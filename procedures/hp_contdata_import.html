CREATE PROCEDURE HP_CONTDATA_IMPORT(
    C_ID D_ID,
    MAX_LINE D_INTEGER = 0)
AS
declare variable opr_id d_id;
declare variable ovs_value d_str_100;
declare variable un_id d_id;
declare variable un_datetime d_datetime;
declare variable prior_value d_str_100;
declare variable cda_id d_id;
declare variable rd_id d_integer;
declare variable rd_code d_str_100;
declare variable cont_kind d_str_code;
declare variable cont_kind_id d_integer;
declare variable cda_status d_smallint;
declare variable cv_password d_str_100;
declare variable cv_status d_smallint;
declare variable pin d_str_code;
declare variable cda_is_main d_smallint;
declare variable sign_kind d_str_code;
declare variable eds_id d_id;
declare variable msg_id d_id;
declare variable clt_id d_integer;
begin
--  for select cda_id
--  from contdata
--  where cda_from_c_id = :c_id
--  into cda_id
--  do
--  begin
--    update edoc_msg
--    set edm_from_cda_id = null
--    where edm_from_cda_id = :cda_id;
--  end

  delete from contdata
  where cda_from_c_id = :c_id;

  for select rd_id, rd_code
      from refdata
      where rd_main_code = 'ContactInfo' and
            rd_main_code <> rd_code
      into rd_id, rd_code
  do
  begin
    opr_id = null;
    select opr_id
    from objectprops
    where opr_object_kind = 0 and
          opr_object_id = :c_id and
          opr_subject_kind = -1 and
          opr_subject_id = :rd_id
    into opr_id;

    if (opr_id is null) then
      continue;

    cont_kind = replace(rd_code, 'ContactInfo.', '');
    if (cont_kind in ('cell_main', 'cell_add')) then
      cont_kind = 'cell';

    cont_kind_id = f_enum('cont_kind', cont_kind, 'id');

    cda_id = 0;
    cda_status = -1;
    prior_value = '***';

    if (rd_code = 'ContactInfo.Cell_Main')
    then cda_is_main = 1;
    else cda_is_main = 0;

    for select ovs_value, ovs_from_un_id, ovs_datetime, ovs_add_value
        from objectvalues
        where ovs_from_opr_id = :opr_id and
              ovs_is_active = 1 and
              (:max_line = 0 or ovs_id <= :max_line)
        order by ovs_date_begin, ovs_id
        into :ovs_value, :un_id, :un_datetime, :cda_status
    do
    begin
      if (cda_status = 2) then
        cda_status = 1;

      if (ovs_value is null) then ovs_value = '';

      if (un_id is null) then
        un_id = 1000000000000;

      select cda_id
      from hp_contdata_update(:c_id, :cont_kind_id, :ovs_value, :cda_is_main, :un_id, :un_datetime)
      into :cda_id;

      if (cda_status = 1 and cda_id <> 0) then
        execute procedure hp_contveri_set_ok(:cda_id, :un_id, :un_datetime);
    end

    prior_value = '***';

    for select cv_contact_value, cv_password, cv_status, coalesce(cv_verify_un_id, cv_from_un_id),
               coalesce(cv_verify_datetime, cv_datetime)
        from contverify
        where cv_object_kind = 0 and
              cv_object_id = :c_id and
              cv_from_rd_id = :rd_id and
              cv_status >= 0
        order by 1, 5, 3
        into :ovs_value, cv_password, cv_status, un_id, un_datetime
    do
    begin
      if (ovs_value is null) then
        ovs_value = '';

      if (ovs_value <> prior_value) then
        cda_id = f_contdata_id(c_id, cont_kind_id, ovs_value, cda_is_main, current_date, 'null', :un_id);

      if (cda_id is not null) then
      begin
        sign_kind = '';
        sign_kind = case cv_status
          when 0 then 'none'
          when 1 then 'finstart'
          when 2 then 'paper'
          when 3 then 'sms'
        end;

        if (cv_status <> 0) then cv_status = 1;

        pin = cv_password;

        insert into contveri (cvr_from_cda_id, cvr_hash_pin, cvr_status, cvr_sign_kind, cvr_datetime)
        values (:cda_id, :pin, :cv_status, f_enum('edoc_sign_kind', :sign_kind, 'id'), :un_datetime);

        prior_value = ovs_value;
      end
    end
  end

  for select edm_from_eds_id, edm_from_msg_id, edm_from_clt_id
  from edoc_msg, edoc_sign
  where edm_from_eds_id = eds_id and
        eds_from_c_id = :c_id and
        coalesce(edm_from_clt_id, 0) <> 0 and
        edm_from_cda_id is null
  into :eds_id, :msg_id, :clt_id
  do
  begin
    cda_id = null;
    select cda_id
    from contdata
    where cda_from_c_id = :c_id and
          cda_from_clt_id = :clt_id and
          cda_cont_kind = 0
    order by cda_status desc, cda_id desc
    rows 1
    into cda_id;

    if (cda_id is not null) then
    begin
      update edoc_msg
      set edm_from_cda_id = :cda_id
      where edm_from_eds_id = :eds_id and
            edm_from_msg_id = :msg_id and
            edm_from_clt_id = :clt_id;
    end else
      exception e_default 'There are null value in eDoc_Msg';
  end

  max_line = 0;
  select count(*)
  from hp_contdata_one_check(:c_id)
  into max_line;
end
