CREATE PROCEDURE HP_DO_KN_OBSUMMARY_CHECK_SALDO
RETURNS (
    ACC_ID D_ID,
    BAL_SUM D_CURRENCY,
    BAL_VAL D_CURRENCY,
    BAL_TAX D_CURRENCY,
    ACTUAL_SUM D_CURRENCY,
    ACTUAL_VAL D_CURRENCY,
    ACTUAL_TAX D_CURRENCY)
AS
begin
  for select acc_id
  from ref_account
  into :acc_id
  do
  begin
    actual_sum = 0;
    actual_val = 0;
    actual_tax = 0;

    bal_sum = 0;
    bal_val = 0;
    bal_tax = 0;

    select sum(kobs_dt_sum - kobs_kt_sum),
           sum(kobs_dt_val - kobs_kt_val),
           sum(kobs_dt_tax - kobs_kt_tax)
    from kn_obsummary
    where kobs_acc_id = :acc_id
    into actual_sum, actual_val, actual_tax;
    if (actual_sum is null) then actual_sum = 0;
    if (actual_val is null) then actual_val = 0;
    if (actual_tax is null) then actual_tax = 0;

    select kobs_bal_sum, kobs_bal_val, kobs_bal_tax
    from kn_obsummary
    where kobs_acc_id = :acc_id
    order by kobs_acc_id desc, kobs_date desc
    rows 1
    into bal_sum, bal_val, bal_tax;

    if (actual_sum <> bal_sum or
        actual_val <> bal_val or
        actual_tax <> bal_tax)
    then
      suspend;
  end
end
