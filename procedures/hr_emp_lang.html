CREATE PROCEDURE HR_EMP_LANG(
    ORDER_ID DECIMAL(18,0))
RETURNS (
    CL_NUMBER D_INTEGER,
    RD_NAME D_STR_255,
    CL_MARK_READ D_SMALLINT,
    CL_MARK_WRITE D_SMALLINT,
    CL_MARK_SPEAK D_SMALLINT,
    CL_MEMO D_STR_100)
AS
begin
  for
    select
        cl.cl_number,
        ddr.rd_name,
        cl.cl_mark_read,
        cl.cl_mark_write,
        cl.cl_mark_speak,
        cl.cl_memo
    from hr_pstatus ps
    inner join clanguage cl on cl.cl_from_c_id = ps.ps_from_c_id
    inner join refdata ddr  on cl.cl_lang_no = ddr.rd_id
    where ps.ps_from_ho_id = :order_id
    into
         :cl_number,
         :rd_name,
         :cl_mark_read,
         :cl_mark_write,
         :cl_mark_speak,
         :cl_memo
  do
    suspend;
end
