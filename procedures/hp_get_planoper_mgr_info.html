CREATE PROCEDURE HP_GET_PLANOPER_MGR_INFO(
    REG_ID D_INTEGER,
    C_ID D_ID,
    RD_CODE D_STR_CODE,
    REP_DATE D_DATE,
    PLAN_PERIOD D_INTEGER)
RETURNS (
    PO_AMOUNT D_CURRENCY)
AS
begin
  -- Показывает план по менеджерам офиса.
  -- Если в офисе один менеджер, показывает план офиса

 if (coalesce(c_id, 0) = 0) then
    exit;

  po_amount = 0;

  if (exists(select * from loanmgrlist
  where lml_from_reg_id = :reg_id and
        lml_from_c_id = :c_id and
        lml_date_begin <= :rep_date and
        lml_date_end >= :rep_date)) then
  begin
    -- Если C_ID является действующим менеджером офиса, берем только его показатели
    select po_amount
    from hp_get_planoper_info(:reg_id, :c_id, :rd_code, :rep_date, :plan_period, 1)
    into :po_amount;
  end else
  begin
    -- Иначе проверяем, нет ли других менеджеров в офисе.
    -- Если нет, значит это офис с одним менеджером, берем показатели офиса
    if (not exists(select * from loanmgrlist
    where lml_from_reg_id = :reg_id and
        lml_from_c_id is not null and
        lml_date_begin <= :rep_date and
        lml_date_end >= :rep_date)) then
    begin
      select po_amount
      from hp_get_planoper_info(:reg_id, 0, :rd_code, :rep_date, :plan_period, 0)
      into :po_amount;
    end
  end

  suspend;
end
