CREATE PROCEDURE HP_COM_LIST_REFDATA_COMBO(
    RD_MAIN_CODE D_STR_CODE)
RETURNS (
    ID D_INTEGER,
    NAME D_STR_255)
AS
begin
  -- Возвращает список выпадающих значений для заполнения параметров запроса на КК

  if (coalesce(rd_main_code, '') = '') then
    exception e_default 'Empty refdata main_code';

  id = 0;
  name = 'Не указан';
  suspend;

  for select rd_id, rd_name
  from refdata
  where rd_main_code = :rd_main_code and
        rd_code <> rd_main_code and
        rd_is_active = 1
  order by rd_orderno, rd_id
  into id, name
  do
    suspend;
end
