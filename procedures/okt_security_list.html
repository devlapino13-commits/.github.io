CREATE PROCEDURE OKT_SECURITY_LIST(
    FOR_ANALIZ D_SMALLINT = 0,
    IGNORE_ALLOWED_CALL_TIME D_BOOL = 0)
RETURNS (
    PANT_ID ID_TYPE,
    CELL_NO D_STR_20,
    PROJECT STR_100,
    TIMEZONE D_INTEGER,
    PERSON_TYPE STR_100,
    PERSON_ID ID_TYPE,
    UN_ID ID_TYPE,
    LANGUAGE STR_100,
    URL D_STR_2048,
    DELINQ_DAYS D_INTEGER,
    PHONE_TYPE STR_100,
    K_ID ID_TYPE,
    KF_NAME STR_100,
    K_KONTRACT_NO STR_100,
    C_NAME_FIO STR_100,
    CC_NAME STR_100,
    UN_NAME STR_100,
    COMMAND_ID D_INTEGER,
    DB_ID D_INTEGER,
    MSK_CALL_TIME_START D_TIME,
    MSK_CALL_TIME_END D_TIME)
AS
declare variable CC_ID ID_TYPE;
declare variable C_ID ID_TYPE;
declare variable CLIENT_CELL_NO STR_100;
declare variable CAN_CALL_2 D_SMALLINT;
declare variable CONT_CALL_IS_ACTIVE_CL_LIST STR_255;
declare variable CELL_MAIN_ID D_INTEGER;
declare variable CELL_ADD_ID D_INTEGER;
declare variable CELL_MAIN_CAN_CALL D_INTEGER;
declare variable CELL_ADD_CAN_CALL D_INTEGER;
declare variable CELL_ADD STR_100;
declare variable CLI_QTY_COUNTER D_INTEGER;
declare variable KONTRACT_CAN_CALL D_INTEGER;
declare variable CONT_OUT_OF_TURN_ACTIVE D_INTEGER;
declare variable C_ID_OLD ID_TYPE;
declare variable UN_ID_OLD ID_TYPE;
declare variable SIRENA1_RD_ID D_INTEGER;
declare variable PANTERA_RD_ID D_INTEGER;
declare variable INV_AND_REC_RD_ID D_INTEGER;
declare variable KS2_RD_ID D_INTEGER;
declare variable PROJECT_ID D_INTEGER;
begin
  db_id = f_get_db_id();
  language = 'Language.RU';

  --OKTELL.  Список задач для обработки OKTELL по проекту "Сирена"
  cell_main_id = f_get_rd_id('ContactInfo.Cell_Main');
  cell_add_id = f_get_rd_id('ContactInfo.Cell_Add');

  sirena1_rd_id = f_get_rd_id('OKTELL.Projects.Sirena1');
  pantera_rd_id = f_get_rd_id('OKTELL.Projects.Pantera');
  inv_and_rec_rd_id = f_get_rd_id('OKTELL.Projects.Investigation_And_Recovery');
  ks2_rd_id = f_get_rd_id('OKTELL.Projects.Ks_finance_2');

  un_id_old = 0;

  for select k_id, pant_id, un_id, 'OKTELL.Projects.Sirena1', delinq_days, kf_name, command_id, :sirena1_rd_id, k_kontract_no, cli_qty_counter
      from okt_sirena_list(current_date, 'OKTELL.Projects.Sirena1', '', 'OKTELL_LIST', '', '', '', 21, 60, 0, 0, 0, 0, 81001)
      union all
      select k_id, pant_id, un_id, 'OKTELL.Projects.Pantera', delinq_days, kf_name, command_id, :pantera_rd_id, k_kontract_no, cli_qty_counter
      from okt_pantera_list(current_date, 'OKTELL.Projects.Pantera', 'OKTELL_LIST', '', '', '', 1, 9999, 0, 0, 0, 0, 81001)
      union all
      select k_id, pant_id, un_id, 'OKTELL.Projects.Investigation_And_Recovery', delinq_days, kf_name, command_id, :inv_and_rec_rd_id, k_kontract_no, cli_qty_counter
      from okt_pantera_list(current_date, 'OKTELL.Projects.Investigation_And_Recovery', 'OKTELL_LIST', '', '', '', 1, 9999, 0, 0, 0, 0, 81001)
      union all
      select k_id, pant_id, un_id, 'OKTELL.Projects.Ks_finance_2', delinq_days, kf_name, command_id, :ks2_rd_id, k_kontract_no,  cli_qty_counter
      from okt_pantera_list(current_date, 'OKTELL.Projects.Ks_finance_2', 'OKTELL_LIST', '', '', '', 1, 9999, 0, 0, 0, 0, 81001)
      into :k_id, :pant_id, :un_id, :project, :delinq_days, :kf_name, :command_id, :project_id, :k_kontract_no, :cli_qty_counter
  do
  begin
    -- проверяю договор по игнор-листу
    select can_call
    from okt_can_call(:project_id, 'Kontract', :k_id, null, null, null, null)
    into :kontract_can_call;

    -- ===================== УЧАСТНИКИ ГРУППЫ ==================================
    if (kontract_can_call = 1) then -- если нет в игнор-листе
    begin
      cc_name = '';
      person_type = 'Clients';
      cell_no = '';

      cont_call_is_active_cl_list = '';

      for select kncl.kc_from_c_id
          from knclient kncl
          where kncl.kc_amount_approved > 0 and
                kncl.kc_from_k_id = :k_id
          into :person_id
      do
      begin
        cont_out_of_turn_active = null;

        -- проверяю, ставилась ли галочка звонить конт лицам не дожидаясь очереди
        select pds.pds_cont_call_is_active
        from pantera_data_svod pds
        where pds.pds_from_pan_id = :pant_id and
              pds.pds_from_c_id = :person_id
        order by pds.pds_id desc
        rows 1
        into :cont_out_of_turn_active;

        cont_out_of_turn_active = coalesce(cont_out_of_turn_active, -1);

        cell_main_can_call = 0;
        cell_add_can_call = 0;

        -- проверка клиента по игнор-листу
        select can_call
        from okt_can_call(null, 'clients', :person_id, null, null, :cell_main_id, null)
        into :cell_main_can_call;

        if (cell_main_can_call = 0) then -- если основной нормер в игнор-лиcте, проверяем доп. номер
        begin
          cell_add = f_cont_info('clients', person_id, 'cell_add', 'last');

          if (cell_add <> '') then
          begin
            select can_call
            from okt_can_call(null, 'Clients', :c_id, null, null, :cell_add_id, null)
            into :cell_add_can_call;
          end
        end

        if ((cell_main_can_call = 1) or (cell_add_can_call = 1)) then -- если можно дозвониться по основному или доп номеру
        begin
          can_call_2 = 1;

          -- проверяю, наступило ли время для следующего звонка
          select can_call,
                 timezone,
                 msk_call_time_start,
                 msk_call_time_end
          from hp_can_call_check(:k_id, 'Clients', :person_id, null, :ignore_allowed_call_time)
          into :can_call_2,
               :timezone,
               :msk_call_time_start,
               :msk_call_time_end;

          if (can_call_2 = 1) then
          begin
            if (cell_main_can_call = 1) then
            begin
              cell_no = f_cont_info('clients', person_id, 'cell_main', 'last');
              phone_type = 'ContactInfo.Cell_Main';
            end
            else
            begin
              cell_no = cell_add;
              phone_type = 'ContactInfo.Cell_Add';
            end

            if (cell_no similar to '[-]?[[:DIGIT:]]+') then
            begin
              select cl.c_name_fio
              from clients cl
              where cl.c_id = :person_id
              into :c_name_fio;

              url = 'http://oktell.molbulak.ru/Sirena_d.aspx?project=' || trim(:project) || '&pant_id=' || pant_id || '&object_id=' || person_id || '&object_kind=' || person_type || '&client_cell_no=' || cell_no || '&K_ID=' || k_id || '&cont_call_active=' || abs(cont_out_of_turn_active) || '&phone_type=' || :phone_type ||'&call_system=OKTELL' || '&un_id=';

              if (for_analiz = 1) then
              begin
                if (un_id <> un_id_old) then
                begin
                  un_id_old = un_id;

                  select un.un_name
                  from username un
                  where un_id = :un_id
                  into :un_name;
                end
              end

              suspend;
            end
          end
        end
        else
          cli_qty_counter = cli_qty_counter - 1;

        -- если для клиента указан внеочередной обзвон конт лиц, добавлю этого клиента в список
        if (cont_out_of_turn_active = 1) then
          cont_call_is_active_cl_list = cont_call_is_active_cl_list || '; ' || cast(:person_id as str_100);
      end
    end

    -- ========== КОНТАКТНЫЕ ЛИЦА ==============================================
    -- обзваниваем контактных лиц только если
    -- а) договор не в игнор-листе
    -- б) все участники группы в игнор-листе или если участники группы у которых выставлена галочка "внеочредной обзвон контаткных лиц"
    c_id_old = 0;

    if ((kontract_can_call = 1) and
        ((cli_qty_counter = 0) or (cont_call_is_active_cl_list <> ''))) then
    begin
      cell_no = '';
      person_type = 'Contacts';
      phone_type = 'ContactInfo.Cell_Main';

      for select cont.cc_id,
                 cont.cc_from_c_id,
                 left(coalesce(cont.cc_cell_no, ''), 20),
                 cont.cc_name
          from knclient kncl
          inner join contacts cont on cont.cc_from_c_id = kncl.kc_from_c_id
          where kncl.kc_amount_approved > 0 and
                kncl.kc_from_k_id = :k_id and
                ((cont.cc_total_res = 1) or
                (cont.cc_verif_ignore = 1)) and -- беру только верифицированные + которых не верифицировали из-за галочки "не верифицировать"
                cont.cc_is_old_contact = 0 and -- беру только последние контакты
                cont.cc_is_delete = 0 -- RFCRU-2246
          into :cc_id,
               :c_id,
               :cell_no,
               :cc_name
      do
      begin
        -- если клиенты для обзвона закончились либо, поставлена галочка - внеочередной обзвон контактных лиц
        if (((cli_qty_counter = 0) or (cont_call_is_active_cl_list containing (cast(:c_id as str_100)))) and
            (cell_no similar to '[-]?[[:DIGIT:]]+')) then
        begin
          person_id = cc_id;

          -- проверяю по игнор листу
          select can_call
          from okt_can_call(null, 'Contacts', :cc_id, null, null, :cell_main_id, null)
          into :cell_main_can_call;

          if (cell_main_can_call = 1) then -- если абонента нет в исключениях на обзвон
          begin
            can_call_2 = 0;

            -- проверяю, наступила ли дата+время следующего звонка
            select can_call,
                   timezone,
                   msk_call_time_start,
                   msk_call_time_end
            from hp_can_call_check(:k_id, 'Contacts', :cc_id, null, :ignore_allowed_call_time)
            into :can_call_2,
                 :timezone,
                 :msk_call_time_start,
                 :msk_call_time_end;

            if (can_call_2 = 1) then
            begin
              client_cell_no = f_cont_info('clients', c_id, 'cell_main', 'last');

              if (client_cell_no similar to '[-]?[[:DIGIT:]]+') then
              begin
                url = 'http://oktell.molbulak.ru/Sirena_d.aspx?project=' || trim(:project) || '&pant_id=' || pant_id || '&object_id=' || person_id || '&object_kind=' || person_type || '&contact_cell_no=' || cell_no || '&client_cell_no=' || client_cell_no || '&c_id=' || c_id || '&K_ID=' || k_id || '&phone_type=' || :phone_type ||'&call_system=OKTELL' || '&un_id=';
                phone_type = 'ContactInfo.Cell_Main';

                if (c_id <> c_id_old) then
                begin
                  select cl.c_name_fio
                  from clients cl
                  where cl.c_id = :c_id
                  into :c_name_fio;

                  c_id_old = c_id;
                end

                if (for_analiz = 1) then
                begin
                  if (un_id <> un_id_old) then
                  begin
                    un_id_old = un_id;

                    select un.un_name
                    from username un
                    where un_id = :un_id
                    into :un_name;
                  end
                end
                suspend;
              end
            end
          end
        end
      end
    end
  end
end
