CREATE PROCEDURE AAREP_UNDERWRITER_TIME(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    FIRST_LINE_TIME_STR STR_10,
    WITHOUT_REVISION D_INTEGER,
    REVISION D_INTEGER,
    UPDATED_APP D_INTEGER,
    IN_PROCESS D_INTEGER,
    REFUSAL D_INTEGER,
    COUNT_ALL D_INTEGER,
    MIDDLE_TIME TIME,
    MIN10 D_INTEGER,
    MIN20 D_INTEGER,
    MIN30 D_INTEGER,
    MIN60 D_INTEGER,
    MIN90 D_INTEGER,
    MIN120 D_INTEGER,
    MIN150 D_INTEGER,
    MIN180 D_INTEGER,
    MIN180_MORE D_INTEGER,
    COUNT_ALL2 D_INTEGER)
AS
declare variable MT_H smallint;
declare variable MT_M smallint;
declare variable MT_S smallint;
begin 

    for select left(cast(first_line_datetime as time), 2),
               avg(cast(iif(left(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time), 1) in (1,2), left(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2), 2), left(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2), 1))as smallint)),
               avg(cast(substring(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time) from 4 for 2)as smallint)),
               avg(cast(substring(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time) from 7 for 2)as smallint)),
               sum(iif(first_status = 'Одобрен', 1, 0)),
               sum(iif(first_status = 'Доработка', 1, 0)),
               sum(iif(first_status <> last_status, 1, 0)),
               sum(iif(first_status = 'Доработка' and last_status= 'Доработка', 1, 0)),
               sum(iif(first_status = 'Отказан', 1, 0)),
               count(acl_id),
               sum(iif(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time) between cast('00:00:00' as time) and cast('00:10:00' as time), 1, 0)),
               sum(iif(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time) between cast('00:10:01' as time) and cast('00:20:00' as time), 1, 0)),
               sum(iif(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time) between cast('00:20:01' as time) and cast('00:30:00' as time), 1, 0)),
               sum(iif(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time) between cast('00:30:01' as time) and cast('01:00:00' as time), 1, 0)),
               sum(iif(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time) between cast('01:00:01' as time) and cast('01:30:00' as time), 1, 0)),
               sum(iif(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time) between cast('01:30:01' as time) and cast('02:00:00' as time), 1, 0)),
               sum(iif(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time) between cast('02:00:01' as time) and cast('02:30:00' as time), 1, 0)),
               sum(iif(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time) between cast('02:30:01' as time) and cast('03:00:00' as time), 1, 0)),
               sum(iif(cast(iif(first_processing2 containing '-дн.', '23:59:59',first_processing2) as time) > cast('03:00:00' as time), 1, 0))
    from aarep_underwriter_consolid_new(:rep_region_id, :rep_date_begin, :rep_date_end)
    group by 1
    order by 1
    into first_line_time_str, mt_h, mt_m, mt_s,
         without_revision, revision, updated_app, in_process, refusal, count_all,
         min10, min20, min30, min60, min90, min120, min150, min180, min180_more
    do
    begin

        middle_time = '00:00:00';

        count_all2 = min10 + min20 + min30 + min60 + min90 + min120 + min150 + min180 + min180_more;

        first_line_time_str = cast(first_line_time_str as smallint) || '-' || iif(cast(first_line_time_str as smallint) + 1 = 24, 0, cast(first_line_time_str as smallint) + 1);

        middle_time = mt_h || ':' || mt_m || ':' || mt_s;

        suspend;
    end 
end
