CREATE PROCEDURE HP_SEND_CODE_TO_CLIENTS(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_8100,
    SHOW_RES D_STR_10)
AS
declare variable TMP_CODE D_INTEGER;
declare variable CNC_CODE D_INTEGER;
declare variable CELL_NO D_STR_100;
declare variable CNC_ID D_ID;
declare variable COUNT_SMS_BY_HOUR D_INTEGER;
declare variable SMS_TEXT_STR D_STR_255;
begin
  -- RFCRU-3522 процедура генерирует код
  res_text = '';
  sms_text_str = '';
  show_res = 'dialog';
  tmp_code = cast(1000 + floor(rand() * 9000) as d_integer); -- новый код
  cell_no = f_cont_info('clients', :c_id, 'cell_main', 'ok');

  select count(*)
  from check_number_code cnc
  where cnc.cnc_from_c_id = :c_id         -- кол-во СМС отправленных за час
    and cnc.cnc_cell_main = :cell_no
    and cast((cnc.cnc_create_datetime - current_timestamp) * 24 as d_float) between -1 and 0
  into count_sms_by_hour;

  if (count_sms_by_hour >= 3) then
    res_text = f_format('Кол-во СМС отправленных за час не должно быть больше %1', count_sms_by_hour);

  if (res_text = '') then
  begin
    select cnc.cnc_code, cnc.cnc_id
    from check_number_code cnc
    where cnc.cnc_from_c_id = :c_id
      and cnc.cnc_is_active = 1
    order by cnc.cnc_create_datetime
    into cnc_code, cnc_id;

    cnc_id = coalesce(cnc_id, 0);

    while(cnc_code = tmp_code) do
    begin
       tmp_code = cast(1000 + floor(rand() * 9000) as d_integer); -- новый код
    end
  
    insert into check_number_code (cnc_from_c_id, cnc_code, cnc_cell_main, cnc_from_un_id) -- создаем новый код
    values (:c_id, :tmp_code, :cell_no, :un_id);

    if(cnc_id <> 0) then
    begin
      update check_number_code cnc   -- удаляем старый
         set cnc.cnc_is_active = 0
      where cnc.cnc_id = :cnc_id;
    end

    sms_text_str = f_format('Dlya izmeneniya vashykh personalnykh dannykh nazovite kod %1 vashemu specialistu', :tmp_code); 
    --отправляем СМС
    execute procedure hp_send_quick_message('sms', :cell_no, '', :sms_text_str, '', -1, :un_id);

    res_text = f_format('Код отправлен на номер %1', cell_no);
  end

  suspend;
end
