CREATE PROCEDURE HP_REPAY_EXTRA_SALDO(
    K_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    RES_LAST_DATE DATE,
    RES_AMOUNT_SALDO DECIMAL(18,4),
    RES_TOTAL_PLUS DECIMAL(18,4),
    RES_TOTAL_MINUS DECIMAL(18,4),
    RES_TOTAL_WRITEOFF DECIMAL(18,4),
    RES_RD_ID INTEGER,
    RES_RD_NAME VARCHAR(255),
    RES_DETAIL VARCHAR(1024))
AS
begin
-- Процедура возвращается существующие обязательства по кредиту
-- Вызывается из окна погашения кредита

-- ID обязательства передается в параметре Rep_RD_ID, если 0 - значит все
-- Тип обязательства передается в параметре Rep_Str_ID, если "" - значит все
-- Тип обязательства могут быть "Money", "Zalog", "Poruch" и т.д.

--  HP_DO_SHOW_REPAYMENT  -- Показывает задолженность
--  HP_DO_REPAYMENT       -- Распределяет погашение

  c_id = 0;
  for select res_last_date, res_rd_id, res_rd_name, res_detail, res_amount_saldo, res_total_plus, res_total_minus,
             res_total_writeoff
      from aarep_operextra_saldo(9, :k_id, 0, 'LoanRepay', :rep_date, 0)
      into res_last_date, res_rd_id, res_rd_name, res_detail, res_amount_saldo, res_total_plus, res_total_minus,
           res_total_writeoff
  do
  begin
    suspend;
  end
end
