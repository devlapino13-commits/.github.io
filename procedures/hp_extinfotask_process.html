CREATE PROCEDURE HP_EXTINFOTASK_PROCESS(
    PROCESS_MODE D_SMALLINT NOT NULL = 0)
AS
declare variable EIC_CODE D_STR_CODE;
declare variable DB_ID D_ID;
declare variable EIL_ID D_ID;
declare variable WAS_ERROR D_BOOL;
declare variable ERROR_COUNT D_INTEGER;
declare variable MAIL_BODY D_STR_8100;
begin
  -- Процедура обработки ответов для таблицы extinfolist.
  -- Выполняется FsService при получении события extinfo_process,
  -- вызов события extinfo_process происходит в EXTINFOLIST_AU0
  -- !!! Не пишите тут реализацию, создайте отдельную процедуру-обработчик,
  -- !!! и вызывайте в зависимости от страны и eic_code
  -- RFCRU-4557, RFCRU-4636

  -- PROCESS_MODE - параметр, задающий режим работы процедуры
  -- 0 - обрабатываются все запросы из extinfotask, кроме ранее помеченных ошибочными.
  --     Eсли возникает исключение, будет отправлено письмо разработчикам,
  --     с информацией об ошибке, а запись в exitinfotask помечена как ошибочная.
  --     Далее будет продолжена обработка оставшихся записей.
  -- 1 - режим, в котором обрабатываются только ошибочные записи extinfotask.
  --     При возникновении исключения процедура прерывает выполнение с выдачей
  --     информации об ошибке.
  -- RFCRU-4639

  db_id = f_get_db_id();
  was_error = 0; -- Признак ошибки, чтобы не отправлять несколько отчетов

  for select eil_id, eic_code
      from extinfotask
      left join extinfolist on eit_from_eil_id = eil_id
      left join extinfocfg on eil_from_eic_id = eic_id
      where eit_status = -:process_mode
      into eil_id, eic_code
  do
  begin
    if (db_id = 81001) then
    begin
       if (eic_code in ('kontur', 'kontur_b2')) then   -- RFCRU-5873
        execute procedure hp_bankrupt_update(eil_id);

      if (eic_code in ('inn_nalog', 'inn_gnivc', 'inn_smev')) then
        execute procedure hp_extinfolist_inn_update(eil_id);

      if (eic_code = 'bki_ssp_40') then
        execute procedure hp_extinfolist_selfblock_update(eil_id);

      if (eic_code = 'passport_smev') then
        execute procedure hp_extinfolist_smevpass_upd(eil_id);

      if (eic_code = 'ligres_fssp_cli_check') then
        execute procedure hp_extinfo_ligres_upd(eil_id);
    end
    else
    if (db_id = 41701) then
    begin
      if (eic_code in ('stop_cr_kib', 'stop_cr_crif')) then
        execute procedure hp_self_prohibition_update(eil_id);

      if (eic_code = 'check_in_ru_db') then
        execute procedure hp_online_loan_ru_kib_check(eil_id);

      if (eic_code in ('passport_status', 'marital_status')) then
        execute procedure hp_extinfo_pass_status_process(eil_id);
      
      if (eic_code = 'policy_info') then
        execute procedure hp_extinfo_policy_info_process(eil_id);

      if (eic_code = 'policy_cancel') then
        execute procedure hp_extinfo_pol_cancel_process(eil_id);

      if (eic_code = 'tunduk') then
        execute procedure hp_extinfo_tunduk_process(eil_id);

      if (eic_code = 'check_npdl') then
        execute procedure hp_npdl_response_update(eil_id);
    end
    else
    if (db_id = 39801) then
    begin
      if (eic_code in ('afm_iin_check', 'afm_name_check')) then
        execute procedure hp_extinfolist_afm_update(eil_id);

      else if (eic_code = 'kdn_score') then
        execute procedure hp_extinfo_kdn_score_process(eil_id);

      else if (eic_code = 'antifraud_client_check') then
        execute procedure hp_extinfo_antifraud_process(eil_id);
      
      else if (eic_code = 'military_status') then
        execute procedure hp_extinfo_military_process(eil_id);

      else if (eic_code = 'salary_check') then
        execute procedure hp_extinfo_salary_process(eil_id);

      else if (eic_code = 'pension_check') then
        execute procedure hp_extinfo_pension_process(eil_id);

      else if (eic_code = 'asp_check') then
        execute procedure hp_extinfo_asp_process(eil_id);

      else if (eic_code = 'bankruptcy') then
        execute procedure hp_extinfo_bankrupt_process(eil_id);

      else if (eic_code = 'pkb_cib_report_negative') then
        execute procedure hp_extinfo_pkb_rep_neg_process(eil_id);

      else if (eic_code = 'pkb_available_reports') then
        execute procedure hp_extinfo_pkb_avlbl_process(eil_id);

      else if (eic_code = 'bmg') then
        execute procedure hp_extinfo_bmg_process(eil_id);

      else if (eic_code = 'gbdfl_get_cli_data') then
        execute procedure hp_extinfo_gbdfl_process(eil_id);

      else if (eic_code = 'stop_credit') then
        execute procedure hp_extinfo_stop_cr_process(eil_id);

      else if (eic_code = 'military_pocket') then
        execute procedure hp_extinfo_military_pocket_process(eil_id);

      else if (eic_code = 'pkb_cib_report') then
      begin
        execute procedure hp_parse_pkb_report_process(eil_id);
        execute procedure hp_refresh_pkb_data(:eil_id);  -- RFCKZ-832
      end

      -- RFCKZ-877: обработка ответа Fraud Score ГКБ
      else if (eic_code = 'gkb_pay_checks_core') then
        execute procedure hp_extinfo_gkb_core_process(eil_id);
    end
    else
    if (db_id = 94401) then
    begin
      if (eic_code = 'fin') then
        execute procedure hp_extinfo_fin_process(eil_id);
    end
    -- Удаляем после обработки. Даже если обработчик не реализован, удаляем,
    -- чтобы не копился мусор.
    delete from extinfotask
    where eit_from_eil_id = :eil_id;

    -- Обработка исключений
    when gdscode deadlock do
    begin
      -- Прячем исключение при параллельном выполнении
      -- Сервис работает в один поток, но кто-то может запустить руками
    end
    when any do
    begin
      if (process_mode = 1) then
        exception; -- Возвращаем оригинальное исключение, для отладки

      -- Если первая ошибка, отправляем письмо
      if (was_error = 0) then
      begin
        was_error = 1;

        --Убеждаемся, что ранее не было ошибок, т.к. письмо отправляем один раз
        select count(*)
        from extinfotask
        where eit_status = -1
        into error_count;

        if (error_count = 0) then
        begin
          -- Отправляем информацию об ошибке разработчикам
          mail_body =
            'Уважаемые, коллеги.<br><br>' ||
            'При обработке задач в hp_extinfotask_process возникло исключение:<br><br>' ||
            f_format('<code>Задача: eil_id: %1, eic_code: %2, Исключение: gdscode:%3, sqlcode:%4, sqlstate:%5</code><br><br>',
              eil_id, eic_code, gdscode, sqlcode, sqlstate) ||
            'Получить детали ошибки можно с помощью вызова:<br><br>' ||
            '<code>execute procedure hp_extinfotask_process(1);</code><br><br>' ||
            'После исправления обработчика/ошибки, необходимо повторно выполнить '||
            'следующий код, чтобы убедиться что все ошибки обработаны:<br><br>' ||
            '<code>execute procedure hp_extinfotask_process(1);</code>';

          execute procedure hp_send_quick_message(
            'email',
            iif(db_id = 81001, 'ru.finstart@molbulak.com', 'kg.finstart@mbulak.kg'),
            'Проблемы при обработке Extinfolist (Extinfotask)',
            mail_body,
            'hp_extinfotask_process',
            1,
            1000000000000);
        end
      end

      -- Маркируем ошибку, чтобы исключить повторную обработку
      update extinfotask
      set eit_status = -1
      where eit_from_eil_id = :eil_id;
    end
  end
end
