CREATE PROCEDURE HP_COM_PACKET_STAT(
    CMP_ID D_INTEGER)
RETURNS (
    REC_ID D_INTEGER,
    REC_PARENT_ID D_INTEGER,
    RES_CAPTION D_STR_255,
    RES_COUNT D_INTEGER)
AS
declare variable cmc_name d_str_255;
declare variable cmc_id d_integer;
declare variable cml_status d_integer;
declare variable enum_send_id d_integer;
begin
  rec_id = 0;

  enum_send_id = f_enum('com_action_status', 'sent', 'id');

  select count(*)
  from com_list
  where cml_last_cmp_id is null and
        cml_last_status = :enum_send_id
  into res_count;

  if (res_count > 0) then
  begin
    res_caption = 'Запросы в очереди';
    rec_parent_id = null;
    rec_id = rec_id + 1;
    suspend;
    rec_parent_id = rec_id;
  
    for select cmc_id, cmc_name
    from com_cfg
    into cmc_id, cmc_name
    do
    begin
      res_caption = cmc_name;
      select count(*)
      from com_list
      where cml_last_cmp_id is null and
            cml_from_cmc_id = :cmc_id and
            cml_last_status = :enum_send_id
      into res_count;
      rec_id = rec_id + 1;
      suspend;
    end
  end

  for select cmc_id, cmc_name
  from com_cfg
  into cmc_id, cmc_name
  do
  begin
    cml_status = -1;
    res_count = 0;
    select count(*)
    from com_list
    where cml_last_cmp_id = :cmp_id and
          cml_from_cmc_id = :cmc_id
    into res_count;
    res_caption = cmc_name;
    rec_parent_id = null;

    if (res_count > 0) then
    begin
      rec_id = rec_id + 1;
      suspend;
  
      rec_parent_id = rec_id;
      for select cml_last_status, count(*)
      from com_list
      where cml_last_cmp_id = :cmp_id and
            cml_from_cmc_id = :cmc_id
      group by cml_last_status
      into cml_status, res_count
      do
      begin
        res_caption = f_enum('com_action_status', cml_status, 'name');
        rec_id = rec_id + 1;
        suspend;
      end
    end
  end
end
