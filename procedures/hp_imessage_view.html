CREATE PROCEDURE HP_IMESSAGE_VIEW(
    UN_ID D_ID,
    ID D_ID = 0,
    IM_TYPE D_INTEGER = 0)
RETURNS (
    IM_ID D_ID,
    IM_FROM_OL_ID D_ID,
    OL_DATE D_DATE,
    IM_FROM_K_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    IM_FROM_C_ID D_ID,
    K_DISB_DATE D_DATE,
    IM_FROM_REG_ID D_INTEGER,
    REG_NAME VARCHAR(50),
    IM_THROUGH_WHOM D_STR_255,
    IM_AMOUNT_PAYMENT CURR_TYPE,
    IM_CREATE_DATETIME D_DATETIME,
    IM_CASHIER_ID D_ID,
    CASHIER_NAME D_STR_100,
    CASHIER_POS D_STR_255,
    IM_CASHIER_SIGN_DATETIME D_DATETIME,
    IM_CONTROLLER_1_ID D_ID,
    CONTROLLER_1_NAME D_STR_100,
    CONTROLLER_1_POS D_STR_255,
    IM_CONTROLLER_1_SIGN_DATETIME D_DATETIME,
    IM_CONTROLLER_2_ID D_ID,
    CONTROLLER_2_NAME D_STR_100,
    CONTROLLER_2_POS D_STR_255,
    IM_CONTROLLER_2_SIGN_DATETIME D_DATETIME,
    IM_STATUS D_SMALLINT,
    STATUS D_STR_100)
AS
declare variable asign_resp d_integer;
declare variable asign_auth d_integer;
declare variable c_name_fio d_str_100;
declare variable dt_sign_loan d_datetime;
declare variable dt_issue_loan d_datetime;
begin
  -- право подписи ответственным лицом
  asign_resp =
    iif(exists(
        select *
        from permissions
        where pm_object_kind = 0
          and pm_object_id = :un_id
          and pm_subject_kind = 3
          and pm_subject_id = f_get_pa_id('FinStart.IM_EarlyRepay_Responsible')
          and pm_is_delete = 0
          and current_date between coalesce(pm_date_begin, '01.01.1900')
            and coalesce(pm_date_end,'31.12.9999')
          and pm_access_kind = 1), 1, 0);
  -- право подписи ответственным лицом
  asign_auth =
    iif(exists(
        select *
        from permissions
        where pm_object_kind = 0
          and pm_object_id = :un_id
          and pm_subject_kind = 3
          and pm_subject_id = f_get_pa_id('FinStart.IM_EarlyRepay_Authorized')
          and pm_is_delete = 0
          and current_date between coalesce(pm_date_begin, '01.01.1900')
            and coalesce(pm_date_end, '31.12.9999')
          and pm_access_kind = 1), 1, 0);

  for
    select
      im_id,
      im_from_ol_id,
      ol_date,
      im_from_k_id,
      k_kontract_no,
      im_from_c_id,
      c_name_fio,
      k_disb_date,
      im_from_reg_id,
      reg_name,
      im_through_whom,
      im_amount_payment,
      im_create_datetime,
      im_cashier_id,
      u.un_fio,
      im_cashier_sign_datetime,
      im_controller_1_id,
      u1.un_fio,
      im_controller_1_sign_datetime,
      im_controller_2_id,
      u2.un_fio,
      im_controller_2_sign_datetime,
      im_status,
      case im_status
        when -1 then 'аннулировано'
        when  0 then 'нужна подпись КОБД'
        when  1 then 'нужна подпись ответственного лица'
        when  2 then 'нужна подпись уполномоченного лица'
        when  3 then 'подписано'
      end
    from internal_message
      left join operlist
        on ol_id = im_from_ol_id
      inner join kontract
        on k_id = im_from_k_id
      inner join region
        on reg_unicode = im_from_reg_id
      inner join username as u
        on u.un_id = im_cashier_id
      left join username as u1
        on u1.un_id = im_controller_1_id
      left join username as u2
        on u2.un_id = im_controller_2_id
      left join clients
        on c_id = im_from_c_id
    where iif((:id > 0) or (:asign_resp = 0 and :asign_auth = 0), (im_cashier_id = :un_id) and (im_status = 0),
      iif(:asign_resp = 1, im_status = 1, im_status = 2)) and (im_type = :im_type)
  into
      :im_id,
      :im_from_ol_id,
      :ol_date,
      :im_from_k_id,
      :k_kontract_no,
      :im_from_c_id,
      :c_name_fio,
      :k_disb_date,
      :im_from_reg_id,
      :reg_name,
      :im_through_whom,
      :im_amount_payment,
      :im_create_datetime,
      :im_cashier_id,
      :cashier_name,
      :im_cashier_sign_datetime,
      :im_controller_1_id,
      :controller_1_name,
      :im_controller_1_sign_datetime,
      :im_controller_2_id,
      :controller_2_name,
      :im_controller_2_sign_datetime,
      :im_status,
      :status
  do
  begin
    if (im_type = 1 and ol_date is null) then
      ol_date = cast(im_cashier_sign_datetime as d_date);

    cashier_pos =
      coalesce(f_get_position(:im_cashier_id, :ol_date), 'Кассир-оператор');
    if (im_controller_1_id > 0)
      then controller_1_pos = f_get_position(:im_controller_1_id, :ol_date);
    if (im_controller_2_id > 0)
      then controller_2_pos = f_get_position(:im_controller_2_id, :ol_date);
    if (im_type = 1) then
    begin
      dt_sign_loan = (
        select first 1
          eds_status_datetime
        from edoc_list
          inner join edoc_sign
            on eds_from_edl_id = edl_id
              and eds_status = 1
        where edl_object_kind = 9
          and edl_object_id = :im_from_k_id
          and edl_from_edc_id = 3
          and edl_status = 1);
      dt_issue_loan = (
        select
          ol_datetime
        from operlist
        where ol_id = :im_from_ol_id);
    end
    suspend;
  end
end
