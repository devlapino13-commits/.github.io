Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXEREFCFG_LIST
RETURNS (
    ERL_CODE D_STR_255,
    ERL_LOOKUP D_SMALLINT,
    RD_NAME D_STR_255,
    RD_MEMO D_STR_255,
    ERC_ERL_UUID D_UUID,
    ERC_SHOW_INACTIVE D_BOOL,
    ERC_INT_TAG_CAPTION D_STR_CODE,
    ERC_STR_TAG_CAPTION D_STR_CODE,
    ERC_VISIBLE_ID D_BOOL,
    ERC_VISIBLE_CODE D_BOOL,
    ERC_VISIBLE_NAME D_BOOL,
    ERC_VISIBLE_MEMO D_BOOL,
    ERC_VISIBLE_ORDERNO D_BOOL,
    ERC_VISIBLE_IS_ACTIVE D_BOOL,
    ERC_VISIBLE_STR_ID D_BOOL,
    ERC_VISIBLE_TAG D_BOOL,
    ERL_UUID_CODE D_STR_CODE)
AS
begin
  for select erl_name,
             erl_code,
             erl_lookup,
             erc_erl_uuid,
             erc_show_inactive,
             erc_int_tag_caption,
             erc_str_tag_caption,
             erc_visible_id,
             erc_visible_code,
             erc_visible_name,
             erc_visible_memo,
             erc_visible_orderno,
             erc_visible_is_active,
             erc_visible_str_id,
             erc_visible_tag
      from ux$erefcfg, ux$ereflist
      where erl_uuid = erc_erl_uuid and
            erl_code <> 'none'
      into :rd_name,
           :erl_code,
           :erl_lookup,
           :erc_erl_uuid,
           :erc_show_inactive,
           :erc_int_tag_caption,
           :erc_str_tag_caption,
           :erc_visible_id,
           :erc_visible_code,
           :erc_visible_name,
           :erc_visible_memo,
           :erc_visible_orderno,
           :erc_visible_is_active,
           :erc_visible_str_id,
           :erc_visible_tag
  do
  begin
    rd_memo = rd_name;
    erl_uuid_code = uuid_to_char(erc_erl_uuid);
    suspend;
  end
end
