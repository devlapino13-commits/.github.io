CREATE PROCEDURE HP_EXTSYSPIN_VERIFY(
    OBJ_KIND D_STR_CODE,
    CONT_KIND D_STR_CODE,
    CONT_VALUE D_STR_CODE,
    PIN_CODE D_STR_CODE)
RETURNS (
    RES_STATUS D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable clt_id d_integer;
declare variable esp_id d_id;
declare variable esp_status d_smallint;
begin
  res_status = 1; -- по умолчанию ошибка
  res_text = '';

  cont_kind = f_enum('cont_kind', cont_kind, 'id', 1);
  obj_kind = f_objkind(obj_kind, 'id');
  clt_id = f_cont_id(cont_value, 'null');

  if (clt_id is null) then
    res_text = f_lang('', 'Неправильный номер телефона');

  if (res_text = '') then
  begin
    pin_code = f_coder_pass(pin_code);
  
    select esp_id, esp_status
    from extsyspin
    where esp_cont_kind = :cont_kind and
          esp_from_clt_id = :clt_id and
          esp_object_kind = :obj_kind and
          esp_pin = :pin_code
    order by esp_status
    rows 1
    into :esp_id, :esp_status;
  
    if (esp_id is null) then
      res_text = f_lang('', 'Неправильный пин-код');
  end

  if (res_text = '') then
  begin
    if (esp_status <> 1) then
    begin
      update extsyspin
      set esp_status = 1
      where esp_id = :esp_id;
    end

    res_status = 0; -- нет ошибки
    res_text = f_lang('', 'OK');
  end

  suspend;
end
