CREATE PROCEDURE FM_CHANGE_DUE_DATE(
    UN_ID D_ID,
    OBJ_KIND D_STR_255,
    OBJ_ID D_ID,
    DISB_DAY D_INTEGER)
RETURNS (
    RES_ID INTEGER,
    RES_TXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable OBJ_KIND_ID D_ID;
begin

    if (:disb_day < 1 or :disb_day > 31 ) then
    begin
       res_id = -1;
       res_txt = 'Некорректный день месяца';
       suspend;
       exit;
    end 

    if (obj_kind = 'miniapp') then
    begin
        update miniapp
        set ma_repay_day = :disb_day
        where (ma_id = :obj_id) and (ma_date_end >= '31.12.2999') and (ma_is_active = 1);

        select res_id, res_txt
        from hp_make_schedule(:un_id, 11, :obj_id, 0, null, 0)
        into res_id, res_txt;
    end
    else
    begin
        update kontract
        set k_repay_day = :disb_day
        where (k_id = :obj_id);

        select res_id, res_txt
        from hp_make_schedule(:un_id, 9, :obj_id, 0, null, 0)
        into res_id, res_txt;

    end

  suspend;
end
