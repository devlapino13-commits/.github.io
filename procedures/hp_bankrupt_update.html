CREATE PROCEDURE HP_BANKRUPT_UPDATE(
    EIL_ID D_ID)
AS
declare variable status_date d_date;
declare variable is_bankrupt d_bool;
declare variable need_check d_bool;
declare variable eil_date d_date;
declare variable c_id d_id;
declare variable is_finished d_bool;
declare variable need_save d_bool;
declare variable json_str d_str_8100;
declare variable last_json_answer d_str_8100;
declare variable k_id d_id;
declare variable client_count d_integer;
begin
  need_save = 1;

  -- Добавление клиента в таблицу банкротов
  select eil_date, c_id, status_date, need_check, is_bankrupt, is_finished
  from hp_get_kontur_info(:eil_id)
  into eil_date, c_id, status_date, need_check, is_bankrupt, is_finished;

  select cast(blx_blob as d_blob_text)
  from extinfolist, extinfocfg, hp_get_blobex_blob(eil_from_bls_id)
  where eic_code in ('kontur', 'kontur_b2')                            -- RFCRU-5873
    and eil_from_eic_id = eic_id
    and eil_status = 2
    and eil_id = :eil_id
    and eil_from_bls_id is not null
  into json_str;

  select cast(blx_blob as d_blob_text)
  from bankrupt brt, extinfolist eil, hp_get_blobex_blob(eil_from_bls_id)
  where brt.brt_from_eil_id = eil.eil_id and
        brt.brt_from_c_id = :c_id and
        eil.eil_id <> :eil_id and
        eil.eil_date <= :eil_date
  order by eil.eil_date desc
  rows 1
  into last_json_answer;

  if (last_json_answer = json_str) then   -- RFCRU-5873 если текущий ответ не отличается от предыдущего, не делать вставку в таблицу bankrupt
    need_save = 0;

  if ((is_bankrupt = 1 or need_check = 1) and need_save = 1) then
  begin
    -- RFCRU-1274 Убрал вставку задним числом, и последующий пересчет oksummary
    -- из-за того, что расторжение делается в конце месяца (HP_KN_OKSUMMARY_BANKRUPT_UPDATE)
    -- Если после даты банкротства до конца месяца приходит погашение, это ломает график
    -- Если тут оставить как есть, а в HP_KN_OKSUMMARY_BANKRUPT_UPDATE расторгать
    -- концом месяца, то будет рассогласование с brt_date_begin.
    -- Удаленный код см. в истории изменений до 06.02.2024

    update or insert into bankrupt (
      brt_from_c_id, brt_from_eil_id, brt_date_begin,
      brt_date_fact, brt_need_check, brt_is_finished)
    values (:c_id, :eil_id, :eil_date,
      :status_date, :need_check, :is_finished)
    matching (brt_from_c_id, brt_date_begin);

    if (is_finished = 1) then
    begin
      for select k_id, count(kc_id)
          from knclient, kontract, oksummary
          where kc_from_c_id = :c_id and
                kc_from_k_id = k_id and
                kc_amount_approved > 0 and
                oks_from_k_id = k_id and
                :status_date between oks_date_begin and oks_date_end and
                oks_close_state in (0, 1)
          group by k_id
          into k_id, client_count
      do
      begin
        if (coalesce(client_count, 0) = 1) then
        begin
          if (not exists(select *
                         from bki_bankrupt
                         where bbt_from_k_id = :k_id)) then
            insert into bki_bankrupt (bbt_from_k_id, bbt_date)
            values (:k_id, :status_date);
        end
      end
    end
  end
end
