CREATE PROCEDURE FM_HTML_EXPRESSAPP_STATUS(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable CL_FIO STR_100;
declare variable LOAN_AMOUNT integer;
declare variable LOAN_PERIOD integer;
declare variable LANG D_STR_LANG;
declare variable EXC_NAME D_STR_255;
declare variable DISB_DATE D_STR_100;
declare variable EXA_FINAL_STATUS D_SMALLINT;
declare variable EXA_WAIT_STATUS D_SMALLINT;
declare variable WAIT_TEXT D_STR_LONG;
declare variable RES_HTML D_STR_LONG;
declare variable EXA_DATETIME_STR D_STR_CODE;
begin
  res_text = (select res_style from fm_html_styles) || '<body>';

  lang = f_get_lang_code(un_id, null);

  select c_name_fio,
         exc_name,
         f_datetostr(exa_disb_date),
         exa_amount,
         exa_period,
         exa_final_status,
         exa_wait_status,
         f_datetimetostr(exa_datetime)
  from expressapp, clients, expresscfg
  where exa_id = :obj_id
  and exa_from_c_id = c_id
  and exa_from_exc_id = exc_id
  into cl_fio,
       exc_name,
       disb_date,
       loan_amount,
       loan_period,
       exa_final_status,
       exa_wait_status,
       exa_datetime_str;

  res_text = BLOB_APPEND(res_text, '<h1 class="green_text">' || :cl_fio || '</h1>');

  res_text = BLOB_APPEND(res_text,
    f_format('%1: <b>%2</b><br>', f_lang(lang, 'Дата выдачи'), disb_date) ||
    f_format('%1: <b>%2%3</b><br>', f_lang(lang, 'Сумма займа'), loan_amount) ||
    f_format('%1: <b>%2 мес.</b><br>', f_lang(lang, 'Срок'),loan_period)||'<br>');

  res_html = '';
  if (exa_wait_status is not null) then
  begin
     lang = f_get_lang_code(un_id, null);

     res_text = BLOB_APPEND(res_text, case exa_wait_status
       when -1 then 'Робот &#128736;' -- черный   &#128736; Молоток и гаечный ключ, настройка
       when  0 then 'Робот &#9989;'   -- зеленый  &#9989; Зеленая галочка
       when  1 then 'Робот &#128293;' -- Красный &#128293; Огонек
       when  2 then 'Робот &#9203;'   -- Желтый  &#9203; песочные часы
     end);
     res_text = BLOB_APPEND(trim(res_text),'');

     if (exa_final_status  in (0, 1)) then
     begin
       wait_text = case exa_wait_status
         when -1 then '&#128736; Ошибка, отправлено разработчикам' -- черный   &#128736; Молоток и гаечный ключ, настройка
         when  0 then '&#9989; В обработке'   -- зеленый  &#9989; Зеленая галочка
         when  1 then '&#128293; Требуется ваше участие' -- Красный &#128293; Огонек
         when  2 then '&#9203; В ожидании'   -- Желтый  &#9203; песочные часы
       end;
       wait_text = trim(wait_text);

       res_html = res_html || '<font color="#02b316"><b>' || f_lang(lang, 'Заявку ведет робот.') || '</b></font><br>';
     end else
     begin
       wait_text = case exa_final_status
         when -2 then '&#10006; Клиент отказал'
         when -1 then '&#10060; Отказ от компании'
         when  2 then '&#9989; Займ выдан'
       end;
       wait_text = trim(wait_text);

       res_html = res_html || '<font color="#02b316"><b>' || f_lang(lang, 'Заявка обработана роботом.') || '</b></font><br>';
     end
     res_html = res_html || exa_datetime_str || ', статус:' || '<br>';
     res_html = res_html || '<b>' || exc_name || '</b><br>';
     res_html = res_html || trim(wait_text) || '<br><br>';
  end
  res_text = BLOB_APPEND(res_text, res_html);

  if (exa_wait_status = 1 and exa_final_status  in (0, 1)) then
  begin
    res_text = BLOB_APPEND(res_text, f_uri_sql(f_lang(lang, 'Продолжить'), 'express_process', '', :obj_id, '', f_lang(lang, 'Подтверждаете?'), '', 0, 0, '', '', 'dialog', 'next') || '<br><br>');
    res_text = BLOB_APPEND(res_text, f_uri_sql(f_lang(lang, 'Отказать'), 'express_process', '', :obj_id, '', f_lang(lang, 'Подтверждаете?'), '', 0, 0, '', '', 'dialog', 'cancel') || '<br><br>');
  end

  suspend;

end
