CREATE PROCEDURE HP_SUD_REQUEST_CREATE
AS
declare variable k_id d_id;
declare variable un_id d_id;
declare variable smr_id d_id;
begin
  -- создание заявок судебного модуля путем импорта из excel (RFCRU-3149)
  -- создание заявок по контрактам, импортированных во временную таблицу из excel

  for select sri_k_id,
             sri_un_id
      from sud_request_import
      where sri_note_id = 1
      into k_id,
           un_id
  do
  begin
    smr_id = null;
    select smr_id
    from hp_sud_new_request(:k_id, :un_id)
    into smr_id;
    smr_id = coalesce(smr_id, 0);

    if (smr_id > 0) then
    begin
      update sud_request_import
      set sri_note = 'Заявка создана',
          sri_note_id = 2
      where sri_k_id = :k_id;
    end
    else
      update sud_request_import
      set sri_note = 'Возникла ошибка при импорте',
          sri_note_id = -2
      where sri_k_id = :k_id;
  end
end
