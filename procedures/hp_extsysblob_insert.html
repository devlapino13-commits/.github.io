CREATE PROCEDURE HP_EXTSYSBLOB_INSERT(
    TRANS_ID D_STR_20,
    FILE_TYPE STR_20,
    BLOB_TXT D_BLOB_TEXT,
    COMMAND D_STR_CODE,
    BLOB_PHOTO D_BLOB_BINARY)
AS
declare variable db_id d_integer;
declare variable esb_id d_id;
declare variable bls_id d_id;
declare variable caption d_str_100;
declare variable esw_id d_id;
begin
  db_id = f_get_db_id();

  if (db_id in (39801, 41701)) then
  begin
    if ((trans_id is null) or (file_type is null)) then
      exit;

    delete from extsysblob eb
    where eb.esb_trans_id = :trans_id and
          eb.esb_type = :file_type and
          eb.esb_command_name = :command;

    /*
    if (command = 'app_save') then
    begin
      select esw.esw_id
      from extsysweb esw
      where esw.esw_trans_id = :trans_id and
            esw.esw_command = :command
      into esw_id;

      delete from extminiapp ema
      where ema.ema_from_esw_id = :esw_id;

      delete from extsysweb esw
      where esw.esw_trans_id = :trans_id and
            esw.esw_command = :command;
    end */

    insert into extsysblob (esb_trans_id, esb_type, esb_blob_txt, esb_command_name)
    values (:trans_id, :file_type, :blob_txt, :command)
    returning esb_id
    into esb_id;

    if (command = 'person_update') then
    begin
      caption = (case file_type
                   when 'pass_front.jpg' then 'Лицевая сторона паспорта'
                   when 'pass_back.jpg' then 'Обратная сторона паспорта'
                   when 'selfie.jpg' then 'Фото лица клиента'
                 end);

      if (file_type in ('pass_front.jpg', 'pass_back.jpg', 'selfie.jpg')) then
      begin
        execute procedure hp_insert_into_blobstore(:file_type, :caption, :blob_photo, 1000000000000, 2, 3)
        returning_values :bls_id;

        update extsysblob esb
        set esb.esb_from_bls_id = :bls_id
        where esb.esb_id = :esb_id;
      end
    end

    if (blob_photo is not null and command = 'app_save') then
    begin
      caption = 'Согласие супруги/супруга';
      file_type = 'spouse_consent.jpg';
      execute procedure hp_insert_into_blobstore(:file_type, :caption, :blob_photo, 1000000000000, 2, 3)
      returning_values :bls_id;

      update extsysblob esb
      set esb.esb_from_bls_id = :bls_id
      where esb.esb_id = :esb_id;
    end 
  end
  else
  begin
    if ((trans_id is null) or (file_type is null)) then
      exit;

    delete from extsysblob eb
    where eb.esb_trans_id = :trans_id and
          eb.esb_type = :file_type;

    insert into extsysblob (esb_trans_id, esb_type, esb_blob_txt)
    values (:trans_id, :file_type, :blob_txt);
  end
end
