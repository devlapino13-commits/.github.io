CREATE PROCEDURE AAREP_REPORT_FOR_TSB_3_CLOSED(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    K_KONTRACT_NO STR_20,
    C_ID_STR STR_20,
    KC_ID ID_TYPE,
    C_NAME_FIO STR_100,
    C_BIRTH_DATE D_DATE,
    PR_NAME D_STR_CODE,
    K_DISB_DATE D_DATE,
    TERM_IN_DAYS D_INTEGER,
    OKS_KONT_LAST_DATE D_DATE,
    OKS_LAST_SH_DATE D_DATE,
    K_STOP_PERC D_DATE,
    CLOSE_DATE D_DATE,
    K_PERCENT_FLAT_ANNUAL CURR_TYPE,
    PERCENT_YEAR_R CURR_TYPE,
    KD_FLOAT_PERC_LEVEL D_SMALLINT,
    PERCENT_EFFECT CURR_TYPE,
    CRED_FULL_COST CURR_TYPE,
    ACC_NO_DISB STR_10,
    ACC_NO STR_10,
    IS_SECURED D_SMALLINT,
    Z_COST_LIKVID CURR_TYPE,
    OKS_IS_RESCHEDULE_LLR D_SMALLINT,
    LOAN_EMP D_SMALLINT,
    KC_AMOUNT CURR_TYPE,
    DELINQ_DAYS D_INTEGER,
    OKS_ECL_ACC_KIND D_SMALLINT,
    OKS_ECL_PERCENT PERC_TYPE,
    FINAL_PDN CURR_TYPE,
    KC_ANKETA_INCOME D_CURRENCY,
    KC_STAT_AVG_INCOME D_CURRENCY,
    KC_FINAL_INCOME D_CURRENCY,
    U_CLI D_STR_255,
    K_FACT_DISB_ONLINE D_SMALLINT,
    REG_NAME STR_100,
    CLIENT_TYPE D_STR_20)
AS
declare variable K_ID_PRIOR ID_TYPE;
declare variable K_ID ID_TYPE;
declare variable C_ID ID_TYPE;
declare variable PR_ID D_INTEGER;
declare variable PR_ID_PRIOR D_INTEGER;
declare variable MAX_SH_DATE D_DATE;
declare variable KC_NON_RESIDENT D_SMALLINT;
declare variable KNR_NON_RESIDENT D_SMALLINT;
declare variable PR_CODE STR_100;
declare variable PR_LOAN_TYPE D_SMALLINT;
declare variable PR_PRODUCT_KIND D_SMALLINT;
declare variable K_KOLICH_CLIENT_APPROVED D_SMALLINT;
declare variable SAB_ID_ZALOG_COST D_ID;
declare variable OKS_DELINQ_DATE_PRV D_DATE;
declare variable REG_ID D_INTEGER;
declare variable REG_ID_PRIOR D_INTEGER;
declare variable PR_CODE_LOC D_STR_100;
declare variable KC_CLIENT_TYPE D_INTEGER;
declare variable KC_IS_MSP D_SMALLINT;
begin
  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Залог'
  into :sab_id_zalog_cost;

  k_id_prior = 0;
  pr_id_prior = 0;
  reg_id_prior = 0;
  for
  select oks_from_kc_id,
         oks_from_k_id,
         oks_date_begin,
         oks_kont_last_date,
         oks_last_sh_date,
         oks_is_reschedule_llr,
         oks_delinq_date_prv,
         oks_ecl_acc_kind,
         oks_ecl_percent
  from kn_oksummary
  where oks_close_state = 1 -- Закрыт в эту дату
  and oks_date_begin between :rep_date_begin and :rep_date_end
  into kc_id,
       k_id,
       close_date,
       oks_kont_last_date,
       oks_last_sh_date,
       oks_is_reschedule_llr,
       oks_delinq_date_prv,
       oks_ecl_acc_kind,
       oks_ecl_percent
  do
  begin
    delinq_days = coalesce(close_date - (oks_delinq_date_prv + 1),0);
    if (delinq_days < 0) then delinq_days = 0;

    if (exists(select * from operkred  -- если закрыт со списанием или были списании то их не включаем
    where ok_from_k_id = :k_id
    and ok_ol_is_active = 1
    and ok_type = 5
    and ok_os + ok_perc < 0)) then
    begin
      Continue;
    end 

    pr_code_loc = '';
    select p.pr_code
    from kontract k, product p
    where p.pr_id = k.k_from_pr_id and
    k.k_id = :k_id
    into:pr_code_loc;

    if (pr_code containing '.selfemployed.') then
    begin
      client_type = 'СЗ';
    end else
    begin
      select kc_client_type, coalesce(kc_is_msp, 0) from knclient
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      rows 1
      into kc_client_type, kc_is_msp;

      if (kc_client_type = 1) then
      begin
        client_type = 'ЮЛ';
      end else
      if (kc_client_type = 2) then
      begin
        if (kc_is_msp = 1) then client_type = 'ИП-МСП';
          else client_type = 'ИП-неМСП';
      end else
      begin
        client_type = 'ФЛ';
      end
    end

    if (k_id_prior <> k_id) then
    begin
      k_id_prior = k_id;
      select k_kontract_no,
             k_from_pr_id,
             k_disb_date,
             k_stop_perc,
             round(k_percent_per_day * 365.00, 3),
             round(k_percent_effect,3),
             round(k_percent_effect_fifd,3),
             k_kolich_client_approved,
             k_fact_disb_online,
             k_fact_closed_reg_id
      from kontract
      where k_id = :k_id
      into k_kontract_no,
           pr_id,
           k_disb_date,
           k_stop_perc,
           k_percent_flat_annual,
           percent_effect,
           cred_full_cost,
           k_kolich_client_approved,
           k_fact_disb_online,
           reg_id
           ;

      if (pr_id_prior <> pr_id) then
      begin
        pr_id_prior = pr_id;
        select pr_name,
               pr_code,
               pr_loan_type,
               pr_product_kind
        from product
        where pr_id = :pr_id
        into pr_name,
             pr_code,
             pr_loan_type,
             pr_product_kind;

        if (pr_product_kind = 4) then loan_emp = 0;
          else loan_emp = 1;
      end 

      select max(sh_date)
      from schedule
      where sh_from_k_id = :k_id
      into max_sh_date;
      term_in_days = max_sh_date - k_disb_date;

      percent_year_r = null;
      select round(kd_percent_per_day * 365, 3), kd_float_perc_level
      from kontdata
      where :close_date between kd_ol_date and kd_date_end and
              kd_ol_is_delete = 0 and
              kd_from_k_id = :k_id
      into percent_year_r, kd_float_perc_level;

      is_secured = null;
      select is_secured
      from aarep_get_secured_loan(:pr_code, :pr_loan_type, :pr_product_kind, :pr_id, :k_disb_date, :k_kolich_client_approved, :k_id)
      into is_secured;

      u_cli = '';
      select sbl_name from sb_akt, sb_list
      where sba_from_k_id = :k_id
      and :rep_date_end between sba_date_begin and sba_date_end
      and sba_status = 2
      and sba_is_delete = 0
      and sba_from_sbl_id = sbl_id
      into u_cli;
      if (u_cli <> 'ук') then u_cli = '';


      if (reg_id_prior <> reg_id) then
      begin
        reg_id_prior = reg_id;
        select reg_code||' '||reg_name from region
        where reg_unicode = :reg_id
        into reg_name;
      end 

    end

    select kc_from_c_id,
           coalesce(kc_non_resident,0),
           kc_amount_approved
    from knclient
    where kc_id = :kc_id
    into c_id,
         kc_non_resident,
         kc_amount;

    c_id_str = 'n'||c_id;

    select c_name_fio, c_birth_date from clients
    where c_id = :c_id
    into c_name_fio, c_birth_date;

    knr_non_resident = null;
    select knr_non_resident
    from kn_resident
    where knr_from_kc_id = :kc_id
    and :k_disb_date between knr_date_begin and knr_date_end
    into knr_non_resident;
    knr_non_resident = coalesce(knr_non_resident, kc_non_resident);
    acc_no_disb = iif(knr_non_resident = 1,'492','488');

    knr_non_resident = null;
    select knr_non_resident
    from kn_resident
    where knr_from_kc_id = :kc_id
    and :rep_date_end between knr_date_begin and knr_date_end
    into knr_non_resident;
    knr_non_resident = coalesce(knr_non_resident, kc_non_resident);
    acc_no = iif(knr_non_resident = 1,'492','488');

    if (pr_code containing '.zalog_car.' or pr_code containing '.zalog.') then
    begin
      z_cost_likvid = -f_get_acc_amount(:kc_id, :sab_id_zalog_cost, :close_date, :close_date, 'bal');  -- Залоговая стоимоть
    end 

    final_pdn = null;
    select kc_pdn_koef from knclient_pdn
    where kc_id = :kc_id
      and kcp_kind = 0
    into final_pdn;

    kc_anketa_income = null;
    kc_stat_avg_income = null;
    kc_final_income = null;
    if (coalesce(final_pdn, 0) <> 0) then
    begin
      select kc_anketa_income, kc_stat_avg_income, kc_final_income
      from aarep_knclient_one_pdnlist(:kc_id)
      where --:rep_only_summary = 0 or
            rec_no = -1
      into kc_anketa_income, kc_stat_avg_income, kc_final_income;
    end 


    suspend;
  end
end
