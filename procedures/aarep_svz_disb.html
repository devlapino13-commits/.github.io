CREATE PROCEDURE AAREP_SVZ_DISB(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_TYPE SMALLINT)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(60),
    AA_CURR_KC_ID ID_TYPE,
    AA_CURR_KC_FIO VARCHAR(100),
    JS_ID INTEGER,
    DISB_CLI_WEEK INTEGER,
    DISB_CLI_MONTH INTEGER,
    CL_END_CAT1_2 INTEGER,
    AA_AMOUNT_ACTIVE_END CURR_TYPE,
    AA_DELIQ_BALANCE7 CURR_TYPE,
    AA_DELIQ_BALANCE1 CURR_TYPE,
    PAR7 CURR_TYPE,
    PAR1 CURR_TYPE,
    JS_NAME VARCHAR(100),
    CB_PERSONAL_STAGE_MONTH INTEGER,
    PUNCT VARCHAR(10),
    REP_GROUPING VARCHAR(50),
    KC_BALANCE_BEGIN CURR_TYPE,
    CB_PORTFEL_C CURR_TYPE,
    CB_PORTFEL_D CURR_TYPE,
    PAR7_BONUS CURR_TYPE)
AS
declare variable cb_personal_stage date;
declare variable k integer;
declare variable t integer;
declare variable week_day smallint;
declare variable rep_date_begin_week date;
declare variable rep_date_begin_month date;
declare variable js_code varchar(100);
begin
  if (rep_type is null) then
    rep_type = 0;
  week_day = extract(weekday from :rep_date_end) - 1;
  if (week_day < 0) then
    week_day = 6;
  rep_date_begin_week = :rep_date_end - week_day;
  rep_date_begin_month = :rep_date_end - extract(day from :rep_date_end) + 1;
-- Если неделя попадает в переходящий месяц то начало недели будет начало месяца т.е. отчет с начало месяца
  if (extract(month from :rep_date_begin_week) <> extract(month from :rep_date_end)) then
    rep_date_begin_week = rep_date_begin_month;
  k = 0;
  t = 1;
  for select aa_curr_office_name, aa_curr_kc_id, aa_curr_kc_fio,
             sum(iif(dd_disb_date between :rep_date_begin_week and :rep_date_end, aa_just_cl_disb, 0)),
             sum(aa_just_cl_disb), sum(iif(aa_category_id in (1, 2), aa_just_cl_end, 0)),
             sum(iif(aa_category_id = 1, aa_amount_active_end, 0)), sum(iif(aa_category_id = 1, aa_deliq_balance7, 0)),
             sum(iif(aa_category_id = 1 and aa_deliq_days > 0, aa_deliq_balance, 0))
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin_month, :rep_date_end, '', 0, 1, 0, 1, 0, 0, 0)
      where js_code <> 'JobSystem.Specialist_SB'
      group by aa_curr_office_name, aa_curr_kc_id, aa_curr_kc_fio
      order by 1, 6
      into aa_curr_office_name, aa_curr_kc_id, aa_curr_kc_fio, disb_cli_week, disb_cli_month, cl_end_cat1_2,
           aa_amount_active_end, aa_deliq_balance7, aa_deliq_balance1
  do
  begin
    rep_grouping = case
                     when disb_cli_month = 0 then '1. не выдавал'
                     when disb_cli_month < 17 then '2. менее 17 выданных клиентов'
                     when disb_cli_month between 17 and 24 then '3. от 17 по 24 выданных клиента'
                     when disb_cli_month > 24 then '4. от 25 и более выданных клиента'
                   end;
    js_name = 'СВЗ';
    select ps_date_begin
    from hr_pstatus --, hr_orders , joblist
    where ps_from_c_id = :aa_curr_kc_id and
          ps_date_begin is not null and
          ps_status = 1
    order by ps_date_begin
    rows 1
    into :cb_personal_stage;
    if (cb_personal_stage is null) then
      cb_personal_stage = :rep_date_end;
    cb_personal_stage_month = datediff(day, :cb_personal_stage, :rep_date_end) / 30;
    par7 = 0;
    par1 = 0;
    if (aa_amount_active_end > 0) then
    begin
      par7 = aa_deliq_balance7 / aa_amount_active_end * 100;
      par1 = aa_deliq_balance1 / aa_amount_active_end * 100;
    end
    if (rep_type = 0) then
    begin
      par7_bonus = 0;
--    Портфель за предыдущий месяц
      kc_balance_begin = null;
      select sum(oks_balance)
      from oksummary
      where (:rep_date_begin_month - 1) between oks_date_begin and oks_date_end and
            oks_kc_id = :aa_curr_kc_id and
            oks_close_state = 0 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, (:rep_date_begin_month - 1), null, null, null, null, null, null)) = 1
      into kc_balance_begin;
      if (kc_balance_begin is null) then
        kc_balance_begin = 0;
-- С - Портфель займов, переданный в категорию 1 от другого СВЗ в течение отчетного месяца
      cb_portfel_c = null;
      select sum(iif((select aa_category_id
                      from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 1,
             (select oks_balance
              from oksummary
              where oks_from_k_id = k_id and
                    oks_close_state = 0 and
                    :rep_date_end between oks_date_begin and oks_date_end), 0))
      from kontstate, kontract, knaktlist kal1, knaktlist kal2
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal1.kal_from_kf_id and
            kal1.is_delete = 0 and
            :rep_date_end between kal1.kal_date and kal1.kal_date_end and
            kal1.kal_new_kf_officer = :aa_curr_kc_id and
            k_from_kf_id = kal2.kal_from_kf_id and
            kal2.is_delete = 0 and
            :rep_date_begin_month between kal2.kal_date and kal2.kal_date_end and
            kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer and
            ks_date_begin < kal1.kal_date -- после передачи выданные группы не включает
      into cb_portfel_c;
      if (cb_portfel_c is null) then
        cb_portfel_c = 0;

-- D - Портфель займов, переданный в категорию 1 от данного СВЗ другому СВЗ в течение отчетного месяца
      cb_portfel_d = null;
      select sum(iif((select aa_category_id
                      from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 1,
             (select oks_balance
              from oksummary
              where oks_from_k_id = k_id and
                    oks_close_state = 0 and
                    :rep_date_end between oks_date_begin and oks_date_end), 0))
      from kontstate, kontract, knaktlist kal1, knaktlist kal2
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal1.kal_from_kf_id and
            kal1.is_delete = 0 and
            :rep_date_end between kal1.kal_date and kal1.kal_date_end and
            k_from_kf_id = kal2.kal_from_kf_id and
            kal2.is_delete = 0 and
            :rep_date_begin_month between kal2.kal_date and kal2.kal_date_end and
            kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer and
            kal2.kal_new_kf_officer = :aa_curr_kc_id and
            ks_date_begin < kal1.kal_date -- после передачи выданные группы не включает
      into cb_portfel_d;
      if (cb_portfel_d is null) then
        cb_portfel_d = 0;
-- По положению от 27.02.2013
      if (aa_deliq_balance7 = 0) then
      begin
        par7_bonus = 0;
      end
      else
      begin
        if (kc_balance_begin + cb_portfel_c - cb_portfel_d > 0) then
        begin
          par7_bonus = aa_deliq_balance7 / (kc_balance_begin + cb_portfel_c - cb_portfel_d) * 100;
        end
        else
        begin
          par7_bonus = 100;
        end
      end

      suspend;
    end
    else
    if (aa_curr_kc_id <> 1000000000000) then -- не включает виртуального и СБ
    begin
      k = k + 1;
      punct = 'Пункт ' || t;
      suspend;
      if (k > 7) then
      begin
        k = 0;
        t = t + 1;
        aa_curr_office_name = null;
        aa_curr_kc_id = null;
        aa_curr_kc_fio = null;
        js_id = null;
        disb_cli_week = null;
        disb_cli_month = null;
        cl_end_cat1_2 = null;
        aa_amount_active_end = null;
        aa_deliq_balance7 = null;
        aa_deliq_balance1 = null;
        par7 = null;
        par1 = null;
        js_name = null;
        cb_personal_stage_month = null;
        punct = null;
        suspend;
      end
    end
  end

end
