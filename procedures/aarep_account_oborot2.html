CREATE PROCEDURE AAREP_ACCOUNT_OBOROT2(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    ACCOUNT VARCHAR(20),
    ACCOUNT_NAME VARCHAR(255),
    ACCOUNT_NUMBER INTEGER,
    ACC_BALANCE VARCHAR(255),
    BALANCE_NAME VARCHAR(255),
    ACC_GROUP VARCHAR(255),
    ACC_CLASS VARCHAR(255),
    CURR_CODE VARCHAR(3),
    OB_DT DECIMAL(18,4),
    OB_KT DECIMAL(18,4),
    LAST_OPER_DATE DATE,
    ACC_GROUP_NO VARCHAR(20),
    ACC_KIND INTEGER,
    ACC_KIND_NAME VARCHAR(255),
    ACCOUNT_TYPE INTEGER,
    CURR_INFO VARCHAR(3),
    OB_DT_CURR DECIMAL(18,4),
    OB_KT_CURR DECIMAL(18,4))
AS
declare variable acc_kind_temp integer;
begin
  for select account, al_name, al_number, al_from_bg_id, b1.bg_name, b2.bg_name, b2.parent, currency, sum(ob_dt),
             sum(ob_kt), sum(ob_dt_curr), sum(ob_kt_curr), max(max_date), b2.bg_id, b2.bg_kind, al_type
      from aarep_account_oborot_temp(:report_region_id, :report_date_begin, :report_date_end), acc_list, balgroup b1, balgroup b2
      where account = al_id and
            al_from_bg_id = b1.bg_id and
            b1.parent = b2.bg_id and
            ((currency = :report_currency) or
            (:report_currency = '---') or
            (:report_currency = '000'))
      group by account, al_name, al_number, al_from_bg_id, b1.bg_name, b2.bg_name, b2.parent, currency, b2.bg_id, b2.bg_kind, al_type
      into :account, :account_name, :account_number, :acc_balance, :balance_name, :acc_group, :acc_class, :curr_code,
           :ob_dt, :ob_kt, :ob_dt_curr, :ob_kt_curr, :last_oper_date, :acc_group_no, :acc_kind_temp, :account_type
  do
  begin
    curr_info = curr_code;
    if (report_currency <> '000') then
    begin
      ob_dt = :ob_dt_curr;
      ob_kt = :ob_kt_curr;
    end

    acc_kind = 0;
    acc_kind_name = 'Не указан';
    if (acc_kind_temp = 1) then
    begin
      acc_kind = 1;
      acc_kind_name = 'Активы';
    end
    if ((acc_kind_temp = 2) or (acc_kind_temp = 3)) then
    begin
      acc_kind = 2;
      acc_kind_name = 'Обязательства и капитал';
    end
    if ((acc_kind_temp = 4) or (acc_kind_temp = 5)) then
    begin
      acc_kind = 3;
      acc_kind_name = 'Доходы и расходы';
    end
    if (report_currency = '000') then
    begin
      curr_code = '000';
    end
    suspend;
  end
end
