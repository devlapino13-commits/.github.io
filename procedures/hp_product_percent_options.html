CREATE PROCEDURE HP_PRODUCT_PERCENT_OPTIONS(
    PR_ID D_INTEGER,
    PRL_ID D_INTEGER,
    OPTION_CODE D_STR_CODE NOT NULL,
    UN_ID D_ID,
    IMPORT_FROM_PR_ID D_INTEGER = 0,
    IMPORT_FROM_PRL_ID D_INTEGER = 0,
    PRR_ID D_ID = 0,
    PRR_MIN_POINT D_INTEGER = 0,
    PRR_MAX_POINT D_INTEGER = 0,
    PRR_MIN_PERIOD D_INTEGER = 0,
    PRR_MAX_PERIOD D_INTEGER = 0,
    PRR_PERCENT D_CURRENCY = 0,
    PRR_PERCENT_2 D_CURRENCY = 0,
    PRR_PERCENT_7 D_CURRENCY = 0)
AS
declare variable tmp_prr_min_point d_integer;
declare variable tmp_prr_max_point d_integer;
declare variable tmp_prr_min_period d_integer;
declare variable tmp_prr_max_period d_integer;
declare variable tmp_prr_percent d_currency;
declare variable tmp_prr_percent_2 d_currency;
declare variable tmp_prr_percent_7 d_currency;
begin
  -- RFCKG-3640 Команды для модуля «Процентная ставка»
  if(option_code = 'update') then
  begin
    update product_rates
    set prr_min_point = :prr_min_point,
           prr_max_point = :prr_max_point,
           prr_min_period = :prr_min_period,
           prr_max_period = :prr_max_period,
           prr_percent = :prr_percent,
           prr_percent_2 = :prr_percent_2,
           prr_percent_7 = :prr_percent_7,
           prr_from_un_id = :un_id
    where prr_id = :prr_id;
  end

  if(option_code = 'delete') then
  begin
    update product_rates
    set prr_from_un_id = :un_id,
        prr_is_actual = 0,
        prr_enum_status = 1
    where (prr_id = :prr_id);
  end

  if(option_code = 'deleteAllForImportFromXLS') then   -- RFCKG-4737
  begin
    --Удаляем старые записи
    for select prr_id
    from product_rates prr
    where prr.prr_from_pr_id = :pr_id
      and prr.prr_from_prl_id = :prl_id
      and prr_is_actual = 1
    into prr_id
    do 
    begin
      update product_rates
      set prr_from_un_id = :un_id,
          prr_is_actual = 0,
          prr_enum_status = 4
      where (prr_id = :prr_id);
    end
  end

  if(option_code = 'addOneForImportFromXLS') then  -- RFCKG-4737
  begin
    insert into product_rates (prr_from_pr_id, prr_from_prl_id, prr_min_point, prr_max_point, prr_min_period, prr_max_period, prr_percent, prr_percent_2, prr_percent_7, prr_is_actual, prr_from_un_id)
    values (:pr_id, :prl_id, :prr_min_point, :prr_max_point, :prr_min_period, :prr_max_period, :prr_percent, :prr_percent_2, :prr_percent_7, 1, :un_id);
  end

  if(option_code = 'add') then
  begin
    insert into product_rates (prr_from_pr_id, prr_from_prl_id, prr_min_point, prr_max_point, prr_min_period, prr_max_period, prr_percent, prr_percent_2, prr_percent_7, prr_is_actual, prr_from_un_id)
    values (:pr_id, :prl_id, 0, 0, 0, 0, 0, 0, 0, 1, :un_id);
  end

  if(option_code = 'import') then
  begin
    --Удаляем старые записи
    for select prr_id
    from product_rates prr
    where prr.prr_from_pr_id = :pr_id
      and prr.prr_from_prl_id = :prl_id
      and prr_is_actual = 1
    into prr_id
    do 
    begin
      update product_rates
      set prr_from_un_id = :un_id,
          prr_is_actual = 0,
          prr_enum_status = 3
      where (prr_id = :prr_id);
    end

    -- Добавляем записи в таблицу
    for select prr.prr_min_point,
               prr.prr_max_point,
               prr.prr_min_period,
               prr.prr_max_period,
               prr.prr_percent,
               prr.prr_percent_2,
               prr.prr_percent_7
    from product_rates prr
    where prr.prr_from_pr_id = :import_from_pr_id
    and prr.prr_from_prl_id = :import_from_prl_id  -- RFCKG-4119
    and prr_is_actual = 1
    into tmp_prr_min_point,
         tmp_prr_max_point,
         tmp_prr_min_period,
         tmp_prr_max_period,
         tmp_prr_percent,
         tmp_prr_percent_2,
         tmp_prr_percent_7
    do
    begin
      insert into product_rates (prr_from_pr_id, prr_from_prl_id, prr_min_point, prr_max_point, prr_min_period, prr_max_period, prr_percent, prr_percent_2, prr_percent_7, prr_is_actual, prr_from_un_id)
      values (:pr_id, :prl_id, :tmp_prr_min_point, :tmp_prr_max_point, :tmp_prr_min_period, :tmp_prr_max_period, :tmp_prr_percent, :tmp_prr_percent_2, :tmp_prr_percent_7, 1, :un_id);
    end
  end
end
