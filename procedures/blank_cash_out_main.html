CREATE PROCEDURE BLANK_CASH_OUT_MAIN(
    OL DECIMAL(18,0))
RETURNS (
    BL_DATE DATE,
    BL_DOC_NO VARCHAR(20),
    BL_MEMO VARCHAR(1024),
    BL_CLIENT_NAME VARCHAR(255),
    BL_PASS_NO VARCHAR(30),
    BL_PASS_DATE DATE,
    BL_PASS_WHO VARCHAR(255),
    BL_CLIENT_REGIST_NO VARCHAR(20),
    BL_CLIENT_OKPO VARCHAR(20),
    BL_CLIENT_INN VARCHAR(20),
    BL_OFFICE_FULL_NAME VARCHAR(255),
    BL_OFFICE_BANK_NAME VARCHAR(255),
    BL_SUMMA DECIMAL(18,4),
    BL_ACCOUNT_DT VARCHAR(20),
    BL_ACCOUNT_KT VARCHAR(20),
    BL_IS_DELETE SMALLINT,
    BL_KASS_PLAN VARCHAR(10),
    BL_CORR_ACC VARCHAR(255),
    BL_COMPANY VARCHAR(255),
    BL_OFFICE_NAME VARCHAR(255),
    BL_BALANCE_ID VARCHAR(20),
    BL_REGION_CODE VARCHAR(10),
    BL_ACCOUNT_NO INTEGER,
    BL_BRANCH_FULL_NAME VARCHAR(255),
    BL_DATE_DAY INTEGER,
    BL_DATE_MONTH INTEGER,
    BL_DATE_YEAR INTEGER,
    BL_KONTRACT_NO VARCHAR(20),
    BL_KONTRACT_DATE DATE,
    BL_C_DOC_NAME VARCHAR(50),
    BL_UN_FIO VARCHAR(100),
    BL_MANAGER_FIO VARCHAR(100),
    BL_CREDIT_ACC VARCHAR(20),
    ACC_DT D_STR_CODE,
    ACC_KT D_STR_CODE)
AS
declare variable OFFICE varchar(255);
declare variable COMPANY_NAME varchar(255);
declare variable BL_BANK_NAME varchar(255);
declare variable BL_BANK_ID varchar(10);
declare variable ST_STAFF integer;
declare variable ST_CLIENT integer;
declare variable OL_TYPE integer;
declare variable OL_SUBTYPE integer;
declare variable DATABASE_ID integer;
declare variable SUB_NO integer;
declare variable DOC_NO integer;
declare variable OL_DATE date;
declare variable REG_CODE varchar(20);
declare variable C_DOC_TYPE integer;
declare variable OL_MANAGER_C_ID decimal(18,0);
declare variable BL_BALANCE_ID_TMP varchar(20);
declare variable BL_CREDIT_ACC_TMP varchar(20);
declare variable C_IS_RESIDENT smallint;
declare variable REG_ID integer;
declare variable TMP_ID D_ID;
declare variable RES_KIND D_STR_CODE;
declare variable OL_DATE_REP D_DATE;
declare variable ATL_ACC_MASK D_STR_20;
declare variable OB_SUMMA D_CURRENCY;

begin
  select ol_date
  from operlist
  where ol_id = :ol
  into :ol_date_rep;

--  for select res_kind
--      from hp_do_moneyoper_import_process(:ol_date_rep, :ol_date_rep, 0, -- :need_clear_old_list,
--      1, -- :need_fill_money_oper
--      0, -- :need_cash,
--      0, -- :need_bank,
--      0) -- :need_1c
--      into :res_kind
--  do
--  begin
--    res_kind = '';
--  end
--
--  for select res_kind
--      from hp_do_moneyoper_import_process(:ol_date_rep, :ol_date_rep, 0, -- :need_clear_old_list,
--      0, -- :need_fill_money_oper
--      1, -- :need_cash,
--      0, -- :need_bank,
--      0) -- :need_1c
--      into :res_kind
--  do
--  begin
--    res_kind = '';
--  end
  -- если передан НЕ OL_ID, возможно это ID из рефинансирования. OL_ID берем оттуда
  if (not exists(select *
                 from operlist
                 where ol_id = :ol)) then
  begin
    select krd_from_ol_id
    from kontrefund_disb
    where krd_id = :ol
    into :tmp_id;

    if (tmp_id is not null) then
      ol = tmp_id;
  end

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Для AZ РКО при выдаче кредита печатается для каждого клиента */

  :database_id = f_get_db_id();
  select ol_oper_type, ol_oper_subtype, ol_date, reg_code, reg_unicode, un_fio, ol_manager_c_id
  from operlist, region, username
  where ol_id = :ol and
        ol_from_reg_id = reg_unicode and
        ol_user_id = un_id
  into :ol_type, :ol_subtype, :ol_date, :reg_code, :reg_id, bl_un_fio, ol_manager_c_id;
  bl_manager_fio = '';
  if (ol_manager_c_id is not null) then
  begin
    select c_name_fio
    from clients
    where c_id = :ol_manager_c_id
    into :bl_manager_fio;
  end

  if ((database_id = 94401 and
      ol_type = 1 and
      ol_subtype = 1) or (database_id = 81001 and
      ol_type = 1 and
      ol_subtype = 1 and
      ol_date >= '01.01.2014')) -- в РФ с 2014 года печатаем расходник на каждого
  then
  begin
    if (database_id = 81001) then
      bl_credit_acc = '50.01';

    select k_kontract_no, k_disb_date
    from kontract, operkred
    where ok_from_ol_id = :ol and
          ok_type = 8 and
          ok_ol_is_active = 1 and
          ok_from_k_id = k_id and
          ok_from_c_id = 0 rows 1
    into :bl_kontract_no, :bl_kontract_date;
    sub_no = 1;
    for select ol_date, ol_doc_no, ol_memo, cm_name, reg_name, b_name, br_from_b_id, operlist.is_delete, b_corr_acc,
               reg_kont_company
        from company, region
        left join(bankregion left join banks on (br_from_b_id = b_id)) on (br_from_reg_id = reg_unicode and
              br_is_main = 1), operlist
        where ol_id = :ol and
              reg_unicode = ol_from_reg_id rows 1
        into :bl_date, :doc_no, :bl_memo, :company_name, :office, :bl_bank_name, :bl_bank_id, bl_is_delete,
             :bl_corr_acc, :bl_branch_full_name
    do
    begin
      bl_office_bank_name = bl_bank_id || ' ' || bl_bank_name;
      bl_office_full_name = company_name || ' ' || office;
      bl_company = company_name;
      bl_office_name = :office;
      for select ob_account_dt, ob_account_kt, ob_from_kp_id, a_dt.al_from_bg_id, reg_code, a_dt.al_number
          from operbook, acc_list a_kt, acctypelist, operlist, region, acc_list a_dt
          where ob_from_ol_id = :ol and
                ol_id = :ol and
                a_kt.al_id = ob_account_kt and
                a_dt.al_id = ob_account_dt and
                a_kt.al_from_bg_id like atl_acc_mask || '%' and
                atl_from_at_id in (11, 16) and
                ol_from_reg_id = reg_unicode
          into :bl_account_dt, :bl_account_kt, :bl_kass_plan, :bl_balance_id, :bl_region_code, :bl_account_no
      do
      begin
        if (database_id = 81001 and
            ol_date >= '01.01.2014') then
        begin
          bl_account_dt = replace(substring(bl_account_dt from 8 for 5), '-', '.');
          bl_account_kt = replace(substring(bl_account_kt from 8 for 5), '-', '.');
          bl_balance_id = replace(bl_balance_id, '-', '.');

          bl_balance_id_tmp = bl_balance_id;
          bl_credit_acc_tmp = bl_credit_acc;
        end
        for select kc_amount_approved, c_name_fio, c_pass_no, c_pass_date, c_pass_who, c_reg_no, c_okpo, c_inn,
                   c_doc_type, iif(kc_non_resident = 1, 0, 1)
            from operkred, knclient, clients cl
            where ok_from_ol_id = :ol and
                  ok_from_k_id = kc_from_k_id and
                  kc_from_c_id = c_id and
                  kc_amount_approved > 0 and
                  ok_type = 8 and
                  ok_from_c_id = 0
            into :bl_summa, :bl_client_name, :bl_pass_no, :bl_pass_date, :bl_pass_who, :bl_client_regist_no,
                 :bl_client_okpo, :bl_client_inn, :c_doc_type, :c_is_resident
        do
        begin
          bl_summa = coalesce(bl_summa, 0);
          if (bl_kass_plan is null) then
            bl_kass_plan = '';
          bl_date_day = extract(day from :bl_date);
          bl_date_month = extract(month from :bl_date);
          bl_date_year = extract(year from :bl_date);
          if (database_id = 94401) then
            bl_doc_no = doc_no || '/' || sub_no;
          else
            bl_doc_no = reg_code || '-' || right('000000' || doc_no, 6) || '/' || sub_no;
          if (database_id = 81001) then
          begin
            c_doc_type = coalesce(c_doc_type, 0);
            bl_c_doc_name = case
                              when c_doc_type in (12, 13) then 'Паспорт РФ'
                              when c_doc_type = 39 then 'Вид на жительство'
                              else 'Паспорт иностранного гражданина'
                            end;
          end

          if (database_id = 81001 and
              ol_date >= '01.01.2018') then
          begin
            bl_balance_id = bl_balance_id_tmp;
            bl_credit_acc = bl_credit_acc_tmp;

            if (c_is_resident = 1) then
            begin
              bl_balance_id = (case bl_balance_id
                                 when '58.03' then '48801'
                                 when '50.01' then '20202'
                                 when '76.09' then '48809'
                                 when '76.02' then '60323'
                                 when '76.06' then '47422'
                                 when '98.01' then '48809'
                                 when '51.01' then '20501'
                                 when '57.04' then '20501'
                                 else bl_balance_id
                               end);
            end
            else
            begin
              bl_balance_id = (case bl_balance_id
                                 when '58.03' then '49201'
                                 when '50.01' then '20202'
                                 when '76.09' then '49209'
                                 when '76.02' then '60323'
                                 when '76.06' then '47422'
                                 when '98.01' then '49209'
                                 when '51.01' then '20501'
                                 when '57.04' then '20501'
                                 else bl_balance_id
                               end);
            end
            if (bl_balance_id starting '70.') then
              bl_balance_id = '60306';

            if (bl_credit_acc = '50.01') then
              bl_credit_acc = '20202';

            if (bl_credit_acc starting '70.') then
              bl_credit_acc = '60306';
          end
          acc_dt = null;
          acc_kt = null;
          select dt.acc_code, kt.acc_code
          from kn_operbook kob
          inner join ref_account dt on dt.acc_id = kob.kob_acc_dt
          inner join ref_account kt on kt.acc_id = kob.kob_acc_kt
          inner join sys_accbalance sab on sab.sab_id = kt.acc_from_sab_id
          where kob_from_ol_id = :ol and
                sab.sab_full_code = 'Деньги.Нал.Касса'
          into :acc_dt, acc_kt;

          suspend;
          sub_no = sub_no + 1;
        end
      end
    end
  end
  else
  begin
    if (database_id = 81001) then
      bl_credit_acc = '50.01';

    for select ol_date, ol_doc_no, ol_memo, c_name_fio, c_pass_no, c_pass_date, c_pass_who, c_reg_no, c_okpo, c_inn,
               cm_name, reg_name, b_name, br_from_b_id, operlist.is_delete, b_corr_acc, reg_kont_company, c_doc_type
        from company, region
        left join(bankregion left join banks on (br_from_b_id = b_id)) on (br_from_reg_id = reg_unicode and
              br_is_main = 1), operlist
        left join clients on (ol_document_client = c_id)
        where ol_id = :ol and
              reg_unicode = ol_from_reg_id rows 1
        into :bl_date, :bl_doc_no, :bl_memo, :bl_client_name, :bl_pass_no, :bl_pass_date, :bl_pass_who,
             :bl_client_regist_no, :bl_client_okpo, :bl_client_inn, :company_name, :office, :bl_bank_name, :bl_bank_id,
             bl_is_delete, :bl_corr_acc, :bl_branch_full_name, :c_doc_type
    do
    begin
      bl_office_bank_name = bl_bank_id || ' ' || bl_bank_name;
      bl_office_full_name = company_name || ' ' || office;
      bl_company = company_name;
      bl_office_name = :office;
      for select ob_account_dt, ob_account_kt, ob_from_kp_id, sum(ob_summa_val), a_dt.al_from_bg_id, reg_code,
                 a_dt.al_number, min(atl_acc_mask) atl_acc_mask
          from operbook, acc_list a_kt, acctypelist, operlist, region, acc_list a_dt
          where ob_from_ol_id = :ol and
                ol_id = :ol and
                a_kt.al_id = ob_account_kt and
                a_dt.al_id = ob_account_dt and
                a_kt.al_from_bg_id like atl_acc_mask || '%' and
                atl_from_at_id in (11, 16) and
                ol_from_reg_id = reg_unicode
          group by ob_account_dt, ob_account_kt, ob_from_kp_id, a_dt.al_from_bg_id, reg_code, a_dt.al_number
          union
          select ob_account_dt, ob_account_kt, ob_from_kp_id, sum(ob_summa_val), al_from_bg_id, reg_code, al_number, null atl_acc_mask
          from operbook, acc_list, region
          where ob_from_ol_id = :ol and
                ob_account_kt = al_id and
                al_from_bg_id = '99287' and
                al_from_reg_id = :reg_id and
                al_from_reg_id = reg_unicode and
                :database_id = 94401
          group by ob_account_dt, ob_account_kt, ob_from_kp_id, al_from_bg_id, reg_code, al_number
          into :bl_account_dt, :bl_account_kt, :bl_kass_plan, :bl_summa, :bl_balance_id, :bl_region_code,
               :bl_account_no, :atl_acc_mask
      do
      begin
        if (database_id = 81001 and
            ol_date >= '01.01.2014') then
        begin
          bl_account_dt = replace(substring(bl_account_dt from 8 for 5), '-', '.');
          bl_account_kt = replace(substring(bl_account_kt from 8 for 5), '-', '.');
          c_doc_type = coalesce(c_doc_type, 0);
          bl_c_doc_name = case
                            when c_doc_type in (12, 13) then 'Паспорт РФ'
                            when c_doc_type = 39 then 'Вид на жительство'
                            else 'Паспорт иностранного гражданина'
                          end;
        end
        if (bl_summa is null) then
          bl_summa = 0;
        if (bl_kass_plan is null) then
          bl_kass_plan = '';
        bl_date_day = extract(day from :bl_date);
        bl_date_month = extract(month from :bl_date);
        bl_date_year = extract(year from :bl_date);

        if (database_id = 81001 and
            ol_date >= '01.01.2018') then
        begin
          if (bl_credit_acc = '50.01') then
            bl_credit_acc = '20202';

          if (bl_balance_id = '57-01') then
            bl_balance_id = '20209';
          if (bl_balance_id = '94-01') then
            bl_balance_id = '60308';
          if (bl_balance_id = '76-06') then
            bl_balance_id = '47422';
          if (bl_balance_id = '51-01') then
            bl_balance_id = '20501';
          if (bl_balance_id = '57-04') then
            bl_balance_id = '20501';

          if (bl_balance_id starting '70-') then
            bl_balance_id = '60306';
        end
        acc_dt = null;
        acc_kt = null;
        select dt.acc_code, kt.acc_code
        from kn_operbook kob
        inner join ref_account dt on dt.acc_id = kob.kob_acc_dt
        inner join ref_account kt on kt.acc_id = kob.kob_acc_kt
        inner join sys_accbalance sab on sab.sab_id = kt.acc_from_sab_id
        where kob_from_ol_id = :ol and
              sab.sab_full_code = 'Деньги.Нал.Касса'
        into :acc_dt, acc_kt;

        if (database_id = 94401 and atl_acc_mask = 10030) then
        begin
          select sum(ob_summa) from operbook
          where ob_from_ol_id = :ol
          into ob_summa;

          if (bl_summa > ob_summa) then bl_summa = ob_summa;
        end

        suspend;
      end
    end
  end
end
