CREATE PROCEDURE HP_PRINT_NO_DEBT_KZ(
    K_ID D_ID)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    CERT_DATE D_STR_CODE,
    CERT_DATE_HEADER D_STR_CODE,
    K_KONTRACT_NO D_STR_20,
    K_KONTRACT_DATE D_STR_CODE)
AS
declare variable total_debt d_currency;
declare variable has_extra_debt d_bool;
begin
    -- RFCKZ-627: Проверка перед формированием справки об отсутствии задолженности
    total_debt = 0;

    -- Если у клиента есть задолженность перед компанией, справка не формируется
    select sum(zad_os + zad_perc + sh_calc_perc + sh_calc_fine)
    from aarep_make_zad_list(:k_id, current_date, 1)
    into :total_debt;

    if (exists(select *
               from hp_repay_extra_saldo(:k_id, 0, current_date)
               where res_amount_saldo > 0)
    ) then
        has_extra_debt = 1;

    if (total_debt > 0 or has_extra_debt = 1) then
      exception e_message 'Для получения справки необходимо погасить задолженность';

    cert_date = f_datetostr(current_date);
    cert_date_header = cert_date;

    select c.c_name_fio,
           c.c_inn,
           k.k_kontract_no,
           f_datetostr(k.k_kontract_date)
    from kontract k
    inner join knclient kc on kc.kc_from_k_id = k.k_id
    inner join clients c on kc.kc_from_c_id = c.c_id
    where k.k_id = :k_id and
          kc.kc_amount_approved > 0
    rows 1
    into :c_name_fio,
         :c_inn, 
         :k_kontract_no,
         :k_kontract_date;

    suspend;
end
