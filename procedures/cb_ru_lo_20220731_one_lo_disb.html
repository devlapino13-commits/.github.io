begin

  -- Результат: количество рабочих дней в месяце
  total_days = 0;
  -- Пока не дошли до конца месяца повторяем цикл
  dd = rep_date_begin;
  while (dd <= rep_date_end) do
  begin
    total_days = total_days + (1 - f_date_is_weekend(dd, 0)); -- функция возвращает 0 если рабочий день, 1 если это выходной
    dd = dateadd(day, 1, dd); -- Переходим на следующий день
  end

  reg_id_prior = 0;
  for select ps_from_c_id,
             (select cont_value
              from hp_contdata_get_email_valid(0, c_id)), ps_from_reg_id
      from hr_pstatus, clients, joblist, jobkind, region_list(:rep_region_id, 1, -1, 1)
      where :rep_c_id = 0 and

            ps_date_begin <= :rep_date_end and

    if (reg_id_prior <> reg_id) then

    aarep_bonus_os = 0;
    aarep_bonus_os_adv = 0;

      aarep_fact_adv_os = 0;

      aarep_fact_adv_perc = 0;
    if (aarep_oklad is null) then
      aarep_oklad = 0;

    aarep_returnment_koef = 1.0; --конэфициент факт делим на план
    --    if (aarep_plan_all <> 0) then
    --      aarep_returnment_koef = round(aarep_fact_planned / aarep_plan_all, 4);
    --    else

    /*
     [17:36, 11.08.2022] Ирода Ахунова: я в тех.задании писала по возвратности формулу, если плановые погашения=0, то 0, надо 0 на 1 исправить
     [17:36, 11.08.2022] Ирода Ахунова: если нет плановых погашений, 100:

    if (aarep_returnment_koef > 1.0) then
      aarep_returnment_koef = 1.0;

    if (rep_date_begin < cast('01.01.2024' as date)) then
    begin
      fine_returnment = case
                          when aarep_returnment_koef < 0.9 then 5000
                          when aarep_returnment_koef < 0.93 then 3000
                          when aarep_returnment_koef < 0.95 then 1500
                          else 0
                        end;
    end
    else
    if (rep_date_begin < cast('01.11.2024' as date)) then
    begin
      fine_returnment = case
                          when aarep_returnment_koef < 0.9 then 5000
                          when aarep_returnment_koef < 0.93 then 2500
                          when aarep_returnment_koef < 0.95 then 1000
                          else 0
                        end;
    end
    else
    if (rep_date_begin < cast('01.02.2026' as date)) then
    begin
      fine_returnment = case
                          when aarep_returnment_koef < 0.9 then 5000
                          when aarep_returnment_koef < 0.93 then 3000
                          when aarep_returnment_koef < 0.95 then 1500
                          else 0
                        end;
    end
    else
    begin
      fine_returnment = case
                          when aarep_returnment_koef < 0.89 then 5000
                          when aarep_returnment_koef < 0.91 then 3000
                          when aarep_returnment_koef < 0.92 then 1500
                          else 0
                        end;
    end

    koef_penalty = case
                     when aarep_returnment_koef < 0.92 then 0.33
                     when aarep_returnment_koef < 0.94 then 0.3
                     else 0.27
                   end;

    aarep_bonus_plus = round((aarep_fact_all * 0.17) + (aarep_fact_delinq_os * 0.09), 0);

    aarep_bonus_ideal = round(aarep_month_plan_perc * 0.17, 0);

    aarep_bonus_minus = round(((aarep_delinq_os + aarep_delinq_perc) * koef_penalty), 0);

    select jk_system_code
    from hr_pstatus h, joblist, jobkind
    where :rep_date_end between ps_date_begin and ps_date_end and
          h.ps_from_c_id = :c_id and
          ps_is_working = 1 and
          ps_status = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id
    into :jk_system_code;

    select lo_stage_month_full
    from hp_get_lo_stage_full(:c_id, :rep_date_end, :jk_system_code)
    into :lo_stage_month_full;

    if (coalesce(lo_stage_month_full, -1) < 0) then
      lo_stage_month_full = 0;

    aarep_bonus_minus_disb = 0;
    kol_disb_kont = 0;

    sfb_fix_bonus = 0;
    sfb_total_work = 0;
    sfb_fact_work = 0;
    sfb_fact_work_eff = 0;

    -- фиксированный бонус будет брасться из таблицы SA_FIXBONUS
    select coalesce(sfb_fix_bonus, 0), coalesce(sfb_total_work, 0), coalesce(sfb_fact_work, 0)
    from sa_fixbonus s
    where sfb_period = :rep_period_end and
          sfb_from_c_id = :c_id
    into :sfb_fix_bonus, sfb_total_work, sfb_fact_work;

    -- добавлено по письму Ироды от 12.4.21 "Поправка в расчет бонусов в базе"
    aarep_bonus_constant = sfb_fix_bonus;

    if ( --sfb_total_work > sfb_fact_work and
        sfb_fact_work > 0) then
      sfb_fact_work_eff = round(sfb_fact_work / cast(sfb_total_work as d_currency) * cast(total_days as d_currency), 0);
    sfb_fact_work_eff = coalesce(sfb_fact_work_eff, 0);
    if (sfb_total_work > 0) then
      sfb_fix_bonus = round((sfb_fix_bonus / sfb_total_work) * sfb_fact_work, 2); --Фикс.бонус к начислению
    else
    if (sfb_total_work = 0) then
      sfb_fix_bonus = 0.00;

    select count(ks.ks_from_k_id)
    from kontstate ks
    where ks.ks_ol_is_delete = 0 and
          ks.ks_state = 1 and -- говорит о том что кредит активный
          ks.ks_date_begin between :rep_date_begin and :rep_date_end and
          ks.ks_kc_id = :c_id and
          not exists(select *
                     from kontract, product
                     where k_id = ks.ks_from_k_id and
                           k_from_pr_id = pr_id and
                           pr_code containing '.express_online.')
    into :kol_disb_kont;

    if (lo_stage_month_full > 4) then
    begin
      if (sfb_fact_work_eff > 0) then
      begin
        disb_norma = round(kol_disb_kont / sfb_fact_work_eff, 2);

        aarep_bonus_minus_disb = case
                                   when disb_norma <= 0.5 then 3000
                                   when disb_norma between 0.49 and 0.75 then 2000
                                   when disb_norma between 0.74 and 0.99 then 1000
                                 end;
      end
    end

    if (sfb_fact_work = 0) then
      aarep_bonus_minus_disb = 0;

    --Кол-во выдач для расчета штрафа
    kol_disb_for_fine = 0;
    if (sfb_fact_work_eff > 0) then
    begin
      kol_disb_for_fine = cast(kol_disb_kont as double precision) / sfb_fact_work_eff * total_days;
      -- округление вверх
      if (kol_disb_for_fine > trunc(kol_disb_for_fine, 0)) then
        kol_disb_for_fine = trunc(kol_disb_for_fine, 0) + 1;
    end

    aarep_bonus = 0;
    aarep_bonus_total_fact = 0;
    aarep_bonus_total_to_pay = 0;
    --    aarep_bonus_constant = 5747;

    aarep_bonus_total = aarep_bonus_plus - aarep_bonus_minus;
    aarep_bonus_total_fact = aarep_bonus_total - aarep_oklad - aarep_bonus_constant;
    aarep_bonus = aarep_bonus_total - aarep_oklad - aarep_bonus_constant;

    if (aarep_bonus < 0) then
      aarep_bonus = 0;

    --    штраф по заявкам, для штрафа аннулированные заявки тоже участвуют
    lp_amount = null;
    select coalesce(sum(lp_amount), 0)
    from lo_penalty, miniapp
    where lp_is_active = 1 and
          lp_date between :rep_date_begin and :rep_date_end and
          lp_from_c_id = :c_id and
          lp_from_ma_id = ma_id and
          ma_date_end = '31.12.9999' /*
          not exists(select * from miniapp, product
          where ma_id = lp_from_ma_id
          and ma_is_active = 1
          and ma_date_end = '31.12.9999'
          and ma_from_pr_id = pr_id
          and pr_code containing '.express_online.') */
    into lp_amount;
    lp_amount = coalesce(lp_amount, 0);

    fine_online_app = null;
    select sum(lp_amount)
    from lo_penalty, extminiapp
    where lp_date between :rep_date_begin and :rep_date_end and
          lp_is_active = 1 and
          lp_from_c_id = :c_id and
          lp_from_ema_id = ema_id
    into fine_online_app;
    fine_online_app = coalesce(fine_online_app, 0);

    aarep_bonus_minus_disb = coalesce(aarep_bonus_minus_disb, 0);

    aarep_bonus_total_to_pay = aarep_bonus + sfb_fix_bonus - lp_amount - aarep_bonus_minus_disb;

    if (coalesce(aarep_bonus_total_to_pay, -1) < 0) then
      aarep_bonus_total_to_pay = 0;

    select lo_stage_date, kol_vydach, kol_del_plan, aarep_bonus_alternative, lo_stage_month
    from cb_ru_lo_alternative(:c_id, :rep_date_end, :sfb_fix_bonus, :lp_amount, :aarep_bonus_minus_disb)
    into :lo_stage_date, :kol_vydach, :kol_del_plan, :aarep_bonus_alternative, lo_stage_month;

    aarep_day_bonus_plus = aarep_bonus_plus;
    aarep_day_bonus_minus = aarep_bonus_minus;
    aarep_day_bonus_total = aarep_bonus_total;
    aarep_day_returnment_koef = aarep_returnment_koef;

    aarep1_plan_all = 0;
    aarep1_fact_all = 0;
    aarep1_fact_planned = 0;
    aarep1_fact_other = 0;

    --    стаж СВЗ
    --    lo_stage_month = 0;
    --    select lo_stage_month
    --    from hp_get_lo_stage(:c_id, :rep_date_end)
    --    into lo_stage_month;

    fine_disb = 0;
    if (sfb_fact_work_eff > 0) then -- кол-во рабочих дней если 0 то и штраф по выдачам 0
    begin
      if (rep_date_begin < cast('01.01.2024' as date)) then
      begin
        if (kol_disb_for_fine < 20 and
            lo_stage_month >= 3) then
          fine_disb = 6000;
        else
        if (kol_disb_for_fine >= 20 and
            kol_disb_for_fine < 30 and
            lo_stage_month >= 4) then
          fine_disb = 4000;
        else
        if (kol_disb_for_fine >= 30 and
            kol_disb_for_fine < 35 and
            lo_stage_month >= 6) then
          fine_disb = 2000;
      end
      else
      if (rep_date_begin < cast('01.11.2024' as date)) then
      begin
        if (kol_disb_for_fine < 15 and
            lo_stage_month >= 3) then
          fine_disb = 6000;
        else
        if (kol_disb_for_fine >= 15 and
            kol_disb_for_fine < 25 and
            lo_stage_month >= 4) then
          fine_disb = 4000;
        else
        if (kol_disb_for_fine >= 25 and
            kol_disb_for_fine < 35 and
            lo_stage_month >= 6) then
          fine_disb = 2000;
      end
      else
      begin
        if (kol_disb_for_fine < 20 and
            lo_stage_month >= 3) then
          fine_disb = 6000;
        else
        if (kol_disb_for_fine >= 20 and
            kol_disb_for_fine < 30 and
            lo_stage_month >= 4) then
          fine_disb = 4000;
        else
        if (kol_disb_for_fine >= 30 and
            kol_disb_for_fine < 35 and
            lo_stage_month >= 6) then
          fine_disb = 2000;
      end
    end

    if ((extract(day from :rep_date_end) > 1) and
        (need_one_day = 1)) then
    begin
      select sum(aarep_plan_all), sum(aarep_fact_all), sum(aarep_fact_planned), sum(aarep_fact_other),
             sum(aarep_fact_delinq_os), sum(aarep_delinq_os), sum(aarep_delinq_perc)
      from cb_ru_lo_202011_one_tmp(:reg_id, :c_id, :rep_date_begin, :rep_date_end - 1)
      into :aarep1_plan_all, :aarep1_fact_all, :aarep1_fact_planned, :aarep1_fact_other, :aarep1_fact_delinq_os,
           :aarep1_delinq_os, :aarep1_delinq_perc;

      if (aarep1_plan_all is null) then
        aarep1_plan_all = 0;
      if (aarep1_fact_all is null) then
        aarep1_fact_all = 0;
      if (aarep1_fact_planned is null) then
        aarep1_fact_planned = 0;
      if (aarep1_fact_other is null) then
        aarep1_fact_other = 0;

      aarep_day_returnment_koef = 1.0;
      if (aarep_plan_all - aarep1_plan_all <> 0) then
        aarep_day_returnment_koef = round((aarep_fact_planned - aarep1_fact_planned) / (aarep_plan_all - aarep1_plan_all), 4);
      else
        aarep_day_returnment_koef = 0.0;

      if (aarep_day_returnment_koef > 1.0) then
        aarep_day_returnment_koef = 1.0;
      if (aarep_day_returnment_koef < 0) then
        aarep_day_returnment_koef = 1.0;

      aarep1_fact_all = aarep1_fact_all;
      aarep_day_bonus_plus = round((aarep1_fact_all * 0.17) + (aarep1_fact_delinq_os * 0.09), 0);

      aarep_day_bonus_minus = round(((aarep1_delinq_os + aarep1_delinq_perc) * koef_penalty), 0);
      aarep_day_bonus_total = aarep_day_bonus_plus - aarep_day_bonus_minus;

      aarep_day_bonus_plus = aarep_bonus_plus - aarep_day_bonus_plus;
      aarep_day_bonus_minus = aarep_bonus_minus - aarep_day_bonus_minus;
      aarep_day_bonus_total = aarep_bonus_total - aarep_day_bonus_total;
    end

    if (rep_c_id <> 0) then
      suspend;
    else
    begin
      if ((aarep_port_balance_all <> 0) or (aarep_bonus_total <> 0)) then
        suspend;
    end
  end
end
