CREATE PROCEDURE AAREP_DELINQREP_K_IDS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_ONLY_WITH_REP INTEGER,
    REP_K_ID DECIMAL(18,0))
RETURNS (
    AAREP_K_ID DECIMAL(18,0),
    AAREP_KF_ID DECIMAL(18,0),
    AAREP_HAS_REPORT INTEGER,
    AAREP_K_DELIQ_BEGIN_DATE DATE,
    AAREP_K_DELIQ_OS DECIMAL(18,4),
    AAREP_K_DELIQ_PERC DECIMAL(18,4),
    AAREP_K_DELIQ_DAYS INTEGER,
    AAREP_REGION_ID INTEGER,
    AAREP_K_BALANCE_BEGIN DECIMAL(18,4),
    AAREP_K_BALANCE_END DECIMAL(18,4),
    AAREP_REPAY_OS DECIMAL(18,4),
    AAREP_REPAY_PERC DECIMAL(18,4),
    AAREP_REPAY_FINE DECIMAL(18,4),
    AAREP_WRITEOFF_OS DECIMAL(18,4),
    AAREP_WRITEOFF_PERC DECIMAL(18,4),
    AAREP_WRITEOFF_FINE DECIMAL(18,4),
    AAREP_LAST_OL_DATE DATE,
    CURR_CODE VARCHAR(3))
AS
declare variable curr_prior varchar(3);
declare variable curr_rate decimal(18,4);
declare variable pr_set_rate_in_kont integer;
declare variable k_index_curr_rate decimal(18,4);
declare variable dk_date date;
declare variable kolich integer;
declare variable deliq_os decimal(18,4);
declare variable deliq_perc decimal(18,4);
declare variable ol_date date;
declare variable ok_type integer;
declare variable ok_sh_date date;
declare variable ok_sh_id integer;
declare variable sh_os decimal(18,4);
declare variable sh_perc decimal(18,4);
declare variable ok_os decimal(18,4);
declare variable ok_perc decimal(18,4);
declare variable is_first_line integer;
declare variable dd_prior date;
declare variable ok_fine decimal(18,4);
begin
  select cm_default_currency
  from company
  into :curr_prior;
  aarep_has_report = 1;
  aarep_k_deliq_begin_date = null;
  aarep_k_deliq_os = 0;
  aarep_k_deliq_perc = 0;
  aarep_k_deliq_days = 0;
  aarep_k_balance_end = 0;
  aarep_repay_os = 0;
  aarep_repay_perc = 0;
  aarep_repay_fine = 0;
  aarep_writeoff_os = 0;
  aarep_writeoff_perc = 0;
  aarep_writeoff_fine = 0;
  if (rep_k_id <> 0) then
    rep_only_with_rep = 1;

  for select distinct dq_from_k_id, k_from_kf_id, k_index_curr_code, k_index_curr_rate, pr_set_rate_in_kont,
                      k_last_reg_id
      from delinqrep, kontract, region_list(:rep_region_id, 1, -1, 1), product
      where dq_from_k_id = k_id and
            k_from_pr_id = pr_id and
            k_last_reg_id = id_list and
            (:rep_k_id = 0 or
            (:rep_k_id <> 0 and
            k_id = :rep_k_id))
      into :aarep_k_id, :aarep_kf_id, :curr_code, :k_index_curr_rate, :pr_set_rate_in_kont, :aarep_region_id
  do
  begin
    aarep_k_deliq_begin_date = null;
    aarep_k_deliq_os = 0;
    aarep_k_deliq_perc = 0;
    aarep_k_deliq_days = 0;
    aarep_k_balance_begin = 0;
    aarep_k_balance_end = 0;
    aarep_repay_os = 0;
    aarep_repay_perc = 0;
    aarep_repay_fine = 0;
    aarep_writeoff_os = 0;
    aarep_writeoff_perc = 0;
    aarep_writeoff_fine = 0;
    aarep_last_ol_date = null;

    if (rep_only_with_rep = 1) then
    begin
      dk_date = :rep_date_end;
      kolich = 0;
      aarep_k_balance_begin = 0;
      aarep_k_balance_end = 0;
      is_first_line = 1;
      aarep_repay_os = 0;
      aarep_repay_perc = 0;
      aarep_repay_fine = 0;
      aarep_writeoff_os = 0;
      aarep_writeoff_perc = 0;
      aarep_writeoff_fine = 0;

      deliq_os = 0;
      deliq_perc = 0;
      dd_prior = null;

      for select ok_ol_date, ok_type, ok_sh_date, ok_sh_id, ok_sh_os, ok_sh_perc, ok_os, ok_perc, ok_fine
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_k_id = :aarep_k_id and
                ok_ol_date <= :rep_date_end and
                ok_type <> 7 and
                ok_from_c_id = 0
          order by ok_sh_date, ok_sh_id, ok_ol_date
          into :ol_date, :ok_type, :ok_sh_date, :ok_sh_id, :sh_os, :sh_perc, :ok_os, :ok_perc, :ok_fine
      do
      begin
        if (((ol_date > aarep_last_ol_date) or (aarep_last_ol_date is null)) and
            ok_type = 4) then
          aarep_last_ol_date = :ol_date;
        if ((ol_date <= :rep_date_end) and
            (ok_sh_date <= :rep_date_end)) then
        begin
          if (ok_sh_id <> 0) then
          begin
            if (ok_type < 8) then
            begin
              if ((is_first_line = 1) or (:ok_sh_date = :dd_prior)) then
              begin
                deliq_os = :deliq_os + :sh_os + :ok_os;
                deliq_perc = :deliq_perc + :sh_perc + :ok_perc;
                if (is_first_line = 1) then
                  dd_prior = :ok_sh_date;
              end
              else
              begin
                if ((deliq_os > 0) or (deliq_perc > 0)) then
                begin
                  aarep_k_deliq_os = :aarep_k_deliq_os + :deliq_os;
                  aarep_k_deliq_perc = :aarep_k_deliq_perc + :deliq_perc;
                  if (dk_date = :rep_date_end) then
                    dk_date = :dd_prior;
                end
                dd_prior = :ok_sh_date;
                deliq_os = :sh_os + :ok_os;
                deliq_perc = :sh_perc + :ok_perc;
              end
              is_first_line = 0;
            end
          end
        end
        if (:ol_date <= :rep_date_end) then
        begin
          if (ok_type <= 8) then
            aarep_k_balance_end = :aarep_k_balance_end + :ok_os;
        end
        if (:ol_date < :rep_date_begin) then
        begin
          if (ok_type <= 8) then
            aarep_k_balance_begin = :aarep_k_balance_begin + :ok_os;
        end
        if ((ol_date >= :rep_date_begin) and
            (ol_date <= :rep_date_end)) then
        begin
          if (ok_type = 4) then
          begin
            aarep_repay_os = aarep_repay_os - ok_os;
            aarep_repay_perc = aarep_repay_perc - ok_perc;
            aarep_repay_fine = aarep_repay_fine - ok_fine;
          end
          if (ok_type in (5, 6)) then
          begin
            aarep_writeoff_os = aarep_writeoff_os - ok_os;
            aarep_writeoff_perc = aarep_writeoff_perc - ok_perc;
            aarep_writeoff_fine = aarep_writeoff_fine - ok_fine;
          end
          kolich = 1;
        end
      end /* Расчет сумм окончен */
      if (kolich > 0) then
      begin
        if ((:deliq_os > 0) or (:deliq_perc > 0)) then
        begin
          aarep_k_deliq_os = :aarep_k_deliq_os + :deliq_os;
          aarep_k_deliq_perc = :aarep_k_deliq_perc + :deliq_perc;
          if (dk_date = :rep_date_end) then
            dk_date = :dd_prior;
        end
      end
      if ((aarep_k_deliq_os + aarep_k_deliq_perc) >= 0.01) then
      begin
        if (curr_code <> curr_prior) then
        begin
          select cr_rate
          from opercurr
          where cr_from_cu_id = :curr_code and
                cr_date <= :rep_date_end and
                cr_date_end >= :rep_date_end
          into :curr_rate;
          if (curr_rate < :k_index_curr_rate) then
          begin
            if (pr_set_rate_in_kont = 1) then
            begin
              curr_rate = :k_index_curr_rate;
            end
          end
          aarep_k_deliq_os = cast(aarep_k_deliq_os * curr_rate as decimal(15,2));
          aarep_k_deliq_perc = cast(aarep_k_deliq_perc * curr_rate as decimal(15,2));
          aarep_k_balance_end = cast(aarep_k_balance_end * curr_rate as decimal(15,2));
          aarep_k_balance_begin = cast(aarep_k_balance_begin * curr_rate as decimal(15,2));
        end
        aarep_k_deliq_days = cast((:rep_date_end - :dk_date) as integer);
      end
      aarep_k_deliq_begin_date = :dk_date;

    end
    suspend;
  end

  if (rep_only_with_rep = 0) then
  begin
    aarep_has_report = 0;
    curr_prior = '';

    for select distinct dk_id, dk_kf_id, dk_date, dk_os, dk_perc, dk_days, dk_region_id, dk_balance_begin,
                        dk_balance_end, aarep_repay_os, aarep_repay_perc, aarep_repay_fine, aarep_writeoff_os,
                        aarep_writeoff_perc, aarep_writeoff_fine, aarep_last_ol_date, curr_code
        from aarep_kontract_deliq_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '000')
        where (dk_os + dk_perc) >= 0.01
        into :aarep_k_id, :aarep_kf_id, :aarep_k_deliq_begin_date, :aarep_k_deliq_os, :aarep_k_deliq_perc,
             :aarep_k_deliq_days, :aarep_region_id, :aarep_k_balance_begin, :aarep_k_balance_end, :aarep_repay_os,
             :aarep_repay_perc, :aarep_repay_fine, :aarep_writeoff_os, :aarep_writeoff_perc, :aarep_writeoff_fine,
             :aarep_last_ol_date, :curr_code
    do
    begin
      suspend;
    end
  end
end
