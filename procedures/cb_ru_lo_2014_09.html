CREATE PROCEDURE CB_RU_LO_2014_09(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    CB_PERSONAL_NAME VARCHAR(100),
    JS_NAME VARCHAR(100),
    CB_IS_WORKING INTEGER,
    PAR7 DECIMAL(18,4),
    KOL_CLI_CATEG_1_2 INTEGER,
    KOL_CLI_BEGIN_CAT_1 INTEGER,
    KOL_CLI_END_CAT_1 INTEGER,
    PRIROST_CLI_CAT_1 INTEGER,
    BONUS_PRIROST CURR_TYPE,
    BONUS_CLI CURR_TYPE,
    SHTRAF_PRIROST CURR_TYPE,
    PAR7_LAST CURR_TYPE,
    SNIJEN_PAR7 CURR_TYPE,
    SHTRAF_PAR7 CURR_TYPE,
    BONUS_SNIJEN_PAR7 CURR_TYPE,
    BONUS_END CURR_TYPE,
    CB_MEMO VARCHAR(1024),
    SHTRAF_TOTAL CURR_TYPE,
    SHTRAF_MEMO VARCHAR(1024),
    CB_PERSONAL_ID DECIMAL(18,0),
    AA_CURR_REGION_ID INTEGER,
    CB_BONUS_TO_PAY DECIMAL(18,4))
AS
declare variable rep_date_begin date;
declare variable balance_categ_1 curr_type;
declare variable delinq_balance_categ_1 curr_type;
declare variable aa_curr_region_id_prior integer;
declare variable prirost_factor curr_type;
declare variable cli_factor curr_type;
declare variable par_factor curr_type;
declare variable js_code varchar(100);
declare variable balance_categ_1_last curr_type;
declare variable delinq_balance_categ_1_last curr_type;
begin
  if ((rep_region_id is null) or (rep_date_end is null)) then
    exit;
  rep_date_begin = rep_date_end - extract(day from rep_date_end);

-- delete from TEMP_KC_REG_INFO;
-- на начало
  insert into temp_kc_reg_info (kc_region_id, kc_officer_id, kc_kol_cli, kc_balance, kc_kol_cli_sks)
  select ac_region_id, ac_kc_id, sum(ac_is_client), sum(ac_cli_balance), 0 -- KC_KOL_CLI_SKS - 0 - на дату начало; 1 - на дату конец
  from aarep_cli_active_list(:rep_date_begin)
  where (select aa_category_id
         from aarep_get_category_id(ac_kontract_id, :rep_date_begin, null, null, null, null, null, null)) in (1) and
        not exists(select *
                   from sb_akt, sb_list
                   where sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_status = 2 and
                         :rep_date_begin between sba_date_begin and sba_date_end and
                         sba_from_k_id = ac_kontract_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1 -- находящийся по внешним коллектором не будет считатся
        )
  group by ac_region_id, ac_kc_id;
-- на конец
  insert into temp_kc_reg_info (kc_region_id, kc_officer_id, kc_kol_cli, kc_balance, kc_kol_cli_sks)

  select ac_region_id, ac_kc_id, sum(ac_is_client), sum(ac_cli_balance), 1 -- KC_KOL_CLI_SKS - 0 - на дату начало; 1 - на дату конец
  from aarep_cli_active_list(:rep_date_end)
  where (select aa_category_id
         from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) in (1) and
        not exists(select *
                   from sb_akt, sb_list
                   where sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_status = 2 and
                         :rep_date_end between sba_date_begin and sba_date_end and
                         sba_from_k_id = ac_kontract_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1 -- находящийся по внешним коллектором не будет считатся
        ) and
        exists(select *
               from knaktlist
               where knaktlist.is_delete = 0 and
                     :rep_date_end between kal_date and kal_date_end and
                     kal_from_kf_id = ac_kf_id and
                     kal_new_kf_officer = ac_kc_id and
                     (kal_date <= :rep_date_begin -- считать только не переданным за текущую неделю
                     or
                     kal_date <= ac_disb_date))
  group by ac_region_id, ac_kc_id;

  rep_date_begin = rep_date_begin + 1;
  aa_curr_region_id_prior = 0;

  for select aa_curr_kc_id, aa_curr_region_id, sum(iif(aa_category_id = 1, aa_amount_active_end, 0)),
             sum(iif(aa_category_id = 1, aa_deliq_balance7 - aa_deliq_balance180, 0)), sum(iif(not exists(select *
                                                                                                          from sb_akt, sb_list
                                                                                                          where sba_is_delete = 0 and
                                                                                                                sba_from_sbl_id <> 0 and
                                                                                                                sba_status = 2 and
                                                                                                                :rep_date_end between sba_date_begin and sba_date_end and
                                                                                                                sba_from_k_id = aa_k_id and
                                                                                                                sba_from_sbl_id = sbl_id and
                                                                                                                sbl_type = 1), aa_just_cl_end, 0))
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 1, 1, 1, 0, 0, 0)
      where aa_category_id in (1, 2) and
            aa_curr_kc_id > 1000000000003
      group by aa_curr_kc_id, aa_curr_region_id
      into cb_personal_id, aa_curr_region_id, balance_categ_1, delinq_balance_categ_1, kol_cli_categ_1_2
  do
  begin
    cb_memo = '';
    shtraf_memo = '';
    if (aa_curr_region_id_prior <> aa_curr_region_id) then
    begin
      aa_curr_region_id_prior = aa_curr_region_id;
      select r1.reg_code || ' ' || r1.reg_name office_name, r2.reg_code || ' ' || r2.reg_name branch_name
      from region r1, region r2
      where r1.reg_unicode = :aa_curr_region_id and
            r1.parent = r2.reg_unicode
      into office_name, branch_name;
    end
    cb_personal_name = (select c_name_fio
                        from clients
                        where c_id = :cb_personal_id);
-- Системная должность
    js_name = null;
    js_code = null;
    select rd_name, jk_system_code
    from hr_pstatus, joblist, jobkind, refdata
    where ps_from_c_id = :cb_personal_id and
          ps_is_working = 1 and
          ps_status = 1 and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_id = rd_id
    into js_name, js_code;
    if (js_name is null) then
      js_name = 'Не определен';
    if (js_code is null) then
    begin
      js_code = '';
      cb_is_working = null;
--    Cтатус на конец даты,  0 - не работает; 1 - работает;
      select ps_is_working
      from hr_pstatus
      where ps_from_c_id = :cb_personal_id and
            ps_date_begin is not null and
            ps_status = 1 and
            :rep_date_end between ps_date_begin and ps_date_end
      into cb_is_working;
      if (cb_is_working is null) then
        cb_is_working = 0;
    end
    else
      cb_is_working = 1;

    if (delinq_balance_categ_1 > 0) then
    begin
      par7 = delinq_balance_categ_1 / balance_categ_1 * 100;
    end
    else
    begin
      par7 = 0;
    end

    prirost_cli_cat_1 = 0;
    prirost_factor = 0;
    cli_factor = 0;
    shtraf_prirost = 0;

--  клиенты на начало
    kol_cli_begin_cat_1 = null;
    select kc_kol_cli
    from temp_kc_reg_info
    where kc_region_id = :aa_curr_region_id and
          kc_officer_id = :cb_personal_id and
          kc_kol_cli_sks = 0 -- выборка начало
    into kol_cli_begin_cat_1;
    if (kol_cli_begin_cat_1 is null) then
      kol_cli_begin_cat_1 = 0;

--  клиенты на конец
    kol_cli_end_cat_1 = null;
    select kc_kol_cli
    from temp_kc_reg_info
    where kc_region_id = :aa_curr_region_id and
          kc_officer_id = :cb_personal_id and
          kc_kol_cli_sks = 1 -- выборка конец
    into kol_cli_end_cat_1;
    if (kol_cli_end_cat_1 is null) then
      kol_cli_end_cat_1 = 0;

--      Расчет прироста клиентов по категории 1
    prirost_cli_cat_1 = kol_cli_end_cat_1 - kol_cli_begin_cat_1;

    if (kol_cli_categ_1_2 < 250) then
    begin
      if (kol_cli_categ_1_2 <= 120) then
      begin
        prirost_factor = case
                           when prirost_cli_cat_1 <= 5 then 0
                           when prirost_cli_cat_1 between 6 and 14 then 400
                           when prirost_cli_cat_1 >= 15 then 500
                         end;
      end
      else
      if (kol_cli_categ_1_2 between 121 and 249) then
      begin
        prirost_factor = case
                           when prirost_cli_cat_1 <= 5 then 0
                           when prirost_cli_cat_1 between 6 and 14 then 1000
                           when prirost_cli_cat_1 >= 15 then 1200
                         end;
      end

      par_factor = case
                     when par7 < 1.0000 then 1.1
                     when par7 <= 7.0000 then 1
                     else 0
                   end;

      if (prirost_cli_cat_1 between 0 and 5) then
      begin
        shtraf_prirost = 2000;
        shtraf_memo = 'Штраф за прирост менее 6 клиентов; ';
      end
      else
      if (prirost_cli_cat_1 < 0) then
      begin
        shtraf_prirost = 5000;
        shtraf_memo = 'Штраф за отток клиентов; ';
      end

      if (prirost_cli_cat_1 <= 5) then
      begin
        cb_memo = 'Прирост менее 6 клиентов;';
      end

    end
    else
    begin
-- Бонус получает более 250кл
      if (kol_cli_categ_1_2 between 250 and 299) then
      begin
        cli_factor = 140;
      end
      else
      if (kol_cli_categ_1_2 >= 300) then
      begin
        cli_factor = 145;
      end

      par_factor = case
                     when par7 < 1.0000 then 1.2
                     when par7 < 3.0000 then 1.1
                     when par7 <= 7.0000 then 1
                     else 0
                   end;

    end
    bonus_prirost = prirost_factor * par_factor * prirost_cli_cat_1;
    bonus_cli = cli_factor * par_factor * kol_cli_categ_1_2;

    if (par7 > 7.0000) then
    begin
      cb_memo = cb_memo || ' PAR7 более 7%;';
    end

--    Бонус за снижение PAR7
    balance_categ_1_last = null;
    delinq_balance_categ_1_last = null;
    par7_last = 0;
    snijen_par7 = 0;
    shtraf_par7 = 0;
    bonus_snijen_par7 = 0;

    if (par7 > 7.0000 and
        :rep_date_end <= '31.10.2014') then
    begin
      select sum(oks_balance), sum(iif(:rep_date_begin - 1 - oks_delinq_date > 7, oks_balance, 0))
      from oksummary
      where :rep_date_begin - 1 between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_kc_id = :cb_personal_id and
            oks_region_id = :aa_curr_region_id and
            oks_close_state = 0 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_begin - 1, null, null, null, null, null, null)) in (1)
      into balance_categ_1_last, delinq_balance_categ_1_last;

      if (delinq_balance_categ_1_last is null) then
        delinq_balance_categ_1_last = 0;
      if (delinq_balance_categ_1_last = 0) then
        par7_last = 0;
      else
        par7_last = delinq_balance_categ_1_last / balance_categ_1_last * 100;

      snijen_par7 = par7_last - par7;

      if (snijen_par7 <= -0.1000) then
      begin
        shtraf_par7 = 2000;
        shtraf_memo = shtraf_memo || 'Штраф за ухудшение показателя PAR7;';
      end
      else
      if (snijen_par7 >= 5.0000) then
      begin
        if (kol_cli_categ_1_2 < 250) then
        begin
          bonus_snijen_par7 = prirost_factor * prirost_cli_cat_1 / 2;
        end
        else
        begin -- исправлено по письму Зары от 30.09.2014 17:34
          bonus_snijen_par7 = cli_factor * kol_cli_categ_1_2 / 2;
        end
      end
      if (snijen_par7 < 5.0000) then
      begin
        cb_memo = cb_memo || ' Снижение PAR7 менее 5%;';
      end

    end

    shtraf_total = shtraf_prirost + shtraf_par7;

    bonus_end = bonus_prirost + bonus_cli + bonus_snijen_par7;
    cb_bonus_to_pay = bonus_end;
    suspend;
  end

end
