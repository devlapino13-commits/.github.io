CREATE PROCEDURE HP_INTEREST_ADJUSTMENT(
    K_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    NEW_OL_ID ID_TYPE)
AS
declare variable k_stop_perc date;
declare variable k_disb_date date;
declare variable k_last_reg_id integer;
declare variable k_kontract_no varchar(20);
declare variable ol_memo varchar(1024);
declare variable un_id id_type;
declare variable sh_id integer;
declare variable sh_date date;
declare variable sh_perc curr_type;
declare variable s_perc curr_type;
declare variable las_ol_date date;
begin
  k_stop_perc = null;
  select k_stop_perc, k_disb_date, k_last_reg_id, k_kontract_no
  from kontract
  where k_id = :k_id
  into k_stop_perc, k_disb_date, k_last_reg_id, k_kontract_no;
  if (k_stop_perc is null) then
    exit;

  select max(ok_ol_date)
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1
  into las_ol_date;
  if (las_ol_date is null) then
    exit;

  if (rep_date is not null) then
  begin
    if (rep_date not between k_disb_date and current_date or rep_date < k_stop_perc or rep_date < las_ol_date) then
    begin
--      rest_text = 'Задана неверная дата!';
--      suspend;
--      exit;
      exception e_default 'Invalid date specified';
    end
  end
  else
  begin
    rep_date = las_ol_date;
    if (rep_date < k_stop_perc) then
      rep_date = k_stop_perc;
  end

  if (un_id is null) then
    un_id = 1000000000001;

  for select sh_id, sh_date, zad_perc, coalesce(sh_calc_perc, 0)
      from aarep_make_zad_list(:k_id, :rep_date, 1)
      where (zad_perc <> 0 or
            coalesce(sh_calc_perc, 0) <> 0)
      into sh_id, sh_date, sh_perc, s_perc
  do
  begin
    if (sh_perc <> s_perc) then
    begin
      if (new_ol_id is null) then
      begin
        select kv
        from get_kv(:k_last_reg_id, 5)
        into :new_ol_id;

        ol_memo = 'Корректировка % по графику в соответствии с накопленными, договора №' || k_kontract_no || '. Дата корректировки ' || :rep_date;
        insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete,
                              ol_user_id)
        values (:new_ol_id, :k_last_reg_id, 1, :rep_date, 8, :ol_memo, 0, 0, :un_id);
      end

      if (sh_perc > 0) then
      begin
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:new_ol_id, -1, :k_id, :sh_id, :sh_date, 3, 0, 0, 0, 3, 0, 0, -:sh_perc, 0, 0);
      end
      if (s_perc > 0) then
      begin
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:new_ol_id, -1, :k_id, :sh_id, :sh_date, 3, 0, 0, 0, 3, 0, 0, :s_perc, 0, 0);
      end
    end
  end
  if (new_ol_id is not null) then
    suspend;
end
