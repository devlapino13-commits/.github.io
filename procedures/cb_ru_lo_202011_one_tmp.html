CREATE PROCEDURE CB_RU_LO_202011_ONE_TMP(
    REP_REGION_ID INTEGER,
    REP_C_ID D_ID,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    AAREP_PLAN_ALL D_CURRENCY,
    AAREP_FACT_ALL D_CURRENCY,
    AAREP_FACT_PLANNED D_CURRENCY,
    AAREP_FACT_OTHER D_CURRENCY,
    AAREP_MONTH_PLAN_OS D_CURRENCY,
    AAREP_MONTH_PLAN_PERC D_CURRENCY,
    AAREP_DELINQ_OS D_CURRENCY,
    AAREP_DELINQ_PERC D_CURRENCY,
    AAREP_FACT_OS D_CURRENCY,
    AAREP_FACT_PERC D_CURRENCY,
    AAREP_FACT_FINE D_CURRENCY,
    AAREP_FACT_DELINQ_OS D_CURRENCY,
    AAREP_FACT_DELINQ_PERC D_CURRENCY,
    AAREP_FACT_ADV_OS D_CURRENCY,
    AAREP_FACT_ADV_PERC D_CURRENCY,
    AAREP_OKLAD D_CURRENCY,
    AAREP_PORT_BALANCE_OWN D_CURRENCY,
    AAREP_PORT_KOLICH_OWN INTEGER,
    AAREP_BONUS_OS D_CURRENCY,
    AAREP_BONUS_OS_ADV D_CURRENCY,
    PR_CODE STR_100,
    K_DISB_DATE DATE,
    AAREP_FACT_KOMMIS D_CURRENCY,
    TERM_PAY_OS D_CURRENCY,
    TERM_CURR_SH_OS D_CURRENCY,
    TERM_PAY_PERC D_CURRENCY,
    TERM_CURR_SH_PERC D_CURRENCY,
    TERM_PAY_KOMMIS D_CURRENCY)
AS
declare variable DD_DISB_DATE date;
declare variable DD date;
declare variable K_STOP_PERC date;
declare variable SUM1 D_CURRENCY = 0;
declare variable SUM2 D_CURRENCY = 0;
declare variable KS_DATE_END date;
declare variable REP_MONTH_END date;
declare variable SH_DD date;
declare variable SUM3 D_CURRENCY = 0;
declare variable SUM4 D_CURRENCY = 0;
declare variable KAL_DATE date;
declare variable KS_KC_ID D_ID;
declare variable CAN_SUSPEND integer;
declare variable KF_ID D_ID;
declare variable OK_WITH_AGENT integer;
declare variable ADVANCED_DAYS integer;
declare variable PR_ID integer;
declare variable PR_ID_PRIOR integer;
declare variable KS_FROM_OL_ID ID_TYPE;
declare variable LOYAL_CALC_DATE date;
declare variable K_STOPPED_PERC_OL_ID_TMP D_ID;
declare variable K_STOPPED_PERC_OL_ID D_ID;
declare variable PENALTY_MINUS_OS D_CURRENCY;
declare variable PENALTY_MINUS_PERC D_CURRENCY;
declare variable PAYED_AFTER_TERMINATION_OS D_CURRENCY;
declare variable PAYED_AFTER_TERMINATION_PERC D_CURRENCY;
declare variable SH_ID D_INTEGER;
declare variable K_STOP_PERC_DATETIME D_DATETIME;
declare variable PRV_PAY_OS D_CURRENCY;
declare variable PRV_PAY_PERC D_CURRENCY;
declare variable LATE_PENALTY D_SMALLINT;
declare variable DATE_CORRECT1 D_DATE;
declare variable DATE_CORRECT2 D_DATE;
declare variable DATE_CORRECT3 D_DATE;
begin
  -->>Если первая дата пустая, выставляется первый день месяца с rep_date_end
  if (rep_date_begin is null) then
    rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;

  -->>Последний день месяца из rep_date_end
  rep_month_end = rep_date_end - extract(day from :rep_date_end) + 1;
  rep_month_end = dateadd(month, 1, rep_month_end) - 1;

  aarep_port_balance_own = 0;
  aarep_port_kolich_own = 0;

  advanced_days = 10;

  -- оклад сотрудника
  aarep_oklad = 0;
  select sum(sao_summa)
  from sa_oklad
  where sao_date <= :rep_date_end and
        sao_from_c_id = :rep_c_id
  into :aarep_oklad;
  aarep_oklad = coalesce(aarep_oklad, 0);

  pr_id_prior = 0;

  for select distinct ks_from_k_id, k_kontract_no, ks_date_begin, ks_date_end, k_kolich_client_approved, ks_kc_id,
                      kn1.kal_date, k_from_kf_id, k_from_pr_id, ks_from_ol_id, kont.k_stopped_perc_ol_id
      from kontstate ks, kontract kont, knaktlist kn1
      where ks_ol_is_delete = 0 and
            ks_date_end >= (:rep_date_begin - 1) and
            ks_date_begin <= :rep_date_end and
            ks_state = 1 and
            kn1.kal_from_kf_id = k_from_kf_id and
            ks_from_k_id = k_id and
            kn1.kal_date <= :rep_date_end and
            kn1.kal_date_end >= :rep_date_end and
            kn1.is_delete = 0 and
            kn1.kal_new_kf_officer = :rep_c_id
--            and k_id = 101010117164000201
--            and k_id = 101010060646600401
      into :k_id, :k_kontract_no, :dd_disb_date, :ks_date_end, :aarep_port_kolich_own, :ks_kc_id, :kal_date, :kf_id,
           pr_id, ks_from_ol_id, k_stopped_perc_ol_id_tmp
  do
  begin
    k_stop_perc = null;
    k_stopped_perc_ol_id = null;
    k_stop_perc_datetime = null;

    penalty_minus_os = 0;
    penalty_minus_perc = 0;

    payed_after_termination_os = 0;
    payed_after_termination_perc = 0;
    prv_pay_os = 0;
    prv_pay_perc = 0;

    --Выплаты за текщий месяц по расторженным договорам с предыдущего месяца
    term_pay_os = 0;
    term_curr_sh_os = 0;
    term_pay_perc = 0;
    term_curr_sh_perc = 0;
    term_pay_kommis = 0;
    select pay_os, curr_sh_os, pay_perc, curr_sh_perc, pay_kommis, prv_pay_os, prv_pay_perc
    from aarep_debt_after_termination(null, :k_id, :rep_date_end)
    into :term_pay_os, :term_curr_sh_os, :term_pay_perc, :term_curr_sh_perc, :term_pay_kommis, prv_pay_os, prv_pay_perc;

    if (term_pay_os is null) then
      term_pay_os = 0;
    if (term_curr_sh_os is null) then
      term_curr_sh_os = 0;
    if (term_pay_perc is null) then
      term_pay_perc = 0;
    if (term_curr_sh_perc is null) then
      term_curr_sh_perc = 0;
    if (term_curr_sh_perc is null) then
      term_pay_kommis = 0;
    if (prv_pay_os is null) then
      prv_pay_os = 0;
    if (prv_pay_perc is null) then
      prv_pay_perc = 0;

    -- если было расторжение договора и дата раторжения больше 01.10.2020, тогда платеж образовавшийся
    -- ввиду схлопывания графика не должен в рамках текущей бонусной политики негативно влиять на бонус
    late_penalty = 0;
    if (k_stopped_perc_ol_id_tmp is not null) then
    begin
      select ol.ol_id, ol.ol_date, ol.ol_datetime
      from operlist ol
      where ol.ol_id = :k_stopped_perc_ol_id_tmp and
            ol.is_delete = 0
      into :k_stopped_perc_ol_id, :k_stop_perc, :k_stop_perc_datetime;

      if (k_stop_perc is not null and extract(month from k_stop_perc) = 1 and extract(day from k_stop_perc) in (30,31)) then
      begin
        late_penalty = 1;
      end

      if (coalesce(k_stop_perc, cast('01.01.1900' as date)) between cast('01.10.2020' as date) and :rep_date_end) then -- не начислять штрафы
      begin
        if (k_stop_perc + 30 > rep_date_end or late_penalty = 1 /*(k_id = 101010046647900501 and rep_date_end < '01.04.2022')*/) then
        begin
          select ok.ok_sh_os, ok.ok_sh_perc, ok.ok_sh_id
          from operkred ok
          where ok.ok_from_k_id = :k_id and
                ok.ok_from_ol_id = :k_stopped_perc_ol_id and
                ok.ok_sh_os > 0
          into :penalty_minus_os, :penalty_minus_perc, :sh_id;

          payed_after_termination_os = 0;
          payed_after_termination_perc = 0;

          -- т.к. после расторжения могло быть погашение охватившее платеж, образовавшийся в входе расторжения,
          -- корректировочную сумму уменьшаю на сумму платежа

          select -sum(ok.ok_os), -sum(ok.ok_perc)
          from operkred ok
          inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
          where ok.ok_from_k_id = :k_id and
                ok.ok_ol_is_active = 1 and
                ok.ok_type = 4 and
                ok.ok_ol_date between :k_stop_perc and :rep_date_end and
                ok.ok_sh_date = :k_stop_perc and
                ok.ok_sh_id = :sh_id and
                ol.ol_datetime >= :k_stop_perc_datetime
          into :payed_after_termination_os, :payed_after_termination_perc;

          payed_after_termination_os = coalesce(payed_after_termination_os, 0);
          payed_after_termination_perc = coalesce(payed_after_termination_perc, 0);

          penalty_minus_os = penalty_minus_os - payed_after_termination_os;
          penalty_minus_perc = penalty_minus_perc - payed_after_termination_perc;
        end
      end
    end

    ok_with_agent = 0;
    can_suspend = 1;

    if (ks_kc_id <> :rep_c_id and
        not exists(select *
                   from krexpert
                   where ke_status = 1 and
                         ke_from_p_id = :rep_c_id and
                         ke_teacher_id = :ks_kc_id and
                         ke_teacher_date_end >= :kal_date -- до этого было :rep_date_end, при передаче наставник действующий должен быть
        )) then
    begin
      if (kal_date >= maxvalue(:rep_date_begin, :dd_disb_date)) then
      begin
        if (not(exists(select *
                       from knaktlist
                       where kal_from_kf_id = :kf_id and
                             is_delete = 0 and
                             kal_new_kf_officer = :rep_c_id and
                             kal_date <= (:kal_date - 1) and
                             kal_date_end >= (:kal_date - 1)))) then
          can_suspend = 0;
      end
    end

    if (can_suspend = 1) then
    begin
      if (pr_id_prior <> pr_id) then
      begin
        pr_id_prior = pr_id;
        select pr_code
        from product

        where pr_id = :pr_id
        into pr_code;
      end

      dd = maxvalue(:rep_date_begin, (dd_disb_date + 1));

      aarep_delinq_os = 0;
      aarep_delinq_perc = 0;
      aarep_month_plan_os = 0;
      aarep_month_plan_perc = 0;

      aarep_fact_all = 0;
      aarep_fact_other = 0;
      aarep_fact_planned = 0;
      aarep_plan_all = 0;

      aarep_fact_os = 0;
      aarep_fact_perc = 0;
      aarep_fact_fine = 0;
      aarep_fact_delinq_os = 0;
      aarep_fact_delinq_perc = 0;
      aarep_fact_adv_os = 0;
      aarep_fact_adv_perc = 0;

      aarep_bonus_os = 0;
      aarep_bonus_os_adv = 0;

      if (rep_date_begin <= k_stop_perc + 30) then
      --      if (k_stop_perc > '01.10.20') then
      begin
        aarep_delinq_os = aarep_delinq_os + term_curr_sh_os;
        aarep_delinq_perc = aarep_delinq_perc + term_curr_sh_perc;
      end

      aarep_port_balance_own = 0;
      select oks_balance
      from oksummary
      where oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_from_k_id = :k_id and
            (:rep_date_end - oks_delinq_date <= 360) and
            oks_close_state = 0
      into :aarep_port_balance_own;
      if (aarep_port_balance_own is null) then
        aarep_port_balance_own = 0;
      if (aarep_port_balance_own = 0) then
      begin
        aarep_port_kolich_own = 0;
      end

      loyal_calc_date = null;
      select ogr_ol_date
      from operagr
      where ogr_from_ol_id = :ks_from_ol_id and
            ogr_sign = 1 and
            ogr_amount > 0
      into loyal_calc_date;

      if (loyal_calc_date is not null) then
        ok_with_agent = 1;
      else
        ok_with_agent = 0;

      -- Просроченные платежи по кредиту за период
      for select ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc),
                 sum(iif(ok_ol_date < :dd, ok_os + ok_sh_os, 0)), sum(iif(ok_ol_date < :dd, ok_perc + ok_sh_perc, 0))
          from operkred
          where ok_from_k_id = :k_id and
                ok_ol_date <= :rep_date_end and
                ok_sh_date between :rep_date_begin and :rep_month_end and
                ok_ol_is_active = 1 and
                ok_type < 7 and
                ok_from_c_id = 0
          group by ok_sh_date
          into :sh_dd, :sum1, :sum2, sum3, sum4
      do
      begin
        if (sum1 is null) then
          sum1 = 0;
        if (sum2 is null) then
          sum2 = 0;
        if (sum3 is null) then
          sum3 = 0;
        if (sum4 is null) then
          sum4 = 0;
        aarep_month_plan_os = aarep_month_plan_os + sum3;
        aarep_month_plan_perc = aarep_month_plan_perc + sum4;
        if (sh_dd between :rep_date_begin and :rep_date_end) then
        begin
          aarep_delinq_os = aarep_delinq_os + sum1;
          aarep_delinq_perc = aarep_delinq_perc + sum2;

          if ((sh_dd = '25.11.2014') and
              (k_stop_perc <= '25.11.2014')) then -- не начислять штрафы если дата возникновения 25.11.2014
          begin
            aarep_delinq_os = aarep_delinq_os - sum1;
            aarep_delinq_perc = aarep_delinq_perc - sum2;
          end

          if (coalesce(k_stop_perc, cast('01.01.1900' as date)) between cast('01.10.2020' as date) and :rep_date_end) then -- не начислять штрафы
          begin
            if (sh_dd = k_stop_perc) then
            begin
              if (sh_dd + 30 > rep_date_end) then
              begin
                aarep_delinq_os = aarep_delinq_os - penalty_minus_os;
                aarep_delinq_perc = aarep_delinq_perc - penalty_minus_perc;

                penalty_minus_os = 0;
                penalty_minus_perc = 0;
              end
            end
            if (late_penalty = 1 /*k_id = 101010046647900501*/) then
            begin
              date_correct1 = '01.01.'||extract(year from :rep_date_end);
              date_correct2 = '01.03.'||extract(year from :rep_date_end);
              date_correct3 = '01.04.'||extract(year from :rep_date_end);

              if (rep_date_end < '01.03.2022') then
              begin
                aarep_delinq_os = aarep_delinq_os - penalty_minus_os;
                aarep_delinq_perc = aarep_delinq_perc - penalty_minus_perc;
              end else
              if (rep_date_end < '01.04.2022') then
              begin
                aarep_delinq_os = aarep_delinq_os + penalty_minus_os;
                aarep_delinq_perc = aarep_delinq_perc + penalty_minus_perc;
              end else
              if (rep_date_end < date_correct2 and rep_date_end >= date_correct1) then
              begin
                aarep_delinq_os = aarep_delinq_os - penalty_minus_os;
                aarep_delinq_perc = aarep_delinq_perc - penalty_minus_perc;
              end else
              if (rep_date_end < date_correct3 and rep_date_end >= date_correct1) then
              begin
                aarep_delinq_os = aarep_delinq_os + penalty_minus_os;
                aarep_delinq_perc = aarep_delinq_perc + penalty_minus_perc;
              end
                penalty_minus_os = 0;
                penalty_minus_perc = 0;

            end

          end

        end
      end

      if (late_penalty = 1 and k_id = 101010038581100701 and rep_date_end < '01.04.2022') then
      begin
        aarep_delinq_os = aarep_delinq_os + penalty_minus_os;
        aarep_delinq_perc = aarep_delinq_perc + penalty_minus_perc;
        penalty_minus_os = 0;
        penalty_minus_perc = 0;
      end

      if (late_penalty = 1 and k_id = 101010097213500101 and rep_date_end < '01.04.2023' and rep_date_end > '01.01.2023') then
      begin
        aarep_delinq_os = aarep_delinq_os + penalty_minus_os;
        aarep_delinq_perc = aarep_delinq_perc + penalty_minus_perc;
        penalty_minus_os = 0;
        penalty_minus_perc = 0;
      end 

      if (late_penalty = 1 and
          coalesce(k_stop_perc, cast('01.01.1900' as date))  = cast('31.01.'||extract(year from :rep_date_end) as date) and
          :rep_date_end < cast('01.04.'||extract(year from :rep_date_end) as date) and
          :rep_date_end > cast('01.01.'||extract(year from :rep_date_end) as date)
          ) then
      begin
        aarep_delinq_os = aarep_delinq_os + penalty_minus_os;
        aarep_delinq_perc = aarep_delinq_perc + penalty_minus_perc;
        penalty_minus_os = 0;
        penalty_minus_perc = 0;
      end 

      select -sum(iif(ok_sh_date between :rep_date_begin and :rep_month_end, ok_os, 0.0)), -- своевременные ОС
             -sum(iif(ok_sh_date between :rep_date_begin and :rep_month_end, ok_perc, 0.0)), -- своевременные %%
             -sum(ok_kommis), -- комиссия
             -sum(ok_fine), -- пени
             -sum(iif(ok_sh_date < :rep_date_begin, ok_os, 0.0)), -- погашение просроченных ОС
             -sum(iif(ok_sh_date < :rep_date_begin, ok_perc, 0.0)), -- погашение просроченных %%
             -sum(iif(ok_sh_date > :rep_month_end, ok_os, 0.0)), -- досрочное погашение ОС
             -sum(iif(ok_sh_date > :rep_month_end, ok_perc, 0.0)), -- досрочное погашение %%
             -sum(iif(ok_sh_date >= :rep_date_begin and datediff(day, ok_ol_date, ok_sh_date) > :advanced_days, ok_os, 0.0)), -- досрочное погашение ОС (более чем 10 дней)
             -sum(iif(ok_sh_date >= :rep_date_begin and datediff(day, ok_ol_date, ok_sh_date) <= :advanced_days, ok_os, 0.0))
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_ol_is_active = 1 and
            ok_type = 4 and
            ok_from_c_id = 0 and
            ok_sh_id > :ok_with_agent -- При расчете дохода не учитываются: погашения за первые два платежа по графику от клиентов, привлеченных агентами;
      into :aarep_fact_os, :aarep_fact_perc, :aarep_fact_kommis, :aarep_fact_fine, :aarep_fact_delinq_os,
