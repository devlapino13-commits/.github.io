declare function getcatval(kind d_str_code, code d_str_code)
  returns d_str_255
  as
  begin
    return coalesce((select bct_name from bki_catalog where bct_kind = :kind and bct_code = :code), 'Неопределен');
  end

declare function f_my_obj_get_arr(handle d_integer, path d_str_long)
returns d_integer
as
  declare variable jsa d_integer;
  declare variable val_type d_str_code;
begin
  if (json.contains(handle, path) = 0) then
    return null;

  val_type = json.val_get_type_name(handle, path);

  if (val_type = 'Array') then
    return json.obj_get_arr(handle, path);
  else if (val_type = 'Object') then
  begin
    handle = json.obj_get_obj(handle, path);
    jsa = json.arr_create();
    json.arr_set_obj(jsa, -1, handle);

    return jsa;
  end
  else
    exception e_default path || ' is not Array/Object';
end

declare function my_get_last_el(jso_in d_integer)
returns d_integer
as
  declare variable jso d_integer;
  declare variable res d_integer;
  declare variable res_tmp d_integer;
  declare variable max_date d_date;
  declare variable cur_date d_date;
begin
  res = null;
  if (jso_in is not null) then
  begin
    max_date = '01.01.1900';
  
    for select jso
    from hp_bki_json_get_list(:jso_in, 1)
    into jso
    do
    begin
      res_tmp = jso;
  
      cur_date = json.obj_get_val(jso, 'last_updated_dt', current_date);
  
      if (max_date < cur_date) then
      begin
        max_date = cur_date;
        res = res_tmp;
      end
    end
  end

  if (res is null) then
    res = json.obj_create();

  return res;
end

begin
  jso = f_bki_bls2json(bls_id);
  jsa_loans = json.obj_get_arr(jso, 'preply.report.account_reply_rutdf');

  cred_no = 0;
  for select distinct f_bki_json_get_date(jso, 'opened_dt', '01.01.1900'), json.obj_get_val(jso, 'uuid', json.obj_get_val(jso, 'acct_num', 'cred_' || i))
  from hp_bki_json_get_list(:jsa_loans, 1)
  order by 1
  into cred_date, cred_uid
  do
  begin
    cred_no = cred_no + 1;
    jso_loan = null;

    select jso
    from hp_bki_json_get_list(:jsa_loans, 1)
    where f_bki_json_get_date(jso, 'opened_dt', '01.01.1900') = :cred_date
      and json.obj_get_val(jso, 'uuid', json.obj_get_val(jso, 'acct_num', 'cred_' || i)) = :cred_uid
    order by f_bki_json_get_date(jso, 'last_updated_dt', '01.01.1900') desc
    rows 1
    into jso_loan;

    if (jso_loan is null) then
      exception e_default f_format('jso_loan not found for cred_no = %1, bls_id = %2', cred_no, bls_id);

    cred_update               = json.obj_get_val(jso_loan, 'last_updated_dt');
    jso_deal                  = my_get_last_el(f_my_obj_get_arr(jso_loan, 'trade'));
    jso_contract_amount       = my_get_last_el(f_my_obj_get_arr(jso_loan, 'account_amt'));
    jso_payments              = my_get_last_el(f_my_obj_get_arr(jso_loan, 'payment'));
    jso_debt                  = my_get_last_el(f_my_obj_get_arr(jso_loan, 'arrear'));
    jso_debt_overdue          = my_get_last_el(f_my_obj_get_arr(jso_loan, 'pastdue_arrear'));

    if (jso_deal is null) then
      continue;

    jsa_collaterals = f_my_obj_get_arr(jso_loan, 'collateral');
    jsa_courts = f_my_obj_get_arr(jso_loan, 'legal_items');
    jsa_changes = f_my_obj_get_arr(jso_loan, 'amendment');
    jsa_guarantee = f_my_obj_get_arr(jso_loan, 'guarantor');

    contract_end_reason       = json.obj_get_val(jso_loan, 'loan_indicator', '');
    cred_end_date_fact_str    = json.obj_get_val(jso_loan, 'loan_indicator_dt', '');
    cred_ratio = json.obj_get_val(jso_deal, 'ownerIndic', null);
    cred_ratio_str = getCatVal('cred_ratio', cred_ratio);

    stop_load_code = json.obj_get_val(my_get_last_el(f_my_obj_get_arr(jso_loan, 'submit_hold')), 'hold_code', '');
    purchaser_commercial_type = json.obj_get_val(my_get_last_el(f_my_obj_get_arr(jso_loan, 'acquirer_legal')), 'right_of_claim_code', '');
    joint_debtors_count = f_bki_json_get_num(my_get_last_el(f_my_obj_get_arr(jso_loan, 'coborrower')), 'solidary_num', 0);
    full_cost = f_bki_json_get_num(my_get_last_el(f_my_obj_get_arr(jso_loan, 'overall_val')), 'credit_total_amt', 0);
    average_payment = f_bki_json_get_num(my_get_last_el(f_my_obj_get_arr(jso_loan, 'month_aver_paymt')), 'aver_paymt_amt', 0);

    cred_sum = f_bki_json_get_num(jso_contract_amount, 'credit_limit', 0);
    cred_end_date = f_bki_json_get_date(jso_deal, 'close_dt', '31.12.2999');

    cred_sum_debt = f_bki_json_get_num(jso_debt, 'amt_outstanding', 0);
    cred_debt_op_sum = f_bki_json_get_num(jso_debt, 'principal_outstanding', 0);
    cred_debt_percent_sum = f_bki_json_get_num(jso_debt, 'int_outstanding', 0);

    cred_sum_overdue = f_bki_json_get_num(jso_debt_overdue, 'amt_past_due', 0);
    cred_sum_overdue_op = f_bki_json_get_num(jso_debt_overdue, 'principal_amt_past_due', 0);
    cred_sum_overdue_percent = f_bki_json_get_num(jso_debt_overdue, 'int_amt_past_due', 0);
    overdue_debt_date = json.obj_get_val(jso_debt_overdue, 'past_due_dt');

    cred_day_overdue = json.obj_get_val(jso_debt_overdue, 'days_past_due', json.obj_get_val(jso_payments, 'days_past_due', 0));
    last_payout_sum = f_bki_json_get_num(jso_payments, 'paymt_amt', 0);
    last_payout_date = f_bki_json_get_date(jso_payments, 'paymt_date');


    begin
      is_our = iif((select obj_id is null from hp_find_cbr_uuid(:cred_uid)), 0, 1);
      when any do is_our = 0;
    end

    cred_end_date_fact = null;
    if (cred_end_date_fact_str <> '') then
    begin
      cred_end_date_fact = cred_end_date_fact_str;
      when any do
        cred_end_date_fact = null;
    end 

    cred_active = case
      when stop_load_code = '3' and purchaser_commercial_type <> '3'  then 2 --'Договор продан (переуступка прав требований)'
      when stop_load_code = '3' and purchaser_commercial_type = '3'   then 5 --'Договор продан коллекторам (переуступка прав требований)'
      when stop_load_code not in ('', '3')                            then 8 --'Договор между Партнером и Бюро об оказании информационных услуг прекратил действие или расторгнут'
      when contract_end_reason = '10'                                 then 4 --'Договор рефинансирован'
      when contract_end_reason not in ('', '10')                      then 0 --'Договор закрыт'
      else 1 --'Договор активен'
    end;

    cred_active_str = case cred_active
      when 0 then  'Договор закрыт'
      when 1 then  'Договор активен'
      when 2 then  'Договор продан (переуступка прав требований)'
      when 4 then  'Договор рефинансирован'
      when 5 then  'Договор продан коллекторам (переуступка прав требований)'
      when 8 then  'Договор между Партнером и Бюро об оказании информационных услуг прекратил действие или расторгнут'
      else 'Неопределен'
    end;

    cred_active_order = case
      when cred_active = 2 then 2
      when cred_active = 5 then 3
      when cred_active in (0,8,4) then 4
      else 1
    end;

    cred_active_order_str = case cred_active_order
      when 1 then 'Активные договоры'
      when 2 then 'Проданные договоры'
      when 3 then 'Переданные коллекторам договоры'
      when 4 then 'Закрытые договоры'
    end;

    cred_partner_type = null;
    if (cred_partner_type_str is not null) then
      cred_partner_type = case cred_partner_type_str
                            when 'REFEREE' then 15
                            when 'BANK' then 1
                            when 'NO_BANK' then 1
                            when 'KPK' then 6
                            when 'COLLECTOR' then 5
                            when 'BROKER' then 4
                            when 'LEASING' then 7
                            when 'LOMBARD' then 8
                            when 'MKK' then 3
                            when 'MFK' then 3
                            when 'MFO' then 3
                            when 'NO_MFO' then 3
                            when 'OIP' then 16
                            when 'RETAIL' then 10
                            when 'STRAKH' then 11
                            when 'FINANCE' then 13
                            else 99
                          end;

    suspend;
  end
