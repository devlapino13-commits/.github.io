CREATE PROCEDURE UXP$PREBLANKS_SIGNER_BY_REGION(
    REP_OFFICE_ID D_INTEGER,
    REP_DATE D_DATE,
    NEED_COMMON_SIGNER D_BOOL)
RETURNS (
    ID D_INTEGER,
    TEXT D_STR_255)
AS
declare variable job_name d_str_255;
declare variable can_sign d_smallint;
declare variable edoc_managers d_str_long;
declare variable c_id d_id;
begin
  if (rep_date is null) then
    rep_date = current_date;

  if (need_common_signer is null) then
    need_common_signer = 0;

  if (rep_office_id is null) then
    exit;

  if (need_common_signer = 1) then
  begin
    for select distinct edc_default_manager_id
    from edoc_cfg
    where edc_default_manager_id is not null
    into c_id
    do
    begin
      id = null;
      select ksr_id
      from kontsignmen, kontsignreg, region_list(:rep_office_id, 1, -1, 0)
      where ksm_from_c_id = :c_id and
            ksr_from_ksm_id = ksm_id and
            ksr_from_reg_id = id_list and
            coalesce(ksr_date_begin, '01.01.1990') <= current_date and
            coalesce(ksr_date_end, '31.12.9999') >= current_date
      order by ksr_is_active desc, ksr_id desc
      rows 1
      into id;

      if (id is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :c_id
        into text;

        if (text = 'Исламов Санжар Якубжанович') then
          Continue;

        suspend;
      end
    end
  end else
  begin
    select list(distinct edc_default_manager_id)
    from edoc_cfg
    where edc_default_manager_id is not null
    into edoc_managers;
  
    for select ksr_id, ksr_fio, ksr_kont_position, ksr_can_sign_loans, ksr_from_c_id
    from print_company(:rep_office_id, 1, :rep_date, 0)
    order by ksr_can_sign_loans desc, ksr_fio
    into :id, :text, :job_name, :can_sign, :c_id
    do
    begin
      job_name = trim(coalesce(job_name, ''));
      if (job_name <> '') then
        text = text || '   - ' || job_name;
  
      if (can_sign = 0) then
      begin
        if (position(c_id, edoc_managers) = 0) then
          Continue;
  
        text = 'Без доверен.: ' || text;
      end
  
      suspend;
    end
  end
end
