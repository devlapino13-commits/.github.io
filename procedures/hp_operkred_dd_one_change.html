CREATE PROCEDURE HP_OPERKRED_DD_ONE_CHANGE(
    K_ID D_ID,
    OL_ID D_ID,
    PLAN_DATE_BEGIN D_DATE,
    PLAN_DATE_END D_DATE,
    NEW_DATE D_DATE)
AS
declare variable sh_id d_integer;
declare variable sh_date d_date;
declare variable aarep_plan_os d_currency;
declare variable aarep_plan_perc d_currency;
begin
  for select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
  from operkred
  where ok_from_k_id = :k_id and
        ok_sh_date between :plan_date_begin and :plan_date_end and
        ok_ol_is_active = 1 and
        ok_ol_date <= :plan_date_begin and
        ok_type <= 6
  group by ok_sh_id, ok_sh_date
  having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
  into sh_id, sh_date, aarep_plan_os, aarep_plan_perc
  do
  begin
    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                          ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
    values (:ol_id, -1, :k_id, :sh_id, :sh_date, 2, 0, 0, 0, 3, 0,
            -1 * :aarep_plan_os, -1 * :aarep_plan_perc, 0, 0);

    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                          ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
    values (:ol_id, -1, :k_id, :sh_id, :new_date, 2, 0, 0, 0, 3, 0, :aarep_plan_os,
            :aarep_plan_perc, 0, 0);
  end
end
