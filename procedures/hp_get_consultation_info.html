CREATE PROCEDURE HP_GET_CONSULTATION_INFO(
    C_ID D_ID,
    REP_DATE DATE)
RETURNS (
    K_ID D_ID,
    SUMM D_CURRENCY,
    KONTRACT_NO D_STR_20,
    REG_NAME VARCHAR(60),
    REG_ID INTEGER,
    OL_DATE DATE,
    RES_TXT D_STR_255,
    DOC_NO D_INTEGER)
AS
declare variable rd_id d_integer;
declare variable ol_date_return d_date;
declare variable count_ d_integer;
begin

  rd_id = f_get_rd_id('CashTransactionIncom.Consultation');

  for select ks.ks_from_k_id, k.k_kontract_no
      from kontstate ks
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id and
            kc.kc_from_c_id = :c_id and
            kc.kc_amount_approved > 0
      inner join kontract k on k.k_id = ks.ks_from_k_id
      inner join product pr on pr.pr_id = k.k_from_pr_id and
            pr.pr_code containing '.consultation.'
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state in (1, 2) and
            :rep_date between ks.ks_date_begin and ks.ks_date_end
      into :k_id, :kontract_no

  do
  begin

    select count(ok.ok_amount), sum(ok.ok_amount), reg.reg_code || ' ' || reg.reg_name,
        reg.reg_unicode, min(ol.ol_date), max(ol.ol_date), ol.ol_doc_no
    from operkontract ok
    inner join operlist ol on ol.ol_id = ok.ok_from_ol_id and
          ol.is_delete = 0 and
          ol.ol_document_client = :c_id
    inner join region reg on ol.ol_from_reg_id = reg.reg_unicode
    where ok.ok_from_k_id = :k_id and
          ok.ok_operation_type = :rd_id
    group by 3, 4, 7
    into :count_, :summ, :reg_name, :reg_id, :ol_date, :ol_date_return, :doc_no;

    select res_txt
    from hp_save_app_return_cons_check(:c_id, :k_id, :rep_date, :ol_date, :summ, :count_, :ol_date_return)
    into :res_txt;

    suspend;

  end
end
