CREATE PROCEDURE IPF_PRODUCT_VIEW(
    K_ID D_ID)
RETURNS (
    ML_NAME D_STR_255,
    AG_CAPTION D_STR_255,
    KML_GUARANTEE_MONTH D_INTEGER,
    KML_COUNT D_CURRENCY,
    KML_COST D_CURRENCY,
    KML_TOTAL D_CURRENCY,
    ML_MODEL D_STR_255,
    ML_ID D_ID,
    KML_ID D_ID)
AS
begin
--RFCKG-4715
--Форма "Правка по продукту ИПФ"
  for select ml.ml_name,
             ag.ag_caption,
             km.kml_guarantee_month,
             km.kml_count,
             km.kml_cost,
             km.kml_total,
             ml.ml_model,
             ml.ml_id,
             km.kml_id
      from kontmat km
      join matlist ml on ml.ml_id = km.kml_from_ml_id
      left join agreements ag on ag.ag_id = km.kml_from_ag_id
      where km.kml_object_kind = 9
            and km.kml_object_id = :k_id
  into
        :ml_name,
        :ag_caption,
        :kml_guarantee_month,
        :kml_count,
        :kml_cost,
        :kml_total,
        :ml_model,
        :ml_id,
        :kml_id
  do
  begin
    suspend;
  end
end
