CREATE PROCEDURE CM_KZ_0039_SQL(
    MA_ID D_INTEGER NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable DECISION_STATUS D_SMALLINT;
declare variable DECISION D_SMALLINT;
declare variable RATING D_STR_CODE;
declare variable GPC_ID D_ID;
begin
  -- RFCKZ-877: автоотказ Fraud Score ГКБ при переводе заявки в договор
  res_info_text = 'Проверка Fraud Score ГКБ';
  res_error_text = '';
  res_error_is_critical = 0;

  if (coalesce(f_get_const('gkb_fraud_is_active'), '0') <> '1') then
    exit;

  select decision_status, decision, rating, gpc_id
  from hp_gkb_fraud_query_status(f_objkind('miniapp', 'id'), :ma_id)
  into :decision_status, :decision, :rating, :gpc_id;

  -- Категория E — сервис был недоступен, заявку не блокируем
  if (decision_status = -3) then
    exit;

  -- Запрос ещё в работе или не создавался
  if (decision_status in (-1, -2)) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Проверка Fraud Score ГКБ не завершена. Пожалуйста, подождите.';
    suspend;
    exit;
  end

  -- Результат получен
  if (decision = 2) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Отказ по результатам проверки Fraud Score.';
    suspend;
  end
  -- decision = 0 или 1 -> пропускаем
end
