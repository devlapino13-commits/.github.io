CREATE PROCEDURE CRM_REFUSAL_LIST(
    CLOSE_REASON_CODE D_STR_100)
RETURNS (
    REFUSAL_CODE STR_100,
    REFUSAL_NAME STR_255)
AS
begin
  close_reason_code = (case close_reason_code when 'website_miniapp_closed_reason.sendtooffice' then 'website_miniapp.sendtoofficeclosereason'
                                              when 'website_miniapp_closed_reason.refuse' then 'website_miniapp.refuse'
                                              when 'loan_close_reason' then 'loan_close_reason'
                                              when 'SVZRefusalReason' then 'SVZRefusalReason'
                                              end);
  if (close_reason_code = 'website_miniapp.sendtoofficeclosereason' or (close_reason_code = 'website_miniapp.refuse'  ) ) then
  begin
    
      for select refusal_code, refusal_name
      from cc_wma_refusal_list_get
      where refusal_type = :close_reason_code
      into :refusal_code, :refusal_name
      do
      begin
        suspend;
      end
  end

  if (close_reason_code = 'loan_close_reason') then
  begin

      for select kr_id, kr_text
      from kontres 
      into :refusal_code, :refusal_name
      do
      begin
        suspend;
      end
  end
  if (close_reason_code = 'SVZRefusalReason') then
  begin 

      for select rd_code, rd_name
      from refdata
      where rd_main_code = :close_reason_code
      into :refusal_code, :refusal_name
      do
      begin
        suspend;
      end
  end
end
