CREATE PROCEDURE PRINT_PERSTEMP_DIFF(
    C_ID ID_TYPE,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    LINE_NO D_INTEGER,
    INFO_NAME D_STR_100,
    INFO_OLD D_STR_255,
    INFO_NEW D_STR_255)
AS
declare variable tmp_id d_id;
begin
  if (coalesce(c_id, 0) = 0 and coalesce(add_params, '') = '') then
    exit;

  if (not exists(select * from clients
                 where c_id = :c_id)) then
  begin
    tmp_id = f_json_str_get_value(add_params, 'c_id', null);
    if (tmp_id is not null) then
      c_id = tmp_id;
  end

  line_no = 0;

  for select line_no, info_name, info_old, info_new
  from hp_perstemp_diff(:c_id)
  into :line_no, :info_name, :info_old, :info_new
  do
    suspend;

  if (line_no = 0) then
  begin
    line_no = null;
    info_name = 'Персональные данные';
    info_old = '***';
    info_new = 'Без изменений';
    suspend;
  end
end
