CREATE PROCEDURE CM_KG_0233_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID ID_TYPE;
declare variable BY_RATING D_SMALLINT;
declare variable MAC_ID D_INTEGER;
begin
  res_error_is_critical = 2;
  for select mac.mac_id, c.c_id, c.c_name_fio, k.k_by_rating
      from kontract k
      inner join miniappclient mac on mac.mac_from_ma_id = k.k_from_ma_app_id and
            mac.mac_amount > 0
      inner join clients c on c.c_id = mac.mac_from_c_id
      inner join product pr on pr.pr_id = k.k_from_pr_id and
            pr.pr_code not containing '.sugarmill.' and
            pr.pr_code not containing '.installment_goods.' and
            pr.pr_code not containing '.online_loan.'
      where k.k_id = :k_id
      into mac_id, c_id, res_info_text, by_rating
  do
  begin
    if (coalesce(by_rating, 0) = 1) then
      exit;

    if (exists(select *
               from cfp_anketa ca
               inner join cfp_positions cp on cp.cp_id = ca.ca_from_cp_id and
                     cp.cp_full_name_ru containing 'Бюджетная сфера' and
                     cp.cp_is_delete = 0
               where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
                     ca.ca_object_id = :mac_id and
                     ca.ca_from_c_id = :c_id)) then
    begin
      if (not exists(select *
                     from extinfolist eil
                     where eil.eil_from_c_id = :c_id and
                           eil.eil_from_eic_id = 10 and
                           eil.eil_status = 2 and
                           coalesce(eil.eil_kk_un_id, 0) = 0)) then
      begin
        if (not exists(select *
                       from clientdocs cds
                       inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
                       inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id and
                             (pdt.pdt_code = 'certificate_of_employment' or pdt.pdt_code = 'pensioners_id')
                       where cds.cds_from_c_id = :c_id and
                             cds.cds_is_delete = 0 and
                             current_date between cds.cds_doc_date_begin and coalesce(cds.cds_doc_date_end, '31.12.9999') and
                             cds.cds_doc_blob_1 > 0)) then
        begin
          res_error_text = 'Справка с места работы/Информация о ЗП не прикреплена';
          suspend;
          exit;
        end
      end
    end
  end
end
