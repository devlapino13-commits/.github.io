CREATE PROCEDURE MKR_ZALOG_LIST(
    K_ID ID_TYPE)
RETURNS (
    COLLATERALTYPECODE VARCHAR(3),
    ANYINFOTODISTING VARCHAR(2048),
    MARKETVALUE CURR_TYPE)
AS
declare variable collateraltypecode_tmp varchar(3);
declare variable z_name str_255;
declare variable z_memo1 varchar(2048);
declare variable z_memo2 varchar(2048);
declare variable marketvalue_tmp curr_type;
declare variable anyinfotodisting_tmp str_255;
declare variable pr_id integer;
begin
  /*
    001 - Недвижимое имущество 
    002 - Движимое имущество
    */

--AnyInfoToDisting

  anyinfotodisting = '';
  marketvalue = 0;

  select kont.k_from_pr_id
  from kontract kont
  where kont.k_id = :k_id
  into :pr_id;

  for select collateraltypecode, z_name, z_memo1, z_memo2, marketvalue
      from (select iif(zal.z_subtype = 2, '001', '002') as collateraltypecode, coalesce(zal.z_name, '') as z_name,
                   coalesce(zal.z_memo1, '') as z_memo1, coalesce(zal.z_memo2, '') as z_memo2,
                   coalesce(zal.z_cost_market, 0) as marketvalue
            from zalog zal
            where zal.z_from_k_id = :k_id and
                  zal.z_type = 1) a
      order by marketvalue desc, collateraltypecode
      into :collateraltypecode_tmp, :z_name, :z_memo1, :z_memo2, :marketvalue_tmp
  do
  begin
    anyinfotodisting_tmp = '';

    if (anyinfotodisting = '') then
    begin
      anyinfotodisting = z_name;
      z_memo1 = trim(z_memo1 || z_memo2);

      if (z_memo1 <> '') then
      begin
        anyinfotodisting = anyinfotodisting || ' (' || z_memo1 || ')';

        if (character_length(anyinfotodisting) > 200) then
          anyinfotodisting = substring(anyinfotodisting from 1 for 200);
      end

      collateraltypecode = collateraltypecode_tmp;
    end
    else
    begin
      if (character_length(anyinfotodisting) < 200) then
      begin
        anyinfotodisting_tmp = z_name;
        z_memo1 = trim(z_memo1 || z_memo2);

        if (z_memo1 <> '') then
        begin
          anyinfotodisting_tmp = anyinfotodisting_tmp || ' (' || z_memo1 || ')';
        end

        if (character_length(anyinfotodisting || '; ' || anyinfotodisting_tmp) <= 200) then
        begin
          anyinfotodisting = anyinfotodisting || '; ' || anyinfotodisting_tmp;
        end
      end
    end
    marketvalue = marketvalue + marketvalue_tmp;
  end

  if (pr_id in (68, 69, 72, 73, 74, 75, 78, 79, 80, 81,
                84, 85)) then
    collateraltypecode = '004';
  if (pr_id in (0, 71, 76, 77, 82, 83)) then
    collateraltypecode = '001';
  if (pr_id in (86, 87)) then
    collateraltypecode = '007';
  if (pr_id in (88, 89, 90, 91)) then
    collateraltypecode = '007';

  if (anyinfotodisting <> '') then
  begin
    anyinfotodisting = left(anyinfotodisting, 200);
    suspend;
  end
end
