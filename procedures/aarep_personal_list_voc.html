CREATE PROCEDURE AAREP_PERSONAL_LIST_VOC(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE,
    REPORT_LIST_KIND INTEGER)
RETURNS (
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    PS_PERIOD_MONTH INTEGER,
    PS_DATE_END_IN_ORDER DATE,
    PS_SALARY DECIMAL(18,4),
    PS_TRANSFER_OFFICE INTEGER,
    PS_TABEL_NO VARCHAR(10),
    HO_DATE DATE,
    HO_FROM_REG_ID INTEGER,
    HO_TEXT VARCHAR(1024),
    HO_REASON VARCHAR(255),
    HO_SAVE_TIME TIMESTAMP,
    HO_ORDER_NO VARCHAR(20),
    JL_NAME VARCHAR(100),
    JL_ORDER_NO INTEGER,
    DEP_NAME VARCHAR(100),
    DEP_ORDER_NO INTEGER,
    REG_NAME VARCHAR(50),
    REG_CODE_NAME VARCHAR(50),
    HO_REG_NAME VARCHAR(50),
    PS_IS_ON_STAFF_TEXT VARCHAR(50),
    PS_IS_WORKING_TEXT VARCHAR(50),
    JL_IS_KRED_PERSON VARCHAR(50),
    JL_KRED_KIND VARCHAR(50),
    JL_IS_FRONT_TEXT VARCHAR(50),
    JL_CATEGORY VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_NATION VARCHAR(100),
    C_CITIZEN VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_EMAIL VARCHAR(100),
    C_FAMILY VARCHAR(100),
    C_EDUCATION VARCHAR(100),
    C_KOL_FAMILY INTEGER,
    C_KOL_CHILD INTEGER,
    C_MEMO VARCHAR(1024),
    C_REG_NO VARCHAR(20),
    C_OKPO VARCHAR(20),
    C_INN VARCHAR(20),
    C_ACCOUNT STR_100,
    C_ORG_FROM_B_ID VARCHAR(10),
    C_HIRED_DATE DATE,
    C_ARMY VARCHAR(100),
    C_PLASTCARD_NO VARCHAR(20),
    C_OTHERS VARCHAR(20),
    C_FULL_ADDRESS VARCHAR(255),
    C_SEX_TEXT VARCHAR(50),
    C_IS_RESIDENT_TEXT VARCHAR(50),
    C_STAGE INTEGER,
    C_STAGE_YEAR INTEGER,
    PS_FROM_C_ID DECIMAL(18,0),
    C_AGE INTEGER,
    JL_ID INTEGER,
    PS_BRANCH VARCHAR(100),
    PS_CURR_STATUS VARCHAR(100),
    JL_KIND INTEGER,
    HRC_NAME VARCHAR(255),
    HRC_IS_FORCE_TXT VARCHAR(30),
    PS_FROM_REG_ID INTEGER,
    C_STAGE_KC_DATE DATE,
    BRANCH_NAME VARCHAR(100),
    REG_STAT_REP_REGION VARCHAR(255),
    SAT_VOC_PAYED DECIMAL(18,4),
    SAT_VOC_END DECIMAL(18,4),
    SAT_VOC_CALC DECIMAL(18,4),
    SAT_VOC_END_YEAR DECIMAL(18,4))
AS
declare variable rep_year integer;
declare variable rep_month integer;
declare variable i integer;
declare variable payed_tmp decimal(18,4);
declare variable calc_tmp decimal(18,4);
declare variable year_end date;
declare variable dd1 date;
declare variable dd2 date;
declare variable hol_count d_smallint;
begin
  rep_year = extract(year from :report_date_end);
  rep_month = extract(month from :report_date_end);
  year_end = cast('31.12.' || :rep_year as date);
  for select ps_date_begin, ps_date_end, ps_period_month, ps_date_end_in_order, ps_salary, ps_transfer_office,
             ps_tabel_no, ho_date, ho_from_reg_id, ho_text, ho_reason, ho_save_time, ho_order_no, jl_name, jl_order_no,
             dep_name, dep_order_no, reg_name, reg_code_name, ho_reg_name, ps_is_on_staff_text, ps_is_working_text,
             jl_is_kred_person, jl_kred_kind, jl_is_front_text, jl_category, c_name_fio, c_birth_date, c_nation,
             c_citizen, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire, c_email, c_family, c_education,
             c_kol_family, c_kol_child, c_memo, c_reg_no, c_okpo, c_inn, c_account, c_org_from_b_id, c_hired_date,
             c_army, c_plastcard_no, c_others, c_full_address, c_sex_text, c_is_resident_text, c_stage, c_stage_year,
             ps_from_c_id, c_age, jl_id, ps_branch, ps_curr_status, jl_kind, hrc_name, hrc_is_force_txt, ps_from_reg_id,
             c_stage_kc_date, branch_name, reg_stat_rep_region
      from aarep_personal_list(:report_region_id, :report_date_end, :report_list_kind)
      into :ps_date_begin, :ps_date_end, :ps_period_month, :ps_date_end_in_order, :ps_salary, :ps_transfer_office,
           :ps_tabel_no, :ho_date, :ho_from_reg_id, :ho_text, :ho_reason, :ho_save_time, :ho_order_no, :jl_name,
           :jl_order_no, :dep_name, :dep_order_no, :reg_name, :reg_code_name, :ho_reg_name, :ps_is_on_staff_text,
           :ps_is_working_text, :jl_is_kred_person, :jl_kred_kind, :jl_is_front_text, :jl_category, :c_name_fio,
           :c_birth_date, :c_nation, :c_citizen, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_email,
           :c_family, :c_education, :c_kol_family, :c_kol_child, :c_memo, :c_reg_no, :c_okpo, :c_inn, :c_account,
           :c_org_from_b_id, :c_hired_date, :c_army, :c_plastcard_no, :c_others, :c_full_address, :c_sex_text,
           :c_is_resident_text, :c_stage, :c_stage_year, :ps_from_c_id, :c_age, :jl_id, :ps_branch, :ps_curr_status,
           :jl_kind, :hrc_name, :hrc_is_force_txt, :ps_from_reg_id, :c_stage_kc_date, :branch_name,
           :reg_stat_rep_region
  do
  begin
    select sat_vocation_end
    from aarep_sa_tabel_one_voc(:ps_from_reg_id, :ps_from_c_id, :rep_year, :rep_month)
    into :sat_voc_end;
    sat_voc_payed = 0;
    sat_voc_calc = 0;
    sat_voc_end_year = sat_voc_end;

    if (rep_month < 12) then
    begin
      i = rep_month + 1;
      while (i <= 12) do
      begin
        for select sat_vocation_curr
            from aarep_sa_tabel_one_voc(:ps_from_reg_id, :ps_from_c_id, :rep_year, :i)
            into :calc_tmp
        do
        begin
          sat_voc_calc = sat_voc_calc + calc_tmp;
        end
        i = i + 1;
      end
    end
    /*
        Было до 23.08.2012 changed by igusev

        for select hrv_date_begin, hrv_date_end from hr_voctravel
        where ((hrv_date_begin>:report_date_end and hrv_date_begin<=:year_end)
        or (hrv_date_end>:report_date_end and hrv_date_end<=:year_end))
        and hrv_from_c_id=:ps_from_c_id
        into :dd1, :dd2
        */

    for select hrv_date_begin, hrv_date_end
        from hr_voctravel, sa_worktype swt, hr_orders
        where ((hrv_date_begin > :report_date_end and
              hrv_date_begin <= :year_end) or
              (hrv_date_end > :report_date_end and
              hrv_date_end <= :year_end)) and
              hrv_from_c_id = :ps_from_c_id and
              ho_order_type = sawt_id and
              hrv_from_ho_id = ho_id and
              hrv_status = 1 and
              swt.sawt_base_type = 4 /* Трудовой отпуск */
              and
              sawt_is_paid = 1 /* опалчиваемы отпуск */
        into :dd1, :dd2
    do
    begin
      if (dd1 <= :report_date_end) then
        dd1 = :report_date_end + 1;
      if (dd2 > :year_end) then
        dd2 = :year_end;

      hol_count = 0;

      select count(*)
      from sa_calendays
      where scd_date between :dd1 and :dd2 and
            scd_work_hours = 0 and
            extract(weekday from scd_date) not in (0, 6)
      into :hol_count;


      payed_tmp = cast((dd2 - dd1 + 1 - hol_count) as integer);
      sat_voc_payed = sat_voc_payed + payed_tmp;
    end
    sat_voc_end_year = sat_voc_end + sat_voc_calc - sat_voc_payed;
    suspend;
  end
end
