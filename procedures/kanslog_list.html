CREATE PROCEDURE KANSLOG_LIST(
    KSO_ID INTEGER)
RETURNS (
    KSG_ID INTEGER,
    KSG_FROM_UN_ID DECIMAL(18,0),
    KSG_DATETIME TIMESTAMP,
    KSG_STATE INTEGER,
    KSG_STATE_TEXT VARCHAR(20),
    KSG_USER_FIO VARCHAR(100))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ksg_id, ksg_from_un_id, ksg_datetime, ksg_state, c_name_fio
      from kanslog, username, clients
      where ksg_from_un_id = un_id and
            un_from_c_id = c_id and
            ksg_from_kso_id = :kso_id
      order by ksg_datetime desc
      into :ksg_id, :ksg_from_un_id, :ksg_datetime, :ksg_state, :ksg_user_fio
  do
  begin
    if (ksg_state = 0) then
      ksg_state_text = 'В процессе';
    if (ksg_state = 1) then
      ksg_state_text = 'Проверен';
    if (ksg_state = 2) then
      ksg_state_text = 'Согласован';
    if (ksg_state = 3) then
      ksg_state_text = 'Утвержден';
    if (ksg_state = 4) then
      ksg_state_text = 'Исполнен';
    suspend;
  end
end
