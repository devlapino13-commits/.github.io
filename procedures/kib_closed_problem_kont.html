CREATE PROCEDURE KIB_CLOSED_PROBLEM_KONT(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID ID_TYPE)
AS
begin
--Процедура предназначена для выявления контрактов, которые были закрыры в FinStart но со статусом "закрыт" не были отправлены в КИБ

  for select ks.ks_from_k_id
      from kontstate ks
      where ks.ks_state = 2 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_ol_is_delete = 0
      into :k_id
  do
  begin
    if (not exists(select *
                   from kibkredit kibkr
                   where kibkr.kkr_from_k_id = :k_id and
                         kibkr.kkr_credstatex_code in ('4-0', '8-0') and
                         kibkr.kkr_transid > 0)) then
    begin
      if (not exists(select *
                     from kib_ignore ki
                     where ki.ki_from_k_id = :k_id)) then
      begin
        suspend;
      end
    end
  end
end
