CREATE PROCEDURE AAREP_PRO_FIZYUR_LOANS_IPF(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    IS_IPF SMALLINT = 0)
RETURNS (
    REP_OBLAST VARCHAR(255),
    REP_BUS_NAME VARCHAR(255),
    REP_BALANCE_BEGIN DECIMAL(18,4),
    REP_BALANCE DECIMAL(18,4),
    REP_BALANCE_BEGIN1000 DECIMAL(18,4),
    REP_BALANCE1000 DECIMAL(18,4),
    REP_DISB_CL_AMOUNT1000 DECIMAL(18,4),
    REP_DISB_CL_KOLICH INTEGER,
    REP_DISB_CL_AMOUNT DECIMAL(18,4),
    REP_PERC_DISB_KOEF DECIMAL(18,7))
AS
declare variable rep_perc_disb_annual decimal(18,4);
declare variable p_rep_balance_begin decimal(18,4);
declare variable p_rep_balance decimal(18,4);
declare variable p_rep_perc_disb_annual decimal(18,4);
declare variable p_rep_disb_cl_amount decimal(18,4);
declare variable p_rep_disb_cl_kolich integer;
declare variable p_rep_perc_disb_koef decimal(18,4);
declare variable p_rep_balance1000 decimal(18,4);
declare variable p_rep_balance_begin1000 decimal(18,4);
declare variable p_rep_disb_cl_amount1000 decimal(18,4);
begin
  for with rep_data
      as (select 'Кыргызстан' as rep_oblast, rep_bus_name, sum(rep_balance_begin) as rep_balance_begin,
                 sum(rep_balance) as rep_balance, sum(rep_perc_disb_annual) as rep_perc_disb_annual,
                 sum(rep_disb_cl_amount) as rep_disb_cl_amount, sum(rep_disb_cl_kolich) as rep_disb_cl_kolich
          from aarep_pro_active_loans_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, 8, :is_ipf)
          group by rep_oblast, rep_bus_name)
      select rep_oblast, rep_bus_name, sum(rep_balance_begin), sum(rep_balance), sum(rep_perc_disb_annual),
             sum(rep_disb_cl_amount), sum(rep_disb_cl_kolich)
      from rep_data
      group by rep_oblast, rep_bus_name
      union
      select rep_oblast, 'х Всего кредитов', sum(rep_balance_begin), sum(rep_balance),
             sum(rep_perc_disb_annual), sum(rep_disb_cl_amount), sum(rep_disb_cl_kolich)
      from rep_data
      group by rep_oblast
      into :rep_oblast, :rep_bus_name, :rep_balance_begin, :rep_balance, :rep_perc_disb_annual, :rep_disb_cl_amount,
           :rep_disb_cl_kolich
  do
  begin
    if (rep_disb_cl_amount <> 0) then
    begin
      rep_perc_disb_koef = cast(rep_perc_disb_annual / rep_disb_cl_amount as decimal(15,7));
      rep_perc_disb_koef = cast(rep_perc_disb_koef * 100 as decimal(15,1));
    end
    else
    begin
      rep_perc_disb_koef = 0;
      rep_perc_disb_koef = 0;
    end

    rep_balance1000 = cast(rep_balance / 1000 as decimal(15,2));
    rep_balance_begin1000 = cast(rep_balance_begin / 1000 as decimal(15,2));
    rep_disb_cl_amount1000 = cast(rep_disb_cl_amount / 1000 as decimal(15,2));
    suspend;
    rep_bus_name = '       юридические лица';
    p_rep_balance_begin = rep_balance_begin;
    p_rep_balance = rep_balance;
    p_rep_perc_disb_annual = rep_perc_disb_annual;
    p_rep_disb_cl_amount = rep_disb_cl_amount;
    p_rep_disb_cl_kolich = rep_disb_cl_kolich;
    p_rep_perc_disb_koef = rep_perc_disb_koef;
    p_rep_balance1000 = rep_balance1000;
    p_rep_balance_begin1000 = rep_balance_begin1000;
    p_rep_disb_cl_amount1000 = rep_disb_cl_amount1000;

    rep_balance_begin = 0;
    rep_balance = 0;
    rep_perc_disb_annual = 0;
    rep_disb_cl_amount = 0;
    rep_disb_cl_kolich = 0;
    rep_perc_disb_koef = 0;
    rep_balance1000 = 0;
    rep_balance_begin1000 = 0;
    rep_disb_cl_amount1000 = 0;
    suspend;
    rep_bus_name = '        физические лица';
    rep_balance_begin = p_rep_balance_begin;
    rep_balance = p_rep_balance;
    rep_perc_disb_annual = p_rep_perc_disb_annual;
    rep_disb_cl_amount = p_rep_disb_cl_amount;
    rep_disb_cl_kolich = p_rep_disb_cl_kolich;
    rep_perc_disb_koef = p_rep_perc_disb_koef;
    rep_balance1000 = p_rep_balance1000;
    rep_balance_begin1000 = p_rep_balance_begin1000;
    rep_disb_cl_amount1000 = p_rep_disb_cl_amount1000;
    suspend;
  end
end
