CREATE PROCEDURE AAREP_SA_TABEL_SYMB_PRINT(
    REPORT_REGION INTEGER,
    REPORT_YEAR INTEGER,
    REPORT_MONTH INTEGER)
RETURNS (
    SAT_FROM_C_ID DECIMAL(18,0),
    SAT_YEAR INTEGER,
    SAT_MONTH INTEGER,
    SAT_FROM_REG_ID INTEGER,
    SAT_OFFICE_ID INTEGER,
    SAT_SDAY_1 VARCHAR(10),
    SAT_SDAY_2 VARCHAR(10),
    SAT_SDAY_3 VARCHAR(10),
    SAT_SDAY_4 VARCHAR(10),
    SAT_SDAY_5 VARCHAR(10),
    SAT_SDAY_6 VARCHAR(10),
    SAT_SDAY_7 VARCHAR(10),
    SAT_SDAY_8 VARCHAR(10),
    SAT_SDAY_9 VARCHAR(10),
    SAT_SDAY_10 VARCHAR(10),
    SAT_SDAY_11 VARCHAR(10),
    SAT_SDAY_12 VARCHAR(10),
    SAT_SDAY_13 VARCHAR(10),
    SAT_SDAY_14 VARCHAR(10),
    SAT_SDAY_15 VARCHAR(10),
    SAT_SDAY_16 VARCHAR(10),
    SAT_SDAY_17 VARCHAR(10),
    SAT_SDAY_18 VARCHAR(10),
    SAT_SDAY_19 VARCHAR(10),
    SAT_SDAY_20 VARCHAR(10),
    SAT_SDAY_21 VARCHAR(10),
    SAT_SDAY_22 VARCHAR(10),
    SAT_SDAY_23 VARCHAR(10),
    SAT_SDAY_24 VARCHAR(10),
    SAT_SDAY_25 VARCHAR(10),
    SAT_SDAY_26 VARCHAR(10),
    SAT_SDAY_27 VARCHAR(10),
    SAT_SDAY_28 VARCHAR(10),
    SAT_SDAY_29 VARCHAR(10),
    SAT_SDAY_30 VARCHAR(10),
    SAT_SDAY_31 VARCHAR(10),
    SAT_VOCATION_PAYED DECIMAL(18,4),
    SAT_VOCATION_BEGIN DECIMAL(18,4),
    SAT_VOCATION_END DECIMAL(18,4),
    SAT_VOCATION_CURR DECIMAL(18,4),
    SAT_OFFICE_NAME VARCHAR(100),
    SAT_PERSONAL_FIO VARCHAR(100),
    SAT_TOTAL_NORMA DECIMAL(18,4),
    SAT_TOTAL_HOURS DECIMAL(18,4),
    SAT_TOTAL_CALC DECIMAL(18,4),
    SAT_NO INTEGER,
    SAT_HOURS DECIMAL(18,4),
    SAT_CALC DECIMAL(18,4),
    SAT_DAYS_TOTAL DECIMAL(18,4),
    SAT_DAYS_CALC DECIMAL(18,4))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  sat_no = 0;
  for select sat_from_c_id, sat_year, sat_month, sat_office_id, max(sat_sday_1), max(sat_sday_2), max(sat_sday_3),
             max(sat_sday_4), max(sat_sday_5), max(sat_sday_6), max(sat_sday_7), max(sat_sday_8), max(sat_sday_9),
             max(sat_sday_10), max(sat_sday_11), max(sat_sday_12), max(sat_sday_13), max(sat_sday_14), max(sat_sday_15),
             max(sat_sday_16), max(sat_sday_17), max(sat_sday_18), max(sat_sday_19), max(sat_sday_20), max(sat_sday_21),
             max(sat_sday_22), max(sat_sday_23), max(sat_sday_24), max(sat_sday_25), max(sat_sday_26), max(sat_sday_27),
             max(sat_sday_28), max(sat_sday_29), max(sat_sday_30), max(sat_sday_31), sum(sat_total_hours),
             sum(sat_total_calc), reg_name, c_name_fio, sum(sat_days_total), sum(sat_days_calc)
      from aarep_sa_tabel_symb_tmp(:report_region, :report_year, :report_month), region, clients
      where reg_unicode = sat_office_id and
            sat_from_c_id = c_id
      group by sat_from_c_id, sat_year, sat_month, sat_office_id, reg_name, c_name_fio
      order by sat_office_id, c_name_fio
      into :sat_from_c_id, :sat_year, :sat_month, :sat_office_id, :sat_sday_1, :sat_sday_2, :sat_sday_3, :sat_sday_4,
           :sat_sday_5, :sat_sday_6, :sat_sday_7, :sat_sday_8, :sat_sday_9, :sat_sday_10, :sat_sday_11, :sat_sday_12,
           :sat_sday_13, :sat_sday_14, :sat_sday_15, :sat_sday_16, :sat_sday_17, :sat_sday_18, :sat_sday_19,
           :sat_sday_20, :sat_sday_21, :sat_sday_22, :sat_sday_23, :sat_sday_24, :sat_sday_25, :sat_sday_26,
           :sat_sday_27, :sat_sday_28, :sat_sday_29, :sat_sday_30, :sat_sday_31, :sat_hours, :sat_calc,
           :sat_office_name, :sat_personal_fio, :sat_days_total, :sat_days_calc
  do
  begin
    sat_vocation_payed = 0;
    sat_vocation_begin = 0;
    sat_vocation_end = 0;
    sat_vocation_curr = 0;
    select sat_vocation_payed, sat_vocation_begin, sat_vocation_end, sat_vocation_curr, sat_total_norma,
           sat_total_hours, sat_total_calc
    from aarep_sa_tabel_one_voc(:sat_office_id, :sat_from_c_id, :report_year, :report_month)
    into :sat_vocation_payed, :sat_vocation_begin, :sat_vocation_end, :sat_vocation_curr, :sat_total_norma,
         :sat_total_hours, :sat_total_calc;

    sat_no = sat_no + 1;
    suspend;
  end
end
