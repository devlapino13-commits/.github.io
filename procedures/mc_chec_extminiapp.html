CREATE PROCEDURE MC_CHEC_EXTMINIAPP(
    C_ID ID_TYPE,
    MCL_ID ID_TYPE)
RETURNS (
    MA_RESULT_TXT D_STR_255)
AS
declare variable c_name_fio d_str_100;
declare variable c_birth_date d_date;
declare variable c_pass_no d_str_long;
declare variable c_inn d_str_20;
declare variable k_id decimal(18,0);
declare variable c_comment d_str_255;
declare variable subject d_str_255;
begin
    -- Предворительная проверка клиента на уровне подачи заявки.
    ma_result_txt = '';

    select
        c.c_name_fio,
        c.c_birth_date,
        c.c_pass_no, 
        c.c_inn
    from clients c
    where c.c_id = :c_id
    into
        :c_name_fio,
        :c_birth_date,
        :c_pass_no,
        :c_inn;


    if (ma_result_txt = ''
        and :c_id is not null
        and exists
        (
            select 1
            from aarep_check_client_fio(:c_id, '', '', '', 0, null, '', 1)
            where c_comment_id = 1
        )
    ) then
        ma_result_txt = 'Вы не можете подать заявку на заем через приложение, обратитесь в офис Компании или к своему специалисту';


    -- RFCKG-2967 Если у клиента есть активная просрочка
    if (ma_result_txt = ''
        and :c_id is not null
        and exists
        (
            select 1
            from oksummary
            inner join knclient on oks_from_k_id = kc_from_k_id
                and kc_amount_approved > 0
                and kc_from_c_id = :c_id
            where current_date between oks_date_begin and oks_date_end
                and current_date - oks_delinq_date > 0
                and oks_close_state in (0, 1)
        )
    ) then
        ma_result_txt = 'У вас имеется активная просрочка';


    -- Если клиент находится в Стоп-листе
    if (ma_result_txt = ''
        and :c_id is not null
        and exists
        (
            select 1
            from hp_blacklist_search(:c_id, :c_name_fio, :c_birth_date, :c_pass_no, :c_inn, f_cont(:c_id, 'cell', 'ok', 'main'), 1, 0, 0)
            where c_comment_id = 1
        )
    ) then
        ma_result_txt = 'Прошу обратиться к своему куратору или в ближайший офис';


    -- активная онлайн-заявка по c_id
    if (ma_result_txt = ''
        and :c_id is not null
        and exists
        (
            select 1
            from extminiapp ema
            where ema.ema_from_c_id = :c_id
                and ema.ema_date between current_date - 30 and current_date
        )
    ) then
        ma_result_txt = 'Вы не можете подать заявку на заем через приложение, обратитесь в офис Компании или к своему специалисту';


    -- активная онлайн-заявка по mcl_id
    if (ma_result_txt = ''
        and :c_id is null
        and exists
        (
            select 1
            from extminiapp ema
            where ema.ema_from_mcl_id = :mcl_id
                and ema.ema_date between current_date - 30 and current_date
        )
    ) then
        ma_result_txt = 'Вы не можете подать заявку на заем через приложение, обратитесь в офис Компании или к своему специалисту';


    --проверка активный кредит
    if (ma_result_txt = ''
        and :c_id is not null
        and exists
        (
            select 1
            from knclient, kontstate
            where kc_from_c_id = :c_id
                and kc_amount_approved > 0
                and kc_from_k_id = ks_from_k_id
                and ks_ol_is_delete = 0
                and ks_state = 1
                and current_date between ks_date_begin and ks_date_end   
        )
    ) then
        ma_result_txt = 'Вы не можете подать заявку на заем через приложение, обратитесь в офис Компании или к своему специалисту';


    --проверка спец учет
    if (ma_result_txt = ''  
        and :c_id is not null
        and exists
        (
            select 1
            from operkroff, knclient
            where kc_from_c_id = :c_id
                and kc_amount_approved > 0
                and kc_from_k_id = oko_from_k_id
                and oko_ol_is_active = 1
           group by oko_from_k_id
           having sum(oko_os + oko_sh_os) > 0 or sum(oko_perc + oko_sh_perc) > 0
        )
    ) then
        ma_result_txt = 'Вы не можете подать заявку на заем через приложение, обратитесь в офис Компании или к своему специалисту';


    -- активная кредитная заявка
    if (ma_result_txt = ''
        and :c_id is not null
        and exists
        (
            select *
            from miniapp
            inner join miniappclient on mac_from_ma_id = ma_id
            left join kontract on k_from_ma_app_id = ma_id
            where ma_date_end = '31.12.9999'
                and iif(k_id is not null, ma_is_active = 1
                and k_status = 0, ma_is_active = 1)
                and ma_date_app between current_date - 30 and current_date
                and mac_from_c_id = :c_id
        )
    ) then
        ma_result_txt = 'Вы не можете подать заявку на заем через приложение, обратитесь в офис Компании или к своему специалисту';


    suspend;
end
