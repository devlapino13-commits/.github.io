CREATE PROCEDURE HP_SA_FIXBONUS_CLEAR(
    REP_DATE_END D_DATE)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable rep_period d_integer;
begin
  rep_period = extract(year from rep_date_end) * 100 + extract(month from rep_date_end);

  if (exists(select * from sa_fixbonus
             where sfb_period > :rep_period)) then
  begin
    res_is_error = 1;
    res_text = 'Удаление допускается только последнего периода';
    Suspend;

    exit;
  end

  res_is_error = 0;

  delete from sa_fixbonus
  where sfb_period = :rep_period;

  if (rep_period > 0) then
    res_text = 'Список за ' || extract(year from rep_date_end) || '-' || right('0' || extract(month from rep_date_end), 2) || ' успешно удален, кол-во: ' || row_count;
  else
    res_text = 'Список пуст';

  suspend;
end
