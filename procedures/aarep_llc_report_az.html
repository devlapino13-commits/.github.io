CREATE PROCEDURE AAREP_LLC_REPORT_AZ(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    CL_CASHIER_NAME VARCHAR(100),
    K_DISB_DATE DATE,
    LK_IS_SECURED_STR VARCHAR(5),
    LK_IS_RESTRUCT_STR VARCHAR(5),
    LK_BALANCE CURR_TYPE,
    LK_CALCPERC CURR_TYPE,
    LK_DELIQ_DAYS INTEGER,
    LK_PERCENT PERC_TYPE,
    RPPU_BALANCE CURR_TYPE,
    RPPU_CALCPERC CURR_TYPE,
    REG_NAME VARCHAR(100))
AS
declare variable k_id decimal(18,0);
declare variable lk_is_secured integer;
declare variable lk_is_restruct integer;
declare variable reg_id_prior integer;
declare variable reg_id integer;
begin
  reg_id_prior = 0;
  for select ks_from_k_id, ks_date_begin, lk_is_secured, lk_is_restruct, lk_balance, lk_calcperc, lk_deliq_days,
             lk_percent, id_list
      from llc_main, llc_kont, kontstate, region_list(:rep_region_id, 1, 3, 1)
      where lm_is_signed = 1 and
            lm_id = lk_from_lm_id and
            :rep_date_end between lk_date and lk_date_end and
            lk_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            lk_from_reg_id = id_list
      into k_id, k_disb_date, lk_is_secured, lk_is_restruct, lk_balance, lk_calcperc, lk_deliq_days, lk_percent,
           reg_id
  do
  begin
    cl_cashier_name = null;
    select k_kontract_no, c_name_fio
    from kontract
    left join clients on (k_clients_cashier_id = c_id)
    where k_id = :k_id
    into k_kontract_no, cl_cashier_name;

    rppu_balance = cast(lk_balance as double precision) * lk_percent / 100;
    rppu_calcperc = cast(lk_calcperc as double precision) * lk_percent / 100;

    if (lk_is_secured is null) then
      lk_is_secured = 0;
    if (lk_is_restruct is null) then
      lk_is_restruct = 0;

    if (lk_is_secured = 1) then
      lk_is_secured_str = 'Да';
    else
      lk_is_secured_str = 'Нет';

    if (lk_is_restruct = 1) then
      lk_is_restruct_str = 'Да';
    else
      lk_is_restruct_str = 'Нет';

    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      reg_name = (select reg_code || ' ' || reg_name
                  from region
                  where reg_unicode = :reg_id);
    end

    suspend;
  end
end
