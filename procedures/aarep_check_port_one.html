CREATE PROCEDURE AAREP_CHECK_PORT_ONE(
    REP_DATE DATE)
RETURNS (
    OL_ID DECIMAL(18,0),
    OL_DATE DATE,
    OL_MEMO VARCHAR(1024),
    OL_KRED_DT DECIMAL(18,4),
    OL_KRED_KT DECIMAL(18,4),
    OL_BOOK_DT DECIMAL(18,4),
    OL_BOOK_KT DECIMAL(15,2))
AS
begin
  for select ol_id, sum(ol_kred_dt), sum(ol_kred_kt), sum(ol_book_dt), sum(ol_book_kt)
      from aarep_check_port_one_tmp(:rep_date)
      group by ol_id
      into :ol_id, :ol_kred_dt, :ol_kred_kt, :ol_book_dt, :ol_book_kt
  do
  begin
    if ((ol_kred_dt <> ol_book_dt) or (ol_kred_kt <> ol_book_kt)) then
    begin
      select ol_date, ol_memo
      from operlist
      where ol_id = :ol_id
      into :ol_date, :ol_memo;
      suspend;
    end
  end
end
