CREATE PROCEDURE CB_RU_LO_2013_11(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    CB_PERSONAL_ID ID_TYPE,
    CB_REGION_ID INTEGER,
    CB_DISB_KOLICH_CLI INTEGER,
    CB_CLOSE_CLI_DO_1_07 INTEGER,
    CB_CLOSE_CLI_S_1_07 INTEGER,
    CB_ACTIV_KOLICH_CLI INTEGER,
    CB_SUM_VZYSKAN CURR_TYPE,
    CB_DELINQ_BALANCE CURR_TYPE,
    CB_ACTIV_BALANCE CURR_TYPE,
    CB_PERSONAL_STAGE DATE,
    PAR7 CURR_TYPE,
    CB_IS_GET INTEGER,
    CB_BONUS_TOTAL CURR_TYPE,
    CB_IS_WORKING INTEGER,
    CB_MEMO VARCHAR(255),
    CB_PORTFEL_C CURR_TYPE,
    CB_PORTFEL_D CURR_TYPE,
    JS_NAME VARCHAR(100),
    CB_DISB_KOLICH_CLI_RE INTEGER,
    CB_CLOSE_CLI_KAT_1_2 INTEGER,
    CB_CLOSE_CLI_KAT_1_2_SB INTEGER,
    PERC_UDERJ CURR_TYPE,
    BONUS_PERCENT INTEGER,
    CB_BONUS_VZYSKAN CURR_TYPE,
    CB_BONUS_DISB_CLOSE CURR_TYPE)
AS
declare variable rep_date_begin date;
declare variable rep_date_end_2 date;
declare variable bonus_issue curr_type;
begin

  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1; -- Начало периода
  rep_date_end_2 = rep_date_begin - 1; -- Конец предыдущего месяца

-- Вытаскивает всех СВЗ которые за текущий месяц какое либо операцию делал или на конец даты есть активный портфель
  for with tabl1
      as (select distinct oks_kc_id
          from oksummary
          where :rep_date_end between oks_date_begin and oks_date_end and
                oks_balance > 0 and
                oks_close_state = 0
          union
          select distinct ks_kc_id
          from kontstate
          where ks_ol_is_delete = 0 and
                cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
                ks_state in (1, 2))
      select distinct oks_kc_id
      from tabl1
      where oks_kc_id <> 1000000000000
--  and oks_kc_id = 1010100014360
            and
            not exists(select *
                       from jobkind, joblist, hr_pstatus
                       where jk_system_code = 'JobSystem.Specialist_SB' and
                             jk_id = jl_kind and
                             jl_id = ps_from_jl_id and
                             :rep_date_end between ps_date_begin and ps_date_end and
                             ps_status = 1 and
                             ps_is_working = 1 and
                             ps_from_c_id = oks_kc_id)
      into cb_personal_id
  do
  begin
-- Системная должность
    js_name = null;
    select rd_name
    from hr_pstatus, joblist, jobkind, refdata
    where ps_from_c_id = :cb_personal_id and
          ps_is_working = 1 and
          ps_status = 1 and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_id = rd_id
    into js_name;
    if (js_name is null) then
      js_name = 'Не определен';
--  Дата Началы работы СВЗ
    cb_personal_stage = null;
    select ps_date_begin
    from hr_pstatus --, hr_orders , joblist
    where ps_from_c_id = :cb_personal_id and
          ps_date_begin is not null and
          ps_status = 1
    order by ps_date_begin
    rows 1
    into :cb_personal_stage;
--  Вычисляет кол-во выдачи данного СВЗ в данном офиса
    cb_disb_kolich_cli = null;
    select sum(k_kolich_client_approved)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :cb_personal_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2
    into cb_disb_kolich_cli;
    if (cb_disb_kolich_cli is null) then
      cb_disb_kolich_cli = 0;
--  Бонус за выдачу
-- пункт 2.4, работающие более одного месяца должны выдать не менее 18 выдач или стаж должно быть меньше месяца
    if (cb_personal_stage > rep_date_begin or cb_disb_kolich_cli >= 18) then
      bonus_issue = 500;
    else
      bonus_issue = 0;

--  Вычисляет кол-во закрытых данного свз по категории 1
--    cb_close_kolich_cli = null;
    cb_close_cli_do_1_07 = null;
    cb_close_cli_s_1_07 = null;

    select sum(iif(k_disb_date < '01.07.2013', k_kolich_client_approved, 0)), -- 1000 руб. - за погашение займа, выданного до 1 июля 2013г.
           sum(iif(k_disb_date >= '01.07.2013', k_kolich_client_approved, 0)) -- 500 руб. - за погашение займа, выданного c 1 июля 2013г.
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 2 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :cb_personal_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 and
          (select aa_category_id
           from aarep_get_category_id(k_id, ks_date_begin - 1, null, null, null, null, null, null)) = 1 and
          not exists(select *
                     from sb_akt, sb_list -- добавлено по приказу №302-06-13, пункт 2, не включать закрытые по СБ
                     where sba_is_delete = 0 and
                           sba_from_sbl_id <> 0 and
                           sba_status = 2 and
                           ks_date_begin between sba_date_begin and sba_date_end and
                           sba_from_k_id = ks_from_k_id and
                           sba_from_sbl_id = sbl_id and
                           sbl_type = 1) and
          not(k_disb_date >= '01.12.2012' and
          ks_date_begin + 10 < k_kredline_end) -- если кредит выдан от 01.12.2012 и после, и закрыт досрочно то их не включает
          and
          not exists(select *
                     from knaktlist -- Пункт 2.5, Если кредит находился 2 и более месяцев у Виртуального СВЗ то за закрытие не будет считатся
                     where knaktlist.is_delete = 0 and
                           kal_new_kf_officer = 1000000000000 and
                           kal_date <= iif(kal_date_end < :rep_date_end, dateadd(month, -2, kal_date_end), dateadd(month, -2, cast(:rep_date_end as date))) and
                           kal_from_kf_id = k_from_kf_id and
                           kal_date_end > k_disb_date and
                           kal_date < :rep_date_end)

    into cb_close_cli_do_1_07, cb_close_cli_s_1_07;
    if (cb_close_cli_do_1_07 is null) then
      cb_close_cli_do_1_07 = 0;
    if (cb_close_cli_s_1_07 is null) then
      cb_close_cli_s_1_07 = 0;

--  Вычисляет кол-во активных клиентво по категории 1 и 2
--  Письмо от Екатерины Бутовой от 12.07.2013
--  Активные клиенты 1 и 2 категории, у которых на руках есть остаток основной суммы займа на отчетную дату.
--  Те клиенты, которые погасили сумму займа, но числятся в БД как активные из-за доначисленных процентов, не входят в число активных клиентов в целях начисления бонусов.
    cb_activ_kolich_cli = null;
    select count(distinct kc_from_c_id)
    from knclient, oksummary
    where kc_amount_approved > 0 and
          kc_from_k_id = oks_from_k_id and
          oks_balance > 0 and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_kc_id = :cb_personal_id and
          oks_close_state = 0 and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
    into :cb_activ_kolich_cli;
    if (cb_activ_kolich_cli is null) then
      cb_activ_kolich_cli = 0;
--  Сумма взыскании по Категории 2
    cb_sum_vzyskan = null;
    select -sum(iif((select aa_category_id
                     from aarep_get_category_id(k_id, ok_ol_date, null, null, null, null, null, null)) = 2, ok_os + ok_perc + ok_fine, 0))
    from operkred, kontract, knaktlist
    where ok_ol_is_active = 1 and
          ok_ol_date between :rep_date_begin and :rep_date_end and
          ok_from_c_id = 0 and
          ok_type = 4 and
          ok_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          ok_ol_date between kal_date and kal_date_end and
          kal_new_kf_officer = :cb_personal_id and
          not exists(select *
                     from sb_akt, sb_list -- добавлено по приказу №302-06-13, пункт 2, не включать закрытые по СБ
                     where sba_is_delete = 0 and
                           sba_from_sbl_id <> 0 and
                           sba_status = 2 and
                           ok_ol_date between sba_date_begin and sba_date_end and
                           sba_from_k_id = k_id and
                           sba_from_sbl_id = sbl_id and
                           sbl_type = 1)
    into cb_sum_vzyskan;
    if (cb_sum_vzyskan is null) then
      cb_sum_vzyskan = 0;
--  Просроченный баланс категории 1 от 8 до 180 дней
    cb_delinq_balance = null;
    select sum(oks_balance)
    from oksummary
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_kc_id = :cb_personal_id and
          oks_delinq_date < :rep_date_end and
          :rep_date_end - oks_delinq_date between 8 and 180 and
          oks_close_state = 0 and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 1
    into cb_delinq_balance;
    if (cb_delinq_balance is null) then
      cb_delinq_balance = 0;
--  Расчет баланса КС на конец предыдущего месяца по Категории 1; B - общий портфель катег.1 на предыдущую отчетную дату
    cb_activ_balance = null;
    select sum(oks_balance)
    from oksummary
    where :rep_date_end_2 between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :cb_personal_id and
          oks_close_state = 0 and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_end_2, null, null, null, null, null, null)) = 1
    into :cb_activ_balance;
    if (:cb_activ_balance is null) then
      cb_activ_balance = 0;
-- С - Портфель займов, переданный в категорию 1 от другого СВЗ в течение отчетного месяца
    cb_portfel_c = null;
    select sum(iif((select aa_category_id
                    from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 1,
           (select oks_balance
            from oksummary
            where oks_from_k_id = k_id and
                  oks_close_state = 0 and
                  :rep_date_end between oks_date_begin and oks_date_end), 0))
    from kontstate, kontract, knaktlist kal1, knaktlist kal2
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal1.kal_from_kf_id and
          kal1.is_delete = 0 and
          :rep_date_end between kal1.kal_date and kal1.kal_date_end and
          kal1.kal_new_kf_officer = :cb_personal_id and
          k_from_kf_id = kal2.kal_from_kf_id and
          kal2.is_delete = 0 and
          :rep_date_begin between kal2.kal_date and kal2.kal_date_end and
          kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer and
          ks_date_begin < kal1.kal_date -- после передачи выданные группы не включает
    into cb_portfel_c;
    if (cb_portfel_c is null) then
      cb_portfel_c = 0;
-- D - Портфель займов, переданный в категорию 1 от данного СВЗ другому СВЗ в течение отчетного месяца
    cb_portfel_d = null;
    select sum(iif((select aa_category_id
                    from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 1,
           (select oks_balance
            from oksummary
            where oks_from_k_id = k_id and
                  oks_close_state = 0 and
                  :rep_date_end between oks_date_begin and oks_date_end), 0))
    from kontstate, kontract, knaktlist kal1, knaktlist kal2
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal1.kal_from_kf_id and
          kal1.is_delete = 0 and
          :rep_date_end between kal1.kal_date and kal1.kal_date_end and
          k_from_kf_id = kal2.kal_from_kf_id and
          kal2.is_delete = 0 and
          :rep_date_begin between kal2.kal_date and kal2.kal_date_end and
          kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer and
          kal2.kal_new_kf_officer = :cb_personal_id and
          ks_date_begin < kal1.kal_date -- после передачи выданные группы не включает
    into cb_portfel_d;
    if (cb_portfel_d is null) then
      cb_portfel_d = 0;
--  Расчет процента удержания пункт 2.10
--  А - Кол-во клиентов, повтоно получивших займы в отчетном периоде
--  повторником не будем считать тех клиентов которые 6 и более месяцев не был активным клиентом
    cb_disb_kolich_cli_re = null;
    select count(kc_from_c_id)
    from kontstate ks1, knclient kc1, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          cast(ks_date_end as date) >= :rep_date_end -- чтобы исключить 2 раза выданные за отчетный месяц т.е. на дату конец клиент должен быть активным
          and
          ks_from_k_id = kc_from_k_id and
          ks_kc_id = :cb_personal_id and
          kc_amount_approved > 0 and
          kc_circle > 1 and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 and
          exists(-- Проверка, клиент не отдахл 6 месяцев
    select *
    from kontstate ks2, knclient kc2, kontract, product
    where ks2.ks_ol_is_delete = 0 and
          ks2.ks_state = 1 and
          ks2.ks_date_begin < :rep_date_begin -- дата выдачи в прошлом месяце должно быть,
          and
          kc2.kc_amount_approved > 0 and
          ks2.ks_date_end >= dateadd(month, -6, ks1.ks_date_begin) - 1 and
          ks2.ks_from_k_id = kc2.kc_from_k_id and
          kc2.kc_from_c_id = kc1.kc_from_c_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2)

          and
          exists(select *
                 from kontstate ks3, knclient kc3, kontract, product -- Для того чтобы брать погашенные за отчетный период
                 where ks3.ks_ol_is_delete = 0 and
                       ks3.ks_state = 2 and
                       cast(ks3.ks_date_begin as date) between :rep_date_begin and :rep_date_end and
                       ks3.ks_from_k_id = kc3.kc_from_k_id and
                       kc3.kc_amount_approved > 0 and
                       kc3.kc_from_c_id = kc1.kc_from_c_id and
                       ks_from_k_id = k_id and
                       k_from_pr_id = pr_id and
                       pr_product_kind <> 2 and
                       (select aa_category_id
                        from aarep_get_category_id(ks3.ks_from_k_id, ks3.ks_date_begin - 1, null, null, null, null, null, null)) = 1)

    into cb_disb_kolich_cli_re;
    if (cb_disb_kolich_cli_re is null) then
      cb_disb_kolich_cli_re = 0;
--  В - Общее кол-во клиентов, категории 1 и 2, полностью погасивших займы в отчетном периоде
    cb_close_cli_kat_1_2 = null;
    select sum(k_kolich_client_approved)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 2 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :cb_personal_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 and
          (select aa_category_id
           from aarep_get_category_id(k_id, ks_date_begin - 1, null, null, null, null, null, null)) = 1
    into cb_close_cli_kat_1_2;
    if (cb_close_cli_kat_1_2 is null) then
      cb_close_cli_kat_1_2 = 0;
--  С - Кол-во клиентов категории 1 и 2, переданные в СБ, которые закрылись за отчетный период
    cb_close_cli_kat_1_2_sb = null;
    select sum(k_kolich_client_approved)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 2 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :cb_personal_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 and
          (select aa_category_id
           from aarep_get_category_id(k_id, ks_date_begin - 1, null, null, null, null, null, null)) = 1 and
          exists(select *
                 from sb_akt, sb_list -- добавлено по приказу №302-06-13, пункт 2, не включать закрытые по СБ
                 where sba_is_delete = 0 and
                       sba_from_sbl_id <> 0 and
                       sba_status = 2 and
                       ks_date_begin between sba_date_begin and sba_date_end and
                       sba_from_k_id = k_id and
                       sba_from_sbl_id = sbl_id and
                       sbl_type = 1)
    into cb_close_cli_kat_1_2_sb;
    if (cb_close_cli_kat_1_2_sb is null) then
      cb_close_cli_kat_1_2_sb = 0;

    if (cb_disb_kolich_cli_re = 0) then
    begin
      perc_uderj = 0;
    end
    else
    if (cb_close_cli_kat_1_2 - cb_close_cli_kat_1_2_sb > 0) then
      perc_uderj = cast(cb_disb_kolich_cli_re as double precision) / (cb_close_cli_kat_1_2 - cb_close_cli_kat_1_2_sb) * 100;
    else
      perc_uderj = 100;

-- Вычисляет стаж КС
    cb_is_working = null;
    cb_region_id = null;
--    Cтатус на конец даты,  0 - не работает; 1 - работает;
    select ps_is_working, ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :cb_personal_id and
          ps_date_begin is not null and
          ps_status = 1 and
          :rep_date_end between ps_date_begin and ps_date_end
    into cb_is_working, cb_region_id;
    if (cb_is_working is null) then
      cb_is_working = 0;

--   if (cb_personal_stage is null) then cb_personal_stage = :rep_date_end;
--   cb_personal_stage_month = datediff(day, :cb_personal_stage, :rep_date_end) / 30;

    cb_is_get = 1;
    cb_memo = '';
--   if (cb_personal_stage_month < 1) then
--   begin
--     cb_is_get = 0;
--     cb_memo = 'Бонус не положено из-за стажа.';
--   end
-- По положению от 27.02.2013
    if (cb_delinq_balance = 0) then
    begin
      par7 = 0;
    end
    else
    begin
      if (cb_activ_balance + cb_portfel_c - cb_portfel_d > 0) then
      begin
        par7 = cb_delinq_balance / (cb_activ_balance + cb_portfel_c - cb_portfel_d) * 100;
      end
      else
      begin
        par7 = 100;
      end
    end

-- Разработано по пункту 3.1.1
    if (not(par7 <= 7.0 and
        cb_is_working = 1) and
        cb_is_get = 1) then
    begin
      cb_is_get = 0;
      if (cb_is_working <> 1) then
        cb_memo = 'Сотрудник в данном офисе не работает.';
      else
        cb_memo = 'Бонус не положено из-за PAR > 7.';
    end
-- Пункт 2.9
    bonus_percent = 0;
    if (cb_close_cli_kat_1_2 < 1) then
      bonus_percent = 100;
    else
    if (par7 < 3) then
    begin
      bonus_percent = case
                        when perc_uderj < 50 then 90
                        when perc_uderj >= 50 and perc_uderj < 60 then 120
                        when perc_uderj >= 60 and perc_uderj < 80 then 130
                        when perc_uderj >= 80 then 150
                      end;
    end
    else
    if (par7 between 3 and 7) then
    begin
      bonus_percent = case
                        when perc_uderj < 50 then 70
                        when perc_uderj >= 50 and perc_uderj < 60 then 100
                        when perc_uderj >= 60 and perc_uderj < 80 then 110
                        when perc_uderj >= 80 then 130
                      end;
    end

    cb_bonus_total = 0;
    cb_bonus_vzyskan = 0;
    cb_bonus_disb_close = 0;
    if (cb_is_get = 1) then
    begin
      cb_bonus_vzyskan = cb_sum_vzyskan * 8 / 100;
      if (cb_activ_kolich_cli >= 200) then
      begin
        cb_bonus_disb_close = (cb_disb_kolich_cli + cb_close_cli_s_1_07) * 500 + (cb_close_cli_do_1_07 * 1000);
      end
      else
        cb_bonus_disb_close = (cb_disb_kolich_cli * bonus_issue) + iif(cb_disb_kolich_cli >= 18, (cb_close_cli_do_1_07 * 1000 + cb_close_cli_s_1_07 * 500), 0);

      cb_bonus_total = cb_bonus_disb_close * bonus_percent / 100 + cb_bonus_vzyskan;
    end

    suspend;
  end
