CREATE PROCEDURE HP_APPMERCHANTSHOP_VIEW(
    IQA_ID D_ID)
RETURNS (
    IQS_ID D_ID,
    IQS_NO D_INTEGER,
    IQS_FROM_CS_ID D_ID,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    ADL_ADDRESS_STR D_STR_255,
    IQS_SHOP_NAME D_STR_255,
    IQS_BANK_DATA D_STR_100,
    IQS_MIN_TRANS_AMOUNT D_CURRENCY,
    IQS_MAX_TRANS_AMOUNT D_CURRENCY,
    IQS_HAS_QR D_BOOL)
AS
declare variable adl_id d_id;
declare variable rd_id_tsp d_integer;
begin
  iqs_bank_data = 'БИК 507001, расчетный счет 1013890209990149';
  iqs_min_trans_amount = 1;
  iqs_max_trans_amount = 200000;
  rd_id_tsp = f_get_rd_id('AgreementKinds.tsp');

  for
    select
      iqs_id,
      iqs_no,
      iqs_from_cs_id,
      c_id,
      c_name_fio,
      cs_from_adl_id,
      iqs_shop_name,
      iqs_has_qr
    from ipc_qr_app_shops
      inner join clientshops
        on cs_id = iqs_from_cs_id and
        cs_from_rd_id = :rd_id_tsp
      inner join clients
        on c_id = cs_from_c_id
    where iqs_from_iqa_id = :iqa_id
  into
    :iqs_id,
    :iqs_no,
    :iqs_from_cs_id,
    :c_id,
    :c_name_fio,
    :adl_id,
    :iqs_shop_name,
    :iqs_has_qr
  do
  begin
    adl_address_str = '';
    select
      adl_address_str
    from addrlist
    where adl_id = :adl_id
    into
      :adl_address_str;
    suspend;
  end
end
