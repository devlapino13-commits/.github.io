CREATE PROCEDURE CM_KZ_0021_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_NAME_FIO D_STR_100;
declare variable C_ID D_ID;
declare variable IS_OVERDUE D_BOOL;
declare variable IS_BANKROT D_BOOL;
declare variable ERROR_TMP D_STR_255;
declare variable eil_id d_id;
declare variable eil_status d_smallint;
begin        
  if (k_id = 101010000739300101) then
    exit; -- RFCKZ-790
    
  -- RFCKZ-222
  error_tmp = 'по данным негативной истории ПКБ ';
  res_error_text = '';
  res_info_text = '';

  for select c.c_id,
             c.c_name_fio
  from knclient kc
  inner join clients c on c.c_id = kc.kc_from_c_id
  where kc.kc_amount_applicate > 0 and
        kc.kc_from_k_id = :k_id
  into c_id,
       c_name_fio
  do
  begin
    if (exists(select 1
               from extinfolist eil
               inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
               where eic.eic_code = 'pkb_available_reports' and
                     eil.eil_from_c_id = :c_id and
                     eil.eil_status = 2
               )) then
    begin
      if ((select priority_id
           from hp_get_priority_rep_id(f_objkind('kontract', 'id'), :k_id)) = -1) then        -- У клиента отсутствует кредитная история
        continue;
    end

    select nkh.nkh_is_overdue,
           nkh.nkh_is_bankrot,
           nkh.nkh_from_eil_id,
           eil.eil_status
    from negativ_kib_history nkh
    inner join extinfolist eil on eil.eil_id = nkh.nkh_from_eil_id
    where nkh.nkh_from_c_id = :c_id and
          nkh.nkh_last = 1
    rows 1
    into is_overdue,
         is_bankrot,
         eil_id,
         eil_status;

    if (coalesce(eil_id, 0) = 0) then                               -- RFCKZ-222
      res_error_text = 'Запрос на получение "Негативной истории клиента в ПКБ" не отправлен';

    if (eil_status < 2) then
      res_error_text = 'Запрос на получение "Негативной истории клиента в ПКБ" еще не обработан';

    if (eil_status = -2) then
      res_error_text = 'Во время запроса на получение "Негативной истории клиента в ПКБ" возникла ошибка';

    if (is_overdue = 1 or is_bankrot = 1) then
    begin
      res_error_text = c_name_fio || ': ';

      if (is_overdue = 1) then
        error_tmp = error_tmp || 'у клиента есть задолжность 30 дней и более';
      if (is_bankrot = 1) then
        error_tmp = error_tmp || iif(error_tmp <> '', ', ', '') || 'клиент проходит процедуру банкротства';

      res_error_text = res_error_text || error_tmp;
    end

    if (res_error_text <> '') then
    begin
      res_error_is_critical = 2;
      suspend;
      exit;
    end
  end
end
