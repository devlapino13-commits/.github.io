CREATE PROCEDURE HP_ACTIONLIST_IN_PROCESS_TEST(
    SERVICE_KIND VARCHAR(255),
    UN_ID DECIMAL(18,0),
    WITH_DETAIL INTEGER)
RETURNS (
    REC_ID INTEGER,
    REC_PARENT_ID INTEGER,
    RD_ID INTEGER,
    RD_CODE VARCHAR(255),
    ACL_OBJECT_KIND INTEGER,
    RES_BOX_NAME VARCHAR(100),
    RES_COUNT INTEGER,
    RES_STATUS INTEGER,
    RES_IS_OWN INTEGER,
    ACL_ID DECIMAL(18,0),
    RES_PICTURE_ID INTEGER,
    RES_NODE_MEMO VARCHAR(255),
    RES_IS_BOLD INTEGER,
    RES_CAN_SELECT INTEGER,
    RD_STR_ID VARCHAR(100))
AS
declare variable c_id decimal(18,0);
declare variable id_new_in_waiting integer;
declare variable id_own_in_waiting integer;
declare variable id_new_in_process integer;
declare variable id_own_in_process integer;
declare variable res_new_in_waiting integer;
declare variable res_own_in_waiting integer;
declare variable res_new_in_process integer;
declare variable res_own_in_process integer;
declare variable all_new_in_waiting integer;
declare variable all_own_in_waiting integer;
declare variable all_new_in_process integer;
declare variable all_own_in_process integer;
declare variable rd_name varchar(255);
declare variable acl_object_id decimal(18,0);
declare variable id_rd_code_line integer;
declare variable rec_id_last integer;
begin
-- Service_kind - имя службы для которой выдается список задач в очереди
-- Service_kind будет закрепляться в свойстве RD_Str_ID элемента ActionGroup
-- Текущие варианты значений Service_kind:
-- UnderWriter, Digitising

-- в будущем могут быть Digitising, CallCenter_Calling, LO_Calling, Visit_Monitoring, Visit_Inspection и т.д.
  res_node_memo = '';
  res_is_bold = 0;
  res_picture_id = 6;
  if (coalesce(service_kind, '') = '') then
    exit;

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  res_can_select = 0;
  rec_id = 0;
  rec_parent_id = null;
  all_new_in_waiting = 0;
  all_own_in_waiting = 0;
  all_new_in_process = 0;
  all_own_in_process = 0;
  acl_id = null;

  for select rd_id, rd_code, rd_name, rd_str_id
      from hp_actionlist_un_access(:service_kind, :un_id)
      into :rd_id, :rd_code, :rd_name, :rd_str_id
  do
  begin
    rec_id = rec_id + 1;
    id_rd_code_line = rec_id;
    rec_id = rec_id + 1;
    id_new_in_waiting = rec_id;
    rec_id = rec_id + 1;
    id_own_in_waiting = rec_id;
    rec_id = rec_id + 1;
    id_new_in_process = rec_id;
    rec_id = rec_id + 1;
    id_own_in_process = rec_id;

    all_new_in_waiting = 0;
    all_own_in_waiting = 0;
    all_new_in_process = 0;
    all_own_in_process = 0;

    res_new_in_waiting = 0;
    res_own_in_waiting = 0;
    res_new_in_process = 0;
    res_own_in_process = 0;

    for select aco_object_kind, sum(iif(aco_status = 0 and aco_respons_c_id is null, 1, 0)),
               sum(iif(aco_status = 0 and aco_respons_c_id is not null, 1, 0)),
               sum(iif(aco_status = 1 and aco_respons_c_id is null, 1, 0)),
               sum(iif(aco_status = 1 and aco_respons_c_id is not null, 1, 0))
        from actionorder, user_reg_list(:un_id)
        where (aco_respons_c_id = :c_id or
              aco_respons_c_id is null) and
              aco_from_rd_id = :rd_id and
              aco_from_reg_id = r_list
        group by aco_object_kind
        into acl_object_kind, :res_new_in_waiting, :res_own_in_waiting, :res_new_in_process, :res_own_in_process
    do
    begin
      acl_id = null;

      if (res_new_in_waiting is null) then
        res_new_in_waiting = 0;
      if (res_own_in_waiting is null) then
        res_own_in_waiting = 0;
      if (res_new_in_process is null) then
        res_new_in_process = 0;
      if (res_own_in_process is null) then
        res_own_in_process = 0;

      if (res_own_in_process <> 0) then
      begin
-- Мои в обработке
        rec_id = rec_id + 1;
        rec_parent_id = id_own_in_process;
        res_picture_id = 5;
        res_status = 1;
        res_is_own = 1;
        res_box_name = rd_name;
        res_count = res_own_in_process;
        all_own_in_process = all_own_in_process + res_own_in_process;
        res_box_name = res_box_name || ' (' || res_count || ')';
        res_is_bold = 1;
        res_node_memo = 'Раскройте ветку и выберите объект для обработки';
        res_can_select = 1;
        suspend;

        if (with_detail = 1) then
        begin
          rec_parent_id = rec_id;
          res_picture_id = 6;
          res_is_bold = 0;
          res_node_memo = 'Нажмите два раза чтоб открыть детали';

          for select aco_id, aco_object_id
              from actionorder
              where aco_respons_c_id = :c_id and
                    aco_status = :res_status and
                    aco_object_kind = :acl_object_kind
              into acl_id, acl_object_id
          do
          begin
            rec_id = rec_id + 1;
            res_box_name = '';
            select res_object_name
            from hp_actionlist_object_name(:acl_object_kind, :acl_object_id)
            into :res_box_name;
            suspend;
          end
          acl_id = null;
        end
      end
      if (res_own_in_waiting <> 0) then
      begin
-- Мои в очереди
        rec_id = rec_id + 1;

        rec_parent_id = id_own_in_waiting;
        res_status = 0;
        res_is_own = 1;
        res_picture_id = 5;
        res_box_name = rd_name;
        res_count = res_own_in_waiting;
        all_own_in_waiting = all_own_in_waiting + res_own_in_waiting;
        res_box_name = res_box_name || ' (' || res_count || ')';
        res_is_bold = 1;
        res_node_memo = 'Раскройте ветку и выберите объект для начала обработки';
        res_can_select = 1;
        suspend;

        if (with_detail = 1) then
        begin
          res_node_memo = 'Нажмите два раза чтоб открыть детали';
          rec_parent_id = rec_id;
          res_picture_id = 6;
          res_is_bold = 0;

          for select aco_id, aco_object_id
              from actionorder
              where aco_respons_c_id = :c_id and
                    aco_status = :res_status and
                    aco_object_kind = :acl_object_kind
              into acl_id, acl_object_id
          do
          begin
            rec_id = rec_id + 1;
            res_box_name = '';
            select res_object_name
            from hp_actionlist_object_name(:acl_object_kind, :acl_object_id)
            into :res_box_name;
            suspend;
          end
          acl_id = null;
        end

      end
      if (res_new_in_process <> 0) then
      begin
-- В обработке у других
        rec_id = rec_id + 1;

        rec_parent_id = id_new_in_process;
        res_status = 1;
        res_picture_id = 5;
        res_is_own = 0;
        res_box_name = rd_name;
        res_count = res_new_in_process;
        all_new_in_process = all_new_in_process + res_new_in_process;
        res_box_name = res_box_name || ' (' || res_count || ')';
        res_is_bold = 1;
        res_node_memo = 'Элементы не доступны вам для обработки';
        res_can_select = 0;
        suspend;
      end

      if (res_new_in_waiting <> 0) then
      begin
-- Новые в ожидании
        rec_id = rec_id + 1;

        rec_parent_id = id_new_in_waiting;
        res_picture_id = 5;
        res_status = 0;
        res_is_own = 0;
        res_box_name = rd_name;
        res_count = res_new_in_waiting;
        all_new_in_waiting = all_new_in_waiting + res_new_in_waiting;
        res_box_name = res_box_name || ' (' || res_count || ')';
        res_is_bold = 1;
        res_node_memo = 'Нажмите два раза чтоб получить очередную для обработки';
        res_can_select = 1;
        suspend;
      end
    end

    acl_id = null;

    res_can_select = 0;
--    RD_CODE='';
--    RD_ID=0;
--    rd_str_id='';
    res_is_bold = 0;

    rec_id_last = rec_id;

    rec_id = id_rd_code_line;
    rec_parent_id = null;
    res_picture_id = 0;
    res_status = -1;
    res_is_own = -1;
    res_node_memo = '';
    res_box_name = rd_name;
    res_node_memo = 'Здесь будут ваши объекты в обработке';
    res_count = all_new_in_waiting + all_own_in_waiting + all_new_in_process + all_own_in_process;
    suspend;

    rec_parent_id = id_rd_code_line;

    rec_id = id_own_in_process;
    res_picture_id = 1;
    res_status = 1;
    res_is_own = 1;
    res_box_name = 'Мои в обработке';
    res_count = all_own_in_process;
    res_is_bold = 0;
    res_node_memo = 'Здесь будут ваши объекты в обработке';
    res_can_select = 0;
    if (res_count <> 0) then
    begin
      res_can_select = 1;
      res_node_memo = 'Объекты в режиме ожидания вашего решения';
      res_is_bold = 1;
      res_box_name = res_box_name || ' (' || res_count || ')';
    end
    suspend;

    rec_id = id_own_in_waiting;
    res_picture_id = 2;
    res_status = 0;
    res_is_own = 1;
    res_box_name = 'Мои в очереди';
    res_count = all_own_in_waiting;
    res_is_bold = 0;
    res_node_memo = 'Здесь будут объекты в очереди, закрепленные за вами';
    res_can_select = 0;
    if (res_count <> 0) then
    begin
      res_can_select = 1;
      res_node_memo = 'Объекты следующие в очереди';
      res_is_bold = 1;
      res_box_name = res_box_name || ' (' || res_count || ')';
    end
    suspend;

    rec_id = id_new_in_process;
    res_picture_id = 3;
    res_status = 1;
    res_is_own = 0;
    res_box_name = 'В обработке у других';
    res_count = all_new_in_process;
    res_is_bold = 0;
    res_node_memo = 'Здесь находятся объекты в очереди, закрепленные за другими сотрудниками';
    res_can_select = 0;
    if (res_count <> 0) then
    begin
      res_node_memo = 'Объекты не доступны для вас';
      res_is_bold = 1;
      res_box_name = res_box_name || ' (' || res_count || ')';
    end
    suspend;

    rec_id = id_new_in_waiting;
    res_picture_id = 4;
    res_status = 0;
    res_is_own = 0;
    res_box_name = 'Новые в ожидании';
    res_count = all_new_in_waiting;
    res_is_bold = 0;
    res_node_memo = 'Здесь будут находиться находятся объекты в общей очереди';
    res_can_select = 0;
    if (res_count <> 0) then
    begin
      res_can_select = 1;
      res_node_memo = 'Нажмите два раза, чтоб забрать очередной объект на обработку';
      res_is_bold = 1;
      res_box_name = res_box_name || ' (' || res_count || ')';
    end
    suspend;

    rec_id = rec_id_last;

  end
end
