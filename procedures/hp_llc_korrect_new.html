CREATE PROCEDURE HP_LLC_KORRECT_NEW(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    KIND SMALLINT,
    K_KONTRACT_NO VARCHAR(20),
    PR_LOAN_TYPE SMALLINT,
    K_DELIQ_DAYS INTEGER,
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    K_FROM_KF_ID DECIMAL(18,0),
    K_FROM_CS_ID INTEGER,
    CS_PERCENT DECIMAL(15,2),
    AAREP_MEMO VARCHAR(50),
    ID_LIST INTEGER,
    KOL_REST INTEGER)
AS
declare variable lm_id decimal(18,0);
declare variable lm_no integer;
declare variable deliq_k_id decimal(18,0);
declare variable kk integer;
begin
  deliq_k_id = 0;
  for select k_id, cast(1 as integer), k_kontract_no,
             case
               when (k_kolich_client_approved > 2 or pr_product_kind = 2) then 0
               else 1
             end as pr_loan_type,
             k_circle, k_circle_internal, k_from_kf_id, id_list, null k_deliq_days
      from kontstate, kontract, product, knaktlist, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between cast(ks_date_begin as date) and ks_date_end and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            k_id = ks_from_k_id and
            k_from_pr_id = pr_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            kal_new_reg_id = id_list

      union

      select dk_id k_id, cast(0 as integer), k_kontract_no,
             case
               when (k_kolich_client_approved > 2 or pr_product_kind = 2) then 0
               else 1
             end as pr_loan_type,
             k_circle, k_circle_internal, k_from_kf_id, dk_region_id id_list, dk_days k_deliq_days
      from aarep_kontract_deliq_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '000'), kontract, product
      where dk_id = k_id and
            k_from_pr_id = pr_id
      order by 1, 2

      into :k_id, :kind, :k_kontract_no, :pr_loan_type, :k_circle, :k_circle_internal, :k_from_kf_id, :id_list, :k_deliq_days

  do
  begin
    if (:k_deliq_days is null) then
      k_deliq_days = 0;
    aarep_memo = null;
    kol_rest = 0;
    if (deliq_k_id <> k_id) then
    begin
      if (:kind = 0) then
        deliq_k_id = k_id;
--        Вычисляет кол-во реструктуризации
      kol_rest = 0;
      select count(ol_id)
      from operlist, operrest
      where or_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            or_from_k_id = :k_id and
            ol_date <= :rep_date_end
      into :kol_rest;
      if (kol_rest is null) then
        kol_rest = 0;
      aarep_memo = null;
      if (kol_rest > 0) then
        aarep_memo = 'Реструктурированный кредит';

      /*Присваивает статусы классификации для групповых кредитов*/
      if (:pr_loan_type = 0) then
      begin
        k_from_cs_id = case
                         when (:k_deliq_days = 0 and (kol_rest = 0)) then 3
                         when ((:k_deliq_days between 1 and 29) or (kol_rest = 1 and :k_deliq_days = 0)) then 4
                         when ((:k_deliq_days between 30 and 59) and (kol_rest < 2)) then 6
                         when ((:k_deliq_days >= 60) or (kol_rest >= 2)) then 7
                       end;
      end
      /*Присваивает статусы классификации для индивидуальных кредитов*/
      if (:pr_loan_type = 1) then
      begin
--1. по своим <30 Дней, в других ФКУ галочка =<30, у родственнков есть галочка (=<30) - то "Активы под наблюдением 10%"
--2. по своим <30 Дней, в других ФКУ галочка >30, у родственнков есть галочка (>30) - то "Субстандартный 25%"
--  Определяет максимальное кол-во просроченных дней для классификации РППУ по индив c другой ФКУ
        kk = null;
        with tabl1
        as (select
                   case kt_status_pogash
                     when 'Платежи по графику' then 2
                     when 'до 30 дней просрочки' then 3
                     when '30 дней до 59 дней просрочки' then 4
                     when '60 дней до 89 дней просрочки' then 5
                     when '90 дней  до 119 дней просрочки' then 6
                     when '120 дней просрочки и более' then 7
                     else 1
                   end as kk
            from knclient, kibtaken
            where kc_amount_approved > 0 and
                  kc_from_k_id = :k_id and
                  kc_from_c_id = kt_from_c_id and
                  kt_year = extract(year from :rep_date_end) and
                  kt_month = extract(month from :rep_date_end))
        select max(kk)
        from tabl1
        into kk;

        k_from_cs_id = case
                         when (:k_deliq_days <= 30 and (kk is null) and (kol_rest = 0)) then 1 -- Стандартный 0%
                         when (:k_deliq_days <= 30 and (kk <= 3) and (kol_rest = 0)) then 4 -- Активы под наблюдением 10%
                         when (:k_deliq_days <= 30 and (kk > 3)) then 5 -- Субстандарт 25%
                         when ((:k_deliq_days between 31 and 90) or (kol_rest > 0 and :k_deliq_days <= 30)) then 5 -- Субстандарт 25%
                         when :k_deliq_days between 91 and 180 then 6 -- Сомнительный 50%
                         when :k_deliq_days > 180 then 7 -- Потери 100%
                       end;
      end

      select
             case :pr_loan_type /*Вытаскивает из таблицы classific процент начисление РППУ*/
               when 0 then cs_is_for_group
               when 1 then cs_is_for_indiv
             end
      from classific
      where cs_id = :k_from_cs_id
      into :cs_percent;

      select lm_id, lm_no
      from llc_main /*Вытаскивает из таблицы llc_main ID РППУ*/
      where lm_date_status = :rep_date_end and
            lm_from_reg_id = :id_list
      into :lm_id, :lm_no;

      /*Обновляет по вышеуказанным параметрам таблицу llc_kont */

      update llc_kont
      set lk_status = :k_from_cs_id,
          lk_percent = :cs_percent,
          lk_percent_orig = :cs_percent,
          lk_is_indiv = :pr_loan_type
      where lk_from_lm_id = :lm_id and
            lk_from_k_id = :k_id and
            lk_is_changed_mannual <> 1;/*Не включает вручную редактированные */

      suspend;
    end
  end
end
