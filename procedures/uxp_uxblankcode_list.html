CREATE PROCEDURE UXP$UXBLANKCODE_LIST
RETURNS (
    UBC_ID D_INTEGER,
    UBC_CODE D_STR_CODE,
    UBC_MEMO D_STR_255,
    UBC_SAVE_KIND D_SMALLINT,
    UBC_SINGLE_SELECT D_BOOL,
    UBC_INACTIVE D_BOOL,
    RES_USED_ACTIONS D_INTEGER,
    RES_BLANK_COUNT D_INTEGER)
AS
begin
  for select ubc_id,
             ubc_code,
             ubc_memo,
             ubc_save_kind,
             ubc_single_select,
             ubc_inactive
      from ux$blankcode
      order by ubc_code
      into :ubc_id,
           :ubc_code,
           :ubc_memo,
           :ubc_save_kind,
           :ubc_single_select,
           :ubc_inactive
  do
  begin
    res_used_actions = 0;
    res_blank_count = 0;

    select count(*)
    from ux$action
    where ua_preblank_kind = 1 and
          ua_preblank_code = :ubc_code
    into res_used_actions;

    select count(*)
    from ux$blanklink
    where ubl_from_ubc_id = :ubc_id
    into res_blank_count;

    if (res_used_actions = 0) then
      res_used_actions = null;

    if (res_blank_count = 0) then
      res_blank_count = null;

    suspend;
  end
end
