CREATE PROCEDURE HP_CONSULT_PAYMENT_INFO(
    K_ID D_ID)
RETURNS (
    NEXT_SH_DATE D_DATE,
    NEXT_SH_AMOUNT D_CURRENCY,
    DELINQ_AMOUNT D_CURRENCY,
    RES_TEXT D_STR_255)
AS
begin
  -- возвращает дату/сумму ближайщего планового платежа + сумму просрочки по предыдущим платежам
  -- платежи по конс. услугам в рассрочку (RFCKG-4900)

  next_sh_amount = 0;
  delinq_amount = 0;
  res_text = '';

  if (not exists(select *
                 from kontract
                 where k_id = :k_id and
                       k_cons_kind > 0)) then
  begin
    suspend;
    exit;
  end

  select oc_schedule_date,
         oc_schedule_principal + oc_paid_principal
  from opercons
  where oc_from_k_id = :k_id and
        oc_ol_is_active = 1 and
        oc_schedule_date >= current_date and
        (oc_schedule_principal + oc_paid_principal) > 0
  order by oc_schedule_date
  rows 1
  into next_sh_date, -- дата планового не закрытого платежа -- 12.04.2026
       next_sh_amount; -- сумма (остаток долга) планового платежа

  if (next_sh_amount is null) then
    next_sh_amount = 0;

  select sum(oc_schedule_principal + oc_paid_principal)
  from opercons
  where oc_from_k_id = :k_id and
        oc_ol_is_active = 1 and
        oc_schedule_date < current_date
  into delinq_amount; -- общая сумма просрочки

  if (delinq_amount is null) then
    delinq_amount = 0;

  -- компоновка res_text
  if (delinq_amount > 0 and
      next_sh_amount > 0) then
  begin
    res_text = 'Просрочка: ' || round(delinq_amount, 2) || ', платеж по графику: ' || f_datetostr(next_sh_date) || ' - ' || round(next_sh_amount, 2) || ' ';
    res_text = res_text || ascii_char(10) || 'Итого рекомендуемая сумма оплаты: ' || round(delinq_amount + next_sh_amount, 2) || ' ';
  end
  else if (delinq_amount > 0) then
    res_text = 'Сумма просрочки: ' || round(delinq_amount, 2) || ' ';
  else if (next_sh_amount > 0) then
    res_text = 'Платеж по графику: ' || f_datetostr(next_sh_date) || ' - ' || round(next_sh_amount, 2) || ' ';

  if (res_text is null) then
    res_text = '';

  suspend;
end
