CREATE PROCEDURE HP_PAYMENT_MONITORING_ONE(
    KC_ID D_ID)
RETURNS (
    K_KONTRACT_NO D_STR_100,
    K_DISB_DATE D_DATE,
    K_PERIOD D_INTEGER,
    K_AMOUNT_APPROVED D_INTEGER,
    BALANCE_OS DECIMAL(18,4),
    K_EXPIRE_DATE D_DATE,
    C_NAME_FIO D_STR_100,
    PR_NAME D_STR_100,
    FIO_KC D_STR_100,
    C_STATUS_BEGIN D_INTEGER,
    C_STATUS_AFTER D_INTEGER,
    REG_NAME D_STR_100)
AS
declare variable k_fact_disb_lo_id d_id;
declare variable c_id d_id;
declare variable k_id d_id;
begin
  select kt.k_kontract_no, kt.k_disb_date, kt.k_period, kt.k_amount_approved, kt.k_expire_date, cl.c_name_fio,
             pr.pr_name, kt.k_id, kt.k_fact_disb_lo_id ,rg.reg_name , kc.kc_from_c_id
      from kontract kt
      inner join kontstate ks on ks.ks_from_k_id = kt.k_id
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      inner join knclient kc on kc.kc_from_k_id = kt.k_id
      inner join clients cl on cl.c_id = kc.kc_from_c_id
      inner join region rg on rg.reg_unicode = ks.ks_region_id
      where kc.kc_id = :kc_id and
            current_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            kt.k_amount_approved > 0
      order by kt.k_disb_date desc
      rows 1
      into :k_kontract_no, :k_disb_date, :k_period, :k_amount_approved, :k_expire_date, :c_name_fio, :pr_name, :k_id, :k_fact_disb_lo_id ,:reg_name ,:c_id;


    select cs.c_name_fio
    from clients cs
    where cs.c_id = :k_fact_disb_lo_id
    into :fio_kc;

    select c_last_status, c_curr_status
    from hp_get_client_status(:c_id, current_date)
    into :c_status_begin, :c_status_after;


    select sum(zad_os)
    from aarep_make_zad_list(:k_id, current_date, 0)
    where sh_date is not null
    into :balance_os;

    suspend;

end
