CREATE PROCEDURE HP_HR_ORDER_TMP(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    HO_ID D_ID,
    REG_ID D_INTEGER,
    STAFF_LIST D_STR_4096)
AS
declare variable staff_exists d_integer;
declare variable min_reg_id d_integer;
declare variable max_reg_id d_integer;
begin
  for select ho_id,
             min(hrv_from_reg_id), max(hrv_from_reg_id),
             left(list(c_name_fio), 4000),
             max(iif(hrv_from_c_id = coalesce(:rep_c_id, hrv_from_c_id), 1, 0))
  from hr_orders, hr_voctravel, clients, region_list(:rep_region_id, 1, 3, 1)
  where ho_date between :rep_date_begin and :rep_date_end and
        hrv_from_ho_id = ho_id and
        hrv_from_reg_id = id_list and
        hrv_from_c_id = c_id
  group by 1
  into ho_id,
       min_reg_id, max_reg_id,
       staff_list, staff_exists
  do
  begin
    if (staff_exists = 0) then
      Continue;

    if (min_reg_id = max_reg_id) then
      reg_id = min_reg_id;
    else
      reg_id = null;

    suspend;
  end

  for select ho_id,
             min(ps_from_reg_id), max(ps_from_reg_id),
             left(list(c_name_fio), 4000),
             max(iif(ps_from_c_id = coalesce(:rep_c_id, ps_from_c_id), 1, 0))
  from hr_orders, hr_pstatus, clients, region_list(:rep_region_id, 1, 3, 1)
  where ho_date between :rep_date_begin and :rep_date_end and
        ps_from_ho_id = ho_id and
        ps_from_reg_id = id_list and
        ps_from_c_id = c_id
  group by 1
  into ho_id,
       min_reg_id, max_reg_id,
       staff_list, staff_exists
  do
  begin
    if (staff_exists = 0) then
      Continue;

    if (min_reg_id = max_reg_id) then
      reg_id = min_reg_id;
    else
      reg_id = null;

    suspend;
  end
end
