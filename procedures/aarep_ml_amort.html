CREATE PROCEDURE AAREP_ML_AMORT(
    BILL_ID DECIMAL(18,0),
    BILL_LINE_NO INTEGER)
RETURNS (
    AMM_DATE DATE,
    AMM_DOC_NO INTEGER,
    AMM_OPER_TYPE_NAME VARCHAR(50),
    AMM_LIFETIME_MONTH INTEGER,
    AMM_SUMMA_OS DECIMAL(18,4),
    AMM_SUMMA_AMORT DECIMAL(18,4),
    AMM_YEAR INTEGER,
    AMM_MONTH INTEGER,
    AMM_MEMO VARCHAR(1024))
AS
declare variable amm_oper_type integer;
declare variable ol_id decimal(18,0);
begin
  for select ol_id, ol_date, ol_doc_no, oal_oper_type, oal_lifetime_month, oal_summa_os, oal_summa_amort, oal_year,
             oal_month, ol_memo
      from operlist, operamorlist
      where oal_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            oal_bill_id = :bill_id and
            oal_bill_line_no = :bill_line_no and
            oal_is_delete = 0
      order by ol_date, oal_year, oal_month, ol_id
      into :ol_id, :amm_date, :amm_doc_no, :amm_oper_type, :amm_lifetime_month, :amm_summa_os, :amm_summa_amort,
           :amm_year, :amm_month, :amm_memo
  do
  begin
    amm_oper_type_name = '';
    if (amm_oper_type = 0) then
      amm_oper_type_name = 'Первично';
    if (amm_oper_type = 1) then
      amm_oper_type_name = 'Начисление';
    if (amm_oper_type = 2) then
      amm_oper_type_name = 'Переоценка';
    if (amm_oper_type = 3) then
      amm_oper_type_name = 'Списание';
    if (amm_oper_type = 4) then
      amm_oper_type_name = 'Модернизация';
    suspend;
  end

end
