CREATE PROCEDURE BLANK_MEM_ORDER_DETAIL(
    OL DECIMAL(18,0),
    POCKET_ID INTEGER,
    REG_ID INTEGER,
    MEM_DATE DATE)
RETURNS (
    MEM_ID DECIMAL(18,0),
    MEM_NUMBER INTEGER,
    MEM_CURRENCY VARCHAR(3),
    MEM_DT VARCHAR(20),
    MEM_KT VARCHAR(20),
    MEM_SUMMA DECIMAL(18,4),
    MEM_CLIENT_DT VARCHAR(255),
    MEM_CLIENT_KT VARCHAR(255),
    MEM_DETAIL VARCHAR(1024))
AS
declare variable mem_client_dt_id decimal(18,0);
declare variable mem_client_kt_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */
  if (ol <> 0) then
  begin
    for select ob_from_ol_id, ob_number, ob_currency, ob_account_dt, ob_account_kt, ob_summa_val, ob_detail,
               dt.c_name_fio, kt.c_name_fio
        from operbook
        left join clients dt on (ob_client_dt = dt.c_id)
        left join clients kt on (ob_client_kt = kt.c_id)
        where ob_from_ol_id = :ol
        order by ob_number
        into :mem_id, :mem_number, :mem_currency, :mem_dt, :mem_kt, :mem_summa, :mem_detail, :mem_client_dt,
             :mem_client_kt
    do
    begin
      if (mem_client_dt is null) then
        mem_client_dt = '';
      if (mem_client_kt is null) then
        mem_client_kt = '';
      suspend;
    end
  end
  else
  begin
    for select distinct ob_from_ol_id, ob_number, ob_currency, ob_account_dt, ob_account_kt, ob_summa_val, ob_detail,
                        ob_client_dt, ob_client_kt
        from operlist, operbook
        where ob_from_ol_id = ol_id and
              ol_pocket = :pocket_id and
              is_delete = 0 and
              ol_from_reg_id = :reg_id and
              ol_date_oper_day = :mem_date
        order by ob_from_ol_id, ob_number
        into :mem_id, :mem_number, :mem_currency, :mem_dt, :mem_kt, :mem_summa, :mem_detail, :mem_client_dt_id,
             :mem_client_kt_id
    do
    begin
      if (not(exists(select *
                     from acc_list, acctypelist
                     where ((al_id = :mem_dt) or
                           (al_id = :mem_kt)) and
                           al_from_bg_id like atl_acc_mask || '%' and
                           atl_from_at_id = 11 /* Касса */ ))) then
      begin
        mem_client_dt = '';
        mem_client_kt = '';
        if (mem_client_dt_id is not null) then
        begin
          select c_name_fio
          from clients
          where c_id = :mem_client_dt_id
          into :mem_client_dt;
        end
        if (mem_client_kt_id is not null) then
        begin
          select c_name_fio
          from clients
          where c_id = :mem_client_kt_id
          into :mem_client_kt;
        end
        if (mem_client_dt is null) then
          mem_client_dt = '';
        if (mem_client_kt is null) then
          mem_client_kt = '';
        suspend;
      end
    end
  end
end
