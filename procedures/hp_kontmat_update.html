CREATE PROCEDURE HP_KONTMAT_UPDATE(
    OBJ_KIND INTEGER,
    OBJ_ID DECIMAL(18,0),
    TOTAL_AMOUNT DECIMAL(18,4),
    ML_BLOB BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable fact_amount decimal(18,4) = 0;
declare variable fact_discount decimal(18,4) = 0;
declare variable json_key varchar(255);
declare variable json_value varchar(255);
declare variable delta decimal(18,4) = 0;
declare variable round_kind integer;
declare variable article varchar(255);
declare variable good_name varchar(255);
declare variable qty decimal(18,4) = 0;
declare variable price_single decimal(18,4) = 0;
declare variable ml_id decimal(18,0);
declare variable real_count decimal(18,4) = 0;
declare variable real_cost decimal(18,4) = 0;
declare variable kml_id integer;
begin
-- Процедура вставляет/обновляет список товаров заявки, контракта
  res_text = '';

  if (obj_kind = 0 and
      obj_id = 0) then
  begin
    res_text = 'Объект не указан';
    suspend;
    exit;
  end
  if (coalesce(ml_blob, '') = '') then -- Оформляем скидку
  begin
    res_text = 'Список товаров не указан';
    suspend;
    exit;
  end

--  ml_blob - JSON-массив в формате:
--  goods имя массива
--    article  Артикул товара  Строка(255)
--    name  Наименование товара  Строка(1000)
--    qty  Число (штук) данного товара  Integer
--    price_single  Цена за единицу товара  Decimal(18,2)

  round_kind = 0; -- Округлять стоиомость и сумму до единицы

  fact_amount = 0.0;
  fact_discount = 0.0;
  delta = 0.0;
  article = '';
  good_name = '';
  qty = 0;
  price_single = 0;

  for select json_key, json_value
      from fm_json_parse(:ml_blob, 0, 'goods')
      into :json_key, :json_value
  do
  begin
    if (json_key = 'endofobject') then
    begin
      fact_amount = fact_amount + round(qty * price_single, round_kind);
      article = '';
      good_name = '';
      qty = 0;
      price_single = 0;
    end
    if (json_key = 'article') then
      article = json_value;
    if (json_key = 'name') then
      good_name = json_value;
    if (json_key = 'qty') then
      qty = json_value;
    if (json_key = 'price_single') then
      price_single = json_value;
  end
  fact_discount = total_amount - fact_amount;
  if (fact_amount = 0) then
  begin
    res_text = 'Общая стоимость товаров нулевая';
    suspend;
    exit; -- Общая сумма товара не может быть нулевой
  end

-- Временно отмечаем как неактуальные
  update kontmat
  set kml_status = -1
  where kml_object_kind = :obj_kind and
        kml_object_id = :obj_id;

  delta = total_amount;
  for select json_key, json_value
      from fm_json_parse(:ml_blob, 0, 'goods')
      into :json_key, :json_value
  do
  begin
    if (json_key = 'endofobject') then
    begin
-- получение ML_ID
      select ml_id
      from hp_matlist_insert(:good_name, :article)
      into :ml_id;

      if (qty <> 0) then
      begin
        real_count = qty;
        real_cost = price_single;
        if (fact_discount <> 0.0 and
            fact_amount <> 0.0) then
        begin
          real_cost = round(((cast(price_single as double precision) * qty * fact_discount) / fact_amount) / qty, round_kind);
          real_cost = real_cost + price_single;
        end
        delta = delta - round(real_cost * real_count, round_kind);

        update or insert into kontmat (kml_object_kind, kml_object_id, kml_from_ml_id, kml_count, kml_cost, kml_status)
        values (:obj_kind, :obj_id, :ml_id, :real_count, :real_cost, 1)
        matching (kml_object_kind, kml_object_id, kml_from_ml_id, kml_cost);
      end
      article = '';
      good_name = '';
      qty = 0;
      price_single = 0;
    end
    if (json_key = 'article') then
      article = json_value;
    if (json_key = 'name') then
      good_name = json_value;
    if (json_key = 'qty') then
      qty = json_value;
    if (json_key = 'price_single') then
      price_single = json_value;
  end
  if (delta <> 0.0) then
  begin
    kml_id = null;
    select kml_id, kml_count, kml_cost
    from kontmat
    where kml_object_kind = :obj_kind and
          kml_object_id = :obj_id and
          kml_status = 1
    order by kml_id desc
    rows 1
    into :kml_id, real_count, real_cost;
    if (kml_id is not null) then
    begin
      if (real_count <> 1.0 and
          real_count <> 0.0) then
        delta = round(delta / real_count, 2);

      update kontmat
      set kml_cost = kml_cost + :delta
      where kml_id = :kml_id;
    end
  end

-- Удаляем из списка если не оказалось в свежем списке
  delete from kontmat
  where kml_object_kind = :obj_kind and
        kml_object_id = :obj_id and
        kml_status = -1;
  suspend;
end
