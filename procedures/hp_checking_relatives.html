CREATE PROCEDURE HP_CHECKING_RELATIVES(
    KF_ID D_ID,
    KF_OFFICER D_ID)
RETURNS (
    IS_RELATIVES D_SMALLINT)
AS
declare variable c_id d_id;
declare variable other_c_id d_str_2048;
begin
  --процедура была создана согласно задачи в jire rfckg-410
  --находим id клиента по kf_id. for для того случия если будет групповой кредит
  is_relatives = 0;
  for select kc.kc_from_c_id
      from kontract k
      inner join knclient kc on kc.kc_from_k_id = k.k_id
      inner join kontstate ks on ks.ks_from_k_id = k.k_id
      where k.k_from_kf_id = :kf_id and
            kc.kc_amount_approved > 0 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            current_date between ks.ks_date_begin and ks.ks_date_end
      into :c_id
  do
  begin
    --находим родственников клиента
    select list(rel_other_c_id, '.')
    from hp_client_relation_tmp(:c_id, 4)
    into :other_c_id;

    --смотрим если среди родственников спфник кому передается папка
    if (:other_c_id containing :kf_officer) then
    begin
      is_relatives = 1;
      suspend;
      exit;
    end
    else
      :is_relatives = 0;
  end
  suspend;
end
