CREATE PROCEDURE HP_CARD_DISB_RESULT_SET(
    OL_ID ID_TYPE,
    ERROR_TEXT STR_255,
    RESPONSE_DATA BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
RETURNS (
    RESULT_TEXT STR_255)
AS
declare variable ol_is_delete smallint;
declare variable ecd_status smallint;
begin
  error_text = trim(coalesce(error_text, ''));
  result_text = 'Saved';

  if (ol_id is null) then
  begin
    result_text = 'Ошибка! OL_ID не передан';
    suspend;
    exit;
  end

  ol_is_delete = null;

  select ol.is_delete, ecd.ecd_status
  from extsys_card_disb ecd
  inner join operlist ol on ol.ol_id = ecd.ecd_from_ol_id
  where ecd.ecd_from_ol_id = :ol_id
  order by ecd.ecd_id
  rows 1
  into :ol_is_delete, :ecd_status;

  ol_is_delete = coalesce(ol_is_delete, -777);

  if (ol_is_delete = 0) then
  begin
    result_text = 'Ошибка! Ранее выдача кредита уже была осуществлена. Отмена операции';
    suspend;
    exit;
  end

  if (ol_is_delete = -777) then
  begin
    result_text = 'Ошибка! Переданная операция не связана с выдачей кредита на карту';
    suspend;
    exit;
  end

  if (ecd_status <> 0) then
  begin
    result_text = 'Ошибка! Операция перевода денежных стредств уже обработана. Отмена операции';
    suspend;
    exit;
  end

  update operlist ol
  set ol.is_delete = 0
  where ol.ol_id = :ol_id;

  update operlist ol
  set ol.is_delete = 0
  where ol.ol_parent_id = :ol_id;

  update extsys_card_disb ecd
  set ecd.ecd_status = iif(:error_text = '', 1, -2),
      ecd.ecd_err_text = :error_text,
      ecd.ecd_response = :response_data
  where ecd.ecd_from_ol_id = :ol_id;

  suspend;
end
