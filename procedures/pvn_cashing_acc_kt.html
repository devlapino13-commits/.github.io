CREATE PROCEDURE PVN_CASHING_ACC_KT(
    OFFICE_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    ACC_KT_SHORT D_STR_10,
    ACC_KT D_STR_20)
AS
declare variable C_ID D_ID;
declare variable NUM_CASH D_INTEGER;
begin
  if (office_id is null) then
    exception e_message 'Офис не передан';

  if (un_id is null) then
    exception e_message 'Пользователь не передан';

  select un.un_from_c_id
  from username un
  where un.un_id = :un_id
  into :c_id;

  if (c_id is null) then
    exception e_message 'Сотрудник не определен';

  select uc_num_cash
  from user_cash uc
  where current_timestamp between uc_date_beg and coalesce(uc_date_end, '31.12.9999')
  and uc.uc_from_reg_id = :office_id
  and uc.uc_from_c_id = :c_id
  into :num_cash;

  if (num_cash is null) then
    exception e_message 'Номер кассы не определен. Пользователь не привязан к кассе';

  acc_kt_short = '10001/'||:num_cash;
  acc_kt = (select al_id from acc_list where al_from_reg_id = :office_id and al_currency = 417 and al_account = :acc_kt_short); -- Касса

  suspend;
end
