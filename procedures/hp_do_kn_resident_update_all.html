CREATE PROCEDURE HP_DO_KN_RESIDENT_UPDATE_ALL(
    REP_DATE D_DATE)
AS
declare variable K_ID D_ID;
declare variable KC_ID D_ID;
declare variable KNR_DATE_BEGIN D_DATE;
declare variable KNR_NON_RESIDENT D_BOOL;
declare variable C_ID D_ID;
declare variable C_NON_RESIDENT D_BOOL;
begin
  if (rep_date = f_date_month_end(rep_date)) then
  begin
    for
    select ks_from_k_id, kc_id, kc_from_c_id from kontstate, knclient
    where ks_ol_is_delete = 0
    and ks_state = 1
    and :rep_date between ks_date_begin and ks_date_end
    and ks_from_k_id = kc_from_k_id
    and kc_amount_approved > 0
    into k_id, kc_id, c_id
    do
    begin
       -- получаем последний статус резидентства на дату
      knr_date_begin = null;
      knr_non_resident = null;
      select knr_date_begin, knr_non_resident
      from kn_resident
      where knr_from_kc_id = :kc_id and
            knr_date_begin <= :rep_date
      order by knr_from_kc_id desc, knr_date_begin desc
      rows 1
      into knr_date_begin, knr_non_resident;

      -- получаем актуальный статус резидентства клиента из документов на дату
      c_non_resident = null;
      select res_non_resident
      from hp_get_client_type(:c_id, 13, 39, :rep_date)
      into c_non_resident;

      -- проводки по переносу сальдо из-за смены резидентства
      if (knr_non_resident <> c_non_resident) then
        execute procedure hp_do_kn_resident_update(:k_id, :rep_date, 13, 39, 1000000000001);
    end
  end
end
