CREATE PROCEDURE HP_EDOC_BLANK_DO_CHECK(
    EDB_ID D_ID,
    NEW_STATUS D_SMALLINT,
    UN_ID D_ID,
    RD_ID D_INTEGER = null)
AS
declare variable staff_id d_id;
declare variable push_body d_str_long;
declare variable rd_name d_str_255;
declare variable c_id d_id;
begin
  update edoc_blank_scan
  set edbs_status = :new_status,
      edbs_from_rd_id = :rd_id,
      edbs_status_un_id = :un_id
  where edbs_from_edb_id = :edb_id and
        ((edbs_from_bls_id is not null and
        edbs_status = 1) or
        (edbs_kind = 2 and
        edbs_status = 0));

  if ((f_get_db_id() = 81001) and (new_status = 2))  then
  begin
    select eds.eds_from_c_id
    from edoc_blank edb, edoc_list edl, edoc_sign eds
    where edl.edl_id = edb.edb_from_edl_id and
          edl.edl_id = eds.eds_from_edl_id and
          eds.eds_status = 1 and
          edb.edb_id = :edb_id
          rows 1
    into :c_id;

    update clientdocs cd
    set cds_verify_status = 2
    where (cds_from_c_id = :c_id and
          cd.cds_from_pk_id = 355 and
          cd.cds_is_delete = 0 and
          cd.cds_verify_status in  (-3, 1))
          order by cd.cds_create_datetime desc
          rows 1;
  end

  if (new_status = -1 and
      rd_id is not null) then
  begin
    select rd_name
    from refdata
    where rd_id = :rd_id
    into rd_name;

    select un_from_c_id,
           f_format('%1, %2, Причина: %3', edf_print_name, f_objname(edl_object_kind, edl_object_id), :rd_name)
    from edoc_list, edoc_blank, username, edoc_frx
    where edb_id = :edb_id and
          edb_from_edl_id = edl_id and
          edl_create_un_id = un_id and
          edb_from_edf_id = edf_id
    into staff_id, push_body;

    if (staff_id is not null) then
      execute procedure hp_send_push_to_staff(:staff_id, 'Доработка скана эл.документа',
          push_body, 1000000000000, 'edoc_blank_scan_abort', 7);
  end
end
