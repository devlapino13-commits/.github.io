CREATE PROCEDURE HP_ACTIONS_SHORT_DETAIL(
    UN_ID D_ID,
    ACL_ID D_ID,
    ACL_DATE D_DATE = null)
RETURNS (
    RES_ACL_ID D_ID,
    RES_QUESTION_ID D_INTEGER,
    RES_RD_ID D_INTEGER,
    RES_STATE D_INTEGER,
    RES_STR_VALUE D_STR_255,
    RD_EDITOR_KIND D_INTEGER,
    RD_CODE D_STR_100,
    RD_NAME D_STR_255,
    RD_DESCRIPTION D_BLOB_TEXT,
    RD_MRU_LIST D_STR_8100,
    RD_ORDERNO FLOAT)
AS
declare variable c_id d_id;
declare variable acl_object_kind d_integer;
declare variable acl_object_id d_id;
declare variable anketa_name d_str_100;
begin
-- Процедура возвращает форму анкеты по ID задачи
-- если ID задачи не указан, возвращает первый из очереди
-- Если очередь пуста, ничего не возвращается

-- Уточняем детали задачи
  if (coalesce(acl_id, 0) = 0) then
  begin
    select
      un_from_c_id
    from username
    where un_id = :un_id
    into :c_id;

    select
      aco_id,
      aco_object_kind,
      aco_object_id,
      rd_code
    from refdata
      inner join actionorder
        on aco_from_rd_id = rd_id
          and aco_respons_c_id = :c_id
          and aco_object_kind = 0
    where rd_main_code = 'ActionGroup'
      and rd_str_id = 'Anketa'
      and rd_code <> rd_main_code
    order by aco_id
    rows 1
   into
     :acl_id,
     :acl_object_kind,
     :acl_object_id,
     :anketa_name;
  end
  else
  begin
    select
      acl_id,
      acl_object_kind,
      acl_object_id,
      rd_code
    from actionlist
      jnner join refdata
        on rd_id = acl_from_rd_id
    where acl_id = :acl_id
    into
      :acl_id,
      :acl_object_kind,
      :acl_object_id,
      :anketa_name;
  end
  if (acl_id is null) then
    acl_id = 0;
  res_acl_id = acl_id;

  if (coalesce(res_acl_id, 0) = 0) then
    exit;

  for
    select
      res_question_id,
      res_rd_id,
      res_state,
      res_str_value,
      rd_editor_kind,
      rd_orderno,
      rd_code,
      rd_name,
      rd_description,
      rd_mru_list
    from hp_actions_short_detail_tmp(
      :acl_object_kind,
      :acl_object_id,
      :anketa_name,
      :acl_date)
    order by rd_orderno
  into
    :res_question_id,
    :res_rd_id,
    :res_state,
    :res_str_value,
    :rd_editor_kind,
    :rd_orderno,
    :rd_code,
    :rd_name,
    :rd_description,
    :rd_mru_list
  do
  begin
    suspend;
  end
end
