CREATE PROCEDURE MC_KONTRACT_DETAIL(
    K_ID D_ID)
RETURNS (
    K_FACT_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_FACT_CLOSED_DATE D_DATE,
    EDL_ID D_ID,
    K_STATUS D_SMALLINT)
AS
declare variable sum_zadol d_currency;
declare variable ks_state d_smallint;
begin
  if (coalesce(k_id, 0) <= 0) then
    exception e_default 'parametr "k_id" has invalid value';

  select ks.ks_state
  from kontstate ks
  where ks.ks_from_k_id = :k_id and
        current_date between ks.ks_date_begin and ks.ks_date_end and
        ks.ks_ol_is_delete = 0
  into :ks_state;

  k_status = 1;

  if (ks_state = 2) then
  begin
    select sum(oko_os + oko_perc + oko_other + oko_sh_os + oko_sh_perc + oko_sh_other)
    from operkroff
    where oko_from_k_id = :k_id and
          oko_ol_is_active = 1
    into sum_zadol;
    sum_zadol = coalesce(sum_zadol, 0);

    if (sum_zadol = 0) then
      k_status = 2;
  end

  if (k_status = 1) then
    exception e_message 'информация доступна только по закрытым контрактам ';

  select k.k_fact_disb_date, k.k_amount_approved, k.k_fact_closed_date
  from kontract k
  where k.k_id = :k_id
  into :k_fact_disb_date, k_amount_approved, k_fact_closed_date;

  select edl_id
  from edoc_list edl, edoc_blank edb
  where edb.edb_from_edl_id = edl.edl_id and
        edl.edl_object_kind = f_objkind('kontract', 'id') and
        edl.edl_object_id = :k_id and
        edl.edl_from_edc_id = f_edoc_kind('loan.debt', 'id') and
        edl.edl_status = 0 rows 1
  into :edl_id;

  suspend;
end
