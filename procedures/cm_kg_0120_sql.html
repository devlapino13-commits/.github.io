CREATE PROCEDURE CM_KG_0120_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable c_id id_type;
declare variable pr_id d_integer;
declare variable pr_name d_str_100;
declare variable prl_amount_cl_min d_currency;
declare variable prl_amount_cl_max d_currency;
begin
  if (ma_id = 3451932) then exit; --Чолпон 11.04.23
  res_error_is_critical = 1;
  select ma_date_oper, pr_id, pr_name, min(pl.prl_amount_cl_min), max(pl.prl_amount_cl_max)
  from miniapp ma
  inner join product pr on ma.ma_from_pr_id = pr_id
  inner join prodlogic pl on pr_id = pl.prl_from_pr_id and
        ma.ma_date_oper between pl.prl_date_begin and pl.prl_date_end and
        ma.ma_kolich_repay between pl.prl_repay_min and pl.prl_repay_max
  where ma_id = :ma_id and
        ma_is_active = 1 and
        ma_date_end = '31.12.9999'
  group by 1, 2, 3
  into kf_disb_date, pr_id, pr_name, prl_amount_cl_min, prl_amount_cl_max;
  for select mcl.mac_from_c_id, c_name_fio, mac_amount
      from miniappclient mcl
      inner join clients on mac_from_c_id = c_id
      where mac_from_ma_id = :ma_id and
            mac_amount > 0
      into c_id, res_info_text, kc_amount
  do
  begin
    if (kc_amount < prl_amount_cl_min) then
    begin
      res_error_text = pr_name || ' - Сумма клиента не может быть ниже: ' || round(prl_amount_cl_min, 2);
      suspend;
    end
    if (kc_amount > prl_amount_cl_max) then
    begin
      res_error_text = pr_name || ' - Сумма клиента не может быть больше: ' || round(prl_amount_cl_max, 2);
      suspend;
    end
  end
end
