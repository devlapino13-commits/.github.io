CREATE PROCEDURE HP_GET_RF_CREDIT_HISTORY(
    OBJECT_KIND D_STR_20,
    OBJECT_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable sql_text d_str_255;
declare variable res_text_tmp d_str_2048;
declare variable c_birth_date d_date;
declare variable c_name_fio d_str_100;
declare variable c_inn d_str_20;
declare variable c_pass_no varchar(30);
declare variable sql_text_tmp d_str_255;
declare variable crel_other_c_id d_id;
declare variable rel_name d_str_20;
declare variable res_html_family d_blob_text;
declare variable counter_tmp d_integer;
declare variable sql_request d_str_long;
declare variable res_text d_str_8100;
declare variable client_type d_str_20;
declare variable c_name_fio_fam d_str_100;
declare variable res_text_tmp_rf d_str_8100;
declare variable ma_id d_id;
declare variable c_id_list d_str_long;
begin
  res_text = '';
  sql_text_tmp = '';
  res_html = '';
  c_id_list = '';
  sql_text = 'select cl.c_birth_date, cl.c_name_fio, cl.c_inn, cl.c_pass_no from clients cl where cl.c_id = ';

  res_html = (select res_style
              from fm_html_styles) || '<body>';
  res_html = BLOB_APPEND(res_html, '<h2>Кредитная история в Мбулак РФ:</h2>');

  if (f_objkind(:object_kind, 'name') = 'kontract') then
  begin
    select kt.k_from_ma_app_id
    from kontract kt
    where kt.k_id = :object_id
    into ma_id;
  end
  else
    ma_id = :object_id;

  for select mac_from_c_id, iif(mac_amount > 0, 1, 0) as is_client
      from miniappclient mac
      where mac_from_ma_id = :ma_id -- условие amount_approved > 0 не добавлено так как проверяем поручителей тоже
      as cursor mac
  do
  begin
    if (c_id_list containing '.' || mac.mac_from_c_id || '.') then
        continue;

    sql_text_tmp = :sql_text || mac.mac_from_c_id;
    res_text_tmp = '';
    c_birth_date = null;
    c_name_fio = null;
    c_inn = null;
    c_pass_no = null;
    client_type = coalesce(iif(mac.is_client = 1, trim('Клиент'), trim('Поручитель')), '');


    execute statement sql_text_tmp
        into c_birth_date, c_name_fio, c_inn, c_pass_no;

    res_text_tmp = '{' || '"json":' || '{' || '"FIO": ' || '"' || coalesce(:c_name_fio, '') || '"' || ',' || '"BD": ' || '"' || f_datetostr(:c_birth_date) || '"' || ',' || '"INN": ' || '"' || coalesce(:c_inn, '') || '"' || ',' || '"PASS_NO": ' || '"' || coalesce(:c_pass_no, '') || '"' || '}' || '}';

    sql_request = 'select res_text from HP_RF_CREDIT_HISTORY_INFO(' || '''' || res_text_tmp || ''')';
    begin
      execute statement :sql_request
      with autonomous transaction
      on external('10.1.2.11/3050:e:\ib_base\db_mbfru.ib')
      as user 'SYSDBA' password 'F!ora80X'
      into :res_text;
      when any do
      begin
        res_html = 'База РФ временно недоступна, повторите запрос немного позже.';
        suspend;
        exit;
      end
    end
    res_html = BLOB_APPEND(res_html, '<b>' || client_type || ': ' || :c_name_fio || '</b><br><br>');
    if (trim(coalesce(:res_text, '')) <> '') then
      res_html = BLOB_APPEND(res_html, 'Долг в РФ ' || trim(:res_text) || ' рублей<br><br>'); -- RFCKG-5062
    else
      res_html = BLOB_APPEND(res_html, 'Кредитная информация в РФ отсутствует<br><br>');

    res_html_family = '';
    for select crel.crel_other_c_id, rel.rel_name
        from clientrelation crel
        inner join relationtype rel on rel.rel_id = crel.crel_from_rel_id
        where crel.crel_from_c_id = mac.mac_from_c_id and
              crel.crel_is_active = 1 and
              crel.crel_from_rel_id in (1, 2, 3, 4, 5, 6) --Супруг, Супруга, Отец, Мать, Сын, Дочь
        into crel_other_c_id, rel_name
    do
    begin
      c_birth_date = null;
      c_name_fio_fam = null;
      c_inn = null;
      c_pass_no = null;
      sql_text_tmp = :sql_text || crel_other_c_id;
      c_id_list = c_id_list || '.' || crel_other_c_id || '.';

      execute statement sql_text_tmp
          into c_birth_date, c_name_fio_fam, c_inn, c_pass_no;

      res_text_tmp = '{' || '"json":' || '{' || '"FIO": ' || '"' || coalesce(:c_name_fio_fam, '') || '"' || ',' || '"BD": ' || '"' || f_datetostr(:c_birth_date) || '"' || ',' || '"INN": ' || '"' || coalesce(:c_inn, '') || '"' || ',' || '"PASS_NO": ' || '"' || coalesce(:c_pass_no, '') || '"' || '}' || '}';

      sql_request = 'select res_text from HP_RF_CREDIT_HISTORY_INFO(' || '''' || res_text_tmp || ''')';
      begin
        execute statement :sql_request
        with autonomous transaction
        on external('10.1.2.11/3050:e:\ib_base\db_mbfru.ib')
        as user 'SYSDBA' password 'F!ora80X'
        into :res_text;
        when any do
        begin
          res_html = 'База РФ временно недоступна, повторите запрос немного позже.';
          suspend;
          exit;
        end 
      end
      if (trim(coalesce(:res_text, '')) <> '') then
        res_html_family = BLOB_APPEND(res_html_family, 'Долг родственника в РФ ' || trim(:res_text) || ' рублей<br><br>'); -- RFCKG-5062
    end
    res_html = BLOB_APPEND(res_html, res_html_family, '</body>');
  end
  suspend;
end
