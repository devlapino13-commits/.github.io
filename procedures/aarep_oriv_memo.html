CREATE PROCEDURE AAREP_ORIV_MEMO(
    REP_DATE D_DATE)
RETURNS (
    K_KONTRACT_NO STR_20,
    SBL_NAME STR_255,
    SBA_DATE_BEGIN D_DATE,
    SBA_DATE_END D_DATE,
    C_NAME_FIO STR_100,
    ORIV_MEMO_DATETIME STR_100,
    ORIV_MEMO D_STR_LONG,
    ORIV_COMMENT_DATETIME STR_100,
    ORIV_COMMENT D_STR_LONG)
AS
declare variable k_id id_type;
declare variable c_id id_type;
declare variable db_id d_integer;
begin

  db_id = f_get_db_id();

  for select k_id, sba_date_begin, sba_date_end, sbl_name, k_kontract_no, c_name_fio, c_id
      from sb_akt
      inner join sb_list on sba_from_sbl_id = sbl_id
      inner join knclient on kc_from_k_id = sba_from_k_id
      inner join kontract on k_id = sba_from_k_id
      inner join clients on c_id = kc_from_c_id
      inner join oksummary on oks_from_k_id = k_id
      where sba_status = 2 and
            :rep_date between sba_date_begin and sba_date_end and
            iif (:db_id = 81001, sba_is_delete = 0 and sbl_name containing 'KG', sba_is_delete = 0) and
            kc_amount_approved > 0 and
            :rep_date >= oks_date_begin and
            :rep_date < oks_date_end
      into k_id, sba_date_begin, sba_date_end, sbl_name, k_kontract_no, c_name_fio, c_id
  do
  begin
    oriv_memo = '';
    oriv_memo_datetime = '';
    for select obm_memo, f_datetimetostr(obm_datetime)
        from hp_objectprops_memo_get('clients', :c_id, 'Refdata', 'ObjectProps.oriv_memo', 1)
        into oriv_memo, oriv_memo_datetime
    do
    begin
      oriv_comment = '';
      oriv_comment_datetime = '';
      for select obm_memo, f_datetimetostr(obm_datetime)
          from hp_objectprops_memo_get('clients', :c_id, 'Refdata', 'ObjectProps.oriv_comment', 1)
          into oriv_comment, oriv_comment_datetime
      do
      begin
        suspend;
      end
    end
    if (oriv_memo = '' or oriv_comment = '') then
      suspend;
  end
end
