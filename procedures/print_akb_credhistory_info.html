CREATE PROCEDURE PRINT_AKB_CREDHISTORY_INFO(
    KBM_ID D_ID)
RETURNS (
    ACCOUNTNO D_STR_20,
    OVERDUEDAYS D_INTEGER,
    HISTORYYEAR D_STR_10,
    HISTORYMONTH D_STR_10,
    CREDITSTATUS D_STR_10)
AS
declare variable CHECKED_DATE D_DATETIME;
declare variable JSON_TEXT D_BLOB_BINARY;
declare variable LIABILITIESCOUNT D_BLOB_TEXT;
declare variable LIABILITY D_BLOB_TEXT;
declare variable I D_INTEGER;
declare variable HISTORY D_BLOB_TEXT;
declare variable COUNTHISTORYITEM D_INTEGER;
declare variable HISTORYITEM D_BLOB_TEXT;
declare variable REPORTINGPERIOD D_STR_10;
declare variable I_CH D_INTEGER;
begin
  json_text = '';
  select bls.bls_blob
  from extinfolist eil
  inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
  inner join kibminiapp kbm on eil.eil_object_id = kbm.kbm_id
  inner join blobstore bls on eil.eil_from_bls_id = bls.bls_id
  where eil.eil_status = 2 and
        kbm.kbm_ver_info = 3 and
        kbm.kbm_id = :kbm_id and
        eic.eic_code = 'akb'
  order by kbm.kbm_checked_datetime desc
  rows 1
  into :json_text;

  if (json_text <> '' or json_text <> '{}' or json_text is not null) then
  begin
  liabilitiescount = f_json_blob_get_count(:json_text, 'liabilities.liability') - 1;

      if (f_json_blob_get_value(:json_text, 'liabilities.liability', '') <> '') then
      begin
        i = 0;
        while (i <= liabilitiescount) do
        begin
          liability = f_json_blob_get_value(:json_text, 'liabilities.liability[' || :i || ']', '');
          accountno = f_json_blob_get_value(:liability, 'accountNo', '');

          history = f_json_blob_get_value(:liability, 'history', '');
        if (history <> '') then
        begin
          counthistoryitem = f_json_blob_get_count(:history, 'historyItem') - 1;

          if (counthistoryitem < 0) then
          begin
            historyitem = f_json_blob_get_value(:history, 'historyItem', '');
            overduedays = f_json_blob_get_value(:historyitem, 'overdueDays', '');
            overduedays = cast(overduedays as d_integer);
            reportingperiod = f_json_blob_get_value(:historyitem, 'reportingPeriod', '');
            if (reportingperiod <> '') then
            begin
              historymonth = left(:reportingperiod, 2);
              historyyear = right(:reportingperiod, 4);
            end
            creditstatus = f_json_blob_get_value(:historyitem, 'creditStatus', '');
            suspend;
            exit;
          end 

          i_ch = 0;
          while (i_ch <= counthistoryitem and counthistoryitem >= 0) do
          begin
            historyitem = f_json_blob_get_value(:history, 'historyItem[' || :i_ch || ']', '');
            overduedays = f_json_blob_get_value(:historyitem, 'overdueDays', '');
            overduedays = cast(overduedays as d_integer);
            reportingperiod = f_json_blob_get_value(:historyitem, 'reportingPeriod', '');
            if (reportingperiod <> '') then
            begin
              historymonth = left(:reportingperiod, 2);
              historyyear = right(:reportingperiod, 4);
            end
            creditstatus = f_json_blob_get_value(:historyitem, 'creditStatus', '');
            i_ch = i_ch + 1;
            suspend;
          end
        end
        i = i + 1;
        end
      end
  end
end
