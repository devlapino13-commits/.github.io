CREATE PROCEDURE AAREP_GET_RESTRUCTURING(
    REP_K_ID ID_TYPE,
    REP_DATE_END DATE,
    REP_DB_ID INTEGER)
RETURNS (
    IS_RESTRUCT SMALLINT,
    RESTRUCT_MEMO VARCHAR(255),
    RESTRUCT_STR VARCHAR(255),
    RESTRUCT_DATE DATE,
    RESTRUCT_KOL INTEGER,
    RESTRUCT_DATE_LAST DATE)
AS
declare variable ol_from_rd_id integer;
declare variable kol_rows integer;
declare variable ol_date date;
declare variable ol_user_id id_type;
declare variable rd_name varchar(255);
declare variable rd_memo varchar(255);
declare variable is_first smallint;
declare variable max_date_graph date;
declare variable max_date_rest date;
declare variable or_from_ol_id id_type;
declare variable max_day_in_last_payment d_integer;
declare variable sh_date_old d_date;
declare variable sh_date_new d_date;
declare variable sh_kol_old d_integer;
declare variable sh_kol_new d_integer;
begin
  ol_from_rd_id = 0;
  is_restruct = 0;
  restruct_date = null;
  restruct_kol = 0;
  is_first = 0;
  rd_name = '';
  rd_memo = '';
  restruct_memo = null;
  restruct_str = null;
  if (rep_db_id is null) then
    rep_db_id = f_get_db_id();

  max_day_in_last_payment = 0;

  if (rep_db_id = 81001 and
      rep_date_end >= cast('30.04.2024' as date)) then
    max_day_in_last_payment = 0; -- по задаче RFCRU-1542 , ДОбавил ubakachiev
  else


  -- с 31.03.2020 в РФ увеличение срока не более на 31 дней не считаем реструктуризацией, Sanjari, 13.05.2020
  if (rep_db_id = 81001 and
      rep_date_end >= cast('01.03.2020' as date)) then
    max_day_in_last_payment = 31;

  for select coalesce(or_from_rd_id, 1), or_ol_date, or_from_ol_id, ol_user_id
      from operrest
      inner join operlist on ol_id = or_from_ol_id
      where or_from_k_id = :rep_k_id and
            or_ol_is_active = 1 and
            :rep_date_end between or_ol_date and or_date_end
      order by ol_date
      into :ol_from_rd_id, :ol_date, :or_from_ol_id, :ol_user_id
  do
  begin
    is_restruct = 0;

    if (rep_date_end < cast('01.12.2016' as date) or rep_db_id not in (39801, 41701)) then
    begin
      is_restruct = 1;
      if (rep_db_id = 81001) then
      begin
        select max(iif(ok_type = 1, ok_sh_date, cast('01.01.1900' as date))),
               max(iif(ok_type = 2 and ok_from_ol_id = :or_from_ol_id and ok_os + ok_sh_os > 0 and ok_perc + ok_sh_perc > 0, ok_sh_date, cast('01.01.1900' as date)))
        from operkred
        where ok_ol_is_active = 1 and
              ok_ol_date <= :rep_date_end and
              ok_from_k_id = :rep_k_id and
              ok_type in (1, 2) and
              ok_from_c_id = 0
        into max_date_graph, max_date_rest;

        if (datediff(day, max_date_graph, max_date_rest) <= max_day_in_last_payment) then
          is_restruct = 0;

        -- проверка, имеется ли отсрочка
        if (is_restruct = 0) then
        begin
          -- график до реструктуризации
          sh_date_old = null;
          with tabl1
          as (select ok_sh_date, sum(ok_os + ok_sh_os) os, sum(ok_perc + ok_sh_perc) perc
              from operkred
              where ok_ol_is_active = 1 and
                    ok_from_k_id = :rep_k_id and
                    ok_type < 7 and
                    ok_ol_date <= :ol_date and
                    ok_from_ol_id <> :or_from_ol_id
              group by 1)
          select min(ok_sh_date)
          from tabl1
          where os > 0 or
                perc > 0
          into sh_date_old;
          -- график после реструктуризации
          sh_date_new = null;
          with tabl1
          as (select ok_sh_date, sum(ok_os + ok_sh_os) os, sum(ok_perc + ok_sh_perc) perc
              from operkred
              where ok_ol_is_active = 1 and
                    ok_from_k_id = :rep_k_id and
                    ok_type < 7 and
                    ok_ol_date <= :ol_date
              group by 1)
          select min(ok_sh_date)
          from tabl1
          where os > 0 or
                perc > 0
          into sh_date_new;

          if ((sh_date_new > sh_date_old) and
              (sh_kol_new = sh_kol_old)) then
            is_restruct = 1;
          --2.  Предоставление отсрочки платежа по займу;

          if ((sh_date_new > sh_date_old) and
              (max_date_graph = max_date_rest)) then
            is_restruct = 1;
        end
      end
    end
    else
    begin
      rd_name = '';
      rd_memo = '';
      if (ol_from_rd_id > 0) then
      begin
        select rd_name, rd_memo
        from refdata
        where rd_id = :ol_from_rd_id
        into rd_name, rd_memo;

        if (rd_memo = 'Реструктуризация') then
          is_restruct = 1;
      end
      else
      begin
        if (ol_user_id = 1000000000000) then
          is_restruct = 1;
      end

      if (is_restruct = 1) then
      begin
        if (is_first = 0) then
        begin
          is_first = 1;
          restruct_memo = rd_name;
          restruct_str = rd_memo;
          restruct_date = ol_date;
        end
        restruct_date_last = ol_date;
      end
    end
    restruct_kol = restruct_kol + is_restruct;

    if (is_restruct = 1) then
    begin
      if (restruct_date is null) then
        restruct_date = ol_date;
      restruct_date_last = ol_date;
    end
  end
  -- нижнее условие сделано для того чтобы если было несколько реструктуризации и из них последний не реструктуризация
  if (restruct_kol > 0) then
    is_restruct = 1;
  else
  if (rd_name <> '' or rd_memo <> '') then
  begin
    restruct_memo = rd_name;
    restruct_str = rd_memo;
  end

  suspend;
end
