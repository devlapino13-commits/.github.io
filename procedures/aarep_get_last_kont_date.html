CREATE PROCEDURE AAREP_GET_LAST_KONT_DATE(
    K_ID DECIMAL(18,0))
RETURNS (
    IS_CRITICAL INTEGER,
    RES_TEXT VARCHAR(255))
AS
declare variable acl_date date;
declare variable rd_id integer;
declare variable ma_id integer;
declare variable ma_date_oper date;
begin
  -- Проведура проверяет дату выдачи кредита после утверждения КК.
  -- Если прошло более 15 дней, нельзя выдавать - решение КК утрачивает силу
  is_critical = 0;
  res_text = '';

  /*select rd_id
  from refdata
  where rd_code = 'ActionGroup.Check_Loan_Data' and
        rd_is_active = 1
  into :rd_id;*/

  select rd_id
  from hp_get_check_loan_data(:k_id)
  into :rd_id;

  if (rd_id is null) then
  begin
    suspend;
    exit;
  end

  select cast(acl_last_datetime as date)
  from actionlist
  where acl_object_kind = 9 and
        acl_object_id = :k_id and
        acl_from_rd_id = :rd_id and
        acl_status = 2
  into :acl_date;
  if (acl_date is null) then
  begin
    suspend;
    exit;
  end

  select k_from_ma_app_id
  from kontract
  where k_id = :k_id
  into :ma_id;
  if (ma_id is null) then
  begin
    suspend;
    exit;
  end

  select ma_date_oper
  from miniapp
  where ma_id = :ma_id and
        ma_date_app <= :acl_date and
        ma_date_end >= :acl_date and
        ma_is_active = 1
  into :ma_date_oper;
  if (ma_date_oper is null) then
  begin
    suspend;
    exit;
  end

  ma_date_oper = maxvalue(ma_date_oper, acl_date);
  ma_date_oper = dateadd(day, 15, ma_date_oper);

  if (exists(select *
             from kontract
             where k_disb_date > :ma_date_oper and
                   k_id = :k_id)) then
  begin
    is_critical = 2;
    res_text = 'После одобрения КК прошло более 15 дней. Необходимо отправить повторно на КК.';
    suspend;
  end
end
