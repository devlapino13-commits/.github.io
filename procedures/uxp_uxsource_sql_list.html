CREATE PROCEDURE UXP$UXSOURCE_SQL_LIST
RETURNS (
    US_UUID D_UUID,
    US_CODE D_STR_CODE,
    US_LOOKUP_TEXT D_STR_255,
    US_NAME D_STR_255,
    US_MEMO D_STR_8100,
    RES_LOOKUP_SQL D_STR_LONG,
    RES_KEYNAME_SQL D_STR_LONG,
    RES_IS_ERROR D_BOOL,
    RES_ERROR_TEXT D_STR_255)
AS
begin
  for select us_uuid, us_code, us_name, us_memo, us_lookup_text
  from ux$source
  where us_lookup_id <> '' and
        us_lookup_text <> ''
  order by us_code
  into us_uuid, us_code, us_name, us_memo, us_lookup_text
  do
  begin
    res_lookup_sql = '';
    res_keyname_sql = '';
    res_error_text = '';
    res_is_error = 0;

    select lookup_sql, keyname_sql, error_text
    from uxp$uxsource_sql_prepare(:us_uuid, '')
    into :res_lookup_sql, :res_keyname_sql, :res_error_text;

    if (res_error_text <> '') then
      res_is_error = 1;

    suspend;
  end
end
