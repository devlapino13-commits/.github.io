    exit;
  for select kt.k_last_reg_id, pr.pr_code, kt.k_amount_approved, kc.kc_from_c_id, kt.k_disb_date
      from kontract kt
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      inner join knclient kc on kc.kc_from_k_id = kt.k_id
      where kt.k_id = :k_id and
