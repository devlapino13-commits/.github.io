CREATE PROCEDURE HP_PUSH_NOTIFY_VIEW(
    DT_BEG D_DATE = current_date,
    DT_END D_DATE = current_date)
RETURNS (
    PNO_ID D_INTEGER,
    PNO_CREATION_DATE D_DATE,
    PNO_PLANNED_DATE D_DATETIME,
    PLANNED_DATE D_DATE,
    PLANNED_TIME D_TIME,
    PNO_APPROVAL_DATE D_DATETIME,
    PNO_EXECUTION_DATE D_DATETIME,
    PNO_TITLE D_STR_255,
    PNO_CONTENT D_STR_LONG,
    PNO_TYPE_RECEIVER D_SMALLINT,
    PNO_DATE_BEGIN D_DATE,
    PNO_DATE_END D_DATE,
    TYPE_RECEIVER D_STR_CODE,
    PNO_REG_CLIENT D_INTEGER,
    REG_NAME D_STR_CODE,
    PNO_PRJ_USER D_SMALLINT,
    PRJ_USER D_STR_20,
    PNO_USER_ADV D_BOOL,
    PNO_NUMB_RECIPIENT D_INTEGER,
    PNO_STATUS D_SMALLINT,
    STATUS D_STR_CODE,
    PNO_AUTHOR_ID D_ID,
    AUTHOR_NAME D_STR_100,
    PNO_CONTROLLER_ID D_ID,
    CONTROLLER_NAME D_STR_100)
AS
begin
  for
    select
      pno_id,
      pno_creation_date,
      pno_planned_date,
      cast(pno_planned_date as d_date),
      cast(pno_planned_date as d_time),
      pno_approval_date,
      pno_execution_date,
      pno_title,
      pno_content,
      pno_type_receiver,
      case pno_type_receiver
        when 0 then 'активные клиенты'
        when 1 then 'активные пользователи КМП'
        when 2 then 'закрытые клиенты'
      end, 
      pno_date_begin,
      pno_date_end,
      pno_reg_client,
      pno_prj_user,
      case pno_prj_user
        when 0 then 'все'
        when 1 then 'КМП'
        when 2 then 'ОЙВО'
      end,
      pno_user_adv,
      pno_numb_recipient,
      pno_status,
      case pno_status
        when -1 then 'отменено'
        when  0 then 'ожидает утверждения'
        when  1 then 'утверждено'
        when  2 then 'исполнено'
      end,
      pno_author_id,
      u1.un_fio, 
      pno_controller_id,
      u2.un_fio
    from push_notify
      inner join username as u1
        on u1.un_id = pno_author_id
      left join username as u2
        on u2.un_id = pno_controller_id
    where pno_creation_date between :dt_beg and :dt_end
  into
    :pno_id,
    :pno_creation_date,
    :pno_planned_date,
    :planned_date,
    :planned_time,
    :pno_approval_date,
    :pno_execution_date,
    :pno_title,
    :pno_content,
    :pno_type_receiver,
    :type_receiver,
    :pno_date_begin,
    :pno_date_end,
    :pno_reg_client,
    :pno_prj_user,
    :prj_user,
    :pno_user_adv,
    :pno_numb_recipient,
    :pno_status,
    :status,
    :pno_author_id,
    :author_name,
    :pno_controller_id,
    :controller_name
  do
  begin
    if (pno_reg_client > 0) then
      reg_name = (select reg_name from region where reg_unicode = :pno_reg_client);
    else
      reg_name = null;
    suspend;
  end
end
