CREATE PROCEDURE HP_DO_REPAYMENT_RECALC_NEW_FIX_2(
    K_ID D_ID,
    RECALC_DATE D_DATE,
    NEED_EDOC D_BOOL)
AS
declare variable OL_ID D_ID;
declare variable REG_ID D_INTEGER;
declare variable OL_MEMO D_STR_LONG;
declare variable UN_ID D_ID;
declare variable KD_PERCENT_EFFECT D_PERCENT;
declare variable SH_ID D_INTEGER;
declare variable SH_DATE D_DATE;
declare variable SH_OS D_CURRENCY;
declare variable SH_PERC D_CURRENCY;
declare variable SH_LAST_SH_ID D_INTEGER;
declare variable SH_LAST_SH_DATE D_DATE;
declare variable SH_LAST_OS_AMOUNT D_CURRENCY;
declare variable NEW_OS D_CURRENCY;
declare variable OK_SH_DATE D_DATE;
declare variable OK_SH_ID D_INTEGER;
declare variable SH_LAST_PERC_AMOUNT D_CURRENCY;
declare variable SH_DAYS D_INTEGER;
declare variable NEW_PERC D_CURRENCY;
declare variable PERCENT_PER_DAY D_PERCENT;
declare variable SUM_ANU D_CURRENCY;
declare variable BAL D_CURRENCY;
declare variable KOL_DAY D_INTEGER;
declare variable SH_LAST_SH_DATE_2 D_DATE;
begin
  -- Процедура перестраивает график создав новую операцию
  -- Является аналогом принятия нулевого платежа

  un_id = 1000000000001;

  select kal_new_reg_id, k_kontract_no || ' от ' || f_datetostr(k_disb_date) || 'г. '
  from kontract, knaktlist kal
  where k_id = :k_id and
        kal.kal_from_kf_id = k_from_kf_id and
        kal.kal_date <= :recalc_date and
        kal.kal_date_end >= :recalc_date and
        kal.is_delete = 0
  into reg_id, ol_memo;

  if (reg_id is null) then
    exception e_default 'reg_id not found';

  ol_memo = 'Корректировка графика погашений по договору ' || ol_memo;

  insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind,
                        ol_memo, ol_parent_id, is_delete, ol_user_id)
  values (:reg_id, 1, 4, :recalc_date, 9, :ol_memo, null, 0, :un_id)
  returning ol_id
  into :ol_id;

  -- находим последний платеж и сумму аннуитета, сумма аннуитета sh_os + sh_perc после выхода из цикла
  for
  select ok_sh_date, ok_sh_id,
         sum(ok_os + ok_sh_os) os,
         sum(ok_perc + ok_sh_perc) perc
  from operkred
  where ok_from_k_id = :k_id
  and ok_ol_is_active = 1
  and ok_type < 7
  group by 1,2
  having sum(ok_os + ok_sh_os) <> 0
  order by ok_sh_date desc
  rows 2
  into ok_sh_date, ok_sh_id, sh_os, sh_perc
  do
  begin
    if (sh_last_sh_id is null) then
    begin
      sh_last_sh_id = ok_sh_id;
      sh_last_sh_date = ok_sh_date;
      sh_last_os_amount = sh_os;
      sh_last_perc_amount = sh_perc;
    end
  end

--  if (row_count < 2) then
--    exception e_message 'Остался только 1 платеж';

  sum_anu = sh_os + sh_perc;
  bal = sh_last_os_amount;
  kol_day = null;

  new_os = sum_anu - sh_last_perc_amount - bal;

  insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_pay_type, 
                        ok_sh_os, ok_sh_perc)
  values (:ol_id, -1, :k_id, :sh_last_sh_id, :sh_last_sh_date, 3, 3,
          :new_os, 0);

  sh_last_sh_date_2 = sh_last_sh_date;
  bal = bal - (sum_anu - sh_last_perc_amount);


  select kd_percent_per_day
  from kontdata
  where kd_ol_date <= :recalc_date and   
        kd_from_k_id = :k_id and
        kd_ol_is_delete = 0 and
        kd_ol_date <= kd_date_end
  order by kd_ol_date desc
  rows 1
  into percent_per_day;

  -- находим укороченные платежи по графику
  for
  select ok_sh_id, ok_sh_date
  from operkred
  where ok_from_k_id = :k_id
  and ok_ol_is_active = 1
  and ok_type < 7
  and ok_sh_date > :sh_last_sh_date
  group by 1,2
  into ok_sh_id, ok_sh_date
  do
  begin
    sh_days = ok_sh_date - sh_last_sh_date_2;
    sh_last_sh_date_2 = ok_sh_date;

    new_perc = round(cast(bal as double precision) * percent_per_day * sh_days / 100.00, 0);

    if (bal + new_perc < sum_anu) then sum_anu = bal + new_perc;

    new_os = sum_anu - new_perc;
    bal = bal - new_os;



    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_pay_type, 
                        ok_sh_os, ok_sh_perc)
    values (:ol_id, -1, :k_id, :ok_sh_id, :ok_sh_date, 3, 3,
            :new_os, :new_perc);

  end
--  execute procedure hp_do_repayment_recalc_sh(:ol_id, :k_id);
--  execute procedure hp_do_repayment_recalc_sh(:ol_id, :k_id);
 /*
  if (need_edoc = 1) then
  begin
    select kps_psk
    from hp_edoc_ins_kontpsk(:k_id, :ol_id, :un_id)
    into :kd_percent_effect;
  end
 */
end
