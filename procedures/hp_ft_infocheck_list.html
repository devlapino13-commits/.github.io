CREATE PROCEDURE HP_FT_INFOCHECK_LIST(
    FIL_ID D_ID)
RETURNS (
    FIC_ID D_ID,
    FIC_FROM_FIL_ID D_ID,
    FIC_DATE D_DATE,
    FIC_DATA_CHANGED D_SMALLINT,
    FIC_FOUND_IN_BL D_SMALLINT,
    FIC_RISK_CATEG D_INTEGER,
    FIC_SCHEDULE_DATE D_DATE,
    FIC_FROM_UN_ID D_ID,
    FIC_DATETIME D_DATETIME,
    FIB_ID D_ID,
    UN_NAME D_STR_100,
    CHECK_MEMO D_STR_LONG)
AS
declare variable last_date d_date;
declare variable last_date_exist d_smallint;
begin
  select coalesce(fil_date_close, current_date), fil_from_fib_id
  from ft_infolist
  where fil_id = :fil_id
  into last_date, fib_id;

  last_date_exist = 0;
  for select fic_id,
             fic_from_fil_id,
             fic_date,
             fic_data_changed,
             fic_found_in_bl,
             fic_risk_categ,
             fic_schedule_date,
             fic_from_un_id,
             fic_datetime,
             fil_from_fib_id
      from ft_infocheck, ft_infolist
      where fic_from_fil_id = :fil_id and
            fic_from_fil_id = fil_id
      order by fic_date, fic_id
      into :fic_id,
           :fic_from_fil_id,
           :fic_date,
           :fic_data_changed,
           :fic_found_in_bl,
           :fic_risk_categ,
           :fic_schedule_date,
           :fic_from_un_id,
           :fic_datetime,
           :fib_id
  do
  begin
    un_name = '';
    check_memo = '';

    select un_name
    from username
    where un_id = :fic_from_un_id
    into un_name;

    select fid_memo
    from ft_infodate
    where fid_from_fib_id = :fib_id and
          fid_check_date = :fic_date
    rows 1
    into check_memo;

    if (fic_date = last_date) then last_date_exist = 1;

    suspend;
  end

  if (last_date_exist = 0) then
  begin
    fic_id = 0;
    fic_date = last_date;
    fic_from_fil_id = fil_id;
    fic_datetime = current_timestamp;
    fic_found_in_bl = 0;
    fic_data_changed = 0;
    Suspend;
  end
end
