CREATE PROCEDURE HP_PDN_REF_BKI(
    UN_ID D_ID NOT NULL,
    IS_KI D_BOOL = 0)
RETURNS (
    RES_TEXT D_STR_2048,
    RES_IS_ERROR D_BOOL)
AS
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable pdn_status_id d_smallint;
declare variable ssp_is_send d_bool;
declare variable ki_is_send d_bool;
declare variable cnt_ssp d_integer;
declare variable cnt_ki d_integer;
declare variable cnt d_integer;
declare variable txt d_str_2048;
declare variable res_text_tmp d_str_255;
declare variable res_is_error_tmp d_bool;
begin
  -- отправка запросов ССП/КИ при пересчете ПДН после 60 дней после выдачи (RFCRU-4309)
  -- используется в форме "ПДН - расчет в связи с неисполнением цели по рефинансированию"

  res_is_error = 0;
  res_text = '';
  txt = '';
  cnt_ssp = 0;
  cnt_ki = 0;

  select count(*)
  from pdn_ref_recalc
  where prr_is_selected = 1
  into cnt;

  if (cnt > 100) then
    exception e_message 'Кол-во выбранных договоров - ' || cnt || ', за раз можно отправить запросы ССП/КИ максимум по 100 договорам';

  for select prr_k_id,
             prr_k_kontract_no,
             prr_pdn_status_id
      from pdn_ref_recalc
      where prr_is_selected = 1
      into k_id,
           k_kontract_no,
           pdn_status_id
  do
  begin
    res_text_tmp = '';
    res_is_error_tmp = 0;
    ssp_is_send = 0;
    ki_is_send = 0;

    if (pdn_status_id not in (4, 7)) then
    begin
      if (txt <> '') then
        txt = txt || ascii_char(10);
      txt = txt || k_kontract_no || ': не соответствует статус ПДН'; -- статус должен быть 4 - "Кандидат" или 7 - "Кандидат: КИ устарела";
      res_is_error_tmp = 1;
      continue;
    end

    select res_text,
           res_is_error,
           ssp_is_send,
           ki_is_send
    from hp_pdn_ref_bki_one(:un_id, :k_id, :is_ki)
    into res_text_tmp,
         res_is_error_tmp,
         ssp_is_send,
         ki_is_send;

    if (res_is_error_tmp = 1) then
    begin
      if (txt <> '') then
        txt = txt || ascii_char(10);
      txt = txt || trim(res_text_tmp);
    end

    if (res_is_error_tmp = 0) then
    begin
      cnt_ssp = cnt_ssp + ssp_is_send;
      cnt_ki = cnt_ki + ki_is_send;

      if (ssp_is_send = 1) then
        update pdn_ref_recalc
        set prr_ssp_note = 'В очереди для отправки',
            prr_ssp_note_id = -2
        where prr_k_id = :k_id;

      if (ki_is_send = 1) then
        update pdn_ref_recalc
        set prr_ki_note = 'В очереди для отправки',
            prr_ki_note_id = -2
        where prr_k_id = :k_id;
    end
  end

  res_text = 'Отправлено запросов ССП/КИ - ' || cnt_ssp || '/' || cnt_ki;

  if (txt <> '') then
    res_text = trim(res_text) || ascii_char(10) || 'Ошибки:' || ascii_char(10) || trim(txt);

  suspend;
end
