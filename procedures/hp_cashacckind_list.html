CREATE PROCEDURE HP_CASHACCKIND_LIST(
    CASH_OPER_KIND D_INTEGER,
    IS_IPF D_BOOL)
RETURNS (
    CAK_ID D_INTEGER,
    CAK_OPER_KIND D_INTEGER,
    CAK_IS_IPF D_BOOL,
    CAK_FROM_BG_ID D_STR_20,
    CAK_ACC_NO D_INTEGER,
    CAK_IS_PERSONAL D_BOOL)
AS
begin
  if (is_ipf is null) then
    is_ipf = 0;

  for select cak_id,
             cak_oper_kind,
             cak_is_ipf,
             cak_from_bg_id,
             cak_acc_no,
             cak_is_personal
      from cashacckind
      where cak_oper_kind = :cash_oper_kind and
            cak_is_ipf = :is_ipf
      into :cak_id,
           :cak_oper_kind,
           :cak_is_ipf,
           :cak_from_bg_id,
           :cak_acc_no,
           :cak_is_personal
  do
  begin
    suspend;
  end
end
