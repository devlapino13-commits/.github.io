CREATE PROCEDURE HR_PERSONAL_LAST_POSITION(
    PER_ID DECIMAL(18,0),
    PER_DATE DATE)
RETURNS (
    PS_TEXT VARCHAR(512),
    PS_DATE_BEGIN DATE,
    PS_DATE_END_IN_ORDER DATE,
    PS_CALC_BONUS INTEGER,
    PS_FROM_REG_ID INTEGER,
    PS_BRANCH_ID INTEGER,
    PS_JL_NAME VARCHAR(100),
    PS_REG_NAME VARCHAR(100),
    PS_FROM_JL_ID INTEGER,
    PS_IS_WORKING SMALLINT)
AS
declare variable ps_dep_name varchar(100);
declare variable ps_is_on_staff smallint;
declare variable ps_from_dep_id integer;
declare variable ps_period_month integer;
declare variable ps_from_ho_id decimal(18,0);
declare variable ps_id integer;
declare variable jk_system_code varchar(100);
begin
  ps_text = '';
  for select ps_id, ps_from_ho_id, ps_date_begin, ps_period_month, ps_date_end_in_order, ps_from_reg_id, ps_from_dep_id,
             ps_from_jl_id, ps_is_on_staff, ps_is_working, reg_name, dep_name, jl_name, reg_head_office,
             jk_system_code
      from hr_pstatus, region, depart, joblist, jobkind
      where ps_from_c_id = :per_id and
            ps_date_begin <= :per_date and
            ps_date_end >= :per_date and
            ps_from_reg_id = reg_unicode and
            ps_from_dep_id = dep_id and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            ps_status = 1
      order by ps_date_begin desc, ps_date_end_in_order desc, ps_is_working desc
      rows 1
      into :ps_id, :ps_from_ho_id, :ps_date_begin, :ps_period_month, :ps_date_end_in_order, :ps_from_reg_id,
           :ps_from_dep_id, :ps_from_jl_id, :ps_is_on_staff, :ps_is_working, :ps_reg_name, :ps_dep_name, :ps_jl_name,
           :ps_branch_id, :jk_system_code
  do
  begin
    if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
      ps_calc_bonus = 1;
    else
      ps_calc_bonus = 0;
    ps_text = ps_reg_name || ', ' || ps_jl_name;
    if (ps_is_working = 0) then
      ps_text = ps_text || ', Уволен';
    else
    begin
      if (ps_is_on_staff = 0) then
        ps_text = ps_text || ', Внештат';
      else
        ps_text = ps_text || ', Штатный';
    end
    if (ps_calc_bonus is null) then
      ps_calc_bonus = 0;
    suspend;
  end
end
