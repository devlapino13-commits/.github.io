    lastpos = 1;

                kncl.kc_from_c_id = :c_id and

                into :et_code, :et_create_datetime;

                do_send_sms = 1;

                if (et_create_datetime is not null) then -- если ранее уже передавалась команда на отправку СМС с ПИН-кодом
                begin
                  if (datediff(minute, et_create_datetime, cast('now' as timestamp)) < 10) then -- если с момента отправки СМС, прошло 10 минут, тогда СМС новый не отправляю, подставляю ПИН из предыдущего
                  begin
                    pincode = :et_code;
                    do_send_sms = 0;
                  end
                  else
                    do_send_sms = 1;
                end

                if (do_send_sms = 1) then
                begin
                  pincode = cast(round(trunc((rand() * 88888) + 10000)) as str_100);
                  execute procedure hp_send_quick_message('sms', :cell_no, '', :pincode, '', null, 1000000000000);
                end

                insert into extsys_token (et_from_esm_id, et_from_c_id, et_code)
                values (:esm_id, :c_id, :pincode);
              end
            end
          end
        end
      end
      result =
        '<?xml version="1.0" encoding="UTF-8"?>' || crlf ||
        '<response>' || crlf ||
        '     <result>' || err_no ||
        '</result>' || crlf ||
        '     <comment>' || err_text || '</comment>' || crlf ||
          iif(err_no = 0,
          '     <client_id>' || c_id || '</client_id>' || crlf ||
          '     <client_name>' || c_name_fio || '</client_name>' || crlf, '') ||
        '</response>';
    end
    else
    if (command = 'pin_check') then -- валидация кода из СМС (выдача на карту)
    begin
      err_no = 0;
      err_text = 'OK';
      pincode = trim(coalesce(pincode, ''));
      et_code = null;

      if (pincode = '') then
        err_text = 'Пин-код не указан';
      else
      if (c_id is null) then
        err_text = 'Идентификатор клиента не передан';
      else
      begin
        select et.et_code
        from extsys_token et
        where et.et_from_c_id = :c_id
        order by et.et_id desc
        rows 1
        into :et_code;

        et_code = coalesce(et_code, '');

        if (et_code = '') then
          err_text = 'Код не отправлялся. Верификация отменена.';
        else
        if (et_code <> pincode) then
          err_text = 'Код не корректный.';
      end

      if (err_text <> 'OK') then
        err_no = 8;

      result = '<?xml version="1.0" encoding="UTF-8"?>' || crlf || '<response>' || crlf || '     <result>' || err_no || '</result>' || crlf || '     <comment>' || err_text || '</comment>' || crlf || '</response>';
    end

    if (command = 'check') then -- всегда возращаю что ОК, т.е. данная операция нужна только для логирования
    begin
      result = '<?xml version="1.0" encoding="UTF-8"?>' || crlf || '<response>' || crlf || '     <result>' || err_no || '</result>' || crlf || '     <comment>' || err_text || '</comment>' || crlf || '</response>';
    end

    if (command = 'authorise') then -- igusev
    begin
      err_no = 0;
      err_text = 'OK';
      token = trim(coalesce(token, ''));
      cardnumber = coalesce((select field_value
                             from hp_extsys_parcer(:params, 'cardnumber')), '');

      if (c_id is null) then
        err_text = 'Идентификатор клиента не передан';
      else
      if (token = '') then
        err_text = 'Токен не передан';
      else
      begin
        select et.et_id, et.et_token_datetime
        from extsys_token et
        where et.et_from_c_id = :c_id
        order by et.et_id desc
        rows 1
        into :et_id, :et_token_datetime;

        if (et_id is null) then
          err_text = 'Код не отправлялся. Отмена привязки токена';
        else
        if (et_token_datetime is not null) then
          err_text = 'Токен уже был присвоен';
        else
        begin
          update extsys_token et
          set et.et_token = :token,
              et.et_cardnumber = :cardnumber,
              et.et_token_esm_id = :esm_id
          where et.et_id = :et_id;

          select edl_id
          from edoc_list
          where edl_object_id = :c_id and
                edl_object_kind = f_objkind('clients', 'id') and
                edl_status = 0 and
                coalesce(edl_date_end, current_date) >= current_date and
                edl_from_edc_id = f_edoc_kind('bankcard.moneta', 'id')
          order by edl_id desc
          rows 1
          into edl_id;

          if (edl_id is not null) then
          begin
            select eds_id
            from edoc_sign
            where eds_from_edl_id = :edl_id and
                  eds_status = 0
            order by eds_status_datetime desc
            rows 1
            into eds_id;
  
            if (eds_id is not null) then
            begin
              update edoc_msg
              set edm_status = 1
              where edm_from_eds_id = :eds_id and
                    edm_status = 0;

              update edoc_sign
              set eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'online', 'id')
              where eds_id = :eds_id;
            end
          end
        end
      end

      if (err_text <> 'OK') then
        err_no = 8;

      result = '<?xml version="1.0" encoding="UTF-8"?>' || crlf || '<response>' || crlf || '     <result>' || err_no || '</result>' || crlf || '     <comment>' || err_text || '</comment>' || crlf || '</response>';
    end

    insert into extsysanswer (esa_id, esa_from_esm_id, esa_message, esa_result_id, esa_status)
    values (-1, :esm_id, :result, :err_no, iif(:err_no = 0, 0, -1))
    returning esa_id
    into :esa_id;

    -- Заполняем оставшиеся поля
    update extsysmain
    set esm_account = :cl_birthdate || ':' || :cl_uniq_acc,
        esm_object_id = :c_id,
        esm_result_id = :err_no,
        esm_status = iif(:err_no = 0, 0, -1)
    where esm_id = :esm_id;

    suspend;
  end

  /* ------------------------------ WEBSITE ----------------------------------*/
  if (project = 'website') then
  begin
    err_no = 0;
    err_text = 'OK';
    for select esf_id, esf_field_code, esv_value_txt
        from extsysfields, extsysvalues
        where esv_from_esf_id = esf_id and
              esv_from_esm_id = :esm_id and
              esf_project = :project and
              esf_field_code in ('command', 'phone_no', 'sms_code', 'cl_birthday', 'cl_account', 'cl_next_call_value', 'uuid', 'k_id')
        order by esf_field_code asc
        into esf_id, field_code, field_value
    do
    begin
      if (field_code = 'command') then
        command = field_value;
      if (field_code = 'phone_no') then
        phone_no = field_value;
      if (field_code = 'sms_code') then
        sms_code = field_value;
      if (field_code = 'cl_account') then
        cl_uniq_acc = field_value;
      if (field_code = 'k_id') then
        k_id = field_value;
      if (field_code = 'uuid') then
      begin
        uuid = char_to_uuid(field_value);
        when any do
        begin
          err_no = 4;
          err_text = 'Неверный формат UUID';
        end
      end
      if (field_code = 'cl_birthday') then
      begin
        cl_birthdate = field_value;
        update extsysvalues
        set esv_value_date = :cl_birthdate
        where esv_from_esm_id = :esm_id and
              esv_from_esf_id = :esf_id;
        when sqlcode -413 do
        -- convert_error
        begin
          err_no = 4;
          err_text = 'Неправильный формат даты рождения клиента (ДД.ММ.ГГГГ)';
        end
      end
      if (field_code = 'cl_next_call_value') then
      begin
        if (field_value <> '') then
        begin
          /* Timestamp передается в формате YYYYMMDDHHNNSS */
          tmp_timestamp = substring(field_value from 7 for 2) || '.' || -- DD.
          substring(field_value from 5 for 2) || '.' || -- MM.
          substring(field_value from 1 for 4) || ' ' || -- YYYY
          substring(field_value from 9 for 2) || ':' || -- HH:
          substring(field_value from 11 for 2) || ':' || -- NN:
          substring(field_value from 13 for 2); -- SS
          -- Обновляем value_date в extvalues
          update extsysvalues esv
          set esv.esv_value_date = :tmp_timestamp,
              esv.esv_value_txt = :tmp_timestamp
          where esv_from_esm_id = :esm_id and
                esv_from_esf_id = :esf_id;

          when sqlcode -413 do
          -- convert_error
          begin
            err_no = 4;
            err_text = 'Неправильный формат даты следующего звонка (YYYYMMDDHHNNSS)';
          end
        end
      end
    end

    update extsysmain
    set esm_account = null,
        esm_action = :command,
        esm_ext_datetime = current_timestamp,
        esm_object_kind = 0, -- По аналогии когда не найден клиент
        esm_object_id = null,
        esm_result_id = 0,
        esm_status = 0
    where esm_id = :esm_id;

    if (command not in ('save_cl_info', 'get_cl_info', 'lp_sign')) then
    begin
      err_no = 3;
      err_text = 'Неверная команда: ' || command;
    end

    if (command = 'save_cl_info') then
    begin
      -- Корректность номера телефона
      if (err_no = 0 and
          phone_no not similar to '(79|992|996|998|994)[[:DIGIT:]]{9}') then
      begin
        err_no = 1;
        err_text = 'Неправильный номер телефона';
      end
      -- Проверка, что в contverify есть пара № телефона + код
      if (err_no = 0) then
      begin
        s = '';
        select iif(res_status = 0, 0, 5), res_text
        from hp_extsyspin_verify('ExtSysMain', 'cell', :phone_no, :sms_code)
        into :err_no, err_text;
      end

      -- Проверки пройдены, сохранение
      t = 0;
      if (err_no <> 0) then
      begin
        t = -1;
        update extsysmain
        set esm_result_id = :err_no,
            esm_status = -1
        where esm_id = :esm_id;
      end
      result =
        '<?xml version="1.0" encoding="UTF-8"?>' || crlf ||
        '<response>' || crlf ||
        '     <result>' || err_no || '</result>' || crlf ||
        '     <comment>' || err_text || '</comment>' || crlf ||
        '</response>';

      -- переношу данные в таблицу CC_WEBSITE_MINIAPP, для дальнейшей обработки КЦ ЗУрры
      for select error_text
          from hp_wma_basic_data_fill(:esm_id, :params)
          where error_text <> ''
          into :err_text
      do
      begin
        err_no = 5;
      end

      -- логика для ОКТЕЛЛ
      --execute procedure OKT_WEBSITE_FILL_ANKETA(:ESM_ID);
    end
    else
    if (command = 'get_cl_info') then
    begin
      -- Отключение возможности оплаты по Монете
--      tmp_timestamp = cast('now' as timestamp);
--      if (extract(hour from tmp_timestamp) = 23) then
--      begin
--        err_no = 8;
--        err_text = 'Сервис временно недоступен, с 23:00 по 24:00 по московскому времени';
--      end


      /* Проверка что uniq_id клиента передали */
      if (err_no = 0 and
          cl_uniq_acc is null) then
      begin
        err_no = 6;
        err_text = 'Лицевой счет не указан';
      end

      /* Проверка что дата рождения клиента указана */
      if (err_no = 0 and
          cl_birthdate is null) then
      begin
        err_no = 7;
        err_text = 'Дата рождения не указана';
      end

      /* Проверка существования плательщика */
      if (err_no = 0) then
      begin
        select c_id
        from hp_check_clientuniq_acc_no(:cl_birthdate, :cl_uniq_acc)
        into c_id;
        if (c_id = -1) then
        begin
          err_no = 8;
          err_text = 'Клиент не найден, обратитесь в офис компании';
        end
        else
        begin
          select cl.c_name_fio
          from clients cl
          where cl.c_id = :c_id
          into :c_name_fio;

          select short_fio
          from hp_fio_clear(:c_name_fio)
          into :c_name_fio;

          if (c_id is not null) then
          begin
            if (k_id is null) then
            begin
                -- получаю рекомендуемую сумму платежа
                select oks_from_k_id
                from oksummary
                inner join knclient on kc_from_k_id = oks_from_k_id and
                      kc_from_c_id = :c_id and
                      kc_amount_approved > 0
                where oks_date_end >= '31.12.2999' and
                oks_close_state = 0
                order by oks_delinq_date, oks_balance
                rows 1
                into :k_id;
            end

            if (k_id is not null) then
            begin
              cur_name = (select cm_currency_name
                          from company);
              sum_repay = (select recom_pay_sum
                           from get_recommended_pay_sum(:k_id, current_date));
              k_kontract_no = (select k_kontract_no from kontract where k_id = :k_id);
              description = 'Погашение займа по договору №'||:k_kontract_no||', '||c_name_fio||', без НДС';
              c_name_fio = c_name_fio || '; Рекомендуемая сумма платежа: ' || round(sum_repay, 2) || ' ' || cur_name;
            end
          end
          --          end
        end
      end
      if (err_no = 0) then
      begin
        -- Заполняем оставшиеся поля
        update extsysmain
        set esm_account = :cl_birthdate || ':' || :cl_uniq_acc,
            esm_object_id = :c_id,
            esm_result_id = :err_no,
            esm_status = iif(:err_no = 0, 0, -1)
        where esm_id = :esm_id;
      end
      description = coalesce(description,'');
      result = '<?xml version="1.0" encoding="UTF-8"?>' || crlf ||
      '<response>' || crlf ||
      '     <result>' || err_no || '</result>' || crlf ||
      '     <comment>' || err_text || '</comment>' || crlf ||
      iif(err_no = 0,
        '     <client_id>' || c_id || '</client_id>' || crlf ||
        '     <client_name>' || c_name_fio || '</client_name>' || crlf ||
        '     <description>' || description || '</description>' || crlf ||
        '     <mnt_id>' || f_get_const('moneta_acc') || '</mnt_id>' || crlf
        , '') ||
      '</response>';
    end
    else
    if (command = 'lp_sign') then
    begin
      err_no = 4;
      err_text = 'Неверный запрос.';

      result =
        '<?xml version="1.0" encoding="UTF-8"?>' || crlf ||
        '<response>' || crlf ||
        '     <result>' || err_no || '</result>' || crlf ||
        '     <comment>' || err_text || '</comment>' || crlf ||
        '</response>';
    end

    insert into extsysanswer (esa_id, esa_from_esm_id, esa_message, esa_result_id, esa_status)
    values (-1, :esm_id, :result, :err_no, iif(:err_no = 0, 0, -1))
    returning esa_id
    into :esa_id;

    suspend;
  end
