CREATE PROCEDURE FM_REPORT_MENU(
    REP_ID VARCHAR(255),
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    DATE_BEGIN DATE,
    DATE_END DATE,
    UN_ID DECIMAL(18,0))
RETURNS (
    M_CODE VARCHAR(100),
    M_KIND VARCHAR(100),
    M_CAPTION VARCHAR(100),
    M_PARAMS VARCHAR(1024))
AS
declare variable txt varchar(255); /* M_CODE */
declare variable db_id integer;
declare variable str_date_end varchar(10);
declare variable str_date_begin varchar(10);
declare variable c_id decimal(18,0);
declare variable def_txt varchar(255);
declare variable ma_id decimal(18,0);
declare variable k_id_tmp decimal(18,0);
declare variable ma_id_tmp decimal(18,0);
declare variable obj_id_tmp decimal(18,0);
declare variable obj_kind_tmp varchar(100);
declare variable has_access d_bool;
declare variable lang d_str_lang;
declare variable un_name d_str_code;
declare variable is_oriv d_smallint;
declare variable head d_str_100;
declare variable conf d_str_100;
declare variable note d_str_100;
declare variable is_manager d_bool;
begin
  -- Процедура возвращает список меню для окна в мобильном приложении.
  -- Если вызывается из основного окна клиента, Rep_ID = Client

  -- Возвращаемые результаты:
  -- M_CODE - варианты: Open, HTML, SQL
  --   Для M_CODE=Open,
  --     в M_KIND передаем кодовое название окна для перехода, например Client
  --     в M_PARAMS - Json, состав параметров зависит от окна. Н-р для access_kind требуется для окна клиента
  --   Для M_CODE=HTML,
  --     в M_KIND передаем Rep_ID из процедуры FM_REPORT_HTML_MAIN
  --     в M_PARAMS передаем
  --       {"title":"Заголовок окна отчета", "obj_kind":"obj_kind", "obj_id":"obj_id",
  --        "date_begin":"Дата нач.", "date_end":"Дата окон.", "date_kind":"Тип даты (0,1,2)",
  --        "find_region":"фильтр по региону (0,1)"}
  --   Для M_CODE=SQL,
  --     в M_KIND передаем название команды, эта команда будет обработана в процедуре fm_report_cmd
  --     m_params={"obj_kind":"obj_kind", "obj_id":"obj_id", }
  -- M_CAPTION - текст пункта меню
  --   Для M_CODE=SQL,
  --     в M_KIND передаем название команды, эта команда будет обработана в процедуре fm_report_cmd
  --     m_params={
  --       "obj_kind":"obj_kind",
  --       "obj_id":"obj_id",
  --       "description":"description", -- Основной текст окна диалогового окна выше редактора
  --       "edit_kind": "none", -- none, text, date, combo, number, по умолчанию - none
  --       "comment":"comment", -- Примечание, в нижней части окна ниже редактора
  --       "max_length":"max_length", -- Макс длина строки для edit_kind = (text, number), по-умолчанию не ограничена
  --       "max_lines":"1", -- Макс. количество строк для edit_kind = text, по-умолчанию: 1
  --       "precision":"0", -- Количество знаков после запятой для edit_kind = number
  --       "empty_err":"empty_err", -- Текст ошибки, когда пользователь ничего не указал в редакторе
  --          если не указано, то допускается пустое значение в редакторе, для edit_kind = (text, date, number)
  --       "def_text":"def_text", -- Текст в редакторе по-умолчанию, для edit_kind = (text, mru, date, number)
  --     }
  --     Для параметров description, comment, empty_err - допустимы базовые теги HTML
  --     Пример, для текстового параметра:
  --     m_params =
  --       '{' ||
  --       '"description":"Вы действительно <b>этого</b> хотите?",' ||
  --       '"obj_kind":"' || obj_kind || '",' ||
  --       '"obj_id":"' || obj_id || '",' ||
  --       '"comment":"Если это сделать будет <b>вообще</b> замечательно, ты даже не представляешь, как!",' ||
  --       '"edit_kind":"text",' ||
  --       '"max_lines":4' ||
  --       '"max_length":255,' ||
  --       '"empty_err":"Вводи давай!",'  ||
  --       '"def_text":"Hello World!"' ||
  --       '}';

  obj_kind = coalesce(obj_kind, '');

  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);

  str_date_begin = f_datetostr(date_begin);
  str_date_end = f_datetostr(date_end);

  un_id = (select un_id
           from hp_test_un_id(:un_id));

  select un_name, un_from_c_id
  from username
  where un_id = :un_id
  into un_name, c_id;

  is_manager = 0;
  if (exists(select *
             from hr_pstatus, joblist, jobkind
             where ps_from_c_id = :c_id and
                   ps_date_begin <= current_date and
                   ps_date_end >= current_date and
                   ps_is_working = 1 and
                   ps_status = 1 and
                   ps_from_jl_id = jl_id and
                   jl_kind = jk_id and
                   jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior', 'JobSystem.Region_Director'))) then
    is_manager = 1;
  if (is_manager = 0) then
  begin
    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.CanManageLoans')
               where has_access = 1)) then
      is_manager = 1;
  end

  is_oriv = 0;
  if (exists(select *
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.ViewAccessOriv')
             where has_access = 1)) then
    is_oriv = 1; --если это специалист отдела взыскания

  if (rep_id = 'expressapp_list' or rep_id = 'express_package_list_app' ) then -- Если это экспресс-заявка, подменяем его на просмотр статуса заявки
    rep_id = 'loan_in_progress_status';

  if (is_oriv = 0) then
  begin
    -- Из окна Заявки в процессе можно отправлять на КК
    if (rep_id = 'loan_in_progress_status') then
    begin
      if (db_id in (41701)) then -- RFCKG-4210
      begin
        head = 'Отправить на проверку в Отдел поддержки сети (ОПС)';
        conf = 'Действительно хотите отправить группу в ОПС?';
        note = 'Примечание для ОПС.';
      end
      else
      if (db_id = 81001) then
      begin
        head = 'Отправить на комитет';
        conf = 'Действительно хотите отправить группу на кредитный комитет?';
        note = 'Здесь можете оставить примечание для КК.';
      end
      else
      if (db_id = 94401) then
      begin
        head = 'Komitəyə göndərilsin';
        conf = 'Kredit həqiqətən Kredit Komitəsinə göndərilsin?';
        note = 'Burda Kredit Komitəsi üçün əlavə qeyd saxlıya bilərsiz.';
      end
      m_caption = head;
      m_code = 'SQL';
      m_kind = 'Send_To_KK';
      m_params =
        '{' ||
        '"description":"' || conf || '",' ||
        '"obj_kind":"' || obj_kind || '",' ||
        '"obj_id":"' || obj_id || '",' ||
        '"max_lines":4,' ||
        '"max_length":255,' ||
        '"def_text":"",' ||
        '"empty_err":"",' ||
        '"comment":"' || note || '",' ||
        '"edit_kind":"text"' ||
        '}';
      suspend;
    end

    if (rep_id = 'loan_in_progress_status' and
        db_id = 81001 and
        (is_manager = 1 or
         exists(select *
                from hr_pstatus, joblist, jobkind
                where ps_from_c_id = :c_id and
                      ps_date_begin <= current_date and
                      ps_date_end >= current_date and
                      ps_is_working = 1 and
                      ps_status = 1 and
                      ps_from_jl_id = jl_id and
                      jl_kind = jk_id and
                      jk_system_code in ('JobSystem.LO_Standart')))) then       -- RFCRU-5604
    begin
      head = 'Видео-сертификация';
      conf = 'Ссылка на Яндекс.Телемост';
      note = '';
      m_caption = head;
      m_code = 'SQL';
      m_kind = 'send_video_certification_link';
      m_params =
        '{' ||
        '"description":"' || conf || '",' ||
        '"obj_kind":"' || obj_kind || '",' ||
        '"obj_id":"' || obj_id || '",' ||
        '"max_length":1024,' ||
        '"def_text":"",' ||
        '"empty_err":"ссылка не может быть пустой",' ||
        '"comment":"' || note || '",' ||
        '"edit_kind":"text"' ||
        '}';
      suspend;
    end

    -- Из окна Заявки в процессе можно отправлять на КИБ
    if (rep_id = 'loan_in_progress_status' and
        db_id = 41701) then
    begin
      m_caption = 'Отправить на проверку КИБ';
      m_code = 'SQL';
      m_kind = 'Send_To_Check_KIB';

      m_params = '{"description":"' || iif(db_id <> 94401, 'Действительно хотите отправить группу на проверку КИБ?', 'Действительно хотите отправить клиента на проверку КИБ?') || '","obj_kind":"' || obj_kind || '","obj_id":"' || obj_id || '"}';
      suspend;
    end
    
    if (rep_id = 'loan_in_progress_status' and
        db_id = 39801) then
    begin
      m_caption = 'Отправить на проверку в ПКБ';
      m_code = 'SQL';
      m_kind = 'send_to_pkb';

      m_params = '{"description":"' || f_lang(lang, 'Действительно хотите отправить на проверку в ПКБ?') || '","obj_kind":"' || obj_kind || '","obj_id":"' || obj_id || '"}';
      suspend;
    end

    if (rep_id = 'loan_in_progress_status' and
       :db_id = 81001) then
    begin
      if (obj_kind in ('Clients', '0')) then -- Если входящий параметр клиент
      begin
        k_id_tmp = null;
        ma_id_tmp = null;
        -- Ищем последний договор в процессе
        select ks_from_k_id
        from kontstate, knclient, kontract
        where ks_state = 0 and
              ks_ol_is_delete = 0 and
              ks_from_k_id = kc_from_k_id and
              kc_from_c_id = :obj_id and
              kc_amount_approved > 0 and
              ks_date_end >= '31.12.2999' and
              ks_date_begin >= current_date - 30 and
              ks_from_k_id = k_id and
              k_status >= 0
        order by ks_date_begin desc
        rows 1
        into :k_id_tmp;
        if (k_id_tmp is not null) then
        begin
          obj_kind = 'Kontract';
          obj_id = k_id_tmp;
        end
        else
        begin
          -- Если договор в процессе не найден, ищем заявку
          select ma_id
          from miniappclient, miniapp
          where ma_is_active = 1 and
                ma_id = mac_from_ma_id and
                mac_from_c_id = :obj_id and
                mac_amount > 0 and
                ma_date_end >= '31.12.2999' and
                ma_date_oper >= current_date - 30
          order by ma_date_app desc
          rows 1
          into ma_id_tmp;
          if (ma_id_tmp is not null) then
          begin
            obj_kind = 'Miniapp';
            obj_id = ma_id_tmp;
          end
        end
      end

      obj_kind = f_objkind(obj_kind, 'name');

      if (obj_kind in ('Miniapp', 'Kontract')) then
      begin
        m_caption = 'Отказать в выдаче';
        m_code = 'SQL';
        m_kind = 'cancel_loan';
        m_params = '{' || '"description":"Укажите причину отказа ниже:",' || '"obj_kind":"' || obj_kind || '",' || '"obj_id":"' || obj_id || '",' || '"edit_kind":"combo"' || '}';
        suspend;
      end
    end


    if (obj_kind in ('miniapp', '11') and
        db_id = 94401) then
    begin
      m_caption = f_lang(lang, 'Отправить запрос о месте работы');
      m_code = 'SQL';
      m_kind = 'Send_To_Check_Job';

      m_params = '{"description":"' || f_lang(lang, 'Действительно хотите отправить запрос о месте работы?') || '","obj_kind":"' || obj_kind || '","obj_id":"' || obj_id || '"}';
      suspend;
    end
    if (obj_kind in ('miniapp', '11') and
        db_id in (41701)) then  -- RFCKG-4210
    begin
      m_caption = f_lang(lang, 'Отправить запрос о фактическом адресе проживания');
      m_code = 'SQL';
      m_kind = 'Send_To_Check_address_fact';

      m_params = '{"description":"' || f_lang(lang, 'Действительно хотите отправить запрос об адресе фактического проживания?') || '","obj_kind":"' || obj_kind || '","obj_id":"' || obj_id || '"}';
      suspend;
    end
    if (obj_kind in ('miniapp', '11') and
        db_id in (41701)) then -- RFCKG-4210
    begin
      m_caption = f_lang(lang, 'Отправить запрос о семье');
      m_code = 'SQL';
      m_kind = 'Send_To_Check_family_members';

      m_params = '{"description":"' || f_lang(lang, 'Действительно хотите отправить запрос о семье?') || '","obj_kind":"' || obj_kind || '","obj_id":"' || obj_id || '"}';
      suspend;
    end
    if (obj_kind in ('miniapp', '11') and
        db_id = 94401) then
    begin
      m_caption = f_lang(lang, 'Отправить запрос о пенсии');
      m_code = 'SQL';
      m_kind = 'Send_To_Check_Pension';

      m_params = '{"description":"' || f_lang(lang, 'Действительно хотите отправить запрос о пенсии?') || '","obj_kind":"' || obj_kind || '","obj_id":"' || obj_id || '"}';
      suspend;
    end
    if (obj_kind in ('miniapp', '11') and
        db_id = 94401) then
    begin
      m_caption = f_lang(lang, 'Отправить запрос ИНН для юр.лица');
      m_code = 'SQL';
      m_kind = 'Send_To_Check_INN_Entity';

      m_params = '{"description":"' || f_lang(lang, 'Действительно хотите отправить запрос ИНН для юр.лица?') || '","obj_kind":"' || obj_kind || '","obj_id":"' || obj_id || '"}';
      suspend;
    end
    if (obj_kind in ('miniapp', '11') and
        db_id = 94401) then
    begin
      m_caption = f_lang(lang, 'Отправить запрос о месте работы поручителя');
      m_code = 'SQL';
      m_kind = 'Send_To_Check_Job_Por';

      m_params = '{"description":"' || f_lang(lang, 'Действительно хотите отправить запрос о месте работы?') || '","obj_kind":"' || obj_kind || '","obj_id":"' || obj_id || '"}';
      suspend;
    end
    if (obj_kind in ('miniapp', '11') and
        db_id = 94401) then
    begin
      m_caption = f_lang(lang, 'Отправить запрос о пенсии поручителя');
      m_code = 'SQL';
      m_kind = 'Send_To_Check_Pension_Por';

      m_params = '{"description":"' || f_lang(lang, 'Действительно хотите отправить запрос о пенсии?') || '","obj_kind":"' || obj_kind || '","obj_id":"' || obj_id || '"}';
      suspend;
    end
    if (obj_kind in ('miniapp', '11') and
        db_id = 94401) then
    begin
      m_caption = f_lang(lang, 'Отправить запрос ИНН поручителя для юр.лица');
      m_code = 'SQL';
      m_kind = 'Send_To_Check_INN_Entity_Por';

      m_params = '{"description":"' || f_lang(lang, 'Действительно хотите отправить запрос ИНН для юр.лица?') || '","obj_kind":"' || obj_kind || '","obj_id":"' || obj_id || '"}';
      suspend;
    end

    if (rep_id = 'loan_in_progress_status') then
    begin
      if (obj_kind in ('miniapp', '11')) then
      begin
        m_caption = iif(db_id <> 94401, 'Оформить договор', 'Müqavilə hazırlanması');
        m_code = 'SQL';
        m_kind = 'cmd_create_loan';
        m_params = f_format('{"description":"%1","obj_kind":"%2","obj_id":"%3"}', iif(db_id <> 94401, 'Вы действительно хотите заявку перенести в статус договора?', 'Ərizəni həqiqətən müqavilə mərhələsinə keçirilsin?'), obj_kind, obj_id);
        suspend;
      end

      if (obj_kind in ('kontract', '9')) then
      begin
        if ((is_manager = 1) or (exists(select *
                                        from kontract k
                                        where k_id = :obj_id and
                                              k_status = 0))) then
        begin
          m_caption = iif(db_id <> 94401, 'Вернуть в статус заявки', 'Kredit müraciəti statusuna qayıdın');
          m_code = 'SQL';
          m_kind = 'kontract_to_miniapp';
          m_params = f_format('{"description":"%1","obj_kind":"%2","obj_id":"%3"}', iif(db_id <> 94401, 'Действительно\nвернуть в статус заявки?', 'Tətbiq statusuna çevrilməyinizə əminsiniz?'), obj_kind, obj_id);
          suspend;
        end
        if (exists(select *
                   from kontract k
                   where k_id = :obj_id and
                         k_status = 1) and
            is_manager = 1) then
        begin
          m_caption = iif(db_id <> 94401, 'Вернуть в статус дог/проц', 'Davam edən müqaviləni statusuna qaytarın');
          m_code = 'SQL';
          m_kind = 'kontract_to_progress_status';
          m_params = f_format('{"description":"%1","obj_kind":"%2","obj_id":"%3"}', iif(db_id <> 94401, 'Действительно\nвернуть в статус договор в процессе?', 'Müqaviləni həqiqətən müddətdə vəziyyətə qaytarmaq??'), obj_kind, obj_id);
          suspend;
        end
      end
    end

    if (rep_id = 'loan_in_progress_status'
        and db_id = 81001
        and is_manager = 1
        and obj_kind in ('kontract', '9')
        ) then
    begin
      m_caption = 'Опрос клиента по БКИ';
      m_code = 'HTML';
      m_kind = 'bki_survey_rf';

      m_params = '{"title":"Опрос клиента по БКИ","obj_kind":"kontract","obj_id":"' || obj_id || '"}';
      suspend;
    end

    if (rep_id = 'loan_in_progress_status' and
        is_manager = 1) then
    begin
      if (obj_kind in ('kontract', 'clients', '0', '9')) then
      begin
        m_caption = iif(db_id <> 94401, 'Утвердить кредит', 'Müqaviləni təsdiqləmək');
        m_code = 'SQL';
        m_kind = 'Approve_loan';
        m_params = f_format('{"description":"%1","obj_kind":"%2","obj_id":"%3"}', iif(db_id <> 94401, 'Действительно\nхотите утвердить кредит?', 'Müqaviləni həqiqətən təsdiqləmək istəyirsiniz?'), obj_kind, obj_id);
        suspend;
      end
    end

    -- Из окна Клиента можно делать запрос на фото документа
    if (rep_id = 'Client' and f_get_const('docrecogrequest_active') = 1) then
    begin
      m_caption = 'Запрос на фото документа';
      m_code = 'open';
      m_kind = 'doc_recog_request';
      m_params = '{"obj_kind":"' || obj_kind || '", "obj_id":"' || obj_id || '"}';
      suspend;
    end

    -- Из окна Клиента можно отправлять на КК
    if (rep_id in ('Client', 'loan_closed_list') and
        db_id = 81001) then
    begin
      m_caption = iif(db_id <> 94401, 'Отправить на комитет', 'Komitəyə göndərilsin');
      m_code = 'SQL';
      m_kind = 'Send_To_KK';

      select c_name_fio || ' ' || trim(iif(:db_id <> 94401, 'и остальные участники.', 'və digər üzvləri'))
      from clients
      where c_id = :obj_id
      into txt;

      m_params = f_format('{"description":"%1","obj_kind":"Clients","obj_id":"%2","max_lines":4,"max_length":255,"def_text":"","empty_err":"","comment":"%3","edit_kind":"text"}', txt, obj_id, trim(iif(db_id <> 94401, 'Здесь можете оставить примечание для КК.', 'Burda Kredit Komitəsi üçün əlavə qeyd saxlıya bilərsiz.')));

      suspend;
    end

    -- Из окна Клиента можно просмотреть анкетные данные
    if (rep_id in ('Client', 'loan_closed_list')) then
    begin
      m_caption = iif(db_id <> 94401, 'Анкетные данные', 'Anket məlumatları');
      m_caption = iif(db_id in (39801, 41701), 'Анкета и Анализ клиента', m_caption);
      m_code = 'HTML';
      m_kind = 'AnketaInfo';

      m_params =
        '{' ||
        '"title":"' || iif(db_id <> 94401, 'Анкетные данные', 'Anket məlumatları') || '",' ||
        '"obj_kind":"Clients",' ||
        '"obj_id":"' || obj_id || '",'||
        '"date_begin":"' || str_date_begin || '",' ||
        '"date_end":"' || str_date_end || '",' ||
        '"date_kind":0,' ||
        '"find_region":0' ||
        '}';
      suspend;
    end

    -- Из окна Клиента можно просмотреть Статус заявки
    if (rep_id in ('Client', 'loan_closed_list')) then
    begin
      obj_id_tmp = null;
