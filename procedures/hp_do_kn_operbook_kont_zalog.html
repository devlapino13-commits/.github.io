CREATE PROCEDURE HP_DO_KN_OPERBOOK_KONT_ZALOG(
    KC_ID D_ID,
    OL_DATE D_DATE,
    OL_REG_ID D_INTEGER,
    UN_ID D_ID)
AS
declare variable ob_detail d_str_255;
declare variable dt_acc_id id_type;
declare variable kt_acc_id id_type;
declare variable loan_acc_group d_str_code;
declare variable kontract_no d_str_255;
declare variable kontract_date date;
declare variable oks_diff_esp_bal curr_type;
declare variable oks_diff_esp_norm curr_type;
declare variable oks_llr_os curr_type;
declare variable oks_llr_perc curr_type;
declare variable oks_diff_ecl_llr curr_type;
declare variable close_state integer;
declare variable oks_llr_kind integer;
declare variable non_resident smallint;
declare variable client_type smallint;
declare variable ol_id d_id;
declare variable k_id d_id;
declare variable cl_count d_integer;
declare variable zalog_cost d_integer;
begin
-- Процедура создает новые доп.проводки по выбытию залога

  loan_acc_group = 'Микрозаймы'; -- в будущем нужно прикрепить название раздела системного счета в таблицу продуктов.

  if (ol_date is null) then
    exception e_default 'Date is empty';

  if (ol_date >= '01.03.2019') then
  begin
    select ol_id
    from hp_get_kn_operbook_ol(:ol_date, :ol_reg_id, 'calc_vbs')
    into :ol_id;
  end
  else
  begin
    select k_kontract_no, k_kontract_date
    from kontract, knclient
    where kc_id = :kc_id and
          kc_from_k_id = k_id
    into :kontract_no, kontract_date;
    kontract_no = 'договору займа ' || kontract_no || ' от ' ||f_datetostr(:kontract_date) || ' г.';

    ob_detail = 'Начисления по ' || kontract_no;
    insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind,
                          ol_memo, ol_parent_id, is_delete, ol_user_id)
    values (:ol_reg_id, 2, 30, :ol_date, 9, :ob_detail, null, 0, :un_id)
    returning ol_id
    into :ol_id;
  end

  select non_resident, client_type
  from hp_get_kn_client_resident(:kc_id, :ol_date)
  into non_resident, client_type;

  for select oks_diff_esp_bal, oks_diff_esp_norm, oks_llr_os, oks_llr_perc, oks_diff_ecl_llr,
             oks_close_state, oks_llr_kind
      from kn_oksummary
      where oks_from_kc_id = :kc_id and
            oks_date_begin = :ol_date and
            oks_close_state <> 2
      into oks_diff_esp_bal, oks_diff_esp_norm, oks_llr_os, oks_llr_perc, oks_diff_ecl_llr, close_state,
           oks_llr_kind
  do
  begin
    if (oks_diff_esp_bal is null) then
      oks_diff_esp_bal = 0;
    if (oks_diff_esp_norm is null) then
      oks_diff_esp_norm = 0;
    if (oks_llr_os is null) then
      oks_llr_os = 0;
    if (oks_llr_perc is null) then
      oks_llr_perc = 0;
    if (oks_diff_ecl_llr is null) then
      oks_diff_ecl_llr = 0;

    if (close_state = 1 and oks_llr_kind = 2) then
    begin
      cl_count = 0;
      select kc_from_k_id from knclient
      where kc_id = :kc_id
      into :k_id;

      select count(*)
      from knclient
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      into cl_count;

      if (cl_count > 1) then
      begin
        ob_detail = 'Списание поручительства по обеспеченному займу за балансом в связи с прекращением признания займа';

        select acc_id
        from get_ref_account_office('Внебаланс', '99999', :ol_reg_id, '', -1, :non_resident, :client_type, :ol_date)
        into :dt_acc_id;

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, 'Поручительство.ВнеБаланс', :kc_id, :non_resident, :client_type, :ol_date)
        into :kt_acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, 0, 0, 0, :ol_date,
            :ol_reg_id, 1, :ob_detail, -1, 0);
      end

      zalog_cost = 0;
      select sum(z_cost_likvid)
      from zalog
      where z_from_k_id = :k_id and
            coalesce(z_cost_likvid, 0) <> 0
      into zalog_cost;
      if (zalog_cost is null) then zalog_cost = 0;

      if (zalog_cost <> 0) then
      begin
        ob_detail = 'Списание залоговой стоимости по займу за балансом';
  
        select acc_id
        from get_ref_account_knclient(:loan_acc_group, 'Залог.ВнеБаланс', :kc_id, :non_resident, :client_type, :ol_date)
        into :dt_acc_id;
  
        select acc_id
        from get_ref_account_office('Внебаланс', '99998', :ol_reg_id, '', -1, :non_resident, :client_type, :ol_date)
        into :kt_acc_id;
  
        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, 0, 0, 0, :ol_date,
            :ol_reg_id, 1, :ob_detail, 1, 0);
      end
    end
  end
end
