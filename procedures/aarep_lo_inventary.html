CREATE PROCEDURE AAREP_LO_INVENTARY(
    REP_DATE DATE)
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    KRED_LAST_DATE DATE,
    KRED_REG_CODE VARCHAR(20),
    KRED_REG_NAME VARCHAR(100),
    JL_NAME VARCHAR(100),
    JL_REG_CODE VARCHAR(20),
    JL_REG_NAME VARCHAR(100),
    JL_IS_ON_STAFF INTEGER)
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select distinct c_id, c_name_fio
      from knaktlist, kontract, kontstate, clients
      where kal_from_kf_id = k_from_kf_id and
            k_id = ks_from_k_id and
            knaktlist.is_delete = 0 and
            ks_ol_is_delete = 0 and
            kal_date <= :rep_date and
            kal_date_end >= :rep_date and
            cast(ks_date_begin as date) <= :rep_date and
            ks_date_end >= :rep_date and
            kal_new_kf_officer = c_id and
            ks_state = 1
      into :c_id, c_name_fio
  do
  begin
    jl_name = '';
    jl_reg_code = '';
    jl_reg_name = '';
    kred_last_date = null;
    kred_reg_code = '';
    kred_reg_name = '';

    select kal_date, reg_code, reg_name
    from knaktlist, region
    where knaktlist.is_delete = 0 and
          kal_new_kf_officer = :c_id and
          kal_new_reg_id = reg_unicode
    order by kal_date
    rows 1
    into :kred_last_date, :kred_reg_code, :kred_reg_name;

    select jl_name, reg_code, reg_name, ps_is_on_staff
    from hr_pstatus, joblist, region
    where ps_from_c_id = :c_id and
          ps_from_jl_id = jl_id and
          ps_from_reg_id = reg_unicode and
          ps_date_begin <= :rep_date and
          ps_date_end >= :rep_date and
          ps_is_working = 1
    order by ps_date_begin desc
    rows 1
    into :jl_name, :jl_reg_code, :jl_reg_name, :jl_is_on_staff;

    suspend;
  end
end
