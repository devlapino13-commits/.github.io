CREATE PROCEDURE HP_BUY_GOODS(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_ID D_INTEGER)
AS
declare variable ag_id d_integer;
declare variable reg_id d_integer;
declare variable kontmat_data blob sub_type 1 segment size 80;
begin

  res_text = 'Покупка успешно совершена.';
  res_id = -1;
  for select kml.kml_from_ag_id, ks.ks_region_id
      from kontract k
      inner join kontmat kml on kml.kml_object_id = k.k_id and
            kml.kml_object_kind = 9
      inner join kontstate ks on ks.ks_from_k_id = k.k_id and
            ks.ks_ol_is_delete = 0 and
            current_date between ks.ks_date_begin and ks.ks_date_end
      where k.k_id = :k_id
      into :ag_id, :reg_id

  do
  begin
    kontmat_data = '{"main":{"seller_id":' || :ag_id || ',"region_id":' || :reg_id || '}}';

    select res_id, res_text
    from hp_kontmat_action(:un_id, 'doLoad', :kontmat_data, 1, :k_id)
    into :res_id, :kontmat_data;

    if (res_id = -1) then
    begin
      res_text = :kontmat_data;
      res_id = 0;
      suspend;
      exit;
    end

    res_id = -1;

    select res_id, res_text
    from hp_kontmat_action(:un_id, 'doSave', :kontmat_data, 1)
    into :res_id, :kontmat_data;

    if (res_id = -1) then
    begin
      res_text = :kontmat_data;
      res_id = 0;
      suspend;
      exit;
    end
  end
  suspend;

end
