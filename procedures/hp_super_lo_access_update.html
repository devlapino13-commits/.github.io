CREATE PROCEDURE HP_SUPER_LO_ACCESS_UPDATE
AS
declare variable pm_id d_integer;
declare variable pm_access_kind d_integer;
declare variable subject_kind d_integer;
declare variable subject_id d_id;
declare variable un_id d_id;
declare variable aslt_super_category d_smallint;
begin
  -- обновляет (закрывает/открывает) доступы для Супер КС к особым кредитным продуктам (RFCAZ-427)
  -- запускается роботом ежемесячно 5-го числа

  execute procedure hp_super_lo_access_update_tmp; -- актуализация списка Супер КС на конец предыдущего месяца (analitica_super_lo_tmp)

  select ukt_id
  from hp_get_uniqtable_id('PermActions')
  into subject_kind;

  subject_id = f_get_pa_id('FinStart.Permission_Products');

  -- проверка имеющихся Супер КС
  for select distinct pm_id, pm_access_kind, un_id, aslt_super_category
      from permissions
      inner join username on un_id = pm_object_id
      left join analitica_super_lo_tmp on aslt_lo_id = un_from_c_id
      where pm_is_delete = 0 and
            pm_object_kind = 0 and
            pm_subject_kind = :subject_kind and
            pm_subject_id = :subject_id
      into pm_id, pm_access_kind, un_id, aslt_super_category
  do
  begin
    aslt_super_category = coalesce(aslt_super_category, 0);

    if (pm_access_kind = 0 and aslt_super_category in (1, 2)) then -- доступа нет, но КС в текущем месяце Супер КС -> открываем доступ
      insert into permissions (pm_object_kind, pm_object_id, pm_subject_kind, pm_subject_id, pm_access_kind)
      values (0, :un_id, :subject_kind, :subject_id, 1);

    if (pm_access_kind = 1 and aslt_super_category not in (1, 2)) then -- доступ есть, но КС в текущем месяце не Супер КС -> закрываем доступ
      update permissions
      set pm_is_delete = 1
      where pm_id = :pm_id;
  end

  -- открываем доступы новым Супер КС
  for select un_id
      from analitica_super_lo_tmp
      inner join username on un_from_c_id = aslt_lo_id
      where aslt_super_category in (1, 2)
      into un_id
  do
  begin
    if (not exists(select *
                   from permissions
                   where current_date between pm_date_begin and pm_date_end and
                         pm_is_delete = 0 and
                         pm_access_kind = 1 and
                         pm_object_kind = 0 and
                         pm_object_id = :un_id and
                         pm_subject_kind = :subject_kind and
                         pm_subject_id = :subject_id)) then
      insert into permissions (pm_object_kind, pm_object_id, pm_subject_kind, pm_subject_id, pm_access_kind)
      values (0, :un_id, :subject_kind, :subject_id, 1);
  end
end
