declare function f_my_add_event(events_jso d_integer, event_code d_str_code, event_time d_time = null)
returns d_bool
as
declare variable jso d_integer;
begin
  if (exists(select 1 from hp_bki_json_get_list(json.obj_get_arr(:events_jso, 'event_list'), 1)
      where :event_code = json.obj_get_val(jso, 'code')))   -- когда потребуется присылать несколько однотипных событий в течении дня, то добавить условие на проверку времени
  then
    return null;

  if (exists(select 1 from hp_bki_json_get_list(json.obj_get_arr(:events_jso, 'sent_events'), 0)
            where :event_code = val))
  then
    return null;

  if (json.count_get(json.obj_get_arr(events_jso, 'allow_events')) = 0
    or exists(select 1 from hp_bki_json_get_list(json.obj_get_arr(:events_jso, 'allow_events'), 0)
              where :event_code = val))
  then
  begin
    jso = json.obj_create();
    json.arr_set_obj(json.obj_get_arr(events_jso, 'event_list'), -1, jso);
    json.obj_set_s(jso, 'code', event_code);
    json.obj_set_s(jso, 'time', event_time);
  end

  return null;
end

declare function f_index(jsa d_integer, paths d_str_255, find_value d_str_255, start_index d_integer = null, delimeter4val d_str_10 = null)
returns d_integer
as
  declare variable i d_integer;
  declare variable val d_str_255;
  declare variable path d_str_255;
  declare variable cnt d_integer;
  declare variable fvl_tmp d_str_255;
begin
  if (jsa is null) then
    return null;

  i = coalesce(start_index, 0);
  cnt = json.count_get(jsa);

  while (i < cnt)
  do
  begin
    for select trim(res_str) from hp_split_str(:paths, ',')
    into path
    do
    begin
      val = json.obj_get_val(json.arr_get_obj(jsa, i), path);
      if (delimeter4val is null) then
      begin
        if (val is not distinct from find_value) then
        begin
          return i;
          exit;
        end
      end
      else
      begin
        for select res_str from hp_split_str(:find_value, :delimeter4val)
        into fvl_tmp
        do
          if (val is not distinct from fvl_tmp) then
          begin
            return i;
            exit;
          end
      end
    end

    i = i + 1;
  end 
end

declare function f_send2reset (bkk_id d_id, object_kind d_integer, object_id d_id, event_date d_date, tag d_integer)
returns d_bool
as
  declare variable c_id d_id;
  declare variable bkt_id d_id;
  declare variable bkl_id d_id;
  declare variable bkc_id d_id;
  declare variable bev_id d_id;
  declare variable packet_id d_id;
  declare variable counter d_integer;
  declare variable packet_name d_str_code;
  declare variable start_date d_date;
begin
  select bkt_id
  from bki_packet_type
  where bkt_from_bkk_id = :bkk_id
    and bkt_action = f_enum('bki_packet_action', 'reset', 'id')
    and bkt_format = f_enum('bki_packet_format', 'loan', 'id')
  into bkt_id;

  select bkp_id
  from bki_packet
  where bkp_from_bkt_id = :bkt_id
    and bkp_date = current_date
    and bkp_status = 0
  order by bkp_id desc
  rows 1
  into packet_id;

  if (packet_id is null) then
  begin
    -- создадим пакет
    select coalesce(max(bkp_number), 0)
    from bki_packet, bki_packet_type
    where bkp_from_bkt_id = bkt_id
      and bkt_from_bkk_id = :bkk_id
      and bkp_date = current_date
      and bkt_format = f_enum('bki_packet_format', 'loan', 'id')
    into counter;

    --имя пакета
    execute procedure hp_bki_get_packet_name(current_date, (select bkk_code from bki_kind where bkk_id = :bkk_id), counter, 0)
    returning_values packet_name, counter;

    insert into bki_packet (bkp_code, bkp_from_bkk_id, bkp_date, bkp_number, bkp_from_bkt_id, bkp_status)
    values (:packet_name, :bkk_id, current_date, :counter, :bkt_id, 0)
    returning bkp_id
    into packet_id;
  end

  select bkl_id
  from bki_loans
  where bkl_from_bkp_id = :packet_id
    and bkl_object_kind = :object_kind
    and bkl_object_id = :object_id
  into bkl_id;

  if (bkl_id is null) then
    insert into bki_loans (bkl_from_bkp_id, bkl_object_kind, bkl_object_id)
    values (:packet_id, :object_kind, :object_id)
    returning bkl_id
    into bkl_id;

  for select distinct mac_from_c_id
  from miniappclient
  where :object_kind = 11
    and mac_from_ma_id = :object_id
  union
  select kc_from_c_id
  from knclient
  where :object_kind = 9
    and kc_from_k_id = :object_id
  into c_id
  do
  begin
    select bkc_id, f_json_str_get_value(bkc_add, 'start_date', null)
    from bki_clients
    where bkc_from_bkl_id = :bkl_id
      and bkc_from_c_id = :c_id
    into bkc_id, start_date;

    if (bkc_id is null) then
      insert into bki_clients (bkc_from_bkl_id, bkc_from_c_id, bkc_add)
      values (:bkl_id, :c_id, f_format('{"start_date":"%1", "change_reason":"1"}', :event_date))
      returning bkc_id
      into bkc_id;
    else if (coalesce(start_date, '31.12.9999') > event_date) then
      update bki_clients
      set bkc_add = f_format('{"start_date":"%1", "change_reason":"1"}', :event_date)
      where bkc_id = :bkc_id;

    select bev_id
    from bki_events
    where bev_from_bkc_id = :bkc_id
      and bev_event_date = :event_date
    into bev_id;

    if (bev_id is null) then
      insert into bki_events (bev_from_bkc_id, bev_event_date, bev_tag)
      values (:bkc_id, :event_date, :tag);
  end

  return 1;
end

begin
  const_null_date = '01.01.1900';

  select bev_event_date, bev_tag, bkl_object_id, bkl_object_kind, bkl_id, bkp_id, f_enum('bki_packet_action', bkt_action, 'code'),
      bkc_add, bkp_from_bkk_id, bkt_format
  from bki_packet, bki_loans, bki_clients, bki_events, bki_packet_type
  where bkl_from_bkp_id = bkp_id
    and bkc_from_bkl_id = bkl_id
    and bev_from_bkc_id = bkc_id
    and bkt_id = bkp_from_bkt_id
    and bev_id = :bev_id_in
  into event_date, tag, object_id, object_kind, bkl_id, bkp_id, bkt_action_code,
      bkc_add, bkk_id, bkt_format;

  if (object_kind = f_objkind('miniapp', 'id')) then
    select k_id
    from kontract k
    where k.k_from_ma_app_id = :object_id
      and k.k_status >= 0
    order by 1 desc
    rows 1
    into k_id;
  else if (object_kind = f_objkind('kontract', 'id')) then
    k_id = object_id;

  if (bkt_format = 0) then
  begin
    if (bkt_action_code = 'reset') then
        select coalesce(max(bev_event_date), :const_null_date)
        from bki_packet, bki_loans, bki_clients, bki_events
        where bkl_from_bkp_id = bkp_id
          and bkc_from_bkl_id = bkl_id
          and bev_from_bkc_id = bkc_id
          and bev_status > 0
          and (bkp_status between 0 and 3 or bkp_id = :bkp_id)
          and coalesce(bev_delete_bev_id, 0) = 0
          and bev_event_id in (9, 10)
          and bkl_object_kind = 9
          and bkl_object_id = :k_id
          and bev_event_date < :event_date
          and bkp_id = :bkp_id
        into last_debt_calc_date;
  
    if (coalesce(last_debt_calc_date, const_null_date) = const_null_date) then
        select coalesce(max(bev_event_date), :const_null_date)
        from bki_packet, bki_loans, bki_clients, bki_events
        where bkl_from_bkp_id = bkp_id
          and bkc_from_bkl_id = bkl_id
          and bev_from_bkc_id = bkc_id
          and bev_status > 0
          and (bkp_status between 0 and 3 or bkp_id = :bkp_id)
          and coalesce(bev_delete_bev_id, 0) = 0
          and bev_event_id in (9, 10)
          and bkl_object_kind = 9
          and bkl_object_id = :k_id
          and bev_event_date < :event_date
        into last_debt_calc_date;
  end

  execute procedure hp_bki_get_loan_info5(:jsa, :object_kind, :object_id, :event_date, :last_debt_calc_date, :xml_version_id,
      iif(:bkt_action_code = cast('delete' as d_str_code), 1, 0), :bkt_format);

  client_timezone = 3; -- заглушка

  for select jso from hp_bki_json_get_list(:jsa, 1)
  into jso
  do
  begin
    -- события
    events_jso = json.obj_create();
    event_list_jsa = json.arr_create();
    json.obj_set_arr(events_jso, 'event_list', event_list_jsa);
    sent_events_jsa = json.arr_create();
    json.obj_set_arr(events_jso, 'sent_events', sent_events_jsa);
    allow_events_jsa = json.arr_create();
    json.obj_set_arr(events_jso, 'allow_events', allow_events_jsa);

    sent_events_jso = json.obj_create();

    uid = json.obj_get_val(jso, 'uid');
    uid_force = coalesce(uid, json.obj_get_val(jso, 'info_app_uid'));
    c_id = json.obj_get_val(:jso, 'c_id');
    bkc_id = null;

    select bkc_id
    from bki_clients
    where bkc_from_bkl_id = :bkl_id
      and bkc_from_c_id = :c_id
      and coalesce(f_bki_data_text(bkc_uid_bkd_id), :uid_force) = :uid_force
    into bkc_id;

    if (bkc_id is null) then
      insert into bki_clients (bkc_from_bkl_id, bkc_from_c_id, bkc_uid_bkd_id, bkc_add)
      values(:bkl_id, :c_id, f_bki_data_id(:uid_force), :bkc_add)
      returning bkc_id
      into bkc_id;
    else
      update bki_clients
      set bkc_uid_bkd_id = f_bki_data_id(:uid_force)
      where bkc_id = :bkc_id;

    -- Заполним титульную часть
    cl_path = c_id || '_' || event_date;
    jso_client = json.obj_get_obj(jso_clients, cl_path);

    if (jso_client is null) then
    begin
      jso_client = json.obj_create(jso_clients, cl_path);
      json.obj_set_s(jso_client, 'bkk_id', bkk_id);
      json.obj_set_s(jso_client, 'deal_sign_credit', json.obj_get_val(jso, 'deal_sign_credit'));
      execute procedure hp_bki_get_client_info5(:c_id, :event_date, :jso_client, :xml_version_id, :bkt_format);
    end

    if (json.contains(jso_clients, 'bkc_id=' || bkc_id) = 0) then
    begin
      json.obj_create(jso_clients, 'bkc_id=' || bkc_id);
      execute procedure hp_bki_calc_save('Title', xml_version_id, 1, bkc_id, jso_client);
    end

    -- если не удаление и не коррекция, то определим события на эту дату
    -- найдем события, которые уже передавались
    if (bkt_action_code = 'default' and tag not in (5, 55)) then -- если переотправка, то отправляем все
    begin
      for select left(event_code, position('_', event_code) - 1), bev_hash, event_date
      from hp_bki_get_sended_loan_events(:bkk_id, :object_id)
      where uid = :uid_force
        and bkp_status between 0 and 3
        and bev_status > 0
        and coalesce(event_code, '') <> ''
        and bkt_format = :bkt_format
      order by event_date
      into event_tmp, old_hash_str, event_date_tmp
      do
      begin
        if (event_date_tmp = event_date) then
        begin
          if (event_tmp in ('1.1', '1.2', '1.3', '2.13.1', '2.13.2', '2.13.3', '2.13.4', '2.13.5')) then
            json.arr_set_s(sent_events_jsa, -1, event_tmp);
          else if (event_tmp not in ('1.7', '1.9')) then
            json.obj_set_s(sent_events_jso, '"'||event_tmp||'"', old_hash_str);
        end
        else
        begin
          if (event_tmp = '2.2') then
            json.obj_set_s(sent_events_jso, 'disb_date', event_date_tmp);
          else if (event_tmp = '2.5') then
            json.obj_set_s(sent_events_jso, 'close_date', event_date_tmp);

          if (event_tmp in ('2.1', '2.3', '2.4', '2.5', '2.6')) then
            json.obj_set_s(sent_events_jso, 'last_date', event_date_tmp);
        end 
      end
    end

    if (bkt_format = 2) then
    begin
      if (json.obj_get_val(jso, 'deal_sign_credit', 0) <> 1) then -- Антифрод только для потреб. кредитов
        continue;

      -- найдем события  если это пакет антифрода (AF)
      -- 2.13.1 (аналог 1.1) Получено обращение заявителя в целях заключения договора потребительского займа (кредита)
      if (json.obj_get_val(jso, 'info_app_date') = event_date) then
        f_my_add_event(events_jso, '2.13.1', cast(json.obj_get_val(jso, 'info_app_datetime') as timestamp));

      -- 2.13.2 (аналог 1.2) Одобрено обращение заявителя в целях заключения договора потребительского займа (кредита)
      if (json.obj_get_val(jso, 'approve_date') = event_date)then
        f_my_add_event(events_jso, '2.13.2', cast(json.obj_get_val(jso, 'approve_datetime') as timestamp));

      -- 2.13.3 (аналог 1.4) С заемщиком заключен договор потребительского займа (кредита)
      if (json.obj_get_val(jso, 'deal_date') = event_date) then
        f_my_add_event(events_jso, '2.13.3', cast(json.obj_get_val(jso, 'deal_datetime') as timestamp));

      -- 2.13.4 (аналог 2.2) Заемщику переданы денежные средства по заключенному договору потребительского займа (кредита)
      if (json.obj_get_val(jso, 'cred_start_debt_date') = event_date) then
        f_my_add_event(events_jso, '2.13.4', cast(json.obj_get_val(jso, 'cred_start_debt_datetime') as timestamp));

      -- 2.13.5 (аналог 1.3) Заявителю отказано в заключении договора потребительского займа (кредита)
      if (json.obj_get_val(jso, 'info_failure_date') = event_date
        and json.obj_get_val(jso, 'status') = 'annul'
        and json.contains(events_jso, '1.2') = 0)
      then
        f_my_add_event(events_jso, '2.13.5', cast(json.obj_get_val(jso, 'info_failure_datetime') as timestamp));
    end
    else if (bkt_action_code = 'delete') then
    begin
      if (object_kind = f_objkind('kontract', 'id')) then
        f_my_add_event(events_jso, '3.3');
      else
        f_my_add_event(events_jso, '3.4');
    end
    else if (json.obj_get_val(jso, 'cession_date') = event_date) then   -- Права по обязательству полностью перешли к другому лицу
        f_my_add_event(events_jso, '2.11');
    else 
    begin
      select max(bev_event_date)
      from bki_packet_type, bki_packet, bki_loans, bki_clients, bki_events
      where bkp_from_bkt_id = bkt_id
        and bkl_from_bkp_id = bkp_id
        and bkc_from_bkl_id = bkl_id
        and bev_from_bkc_id = bkc_id
        and bkt_format = 0
        and bev_status > 0
        and bkp_status between 0 and 3
        and coalesce(bev_delete_bev_id, 0) = 0
        and bkc_from_c_id = :c_id
        and bev_event_date < :event_date
      into last_send_client_date;      -- последняя отправка по клиенту

      -- если это ручное отправление, то найдем разрешенные к отправке события
      if (bkt_action_code in ('reset', 'recreate')) then -- для действия 3.2 не доступны события по обращениям
      begin
        json.arr_set_s(allow_events_jsa, -1, '1.4');
        json.arr_set_s(allow_events_jsa, -1, '2.1');
        json.arr_set_s(allow_events_jsa, -1, '2.2');
        json.arr_set_s(allow_events_jsa, -1, '2.3');
        json.arr_set_s(allow_events_jsa, -1, '2.4');
        json.arr_set_s(allow_events_jsa, -1, '2.5');
        json.arr_set_s(allow_events_jsa, -1, '2.6');
      end
      else
        for select left(bml_allow_events, position('_', bml_allow_events) - 1)
        from bki_manual_main, bki_manual_list
        where bmm_from_bkp_id = :bkp_id
          and bml_from_bmm_id = bmm_id
          and bml_object_kind = :object_kind
          and bml_object_id = :object_id
          and coalesce(bml_allow_events, '') <> ''
        into event_tmp
        do
          json.arr_set_s(allow_events_jsa, -1, event_tmp);


      -- Обращение клиента (новая заявка)
      if (json.obj_get_val(jso, 'info_app_date') = event_date) then
        f_my_add_event(events_jso, '1.1', cast(json.obj_get_val(jso, 'info_app_datetime') as timestamp));

      if (json.obj_get_val(jso, 'info_failure_date') = event_date
          and json.obj_get_val(jso, 'status') = 'cl_refusal') then
        f_my_add_event(events_jso, '1.2', cast(json.obj_get_val(jso, 'info_failure_datetime') as timestamp));
      -- Заявка одобрена
      else if (json.obj_get_val(jso, 'approve_date') = event_date)then
        f_my_add_event(events_jso, '1.2', cast(json.obj_get_val(jso, 'approve_datetime') as timestamp));

      -- Аннулирование заявки
      if (json.obj_get_val(jso, 'info_failure_date') = event_date
        and json.obj_get_val(jso, 'status') = 'annul'
        and json.contains(events_jso, '1.2') = 0)
      then
        f_my_add_event(events_jso, '1.3', cast(json.obj_get_val(jso, 'info_failure_datetime') as timestamp));

      if (uid is not null) then
      begin
        -- Сделка совершена
        if (json.obj_get_val(jso, 'deal_date') = event_date) then
          f_my_add_event(events_jso, '1.4', cast(json.obj_get_val(jso, 'deal_datetime') as timestamp));
  
        -- Субъекту передана сумма займа
        if (json.obj_get_val(jso, 'cred_start_debt_date') = event_date) then
          f_my_add_event(events_jso, '2.2', cast(json.obj_get_val(jso, 'cred_start_debt_datetime') as timestamp));
  
        -- Изменились сведения об исполнении обязательства
        if (coalesce(json.obj_get_val(jso, 'last_payout_date'), const_null_date) = event_date  -- погашение
           or (json.obj_get_val(jso, 'calc_date') = event_date                           -- или произошел пересчет
              and event_date is distinct from json.obj_get_val(jso, 'cred_start_debt_date')      -- не во время выдачи
              and json.obj_get_val(jso, 'contract_end_date') is null))                           -- и договор не закрыт
        then
          if (event_date > json.obj_get_val(jso, 'deal_end_date') or json.obj_get_val(jso, 'is_poruch', 0) = 0) then
            f_my_add_event(events_jso, '2.3', cast(json.obj_get_val(jso, 'calc_datetime') as timestamp));

        -- Изменились сведения об обеспечении
        if (f_index(json.obj_get_arr(jso, 'FL_32_35_Group'), 'collateral_sign', 1) is not null
          and f_index(json.obj_get_arr(jso, 'FL_32_35_Group'), 'date, date_assessment, fact_end_date, calc_date', event_date) is not null)
        then
        begin
          -- если event_date = дате выдачи или дате закрытия, то используем их, если нет, то cast(event_date as timestamp)
          if (event_date = json.obj_get_val(jso, 'contract_end_date', '31.12.9999')) then
            f_my_add_event(events_jso, '2.4', cast(json.obj_get_val(jso, 'contract_end_datetime') as timestamp));

          else if (event_date = json.obj_get_val(jso, 'cred_start_debt_date')) then
            f_my_add_event(events_jso, '2.4', cast(json.obj_get_val(jso, 'cred_start_debt_datetime') as timestamp));

          else if (event_date > json.obj_get_val(jso, 'cred_start_debt_date') and
                   event_date < json.obj_get_val(jso, 'contract_end_date', '31.12.9999')) then
            f_my_add_event(events_jso, '2.4', cast(event_date as timestamp));
        end
        else if (f_index(json.obj_get_arr(jso, '33'), 'warranty_sign', 1) is not null
          and f_index(json.obj_get_arr(jso, '33'), 'date, end_date', event_date) is not null
          and json.obj_get_val(jso, 'deal_ratio') = '1')
        then
        begin
          -- если event_date = дате выдачи или дате закрытия, то используем их, если нет, то cast(event_date as timestamp)
          if (event_date = json.obj_get_val(jso, 'contract_end_date', '31.12.9999')) then
            f_my_add_event(events_jso, '2.4', cast(json.obj_get_val(jso, 'contract_end_datetime') as timestamp));

          else if (event_date = json.obj_get_val(jso, 'cred_start_debt_date')) then
            f_my_add_event(events_jso, '2.4', cast(json.obj_get_val(jso, 'cred_start_debt_datetime') as timestamp));

          else if (event_date > json.obj_get_val(jso, 'cred_start_debt_date') and
                   event_date < json.obj_get_val(jso, 'contract_end_date', '31.12.9999')) then
            f_my_add_event(events_jso, '2.4', cast(event_date as timestamp));
        end
        -- Изменились сведения о судебном споре
        if (f_index(json.obj_get_arr(jso, '39'), 'court_sign', 1) is not null) then
          f_my_add_event(events_jso, '2.6');
  
        -- Изменились сведения об условиях обязательства CКИ
        if (f_index(json.obj_get_arr(jso, '23'), 'sign', 1) is not null) then
          if (event_date <= json.obj_get_val(jso, 'contract_end_date', '31.12.9999')) then
            f_my_add_event(events_jso, '2.1',  cast(json.obj_get_val(jso, '23[0].applyDateTime') as timestamp));
  
        -- Изменились сведения об условиях обязательства CКИ
        if (f_index(json.obj_get_arr(jso, '23_1'), 'sign', 1) is not null
          or json.obj_get_val(jso, 'writeoff_date') is not distinct from event_date   -- списание
          or (json.obj_get_val(jso, 'full_cost_date') is not distinct from event_date
              and json.obj_get_val(jso, 'is_poruch', 0) = 0
              and event_date is distinct from json.obj_get_val(jso, 'cred_start_debt_date'))   -- изменение ПСК
          )
        then
          if (event_date <= json.obj_get_val(jso, 'contract_end_date', '31.12.9999')) then
            f_my_add_event(events_jso, '2.1', cast(json.obj_get_val(jso, 'd_23_1_datetime') as timestamp));

        -- Обязательство субъекта прекратилось
        if (json.obj_get_val(jso, 'contract_end_date') = event_date) then
          f_my_add_event(events_jso, '2.5', cast(json.obj_get_val(jso, 'contract_end_datetime') as timestamp));
      end

      -- Изменились сведения о CКИ в титульной части
      -- Титульная часть передается во всех событиях. Если нет ни одного события а данные изменились, создадим событие 1.7
      if (last_send_client_date is not null
          and json.obj_get_val(jso_client, 'pers_data_change.ch_title', 0) = 1
          and json.obj_get_val(jso_client, 'pers_data_change.set_title', 0) = 0)
      then
      begin
        f_my_add_event(events_jso, '1.7');
        json.obj_set_s(jso_client, 'pers_data_change.set_title', 1);
      end
      -- Изменились сведения о CКИ в основной части
      -- эти изменения передаются только в 1.4. Если нет ни одного события а данные изменились, создадим событие 1.9
      if (last_send_client_date is not null
          and json.obj_get_val(jso_client, 'pers_data_change.ch_base', 0) = 1
          and f_index(event_list_jsa, 'code', '1.4') is null)
      then
        f_my_add_event(events_jso, '1.9', cast(json.obj_get_val(jso_client, 'pers_data_change.signed_datetime') as timestamp));

      -- Изменились сведения по делу о банкротстве CКИ
      if (json.obj_get_val(jso_client, 'bankruptcy_fact', 0) = 1 and json.obj_get_val(jso, 'is_poruch', 0) = 0)
      then
        f_my_add_event(events_jso, '1.12');
    end

    for select json.obj_get_val(jso, 'code'), json.obj_get_val(jso, 'time')
    from hp_bki_json_get_list(:event_list_jsa, 1)
    into event_code, event_time
    do
    begin
      bev_id = null;

      passport_date = null;

      if (event_code = '1.7') then
        passport_date = json.obj_get_val(jso_client, 'doc_date', event_date);

      select bev_id
      from bki_events
      where bev_from_bkc_id = :bkc_id
        and bev_event_id is null
        and bev_status = 0
        and bev_event_date = :event_date
      into bev_id;

      -- по-умолчанию статус "Пустая строка"
      update bki_events
      set bev_status = -3
      where bev_id = :bev_id;

      -- если uid'а договора нет а событие определилось по-умолчанию, то игнорим - это заявка, по ней такое событие невозможно
      if (event_code = '2.3' and uid is null) then
        continue;

      -- выставим ВЫЧИСЛЯЕМЫЕ значения показателей, которые зависят от передаваемого события (остальное указывается в приоритетных значениях)
      if (event_code = '1.2') then
      begin
        if (json.obj_get_val(jso, 'status') = 'cl_refusal') then
        begin
          json.obj_set_s(jso, 'info_app_stage_code', '4');  -- Отозвано субъектом до одобрения источником обращения или отказа от совершения сделки по обращению
          json.obj_set_s(jso, 'info_app_stage_date', json.obj_get_val(jso, 'info_failure_date'));
          json.obj_set_s(jso, 'info_failure_date', null);
          json.obj_set_s(jso, 'info_app_loan_sum', null);
          json.obj_set_s(jso, 'info_app_loan_currency', null);
        end
        else
        begin
          json.obj_set_s(jso, 'info_app_stage_code', '2');  -- Одобрено
          json.obj_set_s(jso, 'info_app_stage_date', json.obj_get_val(jso, 'approve_date'));
        end
      end
      else if (event_code = '3.4') then
      begin
        if (json.obj_get_val(jso, 'status') = 'annul') then
        begin
          json.obj_set_s(jso, 'info_app_stage_code', '3');  -- Отозвано субъектом до одобрения источником обращения или отказа от совершения сделки по обращению
          json.obj_set_s(jso, 'info_app_stage_date', json.obj_get_val(jso, 'info_failure_date'));
          json.obj_set_s(jso, 'info_app_loan_sum', null);
          json.obj_set_s(jso, 'info_app_loan_currency', null);
        end
        else if (json.obj_get_val(jso, 'status') = 'cl_refusal') then
        begin
          json.obj_set_s(jso, 'info_app_stage_code', '4');  -- Отозвано субъектом до одобрения источником обращения или отказа от совершения сделки по обращению
          json.obj_set_s(jso, 'info_app_stage_date', json.obj_get_val(jso, 'info_failure_date'));
          json.obj_set_s(jso, 'info_failure_date', null);
          json.obj_set_s(jso, 'info_app_loan_sum', null);
          json.obj_set_s(jso, 'info_app_loan_currency', null);
        end
        else
        begin
          json.obj_set_s(jso, 'info_app_stage_code', '2');  -- Одобрено
          json.obj_set_s(jso, 'info_app_stage_date', json.obj_get_val(jso, 'approve_date'));
        end
      end
      else if (event_code = '2.13.1') then
        json.obj_set_s(jso, 'info_app_stage_datetime', f_bki_datetime_to_str(json.obj_get_val(jso, 'info_app_datetime'), 3));
      else if (event_code = '2.13.2') then
        json.obj_set_s(jso, 'info_app_stage_datetime', f_bki_datetime_to_str(json.obj_get_val(jso, 'approve_datetime'), 3));
      else if (event_code = '2.13.3') then
        json.obj_set_s(jso, 'info_app_stage_datetime', f_bki_datetime_to_str(json.obj_get_val(jso, 'deal_datetime'), 3));
      else if (event_code = '2.13.4') then
        json.obj_set_s(jso, 'info_app_stage_datetime', f_bki_datetime_to_str(json.obj_get_val(jso, 'cred_start_debt_datetime'), 3));
      else if (event_code = '2.13.5') then
        json.obj_set_s(jso, 'info_app_stage_datetime', f_bki_datetime_to_str(json.obj_get_val(jso, 'info_failure_datetime'), 3));


      if (json.obj_get_val(jso, 'avg_load_range') is not null) then
      begin
        if (event_code in ('1.2', '1.3')) then
          json.obj_set_s(jso, 'avg_load_calculation_fact_sign', '0');
        else
          json.obj_set_s(jso, 'avg_load_calculation_fact_sign', '1');
      end

      event_id = (select el_id from enum_list where el_kind = 'bki_event' and el_code starting with :event_code || '_');
      if (event_id is null) then
        exception e_default 'Not found event_id for event_code = ' || coalesce(event_code, '<NULL>');

      hash_str = hex_encode(crypt_hash(
                                (select list(coalesce(var_value, ''))
                                from hp_bki_get_valuelist4event('Event_' || :event_code, :xml_version_id, :jso))
                              using sha256));

      if (bkt_format = 0 and bkt_action_code = 'default') then
      begin
        old_hash_str = json.obj_get_val(sent_events_jso, '"'||event_code||'"');

        if (hash_str = old_hash_str) then -- если хэш старого события совпадает с новым, то это дубль, не отправляем
          continue;
        else if (old_hash_str is not null) then -- если старое событие есть, но хеши не совпадают, то надо делать переотправку
        begin
          f_send2reset(bkk_id, object_kind, object_id, event_date, tag);
          continue;
        end

        -- если выдача, но ранее передавали выдачу в другую дату, то надо делать переотправку
        if (event_code = '2.2') then
        begin
          event_date_tmp = cast(json.obj_get_val(sent_events_jso, 'disb_date') as date);

          if (event_date_tmp is not null) then
          begin
            f_send2reset(bkk_id, object_kind, object_id, minvalue(event_date_tmp, event_date), tag);
            continue;
          end
        end

        -- если передавали закрытие на более раннюю дату, то надо делать переотправку
        event_date_tmp = cast(json.obj_get_val(sent_events_jso, 'close_date') as date);

        if (event_date_tmp < event_date and event_code in ('2.1', '2.3', '2.4', '2.5', '2.6')) then
        begin
          f_send2reset(bkk_id, object_kind, object_id, event_date_tmp, tag);
          continue;
        end

        -- если закрытие, но ранее передавали что-то на более поздние даты, то надо делать переотправку
        if (event_code = '2.5') then
        begin
          event_date_tmp = cast(json.obj_get_val(sent_events_jso, 'last_date') as date);

          if (event_date_tmp > event_date) then
          begin
            f_send2reset(bkk_id, object_kind, object_id, event_date, tag);
            continue;
          end
        end
      end

      if (bev_id is null) then
        insert into bki_events (bev_from_bkc_id, bev_event_id, bev_event_date, bev_tag, bev_loan_status, bev_status, bev_event_time, bev_hash)
        values (:bkc_id, :event_id, coalesce(:passport_date, :event_date), :tag, json.obj_get_val(:jso, 'status'), 1, :event_time, :hash_str)
        returning bev_id
        into bev_id;
      else
        update bki_events
        set bev_event_id = :event_id,
            bev_event_date = coalesce(:passport_date, :event_date),
            bev_tag = :tag,
            bev_loan_status = json.obj_get_val(:jso, 'status'),
            bev_status = 1,
            bev_event_time = :event_time,
            bev_hash = :hash_str
        where bev_id = :bev_id;

      if (event_code = '2.11') then
      begin
        for select scl_id
        from sud_cession_packet, sud_cession_loans
        where scl_from_scp_id = scp_id
          and scl_from_k_id = :k_id
          and scp_date = :event_date
        into scl_id
        do
          update sud_cession_loans
          set scl_from_bev_id = :bev_id
          where scl_id = :scl_id;
      end 

      if (event_code = '2.6') then
      begin
        for select json.obj_get_val(jso, 'court_sub_id', 0)
        from hp_bki_json_get_list(json.obj_get_arr(:jso, '39'), 1)
        into sub_id
        do
          if (sub_id > 0) then
            update sud_bki
            set sub_from_bev_id = :bev_id,
                sub_status = 2
            where sub_id = :sub_id;
      end

      for select jso from hp_bki_json_get_list(json.obj_get_arr(:jso, '23'), 1)
      into jso_tmp
      do
      begin
        bod_id = json.obj_get_val(jso_tmp, 'bod_id');
        if (bod_id is null) then
          continue;

        update bki_once_data bod
        set bod_from_bkc_id = :bkc_id
        where bod_id = :bod_id;
      end

      for select jso from hp_bki_json_get_list(json.obj_get_arr(:jso_client, 'FL_2_5_Group'), 1)
      into jso_tmp
      do
      begin
        hist_fio = json.obj_get_val(jso_tmp, 'hist_fio');
        hist_cds_id = json.obj_get_val(jso_tmp, 'hist_cds_id');

        if (hist_fio is not null) then
        begin
          bod_id = null;
          select bod_id from bki_once_data bod
          where bod_kind = 2
            and bod_object_kind = 0
            and bod_object_id = :c_id
            and bod_add = :hist_fio
            and (bod_from_bkc_id is null
                or bod_from_bkc_id = :bkc_id)
          rows 1
          into bod_id;

          if (bod_id is not null) then
            update bki_once_data bod
            set bod_from_bkc_id = :bkc_id
            where bod_id = :bod_id;
          else
            insert into bki_once_data (bod_kind, bod_object_kind, bod_object_id, bod_add, bod_from_bkc_id)
            values (2, 0, :c_id, :hist_fio, :bkc_id);
        end

        if (hist_cds_id is not null) then
        begin
          bod_id = null;
          select bod_id from bki_once_data bod
          where bod_kind = 1
            and bod_object_kind = 0
            and bod_object_id = :c_id
            and bod_from_cds_id = :hist_cds_id
            and (bod_from_bkc_id is null
                or bod_from_bkc_id = :bkc_id)
          rows 1
          into bod_id;

          if (bod_id is not null) then
            update bki_once_data bod
            set bod_from_bkc_id = :bkc_id
            where bod_id = :bod_id;
          else
            insert into bki_once_data (bod_kind, bod_object_kind, bod_object_id, bod_from_cds_id, bod_from_bkc_id)
            values (1, 0, :c_id, :hist_cds_id, :bkc_id);
        end
      end

      action_code_tmp = case event_code
                          when '1.1' then 'A'
                          when '1.4' then 'A'
                          when '3.1' then 'C.1'
                          when '3.2' then 'C.1'
                          when '3.3' then 'C.2'
                          when '3.4' then 'C.2'
                          when '3.5' then 'C.2'
                          when '2.13.1' then 'A'
                          else 'B'
                        end;

      -- найдем узел для события и запишем
      execute procedure hp_bki_calc_save('Event_' || event_code, xml_version_id, 2, bev_id, jso, action_code_tmp,
                            event_date + coalesce(event_time, time '0:0:0'), iif(event_time is null, 777, client_timezone));

      if (event_code in ('1.7', '1.4', '1.9', '1.12')) then
        execute procedure hp_bki_calc_save('Event_' || event_code, xml_version_id, 2, bev_id, jso_client, null, null, null, 1);
    end

    -- если не нашлось события
    update bki_events
    set bev_status = -3
    where bev_id = :bev_id_in
      and bev_status = 0
      and bev_event_id is null;
  end
end
