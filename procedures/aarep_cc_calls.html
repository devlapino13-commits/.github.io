CREATE PROCEDURE AAREP_CC_CALLS(
    REP_PROJECT STR_100,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    CC_PROJECT STR_100,
    COMMAND_NAME STR_100,
    ABONENT_TYPE STR_100,
    CALLED_DATE DATE,
    CALLED_DATETIME TIMESTAMP,
    K_KONTRACT_NO STR_100,
    DIALOG_PRESENT INTEGER,
    DIALOG_MISSED INTEGER,
    ABONENT_FIO STR_100,
    ABONENT_CELL_NO STR_100,
    OPER_CALLED STR_100,
    OPER_BLOCKED D_STR_100)
AS
declare variable RD_RES integer;
declare variable C_ID ID_TYPE;
declare variable CC_ID ID_TYPE;
declare variable K_ID ID_TYPE;
declare variable PANT_ID ID_TYPE;
begin
  for select proj.rd_name, 'Звонки клиентам', pcl_called_date, pc.pcl_called_datetime, pc.pcl_rd_res,
             pc.pcl_from_c_id, null, pantk.pant_oper_called, pantk.pant_from_k_id, pantk.pant_id,
             iif(un.un_name is null, pantk.pant_oper_called, un.un_name), f_cont_info('clients', pcl_from_c_id, 'cell_main', 'ok')
      from pantera_clients pc
      inner join pantera_kontracts pantk on pantk.pant_id = pc.pcl_from_pan_id
      inner join refdata proj on proj.rd_id = pantk.pant_rd_project_type
      left join pantera_clients_opers pclo on pclo.pco_from_pcl_id = pc.pcl_id
      left join username un on un.un_id = pclo.pco_from_un_id
      where pc.pcl_called_date between :rep_date_begin and :rep_date_end and
            proj.rd_code containing (:rep_project)
      union all
      select proj.rd_name, 'Звонки контактным лицам', pcon.pcon_called_date,
             pcon.pcon_called_datetime, pcon.pcon_rd_res, null, pcon.pcon_from_cc_id, pantk.pant_oper_called,
             pantk.pant_from_k_id, pantk.pant_id, iif(un.un_name is null, pantk.pant_oper_called, un.un_name), null
      from pantera_contacts pcon
      inner join pantera_kontracts pantk on pantk.pant_id = pcon.pcon_from_pan_id
      inner join refdata proj on proj.rd_id = pantk.pant_rd_project_type
      left join pantera_contacts_opers pcco on pcco.pco_from_pcon_id = pcon.pcon_id
      left join username un on un.un_id = pcco.pco_from_un_id
      where pcon.pcon_called_date between :rep_date_begin and :rep_date_end and
            proj.rd_code containing (:rep_project)
      order by 1
      into :cc_project, :abonent_type, :called_date, :called_datetime, :rd_res, :c_id, :cc_id, :oper_blocked, :k_id, :pant_id, :oper_called, abonent_cell_no
  do
  begin
    command_name = '';

    if (:rep_project in ('OKTELL.PROJECTS.PANTERA','OKTELL.PROJECTS.SIRENA','OKTELL.PROJECTS.KS_FINANCE')) then
    begin
        select pcm.pcm_name
        from pantera_commands pcm
        inner join pantera_kontracts_com pkc on pkc.pkc_from_pcm_id = pcm.pcm_id
        where pkc.pkc_from_pant_id = :pant_id
        into command_name;
    end

    select kont.k_kontract_no
    from kontract kont
    where kont.k_id = :k_id
    into :k_kontract_no;

    dialog_present = null;
    dialog_missed = null;

    select iif(rd.rd_tag > 0, 1, 0), iif(rd.rd_tag <= 0, 1, 0)
    from refdata rd
    where rd.rd_id = :rd_res
    into :dialog_present, :dialog_missed;

    abonent_fio = null;

    if (c_id is not null) then
    begin
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into :abonent_fio;
    end

    if (cc_id is not null) then
    begin
      select cont.cc_name, cc_cell_no
      from contacts cont
      where cont.cc_id = :cc_id
      into :abonent_fio, abonent_cell_no;
    end

    suspend;
  end
end
