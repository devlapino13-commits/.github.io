CREATE PROCEDURE HP_KONTMAT_OWN_WRITEOFF(
    K_ID D_ID,
    OL_MAIN_ID D_ID)
AS
declare variable kml_id d_integer;
declare variable bill_id d_id;
declare variable bill_line_no d_integer;
declare variable wp_id d_id;
declare variable motv_id d_id;
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable un_id d_id;
declare variable quantity d_currency;
declare variable ol_id d_id;
declare variable ol_parent_id d_id;
declare variable ol_date d_date;
declare variable ol_doc_no d_integer;
declare variable ol_doc_kind d_integer;
declare variable ol_oper_type d_integer;
declare variable ol_oper_subtype d_integer;
declare variable ol_summa d_currency;
declare variable ol_memo d_str_255;
declare variable k_currency d_str_lang;
declare variable loan_reg_id d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable res_reg_id d_integer;
declare variable res_acc_dt d_str_20;
declare variable res_acc_kt d_str_20;
declare variable total_line d_integer;
declare variable res_line_no d_integer;
begin
  if (not exists(select * from kontmat_own where kmo_k_id = :k_id))
    then exit;

  un_id = (select ol_user_id from operlist where ol_id = :ol_main_id);
  ol_oper_type = 3;
  ol_oper_subtype = 5;
  select
    'Выбытие ТМЦ по дог. №' || k_kontract_no,
    k_currency,
    k_last_reg_id,
    k_disb_date
  from kontract
  where k_id = :k_id
  into
    :ol_memo,
    :k_currency,
    :loan_reg_id,
    :ol_date;
  acc_dt = (select al_id from acc_list
    where al_from_reg_id = :loan_reg_id and al_account = '10977/1');

  for
    select
      kmo_kml_id,
      kmo_bill_id,
      kmo_bill_line_no,
      kmo_wp_id,
      kmo_motv_id,
      kmo_reg_id,
      kmo_c_id,
      kmo_quantity,
      kmo_total
    from kontmat_own
    where kmo_k_id = :k_id
  into
    :kml_id,
    :bill_id,
    :bill_line_no,
    :wp_id,
    :motv_id,
    :reg_id,
    :c_id,
    :quantity,
    :ol_summa
  do
  begin
    ol_doc_no = (select num from last_doc_number(:ol_date, 2, :loan_reg_id));
    acc_kt = (select al_id from acc_list
      where al_from_reg_id = :reg_id and al_account = '11209/10');
    for
      select
        res_reg_id,
        res_acc_dt,
        res_acc_kt,
        total_line,
        res_line_no
      from aarep_loronostro_link(:loan_reg_id, :acc_dt, :acc_kt)
    into
      :res_reg_id,
      :res_acc_dt,
      :res_acc_kt,
      :total_line,
      :res_line_no
    do
    begin
      ol_id = (select first 1 kv from  get_kv(f_hq_id(), 5));
      ol_parent_id = ol_main_id;
      ol_doc_kind = iif(res_line_no = 1, 2, 9);

      insert into operlist (
        ol_id,
        ol_doc_no,
        ol_from_reg_id,
        ol_date,
        ol_oper_type,
        ol_oper_subtype,
        ol_doc_kind,
        ol_memo,
        ol_user_id,
        ol_document_client,
        ol_is_client,
        ol_parent_id
      )
      values (
        :ol_id,
        :ol_doc_no,
        :res_reg_id,
        :ol_date,
        :ol_oper_type,
        :ol_oper_subtype,
        :ol_doc_kind,
        :ol_memo,
        :un_id,
        :c_id,
        1,
        :ol_parent_id
        );
      insert into operbook (
        ob_from_ol_id,
        ob_curr_rate,
        ob_currency,
        ob_account_dt,
        ob_account_kt,
        ob_summa_val,
        ob_summa,
        ob_detail
      )
      values (
        :ol_id,
        1,
        :k_currency,
        :res_acc_dt,
        :res_acc_kt,
        :ol_summa,
        :ol_summa,
        :ol_memo
      );
      if (res_line_no = 1) then ol_parent_id = ol_id;
    end
    insert into opermat (
      om_from_ol_id,
      om_number,
      om_bill_id,
      om_bill_line_no,
      om_oper_type,
      om_workplace_id,
      om_matotvet_id,
      om_summa_minus,
      om_kolich_minus,
      om_nds_percent,
      om_account,
      om_is_written_off,
      om_account_info
      )
    select first 1
      :ol_parent_id,
      1,
      om_bill_id,
      om_bill_line_no,
      5,
      :wp_id,
      :motv_id,
      :ol_summa,
      :quantity,
      0,
      om_account,
      1,
      om_account_info
    from opermat
    where om_bill_id = :bill_id
      and om_bill_line_no = :bill_line_no
      and om_oper_type = 1;

    update kontmat_own set kmo_ol_id = :ol_parent_id where kmo_kml_id = :kml_id;
  end
end
