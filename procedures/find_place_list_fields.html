CREATE PROCEDURE FIND_PLACE_LIST_FIELDS(
    PLACE_ID DECIMAL(18,0))
RETURNS (
    PL_ID DECIMAL(18,0),
    PARENT DECIMAL(18,0),
    PL_PATH VARCHAR(255),
    PL_NAME VARCHAR(100),
    PL_STATUS INTEGER,
    PL_CAN_SELECT INTEGER,
    PLT_NAME VARCHAR(255),
    PLACE_FULL_REGION VARCHAR(255),
    PLACE_FULL VARCHAR(255),
    PL_NAME_KOMITET VARCHAR(100))
AS
declare variable txt varchar(255);
declare variable p_id decimal(18,0);
declare variable finish integer;
begin
  /* Procedure Text */
  for select pl_id, parent, pl_name, pl_name_komitet, pl_status, plt_name, plt_can_select
      from places, placetype
      where pl_status = plt_id and
            pl_id = :place_id
      into :pl_id, :parent, :pl_name, :pl_name_komitet, :pl_status, :plt_name, :pl_can_select
  do
  begin
    pl_path = '';
    finish = 0;
    for select pl_id, pl_name
        from places_list(:pl_id, 0, -1, 0), places
        where id_list = pl_id
        into :p_id, :txt
    do
    begin
      pl_path = :txt || ' ' || :pl_path;
    end
    place_full_region = '';
    place_full = '';
    select place_full, place_full_region
    from aarep_places_link(:pl_id)
    into :place_full, :place_full_region;
    suspend;
  end
end
