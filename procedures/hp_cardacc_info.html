CREATE PROCEDURE HP_CARDACC_INFO(
    CCA_ID D_INTEGER,
    DT_OPER D_DATE = current_date,
    SHOW_CNAME D_BOOL = 0)
RETURNS (
    AMOUNT D_CURRENCY,
    AMOUNT_BALANCE D_CURRENCY,
    C_NAME_FIO D_STR_100)
AS
declare variable acc d_str_20;
begin
  acc = '20001/1';

  with cardlist_tmp as (select cal_card_id as card_id
                        from cardlist
                        where cal_from_cca_id = :cca_id),
       kontract_tmp as (select clo_from_k_id as k_id,
                               oks_balance
                        from cardlistloan
                        inner join cardlist_tmp on card_id = clo_card_id
                        left join oksummary on oks_from_k_id = clo_from_k_id and oks_close_state = 0 and :dt_oper between oks_date_begin and oks_date_end
                        group by clo_from_k_id, oks_balance)
  select sum(iif(ol_date <= :dt_oper, iif(ob_short_acc_kt = :acc, ob_summa, 0) - iif(ob_short_acc_dt = :acc, ob_summa, 0), 0)),
         (select sum(oks_balance)
          from kontract_tmp)
  from operbook
  inner join(select distinct ok_from_ol_id as olo_id
             from operkred
             inner join kontract_tmp on k_id = ok_from_k_id
             union
             select distinct oko_from_ol_id as olo_id -- RFCKG-3041
             from operkroff
             inner join kontract_tmp on k_id = oko_from_k_id
             where oko_type = 4
             union
             select distinct ol_id as olo_id -- RFCKG-3041
             from cardlistloan
             inner join cardlist_tmp on card_id = clo_card_id
             inner join operkred on ok_from_k_id = clo_from_k_id
             inner join operlist on ol_id = (select coalesce(ol_parent_id, ol_id)
                                             from operlist
                                             where ol_id = ok_from_ol_id)
             where ol_oper_type = 2
             union
             select ol_id as olo_id
             from pvnopers
             inner join cardlist_tmp on card_id = po_card_id
             inner join operlist on ol_parent_id = (select coalesce(ol_parent_id, ol_id)
                                                    from operlist
                                                    where ol_id = po_from_ol_id) or ol_id = (select coalesce(ol_parent_id, ol_id)
                                                                                             from operlist
                                                                                             where ol_id = po_from_ol_id)
             where po_is_mb_card = 1 and
                   po_from_ol_id > 0 and
                   po_is_delete = 0
             union
             select ol_id as olo_id
             from ipc_exd_detail
             inner join cardlist_tmp on card_id = ied_from_card_id
             inner join operlist on ied_from_ol_id in (ol_id, ol_parent_id)
             where ied_from_ol_id > 0) as a on a.olo_id = ob_from_ol_id
  inner join operlist on ol_id = ob_from_ol_id and is_delete = 0
  where ob_short_acc_dt = :acc or ob_short_acc_kt = :acc
  into :amount,
       :amount_balance;

  amount = coalesce(amount, 0);
  amount_balance = coalesce(amount_balance, 0);
  if (show_cname = 1) then
    c_name_fio = (select c_name_fio
                  from cardclientacc
                  inner join cardclientid on cci_id = cca_from_cci_id
                  inner join clients on c_id = cci_from_c_id
                  where cca_id = :cca_id);

  suspend;
end
