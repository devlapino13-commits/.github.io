CREATE PROCEDURE CM_RU_0072_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable c_id id_type;
declare variable c_is_personal smallint;
declare variable pr_name varchar(50);
declare variable min_sum integer;
declare variable max_sum integer;
declare variable k_period decimal(15,2);
declare variable c_citizen integer;
declare variable k_percent perc_type;
declare variable c_hired_date date;    
declare variable ma_date date;
declare variable rep_date date;
declare variable k_from_ma_app_id integer;
declare variable par30 integer;
declare variable c_portfel_del curr_type;
declare variable c_portfel curr_type;
declare variable par30_rd integer;
declare variable c_portfel_del_rd curr_type;
declare variable c_portfel_rd curr_type;
declare variable par30_o integer;
declare variable portfel_del curr_type;
declare variable portfel curr_type;
declare variable par30_o_rd integer;
declare variable portfel_del_rd curr_type;
declare variable portfel_rd curr_type;
declare variable per_reg_id id_type;
declare variable per_rd_code varchar(255); 
declare variable kol_kl integer;
begin
--Разработан по приказу №16-08-16 о/д
--  if (k_id is null) then k_id = 101010015618400101;
  select kc_from_c_id, pr.pr_name,
         KC_AMOUNT_approved,
         k.k_period,
         k.k_percent,
         k_disb_date,
         k.k_from_ma_app_id
  from kontract k, product pr, knclient kc
  where :k_id = k_id
    and k_from_pr_id = pr_id
    and pr_code containing '.IND.'
    and pr_code containing '.EMP.'
    and pr_code containing '.FOREIGN.'
    and k_id = kc_from_k_id
    and kc_amount_approved > 0
  into c_id, pr_name, KC_AMOUNT, k_period, k_percent, kf_disb_date, k_from_ma_app_id;
  if (c_id is not null) then
  begin
    select c_name_fio, c.c_is_personal, c.c_citizen
    from clients c
    where c_id = :c_id
    into res_info_text, c_is_personal, c_citizen;
    
    select distinct ps_from_reg_id, rd.rd_code
    from hr_pstatus, joblist jl, jobkind, refdata rd
    where current_date between ps_date_begin and ps_date_end
    and ps_from_c_id = :c_id
    and ps_is_working = 1
    and ps_status = 1
    and ps_from_jl_id = jl_id
    and jl_kind = jk_id
    and jk_system_id = rd_id
    and rd.rd_main_code containing 'JobSystem'
    --and rd.rd_code = 'JobSystem.Region_KM'
    order by ps_date_begin desc rows 1
    into per_reg_id, per_rd_code;
        
    --Общие правила
    --1.1 Микрозайм иностранному сотруднику(Заем) инд.кред.без залога
    if (per_rd_code is null) then
    begin
      res_error_is_critical=2;
      res_error_text='Только сотрудники компании могут получить займ по продукту: '||pr_name;
      suspend;
    end else
    begin   
      --1.2 Заем предоставляется только иностранным сотрудникам
      if (c_citizen not in(8,9,12)) then
      begin
        res_error_is_critical=iif(c_citizen = 13, 0, 1);
        res_error_text='Только иностранные сотрудники компании могут получить займ по продукту: '||pr_name;
        suspend;
      end else
      --Стаж работы более 9 мес.
      if (kf_disb_date > dateadd(month, 9, (
                                        select max(ps_date_begin)
                                        from hr_pstatus
                                        where ps_status=1
                                        and ps_from_c_id=:C_ID))) then
      begin
        res_error_is_critical=1;
        res_error_text='Стаж сотрудника не позволяет получить займ по продукту: '||pr_name;
        suspend;
      end else
      begin
        --2 Условия кредитования
        --2.1 Сумма займа от 20000 до 35000(не более суммы оклада)
        min_sum = 20000;
        if (kf_disb_date < '12.09.16') then max_sum = 35000;
        if (kf_disb_date >= '12.09.16') then max_sum = 50000;
        
        if (kc_amount not between min_sum and max_sum) then
        begin
          res_error_is_critical=1;
          res_error_text='Сумма займа должен быть от '||min_sum||' до '||max_sum;
          suspend;
        end
        
        --2.2 Срок займа 3 - 6 мес.
        if (k_period not between 3 and 6) then
        begin
          res_error_is_critical=1;
          res_error_text='Срок займа должен быть от 3х до 6 мес.';
          suspend;
        end
        /*
        --2.3 %%ставка = 25%
        if (k_percent <> 25 and kf_disb_date < '12.09.16') then
        begin
          res_error_is_critical=1;
          res_error_text='Процентная ставка должен быть: 25';
          suspend;
        end
        
        --2.3 %%ставка = 24% от 12 сент 16 года
        if (k_percent <> 24 and kf_disb_date >= '12.09.16') then
        begin
          res_error_is_critical=1;
          res_error_text='Процентная ставка должен быть: 24';
          suspend;
        end
        */
        --3 Требование к заемщику
        --3.1 Отсуствие выговора
        if (exists (
                select *
                from rebuke r
                where rb_is_delete=0
                and rb_from_c_id=:c_id
                and r.rb_type = 1
                and :kf_disb_date between r.rb_date and rb_period_start+365)) then
        begin
          res_error_is_critical=1;
          res_error_text='У сотрудника есть выговор, выдача запрещена!';
          suspend;
        end
        --3.1 Показатель ПАР30 менее 7%(на отчетный день и в день заявки)
        select min(ma_date_app)
        from miniapp ma
        where ma.ma_id = :k_from_ma_app_id
        and ma.ma_is_active = 1
        into ma_date;
            
        --Определяем отчетный день
        rep_date = cast('01.'||extract(month from :kf_disb_date)||'.'||extract(year from :kf_disb_date) as date)-1;
        
        if (per_rd_code containing 'LO_Standart') then
        begin
          select sum(oks_balance) c_portfel,
                 sum(iif(:ma_date <= oks_delinq_date, oks_balance, 0)) c_portfel_del
          from oksummary oks
          where oks_close_state = 0
          and oks_kc_id = :c_id
          and :ma_date between oks_date_begin and oks_date_end
          into c_portfel, c_portfel_del;
            
          par30 = coalesce(c_portfel_del / c_portfel, 0);
            
          if (par30 > 7) then
          begin
            res_error_is_critical=1;
            res_error_text='PAR30 сотрудника на дату заявки '||par30||'%!';
            suspend;
          end
            
          select sum(oks_balance) c_portfel,
                 sum(iif(:rep_date <= oks_delinq_date, oks_balance, 0)) c_portfel_del
          from oksummary oks
          where oks_close_state = 0
          and oks_kc_id = :c_id
          and :rep_date between oks_date_begin and oks_date_end
          into c_portfel_rd, c_portfel_del_rd;
        
          par30_rd = coalesce(c_portfel_del_rd / c_portfel_rd, 0);
          if (par30_rd > 7) then
          begin
            res_error_is_critical=1;
            res_error_text='PAR30 сотрудника на последнюю отчетную дату '||par30_rd||'%!';
            suspend;
          end
            
          select count(distinct kc_from_c_id)
          from oksummary oks, knclient
          where oks_close_state = 0
          and oks_kc_id = :c_id
          and :rep_date between oks_date_begin and oks_date_end
          and oks_From_K_id = kc_from_K_id
          and kc_amount_approved > 0
          into kol_kl;
        
          kol_kl = coalesce(kol_kl, 0);
        
          if (kol_kl < 200) then
          begin
            res_error_is_critical=1;
            res_error_text='Количество активных клиентов на последнюю отчетную дату меньше 200!';
            suspend;
          end
        end
        
        if (per_rd_code containing 'Region_KM') then
        begin
          select sum(oks_balance) c_portfel,
                 sum(iif(:ma_date <= oks_delinq_date, oks_balance, 0)) c_portfel_del
          from oksummary oks
          where oks_close_state = 0
          and oks_region_id = :per_reg_id
          and :ma_date between oks_date_begin and oks_date_end
          into portfel, portfel_del;
        
          par30_o = coalesce(portfel_del / portfel, 0);
    
          if (par30_o > 7) then
          begin
            res_error_is_critical=1;
            res_error_text='PAR30 офиса на дату заявки '||par30_o||'%!';
            suspend;
          end
    
          select sum(oks_balance) c_portfel,
                 sum(iif(:rep_date <= oks_delinq_date, oks_balance, 0)) c_portfel_del
          from oksummary oks
          where oks_close_state = 0
          and oks_region_id = :per_reg_id
          and :rep_date between oks_date_begin and oks_date_end
          into portfel_rd, portfel_del_rd;
        
          par30_o_rd = coalesce(portfel_del_rd / portfel_rd, 0);
        
          if (par30_o_rd > 7) then
          begin
            res_error_is_critical=1;
            res_error_text='PAR30 офиса на последнюю отчетную дату '||par30_o_rd||'%!';
            suspend;
          end
        end
      end
    end
  end
end
