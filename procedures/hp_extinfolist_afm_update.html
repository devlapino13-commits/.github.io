CREATE PROCEDURE HP_EXTINFOLIST_AFM_UPDATE(
    EIL_ID D_ID)
AS
declare variable BLS_ID D_ID;
declare variable C_ID D_ID;
declare variable UN_ID D_ID;
declare variable MA_ID D_ID;
declare variable EIC_CODE D_STR_CODE;
declare variable SQL_TEXT D_STR_LONG;
declare variable JSON_INFO D_BLOB_TEXT;
declare variable CATEGORY_ID D_SMALLINT;
declare variable STATUS D_SMALLINT;
declare variable AFM_FLAG D_SMALLINT;
declare variable PDL_FLAG D_SMALLINT;
declare variable EIL_STATUS D_SMALLINT;
declare variable STATUS_ERR D_SMALLINT;
begin
    select
        eil_from_bls_id,
        eil_from_c_id,
        eil_from_un_id,
        eil_object_id,
        eic_code,
        eil_status
    from extinfolist
    inner join extinfocfg on eil_from_eic_id = eic_id
        and eic_code in ('afm_iin_check', 'afm_name_check')
        and eil_status in (-2, 2)
    where eil_id = :eil_id
    into
        :bls_id,
        :c_id,
        :un_id,
        :ma_id,
        :eic_code,
        :eil_status;

    if (:eic_code not in ('afm_iin_check', 'afm_name_check') or :eic_code is null or bls_id is null) then
        exit;

    -- Проверяем наличие блоба
    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into :json_info;


    -- Все проверки пройдены, можно обрабатывать результат
    afm_flag = 0;
    pdl_flag = 0;

    -- В момент проверки клиента по всем перечням ПОД/ФТ, данный триггер вызывается дважды, при 'afm_iin_check' и 'afm_name_check'
    -- чтобы не делать лишнего insert или update or insert, добавляю такоую проверку по клиенту на дату
    if (not exists
        (
            select *
            from afm_check ac
            where ac.ac_from_c_id = :c_id
                and ac_from_ma_id = :ma_id
                and ac.ac_is_actual = 1
        )
    ) then
    begin
        if (eil_status = -2 or coalesce(:json_info, '') = '') then
        begin 
            status_err = f_enum('afm_status', 'afm_error', 'id');
    
            insert into afm_check
            (
                ac_from_c_id, ac_from_ma_id, ac_from_un_id, ac_date, ac_is_actual, ac_category_1,
                ac_category_2, ac_category_3, ac_category_4, ac_category_5, ac_category_6, ac_category_7,
                ac_category_8, ac_category_9, ac_category_10, ac_category_11, ac_category_12, ac_category_13, ac_category_14, ac_category_15
            )
            values
            (
                :c_id, :ma_id, :un_id, current_date, 1, :status_err, :status_err,
                :status_err, :status_err, :status_err, :status_err, :status_err, :status_err, :status_err,
                :status_err, :status_err, :status_err, :status_err, :status_err, :status_err
            );
    
            exit;
        end
        else
        begin
            -- так как мы отправляем и получаем два запроса, нам необходимо только один и самый первый раз изменять c_pft_risk_koef
            update clients c
            set c_pft_risk_koef = 0
            where c_id = :c_id;
    
            -- Делаем не актуальными все проверки по АФМ, которые были ранее
            update afm_check
            set ac_is_actual = 0
            where ac_from_c_id = :c_id
                and ac_from_ma_id = :ma_id
                and ac_is_actual = 1;
    
            -- Создаём запись (по умолчанию клиент, все категории выставлены на 1 или же не найден в списках)
            insert into afm_check (ac_from_c_id, ac_from_ma_id, ac_from_un_id, ac_date)
            values (:c_id, :ma_id, coalesce(:un_id, 1000000000000), current_date);
        end
    end

    if (coalesce(:json_info, '') = '') then
        exit;

    -- Сохраняем и обрабатываем все ответы, где status <> не найден в списках ПОД/ФТ
    -- К сожалению АФМ не оперирует таким понятием, как кодовое навзание категории, поэтому работаем с целочисленным значением (category_id)
    for select distinct category_id, status
    from hp_parse_json_afm_check(:json_info, :eic_code)
    where status <> 1
    into :category_id, :status
    do
    begin
        if (:eic_code = 'afm_name_check' and :category_id in (11,12)) then
            continue;

        sql_text = f_format(
        '
            update afm_check
            set ac_category_%1 = %2
            where ac_from_c_id = %3
                and ac_from_ma_id = %4
                and ac_is_actual = 1
        ', :category_id, :status, :c_id, :ma_id);

        execute statement(:sql_text);

        -- Обработка перечней террористов и экстремистов
        if (afm_flag = 0
            and ((:eic_code = 'afm_iin_check' and :category_id in (11,12)) or (:eic_code = 'afm_name_check' and :category_id in (1,4,5,6,7,8,9,10,14,15)))
        ) then
        begin
            update clients
            set c_pft_risk_koef = 2
            where c_id = :c_id;

            afm_flag = 1;
            continue;
        end
        -- Обработка перечня ПДЛ (Публичные Должностные Лица)
        if (pdl_flag = 0
            and (:eic_code = 'afm_iin_check' and :category_id in (13))
        ) then
        begin
            update clients
            set c_is_public = 1
            where c_id = :c_id;

            update or insert into clientbenefic (cb_from_c_id, cb_is_owner)
            values (:c_id, 0)
            matching(cb_from_c_id, cb_is_owner);

            pdl_flag = 1;
            continue;
        end
    end

    -- RFCKZ-128
    execute procedure hp_afm_send_compliance_mail(f_objkind('clients', 'id'), :c_id, 'afm_active');
    execute procedure hp_afm_send_compliance_mail(f_objkind('clients', 'id'), :c_id, 'afm_excluded');

    if (:eic_code = 'afm_name_check') then
        execute procedure hp_afm_process_fio_list(:c_id, :ma_id);
end
