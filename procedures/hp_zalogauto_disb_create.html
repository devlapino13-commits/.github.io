CREATE PROCEDURE HP_ZALOGAUTO_DISB_CREATE(
    K_ID D_ID,
    UN_ID D_ID)
AS
declare variable Z_ID D_INTEGER;
declare variable Z_FROM_C_ID D_ID;
declare variable Z_COST_MARKET D_CURRENCY;
declare variable Z_COST_LIKVID D_CURRENCY;
declare variable Z_COST_KONTRACT D_CURRENCY;
declare variable K_DISB_DATE D_DATE;
declare variable C_ID D_ID;
declare variable ZAL_ID D_INTEGER;
declare variable ZAP_ID D_INTEGER;
declare variable PR_CODE D_STR_255;
declare variable RES_IS_ERROR D_SMALLINT;
declare variable RES_TEXT D_STR_255;
declare variable ZA_ID D_INTEGER;
declare variable KS_FROM_OL_ID D_ID;
declare variable STS_ID D_ID;
declare variable PTS_ID D_ID;
begin
  -- Восстановление данных о профсуждении из таблицы Zalog при выдаче займа

  select ks_date_begin, pr_code, ks_from_ol_id
  from kontstate, kontract, product
  where k_id = :k_id and
        ks_from_k_id = k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        k_from_pr_id = pr_id
  into k_disb_date, pr_code, ks_from_ol_id;

  if (position('.zalog_car.', pr_code) = 0 and position('.zalog.', pr_code) = 0) then
    exit;


  select z_id, z_from_c_id, z_cost_market, z_cost_likvid, z_cost_kontract
  from zalog
  where z_from_k_id = :k_id and
        z_type = 1 and
        z_subtype = 4
  order by z_id
  rows 1
  into z_id, z_from_c_id, z_cost_market, z_cost_likvid, z_cost_kontract;

  if (z_id is null) then
    exception e_default f_format('Zalog not found, k_id: %1', k_id);

  select za_id, za_from_zal_id
  from zalogauto
  where za_from_k_id = :k_id
  into za_id, zal_id;

  if (za_id is null or zal_id is null) then
  begin
    select kc_from_c_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into c_id;

    -- Создаем первичное проф.суждение при выдаче
    select res_is_error, res_text
    from hp_zalogauto_edit(null, 'kontract', :k_id, :z_from_c_id, :z_cost_kontract, 0, 0, null, null, '', :un_id)
    into res_is_error, res_text;
  
    if (res_is_error = 1) then
      exception e_default res_text;

    update zalogauto
    set za_from_z_id = :z_id
    where za_from_k_id = :k_id
    returning za_id, za_from_zal_id
    into za_id, zal_id;

    select cds_id
    from clientdocs
    where cds_from_c_id in (:c_id, :z_from_c_id) and
          cds_from_pk_id = 310 and -- СТС
          cds_doc_date_begin <= (:k_disb_date) and
          (cds_is_delete = 0 or
          (cds_is_delete = 1 and
          cds_delete_datetime > :k_disb_date)) and
          cds_verify_status = 2
    order by cds_create_datetime desc
    rows 1
    into sts_id;
    if (sts_id is null) then
      exception e_message 'В документах клиента не найден верифицированный СТС';

    select cds_id
    from clientdocs
    where cds_from_c_id in (:c_id, :z_from_c_id) and
          cds_from_pk_id = 315 and -- ПТС
          cds_doc_date_begin <= (:k_disb_date) and
          (cds_is_delete = 0 or
          (cds_is_delete = 1 and
          cds_delete_datetime > :k_disb_date)) and
          cds_verify_status = 2
    order by cds_create_datetime desc
    rows 1
    into pts_id;

    if (pts_id is null) then
      exception e_message 'В документах клиента не найден верифицированный ПТС';

    update zalogauto
    set za_orig_sts_id = :sts_id,
        za_orig_pts_id = :pts_id
    where za_id = :za_id;

    insert into zalogautoblob (zab_from_zal_id, zab_from_bls_id)
    select :zal_id, cds_doc_blob_1
    from clientdocs
    where cds_from_c_id = :c_id and
          cds_from_pk_id = 317 and
          cds_verify_status >= 2 and
          cds_is_delete = 0 and
          cds_doc_blob_1 is not null
    union
    select :zal_id, cds_doc_blob_2
    from clientdocs
    where cds_from_c_id = :c_id and
          cds_from_pk_id = 317 and
          cds_verify_status >= 2 and
          cds_is_delete = 0 and
          cds_doc_blob_2 is not null;
  end

  update zalogautolist
  set zal_status = 2
  where zal_id = :zal_id
  returning zal_from_zap_id
  into zap_id;

  update zalogautoprof
  set zap_date = :k_disb_date,
      zap_avg_cost = :z_cost_market,
      zap_zalog_cost = :z_cost_likvid,
      zap_status = 2,
      zap_from_ol_id = :ks_from_ol_id
  where zap_id = :zap_id;
end
