CREATE PROCEDURE UXP$UI_EXTFORMACTIONS
RETURNS (
    UA_NAME D_STR_100,
    UA_HINT D_STR_255,
    UNIT_NAME D_STR_CODE,
    COMPONENT_NAME D_STR_CODE,
    DATASET_NAME D_STR_CODE,
    ACTION_KIND D_SMALLINT,
    REPLACE_KIND D_SMALLINT,
    DATA_LINKED D_BOOL,
    UA_UUID D_UUID,
    FIELD_PAIRS D_STR_8100)
AS
declare variable efm_uuid d_uuid;
declare variable prv_component d_str_code;
declare variable replace_kind_tmp d_smallint;
begin
  field_pairs = '';

  for select efm_uuid, efm_code
  from ux$extform
  into efm_uuid, unit_name
  do
  begin
     prv_component = '';
     replace_kind = 0;
     for select ua_uuid, ua_ext_component, ua_ext_dataset, ua_ext_action, ua_ext_replace_kind,
                ua_name, ua_hint
     from ux$action
     where ua_owner = :efm_uuid
     order by ua_ext_component
     into :ua_uuid, :component_name, :dataset_name, :action_kind, :replace_kind_tmp,
          :ua_name, :ua_hint
     do
     begin
       data_linked = 0;
       if (prv_component = component_name and replace_kind_tmp > 0) then
         replace_kind = replace_kind + replace_kind_tmp;
       else
         replace_kind = replace_kind_tmp;

       if (exists(select * from ux$actionparam
       where uap_ua_uuid = :ua_uuid and
             uap_use_default = 0 and
             uap_is_output = 0 and
             coalesce(uap_owner_ext_field, '') <> '')) then
         data_linked = 1;

       suspend;
       prv_component = component_name;
     end
  end
end
