CREATE PROCEDURE MC_UI_VISIBILITY(
    MCL_ID D_ID,
    LOAN_QTY D_INTEGER,
    EXTMINIAPP_QTY D_INTEGER)
RETURNS (
    UI_PART D_STR_100,
    UI_PART_DESCRIPTION D_STR_255,
    IS_VISIBLE D_SMALLINT,
    MESSAGE_TXT D_STR_255)
AS
declare variable EXTMINIAPP_QTY_PROCESS D_SMALLINT;
declare variable EXTMINIAPP_QTY_ALL D_SMALLINT;
declare variable MAX_REJECT_DATE D_DATE;
declare variable DB_ID D_INTEGER;
declare variable LANG_CODE D_STR_LANG;
declare variable C_ID D_ID;
declare variable C_VERIFY_STATUS D_SMALLINT;
declare variable USER_CODE D_INTEGER;
declare variable EDS_EXIST D_INTEGER; /* Наличие договора ЭЦП */
declare variable OS D_INTEGER;
declare variable HIDE_IOS_PRODUCTS D_INTEGER;
declare variable MCL_PROJECT D_STR_20;
begin
  db_id = f_get_db_id();
  lang_code = f_get_lang_code(null, :mcl_id);
  hide_ios_products = f_get_const('hide_ios_products');
  /*
        Примечание: изначально рассматривался вариант, что в данную процедуру будут передаваться:
        1)число активных кредитов и
        2) число активных заявок (процедура MC_CLI_EXTMINIAPP_LIST),
        Но, потом, было принято решение, что процедура MC_CLI_EXTMINIAPP_LIST будет возращать заявки, по которым был в итоге отказ. Т.е. не только активные.
        Поэтому, EXTMINIAPP_QTY я перезатираю внутри процедуры и вычисляю сам.
        Т.к. мне нужно исключать заявки, с негативным статусом, возможно логика поменяется и их так же нужно будет учитывать.
        Что касается LOAN_QTY, его я не перезатираю, т.к. в него попадает число активный кредитов.
   */

  mcl_id = coalesce(mcl_id, 0);

  if (mcl_id <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  if ((loan_qty is null) or (loan_qty < 0)) then
    exception e_default 'parametr "loan_qty" has invalid value';

  if ((extminiapp_qty is null) or (extminiapp_qty < 0)) then
    exception e_default 'parametr "extminiapp_qty" has invalid value';

  extminiapp_qty = 0;

  select
    mcl.mcl_from_c_id,
    mcl.mcl_user_code,
    mcl.mcl_os,
    f_enum('mcl_project', mcl.mcl_project, 'code', 0)
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into
    :c_id,
    :user_code,
    :os,
    :mcl_project;

  select c.c_verify_status
  from clients c
  where c.c_id = :c_id
  into c_verify_status;

  if (db_id in (39801, 41701) and
      c_id is not null) then
  begin
    eds_exist = f_asp_signed(:c_id, current_date);
  end

  select sum(iif(point_3_state = 0, 1, 0)), count(point_3_state),
         max(iif(point_3_state = -1, cast(point_3_date as date), null))
  from mc_cli_extminiapp_list(:mcl_id)
  into :extminiapp_qty_process, :extminiapp_qty_all, :max_reject_date;

  --------------------------------------------->
  ui_part = 'mc_ui_parts.loan_calculate';
  ui_part_description = f_lang(lang_code, 'Рассчитать заем');
  message_txt = '';

  is_visible = 1;
  --if (db_id = 81001 or db_id in (39801, 41701)) then  -- деактивировали по запросу от Акинай и Жыпара
  --is_visible = 0;

  if (db_id = 81001) then
  begin
    if (max_reject_date is not null) then
    begin
      if ((dateadd(day, 30, max_reject_date)) > current_date) then -- для РФ через 30 дней.
      begin
        is_visible = 0;
        message_txt = f_lang(lang_code, 'С последнего отказа не прошло 30 дней. Подача новой онлайн-заявки через приложение возможна: ') || f_datetostr(dateadd(day, 30, max_reject_date)) || '. Чтобы подать заявку на заем обратитесь в офис Компании или к своему специалисту.';
      end
    end
    if (extminiapp_qty_process >= 1) then
    begin
      is_visible = 0;
      message_txt = f_lang(lang_code, 'Есть активная заявка, подача новой онлайн-заявки пока невозможна');
    end

    -- RFCRU-3766
    if (mcl_project = 'oivo') then
    begin
        is_visible = 0;
        message_txt = '';
    end

  end
  else
  begin
    is_visible = 1;
    if (extminiapp_qty_process >= 1) then
    begin
      is_visible = 0;
      message_txt = f_lang(lang_code, 'Есть активная заявка, подача новой онлайн-заявки пока невозможна');
    end

    if (max_reject_date is not null) then
    begin
      if ((dateadd(day, 7, max_reject_date)) > current_date) then
      begin
        is_visible = 0;
        message_txt = f_lang(lang_code, 'С последнего отказа не прошло 7 дней. Подача новой онлайн-заявки возможна: ') || f_datetostr(dateadd(day, 7, max_reject_date));
      end
    end
    if ((db_id = 41701)
        and (f_get_const('ces_active') = 0 or c_verify_status <> 2)
    ) then
    begin
      if ((select cv.cv_value
           from const_var cv
           where cv.cv_code = 'mc_facerecog') = 1) then
      begin
        if (c_id is not null and
            (:eds_exist = 0)) then
        begin
          message_txt = f_lang(lang_code, 'Вам необходимо подойти в ближайший офис. Адреса и режим работы можно посмотреть по кнопке "На карте". Подача новой онлайн-заявки пока невозможна.');
          is_visible = 0; -- для клиентов у которых нет активный ЭЦП или не авторизован
        end
      end
    end
  end

  if (db_id = 39801) then
  begin
     is_visible = 1;
  end

  suspend;
  ---------------------------------------------<
  --------------------------------------------->
  ui_part = 'mc_ui_parts.loan_list';

  ui_part_description = f_lang(lang_code, 'Текущий заем');
  message_txt = '';

  is_visible = 1;

  if (loan_qty = 0) then
    is_visible = 0;
  suspend;
  ---------------------------------------------<
  --------------------------------------------->
  ui_part = 'mc_ui_parts.hide_product_list';

  ui_part_description = f_lang(lang_code, 'Карусель продуктов');
  message_txt = '';

  is_visible = 1;

  if (db_id = 39801) then
    is_visible = 0;
  suspend;
  ---------------------------------------------<
  --------------------------------------------->
  ui_part = 'mc_ui_parts.extminiapp_list';
  ui_part_description = f_lang(lang_code, 'Ваша заявка');
  message_txt = '';

  is_visible = 1;

  if (extminiapp_qty_all = 0) then
    is_visible = 0;

  suspend;
  ---------------------------------------------<
  -------------------------------------------->

  ui_part = 'mc_ui_parts.client_rating';
  ui_part_description = 'Надежный клиент';
  message_txt = '';

  if (db_id in (94401, 41701)) then --kg az
    is_visible = 1;

  -- RFCRU-3766
  if (db_id = 81001) then --rf
  begin
    if (mcl_project = 'kmp') then
      is_visible = 1;

    if (mcl_project = 'oivo') then
      is_visible = 0;

  end

  if (db_id = 39801) then --kz
    is_visible = 0;

  if (hide_ios_products = 1 and os = 2) then
    is_visible = 0;

  suspend;
  ---------------------------------------------<
  --------------------------------------------->

  ui_part = 'mc_ui_parts.legal_advice';
  ui_part_description = 'Юридические консультации';
  message_txt = '';
  is_visible = 0;

  -- RFCRU-3766
  if (db_id = 81001) then --rf
  begin
    if (mcl_project = 'kmp') then
      is_visible = 1;

    if (mcl_project = 'oivo') then
      is_visible = 0;

  end

  suspend;
  ---------------------------------------------<
  --------------------------------------------->
  if (db_id in (39801, 41701)) then
  begin
    ui_part = 'mc_ui_parts.my_point';
    ui_part_description = 'Мои баллы';
    message_txt = '';
    is_visible = 0;

    suspend;
  end
  ---------------------------------------------<
  -------------------------------------------->
  is_visible = 0;
  ui_part = 'mc_ui_parts.installment_goods';
  ui_part_description = 'Товары в рассрочку';
  message_txt = '';


  if ((db_id = 41701) and
     (f_get_const('installment_goods_is_active') = 1)) then
  begin
    if (f_get_const('installment_goods_add_asp') = 1) then
        is_visible = 1;
    else if (eds_exist >= 1) then
        is_visible = 1;
  end

  -- RFCKG-3447
  if ((f_get_const('installment_goods_add_asp') = 0) and
     (F_ASP_SIGNED(c_id, current_date) = 0)) then
  begin
    is_visible = 0;
  end

  suspend;
  ---------------------------------------------<
  --------------------------------------------->
  is_visible = 1;

  ui_part = 'mc_ui_parts.total_points'; --вкладка "Всего баллов" недоступны если вход как агент
  ui_part_description = 'Всего баллов';
  message_txt = '';

  if (:user_code = 1) then --kg, агент
    is_visible = 0;

  -- RFCRU-3766
  if (db_id = 81001) then --rf
  begin
    if (mcl_project = 'kmp') then
      is_visible = 1;

    if (mcl_project = 'oivo') then
      is_visible = 0;

    if (hide_ios_products = 1 and os = 2) then
      is_visible = 0;
  end

  suspend;
  ---------------------------------------------<
  --------------------------------------------->
 is_visible = 0;

  ui_part = 'mc_ui_parts.edit_personal_phone';
  ui_part_description = 'Редактирвоание телефонного номера';
  message_txt = '';

  if (c_id is not null) then
    is_visible = 1;

  if (db_id = 81001) then
  begin
    if (c_id is not null and is_visible = 1) then
      is_visible = (select phone_change_allowed from hp_phone_change_is_allowed(:mcl_id));

    if (is_visible = 1
        and exists
        (
            select *
            from hr_pstatus
            where ps_from_c_id = :c_id
                and ps_date_begin <= current_date
                and ps_date_end >= current_date
                and ps_is_working = 1
        )
    ) then
      is_visible = 0;

  end

  suspend;
  ---------------------------------------------<
  --------------------------------------------->
  ui_part = 'mc_ui_parts.partner_program_show';
  ui_part_description = 'Отображение элементов по партнерской программе РФ';
  message_txt = '';
  is_visible = 0;

  if (db_id = 81001 and
      c_id is not null) then
  begin
    if (mcl_project = 'kmp' and
        not exists
        (
            select ps_from_jl_id
            from hr_pstatus
            where ps_from_c_id = :c_id and
                current_date between ps_date_begin and ps_date_end and
                ps_is_working = 1 and
                ps_status = 1
        )
    ) then
      is_visible = 1;

    if (mcl_project = 'oivo') then
      is_visible = 0;

  end

  if (hide_ios_products = 1 and
      os = 2) then
    is_visible = 0;
  --  if (c_id in (1010100347157)) then is_visible = 1; -- для теста Бекбоева Кулуя

  suspend;
  ---------------------------------------------<
  --------------------------------------------->
  ui_part = 'mc_ui_parts.partner_recommend';
  ui_part_description = 'Вместо получить заем отобразить рекомендовать РФ';
  message_txt = '';
  is_visible = 0;

  if (db_id = 81001) then
  begin
    -- Если у клиента имеется активный заем то вместо получить заем нужно отбразить рекомендовать
    if (mcl_project = 'kmp' and
        exists(select *
               from kontstate, knclient
               where ks_ol_is_delete = 0 and
                     ks_state = 1 and
                     current_date between ks_date_begin and ks_date_end and
                     ks_from_k_id = kc_from_k_id and
                     kc_amount_approved > 0 and
                     kc_from_c_id = :c_id)
    ) then
      is_visible = 1;

    if (mcl_project = 'oivo') then
      is_visible = 0;

    --    if (c_id in (1010100347157)) then is_visible = 1; -- для теста Бекбоева Кулуя
  end
  suspend;
  ---------------------------------------------<
  --------------------------------------------->

  ui_part = 'mc_ui_parts.loans_for_ref';
  ui_part_description = 'Отображение "История займов" в РФ';
  message_txt = '';
  is_visible = 0;

  if (db_id = 81001) then
    is_visible = 1;
  suspend;
  ---------------------------------------------<

  ui_part = 'mc_ui_parts.cards_tokenization';
  ui_part_description = 'Токенизация карт';
  message_txt = '';
  is_visible = 0;

  -- RFCKG-4370
  if (db_id = 39801) then
    is_visible = 1;

  suspend;
  ---------------------------------------------<

  ui_part = 'mc_ui_parts.show_loan_limit';
  ui_part_description = 'Показывать кредитный лимит';
  message_txt = '';
  is_visible = 0;

  if ((db_id = 41701) and
      exists(select *
             from knclient kc
             inner join kontract k on k.k_id = kc.kc_from_k_id
             inner join product pr on pr.pr_id = k.k_from_pr_id
             inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id and
                   ks.ks_ol_is_delete = 0 and
                   current_date between ks.ks_date_begin and ks.ks_date_end and
                   ks.ks_state in (1, 2)
             where kc.kc_from_c_id = :c_id and
                   pr.pr_code not containing '.installment_goods.' and
                   pr_code not containing '.sugarmill.' and
                   kc.kc_amount_approved > 0)) then
    is_visible = 1;

  suspend;
  ---------------------------------------------<

  ui_part = 'mc_ui_parts.online_loan';
  ui_part_description = 'Онлайн-займ в OiVO';
  message_txt = '';
  is_visible = 0;

  -- Пока отключаем эту часть, чтобы не делать каждый раз лишний запрос. Включу уже перед запуском проекта
--  if (db_id = 81001 and mcl_project = 'oivo') then
--    is_visible = (select is_visible from hp_oivo_online_loan_visibility(:mcl_id));

  suspend;
  ---------------------------------------------<

end
