CREATE PROCEDURE AAREP_KONTRACT_INFO(
    K_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    CL_NO INTEGER,
    KF_NAME VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    CLIENTS_QTY INTEGER,
    REG_NAME VARCHAR(100),
    REG_BRANCH VARCHAR(100),
    KONT_ZAD_OS DECIMAL(18,4),
    KONT_ZAD_PERC DECIMAL(18,4),
    KONT_FINE DECIMAL(18,4),
    KONT_SUM_TOTAL DECIMAL(18,4),
    CLIENT_FIO VARCHAR(255),
    C_PLASTCARD_NO VARCHAR(20),
    C_PASS_NO VARCHAR(30),
    CLIENT_REG_ADR VARCHAR(255),
    CLIENT_FACT_ADR VARCHAR(255),
    CLIENT_ROD_REG_ADR VARCHAR(255),
    CLIENT_ROD_FACT_ADR VARCHAR(255),
    BLOB_PHOTO BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    BLOB_PASS1 BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    BLOB_PASS2 BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    C_ID DECIMAL(18,0),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    DELIQ_DAYS INTEGER,
    LAST_PAY_DATE DATE)
AS
declare variable adl_type smallint;
declare variable adl_address_str varchar(255);
declare variable c_blob_photo_id decimal(18,0);
declare variable c_blob_pass1_id decimal(18,0);
declare variable c_blob_pass2_id decimal(18,0);
declare variable dd_begin date;
declare variable aarep_delinq_date date;
begin
--101010000732700701
  clients_qty = 0;

  select count(*)
  from knclient kncl
  where kncl.kc_from_k_id = :k_id and
        kncl.kc_amount_approved > 0
  into :clients_qty;

  if (clients_qty is null) then
    clients_qty = 0;
  if (clients_qty = 0) then
    exit;

-- получаю номер группы
  select kal_new_kf_name, reg1.reg_code || ' ' || reg1.reg_name, reg2.reg_name, k_kontract_no, k_amount_approved,
         k_disb_date
  from kontract, knaktlist, region reg1, region reg2
  where k_from_kf_id = kal_from_kf_id and
        knaktlist.is_delete = 0 and
        :rep_date between kal_date and kal_date_end and
        kal_new_reg_id = reg1.reg_unicode and
        reg1.parent = reg2.reg_unicode and
        k_id = :k_id
  into :kf_name, :reg_name, :reg_branch, k_kontract_no, :k_amount_approved, :k_disb_date;
  cl_no = 0;
-- баланс по контракту
  kont_sum_total = 0;
  select sum(zad_os), sum(zad_perc), sum(sh_calc_fine)
  from aarep_make_zad_list(:k_id, :rep_date, 1)
  where sh_date is not null
  into :kont_zad_os, :kont_zad_perc, :kont_fine;
  kont_sum_total = kont_zad_os + kont_zad_perc + kont_fine;

-- вытаскиваю число дней просрочки
  select oks_date_begin, oks_delinq_date
  from oksummary
  where oks_from_k_id = :k_id and
        oks_date_begin <= :rep_date and
        oks_date_end >= :rep_date and
        oks_close_state = 0
  into :dd_begin, :aarep_delinq_date;
  if (dd_begin is not null) then
  begin
-- контракт активный
    if (aarep_delinq_date < rep_date) then
    begin
      deliq_days = datediff(day, :aarep_delinq_date, :rep_date);
    end
  end

  if (deliq_days is null) then
    deliq_days = 0;

  last_pay_date = null;
-- Дата последнего погашения
  select ok.ok_ol_date
  from operkred ok
  where ok.ok_from_k_id = :k_id and
        ok.ok_ol_is_active = 1 and
        ok.ok_type in (4, 9) and
        ok_from_c_id = 0
  order by ok.ok_ol_date desc
  rows 1
  into :last_pay_date;

  for select c_id, c_name_fio, cl.c_pass_no,
             (select cont_value
              from hp_contdata_get_cell_main(0, cl.c_id)), c_blob_photo_id, c_blob_pass1_id, c_blob_pass2_id,
             kc_amount_approved
      from knclient kncl, clients cl
      where kncl.kc_from_k_id = :k_id and
            kncl.kc_amount_approved > 0 and
            cl.c_id = kncl.kc_from_c_id
      into :c_id, :client_fio, :c_pass_no, :c_plastcard_no, :c_blob_photo_id, :c_blob_pass1_id, :c_blob_pass2_id,
           :kc_amount_approved
  do
  begin
    cl_no = cl_no + 1;
    select bls_blob
    from aarep_blobstore_show(:c_blob_photo_id)
    into :blob_photo;

    select bls_blob
    from aarep_blobstore_show(:c_blob_pass1_id)
    into :blob_pass1;

    select bls_blob
    from aarep_blobstore_show(:c_blob_pass2_id)
    into :blob_pass2;

-- получаю адреса
    client_reg_adr = '';
    client_fact_adr = '';
    client_rod_reg_adr = '';
    client_rod_fact_adr = '';

    for select adc_address_type, adl.adl_address_str
        from addrclient adcl
        inner join addrlist adl on adcl.adc_from_adl_id = adl.adl_id
        where adcl.adc_from_c_id = :c_id and
              adcl.adc_address_type in (1, 2, 4, 5)
        into :adl_type, :adl_address_str
    do
    begin
      if (adl_type = 1) then
        client_reg_adr = adl_address_str;
      if (adl_type = 2) then
        client_fact_adr = adl_address_str;
      if (adl_type = 4) then
        client_rod_reg_adr = adl_address_str;
      if (adl_type = 5) then
        client_rod_fact_adr = adl_address_str;
    end

    suspend;
  end

end
