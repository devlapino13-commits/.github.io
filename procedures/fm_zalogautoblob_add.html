CREATE PROCEDURE FM_ZALOGAUTOBLOB_ADD(
    ZA_ID D_ID,
    BLS_FILENAME D_STR_255,
    BLS_BLOB D_BLOB_BINARY,
    BLS_KIND D_INTEGER,
    UN_ID D_ID)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable zal_id d_integer;
declare variable bls_id d_id;
begin
  err_text = '';

  select za_from_zal_id
  from zalogauto
  where za_id = :za_id
  into zal_id;

  if (zal_id is null) then
  begin
    err_text = f_format('zal_id not found for za_id:', za_id);
    suspend;
    exit;
  end

  execute procedure hp_insert_into_blobstore (:bls_filename, 'Фото автомобиля', :bls_blob, :un_id, :bls_kind, 1)
  returning_values :bls_id;

  insert into zalogautoblob(zab_from_zal_id, zab_from_bls_id)
  values(:zal_id, :bls_id);

  suspend;
end
