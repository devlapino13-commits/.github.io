CREATE PROCEDURE HP_REBUKE_SCORECFG_UPDATE(
    RSC_ID D_INTEGER,
    RSC_FROM_RBK_ID D_INTEGER,
    RSC_REBUKE_TYPE D_INTEGER,
    RSC_SCORE_MIN D_INTEGER,
    RSC_SCORE_MAX D_INTEGER,
    RES_CHECKED D_BOOL)
AS
begin
  rsc_score_min = coalesce(rsc_score_min, 0);
  rsc_score_max = coalesce(rsc_score_max, 0);

  if (rsc_id is not null) then
  begin
    if (res_checked = 1) then
    begin
      update rebuke_scorecfg
      set rsc_from_rbk_id = :rsc_from_rbk_id,
          rsc_rebuke_type = :rsc_rebuke_type,
          rsc_score_min = :rsc_score_min,
          rsc_score_max = :rsc_score_max
      where (rsc_id = :rsc_id);
    end else
    begin
      delete from rebuke_scorecfg
      where rsc_id = :rsc_id;
    end
  end else
  begin
    if (res_checked = 1) then
    begin
      update or insert into rebuke_scorecfg (rsc_id, rsc_from_rbk_id, rsc_rebuke_type, rsc_score_min, rsc_score_max)
      values (:rsc_id, :rsc_from_rbk_id, :rsc_rebuke_type, :rsc_score_min, :rsc_score_max)
      matching (rsc_from_rbk_id, rsc_rebuke_type);
    end
  end
end
