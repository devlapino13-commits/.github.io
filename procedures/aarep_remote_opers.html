CREATE PROCEDURE AAREP_REMOTE_OPERS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OL_ID VARCHAR(100),
    REG_NAME D_STR_100,
    OL_DATE D_DATE,
    RD_NAME D_STR_255,
    PAYED_OS D_CURRENCY,
    PAYED_PERC D_CURRENCY,
    PAYED_FINE D_CURRENCY,
    PAYED_KOMMIS D_CURRENCY,
    PAYED_BALANCE D_CURRENCY,
    UN_FIO D_STR_255,
    K_KONTRACT_NO D_STR_100,
    PR_LOAN_TYPE_STR VARCHAR(20))
AS
declare variable k_id d_id;
declare variable reg_id d_id;
declare variable rd_id d_id;
declare variable un_id d_id;
declare variable pr_loan_type integer;
begin
 payed_os = 0;
 payed_perc = 0;
 payed_fine = 0;
 payed_kommis = 0;
 
  for select ol.ol_id,
             ol.ol_from_reg_id,
             ol.ol_date,
             ol.ol_user_id,
             ol.ol_from_rd_id,
             ok.ok_from_k_id,
             sum(-ok.ok_os),
             sum(-ok.ok_perc),
             sum(-ok.ok_perc),
             sum(-ok.ok_kommis)
      from operlist ol
      inner join operkred ok on ok.ok_from_ol_id = ol.ol_id
      inner join region_list(:rep_region_id, 1, 3, 1) rl on rl.id_list = ol.ol_from_reg_id
      where ol.is_delete = 1 and
            ok.ok_ol_is_active = 0 and
            ok.ok_type = 4 and
            ol.ol_date between :rep_date_begin and :rep_date_end
      group by ol_id, ol_from_reg_id, ol_date, ol_user_id, ol_from_rd_id,ok_from_k_id
      into ol_id,
           reg_id,
           ol_date,
           un_id,
           rd_id,
           k_id,
           payed_os,
           payed_perc,
           payed_fine,
           payed_kommis
  do
  begin
    payed_balance = 0;
    payed_balance = :payed_os + :payed_perc + :payed_fine + :payed_kommis;

    k_kontract_no = '';

    select k.k_kontract_no,
           pr.pr_loan_type
    from kontract k
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where k.k_id = :k_id and
          pr.pr_id > 0
    into k_kontract_no,
         pr_loan_type;

    pr_loan_type_str = '';
    if (pr_loan_type = 0) then
      pr_loan_type_str = 'Группа';
    else
    if (pr_loan_type = 1) then
      pr_loan_type_str = 'Индив.';

    select reg.reg_name
    from region reg
    where reg.reg_unicode = :reg_id
    into reg_name;
    
    rd_name='';
    
    select rd.rd_name
    from refdata rd
    where rd.rd_id = :rd_id
    into rd_name;
    
    un_fio='';
    
    select un.un_fio
    from username un
    where un.un_id = :un_id
    into un_fio;

    suspend;
  end
  for select ol.ol_id,
             ol.ol_from_reg_id,
             ol.ol_date,
             ol.ol_user_id,
             ol.ol_from_rd_id,
             oko_from_k_id,
             sum(-okf.oko_os),
             sum(-okf.oko_perc)
      from operlist ol
      inner join operkroff okf on okf.oko_from_ol_id = ol.ol_id
      inner join region_list(:rep_region_id, 1, 3, 1) rl on rl.id_list = ol.ol_from_reg_id
      where ol.is_delete = 1 and
            okf.oko_type = 4 and
            ol.ol_date between :rep_date_begin and :rep_date_end
      group by ol_id, ol_from_reg_id, ol_date, ol_user_id, ol_from_rd_id,oko_from_k_id
      into ol_id,
           reg_id,
           ol_date,
           un_id,
           rd_id,
           k_id,
           payed_os,
           payed_perc
  do
  begin
    payed_balance = 0;
    payed_balance = :payed_os + :payed_perc;

    k_kontract_no = '';

    select k.k_kontract_no,
           pr.pr_loan_type
    from kontract k
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where k.k_id = :k_id and
          pr.pr_id > 0
    into k_kontract_no,
         pr_loan_type;

    pr_loan_type_str = '';
    if (pr_loan_type = 0) then
      pr_loan_type_str = 'Группа';
    else
    if (pr_loan_type = 1) then
      pr_loan_type_str = 'Индив.';

    select reg.reg_name
    from region reg
    where reg.reg_unicode = :reg_id
    into reg_name;
    
    rd_name='';
    
    select rd.rd_name
    from refdata rd
    where rd.rd_id = :rd_id
    into rd_name;
    
    un_fio='';
    
    select un.un_fio
    from username un
    where un.un_id = :un_id
    into un_fio;

    suspend;
  end
end
