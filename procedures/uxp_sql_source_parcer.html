CREATE PROCEDURE UXP$SQL_SOURCE_PARCER(
    FROM_SQL D_STR_LONG)
RETURNS (
    SOURCE_CODE D_STR_CODE,
    SOURCE_PARAMS D_STR_LONG)
AS
declare variable bracket_count d_integer;
declare variable in_str d_bool;
declare variable sym d_str_sym;
declare variable len d_integer;
declare variable pos d_integer;
declare variable spec_symbols d_str_code;
declare variable source_pos_start d_integer;
declare variable source_pos_end d_integer;
declare variable source_param_start d_integer;
declare variable source_param_end d_integer;
begin
  while (position(ascii_char(13) || ascii_char(10), from_sql) > 0) do
    from_sql = replace(from_sql, ascii_char(13) || ascii_char(10), ' ');

  while (position(ascii_char(13), from_sql) > 0) do
    from_sql = replace(from_sql, ascii_char(13), ' ');

  while (position(ascii_char(10), from_sql) > 0) do
    from_sql = replace(from_sql, ascii_char(10), ' ');

  spec_symbols = ',:(';
  pos = 0;
  while (pos < char_length(spec_symbols)) do
  begin
    pos = pos + 1;
    sym = substring(spec_symbols from pos for 1);

    while (position(sym || ' ', from_sql) > 0) do
      from_sql = replace(from_sql, sym || ' ', sym);

    while (position(' ' || sym, from_sql) > 0) do
      from_sql = replace(from_sql, ' ' || sym, sym);
  end

  while (position('  ', from_sql) > 0) do
    from_sql = replace(from_sql, '  ', '');

  from_sql = trim(from_sql);

  len = char_length(from_sql);
  pos = 0;
  bracket_count = 0;
  source_pos_start = 1;
  source_param_start = 0;
  source_params = '';
  while (pos < len) do
  begin
    pos = pos + 1;
    source_pos_end = 0;

    sym = substring(from_sql from pos for 1);

    if (sym = '''') then
      in_str = 1 - in_str;

    if (in_str = 1) then
      continue;

    if (sym = '''') then
      continue;

    if (sym = '(') then
    begin
      if (bracket_count = 0) then
      begin
        source_param_start = pos;
        source_code = substring(from_sql from source_pos_start for (pos - source_pos_start));
      end

      bracket_count = bracket_count + 1;
      continue;
    end else
    if (sym = ')') then
    begin
      bracket_count = bracket_count - 1;

      if (bracket_count = 0) then
      begin
        source_param_end = pos;
        source_pos_end = pos;
        source_params = substring(from_sql from source_param_start + 1 for (source_param_end - source_param_start - 1));
        suspend;

        source_code = '';
        source_params = '';
        source_pos_start = source_pos_end + 1;
      end
      continue;
    end

    if (bracket_count = 0 and (sym in (' ', ',') or pos = len)) then
    begin
      source_param_end = pos;
      source_pos_end = pos;

      if (pos < len) then
        source_pos_end = source_pos_end - 1;

      source_code = substring(from_sql from source_pos_start for (source_pos_end - source_pos_start + 1));

      if (source_code <> '') then
        suspend;

      source_code = '';
      source_params = '';
      source_pos_start = source_pos_end + 1;

      if (sym = ',') then
        source_pos_start = source_pos_start + 1;
    end
  end
end
