CREATE PROCEDURE CM_KG_0117_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable c_id id_type;
declare variable pr_id d_integer;
declare variable pr_code d_str_100;
declare variable date_oper d_date;
declare variable pr_id_tmp d_integer;
declare variable is_ok d_smallint;
declare variable pr_code_tmp d_str_100;
declare variable pr_name d_str_100;
begin
  res_error_is_critical = 2;
  select p.pr_id, p.pr_code, m.ma_date_oper, p.pr_name
  from miniapp m, product p
  where ma_id = :ma_id and
        ma_is_active = 1 and
        ma_date_end = '31.12.9999' and
        m.ma_refund = 1 and
        p.pr_id = m.ma_from_pr_id
  into pr_id, pr_code, date_oper, pr_name;
  if (pr_code containing '.IPF.' and
      pr_code not containing '.KARDHASAN.') then -- если не Карт-Хасан
  begin
    res_error_text = 'Рефинансирование ИПФ-займов возможно только по Карт-Хасан';
    suspend;
    exit;
  end
  if (pr_code is not null) then
    for select mac_from_c_id
        from miniappclient mcl
        where mac_from_ma_id = :ma_id and
              mac_amount > 0
        into :c_id
    do
    begin
      select c_name_fio
      from clients
      where c_id = :c_id
      into :res_info_text;
      pr_id_tmp = null;
      is_ok = 0;
      for select k.k_from_pr_id, pr.pr_code
          from kontstate ks
          inner join knclient knc on knc.kc_from_k_id = ks.ks_from_k_id
          inner join kontract k on k.k_id = knc.kc_from_k_id
          inner join product pr on pr.pr_id = k.k_from_pr_id
          where knc.kc_from_c_id = :c_id and
                knc.kc_amount_approved > 0 and
                ks.ks_ol_is_delete = 0 and -- говорит о том что операция не удалена
                ks.ks_state = 1 and -- говорит о том, что кредит активный
                :date_oper between ks.ks_date_begin and ks.ks_date_end and
                :is_ok = 0
          into pr_id_tmp, pr_code_tmp
      do
      begin
        if (coalesce(pr_id_tmp, '') = '') then
        begin
          res_error_text = 'Не найден активный заем для рефинансирования';
          suspend;
        end
        /* MRAKHMANOV
    Продукт  список ID
    Бирге Стандарт   214, 215, 335
    Бирге Экспресс  235, 236, 340
    Бирге Льготный  171, 176, 341
    Бирге Мугалим  218, 219, 339
    Бирге Мугалим Льготный  232, 233, 342
    */
        if (pr_code containing '.BIRGE.') then
        begin
          if (pr_code containing '.STANDART.' and --            pr_id_tmp in (214, 215, 335)
              pr_code_tmp not containing '.IPF.') then
          begin
            is_ok = 1;
          end
          else
          if (pr_code containing '.express.' and
              pr_id_tmp in (235, 236, 340)) then
          begin
            is_ok = 1;
          end
          else
          if (pr_code containing '.MUGALIM.') then
          begin
            if (pr_code containing '.par.' and
                pr_id_tmp in (232, 233, 342)) then
            begin
              is_ok = 1;
            end
            else
            if (pr_id_tmp in (218, 219, 339)) then
            begin
              is_ok = 1;
            end
          end
          else
          if (pr_code containing '.PAR.' and
              pr_id_tmp in (171, 176, 341, 272)) then     --добавил 272 письмо от Анары Кошматов
          begin
            is_ok = 1;
          end
        end
        else
        if (pr_code containing '.GR.') then
        begin
          /*
          MRAKHMANOV
        Групповой   208, 209, 334
        Групповой Льготный  153, 156, 337
        Групповой Мугалим   212, 213, 336
        Групповой Мугалим Льготный  230, 231, 338
        Фермер  252, 253, 296
            
        Кард Хасан  237, 269
        
        select * from product
        where pr_code containing '.farmer.'      */
          if (pr_code containing '.MUGALIM.') then
          begin
            if (pr_code containing '.par.' and
                pr_id_tmp in (230, 231, 338)) then
            begin
              is_ok = 1;
            end
            else
            if (pr_id_tmp in (212, 213, 261, 336)) then
            begin
              is_ok = 1;
            end
          end
          else
          if (pr_code containing '.PAR.' and
              pr_id_tmp in (153, 156, 337)) then
          begin
            is_ok = 1;
          end
          else
          if (pr_code not containing '.PAR.' and
              pr_id_tmp in (208, 209, 334, 260)) then  --260 добавлен отдельно, в письме от Алмаза не было
          begin
            is_ok = 1;
          end
        end
        else
        if (pr_code containing '.IPF.' and
            pr_code containing '.KARDHASAN.' and
            pr_id_tmp not in (237, 269)) then
        begin
          is_ok = 1;
        end
        else
        if (pr_code containing '.farmer.' and
            pr_id_tmp in (252, 253, 296)) then
        begin
          is_ok = 1;
        end
      end
      if (is_ok = 0) then
      begin
        res_error_text = 'Не найден активный заем по продукту: ' || pr_name || --
        iif(pr_code containing '.KARDHASAN.', '. Рефинансирование ИПФ-займов возможно только по Карт-Хасан', --
        '. Рефинансирование классического займа возможно либо в рамках одного продукта либо через Бирге Стандарт');
        suspend;
      end
    end
end
