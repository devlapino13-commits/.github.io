CREATE PROCEDURE AAREP_CALC_PERC_ACCLIST(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    AAREP_OFFICE_ID INTEGER,
    AAREP_ACCOUNT VARCHAR(20),
    AAREP_ACCOUNT_TYPE INTEGER)
AS
declare variable aarep_account_profit varchar(20);
declare variable aarep_account_calced varchar(20);
declare variable aarep_account_repayed varchar(20);
declare variable aarep_account_portfolio varchar(20);
declare variable db_id d_integer;
begin
  /* Procedure Text */
  db_id = f_get_db_id();

  for select distinct aarep_office_id, aarep_account_profit, aarep_account_calced, aarep_account_repayed,
                      aarep_account_portfolio
      from aarep_calc_perc_operbook(:report_region_id, :report_date_begin, :report_date_end, 0)
      into :aarep_office_id, :aarep_account_profit, :aarep_account_calced, :aarep_account_repayed,
           :aarep_account_portfolio
  do
  begin
    aarep_account_type = 1;
    aarep_account = aarep_account_portfolio;
    if ((aarep_account is not null) and
        (aarep_account <> '')) then
      suspend;
    aarep_account_type = 2;
    aarep_account = aarep_account_calced;
    if ((aarep_account is not null) and
        (aarep_account <> '')) then
      suspend;

    if (db_id <> 39801) then
    begin
      aarep_account_type = 3;
      aarep_account = aarep_account_repayed;
      if ((aarep_account is not null) and
          (aarep_account <> '')) then
        suspend;
    end

    aarep_account_type = 4;
    aarep_account = aarep_account_profit;
    if ((aarep_account is not null) and
        (aarep_account <> '')) then
      suspend;
  end
end
