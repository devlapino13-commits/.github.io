    UX_UUID D_UUID,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    UA_UUID D_UUID,
    UA_OWNER D_UUID,
    UA_UXA_UUID D_UUID,
    UA_USR_ID_FIELD D_STR_CODE,
    UA_KIND D_SMALLINT,
    UA_NAME D_STR_100,
    UA_HINT D_STR_255,
    UA_DISABLE_ON_EDIT D_BOOL,
    UA_READONLY D_BOOL,
    UA_SAFE_MODE D_BOOL,
    UA_INACTIVE D_BOOL,
    RES_UX_KIND D_INTEGER,
    RES_IS_OWN D_BOOL,
    RES_CAPTION D_STR_255,
    RES_MENU_COUNT D_INTEGER,
    RES_OWNER_CODE D_STR_100,
    RES_UXA_CODE D_STR_CODE,
    UA_UUID_STR D_STR_CODE,
    RES_OWNER_CAPTION D_STR_255,
    UA_IMG_UUID D_UUID,
    UA_IMG_UUID_CODE D_STR_CODE,
    RES_OWNER_KIND D_STR_CODE,
    UA_EXT_COMPONENT D_STR_CODE,
    UA_EXT_DATASET D_STR_CODE,
    UA_EXT_REPLACE_KIND D_SMALLINT,
    UA_HOTKEY D_STR_10,
    UA_ADD_DELIMITER D_SMALLINT,
    RES_HAS_EXPRESSION D_BOOL,
    RES_EXPRESSION D_STR_LONG,
    UA_IS_GROUP D_BOOL,
    UA_ALIAS_GROUP D_STR_8100,
    UA_AUTO_REFRESH_SECONDS D_INTEGER,
    UA_TRIGGER_KIND D_INTEGER,
    UA_PREBLANK_KIND D_INTEGER,
    UA_PREBLANK_CODE D_STR_CODE,
    UA_PREBLANK_SAVE D_SMALLINT)
AS
declare variable tmp_name d_str_100;
declare variable mm_uuid d_uuid;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  mm_uuid = uxf$mainmenu();

  -- если передан ua_code, возващаем одну запись
  if (uxf$code(ux_uuid, 'kind') = 'uxaction') then
  begin
    res_is_own = 1;
    for select ua_uuid,
               ua_owner,
               ua_uxa_uuid,
               ua_uxa_fcode,
               ua_usr_id_field,
               ua_name,
               ua_hint,
               ua_disable_on_edit,
               ua_readonly,
               ua_safe_mode,
               ua_inactive,
               ua_img_uuid,
               uuid_to_char(ua_img_uuid),
               ua_ext_component,
               ua_ext_dataset,
               ua_ext_replace_kind,
               ua_hotkey,
               ua_add_delimiter,
               ua_auto_refresh_seconds,
               ua_is_group,
               ua_alias_group,
               ua_trigger_kind,
               ua_preblank_kind,
               ua_preblank_code,
               ua_preblank_save
        from ux$action
        where ua_uuid = :ux_uuid
        into :ua_uuid,
             :ua_owner,
             :ua_uxa_uuid,
             :res_uxa_code,
             :ua_usr_id_field,
             :ua_name,
             :ua_hint,
             :ua_disable_on_edit,
             :ua_readonly,
             :ua_safe_mode,
             :ua_inactive,
             :ua_img_uuid,
             :ua_img_uuid_code,
             :ua_ext_component,
             :ua_ext_dataset,
             :ua_ext_replace_kind,
             :ua_hotkey,
             :ua_add_delimiter,
             :ua_auto_refresh_seconds,
             :ua_is_group,
             :ua_alias_group,
             :ua_trigger_kind,
             :ua_preblank_kind,
             :ua_preblank_code,
             :ua_preblank_save
    do
    begin
      -- если Action добавлен в меню или кнопке, указываем этот объект
      res_menu_count = 0;
      res_owner_code = uxf$code(ua_owner, 'fcode');
      res_owner_kind = uxf$code(ua_owner, 'kind');

      if (ua_uxa_uuid is not null) then
      begin
        select uxa_name, uxa_kind, uu_kind
        from ux$alias, ux$code, ux$uniq
        where uxa_uuid = :ua_uxa_uuid and
              uxa_owner = ux_uuid and
              ux_uuid = uu_id
        into tmp_name, ua_kind, res_ux_kind;

        res_caption = tmp_name;
        if (tmp_name not containing res_uxa_code) then
          res_caption = res_caption || '   - ' || res_uxa_code;
      end else
      begin
        res_caption = res_uxa_code;
        ua_kind = 0;
        res_ux_kind = null;
      end

      select count(*) from ux$aliasaction
      where uaa_ua_uuid = :ua_uuid
      into res_menu_count;
  
      tmp_name = uxf$code(ua_owner, 'name');
      res_owner_caption = tmp_name;
      if (tmp_name not containing res_owner_code) then
        res_owner_caption = res_owner_caption || '   - ' || res_owner_code;

      res_has_expression = 0;
      res_expression = '';
  
      select exp_expression
      from ux$expression
      where exp_ux_uuid = :ua_uuid and
            exp_kind = 2
      into res_expression;
  
      if (coalesce(res_expression, '') <> '') then
        res_has_expression = 1;

      ua_uuid_str = lower(uuid_to_char(ua_uuid));
      suspend;
    end
  end else
  begin
    res_is_own = 1;
    -- Список действий объекта
    for select ua_uuid,
               ua_owner,
               ua_uxa_uuid,
               ua_uxa_fcode,
               ua_usr_id_field,
               ua_name,
               ua_hint,
               ua_disable_on_edit,
               ua_readonly,
               ua_safe_mode,
               ua_inactive,
               ua_img_uuid,
               uuid_to_char(ua_img_uuid),
               ua_ext_component,
               ua_ext_dataset,
               ua_ext_replace_kind,
               ua_hotkey,
               ua_add_delimiter,
               ua_auto_refresh_seconds,
               ua_is_group,
               ua_alias_group,
               ua_trigger_kind,
               ua_preblank_kind,
               ua_preblank_code,
               ua_preblank_save
        from ux$action
        where ua_owner = :ux_uuid
        into :ua_uuid,
             :ua_owner,
             :ua_uxa_uuid,
             :res_uxa_code,
             :ua_usr_id_field,
             :ua_name,
             :ua_hint,
             :ua_disable_on_edit,
             :ua_readonly,
             :ua_safe_mode,
             :ua_inactive,
             :ua_img_uuid, 
             :ua_img_uuid_code,
             :ua_ext_component,
             :ua_ext_dataset,
             :ua_ext_replace_kind,
             :ua_hotkey,
             :ua_add_delimiter,
             :ua_auto_refresh_seconds,
             :ua_is_group,
             :ua_alias_group,
             :ua_trigger_kind,
             :ua_preblank_kind,
             :ua_preblank_code,
             :ua_preblank_save

    do
    begin
      -- если Action добавлен в меню или кнопке, указываем этот объект
      res_menu_count = 0;
      res_owner_code = uxf$code(ua_owner, 'fcode');
      res_owner_kind = uxf$code(ua_owner, 'kind');

      tmp_name = uxf$code(ua_uxa_uuid, 'name');
  
      select count(*) from ux$aliasaction
      where uaa_ua_uuid = :ua_uuid
      into res_menu_count;
  
      if (ua_uxa_uuid is not null) then
      begin
        select uxa_name, uxa_kind, uu_kind
        from ux$alias, ux$code, ux$uniq
        where uxa_uuid = :ua_uxa_uuid and
              uxa_owner = ux_uuid and
              ux_uuid = uu_id
        into tmp_name, ua_kind, res_ux_kind;

        res_caption = tmp_name;
        if (tmp_name not containing res_uxa_code) then
          res_caption = res_caption || '   - ' || res_uxa_code;
      end else
      begin
        res_caption = res_uxa_code;
        ua_kind = 0;
        res_ux_kind = null;
      end

      tmp_name = uxf$code(ua_owner, 'name');
      res_owner_caption = tmp_name;
      if (tmp_name not containing res_owner_code) then
        res_owner_caption = res_owner_caption || '   - ' || res_owner_code;

      res_has_expression = 0;
      res_expression = '';
  
      select exp_expression
      from ux$expression
      where exp_ux_uuid = :ua_uuid and
            exp_kind = 2
      into res_expression;
  
      if (coalesce(res_expression, '') <> '') then
        res_has_expression = 1;

      ua_uuid_str = lower(uuid_to_char(ua_uuid));
      suspend;
    end
  
    -- Список действий где вызывается данный объект
    res_is_own = 0;
    for select ua_uuid,
               ua_owner,
               ua_uxa_uuid,
               ua_uxa_fcode,
               ua_usr_id_field,
               uxa_kind,
               ua_name,
               ua_hint,
               ua_disable_on_edit,
               ua_readonly,
               ua_safe_mode,
               ua_inactive,
               ua_img_uuid,
               uuid_to_char(ua_img_uuid),
               uu_kind,
               ux_name,
               ua_ext_component,
               ua_ext_dataset,
               ua_ext_replace_kind,
               ua_hotkey,
               ua_add_delimiter,
               ua_auto_refresh_seconds,
               ua_is_group,
               ua_alias_group,
               ua_trigger_kind,
               ua_preblank_kind,
               ua_preblank_code,
               ua_preblank_save
        from ux$action, ux$alias, ux$form, ux$uniq, ux$code
        where ufm_uuid = :ux_uuid and
              ufm_ux_uuid = uxa_owner and
              ua_uxa_uuid = uxa_uuid and
              ua_owner <> :ux_uuid and
              ufm_uuid = uu_id and
              ux_uuid = ufm_ux_uuid
        into :ua_uuid,
             :ua_owner,
             :ua_uxa_uuid,
             :res_uxa_code,
             :ua_usr_id_field,
             :ua_kind,
             :ua_name,
             :ua_hint,
             :ua_disable_on_edit,
             :ua_readonly,
             :ua_safe_mode,
             :ua_inactive,
             :ua_img_uuid,
             :ua_img_uuid_code,
             :res_ux_kind,
             :tmp_name,
             :ua_ext_component,
             :ua_ext_dataset,
             :ua_ext_replace_kind,
             :ua_hotkey,
             :ua_add_delimiter,
             :ua_auto_refresh_seconds,
             :ua_is_group,
             :ua_alias_group,
             :ua_trigger_kind,
             :ua_preblank_kind,
             :ua_preblank_code,
             :ua_preblank_save
    do
    begin
      -- если Action добавлен в меню или кнопке, указываем этот объект
      res_menu_count = 0;
      res_owner_code = uxf$code(ua_owner, 'fcode');
      res_owner_kind = uxf$code(ua_owner, 'kind');

      tmp_name = uxf$code(ua_uxa_uuid, 'name');
      res_caption = tmp_name;
      if (tmp_name not containing res_uxa_code) then
        res_caption = res_caption || '   - ' || res_uxa_code;

      tmp_name = uxf$code(ua_owner, 'name');
      res_owner_caption = tmp_name;
      if (tmp_name not containing res_owner_code) then
        res_owner_caption = res_owner_caption || '   - ' || res_owner_code;

      if (ua_owner <> mm_uuid) then
      begin
        select count(*) from ux$aliasaction
        where uaa_ua_uuid = :ua_uuid
        into res_menu_count;
      end else
      begin
        res_owner_caption = 'Главное меню';

        if (exists(select * from uxp$actionmenu(:ua_uuid))) then
          res_menu_count = 1;
      end
  
      res_has_expression = 0;
      res_expression = '';
  
      select exp_expression
      from ux$expression
      where exp_ux_uuid = :ua_uuid and
            exp_kind = 2
      into res_expression;
  
      if (coalesce(res_expression, '') <> '') then
        res_has_expression = 1;

      ua_uuid_str = lower(uuid_to_char(ua_uuid));
      suspend;
    end
  end
