CREATE PROCEDURE FM_HTML_LO_BALANCE_CHANGE(
    UN_ID DECIMAL(18,0),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable color_black varchar(100);
declare variable color_yellow varchar(100);
declare variable color_green varchar(100);
declare variable color_red varchar(100);
declare variable lo_id decimal(18,0);
declare variable s_text varchar(100);
declare variable line_str varchar(255);
declare variable cm_database_id integer;
declare variable plan_date date;
declare variable sum_plan decimal(18,4) = 0;
declare variable sum_last decimal(18,4) = 0;
declare variable sum_advance decimal(18,4) = 0;
declare variable sum_disb decimal(18,4) = 0;
declare variable sum_writeoff decimal(18,4) = 0;
declare variable line_amount decimal(18,4) = 0;
declare variable line_amount_end decimal(18,4) = 0;
declare variable space_str varchar(20);
declare variable dd_str varchar(20);
declare variable count_disb decimal(18,4) = 0;
declare variable count_close decimal(18,4) = 0;
declare variable count_close_adv decimal(18,4) = 0;
declare variable count_tmp decimal(18,4) = 0;
declare variable count_client_end decimal(18,4) = 0;
begin
  res_text = (select res_style
              from fm_html_styles) || '<body>';
  color_black = 'black';
  color_yellow = '#F08205';
  color_green = '#02B316';
  color_red = 'red';

  space_str = '&nbsp;' || '&nbsp;' || '&nbsp;';

  cm_database_id = f_get_db_id();

  color_black = 'black';
  color_yellow = '#F08205';
  color_green = '#02B316';
  color_red = 'red';

  res_text = BLOB_APPEND(res_text, f_format('<h1 class="green_text">%1</h1>', iif(cm_database_id <> 94401, 'Планируемое изменение баланса', 'Balansda planlaşdırılmış dəyişiklər')));

  select un_from_c_id
  from username
  where un_id = :un_id
  into :lo_id;

  if (lo_id is null) then
  begin
    res_text = BLOB_APPEND(res_text, f_format('<font color="%1">%2</font><br>', color_red, iif(:cm_database_id <> 94401, 'Пользователь не определен', 'İstifadəçi təyin olunmadı')));
    suspend;
    exit;
  end
  line_str = iif(cm_database_id <> 94401, 'Дата', 'Tarix') || ': ' || iif(rep_date_end is not null, right('0' || extract(day from rep_date_end), 2) || '.' || right('0' || extract(month from rep_date_end), 2) || '.' || extract(year from rep_date_end), iif(cm_database_id <> 94401, 'Не указан', 'Qeyd olunmayıb'));
  line_str = '<font color="' || color_black || '">' || :line_str || '</font>';
  res_text = BLOB_APPEND(res_text, line_str || '<br>');

  s_text = '';
  select reg_code || ' ' || reg_name
  from region, hr_pstatus
  where reg_unicode = ps_from_reg_id and
        ps_from_c_id = :lo_id and
        ps_date_begin <= :rep_date_begin and
        ps_date_end >= :rep_date_begin and
        ps_is_working = 1 and
        ps_status = 1
  into :s_text;
  if (s_text is null) then
    s_text = '';
  if (s_text = '') then
  begin
    res_text = BLOB_APPEND(res_text, f_format('<font color="%1">%2</font><br>', color_red, iif(cm_database_id <> 94401, 'У вас нет действующего приказа.', 'Aktiv əmriniz mövcüd deyil')));
    suspend;
    exit;
  end

  line_str = iif(cm_database_id <> 94401, 'Офис', 'Filial') || ': ' || s_text;
  line_str = '<font color="' || color_black || '">' || :line_str || '</font>';
  res_text = BLOB_APPEND(res_text, line_str || '<br>');

  select c_name_fio
  from clients
  where c_id = :lo_id
  into :s_text;

  line_str = s_text;
  line_str = '<font color="' || color_black || '">' || :line_str || '</font>';
  res_text = BLOB_APPEND(res_text, line_str || '<br>');

  res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="70%">
       <col width="10%">
       <col width="20%">
       <tr>' || f_format('<th>%1</th><th>%2</th><th>%3</th></tr>', iif(cm_database_id <> 94401, 'Период', 'Müddət'),
                                                                   iif(cm_database_id <> 94401, 'Кредиты', 'Kreditlər'),
                                                                   iif(cm_database_id <> 94401, 'Осн.сумма', 'Əsas məbləğ')));

  dd_str = iif(rep_date_begin is not null, right('0' || extract(day from rep_date_begin), 2) || '.' || right('0' || extract(month from rep_date_begin), 2) || '.' || extract(year from rep_date_begin), iif(cm_database_id <> 94401, 'Не указан', 'Не указан'));

  line_str = f_format('<font color="%1"><b>%2 %3</b></font>', color_green,
                                                              iif(cm_database_id <> 94401, 'КП на утро', 'Sabaha KrPort'),
                                                              :dd_str);

  select sum(oks_balance), sum(k_kolich_client_approved)
  from oksummary, kontract
  where oks_kc_id = :lo_id and
        oks_date_begin <= (:rep_date_begin - 1) and
        oks_date_end >= (:rep_date_begin - 1) and
        oks_from_k_id = k_id and
        oks_close_state = 0
  into :line_amount, count_tmp;
  if (line_amount is null) then
    line_amount = 0;
  if (count_tmp is null) then
    count_tmp = 0;
  res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || round(count_tmp, 0) || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');
  line_amount_end = line_amount;
  count_client_end = count_tmp;

  for select plan_date, sum(sum_plan), sum(sum_last), sum(sum_advance), sum(sum_disb), sum(sum_writeoff),
             sum(count_disb), sum(count_close), sum(count_close_adv)
      from fm_loan_total_plan_tmp(:un_id, :rep_date_begin, :rep_date_end)
      group by plan_date
      into :plan_date, :sum_plan, :sum_last, :sum_advance, :sum_disb, :sum_writeoff, count_disb, count_close,
           count_close_adv
  do
  begin
    line_str = '';
    line_amount = 0.0;
    line_amount_end = line_amount_end + sum_plan + sum_last + sum_advance + sum_disb + sum_writeoff;
    count_client_end = count_client_end + count_disb + count_close + count_close_adv;
    if (plan_date < rep_date_begin) then
    begin
-- Фактические операции
      line_str = f_format('<b>%1</b>', iif(cm_database_id <> 94401, 'Фактические за сегодня', 'Bu günə faktiki'));
      line_amount = sum_plan + sum_last + sum_advance + sum_disb + sum_writeoff;
      res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || round(count_disb + count_close + count_close_adv, 0) || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');

      line_str = space_str || iif(cm_database_id <> 94401, 'Погашено по графику', 'İlkin cədvəl ilə ödəniş');
      line_amount = sum_plan;
      res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || round(count_close, 0) || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');

      line_str = space_str || iif(cm_database_id <> 94401, 'Погашены просрочки', 'Ödənilən gecikmələr');
      line_amount = sum_last;
      res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || 0 || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');

      line_str = space_str || iif(cm_database_id <> 94401, 'Погашены досрочно', 'Ödənilən gecikmələr');
      line_amount = sum_advance;
      res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || round(count_close_adv, 0) || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');

      line_str = space_str || iif(cm_database_id <> 94401, 'Списано', 'Silinib');
      line_amount = sum_writeoff;
      res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || 0 || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');

      line_str = space_str || iif(cm_database_id <> 94401, 'Выданы', 'Verilən');
      line_amount = sum_disb;
      res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || round(count_disb, 0) || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');
    end
    else
    begin
-- Планируемые операции
      dd_str = iif(plan_date is not null, right('0' || extract(day from plan_date), 2) || '.' || right('0' || extract(month from plan_date), 2) || '.' || extract(year from plan_date), iif(cm_database_id <> 94401, 'Не указан', 'Qeyd olunmayıb'));

      if (plan_date = rep_date_begin) then
        line_str = f_format('<b>%1</b>', iif(cm_database_id <> 94401, 'Ожидаемые за сегодня', 'Bu günə gözlənilən'));
      else
        line_str = f_format('<b>%1 %2</b></br>', iif(cm_database_id <> 94401, 'Ожидаемые за', 'Gözlənilən'), :dd_str);
      line_amount = sum_plan + sum_last + sum_advance + sum_disb + sum_writeoff;
      res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || round(count_disb + count_close + count_close_adv, 0) || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');

      line_str = space_str || iif(cm_database_id <> 94401, 'по графику', 'ödəniş cədvəli üzrə');
      line_amount = sum_plan;
      res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || 0 || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');

      line_str = space_str || iif(cm_database_id <> 94401, 'последние платежи', 'sonuncu ödənişlər');
      line_amount = sum_last;
      res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || round(count_close, 0) || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');

      line_str = space_str || iif(cm_database_id <> 94401, 'по заявкам на досрочно', 'Vaxtından əvvəl bağlanma ərizələri üzrə');
      line_amount = sum_advance;
      res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || round(count_close_adv, 0) || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');

      line_str = space_str || iif(cm_database_id <> 94401, 'выдачи по заявкам', 'Ərizələr üzrə verilənlər');
      line_amount = sum_disb;
      res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || round(count_disb, 0) || '</td>' || '<td align="right">' || round(line_amount, 0) || '</td>' || '</tr>');
    end
  end
  dd_str = iif(rep_date_end is not null, right('0' || extract(day from rep_date_end), 2) || '.' || right('0' || extract(month from rep_date_end), 2) || '.' || extract(year from rep_date_end), iif(cm_database_id <> 94401, 'Не указан', 'Не указан'));

  line_str = f_format('<b>%1 %2</b>', iif(cm_database_id <> 94401, 'Ожидаемое КП на', 'Gözlənilən Kr.P.'), :dd_str);
  line_str = '<font color="' || color_green || '">' || :line_str || '</font>';
  res_text = BLOB_APPEND(res_text, '<tr>' || '<td align="left">' || line_str || '</td>' || '<td align="right">' || round(count_client_end, 0) || '</td>' || '<td align="right">' || round(line_amount_end, 0) || '</td>' || '</tr>');

  res_text = BLOB_APPEND(res_text, '</table>');
  res_text = BLOB_APPEND(res_text, '</body>');

  suspend;
end
