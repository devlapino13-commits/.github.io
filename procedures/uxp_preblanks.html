CREATE PROCEDURE UXP$PREBLANKS(
    PREBLANK$ACTION_ID D_STR_CODE,
    PREBLANK$KIND D_STR_CODE,
    PREBLANK$CODE D_STR_CODE,
    IN_OBJ_KIND D_STR_CODE,
    IN_OBJ_ID D_STR_CODE)
RETURNS (
    PRB_ID D_INTEGER,
    PRB_KIND D_INTEGER,
    PRB_CODE D_STR_CODE,
    PRB_OBJECT_KIND D_INTEGER,
    PRB_OBJECT_ID D_ID,
    PRB_UA_UUID D_UUID,
    PRB_SELECTED D_BOOL,
    PRB_PREPARED D_BOOL,
    PRB_REQUIRED D_BOOL,
    PRB_BLANK_BLOB D_BLOB_BINARY,
    PRB_BLANK_BLN_ID D_INTEGER,
    PRB_BLANK_BLS_ID D_ID,
    PRB_SYNC_STATE D_SMALLINT,
    RES_UA_NAME D_STR_100,
    RES_UA_MEMO D_STR_255,
    RES_USED_PARAMS D_STR_LONG,
    EDL_ADD_PARAMS D_STR_255,
    PRB_SINGLE_SELECT D_BOOL,
    FRX$BLOB D_BLOB_BINARY)
AS
declare variable ua_uuid d_uuid;
declare variable kind_id d_integer;
declare variable is_required d_bool;
declare variable bls_id d_id;
declare variable is_prepared d_bool;
declare variable is_selected d_bool;
declare variable preblank_kind d_str_code;
declare variable edl_object_kind d_integer;
declare variable edl_object_id d_id;
declare variable obj_kind_id d_integer;
declare variable edl_id d_id;
declare variable oml_id d_integer;
begin
  preblank$code = trim(coalesce(preblank$code, ''));

  kind_id = f_enum('uxaction_preblank_kind', preblank$kind, 'id');
  if (kind_id is null) then
    kind_id = 0;

  preblank_kind = f_enum('uxaction_preblank_kind', preblank$kind, 'code');

  if (preblank$action_id is null) then
    preblank$action_id = '';

  obj_kind_id = f_objkind(in_obj_kind, 'id');

  delete from ux$preblanks
  where 1 = 1;

--  update ux$preblanks
--  set prb_sync_state = -1,
--      prb_prepared = 0,
--      prb_selected = 0,
--      prb_from_oml_id = null
--  where prb_kind = :kind_id and
--        prb_code = :preblank$code;

  edl_add_params = '';
  if (preblank_kind = 'eblank') then
  begin
    select edl_object_kind, edl_object_id
    from edoc_list
    where edl_id = :in_obj_id
    into edl_object_kind, edl_object_id;

    if (row_count = 0) then
      exit;

    edl_add_params = '{"edl_id":' || in_obj_id || '}';
  end

  if (f_objkind(in_obj_kind, 'code') = 'hr_orders') then
  begin
    select edl_id
    from edoc_list
    where edl_object_kind = :obj_kind_id and
          edl_object_id = :in_obj_id
    order by edl_id desc
    rows 1
    into edl_id;
  end

  select oml_id
  from uxp$preblanks_presave(:preblank$action_id, :preblank$kind, :preblank$code, :in_obj_kind, :in_obj_id)
  into :oml_id;

  for select res_ua_uuid, res_bln_id, res_required, res_bls_id, res_blank_name
  from uxp$preblank_actions(:preblank$kind, :preblank$code, :in_obj_kind, :in_obj_id)
  into :ua_uuid, prb_blank_bln_id, is_required, bls_id, res_ua_name
  do
  begin
    is_prepared = 0;
    if (bls_id is not null) then
      is_prepared = 1;

    is_selected = iif(is_required = 1 or is_prepared = 1, 1, 0);

    if (is_selected = 0 and edl_id is not null) then
    begin
      if (exists(select * from edoc_blank
                 where edb_from_edl_id = :edl_id and
                       edb_from_bln_id = :prb_blank_bln_id)) then
        is_selected = 1;
    end

    is_selected = 0;

    update or insert into ux$preblanks (prb_kind, prb_code, prb_object_kind, prb_object_id,
                                        prb_name, prb_ua_uuid, prb_blank_bln_id,
                                        prb_required, prb_prepared, prb_selected, prb_blank_blob, prb_blank_bls_id,
                                        prb_from_oml_id, prb_sync_state)
    values (:kind_id, :preblank$code, coalesce(:edl_object_kind, :obj_kind_id), coalesce(:edl_object_id, :in_obj_id),
            :res_ua_name, :ua_uuid, :prb_blank_bln_id,
            :is_required, :is_prepared, :is_selected, null, :bls_id, :oml_id, 0)
    matching (prb_kind, prb_code, prb_ua_uuid);
  end

  delete from ux$preblanks
  where prb_kind = :kind_id and
        prb_code = :preblank$code and
        prb_sync_state = -1;

  select ubc_single_select
  from ux$blankcode
  where ubc_code = :preblank$code
  into prb_single_select;

  if (prb_single_select is null) then
    prb_single_select = 0;

  for select prb_id,
             prb_kind,
             prb_code,
             prb_object_kind,
             prb_object_id,
             prb_ua_uuid,
             prb_name,
             prb_selected,
             prb_prepared,
             prb_required,
             prb_blank_blob,
             prb_blank_bln_id,
             prb_blank_bls_id,
             prb_sync_state
      from ux$preblanks
      where prb_kind = :kind_id and
            prb_code = :preblank$code
      into :prb_id,
           :prb_kind,
           :prb_code,
           :prb_object_kind,
           :prb_object_id,
           :prb_ua_uuid,
           :res_ua_name,
           :prb_selected,
           :prb_prepared,
           :prb_required,
           :prb_blank_blob,
           :prb_blank_bln_id,
           :prb_blank_bls_id,
           :prb_sync_state
  do
  begin
    res_ua_memo = '';
    res_used_params = '';

    if (prb_blank_bls_id is not null) then
    begin
      res_ua_memo = 'Документ из хранилища';
      res_used_params = uxf$blob(prb_blank_bls_id, 1, 1, 1);
    end
    else
    begin
      select list(distinct uf_field_name)
      from ux$actionparam, ux$frparam, ux$field
      where uap_ua_uuid = :prb_ua_uuid and
            uap_is_output = 0 and
            uap_owner_ufp_uuid = ufp_uuid and
            ufp_uf_uuid = uf_uuid
      into res_used_params;
  
      if (res_used_params is null) then
        res_used_params = '';
    end

    suspend;
  end
end
