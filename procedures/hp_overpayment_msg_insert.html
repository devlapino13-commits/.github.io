CREATE PROCEDURE HP_OVERPAYMENT_MSG_INSERT(
    K_ID D_ID)
AS
declare variable over_date d_date;
declare variable ovm_id d_integer;
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable kc_status d_smallint;
declare variable phone d_str_100;
begin
  over_date = (select max(ok_ol_date)
              from operkred
              where ok_from_k_id = :k_id
                and ok_ol_is_active = 1
                and ok_os < 0
                and ok_type = 9);

  -- определим казначея
  for select kc_id, kc_from_c_id, kc_status
  from knclient kc
  where kc_from_k_id = :k_id
    and kc.kc_amount_approved > 0
  order by kc_status desc, kc_number
  into kc_id, c_id, kc_status
  do
  begin
    if (kc_status = 2) then -- казначей
      break;

    -- если казначея нет (ушел через амнистию), то ищем первого у кого есть основной телефон
    phone = (select cci.ok_cell_main from continfo cci
            where cci.cci_object_id = :c_id
              and cci.cci_object_kind = 0);
    if (coalesce(phone, '') <> '') then
      break;
  end 

  ovm_id = (select ovm_id
            from overpayment_msg ovm
            where ovm_from_kc_id = :kc_id
              and ovm.ovm_from_msg_id is null  -- не отправлено
              and ovm.ovm_is_delete = 0);

  if (ovm_id is null) then
    insert into overpayment_msg (ovm_from_kc_id, ovm_over_date)
    values (:kc_id, :over_date);
  else
    update overpayment_msg
    set ovm_over_date = :over_date,
        ovm_status = 0
    where ovm_id = :ovm_id;
end
