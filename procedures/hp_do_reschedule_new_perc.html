CREATE PROCEDURE HP_DO_RESCHEDULE_NEW_PERC(
    K_ID D_ID,
    RECALC_DATE D_DATE,
    PERCENT_PER_DAY D_PERCENT,
    OL_MEMO D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable ol_id d_id;
declare variable reg_id d_integer;
declare variable kd_percent_effect d_percent;
declare variable kd_percent d_percent;
declare variable orig_percent_per_day d_percent;
declare variable kps_psk d_percent;
declare variable need_edoc d_bool;
begin
  -- Процедура для изменения процентной ставки и перестроения графика

  res_is_error = 0;
  res_text = 'Ставка успешно изменена';

  select kal_new_reg_id, k_kontract_no || ' от ' || f_datetostr(k_disb_date) || 'г. '
  from kontract, knaktlist kal
  where k_id = :k_id and
        kal.kal_from_kf_id = k_from_kf_id and
        kal.kal_date <= :recalc_date and
        kal.kal_date_end >= :recalc_date and
        kal.is_delete = 0
  into reg_id, ol_memo;

  if (reg_id is null) then
    exception e_default 'reg_id not found';

  if (coalesce(ol_memo, '') = '') then
    ol_memo = 'Изменение процентной ставки по договору ' || ol_memo;

  select kd_percent_per_day, kd_percent_effect_fifd
  from kontdata
  where kd_from_k_id = :k_id and
        kd_ol_date <= :recalc_date and
        kd_date_end >= kd_ol_date and
        kd_ol_is_delete = 0
  order by kd_ol_date desc
  rows 1
  into orig_percent_per_day, kd_percent_effect;

  if (orig_percent_per_day is not distinct from :percent_per_day) then
  begin
    res_text = 'Новая ставка не отличается от ставки на дату';
    suspend;

    exit;
  end

  insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind,
                        ol_memo, ol_parent_id, is_delete, ol_user_id)
  values (:reg_id, 1, 4, :recalc_date, 9, :ol_memo, null, 0, :un_id)
  returning ol_id
  into :ol_id;

  kd_percent = round(percent_per_day * 30.0000000, 3);
  insert into kontdata (kd_from_k_id, kd_from_ol_id, kd_ol_date, kd_ol_is_delete, kd_percent, kd_percent_per_day,
                        kd_percent_effect, kd_percent_effect_fifd, kd_fine_percent, kd_float_perc_level, kd_vac_status,
                        kd_memo)
  select kd_from_k_id, :ol_id, :recalc_date, 0, :kd_percent, :percent_per_day,
                       kd_percent_effect, kd_percent_effect_fifd, kd_fine_percent, kd_float_perc_level, kd_vac_status,
                       'Изменение ставки от ' || :recalc_date
  from kontdata
  where kd_from_k_id = :k_id and
        kd_ol_date <= :recalc_date and
        kd_date_end >= kd_ol_date and
        kd_ol_is_delete = 0
  rows 1;

  execute procedure hp_do_repayment_recalc_sh(:ol_id, :k_id);

  select kps_psk
  from hp_kontpsk_insert(:k_id, :ol_id)
  into kps_psk;

  if (coalesce(kps_psk, 0) = 0) then
    kps_psk = kd_percent_effect;

  need_edoc = 0;

  if (kps_psk <> kd_percent_effect) then
  begin
    update kontdata
    set kd_percent_effect_fifd = :kps_psk
    where kd_from_k_id = :k_id and
          kd_ol_date = :recalc_date and
          kd_from_ol_id = :ol_id;

    if (f_get_db_id() = 81001) then
      need_edoc = 1;
  end

  if (need_edoc = 1) then
  begin
    select kps_psk
    from hp_edoc_ins_kontpsk(:k_id, :ol_id, :un_id)
    into :kps_psk;
  end

  suspend;
end
