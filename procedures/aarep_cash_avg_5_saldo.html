CREATE PROCEDURE AAREP_CASH_AVG_5_SALDO(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_ID INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    REG_OKRUG VARCHAR(100),
    REG_SALDO_CASH DECIMAL(18,4),
    REG_CASH_LIMIT DECIMAL(18,4),
    REG_CASH_LIMIT_MEMO VARCHAR(100),
    REG_DAY_STATUS VARCHAR(100),
    REG_OVERLIMIT_COUNT INTEGER,
    REG_DAY_MEMO VARCHAR(255))
AS
declare variable n integer;
declare variable dd date;
declare variable day_closed integer;
begin
  /* отображает остатки кассы за последние 5 недель */
  for select reg_unicode, reg_code, reg_name
      from region_list(:rep_region_id, 1, 3, 1), region
      where id_list = reg_unicode and
            coalesce(reg_date_close, '31.12.2999') >= :rep_date_end
      order by 2
      into :reg_id, :reg_code, :reg_name
  do
  begin
    reg_day_status = 'Опер.день НЕ закрыт';
    select day_closed, day_memo
    from hp_is_day_closed(:reg_id, :rep_date_end, 1)
    into :day_closed, :reg_day_memo;
    if (day_closed = 1) then
      reg_day_status = 'Закрыт';

    n = 0;
    select aa_curr_level3_name
    from aarep_get_office_categ(:reg_id, :rep_date_end)
    into :reg_okrug;

    reg_cash_limit = 0;
    select rlm_limit_cash_max
    from reglimits
    where rlm_from_reg_id = :reg_id and
          rlm_date <= :rep_date_end
    order by rlm_date desc, rlm_id desc
    rows 1
    into :reg_cash_limit;
    if (reg_cash_limit is null) then
      reg_cash_limit = 0;
    reg_overlimit_count = 0;

    while (n < 5) do
    begin
      n = n + 1;
      dd = (rep_date_end + n - 5);
      reg_saldo_cash = 0;
      select res_saldo_cash
      from aarep_money_cash_saldo(:reg_id, :dd)
      into :reg_saldo_cash;
      if (reg_saldo_cash > reg_cash_limit) then
        reg_overlimit_count = reg_overlimit_count + 1;
    end
    reg_cash_limit_memo = '';
    if (reg_saldo_cash > reg_cash_limit) then
      reg_cash_limit_memo = 'Остаток превышает лимит';
    suspend;
  end
end
