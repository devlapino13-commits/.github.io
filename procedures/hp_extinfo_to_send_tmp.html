CREATE PROCEDURE HP_EXTINFO_TO_SEND_TMP(
    PROJECT_CODE D_STR_CODE)
RETURNS (
    ID D_ID,
    C_ID D_ID,
    KIB_NAME D_STR_CODE,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID ID_TYPE,
    DELAY_DAYS D_INTEGER)
AS
declare variable DB_ID D_ID;
declare variable EIC_ID D_ID;
declare variable MC_LOGIN_ID D_INTEGER;
declare variable KMP_REGISTRATION_NEW_PROCESS D_DATE;
begin
  delay_days = 3;
  db_id = f_get_db_id();
  if (coalesce(project_code, '') <> '') then
  begin
    select eic_id
    from extinfocfg
    where eic_code = :project_code
    into eic_id;

    if (eic_id is null) then
      exception e_default 'Project code not found: ' || coalesce(project_code, '<NULL>');
  end

  kmp_registration_new_process = '31.12.9999';
  if ((db_id in (39801, 41701)) and (project_code = 'Tunduk')) then
    kmp_registration_new_process = cast(f_get_const('kmp_registration_new_process') as date);

  -- с даты X задачи по проверке чз ГРС для КМП начинают работать по новой логике, т.к. запрос в ГРС делается сразу в процессе регистрации пользователя  в КМП, чз метод сервиса КМПб
  -- поэтому с даты X текущая процедура не должна возвращать задачи на проверку в ГРС полученные от КМП иначе может произойти задвоение. Т.е. задача обработается разными сервисами Володи.
  if ((db_id in (39801, 41701)) and
      (project_code = 'Tunduk') and
      (current_date >= kmp_registration_new_process)) then
  begin
      mc_login_id = f_objkind('mc_login', 'id');

      for select eil_id, eil_from_c_id, eic_code, eil_object_kind, eil_object_id
      from extinfolist, extinfocfg
      where eil_from_eic_id = eic_id and
            eil_is_json = 1 and
            eil_status in (-1, 0) and -- новые запросы и запросы с временной ошибкой
            eil_date >= current_date - :delay_days and
            eil_from_eic_id = :eic_id and
            eil_object_kind <> :mc_login_id
      order by eil_id
      rows 500
      into :id, :c_id, kib_name, object_kind, object_id
      do
      begin
        suspend;
      end
  end
  else
  begin
      for select eil_id, eil_from_c_id, eic_code, eil_object_kind, eil_object_id
      from extinfolist, extinfocfg
      where eil_from_eic_id = eic_id and
            eil_is_json = 1 and
            eil_status in (-1, 0) and -- новые запросы и запросы с временной ошибкой
            eil_date >= current_date - :delay_days and
            (:eic_id is null or
            eil_from_eic_id = :eic_id)
      order by iif (:db_id = 81001, -eil_id, eil_id)
      rows 500
      into :id, :c_id, kib_name, object_kind, object_id
      do
      begin
        suspend;
      end
  end
end
