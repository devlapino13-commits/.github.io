CREATE PROCEDURE GET_WORKPLACE_FIO(
    WP_ID DECIMAL(18,0),
    DD DATE)
RETURNS (
    PER_ID DECIMAL(18,0),
    PER_FIO VARCHAR(100))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */
  for select ps_from_c_id, c_name_fio
      from hr_pstatus, clients, workplace
      where wp_id = :wp_id and
            ((ps_from_reg_id = wp_default_reg_id) or
            (wp_default_reg_id is null) or
            (ps_from_reg_id = wp_from_reg_id)) and
            ((ps_from_dep_id = wp_default_dep_id) or
            (wp_default_dep_id is null)) and
            ((ps_from_jl_id = wp_default_job_id) or
            (wp_default_job_id is null)) and
            ps_date_begin <= :dd and
            ps_date_end >= :dd and
            c_id = ps_from_c_id
      order by c_name_fio
      into :per_id, :per_fio
  do
  begin
    suspend;
  end
end
