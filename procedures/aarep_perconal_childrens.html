CREATE PROCEDURE AAREP_PERCONAL_CHILDRENS(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    BRANCH_NAME VARCHAR(100),
    REG_NAME VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    C_SEX_TEXT VARCHAR(50),
    C_FAMILY VARCHAR(100),
    JL_NAME VARCHAR(100),
    C_KOL_CHILD INTEGER,
    R_C_NAME_FIO VARCHAR(100),
    R_C_BIRTH_DATE DATE,
    AGE INTEGER,
    PS_CURR_STATUS VARCHAR(100))
AS
declare variable c_id id_type;
begin
    for select branch_name, reg_name, c_name_fio, c_sex_text, c_family, jl_name, c_kol_child, ps_curr_status, ps_from_c_id
    from aarep_personal_list(:rep_region_id, :rep_date_end, 1)
    into :branch_name, :reg_name, :c_name_fio, :c_sex_text, :c_family, :jl_name, :c_kol_child, :ps_curr_status, :c_id

    do
    begin
    for select  c_name_fio, c_birth_date, f_age(c_birth_date, current_date)
    from hp_client_relation_list(:c_id, 0)
    where crel_from_rel_id in ('5','6')
    into  :r_c_name_fio, :r_c_birth_date, :age
    do
    begin
      suspend;
    end
  end
end
