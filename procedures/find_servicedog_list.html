    SD_ID INTEGER,
    SD_DOGOVOR_NO VARCHAR(20),
    SD_FROM_REG_ID INTEGER,
    SD_FROM_C_ID DECIMAL(18,0),
    SD_CATEGORY VARCHAR(255),
    SD_MEMO VARCHAR(1024),
    SD_DATE_BEGIN DATE,
    SD_DATE_END DATE,
    SD_DATE_CLOSED DATE,
    SD_ACC_MAIN VARCHAR(20),
    SD_ACC_ADVANCE VARCHAR(20),
    SD_ACC_EXPENSE VARCHAR(20),
    SD_USER_ID DECIMAL(18,0),
    SD_DATETIME TIMESTAMP,
    SD_REGION_NAME VARCHAR(100),
    SD_IS_INDIV_STR VARCHAR(50),
    SD_ONCE_ONLY_PAY_STR VARCHAR(50),
    SD_NEED_PATENT_STR VARCHAR(50),
    SD_NEED_SPOLIS_STR VARCHAR(50),
    SD_ONLY_WORKDAYS_STR VARCHAR(50),
    SD_CLIENT_NAME VARCHAR(100),
    SD_CLIENT_PASS_NO VARCHAR(30),
    SD_STATUS_STR VARCHAR(50),
    SD_STATUS INTEGER,
    SD_WARNING VARCHAR(255),
    SD_WARNING_INT INTEGER,
    SD_KIND_STR VARCHAR(100))
AS
declare variable sd_is_indiv integer;
declare variable sd_once_only_pay integer;
declare variable sd_need_patent integer;
declare variable sd_need_spolis integer;
declare variable sd_only_workdays integer;
declare variable rep_find_txt_up varchar(100);
declare variable dd_max date;
declare variable sd_warning1 varchar(255);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  rep_find_txt_up = upper(:rep_find_txt);
  if (((rep_find_txt <> '') or (rep_date_kind > 0)) and
      (rep_date_kind < 4)) then
  begin
    for select sd_id, sd_dogovor_no, sd_from_reg_id, sd_from_c_id, sd_category, sd_memo, sd_is_indiv, sd_date_begin,
               sd_date_end, sd_date_closed, sd_once_only_pay, sd_acc_main, sd_acc_advance, sd_acc_expense,
               sd_need_patent, sd_need_spolis, sd_only_workdays, sd_user_id, sd_datetime, reg_name || ' ' || reg_code,
               c_name_fio, c_pass_no, sd_status, sk_name
        from servicedog, region, region_list(:rep_region_id, 1, -1, 1), clients, servicekind
        where reg_unicode = sd_from_reg_id and
              (sd_from_reg_id = id_list) and
              c_id = sd_from_c_id and
              sd_kind = sk_id
--      and (c_index_c_name_fio like '%'||:rep_find_txt_up||'%'
              and
              (c_name_fio like '%' || :rep_find_txt_up || '%' or
              c_index_c_pass_no like '%' || :rep_find_txt_up || '%' or
              sd_memo like '%' || :rep_find_txt || '%') and
              (:rep_date_kind = 0 or
              ((:rep_date_kind = 1) and
              (sd_date_begin between :rep_date_begin and :rep_date_end)) or
              ((:rep_date_kind = 2) and
              (sd_date_end between :rep_date_begin and :rep_date_end)) or
              ((:rep_date_kind = 3) and
              (sd_date_closed between :rep_date_begin and :rep_date_end))) and
              (:rep_sd_kind = -1 or
              sd_kind = :rep_sd_kind)
        into :sd_id, :sd_dogovor_no, :sd_from_reg_id, :sd_from_c_id, :sd_category, :sd_memo, :sd_is_indiv,
             :sd_date_begin, :sd_date_end, :sd_date_closed, :sd_once_only_pay, :sd_acc_main, :sd_acc_advance,
             :sd_acc_expense, :sd_need_patent, :sd_need_spolis, :sd_only_workdays, :sd_user_id, :sd_datetime,
             :sd_region_name, :sd_client_name, :sd_client_pass_no, :sd_status, :sd_kind_str

    begin
      if (sd_is_indiv = 0) then
        sd_is_indiv_str = 'Юр.лицо';
      else
        sd_is_indiv_str = 'Физ.лицо';

        sd_once_only_pay_str = 'Разовый';
      else
        sd_once_only_pay_str = 'Ежемесячный';
      if (sd_need_patent = 1) then
        sd_need_patent_str = 'Требуется';
      else
        sd_need_patent_str = 'Не требуется';
      if (sd_need_spolis = 1) then
        sd_need_spolis_str = 'Требуется';
      else
        sd_need_spolis_str = 'Не требуется';
      if (sd_only_workdays = 1) then
        sd_only_workdays_str = 'Почасовая оплата';
      else
        sd_only_workdays_str = 'Оплата за период';
      if (sd_status = -1) then
        sd_status_str = 'Аннулирован';
      if (sd_status = 0) then
        sd_status_str = 'В процессе';
      if (sd_status = 1) then
        sd_status_str = 'Утвержден';
      if (sd_status = 2) then
        sd_status_str = 'Закрыт';
      dd_max = null;
      sd_warning = '';
      sd_warning1 = '';
      sd_warning_int = 0;
      if (sd_need_patent = 1) then
      begin
        select sdc_patent_date_end
        from servicedocs
        where sdc_patent_date_begin <= :sd_date_end and
              sdc_patent_date_end >= :sd_date_begin and
              sdc_from_sd_id = :sd_id
        order by sdc_patent_date_end desc
        rows 1
        into :dd_max;
        if (dd_max is null) then
        begin
          sd_warning = 'Нет патента';
        end
        else
        begin
          if (:dd_max < :sd_date_end) then
            sd_warning = 'Срок патента не соответствует';
        end
      end
      dd_max = null;
      if (sd_need_spolis = 1) then
      begin
        select sdc_spolis_date_end
        from servicedocs
        where sdc_spolis_date_begin <= :sd_date_end and
              sdc_spolis_date_end >= :sd_date_begin and
              sdc_from_sd_id = :sd_id
        order by sdc_spolis_date_end desc
        rows 1
        into :dd_max;
        if (dd_max is null) then
        begin
          sd_warning1 = 'Нет соцполиса';
        end
        else
        begin
          if (:dd_max < :sd_date_end) then
            sd_warning1 = 'Срок соцполиса не соответствует';
        end
      end
      if ((sd_warning <> '') and
          (sd_warning1 <> '')) then
        sd_warning = :sd_warning || ',  ' || :sd_warning1;
      else
      begin
        if (sd_warning1 <> '') then
          sd_warning = :sd_warning1;
      end
      if (sd_warning <> '') then
        sd_warning_int = 1;
      suspend;
    end
  end
  else
  begin
    if ((rep_find_txt <> '') or (rep_date_kind = 4)) then
    begin
      for select distinct sd_id, sd_dogovor_no, sd_from_reg_id, sd_from_c_id, sd_category, sd_memo, sd_is_indiv,
                          sd_date_begin, sd_date_end, sd_date_closed, sd_once_only_pay, sd_acc_main, sd_acc_advance,
                          sd_acc_expense, sd_need_patent, sd_need_spolis, sd_only_workdays, sd_user_id, sd_datetime,
                          reg_name || ' ' || reg_code, c_name_fio, c_pass_no, sd_status, sk_name
          from servicedog, region, region_list(:rep_region_id, 1, -1, 1), clients, servicesh, servicekind
          where reg_unicode = sd_from_reg_id and
                (sd_from_reg_id = id_list) and
                sd_kind = sk_id and
                c_id = sd_from_c_id and
                sd_id = ssh_from_sd_id and
                ssh_amount_dogovor <> 0
--          and (c_index_c_name_fio like '%'||:rep_find_txt_up||'%'
                and
                (c_name_fio like '%' || :rep_find_txt_up || '%' or
                c_index_c_pass_no like '%' || :rep_find_txt_up || '%' or
                sd_memo like '%' || :rep_find_txt || '%') and
                (:rep_date_kind = 0 or
                ((:rep_date_kind = 4) and
                (ssh_date between :rep_date_begin and :rep_date_end))) and
                (:rep_sd_kind = -1 or
                sd_kind = :rep_sd_kind)
          into :sd_id, :sd_dogovor_no, :sd_from_reg_id, :sd_from_c_id, :sd_category, :sd_memo, :sd_is_indiv,
               :sd_date_begin, :sd_date_end, :sd_date_closed, :sd_once_only_pay, :sd_acc_main, :sd_acc_advance,
               :sd_acc_expense, :sd_need_patent, :sd_need_spolis, :sd_only_workdays, :sd_user_id, :sd_datetime,
               :sd_region_name, :sd_client_name, :sd_client_pass_no, :sd_status, :sd_kind_str
      do
      begin
        if (sd_is_indiv = 0) then
          sd_is_indiv_str = 'Юр.лицо';
        else
          sd_is_indiv_str = 'Физ.лицо';
        if (sd_once_only_pay = 1) then
          sd_once_only_pay_str = 'Разовый';
        else
          sd_once_only_pay_str = 'Ежемесячный';
        if (sd_need_patent = 1) then
          sd_need_patent_str = 'Требуется';
        else
          sd_need_patent_str = 'Не требуется';
        if (sd_need_spolis = 1) then
          sd_need_spolis_str = 'Требуется';
        else
          sd_need_spolis_str = 'Не требуется';
        if (sd_only_workdays = 1) then
          sd_only_workdays_str = 'Почасовая оплата';
        else
          sd_only_workdays_str = 'Оплата за период';
        if (sd_status = -1) then
          sd_status_str = 'Аннулирован';
        if (sd_status = 0) then
          sd_status_str = 'В процессе';
        if (sd_status = 1) then
          sd_status_str = 'Утвержден';
        if (sd_status = 2) then
          sd_status_str = 'Закрыт';
        dd_max = null;
        sd_warning = '';
        sd_warning1 = '';
        sd_warning_int = 0;
        if (sd_need_patent = 1) then
        begin
          select sdc_patent_date_end
          from servicedocs
          where sdc_patent_date_begin <= :sd_date_end and
                sdc_patent_date_end >= :sd_date_begin and
                sdc_from_sd_id = :sd_id
          order by sdc_patent_date_end desc
          rows 1
          into :dd_max;
          if (dd_max is null) then
          begin
            sd_warning = 'Нет патента';
          end
          else
          begin
            if (:dd_max < :sd_date_end) then
              sd_warning = 'Срок патента не соответствует';
          end
        end
        dd_max = null;
        if (sd_need_spolis = 1) then
        begin
          select sdc_spolis_date_end
          from servicedocs
          where sdc_spolis_date_begin <= :sd_date_end and
                sdc_spolis_date_end >= :sd_date_begin and
                sdc_from_sd_id = :sd_id
          order by sdc_spolis_date_end desc
          rows 1
          into :dd_max;
          if (dd_max is null) then
          begin
            sd_warning1 = 'Нет соцполиса';
          end
          else
          begin
            if (:dd_max < :sd_date_end) then
              sd_warning1 = 'Срок соцполиса не соответствует';
          end
        end
        if ((sd_warning <> '') and
            (sd_warning1 <> '')) then
          sd_warning = :sd_warning || ',  ' || :sd_warning1;

        begin
          if (sd_warning1 <> '') then
            sd_warning = :sd_warning1;

    end
  end
