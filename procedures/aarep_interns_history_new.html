CREATE PROCEDURE AAREP_INTERNS_HISTORY_NEW(
    REP_REG_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    JL_I_NAME D_STR_100,
    REG_OFFICE_NAME VARCHAR(50),
    REG_NAME VARCHAR(60),
    FIO D_STR_100,
    BIRTH_DATE D_DATE,
    PASS_NO VARCHAR(30),
    PASS_DATE D_DATE,
    PASS_WHO D_STR_255,
    INN D_STR_20,
    NATIONALITY D_STR_255,
    AGE D_INTEGER,
    SEX D_STR_10,
    PHONE_NUM D_STR_20,
    PERSONNEL_DATE D_DATE,
    EDUCATIONAL_DATE D_DATE,
    PS_DATE_BEGIN D_DATE,
    JL_NAME D_STR_100,
    ADDRESS D_STR_255,
    ADDRESS_FACT D_STR_255,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    PERSONNEL_RES D_STR_20,
    EDUCATIONAL_RES D_STR_20)
AS
declare variable reg_id_tmp d_integer;
declare variable reg_id d_integer;
declare variable nation_tmp d_integer;
declare variable nation d_integer;
declare variable i_jl_id d_integer;
declare variable i_jl_id_tmp d_integer;
declare variable jl_id d_integer;
declare variable jl_id_tmp d_integer;
declare variable addr_empty d_str_255;
declare variable adl_id d_id;
declare variable address_type d_smallint;
declare variable addr_str_tmp d_str_255;
declare variable c_id d_id;
declare variable jl_name_tmp d_str_100;
begin
  reg_id = 0;
  i_jl_id = 0;
  jl_id = 0;

  select adl.adl_address_str
  from addrlist adl
  where adl.adl_id = 41701000000000
  into :addr_empty;

  for select i.i_from_jl_id, i.i_from_reg_id, rl.parent_name, c.c_name_fio, c.c_birth_date, c.c_pass_no, c.c_pass_date,
             c.c_pass_who, c.c_inn, c.c_nation, f_age(c.c_birth_date, cast(i.i_registred_datetime as date)),
             case c.c_sex
               when 0 then 'женский'
               when 1 then 'мужской'
             end,
             f_cont_info('clients', c.c_id, 'cell', 'last'), cast(icr.icr_personnel_com_timestamp as date),
             case icr.icr_personnel_com_res
               when 0 then 'Отклонен'
               when 1 then 'Утвержден'
               else null
             end,
             cast(icr.icr_educational_com_timestamp as date),
             case icr.icr_educational_com_res
               when 0 then 'Отклонен'
               when 1 then 'Утвержден'
               else null
             end,
             i.i_from_c_id, i.i_date_begin, i.i_date_end
      from interns i
      inner join region_list(:rep_reg_id, 1, 3, 1) rl on i.i_from_reg_id = rl.id_list
      inner join clients c on i.i_from_c_id = c.c_id
      left join intern_committee_result icr on icr.icr_from_i_id = i.i_id
      where (:rep_date_end between i.i_date_begin and i.i_date_end or
            i.i_date_begin is null)
      into :i_jl_id_tmp, :reg_id_tmp, :reg_name, :fio, :birth_date, :pass_no, :pass_date, :pass_who, :inn, :nation_tmp,
           :age, :sex, :phone_num, :personnel_date, :personnel_res, :educational_date, :educational_res, :c_id,
           :date_begin, :date_end
  do
  begin

    ps_date_begin = null;
    jl_id_tmp = null;

    if (:date_begin is not null) then
      select ps.ps_date_begin, ps.ps_from_jl_id
      from hr_pstatus ps
      inner join hr_orders ho on ps.ps_from_ho_id = ho.ho_id and
            ho.ho_order_type = 1 and
            ho.ho_status = 1
      where ps.ps_from_c_id = :c_id and
            ps.ps_status = 1 and
            ps.ps_date_begin >= :date_begin
      order by ps.ps_id, ps.ps_from_ho_id
      rows 1
      into :ps_date_begin, :jl_id_tmp;

    if (reg_id_tmp <> reg_id) then
    begin
      reg_id = :reg_id_tmp;

      select reg.reg_code || ' ' || reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :reg_office_name;
    end

    if (:i_jl_id_tmp <> :i_jl_id) then
    begin
      i_jl_id = :i_jl_id_tmp;
      select jl.jl_name
      from joblist jl
      where jl.jl_id = :i_jl_id
      into :jl_i_name;
    end

    if (:jl_id_tmp is not null and
        :jl_id_tmp <> :jl_id) then
    begin
      jl_id = :jl_id_tmp;
      select jl.jl_name
      from joblist jl
      where jl.jl_id = :jl_id
      into :jl_name_tmp;

      jl_name = jl_name_tmp;
    end
    else
    begin
      if (:jl_id_tmp is null) then
        jl_name = null;
      if (:jl_id_tmp = :jl_id) then
        jl_name = :jl_name_tmp;
    end

    if (:nation_tmp is null) then
      nationality = null;
    else
    if (:nation_tmp <> :nation or :nation is null) then
    begin
      nation = :nation_tmp;
      select rd.rd_name
      from refdata rd
      where rd.rd_id = :nation
      into :nationality;
    end

    for select adc.adc_address_type, adc.adc_from_adl_id
        from addrclient adc
        where adc.adc_from_c_id = :c_id and
              adc.adc_address_type in (1, 2)
        into :address_type, :adl_id

    do
    begin
      if (:adl_id <> 41701000000000) then
        select adl.adl_address_str
        from addrlist adl
        where adl.adl_id = :adl_id
        into :addr_str_tmp;
      else
        addr_str_tmp = :addr_empty;

      if (:address_type = 1) then
        address = :addr_str_tmp;
      if (:address_type = 2) then
        address_fact = :addr_str_tmp;
    end

    suspend;
  end
end
