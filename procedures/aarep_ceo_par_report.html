CREATE PROCEDURE AAREP_CEO_PAR_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_C_ID DECIMAL(18,0),
    REP_DAYS_IN_PERIOD INTEGER)
RETURNS (
    REG_ID INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    REP_LINE_NO INTEGER,
    REP_DATE DATE,
    REP_BALANCE_ALL DECIMAL(18,4),
    REP_BALANCE_7 DECIMAL(18,4),
    REP_BALANCE_30 DECIMAL(18,4),
    REP_KOLICH_7 INTEGER,
    REP_KOLICH_30 INTEGER,
    LO_ID DECIMAL(18,0),
    LO_INFO VARCHAR(100),
    REP_DISB_AMOUNT DECIMAL(18,4),
    REP_DISB_CLIENT INTEGER,
    REP_BALANCE_1 DECIMAL(18,4),
    REP_KOLICH_1 INTEGER,
    REP_KOLICH_ALL INTEGER,
    REP_BALANCE_8_180 CURR_TYPE,
    REP_BALANCE_31_180 CURR_TYPE,
    PORTFEL_BONUS CURR_TYPE,
    REP_KOLICH_CLI INTEGER,
    REP_DISB_LGOT_AMOUNT CURR_TYPE,
    REP_DISB_IND_AMOUNT CURR_TYPE,
    REP_BALANCE_LGOT CURR_TYPE,
    REP_BALANCE_IND CURR_TYPE,
    REP_BALANCE_180_ CURR_TYPE,
    REP_KOLICH_180_ CURR_TYPE,
    REP_KOLICH_CLI_180_ CURR_TYPE,
    REP_KOLICH_IND INTEGER,
    REP_TRANSFER_CLI INTEGER,
    REP_DISB_LGOT_KOLICH INTEGER,
    REP_DISB_IND_KOLICH INTEGER,
    REP_KOLICH_LGOT INTEGER,
    REP_KOLICH_CLI_EX_TRANSFER INTEGER)
AS
declare variable LO_ID_PRIOR decimal(18,0);
declare variable LO_FIO varchar(100);
declare variable LO_JOBNAME varchar(100);
declare variable LO_CELL_NO varchar(100);
declare variable LINE integer;
declare variable STAGE_DATE date;
declare variable REP_DATE_END_2 date;
declare variable PORTFEL_B decimal(18,4);
declare variable PORTFEL_C decimal(18,4);
declare variable PORTFEL_D decimal(18,4);
declare variable LO_STAGE_MONTH_FULL D_SMALLINT;
begin
  lo_id_prior = 0;
  rep_line_no = 0;
  lo_jobname = '';
  lo_cell_no = '';
  if ((rep_c_id is null) or (rep_c_id = -1)) then
    rep_c_id = 0;
  rep_date_end_2 = rep_date_end - extract(day from rep_date_end); -- конец предыдущего месяца
  portfel_b = 0;
  for select reg_code, reg_name, reg_id, lo_id, rep_date, rep_line_no, sum(rep_balance_all), sum(rep_kolich_all),
             sum(rep_balance_1), sum(rep_kolich_1), sum(rep_balance_7), sum(rep_kolich_7), sum(rep_balance_30),
             sum(rep_kolich_30), sum(rep_disb_amount), sum(rep_disb_client), sum(rep_balance_8_180),
             sum(rep_balance_31_180), sum(rep_kolich_cli), sum(rep_disb_lgot_amount), sum(rep_disb_ind_amount),
             sum(rep_balance_lgot), sum(rep_balance_ind), sum(rep_balance_180_), sum(rep_kolich_180_),
             sum(rep_kolich_cli_180_), sum(rep_kolich_ind), sum(rep_transfer_cli), sum(rep_disb_lgot_kolich),
             sum(rep_disb_ind_kolich), sum(rep_kolich_lgot), sum(rep_kolich_cli_ex_transfer)
      from aarep_ceo_par_report_tmp(:rep_region_id, :rep_date_end, :rep_c_id, :rep_days_in_period)
      group by reg_code, reg_name, reg_id, lo_id, rep_date, rep_line_no, parent2, parent
      order by parent2, parent, reg_code, lo_id, rep_date desc
      into :reg_code, :reg_name, reg_id, :lo_id, :rep_date, rep_line_no, :rep_balance_all, :rep_kolich_all,
           :rep_balance_1, :rep_kolich_1, :rep_balance_7, :rep_kolich_7, :rep_balance_30, :rep_kolich_30,
           :rep_disb_amount, rep_disb_client, rep_balance_8_180, rep_balance_31_180, :rep_kolich_cli,
           :rep_disb_lgot_amount, :rep_disb_ind_amount, rep_balance_lgot, rep_balance_ind, rep_balance_180_,
           rep_kolich_180_, rep_kolich_cli_180_, rep_kolich_ind, rep_transfer_cli, rep_disb_lgot_kolich,
           rep_disb_ind_kolich, rep_kolich_lgot, rep_kolich_cli_ex_transfer
  do
  begin
    if (lo_id <> lo_id_prior) then
    begin
      line = 1;
      select c_name_fio,
             (select cont_value
              from hp_contdata_get_cell_main(0, :lo_id))
      from clients
      where c_id = :lo_id
      into :lo_fio, :lo_cell_no;

      lo_jobname = '';
      /*
      select jl_name
      from hr_pstatus, joblist
      where ps_from_c_id = :lo_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_status = 1 and
            ps_from_jl_id = jl_id and
            ps_is_working = 1
      order by ps_is_working desc
      rows 1
      into :lo_jobname;
      if ((lo_jobname is null) or (lo_jobname = '')) then
        lo_jobname = 'Должность не известна';

      lo_jobname = '';
      stage_date = null;
      /*
      select ps_date_begin
      from hr_pstatus, joblist, jobkind
      where ps_from_c_id = :lo_id and
            ps_status = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            ps_is_working = 1 and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')
      order by ps_date_begin
      rows 1
      into :stage_date;
      if (stage_date is not null) then
      begin
        if (lo_jobname <> 'Должность не известна') then
          lo_jobname = lo_jobname || ', стаж СВЗ: ' || datediff(month, stage_date, :rep_date_end) || ' мес.';
      end
      */

      select lo_stage_month_full
      from hp_get_lo_stage_full(:lo_id, :rep_date_end, null, 1)
      into lo_stage_month_full;

      if (lo_stage_month_full > 0) then
      lo_jobname = lo_jobname || ', стаж СВЗ: ' || lo_stage_month_full || ' мес.';

      lo_id_prior = lo_id;
    end
    else
      line = line + 1;

    portfel_bonus = 0;

    if (line = 1) then
      lo_info = lo_fio;
    if (line = 2) then
      lo_info = lo_jobname;
    if (line = 3) then
      lo_info = lo_cell_no;
    if (line > 3) then
      lo_info = '';
    suspend;
  end
end
