CREATE PROCEDURE HP_TICKET_UPD(
    TKT_ID D_ID,
    TKT_STATUS D_SMALLINT,
    TKT_OBJECT_KIND D_INTEGER,
    TKT_OBJECT_ID D_ID,
    TKT_DATETIME D_DATETIME = null)
AS
begin
  if (coalesce(tkt_id, 0) = 0) then
    exception e_default 'Emptry Ticket id';

  -- если последняя запись идентична передаваемым параметрам,
  -- ничего не меняем и завершаем работу
  if (exists(select *
             from ticket, ticketlog
             where tkt_id = :tkt_id and
                   tkl_from_tkt_id = tkt_id and
                   tkt_actual_version = tkl_version and
                   tkt_status = :tkt_status and
                   tkl_object_kind = :tkt_object_kind and
                   tkl_object_id = :tkt_object_id)) then
    exit;

  -- перед закрытием тикета, нужно убедиться, что тикет находится в передаваемом Object_Kind
  if (tkt_status in (-2, 2)) then
  begin
    if (exists(select *
               from ticket, ticketlog
               where tkt_id = :tkt_id and
                     tkl_from_tkt_id = tkt_id and
                     tkt_actual_version = tkl_version and
                     tkl_object_kind <> :tkt_object_kind)) then
      exit;
  end

  update ticket
  set tkt_status = :tkt_status
  where tkt_id = :tkt_id and
        not(tkt_status = :tkt_status);

  insert into ticketlog (tkl_from_tkt_id, tkl_object_kind, tkl_object_id, tkl_datetime)
  values (:tkt_id, :tkt_object_kind, :tkt_object_id, :tkt_datetime);
end
