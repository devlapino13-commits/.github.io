Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE AAREP_REVALUE_OF_CHANGES_T1(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RO_1_BEGIN DECIMAL(18,4),
    RO_2_BEGIN DECIMAL(18,4),
    RO_3_BEGIN DECIMAL(18,4),
    GOTO_RO1 DECIMAL(18,4),
    GOTO_RO1_OF_RO2 DECIMAL(18,4),
    GOTO_RO2_OF_RO1 DECIMAL(18,4),
    GOTO_RO2 DECIMAL(18,4),
    GOTO_RO3_OF_RO1 DECIMAL(18,4),
    GOTO_RO3_OF_RO2 DECIMAL(18,4),
    GOTO_RO3 DECIMAL(18,4),
    RECOVERY_RO1 DECIMAL(18,4),
    RECOVERY_RO2 DECIMAL(18,4),
    RECOVERY_RO3 DECIMAL(18,4),
    INCREASE_RO1 DECIMAL(18,4),
    INCREASE_RO2 DECIMAL(18,4),
    INCREASE_RO3 DECIMAL(18,4),
    NEW_RO1 DECIMAL(18,4),
    RETIREMENT_RO3 DECIMAL(18,4),
    DISPOSAL_RO1 CURR_TYPE,
    DISPOSAL_RO2 CURR_TYPE,
    DISPOSAL_RO3 CURR_TYPE)
AS
begin
  select sum(ro_1_begin), sum(ro_2_begin), sum(ro_3_begin),
         sum(goto_ro1), sum(goto_ro1_of_ro2),
         sum(goto_ro2_of_ro1), sum(goto_ro2),
         sum(goto_ro3_of_ro1), sum(goto_ro3_of_ro2), sum(goto_ro3),
         sum(recovery_ro1), sum(recovery_ro2), sum(recovery_ro3),
         sum(new_ro1),
         sum(increase_ro1), sum(increase_ro2), sum(increase_ro3),
         sum(retirement_ro3),
         sum(disposal_ro1), sum(disposal_ro2), sum(disposal_ro3)
  from aarep_revalue_of_changes_reser(:rep_date_begin, :rep_date_end)
  into ro_1_begin, ro_2_begin, ro_3_begin,
       goto_ro1, goto_ro1_of_ro2,
       goto_ro2_of_ro1, goto_ro2,
       goto_ro3_of_ro1, goto_ro3_of_ro2, goto_ro3,
       recovery_ro1, recovery_ro2, recovery_ro3,
       new_ro1,
       increase_ro1, increase_ro2, increase_ro3,
       retirement_ro3,
       disposal_ro1, disposal_ro2, disposal_ro3;
  suspend;
end
