CREATE PROCEDURE HP_REPAY_CALC_KOMMIS_RU(
    K_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    RES_KOMMIS_AMOUNT DECIMAL(18,4))
AS
declare variable need_kommis decimal(18,4) = 0;
declare variable kc_id decimal(18,0);
declare variable last_date date;
declare variable base_amount decimal(18,4) = 0;
declare variable k_percent_per_day decimal(18,7);
declare variable aa_date_repay date;
declare variable kol_days integer;
declare variable gr_amount decimal(18,4) = 0;
declare variable tmp decimal(18,4) = 0;
declare variable pr_interest_method integer;
declare variable k_stop_perc date;
declare variable k_last_status integer;
declare variable min_qty_day_before_pay integer;
begin
-- Вычисляет размер комиссии за досрочное погашение, РФ

  need_kommis = 1;
  res_kommis_amount = 0;

  if (rep_date >= cast('23.03.2018' as date)) then -- реализовано на основании письма от Зары. Тема: ТЗ по изменениям в продукты, требования к клиентам и кредитный процесс
  begin
    need_kommis = 0;
    suspend;
    exit;
  end

-- Если папка числится в РФ ГО, комиссию не начислять
  if ((select kal_new_reg_id
       from knaktlist, kontract
       where kal_from_kf_id = k_from_kf_id and
             k_id = :k_id and
             knaktlist.is_delete = 0 and
             kal_date <= :rep_date and
             kal_date_end >= :rep_date) = 1010100) then
    need_kommis = 0;

  select k_stop_perc, pr_interest_method
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into :k_stop_perc, pr_interest_method;

  if (k_stop_perc is not null) then
    need_kommis = 0;

  kol_days = 10;

  k_last_status = 0;

-- беру статус который был при выдаче займа
  select k_last_status
  from hp_get_kontract_status(:k_id, :rep_date)
  into :k_last_status;

  k_last_status = coalesce(k_last_status, 0);
  if (k_last_status in (3, 4, 5)) then
    need_kommis = 0;

-- Проверяем, есть ли заявка на досрочное погашение
  if (need_kommis = 1) then
  begin
    aa_date_repay = null;

    min_qty_day_before_pay = 10;
    if (k_last_status in (1, 2)) then
      min_qty_day_before_pay = 5;

    select aa_date_repay
    from apadvanced
    where aa_is_deleted = 0 and
          aa_from_k_id = :k_id and
          (:c_id = 0 or
          aa_from_c_id = :c_id) and
          aa_date_app <= :rep_date and
          datediff(day, aa_date_app, aa_date_repay) >= :min_qty_day_before_pay
    order by aa_date_app desc
    rows 1
    into :aa_date_repay;

    if (aa_date_repay is not null) then
    begin
-- если дата в заявке и фактическое совпадает
      if (aa_date_repay = rep_date) then
        need_kommis = 0;

-- если дата в заявке и фактическое плюс-минус 1 день
      if (abs(datediff(day, :rep_date, :aa_date_repay)) <= 1) then
        need_kommis = 0;

-- если дата погашения в заявке попадает на выходные
      if (extract(weekday from aa_date_repay) in (0, 6)) then
      begin
-- если дата фактического погашения в заявке попадает на ближайшие пт-пн
        if (extract(weekday from rep_date) in (0, 1, 5, 6)) then
        begin
-- если между ними не более 2 дней, то не считаем комиссию.
          if (abs(datediff(day, :rep_date, :aa_date_repay)) <= 2) then
            need_kommis = 0;
        end
      end

      if (need_kommis = 1) then
      begin
        if (aa_date_repay > rep_date and
            datediff(day, :rep_date, :aa_date_repay) < 10) then
          kol_days = datediff(day, :rep_date, :aa_date_repay);
      end
    end
  end

-- если нет заявки, проверяем исключения
  if (need_kommis = 1) then
  begin
-- Если не прошло 14 дней с момента выдачи, могут погашать без комиссии
    if (exists(select *
               from kontstate
               where ks_from_k_id = :k_id and
                     ks_state = 1 and
                     ks_ol_is_delete = 0 and
                     datediff(day, ks_date_begin, :rep_date) < 14)) then
      need_kommis = 0;
  end

  if (need_kommis = 1) then
  begin
-- Если до конца срока кредита осталось меньше 10 дней, могут погашать без комиссии
    select ok_sh_date
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_from_c_id = 0
    group by ok_sh_date
    having sum(ok_sh_os + ok_os) > 0
    order by ok_sh_date desc
    rows 1
    into :last_date;

    if (last_date is null) then
    begin
      select sh_date
      from schedule
      where sh_from_k_id = :k_id
      order by sh_from_k_id desc, sh_date desc
      rows 1
      into :last_date;
    end

    if (datediff(day, :rep_date, :last_date) < 10) then
      need_kommis = 0;
  end

  if (need_kommis = 1) then
  begin
-- Проверяем, есть ли заявка на следующий цикл у группы или участника группы
    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              (:c_id = 0 or
              kc_from_c_id = :c_id)
        into :kc_id
    do
    begin
      if (exists(select *
                 from miniapp, miniappclient
                 where ma_id = mac_from_ma_id and
                       mac_amount > 0 and
                       mac_from_c_id = :kc_id and
                       ma_date_end >= '31.12.2999' and
                       ma_date_oper >= :rep_date and
                       ma_date_app <= :rep_date and
                       ma_is_active = 1)) then
        need_kommis = 0;
    end
  end

  if (need_kommis = 1 and
      rep_date >= cast('01.04.2016' as date)) then
  begin
-- Разработана по приказу №59-03-16 о/д  от 11 марта, добавил Улан. от 30.03.2016
--  2,5 в случае, если клиент полностью досрочно погашает микрозаем, по которому имеется активная (действующая) просрочка
    if (c_id > 0) then
    begin
      if (exists(select *
                 from knclient, oksummary
                 where kc_from_c_id = :c_id and
                       kc_amount_approved > 0 and
                       kc_from_k_id = oks_from_k_id and
                       oks_close_state = 0 and
                       cast(:rep_date as date) between oks_date_begin and oks_date_end and
                       cast(:rep_date as date) - oks_delinq_date > 0)) then
      begin
        need_kommis = 0;
      end
    end
    else
    begin
      if (exists(select *
                 from oksummary
                 where oks_from_k_id = :k_id and
                       oks_close_state = 0 and
                       cast(:rep_date as date) between oks_date_begin and oks_date_end and
                       cast(:rep_date as date) - oks_delinq_date > 0)) then
      begin
        need_kommis = 0;
      end
    end
--  дополнение по пункту 2,1 в течении 30ти календарных дней со дня получения микрозайма для ЦЕЛЕВЫХ мирозаймов
--  по письму Зары получено от March 14, 2016 8:54 PM
--  где "К целевым займам сейчас относится только один продукт (это для п. 2.1 Приказа) – Микрозаем на патент."
    if (need_kommis = 1 and
        exists(select *
               from kontstate, kontract, product
               where k_id = :k_id and
                     k_from_pr_id = pr_id and
                     pr_code containing '.PATENT.' and
                     ks_from_k_id = k_id and
                     ks_state = 1 and
                     ks_ol_is_delete = 0 and
                     datediff(day, ks_date_begin, :rep_date) < 30)) then
      need_kommis = 0;
  end

  if (need_kommis = 1) then
  begin
-- если вышеуказанные условия не соблюдались
-- вычисляем сумму комиссии
    base_amount = 0;

    if (pr_interest_method > 0) then -- На остаток
    begin
      select oks_balance, kd_percent_per_day
      from oksummary, kontdata
      where oks_from_k_id = :k_id and
            oks_date_begin <= :rep_date and
            oks_date_end >= :rep_date and
            kd_from_k_id = :k_id and
            kd_ol_is_delete = 0 and
            :rep_date between kd_ol_date and kd_date_end and
            oks_close_state = 0
      into :base_amount, k_percent_per_day;

      if (c_id > 0) then
      begin
        select k_amount_approved, kc_amount_approved
        from knclient, kontract
        where kc_from_c_id = :c_id and
              kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              k_id = :k_id
        into :gr_amount, :tmp;
        base_amount = round(cast(base_amount as double precision) * (tmp / cast(gr_amount as double precision)), 0);
      end
    end
    else
    begin
      if (c_id > 0) then
      begin
        select kc_amount_approved, kd_percent_per_day
        from knclient, kontdata
        where kc_from_c_id = :c_id and
              kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              kd_from_k_id = :k_id and
              kd_ol_is_delete = 0 and
              :rep_date between kd_ol_date and kd_date_end
        into :base_amount, :k_percent_per_day;
      end
      else
      begin
        select k_amount_approved, kd_percent_per_day
        from kontract, kontdata
        where k_id = :k_id and
              k_amount_approved > 0 and
              kd_from_k_id = :k_id and
              kd_ol_is_delete = 0 and
              :rep_date between kd_ol_date and kd_date_end
        into :base_amount, :k_percent_per_day;
      end
    end
    res_kommis_amount = round((cast(base_amount as double precision) / 100.0 * :kol_days) * k_percent_per_day, 0);
  end
  suspend;

end
