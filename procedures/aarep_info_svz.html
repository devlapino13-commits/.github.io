declare variable rep_date date;
declare variable def_curr varchar(3);
declare variable curr_rate curr_type;
begin
  rep_date = current_date - 1;
  def_curr = (select cm_default_currency
              from company);
  curr_rate = (select cr_rate
               from opercurr o
               where :rep_date between cr_date and cr_date_end and
                     o.cr_from_cu_id = '840');
  disb_date_first = null;
  disb_date_last = null;
  disb_kolich_new = null;
  disb_kolish_old = null;
  disb_kol_new_last_3 = null;
  disb_kol_old_last_3 = null;
-- Вычисляет первую и последнию даты выдачи
  select min(ks_date_begin), -- первая дата выдачи
         max(ks_date_begin), -- последняя дата выдачи
         sum((select count(*)
              from knclient -- выданные новые кредиты
              where kc_from_k_id = ks_from_k_id and
                    kc_amount_approved > 0 and
                    kc_circle <= 1)), sum((select count(*)
                                           from knclient -- выденные повторные кредиты
                                           where kc_from_k_id = ks_from_k_id and
                                                 kc_amount_approved > 0 and
                                                 kc_circle > 1)), sum(iif(ks_date_begin between :rep_date - 90 and :rep_date,
-- выданные новые кредиты за последние 90 дней
         (select count(*)
          from knclient
          where kc_from_k_id = ks_from_k_id and
                kc_amount_approved > 0 and
                kc_circle <= 1), 0)), sum(iif(ks_date_begin between :rep_date - 90 and :rep_date,
-- выданные повторные кредиты за последние 90 дней
         (select count(*)
          from knclient
          where kc_from_k_id = ks_from_k_id and
                kc_amount_approved > 0 and
                kc_circle > 1), 0)),

         sum((select sum(kc_amount_approved * k_index_curr_rate)
              from knclient, kontract
              where kc_from_k_id = ks_from_k_id and
                    kc_amount_approved > 0 and
                    kc_circle <= 1 and
                    kc_from_k_id = k_id and
                    ks_from_k_id = k_id)),

         sum((select sum(kc_amount_approved * k_index_curr_rate)
              from knclient, kontract
              where kc_from_k_id = ks_from_k_id and
                    kc_amount_approved > 0 and
                    kc_circle > 1 and
                    kc_from_k_id = k_id and
                    ks_from_k_id = k_id))

  from kontstate
  where ks_ol_is_delete = 0 and
        ks_state = 1 and
        ks_kc_id = :rep_officer_id and
        ks_date_begin <= :rep_date
  into disb_date_first, disb_date_last, disb_kolich_new, disb_kolish_old, disb_kol_new_last_3, disb_kol_old_last_3,
       disb_amount_new, disb_amount_old;

  if (disb_kolich_new is null) then
    disb_kolich_new = 0;
  if (disb_kolish_old is null) then
    disb_kolish_old = 0;
  if (disb_kol_new_last_3 is null) then
    disb_kol_new_last_3 = 0;
  if (disb_kol_old_last_3 is null) then
    disb_kol_old_last_3 = 0;

  disb_kol_new_last_3 = disb_kol_new_last_3 / 3;
  disb_kol_old_last_3 = disb_kol_old_last_3 / 3;
-- Вытаскивает данные по своим выданным кредитам
  with tabl1
  as (select (select aa_category_id
              from aarep_get_category_id(ks_from_k_id, :rep_date, null, null, null, null, null, null)) aa_category_id, iif(k_index_curr_code = :def_curr,
-- Расчет портфеля контракта в сводном эквиваленте
              (select oks_balance
               from oksummary
               where oks_from_k_id = k_id and
                     oks_close_state = 0 and
                     :rep_date between oks_date_begin and oks_date_end), iif(k_index_curr_rate >= :curr_rate,
              (select oks_balance
               from oksummary
               where oks_from_k_id = k_id and
                     oks_close_state = 0 and
                     :rep_date between oks_date_begin and oks_date_end) * k_index_curr_rate,
              (select oks_balance
               from oksummary
               where oks_from_k_id = k_id and
                     oks_close_state = 0 and
                     :rep_date between oks_date_begin and oks_date_end) * :curr_rate)) k_balance, coalesce((select cast(:rep_date as date) - oks_delinq_date
                                                                                                            from oksummary -- Расчет кол-во просроченных дней
                                                                                                            where oks_from_k_id = k_id and
                                                                                                                  oks_close_state = 0 and
                                                                                                                  :rep_date between oks_date_begin and oks_date_end and
                                                                                                                  oks_delinq_date < :rep_date), 0) k_kolich_delinq_day,
              k_kolich_client_approved
      from kontstate, kontract
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date between ks_date_begin and ks_date_end and
            ks_kc_id = :rep_officer_id and
            ks_from_k_id = k_id)
  select sum(k_kolich_client_approved),

         sum(iif(aa_category_id = 1, k_kolich_client_approved, 0)),
         sum(iif(aa_category_id = 2, k_kolich_client_approved, 0)),
         sum(iif(aa_category_id = 3, k_kolich_client_approved, 0)),

         sum(iif(aa_category_id = 1, k_balance, 0)), sum(iif(aa_category_id = 2, k_balance, 0)),
         sum(iif(aa_category_id = 3, k_balance, 0)),

         sum(iif(aa_category_id = 1 and k_kolich_delinq_day > 7, k_balance, 0)),
         sum(iif(aa_category_id = 2 and k_kolich_delinq_day > 7, k_balance, 0)),
         sum(iif(aa_category_id = 3 and k_kolich_delinq_day > 7, k_balance, 0)),

         sum(iif(aa_category_id = 1 and k_kolich_delinq_day > 30, k_balance, 0)),
         sum(iif(aa_category_id = 2 and k_kolich_delinq_day > 30, k_balance, 0)),
         sum(iif(aa_category_id = 3 and k_kolich_delinq_day > 30, k_balance, 0)),

         sum(iif(aa_category_id = 1 and k_kolich_delinq_day > 60, k_balance, 0)),
         sum(iif(aa_category_id = 2 and k_kolich_delinq_day > 60, k_balance, 0)),
         sum(iif(aa_category_id = 3 and k_kolich_delinq_day > 60, k_balance, 0)),

         sum(k_balance), sum(iif(k_kolich_delinq_day > 7, k_balance, 0)),
         sum(iif(k_kolich_delinq_day > 30, k_balance, 0)), sum(iif(k_kolich_delinq_day > 60, k_balance, 0))

  from tabl1
  into own_kolich_cl, own_categ1_kolich_cl, own_categ2_kolich_cl, own_categ3_kolich_cl, own_categ1_balance,
       own_categ2_balance, own_categ3_balance, own_categ1_delinq_balance7, own_categ2_delinq_balance7,
       own_categ3_delinq_balance7, own_categ1_delinq_balance30, own_categ2_delinq_balance30,
       own_categ3_delinq_balance30, own_categ1_delinq_balance60, own_categ2_delinq_balance60,
       own_categ3_delinq_balance60, own_balance, own_delinq_balance7, own_delinq_balance30, own_delinq_balance60;

  own_categ1_par7 = 0;
  own_categ1_par30 = 0;
  own_categ1_par60 = 0;
  own_categ2_par7 = 0;
  own_categ2_par30 = 0;
  own_categ2_par60 = 0;
  own_categ3_par7 = 0;
  own_categ3_par30 = 0;
  own_categ3_par60 = 0;
  own_all_par7 = 0;
  own_all_par30 = 0;
  own_all_par60 = 0;

  own_categ1_doly_cl = 0;
  own_categ2_doly_cl = 0;
  own_categ3_doly_cl = 0;
  if (own_kolich_cl > 0) then
  begin
    own_categ1_doly_cl = own_categ1_kolich_cl / own_kolich_cl * 100;
    own_categ2_doly_cl = own_categ2_kolich_cl / own_kolich_cl * 100;
    own_categ3_doly_cl = own_categ3_kolich_cl / own_kolich_cl * 100;
  end

  if (own_categ1_balance > 0) then
  begin
    own_categ1_par7 = own_categ1_delinq_balance7 / own_categ1_balance * 100;
    own_categ1_par30 = own_categ1_delinq_balance30 / own_categ1_balance * 100;
    own_categ1_par60 = own_categ1_delinq_balance60 / own_categ1_balance * 100;
  end
  if (own_categ2_balance > 0) then
  begin
    own_categ2_par7 = own_categ2_delinq_balance7 / own_categ2_balance * 100;
    own_categ2_par30 = own_categ2_delinq_balance30 / own_categ2_balance * 100;
    own_categ2_par60 = own_categ2_delinq_balance60 / own_categ2_balance * 100;
  end
  if (own_categ3_balance > 0) then
  begin
    own_categ3_par7 = own_categ3_delinq_balance7 / own_categ3_balance * 100;
    own_categ3_par30 = own_categ3_delinq_balance30 / own_categ3_balance * 100;
    own_categ3_par60 = own_categ3_delinq_balance60 / own_categ3_balance * 100;
  end
  if (own_balance > 0) then
  begin
    own_all_par7 = own_delinq_balance7 / own_balance * 100;
    own_all_par30 = own_delinq_balance30 / own_balance * 100;
    own_all_par60 = own_delinq_balance60 / own_balance * 100;
  end

-- Вычисляет данные кредитов которые находится в текущем СВЗ
  with tabl1
  as (select (select aa_category_id
              from aarep_get_category_id(ks_from_k_id, :rep_date, null, null, null, null, null, null)) aa_category_id, iif(k_index_curr_code = :def_curr,
-- Расчет портфеля контракта в сводном эквиваленте
              (select oks_balance
               from oksummary
               where oks_from_k_id = k_id and
                     oks_close_state = 0 and
                     :rep_date between oks_date_begin and oks_date_end), iif(k_index_curr_rate >= :curr_rate,
              (select oks_balance
               from oksummary
               where oks_from_k_id = k_id and
                     oks_close_state = 0 and
                     :rep_date between oks_date_begin and oks_date_end) * k_index_curr_rate,
              (select oks_balance
               from oksummary
               where oks_from_k_id = k_id and
                     oks_close_state = 0 and
                     :rep_date between oks_date_begin and oks_date_end) * :curr_rate)) k_balance, coalesce((select cast(:rep_date as date) - oks_delinq_date
                                                                                                            from oksummary -- Расчет кол-во просроченных дней
                                                                                                            where oks_from_k_id = k_id and
                                                                                                                  oks_close_state = 0 and
                                                                                                                  :rep_date between oks_date_begin and oks_date_end and
                                                                                                                  oks_delinq_date < :rep_date), 0) k_kolich_delinq_day,
              k_kolich_client_approved
      from kontstate, kontract, knaktlist
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date between kal_date and kal_date_end and
            kal_new_kf_officer = :rep_officer_id)
  select sum(k_kolich_client_approved), sum(iif(aa_category_id = 1, k_kolich_client_approved, 0)),
         sum(iif(aa_category_id = 2, k_kolich_client_approved, 0)),
         sum(iif(aa_category_id = 3, k_kolich_client_approved, 0)),

         sum(iif(aa_category_id = 1, k_balance, 0)), sum(iif(aa_category_id = 2, k_balance, 0)),
         sum(iif(aa_category_id = 3, k_balance, 0)),

         sum(iif(aa_category_id = 1 and k_kolich_delinq_day > 7, k_balance, 0)),
         sum(iif(aa_category_id = 2 and k_kolich_delinq_day > 7, k_balance, 0)),
         sum(iif(aa_category_id = 3 and k_kolich_delinq_day > 7, k_balance, 0)),

         sum(iif(aa_category_id = 1 and k_kolich_delinq_day > 30, k_balance, 0)),
         sum(iif(aa_category_id = 2 and k_kolich_delinq_day > 30, k_balance, 0)),
         sum(iif(aa_category_id = 3 and k_kolich_delinq_day > 30, k_balance, 0)),

         sum(iif(aa_category_id = 1 and k_kolich_delinq_day > 60, k_balance, 0)),
         sum(iif(aa_category_id = 2 and k_kolich_delinq_day > 60, k_balance, 0)),
         sum(iif(aa_category_id = 3 and k_kolich_delinq_day > 60, k_balance, 0)),

         sum(k_balance), sum(iif(k_kolich_delinq_day > 7, k_balance, 0)),
         sum(iif(k_kolich_delinq_day > 30, k_balance, 0)), sum(iif(k_kolich_delinq_day > 60, k_balance, 0))
  from tabl1
  into kolich_cl, categ1_kolich_cl, categ2_kolich_cl, categ3_kolich_cl, categ1_balance, categ2_balance, categ3_balance,
