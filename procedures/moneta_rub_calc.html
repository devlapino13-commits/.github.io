CREATE PROCEDURE MONETA_RUB_CALC(
    AMOUNT_KGS D_CURRENCY,
    REP_DATE D_DATE = current_date)
RETURNS (
    AMOUNT_RUB D_CURRENCY)
AS
declare variable error_txt d_str_100;
declare variable moneta_commission d_currency;
declare variable rub_course d_currency;
declare variable rub d_currency;
declare variable rub_moneta d_currency;
begin
  amount_rub = 0;
  moneta_commission = 0;
  rub_course = 0;
  rub = 0;

  if (amount_kgs < 1) then
  begin
     exception e_message 'Минимальная сумма погашения 1 сом';
     exit;
  end

  select error_txt,
         moneta_commission,
         rub_course
  from moneta_get_params(:rep_date)
  into error_txt, -- текст ошибки
       moneta_commission, -- процент комиссии Монеты
       rub_course; -- курс рубля
  
  if (coalesce(error_txt, '') <> '') then
  begin
     exception e_message error_txt;
     exit;
  end

  if (rub_course = 0) then
  begin
    exception e_message 'Курс рубля не может быть равен нулю';
    exit;
  end

  -- сумма погашения в рублях после округления
  rub = round_currency_up(amount_kgs / rub_course);

  -- сумма комиссии Монеты в рублях после округления
  rub_moneta = round_currency_up(rub * moneta_commission / 100);

  -- полная сумма оплаты в рублях (сумма отправки в сервис Монеты)
  amount_rub = rub + rub_moneta;

  suspend;
end
