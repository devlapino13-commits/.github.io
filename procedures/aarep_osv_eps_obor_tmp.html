CREATE PROCEDURE AAREP_OSV_EPS_OBOR_TMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_DT_AB_ID ID_TYPE = null,
    REP_KT_AB_ID ID_TYPE = null,
    REP_FOLD_ACCOUNT INTEGER = 0)
RETURNS (
    DT_ACC_ID ID_TYPE,
    KT_ACC_ID ID_TYPE,
    OB_SUMMA CURR_TYPE,
    OB_SUMMA_VAL CURR_TYPE,
    OB_SUMMA_TAX_DT CURR_TYPE,
    OB_SUMMA_TAX_KT CURR_TYPE)
AS
declare variable has_filter integer;
begin
  has_filter = 0;
  if (exists(select *
             from temp_ref_account_rep
             rows 1)) then
    has_filter = 1;

  if (has_filter = 0) then
  begin
    if (rep_dt_ab_id is null) then
    begin
      for select iif(:rep_fold_account = 0 or a_dt.acc_object_kind is null, kob_acc_dt, f_get_first_acc_id(kob_acc_dt, a_dt.acc_code)),
                 iif(:rep_fold_account = 0 or a_kt.acc_object_kind is null, kob_acc_kt, f_get_first_acc_id(kob_acc_kt, a_kt.acc_code)),
                 sum(kob_summa) as ob_summa, sum(kob_summa_val) as ob_summa_val,
                 sum(kob_summa_tax_dt) as ob_summa_tax_dt, sum(kob_summa_tax_kt) as ob_summa_tax_kt
          from kn_operbook, ref_account a_dt, ref_account a_kt
          where kob_ol_date between :rep_date_begin and :rep_date_end and
                kob_ol_is_active = 1 and
                a_dt.acc_id = kob_acc_dt and
                a_kt.acc_id = kob_acc_kt
          group by 1, 2
          into dt_acc_id, kt_acc_id, ob_summa, ob_summa_val, ob_summa_tax_dt, ob_summa_tax_kt
      do
        suspend;
    end
    else
    begin
      for select iif(:rep_fold_account = 0 or a_dt.acc_object_kind is null, kob_acc_dt, f_get_first_acc_id(kob_acc_dt, a_dt.acc_code)),
                 iif(:rep_fold_account = 0 or a_kt.acc_object_kind is null, kob_acc_kt, f_get_first_acc_id(kob_acc_kt, a_kt.acc_code)),
                 sum(kob_summa) as ob_summa, sum(kob_summa_val) as ob_summa_val,
                 sum(kob_summa_tax_dt) as ob_summa_tax_dt, sum(kob_summa_tax_kt) as ob_summa_tax_kt
          from kn_operbook, ref_account a_dt, ref_account a_kt
          where kob_ol_date between :rep_date_begin and :rep_date_end and
                kob_ol_is_active = 1 and
                a_dt.acc_id = kob_acc_dt and
                a_kt.acc_id = kob_acc_kt and
                a_dt.acc_from_ab_id = :rep_dt_ab_id and
                a_kt.acc_from_ab_id = :rep_kt_ab_id
          group by 1, 2
          into dt_acc_id, kt_acc_id, ob_summa, ob_summa_val, ob_summa_tax_dt, ob_summa_tax_kt
      do
        suspend;
    end
  end
  else
  begin
    if (rep_dt_ab_id is null) then
    begin
      for with tab as (
         select distinct kob_id,
                iif(:rep_fold_account = 0 or a_dt.acc_object_kind is null, kob_acc_dt, f_get_first_acc_id(kob_acc_dt, a_dt.acc_code)) as dt_acc,
                iif(:rep_fold_account = 0 or a_kt.acc_object_kind is null, kob_acc_kt, f_get_first_acc_id(kob_acc_kt, a_kt.acc_code)) as kt_acc,
                kob_summa as ob_summa, kob_summa_val as ob_summa_val,
                kob_summa_tax_dt as ob_summa_tax_dt, kob_summa_tax_kt as ob_summa_tax_kt
          from kn_operbook, temp_ref_account_rep, ref_account a_dt, ref_account a_kt
          where kob_ol_date between :rep_date_begin and :rep_date_end and
                kob_ol_is_active = 1 and
                a_dt.acc_id = kob_acc_dt and
                a_kt.acc_id = kob_acc_kt and
                (kob_acc_dt = rep_acc_id or
                kob_acc_kt = rep_acc_id))
      select dt_acc, kt_acc, sum(ob_summa), sum(ob_summa_val), sum(ob_summa_tax_dt), sum(ob_summa_tax_kt)
      from tab
      group by 1, 2
      into dt_acc_id, kt_acc_id, ob_summa, ob_summa_val, ob_summa_tax_dt, ob_summa_tax_kt
      do
        suspend;

    end
    else
    begin
      for with tab as (
         select distinct kob_id,
                iif(:rep_fold_account = 0 or a_dt.acc_object_kind is null, kob_acc_dt, f_get_first_acc_id(kob_acc_dt, a_dt.acc_code)) as dt_acc,
                iif(:rep_fold_account = 0 or a_kt.acc_object_kind is null, kob_acc_kt, f_get_first_acc_id(kob_acc_kt, a_kt.acc_code)) as kt_acc,
                kob_summa as ob_summa, kob_summa_val as ob_summa_val,
                kob_summa_tax_dt as ob_summa_tax_dt, kob_summa_tax_kt as ob_summa_tax_kt
          from kn_operbook, temp_ref_account_rep, ref_account a_dt, ref_account a_kt
          where kob_ol_date between :rep_date_begin and :rep_date_end and
                kob_ol_is_active = 1 and
                (kob_acc_dt = rep_acc_id or
                kob_acc_kt = rep_acc_id) and
                a_dt.acc_id = kob_acc_dt and
                a_kt.acc_id = kob_acc_kt and
                a_dt.acc_from_ab_id = :rep_dt_ab_id and
                a_kt.acc_from_ab_id = :rep_kt_ab_id)
      select dt_acc, kt_acc, sum(ob_summa), sum(ob_summa_val), sum(ob_summa_tax_dt), sum(ob_summa_tax_kt)
      from tab
      group by 1, 2
      into dt_acc_id, kt_acc_id, ob_summa, ob_summa_val, ob_summa_tax_dt, ob_summa_tax_kt
      do
        suspend;
    end
  end
end
