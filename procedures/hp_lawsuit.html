CREATE PROCEDURE HP_LAWSUIT(
    K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    COMPANY_NAME D_STR_255,
    CURRENCY_NAME D_STR_20,
    K_KONTRACT_NO D_STR_20,
    K_KONTRACT_DATE D_DATE,
    K_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_PERCENT_EFFECT D_PERCENT,
    K_PERCENT_INPUT D_PERCENT,
    K_PERIOD D_INTEGER,
    CARD_NUMB D_STR_100,
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    C_BIRTH_DATE D_DATE,
    C_PASS_NO D_STR_CODE,
    C_PASS_DATE D_DATE,
    C_PASS_WHO D_STR_255,
    C_PASS_DATE_EXPIRE D_DATE,
    C_FULL_ADDRESS D_STR_255,
    C_CELL_NO D_STR_255,
    ZAD_OS D_CURRENCY,
    ZAD_PERC D_CURRENCY,
    ZAD_FINE D_CURRENCY,
    ZAD_ALL D_CURRENCY,
    DEBT_WRITEXEC D_CURRENCY,
    DEBT_WRITEXEC_ADD D_CURRENCY,
    DEBT_STATEDUTY D_CURRENCY,
    DELINQ_DAYS D_INTEGER,
    DATE_WRITEXEC D_DATE,
    DATE_WRITEXEC_CANCEL D_DATE,
    LAM_DELO_NO D_STR_100)
AS
declare variable c_id d_id;
declare variable lam_id d_integer;
begin
  rep_date = coalesce(rep_date, current_date);
  for
    select
      k_kontract_no,
      k_kontract_date,
      k_disb_date,
      k_amount_approved,
      round(k_percent_effect, 2),
      k_percent_input,
      k_period,
      c_id,
      c_name_fio,
      c_inn,
      c_birth_date,
      c_pass_no,
      c_pass_date,
      c_pass_who,
      c_pass_date_expire,
      adl_address_str,
      lam_id,
      lam_delo_no
    from kontract
      inner join knclient
        on kc_from_k_id = k_id
          and kc_amount_approved > 0
          and kc_status = 2
      inner join lawkredit
        on lak_from_k_id = kc_from_k_id
          and lak_from_c_id = kc_from_c_id
      inner join lawmain
        on lam_id = lak_from_lam_id
          and lam_is_active = 1
      inner join clients
        on c_id = kc_from_c_id
      left join addrclient
        on adc_from_c_id = c_id
          and adc_address_type = 1
      left join addrlist
        on adl_id = adc_from_adl_id
    where k_id = :k_id
  into
    :k_kontract_no,
    :k_kontract_date,
    :k_disb_date,
    :k_amount_approved,
    :k_percent_effect,
    :k_percent_input,
    :k_period,
    :c_id,
    :c_name_fio,
    :c_inn,
    :c_birth_date,
    :c_pass_no,
    :c_pass_date,
    :c_pass_who,
    :c_pass_date_expire,
    :c_full_address,
    :lam_id,
    :lam_delo_no
  do
  begin
    select first 1
      cm_company_full_name,
      cm_currency_name
    from company
    into
      :company_name,
      :currency_name;

    c_cell_no = (select cont_value
      from hp_contdata_get_cell_main(0, :c_id));

    card_numb = (select first 1 et_cardnumber
      from extsys_token
      where et_from_c_id = :c_id
        and et_is_actual = 1);

    select
      sum(zad_os),
      sum(sh_calc_perc),
      sum(sh_calc_fine),
      sum(zad_os + sh_calc_perc + sh_calc_fine)
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    into
      :zad_os,
      :zad_perc,
      :zad_fine,
      :zad_all;

    select
      debt_writexec,
      debt_stateduty
    from hp_debtamount_legalcosts(:k_id, :rep_date, 1)
    into
      :debt_writexec,
      :debt_stateduty;
    debt_writexec_add = debt_writexec + 2300;
    debt_stateduty = round(zad_all * 0.03, 0);

    delinq_days = (select rep_delinq_days
      from aarep_kont_delinq_status(:k_id, :rep_date));

    date_writexec = (select lad_date
      from lawdata
      where lad_from_lam_id = :lam_id
        and lad_from_laf_id = 7);
    date_writexec_cancel = (select lad_date
      from lawdata
      where lad_from_lam_id = :lam_id
        and lad_from_laf_id = 8);

    currency_name = 'теңге';

    suspend;
  end
end
