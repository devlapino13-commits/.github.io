CREATE PROCEDURE UXP$UXEXPORT_SRC_PREPARE(
    UFM_UUID D_UUID,
    EXISTING_EXP_ID D_INTEGER,
    EXISTING_EXP_INDEX D_INTEGER)
AS
declare variable exi_uuid d_uuid;
declare variable tmp_uuid d_uuid;
begin
  for select ux_us_uuid
  from ux$code, ux$form
  where ufm_uuid = :ufm_uuid and
        ux_uuid = ufm_ux_uuid
  into exi_uuid
  do
    execute procedure uxp$uxexpoitem_add(:existing_exp_id, :existing_exp_index, :exi_uuid, 1);

  for select ufr_us_uuid
  from ux$frame
  where ufr_owner = :ufm_uuid
  into exi_uuid
  do
    execute procedure uxp$uxexpoitem_add(:existing_exp_id, :existing_exp_index, :exi_uuid, 1);

  for select distinct us_uuid
  from ux$frame, ux$source
  where ufr_owner = :ufm_uuid and
        ufr_add_table_fields = us_code
  into exi_uuid
  do
    execute procedure uxp$uxexpoitem_add(:existing_exp_id, :existing_exp_index, :exi_uuid, 1);

  for select distinct ufd_us_uuid
  from ux$frdata, ux$frame
  where ufr_owner = :ufm_uuid and
        ufd_ufr_uuid = ufr_uuid
  into exi_uuid
  do
    execute procedure uxp$uxexpoitem_add(:existing_exp_id, :existing_exp_index, :exi_uuid, 1);

  for select distinct uf_owner
  from ux$frame, ux$frparam, ux$field
  where ufr_owner = :ufm_uuid and
        ufp_ufr_uuid = ufr_uuid and
        ufp_uf_uuid = uf_uuid
  into exi_uuid
  do
    execute procedure uxp$uxexpoitem_add(:existing_exp_id, :existing_exp_index, :exi_uuid, 1);

  for select distinct ufl_us_uuid
  from ux$frparam, ux$frame, ux$frlookup
  where ufr_owner = :ufm_uuid and
        ufp_ufr_uuid = ufr_uuid and
        ufl_ufp_uuid = ufp_uuid
  into exi_uuid
  do
    execute procedure uxp$uxexpoitem_add(:existing_exp_id, :existing_exp_index, :exi_uuid, 1);

  for select distinct us_uuid
  from ux$frcombo, ux$frparam, ux$frame, ux$source
  where ufr_owner = :ufm_uuid and
        ufp_ufr_uuid = ufr_uuid and
        ufc_ufp_uuid = ufp_uuid and
        ufc_combo_source = us_code
  into exi_uuid
  do
    execute procedure uxp$uxexpoitem_add(:existing_exp_id, :existing_exp_index, :exi_uuid, 1);

  for select distinct s2.us_uuid
  from ux$expoitem, ux$source s1, ux$source s2
  where exi_exp_id = :existing_exp_id and
        exi_ux_uuid = s1.us_uuid and
        s1.us_main_table = s2.us_code
  into :exi_uuid
  do
    execute procedure uxp$uxexpoitem_add(:existing_exp_id, :existing_exp_index, :exi_uuid, 1);

  for select exi_ux_uuid
  from ux$expoitem, ux$source
  where exi_exp_id = :existing_exp_id and
        exi_ux_uuid = us_uuid
  into :tmp_uuid
  do
  begin
    for select uf_uuid
    from ux$field
    where uf_owner = :tmp_uuid
    order by uf_is_param, uf_position
    into exi_uuid
    do
      execute procedure uxp$uxexpoitem_add(:existing_exp_id, :existing_exp_index, :exi_uuid, 1);
  end
end
