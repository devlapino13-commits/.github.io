CREATE PROCEDURE MC_FIND_CLIENT_BY_PHONE(
    PHONE_NUMBER D_STR_20,
    IS_MAIN D_BOOL = 1)
RETURNS (
    C_ID D_ID,
    INN D_STR_20)
AS
begin
  -- RFCKG-4165 поиск клиента по верифицированному номеру телефона
  -- если клиент не найден тогда c_id = 0, inn = ''
  -- если is_main = 1 то поиск идет только по основному номеру
  inn = '';
  c_id = 0;

  if (coalesce(phone_number, '') = '' ) then
    exception e_message('Не указан номер телефона');

  select coalesce(c.c_id, 0),
         coalesce(c.c_inn, '')
  from contdata cda
  inner join contlist clt on clt.clt_id = cda.cda_from_clt_id
  inner join clients c on c.c_id = cda.cda_from_c_id
  where clt.clt_code = :phone_number
    and cda.cda_is_delete = 0
    and (:is_main <> 1 or cda.cda_is_main = 1)
    and cda.cda_status = 1
    and cda.cda_cont_kind in (-1, 0)
  order by cda.cda_cont_kind
  rows 1
  into c_id, inn;

  suspend;
end
