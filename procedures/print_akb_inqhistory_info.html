CREATE PROCEDURE PRINT_AKB_INQHISTORY_INFO(
    KBM_ID D_ID)
RETURNS (
    PURPOSEDESCRIPTION D_STR_255,
    I_IH D_INTEGER,
    INQBANKNAME D_STR_255,
    INQDATE D_DATETIME)
AS
declare variable CHECKED_DATE D_DATE;
declare variable JSON_TEXT D_BLOB_BINARY;
declare variable INQHISTORYCOUNT integer;
declare variable INQHISTORYITEM blob sub_type 0 segment size 80;
declare variable INQPURPOSEID integer;
declare variable INQDATE_STR varchar(30);
declare variable DB_ID D_ID;
begin
  db_id = f_get_db_id();
  json_text = '';
  select bls.bls_blob
  from extinfolist eil
  inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
  inner join kibminiapp kbm on eil.eil_object_id = kbm.kbm_id
  inner join blobstore bls on eil.eil_from_bls_id = bls.bls_id
  where eil.eil_status = 2 and
        kbm.kbm_ver_info = 3 and
        kbm.kbm_id = :kbm_id and
        eic.eic_code = 'akb'
  order by kbm.kbm_checked_datetime desc
  rows 1
  into :json_text;

  if (json_text <> '' or json_text <> '{}' or json_text is not null) then
  begin
  inqhistorycount = f_json_blob_get_count(:json_text, 'inquiryHistory.inquiryHistoryItem') - 1;

        i_ih = 0;
        while (i_ih <= inqhistorycount) do
        begin
          inqhistoryitem = f_json_blob_get_value(:json_text, 'inquiryHistory.inquiryHistoryItem[' || :i_ih || ']', '');

          inqbankname = f_json_blob_get_value(:inqhistoryitem, 'inqBankName', '');
          inqdate_str = f_json_blob_get_value(:inqhistoryitem, 'inqDate', '');
          inqdate_str = substring(inqdate_str from 1 for 19);
          inqdate_str = replace(inqdate_str, 'T', ' ');
          inqdate = cast(inqdate_str as d_datetime);
          inqpurposeid = f_json_blob_get_value(:inqhistoryitem, 'inqPurposeId', '');

          purposedescription = case
                             when inqpurposeid = '001' then iif(db_id <> 94401, 'Обращение по кредиту', 'Kredit müraciəti')
                             when inqpurposeid = '002' then iif(db_id <> 94401, 'Обращение по кредитной линии', 'Kredit xətti ücün müraciət')
                             when inqpurposeid = '003' then iif(db_id <> 94401, 'Запрос по поручителю', 'Zaminin sorğulanması')
                             when inqpurposeid = '004' then iif(db_id <> 94401, 'Запрос по основателю', 'Təsisçinin sorğulanması')
                             when inqpurposeid = '005' then iif(db_id <> 94401, 'Запрос по текущему заемщику', 'Cari borcalanın sorğulanması')
                           end;
          i_ih = i_ih + 1;
          suspend;
        end
     end
end
