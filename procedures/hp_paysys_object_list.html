CREATE PROCEDURE HP_PAYSYS_OBJECT_LIST(
    OBJ_ID INTEGER)
RETURNS (
    RES_ID DECIMAL(18,0),
    RES_NAME VARCHAR(255),
    RES_IS_CARD INTEGER)
AS
begin
  if (obj_id = 0) then -- Проекты платежной системы
  begin
    for select psp_id, psp_name, psp_is_card
        from paysys_project
        where psp_is_active = 1
        order by psp_id
        into :res_id, :res_name, res_is_card
    do
    begin
      suspend;
    end
  end
  if (obj_id = 1) then -- Операционные офис
  begin
    for select reg_unicode, reg_name || ' (' || reg_code || ')'
        from region
        where reg_status = 3
        into :res_id, :res_name
    do
    begin
      suspend;
    end
  end
  if (obj_id = 2) then -- Сотрудник
  begin
    for with clients_list
        as (select psm_from_c_source as client_id
            from paysys_main
            where psm_from_c_source is not null
            union
            select psm_from_c_destin as client_id
            from paysys_main
            where psm_from_c_destin is not null),
        staff_list
        as (select distinct ps_from_c_id as staff_id
            from hr_pstatus, clients_list
            where client_id = ps_from_c_id and
                  ps_is_working = 1)
        select c_id, c_name_fio
        from clients, staff_list
        where staff_id = c_id
        into :res_id, :res_name
    do
    begin
      suspend;
    end
  end
end
