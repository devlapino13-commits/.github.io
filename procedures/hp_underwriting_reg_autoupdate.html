CREATE PROCEDURE HP_UNDERWRITING_REG_AUTOUPDATE
AS
declare variable ucf_id d_integer;
declare variable ucf_system d_smallint;
begin
  ucf_system = f_enum_id('access_sytem', 'undervriter_certif');

  /*Суббота вкл*/
  if (extract(weekday from current_date) = 6) then
  begin
    for select ucf.ucf_id
        from underwritingcertif ucf
        inner join sa_office_working_days sowd on ucf.ucf_from_reg_id = sowd.sowd_from_reg_id and
              sowd.sowd_days > 5 and
              current_date between sowd.sowd_date_begin and sowd.sowd_date_end
        where ucf.ucf_active = 0 and
              current_date between ucf.ucf_date_begin and ucf.ucf_date_end and
              ucf.ucf_system = :ucf_system
        into :ucf_id
    do
    begin
      update underwritingcertif
      set ucf_active = 1,
          ucf_from_un_id = 1000000000000
      where (ucf_id = :ucf_id);
    end
  end
  /*Понедельник откл*/
  if (extract(weekday from current_date) = 1) then
  begin
    for select ucf.ucf_id
        from underwritingcertif ucf
        inner join sa_office_working_days sowd on ucf.ucf_from_reg_id = sowd.sowd_from_reg_id and
              sowd.sowd_days > 5 and
              current_date between sowd.sowd_date_begin and sowd.sowd_date_end
        where ucf.ucf_active = 1 and
              current_date between ucf.ucf_date_begin and ucf.ucf_date_end and
              ucf.ucf_system = :ucf_system and
              ucf.ucf_from_un_id = 1000000000000 and
              cast(ucf.ucf_change_time as date) = current_date - 2
        into :ucf_id
    do
    begin
      update underwritingcertif
      set ucf_active = 0,
          ucf_from_un_id = 1000000000000
      where (ucf_id = :ucf_id);
    end
  end
end
