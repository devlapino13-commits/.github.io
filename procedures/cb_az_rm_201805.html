CREATE PROCEDURE CB_AZ_RM_201805(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    AD_REPAY_PERC CURR_TYPE,
    AD_REPAY_FINE CURR_TYPE,
    AD_REPAY_KOMMIS CURR_TYPE,
    AD_RETURN_PLAN_OS CURR_TYPE,
    AD_RETURN_FACT_OS CURR_TYPE,
    REPAY_DELINQ CURR_TYPE,
    SUM_VZYSKAN CURR_TYPE,
    KC6_300 INTEGER,
    KC6 INTEGER,
    Y_SUM CURR_TYPE)
AS
declare variable rep_date_begin date;
declare variable res_period_id integer;
declare variable reg_id integer;
declare variable officer_id id_type;
declare variable aarep_bonus_total_svz curr_type;
declare variable c_hired_date date;
begin
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;

  execute procedure hp_analitica_create_period(:rep_region_id, :rep_date_begin, :rep_date_end, 1, null, null, 30, 0,
      'standart')
      returning_values :res_period_id;

  if (res_period_id is null) then
  begin
--     res_text='Не удалось расчитать данные за период';
--     Suspend;
    exit;
  end

  for select res_field_1, res_field_2, sum(ad_repay_perc), sum(ad_repay_fine), sum(ad_repay_kommis),
             sum(ad_return_plan_os), sum(ad_return_fact_os)
      from hp_analitica_report('reg_id, reg_name', 'tree_group', 8, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 360, 0, 'standart')
      group by 1, 2
      having sum(ad_active_balance) > 0.0
      into reg_id, reg_name, ad_repay_perc, ad_repay_fine, ad_repay_kommis, ad_return_plan_os, ad_return_fact_os
  do
  begin

    repay_delinq = null;
    sum_vzyskan = null;
    select -sum(iif(ok_sh_date >= '1.5.2016', ok_os, 0)), -sum(iif(ok_sh_date < '1.5.2016', ok_os, 0))
    from operkred, kontract, knaktlist
    where ok_ol_is_active = 1 and
          ok_ol_date between :rep_date_begin and :rep_date_end and
          ok_sh_date < :rep_date_begin and
          ok_type = 4 and
          ok_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          kal_new_reg_id = :reg_id and
          :rep_date_end between kal_date and kal_date_end
    into repay_delinq, sum_vzyskan;
    if (repay_delinq is null) then
      repay_delinq = 0;
    if (sum_vzyskan is null) then
      sum_vzyskan = 0;

    kc6_300 = 0;
    for select c_id, aarep_bonus_total
        from cb_az_lo_201602_one_lo(:reg_id, 0, :rep_date_end, 0)
        into officer_id, :aarep_bonus_total_svz
    do
    begin
      if (aarep_bonus_total_svz > 300.0) then
      begin
        c_hired_date = null;
        select c_hired_date
        from clients
        where c_id = :officer_id
        into c_hired_date;
        if (c_hired_date is not null) then
        begin
          if (c_hired_date <= dateadd(month, -6, :rep_date_end)) then
          begin
            kc6_300 = kc6_300 + 1;
          end
        end
      end
    end

    kc6 = 0;
    select count(*)
    from hr_pstatus, joblist, jobkind, clients
    where ps_status = 1 and
          ps_is_working = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code = 'JobSystem.LO_Standart' and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_reg_id = :reg_id and
          ps_from_c_id = c_id and
          c_hired_date <= dateadd(month, -6, :rep_date_end) and
          exists(select *
                 from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                 where is_decret = 0)
    into kc6;

    y_sum = case
              when sum_vzyskan > 10000.0 then sum_vzyskan * 5 / 100
              when sum_vzyskan > 5000.0 then sum_vzyskan * 4 / 100
              when sum_vzyskan >= 1000.0 then sum_vzyskan * 3 / 100
              else 0
            end;

    suspend;
  end
end
