CREATE PROCEDURE AAREP_GET_TEAM_OFFICE(
    TEAM_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    RES_REGION_NAME VARCHAR(255))
AS
declare variable rd_id integer;
declare variable c_id decimal(18,0);
declare variable reg_name varchar(100);
begin
  res_region_name = '';
  if (coalesce(team_id, 0) = 0) then
  begin
    suspend;
    exit;
  end
  rd_id = (select rd_id
           from refdata
           where rd_code = 'objectprops.staffteam');
  if (rd_id is null) then
  begin
    suspend;
    exit;
  end

  if (exists(select rd_id
             from refdata
             where rd_id = :team_id and
                   (rd_code containing 'StaffTeamList.GoldChamp' or
                   rd_code containing 'StaffTeamList.SilverChamp'))) then
  begin
    suspend;
    exit;
  end

  for select opr_object_id
      from objectprops, objectvalues, objectvalres
      where opr_id = ovs_from_opr_id and
            opr_object_kind = 0 and
            opr_subject_kind = -1 and
            opr_subject_id = :rd_id and
            ovr_from_opr_id = ovs_from_opr_id and
            ovr_from_ovs_id = ovs_id and
            ovr_result_rd_id = :team_id and
            ovs_date_begin <= :rep_date and
            ovs_date_end >= :rep_date
      into :c_id
  do
  begin
    reg_name = '';
    select reg_code || ' ' || reg_name
    from hr_pstatus, region
    where ps_from_c_id = :c_id and
          ps_date_begin <= :rep_date and
          ps_date_end >= :rep_date and
          ps_is_working = 1 and
          ps_status = 1 and
          ps_from_reg_id = reg_unicode
    into :reg_name;

    if (position(reg_name, res_region_name) <= 0) then
    begin
      if (res_region_name = '') then
        res_region_name = reg_name;
      else
        res_region_name = res_region_name || ', ' || reg_name;
    end
  end
  suspend;
end
