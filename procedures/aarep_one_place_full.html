CREATE PROCEDURE AAREP_ONE_PLACE_FULL(
    USER_ID DECIMAL(18,0))
RETURNS (
    PLACE_ID DECIMAL(18,0),
    PLACE_FULL_PATH VARCHAR(512),
    PLACE_ACCESS_KOLICH INTEGER,
    PLACE_SHORT_PATH VARCHAR(255),
    PLACE_NAME VARCHAR(255))
AS
declare variable str varchar(255);
declare variable p_id decimal(18,0);
begin
  /* Procedure Text */
  for select ups_from_pl_id, ups_access_kolich
      from userplacestat
      where ups_from_un_id = :user_id
      into :place_id, :place_access_kolich
  do
  begin
    if (place_access_kolich is null) then
      place_access_kolich = 0;
    place_full_path = '';
    place_name = '';
    place_short_path = '';
    for with recursive pl_list
        as (select p1.pl_id, p1.parent, p1.pl_status, p1.pl_name
            from places p1
            where p1.pl_id = :place_id and
                  p1.is_delete = 0
            union all
            select p2.pl_id, p2.parent, p2.pl_status, p2.pl_name
            from places p2
            join pl_list on (pl_list.parent = p2.pl_id)
            where p2.is_delete = 0)
        select pl_id, pl_name
        from pl_list, placetype
        where pl_status = plt_id and
              plt_show_in_address = 1
        into :p_id, :str
    do
    begin
      if (place_full_path = '') then
        place_full_path = str;
      else
        place_full_path = place_full_path || ' ' || str;
      if (p_id <> :place_id) then
      begin
        if (place_short_path = '') then
          place_short_path = str;
        else
          place_short_path = place_short_path || ' ' || str;
      end
      else
        place_name = :str;
    end
    suspend;
  end
end
