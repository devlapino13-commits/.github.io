CREATE PROCEDURE FS_REPORT_PARAM_LIST(
    UN_ID D_ID,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    REP_ID D_STR_255,
    REP_OBJ_KIND D_STR_CODE = '',
    REP_OBJ_ID D_ID = 0,
    FIND_TEXT D_STR_100 = '')
RETURNS (
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID,
    TEXT_1 D_STR_255,
    TEXT_2 D_STR_255,
    TEXT_3 D_STR_255,
    TEXT_4 D_STR_255,
    REP_SQL D_STR_8100)
AS
declare variable sql_html d_str_8100;
begin
  -- Процедура возвращает результат для FinStart
  sql_html = 'select * from fm_report_html_main(:un_id, :rep_id, :obj_kind, :obj_id, :date_begin, :date_end, '''', 0, :un_ip)';

  for select obj_kind, obj_id, text_1, text_2, text_3, text_4
  from fm_report_param_list(:un_id, :date_begin, :date_end, :rep_id, :rep_obj_kind, :rep_obj_id, :find_text)
  into :obj_kind, :obj_id, :text_1, :text_2, :text_3, :text_4
  do
  begin
    rep_sql = sql_html;

    if (obj_kind is null) then
      obj_kind = '';

    if (obj_id is null) then
      obj_id = 0;
  
    rep_sql = replace(rep_sql, ':un_id', un_id);
    rep_sql = replace(rep_sql, ':rep_id', '''' || rep_id || '''');
    rep_sql = replace(rep_sql, ':obj_kind', '''' || obj_kind || '''');
    rep_sql = replace(rep_sql, ':obj_id', obj_id);
    rep_sql = replace(rep_sql, ':date_begin', '''' || f_datetostr(date_begin) || '''');
    rep_sql = replace(rep_sql, ':date_end', '''' || f_datetostr(date_end) || '''');

    suspend;
  end
end
