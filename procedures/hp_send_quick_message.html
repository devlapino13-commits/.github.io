CREATE PROCEDURE HP_SEND_QUICK_MESSAGE(
    SEND_KIND VARCHAR(20),
    MAIL_TO VARCHAR(255),
    MAIL_SUBJECT VARCHAR(255),
    MAIL_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    COMMAND_KIND VARCHAR(255),
    LIFETIME_DAYS INTEGER,
    UN_ID DECIMAL(18,0))
AS
declare variable msg_object_kind integer;
declare variable msg_object_id decimal(18,0);
declare variable msg_type integer;
begin
-- SMS или EMAIL отправляется мгновенно, без ожидания расписания из fsrobot_sh_main

  if (send_kind = 'email') then
  begin
    msg_type = 4;

    select res_object_kind, res_object_id
    from hp_contdata_find_object(:mail_to, 'email')
    rows 1
    into :msg_object_kind, :msg_object_id;

    if (lifetime_days is null) then
      lifetime_days = 7;
    if (coalesce(command_kind, '') = '') then
      command_kind = 'quick email';
  end
  if (send_kind = 'sms') then
  begin
    msg_type = 3;
    select res_object_kind, res_object_id
    from hp_contdata_find_object(:mail_to, 'cell')
    rows 1
    into :msg_object_kind, :msg_object_id;

    if (lifetime_days is null) then
      lifetime_days = 7;
    if (coalesce(command_kind, '') = '') then
      command_kind = 'quick sms';
    if (coalesce(mail_subject, '') = '') then
      mail_subject = command_kind;
  end
  if (msg_type is not null) then
  begin
    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                msg_command_kind, msg_object_kind, msg_object_id)
    values (:msg_type, :mail_to, :mail_subject, :mail_body, :lifetime_days, :un_id, :command_kind, :msg_object_kind,
            :msg_object_id);
  end
end
