CREATE PROCEDURE HP_SUD_GET_AVIABLE_DOCS(
    REQUEST_ID D_ID,
    ONLY_ACTUAL_ON_DISB D_BOOL,
    UN_ID D_ID)
RETURNS (
    REC_ID D_INTEGER,
    DOC_KIND D_STR_100,
    DOC_NAME D_STR_255,
    DOC_MEMO D_STR_LONG,
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    BLS_ID D_ID,
    DOC_CREATE_DATETIME D_DATETIME,
    DOC_DELETE_DATE D_DATE,
    DOC_ADDED D_BOOL,
    DOC_ACTUAL_ON_DISB D_BOOL)
AS
declare variable edl_obj_kind d_integer;
declare variable edl_obj_id d_id;
declare variable k_id d_id;
declare variable cds_doc_blob_1 d_id;
declare variable cds_doc_blob_2 d_id;
declare variable tmp_memo d_str_long;
declare variable disb_datetime d_datetime;
declare variable orig_actual_on_disb d_bool;
begin
  doc_memo = '';
  rec_id = 0;

  select smr_from_k_id
  from sud_mainreq
  where smr_id = :request_id
  into k_id;

  select ol_datetime
  from kontstate, operlist
  where ks_from_ol_id = ol_id and
        ks_from_k_id = :k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0
  into disb_datetime;

  edl_obj_kind = f_objkind('kontract', 'id');

  for select k_id, f_objkind('kontract', 'id')
  from kontract
  where k_id = :k_id
  union
  select kc_id, f_objkind('knclient', 'id')
  from knclient
  where kc_from_k_id = :k_id
  union
  select kc_from_c_id, f_objkind('clients', 'id')
  from knclient
  where kc_from_k_id = :k_id
  into edl_obj_id, edl_obj_kind
  do
  begin
    doc_name = f_objname(edl_obj_kind, edl_obj_id);

    orig_actual_on_disb = 1;
    if (edl_obj_kind = 0) then
      orig_actual_on_disb = 0;

    obj_kind = f_objkind('edoc_blank', 'id');

    for select edf_print_name,
              'Эл.документ от ' || f_datetostr(edl_blank_date),
              edb_id,
              coalesce(edb_scan_bls_id, edb_from_bls_id),
              edl_create_datetime,
              edl_date_end
    from edoc_list, edoc_blank, edoc_frx
    where edb_from_edl_id = edl_id and
          edb_from_edf_id = edf_id and
          edb_from_c_id is null and
          edl_object_kind = :edl_obj_kind and
          edl_object_id = :edl_obj_id and
          edl_status = 1 and
          coalesce(edb_scan_bls_id, edb_from_bls_id) is not null
    into doc_kind, doc_memo, obj_id, bls_id, doc_create_datetime, doc_delete_date
    do
    begin
      doc_actual_on_disb = orig_actual_on_disb;
      if (disb_datetime >= doc_create_datetime and
          disb_datetime <= coalesce(doc_delete_date, '31.12.2999')) then
        doc_actual_on_disb = 1;

      if (only_actual_on_disb <= doc_actual_on_disb) then
      begin
        rec_id = rec_id + 1;

        doc_added = 0;
        if (exists(select * from sud_loanbls
        where slb_from_bls_id = :bls_id and
              slb_object_kind = :obj_kind and
              slb_object_id = :obj_id and
              slb_is_delete = 0)) then
          doc_added = 1;
  
        suspend;
      end
    end

    doc_create_datetime = null;
    if (edl_obj_kind = 0) then
    begin
      obj_kind = f_objkind('clientdocs', 'id');

      for select pk_name,
                 trim(iif(trim(coalesce(cds_doc_no, '')) <> '', '№ ' || cds_doc_no, '') ||
                 iif(cds_doc_date_begin is not null, ' от ' || f_datetostr(cds_doc_date_begin), '') ||
                 iif(cds_doc_date_end is not null, ' по ' || f_datetostr(cds_doc_date_end), '')),
                 cds_id, cds_doc_blob_1, cds_doc_blob_2,
                 cds_create_datetime,
                 cds_delete_datetime
      from clientdocs, passkind
      where cds_from_pk_id = pk_id and
            cds_from_c_id = :edl_obj_id and
            (coalesce(cds_doc_blob_1, 0) > 0 or
             coalesce(cds_doc_blob_2, 0) > 0)
      into doc_kind, tmp_memo, obj_id, cds_doc_blob_1, cds_doc_blob_2, doc_create_datetime, doc_delete_date
      do
      begin
        doc_actual_on_disb = orig_actual_on_disb;
        if (disb_datetime >= doc_create_datetime and
            disb_datetime <= coalesce(doc_delete_date, '31.12.2999')) then
          doc_actual_on_disb = 1;

        if (only_actual_on_disb <= doc_actual_on_disb) then
        begin
          if (coalesce(cds_doc_blob_1, 0) > 0) then
          begin
            bls_id = cds_doc_blob_1;
            rec_id = rec_id + 1;
  
            doc_added = 0;
            if (exists(select * from sud_loanbls
            where slb_from_bls_id = :bls_id and
                  slb_object_kind = :obj_kind and
                  slb_object_id = :obj_id and
                  slb_is_delete = 0)) then
              doc_added = 1;
  
            doc_memo = tmp_memo;
            if (coalesce(cds_doc_blob_1, 0) > 0 and
                coalesce(cds_doc_blob_2, 0) > 0 and
                coalesce(cds_doc_blob_2, 0) <> coalesce(cds_doc_blob_1, 0)) then
              doc_memo = 'стр.1 ' || doc_memo;
  
            suspend;
          end
          if (coalesce(cds_doc_blob_2, 0) > 0 and
              coalesce(cds_doc_blob_2, 0) <> coalesce(cds_doc_blob_1, 0)) then
          begin
            bls_id = cds_doc_blob_2;
            rec_id = rec_id + 1;
  
            doc_added = 0;
            if (exists(select * from sud_loanbls
            where slb_from_bls_id = :bls_id and
                  slb_object_kind = :obj_kind and
                  slb_object_id = :obj_id and
                  slb_is_delete = 0)) then
              doc_added = 1;
  
            doc_memo = tmp_memo;
            if (coalesce(cds_doc_blob_1, 0) > 0 and
                coalesce(cds_doc_blob_2, 0) > 0 and
                coalesce(cds_doc_blob_2, 0) <> coalesce(cds_doc_blob_1, 0)) then
              doc_memo = 'стр.2 ' || doc_memo;
  
            suspend;
          end
        end
      end
    end
  end
end
