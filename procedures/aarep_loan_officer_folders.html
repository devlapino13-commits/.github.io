CREATE PROCEDURE AAREP_LOAN_OFFICER_FOLDERS(
    REP_KC_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    KF_ID ID_TYPE,
    KF_NAME VARCHAR(30),
    K_BALANCE CURR_TYPE,
    K_DELINQ_BALANCE CURR_TYPE,
    DELINQ_DAY INTEGER,
    KOL_CLI INTEGER,
    KOL_KONTRACT INTEGER,
    DELINQ_PERIOD SMALLINT)
AS
begin
  for select kal_from_kf_id, kal_new_kf_name
      from knaktlist
      where knaktlist.is_delete = 0 and
            :rep_date between kal_date and kal_date_end and
            kal_new_kf_officer = :rep_kc_id
      into kf_id, kf_name
  do
  begin
    select sum(oks_balance), sum(iif(:rep_date > oks_delinq_date, oks_balance, 0)),
           max(iif(:rep_date - oks_delinq_date < 0, 0, :rep_date - oks_delinq_date)), count(distinct k_id)
    from kontract, oksummary
    where k_from_kf_id = :kf_id and
          k_id = oks_from_k_id and
          oks_close_state = 0 and
          :rep_date between oks_date_begin and oks_date_end
    into k_balance, k_delinq_balance, delinq_day, kol_kontract;

    if (k_balance is not null) then
    begin
      select count(distinct kc_from_c_id)
      from kontract, oksummary, knclient
      where k_from_kf_id = :kf_id and
            kc_from_k_id = k_id and
            oks_from_k_id = k_id and
            oks_close_state = 0 and
            :rep_date between oks_date_begin and oks_date_end
      into kol_cli;

      delinq_period = case
                        when delinq_day between 1 and 8 then 1
                        when delinq_day between 8 and 30 then 2
                        when delinq_day > 30 then 3
                        else 0
                      end;
      suspend;
    end
  end
end
