CREATE PROCEDURE HP_IMESSAGE_CREATE(
    OL_ID D_ID,
    PLAN_DATE D_DATE = current_date,
    PLAN_REPAY CURR_TYPE = 0,
    IM_TYPE D_INTEGER = 0,
    IN_UN_ID D_ID = null,
    IN_K_ID D_ID = null)
RETURNS (
    IM_ID D_ID)
AS
declare variable ol_date d_date;
declare variable reg_id d_integer;
declare variable cl_name d_str_255;
declare variable un_id d_id;
declare variable k_id d_id;
declare variable actual_date d_date;
declare variable actual_repay curr_type;
declare variable amount_payment curr_type;
begin

  im_id = 0;
  reg_id = (select ol_from_reg_id
    from operlist where ol_id = :ol_id);
  select first 1
    ol_id,
    ol_date,
    ol_doc_client_name,
    ol_user_id
  from operlist
  where (ol_id = :ol_id
    or ol_parent_id = :ol_id)
    and ol_oper_type = 1
    and ol_oper_subtype = 2
    and is_delete = 0
    and ol_doc_client_name <> ''
  into
    :ol_id,
    :ol_date,
    :cl_name,
    :un_id;

  if (ol_id > 0 or
     (im_type = 1 and coalesce(:in_k_id, 0) > 0)) then begin
    actual_date = ol_date;
    select
      max(ok_from_k_id),
      -sum(ok_os + ok_perc + ok_fine + ok_kommis + ok_kommis_adv)
    from operkred
    where ok_from_ol_id = :ol_id
    into
      :k_id,
      :actual_repay;

    amount_payment = 0;
    if (actual_date < plan_date) then
    begin
      if (actual_repay >= plan_repay) then
        amount_payment = actual_repay;
      -- в случае полного досрочного погашения сумма plan_repay может оказаться больше чем actual_repay
      -- поэтому проверим через oksummary закрыт ли кредит
      else if (not exists(select * from oksummary oks
                          where oks.oks_from_k_id = :k_id
                            and :actual_date between oks.oks_date_begin and oks.oks_date_end
      )) then
        amount_payment = actual_repay;
    end
    if (actual_date = plan_date and actual_repay > plan_repay )
      then amount_payment = actual_repay - plan_repay;

    if (amount_payment > 0) then
    begin
      insert into internal_message(
        im_from_ol_id, im_from_k_id, im_from_reg_id, im_through_whom, im_amount_payment, im_cashier_id, im_type)
      values(:ol_id, :k_id, :reg_id, :cl_name, :amount_payment, :un_id, :im_type)
      returning im_id into im_id;
    end
    else if (im_type = 1) then
    begin
      if (un_id is null) then
        un_id = in_un_id;
      if (k_id is null) then
        k_id = in_k_id;
      if (coalesce(amount_payment, 0) = 0 or coalesce(reg_id, 0) = 0 ) then
      begin
        select k.k_amount_approved, k.k_last_reg_id
        from kontract k
        where k.k_id = :k_id
        into amount_payment, reg_id;
      end

      insert into internal_message(
        im_from_ol_id, im_from_k_id, im_from_reg_id, im_through_whom, im_amount_payment, im_cashier_id, im_type)
      values(:ol_id, :k_id, :reg_id, coalesce(:cl_name, ''), :amount_payment, :un_id, :im_type)
      returning im_id into im_id;
    end
  end
  suspend;
end
