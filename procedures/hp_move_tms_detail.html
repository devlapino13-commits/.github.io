CREATE PROCEDURE HP_MOVE_TMS_DETAIL(
    REP_DATE DATE,
    OL_ID_FIRST DECIMAL(18,0),
    OL_ID_LAST DECIMAL(18,0),
    REG_FROM INTEGER,
    REG_TO INTEGER,
    WP_FROM DECIMAL(18,0),
    WP_TO DECIMAL(18,0),
    MO_FROM DECIMAL(18,0),
    MO_TO DECIMAL(18,0))
RETURNS (
    OM_ACCOUNT VARCHAR(20),
    OM_AMOUNT_TMS DECIMAL(18,4),
    OM_AMOUNT_AMORT DECIMAL(18,4))
AS
declare variable om_bill_id decimal(18,0);
declare variable om_bill_line_no integer;
declare variable om_matotvet_id decimal(18,0);
declare variable om_kolich_tms decimal(18,4);
declare variable ol_number integer;
declare variable acc1 varchar(20);
declare variable acc2 varchar(20);
declare variable acc_prior varchar(20);
declare variable reg_loronostro_head integer;
declare variable mc_id decimal(18,0);
declare variable cm_default_currency varchar(3);
declare variable acc_amort varchar(20);
declare variable om_os decimal(18,4);
declare variable ol_year integer;
declare variable ol_month integer;
declare variable om_account_to varchar(20);
declare variable om_workplace_id decimal(18,0);
declare variable mo_tmp decimal(18,0);
declare variable wp_tmp decimal(18,0);
begin
  om_amount_tms = 0;
  om_amount_amort = 0;
  ol_number = 1;
  select cm_default_currency
  from company
  into :cm_default_currency;
  if (ol_id_last = 0) then
    ol_id_last = ol_id_first;

  /*
  REP_DATE='02.07.2013';
  WP_From=1010100000007;
  WP_TO=1090100000038;
  OL_ID_First=1090100001883813;
  OL_ID_Last=1090100001883814;
  REG_FROM=1010100;
  */

  ol_year = extract(year from :rep_date);
  ol_month = extract(month from :rep_date);

  select reg_loronostro_head
  from region
  where reg_unicode = :reg_from
  into :reg_loronostro_head;
  acc_prior = '';
-- Счет куда улетает (Лороностро передающего)
  acc1 = null;
  acc2 = null;
  if (reg_from <> reg_to) then
  begin
    if (reg_from <> reg_loronostro_head) then
    begin
      select ln_account
      from loronostro
      where ln_region_loro = :reg_from and
            ln_region_nostro = :reg_loronostro_head
      into :acc1;
    end
    else
    begin
      select ln_account
      from loronostro
      where ln_region_loro = :reg_from and
            ln_region_nostro = :reg_to
      into :acc1;
    end

-- Счет откуда прилетает(Лороностро принимающего)
    if (reg_to <> reg_loronostro_head) then
    begin
      select ln_account
      from loronostro
      where ln_region_loro = :reg_to and
            ln_region_nostro = :reg_loronostro_head
      into :acc2;
    end
    else
    begin
      select ln_account
      from loronostro
      where ln_region_loro = :reg_to and
            ln_region_nostro = :reg_from
      into :acc2;
    end
  end

  for select om_account, om_bill_id, om_bill_line_no, om_workplace_id, om_matotvet_id, om_kolich_tms, om_amount_tms
      from hp_move_tms_list(:rep_date, :reg_from, :wp_from, :mo_from)
      into :om_account, om_bill_id, om_bill_line_no, :om_workplace_id, :om_matotvet_id, :om_kolich_tms, om_amount_tms
  do
  begin
    ol_number = ol_number + 1;
    om_amount_amort = 0;
    if (reg_from <> reg_to) then
    begin
      if (acc_prior <> om_account) then
      begin
        select reg_code || right(:om_account, 14)
        from region
        where reg_unicode = :reg_to
        into :om_account_to;
        acc_prior = om_account;
      end
    end
    else
    begin
      om_account_to = :om_account;
      acc1 = om_account_to;
      acc2 = om_account;
    end
    wp_tmp = :wp_to;
    mo_tmp = :mo_to;
    if (wp_tmp = 0) then
      wp_tmp = :om_workplace_id;
    if (mo_tmp = 0) then
      mo_tmp = :om_matotvet_id;

-- Передать из офиса
    insert into opermat (om_from_ol_id, om_number, om_bill_id, om_bill_line_no, om_oper_type, om_workplace_id,
                         om_matotvet_id, om_summa_plus, om_summa_minus, om_kolich_plus, om_kolich_minus, om_nds_percent,
                         om_account, om_is_written_off, om_workplace_info, om_matotvet_info, om_account_info,
                         om_summa_total, om_main_line)
    values (:ol_id_first, :ol_number, :om_bill_id, :om_bill_line_no, 3, :om_workplace_id, :om_matotvet_id, 0,
            :om_amount_tms, 0, :om_kolich_tms, 0, :om_account, 1, :wp_tmp, :mo_tmp, :acc1, :om_amount_tms, 1);

    if (reg_from = reg_to) then
    begin
      ol_number = ol_number + 1;
    end

    insert into opermat (om_from_ol_id, om_number, om_bill_id, om_bill_line_no, om_oper_type, om_workplace_id,
                         om_matotvet_id, om_summa_plus, om_summa_minus, om_kolich_plus, om_kolich_minus, om_nds_percent,
                         om_account, om_is_written_off, om_workplace_info, om_matotvet_info, om_account_info,
                         om_summa_total, om_main_line)
    values (:ol_id_last, :ol_number, :om_bill_id, :om_bill_line_no, 3, :wp_tmp, :mo_tmp, :om_amount_tms, 0,
            :om_kolich_tms, 0, 0, :om_account_to, 0, :om_workplace_id, :om_matotvet_id, :acc2, :om_amount_tms, 1);

    if (reg_from <> reg_to) then
      suspend; -- Передаем суммы, чтоб построить проводки потом

-- считаем накопленный износ, если есть передаем тоже
    if (reg_from <> reg_to) then
    begin
      om_amount_tms = 0;
      select sum(oal_summa_amort), sum(oal_summa_amort)
      from operamorlist, operlist
      where oal_bill_id = :om_bill_id and
            oal_bill_line_no = :om_bill_line_no and
            oal_from_ol_id = ol_id and
            ol_date <= :rep_date and
            operlist.is_delete = 0
      into :om_amount_amort, :om_os;
      if (om_amount_amort is null) then
        om_amount_amort = 0;
      if (om_os is null) then
        om_os = 0;
      if (om_amount_amort <> 0) then
      begin
        select ob_from_mc_id
        from operbill
        where ob_is_delete = 0 and
              ob_date_begin <= :rep_date and
              ob_date_end >= :rep_date and
              ob_bill_id = :om_bill_id and
              ob_bill_line_no = :om_bill_line_no
        into :mc_id;
        if (mc_id is not null) then
        begin
          select reg_code || :cm_default_currency || mc_acc_amort || '00000' || mc_acc_amort_no
          from matcateg, region
          where mc_id = :mc_id and
                reg_unicode = :reg_from
          into :acc_amort;

          insert into operamorlist (oal_bill_id, oal_bill_line_no, oal_from_ol_id, oal_is_delete, oal_oper_type,
                                    oal_lifetime_month, oal_summa_os, oal_summa_amort, oal_year, oal_month,
                                    oal_dont_calc)
          values (:om_bill_id, :om_bill_line_no, :ol_id_first, 0, 3, 0, -:om_os, -:om_amount_amort, :ol_year, :ol_month,
                  0);

          insert into operamorlist (oal_bill_id, oal_bill_line_no, oal_from_ol_id, oal_is_delete, oal_oper_type,
                                    oal_lifetime_month, oal_summa_os, oal_summa_amort, oal_year, oal_month,
                                    oal_dont_calc)
          values (:om_bill_id, :om_bill_line_no, :ol_id_last, 0, 0, 0, :om_os, :om_amount_amort, :ol_year, :ol_month, 0);

          om_account = :acc_amort;
          suspend;
        end
      end
    end
  end
end
