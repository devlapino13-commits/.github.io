CREATE PROCEDURE FM_GET_REPORT_BLOB(
    REP_ID D_STR_CODE,
    REP_LANG D_STR_LANG,
    REP_DATE D_DATE,
    MCL_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80)
AS
declare variable frm_id integer;
declare variable def_lang d_str_lang;
begin
  def_lang = 'ru';

  if (rep_date is null) then
    rep_date = current_date;

-- Подмена отчета, при необходимости
  select res_rep_id
  from mc_replace_rep_id(:mcl_id, :rep_id)
  into rep_id;

  select frm_id
  from fm_report_main
  where frm_code = :rep_id
  into :frm_id;

  if (coalesce(rep_lang, '') = '') then
    rep_lang = f_get_lang_code(:un_id, :mcl_id);

  select frb_blob
  from fm_report_blob
  where frb_from_frm_id = :frm_id and
        frb_lang = :rep_lang and
        frb_date <= :rep_date
  order by frb_from_frm_id desc, frb_lang desc, frb_date desc
  rows 1
  into :res_blob;

  if (row_count = 0 and
      rep_lang <> :def_lang) then
  begin
    select frb_blob
    from fm_report_blob
    where frb_from_frm_id = :frm_id and
          frb_lang = :def_lang and
          frb_date <= :rep_date
    order by frb_from_frm_id desc, frb_lang desc, frb_date desc
    rows 1
    into :res_blob;
  end

  suspend;
end
