CREATE PROCEDURE HP_GET_DISB_KIND(
    K_ID D_ID)
RETURNS (
    K_DISB_KIND D_SMALLINT)
AS
declare variable K_IS_ONLINE D_SMALLINT;
begin
  select kont.k_disb_kind, kont.k_is_online
  from kontract kont
  where kont.k_id = :k_id
  into :k_disb_kind, :k_is_online;

  if ((k_is_online = 1) and (k_disb_kind = 5)) then
    k_disb_kind = -1;

  suspend;
end
