CREATE PROCEDURE AAREP_FT_CLIENT_LIST(
    REP_DATE D_DATE)
RETURNS (
    C_ID D_ID,
    REC_NO D_INTEGER,
    FIO D_STR_255,
    BIRTH_DATE D_DATE,
    RESIDENT D_STR_10,
    DEBT_TXT D_STR_LONG,
    ZALOG_DEBT_TXT D_STR_LONG,
    OVERPAY_TXT D_STR_LONG,
    LAST_PAYMENT_DATE D_DATE,
    ASP_DATE_OPEN D_DATE,
    ASP_DATE_REJECT D_DATE)
AS
declare variable asp_kind d_integer;
declare variable max_date_const d_date;
declare variable min_date_const d_date;
begin
  -- выборка очень большая, поэтому пришлось делать почти всё через один запрос, так скорость гораздо выше
  asp_kind = f_edoc_kind('asp.agreement', 'id');
  rec_no = 0;
  max_date_const = '31.12.9999';
  min_date_const = '01.01.1900';

  for with t_overpay(ok_from_k_id, overpay_sum) as (
    select ok_from_k_id, -sum(ok_os)
    from operkred
    where ok_ol_is_active = 1
      and ok_type = 9
      and ok_ol_date <= :rep_date
    group by 1
    having sum(ok_os) <> 0
  ), t1(c_id, debt_str, zalog_debt_str, overpay_str, asp_date_open, asp_date_reject, last_payment_date) as (
  -- 1. активные клиенты (имеют активные договоры)
    select kc_from_c_id, cast(list(f_numtostr(oks_balance, 2) || ' - ' || k_kontract_no, ', ') as d_str_long), null, null, :max_date_const, :min_date_const, :min_date_const
    from oksummary, knclient, kontract
    where kc_from_k_id = oks_from_k_id
      and k_id = oks_from_k_id
      and oks_close_state = 0
      and :rep_date between oks_date_begin and oks_date_end
    group by 1,3,4,5,6,7
  union  -- добавляем так же залогодателей
    select z_from_c_id, null, cast(list(f_numtostr(oks_balance, 2) || ' - ' || k_kontract_no, ', ') as d_str_long), null, :max_date_const, :min_date_const, :min_date_const
    from oksummary, zalog, kontract
    where z_from_k_id = oks_from_k_id
      and k_id = oks_from_k_id
      and oks_close_state = 0
      and :rep_date between oks_date_begin and oks_date_end
    group by 1,2,4,5,6,7
  union   -- 2. имеют непогашенные переплаты
    select kc_from_c_id, null, null, cast(list(f_numtostr(overpay_sum, 2) || ' - ' || k_kontract_no, ', ') as d_str_long), :max_date_const, :min_date_const, :min_date_const
    from t_overpay, knclient, kontract
    where kc_from_k_id = ok_from_k_id
      and k_id = kc_from_k_id
    group by 1,2,3,5,6,7
  union -- 3. замороженные переплаты
    select acc_object_id, null, null, cast(list(f_numtostr(-f_acc_bal(acc_id, current_date, 'sum'), 2) || ' - Заморожено', ', ') as d_str_long),
        :max_date_const, :min_date_const, :min_date_const
    from ref_account, ref_accbalance
    where ab_code = '60322'
      and acc_from_ab_id = ab_id
      and acc_add_code = '60322 (83)'
      and acc_object_kind = 0
      and coalesce(f_acc_bal(acc_id, current_date, 'sum'), 0) <> 0
    group by 1,2,3,5,6,7
  union  -- 4. активный АСП
    select edl_object_id, null, null, null, cast(eds_status_datetime as date), coalesce(edl_date_end + 1, :min_date_const), :min_date_const
    from edoc_sign, edoc_list, edoc_blank edb
    where eds_from_edl_id = edl_id
      and edb_from_edl_id = edl_id
      and edl_from_edc_id = :asp_kind
      and edl_object_kind = 0
      and eds_status = 1
      and eds_status_datetime < :rep_date + 1
      and coalesce(edl_date_end, :max_date_const) >= :rep_date
  union -- 5. платеж был в течении двух лет
    select kc_from_c_id, null, null, null, :max_date_const, :min_date_const, max(ks_date_begin)
    from kontstate, knclient
    where kc_from_k_id = ks_from_k_id
      and ks_ol_is_delete = 0
      and ks_state = 2
      and ks_date_begin >= dateadd(year, -2, :rep_date)
    group by 1,2,3,4,5,6
    having max(ks_date_begin) < :rep_date
  ), t2 (c_id, debt_str, zalog_debt_str, overpay_str, asp_date_open, asp_date_reject, last_payment_date) as (
    select c_id, list(debt_str, ', '), list(zalog_debt_str, ', '), list(overpay_str, ', '),
        min(asp_date_open), max(asp_date_reject), max(last_payment_date)
    from t1
    group by 1
  )
  select t2.c_id, c_name_fio, c_birth_date, iif(c_citizen = 13, '+', ''),
      debt_str, zalog_debt_str, overpay_str,
      iif(asp_date_open = :max_date_const, null, asp_date_open),
      iif(asp_date_reject = :min_date_const, null, asp_date_reject),
      iif(last_payment_date = :min_date_const, null, last_payment_date)
  from t2, clients c
  where c.c_id = t2.c_id
  and not exists(select 1 from ft_info_stoplist
              where current_date between fisl_freeze_date_begin and coalesce(fisl_freeze_date_end, :max_date_const)
                and fisl_from_c_id = t2.c_id)
  into c_id, fio, birth_date, resident, debt_txt, zalog_debt_txt, overpay_txt,
       asp_date_open, asp_date_reject, last_payment_date
  do
  begin
    rec_no = rec_no + 1;

    if (asp_date_open is null) then
      select min(cast(eds_status_datetime as date)), max(edl_date_end + 1)
      from edoc_sign, edoc_list, edoc_blank edb
      where eds_from_edl_id = edl_id
        and edb_from_edl_id = edl_id
        and edl_from_edc_id = :asp_kind
        and edl_object_kind = 0
        and edl_object_id = :c_id
        and eds_status = 1
        and eds_status_datetime < :rep_date + 1
      into asp_date_open, asp_date_reject;

    if (last_payment_date is null) then
      select max(ok_ol_date)
      from knclient, operkred
      where kc_from_c_id = :c_id
        and ok_from_k_id = kc_from_k_id
        and ok_type = 4
        and ok_ol_is_active = 1
        and ok_ol_date <= :rep_date
      into last_payment_date;

    suspend;
  end 
end
