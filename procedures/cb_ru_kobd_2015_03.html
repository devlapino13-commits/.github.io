CREATE PROCEDURE CB_RU_KOBD_2015_03(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(60),
    KOL_TRANSACTION INTEGER,
    KOL_KOBD INTEGER,
    C_NAME_FIO VARCHAR(100),
    JL_NAME VARCHAR(100),
    PS_DATE_BEGIN DATE,
    TRANS_KOBD INTEGER,
    BRANCH_NAME VARCHAR(100))
AS
declare variable reg_id integer;
declare variable c_id id_type;
declare variable c_sex smallint;
declare variable hrv_day_back date;
begin
  for select ol_from_reg_id, count(distinct ol_id)
      from operkred, operlist
      where operlist.is_delete = 0 and
            ol_date_oper_day between :rep_date_begin and :rep_date_end and
            ol_id = ok_from_ol_id and
            ok_type in (8, 4) and
            ok_os + ok_perc + ok_fine <> 0 and
            (:rep_date_end < '01.01.2017' or
            (ol_doc_kind in (0, 1))) -- Кассовый расходный и приходный ордер
      group by ol_from_reg_id
      into reg_id, kol_transaction
  do
  begin
--      reg_name = (select reg_code||' '||reg_name from region where reg_unicode = :reg_id);

    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
    from region r1, region r2
    where r1.reg_unicode = :reg_id and
          r1.parent = r2.reg_unicode
    into reg_name, branch_name;

    /*
      kol_kobd = null;
      select count(*) from hr_pstatus, joblist, jobkind
      where :rep_date_end between ps_date_begin and ps_date_end
      and ps_is_working = 1
      and ps_status = 1
      and ps_from_jl_id = jl_id
      and jl_kind = jk_id
      and jk_system_code = 'JobSystem.Specialist_KOBD'
      and ps_from_reg_id = :reg_id
      and coalesce((select sawt_id from HR_VOCTRAVEL, hr_orders, sa_worktype
          where hrv_from_ho_id = ho_id
          and ho_order_type = sawt_id
          and hrv_from_c_id = ps_from_c_id
          and :rep_date_end between hrv_date_begin and hrv_date_end_fact
          order by hrv_date_begin desc
          rows 1),0) not in (16,17,18,8) -- Декретников не включает
      into kol_kobd;

      if (kol_kobd <> 0) then trans_kobd = kol_transaction / kol_kobd;
      else trans_kobd = kol_transaction;
      */

--      kol_trans_temp = kol_transaction;

    c_name_fio = null;
    for select c_name_fio, jl_name --, ps_date_begin
               , c_id, c_sex
        from hr_pstatus, joblist, jobkind, clients
        where :rep_date_end between ps_date_begin and ps_date_end
--      and ps_date_begin <= :REP_DATE_BEGIN -- Добавлено от 11.01.2016 Улан, нужно включать кто полный период отработал
              and
              ps_is_working = 1 and
              ps_status = 1 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code = 'JobSystem.Specialist_KOBD' and
              ps_from_reg_id = :reg_id and
              ps_from_c_id = c_id and
              coalesce((select sawt_id
                        from hr_voctravel, hr_orders, sa_worktype
                        where hrv_from_ho_id = ho_id and
                              ho_order_type = sawt_id and
                              hrv_from_c_id = ps_from_c_id and
                              :rep_date_end between hrv_date_begin and hrv_date_end_fact
                        order by hrv_date_begin desc
                        rows 1), 0) not in (16, 17, 18, 8) -- Декретников не включает
        into c_name_fio, jl_name --, ps_date_begin
             , c_id, c_sex
    do
    begin
--      вычисляем дату найма кассира
      select min(ps_date_begin)
      from hr_pstatus, joblist, jobkind
      where ps_is_working = 1 and
            ps_status = 1 and
            ps_from_c_id = :c_id and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.Specialist_KOBD'
      into ps_date_begin;

      if (c_sex = 0 and
          ps_date_begin <= :rep_date_begin) then
      begin
--        если сотрудник женского рода, то проверяет когда в последний раз вернулся из декретного отпуска
        hrv_day_back = null;
        select max(hrv_day_back)
        from hr_voctravel, hr_orders, sa_worktype
        where hrv_from_ho_id = ho_id and
              ho_order_type = sawt_id and
              hrv_from_c_id = :c_id and
              sawt_id in (16, 17, 18, 8) and
              hrv_date_end_fact - hrv_date_begin > 30 and
              hrv_date_end_fact < :rep_date_end -- приказы после заданной даты не показывало
        into hrv_day_back;

        if (hrv_day_back is not null) then
        begin
--          дата возвращение декретного отпска будет датой началы работы.
          if (ps_date_begin < hrv_day_back) then
            ps_date_begin = hrv_day_back;
        end
      end

      if (ps_date_begin <= :rep_date_begin) then
      begin
        kol_kobd = 1;
        suspend; -- нужно вытаскивать кто полный период отработал
        kol_transaction = null;
--          kol_kobd = null;
      end
    end

    if (c_name_fio is null) then
    begin
      kol_kobd = 0;
      suspend;
    end
  end
end
