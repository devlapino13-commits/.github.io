CREATE PROCEDURE AAREP_CLIENT_VERIFY_DATA(
    C_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    CLIENT_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_SEX_STR VARCHAR(20),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_DATE_EXPIRE DATE,
    C_VERIFY_STATUS INTEGER,
    C_VERIFY_DATETIME TIMESTAMP,
    C_BLOB_PHOTO BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    C_BLOB_PASS1 BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    C_BLOB_PASS2 BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    C_INN VARCHAR(20),
    C_ALTERNAT_FIO VARCHAR(100),
    C_PASS_WHO VARCHAR(255),
    C_PASS_WHO_CODE VARCHAR(20),
    C_BIRTH_COUNTRY VARCHAR(100),
    C_BIRTH_CITY VARCHAR(100),
    C_CITIZEN INTEGER,
    C_DOC_TYPE INTEGER,
    C_BLOB_PHOTO_ID DECIMAL(18,0),
    C_BLOB_PASS1_ID DECIMAL(18,0),
    C_BLOB_PASS2_ID DECIMAL(18,0),
    C_SOURCE_SYSTEM VARCHAR(100),
    C_PINCODE VARCHAR(20),
    C_STAFF_STATUS D_STR_255,
    C_NAME_ENG D_STR_255,
    C_CITIZEN_CODE D_STR_CODE,
    ALTERNATE_FIO_DISABLED D_SMALLINT,
    BLOB_LIST D_STR_4096,
    ADL_ADDRESS_STR_LEG D_STR_255,
    ADL_ID D_ID,
    BIO_PHOTO D_SMALLINT,
    CDS_DOC_BLOB_1 DECIMAL(18,0),
    CDS_DOC_BLOB_2 DECIMAL(18,0))
AS
declare variable c_sex          integer;
declare variable cs_staff_state integer;
declare variable c_from_un_id   decimal(18,0);
declare variable pk_name        d_str_255;
declare variable rd_id          integer;
declare variable acl_from_un_id id_type;
declare variable jl_name        str_100;
declare variable jk_system_code str_100;
declare variable un_fio         str_100;
declare variable acl_date       date;
declare variable un_from_c_id   id_type;
declare variable c_ext_inn      d_str_20;
declare variable db_id          d_integer;
begin
  c_blob_photo = null;
  c_blob_pass1 = null;
  c_blob_pass2 = null;
  c_staff_status = '';
  bio_photo = -1;
  blob_list = '';
  db_id = f_get_db_id();

  for select c_name_fio, c_sex, c_birth_date, c_pass_no, c_pass_date, c_pass_date_expire, c_blob_photo_id,
             c_blob_pass1_id, c_blob_pass2_id, c_verify_status, c_verify_datetime, c_inn, c_ext_inn, c_alternat_fio, c_pass_who,
             c_pass_who_code, c_birth_country, c_birth_city, c_citizen, c_doc_type, c_pincode,
             iif(c_source_system = 1, 'FS Mobile', 'FinStart'), c_from_un_id, c_name_eng, adl.adl_address_str,
             adl.adl_id
      from clients
      inner join addrclient adc on adc.adc_from_c_id = c_id and
            adc.adc_address_type = 1
      inner join addrlist adl on adc.adc_from_adl_id = adl.adl_id
      where c_id = :c_id
      into c_name_fio, c_sex, c_birth_date, c_pass_no, c_pass_date, c_pass_date_expire, c_blob_photo_id,
           c_blob_pass1_id, c_blob_pass2_id, c_verify_status, c_verify_datetime, c_inn, c_ext_inn, c_alternat_fio, c_pass_who,
           c_pass_who_code, c_birth_country, c_birth_city, c_citizen, c_doc_type, c_pincode, c_source_system,
           c_from_un_id, c_name_eng, adl_address_str_leg, adl_id
  do
  begin
    alternate_fio_disabled = 1;

    if (left(replace(:c_pass_no, ' ', ''), 2) = 'ID') then
    begin
      :bio_photo = 0;
      select coalesce(cds.cds_doc_blob_1, 0), coalesce(cds.cds_doc_blob_2, 0)
      from clientdocs cds
      inner join passkind pk on pk.pk_id = cds_from_pk_id
      inner join passdoctype pdt on pk.pk_from_pdt_id = pdt.pdt_id and
        pdt.pdt_code = 'info_with_chip' --информация с чипа биомет.паспорта
      where cds_from_c_id = :c_id and
        cds_is_delete = 0
      order by cds_id desc
      rows 1
      into :cds_doc_blob_1, :cds_doc_blob_2;

      if (cds_doc_blob_1 <> 0 or cds_doc_blob_2 <> 0) then
        :bio_photo = 1;
    end

    select pk_name
    from passkind
    where pk_id = :c_doc_type
    into pk_name;

    if (:cds_doc_blob_1 is null) then
        :cds_doc_blob_1 = 0;
    if (:cds_doc_blob_2 is null) then
        :cds_doc_blob_2 = 0;

    if (coalesce(c_blob_pass2_id, 0) > 0) then
    begin
      if (:bio_photo < 1) then
        blob_list = '%1,%2';
      else
      if (cds_doc_blob_1 <> 0 and
          cds_doc_blob_2 <> 0) then
        blob_list = '%1,%2,%3,%4';
      else
        blob_list = '%1,%2,%3';

    end
    else
    begin
      if (:bio_photo < 1) then
        blob_list = '%1';
      else
      if (cds_doc_blob_1 <> 0 and
          cds_doc_blob_2 <> 0) then
        blob_list = '%1,%2,%3';
      else
        blob_list = '%1,%2';
    end

    if (cds_doc_blob_1 = 0 and
        cds_doc_blob_2 <> 0) then
      cds_doc_blob_1 = cds_doc_blob_2;

    blob_list = f_format('[{"caption": "%1","blobs":[%2]},{"caption": "%3","blobs":[%4]}]',
        c_name_fio,
        f_image_item('blobstore', c_blob_photo_id, 'Фото', un_id, 0),
        pk_name,
        f_format(blob_list,
          f_image_item('blobstore', c_blob_pass1_id, 'стр.1', un_id),
          f_image_item('blobstore', c_blob_pass2_id, 'стр.2', un_id),
          f_image_item('blobstore', cds_doc_blob_1, 'id стр.1', un_id),
          f_image_item('blobstore', cds_doc_blob_2, 'id стр.2', un_id)));

    if (c_sex = 0) then
      c_sex_str = 'ЖЕН.';
    else
      c_sex_str = 'МУЖ.';

    client_id = :c_id;

    select replace(rd_code, 'citizen.', '')
    from refdata
    where rd_id = :c_citizen
    into c_citizen_code;

    if (char_length(coalesce(c_citizen_code, '')) <> 2) then
      c_citizen_code = '';

    if (db_id = 81001 and c_citizen_code = 'uz') then
      alternate_fio_disabled = 0;

    if (db_id = 81001 and c_citizen_code = 'kg') then
      c_inn = c_ext_inn;

    select cs_staff_state
    from clientstate
    where cs_from_c_id = :c_id
    into cs_staff_state;
    if (cs_staff_state = 0) then
      c_staff_status = 'Бывший сотрудник';
    else
    if (cs_staff_state = 1) then
      c_staff_status = 'Действующий сотрудник';
    else
    begin
      if (c_from_un_id is not null) then
      begin
        if (exists(select *
                   from hp_permission_access_check(:c_from_un_id, 'Menulist', 0, 'N62')
                   where has_access = 1)) then
        begin
          if (not(exists(select *
                         from hp_permission_access_check(:c_from_un_id, 'PermActions', 0, 'FinStart.Superuser')
                         where has_access = 1))) then
          begin
            c_staff_status = 'Возможно будущий сотрудник';
          end
        end
      end
    end

    select acl_from_un_id, -- ID пользователя кто отправил на верификацию
           acl_date
    from actionlist
    where acl_object_kind = 0 and
          acl_object_id = :c_id and
          acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO'))
    order by acl_datetime desc rows 1
    into acl_from_un_id, acl_date;

    jl_name = '';
    jk_system_code = '';
    select jl_name, jk_system_code
    from hr_pstatus, joblist, jobkind
    where ps_from_c_id = :un_from_c_id and
          :acl_date between ps_date_begin and ps_date_end and
          ps_status = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          ps_is_working = 1
    into jl_name, jk_system_code;

    if (jk_system_code in ('JobSystem.HR_Department', 'JobSystem.HR_Service') and
        c_staff_status = '') then
      c_staff_status = 'Будущий сотрудник';

    if (c_staff_status <> '') then
      c_staff_status = c_staff_status || '; ';

    if (exists(
        select *
        from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.ClientVerifyShowInitiator')
        where has_access = 1)) then
    begin
      un_fio = '';
      select un_fio, un_from_c_id
      from username
      where un_id = :acl_from_un_id
      into un_fio, un_from_c_id;
  
      if (un_fio <> '') then
      begin
        select short_fio
        from hp_fio_clear(:un_fio)
        into un_fio;
        c_staff_status = c_staff_status || 'Инициатор:' || un_fio;
      end
    end

    if (jl_name <> '') then
      c_staff_status = c_staff_status || ' ' || jl_name;
    suspend;
  end
end
