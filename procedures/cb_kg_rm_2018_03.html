CREATE PROCEDURE CB_KG_RM_2018_03(
    REP_DATE_END DATE)
RETURNS (
    REGION_NAME VARCHAR(100),
    TOTAL_BONUS_OKLAD_SPF CURR_TYPE,
    KOL_KC INTEGER,
    BONUS_OKLAD_SPF CURR_TYPE,
    BALANCE_END CURR_TYPE,
    BALANCE_END_IPF CURR_TYPE,
    DELINQ_BALANCE_30 CURR_TYPE,
    BALANCE_BEGIN CURR_TYPE,
    BALANCE_BEGIN_IPF CURR_TYPE,
    BASIC_BONUS CURR_TYPE,
    PRIROST CURR_TYPE,
    PRIROST_IPF CURR_TYPE,
    KPI1 CURR_TYPE,
    KPI2 CURR_TYPE,
    KPI3 CURR_TYPE,
    BONUS_RM CURR_TYPE,
    PLAN_GROWTH DECIMAL(18,4),
    PLAN_GROWTH_IPF DECIMAL(18,4),
    PLAN_SPF INTEGER)
AS
declare variable rep_date_begin date;
declare variable region_id integer;
declare variable rep_period integer;
begin
  rep_date_begin = rep_date_end - extract(day from rep_date_end);
  rep_period = extract(year from rep_date_end) || right((extract(month from rep_date_end) + 100), 2);

  /*
  for
  select c_name_fio, jl_name, ps_from_reg_id
  from jobsystem, joblist, hr_pstatus, clients
  where js_memo = 'JobSystem.Region_Director'
  and js_id = jl_kind
  and jl_id = ps_from_jl_id
  and current_date between ps_date_begin and ps_date_end
  and ps_status = 1
  and ps_is_working = 1
  and ps_from_c_id = c_id
  and ps_from_reg_id <> 1090100
  into rm_name, job_name, reg_id
  do
  begin

    select r2.reg_name, r1.parent
    from region r1, region r2
    where r1.reg_unicode = :reg_id
    and r1.parent = r2.reg_unicode
    into region_name, parent;

    suspend;
  end
  */

-- Сумма заработка всех СПФ компании
--  delete from temp_cb_rm;
  total_bonus_oklad_spf = 0;

  for select parent,
--         count(*),
             sum(aarep_bonus_total)
      from cb_kg_lo_from_km(1000000, :rep_date_end), region
      where reg_id = reg_unicode
      group by parent
      into region_id,
--       kol_kc,
           bonus_oklad_spf
  do
  begin
    insert into temp_cb_rm (tcr_region_id, /* tcr_kol_spf, */ tcr_bonus_oklad_spf)
    values (:region_id, /*:kol_kc,*/ :bonus_oklad_spf);
    total_bonus_oklad_spf = total_bonus_oklad_spf + bonus_oklad_spf;
  end

  for select prs_region, prs_plan_growth, prs_plan_growth_ipf, prs_plan_spf
      from planregionspf
      where prs_period = :rep_period
      into region_id, plan_growth, plan_growth_ipf, plan_spf
  do
  begin
    select reg_name
    from region
    where reg_unicode = :region_id
    into region_name;

    select --tcr_kol_spf,
tcr_bonus_oklad_spf
    from temp_cb_rm
    where tcr_region_id = :region_id
    into --kol_kc,
         bonus_oklad_spf; -- бонус КС до вычета оклада

--  Базовый бонус РМ
    basic_bonus = round(bonus_oklad_spf / total_bonus_oklad_spf * 250000, 0);
--  портфель на конец, баланс просрочки >30 дней
    select sum(oks_balance), sum(iif(pr_code containing '.ipf.', oks_balance, 0)),
           sum(iif(:rep_date_end - oks_delinq_date > 30, oks_balance, 0)), count(distinct oks_kc_id)
    from oksummary, region_list(:region_id, 1, 3, 1), kontract, product
    where oks_region_id = id_list and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          oks_balance > 0 and
          oks_close_state = 0 
    into balance_end, balance_end_ipf, delinq_balance_30, kol_kc;
    balance_end = balance_end - balance_end_ipf;

--  портфель на начало
    select sum(oks_balance), sum(iif(pr_code containing '.ipf.', oks_balance, 0))
    from oksummary, region_list(:region_id, 1, 3, 1), kontract, product
    where oks_region_id = id_list and
          :rep_date_begin between oks_date_begin and oks_date_end and
          oks_from_k_id = k_id and
          oks_close_state = 0 and
          k_from_pr_id = pr_id
    into balance_begin, balance_begin_ipf;
    balance_begin = balance_begin - balance_begin_ipf;

    prirost = balance_end - balance_begin;
    prirost_ipf = balance_end_ipf - balance_begin_ipf;

--  KPI1
    kpi1 = (prirost / plan_growth) * (prirost_ipf / plan_growth_ipf);
    if (kpi1 > 1.1) then
      kpi1 = 1.1;
--  KPI2
    kpi2 = 1 - (delinq_balance_30 / 1000000) * 3 / 100;
--  KPI3
    kpi3 = 1 - (cast(plan_spf as double precision) - kol_kc) * 3 / 100;
    if (kpi3 > 1.0) then
      kpi3 = 1.0;
--  Бонус РМ к начислению
    bonus_rm = round(cast(basic_bonus as double precision) * kpi1 * kpi2 * kpi3, 0);

    suspend;
  end

end
