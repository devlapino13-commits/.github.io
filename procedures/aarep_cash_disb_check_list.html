CREATE PROCEDURE AAREP_CASH_DISB_CHECK_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    END_DISB_SALDO DECIMAL(18,4),
    END_OTHER_SALDO DECIMAL(18,4),
    END_TOTAL_SALDO DECIMAL(18,4),
    BEGIN_DISB_SALDO DECIMAL(18,4),
    BEGIN_OTHER_SALDO DECIMAL(18,4),
    BEGIN_TOTAL_SALDO DECIMAL(18,4),
    IN_FROM_BANK DECIMAL(18,4),
    IN_FROM_REPAY DECIMAL(18,4),
    OUT_TO_BANK DECIMAL(18,4),
    OUT_TO_DISB DECIMAL(18,4),
    OUT_TO_BANK_FROM_DISB DECIMAL(18,4),
    REP_REGION_NAME VARCHAR(100))
AS
declare variable reg_id integer;
begin
  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into :reg_id
  do
  begin
    select end_disb_saldo, end_other_saldo, end_total_saldo, begin_disb_saldo, begin_other_saldo, begin_total_saldo,
           in_from_bank, in_from_repay, out_to_bank, out_to_disb, out_to_bank_from_disb, rep_region_name
    from aarep_cash_disb_check(:reg_id, :rep_date_end)
    into :end_disb_saldo, :end_other_saldo, :end_total_saldo, :begin_disb_saldo, :begin_other_saldo, :begin_total_saldo,
         :in_from_bank, :in_from_repay, :out_to_bank, :out_to_disb, :out_to_bank_from_disb, :rep_region_name;
    suspend;
  end
end
