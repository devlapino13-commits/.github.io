CREATE PROCEDURE HP_GRS_ANSWER_EXEC(
    EIL_ID D_ID)
AS
declare variable C_ID D_ID;
declare variable ADL_ID D_ID;
declare variable UN_VIRT D_ID;
declare variable PARENT_ID D_ID;
declare variable C_NAME_FIO D_STR_100;
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable ADDRESS_ARRAY D_BLOB_TEXT;
declare variable POST D_INTEGER;
declare variable COUNTRY_TXT D_BLOB_TEXT;
declare variable ADDRESS_STR D_BLOB_TEXT;
declare variable ADDRESS_HOUSE D_BLOB_TEXT;
declare variable ADDRESS_BUILDING D_BLOB_TEXT;
declare variable ADDRESS_APARTMENT D_BLOB_TEXT;
declare variable SURNAME D_STR_100;
declare variable NAME D_STR_100;
declare variable PATRONYMIC D_STR_100;
declare variable FAMILY_STATUS D_INTEGER;
declare variable GENDER D_STR_SYM;
declare variable DATE_OF_BIRTH D_DATE;
declare variable PASSPORT_SERIES D_STR_10;
declare variable PASSPORT_NUMBER D_STR_20;
declare variable PIN D_STR_20;
declare variable ISSUED_DATE D_DATE;
declare variable PASSPORT_AUTHORITY D_STR_64;
declare variable PASSPORT_AUTHORITY_CODE D_STR_64;
declare variable EXPIRED_DATE D_DATE;
declare variable VOIDSTATUS D_INTEGER;
declare variable MCL_ID D_ID;
declare variable PL_STREET_ID D_ID;
declare variable RD_MARITAL_STATUS D_INTEGER;
declare variable DOC_TYPE D_INTEGER;
declare variable FL_ID D_ID;
declare variable FL_ID_ENG D_ID;
begin
  un_virt = 1000000000000;
  mcl_id = 0;
  for select cl.c_id, cl.c_name_fio, eil.eil_from_bls_id
      from extinfolist eil
      inner join clients cl on cl.c_id = eil.eil_from_c_id
      where eil.eil_id = :eil_id
      into :c_id, :c_name_fio, :bls_id
  do
  begin
    select tg.json_text
    from temp_grs tg
    where tg.bls_id = :bls_id
    into json_text;
    if (coalesce(json_text, '') = '') then
    begin
      select blx_blob
      from hp_get_blobex_blob(:bls_id)
      into json_text;
    end

    select mcl.mcl_id
    from mc_login mcl
    where mcl.mcl_from_c_id = :c_id and
          mcl.mcl_is_delete = 0
    order by mcl.mcl_last_login_datetime desc
    rows 1
    into mcl_id;

    select voidstatus, surname, name, patronymic, family_status, date_of_birth, passport_series, passport_number, pin,
           issued_date, passport_authority, passport_authority_code, expired_date, gender, country_txt, post, parent_id,
           pl_street_id, address_str, address_building, address_apartment, address_array, rd_marital_status, doc_type
    from hp_parse_json_grs(:json_text)
    into :voidstatus, :surname, :name, :patronymic, :family_status, :date_of_birth, :passport_series, :passport_number,
         :pin, :issued_date, :passport_authority, :passport_authority_code, :expired_date, :gender, :country_txt, :post,
         :parent_id, :pl_street_id, :address_str, :address_building, :address_apartment, :address_array, :rd_marital_status,
         :doc_type;

    ---------------------------------------------------------------------- >
    if (voidstatus = 1) then
    begin
      update grs_clients
      set gc_is_last = 0
      where gc_from_c_id = :c_id and
            gc_is_last = 1;

      insert into grs_clients (gc_from_c_id, gc_from_eil_id, gc_status, gc_status_datetime, gc_is_last)
      values (:c_id, :eil_id, -4, current_timestamp, 1);

      update clients c
      set c.c_verify_status = -2
      where c.c_id = :c_id;

      if (coalesce(:mcl_id, 0) > 0) then
        execute procedure mc_send_push('clients', :c_id, 'Верификация паспорта',
            'Ваш паспорт не прошел верификацию!');

      exit;
    end
    if (coalesce(:parent_id, 0) > 0) then
    begin
      insert into addrlist (adl_from_pl_id, adl_depart_no, adl_depart_pref, adl_flat_no, adl_building_no,
                            adl_section_no, adl_cross_street_id, adl_address_str, adl_comment, adl_post_index,
                            adl_is_edited, adl_from_un_id, adl_street_id, adl_is_without_street, adl_is_filled,
                            adl_long, adl_lat, adl_geo_count, adl_place_pls_id, adl_street_pls_id, adl_lang_kind)
      values (:parent_id, 0, :address_building, :address_apartment, 0, 0, null, :address_str, null, :post, 0, :un_virt,
              :pl_street_id, iif(:pl_street_id is null, 1, 0), 1, null, null, -1, null, null, 0)
      returning adl_id
      into :adl_id;

      update or insert into addrclient (adc_from_c_id, adc_address_type, adc_from_adl_id, adc_from_un_id)
      values (:c_id, 1, :adl_id, :un_virt)
      matching (adc_from_c_id, adc_address_type);

      insert into grs_addrlist (ga_from_eil_id, ga_from_adl_id)
      values (:eil_id, :adl_id);
    end
    -- address ----------------------------------------------------------- <
    if (coalesce(address_array, '') = '' or address_str containing 'ликвидирован') then
    begin
      update grs_clients gc
      set gc.gc_status = 3
      where gc.gc_from_c_id = :c_id and
            gc.gc_status = 2 and
            gc.gc_is_last = 1 and
            gc.gc_from_eil_id = :eil_id;

      update addrclient adc
      set adc.adc_from_adl_id = 41701000000000
      where adc.adc_address_type = 1 and
            adc.adc_from_c_id = :c_id;
    end

    select fl_id, fl_id_eng
    from hp_prepare_fio(:surname, :name, :patronymic, null, null, null)
    into :fl_id, fl_id_eng;

    update clients c
    set c.c_name_fio = f_translit_fio_grs('cyr', 'cyr', coalesce(:surname, '') || ' ' || coalesce(:name, '') || ' ' || coalesce(:patronymic, ''), 'kg'),
        c.c_sex = iif(:gender = 'M', 1, 0),
        c.c_birth_date = :date_of_birth,
        c.c_pass_no = :passport_series || :passport_number,
        c.c_doc_type = :doc_type,
        c.c_pass_date = :issued_date,
        c.c_pass_date_expire = :expired_date,
        c.c_family = :family_status,
        c.c_index_c_pass_no = :passport_series || :passport_number,
        c.c_from_fl_id = :fl_id,
        c.c_name_eng_fl_id = :fl_id_eng
    where c.c_id = :c_id and
          c.c_inn = :pin;

    update clientdocs cds
    set cds.cds_create_datetime = current_timestamp,
        cds.cds_doc_date_begin = :issued_date,
        cds.cds_doc_date_end = :expired_date
    where cds.cds_from_c_id = :c_id and
          cds.cds_is_main_doc = 1 and
          cds.cds_is_delete = 0;

    if (coalesce(:mcl_id, 0) > 0) then
    begin
      update clients c
      set c.c_verify_status = 2
      where c.c_id = :c_id and
            c.c_inn = :pin;
      execute procedure mc_send_push('clients', :c_id, 'Верификация паспорта',
          'Ваш паспорт успешно прошел верификацию!');
    end

    execute procedure hp_objectprops_set_with_rd(f_objkind('clients', 'id'), :c_id, -1,
        f_get_rd_id('AnketaQuestions.Family'), current_date, '', 0, :family_status, :un_virt);

    execute procedure hp_objectprops_set_with_rd(f_objkind('clients', 'id'), :c_id, -1,
        f_get_rd_id('AnketaQuestions.Family_tunduk'), current_date, '', 0, :rd_marital_status, :un_virt);

  end
end
