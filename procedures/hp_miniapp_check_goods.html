CREATE PROCEDURE HP_MINIAPP_CHECK_GOODS(
    MA_ID D_INTEGER)
RETURNS (
    RES_VALUE D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable k_id d_id;
begin
  --Проверка, что по продукту должен быть указан товар и указан ли он

  res_text = '';
  res_value  = 0;

  select kont.k_id
  from kontract kont, product pr
  where kont.k_from_ma_app_id = :ma_id and
        k_from_pr_id = pr_id and
        pr_code containing '.kontmat.'
  into k_id;

  if (k_id is null) then
  begin
    res_text = '';
    suspend;
    exit;
  end

  res_value = 1;

  if (not exists(select *
                 from kontmat km
                 where km.kml_object_kind = 9
                 and km.kml_object_id = :k_id)) then
  begin
    res_value = -1;
    res_text = 'Товар не найден';
  end

  suspend;
end
