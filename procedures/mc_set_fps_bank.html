CREATE PROCEDURE MC_SET_FPS_BANK(
    MCL_ID D_ID,
    FPB_ID D_ID,
    MA_ID D_ID)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255)
AS
declare variable mac_id d_id;
declare variable c_id d_id;
declare variable res_is_error d_smallint;
begin
    c_id = (select mcl_from_c_id from mc_login where mcl_id = :mcl_id);
    res_is_error = 0;

    select first 1 mac.mac_id
    from miniappclient mac
    where mac.mac_from_ma_id = :ma_id and
          mac.mac_from_c_id = :c_id and
          mac.mac_amount > 0 and
          mac.mac_is_cashier = 1
    into :mac_id;

    if (coalesce(c_id, 0) = 0 or coalesce(mac_id, 0) = 0 or fpb_id is null) then
    begin
        res_code = 'error';
        res_text = 'Не удалось загрузить данные. Пожалуйста, попробуйте позже';
        suspend;
        exit;
    end

    -- Устанавливаем банк
    select res_is_error
    from fm_set_fps_loan_banks(f_objkind('miniappclient', 'id'), :mac_id, :fpb_id, 1000000000000)
    into :res_is_error;

    -- При установке банка произошла ошибка
    if (res_is_error = 1) then
    begin
        res_code = 'error';
        res_text = 'Не удалось установить банк для выдачий займа. Пожалуйста, повторите попытку позже или обратитесь к технической поддержке';
        suspend;
        exit;
    end

    -- Обновляем этап, на котором находится клиент
    update mc_oivo_online_loan ool
    set ool.ool_stage = f_enum_id('oivo_online_loan_stage', 'fps_bank_selected')
    where ool.ool_from_mcl_id = :mcl_id and
          ool.ool_from_ma_id = :ma_id;

    -- Банк успешно установлен
    res_code = 'success';
    res_text = '';

    suspend;

    when any do
    begin
        res_code = 'error';
        res_text = 'Что-то пошло не так. Повторите попытку позже или обратитесь к технической поддержке';
        suspend;
    end
end
