CREATE PROCEDURE AAREP_CALC_PERC_LIST_TMP(
    REPORT_REGION_ID D_INTEGER,
    REPORT_DATE_BEGIN D_DATE,
    REPORT_DATE_END D_DATE)
RETURNS (
    AAREP_K_ID D_ID,
    AAREP_FROM_REG_ID D_INTEGER,
    AAREP_PERC_BEGIN D_CURRENCY,
    AAREP_PERC_CALCED D_CURRENCY,
    AAREP_PERC_PAYED D_CURRENCY,
    AAREP_PERC_WRITEOFF D_CURRENCY,
    AAREP_PERC_END D_CURRENCY,
    AAREP_PERC_TRANSFER D_CURRENCY,
    AAREP_OUT_PERC_BEGIN D_CURRENCY,
    AAREP_OUT_PERC_CALCED D_CURRENCY,
    AAREP_OUT_PERC_END D_CURRENCY,
    AAREP_DELAYED_PERC_BEGIN D_CURRENCY,
    AAREP_DELAYED_PERC_END D_CURRENCY,
    AAREP_DELAYED_PERC_TRANSFER D_CURRENCY)
AS
declare variable rep_period_begin d_integer;
declare variable rep_period_end d_integer;
declare variable db_id d_integer;
declare variable k_id d_id;
begin
  db_id = f_get_db_id();
  if (db_id = 94401 and report_date_end <= '31.12.2011') then
    exit;

  rep_period_begin = 0;
  rep_period_end = 0;
  if (extract(day from :report_date_begin) = 1) then
    rep_period_begin = extract(year from :report_date_begin) * 100 + extract(month from :report_date_begin);
  if (extract(day from (:report_date_end + 1)) = 1) then
    rep_period_end = extract(year from :report_date_end) * 100 + extract(month from :report_date_end);
  if (rep_period_begin = 0 or rep_period_end = 0) then
    exit;

  aarep_perc_begin = 0;
  aarep_perc_calced = 0;
  aarep_perc_payed = 0;
  aarep_perc_writeoff = 0;
  aarep_perc_end = 0;
  aarep_perc_transfer = 0;
  aarep_out_perc_begin = 0;
  aarep_out_perc_calced = 0;
  aarep_out_perc_end = 0;

  for
    with tab as
    (
       select
         kcp_from_k_id as aarep_k_id, kcp_from_reg_id as aarep_from_reg_id, -- новый офис
         iif(kcp_from_reg_id = kcp_transfer_reg_id and kcp_period = :rep_period_begin, kcp_perc_begin_saldo, 0) as aarep_perc_begin, -- сальдо на начало, если была передача, сальдо = 0
         kcp_perc_calced - iif(kcp_from_reg_id <> kcp_transfer_reg_id, iif(kcp_perc_transfer_payed + kcp_perc_transfer_writeoff - kcp_perc_begin_saldo > 0, kcp_perc_transfer_payed + kcp_perc_transfer_writeoff - kcp_perc_begin_saldo, 0), 0) as aarep_perc_calced, -- начислено в текущем офисе
         kcp_perc_payed - kcp_perc_transfer_payed as aarep_perc_payed, -- выплачено в новом офисе
         kcp_perc_writeoff - kcp_perc_transfer_writeoff as aarep_perc_writeoff, -- списано в новом офисе
         iif(kcp_period = :rep_period_end, kcp_perc_saldo, 0) as aarep_perc_end, -- сальдо на конец
         iif(kcp_from_reg_id <> kcp_transfer_reg_id, iif(kcp_perc_begin_saldo - kcp_perc_transfer_payed - kcp_perc_transfer_writeoff > 0, kcp_perc_begin_saldo - kcp_perc_transfer_payed - kcp_perc_transfer_writeoff, 0), 0) as aarep_perc_transfer, -- сумма передачи из другого офиса
         iif(kcp_from_reg_id = kcp_transfer_reg_id and kcp_period = :rep_period_begin, kcp_out_perc_begin_saldo, 0) as aarep_out_perc_begin, -- внебалансовое сальдо на начало, если была передача, сальдо = 0
         kcp_out_perc_calced - iif(kcp_from_reg_id <> kcp_transfer_reg_id, iif(kcp_perc_transfer_payed + kcp_perc_transfer_writeoff - kcp_out_perc_begin_saldo > 0, kcp_perc_transfer_payed + kcp_perc_transfer_writeoff - kcp_out_perc_begin_saldo, 0), 0) as aarep_out_perc_calced, -- внебалансовое начислено в текущем офисе
         iif(kcp_period = :rep_period_end, kcp_out_perc_saldo, 0) as aarep_out_perc_end, -- внебалансовое сальдо на конец
         iif(kcp_from_reg_id <> kcp_transfer_reg_id, 0, kcp_delayed_perc_begin) as aarep_delayed_perc_begin,
         kcp_delayed_perc_end as aarep_delayed_perc_end,
         iif(kcp_from_reg_id <> kcp_transfer_reg_id, kcp_delayed_perc_begin, 0) as aarep_delayed_perc_transfer
       from kontcalcperc
         inner join region_list(:report_region_id, 1, 3, 1)
           on id_list = kcp_from_reg_id
       where kcp_period between :rep_period_begin and :rep_period_end
       union all
       select
         kcp_from_k_id as aarep_k_id, kcp_transfer_reg_id as aarep_from_reg_id, -- текущий офис
         iif(kcp_period = :rep_period_begin, kcp_perc_begin_saldo, 0) as aarep_perc_begin, -- сальдо на начало, если была передача, сальдо = 0
         iif(kcp_perc_transfer_payed + kcp_perc_transfer_writeoff - kcp_perc_begin_saldo > 0, kcp_perc_transfer_payed + kcp_perc_transfer_writeoff - kcp_perc_begin_saldo, 0) as aarep_perc_calced, -- начислено в текущем офисе
         kcp_perc_transfer_payed as aarep_perc_payed, -- выплачено в новом офисе
         kcp_perc_transfer_writeoff as aarep_perc_writeoff, -- списано в новом офисе
         0 as aarep_perc_end, -- сальдо на конец
         -iif(kcp_perc_begin_saldo - kcp_perc_transfer_payed - kcp_perc_transfer_writeoff > 0, kcp_perc_begin_saldo - kcp_perc_transfer_payed - kcp_perc_transfer_writeoff, 0) as aarep_perc_transfer, -- сумма передачи из другого офиса
         iif(kcp_period = :rep_period_begin, kcp_out_perc_begin_saldo, 0) as aarep_out_perc_begin, -- внебалансовое сальдо на начало, если была передача, сальдо = 0
         iif(kcp_perc_transfer_payed + kcp_perc_transfer_writeoff - kcp_out_perc_begin_saldo > 0, kcp_perc_transfer_payed + kcp_perc_transfer_writeoff - kcp_out_perc_begin_saldo, 0) as aarep_out_perc_calced, -- внебалансовое начислено в текущем офисе
         0 as aarep_out_perc_end, -- сальдо на конец
         kcp_delayed_perc_begin as aarep_delayed_perc_begin,
         0 as aarep_delayed_perc_end,
         -kcp_delayed_perc_begin as aarep_delayed_perc_transfer
       from kontcalcperc
         inner join region_list(:report_region_id, 1, 3, 1)
           on id_list = kcp_transfer_reg_id
       where kcp_period between :rep_period_begin and :rep_period_end
         and kcp_from_reg_id <> kcp_transfer_reg_id
    )
    select
      aarep_k_id,
      aarep_from_reg_id,
      sum(aarep_perc_begin),
      sum(aarep_perc_calced),
      sum(aarep_perc_payed),
      sum(aarep_perc_writeoff),
      sum(aarep_perc_end),
      sum(aarep_perc_transfer),
      sum(aarep_out_perc_begin),
      sum(aarep_out_perc_calced),
      sum(aarep_out_perc_end),
      sum(aarep_delayed_perc_begin),
      sum(aarep_delayed_perc_end),
      sum(aarep_delayed_perc_transfer)
    from tab
    group by aarep_k_id, aarep_from_reg_id
  into
    :aarep_k_id,
    :aarep_from_reg_id,
    :aarep_perc_begin,
    :aarep_perc_calced,
    :aarep_perc_payed,
    :aarep_perc_writeoff,
    :aarep_perc_end,
    :aarep_perc_transfer,
    :aarep_out_perc_begin,
    :aarep_out_perc_calced,
    :aarep_out_perc_end,
    :aarep_delayed_perc_begin,
    :aarep_delayed_perc_end,
    :aarep_delayed_perc_transfer
  do
  begin
    suspend;

    k_id = aarep_k_id;
    aarep_perc_begin = 0;
    aarep_perc_calced = 0;
    aarep_perc_payed = 0;
    aarep_perc_writeoff = 0;
    aarep_perc_end = 0;
    aarep_perc_transfer = 0;
    aarep_out_perc_begin = 0;
    aarep_out_perc_calced = 0;
    aarep_out_perc_end = 0;

    for
      select
        kd_k_new
      from kontdevides
      where kd_k_orig = :k_id
        and kd_date > :report_date_end
    into
      :aarep_k_id
    do
    begin
      if (not exists(
        select *
        from kontcalcperc
        where kcp_period = :rep_period_end
          and kcp_from_k_id = :aarep_k_id)) then
      suspend;
    end
  end
end
