CREATE PROCEDURE HP_EDOC_INS_KONTPSK(
    K_ID D_ID,
    OL_ID D_ID,
    UN_ID D_ID)
RETURNS (
    KPS_ID D_ID,
    KPS_PSK D_PERCENT,
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
begin
  edl_id = null;
  res_text = '';

  select kps_id, kps_psk
  from hp_kontpsk_insert(:k_id, :ol_id)
  into kps_id, kps_psk;

  if (kps_id is not null) then
  begin
    if (exists(
        select * from knclient
        where kc_from_c_id = :k_id
          and kc_amount_approved > 0
          and f_clientcom_allowed(kc_from_c_id, 0, 1, 0) = 0))
    then
      res_text = 'Отправка сообщений данному клиенту запрещена по причине отсутствия согласий';
    else
      select edl_id, res_text
      from hp_edoc_insert('loan.psk', f_objkind('kontpsk', 'id'), :kps_id, :un_id)
      into edl_id, res_text;
  end

  suspend;
end
