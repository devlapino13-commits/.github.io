CREATE PROCEDURE FM_MAKE_KONTRACT(
    UN_ID DECIMAL(18,0),
    MA_ID INTEGER)
RETURNS (
    RES_RESULT_ID INTEGER,
    RES_RESULT_TXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable RES_FIELD_NAME varchar(255);
declare variable RES_FIELD_VALUES varchar(2048);
declare variable IS_CRITICAL integer;
declare variable RES_FIELD_NAME_PRIOR varchar(255);
declare variable K_KONTRACT_NO varchar(20);
declare variable DB_ID integer;
declare variable MIN_CRITICAL integer;
declare variable HEADER_ADDED integer;
begin
  res_result_id = 1;
  header_added = 0;
  res_result_txt = (select res_style
                    from FM_HTML_STYLES) || '<body>';
  :db_id = f_get_db_id();
  -- РКЗ: показываем и некритичные сообщения, остальные БД - как раньше
  min_critical = iif(db_id = 39801, 0, 1);
  select k_kontract_no
  from KONTRACT
  where k_from_ma_app_id = :ma_id and
        k_status <> -2
  order by k_disb_date desc
  rows 1
  into :k_kontract_no;
  if (k_kontract_no is not null) then
  begin
    res_result_txt = BLOB_APPEND(res_result_txt, f_format('<h1 class="red_text">%1</h1>', trim(iif(db_id <> 94401, 'Внимание!', 'Diqqət!'))));
    res_result_txt = BLOB_APPEND(res_result_txt, f_format('<br>%1.', trim(iif(db_id <> 94401, 'Заявка ранее была переведена в договор', 'Ərizə əvvəlcə müqaviləyə keçirilmişdir'))));
    res_result_txt = BLOB_APPEND(res_result_txt, f_format('<br>%1: <b>%2</b>', trim(iif(db_id <> 94401, 'Номер договора', 'Müqavilə nömrəsi')), k_kontract_no));
    res_result_txt = BLOB_APPEND(res_result_txt, '</body>');
    suspend;
    exit;
  end
  res_field_name_prior = '';
  for select coalesce(kf_client_fio, ''), coalesce(kf_error_text, ''), kf_error_is_critical
      from HP_CHECK_LOAN_KONDITIONS(0, 1000000, :ma_id, 'Create_Loan', :un_id)
      where kf_error_is_critical >= :min_critical
      order by kf_error_is_critical desc, kf_client_fio
      into res_field_name, res_field_values, is_critical
  do
  begin
    if (header_added = 0) then
    begin
      res_result_txt = BLOB_APPEND(res_result_txt, f_format('<h1 class="green_text">%1</h1>', iif(db_id <> 94401, 'Результаты проверки', 'Yoxlamanın nəticələri')));
      header_added = 1;
    end
    if (is_critical >= 1) then
      res_result_id = -1;
    if (res_field_name_prior = '' or res_field_name <> res_field_name_prior) then
    begin
      res_result_txt = BLOB_APPEND(res_result_txt, '<br><br><b>' || res_field_name || '</b>:');
    end
    if (res_field_values <> '') then
    begin
      if (is_critical = 0) then
        res_result_txt = BLOB_APPEND(res_result_txt, '<br><font color="#808080">Информация</font>: ' || res_field_values);
      else if (is_critical = 1) then
        res_result_txt = BLOB_APPEND(res_result_txt, f_format('<br><font color="#F08205">%1</font>: ', iif(db_id <> 94401, 'Критично', 'Kritik')) || res_field_values);
      else
        res_result_txt = BLOB_APPEND(res_result_txt, f_format('<br><font color="red">%1</font>: ', iif(db_id <> 94401, 'Очень критично', 'Kritik hədd')) || res_field_values);
      res_result_txt = BLOB_APPEND(res_result_txt, '<br>');
    end
    res_field_name_prior = res_field_name;
  end
  if (res_result_id = -1) then
  begin
    res_result_txt = BLOB_APPEND(res_result_txt, '</body>');
    suspend;
    exit;
  end
  select k_kontract_no
  from MAKE_NEW_KONTRACT(:ma_id, 0, :un_id)
  into :k_kontract_no;
  res_result_txt = BLOB_APPEND(res_result_txt, f_format('<h1 class="green_text">%1</h1>', iif(db_id <> 94401, 'Успешно', 'Uğurlu')));
  res_result_txt = BLOB_APPEND(res_result_txt, f_format('<br>%1.', iif(db_id <> 94401, 'Заявка успешно переведена в договор', 'Ərizə uğurla müqaviləyə dəyişdirildi')));
  res_result_txt = BLOB_APPEND(res_result_txt, f_format('<br>%1: <b>%2</b>', iif(db_id <> 94401, 'Номер договора', 'Müqavilə nömrəsi'), k_kontract_no));
  res_result_txt = BLOB_APPEND(res_result_txt, '</body>');
  suspend;
end
