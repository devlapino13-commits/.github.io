CREATE PROCEDURE NBKI_GET_STATUS_DATE(
    K_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    LOAN_STATE VARCHAR(50),
    STATE_DATE DATE)
AS
declare variable delinq_date date;
declare variable oks_date_end date;
declare variable k_disb_date date;
declare variable k_fact_closed_date date;
begin
  delinq_date = null;
  select oks_delinq_date
  from oksummary
  where :rep_date between oks_date_begin and oks_date_end and
        oks_close_state = 0 and
        oks_from_k_id = :k_id
  into delinq_date;

  state_date = null;
  if (delinq_date is null) then
    loan_state = 'Closed';
  else
  if (delinq_date < rep_date) then
  begin
    loan_state = 'Delinq';
    state_date = delinq_date;
  end
  else
  begin
    loan_state = 'NonDelinq';

    oks_date_end = null;
    select oks_date_end
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_end > oks_delinq_date and
          oks_date_end <= :rep_date and
          oks_close_state = 0
    order by oks_date_end desc
    rows 1
    into oks_date_end;
    if (oks_date_end is not null) then
      state_date = oks_date_end + 1;
  end

  if (state_date is null) then
  begin
    select k_disb_date, k_fact_closed_date
    from kontract kont
    where kont.k_id = :k_id
    into k_disb_date, k_fact_closed_date;

    if (loan_state = 'Closed') then
      state_date = k_fact_closed_date;
    else
    if (loan_state = 'NonDelinq') then
      state_date = k_disb_date;
  end
  suspend;
end
