CREATE PROCEDURE HP_OBSUMRECOST_CHECK(
    CHECK_DATE D_DATE)
RETURNS (
    RES_AL_ID D_STR_20,
    RES_SALDO_BEGIN D_CURRENCY,
    RES_OB_DT D_CURRENCY,
    RES_OB_KT D_CURRENCY,
    RES_SALDO_END D_CURRENCY,
    NEW_OB_DT D_CURRENCY,
    NEW_OB_KT D_CURRENCY,
    NEW_SALDO_END D_CURRENCY)
AS
begin
  for select al_id
  from acc_list
  order by al_id
  into res_al_id
  do
  begin
    res_saldo_begin = f_al_bal(res_al_id, check_date - 1);

    res_ob_dt = 0;
    select sum(ob_summa)
    from operbook
    where ob_account_dt = :res_al_id and  
          ob_ol_date = :check_date and
          ob_ol_is_active = 1
    into res_ob_dt;

    if (res_ob_dt is null) then
      res_ob_dt = 0;

    res_ob_kt = 0;
    select sum(ob_summa)
    from operbook
    where ob_account_kt = :res_al_id and
          ob_ol_date = :check_date and
          ob_ol_is_active = 1
    into res_ob_kt;

    if (res_ob_kt is null) then
      res_ob_kt = 0;

    res_saldo_end = res_saldo_begin + res_ob_dt - res_ob_kt;

    select dt_ekv, kt_ekv
    from hp_al_obor(:res_al_id, :check_date, :check_date)
    into :new_ob_dt, :new_ob_kt;

    new_saldo_end = f_al_bal(res_al_id, check_date);

    if (res_ob_dt <> new_ob_dt or
        new_ob_kt <> new_ob_kt or
        res_saldo_end <> new_saldo_end) then
      suspend;
  end
end
