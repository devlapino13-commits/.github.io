CREATE PROCEDURE OKT_PANTERA_LIST_TMP(
    REP_DATE DATE,
    PROJECT_CODE STR_100,
    K_FOLDER_NAME STR_100,
    KONTRACT_NO STR_100,
    DELINQ_DAYS_MIN D_INTEGER,
    DELINQ_DAYS_MAX D_INTEGER,
    CALLED_OPER STR_100 = '')
RETURNS (
    K_ID ID_TYPE,
    KF_ID ID_TYPE,
    K_KONTRACT_NO VARCHAR(50),
    DELINQ_DAYS D_INTEGER,
    OKS_KC_ID ID_TYPE,
    BALANCE_OS CURR_TYPE,
    DELINQ_OS CURR_TYPE,
    REG_ID D_INTEGER,
    CLI_QTY_COUNTER D_INTEGER)
AS
declare variable SBL_ID D_INTEGER;
declare variable K_ID_TMP ID_TYPE;
declare variable SQL_TEXT D_STR_8100;
declare variable KF_ID_TMP D_ID;
declare variable DB_ID D_INTEGER;
declare variable RD_ID D_INTEGER;
begin
  if (CALLED_OPER = 'OKTELL_LIST') then
  begin
    rd_id = f_get_rd_id(:project_code);

    for select pee.ppe_from_k_id
    from pantera_preexec pee
    where pee.ppe_project_rd_id = :rd_id
    into :k_id
    do
    begin
      select kont.k_last_reg_id,
             kont.k_kolich_client_approved,
             kont.k_kontract_no
       from kontract kont
       where kont.k_id = :k_id
       into :reg_id,
            :cli_qty_counter,
            :k_kontract_no;
              
      suspend;
    end

    exit;
  end

  db_id = f_get_db_id();

  k_folder_name = trim(coalesce(k_folder_name, ''));
  kontract_no = trim(coalesce(kontract_no, ''));
  delinq_days_min = coalesce(delinq_days_min, 0);
  delinq_days_max = coalesce(delinq_days_max, 0);

  if ((project_code containing ('OKTELL.Projects.Pantera')) or
      (project_code containing ('OKTELL.Projects.ks_finance')) or
      (project_code containing ('OKTELL.Projects.Investigation_And_Recovery'))) then
  begin
    if (delinq_days_min < 1) then
      delinq_days_min = 1;
    if (delinq_days_max = 0) then
      delinq_days_max = 9999;

    sbl_id = (select cast(rd.rd_str_id as d_integer) from refdata rd where rd.rd_code = :project_code) ;
  end

  if (kontract_no <> '') then
  begin
    select kont.k_id
    from kontract kont
    where kont.k_kontract_no = :kontract_no
    into :k_id_tmp;
  end

  if (k_folder_name <> '') then
  begin
    select kal.kal_from_kf_id
    from knaktlist kal
    where kal.kal_new_kf_name = :k_folder_name
    and :rep_date between kal.kal_date and kal.kal_date_end
    and kal.is_delete = 0
    into :kf_id_tmp;
  end

  sql_text = 'select sba.sba_from_k_id,
              kont.k_from_kf_id,
              cast(:rep_date as date) - oks_delinq_date as delinq_days,
              oks.oks_kc_id,
              oks.oks_balance,
              kont.k_last_reg_id,
              kont.k_kolich_client_approved
            from oksummary oks
            inner join sb_akt sba on sba.sba_from_k_id = oks.oks_from_k_id
            inner join kontract kont on kont.k_id = sba.sba_from_k_id
            where :rep_date between sba.sba_date_begin and sba.sba_date_end and
                sba.sba_date_begin <= sba.sba_date_end and
                sba_is_delete = 0 and
                sba.sba_from_sbl_id = :sbl_id and
                sba.sba_status = 2 and
                :rep_date between oks_date_begin and oks_date_end and
                oks_delinq_date < :rep_date and
                cast(:rep_date as date) - oks_delinq_date between :delinq_days_min and :delinq_days_max and
                oks.oks_close_state = 0';

  if (k_id_tmp is not null) then
    sql_text = sql_text||' and kont.k_id = :k_id_tmp';
  else
    sql_text = sql_text||' and :k_id_tmp is null';

  if (kf_id_tmp is not null) then
    sql_text = sql_text||' and kont.k_from_kf_id = :kf_id_tmp';
  else
    sql_text = sql_text||' and :kf_id_tmp is null';

  for execute statement(sql_text)
    (rep_date := rep_date,
     sbl_id := sbl_id,
     delinq_days_min := delinq_days_min,
     delinq_days_max := delinq_days_max,
     k_id_tmp := k_id_tmp,
     kf_id_tmp := kf_id_tmp)
  into :k_id, :kf_id, :delinq_days, :oks_kc_id, :balance_os,
       :reg_id, :cli_qty_counter
  do
  begin
    select kont.k_kontract_no
    from kontract kont
    where kont.k_id = :k_id
    into :k_kontract_no;

    delinq_os = 0;
    select sum(ok_os + ok_sh_os)
    from operkred
    where ok_from_k_id = :k_id and
          ok_from_c_id = 0 and
          ok_ol_date <= :rep_date and
          ok_sh_date < :rep_date and
          ok_ol_is_active = 1 and
          ok_type <= 7
    into :delinq_os;

    balance_os = coalesce(balance_os, 0);
    delinq_os = coalesce(delinq_os, 0);

    if (db_id = 81001) then
    begin
        if (not((balance_os < 500) or (delinq_os < 500))) then
            suspend;
    end
    else
        suspend;
  end

  --if (project_code containing ('OKTELL.Projects.Investigation_And_Recovery')) then
  if (sbl_id is not null) then
  begin
    sql_text = '
    select sba.sba_from_k_id,
           k_kontract_no,
           kont.k_from_kf_id,
           kal.kal_new_kf_officer,
           sum(okf.oko_os + okf.oko_sh_os),
           sum(okf.oko_os + okf.oko_perc + okf.oko_other + okf.oko_sh_os + okf.oko_sh_perc + okf.oko_sh_other),
           kont.k_last_reg_id,
           kont.k_kolich_client_approved
    from operkroff okf
    inner join sb_akt sba on sba.sba_from_k_id = okf.oko_from_k_id
    inner join kontract kont on kont.k_id = okf.oko_from_k_id
    inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
    where okf.oko_ol_is_active = 1 and
    :rep_date between sba.sba_date_begin and sba.sba_date_end and
    sba.sba_date_begin <= sba.sba_date_end and
    sba_is_delete = 0 and
    sba.sba_from_sbl_id = :sbl_id and
    sba.sba_status = 2 and
    kal.is_delete = 0 and
    :rep_date between kal.kal_date and kal.kal_date_end';

    if (k_id_tmp is not null) then
        sql_text = sql_text||' and kont.k_id = :k_id_tmp';
    else
        sql_text = sql_text||' and :k_id_tmp is null';
    
    if (kf_id_tmp is not null) then
        sql_text = sql_text||' and kont.k_from_kf_id = :kf_id_tmp';
    else
        sql_text = sql_text||' and :kf_id_tmp is null';

    sql_text = sql_text||'
    group by sba.sba_from_k_id, k_kontract_no, kont.k_from_kf_id, kal.kal_new_kf_officer, kont.k_last_reg_id, kont.k_kolich_client_approved
    having sum(okf.oko_os + okf.oko_perc + okf.oko_other + okf.oko_sh_os + okf.oko_sh_perc + okf.oko_sh_other) > 0';

    for execute statement(sql_text)
    (rep_date := rep_date,
     sbl_id := sbl_id,
     k_id_tmp := k_id_tmp,
     kf_id_tmp := kf_id_tmp)
    into :k_id, :k_kontract_no, :kf_id, :oks_kc_id, :balance_os, delinq_os,
         :reg_id, :cli_qty_counter
    do
    begin
        delinq_os = balance_os;
        delinq_days = 999;
        
        balance_os = coalesce(balance_os, 0);
        delinq_os = coalesce(delinq_os, 0);
        
        if (db_id = 81001) then
        begin
            if (not((balance_os < 500) or (delinq_os < 500))) then
                suspend;
        end
        else
            suspend;
    end
  end
end
