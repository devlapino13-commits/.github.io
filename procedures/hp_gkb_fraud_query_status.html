CREATE PROCEDURE HP_GKB_FRAUD_QUERY_STATUS(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID)
RETURNS (
    DECISION_STATUS D_SMALLINT,
    DECISION D_SMALLINT,
    RATING D_STR_CODE,
    GPC_ID D_ID)
AS
declare variable EIL_STATUS D_SMALLINT;
declare variable EIL_ID D_ID;
declare variable IS_ERROR D_SMALLINT;
begin
  decision_status = null;
  decision = null;
  rating = null;
  gpc_id = null;

  -- 1. Есть ли уже сохранённый результат по этому объекту?
  select first 1 gpc_id, gpc_rating, gpc_decision, gpc_is_error
  from gkb_pay_check
  where gpc_object_kind = :obj_kind
    and gpc_object_id = :obj_id
  order by gpc_id desc
  into :gpc_id, :rating, :decision, :is_error;

  if (gpc_id is not null) then
  begin
    if (is_error = 1) then
      decision_status = -3;  -- категория E — пропускаем
    else
      decision_status = 0;   -- решение получено
    suspend;
    exit;
  end

  -- 2. Результата ещё нет. Смотрим статус запроса в extinfolist по объекту.
  select first 1 eil.eil_id, eil.eil_status
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eic.eic_code = 'gkb_pay_checks_core'
    and eil.eil_object_kind = :obj_kind
    and eil.eil_object_id = :obj_id
  order by eil.eil_id desc
  into :eil_id, :eil_status;

  if (eil_status is null) then
    decision_status = -1;              -- запрос не создавался
  else if (eil_status in (-1, 0, 1)) then
    decision_status = -2;              -- в работе
  else if (eil_status in (-2, -3)) then
    decision_status = -3;              -- ошибка сервиса → E
  else
    decision_status = -2;              -- статус 2, но результат ещё не разобран → ждём

  suspend;
end
