CREATE PROCEDURE CB_RU_SB_2015_08_TEMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    SBA_DATE_BEGIN DATE,
    SBA_DATE_END DATE,
    DELINQ_DAY_BEGIN INTEGER,
    ACTUAL_DATE_BEGIN DATE,
    ACTUAL_DATE_END DATE,
    DELINQ_DAY_END INTEGER,
    IS_GRAPH SMALLINT,
    KOL_CLI_BEFORE_60 INTEGER,
    OS_REPAY CURR_TYPE,
    PERC_REPAY CURR_TYPE,
    FINE_REPAY CURR_TYPE,
    KOL_CLI_BEGIN INTEGER,
    KOL_CLI_PEREDAN INTEGER,
    KOL_CLI_OTOZVAN INTEGER,
    IS_WOFF D_SMALLINT)
AS
declare variable SBA_FROM_K_ID decimal(18,0);
begin
  for select sba_from_k_id, sba_date_begin, sba_date_end, 0 is_woff
      from sb_list, sb_akt, kontstate
      where sbl_category = 'SB_MBF_RU' and
            sbl_id = sba_from_sbl_id and
            sba_is_delete = 0 and
            sba_date_begin <= :rep_date_end -- будет рассматриватьс¤ все переданные папки до конец период
            and
            sba_date_end >= :rep_date_begin and
            sba_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_begin - 1 between ks_date_begin and ks_date_end -- чтобы кредиты на начало периода были активными
--            and sba_from_k_id = 101010026853200101
      union
      select oko_from_k_id, sba_date_begin, sba_date_end, 1
      from operkroff, sb_list, sb_akt
      where oko_ol_is_active = 1
      and oko_ol_date < :rep_date_begin
      and oko_from_k_id = sba_from_k_id
      and sbl_category = 'SB_MBF_RU'
      and sbl_id = sba_from_sbl_id
      and sba_is_delete = 0
      and sba_date_begin <= :rep_date_end
      and sba_date_end >= :rep_date_begin
--      and sba_from_k_id = 101010026853200101
      group by  1,2,3
      having sum(oko_os + oko_perc + oko_sh_os + oko_sh_perc) <> 0
      into sba_from_k_id, sba_date_begin, sba_date_end, is_woff
  do
  begin

    select k_kontract_no, k_kolich_client_approved
    from kontract
    where k_id = :sba_from_k_id
    into k_kontract_no, k_kolich_client_approved;

--  проверка просрочки на начало даты
    if (sba_date_begin < rep_date_begin) then
      actual_date_begin = rep_date_begin;
    else
      actual_date_begin = sba_date_begin;
    if (sba_date_end > rep_date_end) then
      actual_date_end = rep_date_end;
    else
      actual_date_end = sba_date_end;

    delinq_day_begin = null;
    select cast(:actual_date_begin as date) - 1 - oks_delinq_date -- провер¤етс¤ на начало дн¤ т.е. на конец предыдущего дн¤
    from oksummary
    where oks_from_k_id = :sba_from_k_id and
          :actual_date_begin - 1 between oks_date_begin and oks_date_end and
          :actual_date_begin - 1 >= oks_delinq_date
    into delinq_day_begin;
--    if (delinq_day_begin is null) then delinq_day_begin = 0;
--  если delinq_day_begin = null в результате то в —Ѕ был передан без просрочки
    delinq_day_end = null;
    is_graph = 0;
    os_repay = 0;
    perc_repay = 0;
    fine_repay = 0;
    kol_cli_before_60 = 0;
    kol_cli_begin = 0;
    kol_cli_peredan = 0;
    kol_cli_otozvan = 0;

    if (delinq_day_begin is not null or is_woff = 1) then
    begin
--    ѕогашенна¤ сумма группы
      os_repay = null;
      perc_repay = null;
      fine_repay = null;
      select -sum(ok_os), -sum(ok_perc), -sum(ok_fine)
      from operkred, operlist
      where ok_from_k_id = :sba_from_k_id and
            ok_ol_is_active = 1 and
            ok_type = 4 and
            ok_from_c_id = 0 and
            ok_ol_date between :actual_date_begin and :actual_date_end and
            ok_from_ol_id = ol_id and
            ol_memo not starting 'Корректировка согласно протокола кредитного комитета №29 от 01.04.2019г. Контракт №:'
      into os_repay, perc_repay, fine_repay;

      if (is_woff = 1) then
      begin
        select -sum(oko_os), -sum(oko_perc), -sum(oko_other) from operkroff
        where oko_from_k_id = :sba_from_k_id
        and oko_ol_is_active = 1
        and oko_ol_date between :actual_date_begin and :actual_date_end
        into os_repay, perc_repay, fine_repay;
      end

      if (os_repay is null) then
        os_repay = 0;
      if (perc_repay is null) then
        perc_repay = 0;
      if (fine_repay is null) then
        fine_repay = 0;

      if (delinq_day_begin <= 60 /*or is_woff = 1*/) then
      begin
--    проверка просрочки на конец даты
        select cast(:rep_date_end as date) - oks_delinq_date
        from oksummary
        where oks_from_k_id = :sba_from_k_id and
              :rep_date_end between oks_date_begin and oks_date_end and
              :rep_date_end >= oks_delinq_date and
              oks_close_state = 0
        into delinq_day_end;

--    если на конец дату не имеетс¤ просрочка то данный контракт встал в график
--    если delinq_day_end <> null то имеет просрочку
        is_graph = 0;
        if (delinq_day_end is null and is_woff = 0) then
        begin
          is_graph = 1;
        end

--        кол-во клиентов на начало у которых просрочка менее 60 дней
        if (sba_date_begin < :rep_date_begin) then
          kol_cli_begin = k_kolich_client_approved;
--        кол-во клиентов переданных в течении периода у которых просрочка менее 60 дней
        if (sba_date_begin between :rep_date_begin and :rep_date_end) then
          kol_cli_peredan = k_kolich_client_approved;
--        кол-во клиентов вставших в график или погасившейс¤ 4 тыс и более сумму.
        if (is_graph = 1 or os_repay + perc_repay >= 4000.0) then
          kol_cli_before_60 = k_kolich_client_approved;
--        кол-во отозванных клиентов в которых никаких действии не были
        if (sba_date_end between :rep_date_begin and :rep_date_end and
            kol_cli_before_60 = 0) then
          kol_cli_otozvan = k_kolich_client_approved;
--        добавлено условие по задаче на Jira 2243
        if (os_repay + perc_repay + fine_repay = 0) then
          kol_cli_before_60 = 0;

      end
    end

    suspend;
  end
end
