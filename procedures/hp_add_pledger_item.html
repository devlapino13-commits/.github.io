CREATE PROCEDURE HP_ADD_PLEDGER_ITEM(
    UN_ID D_ID,
    C_ID D_ID,
    DRL_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable drm_code d_str_255;
declare variable obj_kind d_integer;
declare variable rd_id d_integer;
declare variable k_id d_id;
declare variable z_id d_id;
declare variable acl_id d_id;
begin
    -- RFCRU-2770 / RFCRU-2771
    if (f_get_db_id() <> 81001 and f_get_const('fm_pledger_add') <> 1) then
        exit;

    drm_code = '';        
    select coalesce(drm.drm_code, '')
    from docrecoglist drl
    inner join docrecogmain drm on drl.drl_from_drm_id = drm.drm_id
        and drm.drm_code in ('rus.pts.type1', 'rus.sts.new')
    where drl.drl_id = :drl_id
    into :drm_code;

    if (drm_code not in ('rus.pts.type1', 'rus.sts.new')) then
        exit;

    k_id = 0;
    z_id = 0;
    res_text = '';
    obj_kind = f_objkind('kontract', 'id');
    --rd_id = f_get_rd_id('ActionGroup.Check_Loan_Data');

    select
        coalesce(kc.kc_from_k_id, 0),
        coalesce(z.z_id, 0)
    from knclient kc
    inner join kontract k on kc.kc_from_k_id = k.k_id
        and k.k_status = 0
        and k.k_fact_disb_date is null
    inner join product pr on k.k_from_pr_id = pr.pr_id
        and (pr.pr_code containing '.zalog.' or pr.pr_code containing '.zalog_car.')
    inner join zalog z on kc.kc_from_k_id = z.z_from_k_id
        and z_type = 1
        and z_subtype = 4
    where kc.kc_from_c_id = :c_id -- RFCRU-2842 / Раньше было привязка на залогодателя, теперь на клиента
    into
        :k_id,
        :z_id;

    select rd_id
    from hp_get_check_loan_data(:k_id)
    into:rd_id;

    if (k_id = 0) then
        exit;

    -- проверка в КК - на этапе обработки
    if (exists
        (
            select 1
            from actionlist
            where acl_object_kind = :obj_kind
                and acl_object_id = :k_id
                and acl_from_rd_id = :rd_id
                and acl_status = 1
        )
    ) then
    begin
        res_text = 'Договор находится на рассмотрении КК. Изменения запрещены.';
        suspend;
        exit;
    end

    for select acl_id
    from actionlist
    where acl_object_kind = :obj_kind
        and acl_object_id = :k_id
        and acl_from_rd_id = :rd_id
        and acl_status = 2
    into :acl_id
    do
    begin
        update actionlist
        set acl_status = 0,
            acl_from_un_id = :un_id
        where acl_id = :acl_id;

        res_text = 'Повторно отправлен на КК в связи с изменениями в залоге';
        execute procedure hp_objectprops_memo_set(:obj_kind, :k_id, -1, :rd_id, :res_text, :un_id);
    end
    res_text = '';

    -- RFCRU-2770 Добавление ПТС через МП
    if (drm_code = 'rus.pts.type1') then
        update zalog
        set z_name = f_docrecog_value(:drl_id, 'model')
        where z_from_k_id = :k_id
            and z_id = :z_id;

    -- RFCRU-2771 Добавление СТС через МП
    if (drm_code = 'rus.sts.new') then
        update zalog
        set z_registry_no = f_docrecog_value(:drl_id, 'plate'),
            z_registry_date = f_docrecog_value(:drl_id, 'date_begin')
        where z_from_k_id = :k_id
            and z_id = :z_id;

    suspend;
end
