CREATE PROCEDURE FM_FIND_CLIENT(
    FIND_FIO VARCHAR(100),
    UN_ID DECIMAL(18,0))
RETURNS (
    C_ID TYPE OF COLUMN CLIENTS.C_ID,
    C_NAME_FIO TYPE OF COLUMN CLIENTS.C_NAME_FIO,
    C_BIRTH_DATE TYPE OF COLUMN CLIENTS.C_BIRTH_DATE,
    C_PASS_NO TYPE OF COLUMN CLIENTS.C_PASS_NO,
    ACCESS_LEVEL INTEGER)
AS
declare variable rows_count_max integer;
declare variable isfound integer;
declare variable db_id d_integer;
declare variable c_inn d_str_20;
begin
  db_id = f_get_db_id();
  rows_count_max = 40; -- Максимальное количество возвращаемых строк
  access_level = -1;

  find_fio = coalesce(find_fio, '');

  isfound = 0;
  for select distinct fc_id
      from fm_find_client_tmp(:find_fio)
      rows :rows_count_max
      into :c_id
  do
  begin
    isfound = isfound + 1;
    for select c_name_fio, c_birth_date, c_pass_no, c_inn
        from clients
        where c_id = :c_id and
              c_type = 0 -- Берем только физ. лиц
        into c_name_fio, c_birth_date, c_pass_no, c_inn
    do
    begin
      if (char_length(c_pass_no) >= 5) then
      begin
        c_pass_no = left(c_pass_no, char_length(c_pass_no) - 5) || '***' || right(c_pass_no, 2);
      end

      if (char_length(c_inn) >= 7 and db_id = 39801) then  -- RFCKG-4140 Поменяли номер паспорта на ИНН при отображении стр. Клиенты в МП
      begin
        c_inn = left(c_inn, char_length(c_inn) - 7) || '****' || right(c_inn, 2);
        c_pass_no = c_inn;
      end

      select access_lev
      from fm_client_status(:un_id, :c_id)
      into access_level;

      suspend;
    end
  end
  if (isfound > 0) then
    exit;

  if (find_fio <> '') then
  begin
    -- Поиск клиентов через fiolist
    for select distinct c_id, c_name_fio, c_birth_date, c_pass_no, c_inn
        from clients, hp_find_from_namelist(0, 0, :find_fio)
        where c_id = object_id and
              c_type = 0 -- Берем только физ. лиц
        order by res_priority, c_name_fio
        rows :rows_count_max
        into c_id, c_name_fio, c_birth_date, c_pass_no, c_inn
    do
    begin
      if (char_length(c_pass_no) >= 5) then
      begin
        c_pass_no = left(c_pass_no, char_length(c_pass_no) - 5) || '***' || right(c_pass_no, 2);
      end

      if (char_length(c_inn) >= 7 and db_id = 39801) then  -- RFCKG-4140 Поменяли номер паспорта на ИНН при отображении стр. Клиенты в МП
      begin
        c_inn = left(c_inn, char_length(c_inn) - 7) || '****' || right(c_inn, 2);
        c_pass_no = c_inn;
      end

      select access_lev
      from fm_client_status(:un_id, :c_id)
      into access_level;
      suspend;
    end
  end
  else
  begin
    -- Если строка поиска пустая, выдаем список клиентов офиса/ов пользователя
    -- у которых есть заявка, и нет фото
    for with last_clients
        as (select cul_from_c_id, max(cul_datetime) as cul_datetime
            from clientlast
            where cul_from_un_id = :un_id
            group by cul_from_c_id
            order by 2 desc)
        select c_id, c_name_fio, c_birth_date, c_pass_no, c_inn
        from clients, last_clients
        where c_id = cul_from_c_id
        order by cul_datetime desc
        rows :rows_count_max
        into c_id, c_name_fio, c_birth_date, c_pass_no, c_inn
    do
    begin
      if (char_length(c_pass_no) >= 5) then
      begin
        c_pass_no = left(c_pass_no, char_length(c_pass_no) - 5) || '***' || right(c_pass_no, 2);
      end

      if (char_length(c_inn) >= 7 and db_id = 39801) then  -- RFCKG-4140 Поменяли номер паспорта на ИНН при отображении стр. Клиенты в МП
      begin
        c_inn = left(c_inn, char_length(c_inn) - 7) || '****' || right(c_inn, 2);
        c_pass_no = c_inn;
      end

      select access_lev
      from fm_client_status(:un_id, :c_id)
      into access_level;
      suspend;
    end
  end
end
