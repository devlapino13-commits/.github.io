CREATE PROCEDURE FIND_PAYSYS_K_ID(
    PROJECT_ID INTEGER,
    CARD_NO VARCHAR(20),
    REP_DATE DATE)
RETURNS (
    K_CARD_NO VARCHAR(20),
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    K_KF_NAME VARCHAR(20),
    K_REGION_ID INTEGER,
    K_REGION_NAME VARCHAR(100),
    K_STATE INTEGER,
    K_PAYSYS_ID INTEGER)
AS
declare variable psi_id integer;
declare variable psi_date_begin date;
declare variable psi_date_end date;
begin
  for select psi_id, psi_code, psi_date_begin, psi_date_end
      from paysys_item
      where psi_from_psp_id = :project_id and
            psi_code like '%' || :card_no || '%'
      into :psi_id, :k_card_no, :psi_date_begin, :psi_date_end
  do
  begin
    k_paysys_id = psi_id;
    for select psl_object_id
        from paysys_list
        where psl_object_kind = 3 -- Кредиты
              and
              psl_from_psi_id = :psi_id and
              psl_is_delete = 0 and
              psl_date <= :rep_date
        group by psl_object_id
        having sum(psl_sign) > 0
        into :k_id
    do
    begin
      select k_kontract_no, kal_new_kf_name, kal_new_reg_id, reg_code || ' ' || reg_name, ks_state
      from region, kontract, knaktlist, kontstate
      where k_id = :k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_date <= :rep_date and
            kal_date_end >= :rep_date and
            kal_new_reg_id = reg_unicode and
            ks_from_k_id = :k_id and
            ks_ol_is_delete = 0 and
            ks_date_begin <= :rep_date and
            ks_date_end >= :rep_date
      into :k_kontract_no, :k_kf_name, :k_region_id, :k_region_name, :k_state;
      suspend;
    end
  end
end
