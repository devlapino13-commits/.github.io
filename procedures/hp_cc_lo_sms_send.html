CREATE PROCEDURE HP_CC_LO_SMS_SEND(
    CK_ID INTEGER,
    C_ID ID_TYPE,
    SMS_TXT STR_255)
AS
begin
  sms_txt = trim(coalesce(sms_txt, ''));
  sms_txt = replace(sms_txt, ascii_char(13), ' ');
  sms_txt = replace(sms_txt, ascii_char(10), ' ');
  sms_txt = f_replace_all(sms_txt, '  ', ' ');

  if ((sms_txt <> '') and
      (c_id is not null)) then
  begin
    insert into cc_website_lo_sms (wls_from_ck_id, wls_lo_id, wls_sms_txt)
    values (:ck_id, :c_id, :sms_txt);

    execute procedure fsrobot_insert_event('DoMailing', 'WEBMINIAPP_LO_NOTICE',
        left(cast(c_id as varchar(13)) || ';' || sms_txt, 255), 1000000000000);
  end
end
