CREATE PROCEDURE FM_HTML_MAX_DELINQ_DAYS(
    C_ID D_ID)
RETURNS (
    RES_TEXT D_INTEGER)
AS
declare variable K_ID D_ID;
declare variable DELING D_SMALLINT;
declare variable KS_STATE D_SMALLINT;
begin
  --процедура для подсчета максимальных просроченных дней клиента (будет отоброжаться в меню "информация о клиенте" (после "ограниченный режим просмотра"))
  res_text = 0;
  for select k.k_id, ks.ks_state
      from kontract k
      inner join kontstate ks on ks.ks_from_k_id = k.k_id
      inner join knclient kc on kc.kc_from_k_id = k.k_id
      where kc.kc_from_c_id = :c_id and
            kc.kc_amount_approved > 0 and
            ks.ks_state in (1, 2) and
            current_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_ol_is_delete = 0
      order by k.k_disb_date desc
      rows 3
      into :k_id, :ks_state
  do
  begin
    deling = 0;
    if (:ks_state = 1) then
    begin
      select coalesce(current_date - oks.oks_delinq_date, 0)
      from oksummary oks
      where current_date between oks.oks_date_begin and oks.oks_date_end and
            current_date - oks.oks_delinq_date > 0 and
            oks_close_state = 0 and
            oks.oks_from_k_id = :k_id
      into :deling;
    end
    if (:ks_state = 2) then
    begin
      select coalesce(max(ok.ok_ol_date - ok_sh_date), 0)
      from operkred ok
      where ok.ok_ol_is_active = 1 and
            ok.ok_ol_date <= current_date and
            ok.ok_ol_date > ok_sh_date and
            ok_type < 7 and
            ok_from_k_id = :k_id and
            ok_os + ok_perc < 0 and
            ok_from_c_id = 0
      into :deling;
    end
    if (:deling > :res_text) then
      :res_text = :deling;
  end
  suspend;
end
