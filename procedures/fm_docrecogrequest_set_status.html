CREATE PROCEDURE FM_DOCRECOGREQUEST_SET_STATUS(
    DRR_ID D_ID,
    NEW_STATUS D_INTEGER,
    UN_ID D_ID,
    NEW_REASON_RD_ID D_INTEGER = null)
RETURNS (
    RES_TEXT D_STR_255,
    NEED_REFRESH D_BOOL,
    RES_ID D_BOOL)
AS
declare variable drr_status d_integer;
declare variable drl_status d_integer;
begin
  res_id = 1;
  need_refresh = 0;
  res_text = 'Не реализовано, обратитесь к программистам';

  select drr_status, drl_status
  from docrecogrequest
    left join docrecoglist on drl_id = drr_from_drl_id
  where drr_id = :drr_id
  into drr_status, drl_status;

  if (drr_status is null or drr_status < -2 or drr_status > 2) then
  begin
    res_text = 'Некорректный статус запроса:' || coalesce(drr_status, 'null');
    suspend;
    exit;
  end

  if (new_status = -2) then
  begin
    if (drr_status in (-1, 0)) then
    begin
      execute procedure hp_docrecogrequest_set_status(drr_id, -2, new_reason_rd_id);
      res_text = 'Запрос аннулирован';
      res_id = 0;
      need_refresh = 1;
    end
    else
      res_text = 'Запрос не может быть аннулирован в этом статусе';
  end

  if (new_status = -1) then
  begin
    if (drr_status = 2) then
      res_text = 'Документ ранее подтвержден';
    else if (drr_status = -2) then
      res_text = 'Документ ранее аннулирован';
    else if (coalesce(drl_status, -2) <> -1) then
      res_text = 'Некорректный статус документа';
    else if (new_reason_rd_id is null) then
      res_text = 'Не указана причина доработки';
    else
    begin
      execute procedure hp_docrecogrequest_set_status(drr_id, -1, new_reason_rd_id);
      res_text = 'Запрос отправлен на доработку';
      res_id = 0;
      need_refresh = 1;
    end
  end

  if (new_status = 2) then
  begin
    execute procedure hp_docrecogrequest_set_status(drr_id, 2);
    res_id = 0;
    res_text = 'Документ успешно сохранен';
    need_refresh = 1;
  end
  suspend;
end
