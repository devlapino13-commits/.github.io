CREATE PROCEDURE CRM_WMA_ANKETA_INFO_PART(
    WMA_ID INTEGER,
    FIO STR_255,
    BIRTH_DATE DATE,
    PHONE_NO STR_100,
    FIND_BY_WMA_ID SMALLINT)
RETURNS (
    JSON_VALUE D_BLOB_TEXT)
AS
declare variable c_id id_type;
declare variable res_comment str_255;
declare variable c_name_fio str_100;
declare variable c_birth_date str_10;
declare variable anketa_memo d_str_8100;
declare variable un_name str_100;
declare variable wmat_datetime timestamp;
declare variable fio_tmp str_255;
declare variable birth_date_tmp date;
declare variable phone_no_tmp str_100;
declare variable wma_c_id id_type;
declare variable is_first_row d_smallint;
declare variable by_closed_ank d_smallint;
begin
  if (find_by_wma_id = 1) then
  begin
      select wmt.wmat_name,
             wmt.wmat_birthday,
             wma.wma_phone_no,
             wma.wma_from_c_id
      from cc_website_miniapp_tmp wmt
      inner join cc_website_miniapp wma on wma.wma_last_wmat_id = wmt.wmat_id
      where wma.wma_id = :wma_id
      into :fio_tmp,
           :birth_date_tmp,
           :phone_no_tmp,
           :wma_c_id;

      fio = fio_tmp;
      birth_date = birth_date_tmp;
      phone_no = phone_no_tmp;
  end

  json_value = '{ "wma_info_part": {' ||
                     '"operator_comments": [';

  is_first_row = 1;

  for select coalesce(wmc.wmc_memo,''),
             iif(un.un_name is null, '[Не указан]', un.un_name),
             wmc.wmc_datetime,
             1 as by_closed_ank
  from cc_website_miniapp_close wmc
  left join username un on wmc.wmc_un_id = un.un_id
  where wmc.wmc_from_wma_id = :wma_id
  and wmc.wmc_memo <> ''
  union all
  select wmat_anketa_memo,
         un_name,
         wmat_datetime,
         by_closed_ank
  from
  (
      select wmt.wmat_anketa_memo,
             iif(un.un_name is null, '[Не указан]', un.un_name) un_name,
             f_datetimetostr(wmt.wmat_datetime) wmat_datetime,
             0 as by_closed_ank
      from cc_website_miniapp_tmp wmt
      left join username un on wmt.wmat_from_un_id = un.un_id
      where wmt.wmat_from_wma_id = :wma_id
      and wmt.wmat_anketa_memo <> ''
      order by wmt.wmat_no desc
  ) A
  into :anketa_memo, :un_name, :wmat_datetime, :by_closed_ank
  do
  begin
    if (is_first_row = 0) then
        json_value = BLOB_APPEND(json_value, ', ');

    json_value = BLOB_APPEND(json_value, '{ ' || f_json_make_pair('memo', anketa_memo, 'str') || ', '
                            || f_json_make_pair('user', un_name, 'str') || ', '
                            || f_json_make_pair('saved_datetime', wmat_datetime, 'datetime') || ', '
                            || f_json_make_pair('by_closed_ank', by_closed_ank, 'str')
                            || '} ');
    if (is_first_row = 1) then
        is_first_row = 0;
  end 
  json_value = BLOB_APPEND(json_value, ' ],');

  if (wma_c_id is null) then
  begin
    json_value = BLOB_APPEND(json_value, ' "coincidence_with_persons": [ ');

    is_first_row = 1;
    for select c_id, res_comment, c_name_fio, f_datetostr(c_birth_date) c_birth_date
      from hp_find_clients(null, :fio, :birth_date, null, null, null, :phone_no, null, null)
      where is_critical = 1
      into :c_id, :res_comment, :c_name_fio, :c_birth_date
    do
    begin
        if (is_first_row = 0) then
            json_value = BLOB_APPEND(json_value, ', ');

        json_value = BLOB_APPEND(json_value, '{ ' || f_json_make_pair('c_id', :c_id, 'str') || ', '
                                || f_json_make_pair('comment',:res_comment, 'str') || ', '
                                || f_json_make_pair('fio', c_name_fio, 'str') || ', '
                                || f_json_make_pair('birth_date', c_birth_date, 'date') || ', '
                                || (select txt from cc_wma_info_part_tmp(:c_id, 0))
                                || '} ');
        if (is_first_row = 1) then
            is_first_row = 0;        
    end

    json_value = BLOB_APPEND(json_value, ' ]');
    json_value = BLOB_APPEND(json_value, ' ');
  end
  else
  begin
    json_value = BLOB_APPEND(json_value, ' "coincidence_with_persons": [ ], ');
    json_value = BLOB_APPEND(json_value, ' "client_data": {');

    c_id = wma_c_id;
    select cl.c_name_fio, cl.c_birth_date
    from clients cl
    where cl.c_id = :c_id
    into :c_name_fio, :c_birth_date;

    json_value = BLOB_APPEND(json_value, f_json_make_pair('fio', :c_name_fio, 'str') || ', '
                            || f_json_make_pair('birth_date', c_birth_date, 'date') || ', '
                            || (select txt from cc_wma_info_part_tmp(:c_id, 0))
                            || ' } ');

  end
  json_value = BLOB_APPEND(json_value, ' }');
  json_value = BLOB_APPEND(json_value, ' }');
  suspend;
end
