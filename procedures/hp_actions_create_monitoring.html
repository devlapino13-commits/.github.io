CREATE PROCEDURE HP_ACTIONS_CREATE_MONITORING
AS
declare variable KF_ID D_ID;
declare variable REG_ID D_INTEGER;
declare variable LO_ID D_ID;
declare variable K_ID D_ID;
declare variable K_DISB_DATE D_DATE;
declare variable REP_DATE D_DATE;
declare variable RD_ANKETA D_INTEGER;
declare variable PR_CODE D_STR_100;
declare variable IS_IDJARA D_BOOL;
declare variable DAYS_TO_WAIT D_INTEGER;
declare variable DATEX D_DATE;
declare variable OBJ_KIND1 D_INTEGER;
declare variable OBJ_KIND2 D_INTEGER;
declare variable UN_ID D_ID;
declare variable MAI_PROVISION_SERVIC_DATE D_DATE;
begin
  -- процедура должна запускаться каждый вечер и создавать задачи на заполнение анкеты по @мониторингу целевого использования кредита"
  -- Процедура должна запускаться после 21:30
  rd_anketa = f_get_rd_id('Anketa.TargetUseLoan');
  if (rd_anketa is null) then
    exit;

  rep_date = current_date;
  dateX = '03.03.2022';
  obj_kind1 = f_objkind('Clients', 'id');
  obj_kind2 = f_objkind('KnClient', 'id');
  un_id = 1000000000001;

  -- RFCKG-4219
  for select
    k_id,
    k_disb_date,
    k_from_kf_id,
    pr_code,
    iif(pr_code containing '.IDJARA.', 1, 0)
  from kontstate
  inner join knclient on ks_from_k_id = kc_from_k_id
    and kc_amount_approved > 0
  inner join kontract on k_id = ks_from_k_id
    and k_amount_approved >= 100000
    and k_disb_kind <> 6 -- RFCKG-2548
  inner join product on pr_id = k_from_pr_id
    and ((pr_code containing '.ipf.' and f_get_rd_code(kc_loan_target) in ('Loan_Target.BusinessGrowth', 'Loan_Target.BusinessCreation'))
      or pr_code not containing '.ipf.')
    and pr_code not containing '.express_online.'
    and pr_code not containing '.express_similar.'
  where ks_date_begin = :rep_date
    and ks_ol_is_delete = 0
    and ks_state = 1
  into
    k_id,
    k_disb_date,
    kf_id,
    pr_code,
    is_idjara
  do
  begin
    --RFCKG-3098
    -- Расчет количества дней до формирования задачи по ЦМ
    if (:pr_code containing '.ipf.') then
      days_to_wait = 3; -- Для ИПФ задача через 3 дня
    else
      days_to_wait = 10; -- Для Классики задача через 10 дней

    select
      kal_new_reg_id,
      kal_new_kf_officer
    from knaktlist
    where kal_from_kf_id = :kf_id
      and :rep_date between kal_date and kal_date_end
      and is_delete = 0
    into
      :reg_id,
      :lo_id;

       if (:is_idjara = 1) then
       begin
        select mi.mai_provision_servic_date
        from miniapp_ipf mi
        inner join kontract k on k.k_from_ma_app_id= mi.mai_from_ma_id
        where k.k_id = :k_id
        into mai_provision_servic_date;
        if (mai_provision_servic_date is null) then
          :mai_provision_servic_date = :rep_date;
        else
          :rep_date = :mai_provision_servic_date;
       end

    -- Создаем задачу завтрашним днем
    insert into actionlist (
      acl_date,
      acl_from_reg_id,
      acl_object_kind,
      acl_object_id,
      acl_from_rd_id,
      acl_status,
      acl_respons_c_id,
      acl_from_un_id)
    select
      :rep_date+:days_to_wait, -- текущая дата + количество дней до создания задачи
      :reg_id,
      iif(:k_disb_date >= :dateX, :obj_kind2, :obj_kind1),
      iif(:k_disb_date >= :dateX, kc_id, kc_from_c_id),
      :rd_anketa,
      0,
      :lo_id,
      :un_id
    from knclient
    where kc_from_k_id = :k_id
      and kc_amount_approved > 0;
    rep_date = current_date;
  end
end
