CREATE PROCEDURE HP_FINCERT_PHONE_CHECK(
    PHONE_NUMBER D_STR_20,
    CHECK_DATE_TIME D_DATETIME)
RETURNS (
    RES_DATE D_DATE,
    RES_STATUS D_SMALLINT,
    FF_CREATE_DATETIME D_DATETIME,
    RES_CODE D_STR_CODE)
AS
declare variable ff_file_body d_str_255;
declare variable res_datetime d_datetime;
begin
  -- RFCRU-3509 Получаем дату
  res_status = 0;
  ff_file_body = '';

  if(coalesce(phone_number, '') = '') then
    exit;

  for with t as(
    select ff_file_body, ff_create_datetime, 'fastpay_number' res_code
    from hp_fincert_get_file('fastpay_number', :check_date_time, 1) t
    union all
    select ff_file_body, ff_create_datetime, 'phone_number'
    from hp_fincert_get_file('phone_number', :check_date_time, 1) t
  )
  select ff_create_datetime, res_code,
    case
        when position(:phone_number in t.ff_file_body) > 0 then
          -- найдена позиция номера
          case
            when position(',' in substring(t.ff_file_body from position(:phone_number in t.ff_file_body))) > 0 then
              -- есть запятая после номера — берем до неё
              substring(
                t.ff_file_body from position(:phone_number in t.ff_file_body)
                for position(',' in substring(t.ff_file_body from position(:phone_number in t.ff_file_body))) - 1
              )
            else
              -- запятая не найдена — берем до конца
              substring(t.ff_file_body from position(:phone_number in t.ff_file_body))
          end
        else
          ''
     end
  from t
  into ff_create_datetime, res_code, ff_file_body
  do
  begin
    if (ff_file_body <> '') then
    begin
      res_status = 1;
  
      res_datetime = cast(substring(ff_file_body from
                                    position('|', ff_file_body) + 1 for
                                    position('|', substring(ff_file_body from position('|', ff_file_body) + 1)) - 1) as d_datetime);
  
      res_date = cast(res_datetime as d_date);
      break;
    end
  end

  suspend;
end
