CREATE PROCEDURE AAREP_FOLDER_TRANSFER_FOR_DISB(
    REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    NUM D_INTEGER,
    DISB_OFFICE_NAME D_STR_64,
    K_FACT_DISB_KF_NO D_STR_100,
    DISB_OFFICER_NAME D_STR_100,
    K_FACT_DISB_DATE D_DATE,
    KAL_DATE D_DATE,
    MA_OFFICE_NAME D_STR_64,
    KAL_NEW_KF_NAME D_STR_100,
    MA_OFFICER_NAME D_STR_100)
AS
declare variable reg_id d_integer;
    declare variable new_reg_id d_integer;
    declare variable prior_reg_id d_integer;
    declare variable kf_id d_id;
    declare variable ma_id d_integer;
    declare variable k_fact_disb_lo_id id_type;
    declare variable kal_new_kf_officer id_type;
begin
  /*
  создана в рамках RFCKG-1011
  параметры: регистрировавший заявку офис или офисы; отчетный период.
  результат: а) офис, выдавший займ: название, № папки, КС, дата выдачи;
             б) офис, регистрировавший заявку: дата возврата, название, № папки, КС.
*/

  num = 0;

  for select id_list
      from region_list(:region_id, 1, 3, 1)
      into reg_id -- беру все офисы по переданному региону/стране
  do
  begin
    -- получаю передачи папки из одного офиса в другой за отчетный период
    for select kal.kal_from_kf_id,
               kal.kal_new_reg_id,
               kal.kal_prior_reg_id,
               kal.kal_date,
               kal.kal_new_kf_officer,
               kal.kal_new_kf_name
        from knaktlist kal
        where kal.kal_new_reg_id = :reg_id and -- текущий офис = офис, регистрировавший заявку
              kal.kal_date between :rep_date_begin and :rep_date_end and
              kal.is_delete = 0 and
              kal.kal_new_reg_id <> kal.kal_prior_reg_id -- была передача из офиса в офис
        into kf_id, -- id папки
             new_reg_id, -- последний офис за период
             prior_reg_id, -- предыдущий офис за период
             kal_date, -- дата создания папки
             kal_new_kf_officer, -- КС за которым папка
             kal_new_kf_name -- номер папки
    do
    begin
      -- поднимаю контракты в рамках передачи где [офис выдачи = офису откуда ушла парка] и [выдача займа была в рамках отчетного периода]
      for select k.k_from_ma_app_id,
                 k.k_fact_disb_date,
                 k_fact_disb_lo_id,
                 k.k_fact_disb_kf_no
          from kontract k
          where k.k_from_kf_id = :kf_id and
                k.k_fact_disb_reg_id = :prior_reg_id and
                k.k_disb_date between :rep_date_begin and :rep_date_end
          into ma_id,
               k_fact_disb_date,
               k_fact_disb_lo_id,
               k_fact_disb_kf_no
      do
      begin
        if (exists( -- если офис где оформлялась заявка = офису куда была передача после выдачи - это контракт, который требовалось отловить
        select ma_id
        from miniapp ma
        where ma.ma_id = :ma_id and
              ma.ma_is_active = 1 and
              ma.ma_date_end = '31.12.9999' and
              ma.ma_office_id = :new_reg_id)) then
        begin
          ma_office_name = (select reg.reg_name
                            from region reg
                            where reg.reg_unicode = :new_reg_id);

          disb_office_name = (select reg.reg_name
                              from region reg
                              where reg.reg_unicode = :prior_reg_id);

          disb_officer_name = (select c_name_fio
                               from clients
                               where c_id = :k_fact_disb_lo_id);

          ma_officer_name = (select c_name_fio
                             from clients
                             where c_id = :kal_new_kf_officer);

          num = num + 1;

          suspend;
        end
      end
    end
  end
end
