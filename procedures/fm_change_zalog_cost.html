CREATE PROCEDURE FM_CHANGE_ZALOG_COST(
    M_KIND D_STR_CODE,
    UN_ID D_ID,
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID,
    COST D_CURRENCY)
RETURNS (
    RES_ID D_INTEGER,
    RES_TXT D_STR_255)
AS
begin
    --RFCRU-2906
    res_id = 0;
    res_txt = '';

    if(obj_kind <> 'zalog') then res_txt = 'Произошла ошибка: Некорректный obj_kind';

    if( cost < 0) then res_txt = 'Цена должна быть больше 0';

    if(cast(cost as int) <> cost) then res_txt = 'Цена должна быть целым числом';

    if (m_kind = 'set_cost_market' and res_txt = '' and obj_id is not null) then
    begin
        update zalog z
        set z.z_cost_market = :cost
        where z.z_id = :obj_id;
        res_id = 1;
    end

    if (m_kind = 'set_cost_kontract' and res_txt = '' and obj_id is not null) then
    begin
        update zalog z
        set z.z_cost_kontract = :cost
        where z.z_id = :obj_id;
        res_id = 1;
    end

    suspend;
end
