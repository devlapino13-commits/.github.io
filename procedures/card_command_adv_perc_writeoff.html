CREATE PROCEDURE CARD_COMMAND_ADV_PERC_WRITEOFF(
    OBJECT_KIND D_STR_100 NOT NULL,
    OBJECT_ID D_ID NOT NULL,
    OL_ID D_ID = null)
AS
declare variable c_id d_id;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable cca_code d_str_code;
declare variable cs_request_json d_str_8100;
declare variable comment_txt varchar(140);
declare variable cs_id d_integer;
declare variable cst_packet_no varchar(8);
declare variable cst_ticket_no varchar(7);
declare variable trans_datetime d_datetime;
declare variable object_kind_str d_str_100;
declare variable k_id d_id;
declare variable ol_parent_id d_id;
declare variable ol_from_reg_id d_integer;
declare variable un_id d_id;
declare variable adv_kommis d_currency;
declare variable ol_id_disb d_id;
declare variable cs_parent_id d_integer;
begin
    if (f_get_const('cardprjclose') = 1) then
      exit;

    object_kind_str = f_objkind(object_kind, 'name');

    if (object_kind_str = 'kontract') then
        k_id = object_id;
    else
        exception e_message 'object_kind = '||object_kind_str||' не реализован';

    if (ol_id is null) then exception e_message 'ol_id не определен';

    select ol.ol_parent_id,
           ol.ol_from_reg_id,
           ol.ol_user_id
    from operlist ol
    where ol.ol_id = :ol_id
    into :ol_parent_id,
         :ol_from_reg_id,
         :un_id;

    if (ol_parent_id is not null) then
    begin
      select ol.ol_from_reg_id
      from operlist ol
      where ol.ol_id = :ol_parent_id
      into :ol_from_reg_id;
    end

    ------- получаем cs_id родительсткой запсиии. Т.к. погашение до % идет сразу после выдачи займа, => что cs_id родителя - это cs_id операции присования кред. лимита по факту выдачи займа
    select kont.k_fact_disb_ol_id
    from kontract kont
    where kont.k_id = :k_id
    into :ol_id_disb;

    if (ol_id_disb is null) then
        exception e_message 'ol_id_disb is null';

    select cso.cso_from_cs_id
    from cardsyncopers cso
    inner join cardsync cs on cs.cs_id = cso.cso_from_cs_id
    where cso.cso_from_ol_id = :ol_id_disb
    and cs.cs_from_csc_id = f_get_cardsynccfg('card_disb', 'id')
    into :cs_parent_id;

    if (cs_parent_id is null) then
        exception e_message 'cs_parent_id is null';
    ---------------------- <<

    adv_kommis = 0;

    select -sum(ok.ok_kommis)
    from operkred ok
    where ok.ok_from_k_id = :k_id
    and ok.ok_ol_is_active = 1
    and ok.ok_type = 4
    and ok.ok_from_ol_id = :ol_id
    into :adv_kommis;
    
    if (adv_kommis < 0) then
        exception e_message 'По комисии не может быть переплаты';

    comment_txt = f_format('Комисиия: %1', :adv_kommis);

    select kncl.kc_from_c_id
    from knclient kncl
    where kncl.kc_from_k_id = :k_id
    into :c_id;
    
    select
        card_id,
        cca_id,
        cca_code
    from hp_client_bank_card_mb(:c_id, current_date)
    into
        :card_id,
        :cca_id,
        :cca_code;

    select packet_no,
           ticket_no
    from cardsynctrn_create
    into :cst_packet_no,
         :cst_ticket_no;
    
    if (cst_packet_no is null) then
        exception e_message '№ пакета не опредлен';
    
    if (cst_ticket_no is null) then
        exception e_message '№ тикета не опредлен';
    
    select ol.ol_datetime
    from operlist ol
    where ol.ol_id = :ol_id
    into :trans_datetime;

    -- payment_mode = 1 - моментальное выполнение без учета ОТБ (баланса), т.е может уходить в минус

    cs_request_json = f_format('
    {
        "card_acct": "%1",
        "card_id": '||:card_id||',
        "card_acct_ccy": "KGS",
        "tran_type": "120",
        "tran_ccy": "KGS",
        "tran_amount": %2,
        "tran_pocket": "%3",
        "tran_ticket": "%4",
        "tran_comment": "%5",
        "bank_c": "21",
        "groupc": "01",
        "payment_mode": 2,
        "tran_date_time": "' || f_datetimetostr(:trans_datetime) || '"
    }',
    :cca_code,
    round(:adv_kommis, 2),
    :cst_packet_no,
    :cst_ticket_no,
    :comment_txt
    );

    insert into cardsync(
      cs_parent_id,
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_amount,
      cs_request_json,
      cs_office_id,
      cs_from_un_id,
      cs_from_k_id,
      cs_from_c_id
    )
    values(
      :cs_parent_id,
      f_get_cardsynccfg('adv_kommis_writeoff', 'id'),
      :card_id,
      :cca_id,
      f_objkind('kontract', 'id'),
      :k_id,
      :adv_kommis,
      :cs_request_json,
      :ol_from_reg_id,
      :un_id,
      :k_id, 
      :c_id
    )
    returning cs_id
    into :cs_id;
    
    update cardsynctrn cst
    set cst.cst_from_cs_id = :cs_id
    where cst.cst_packet_no = :cst_packet_no and
          cst.cst_ticket_no = :cst_ticket_no;

    insert into cardsyncopers (cso_from_cs_id, cso_from_ol_id)
    values (:cs_id, :ol_id);
end
