CREATE PROCEDURE AAREP_REPORT_BANK_KART(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_REGION_ID D_INTEGER)
RETURNS (
    LINE D_INTEGER,
    CREDDATE D_DATE,
    CREDTIME TIME,
    KAL_NEW_KF_NAME VARCHAR(100),
    K_KONTRACT_NO D_STR_CODE,
    C_NAME_FIO D_STR_100,
    CREDSUM D_CURRENCY,
    PR_NAME VARCHAR(50),
    REG_NAME D_STR_100,
    SOTR_FIO D_STR_100,
    ET_CARDNUMBER STR_100 COLLATE UNICODE_CI,
    JOB_NAME STR_100 COLLATE UNICODE_CI,
    PAY_PROJECT D_STR_CODE,
    FPS_APPROVE_KIND D_STR_CODE,
    FPS_APPROVE_FIO D_STR_100,
    REASON D_STR_2048,
    REASON_DATETIME D_DATETIME,
    CITIZEN STR_255,
    C_NAME_FIO_SBP D_STR_100,
    MA_DATE_SAVE D_DATETIME,
    EDS_STATUS_DATETIME D_DATETIME)
AS
declare variable db_id d_integer;
declare variable k_disb_kind d_integer;
declare variable kc_id d_id;
declare variable k_id d_id;
declare variable rd_id d_integer;
declare variable kc_from_c_id id_type;
begin
  line = 0;
  db_id = f_get_db_id();
  for select distinct kc_id, k.k_disb_date, cast(ol.ol_datetime as time) credtime, kal.kal_new_kf_name, k.k_kontract_no,
                      et.et_cardnumber, c.c_name_fio, kc.kc_amount_approved credsum, pr.pr_name,
                      reg.reg_code || ' ' || reg_name as reg_name, un.un_fio sotr_fio, jl.jl_name, k.k_disb_kind,
                      k.k_id, rd_name, kc.kc_from_c_id, ma_date_save
      from kontstate ks
      inner join kontract k on k.k_id = ks.ks_from_k_id --and
            -- k.k_kontract_no = '1801-207294-14'
      inner join knclient kc on k.k_id = kc.kc_from_k_id
      inner join clients c on kc.kc_from_c_id = c.c_id
      inner join refdata on rd_id = c.c_citizen
      inner join miniapp ma on ma.ma_id = k.k_from_ma_app_id
      left join extsys_token et on et.et_from_c_id = c.c_id and
            et.et_is_actual = 1
      inner join knaktlist kal on kal.kal_from_kf_id = ma.ma_from_kf_id
      inner join product pr on pr.pr_id = k.k_from_pr_id
      inner join region reg on reg.reg_unicode = ma.ma_office_id
      left join operlist ol on ol.ol_id = ks.ks_from_ol_id
      left join username un on ol.ol_user_id = un.un_id
      left join hr_pstatus ps on ps.ps_from_c_id = un.un_from_c_id and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_status = 1 and
            -- только утвержденные приказы
            ps_is_working = 1 -- только работающие сотрудники
      left join joblist jl on ps.ps_from_jl_id = jl.jl_id

      where ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            :rep_date_end between kal_date and kal_date_end

            and
            kc.kc_amount_approved > 0 and
            ma_date_end = '31.12.9999' and
            ks.ks_state in (0, 1) and
            k.k_is_online = 1 -- только онлайн выдачи
            and
            k.k_disb_kind in (5, 6, 7) and
            ks.ks_ol_is_delete = 0 and
            (reg_unicode = :rep_region_id or
            :rep_region_id =1000000)
      order by k.k_disb_date
      into kc_id, creddate, credtime, kal_new_kf_name, k_kontract_no, et_cardnumber, c_name_fio, credsum, pr_name,
           reg_name, sotr_fio, job_name, k_disb_kind, k_id, citizen, kc_from_c_id, ma_date_save
  do
  begin
    select eds_status_datetime
    from hp_edoc_list('kontract', :k_id, null)
    rows 1
    into eds_status_datetime;

    fps_approve_kind = '';
    pay_project = '';
    fps_approve_fio = '';
    fps_approve_kind = '';
    if (db_id = 41701) then
      pay_project = 'elcard';
    else
    if (db_id = 81001) then
      pay_project = 'Карта';

    if (db_id = 81001 and
        k_disb_kind = 7) then
    begin
      -- Доп. инфо по выдачам через СБП
      pay_project = 'СБП';
      fps_approve_kind = 'Авто';
      fps_approve_fio = '';

      select fcn_approve_un_id, fps_client_names.fcn_name
      from fps_loan_banks
      inner join fps_client_banks on fbl_from_fcb_id = fcb_id
      inner join fps_banks on fcb_from_fpb_id = fpb_id
      inner join fps_client_names on fcn_from_fcb_id = fcb_id
      left join username on un_id = fcn_approve_un_id
      left join clients on c_id = un_from_c_id
      where flb_from_kc_id = :kc_id and
            fcn_status = 2
      order by fcn_datetime desc
      rows 1
      into fps_approve_fio, c_name_fio_sbp;

      if (fps_approve_fio <> '') then
        fps_approve_kind = 'КК';

      if (pay_project = 'СБП') then
        :et_cardnumber = '';
    end

    if (credtime is null) then
    begin
      :c_name_fio_sbp = '';
      select fps_client_names.fcn_name
      from fps_client_names
      inner join fps_client_banks on fcn_from_fcb_id = fcb_id
      left join fps_loan_banks on fbl_from_fcb_id = fcb_id
      where fcb_from_c_id = :kc_from_c_id and
            fcn_status = 0
      order by fcn_datetime asc
      rows 1
      into :c_name_fio_sbp;

    end

    if (db_id = 81001) then
    begin

      reason = '';
      reason_datetime = null;
      select rd_id
      from hp_get_check_loan_data(:k_id)
      into :rd_id;

      select obm_memo, obm_datetime
      from objectprops
      inner join objectvalues on ovs_from_opr_id = opr_id
      inner join objectmemo on obm_from_opr_id = opr_id
      where opr_object_kind = 9 and
            opr_object_id = :k_id and
            opr_subject_kind = -1 and
            opr_subject_id = :rd_id and
            ovs_is_active = 1 and
            ovs_value = -1 and
            f_get_verify_memo(obm_memo, 'code') in (select vcm_code
                                                    from verifycomment
                                                    where vcm_is_payout = 1)
      order by obm_id desc
      rows 1
      into reason, reason_datetime;

      if ((reason = '') and
          (:c_name_fio_sbp = '')) then
        reason = 'тех. ошибка';
    end

    line = line + 1;
    suspend;
  end
end
