CREATE PROCEDURE AAREP_PAR_KM_LIDER(
    REP_REGION_ID ID_TYPE,
    REP_DATE DATE,
    REP_STEP SMALLINT)
RETURNS (
    LDAY_PMONTH DATE,
    LDAY_CMONTH DATE,
    MEASURE STR_100,
    OFFICE_ID ID_TYPE,
    C_ID ID_TYPE,
    PREV_BALANCE CURR_TYPE,
    PREV_BALANCE_8_180 CURR_TYPE,
    PREV_MONTH_PAR_8_180 CURR_TYPE,
    CURRENT_BALANCE_8_180 CURR_TYPE,
    RISE_BALANCE_PAR_8_180 DECIMAL(18,1),
    MEASURE_STEP SMALLINT,
    PREV_KS_QTY INTEGER,
    CURRENT_KS_QTY INTEGER,
    PREV_CLIENTS_QTY INTEGER,
    CURRENT_CLIENTS_QTY INTEGER,
    PREV_AVERAGE_CLIENTS CURR_TYPE,
    CURRENT_AVERAGE_CLIENTS CURR_TYPE,
    RISE_AVERAGE_CLIENTS DECIMAL(18,1),
    MAX_QTY INTEGER,
    CURRENT_BALANCE CURR_TYPE,
    RISE_BALANCE DECIMAL(18,1),
    MAX_VAL DECIMAL(18,1),
    RISE_BALANCE_PAR_8_180_MAX DECIMAL(18,1),
    RISE_AVERAGE_CLIENTS_MAX DECIMAL(18,1),
    OFFICE_NAME STR_255,
    KM_FIO STR_100,
    RESULT_VAL DECIMAL(18,1))
AS
declare variable qty smallint;
declare variable is_culc smallint;
declare variable rise_average_clients_tmp decimal(18,1);
declare variable rise_balance_max decimal(18,1);
declare variable max_qty2 smallint;
declare variable km_qty integer;
declare variable is_ok smallint;
declare variable c_id_tmp id_type;
declare variable end_otpusk_date date;
declare variable rise_balance_par_8_180_tmp decimal(18,1);
declare variable ps_date_begin date;
declare variable max_qty_tmp integer;
begin
-- Первоначальная выборка данных и расчет показателя "1. Наибольший рост баланса PAR8-180"
  if (rep_step = 1) then
  begin
    measure = '1. Наибольший рост баланса PAR8-180';
    rise_balance_par_8_180_max = 0;
    max_val = 0;

    lday_pmonth = (select cast('01.' || cast(extract(month from cast(:rep_date as date)) as varchar(2)) || '.' || cast(extract(year from cast(:rep_date as date)) as varchar(4)) as date) - 1
                   from rdb$database); -- последняя дата предыдущего месяца

    lday_cmonth = /*(select cast('01.'||cast(extract(month from dateadd(month, 1, cast(:rep_date as date))) as varchar(2))||'.'||
                        cast(extract(year from cast(:rep_date as date)) as varchar(4)) as date) - 1
                     from rdb$database);  -- последняя дата текущего месяца    */
    rep_date;

    for select oks_region_id, sum(oks_balance)
        from oksummary oks, region_list(:rep_region_id, 1, -1, 1)
        where oks_close_state = 0 and
              :lday_pmonth between oks_date_begin and oks_date_end and
              (cast(:lday_pmonth as date) - oks_delinq_date) between 8 and 180 and
              oks_region_id = id_list
        group by oks_region_id
        order by 2 desc
        into :office_id, prev_balance_8_180 -- просроченный баланс офиса (8-180 дн. просрочки), за предыдущ месяц
    do
    begin
      select sum(oks_balance)
      from oksummary
      where oks_region_id = :office_id and
            :lday_pmonth between oks_date_begin and oks_date_end
      into :prev_balance; --Потрфель за предыдущий месяц

      prev_month_par_8_180 = prev_balance_8_180 / prev_balance * 100; --Процент ПАР8_180 на конец предыдущ месяца

      if (prev_month_par_8_180 > 1.5) then --Если ПАР8_180 (округляя до 1 символа) больше 1,5% то узнаем менеджера офиса
      begin
        km_qty = 0;
        is_ok = 0;
-- проверяю, сколько КМ со стажем стаже работы более 3 месяцев, в одном офисе
        for select hps.ps_from_c_id, ps_date_begin
            from hr_pstatus hps
            inner join joblist jl on jl.jl_id = ps_from_jl_id
            inner join jobkind jk on jk.jk_id = jl.jl_kind
            where :lday_cmonth between ps_date_begin and ps_date_end and
                  ps_is_working = 1 and
                  jk.jk_system_code = 'JobSystem.Region_KM' and
                  hps.ps_status = 1 and
                  ps_from_reg_id = :office_id and
                  ps_date_end_in_order > :lday_cmonth and
                  datediff(month, ps_date_begin, :lday_cmonth) >= 3
            into :c_id, ps_date_begin
        do
        begin
          ps_date_begin = dateadd(month, datediff(month, ps_date_begin, :lday_cmonth), ps_date_begin);

          if (datediff(day, ps_date_begin, lday_cmonth) >= 1) then
          begin
            km_qty = :km_qty + 1;
          end
        end

        if (km_qty = 1) then -- если только 1 КМ, в подотчетном офисе, со стажем более 3 месяцев
        begin
          is_ok = 1;
        end
        else
        if (km_qty > 1) then -- если несколько КМ, в подотчетном офисе, со стажем более 3 месяцев, проверяю кто из КМ га даннй момент в отпуске
        begin
          is_ok = 0;

          for select hps.ps_from_c_id
              from hr_pstatus hps
              inner join joblist jl on jl.jl_id = ps_from_jl_id
              inner join jobkind jk on jk.jk_id = jl.jl_kind
              where :lday_cmonth between ps_date_begin and ps_date_end and
                    ps_is_working = 1 and
                    jk.jk_system_code = 'JobSystem.Region_KM' and
                    hps.ps_status = 1 and
                    ps_from_reg_id = :office_id and
                    ps_date_end_in_order > :lday_cmonth and
                    datediff(month, ps_date_begin, :lday_cmonth) > 3
              into :c_id_tmp
          do
          begin
            if (is_ok = 0) then
            begin
              c_id = c_id_tmp;

-- проверяю, находится ли КМ сейчас в отпуске
              end_otpusk_date = null;
              select hrv.hrv_date_end_fact
              from hr_orders hro, hr_voctravel hrv, sa_worktype saw
              where ho_order_type in (7, 8, 9, 10, 16, 17, 18, 20) --Все виды отпусков
                    and
                    hro.ho_id = hrv.hrv_from_ho_id and
                    hrv.hrv_from_c_id = :c_id and
                    hro.ho_order_type = saw.sawt_id and
                    :lday_cmonth between hrv.hrv_date_begin and hrv.hrv_date_end_fact and
                    hro.ho_status = 1 and
                    hrv.hrv_status = 1
              order by hro.ho_date desc
              into :end_otpusk_date;

              if (end_otpusk_date is not null) then -- Если Да, он в отпуске, проверяю нет ли отзыва из отпуска
              begin
                if (exists(select *
                           from hr_orders hro, hr_voctravel hrv, sa_worktype saw
                           where ho_order_type in (15, 19) --Отзывы из отпуска
                                 and
                                 hro.ho_id = hrv.hrv_from_ho_id and
                                 hrv.hrv_from_c_id = :c_id and
                                 hro.ho_order_type = saw.sawt_id and
                                 hrv.hrv_date_begin <= :lday_cmonth and
                                 :lday_cmonth <= hrv.hrv_date_end_fact and
                                 hro.ho_status = 1 and
                                 hrv.hrv_status = 1)) then
                begin -- Если на дату приготовления отчета сотрдуник не был в отпуске (был оозван)
                  is_ok = 1;
                end
              end
              else
              begin
                is_ok = 1;
              end
            end
          end
        end

        if (is_ok = 1) then
        begin
          current_balance_8_180 = 0;

          for select sum(oks_balance)
              from oksummary
              where oks_close_state = 0 and
                    oks_region_id = :office_id and
                    :lday_cmonth between oks_date_begin and oks_date_end and
                    (cast(:lday_cmonth as date) - oks_delinq_date) between 8 and 180
              into current_balance_8_180 --Текущий ПАР8_180
          do
          begin
            rise_balance_par_8_180 = round(((current_balance_8_180 - prev_balance_8_180) / prev_balance) * 100, 1);

            if (rise_balance_par_8_180 > rise_balance_par_8_180_max) then
            begin
              rise_balance_par_8_180_max = rise_balance_par_8_180;
              max_qty = 1;
              max_val = rise_balance_par_8_180_max;
            end
            else
            if (rise_balance_par_8_180 = rise_balance_par_8_180_max) then
              max_qty = max_qty + 1;

            suspend;
          end
        end
      end
    end
  end
------------------------
--- Опредление числа записей из пункта 1 и расчет показателя "2. Наибольшее снижение продуктивности КС"
  if (rep_step = 2) then
  begin
    rise_balance_par_8_180_tmp = -9999;
    qty = 0;
    rise_average_clients_max = -9999;
    max_val = -9999;
    max_qty2 = 0;

    for select lday_pmonth, lday_cmonth, measure, office_id, c_id, prev_balance, prev_balance_8_180,
               prev_month_par_8_180, current_balance_8_180, rise_balance_par_8_180, 1, max_qty, max_val
        from aarep_par_km_lider(:rep_region_id, :rep_date, 1)
        order by rise_balance_par_8_180 desc, max_val desc, max_qty desc
        into lday_pmonth, lday_cmonth, measure, office_id, c_id, prev_balance, prev_balance_8_180, prev_month_par_8_180,
             current_balance_8_180, rise_balance_par_8_180, measure_step, max_qty, max_val
    do
    begin
      max_val = 0;

      is_culc = 0;

      if ((rise_balance_par_8_180_tmp < rise_balance_par_8_180) and
          (qty = 0)) then
      begin
        rise_balance_par_8_180_tmp = rise_balance_par_8_180;
        qty = qty + 1;
        if (max_qty > 1) then
          is_culc = 1;

        max_qty = 0;

        suspend;
      end
      else
      if (rise_balance_par_8_180_tmp = rise_balance_par_8_180) then
      begin
        qty = qty + 1;
        is_culc = 1;
        max_qty = 0;
        suspend;
      end
      else
      if ((rise_balance_par_8_180_tmp > rise_balance_par_8_180) and
          (qty < 5)) then
      begin
        max_qty = 0;
        qty = qty + 1;
        suspend;
      end

      if (is_culc = 1) then
      begin

        prev_ks_qty = (select count(hps.ps_from_c_id)
                       from hr_pstatus hps
                       inner join joblist jl on jl.jl_id = ps_from_jl_id
                       inner join jobkind jk on jk.jk_id = jl.jl_kind
                       where :lday_pmonth between ps_date_begin and ps_date_end and
                             ps_is_working = 1 and
                             jk.jk_system_code = 'JobSystem.LO_Standart' and
                             hps.ps_status = 1 and
                             ps_from_reg_id = :office_id and
                             ps_date_end_in_order > :lday_pmonth);

        current_ks_qty = (select count(hps.ps_from_c_id)
                          from hr_pstatus hps
                          inner join joblist jl on jl.jl_id = ps_from_jl_id
                          inner join jobkind jk on jk.jk_id = jl.jl_kind
                          where :lday_cmonth between ps_date_begin and ps_date_end and
                                ps_is_working = 1 and
                                jk.jk_system_code = 'JobSystem.LO_Standart' and
                                hps.ps_status = 1 and
                                ps_from_reg_id = :office_id and
                                ps_date_end_in_order > :lday_cmonth);

        prev_clients_qty = (select count(distinct kc_from_c_id)
                            from knclient, oksummary oks
                            where kc_from_k_id = oks_from_k_id and
                                  kc_amount_approved > 0 and
                                  oks_close_state = 0 and
                                  :lday_pmonth between oks_date_begin and oks_date_end and
                                  oks_region_id = :office_id);

        current_clients_qty = (select count(distinct kc_from_c_id)
                               from knclient, oksummary oks
                               where kc_from_k_id = oks_from_k_id and
                                     kc_amount_approved > 0 and
                                     oks_close_state = 0 and
                                     :lday_cmonth between oks_date_begin and oks_date_end and
                                     oks_region_id = :office_id);

        prev_average_clients = prev_clients_qty / prev_ks_qty;

        current_average_clients = current_clients_qty / current_ks_qty;

        rise_average_clients = round(((current_average_clients / prev_average_clients) - 1) * 100, 1);

        if (rise_average_clients <> 0) then
          rise_average_clients = rise_average_clients * -1; -- сделано cпециально потому что ищем наибольшее снижение

        if (rise_average_clients > rise_average_clients_max) then
        begin
          rise_average_clients_max = rise_average_clients;
          max_qty2 = 1;
          max_val = rise_average_clients_max;
        end
        else
        if (rise_average_clients = rise_average_clients_max) then
          max_qty2 = max_qty2 + 1;

        max_qty = max_qty2;

        measure_step = 2;
        measure = '2. Наибольшее снижение продуктивности КС';

        suspend;
      end
    end
  end
------------
--- Опредление числа записей из пункта 2 и расчет показателя "3. Наибольшее cнижение баланса КП офиса"
  if (rep_step = 3) then
  begin
    rise_average_clients_tmp = -9999;
    rise_balance_max = -9999;
    qty = 0;
    max_val = 0;
    max_qty2 = 0;
    for select lday_pmonth, lday_cmonth, measure, office_id, c_id, prev_balance, prev_balance_8_180,
               prev_month_par_8_180, current_balance_8_180, rise_balance_par_8_180, measure_step, prev_ks_qty,
               current_ks_qty, prev_clients_qty, current_clients_qty, prev_average_clients, current_average_clients,
               rise_average_clients, max_qty, max_val
        from aarep_par_km_lider(:rep_region_id, :rep_date, 2)
        order by measure_step desc, rise_average_clients desc, max_val desc, max_qty desc, rise_balance_par_8_180 desc
        into lday_pmonth, lday_cmonth, measure, office_id, c_id, prev_balance, prev_balance_8_180, prev_month_par_8_180,
             current_balance_8_180, rise_balance_par_8_180, measure_step, prev_ks_qty, current_ks_qty, prev_clients_qty,
             current_clients_qty, prev_average_clients, current_average_clients, rise_average_clients, max_qty,
             max_val
    do
    begin
      max_val = 0;
      max_qty_tmp = max_qty;
      max_qty = 0;

      suspend;
      is_culc = 0;

      if (measure_step = 2) then
      begin
        if ((rise_average_clients_tmp < rise_average_clients) and
            (qty = 0)) then
        begin
          rise_average_clients_tmp = rise_average_clients;
          qty = qty + 1;
          if (max_qty_tmp > 1) then
            is_culc = 1;
        end
        else
        if (rise_average_clients_tmp = rise_average_clients) then
        begin
          qty = qty + 1;
          is_culc = 1;
        end
        else
        if ((rise_average_clients_tmp > rise_average_clients)) then
        begin
          qty = qty + 1;
        end

        if (is_culc = 1) then
        begin
          select sum(oks_balance)
          from oksummary
          where oks_region_id = :office_id and
                :lday_cmonth between oks_date_begin and oks_date_end
          into :current_balance; --Потрфель за отчетный месяц

          rise_balance = round(((prev_balance / current_balance) - 1) * 100, 1);
          if (rise_balance <> 0) then
            rise_balance = rise_balance * -1; -- сделано cпециально потому что ищем наибольшее снижение

          if (rise_balance > rise_balance_max) then
          begin
            rise_balance_max = rise_balance;
            max_qty2 = 1;
            max_val = rise_balance_max;
          end
          else
          if (rise_balance = rise_balance_max) then
            max_qty2 = max_qty2 + 1;
          max_qty = max_qty2;

          measure_step = 3;
          measure = '3. Наибольшее cнижение баланса КП офиса';

          suspend;
        end
      end
    end
  end
-------------------------
-- Итоговая выборка данных для отчета
  if (rep_step = 4) then
  begin
    for select lday_pmonth, lday_cmonth, measure, office_id, c_id, prev_balance, prev_balance_8_180,
               current_balance_8_180, rise_balance_par_8_180, prev_clients_qty, current_average_clients,
               rise_average_clients, current_balance, rise_balance, measure_step
        from aarep_par_km_lider(:rep_region_id, :rep_date, 3)
        order by measure_step desc, rise_average_clients desc, max_val desc, max_qty desc, rise_average_clients desc, rise_balance_par_8_180 desc
        into lday_pmonth, lday_cmonth, measure, office_id, c_id, prev_balance, prev_balance_8_180,
             current_balance_8_180, rise_balance_par_8_180, prev_average_clients, current_average_clients,
             rise_average_clients, current_balance, rise_balance, measure_step
    do
    begin
      select reg.reg_name
      from region reg
      where reg.reg_unicode = :office_id
      into :office_name;

      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into :km_fio;

      if (measure_step = 1) then
        result_val = rise_balance_par_8_180;
      else
      if (measure_step = 2) then
        result_val = rise_average_clients;
      else
      if (measure_step = 3) then
        result_val = rise_balance;

      prev_balance = coalesce(prev_balance, 0);
      prev_balance_8_180 = coalesce(prev_balance_8_180, 0);
      current_balance_8_180 = coalesce(current_balance_8_180, 0);
      prev_average_clients = coalesce(prev_average_clients, 0);
      current_average_clients = coalesce(current_average_clients, 0);
      rise_average_clients = coalesce(rise_average_clients, 0);
      current_balance = coalesce(rise_average_clients, 0);

      suspend;
    end
  end
end
