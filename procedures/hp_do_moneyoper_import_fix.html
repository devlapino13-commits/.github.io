CREATE PROCEDURE HP_DO_MONEYOPER_IMPORT_FIX(
    OL_ID D_ID,
    CURRECT_AMOUNT D_CURRENCY,
    WRONG_AMOUNT D_CURRENCY)
AS
declare variable first_line d_integer;
declare variable kob_id d_id;
declare variable kob_summa d_currency;
declare variable stay_kob_id d_id;
declare variable total_amount d_currency;
declare variable ol_date d_date;
declare variable k_id d_id;
declare variable kc_id d_id;
declare variable acc_dt d_id;
declare variable acc_kt d_id;
declare variable pcb_summa d_currency;
declare variable row_kol d_integer;
declare variable fixed d_bool;
begin
  select kob_id, kob_from_kc_id, kob_ol_date
  from kn_operbook
  where kob_from_ol_id = :ol_id and
        kob_from_kc_id is not null
  rows 1
  into kob_id, kc_id, ol_date;

  if (kc_id is not null) then
  begin
    if (not exists(
    select * from kn_operbook
    where kob_from_ol_id = :ol_id and
          (kob_from_kc_id is null or kob_from_kc_id <> :kc_id))) then
    begin
      select kc_from_k_id
      from knclient
      where kc_id = :kc_id
      into k_id;

      execute procedure hp_do_kn_operbook_one_online(:k_id, current_date, null, :ol_date - 1);
      Exit;
    end
  end


  -- Процедура удаляет дубликаты проводок после импорта из 1С

  first_line = 1;
  total_amount = 0;

  for select kob_id, kob_summa
  from kn_operbook
  where kob_from_ol_id = :ol_id
  into kob_id, kob_summa
  do
  begin
    if (first_line = 1) then
    begin
      if (currect_amount = kob_summa) then
        stay_kob_id = kob_id;
    end else
      total_amount = total_amount + kob_summa;

    first_line = 0;
  end


  if (stay_kob_id is not null and total_amount = wrong_amount - currect_amount) then
  begin
    delete from kn_operbook
    where kob_from_ol_id = :ol_id and
          kob_id <> :stay_kob_id;
  end else
  begin
    select kob_id, kob_ol_date, kob_acc_dt, kob_acc_kt
    from kn_operbook
    where kob_from_ol_id = :ol_id and
          kob_from_kc_id is null
    rows 1
    into kob_id, ol_date, acc_dt, acc_kt;

    pcb_summa = 0;
    total_amount = 0;
    fixed = 0;

    for select pcb_amount_sum
    from packet_1c_imp_bank
    where pcb_from_ol_id = :ol_id and
          pcb_acc_dt_id = :acc_dt and
          pcb_acc_kt_id = :acc_kt and
          pcb_date = :ol_date
    into pcb_summa
    do
    begin
      total_amount = total_amount + pcb_summa;

      if (currect_amount = total_amount) then
      begin
        fixed = 1;

        delete from kn_operbook
        where kob_from_ol_id = :ol_id and
              kob_id <> :kob_id;
  
        update kn_operbook
        set kob_summa = :currect_amount,
            kob_summa_tax_dt = :currect_amount,
            kob_summa_tax_kt = :currect_amount
        where kob_id = :kob_id
        rows 1;

        exit;
      end
    end

    if (fixed = 0) then
      exception e_default 'Something wrong, ol_id = ' || ol_id;
  end
end
