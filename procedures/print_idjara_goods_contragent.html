CREATE PROCEDURE PRINT_IDJARA_GOODS_CONTRAGENT(
    OBJ_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    PERIOD D_INTEGER,
    K_DISB_DATE D_DATE,
    AG_CAPTION D_STR_255,
    AG_DATE_BEGIN D_DATE,
    AG_AGREEMENT_NO VARCHAR(30),
    K_AGENT_FIO D_STR_100,
    K_AGENT_INN D_STR_20,
    K_AGENT_ADDRESS D_STR_255,
    QR_CODE D_STR_LONG,
    ML_NAME D_STR_100,
    ML_MODEL D_STR_255,
    KML_COST DECIMAL(18,2),
    KML_COUNT DECIMAL(18,2),
    KML_TOTAL DECIMAL(18,2),
    SUM_KML_TOTAL DECIMAL(18,2),
    K_AGENT_EDOC_DATETIME D_DATETIME,
    REG_NAME D_STR_100,
    SIGN_DATE D_DATE,
    K_KONTRACT_NO D_STR_20,
    KMI_IMEI D_STR_20)
AS
declare variable OBJ_KIND D_INTEGER;
declare variable EDL_ID D_ID;
declare variable AG_C_ID D_ID;
declare variable REG_ID D_INTEGER;
declare variable K_ID D_ID;
declare variable MA_ID D_INTEGER;
declare variable AG_ID D_ID;
declare variable KML_ID D_INTEGER;
declare variable KMA_FROM_AG_ID D_ID;
begin
  obj_kind = f_objkind('kontract', 'id');

  select ag_object_id,
         k_id,
         k_kontract_no,
         k_period,
         k_disb_date,
         k_from_ma_app_id,
         kma_from_ag_id
  from kontmatag
  inner join agreements on ag_id = kma_from_ag_id
  inner join kontract on k_id = kma_from_k_id
  where kma_id = :obj_id and
        ag_object_kind = 0 and
        ag_status = 1 and
        k_disb_date between ag_date_begin and coalesce(ag_date_end, '31.12.9999')
  into ag_c_id,
       k_id,
       k_kontract_no,
       period,
       k_disb_date,
       ma_id,
       kma_from_ag_id;

  select reg_name
  from region
  inner join miniapp on ma_office_id = reg_unicode
  where ma_id = :ma_id and
        ma_is_active = 1 and
        ma_date_end = '31.12.9999'
  into reg_name; -- офис берем из заявки, т.к. акт приема-передачи узуфрукта формируется до выдачи займа (rfckg-2091)

  select ag_id,
         ag_caption,
         ag_date_begin,
         ag_agreement_no
  from agreements
  inner join edoc_list on edl_object_kind = f_objkind('agreements', 'id') and edl_object_id = ag_id
  inner join edoc_sign on eds_from_edl_id = edl_id
  where ag_object_kind = 0 and
        ag_object_id = :ag_c_id and
        ag_from_rd_id = f_get_rd_id('agreementkinds.ipf_idjara') and
        ag_status = 1 and
        eds_status = 1
  rows 1
  into ag_id, -- подписанный продавцом ген. договор о сотрудничестве (rfckg-2091)
       ag_caption,
       ag_date_begin,
       ag_agreement_no;

  select cast(eds.eds_status_datetime as date)
  from edoc_list edl
  inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
  where edl.edl_object_kind = 35 and
        edl.edl_object_id = :ag_id and
        edl.edl_from_edc_id = f_edoc_kind('gpsi.agreement', 'id') and
        edl.edl_status = 1
  into sign_date;

  select cl.c_name_fio,
         cl.c_inn,
         adl.adl_address_str
  from clients cl
  inner join addrclient adc on adc.adc_from_c_id = cl.c_id
  inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
  where cl.c_id = :ag_c_id and
        adc.adc_address_type = 1
  order by adl.adl_datetime desc
  rows 1
  into k_agent_fio,
       k_agent_inn,
       k_agent_address;

  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);
    qr_code = f_edoc_qr_by_edl(:edl_id, :ag_c_id);

    select eds.eds_status_datetime
    from edoc_list edl
    inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
    where edl.edl_id = :edl_id and
          eds.eds_from_c_id = :ag_c_id and
          eds.eds_status = 1 and
          edl.edl_status = 1
    into k_agent_edoc_datetime;
  end

  for select ml.ml_name,
             ml.ml_model,
             kml.kml_cost,
             kml.kml_count,
             kml.kml_total,
             sum(kml.kml_total) over(partition by kml.kml_object_id) as sum_kml_total,
             kml.kml_id
      from kontmat kml
      inner join matlist ml on ml.ml_id = kml.kml_from_ml_id
      where kml.kml_object_kind = :obj_kind and
            kml.kml_object_id = :k_id and
            kml.kml_from_ag_id = :kma_from_ag_id
      into ml_name,
           ml_model,
           kml_cost,
           kml_count,
           kml_total,
           sum_kml_total,
           kml_id
  do
  begin
    kmi_imei = '';

    select cast(kmi_imei as d_str_20)
    from kontmat_imei
    where kmi_kml_id = :kml_id
    into kmi_imei;

    suspend;
  end
end
