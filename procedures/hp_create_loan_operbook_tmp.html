CREATE PROCEDURE HP_CREATE_LOAN_OPERBOOK_TMP(
    OL_ID DECIMAL(18,0),
    K_ID DECIMAL(18,0),
    OPER_TYPE INTEGER,
    REPAY_FACT_OFFICE_ID INTEGER,
    IS_REPAY_FROM_SALARY INTEGER,
    REPAY_DT_ACCOUNT_KIND INTEGER,
    REPAY_DT_ACCOUNT_NO VARCHAR(20),
    REPAY_CURR_RATE DECIMAL(18,4))
RETURNS (
    NEW_OL_ID DECIMAL(18,0),
    NEW_OL_PARENT_ID DECIMAL(18,0),
    RES_REG_ID INTEGER,
    RES_CURRENCY VARCHAR(3),
    RES_ACC_DT VARCHAR(20),
    RES_ACC_KT VARCHAR(20),
    TOTAL_LINE INTEGER,
    RES_LINE_NO INTEGER,
    RES_AMOUNT DECIMAL(18,4),
    RES_MEMO VARCHAR(255),
    RES_ACTION VARCHAR(20),
    CM_DATABASE_ID INTEGER,
    CURR_CURRENCY VARCHAR(3),
    RES_RECORD_NO INTEGER)
AS
declare variable repay_loan_office_id integer;
declare variable repay_user_id decimal(18,0);
declare variable repay_doc_date date;
declare variable repay_loan_office_code varchar(10);
declare variable acc_kt varchar(20);
declare variable parent_tmp decimal(18,0);
declare variable amount_os decimal(18,4) = 0;
declare variable amount_perc decimal(18,4) = 0;
declare variable amount_fine decimal(18,4) = 0;
declare variable amount_tax decimal(18,4) = 0;
declare variable amount_kommis decimal(18,4) = 0;
declare variable amount_pereplat decimal(18,4) = 0;
declare variable total_amount decimal(18,4) = 0;
declare variable pr_id integer;
declare variable ac_type integer;
declare variable amount_os_normal decimal(18,4) = 0;
declare variable amount_os_delinq decimal(18,4) = 0;
declare variable ac_no varchar(20);
declare variable ac_delinq varchar(20);
declare variable ac_writeoff varchar(20);
declare variable res_acc_dt_1 varchar(20);
declare variable res_acc_dt_2 varchar(20);
declare variable dm_loan_name varchar(20);
declare variable reg_current integer;
declare variable loan_currency varchar(3);
begin
-- тестовые данные:
-- Основная сумма кредита Погашение кредита согл. договора 0401-058166-05 от 24.10.2014 Кр.папка № 4-1-69-132
-- Офис учета: 0401 - Ош
--  ol_id=1090100004123142;
--  k_id=109010005816600501;
-- конец парамтеров

  /* select cm_database_id, cm_default_currency
  from company
  into cm_database_id, curr_currency;   */
  :cm_database_id = f_get_db_id();
  select cm_default_currency
  from company
  into curr_currency;

  res_record_no = 0;

  if (cm_database_id = 81001) then
    dm_loan_name = 'займ';
  else
    dm_loan_name = 'кредит';

  if (oper_type is null) then
    oper_type = 0;

  select ol_date, ol_user_id
  from operlist
  where ol_id = :ol_id
  into :repay_doc_date, repay_user_id;

  select kal_new_reg_id, k_from_pr_id, k_index_curr_code
  from knaktlist, kontract
  where k_id = :k_id and
        k_from_kf_id = kal_from_kf_id and
        knaktlist.is_delete = 0 and
        kal_date <= :repay_doc_date and
        kal_date_end >= :repay_doc_date
  into :repay_loan_office_id, :pr_id, :loan_currency;

  if (oper_type = 1) then
  begin
-- Если операция СПИСАНИЕ, то строим проводки без Лоро-ностро
    repay_fact_office_id = repay_loan_office_id;

    select res_account
    from hp_make_kredit_acclist(:pr_id, :repay_loan_office_id, 10)
    into :repay_dt_account_no;
  end

  amount_os = 0;
  amount_perc = 0;
  amount_fine = 0;
  amount_kommis = 0;
  amount_tax = 0;
  amount_pereplat = 0;
  amount_os_normal = 0;
  amount_os_delinq = 0;

  select -sum(ok_os), -sum(ok_perc), -sum(ok_fine), -sum(ok_kommis), -sum(ok_tax),
         sum(iif(ok_sh_date >= :repay_doc_date, -ok_os, 0.0)), sum(iif(ok_sh_date < :repay_doc_date, -ok_os, 0.0))
  from operkred
  where ok_from_ol_id = :ol_id and
        ok_from_k_id = :k_id and
        ok_from_c_id = 0 and
        ok_ol_is_active = 1
--  and ok_type=(4+:oper_type)
        and
        ok_type in (4, 5)
  into :amount_os, :amount_perc, :amount_fine, :amount_kommis, :amount_tax, :amount_os_normal, :amount_os_delinq;

  if (amount_os is null) then
    amount_os = 0;
  if (amount_os_normal is null) then
    amount_os_normal = 0;
  if (amount_os_delinq is null) then
    amount_os_delinq = 0;
  if (amount_perc is null) then
    amount_perc = 0;
  if (amount_fine is null) then
    amount_fine = 0;
  if (amount_kommis is null) then
    amount_kommis = 0;
  if (amount_tax is null) then
    amount_tax = 0;

  if (oper_type = 0) then
  begin
    select -sum(ok_os)
    from operkred
    where ok_from_ol_id = :ol_id and
          ok_from_k_id = :k_id and
          ok_from_c_id = 0 and
          ok_ol_is_active = 1 and
          ok_type = 9
    into :amount_pereplat;
    if (amount_pereplat is null) then
      amount_pereplat = 0;
  end

  total_amount = amount_os + amount_perc + amount_fine + amount_kommis + amount_tax + amount_pereplat;

  res_acc_dt = repay_dt_account_no;
  new_ol_id = ol_id;
  new_ol_parent_id = null;
  res_reg_id = repay_fact_office_id;
  res_acc_kt = '';
  reg_current = 1;
  total_line = 1;
  res_line_no = 0;

  res_currency = loan_currency;
  res_action = 'No_Action';

  if (repay_loan_office_id <> repay_fact_office_id) then
  begin
    res_amount = total_amount;
    new_ol_id = 0;

    repay_loan_office_code = (select reg_code
                              from region
                              where reg_unicode = :repay_loan_office_id);
    acc_kt = repay_loan_office_code || substring(repay_dt_account_no from 5 for 20);

    for select res_reg_id, res_acc_dt, res_acc_kt, reg_current, total_line, res_line_no
        from aarep_loronostro_link(:repay_fact_office_id, :repay_dt_account_no, :acc_kt)
        into :res_reg_id, :res_acc_dt, :res_acc_kt, :reg_current, :total_line, :res_line_no
    do
    begin
      res_action = 'No_Action';

      if (res_line_no <> total_line) then
        new_ol_id = new_ol_id + 1;
      else
        new_ol_id = ol_id;
      if (res_line_no = 1) then
        parent_tmp = new_ol_id;

      if (new_ol_parent_id is null) then
      begin
        if (res_line_no = 1) then
          new_ol_parent_id = null;
        else
          new_ol_parent_id = parent_tmp;
      end

-- Здесь обновляем OL_Parent_ID для New_OL_ID
      if ((new_ol_parent_id is not null) and
          (new_ol_id <> new_ol_parent_id)) then
        res_action = 'Update_Parent';

-- Здесь надо создать новую запись в Operlist
      if (new_ol_id <> ol_id) then
        res_action = 'Appen_Operlist';

      if ((res_line_no < total_line) and
          (res_amount <> 0)) then
      begin
        res_memo = 'Погашение ' || dm_loan_name || 'а через другой офис';
        res_record_no = res_record_no + 1;
        suspend;
      end
      repay_dt_account_no = res_acc_dt;
    end
  end

-- Если это кредитная операция (Не лоро-ностро)
  res_amount = amount_os;
  if (res_amount <> 0) then
  begin
    res_acc_kt = '';
    ac_type = 1; -- Основная сумма
    select res_account, res_acc_delinq, res_acc_writeoff
    from hp_make_kredit_acclist(:pr_id, :repay_loan_office_id, :ac_type)
    into :ac_no, :ac_delinq, :ac_writeoff;

    if (oper_type = 1) then
    begin
      ac_no = ac_writeoff;
      ac_delinq = ac_writeoff;

      ac_type = 10; -- РППУ по ОС
      select res_account, res_acc_delinq
      from hp_make_kredit_acclist(:pr_id, :repay_loan_office_id, :ac_type)
      into :res_acc_dt_1, :res_acc_dt_2;
    end
    if (ac_no = ac_delinq) then
    begin
      res_amount = amount_os;
      res_memo = 'Основная сумма ' || dm_loan_name || 'а';
      res_acc_kt = ac_no;
      if (oper_type = 1) then
        res_acc_dt = res_acc_dt_1;
      res_record_no = res_record_no + 1;
      suspend;
    end
    else
    begin
      res_amount = amount_os_normal;
      res_memo = 'Основная сумма ' || dm_loan_name || 'а';
      res_acc_kt = ac_no;
      if (oper_type = 1) then
        res_acc_dt = res_acc_dt_1;
      res_record_no = res_record_no + 1;
      suspend;
      res_amount = amount_os_delinq;
      res_memo = 'Основная сумма ' || dm_loan_name || 'а (просроченная)';
      res_acc_kt = ac_delinq;
      if (oper_type = 1) then
        res_acc_dt = res_acc_dt_2;
      res_record_no = res_record_no + 1;
      suspend;
    end
  end

  res_amount = amount_perc;
  res_memo = 'Проценты по ' || dm_loan_name || 'у';
  if (res_amount <> 0) then
  begin
    res_acc_kt = '';
    ac_type = 3; -- Проценты
    select res_account, res_acc_writeoff
    from hp_make_kredit_acclist(:pr_id, :repay_loan_office_id, :ac_type)
    into :ac_no, :ac_writeoff;
    res_acc_kt = ac_no;

    if (oper_type = 1) then
    begin
      ac_no = ac_writeoff;

      ac_type = 15; -- РППУ по процентам
      select res_account
      from hp_make_kredit_acclist(:pr_id, :repay_loan_office_id, :ac_type)
      into :res_acc_dt_1;

      res_acc_dt = res_acc_dt_1;
    end
    res_record_no = res_record_no + 1;
    suspend;
  end

  res_acc_dt = repay_dt_account_no;

  res_amount = amount_fine;
  res_memo = 'Пеня за просрочку';
  if ((res_amount <> 0) and
      (oper_type = 0)) then
  begin
    res_acc_kt = '';
    ac_type = 5; -- Штрафы
    select res_account
    from hp_make_kredit_acclist(:pr_id, :repay_loan_office_id, :ac_type)
    into :ac_no;
    res_acc_kt = ac_no;
    res_record_no = res_record_no + 1;
    suspend;
  end

  res_amount = amount_pereplat;
  res_memo = 'Переплата ' || dm_loan_name || 'а';
  if ((res_amount <> 0) and
      (oper_type = 0)) then
  begin
    res_acc_kt = '';
    ac_type = 8; -- Переплата
    select res_account
    from hp_make_kredit_acclist(:pr_id, :repay_loan_office_id, :ac_type)
    into :ac_no;
    res_acc_kt = ac_no;
    res_record_no = res_record_no + 1;
    suspend;
  end

  res_amount = amount_tax;
  res_memo = 'Выплачен НсП на проценты ' || dm_loan_name || 'а';
  if (res_amount <> 0) then
  begin
    select nsp_acc_int_repay, nsp_acc_int_main
    from hp_get_kontract_nsp_acc(:repay_loan_office_id, :repay_doc_date)
    into :ac_no, res_acc_dt_1;
    res_acc_kt = ac_no;
    if (oper_type = 1) then
      res_acc_dt = res_acc_dt_1;
    res_record_no = res_record_no + 1;
    suspend;
  end

  res_amount = amount_kommis;
  res_memo = 'Штраф за досрочное погашение от суммы переплаты';
  if ((res_amount <> 0) and
      (oper_type = 0)) then
  begin
    res_acc_kt = '';
    ac_type = 26; -- Комиссия
    select res_account
    from hp_make_kredit_acclist(:pr_id, :repay_loan_office_id, :ac_type)
    into :ac_no;
    res_acc_kt = ac_no;
    res_record_no = res_record_no + 1;
    suspend;
  end

end
