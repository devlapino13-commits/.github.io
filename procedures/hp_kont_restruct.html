CREATE PROCEDURE HP_KONT_RESTRUCT(
    REP_KONTRACT_NO D_STR_20,
    REP_DATE D_DATE)
RETURNS (
    OK_NUMBER D_INTEGER,
    OK_SH_DATE D_DATE,
    OK_SH_OS D_CURRENCY,
    OK_SH_PERC D_CURRENCY,
    INFO BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable k_id d_id;
declare variable k_disb_date d_date;
declare variable k_amount_approved d_currency;
declare variable kf_id d_id;
declare variable kal_new_kf_name d_str_100;
declare variable pr_name d_str_code;
declare variable reg_name d_str_code;
declare variable clients_fio d_str_255;
declare variable ol_id d_id;
declare variable ol_datetime d_datetime;
declare variable ol_memo d_str_long;
declare variable ol_from_rd_id d_integer;
declare variable rd_name d_str_255;
declare variable un_name d_str_code;
declare variable eol d_str_10;
begin
  -- процедура возвращает информацию о проведенной реструктуризации контракта (RFCKG-3136)

  select k_id,
         k_from_kf_id,
         k_disb_date,
         k_amount_approved,
         pr_name
  from kontract
  inner join kontstate on ks_from_k_id = k_id
  inner join product on pr_id = k_from_pr_id
  where k_kontract_no = :rep_kontract_no and
        current_date between ks_date_begin and ks_date_end and
        ks_ol_is_delete = 0 and
        ks_state = 1
  into k_id,
       kf_id,
       k_disb_date,
       k_amount_approved,
       pr_name;

  if (k_id is null) then
    exit;

  select kal.kal_new_kf_name,
         left(reg.reg_code || ' ' || reg.reg_name, 50)
  from knaktlist kal
  inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
  where kal.kal_from_kf_id = :kf_id and
        kal.kal_date_end = '31.12.9999' and
        kal.is_delete = 0
  into kal_new_kf_name,
       reg_name;

  select left(list(c_name_fio, ', '), 255)
  from clients
  inner join knclient on kc_from_c_id = c_id
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into clients_fio;

  if (char_length(clients_fio) = 255) then
    clients_fio = left(clients_fio, 252) || '...';

  select ok_from_ol_id
  from operkred
  where ok_ol_is_active = 1 and
        ok_from_k_id = :k_id and -- фильтр по контракту
        ok_ol_date = :rep_date and -- фильтр по дате
        ok_type = 2 -- реструктуризация
  order by ok_number desc rows 1
  into ol_id;

  select ol_datetime,
         ol_memo,
         ol_from_rd_id,
         un_name
  from operlist
  inner join username on un_id = ol_user_id
  where ol_id = :ol_id
  into ol_datetime,
       ol_memo,
       ol_from_rd_id,
       un_name;

  select rd_name
  from refdata
  where rd_id = :ol_from_rd_id
  into rd_name;

  eol = ascii_char(13) || ascii_char(10) || ascii_char(13) || ascii_char(10);
  info = blob_append(info, 'Контракт: ID ' || k_id || eol);
  info = blob_append(info, 'Cумма: ' || f_numtostr(k_amount_approved, 0) || ', дата выдачи ' || f_datetostr(k_disb_date) || eol);
  info = blob_append(info, 'Кредитный продукт: ' || pr_name || eol);
  info = blob_append(info, 'Кредитная папка: ' || kal_new_kf_name || ', офис ' || reg_name || eol);
  info = blob_append(info, 'Клиенты: ' || clients_fio || eol);
  info = blob_append(info, 'Операция: ' || ol_memo || ', ID ' || ol_id || ', дата ' || left(ol_datetime, 16) || ' (' || lower(un_name) || ')' || eol);

  for select ok_number,
             ok_sh_date,
             ok_sh_os,
             ok_sh_perc
      from operkred
      where ok_ol_is_active = 1 and
            ok_from_k_id = :k_id and -- фильтр по контракту
            ok_ol_date = :rep_date and -- фильтр по дате
            ok_type = 2 -- реструктуризация
      order by ok_number
      into ok_number,
           ok_sh_date,
           ok_sh_os,
           ok_sh_perc
  do
    suspend;

end
