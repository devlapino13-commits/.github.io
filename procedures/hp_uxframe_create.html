CREATE PROCEDURE HP_UXFRAME_CREATE(
    CREATE_KIND D_STR_CODE,
    TABLE_NAME D_STR_CODE,
    FR_CODE D_STR_CODE,
    FR_NAME D_STR_100,
    FR_SOURCE D_STR_CODE)
RETURNS (
    RES_FR_CODE D_STR_CODE)
AS
declare variable ufr_show_kind d_smallint;
declare variable grid_ufi_code d_str_100;
declare variable root_ufi_code d_str_100;
declare variable ux_code d_str_code;
declare variable tmp_ufi_code d_str_code;
begin
  if (exists(select * from uxcode
  where ux_code = :fr_code)) then
    exception e_message 'Кодовое название занято, придумайте новую';

  ufr_show_kind = 2;
  if (create_kind = 'uxform_one') then
  begin
    ufr_show_kind = 1;
    create_kind = 'uxform';
  end
  else
    if (create_kind = 'uxcommand') then
      ufr_show_kind = 0;

  -- Создаем фрейм
  select res_ux_code
  from hp_uxframe_update('', :fr_code, :table_name, :fr_name, :ufr_show_kind,
                             1, 0, 0, 0, :fr_source, '', '', '', '')
  into :fr_code;

  res_fr_code = fr_code;

  -- Синхронизируем поля источника
  execute procedure hp_uxfrdata_sync_params(:fr_code, 1);

  root_ufi_code = fr_code || '#';
  grid_ufi_code = '';

  -- Если набор данных, создаем грид
  if (ufr_show_kind = 2) then
  begin
    select ufi_code
    from hp_uxfritem_aux_add(:fr_code, 'group', 'grid', :root_ufi_code)
    into :grid_ufi_code;
  end

  -- Если грид не создан, за ветку грида берем главную ветку
  if (grid_ufi_code = '') then
    grid_ufi_code = root_ufi_code;

  -- Добавляем все поля и параметры в интерфейс
  insert into uxfritem (ufi_ux_code, ufi_ufp_code, ufi_kind, ufi_position, ufi_parent_code)
  select :fr_code, ufp_code, 0, ufp_index, iif(ufp_is_param = 0, :grid_ufi_code, :root_ufi_code)
  from hp_uxframe_params(:fr_code)
  where not(:create_kind = 'uxcommand' and ufp_is_param = 0) and
        ufp_in_ui = 0;

  if (create_kind <> 'uxframe') then
  begin
    ux_code = uxf$kind(create_kind, 'prefix');
    ux_code = ux_code || substring(fr_code from 3);

    select res_ux_code
    from hp_uxcode_insert(:ux_code, :table_name, :fr_name, :create_kind)
    into :ux_code;

    if (ux_code <> '') then
    begin
      select ufi_code
      from hp_uxfritem_add(:ux_code, 'form', '')
      into :tmp_ufi_code;

      select ufi_code
      from hp_uxfritem_add(:ux_code, 'frame', :fr_code)
      into :tmp_ufi_code;
    end
  end

  suspend;
end
