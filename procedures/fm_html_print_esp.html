CREATE PROCEDURE FM_HTML_PRINT_ESP(
    UN_ID D_ID,
    OBJ_ID D_ID)
RETURNS (
    RES_TEXT D_STR_2048)
AS
declare variable REG_ID D_INTEGER;
declare variable LANG D_STR_LANG;
begin
  select un.un_curr_reg_id
  from username un
  where un.un_id = :un_id
  into :reg_id;
  res_text = '<body>' || f_uri_blank(f_lang(lang, 'Распечатать договор НЦП русский'), 'print', f_objkind('username', 'id'), :un_id, 1643,
             f_lang(lang, 'Печать бланка НЦП на русском'), 1, 0, f_format('{"reg_id":%1}', :reg_id)) || '<br><br>' ||
             f_uri_blank(f_lang(lang, 'Распечатать договор НЦП кыргызский'), 'print', f_objkind('username', 'id'), :un_id, 1706,
             f_lang(lang, 'Печать бланка НЦП на кыргызском'), 1, 0, f_format('{"reg_id":%1}', :reg_id)) || '<br>' ||
             '</body>';

  suspend;
end
