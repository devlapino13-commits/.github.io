CREATE PROCEDURE UXP$UXGRID_SET_EXCEL(
    FRAME_UUID_CODE D_STR_CODE,
    EXCEL_BLOB D_BLOB_BINARY,
    EXCEL_DATASET D_STR_CODE,
    EXCEL_GROUP_FIELD D_STR_255)
AS
declare variable fr_uuid d_uuid;
declare variable grid_uuid d_uuid;
begin
  fr_uuid = char_to_uuid(frame_uuid_code);

  if (exists(select * from ux$grid
             where ugr_uuid = :fr_uuid)) then
    grid_uuid = fr_uuid;
  else
  begin
    select ugr_uuid
    from ux$grid
    where ugr_ufr_uuid = :fr_uuid
    order by ugr_datetime desc
    rows 1
    into grid_uuid;

    if (row_count = 0) then
    begin
      select ugr_uuid
      from ux$frame, ux$grid
      where ufr_owner = :fr_uuid and
            ugr_ufr_uuid = ufr_uuid
      order by ugr_datetime desc
      rows 1
      into grid_uuid;

      if (row_count = 0) then
        exception e_message 'Grid not found: ' || frame_uuid_code;
    end
  end

  update ux$grid
  set ugr_excel = :excel_blob,
      ugr_ds_code = :excel_dataset,
      ugr_group_field = :excel_group_field
  where ugr_uuid = :grid_uuid;

  if (row_count = 0) then
    exception e_message 'Grid not found: ' || frame_uuid_code;
end
