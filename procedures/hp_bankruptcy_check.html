CREATE PROCEDURE HP_BANKRUPTCY_CHECK
AS
declare variable K_ID D_ID;
declare variable BFE_ID D_ID;
declare variable BANK_DATE D_DATE;
declare variable OPER_COUNT D_INTEGER;
begin
    for select bfe_id,
               bfe_from_k_id,
               bfe_bankruptcy_date
        from bankruptcy_for_exec bfe
        where bfe_status = 0  -- только не обработанные
        into bfe_id,
             k_id,
             bank_date
    do
    begin
        select count(*)
        from operkred ok
        where ok.ok_from_k_id = :k_id and
              ok.ok_ol_is_active = 1 and
              ok.ok_ol_date >= :bank_date
        into :oper_count;

        if (:oper_count = 0) then
        begin
            in autonomous transaction do
            begin
                execute procedure hp_kontract_stop_perc(:k_id, :bank_date, :bank_date);
    
                update bankruptcy_for_exec
                set bfe_status = 1
                where bfe_id = :bfe_id;

                update kontract k
                set k.k_stop_perc = :bank_date,
                    k.k_stop_fine = :bank_date
                where k.k_id = :k_id;
            end
        end
        else
        begin
            update bankruptcy_for_exec
            set bfe_status = -1
            where bfe_id = :bfe_id;
        end
    end
end
