CREATE PROCEDURE HP_FIND_NAMELIST_TMP(
    FIND_TEXT D_STR_100,
    FIND_OBJECT_KIND D_SMALLINT,
    FIND_OBJECT_ID D_ID)
RETURNS (
    RES_ID D_ID)
AS
begin
  if (coalesce(find_object_id, 0) = 0) then
  begin
    if (char_length(:find_text) >= 3) then
    begin
      for select distinct nlk_object_id
          from namelist, namelinks
          where nl_name starting :find_text and
                nl_id = nlk_from_nl_id and
                nlk_object_kind = :find_object_kind
          into res_id
      do
        suspend;
    end
    else
    begin
      for select distinct nlk_object_id
          from namelist, namelinks
          where nl_name = :find_text and
                nl_id = nlk_from_nl_id and
                nlk_object_kind = :find_object_kind
          into res_id
      do
        suspend;
    end
  end else
  begin
    res_id = find_object_id;
    if (exists(select *
    from namelist, namelinks
    where nl_name starting :find_text and
          nl_id = nlk_from_nl_id and
          nlk_object_kind = :find_object_kind and
          nlk_object_id = :find_object_id)) then
      suspend;
  end
end
