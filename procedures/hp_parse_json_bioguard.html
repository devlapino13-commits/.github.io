    JSON_INFO D_BLOB_TEXT,
    CLIENT_PASS_DOC_NO D_STR_255)
RETURNS (
    IIN D_STR_20,
    SURNAME D_STR_100,
    FIRSTNAME D_STR_100,
    PATRONYMIC D_STR_100,
    ENG_SURNAME D_STR_100,
    ENG_FIRSTNAME D_STR_100,
    DOB D_DATE,
    GENDER D_STR_10,
    NATIONALITY D_STR_255,
    NATIONALITY_KZ D_STR_255,
    CITIZENSHIP D_STR_255,
    CITIZENSHIP_KZ D_STR_255,
    COUNTRY D_STR_255,
    BIRTH_PLACE D_STR_255,
    DOC_NUMBER D_STR_100,
    DOC_ISSUE_DATE D_DATE,
    DOC_END_DATE D_DATE,
    ISSUE_ORG_NAME D_STR_255,
    C_NAME_FIO D_STR_255,
    IS_DOCUMENT_VALID D_SMALLINT,
    ERROR_TEXT D_STR_255,
    PHONE_NUMBER D_STR_20,
    COUNTRY_CODE D_STR_100,
    COUNTRY_TXT D_STR_255,
    DISTRICT_CODE D_STR_20,
    DISTRICT_TXT D_STR_255,
    REGION_CODE D_STR_100,
    REGION_TXT D_STR_255,
    CITY_TXT D_STR_255,
    STREET_TXT D_STR_255,
    BUILDING_TXT D_STR_255,
    FLAT_TXT D_STR_255,
    COUNTRY_CODE_FACT D_STR_100,
    COUNTRY_TXT_FACT D_STR_255,
    DISTRICT_CODE_FACT D_STR_20,
    DISTRICT_TXT_FACT D_STR_255,
    REGION_CODE_FACT D_STR_100,
    REGION_TXT_FACT D_STR_255,
    CITY_TXT_FACT D_STR_255,
    STREET_TXT_FACT D_STR_255,
    BUILDING_TXT_FACT D_STR_255,
    FLAT_TXT_FACT D_STR_255)
AS
declare variable DOC_COUNT D_INTEGER;
declare variable STATUS_PASS D_INTEGER;
declare variable DOC_ITEM D_BLOB_TEXT;
declare variable I D_INTEGER;
declare variable DOC_PATH D_STR_LONG;
declare variable PERSONAL_DATA_PATH D_STR_255;
declare variable FACT_ADDRESS_PATH D_STR_255;
declare variable TEMP_TYPE_CODE D_STR_10;
begin
    error_text = '';
    personal_data_path = 'digital_docs.data.person_data';
    
    phone_number = f_json_blob_get_value(:json_info, 'phone_number', 1);  -- номер телефона
    iin = f_json_blob_get_value(:json_info, :personal_data_path||'.iin', '');  -- инн
    eng_surname = f_json_blob_get_value(:json_info, :personal_data_path||'.engSurname', '');  -- фамилия на латинице
    eng_firstname = f_json_blob_get_value(:json_info, :personal_data_path||'.engFirstName', '');  -- имя на латинице
    dob = cast(f_json_blob_get_value(:json_info, :personal_data_path||'.dob', '1900-01-01') as date);  -- дата рождения
    gender = f_json_blob_get_value(:json_info, :personal_data_path||'.gender.code', '');  -- пол
    nationality = f_json_blob_get_value(:json_info, :personal_data_path||'.nationality.name_ru', '');  -- Национальность
    nationality_kz = f_json_blob_get_value(:json_info, :personal_data_path||'.nationality.name_kz', '');  -- Национальность
    citizenship = f_json_blob_get_value(:json_info, :personal_data_path||'.citizenship.name_ru', '');  -- Гражданство
    citizenship_kz = f_json_blob_get_value(:json_info, :personal_data_path||'.citizenship.name_kz', '');  -- Гражданство
    country = f_json_blob_get_value(:json_info, :personal_data_path||'.birth_place.country.name_ru', '');  -- Страна рождения
    birth_place = f_json_blob_get_value(:json_info, :personal_data_path||'.birth_place.region.name_ru', '');  -- Место рождения
    
                                  -- ADDRESS --
    city_txt = f_json_blob_get_value(:json_info, :personal_data_path||'.address.city', '');  -- город
    street_txt = f_json_blob_get_value(:json_info, :personal_data_path||'.address.street', ''); -- улица
    building_txt = f_json_blob_get_value(:json_info, :personal_data_path||'.address.building', ''); -- дом
    flat_txt = f_json_blob_get_value(:json_info, :personal_data_path||'.address.flat', '');  -- квартира
    country_code = f_json_blob_get_value(:json_info, :personal_data_path||'.address.country.code', '');  -- страна (код)
    country_txt = f_json_blob_get_value(:json_info, :personal_data_path||'.address.country.name_ru', '') ;  -- страна (текст)
    district_code = f_json_blob_get_value(:json_info, :personal_data_path||'.address.district.code', ''); -- область (код)
    district_txt  = f_json_blob_get_value(:json_info, :personal_data_path||'.address.district.name_ru', ''); -- область (текст)
    region_code = f_json_blob_get_value(:json_info, :personal_data_path||'.address.region.code', '');   -- район (код)
    region_txt  = f_json_blob_get_value(:json_info, :personal_data_path||'.address.region.name_ru', ''); -- район (текст)


    fact_address_path = :personal_data_path||'.addressTemp';
    if (f_json_blob_get_count(:json_info, :fact_address_path||'.Address') > 0) then
    begin
        city_txt_fact = f_json_blob_get_value(:json_info, :fact_address_path||'.Address[0].city', '');  -- город
        street_txt_fact = f_json_blob_get_value(:json_info, :fact_address_path||'.Address[0].street', ''); -- улица
        building_txt_fact = f_json_blob_get_value(:json_info, :fact_address_path||'.Address[0].building', ''); -- дом
        flat_txt_fact = f_json_blob_get_value(:json_info, :fact_address_path||'.Address[0].flat', '');  -- квартира
        country_code_fact = f_json_blob_get_value(:json_info, :fact_address_path||'.Address[0].country.code', '');  -- страна (код)
        country_txt_fact = f_json_blob_get_value(:json_info, :fact_address_path||'.Address[0].country.name_ru', '') ;  -- страна (текст)
        district_code_fact = f_json_blob_get_value(:json_info, :fact_address_path||'.Address[0].district.code', ''); -- область (код)
        district_txt_fact  = f_json_blob_get_value(:json_info, :fact_address_path||'.Address[0].district.name_ru', ''); -- область (текст)
        region_code_fact = f_json_blob_get_value(:json_info, :fact_address_path||'.Address[0].region.code', '');   -- район (код)
        region_txt_fact  = f_json_blob_get_value(:json_info, :fact_address_path||'.Address[0].region.name_ru', ''); -- район (текст)
        
    end
    else
    begin
        city_txt_fact = :city_txt;
        street_txt_fact = :street_txt;
        building_txt_fact = :building_txt;
        flat_txt_fact = :flat_txt;
        country_code = :country_code_fact;
        country_txt = :country_txt_fact;
        district_code_fact = :district_code;
        district_txt_fact = :district_txt;
        region_code_fact = :region_code;
        region_txt_fact = :region_txt;
    end
                                      -- DOCUMENTS --
    i = 0;
    is_document_valid = 0;
    doc_count = coalesce(f_json_blob_get_count(:json_info, :personal_data_path || '.documents.document'), 0);

    while (i < doc_count) do
    begin
        doc_path = f_format('%1.documents.document[%2]', :personal_data_path, :i);
        doc_item = cast(f_json_blob_get_value(:json_info, :doc_path, '')as d_blob_text);

        if (doc_item is null or doc_item = '') then
        begin
            i = i + 1;
            continue;
        end
    
        temp_type_code   = f_json_str_get_value(:doc_item, 'type.code', '');
        status_pass = f_json_str_get_value(:doc_item, 'status.code', '');
        doc_end_date = cast(left(f_json_str_get_value(:doc_item, 'end_date', ''), 10) as date);
        doc_number = f_json_str_get_value(:doc_item, 'number', '');
    
        if (temp_type_code = '002'
          and status_pass not in (1, 5, 7, 8, 9, 10, 11, 12, 13, 14, 16, 17, 18, 19, 20, 21)
          and doc_end_date > current_date + 14
          and doc_number = :client_pass_doc_no -- Ищем в ответе от биогарда именно тот номер документа, который был распознан с загруженного клиентом паспорта
        ) then
        begin
            is_document_valid = 1;
            surname = f_json_blob_get_value(:doc_item, 'surname', '');  -- фамилия
            firstname = f_json_blob_get_value(:doc_item, 'name', '');  -- имя
            patronymic = f_json_blob_get_value(:doc_item, 'patronymic', '');  -- отчество
            
            doc_issue_date = cast(left(f_json_str_get_value(:doc_item, 'begin_date', ''), 10)as date);
            issue_org_name = f_json_str_get_value(:doc_item, 'issue_organization.name_ru', '');
            
            leave;
        end
    
        i = i + 1;
    end

    c_name_fio = f_translit_fio_grs('orig', 'orig', coalesce(:surname, '') || ' ' || coalesce(:firstname, '') || ' ' || coalesce(:patronymic, ''), 'kz');
    suspend;
end
