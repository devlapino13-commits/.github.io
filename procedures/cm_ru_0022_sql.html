CREATE PROCEDURE CM_RU_0022_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kf_client_pass_no varchar(30);
declare variable c_id id_type;
declare variable c_birth_date date;
declare variable c_cell_no_date date;
declare variable c_verify_status integer;
begin
    for
    select mac_from_c_id, c_name_fio, C_pass_no, c_birth_date, --c_cell_no_date,
    c_verify_status
    from miniappclient, clients
    where mac_from_ma_id=:ma_id
    and (mac_amount>0 or mac_is_poruch=1)
    and c_type = 0
    and mac_from_c_id=c_id
    into :C_ID, :res_info_text, :kf_client_pass_no, :c_birth_date, --:c_cell_no_date,
    c_verify_status
    do
    begin
      c_cell_no_date = null;
      select c_cell_no_date
      from AAREP_GET_CONTINFO('clients', :c_id)
      into c_cell_no_date;
        res_error_text='';
        res_error_is_critical=0;
        if (:c_birth_date is null) then
        begin 
          res_error_is_critical=1;
          res_error_text='Дата рождения не заполнена';
          Suspend;
        end
        if (coalesce(:kf_client_pass_no,'') = '') then
        begin 
          res_error_is_critical=1;
          res_error_text='Номер паспорта не заполнен';
          Suspend;
        end
    end
end
