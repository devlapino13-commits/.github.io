CREATE PROCEDURE PRINT_CURR_SCHEDULE(
    PRINT_K_ID DECIMAL(18,0))
RETURNS (
    SH_ID INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_TOTAL DECIMAL(18,4),
    SH_FINE DECIMAL(18,4))
AS
declare variable sh_dd date;
declare variable sh_calc_fine decimal(18,4);
begin
  /* Получение даты разделения */
  select max(kd_date)
  from kontdevides
  where kd_k_orig = :print_k_id or
        kd_k_new = :print_k_id
  into sh_dd;

  /* Получение обновленного графика*/
  sh_fine = 0;
  for select sh_id, sh_date, zad_os, zad_perc, sh_calc_fine
      from aarep_make_zad_list(:print_k_id, coalesce(:sh_dd, current_date), 1)
      where sh_date is not null and
            zad_os + zad_perc <> 0
      into sh_id, sh_date, sh_os, sh_perc, sh_calc_fine
  do
  begin
    sh_total = sh_os + sh_perc;
    sh_fine = sh_fine + sh_calc_fine;
    suspend;
  end

end
