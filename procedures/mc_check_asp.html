CREATE PROCEDURE MC_CHECK_ASP(
    MCL_ID D_ID)
RETURNS (
    HAS_ASP D_SMALLINT)
AS
declare variable C_ID D_ID;
declare variable MCL_LOGIN D_STR_CODE;
declare variable C_VERIFY_STATUS D_INTEGER;
begin
  has_asp = 1;
  if (f_get_db_id() in (41701, 39801) and
      f_get_const('ces_active') = 1) then
  begin
    c_verify_status = 0;
    select mcl.mcl_from_c_id
    from mc_login mcl
    where mcl.mcl_id = :mcl_id
    into c_id;

    select c.c_verify_status
    from clients c
    where c.c_id = :c_id
    into c_verify_status;

    if (f_asp_signed(:c_id, current_date) <> 1 and
        c_verify_status = 2) then
      has_asp = 0;
  end
  suspend;

end
