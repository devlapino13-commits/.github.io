CREATE PROCEDURE MC_CREATE_ONLINE_APPLICATION(
    MCL_ID D_ID,
    AMOUNT D_CURRENCY,
    PERIOD D_SMALLINT)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255,
    MA_ID D_ID)
AS
declare variable ma_no d_integer;
declare variable lo_region_id d_id;
declare variable loanfolderid d_id;
declare variable ma_is_online d_bool;
declare variable disb_kind d_integer;
declare variable currency_id varchar(3);
declare variable loanpercent d_percent;
declare variable loanmanagerid d_id;
declare variable loanrepayday d_smallint;
declare variable is_refund d_bool;
declare variable loanproductid d_id;
declare variable loankommispercent d_percent;
declare variable add_month d_smallint;
declare variable add_info d_str_255;
declare variable tkt_id d_id;
declare variable c_id d_id;
declare variable mac_loan_target d_integer;
declare variable mac_is_cashier d_smallint;
declare variable mac_business_subtype d_integer;
declare variable mac_kommis_lgota d_currency;
declare variable mac_allow_transfer_rights d_smallint;
declare variable loandisbdate d_date;
declare variable res_schedule_id d_integer;
declare variable ool_id d_id;
declare variable edl_id d_id;
begin
    if (f_get_db_id() <> 81001) then
        exit;

    c_id = (select mcl_from_c_id from mc_login where mcl_id = :mcl_id);
    if (coalesce(c_id, 0) = 0) then
    begin
        res_code = 'error';
        res_text = 'Не удалось загрузить данные профиля. Попробуйте войти в приложение заново';
        suspend;
        exit;
    end

    /*

        Ключевые параметры создаваемой кредитной заявки



        !!! ВАЖНО !!!
        Логика согласованная с Зарой: папки создаются на ГО, сотрудник виртуальный, зарегистрирован также на ГО,
        НО его код (из krexpert): ГО OiVO. То есть, номера папок будут строиться от этого кода следующим образом:
        ГО OiVO-1, ГО-OiVO-2, ГО OiVO-3, ...

    */

    lo_region_id = 1010100; -- Офис выдачи займа (ГО)
    loanmanagerid = 1000000000005; -- Кредитный специалист (Виртуальный сотрудник OiVO)

    -- Обнуляем переменные (эти значения либо не нужны, либо потом перезапишутся)
    loanpercent = 0; -- Процентная ставка по займу
    loankommispercent = 0;
    add_info = ''; -- Дополнительная информация по займу
    tkt_id = null; -- ID тикета

    -- Дефолтные значения переменных (эти значения попадут в кредитную заявку)
    loandisbdate = current_date; -- Дата выдачи займа
    loanproductid = 156; -- ID кредитного продукта (Для тестов временно берем стандартный продукт OiVO)
    loanrepayday = 0; -- Предпочитаемый день погашения займа. 0 - определить автоматически
    is_refund = 0; -- Рефинансирование (выдача займа имея другой активный основной заем)
    add_month = 0; -- Сколько месяцев дополнительно нужно добавить к первому погашению по графику (НЕ добавляем)
    disb_kind = 7; -- Способ выдачи займа (СБП по номеру телефона)
    ma_is_online = 1; --  Выдача онлайн (без посещения офиса)

    -- Валюта выдачи займа
    currency_id = (select cm_default_currency from company);

    select perc_default, kommis_perc_default, add_info
    from hp_get_percent(:loanproductid, :loandisbdate, 1, :amount, :period, :amount, :amount, cast(:c_id as d_str_255), :add_month, :is_refund, :disb_kind)
    into :loanpercent, :loankommispercent, :add_info;

    -- Создание кредитной папки
    execute procedure insert_new_kfolder(:lo_region_id, :loanmanagerid, current_date, 0, '', 0, '')
    returning_values :loanfolderid;

    -- Создание кредитной заявки
    update or insert into miniapp
    (
        ma_id, ma_number, ma_no, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id,
        ma_kolich_repay, ma_percent, ma_kommis_perc, ma_kolich_client, ma_memo, ma_is_active,
        ma_is_auto, ma_summa_minus, ma_from_un_id, ma_office_id, ma_currency_id,
        ma_percent_category, ma_manager_id, ma_repay_day, ma_source_system, ma_refund,
        ma_add_month, ma_add_info, ma_disb_kind, ma_from_tkt_id, ma_is_online
    )
    values
    (
        -1, -1, -1, current_date, :loandisbdate, :loanfolderid, :loanproductid,
        :period, :loanpercent, :loankommispercent, 1, '', 1, 0, :amount, 1000000000000,
        :lo_region_id, :currency_id, :loanpercent, :loanmanagerid, :loanrepayday, 1, :is_refund, :add_month, :add_info,
        :disb_kind, :tkt_id, :ma_is_online
    )
    matching (ma_id, ma_number)
    returning ma_id, ma_no
    into :ma_id, :ma_no;

    if (coalesce(ma_id, 0) = 0) then
    begin
        res_code = 'error';
        res_text = 'Не удалось создать заявку на займ. Пожалуйста, попробуйте позже';
        suspend;
        exit;
    end

    /*
        
        Ключевые параметры добавления клиента в заявку

    */

    mac_loan_target = 6; -- Цель получения займа (Потребительский)
    mac_is_cashier = 1; -- Является ли клиент казначеем в заявке
    mac_business_subtype = 21; -- Субтип бизнеса (Потребительский)
    mac_kommis_lgota = 0; -- ????
    mac_allow_transfer_rights = 1; -- Разрешение на уступку прав требования (оставляем дефолтное значение)

    insert into miniappclient
    (
        mac_from_reg_id, mac_from_ma_id, mac_from_c_id, mac_amount, mac_is_poruch,
        mac_loan_target, mac_is_cashier, mac_business_subtype, mac_kommis_lgota,
        mac_allow_transfer_rights
    )
    values
    (
        0, :ma_id, :c_id, :amount, 0,
        :mac_loan_target, :mac_is_cashier, :mac_business_subtype, :mac_kommis_lgota,
        :mac_allow_transfer_rights
    );

    execute procedure hp_miniapp_kontract_sync(f_objkind('miniapp', 'id'), :ma_id);

    -- Строим график для заявки
    select res_id
    from hp_make_schedule(1000000000000, f_objkind('miniapp', 'id'), :ma_id, 0, null, 0)
    into res_schedule_id;

    -- Запись о том, что создана онлайн-заявка через OiVO
    insert into mc_oivo_online_loan (ool_from_mcl_id, ool_stage, ool_from_ma_id, ool_from_k_id)
    values (:mcl_id, f_enum_id('oivo_online_loan_stage', 'miniapp_created'), :ma_id, null)
    returning ool_id
    into :ool_id;

    -- Отправляем клиенту эл. документ заявки на займ. Не используем hp_edoc_ins_loan_application,
    -- т.к. проверки здесь не нужны
    select edl_id, res_text
    from hp_edoc_insert('loan.application', f_objkind('miniapp', 'id'), :ma_id, 1000000000000)
    into :edl_id, :res_text;

    insert into mc_oivo_online_loan_edoc (oole_from_ool_id, oole_packet_miniapp, oole_packet_kontract)
    values (:ool_id, :edl_id, null);

    if (coalesce(res_text, '') <> '') then
    begin
        -- Заменяем системный текст из процедуры, его нельзя показывать клиенту
        res_code = 'error';
        res_text = 'Не удалось отправить электронный документ. Пожалуйста, попробуйте снова или обратитесь к техническое поддержке';
        suspend;
        exit;
    end

    res_code = 'success';
    res_text = '';

    suspend;

    when any do
    begin
        res_code = 'error';
        res_text = 'Не удалось создать заявку на займ. Пожалуйста, попробуйте позже';
        suspend;
    end
end
