CREATE PROCEDURE HP_OPERCASH_DAILY_TOTAL(
    REP_CASH_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    RES_CATEGORY D_STR_100,
    RES_LINE_CATION D_STR_100,
    RES_BEGIN D_CURRENCY,
    RES_PLUS D_CURRENCY,
    RES_MINUS D_CURRENCY,
    RES_END D_CURRENCY)
AS
declare variable al_id d_str_20;
declare variable cash_name d_str_20;
declare variable res_line_no d_integer;
declare variable cls_al_id d_str_20;
declare variable ipf_al_id d_str_20;
begin
  res_line_no = 0;

  select '# ' || trim(iif(csp_kind = 0, 'Касса', 'Расчет')) || ' ' || csp_cash_no,
         csp_cls_al_id, csp_ipf_al_id
  from cashplace
  where csp_id = :rep_cash_id
  into cash_name, cls_al_id, ipf_al_id;

  if (row_count = 0) then
    exit;

  res_category = cash_name;

  for select csp_cls_al_id
  from cashplace
  where csp_id = :rep_cash_id and
        csp_cls_al_id is not null
  union
  select csp_ipf_al_id
  from cashplace
  where csp_id = :rep_cash_id and
        csp_ipf_al_id is not null
  into al_id
  do
  begin
    res_line_cation = '';
    res_begin = 0;
    res_plus = 0;
    res_minus = 0;
    res_end = 0;

    select al_account
    from acc_list
    where al_id = :al_id
    into res_line_cation;

    res_begin = f_al_bal(al_id, rep_date - 1, 'sum');

    select sum(ob_summa)
    from operbook
    where ob_account_dt = :al_id and
          ob_ol_date = :rep_date and
          ob_ol_is_active = 1
    into res_plus;

    if (res_plus is null) then
      res_plus = 0;

    select sum(ob_summa)
    from operbook
    where ob_account_kt = :al_id and
          ob_ol_date = :rep_date and
          ob_ol_is_active = 1
    into res_minus;

    if (res_minus is null) then
      res_minus = 0;

    res_end = res_begin + res_plus - res_minus;
    res_line_no = res_line_no + 1;

    suspend;
  end

  res_category = 'В процессе';

  if (cls_al_id is not null) then
  begin
    select al_account
    from acc_list
    where al_id = :cls_al_id
    into res_line_cation;
  
    res_begin = 0;
    res_plus = 0;
    res_minus = 0;
    res_end = 0;
  
    select sum(iif(opc_sign = 1, opc_amount, 0)), sum(iif(opc_sign = -1, opc_amount, 0))
    from opercash o
    where opc_from_csp_id = :rep_cash_id and
          opc_date = :rep_date and
          opc_status >= 0 and
          opc_from_ol_id is null and
          opc_is_ipf = 0 and
          opc_is_delete = 0
    into res_plus, res_minus;
  
    if (res_plus is null) then
      res_plus = 0;
  
    if (res_minus is null) then
      res_minus = 0;

    suspend;
  end

  if (ipf_al_id is not null) then
  begin
    select al_account
    from acc_list
    where al_id = :ipf_al_id
    into res_line_cation;

    res_begin = 0;
    res_plus = 0;
    res_minus = 0;
    res_end = 0;
  
    select sum(iif(opc_sign = 1, opc_amount, 0)), sum(iif(opc_sign = -1, opc_amount, 0))
    from opercash o
    where opc_from_csp_id = :rep_cash_id and
          opc_date = :rep_date and
          opc_status >= 0 and
          opc_from_ol_id is null and
          opc_is_ipf = 1 and
          opc_is_delete = 0
    into res_plus, res_minus;
  
    if (res_plus is null) then
      res_plus = 0;
  
    if (res_minus is null) then
      res_minus = 0;
  
    suspend;
  end
end
