CREATE PROCEDURE CM_RU_0265_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable DISB_DATE D_DATE;
declare variable KOLICH_REPAY D_INTEGER;
declare variable C_ID D_ID;
declare variable POINT D_INTEGER;
declare variable PR_ID D_INTEGER;
declare variable AMOUNT_MAX D_CURRENCY;
declare variable AMOUNT D_CURRENCY;
declare variable CLI_BALANCE D_CURRENCY;
declare variable MA_REFUND D_INTEGER;
declare variable REFUND_BALANCE D_CURRENCY;
declare variable LIMIT_MAX D_CURRENCY;
declare variable REFUND_BALANCE_ONE D_CURRENCY;
declare variable AUTHORIZED_AMOUNT D_CURRENCY;
declare variable MDS D_CURRENCY;
declare variable PR_CODE D_STR_100;
declare variable MIN_POINT D_SMALLINT;
declare variable PR_NAME D_STR_100;
declare variable K_ID D_ID;
declare variable CLOSE_DATE D_DATE;
declare variable MAX_DELINQ_DAY D_INTEGER;
declare variable K_KONTRACT_NO D_STR_20;
declare variable IS_ACTIVE D_SMALLINT;
declare variable HAS_INTERNAL_DELINQ D_SMALLINT;
declare variable EXT_OVERDUE D_CURRENCY;
begin
  if (ma_id = 2452418) then
    exit;

  limit_max = 200000;
  -- проверка по продукту Привилегия
  for
  select ma_date_oper, ma_kolich_repay, pr_id, ma_refund, pr_code, pr_name
  from miniapp, product
  where ma_id = :ma_id
  and ma_is_active = 1
  and ma_date_end = '31.12.9999'
  and ma_from_pr_id = pr_id
  and (pr_code containing '.privilege' or pr_code containing '.refinance.' or pr_code containing '.refinance_oivo.')
  into disb_date, kolich_repay, pr_id, ma_refund, pr_code, pr_name
  do
  begin
    -- если имеется активный займ то запрещено брать залоговый
    select mac_from_c_id, mac_amount from miniappclient
    where mac_from_ma_id = :ma_id
    and mac_amount > 0
    rows 1
    into c_id, amount;

    point = f_point_get(11, ma_id, c_id, disb_date);

    min_point = 0;
    if (pr_code containing '.privilege.') then min_point = 37;
    else if (pr_code containing '.privilege_plus.') then min_point = 16;

    if (pr_code containing '.refinance.') then min_point = 25;

    if (pr_code containing '.refinance_oivo.') then min_point = 0;

    if (pr_code containing '.refinance.') then
    begin
      has_internal_delinq = 0;

      if (exists (
        select * from knclient kc
        inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
        inner join oksummary oks on oks.oks_from_k_id = kc.kc_from_k_id
        where kc.kc_from_c_id = :c_id
          and kc.kc_amount_approved > 0
          and ks.ks_ol_is_delete = 0
          and ks.ks_state = 1
          and :disb_date between ks.ks_date_begin and ks.ks_date_end
          and :disb_date between oks.oks_date_begin and oks.oks_date_end
          and oks.oks_close_state = 0
          and :disb_date > oks.oks_delinq_date
      )) then has_internal_delinq = 1;

      if (has_internal_delinq = 1) then 
      begin
        if (point < 25) then
        begin
          res_error_text = 'У клиента меньше 25 баллов и есть активная просрочка, кредитование запрещено';
          res_error_is_critical = 2;
          suspend;
          min_point = :point;
        end
        else
        begin
          min_point = 25; 
        end
      end
      else
      begin
        min_point = 0;
      end
    end

    if (point < min_point) then
    begin
      res_error_text = f_format('Для продукта: %1, минимальный бал : %2, Бал кл: %3', pr_name, min_point, point);
      res_error_is_critical = 2;
      suspend;
    end

    -- по активным кредитам не должно быть просрочки
    -- проверка на просрочку
    is_active = 0;
    for
    select ks_from_k_id
    from knclient, kontstate
    where kc_from_c_id = :c_id
    and kc_amount_approved > 0
    and kc_from_k_id = ks_from_k_id
    and ks_ol_is_delete = 0
    and ks_state = 1
    and :disb_date between ks_date_begin and ks_date_end
    into k_id
    do
    begin
      is_active = 1;

      -- клиента активный
        max_delinq_day = null;
        select :disb_date - oks_delinq_date
        from oksummary
        where oks_from_k_id = :k_id
        and :disb_date between oks_date_begin and oks_date_end
        and oks_close_state = 0
        and :disb_date > oks_delinq_date
        into max_delinq_day;

        if (max_delinq_day is not null) then
        begin
          select k_kontract_no from kontract
          where k_id = :k_id
          into k_kontract_no;
  
          res_error_text = f_format('В контракте: %1 имеется просрочка, кол-во просроченных дней: %2', k_kontract_no, max_delinq_day);
          res_error_is_critical = 2;
          suspend;
        end

    end 

    -- если не имеется активных контрактов то проверяется по последнему контракт
    if (is_active = 0) then
    begin
      for
      select ks_from_k_id
      from knclient, kontstate
      where kc_from_c_id = :c_id
      and kc_amount_approved > 0
      and kc_from_k_id = ks_from_k_id
      and ks_ol_is_delete = 0
      and ks_state = 2
      order by ks_date_begin desc
      rows 1
      into k_id
      do
      begin
        max_delinq_day = null;
        select max(ok_ol_date - ok_sh_date) from operkred
        where ok_from_k_id = :k_id
        and ok_ol_is_active = 1
        and ok_ol_date <= :disb_date
        and ok_type in (4,5)
        and ok_os + ok_perc < 0
        and ok_ol_date > ok_sh_date
        into max_delinq_day;
  
        if (max_delinq_day is not null and max_delinq_day > 7) then
        begin
          select k_kontract_no from kontract
          where k_id = :k_id
          into k_kontract_no;
  
          res_error_text = f_format('В контракте: %1 имеется просрочка, кол-во просроченных дней: %2', k_kontract_no, max_delinq_day);
          res_error_is_critical = 2;
          suspend;
        end
      end
    end
  end
end
