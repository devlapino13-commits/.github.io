CREATE PROCEDURE HP_FIND_CONST_USAGE(
    CONST_NAME D_STR_CODE,
    IS_TMP D_SMALLINT)
RETURNS (
    OBJECT_KIND D_STR_CODE,
    OBJECT_NAME D_STR_255,
    OBJECT_DESCRIPTION D_STR_8100)
AS
declare variable find_txt d_str_100;
begin
  if (coalesce(const_name, '') = '') then
    exit;

  if (is_tmp = 1)
  then find_txt = 'f_tmp_const(''' || const_name || ''')';
  else find_txt = 'f_get_const(''' || const_name || ''')';

  find_txt = lower(find_txt);

  for select 'procedure', trim(lower(rdb$procedure_name)), coalesce(rdb$description, '')
  from rdb$procedures
  where lower(rdb$procedure_source) containing :find_txt
  union
  select 'function', trim(lower(rdb$function_name)), coalesce(rdb$description, '')
  from rdb$functions
  where lower(rdb$function_source) containing :find_txt
  union
  select 'trigger',
         trim(lower(rdb$trigger_name) || ' ' ||
         trim(iif(position(trim(lower(rdb$relation_name)), trim(lower(rdb$trigger_name))) > 0,
                  '', '(' || trim(lower(rdb$relation_name)) || ')'))),
         coalesce(rdb$description, '')
  from rdb$triggers
  where lower(RDB$TRIGGER_SOURCE) containing :find_txt
  order by 2
  into object_kind, object_name, object_description
  do
    suspend;
end
