CREATE PROCEDURE HP_SURVEY_VIEW_KONT(
    C_ID D_ID)
RETURNS (
    K_ID D_ID,
    K_DISB_DATE D_DATE,
    K_KONTRACT_NO D_STR_20,
    K_STATUS_NAME D_STR_20,
    KC_ID D_ID,
    IS_IPF D_BOOL)
AS
declare variable dtcurr d_date;
begin
  dtcurr = current_date;
  for
    select
      k_id,
      k_disb_date,
      k_kontract_no,
      case ks_state
        when 0 then 'В процессе'
        when 1 then 'Активный'
        when 2 then 'Закрытый'
      end,
      kc_id,
      iif(pr_code containing '.ipf.', 1, 0)
    from knclient
      inner join kontract
        on k_id = kc_from_k_id
          and k_status in (0, 1)
      inner join kontstate
        on ks_from_k_id = k_id
          and ks_date_end >= :dtcurr
          and ks_ol_is_delete = 0
      inner join product
        on pr_id = k_from_pr_id
    where kc_from_c_id = :c_id
      and kc_amount_approved > 0
  into
    :k_id,
    :k_disb_date,
    :k_kontract_no,
    :k_status_name,
    :kc_id,
    :is_ipf
  do
  begin
    suspend;
  end
end
