CREATE PROCEDURE AAREP_CLI_CIRCLE_AE4(
    REP_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(20),
    C_BIRTH_DATE DATE,
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KC_CIRCLE INTEGER,
    KF_OFFICER VARCHAR(100),
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    OKRUG_NAME VARCHAR(50))
AS
declare variable old_k_id decimal(18,0);
declare variable k_deliq_date_begin date;
declare variable is_delinq smallint;
declare variable kc_from_k_id decimal(18,0);
declare variable k_from_kf_id decimal(18,0);
declare variable kc_from_c_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select k_id, k_kontract_no, k_disb_date, k_amount_approved, k_circle, k_circle_internal, kc_amount_approved,
             kc_circle, kc_from_c_id, k_from_kf_id
      from kontstate, knclient, kontract
      where :rep_date_end between ks_date_begin and ks_date_end and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            kc_from_k_id = ks_from_k_id and
            kc_circle > 3 and
            k_id = ks_from_k_id and
            kc_amount_approved > 0
      order by k_id
      into :k_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :k_circle, :k_circle_internal, :kc_amount_approved,
           :kc_circle, :kc_from_c_id, :k_from_kf_id
  do
  begin

    if (:k_id <> :old_k_id) then
    begin
      k_deliq_date_begin = null;
      select ok_sh_date
      from operkred /*Проверка на просрочку или были просрочки в текущем цикле*/
      where ok_ol_is_active = 1 and
            ok_type < 7 and
            ok_ol_date <= :rep_date_end and
            ok_sh_date <= :rep_date_end and
            ok_from_k_id = :k_id and
            ok_from_c_id = 0
      group by ok_from_k_id, ok_sh_date
      having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0 or (max(ok_ol_date - ok_sh_date) > 0)
      order by ok_from_k_id, ok_sh_date
      rows 1
      into :k_deliq_date_begin;

    end
    is_delinq = 0;
    if (:k_deliq_date_begin is not null) then
      is_delinq = 1;

    if (:is_delinq = 0) then
    /*Если не просрочен то выполняет следующее*/
    begin
      for
      /*Вытаскивает по клиенту все предыдущие закрытые контракты*/
          select kc_from_k_id
          from kontstate, knclient
          where ks_ol_is_delete = 0 and
                ks_state = 2 and
                :rep_date_end between ks_date_begin and ks_date_end and
                kc_from_k_id = ks_from_k_id and
                kc_from_c_id = :kc_from_c_id
          into :kc_from_k_id
      do
      begin
        if (is_delinq = 0) then
        begin
          if (exists(select ok_sh_date
                     from operkred /*Проверка были ли по закрытым контрактам просрочки*/
                     where ok_ol_is_active = 1 and
                           ok_type < 7 and
                           ok_ol_date <= :rep_date_end and
                           ok_sh_date <= :rep_date_end and
                           ok_from_k_id = :kc_from_k_id and
                           ok_from_c_id = 0
                     group by ok_from_k_id, ok_sh_date
                     having max(ok_ol_date - ok_sh_date) > 0)) then
            is_delinq = 1;
        end
      end
    end
    old_k_id = :k_id;

    if (is_delinq = 0) then
    begin
      reg_name = null;
      parent_name = null;
      okrug_name = null;
      kf_officer = null;

      select reg1.reg_code || ' ' || reg1.reg_name, reg2.reg_code || ' ' || reg2.reg_name,
             reg3.reg_code || ' ' || reg3.reg_name, c_name_fio
      from knaktlist, region reg1, region reg2, region reg3, clients
      where kal_from_kf_id = :k_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            kal_new_reg_id = reg1.reg_unicode and
            reg1.parent = reg2.reg_unicode and
            reg2.parent = reg3.reg_unicode and
            kal_new_kf_officer = c_id
      into :reg_name, :parent_name, :okrug_name, :kf_officer;

      c_name_fio = null;
      c_pass_no = null;
      c_birth_date = null;

      select c_name_fio, c_pass_no, c_birth_date
      from clients
      where c_id = :kc_from_c_id
      into :c_name_fio, :c_pass_no, :c_birth_date;

      suspend;
    end
  end
end
