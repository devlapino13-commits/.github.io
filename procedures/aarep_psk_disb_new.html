CREATE PROCEDURE AAREP_PSK_DISB_NEW(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    NUMBER_ITEM VARCHAR(20),
    CATEGORY_LOANS VARCHAR(255),
    KC_AMOUNT_APPROVED CURR_TYPE,
    PERCENT_EFFECT CURR_TYPE,
    KOL_KONTRACT INTEGER)
AS
declare variable ITEM_1 integer;
declare variable ITEM_2 integer;
declare variable ITEM_3 integer;
declare variable SUM_PERCENT_EFFECT decimal(18,4);
begin
    -- RFCRU-3364
  for select item_1, item_2, item_3, sum(kc_amount_approved),
             sum(cast(kc_amount_approved as double precision) * k_percent_effect), sum(is_kontract)
      from aarep_psk_disb_temp_new(:rep_date_begin, :rep_date_end)
      group by item_1, item_2, item_3
      into item_1, item_2, item_3, kc_amount_approved, sum_percent_effect, kol_kontract
  do
  begin

    percent_effect = cast(sum_percent_effect as double precision) / kc_amount_approved;

    number_item = item_1 || iif(item_2 = 0, '', '.' || item_2) || iif(item_3 = 0, '', '.' || item_3);
    category_loans = case number_item
                       when '1' then 'Потребительские займы с обеспечением в виде залога (кроме POS-займов и ипотеки)'
                       when '2' then 'Потребительские займы с иным обеспечением (кроме POS-займов и ипотеки)'
                       when '2.1' then 'до 365 дней включительно'
                       when '2.2' then 'свыше 365 дней'
                       when '3' then 'Потребительские займы без обеспечения (кроме POS-займов)'
                       when '3.1' then 'до 30 дней включительно, в том числе:'
                       when '3.1.1' then 'до 30 тыс. руб. включительно'
                       when '3.1.2' then 'свыше 30 тыс. руб.'
                       when '3.2' then 'от 31 до 60 дней включительно, в том числе:'
                       when '3.2.1' then 'до 30 тыс. руб. включительно'
                       when '3.2.2' then 'свыше 30 тыс. руб.'
                       when '3.3' then 'от 61 до 180 дней включительно, в том числе:'
                       when '3.3.1' then 'до 30 тыс. руб. включительно'
                       when '3.3.2' then 'свыше 30 тыс. руб. до 100 тыс. руб. включительно'
                       when '3.3.3' then 'свыше 100 тыс. руб.'
                       when '3.4' then 'от 181 дня до 365 дней включительно, в том числе:'
                       when '3.4.1' then 'до 100 тыс. руб. включительно'
                       when '3.4.2' then 'свыше 100 тысяч рублей'

                       when '3.5' then 'свыше 365 дней, в том числе:'
                       when '3.5.1' then 'до 100 тыс. руб. включительно'
                       when '3.5.2' then 'свыше 100 тысяч рублей'

                       when '4' then 'POS-займы'
                       when '4.1' then 'до 180 дней включительно, в том числе:'
                       when '4.1.1' then 'до 30 тыс. руб. включительно'
                       when '4.1.2' then 'свыше 30 тыс. руб. '
                       when '4.2' then 'от 181 до 305 дней включительно, в том числе:'

                       when '4.2.1' then 'до 30 тысяч рублей включительно'
                       when '4.2.2' then 'свыше 30 тысяч рублей'

                       when '4.3' then 'от 306 до 365 дней включительно, в том числе:'
                       when '4.3.1' then 'до 30 тысяч рублей включительно'
                       when '4.3.2' then 'свыше 30 тысяч рублей'

                       when '4.4' then 'свыше 365 дней'

                       when '5' then 'Потребительские займы, обязательства по которым обеспечены ипотекой'

                       when '6' then 'Потребительские займы, полная стоимость потребительского кредита (займа) по которым равна нулю, в том числе:'
                       when '6.1' then 'потребительские займы без обеспечения (кроме POS-займов)'
                       when '6.2' then 'POS-займы '

                     end;
    number_item = '2.' || number_item;

  end
end
