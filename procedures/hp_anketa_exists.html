CREATE PROCEDURE HP_ANKETA_EXISTS(
    K_ID D_ID,
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    IS_ERROR D_INTEGER,
    RES_TEXT D_STR_255)
AS
declare variable mac_id d_integer;
begin
  is_error = 1;
  res_text = 'Анкета не создана';

  select mac_id
  from kontract
  inner join miniappclient on mac_from_ma_id = k_from_ma_app_id
  where k_id = :k_id and
        mac_from_c_id = :c_id
  into :mac_id;

  if (exists(select *
             from cfp_anketa
             where ca_date = current_date and
                   ca_from_un_id = :un_id and
                   ca_object_kind = f_objkind('miniappclient', 'id') and
                   ca_object_id = :mac_id)) then
  begin
    is_error = 0;
    res_text = '';
  end

  suspend;
end
