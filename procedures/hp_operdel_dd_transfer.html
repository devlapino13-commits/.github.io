CREATE PROCEDURE HP_OPERDEL_DD_TRANSFER(
    REPORT_REGION_ID INTEGER,
    REPORT_ONE_K_ID D_ID,
    REPORT_DATE_BEGIN D_DATE,
    REPORT_DATE_END D_DATE,
    REPORT_DATE_NEW D_DATE,
    REPORT_USER_ID D_ID)
RETURNS (
    NEW_OL_ID D_ID)
AS
declare variable aarep_k_id decimal(18,0);
declare variable curr_reg integer;
declare variable ss_memo varchar(200);
declare variable kont_name d_str_255;
begin
  -- новый вариант переноса даты погашения по графику, просрочки не переносим
  for select oks_from_k_id, k_kontract_no || ' от ' || f_datetostr(k_disb_date)
  from oksummary, kontract
  where oks_date_end >= '31.12.2999' and
        oks_delinq_date between :report_date_begin and :report_date_end and
        oks_from_k_id = :report_one_k_id and
        oks_close_state = 0 and
        oks_from_k_id = k_id
  into :aarep_k_id, kont_name
  do
  begin
    if (new_ol_id is null) then
    begin
      curr_reg = f_hq_id();
    
      select kv
      from get_kv(:curr_reg, 5)
      into :new_ol_id;
    
      if (report_date_begin <> report_date_end) then
        ss_memo = 'Перевод даты погашения (' || :report_date_begin || ' - ' || :report_date_end || ') -> ' || :report_date_new;
      else
        ss_memo = 'Перевод даты погашения ' || :report_date_begin || ' -> ' || :report_date_new;

      ss_memo = ss_memo || ', № ' || kont_name;

      insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                            ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                            ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                            ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month)
      values (:new_ol_id, :report_region_id, 1, :report_date_begin, null, '', 3, :ss_memo, 0, 0, null, :report_user_id,
              null, null, 0, null, null, null, 9, :report_date_begin, null, null, 0, null, null);
    end

    execute procedure hp_operkred_dd_one_change(:aarep_k_id, :new_ol_id, :report_date_begin, :report_date_end, :report_date_new);
  end

  suspend;
end
