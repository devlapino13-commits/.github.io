CREATE PROCEDURE PRINT_KONTPSK_PSK(
    KPS_ID D_ID)
RETURNS (
    KPS_PSK D_PERCENT,
    KPS_PSK_AMOUNT D_CURRENCY)
AS
declare variable k_id d_id;
declare variable ol_id d_id;
declare variable ol_date d_date;
declare variable kc_client_type d_integer;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  -- kps_psk - Полная стоимость кредита (займа) в процентах годовых
  -- kps_psk_amount - Полная стоимость кредита (займа) в денежном выражении
  -- в РФ используется в bln_id = 1538

  select kps_from_k_id,
         kps_psk,
         ol_id,
         ol_date
  from kontpsk, operlist
  where kps_id = :kps_id and
        kps_from_ol_id = ol_id
  into k_id,
       kps_psk,
       ol_id,
       ol_date;

  if (db_id = 81001) then
  begin
    select kc_client_type
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 rows 1
    into kc_client_type;

    if (kc_client_type is null) then
      kc_client_type = 0;

    if (kc_client_type = 2 or
        exists(select *
               from kontract
               inner join product on pr_id = k_from_pr_id
               where k_id = :k_id and
                     pr_code containing '.selfemployed.')) then
      exit; -- в РФ клиентам СЗ и ИП отправляется инфо-лист без ПСК, RFCRU-6031
  end

  select sum(ok_sh_perc + ok_kommis)
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type < 4 and
        ok_ol_date <= :ol_date and
        ok_from_ol_id <= :ol_id
  into kps_psk_amount;

  suspend;
end
