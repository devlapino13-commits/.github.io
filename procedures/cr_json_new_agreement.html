CREATE PROCEDURE CR_JSON_NEW_AGREEMENT(
    C_ID D_ID,
    SHOW_ADD_INFO SMALLINT = 0)
RETURNS (
    NEW_AGREEMENT D_STR_2048)
AS
declare variable bank_code d_integer;
declare variable branch d_str_10;
declare variable distrib_mode d_str_lang;
declare variable status d_integer;
declare variable country d_str_lang;
declare variable city d_str_100;
declare variable street d_str_100;
declare variable e_mails d_str_100;
declare variable agre_nom d_integer;
declare variable client d_str_20;
declare variable coment d_str_255;
declare variable contract d_str_20;
declare variable ctime timestamp;
declare variable a_product d_str_lang;
declare variable enrolled date;
declare variable b_br_id varchar(4);
declare variable office varchar(50);
declare variable office_id d_id;
declare variable bincod varchar(50);
declare variable risk_level d_str_20;
declare variable rep_lang integer;
begin
    :bank_code = 21;
    :branch = '000';
    :distrib_mode = '01';
    :status = 10;
    :country = 'KGZ';
    :enrolled = current_date;
    :city = '_';
    :street = '_';
    :e_mails = '_';
    :client = null;
    :coment = null;
    :contract = null;
    :ctime = current_timestamp;

    select /*cca.cca_id, */cal.cal_product,
        reg.reg_code, reg.reg_name,
        reg.reg_unicode, ckd.ckd_code,
        iif(cal.cal_risk_group is null, 'A', cal.cal_risk_group),
        case
            when c.c_blank_lang is null or (c.c_blank_lang = 'ru') then 3
            when c.c_blank_lang = 'kg' then 3
        end
    from cardaktclient cac
    inner join cardclientacc cca on cac.cac_from_cca_id = cca.cca_id
    inner join cardlist cal on cal.cal_card_id = cac.cac_card_id
    inner join cardembomain cem on cem.cem_id = cal.cal_from_cem_id
    inner join cardkind ckd on ckd.ckd_id = cem.cem_from_ckd_id
    inner join region reg on cac.cac_from_reg_id = reg.reg_unicode
    inner join clients c on cac.cac_from_c_id = c.c_id
    where cac.cac_from_c_id = :c_id
      and cac.cac_status = 1
    into /*:agre_nom,*/ :a_product, :b_br_id, :office, :office_id, :bincod, :risk_level, :rep_lang;

    new_agreement = '{' ||
        f_json_make_pair('product', :a_product, 'str', 1) ||
        f_json_make_pair('bank_code', :bank_code, 'num', 1) ||
        f_json_make_pair('branch', :branch, 'str', 1) ||
        f_json_make_pair('bincod', :bincod, 'str', 1) ||
        f_json_make_pair('enrolled', :enrolled, 'date', 1) ||
        f_json_make_pair('distrib_mode', :distrib_mode, 'str', 1) ||
        f_json_make_pair('status', :status, 'num', 1) ||
        f_json_make_pair('country', :country, 'str', 1) ||
        f_json_make_pair('city', :city, 'str', 1) ||
        f_json_make_pair('street', :street, 'str', 1) ||
        f_json_make_pair('e_mails', :e_mails, 'str', 1) ||
        f_json_make_pair('rep_lang', :rep_lang, 'num', iif(show_add_info = 1, 1, 0));

    if(:show_add_info = 1) then
        begin
        :new_agreement = :new_agreement ||
            f_json_make_pair('client', :client, 'str', 1) ||
            f_json_make_pair('b_br_id', :b_br_id, 'str', 1) ||
            f_json_make_pair('office', :office, 'str', 1) ||
            f_json_make_pair('office_id', :office_id, 'num', 1) ||
            f_json_make_pair('coment', :coment, 'str', 1) ||
            f_json_make_pair('contract', :contract, 'str', 1) ||
            f_json_make_pair('ctime', :ctime, 'datetime', 1) ||
            f_json_make_pair('risk_level', :risk_level, 'str');
        end

    :new_agreement = :new_agreement || '}';

    suspend;
end
