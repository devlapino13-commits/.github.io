CREATE PROCEDURE CB_RU_LO_2012_12(
    REP_REGION_ID INTEGER,
    REP_BONUS_ID INTEGER,
    REP_USER_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    CB_REGION_ID INTEGER,
    CB_PERSONAL_ID ID_TYPE,
    CB_DISB_KOLICH_CLI INTEGER,
    CB_CLOSE_KOLICH_CLI INTEGER,
    CB_ACTIV_KOLICH_CLI INTEGER,
    CB_SUM_VZYSKAN CURR_TYPE,
    CB_DELINQ_BALANCE CURR_TYPE,
    CB_ACTIV_BALANCE CURR_TYPE,
    CB_DISB_KOLICH_CLI_3 INTEGER,
    CB_PERSONAL_STAGE DATE,
    CB_PERSONAL_STAGE_MONTH DECIMAL(15,1),
    PAR7 DECIMAL(15,1),
    CB_IS_GET INTEGER,
    CB_BONUS_TOTAL CURR_TYPE,
    CB_IS_WORKING INTEGER,
    CB_MEMO VARCHAR(255))
AS
declare variable date_begin date;
declare variable date_end date;
declare variable date_end_2 date;
declare variable date_begin_3 date;
begin
  date_end = :rep_date_end;
  date_begin = cast('01.' || extract(month from :date_end) || '.' || extract(year from :date_end) as date); -- Начало периода
--  date_begin = '26.12.2012'; -- временная строка, с февралая 2013 нужно убрать
  date_end_2 = date_begin - 1; -- Конец предыдущего месяца
  if (:rep_date_end between '02.01.2013' and '31.01.2013') then
    date_end_2 = '25.12.2013'; -- Для того чтобы если за январь2013 считать то за пред-й портфель брал 25.12.2012

--  date_end_2 = date_end; -- Для проверки с отчетом о портфеле
  date_begin_3 = date_begin - 32;
  date_begin_3 = cast('01.' || extract(month from :date_begin_3) || '.' || extract(year from :date_begin_3) as date); -- Начало 3 месяц назад
  for with tabl1
      as (select
                 case
                   when ks_date_begin between :date_begin and :date_end then ks_region_id -- если выдан то Офис где был выдан
                   else case
                          when ks_date_end < :date_end then (select ks2.ks_region_id
                                                             from kontstate ks2
                                                             where ks2.ks_ol_is_delete = 0 and
                                                                   ks2.ks_state = 2 and
                                                                   ks2.ks_from_k_id = k_id rows 1) -- если закрыт то Офис где был закрыт
                          else kal_new_reg_id -- если активный то текущий Офис
                        end
                 end as kal_new_reg_id,
-- Расчет актуального офиса
                 kal_new_kf_officer, ks_from_k_id,
                 case
                   when coalesce((select rep_delinq_days
                                  from aarep_kont_delinq_status(ks_from_k_id, :rep_date_end)), 0) > 180 then 'Категория 3'
                   else case
                          when (kal_new_kf_officer = ks_kc_id) then case
                                                                      when ks_date_begin >= '01.06.2012' then 'Категория 1'
                                                                      else case
                                                                             when coalesce((select rep_delinq_days
                                                                                            from aarep_kont_delinq_status(ks_from_k_id, '01.12.2012')), 0) <= 0 then 'Категория 1'
                                                                             when coalesce((select rep_delinq_days
                                                                                            from aarep_kont_delinq_status(ks_from_k_id, '01.12.2012')), 0) between 1 and 180 then 'Категория 2'
                                                                             else 'Категория 3'
                                                                           end
                                                                    end
                          else case
                                 when coalesce((select rep_delinq_days
                                                from aarep_kont_delinq_status(ks_from_k_id, (case
                                                                                               when kal_date > '01.12.2012' then kal_date
                                                                                               else '01.12.2012'
                                                                                             end))), 0) <= 0 then 'Категория 1'
                                 when coalesce((select rep_delinq_days
                                                from aarep_kont_delinq_status(ks_from_k_id, (case
                                                                                               when kal_date > '01.12.2012' then kal_date
                                                                                               else '01.12.2012'
                                                                                             end))), 0) between 1 and 180 then 'Категория 2'
                                 else 'Категория 3'
                               end
                        end
                 end as categ_str,
-- категория контракта
                 case
                   when ks_date_begin between :date_begin and :date_end then k_kolich_client_approved
                   else 0
                 end as disb_kolich_cli,
-- кол-во выданным клиентам
                 case
                   when ks_date_end < :date_end then k_kolich_client_approved
                   else 0
                 end as close_kolich_cli,
-- кол-во закрытых клиентов
                 ks_date_begin as disb_date, -- Дата выдачи контракта
                 case
                   when ks_date_end < :date_end then ks_date_end + 1
                   else null
                 end as close_date,
-- дата закрытия контракта
                 k_kredline_end as close_date_graph, -- дата закрытия по графику
                 case
                   when :date_end between ks_date_begin and ks_date_end then k_kolich_client_approved
                   else 0
                 end as activ_kolich_cli,
-- кол-во активных клиентов
                 coalesce((select rep_delinq_days
                           from aarep_kont_delinq_status(ks_from_k_id, :date_end)), 0) as delinq_day, -- кол-во просроченных дней
                 case
                   when (coalesce((select rep_delinq_days
                                   from aarep_kont_delinq_status(ks_from_k_id, :date_end)), 0) between 8 and 180) then coalesce((select sum(ok_os)
                                                                                                                                 from operkred
                                                                                                                                 where ok_from_k_id = k_id and
                                                                                                                                       ok_from_c_id = 0 and
                                                                                                                                       ok_type not in (1, 2, 3, 9, 10) and
                                                                                                                                       ok_ol_is_active = 1 and
                                                                                                                                       ok_ol_date <= :date_end), 0)
                   else 0
                 end as delinq_balanse,
-- Просроченный баланc от 8-180 дней на конец текущего месяца
                 case
                   when ks_date_begin between :date_begin and :date_end then ks_kc_id -- если выдан то выдавший КС
                   else case
                          when ks_date_end < :date_end then (select ks2.ks_kc_id
                                                             from kontstate ks2
                                                             where ks2.ks_ol_is_delete = 0 and
                                                                   ks2.ks_state = 2 and
                                                                   ks2.ks_from_k_id = k_id rows 1) -- если закрыт то закрывший КС
                          else kal_new_kf_officer -- если активный то текущий КС
                        end
                 end as kc_action,
-- актуальный КС
                 case(case
                        when coalesce((select rep_delinq_days
                                       from aarep_kont_delinq_status(ks_from_k_id, :rep_date_end)), 0) > 180 then 'Категория 3'
                        else case
                               when (kal_new_kf_officer = ks_kc_id) then case
                                                                           when ks_date_begin >= '01.06.2012' then 'Категория 1'
                                                                           else case
                                                                                  when coalesce((select rep_delinq_days
                                                                                                 from aarep_kont_delinq_status(ks_from_k_id, '01.12.2012')), 0) <= 0 then 'Категория 1'
                                                                                  when coalesce((select rep_delinq_days
                                                                                                 from aarep_kont_delinq_status(ks_from_k_id, '01.12.2012')), 0) between 1 and 180 then 'Категория 2'
                                                                                  else 'Категория 3'
                                                                                end
                                                                         end
                               else case
                                      when coalesce((select rep_delinq_days
                                                     from aarep_kont_delinq_status(ks_from_k_id, (case
                                                                                                    when kal_date > '01.12.2012' then kal_date
                                                                                                    else '01.12.2012'
                                                                                                  end))), 0) <= 0 then 'Категория 1'
                                      when coalesce((select rep_delinq_days
                                                     from aarep_kont_delinq_status(ks_from_k_id, (case
                                                                                                    when kal_date > '01.12.2012' then kal_date
                                                                                                    else '01.12.2012'
                                                                                                  end))), 0) between 1 and 180 then 'Категория 2'
                                      else 'Категория 3'
                                    end
                             end
                      end)
                   when 'Категория 2' then -coalesce((select sum(ok_os + ok_perc + ok_fine)
                                                               from operkred
                                                               where ok_ol_is_active = 1 and
                                                                     ok_ol_date between :date_begin and :date_end and
                                                                     ok_from_c_id = 0 and
                                                                     ok_type = 4 and
                                                                     ok_from_k_id = k_id), 0)
                   else 0
                 end as sum_vzyskan,
                 k_kontract_no

          from kontstate, kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1)
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_begin <= :date_end and
                ks_date_end >= :date_begin and
                ks_from_k_id = k_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                :date_end between kal_date and kal_date_end and
                kal_new_reg_id = id_list)

      select kal_new_reg_id, kc_action, sum(disb_kolich_cli), -- кол-во выданных клиентов
             sum(
             case
               when (categ_str = 'Категория 1' and (disb_date < '01.12.2012' or (coalesce(close_date, '01.01.1990') >= close_date_graph))) -- если кредит выдан от 01.12.2012 и после, и закрыт досрочно то их не включает
                 then close_kolich_cli
               else 0
             end),
-- кол-во закрытых по категории 1
             sum(
             case
               when (categ_str in ('Категория 1', 'Категория 2') and delinq_day < 30) then activ_kolich_cli
               else 0
             end),
-- кол-во клиентов катег 1 и 2, которые просроч. дни < 30 (пункт 3.1.1)
             sum(
             case categ_str
               when 'Категория 2' then sum_vzyskan
               else 0
             end),
-- взысканная сумма по категории 2
             sum(
             case categ_str
               when 'Категория 1' then delinq_balanse
               else 0
             end) -- просроченный баланc по категории 1 от 8-180 дней

      from tabl1
      group by kal_new_reg_id, kc_action
      into cb_region_id, cb_personal_id, cb_disb_kolich_cli, cb_close_kolich_cli, cb_activ_kolich_cli, cb_sum_vzyskan,
           cb_delinq_balance
  do
  begin
--  Расчет баланса КС на конец предыдущего месяца по Категории 1
    select sum(
           case(case
                  when coalesce((select rep_delinq_days
                                 from aarep_kont_delinq_status(ks_from_k_id, :rep_date_end)), 0) > 180 then 'Категория 3'
                  else case
                         when (kal_new_kf_officer = ks_kc_id) then case
                                                                     when ks_date_begin >= '01.06.2012' then 'Категория 1'
                                                                     else case
                                                                            when coalesce((select rep_delinq_days
                                                                                           from aarep_kont_delinq_status(ks_from_k_id, '01.12.2012')), 0) <= 0 then 'Категория 1'
                                                                            when coalesce((select rep_delinq_days
                                                                                           from aarep_kont_delinq_status(ks_from_k_id, '01.12.2012')), 0) between 1 and 180 then 'Категория 2'
                                                                            else 'Категория 3'
                                                                          end
                                                                   end
                         else case
                                when coalesce((select rep_delinq_days
                                               from aarep_kont_delinq_status(ks_from_k_id, (case
                                                                                              when kal_date > '01.12.2012' then kal_date
                                                                                              else '01.12.2012'
                                                                                            end))), 0) <= 0 then 'Категория 1'
                                when coalesce((select rep_delinq_days
                                               from aarep_kont_delinq_status(ks_from_k_id, (case
                                                                                              when kal_date > '01.12.2012' then kal_date
                                                                                              else '01.12.2012'
                                                                                            end))), 0) between 1 and 180 then 'Категория 2'
                                else 'Категория 3'
                              end
                       end
                end)
             when 'Категория 1' then coalesce((select sum(ok_os)
                                                        from operkred
                                                        where ok_from_k_id = k_id and
                                                              ok_from_c_id = 0 and
                                                              ok_type not in (1, 2, 3, 9, 10) and
                                                              ok_ol_is_active = 1 and
                                                              ok_ol_date <= :date_end_2), 0)
             else 0
           end)
    from kontstate, kontract, knaktlist
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :date_end_2 between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :date_end_2 between kal_date and kal_date_end -- нужно добавить на расчеты с февраля
--    and '25.12.2012' between kal_date and kal_date_end
          and
          kal_new_kf_officer = :cb_personal_id
    into :cb_activ_balance;
    if (:cb_activ_balance is null) then
      cb_activ_balance = 0;

-- расчет новых клиентов за последние 3 месяца

    if (:rep_date_end > '31.01.2013') then -- С февраля будет действовать то что новых клиентов больше 18ти
    begin
      cb_disb_kolich_cli_3 = null;
      select sum(
             case
               when kc1.kc_circle <= 1 then 1
               else case
                      when (select coalesce(max(ks2.ks_date_begin), '01.01.1990') -- если цикл свыше одного и из предыдущих контрактов нет то будет считать как новый
                            from kontstate ks2, knclient kc2
                            where ks2.ks_ol_is_delete = 0 and
                                  ks2.ks_state = 2 and
                                  ks2.ks_date_begin < ks1.ks_date_begin and
                                  ks2.ks_from_k_id = kc2.kc_from_k_id and
                                  kc2.kc_amount_approved > 0 and
                                  kc2.kc_from_c_id = kc1.kc_from_c_id) < ks1.ks_date_begin - 180 then 1
                      else 0
                    end
             end)
      from kontstate ks1, knclient kc1
      where ks1.ks_ol_is_delete = 0 and
            ks1.ks_state = 1 and
            cast(ks1.ks_date_begin as date) between :date_begin_3 and :date_end and
            ks1.ks_kc_id = :cb_personal_id and
            kc1.kc_from_k_id = ks1.ks_from_k_id and
            kc1.kc_amount_approved > 0
      into :cb_disb_kolich_cli_3;
      if (:cb_disb_kolich_cli_3 is null) then
        cb_disb_kolich_cli_3 = 0;
    end
    else
    begin
      cb_disb_kolich_cli_3 = 18;
    end

-- Вычисляет стаж КС     .
    cb_personal_stage = null;
    cb_is_working = null;
    select cb_personal_stage, cb_is_working
    from cb_personal_data(:cb_region_id, :rep_date_end, :cb_personal_id)
    into cb_personal_stage, cb_is_working;

    /*
      select ps_date_begin from hr_pstatus --, hr_orders , joblist
      where ps_from_c_id = :cb_personal_id
      and ps_date_begin is not null
      order by ps_date_begin
      rows 1
      into :cb_personal_stage;      */

    if (cb_personal_stage is null) then
      cb_personal_stage = :rep_date_end;
    if (cb_is_working is null) then
      cb_is_working = 0;
    cb_personal_stage_month = datediff(day, :cb_personal_stage, :rep_date_end) / 30;

    if (cb_delinq_balance = 0 and
        cb_activ_balance = 0) then
    begin
      par7 = 0;
    end
    else
    begin
      if (cb_activ_balance > 0) then
      begin
        par7 = cb_delinq_balance / cb_activ_balance * 100;
      end
      else
      begin
        par7 = 100;
      end
    end

-- Разработано по пункту 3.1.1

    if (par7 <= 7.0 and
        cb_is_working = 1) then
    begin
      if (cb_disb_kolich_cli_3 >= 18) then
      begin
        if ((cb_personal_stage_month >= 1 and
            cb_personal_stage_month < 6) or (cb_personal_stage_month >= 6 and
            cb_personal_stage_month < 9 and
            cb_activ_kolich_cli >= 50) or (cb_personal_stage_month >= 9 and
            cb_personal_stage_month < 12 and
            cb_activ_kolich_cli >= 75) or (cb_personal_stage_month >= 12 and
            cb_personal_stage_month < 15 and
            cb_activ_kolich_cli >= 100) or (cb_personal_stage_month >= 15 and
            cb_personal_stage_month < 18 and
            cb_activ_kolich_cli >= 125) or (cb_personal_stage_month >= 18 and
            cb_activ_kolich_cli >= 150)) then
        begin
          cb_is_get = 1;
          cb_memo = '';
        end
        else
        begin
          cb_is_get = 0;
          cb_memo = 'Бонус не положено из-за стажа';
        end
      end
      else
      begin
        if (cb_personal_stage_month >= 18 and
            cb_activ_kolich_cli >= 150) then
        begin
          cb_is_get = 1;
          cb_memo = '';
        end
        else
        begin
          cb_is_get = 0;
          cb_memo = 'Бонус не положено из-за кол-во новых клиентов за последние 3 месяца.';
        end
      end
    end
    else
    begin
      cb_is_get = 0;
      if (cb_is_working <> 1) then
        cb_memo = 'Сотрудник был уволен.';
      else
        cb_memo = 'Бонус не положено из-за PAR > 7.';
    end

    cb_bonus_total = 0;
    if (cb_is_get = 1) then
    begin
      cb_bonus_total = (cb_disb_kolich_cli * 500) + (cb_close_kolich_cli * 1000) + (cb_sum_vzyskan * 8 / 100);
    end

    suspend;
  end
end
