CREATE PROCEDURE UXP$UXCOMBO_INFO(
    UFP_UUID D_UUID,
    UFP_UUID_CODE D_STR_CODE = null)
RETURNS (
    UF_SOURCE_CODE D_STR_CODE,
    UF_SOURCE_NAME D_STR_CODE,
    UF_FIELD_NAME D_STR_CODE,
    UF_UUID D_UUID,
    UFC_UFP_UUID D_UUID,
    UFC_HIDE_TEXT D_BOOL,
    UFC_KIND D_INTEGER)
AS
begin
  if (coalesce(ufp_uuid, '') = '') then
    ufp_uuid = uxf$uniq(ufp_uuid_code);

  ufc_ufp_uuid = ufp_uuid;

  select ufc_hide_text, uf_uuid, ufc_kind
  from ux$frcombo, ux$source, ux$field
  where ufc_ufp_uuid = :ufp_uuid and
        us_code = ufc_combo_source and
        uf_owner = us_uuid and
        uf_field_name = ufc_combo_field
  into ufc_hide_text, uf_uuid, ufc_kind;

  select us_code, us_name, uf_field_name
  from ux$field, ux$source
  where uf_uuid = :uf_uuid and
        uf_owner = us_uuid
  into uf_source_code, uf_source_name, uf_field_name;

  suspend;
end
