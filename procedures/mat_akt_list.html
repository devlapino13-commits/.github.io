CREATE PROCEDURE MAT_AKT_LIST(
    AKT_OL_ID DECIMAL(18,0),
    AKT_IN_ID DECIMAL(18,0),
    AKT_OUT_ID DECIMAL(18,0))
RETURNS (
    MAL_PERSON_IN DECIMAL(18,0),
    MAL_PERSON_OUT DECIMAL(18,0),
    MAL_PERSON_NAME_IN VARCHAR(100),
    MAL_PERSON_NAME_OUT VARCHAR(100),
    MAL_ML_NAME VARCHAR(255),
    MAL_ML_MODEL VARCHAR(255),
    MAL_ML_INVENT_STR VARCHAR(50),
    MAL_ML_FULL_DETAIL VARCHAR(2048),
    MAL_ML_YEDIZ VARCHAR(20),
    MAL_KOLICH DECIMAL(18,4),
    MAL_SUMMA DECIMAL(18,4),
    MAL_DOC_DATE DATE,
    MAL_DOC_OFFICE VARCHAR(100),
    MAL_NAME_CLIENT VARCHAR(100),
    MAL_NAME_OFFICE VARCHAR(100),
    MAL_TRANSFER_TYPE INTEGER,
    MAL_DOC_NO INTEGER,
    MAL_WP_IN VARCHAR(100),
    MAL_WP_OUT VARCHAR(100))
AS
declare variable ww1 varchar(100);
declare variable rr1 varchar(100);
declare variable ww2 varchar(100);
declare variable rr2 varchar(100);
declare variable st_staff integer;
declare variable st_client integer;
begin
  for select ol_date, r1.reg_name, c_name_fio, r2.reg_name, ol_doc_no
      from operlist
      left join clients on (ol_document_client = c_id)
      left join region r1 on (ol_from_reg_id = r1.reg_unicode)
      left join region r2 on (ol_document_office = r2.reg_unicode)
      where ol_id = :akt_ol_id
      into :mal_doc_date, :mal_doc_office, :mal_name_client, :mal_name_office, :mal_doc_no
  do
  begin
    if (mal_doc_office is null) then
      mal_doc_office = '';
    if (mal_name_client is null) then
      mal_name_client = '';
    if (mal_name_office is null) then
      mal_name_office = '';
  end

  for select distinct om_matotvet_id, om_matotvet_info, c1.c_name_fio, c2.c_name_fio, ml_name, ml_model, ob_invent_str,
                      ob_full_detail, yl_code, om_kolich_plus, om_summa_plus, w1.wp_name, w2.wp_name, r1.reg_name,
                      r2.reg_name
      from opermat
      left join clients c1 on (c1.c_id = om_matotvet_id)
      left join clients c2 on (c2.c_id = om_matotvet_info)
      left join(workplace w1 left join region r1 on (r1.reg_unicode = w1.wp_from_reg_id)) on (w1.wp_id = om_workplace_id)
      left join(workplace w2 left join region r2 on (r2.reg_unicode = w2.wp_from_reg_id)) on (w2.wp_id = om_workplace_info)
      left join operbill on ob_bill_id = om_bill_id and ob_bill_line_no = om_bill_line_no
      left join matlist on ml_id = ob_from_ml_id
      left join yedizlist on yl_id = ml_from_yl_id
      where om_from_ol_id = :akt_ol_id and
            om_main_line = 1 and
            om_oper_type <= 5 and
            (:akt_in_id = 0 or
            om_matotvet_id = :akt_in_id) and
            (:akt_out_id = 0 or
            om_matotvet_info = :akt_out_id) and
            (om_kolich_plus > 0 or
            om_summa_plus > 0) and
            ob_date_begin <= :mal_doc_date and
            ob_date_end >= :mal_doc_date
      union
      select distinct om_matotvet_info, om_matotvet_id, c2.c_name_fio, c1.c_name_fio, ml_name, ml_model, ob_invent_str,
                      ob_full_detail, yl_code, om_kolich_minus, om_summa_minus, w2.wp_name, w1.wp_name, r2.reg_name,
                      r1.reg_name
      from opermat
      left join clients c1 on (c1.c_id = om_matotvet_id)
      left join clients c2 on (c2.c_id = om_matotvet_info)
      left join(workplace w1 left join region r1 on (r1.reg_unicode = w1.wp_from_reg_id)) on (w1.wp_id = om_workplace_id)
      left join(workplace w2 left join region r2 on (r2.reg_unicode = w2.wp_from_reg_id)) on (w2.wp_id = om_workplace_info)
      left join operbill on ob_bill_id = om_bill_id and ob_bill_line_no = om_bill_line_no
      left join matlist on ml_id = ob_from_ml_id
      left join yedizlist on yl_id = ml_from_yl_id
      where om_from_ol_id = :akt_ol_id and
            om_main_line = 1 and
            om_oper_type <= 5 and
            (:akt_out_id = 0 or
            om_matotvet_id = :akt_out_id) and
            (:akt_in_id = 0 or
            om_matotvet_info = :akt_in_id) and
            (om_kolich_minus > 0 or
            om_summa_minus > 0) and
            ob_date_begin <= :mal_doc_date and
            ob_date_end >= :mal_doc_date
      into :mal_person_in, :mal_person_out, :mal_person_name_in, :mal_person_name_out, :mal_ml_name, :mal_ml_model,
           :mal_ml_invent_str, :mal_ml_full_detail, :mal_ml_yediz, :mal_kolich, :mal_summa, :ww1, :ww2, :rr1, :rr2
  do
  begin
    mal_wp_in = '';
    mal_wp_out = '';
    if (mal_person_name_in is null) then
      mal_person_name_in = '';
    if (mal_person_name_out is null) then
      mal_person_name_out = '';

    if ((ww1 is not null) and
        (rr1 is not null)) then
      mal_wp_in = rr1 || ' ' || ww1;
    if ((ww2 is not null) and
        (rr2 is not null)) then
      mal_wp_out = rr2 || ' ' || ww2;

    mal_transfer_type = 5;
    if ((mal_name_client <> '') and
        (mal_person_name_out = '')) then
      mal_transfer_type = 1;
    if ((mal_name_client <> '') and
        (mal_person_name_in = '')) then
      mal_transfer_type = 2;
    if ((mal_name_office <> '') and
        (mal_person_name_out = '')) then
      mal_transfer_type = 3;
    if ((mal_name_office <> '') and
        (mal_person_name_in = '')) then
      mal_transfer_type = 4;
    suspend;
  end
end
