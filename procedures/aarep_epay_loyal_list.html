RETURNS (
    NUM D_INTEGER,
    C_ID D_ID,
    EPD_ID D_INTEGER,
    EPD_STATUS D_INTEGER,
    EPD_DATE_END D_DATE,
    NAME STR_255,
    SURNAME STR_255,
    MIDLENAME STR_255,
    BIRTHDAY D_DATE,
    IIN STR_100,
    CITIZEN STR_100,
    RESIDENT D_SMALLINT,
    POST_INDEX STR_20,
    REGION_ STR_255,
    COUNTRY STR_255,
    OBL STR_255,
    DISTRICT STR_255,
    CITY STR_255,
    STREET STR_255,
    DEPART_NO STR_255,
    BUILDING_NO STR_255,
    FLAT_NO STR_255,
    SECTION_NO STR_255,

    PASS_COUNTRY STR_100,
    PASS_NO D_STR_100,
    PASS_WHO STR_255,
    PASS_DATE D_DATE,
    PASS_ADD_DATE D_DATE,
    SEX D_STR_CODE,
    AG_ID D_INTEGER,
    C_NAME_FIO D_STR_255,
    AG_DATE_BEGIN D_DATE,
    AG_DATE_END D_DATE,
    REG_NAME D_STR_100,
    DURATION_FROM DATE,
    DURATION_TO DATE,
    DOCUMENT STR_100,
    MIGRATION_SER STR_255,
    MIGRATION_NO STR_255,
    FULL_ADDRESS_REGISTR D_STR_255,
    FULL_ADDRESS_FACT D_STR_255,
    EPAY_AG_STATE D_STR_CODE,
    EPAY_AG_FIRST_DATE D_DATE,
    EPAY_AG_LAST_DATE D_DATE,
    EPAY_CURR_CIRCLE D_INTEGER,
    EPAY_LAST_DATE_STOP D_DATE,
    EPAY_LINK_COUNT_ALL D_INTEGER,
    EPAY_LINK_COUNT_DISB D_INTEGER,
    EPAY_PROFIT_PERIOD D_CURRENCY,
    EPAY_PROFIT_ALL D_CURRENCY,
    EPAY_REPAY_ALL D_CURRENCY,
    EPAY_REPAY_PERIOD D_CURRENCY,
    EPAY_TAX_ALL D_CURRENCY,
    EPAY_TAX_PERIOD D_CURRENCY,
    EPAY_RESIDENT_STR D_STR_CODE,

    EPAY_SALDO D_CURRENCY,
    EPAY_HAS_BANKCARD D_SMALLINT,
    EPAY_BANKCARD_NO D_STR_CODE,
    AE_IS_EXCHANGE_TEXT D_STR_20,
    AGT_MEMO D_STR_255,
    RD_NAME VARCHAR(255),
    AH_K_KONTRACT_NO_DELIQ D_STR_20,
    AH_K_KONTRACT_NO_24 D_STR_20,
    AH_TER D_STR_10,
    AH_DATE D_DATETIME,
    AH_DATE_TER D_DATETIME,
    C_NAME_FIO_KC D_STR_100,
    EPAY_WRITEOFF_PERIOD D_CURRENCY,
    EPAY_WRITEOFF_ALL D_CURRENCY,
    EPAY_SALDO_BEGIN D_CURRENCY,
    EPAY_AMOUNT D_CURRENCY,
    EPAY_TAX_FL D_CURRENCY,
    EPAY_SALDO_DIFF D_CURRENCY)
AS
declare variable k_id d_id;
declare variable rd_id d_integer;
declare variable obj_kind d_integer;
declare variable min_amount d_currency;
declare variable ae_is_exchange d_smallint;
declare variable agt_from_rd_id d_id;
declare variable epd_amount d_currency;
declare variable epay_payed_year d_currency;
declare variable rep_year_begin d_date;
begin
  -- сведения о клиентах по заключенным договорам по программее лояльности
  num = 0;

  rd_id = f_get_rd_id('AgreementKinds.loyal_agreement');
  obj_kind = f_objkind('clients', 'id');

  if (rd_id is null) then
    exception e_default 'rep_rd_code is empty';

  if (coalesce(rep_kind, '') = '') then
    exception e_default 'rep_kind is empty';

  -- rep_kind:
  -- pay   - список для оплаты, отображаются только проверенные
  -- check - список для проверки, отображаются у кого баллы больше 1000
  -- full  - полный список для проверки и анализа

  if (rep_kind not in ('pay', 'check', 'full')) then
    exception e_default 'Unknown rep_kind: ' || rep_kind;

  min_amount = 1000; -- Минимальные баллы для оплаты и отображения в ведомости
  rep_year_begin = cast('01.01.' || extract(year from rep_date_end) as date);

  for select ag_id, c_id, c_name_fio, ag_date_begin, ag_date_end
      from agreements, clients
      where ag_from_rd_id = :rd_id and
            ag_status = 1 and
            ag_object_kind = :obj_kind and
            ag_object_id = c_id and
            ag_date_begin <= :rep_date_end  --and AG_ID=5058
      into :ag_id, c_id, c_name_fio, ag_date_begin, ag_date_end
  do
  begin
    epay_ag_state = '';
    if (ag_date_end is null) then
      epay_ag_state = 'А';
    else
      epay_ag_state = 'ВСИ';

    epay_ag_first_date = null;
    epay_ag_last_date = null;
    epay_curr_circle = 1;
    epay_last_date_stop = null;

    select min(ag_date_begin), max(ag_date_begin), max(ag_date_end), sum(iif(ag_date_begin <= :ag_date_begin, 1, 0))
    from agreements
    where ag_from_rd_id = :rd_id and
          ag_status = 1 and
          ag_object_kind = :obj_kind and
          ag_object_id = :c_id
    into epay_ag_first_date, epay_ag_last_date, epay_last_date_stop, epay_curr_circle;

    epd_id = null;
    epd_date_end = null;
    epd_status = 0;
    select epd_id, epd_status, epd_date_end
    from epay_doc
    where epd_from_c_id = :c_id and
          epd_object_kind = f_objkind('agreements', 'id') and
          epd_object_id = :ag_id
    order by epd_date_end desc, epd_id desc
    rows 1
    into epd_id, epd_status, epd_date_end;

    if (rep_kind = 'pay') then
    begin
      if (coalesce(epd_status, 0) <> 1 or epd_date_end is null or epd_date_end < current_date) then
        continue;
    end

    epay_link_count_all = 0;
    epay_link_count_disb = 0;

    select count(*)
    from clientlink
    where clk_from_c_id = :c_id and
          clk_add_datetime < (:rep_date_end + 1)
    into epay_link_count_all;

    epay_profit_period = 0;
    epay_profit_all = 0;
    epay_link_count_disb = 0;
    epay_repay_all = 0;
    epay_repay_period = 0;
    epay_tax_all = 0;
    epay_tax_period = 0;
    epay_saldo = 0;
    epay_writeoff_period = 0;
    epay_writeoff_all = 0;
    epay_saldo_begin = 0;

    /*
    select sum(iif(ogr_ol_date between :rep_date_begin and :rep_date_end and ogr_sign = 1, ogr_amount, 0)), -- начислено за период
           sum(iif(ogr_sign = 1, ogr_amount, 0)), -- начисленные баллы за все периоды
           sum(iif(ogr_object_id is not null, 1, 0)), -- друзья получившие займ
           sum(iif(ogr_sign = -1, ogr_amount + ogr_tax_fl, 0)), -- выплаченные баллы за все периоды
           sum(iif(ogr_ol_date between :rep_date_begin and :rep_date_end and ogr_sign = -1, ogr_amount + ogr_tax_fl, 0)), -- выплаченные баллы за период
           sum(iif(ogr_sign = -1, ogr_tax_fl, 0)), -- Сумма удержанного НДФЛ за все периоды
           sum(iif(ogr_ol_date between :rep_date_begin and :rep_date_end and ogr_sign = -1, ogr_tax_fl, 0)), -- Сумма удержанного НДФЛ за период
           sum(ogr_amount * ogr_sign * f_loyal(ogr_from_ol_id, :rep_date_end) + ogr_tax_fl * ogr_sign * f_loyal(ogr_from_ol_id, :rep_date_end)) -- Остаток накопленных баллов
    from operagr
    where ogr_ol_date <= :rep_date_end and
          ogr_ol_is_active = 1 and
          ogr_from_ag_id = :ag_id
    into :epay_profit_period, epay_profit_all, epay_link_count_disb, epay_repay_all, epay_repay_period, :epay_tax_all,
         epay_tax_period, epay_saldo;
    */
    select sum(iif(ogr_ol_date between :rep_date_begin and :rep_date_end and ogr_sign = 1, ogr_amount, 0)), -- начислено за период
           sum(iif(ogr_sign = 1, ogr_amount, 0)), -- начисленные баллы за все периоды
           sum(iif(ogr_object_id is not null, 1, 0)), -- друзья получившие займ
           sum(iif(ogr_sign = -1 and ogr_object_id is null, ogr_amount + ogr_tax_fl, 0)), -- выплаченные баллы за все периоды
           sum(iif(ogr_ol_date between :rep_date_begin and :rep_date_end and ogr_sign = -1  and ogr_object_id is null, ogr_amount + ogr_tax_fl, 0)), -- выплаченные баллы за период
           sum(iif(ogr_sign = -1 and ogr_object_id is null, ogr_tax_fl, 0)), -- Сумма удержанного НДФЛ за все периоды
           sum(iif(ogr_ol_date between :rep_date_begin and :rep_date_end and ogr_sign = -1 and ogr_object_id is null, ogr_tax_fl, 0)), -- Сумма удержанного НДФЛ за период
           sum((ogr_amount + ogr_tax_fl) * ogr_sign * f_loyal(ogr_from_ol_id, :rep_date_end)), -- Остаток накопленных баллов
    
           sum(iif(ogr_ol_date between :rep_date_begin and :rep_date_end and ogr_sign = -1 and ogr_object_id is not null, ogr_amount + ogr_tax_fl, 0)), -- списано за период
           sum(iif(ogr_sign = -1 and ogr_object_id is not null, ogr_amount + ogr_tax_fl, 0)), -- списано баллы за все периоды
           sum(iif(ogr_ol_date < :rep_date_begin, (ogr_amount + ogr_tax_fl) * ogr_sign * f_loyal(ogr_from_ol_id, :rep_date_begin - 1), 0)) -- Остаток накопленных баллов на начало периода
    from operagr
    where ogr_ol_date <= :rep_date_end and
          ogr_ol_is_active = 1 and
          ogr_from_ag_id = :ag_id
    into :epay_profit_period, epay_profit_all, epay_link_count_disb, epay_repay_all, epay_repay_period, :epay_tax_all,
          epay_tax_period, epay_saldo,
          epay_writeoff_period, epay_writeoff_all, epay_saldo_begin;

    epd_amount = 0;
    epay_payed_year = 0;
    select sum(ogr_amount * ogr_sign * f_loyal(ogr_from_ol_id, current_date) + ogr_tax_fl * ogr_sign * f_loyal(ogr_from_ol_id, current_date)),
           sum(iif(ogr_sign = -1 and ogr_object_kind is null and ogr_ol_date >= :rep_year_begin, ogr_amount + ogr_tax_fl, 0))
    from operagr
    where ogr_from_ag_id = :ag_id and
          ogr_ol_is_active = 1
    into epd_amount, epay_payed_year;

    if (epd_amount is null) then
      epd_amount = 0;

    if (epay_payed_year is null) then
      epay_payed_year = 0;

    if (epay_profit_period is null) then
      epay_profit_period = 0;
    if (epay_profit_all is null) then
      epay_profit_all = 0;
    if (epay_link_count_disb is null) then
      epay_link_count_disb = 0;
    if (epay_repay_all is null) then
      epay_repay_all = 0;
    if (epay_repay_period is null) then
      epay_repay_period = 0;
    if (epay_tax_all is null) then
      epay_tax_all = 0;
    if (epay_tax_period is null) then
      epay_tax_period = 0;
    if (epay_saldo is null) then
      epay_saldo = 0;

    if (epay_writeoff_period is null) then
      epay_writeoff_period = 0;
    if (epay_writeoff_all is null) then
      epay_writeoff_all = 0;
    if (epay_saldo_begin is null) then
      epay_saldo_begin = 0;

    -----------------------------------

    ah_k_kontract_no_deliq = null;
    ah_k_kontract_no_24 = null;
    ah_ter = null;
    ah_date = null;
    ah_date_ter = null;

    select k30.k_kontract_no, k24.k_kontract_no, iif(agc_terr_match = 1, 'Да', ''), agc_kont_check_datetime,
           agc_terr_check_datetime
    from agreements_checks
    left join kontract k30 on agc_delinq_k_id = k30.k_id
    left join kontract k24 on agc_last_k_id = k24.k_id
    where agc_from_ag_id = :ag_id
    into ah_k_kontract_no_deliq, ah_k_kontract_no_24, ah_ter, ah_date, ah_date_ter;

    agt_from_rd_id = null;
    agt_memo = null;
    select agt_from_rd_id, agt_memo
    from agreements_termination ag
    where agt_from_ag_id = :ag_id rows 1
    into agt_from_rd_id, agt_memo;

    rd_name = null;
    select rd_name
    from refdata
    where rd_main_code = 'agreements_termination' and
          rd_id = :agt_from_rd_id
    into rd_name;

    -----------------------------------

    if (rep_kind in ('check', 'pay')) then
      if (epay_saldo < min_amount and
          c_id <> 1010100002561) then
        continue;

    reg_name = '';
    select reg_code || ' ' || reg_name
    from clientstate, region
    where cs_from_c_id = :c_id and
          cs_region_id = reg_unicode
    into reg_name;

    epay_ag_account = '';
    select f_fmt_acc(acc_id)
    from ref_account
    where acc_object_kind = f_objkind('agreements', 'id') and
          acc_object_id = :ag_id
    order by acc_id + 0
    rows 1
    into epay_ag_account;

    epay_has_bankcard = 0;
    epay_bankcard_no = '';

    -- Закрепление действует полтора года (543 дня), см. RFCRU-1734
    -- После истечения этого периода карту необходимо перезакрепить
    select et_cardnumber
    from extsys_token
    where et_from_c_id = :c_id and
          et_is_actual = 1 and
          dateadd(day,  543, et_token_datetime) > :rep_date_end
    order by et_id desc
    rows 1
    into epay_bankcard_no;
    if (coalesce(epay_bankcard_no, '') <> '') then
      epay_has_bankcard = 1;

    ae_is_exchange = 0;
    ae_is_exchange_text = 'Копить';

    select ae_is_exchange
    from agreement_exchange
    where ae_from_ag_id = :ag_id
    into ae_is_exchange;
    if (ae_is_exchange is null) then
      ae_is_exchange = 0;

    if (ae_is_exchange = 1) then
      ae_is_exchange_text = 'Обменять';

    select name, surname, midlename, birthday, iin, citizen, resident, post_index, region_, country, obl, district,
           city, street, depart_no, building_no, flat_no, section_no, pass_type, pass_ser, pass_country, pass_no,
           pass_who, pass_date, sex, duration_from, duration_to, document, migration_ser, migration_no,
           full_address_registr, full_address_fact, pass_add_date
    from aarep_epay_client_info(:c_id, :rep_doc_date)
    into :name, :surname, :midlename, :birthday, :iin, :citizen, :resident, :post_index, :region_, :country, :obl,
         :district, :city, :street, :depart_no, :building_no, :flat_no, :section_no, :pass_type, :pass_ser,
         :pass_country, :pass_no, :pass_who, :pass_date, :sex, :duration_from, :duration_to, :document, :migration_ser,
         :migration_no, :full_address_registr, :full_address_fact, pass_add_date;

    epay_resident_str = 'Р';
    if (resident = 0) then
      epay_resident_str = 'НР';

    epay_amount = 0;
    epay_tax_fl = 0;

    select epay_new_amount, epay_new_tax
    from hp_epay_local_calc_tax(:ag_id, :resident, :rep_date_end, 0)
    into :epay_amount, :epay_tax_fl;

    epay_saldo_diff = epay_saldo - epd_amount;

    k_id = null;
    c_name_fio_kc = '';
    select k.k_id
    from knclient knc
    inner join kontract k on k.k_id = knc.kc_from_k_id
    where knc.kc_from_c_id = :c_id and
          knc.kc_amount_approved > 0
    order by k.k_fact_disb_date desc
    rows 1
    into :k_id;

    if (k_id is not null) then
    begin
      select cl.c_name_fio
      from kontract k
      inner join knaktlist kn on kn.kal_from_kf_id = k.k_from_kf_id
      inner join clients cl on cl.c_id = kn.kal_new_kf_officer
      where k.k_id = :k_id and
            :rep_date_end between kn.kal_date and kn.kal_date_end and
            kn.is_delete = 0
      into :c_name_fio_kc;
    end
    if (rep_kind in ('full', 'check', 'pay')) then
    begin
      num = num + 1;
      suspend;
    end
  end
end
