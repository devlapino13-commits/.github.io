CREATE PROCEDURE HP_EXTINFO_TO_SEND_MILITARY_PC(
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    JSON_VALUE D_STR_8100)
AS
declare variable MCP_ID D_INTEGER;
begin
  -- RFCRU-3817 Отправка пакета для запроса на получения информации о статусе прохождения воинской службы в КЗ
  if (f_objkind('military_pocket_check', 'id') <> :object_kind) then
        exception e_default 'object_kind is not military_pocket_check';

  select mcp.mcp_id
  from military_check_pocket mcp
  where mcp.mcp_id = :object_id
  into :mcp_id;

  if (coalesce(mcp_id, -1) = -1) then
    exception e_default 'incorrect id in military_pocket_check';

  json_value = '{}'; -- Для отправки запроса не требуется данных от БД
  suspend;
end
