CREATE PROCEDURE HP_PANTERA_IGNORE_INSERT(
    PROJECT STR_255,
    OBJECT_KIND STR_255,
    OBJECT_ID ID_TYPE,
    SUBJECT_KIND INTEGER,
    REASON STR_255,
    PANT_ID ID_TYPE,
    REP_DATE DATE,
    IGNORE_DATE_END D_DATE = null)
RETURNS (
    PI_ID INTEGER)
AS
declare variable OBJECT_KIND_ID integer;
declare variable PROJECT_ID integer;
declare variable REASON_ID integer;
declare variable PI_ID_OLD integer;
begin
  pi_id = null;
  if (rep_date is null) then
    rep_date = current_date;

  object_kind_id = f_objkind(object_kind, 'id');

  if (project similar to '[-]?[[:DIGIT:]]+') then
    project_id = cast(project as integer);
  else
  begin
    select rd_id
    from refdata
    where rd_code = :project
    into :project_id;
  end

  if (reason similar to '[-]?[[:DIGIT:]]+') then
    reason_id = cast(reason as integer);
  else
  begin
    select rd_id
    from refdata
    where rd_code = :reason
    into :reason_id;
  end

  pi_id_old = null;

  if (project_id is null) then
  begin
    select pi.pi_id
    from pantera_ignore pi
    where pi.pi_rd_project_id is null and
          pi.pi_object_kind = :object_kind_id and
          pi.pi_object_id = :object_id and
          pi.pi_subject_kind = :subject_kind and
          pi_date_end > :rep_date rows 1
    into :pi_id_old;
  end
  else
  begin
    select pi.pi_id
    from pantera_ignore pi
    where pi.pi_rd_project_id = :project_id and
          pi.pi_object_kind = :object_kind_id and
          pi.pi_object_id = :object_id and
          pi.pi_subject_kind = :subject_kind and
          pi_date_end > :rep_date rows 1
    into :pi_id_old;
  end

  if (pi_id_old is null) then
  begin
    insert into pantera_ignore (pi_rd_project_id, pi_from_pant_id, pi_object_kind, pi_object_id, pi_reason_rd_id,
                                pi_date_begin, pi_date_end, pi_subject_kind)
    values (:project_id, :pant_id, :object_kind_id, :object_id, :reason_id, :rep_date, :ignore_date_end, :subject_kind)
    returning pi_id
    into :pi_id;
  end
  suspend;
end
