CREATE PROCEDURE CM_AZ_0032_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable KF_CLIENT_PASS_NO varchar(30);
declare variable PR_CODE D_STR_100;
declare variable K_CURRENCY D_STR_10;
declare variable KC_FROM_C_ID D_ID;
declare variable KC_CIRCLE integer;
declare variable C_ID D_ID;
declare variable CIZIZEN_NAME D_STR_255;
declare variable C_BIRTH_DATE date;
declare variable C_INN D_STR_20;
declare variable K_ID_OLD D_ID;
declare variable K_OLD_OL_DATE date;
declare variable K_OLD_DELINQ_DATE date;
declare variable K_OL_DEL_DAY_TMP integer;
declare variable MAX_DELINQ_DAYS integer;
declare variable SUM_DELINQ_DAYS integer;
declare variable PORUCH_K_ID D_ID;
declare variable Z_FROM_C_ID D_ID;
declare variable Z_COST_MARKET D_CURRENCY;
declare variable Z_COST_MARKET_ALL D_CURRENCY;
declare variable C_VERIFY_STATUS integer;
declare variable CL_QTY integer;
declare variable OLD_CONT integer;
declare variable NEW_CONT integer;
declare variable IS_DR integer;
declare variable UN_NAME D_STR_100;
begin
  if (k_id = 101010015313700101) then exit;

  select un.un_name
  from username un
  where un.un_id = :un_id
  into :un_name;

  is_dr = 0;
  if (un_name in ('SKARIMOV', 'ERASULOV', 'VHASANOV', 'ATABDULLAYEV', 'VHADIYEVA', 'SAZIMOV')) then --РМ - список предоставил Саид Мамедов
    is_dr = 1;

  if (check_kind like '.send_kk.') then
    is_dr = 1;

  --РАЗДЕЛ 1. ОБ�©ИЕ УСЛОВИЯ КРЕДИТНЫХ ПРОДУКТОВ
  /* Пункт 1.1 не автоматизирован!!!!!!!!!!!!!!!!!!!
  1.1.    Цель кредитования    Кредитные средства выдаются для следующих целей:
    а.    Создание начального бизнеса, хозяйства, start up и прочее;
    б.    Развитие существующего бизнеса хозяйства и прочее;
    в.    Потребительские нужды.
    
    Запрещённые направления при кредитовании: торговля оружием и боевой техникой,
    производство и продажа наркотических веществ, производство алкогольных и табачных изделий,
    а также любая деятельность, запрещённая действующим законодательством Азербайджанской Республики.
  */
  --1.2.    Валюта     Манат
  select k_disb_date, k.k_currency, pr.pr_code
  from kontract k
  inner join product pr on k_from_pr_id = pr_id
  where k.k_id = :k_id
  into kf_disb_date, k_currency, pr_code;
  if (k_currency <> '944') then
  begin
    res_error_text = 'Валюта выбрана неправильно. Выберите - Манат';
    res_error_is_critical = 1;
    suspend;
  end
  --1.3.    Требования к клиентам/ поручителям
  /*
  1. Требование к клиенту:
    a.  Гражданство Азербайджанской Республики;
    b.  Возрастная категория - от 21 до 70 лет (включительно).
    c.  Во время повторного обращения за кредитом, верхний предел возрастной
          категории для клиентов, которые при предыдущем кредите не попадали в
          просрочку или просрочка возникала не по их вине, поднимается до 72 лет (включительно);
    d.  Трудоспособность;
    e.  Дееспособность;
  */
  for select kc_from_c_id, kc_amount_approved, kc_circle,
             --clients
             c_name_fio, cl.c_pass_no, c_birth_date, cl.c_inn, cl.c_id, --
             --refdata
             rd.rd_name
      from knclient kc --
      inner join clients cl on (kc_from_c_id = c_id)
      inner join refdata rd on (cl.c_citizen = rd.rd_id)
      where (--
            kc_from_k_id = :k_id and
            kc_amount_approved > 0 --
            )
      into kc_from_c_id, --
           kc_amount, --Сумма кредита
           kc_circle, --Цикл клиента
           res_info_text, --ФИО клиента
           kf_client_pass_no, --Номер паспорта
           c_birth_date, --Дата рождения
           c_inn, c_id, cizizen_name --Страна
  do
  begin
    if (:c_id in (1010100364750, 1010100286777)) then
        exit; -- RFCAZ-511

    c_birth_date = coalesce(c_birth_date, current_date);
    cl_qty = coalesce(cl_qty, 0) + 1;
    if (pr_code containing '.IND.' and
        cl_qty > 1) then
    begin
      res_error_text = 'Добавлено несколько учатников. Исправьте сумму клиента!';
      res_error_is_critical = 2;
      suspend;
    end
    --1. Требование к клиенту: a. Гражданство Азербайджанской Республики;
    if (cizizen_name <> 'Азербайджан') then
    begin
      --      res_error_text = 'Требование к клиенту: a. Гражданство Азербайджанской Республики. Указан: ' || CIZIZEN_NAME;
      res_error_text = 'Müştəriyə tələb. Azərbaycan Respublikasının vətəndaşı olmaq zəruridir. Указан: ' || cizizen_name;
      res_error_is_critical = 2;
      suspend;
    end
    if (kc_circle > 1) then
    begin
      --Повторники
      select kc_from_k_id
      from knclient
      inner join kontstate ks on (--
            kc_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin < :kf_disb_date)
      inner join kontract k on kc_from_k_id = k_id
      inner join product pr on (k_from_pr_id = pr_id
            --and PR_CODE containing '.IND.'
            )
      where kc_from_c_id = :kc_from_c_id and
            kc_amount_approved > 0 and
            kc_from_k_id <> :k_id
      order by ks_date_begin desc
      rows 1
      into k_id_old;
      if (k_id_old is null) then
      begin
        res_error_text = 'Последний цикл не определен, обратитесь к разработчику';
        res_error_is_critical = 2;
        suspend;
      end
      else
      begin
        for select max(iif(oks_date_end >= current_date, current_date, oks_date_end + 1)) ol_date, oks_delinq_date
            from oksummary
            where oks_close_state = 0 and
                  oks_from_k_id = :k_id_old and
                  oks_delinq_date <= :kf_disb_date and
                  oks_delinq_date <= oks_date_end
            group by oks_delinq_date
            into k_old_ol_date, k_old_delinq_date
        do
        begin
          k_ol_del_day_tmp = coalesce(k_old_ol_date - k_old_delinq_date, 0);
          if (coalesce(max_delinq_days, 0) < k_ol_del_day_tmp) then
            max_delinq_days = k_ol_del_day_tmp;
          sum_delinq_days = coalesce(sum_delinq_days, 0) + k_ol_del_day_tmp;
        end
        max_delinq_days = coalesce(max_delinq_days, 0);
        sum_delinq_days = coalesce(sum_delinq_days, 0);
      end
      if (sum_delinq_days > 0) then
      begin
        --a. Общее количество просроченных дней не должно превышать 30 дней. Просрочки более 30 дней рассматриваются и утверждаются Уполномоченным лицом;
        res_error_text = 'Sonuncu bağlanmış fərdi kredit müqaviləsi üzrə ümumi gecikmə günlərin sayı 30 gündən çox olmamalıdır.';
        res_error_is_critical = 0;
        if (sum_delinq_days > 30) then
          res_error_is_critical = 1;
        /*"если  у клиента/поручителя по ПОСЛЕДНЕМУ  НЕАКТИВНОМУ кредиту была просрочка до 30 дней утверждает Старший КС
        31-60 дней - РМ
        61 и более дней Эрнест и Алтын" */
        if (sum_delinq_days between 31 and 60) then
        begin
          if (is_dr = 1) then
            res_error_is_critical = 0;
          else
            res_error_text = 'Əvvəlki dövr kredit üzrə ümumi gecikmə günlərin sayı 60 günə qədərdir. Müqaviləni regional menecer təsdiqləyə bilər.';
        end
        suspend;
      end
    end
    --1. Требование к клиенту: b.  Возрастная категория - от 21 до 76 лет (включительно).
    if (((kf_disb_date < dateadd(year, 21, c_birth_date)) or ( kf_disb_date > dateadd(year, 77,c_birth_date)))) then
    begin
      --res_error_text = 'Возрастная категория ' || C_AGE_TEXT || ' лет (включительно).';
      res_error_text = 'Müştərinin yaşı 21-76 arasında olmalıdır.';
      res_error_is_critical = 2;
      suspend;
    end
    if (kc_circle = 1 and
        kf_disb_date > dateadd(year, 76, c_birth_date)) then
    begin
      --res_error_text = 'Возрастная категория ' || C_AGE_TEXT || ' лет (включительно).';
      res_error_text = '1 dövr müştərinin yaşı 21-75 arasında olmalıdır.';
      res_error_is_critical = 2;
      suspend;
    end
    /*c.  Во время повторного обращения за кредитом, верхний предел возрастной
        категории для клиентов, которые при предыдущем кредите не попадали в
        просрочку или просрочка возникала не по их вине, поднимается до 76 лет (включительно);*/
    if (kc_circle > 1 and
        coalesce(max_delinq_days, 0) > 0 and
        kf_disb_date > dateadd(year, 77, c_birth_date)) then
    begin
      --res_error_text = 'Возрастная категория ' || C_AGE_TEXT || ' лет (включительно).';
      res_error_text = 'Təkrar dövrdə müştərinin maksimal yaşı 76 ola bilər.';
      res_error_is_critical = 2;
      suspend;
    end
    ----------------------------------------------------------------------------
    --1.4.    Документы, запрашиваемые от клиента
    --От клиента требуются следующие документы:
    --a. Документ, удостоверяющий личность: действующий паспорт;
    --b. Разрешение клиента и поручителя на получение дополнительной личной информации о нем из других источников (АК-01);
    --c. Обращение по получению кредита (АК-02);
    --d. Если будут приняты во внимание официально подтверждённые  доходы супруги/супруга, то у нее/него берется письменное согласие на  получение личной информации.
    --e. ИНН (в зависимости от продукта)
    --f. документы, подтверждающие доход (в зависимости от продукта)
    if (coalesce(kf_client_pass_no, '') = '') then
    begin
      res_error_text = 'Şəxsiyyət vəsiqəsi üzrə məlumatları doldurun'; --'Прикрепите действующий паспорт.';
      res_error_is_critical = 1;
      suspend;
    end
    --Наличие добавленных файлов во вкладке Документы у клиента и (если есть) поручителя: "Кредитная история от МКР",  "Согласие на проверку МКР". Очень критично.
    if (not exists(select *
                   from clientdocs cdc, passkind, passdoctype
                   where cds_from_c_id = :kc_from_c_id and
                         cds_from_pk_id = pk_id and
                         pk_from_pdt_id = pdt_id and
                         pdt_code in ('credit_history', 'mkr_agreement') and
                         cds_is_delete = 0)) then
    begin
      res_error_text = 'Müştəri məlumatlarında"Кредитная история от МКР",  "Согласие на проверку МКР" faylların əlavə olunması zəruridir';
      res_error_is_critical = 2;
      suspend;
    end
    --Для потребительских кредитов Документы у клиента обязательно наличие файла "Справка с места работы". Очень критично.
    if (pr_code containing '.AILE.' and
        not exists(select *
                   from clientdocs cdc, passkind, passdoctype
                   where cds_from_c_id = :kc_from_c_id and
                         cds_from_pk_id = pk_id and
                         pk_from_pdt_id = pdt_id and
                         pdt_code in ('job_certificate') and
                         cds_is_delete = 0)) then
    begin
      res_error_text = 'İstehlak kreditləri üçün müştərinin məlumatlarında "Справка с места работы" faylın əlavə olunması zəruridir.';
      res_error_is_critical = 2;
      suspend;
    end
    --Обязательно заполненное поле "ИНН" в личной инф. клиента. Очень критично.
    if (pr_code containing '.Micro_Biznes.' and
        coalesce(c_inn, '') = '') then
    begin
      res_error_text = '"Mikro_biznes" məhsulu üzrə müştərilərin məlumatlarında "ИНН" bəndin doldurulması zəruridir.';
      res_error_is_critical = 2;
      suspend;
    end
    -- проверяю договора где клиент выступает поручителем >>>>>>>>>>>>>>>>>>>>
    for select distinct z.z_from_k_id
        from zalog z
        inner join kontstate ks on (z_from_k_id = ks.ks_from_k_id and
              :kf_disb_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1)
        where z_from_c_id = :kc_from_c_id and
              z_type = 0 and
              z.z_from_k_id <> :k_id
        into :poruch_k_id
    do
    begin
      max_delinq_days = null;
      sum_delinq_days = null;
      for select oks_delinq_date, --
                 max(iif(oks_date_end < current_date, oks_date_end + 1, current_date)) ol_date
          from oksummary
          where oks_close_state = 0 and
                oks_from_k_id = :poruch_k_id and
                oks_delinq_date <= oks_date_end and
                oks_delinq_date <= :kf_disb_date
          group by oks_delinq_date
          into k_old_delinq_date, k_old_ol_date
      do
      begin
        k_ol_del_day_tmp = coalesce(k_old_ol_date - k_old_delinq_date, 0);
        if (coalesce(max_delinq_days, 0) < k_ol_del_day_tmp) then
          max_delinq_days = k_ol_del_day_tmp;
        sum_delinq_days = coalesce(sum_delinq_days, 0) + k_ol_del_day_tmp;
      end
      max_delinq_days = coalesce(max_delinq_days, 0);
      sum_delinq_days = coalesce(sum_delinq_days, 0);
      --По активному и по закрытому кредитам, где клиент указан как поручитель не должно быть просрочек.
      if (max_delinq_days > 0) then
      begin
        res_error_text = 'Zamin qismində çıxış edən aktiv və bağlı kreditlər üzrə gecikmə olmamalıdır.';
        res_error_is_critical = 1;
        /*"если  у клиента/поручителя по ПОСЛЕДНЕМУ  НЕАКТИВНОМУ кредиту была просрочка до 30 дней утверждает Старший КС
        31-60 дней - РМ
        61 и более дней Эрнест и Алтын" */
        if (sum_delinq_days between 31 and 60) then
        begin
          if (is_dr = 1) then
            res_error_is_critical = 0;
          else
            res_error_text = 'Əvvəlki dövr kredit üzrə ümumi gecikmə günlərin sayı 60 günə qədərdir. Müqaviləni regional menecer təsdiqləyə bilər.';
        end
        suspend;
      end
    end
    --Родственники сотрудников, указанные во вкладке Связи как "Ata","Ana", "Həyat yoldaşı (kişi)", "Həyat yoldaşı (qadın)", "Oğul", "Qızı" не могут брать кредит.
    /*
    if (exists(select *
               from hr_pstatus
               inner join hp_client_relation_list(ps_from_c_id, 1) rl on (rl.crel_other_c_id = :kc_from_c_id and
                     crel_from_rel_id in (1, 2, 3, 4, 5, 6))
               where ps_is_working = 1 and
                     :kf_disb_date between ps_date_begin and ps_date_end)) then
    begin
      --Родственники сотрудников, указанные во вкладке Связи как "Ata","Ana", "Həyat yoldaşı (kişi)", "Həyat yoldaşı (qadın)", "Oğul", "Qızı" не могут брать кредит.
      res_error_text = 'Şirkətin əməkdaşlarının "Ata","Ana", "Həyat yoldaşı (kişi)", "Həyat yoldaşı (qadın)", "Oğul", "Qızı" kimi qeydə alınan qohumları kredit götürə bilməzlər.';
      res_error_is_critical = 1;
      suspend;
    end
    */
    -- скрипт выше был заменен на ниже, то что выше было более 28 тыс. обращение к таблице CLIENTRELATION, после изменения не более 50 раз.
    if (exists(select * from hp_client_relation_list(:kc_from_c_id, 1)
        where crel_from_rel_id <> ''
        and crel_from_rel_id in (1, 2, 3, 4, 5, 6)
        and crel_other_c_id is not null
        and exists(select * from hr_pstatus
            where ps_from_c_id = crel_other_c_id
            and :kf_disb_date between ps_date_begin and ps_date_end
            and ps_is_working = 1
            and ps_status = 1))) then
    begin
      --Родственники сотрудников, указанные во вкладке Связи как "Ata","Ana", "Həyat yoldaşı (kişi)", "Həyat yoldaşı (qadın)", "Oğul", "Qızı" не могут брать кредит.
      res_error_text = 'Şirkətin əməkdaşlarının "Ata","Ana", "Həyat yoldaşı (kişi)", "Həyat yoldaşı (qadın)", "Oğul", "Qızı" kimi qeydə alınan qohumları kredit götürə bilməzlər.';
      res_error_is_critical = 1;
      suspend;
    end
    /*5.   В личной инф. клиента должно быть минимум 6 контактных лиц. Очень критично.
        */
    select sum(iif(ct.cc_create_date < :kf_disb_date - 14, 1, 0)) old_cont,
           sum(iif(ct.cc_create_date < :kf_disb_date - 14, 0, 1)) new_cont
    from contacts ct
    where ct.cc_from_c_id = :kc_from_c_id
    into old_cont, new_cont;
    old_cont = coalesce(old_cont, 0);
    new_cont = coalesce(new_cont, 0);
    if (new_cont < 1 and
        old_cont > 5) then
    begin
      res_error_text = 'Yeni əlaqələr mövcud deyil';
      res_error_is_critical = 0;
      suspend;
    end
    if (new_cont + old_cont < 6) then
    begin
      res_error_text = 'Müştərinin məlumatında en azı 6 nəfər əlaqə üçün şəxs qeyd olunmalıdır';
      res_error_is_critical = 2;
      suspend;
    end
    --контактныe лицa<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<
  end
  --Залоги
  for select z.z_from_c_id, sum(coalesce(z.z_cost_market, 0))
      from zalog z
      where z.z_from_k_id = :k_id and
            z.z_type = 1
      group by 1
      into z_from_c_id, z_cost_market
  do
  begin
    if (pr_code not containing '.ZALOG.') then
    begin
      --Не должны регистрироваться залоги (касается продуктов где не предусмотрен залог). Очень критично.
      res_error_text = 'Carı məhsul üzrə girov qeyd oluna bilməz.';
      res_error_is_critical = 2;
      suspend;
    end
    else
    begin
      res_error_text = ''; --vajno
      z_cost_market_all = coalesce(z_cost_market_all, 0) + z_cost_market;
      if (coalesce(z_from_c_id, 0) = 0) then
      begin
        --res_error_text = 'Неполные данные залогодателя';
        res_error_text = 'Zamin üzrə məlumatlar tam qeyd olunmalıdır. Məlumatları tamamlayın.';
        res_error_is_critical = 1;
        suspend;
      end
      else
      begin
        res_info_text = '';
        c_birth_date = null; --Дата рождения
        cizizen_name = '';
        c_verify_status = null;
        select c_name_fio, c_birth_date, cl.c_pass_no, --
               --refdata
               rd.rd_name
        from clients cl
        inner join refdata rd on (cl.c_citizen = rd.rd_id)
        where cl.c_id = :z_from_c_id
        into res_info_text, c_birth_date, kf_client_pass_no, cizizen_name; --Страна;
        c_birth_date = coalesce(c_birth_date, current_date);
        --Гражданство Залогодателя должно быть "Азербайджан", возраст минимально 21 год. Очень критично.
        if (coalesce(cizizen_name, '') <> 'Азербайджан') then
        begin
          res_error_text = 'Girovqoyan şəxsin Azərbaycan Respublikasının vətəndaşı olmalıdır.';
          res_error_is_critical = 2;
          suspend;
        end
        if (kf_disb_date < dateadd(year, 21, c_birth_date)) then
        begin
          res_error_text = 'Girovqoyan şəxsin minimal olaraq 21 yaşı olmalıdır.  Указан: ' || cizizen_name;
          res_error_is_critical = 2;
          suspend;
        end
        --1.5. Документы, запрашиваемые от поручителя
        --a. Документ, удостоверяющий личность: действующий паспорт;
        if (coalesce(kf_client_pass_no, '') = '') then
        begin
          res_error_text = 'Şəxsiyyət vəsiqəsi üzrə məlumatları doldurun'; --'Прикрепите действующий паспорт.';
          res_error_is_critical = 1;
          suspend;
        end
      end
    end
  end
  --Сумма рыночной стоимости всех залогов не должна быть меньше 50 % от суммы кредита. Очень критично.
  if (pr_code containing '.ZALOG.' and
      coalesce(z_cost_market_all, 0) < kc_amount / 2) then
  begin
    res_error_text = 'Girovların ümumi bazar dəyəri kredit məbləğinin 50 %-dən az ola bilməz.';
    res_error_is_critical = 2;
    suspend;
  end
  --Поручители
  res_error_text = ''; --vajno
  for select z.z_from_c_id
      from zalog z
      where z.z_from_k_id = :k_id and
            z.z_type = 0
      into z_from_c_id
  do
  begin
    if (:z_from_c_id = 1010100286777) then
        exit; -- RFCAZ-511

    if (pr_code not containing '.PORUCH.') then
    begin
      --Не должны регистрироваться поручители (касается продуктов где не предусмотрены поручители). Очень критично.
      res_error_text = 'Carı məhsul üzrə zamin qeyd oluna bilməz.';
      res_error_is_critical = 2;
      suspend;
    end
    else
    begin
      if (coalesce(z_from_c_id, 0) = 0) then
      begin
        --res_error_text = 'Неполные данные поручителя';
        res_error_text = 'Zamin üzrə məlumatlar tam qeyd olunmalıdır. Məlumatları tamamlayın.';
        res_error_is_critical = 1;
        suspend;
      end
      else
      begin
        --Наличие добавленных файлов во вкладке Документы у клиента и (если есть) поручителя: "Кредитная история от МКР",  "Согласие на проверку МКР". Очень критично.
        if (not exists(select *
                       from clientdocs cdc, passkind, passdoctype
                       where cds_from_c_id = :z_from_c_id and
                             cds_from_pk_id = pk_id and
                             pk_from_pdt_id = pdt_id and
                             pdt_code in ('credit_history', 'mkr_agreement') and
                             cds_is_delete = 0)) then
        begin
          res_error_text = 'Zamin məlumatlarında"Кредитная история от МКР",  "Согласие на проверку МКР" faylların əlavə olunması zəruridir';
          res_error_is_critical = 2;
          suspend;
        end
        res_info_text = '';
        c_birth_date = null; --Дата рождения
        cizizen_name = '';
        kf_client_pass_no = '';
        c_verify_status = null;
        select c_name_fio, c_birth_date, cl.c_pass_no, cl.c_verify_status, --
               --refdata
               rd.rd_name
        from clients cl
        inner join refdata rd on (cl.c_citizen = rd.rd_id)
        where cl.c_id = :z_from_c_id
        into res_info_text, c_birth_date, kf_client_pass_no, c_verify_status, cizizen_name; --Страна;
        c_birth_date = coalesce(c_birth_date, current_date);
        c_verify_status = coalesce(c_verify_status, 0);
        if (c_verify_status <> 2) then
        begin
          --Текст: Статус верификации поручителя должен быть “Успешная верификация”
          res_error_text = 'Zaminin verifikasiya statusu “Успешная верификация” olmalıdır.';
          res_error_is_critical = 2;
          suspend;
        end
        if (coalesce(cizizen_name, '') <> 'Азербайджан') then
        begin
          --res_error_text = 'Поручителем может быть гражданин Азербайджанской Республики. Указан: ' || CIZIZEN_NAME;
          res_error_text = 'Zamin üçün tələblər. Zamin yalnız Azərbaycan Respublikasının vətəndaşı ola bilər. Указан: ' || cizizen_name;
          res_error_is_critical = 1;
          suspend;
        end
        --1. Требование к поручителю: b.  Возрастная категория - от 21 до 70 лет (включительно).
        if (kf_disb_date < dateadd(year, 21, c_birth_date)) then
        begin
          --res_error_text = 'Возрастная категория ' || C_AGE_TEXT || ' лет (включительно).';
          res_error_text = 'Zaminin yaşı 21-70 arasında olmalıdır.';
          res_error_is_critical = 2;
          suspend;
        end
        if (kf_disb_date > dateadd(year, 76, c_birth_date)) then
        begin
          --res_error_text = 'Возрастная категория ' || C_AGE_TEXT || ' лет (включительно).';
          res_error_text = 'Zaminin maksimal yaşı 75 ola bilər.';
          res_error_is_critical = 2;
          suspend;
        end
        --1.5. Документы, запрашиваемые от поручителя
        --a. Документ, удостоверяющий личность: действующий паспорт;
        if (coalesce(kf_client_pass_no, '') = '') then
        begin
          res_error_text = 'Şəxsiyyət vəsiqəsi üzrə məlumatları doldurun'; --'Прикрепите действующий паспорт.';
          res_error_is_critical = 1;
          suspend;
        end
      end
      ---- проверяю договора где поручител выступает клиентoм >>>>>>>>>>>>>>>>>>>>
      for select distinct ks_from_k_id
          from knclient
          inner join kontstate ks on (kc_from_k_id = ks_from_k_id and
                :kf_disb_date > ks_date_begin and
                ks.ks_ol_is_delete = 0 and
                ks.ks_state = 1)
          where kc_from_c_id = :z_from_c_id and
                kc_amount_approved > 0
          order by ks_date_begin desc
          rows 1
          into :poruch_k_id
      do
      begin
        max_delinq_days = null;
        sum_delinq_days = null;
        k_old_delinq_date = null;
        k_old_ol_date = null;
        for select oks_delinq_date, --
                   max(iif(oks_date_end < current_date, oks_date_end + 1, current_date)) ol_date
            from oksummary
            where oks_close_state = 0 and
                  oks_from_k_id = :poruch_k_id and
                  oks_delinq_date <= oks_date_end and
                  oks_delinq_date <= :kf_disb_date
            group by oks_delinq_date
            into k_old_delinq_date, k_old_ol_date
        do
        begin
          k_ol_del_day_tmp = coalesce(k_old_ol_date - k_old_delinq_date, 0);
          if (coalesce(max_delinq_days, 0) < k_ol_del_day_tmp) then
            max_delinq_days = k_ol_del_day_tmp;
          sum_delinq_days = coalesce(sum_delinq_days, 0) + k_ol_del_day_tmp;
        end
        max_delinq_days = coalesce(max_delinq_days, 0);
        sum_delinq_days = coalesce(sum_delinq_days, 0);
        --По активному и по закрытому кредитам, где клиент указан как поручитель не должно быть просрочек.
        if (max_delinq_days > 0) then
        begin
          res_error_text = 'Zamin qismində çıxış edənin aktiv və bağlı kreditlər üzrə gecikməsi olmamalıdır.';
          res_error_is_critical = 0;
          if (sum_delinq_days >= 31) then
            res_error_is_critical = 1;
          /*"если  у клиента/поручителя по ПОСЛЕДНЕМУ  НЕАКТИВНОМУ кредиту была просрочка до 30 дней утверждает Старший КС
        31-60 дней - РМ
        61 и более дней Эрнест и Алтын" */
          if (sum_delinq_days between 31 and 60) then
          begin
            if (is_dr = 1) then
              res_error_is_critical = 0;
            else
              res_error_text = 'Əvvəlki dövr kredit üzrə ümumi gecikmə günlərin sayı 60 günə qədərdir. Müqaviləni regional menecer təsdiqləyə bilər.';
          end
          suspend;
        end
      end
    end
  end
  if (pr_code containing '.PORUCH.' and
      coalesce(z_from_c_id, 0) = 0 and
      res_error_text = '') then
  begin
    --res_error_text = 'Добавьте поручителя';
    res_error_text = 'Carı məhsul üzrə yığılan kredit müqaviləsində ən azı 1 zaminin qeyd olunmalıdır.';
    res_error_is_critical = 1;
    suspend;
  end
end
