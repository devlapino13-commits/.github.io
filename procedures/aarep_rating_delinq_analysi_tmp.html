CREATE PROCEDURE AAREP_RATING_DELINQ_ANALYSI_TMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    CONSTANTA_DATE DATE)
RETURNS (
    KS_FROM_K_ID ID_TYPE,
    REP_DELINQ_DAYS INTEGER,
    AMOUNT_ACTIVE_BEGIN CURR_TYPE,
    AMOUNT_ACTIVE_END CURR_TYPE,
    OS_REPAY CURR_TYPE,
    OS_WOFF CURR_TYPE)
AS
declare variable ol_date date;
declare variable ok_type integer;
declare variable ok_os curr_type;
begin
  for select ks_from_k_id
      from kontstate
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :constanta_date between ks_date_begin and ks_date_end
      into ks_from_k_id
  do
  begin
    select coalesce(rep_delinq_days, 0)
    from aarep_kont_delinq_status(:ks_from_k_id, :constanta_date)
    into rep_delinq_days;

    if (rep_delinq_days > 7) then
    begin
      ok_os = 0;
      amount_active_begin = 0;
      amount_active_end = 0;
      os_repay = 0;
      os_woff = 0;

      for select ok_ol_date, ok_type, sum(ok_os)
          from operkred
          where ok_ol_is_active = 1 and
                ok_ol_date <= :rep_date_end and
                ok_from_k_id = :ks_from_k_id and
                ok_type in (4, 5, 8) and
                ok_from_c_id = 0
          group by ok_ol_date, ok_type, ok_os
          into ol_date, ok_type, ok_os
      do
      begin
        if (ok_os is null) then
          ok_os = 0;
        if (:ol_date < :rep_date_begin) then
          amount_active_begin = amount_active_begin + ok_os;
        if (:ol_date between :rep_date_begin and :rep_date_end) then
        begin
          if (:ok_type = 4) then
            os_repay = os_repay - ok_os;
          if (:ok_type = 5) then
            os_woff = os_woff - ok_os;
        end
      end
      if (amount_active_begin is null) then
        amount_active_begin = 0;
      amount_active_end = :amount_active_begin - (:os_repay + :os_woff);
      if (amount_active_begin <> 0) then
        suspend;
    end
  end

end
