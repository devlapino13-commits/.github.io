CREATE PROCEDURE CM_KG_0131_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable k_from_kf_id id_type;
declare variable c_id id_type;
declare variable last_par_k_id id_type;
declare variable last_k_id id_type;
declare variable last_k_id_1 id_type;
declare variable last_k_id_2 id_type;
declare variable last_k_id_3 id_type;
declare variable last_k_id_4 id_type;
declare variable last_k_kontract_no varchar(50);
declare variable last_pr_name varchar(50);
declare variable pr_code varchar(100);
declare variable last_pr_code varchar(100);
declare variable k_kontract_no varchar(100);
declare variable last_ks_date_end date;
declare variable all_delinq_days integer;
declare variable curr_delinq_days integer;
declare variable max_delinq_days integer;
declare variable k_kolich_client_approved integer;
declare variable kol_cl integer;
declare variable kol_cl_2 integer;
declare variable kol_cl_3 integer;
declare variable kol_cl_4 integer;
declare variable k_refund integer;
begin
if (k_id = 109010006023701301) then exit;
if (k_id = 109010070876500501) then exit;
if (k_id = 109010040915800401) then exit;
if (k_id = 109010100397400301) then exit;
if (k_id = 109010028959300601) then exit;
if (k_id = 109010025209900601) then exit;
if (k_id = 109010119696200201) then exit;
if (k_id = 109010027455800601) then exit;
if (k_id = 109010046229500501) then exit;
if (k_id = 109010119840200201) then exit;
  res_error_is_critical = 1;
  for select k.k_disb_date, pr.pr_code
  from kontract k
  inner join product pr on k_from_pr_id = pr_id and
        pr_code not containing '.par.' and
        pr_code not containing '.express.' and
        pr_code not containing '.ipf.' and
        ((pr_code containing '.gr.') or (pr_code containing '.ind.')) and
        k.k_disb_date >= '14.02.2021'
  where k_id = :k_id
  into :kf_disb_date, :pr_code
  do
    for select kc_from_c_id
      from knclient kc
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into c_id
    do
    begin
      if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
        exit;

        if (exists (
            select *
            from kontstate ks
            inner join kontract kont on kont.k_id = ks.ks_from_k_id
            inner join product pr on pr.pr_id = kont.k_from_pr_id
            inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
            where ks.ks_state = 1
            and :kf_disb_date between ks.ks_date_begin and ks.ks_date_end
            and ks.ks_ol_is_delete = 0
            and pr_code not containing '.par.'
            and pr_code not containing '.express.'
            and pr_code not containing '.ipf.'
            and ((pr_code containing '.gr.') or (pr_code containing '.ind.'))
            and kncl.kc_from_c_id = :c_id
            and kncl.kc_amount_approved > 0)) then
        begin
            select c_name_fio
            from clients
            where c_id = :c_id
            into :res_info_text;
            res_error_text = 'У заемщика имеется активный Основной кредит в Компании';
            suspend;
        end
    end
end
