CREATE PROCEDURE AAREP_RESTRUKT_CLIENTS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    OL_ID DECIMAL(18,0),
    REG_ID INTEGER,
    OL_DATE DATE,
    OL_MEMO VARCHAR(1024),
    KF_NAME VARCHAR(100),
    REG_NAME VARCHAR(50),
    PARENT_ID INTEGER,
    PARENT_NAME VARCHAR(50),
    K_KREDLINE_END DATE,
    K_DATE_END DATE,
    KOL_PROLOG INTEGER,
    REP_ACTIVE_BALANCE DECIMAL(18,2),
    RT_KIND VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    PR_NAME VARCHAR(50),
    PR_COMMON_NAME VARCHAR(100),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    KC_FROM_C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    BT_NAME VARCHAR(50),
    BS_NAME VARCHAR(50),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    REG_STAT_REP_REGION VARCHAR(255),
    K_DISB_KINDD D_STR_100,
    K_BY_RATING D_STR_64)
AS
declare variable G integer;
declare variable CURR varchar(3);
declare variable DEF_CURR varchar(3);
declare variable CURR_RATE decimal(18,4);
declare variable K_INDEX_CURR_RATE decimal(18,4);
declare variable PR_SET_RATE_IN_KONT integer;
declare variable REG_PRIOR integer;
declare variable ST_STAFF integer;
declare variable ST_CLIENT integer;
declare variable CM_IS_CLIENT_MODE integer;
declare variable K_ID_PRIOR ID_TYPE;
declare variable KC_BUSUNESS_SUBTYPE_PRIOR smallint = 0;
declare variable KC_BUSUNESS_SUBTYPE smallint;
declare variable GR_BALANCE CURR_TYPE;
begin
-- operkred_fix

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select cm_default_currency, cm_is_client_mode
  from company
  into :def_curr, cm_is_client_mode;
  kc_busuness_subtype_prior = -1;
  if (cm_is_client_mode = 0) then
  begin
    for select k_id, ol_date, ol_memo, kf_name, reg_name, parent_name, k_kredline_end, k_date_end, kol_prolog,
               balance_rest, rt_kind, k_kontract_no, pr_name, pr_common_name, k_kolich_client_approved,
               k_amount_approved, reg_stat_rep_region, k_disb_kindd, k_by_rating
        from aarep_restrukt(:rep_region_id, :rep_date_begin, :rep_date_end, 'Restructuring')
        into :k_id, :ol_date, :ol_memo, :kf_name, :reg_name, :parent_name, :k_kredline_end, :k_date_end, :kol_prolog,
             :gr_balance, :rt_kind, :k_kontract_no, :pr_name, :pr_common_name, :k_kolich_client_approved,
             :k_amount_approved, :reg_stat_rep_region, :k_disb_kindd, :k_by_rating
    do
    begin
      for select kc_amount_approved, c_name_fio, kc_busuness_subtype
          from knclient, clients
          where kc_from_k_id = :k_id and
                kc_amount_approved > 0 and
                kc_from_c_id = c_id
          into kc_amount_approved, c_name_fio, kc_busuness_subtype
      do
      begin
        if (kc_busuness_subtype_prior <> kc_busuness_subtype) then
        begin
          kc_busuness_subtype_prior = kc_busuness_subtype;
          select bt_name, bs_name
          from bustype, bussubtype
          where bs_id = :kc_busuness_subtype and
                bs_from_bt_id = bt_id
          into bt_name, bs_name;
        end
        rep_active_balance = cast(kc_amount_approved as double precision) / k_amount_approved * gr_balance;
        suspend;
      end
    end
--    Exit;
  end
end
