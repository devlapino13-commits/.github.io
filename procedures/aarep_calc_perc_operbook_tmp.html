CREATE PROCEDURE AAREP_CALC_PERC_OPERBOOK_TMP(
    REPORT_REGION_ID D_INTEGER,
    REPORT_DATE_BEGIN D_DATE,
    REPORT_DATE_END D_DATE,
    REPORT_NEED_PERC D_INTEGER)
RETURNS (
    AAREP_BALANCE_BEGIN D_CURRENCY,
    AAREP_PERC_BEGIN D_CURRENCY,
    AAREP_PERC_CALCED D_CURRENCY,
    AAREP_PERC_PAYED D_CURRENCY,
    AAREP_PERC_WRITEOFF D_CURRENCY,
    AAREP_PERC_END D_CURRENCY,
    AAREP_BALANCE_END D_CURRENCY,
    AAREP_DISB_AMOUNT D_CURRENCY,
    AAREP_CALCED_ALL D_CURRENCY,
    AAREP_PAYED_ALL D_CURRENCY,
    AAREP_WRITEOFF_ALL D_CURRENCY,
    AAREP_OFFICE D_STR_100,
    AAREP_PRODUCT_ID D_INTEGER,
    AAREP_FOND D_STR_20,
    AAREP_OFFICE_ID D_INTEGER,
    AAREP_ACCOUNT_PROFIT D_STR_20,
    AAREP_ACCOUNT_CALCED D_STR_20,
    AAREP_ACCOUNT_REPAYED D_STR_20,
    AAREP_ACCOUNT_PORTFOLIO D_STR_20,
    AAREP_OUT_PERC_BEGIN D_CURRENCY,
    AAREP_OUT_PERC_CALCED D_CURRENCY,
    AAREP_OUT_PERC_END D_CURRENCY)
AS
declare variable k_stop_perc d_date;
declare variable aarep_balance_begin1 d_currency;
declare variable aarep_perc_begin1 d_currency;
declare variable prior_office d_integer;
declare variable prior_fond d_str_20;
declare variable aarep_k_id d_id;
declare variable prior_product_id d_integer;
declare variable aarep_office_begin d_str_100;
declare variable aarep_office_id_begin d_integer;
declare variable kf_id d_id;
declare variable db_id d_integer;
declare variable k_kredline_end d_date;
declare variable k_zero_balance_date d_date;
declare variable rep_period d_integer;
declare variable rep_period_prior d_integer;
declare variable tmp1 d_currency = 0;
declare variable tmp2 d_currency = 0;
declare variable k_id d_id;
declare variable aarep_out_perc_begin1 d_currency = 0;
declare variable aarep_office_id_tmp d_integer;
declare variable aarep_office_tmp d_str_100;
declare variable aarep_balance_end_tmp d_currency;
begin
  -- Временная процедура по начисленным процентам, офисы на начало
  -- и конец периода могут отличаться, если папка передана
  db_id = f_get_db_id();

  aarep_balance_begin = 0;
  aarep_perc_begin = 0;
  aarep_perc_calced = 0;
  aarep_perc_payed = 0;
  aarep_perc_writeoff = 0;
  aarep_perc_end = 0;
  aarep_balance_end = 0;
  aarep_calced_all = 0;
  aarep_payed_all = 0;
  aarep_writeoff_all = 0;
  aarep_disb_amount = 0;
  aarep_out_perc_begin = 0;
  aarep_out_perc_calced = 0;
  aarep_out_perc_end = 0;
  prior_office = -1;
  prior_fond = -1;
  prior_product_id = 0;
  aarep_account_profit = '';
  aarep_account_calced = '';
  aarep_account_repayed = '';

  rep_period = extract(year from :report_date_end) * 100 + extract(month from :report_date_end);
  rep_period_prior = extract(year from dateadd(month, -1, :report_date_end)) * 100 + extract(month from dateadd(month, -1, :report_date_end));

  -- Список офисов и продуктов, по которым были движения по начислению %%
  for
    select distinct
      k_from_f_id,
      reg_code || ' ' || reg_name,
      reg_unicode,
      k_from_pr_id
    from kontcalcperc
      inner join kontract
        on k_id = kcp_from_k_id
      inner join region_list(:report_region_id, 1, -1, 1)
        on id_list = kcp_from_reg_id
      inner join region
        on reg_unicode = id_list
    where kcp_period = :rep_period
    union
    select distinct
      k_from_f_id,
      reg_code || ' ' || reg_name,
      reg_unicode,
      k_from_pr_id
    from kontcalcperc
      inner join kontract
        on k_id = kcp_from_k_id
      inner join region_list(:report_region_id, 1, -1, 1)
        on id_list = kcp_from_reg_id
      inner join region
        on reg_unicode = kcp_transfer_reg_id
    where kcp_period = :rep_period
      and kcp_from_reg_id <> kcp_transfer_reg_id
    order by 3, 4, 1
  into
    :aarep_fond,
    :aarep_office,
    :aarep_office_id,
    :aarep_product_id
  do
  begin
    if (prior_office <> aarep_office_id or prior_fond <> aarep_fond or prior_product_id <> aarep_product_id) then
    begin
      select res_account
      from hp_make_kredit_acclist(:aarep_product_id, :aarep_office_id, 1) -- Кредитный портфель
      into :aarep_account_portfolio;

      select res_account
      from hp_make_kredit_acclist(:aarep_product_id, :aarep_office_id, 2) -- Начисленные проценты
      into :aarep_account_calced;

      select res_account
      from hp_make_kredit_acclist(:aarep_product_id, :aarep_office_id, 3) -- Выплаченные проценты
      into :aarep_account_repayed;

      select res_account
      from hp_make_kredit_acclist(:aarep_product_id, :aarep_office_id, 36) -- Процентный доход
      into :aarep_account_profit;

      suspend;
      prior_office = aarep_office_id;
      prior_fond = aarep_fond;
      prior_product_id = aarep_product_id;
    end
  end

  prior_office = -1;
  prior_fond = -1;
  prior_product_id = 0;
  aarep_account_profit = '';
  aarep_account_calced = '';
  aarep_account_repayed = '';
  aarep_account_portfolio = '';

  if (report_need_perc = 1) then
  begin
    -- Если нужно показать начисленные проценты
    for
      select
        kcp_from_k_id,
        k_from_f_id,
        k_stop_perc,
        creg.reg_code || ' ' || creg.reg_name,
        creg.reg_unicode,
        k_from_pr_id,
        k_from_kf_id,
        k_kredline_end,
        k_zero_balance_date,
        kcp_transfer_reg_id,
        lastreg.reg_code || ' ' || lastreg.reg_name
      from kontcalcperc
        inner join kontract
          on k_id = kcp_from_k_id
        inner join region_list(:report_region_id, 1, -1, 1)
          on id_list = kcp_from_reg_id
        inner join region creg
          on creg.reg_unicode = id_list
        inner join region lastreg
          on lastreg.reg_unicode = kcp_transfer_reg_id
      where kcp_period = :rep_period
    into
      :aarep_k_id,
      :aarep_fond,
      :k_stop_perc,
      :aarep_office,
      :aarep_office_id,
      :aarep_product_id,
      :kf_id,
      :k_kredline_end,
      :k_zero_balance_date,
      :aarep_office_id_begin,
      :aarep_office_begin
    do
    begin
      aarep_balance_begin = 0;
      aarep_perc_begin = 0;
      aarep_perc_calced = 0;
      aarep_perc_payed = 0;
      aarep_perc_writeoff = 0;
      aarep_perc_end = 0;
      aarep_balance_end = 0;
      aarep_calced_all = 0;
      aarep_payed_all = 0;
      aarep_writeoff_all = 0;
      aarep_disb_amount = 0;
      aarep_out_perc_begin = 0;
      aarep_out_perc_calced = 0;
      aarep_out_perc_end = 0;

      -- Для амнистированных групп, добавлно от 14.01.2014
      k_id = aarep_k_id;
      for
        select
          kd_k_new
        from kontdevides
          where kd_k_orig = :k_id
            and kd_date > :report_date_end
      into
        :aarep_k_id
      do
      begin
        if (not exists(
          select *
          from kontcalcperc
          where kcp_period = :rep_period
            and kcp_from_k_id = :aarep_k_id)) then
        begin
          aarep_balance_begin = null;
          select
            oks_balance
          from oksummary
          where oks_from_k_id = :aarep_k_id
            and oks_date_begin <= (:report_date_begin - 1)
            and oks_date_end >= (:report_date_begin - 1)
            and oks_close_state = 0
          into
            :aarep_balance_begin;
          aarep_balance_begin = coalesce(aarep_balance_begin, 0);

          aarep_balance_end = null;
          select
            oks_balance
          from oksummary
          where oks_from_k_id = :aarep_k_id
            and :report_date_end between oks_date_begin and oks_date_end
            and oks_close_state = 0
          into
            :aarep_balance_end;
          aarep_balance_end = coalesce(aarep_balance_end, 0);

          if (aarep_office_id_begin = aarep_office_id) then
          begin
            suspend;
          end else
          begin
            aarep_office_id_tmp = aarep_office_id;
            aarep_office_tmp = aarep_office;
            aarep_office_id = aarep_office_id_begin;
            aarep_office = aarep_office_begin;
            aarep_balance_end_tmp = aarep_balance_end;
            aarep_balance_end = 0;
            suspend;

            aarep_office_id = aarep_office_id_tmp;
            aarep_office = aarep_office_tmp;
            aarep_balance_begin = 0;
            aarep_balance_end = aarep_balance_end_tmp;
            suspend;
          end
        end
      end

      aarep_k_id = k_id;
      aarep_balance_begin = 0;
      aarep_perc_begin = 0;
      aarep_perc_calced = 0;
      aarep_perc_payed = 0;
      aarep_perc_writeoff = 0;
      aarep_perc_end = 0;
      aarep_balance_end = 0;
      aarep_calced_all = 0;
      aarep_payed_all = 0;
      aarep_writeoff_all = 0;
      aarep_disb_amount = 0;
      aarep_out_perc_begin = 0;
      aarep_out_perc_calced = 0;
      aarep_out_perc_end = 0;

      select
        oks_balance
      from oksummary
      where oks_from_k_id = :aarep_k_id
        and oks_date_begin <= (:report_date_begin - 1)
        and oks_date_end >= (:report_date_begin - 1)
        and oks_close_state = 0
      into
        :aarep_balance_begin;

      select
        oks_balance
      from oksummary
      where oks_from_k_id = :aarep_k_id
        and :report_date_end between oks_date_begin and oks_date_end
        and oks_close_state = 0
      into
        :aarep_balance_end;

      select
        kcp_perc_saldo,
        kcp_perc_calced + iif(kcp_from_reg_id <> kcp_transfer_reg_id, kcp_perc_begin_saldo - kcp_perc_transfer_payed - kcp_perc_transfer_writeoff, 0),
        kcp_perc_payed - kcp_perc_transfer_payed,
        kcp_perc_writeoff - kcp_perc_transfer_writeoff,
        kcp_perc_begin_saldo,
        kcp_perc_transfer_payed,
        kcp_perc_transfer_writeoff,
        kcp_out_perc_saldo,
        kcp_out_perc_calced + iif(kcp_from_reg_id <> kcp_transfer_reg_id, kcp_out_perc_begin_saldo - kcp_perc_transfer_payed - kcp_perc_transfer_writeoff, 0),
        kcp_out_perc_begin_saldo
      from kontcalcperc
      where kcp_period = :rep_period
        and kcp_from_k_id = :aarep_k_id
        and kcp_from_c_id = 0
      into
        :aarep_perc_end,
        :aarep_perc_calced,
        :aarep_perc_payed,
        :aarep_perc_writeoff,
        :aarep_perc_begin,
        :tmp1,
        :tmp2,
        :aarep_out_perc_end,
        :aarep_out_perc_calced,
        :aarep_out_perc_begin;

      aarep_balance_begin = coalesce(aarep_balance_begin, 0);
      aarep_perc_begin = coalesce(aarep_perc_begin, 0);
      aarep_perc_calced = coalesce(aarep_perc_calced, 0);
      aarep_perc_payed = coalesce(aarep_perc_payed, 0);
      aarep_perc_writeoff = coalesce(aarep_perc_writeoff, 0);
      aarep_perc_end = coalesce(aarep_perc_end, 0);
      aarep_balance_end = coalesce(aarep_balance_end, 0);
      aarep_calced_all = coalesce(aarep_calced_all, 0);
      aarep_payed_all = coalesce(aarep_payed_all, 0);
      aarep_writeoff_all = coalesce(aarep_writeoff_all, 0);
      aarep_disb_amount = coalesce(aarep_disb_amount, 0);
      aarep_out_perc_begin = coalesce(aarep_out_perc_begin, 0);
      aarep_out_perc_calced = coalesce(aarep_out_perc_calced, 0);
      aarep_out_perc_end = coalesce(aarep_out_perc_end, 0);

      if (aarep_office_id_begin = aarep_office_id) then
      begin
        suspend;
      end else
      begin
        aarep_balance_begin1 = aarep_balance_begin;
        aarep_perc_begin1 = aarep_perc_begin;
        aarep_balance_begin = 0;
        aarep_perc_begin = 0;
        aarep_out_perc_begin1 = aarep_out_perc_begin;
        suspend;

        aarep_office_id = aarep_office_id_begin;
        aarep_office = aarep_office_begin;
        aarep_balance_begin = aarep_balance_begin1;
        aarep_perc_begin = aarep_perc_begin1;
        aarep_out_perc_begin = aarep_out_perc_begin1;
        aarep_perc_calced = 0;
        aarep_perc_payed = 0;
        aarep_perc_writeoff = 0;
        aarep_perc_end = 0;
        aarep_balance_end = 0;
        aarep_calced_all = 0;
        aarep_payed_all = 0;
        aarep_writeoff_all = 0;
        aarep_disb_amount = 0;
        aarep_perc_payed = tmp1;
        aarep_perc_writeoff = tmp2;
        aarep_out_perc_calced = 0;
        aarep_out_perc_end = 0;
        aarep_perc_calced = -(aarep_perc_begin - aarep_perc_payed - aarep_perc_writeoff);
        aarep_out_perc_calced = -(aarep_out_perc_begin - aarep_perc_payed - aarep_perc_writeoff);
        suspend;
      end
    end
  end
end
