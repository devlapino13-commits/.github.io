CREATE PROCEDURE HP_AZ_DELINQ_TRANSFER(
    REP_DATE_END DATE)
RETURNS (
    OK_FROM_K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    MAX_SH_DATE DATE,
    DELINQ_OS DECIMAL(18,4),
    DELINQ_PERC DECIMAL(18,4),
    OK_SH_ID INTEGER,
    OK_SH_DATE DATE,
    DELINQ_SH_OS DECIMAL(18,4),
    DELINQ_SH_PERC DECIMAL(18,4),
    SH_ID_NEW INTEGER,
    SH_DATE_NEW DATE,
    SH_OS_NEW DECIMAL(18,4),
    SH_PERC_NEW DECIMAL(18,4),
    IS_PAYMENT SMALLINT,
    RESTRUCT_MEMO VARCHAR(50),
    NEW_OL_ID DECIMAL(18,0))
AS
declare variable k integer;
declare variable is_restruct smallint;
begin
  /*Формирует все просроченные кредити индив. кредитов которые по ОС<=10*/
-- operkred_fix

  for select ok_from_k_id, k_kontract_no, max(ok_sh_date), sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
      from operkred, kontstate, kontract, product, knaktlist
      where ok_ol_is_active = 1 and
            ok_from_c_id = 0
            /*and k_kontract_no='BK/01668'*/
            /*and k_id=102010000024900201*/
            and
            ok_type < 7 and
            ok_ol_date <= :rep_date_end and
            ok_sh_date <= :rep_date_end and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_from_k_id = ok_from_k_id and
            ok_from_k_id = k_id and
            k_from_pr_id = pr_id and
            product.pr_loan_type = 1 and
            knaktlist.is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            :rep_date_end between kal_date and kal_date_end and
            kal_new_reg_id = 1020100 /*Ограничение по офису Нариманова*/
      group by ok_from_k_id, k_kontract_no
      having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0 and sum(ok_os + ok_sh_os) <= 10.0
      into :ok_from_k_id, :k_kontract_no, :max_sh_date, :delinq_os, :delinq_perc
  do
  begin
    is_payment = 0;
    k = 0;
    /*Определяет после 30.06.2012 были ли погашении*/
    if (exists(select *
               from operkred
               where ok_ol_is_active = 1 and
                     ok_from_c_id = 0 and
                     ok_type = 4 and
                     ok_ol_date > '30.06.2012' and
                     ok_from_k_id = :ok_from_k_id)) then
      is_payment = 1;

    sh_id_new = null;
    sh_date_new = null;
    sh_os_new = null;
    sh_perc_new = null;
    select ok_sh_id, /*Вытаскивает следующую выплату, кроме просроченных выплат*/
           ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
    from operkred
    where ok_ol_is_active = 1 and
          ok_from_c_id = 0 and
          ok_from_k_id = :ok_from_k_id and
          ok_type < 7 and
          ok_sh_date > :max_sh_date
    group by ok_sh_id, ok_sh_date
    having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0
    order by ok_sh_id, ok_sh_date
    rows 1
    into :sh_id_new, :sh_date_new, :sh_os_new, :sh_perc_new;
    restruct_memo = 'Успешно!';
    is_restruct = 1;
    new_ol_id = null;
    if (:sh_id_new is null or is_payment = 1) then
    begin
      restruct_memo = 'Перенос даты не произведено!';
      is_restruct = 0;
    end
    else
    begin
      /*Генерирует ID operlist*/
      select result
      from get_kv_id(5)
      into :new_ol_id;
      /*Создает операцию в operlist*/
      insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_memo, ol_status, is_delete, ol_user_id,
                            ol_oper_subtype, ol_date_oper_day)
      values (:new_ol_id, 1020100, 1, :rep_date_end, 'Delinq_correct', 0, 0, 1000000000001, 2, :rep_date_end);

      /*Создает запись, перевод на следующую дату */
      k = :k + 1;
      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
      values (:new_ol_id, :k, :ok_from_k_id, :sh_id_new, :sh_date_new, 3, 0.0, 0.0, 0.0, 3, 0, :delinq_os, :delinq_perc,
              0.0, 0.0);

    end

    for select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred /*Вытаскивает просроченные выплаты*/
        where ok_ol_is_active = 1 and
              ok_from_c_id = 0 and
              ok_type < 7 and
              ok_ol_date <= :rep_date_end and
              ok_sh_date <= :rep_date_end and
              ok_from_k_id = :ok_from_k_id
        group by ok_sh_id, ok_sh_date
        having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0
        into :ok_sh_id, :ok_sh_date, :delinq_sh_os, :delinq_sh_perc

    do
    begin
      /*Создает запись, убирает остатки*/
      if (:is_restruct = 1) then
      begin
        k = :k + 1;
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:new_ol_id, :k, :ok_from_k_id, :ok_sh_id, :ok_sh_date, 3, 0.0, 0.0, 0.0, 3, 0, -1 * :delinq_sh_os,
                -1 * :delinq_sh_perc, 0.0, 0.0);
      end

      suspend;
    end
    /*   end*/
  end
end
