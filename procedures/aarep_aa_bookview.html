CREATE PROCEDURE AAREP_AA_BOOKVIEW(
    REGION_ID D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    DEL_STATUS D_INTEGER,
    ACC_DT D_STR_20,
    ACC_KT D_STR_20,
    MEMO_TEXT D_STR_255,
    CLIENT_TEXT D_STR_100,
    USER_ID D_ID,
    SHOW_CORR D_INTEGER = 0)
RETURNS (
    OB_CURRENCY D_STR_LANG,
    OB_CURR_RATE D_CURRENCY,
    OB_CLIENT_DT D_ID,
    OB_ACCOUNT_DT D_STR_20,
    OB_SUMMA_VAL D_CURRENCY,
    OB_SUMMA D_CURRENCY,
    OB_FROM_PC_ID D_STR_20,
    OB_FROM_KP_ID D_STR_10,
    OB_ACCOUNT_KT D_STR_20,
    OB_CLIENT_KT D_ID,
    OB_DETAIL D_STR_LONG,
    OB_SUBKONTO_DT D_INTEGER,
    OB_SUBKONTO_KT D_INTEGER,
    OL_ID D_ID,
    OL_FROM_REG_ID D_INTEGER,
    OL_DATE D_DATE,
    OL_DOC_NO D_INTEGER,
    OL_DOC_NO_STR D_STR_20,
    OL_MEMO D_STR_LONG,
    IS_DELETE D_SMALLINT,
    OL_DATETIME D_DATETIME,
    OL_DOC_CLIENT_NAME D_STR_255,
    OL_YEAR D_INTEGER,
    OL_MONTH D_INTEGER,
    UN_NAME VARCHAR(30),
    DN_NAME D_STR_255,
    MO_NAME D_STR_100,
    CLIENT_DT D_STR_255,
    CLIENT_KT D_STR_255,
    IS_DELETE_NAME D_STR_20,
    OL_CALC_MEMO D_STR_LONG,
    OB_ACC_SHORT_DT D_STR_20,
    OB_ACC_SHORT_KT D_STR_20,
    OL_REGION_NAME D_STR_100,
    OL_BRANCH_NAME D_STR_100,
    OB_LOOK_DT_NAME VARCHAR(200),
    OB_LOOK_KT_NAME VARCHAR(200),
    OL_OPER_TYPE D_INTEGER,
    OL_DOC_KIND D_INTEGER,
    OL_IS_CLIENT D_INTEGER,
    OL_POCKET D_INTEGER,
    OB_SUB_DT_NAME D_STR_255,
    OB_SUB_KT_NAME D_STR_255,
    OL_PARENT_ID D_ID,
    OL_IS_SALARY_LIST D_INTEGER,
    OL_OPER_SUBTYPE D_INTEGER,
    OB_NUMBER D_INTEGER,
    REGION_NAME_CORR STR_100,
    ACC_SHORT_CORR STR_100,
    OLA_OPER_NO D_STR_CODE,
    OLA_DTSUPPLY D_DATE)
AS
declare variable db_id d_integer;
declare variable can_text d_integer;
declare variable can_client d_integer;
declare variable txt1 d_str_255;
declare variable txt2 d_str_255;
declare variable ol_new_id d_id;
declare variable ob_account_dt_prior d_str_20;
declare variable ob_account_kt_prior d_str_20;
declare variable ol_user_id d_id;
declare variable ol_user_id_prior d_id;
declare variable st_staff d_integer;
declare variable st_client d_integer;
declare variable jk_system_id d_integer;
declare variable can_suspend d_integer;
declare variable un_id curr_type;
declare variable ol_doc_kind_prior d_integer;
declare variable ol_pocket_prior d_integer;
declare variable ol_from_reg_id_prior d_integer;
declare variable is_active d_smallint;
declare variable sql_text d_str_2048;
declare variable ol_document_client d_id;
declare variable dt_yes d_integer;
declare variable kt_yes d_integer;
declare variable ol_id_corr id_type;
declare variable region_id_corr d_integer;
declare variable def_curr d_str_lang;
declare variable ag_caption d_str_255;
begin
  db_id = f_get_db_id();
  def_curr = f_get_def_currency();

  select st_user_client_access, st_user_staff_access, jk_system_id
  from get_stored_proc_options
  into :st_client, :st_staff, :jk_system_id;
  un_id = (select un_id
           from username
           where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN'));
  is_active = del_status;

  ol_branch_name = '';
  ol_user_id_prior = 0;
  ob_account_dt_prior = '';
  ob_account_kt_prior = '';
  ol_is_salary_list = 0;
  ob_sub_dt_name = 'Не указан (Субконто)';
  ob_sub_kt_name = 'Не указан (Субконто)';
  ob_subkonto_dt = 0;
  ob_subkonto_kt = 0;

  ol_pocket_prior = -1;
  ol_doc_kind_prior = -1;
  ol_from_reg_id_prior = 0;

  if (acc_dt is null) then
    acc_dt = '';
  if (acc_kt is null) then
    acc_kt = '';
  if (acc_dt <> '') then
    acc_dt = replace(:acc_dt, '%', '/');
  if (acc_kt <> '') then
    acc_kt = replace(:acc_kt, '%', '/');

  if (memo_text is null) then
    memo_text = '';
  if (client_text is null) then
    client_text = '';

  sql_text = 'select ol_id, operlist.is_delete, ol_oper_type, ol_oper_subtype, ol_is_client, ol_pocket,
             ol_doc_kind, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
             ob_summa, ob_from_pc_id, ob_from_kp_id, ob_account_kt, ob_client_kt,
             ob_detail, ol_from_reg_id, ol_date, ol_doc_no, ol_doc_no_str,
             ol_memo, ol_datetime, ol_doc_client_name, ol_year,
             ol_month, ol_document_client, ol_user_id,
             OL_PARENT_ID, OB_NUMBER, OL_Doc_Kind, OL_Pocket
       from Operlist, Region_List(:Region_ID, 1, 3, 1), OperBook
       Where ob_ol_date between :Date_Begin and :Date_End';
  if (is_active > -1) then
    sql_text = sql_text || ' and OB_OL_Is_Active=' || :is_active;
  if (acc_dt <> '') then
  begin
    if (position('/', acc_dt) > 0) then
      sql_text = sql_text || ' and (OB_short_acc_dt=''' || :acc_dt || ''')';
    else
      sql_text = sql_text || ' and (OB_short_acc_dt starting ''' || :acc_dt || ''')';
  end
  if (acc_kt <> '') then
  begin
    if (position('/', acc_kt) > 0) then
      sql_text = sql_text || ' and (OB_short_acc_kt=''' || :acc_kt || ''')';
    else
      sql_text = sql_text || ' and (OB_short_acc_kt starting ''' || :acc_kt || ''')';
  end
  sql_text = sql_text || ' and OB_From_OL_ID=OL_ID and ob_ol_reg_id=ID_List';
  for execute statement(:sql_text)(region_id := :region_id, date_begin := :date_begin, date_end := :date_end)
          into :ol_id, :is_delete, :ol_oper_type, :ol_oper_subtype, :ol_is_client, :ol_pocket, :ol_doc_kind,
          :ob_currency, :ob_curr_rate, :ob_client_dt, :ob_account_dt, :ob_summa_val, :ob_summa, :ob_from_pc_id,
          :ob_from_kp_id, :ob_account_kt, :ob_client_kt, :ob_detail, :ol_from_reg_id, :ol_date, :ol_doc_no,
          :ol_doc_no_str, :ol_memo, :ol_datetime, :ol_doc_client_name, :ol_year, :ol_month, :ol_document_client,
          :ol_user_id, :ol_parent_id, :ob_number, :ol_doc_kind, :ol_pocket
  do
  begin
    if ((:user_id = -1) or (ol_user_id = :user_id)) then
    begin
      can_suspend = 1;
      if (ol_oper_type = 2 and
          ol_oper_subtype in (24, 25)) then -- {Хоздоговора не показывать никому кроме бухгалтеров
      begin
        if (not(jk_system_id in (3, 4))) then
          can_suspend = 0;
      end
      if (un_id in (1090100000752)) then
        can_suspend = 1; -- Доступ для Ироды на хоздоговора, добавлено от 20130816
      if (can_suspend = 1) then
      begin
        if (ob_summa_val = ob_summa and ob_currency = def_curr) then
          ob_summa_val = 0;
        can_client = 1;
        can_text = 1;
        client_kt = '';
        client_dt = '';
        if (memo_text <> '') then
        begin
          can_text = 0;
          if (ob_detail is null) then
            ob_detail = '';
          if (ol_memo is null) then
            ol_memo = '';
          txt1 = left(ob_detail, 250);
          txt2 = left(ol_memo, 250);
          if ((position(:memo_text, :txt1) > 0) or (position(:memo_text, :txt2) > 0)) then
            can_text = 1;
        end
        if (can_text = 1) then
        begin
          if (ob_client_dt is not null) then
          begin
            select c_name_fio
            from clients
            where c_id = :ob_client_dt
            into :client_dt;
          end
          if (ob_client_kt is not null) then
          begin
            select c_name_fio
            from clients
            where c_id = :ob_client_kt
            into :client_kt;
          end
          if ((ob_client_dt is null) and
              (ob_client_kt is null)) then
          begin
            if (ol_document_client is not null) then
            begin
              select c_name_fio
              from clients
              where c_id = :ol_document_client
              into :client_dt;
              client_dt = 'Док.на имя: ' || :client_dt;

              if ((ob_account_dt like '%50-01%') or (ob_account_dt like '%51-01%')) then
              begin
                ob_client_kt = ol_document_client;
              end
              if ((ob_account_kt like '%50-01%') or (ob_account_kt like '%51-01%')) then
              begin
                ob_client_dt = ol_document_client;
              end

            end
            else
            begin
              client_dt = 'Док.на имя: ' || :ol_doc_client_name;
            end
          end
          if ((ob_account_kt like '%58-03%') and
              (ob_client_kt is null)) then
          begin
            ob_client_kt = ol_document_client;
          end
          if ((ob_account_dt like '%58-03%') and
              (ob_client_dt is null)) then
          begin
            ob_client_dt = ol_document_client;
          end
          if ((ob_account_kt like '%76-09%') and
              (ob_client_kt is null)) then
          begin
            ob_client_kt = ol_document_client;
          end
          if ((ob_account_dt like '%76-09%') and
              (ob_client_dt is null)) then
          begin
            ob_client_dt = ol_document_client;
          end
          if ((ob_account_kt like '%76-02%') and
              (ob_client_kt is null)) then
          begin
            ob_client_kt = ol_document_client;
          end
          if ((ob_account_dt like '%76-02%') and
              (ob_client_dt is null)) then
          begin
            ob_client_dt = ol_document_client;
          end
          if (client_text <> '') then
          begin
            can_client = 0;
            if (position(:client_text, client_dt) > 0) then
              can_client = 1;
            else
            if (position(:client_text, client_kt) > 0) then
              can_client = 1;
          end
        end
        if (can_client * can_text = 1) then
        begin
          if (ol_doc_kind <> ol_doc_kind_prior) then
          begin
            select dn_name
            from docname
            where dn_id = :ol_doc_kind
            into :dn_name;
            ol_doc_kind_prior = :ol_doc_kind;
          end
          if (ol_pocket <> ol_pocket_prior) then
          begin
            select mo_name
            from mopocket
            where mo_id = :ol_pocket
            into :mo_name;
            ol_pocket_prior = :ol_pocket;
          end

          if (ol_from_reg_id_prior <> ol_from_reg_id) then
          begin
            select r1.reg_name, r2.reg_name
            from region r1, region r2
            where r1.reg_unicode = :ol_from_reg_id and
                  r1.parent = r2.reg_unicode
            into :ol_region_name, ol_branch_name;
            ol_from_reg_id_prior = ol_from_reg_id;
          end

          if (ol_user_id <> ol_user_id_prior) then
          begin
            select un_name
            from username
            where un_id = :ol_user_id
            into :un_name;
            ol_user_id_prior = ol_user_id;
          end
          if (ob_account_dt_prior <> :ob_account_dt) then
          begin
            select al_account, al_name
            from acc_list
            where al_id = :ob_account_dt
            into :ob_acc_short_dt, :ob_look_dt_name;
            ob_account_dt_prior = :ob_account_dt;
          end
          if (ob_account_kt_prior <> :ob_account_kt) then
          begin
            select al_account, al_name
            from acc_list
            where al_id = :ob_account_kt
            into :ob_acc_short_kt, :ob_look_kt_name;
            ob_account_kt_prior = :ob_account_kt;
          end

          is_delete_name = null;
          if (is_delete = 1) then
            is_delete_name = 'Аннулирован';
          if (is_delete_name is null) then
            is_delete_name = '';
          if ((ob_detail is not null) and
              (ol_memo is not null)) then
          begin
            if (ob_detail <> ol_memo) then
              ol_calc_memo = left(:ob_detail || ' ' || :ol_memo, 1024);
            else
              ol_calc_memo = :ol_memo;
          end
          else
          begin
            if (ob_detail is not null) then
              ol_calc_memo = :ob_detail;
            else
              ol_calc_memo = :ol_memo;
          end
          if ((ol_oper_type = 2) and
              ((ol_oper_subtype = 21) or (ol_oper_subtype = 25)) and
              ((ol_doc_kind = 6) or (ol_doc_kind = 9))) then
          begin
            ol_new_id = 0;
            ol_is_salary_list = 0;
            select distinct osc_from_ol_id
            from opersacard
            where (osc_from_ol_id = :ol_id or
                  osc_from_ol_id = :ol_parent_id) rows 1
            into :ol_new_id;
            if ((ol_new_id is not null) and
                (ol_new_id <> 0)) then
            begin
              ol_is_salary_list = 1;
            end
          end
          -- S.K.
          ol_id_corr = null;
          acc_short_corr = null;
          region_id_corr = null;
          region_name_corr = null;
          if (db_id in (39801, 41701) and show_corr = 1) then begin
            dt_yes = iif(left(ob_acc_short_dt,5) in ('11510', '11512'), 1, 0);
            kt_yes = iif(left(ob_acc_short_kt,5) in ('11510', '11512'), 1, 0);
            IF (dt_yes <> kt_yes) then begin
              if (ol_parent_id is null) then begin
                select first 1
                  ol_id, ol_from_reg_id
                from operlist
                  inner join operbook
                    on ob_from_ol_id = ol_id and ob_ol_is_active = 1
                      and not (left(ob_short_acc_dt,5) in ('11510', '11512') and
                               left(ob_short_acc_kt,5) in ('11510', '11512'))
                where ol_parent_id = :ol_id
                  and ob_summa = :ob_summa
                order by ol_id, ol_parent_id
                into
                  :ol_id_corr, :region_id_corr;
                if (ol_id_corr is null) then
                  select first 1
                    ol_id, ol_from_reg_id
                  from operlist
                    inner join operbook
                      on ob_from_ol_id = ol_id and ob_ol_is_active = 1
                        and not (left(ob_short_acc_dt,5) in ('11510', '11512') and
                                 left(ob_short_acc_kt,5) in ('11510', '11512'))
                  where ol_parent_id = :ol_id
                  order by ol_id, ol_parent_id
                  into
                    :ol_id_corr, :region_id_corr;
              end else begin
                select first 1
                  ol_id, ol_from_reg_id
                from operlist
                  inner join operbook
                    on ob_from_ol_id = ol_id and ob_ol_is_active = 1
                      and not (left(ob_short_acc_dt,5) in ('11510', '11512') and
                               left(ob_short_acc_kt,5) in ('11510', '11512'))
                where ol_parent_id = :ol_parent_id
                  and ol_id <> :ol_id
                order by ol_id, ol_parent_id
                into
                  :ol_id_corr, :region_id_corr;

                if (row_count = 0) then
                select
                  ol_id,
                  ol_from_reg_id
                from operlist
                where ol_id = :ol_parent_id
                  and is_delete = 0
                into
                  :ol_id_corr, :region_id_corr;
              end

              if (ol_id_corr is not null) then begin
                region_name_corr = (select reg_name from region where reg_unicode = :region_id_corr);
                select
                  list(acc)
                from (
                  select distinct acc from (
                    select first 3
                      iif(left(ob_short_acc_dt,5) in ('11510', '11512'), ob_short_acc_kt, ob_short_acc_dt) as acc
                    from operbook
                    where ob_from_ol_id = :ol_id_corr and ob_ol_is_active = 1
                        and ob_summa = :ob_summa
                        and not (left(ob_short_acc_dt,5) in ('11510', '11512') and
                                 left(ob_short_acc_kt,5) in ('11510', '11512'))
                    order by ob_number )
                )
                into
                  :acc_short_corr;
                if (acc_short_corr is null) then
                  select
                    list(acc)
                  from (
                    select distinct acc from (
                      select first 3
                        iif(left(ob_short_acc_dt,5) in ('11510', '11512'), ob_short_acc_kt, ob_short_acc_dt) as acc
                      from operbook
                      where ob_from_ol_id = :ol_id_corr and ob_ol_is_active = 1
                          and not (left(ob_short_acc_dt,5) in ('11510', '11512') and
                                   left(ob_short_acc_kt,5) in ('11510', '11512'))
                      order by ob_number )
                  )
                  into
                    :acc_short_corr;
              end
            end
          end
          if (db_id in (39801, 41701)) then
          begin
            ola_oper_no = null;
            ola_dtsupply = null;
            select
              ola_oper_no,
              ola_dtsupply
            from operlist_add
            where ola_from_ol_id = :ol_id
            into
              :ola_oper_no,
              :ola_dtsupply;

            if (ob_acc_short_dt = '21122/300') then
            begin
              ag_caption = '';
              select ag_caption
              from agreements
              where ag_status = 1 and
                    ag_from_rd_id = f_get_rd_id('AgreementKinds.insurance') and
                    :ol_date between ag_date_begin and ag_date_end
              order by ag_date_begin desc rows 1
              into ag_caption;
              client_dt = 'Док.на имя: ' || ag_caption; -- RFCKG-3661
            end
          end 
          --
          suspend;
        end
      end
    end
  end
end
