CREATE PROCEDURE PRINT_BEN_CLI(
    C_ID ID_TYPE)
RETURNS (
    ANK_FILLED_FIRST_TIME_TXT STR_100,
    ANK_UPDATE_DATA_TXT STR_100,
    PRINT_DATE DATE,
    C_TYPE SMALLINT,
    C_NAME_FIO STR_100,
    C_INN STR_20,
    CREATE_DATE DATE,
    LO_FIO STR_100,
    LO_JL_NAME STR_100,
    BEN_C_TYPE INTEGER,
    BEN_C_NAME_FIO STR_100,
    BEN_C_NAME_FULL STR_255,
    BEN_C_BIRTH_DATE DATE,
    BEN_C_PASS_NO STR_100,
    BEN_C_PASS_DATE DATE,
    BEN_C_PASS_WHO STR_255,
    BEN_C_PASS_DATE_EXPIRE DATE,
    BEN_C_SEX INTEGER,
    BEN_C_BIRTH_CITY STR_100,
    BEN_C_INN VARCHAR(20),
    BEN_C_NATION_STR STR_100,
    BEN_C_CITIZEN_STR STR_255,
    BEN_C_FAMILY_STR STR_255,
    BEN_C_CELL_NO STR_100,
    BEN_C_EMAIL STR_100,
    BEN_C_ADDRESS_YUR STR_255,
    BEN_C_ADDRESS_FACT STR_255,
    BEN_CB_PERCENT DECIMAL(5,2),
    BEN_C_PFT_RISK_KOEF INTEGER,
    BEN_C_PFT_RISK_MEMO VARCHAR(1024),
    BEN_C_IS_RESIDENT SMALLINT,
    BEN_C_IS_PUBLIC SMALLINT,
    FIRST_TIME SMALLINT,
    C_DATETIME TIMESTAMP,
    BEN_QR_CODE D_STR_CODE,
    BEN_C_ALTERNAT_FIO D_STR_100,
    BEN_C_OKPO D_STR_100,
    BEN_C_FORM_STATUS_LONG D_STR_255,
    BEN_C_FORM_STATUS_SHORT D_STR_20,
    BEN_C_ORG_TYPE_LONG D_STR_255,
    BEN_C_ORG_TYPE_SHORT D_STR_20)
AS
declare variable rd_id id_type;
declare variable kf_id id_type;
declare variable lo_id id_type;
declare variable rd_id_1 id_type;
begin
  ank_filled_first_time_txt = 'Заполняется впервые';
  ank_update_data_txt = 'Обновление анкетных данных';
  print_date = current_date;

  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.IPF_Sellers'
  into :rd_id;

  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.tsp_and_ipf'
  into :rd_id_1;

  first_time = 0;
  if (exists(select *
             from kontmat kml, agreements ag, kontstate ks
             where kml_from_ag_id = ag_id and
                   kml_object_kind = 9 and
                   kml_from_ml_id is not null and
                   ag_object_kind = 0 and
                   ag.ag_from_rd_id in (:rd_id, :rd_id_1) and
                   ag_object_id = :c_id and
                   ks.ks_date_begin < current_date and
                   ks.ks_from_k_id = kml.kml_object_id and
                   current_date between ks.ks_date_begin and ks.ks_date_end and
                   ks.ks_ol_is_delete = 0 and
                   ks.ks_state = 1
             order by kml_id)) then
    first_time = 1;

  print_date = current_date;

  select cl.c_type, cl.c_name_fio, cl.c_inn, c_datetime
  from clients cl
  where cl.c_id = :c_id
  into :c_type, c_name_fio, c_inn, c_datetime;

  select cast(m.ma_date_save as date), ma_from_kf_id
  from miniappclient ml
  inner join miniapp m on ml.mac_from_ma_id = m.ma_id
  where ml.mac_from_c_id = :c_id and
        ml.mac_amount > 0 and
        m.ma_date_end >= '31.12.2999'
  order by m.ma_date_save desc
  rows 1
  into :create_date, :kf_id;

  select pcc.qr_code
  from print_client_cell(:c_id) pcc
  into ben_qr_code;

  select cl.c_name_fio, cl.c_id
  from knaktlist, clients cl
  where kal_from_kf_id = :kf_id and
        knaktlist.is_delete = 0 and
        kal_date <= current_date and
        kal_date_end >= current_date and
        kal_new_kf_officer = c_id

  into :lo_fio, lo_id;

  select jl_name
  from joblist, hr_pstatus
  where ps_from_jl_id = jl_id and
        ps_from_c_id = :lo_id and
        ps_is_working = 1 and
        current_date between ps_date_begin and ps_date_end
  into lo_jl_name;

  for select ben_c_type, ben_c_name_fio, ben_c_name_full, ben_c_birth_date, ben_c_pass_no, ben_c_pass_date,
             ben_c_pass_who, ben_c_pass_date_expire, ben_c_sex, ben_c_birth_city, ben_c_inn, ben_c_nation_str,
             ben_c_citizen_str, ben_c_family_str, ben_c_cell_no, ben_c_email, ben_c_address_yur, ben_c_address_fact,
             ben_cb_percent, ben_c_pft_risk_koef, ben_c_pft_risk_memo, ben_c_is_resident, ben_c_is_public,
             ben_c_alternat_fio, ben_c_okpo, ben_c_form_status_long, ben_c_form_status_short, ben_c_org_type_long,
             ben_c_org_type_short
      from print_client_ben(:c_id)
      into :ben_c_type, :ben_c_name_fio, :ben_c_name_full, :ben_c_birth_date, :ben_c_pass_no, :ben_c_pass_date,
           :ben_c_pass_who, :ben_c_pass_date_expire, :ben_c_sex, :ben_c_birth_city, :ben_c_inn, :ben_c_nation_str,
           :ben_c_citizen_str, :ben_c_family_str, :ben_c_cell_no, :ben_c_email, :ben_c_address_yur, :ben_c_address_fact,
           :ben_cb_percent, :ben_c_pft_risk_koef, :ben_c_pft_risk_memo, ben_c_is_resident, ben_c_is_public,
           :ben_c_alternat_fio, :ben_c_okpo, :ben_c_form_status_long, :ben_c_form_status_short, :ben_c_org_type_long,
           :ben_c_org_type_short
  do
  begin
    suspend;
  end
end
