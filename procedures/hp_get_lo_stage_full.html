CREATE PROCEDURE HP_GET_LO_STAGE_FULL(
    LO_ID ID_TYPE,
    REP_DATE DATE,
    SYSTEM_CODE VARCHAR(100),
    IS_ALL_POSITION SMALLINT = 0)
RETURNS (
    LO_STAGE_MONTH DECIMAL(4,2),
    LO_DEKRET_MONTH DECIMAL(4,2),
    LO_STAGE_MONTH_FULL DECIMAL(4,2))
AS
declare variable ps_date_begin date;
declare variable ps_date_end date;
declare variable stage_temp decimal(4,2);
declare variable day_razn integer;
declare variable hrv_date_end_fact date;
declare variable hrv_date_begin date;
begin
  lo_stage_month = 0;
  lo_dekret_month = 0;
  if (system_code is null or system_code = '') then
    system_code = '-1';
  for select ps_date_begin, ps_date_end
      from hr_pstatus, joblist, jobkind
      where ps_from_c_id = :lo_id and
            ps_is_working = 1 and
            ps_status = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            (coalesce(jk_system_code, '') = :system_code or
            :is_all_position = 1) and
            ((not(jk_system_code = 'JobSystem.Region_KM' and
            jl_name = 'И.о. заместителя менеджера офиса') and
            :is_all_position = 0) or
            :is_all_position = 1) and
            ps_date_begin <= ps_date_end and
            ps_date_begin <= :rep_date
      into ps_date_begin, ps_date_end
  do
  begin
    if (ps_date_end > :rep_date) then
      ps_date_end = :rep_date;
    stage_temp = datediff(month, ps_date_begin, ps_date_end);
    if (extract(day from ps_date_begin) > extract(day from ps_date_end) and
        stage_temp > 0) then
      stage_temp = stage_temp - 1;
    day_razn = ps_date_end - dateadd(month, stage_temp, ps_date_begin);
    stage_temp = stage_temp + cast(day_razn as double precision) / 30;

    lo_stage_month = lo_stage_month + stage_temp;

    for select hrv_date_begin, hrv_date_end_fact
        from hr_voctravel, hr_orders, sa_worktype
        where hrv_from_ho_id = ho_id and
              ho_order_type = sawt_id and
              hrv_from_c_id = :lo_id and
              hrv_status = 1 and
              sawt_id in (16, 17, 18, 21) and
              hrv_date_begin <= hrv_date_end_fact and
              hrv_date_begin <= :ps_date_end and
              hrv_date_end_fact >= :ps_date_begin
        into hrv_date_begin, hrv_date_end_fact
    do
    begin
      if (hrv_date_begin < ps_date_begin) then
        hrv_date_begin = ps_date_begin;
      if (hrv_date_end_fact > ps_date_end) then
        hrv_date_end_fact = ps_date_end;

      if (hrv_date_end_fact > rep_date) then
        hrv_date_end_fact = rep_date;
      stage_temp = datediff(month, hrv_date_begin, hrv_date_end_fact);
      if (extract(day from hrv_date_begin) > extract(day from hrv_date_end_fact) and
          stage_temp > 0) then
        stage_temp = stage_temp - 1;
      day_razn = hrv_date_end_fact - dateadd(month, stage_temp, hrv_date_begin);
      stage_temp = stage_temp + cast(day_razn as double precision) / 30;

      lo_dekret_month = lo_dekret_month + stage_temp;
    end
  end

  lo_stage_month_full = lo_stage_month - lo_dekret_month;
  suspend;
end
