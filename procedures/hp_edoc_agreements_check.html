CREATE PROCEDURE HP_EDOC_AGREEMENTS_CHECK(
    EDOC_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    SIGNER_C_ID D_ID)
AS
declare variable cell_no d_str_100;
begin
  res_text = '';
  if (edoc_kind = 'storage_agreement') then
  begin
    signer_c_id = 1090100331989; ---Подписантом может быть только Шавкат и доступ только у него.
    cell_no = f_cont_info('clients', :obj_id, 'cell_main', 'ok');
    if (cell_no = '') then
    begin
      res_text = 'У сотрудника нет верифицированного номера';
      suspend;
      exit;
    end
    if (f_edoc_status('storage_agreement', :obj_id, current_date) = 'signed') then
    begin
      res_text = 'Договор хранения уже подписан';
      suspend;
      exit;
    end
    if (not exists(select *
                   from hr_pstatus ps
                   inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
                   inner join jobkind jk on jk.jk_id = jl.jl_kind and
                         jk.jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Region_Director')
                   where ps.ps_is_working = 1 and
                         ps.ps_from_c_id = :obj_id and
                         current_date between ps.ps_date_begin and ps.ps_date_end)) then
    begin
      res_text = 'Договор может быть отправлен только КМ и РМ';
      suspend;
      exit;
    end
    /* if (not exists(select *
                   from edoc_list edl
                   inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id and
                         eds.eds_status = 1 and
                         cast(eds.eds_status_datetime as date) >= '11.01.2022'
                   where edl.edl_object_kind = f_objkind('clients', 'id') and
                         edl.edl_object_id = :obj_id and
                         edl.edl_from_edc_id = f_edoc_kind('asp.agreement', 'id') and
                         current_date between edl.edl_blank_date and coalesce(edl.edl_date_end, current_date) and
                         edl.edl_status = 1
                   order by edl.edl_status desc, edl.edl_id desc
                   rows 1)) then
    begin
      res_text = 'Нет подписанного ЭЦП, от 11.01.2022.';
      suspend;
      exit;
    end  */
    if (not exists(select *
                   from clientdocs cds
                   inner join passkind pk on cds.cds_from_pk_id = pk.pk_id and
                         pk.pk_is_delete = 0
                   inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id and
                         pdt.pdt_is_main_doc = 1
                   where cds.cds_from_c_id = :obj_id and
                         current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end and
                         cds.cds_is_delete = 0)) then
    begin
      res_text = 'Нет действующего паспорта.';
      suspend;
      exit;
    end
    suspend;
  end
end
