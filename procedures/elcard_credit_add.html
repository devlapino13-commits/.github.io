CREATE PROCEDURE ELCARD_CREDIT_ADD(
    ESM_ID ID_TYPE,
    FULL_SOURCE D_BLOB_TEXT,
    ANSWER D_BLOB_TEXT,
    ERROR_CODE D_BLOB_TEXT)
AS
declare variable esm_date d_date;
declare variable code d_str_20;
declare variable pos d_integer;
declare variable str_blob d_blob_text;
begin
  error_code = trim(coalesce(error_code, ''));
  answer = trim(coalesce(answer, ''));
  full_source = trim(coalesce(full_source, ''));

  if (esm_id is null) then
    exception e_default 'ESM_ID is null';

  select esm.esm_date
  from extsysmain esm
  where esm.esm_id = :esm_id
  and esm.esm_project = 'elcard_disb'
  and esm.esm_action = 'CardDisb'
  into :esm_date;

  if (esm_date is null) then -- исхожу из того, что не может быть такого, чтобы  у extsysmain не был заполнен esm_date, поэтому если esm_date не заполняется считаю что записи вообще нет и поднимаю exception
    exception e_default 'ESM_ID is not assigned to "elcard_disb"';

  delete from extsysanswer ea
  where ea.esa_from_esm_id = :esm_id;

  if (error_code <> '') then
  begin
    insert into extsysanswer (esa_from_esm_id, esa_message)
    values (:esm_id, 'error: ' || :error_code);
  end
  else
  begin
    pos = (position('"confirm"', answer));
    if (pos <= 0) then -- если не найдено json-а связанного с переводом
    begin
         if (position('"create"', answer) <= 0) then  -- если не найдено json-а связанного с созданием платежного поручения
         begin
            error_code = '-2'; -- нет ни json связанного с "Переводом" , ни jsoт связанного с "Платежным поручением"
         end
         else
         begin
            pos = position('"create"', answer);
            str_blob = substring(answer from pos + 9);
            pos = position('}', str_blob);
            str_blob = left(str_blob, pos);

            error_code = f_json_str_get_value(str_blob, 'status', '-3');
         end
    end
    else
    begin
        str_blob = substring(answer from pos + 10);
        pos = position('}', str_blob);
        str_blob = left(str_blob, pos); -- на этом этапе у меня должен получиться json вида:{"status": 310, "message": "301 Неверный идентификатор платежного поручения"}
        error_code = f_json_str_get_value(str_blob, 'status', '-4');

        if (error_code = '300') then -- при переводе денег, код 300 - это не ошибка, а подтверждение успешной обработки
            error_code = '';
    end

    if (error_code <> '') then
    begin
        insert into extsysanswer (esa_from_esm_id, esa_message)
        values (:esm_id, 'error: ' || :error_code);
    end
  end

  if (answer <> '') then
  begin
    insert into extsysanswer (esa_from_esm_id, esa_message)
    values (:esm_id, 'answer: ' || :answer);
  end

  update extsysmain em
  set em.esm_status = iif(:error_code = '', 1, -1),
      em.esm_full_source = :full_source
  where em.esm_id = :esm_id;
end
