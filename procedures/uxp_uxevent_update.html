Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXEVENT_UPDATE(
    NEW_EVT_UUID TYPE OF D_UUID,
    NEW_EVT_SENDER_UUID TYPE OF D_UUID,
    NEW_EVT_EVENT_KIND TYPE OF D_INTEGER,
    NEW_EVT_TARGET_UUID TYPE OF D_UUID,
    NEW_EVT_HANDLER_KIND TYPE OF D_INTEGER,
    NEW_EVT_POSITION TYPE OF D_INTEGER)
RETURNS (
    EVT_UUID TYPE OF D_UUID,
    EVT_SENDER_UUID TYPE OF D_UUID,
    EVT_EVENT_KIND TYPE OF D_INTEGER,
    EVT_TARGET_UUID TYPE OF D_UUID,
    EVT_HANDLER_KIND TYPE OF D_INTEGER,
    EVT_POSITION TYPE OF D_INTEGER)
AS
begin
  if (new_evt_uuid is not null and exists(select evt_uuid from ux$event where (evt_uuid = :new_evt_uuid))) then
  begin
    update ux$event
    set evt_sender_uuid = :new_evt_sender_uuid,
        evt_event_kind = :new_evt_event_kind,
        evt_target_uuid = :new_evt_target_uuid,
        evt_handler_kind = :new_evt_handler_kind,
        evt_position = :new_evt_position
    where (evt_uuid = :new_evt_uuid)
    returning evt_uuid, evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position
    into evt_uuid, evt_sender_uuid, evt_event_kind,  evt_target_uuid, evt_handler_kind, evt_position;
  end
  else
  begin
    insert into ux$event (evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position)
    values (:new_evt_sender_uuid, :new_evt_event_kind, :new_evt_target_uuid, :new_evt_handler_kind, :new_evt_position)
    returning evt_uuid, evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position
    into evt_uuid, evt_sender_uuid, evt_event_kind,  evt_target_uuid, evt_handler_kind, evt_position;
   end

   suspend;
end
