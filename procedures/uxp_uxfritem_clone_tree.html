CREATE PROCEDURE UXP$UXFRITEM_CLONE_TREE(
    NEW_UX_UUID D_UUID,
    NEW_PARENT_UUID D_UUID,
    ORIG_PARENT_UUID D_UUID)
AS
declare variable ufi_uuid d_uuid;
declare variable new_ufi_uuid d_uuid;
begin
  for select ufi_uuid, ufi_parent_uuid, ufi_owner, ufi_ufr_uuid,
             ufi_uf_uuid, ufi_kind, ufi_control, ufi_position,
             ufi_name, ufi_caption_align, ufi_hint, ufi_hidden, ufi_readonly,
             ufi_group_direction, ufi_h_align, ufi_v_align, ufi_width, ufi_height,
             ufi_sizable, ufi_order_index, ufi_group_index, ufi_fixed_kind, ufi_summary_kind, ufi_font_size
  from ux$fritem
  where ufi_parent_uuid = :orig_parent_uuid
  order by ufi_position
  as cursor cur
  do
  begin
    ufi_uuid = cur.ufi_uuid;

    insert into ux$fritem (ufi_parent_uuid, ufi_owner, ufi_ufr_uuid,
                           ufi_uf_uuid, ufi_kind, ufi_control, ufi_position,
                           ufi_name, ufi_caption_align, ufi_hint, ufi_hidden, ufi_readonly,
                           ufi_group_direction, ufi_h_align, ufi_v_align, ufi_width, ufi_height,
                           ufi_sizable, ufi_order_index, ufi_group_index, ufi_fixed_kind, ufi_summary_kind, ufi_font_size)
    values (:new_parent_uuid, :new_ux_uuid, uxf$pair(cur.ufi_ufr_uuid),
            cur.ufi_uf_uuid, cur.ufi_kind, cur.ufi_control, cur.ufi_position,
            cur.ufi_name, cur.ufi_caption_align, cur.ufi_hint, cur.ufi_hidden, cur.ufi_readonly,
            cur.ufi_group_direction, cur.ufi_h_align, cur.ufi_v_align, cur.ufi_width, cur.ufi_height,
            cur.ufi_sizable, cur.ufi_order_index, cur.ufi_group_index, cur.ufi_fixed_kind, cur.ufi_summary_kind, cur.ufi_font_size)
    returning ufi_uuid
    into :new_ufi_uuid;

    execute procedure uxp$uxpairuuid_add(:ufi_uuid, :new_ufi_uuid);

    execute procedure uxp$uxfritem_clone_tree(:new_ux_uuid, :new_ufi_uuid, ufi_uuid);
  end
end
