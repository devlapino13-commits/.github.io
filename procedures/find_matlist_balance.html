CREATE PROCEDURE FIND_MATLIST_BALANCE(
    WORKPLACE_ID DECIMAL(18,0),
    DD DATE,
    TEXT_STR VARCHAR(50),
    INVENT_STR VARCHAR(50),
    WITH_CPL SMALLINT)
RETURNS (
    MBAL_BILL_ID DECIMAL(18,0),
    MBAL_BILL_LINE_NO INTEGER,
    MBAL_WP_ID DECIMAL(18,0),
    MBAL_MO_ID DECIMAL(18,0),
    MBAL_ML_ID DECIMAL(18,0),
    MBAL_KOLICH DECIMAL(18,4),
    MBAL_SUMMA DECIMAL(18,4),
    MBAL_BILL_COUNT INTEGER,
    MBAL_CPL_ID DECIMAL(18,0),
    MBAL_MC_ID DECIMAL(18,0),
    MBAL_INVENT_NO INTEGER,
    MBAL_INVENT_STR VARCHAR(50),
    MBAL_ACCOUNT VARCHAR(20),
    MBAL_IS_SKLAD SMALLINT,
    MBAL_FULL_DETAIL VARCHAR(2048),
    CPL_NAME VARCHAR(1024),
    C_NAME_FIO VARCHAR(100),
    ML_NAME VARCHAR(255),
    YL_CODE VARCHAR(20),
    WP_NAME VARCHAR(100),
    OL_DATE DATE)
AS
declare variable has_text smallint;
declare variable has_invent smallint;
declare variable text_str1 varchar(50);
declare variable wp integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  has_text = 0;
  has_invent = 0;
  text_str1 = :text_str;
  text_str = upper(:text_str);
  invent_str = upper(:invent_str);
  if (text_str <> '') then
    has_text = 1;
  if (invent_str <> '') then
    has_invent = 1;
  if ((has_text = 0) and
      (has_invent = 0)) then
  begin
    for select distinct mbal_bill_id, mbal_bill_line_no, mbal_wp_id, mbal_mo_id, mbal_ml_id, mbal_kolich, mbal_summa,
                        mbal_bill_count, mbal_cpl_id, mbal_mc_id, mbal_invent_str, mbal_invent_no, mbal_account,
                        mbal_is_sklad, mbal_full_detail, c_name_fio, ml_name, yl_code, mbal_wp_name, ol_date
        from matlist_balance_list(:workplace_id, 0, :dd, 0, 1), clients, matlist, yedizlist, operlist
        where ((:with_cpl = -1) or
              (mbal_cpl_id > 0 and
              :with_cpl = 1) or
              ((mbal_cpl_id is null) and
              :with_cpl = 0)) and
              mbal_mo_id = c_id and
              ml_from_yl_id = yl_id and
              ml_id = mbal_ml_id and
              ol_id = mbal_bill_id
        order by c_name_fio, ml_name
        into :mbal_bill_id, :mbal_bill_line_no, :mbal_wp_id, :mbal_mo_id, :mbal_ml_id, :mbal_kolich, :mbal_summa,
             :mbal_bill_count, :mbal_cpl_id, :mbal_mc_id, :mbal_invent_str, :mbal_invent_no, :mbal_account,
             :mbal_is_sklad, :mbal_full_detail, :c_name_fio, :ml_name, :yl_code, :wp_name, :ol_date
    do
    begin
      cpl_name = '';
      if (mbal_cpl_id is not null) then
      begin
        select cpl_name
        from complect
        where cpl_id = :mbal_cpl_id
        into :cpl_name;
      end
      suspend;
    end
  end
  else
  begin
    /* Поиск с критериями */
    mbal_bill_count = 1;
    for select distinct ob_bill_id, ob_bill_line_no
        from operbill, matlist
        where ml_id = ob_from_ml_id and
              ob_date_begin <= :dd and
              ob_date_end >= :dd and
              ob_is_delete = 0 and
              upper(ml_name) like '%' || :text_str || '%' and
              upper(ob_invent_str) like '%' || :invent_str || '%' and
              ((:with_cpl = -1) or
              (ob_complect_id > 0 and
              :with_cpl = 1) or
              ((ob_complect_id is null) and
              :with_cpl = 0))
        into :mbal_bill_id, :mbal_bill_line_no
    do
    begin
      for select om_workplace_id, om_matotvet_id, ob_from_ml_id,
                 sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) as kol_balance,
                 sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) as sum_balance, ob_complect_id,
                 ob_from_mc_id, ob_invent_str, ob_invent_no, om_account, fwp_status, ob_full_detail, fwp_name, ml_name,
                 yl_code, c_name_fio
          from opermat, operlist, full_wp_list(:workplace_id, 0), operbill, matlist, yedizlist, clients
          where om_bill_id = :mbal_bill_id and
                om_bill_line_no = :mbal_bill_line_no and
                om_from_ol_id = ol_id and
                om_workplace_id = fwp_id and
                is_delete = 0 and
                ol_date <= :dd and
                ob_bill_id = :mbal_bill_id and
                ob_bill_line_no = :mbal_bill_line_no and
                ob_date_begin <= :dd and
                ob_date_end >= :dd and
                ob_is_delete = 0 and
                ob_from_ml_id = ml_id and
                ml_from_yl_id = yl_id and
                om_matotvet_id = c_id
          group by om_workplace_id, om_matotvet_id, ob_from_ml_id, ob_complect_id, ob_from_mc_id, ob_invent_str, ob_invent_no, om_account, fwp_status, ob_full_detail, fwp_name, ml_name, yl_code, c_name_fio
          having(sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) <> 0) or (sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) <> 0)
          into :mbal_wp_id, :mbal_mo_id, :mbal_ml_id, :mbal_kolich, :mbal_summa, :mbal_cpl_id, :mbal_mc_id,
               :mbal_invent_str, :mbal_invent_no, :mbal_account, :wp, :mbal_full_detail, :wp_name, :ml_name, :yl_code,
               :c_name_fio
      do
      begin
        cpl_name = '';
        if (mbal_cpl_id is not null) then
        begin
          select cpl_name
          from complect
          where cpl_id = :mbal_cpl_id
          into :cpl_name;
        end

        if (wp = 6) then
          mbal_is_sklad = 1;
        else
          mbal_is_sklad = 0;

        select distinct ol_date
        from operlist
        where ol_id = :mbal_bill_id rows 1
        into :ol_date;

        suspend;
      end
    end
  end
end
