CREATE PROCEDURE FM_LOANS_CLOSED_TMP(
    C_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REG_ID D_INTEGER)
RETURNS (
    K_ID D_ID,
    CL_DATE D_DATE,
    KF_NAME D_STR_100)
AS
begin
    if (reg_id = -1) then
    begin
        for select ks_from_k_id, ks_date_begin, ks_kfolder_no
        from kontstate
        where ks_ol_is_delete = 0 and
              ks_date_begin between :rep_date_begin and :rep_date_end and
              ks_state = 2 and
              ks_kc_id = :c_id
        order by ks_date_begin desc
        into :k_id, cl_date, kf_name
        do
            suspend;
    end
    else
    begin
        for select ks_from_k_id, ks_date_begin, ks_kfolder_no
        from kontstate
        where ks_ol_is_delete = 0 and
              ks_date_begin between :rep_date_begin and :rep_date_end and
              ks_state = 2 and
              ks_region_id = :reg_id
        order by ks_date_begin desc
        into :k_id, cl_date, kf_name
        do
            suspend;
    end
end
