CREATE PROCEDURE HP_BKI_ADD_TO_BUFFER(
    SOURCE_ALIAS D_STR_100 NOT NULL,
    OBJECT_KIND_NAME D_STR_100 NOT NULL,
    OBJECT_ID D_ID NOT NULL,
    ACTUALITY_DATE D_DATE NOT NULL)
AS
declare variable object_kind d_integer;
declare variable tag d_integer;
declare variable bkk_id d_id;
declare variable bkk_code d_str_code;
declare variable eds_id d_id;
declare variable c_id d_id;
declare variable object_id_tmp d_id;
declare variable app_summ d_currency;
declare variable cnt d_integer;
begin
  -- пока только для РФ
  if (f_get_db_id() <> 81001) then
    exit;

  --- если идет процедура разделения договоров (амнистирование)
  if (coalesce(rdb$get_context('USER_TRANSACTION', 'DEVIDE_IN_PROGRESS'), 0) = 1) then  
    exit;

  tag = case
          when source_alias containing ('.operkred.')   then 1
          when source_alias containing ('.kontdata.')   then 2
          when source_alias containing ('.operkroff.')  then 3
          when source_alias containing ('.kontstate.')  then 4
          when source_alias containing ('.bki_errors.') then 6
          when source_alias containing ('.miniapp.')    then 7
          when source_alias containing ('.lawmain.')    then 8
          when source_alias containing ('.lawkredit.')  then 9
          when source_alias containing ('.hp_bki_get_packet.')            then 10
          when source_alias containing ('.hp_bki_get_loan_list_equifax.') then 11 -- переход на следующий статус
          when source_alias containing ('.hp_bki_packet_set_status.')     then 12 -- повторная отправка "затертых" пакетов после корректировки
          when source_alias containing ('.operextra.')                    then 13
          when source_alias containing ('.hp_edoc_after_sign_action.')    then 14 -- подписание заявки
          when source_alias containing ('.zalog.')      then 15 -- залоги
                                                             -- 16 - robot
          when source_alias containing ('.konteffrate.')then 17 -- ПСК
          when source_alias containing ('.kontpsk.')    then 18 -- ПСК
          when source_alias containing ('.bki_bankrupt.')then 19 -- Банкроты
          when source_alias containing ('.edoc_msg.')   then 20 -- Отправка документа
          when source_alias containing ('.perslist.')   then 21 -- Изменение перс. данных
          when source_alias containing ('.bankrupt.')   then 22 -- Банкротства
          when source_alias containing ('.knclient_pdn.')   then 23 -- Расчет ПДН 60 дней
          when source_alias containing ('.hp_edoc_ins_loan_agreement.')   then 24 -- Утверждение договора
--          when source_alias containing ('.perstemp.')   then 25
--          when source_alias containing ('.kn_oksummary.')   then 26 -- изменилась дата окончания договора
          when source_alias containing ('.kontdevides.')   then 27 -- разделение
          when source_alias containing ('.hp_sud_cession_send_bki.')   then 28 -- Цессии (продажа договора)
          when source_alias containing ('.extsysmain.')   then 29 -- выдача
          when source_alias containing ('.hp_sud_bki_send.')   then 30 -- суд (новый модуль)
        end;

  if (tag is null) then
    exception e_default 'unknoun tag for source_alias: ' || source_alias;

  -- стали происходить изменения в БД по очень старым контрактам, что рождает кучу ненужных событий в прошлом. Ограничим.
  -- Если это событие не по судебным делам и не отправка из процедур, то ограничим тремя месяцами
  -- При необходимости отправим вручную
  if (actuality_date < current_date - 90 and tag not in (6,8,9,10,11,12,13,15,30)) then
    exit;

  -- правила отбора событий для записи в буфер.
  if (tag = 4 and source_alias not containing ('.ks_state1.') and source_alias not containing ('.ks_state2.')) then
    exit;
  else if (tag = 13 and source_alias not containing ('.ox_from_rd_id' || f_get_rd_id('OperExtraKind.LawClientExpense') || '.')) then
    exit;
  else if (tag = 21 and source_alias not containing ('.pls_edit_kind1.')) then
    tag = 25; -- корректировка / изменения без подписания перс. данных
  else if (tag = 22 and exists(select 1
                              from bankrupt
                              where brt_from_c_id = :object_id
                                and brt_date_fact = :actuality_date
                                and brt_date_end < current_date)) --  не первая запись
  then
    exit;

  object_kind = f_objkind(object_kind_name, 'id');

  if (object_kind_name = 'miniapp') then -- если это заявка и по ней есть договор, то в буфер закинем договор, что бы избежать ситуации когда по одной паре заявка-договор были бы две записи
  begin
    object_id_tmp = object_id;

    select f_objkind('kontract', 'id'), k_id
    from kontract k
    where k.k_from_ma_app_id = :object_id_tmp
    order by k_status desc
    rows 1
    into object_kind, object_id;
  end

  -- если контракт есть в списке заблокированных (удаляли ранее по разным причинам), то запись не производим
  if (object_kind = f_objkind('kontract', 'id')
    and exists(select bsl_id from bki_stoplist_loans bsl where bsl.bsl_from_k_id = :object_id))
  then
    exit;

  -- если это подписание заявки, то сохраним сумму
  if (tag = 14) then
  begin
    select eds_id, eds_from_c_id
    from edoc_sign, edoc_list
    where eds_from_edl_id = edl_id
      and edl_from_edc_id = f_edoc_kind('loan.application','id')
      and edl_object_kind = :object_kind
      and edl_object_id = :object_id
      and edl_status = 1
      and eds_status = 1
    into eds_id, c_id;

    select mac_amount
    from miniappclient
    where mac_from_ma_id = :object_id
      and mac_from_c_id = :c_id
    into app_summ;

    if (eds_id is not null and app_summ is not null) then
      update or insert into bki_sign_app_data (bad_from_eds_id, bad_summ)
      values (:eds_id, :app_summ)
      matching (bad_from_eds_id);
  end
  else if (tag = 20) then -- ловим первую отправку договора
  begin
    cnt = 0;

    select edl_object_kind, edl_object_id, min(msg_datetime), count(distinct msg_id)
    from edoc_sign, edoc_list, edoc_msg, fsrobot_msgbox
    where eds_from_edl_id = edl_id
      and edl_from_edc_id = f_edoc_kind('loan.agreement','id')
      and edm_from_eds_id = eds_id
      and msg_id = edm_from_msg_id
      and eds_id = :object_id
    group by 1,2
    into object_kind, object_id, actuality_date, cnt;

    if (cnt <> 1) then
      exit;
  end 

  -- RFCRU-5786
  if (tag in (7, 14, 24, 29) -- заявка, одобрение (утверждение), выдача/договор, отказ
      and actuality_date > current_date - 2) -- максимум вчерашние
  then
  begin
    if (not exists (select *
            from bki_buffer
            where bkb_from_bkk_id = :bkk_id
              and bkb_object_kind = :object_kind
              and bkb_object_id   = :object_id
              and bkb_date        = :actuality_date
              and bkb_type        = 2)   --- AF
    )
    then
      insert into bki_buffer (bkb_from_bkk_id, bkb_object_kind, bkb_object_id, bkb_date, bkb_tag, bkb_type)
      values (1, :object_kind, :object_id, :actuality_date, :tag, 2);
  end

  for select bkk.bkk_id, bkk.bkk_code
    from bki_kind bkk
    where bkk.bkk_inactive = 0
  into bkk_id, bkk_code
  do
  begin
    if (bkk_code = 'ru.nbki' and source_alias not containing ('.nbki.')) then
      continue;

    if (not exists (select *
            from bki_buffer
            where bkb_from_bkk_id = :bkk_id
              and bkb_object_kind = :object_kind
              and bkb_object_id   = :object_id
              and bkb_date        = :actuality_date
              and bkb_type        = 0)
    )
    then
      insert into bki_buffer (bkb_from_bkk_id, bkb_object_kind, bkb_object_id, bkb_date, bkb_tag)
      values (:bkk_id, :object_kind, :object_id, :actuality_date, :tag);
  end
end
