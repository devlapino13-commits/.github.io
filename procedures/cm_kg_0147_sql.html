CM_KG_0147_SQL

CREATE PROCEDURE CM_KG_0147_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable c_id d_id;
declare variable is_dr d_smallint;
declare variable c_id_approve d_id;
begin
  res_error_is_critical = 1;
  -- если у клиент есть активный заем по классике по старому продукту (действующий продукт до 01.04.2021)
  --тогда клиент сможет взять еще один основной заем по Ишеним групповому или индив до 100 000 сом с консультацией
  for select k_disb_date,
             kc.kc_amount_approved,
             cl.c_name_fio,
             cl.c_id
  from kontract kont
  inner join product pr on k_from_pr_id = pr_id and
        pr_code containing '.STANDART.' and
        pr_code containing '.CONSULTATION.' and
        kont.k_refund = 0 -- не рефинансирование
  inner join knclient kc on k_id = kc_from_k_id and
        kc_amount_approved > 0
  inner join clients cl on cl.c_id = kc.kc_from_c_id
  where k_id = :k_id
  into :kf_disb_date, :kc_amount, :res_info_text, :c_id
  do
  begin
        if (exists(
            select *
            from kontract kont
            inner join product pr on k_from_pr_id = pr_id and
                pr_code not containing '.PAR.' and
                pr_code not containing '.IPF.'
            inner join knclient kc on k_id = kc_from_k_id and
                kc_amount_approved > 0
            inner join kontstate ks on ks.ks_from_k_id = kont.k_id
            where kc.kc_from_c_id = :c_id
            and kont.k_disb_date < '01.04.2021'
            and kont.k_id <> :k_id
            and ks.ks_ol_is_delete = 0
            and ks.ks_state = 1
            and :kf_disb_date between ks.ks_date_begin and ks.ks_date_end
        )) then
        begin
            if (kc_amount > 100000) then
            begin
                is_dr = 0;

                select un.un_from_c_id
                from username un
                where un.un_id = :un_id
                into c_id_approve;
            
                if (exists (select *
                           from hr_pstatus  hps
                           inner join joblist jl on jl.jl_id =  ps_from_jl_id
                           inner join jobkind jk on jk.jk_id = jl.jl_kind
                           where current_date between ps_date_begin and ps_date_end
                           and ps_is_working = 1
                           and jk.jk_system_code = 'JobSystem.Region_Director'
                           and hps.ps_from_c_id  = :c_id_approve)) then
                is_dr = 1;
    
                if (is_dr = 0) then
                begin
                  res_error_text = 'Если у клиент есть активный заем по классике по старому продукту, тогда клиент сможет взять еще один основной заем по Ишеним (групповому или индив.) до 100 000 сом (с консультацией)';
                  suspend;
                end
            end
        end
  end
end
