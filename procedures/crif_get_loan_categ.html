CREATE PROCEDURE CRIF_GET_LOAN_CATEG(
    K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    OPER_DATE DATE,
    DATE_TYPE D_STR_20,
    ACTIVE_DELINQ_DAYS D_INTEGER,
    DAYS_PAST_DUE_DOMAIN D_STR_10)
AS
begin
  for select :rep_date,
         'rep_date' as date_type from rdb$database
  union all
  select :rep_date - 1,
         'prev_date' as date_type from rdb$database
  into :oper_date, :date_type
  do
  begin
    -- получаем число дней активной просрочки -->
    active_delinq_days = 0;

    select iif(:oper_date - oks.oks_delinq_date <= 0, 0, :oper_date - oks.oks_delinq_date)
    from oksummary oks
    where oks.oks_from_k_id = :k_id and
          oks.oks_close_state = 0 and
          :oper_date between oks.oks_date_begin and oks.oks_date_end
    into :active_delinq_days;
    -- <-------------------------------------------

    if (active_delinq_days <= 0) then
        days_past_due_domain = '1';
    else
    if (active_delinq_days < 30) then
        days_past_due_domain = '2'; --до 30 дней просрочки
    else
    if ((active_delinq_days >= 30) and
        (active_delinq_days < 60)) then
        days_past_due_domain = '3'; --30 дней до 59 дней просрочки
    else
    if ((active_delinq_days >= 60) and
        (active_delinq_days < 90)) then
        days_past_due_domain = '4'; --60 дней до 89 дней просрочки
    else
    if ((active_delinq_days >= 90) and
        (active_delinq_days < 120)) then
        days_past_due_domain = '5'; --90 дней  до 119 дней просрочки
    else
    if ((active_delinq_days >= 120) and
        (active_delinq_days < 150)) then
        days_past_due_domain = '6'; --90 дней  до 119 дней просрочки
    else
    if ((active_delinq_days >= 150) and
        (active_delinq_days < 180)) then
        days_past_due_domain = '7'; --90 дней  до 119 дней просрочки
    else
    if (active_delinq_days >= 180) then
        days_past_due_domain = '8'; --120 дней просрочки и более
    else
        days_past_due_domain = '0';
    suspend;
  end
end
