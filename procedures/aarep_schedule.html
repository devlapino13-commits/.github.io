CREATE PROCEDURE AAREP_SCHEDULE(
    K_ID DECIMAL(18,0),
    K_DATE DATE)
RETURNS (
    SH_ID INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_PERC_TAX DECIMAL(18,4),
    SH_ALL DECIMAL(18,4))
AS
declare variable k_nsp_percent decimal(18,7);
declare variable sh_kommis decimal(18,4);
declare variable k_disb_date date;
begin
  k_nsp_percent = 0;

  select k_disb_date
  from kontract
  where k_id = :k_id
  into :k_disb_date;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end

  /* Procedure Text */
  if (k_date = '30.12.1899') then
  begin
    for select sh_id, sh_date, sh_os, sh_perc, sh_perc_tax, sh_kommis
        from schedule
        where sh_from_k_id = :k_id and
              sh_id > 0
        order by sh_date
        into :sh_id, :sh_date, :sh_os, :sh_perc, :sh_perc_tax, :sh_kommis
    do
    begin
      sh_perc_tax = round((sh_perc + sh_kommis) * k_nsp_percent / 100);

      sh_all = :sh_os + sh_perc + sh_perc_tax;
      suspend;
    end
  end
  else
  begin
    sh_perc_tax = 0;
    for select ok_sh_id, ok_sh_date, sum(ok_sh_os), sum(ok_sh_perc), sum(ok_kommis)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_date <= :k_date and
              ok_ol_is_active = 1 and
              ok_type = 1 and
              ok_sh_id > 0 and
              ok_from_c_id = 0
        group by ok_sh_id, ok_sh_date
        order by ok_sh_date
        into :sh_id, :sh_date, :sh_os, :sh_perc, :sh_kommis
    do
    begin
      sh_perc_tax = round((sh_perc + sh_kommis) * k_nsp_percent / 100);

      sh_all = :sh_os + sh_perc + sh_perc_tax;
      suspend;
    end
  end
end
