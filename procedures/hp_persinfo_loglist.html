CREATE PROCEDURE HP_PERSINFO_LOGLIST(
    C_ID D_ID,
    IN_SOURCE D_STR_CODE = null,
    IN_FIELD_NAME D_STR_CODE = null)
RETURNS (
    PVA_ID D_ID,
    PVA_DATETIME D_DATETIME,
    PIF_SOURCE D_STR_CODE,
    PIF_FIELD_NAME D_STR_CODE,
    PIF_CAPTION D_STR_CODE,
    PIF_POSITION D_INTEGER,
    PVA_TEXT_VALUE D_STR_255,
    PVA_IS_SIGNED D_BOOL,
    PED_FROM_EDL_ID D_ID,
    EDL_CAPTION D_STR_100,
    EDL_CREATE_DATETIME D_DATETIME,
    UN_FIO D_STR_100)
AS
declare variable ped_pva_id d_id;
declare variable un_fio_hide boolean;
begin
  if (c_id is null) then
    exit;

  if (coalesce(in_source, '') = '') then
    in_source = null;

  if (coalesce(in_field_name, '') = '') then
    in_field_name = null;

  un_fio_hide = not exists(
    select *
    from hp_permission_access_check(
      uxf$user(),
      'PermActions',
      0,
      'FinStart.PersInfoLog.ViewFIO')
    where has_access = 1);

  for select pva_id, pva_datetime,
             pif_source, pif_field_name, pif_caption, pif_position,
             ped_pva_id, ped_from_edl_id, c_name_fio
  from persinfo
       inner join persdata on (pda_from_pin_id = pin_id)
       inner join ux$persfield on (pda_from_pif_uuid = pif_uuid and
                                   coalesce(:in_source, pif_source) = pif_source and
                                   coalesce(:in_field_name, pif_field_name) = pif_field_name)
       inner join persvalue on (pva_from_pda_id = pda_id)
       left join ux$users on (usr_id = pva_usr_id)
       left join clients on (c_id = usr_from_c_id)
       left join persedoc on (pva_id = ped_pva_id)
  where pin_object_kind = 0 and
        pin_object_id = :c_id
  into pva_id, pva_datetime,
       pif_source, pif_field_name, pif_caption, pif_position,
       ped_pva_id, ped_from_edl_id, un_fio
  do
  begin
    pva_text_value = pif$id_val(pva_id);
    pva_is_signed = iif(ped_pva_id is null, 0, 1);

    edl_caption = '';
    edl_create_datetime = null;

    if (ped_from_edl_id is not null) then
    begin
      select edc_name, edl_create_datetime
      from edoc_list, edoc_cfg
      where edl_id = :ped_from_edl_id and
            edl_from_edc_id = edc_id
      into edl_caption, edl_create_datetime;
    end

    if (un_fio_hide) then
      un_fio = null;

    suspend;
  end
end
