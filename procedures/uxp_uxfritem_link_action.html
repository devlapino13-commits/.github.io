CREATE PROCEDURE UXP$UXFRITEM_LINK_ACTION(
    UX_UUID D_UUID,
    CURRENT_UFI_UUID D_UUID,
    UA_UUID D_UUID)
AS
declare variable ufi_kind d_integer;
declare variable ufi_control d_str_code;
begin
  -- Прикрепляет действие в редактору
  if (trim(coalesce(current_ufi_uuid, '')) = '') then
    Exit;

  select ufi_control, ufi_kind
  from ux$fritem
  where ufi_owner = :ux_uuid and
        ufi_uuid = :current_ufi_uuid
  into ufi_control, ufi_kind;

  if (row_count = 0) then
    exit;

  if (not exists(select * from ux$action
                 where ua_uuid = :ua_uuid)) then
    exit;

  if (ufi_control in ('buttonedit', 'dialogedit')) then
    exception e_message 'Нельзя заменять действие в данном редакторе';

  if (ufi_kind not in (0, 2)) then
    exception e_message 'Нельзя добавлять действие в данный элемент';

  update or insert into ux$frbutton (ufb_ufi_uuid, ufb_ua_uuid)
  values (:current_ufi_uuid, :ua_uuid)
  matching (ufb_ufi_uuid);
end
