CREATE PROCEDURE HP_MIRAPOLIS(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    C_ID ID_TYPE,
    PL_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    C_SEX_TEXT VARCHAR(50),
    C_BIRTH_DATE DATE,
    C_PLASTCARD_NO VARCHAR(100),
    C_EMAIL VARCHAR(255),
    REG_CODE VARCHAR(100),
    REG_NAME VARCHAR(100),
    JL_NAME VARCHAR(100),
    C_UN_NAME VARCHAR(30),
    M_UN_NAME VARCHAR(30),
    C_HIRED_DATE DATE,
    C_HO_DATE DATE,
    C_UN_ID ID_TYPE,
    M_UN_ID ID_TYPE,
    C_STATUS VARCHAR(10),
    C_UN_TRILLIONER_PASS VARCHAR(30),
    FIRSTNAME VARCHAR(100),
    LASTNAME VARCHAR(100),
    PATRONOMIC VARCHAR(100),
    M_C_ID ID_TYPE)
AS
declare variable reg_address_fact d_id;
declare variable reg_id d_id;
declare variable ho_status_date d_date;
declare variable c_id_temp d_id;
declare variable mirapolis_mgr d_id;
begin
  mirapolis_mgr = 1010100780677; -- Джамалова Загра Загидовна

  for select re.reg_address_fact, reg_unicode, re.reg_code, re.reg_name
      from region_list(:report_region_id, 1, 3, 1), region re
      where id_list = reg_unicode
      into reg_address_fact, reg_id, reg_code, reg_name
  do
  begin
    c_id_temp = -1;
    for with tab
        as (select ps_from_c_id, c_name_fio, c_sex_text, c_birth_date,
                   c_plastcard_no, c_email, jl_name, c_hired_date,
                   iif(ps_curr_status = 'Декретный отпуск', 'Неактивный', 'Активный') c_status,
                   null as ho_status_datetime
            from aarep_personal_list(:reg_id, :report_date_end, 1)

            union

            select ps_from_c_id, c_name_fio, c_sex_text, c_birth_date,
                   c_plastcard_no, c_email, jl_name, c_hired_date,
                   'Неактивный', null
            from aarep_personal_list(:reg_id, :report_date_end, 0)
            where ps_date_begin >= :report_date_end - 14

            union

            select ps_from_c_id, c_name_fio,
                   iif(c_sex = 1, 'Мужчина', 'Женщина') as c_sex_text,
                   c_birth_date, c_plastcard_no, c_email, jl_name,
                   ps_date_begin, 'Неактивный', ho_status_datetime
            from hr_pstatus ps1, hr_orders, clients, joblist
            where ps_from_ho_id = ho_id and
                  ps_status = -1 and
                  cast(ho_status_datetime as date) between :report_date_end - 14 and :report_date_end
                  -- на дату аннулирования нужно проверить сотрудник работает или нет, если не работает то только тогда нужно отправлять
                  and
                  not exists(select *
                             from hr_pstatus ps2
                             where ps2.ps_from_c_id = ps1.ps_from_c_id and
                                   ps2.ps_is_working = 1 and
                                   ps2.ps_status = 1 and
                                   cast(ho_status_datetime as date) between ps2.ps_date_begin and ps2.ps_date_end) and
                  ps_from_c_id = c_id and
                  ps_from_jl_id = jl_id and
                  ps_from_reg_id = :reg_id)
        select ps_from_c_id, c_name_fio, c_sex_text, c_birth_date,
               c_plastcard_no, c_email, jl_name, c_hired_date, c_status,
               ho_status_datetime
        from tab
        where (ps_from_c_id = :mirapolis_mgr and c_status = 'Активный') or
              jl_name not in ('Специалист службы безопасности', 'Водитель')
        order by 1, 8 desc
        into c_id, c_name_fio, c_sex_text, c_birth_date, c_plastcard_no,
             c_email, jl_name, c_hired_date, c_status, ho_status_date
    do
    begin
      if (c_id = c_id_temp) then
        continue;
      c_id_temp = c_id;

      c_plastcard_no = coalesce(c_plastcard_no, '');
      c_email = coalesce(c_email, '');

      c_un_name = null;
      c_un_id = null;
      c_un_trillioner_pass = null;

      select un_name, un_id, un_trillioner_pass
      from username
      where un_from_c_id = :c_id
      order by un_is_active desc
      rows 1
      into c_un_name, c_un_id, c_un_trillioner_pass;

      if (c_un_name is null) then
      begin
        select left(fl_name2, 1) || fl_name1 || extract(day from c_birth_date)
        from clients, fiolist
        where c_id = :c_id and
              c_alternat_fl_id = fl_id
        into c_un_name;
        c_un_name = replace(c_un_name, '-', '');
        c_un_name = replace(c_un_name, '''', '');
      end
      if (coalesce(c_un_trillioner_pass, '') = '') then
        c_un_trillioner_pass = c_un_name || extract(day from c_birth_date);

      select name_first, name_last, name_middle
      from hp_fio_clear_ru(:c_name_fio)
      into firstname, lastname, patronomic;

      m_un_name = null;
      m_un_id = null;
      m_c_id = null;

      -- Для мега-офиса данные менеджера берем из loanmgrlist
      select un_name, un_id, un_from_c_id
      from krexpert
      left join loanmgrlist on ke_from_lml_id = lml_id
      left join username on un_from_c_id = lml_from_c_id
      where ke_from_p_id = :c_id and
            ke_from_reg_id = :reg_id and
            ke_date <= :report_date_end
      order by ke_date desc
      rows 1
      into m_un_name, m_un_id, m_c_id;

      if (m_c_id is null) then
      begin
        select un.un_name, un.un_id, un_from_c_id
        from hr_pstatus ps, joblist, jobkind, username un
        where ps_from_reg_id = :reg_id and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              :report_date_end between ps_date_begin and ps_date_end and
              ps_status = 1 and
              ps_is_working = 1 and
              ps.ps_from_c_id = un.un_from_c_id and
              ps.ps_from_c_id not in (select hrv.hrv_from_c_id
                                      from hr_orders ho, hr_voctravel hrv
                                      where ps.ps_from_c_id = hrv.hrv_from_c_id and
                                            hrv.hrv_from_ho_id = ho_id and
                                            ho_order_type = 16 and
                                            :report_date_end between hrv.hrv_date_begin and hrv.hrv_date_end) and
              ((jk_system_code = 'JobSystem.Region_KM' and
              (jl_name = 'Менеджер офиса' or
              jl_name containing 'менеджера' or
              jl_name containing 'Старший')) or
              jk_system_code = 'JobSystem.Region_Director' or
              ps_from_c_id = 1010100210250) rows 1
        into m_un_name, m_un_id, m_c_id;
      end

      if (m_c_id is null) then
      begin
        select un_name, un_id, un_from_c_id
        from username
        where un_from_c_id = :mirapolis_mgr
        into m_un_name, m_un_id, m_c_id;
      end

      c_ho_date = null;
      select ps_date_begin
      from hr_pstatus, hr_orders, sa_worktype
      where ps_status = 1 and
            ps_from_ho_id = ho_id and
            ho_order_type = sawt_id and
            ps_from_c_id = :c_id and
            sawt_base_type = 1
      order by ps_date_begin desc
      rows 1
      into c_ho_date;
      if (c_ho_date is null) then
        c_ho_date = ho_status_date;

      pl_name = null;
      select pl.pl_name
      from addrlist ad, places pl
      where ad.adl_from_pl_id = pl.pl_id and
            ad.adl_id = :reg_address_fact
      into pl_name;

      if (m_c_id is not null and
          reg_code is not null and
          reg_name is not null and
          jl_name is not null and
          c_hired_date is not null and
          c_ho_date is not null and
          c_un_name is not null and
          c_un_trillioner_pass is not null and
          c_id is not null) then
        suspend;
    end
  end
end
