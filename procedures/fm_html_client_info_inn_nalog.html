CREATE PROCEDURE FM_HTML_CLIENT_INFO_INN_NALOG(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable eil_status d_smallint;
declare variable eil_status_datetime d_datetime;
declare variable request_inn_nalog d_str_long;
declare variable inn_nalog_kk_search_request d_str_long;
declare variable inn_fill d_str_long;
declare variable eil_status_str d_str_code;
declare variable c_inn d_str_20;
declare variable inn_ru_source d_integer;
declare variable inn_ru_manual_input d_integer;
declare variable query_status_str d_str_100;
declare variable query_status_color d_str_20;
declare variable status_result d_str_255;
declare variable is_query_already_sended_today d_bool;
declare variable inn_received d_bool;
declare variable is_auto_querys_finished d_bool;
declare variable obm_memo d_blob_text;
declare variable acl_status d_integer;
declare variable is_finished d_smallint;
declare variable can_query_inn d_bool;
begin
  inn_ru_source = f_get_const('inn_ru_source');
  inn_ru_manual_input = f_get_const('inn_ru_manual_input');
  inn_nalog_KK_search_request = '';

  res_text = '<b>Статус запрос ИНН в России:</b><br>';

  select eil.eil_status, eil.eil_status_datetime
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id and
        eic.eic_code = 'inn_nalog'
  where eil.eil_object_kind = f_objkind('clients', 'id') and
        eil.eil_object_id = :c_id
  order by eil.eil_create_datetime desc
  rows 1
  into eil_status, eil_status_datetime;

  --Получение Статуса
  select query_status_str,
         query_status_color,
         inn_received,
         is_auto_querys_finished,
         is_finished,
         can_query_inn
  from inn_query_status(:c_id)
  into query_status_str,
       query_status_color,
       inn_received,
       is_auto_querys_finished,
       is_finished,
       can_query_inn;

  select is_query_already_sended_today,
         acl_status,
         obm_memo
  from hp_get_actionlist_task_state('ActionGroup.Manual_Inn_Check', 0, :c_id)
  into is_query_already_sended_today,
       acl_status,
       obm_memo;

  if (inn_ru_source = 0) then
  begin
    select cl.c_inn
    from clients cl
    where cl.c_id = :c_id
    into c_inn;
  end
  else
  begin
    select icm.icm_inn
    from inn_check_main icm
    where icm.icm_from_c_id = :c_id
    into c_inn;
  end

  c_inn = coalesce(c_inn, '');

  eil_status_str = case
                     when eil_status = 2 then 'Успешно обработан'
                     when eil_status between 0 and 1 then 'В обработке'
                     when eil_status < 0 then 'Возникла ошибка'
                     else 'Запрос не отправлен'
                   end;

  status_result = f_format('<span style="margin: 0; font-weight: bold; color: %2;">%1</span>', coalesce(query_status_str, ''), coalesce(query_status_color, 'black'));
  res_text = blob_append(res_text, f_format('%1: %2 <br>', 'Статус', iif(inn_ru_source = 0, eil_status_str, status_result)));

  if(inn_ru_manual_input in (1,3)) then
  begin
    inn_fill = f_uri_sql('Внести ИНН вручную', 'inn_fill', 'clients', c_id, 'text', 'Введите ИНН', '', 4, 12, '', '', 'dialog', '');
    res_text = blob_append(res_text, f_format('%1: %2 <br>', 'Время обработки', f_datetimetostr(eil_status_datetime)));
  end
  else
    inn_fill = '';

  res_text = blob_append(res_text, f_format('%1: <b>%2</b>  <br>', 'ИНН',  iif(c_inn = '', 'Отсутствует', c_inn)));
  if(acl_status in (-1, -2, -3)) then
    res_text = blob_append(res_text, f_format('%1: <b>%2</b>  <br>', 'Коментарий КК',  obm_memo));

  request_inn_nalog = f_uri_sql('Запросить ИНН', 'inn_nalog_request', 'clients', :c_id, 'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', '');

  if((is_finished = 0 or acl_status in (0, 1)) and can_query_inn <> 1) then
     request_inn_nalog = '';

  --Если Получение ИНН по новой схеме И Запрос на получчение ИНН сегодня уже отправлен ИЛИ ИНН уже получен ИЛИ Задача на ручной поиск ИНН у КК
  if(inn_ru_source > 0 and (is_query_already_sended_today = 1 or inn_received = 1)) then
    request_inn_nalog = '';

  if((inn_received = 0 and is_finished = 1) and is_query_already_sended_today = 0 and acl_status is null) then
    inn_nalog_KK_search_request = f_uri_sql('Отправить на ручной поиск ИНН КК', 'inn_nalog_KK_search_request', 'clients', :c_id, 'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', '');
  if(acl_status is not null and acl_status not in  (0, 1) and inn_received <> 1) then -- RFCRU-3445
    inn_nalog_KK_search_request = f_uri_sql('Отправить на повторную ручную проверку КК', 'inn_nalog_KK_search_request', 'clients', :c_id, 'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', '');

  res_text = blob_append(res_text,  iif(request_inn_nalog <> '', request_inn_nalog || '<br>', request_inn_nalog), iif(inn_nalog_KK_search_request <> '', inn_nalog_KK_search_request || '<br>', inn_nalog_KK_search_request), iif(inn_fill <> '', inn_fill || '<br>', inn_fill));
  res_text = blob_append(res_text, '<br>');

  suspend;
end
