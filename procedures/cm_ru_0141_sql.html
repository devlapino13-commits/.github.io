CREATE PROCEDURE CM_RU_0141_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare variable DOC_CODE STR_100;
declare variable PK_NAME STR_100;
begin
  for select CL.C_NAME_FIO, PDT_CODE, PK.PK_NAME
      from KNCLIENT KNCL
      inner join CLIENTS CL on KNCL.KC_FROM_C_ID = CL.C_ID
      inner join CLIENTDOCS CD
      inner join PASSKIND PK on CDS_FROM_PK_ID = PK_ID
      inner join PASSDOCTYPE on PDT_ID = PK_FROM_PDT_ID on (CDS_FROM_C_ID = KC_FROM_C_ID and
            CD.CDS_IS_DELETE = 0 and
            current_date < CDS_DOC_DATE_BEGIN and
            current_date < CDS_DOC_DATE_END)
      where KNCL.KC_AMOUNT_APPROVED > 0 and
            KNCL.KC_FROM_K_ID = :K_ID
      into :res_info_text, DOC_CODE, PK_NAME
  do
  begin
    res_error_is_critical = 0;
    res_error_text = 'Дата выдачи документа больше текущей даты. Исправьте дату документа: ' || PK_NAME;
    suspend;
  end --Ограничения по дате документа (не позднее даты ввода)
end
