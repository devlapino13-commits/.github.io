CREATE PROCEDURE HP_SEND_KONTRACT_TO_KK(
    K_ID ID_TYPE,
    UN_ID ID_TYPE)
RETURNS (
    ACL_ID D_ID,
    WAIT_STATUS D_SMALLINT,
    RES_TEXT D_STR_LONG)
AS
declare variable k_status d_smallint;
declare variable acl_status d_integer;
declare variable rd_id d_integer;
declare variable auto_approve d_bool;
declare variable cs_info_state d_bool;
declare variable pr_id d_integer;
declare variable k_disb_date d_date;
declare variable c_id d_id;
declare variable db_id d_id;
begin
  wait_status = 0;
  res_text = '';
  db_id = f_get_db_id();

  select k_status, k_from_pr_id, k_disb_date
  from kontract kont
  where k_id = :k_id
  into :k_status, pr_id, k_disb_date;

  if (k_status = -2) then
  begin
    wait_status = 1;
    res_text = 'Заявка еще не переведена в договор';
  end

  if (k_status = -1) then
  begin
    wait_status = 1;
    res_text = 'Договор аннулирован, обратитесь к КОБД';
  end

  if (k_status = 1) then
    res_text = 'Договор уже утвержден, нет необходимости повторной отправки';

  if (k_status = 0) then
  begin
    select rd_id
    from hp_get_check_loan_data(:k_id)
    into:rd_id;


    /*select rd_id
    from refdata
    where rd_code = 'ActionGroup.Check_Loan_Data'
    into rd_id; */

    select acl_id, acl_status
    from actionlist
    where acl_object_kind = 9 and
          acl_object_id = :k_id and
          acl_from_rd_id = :rd_id
    order by acl_id desc
    rows 1
    into acl_id, acl_status;

    if (acl_id is null) then
    begin
      -- Отправка на КК
      select acl_id, res_message
      from hp_actionlist_object_insert('kontract', :k_id, 'actiongroup.check_loan_data', :un_id)
      into acl_id, :res_text;

      -- Запись не создан, возвращаем как ошибку
      if (res_text <> '') then
      begin
        wait_status = -1;
        acl_status = -1;
      end
      else
        acl_status = 0;
    end

    if (wait_status = 0 and acl_status in (0, 1)) then
    begin
      auto_approve = 1;

      -- Если боевая база, проверяем основные условия.
      -- Иначе подтверждаем автоматически
      if (exists(select * from hp_db_is_actual)) then
      begin
        for select kc_from_c_id, cs_info_state
        from knclient, clientstate
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = cs_from_c_id
        into c_id, cs_info_state
        do
        begin
          cs_info_state = 1; -- Отмечаем что данные не менялись

          if (cs_info_state = 0) then
            auto_approve = 0;

          if (auto_approve = 1) then
          begin
            if (f_docprloan_valid(c_id, pr_id, 0, k_disb_date) = -1) then
              auto_approve = 0;
          end
        end
      end

      if (auto_approve = 1) then
      begin
        update actionlist
        set acl_status = 2,
            acl_respons_c_id = iif(:db_id = 81001, 1000000000001 ,1000000000000),
            acl_from_un_id = 1000000000000
        where acl_id = :acl_id;
      end
    end
  end
  suspend;
end
