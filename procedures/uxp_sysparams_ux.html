CREATE PROCEDURE UXP$SYSPARAMS_UX
RETURNS (
    UX_UUID D_UUID)
AS
declare variable us_uuid d_uuid;
declare variable us_code d_str_code;
declare variable ufr_uuid d_uuid;
declare variable ufm_uuid d_uuid;
begin
  us_code = 'sys$params';

  select us_uuid
  from ux$source
  where us_code = :us_code
  into us_uuid;

  if (us_uuid is null) then
  begin
    insert into ux$source (us_code, us_name, us_is_table, us_kind)
    values (:us_code, 'Параметры отчетов', 0, 2) -- Создаем виртуальный источник
    returning us_uuid
    into :us_uuid;
  end

  select ux_uuid
  from ux$code
  where ux_us_uuid = :us_uuid
  into ux_uuid;

  if (ux_uuid is null) then
  begin
    select ux_uuid
    from uxp$uxuniq_insert(:us_uuid, 'uxform')
    into ux_uuid;

    insert into ux$code (ux_uuid, ux_us_uuid, ux_name)
    values (:ux_uuid, :us_uuid, 'Форма отчетов');
  end

  select ufm_uuid
  from ux$form
  where ufm_ux_uuid = :ux_uuid
  into ufm_uuid;

  if (ufm_uuid is null) then
  begin
    insert into ux$form (ufm_uuid, ufm_ux_uuid)
    values (:ux_uuid, :ux_uuid)
    returning ufm_uuid
    into :ufm_uuid;
  end

  select ufr_uuid
  from ux$frame
  where ufr_owner = :ufm_uuid
  into ufr_uuid;

  if (ufr_uuid is null) then
  begin
    insert into ux$frame (ufr_uuid, ufr_owner, ufr_us_uuid, ufr_code, ufr_is_grid)
    values (:ufm_uuid, :ufm_uuid, :us_uuid, '', 0);
  end

  suspend;
end
