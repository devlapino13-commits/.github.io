CREATE PROCEDURE HP_APPEND_ARR_FROM_JSON(
    ARR_NAME D_STR_20,
    JSON_INP D_BLOB_TEXT,
    NEW_VALUE D_BLOB_TEXT,
    CLEAR_SCOPES D_SMALLINT = 0)
RETURNS (
    RESULT D_BLOB_TEXT)
AS
declare variable array d_blob_text;
begin
  array = f_json_blob_get_value(:json_inp, :arr_name, '');
  if (array <> '') then
  begin
    array = replace(array, ']', ',' || :new_value || ']');
    result = f_json_blob_set_value(:json_inp, :arr_name, :array);
  end
  else
  begin
    array = '[' || :new_value || ']';
    result = overlay(:json_inp placing ', "' || :arr_name || '":' || :array || '}' from char_length(:json_inp));
  end
  if (clear_scopes = 1) then
  begin
     result = overlay(:result placing '' from 1 for 1);
     result = overlay(:result placing '' from char_length(:result) for char_length(:result));
  end
  suspend;
end
