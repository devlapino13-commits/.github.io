CREATE PROCEDURE GET_OPERCARD_LIST(
    REP_DATE DATE,
    ONLY_NEW INTEGER)
RETURNS (
    OPC_ID INTEGER,
    OPC_DATE DATE,
    OPC_FROM_REG_ID INTEGER,
    OPC_FROM_OL_ID DECIMAL(18,0),
    OPC_FROM_K_ID DECIMAL(18,0),
    OPC_FROM_C_ID DECIMAL(18,0),
    OPC_FROM_PSI_ID DECIMAL(18,0),
    OPC_AMOUNT DECIMAL(18,4),
    OPC_STATUS INTEGER,
    OPC_MEMO VARCHAR(255),
    OPC_FROM_UN_ID DECIMAL(18,0),
    OPC_DATETIME TIMESTAMP,
    OPC_KONTRACT_NO VARCHAR(20),
    OPC_DISB_DATE DATE,
    OPC_REGION_NAME VARCHAR(100),
    OPC_CLIENT_NAME VARCHAR(100),
    OPC_CARD_NO VARCHAR(20),
    OPC_CARD_DATE_END DATE,
    OPC_POCKET_ID INTEGER)
AS
begin
  for select opc_id, opc_date, opc_from_reg_id, opc_from_ol_id, opc_from_k_id, opc_from_c_id, opc_from_psi_id,
             opc_amount, opc_status, opc_memo, opc_from_un_id, opc_datetime, opc_pocket_id
      from opercard
      where opc_date = :rep_date and
            ((:only_new = 1 and
            opc_status <> 2) or
            (:only_new = 0))
      into :opc_id, :opc_date, :opc_from_reg_id, :opc_from_ol_id, :opc_from_k_id, :opc_from_c_id, :opc_from_psi_id,
           :opc_amount, :opc_status, :opc_memo, :opc_from_un_id, :opc_datetime, :opc_pocket_id
  do
  begin
    opc_kontract_no = '';
    opc_disb_date = null;
    opc_region_name = '';
    opc_client_name = '';
    opc_card_no = 'Не найдена';

    select c_name_fio
    from clients
    where c_id = :opc_from_c_id
    into opc_client_name;

    select k_kontract_no, k_disb_date
    from kontract
    where k_id = :opc_from_k_id
    into opc_kontract_no, opc_disb_date;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :opc_from_reg_id
    into opc_region_name;

    select psi_code, psi_date_end
    from paysys_item
    where psi_id = :opc_from_psi_id
    into :opc_card_no, :opc_card_date_end;

    suspend;
  end
end
