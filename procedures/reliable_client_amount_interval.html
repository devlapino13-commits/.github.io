CREATE PROCEDURE RELIABLE_CLIENT_AMOUNT_INTERVAL(
    C_ID D_ID,
    PR_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    AMOUNT_MIN D_INTEGER,
    AMOUNT_MAX D_INTEGER,
    POINT_CL D_INTEGER)
AS
declare variable pr_code d_str_100;
begin
  point_cl = f_point(c_id, rep_date);

  select amount_min, amount_max
  from perc_amount_interval_by_point(:pr_id, :point_cl, :rep_date)
  into :amount_min, :amount_max;

  suspend;
end
