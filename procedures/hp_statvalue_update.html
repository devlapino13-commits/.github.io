CREATE PROCEDURE HP_STATVALUE_UPDATE(
    STI_ID D_INTEGER NOT NULL,
    STAT_NAME D_STR_255 NOT NULL,
    PL_ID D_ID,
    PERIOD_BEGIN D_INTEGER NOT NULL,
    NEW_VALUE D_CURRENCY NOT NULL,
    DATE_BEGIN D_DATE,
    UN_ID D_ID)
AS
declare variable str_id d_integer;
declare variable stv_id d_id;
declare variable stv_value d_currency;
declare variable str_from_pl_id d_id;
declare variable stp_id d_integer;
begin
  -- Процедура добавляет или обновляет значение индикатора по его ID и названии территории

  if (stat_name is null) then exit;
  stat_name = trim(stat_name);

  -- Получаем по названию ID территории
  select str_id, str_from_pl_id
  from statregion
  where str_name = :stat_name
  into :str_id, str_from_pl_id;

  -- Если нет, добавляем новую
  if (str_id is null) then
  begin
    insert into statregion (str_name, str_from_pl_id)
    values (:stat_name, :pl_id)
    returning str_id
    into :str_id;
  end else
  begin
    if (coalesce(str_from_pl_id, -1) <> coalesce(pl_id, -1) and
        pl_id is not null) then
    begin
      update statregion
      set str_from_pl_id = :pl_id
      where str_id = :str_id;
    end
  end

  select stp_id
  from statpocket
  where stp_from_sti_id = :sti_id and
        stp_period_begin = :period_begin and
        stp_date_begin = :date_begin
  into stp_id;

  if (stp_id is null) then
  begin
    insert into statpocket (stp_from_sti_id, stp_period_begin, stp_date_begin, stp_from_un_id)
    values (:sti_id, :period_begin, :date_begin, :un_id)
    returning stp_id
    into :stp_id;
  end

  select stv_id, stv_value
  from statvalue
  where stv_from_stp_id = :stp_id and
        stv_from_str_id = :str_id
  into stv_id, stv_value;

  if (stv_id is not null) then
  begin
    -- если есть совпадающая строка и значение отличается, обновляем
    if (stv_value <> new_value) then
    begin
      update statvalue
      set stv_value = :stv_value
      where stv_id = :stv_id;
    end
  end else
  begin
    -- если нет строки, добавляем новую
    insert into statvalue (stv_from_stp_id, stv_from_str_id, stv_value)
    values (:stp_id, :str_id, :new_value);
  end
end
