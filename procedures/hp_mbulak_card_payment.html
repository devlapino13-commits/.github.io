CREATE PROCEDURE HP_MBULAK_CARD_PAYMENT(
    K_ID D_ID,
    OL_ID D_ID)
AS
begin
  if (f_get_db_id() not in (39801, 41701)) then exit;

  -- если это кредит по карте МБулак
  if (exists(
    select *
    from kontract kont
    where kont.k_id = :k_id
      and kont.k_disb_kind = 6)) then
  begin
    -- если карта не активна
    if (not exists(select * from hp_client_bank_card_mb(-:k_id, current_date, 0)))
    then exception e_message 'Не определена активная карта МБулак.';

    -- меняем счёт переплаты (если она есть)
    update operbook set
      ob_account_kt = left(ob_account_kt, 14) || '1101'
    where ob_from_ol_id = :ol_id
      and ob_short_acc_kt starting '21199/';

    -- если было списание по закрытию займа, то ничего не делаем
    /*
    if (exists(
      select * from cardsync
      where cs_from_k_id = :k_id
        and cs_from_csc_id = f_get_cardsynccfg_id('write-off_on_closing')))
    then exit;
    */

    -- если это плата комиссии за карточное обслуживание
    if (exists(
      select *
      from operkred ok
      where ok.ok_from_ol_id = :ol_id
        and ok.ok_from_k_id = :k_id
        and ok.ok_type = 4
        and ok.ok_ol_is_active = 1
        and ok.ok_kommis < 0))
    then
      -- комиссия за карточное обслуживание
      execute procedure card_command_adv_perc_writeoff(f_objkind('kontract', 'id'), :k_id, :ol_id);
    else
      -- изменение кредитного лимита по факту платежа
      execute procedure card_payment_card_limit_update(f_objkind('kontract', 'id'), :k_id, :ol_id);
  end
end
