CREATE PROCEDURE HP_GET_KN_OKSUMMARY_ZAD_INFO(
    NEW_OKS_FROM_KC_ID D_ID,
    NEW_OKS_DATE_BEGIN D_DATE,
    NEW_OKS_PERCENT_ESP D_PERCENT)
RETURNS (
    RES_OKS_BAL_ESP_NORM D_CURRENCY,
    RES_SH_COUNT D_INTEGER,
    RES_DELINQ_DATE D_DATE,
    RES_PLAN_SH_DATE D_DATE,
    RES_VBS_FINAL D_CURRENCY,
    RES_VBS_STOPPED D_CURRENCY,
    RES_EAD_LAST_SH_DATE D_DATE,
    RES_STR_VBS_NORMAL D_STR_LONG,
    RES_STR_VBS_DATES D_STR_LONG,
    RES_BAL_ZAD_OS D_CURRENCY,
    RES_BAL_ZAD_PERC D_CURRENCY,
    RES_SH_PERC D_CURRENCY,
    RES_PROLONG_DATE_CURR D_DATE,
    RES_PROLONG_DATE_MAX D_DATE,
    MAX_DELINQ_DAYS D_INTEGER,
    RES_BAL_MOVED_OS D_CURRENCY,
    RES_RESH_DELINQ D_CURRENCY)
AS
declare variable ok_sh_date d_date;
declare variable amount_begin d_currency;
declare variable amount_final d_currency;
declare variable amount_zad_os d_currency;
declare variable amount_zad_perc d_currency;
declare variable sh_perc d_currency;
declare variable has_max_restruc d_integer;
declare variable has_curr_restruc d_integer;
declare variable tmp_delinq_days d_integer;
declare variable moved_sh_id d_integer;
declare variable tmp_moved_os d_currency;
declare variable tmp_resh_delinq d_currency;
begin
-- Вычисляем задолженность по графику, дата просрочки и плановую
  res_vbs_final = 0;
  res_vbs_stopped = 0;
  res_oks_bal_esp_norm = 0;

  amount_begin = 0;
  amount_final = 0;
  amount_zad_os = 0;
  amount_zad_perc = 0;

  res_sh_count = 0;
  res_str_vbs_dates = '';
  res_str_vbs_normal = '';
  res_plan_sh_date = null;
  res_delinq_date = null;

  res_bal_zad_os = 0;
  res_bal_zad_perc = 0;
  res_sh_perc = 0;
  max_delinq_days = 0;
  res_bal_moved_os = 0; -- пересенный платеж ОС, просроченный который стал не просроченным
  tmp_resh_delinq = 0; -- пересенный платеж через реструктуризацию
  res_resh_delinq = 0; -- пересенный платеж через реструктуризацию

  for select ok_sh_date,
             max(iif(ok_sh_date >= '01.03.2020' and ok_sh_date > :new_oks_date_begin and ok_type = 2 and ol_oper_type = 1 and ol_oper_subtype = 9 and ok_sh_os > 0, ok_sh_id, 0)),
             sum(iif(ok_ol_date < :new_oks_date_begin or ok_type = 1, ok_os + ok_perc + ok_sh_os + ok_sh_perc, 0.0)),
             sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc), sum(ok_sh_perc),
             sum(iif(ok_type = 2 and (ok_sh_os <> 0 or ok_sh_perc <> 0), 1, 0)),
             sum(iif(ok_type = 2 and (ok_sh_os <> 0 or ok_sh_perc <> 0) and ok_ol_date = :new_oks_date_begin, 1, 0)),
             max(iif(ok_type in (4, 5, 6) and (ok_os < 0 or ok_perc < 0) and ok_ol_date > ok_sh_date,
                 datediff(day, ok_sh_date, ok_ol_date), 0)),
             sum(iif(ok_type = 2 and ok_ol_date = :new_oks_date_begin and
                     ok_sh_date < ok_ol_date and (ok_sh_os < 0 or ok_sh_perc < 0), -(ok_sh_os + ok_sh_perc), 0))
      from kn_operkred, operlist
      where ok_from_kc_id = :new_oks_from_kc_id and
            ok_type < 7 and
            ok_ol_date <= :new_oks_date_begin and
            ok_ol_is_active = 1 and
            ok_from_ol_id = ol_id
      group by ok_sh_date
      into ok_sh_date, moved_sh_id, amount_begin, amount_zad_os, amount_zad_perc, sh_perc, has_max_restruc, has_curr_restruc, tmp_delinq_days, tmp_resh_delinq
  do
  begin
    if (moved_sh_id is null) then
      moved_sh_id = 0;

    if (tmp_resh_delinq is null) then
      tmp_resh_delinq = 0;

    if (tmp_resh_delinq <> 0) then
    begin
      if (not exists(select * from knclient, operrest
      where kc_id = :new_oks_from_kc_id and
            kc_from_k_id = or_from_k_id and
            or_ol_date = :new_oks_date_begin and
            or_ol_is_active = 1)) then
        tmp_resh_delinq = 0;
    end

    amount_final = amount_zad_os + amount_zad_perc;
    if (tmp_delinq_days > max_delinq_days) then
      max_delinq_days = tmp_delinq_days;

    if (ok_sh_date = new_oks_date_begin) then
      res_sh_perc = sh_perc;

    if (has_max_restruc > 0) then
      res_prolong_date_max = ok_sh_date;
    if (has_curr_restruc > 0) then
      res_prolong_date_curr = ok_sh_date;

    res_resh_delinq = res_resh_delinq + tmp_resh_delinq;

    if (ok_sh_date <= new_oks_date_begin) then
    begin
      if (res_delinq_date is null and
          amount_final > 0) then
        res_delinq_date = ok_sh_date; -- дата просрочки

      res_bal_zad_os = res_bal_zad_os + amount_zad_os;
      res_bal_zad_perc = res_bal_zad_perc + amount_zad_perc;
    end else
    begin
      -- если не просроченный платеж был перенесен из-за выходных,
      -- то проверяем был ли он первоначально до даты отчета
      -- если да, считаем его как перенесенный и на него не будет начисляться процент как и просроченным
      if (moved_sh_id > 0 and amount_zad_os > 0 and new_oks_date_begin >= '31.03.2020') then
      begin
        tmp_moved_os = 0;

        if (exists(select *
        from kn_operkred, operlist
        where ok_from_kc_id = :new_oks_from_kc_id and
              ok_type = 2 and
              ok_sh_id = :moved_sh_id and
              ok_ol_date <= :new_oks_date_begin and
              ok_sh_date <= :new_oks_date_begin and
              ok_ol_is_active = 1 and
              ok_sh_os < 0 and
              ok_from_ol_id = ol_id and
              ol_oper_type = 1 and
              ol_oper_subtype = 9)) then
          tmp_moved_os = amount_zad_os;

        res_bal_moved_os = res_bal_moved_os + tmp_moved_os;
      end
    end

    if (ok_sh_date >= new_oks_date_begin) then
    begin
      if (res_plan_sh_date is null) then
        res_plan_sh_date = ok_sh_date; -- ближайшая дата погашения

      res_oks_bal_esp_norm = res_oks_bal_esp_norm + amount_begin / power(1.0 + cast(new_oks_percent_esp as double precision) / 100.0, cast(datediff(day, new_oks_date_begin, ok_sh_date) as double precision) / 365.0);

      res_vbs_final = res_vbs_final + amount_final / power(1.0 + cast(new_oks_percent_esp as double precision) / 100.0, cast(datediff(day, new_oks_date_begin, ok_sh_date) as double precision) / 365.0);
      res_vbs_stopped = res_vbs_stopped + amount_zad_os;

      if (amount_final <> 0) then
      begin
        res_ead_last_sh_date = ok_sh_date;
        res_sh_count = res_sh_count + 1;
        res_str_vbs_dates = res_str_vbs_dates || f_datetostr(ok_sh_date);
        if (amount_final >= 0) then
          res_str_vbs_normal = res_str_vbs_normal || lpad(amount_final, 20, '0');
        else
          res_str_vbs_normal = res_str_vbs_normal || '-' || lpad(-amount_final, 19, '0');
      end
    end
  end

  if (res_delinq_date is not null) then
    max_delinq_days = maxvalue(max_delinq_days, datediff(day, res_delinq_date, new_oks_date_begin));

  if (res_vbs_final is null) then
    res_vbs_final = 0;
  if (res_vbs_stopped is null) then
    res_vbs_stopped = 0;
  if (res_oks_bal_esp_norm is null) then
    res_oks_bal_esp_norm = 0;

  suspend;
end
