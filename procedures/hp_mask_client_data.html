CREATE PROCEDURE HP_MASK_CLIENT_DATA(
    BLS_ID D_ID)
RETURNS (
    FINISH_JSON_TEXT D_BLOB_TEXT)
AS
declare variable jso d_integer;
declare variable db_id d_id;
declare variable json_text d_blob_text;
declare variable eic_code d_str_255;

declare function mask_field(
    p_json d_integer,
    p_path d_str_255)
returns d_integer
as
declare variable dummy d_integer;
begin
  if (json.contains(p_json, p_path) = 1) then
    dummy = json.obj_set_s(p_json, p_path, 'xxx');
  return 1;
end

begin
  select cast(hp.blx_blob as d_blob_text)
  from hp_get_blobex_blob(:bls_id) hp
  into :json_text;

  jso = json.parse(:json_text);

  if (jso is null) then
    exception e_message('не валидный json');

  db_id = f_get_db_id();

  select eic.eic_code
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id =eil.eil_from_eic_id
  where eil.eil_from_bls_id = :bls_id
  into :eic_code;

  if (db_id = 81001) then
  begin
    if (:eic_code = 'equifax_40') then
    begin
      mask_field(jso, 'response.titlePart');
      mask_field(jso, 'response.basePart');
    end
    else if (:eic_code = 'nbki_40') then
    begin
      mask_field(jso, 'prequest.req');
      mask_field(jso, 'preply.report.person_reply');
      mask_field(jso, 'preply.report.id_reply');
      mask_field(jso, 'preply.report.address_reply');
      mask_field(jso, 'preply.report.address_reply_rutdf');
      mask_field(jso, 'preply.report.phone_reply');
      mask_field(jso, 'preply.report.contact_reply');
      mask_field(jso, 'preply.report.contact_reply');
    end 
    else if (:eic_code = 'equifax') then
    begin
      mask_field(jso, 'response.basePart.addrFact');
      mask_field(jso, 'response.basePart.addrReg');
      mask_field(jso, 'response.basePart.addrRequests');
      mask_field(jso, 'response.basePart.historyAddr');
      mask_field(jso, 'response.titlePart._private');

    end
    else if (:eic_code = 'nbki') then
    begin
      mask_field(jso, 'product.preply.report.addressReply');
      mask_field(jso, 'product.preply.report.idReply');
      mask_field(jso, 'product.preply.report.inqControlNum');
      mask_field(jso, 'product.preply.report.personReply');
      mask_field(jso, 'product.prequest.req');

    end
    else
      exception e_message('проект не найден');
  end
  else if (db_id = 41701) then
  begin
    if (:eic_code = 'Ishenim') then
    begin
      mask_field(jso, 'individual');
      mask_field(jso, 'reportInfo.requestedBy');
      mask_field(jso, 'currentRelations');
    end
    else if (:eic_code = 'ss_cib_report') then
    begin
      mask_field(jso, 'other_element.monitoring_enquiry.enquired_data.subject');
      mask_field(jso, 'other_element.monitoring_enquiry.credit_report.subject_info');
    end
    else
      exception e_message('проект не найден');
  end
  else if (:db_id = 39801) then
  begin
    mask_field(jso, 'Report.Root.Header');
    mask_field(jso, 'Report.Root.SubjectDetails');
    mask_field(jso, 'Report.Root.SubjectsAddress');
    mask_field(jso, 'Report.Root.IdentificationDocuments');
  end

  -- собираем итоговый json
  finish_json_text = json.obj_to_json(jso);

  suspend;

  when any do
  begin
    finish_json_text = null;
    suspend;
  end
end
