CREATE PROCEDURE DO_CLIENT_CIRCLE(
    CL_ID DECIMAL(18,0))
AS
declare variable cir_no integer;
declare variable amount decimal(18,4);
declare variable kc_id decimal(18,0);
declare variable kc_circle integer;
declare variable ks_state integer;
begin
  /* Procedure Text */
  cir_no = 1;
  for select kc_id, kc_circle, kc_amount_approved, ks_state
      from knclient, kontract, kontstate
      where k_id = kc_from_k_id and
            kc_from_c_id = :cl_id and
            ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            ks_date_end = '31.12.9999'
      order by k_disb_date, kc_from_k_id, kc_circle
      into :kc_id, :kc_circle, amount, ks_state
  do
  begin
    if (kc_circle is null) then
      kc_circle = 0;
    if (cir_no <> kc_circle) then
    begin
      update knclient
      set kc_circle = :cir_no
      where kc_id = :kc_id;
    end
    if (coalesce(amount, 0) > 0 and
        ks_state >= 1) then
      cir_no = :cir_no + 1;
  end
end
