CREATE PROCEDURE MC_ONLINE_APP_STATUS(
    MCL_ID D_ID,
    ONLINE_APP_ID D_INTEGER)
RETURNS (
    APP_STATUS D_STR_CODE)
AS
declare variable exa_id d_id;
declare variable oa_status d_smallint;
begin
  app_status = 'in_process';

  select oa.oa_from_exa_id, oa.oa_status
  from online_app oa
  where oa.oa_id = :online_app_id
  into exa_id, oa_status;

  --Заявка отменена, до создания записи в expressapp процесс не дошёл
  if (oa_status = -1) then
  begin
    app_status = 'app_canceled';
    suspend;
    exit;
  end

  --В процессе возникла ошибка
  if (exists(select exa_final_status
             from expressapp exa
             where exa.exa_id = :exa_id and
                   exa.exa_final_status < 0)) then
  begin
    app_status = 'app_error';
    suspend;
    exit;
  end

  --Процесс прошёл успешно, документы созданы, кредит готов к выдаче
  if (exists(select *
             from expresslog exl
             inner join expresscfg exc on exc.exc_id = exl.exl_from_exc_id and
                   exc.exc_action_code = 'loan_disb'
             where exl.exl_from_exa_id = :exa_id and
                   exl.exl_wait_status = 0)) then
    app_status = 'loan_docs_created';

  suspend;
end
