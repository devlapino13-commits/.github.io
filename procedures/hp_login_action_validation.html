CREATE PROCEDURE HP_LOGIN_ACTION_VALIDATION(
    ACTION_CODE D_STR_CODE,
    ACTION_DATA D_STR_LONG,
    IF_TEST_PHONE D_BOOL = 0)
RETURNS (
    RES_CODE D_STR_CODE,
    ERR_TEXT D_STR_255)
AS
declare variable mcl_login d_str_code;
declare variable birth_date_str d_str_code;
declare variable operating_system d_str_10;
declare variable application_version d_str_10;
declare variable mcl_project_text d_str_code;
declare variable is_biom d_smallint;
declare variable db_id d_integer;
declare variable pin_code d_integer;
declare variable mcl_pass_hash d_str_long;
declare variable db_is_actual d_smallint;
declare variable country d_str_code;
declare variable ip_address d_str_code;
declare variable is_local_ip d_bool;
declare variable is_test_phone d_bool;
declare variable country_condition d_bool;
begin
    /*
        RFCKG-4671
        Процедура отвечает за валидацию входящих параметров процедуры mc_login_action
    */

    if (action_code not in ('send_code', 'check_code', 'update_pass', 'login')) then
        exception e_message f_format('Недопустимое действие: %1', :action_code);

    db_id = f_get_db_id();
    db_is_actual = 0;

    if (exists(select * from hp_db_is_actual)) then
        db_is_actual = 1;

    is_biom = iif(db_id in (39801, 41701), 1, 0);

    res_code = case action_code
                   when 'send_code' then 'send_error'
                   when 'check_code' then 'check_error'
                   when 'update_pass' then 'pass_error'
                   when 'login' then 'login_error'
               end;

    -------------------- ОБ�©ИЕ ПРОВЕРКИ ДЛЯ ВСЕХ ДЕЙСТВИЙ --------------------
    mcl_login = f_json_str_get_value(action_data, 'login', '');
    operating_system = f_json_str_get_value(action_data, 'operating_system', '');
    application_version = f_json_str_get_value(action_data, 'application_version', '');

    if (coalesce(mcl_login, '') = '') then
    begin
        err_text = 'Отсутствует номер телефона';
        suspend;
        exit;
    end

    -- RFCRU-5785
    if (f_phone_validate_mask(:mcl_login, :db_id) = 0) then
    begin
        err_text = 'Неправильный номер телефона';
        suspend;
        exit;
    end

    if (operating_system = '') then
    begin
        err_text = 'Название операционной системы не передано';
        suspend;
        exit;
    end

    if (operating_system not in ('ios', 'android')) then
    begin
        err_text = 'Недопустимая операционная система';
        suspend;
        exit;
    end

    if (application_version = '') then
    begin
        err_text = 'Версия приложения не передана';
        suspend;
        exit;
    end

    --------------------------------------------------------------------------


    --------- ПРОВЕРКА ВОЗМОЖНОСТИ РЕГИСТРАЦИИ В ОПРЕДЕЛЕННОЙ СТРАНЕ ---------                    -- временное решение
    if (if_test_phone = 0 and db_is_actual = 1 and action_code in ('send_code', 'check_code') and db_id <> 39801) then
    begin
        res_code = 'send_error'; -- Для этих проверок res_code всегда будет "send_error"
        ip_address = f_json_str_get_value(action_data, 'ip', '');
        is_test_phone = 0;

        country =
          case db_id
               when 81001 then 'РФ'
               when 39801 then 'РК'
               when 41701 then 'КГ'
               when 94401 then 'AZ'
          end;

        -- RFCRU-988 Регистрация в КМП возможна только с территории РФ
        if (ip_address = '') then
        begin
            err_text = 'Не получен ip-адрес';
            suspend;
            exit;
        end

        select ipr_is_local
        from hp_get_ip_info(:ip_address)
        into is_local_ip;

        if (exists(select 1
                   from mc_test_access mta
                   where mta.mta_phone_number = :mcl_login)) then
            is_test_phone = 1;
        
        if (is_local_ip = 0 and is_test_phone = 0) then  -- тестовые номера телефонов для РФ
        begin
            err_text = 'Регистрация возможна только на территории '||country||'. Если вы находитесь в '||country||', но не можете зарегистрироваться, проверьте, отключен ли у вас VPN';
            suspend;
            exit;
        end

        country_condition = 0;
        if ((db_id = 81001) and char_length(mcl_login) = 11 and left(mcl_login, 2) = '79') then
            country_condition = 1;
        if ((db_id = 39801) and char_length(mcl_login) = 11 and left(mcl_login, 2) in ('70', '74', '77')) then
            country_condition = 1;
        if ((db_id = 41701) and char_length(mcl_login) = 12 and left(mcl_login, 3) = '996') then
            country_condition = 1;
        if ((db_id = 94401) and char_length(mcl_login) = 12 and left(mcl_login, 3) = '994') then
            country_condition = 1;
    
        if (country_condition = 0) then
        begin
            err_text = 'Регистрация возможна только на территории '||country||'.';
            suspend;
            exit;
        end
    end
    --------------------------------------------------------------------------


    --------------------- ПРОВЕРКА ФОРМАТА ДАТЫ РОЖДЕНИЯ ---------------------
    if (action_code in ('send_code', 'update_pass')) then
    begin
        birth_date_str = f_json_str_get_value(action_data, 'birth_date', '');

        if (is_biom = 0 and f_is_valid_date(:birth_date_str) = 0) then
        begin
            err_text = 'Неправильный формат даты рождения';
            suspend;
            exit;
        end

    end
    --------------------------------------------------------------------------

    ----------------------- ПРОВЕРКА ВЕРСИИ ПРИЛОЖЕНИЯ -----------------------
    if (action_code in ('send_code', 'update_pass', 'login')) then
    begin
        mcl_project_text = f_json_str_get_value(action_data, 'project', 'kmp');

        select res_code, err_text
        from hp_get_mc_app_version(:mcl_login, :mcl_project_text, :application_version, :operating_system)
        into res_code, err_text;
    
        if (res_code = 'version_error') then
        begin
          suspend;
          exit;
        end
    end
    --------------------------------------------------------------------------


    if (action_code = 'send_code') then
    begin
        -- Специфические проверки для действия "send_code"
    end


    if (action_code = 'check_code') then
    begin
        -- Специфические проверки для действия "check_code"
        pin_code = f_json_str_get_value(action_data, 'pin_code', -1);

        if (coalesce(pin_code, -1) = -1) then
        begin
          err_text = 'Код не указан';
          suspend;
          exit;
        end

    end


    if (action_code = 'update_pass') then
    begin
        -- Специфические проверки для действия "update_pass"
        pin_code = f_json_str_get_value(action_data, 'pin_code', -1);
        mcl_pass_hash = f_json_str_get_value(action_data, 'password', null);

        if (coalesce(pin_code, -1) = -1) then
        begin
            err_text = 'Код не указан';
            suspend;
            exit;
        end
    
        if (coalesce(mcl_pass_hash, '') = '') then
        begin
            err_text = 'Пустой пароль';
            suspend;
            exit;
        end

    end


    if (action_code = 'login') then
    begin
        -- Специфические проверки для действия "login"
        mcl_pass_hash = f_json_str_get_value(action_data, 'password', '');

        if (coalesce(mcl_pass_hash, '') = '') then
        begin
            err_text = 'Пустой пароль';
            suspend;
            exit;
        end

    end

    if (coalesce(err_text, '') = '') then
        res_code = null;

    suspend;

end
