CREATE PROCEDURE AAREP_LLC_REPORT_TEMP(
    REG_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    LLC_K_ID D_ID,
    LLC_REGION_ID D_INTEGER,
    LLC_STATUS D_INTEGER,
    LLC_NAME D_STR_CODE,
    LLC_RESERV_KIND D_STR_CODE,
    LLC_OFFICE_NAME D_STR_100,
    LLC_BRANCH_NAME D_STR_100,
    LLC_GR_BALANCE D_CURRENCY,
    LLC_GR_RPPU D_CURRENCY,
    LLC_IND_BALANCE D_CURRENCY,
    LLC_IND_RPPU D_CURRENCY,
    LLC_GR_PERCENT D_PERCENT,
    LLC_IND_PERCENT D_PERCENT,
    LLC_K_BALANCE D_CURRENCY,
    LLC_PERCENT DECIMAL(15,7),
    LLC_RPPU D_CURRENCY,
    LLC_OFFICE_ID D_INTEGER,
    LLC_IS_SPECIAL D_INTEGER)
AS
declare variable LM_ID D_ID;
declare variable LLC_LOAN_TYPE D_SMALLINT;
begin
  for
    select
      lm_id,
      lk_from_k_id,
      lk_balance,
      lk_percent,
      lk_is_indiv,
      lk_from_reg_id,
      lk_status,
      cs_name,
      cs_llr_kind_text,
      r1.reg_code || ' ' || r1.reg_name,
      r2.reg_code || ' ' || r2.reg_name,
      r1.reg_unicode
    from llc_main
      inner join region_list(:reg_id, 1, -1, 1)
        on id_list = lm_from_reg_id
      inner join region as r1
        on r1.reg_unicode = id_list
      inner join region as r2
        on r2.reg_unicode = r1.parent
      inner join llc_kont
        on lk_from_lm_id = lm_id
      inner join kontstate
        on ks_from_k_id = lk_from_k_id
          and :rep_date between ks_date_begin and ks_date_end
          and ks_state = 1
          and ks_ol_is_delete = 0
      inner join classific
        on cs_id = lk_status
    where lm_date = :rep_date
      and lm_is_signed = 1
  into
    :lm_id,
    :llc_k_id,
    :llc_k_balance,
    :llc_percent,
    :llc_loan_type,
    :llc_region_id,
    :llc_status,
    :llc_name,
    :llc_reserv_kind,
    :llc_office_name,
    :llc_branch_name,
    :llc_office_id
  do
  begin
    llc_is_special = iif(llc_reserv_kind = 'Общий резерв', 0, 1);
    if (f_get_db_id() in (39801, 41701) ) then
    begin
      llc_rppu = (
        select
          sum(cast(lkc_balance as double precision) * lkc_percent / 100)
        from llc_kont_cli
        where lkc_from_k_id = :llc_k_id
          and lkc_from_lm_id = :lm_id);
    end else begin
      llc_rppu = cast(llc_k_balance as double precision) * llc_percent / 100;
    end
    llc_gr_balance = 0;
    llc_gr_rppu = 0;
    llc_ind_balance = 0;
    llc_ind_rppu = 0;
    llc_gr_percent = 0;
    llc_ind_percent = 0;
    if (llc_loan_type = 0) then
    begin
      llc_gr_balance = :llc_k_balance;
      llc_gr_rppu = :llc_rppu;
      llc_gr_percent = :llc_percent;
    end
    else
    begin
      llc_ind_balance = :llc_k_balance;
      llc_ind_rppu = :llc_rppu;
      llc_ind_percent = :llc_percent;
    end
    suspend;
  end
end
