CREATE PROCEDURE UXP$UXSEARCH_INFO(
    UFR_UUID D_UUID,
    UF_UUID D_UUID,
    UXA_CODE D_STR_CODE)
RETURNS (
    RES_UXA_CODE D_STR_CODE,
    RES_UXA_NAME D_STR_100,
    RES_FRAME_ID_FIELD D_STR_CODE,
    RES_FRAME_TEXT_FIELD D_STR_CODE,
    RES_UA_NAME D_STR_100,
    RES_MULTI_SELECT D_STR_CODE)
AS
begin
  res_frame_id_field = uxf$code(uf_uuid, 'code');

  select uxa_name
  from ux$alias
  where uxa_uuid = uxf$uniq(:uxa_code)
  into res_uxa_name;

  res_frame_text_field = '';
  res_multi_select = '';
  res_uxa_code = '';
  res_ua_name = '';

  select ua_uxa_fcode, ua_usr_text_field, ua_name
  from ux$action
  where ua_owner = :ufr_uuid and
        ua_usr_id_field = :res_frame_id_field
  into res_uxa_code, res_frame_text_field, res_ua_name;

  if (res_uxa_code <> uxa_code) then
  begin
    res_frame_text_field = '';
    res_uxa_code = uxa_code;
    res_ua_name = res_uxa_name;
  end

  if (res_frame_text_field containing '$') then
  begin
    -- Редактор может возвращает более одной строки, разделители могут быть:
    -- $delimiter_comma     // запятая
    -- $delimiter_semicolon // точка с запятой
    -- $square_brackets     // каждое значение в квадратной скобке

    res_multi_select = res_frame_text_field;
    res_frame_text_field = '';
  end

  suspend;
end
