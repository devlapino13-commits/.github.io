CREATE PROCEDURE HP_POCKET_WRITEOFF_RU(
    REP_REGION_ID INTEGER,
    REP_DATE DATE,
    BALANCE_DATE DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    REG_ID INTEGER,
    PR_NAME VARCHAR(255),
    PR_ID INTEGER,
    KONTRACT_NO VARCHAR(20),
    K_ID ID_TYPE,
    ZAD_OS CURR_TYPE,
    SH_CALC_PERC CURR_TYPE,
    SH_CALC_FINE CURR_TYPE,
    SH_DATE DATE,
    SH_ID INTEGER,
    K_DISB_DATE DATE)
AS
declare variable reg_id_old integer;
declare variable k_id_old id_type;
declare variable id_no integer;
declare variable ol_id id_type;
declare variable ok_number integer;
declare variable pr_id_old integer;
declare variable account_kt_perc varchar(100);
declare variable account_kt_os varchar(100);
declare variable change_accounts smallint;
declare variable zad_os_sum curr_type;
declare variable sh_calc_perc_sum curr_type;
declare variable sh_calc_fine_sum curr_type;
declare variable ob_number integer;
declare variable zad_perc curr_type;
declare variable ks_number integer;
declare variable is_close smallint;
declare variable sum_total curr_type;
declare variable c_id decimal(18,0);
declare variable okf_id integer;
declare variable account_dt_os varchar(100);
declare variable account_dt_perc varchar(100);
declare variable account_dt_perc_1 varchar(100);
declare variable account_dt_perc_2 varchar(100);
declare variable reg_code varchar(10);
declare variable summa decimal(18,4);
declare variable ks_state integer;
declare variable ol_date date;
declare variable max_ol_date date;
declare variable reason_rd_id integer;
begin

  reg_id_old = 0;
  k_id_old = 0;
  pr_id_old = 0;

  zad_os_sum = 0;
  sh_calc_perc_sum = 0;
  sh_calc_fine_sum = 0;

  ob_number = 0;

  reason_rd_id = (select rd.rd_id
                  from refdata rd
                  where rd.rd_code = 'Reason_WriteOff.More_3_Year');

  for select reg_id, k_id, kontract_no, zad_os, zad_perc, sh_calc_perc, sh_calc_fine, sh_date, sh_id, pr_id, pr_name,
             k_disb_date, reg_name, reg_code
      from aarep_writeoff_plan_ru(:rep_region_id, :rep_date, :balance_date, 0)
      where mess_str = 'Подлежит списанию'
      order by reg_id, pr_id, k_id, sh_id, sh_date
      into :reg_id, :k_id, :kontract_no, :zad_os, :zad_perc, :sh_calc_perc, :sh_calc_fine, :sh_date, :sh_id, :pr_id,
           :pr_name, :k_disb_date, :reg_name, :reg_code
  do
  begin
    change_accounts = 0;
    if ((reg_id <> reg_id_old) or (pr_id_old <> pr_id)) then
      change_accounts = 1;

    if (k_id <> k_id_old) then
    begin
      if (k_id_old <> 0) then
      begin
        if (zad_os_sum <> 0) then
        begin
          ob_number = ob_number + 1;

          insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                                ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                                ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date,
                                ob_ol_is_active, ob_ol_reg_id)
          values (:ol_id, :ob_number, '810', 1, null, :account_dt_os, :zad_os_sum, :zad_os_sum, null, null, null,
                  :account_kt_os, null, 'Основная сумма', null, :rep_date, 1, :reg_id);
        end
        if (sh_calc_perc_sum <> 0) then
        begin
          ob_number = ob_number + 1;

          if (k_disb_date < cast('28.02.2012' as date)) then
            account_dt_perc = account_dt_perc_1;
          else
            account_dt_perc = account_dt_perc_2;

          insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                                ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                                ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date,
                                ob_ol_is_active, ob_ol_reg_id)
          values (:ol_id, :ob_number, '810', 1, null, :account_dt_perc, :sh_calc_perc_sum, :sh_calc_perc_sum, null,
                  null, null, :account_kt_perc, null, 'Проценты по кредиту', null, :rep_date, 1,
                  :reg_id);
        end
      end
      zad_os_sum = zad_os;
      sh_calc_perc_sum = sh_calc_perc;
      sh_calc_fine_sum = sh_calc_fine;
    end
    else
    begin
      zad_os_sum = zad_os_sum + zad_os;
      sh_calc_perc_sum = sh_calc_perc_sum + sh_calc_perc;
      sh_calc_fine_sum = sh_calc_fine_sum + sh_calc_fine;
    end

    if (reg_id <> reg_id_old) then
    begin
      select kv
      from get_kv(:reg_id, 5)
      into :ol_id;

      ok_number = 0;
      ob_number = 0;
      okf_id = 0;

      insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_memo,
                            ol_status, is_delete, ol_user_id, ol_from_rd_id)
      values (:ol_id, :reg_id, 1, 6, :rep_date, 4,
              'Пакетное списание платежей с просрочкой 3 года и более.', 0,
              0, 1010100000278, :reason_rd_id);
    end

    if (change_accounts = 1) then
    begin
      select res_account
      from hp_make_kredit_acclist(:pr_id, :reg_id, 9)
      into :account_dt_os;

      select res_account
      from hp_make_kredit_acclist(:pr_id, :reg_id, 1)
      into :account_kt_os;

      account_dt_perc_1 = reg_code || '810' || '91-02' || '00000' || '2';
      account_dt_perc_2 = reg_code || '810' || '63-01' || '00000' || '1';

      /*
        select RES_ACCOUNT
        from HP_MAKE_KREDIT_ACCLIST(:pr_id, :REG_ID, 9)
        into :ACCOUNT_DT_PERC_1;

        select RES_ACCOUNT
        from HP_MAKE_KREDIT_ACCLIST(:pr_id, :REG_ID, 9)
        into :ACCOUNT_DT_PERC_2;
        */

      select res_account
      from hp_make_kredit_acclist(:pr_id, :reg_id, 2)
      into :account_kt_perc;
    end

-- Если задолженность по контракту  = 0, тогда
    /*
    if ((K_ID <> K_ID_OLD) and (K_ID_OLD <> 0)) then
    begin
      ks_number = (select max(ks_number)
                   from kontstate ks
                   where ks_from_k_id = :k_id_old );
    
      is_close = 0;
      select 1 from kontstate
      where ks_ol_is_delete = 0
      and ks_state = 2
      and ks_from_k_id = :k_id_old
      into is_close;
      if (is_close is null) then is_close = 0;
    
      if (is_close = 0) then
      begin
        sum_total = null;

        select sum(ZAD_OS + SH_CALC_PERC + SH_CALC_FINE)
        from aarep_make_zad_list(:k_id_old, :balance_date, 1)
        into :sum_total;

        if (sum_total is null) then sum_total = 0;

        if (sum_total = 0) then
        begin
            insert into KONTSTATE (KS_FROM_OL_ID, KS_FROM_K_ID, KS_DATE_BEGIN, KS_DATE_END, KS_OL_IS_DELETE, KS_STATE, KS_NUMBER)
            values (:ol_id, :k_id_old, :rep_date, '31.12.9999', 0, 2, :KS_NUMBER + 1);            
        end
      end
    end
    */

    ok_number = ok_number + 1;

    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                          ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                          ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
    values (:ol_id, :ok_number, :k_id, :sh_id, :sh_date, 5, -:zad_os, -:sh_calc_perc, -:sh_calc_fine, 3, 0, 0, 0, 0, 0,
            :rep_date, :reg_id, 1, 0);

    if ((zad_perc - sh_calc_perc) <> 0) then
    begin
      ok_number = ok_number + 1;

      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                            ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
      values (:ol_id, :ok_number, :k_id, :sh_id, :sh_date, 3, 0, 0, 0, 3, 0, 0, -(:zad_perc - :sh_calc_perc), 0, 0,
              :rep_date, :reg_id, 1, 0);
    end

    select kncl.kc_from_c_id
    from knclient kncl
    where kncl.kc_from_k_id = :k_id and
          kncl.kc_amount_approved > 0
    order by kncl.kc_amount_approved desc
    rows 1
    into :c_id;

    okf_id = okf_id + 1;

    insert into operkroff (oko_from_ol_id, oko_id, oko_from_k_id, oko_from_c_id, oko_type, oko_os, oko_perc, oko_other,
                           oko_sh_os, oko_sh_perc, oko_sh_other)
    values (:ol_id, :okf_id, :k_id, :c_id, 1, 0, 0, 0, :zad_os, :sh_calc_perc, 0);

    pr_id_old = pr_id;
    reg_id_old = reg_id;
    k_id_old = k_id;

    suspend;
  end

  if (ol_id is not null) then
  begin
    if (zad_os_sum <> 0) then
    begin
      ob_number = ob_number + 1;

      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                            ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                            ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date,
                            ob_ol_is_active, ob_ol_reg_id)
      values (:ol_id, :ob_number, '810', 1, null, :account_dt_os, :zad_os_sum, :zad_os_sum, null, null, null,
              :account_kt_os, null, 'Основная сумма', null, :rep_date, 1, :reg_id);
    end
    if (sh_calc_perc_sum <> 0) then
    begin
      ob_number = ob_number + 1;

      if (k_disb_date < cast('28.02.2012' as date)) then
        account_dt_perc = account_dt_perc_1;
      else
        account_dt_perc = account_dt_perc_2;

      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                            ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                            ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date,
                            ob_ol_is_active, ob_ol_reg_id)
      values (:ol_id, :ob_number, '810', 1, null, :account_dt_perc, :sh_calc_perc_sum, :sh_calc_perc_sum, null, null,
              null, :account_kt_perc, null, 'Проценты по кредиту', null, :rep_date, 1, :reg_id);
    end
    /*
    ks_number = (select max(ks_number)
                 from kontstate
                 where ks_from_k_id = :k_id_old);
    
    is_close = 0;
    select 1 from kontstate
    where ks_ol_is_delete = 0
    and ks_state = 2
    and ks_from_k_id = :k_id_old
    into is_close;
    if (is_close is null) then is_close = 0;
    
    if (is_close = 0) then
    begin
      sum_total = null;
    
      select sum(ZAD_OS + SH_CALC_PERC + SH_CALC_FINE)
      from aarep_make_zad_list(:k_id_old, :rep_date, 1)
      into :sum_total;
    
      if (sum_total is null) then sum_total = 0;
    
      if (sum_total = 0) then
      begin
          insert into KONTSTATE (KS_FROM_OL_ID, KS_FROM_K_ID, KS_DATE_BEGIN, KS_DATE_END, KS_OL_IS_DELETE, KS_STATE, KS_NUMBER)
          values (:ol_id, :k_id_old, :rep_date, '31.12.9999', 0, 2, :KS_NUMBER + 1);
      end
    end
    */
  end

  ol_id = null;
  k_id = null;
  ks_number = 0;

  for select distinct ok.ok_from_k_id, ol.ol_id
      from operkred ok, operlist ol
      where ok.ok_ol_date = :rep_date and
            ok.ok_type = 5 and
            ok.ok_ol_is_active = 1 and
            ok.ok_from_c_id = 0 and
            ok.ok_from_ol_id = ol.ol_id and
            ol.ol_memo = 'Пакетное списание платежей с просрочкой 3 года и более.'
      into :k_id, :ol_id
  do
  begin
    summa = null;

    select sum(zad_os + sh_calc_perc + sh_calc_fine)
    from aarep_make_zad_list(:k_id, :balance_date, 1)
    where sh_date is not null
    into :summa;

    if (summa is null) then
      summa = 0;

    select ks.ks_state
    from kontstate ks
    where ks.ks_from_k_id = :k_id and
          ks.ks_ol_is_delete = 0 and
          :balance_date between ks.ks_date_begin and ks.ks_date_end
    into :ks_state;

    if ((summa = 0) and
        (ks_state <> 2)) then
    begin
      ks_number = (select max(ks_number)
                   from kontstate
                   where ks_from_k_id = :k_id);

      insert into kontstate (ks_from_ol_id, ks_from_k_id, ks_date_begin, ks_date_end, ks_ol_is_delete, ks_state,
                             ks_number)
      values (:ol_id, :k_id, :rep_date, '31.12.9999', 0, 2, :ks_number + 1);
    end
  end
end
