CREATE PROCEDURE AHPLIST_AKTTRANSFER(
    AKT_ID DECIMAL(18,0))
RETURNS (
    AKT_ID_NO DECIMAL(18,0),
    AKT_NO INTEGER,
    AKT_NEW_KF_NAME VARCHAR(100),
    AKT_NEW_OFFICER_NAME VARCHAR(100),
    AKT_NEW_REGION_NAME VARCHAR(50),
    AKT_OLD_KF_NAME VARCHAR(100),
    AKT_OLD_OFFICER_NAME VARCHAR(100),
    AKT_OLD_REGION_NAME VARCHAR(50),
    AKT_DATE DATE)
AS
begin
  for
    select
      kal_from_ka_id,
      kal_number,
      kal_date,
      kal_new_kf_name,
      c1.c_name_fio,
      r1.reg_name,
      kal_prior_kf_name,
      c2.c_name_fio,
      r2.reg_name
    from knaktlist
      inner join clients as c1
        on c1.c_id = kal_new_kf_officer
      inner join region as r1
        on r1.reg_unicode = kal_new_reg_id
      inner join clients as c2
        on c2.c_id = kal_prior_lo_id
      inner join region as r2
        on r2.reg_unicode = kal_prior_reg_id
    where kal_from_ka_id = :akt_id
      and knaktlist.is_delete = 0
    order by kal_number
  into
    :akt_id_no,
    :akt_no,
    :akt_date,
    :akt_new_kf_name,
    :akt_new_officer_name,
    :akt_new_region_name,
    :akt_old_kf_name,
    :akt_old_officer_name,
    :akt_old_region_name
  do suspend;
end
