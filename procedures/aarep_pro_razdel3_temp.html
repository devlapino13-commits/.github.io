CREATE PROCEDURE AAREP_PRO_RAZDEL3_TEMP(
    REP_DATE_END DATE)
RETURNS (
    BSL_NAME STR_255,
    IS_RESTRUCT SMALLINT,
    IS_DELINQ SMALLINT,
    LKC_STATUS SMALLINT,
    NO_ACCRUAL_STATUS SMALLINT,
    BALANCE_KC CURR_TYPE)
AS
declare variable dolya double precision;
declare variable oks_balance decimal(18,4);
declare variable k_id id_type;
declare variable c_id id_type;
declare variable k_id_prior id_type;
begin
  k_id_prior = 0;
  for select lkc_from_k_id, lkc_from_c_id, iif(lkc_deliq_days > 0, 1, 0) as is_delinq, lkc_balance, lkc_status
      from llc_main, llc_kont_cli
      where lm_date_status = :rep_date_end and
            lm_is_signed = 1 and
            lm_id = lkc_from_lm_id
      into k_id, c_id, is_delinq, balance_kc, lkc_status
  do
  begin
    bsl_name = null;
    select bsl_name
    from knclient, busstandartlink, busstandartlist
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = :k_id and
          kc_busuness_subtype = bslk_from_bs_id and
          bslk_from_bsm_id = 7 -- Классификация НБКР ПРО
          and
          bslk_from_bsm_id = bsl_from_bsm_id and
          bslk_from_bsl_id = bsl_id
    into bsl_name;

    if (k_id_prior <> k_id) then
    begin
      k_id_prior = k_id;
      is_restruct = (select is_restruct
                     from aarep_get_restructuring(:k_id, :rep_date_end, 41701));

      no_accrual_status = 0;
      select iif(k_stop_perc <= :rep_date_end, 1, 0)
      from kontract
      where k_id = :k_id
      into no_accrual_status;
    end
    suspend;
  end
end
