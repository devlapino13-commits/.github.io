CREATE PROCEDURE PRINT_FOR_BLANK_ASP(
    UN_ID D_ID)
RETURNS (
    FIO D_STR_255,
    UN_POSITION D_STR_100,
    DESERTED_POINT D_STR_255,
    DESERTED_POINT_KG D_STR_255)
AS
begin
    select
        plt.plt_short_name || pl.pl_name,
        pl.pl_name || ' ' || plt.plt_alt_short_name,
        un.un_fio,
        coalesce(jl.jl_name, 'сотрудник')
    from addrlist adl
    inner join region reg on reg.reg_address_fact = adl.adl_id
    inner join username un on un.un_curr_reg_id = reg.reg_unicode
        and un.un_id = :un_id
    inner join hr_pstatus ps on un.un_from_c_id = ps.ps_from_c_id
        and ps.ps_is_working = 1
        and current_date between ps_date_begin and ps_date_end
    inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
    inner join places pl on pl.pl_id = adl.adl_from_pl_id
    inner join placetype plt on plt.plt_id = pl.pl_status
    into
        deserted_point,
        deserted_point_kg,
        fio,
        un_position;

    suspend;
end
