CREATE PROCEDURE HP_KONTDISBPLAN_APP_REPEAT_SUM(
    KDP_ID D_INTEGER)
RETURNS (
    APP_REPEAT_SUM D_CURRENCY)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable k_amount_approved d_currency;
begin
  -- возвращает сумму договоров в процессе по повторным клиентам (RFCAZ-244)

  app_repeat_sum = 0;

  if (f_get_db_id() <> 94401) then
  begin
    suspend;
    exit;
  end

  for select k_id,
             k_amount_approved
      from kontract
      inner join kontdisbplan on kdp_date_begin = k_disb_date and
            kdp_office_id = k_last_reg_id
      where k_status = 0 and
            kdp_id = :kdp_id
      into k_id,
           k_amount_approved
  do
  begin
    c_id = null;
    select kc_from_c_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 rows 1
    into c_id;

    if (exists(select *
               from kontstate
               inner join knclient on kc_from_k_id = ks_from_k_id
               where kc_from_c_id = :c_id and
                     kc_amount_approved > 0 and
                     ks_ol_is_delete = 0 and
                     ks_state in (1, 2) and
                     current_date between ks_date_begin and ks_date_end)) then
      app_repeat_sum = app_repeat_sum + k_amount_approved;
  end

  suspend;
end
