CREATE PROCEDURE AAREP_CEO_DAILY_OLD(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(50),
    SVZ_KOL_ACT INTEGER,
    SVZ_DISB INTEGER,
    DISB_KOL_KRED INTEGER,
    DISB_KOL_PARALEL INTEGER,
    DISB_KOL_VSEGO INTEGER,
    SVZ_ONE_DISB CURR_TYPE,
    PAR1 CURR_TYPE,
    PAR7 CURR_TYPE,
    PAR30 CURR_TYPE,
    PERC_UDERJ CURR_TYPE,
    REP_BALANCE CURR_TYPE,
    PRIROST_PORTFEL CURR_TYPE,
    KOL_CLI INTEGER,
    PRIROST_CLI INTEGER)
AS
declare variable reg_unicode integer;
declare variable rep_date_begin date;
begin
  rep_date_begin = rep_date_end;
  for select reg_unicode, reg_name
      from region, region_list(:rep_region_id, 1, 3, 1)
      where reg_unicode = id_list
      order by reg_name
      into reg_unicode, office_name
  do
  begin
    select svz_kol_act, svz_disb, disb_kol_kred, disb_kol_paralel, disb_kol_vsego, svz_one_disb, par1, par7, par30,
           perc_uderj, rep_balance, prirost_portfel, kol_cli, prirost_cli
    from aarep_info_office_old(:reg_unicode, :rep_date_begin, :rep_date_end)
    into svz_kol_act, svz_disb, disb_kol_kred, disb_kol_paralel, disb_kol_vsego, svz_one_disb, par1, par7, par30,
         perc_uderj, rep_balance, prirost_portfel, kol_cli, prirost_cli;

    suspend;
  end
end
