CREATE PROCEDURE FT_ACARD_PROPS_GET(
    C_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    RES_TEXT D_STR_LONG)
AS
declare variable cds_doc_no str_100;
declare variable cds_doc_date_begin str_100;
declare variable cds_doc_date_end str_100;
declare variable citizen_id d_integer;
declare variable is_found d_smallint;
declare variable is_old d_bool;
declare variable doc_name d_str_100;
begin
  cds_doc_no = '';
  is_found = 0;
  is_old = iif(rep_date < date'30.10.2024', 1, 0);

  select pk_country_id
  from clientdocs cds
  inner join passkind pask on pask.pk_id = cds.cds_from_pk_id
  where cds_from_c_id = :c_id and
        cds_create_datetime < (:rep_date + 1) and
        cds_is_main_doc = 1
  order by cds_create_datetime desc
  rows 1
  into :citizen_id;

  if (citizen_id is null) then
  begin
    select cl.c_citizen
    from clients cl
    where cl.c_id = :c_id
    into citizen_id;
  end

  if (citizen_id <> 13) then
  begin
    select coalesce(cds_doc_no, ''), f_datetostr(cds_doc_date_begin), f_datetostr(cd.cds_doc_date_end), 1
    from clientdocs cd, passkind, passdoctype
    where cds_from_c_id = :c_id and
          cds_from_pk_id = pk_id and
          pk_from_pdt_id = pdt_id and
          pdt_code = 'registration_vnj' and
          cds_is_delete = 0 and
          :rep_date between cd.cds_doc_date_begin and cd.cds_doc_date_end and
          cds_create_datetime < (:rep_date + 1)
    order by cds_create_datetime desc
    rows 1
    into :cds_doc_no, :cds_doc_date_begin, :cds_doc_date_end, :is_found;
    doc_name = 'ВНЖ';

    if (is_found  = 0) then
    begin
      select coalesce(cds_doc_no, ''), f_datetostr(cds_doc_date_begin), f_datetostr(cd.cds_doc_date_end), 1
      from clientdocs cd, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = pdt_id and
            pdt_code = 'registration_rvp' and
            cds_is_delete = 0 and
            :rep_date between cd.cds_doc_date_begin and cd.cds_doc_date_end and
            cds_create_datetime < (:rep_date + 1)
      order by cds_create_datetime desc
      rows 1
      into :cds_doc_no, :cds_doc_date_begin, :cds_doc_date_end, :is_found;
      doc_name = 'РВП';
    end

    if (is_found  = 0 and is_old = 1) then
    begin
      select coalesce(cds_doc_no, ''), f_datetostr(cds_doc_date_begin), f_datetostr(cd.cds_doc_date_end), 1
      from clientdocs cd, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = pdt_id and
            pdt_code = 'work_permit' and
            cds_is_delete = 0 and
            :rep_date between cd.cds_doc_date_begin and cd.cds_doc_date_end and
            cds_create_datetime < (:rep_date + 1)
      order by cds_create_datetime desc
      rows 1
      into :cds_doc_no, :cds_doc_date_begin, :cds_doc_date_end, :is_found;
      doc_name = 'Разрешение на работу';
    end

    if (is_found  = 0 and is_old = 1) then
    begin
      select coalesce(cds_doc_no, ''), f_datetostr(cds_doc_date_begin), f_datetostr(cd.cds_doc_date_end), 1
      from clientdocs cd, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = pdt_id and
            pdt_code = 'work_patent' and
            cds_is_delete = 0 and
            :rep_date between cd.cds_doc_date_begin and cd.cds_doc_date_end and
            cds_create_datetime < (:rep_date + 1)
      order by cds_create_datetime desc
      rows 1
      into :cds_doc_no, :cds_doc_date_begin, :cds_doc_date_end, :is_found;
    doc_name = 'Патент';
    end

    if (is_found  = 0 and is_old = 1) then
    begin
      select coalesce(cds_doc_no, ''), f_datetostr(cds_doc_date_begin), f_datetostr(cd.cds_doc_date_end), 1
      from clientdocs cd, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = pdt_id and
            pdt_code = 'job_agreement' and
            cds_is_delete = 0 and
            :rep_date between cd.cds_doc_date_begin and cd.cds_doc_date_end and
            cds_create_datetime < (:rep_date + 1)
      order by cds_create_datetime desc
      rows 1
      into :cds_doc_no, :cds_doc_date_begin, :cds_doc_date_end, :is_found;
      doc_name = 'Трудовой договор';
    end

    if (is_found  = 0 and is_old = 1) then
    begin
      select coalesce(cds_doc_no, ''), f_datetostr(cds_doc_date_begin), f_datetostr(cd.cds_doc_date_end), 1
      from clientdocs cd, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = pdt_id and
            pdt_code = 'relocation_docs' and
            cds_is_delete = 0 and
            :rep_date between cd.cds_doc_date_begin and cd.cds_doc_date_end and
            cds_create_datetime < (:rep_date + 1)
      order by cds_create_datetime desc
      rows 1
      into :cds_doc_no, :cds_doc_date_begin, :cds_doc_date_end, :is_found;
      doc_name = 'Свидетельство участника программы добровольного переселения';
    end

    if (is_found  = 0) then
    begin
      select coalesce(cds_doc_no, ''),
          f_datetostr(cds_doc_date_begin),
          f_datetostr(iif(:rep_date < cast('24.07.2025' as date), cds_doc_date_end, cds_doc_date_begin + 89)),
          1
      from clientdocs, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = pdt_id and
            pdt_code = 'magration_card' and
            cds_is_delete = 0 and
            :rep_date between cds_doc_date_begin and iif(:rep_date < cast('24.07.2025' as date), cds_doc_date_end, cds_doc_date_begin + 89) and
            cds_create_datetime < (:rep_date + 1)
      order by cds_create_datetime desc
      rows 1
      into :cds_doc_no, :cds_doc_date_begin, :cds_doc_date_end, :is_found;
      doc_name = 'Миграционная карта';
    end

    if (is_found  = 0 and rep_date >= cast('24.07.2025' as date)) then -- RFCRU-4992
    begin
      select pdt_name, coalesce(cds_doc_no, ''), f_datetostr(cds_doc_date_begin),
          f_datetostr(iif(pdt_code = 'magration_card', cds_doc_date_begin + 89, cds_doc_date_end)), 1
      from clientdocs, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = pdt_id and
            pdt_code in('registration_vnj','registration_rvp','magration_card') and
            cds_is_delete = 0 and
            cds_create_datetime < (:rep_date + 1)
      order by cds_create_datetime desc
      rows 1
      into doc_name, :cds_doc_no, :cds_doc_date_begin, :cds_doc_date_end, is_found;

    end
  end

  if (is_found = 1) then
    res_text = doc_name || ': Серия/номер: ' || cds_doc_no || '; Дата начала срока действия права пребывания (проживания): ' || cds_doc_date_begin || '; Дата окончания срока действия права пребывания: ' || cds_doc_date_end || ';';
  else
    res_text = '---';
  suspend;
end
