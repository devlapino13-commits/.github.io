CREATE PROCEDURE HP_CARD_BLOCK_OPERS_LOG(
    REGION_ID D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE)
RETURNS (
    REGION_NAME D_STR_CODE,
    OPER_DATE_TIME D_DATETIME,
    CARD_ID D_INTEGER,
    PAN_MASKED D_STR_20,
    NAME_FIO D_STR_100,
    OPER_TYPE D_STR_CODE,
    REASON_TEXT VARCHAR(30),
    STATUS_NAME D_STR_CODE,
    ERROR_TEXT D_STR_8100,
    STATUS_ID INTEGER,
    CBO_ID D_INTEGER,
    OPER_TYPE_ID D_INTEGER)
AS
begin
  for select r.reg_name, cbo.cbo_datetime, cbo.cbo_from_cal_id, cal.cal_pan_short as pan_masked, cl.c_name_fio,
    bl.name || iif(cbo.cbo_type = 1 and cbo.cbo_is_hard_stoplist = 1, ' (жесткий стоп-лист)', '') as oper_type, cbr.cbr_reason as block_reason, el.el_name as status_name, iif(cbo.cbo_status = -1, cs.cs_answer_json, '') as error_text, cbo.cbo_status, cbo.cbo_id,
    cbo.cbo_type
  from cardblockopers cbo
    inner join cardlist cal on cbo.cbo_from_cal_id = cal.cal_card_id
    inner join clients cl on cbo.cbo_from_c_id = cl.c_id
    inner join cardblocktype_list bl on cbo.cbo_type = bl.id
    inner join cardblockreasons cbr on cbo.cbo_reason = cbr.cbr_id
    inner join enum_list el on cbo.cbo_status = el.el_id and el.el_kind = 'cardblock_oper_status'
    inner join region r on cbo.cbo_region_id = r.reg_unicode
    inner join region_list(:region_id, 1, 3, 1) rl on r.reg_unicode = rl.id_list
    left join cardsync cs on cbo.cbo_from_cs_id = cs.cs_id
  where cbo.cbo_date between :date_begin and :date_end
  order by cbo.cbo_id
  into :region_name, :oper_date_time,:card_id, :pan_masked,:name_fio, :oper_type, :reason_text, :status_name, :error_text, :status_id, :cbo_id, :oper_type_id
  do
    suspend;

end
