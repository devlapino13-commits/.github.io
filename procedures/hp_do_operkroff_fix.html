CREATE PROCEDURE HP_DO_OPERKROFF_FIX(
    K_ID D_ID,
    REP_DATE D_DATE)
AS
declare variable sum_os d_currency;
declare variable sum_perc d_currency;
declare variable off_os d_currency;
declare variable off_perc d_currency;
declare variable oko_id d_integer;
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable ol_id d_id;
begin
  for select kc_id, kc_from_c_id
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into kc_id, c_id
  do
  begin
    for select ok_from_ol_id, sum(-ok_os), sum(-ok_perc)
        from kn_operkred
        where ok_ol_is_active = 1 and
              ok_ol_date = :rep_date and
              ok_from_kc_id = :kc_id and
              ok_type = 5 and
              (ok_os <> 0 or
              ok_perc <> 0)
        group by 1
        into ol_id, sum_os, sum_perc
    do
    begin
      off_os = 0;
      off_perc = 0;
      oko_id = 0;
      select max(oko_id), sum(oko_sh_os), sum(oko_sh_perc)
      from operkroff
      where oko_from_ol_id = :ol_id and
            oko_from_k_id = :k_id and
            oko_from_c_id = :c_id
      into oko_id, :off_os, :off_perc;

      if (sum_os - off_os <> 0 or sum_perc - off_perc <> 0) then
      begin
        update operkroff
        set oko_sh_os = oko_sh_os + (:sum_os - :off_os),
            oko_sh_perc = oko_sh_perc + (:sum_perc - :off_perc)
        where oko_from_ol_id = :ol_id and
              oko_from_k_id = :k_id and
              oko_from_c_id = :c_id and
              oko_id = :oko_id;
      end
    end
  end
end
