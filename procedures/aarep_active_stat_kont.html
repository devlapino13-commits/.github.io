CREATE PROCEDURE AAREP_ACTIVE_STAT_KONT(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    REG_OKRUG VARCHAR(255),
    REG_REGION VARCHAR(255),
    REG_OFFICE VARCHAR(255),
    LOAN_TYPE_TEXT VARCHAR(255),
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    KAL_NEW_KF_NAME VARCHAR(255),
    K_BONUS_TEXT VARCHAR(255),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    ZAD_OS_TOTAL DECIMAL(18,4),
    SH_LAST_DATE DATE,
    OKS_DELINQ_DATE DATE,
    KD_PERCENT_PER_DAY DECIMAL(18,7),
    CIRCLE_STAGE INTEGER,
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER)
AS
declare variable C_CURR_BONUS_STATUS integer;
declare variable CURR varchar(3);
declare variable CURR_CURR varchar(3);
declare variable PR_LOAN_TYPE integer;
declare variable EMP integer;
declare variable PARAL integer;
declare variable C_LAST_BONUS_STATUS integer;
declare variable REG_ID integer;
declare variable REG_ID_PRIOR integer;
declare variable KF_ID ID_TYPE;
declare variable K_ID_FROM_STATUS ID_TYPE;
declare variable KS_CLOSE_DATE date;
declare variable LAST_STATUS integer;
declare variable CURR_STATUS integer;
declare variable CAN_NEXT integer;
declare variable CR_RATE CURR_TYPE;
declare variable CURR_RATE CURR_TYPE;
begin
  select cm_default_currency
  from company
  into :curr_curr;

  select cr_rate
  from opercurr
  where cr_from_cu_id = '840' and
        :rep_date between cr_date and cr_date_end
  into cr_rate;

  /*
    for select
           reg3.reg_name,
           reg2.reg_name,
           reg1.reg_name,
           k_id,
           kont.k_kontract_no,
           kont.k_disb_date,
           kont.k_amount_approved,
           kal_new_kf_name,
           kont.k_kolich_client_approved,
           k_index_curr_code,
           pr_loan_type,
           pr_is_employer,
           pr_category
    from kontract kont, kontstate ks, region_list(:rep_region_id,1,3,1),  knaktlist kal,
         region reg1, region reg2, region reg3, product pr
    where ks.ks_ol_is_delete = 0
    and :rep_date between ks.ks_date_begin and ks.ks_date_end
    and ks.ks_state = 1
    and ks_region_id=id_list
    and ks.ks_from_k_id = kont.k_id
    and k_from_kf_id = kal_from_kf_id
    and kal.is_delete = 0
    and :rep_date between kal_date and kal_date_end
    and reg1.reg_unicode = id_list
    and reg2.reg_unicode = reg1.parent
    and reg3.reg_unicode = reg2.parent
    and k_from_pr_id=pr_id
    into
         :reg_okrug, :reg_region, :reg_office, :k_id, :k_kontract_no, :k_disb_date,
         :k_amount_approved, :kal_new_kf_name, :k_kolich_client_approved,
         :curr, :pr_loan_type, :emp, :paral
                                                */
  reg_id_prior = 0;
  for select oks_region_id, oks_from_k_id, oks_balance, oks_delinq_date
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where oks_region_id = id_list and
            :rep_date between oks_date_begin and oks_date_end
            and oks_close_state = 0 
      into reg_id, k_id, zad_os_total, oks_delinq_date
  do
  begin
    if (oks_delinq_date >= rep_date) then
      oks_delinq_date = null;

    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
      from region r1, region r2, region r3
      where r1.reg_unicode = :reg_id and
            r1.parent = r2.reg_unicode and
            r2.parent = r3.reg_unicode
      into reg_office, reg_region, reg_okrug;
    end

    select k_kontract_no, k_disb_date, k_amount_approved, kal_new_kf_name, k_kolich_client_approved, k_index_curr_code,
           pr_loan_type, pr_is_employer, pr_category, k_kredline_end, k_from_kf_id, k_index_curr_rate, k_circle,
           k_circle_internal
    from kontract, product, knaktlist
    where k_id = :k_id and
          k_from_pr_id = pr_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date between kal_date and kal_date_end
    into :k_kontract_no, :k_disb_date, :k_amount_approved, :kal_new_kf_name, :k_kolich_client_approved, :curr,
         :pr_loan_type, :emp, :paral, :sh_last_date, :kf_id, :curr_rate, :k_circle, :k_circle_internal;

    /* >> определяю статус группы                           */
    k_bonus_text = '';
    select c_last_bonus_status, c_curr_bonus_status, c_curr_bonus_status_str
    from hp_kontstate_loan_status(:k_id, 0, 0, :rep_date, '')
    into c_last_bonus_status, c_curr_bonus_status, k_bonus_text;

    circle_stage = 0;
    if (c_last_bonus_status <> 0 and
        c_last_bonus_status = c_curr_bonus_status) then
    begin
      circle_stage = 1;
      can_next = 1;
      for select ks_from_k_id, ks_date_end
          from kontstate, kontract
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_begin < :k_disb_date and
                ks_from_k_id = k_id and
                k_from_kf_id = :kf_id
          order by ks_date_begin desc
          into k_id_from_status, ks_close_date
      do
      begin
        if (ks_close_date > rep_date) then
          ks_close_date = rep_date;

        if (can_next = 1) then
        begin
          k_bonus_text = '';
          select c_last_bonus_status, c_curr_bonus_status, c_curr_bonus_status_str
          from hp_kontstate_loan_status(:k_id_from_status, 0, 0, :ks_close_date, '')
          into last_status, curr_status, :k_bonus_text;
          if (last_status is null) then
            last_status = 0;
          if (curr_status is null) then
            curr_status = 0;

          if (last_status <> 0 and
              last_status = c_curr_bonus_status and
              last_status = curr_status) then
          begin
            circle_stage = circle_stage + 1;
            can_next = 1;
          end
          else
          begin
            can_next = 0;
          end
        end
      end
    end

    loan_type_text = '';
    if (curr = :curr_curr) then
    begin
      if ((pr_loan_type = 0) and
          (emp = 0) and
          (paral = 0)) then
        loan_type_text = 'A Групповой';
      if (paral = 1) then
        loan_type_text = 'C Параллельный';
      if ((pr_loan_type = 1) and
          (emp = 0) and
          (paral = 0)) then
        loan_type_text = 'E Индивидуальный';
      if (emp = 1) then
        loan_type_text = 'G Кред.сотрудникам';
    end
    else
    begin
      if ((pr_loan_type = 0) and
          (emp = 0) and
          (paral = 0)) then
        loan_type_text = 'B Индекс.Групповой';
      if (paral = 1) then
        loan_type_text = 'D Индекс.Параллельный';
      if ((pr_loan_type = 1) and
          (emp = 0) and
          (paral = 0)) then
        loan_type_text = 'F Индекс.Индивидуальный';
      if (emp = 1) then
        loan_type_text = 'H Индекс.Кред.сотрудникам';

      if (cr_rate > curr_rate) then
        zad_os_total = zad_os_total * cr_rate;
      else
        zad_os_total = zad_os_total * curr_rate;
    end

    kd_percent_per_day = 0;
    select kd_percent_per_day
    from kontdata
    where kd_from_k_id = :k_id and
          kd_ol_is_delete = 0 and
          :rep_date between kd_ol_date and kd_date_end
    into kd_percent_per_day;

    suspend;
  end
end
