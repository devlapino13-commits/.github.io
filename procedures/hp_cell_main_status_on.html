CREATE PROCEDURE HP_CELL_MAIN_STATUS_ON(
    C_ID D_ID,
    OK_CELL_MAIN D_STR_100)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable phone d_str_100;
declare variable clt_id d_integer;
declare variable cda_status d_smallint;
declare variable cda_id d_id;
begin
  -- процедура устанавливает contdata.cda_status = 1 у заданного номера телефона клиента (RFCKG-2578)

  res_info_text = '';
  res_error_text = '';

  select f_cont_info('clients', :c_id, 'cell_main', 'ok')
  from rdb$database
  into phone;

  if (coalesce(phone, '') = '') then
    res_error_text = 'Ошибка: основной верифицированный номер телефона клиента не найден';
  else if (ok_cell_main <> phone) then
    res_error_text = 'Ошибка: заданный номер телефона не совпадает с имеющимся в БД';
  else
  begin
    select clt_id
    from contlist
    where clt_code = :ok_cell_main
    into clt_id;

    select cda_id,
           cda_status
    from contdata
    where cda_from_c_id = :c_id and
          cda_from_clt_id = :clt_id and
          cda_is_delete = 0 and
          cda_is_main = 1
    order by cda_date desc
    rows 1
    into cda_id,
         cda_status;

    if (cda_status = 1) then
      res_error_text = 'Ошибка: статус номера уже "Подтвержден", повторно подтверждать нет необходимости';
  end

  if (res_error_text = '') then
  begin
    update contdata
    set cda_status = 1
    where cda_id = :cda_id;

    if (row_count = 1) then
      res_info_text = 'Статус номера телефона успешно подтвержден';
    else
      res_error_text = 'Ошибка: подтвердить номер телефона не удалось';
  end

  suspend;
end
