CREATE PROCEDURE RCIB_CRED_FIP_LIST(
    BASE_RCP_ID ID_TYPE,
    REP_K_ID ID_TYPE,
    REP_MA_ID INTEGER,
    REP_C_ID ID_TYPE,
    REP_DATE_END DATE,
    STEP SMALLINT)
RETURNS (
    REP_DATE_BEGIN DATE,
    K_ID ID_TYPE,
    MA_ID INTEGER,
    MA_NUMBER INTEGER,
    FAILURE_REASON INTEGER,
    FAILURE_REASON_TEXT STR_100,
    PR_ID INTEGER,
    APPLICATION_WAY INTEGER,
    APP_CRED_TYPE INTEGER,
    APP_CRED_CURRENCY VARCHAR(3),
    LOAN_CREDIT INTEGER,
    CRED_TYPE INTEGER,
    CRED_CURRENCY VARCHAR(3),
    APPLICATION_DATE TIMESTAMP,
    APPROVAL_DATE DATE,
    MA_OFFICE_ID INTEGER,
    APP_CRED_DURATION INTEGER,
    REGION_ VARCHAR(10),
    STATUS_DATE DATE,
    STATUS_CODE INTEGER,
    STATUS VARCHAR(20),
    C_ID ID_TYPE,
    APPLICATION_NUM STR_100,
    APP_CRED_SUM DECIMAL(15,2),
    APP_CRED_TYPECB INTEGER,
    CRED_TYPECB INTEGER,
    CRED_MISSEDPAY_120 INTEGER,
    CRED_FULL_PAY INTEGER,
    CRED_SUM DECIMAL(15,2),
    CRED_DURATION INTEGER,
    CRED_ENDDATE DATE,
    CRED_DATE DATE,
    CRED_NO STR_100,
    TYPE_ SMALLINT,
    IS_PORUCH SMALLINT,
    IS_OBESP SMALLINT)
AS
declare variable k_id_old id_type;
declare variable pl_fs_code varchar(10);
declare variable k_kontract_no varchar(50);
declare variable kc_number integer;
declare variable c_id_tmp id_type;
declare variable pr_code varchar(20);
declare variable k_id_real id_type;
declare variable ma_id_tmp integer;
declare variable ma_number_tmp integer;
declare variable cl_not_found integer;
begin
  if (rep_k_id = 0) then
    rep_k_id = null;
  if (rep_ma_id = 0) then
    rep_ma_id = null;
  if (base_rcp_id = 0) then
    base_rcp_id = null;
  if (rep_c_id = 0) then
    rep_c_id = null;

  if (step = 1) then
  begin
    application_way = 3; -- Прямой (прямое обращение в отделение или офис Партнера)
    app_cred_type = 19; -- Микрозайм
    loan_credit = 1; -- Договор займа
    cred_type = 19; -- Микрозайм

    k_id_old = 0;
    for select k_id, status_date, cred_missedpay_120, cred_full_pay, ma_id, failure_reason, failure_reason_text, pr_id,
               status, status_code, k_kontract_no, k_kredline_end, k_currency, rep_date_begin, k_id_real,
               cred_duration
        from rcib_cred_fip_list_temp(:base_rcp_id, :rep_k_id, :rep_ma_id, :rep_date_end)
--where k_ID = 101010006814000101 /* <-- */
        order by k_id, order_no
        into :k_id, :status_date, :cred_missedpay_120, :cred_full_pay, :ma_id, :failure_reason, :failure_reason_text,
             :pr_id, :status, :status_code, :k_kontract_no, :cred_enddate, :cred_currency, :rep_date_begin, :k_id_real,
             :cred_duration
    do
    begin
      if (k_id <> k_id_old) then
      begin
        app_cred_sum = null;

        application_date = null;
        approval_date = null;
        ma_office_id = null;
        app_cred_duration = null;
        pl_fs_code = null;
        region_ = null;
        k_id_old = k_id;

-- т.к. из заявки могут просто удалить учатника группы,
-- а ранее в КИБ он уже был отправлен, важно отправить в киб по нему статус отмены, чтобы этот клиент не осталвся висеть в заявке, в которой его фактически уже нет.

        cl_not_found = 0;

        if ((rep_c_id is not null) and
            (rep_ma_id is not null)) then
        begin
          if (not exists(select *
                         from miniappclient macl
                         where macl.mac_from_ma_id = :rep_ma_id and
                               macl.mac_from_c_id = :rep_c_id and
                               macl.mac_amount > 0)) then
          begin
            failure_reason = 99;
            failure_reason_text = 'Аннулирование для перерегистрации заявки'; --'Кредитная политика заимодавца (кредитора)';
            status_code = 6; -- Отмена
            status = 'annul';

            cl_not_found = 1;
          end
        end

        if (status = 'annul') then
          k_id = k_id_real; -- // Важно!

        cred_sum = null;
        k_kontract_no = null;
        cred_enddate = null;
        kc_number = null;
        cred_date = null;
        cred_no = null;

--if (STATUS in ('disb', 'process', 'annul')) then -- для этих статусов нужны данные по заявке
--begin
        if (ma_id is null) then
        begin
          select kont.k_from_ma_app_id
          from kontract kont
          where kont.k_id = :k_id
          into :ma_id;
        end

        select maa.ma_date_save
        from miniapp maa
        where maa.ma_id = :ma_id and
              maa.ma_number = 0
        into :application_date; -- Дата и время подачи кредитной заявки

        select maa.ma_office_id, ma_kolich_repay, ma_from_pr_id, ma_number, maa.ma_currency_id
        from miniapp maa
        where maa.ma_date_end = '31.12.9999' and
              maa.ma_id = :ma_id
        into :ma_office_id, :app_cred_duration, -- Длительность договора займа (кредита)
             :pr_id, :ma_number, :app_cred_currency;

        app_cred_currency = trim(coalesce(app_cred_currency, ''));
        if (app_cred_currency = '') then
          app_cred_currency = '810';

        if (ma_number <> 0) then
        begin
          select maa.ma_date_save
          from miniapp maa
          where maa.ma_number = 0 -- нужна именно первая запись
                and
                maa.ma_id = :ma_id
          into :application_date; -- Дата и время подачи кредитной заявки
        end

        approval_date = dateadd(day, 30, application_date); -- Дата окончания действия одобрения кредитной заявки.Считаю, что 30 дней

        select substring(pl_code from 4 for 2)
        from region reg
        inner join addrlist adl on adl.adl_id = reg_address_legal
        inner join places pl on pl.pl_id = adl.adl_from_pl_id
        where reg.reg_unicode = :ma_office_id
        into :pl_fs_code;

        select rcr.rcr_cib_code
        from rcib_region rcr
        where rcr.rcr_fs_code = coalesce(:pl_fs_code, '00')
        into :region_; -- Код региона РФ по месту регистрации Офиса  (Регион подачи кредитной заявки)

        if (status = 'annul') then
        begin
          if (cl_not_found = 0) then
          begin
            for select macl.mac_from_c_id, macl.mac_amount
                from miniappclient macl
                inner join clients cl on cl.c_id = macl.mac_from_c_id
                where macl.mac_from_ma_id = :ma_id and
                      macl.mac_amount > 0 and
                      cl.c_pass_date is not null and
                      coalesce(cl.c_pass_who, '') <> ''
                into :c_id, :app_cred_sum
            do
            begin
              application_num = cast(ma_id as varchar(18)) || '-' || cast(c_id as varchar(18));

              if (app_cred_duration <= 12) then
              begin
                if (app_cred_sum < 30000) then
                  app_cred_typecb = 401;
                if (app_cred_sum between 30000 and 100000) then
                  app_cred_typecb = 402;
                if (app_cred_sum between 100000 and 300000) then
                  app_cred_typecb = 403;
                if (app_cred_sum > 300000) then
                  app_cred_typecb = 404;
              end
              else
              begin
                if (app_cred_sum < 30000) then
                  app_cred_typecb = 405;
                if (app_cred_sum between 30000 and 100000) then
                  app_cred_typecb = 406;
                if (app_cred_sum between 100000 and 300000) then
                  app_cred_typecb = 407;
                if (app_cred_sum > 300000) then
                  app_cred_typecb = 408;
              end
              suspend;
            end
          end
          else
          begin
            c_id = null;
            if (exists(select *
                       from clients cl
                       where cl.c_id = :rep_c_id)) then
              c_id = :rep_c_id;
            else
            begin
              select jcl.jc_stay_c_id
              from joinedclients jcl
              where jcl.jc_remove_c_id = :rep_c_id rows 1
              into :c_id;
            end

            application_num = cast(:ma_id as varchar(18)) || '-' || cast(:rep_c_id as varchar(18));

            select rcf.rcf_app_cred_sum, rcf.rcf_app_cred_typecb
            from rcib_fip rcf
            where rcf.rcf_from_ma_id = :rep_ma_id and
                  rcf.rcf_from_c_id = :rep_c_id
            order by rcf.rcf_from_rcp_id desc
            rows 1
            into :app_cred_sum, :app_cred_typecb;

            suspend;
          end
        end
        else
        begin
          cred_typecb = null;
          cred_duration = null;

          if (status in ('disb', 'cont_120', 'closed')) then
          begin
            cred_date = status_date;

            if (k_kontract_no is null) then -- ели значения ниже я не получил в процедуре RCIB_CRED_FIP_LIST_TEMP
            begin
              select kont.k_kontract_no, kont.k_kredline_end, kont.k_currency, round(kont.k_period)
              from kontract kont
              where kont.k_id = :k_id
              into :k_kontract_no, :cred_enddate, :cred_currency, :cred_duration;
            end
          end

          if (position('/' in k_kontract_no) > 0) then
          begin
            k_kontract_no = substring(:k_kontract_no from 1 for (position('/' in k_kontract_no)) - 1);
          end

          for select kncl.kc_from_c_id, kncl.kc_amount_applicate, kc_number, kc_amount_approved
              from knclient kncl
              where kncl.kc_from_k_id = :k_id and
                    kncl.kc_amount_approved > 0
              into :c_id, :app_cred_sum, :kc_number, :cred_sum
          do
          begin
            if (status in ('disb', 'cont_120', 'closed')) then
            begin
              cred_no = k_kontract_no || '(' || cast(kc_number as varchar(10)) || ')';
            end
            else
            begin
              cred_sum = null;
              kc_number = null;
            end

-- application -------------------------------------
            if ((coalesce(app_cred_sum, 0) = 0) and
                (coalesce(cred_sum, 0) > 0)) then
              app_cred_sum = cred_sum;

            application_num = cast(ma_id as varchar(18)) || '-' || cast(c_id as varchar(18));

            if (app_cred_duration <= 12) then
            begin
              if (app_cred_sum < 30000) then
                app_cred_typecb = 401;
              if (app_cred_sum between 30000 and 100000) then
                app_cred_typecb = 402;
              if (app_cred_sum between 100000 and 300000) then
                app_cred_typecb = 403;
              if (app_cred_sum > 300000) then
                app_cred_typecb = 404;
            end
            else
            begin
              if (app_cred_sum < 30000) then
                app_cred_typecb = 405;
              if (app_cred_sum between 30000 and 100000) then
                app_cred_typecb = 406;
              if (app_cred_sum between 100000 and 300000) then
                app_cred_typecb = 407;
              if (app_cred_sum > 300000) then
                app_cred_typecb = 408;
            end
-- credit ------------------------------------------
            if (status in ('disb', 'cont_120', 'closed')) then
            begin
              if (cred_duration <= 12) then
              begin
                if (cred_sum < 30000) then
                  cred_typecb = 401;
                if (cred_sum between 30000 and 100000) then
                  cred_typecb = 402;
                if (cred_sum between 100000 and 300000) then
                  cred_typecb = 403;
                if (cred_sum > 300000) then
                  cred_typecb = 404;
              end
              else
              begin
                if (cred_sum < 30000) then
                  cred_typecb = 405;
                if (cred_sum between 30000 and 100000) then
                  cred_typecb = 406;
                if (cred_sum between 100000 and 300000) then
                  cred_typecb = 407;
                if (cred_sum > 300000) then
                  cred_typecb = 408;
              end
            end
            suspend;
          end
        end
--end
--else if (status in ('cont_120', 'closed')) then -- для этих статусов нет смысла заполнять данные по заявке
--begin

        /*
                    if (k_kontract_no is null) then  -- ели значения ниже я не получил в процедуре RCIB_CRED_FIP_LIST_TEMP
                    begin
                        select kont.k_kontract_no,
                               kont.k_kredline_end,
                               kont.k_currency
                        from kontract kont where kont.k_id = :k_id
                        into :k_kontract_no,
                             :cred_enddate,
                             :cred_currency;
                    end

    
                    for select kncl.kc_from_c_id,
                               kncl.kc_amount_approved,
                               kc_number
                    from knclient kncl
                    where kncl.kc_from_k_id = :k_id
                    and kncl.kc_amount_approved > 0
                    into :c_id,
                         :cred_sum,
                         :kc_number
                    do
                    begin
                        CRED_NO  = k_kontract_no || '(' || cast(kc_number as varchar(10))||')';
    
                        suspend;
                    end  */
--end
      end
    end
  end
  else
  if (step = 2) then
  begin
    for select k_id, ma_id, failure_reason, failure_reason_text, pr_id, application_way, app_cred_type,
               app_cred_currency, loan_credit, cred_type, cred_currency, application_date, approval_date, ma_office_id,
               app_cred_duration, region_, status_date, status_code, status, c_id, application_num, app_cred_typecb,
               cred_missedpay_120, cred_full_pay, cred_sum, cred_enddate, cred_date, cred_no, rep_date_begin, ma_number,
               app_cred_sum, cred_typecb, cred_duration
        from rcib_cred_fip_list(:base_rcp_id, :rep_k_id, :rep_ma_id, :rep_c_id, :rep_date_end, 1)
        into :k_id, :ma_id, :failure_reason, :failure_reason_text, :pr_id, :application_way, :app_cred_type,
             :app_cred_currency, :loan_credit, :cred_type, :cred_currency, :application_date, :approval_date,
             :ma_office_id, :app_cred_duration, :region_, :status_date, :status_code, :status, :c_id_tmp,
             :application_num, :app_cred_typecb, :cred_missedpay_120, :cred_full_pay, :cred_sum, :cred_enddate,
             :cred_date, :cred_no, :rep_date_begin, :ma_number, :app_cred_sum, :cred_typecb, :cred_duration
    do
    begin
      ma_id_tmp = ma_id;
      ma_number_tmp = ma_number;

      if (status in ('cont_120', 'closed', 'disb')) then
      begin

        select pr.pr_code
        from product pr
        where pr.pr_id = :pr_id
        into :pr_code;

        is_obesp = 1;
        /*
                1   Групповой микрозаем
                    01.07.14 - 14.12.14    обеспечен      Приказ №123-06-14 от 25.06.14
                    15.12.14 - 14.01.15    не обеспечен   Приказ №249-12-14 от 08.12.14
                    15.01.15 - 30.06.15    обеспечен      Приказ №274-12-14 от 31.12.14
                    01.07.15 - наст.время  не обеспечен   Приказ №26-06-15 от 25.06.15
    
                */
        if (pr_code containing ('.GR.') and
            ((cred_date between '15.12.2014' and '14.01.2015') or (cred_date >= '01.07.2015'))) then
          is_obesp = 0;
        /*
                2   Микрозаем 1+1
                    15.12.14 - 14.01.15    не обеспечен   Приказ №250-12-14 от 08.12.14
                    15.01.15 - 30.06.15    обеспечен      Приказ №274-12-14 от 31.12.14
                    01.07.15 - наст.время  не обеспечен   Приказ №26-06-15 от 25.06.15
                */
        if ((pr_code containing ('.IND.')) and
            (pr_code containing ('.ONE.')) and
            ((cred_date between '15.12.2014' and '14.01.2015') or (cred_date >= '01.07.2015'))) then
          is_obesp = 0;
        /*
                3.  Индивидуальный с поручительством
                    04.08.14 - 30.06.15    обеспечен      Приказ №157-08-14 от 01.08.14
                    30.06.2015             отменен        Приказ №26-06-15 от 25.06.15
    
                4,5. Семья, Семья Плюс
                    05.10.15 - 21.10.15    не обеспечен   Приказ №23-09-15 от 30.09.15
                                                          Приказ №18-10-15 от 26.10.15
                    22.10.15 - наст.время  обеспечен      Приказ №11/1-10-15 от 19.10.15
                */
        if (((pr_code containing ('.GR.FAMILY.')) or (pr_code containing ('.GR.FAMILY PLUS.'))) and
            (cred_date between '05.10.2015' and '21.10.2015')) then
          is_obesp = 0;
        /*
                6   Баракат
                    05.10.15  - наст.время не обеспечен   Приказ №23-09-15 от 30.09.15
                7   Баракат Плюс
                    05.10.15  - наст.время не обеспечен   Приказ №23-09-15 от 30.09.15
                */

        if (((pr_code containing ('.IND.BARAKAT.')) or (pr_code containing ('.IND.BARAKAT PLUS.'))) and
            (cred_date >= '05.10.2015')) then
          is_obesp = 0;
        /*
                8   Будущий сотрудник
                    01.07.14 - наст.время  обеспечен      Приказ №59-03-14 от 14.03.14
    
                9    Микрозаем сотруднику
                     05.06.15 - наст.время обеспечен      Приказ № 02-06-15 от 01.06.15
                */

        if (is_obesp = 0) then -- если кредит не обчеспечен участники группы друг за друга не отвечают
        begin
          type_ = 0;
          is_poruch = 0;
          c_id = :c_id_tmp;

          suspend;
        end
        else
        begin
-- :type_ = 3, говорит о том что участник группы выступает в качестве поручителя для остальных участников группы
          for select kc_from_c_id, iif(kncl.kc_from_c_id = :c_id_tmp, 0, 3), iif(kncl.kc_from_c_id = :c_id_tmp, 0, 1)
              from knclient kncl
              where kncl.kc_from_k_id = :k_id and
                    kncl.kc_amount_approved > 0
              into :c_id, :type_, :is_poruch
          do
          begin

            ma_id = ma_id_tmp;
            ma_number = ma_number_tmp;
            if (type_ = 3) then -- для получителя не заполняются данные по кредитной заявке
            begin
              ma_id = null;
              ma_number = null;

              /*

                            application_num     = null;
                            application_date    = null;
                            region_             = null;
                            application_way     = null;
                             status_date         = null;
                            approval_date       = null;
                            app_cred_type       = null;
                            app_cred_typecb     = null;
                            app_cred_sum        = null;
                            app_cred_duration   = null;
                            app_cred_currency   = null;
                            failure_reason      = null;
                            failure_reason_text = null;  */
            end

            suspend;
          end
        end
      end
      else
      if (status in ('process', 'annul')) then
      begin
        type_ = 0;
        is_poruch = 0;
        c_id = :c_id_tmp;

        suspend;
      end
    end
  end
end
