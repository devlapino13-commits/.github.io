CREATE PROCEDURE CES_SIGN_CHECK_GET_STATUS(
    CSC_ID D_ID)
RETURNS (
    STATUS D_SMALLINT,
    STATUS_STR D_STR_CODE,
    PERSONIDNP D_STR_20,
    PERSONFIO D_STR_100,
    ORGANIZATIONINN D_STR_20,
    ORGANIZATIONNAME D_STR_100,
    VERIFIED D_SMALLINT,
    TIMESTAMP_VALUE D_STR_20,
    ERROR_CODE D_STR_20,
    ERROR_MESSAGE D_STR_255)
AS
declare variable ESM_ID D_ID;
declare variable JSON_STR D_BLOB_TEXT;
declare variable VERIFIED_STR D_STR_10;
begin
  select
    csc_status,
    csc_from_esm_id
  from ces_sign_check
  where csc_id = :csc_id
  and csc_project = 0
  into
    :status,
    :esm_id;

  status_str =
    case
      when status = 0 then 'запрос в обработке, ожидайте результат ...'
      when status = 1 then 'запрос успешно обработан'
      when status = 2 then 'ошибка обработки запроса'
    end;

  select
    esa_message
  from extsysanswer
  where esa_from_esm_id = :esm_id
  into
    :json_str;

  if (json_str is null) then
  begin
    personIdnp = '';
    personFio = '';
    organizationInn = '';
    organizationName = '';
    verified_str = '';
    timestamp_value = '';
    error_code = '';
    error_message = '';
  end else
  begin
    personIdnp = f_json_blob_get_value(:json_str, 'personIdnp', '');
    personFio = f_json_blob_get_value(:json_str, 'personFio', '');
    organizationInn = f_json_blob_get_value(:json_str, 'organizationInn', '');
    organizationName = f_json_blob_get_value(:json_str, 'organizationName', '');
    verified_str = f_json_blob_get_value(:json_str, 'verified', '');
    timestamp_value = f_json_blob_get_value(:json_str, 'timestamp', '');
    error_code = f_json_blob_get_value(:json_str, 'errorCode', '');
    error_message = f_json_blob_get_value(:json_str, 'errorMessage', '');
  end

  if (verified_str = 'true') then
    verified = 1;
  else
    verified = 0;
  suspend;
end
