CREATE PROCEDURE HP_DO_PERC_TO_OS(
    K_ID D_ID,
    OL_DATE D_DATE,
    OLD_AMOUNT D_CURRENCY,
    NEW_AMOUNT D_CURRENCY,
    ACTION_KIND D_STR_CODE)
AS
declare variable reg_id d_integer;
declare variable pr_id d_integer;
declare variable kf_id d_id;
declare variable last_ol_id d_id;
declare variable main_ol_id d_id;
declare variable opl_acc_kind d_integer;
declare variable al_id d_str_code;
declare variable acc_id d_id;
declare variable ol_memo d_str_long;
declare variable un_id d_id;
declare variable acc_calced d_str_code;
declare variable acc_profit d_str_code;
declare variable acc_payed d_str_code;
declare variable acc_pereplata d_str_code;
declare variable sh_id d_integer;
declare variable sh_date d_date;
declare variable ac_type d_integer;
declare variable total_amount d_currency;
declare variable tmp_amount d_currency;
declare variable ok_amount d_currency;
declare variable total_os d_currency;
declare variable total_pereplata d_currency;
declare variable acc_dt d_str_code;
declare variable acc_kt d_str_code;
declare variable def_curr d_str_code;
declare variable acc_os d_str_code;
declare variable total_balance d_currency;
declare variable total_perc d_currency;
declare variable amount d_currency;
declare variable ol_id d_id;
begin
  if (action_kind not in ('repay', 'type_7')) then
    exception e_default 'Incorrect action_kind: ' || coalesce(action_kind, '<NULL>');

  un_id = 1000000000001;
  if (new_amount < 0)
  then amount = abs(new_amount);
  else amount = abs(new_amount - old_amount);

  if (f_get_db_id() <> 81001) then
    exception e_default 'This procedure only for RU DB';

  def_curr = '810';

  total_os = 0;
  total_pereplata = 0;

  select ok_from_ol_id
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_date <= :ol_date and
        ok_ol_is_active = 1 and
        ok_type = 4 and
        (ok_os <> 0 or ok_perc <> 0)
  order by ok_ol_date desc
  rows 1
  into last_ol_id;

  if (last_ol_id is not null) then
  begin
    select iif(ol_parent_id is null, ol_id, ol_parent_id)
    from operlist
    where ol_id = :last_ol_id
    into main_ol_id;
  
    select opl_acc_kind, opl_from_al_id, opl_from_acc_id
    from operloan
    where opl_from_ol_id = :main_ol_id
    into opl_acc_kind, al_id, acc_id;
  
    if (al_id is not null and acc_id is null) then
    begin
      if (al_id = '010181070-01000001') then
      begin
        select kob_acc_dt
        from kn_operbook, ref_account
        where kob_from_ol_id = :main_ol_id and
              kob_acc_dt = acc_id and
              acc_code like '4742_8100000%'
        order by kob_id
        rows 1
        into acc_id;
      end
  
      if (acc_id is null) then
        exception e_default f_format('Dt acc_id not found, k_id: %1, al_id: %2', k_id, al_id);
    end
  
    if (al_id is null) then
      exception e_default f_format('Dt al_id not found, k_id: %1', k_id);
  end

  select k_from_kf_id, k_kontract_no || ' от ' || f_datetostr(k_disb_date), k_from_pr_id
  from kontract
  where k_id = :k_id
  into kf_id, ol_memo, pr_id;

  select kal_new_reg_id
  from knaktlist k
  where k.kal_from_kf_id = :kf_id and
        k.is_delete = 0 and
        k.kal_date <= :ol_date and
        k.kal_date_end >= :ol_date
  into reg_id;

  ol_memo = 'Корректировка накопленных процентов по договору ' || ol_memo;

  insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind,
                        ol_memo, ol_parent_id, is_delete, ol_user_id)
  values (:reg_id, 1, 13, :ol_date, 9, :ol_memo, null, 0, :un_id)
  returning ol_id
  into :ol_id;

  update or insert into operloan (opl_from_ol_id, opl_acc_kind, opl_from_al_id, opl_from_acc_id)
  values (:ol_id, :opl_acc_kind, :al_id, :acc_id)
  matching (opl_from_ol_id);

  if (action_kind = 'type_7') then
  begin
    amount = old_amount - new_amount;

    select ok_sh_id, ok_sh_date
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_sh_date <= :ol_date and
          ok_ol_date <= :ol_date
    order by ok_ol_is_active desc, ok_sh_date desc, ok_sh_id desc
    rows 1
    into sh_id, sh_date;

    if (sh_id is null) then
      exception e_default f_format('SH_ID not found, k_id: %1', k_id);

    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                          ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                          ok_ol_is_active)
    values (:ol_id, -1, :k_id, :sh_id, :sh_date, 7, 0, :amount, 0,
            3, 0, 0, 0, 0, 0, :ol_date, :reg_id, 1);

  end

  if (action_kind = 'repay') then
  begin
    ac_type = 1; -- Кредитный портфель
    select res_account
    from hp_make_kredit_acclist(:pr_id, :reg_id, :ac_type)
    into :acc_os;

    ac_type = 2; -- Начисленные проценты
    select res_account
    from hp_make_kredit_acclist(:pr_id, :reg_id, :ac_type)
    into :acc_calced;
  
    ac_type = 36; -- Доход процент
    select res_account
    from hp_make_kredit_acclist(:pr_id, :reg_id, :ac_type)
    into :acc_profit;
  
    ac_type = 3; -- Проценты полученные
    select res_account
    from hp_make_kredit_acclist(:pr_id, :reg_id, :ac_type)
    into :acc_payed;
  
    ac_type = 8; -- Переплата
    select res_account
    from hp_make_kredit_acclist(:pr_id, :reg_id, :ac_type)
    into :acc_pereplata;

    if (al_id is not null) then
    begin
      total_amount = amount;
      total_perc = 0;
      for select ok_sh_id, ok_sh_date, sum(-ok_perc)
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_ol_date <= :ol_date and
            ok_type = 4
      group by ok_sh_id, ok_sh_date
      order by ok_sh_date desc, ok_sh_id desc
      into sh_id, sh_date, ok_amount
      do
      begin
        if (total_amount > 0) then
        begin
          if (ok_amount >= total_amount) then
          begin
            tmp_amount = total_amount;
            total_amount = 0;
          end else
          begin
            tmp_amount = ok_amount;
            total_amount = total_amount - ok_amount;
          end
  
          total_perc  = total_perc + tmp_amount;
    
          insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                                ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                                ok_ol_is_active)
          values (:ol_id, -1, :k_id, :sh_id, :sh_date, 4, 0, :tmp_amount, 0,
                  3, 0, 0, 0, 0, 0, :ol_date, :reg_id, 1);
    
          insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                                ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                                ok_ol_is_active)
          values (:ol_id, -1, :k_id, :sh_id, :sh_date, 3, 0, 0, 0,
                  3, 0, 0, -:tmp_amount, 0, 0, :ol_date, :reg_id, 1);
        end
      end
  
      -- Если здесь будет возникать ошибка, рассмотреть вариант корректировки через OK_TYPE = 7
      if (total_amount > 0) then
        exception e_default f_format('Repayed perc is less than diff amount, k_id: %1', k_id);
  
      total_amount = total_perc;
      total_balance = 0;
      for select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os)
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_type <= 7
      group by ok_sh_id, ok_sh_date
      having sum(ok_os + ok_sh_os) > 0
      into sh_id, sh_date, ok_amount
      do
      begin
        total_balance = total_balance + ok_amount;
        if (total_amount > 0) then
        begin
          if (ok_amount >= total_amount) then
          begin
            tmp_amount = total_amount;
            total_amount = 0;
          end else
          begin
            tmp_amount = ok_amount;
            total_amount = total_amount - ok_amount;
          end
    
          total_os = total_os + tmp_amount;
    
          insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                                ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                                ok_ol_is_active)
          values (:ol_id, -1, :k_id, :sh_id, :sh_date, 4, -:tmp_amount, 0, 0,
                  3, 0, 0, 0, 0, 0, :ol_date, :reg_id, 1);
          end
      end
  
      -- Осталась сумма, перекидываем на переплату и закрываем займ
      if (total_amount > 0) then
      begin
        total_pereplata = total_amount;
  
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                              ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
                              ok_ol_is_active)
        values (:ol_id, -1, :k_id, :sh_id, :sh_date, 9, -:total_pereplata, 0, 0,
                3, 0, 0, 0, 0, 0, :ol_date, :reg_id, 1);
      end
  
      if (total_balance = total_os) then
      begin
        if (not exists(select * from kontstate
        where ks_from_k_id = :k_id and
              ks_ol_is_delete = 0 and
              ks_state = 2)) then
        begin
          insert into kontstate (ks_from_ol_id, ks_from_k_id, ks_date_begin, ks_ol_is_delete, ks_state)
          values (:ol_id, :k_id, :ol_date, 0, 2);
        end
      end
    end

    acc_dt = acc_calced;
    acc_kt = acc_profit;
    tmp_amount = -abs(new_amount - old_amount);
    ol_memo = 'Сторнирование излишне начисленного дохода по займу';

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:ol_id, -1, :def_curr, 1, null, :acc_dt, :tmp_amount,
            :tmp_amount, null, null, 0, :acc_kt, null, :ol_memo,
            :ol_date, 1, :reg_id);

    if (al_id is not null) then
    begin
      acc_dt = al_id;
      acc_kt = acc_payed;
      tmp_amount = -total_perc;
      ol_memo = 'Сторнирование излишне полученных процентов';
  
      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                            ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                            ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
      values (:ol_id, -1, :def_curr, 1, null, :acc_dt, :tmp_amount,
              :tmp_amount, null, null, 0, :acc_kt, null, :ol_memo,
              :ol_date, 1, :reg_id);
  
      if (total_os > 0) then
      begin
        acc_dt = al_id;
        acc_kt = acc_os;
        tmp_amount = total_os;
        ol_memo = 'Отражение основной суммы на излишне начисленных и полученных процентов';
    
        insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                              ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                              ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
        values (:ol_id, -1, :def_curr, 1, null, :acc_dt, :tmp_amount,
                :tmp_amount, null, null, 0, :acc_kt, null, :ol_memo,
                :ol_date, 1, :reg_id);
      end
  
      if (total_pereplata > 0) then
      begin
        acc_dt = al_id;
        acc_kt = acc_pereplata;
        tmp_amount = total_pereplata;
        ol_memo = 'Отражение переплаты на сумму излишне начисленных и полученных процентов';
    
        insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                              ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                              ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
        values (:ol_id, -1, :def_curr, 1, null, :acc_dt, :tmp_amount,
                :tmp_amount, null, null, 0, :acc_kt, null, :ol_memo,
                :ol_date, 1, :reg_id);
      end
    end
  end
end
