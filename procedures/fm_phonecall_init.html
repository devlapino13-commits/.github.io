CREATE PROCEDURE FM_PHONECALL_INIT(
    C_ID D_ID,
    PHONE_ID D_ID,
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    UN_ID D_ID,
    IS_CONTPERSON D_BOOL = 0)
RETURNS (
    PNC_ID D_ID,
    ERR_TEXT D_STR_255,
    PRV_PHONE_NO D_STR_CODE)
AS
declare variable pnc_src_c_id d_id;
declare variable pnc_src_phone_id d_integer;
declare variable is_delinq d_bool;
declare variable msg_body d_str_long;
declare variable cl_fio str_100;
declare variable address_str str_255;
declare variable msk_call_time_start d_time;
declare variable msk_call_time_end d_time;
begin
  err_text = '';
  is_contperson = coalesce(is_contperson, 0);

  select un_from_c_id
  from username
  where un_id = :un_id
  into pnc_src_c_id;

  if (not exists(select * from hp_db_is_actual)) then
  begin
    err_text = 'Вы подключены к тестовой базе. Звонки недоступны.';
    suspend;
    exit;
  end

  select msk_call_time_start, msk_call_time_end
  from hp_places_get_timezone_c(:c_id, f_cont_code(:phone_id))
  into msk_call_time_start, msk_call_time_end;

  if (msk_call_time_start is null) then
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into cl_fio;

    select adl_address_str
    from addrclient, addrlist
    where adl_id = adc_from_adl_id and
          adc_from_c_id = :c_id and
          adc_address_type = 1
    into address_str;

    msg_body = f_format('<b>Источник:</b> МП<br> <b>ФИО клиента:</b> %1<br> <b>Адрес регистрации клиента:</b> %2<br> <b>Номер телефона:</b> %3<br><b>Текущее время:</b> %4', cl_fio, address_str, f_cont_info('clients', :c_id, 'cell_main', 'ok'), f_datetimetostr(current_timestamp, 'dd.mm.yyyy hh:nn:ss'));

    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                msg_command_kind)
    values (0, 'i.onischenko@molbulak.com',
            'Часовой пояс клиента не определен. Звонок невозможен', :msg_body, 1, 1000000000001, '');

    err_text = 'Часовой пояс клиента не определен в связи с некорректным юр.адресом клиента. Звонок невозможен.';
    suspend;
    exit;
  end

  if (msk_call_time_start is not null and
      cast(current_time as time) not between msk_call_time_start and msk_call_time_end) then
  begin
    err_text = 'Совершать звонки необходимо в период с 9:00 до 20:00';
    suspend;
    exit;
  end

  -- Находим номер телефона пользователя
  pnc_src_phone_id = f_cont_id(f_cont_info('clients', pnc_src_c_id, 'cell_main', 'ok'), 'null');

  if (coalesce(pnc_src_phone_id, 0) = 0) then
  begin
    err_text = 'У вас нет подтвержденного основного номера телефона, звонок невозможен';
    suspend;
    exit;
  end

  is_delinq = 0;
  if (obj_kind = 9) then
  begin
    select is_delinq
    from hp_is_client_delinqed(:c_id, :obj_id)
    into is_delinq;
  end
  else if (obj_kind = 0) then
  begin
    select is_delinq
    from hp_is_client_delinqed(:c_id)
    into is_delinq;
  end

  if (is_delinq = 1) then
  begin
    err_text = 'Клиент в просрочке, звонки запрещены.';
    suspend;
    exit;
  end

  if (exists(
      select *
      from phonecalls
      where pnc_dst_c_id = :c_id
        and pnc_dst_phone_id = :phone_id
        and pnc_call_result_id = f_enum_id('phone_call_result', 'wrong_number')
     )) then
  begin
    err_text = 'Номер заблокирован для звонков так как ранее был установлен статус "Чужой номер"';
    suspend;
    exit;
  end

  -- № телефона провайдера генерируется случайным образом
  -- У Oivo и MBulak разные пулы номеров
  if (f_is_employee_oivo(pnc_src_c_id, current_date) = 1) then
    prv_phone_no = '74952258057';
  else
    prv_phone_no = decode(round(rand() * 7),
      0, '74952216801',
      1, '74952216802',
      2, '74952217205',
      3, '74952217319',
      4, '74952217829',
      5, '74952217841',
      6, '74952258063',
      7, '74952258089');

  insert into phonecalls (
    pnc_src_phone_id, pnc_dst_phone_id, pnc_prv_phone_id, pnc_src_c_id,
    pnc_dst_c_id, pnc_direction, pnc_object_kind, pnc_object_id, pnc_is_delinq,
    pnc_is_contperson)
  values (:pnc_src_phone_id, :phone_id, f_cont_id(:prv_phone_no, 'add'),
    :pnc_src_c_id, :c_id, 1, :obj_kind, :obj_id, :is_delinq, :is_contperson)
  returning pnc_id
  into pnc_id;

  suspend;
end
