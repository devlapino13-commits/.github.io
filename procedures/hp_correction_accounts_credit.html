CREATE PROCEDURE HP_CORRECTION_ACCOUNTS_CREDIT(
    REP_DATE_END DATE)
RETURNS (
    OL_ID ID_TYPE,
    K_ID ID_TYPE,
    OK_AMMOUNT CURR_TYPE,
    OL_DATE DATE,
    REGION_ID INTEGER,
    OLD_AL_ID VARCHAR(20),
    PR_LOAN_TYPE INTEGER,
    PR_PRODUCT_KIND INTEGER,
    CURR_CODE VARCHAR(3),
    OLD_ACCOUNT VARCHAR(128),
    NEW_ACCOUNT VARCHAR(128),
    NEW_AL_ID VARCHAR(20),
    MEMO VARCHAR(128),
    IS_CHECK_OPERBOOK SMALLINT)
AS
begin
--  Выдача
  memo = 'Выдача';
  for with tabl1
      as (select ok_from_ol_id, ok_from_k_id, ok_os, ok_ol_date, ok_ol_reg_id
          from operkred
          where ok_ol_is_active = 1 and
                ok_ol_date between '01.01.2014' and :rep_date_end and
                ok_type = 8 and
                ok_from_c_id = 0 and
                ok_os <> 0),
      tabl2
      as (select tabl1.*, sa_account, pr_loan_type, pr_product_kind, k_index_curr_code,
                 (select al_account
                  from acc_list
                  where al_id = sa_account),
                 iif(pr_product_kind in (1, 2), '10931/1', iif(pr_product_kind = 3 or (pr_product_kind = 4 and k_index_curr_code = '417'), '10941/1', iif(pr_product_kind = 4 and k_index_curr_code <> '417', '10941/10', ''))) new_account
          from tabl1, kontract, product, standacc
          where ok_from_k_id = k_id and
                k_from_pr_id = pr_id and
                pr_id = sa_from_pr_id and
                sa_from_reg_id = ok_ol_reg_id and
                sa_account_type = 1)
      select tabl2.*,
             (select al_id
              from acc_list
              where al_account = new_account and
                    al_from_reg_id = ok_ol_reg_id and
                    al_currency = k_index_curr_code) new_al_id
      from tabl2
      where al_account <> new_account
      into ol_id, k_id, ok_ammount, ol_date, region_id, old_al_id, pr_loan_type, pr_product_kind, curr_code,
           old_account, new_account, new_al_id
  do
  begin
    is_check_operbook = 0;
    select 1
    from operbook
    where ob_from_ol_id = :ol_id and
          ob_account_dt = :old_al_id and
          ob_summa_val = :ok_ammount
    into is_check_operbook;
    suspend;
  end
--  Погашение ОС
  memo = 'Погашение ОС';
  is_check_operbook = 0;
  for with tabl1
      as (select ok_from_ol_id, ok_from_k_id, sum(ok_os) ok_os, ok_ol_date, ok_ol_reg_id
          from operkred
          where ok_ol_is_active = 1 and
                ok_ol_date between '01.01.2014' and :rep_date_end and
                ok_type = 4 and
                ok_os <> 0 and
                ok_from_c_id = 0
          group by ok_from_ol_id, ok_from_k_id, ok_ol_date, ok_ol_reg_id),
      tabl2
      as (select tabl1.*, sa_account, pr_loan_type, pr_product_kind, k_index_curr_code,
                 (select al_account
                  from acc_list
                  where al_id = sa_account),
                 iif(pr_product_kind in (1, 2), '10931/1', iif(pr_product_kind = 3 or (pr_product_kind = 4 and k_index_curr_code = '417'), '10941/1', iif(pr_product_kind = 4 and k_index_curr_code <> '417', '10941/10', ''))) new_account
          from tabl1, kontract, product, standacc
          where ok_from_k_id = k_id and
                k_from_pr_id = pr_id and
                pr_id = sa_from_pr_id and
                sa_from_reg_id = ok_ol_reg_id and
                sa_account_type = 1)
      select tabl2.*,
             (select al_id
              from acc_list
              where al_account = new_account and
                    al_from_reg_id = ok_ol_reg_id and
                    al_currency = k_index_curr_code) new_al_id
      from tabl2
      where al_account <> new_account
      into ol_id, k_id, ok_ammount, ol_date, region_id, old_al_id, pr_loan_type, pr_product_kind, curr_code,
           old_account, new_account, new_al_id
  do
  begin
    is_check_operbook = 0;
    select 1
    from operbook
    where ob_from_ol_id = :ol_id and
          ob_account_kt = :old_al_id and
          ob_summa_val = -:ok_ammount
    into is_check_operbook;
    suspend;
  end
--  Погашение процентов
  memo = 'Полученные проценты';
  is_check_operbook = 0;
  for with tabl1
      as (select ok_from_ol_id, ok_from_k_id, sum(ok_perc) ok_perc, ok_ol_date, ok_ol_reg_id
          from operkred
          where ok_ol_is_active = 1 and
                ok_ol_date between '01.01.2014' and :rep_date_end and
                ok_type = 4 and
                ok_perc <> 0 and
                ok_from_c_id = 0
          group by ok_from_ol_id, ok_from_k_id, ok_ol_date, ok_ol_reg_id),
      tabl2
      as (select tabl1.*, sa_account, pr_loan_type, pr_product_kind, k_index_curr_code,
                 (select al_account
                  from acc_list
                  where al_id = sa_account),
                 iif(pr_product_kind in (1, 2), '21141/1', iif(pr_product_kind = 3 or (pr_product_kind = 4 and k_index_curr_code = '417'), '21141/2', iif(pr_product_kind = 4 and k_index_curr_code <> '417', '21141/10', ''))) new_account
          from tabl1, kontract, product, standacc
          where ok_from_k_id = k_id and
                k_from_pr_id = pr_id and
                pr_id = sa_from_pr_id and
                sa_from_reg_id = ok_ol_reg_id and
                sa_account_type = 3)
      select tabl2.*,
             (select al_id
              from acc_list
              where al_account = new_account and
                    al_from_reg_id = ok_ol_reg_id and
                    al_currency = k_index_curr_code) new_al_id
      from tabl2
      where al_account <> new_account
      into ol_id, k_id, ok_ammount, ol_date, region_id, old_al_id, pr_loan_type, pr_product_kind, curr_code,
           old_account, new_account, new_al_id
  do
  begin
--    Тут имеются еще досрочные погашении, при создании скрипта если общая сумма ровняется к аналитике то по ним нужно счета менять
    is_check_operbook = 0;
    if ((select -sum(ob_summa_val)
         from operbook
         where ob_from_ol_id = :ol_id and
               ob_account_kt = :old_al_id) = :ok_ammount) then
      is_check_operbook = 1;
    suspend;
  end
--  Погашения пени
  memo = 'Полученные штрафы';
  is_check_operbook = 0;
  for with tabl1
      as (select ok_from_ol_id, ok_from_k_id, sum(ok_fine) ok_fine, ok_ol_date, ok_ol_reg_id
          from operkred
          where ok_ol_is_active = 1 and
                ok_ol_date between '01.01.2014' and :rep_date_end and
                ok_type = 4 and
                ok_fine <> 0 and
                ok_from_c_id = 0
          group by ok_from_ol_id, ok_from_k_id, ok_ol_date, ok_ol_reg_id),
      tabl2
      as (select tabl1.*, sa_account, pr_loan_type, pr_product_kind, k_index_curr_code,
                 (select al_account
                  from acc_list
                  where al_id = sa_account),
                 iif(pr_product_kind in (1, 2), '60631/1', iif(pr_product_kind = 3 or (pr_product_kind = 4 and k_index_curr_code = '417'), '60631/2', iif(pr_product_kind = 4 and k_index_curr_code <> '417', '60631/2', ''))) new_account
          from tabl1, kontract, product, standacc
          where ok_from_k_id = k_id and
                k_from_pr_id = pr_id and
                pr_id = sa_from_pr_id and
                sa_from_reg_id = ok_ol_reg_id and
                sa_account_type = 5)
      select tabl2.*,
             (select al_id
              from acc_list
              where al_account = new_account and
                    al_from_reg_id = ok_ol_reg_id and
                    al_currency = k_index_curr_code) new_al_id
      from tabl2
      where al_account <> new_account
      into ol_id, k_id, ok_ammount, ol_date, region_id, old_al_id, pr_loan_type, pr_product_kind, curr_code,
           old_account, new_account, new_al_id
  do
  begin
    is_check_operbook = 0;
    select 1
    from operbook
    where ob_from_ol_id = :ol_id and
          ob_account_kt = :old_al_id and
          ob_summa_val = -:ok_ammount
    into is_check_operbook;
    suspend;
  end
--  Переплата
  memo = 'Переплата по ОС';
  is_check_operbook = 0;
  for with tabl1
      as (select ok_from_ol_id, ok_from_k_id, sum(ok_os) ok_os, ok_ol_date, ok_ol_reg_id
          from operkred
          where ok_ol_is_active = 1 and
                ok_ol_date between '01.01.2014' and :rep_date_end and
                ok_type = 9 and
                ok_os <> 0 and
                ok_from_c_id = 0
          group by ok_from_ol_id, ok_from_k_id, ok_ol_date, ok_ol_reg_id),
      tabl2
      as (select tabl1.*, sa_account, pr_loan_type, pr_product_kind, k_index_curr_code,
                 (select al_account
                  from acc_list
                  where al_id = sa_account),
                 iif(pr_product_kind in (1, 2), '21199/1', iif(pr_product_kind = 3 or (pr_product_kind = 4 and k_index_curr_code = '417'), '21199/1', iif(pr_product_kind = 4 and k_index_curr_code <> '417', '21199/1', ''))) new_account
          from tabl1, kontract, product, standacc
          where ok_from_k_id = k_id and
                k_from_pr_id = pr_id and
                pr_id = sa_from_pr_id and
                sa_from_reg_id = ok_ol_reg_id and
                sa_account_type = 8)
      select tabl2.*,
             (select al_id
              from acc_list
              where al_account = new_account and
                    al_from_reg_id = ok_ol_reg_id and
                    al_currency = k_index_curr_code) new_al_id
      from tabl2
      where al_account <> new_account
      into ol_id, k_id, ok_ammount, ol_date, region_id, old_al_id, pr_loan_type, pr_product_kind, curr_code,
           old_account, new_account, new_al_id
  do
  begin
    is_check_operbook = 0;
    select 1
    from operbook
    where ob_from_ol_id = :ol_id and
          ob_account_kt = :old_al_id and
          ob_summa_val = -:ok_ammount
    into is_check_operbook;
    suspend;
  end
--  Полученные комиссионные (ДБП)
  memo = 'Полученные комиссионные (ДБП)';
  is_check_operbook = 0;
  for with tabl1
      as (select ok_from_ol_id, ok_from_k_id, sum(ok_kommis) ok_kommis, ok_ol_date, ok_ol_reg_id
          from operkred
          where ok_ol_is_active = 1 and
                ok_ol_date between '01.01.2014' and :rep_date_end and
                ok_type = 4 and
                ok_kommis <> 0 and
                ok_from_c_id = 0
          group by ok_from_ol_id, ok_from_k_id, ok_ol_date, ok_ol_reg_id),
      tabl2
      as (select tabl1.*, sa_account, pr_loan_type, pr_product_kind, k_index_curr_code,
                 (select al_account
                  from acc_list
                  where al_id = sa_account),
                 iif(pr_product_kind in (1, 2), '21141/101', iif(pr_product_kind = 3 or (pr_product_kind = 4 and k_index_curr_code = '417'), '21141/102', iif(pr_product_kind = 4 and k_index_curr_code <> '417', '21141/102', ''))) new_account
          from tabl1, kontract, product, standacc
          where ok_from_k_id = k_id and
                k_from_pr_id = pr_id and
                pr_id = sa_from_pr_id and
                sa_from_reg_id = ok_ol_reg_id and
                sa_account_type = 7)
      select tabl2.*,
             (select al_id
              from acc_list
              where al_account = new_account and
                    al_from_reg_id = ok_ol_reg_id and
                    al_currency = k_index_curr_code) new_al_id
      from tabl2
      where al_account <> new_account
      into ol_id, k_id, ok_ammount, ol_date, region_id, old_al_id, pr_loan_type, pr_product_kind, curr_code,
           old_account, new_account, new_al_id
  do
  begin
    is_check_operbook = 0;
    select 1
    from operbook
    where ob_from_ol_id = :ol_id and
          ob_account_kt = :old_al_id and
          ob_summa_val = -:ok_ammount
    into is_check_operbook;
    suspend;
  end
end
