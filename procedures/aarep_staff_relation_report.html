CREATE PROCEDURE AAREP_STAFF_RELATION_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    REG_NAME VARCHAR(100),
    JL_NAME VARCHAR(100),
    REL_0_FAMILY INTEGER,
    REL_3_FATHER INTEGER,
    REL_4_MOTHER INTEGER,
    REL_5_SUN INTEGER,
    REL_6_DAUGHTER INTEGER,
    REL_7_BROTHER INTEGER,
    REL_8_SISTER INTEGER,
    REL_OTHERS INTEGER,
    REL_TOTAL_OWN INTEGER,
    REL_TOTAL_ALL INTEGER,
    REL_STATUS VARCHAR(100))
AS
declare variable c_id decimal(18,0);
declare variable rel_code varchar(50);
begin
  for select c_name_fio, c_id, reg_code || ' ' || reg_name, jl_name
      from hr_pstatus, clients, region, joblist, region_list(:rep_region_id, 1, -1, 1)
      where ps_from_c_id = c_id and
            ps_from_reg_id = reg_unicode and
            ps_from_reg_id = id_list and
            ps_from_jl_id = jl_id and
            ps_is_working = 1 and
            ps_status = 1 and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end
      order by 3, 1
      into :c_name_fio, :c_id, :reg_name, :jl_name
  do
  begin
-- 1  Супруг
-- 2  Супруга
-- 3  Отец
-- 4  Мать
-- 5  Сын
-- 6  Дочь
-- 7  Брат
-- 8  Сестра

    rel_0_family = 0;
    rel_3_father = 0;
    rel_4_mother = 0;
    rel_5_sun = 0;
    rel_6_daughter = 0;
    rel_7_brother = 0;
    rel_8_sister = 0;
    rel_others = 0;

    rel_total_own = 0;
    rel_total_all = 0;

    for select rel_code
        from hp_client_relation_tmp(:c_id, 4)
        into :rel_code
    do
    begin
      if (rel_code = '1') then
        rel_0_family = rel_0_family + 1;
      else
      if (rel_code = '2') then
        rel_0_family = rel_0_family + 1;
      else
      if (rel_code = '3') then
        rel_3_father = rel_3_father + 1;
      else
      if (rel_code = '4') then
        rel_4_mother = rel_4_mother + 1;
      else
      if (rel_code = '5') then
        rel_5_sun = rel_5_sun + 1;
      else
      if (rel_code = '6') then
        rel_6_daughter = rel_6_daughter + 1;
      else
      if (rel_code = '7') then
        rel_7_brother = rel_7_brother + 1;
      else
      if (rel_code = '8') then
        rel_8_sister = rel_8_sister + 1;
      else
        rel_others = rel_others + 1;
    end
    rel_total_own = rel_0_family + rel_3_father + rel_4_mother + rel_5_sun + rel_6_daughter + rel_7_brother + rel_8_sister;
    rel_total_all = rel_total_own + rel_others;

    rel_status = 'Все заполнено';
    if (rel_0_family = 0) then
    begin
      if (not(exists(select *
                     from clientrelabsent
                     where cra_from_c_id = :c_id and
                           cra_family = 1))) then
      begin
        rel_0_family = null;
        rel_status = 'Есть пустые';
      end
    end
    if (rel_3_father = 0) then
    begin
      if (not(exists(select *
                     from clientrelabsent
                     where cra_from_c_id = :c_id and
                           cra_3 = 1))) then
      begin
        rel_3_father = null;
        rel_status = 'Есть пустые';
      end
    end
    if (rel_4_mother = 0) then
    begin
      if (not(exists(select *
                     from clientrelabsent
                     where cra_from_c_id = :c_id and
                           cra_4 = 1))) then
      begin
        rel_4_mother = null;
        rel_status = 'Есть пустые';
      end
    end
    if (rel_5_sun = 0) then
    begin
      if (not(exists(select *
                     from clientrelabsent
                     where cra_from_c_id = :c_id and
                           cra_5 = 1))) then
      begin
        rel_5_sun = null;
        rel_status = 'Есть пустые';
      end
    end
    if (rel_6_daughter = 0) then
    begin
      if (not(exists(select *
                     from clientrelabsent
                     where cra_from_c_id = :c_id and
                           cra_6 = 1))) then
      begin
        rel_6_daughter = null;
        rel_status = 'Есть пустые';
      end
    end
    if (rel_7_brother = 0) then
    begin
      if (not(exists(select *
                     from clientrelabsent
                     where cra_from_c_id = :c_id and
                           cra_7 = 1))) then
      begin
        rel_7_brother = null;
        rel_status = 'Есть пустые';
      end
    end
    if (rel_8_sister = 0) then
    begin
      if (not(exists(select *
                     from clientrelabsent
                     where cra_from_c_id = :c_id and
                           cra_8 = 1))) then
      begin
        rel_8_sister = null;
        rel_status = 'Есть пустые';
      end
    end
    suspend;
  end
end
