CREATE PROCEDURE AAREP_CELLAKTLIST_ONE(
    CELL_NO VARCHAR(20),
    REP_DATE DATE,
    IS_MOUNT D_BOOL = 0)
RETURNS (
    CAL_FROM_CAM_ID DECIMAL(18,0),
    CAL_ID INTEGER,
    CAL_DATE_BEGIN DATE,
    CAL_DATE_END DATE,
    CAL_FROM_C_ID DECIMAL(18,0),
    CAL_CELL_NO VARCHAR(20),
    CAL_STATE INTEGER,
    CAL_IS_CONTACT INTEGER,
    CAL_IS_CORPORATE INTEGER,
    CAL_HAS_LIMIT INTEGER,
    CAL_FROM_COP_ID INTEGER,
    CAL_PS_TEXT VARCHAR(255),
    CAL_FIO VARCHAR(100),
    CAL_REGION_ID INTEGER,
    CAL_BRANCH_ID INTEGER,
    CAL_FROM_JL_ID INTEGER,
    CAL_FROM_JL_NAME VARCHAR(100),
    CAL_REGION_NAME VARCHAR(100),
    CAL_IS_WORKING INTEGER,
    CLK_LIMIT_KIND VARCHAR(100),
    CLK_LIMIT_AMOUNT DECIMAL(18,4),
    CAL_NUMBER_WITH_OPERATOR D_STR_100)
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select cal_from_cam_id, cal_id, cal_date_begin, cal_date_end, cal_from_c_id, cal_cell_no, cal_state,
             cal_is_contact, cal_is_corporate, cal_has_limit, cal_from_cop_id, clk_name, clk_limit_amount, cal_number_with_operator
      from cellaktlist, cellimitkind
      where cal_cell_no = :cell_no and
            cal_date_begin <= :rep_date and
            (
              ((:is_mount = 1) and (extract(year from cal_date_end) = extract(year from :rep_date)) and (extract(month from cal_date_end) = extract(month from :rep_date)))
              or (cal_date_end >= :rep_date)
            ) and
            clk_id = cal_has_limit
      order by cal_date_begin desc
      into :cal_from_cam_id, :cal_id, :cal_date_begin, :cal_date_end, :cal_from_c_id, :cal_cell_no, :cal_state,
           :cal_is_contact, :cal_is_corporate, :cal_has_limit, :cal_from_cop_id, :clk_limit_kind, :clk_limit_amount, :cal_number_with_operator
  do
  begin
    select ps_text, c_name_fio, ps_from_reg_id, ps_branch_id, ps_jl_name, ps_reg_name, ps_from_jl_id, ps_is_working
    from hr_personal_last_position(:cal_from_c_id, :rep_date), clients
    where c_id = :cal_from_c_id rows 1
    into :cal_ps_text, :cal_fio, :cal_region_id, :cal_branch_id, :cal_from_jl_name, :cal_region_name, :cal_from_jl_id,
         cal_is_working;

    suspend;
  end
end
