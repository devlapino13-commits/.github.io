CREATE PROCEDURE OKT_PANTERA_CONTACTS_LIST(
    PANT_ID ID_TYPE,
    C_ID ID_TYPE)
RETURNS (
    CC_STATUS VARCHAR(50),
    CC_TYPE VARCHAR(50),
    CC_NAME STR_100,
    CELL_NO VARCHAR(50),
    CALL_RES STR_100,
    OLD_CONTACT INTEGER,
    CC_ID INTEGER,
    PDS_TRY_QTY INTEGER,
    PDS_CALL_QTY INTEGER,
    PDS_LAST_CALL_DATE DATE,
    PDS_LAST_TRY_DATE DATE,
    PDS_MEMO_QTY INTEGER,
    HAS_MEMO SMALLINT,
    CONTACT_TYPE SMALLINT,
    PORUCH_ID ID_TYPE,
    NEXT_CALL_DATE TIMESTAMP,
    CAN_CALL INTEGER,
    REASON STR_255)
AS
declare variable K_ID ID_TYPE;
begin
  poruch_id = 0;
  contact_type = 0; --  contacts

  select pantk.pant_from_k_id
  from pantera_kontracts pantk
  where pantk.pant_id = :pant_id
  into :k_id;

  for select cont.cc_id, cont.cc_status,
             (
             case cont.cc_type
               when 0 then 'РФ'
               else 'Родина'
             end),
             cont.cc_name, cont.cc_cell_no,
             (
             case cont.cc_total_res
               when -1 then 'не дозвонился'
               when 0 then 'не обработан'
               when 1 then 'подтвержден'
               when -2 then 'не подтвержден'
             end),
             cont.cc_is_old_contact
      from contacts cont
      where cont.cc_from_c_id = :c_id
      order by cc_is_old_contact desc
      into cc_id, cc_status, cc_type, cc_name, cell_no, call_res, old_contact
  do
  begin
    if ((trim(coalesce(cc_name, '')) <> '') and
        (trim(coalesce(cell_no, '')) <> '')) then
    begin
      pds_try_qty = 0;
      pds_call_qty = 0;
      pds_memo_qty = 0;
      pds_last_call_date = null;
      pds_last_try_date = null;
      next_call_date = null;

      select pds.pds_try_qty, pds.pds_call_qty, pds.pds_last_call_date, pds.pds_last_try_date, pds.pds_memo_qty,
             pds.pds_next_call_datetime
      from pantera_data_svod pds
      where pds.pds_from_pan_id = :pant_id and
             pds.pds_from_cc_id = :cc_id
      into :pds_try_qty, :pds_call_qty, :pds_last_call_date, :pds_last_try_date, :pds_memo_qty, :next_call_date;

      if (pds_try_qty is null) then
        pds_try_qty = 0;
      if (pds_call_qty is null) then
        pds_call_qty = 0;
      if (pds_memo_qty is null) then
        pds_memo_qty = 0;

      has_memo = 0;
      if (pds_memo_qty > 0) then
        has_memo = 1;

      select can_call, reason
      from hp_can_call_check(:k_id, 'Contacts', :cc_id, null)
      into :can_call, reason;
      suspend;
    end
  end

  cc_status = 'Поручитель';
  contact_type = 1; --  поручители
  cc_id = 0;

  for select cl.c_name_fio, f_cont_info('clients', cl.c_id, 'cell_main', 'last'), cl.c_id
      from zalog zal
      inner join pantera_kontracts pantk on zal.z_from_k_id = pantk.pant_from_k_id
      inner join clients cl on cl.c_id = zal.z_from_c_id
      where pantk.pant_id = :pant_id
      into :cc_name, :cell_no, :poruch_id
  do
  begin
    cc_type = '-';
    call_res = '-';
    old_contact = 0;

    pds_try_qty = 0;
    pds_call_qty = 0;
    pds_memo_qty = 0;
    pds_last_call_date = null;
    pds_last_try_date = null;
    next_call_date = null;

    select pds.pds_try_qty, pds.pds_call_qty, pds.pds_last_call_date, pds.pds_last_try_date, pds.pds_memo_qty,
           pds.pds_next_call_datetime
    from pantera_data_svod pds
    where pds.pds_from_pan_id = :pant_id and
          pds.pds_from_poruch_id = :poruch_id
    into :pds_try_qty, :pds_call_qty, :pds_last_call_date, :pds_last_try_date, :pds_memo_qty, :next_call_date;

    if (pds_try_qty is null) then
      pds_try_qty = 0;
    if (pds_call_qty is null) then
      pds_call_qty = 0;
    if (pds_memo_qty is null) then
      pds_memo_qty = 0;

    has_memo = 0;
    if (pds_memo_qty > 0) then
      has_memo = 1;

    select can_call, reason
    from hp_can_call_check(:k_id, 'Clients', :poruch_id, null)
    into :can_call, reason;

    suspend;
  end
end
