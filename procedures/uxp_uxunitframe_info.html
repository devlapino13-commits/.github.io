Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXUNITFRAME_INFO(
    UNIT_UUID D_UUID,
    USR_UUID D_UUID)
RETURNS (
    UNF_UUID D_UUID,
    UNF_OWNER D_UUID,
    UNF_USR_UUID D_UUID,
    UNF_CODE D_STR_CODE,
    UNF_NAME D_STR_100,
    UNF_IS_MAINFRAME D_BOOL,
    UNF_IS_GRID D_BOOL,
    UNF_TREE_PARENT_FIELD D_STR_CODE,
    UNF_MASTER_UNF_CODE D_STR_CODE,
    UNF_MASTER_FIELD D_STR_CODE,
    UNF_DETAIL_FIELD D_STR_CODE,
    UNF_SQL_SELECT D_STR_4096,
    UNF_SQL_UPDATE D_STR_4096,
    UNF_SQL_INSERT D_STR_4096,
    UNF_SQL_DELETE D_STR_LONG)
AS
begin
  for select unf_uuid,
             unf_owner,
             unf_usr_uuid,
             unf_code,
             unf_name,
             unf_is_mainframe,
             unf_is_grid,
             unf_tree_parent_field,
             unf_master_unf_code,
             unf_master_field,
             unf_detail_field,
             unf_sql_select,
             unf_sql_update,
             unf_sql_insert,
             unf_sql_delete
      from ux$unitframe
      where unf_owner = :unit_uuid and
            unf_usr_uuid = :usr_uuid
      into :unf_uuid,
           :unf_owner,
           :unf_usr_uuid,
           :unf_code,
           :unf_name,
           :unf_is_mainframe,
           :unf_is_grid,
           :unf_tree_parent_field,
           :unf_master_unf_code,
           :unf_master_field,
           :unf_detail_field,
           :unf_sql_select,
           :unf_sql_update,
           :unf_sql_insert,
           :unf_sql_delete
  do
  begin
    suspend;
  end
end
