CREATE PROCEDURE HP_SUSN_QUERY(
    MA_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable C_ID D_ID;
declare variable C_INN D_STR_20;
declare variable EIC_ID D_ID;
declare variable OBJ_KIND_MA D_INTEGER;
declare variable TRY_COUNT D_INTEGER;
declare variable MAX_TRY D_INTEGER;
declare variable LAST_STATUS D_SMALLINT;
declare variable LAST_DT timestamp;
declare variable MINS_LEFT D_INTEGER;
begin
  -- RFCKZ-996
  res_text = '';
  max_try = 4;                                  -- 1 автозапрос + 3 ручных
  obj_kind_ma = f_objkind('miniapp', 'id');

  eic_id = (select eic.eic_id
            from extinfocfg eic
            where eic.eic_code = 'susn');

  if (eic_id is null) then
  begin
    res_text = 'Сервис СУСН не настроен';
    suspend;
    exit;
  end

  select first 1 mac.mac_from_c_id
  from miniappclient mac
  where mac.mac_from_ma_id = :ma_id
    and mac.mac_amount > 0
  into c_id;

  if (c_id is null) then
  begin
    res_text = 'Заявка не найдена';
    suspend;
    exit;
  end

  select coalesce(cl.c_inn, '')
  from clients cl
  where cl.c_id = :c_id
  into c_inn;

  if (c_inn = '') then
  begin
    res_text = 'У клиента отсутствует ИИН';
    suspend;
    exit;
  end

  select count(*)
  from extinfolist eil
  where eil.eil_from_eic_id = :eic_id
    and eil.eil_object_kind = :obj_kind_ma
    and eil.eil_object_id = :ma_id
  into try_count;

  select first 1 eil.eil_status, eil.eil_status_datetime
  from extinfolist eil
  where eil.eil_from_eic_id = :eic_id
    and eil.eil_object_kind = :obj_kind_ma
    and eil.eil_object_id = :ma_id
  order by eil.eil_id desc
  into last_status, last_dt;

  if (last_status in (-1, 0, 1)) then
  begin
    res_text = 'Запрос уже отправлен, ожидается ответ';
    suspend;
    exit;
  end

  if (try_count >= max_try) then
  begin
    res_text = 'Сервис СУСН не отвечает';
    suspend;
    exit;
  end

  -- ПКБ возвращает 400 DUPLICATE на повторный запрос по тому же ИИН
  -- в течение 10 минут
  if (last_dt is not null) then
  begin
    mins_left = 10 - datediff(minute from :last_dt to current_timestamp);

    if (mins_left > 0) then
    begin
      res_text = f_format('Повторный запрос возможен через %1 мин.', :mins_left);
      suspend;
      exit;
    end
  end

  -- Вставляем в EXTINFOLIST напрямую, как HP_GKB_FRAUD_ROBOT (RFCKZ-877)
  -- и HP_EXTINFO_INSERT_ANTIFRAUD.
  -- HP_EXTINFO_INSERT здесь непригодна: она дедуплицирует по паре
  -- (клиент, конфигурация) в пределах eic_info_actual_days и при
  -- совпадении не вставляет новую запись, а лишь переставляет
  -- eil_object_kind/eil_object_id у старой. Для 'susn' это убило бы
  -- и повторную отправку, и новый запрос по новой заявке в тот же день.
  insert into extinfolist (eil_from_eic_id, eil_from_c_id, eil_date,
                           eil_object_kind, eil_object_id, eil_is_json,
                           eil_from_un_id)
  values (:eic_id, :c_id, current_date,
          :obj_kind_ma, :ma_id, 1,
          :un_id);

  post_event 'susn';

  res_text = 'Запрос отправлен';
  suspend;
end
