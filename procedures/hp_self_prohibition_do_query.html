CREATE PROCEDURE HP_SELF_PROHIBITION_DO_QUERY(
    C_ID D_ID NOT NULL,
    OBJECT_KIND D_STR_CODE NOT NULL,
    OBJECT_ID D_ID NOT NULL,
    UN_ID D_ID NOT NULL)
AS
declare variable EIC_STOP_CR_KIB D_ID;
declare variable EIC_STOP_CR_CRIF D_ID;
declare variable EIL_ISHENIM D_ID;
declare variable EIL_SSOUND D_ID;
declare variable OBJECT_KIND_ID D_INTEGER;
declare variable SPM_ID D_ID;
begin
    -- Процедура инициирует запрос на проверку наличия "самозапрета" в КИБы: Ишеним, Сэйф энд Сааунд

    eic_stop_cr_kib = (select ec.eic_id from extinfocfg ec where ec.eic_code = 'stop_cr_kib'); -- Ишеним
    eic_stop_cr_crif = (select ec.eic_id from extinfocfg ec where ec.eic_code = 'stop_cr_crif'); -- Сэйф энд Саунд
    object_kind_id = f_objkind(:object_kind, 'id');

    -- Инициирует запрос в Ишеним
    insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_is_json, eil_from_eip_id, eil_from_un_id)
    values (:c_id, :eic_stop_cr_kib, current_date, :object_kind_id, :object_id, 1, null, :un_id)
    returning eil_id
    into :eil_ishenim;

    -- Инициирует запрос в Сэйф энд Саунд
    insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_is_json, eil_from_eip_id, eil_from_un_id)
    values (:c_id, :eic_stop_cr_crif, current_date, :object_kind_id, :object_id, 1, null, :un_id)
    returning eil_id
    into :eil_ssound;

    -- Проверяем, инициировался ли запрос на проверку (по данному клиенту, в разрезе данного объекта) ранее
    select spm_id
    from self_prohibition_main
    where spm_object_kind = :object_kind_id and
          spm_object_id = :object_id and
          spm_from_c_id = :c_id
    into :spm_id;

    -- Если нет, создаем базовую запись
    if (spm_id is null) then
    begin
        insert into self_prohibition_main (spm_object_kind, spm_object_id, spm_from_c_id, spm_eil_ishenim,
                                           spm_eil_ssound, spm_total_result, spm_query_qty, spm_last_query_datetime)
        values (:object_kind_id, :object_id, :c_id, :eil_ishenim, :eil_ssound, 0, 1, current_timestamp)
        returning spm_id
        into :spm_id;
    end
    else  -- Если да, делаем апдейт базовой записи
    begin
        update SELF_PROHIBITION_MAIN
        set spm_eil_ishenim = :eil_ishenim,
            spm_eil_ssound = :eil_ssound,
            spm_total_result = 0, -- обнуляем результат
            spm_query_qty = spm_query_qty + 1, -- увеличиваем счетчик попыток
            spm_last_query_datetime = current_timestamp
        where (spm_id = :spm_id);
    end

    -- всегда создаем запись с деталями. По идее в рамках одного SELF_PROHIBITION_MAIN может быть больше 1 записи с деталями толлько если
    insert into self_prohibition_detail (spd_from_spm_id, spd_eil_ishenim, spd_eil_ssound, spd_ishenim_result,
                                         spd_ssound_result, spd_total_result)
    values (:spm_id, :eil_ishenim, :eil_ssound, 0, 0, 0);
end
