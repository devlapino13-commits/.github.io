CREATE PROCEDURE HP_WMA_SMS_SEND(
    WMA_ID INTEGER,
    PHONE_NO STR_100,
    SMS_TEXT STR_255,
    UN_ID ID_TYPE)
AS
begin
  sms_text = trim(coalesce(sms_text, ''));
  sms_text = replace(sms_text, '~', '''');

  insert into cc_website_sms_log (wsl_wma_id, wsl_phone, wsl_sms_text, wsl_un_id)
  values (:wma_id, :phone_no, :sms_text, :un_id);

  execute procedure okt_send_sms(:phone_no, :sms_text);
end
