CREATE PROCEDURE HP_SUD_EXTINFO_INSERT(
    EIC_CODE D_STR_CODE,
    OBJ_KIND_STR D_STR_CODE,
    OBJ_ID D_ID,
    C_ID D_ID,
    FORCE_INSERT D_BOOL = 0)
RETURNS (
    EIL_ID D_ID)
AS
declare variable eic_id d_id;
declare variable eic_has_api d_smallint;
declare variable eic_info_actual_days d_integer;
declare variable obj_kind d_integer;
begin
  select eic_id, eic_has_api, eic_info_actual_days
  from extinfocfg
  where eic_code = :eic_code
  into eic_id, eic_has_api, eic_info_actual_days;

  if (eic_id is null) then
    exception e_default 'ExtinfoCfg not found: ' || coalesce(eic_code, '<NULL>');

  obj_kind = f_objkind(obj_kind_str, 'id');

  if (force_insert = 1 or not exists(select eil_id, eil_date, eil_object_kind
                                    from extinfolist
                                    where eil_from_c_id = :c_id
                                      and eil_from_eic_id = :eic_id
                                      and eil_status in (-1, 0, 1, 2)
                                      and eil_object_kind = :obj_kind
                                      and eil_object_id = :obj_id
                                      and datediff(day, eil_date, current_date) <= :eic_info_actual_days)
  )then
  begin
    insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_is_json)
    values (:c_id, :eic_id, current_date, :obj_kind, :obj_id, :eic_has_api)
    returning eil_id
    into eil_id;

    suspend;
  end
end
