CREATE PROCEDURE HP_UXALIAS_LIST(
    UX_CODE D_STR_CODE)
RETURNS (
    RES_ID D_STR_CODE,
    RES_CAPTION D_STR_255,
    RES_SHORT_CODE D_STR_CODE,
    UXA_UX_CODE D_STR_CODE,
    UXA_CODE D_STR_CODE,
    UXA_NAME D_STR_100,
    RES_ALIAS_KIND D_SMALLINT,
    RES_MENU_COUNT D_INTEGER,
    RES_SEARCH_USED D_INTEGER)
AS
declare variable ux_kind d_str_code;
declare variable ux_name d_str_code;
declare variable uxa_kind d_smallint;
begin
  -- Список алиасов для показа действий
  uxa_ux_code = ux_code;
  res_search_used = 0;

  select count(*)
  from uxaction
  where ua_ux_code = :ux_code
  into res_menu_count;

  res_id = ux_code || '_';
  uxa_code = '';
  res_alias_kind = 0;

  select ux_name, uxf$kind(ux_kind, 'code')
  from uxcode
  where ux_code = :ux_code
  into ux_name, ux_kind;

  uxa_name = ux_name;
  res_short_code = '';

  if (ux_kind <> 'uxcommand') then
  begin
    res_caption = ux_name;
    res_short_code = '';

    if (res_caption not containing ux_code) then
      res_caption = res_caption || ' [' || ux_code || ']';
  
    suspend;
  end

  for select uxa_code, uxa_name, uxa_kind
  from uxalias
  where uxa_ux_code = :ux_code
  into uxa_code, uxa_name, uxa_kind
  do
  begin
    res_alias_kind = 1;

    res_short_code = '';
    if (uxa_code <> ux_name) then
    begin
      res_short_code = substring(uxa_code from char_length(ux_code) + 1);
      if (left(res_short_code, 1) = '.') then
        res_short_code = substring(res_short_code from 2);
    end

    if (exists(select * from uxsearch
    where usr_code = :uxa_code)) then
      res_alias_kind = 2;
    else
      if (uxa_kind = 2) then
        res_alias_kind = 3;

    res_menu_count = 0;

    select count(*)
    from uxaction
    where ua_uxa_code = :uxa_code and
          ua_usr_id_field <> ''
    into res_search_used;
  
    res_caption = uxa_name;
    if (res_short_code <> '') then
      res_caption = res_caption || ' (' || res_short_code || ')';

    if (ux_kind <> 'uxcommand') then
    begin
      select count(*)
      from uxaliasaction
      where uaa_uxa_code = :uxa_code
      into res_menu_count;
    end

    suspend;
  end
end
