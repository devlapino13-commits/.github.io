CREATE PROCEDURE HP_CORRECT_TERMPAY(
    K_FROM_NO STR_20,
    K_TO_NO STR_20,
    PAY_NAME_FROM STR_100,
    PAY_NAME_TO STR_100,
    PAY_DATE DATE,
    PAY_AMOUNT CURR_TYPE,
    PAY_AMOUNT_FROM CURR_TYPE,
    PAY_AMOUNT_TO CURR_TYPE)
RETURNS (
    ERRMSG STR_255)
AS
declare variable un_id id_type;
declare variable ol_parent_id id_type;
declare variable c_from_id id_type;
declare variable c_to_id id_type;
declare variable ol_id id_type;
declare variable esm_id id_type;
declare variable new_esm_id id_type;
declare variable esm_source d_blob_text;
declare variable k_from_id id_type;
declare variable k_to_id id_type;
declare variable esm_full_source blob sub_type 0 segment size 80;
declare variable esm_full_source_new blob sub_type 0 segment size 80;
declare variable esm_amount curr_type;
declare variable amount_first_kontract curr_type;
declare variable esm_project str_100;
declare variable esm_trans_id str_20;
declare variable esm_account str_20;
begin
  select un.un_id
  from username un
  where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN')
  into un_id;

  if (un_id is null) then
    un_id = 1000000000001;

  select k_id
  from kontract k
  where k_kontract_no = :k_from_no
  order by k_id desc
  rows 1
  into :k_from_id;

  select c_id
  from clients
  where c_name_fio = :pay_name_from
  into c_from_id;

  select c_id
  from clients
  where c_name_fio = :pay_name_to
  into c_to_id;

  if (k_to_no = 'Перепринятие') then
    k_to_id = k_from_id;
  else
    select k_id
    from kontract k
    where k_kontract_no = :k_to_no
    order by k_id desc
    rows 1
    into :k_to_id;

  if (k_to_id is null) then
    k_to_id = k_from_id;

  if (exists(select *
             from operkred
             where ok_from_k_id in (:k_from_id, :k_to_id) and
                   ok_ol_is_active = 1 and
                   ok_ol_date > :pay_date)) then
  begin
    errmsg = 'Есть позднее погашение в контракте';
    suspend;
    exit;
  end

  if (k_from_id is not null and
      k_to_id is not null and
      pay_amount_to > 0) then
  begin
    select eo.eso_from_ol_id
    from extsysmain em
    inner join extsysoper eo on em.esm_id = eo.eso_from_esm_id
    where em.esm_from_k_id = :k_from_id and
          em.esm_date = :pay_date and
          em.esm_amount = :pay_amount and
          em.esm_status = 1 rows 1








    into ol_parent_id;

    if (coalesce(ol_parent_id, 0) = 0) then
    begin
      errmsg = 'Не найден терм.платёж по ' || :k_from_no ||
        ' за ' || f_datetostr(:pay_date) || ' в сумме ' || :pay_amount;
      suspend;
      exit;
    end

    if (pay_amount <> pay_amount_from + pay_amount_to) then
    begin
      errmsg = 'Ошибка: ' || pay_amount || ' <> ' || pay_amount_from || ' + ' || pay_amount_to;
      suspend;
      exit;
    end

    select esm.eso_from_esm_id
    from extsysoper esm
    where esm.eso_from_ol_id = :ol_parent_id rows 1
    into esm_id;

    select em.esm_full_source, em.esm_project, esm_trans_id
    from extsysmain em
    where em.esm_id = :esm_id
    into esm_full_source, esm_project, esm_trans_id;

    amount_first_kontract = pay_amount - pay_amount_to;

    if (pay_amount_from <> amount_first_kontract) then
    begin
      errmsg = 'Общая сумма для разделения не совпадает с суммой оплаты!';
      suspend;
      exit;
    end

    esm_full_source_new = replace(esm_full_source, '&cl_date', '_fix2&cl_date');
    --    ESM_TRANS_ID = ESM_TRANS_ID || '_fix2';

    if (coalesce(esm_id, 0) = 0) then
    begin
      errmsg = 'Не найден ESM_ID';
      suspend;
      exit;
    end
    else
    begin

      update operlist
      set is_delete = 1,
          ol_user_id = :un_id
      where ol_id = :ol_parent_id;

      if (pay_amount_to <> 0) then
      begin
        update extsysmain em
        set em.esm_amount = :amount_first_kontract,
            em.esm_from_k_id = :k_from_id,
            em.esm_status = 0
        where em.esm_id = :esm_id;

        insert into extsysmain (esm_id, esm_trans_id, esm_project, esm_action, esm_date, esm_account, esm_object_kind,
                                esm_object_id, esm_datetime, esm_result_id, esm_status, esm_full_source,
                                esm_ext_datetime, esm_amount, esm_from_k_id)
        select -1, esm_trans_id, esm_project, esm_action, esm_date, esm_account, esm_object_kind, esm_object_id,
               esm_datetime, esm_result_id, esm_status, :esm_full_source_new, esm_ext_datetime, :pay_amount_to,
               :k_to_id
        from extsysmain
        where esm_id = :esm_id
        returning esm_id
        into new_esm_id;

        if (new_esm_id is null) then
        begin
          errmsg = 'Ошибка при разделении!';
          suspend;
          exit;
        end
        else
          errmsg = 'Сохранено успешно, проверьте погашения!';
        suspend;
      end
      else
      begin
        errmsg = 'Сумма не указана!';
        suspend;
      end
    end
  end
  else
  begin
    if (coalesce(pay_amount_to, 0) > 0) then
    begin
      errmsg = 'Ошибка: ' || coalesce(pay_amount_to, 0) || ' > 0';
      suspend;
      exit;
    end

    select eo.eso_from_ol_id
    from extsysmain em
    inner join extsysoper eo on em.esm_id = eo.eso_from_esm_id
    where em.esm_from_k_id = :k_from_id and
          em.esm_date = :pay_date and
          em.esm_amount = :pay_amount and
          em.esm_status = 1 rows 1








    into ol_parent_id;

    if (coalesce(ol_parent_id, 0) = 0) then
    begin
      errmsg = 'Не найдено аннулированное погашение';
      suspend;
      exit;
    end

    select esm.eso_from_esm_id
    from extsysoper esm
    where esm.eso_from_ol_id = :ol_parent_id rows 1
    into esm_id;

    if (coalesce(esm_id, 0) = 0) then
    begin
      errmsg = 'Не найден ESM_ID';
      suspend;
      exit;
    end
    else
    begin
      update operlist
      set is_delete = 1,
          ol_user_id = :un_id
      where ol_id = :ol_parent_id;

      update extsysmain em
      set em.esm_from_k_id = :k_to_id,
          em.esm_status = 0
      where em.esm_id = :esm_id;

      errmsg = 'Сохранено успешно, проверьте погашения!';
      suspend;
    end
  end
end
