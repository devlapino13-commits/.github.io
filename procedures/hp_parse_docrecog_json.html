CREATE PROCEDURE HP_PARSE_DOCRECOG_JSON(
    DOCRECOG_JSON D_BLOB_TEXT,
    FIELDS_ARRAY_PATH D_STR_255 = 'result.fields')
RETURNS (
    FIELD_CODE D_STR_CODE,
    FIELD_VALUE D_STR_255,
    FIELD_ACCEPTED D_SMALLINT)
AS
declare variable ARRAY_SIZE D_INTEGER;
declare variable ITEM_NO D_INTEGER;
declare variable TMP_JSON D_STR_8100;
begin
    array_size = f_json_blob_get_count(docrecog_json, :fields_array_path);
    item_no = 0;

    while (item_no < array_size) do
    begin
        tmp_json = f_json_blob_get_value(docrecog_json, :fields_array_path || '[' || item_no || ']', null);
        field_code = f_json_str_get_value(tmp_json, 'code', '');
        field_value = left(trim(f_json_str_get_value(tmp_json, 'value', '')), 255);
        field_accepted = f_json_str_get_value(tmp_json, 'is_accepted', '');

        item_no = item_no + 1;
        suspend;
    end

end
