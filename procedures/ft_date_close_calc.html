CREATE PROCEDURE FT_DATE_CLOSE_CALC(
    K_ID D_ID,
    C_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    LAST_DATE D_DATE)
AS
declare variable last_date_tmp d_date;
declare variable pereplata d_currency;
declare variable acc_id d_id;
begin
  -- Процедура возвращает дату закрытия договора
  -- Если дата закрытия больше rep_date, возвращаем null
  -- Учитываются переплаты и судебные задолженности, внесистемка
  --   Если они не погашены на дату rep_date возвращаем null
  --   Если они погашены до rep_date, возвращаем дату закрытия последней задолженности/переплаты
  -- Учитывается подписанный договор АСП
  --  Если на дату закрытия всех задолженностей АСП действителен, возвращаем его дату закрытия
  --  Если он не аннулирован, считаем что договор продолжает действовать
  -- Большинство деталей описаны в RFCRU-2833

  last_date = null; -- Признак, что есть задолженность или действующий АСП на rep_date

  -- RFCRU-4884
  if (exists(select 1 from ft_info_stoplist
            where current_date between fisl_freeze_date_begin and coalesce(fisl_freeze_date_end, '31.12.9999')
              and fisl_from_c_id = :c_id))
  then
  begin
    suspend;
    exit;
  end 

  select oks_date_end
  from oksummary oks
  where oks_from_k_id = :k_id
    and oks_close_state = 0 -- RFCRU-4701 списанные и прочие долги не учитываем
  order by oks_from_k_id, oks_date_end desc
  rows 1
  into last_date_tmp;

  -- Увеличиваем 1 день только если кредит закрыт ранее
  -- Т.к. в oksummary date_end меньше даты закрытия на 1 день
  if (last_date_tmp is not null and last_date_tmp <> '31.12.9999')
    then last_date_tmp = last_date_tmp + 1;

  -- Если кредит закрыт ранее, проверяем переплату
  if (last_date_tmp < rep_date) then
  begin
    last_date = last_date_tmp;

    last_date_tmp = null;
    pereplata = null;
    select sum(ok_os),
           max(ok_ol_date)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date and
          ok_type = 9
    into pereplata, last_date_tmp;

    -- Если есть переплата в любую сторону, считаем что договор не закрыт
    if (coalesce(pereplata, 0) <> 0) then
      last_date = null;
    else if (last_date_tmp > last_date) then
      last_date = last_date_tmp;

    -- Переплату могли перенести на счет заморозки
    -- Если там есть остаток, считаем что договор не закрыт RFCRU-2833 (коммент)
    if (last_date is not null and last_date_tmp is not null) then
    begin
      -- Получаем счет заморозки клиента
      select acc_id
      from ref_account, ref_accbalance
      where ab_code = '60322'
      and acc_from_ab_id = ab_id
      and acc_add_code = '60322 (83)'
      and acc_object_kind = 0
      and acc_object_id = :c_id
      into acc_id;
      -- Проверяем баланс
      if (acc_id is not null and f_acc_bal(acc_id, rep_date, 'sum') <> 0) then
        last_date = null;
      -- Счета открывают на клиента, а не на договор. Учитывая то, что
      -- разморозки счетов еще не было, считаем что наличие остатка, это вечная
      -- активность договора. Если будут размораживать, придется вытаскивать
      -- проводки со счета договора на счет заморозки клиента и брать баланс
      -- и дату последей операции.
    end
  end

  -- Если на дату закрытия есть АСП, берем его дату (если есть)
  if (last_date is not null) then
  begin
    if (f_edoc_status('asp.agreement', c_id, last_date) = 'signed') then
    begin
      select edl_date_end
      from edoc_list
      where edl_object_kind = 0 and
            edl_object_id = :c_id and
            edl_from_edc_id = f_edoc_kind('asp.agreement','id') and
            edl_status = 1 and
            :last_date between edl_blank_date and coalesce(edl_date_end, :last_date)
      order by edl_id desc
      rows 1
      into last_date_tmp;
  
      if (last_date_tmp is null) then
        last_date = null;
      else if (last_date_tmp > last_date) then
        last_date = last_date_tmp;
    end
  end

  if (last_date is not null) then
    last_date = last_date + 1;

  -- RFCRU-4701
  select min(cds_create_datetime)
  from clientdocs cds
  where cds_from_c_id = :c_id
    and cds_is_delete = 0
    and cds_from_pk_id = 91 -- свидетельство о смерти
  into last_date_tmp;

  if (last_date_tmp < coalesce(last_date, current_date)) then
    last_date = last_date_tmp;

  suspend;
end
