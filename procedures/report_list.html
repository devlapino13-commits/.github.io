CREATE PROCEDURE REPORT_LIST(
    ID INTEGER,
    SELF_INCLUDE INTEGER,
    KIND INTEGER,
    IS_DOWN INTEGER)
RETURNS (
    ID_LIST INTEGER,
    REP_NAME_RUS VARCHAR(255),
    REP_NAME_ENG VARCHAR(255),
    ORDER_NO INTEGER)
AS
begin
  for with recursive mc_list
      as (select rl_id, parent, rl_order_no, rl_type, rl_name_rus, rl_name_eng
          from reportlist
          where rl_id = :id
          union all
          select r.rl_id, r.parent, r.rl_order_no, r.rl_type, r.rl_name_rus, r.rl_name_eng
          from reportlist r
          join mc_list on ((:is_down = 1 and
                mc_list.rl_id = r.parent) or (:is_down = 0 and
                mc_list.parent = r.rl_id)))
      select rl_id, rl_order_no, rl_name_rus, rl_name_eng
      from mc_list
      where ((:self_include = 0 and
            rl_id <> :id) or
            (:self_include = 1)) and
            ((:kind = -1) or
            (:kind <> -1 and
            rl_type = :kind))
      into :id_list, :order_no, :rep_name_rus, :rep_name_eng
  do
    suspend;
end
