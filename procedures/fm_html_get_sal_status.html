CREATE PROCEDURE FM_HTML_GET_SAL_STATUS(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable esm_status d_smallint;
declare variable c_name_fio d_str_255;
declare variable req_info_text d_str_255;
declare variable res_info_text d_str_255;
declare variable req_permission d_str_long;
declare variable lang d_str_lang;
declare variable send_code d_str_long;
declare variable get_salary_info d_str_long;
declare variable eil_status d_smallint;
declare variable eil_status_text d_str_255;
declare variable code_status d_smallint;
declare variable res_temp_html d_str_long;
declare variable esm_datetime d_datetime;
declare variable send_code_count d_integer;
declare variable req_perm_count d_integer;
begin
  esm_status = -2;
  lang = f_get_lang_code(:un_id, null);
  res_html = f_format('<h2>%1</h2>', 'Справка о заработной плате');
  select c.c_name_fio
  from clients c
  where c.c_id = :c_id
  into c_name_fio;
  req_info_text = 'Запрос не отправлен';
  res_info_text = '';
  req_permission = f_uri_sql(f_lang(lang, 'Запросить'), 'req_perm_sal', 'Clients', :c_id, 'none', f_lang(lang, 'Срок согласия от клиента на получение его перс данных 6 месяцев?'), '', 0, 0, '', '', 'dialog', '');
  send_code = f_uri_sql('Отправить код', 'send_code_sal', 'clients', c_id, 'text', 'Укажите код', '', 4, 6, '', '', 'dialog', '');
  get_salary_info = f_uri_sql(f_lang(lang, 'Запросить'), 'get_salary_info', 'Clients', :c_id, 'none', f_lang(lang, 'Хотите отправить запрос для получения справки?'), '', 0, 0, '', '', 'dialog', '');

  if (exists(select *
             from extsysmain esm
             where esm.esm_project = 'salary_info' and
                   esm.esm_action = 'req_permission' and
                   esm.esm_object_kind = f_objkind('clients', 'id') and
                   esm.esm_object_id = :c_id and
                   current_date between esm.esm_date and dateadd(month, 6, esm.esm_date))) then
  begin
    eil_status = -1;
    eil_status_text = '';

    select sum(iif(esm.esm_action = 'send_code', 1, 0)), sum(iif(esm.esm_action = 'req_permission', 1, 0))
    from extsysmain esm
    where esm.esm_project = 'salary_info' and
          esm.esm_object_kind = f_objkind('clients', 'id') and
          esm.esm_object_id = :c_id and
          esm.esm_date = current_date
    into send_code_count, req_perm_count;

    send_code_count = coalesce(send_code_count, 0);
    req_perm_count = coalesce(req_perm_count, 0);

    select (case eil.eil_status
              when 2 then 'Справка получена'
              when -2 then 'Ошибка при получении справки'
              else 'Запрос в обработке'
            end), eil.eil_status
    from extinfolist eil
    where eil.eil_from_eic_id = 10 and
          eil.eil_is_json = 1 and
          eil.eil_object_kind = f_objkind('clients', 'id') and
          eil.eil_object_id = :c_id and
          eil.eil_from_c_id = :c_id
    order by eil.eil_create_datetime desc
    rows 1
    into eil_status_text, eil_status;
    select (case esm.esm_status
              when 1 then 'Успешно отправлен'
              when -1 then 'Не обработан'
              else 'В обработке'
            end), esm.esm_status, esm.esm_datetime
    from extsysmain esm
    where esm.esm_project = 'salary_info' and
          esm.esm_action = 'req_permission' and
          esm.esm_object_kind = f_objkind('clients', 'id') and
          esm.esm_object_id = :c_id
    order by esm.esm_datetime desc
    rows 1
    into req_info_text, esm_status, esm_datetime;

    select (case esm.esm_status
              when 1 then 'Разрешен'
              when -1 then 'Запрос не обработан'
              else 'Код отправлен'
            end), esm.esm_status
    from extsysmain esm
    where esm.esm_project = 'salary_info' and
          esm.esm_action = 'send_code' and
          esm.esm_object_kind = f_objkind('clients', 'id') and
          esm.esm_object_id = :c_id and
          esm.esm_datetime > :esm_datetime
    order by esm.esm_datetime desc
    rows 1
    into res_info_text, code_status;

    res_info_text = coalesce(res_info_text, '');

    esm_status = coalesce(esm_status, -2);
    res_temp_html = (case :esm_status
                       when 1 then f_html_table_row(:c_name_fio, 'Запрос на получения справки', :req_info_text || '<br>' || iif(coalesce(:code_status, -2) = -2, :send_code, ''), '', -1, -1, -1, -1, 3, '')
                       when 0 then f_html_table_row(:c_name_fio, 'Запрос на получения справки', :req_info_text, '', -1, -1, -1, -1, 3, '')
                       else f_html_table_row(:c_name_fio, 'Запрос на получения справки', :req_info_text || '<br>' || :req_permission, '', -1, -1, -1, -1, 3, '')
                     end);
    if (esm_status = 1 and
        code_status = -1 and
        send_code_count < 5) then
      res_temp_html = f_html_table_row(:c_name_fio, 'Запрос на получения справки', 'Код веден неправильно!' || '<br>' || :send_code, '', -1, -1, -1, -1, 3, '');
    else
    if (send_code_count >= 5) then
      res_temp_html = f_html_table_row(:c_name_fio, 'Запрос на получения справки', 'Ошибка при получении справки' || '<br>' || :req_permission, '', -1, -1, -1, -1, 3, '');
    if (esm_status = 1 and
        code_status = 1) then
    begin
      res_temp_html = f_html_table_row(:c_name_fio, 'Запрос на получения справки', :req_info_text || '<br>' || :res_info_text, '', -1, -1, -1, -1, 3, '');
      res_temp_html = blob_append(res_temp_html, f_html_table_row('', 'Справка заработной платы', iif(:eil_status < 0, :eil_status_text || '<br>' || :get_salary_info, :eil_status_text), '', -1, -1, -1, -1, 3, ''));
    end
    if (req_perm_count >= 1 and
        send_code_count >= 5 and
        coalesce(code_status, -2) < 0) then
      res_temp_html = f_html_table_row(:c_name_fio, 'Запрос на получения справки', 'Превышен лимит отправки запроса на Соц фонд', '', -1, -1, -1, -1, 3, '');
  end
  else
    res_temp_html = f_html_table_row(:c_name_fio, 'Запрос на получения справки', :req_info_text || '<br>' || :req_permission, '', -1, -1, -1, -1, 3, '');
  res_html = blob_append(res_html, '<table width="100%">
       <col width="40%">
       <col width="30%">
       <col width="30%">
       <tr>');
  res_html = blob_append(res_html, res_temp_html);
  suspend;
end
