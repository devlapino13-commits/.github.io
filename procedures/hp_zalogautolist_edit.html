CREATE PROCEDURE HP_ZALOGAUTOLIST_EDIT(
    ZAL_ID D_INTEGER,
    ZAL_FROM_ZAP_ID D_INTEGER,
    ZAL_COST D_CURRENCY,
    ZAL_WEBPAGE D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_ID D_INTEGER,
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
begin
  res_is_error = 0;
  res_text = '';
  res_id = 0;

  if (zal_webpage is null) then
    zal_webpage = '';

  zal_webpage = trim(zal_webpage);

  if (coalesce(zal_id, 0) = 0) then
  begin
    if (not exists(select * from zalogautolist
    where zal_from_zap_id = :zal_from_zap_id and
          zal_webpage = :zal_webpage)) then
    begin
      insert into zalogautolist (zal_from_zap_id, zal_cost, zal_webpage, zal_status, zal_from_un_id)
      values (:zal_from_zap_id, :zal_cost, :zal_webpage, 1, :un_id)
      returning zal_id
      into :res_id;
    end else
    begin
      res_text = 'Объявление с этой ссылкой уже добавлено';
      res_is_error = 1;
    end
  end else
  begin
    res_id = zal_id;
  
    if (zal_webpage = '' and zal_cost = 0) then
    begin
      update zalogautolist
      set zal_is_delete = 1,
          zal_from_un_id = :un_id
      where zal_id = :zal_id;
    end else
    begin
      update zalogautolist
      set zal_from_zap_id = :zal_from_zap_id,
          zal_cost = :zal_cost,
          zal_webpage = :zal_webpage,
          zal_status = iif(coalesce(zal_webpage, '') <> coalesce(:zal_webpage, '') or zal_status = -2, 1, zal_status),
          zal_add_params = iif(coalesce(zal_webpage, '') <> coalesce(:zal_webpage, '') or zal_status = -2, '', zal_add_params),
          zal_from_un_id = :un_id
      where (zal_id = :zal_id);
    end
  end

  suspend;
end
