CREATE PROCEDURE HP_UXFRITEM_UNGROUP(
    UFR_CODE D_STR_CODE,
    UFI_CODE D_STR_CODE)
AS
declare variable new_parent_code d_str_100;
declare variable new_position d_integer;
declare variable tmp_ufi_code d_str_100;
begin
  -- Процедура удаляет доп.элементы из UxFrItem (group, label  и т.д.)
  if (ufi_code = '') then
    exit;

  select ufi_parent_code, ufi_position
  from uxfritem
  where ufi_code = :ufi_code
  into new_parent_code, new_position;

  for select ufi_code
  from uxfritem
  where ufi_parent_code = :ufi_code and
        ufi_ux_code = :ufr_code
  order by ufi_position desc
  into tmp_ufi_code
  do
    execute procedure hp_uxfritem_move(:tmp_ufi_code, :new_parent_code, :new_position);

  delete from uxfritem
  where ufi_ux_code = :ufr_code and
        ufi_code = :ufi_code and
        ufi_kind = -1 and
        ufi_parent_code is not null;
end
