CREATE PROCEDURE HP_REPLACEMENTPLEDGER_SET(
    Z_OPER D_INTEGER,
    Z_ID_OLD D_INTEGER,
    Z_FROM_K_ID ID_TYPE,
    Z_FROM_C_ID ID_TYPE,
    Z_TYPE D_INTEGER,
    Z_SUBTYPE D_INTEGER,
    Z_NAME D_STR_100,
    Z_NAME_FIO D_STR_255,
    Z_FROM_ADL_ID ID_TYPE,
    Z_PHONE D_STR_100,
    Z_DOCUMENT D_STR_255,
    Z_MEMO1 VARCHAR(1024),
    Z_MEMO2 D_STR_2048,
    Z_REGISTRY_NO VARCHAR(50),
    Z_REGISTRY_DATE D_DATE,
    Z_REGISTRY_AGENCY VARCHAR(80),
    Z_COST_MARKET CURR_TYPE,
    Z_COST_LIKVID CURR_TYPE,
    Z_COST_KONTRACT CURR_TYPE)
RETURNS (
    Z_ID_NEW D_INTEGER)
AS
declare dt_set d_date;
begin
  dt_set = iif(z_oper=0, current_date, null);
  if (z_oper = 0) then
    update zalog set z_date_end = :dt_set - 1 where z_id = :z_id_old;
  insert into zalog (
    z_from_k_id, z_from_c_id, z_type, z_subtype, z_name, z_name_fio, z_from_adl_id,
    z_phone, z_document, z_memo1, z_memo2, z_registry_no, z_registry_date, z_registry_agency, 
    z_cost_market, z_cost_likvid, z_cost_kontract, z_date_begin)
  values (
    :z_from_k_id, :z_from_c_id, :z_type, :z_subtype, :z_name, :z_name_fio, :z_from_adl_id,
    :z_phone, :z_document, :z_memo1, :z_memo2, :z_registry_no, :z_registry_date, :z_registry_agency, 
    coalesce(:z_cost_market,0), coalesce(:z_cost_likvid,0), coalesce(:z_cost_kontract,0), :dt_set)
  returning z_id into :z_id_new;
  suspend;
end
