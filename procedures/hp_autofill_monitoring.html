CREATE PROCEDURE HP_AUTOFILL_MONITORING
AS
declare variable obj_kind d_integer;
declare variable rd_id d_integer;
declare variable acl_id d_id;
declare variable kc_id d_id;
declare variable opr_id_loan_use d_id;
declare variable opr_id_memo d_id;
declare variable ovs_id_loan_use d_id;
declare variable ovs_id_memo d_id;
declare variable k_disb_date d_date;
declare variable obj_kind_clients d_integer;
declare variable opr_id d_id;
declare variable ovs_id d_id;
declare variable c_id d_id;
begin
  obj_kind = f_objkind('knclient', 'id');
  rd_id = f_get_rd_id('Anketa.TargetUseLoan');
  obj_kind_clients = f_objkind('clients', 'id');
  for select acl.acl_id, kc.kc_id, kt.k_disb_date, kc.kc_from_c_id
      from actionlist acl
      inner join knclient kc on kc.kc_id = acl.acl_object_id
      inner join kontract kt on kt.k_id = kc.kc_from_k_id
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      where acl.acl_object_kind = :obj_kind and
            acl.acl_from_rd_id = :rd_id and
            acl.acl_status = 0 and
            pr.pr_code containing '.july.' and
            kc.kc_amount_approved > 0
      order by acl.acl_id desc
      into acl_id, kc_id, k_disb_date, c_id
  do
  begin
    insert into objectprops (opr_object_kind, opr_object_id, opr_subject_kind, opr_subject_id)
    values (8, :kc_id, -1, 1624)
    returning opr_id
    into opr_id_loan_use;

    insert into objectvalues (ovs_from_opr_id, ovs_date_begin, ovs_value, ovs_add_value, ovs_from_un_id)
    values (:opr_id_loan_use, :k_disb_date, '', 0, 1000000000000)
    returning (ovs_id)
    into :ovs_id_loan_use;

    insert into objectvalres (ovr_from_opr_id, ovr_from_ovs_id, ovr_result_rd_id, ovr_state, ovr_value)
    values (:opr_id_loan_use, :ovs_id_loan_use, 63, 1, '');

    insert into objectprops (opr_object_kind, opr_object_id, opr_subject_kind, opr_subject_id)
    values (8, :kc_id, -1, 4083)
    returning opr_id
    into opr_id_memo;

    insert into objectvalues (ovs_from_opr_id, ovs_date_begin, ovs_value, ovs_add_value, ovs_from_un_id)
    values (:opr_id_memo, :k_disb_date, '', 0, 1000000000000)
    returning (ovs_id)
    into :ovs_id_memo;

    insert into objectvalres (ovr_from_opr_id, ovr_from_ovs_id, ovr_result_rd_id, ovr_state, ovr_value)
    values (:opr_id_memo, :ovs_id_memo, 0, 1,
            'Удовлетворение потребительских нужд');

    select opr.opr_id
    from objectprops opr
    where opr.opr_object_kind = :obj_kind_clients and --clients
          opr.opr_object_id = :c_id and
          opr.opr_subject_kind = -1 and
          opr.opr_subject_id = 4317
    order by opr.opr_id desc
    rows 1
    into opr_id;

    insert into objectvalues (ovs_from_opr_id, ovs_date_begin, ovs_value, ovs_add_value, ovs_from_un_id)
    values (:opr_id, :k_disb_date, '', 0, 1000000000000)
    returning (ovs_id)
    into :ovs_id;

    insert into objectvalres (ovr_from_opr_id, ovr_from_ovs_id, ovr_result_rd_id, ovr_state, ovr_value)
    values (:opr_id, :ovs_id, 1435, 1,
            'Удовлетворение потребительских нужд');

    update actionlist acl
    set acl.acl_status = 2
    where acl.acl_id = :acl_id;
  end
end
