CREATE PROCEDURE AGREEMENT_QR_SAVE_RESULT(
    APQ_ID D_INTEGER,
    URL D_STR_LONG,
    URL_JSON_DATA D_STR_8100,
    ERROR_TEXT D_STR_255)
AS
begin
    error_text = coalesce(error_text, '');
    url = coalesce(url, '');
    url_json_data = coalesce(url_json_data, '');
    if ((error_text = '') and (url = '') and (url_json_data in ('', '{}'))) then   -- временное решение, пока Владимир сервис доработает
      error_text = 'Ошибка при считывании QR код';
    if ((error_text = '') and (url = '')) then
         exception e_message('url не передан');

    if ((error_text = '') and (url_json_data in ('', '{}'))) then
         exception e_message('url_json_data не передан');

    if (apq_id is null) then
        exception e_message('apq_id is null');

    update agreement_payment_qr apq
    set apq.apq_status = iif(:error_text = '', f_enum_id('agreement_payment_qr_status', 'assigned'),
                                               f_enum_id('agreement_payment_qr_status', 'error')),
        apq.apq_url = :url,
        apq.apq_url_data = :url_json_data,
        apq.apq_error_text = :error_text
    where apq_id = :apq_id;
end
