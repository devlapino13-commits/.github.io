CREATE PROCEDURE HP_USER_CASH_NUM(
    OFFICE_ID D_INTEGER,
    ACC_TYPE D_INTEGER,
    UN_ID D_ID)
RETURNS (
    NUM_CASH D_INTEGER)
AS
declare variable bg_id d_str_20;
declare variable db_id d_integer;
begin
  -- возвращает номера касс по офису (RFCKZ-396)
  -- используется в форме "Документы -> Соответствие кассы пользователю", поле "Счет кассы"

  db_id = f_get_db_id();

  if (db_id <> 39801) then
    exit;

  if (acc_type = 1) then
    bg_id = '1021'; -- денежные средства в пути
  else
    exit;

  for select numb_cash
      from hp_numb_cash(:office_id, :bg_id, 0, :un_id)
      into num_cash
  do
    suspend;
end
