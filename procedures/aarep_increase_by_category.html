CREATE PROCEDURE AAREP_INCREASE_BY_CATEGORY(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(60),
    AA_CURR_BRANCH_NAME VARCHAR(60),
    AA_CURR_LEVEL_NAME VARCHAR(60),
    AA_CURR_KC_FIO VARCHAR(100),
    KOL_CL_BEG_CAT1 INTEGER,
    PAR_BEG_CAT1 CURR_TYPE,
    KOL_CL_BEG_CAT2 INTEGER,
    PAR_BEG_CAT1_2 CURR_TYPE,
    KOL_CL_END_CAT1 INTEGER,
    PAR_END_CAT1 CURR_TYPE,
    KOL_CL_END_CAT2 INTEGER,
    PAR_END_CAT1_2 CURR_TYPE)
AS
begin
  for with tabl1
      as (select aa_loan_kind_new, aa_category_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name,
                 aa_curr_kc_fio, aa_kolich_cl_active_end as aa_kolich_cl_active_begin,
                 aa_amount_active_end as aa_amount_active_begin,
                 iif(aa_deliq_days between 8 and 180, aa_amount_active_end, 0) as delinq_balance_begin,
                 0 as aa_kolich_cl_active_end, 0 as aa_amount_active_end, 0 as delinq_balance_end
          from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin - 1, :rep_date_begin - 1, '', 0, 0, 0, 0, 0, 0, 0)
          union
          select aa_loan_kind_new, aa_category_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name,
                 aa_curr_kc_fio, 0 as aa_kolich_cl_active_begin, 0 as aa_amount_active_begin, 0 as delinq_balance_begin,
                 aa_kolich_cl_active_end, aa_amount_active_end,
                 iif(aa_deliq_days between 8 and 180, aa_amount_active_end, 0) as delinq_balance_end
          from aarep_analitica_port_temp(:rep_region_id, :rep_date_end, :rep_date_end, '', 0, 0, 0, 0, 0, 0, 0)),
      tabl2
      as (select aa_loan_kind_new, aa_category_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name,
                 aa_curr_kc_fio, sum(aa_kolich_cl_active_begin) as kolich_cl_begin,
                 sum(aa_amount_active_begin) as balance_begin, sum(delinq_balance_begin) as delinq_balance_begin,
                 sum(aa_kolich_cl_active_end) as kolich_cl_end, sum(aa_amount_active_end) as balance_end,
                 sum(delinq_balance_end) as delinq_balance_end
          from tabl1
          group by aa_loan_kind_new, aa_category_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_kc_fio)

      select aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_kc_fio,
             sum(iif(aa_category_id = 1, kolich_cl_begin, 0)) as kol_cl_beg_cat1, -- кол-во по категории 1 на начало
             iif(sum(iif(aa_category_id = 1, balance_begin, 0)) <> 0, sum(iif(aa_category_id = 1, delinq_balance_begin, 0)) / sum(iif(aa_category_id = 1, balance_begin, 0)) * 100, 0) as par_beg_cat1,
             sum(iif(aa_category_id = 2, kolich_cl_begin, 0)) as kol_cl_beg_cat2, -- кол-во по категории 2 на начало
             iif(sum(iif(aa_category_id in (1, 2), balance_begin, 0)) <> 0, sum(iif(aa_category_id in (1, 2), delinq_balance_begin, 0)) / sum(iif(aa_category_id in (1, 2), balance_begin, 0)) * 100, 0) as par_beg_cat1_2,

             sum(iif(aa_category_id = 1, kolich_cl_end, 0)) as kol_cl_end_cat1, -- кол-во по категории 1 на начало
             iif(sum(iif(aa_category_id = 1, balance_end, 0)) <> 0, sum(iif(aa_category_id = 1, delinq_balance_end, 0)) / sum(iif(aa_category_id = 1, balance_end, 0)) * 100, 0) as par_end_cat1,
             sum(iif(aa_category_id = 2, kolich_cl_end, 0)) as kol_cl_end_cat2, -- кол-во по категории 2 на начало
             iif(sum(iif(aa_category_id in (1, 2), balance_end, 0)) <> 0, sum(iif(aa_category_id in (1, 2), delinq_balance_end, 0)) / sum(iif(aa_category_id in (1, 2), balance_end, 0)) * 100, 0) as par_end_cat1_2
      from tabl2
      group by aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_kc_fio
      into aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_kc_fio, kol_cl_beg_cat1, par_beg_cat1,
           kol_cl_beg_cat2, par_beg_cat1_2, kol_cl_end_cat1, par_end_cat1, kol_cl_end_cat2, par_end_cat1_2
  do
  begin
    suspend;
  end

end
