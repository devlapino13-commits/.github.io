CREATE PROCEDURE HP_DO_KN_OPERBOOK_OL_MEMORDER(
    EXIST_OL_ID D_ID,
    KC_ID D_ID,
    ACC_CODE_DT D_STR_CODE,
    ACC_CODE_KT D_STR_CODE,
    AMOUNT_SUM D_CURRENCY,
    AMOUNT_VAL D_CURRENCY,
    AMOUNT_TAX_DT D_CURRENCY,
    AMOUNT_TAX_KT D_CURRENCY,
    OB_DETAIL D_STR_255,
    ACC_DT D_ID,
    ACC_KT D_ID,
    OL_DATE D_DATE = null,
    OL_OPER_TYPE D_INTEGER = null,
    OL_OPER_SUBTYPE D_INTEGER = null,
    OL_DOC_TYPE D_INTEGER = null,
    OL_REG_ID D_INTEGER = null,
    UN_ID D_ID = null)
RETURNS (
    OL_ID D_ID)
AS
declare variable obd_id d_id;
begin
  if (exist_ol_id is null) then
  begin
    if (ol_date is null) then
      exception e_default 'OL_Date is empty';

    if (ol_oper_type is null or ol_oper_subtype is null or ol_reg_id is null) then
      exception e_default 'Some params is empty';

    insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_doc_no, ol_doc_no_str,
                          ol_memo, ol_parent_id, is_delete, ol_user_id)
    values (:ol_reg_id, :ol_oper_type, :ol_oper_subtype, :ol_date, :ol_doc_type, 0, '', :ob_detail, null, 0, :un_id)
    returning ol_id
    into :ol_id;
  end
  else
  begin
    ol_id = exist_ol_id;

    ol_date = null;
    select ol_date
    from operlist
    where ol_id = :ol_id
    into :ol_date;

    if (ol_date is null) then
      exception e_default 'OL_ID not found';
  end

  if (ol_date < '31.12.2017') then
    exception e_default 'Can not make record early 31.12.2017';

  acc_code_dt = replace(acc_code_dt, '.', '');
  acc_code_dt = replace(acc_code_dt, ' ', '');

  acc_code_kt = replace(acc_code_kt, '.', '');
  acc_code_kt = replace(acc_code_kt, ' ', '');

  if (acc_dt is null) then
  begin
    select acc_id
    from ref_account
    where acc_code = :acc_code_dt
    into acc_dt;
  end

  if (acc_kt is null) then
  begin
    select acc_id
    from ref_account
    where acc_code = :acc_code_kt
    into acc_kt;
  end

  if (acc_dt is null) then
    exception e_default 'Dt account not found: ' || coalesce(acc_code_dt, '<null>');

  if (acc_kt is null) then
    exception e_default 'Kt account not found: ' || coalesce(acc_code_kt, '<null>');

  select obd_id
  from kn_obdetail
  where obd_detail = :ob_detail
  into obd_id;

  if (obd_id is null) then
  begin
    insert into kn_obdetail (obd_detail)
    values (:ob_detail)
    returning obd_id
    into obd_id;
  end

  if (amount_tax_dt is null) then
    amount_tax_dt = amount_sum;
  if (amount_tax_kt is null) then
    amount_tax_kt = amount_sum;

  insert into kn_operbook (kob_from_ol_id, kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                           kob_summa_tax_dt, kob_summa_tax_kt, kob_detail_id, kob_from_kc_id)
  values (:ol_id, '810', 1, :acc_dt, :acc_kt, :amount_sum, :amount_val, :amount_tax_dt, :amount_tax_kt, :obd_id, :kc_id);

  suspend;
end
