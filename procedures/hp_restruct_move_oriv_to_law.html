CREATE PROCEDURE HP_RESTRUCT_MOVE_ORIV_TO_LAW(
    REP_DATE D_DATE,
    PROCESS_MODE D_SMALLINT)
AS
declare variable rbc_id d_id;
declare variable k_id d_id;
declare variable res_k_id d_id;
declare variable k_kontract_no d_str_20;
begin
    /*
      RFCKZ-705. Процедура проводит автоматическую передачу контрактов с баланса ОРиВ
      на баланс юр. отдела на 30-й с момента отмены исполнительной надписи, если по контракту
      не была произведена реструктуризация или он не был погашен.

      Примечание: при отмене испол. надписи через специальный модуль происходит передача контракта
      с баланса юр. отдела на баланс ОРиВ, в данной же процедуре происходит обратный процесс

      process_mode:
        1 - обработка всех записей на rep_date (стандартная обработка)
        2 - обработка только ошибочных записей
    */

    if (process_mode not in (1, 2)) then
        exception e_message 'Некорректный process_mode';

    for
        select rbc.rbc_id, rbc.rbc_from_k_id
        from restruct_block_cancel rbc
        where rbc.rbc_date_end = :rep_date and
              rbc.rbc_moved_law_to_oriv = 1 and -- перемещенные от юр. отдела в ОРиВ
              rbc.rbc_moved_oriv_to_law = 0 and  -- но не перемещенные обратно
              (
                (:process_mode = 1 and coalesce(rbc.rbc_move_oriv_to_law_error, '') = '') or
                (:process_mode = 2 and coalesce(rbc.rbc_move_oriv_to_law_error, '') <> '')
              )
        into :rbc_id, :k_id
    do
    begin
        begin
            -- Если по контракту была проведена реструктуризация, не переводим на юр. отдел
            if (exists(select *
                       from aarep_get_restructuring(:k_id, :rep_date, f_get_db_id())
                       where is_restruct = 1)
            ) then
                continue;
    
            -- Если контракт полностью закрыт, не переводим на юр. отдел
            if (exists(select *
                       from kontstate ks
                       where :rep_date between ks.ks_date_begin and ks.ks_date_end and
                             ks.ks_from_k_id = :k_id and
                             ks.ks_state = 2 and
                             ks.ks_ol_is_delete = 0)
            ) then
                continue;
    
            k_kontract_no = (select k.k_kontract_no from kontract k where k.k_id = :k_id);
        
            -- Передаем кредит из ОРиВ в юр. отдел
            select k_id from insert_sb_akt(:k_kontract_no, :rep_date, 2, 2, 1000000000000)
            into :res_k_id;
    
            -- Перемещение прошло успешно, обновляем запись
            update restruct_block_cancel rbc
            set rbc.rbc_moved_oriv_to_law = 1,
                rbc.rbc_move_oriv_to_law_datetime = current_timestamp,
                rbc.rbc_move_oriv_to_law_error = null
            where rbc.rbc_id = :rbc_id;
    
            rbc_id = null;
            k_id = null;
    
            when any do
            begin
                update restruct_block_cancel rbc
                set rbc.rbc_move_oriv_to_law_error = left(rdb$error(message), 255)
                where rbc.rbc_id = :rbc_id;

                execute procedure hp_send_quick_message('email', 't.mavlyanov@mbulak.kg',
                    'Ошибка при переводе кредита ОРиВ -> Юр. отдел',
                    f_format('Возникла ошибка в контракте: %1, rbc_id: %2, текст ошибки: %3', :k_id, :rbc_id, left(rdb$error(message), 2048)),
                    '', 7, 1000000000000
                );

                rbc_id = null;
                k_id = null;
            end
        end
    end

end
