CREATE PROCEDURE MC_DO_REJECTION(
    MCL_ID D_ID,
    MA_ID D_ID,
    REJECTION_CODE D_STR_CODE)
RETURNS (
    RES_CODE D_STR_255,
    IS_OK D_BOOL,
    ERROR_TEXT D_STR_255,
    EDS_ID D_ID)
AS
declare variable k_id d_id;
declare variable edl_id d_id;
declare variable freeze_datetime_end d_datetime;
declare variable db_id d_integer;
declare variable un_id d_id;
begin
  -- RFCKG-4445
  -- Если не добавить это поле, то КМП считает error_text фатальной ошибкой и ломается
  res_code = 'success';
  is_ok = 0;
  error_text = '';
  db_id = f_get_db_id();
  if (rejection_code <> 'complete') then
    exception e_default 'Unkown rejection_code = ' || coalesce(rejection_code, '<NULL>');
  if (not exists(select 1
            from miniappclient, mc_login mcl
            where mcl_from_c_id = mac_from_c_id
              and mcl_id = :mcl_id
              and mac_from_ma_id = :ma_id))
  then
    exception e_default f_format('Not found ma_id = %1 for mcl_id = %2', ma_id, mcl_id);
  select k_id
  from kontract
  where k_from_ma_app_id = :ma_id
    and k_status >= 0
  order by k_status desc, k_id desc
  rows 1
  into k_id;
  if (k_id is null) then
    error_text = 'Договор не найден';
  if (error_text = '') then
  begin
    select freeze_datetime_end
    from hp_get_freeze_data(:k_id)
    into :freeze_datetime_end;
  
    if (freeze_datetime_end is null) then
      error_text = 'Договор не подписан';
  end
  -- RFCRU-5900 для РФ (db_id = 81001) блок "Период охлаждения окончен" отключён
  if (error_text = '' and db_id <> 81001) then
    if (current_timestamp > freeze_datetime_end) then
      error_text = 'Период охлаждения окончен';
  if (error_text = '') then
  begin
    is_ok = 1;
    if (db_id = 81001) then
    begin
      select eds_id
      from edoc_sign, edoc_list
      where eds_from_edl_id = edl_id
        and edl_from_edc_id = f_edoc_kind('loan.cancel', 'id')
        and edl_object_kind = 9
        and edl_object_id = :k_id
        and edl_status = 1
      order by 1 desc
      rows 1
      into eds_id;
  
      if (eds_id is null) then
      begin
        select edl_id, res_text
        from hp_edoc_insert('loan.cancel', f_objkind('kontract', 'id'), :k_id, 1000000000000)
        into edl_id, error_text;
  
        if (error_text = '') then
          select eds_id
          from edoc_sign
          where eds_from_edl_id = :edl_id
          into eds_id;
        else
          is_ok = 0;
      end 
    end
    else if (db_id = 39801) then
    begin
      un_id = 1000000000000;
      eds_id = 0;
      delete from freeze_main
      where fzm_from_k_id = :k_id;
      insert into kontreslist (krl_id, krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                               krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
      values (-1, 1, :un_id, :k_id, 0, null, current_timestamp, current_date, current_date, 0);
      
      rdb$set_context('USER_TRANSACTION', 'FREEZE_APPROVE_CANCEL_K_ID', :k_id);
      
      update kontract
      set k_status = -1,
          k_user_id = :un_id
      where k_id = :k_id;
    end
  end
  suspend;
end
