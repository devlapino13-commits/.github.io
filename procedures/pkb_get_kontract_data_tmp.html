CREATE PROCEDURE PKB_GET_KONTRACT_DATA_TMP(
    POCKET_ID INTEGER)
RETURNS (
    K_ID ID_TYPE,
    K_KONTRACT_NO D_STR_CODE,
    ORDER_MARK D_SMALLINT,
    PHASEOFCONTRACT D_INTEGER,
    DAYS_PAST_DUE_DOMAIN D_INTEGER,
    DAYS_PAST_DUE_DOMAIN_PREV D_INTEGER,
    LOCAL_STATUS D_STR_CODE,
    ACTIVE_DELINQ_DAYS D_INTEGER)
AS
declare variable k_id_old id_type;
declare variable rep_date_begin d_date;
declare variable rep_date_end d_date;
declare variable pocket_type d_str_10;
declare variable do_suspend d_bool;
declare variable disb_date d_date;
declare variable date_type d_str_20;
declare variable contract_status_tmp d_integer;
declare variable is_fixed d_smallint;
declare variable gkb d_integer;
declare variable pkb d_integer;
declare variable cib_code d_str_20;
begin
  k_id_old = 0;
  gkb = null;
  pkb = null;
  select cp.cp_date_begin, --
         cp.cp_date_end, --
         cp.cp_pocket_type,
         f_get_rd_code(cp.cp_cib_id)
  from cib_pocket cp
  where cp.cp_id = :pocket_id
  into :rep_date_begin,
       :rep_date_end,
       :pocket_type,
       cib_code;

  if (cib_code = 'CIB.PKB') then
   pkb = :pocket_id;
  else
   gkb = :pocket_id;

  if ((rep_date_end is null) or (pocket_type is null)) then
    exit;

  for select ks_from_k_id,
             order_mark,
             phaseofcontract,
             is_fixed
      from (
            select ks_from_k_id, order_mark, phaseofcontract, is_fixed
            from (select ks.ks_from_k_id,
                  0 as order_mark,
                  iif(ks.ks_state = 1, 4, 5) as phaseofcontract, 0 as is_fixed
                  from cib_to_send_kontracts cts
                  inner join kontstate ks  on ks.ks_from_k_id = cts.cts_from_k_id
                  where :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
                        ks.ks_ol_is_delete = 0 and
                        iif(:cib_code = 'CIB.PKB', cts.cts_from_cp_pkb_id = :pkb, cts.cts_from_cp_gkb_id = :gkb)
            ) Kont
            union
            select ks_from_k_id, order_mark, phaseofcontract, is_fixed
            from (select ks.ks_from_k_id,
                  0 as order_mark,
                  iif(ks.ks_state = 1, 4, 5) as phaseofcontract, 1 as is_fixed
                  from CIB_KONTRACT_ERRORS cke
                  inner join kontstate ks  on ks.ks_from_k_id = cke.cke_from_k_id
                  where :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
                        ks.ks_ol_is_delete = 0 and
                        cke.cke_checked_datetime  >= :rep_date_end - 1 and
                        cke.cke_checked_datetime < :rep_date_end
            ) CheckedKont
      ) total
      order by ks_from_k_id,
               order_mark
      into :k_id,
           :order_mark,
           :phaseofcontract,
           :is_fixed
  do
  begin
    if (k_id <> k_id_old) then
    begin
      k_id_old = k_id;



      -- если крежит в игнор листе, его не отправляем
      if (exists(select * from kib_ignore where ki_from_k_id = :k_id)) then
        continue;

      select kont.k_kontract_no,
             kont.k_disb_date
      from kontract kont
      where kont.k_id = :k_id
      into :k_kontract_no,
           :disb_date;
    
      if (phaseofcontract = 5) then  -- если крежит был закрыт отпраляем его в любом случае
      begin
        days_past_due_domain = 1;
        local_status = 'closed';
      end
      else -- если это активный заем
      begin
           days_past_due_domain = 10;
        
           -- если у кредита изменился статус просрочки по отновшению к предыдущему дню, отправляем его
           for select date_type,
                      contract_status ,
                      active_delinq_days
           from pkb_get_loan_categ(:k_id, :rep_date_end)
           into :date_type,
                :contract_status_tmp,
                :active_delinq_days
           do
           begin
               if (date_type = 'rep_date') then
                   days_past_due_domain = contract_status_tmp;
               else
                   days_past_due_domain_prev = contract_status_tmp;
           end
           if (days_past_due_domain <> days_past_due_domain_prev) then
           begin
               local_status = 'changed_delinq_status';
           end
           -- < ---------
           -- если это выдача займа
           if (disb_date between rep_date_begin and rep_date_end) then
           begin
             if (not exists(select * from cib_to_send_kontracts cts
                        where cts.cts_from_k_id = :k_id and
                              iif(:cib_code = 'CIB.PKB', cts.cts_from_cp_pkb_id <> :pkb, cts.cts_from_cp_gkb_id <> :gkb))) then   -- если выдача была сегодня, но контракт снова попал в список, то отправляем уже не как новый контракт
             begin
               local_status = 'disb';
               days_past_due_domain = 27;
             end
             else
             begin
               local_status = 'payed';
               days_past_due_domain = 10;
             end

           end
           -- если на период пакета по кредиту была оплата, добавляем кредит в пакет

               if (exists(select *
                       from operkred ok
                       where ok.ok_type in (2, 4, 9) and
                             ok.ok_ol_is_active = 1 and
                             ok.ok_from_k_id = :k_id and
                             ok.ok_ol_date between :rep_date_begin and :rep_date_end)) then
                   local_status = 'payed';
           -- < --
      end
           suspend;
    end
  end
end
