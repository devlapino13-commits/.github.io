CREATE PROCEDURE AAREP_REPORT_FOR_TSB_2(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    C_NAME_FIO VARCHAR(100),
    K_DISB_DATE DATE,
    RES_CLIENT_IS_RESIDENT INTEGER,
    IS_SECURED INTEGER,
    OL_DATE DATE,
    DELINQ_DAY INTEGER,
    CLI_WOFF_OS CURR_TYPE,
    CLI_WOFF_PERC CURR_TYPE,
    CLI_WOFF_FINE CURR_TYPE)
AS
declare variable k_id id_type;
declare variable sh_date date;
declare variable woff_os curr_type;
declare variable woff_perc curr_type;
declare variable woff_fine curr_type;
declare variable k_amount_approved curr_type;
declare variable kc_amount_approved curr_type;
declare variable dolya double precision;
declare variable c_id id_type;
declare variable citizen_ru_id integer;
declare variable doc_vnj_id integer;
declare variable pr_code varchar(100);
declare variable pr_loan_type integer;
declare variable k_kolich_client_approved integer;
declare variable k_id_prior id_type;
declare variable pr_product_kind integer;
declare variable pr_id integer;
begin
  citizen_ru_id = null;
  select rd_id
  from refdata
  where rd_code = 'Citizen.RU'
  into :citizen_ru_id;

  doc_vnj_id = null;
  select pk_id
  from passkind, passdoctype
  where pdt_id = pk_from_pdt_id and
        pk_is_delete = 0 and
        pdt_code = 'registration_vnj' rows 1
  into :doc_vnj_id;

  k_id_prior = 0;
  for select ok_from_k_id, ok_ol_date, min(ok_sh_date), sum(ok_os), sum(ok_perc), sum(ok_fine)
      from operkred
      where ok_ol_is_active = 1 and
            ok_type in (5, 6) and
            ok_ol_date between :rep_date_begin and :rep_date_end
      group by ok_from_k_id, ok_ol_date
      into k_id, ol_date, sh_date, woff_os, woff_perc, woff_fine
  do
  begin

    if (k_id_prior <> k_id) then
    begin
      select k_kontract_no, k_amount_approved, k_disb_date, pr_code, pr_loan_type, pr_product_kind, pr_id,
             k_kolich_client_approved
      from kontract, product
      where k_id = :k_id and
            k_from_pr_id = pr_id
      into k_kontract_no, k_amount_approved, k_disb_date, pr_code, pr_loan_type, pr_product_kind, pr_id,
           k_kolich_client_approved;

      is_secured = null;
      select is_secured
      from aarep_get_secured_loan(:pr_code, :pr_loan_type, :pr_product_kind, :pr_id, :k_disb_date, :k_kolich_client_approved, :k_id)
      into is_secured;
      k_id_prior = k_id;
    end

    delinq_day = ol_date - sh_date;
    if (delinq_day < 0) then
      delinq_day = 0;

    for select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id, kc_amount_approved
    do
    begin
      c_name_fio = 'n' || :c_id;
      dolya = cast(kc_amount_approved as double precision) / k_amount_approved;

      select 1 - res_non_resident
      from hp_get_client_type(:c_id, :citizen_ru_id, :doc_vnj_id, :rep_date_end)
      into :res_client_is_resident;

      cli_woff_os = dolya * woff_os;
      cli_woff_perc = dolya * woff_perc;
      cli_woff_fine = dolya * woff_fine;

      suspend;
    end
  end
end
