CREATE PROCEDURE UXP$USER_REGIONS(
    UN_ID D_ID)
RETURNS (
    ID D_INTEGER,
    TEXT D_STR_100,
    IS_OFFICE D_BOOL)
AS
declare variable reg_status d_integer;
declare variable reg_date_close d_date;
declare variable closed_date d_date;
begin
  closed_date = cast('01.01.' || (extract(year from current_date) - 1) as date);

  if (un_id is null) then
    exit;

  for select distinct reg_list, iif(reg_code <> reg_name, reg_code || '  ' || reg_name, reg_code),
             reg_status, reg_date_close
  from aarep_region_user_tmp(:un_id), region
  where reg_unicode = reg_list
  order by reg_code, reg_unicode
  into :id, :text, reg_status, reg_date_close
  do
  begin
    is_office = iif(reg_status = 3, 1, 0);

    if (coalesce(reg_date_close, current_date) < closed_date) then
      Continue;

    if (is_office = 0) then
    begin
      if (not exists(select * from region
                     where parent = :id and
                     reg_status = 3 and
                     coalesce(reg_date_close, current_date) >= :closed_date)) then
      begin
        if (not exists(select * from region r1, region r2
        where r1.parent = :id and r1.reg_unicode = r2.parent and
              r2.reg_status = 3 and
              coalesce(r2.reg_date_close, current_date) >= :closed_date)) then
          Continue;
      end
    end

    if (reg_date_close is not null) then
      text = text || ', до ' || f_datetostr(reg_date_close);

    suspend;
  end
end
