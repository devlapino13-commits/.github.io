CREATE PROCEDURE AAREP_FIO_TRANSLITERATION_CHECK(
    REP_DATE D_DATE NOT NULL)
RETURNS (
    C_ID ID_TYPE,
    C_NAME_FIO D_STR_100,
    C_NAME_ENG D_STR_100,
    CITIZEN D_STR_100,
    KONT_STATUS D_STR_100)
AS
declare variable has_active d_smallint;
declare variable has_offbalance d_smallint;
begin
  for select distinct c.c_id,
                      c.c_name_fio,
                      c.c_name_eng,
                      rd.rd_name
      from knclient kc
      join clients c on c.c_id = kc.kc_from_c_id
      left join refdata rd on rd.rd_id = c.c_citizen --выводит гражданство письменно
      where c.c_name_eng similar to '%[а-яА-ЯёЁ]%'
      into c_id,
           c_name_fio,
           c_name_eng,
           citizen
  do
  begin
    has_active = 0;
    has_offbalance = 0;

    if (exists(select 1
               from knclient kc
               join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
               where kc.kc_from_c_id = :c_id and
                     ks.ks_ol_is_delete = 0 and
                     :rep_date between ks.ks_date_begin and ks.ks_date_end and
                     ks.ks_state = 1)) then
      has_active = 1;

    if (exists(select 1
               from knclient kc
               join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
               inner join operkroff okf on okf.oko_from_k_id = ks.ks_from_k_id
               where kc.kc_from_c_id = :c_id and
                     ks.ks_ol_is_delete = 0 and
                     ks.ks_state = 2 and
                     :rep_date between ks.ks_date_begin and ks.ks_date_end and
                     okf.oko_ol_is_active = 1 and
                     okf.oko_ol_date <= :rep_date
               group by ks.ks_from_k_id
               having sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc) > 0)) then
      has_offbalance = 1;

    if (has_active = 1 and
        has_offbalance = 1) then
      kont_status = 'Активный+Внебаланс';
    else if (has_active = 1) then
      kont_status = 'Активный';
    else if (has_offbalance = 1) then
      kont_status = 'Внебаланс';
    else
      kont_status = 'Не активный';

    suspend;
  end
end
