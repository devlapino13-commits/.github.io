CREATE PROCEDURE AAREP_ACCOUNT_POSITION(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    POS_CAPITAL_EKV DECIMAL(18,4),
    POS_CURR_CODE VARCHAR(3),
    POS_CURR_SYMBOL VARCHAR(10),
    POS_CURR_NAME VARCHAR(50),
    POS_CURR_RATE DECIMAL(18,4),
    POS_CAPITAL_VAL DECIMAL(18,4),
    POS_ASSETS DECIMAL(18,4),
    POS_LIAB DECIMAL(18,4),
    POS_DIFF DECIMAL(18,4),
    POS_POSITION DECIMAL(18,4),
    POS_KIND VARCHAR(100),
    POS_LIAB_EKV DECIMAL(18,4))
AS
declare variable summa decimal(18,4);
declare variable kind integer;
declare variable curr_id varchar(3);
declare variable curr_id_prior varchar(3);
declare variable def_currency varchar(3);
begin
  select cm_default_currency
  from company
  into :def_currency;

  curr_id_prior = '   ';
  select sum(summa_ekv)
  from aarep_account_saldo_temp(:report_region_id, :report_date_end), acc_list, balgroup
  where account = al_id and
        al_from_bg_id = bg_id and
        bg_kind >= 3
  into :pos_capital_ekv;
  if (pos_capital_ekv is null) then
    pos_capital_ekv = 0;
  pos_capital_ekv = pos_capital_ekv * (-1);

  if (pos_capital_ekv <> 0) then
  begin
    for select currency, bg_kind, sum(summa_val)
        from aarep_account_saldo_temp(:report_region_id, :report_date_end), acc_list, balgroup
        where account = al_id and
              al_from_bg_id = bg_id and
              bg_kind <= 2 and
              bg_is_recost = 1
        group by currency, bg_kind
        order by currency, bg_kind
        into :curr_id, :kind, :summa
    do
    begin
      summa = cast(summa as decimal(15,2));
      if ((curr_id <> curr_id_prior) and
          (curr_id_prior <> '   ')) then
      begin
        select ap_from_cl_id, cu_symbol, cu_name, cr_rate
        from accposit, currlist, opercurr
        where ap_from_cl_id = cu_id and
              ap_from_cl_id = cr_from_cu_id and
              cr_date <= :report_date_end and
              cr_date_end >= :report_date_end and
              cr_from_cu_id = :curr_id_prior
        order by ap_from_cl_id
        into :pos_curr_code, :pos_curr_symbol, :pos_curr_name, :pos_curr_rate;

        pos_capital_val = cast(:pos_capital_ekv / :pos_curr_rate as decimal(18,2));
        pos_diff = :pos_assets - :pos_liab;
        pos_position = cast((:pos_diff / :pos_capital_val) * 100 as decimal(15,2));
        if (pos_curr_code <> def_currency) then
          pos_liab_ekv = cast(pos_liab * :pos_curr_rate as decimal(15,2));
        else
          pos_liab_ekv = 0;

        pos_kind = '';
        if ((pos_assets <> 0) or (pos_liab <> 0)) then
        begin
          if (pos_assets = pos_liab) then
            pos_kind = 'Закрытая позиция';
          else
          if (pos_assets > pos_liab) then
            pos_kind = 'Длинная позиция';
          else
          if (pos_assets < pos_liab) then
            pos_kind = 'Короткая позиция';
          suspend;
        end

        pos_assets = 0;
        pos_liab = 0;
        pos_diff = 0;
        pos_position = 0;
        if (kind = 1) then
        begin
          pos_assets = :summa;
          summa = 0;
        end
        if (kind = 2) then
        begin
          pos_liab = :summa * (-1);
          summa = 0;
        end
      end
      else
      begin
        if (kind = 1) then
        begin
          pos_assets = :summa;
          summa = 0;
        end
        if (kind = 2) then
        begin
          pos_liab = :summa * (-1);
          summa = 0;
        end
      end
      curr_id_prior = curr_id;
    end
    select ap_from_cl_id, cu_symbol, cu_name, cr_rate
    from accposit, currlist, opercurr
    where ap_from_cl_id = cu_id and
          ap_from_cl_id = cr_from_cu_id and
          cr_date <= :report_date_end and
          cr_date_end >= :report_date_end and
          cr_from_cu_id = :curr_id
    order by ap_from_cl_id
    into :pos_curr_code, :pos_curr_symbol, :pos_curr_name, :pos_curr_rate;

    pos_capital_val = cast(:pos_capital_ekv / :pos_curr_rate as decimal(18,2));
    pos_diff = :pos_assets - :pos_liab;
    pos_position = cast((:pos_diff / :pos_capital_val) * 100 as decimal(15,2));
    if (pos_curr_code <> def_currency) then
      pos_liab_ekv = cast(pos_liab * :pos_curr_rate as decimal(15,2));
    else
      pos_liab_ekv = 0;

    pos_kind = '';
    if ((pos_assets <> 0) or (pos_liab <> 0)) then
    begin
      if (pos_assets = pos_liab) then
        pos_kind = 'Закрытая позиция';
      else
      if (pos_assets > pos_liab) then
        pos_kind = 'Длинная позиция';
      else
      if (pos_assets < pos_liab) then
        pos_kind = 'Короткая позиция';
      suspend;
    end

  end
end
