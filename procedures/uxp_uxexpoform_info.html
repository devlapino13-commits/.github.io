CREATE PROCEDURE UXP$UXEXPOFORM_INFO(
    UFM_UUID D_UUID,
    UA_UUID D_UUID,
    USE_RELEASE D_BOOL)
RETURNS (
    EXF_UFM_UUID D_UUID,
    EXF_CAPTION D_STR_100,
    EXF_UX_KIND D_SMALLINT,
    EXF_UX_UUID D_UUID,
    EXF_UX_SCODE D_STR_CODE,
    EXF_UX_FCODE D_STR_100,
    EXF_UX_NAME D_STR_100,
    EXF_UX_RELEASE_UUID D_UUID,
    EXF_UX_DRAFT_UUID D_UUID,
    EXF_US_CODE D_STR_CODE,
    EXF_US_NAME D_STR_100,
    EXF_US_IS_TABLE D_BOOL,
    EXF_US_KIND D_SMALLINT,
    EXF_US_MEMO D_STR_8100,
    EXF_US_KEYFIELDS D_STR_100,
    EXF_US_AUTOINCFIELD D_STR_CODE,
    EXF_US_LOOKUP_ID D_STR_CODE,
    EXF_US_LOOKUP_TEXT D_STR_255,
    EXF_US_MAIN_TABLE D_STR_CODE,
    EXF_US_SQL D_STR_4096,
    EXF_UA_UXA_UUID D_UUID,
    EXF_UA_UXA_CODE D_STR_CODE,
    EXF_UA_UXA_FCODE D_STR_100,
    EXF_UA_UXA_NAME D_STR_100,
    EXF_UA_UXA_KIND D_SMALLINT,
    EXF_UA_IN_UI D_BOOL,
    EXF_CHILD_ACTION_COUNT D_INTEGER,
    EXF_USE_RELEASE D_BOOL,
    EXF_DB_ID D_INTEGER,
    EXF_DB_IS_ACTUAL D_BOOL,
    EXF_DB_PATH D_STR_LONG)
AS
declare variable ua_owner d_uuid;
declare variable tmp_count d_integer;
begin
  if (ufm_uuid is null and ua_uuid is null) then
    exception e_default 'Empty input params';

  exf_ua_uxa_uuid = null;
  exf_ua_uxa_code = '';
  exf_ua_uxa_fcode = '';
  exf_ua_uxa_name = '';
  exf_ua_uxa_kind = 0;
  exf_ua_in_ui = 0;
  exf_caption = '';

  exf_db_is_actual = 0;
  if (uxf$db_actual()) then
    exf_db_is_actual = 1;

  exf_db_path = rdb$get_context('SYSTEM', 'DB_NAME');

  exf_db_id = uxf$db_id();

  exf_use_release = use_release;

  if (ua_uuid is not null) then
  begin
    select ua_owner, ua_name, uu_kind,
           ux_uuid, ux_scode, ux_fcode, ux_name, ux_release_uuid, ux_draft_uuid,
           us_code, us_name, us_is_table, us_kind, us_memo, us_keyfields,
           us_autoincfield, us_lookup_id, us_lookup_text, us_main_table, us_sql,
           uxa_uuid, uxa_code, uxa_fcode, uxa_name, uxa_kind
    from ux$action, ux$alias, ux$code, ux$source, ux$uniq
    where ua_uuid = :ua_uuid and
          uxa_uuid = ua_uxa_uuid and
          uxa_owner = ux_uuid and
          ux_us_uuid = us_uuid and
          uu_id = ux_uuid
    into ua_owner, exf_caption, exf_ux_kind,
         exf_ux_uuid, exf_ux_scode, exf_ux_fcode, exf_ux_name, exf_ux_release_uuid, exf_ux_draft_uuid,
         exf_us_code, exf_us_name, exf_us_is_table, exf_us_kind, exf_us_memo, exf_us_keyfields,
         exf_us_autoincfield, exf_us_lookup_id, exf_us_lookup_text, exf_us_main_table, exf_us_sql,
         exf_ua_uxa_uuid, exf_ua_uxa_code, exf_ua_uxa_fcode, exf_ua_uxa_name, exf_ua_uxa_kind;

    if (row_count = 0) then
    begin
      select ua_owner, ua_name
      from ux$action
      where ua_uuid = :ua_uuid and
            ua_uxa_uuid is null
      into ua_owner, exf_caption;
  
      if (row_count = 0) then
        exit;
    end

    if (exf_use_release = 1 and exf_ux_release_uuid is null) then
      exf_use_release = 0;

    if (exf_use_release = 1) then
      exf_ufm_uuid = exf_ux_release_uuid;
    else
      exf_ufm_uuid = exf_ux_draft_uuid;

    if (exists(select * from ux$frame
               where ufr_uuid = :ua_owner)) then
      exf_ua_in_ui = 1;

    if (exf_ua_in_ui = 0) then
    begin
      if (exists(select * from ux$frbutton
                 where ufb_ua_uuid = :ua_uuid)) then
        exf_ua_in_ui = 1;
    end
  end else
  begin
    exf_ufm_uuid = ufm_uuid;

    select uu_kind, ux_uuid, ux_scode, ux_fcode, ux_name, ux_release_uuid, ux_draft_uuid,
           us_code, us_name, us_is_table, us_kind, us_memo, us_keyfields, 
           us_autoincfield, us_lookup_id, us_lookup_text, us_main_table, us_sql
    from ux$form, ux$code, ux$source, ux$uniq
    where ufm_uuid = :exf_ufm_uuid and
          ufm_ux_uuid = ux_uuid and
          ux_us_uuid = us_uuid and
          uu_id = ux_uuid
    into exf_ux_kind, exf_ux_uuid, exf_ux_scode, exf_ux_fcode, exf_ux_name, exf_ux_release_uuid, exf_ux_draft_uuid,
         exf_us_code, exf_us_name, exf_us_is_table, exf_us_kind, exf_us_memo, exf_us_keyfields, 
         exf_us_autoincfield, exf_us_lookup_id, exf_us_lookup_text, exf_us_main_table, exf_us_sql;

    if (row_count = 0) then
      exit;

    exf_caption = exf_ux_name;
  end

  exf_child_action_count = 0;
  select count(*)
  from ux$action
  where ua_owner = :exf_ufm_uuid
  into tmp_count;

  exf_child_action_count = exf_child_action_count + tmp_count;

  select count(*)
  from ux$frame, ux$action
  where ufr_owner = :exf_ufm_uuid and
        ua_owner = ufr_uuid
  into tmp_count;

  exf_child_action_count = exf_child_action_count + tmp_count;

  suspend;
end
