Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_KN_OKSUMMARY_GET_PRV(
    KC_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    PRV_DATE D_DATE,
    PRV_VBS D_CURRENCY,
    PRV_PERCENT_NORM D_PERCENT,
    PRV_BAL_TO_PERC D_CURRENCY,
    PRV_BAL_ESP D_CURRENCY,
    PRV_BAL_PERC D_CURRENCY,
    PRV_BAL_KOMMIS D_CURRENCY,
    PRV_DELINQ_DATE D_DATE,
    PRV_KOMMIS_AMOUNT D_CURRENCY,
    PRV_BALANCE D_CURRENCY,
    PRV_BAL_ZAD_OS D_CURRENCY,
    PRV_BAL_ZAD_PERC D_CURRENCY,
    PRV_LAST_SH_DATE D_DATE,
    PRV_BAL_LAW D_CURRENCY,
    PRV_IS_KONT_DEVIDED D_SMALLINT,
    PRV_MAX_PROLONG D_INTEGER)
AS
declare variable k_id d_id;
begin
  -- Возвращает значения из предыдущей строки

  select oks_date_begin, oks_vbs_end, oks_percent_norm, oks_bal_to_perc, oks_bal_esp_end - oks_off_esp_bal,
         oks_bal_perc, oks_bal_kommis, oks_delinq_date, oks_kommis_amount, oks_balance, oks_bal_zad_os,
         oks_bal_zad_perc, oks_last_sh_date, oks_bal_law, oks_is_kont_devided, oks_max_prolong
  from kn_oksummary
  where oks_from_kc_id = :kc_id and
        oks_date_begin < :rep_date
  order by oks_from_kc_id desc, oks_date_begin desc
  rows 1
  into prv_date, prv_vbs, prv_percent_norm, prv_bal_to_perc, prv_bal_esp,
       prv_bal_perc, prv_bal_kommis, prv_delinq_date, prv_kommis_amount,
       prv_balance, prv_bal_zad_os, prv_bal_zad_perc, prv_last_sh_date,
       prv_bal_law, prv_is_kont_devided, prv_max_prolong;

  if (row_count = 0) then
  begin
    select kc_from_k_id
    from knclient
    where kc_id = :kc_id
    into k_id;

    select kd_percent_per_day * 365.0000000
    from kontdata
    where kd_from_k_id = :k_id and
          kd_ol_date <= :rep_date and
          kd_ol_is_delete = 0 and
          kd_ol_date <= kd_date_end
    order by kd_from_k_id desc, kd_ol_date desc
    rows 1
    into prv_percent_norm;
  end

  if (prv_date is null) then prv_date = rep_date;
  if (prv_percent_norm is null) then prv_percent_norm = 0;
  if (prv_bal_to_perc is null) then prv_bal_to_perc = 0;
  if (prv_bal_esp is null) then prv_bal_esp = 0;
  if (prv_bal_perc is null) then prv_bal_perc = 0;
  if (prv_kommis_amount is null) then prv_kommis_amount = 0;
  if (prv_balance is null) then prv_balance = 0;
  if (prv_bal_zad_os is null) then prv_bal_zad_os = 0;
  if (prv_bal_zad_perc is null) then prv_bal_zad_perc = 0;
  if (prv_bal_zad_perc is null) then prv_bal_zad_perc = 0;
  if (prv_bal_law is null) then prv_bal_law = 0;
  if (prv_max_prolong is null) then prv_max_prolong = 0;

  suspend;
end
