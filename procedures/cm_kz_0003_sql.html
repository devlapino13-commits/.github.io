CREATE PROCEDURE CM_KZ_0003_SQL(
    MA_ID D_INTEGER NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable OBJ_KIND D_INTEGER;
declare variable STOP_CREDIT_STATUS D_SMALLINT;
declare variable STOP_CREDIT_ANSWER_DATE D_DATE;
declare variable STOP_CREDIT_QUERY_DATETIME D_DATETIME;
declare variable SERVICE_MAX_DELAY_SECONDS D_INTEGER;
declare variable C_ID D_ID;
begin
    -- RFCKG-4392, RFCKZ-23

    if (f_get_const('stop_credit_is_active') = 0) then
        exit;

    res_error_text = '';
    res_error_is_critical = 2;
    obj_kind = f_objkind('miniapp', 'id');
    service_max_delay_seconds = cast(f_get_const('stop_credit_service_max_delay_seconds') as d_integer);

    select status, answer_date, query_datetime
    from hp_stop_credit_query_status(:obj_kind, :ma_id)
    into :stop_credit_status, :stop_credit_answer_date, :stop_credit_query_datetime;

    if (stop_credit_status = 1 and f_is_stopcred_actual(:stop_credit_answer_date) = 1) then -- Есть активный стоп-кредит
        res_error_text = 'У клиента действует стоп-кредит, выдача невозможна';

    if (stop_credit_status in (0, 1) and f_is_stopcred_actual(:stop_credit_answer_date) = 0) then -- Есть ответ по стоп-кредиту, но он уже неактуален
        res_error_text = 'Полученный статус клиента по стоп-кредиту уже не актуален, запросите новый';

    if (stop_credit_status = -1) then
        res_error_text = 'Запрос на сервис стоп-кредита еще не осуществлялся';

    if (stop_credit_status = -2) then
    begin
        if (datediff(second, stop_credit_query_datetime, current_timestamp) >= service_max_delay_seconds) then
        begin
            if (f_is_stopcred_doc_loaded(:obj_kind, :ma_id) = 0) then
                res_error_text = 'Сервис стоп-кредит не отвечает, приложите документ или дождитесь ответа';
        end
        else
            res_error_text = 'Ответ от сервиса Стоп-кредит ещё не получен';
    end

    if (stop_credit_status = -3) then
    begin
        -- RFCKZ-23
        if (f_is_stopcred_doc_loaded(:obj_kind, :ma_id) = 0) then
            res_error_text = 'Сервис стоп-кредит ответил с ошибкой, приложите документ';
    end

    if (coalesce(res_error_text, '') <> '') then
        suspend;


    -- Проверка банковской карты
    select mac.mac_from_c_id
    from miniappclient mac
    where
        mac.mac_from_ma_id = :ma_id and
        mac.mac_is_cashier = 1
    rows 1
    into :c_id;

    if (check_kind = '.create_loan.' and not exists (
                                                    select et_cardnumber
                                                    from extsys_token
                                                    where
                                                        et_from_c_id = :c_id and
                                                        et_is_actual = 1
                                                    )
    ) then
    begin
        res_error_text = 'У клиента не добавлена банковская карта в КМП';
        -- RFCKZ-877
        res_error_is_critical = 2;
        suspend;
    end

end
