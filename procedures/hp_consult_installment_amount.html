CREATE PROCEDURE HP_CONSULT_INSTALLMENT_AMOUNT(
    K_ID D_ID)
RETURNS (
    KC_ID D_ID,
    AMOUNT_TOTAL D_CURRENCY,
    AMOUNT_BALANCE D_CURRENCY,
    AMOUNT_RECOM D_CURRENCY,
    TEXT_RECOM D_STR_100,
    TEXT_INFO D_STR_100,
    K_INFO D_STR_100)
AS
declare variable k_cons_kind d_smallint;
declare variable res_amount d_currency;
declare variable payed_amount d_currency;
declare variable delinq_amount d_currency;
begin
  -- возвращает kc_id и суммы конс. услуг в рассрочку (RFCKG-4900)
  -- amount_total - полная стоимость конс. услуг
  -- amount_balance - общая задолженность (остаток долга от полной стоимости за вычетом погашений)
  -- amount_recom - рекомендуемая сумма к оплате (сумма ближайщего планового платежа + сумма просрочек)
  -- text_recom - текстовка к рекомендуемой сумме
  -- text_info - текстовка к типу оплаты

  amount_total = 0;
  amount_balance = 0;
  amount_recom = 0;
  text_recom = '';
  text_info = '';

  select k_cons_kind,
         k_kontract_no || ' от ' || f_datetostr(k_fact_disb_date)
  from kontract
  where k_id = :k_id
  into k_cons_kind,
       k_info;

  if (k_cons_kind is null) then
    k_cons_kind = 0;

  if (k_cons_kind = 0) then
  begin
    suspend;
    exit;
  end

  select kc_id
  from knclient
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0 rows 1
  into kc_id;

  select price
  from clt_get_price_for_consultation(:kc_id)
  into res_amount;

  amount_total = res_amount;

  if (res_amount > 0) then
  begin
    select sum(opk.ok_amount)
    from operkontract opk
    inner join operlist ol on ol.ol_id = opk.ok_from_ol_id and ol.is_delete = 0
    where opk.ok_from_k_id = :k_id and
          opk.ok_operation_type = f_get_rd_id('CashTransactionIncom.Consultation')
    into payed_amount;

    if (payed_amount is null) then
      payed_amount = 0;

    amount_balance = res_amount - payed_amount;

    if (amount_balance > 0) then -- есть задолженность
    begin
      select next_sh_amount + delinq_amount
      from hp_consult_payment_info(:k_id)
      into amount_recom;

      text_recom = 'Рекомендуемая сумма ' || round(amount_recom, 2) || ' coм';
      text_info = 'Оплата консультационных услуг ';
    end
  end

  if (kc_id is null) then
    kc_id = 0;

  if (amount_total is null) then
    amount_total = 0;

  if (amount_balance is null) then
    amount_balance = 0;

  if (amount_recom is null) then
    amount_recom = 0;

  suspend;
end
