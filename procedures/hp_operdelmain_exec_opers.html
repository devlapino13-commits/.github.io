CREATE PROCEDURE HP_OPERDELMAIN_EXEC_OPERS(
    IN_ODM_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable k_id d_id;
declare variable main_ol_id d_id;
declare variable loan_ol_id d_id;
declare variable oper_type d_integer;
declare variable oper_subtype d_integer;
declare variable res_code d_str_code;
declare variable odl_id d_integer;
declare variable odm_main_ol_id d_id;
declare variable new_ol_id d_id;
declare variable ol_date d_date;
declare variable is_writeoff d_smallint;
declare variable esm_id d_id;
declare variable odl_new_amount d_currency;
declare variable odl_new_account d_str_20;
declare variable odl_new_reg_id d_integer;
declare variable odl_new_payment_kind d_smallint;
declare variable ol_from_reg_id d_integer;
declare variable min_dd_date d_date;
declare variable max_dd_date d_date;
declare variable client_id d_id;
declare variable ol_doc_no d_integer;
declare variable ol_doc_no_str d_str_20;
begin
  res_is_error = 0;
  res_text = '';

  select odm_main_ol_id
  from operdelmain, operlist
  where odm_id = :in_odm_id and
        odm_main_ol_id = ol_id
  into odm_main_ol_id;

  if (row_count = 0) then
    res_text = 'Заявка на удаление операции не найдена: ' || coalesce(in_odm_id, '<Null>');

  if (res_text <> '') then
  begin
    res_is_error = 1;
    suspend;
    exit;
  end

  for select odk_from_k_id
  from operdelkont
  where odk_from_odm_id = :in_odm_id
  order by iif(odk_loan_ol_id is not null, 0, 1), odk_id
  into :k_id
  do
  begin
    for select odl_id, odl_main_ol_id, odl_loan_ol_id,
               iif(m.odm_main_ol_id = d.odl_main_ol_id, m.odm_oper_date, ol_date),
               odl_new_amount, odl_new_account, odl_new_reg_id, odl_new_payment_kind,
               coalesce(ol_oper_type, 0), coalesce(ol_oper_subtype, 0), ol_from_reg_id, ol_doc_no, ol_doc_no_str
    from operdellist d, operdelmain m, operlist ol
    where odl_from_odm_id = :in_odm_id and
          odl_from_k_id = :k_id and
          odl_new_ol_id is null and
          odl_new_action = 1 and
          odl_state = -1 and
          odl_loan_ol_id = ol.ol_id and
          m.odm_id = odl_from_odm_id
    order by ol_date, odl_id
    into :odl_id, :main_ol_id, :loan_ol_id,
         :ol_date, :odl_new_amount, :odl_new_account, :odl_new_reg_id, :odl_new_payment_kind,
         :oper_type, :oper_subtype, :ol_from_reg_id, :ol_doc_no, :ol_doc_no_str
    do
    begin
      res_code = '';

      select res_code
      from hp_operdel_operkind(:loan_ol_id, :oper_type, :oper_subtype)
      into :res_code;

      if (res_code = '' and odl_new_amount <> 0) then
        exception e_message 'Тип операции не определен, ol_id: ' || loan_ol_id;

      /*
      when 1 then 'disb'          X
      when 2 then 'repay'         ok
      when 3 then 'writeoff'      ok
      when 4 then 'reschedule'    X
      when 9 then 'transfer'      ok
      when 10 then 'overpayback'  X  они не должны попадать в список
      when 11 then 'overpayoff'   X  они не должны попадать в список
      when 12 then 'floatperc'
      when 13 then 'reverse'      X
      */

      esm_id = null;
      new_ol_id = null;
      is_writeoff = 0;
      min_dd_date = null;
      max_dd_date = null;
      client_id = null;

      if (odl_new_amount is null) then
        odl_new_amount = 0;

      res_text = '';

      if (res_code in ('repay', 'writeoff') and odl_new_amount <> 0) then
      begin
        is_writeoff = 0;
        if (res_code = 'writeoff') then
          is_writeoff = 1;

        if (res_code = 'repay' and odl_new_payment_kind = 3) then
        begin
          select eso_from_esm_id
          from extsysoper
          where eso_from_ol_id = :main_ol_id
          into esm_id;

          if (esm_id is null) then
          begin
            res_text = 'Онлайн погашение не найдено, ol_id: ' || main_ol_id;
            res_is_error = 1;
            suspend;
            exit;
          end

          select kc_from_c_id
          from knclient
          where kc_from_k_id = :k_id and 
                kc_amount_approved > 0
          order by kc_status desc
          rows 1
          into client_id;

          update extsysmain
          set esm_from_k_id = :k_id,
              esm_date = :ol_date
          where esm_id = :esm_id and
                esm_from_k_id is distinct from :k_id;

          update extsysmain
          set esm_object_id = :client_id
          where esm_id = :esm_id and
                esm_object_kind = 0;

          update extsysmain
          set esm_status = 0
          where esm_id = :esm_id;
    
          select err_text
          from hp_extsys_process_one(:esm_id, :odl_id)
          into res_text;
        end else
        begin
          for select * from operlist
          where ol_id = :loan_ol_id
          as cursor cur
          do
          begin
            select ol_id, error_text
            from hp_do_repayment(:k_id, 0, :ol_date, :odl_new_amount, 1, 1, 1, :odl_id, 1, 0, :is_writeoff, :odl_new_reg_id, :odl_new_payment_kind, :odl_new_account, 1, :un_id,
                                 cur.ol_document_client, cur.ol_doc_client_name, cur.ol_manager_c_id, cur.ol_from_rd_id)
            into :new_ol_id, :res_text;

            update operlist
            set ol_doc_no = :ol_doc_no,
                ol_doc_no_str = :ol_doc_no_str
            where ol_id = :new_ol_id or ol_parent_id = :new_ol_id;
          end
        end
      end else
      if (res_code = 'transfer') then
      begin
        select min(ok_sh_date), max(ok_sh_date)
        from operkred
        where ok_from_ol_id = :loan_ol_id and
              ok_from_k_id = :k_id
        into min_dd_date, max_dd_date;

        if (min_dd_date is not null) then
        begin
          select new_ol_id
          from hp_operdel_dd_transfer(:ol_from_reg_id, :k_id, :min_dd_date, :max_dd_date, :max_dd_date + 1, :un_id)
          into :new_ol_id;
        end
      end else
      if (res_code = 'floatperc') then
      begin
        execute procedure hp_do_reschedule_float(:k_id, :ol_date, :un_id);

        select distinct ol_id
        from operkred, operlist
        where ok_from_k_id = :k_id and
              ok_ol_date = :ol_date and
              ok_ol_is_active = 1 and
              ok_from_ol_id = ol_id and
              ol_oper_type = 1 and
              ol_oper_subtype = 12
        into new_ol_id;
      end

      if (res_text <> '') then
      begin
        res_is_error = 1;
        suspend;
        exit;
      end

      update operdellist
      set odl_new_ol_id = coalesce(:new_ol_id, odl_new_ol_id),
          odl_state = 1
      where odl_id = :odl_id;
    end
  end

  update operdellist
  set odl_state = 1
  where odl_from_odm_id = :in_odm_id and
        odl_main_ol_id = :odm_main_ol_id;

  suspend;
end
