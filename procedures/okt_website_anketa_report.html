CREATE PROCEDURE OKT_WEBSITE_ANKETA_REPORT(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ESM_DATETIME TIMESTAMP,
    ESM_DATE DATE,
    RESULT STR_255,
    OPERATOR_FIO STR_100,
    CL_NAME STR_255,
    GENDER STR_255,
    PHONE_NO STR_255,
    PHONE_NO_2 STR_255,
    CL_CITIZENSHIP STR_255,
    CL_CITY STR_255,
    CL_BIRTHDATE STR_255,
    CL_KOLICH_REPAY INTEGER,
    CL_LOAN_AMOUNT STR_255,
    DOCS STR_255,
    PASSNO STR_255,
    PASSDISBDATE STR_255,
    PASSEXPIRATIONDATE STR_255,
    ISEMPLOYED STR_255,
    OTHERDEBTS STR_255,
    PRODUCT STR_255,
    SVZRUN STR_255,
    SVZRUNDATE STR_255,
    SVZRUNADR STR_255,
    OFFICEADRSMSSEND STR_255,
    INFOSOURCE STR_255,
    MOSMSSEND STR_255,
    CL_NEXT_CALL_VALUE STR_255,
    MEMO STR_255,
    ACL_LAST_DATE DATE,
    ACL_LAST_DATETIME TIMESTAMP)
AS
declare variable rd_id integer;
declare variable acl_id id_type;
declare variable acl_object_kind integer;
declare variable acl_object_id id_type;
declare variable ques_code str_255;
declare variable ques_text str_255;
declare variable res_text str_255;
declare variable un_id id_type;
declare variable un_id_old id_type;
begin
  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'OKTELL.Projects.WebSite'
  into :rd_id;

  un_id_old = 0;

  for select al.acl_id, al.acl_object_kind, al.acl_object_id, al.acl_date, al.acl_last_datetime, al.acl_from_un_id
      from actionlist al
      where al.acl_from_rd_id = :rd_id and
            al.acl_status >= 0 and
            al.acl_parent_id is not null and
            al.acl_date between :rep_date_begin and :rep_date_end
      union all
      select al.acl_id, al.acl_object_kind, al.acl_object_id, al.acl_date, al.acl_last_datetime, al.acl_from_un_id
      from actionlist al
      where al.acl_from_rd_id = :rd_id and
            al.acl_status = -2 and
            al.acl_parent_id is null and
            al.acl_date between :rep_date_begin and :rep_date_end
      into :acl_id, :acl_object_kind, :acl_object_id, :acl_last_date, :acl_last_datetime, :un_id
  do
  begin
    if (un_id <> un_id_old) then
    begin
      select cl.c_name_fio
      from username un
      inner join clients cl on cl.c_id = un.un_from_c_id
      where un.un_id = :un_id
      into :operator_fio;

      un_id_old = un_id;
    end

    select em.esm_datetime, em.esm_date
    from extsysmain em
    where em.esm_id = :acl_object_id
    into :esm_datetime, :esm_date;

    for select ques_code, ques_text, res_text
        from hp_anketa_print(:acl_object_kind, :acl_object_id, 'OKTELL.Projects.WebSite', null)
        into :ques_code, ques_text, res_text
    do
    begin
      res_text = replace(res_text, 'Значение:', '');
      res_text = replace(res_text, 'Значение', '');

      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Cl_Name') then
        cl_name = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Gender') then
        gender = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Phone_No') then
        phone_no = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Phone_No_2') then
        phone_no_2 = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Cl_Citizenship') then
        cl_citizenship = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Cl_City') then
        cl_city = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Cl_BirthDate') then
        cl_birthdate = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Cl_Kolich_Repay') then
        cl_kolich_repay = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Cl_Loan_Amount') then
        cl_loan_amount = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Docs') then
        docs = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.PassNo') then
        passno = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.PassDisbDate') then
        passdisbdate = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.PassExpirationDate') then
        passexpirationdate = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.IsEmployed') then
        isemployed = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.OtherDebts') then
        otherdebts = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Product') then
        product = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.SVZrun') then
        svzrun = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.SVZrunDate') then
        svzrundate = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.SVZrunAdr') then
        svzrunadr = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.OfficeAdrSMSSend') then
        officeadrsmssend = res_text;
      else
      if (ques_code = 'AnketaQuestions.InfoSource') then
        infosource = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.MOSmsSend') then
        mosmssend = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Result') then
        result = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Cl_Next_Call_Value') then
        cl_next_call_value = res_text;
      else
      if (ques_code = 'AnketaQuestions.OKTELL.WebSite.Memo') then
        memo = res_text;
    end

    suspend;
  end
end
