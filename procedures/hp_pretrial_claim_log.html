CREATE PROCEDURE HP_PRETRIAL_CLAIM_LOG(
    EDL_ID D_ID)
AS
declare variable edl_object_id d_id;
declare variable k_id d_id;
declare variable c_id d_id;
declare variable msg_datetime d_datetime;
declare variable smr_lawyer_fio d_str_100;
begin
  -- добавление отправленнной клиенту претензии досудебной в журнал учета действий модуля досудебных взысканий (RFCRU-5302)

  select edl_object_id
  from edoc_list
  where edl_id = :edl_id
  into edl_object_id;

  select smr_from_k_id,
         smr_from_c_id
  from sud_mainreq
  where smr_id = :edl_object_id
  into k_id,
       c_id;

  if (exists(select *
             from pre_sud_kontract_list
             where pskl_k_id = :k_id)) then
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into smr_lawyer_fio; -- ФИО сотрудника создавшего запрос на отправку в суд

    select kc_from_c_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into c_id; -- клиент

    select msg_datetime
    from fsrobot_msgbox
    inner join edoc_msg on edm_from_msg_id = msg_id
    inner join edoc_sign on eds_id = edm_from_eds_id
    where msg_type = 1 and
          eds_from_edl_id = :edl_id and
          eds_from_c_id = :c_id
    order by msg_datetime desc
    rows 1
    into msg_datetime; -- дата и время отправки СМС

    execute procedure hp_pre_sud_action_log_ins(f_enum_id('pre_sud_actions', 'pretrial_claim', 1), :edl_id, :k_id, :msg_datetime, :smr_lawyer_fio);
  end
end
