CREATE PROCEDURE AAREP_RESTRUKT_DT_CLIENT(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    OL_ID DECIMAL(18,0),
    REG_ID INTEGER,
    RESTRUCT_DATE DATE,
    OL_DATE DATE,
    OL_MEMO VARCHAR(1024),
    KF_NAME VARCHAR(100),
    REG_NAME VARCHAR(50),
    PARENT_ID INTEGER,
    PARENT_NAME VARCHAR(50),
    K_KREDLINE_END DATE,
    K_DATE_END DATE,
    KOL_PROLOG INTEGER,
    REP_ACTIVE_BALANCE DECIMAL(18,2),
    RT_KIND VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    PR_NAME VARCHAR(50),
    PR_COMMON_NAME VARCHAR(100),
    REP_ACTIVE_BALANCE_NEW DECIMAL(18,2),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    KC_FROM_C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    BT_NAME VARCHAR(50),
    BS_NAME VARCHAR(50),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    REG_STAT_REP_REGION VARCHAR(255),
    K_PERIOD DECIMAL(15,2),
    K_FACT_DISB_DATE DATE,
    K_PERCENT DECIMAL(18,7),
    K_PERCENT_AFTER DECIMAL(18,7),
    TYPE_FINANCING VARCHAR(20),
    ZAD_PERC_CL DECIMAL(18,2),
    SH_CALC_FINE_CL DECIMAL(18,2),
    CS_NAME D_STR_100,
    KOL_RESTR D_INTEGER,
    OL_DATETIME_FIRST DATE,
    OL_DATETIME_LAST DATE,
    LKC_STATUS SMALLINT,
    IS_DELINQ SMALLINT,
    NO_ACCRUAL_STATUS SMALLINT,
    OKS_DELINQ_DAYS INTEGER,
    K_DISB_KINDD D_STR_100,
    K_BY_RATING D_STR_64)
AS
declare variable G integer;
declare variable OL_DATETIME date;
declare variable OR_FROM_OL_ID D_ID;
declare variable OL_DT date;
declare variable OS_SUM decimal(18,4);
declare variable DT_MAX date;
declare variable OK_TYPE integer;
declare variable CURR varchar(3);
declare variable DEF_CURR varchar(3);
declare variable CURR_RATE decimal(18,4);
declare variable K_INDEX_CURR_RATE decimal(18,4);
declare variable PR_SET_RATE_IN_KONT integer;
declare variable REG_PRIOR integer;
declare variable ST_STAFF integer;
declare variable ST_CLIENT integer;
declare variable CM_IS_CLIENT_MODE integer;
declare variable K_ID_PRIOR ID_TYPE;
declare variable KC_BUSUNESS_SUBTYPE integer;
declare variable KC_BUSUNESS_SUBTYPE_PRIOR integer;
declare variable GR_BALANCE_REST CURR_TYPE;
declare variable GR_BALANCE CURR_TYPE;
declare variable PR_CODE varchar(100);
declare variable DB_ID integer;
declare variable ZAD_PERC decimal(18,2);
declare variable SH_CALC_FINE decimal(18,2);
declare variable C_ID D_ID;
declare variable RD_MEMO D_STR_255;
begin
  -- operkred_fix
  -- RFCKG-3540

  db_id = f_get_db_id();

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  select cm_default_currency, cm_is_client_mode
  from company
  into :def_curr, cm_is_client_mode;

  reg_prior = 0;
  kc_busuness_subtype_prior = -1;
  if (cm_is_client_mode = 0) then
  begin
    for select k_id, reg_id, restruct_date, kf_name, reg_name, parent_name, k_date_end, kol_prolog, rep_active_balance,
               rt_kind, k_kontract_no, pr_name, pr_common_name, rep_active_balance_new, k_kolich_client_approved,
               k_amount_approved, reg_stat_rep_region, k_period, k_fact_disb_date, k_percent, pr_code, ol_date, k_disb_kindd, k_by_rating
        from aarep_restrukt_dt(:rep_region_id, :rep_date_end)
        into :k_id, :reg_id, :restruct_date, :kf_name, :reg_name, :parent_name, :k_date_end, :kol_prolog, :gr_balance,
             :rt_kind, :k_kontract_no, :pr_name, :pr_common_name, :gr_balance_rest, :k_kolich_client_approved,
             :k_amount_approved, :reg_stat_rep_region, :k_period, :k_fact_disb_date, :k_percent, :pr_code, :ol_date, :k_disb_kindd, :k_by_rating
    do
    begin
      kol_restr = 0;

      for select ol.ol_id,
                 cast(ol_datetime as date),
                 rd.rd_memo,
                 ol.ol_memo
          from operrest ot
          inner join operlist ol on ol.ol_id = ot.or_from_ol_id
          inner join refdata rd on rd_id = or_from_rd_id
          where or_from_k_id = :k_id and
                or_ol_is_active = 1 and
                ol.is_delete = 0 and
                ot.or_ol_date <= :rep_date_end and
                rd_code not containing 'Reason_Reschedule.4085'
          order by or_ol_date asc, ol.ol_datetime asc
          into :ol_id,
               :ol_datetime,
               :rd_memo,
               :ol_memo
      do
      begin
        if (db_id = 41701 and :rd_memo not containing 'Реструктуризация') then
            continue;

        -- RFCKZ-1089
        if (db_id = 39801 and :ol_memo not containing 'Реструктуризация') then
            continue;

        kol_restr = kol_restr + 1;
        if (kol_restr = 1) then
          ol_datetime_first = ol_datetime;
      end
      ol_datetime_last = ol_datetime;
      /*   ---===
      kol_restr = 0;

      select count(distinct ot.or_from_ol_id)
      from operrest ot
      inner join refdata on rd_id = or_from_rd_id
      where or_from_k_id = :k_id and
            or_ol_is_active = 1 and
            ot.or_ol_date <= :rep_date_end and
            rd_code not containing 'Reason_Reschedule.4085'
      into :kol_restr;

      or_ol_date = null;
      select ot.or_ol_date
      from operrest ot
      inner join refdata on rd_id = or_from_rd_id
      where or_from_k_id = :k_id and
            or_ol_is_active = 1 and
            ot.or_ol_date <= :rep_date_end and
            rd_code not containing 'Reason_Reschedule.4085'
      order by  or_ol_date desc
      rows 1
      into :or_ol_date;

     --===    */
      select sh_date
      from schedule
      where sh_from_k_id = :k_id
      order by sh_id desc
      rows 1
      into :k_kredline_end; --BOARD-2047

      zad_perc = 0;
      select sum(zad_perc + sh_add_perc), sum(sh_calc_fine)
      from aarep_make_zad_list(:k_id, :rep_date_end, 1)
      where sh_date < :rep_date_end
      into :zad_perc, sh_calc_fine; --BOARD-2047

      k_percent_after = null;
      select round(kd_percent, 2) k_percent
      from kontdata kd
      where kd.kd_ol_is_delete = 0 and
            kd.kd_from_k_id = :k_id and
            :rep_date_end between kd_ol_date and kd.kd_date_end
      into k_percent_after;

      if (k_percent_after = 0) then
        k_percent_after = k_percent;
      /*
      select k_period,
             k_fact_disb_date,
             k_percent
      from kontract
      where k_id = :k_id
      into k_period,
           k_fact_disb_date,
           k_percent;
      */
      if (db_id in (39801, 41701)) then
      begin
        if (pr_code containing '.ipf.') then
          type_financing = 'ИПФ';
        else
          type_financing = 'Классика';
      end

      if (db_id = 39801 and (:rt_kind is null or :rt_kind = '')) then
      begin
        :rt_kind = 'Фин. трудности  (иное/другое)';
      end

      for select kc_amount_approved, c_name_fio, kc_busuness_subtype, c_id
          from knclient, clients
          where kc_from_k_id = :k_id and
                kc_amount_approved > 0 and
                kc_from_c_id = c_id
          into kc_amount_approved, c_name_fio, kc_busuness_subtype, c_id
      do
      begin
        if (kc_busuness_subtype_prior <> kc_busuness_subtype) then
        begin
          kc_busuness_subtype_prior = kc_busuness_subtype;
          select bt_name, bs_name
          from bustype, bussubtype
          where bs_id = :kc_busuness_subtype and
                bs_from_bt_id = bt_id
          into bt_name, bs_name;
        end

        rep_active_balance = cast(kc_amount_approved as double precision) / k_amount_approved * gr_balance;
        rep_active_balance_new = cast(kc_amount_approved as double precision) / k_amount_approved * gr_balance_rest;

        zad_perc_cl = 0;
        zad_perc_cl = (zad_perc * kc_amount_approved) / k_amount_approved;

        sh_calc_fine_cl = 0;
        sh_calc_fine_cl = (sh_calc_fine * kc_amount_approved) / k_amount_approved;

        cs_name = null;
        select cs_name, lkc_status
        from llc_kont_cli, llc_kont, classific
        where lk_from_k_id = :k_id and
              lkc_from_k_id = :k_id and
              lkc_from_c_id = :c_id and
              lk_from_lm_id = lkc_from_lm_id and
              cs_id = lkc_status and
              :rep_date_end between lk_date and lk_date_end
        into :cs_name, :lkc_status;

        select iif(lkc_deliq_days > 0, 1, 0) as is_delinq
        from llc_main, llc_kont_cli
        where lm_date_status = :rep_date_end and
              lkc_from_k_id = :k_id and
              lm_id = lkc_from_lm_id rows 1
        into :is_delinq;

        no_accrual_status = 0;
        select iif(k_stop_perc <= :rep_date_end, 1, 0), cast(:rep_date_end as date) - oks_delinq_date
        from kontract
        inner join oksummary oks on oks.oks_from_k_id = k_id
        where k_id = :k_id and
              oks.oks_close_state = 0 and
              :rep_date_end between oks.oks_date_begin and oks.oks_date_end
        into no_accrual_status, oks_delinq_days;

        if (oks_delinq_days < 0) then
          oks_delinq_days = null;
        suspend;
      end
    end
  end
