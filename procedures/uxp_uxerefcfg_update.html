Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXEREFCFG_UPDATE(
    NEW_ERC_ERL_UUID D_UUID,
    NEW_ERC_SHOW_INACTIVE D_BOOL,
    NEW_ERC_INT_TAG_CAPTION D_STR_CODE,
    NEW_ERC_STR_TAG_CAPTION D_STR_CODE,
    NEW_ERC_VISIBLE_ID D_BOOL,
    NEW_ERC_VISIBLE_CODE D_BOOL,
    NEW_ERC_VISIBLE_NAME D_BOOL,
    NEW_ERC_VISIBLE_MEMO D_BOOL,
    NEW_ERC_VISIBLE_ORDERNO D_BOOL,
    NEW_ERC_VISIBLE_IS_ACTIVE D_BOOL,
    NEW_ERC_VISIBLE_STR_ID D_BOOL,
    NEW_ERC_VISIBLE_TAG D_BOOL,
    NEW_ERL_LOOKUP D_SMALLINT)
RETURNS (
    ERC_ERL_UUID D_UUID,
    ERC_SHOW_INACTIVE D_BOOL,
    ERC_INT_TAG_CAPTION D_STR_CODE,
    ERC_STR_TAG_CAPTION D_STR_CODE,
    ERC_VISIBLE_ID D_BOOL,
    ERC_VISIBLE_CODE D_BOOL,
    ERC_VISIBLE_NAME D_BOOL,
    ERC_VISIBLE_MEMO D_BOOL,
    ERC_VISIBLE_ORDERNO D_BOOL,
    ERC_VISIBLE_IS_ACTIVE D_BOOL,
    ERC_VISIBLE_STR_ID D_BOOL,
    ERC_VISIBLE_TAG D_BOOL,
    ERL_LOOKUP D_SMALLINT,
    ERL_UUID_CODE D_STR_CODE)
AS
begin
  update ux$ereflist
  set erl_lookup = :new_erl_lookup
  where erl_uuid = :new_erc_erl_uuid
  returning erl_lookup
  into erl_lookup;

  update ux$erefcfg
  set erc_show_inactive = :new_erc_show_inactive,
      erc_int_tag_caption = :new_erc_int_tag_caption,
      erc_str_tag_caption = :new_erc_str_tag_caption,
      erc_visible_id = :new_erc_visible_id,
      erc_visible_code = :new_erc_visible_code,
      erc_visible_name = :new_erc_visible_name,
      erc_visible_memo = :new_erc_visible_memo,
      erc_visible_orderno = :new_erc_visible_orderno,
      erc_visible_is_active = :new_erc_visible_is_active,
      erc_visible_str_id = :new_erc_visible_str_id,
      erc_visible_tag = :new_erc_visible_tag
  where erc_erl_uuid = :new_erc_erl_uuid
  returning erc_erl_uuid,
            erc_show_inactive,
            erc_int_tag_caption,
            erc_str_tag_caption,
            erc_visible_id,
            erc_visible_code,
            erc_visible_name,
            erc_visible_memo,
            erc_visible_orderno,
            erc_visible_is_active,
            erc_visible_str_id,
            erc_visible_tag
  into erc_erl_uuid, 
       erc_show_inactive, 
       erc_int_tag_caption, 
       erc_str_tag_caption, 
       erc_visible_id, 
       erc_visible_code, 
       erc_visible_name, 
       erc_visible_memo, 
       erc_visible_orderno, 
       erc_visible_is_active, 
       erc_visible_str_id, 
       erc_visible_tag;

  erl_uuid_code = uuid_to_char(erc_erl_uuid);

  suspend;
end
