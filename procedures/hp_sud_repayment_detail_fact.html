CREATE PROCEDURE HP_SUD_REPAYMENT_DETAIL_FACT(
    SMR_ID D_ID)
RETURNS (
    OKS_DATE D_DATE,
    OKS_PMT_OS D_CURRENCY,
    OKS_PMT_PERC D_CURRENCY,
    OKS_PMT_ALL D_CURRENCY,
    OKS_DISB_AMOUNT D_CURRENCY,
    OKS_BAL_TO_PERC D_CURRENCY,
    OKS_DAYS D_INTEGER,
    OKS_CALC_PERC D_CURRENCY,
    OKS_BALANCE D_CURRENCY,
    OKS_BAL_PERC D_CURRENCY,
    OKS_PERCENT_NORM D_PERCENT)
AS
declare variable k_id d_id;
begin
  select smr_from_k_id
  from sud_mainreq
  where smr_id = :smr_id
  into k_id;

  for select oks_date, oks_pmt_os, oks_pmt_perc, oks_pmt_all, oks_disb_amount, oks_bal_to_perc, oks_days, oks_calc_perc,
         oks_balance, oks_bal_perc, oks_percent_norm
  from hp_repayment_detail_fact(:k_id, current_date)
  into :oks_date, :oks_pmt_os, :oks_pmt_perc, :oks_pmt_all, :oks_disb_amount, :oks_bal_to_perc, :oks_days, :oks_calc_perc,
       :oks_balance, :oks_bal_perc, :oks_percent_norm
  do
    suspend;
end
