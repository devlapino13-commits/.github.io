CREATE PROCEDURE AAREP_INSURANCE_PAYMENTS_REPORT(
    REP_REG_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_64,
    KAL_NEW_KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    C_NAME_FIO D_STR_255,
    LO_NAME_FIO D_STR_255,
    K_FACT_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    PAYED_CONSULT_AMOUNT D_CURRENCY,
    K_KREDLINE_END D_DATE,
    INSURANCE_COMP_NAME D_STR_100,
    POLICY_NO D_STR_64,
    COMMISSION D_CURRENCY,
    NSP D_CURRENCY,
    MBULAK_SHARE D_CURRENCY,
    SETTLEMENT_AMOUNT D_CURRENCY,
    K_FACT_CLOSED_DATE D_DATE,
    STATUS D_STR_64,
    LO_INN D_STR_20,
    C_INN D_STR_20,
    PR_NAME D_STR_CODE,
    PR_KIND D_STR_20,
    TARIF PERC_TYPE)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable ag_id d_id;
declare variable reg_unicode d_integer;
declare variable lo_id d_id;
declare variable n_value d_integer;
declare variable pr_code d_str_100;
declare variable k_disb_date d_date;
declare variable k_score d_integer;
begin
    for select distinct
        t.id_list,
        kal.kal_new_kf_name,
        k.k_fact_disb_lo_id,
        k.k_kontract_no,
        k.k_fact_disb_date,
        k.k_amount_approved,
        k.k_kredline_end,
        ik.ik_policy_no,
        ik.ik_commission / 102 * 100,
        ik.ik_policy_company_share,
        ik.ik_tax,
        ik.ik_mbulak_share,
        ik.ik_from_ag_id,
        k.k_fact_closed_date,
        iif(k.k_fact_closed_date is null, 'активный', 'закрыт') as status,
        kc.kc_from_c_id,
        k.k_id,
        pr.pr_name,
        pr.pr_code,
        iif(pr.pr_code containing '.ipf.', 'ИПФ', 'Классика'),
        k.k_disb_date,
        k.k_kredline_end - k.k_disb_date
    from kontstate ks
    inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
    inner join insurance_kontract ik on ik.ik_from_k_id = ks.ks_from_k_id
        and ((ik.ik_enum_status = f_enum_id('insurance_kontract', 'insurance_policy_obtained')
             or ik.ik_enum_status = f_enum_id('insurance_kontract', 'insurance_policy_app_was_sent')) or
             ik.ik_from_ag_id in (7414, 7570))
    inner join kontract k on k.k_id = ks.ks_from_k_id
    inner join product pr on k.k_from_pr_id = pr.pr_id
    inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
        and kal.is_delete = 0
        and :rep_date_end between kal.kal_date and kal.kal_date_end
    inner join region_list(:rep_reg_id, 1, 3, 1) t on t.id_list = ks.ks_region_id
    where ks.ks_ol_is_delete = 0
        and ks.ks_state = 1
        and ks.ks_date_begin between :rep_date_begin and :rep_date_end
        -- and pr.pr_code not containing '.ipf.' -- RFCKG-5348
        and (
            (k.k_fact_disb_date < '10.11.2025'
                and kc.kc_amount_approved >= 70000
                and pr.pr_code containing '.IND.'
                and (pr.pr_code containing '.STANDART.' or
                     pr.pr_code containing '.IPF.') -- RFCKG-5348
                and pr.pr_code not containing '.consultation.')
            or
            (k.k_fact_disb_date >= '10.11.2025')
        )
    into
        :reg_unicode,
        :kal_new_kf_name,
        :lo_id,
        :k_kontract_no,
        :k_fact_disb_date,
        :k_amount_approved,
        :k_kredline_end,
        :policy_no,
        :commission,
        :settlement_amount,
        :nsp,
        :mbulak_share,
        :ag_id,
        :k_fact_closed_date,
        :status,
        :c_id,
        :k_id,
        :pr_name,
        :pr_code,
        :pr_kind,
        :k_disb_date,
        :n_value
    do
    begin
        select reg.reg_name
        from region reg
        where reg.reg_unicode = :reg_unicode
        into :reg_name;

        select c.c_name_fio,
               c.c_inn
        from clients c
        where c.c_id = :c_id
        into :c_name_fio,
             :c_inn;

        select c.c_name_fio,
               c.c_inn
        from clients c
        where c.c_id = :lo_id
        into :lo_name_fio,
             :lo_inn;

        select sum(oko.ok_amount)
        from operkontract oko
        inner join operlist ol on oko.ok_from_ol_id = ol.ol_id
            and ol.is_delete = 0
        where oko.ok_from_k_id = :k_id
            and oko.ok_operation_type = f_get_rd_id('CashTransactionIncom.Insurance')
            and ol.ol_date <= :rep_date_end
        into :payed_consult_amount;

        select c_insur.c_name_fio
        from agreements ag
        inner join clients c_insur on c_insur.c_id = ag.ag_object_id
        where ag.ag_id = :ag_id
        into :insurance_comp_name;

        -- RFCKG-4966
        /*
        Тариф был очень маленьким (0.00005479452), и при умножении на большие суммы возникало переполнение (Integer overflow).
        Чтобы избежать ошибки и сохранить точность, тариф увеличили на 100000,
        а в формуле расчёта комиссии добавили деление на 100000 для восстановления исходного масштаба.
        */
        --tarif = iif(f_get_const('new_loan_conditions') = 1, 0.0000548, 0.00005);
        tarif = iif(f_get_const('new_loan_conditions') = 1, 5.479452, 5);

        if (pr_code not containing '.ipf.' and n_value <= 365) then
            n_value = 365;

        k_score = coalesce(f_point(:c_id, :k_disb_date, iif(:pr_code containing '.ipf.', 2, 0)), 0);

        if (pr_code containing '.sputnik.' and :k_fact_disb_date between '17.11.2025' and '17.02.2026') then
            tarif = 10.958904;
        else if (pr_code not containing '.consultation.' and pr_code not containing '.sputnik.' and
                 pr_code not containing '.ipf.' and pr_code containing '.standart.' and
                 pr_code containing '.ind.' and :k_fact_disb_date between '15.01.2026' and '17.02.2026') then
        begin
            tarif = case
                        when :k_score between 0 and 12 then 16.438000  -- 0.06
                        when :k_score between 13 and 30 then 13.699000 -- 0.05
                        when :k_score between 31 and 52 then 10.959000 -- 0.04
                        else 8.219000 -- 0.03
                    end;
        end
        else if (pr_code not containing '.consultation.' and pr_code not containing '.sputnik.' and
                 pr_code not containing '.ipf.' and pr_code containing '.standart.' and
                 pr_code containing '.ind.' and
                 :k_fact_disb_date between '17.11.2025' and '15.01.2026') then
            tarif = 8.219000;

        if (:k_disb_date >= '18.02.2026') then
        begin
            if (:k_disb_date >= '12.03.2026') then
            begin
                if (pr_code containing '.sputnik.' or
                    pr_code containing '.insur_base.') then
                begin
                    tarif = case
                              when :k_score between 0 and 19 then 13.699000  -- 5%
                              when :k_score between 20 and 41 then 12.329000  -- 4.5%
                              else 10.959000                                 -- 4%
                            end;
                end
                else if (pr_code not containing '.consultation.'
                         and pr_code not containing '.ipf.'
                         and pr_code containing '.standart.'
                         and pr_code containing '.ind.') then
                begin
                    tarif = case
                                when :k_score between 0 and 41 then 16.438000  -- 6%
                                else 13.699000                                 -- 5%
                            end;
                end
                else if (pr_code containing '.ipf.') then -- RFCKG-5348
                begin
                    tarif = case
                                when :k_score between 0 and 19 then 19.178000 -- 7%

                                when :k_score between 20 and 30 and
                                     :k_kredline_end <= dateadd(month, 6, :k_disb_date) then 10.958904 -- 4%

                                when :k_score between 20 and 30 then 16.438000 -- 6%

                                when :k_kredline_end <= dateadd(month, 6, :k_disb_date) then 8.219000 -- 3%

                                else 13.699000 -- 5%
                            end;
                end
            end
            else
            begin
                if (pr_code containing '.sputnik.') then
                begin
                    tarif = case
                              when :k_score between 0 and 19 then 13.699000  -- 5%
                              else 10.959000                                 -- 4%
                            end;
                end
                else if (pr_code not containing '.consultation.'
                         and pr_code not containing '.ipf.'
                         and pr_code containing '.standart.'
                         and pr_code containing '.ind.') then
                begin
                    if (:k_disb_date >= '12.03.2026') then
                    tarif = case
                              when :k_score between 0 and 41 then 16.438000  -- 6%
                              else 13.699000                                 -- 5%
                            end;
                    else
                    tarif = case
                                when :k_score between 0 and 12 then 16.438000  -- 6%
                                else 13.699000                                 -- 5%
                            end;
                end
            end
        end

        tarif = round(:tarif / 273.98 * 100, 2);

        suspend;
    end
end
