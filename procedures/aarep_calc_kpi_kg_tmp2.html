CREATE PROCEDURE AAREP_CALC_KPI_KG_TMP2(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    K_KOLICH INTEGER,
    K_KC_ID DECIMAL(18,0),
    K_REGION_ID INTEGER,
    K_BALANCE DECIMAL(18,4),
    K_JOBNAME VARCHAR(255),
    K_IS_KC INTEGER,
    K_STAGE_BEGIN DATE,
    K_IS_WORKING VARCHAR(20),
    K_NEED_KPI INTEGER,
    K_BRANCH_CODE VARCHAR(20),
    K_REGION_NAME VARCHAR(100),
    K_KC_NAME VARCHAR(100),
    K_BRANCH_NAME VARCHAR(100))
AS
declare variable k_is_work integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable jk_code varchar(100);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select sum(k_kolich), k_kc_id, k_region_id, sum(k_balance)
      from aarep_calc_kpi_kg_tmp(:rep_region_id, :rep_date_end)
      group by k_kc_id, k_region_id
      into :k_kolich, :k_kc_id, :k_region_id, :k_balance
  do
  begin
    k_is_kc = 0;
    k_jobname = '';
    k_stage_begin = rep_date_end + 1;
    k_is_work = 0;
    k_is_working = '';
    k_need_kpi = 0;

    k_is_kc = 0;
    for select jl_name, jk_system_code, ps_is_working
        from hr_pstatus, joblist, jobkind
        where ps_from_c_id = :k_kc_id and
              ps_from_reg_id = :k_region_id and
              ps_date_begin <= :rep_date_end and
              ps_date_end >= :rep_date_end and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id
        order by ps_is_working desc
        rows 1
        into :k_jobname, :jk_code, :k_is_work
    do
    begin
      if (jk_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
        k_is_kc = 1;
      if (k_is_work is null) then
        k_is_work = 0;
    end
    if (k_jobname is null) then
      k_jobname = '';
    if (k_jobname = '') then
    begin
      for select reg_code || ' ' || reg_name || ', ' || jl_name
          from hr_pstatus, joblist, region
          where ps_from_c_id = :k_kc_id and
                ps_from_reg_id = reg_unicode and
                ps_date_begin <= :rep_date_end and
                ps_date_end >= :rep_date_end and
                ps_from_jl_id = jl_id
          order by ps_is_working desc
          rows 1
          into :k_jobname
      do
      begin
        if (k_jobname is null) then
          k_jobname = '';
      end
    end

    if (k_is_work = 1) then
      k_is_working = 'Работает';
    else
      k_is_working = 'Не работает';

    select min(ps_date_begin)
    from hr_pstatus, joblist
    where ps_from_c_id = :k_kc_id and
          ps_from_reg_id = :k_region_id and
          ps_from_jl_id = jl_id and
          jl_kind = 5 and
          ps_status = 1
    into :k_stage_begin;
    if (k_stage_begin is null) then
      k_stage_begin = rep_date_end + 1;

    if ((k_is_work = 1) and
        (k_is_kc = 1)) then
      k_need_kpi = 1;
    else
      k_need_kpi = 0;

    select left(r1.reg_code, 2), r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
    from region r1, region r2
    where r1.reg_unicode = :k_region_id and
          r1.parent = r2.reg_unicode
    into :k_branch_code, :k_region_name, :k_branch_name;

    select c_name_fio
    from clients
    where c_id = :k_kc_id
    into :k_kc_name;

    suspend;
  end
  /* Нулевые данные, список КС по модулю кадров */
  k_kolich = 0;
  k_balance = 0;
  for select ps_from_c_id, ps_from_reg_id
      from hr_pstatus, region_list(:rep_region_id, 1, -1, 1), joblist, jobkind
      where ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            (ps_from_reg_id = id_list) and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')
      into :k_kc_id, :k_region_id
  do
  begin
    k_is_kc = 0;
    k_jobname = '';
    k_stage_begin = rep_date_end + 1;
    k_is_work = 0;
    k_is_working = '';
    k_need_kpi = 0;

    for select jl_name, jk_system_code, ps_is_working
        from hr_pstatus, joblist, jobkind
        where ps_from_c_id = :k_kc_id and
              ps_from_reg_id = :k_region_id and
              ps_date_begin <= :rep_date_end and
              ps_date_end >= :rep_date_end and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id
        order by ps_is_working desc
        rows 1
        into :k_jobname, :jk_code, :k_is_work
    do
    begin
      if (jk_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
        k_is_kc = 1;
      if (k_is_work is null) then
        k_is_work = 0;
    end
    if (k_jobname is null) then
      k_jobname = '';
    if (k_jobname = '') then
    begin
      for select reg_code || ' ' || reg_name || ', ' || jl_name
          from hr_pstatus, joblist, region
          where ps_from_c_id = :k_kc_id and
                ps_from_reg_id = reg_unicode and
                ps_date_begin <= :rep_date_end and
                ps_date_end >= :rep_date_end and
                ps_from_jl_id = jl_id
          order by ps_is_working desc
          rows 1
          into :k_jobname
      do
      begin
        if (k_jobname is null) then
          k_jobname = '';
      end
    end

    if (k_is_work = 1) then
      k_is_working = 'Работает';
    else
      k_is_working = 'Не работает';

    select min(ps_date_begin)
    from hr_pstatus, joblist, jobkind
    where ps_from_c_id = :k_kc_id and
          ps_from_reg_id = :k_region_id and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
          ps_status = 1
    into :k_stage_begin;
    if (k_stage_begin is null) then
      k_stage_begin = rep_date_end + 1;

    if ((k_is_work = 1) and
        (k_is_kc = 1)) then
      k_need_kpi = 1;
    else
      k_need_kpi = 0;

    select left(r1.reg_code, 2), r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
    from region r1, region r2
    where r1.reg_unicode = :k_region_id and
          r1.parent = r2.reg_unicode
    into :k_branch_code, :k_region_name, :k_branch_name;

    select c_name_fio
    from clients
    where c_id = :k_kc_id
    into :k_kc_name;

    suspend;
  end
end
