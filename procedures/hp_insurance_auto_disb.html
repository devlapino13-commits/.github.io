CREATE PROCEDURE HP_INSURANCE_AUTO_DISB(
    K_ID D_ID,
    UN_ID D_ID)
AS
declare variable reg_id d_integer;
declare variable parent_k_id d_id;
declare variable ag_id d_integer;
declare variable c_id d_id;
declare variable manager_id d_id;
declare variable acc_kt d_str_20;
declare variable al_account_kt varchar(32);
declare variable acc_dt d_str_20;
declare variable al_account_dt varchar(32);
declare variable ol_id d_id;
declare variable c_name_fio d_str_100;
declare variable insurance_c_id d_id;
declare variable parent_kontract_no d_str_20;
declare variable parent_disb_date d_date;
declare variable policy_no d_str_20;
declare variable ik_commission d_currency;
declare variable ik_mbulak_share d_currency;
declare variable doc_no d_integer;
declare variable res_text d_str_255;
begin
  -- RFCKG-4831

  if (exists(select *                        -- Если контракт уже активный
             from kontstate ks
             where ks.ks_ol_is_delete = 0 and
                   ks.ks_from_k_id = :k_id and
                   ks.ks_state = 1 )) then
    exit;


  select ks.ks_region_id
  from kontstate ks
  where ks.ks_from_k_id = :k_id and
        ks.ks_state = 0
  into reg_id;

  -- Счета к получению по оплате страхового полиса
  select al_id,
         al_account
  from acc_list
  where al_from_reg_id = :reg_id and
        al_currency = '417' and
        al_account = '11531/301'
  into acc_dt,
       al_account_dt;

  select lir.lir_kontract_base
  from loan_insurance_relation lir
  where lir.lir_kontract_insurance = :k_id
  into parent_k_id;

  select k.k_kontract_no,
         k.k_disb_date,
         k.k_approved_manager,
         c.c_name_fio,
         c.c_id
  from kontract k
  inner join clients c on c.c_id = k.k_clients_cashier_id
  where k.k_id = :parent_k_id
  into parent_kontract_no,
       parent_disb_date,
       manager_id,
       c_name_fio,
       c_id;

  select ik.ik_from_ag_id,
         ik.ik_policy_no,
         ik.ik_commission,
         ik.ik_mbulak_share
  from insurance_kontract ik
  where ik.ik_from_k_id = :parent_k_id
  into ag_id,
       policy_no,
       ik_commission,
       ik_mbulak_share;

  select ag.ag_object_id,
         al.al_account
  from acc_list al
  inner join agreements ag on ag.ag_from_al_id = al.al_id
  where ag.ag_id = :ag_id
  into insurance_c_id,
       al_account_kt; -- короткий код счета (например 21122/301)

  select al_id
  from acc_list
  where al_from_reg_id = :reg_id and
        al_currency = '417' and
        al_account = :al_account_kt
  into acc_kt; -- счет страхования в офисе, где принимается платеж (RFCKG-5266)

  select num
  from last_doc_number(current_date, 9, :reg_id)
  into :doc_no;

  -- таблтца OPTLIST
  insert into operlist (ol_from_reg_id, ol_oper_type, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                        ol_status, is_delete, ol_user_id, ol_document_client,
                        ol_document_office, ol_oper_subtype, ol_date)
  values (:reg_id,
          2, -- денежная операция
          :doc_no, :doc_no, 7, f_format('Оплата за страхование кредита %1 от %2', :parent_kontract_no, f_datetostr(:parent_disb_date)), 1, -- ol_status
          0, -- is_delete
          :un_id, :c_id,
          :reg_id,
          9, -- Мемориальный ордер
          current_date)
  returning (ol_id)
  into ol_id;

  insert into operkontract ( ok_from_k_id, ok_from_ol_id, ok_amount, ok_operation_type)
  values (:parent_k_id, :ol_id, :ik_commission, f_get_rd_id('CashTransactionIncom.Insurance'));

  -- Создание операции и сохранение Operbook
  -- 1.  Признание счета к получению от клиента и задолженности по сумме страхового полиса перед Страховой Компанией
  insert into operbook (ob_from_ol_id, ob_number, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_curr_rate, ob_curr_unit, ob_account_kt, ob_client_kt, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt)
  values (:ol_id, 1, :c_id, :acc_dt, :ik_commission, :ik_commission, 1, 1, :acc_kt, :insurance_c_id,
          f_format('Страховая премия по страховому полису №%1, №%2 от %3 %4', :policy_no, :parent_kontract_no, f_datetostr(:parent_disb_date), :c_name_fio),
          current_date, 1, :reg_id, :al_account_dt, :al_account_kt);

  -- 2.  Признание комиссии М Булак
  acc_dt = acc_kt;
  al_account_dt = al_account_kt;

  -- ДБП комиссия агента по Страхованию
  select al_id,
         al_account
  from acc_list
  where al_from_reg_id = :reg_id and
        al_currency = '417' and
        al_account = '21114/20'
  into acc_kt,
       al_account_kt;

  insert into operbook (ob_from_ol_id, ob_number, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_curr_rate, ob_curr_unit, ob_account_kt, ob_client_kt, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt)
  values (:ol_id, 2, :insurance_c_id, :acc_dt, :ik_mbulak_share, :ik_mbulak_share, 1, 1, :acc_kt, :insurance_c_id,
          f_format('Комиссия МБулак по дог. о сотрудничестве по страхованию (№%1, № %2 от %3, %4)', :policy_no, :parent_kontract_no, f_datetostr(:parent_disb_date), :c_name_fio),
          current_date, 1, :reg_id, :al_account_dt, :al_account_kt);

  -- Выдача займа на страховой полис
  select al_id
  from acc_list
  where al_from_reg_id = :reg_id and
        al_currency = '417' and
        al_account = '10941/15'
  into acc_dt;

  select al_id
  from acc_list
  where al_from_reg_id = :reg_id and
        al_currency = '417' and
        al_account = '11531/301'
  into acc_kt;

  select res_text
  from hp_do_disbursement(:un_id, :k_id, :reg_id, 3, :acc_kt, :reg_id, 2, :acc_dt, :c_id, :manager_id)
  into res_text;

end
