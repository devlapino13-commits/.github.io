CREATE PROCEDURE HP_HRORDERS_TMP(
    REP_REGION_ID D_INTEGER,
    IN_C_ID D_ID,
    IN_FIND_TEXT D_STR_100,
    IN_ORDER_TYPE D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    IN_FIND_TYPE D_INTEGER)
RETURNS (
    ORDER_ID D_ID,
    FIO_LIST D_STR_LONG,
    HRV_CLASS D_STR_CODE)
AS
begin
  if (in_order_type = 0) then
    in_order_type = null;

  if (in_find_text is null) then
    in_find_text = '';

  hrv_class = '';

  if (:in_find_type in (-1, 1)) then
  begin
    if (in_c_id is not null) then
    begin
      -- Поиск приказа по сотруднику
      for select ps_from_ho_id, left(list(distinct c_name_fio), 1020)
          from region_list(:rep_region_id, 1, -1, 1), hr_pstatus, hr_orders, clients
          where ps_from_ho_id = ho_id and
                ps_from_c_id = :in_c_id and
                ps_from_c_id = c_id and
                ho_order_type = coalesce(:in_order_type, ho_order_type) and
                (:in_find_text = '' or ho_order_no containing :in_find_text or ho_text containing :in_find_text or ho_reason containing :in_find_text) and
                ((ho_order_type = 2 and (ps_from_reg_id = id_list or ho_from_reg_id = id_list))
                  or
                  (coalesce(ho_order_type, 0) <> 2 and ps_from_reg_id = id_list))
      group by 1
      into :order_id, :fio_list
      do
        suspend;
    end else
    begin
      -- Поиск приказов за период
      for select ps_from_ho_id, left(list(distinct c_name_fio), 1020)
          from region_list(:rep_region_id, 1, -1, 1), hr_pstatus, hr_orders, clients
          where ho_date between :rep_date_begin and :rep_date_end and
                ps_from_c_id = c_id and
                ps_from_ho_id = ho_id and
                ho_order_type = coalesce(:in_order_type, ho_order_type) and
                (:in_find_text = '' or ho_order_no containing :in_find_text or ho_text containing :in_find_text or ho_reason containing :in_find_text) and
                ((ho_order_type = 2 and (ps_from_reg_id + 0 = id_list or ho_from_reg_id + 0 = id_list))
                  or
                  (coalesce(ho_order_type, 0) <> 2 and ps_from_reg_id + 0 = id_list))
      group by 1
      into :order_id, :fio_list
      do
        suspend;
    end
  end

  if (:in_find_type in (-1, 0)) then
  begin
    if (in_c_id is not null) then
    begin
      -- Поиск приказа по сотруднику
      for select hrv_from_ho_id, left(list(distinct c_name_fio), 1020), min(hrv_class)
          from region_list(:rep_region_id, 1, -1, 1), hr_voctravel, hr_orders, clients
          where hrv_from_ho_id = ho_id and
                hrv_from_c_id = :in_c_id and
                hrv_from_c_id = c_id and
                ho_order_type = coalesce(:in_order_type, ho_order_type) and
                (:in_find_text = '' or ho_order_no containing :in_find_text or ho_text containing :in_find_text or ho_reason containing :in_find_text) and
                ((ho_order_type = 2 and (hrv_from_reg_id = id_list or ho_from_reg_id = id_list))
                  or
                  (coalesce(ho_order_type, 0) <> 2 and hrv_from_reg_id = id_list))
      group by 1
      into :order_id, :fio_list, hrv_class
      do
        suspend;
    end else
    begin
      -- Поиск приказов за период
      for select hrv_from_ho_id, left(list(distinct c_name_fio), 1020), min(hrv_class)
          from region_list(:rep_region_id, 1, -1, 1), hr_voctravel, hr_orders, clients
          where ho_date between :rep_date_begin and :rep_date_end and
                hrv_from_c_id = c_id and
                hrv_from_ho_id = ho_id and
                ho_order_type = coalesce(:in_order_type, ho_order_type) and
                (:in_find_text = '' or ho_order_no containing :in_find_text or ho_text containing :in_find_text or ho_reason containing :in_find_text) and
                ((ho_order_type = 2 and (hrv_from_reg_id + 0 = id_list or ho_from_reg_id + 0 = id_list))
                  or
                  (coalesce(ho_order_type, 0) <> 2 and hrv_from_reg_id + 0 = id_list))
      group by 1
      into :order_id, :fio_list, hrv_class
      do
        suspend;
    end
  end
end
