RETURNS (
    TOTAL_1 DECIMAL(18,4),

    TOTAL_3 DECIMAL(18,4),
    TOTAL_4 DECIMAL(18,4),
    TOTAL_5 DECIMAL(18,4),

    TOTAL_7 DECIMAL(18,4),

    TOTAL_11 DECIMAL(18,4),
    TOTAL_12 DECIMAL(18,4),
    TOTAL_13 DECIMAL(18,4),
    TOTAL_14 DECIMAL(18,4),
    TOTAL_15 DECIMAL(18,4),

    TOTAL_17 DECIMAL(18,4),
    TOTAL_18 DECIMAL(18,4),

    TOTAL_20 DECIMAL(18,4))

begin
  select sum(data_1), sum(data_2), sum(data_3), sum(data_4), sum(data_5), sum(data_6), sum(data_7), sum(data_8),

       :total_12, :total_13, :total_14, :total_15, :total_16, :total_17, :total_18, :total_19, :total_20;

end
