CREATE PROCEDURE PRINT_SUD_LAWSUIT(
    SMR_ID D_ID)
RETURNS (
    K_KONTRACT_DATE D_DATE,
    K_DISB_DATE D_DATE,
    K_ORIG_KONTRACT_NO D_STR_20,
    K_KOLICH_REPAY D_INTEGER,
    K_PERC_YEAR_RU D_PERCENT,
    K_DATE_STOP_PERC D_DATE,
    K_PERCENT D_PERCENT,
    K_STOP_KONTRACT D_STR_20,
    K_ZAD_OS D_CURRENCY,
    K_ZAD_CALC_PERC D_CURRENCY,
    K_ZAD_OS_PERC D_CURRENCY,
    FORMULA D_CURRENCY,
    TEXT_FIO D_STR_LONG,
    TEXT_FIO_2 D_STR_LONG,
    AMOUNT_APPROVED D_CURRENCY,
    FULL_PAYMENT_DEBT D_STR_LONG,
    DELINQ_DATE D_DATE,
    SUD_NAME D_STR_255,
    SUD_ADDRESS D_STR_255,
    STS_SERIES_NUMBER STR_255,
    STS_ISSUE_DATE STR_255,
    STS_AUTHORITY STR_255,
    STS_PLATE STR_255,
    PTS_SERIES_NUMBER STR_255,
    PTS_VIN STR_255,
    PTS_MODEL STR_255,
    PTS_CHASSIS_NO STR_255,
    PTS_BODY_NO STR_255,
    Z_COST_LIKVID D_CURRENCY,
    CLAIM_DATE D_DATE,
    CLAIM_DATE_END D_DATE,
    RESPONSIBLE_NAME D_STR_255,
    KIND D_SMALLINT,
    IS_ZALOG D_BOOL,
    IS_CASH D_BOOL)
AS
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable pr_code d_str_code;
declare variable responsible_id d_id;
begin
  select smr_from_k_id
  from sud_mainreq smr
  where smr_id = :smr_id
  into k_id;

  select smd_proc_type
  from sud_maindocs
  where smd_id = :smr_id
  into kind;              -- тип заявления. 0 - Судебный приказ,  1 -  Исковое заявление

  select iif(smd_disb_kind = 1, 1, 0)
  from sud_maindocs
  where smd_id = :smr_id
  into is_cash;  -- 0 - на карту, 1 - наличными

  select k_kontract_date, k_kontract_no, k_kolich_repay, k_percent_per_day * 360, k_stop_perc, k_percent, k_disb_date, pr_code
  from kontract, product
  where k_id = :k_id
    and pr_id = k_from_pr_id
  into k_kontract_date, k_kontract_no, k_kolich_repay, k_perc_year_ru, k_date_stop_perc, k_percent, k_disb_date, pr_code;

  k_orig_kontract_no = k_kontract_no;
  select k_kontract_no
  from kontdevides, kontract
  where kd_k_orig = k_id
    and kd_k_new = :k_id
  rows 1
  into k_orig_kontract_no;

  select f_datetostr(date_stop_kontract)
  from hp_find_stop_kontract_date(:k_id)
  into k_stop_kontract;

  select iif(:kind = 1, formula_new, formula_sud_new), text_fio, text_fio_2, amount_approved, full_payment_debt, k_zad_os, k_zad_calc_perc, k_zad_os_perc
  from print_amnistia(:k_id)
  into formula, text_fio, text_fio_2, amount_approved, full_payment_debt, k_zad_os, k_zad_calc_perc, k_zad_os_perc;

  select delinq_date
  from print_client_zadol(:k_id)
  into delinq_date;

  select sag_name, sag_address, smd_respons_sag_id
  from sud_maindocs, sud_agents
  where smd_id = :smr_id
    and sag_id = smd_sud_sag_id
  into sud_name, sud_address, responsible_id;

  responsible_name = (select sag_name from sud_agents where sag_id = :responsible_id);

  is_zalog = 0;
  if (pr_code containing '.zalog') then -- если это займ на покупку авто
  begin
    is_zalog = 1;
    select pts_model, pts_vin, pts_body_no, pts_chassis_no, pts_series_number, sts_plate, sts_series_number, sts_issue_date, sts_authority
    from print_zal_auto(:k_id)
    into pts_model, pts_vin, pts_body_no, pts_chassis_no, pts_series_number, sts_plate, sts_series_number, sts_issue_date, sts_authority;

    select z_cost_likvid
    from print_zalog(:k_id)
    into z_cost_likvid;

    select smd_claim_date
    from sud_maindocs
    where smd_id = :smr_id
    into claim_date;

    if (claim_date is null) then
      exception e_message 'Не заполнена дата отправки претензии';

    claim_date_end = dateadd(day, 30, claim_date);
  end

  suspend;
end
