CREATE PROCEDURE HP_BKI_GET_INFO4REQ_FIO(
    C_ID ID_TYPE,
    FORCED_CYR D_BOOL = 0)
RETURNS (
    LASTNAME D_STR_100,
    FIRSTNAME D_STR_100,
    MIDDLENAME D_STR_100)
AS
declare variable c_name_fio str_100;
declare variable pk_id d_id;
declare variable fio_cyr d_str_255;
declare variable fio_eng d_str_255;
declare variable fio_last_datetime d_datetime;
declare variable fio_doc_date_begin d_date;
declare variable cds_create_datetime d_datetime;

declare function del_twice_space(str d_str_255)
  returns d_str_255
  as
  begin
    str = replace(str, ascii_char(9), ' ');
    str = replace(str, '\', '');
    str = f_replace_all(str, ' -', '-');
    str = f_replace_all(str, '- ', '-');
    str = f_replace_all(str, '  ', ' ');

    return trim(str);
  end

declare function f_my_fix_wrong_lat(in_str d_str_255)
  returns d_str_255
  as
  declare variable res_text d_str_255;
  begin
    res_text = in_str;
    if(res_text similar to '%[а-яА-Я]%') then
    begin
      res_text = replace(res_text collate unicode, 'А' collate unicode, 'A');
      res_text = replace(res_text collate unicode, 'В' collate unicode, 'B');
      res_text = replace(res_text collate unicode, 'С' collate unicode, 'C');
      res_text = replace(res_text collate unicode, 'Е' collate unicode, 'E');
      res_text = replace(res_text collate unicode, 'Г' collate unicode, 'F');
      res_text = replace(res_text collate unicode, 'Н' collate unicode, 'H');
      res_text = replace(res_text collate unicode, 'К' collate unicode, 'K');
      res_text = replace(res_text collate unicode, 'М' collate unicode, 'M');
      res_text = replace(res_text collate unicode, 'О' collate unicode, 'O');
      res_text = replace(res_text collate unicode, 'Р' collate unicode, 'P');
      res_text = replace(res_text collate unicode, 'Т' collate unicode, 'T');
      res_text = replace(res_text collate unicode, 'И' collate unicode, 'U');
      res_text = replace(res_text collate unicode, 'Х' collate unicode, 'X');
      res_text = replace(res_text collate unicode, 'У' collate unicode, 'Y');

      res_text = replace(res_text collate unicode, 'а' collate unicode, 'a');
      res_text = replace(res_text collate unicode, 'с' collate unicode, 'c');
      res_text = replace(res_text collate unicode, 'е' collate unicode, 'e');
      res_text = replace(res_text collate unicode, 'г' collate unicode, 'f');
      res_text = replace(res_text collate unicode, 'н' collate unicode, 'h');
      res_text = replace(res_text collate unicode, 'к' collate unicode, 'k');
      res_text = replace(res_text collate unicode, 'м' collate unicode, 'm');
      res_text = replace(res_text collate unicode, 'п' collate unicode, 'n');
      res_text = replace(res_text collate unicode, 'о' collate unicode, 'o');
      res_text = replace(res_text collate unicode, 'р' collate unicode, 'p');
      res_text = replace(res_text collate unicode, 'и' collate unicode, 'u');
      res_text = replace(res_text collate unicode, 'х' collate unicode, 'x');
      res_text = replace(res_text collate unicode, 'у' collate unicode, 'y');

    end
    return res_text;
  end

begin
  fio_last_datetime = '01.01.1900';

  for select del_twice_space(trim(c.c_name_fio)), del_twice_space(f_my_fix_wrong_lat(trim(c.c_name_eng))), '31.12.2999', '31.12.2999'
    from clients c
    where c_id = :c_id
  union
    select  del_twice_space(trim(f_docrecog_value(cds.cds_from_drl_id, 'client_name_old'))),
            del_twice_space(f_my_fix_wrong_lat(trim(f_docrecog_value(cds.cds_from_drl_id, 'client_name_old_eng')))),
            cds.cds_doc_date_begin,
            cds.cds_create_datetime
    from clientdocs cds
    where cds_from_pk_id = 66 -- Свидетельство о перемене Фамилии, Имени, Отчества
      and cds_from_c_id = :c_id
      and cds_is_delete = 0
  order by 3,4
  into fio_cyr, fio_eng, fio_doc_date_begin, cds_create_datetime
  do
  begin
    if (fio_cyr || fio_eng = '') then
      continue;

    if (fio_cyr = '') then
      fio_cyr = fio_eng;

    if (fio_eng = '') then
      fio_eng = fio_cyr;

    if (datediff(day, fio_last_datetime, fio_doc_date_begin) < 1) then
      continue;

    pk_id = null;

    -- ищем последний паспорт прикрепленный не позднее чем за неделю до документа смены ФИО
    select cds_from_pk_id
    from clientdocs cds
    where cds_from_c_id = :c_id
      and cds_is_main_doc = 1
      and cds_verify_status = 2
      and cds_doc_date_begin < dateadd(day, -7, :fio_doc_date_begin)
    order by cds_create_datetime desc
    rows 1
    into pk_id;

    if (pk_id is null) then  -- если не нашли, то поищем без ограничения в 7 дней
      select cds_from_pk_id
      from clientdocs cds
      where cds_from_c_id = :c_id
        and cds_is_main_doc = 1
        and cds_verify_status = 2
        and cds_doc_date_begin <= :fio_doc_date_begin
      order by cds_create_datetime desc
      rows 1
      into pk_id;

    c_name_fio = iif(pk_id = 12, fio_cyr, fio_eng);

    if (trim(coalesce(c_name_fio, '')) <> '') then
    begin
      select name_last, name_first, name_middle
      from hp_fio_clear_ru(:c_name_fio)
      into :lastname, :firstname, :middlename;

      if (middlename = '') then middlename = null;

      suspend;
    end

    fio_last_datetime = fio_doc_date_begin;

    if (pk_id <> 12 and coalesce(forced_cyr, 0) = 1) then  -- для иностранцев запрос на обоеих раскладках
    begin
      c_name_fio = fio_cyr;
  
      if (trim(coalesce(c_name_fio, '')) <> '') then
      begin
        select name_last, name_first, name_middle
        from hp_fio_clear_ru(:c_name_fio)
        into :lastname, :firstname, :middlename;

        if (middlename = '') then middlename = null;

        suspend;
      end
    end
  end
end
