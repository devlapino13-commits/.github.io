CREATE PROCEDURE MILLICARD_CHECKCREDIT_BY_HAND(
    ESM_ID ID_TYPE,
    UN_ID ID_TYPE)
RETURNS (
    RESULT_STR STR_255)
AS
declare variable k_id id_type;
begin
-- UN_ID будет использоваться для логирования
-- если найдена задача, по которой операция не прошла

  if (esm_id is null) then
  begin
    result_str = 'Задача не определена';
    suspend;
    exit;
  end

  if (not exists (select * from extsysmain em where em.esm_id = :esm_id)) then
  begin
    result_str = 'Задача не определена';
    suspend;
    exit;
  end

  if (un_id is null) then
  begin
    result_str = 'Пользователь не определен';
    suspend;
    exit;
  end

  insert into millicard_checkcredit_log (mcl_from_esm_id, mcl_un_id)
  values (:esm_id, :un_id);

  select em.esm_from_k_id
  from extsysmain em
  where em.esm_project in ('millicard_disb', 'millicard_biz_disb') and
        em.esm_action = 'checkCredit' and
        em.esm_status = -1 and
        em.esm_id = :esm_id
  into :k_id;

  if (k_id is null) then
  begin
    result_str = 'Задача на подтверждение выдачи не найдена (Поиск среди задач на '||f_datetostr(current_date)||')';
    suspend;
    exit;
  end

-- проверяю нет ли уже успешной задачи по этому же кредиту
  if (exists(select em.esm_id, em.esm_from_k_id
             from extsysmain em
             where em.esm_project in ('millicard_disb', 'millicard_biz_disb') and
                   em.esm_action = 'checkCredit' and
                   em.esm_status = 1 and
                   em.esm_from_k_id = :k_id)) then
  begin
    result_str = 'По кредиту уже перечислили деньги';
    suspend;
    exit;
  end

-- update задачи, которая ранее была обработана с ошибкой
  update extsysmain em
  set em.esm_status = 0
  where em.esm_id = :esm_id;

  result_str = 'Задача на подтверждение выдачи была успешно заново отправлена в Millicard';

  suspend;
end
