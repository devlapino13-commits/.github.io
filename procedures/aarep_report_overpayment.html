CREATE PROCEDURE AAREP_REPORT_OVERPAYMENT(
    REP_DATE_END DATE)
RETURNS (
    KC_ID ID_TYPE,
    SALDO_PEREPLAT CURR_TYPE,
    KONTRACT_NO STR_20,
    C_NAME_FIO STR_100,
    CLOSE_DATE D_DATE,
    REPAY_FIRST_PEREPLAT D_DATE,
    REPAY_LAST_PEREPLAT D_DATE,
    DATE_LAW_REPAY D_DATE,
    DATE_LAW_CALC D_DATE,
    KOL_ACTIV_KRED D_INTEGER,
    BAL_ACTIV_OS CURR_TYPE,
    BAL_ACTIV_OS_DEL CURR_TYPE,
    BAL_ACTIV_PERC CURR_TYPE,
    BAL_ACTIV_PERC_DEL CURR_TYPE,
    KOL_KROFF_KRED D_INTEGER,
    BAL_KROFF_OS CURR_TYPE,
    BAL_KROFF_PERC CURR_TYPE,
    C_ID_STR STR_20,
    PR_NAME VARCHAR(50),
    K_DISB_DATE D_DATE,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    CLIENT_TYPE D_STR_20)
AS
declare variable k_id id_type;
declare variable rd_id d_integer;
declare variable c_id decimal(18,0);
declare variable pr_code d_str_100;
begin
  select rd_id
  from refdata
  where rd_code = 'OperExtraKind.LawClientExpense'
  into rd_id;

  for select ok_from_kc_id, sum(ok_os), min(ok_ol_date), max(ok_ol_date)
      from kn_operkred
      where ok_ol_is_active = 1 and
            ok_type = 9 and
            ok_ol_date <= :rep_date_end
            -- and ok_from_kc_id = 28554
      group by 1
      having sum(ok_os) < 0
      --rows 100
      into kc_id, saldo_pereplat, repay_first_pereplat, repay_last_pereplat
  do
  begin
    select kc_from_k_id, kc_from_c_id
    from knclient
    where kc_id = :kc_id

    into k_id, c_id;

    c_id_str = 'n' || c_id;

    select k.k_kontract_no, k.k_fact_closed_date, k.k_disb_date, pr.pr_name, k.k_kolich_client_approved, pr.pr_code
    from kontract k, product pr
    where k.k_id = :k_id and
          pr.pr_id = k.k_from_pr_id
    into kontract_no, close_date, k_disb_date, pr_name, k_kolich_client_approved, pr_code;

    client_type = 'ФЛ';
    if (pr_code containing '.selfemployed.') then
      client_type = 'СЗ';

    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;

    date_law_calc = null;
    date_law_repay = null;
    select min(iif(ox_amount_plus > 0, ox_ol_date, null)), min(iif(ox_amount_minus > 0, ox_ol_date, null))
    from kn_operextra
    where ox_from_kc_id = :kc_id and
          ox_ol_is_active = 1 and
          ox_from_rd_id = :rd_id
    into date_law_calc, date_law_repay;

    kol_activ_kred = null;
    bal_activ_os = null;
    bal_activ_os_del = null;

    select count(distinct kc_id), sum(oks_balance), sum(iif(oks_delinq_days > 0, oks_balance, 0))
    from kn_oksummary, knclient
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_close_state = 0 and
          oks_from_kc_id = kc_id and
          kc_from_c_id = :c_id
    into kol_activ_kred, bal_activ_os, bal_activ_os_del;

    --столбец 14 и 15
    bal_activ_perc = null;
    bal_activ_perc_del = null;

    select sum(oks_bal_perc) bal_activ_perc, sum(iif(oks_delinq_days > 0, oks_bal_zad_perc, 0)) bal_activ_perc_del
    from kn_oksummary, knclient
    where :rep_date_end between oks_date_begin and oks_date_end and
          (oks_close_state in (0, 1) or
          oks_bal_law <> 0 or
          oks_reserv_law <> 0) and
          oks_from_kc_id = kc_id and
          kc_from_c_id = :c_id
    into bal_activ_perc, bal_activ_perc_del;

    if (kol_activ_kred = 0) then
      kol_activ_kred = null;
    kol_kroff_kred = null;
    bal_kroff_os = null;
    bal_kroff_perc = null;
    select count(distinct oko_from_k_id), sum(oko_os + oko_sh_os), sum(oko_perc + oko_sh_perc)
    from operkroff
    where oko_ol_is_active = 1 and
          oko_ol_date <= :rep_date_end and
          oko_from_c_id = :c_id having sum(oko_os + oko_sh_os) > 0 or
          sum(oko_perc + oko_sh_perc) > 0
    into kol_kroff_kred, bal_kroff_os, bal_kroff_perc;

    suspend;
  end
end
