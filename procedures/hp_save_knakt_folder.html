CREATE PROCEDURE HP_SAVE_KNAKT_FOLDER(
    REP_REGION_ID INTEGER,
    REP_DATE DATE,
    REP_MEMO VARCHAR(255),
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT VARCHAR(255),
    KA_ID DECIMAL(18,0))
AS
declare variable ka_akt_no integer;
declare variable kf_id id_type;
declare variable lo_id id_type;
declare variable lo_id_prior id_type;
declare variable kal_new_no integer;
declare variable kal_new_officer_code varchar(20);
declare variable kal_new_kf_name varchar(100);
declare variable kf_count integer;
declare variable c_name_fio varchar(100);
begin
  res_text = '';
  ka_id = 0;

  for select distinct tkf_lo_id
      from temp_knakt_folder
      where tkf_lo_id is not null
      into lo_id
  do
  begin
    kal_new_officer_code = null;
    select ke_code
    from krexpert
    where :rep_date between ke_date and ke_date_end and
          ke_from_reg_id = :rep_region_id and
          ke_from_p_id = :lo_id
    into kal_new_officer_code;
    if (kal_new_officer_code is null) then
    begin
      select c_name_fio
      from clients
      where c_id = :lo_id
      into :c_name_fio;

      if (res_text = '') then
        res_text = 'Отсутствует код СВЗ: ' || c_name_fio;
      else
        res_text = res_text || ', ' || c_name_fio;
    end
  end
  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  ka_id = (select kv
           from get_kv(:rep_region_id, 7));
  ka_akt_no = coalesce((select max(cast(ka_akt_no as integer))
                        from knakt
                        where ka_from_reg_id = :rep_region_id), 0) + 1;
-- создаем запись в табилце KNAKT
  insert into knakt (ka_id, ka_date, ka_from_reg_id, ka_akt_no, ka_memo, is_delete, ka_from_ol_id)
  values (:ka_id, :rep_date, :rep_region_id, :ka_akt_no, :rep_memo, 0, null);

  lo_id_prior = 0;
  res_text = '';
  for select tkf_kf_id, tkf_lo_id
      from temp_knakt_folder
      where tkf_lo_id is not null
      order by tkf_lo_id
      into kf_id, lo_id
  do
  begin
    if (lo_id_prior <> lo_id) then
    begin
      lo_id_prior = lo_id;

      kal_new_no = null;
      kal_new_officer_code = null;
      kal_new_no = 0;

--    Если у СВЗ нету ни одной папки то код СВЗ вытаскиваем из таблицы KREXPERT
      select ke_code
      from krexpert
      where :rep_date between ke_date and ke_date_end and
            ke_from_reg_id = :rep_region_id and
            ke_from_p_id = :lo_id
      into kal_new_officer_code;

--    Находим последнию папку данного СВЗ для продолжения нумерации
      select kal_new_no
      from knaktlist
      where knaktlist.is_delete = 0 and
            kal_date_end >= '31.12.2999' and
            kal_new_officer_code = :kal_new_officer_code and
            kal_new_reg_id = :rep_region_id
      order by kal_new_officer_code desc, kal_new_no desc
      rows 1
      into kal_new_no;
      if (kal_new_no is null) then
        kal_new_no = 0;
    end
    kal_new_no = kal_new_no + 1;
    kal_new_kf_name = kal_new_officer_code || '-' || kal_new_no;
--  Сохраняем передачу
    insert into knaktlist (kal_from_ka_id, kal_number, kal_from_kf_id, kal_from_k_id, kal_date, kal_date_end,
                           kal_new_kf_officer, kal_new_kf_name, is_delete, kal_new_reg_id, kal_new_no,
                           kal_new_officer_code)
    values (:ka_id, -1, :kf_id, null, :rep_date, '31.12.9999', :lo_id, :kal_new_kf_name, 0, :rep_region_id, :kal_new_no,
            :kal_new_officer_code);
  end

-- Формируем push-уведомления
  for select tkf_lo_id, count(*)
      from temp_knakt_folder
      group by tkf_lo_id
      into :lo_id, kf_count
  do
  begin
    execute procedure hp_send_push_to_staff(:lo_id, 'Прием-передача папок',
        'Вам передано ' || kf_count || ' кредитных папок на обслуживание. Список можете увидеть в мобильном приложении через Кредиты - Принятые активные папки.',
        :un_id, 'Knaktlist_auto_transfer', 7);
  end
  suspend;
end
