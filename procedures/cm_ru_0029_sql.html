CREATE PROCEDURE CM_RU_0029_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable reg_id integer;
declare variable pr_is_employer integer;
declare variable pr_id integer;
declare variable K_KOMMIS_PERC perc_type;
declare variable prl_admin_perc_min perc_type;
declare variable prl_admin_perc_max perc_type;
declare variable k_kolich_repay integer;
declare variable k_disb_date date;
begin
  for
  select k_disb_date,
         k_amount_approved,
         pr_is_employer,
         pr_id,
         K_KOMMIS_PERC,
         k_kolich_repay,
         k_disb_date
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  into kf_disb_date, KC_AMOUNT, pr_is_employer, pr_id, K_KOMMIS_PERC, k_kolich_repay, k_disb_date
  do
  begin
    prl_admin_perc_min = null;
    prl_admin_perc_max = null;
    select min(prl_admin_perc_min), max(prl_admin_perc_max)
    from PRODLOGIC
    where prl_from_pr_id = :pr_id
    and :k_disb_date between prl_date_begin and prl_date_end
    and :k_kolich_repay between prl_repay_min and prl_repay_max
    into prl_admin_perc_min, prl_admin_perc_max;
    if (prl_admin_perc_min is null) then prl_admin_perc_min = 0;
    if (prl_admin_perc_max is null) then prl_admin_perc_max = 100;
    if (K_KOMMIS_PERC is null) then K_KOMMIS_PERC = 0;
    if (not(K_KOMMIS_PERC between prl_admin_perc_min and prl_admin_perc_max)) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Доп.% не соответствует к настройкам кр. продукта';
      suspend;
    end
  end
end
