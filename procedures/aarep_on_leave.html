CREATE PROCEDURE AAREP_ON_LEAVE(
    REP_REGION_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    C_ID ID_TYPE,
    C_NAME_FIO VARCHAR(100),
    JL_NAME VARCHAR(100),
    HO_DATE DATE,
    HO_ORDER_NO VARCHAR(20),
    HRV_DATE_BEGIN DATE,
    HRV_DATE_END DATE,
    HRV_DAYS INTEGER,
    SAWT_NAME VARCHAR(100),
    REG_NAME VARCHAR(50))
AS
declare variable date_temp date;
declare variable diff_days integer;
begin
  for select distinct c_name_fio, sw.sawt_name, hrv.hrv_date_begin, hrv.hrv_date_end_fact, hrv.hrv_days, --added by Aleksei 18/03/19
                      ho_date, ho_order_no, jl_name, re.reg_name,cl.c_id
      from hr_orders ho, sa_worktype sw, hr_voctravel hrv, clients cl, hr_pstatus hp, joblist, region re, region_list(:rep_region_id, 1, -1, 1)
      where ho.ho_status = 1 and
            ho.ho_order_type = sw.sawt_id and
            sw.sawt_base_type in (4, 5) and
            ho.ho_order_type <> 20 and
            hrv.hrv_date_begin <= :rep_date_end and
            hrv.hrv_date_end_fact >= :rep_date_begin and
            ho.ho_id = hrv.hrv_from_ho_id and
            hrv.hrv_status = 1 and
            hrv.hrv_from_c_id = c_id and
            hrv.hrv_from_c_id = hp.ps_from_c_id and
            hp.ps_status = 1 and
            hp.ps_is_working = 1 and
            :rep_date_begin between hp.ps_date_begin and hp.ps_date_end and
            hp.ps_from_jl_id = jl_id and
            hp.ps_from_reg_id = id_list and
            hp.ps_from_reg_id = reg_unicode
      into c_name_fio, sawt_name, hrv_date_begin, hrv_date_end, hrv_days, --added by Aleksei 18/03/19
           ho_date, ho_order_no, jl_name, reg_name,c_id
  do
  begin ----added by Aleksei on 18/03/2019
    date_temp = :hrv_date_begin;
    diff_days = 0;

    while (date_temp <= :hrv_date_end) do
    begin
      if (exists(select *
                 from sa_calendar scd
                 where :date_temp = cast(cast(scd.scal_day as varchar(2)) || '.' || cast(scd.scal_month as varchar(2)) || '.' || cast(extract(year from :ho_date) as varchar(4)) as date))) then
        diff_days = diff_days + 1;

      date_temp = date_temp + 1;
    end

    :hrv_days = :hrv_days - diff_days;

    suspend;
  end
end
