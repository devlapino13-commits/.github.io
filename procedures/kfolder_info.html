CREATE PROCEDURE KFOLDER_INFO(
    ID DECIMAL(18,0),
    DD DATE)
RETURNS (
    KFI_ID DECIMAL(18,0),
    KFI_OFFICER_ID DECIMAL(18,0),
    KFI_OFFICER_NAME VARCHAR(100),
    KFI_NAME VARCHAR(100),
    KFI_REG_ID INTEGER,
    KFI_REG_NAME VARCHAR(100),
    KFI_CODE VARCHAR(20),
    KFI_NO INTEGER,
    KFI_DATE DATE,
    KFI_REG_CODE VARCHAR(10),
    KFI_IS_YUR INTEGER,
    KFI_IS_YUR_TEXT VARCHAR(20))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  /* procedure text */
  select kal_from_kf_id, kal_new_kf_officer, c_name_fio, kal_new_kf_name, kal_new_reg_id, reg_name,
         kal_new_officer_code, kal_new_no, reg_code, kf_is_yur
  from knaktlist, clients, region, kfolder
  where kal_new_kf_officer = c_id and
        kal_new_reg_id = reg_unicode and
        kal_from_kf_id = :id and
        kal_date <= :dd and
        kal_date_end >= :dd and
        kf_id = kal_from_kf_id and
        knaktlist.is_delete = 0
  into :kfi_id, :kfi_officer_id, :kfi_officer_name, :kfi_name, :kfi_reg_id, :kfi_reg_name, :kfi_code, :kfi_no,
       :kfi_reg_code, :kfi_is_yur;
  if (kfi_is_yur = 0) then
    kfi_is_yur_text = 'Физическое лицо';
  else
    kfi_is_yur_text = 'Юридическое лицо';

  select kal_date
  from knaktlist
  where is_delete = 0 and
        kal_from_ka_id = 0 and
        kal_from_kf_id = :id
  into :kfi_date;

  suspend;
end
