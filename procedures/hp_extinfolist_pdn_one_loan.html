declare procedure get_exept_data(kc_id d_id, rep_date d_date)
returns (
  k_kontract_no d_str_code,

  suspend;
end

declare procedure get_period_month(
    creddate d_date,
    credenddate d_date,
    calc_date d_date)
returns (
    calc_koef_period d_currency,
    calc_period_month d_currency)
as
begin
  calc_koef_period = 1;
  -- Поправ коэф ЧБП/30 (для займов до 30 дней)
  if (datediff(day, creddate, credenddate) <= 30 and datediff(day, creddate, credenddate) > 0) then
  begin
    -- с 01.10.2019 действует новый вариант расчета ЧБП/30, письмо от Елены в январе 2020 г.
    if (calc_date < '01.10.2019')
    then calc_koef_period = (cast(365.0 as curr_type) / cast(datediff(day, creddate, credenddate) as curr_type) / cast(30.0 as curr_type));
    else calc_koef_period = cast(datediff(day, creddate, credenddate) as curr_type) / cast(30.0 as curr_type);
  end

  calc_period_month = 0;
  if (calc_date <= credenddate) then
    calc_period_month = (extract(year from credenddate) - extract(year from calc_date)) * 12 +
                        extract(month from credenddate) - extract(month from calc_date) +
                        iif(extract(day from credenddate) > extract(day from calc_date), 1, 0);

  suspend;
end

  cred_owner_type_id = null;
  debtcurr_opsum = 0;
  debtcurr_percentsum = 0;
  debtcurr_othersum = 0;
  debtoverdue_opsum = 0;
  debtoverdue_percentsum = 0;
  debtoverdue_othersum = 0;
  fullcost_sum = 0;
  fullcost_date = null;
  future_percent_sum = 0;
  rec_no = 2000;
  average_payment_date = null;
  exclusion_reason = null;
  debtoverdue_date = null;
  gracedate = null;
  graceenddate = null;
  minsumpaycc = null;

  for select kpa_uid, kpa_calc_date, kpa_sum, kpa_currency, kpa_currency, kpa_id
  from knclient_pdn_amplist
  where kpa_from_kcp_id = :kcp_id
    and kpa_is_actual = 1
    and json.contains(:uids_jso, kpa_uid) = 0
  into creduid, amp_update, amp_sum, amp_currency, credcurrency, kpa_id
  do
  begin
    debtoverdue_date = json.obj_get_val(nbki_active_jso, creduid);
    exclusion_reason = null;
    calc_month_repay_final = amp_sum;

    if (kcp_kind = 0 and need_exclude(new_kc_id, creduid)) then
      exclusion_reason = 1;
    else if (kcp_kind = 0 and amp_update < dateadd(month, -12, calc_date))
    then
      exclusion_reason = 2;
    else
      sum_month_repay_final = sum_month_repay_final + calc_month_repay_final;



    rec_no = rec_no + 1;

    suspend;
  end 

  if (kcp_kind = 0) then
  begin
    select score_level, card_id, score
    from hp_bki_scoring_check(:jso_full, :calc_date)
    into score_level, card_id, score;
    
    if ((card_id + score) is not null) then
      update or insert into bki_scoring (bsc_from_kc_id, bsc_card_id, bsc_score, bsc_scoring_level)
      values (:new_kc_id, :card_id, :score, :score_level)
      matching (bsc_from_kc_id);
  end

  -- итоговая строка
  rec_no = -1;
  calc_month_repay_final = sum_month_repay_final;
  creduid = null;
  exclusion_reason = null;

  kc_stat_avg_income = 0.00;
  final_pdn = 0;
  average_payment_date = calc_date;

  select res_last_period, res_income_avg, res_income_01, res_income_02, res_income_03, res_income_04, res_income_05,
         res_income_06, res_income_07, res_income_08, res_income_09, res_income_10, res_income_11, res_income_12,
         str_id, adl_id, res_address
  from hp_extinfolist_pdn_one_income(:new_kc_id, :calc_date)
  into :res_last_income_period, :kc_stat_avg_income, :res_income_01, :res_income_02, :res_income_03, :res_income_04, :res_income_05,
       :res_income_06, :res_income_07, :res_income_08, :res_income_09, :res_income_10, :res_income_11, :res_income_12,
       str_id, adl_id, res_address;

  if (kc_stat_avg_income <= 0) then  -- изменено RFCRU-2285
  begin
    select k_kontract_no, uname, office
    from get_exept_data(:new_kc_id, :calc_date)
    into k_kontract_no, uname, office;

    msg_subject = 'Указан некорректный адрес регистрации клиента.';
    param_str = ' № договора = ' || k_kontract_no || 
                ', Офис = ' || office || 
                ', Пользователь = ' || uname || 
                ', ID займа = ' || new_kc_id || 
                ', Клиент = ' || coalesce(orig_c_name_fio, '<NULL>') || 
                ', Дата = ' || f_datetostr(calc_date) ||
                ', Основание ввода адреса = ' || trim(iif(cml_id is not null, 'КК', 'Выдача')) ||
                ', Адрес = ' || coalesce(res_address, '<NULL>');

    in autonomous transaction do
      insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id, msg_command_kind)
      values (0, 's.kravtsov@mbfplanet.com;e.grabovskaya@molbulak.com;zara@molbulak.com;y.kashtanova@molbulak.com', :msg_subject, :param_str, 7, 1000000000001, '');

    exception e_message 'Внимание! ПДН еще не рассчитан. Требуется 1-1.5 часа в рабочие дни. Не предпринимайте действий с заявкой/договором, пока не получите сообщение о рассчитанном ПДН от Опер. отдела';
  end

  kc_anketa_income = 0;
  kc_anketa_income_calc = null;
  is_documented = null;

  select monthly_income
  from hp_knclient_anketa_income(:new_kc_id, :calc_date)
  into kc_anketa_income;

  -- RFCRU-2288 (введена переменная is_documented)
  select maci_is_documented
  from knclient, kontract, miniappclient, miniappclient_income
  where kc_id = :new_kc_id
    and k_id = kc_from_k_id
    and mac_from_ma_id = k_from_ma_app_id
    and mac_from_c_id = kc_from_c_id
    and maci_from_mac_id = mac_id
  into is_documented;

  c_name_fio = orig_c_name_fio;
  kc_documented_income = 0;

  if (kcp_kind = 0 and f_get_const('documented_income_active') = 1 and is_documented = 1) then
  begin
    select ki_verify_income
    from knclient, kk_income
    where kc_id = :new_kc_id
      and ki_from_k_id = kc_from_k_id
      and ki_from_c_id = kc_from_c_id
    into kc_documented_income;
  end

  -- RFCRU-2301 -- Для продуктов "Самозанятый", "Бизнес" ПДН не рассчитывается
  if (new_pr_code containing '.selfemployed.'
    or new_pr_code containing '.business.')
  then
  begin
    suspend;
    exit;
  end 

  -- если расчет задним числом или расчет по рефинасам 60 дней, то доход остается первоначальный
  if ((calc_date < current_date) or kcp_kind = 2) then
  begin
   if (kcp_kind = 2) then
      select kcp_id
      from knclient_pdn
      where kc_id = :new_kc_id
        and kcp_kind = 0
      into kcp_id;

    select kc_pdn_koef
    from knclient_pdn
    where kcp_id = :kcp_id
    into final_pdn;

    if (coalesce(final_pdn, 0) <> 0) then
    begin
      select kc_stat_last_period, kc_stat_avg_income, kc_anketa_income, kc_documented_income, kc_final_income, kcp_income_discount_perc
      from knclient_pdn
      where kcp_id = :kcp_id
      into res_last_income_period, kc_stat_avg_income, kc_anketa_income, kc_documented_income, kc_final_income, income_discount_perc;
  
      final_pdn = round(calc_month_repay_final / kc_final_income * 100.00, 1);
      suspend;
      exit;
    end
  end

  -- 1. Если есть документально утвержденный доход, то считаем по нему
  if (coalesce(kc_documented_income, 0) > 0) then
  begin
    kc_final_income = kc_documented_income;
    final_pdn = round(calc_month_repay_final / kc_final_income * 100.00, 1);
    suspend;
    exit;
  end

  -- 2. Считаем по минимальной сумме от анкетного дохода и среднего дохода по Росстату
  if (kc_anketa_income = 0) then
    kc_final_income = kc_stat_avg_income;
  else
    kc_final_income = minvalue(kc_anketa_income, kc_stat_avg_income);

  ---RFCRU-5664  -- после 01.07.2026 можно удалить условие, если не будет пересчетов "задним числом"
  if (k_disb_date >= '01.07.2026') then
    kc_final_income = kc_final_income * 0.90;

  final_pdn = round(calc_month_repay_final / kc_final_income * 100.00, 1);

  suspend;
end
