CREATE PROCEDURE HP_UNIQKEYS_UPDATE(
    KIND INTEGER,
    TABLE_NAME VARCHAR(100),
    OLD_VALUE DECIMAL(18,0),
    NEW_VALUE DECIMAL(18,0))
AS
declare variable t_id integer;
begin
  t_id = f_objkind(table_name, 'id');

  if (t_id is not null) then
  begin
    if (kind = 2) then -- After Update
    begin
      if (coalesce(old_value, -777) <> coalesce(new_value, -777)) then
      begin
        update objectprops
        set opr_object_id = :new_value
        where opr_object_kind = :t_id and
              opr_object_id = :old_value;

        update operextra
        set ox_object_id = :new_value
        where ox_object_kind = :t_id and
              ox_object_id = :old_value;

        update objectprops
        set opr_subject_id = :new_value
        where opr_subject_kind = :t_id and
              opr_subject_id = :old_value;

        update contverify
        set cv_object_id = :new_value
        where cv_object_kind = :t_id and
              cv_object_id = :old_value;
      end
    end
    if (kind = 3) then -- After Delete
    begin
      if (exists(select *
                 from objectprops
                 where (opr_object_kind = :t_id and
                       opr_object_id = :old_value) or
                       (opr_subject_kind = :t_id and
                       opr_subject_id = :old_value))) then
        exception e_default 'Foreign key error. Values [' || :t_id || ';' || :old_value || '] used in ObjectProps';
    end
  end
end
