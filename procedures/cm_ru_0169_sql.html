CREATE PROCEDURE CM_RU_0169_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable pr_code varchar(100);
declare variable c_id id_type;
declare variable kc_balance curr_type;
declare variable c_curr_status integer;
declare variable kc_balance_tmp curr_type;
declare variable delinq_days d_integer;
begin
  for select kc_from_c_id, kc_amount_approved, pr.pr_code, kont.k_disb_date
      from knclient kncl
      inner join kontract kont on kont.k_id = kncl.kc_from_k_id
      inner join product pr on pr.pr_id = kont.k_from_pr_id
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into c_id, kc_amount, pr_code, kf_disb_date
  do
  begin
    select c_name_fio, c_pass_no
    from clients
    where c_id = :c_id
    into res_info_text, kf_client_pass_no;
    -- если у клиента нет активного займа по Выгодная сторона, тогда не проверяем
    if (not exists(select *
                   from kontstate ks
                   inner join kontract kont on kont.k_id = ks.ks_from_k_id
                   inner join product pr on pr.pr_id = kont.k_from_pr_id
                   inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
                   where ks.ks_ol_is_delete = 0 and
                         ks.ks_state = 1 and
                         :kf_disb_date between ks.ks_date_begin and ks.ks_date_end and
                         kncl.kc_amount_approved > 0 and
                         kncl.kc_from_c_id = :c_id and
                         pr.pr_code like '%.float_percent.%')) then
      exit;
    if (not((pr_code like '%.blago.%') or (pr_code like '%.uspeh.%') or (pr_code like '%.express.%'))) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Клиент может получить второй заем по основному продукту: «Благо» или «Успех» при наличии «Выгодная сторона»';
      suspend;
      exit;
    end
    -- определяем статус клиента
    select c_curr_status
    from hp_get_client_status_str(:c_id, :kf_disb_date)
    into c_curr_status;
    --  вычисляем всех активных кредитов данного клиента
    kc_balance = null;
    kc_balance = 0;
    for select --sum(cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance)
:kf_disb_date - oks.oks_delinq_date, cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance
        from knclient, oksummary oks, kontract
        where kc_from_c_id = :c_id and
              kc_amount_approved > 0 and
              kc_from_k_id = oks_from_k_id and
              kc_from_k_id = k_id and
              :kf_disb_date between oks_date_begin and oks_date_end and
              --oks_balance > 0 and
              oks_close_state = 0 and
              k_id <> :k_id
        into :delinq_days, :kc_balance_tmp
    do
    begin
      kc_balance_tmp = coalesce(kc_balance_tmp, 0);
      kc_balance = kc_balance + kc_balance_tmp;
      if (delinq_days > 0) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Есть активный просроченный заем в "М Булак", новый заем Вы получить не можете';
        suspend;
      end
    end
    kc_balance = coalesce(kc_balance, 0);
    if (c_curr_status = 0 and
        kc_balance + kc_amount > 30000.00) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Клиента без статуса, баланс активных кредитов [' || :kc_balance || '] + сумма нового кредита [' || :kc_amount || ']  не должно превышать 30 тыс.';
      suspend;
    end
    if (c_curr_status = 1 and
        kc_balance + kc_amount > 45000.00) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Статус клиента 1 звезды, баланс активных кредитов [' || :kc_balance || '] + сумма нового кредита [' || :kc_amount || '] не должно превышать 45 тыс.';
      suspend;
    end
    if (c_curr_status = 2 and
        kc_balance + kc_amount > 60000.00) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Статус клиента 2 звезды, баланс активных кредитов [' || :kc_balance || ' + сумма нового кредита [' || :kc_amount || '] не должно превышать 60 тыс.';
      suspend;
    end
    if (c_curr_status = 3 and
        kc_balance + kc_amount > 80000.00) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Статус клиента 3 звезды, баланс активных кредитов [' || :kc_balance || '] + сумма нового кредита [' || :kc_amount || '] не должно превышать 80 тыс.';
      suspend;
    end
    if (c_curr_status >= 4 and
        kc_balance + kc_amount > 90000.00) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Статус клиента 4 и более звезд, баланс активных кредитов [' || :kc_balance || '] + сумма нового кредита [' || :kc_amount || '] не должно превышать 90 тыс.';
      suspend;
    end
  end
end
