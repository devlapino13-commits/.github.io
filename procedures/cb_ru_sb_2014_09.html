CREATE PROCEDURE CB_RU_SB_2014_09(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    BALANCE CURR_TYPE,
    DELINQ_BALANCE7 CURR_TYPE,
    DELINQ_BALANCE30 CURR_TYPE,
    KOL_CLI_CTEG_1_2 INTEGER,
    PAR7 CURR_TYPE,
    PAR30 CURR_TYPE,
    BONUS_PERCENT CURR_TYPE,
    KOL_SVZ INTEGER,
    BONUS_LO CURR_TYPE,
    AVG_BONUS_LO CURR_TYPE,
    BONUS_KOL_CLI CURR_TYPE,
    BALANCE_LAST CURR_TYPE,
    DELINQ_BALANCE7_LAST CURR_TYPE,
    PAR7_LAST CURR_TYPE,
    SNIJEN_PAR7 CURR_TYPE,
    BONUS_SNIJEN_PAR CURR_TYPE,
    SHTRAF_SNIJEN_PAR CURR_TYPE,
    BONUS_END CURR_TYPE,
    KOL_SB INTEGER)
AS
declare variable reg_id integer;
declare variable rep_date_begin date;
begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end);

-- во временную таблицу сохраняет бонусы СВЗ
  insert into temp_bonusru_lo (bl_personal_id, bl_region_id, bl_bonus_total)
  select cb_personal_id, aa_curr_region_id, bonus_end
  from cb_ru_lo_2014_09(:rep_region_id, :rep_date_end)
  where cb_is_working = 1;

  for select ac_region_id, sum(ac_cli_balance), sum(iif(ac_delinq_day > 7, ac_cli_balance, 0)),
             sum(iif(ac_delinq_day > 30, ac_cli_balance, 0)), sum(iif(ac_is_client = 1 and not exists(select *
                                                                                                      from sb_akt, sb_list
                                                                                                      where sba_is_delete = 0 and
                                                                                                            sba_from_sbl_id <> 0 and
                                                                                                            sba_status = 2 and
                                                                                                            :rep_date_end between sba_date_begin and sba_date_end and
                                                                                                            sba_from_k_id = ac_kontract_id and
                                                                                                            sba_from_sbl_id = sbl_id and
                                                                                                            sbl_type = 1), 1, 0))
      from aarep_cli_active_list(:rep_date_end)
      where ac_delinq_day < 181 and -- добавлено для того чтобы у кого просрочка более 180 дней то их на категория не проверять
            (select aa_category_id
             from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
      group by ac_region_id
      into reg_id, balance, delinq_balance7, delinq_balance30, kol_cli_cteg_1_2
  do
  begin
    reg_name = (select reg_code || ' ' || reg_name
                from region
                where reg_unicode = :reg_id);
--  Расчет кол-во СВЗ
    kol_svz = null;
    select count(*)
    from hr_pstatus, joblist, jobkind
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          ps_status = 1 and
          ps_from_reg_id = :reg_id and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
          coalesce((select sawt_id
                    from hr_voctravel, hr_orders, sa_worktype
                    where hrv_from_ho_id = ho_id and
                          ho_order_type = sawt_id and
                          hrv_from_c_id = ps_from_c_id and
                          :rep_date_end between hrv_date_begin and hrv_date_end_fact
                    order by hrv_date_begin desc
                    rows 1), 0) not in (16, 17, 18) -- Декретников не включает
    into kol_svz;
    if (kol_svz is null) then
      kol_svz = 0;

    if (delinq_balance7 = 0) then
      par7 = 0;
    else
      par7 = delinq_balance7 / balance * 100;
    if (delinq_balance30 = 0) then
      par30 = 0;
    else
      par30 = delinq_balance30 / balance * 100;

    bonus_percent = case
                      when kol_cli_cteg_1_2 <= 500 then case
                                                          when par7 <= 3.0000 then 0.6
                                                          when par7 <= 7.0000 then 0.4
                                                          when par30 <= 7.0000 then 0.2
                                                          else 0
                                                        end
                      when kol_cli_cteg_1_2 <= 750 then case
                                                          when par7 <= 3.0000 then 0.7
                                                          when par7 <= 7.0000 then 0.6
                                                          when par30 <= 7.0000 then 0.3
                                                          else 0
                                                        end
                      when kol_cli_cteg_1_2 <= 1000 then case
                                                           when par7 <= 3.0000 then 0.9
                                                           when par7 <= 7.0000 then 0.7
                                                           when par30 <= 7.0000 then 0.4
                                                           else 0
                                                         end
                      when kol_cli_cteg_1_2 <= 1500 then case
                                                           when par7 <= 3.0000 then 1.0
                                                           when par7 <= 7.0000 then 0.9
                                                           when par30 <= 7.0000 then 0.5
                                                           else 0
                                                         end
                      when kol_cli_cteg_1_2 <= 2000 then case
                                                           when par7 <= 3.0000 then 1.2
                                                           when par7 <= 7.0000 then 1.0
                                                           when par30 <= 7.0000 then 0.6
                                                           else 0
                                                         end
                      when kol_cli_cteg_1_2 <= 3000 then case
                                                           when par7 <= 3.0000 then 1.3
                                                           when par7 <= 7.0000 then 1.2
                                                           when par30 <= 7.0000 then 0.7
                                                           else 0
                                                         end
                      when kol_cli_cteg_1_2 <= 4000 then case
                                                           when par7 <= 3.0000 then 1.5
                                                           when par7 <= 7.0000 then 1.3
                                                           when par30 <= 7.0000 then 0.8
                                                           else 0
                                                         end
                      else case
                             when par7 <= 3.0000 then 2.0
                             when par7 <= 7.0000 then 1.5
                             when par30 <= 7.0000 then 0.9
                             else 0
                           end
                    end;

--    kol_svz = null;
    bonus_lo = null;
    select -- count(*),
sum(bl_bonus_total)
    from temp_bonusru_lo
    where bl_region_id = :reg_id
    into -- kol_svz,
         bonus_lo;

--    if (kol_svz is null) then kol_svz = 0;
    if (bonus_lo is null) then
      bonus_lo = 0;

    if (kol_svz <> 0) then
      avg_bonus_lo = bonus_lo / kol_svz;
    else
      avg_bonus_lo = 0;

    bonus_kol_cli = avg_bonus_lo * bonus_percent;

    balance_last = null;
    delinq_balance7_last = null;
    par7_last = null;
    snijen_par7 = null;
    bonus_snijen_par = null;
    shtraf_snijen_par = null;
    if (par30 > 7.0000) then
    begin
      select sum(oks_balance), sum(iif(:rep_date_begin - oks_delinq_date > 7, oks_balance, 0))
      from oksummary
      where oks_region_id = :reg_id and
            :rep_date_begin between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_close_state = 0 and
            :rep_date_begin - oks_delinq_date < 181 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_begin, null, null, null, null, null, null)) in (1, 2)
      into balance_last, delinq_balance7_last;
      if (balance_last is null) then
        balance_last = 0;
      if (delinq_balance7_last is null) then
        delinq_balance7_last = 0;

      if (delinq_balance7_last = 0) then
        par7_last = 0;
      else
        par7_last = delinq_balance7_last / balance_last * 100;

      snijen_par7 = par7_last - par7;

      if (snijen_par7 > 0) then
      begin
        if (:rep_date_end <= '30.09.2014') then
        begin
          bonus_snijen_par = case
                               when snijen_par7 >= 4.0000 then 12000
                               when snijen_par7 >= 3.0000 then 10000
                               when snijen_par7 >= 2.0000 then 8000
                               when snijen_par7 >= 1.0000 then 7000
                               else 0
                             end;
        end
        else
        if (:rep_date_end <= '31.10.2014') then
        begin
          bonus_snijen_par = case
                               when snijen_par7 >= 4.0000 then 10000
                               when snijen_par7 >= 3.0000 then 9000
                               when snijen_par7 >= 2.0000 then 7000
                               when snijen_par7 >= 1.0000 then 6000
                               else 0
                             end;
        end
        else
        if (:rep_date_end <= '30.11.2014') then
        begin
          bonus_snijen_par = case
                               when snijen_par7 >= 4.0000 then 9000
                               when snijen_par7 >= 3.0000 then 8000
                               when snijen_par7 >= 2.0000 then 6000
                               when snijen_par7 >= 1.0000 then 5000
                               else 0
                             end;
        end
      end
      else
      if (snijen_par7 <= -0.1000) then
      begin
        if (snijen_par7 < -1.0000) then
          shtraf_snijen_par = 6000;
        else
          shtraf_snijen_par = 4000;
      end
    end

    bonus_end = bonus_kol_cli + coalesce(bonus_snijen_par, 0);

--  Вычисляет кол-во СБ
    kol_sb = null;
    with tabl1
    as (select ps1.ps_from_c_id,
               (select ps2.ps_date_begin
                from hr_pstatus ps2, joblist jl2
                where ps2.ps_from_c_id = ps1.ps_from_c_id and
                      ps2.ps_date_begin is not null and
                      ps2.ps_status = 1 and
                      ps2.ps_from_jl_id = jl2.jl_id and
                      jl2.jl_kind = 8
                order by ps2.ps_date_begin
                rows 1) as ps_stage
        from hr_pstatus ps1, joblist jl1
        where :rep_date_end between ps1.ps_date_begin and ps1.ps_date_end and
              ps1.ps_status = 1 and
              ps1.ps_is_working = 1 and
              ps1.ps_from_reg_id = :reg_id and
              ps1.ps_from_jl_id = jl1.jl_id and
              jl1.jl_kind = 8)
    select count(*)
    from tabl1
--    where cast(:rep_date_end as date) - ps_stage >= 30
    into kol_sb;
    if (kol_sb is null) then
      kol_sb = 0;

    suspend;
  end

end
