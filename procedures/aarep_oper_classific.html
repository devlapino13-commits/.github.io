CREATE PROCEDURE AAREP_OPER_CLASSIFIC(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_RES_ID INTEGER)
RETURNS (
    RES_ID INTEGER,
    RES_OL_ID D_STR_20,
    RES_OB_NUMBER INTEGER,
    RES_DATE DATE,
    RES_DOC_NO INTEGER,
    RES_DOC_NAME VARCHAR(20),
    RES_ACCOUNT_DT VARCHAR(20),
    RES_ACCOUNT_KT VARCHAR(20),
    RES_CURRENCY VARCHAR(3),
    RES_AMOUNT DECIMAL(18,4),
    RES_IS_BANK INTEGER,
    RES_OPER_SIGN INTEGER,
    RES_CLIENT_ID DECIMAL(18,0),
    RES_BANK_ID VARCHAR(10),
    RES_ERROR INTEGER,
    RES_CLIENT_NAME VARCHAR(100),
    RES_CLIENT_IS_RESIDENT INTEGER,
    RES_CLIENT_TYPE VARCHAR(10),
    RES_CLIENT_RESIDENT_STR VARCHAR(20),
    RES_CLIENT_COUNTRY VARCHAR(10),
    RES_OPERCODE_RD_ID INTEGER,
    RES_OPERCODE_STR VARCHAR(20),
    RES_OPER_SIGN_STR VARCHAR(20),
    RES_REGION_ID INTEGER,
    ERR_TEXT VARCHAR(2048),
    RES_CLIENT_DATE D_DATE,
    RES_FROM_KC_ID D_ID,
    RES_KONTRACT_NO D_STR_CODE,
    RES_ACC_CODE D_STR_CODE,
    RES_OL_MEMO VARCHAR(250))
AS
begin
  err_text = '';
  if (not(exists(select *
                 from temp_moneyclassific
                 rows 1))) then
  begin
-- Отчет запускается первый раз, заполняем временную таблицу.
    select err_text
    from aarep_oper_classific_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
    into :err_text;
    if (err_text <> '') then
    begin
      suspend;
      exit;
    end
  end
-- Ниже данные будут браться из временной таблице

  for select res_id, res_ol_id, res_ob_number, res_date, res_doc_no, res_doc_name, res_account_dt, res_account_kt,
             res_currency, res_amount, res_is_bank, res_oper_sign, res_client_id, res_bank_id, res_error,
             res_client_name, res_client_is_resident, res_client_type, res_client_resident_str, res_client_country,
             res_opercode_rd_id, res_opercode_str, res_oper_sign_str, res_region_id,
             res_client_date, res_from_kc_id, res_kontract_no, res_acc_code,res_ol_memo
      from temp_moneyclassific
      where (:rep_res_id = 0 or
            res_id = :rep_res_id)
      into :res_id, :res_ol_id, :res_ob_number, :res_date, :res_doc_no, :res_doc_name, :res_account_dt, :res_account_kt,
           :res_currency, :res_amount, :res_is_bank, :res_oper_sign, :res_client_id, :res_bank_id, :res_error,
           :res_client_name, :res_client_is_resident, :res_client_type, :res_client_resident_str, :res_client_country,
           :res_opercode_rd_id, :res_opercode_str, :res_oper_sign_str, res_region_id,
           :res_client_date, :res_from_kc_id, :res_kontract_no, :res_acc_code,:res_ol_memo
  do
  begin
    suspend;
  end
end
