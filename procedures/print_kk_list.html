CREATE PROCEDURE PRINT_KK_LIST(
    CMP_ID D_INTEGER)
RETURNS (
    K_NUMBER D_INTEGER,
    K_OFFICE D_STR_100,
    K_KONTRACT_NO D_STR_CODE,
    K_FOLDER D_STR_CODE,
    K_CASHIER_FIO D_STR_100,
    K_ZAPROS_DATE D_DATE,
    K_OPER_DATE D_DATE,
    PARAM_PROLONG D_STR_CODE,
    PARAM_NEW_DAY D_STR_CODE,
    PARAM_LGOTA D_STR_CODE,
    PARAM_ADD_MONTH D_STR_CODE,
    STR_PROLONG D_STR_CODE,
    STR_NEW_DAY D_STR_CODE,
    STR_LGOTA D_STR_CODE,
    STR_ADD_MONTH D_STR_CODE)
AS
declare variable enum_approved d_integer;
declare variable enum_done d_integer;
declare variable obj_kind d_integer;
declare variable k_id d_id;
declare variable kf_id d_id;
declare variable reg_id d_integer;
declare variable add_params d_str_long;
declare variable cml_last_status d_integer;
declare variable cml_id d_id;
begin
  k_number = 0;

  enum_approved = f_enum('com_action_status', 'approved', 'id');
  enum_done = f_enum('com_action_status', 'done', 'id');
  obj_kind = f_objkind('kontract', 'id');

  for select cml_id, cml_date, cml_from_reg_id, cml_object_id, cml_add_params, cml_last_status
  from com_list, com_cfg
  where cml_last_cmp_id = :cmp_id and
        cml_last_status in (:enum_approved, :enum_done) and
        cml_object_kind = :obj_kind and
        cml_from_cmc_id = cmc_id
  into :cml_id, :k_zapros_date, :reg_id, :k_id, :add_params, cml_last_status
  do
  begin
    k_number = k_number + 1;
    k_office = '';
    k_kontract_no = '';
    k_folder = '';
    k_cashier_fio = '';
    kf_id = null;

    k_oper_date = null;
    if (cml_last_status = enum_done) then
    begin
      select ol.ol_date
      from com_action, com_operlist, operlist ol
      where cma_from_cml_id = :cml_id and
            cmo_from_cma_id = cma_id and
            cmo_from_ol_id = ol.ol_id and
            ol.is_delete = 0
      order by cmo_id desc
      rows 1
      into k_oper_date;
    end


    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into k_office;

    select k_kontract_no, k_from_kf_id
    from kontract
    where k_id = :k_id
    into k_kontract_no, kf_id;

    select kal_new_kf_name
    from knaktlist a
    where a.kal_from_kf_id = :kf_id and
          a.is_delete = 0 and
          a.kal_date <= :k_zapros_date and
          a.kal_date_end >= :k_zapros_date
    into k_folder;

    select c_name_fio
    from knclient, clients
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = c_id
    order by kc_status desc
    rows 1
    into k_cashier_fio;

    if (coalesce(add_params, '') = '') then
      add_params = '{}';

    param_prolong = f_json_str_get_value(add_params, 'add_month_count', '0');
    param_new_day = f_json_str_get_value(add_params, 'new_pay_day', '0');
    param_lgota = f_json_str_get_value(add_params, 'perc_lgota_value', '0');
    param_add_month = f_json_str_get_value(add_params, 'add_month_kind', '0');

    str_prolong = param_prolong;
    str_new_day = param_new_day;
    str_lgota = param_lgota;
    str_add_month = param_add_month;

    if (str_prolong = '0')
    then str_prolong = 'Без пролонгации';
    else str_prolong = f_format('Пролонг. на %1 мес.', str_prolong);

    if (str_lgota = 0)
    then str_lgota = 'Нет скидки';
    else str_lgota = f_format('Скидка %1%', str_lgota);

    if (str_add_month = 0)
    then str_add_month = 'Без сдвига';
    else str_add_month = f_format('Сдвиг на %1 мес', str_add_month);

    if (str_new_day = '0') then
      str_new_day = '-';

    suspend;
  end
end
