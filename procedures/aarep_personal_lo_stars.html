CREATE PROCEDURE AAREP_PERSONAL_LO_STARS(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    PS_PERIOD_MONTH INTEGER,
    PS_DATE_END_IN_ORDER DATE,
    PS_TRANSFER_OFFICE INTEGER,
    PS_TABEL_NO VARCHAR(10),
    HO_DATE DATE,
    HO_FROM_REG_ID INTEGER,
    HO_TEXT VARCHAR(1024),
    HO_REASON VARCHAR(255),
    HO_SAVE_TIME TIMESTAMP,
    HO_ORDER_NO VARCHAR(20),
    JL_NAME VARCHAR(100),
    JL_ORDER_NO INTEGER,
    DEP_NAME VARCHAR(100),
    DEP_ORDER_NO INTEGER,
    REG_NAME VARCHAR(50),
    REG_CODE_NAME VARCHAR(50),
    HO_REG_NAME VARCHAR(50),
    PS_IS_ON_STAFF_TEXT VARCHAR(50),
    PS_IS_WORKING_TEXT VARCHAR(50),
    JL_IS_KRED_PERSON VARCHAR(50),
    JL_IS_FRONT_TEXT VARCHAR(50),
    JL_CATEGORY VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_NATION VARCHAR(100),
    C_CITIZEN VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_EMAIL VARCHAR(100),
    C_FAMILY VARCHAR(100),
    C_EDUCATION VARCHAR(100),
    C_KOL_FAMILY INTEGER,
    C_KOL_CHILD INTEGER,
    C_MEMO VARCHAR(1024),
    C_REG_NO VARCHAR(20),
    C_OKPO VARCHAR(20),
    C_INN VARCHAR(20),
    C_ACCOUNT VARCHAR(20),
    C_ORG_FROM_B_ID VARCHAR(10),
    C_HIRED_DATE DATE,
    C_ARMY VARCHAR(100),
    C_PLASTCARD_NO VARCHAR(20),
    C_OTHERS VARCHAR(20),
    C_FULL_ADDRESS VARCHAR(255),
    C_SEX_TEXT VARCHAR(50),
    C_IS_RESIDENT_TEXT VARCHAR(50),
    C_STAGE INTEGER,
    C_STAGE_YEAR INTEGER,
    PS_FROM_C_ID DECIMAL(18,0),
    JL_ID INTEGER,
    PS_BRANCH VARCHAR(100),
    PS_CURR_STATUS VARCHAR(100),
    JL_KIND INTEGER,
    HRC_NAME VARCHAR(255),
    PS_FROM_REG_ID INTEGER,
    BRANCH_NAME VARCHAR(100),
    REG_STAT_REP_REGION VARCHAR(255),
    LO_OWN_PAR7 DECIMAL(18,7),
    LO_STAGE DECIMAL(18,4),
    LO_OWN_CLIENT_KOLICH INTEGER,
    HRC_IS_FORCED INTEGER)
AS
declare variable ps_is_on_staff smallint;
declare variable ps_is_working smallint;
declare variable jl_is_front integer;
declare variable c_sex smallint;
declare variable c_is_resident smallint;
declare variable st float;
declare variable st_staff integer;
declare variable st_client integer;
declare variable jl_code varchar(100);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ps_from_c_id, ps_date_begin, ps_date_end, ps_period_month, ps_date_end_in_order, ps_is_on_staff,
             ps_is_working, ps_transfer_office, ps_tabel_no, ho_date, ho_from_reg_id, ho_text, ho_reason, ho_save_time,
             ho_order_no, jl_id, jl_name, jl_order_no, dep_name, dep_order_no, r1.reg_name,
             r1.reg_code || ' ' || r1.reg_name, r2.reg_name, jl_is_front, jl_category, c_name_fio, c_sex, c_birth_date,
             d1.rd_name, d2.rd_name, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire,
             (select cont_value
              from hp_contdata_get_email_main(0, c_id)), d3.rd_name, d4.rd_name, c_kol_family, c_kol_child, c_memo,
             c_reg_no, c_okpo, c_inn, c_account, c_org_from_b_id, c_hired_date, c_is_resident, d5.rd_name,
             (select cont_value
              from hp_contdata_get_cell_main(0, c_id)), c_others, adl_address_str, r3.reg_code || ' ' || r3.reg_name,
             jk_system_id, jk_system_code, hrc_name, hrc_is_forced, ps_from_reg_id, br.reg_code || ' ' || br.reg_name, r1.reg_stat_rep_region

      from hr_pstatus, hr_orders, region_list(:report_region_id, 1, -1, 1), joblist, depart, region r1, region r2, region r3, clients, region br, refdata d1, refdata d2, refdata d3, refdata d4, refdata d5, hr_class, addrclient, addrlist, jobkind
      where ho_id = ps_from_ho_id and
            (ps_from_reg_id = id_list) and
            ps_date_begin <= :report_date_end and
            ps_date_end >= :report_date_end and
            r3.reg_unicode = r1.reg_head_office and
            ps_from_reg_id = r1.reg_unicode and
            br.reg_unicode = r1.parent and
            ps_from_dep_id = dep_id and
            ps_from_jl_id = jl_id and
            ho_from_reg_id = r2.reg_unicode and
            ps_class = hrc_id and
            ps_is_working = 1 and
            ps_from_c_id = c_id and
            c_nation = d1.rd_id and
            c_citizen = d2.rd_id and
            c_family = d3.rd_id and
            c_education = d4.rd_id and
            c_army = d5.rd_id and
            adc_address_type = 1 and
            adc_from_c_id = c_id and
            adc_from_adl_id = adl_id and
            jl_kind = jk_id
      order by jl_order_no, c_name_fio

      into :ps_from_c_id, :ps_date_begin, :ps_date_end, :ps_period_month, :ps_date_end_in_order, :ps_is_on_staff,
           :ps_is_working, :ps_transfer_office, :ps_tabel_no, :ho_date, :ho_from_reg_id, :ho_text, :ho_reason,
           :ho_save_time, :ho_order_no, :jl_id, :jl_name, :jl_order_no, :dep_name, :dep_order_no, :reg_name,
           :reg_code_name, :ho_reg_name, :jl_is_front, :jl_category, :c_name_fio, :c_sex, :c_birth_date, :c_nation,
           :c_citizen, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_email, :c_family, :c_education,
           :c_kol_family, :c_kol_child, :c_memo, :c_reg_no, :c_okpo, :c_inn, :c_account, :c_org_from_b_id,
           :c_hired_date, :c_is_resident, :c_army, :c_plastcard_no, :c_others, :c_full_address, :ps_branch, :jl_kind,
           :jl_code, :hrc_name, :hrc_is_forced, :ps_from_reg_id, :branch_name, :reg_stat_rep_region

  do
  begin
    if (ps_is_on_staff = 0) then
      ps_is_on_staff_text = 'Вне штатный';
    else
      ps_is_on_staff_text = 'Штатный';

    if (ps_is_working = 1) then
      ps_is_working_text = 'Работает';
    else
      ps_is_working_text = 'Уволен';

    if (jl_code in ('JobSystem.LO_Standart', 'JobSystem.Region_KM', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
      jl_is_kred_person = 'Кредит.персонал';
    else
      jl_is_kred_person = 'Админ.персонал';

    if (jl_is_front = 1) then
      jl_is_front_text = 'Фронт офис';
    else
      jl_is_front_text = 'Бэк офис';

    if (c_sex = 1) then
      c_sex_text = 'Мужчина';
    else
      c_sex_text = 'Женщина';

    if (c_is_resident = 1) then
      c_is_resident_text = 'Резидент';
    else
      c_is_resident_text = 'Не резидент';

    if (c_hired_date is not null) then
      c_stage = datediff(month, c_hired_date, report_date_end);
    else
      c_stage = 0;
    st = c_stage / 12;
    c_stage_year = cast(st as integer);
    if (c_stage_year > (c_stage / 12)) then
      c_stage_year = c_stage_year - 1;

    ps_curr_status = '';
    select sawt_name
    from hr_voctravel, hr_orders, sa_worktype
    where hrv_from_ho_id = ho_id and
          ho_order_type = sawt_id and
          hrv_from_c_id = :ps_from_c_id and
          hrv_date_begin <= :report_date_end and
          hrv_date_end_fact >= :report_date_end and
          hrv_status = 1
    order by hrv_date_begin desc
    rows 1
    into :ps_curr_status;

    select lo_own_client_kolich, lo_own_par7, lo_stage
    from aarep_analitica_port_one_lo(:ps_from_c_id, :report_date_end)
    into :lo_own_client_kolich, :lo_own_par7, :lo_stage;
    if ((lo_own_par7 <= 1) and
        (lo_own_client_kolich >= 200) and
        (lo_stage >= 12)) then
      suspend;
  end
end
