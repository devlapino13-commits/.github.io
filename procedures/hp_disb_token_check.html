CREATE PROCEDURE HP_DISB_TOKEN_CHECK(
    K_ID ID_TYPE)
RETURNS (
    ERROR_TEXT VARCHAR(1000))
AS
declare variable has_err smallint;
declare variable c_id id_type;
declare variable c_name_fio str_100;
declare variable cli_qty integer;
begin
  has_err = 0;
---------- Проверка наличия активных Токенов -------------------------------->
  error_text = 'Отсутствуют токены у клиентов: ';
  cli_qty = 0;

  for select kncl.kc_from_c_id
      from knclient kncl
      where kncl.kc_from_k_id = :k_id and
            kncl.kc_amount_approved > 0
      into :c_id
  do
  begin
    cli_qty = cli_qty + 1;

    if (not exists(select *
                   from extsys_token et
                   where et.et_from_c_id = :c_id and
                         et.et_is_actual = 1)) then
    begin
      if (has_err = 0) then
        has_err = 1;

      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into :c_name_fio;

      error_text = error_text || ' ' || c_name_fio || '; ';
    end
  end

  if (cli_qty > 1) then
  begin
    has_err = 1;
    error_text = 'Выдача на кредитные карты доступна только для индивидуальных кредитов';
  end
-----------------------------------------------------------------------------<
---------- Проверка наличия операции по выдаче ------------------------------>
  if (has_err = 0) then
  begin
    if (exists(select *
               from extsys_card_disb ecd
               where ecd.ecd_from_k_id = :k_id and
                     ecd.ecd_status >= 0)) then
    begin
      has_err = 1;
      error_text = 'Выдача отменена. Имеется активная операция по выдаче на карту';
    end
  end
-----------------------------------------------------------------------------<
  if (has_err = 0) then
    error_text = '';

  suspend;
end
