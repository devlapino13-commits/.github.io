CREATE PROCEDURE HP_LISTSMSSEND_UPDATE_STATUS
AS
declare variable msg_send_status smallint;
declare variable lss_id integer;
begin
  for
  select distinct lss_id, msg_send_status
  from listsmssend, fsrobot_msgbox
  where lss_cell_no = msg_mail_to
  and (msg_body = lss_sms_text1 or msg_body = lss_sms_text2)
  and msg_body is not null
  and msg_body <> ''
  and lss_send_status <> 1
  and cast(msg_datetime as date) = '20.06.2019'   -- дата последней отправки
  into lss_id, msg_send_status
  do
  begin
    if (msg_send_status is not null) then
    begin
      update listsmssend set lss_send_status = :msg_send_status
      where lss_id = :lss_id
      and lss_send_status <> :msg_send_status;
    end
  end 
end
