CREATE PROCEDURE HP_KONTREFINANCE_LIST_TMP(
    C_ID D_ID,
    CALC_DATE D_DATE)
RETURNS (
    RES_K_ID D_ID,
    RES_AVG_REPAY D_CURRENCY,
    LOAN_BALANCE D_CURRENCY,
    LOAN_PERC D_CURRENCY)
AS
declare variable last_date d_date;
declare variable repay_count d_integer;
declare variable res_delinq_repay d_currency;
begin
  for select ks_from_k_id
  from knclient, kontstate
  where kc_from_c_id = :c_id and
        kc_amount_approved > 0 and
        kc_from_k_id = ks_from_k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_date_begin + 0 <= :calc_date and
        ks_date_end + 0 >= :calc_date
  into res_k_id
  do
  begin
    last_date = null;
    res_avg_repay = 0;
    res_delinq_repay = 0;
    repay_count = 0;
    loan_balance = 0;
    loan_perc = 0;
    for select ok_sh_date, sum(ok_os + ok_sh_os) as zad_os, sum(ok_perc + ok_sh_perc) as zad_perc
    from operkred
    where ok_from_k_id = :res_k_id and
          ok_ol_date <= :calc_date and
          ok_ol_is_active = 1 and
          ok_type < 8
    group by 1
    having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0
    as cursor cur
    do
    begin
      if (cur.zad_os <= 0 and cur.zad_perc <= 0) then
        Continue;

      if (last_date is null or last_date < cur.ok_sh_date) then
        last_date = cur.ok_sh_date;

      if (cur.ok_sh_date >= calc_date) then
        res_avg_repay = res_avg_repay + cur.zad_os + cur.zad_perc;
      else
        res_delinq_repay = res_delinq_repay + cur.zad_os + cur.zad_perc;

      loan_balance = loan_balance + cur.zad_os;
    end

    loan_perc = 0;
    select result_perc_end
    from aarep_calc_perc_one_saldo(:res_k_id, :calc_date, :calc_date)
    into loan_perc;

    repay_count = 0;
    if (calc_date <= last_date) then
      repay_count = (extract(year from last_date) - extract(year from calc_date)) * 12 +
                          extract(month from last_date) - extract(month from calc_date) +
                          iif(extract(day from last_date) > extract(day from calc_date), 1, 0);

    if (repay_count > 0) then
    begin
      res_avg_repay = round(res_avg_repay / repay_count);
      suspend;
    end

    res_avg_repay = res_avg_repay + res_delinq_repay;
  end
end
