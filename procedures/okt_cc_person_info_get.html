CREATE PROCEDURE OKT_CC_PERSON_INFO_GET(
    REP_CP_ID INTEGER)
RETURNS (
    CP_ID INTEGER,
    CP_CELL_NO STR_20,
    CP_FIO STR_255,
    CP_CITIZEN INTEGER,
    CP_CITY STR_255,
    CP_ADDIT_CELL_NO STR_20,
    CP_FROM_C_ID ID_TYPE,
    CP_FROM_IS_ID INTEGER,
    CP_MEMO STR_255,
    CP_EXT_INFO STR_255)
AS
begin
  for select cp.cp_id, cp.cp_cell_no, cp.cp_fio, cp.cp_citizen, cp.cp_city, cp.cp_addit_cell_no,
             coalesce(cp.cp_from_c_id, 0), cp.cp_from_is_id, cp.cp_memo, cp.cp_ext_info
      from cc_persons cp
      where cp.cp_id = :rep_cp_id
      into :cp_id, :cp_cell_no, :cp_fio, :cp_citizen, :cp_city, :cp_addit_cell_no, :cp_from_c_id, :cp_from_is_id,
           :cp_memo, :cp_ext_info
  do
  begin
    suspend;
  end
end
