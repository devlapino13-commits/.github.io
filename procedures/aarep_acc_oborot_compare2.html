CREATE PROCEDURE AAREP_ACC_OBOROT_COMPARE2(
    REPORT_REGION_ID INTEGER,
    REPORT_YEAR INTEGER,
    REPORT_MONTH INTEGER,
    REPORT_PERIOD_KIND INTEGER,
    REPORT_PERIOD_KOLICH INTEGER,
    REPORT_SHOW_ACC_NO INTEGER,
    REPORT_SHOW_OFFICE INTEGER)
RETURNS (
    REP_ACCOUNT VARCHAR(20),
    REP_ACCOUNT_NO INTEGER,
    REP_BALANCE_NAME VARCHAR(255),
    REP_ACCOUNT_NAME VARCHAR(255),
    REP_GROUP_NO VARCHAR(255),
    REP_SUMMA1 DECIMAL(18,4),
    REP_SUMMA2 DECIMAL(18,4),
    REP_SUMMA3 DECIMAL(18,4),
    REP_SUMMA4 DECIMAL(18,4),
    REP_SUMMA5 DECIMAL(18,4),
    REP_SUMMA6 DECIMAL(18,4),
    REP_SUMMA7 DECIMAL(18,4),
    REP_SUMMA8 DECIMAL(18,4),
    REP_SUMMA9 DECIMAL(18,4),
    REP_SUMMA10 DECIMAL(18,4),
    REP_SUMMA11 DECIMAL(18,4),
    REP_SUMMA12 DECIMAL(18,4),
    BG_KIND_TEXT VARCHAR(30),
    OFFICE_ID INTEGER,
    OFFICE_NAME VARCHAR(100),
    REGION_ID INTEGER,
    REGION_NAME VARCHAR(100))
AS
declare variable rep_summa decimal(18,4);
declare variable rep_period_id integer;
declare variable bg_kind integer;
declare variable office_id_old integer;
begin
  rep_account_no = 0;
  office_id_old = 0;
  for select al_from_bg_id, al_number, b1.bg_name, b2.bg_id || ' ' || b2.bg_name, sum(rep_summa), rep_period_id,
             max(al_name), b1.bg_kind, al.al_from_reg_id
      from acc_list al, balgroup b1, balgroup b2, aarep_acc_oborot_compare_temp(:report_region_id, :report_year, :report_month, :report_period_kind, :report_period_kolich)
      where rep_account = al_id and
            al_from_bg_id = b1.bg_id and
            b1.parent = b2.bg_id and
            b1.bg_kind in (4, 5)
      group by al_from_bg_id, al_number, al_from_reg_id, b1.bg_name, b2.bg_id, b2.bg_name, rep_period_id, b1.bg_kind
      into :rep_account, :rep_account_no, :rep_balance_name, :rep_group_no, :rep_summa, :rep_period_id,
           rep_account_name, bg_kind, office_id
  do
  begin
    if (bg_kind = 4) then
      bg_kind_text = 'Доходы';
    if (bg_kind = 5) then
      bg_kind_text = 'Расходы';
    if (report_show_acc_no = 0) then
      rep_account_no = 0;
    if (report_show_office = 0) then
      office_id = 1000000;

    if (office_id <> 1000000) then
    begin
      if (office_id <> office_id_old) then
      begin
        office_id_old = office_id;

        select reg.reg_name, reg2.reg_unicode, reg2.reg_name
        from region reg
        inner join region reg2 on reg.parent = reg2.reg_unicode
        where reg.reg_unicode = :office_id
        into :office_name, :region_id, :region_name;
      end
    end
    else
    begin
      if (office_name is null) then
      begin
        office_name = '';
        region_id = 1000000;
        region_name = '';
      end
    end

    rep_summa1 = 0;
    rep_summa2 = 0;
    rep_summa3 = 0;
    rep_summa4 = 0;
    rep_summa5 = 0;
    rep_summa6 = 0;
    rep_summa7 = 0;
    rep_summa8 = 0;
    rep_summa9 = 0;
    rep_summa10 = 0;
    rep_summa11 = 0;
    rep_summa12 = 0;

    if (rep_period_id = 1) then
      rep_summa1 = :rep_summa;
    if (rep_period_id = 2) then
      rep_summa2 = :rep_summa;
    if (rep_period_id = 3) then
      rep_summa3 = :rep_summa;
    if (rep_period_id = 4) then
      rep_summa4 = :rep_summa;
    if (rep_period_id = 5) then
      rep_summa5 = :rep_summa;
    if (rep_period_id = 6) then
      rep_summa6 = :rep_summa;
    if (rep_period_id = 7) then
      rep_summa7 = :rep_summa;
    if (rep_period_id = 8) then
      rep_summa8 = :rep_summa;
    if (rep_period_id = 9) then
      rep_summa9 = :rep_summa;
    if (rep_period_id = 10) then
      rep_summa10 = :rep_summa;
    if (rep_period_id = 11) then
      rep_summa11 = :rep_summa;
    if (rep_period_id = 12) then
      rep_summa12 = :rep_summa;

    suspend;
  end
end
