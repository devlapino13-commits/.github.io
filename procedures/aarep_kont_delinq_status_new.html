CREATE PROCEDURE AAREP_KONT_DELINQ_STATUS_NEW(
    REP_K_ID DECIMAL(18,0),
    REP_DATE_END DATE)
RETURNS (
    REP_IS_DELINQ INTEGER,
    REP_DELINQ_BEGIN DATE,
    REP_DELINQ_DAYS INTEGER,
    REP_DELINQ_BALANCE CURR_TYPE)
AS
/*Проверка контракта на просрочку, на указанную дату*/
begin
  rep_is_delinq = 0;
  rep_delinq_days = 0;
  rep_delinq_begin = null;
  rep_delinq_balance = 0;

  select oks_delinq_date, oks_balance
  from oksummary
  where :rep_date_end between oks_date_begin and oks_date_end and
        oks_delinq_date <= :rep_date_end and
        oks_from_k_id = :rep_k_id and
        oks_close_state = 0
  into :rep_delinq_begin, :rep_delinq_balance;
  if (rep_delinq_begin is not null) then
  begin
    rep_is_delinq = 1;
    rep_delinq_days = cast(:rep_date_end - :rep_delinq_begin as integer);
  end
  suspend;
end
