CREATE PROCEDURE CB_AZ_RF_2016_05(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    KOL_KC INTEGER,
    ZARABOT_KC CURR_TYPE,
    SRED_ZARAB_KC CURR_TYPE,
    X CURR_TYPE,
    SUM_VZYSKAN CURR_TYPE,
    Y CURR_TYPE,
    PERC_SBORA CURR_TYPE,
    AAREP_BONUS_TOTAL CURR_TYPE,
    KOL_SSVZ INTEGER,
    FIO_SSVZ VARCHAR(1024))
AS
declare variable reg_id integer;
declare variable rep_date_begin date;
declare variable officer_id id_type;
declare variable zarabot_kc_temp curr_type;
begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;
  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into reg_id
  do
  begin

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;

    kol_kc = 0;
    zarabot_kc = 0;
    for select ps_from_c_id
        from hr_pstatus, joblist, jobkind
        where :rep_date_end between ps_date_begin and ps_date_end and
              ps_from_reg_id = :reg_id and
              ps_status = 1 and
              ps_is_working = 1 and
              ps_from_c_id > 1000000000003 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code in ('JobSystem.LO_Standart')
        into officer_id
    do
    begin
      kol_kc = kol_kc + 1;

      zarabot_kc_temp = null;
      select aarep_bonus_total
      from cb_az_lo_201602_one_lo(:reg_id, :officer_id, :rep_date_end, 0)
      into zarabot_kc_temp;
      zarabot_kc_temp = coalesce(zarabot_kc_temp, 0);

--      if (zarabot_kc_temp > 0) then
--      begin
      zarabot_kc = zarabot_kc + zarabot_kc_temp;
--      end
    end

    if (kol_kc > 0) then
      sred_zarab_kc = cast(zarabot_kc as double precision) / kol_kc;
    else
      sred_zarab_kc = 0;

    x = case
          when sred_zarab_kc < 300 then 0
          when sred_zarab_kc <= 400 then 0.07
          when sred_zarab_kc <= 500 then 0.13
          when sred_zarab_kc <= 600 then 0.20
          when sred_zarab_kc <= 700 then 0.27
          when sred_zarab_kc <= 800 then 0.30
          when sred_zarab_kc <= 900 then 0.33
          when sred_zarab_kc <= 1000 then 0.35
          else 0.4
        end;
-- Расчет суммы взысканий
    sum_vzyskan = null;
    select -sum(ok_os)
    from operkred
    where ok_ol_reg_id = :reg_id and
          ok_ol_is_active = 1 and
          ok_ol_date between :rep_date_begin and :rep_date_end
--    and ok_sh_date < cast('01.05.2016' as date)
          and
          ok_sh_date < :rep_date_begin -- при расчете за июнь июль итд нужно уточнить, будем брать просроченные до 1 мая
          and
          ok_type in (4, 9)
    into sum_vzyskan;
    sum_vzyskan = coalesce(sum_vzyskan, 0);

    if (sum_vzyskan < 1000.0000) then
    begin
      y = -200;
      perc_sbora = 0;
    end
    else
    begin
      perc_sbora = case
                     when sum_vzyskan <= 5000.000 then 3
                     when sum_vzyskan <= 10000.000 then 4
                     else 5
                   end;
      y = sum_vzyskan * perc_sbora / 100;
    end

    aarep_bonus_total = zarabot_kc * x + y;
    if (aarep_bonus_total < 0) then
      aarep_bonus_total = 0;
    aarep_bonus_total = round(aarep_bonus_total, 0);
--    Вычисление кол-во ССВЗ
    kol_ssvz = 0;
    fio_ssvz = '';
    select count(*), list(c_name_fio, '; ')
    from hr_pstatus, joblist, jobkind, clients
    where ps_status = 1 and
          ps_is_working = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code = 'JobSystem.Region_KM' and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_reg_id = :reg_id and
          ps_from_c_id = c_id
    into kol_ssvz, fio_ssvz;
    if (kol_ssvz is null) then
      kol_ssvz = 0;

    if (kol_kc > 0 or zarabot_kc > 0 or sum_vzyskan > 0 and
        kol_ssvz > 0) then
      suspend;
  end
end
