CREATE PROCEDURE UXP$UXFRLOOKUP_INFO(
    UFP_UFR_UUID D_UUID,
    UFP_UF_UUID D_UUID)
RETURNS (
    US_CODE D_STR_CODE,
    LIST_KIND D_SMALLINT,
    US_LOOKUP_ID D_STR_CODE,
    US_LOOKUP_TEXT D_STR_255)
AS
declare variable ufp_uuid d_uuid;
declare variable main_lookup_id d_str_code;
declare variable main_lookup_text d_str_255;
declare variable us_uuid d_uuid;
declare variable us_main_table d_str_code;
begin
  select ufp_uuid
  from ux$frparam
  where ufp_ufr_uuid = :ufp_ufr_uuid and
        ufp_uf_uuid = :ufp_uf_uuid
  into ufp_uuid;

  us_code = '';

  select us_uuid, us_code, ufl_list_kind, ufl_lookup_id, ufl_lookup_text, us_main_table
  from ux$frlookup, ux$source
  where ufl_ufp_uuid = :ufp_uuid and
        ufl_us_uuid = us_uuid
  into :us_uuid, :us_code, :list_kind, us_lookup_id, us_lookup_text, us_main_table;

  if (us_lookup_id is null) then
    us_lookup_id = '';

  if (us_lookup_text is null) then
    us_lookup_text = '';

  if (us_main_table is null) then
    us_main_table = '';

  if ((us_lookup_id = '' or us_lookup_text = '') and us_main_table <> '') then
  begin
    for select us_lookup_id, us_lookup_text
    from ux$source
    where us_code = :us_main_table
    into :main_lookup_id, :main_lookup_text
    do
    begin
      if (main_lookup_id is null) then
        main_lookup_id = '';

      if (main_lookup_text is null) then
        main_lookup_text = '';

      if (us_lookup_id = '' and main_lookup_id <> '') then
      begin
        if (exists(select * from ux$field
                   where uf_owner = :us_uuid and
                         uf_field_name = :main_lookup_id)) then
          us_lookup_id = main_lookup_id;
      end

      if (us_lookup_text = '' and main_lookup_text <> '') then
      begin
        if (exists(select * from ux$field
                   where uf_owner = :us_uuid and
                         uf_field_name = :main_lookup_text)) then
          us_lookup_text = main_lookup_text;
      end
    end
  end

  if (list_kind is null) then
    list_kind = 0;

  suspend;
end
