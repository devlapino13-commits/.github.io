CREATE PROCEDURE HP_FILL_KN_KONTCALCPERC_FIX
RETURNS (
    K_ID DECIMAL(18,0),
    IS_DONE INTEGER,
    K_IS_DEVIDED INTEGER,
    ITER INTEGER)
AS
declare variable kc_id_minus id_type;
declare variable kc_id_plus id_type;
declare variable kcp_period integer;
declare variable min_saldo curr_type;
declare variable prior_calced_period integer;
declare variable plus_saldo curr_type;
declare variable total_done integer;
begin
  iter = 0;

  while (iter < 20) do
  begin
    iter = iter + 1;
    total_done = 0;

    for select distinct kcp_from_k_id, k_is_devided
        from kn_kontcalcperc, kontract
        where kcp_perc_saldo < 0 and
              k_id = kcp_from_k_id
        into k_id, k_is_devided
    do
    begin
      is_done = 0;
      kcp_period = null;
      min_saldo = 0;
      kc_id_minus = null;
      kc_id_plus = null;
      prior_calced_period = null;

      select kcp_period, min(kcp_perc_saldo)
      from kn_kontcalcperc
      where kcp_from_k_id = :k_id
      group by kcp_period
      having min(kcp_perc_saldo) < 0 and sum(kcp_perc_saldo) >= 0
      order by kcp_period desc
      rows 1
      into :kcp_period, min_saldo;

      if (kcp_period is not null) then
      begin
        select kcp_from_kc_id
        from kn_kontcalcperc
        where kcp_from_k_id = :k_id and
              kcp_period = :kcp_period and
              kcp_perc_saldo = :min_saldo rows 1
        into kc_id_minus;

        select kcp_from_kc_id
        from kn_kontcalcperc
        where kcp_from_k_id = :k_id and
              kcp_period = :kcp_period and
              kcp_perc_saldo >= -:min_saldo rows 1
        into kc_id_plus;

        if (kc_id_minus is not null and
            kc_id_plus is not null) then
        begin
          select kcp_period
          from kn_kontcalcperc
          where (kcp_from_kc_id = :kc_id_minus or
                kcp_from_kc_id = :kc_id_plus) and
                kcp_period <= :kcp_period and
                kcp_perc_calced > 0
          order by kcp_from_kc_id desc, kcp_period desc
          rows 1
          into :prior_calced_period;

          if (prior_calced_period is not null) then
          begin
            is_done = 1;
            update kn_kontcalcperc
            set kcp_perc_calced = kcp_perc_calced - :min_saldo
            where kcp_from_kc_id = :kc_id_minus and
                  kcp_period = :prior_calced_period;

            update kn_kontcalcperc
            set kcp_perc_calced = kcp_perc_calced + :min_saldo
            where kcp_from_kc_id = :kc_id_plus and
                  kcp_period = :prior_calced_period;
          end
        end
        else
        begin
          for select kcp_from_kc_id, kcp_perc_saldo
              from kn_kontcalcperc
              where kcp_from_k_id = :k_id and
                    kcp_period = :kcp_period and
                    kcp_perc_saldo > 0
              into kc_id_plus, plus_saldo
          do
          begin
            if (plus_saldo < abs(min_saldo)) then
            begin
              prior_calced_period = null;
              select kcp_period
              from kn_kontcalcperc
              where (kcp_from_kc_id = :kc_id_minus or
                    kcp_from_kc_id = :kc_id_plus) and
                    kcp_period <= :kcp_period and
                    kcp_perc_calced > 0
              order by kcp_from_kc_id desc, kcp_period desc
              rows 1
              into :prior_calced_period;

              if (prior_calced_period is not null) then
              begin
                is_done = 1;
                update kn_kontcalcperc
                set kcp_perc_calced = kcp_perc_calced + :plus_saldo
                where kcp_from_kc_id = :kc_id_minus and
                      kcp_period = :prior_calced_period;

                update kn_kontcalcperc
                set kcp_perc_calced = kcp_perc_calced - :plus_saldo
                where kcp_from_kc_id = :kc_id_plus and
                      kcp_period = :prior_calced_period;

                min_saldo = min_saldo + plus_saldo;
              end
            end
          end
        end
      end
      total_done = total_done + is_done;
      suspend;
    end

    if (total_done = 0) then
    begin
      k_id = -1;
      iter = 21;
      suspend;
      exit;
    end
  end
end
