CREATE PROCEDURE OKT_KONTRACT_CONTACT_LIST(
    K_ID ID_TYPE)
RETURNS (
    C_ID ID_TYPE,
    C_NAME_FIO STR_100,
    CC_CLIENT_OK SMALLINT,
    CC_NAME_OK SMALLINT,
    CC_STATUS_OK SMALLINT,
    CC_ADDRESS_OK SMALLINT,
    CC_NAME STR_100,
    CC_STATUS STR_100,
    CC_ADDRESS STR_255,
    CC_CELL_NO STR_255,
    CC_TYPE_STR VARCHAR(20),
    CC_TIME_END TIMESTAMP,
    CC_TRY_KOL INTEGER,
    CC_CHECK_MEMO STR_255,
    CC_CHECK_RESULT SMALLINT,
    IS_REPEATED SMALLINT,
    IS_OLD_CONTACT SMALLINT,
    CC_CALL_USER STR_255,
    CC_CALL_BACK TIMESTAMP,
    CC_CREATE_DATE TIMESTAMP,
    CC_BIRTH_DATE_FILLED INTEGER,
    CC_VERIF_IGNORE INTEGER,
    CC_CHECKED_DATETIME TIMESTAMP)
AS
declare variable cm_check_contact integer;
declare variable cc_id id_type;
declare variable kol_client integer;
declare variable kol_contact integer;
declare variable kol_region integer;
declare variable kol_rodina integer;
declare variable cc_type integer;
declare variable default_curr varchar(3);
declare variable is_suspend integer;
declare variable cc_date date;
begin
  /*
   CREATED BY SISLAMOV
   PURPOSE: СТАТУС ПРОВЕРКИ КОНТАКТОВ ОПЕРАТОРАМИ КОЛЛ-ЦЕНТРА

   CHANGED BY IGUSEV 10.08.2012
   CHANGES: Добавлен исходящий параметр cc_check_result. А также условие    if ((cc_check_result = -1) and (cc_try_kol<3))  then  cc_check_result = 0;

   */

  select cm_check_contact, cm_default_currency
  from company
  into :cm_check_contact, :default_curr;
  if (cm_check_contact is null) then
    cm_check_contact = 0;

  kol_client = 0;
  kol_contact = 0;

  for select kc_from_c_id, c_name_fio
      from knclient, clients
      where kc_from_k_id = :k_id and
            kc_from_c_id = c_id and
            kc_amount_approved is not null and
            kc_amount_approved <> 0
      into :c_id, :c_name_fio
  do
  begin
    if (exists(select ks.ks_from_k_id
               from kontstate ks
               inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
               where ks.ks_state = 2 and
                     ks.ks_ol_is_delete = 0 and
                     kncl.kc_from_c_id = :c_id and
                     kncl.kc_amount_approved > 0 and
                     current_date between ks.ks_date_begin and ks.ks_date_end)) then
    begin
      is_repeated = 1;
    end
    else
      is_repeated = 0;

    kol_client = kol_client + 1;
    kol_region = 0;
    kol_rodina = 0;
    cc_client_ok = 0;
    cc_name_ok = 0;
    cc_status_ok = 0;
    cc_address_ok = 0;
    is_suspend = 0;

    for select cc_name, cc_status, adl_address_str, cc_cell_no, cc_client_ok, cc_name_ok, cc_status_ok, cc_address_ok,
               cc_type, cc_time_end, cc_try_kol, cc_check_memo, cont.cc_total_res, cc_date, cont.cc_is_old_contact,
               cont.cc_call_user, cont.cc_call_back, cont.cc_create_date, iif(cont.cc_birth_date is null, 0, 1),
               cont.cc_verif_ignore, cont.cc_checked_datetime
        from contacts cont, addrlist
        where cc_from_c_id = :c_id and
              adl_id = cont.cc_from_adl_id
        into :cc_name, :cc_status, :cc_address, :cc_cell_no, :cc_client_ok, :cc_name_ok, :cc_status_ok, :cc_address_ok,
             :cc_type, :cc_time_end, :cc_try_kol, :cc_check_memo, :cc_check_result, :cc_date, :is_old_contact,
             :cc_call_user, :cc_call_back, :cc_create_date, :cc_birth_date_filled, :cc_verif_ignore,
             :cc_checked_datetime
    do
    begin
      if (cc_check_memo is null) then
        cc_check_memo = '';
      if (cc_type = 0) then
      begin
        if (default_curr = '417') then
          cc_type_str = 'KG';
        if (default_curr = '810') then
          cc_type_str = 'РФ';
      end
      else
        cc_type_str = 'Родина';

      if ((cc_check_result = 0) and
          (cc_verif_ignore = 1)) then
        cc_check_result = -999;

      kol_region = 0;
      kol_rodina = 0;
      if ((cc_client_ok = 1) and
          (cc_name_ok = 1) and
          (cc_status_ok = 1) and
          (cc_address_ok = 1) and
          (cc_check_memo = '')) then
      begin
        if (cc_type = 0) then
          kol_region = 1;
        if (cc_type = 1) then
          kol_rodina = 1;
      end
      is_suspend = 1;
      suspend;
    end
    if (is_suspend = 0) then
      suspend;
  end
end
