CREATE PROCEDURE HP_OPERCASH_JOURNAL_LIST(
    OFFICE_ID D_INTEGER,
    REP_DATE_END D_DATE,
    ACCOUNT_DETAILED D_BOOL,
    UN_ID D_ID)
RETURNS (
    OPC_DOC_NO D_INTEGER,
    OPC_OFFICE_ID D_INTEGER,
    OPC_SIGN D_SMALLINT,
    OPC_IS_IPF D_BOOL,
    OPC_CORR_REG_ID D_INTEGER,
    OPC_FROM_OL_ID D_ID,
    RES_CLIENT_FIO D_STR_100,
    RES_AMOUNT_PLUS D_CURRENCY,
    RES_AMOUNT_MINUS D_CURRENCY,
    RES_CASH_ACCOUNT D_STR_20,
    RES_CORR_ACCOUNT D_STR_20,
    RES_ACCOUNT D_STR_255,
    RES_CASH_ACC_NO D_STR_20,
    RES_ACC_BALANCE D_STR_20,
    RES_ACC_NO D_INTEGER)
AS
declare variable is_suspended d_bool;
declare variable ol_id d_id;
declare variable tmp_accounts d_str_255;
declare variable opc_from_k_id d_id;
declare variable opc_corr_al_id d_str_20;
declare variable tmp_client_fio d_str_255;
begin
  if (account_detailed is null) then
    account_detailed = 0;

  for select opc_doc_no, opc_office_id, opc_sign, opc_is_ipf, opc_corr_reg_id, opc_from_ol_id, res_client_fio,
             res_amount_plus, res_amount_minus, res_cash_account, res_corr_account, opc_from_k_id, opc_corr_al_id
  from hp_opercash_office(:office_id, 0, :rep_date_end, 1, :un_id)
  where opc_sign <> 0 and
        opc_amount <> 0
  order by opc_sign desc, opc_doc_no, opc_id
  into opc_doc_no, opc_office_id, opc_sign, opc_is_ipf, opc_corr_reg_id, opc_from_ol_id, tmp_client_fio,
       res_amount_plus, res_amount_minus, res_cash_account, res_corr_account, opc_from_k_id, opc_corr_al_id
  do
  begin
    if (char_length(tmp_client_fio) > 100) then
    begin
      while (position('  ', tmp_client_fio) > 0) do
        tmp_client_fio = replace(tmp_client_fio, '  ', ' ');
    end

    res_client_fio = left(tmp_client_fio, 100);

    res_cash_acc_no = substring(res_cash_account from 7);
    res_account = res_corr_account;

    if (coalesce(opc_corr_al_id, '') = '') then
    begin
      is_suspended = 0;
      ol_id = null;
  
      if (opc_office_id <> opc_corr_reg_id) then
      begin
        select ol_id
        from operlist OL
        where ol_parent_id = :opc_from_ol_id and
              ol_from_reg_id = :opc_corr_reg_id and
              OL.is_delete = 0
        order by ol_id desc
        rows 1
        into ol_id;
      end else
      if (opc_from_k_id is not null and coalesce(opc_corr_al_id, '') = '') then
        ol_id = opc_from_ol_id;
  
      if (ol_id is not null) then
      begin
        if (account_detailed = 1) then -- Каждый лицевой счет раздельно, используется для группировки итогов по счета
        begin
          if (opc_sign = 1) then
          begin
            for select ob_summa, ob_short_acc_kt
            from operbook
            where ob_from_ol_id = :ol_id and
                  ob_summa <> 0
            into res_amount_plus, res_account
            do
            begin
              is_suspended = 1;
  
              res_acc_balance = left(res_account, 5);
              res_acc_no = 0;
              if (position(',', res_account) = 0) then
              begin
                res_acc_no = substring(res_account from 7);
                when any do
                begin
                end
              end
  
              suspend;
            end
          end else
          if (opc_sign = -1) then
          begin
            for select ob_summa, ob_short_acc_dt
            from operbook
            where ob_from_ol_id = :ol_id and
                  ob_summa <> 0
            into res_amount_plus, res_account
            do
            begin
              is_suspended = 1;
  
              res_acc_balance = left(res_account, 5);
              res_acc_no = 0;
              if (position(',', res_account) = 0) then
              begin
                res_acc_no = substring(res_account from 7);
                when any do
                begin
                end
              end
  
              suspend;
            end
          end
        end else
        begin
          -- Лицевые счета в одной строке, используется для отражения в кассовом журнале
          tmp_accounts = '';
          if (opc_sign = 1) then
          begin
            select list(distinct ob_short_acc_kt)
            from operbook
            where ob_from_ol_id = :ol_id and
                  ob_summa <> 0
            into tmp_accounts;
          end else
          if (opc_sign = -1) then
          begin
            select list(distinct ob_short_acc_dt)
            from operbook
            where ob_from_ol_id = :ol_id and
                  ob_summa <> 0
            into tmp_accounts;
          end
  
          if (coalesce(tmp_accounts, '') <> '') then
            res_account = tmp_accounts;
        end
      end
  
      if (is_suspended = 0) then
      begin
        res_acc_balance = left(res_account, 5);
        res_acc_no = 0;
        if (position(',', res_account) = 0) then
        begin
          res_acc_no = substring(res_account from 7);
          when any do
          begin
          end
        end
  
        suspend;
      end
    end else
    begin
      res_acc_balance = left(res_account, 5);
      res_acc_no = 0;
      if (position(',', res_account) = 0) then
      begin
        res_acc_no = substring(res_account from 7);
        when any do
        begin
        end
      end

      suspend;
    end
  end
end
