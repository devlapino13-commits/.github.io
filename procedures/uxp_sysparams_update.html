CREATE PROCEDURE UXP$SYSPARAMS_UPDATE(
    UFP_UFR_UUID D_UUID,
    UFP_UF_UUID D_UUID,
    UFP_IN_USE D_SMALLINT,
    UFP_DEFAULT D_STR_100,
    UFP_LOCKED D_BOOL,
    UFP_DOMAIN D_STR_CODE,
    UFP_EDITOR D_STR_CODE,
    UF_NAME D_STR_100,
    UF_FIELD_NAME D_STR_CODE,
    UF_DEFAULT D_STR_100,
    EDI_SOURCE D_STR_CODE,
    EDI_FIELD_NAME D_STR_CODE,
    EDI_POPUP_MENU D_SMALLINT,
    EDI_INDEX D_INTEGER,
    EDI_IMG_UUID_CODE D_STR_CODE,
    EDI_USE_LAST D_BOOL)
AS
declare variable ufp_uuid d_uuid;
declare variable us_uuid d_uuid;
declare variable uf_type d_smallint;
declare variable edi_img_uuid d_uuid;
begin
  uf_type = 1;
  us_uuid = ufp_ufr_uuid;
  if (ufp_in_use = 1 or ufp_uf_uuid is null) then
  begin
    if (uf_field_name is null) then
      uf_field_name = '';

    if (uf_field_name <> '' and ufp_uf_uuid is null) then
    begin
      if (coalesce(ufp_domain, '') = '') then
        ufp_domain = 'd_integer';

      insert into ux$field (uf_owner, uf_field_name, uf_name, uf_is_param, uf_type, uf_domain, uf_default)
      values (:us_uuid, :uf_field_name, :uf_name, 1, :uf_type, :ufp_domain, :uf_default)
      returning uf_uuid
      into :ufp_uf_uuid;
    end else
    begin
      update ux$field
      set uf_name = :uf_name,
          uf_field_name = :uf_field_name,
          uf_type = :uf_type,
          uf_default = :uf_default
      where uf_uuid = :ufp_uf_uuid
      returning uf_uuid
      into ufp_uf_uuid;
  
      update ux$field
      set uf_domain = :ufp_domain,
          uf_data_type = ''
      where uf_uuid = :ufp_uf_uuid and
            uf_domain is distinct from :ufp_domain;
    end

    update or insert into ux$frparam (ufp_uuid, ufp_ufr_uuid, ufp_uf_uuid, ufp_editor)
    values (:ufp_uf_uuid, :ufp_ufr_uuid, :ufp_uf_uuid, :ufp_editor)
    matching (ufp_ufr_uuid, ufp_uf_uuid)
    returning ufp_uuid
    into ufp_uuid;

    execute procedure uxp$uxparamdef_update(:ufp_uuid, null, :ufp_default, :ufp_locked);

    edi_img_uuid = null;
    if (coalesce(edi_img_uuid_code, '') <> '') then
      edi_img_uuid = char_to_uuid(edi_img_uuid_code);

    update or insert into ux$editor
           (edi_uuid, edi_source, edi_field_name, edi_popup_menu, edi_index, edi_img_uuid, edi_use_last)
    values (:ufp_uuid, :edi_source, :edi_field_name, :edi_popup_menu, :edi_index, :edi_img_uuid, :edi_use_last)
    matching (edi_uuid);

  end else
  if (ufp_in_use = 0) then
  begin
    delete from ux$frparam
    where ufp_ufr_uuid = :ufp_ufr_uuid and
          ufp_uf_uuid = :ufp_uf_uuid;
  end else
  if (ufp_in_use = -1) then
  begin
    delete from ux$frparam
    where ufp_ufr_uuid = :ufp_ufr_uuid and
          ufp_uf_uuid = :ufp_uf_uuid;

    delete from ux$field
    where uf_uuid = :ufp_uf_uuid;
  end
end
