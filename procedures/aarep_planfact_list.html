CREATE PROCEDURE AAREP_PLANFACT_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_AUTO_PERIOD INTEGER,
    REP_AUTO_KOEF DECIMAL(18,7),
    REP_INCLUDE_FIRST_DAY INTEGER)
RETURNS (
    MA_DATE_OPER DATE,
    MA_SUMMA_PLUS_OS DECIMAL(18,4),
    MA_SUMMA_PLUS_PERC DECIMAL(18,4),
    MA_SUMMA_PLUS_ADMIN DECIMAL(18,4),
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_SUMMA_MINUS_REPEAT DECIMAL(18,4),
    MA_SUMMA_MINUS_NEW DECIMAL(18,4),
    MA_SUMMA_MINIAPPNEW DECIMAL(18,4),
    MA_REGION_ID INTEGER,
    MA_FACT_OS DECIMAL(18,4),
    MA_FACT_PERC DECIMAL(18,4),
    MA_FACT_ADMIN DECIMAL(18,4),
    MA_FACT_FINE DECIMAL(18,4),
    MA_FACT_DISB_REPEAT DECIMAL(18,4),
    MA_FACT_DISB_NEW DECIMAL(18,4),
    MA_DELIQ_OS DECIMAL(18,4),
    MA_DELIQ_PERC DECIMAL(18,4),
    MA_ADV_OS DECIMAL(18,4),
    MA_ADV_PERC DECIMAL(18,4),
    MA_TOTAL_OS DECIMAL(18,4),
    MA_TOTAL_PERC DECIMAL(18,4),
    MA_PLAN_DISB_NEW DECIMAL(18,4),
    MA_PLAN_DISB_REPEAT DECIMAL(18,4),
    MA_PLUS_PLAN DECIMAL(18,4),
    MA_PLUS_FACT DECIMAL(18,4),
    MA_MINUS_PLAN DECIMAL(18,4),
    MA_MINUS_FACT DECIMAL(18,4),
    MA_DIFF_SUMMA DECIMAL(18,4),
    MA_OFFICE_NAME VARCHAR(100),
    MA_BRANCH_NAME VARCHAR(100),
    MA_HQ_NAME VARCHAR(100))
AS
declare variable res_hq integer;
begin

  for select ma_date_oper, ma_region_id, sum(ma_summa_plus_os), sum(ma_summa_plus_perc), sum(ma_summa_plus_admin),
             sum(ma_summa_minus), sum(ma_summa_minus_repeat), sum(ma_summa_minus_new), sum(ma_summa_miniappnew),
             sum(-1 * ma_fact_os), sum(-1 * ma_fact_perc), sum(-1 * ma_fact_admin), sum(-1 * ma_fact_fine),
             sum(ma_fact_disb_repeat), sum(ma_fact_disb_new), sum(-1 * ma_deliq_os), sum(-1 * ma_deliq_perc),
             sum(-1 * ma_adv_os), sum(-1 * ma_adv_perc), sum(-1 * ma_total_os), sum(-1 * ma_total_perc),
             r1.reg_code || ' ' || r1.reg_name
      from aarep_planfact_temp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_auto_period, :rep_auto_koef, :rep_include_first_day), region r1
      where ma_date_oper >= :rep_date_begin and
            ma_region_id = r1.reg_unicode
      group by ma_date_oper, ma_region_id, r1.reg_code, r1.reg_name
      order by ma_date_oper, ma_region_id
      into :ma_date_oper, :ma_region_id, :ma_summa_plus_os, :ma_summa_plus_perc, :ma_summa_plus_admin, :ma_summa_minus,
           :ma_summa_minus_repeat, :ma_summa_minus_new, :ma_summa_miniappnew, :ma_fact_os, :ma_fact_perc,
           :ma_fact_admin, :ma_fact_fine, :ma_fact_disb_repeat, :ma_fact_disb_new, :ma_deliq_os, :ma_deliq_perc,
           :ma_adv_os, :ma_adv_perc, :ma_total_os, :ma_total_perc, :ma_office_name
  do
  begin
    ma_plan_disb_new = :ma_summa_minus_new + :ma_summa_miniappnew;
    ma_plan_disb_repeat = :ma_summa_minus + :ma_summa_minus_repeat;

    ma_plus_plan = :ma_summa_plus_os + :ma_summa_plus_perc + :ma_summa_plus_admin;

    ma_plus_fact = :ma_total_os + :ma_total_perc + :ma_fact_admin + :ma_fact_fine;
    ma_minus_plan = :ma_summa_minus + :ma_summa_minus_repeat + :ma_summa_minus_new + :ma_summa_miniappnew;
    ma_minus_fact = :ma_fact_disb_repeat + :ma_fact_disb_new;
    ma_diff_summa = (:ma_minus_plan - :ma_plus_plan) - (:ma_minus_fact - :ma_plus_fact);

    select r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name, r3.reg_unicode
    from region r1, region r2, region r3
    where r1.reg_unicode = :ma_region_id and
          r2.reg_unicode = r1.parent and
          r3.reg_unicode = r2.parent
    into :ma_branch_name, :ma_hq_name, :res_hq;
    if (res_hq = 1000000) then
    begin
      ma_hq_name = ma_branch_name;
    end
    suspend;
  end

end
