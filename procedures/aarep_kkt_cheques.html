CREATE PROCEDURE AAREP_KKT_CHEQUES(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    FOR_FISC_DATE D_BOOL = 0,
    INCLUDE_DUBLICATES D_BOOL = 0)
RETURNS (
    REC_NO D_INTEGER,
    KC_ID ID_TYPE,
    K_KONTRACT_NO STR_20,
    REP_OL_ID D_STR_CODE,
    OL_DATE D_DATE,
    OL_DATETIME D_DATETIME,
    OL_MEMO D_STR_LONG,
    KKM_ID D_ID,
    KKM_STATUS D_STR_CODE,
    KKM_FD D_INTEGER,
    KKM_FPD D_ID,
    KKF_CODE D_STR_20,
    KKM_SUMM D_CURRENCY,
    KKM_CREATE_DATETIME D_DATETIME,
    KKM_SEND_DATETIME D_DATETIME,
    KKM_FISC_DATETIME D_DATETIME,
    KKM_IS_RETURN D_STR_10,
    OFD_LINK D_STR_255,
    CORR_OL_ID_STR D_STR_CODE,
    CORR_OL_DATE D_DATE,
    REG_NAME D_STR_CODE,
    IS_DUBLICATE D_BOOL,
    KKM_SEND_TO D_STR_255)
AS
declare variable k_id d_id;
declare variable ol_id d_id;
declare variable corr_ol_id d_id;
declare variable k_clients_cashier_id d_id;
declare variable kkd_id d_id;
declare variable kkf_id d_id;
begin
  rec_no=0;
  if (for_fisc_date is null) then
    for_fisc_date = 0;
  if (include_dublicates is null) then
    include_dublicates = 0;

  for with t1 (id) as (
    select kkm_id
    from operlist, kkt_main
    where :for_fisc_date = 0
      and kkm_from_ol_id = ol_id
      and ol_date between :rep_date_begin and :rep_date_end
    union
    select kkm_id
    from kkt_main
    where :for_fisc_date = 1
      and kkm_fisc_datetime between :rep_date_begin and cast(:rep_date_end as timestamp) + 0.999999
  )
  select distinct k_id, k_kontract_no, k_clients_cashier_id, ol_id, ol_date, ol_datetime, ol_memo, kkm_corr_ol_id,
         kkm_id, kkm_fd, kkm_fpd, kkm_create_datetime, kkm_send_datetime, kkm_fisc_datetime, kkm_from_kkf_id, kkm_send_to,
         iif(kkm_duplicated_kkm_id is null, 0, 1),
         iif(kkm_is_return = 1, 'Да', 'Нет'),
         case kkm_status
            when -2 then 'Ошибка'
            when -1 then 'Ошибка временная'
            when  0 then 'Новый'
            when  1 then 'Отправлен'
            when  2 then 'Обработан'
            else 'Неизветно'
         end
  from t1, kkt_main, operlist, kontract
  where kkm_id = id
    and ol_id = kkm_from_ol_id
    and k_id = kkm_from_k_id
    and (:include_dublicates = 1 or kkm_duplicated_kkm_id is null)
  into k_id, k_kontract_no, k_clients_cashier_id, ol_id, ol_date, ol_datetime, ol_memo, corr_ol_id,
       kkm_id, kkm_fd, kkm_fpd, kkm_create_datetime, kkm_send_datetime, kkm_fisc_datetime, kkf_id, kkm_send_to, is_dublicate,
       kkm_is_return, kkm_status
  do
  begin
    kc_id = null;
    corr_ol_date = null;
    corr_ol_id_str = null;
    rep_ol_id = ol_id;
    kkm_summ = 0;
    kkf_code = null;
    kkd_id = null;
    ofd_link = null;
    reg_name = null;

    select sum(kki_amount)
    from kkt_items
    where kki_from_kkm_id = :kkm_id
    into kkm_summ;

    select kkf_code, kkf_from_kkd_id
    from kkt_fn
    where kkf_id = :kkf_id
    into kkf_code, kkd_id;


    if (corr_ol_id is not null) then
    begin
      corr_ol_id_str = corr_ol_id;

      select ol_date
      from operlist
      where ol_id = :corr_ol_id
      into corr_ol_date;
    end 

    select kc_id
    from knclient
    where kc_from_k_id = :k_id
      and kc_from_c_id = :k_clients_cashier_id
    rows 1
    into :kc_id;

    if (kc_id is null) then
      select min(kc_id)
      from knclient
      where kc_from_k_id = :k_id
      into :kc_id;

    select reg_name from kkt_devices, region
    where kkd_id = :kkd_id
      and reg_unicode = kkd_from_reg_id
    into reg_name;

    select link
    from hp_kkt_get_ofd_link(:kkm_id)
    into ofd_link;

    rec_no = rec_no + 1;

    suspend;
  end
end
