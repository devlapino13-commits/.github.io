CREATE PROCEDURE FM_COM_BLOB_SET(
    CML_ID D_ID,
    BLS_CAPTION D_STR_100,
    BLS_FILENAME D_STR_100,
    BLS_BLOB D_BLOB_BINARY,
    BLS_KIND D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    ERR_TEXT D_STR_100)
AS
declare variable bls_id d_id;
declare variable lang d_str_lang;
begin
  err_text = '';
  lang = f_get_lang_code(un_id, null);

  if (bls_blob is null) then
    err_text = f_lang(lang, 'Файл не передан. Сохранение невозможно.');
  else if (not exists(select * from com_list where cml_id = :cml_id)) then
    err_text = f_format(f_lang(lang, 'Неверный идентификатор запроса, cml_id: %1'), coalesce(cml_id, 'NULL'));
  else
  begin
    -- Вставка блоба
    execute procedure hp_insert_into_blobstore (:bls_filename, :bls_caption, :bls_blob,
      :un_id, :bls_kind, 1)
    returning_values :bls_id;
  
    insert into com_blob (cmb_from_cml_id, cmb_from_bls_id)
    values (:cml_id, :bls_id);
  end
  suspend;
end
