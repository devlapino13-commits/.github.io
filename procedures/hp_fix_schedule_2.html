CREATE PROCEDURE HP_FIX_SCHEDULE_2(
    K_ID ID_TYPE,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    OL_DATE D_DATE,
    OL_DOC_CLIENT_NAME STR_255,
    OL_DOC_NO_STR STR_20,
    OPL_NAME STR_255,
    ERR_TEXT D_STR_8100,
    REPAY_MEMO D_STR_100,
    OL_ID ID_TYPE)
AS
declare variable OL_OPER_TYPE D_SMALLINT;
declare variable OL_OPER_SUBTYPE D_SMALLINT;
declare variable ESM_ID ID_TYPE;
declare variable OL_MEMO varchar(255);
begin
--  update company set cm_loan_history_action = 2;

  delete from temp_fix_schedule
  where 1 = 1;

  for
  with tabl1 as (
    select distinct ol_date, coalesce(ol_parent_id, ol_id) ol_id, ol_doc_client_name, ol_doc_no_str, ol_memo, ol_oper_type, ol_oper_subtype
    from operkred, operlist
    where ok_from_k_id = :k_id
    and ok_ol_is_active = 1
    and ok_ol_date between :rep_date_begin and :rep_date_end
    and ok_from_ol_id = ol_id
    union
    select kd_ol_date, kd_from_ol_id, null, null, null, 1, 12
    from kontdata
    where kd_from_k_id = :k_id
    and kd_ol_date between :rep_date_begin and :rep_date_end)
  select ol_date, ol_id, max(ol_doc_client_name), max(ol_doc_no_str), max(ol_memo), max(ol_oper_type), max(ol_oper_subtype)
  from tabl1
  group by ol_date, ol_id
  into ol_date, ol_id, ol_doc_client_name, ol_doc_no_str, ol_memo, ol_oper_type, ol_oper_subtype
  do
  begin
    esm_id = null;

    if (ol_oper_subtype = 2) then
    begin
      select esm_id from extsysmain
      where esm_from_k_id = :k_id
      and esm_date = :ol_date
      and esm_action = 'pay'
      and esm_trans_id = :ol_doc_no_str
      into esm_id;
    end

    insert into temp_fix_schedule (ol_date, ol_doc_client_name, ol_doc_no_str, ol_oper_type, ol_oper_subtype, esm_id, ol_id)
    values (:ol_date, :ol_doc_client_name, :ol_doc_no_str, :ol_oper_type, :ol_oper_subtype, :esm_id, :ol_id);

--    if (ol_doc_client_name in ('Погашение через moneta.', 'Погашение через korona.') or (ol_oper_subtype = 12)) then
--    if ((esm_id is not null) or (ol_oper_subtype = 12)) then
    if (ol_oper_subtype = 12) then
    update operlist set is_delete = 1
    where ol_id = :ol_id;
  end

  for
  select ol_date, ol_doc_client_name, ol_doc_no_str, ol_oper_type, ol_oper_subtype, esm_id, ol_id
  from temp_fix_schedule
  order by ol_date
  into :ol_date, :ol_doc_client_name, :ol_doc_no_str, :ol_oper_type, :ol_oper_subtype, :esm_id, ol_id
  do
  begin

    opl_name = 0;
    select opl_name from optlist
    where opl_main = :ol_oper_type
    and opl_detail = :ol_oper_subtype
    into opl_name;
    
    if (ol_oper_subtype = 12) then
    begin
      execute procedure hp_do_reschedule_float(:k_id, :ol_date, 1000000000000);
    end

    err_text = null;
    repay_memo = null;
    if (ol_oper_subtype = 2) then -- обрабатывает погашении
    begin
--      if (ol_doc_client_name in ('Погашение через moneta.', 'Погашение через korona.')) then
      execute procedure hp_do_reschedule_float(:k_id, :ol_date, 1000000000000);

      execute procedure hp_repeat_repayment_current(:k_id, :ol_date, 1000000000000);
      if (esm_id is not null) then
      begin
        /*
        update extsysmain set esm_status = 0 -- перепринять погашене
        where esm_id = :esm_id;

        select err_text from hp_extsys_process_one(:esm_id)
        into err_text;
        */
        repay_memo = 'погашение через платежную систему';
      end else
      begin -- погашение через кассу
        repay_memo = 'погашение через кассу';
--        execute procedure hp_repeat_repayment_tmp(null, :k_id, :ol_date, :ol_date, 1, 2,  'empty', 1000000000000);
      end 
    end

    suspend;
  end
  if (exists(
  select * from kontstate
  where ks_ol_is_delete = 0
  and ks_state = 1
  and ks_from_k_id = :k_id
  and :rep_date_end between ks_date_begin and ks_date_end)) then
  begin
    execute procedure hp_do_reschedule_float(:k_id, :rep_date_end, 1000000000000);
    execute procedure hp_do_reschedule_float(:k_id, :rep_date_end, 1000000000000);
    execute procedure hp_do_reschedule_float(:k_id, :Rep_Date_End, 1000000000000);
  end
--  update company set cm_loan_history_action = 0;
end
