CREATE PROCEDURE FM_LOANS_CLOSED(
    UN_ID DECIMAL(18,0),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REG_ID INTEGER)
RETURNS (
    K_ID DECIMAL(18,0),
    KF_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    REPEAT_STATUS VARCHAR(100),
    CLOSE_DATE VARCHAR(20),
    CLIENT_ID DECIMAL(18,0))
AS
declare variable C_ID decimal(18,0);
declare variable CL_DATE date;
declare variable REGIONS_STR D_STR_4096;
begin
  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  if (c_id is null) then
    exit;

  if (not exists(select ps_from_reg_id
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :c_id and
                       ps_date_begin <= current_date and
                       ps_date_end >= current_date and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior') and
                       ps_is_working = 1)) then
    reg_id = -1;

  regions_str = null;

  if (reg_id <> -1) then
  begin
    regions_str = '.';

    for select id_list
        from region_list(:reg_id, 1, 3, 1)
        into :reg_id
    do
        regions_str = regions_str || :reg_id || '.' ;
  end
  else
    regions_str = '.-1.';

  for select value_out
  from str_with_separator_parser(:regions_str, '.')
  into :reg_id
  do
  begin
      for select res_c_id
          from hp_staff_changer_list(:un_id, 1)
          where :reg_id = -1
          union
          select res_c_id
          from hp_staff_changer_list(:un_id, -1)
          where :reg_id <> -1
          into :c_id
      do
      begin
        for select k_id, cl_date, kf_name
        from fm_loans_closed_tmp(:c_id, :rep_date_begin, :rep_date_end, :reg_id)
        into :k_id, :cl_date, :kf_name
        do
        begin
          close_date = iif(cl_date is not null, right('0' || extract(day from cl_date), 2) || '.' || right('0' || extract(month from cl_date), 2) || '.' || extract(year from cl_date), 'Не указан');
    
          for select kc_from_c_id, c_name_fio
              from knclient, clients
              where kc_from_c_id = c_id and
                    kc_from_k_id = :k_id and
                    kc_amount_approved > 0
              into :client_id, :c_name_fio
          do
          begin
            repeat_status = '';
            if (exists(select *
                       from kontstate, knclient
                       where kc_from_c_id = :client_id and
                             kc_amount_approved > 0 and
                             kc_from_k_id = ks_from_k_id and
                             ks_state = 1 and
                             ks_ol_is_delete = 0 and
                             ks_date_begin >= :cl_date)) then
              repeat_status = 'выдан';

            suspend;
          end
        end
     end
  end
end
