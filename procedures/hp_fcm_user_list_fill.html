CREATE PROCEDURE HP_FCM_USER_LIST_FILL(
    CMD_LIST D_STR_255)
AS
declare variable cmd d_str_code;
declare variable ful_id d_id;
declare variable fcm_id d_str_255;
begin
  -- Сервис: FsSvc
  -- Заполняем fcm_user_list пользователями, для которых будут
  -- синхронизироваться топики в FsService
  -- Допустимые значения команд в списке cmd_list:
  --   clear_username - очистка списка из username
  --   clear_mc_login - очистка списка из mc_login
  --   clear_all - полная очистка таблицы fcm_user_list
  --   add_username - добавить пользователей из username
  --   add_mc_login - добавить пользователей из mc_login

  for select res_str
  from hp_split_str(:cmd_list, ',')
  into cmd
  do
  begin
    if (cmd = 'clear_username') then
    begin
      delete from fcm_user_list
      where ful_un_id is not null;
    end
    else if (cmd = 'clear_mc_login') then
    begin
      delete from fcm_user_list
      where ful_mcl_id is not null;
    end
    else if (cmd = 'clear_all') then
    begin
      delete from fcm_user_list
      where 1=1;
    end
    else if (cmd = 'add_username') then
    begin
      -- Добавляем отсутствующие в fcm_user_list, но присутствующие в username
      insert into fcm_user_list (ful_un_id, ful_mcl_id, ful_fcm_id)
      select un_id, null, un_google_id
      from username left join fcm_user_list on ful_un_id = un_id
      where coalesce(un_google_id, '') <> ''
        and ful_un_id is null;

      -- Обновляем fcm_id в fcm_user_list, если поменялся google_id в username
      -- Удаляем из fcm_user_list, если google_id очищен в username
      for select ful_id, un_google_id
      from fcm_user_list left join username on ful_un_id = un_id
      where un_google_id is distinct from ful_fcm_id
        and ful_mcl_id is null
      into ful_id, fcm_id
      do
      begin
        if (coalesce(fcm_id, '') <> '') then
        begin
          update fcm_user_list
          set ful_fcm_id = :fcm_id
          where ful_id = :ful_id;
        end
        else
          execute procedure hp_fcm_user_remove(ful_id);
      end
    end
    else if (cmd = 'add_mc_login') then
    begin
      -- Добавляем отсутствующие в fcm_user_list, но присутствующие в mc_login
      insert into fcm_user_list (ful_un_id, ful_mcl_id, ful_fcm_id)
      select null, mcl_id, mcl_google_id
      from mc_login left join fcm_user_list on ful_mcl_id = mcl_id
      where coalesce(mcl_google_id, '') <> ''
        and mcl_is_delete = 0
        and ful_mcl_id is null;

      -- Обновляем fcm_id в fcm_user_list, если поменялся google_id в mc_login
      -- Удаляем из fcm_user_list, если google_id очищен в mc_login
      for select ful_id, mcl_google_id
      from fcm_user_list left join mc_login on ful_mcl_id = mcl_id
      where mcl_google_id is distinct from ful_fcm_id
        and mcl_is_delete = 0
        and ful_un_id is null
      into ful_id, fcm_id
      do
      begin
        if (coalesce(fcm_id, '') <> '') then
        begin
          update fcm_user_list
          set ful_fcm_id = :fcm_id
          where ful_id = :ful_id;
        end
        else
          execute procedure hp_fcm_user_remove(ful_id);
      end
    end
  end
end
