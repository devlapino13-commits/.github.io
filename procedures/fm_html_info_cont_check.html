CREATE PROCEDURE FM_HTML_INFO_CONT_CHECK(
    C_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable rd_id d_id;
declare variable ver_status d_smallint;
declare variable queue_no d_integer;
declare variable aco_last_id d_id;
declare variable aco_id d_id;
declare variable decline_reason d_str_2048;
declare variable db_id d_integer;
declare variable bc_status d_integer;
begin
  db_id = f_get_db_id();
  ver_status = 2;
  res_text = f_format('%1Статус проверки %2 номера: ',iif(db_id <> 39801, '', '<br>'), iif(db_id <> 39801, 'контактного', 'сотового'));  -- RFCKG-4140
  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ActionGroup.SimpleVerify_Phone_Number_Verify'
  into :rd_id;

  select acl.acl_status
  from actionlist acl
  where acl.acl_object_kind = f_objkind('clients', 'id') and
        acl.acl_object_id = :c_id and
        acl.acl_from_rd_id = :rd_id
  order by acl.acl_datetime desc
  rows 1
  into ver_status;

  if (db_id = 39801) then
  begin
    select bc.bc_status       -- RFCKG-4180  проверка номера телефона
    from bmg_clients bc
    where bc.bc_from_c_id = :c_id
      and bc.bc_is_last = 1
    order by bc.bc_status_datetime
    rows 1
    into bc_status;

    bc_status = coalesce(bc_status, -100);

    res_text = blob_append(res_text, case bc_status
                                       when -2 then '<b style="color: red;">Не подтвержден</b>'
                                       when 1 then '<b style="color: green;">Подтвержден</b>'
                                       when -3 then '<b style="color: red;">Ошибка запроса</b>'
                                       when -4 then '<b style="color: red;">Ошибка запроса</b>'
                                       when -100 then '<b style="color: red;">Нет активных запросов</b>'
                                       else '<b>Запрос в процессе</b>'
                                     end);

    res_text = blob_append(res_text, iif(bc_status = -2, '<br><span style="color: red;">Клиенту необходимо сменить в КМП номер телефона на тот, который привязанный к ИИН</span><br>', ''));
  end
  else
  begin
    res_text = BLOB_APPEND(res_text, '<b>' || case ver_status
                           when -3 then 'Отмена заполнения'
                           when -2 then 'Отказан'
                           when -1 then 'На доработку'
                           when 0 then 'В ожидании'
                           when 1 then 'Принят на обработку'
                           else 'Подтвержден'
                         end || '</b>');
  end

  if (ver_status = 0) then
  begin
    queue_no = 0;
    for select aco_id
        from actionorder
        where aco_object_kind = 0 and
              aco_object_id = :c_id and
              aco_from_rd_id = :rd_id
        order by aco_id desc
        rows 1
        into :aco_last_id
    do
    begin
      for select aco_id
          from actionorder
          where aco_from_rd_id = :rd_id and
                aco_status = 0
          order by aco_date
          into aco_id
      do
      begin
        if (aco_id <= aco_last_id) then
          queue_no = queue_no + 1;
      end
    end
    res_text = BLOB_APPEND(res_text, '<b> № ' || queue_no || '</b>');
  end

  if (ver_status = -2) then
  begin
    decline_reason = '';
    select f_get_verify_memo(obm_memo, 'reason_memo')
    from objectprops
    inner join objectmemo on obm_from_opr_id = opr_id
    where opr_object_kind = 0 and
          opr_object_id = :c_id and
          opr_subject_id = :rd_id
    order by obm_datetime desc
    rows 1
    into decline_reason;

    decline_reason = coalesce(decline_reason,'');

    res_text = BLOB_APPEND(res_text, '<br>' || '<b>Причина отказа: </b>' || decline_reason || '</br>');

    res_text = BLOB_APPEND(res_text, '<br>' || f_uri_sql('Переотправить на верификацию', 'resend_to_verify','clients', :c_id, 'text', 'Наберите комбинацию', '', 4, 255, '', '', 'dialog', '') || '</br>');
  end

  suspend;
end
