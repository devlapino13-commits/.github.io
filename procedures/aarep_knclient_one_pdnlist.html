CREATE PROCEDURE AAREP_KNCLIENT_ONE_PDNLIST(
    REP_KC_ID D_ID)
RETURNS (
    REG_NAME D_STR_100,
    LINE D_INTEGER,
    NEW_KONTRACT_NO D_STR_CODE,
    NEW_KC_ID D_ID,
    REC_NO D_INTEGER,
    CREDID D_STR_CODE,
    CREDCURRENCY D_STR_10,
    CREDDATE D_DATE,
    CREDENDDATE D_DATE,
    CREDENDDATEFACT D_DATE,
    CREDUPDATE D_DATE,
    CREDSUMDEBT D_CURRENCY,
    CREDSUMOVERDUE D_CURRENCY,
    CREDDAYOVERDUE D_CURRENCY,
    CREDFULLCOST D_CURRENCY,
    CREDSUM D_CURRENCY,
    CREDPARTNERTYPE_STR D_STR_255,
    CREDTYPE_STR D_STR_255,
    CREDACTIVE_STR D_STR_255,
    CALC_KOEF_PERIOD D_CURRENCY,
    CALC_PERIOD_MONTH D_CURRENCY,
    CALC_MONTH_REPAY D_CURRENCY,
    CALC_IS_CREDIT_CARD D_SMALLINT,
    CALC_EIL_DATE D_DATE,
    CALC_STATUS D_STR_255,
    CALC_IN_AVG D_SMALLINT,
    KC_ID D_ID,
    K_KONTRACT_NO D_STR_CODE,
    AVG_INCOME_DOC D_STR_10,
    AVG_INCOME_BKI D_STR_10,
    KC_STAT_AVG_INCOME D_CURRENCY,
    KC_ANKETA_INCOME D_CURRENCY,
    KC_FINAL_INCOME D_CURRENCY,
    C_NAME_FIO D_STR_100,
    ADL_ADDRESS_STR D_STR_255,
    FINAL_PDN D_CURRENCY,
    RES_LAST_INCOME_PERIOD D_INTEGER,
    RES_INCOME_01 D_CURRENCY,
    RES_INCOME_02 D_CURRENCY,
    RES_INCOME_03 D_CURRENCY,
    RES_INCOME_04 D_CURRENCY,
    RES_INCOME_05 D_CURRENCY,
    RES_INCOME_06 D_CURRENCY,
    RES_INCOME_07 D_CURRENCY,
    RES_INCOME_08 D_CURRENCY,
    RES_INCOME_09 D_CURRENCY,
    RES_INCOME_10 D_CURRENCY,
    RES_INCOME_11 D_CURRENCY,
    RES_INCOME_12 D_CURRENCY,
    EIL_ID D_ID,
    ADL_ID D_ID,
    STR_ID D_INTEGER,
    C_ID D_ID,
    KC_CHECK_LAST_DATE D_DATE,
    KC_CHECK_COUNT D_INTEGER,
    ANKETA_WORK_INCOME D_CURRENCY,
    ANKETA_WORK_PERIOD D_STR_100,
    ANKETA_EXTRA_INCOME D_CURRENCY,
    PORUCH_STR D_STR_LANG,
    ZALOG_STR D_STR_LANG,
    KPL_IS_ZEROPERC D_SMALLINT,
    PR_NAME D_STR_255,
    KOL_ACTIV_DOLG_KLIENT D_INTEGER,
    KOL_ACTIV_ZAIM_KLIENT D_INTEGER,
    KOL_CLOSE_DOLG_KLIENT D_INTEGER,
    KOL_CLOSE_DOLG_ALL D_INTEGER,
    UID D_STR_CODE,
    OL_DATETIME D_DATE,
    K_DISB_DATE DATE)
AS
declare variable KC_STAT_AVG_INCOME_TMP D_CURRENCY;
declare variable KC_ANKETA_INCOME_TMP D_CURRENCY;
declare variable KC_FINAL_INCOME_TMP D_CURRENCY;
declare variable C_NAME_FIO_TMP D_STR_100;
declare variable ADL_ADDRESS_STR_TMP D_STR_255;
declare variable FINAL_PDN_TMP D_CURRENCY;
declare variable RES_LAST_INCOME_PERIOD_TMP D_INTEGER;
declare variable KC_LOAN_EXPENSE D_CURRENCY;
declare variable KC_DATE D_DATE;
declare variable ANKETA_WORK_INCOME_TMP D_CURRENCY;
declare variable ANKETA_WORK_PERIOD_TMP D_STR_100;
declare variable ANKETA_EXTRA_INCOME_TMP D_CURRENCY;
declare variable PR_NAME_TMP D_STR_255;
declare variable PR_CODE D_STR_100;
declare variable K_KONT_LAST_DATE D_DATE;
declare variable REG_ID D_INTEGER;
declare variable K_ID D_ID;
declare variable KCP_ID D_ID;
begin
  line = 0;
  avg_income_doc = 'нет';
  avg_income_bki = 'нет';
  for select k_kontract_no, kc.kc_id, c_name_fio, kpl.kc_from_eil_id, kpl.kc_stat_last_period, kpl.kc_pdn_koef,
             kpl.kc_stat_avg_income, kpl.kc_loan_expense, c_id, kc_check_last_date, kc_check_count, kc_date,
             kpl.kc_anketa_income, kpl.kc_final_income, pr_code, pr_name, k_disb_date, k_kont_last_date, k_id, kcp_id
      from knclient kc, kontract, clients, knclient_pdn kpl, product
      where k_id = kc.kc_from_k_id and
            kc.kc_from_c_id = c_id and
            kc.kc_id = kpl.kc_id and
            k_from_pr_id = pr_id and
            kc.kc_id = :rep_kc_id and
            kcp_kind = 0
      into new_kontract_no, new_kc_id, c_name_fio_tmp, eil_id, res_last_income_period_tmp, final_pdn_tmp,
           kc_stat_avg_income_tmp, kc_loan_expense, c_id, kc_check_last_date, kc_check_count, kc_date,
           kc_anketa_income_tmp, kc_final_income_tmp, pr_code, pr_name_tmp, k_disb_date, k_kont_last_date, k_id, kcp_id
  do
  begin
  --время выдачи
   ol_datetime = null;
    select first 1 ol_datetime
    from operkred, operlist
    where ok_ol_is_active = 1 and
          ok_from_k_id = :k_id and
          ok_type = 8 and
          ol_id = ok_from_ol_id and
          is_delete = 0
    order by ol_datetime desc

    into ol_datetime;

    reg_name = '';
    select oks.oks_region_id
    from oksummary oks
    where oks.oks_from_k_id = :k_id and
          oks_close_state = 0 rows 1
    into reg_id;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;

    select work_income, work_income_period, work_extra_income
    from hp_knclient_anketa_income(:new_kc_id, :kc_date)
    into anketa_work_income_tmp, anketa_work_period_tmp, anketa_extra_income_tmp;

    adl_id = null;
    str_id = null;

    calc_eil_date = null;
    adl_address_str_tmp = '';

    c_name_fio = '';
    k_kontract_no = '';
    adl_address_str = '';
    final_pdn = null;
    res_last_income_period = null;

    kc_stat_avg_income = 0;
    kc_anketa_income = 0;
    kc_final_income = 0;
    res_income_01 = 0;
    res_income_02 = 0;
    res_income_03 = 0;
    res_income_04 = 0;
    res_income_05 = 0;
    res_income_06 = 0;
    res_income_07 = 0;
    res_income_08 = 0;
    res_income_09 = 0;
    res_income_10 = 0;
    res_income_11 = 0;
    res_income_12 = 0;

    select eil_date
    from extinfolist
    where eil_id = :eil_id
    into calc_eil_date;

    select res_last_period, res_income_01, res_income_02, res_income_03, res_income_04, res_income_05, res_income_06,
           res_income_07, res_income_08, res_income_09, res_income_10, res_income_11, res_income_12, str_id, adl_id,
           res_address
    from hp_extinfolist_pdn_one_income(:new_kc_id, :kc_date)
    into :res_last_income_period, :res_income_01, :res_income_02, :res_income_03, :res_income_04, :res_income_05,
         :res_income_06, :res_income_07, :res_income_08, :res_income_09, :res_income_10, :res_income_11, :res_income_12,
         :str_id, :adl_id, :adl_address_str_tmp;

    for select kpl_rec_no, kpl_credid, kpl_credcurrency, kpl_creddate, kpl_credenddate, kpl_credenddatefact,
               kpl_credupdate, kpl_credsumdebt, kpl_credsumoverdue, kpl_creddayoverdue, kpl_credfullcost, kpl_credsum,
               f_memo_text(kpl_credpartnertype_id), f_memo_text(kpl_credtype_id), f_memo_text(kpl_credactive_id),
               kpl_koef_period, kpl_period_month, kpl_month_repay, kpl_is_credit_card, f_memo_text(kpl_status_id),
               kpl_in_avg, kpl_history_kc_id, iif(kpl_poruch = 1, 'Да', 'Нет'),
               iif(kpl_zalog = 1, 'Да', 'Нет'), kpl_is_zeroperc
        from knclient_pdnlist
        where kpl_from_kcp_id = :kcp_id
        order by kpl_rec_no
        into rec_no, credid, credcurrency, creddate, credenddate, credenddatefact, credupdate, credsumdebt,
             credsumoverdue, creddayoverdue, credfullcost, credsum, credpartnertype_str, credtype_str, credactive_str,
             calc_koef_period, calc_period_month, calc_month_repay, calc_is_credit_card, calc_status, calc_in_avg,
             kc_id, poruch_str, zalog_str, kpl_is_zeroperc
    do
    begin
      c_name_fio = '';
      k_kontract_no = '';
      adl_address_str = '';
      final_pdn = null;
      pr_name = '';

      if (rec_no = 0) then
      begin
        credid = 'Новый';
        c_name_fio = c_name_fio_tmp;
        adl_address_str = adl_address_str_tmp;
        res_last_income_period = res_last_income_period_tmp;
        creddate = k_disb_date;
        credenddate = k_kont_last_date;
        credupdate = kc_date;
        kc_stat_avg_income = kc_stat_avg_income_tmp;
        kc_anketa_income = kc_anketa_income_tmp;
        kc_final_income = kc_final_income_tmp;
        c_name_fio = c_name_fio_tmp;
        k_kontract_no = new_kontract_no;
        final_pdn = final_pdn_tmp;
        anketa_work_income = anketa_work_income_tmp;
        anketa_work_period = anketa_work_period_tmp;
        anketa_extra_income = anketa_extra_income_tmp;

        if (pr_code containing '.zalog_car.' or pr_code containing '.zalog.') then
          zalog_str = 'Да';
        else
          zalog_str = 'Нет';

        pr_name = pr_name_tmp;
      end
      else
      begin
        str_id = null;
        adl_id = null;
        res_last_income_period = null;

        kc_stat_avg_income = 0;
        kc_anketa_income = 0;
        kc_final_income = 0;
        res_income_01 = 0;
        res_income_02 = 0;
        res_income_03 = 0;
        res_income_04 = 0;
        res_income_05 = 0;
        res_income_06 = 0;
        res_income_07 = 0;
        res_income_08 = 0;
        res_income_09 = 0;
        res_income_10 = 0;
        res_income_11 = 0;
        res_income_12 = 0;
        anketa_work_income = 0;
        anketa_work_period = '';
        anketa_extra_income = 0;
      end

      if (kc_id is not null) then
      begin
        select k_kontract_no, pr_name
        from kontract, knclient, product
        where kc_id = :kc_id and
              kc_from_k_id = k_id and
              k_from_pr_id = pr_id
        into k_kontract_no, pr_name;
      end

      line = line + 1;
      suspend;
      kc_id = null;
    end

    -- Итоговая часть
    rec_no = -1;
    credid = 'Итог';
    creddate = null;
    credenddate = null;
    credenddatefact = null;
    credupdate = null;
    credsumdebt = null;
    credsumoverdue = null;
    creddayoverdue = null;
    credfullcost = null;
    credsum = null;
    credpartnertype_str = 'Итог';
    credtype_str = '';
    credactive_str = '';
    poruch_str = '';
    zalog_str = '';
    calc_koef_period = null;
    calc_period_month = null;
    calc_is_credit_card = null;
    calc_in_avg = 0;
    kpl_is_zeroperc = 0;
    pr_name = pr_name_tmp;

    select kpl_credcurrency, kpl_credenddate, kpl_credenddatefact, kpl_credsumdebt, kpl_credsumoverdue,
           kpl_creddayoverdue, kpl_credfullcost, kpl_credsum, f_memo_text(kpl_credpartnertype_id),
           f_memo_text(kpl_credtype_id), f_memo_text(kpl_credactive_id), kpl_koef_period, kpl_period_month,
           kpl_is_credit_card, f_memo_text(kpl_status_id), kpl_in_avg, iif(kpl_poruch = 1, 'Да', 'Нет'),
           iif(kpl_zalog = 1, 'Да', 'Нет'), kpl_is_zeroperc
    from knclient_pdnlist
    where kpl_from_kcp_id = :kcp_id
    order by kpl_rec_no
    rows 1
    into credcurrency, credenddate, credenddatefact, credsumdebt, credsumoverdue, creddayoverdue, credfullcost, credsum,
         credpartnertype_str, credtype_str, credactive_str, calc_koef_period, calc_period_month, calc_is_credit_card,
         calc_status, calc_in_avg, poruch_str, zalog_str, kpl_is_zeroperc;

    --1. «Кол-во актив дог-ров в КИ клиента» 2. «Кол-во актив займов от М Булак в КИ клиента» 3.«Кол-во закрытых займов от М Булак в КИ клиента»
    kol_activ_dolg_klient = 0;
    kol_activ_zaim_klient = 0;
    kol_close_dolg_klient = 0;
    kol_close_dolg_all = 0;
    select count(
           case
             when f_memo_text(kpl_credactive_id) <> 'договор закрыт' then kpl_credactive_id
             else null
           end) kol_activ_dolg_klient,
           count(
           case
             when f_memo_text(kpl_credpartnertype_id) = 'Мол Булак' and f_memo_text(kpl_credactive_id) <> 'договор закрыт' then kpl_credpartnertype_id
             else null
           end) kol_activ_zaim_klient,
           count(
           case
             when f_memo_text(kpl_credpartnertype_id) = 'Мол Булак' and f_memo_text(kpl_credactive_id) = 'договор закрыт' then kpl_credpartnertype_id
             else null
           end) kol_close_dolg_klient,
           count(
           case
             when f_memo_text(kpl_credactive_id) = 'договор закрыт' then kpl_credpartnertype_id
             else null
           end) kol_close_dolg_all
    from knclient_pdnlist
    where kpl_from_kcp_id = :kcp_id and
          kpl_rec_no <> 0
    into kol_activ_dolg_klient, kol_activ_zaim_klient, kol_close_dolg_klient, kol_close_dolg_all;

    if (pr_code containing '.zalog_car.' or pr_code containing '.zalog.') then
      zalog_str = 'Да';
    else
      zalog_str = 'Нет';

    creddate = k_disb_date;
    credenddate = k_kont_last_date;
    credupdate = kc_date;
    kc_id = new_kc_id;
    adl_address_str = adl_address_str_tmp;
    calc_month_repay = kc_loan_expense;
    kc_stat_avg_income = kc_stat_avg_income_tmp;
    kc_anketa_income = kc_anketa_income_tmp;
    kc_final_income = kc_final_income_tmp;
    c_name_fio = c_name_fio_tmp;
    k_kontract_no = new_kontract_no;
    final_pdn = final_pdn_tmp;
    res_last_income_period = res_last_income_period_tmp;
    anketa_work_income = anketa_work_income_tmp;
    anketa_work_period = anketa_work_period_tmp;
    anketa_extra_income = anketa_extra_income_tmp;

    select res_last_period, res_income_01, res_income_02, res_income_03, res_income_04, res_income_05, res_income_06,
           res_income_07, res_income_08, res_income_09, res_income_10, res_income_11, res_income_12, str_id, adl_id
    from hp_extinfolist_pdn_one_income(:new_kc_id, :kc_date)
    into :res_last_income_period, :res_income_01, :res_income_02, :res_income_03, :res_income_04, :res_income_05,
         :res_income_06, :res_income_07, :res_income_08, :res_income_09, :res_income_10, :res_income_11, :res_income_12,
         :str_id, :adl_id;
    uid = '';
    uid = f_uuid_cbr(f_uuid('knclient', :new_kc_id));
    line = line + 1;
    suspend;
  end
end
