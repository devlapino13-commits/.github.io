CREATE PROCEDURE AAREP_KONTRACT_LIST_BUX(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    K_ID VARCHAR(18),
    KF_REGION_NAME D_STR_100,
    K_KONTRACT_NO VARCHAR(20),
    CL_CASHIER_NAME VARCHAR(100),
    K_DISB_DATE DATE,
    KC_ID ID_TYPE,
    K_AMOUNT_APPROVED CURR_TYPE,
    PR_NAME VARCHAR(50),
    K_KREDLINE_END D_DATE,
    TERM_IN_DAYS D_INTEGER,
    PERCENT_YEAR PERC_TYPE,
    PERCENT_EFFECT PERC_TYPE,
    CRED_FULL_COST DECIMAL(15,3),
    LOAN_EMP INTEGER,
    OKS_IS_REFUND_LLR_STR D_STR_10,
    ACC_NO D_STR_10,
    FINAL_PDN CURR_TYPE,
    STR_ID D_STR_100,
    K_FACT_DISB_ONLINE D_INTEGER,
    C_BIRTH_DATE D_DATE,
    OL_DATETIME D_TIME,
    KC_CIRCLE D_INTEGER,
    C_ID_N D_STR_100,
    UID D_STR_CODE,
    CLI_IS_RESIDENT_ON_DISB D_INTEGER,
    PR_CODE D_STR_100)
AS
declare variable max_sh_date d_date;
declare variable oks_is_refund_llr d_integer;
declare variable knr_non_resident d_smallint;
declare variable kc_date d_date;
declare variable c_id id_type;
begin

  for select ak.aarep_k_id, k_kontract_no, k_disb_date, k_amount_approved, pr_name, k_kredline_end,
             cast(kon.k_percent_effect_fifd as decimal(15,3)) as cred_full_cost,
             case
               when pr_product_kind = 4 then 0
               else 1
             end loan_emp,
             k_fact_disb_online, c3.c_name_fio, c3.c_birth_date, r1.reg_code || ' ' || r1.reg_name, k_clients_cashier_id, pr_code

      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 2) ak
      left join kontract kon on ak.aarep_k_id = kon.k_id
      left join product pr on kon.k_from_pr_id = pr.pr_id
      left join knaktlist akt on (ak.aarep_kf_id = akt.kal_from_kf_id and
            akt.kal_date <= ak.aarep_date_action and
            akt.kal_date_end >= ak.aarep_date_action and
            akt.is_delete = 0)
      left join clients c3 on kon.k_clients_cashier_id = c3.c_id
      left join region r1 on akt.kal_new_reg_id = r1.reg_unicode
      into :k_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :pr_name, :k_kredline_end, :cred_full_cost,
           :loan_emp, :k_fact_disb_online, :cl_cashier_name, :c_birth_date, :kf_region_name, c_id, pr_code
  do
  begin
    c_id_n = 'n' || :c_id;

    kc_id = null;
    kc_circle = null;
    cli_is_resident_on_disb = null;

    select distinct kc_id, kc_circle, 1 - coalesce(kc_non_resident, 0)
    from knclient
    where kc_from_k_id = :k_id
    into kc_id, kc_circle, cli_is_resident_on_disb;

    max_sh_date = null;
    term_in_days = 0;
    select max(sh_date)
    from schedule
    where sh_from_k_id = :k_id
    into max_sh_date;

    if (max_sh_date is not null) then
      term_in_days = max_sh_date - k_disb_date;
    else
      term_in_days = null;

    percent_year = null;
    select kd_percent_per_day * 365
    from kontdata
    where :k_disb_date between kd_ol_date and kd_date_end and
          kd_ol_is_delete = 0 and
          kd_from_k_id = :k_id
    into percent_year;
    percent_effect = 0;
    oks_is_refund_llr = 0;
    select first 1 oks_percent_esp, oks_is_refund_llr
    from kn_oksummary
    where oks_from_kc_id = :kc_id and
          :report_date_end between oks_date_begin and oks_date_end
    into :percent_effect, :oks_is_refund_llr;

    if (oks_is_refund_llr = 1) then
      oks_is_refund_llr_str = 'Да';
    else
      oks_is_refund_llr_str = 'Нет';

    knr_non_resident = 1 - cli_is_resident_on_disb;
    select knr_non_resident
    from kn_resident
    where knr_from_kc_id = :kc_id and
          :report_date_end between knr_date_begin and knr_date_end
    into knr_non_resident;

    acc_no = null;
    if (knr_non_resident = 1) then
      acc_no = '492';
    else
    if (knr_non_resident = 0) then
      acc_no = '488';

    final_pdn = null;
    kc_date = null;
    str_id = null;
    select kc_pdn_koef, kc_date
    from knclient_pdn
    where kc_id = :kc_id and
          kcp_kind = 0
    into final_pdn, :kc_date;

    if (:kc_date is not null) then
    begin
      select str_id
      from hp_extinfolist_pdn_one_income(:kc_id, :kc_date)
      into :str_id;
    end
    ol_datetime = null;
    ol_datetime = null;
    select first 1 ol_datetime
    from operkred, operlist
    where ok_ol_is_active = 1 and
          ok_from_k_id = :k_id and
          ok_type = 8 and
          ol_id = ok_from_ol_id and
          is_delete = 0
    order by ol_datetime desc

    into ol_datetime;

    uid = '';
    uid = f_uuid_cbr(f_uuid('knclient', :kc_id));

    /*ol_datetime = null;
    select first 1 ol_datetime
    from operlist
    where ol_id = :ok_from_ol_id and
          is_delete = 0
    order by ol_datetime desc
    into ol_datetime; */

    suspend;
  end ----end of 'first for"
end
