CREATE PROCEDURE HP_KONT_FOLDER_INFO(
    REP_KONTRACT_NO D_STR_20)
RETURNS (
    INFO BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable k_id d_id;
declare variable tmp_k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable k_amount_approved d_currency;
declare variable kf_id d_id;
declare variable kal_new_kf_name d_str_100;
declare variable pr_name d_str_code;
declare variable reg_name d_str_code;
declare variable ks_state_name d_str_20;
declare variable clients_fio d_str_255;
declare variable line d_str_100;
declare variable eol d_str_10;
declare variable i d_integer;
begin

  select k_id,
         k_kontract_no,
         k_from_kf_id,
         k_disb_date,
         k_amount_approved,
         iif(ks_state = 1, 'активный', 'закрыт'),
         pr_name
  from kontract
  inner join kontstate on ks_from_k_id = k_id
  inner join product on pr_id = k_from_pr_id
  where k_kontract_no = :rep_kontract_no and
        ks_ol_is_delete = 0 and
        ks_date_end = '31.12.9999' and
        ks_state in (1, 2)
  into k_id,
       k_kontract_no,
       kf_id,
       k_disb_date,
       k_amount_approved,
       ks_state_name,
       pr_name;

  if (k_id is null) then
    exit;

  if (exists(select oko_from_k_id
             from operkroff
             where oko_from_k_id = :k_id and
                   oko_ol_is_active = 1 and
                   oko_sh_os - oko_os > 0
             group by oko_from_k_id)) then
    ks_state_name = 'спец. учет';

  select kal.kal_new_kf_name,
         left(reg.reg_code || ' ' || reg.reg_name, 50)
  from knaktlist kal
  inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
  where kal.kal_from_kf_id = :kf_id and
        kal.kal_date_end = '31.12.9999' and
        kal.is_delete = 0
  into kal_new_kf_name,
       reg_name;

  select left(list(c_name_fio, ', '), 255)
  from clients
  inner join knclient on kc_from_c_id = c_id
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into clients_fio;

  if (char_length(clients_fio) = 255) then
    clients_fio = left(clients_fio, 252) || '...';

  line = '-----------------------------------------------------------------------------------------------';
  eol = ascii_char(13) || ascii_char(10);
  info = blob_append(info, eol || 'Группа: ' || kal_new_kf_name || ', офис ' || reg_name || eol || eol || line || line || eol || eol);
  info = blob_append(info, 'Договор № ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date) || ' (' || trim(ks_state_name));
  info = blob_append(info, '), продукт "' || pr_name || '", cумма ' || f_numtostr(k_amount_approved, 0) || eol);
  info = blob_append(info, 'Клиенты: ' || clients_fio || eol || eol || line || line || eol || eol || 'Другие договора в группе: ');

  tmp_k_id = null;
  k_kontract_no = null;
  k_disb_date = null;
  k_amount_approved = null;
  ks_state_name = null;
  pr_name = null;
  i = 0;

  for select k_id,
             k_kontract_no,
             k_disb_date,
             k_amount_approved,
             iif(ks_state = 1, 'активный', 'закрыт'),
             pr_name
      from kontract
      inner join kontstate on ks_from_k_id = k_id
      inner join product on pr_id = k_from_pr_id
      where k_id <> :k_id and
            k_from_kf_id = :kf_id and
            ks_ol_is_delete = 0 and
            ks_date_end = '31.12.9999' and
            ks_state in (1, 2)
      order by k_disb_date desc
      into tmp_k_id,
           k_kontract_no,
           k_disb_date,
           k_amount_approved,
           ks_state_name,
           pr_name
  do
  begin
    i = i + 1;

    if (exists(select oko_from_k_id
               from operkroff
               where oko_from_k_id = :tmp_k_id and
                     oko_ol_is_active = 1 and
                     oko_sh_os - oko_os > 0
               group by oko_from_k_id)) then
      ks_state_name = 'спец. учет';

    clients_fio = null;
    select left(list(c_name_fio, ', '), 255)
    from clients
    inner join knclient on kc_from_c_id = c_id
    where kc_from_k_id = :tmp_k_id and
          kc_amount_approved > 0
    into clients_fio;

    if (char_length(clients_fio) = 255) then
      clients_fio = left(clients_fio, 252) || '...';

    info = blob_append(info, eol || eol || i || ') № ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date));
    info = blob_append(info, ' (' || trim(ks_state_name) || '), продукт "' || pr_name || '", cумма ' || f_numtostr(k_amount_approved, 0));
    info = blob_append(info, eol || 'Клиенты: ' || clients_fio);
  end

  suspend;
end
