CREATE PROCEDURE HP_JIRA_LOG_SUMMARY(
    ISSUE_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE = null,
    REP_DATE_END D_DATE = null)
RETURNS (
    JRS_CODE D_STR_CODE,
    JRS_IS_LAST D_BOOL,
    MIN_DATETIME D_DATETIME,
    MAX_DATETIME D_DATETIME,
    SECOND_TOTAL D_INTEGER,
    SECOND_PERIOD D_INTEGER,
    TIME_TOTAL D_STR_CODE,
    TIME_PERIOD D_STR_CODE)
AS
declare variable total_min d_datetime;
declare variable total_max d_datetime;
declare variable sum_second d_integer;
declare variable sum_period d_integer;
begin
  total_min = null;
  total_max = null;
  sum_second = 0;
  sum_period = 0;

  for select jrs_code,
             min(min_datetime),
             max(max_datetime),
             sum(second_total),
             sum(second_period),
             max(jrs_is_last)
  from hp_jira_log_detail(:issue_id, :rep_date_begin, :rep_date_end)
  where jrs_code <> 'summary'
  group by 1
  having sum(second_total) <> 0
  order by 3, 2
  into jrs_code, min_datetime, max_datetime, second_total, second_period, jrs_is_last
  do
  begin
    if (second_total is null) then
      second_total = 0;

    if (second_period is null) then
      second_period = 0;

    time_total = f_jira_period(second_total);
    time_period = f_jira_period(second_period);

    suspend;

    sum_second = sum_second + second_total;
    sum_period = sum_period + second_period;

    if (total_min is null or total_min > min_datetime) then
      total_min = min_datetime;

    if (total_max is null or total_max < max_datetime) then
      total_max = max_datetime;
  end

  jrs_is_last = 0;
  jrs_code = 'Итоги';
  min_datetime = total_min;
  max_datetime = total_max;
  second_total = sum_second;
  second_period = sum_period;
  time_total = f_jira_period(second_total);
  time_period = f_jira_period(second_period);

  suspend;
end
