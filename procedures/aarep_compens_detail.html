CREATE PROCEDURE AAREP_COMPENS_DETAIL(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_255,
    KF_NAME D_STR_100,
    FIO D_STR_100,
    C_INN D_STR_20,
    PAY_TYPE D_STR_100,
    K_KONTRACT_NO D_STR_100,
    OL_DATE D_DATE,
    GOS_PAYED_PERC DECIMAL(15,2))
AS
declare variable k_id d_id;
declare variable ol_id d_id;
declare variable k_id_old d_id;
declare variable c_id d_id;
declare variable kf_id d_id;
declare variable qty_active d_integer;
declare variable is_found d_integer;
begin
  k_id_old = 0;

  for select distinct kont.k_id, ok.ok_from_ol_id, ok.ok_ol_date, kont.k_kontract_no, kont.k_from_kf_id
      from operkred ok
      inner join kontract kont on ok.ok_from_k_id = kont.k_id
      where ok.ok_type = 4 and
            ok.ok_ol_date between :rep_date_begin and :rep_date_end and
            ok.ok_ol_is_active = 1 and
            ok.ok_perc < 0 and
            kont.k_kolich_client_approved = 1
      order by k_id, ok.ok_from_ol_id
      into :k_id, :ol_id, :ol_date, :k_kontract_no, :kf_id
  do
  begin
    gos_payed_perc = 0;

    select sum(ob_summa)
    from operbook
    where ob_from_ol_id = :ol_id and
          substring(ob_account_dt from 8 for 5) = '11531'
    into :gos_payed_perc;

    if (gos_payed_perc <> 0) then
    begin
      if (k_id <> k_id_old) then
      begin
        k_id_old = k_id;
        c_inn = null;

        select cl.c_name_fio, cl.c_inn
        from knclient kncl
        inner join clients cl on cl.c_id = kncl.kc_from_c_id
        where kncl.kc_from_k_id = :k_id and
              kncl.kc_amount_approved > 0
        into :fio, :c_inn;

        select kal.kal_new_kf_name, reg.reg_name
        from knaktlist kal
        inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
        where kal.kal_from_kf_id = :kf_id and
              :rep_date_end between kal.kal_date and kal.kal_date_end and
              kal.is_delete = 0
        into :kf_name, :reg_name;

        pay_type = 'Индив. Обычное погашение';

        if (exists(select *
                   from operlist ol
                   where ol.ol_id = :ol_id and
                         ol.ol_doc_client_name = 'погашение процентов государством')) then
          pay_type = 'Индив. Погаш. роботом';
      end

      suspend;
    end
  end
  -------------
  k_id_old = 0;

  for select distinct kont.k_id, ok.ok_from_ol_id, ok.ok_ol_date, kont.k_kontract_no, kont.k_from_kf_id
      from operkred ok
      inner join kontract kont on ok.ok_from_k_id = kont.k_id
      where ok.ok_type = 4 and
            ok.ok_ol_date between :rep_date_begin and :rep_date_end and
            ok.ok_ol_is_active = 1 and
            ok.ok_perc < 0 and
            kont.k_kolich_client_approved > 1
      order by k_id, ok.ok_from_ol_id
      into :k_id, :ol_id, :ol_date, :k_kontract_no, :kf_id
  do
  begin
    c_inn = null;
    if (k_id <> k_id_old) then
    begin
      k_id_old = k_id;

      select kal.kal_new_kf_name, reg.reg_name
      from knaktlist kal
      inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
      where kal.kal_from_kf_id = :kf_id and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into :kf_name, :reg_name;
    end

    gos_payed_perc = 0;

    select sum(ob_summa)
    from operbook
    where ob_from_ol_id = :ol_id and
          substring(ob_account_dt from 8 for 5) = '11531'
    into :gos_payed_perc;

    if (gos_payed_perc <> 0) then
    begin
      k_id_old = k_id;
      c_id = null;

      select kncl.kc_from_c_id
      from knclient kncl
      where kncl.kc_from_k_id = :k_id and
            kncl.kc_closed_ol_id = :ol_id
      into :c_id;

      fio = '';

      if (c_id is not null) then
      begin
        select cl.c_name_fio, c_inn
        from clients cl
        where cl.c_id = :c_id
        into :fio, c_inn;

        pay_type = 'Груп. Полное погаш. доли участника';
        suspend;
      end
      else
      begin
        select ol.ol_document_client
        from operlist ol
        where ol.ol_id = :ol_id and
              ol.ol_doc_client_name = 'погашение процентов государством'
        into :c_id;

        if (c_id is not null) then
        begin
          select cl.c_name_fio, c_inn
          from clients cl
          where cl.c_id = :c_id
          into :fio, c_inn;

          pay_type = 'Груп. Погаш. роботом';
          suspend;
        end
        else
        begin
          is_found = 0;
          for select c.c_name_fio, gos_payed_perc, c.c_inn
              from cli_part_payment(:ol_id) cpp
              inner join clients c on cpp.c_id = c.c_id
              into :fio, :gos_payed_perc, c_inn
          do
          begin
            is_found = 1;
            pay_type = 'Груп. Обычное погашение';
            suspend;
          end

          if (is_found = 0) then
          begin
              /*
              if (exists(select *
                         from kontstate ks
                         where ks.ks_ol_is_delete = 0
                         and ks.ks_state = 2
                         and ks.ks_from_ol_id =:ol_id)) then
              begin
              */
                qty_active = 0;

                select count(kncl.kc_id) - count(ol.ol_id)
                from knclient kncl
                left join operlist ol on ol.ol_id = kncl.kc_closed_ol_id and ol.is_delete = 0
                where kncl.kc_from_k_id = :k_id
                and kncl.kc_amount_approved > 0
                into :qty_active;

                if (qty_active = 1) then
                begin
                    select kncl.kc_from_c_id
                    from knclient kncl
                    inner join knclient_compens kcc on kcc.kcc_from_kc_id = kncl.kc_id
                    left join operlist ol on ol.ol_id = kncl.kc_closed_ol_id and ol.is_delete = 0
                    where kncl.kc_from_k_id = :k_id
                    and ol.ol_id is null
                    and kncl.kc_amount_approved > 0
                    and kcc.kcc_date_end > kcc.kcc_date_begin
                    into :c_id;

                    if (c_id is not null) then
                    begin
                        select cl.c_name_fio, c_inn
                        from clients cl
                        where cl.c_id = :c_id
                        into :fio, c_inn;
                    end
                    else
                    begin
                        fio = '[ПРОБЛЕМА. Не могу понять, на кого распределить сумму]';
                        c_inn = '';
                    end

                    pay_type = 'Груп. Обычное погашение. Остался один участник';
                    suspend;
                end
              --end
          end
        end
      end
    end
  end
end
