CREATE PROCEDURE MC_GET_PHONE(
    MCL_ID D_ID)
RETURNS (
    PHONE D_STR_20,
    IS_EXISTS D_BOOL)
AS
declare variable C_ID D_ID;
begin
  is_exists = 1;
  phone = '';

  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :c_id;

  if (c_id is not null) then
    phone = coalesce(f_cont_info('clients', :c_id, 'cell', 'ok'), '');

  if (phone = '') then
  begin
    is_exists = 0;
    phone = 'номер отсутствует';
  end
  else
  begin
    if (f_get_db_id() = 39801) then
        phone = right(:phone, 10);
  end


  suspend;
end
