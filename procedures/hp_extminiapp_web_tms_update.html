CREATE PROCEDURE HP_EXTMINIAPP_WEB_TMS_UPDATE(
    IN_UUID D_STR_CODE,
    IN_TARIF D_STR_CODE,
    IN_AMOUNT D_CURRENCY,
    IN_PERIOD D_INTEGER)
RETURNS (
    RES_JSON D_STR_8100)
AS
declare variable EMA_UUID D_UUID;
declare variable EMA_ID D_ID;
declare variable RES_TEXT D_STR_255;
declare variable OBJ_KIND D_INTEGER;
declare variable PR_ID D_INTEGER;
begin
  res_text = '';
  res_json = '{"result":{"success":%1, "text":"%2"}}';

  begin
    ema_uuid = char_to_uuid(in_uuid);
    when any do
      res_text = 'Неверная ссылка';
  end

  if (res_text = '') then
  begin
    obj_kind = f_objkind('extminiapp', 'id');

    select udl_object_id
    from uuidlist
    where udl_uuid = char_to_uuid(:in_uuid) and -- :ema_uuid and
          udl_object_kind = :obj_kind
    into ema_id;

    if (ema_id is not null) then
    begin
      pr_id = 490; -- по IN_TARIF вычислить продукт

      update extminiapp
      set ema_period = :in_period,
          ema_amount = :in_amount,
          ema_from_pr_id = :pr_id,
          ema_address_str = :in_tarif
      where ema_id = :ema_id and
            ema_period = 0 and
            ema_amount = 0;

      if (row_count = 0) then
        res_json = f_format(res_json, 'false', 'Ссылка более не актуальна, принят на обработку в МБулак');
      else
      begin
        execute procedure hp_expressapp_by_extminiapp_tms(:ema_id);

        res_json = f_format(res_json, 'true', 'Заявка на рассроку сформирована, ждите ссылку по СМС');
      end
    end else
      res_json = f_format(res_json, 'false', 'Не найдено приглашение по запросу');
  end else
    res_json = f_format(res_json, 'false', res_text);

  suspend;
end
