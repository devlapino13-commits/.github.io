CREATE PROCEDURE PRINT_OVERPAY(
    OPM_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    EDL_BLANK_DATE D_DATE,
    C_FIO D_STR_100,
    ADDRESS_YUR D_STR_255,
    C_PASS_NO D_STR_CODE,
    C_PASS_WHO D_STR_255,
    C_PASS_DATE D_DATE,
    ACTIVE_LOANS D_STR_8100,
    DEBT_FOR_PAYMENT D_CURRENCY,
    QR_CODE D_STR_LONG)
AS
declare variable C_ID D_ID;
declare variable OVERPAY_SUM D_CURRENCY;
declare variable DEBT_SUM D_CURRENCY;
declare variable EDL_ID D_ID;
begin
  select c_id, c_name_fio, c_pass_no, c_pass_who, c_pass_date
  from overpay_main, clients
  where opm_id = :opm_id
    and c_id = opm_from_c_id
  into c_id, c_fio, c_pass_no, c_pass_who, c_pass_date;

  select adl_address_str
  from addrclient, addrlist
  where adc_from_c_id = :c_id
    and adc_address_type = 1
    and adc_from_adl_id = adl_id
  into address_yur;

  if (address_yur is null or address_yur = 'не указан') then
    address_yur = 'адрес не указан';

  select list(k_kontract_no || ' от ' || f_datetostr(k_disb_date), ascii_char(10)), sum(opl_sum)
  from overpay_loans, kontract
  where k_id = opl_from_k_id
    and opl_from_opm_id = :opm_id
    and opl_type = 2
  into active_loans, debt_sum;

  select sum(opl_sum)
  from overpay_loans
  where opl_from_opm_id = :opm_id
    and opl_type = 1
  into overpay_sum;

  debt_for_payment = iif(overpay_sum > debt_sum , debt_sum, overpay_sum);

  -- QR-код электронной подписи клиента. Текст пустой если не подписан
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);
    qr_code = f_edoc_qr_by_edl(edl_id, c_id);
    select edl_blank_date
    from edoc_list
    where edl_id = :edl_id
    into edl_blank_date;
  end

  suspend;
end
