CREATE PROCEDURE FM_LOANS_IN_PROCESS_TMP(
    REG_ID D_INTEGER,
    C_ID D_ID)
RETURNS (
    K_ID D_ID,
    KS_STATE D_INTEGER,
    K_STATUS D_INTEGER,
    KF_ID D_ID,
    DISB_DATE D_DATE,
    MA_ID_TMP D_INTEGER,
    PR_ID D_INTEGER)
AS
begin
    if (reg_id = -1) then
    begin
        for select ks_from_k_id, ks_state, k_status, k_from_kf_id, k_disb_date, k_from_ma_app_id, k_from_pr_id
            from kontstate, kontract, knaktlist
            where ks_ol_is_delete = 0 and
                  k_status <> -2 and
                  ks_date_end >= '31.12.2999' and
                  ((ks_state = 0 and
                  k_status >= 0 and
                  ks_date_begin between(current_date - 20) and current_date) -- Договора в процессе
                  or (ks_state = 0 and
                  k_status = -1 and
                  ks_date_begin between(current_date - 1) and current_date) -- Отказанные договора в процессе
                  or (ks_state = 1 and
                  ks_date_begin = current_date) -- выданные за сегодня
                  ) and
                  ks_from_k_id = k_id and
                  knaktlist.is_delete = 0 and
                  kal_date <= current_date and
                  kal_date_end >= current_date and
                  kal_from_kf_id = k_from_kf_id and
                  kal_new_kf_officer + 0 = :c_id
            order by ks_date_begin desc
            into :k_id, ks_state, k_status, kf_id, disb_date, ma_id_tmp, :pr_id
        do
            suspend;
        end
        else
        begin
            for select ks_from_k_id, ks_state, k_status, k_from_kf_id, k_disb_date, k_from_ma_app_id, k_from_pr_id
                from kontstate, kontract, knaktlist
                where ks_ol_is_delete = 0 and
                      k_status <> -2 and
                      ks_date_end >= '31.12.2999' and
                      ((ks_state = 0 and
                      k_status >= 0 and
                      ks_date_begin between(current_date - 20) and current_date) -- Договора в процессе
                      or (ks_state = 0 and
                      k_status = -1 and
                      ks_date_begin between(current_date - 1) and current_date) -- Отказанные договора в процессе
                      or (ks_state = 1 and
                      ks_date_begin = current_date) -- выданные за сегодня
                      ) and
                      ks_from_k_id = k_id and
                      knaktlist.is_delete = 0 and
                      kal_date <= current_date and
                      kal_date_end >= current_date and
                      kal_from_kf_id = k_from_kf_id and
                      kal_new_reg_id + 0 = :reg_id
                order by ks_date_begin desc
                into :k_id, ks_state, k_status, kf_id, disb_date, ma_id_tmp, :pr_id
            do
                suspend;
        end
    end
