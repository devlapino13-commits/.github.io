CREATE PROCEDURE AAREP_NEW_SCHEDULE_KONT(
    KONT_ID DECIMAL(18,0))
RETURNS (
    SH_ID INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_DISB_AMOUNT DECIMAL(18,4),
    SH_NAME_FIO VARCHAR(100),
    REG_NAME VARCHAR(50),
    REG_ADDRESS VARCHAR(255),
    KF_NAME VARCHAR(100),
    KF_OFFICER VARCHAR(100),
    OFFISER_PHONE VARCHAR(20),
    SIGNMEN VARCHAR(100),
    SIGNMEN_PHON VARCHAR(20),
    REG_PHONE VARCHAR(255),
    K_KONTRACT_NO VARCHAR(20),
    REP_DD DATE,
    K_PERCENT DECIMAL(18,7),
    C_NAMES VARCHAR(1024),
    K_DISB_DATE DATE,
    K_PERCENT_MONTH DECIMAL(18,7),
    SH_PERC_TAX DECIMAL(18,4),
    K_NEW_DATE DATE)
AS
declare variable reg_address_fact decimal(18,0);
declare variable c_name_fio varchar(100);
declare variable k_nsp_percent decimal(18,7);
declare variable k_percent_temp decimal(18,7);
declare variable k_percent_day decimal(18,7);
declare variable kd_is_primary d_smallint;
begin
  select kd_ol_date, kd_is_primary, kd_percent_per_day
  from kontdata
  where kd_from_k_id = :kont_id and
        kd_ol_is_delete = 0
  order by kd_from_k_id desc, kd_ol_date desc
  rows 1
  into :rep_dd, kd_is_primary, k_percent_day; -- последняя дата изменения условий %
  k_new_date = :rep_dd;

  if (kd_is_primary = 1) then
    exit;

  k_percent = cast(k_percent_day * 365.0 as decimal(15,1));

  k_nsp_percent = 0;
  select k_disb_date, k_percent
  from kontract
  where k_id = :kont_id
  into :k_disb_date, :k_percent_temp;

  if (k_percent is null) then
    k_percent = k_percent_temp;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end

-- Вытаскивает параметры офиса и КС
  select reg_name, reg_address, kal_new_kf_name, c_name_fio,
         (select cont_value
          from hp_contdata_get_cell_main(0, c_id)), k_disb_date, reg_phone, reg_address_fact, k_amount_approved,
         k_kontract_no
  from kontract, knaktlist, region, clients
  where k_from_kf_id = kal_from_kf_id and
        knaktlist.is_delete = 0 and
        :rep_dd between kal_date and kal_date_end and
        k_id = :kont_id and
        reg_unicode = kal_new_reg_id and
        c_id = kal_new_kf_officer
  into reg_name, reg_address, kf_name, kf_officer, offiser_phone, k_disb_date, reg_phone, reg_address_fact,
       :sh_disb_amount, :k_kontract_no;
-- вытаскивает уполномоченное лицо кто утверждал в последний раз
  if (reg_address_fact > 0) then
  begin
    select adl_address_str
    from addrlist
    where adl_id = :reg_address_fact
    into :reg_address;
  end
  k_percent_month = k_percent_day * 30; --k_percent/365*30;
  c_names = '';
  for select c_name_fio
      from knclient, clients
      where kc_from_c_id = c_id and
            kc_from_k_id = :kont_id and
            kc_amount_approved > 0
      into :c_name_fio
  do
  begin
    if (c_names = '') then
      c_names = c_name_fio;
    else
      c_names = c_names || ', ' || c_name_fio;
  end

  select un_fio,
         (select cont_value
          from hp_contdata_get_cell_main(0, c_id))
  from kontlog, username, kontsignmen, clients
  where kl_state = 1 and
        kl_user_id = un_id and
        ksm_is_active = 1 and
        ksm_from_c_id = un_from_c_id and
        un_from_c_id = c_id and
        kl_from_k_id = :kont_id and
        cast(kl_datetime as date) <= :k_disb_date
  order by kl_datetime desc
  rows 1
  into signmen, signmen_phon;
  if (reg_phone is null) then
    reg_phone = signmen_phon;

  sh_name_fio = 'Тобунун катышуучуларына / Участникам Группы';
  for select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
      from operkred
      where ok_from_k_id = :kont_id and
            ok_type < 7 and
            ok_ol_is_active = 1 and
            ok_ol_date <= :rep_dd and
            ok_from_c_id = 0
      group by ok_sh_id, ok_sh_date
      having sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc) <> 0
      order by ok_sh_date
      into :sh_id, :sh_date, :sh_os, :sh_perc
  do
  begin
    sh_perc_tax = round(sh_perc * k_nsp_percent / 100);
    suspend;
  end

end
