CREATE PROCEDURE AAREP_TALON_SALDO(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    PEO_BILL_ID DECIMAL(18,0),
    PEO_BILL_LINE_NO INTEGER,
    PEO_FROM_C_ID DECIMAL(18,0),
    PEO_COST DECIMAL(18,4),
    PEO_FROM_PEN_ID INTEGER,
    PEO_KOLICH DECIMAL(18,4),
    PEO_KOLICH_REPORT DECIMAL(18,4),
    PEO_LITER DECIMAL(18,4),
    PEO_LITER_REPORT DECIMAL(18,4),
    PEO_SUMMA DECIMAL(18,4),
    PEO_SUMMA_REPORT DECIMAL(18,4),
    PEO_OPER_DATE DATE,
    PEO_FROM_REG_ID INTEGER,
    PEO_MATOTVET_FIO VARCHAR(100),
    PEO_PETROL_NAME VARCHAR(100),
    PEO_ACC_BALANCE VARCHAR(20),
    PEO_ACC_NUMBER INTEGER,
    PEO_LITER_IN_ONE DECIMAL(18,4),
    PEO_BRANCH_ID INTEGER)
AS
begin
  for select peo_bill_id, peo_bill_line_no, peo_from_c_id, peo_cost, peo_from_pen_id, sum(peo_kolich), sum(peo_liter),
             sum(peo_summa), peo_oper_date, peo_from_reg_id, peo_matotvet_fio, peo_petrol_name, peo_acc_balance,
             peo_acc_number, peo_liter_in_one, peo_branch_id, sum(peo_liter_report), sum(peo_kolich_report),
             sum(peo_summa_report)
      from aarep_talon_saldo_new(:rep_region_id, :rep_date)
      group by peo_bill_id, peo_bill_line_no, peo_from_c_id, peo_cost, peo_from_pen_id, peo_oper_date, peo_from_reg_id, peo_matotvet_fio, peo_petrol_name, peo_acc_balance, peo_acc_number, peo_liter_in_one, peo_branch_id

      into :peo_bill_id, :peo_bill_line_no, :peo_from_c_id, :peo_cost, :peo_from_pen_id, :peo_kolich, :peo_liter,
           :peo_summa, :peo_oper_date, :peo_from_reg_id, :peo_matotvet_fio, :peo_petrol_name, :peo_acc_balance,
           :peo_acc_number, :peo_liter_in_one, :peo_branch_id, :peo_liter_report, :peo_kolich_report,
           :peo_summa_report
  do
  begin
    suspend;
  end
end
