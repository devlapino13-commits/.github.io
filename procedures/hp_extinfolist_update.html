CREATE PROCEDURE HP_EXTINFOLIST_UPDATE(
    OLD_ID D_ID,
    OLD_FROM_C_ID D_ID,
    OLD_FROM_EIC_ID D_ID,
    OLD_DATE D_DATE,
    OLD_OBJECT_KIND D_INTEGER,
    OLD_OBJECT_ID D_ID,
    OLD_EXT_ID D_STR_CODE,
    OLD_IS_JSON D_SMALLINT,
    OLD_FROM_BLS_ID D_ID,
    OLD_STATUS D_SMALLINT,
    OLD_KK_MEMO_STR D_STR_LONG,
    OLD_KK_UN_ID D_ID,
    OLD_KK_RESULT D_SMALLINT)
RETURNS (
    EIL_ID D_ID,
    EIL_FROM_C_ID D_ID,
    EIL_FROM_EIC_ID D_ID,
    EIL_DATE D_DATE,
    EIL_OBJECT_KIND D_INTEGER,
    EIL_OBJECT_ID D_ID,
    EIL_EXT_ID D_STR_CODE,
    EIL_IS_JSON D_SMALLINT,
    EIL_FROM_BLS_ID D_ID,
    EIL_STATUS D_SMALLINT,
    EIL_CREATE_DATETIME D_DATETIME,
    EIL_STATUS_DATETIME D_DATETIME,
    EIL_KK_MEMO_ID D_ID,
    EIL_KK_UN_ID D_ID,
    EIL_KK_RESULT D_SMALLINT,
    EIL_KK_DATETIME D_DATETIME,
    EIC_NAME D_STR_255,
    EIC_HAS_API D_SMALLINT,
    EIL_KK_MEMO_STR D_STR_LONG,
    EIL_HAS_BLOB D_SMALLINT,
    EIL_STATUS_STR D_STR_CODE,
    EIC_NEED_KK_RESULT D_SMALLINT,
    EIL_IS_OLD_INFO D_SMALLINT)
AS
declare variable old_kk_memo_id d_id;
begin
  select mml_id
  from hp_memolist_ins(:old_kk_memo_str)
  into :old_kk_memo_id;

  if (old_date is null or old_date > current_date) then
    old_date = current_date;

  if (old_object_id is null) then
  begin
    select eil_object_kind, eil_object_id
    from extinfolist
    where eil_from_c_id = :old_from_c_id and
          abs(datediff(day, eil_date, :old_date)) <= 2 and
          eil_from_eic_id = 1
    order by abs(datediff(day, eil_date, :old_date))
    rows 1
    into old_object_kind, old_object_id;
  end

  if (old_object_id is null) then
  begin
    select k_id
    from knclient, kontract
    where kc_from_k_id = k_id and
          kc_from_c_id = :old_from_c_id and
          kc_amount_approved > 0 and
          k_status >= 0 and
          k_disb_date >= :old_date
    order by k_disb_date desc
    rows 1
    into old_object_id;

    if (old_object_id is not null) then
      old_object_kind = f_objkind('kontract', 'id');
  end

  if (old_object_id is null) then
  begin
    select ma_id
    from miniappclient, miniapp
    where mac_from_ma_id = ma_id and
          mac_from_c_id = :old_from_c_id and
          mac_amount > 0 and
          ma_date_oper >= :old_date
    order by ma_date_app desc
    rows 1
    into old_object_id;

    if (old_object_id is not null) then
      old_object_kind = f_objkind('miniapp', 'id');
  end

  if (old_id is null) then
  begin
    insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_ext_id,
                             eil_is_json, eil_from_bls_id, eil_status,
                             eil_kk_memo_id, eil_kk_un_id, eil_kk_result)
    values (:old_from_c_id, :old_from_eic_id, current_date, :old_object_kind, :old_object_id, :old_ext_id,
            :old_is_json, :old_from_bls_id, :old_status,
            :old_kk_memo_id, :old_kk_un_id, :old_kk_result)
    returning eil_id
    into eil_id;
  end else
  begin
    update extinfolist
    set eil_from_c_id = :old_from_c_id,
        eil_from_eic_id = :old_from_eic_id,
        eil_date = :old_date,
        eil_object_kind = :old_object_kind,
        eil_object_id = :old_object_id,
        eil_ext_id = :old_ext_id,
        eil_is_json = :old_is_json,
        eil_from_bls_id = :old_from_bls_id,
        eil_status = :old_status,
        eil_kk_memo_id = :old_kk_memo_id,
        eil_kk_un_id = :old_kk_un_id,
        eil_kk_result = :old_kk_result
    where (eil_id = :old_id);

    eil_id = old_id;
  end

  select eil_from_c_id,
         eil_from_eic_id,
         eil_date,
         eil_object_kind,
         eil_object_id,
         eil_ext_id,
         eil_is_json,
         eil_from_bls_id,
         eil_status,
         eil_create_datetime,
         eil_status_datetime,
         eil_kk_memo_id,
         eil_kk_un_id,
         eil_kk_result,
         eil_kk_datetime,
         eic_name,
         eic_has_api,
         eic_need_kk_result,
         iif(eil_status = 2 and datediff(day, eil_date, current_date) > eic_info_actual_days, 1, 0)
  from extinfolist, extinfocfg
  where eil_id = :eil_id and
        eil_from_eic_id = eic_id
  into :eil_from_c_id,
       :eil_from_eic_id,
       :eil_date,
       :eil_object_kind,
       :eil_object_id,
       :eil_ext_id,
       :eil_is_json,
       :eil_from_bls_id,
       :eil_status,
       :eil_create_datetime,
       :eil_status_datetime,
       :eil_kk_memo_id,
       :eil_kk_un_id,
       :eil_kk_result,
       :eil_kk_datetime,
       :eic_name,
       :eic_has_api,
       :eic_need_kk_result,
       :eil_is_old_info;

  eil_kk_memo_str = '';
  if (eil_kk_memo_id <> 0) then
  begin
    select mml_memo
    from memolist
    where mml_id = :eil_kk_memo_id
    into eil_kk_memo_str;
  end

  eil_status_str = trim(case eil_status
                          when -3 then 'Отменен'
                          when -2 then 'Ошибка'
                          when -1 then 'Недоступен'
                          when 0 then 'В очереди'
                          when 1 then 'В обработке'
                          when 2 then 'Готов'
                        end);

  if (eil_from_bls_id is null)
  then eil_has_blob = 0;
  else eil_has_blob = 1;

  suspend;
end
