CREATE PROCEDURE NBKI_SET_NEW_KONTRACT(
    REP_K_ID ID_TYPE = 0,
    REP_CP_ID INTEGER = null)
AS
declare variable rep_date date;
begin
  if (rep_cp_id is null) then
    rep_date = current_date;
  else
  begin
    select cp.cp_date_end
    from cib_pocket cp
    where cp.cp_id = :rep_cp_id
    into :rep_date;
  end

  rep_k_id = coalesce(rep_k_id, 0);

  if (rep_k_id = 0) then -- если не задан конкретный кредит, тогда автоматически вычисляем нужный кредит
  begin
    -- получаю новый кредит
    select ks.ks_from_k_id
    from kontstate ks
    inner join kontract kont on kont.k_id = ks.ks_from_k_id and
        kont.k_kolich_client_approved = 1 and
          kont.k_base_k_id is null
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    where ks.ks_state = 1 and
          ks.ks_date_begin between :rep_date - 14 and :rep_date - 7 and
          ks.ks_ol_is_delete = 0   and
          pr.pr_code not like '.ZALOG_CAR.'
    order by ks.ks_date_begin
    rows 1
    into :rep_k_id;
  end

  if (rep_k_id <> 0) then
  begin
    update nbki_kontract_list
    set nkl_date_end = :rep_date - 1
    where nkl_date_end = '31.12.9999';

    insert into nbki_kontract_list (nkL_from_k_id, nkl_date_begin, nkl_date_end, nkl_poket_id)
    values (:rep_k_id, :rep_date, '31.12.9999', :rep_cp_id);
  end
end
