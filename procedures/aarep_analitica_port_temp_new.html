              aa_deliq_days <= 60)) then
          begin
            aa_by_deliq_days = 'D Просрочен. (31-60)';
            aa_deliq_kolich30 = aa_deliq_kolich;
            aa_deliq_cl_kolich30 = aa_deliq_cl_kolich;
            aa_deliq_balance30 = aa_deliq_balance;
            aa_deliq_os30 = aa_deliq_os_all;
            aa_deliq_perc30 = aa_deliq_perc_all;
          end
        end

        if (aa_deliq_days > 60) then
        begin
          if ((rep_period_kind = 0) or (rep_period_kind = 1 and
              aa_deliq_days <= 90)) then
          begin
            aa_by_deliq_days = 'E Просрочен. (61-90)';
            aa_deliq_kolich60 = aa_deliq_kolich;
            aa_deliq_cl_kolich60 = aa_deliq_cl_kolich;
            aa_deliq_balance60 = aa_deliq_balance;
            aa_deliq_os60 = aa_deliq_os_all;
            aa_deliq_perc60 = aa_deliq_perc_all;
          end
        end

        if (aa_deliq_days > 90) then
        begin
          if ((rep_period_kind = 0) or (rep_period_kind = 1 and
              aa_deliq_days <= 180)) then
          begin
            aa_by_deliq_days = 'F Просрочен. (91-180)';
            aa_deliq_kolich90 = aa_deliq_kolich;
            aa_deliq_cl_kolich90 = aa_deliq_cl_kolich;
            aa_deliq_balance90 = aa_deliq_balance;
            aa_deliq_os90 = aa_deliq_os_all;
            aa_deliq_perc90 = aa_deliq_perc_all;
          end
        end

        if (aa_deliq_days > 180) then
        begin
          if ((rep_period_kind = 0) or (rep_period_kind = 1 and
              aa_deliq_days <= 360)) then
          begin
            aa_by_deliq_days = 'G Просрочен. (181-360)';
            aa_deliq_kolich180 = aa_deliq_kolich;
            aa_deliq_cl_kolich180 = aa_deliq_cl_kolich;
            aa_deliq_balance180 = aa_deliq_balance;
            aa_deliq_os180 = aa_deliq_os_all;
            aa_deliq_perc180 = aa_deliq_perc_all;
          end
        end

        if (aa_deliq_days > 360) then
        begin
          aa_by_deliq_days = 'H Просрочен.свыше 360 дн.';
          aa_deliq_kolich360 = aa_deliq_kolich;
          aa_deliq_cl_kolich360 = aa_deliq_cl_kolich;
          aa_deliq_balance360 = aa_deliq_balance;
          aa_deliq_os360 = aa_deliq_os_all;
          aa_deliq_perc360 = aa_deliq_perc_all;
        end
        if ((aa_deliq_days > 7) and
            (rep_period_kind = 1)) then
        begin
          aa_deliq_kolich = 0;
          aa_deliq_cl_kolich = 0;
          aa_deliq_balance = 0;
          aa_deliq_os_all = 0;
          aa_deliq_perc_all = 0;
        end

        if (is_restruct = 1) then
          aa_by_deliq_days = aa_by_deliq_days || ', Реструкт.';
        aa_delinq_begin_date = first_deliq_dd;

-- 16.0 GB

        if (aa_kolich_cl_disb = 0) then
        begin
          aa_amount_max_disb = 0;
          aa_amount_min_disb = 999999999999;
        end

        if (rep_need_clients_kol = 1) then -- если требуется вычислять кол-во клиентов, а не только кредиты
        begin
-- Начало вычисления по клиентам
          aa_just_cl_begin = 0;
          aa_just_cl_end = 0;
          aa_just_cl_disb = 0;
          aa_just_cl_main_begin = 0;
          aa_just_cl_main_end = 0;
          aa_just_cl_main_disb = 0;

          for select kc_from_c_id, kc_amount_approved
              from knclient -- список всех клиентов в данном контракте
              where kc_from_k_id = :aa_k_id and
                    kc_amount_approved > 0
              into :c_id_tmp, :kc_loan_amount
          do
          begin
            if (aa_kolich_cl_disb > 0) then
            begin
              if (aa_amount_max_disb < (cast(kc_loan_amount * kont_curr_rate as decimal(18,4)))) then
                aa_amount_max_disb = kc_loan_amount * kont_curr_rate;
              if (aa_amount_min_disb > (cast(kc_loan_amount * kont_curr_rate as decimal(18,4)))) then
                aa_amount_min_disb = kc_loan_amount * kont_curr_rate;
            end

            chet_kol_begin = 0;
            chet_kol_end = 0;
            chet_kol_disb = 0;
            if (aa_kolich_cl_active_begin > 0) then
              chet_kol_begin = 1;
            if (aa_kolich_cl_active_end > 0) then
              chet_kol_end = 1;
            if (aa_kolich_disb = 1) then
              chet_kol_disb = 1;
            aa_just_cl_amount_begin = 0;
            aa_just_cl_amount_end = 0;
            aa_just_cl_amount_disb = 0;

            for select kc_from_k_id, kc_date_close_os
                from knclient
                where ((kc_date_disb between '01.01.1900' and (:dd_disb_date - 1)) or
                      (kc_date_disb = :dd_disb_date and
                      kc_from_k_id < :aa_k_id)) and
                      kc_date_close_os between :dd_disb_date and '31.12.9999' and
                      kc_from_c_id = :c_id_tmp
--              order by kc_date_close_os desc
                into :k_id_tmp, :dd_close_tmp
            do
            begin
              if ((aa_kolich_cl_active_begin > 0) and
                  (chet_kol_begin = 1)) then
              begin
                if (dd_close_tmp >= (:rep_date_begin - 1)) then
                begin
                  chet_kol_begin = 0;
                end
              end
              else
                chet_kol_begin = 0;

              if ((aa_kolich_cl_active_end > 0) and
                  (chet_kol_end = 1)) then
              begin
                if (dd_close_tmp >= :rep_date_end) then
                begin
                  chet_kol_end = 0;
                end
              end
              else
                chet_kol_end = 0;

              chet_kol_disb = 0;
            end
            if (chet_kol_begin = 1) then
            begin
              tmp = aa_amount_active_begin / kont_amount;
              aa_just_cl_amount_begin = aa_just_cl_amount_begin + cast(tmp * kc_loan_amount as decimal(15,2));
            end
            if (chet_kol_end = 1) then
            begin
              tmp = aa_amount_active_end / kont_amount;
              aa_just_cl_amount_end = cast(tmp * kc_loan_amount as decimal(15,2));
            end
            if (chet_kol_disb = 1) then
            begin
              aa_just_cl_amount_disb = :kc_loan_amount;
              if (kont_curr_rate <> 1) then
                aa_just_cl_amount_disb = cast(aa_just_cl_amount_disb * :kont_curr_rate as decimal(15,2));
            end

            aa_just_cl_begin = aa_just_cl_begin + chet_kol_begin;
            aa_just_cl_end = aa_just_cl_end + chet_kol_end;
            aa_just_cl_disb = aa_just_cl_disb + chet_kol_disb;

            aa_just_cl_main_begin = aa_just_cl_main_begin + aa_just_cl_amount_begin;
            aa_just_cl_main_end = aa_just_cl_main_end + aa_just_cl_amount_end;
            aa_just_cl_main_disb = aa_just_cl_main_disb + aa_just_cl_amount_disb;
          end -- конеч вычисления по клиентам
        end
        suspend;
        kc_id_prior = aa_curr_kc_id;
        office_prior = :aa_curr_region_id;
      end
    end
