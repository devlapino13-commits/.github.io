CREATE PROCEDURE FP_FILL_ANKETA(
    REP_C_ID ID_TYPE,
    REP_DATE DATE,
    REP_UN_ID ID_TYPE)
AS
declare variable c_id id_type;
declare variable fpa_city id_type;
declare variable fpa_client_fio str_255;
declare variable fpa_citizen integer;
declare variable fpa_date_place_birth str_255;
declare variable fpa_pass_kind integer;
declare variable fpa_pass_props str_255;
declare variable fpa_mcard_props str_255;
declare variable fpa_acard_props str_255;
declare variable fpa_inn str_255;
declare variable fpa_address_legal str_255;
declare variable fpa_rel_start date;
declare variable fpa_ank_resp id_type;
declare variable ank_date date;
declare variable fpa_id id_type;
declare variable fpa_date_check date;
declare variable fpa_cell_no str_20;
declare variable fpa_email str_100;
declare variable fpa_snils str_100;
begin
  if (rep_c_id is null) then
    rep_c_id = 0;
  if (rep_c_id = 0) then -- Если по всем активным клиентам
  begin
    for select distinct c_id
    from
    (   -- клиенты активных кредитов
        select kc_from_c_id as c_id
        from kontstate, knclient
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date between ks_date_begin and ks_date_end and
              ks_from_k_id = kc_from_k_id and
              kc_amount_approved > 0
        union all
        -- залогодатели активных кредитов
        select zal.z_from_c_id as c_id
        from kontstate ks, zalog zal
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date between ks_date_begin and ks_date_end and
              zal.z_from_k_id  = ks.ks_from_k_id and
              zal.z_from_c_id is not null and
              :rep_date between coalesce(zal.z_date_begin, '01.01.1900') and coalesce(zal.z_date_end, '31.12.9999') and 
              zal.z_is_ready = 1
    ) A
    into :c_id
    do
    begin
      select fpa_city, fpa_client_fio, fpa_citizen, fpa_date_place_birth, fpa_pass_kind, fpa_pass_props,
             fpa_mcard_props, fpa_acard_props, fpa_inn, fpa_address_legal, fpa_rel_start, fpa_ank_resp, fpa_date_check,
             fpa_cell_no, fpa_email, fpa_snils
      from fp_anketa_tmp(:rep_date, :c_id)
      into :fpa_city, :fpa_client_fio, :fpa_citizen, :fpa_date_place_birth, :fpa_pass_kind, :fpa_pass_props,
           :fpa_mcard_props, :fpa_acard_props, :fpa_inn, :fpa_address_legal, :fpa_rel_start, :fpa_ank_resp,
           :fpa_date_check, :fpa_cell_no, :fpa_email, :fpa_snils;

      fpa_id = 0;
      ank_date = null;

      select fp.fpa_id, fp.fpa_date
      from fp_anketa_cl fp
      where fp.fpa_c_id = :c_id
      order by fp.fpa_date desc
      rows 1
      into :fpa_id, :ank_date;

      ank_date = coalesce(ank_date, '01.05.2014');
      if (abs(datediff(day, cast(:ank_date as date), cast(:rep_date as date))) > 3) then
        fpa_id = 0;

      execute procedure hp_fp_anketa_ins_upd(:fpa_id, :c_id, :rep_date, :rep_un_id, 0, :fpa_city, :fpa_client_fio,
          :fpa_citizen, :fpa_date_place_birth, :fpa_pass_kind, :fpa_pass_props, :fpa_mcard_props, :fpa_acard_props,
          :fpa_inn, :fpa_address_legal, :fpa_rel_start, :fpa_ank_resp, :fpa_date_check, :fpa_cell_no, :fpa_email,
          :fpa_snils, 0);
    end
  end
  else
  begin
    select fpa_city, fpa_client_fio, fpa_citizen, fpa_date_place_birth, fpa_pass_kind, fpa_pass_props, fpa_mcard_props,
           fpa_acard_props, fpa_inn, fpa_address_legal, fpa_rel_start, fpa_ank_resp, fpa_date_check, fpa_cell_no,
           fpa_email, fpa_snils
    from fp_anketa_tmp(:rep_date, :rep_c_id)
    into :fpa_city, :fpa_client_fio, :fpa_citizen, :fpa_date_place_birth, :fpa_pass_kind, :fpa_pass_props,
         :fpa_mcard_props, :fpa_acard_props, :fpa_inn, :fpa_address_legal, :fpa_rel_start, :fpa_ank_resp,
         :fpa_date_check, :fpa_cell_no, :fpa_email, :fpa_snils;

    fpa_id = 0;
    ank_date = null;

    select fp.fpa_date, fp.fpa_id
    from fp_anketa_cl fp
    where fp.fpa_c_id = :rep_c_id
    order by fp.fpa_date desc
    rows 1
    into :ank_date, :fpa_id;

    ank_date = coalesce(ank_date, '01.05.2014');
    if (abs(datediff(day, cast(:ank_date as date), cast(:rep_date as date))) > 3) then
      fpa_id = 0;

    execute procedure hp_fp_anketa_ins_upd(:fpa_id, :rep_c_id, :rep_date, :rep_un_id, 1, :fpa_city, :fpa_client_fio,
        :fpa_citizen, :fpa_date_place_birth, :fpa_pass_kind, :fpa_pass_props, :fpa_mcard_props, :fpa_acard_props,
        :fpa_inn, :fpa_address_legal, :fpa_rel_start, :fpa_ank_resp, :fpa_date_check, :fpa_cell_no, :fpa_email,
        :fpa_snils, 0);
  end
end
