CREATE PROCEDURE PRINT_KONTMAT1(
    K_ID ID_TYPE,
    C_ID ID_TYPE = null)
RETURNS (
    KML_ID INTEGER,
    KML_OBJECT_KIND INTEGER,
    KML_OBJECT_ID DECIMAL(18,0),
    KML_FROM_ML_ID DECIMAL(18,0),
    KML_COUNT DECIMAL(18,4),
    KML_COST DECIMAL(18,4),
    KML_TOTAL DECIMAL(18,4),
    KML_STATUS INTEGER,
    KML_SELL_PROFIT DECIMAL(18,4),
    KML_SELL_PROFIT_ALL DECIMAL(18,4),
    KML_SELL_NSP DECIMAL(18,4),
    KML_SELL_NSP_ALL DECIMAL(18,4),
    KML_SELL_COST DECIMAL(18,4),
    KML_SELL_COST_NSP DECIMAL(18,4),
    KML_SELL_TOTAL DECIMAL(18,4),
    KML_BILL_ID DECIMAL(18,0),
    KML_BILL_LINE_NO INTEGER,
    KML_NAME VARCHAR(255),
    KML_MODEL VARCHAR(255),
    KML_CUR_TIMESTAMP TIMESTAMP,
    KML_SELLER_FIO VARCHAR(100),
    KML_GUARANTEE_MONTH DECIMAL(18,4),
    KML_BIRTH_DATE DATE,
    KML_BITTH_CITY VARCHAR(100),
    KML_SEX SMALLINT,
    KML_NATION_TXT VARCHAR(255),
    KML_FAMILY_TXT VARCHAR(255),
    KML_SEX_TXT VARCHAR(10),
    KML_CITIZEN_TXT VARCHAR(255),
    KML_EMAIL VARCHAR(255),
    KML_PHONE_NO VARCHAR(255),
    KML_ADDRESS_FACT VARCHAR(255),
    KML_PANT_TYPE STR_100,
    KML_PANT_DOC_NO STR_100,
    KML_PANT_DATE_BEGIN DATE,
    KML_PANT_DATE_END DATE,
    KML_PANT_ISSUED_BY STR_255,
    KML_PANT_MEMO_1 STR_255,
    KML_REGISTR_TYPE STR_100,
    KML_REGISTR_DOC_NO STR_100,
    KML_REGISTR_DATE_BEGIN DATE,
    KML_REGISTR_DATE_END DATE,
    KML_REGISTR_ISSUED_BY STR_255,
    KML_POLIS_TYPE STR_100,
    KML_POLIS_DOC_NO STR_100,
    KML_POLIS_DATE_BEGIN DATE,
    KML_POLIS_DATE_END DATE,
    KML_POLIS_ISSUED_BY STR_255,
    KML_C_MEMO VARCHAR(1024),
    KML_PURCHASE_DATETIME TIMESTAMP,
    KML_ISSUE_DATETIME TIMESTAMP,
    KML_SELL_TOTAL_NSP CURR_TYPE,
    KML_INN STR_20,
    KML_C_ID ID_TYPE,
    KML_FEG_CODE D_STR_10,
    KML_IS_RELATIVE D_INTEGER,
    KML_IS_USED D_SMALLINT,
    KMI_IMEI D_STR_20)
AS
declare variable ag_id integer; /* KML_SELLER_FIO */
declare variable kml_nation integer;
declare variable kml_citizen integer;
declare variable kml_family integer;
declare variable om_ol_datetime timestamp;
declare variable om_oper_type integer;
declare variable rd_id id_type;
declare variable rd_id1 id_type;
declare variable feg_id d_integer;
declare variable ma_k_id d_id;
begin
  if (not exists(select * from kontract
  where k_id = :k_id)) then
  begin
    select k_id from kontract
    where k_from_ma_app_id = :k_id
    order by k_id desc
    rows 1
    into ma_k_id;

    if (ma_k_id is not null)
    then k_id = ma_k_id;
    else exit;
  end

  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.IPF_Sellers'
  into :rd_id;

  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.tsp_and_ipf'
  into :rd_id1;

  if (k_id = 109010256266400101) then c_id = 1090101226816;

--Используется как "Время для печати"
  kml_cur_timestamp = current_timestamp;

  for select kml_id, kml_object_kind, kml_object_id, kml_from_ml_id, kml_count, kml_cost, kml_total, kml_status,
             kml_sell_profit, kml_sell_nsp, kml_sell_cost, kml_sell_total, kml_bill_id, kml_bill_line_no,
             kml_guarantee_month, kml_from_ag_id, ag_object_id, kml_from_feg_id, kml_is_relative, KML_IS_USED
      from kontmat km, agreements ag
      where km.kml_object_kind = 9 and
            km.kml_object_id = :k_id and
            km.kml_from_ml_id is not null and
            --km.kml_from_ag_id = ag_id and
            ag.ag_from_rd_id in (:rd_id, :rd_id1) and
            ag_object_kind = 0 and
            ((ag.ag_object_id = :c_id) or
            (:c_id is null))
      into :kml_id, :kml_object_kind, :kml_object_id, :kml_from_ml_id, :kml_count, :kml_cost, :kml_total, :kml_status,
           :kml_sell_profit, :kml_sell_nsp, :kml_sell_cost, :kml_sell_total, :kml_bill_id, :kml_bill_line_no,
           :kml_guarantee_month, :ag_id, :kml_c_id, :feg_id, :kml_is_relative, :KML_IS_USED
  do
  begin
    kml_seller_fio = '';
    kml_inn = '';
    kml_birth_date = null;
    kml_bitth_city = null;
    kml_citizen = null;
    kml_family = null;
    kml_nation = null;
    kml_nation_txt = null;
    kml_sex_txt = null;
    kml_email = null;
    kml_family_txt = null;
    kml_citizen_txt = null;
    kml_sex = null;
    kml_phone_no = null;
    kml_address_fact = null;
    kml_sell_cost_nsp = kml_sell_cost + kml_sell_nsp;
    kml_sell_total_nsp = round(kml_sell_cost_nsp * kml_count, 0);

    if (ag_id is not null) then
    begin
      select c_name_fio, c.c_birth_date, c.c_birth_city, c.c_nation, c.c_sex, c.c_citizen, c.c_family, c.c_memo,c.c_inn
      from clients c
      where c.c_id = :kml_c_id
      into kml_seller_fio, kml_birth_date, kml_bitth_city, kml_nation, kml_sex, kml_citizen, kml_family, kml_c_memo,kml_inn;

      select rd.rd_name
      from refdata rd
      where rd.rd_id = :kml_nation
      into :kml_nation_txt;

      kml_sex_txt = case kml_sex
                      when 0 then 'Женщина'
                      when 1 then 'Мужчина'
                      else 'Юр.лицо'
                    end;

      select rd.rd_name
      from refdata rd
      where rd.rd_id = :kml_family
      into :kml_family_txt;

      select rd.rd_name
      from refdata rd
      where rd.rd_id = :kml_citizen
      into :kml_citizen_txt;

      select cont_value
      from hp_contdata_get_email_valid(0, :kml_c_id)
      into kml_email;

      select cont_value
      from hp_contdata_get_cell_valid(0, :kml_c_id)
      into kml_phone_no;

      select adl_address_str
      from addrlist, addrclient
      where adc_from_c_id = :kml_c_id and
            adl_id = adc_from_adl_id and
            adc_address_type = 2
      into kml_address_fact;

      kml_pant_type = null;
      kml_pant_doc_no = null;
      kml_pant_date_begin = null;
      kml_pant_date_end = null;
      kml_pant_issued_by = null;

      select pk.pk_name, cd.cds_doc_no, cd.cds_doc_date_begin, cd.cds_doc_date_end, cd.cds_doc_issued_by,
             cd.cds_memo_1
      from clientdocs cd
      inner join passkind pk on cd.cds_from_pk_id = pk_id
      inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cds_from_c_id = :kml_c_id and
            current_date between cds_doc_date_begin and cds_doc_date_end and
            cds_is_delete = 0 and
            pdt.pdt_code in ('work_patent', 'License') rows 1
      into :kml_pant_type, :kml_pant_doc_no, :kml_pant_date_begin, :kml_pant_date_end, :kml_pant_issued_by,
           :kml_pant_memo_1;

      kml_registr_type = null;
      kml_registr_doc_no = null;
      kml_registr_date_begin = null;
      kml_registr_date_end = null;
      kml_registr_issued_by = null;

      select pk.pk_name, cd.cds_doc_no, cd.cds_doc_date_begin, cd.cds_doc_date_end, cd.cds_doc_issued_by
      from clientdocs cd
      inner join passkind pk on cd.cds_from_pk_id = pk_id
      inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cds_from_c_id = :kml_c_id and
            current_date between cds_doc_date_begin and cds_doc_date_end and
            cds_is_delete = 0 and
            (pdt.pdt_code = 'certificate_chp' or
            pdt.pdt_code = 'certificate_ip') rows 1
      into :kml_registr_type, :kml_registr_doc_no, :kml_registr_date_begin, :kml_registr_date_end,
           :kml_registr_issued_by;

      kml_polis_type = null;
      kml_polis_doc_no = null;
      kml_polis_date_begin = null;
      kml_polis_date_end = null;
      kml_polis_issued_by = null;

      select pk.pk_name, cd.cds_doc_no, cd.cds_doc_date_begin, cd.cds_doc_date_end, cd.cds_doc_issued_by
      from clientdocs cd
      inner join passkind pk on cd.cds_from_pk_id = pk_id
      inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cds_from_c_id = :kml_c_id and
            current_date between cds_doc_date_begin and cds_doc_date_end and
            cds_is_delete = 0 and
            pdt.pdt_code = 'Soc_polis' rows 1
      into :kml_polis_type, :kml_polis_doc_no, :kml_polis_date_begin, :kml_polis_date_end, :kml_polis_issued_by;

    end

    kml_name = null;
    kml_model = null;
    select ml_name, ml_model
    from matlist ml
    where ml.ml_id = :kml_from_ml_id
    into kml_name, kml_model;

--ДОбавлено по просьбе Кыял 17-января 17-года
    kml_sell_profit_all = 0;
    kml_sell_nsp_all = 0;
    if (coalesce(kml_sell_profit, 0) <> 0) then
      kml_sell_profit_all = kml_sell_profit * kml_count;
    if (coalesce(kml_sell_nsp, 0) <> 0) then
      kml_sell_nsp_all = kml_sell_nsp * kml_count;

    kml_purchase_datetime = null;
    kml_issue_datetime = null;
    for select ol.ol_datetime, om.om_oper_type
        from opermat om, operlist ol
        where om_bill_id = :kml_bill_id and
              om_bill_line_no = :kml_bill_line_no and
              om_oper_type in (1, 5) and
              om_from_ol_id = ol_id and
              ol.is_delete = 0
        into om_ol_datetime, om_oper_type
    do
    begin
      if (om_oper_type = 1) then
        kml_purchase_datetime = om_ol_datetime;
      if (om_oper_type = 5) then
        kml_issue_datetime = om_ol_datetime;
    end
    kml_feg_code = (select feg_code from foreign_econom_goods_ref where feg_id = :feg_id);
    kmi_imei = (select cast(kmi_imei as d_str_20) from kontmat_imei where kmi_kml_id = :kml_id);

    suspend;
  end
end
