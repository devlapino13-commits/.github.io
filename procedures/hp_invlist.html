CREATE PROCEDURE HP_INVLIST(
    IM_ID D_ID)
RETURNS (
    IL_FROM_IM_ID D_ID,
    IL_NUMBER D_INTEGER,
    IL_BILL_ID D_ID,
    IL_BILL_LINE_NO D_INTEGER,
    IL_INVENT_STR D_STR_20,
    IL_NAME D_STR_255,
    IL_YEDIZ D_STR_20,
    IL_DATE_BUY D_DATE,
    IL_COST CURR_TYPE,
    IL_KOLICH CURR_TYPE,
    IL_SUMMA CURR_TYPE,
    IL_FACT_KOLICH CURR_TYPE,
    IL_FACT_SUMMA CURR_TYPE,
    IL_MATOTVET_ID ID_TYPE,
    IL_WORKPLACE_ID ID_TYPE,
    IL_MEMO D_STR_255,
    IL_AUTO_FILL D_INTEGER,
    IL_IS_FOUND D_INTEGER,
    IL_KIND D_INTEGER,
    IL_FROM_MC_ID ID_TYPE,
    C_NAME_FIO D_STR_100,
    ML_MODEL D_STR_255)
AS
begin
  for
    select
      il_from_im_id,
      il_number,
      il_bill_id,
      il_bill_line_no,
      il_invent_str,
      il_name,
      il_yediz,
      il_date_buy,
      il_cost,
      il_kolich,
      il_summa,
      il_fact_kolich,
      il_fact_summa,
      il_matotvet_id,
      il_workplace_id,
      il_memo,
      il_auto_fill,
      il_is_found,
      il_kind,
      il_from_mc_id,
      c_name_fio
    from invlist
      left join clients
        on il_matotvet_id = c_id
    where il_from_im_id = :im_id
  into
      :il_from_im_id,
      :il_number,
      :il_bill_id,
      :il_bill_line_no,
      :il_invent_str,
      :il_name,
      :il_yediz,
      :il_date_buy,
      :il_cost,
      :il_kolich,
      :il_summa,
      :il_fact_kolich,
      :il_fact_summa,
      :il_matotvet_id,
      :il_workplace_id,
      :il_memo,
      :il_auto_fill,
      :il_is_found,
      :il_kind,
      :il_from_mc_id,
      :c_name_fio
  do
  begin
    select first 1
      ml_model
    from operbill
      inner join matlist
        on ml_id = ob_from_ml_id
    where ob_bill_id = :il_bill_id
      and ob_bill_line_no = :il_bill_line_no
    into
      :ml_model;
    suspend;
  end
end
