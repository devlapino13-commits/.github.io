CREATE PROCEDURE AAREP_PLANNING_AUTO_REPEAT(
    REP_REGION_ID INTEGER,
    REP_APLIC_DATE DATE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_AUTO_PERIOD INTEGER,
    REP_AUTO_KOEF DECIMAL(18,7))
RETURNS (
    MA_DATE_OPER DATE,
    MA_SUMMA_PLUS_OS DECIMAL(18,4),
    MA_SUMMA_PLUS_PERC DECIMAL(18,4),
    MA_SUMMA_PLUS_ADMIN DECIMAL(18,4),
    MA_ADMIN_PLUS DECIMAL(18,4),
    MA_ADMIN_MINUS DECIMAL(18,4),
    MA_TRANSFER_PLUS DECIMAL(18,4),
    MA_TRANSFER_MINUS DECIMAL(18,4),
    MA_SUMMA_MINUS_REPEAT DECIMAL(18,4),
    MA_SUMMA_MINUS_NEW DECIMAL(18,4),
    MA_SUMMA_MINIAPPNEW DECIMAL(18,4),
    MA_ADMIN_MINUS_PROCESS DECIMAL(18,4),
    MA_KF_ID DECIMAL(18,0),
    SH_AMOUNT DECIMAL(18,4),
    K_ID DECIMAL(18,0))
AS
declare variable reg_id_prior integer;
declare variable ma_id_prior integer;
declare variable ma_admin_perc decimal(18,7);
declare variable dd_new date;
declare variable sh_dd date;
declare variable k_id_prior decimal(18,0);
declare variable kf_id decimal(18,0);
declare variable kf_id1 decimal(18,0);
declare variable ma_is_new integer;
declare variable prod_id integer;
declare variable ma_pr_id_prior integer;
begin
  dd_new = :rep_date_begin + :rep_auto_period;
  reg_id_prior = -1;
  ma_id_prior = -1;
  ma_summa_minus_repeat = 0;
  ma_summa_minus_new = 0;
  ma_summa_plus_os = 0;
  ma_summa_plus_perc = 0;
  ma_summa_plus_admin = 0;
  ma_transfer_plus = 0;
  ma_transfer_minus = 0;
  ma_admin_plus = 0;
  ma_admin_minus = 0;
  ma_is_new = 0;
  ma_summa_miniappnew = 0;
  ma_admin_minus_process = 0;
  ma_pr_id_prior = -1;

  k_id_prior = 0;
  prod_id = 0;
  /* Список последних погашений */
  for select distinct k_from_kf_id, k_from_pr_id, ok_from_k_id, ok_sh_date, k_amount_approved * k_index_curr_rate
      from operkred, kontract, product, oksummary, region_list(:rep_region_id, 1, -1, 1)
      where ok_from_k_id = k_id and
            ok_type <> 8 and
            ok_ol_date <= :rep_aplic_date and
            ok_sh_date between :dd_new and :rep_date_end and
            ok_sh_id >= k_kolich_repay and
            ok_ol_is_active = 1 and
            k_from_pr_id = pr_id and
            oks_from_k_id = k_id and
            oks_date_begin <= :dd_new and
            oks_date_end >= :dd_new and
            oks_region_id = id_list and
            pr_category = 0 and
            ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
            and
            ok_from_c_id = 0 and
            oks_close_state = 0
      group by k_from_kf_id, k_from_pr_id, ok_from_k_id, ok_sh_date, k_amount_approved, k_index_curr_rate
      having(sum(ok_os + ok_sh_os) <> 0) or (sum(ok_perc + ok_sh_perc) <> 0)
      order by ok_from_k_id, ok_sh_date desc
      into :ma_kf_id, :prod_id, :k_id, :sh_dd, :sh_amount
  do
  begin
    kf_id = :ma_kf_id;
    if (:k_id_prior <> :k_id) then
    begin
      kf_id1 = 0;
      /* Наличие повторных заявок для групп с последними выплатами */
      if (sh_dd <= :rep_date_end) then
      begin
        select distinct ma_from_kf_id
        from miniapp, product
        where ma_date_oper >= :rep_aplic_date and
              ma_date_app <= :rep_aplic_date and
              ma_date_end >= :rep_aplic_date and
              ma_from_kf_id = :kf_id and
              ma_from_pr_id = pr_id and
              pr_category = 0
        into :kf_id1;
        if ((kf_id1 is null) or (kf_id1 = 0)) then
        begin
          kf_id1 = 0;
          ma_date_oper = :sh_dd;
          ma_summa_minus_repeat = :sh_amount + :sh_amount * rep_auto_koef / 100;

          if (prod_id <> ma_pr_id_prior or ma_pr_id_prior is null) then
          begin
            ma_admin_perc = 0;
            select prl_admin_perc_min
            from prodlogic
            where prl_from_pr_id = :prod_id and
                  prl_date_begin <= :ma_date_oper and
                  prl_date_end >= :ma_date_oper rows 1
            into :ma_admin_perc;
            if (ma_admin_perc is null) then
              ma_admin_perc = 0;
          end
          ma_summa_plus_admin = :ma_summa_minus_repeat * :ma_admin_perc / 100;
          suspend;
        end
      end
    end
    k_id_prior = :k_id;
  end
end
