CREATE PROCEDURE CM_AZ_0036_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable LO_NAME varchar(100);
begin
  for select C_NAME_FIO
      from KONTRACT K
      inner join KNAKTLIST KAL on (K_FROM_KF_ID = KAL_FROM_KF_ID and
            KAL.IS_DELETE = 0 and
            K_DISB_DATE between KAL_DATE and KAL_DATE_END)
      inner join CLIENTS on (KAL.KAL_NEW_KF_OFFICER = C_ID and
            C_NAME_FIO in ('Əzizova Pərvanə Natiq qızı', 'Tağıyev Süleyman Nizami oğlu', 'Quliyev Yaşar Ağabəy oğlu', 'Qurbanova Ülviyyə Bayram qızı', 'Muradov Cavad Elxan oğlu', 'Abdullayev Elmar Nütvəli oğlu', 'Bağırov Kamil Heydərəli oğlu', 'Quliyev Nicat Zəki oğlu', 'Məmmədov İsgəndər Muxtar oğlu', 'Rəcəbov Elməddin İbadət oğlu', 'Əbdürəhmanov Zakir Adil oğlu',
                           'Abdullazadə Urfan Natiq oğlu', 'Əmiraslanova Jalə Ərəstun qızı', 'Hümbətov Yaşar Fizuli oğlu', 'Vəliyev Üzeyir Cahangir oğlu', 'Məmmədzadə İbrahim Rövşən oğlu', 'Qulamzadə Elmin Sücaət oğlu', 'Əliyeva Çinarə İsmayıl qızı', 'Xəlilli Xalid Abdurahman oğlu', 'Məmmədova Zərifə Ağasəf qızı', 'Ramazanlı Asim Ehtibar oğlu',
                           'Seyidli Firqət Saməddin oğlu', 'Mirzəmmədli Nərmin Çingiz qızı', 'Qəhrəmanlı Nərmin Elman qızı', 'Zəkəryəyev Teymur Feyruz oğlu', 'Əlizadə Günay Bədəl qızı', 'Əzimov Dərgah Hüseyn oğlu', 'Həmidov Tural Mirzəhməd oğlu', 'Mehdiyeva Lalə Eldar qızı'))
      where K_DISB_DATE between '23.04.2019' and '30.04.2019' and
            K_ID = :K_ID
      into LO_NAME
  do
  begin
    res_error_is_critical = 1;
    res_error_text = 'Для ' || LO_NAME || ' выдачи приостановлены от 23.04.2019 до 30.04.2019';
    suspend;
  end
end
