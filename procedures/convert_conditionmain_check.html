CREATE PROCEDURE CONVERT_CONDITIONMAIN_CHECK
RETURNS (
    CDM_ID D_ID,
    UN_NAME D_STR_CODE)
AS
declare variable proc_name d_str_code;
declare variable db_id d_integer;
declare variable db_code d_str_10;
begin
  db_id = f_get_db_id();
  db_code = (case db_id
          when 81001 then 'RU'
          when 41701 then 'KG'
          when 94401 then 'AZ'
          end);

  for select cdm_id, un_name
  from conditionmain, username
  where cdm_sql_tmp_check = 1 and
        coalesce(cdm_from_un_id, 1000000000001) = un_id
  into cdm_id, un_name
  do
  begin
    proc_name = f_format('CM_%1_%2_SQL', :db_code, lpad(:cdm_id, 4, '0'));

    if (not exists(select * from rdb$procedures
    where rdb$procedure_name = :proc_name and
          rdb$package_name is null)) then
      suspend;
  end
end
