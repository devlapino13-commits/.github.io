CREATE PROCEDURE AAREP_CALC_AMORT_KOMMIS_OBOOK(
    REPORT_REGION_ID DECIMAL(18,0),
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_NEED_PERC INTEGER)
RETURNS (
    RES_AMOUNT_BEGIN DECIMAL(18,4),
    RES_AMOUNT_END DECIMAL(18,4),
    RES_AMOUNT_PERIOD DECIMAL(18,4),
    RES_KRED_AMOUNT DECIMAL(18,4),
    RES_KOMMIS_AMOUNT DECIMAL(18,4),
    AAREP_PRODUCT_NAME VARCHAR(50),
    AAREP_CURRENCY VARCHAR(3),
    AAREP_OFFICE VARCHAR(100),
    AAREP_PRODUCT_ID INTEGER,
    AAREP_OFFICE_ID INTEGER,
    AAREP_FOND VARCHAR(20),
    AAREP_ACCOUNT_PROFIT VARCHAR(20),
    AAREP_ACCOUNT_PAYED VARCHAR(20),
    RES_KOMMIS_PAYED DECIMAL(18,4))
AS
declare variable is_new_variant integer;
begin
  is_new_variant = 1;
  res_kred_amount = 0;
  if (is_new_variant = 1) then
  begin
    for select sum(res_amount_begin), sum(res_amount_end), sum(res_amount_period), sum(res_kommis_amount),
               sum(res_kommis_payed), aarep_product_name, aarep_currency, aarep_region_name, aarep_pr_id,
               aarep_region_id, aarep_k_fond, max(aarep_account_profit), max(aarep_account_payed)
        from aarep_calc_amort_kommis(:report_region_id, :report_date_begin, :report_date_end, :report_need_perc)
        group by aarep_product_name, aarep_currency, aarep_region_name, aarep_pr_id, aarep_region_id, aarep_k_fond
        into :res_amount_begin, :res_amount_end, :res_amount_period, :res_kommis_amount, :res_kommis_payed,
             :aarep_product_name, :aarep_currency, :aarep_office, :aarep_product_id, :aarep_office_id, :aarep_fond,
             :aarep_account_profit, :aarep_account_payed
    do
    begin
      suspend;
    end
  end

  if (is_new_variant = 0) then
  begin
    for select sum(res_amount_begin), sum(res_amount_end), sum(res_amount_period), sum(res_kred_amount),
               sum(res_kommis_amount), sum(res_kommis_payed), aarep_product_name, aarep_currency, aarep_region_name,
               aarep_pr_id, aarep_region_id, aarep_k_fond
        from aarep_calc_amort_kommis2(:report_region_id, :report_date_begin, :report_date_end, :report_need_perc)
        group by aarep_product_name, aarep_currency, aarep_region_name, aarep_pr_id, aarep_region_id, aarep_k_fond
        into :res_amount_begin, :res_amount_end, :res_amount_period, :res_kred_amount, :res_kommis_amount,
             :res_kommis_payed, :aarep_product_name, :aarep_currency, :aarep_office, :aarep_product_id,
             :aarep_office_id, :aarep_fond
    do
    begin
      select res_account
      from hp_make_kredit_acclist(:aarep_product_id, :aarep_office_id, 6) -- Начисленный доп.процент
      into :aarep_account_profit;

      select res_account
      from hp_make_kredit_acclist(:aarep_product_id, :aarep_office_id, 7) -- Выплаченный доп.процент
      into :aarep_account_payed;
      suspend;
    end
  end

end
