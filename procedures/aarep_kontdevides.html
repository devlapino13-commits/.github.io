CREATE PROCEDURE AAREP_KONTDEVIDES(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KD_K_ORIG DECIMAL(18,0),
    KD_K_NEW DECIMAL(18,0),
    KD_DATE DATE,
    KD_ORIG_KONT_NO VARCHAR(20),
    KD_ORIG_C_ID VARCHAR(18),
    KD_ORIG_C_NAME_FIO VARCHAR(100),
    KD_NEW_KONT_NO VARCHAR(20),
    KD_NEW_C_ID VARCHAR(18),
    KD_NEW_C_NAME_FIO VARCHAR(100),
    KD_NEW_AMOUNT DECIMAL(18,4),
    KD_DISB_DATE DATE,
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    KD_NEW_BALANCE DECIMAL(18,4))
AS
declare variable cm_default_currency varchar(3);
declare variable k_id decimal(18,0);
declare variable k_kontract_balance decimal(18,4);
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate decimal(18,4);
declare variable rep_currency varchar(3);
declare variable cr_rate decimal(18,4);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select company.cm_default_currency
  from company
  into :cm_default_currency;

  for select kd_k_orig, kd_k_new, kd_date, k_orig.k_kontract_no, cast(cli_orig.c_id as varchar(18)),
             cli_orig.c_name_fio, k_new.k_kontract_no, k_new.k_amount_approved, k_new.k_disb_date,
             cast(cli_new.c_id as varchar(18)), cli_new.c_name_fio, reg_code || ' ' || reg_name, parent_name,
             k_orig.k_index_curr_code, k_orig.k_index_curr_rate

      from kontdevides, kontract k_new, kontract k_orig, clients cli_orig, clients cli_new, region, region_list(:rep_region_id, 1, 3, 1), knaktlist
      where kd_date between :rep_date_begin and :rep_date_end and
            k_orig.k_id = kd_k_orig and
            k_new.k_id = kd_k_new and
            k_orig.k_clients_cashier_id = cli_orig.c_id and
            kd_c_id = cli_new.c_id and
            reg_unicode = id_list and
            kal_new_reg_id = id_list and
            knaktlist.is_delete = 0 and
            kd_date between kal_date and kal_date_end and
            k_orig.k_from_kf_id = kal_from_kf_id
      into :kd_k_orig, :kd_k_new, :kd_date, :kd_orig_kont_no, :kd_orig_c_id, :kd_orig_c_name_fio, :kd_new_kont_no,
           :kd_new_amount, :kd_disb_date, :kd_new_c_id, :kd_new_c_name_fio, :reg_name, :parent_name, :k_index_curr_code, :k_index_curr_rate

  do
  begin
    k_id = kd_k_new;
    select oks_balance
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin <= :kd_date and
          oks_date_end >= :kd_date and
          oks_close_state = 0
    into :k_kontract_balance;

    /*Если код валюты контракта не совпадает с валютой из таблицы  Company то вычисляет текущий курс валюты*/
    if (:k_index_curr_code <> :cm_default_currency) then
    begin
      kd_new_amount = kd_new_amount * k_index_curr_rate;/*вычисляет еквивалент выдачи в местной валюте*/
      rep_currency = '000';
      select cr_rate
      from opercurr
      where cr_from_cu_id = :k_index_curr_code and
            :kd_date between cr_date and cr_date_end /*курс на дату выделеия*/
      into :cr_rate;
    end
    else
    begin
      rep_currency = k_index_curr_code;
    end

    k_kontract_balance = case
                           when :rep_currency = '000' then (case
                                                              when :cr_rate < :k_index_curr_rate then :k_kontract_balance * k_index_curr_rate
                                                              else :k_kontract_balance * :cr_rate
                                                            end)
                           when :rep_currency = '---' then :k_kontract_balance
                           when :rep_currency = k_index_curr_code then :k_kontract_balance
                         end;

    if (:k_kontract_balance is not null) then
      kd_new_balance = :k_kontract_balance;
    else
      kd_new_balance = 0;

    suspend;
  end
end
