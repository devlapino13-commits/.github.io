CREATE PROCEDURE HP_REPAY_ALLOC_LITE(
    REP_DATE D_DATE,
    REP_K_ID D_ID,
    REPAY_AMOUNT D_CURRENCY,
    PAYED_PERC D_CURRENCY)
AS
declare variable pay_os d_currency;
begin
  update temp_vrepay
  set pay_os = 0,
      pay_perc = 0,
      pay_fine = 0,
      pay_tax = 0,
      pay_kommis = 0,
      pay_extra = 0,
      resh_perc = 0,
      sh_new_os = 0,
      sh_new_perc = 0,
      depo_amount = 0,
      pay_pereplat = 0;

  update temp_vrepay
  set pay_perc = -:payed_perc
  order by s_date
  rows 1;

  repay_amount = repay_amount + payed_perc;

  for select s_rec_no, zad_os - pay_os as zad_os
  from temp_vrepay
  where zad_os - pay_os > 0
  order by s_date, s_id
  as cursor cur
  do
  begin
    if (repay_amount > cur.zad_os) then
    begin
      pay_os = cur.zad_os;
      repay_amount = repay_amount - cur.zad_os;
    end else
    begin
      pay_os = repay_amount;
      repay_amount = 0;
    end

    if (pay_os > 0 ) then
    begin
      update temp_vrepay
      set pay_os = :pay_os
      where current of cur;
    end
  end

  if (repay_amount > 0) then
  begin
    update temp_vrepay
    set pay_pereplat = :repay_amount
    order by s_date desc, s_id desc
    rows 1;
  end
end
