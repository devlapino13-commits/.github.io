CREATE PROCEDURE HP_AGREEMENTS_CHEKS(
    C_ID D_ID,
    REP_DATE_END D_DATE,
    TYPE_CHECKS D_SMALLINT,
    AGREEMENTS_ID D_INTEGER)
RETURNS (
    RESULT D_SMALLINT)
AS
declare variable k_id_delinq_30 d_id;
declare variable k_id_last d_id;
declare variable k_close_date d_date;
declare variable ks_from_k_id d_id;
begin
--  TYPE_CHECKS
--  1) Проверка на просрочку 30+», Давность кредитования 24мес»,
--  2) Совпадение по террористам
  result = 0;
  if (type_checks = 1) then
  begin
    -- проверка на просрочку более 30 дней
    k_id_delinq_30 = null;
    select oks_from_k_id from knclient, oksummary
    where kc_from_c_id = :c_id
    and kc_amount_approved > 0
    and kc_from_k_id = oks_from_k_id
    and oks_close_state = 0
    and :rep_date_end between oks_date_begin and oks_date_end
    and :rep_date_end - oks_delinq_date > 30
    rows 1
    into k_id_delinq_30;

    -- •  С момента закрытия последнего договора займа на дату проверки прошло более 730 дней (2 лет);
    ks_from_k_id = null;
    select ks_from_k_id, ks_date_end
    from knclient, kontstate
    where kc_from_c_id = :c_id
    and kc_amount_approved > 0
    and kc_from_k_id = ks_from_k_id
    and ks_ol_is_delete = 0
    and ks_state = 1
    order by ks_date_end desc
    rows 1
    into ks_from_k_id, k_close_date;

    if (k_close_date < dateadd(month, -24, :rep_date_end)) then k_id_last = ks_from_k_id;
    else k_id_last = null;

--    if (k_id_delinq_30 is not null or k_id_last is not null) then
--    begin
      update or insert into agreements_checks (agc_from_ag_id, agc_kont_check_datetime, agc_delinq_k_id, agc_last_k_id)
      values (:agreements_id, current_timestamp, :k_id_delinq_30, :k_id_last)
      matching (agc_from_ag_id);
      result = 1;
--    end
  end else
  if (type_checks = 2) then
  begin
    result = 0;
    if (exists(select *
    from hp_check_client_fio_ft(:c_id, 1))) then
    begin
       result = 1;
    end
    update or insert into agreements_checks (agc_from_ag_id, agc_terr_match, agc_terr_check_datetime)
    values (:agreements_id, :result, current_timestamp)
    matching (agc_from_ag_id);
    result = 1;
  end
  suspend;
end
