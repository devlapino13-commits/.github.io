CREATE PROCEDURE HP_QR_VERIFY_INFO_LIST(
    C_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_SHOW_VERIFIED D_SMALLINT,
    REP_SHOW_DELETED D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    QRL_ID D_ID,
    QRL_GPS_LAT D_DOUBLE,
    QRL_GPS_LONG D_DOUBLE,
    QRL_DATETIME D_DATETIME,
    QRV_DATETIME D_DATETIME,
    LO_NAME_FIO D_STR_100,
    LO_REGION D_STR_100,
    QRV_GPS_LAT D_DOUBLE,
    QRV_GPS_LONG D_DOUBLE,
    GPS_DISTANCE D_CURRENCY,
    DIFF_SECOND D_INTEGER,
    RES_IMAGE_LIST D_STR_4096,
    QRF_REASON_STR D_STR_100,
    RES_KIND D_SMALLINT)
AS
declare variable rep_datetime_end d_datetime;
declare variable lo_c_id d_id;
declare variable qrf_from_bls_id d_id;
declare variable qrf_from_un_id d_id;
declare variable client_image_list d_str_4096;
declare variable qrf_reason d_smallint;
begin
  -- Процедура возвращает список QR-верификаций клиента
  if (c_id is null) then
    exit;

  -- Время на конец дня rep_date_begin
  rep_datetime_end = dateadd(millisecond, -1, rep_date_end + 1);

  -- Фото клиента
  select res_image_list
  from hp_get_client_doc_images(:c_id, 1, 1, '', :un_id)
  into client_image_list;

  for select
    qrl_id, qrl_gps_lat, qrl_gps_long, qrl_datetime,
    qrv_from_c_id, qrv_gps_lat, qrv_gps_long, qrv_datetime,
    f_gps_distance(qrl_gps_lat, qrl_gps_long, qrv_gps_lat, qrv_gps_long),
    datediff(second, qrl_datetime, qrv_datetime),
    qrf_from_un_id, qrf_from_bls_id, qrf_reason,
    case
      when qrl_state = 2 and qrf_from_qrl_id is not null then 3
      when qrl_state = 2 then 2
      when qrv_from_qrl_id is not null then 1
      else 0
    end
  from qr_list
    left join qr_verify on qrl_id = qrv_from_qrl_id
    left join qr_photo on qrf_from_qrl_id = qrl_id
  where
    (qrl_is_delete = 0 or :rep_show_deleted = 1) and
    (qrl_state = 2 or :rep_show_verified = 0) and
    qrl_datetime between :rep_date_begin and :rep_datetime_end and
    qrl_from_c_id = :c_id
  into qrl_id, qrl_gps_lat, qrl_gps_long, qrl_datetime,
    lo_c_id, qrv_gps_lat, qrv_gps_long, qrv_datetime,
    gps_distance, diff_second, qrf_from_un_id, qrf_from_bls_id,
    qrf_reason, res_kind
  do
  begin
    res_image_list = '';
    lo_name_fio = '';
    lo_region = '';
    qrf_reason_str = '';

    if (res_kind = 3) then
    begin
      gps_distance = 0;
    end
    if (qrf_from_un_id is not null) then
    begin
      -- При фото верификации верификатор - фотограф
      select un_from_c_id from username
      where un_id = :qrf_from_un_id
      into lo_c_id;
    end

    select c_name_fio
    from clients
    where c_id = :lo_c_id
    into lo_name_fio;

    -- Офис пользователя
    select reg_name
    from hr_pstatus, region
    where current_date between ps_date_begin and ps_date_end and
          ps_status = 1 and
          ps_is_working = 1 and
          ps_from_c_id = :lo_c_id
          and ps_from_reg_id = reg_unicode
    into lo_region;

    -- Фото верификации
    if (qrf_from_bls_id is not null) then
    begin
      res_image_list = f_image_item('blobstore', qrf_from_bls_id, 'Фото верификации', un_id, 0);
      res_image_list = f_format('{"caption": "Клиент и специалист", "blobs":[%1]}', res_image_list);
    end

    -- Собираем все фото
    if (client_image_list <> '' and res_image_list <> '') then
      res_image_list = client_image_list || ', ' || res_image_list;
    if (res_image_list <> '') then
      res_image_list = '[' || res_image_list || ']';

    select rd_name
    from fm_get_ref('qr_reason')
    where rd_id = :qrf_reason
    into qrf_reason_str;

    suspend;
  end
end
