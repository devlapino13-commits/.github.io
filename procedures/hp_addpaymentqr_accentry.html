CREATE PROCEDURE HP_ADDPAYMENTQR_ACCENTRY(
    ESM_ID D_ID)
RETURNS (
    OL_ID D_ID,
    ERR_TEXT D_STR_255)
AS
declare variable esm_trans_id d_str_20;
declare variable esm_date d_date;
declare variable c_id d_id;
declare variable esm_amount d_currency;
declare variable kpt_id d_integer;
declare variable un_id d_id;
declare variable kqa_id d_integer;
declare variable kqa_reg_id d_integer;
declare variable kqa_detail d_str_long;
declare variable is_ipf boolean;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable res_reg_id d_integer;
declare variable res_acc_dt d_str_20;
declare variable res_acc_kt d_str_20;
declare variable res_line_no d_integer;
declare variable ol_tmp_id d_id;
begin
  err_text = '';
  for
    select
      esm_trans_id,
      esm_date,
      kpt_id,
      kpt_seller_id,
      kpt_amount,
      kpt_from_un_id,
      kqa_id,
      kqa_reg_id,
      kqa_detail,
      ag_from_rd_id =
        f_get_rd_id('AgreementKinds.IPF_Sellers')
    from extsysmain
      inner join kontpaytasks
        on kpt_from_esm_id = esm_id
      inner join kontpaytasks_qr_add
        on kqa_kpt_id = kpt_id
          and kqa_ol_id is null
      inner join agreements
        on ag_id = kqa_ag_id
    where esm_id = :esm_id
      and esm_project = 'qr'
      and esm_action = 'pay'
      and esm_status = 0
  into
    :esm_trans_id,
    :esm_date,
    :kpt_id,
    :c_id,
    :esm_amount,
    :un_id,
    :kqa_id,
    :kqa_reg_id,
    :kqa_detail,
    :is_ipf
  do
  begin
    acc_dt = (
      select al_id
      from acc_list
      where al_from_reg_id = :kqa_reg_id
        and al_currency = '417'
        and al_account = '21199/99');
    acc_kt = (
      select al_id
      from acc_list
      where al_from_reg_id = 1090100
        and al_currency = '417'
        and al_account = '10024/117');

    for
      select
        res_reg_id,
        res_acc_dt,
        res_acc_kt,
        res_line_no
      from aarep_loronostro_link(:kqa_reg_id, :acc_dt, :acc_kt)
    into
      :res_reg_id,
      :res_acc_dt,
      :res_acc_kt,
      :res_line_no
    do
    begin
      insert into operlist (
        ol_parent_id,
        ol_from_reg_id,
        ol_oper_type,
        ol_date,
        ol_doc_no,
        ol_doc_kind,
        ol_document_client,
        ol_memo,
        ol_user_id
       )
      values (
        :ol_id,
        :res_reg_id,
        2,
        current_date,
        :kqa_id,
        9,
        :c_id,
        :kqa_detail,
        :un_id
      )
      returning ol_id
      into :ol_tmp_id;

      if (res_line_no = 1)
        then ol_id = ol_tmp_id;

      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_tmp_id,
        '417',
        1,
        :res_acc_dt,
        :res_acc_kt,
        :esm_amount,
        :esm_amount
      );
    end

    update kontpaytasks_qr_add set
      kqa_ol_id = :ol_id
    where kqa_id = :kqa_id
      and kqa_ol_id is null;
  end

  if (ol_id is null) then
    err_text = f_format('Ошибка построения бух.проводки при оплате по QR(%1)',
      coalesce(:kqa_id, '?'));
  suspend;
end
