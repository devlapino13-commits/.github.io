CREATE PROCEDURE AAREP_SA_TABEL_PRINT_MTEMP(
    REPORT_REGION INTEGER,
    REPORT_YEAR INTEGER,
    REPORT_MONTH INTEGER)
RETURNS (
    SAT_FROM_C_ID DECIMAL(18,0),
    SAT_YEAR INTEGER,
    SAT_MONTH INTEGER,
    SAT_OFFICE_ID INTEGER,
    SAT_OFFICE_NAME VARCHAR(100),
    SAT_TOTAL_NORMA INTEGER,
    SAT_TOTAL_HOURS INTEGER,
    SAT_TOTAL_CALC INTEGER,
    SAT_PERSONAL_FIO VARCHAR(100),
    SAT_TOTAL_VOCATION INTEGER,
    SAT_DAYS_NORMA DECIMAL(18,4),
    SAT_DAYS_TOTAL DECIMAL(18,4),
    SAT_DAYS_VOCATION DECIMAL(18,4),
    SAT_DAYS_CALC DECIMAL(18,4))
AS
declare variable st_staff integer;
declare variable st_client integer;
declare variable sat_year_month integer;
declare variable tnorm integer;
declare variable tnorm_days decimal(18,4) = 0;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select sum(scd_work_hours), sum(iif(scd_work_hours <> 0, 1, 0))
  from sa_calendays
  where scd_year_month = (:report_year * 100 + :report_month)
  into :tnorm, :tnorm_days;
  if (tnorm is null) then
    tnorm = 0;
  if (tnorm_days is null) then
    tnorm_days = 0;

  for select sat_from_c_id, sat_year_month, sat_office_id, reg_name, c_name_fio,
             sum(iif(sat_from_sawt_id = 0, :tnorm, 0)), sum(sat_total_hours * sawt_hours_real), sum(0),
             sum(sat_total_calc),

             sum(iif(sat_from_sawt_id = 0, :tnorm_days, 0)), sum(sat_days_total * sawt_hours_real), sum(0),
             sum(sat_days_calc)
      from sa_tabel, clients, region_list(:report_region, 1, -1, 1), region, sa_worktype
      where sat_from_c_id = c_id and
            sat_year_month = (:report_year * 100 + :report_month) and
            (sat_office_id = id_list) and
            sat_is_ready = 1 and
            sawt_id = sat_from_sawt_id and
            sawt_base_type <> 4 and
            sat_office_id = reg_unicode
      group by sat_from_c_id, sat_year_month, sat_office_id, reg_name, c_name_fio
      union
      select sat_from_c_id, sat_year_month, sat_office_id, reg_name, c_name_fio,
             sum(iif(sat_from_sawt_id = 0, :tnorm, 0)), sum(0), sum(sat_total_hours * sawt_hours_real),
             sum(sat_total_calc),

             sum(iif(sat_from_sawt_id = 0, :tnorm_days, 0)), sum(0), sum(sat_days_total * sawt_hours_real), sum(sat_days_calc)

      from sa_tabel, clients, region_list(:report_region, 1, -1, 1), region, sa_worktype
      where sat_from_c_id = c_id and
            sat_year_month = (:report_year * 100 + :report_month) and
            (sat_office_id = id_list) and
            sat_is_ready = 1 and
            sawt_id = sat_from_sawt_id and
            sawt_base_type = 4 and
            sat_office_id = reg_unicode
      group by sat_from_c_id, sat_year_month, sat_office_id, reg_name, c_name_fio
      into :sat_from_c_id, :sat_year_month, :sat_office_id, :sat_office_name, :sat_personal_fio, :sat_total_norma,
           :sat_total_hours, :sat_total_vocation, :sat_total_calc,

           :sat_days_norma, :sat_days_total, :sat_days_vocation, :sat_days_calc
  do
  begin
    sat_month = mod(sat_year_month, 100);
    sat_year = (sat_year_month - sat_month) / 100;
    suspend;
  end
end
