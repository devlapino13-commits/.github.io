CREATE PROCEDURE HP_PVNOPER_CASHING(
    PO_ID D_INTEGER,
    CS_ID D_INTEGER = null)
AS
declare variable acc_dt d_str_20;
declare variable c_id d_id;
declare variable ol_doc_no d_integer;
declare variable ol_memo d_str_long;
declare variable card_id d_integer;
declare variable amount d_currency;
declare variable un_id d_id;
declare variable office_id d_integer;
declare variable acc_kt d_str_20;
declare variable ol_id d_id;
declare variable oper_date d_date;
declare variable card_acc_no d_str_code;
declare variable card_disb_office_id d_integer;
declare variable acc_bish_office d_str_code;
declare variable bish_office_id d_integer;
declare variable ol_id_child d_id;
declare variable acc_bo_office_cashing d_str_code;
declare variable acc_bo_office_disb_card d_str_code;
declare variable ol_id1 d_id;
declare variable res_reg_id d_integer;
declare variable res_acc_dt d_str_20;
declare variable res_acc_kt d_str_20;
declare variable total_line d_integer;
declare variable res_line_no d_integer;
begin
  if (po_id is null) then
    exception e_message 'po_id не определен';

  select po_card_id,
         po_from_c_id,
         po_amount,
         po_un_id,
         po_office_id,
         po_acc_kt,
         po_date,
         po.po_card_office_id
  from pvnopers po
  where po.po_id = :po_id
  into :card_id,
       :c_id,
       :amount,
       :un_id,
       :office_id,
       :acc_kt,
       :oper_date,
       :card_disb_office_id;

  if (office_id = 1090100) then
    exception e_message 'ПВН операции не могут быть осуществлены в Бишкек ГО';

  if (card_disb_office_id is null) then
  begin
      select cpl.cpl_from_reg_id
      from  cardlist cal
      inner join cardplace cpl on cal.cal_last_cpl_id = cpl.cpl_id
      where cal.cal_card_id = :card_id
      into :card_disb_office_id;
  end

  if (card_disb_office_id is null) then
    exception e_message 'Офис выдачи карты не определен';

   -- Операции по ПВН
   select al_id
   from acc_list
   where al_from_reg_id = :card_disb_office_id
     and al_currency = 417
     and al_account = '20001/1'
   into :acc_dt;

  if (acc_dt is null) then
    exception e_message 'Счет Дт не определен';

  select cca.cca_code
  from cardlist cal
    inner join cardclientacc cca on cal.cal_from_cca_id = cca.cca_id
  where cal.cal_card_id = :card_id
  into :card_acc_no;

  if (card_acc_no is null) then
    exception e_message 'Карт счет не определен';

--------------------------------------------------------------------------------
  ol_id = null;

  select num
  from last_doc_number(:oper_date, 0, :office_id) -- 0 - т.к. нас интерсует расходный кассовый ордер
  into :ol_doc_no;

  for
    select
      res_reg_id,
      res_acc_dt,
      res_acc_kt,
      total_line,
      res_line_no
    from aarep_loronostro_link(:card_disb_office_id, :acc_dt, :acc_kt)
  into
    :res_reg_id,
    :res_acc_dt,
    :res_acc_kt,
    :total_line,
    :res_line_no
  do
  begin
    insert into operlist (
      ol_parent_id,
      ol_from_reg_id,
      ol_oper_type,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_document_client,
      ol_memo,
      ol_user_id
    )
    values (
      :ol_id,
      :res_reg_id,
      2,
      :oper_date,
      :ol_doc_no,
      0,
      :c_id,
      f_format('Обналичивание средств по ПВН: ID %1 (сч.%2)', :card_id, :card_acc_no),
      :un_id
    )
    returning ol_id, ol_memo
    into :ol_id1, :ol_memo;

    if (res_line_no = 1)
      then ol_id = ol_id1;

    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_detail,
      ob_summa,
      ob_summa_val,
      ob_client_dt
    )
    values (
      :ol_id1,
      '417',
      1,
      :res_acc_dt,
      :res_acc_kt,
      :ol_memo,
      :amount,
      :amount,
      :c_id
    );
  end
--------------------------------------------------------------------------------

  update pvnopers po
  set po.po_status = f_enum('pvnopers_status', 'operation_ok', 'id'),
      po.po_from_ol_id = :ol_id1 -- последняя операция, иначе пустой расходник
  where po.po_id = :po_id;

  if (:cs_id is not null) then
    insert into cardsyncopers (cso_from_cs_id, cso_from_ol_id, cso_after_exec)
    values (:cs_id, :ol_id, 1);
end
