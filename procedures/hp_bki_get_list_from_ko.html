CREATE PROCEDURE HP_BKI_GET_LIST_FROM_KO(
    C_ID D_ID,
    REP_DATE D_DATE = null,
    START_DATE_IN D_DATE = null,
    INCLUDE_NBKI_KO D_BOOL = null)
RETURNS (
    UUID D_STR_CODE,
    CRED_DATE D_DATE,
    CRED_SUM D_CURRENCY,
    CRED_PARTNER_TYPE_STR D_STR_CODE,
    CRED_DEBT_SUM D_CURRENCY,
    CRED_DEBT_OP_SUM D_CURRENCY,
    CRED_DEBT_PERCENT_SUM D_CURRENCY,
    IS_IN_KO D_BOOL,
    IS_IN_SSP D_BOOL,
    IS_OUR D_BOOL,
    SSP_SUMM D_CURRENCY,
    ERROR_TEXT D_STR_100,
    CRED_SUM_OVERDUE D_CURRENCY)
AS
declare variable bls_id d_id;
  declare variable ssp_bls_id d_id;
  declare variable jso d_integer;
  declare variable ssp_jso d_integer;
  declare variable ko_jso d_integer;
  declare variable json_full d_blob_text;
  declare variable kbki_jso d_integer;
  declare variable bki_jso d_integer;
  declare variable loan_jso d_integer;
  declare variable ssp_calc_date d_date;
  declare variable start_date d_date;
begin
  if (rep_date is null) then
    rep_date = current_date;

  if (include_nbki_ko is null) then
    include_nbki_ko = 0;

  error_text = '';
  start_date = start_date_in;

  -- ССП
  if (start_date is null) then
    select :rep_date - eic_info_actual_days
    from extinfocfg
    where eic_code = 'bki_ssp_40'
    into start_date;

  select eil_from_bls_id
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id
    and eil_from_eic_id = eic_id
    and eic_code = 'bki_ssp_40'
    and eil_status = 2
    and eil_date between :start_date and :rep_date
  order by eil_date desc, eil_id
  rows 1
  into ssp_bls_id;

  if (ssp_bls_id is null) then
  begin
    error_text = 'Ошибка получения КБКИ: не получен отчет СП';
    suspend;
    exit;
  end

  select blx_blob
  from hp_get_blobex_blob(:ssp_bls_id)
  into json_full;

  if (json_full is null) then
  begin
    error_text = 'Ошибка получения КБКИ: json пустой';
    suspend;
    exit;
  end

  ssp_jso = json.obj_create();
  jso = json.parse(json_full);

  for select jso
  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'КБКИ'), 1)
  into kbki_jso
  do
  begin
    for select jso
    from hp_bki_json_get_list(json.obj_get_arr(:kbki_jso, 'Обязательства.БКИ'), 1)
    into bki_jso
    do
    begin
      for select jso
      from hp_bki_json_get_list(json.obj_get_arr(:bki_jso, 'Договор'), 1)
      into loan_jso
      do
      begin
        uuid = json.obj_get_val(loan_jso, 'УИД', '');

        if (uuid = '') then
          continue;

        ssp_calc_date = json.obj_get_val(loan_jso, 'СреднемесячныйПлатеж.ДатаРасчета', '01.01.1900');

        if (ssp_calc_date < json.obj_get_val(ssp_jso, uuid || '.calc_date', current_date)) then
        begin
          json.obj_set_s(ssp_jso, uuid || '.calc_date', ssp_calc_date);
          json.obj_set_s(ssp_jso, uuid || '.summ', json.obj_get_val(loan_jso, 'СреднемесячныйПлатеж.value', 0));

          if (json.contains(ssp_jso, 'arr') = 0) then
            json.arr_create(ssp_jso, 'arr');

          json.arr_set_s(json.obj_get_arr(ssp_jso, 'arr'), -1, uuid);
        end 
      end 
    end
  end 

  -- КО СБ
  ko_jso = json.obj_create();
  start_date = start_date_in;

  if (start_date is null) then
    select :rep_date - eic_info_actual_days
    from extinfocfg
    where eic_code = 'equifax_40'
    into start_date;

  select eil_from_bls_id
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id
    and eil_from_eic_id = eic_id
    and eic_code = 'equifax_40'
    and eil_status = 2
    and eil_date between :start_date and :rep_date
  order by eil_date desc, eil_id
  rows 1
  into bls_id;

  if (bls_id is null) then
    error_text = 'Кредитная история не получена';
  else
  begin
    for select coalesce(cred_uid, cred_id), cred_partner_type_str, cred_date, cred_sum, cred_sum_debt, cred_debt_op_sum, cred_debt_percent_sum,
        average_payment, cred_sum_overdue
    from aarep_equifax_cred(:bls_id, 'equifax_40')
    order by cred_date
    into uuid, cred_partner_type_str, cred_date, cred_sum, cred_debt_sum, cred_debt_op_sum, cred_debt_percent_sum,
        ssp_summ, cred_sum_overdue
    do
    begin
      is_in_ko = 1;
      is_in_ssp = 0;
      is_our = 0;
      json.obj_set_s(ko_jso, uuid, '1');
  
      if (char_length(uuid) = 38 and exists(select 1 from hp_find_cbr_uuid(:uuid) where obj_id is not null)) then
      begin
        is_our = 1;
        cred_partner_type_str = 'М Булак';
      end 
  
        if (json.contains(ssp_jso, uuid) = 1
            and ssp_summ = json.obj_get_val(ssp_jso, uuid || '.summ', ssp_summ))
        then
          is_in_ssp = 1;

      suspend;
    end
  end

  -- КО НБКИ
  if (include_nbki_ko = 1) then
  begin
    start_date = start_date_in;
  
    if (start_date is null) then
      select :rep_date - eic_info_actual_days
      from extinfocfg
      where eic_code = 'nbki_40'
      into start_date;
  
    select eil_from_bls_id
    from extinfolist, extinfocfg
    where eil_from_c_id = :c_id
      and eil_from_eic_id = eic_id
      and eic_code = 'nbki_40'
      and eil_status = 2
      and eil_date between :start_date and :rep_date
    order by eil_date desc, eil_id
    rows 1
    into bls_id;
  
    if (bls_id is null) then
      error_text = 'Кредитная история (Источник 2) не получена';
    else
    begin
      for select cred_uid, cred_partner_type_str, cred_date, cred_sum, cred_sum_debt, cred_debt_op_sum, cred_debt_percent_sum,
          average_payment, cred_sum_overdue
      from aarep_ko_nbki_cred(:bls_id)
      where cred_uid is not null
      order by cred_date
      into uuid, cred_partner_type_str, cred_date, cred_sum, cred_debt_sum, cred_debt_op_sum, cred_debt_percent_sum,
          ssp_summ, cred_sum_overdue
      do
      begin
        -- Если уже показывали, то пропускаем
        if (json.contains(ko_jso, uuid) = 1) then
          continue;

        is_in_ko = 1;
        is_in_ssp = 0;
        is_our = 0;
        json.obj_set_s(ko_jso, uuid, '1');
    
        if (char_length(uuid) = 38 and exists(select 1 from hp_find_cbr_uuid(:uuid) where obj_id is not null)) then
        begin
          is_our = 1;
          cred_partner_type_str = 'М Булак';
        end 
    
        if (json.contains(ssp_jso, uuid) = 1
            and ssp_summ = json.obj_get_val(ssp_jso, uuid || '.summ', ssp_summ))
        then
          is_in_ssp = 1;

        suspend;
      end
    end
  end

  -- оставшиеся в ССП
  cred_partner_type_str = null;
  cred_date = null;
  cred_sum = null;
  cred_debt_sum = null;
  cred_debt_op_sum = null;
  cred_debt_percent_sum = null;
  cred_sum_overdue = null;

  for select val
  from hp_bki_json_get_list(json.obj_get_arr(:ssp_jso, 'arr'), 0)
  into uuid
  do
  begin
    -- Если уже показывали, то пропускаем
    if (json.contains(ko_jso, uuid) = 1) then
      continue;

    is_in_ko = 0;
    is_in_ssp = 1;
    is_our = 0;
    cred_partner_type_str = 'МФО';
    ssp_summ = json.obj_get_val(ssp_jso, uuid || '.summ', 0);

    if (exists(select 1 from hp_find_cbr_uuid(:uuid) where obj_id is not null)) then
    begin
      is_our = 1;
      cred_partner_type_str = 'М Булак';
    end 

    suspend;
  end 
end
