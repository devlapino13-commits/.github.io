CREATE PROCEDURE AAREP_FOR_DRR(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    C_NAME_FIO STR_100,
    REG_NAME STR_100,
    BRANCH_NAME STR_100,
    LO_STAGE_MONTH INTEGER,
    AAREP_PORT_KOLICH_BEGIN INTEGER,
    AAREP_PORT_BALANCE_BEGIN INTEGER,
    AAREP_PORT_KOLICH_ALL INTEGER,
    AAREP_PORT_BALANCE_ALL INTEGER,
    AAREP_DISB_KOLICH INTEGER,
    AAREP_DISB_KOLICH_NEW INTEGER,
    AAREP_RETURNMENT_KOEF PERC_TYPE,
    AAREP_BONUS_PLUS CURR_TYPE,
    AAREP_BONUS_MINUS CURR_TYPE,
    AAREP_BONUS_TOTAL CURR_TYPE,
    AAREP_BONUS_IDEAL CURR_TYPE)
AS
declare variable C_ID ID_TYPE;
declare variable REP_DATE_BEGIN_YESTERDAY date;
declare variable REP_DATE_BEGIN date;
declare variable K_ID D_ID; /* ID контракта */
declare variable CLI_ID D_ID; /* ID клиента из контракта */
declare variable IS_REPEAT_CLIENT D_BOOLEAN; /* Является ли клиент повторным */
begin
-- отчет для  Мирзаева Эркин ДРР ООО «МКК М Булак», письмо от 16.08.2019
-- все данные взяты из бонуса СВЗ
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;
  rep_date_begin_yesterday = rep_date_begin - 1;

  for select c_name_fio,
             reg_name,
             branch_name,
             lo_stage_month,
             aarep_port_kolich_all,
             aarep_port_balance_all,
             aarep_returnment_koef,
             aarep_bonus_plus,
             aarep_bonus_minus,
             aarep_bonus_total,
             c_id,
             aarep_bonus_ideal
--      from cb_ru_lo_201410_one_lo(:rep_region_id, 0, :rep_date_begin, :rep_date_end, 0)
      from CB_RU_LO_202011_ONE_LO_DISB(:rep_region_id, 0, :rep_date_begin, :rep_date_end, 0)
        into c_name_fio,
             reg_name,
             branch_name,
             lo_stage_month,
             aarep_port_kolich_all,
             aarep_port_balance_all,
             aarep_returnment_koef,
             aarep_bonus_plus,
             aarep_bonus_minus,
             aarep_bonus_total,
             c_id,
             aarep_bonus_ideal
  do
  begin
    aarep_port_balance_begin = 0;
    aarep_port_kolich_begin = 0;
    select sum(oks_balance), sum(k_kolich_client_approved)
    from oksummary, kontract kont
    where oks_date_begin <= :rep_date_begin_yesterday and
          oks_date_end >= :rep_date_begin_yesterday and
          oks_kc_id = :c_id and
          oks_balance <> 0 and
          oks_from_k_id = k_id and
          (:rep_date_begin_yesterday - oks_delinq_date <= 360) and
          oks_close_state = 0
    into aarep_port_balance_begin, aarep_port_kolich_begin;
    aarep_port_balance_begin = coalesce(aarep_port_balance_begin,0);
    aarep_port_kolich_begin = coalesce(aarep_port_kolich_begin,0);

    /*
    aarep_disb_kolich = 0;
    select sum(k_kolich_client_approved) from kontstate, kontract
    where ks_kc_id = :c_id
    and ks_ol_is_delete = 0
    and ks_state = 1
    and ks_date_begin between :rep_date_begin and :rep_date_end
    and ks_from_k_id = k_id
    into aarep_disb_kolich;

    aarep_disb_kolich = coalesce(aarep_disb_kolich,0);
    */

    aarep_disb_kolich = 0;
    aarep_disb_kolich_new = 0;

    -- RFCRU-3690: Добавление показателя выдачи новым клиентам
    for
        select
            k_id
        from kontstate, kontract
        where
            ks_kc_id = :c_id
            and ks_ol_is_delete = 0
            and ks_state = 1
            and ks_date_begin between :rep_date_begin and :rep_date_end
            and ks_from_k_id = k_id
        into
            :k_id
    do
    begin
        for
            select
                c.c_id
            from knclient kc
            inner join clients c on kc.kc_from_c_id = c.c_id
            where
                kc.kc_from_k_id = :k_id
                and kc.kc_amount_approved > 0
            into
                :cli_id
        do
        begin
            -- Определяем, брал ли клиент хоть раз займ до текущей даты. Если брал, то он повторный
            is_repeat_client = exists(
                select 1
                from kontstate ks
                inner join knclient kcc on kcc.kc_from_k_id = ks.ks_from_k_id
                where
                    ks.ks_date_begin <= :rep_date_end
                    and ks.ks_state = 1
                    and ks.ks_ol_is_delete = 0
                    and kcc.kc_from_c_id = :cli_id
                    and kcc.kc_amount_approved > 0
                    and ks.ks_from_k_id <> :k_id -- Не считаем текущий займ клиента, только другие, иначе все клиенты всегда будут считаться повторными
            );
    
            if (not is_repeat_client) then
                aarep_disb_kolich_new = aarep_disb_kolich_new + 1; -- Выдача только новым

            aarep_disb_kolich = aarep_disb_kolich + 1; -- Всего выдач
    
            is_repeat_client = null;
        end
    end


    aarep_disb_kolich = coalesce(aarep_disb_kolich, 0);
    aarep_disb_kolich_new = coalesce(aarep_disb_kolich_new, 0);

    suspend;
  end
end
