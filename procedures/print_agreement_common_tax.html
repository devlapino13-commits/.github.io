CREATE PROCEDURE PRINT_AGREEMENT_COMMON_TAX(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    C_ID D_ID,
    AGREEMENT_TYPE D_INTEGER)
RETURNS (
    C_INN D_STR_20,
    AG_CAPTION D_STR_255,
    ADL_ADDRESS_STR D_STR_255,
    CDS_DOC_ISSUED_BY D_STR_255,
    B_NAME D_STR_255,
    C_BIK D_STR_10,
    C_ACCOUNT D_STR_CODE,
    ML_NAME D_STR_255,
    UNIT_MEASURE D_STR_20,
    KML_COST DECIMAL(18,2),
    QTY D_INTEGER,
    KML_TOTAL DECIMAL(18,2),
    COST_WITHOUT_TAX DECIMAL(18,2),
    COST_WITH_TAX DECIMAL(18,2),
    TAX_PERCENT D_PERCENT,
    COST_WITHOUT_TAX_TOTAL DECIMAL(18,2),
    COST_WITH_TAX_TOTAL DECIMAL(18,2),
    KML_SUM_FINAL DECIMAL(18,2),
    KML_COUNT D_INTEGER,
    FEG_CODE D_STR_10,
    PRINT_DATE D_DATE)
AS
declare variable ag_id d_id;
declare variable adl_id d_id;
declare variable kontract_obj_kind d_integer;
declare variable agreement_no d_str_255;
begin
  if (coalesce(rep_region_id, 0) = 0 or
      rep_date_begin is null or
      rep_date_end is null or
      coalesce(c_id, 0) = 0 or
      coalesce(agreement_type, 0) = 0) then
    exception e_message('Необходимо заполнить все поля!');

  -- Сделано потому что в разных офисах у контрагентов могут быть разные процентные ставки единого налога
  if (rep_region_id = 1000000) then
    exception e_message('Выгрузка на Кыргызстан невозможна!');

  print_date = rep_date_end;

  select rd.rd_name
  from refdata rd
  where rd.rd_id = :agreement_type
  into agreement_no;

  select ag.ag_id, ag.ag_caption, ag.ag_tax_percent
  from agreements ag
  inner join region_list(:rep_region_id, 1, -1, 0) t on t.id_list = ag.ag_from_reg_id
  where ag.ag_object_kind = f_objkind('clients', 'id') and
        ag.ag_object_id = :c_id and
        :rep_date_end between ag.ag_date_begin and coalesce(ag.ag_date_end, '31.12.9999') and
        ag.ag_status = 1 and
        ag.ag_agreement_no = :agreement_no
  into ag_id, ag_caption, tax_percent;

  select cds.cds_memo_3
  from clientdocs cds
  inner join passkind p on p.pk_id = cds.cds_from_pk_id
  inner join passdoctype pdt on pdt.pdt_id = p.pk_from_pdt_id
  where cds.cds_from_c_id = :c_id and
        cds.cds_is_delete = 0 and
        cds.cds_verify_status = 2 and
        :rep_date_end between cds.cds_doc_date_begin and cds.cds_doc_date_end and
        pdt.pdt_code in ('certificate_ip', 'certificate_chp') and
        p.pk_is_delete = 0
  order by pdt_code desc
  rows 1
  into cds_doc_issued_by;

  select adc.adc_from_adl_id
  from addrclient adc
  where adc.adc_from_c_id = :c_id and
        adc.adc_address_type = 1 --юрид. адрес
  order by adc.adc_datetime desc
  rows 1
  into adl_id;

  select adl.adl_address_str
  from addrlist adl
  where adl.adl_id = :adl_id
  into adl_address_str;

  select cl.c_account, cl.c_org_from_b_id, b.b_name, cl.c_inn
  from clients cl
  inner join banks b on b.b_id = cl.c_org_from_b_id
  where cl.c_id = :c_id
  into c_account, c_bik, b_name, c_inn;

  kontract_obj_kind = f_objkind('kontract', 'id');

  cost_without_tax_total = 0;
  cost_with_tax_total = 0;
  kml_sum_final = 0;

  for select ml.ml_name,
             kml.kml_cost,
             kml.kml_total,
             (kml_total / (:tax_percent + 100) * 100) as cost_without_tax,
             kml_total - (kml_total / (:tax_percent + 100) * 100) as cost_with_tax,
             yl.yl_code,
             count(*) over(partition by ag_object_id),
             sum(kml_total / (:tax_percent + 100) * 100) over(partition by ag_object_id),
             sum((kml_total - (kml_total / (:tax_percent + 100) * 100))) over(partition by ag_object_id),
             sum(kml_total) over(partition by ag_object_id),
             kml_count,
             feg.feg_code
      from kontstate ks
      inner join kontract kt on kt.k_id = ks.ks_from_k_id
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      inner join kontmat kml on kml.kml_object_id = kt.k_id
      inner join agreements ag on ag.ag_id = kml.kml_from_ag_id
      inner join matlist ml on ml.ml_id = kml.kml_from_ml_id
      inner join yedizlist yl on yl.yl_id = ml.ml_from_yl_id
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(ks.ks_region_id as d_integer)
      inner join foreign_econom_goods_ref feg on feg.feg_id = kml.kml_from_feg_id
      where ks.ks_state = 1 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_ol_is_delete = 0 and
            pr.pr_code containing '.ipf.' and
            kml.kml_object_kind = :kontract_obj_kind and
            ag.ag_object_kind = 0 and
            ag.ag_object_id = :c_id
--            ag.ag_status = 1
--            ag.ag_id = :ag_id
      into ml_name, kml_cost, kml_total, cost_without_tax, cost_with_tax, unit_measure, qty, cost_without_tax_total,
           cost_with_tax_total, kml_sum_final, kml_count, feg_code
  do
  begin
    suspend;
  end
end
