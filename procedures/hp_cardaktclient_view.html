CREATE PROCEDURE HP_CARDAKTCLIENT_VIEW(
    ID D_INTEGER = null,
    DTBEG D_DATE = null,
    DTEND D_DATE = null,
    REGP_ID D_ID = null)
RETURNS (
    CAC_ID D_INTEGER,
    CAC_DATE D_DATE,
    CAC_FROM_RD_ID D_INTEGER,
    RD_CODE D_STR_255,
    RD_NAME D_STR_255,
    CAC_DATETIME D_DATETIME,
    CAC_STATUS D_INTEGER,
    STATUS_STR D_STR_CODE,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    CARD_ID D_INTEGER,
    REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    EDL_ID D_ID,
    CCA_ID D_INTEGER,
    IS_PINCOVER D_BOOL,
    C_PIN_ID D_ID,
    C_PIN_NAME_FIO D_STR_100,
    OL_ID D_ID,
    UN_ID D_ID,
    UN_FIO D_STR_100,
    VID_STATUS D_INTEGER)
AS
declare variable k_id d_id;
declare variable reg_lst d_str_8100;
declare variable vwp_type d_bool;
declare variable vwp_status d_integer;
declare variable acl_id d_id;
declare variable acl_status d_integer;
declare variable status_delayed d_integer;
begin
  dtbeg = coalesce(dtbeg, '30.12.1899');
  dtend = coalesce(dtend, current_date);
  regp_id = coalesce(regp_id, 1000000);
  status_delayed = f_enum_id('cardaktclient', 'issued_delayed');

  if (id is null) then
  begin
    reg_lst = '';
    select
      list(id_list, ';')
    from region_list(:regp_id,1,3,1)
    into
      :reg_lst;
  end

  for
    select
      cac_id,
      cac_date,
      cac_from_rd_id,
      rd_code,
      rd_name,
      cac_datetime,
      cac_status,
      f_enum_name('cardaktclient', cac_status),
      cac_from_k_id,
      cac_from_c_id,
      c_name_fio,
      cac_card_id,
      cac_from_reg_id,
      reg_name,
      cac_from_edl_id,
      cac_from_cca_id,
      cac_from_ol_id,
      cac_from_un_id,
      un_fio
    from cardaktclient
      inner join refdata
        on rd_id = cac_from_rd_id
      inner join clients
        on c_id = cac_from_c_id
      inner join region
        on reg_unicode = cac_from_reg_id
      left join edoc_list
        on edl_id = cac_from_edl_id
      inner join username
        on un_id = cac_from_un_id
    where cac_id = coalesce(:id, cac_id)
      and cac_date between :dtbeg and :dtend
  into
    :cac_id,
    :cac_date,
    :cac_from_rd_id,
    :rd_code,
    :rd_name,
    :cac_datetime,
    :cac_status,
    :status_str,
    :k_id,
    :c_id,
    :c_name_fio,
    :card_id,
    :reg_id,
    :reg_name,
    :edl_id,
    :cca_id,
    :ol_id,
    :un_id,
    :un_fio
  do
  begin
    if (not (reg_lst containing reg_id))
         then continue;

    select
      cem_is_pincover,
      c_id,
      c_name_fio
    from cardlist
      inner join cardembomain
        on cem_id = cal_from_cem_id
      left join cardplace
        on cpl_id = cal_last_pin_cpl_id
      left join clients
        on c_id = cpl_from_c_id
    where cal_card_id = :card_id
    into
      :is_pincover,
      :c_pin_id,
      :c_pin_name_fio;

    -- Видеоидентификация
    vwp_status = null;
    vid_status = -2;
    select
      vwp_type,
      vwp_status,
      acl_id,
      acl_status
    from verify_webcam_photo
      left join actionlist
        on acl_id = vwp_acl_id
    where vwp_card_id = :card_id
    order by vwp_id desc
    rows 1
    into
      :vwp_type,
      :vwp_status,
      :acl_id,
      :acl_status;
    if (vwp_type = 1) then
      vid_status =
        case
          when vwp_status = 0 and acl_status < 0 then -1
          when vwp_status = 0 and acl_status in (0, 1) then 0
          when vwp_status = 1 and acl_id is null then 1
          when vwp_status = 1 and acl_status = 2 then 2
        end;
    if (cac_status = status_delayed and edl_id > 0) then
      status_str = 'Ожидание подписи';

    suspend;
  end
end
