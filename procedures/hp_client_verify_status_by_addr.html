CREATE PROCEDURE HP_CLIENT_VERIFY_STATUS_BY_ADDR(
    ADL_ID D_ID)
AS
declare variable c_id d_id;
declare variable verify_status d_smallint;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  for select adl.adc_from_c_id, c.c_verify_status
      from addrclient adl
      inner join clients c on c.c_id = adl.adc_from_c_id
      where adl.adc_from_adl_id = :adl_id and
            adl.adc_address_type = 1 and
            coalesce(c_name_fio, '') <> '' and
            c_birth_date is not null and
            coalesce(c_pass_no, '') <> '' and
            coalesce(c_pass_date, '') <> '' and
            coalesce(c_blob_photo_id, 0) > 0 and
            coalesce(c_blob_pass1_id, 0) > 0 and
            (coalesce(c_blob_pass2_id, 0) > 0)
      into :c_id, :verify_status
  do
  begin
    if (db_id in (39801, 41701)) then
    begin
      if (exists(select k.k_id
                 from knclient knc
                 inner join kontract k on k.k_id = knc.kc_from_k_id
                 where knc.kc_from_c_id = :c_id and
                       knc.kc_amount_approved > 0 and
                       k.k_fact_disb_date is not null)) then
           exit;

        /* if (not exists(select *
                     from clientstate cs
                     where cs.cs_from_c_id = :c_id and
                           cs.cs_client_state < 1)) then
        exit;   */
      end

      if (verify_status = 2) then
      begin
        if (db_id in (39801, 41701)) then
        begin --Доп прооверки для КГ
          if (exists(select *
                     from hp_status_param_verif(:c_id)
                     where result = 0)) then
          begin
            update clients
            set c_verify_status = 1,
                c_verify_datetime = current_timestamp
            where (c_id = :c_id);
        end
          else
          begin
            update clients
            set c_verify_status = 0,
                c_verify_datetime = current_timestamp
            where (c_id = :c_id);
          end
        end
        else
        begin
          update clients
          set c_verify_status = 1,
              c_verify_datetime = current_timestamp
          where (c_id = :c_id);
        end
      end
    end
  end
