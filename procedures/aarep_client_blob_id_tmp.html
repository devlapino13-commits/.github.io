CREATE PROCEDURE AAREP_CLIENT_BLOB_ID_TMP
RETURNS (
    C_ID DECIMAL(18,0),
    C_REGION_ID INTEGER)
AS
declare variable hq_id integer;
declare variable db_id integer;
begin

  db_id = f_get_db_id();
  hq_id = f_hq_id(db_id);

  for select distinct c_id
      from clients, clientblob
      where cb_from_c_id = c_id and
            cb_blob is not null and
            (cb_from_bls_id = 0 or
            cb_from_bls_id is null)
      into :c_id
  do
  begin
    c_region_id = 0;
    select ks_region_id
    from kontstate, knclient
    where ks_from_k_id = kc_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          kc_from_c_id = :c_id and
          kc_amount_approved > 0
    order by ks_date_begin
    rows 1
    into :c_region_id;

    if ((c_region_id is null) or (c_region_id = 0)) then
    begin
      select ps_from_reg_id
      from hr_pstatus
      where ps_from_c_id = :c_id and
            ps_date_end >= '31.12.2999'
      order by ps_date_begin desc
      rows 1
      into :c_region_id;
    end
    if ((c_region_id is null) or (c_region_id = 0)) then
    begin
      c_region_id = :hq_id;
    end
    suspend;
  end
end
