CREATE PROCEDURE MC_NAME_BY_BARCODE(
    MCL_ID BIGINT,
    BARCODE D_INTEGER)
RETURNS (
    GOODS_NAME D_STR_LONG,
    UNIT_PRICE D_CURRENCY,
    UNIT_TYPE D_STR_LONG,
    DATETIME D_DATETIME)
AS
declare variable c_id id_type;
begin
  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into c_id;

  if (c_id is null or (barcode is null)) then
  begin
    exit;
  end
  else
    select cll.gll_name, cll.gll_unit_price, cll.gll_unit_type, cll.gll_datetime
    from gi_local_list cll
    where cll.gll_barcode = :barcode
    into :goods_name, :unit_price, :unit_type, :datetime;
    suspend;
end
