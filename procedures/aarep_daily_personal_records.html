CREATE PROCEDURE AAREP_DAILY_PERSONAL_RECORDS(
    ACTION_CODE D_INTEGER,
    IS_MAIN_OFFICE D_BOOL)
RETURNS (
    HO_ORDER_NO D_STR_20,
    HO_REP_DATE D_DATE,
    HO_ORDER_TYPE D_STR_100,
    C_INN D_STR_20,
    C_NAME_FIO D_STR_100,
    C_CITIZENSHIP D_STR_100,
    JOB_POSITION D_STR_100,
    JOB_CODE D_STR_100,
    JOB_CATEGORY D_STR_100,
    OFFICE_NAME D_STR_100,
    REG_NAME D_STR_100,
    PS_REP_DATE D_DATE,
    PREVIOUS_OFFICE_NAME D_STR_100,
    PREVIOUS_JOB_CODE D_STR_100,
    PREVIOUS_JOB_POSITION D_STR_100)
AS
declare variable C_ID D_ID;
declare variable JL_ID D_ID;
declare variable HO_ORDER_TYPE_ID D_ID;
declare variable OFFICE_ID D_INTEGER;
declare variable REGION_ID D_INTEGER;
begin
    -- RFCKG-3062

    /*
        важно знать

        ACTION_CODE in (1, 2, 3, 6)
        1 - новый сотрудник
        2 - Ротация (переводы) между офисами
        3 - Изменение условий договора
        6 - увольнение
        
        IS_MAIN_OFFICE in (0, 1)
        0 - Филиалы
        1 - Бишкек г. (ГО)

        есть три типа отчётов, которые можно пмолучить из данной процедуры
        найм сотрудников - когда вы передаёте ACTION_CODE = 1
        ротация мужду офисами и изменения условий договора - когда вы передаёте ACTION_CODE = 2 либо 3
        увольнения - когда вы передаёте ACTION_CODE = 6
    */
    if (:action_code not in (1, 2, 3, 6)) then
        exit;

    if (:is_main_office not in (0, 1)) then
        exit;

    :job_category = '';
    :previous_job_code = '';
    :previous_job_position = '';
    :previous_office_name = '';

    region_id = f_hq_id();

    for select
        ho.ho_order_no,
        ho.ho_date,
        ho.ho_order_type,
        sawt.sawt_name,
        ps.ps_from_c_id,
        ps.ps_from_jl_id,
        ps.ps_from_reg_id,
        iif(:action_code = 6, ps.ps_date_end_in_order - 1, ps.ps_date_begin)
    from hr_orders ho
    inner join hr_pstatus ps on ho.ho_id = ps.ps_from_ho_id
        and iif(:is_main_office = 1, ps.ps_from_reg_id = :region_id, ps.ps_from_reg_id <> :region_id)
        and ps.ps_status = 1
        and iif(:action_code = 6, ps.ps_date_end_in_order - 1 = current_date, ps.ps_date_begin = current_date)
    inner join sa_worktype sawt on ho.ho_order_type = sawt.sawt_id
    where ho.ho_status = 1
        and iif(:action_code in (2,3), ho.ho_order_type in (2,3), ho.ho_order_type = :action_code)
    into
        :ho_order_no,
        :ho_rep_date,
        :ho_order_type_id,
        :ho_order_type,
        :c_id,
        :jl_id,
        :office_id,
        :ps_rep_date
    do
    begin
        select
            c.c_inn,
            c.c_name_fio,
            rd.rd_str_id
        from clients c
        inner join refdata rd on c.c_citizen = rd.rd_id
        where c_id = :c_id
        into
            :c_inn,
            :c_name_fio,
            :c_citizenship;

        select
            reg1.reg_name,
            reg2.reg_name
        from region reg1
        inner join region reg2 on reg1.parent = reg2.reg_unicode
        where reg1.reg_unicode = :office_id
        into
            :office_name,
            :reg_name;

        select
            jl.jl_gni_code,
            jl.jl_name
        from joblist jl
        where jl_id = :jl_id
        into
            :job_code,
            :job_position;

        if (:action_code in (2,3)) then
        begin
            select first 1 skip 1
                jl.jl_gni_code,
                jl.jl_name,
                reg.reg_name
            from hr_pstatus ps
            inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
            inner join region reg on ps.ps_from_reg_id = reg.reg_unicode
            where ps.ps_from_c_id = :c_id
                and ps.ps_status = 1
                and ps.ps_is_working = 1
                and ps.ps_is_on_staff = 1
            order by ps.ps_from_ho_id desc
            into
                :previous_job_code,
                :previous_job_position,
                :previous_office_name;

            -- RFCKG-3090
            if (:action_code = 2 and :is_main_office = 1 and :office_id = 1090100 and :ho_order_type_id = 2) then
                -- выводим, если action_code = 2 и текущий офис - ГО, а предыдущий это филиал
                -- примечание: данное условие будет работать, если изменения происходят только с ГО
                suspend;

            if (:action_code = 3) then
                -- выводим все остальные случаи
                suspend;

        end
        else
        begin
            select rd1.rd_str_id  -- категория
            from client_gni_category cgc
            inner join refdata rd1 on rd1.rd_id = cgc.cgc_from_category
            where cgc.cgc_from_c_id = :c_id
                and cgc.cgc_is_delete = 0
                and current_date between cgc.cgc_date_begin and cgc.cgc_date_end
            into :job_category;

            suspend;
        end
    end
end
