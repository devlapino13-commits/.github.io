Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_DELETE_FROM_EXPRESSPKG(
    EXA_ID D_INTEGER,
    UN_ID D_ID)
AS
begin
  -- Данная процедура косвенно удаляет кандидата из пакета, сама запись в EXPRESSAPP не удаляется
  -- т.е. Кандидат становится свободным для внесения в другой пакет
  update expressapp exa set
    exa.exa_from_exp_id = -1,
    exa.exa_from_un_id = :un_id
  where exa.exa_id = :exa_id;
end
