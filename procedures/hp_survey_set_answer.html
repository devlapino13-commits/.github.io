CREATE PROCEDURE HP_SURVEY_SET_ANSWER(
    UN_ID DECIMAL(18,0),
    OBJ_KIND INTEGER,
    OBJ_ID DECIMAL(18,0),
    ANSWER_ID INTEGER)
AS
declare variable svq_id integer;
begin
  if (un_id is null) then
    exit;
  if (answer_id is null) then
    exit;

  if (obj_id is null) then
  begin
    obj_kind = 0;
    select un_from_c_id
    from username
    where un_id = :un_id
    into :obj_id;
  end

  select sva_from_svq_id
  from survey_answer
  where sva_id = :answer_id
  into :svq_id;

  if (svq_id is null) then
    exit;

  insert into survey_result (svr_source_kind, svr_source_id, svr_text, svr_server_sim, svr_from_sva_id, svr_from_svq_id)
  values (:obj_kind, :obj_id, '', '', :answer_id, :svq_id);
end
