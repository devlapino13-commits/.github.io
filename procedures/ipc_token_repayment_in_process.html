CREATE PROCEDURE IPC_TOKEN_REPAYMENT_IN_PROCESS
RETURNS (
    TRANS_ID D_STR_CODE,
    TOKEN_ID D_STR_CODE,
    K_ID D_ID,
    C_ID D_ID,
    AMOUNT D_CURRENCY,
    STATE D_INTEGER)
AS
begin
  for select etr.etr_transaction_id,
             eta.eta_token_id,
             etr.etr_from_k_id,
             et.et_from_c_id,
             etr.etr_amount,
             etr.etr_state
  from extsys_token_repayment etr
  inner join extsys_token et on et.et_id = etr.etr_from_et_id
  inner join extsys_token_add eta on eta.eta_from_et_id = et.et_id
  where etr.etr_is_finished = 0
  into :trans_id,
       :token_id,
       :k_id,
       :c_id,
       :amount,
       :state
  do
    suspend;
end
