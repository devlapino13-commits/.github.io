CREATE PROCEDURE HP_COMPET_KG_OLIMPIC_2022_MO(
    CPP_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable rd_id integer;
declare variable stage integer;
declare variable res_date_begin date;
declare variable res_date_end date;
declare variable rep_period integer;
declare variable res_period_id smallint;
declare variable mo_c_id d_id;
declare variable reg_id integer;
declare variable res_jobname varchar(100);
declare variable res_team_date date;
declare variable cpr_id integer;
declare variable cpd_points_1 decimal(18,2) = 0;
declare variable cpd_points_2 decimal(18,2) = 0;
declare variable cpd_points_3 decimal(18,2) = 0;
declare variable cpd_points_4 decimal(18,2) = 0;
declare variable cpd_points_5 decimal(18,2) = 0;
declare variable cpd_points_6 decimal(18,2) = 0;
declare variable cpd_points_7 decimal(18,2) = 0;
declare variable cpd_points_8 decimal(18,2) = 0;
declare variable cpd_points_9 decimal(18,2) = 0;
declare variable cpd_points_all decimal(18,2) = 0;
declare variable cpd_comment blob sub_type 1 segment size 160;
declare variable cpd_staff_kolich integer;
declare variable growth_without_delinq decimal(15,2);
declare variable reg_qty smallint;
declare variable growth_without_delinq_rank decimal(15,2);
declare variable super_pokazatel decimal(15,2);
declare variable sp_rank smallint;
declare variable avg_res_qual_koef decimal(15,2);
declare variable avg_res_qual_koef_rank smallint;
declare variable new_disb_rep_period decimal(15,2);
declare variable new_disb_rank smallint;
declare variable disb_1_15 decimal(15,2);
declare variable disb_1_15_rank smallint;
declare variable cpr_additional_points decimal(15,2);
declare variable old_points smallint;
declare variable penalty_points smallint;
declare variable disb_loan_avg decimal(15,2);
declare variable diff_client_avg decimal(15,2);
declare variable new_cpr_final_orderno smallint;
declare variable lo_qty_par30_2 decimal(15,2);
declare variable super_pokazatel_txt d_str_255;
declare variable mo_c_id_tmp d_id;
declare variable kol_kc d_integer;
declare variable chl_c_id d_id;
declare variable cpd_add_data_6 decimal(18,2) = 0;
declare variable cpd_add_data_7 decimal(18,2) = 0;
declare variable cpd_add_data_6_temp decimal(18,2) = 0;
declare variable cpd_add_data_7_temp decimal(18,2) = 0;
declare variable cpd_add_data_8 d_integer;
declare variable cpr1_id d_id;
declare variable cpd_id integer;
declare variable kolvo integer;
declare variable point decimal(15,2);
declare variable lo_id bigint;
declare variable koef_return decimal(15,2);
declare variable lo_par1_diff decimal(15,2);
declare variable lo_par1 decimal(15,2);
declare variable lo_par1_perc decimal(15,2);
declare variable lo_par1_perc_end decimal(15,2);
declare variable lo_par1_perc_diff decimal(15,2);
declare variable k_writeoff_sum decimal(15,2);
declare variable reg_par1_perc decimal(15,2);
declare variable reg_par1_diff decimal(15,2);
declare variable reg_par1_perc_diff decimal(15,2);
declare variable reg_par1_perc_end decimal(15,2);
declare variable reg_par1 decimal(15,2);
declare variable reg_koef_return decimal(15,2);
declare variable par1_end decimal(15,2);
declare variable par1_end_rank integer;
declare variable avg_koef_return_tmp decimal(15,2);
declare variable avg_koef_return decimal(15,2);
declare variable team_qty decimal(15,2);
declare variable kol_office integer;
declare variable avg_disb_1_15 decimal(15,2);
declare variable avg_disb_loan decimal(15,2);
declare variable sp_avg decimal(15,2);
declare variable avg_koef_return_rank integer;
declare variable prirost_classic decimal(15,2);
declare variable reg_balance decimal(15,2);
declare variable active_balance decimal(15,2);
declare variable lo_super_pokazatel decimal(15,2);
declare variable cpd_add_data_1 decimal(15,2);
declare variable cpd_add_data_2 decimal(15,2);
declare variable par30_begin decimal(15,2);
declare variable par30_end decimal(15,2);
begin
  /* Олимпийские игры МО */
  -->> Удаляем существующие расчеты
  delete from competresult
  where (cpr_from_cpp_id = :cpp_id);

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end;

  /* Stage:      0 - период (неделя)
                 1 - Промежуточные итоги
                 2 - Итоги
                 3 - Финальные результаты  */

  rep_period = 100 * extract(year from res_date_end) + extract(month from res_date_end);

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end
  mo_c_id_tmp = 0;

  -- Формируем в памяти данные по компании
  -- Тип отчета передаем "olimpic", чтобы система учитывала особенности расчета для олимпиады
  execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
      'standart')
      returning_values :res_period_id;

  if (res_period_id is null) then
  begin
    res_text = 'Не удалось расчитать данные за период';
    suspend;
    exit;
  end

  for --Список сотрудников, окно добавление в команд
      select distinct res_lo_id as mo_c_id, --
                      res_region_id as reg_id, --
                      res_jobname, --
                      res_team_date --
      from --
      aarep_staffteam_list(1000000, :res_date_end)
      where res_jobkind = 'Кредитный менеджер' and
            res_jobmemo <> 'Декретный отпуск'
      into mo_c_id, --
           reg_id, --
           res_jobname, --
           res_team_date --
  do
  begin
    if (mo_c_id = 1090100333782) then
      continue; -- Нуржан попросила исключить 09.08.2022
    cpd_points_1 = 0.0;
    cpd_points_2 = 0.0;
    cpd_points_3 = 0.0;
    cpd_points_4 = 0.0;
    cpd_points_5 = 0.0;
    cpd_points_6 = 0.0;
    cpd_points_7 = 0.0;
    cpd_points_8 = 0.0;
    cpd_points_9 = 0.0;
    cpd_comment = 'Дата вступления: <b>' || f_datetostr(res_team_date) || '</b><br>';

    if (mo_c_id <> mo_c_id_tmp) then
    begin
      mo_c_id_tmp = mo_c_id;
      insert into competresult (cpr_id, cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id,
                                cpr_calced_points, cpr_additional_points, cpr_order_in_group, cpr_final_points,
                                cpr_final_orderno, cpr_result_kind, cpr_comment, cpr_from_un_id, cpr_datetime)
      values (null, :cpp_id, :res_jobname, 0, :mo_c_id, 0, 0, 0, 0, 0, -1, '', :un_id, current_timestamp)
      returning cpr_id
      into :cpr_id;
    end

    cpd_staff_kolich = 0;
    cpd_points_all = -->>
    cpd_points_1 + cpd_points_2 + cpd_points_3 + cpd_points_4 + cpd_points_5 + cpd_points_6 + cpd_points_7 + cpd_points_8 + cpd_points_9;

    insert into competdetails (cpd_from_cpr_id, cpd_id, cpd_object_kind, cpd_object_id, --
                               cpd_points_all, cpd_points_1, cpd_points_2, cpd_points_3, cpd_points_4, --
                               cpd_points_5, cpd_points_6, cpd_points_7, cpd_points_8, cpd_points_9, --
                               cpd_comment, --
                               cpd_add_data_1, cpd_add_data_2, cpd_add_data_3, cpd_add_data_4, cpd_add_data_5,
                               cpd_add_data_6, cpd_add_data_7, cpd_add_data_8, cpd_add_data_9, cpd_staff_kolich)
    values (:cpr_id, null, 7, :reg_id, --
            :cpd_points_all, :cpd_points_1, :cpd_points_2, :cpd_points_3, :cpd_points_4, --
            :cpd_points_5, :cpd_points_6, :cpd_points_7, :cpd_points_8, :cpd_points_9, --
            :cpd_comment, --
            0, 0, 0, 0, 0, --
            0, 0, 0, 0, :cpd_staff_kolich); -- cpd_staff_kolich 1 если не отпуск, иначе 0

  end

  for select chl.chl_c_id, chl.chl_reg_id
      from compet_hr_list chl
      where :res_date_end between chl.chl_date_begin and chl.chl_date_end
      into chl_c_id, reg_id
  do
  begin
    if (not exists(select *
                   from competresult cpr
                   inner join competdetails cpd on cpd.cpd_from_cpr_id = cpr.cpr_id and
                         cpd.cpd_object_kind = f_objkind('region', 'id') and
                         cpd.cpd_object_id = :reg_id
                   where cpr.cpr_from_cpp_id = :cpp_id)) then
    begin
      insert into competresult (cpr_id, cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id,
                                cpr_calced_points, cpr_additional_points, cpr_order_in_group, cpr_final_points,
                                cpr_final_orderno, cpr_result_kind, cpr_comment, cpr_from_un_id, cpr_datetime)
      values (null, :cpp_id, 'Менеджер офиса', 0, :chl_c_id, 0, 0, 0, 0, 0, -1, '', :un_id,
              current_timestamp)
      returning cpr_id
      into :cpr_id;

      insert into competdetails (cpd_from_cpr_id, cpd_id, cpd_object_kind, cpd_object_id, --
                                 cpd_points_all, cpd_points_1, cpd_points_2, cpd_points_3, cpd_points_4, --
                                 cpd_points_5, cpd_points_6, cpd_points_7, cpd_points_8, cpd_points_9, --
                                 cpd_comment, --
                                 cpd_add_data_1, cpd_add_data_2, cpd_add_data_3, cpd_add_data_4, cpd_add_data_5,
                                 cpd_add_data_6, cpd_add_data_7, cpd_add_data_8, cpd_add_data_9, cpd_staff_kolich)
      values (:cpr_id, null, 7, :reg_id, --
              :cpd_points_all, :cpd_points_1, :cpd_points_2, :cpd_points_3, :cpd_points_4, --
              :cpd_points_5, :cpd_points_6, :cpd_points_7, :cpd_points_8, :cpd_points_9, --
              :cpd_comment, --
              0, 0, 0, 0, 0, --
              0, 0, 0, 0, :cpd_staff_kolich);
    end
  end

  reg_qty = 0;
  select count(cpr_id)
  from competresult cr
  inner join competdetails cd on cd.cpd_from_cpr_id = cr.cpr_id and
        cd.cpd_object_id > 0
  where cpr_from_cpp_id = :cpp_id
  into reg_qty;
  cpr_additional_points = 0;
  if (stage = 0) then
  begin
    for select cd.cpd_object_id as reg_id, cr.cpr_id, cd.cpd_id
        from competresult cr
        inner join competdetails cd on cr.cpr_id = cd.cpd_from_cpr_id and
              cd.cpd_staff_kolich = 0
        where cr.cpr_from_cpp_id = :cpp_id
        into reg_id, cpr_id, cpd_id
    do
    begin
      select sum(kc.kc_amount_approved)
      from kontstate ks
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id and
            kc.kc_circle = 1 and
            kc.kc_amount_approved > 0
      where ks.ks_state = 1 and
            ks.ks_date_begin between :res_date_begin and :res_date_end and
            ks.ks_region_id = :reg_id
      into new_disb_rep_period;

      select sum(kc.kc_amount_approved)
      from kontstate ks
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id and
            kc.kc_amount_approved > 0
      where ks.ks_state = 1 and
            ks.ks_date_begin between :res_date_begin and :res_date_end and
            ks.ks_region_id = :reg_id
      into disb_1_15;

      select count(*)
      from aarep_personal_list(:reg_id, :res_date_end, 1)
      where (((cast(:res_date_end as date) - ps_date_begin) >= 5) or
            (ho_text like '%Перевести%')) and
            jl_name in ('Специалист по финансированию', 'Старший специалист по финансированию') and
            ps_curr_status not in ('Отпуск без содерж.по уходу за ребенком', 'Отпуск по беременности и родам', 'Трудовой отпуск декретницы')
      into kol_kc;

      update competdetails
      set cpd_add_data_6 = :disb_1_15,
          cpd_add_data_7 = :new_disb_rep_period,
          cpd_add_data_8 = :kol_kc
      where cpd_from_cpr_id = :cpr_id and
            cpd_id = :cpd_id and
            cpd_object_id = :reg_id;
    end

    for select cpr_id, cd.cpd_add_data_6, cd.cpd_add_data_7, cpd_add_data_8
        from competdetails cd
        inner join competresult cr on cd.cpd_from_cpr_id = cr.cpr_id and
              cr.cpr_from_cpp_id = :cpp_id
        where cd.cpd_object_kind = 7
        into cpr_id, cpd_add_data_6, cpd_add_data_7, cpd_add_data_8
    do
    begin
      if (:cpd_add_data_8 = 0) then
        cpd_add_data_8 = 1;
      if (cpd_add_data_6 <> 0) then
        cpd_add_data_6_temp = :cpd_add_data_6 / :cpd_add_data_8;
      if (cpd_add_data_7 <> 0) then
        cpd_add_data_7_temp = :cpd_add_data_7 / :cpd_add_data_8;

      update competdetails
      set cpd_add_data_6 = :cpd_add_data_6_temp,
          cpd_add_data_7 = :cpd_add_data_7_temp
      where cpd_from_cpr_id = :cpr_id;
    end
    kol_office = (select count(cpd.cpd_from_cpr_id)
                  from competresult cpr
                  inner join competdetails cpd on cpd.cpd_from_cpr_id = cpr.cpr_id
                  where cpr.cpr_from_cpp_id = :cpp_id);
    cpr1_id = 0;

    for select distinct cr.cpr_id, cd.cpd_add_data_6, rank() over(order by cd.cpd_add_data_6 desc)
        from competresult cr
        inner join competdetails cd on cr.cpr_id = cd.cpd_from_cpr_id
        where cr.cpr_from_cpp_id = :cpp_id
        into cpr_id, avg_disb_1_15, disb_1_15_rank
    do
    begin
      cpd_comment = '';
      if (cpr_id <> cpr1_id) then
      begin
        cpr1_id = :cpr_id;
        cpd_comment = 'Cр. Сумма выдач с 1-15 число: <b>' || avg_disb_1_15 || '</b><br>';
        cpr_additional_points = 0;
        if (avg_disb_1_15 > 0 and
            disb_1_15_rank <= 15) then
        begin
          cpr_additional_points = 5;
          cpd_comment = cpd_comment || '    Доп.балл за выдачи 1-15: <b>' || cpr_additional_points || ';</b><br>    Ранг: ' || disb_1_15_rank || ';<br>';
        end
        if (avg_disb_1_15 > 0 and
            :kol_office - disb_1_15_rank < 15) then
        begin
          cpr_additional_points = -5;
          cpd_comment = cpd_comment || '    Доп.балл за выдачи 1-15: <b>' || cpr_additional_points || ';</b><br>    Ранг: ' || disb_1_15_rank || ';<br>';
        end
        update competdetails
        set cpd_comment = cpd_comment || :cpd_comment
        where cpd_from_cpr_id = :cpr_id;

        update competresult
        set cpr_additional_points = cpr_additional_points + :cpr_additional_points
        where cpr_id = :cpr_id;
      end
    end
    for select distinct cpr_id, cd.cpd_add_data_7, rank() over(order by cpd_add_data_7 desc) new_disb_rank
        from competdetails cd
        inner join competresult cr on cd.cpd_from_cpr_id = cr.cpr_id and
              cr.cpr_from_cpp_id = :cpp_id
        order by cpd_add_data_7 desc
        into cpr_id, cpd_add_data_7, new_disb_rank
    do
    begin
      cpd_comment = '';
      cpr1_id = 0;
      if (cpr_id <> cpr1_id) then
      begin
        cpr1_id = :cpr_id;
        cpd_comment = cpd_comment || 'Сумма выдач новым клиентам: <b>' || cpd_add_data_7 || '</b><br>';
        cpr_additional_points = 0;
        if (:cpd_add_data_7 > 0 and
            new_disb_rank <= 10) then
        begin
          cpr_additional_points = 11 - new_disb_rank;
          cpd_comment = cpd_comment || '    Доп.балл за выдачи новым клиентам: <b>' || cpr_additional_points || ';</b><br>    Ранг: ' || new_disb_rank || ';<br>';
          update competdetails
          set cpd_comment = cpd_comment || :cpd_comment
          where cpd_from_cpr_id = :cpr_id;

          update competresult
          set cpr_additional_points = cpr_additional_points + :cpr_additional_points
          where cpr_id = :cpr_id;
        end
      end
    end
  end
  if (stage = 1) then
  begin
    for select distinct cd.cpd_object_id as reg_id, cr.cpr_id, cd.cpd_id
        from competresult cr
        inner join competdetails cd on cr.cpr_id = cd.cpd_from_cpr_id and
              cd.cpd_staff_kolich = 0
        where cr.cpr_from_cpp_id = :cpp_id
        into :reg_id, cpr_id, cpd_id -- ID офиса
    do
    begin
      lo_qty_par30_2 = 0.0;
      lo_par1_perc_diff = 0;
      lo_par1_diff = 0;
      reg_par1_perc = 0;
      reg_par1_diff = 0;
      reg_par1_perc_diff = 0;
      reg_par1_perc_end = 0;
      reg_par1 = 0;
      reg_koef_return = 0;
      kol_kc = 0;
      reg_balance = 0;
      active_balance = 0;
      super_pokazatel = 0;
      lo_super_pokazatel = 0;
      cpd_comment = '';
      for select res_field_2, ad_return_os_koef as koef_return, ad_delinq_balance_diff1 as lo_par1_diff,
                 ad_delinq_par_end1 as lo_par1,
                 iif((ad_delinq_par_end1 / ad_active_balance) > 0.02, -2, 0) as lo_par1_perc,
                 (ad_delinq_par_end1 / ad_active_balance) as lo_par1_perc_end,
                 (iif(ad_active_balance = 0, 0,
                 (ad_delinq_balance_end1 / ad_active_balance)) - iif(ad_begin_active_balance = 0, 0,
                 (ad_delinq_balance_begin1 / ad_begin_active_balance))) * 100 lo_par1_perc_diff,
                 ad_active_balance as active_balance, ad_delinq_par_begin3 as par30_begin, ad_delinq_par_end3 as par30_end
          from hp_analitica_report('reg_id, lo_id, own_portfolio', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart')
          where :reg_id = res_field_1 and
                res_field_1 <> 0 and
                exists(select *
                       from aarep_personal_list(:reg_id, :res_date_end, 1)
                       where res_field_2 = ps_from_c_id and
                             (((cast(:res_date_end as date) - ps_date_begin) >= 15) or
                             (ho_text like '%Перевести%')) and
                             jl_name in ('Специалист по финансированию', 'Старший специалист по финансированию') and
                             ps_curr_status not in ('Отпуск без содерж.по уходу за ребенком', 'Отпуск по беременности и родам', 'Трудовой отпуск декретницы')) and
                res_field_3 = 'Собственный' and
                ad_active_balance > 0
          into lo_id, koef_return, lo_par1_diff, lo_par1, lo_par1_perc, lo_par1_perc_end, lo_par1_perc_diff,
               active_balance, par30_begin, par30_end
      do
      begin
        kol_kc = kol_kc + 1;
        if (koef_return = 0) then
          koef_return = 100;
        select -sum(ok_os)
        from operkred ok
        inner join kontract k on k.k_id = ok.ok_from_k_id and
              k.k_fact_disb_lo_id = :lo_id
        inner join operlist ol on ol.ol_id = ok.ok_from_ol_id and
              ol.is_delete = 0 and
              ol.ol_from_rd_id <> 921
        where ok_ol_is_active = 1 and
              ok_ol_date between :res_date_begin and :res_date_end and
              ok_type = 5 --списания
        into k_writeoff_sum;

        /*select super_pokazatel
        from hp_compet_kg_2021_superp(:lo_id, null, :rep_period, :res_date_begin, :res_date_end)
        into :lo_super_pokazatel;*/
        :lo_super_pokazatel = par30_end - par30_begin;
        super_pokazatel = super_pokazatel + lo_super_pokazatel;
        reg_balance = reg_balance + active_balance;
        k_writeoff_sum = coalesce(k_writeoff_sum, 0);
        lo_par1_diff = lo_par1_diff + k_writeoff_sum;
        reg_par1_perc = reg_par1_perc + lo_par1_perc;
        reg_par1_diff = reg_par1_diff + lo_par1_diff;
        reg_par1_perc_diff = reg_par1_perc_diff + lo_par1_perc_diff;
        reg_par1_perc_end = reg_par1_perc_end + lo_par1_perc_end;
        reg_par1 = reg_par1 + lo_par1;
        reg_koef_return = reg_koef_return + koef_return;
      end

      if (:kol_kc = 0) then
        :kol_kc = 1;

          inner join competdetails cd on cr.cpr_id = cd.cpd_from_cpr_id and
                cd.cpd_object_id > 0
          inner join hp_analitica_report('reg_id, lo_id', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart') on res_field_1 = cd.cpd_object_id and
                res_field_2 > 1000000000010 and
                ad_active_balance > 0.0
          where cr.cpr_from_cpp_id = :cpp_id and
                cpr_object_id > 0
          group by 1, 2)

      select cpr_id, disb_loan_avg, diff_client_avg, --
             rank() over(order by cpr_final_orderno, disb_loan_avg desc, diff_client_avg desc) new_cpr_final_orderno
      from t_cpr
      into cpr_id, disb_loan_avg, diff_client_avg, new_cpr_final_orderno
  do
  begin
    cpd_comment = 'Ср.выдачи: ' || round(:disb_loan_avg) || '; Ср.прирост актив.кл.: ' || round(:diff_client_avg) || '; ';

    update competresult
    set cpr_final_orderno = :new_cpr_final_orderno,
        cpr_comment = cpr_comment || :cpd_comment
    where cpr_from_cpp_id = :cpp_id and
          cpr_id = :cpr_id;
  end

  if (stage = 1) then
    update competresult
    set cpr_additional_points = cpr_additional_points + 1,
        cpr_comment = cpr_comment || 'Доп. бал за 1 место: 1'
    where cpr_from_cpp_id = :cpp_id and
          cpr_final_orderno = 1;
  else
  if (stage = 2) then
  begin
    for select cpr.cpr_id
        from competdetails cpd
        inner join competresult cpr on cpr.cpr_id = cpd.cpd_from_cpr_id and
              cpr.cpr_from_cpp_id = :cpp_id
        where cpd.cpd_id > 1
        into cpr_id
    do
    begin

      update competdetails
      set cpd_points_all = (cpd_points_all / 2)
      where cpd_from_cpr_id = :cpr_id;

      update competresult cr
      set cpr_additional_points = (cpr_additional_points / 2),
          cr.cpr_calced_points = (cpr_calced_points / 2)
      where cpr_id = :cpr_id;
    end

    execute procedure hp_competition_recalc_order(:cpp_id);

    update competresult
    set cpr_additional_points = cpr_additional_points + 10,
        cpr_comment = cpr_comment || 'Доп. бал за 1-место: 10'
    where cpr_from_cpp_id = :cpp_id and
          cpr_final_orderno = 1;

    update competresult
    set cpr_additional_points = cpr_additional_points + 6,
        cpr_comment = cpr_comment || 'Доп. бал за 2-место: 6'
    where cpr_from_cpp_id = :cpp_id and
          cpr_final_orderno = 2;

    update competresult
    set cpr_additional_points = cpr_additional_points + 3,
        cpr_comment = cpr_comment || 'Доп. бал за 3-место: 3'
    where cpr_from_cpp_id = :cpp_id and
          cpr_final_orderno = 3;
  end

  suspend;
end
