CREATE PROCEDURE FM_HTML_CLIENT_DEATH_INFO(
    C_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable GBD_BLS_ID D_ID;
declare variable GBD_JSON_TEXT D_BLOB_TEXT;
declare variable L_STATUS_CODE STR_10;
declare variable D_CERT_NUM STR_100;
declare variable D_DATE_STR STR_20;
declare variable D_ISSUE_ORG STR_255;
declare variable STR_TMP varchar(1000);
begin
  res_text = '';

  select first 1 eil.eil_from_bls_id
  from extinfolist eil
  where eil.eil_from_c_id = :c_id
    and eil.eil_from_eic_id = 3
    and eil.eil_status = 2
  order by eil.eil_create_datetime desc
  into :gbd_bls_id;

  if (gbd_bls_id is not null) then
  begin
    select cast(blx_blob as d_blob_text)
    from hp_get_blobex_blob(:gbd_bls_id)
    into :gbd_json_text;

    if (gbd_json_text is not null) then
    begin
      l_status_code = f_json_blob_get_value(gbd_json_text, 'life_status.code', '');

      if (l_status_code = '2') then
      begin
        d_cert_num  = f_json_blob_get_value(gbd_json_text, 'deathcertificate.number', '');
        d_date_str  = f_json_blob_get_value(gbd_json_text, 'deathcertificate.begindate', '');
        d_issue_org = f_json_blob_get_value(gbd_json_text, 'deathcertificate.issueorganisation', '');

        str_tmp = '<br><br>&nbsp;&nbsp;Жизненный статус: <span style="color: red; font-weight: bold;">Умерший</span>';

        if (d_cert_num <> '') then
          str_tmp = str_tmp || '<br>&nbsp;&nbsp;номер свидетельства о смерти: <b>' || d_cert_num || '</b>';

        if (d_date_str <> '') then
          str_tmp = str_tmp || '<br>&nbsp;&nbsp;дата смерти: <b>' || d_date_str || '</b>';

        if (d_issue_org <> '') then
          str_tmp = str_tmp || '<br>&nbsp;&nbsp;орган выдавший свидетельство: <b>' || d_issue_org || '</b>';

        res_text = str_tmp || '<br>';
      end
    end
  end

  suspend;
end
