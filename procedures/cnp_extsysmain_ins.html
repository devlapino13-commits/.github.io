CREATE PROCEDURE CNP_EXTSYSMAIN_INS(
    CPR_ID D_ID,
    PAY_ACTION D_STR_CODE,
    TRANS_ID D_STR_20,
    ERROR_TXT D_STR_8100,
    PAY_STATUS D_SMALLINT,
    REQUEST_BLOB BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    ANSWER_BLOB BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
RETURNS (
    ESM_ID ID_TYPE)
AS
declare variable REP_DATE D_DATE;
declare variable PROJECT D_STR_CODE;
declare variable C_ID D_ID;
declare variable K_ID D_ID;
declare variable ESM_OLD_STATUS D_SMALLINT;
declare variable ACCOUNT D_STR_100;
declare variable AMOUNT D_CURRENCY;
declare variable UN_ID D_ID;
begin
  error_txt = left(coalesce(error_txt, ''), 255);
  rep_date = current_date;
  project = 'cnp';

  for select cpr.cpr_from_c_id, cpr.cpr_from_k_id, cpr.cpr_amount, cpr.cpr_from_un_id
      from cnp_payment_requests cpr
      where cpr.cpr_id = :cpr_id
      into c_id, k_id, amount, un_id
  do
  begin
    select et.et_token
    from extsys_token et
    where et.et_from_c_id = :c_id and
          et.et_is_actual = 1
    into :account;
    /*if (account is null) then
      account = '151408-99371' ;  тест */
    select em.esm_id, em.esm_status
    from extsysmain em
    where em.esm_trans_id = :trans_id and
          em.esm_project = :project and
          em.esm_action = :pay_action and
          em.esm_from_k_id = :k_id and
          em.esm_object_id = :c_id rows 1
    into :esm_id, esm_old_status;

    if (esm_id is null) then
    begin
      insert into extsysmain (esm_trans_id, esm_project, esm_action, esm_date, esm_account, esm_object_kind,
                              esm_object_id, esm_datetime, esm_result_id, esm_status, esm_full_source, esm_amount,
                              esm_from_k_id)
      values (:trans_id, :project, :pay_action, :rep_date, :account, 0, :c_id, current_timestamp, :pay_status,
              :pay_status, :request_blob, :amount, :k_id)
      returning esm_id
      into :esm_id;

      insert into extsysusers (esu_from_esm_id, esu_user_id)
      values (:esm_id, :un_id);

      insert into extsysanswer (esa_from_esm_id, esa_message, esa_result_id, esa_status)
      values (:esm_id, :answer_blob, :pay_status, :pay_status);

    end
    update cnp_payment_requests cpr
    set cpr.cpr_from_esm_id = :esm_id,
        cpr.cpr_status = :pay_status,
        cpr.cpr_error_text = :error_txt
    where cpr.cpr_id = :cpr_id;
    suspend;
  end
end
