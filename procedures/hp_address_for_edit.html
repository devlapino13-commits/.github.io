CREATE PROCEDURE HP_ADDRESS_FOR_EDIT(
    IN_ADL_ID D_ID,
    UN_ID D_ID)
RETURNS (
    ADL_ID D_ID,
    ADL_FROM_PL_ID D_ID,
    ADL_DEPART_NO D_STR_20,
    ADL_DEPART_PREF D_STR_20,
    ADL_FLAT_NO D_STR_20,
    ADL_BUILDING_NO D_STR_20,
    ADL_SECTION_NO D_STR_20,
    ADL_VLADEN_NO D_STR_20,
    ADL_ADDRESS_STR D_STR_255,
    ADL_COMMENT D_STR_255,
    ADL_POST_INDEX D_INTEGER,
    ADL_IS_WITHOUT_STREET D_SMALLINT,
    ADL_LANG_KIND D_SMALLINT,
    ADL_STREET_ID D_ID,
    ADL_IS_EDITED D_SMALLINT,
    COUNTRY_PL_ID D_ID,
    OBLAST_PL_ID D_ID,
    FACT_ADDRESS D_STR_255,
    PL_ADD_CODE D_STR_20)
AS
declare variable CNT_PARENT D_ID;
declare variable TMP_ID D_ID;
begin
  adl_is_without_street = 0;
  adl_lang_kind = 0;
  adl_is_edited = 0;

  if (coalesce(in_adl_id, 0) = 0) then
  begin
    adl_id = null;
    suspend;
    exit;
  end

  select adl_id, adl_from_pl_id, adl_depart_no, adl_depart_pref, adl_flat_no, adl_building_no, adl_section_no, adl_address_str,
         adl_comment, adl_post_index, adl_street_id, adl_is_without_street, adl_is_edited, adl_lang_kind, adl_vladen_no
  from addrlist
  where adl_id = :in_adl_id
  into adl_id, :adl_from_pl_id, :adl_depart_no, :adl_depart_pref, :adl_flat_no, :adl_building_no, :adl_section_no,
       adl_address_str, :adl_comment, :adl_post_index, :adl_street_id, :adl_is_without_street, :adl_is_edited, :adl_lang_kind, adl_vladen_no;

  fact_address = adl_address_str;
  oblast_pl_id = null;
  country_pl_id = null;

  if (adl_vladen_no is null) then
    adl_vladen_no = '';

  if (adl_from_pl_id is not null) then  -- RFCRU-4498
  begin
    select pl.pl_add_code
    from places pl
    where pl.pl_id = :adl_from_pl_id
    into pl_add_code;

    pl_add_code = coalesce(pl_add_code, '');
  end

  if (adl_from_pl_id is not null) then
  begin
    cnt_parent = adl_from_pl_id;
    oblast_pl_id = adl_from_pl_id;

    while (cnt_parent is not null) do
    begin
      select obl.parent, cnt.pl_id, cnt.parent
      from places obl, places cnt
      where obl.pl_id = :oblast_pl_id and
            obl.parent = cnt.pl_id
      into tmp_id, country_pl_id, cnt_parent;

      if (cnt_parent is not null) then
        oblast_pl_id = tmp_id;
    end
  end

  if (adl_from_pl_id is not null and adl_street_id is distinct from adl_from_pl_id) then
    adl_is_without_street = 0;
  else
  begin
    adl_street_id = null;
    adl_is_without_street = 1;
  end

  if (adl_lang_kind is null) then
    adl_lang_kind = 0;

  suspend;
end
