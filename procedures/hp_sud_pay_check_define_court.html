CREATE PROCEDURE HP_SUD_PAY_CHECK_DEFINE_COURT
AS
declare variable proc_type d_str_code;
declare variable bls_id d_id;
declare variable json_str d_str_8100;
declare variable target d_str_code;
declare variable sag_kind d_integer;
declare variable sag_code d_str_code;
declare variable sag_name d_str_255;
declare variable sag_address d_str_255;
declare variable sag_email d_str_255;
declare variable sag_phone d_str_255;
declare variable sag_bankname d_str_255;
declare variable sag_bik d_str_10;
declare variable sag_inn d_str_20;
declare variable sag_kpp d_str_10;
declare variable sag_acc d_str_20;
declare variable sag_treasure_acc d_str_20;
declare variable sag_payee_name d_str_255;
declare variable sag_kbk d_str_20;
declare variable sag_oktmo d_str_20;
declare variable sag_id d_id;
declare variable slc_id d_id;
declare variable kind d_smallint;
begin
  -- процедура ищет ответы на запросы определения подсудности

  -- найдем заявки ожидающие подсудности
  for select slc_id, f_enum('sud_proc_type', smd_proc_type, 'code')
  from sud_maindocs, sud_loanclient
  where smd_status < 2
    and slc_from_smr_id = smd_id
    and ((smd_proc_type = 0 and slc_auto_worldcourt is null)
      or (smd_proc_type = 1 and slc_auto_federalcourt is null))
  into slc_id, proc_type
  do
  begin
    bls_id = null;

    kind =  case proc_type
              when 'court_order' then 1
              when 'petition' then 2
              else -1
            end;

    if (kind = -1) then
      exception e_default 'Unknown proc_type = ' || coalesce(proc_type, '<NULL>');

    select bls_id
    from hp_sud_get_request_status(:slc_id, :kind)
    where eil_status = 2
    into bls_id;

    if (bls_id is null) then
      continue;

    json_str = f_json_blob_get_value('{"items":'||(select blx_blob from hp_get_blobex_blob(:bls_id))||'}', 'items[0]', '{}');

    target = f_json_str_get_value(json_str, 'TARGET', '');
    sag_kind = f_enum('sud_agent_kind', iif(target = 'COURT', 'world_court', 'federal_court'), 'id');
    sag_code = f_json_str_get_value(json_str, 'COURT_CODE', '');
    sag_name = f_json_str_get_value(json_str, 'COURT_NAME', '');
    sag_address = f_json_str_get_value(json_str, 'COURT_POSTCODE', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'COURT_REGION_NAME', '') || ' ' || f_json_str_get_value(json_str, 'COURT_REGION_TYPE', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'COURT_DISTRICT_NAME', '') || ' ' || f_json_str_get_value(json_str, 'COURT_DISTRICT_TYPE', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'COURT_CITY_TYPE', '-') || '. ' || f_json_str_get_value(json_str, 'COURT_CITY_NAME', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'COURT_LOCALITY_TYPE', '-') || '. ' || f_json_str_get_value(json_str, 'COURT_LOCALITY_NAME', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'COURT_STREET_TYPE', '-') || '. ' || f_json_str_get_value(json_str, 'COURT_STREET_NAME', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'COURT_HOUSE', '') || ' ' || f_json_str_get_value(json_str, 'COURT_BLOCK', '');

    sag_address = replace(sag_address,'-. ', '');
    sag_address = replace(sag_address,'-, ', '');

    sag_email = f_json_str_get_value(json_str, 'COURT_EMAIL', '');
    sag_phone = f_json_str_get_value(json_str, 'COURT_PHONE', '');
    sag_bankname = f_json_str_get_value(json_str, 'BANK_NAME', '');
    sag_bik = f_json_str_get_value(json_str, 'BIK', '');
    sag_inn = f_json_str_get_value(json_str, 'INN', '');
    sag_kpp = f_json_str_get_value(json_str, 'KPP', '');
    sag_acc = f_json_str_get_value(json_str, 'EKS', '');
    sag_treasure_acc = f_json_str_get_value(json_str, 'NKS', '');
    sag_payee_name = f_json_str_get_value(json_str, 'PAYEE_NAME', '');
    sag_kbk = f_json_str_get_value(json_str, 'KBK', '');
    sag_oktmo = f_json_str_get_value(json_str, 'OKTMO', '');

    update or insert into sud_agents (sag_kind, sag_code, sag_name, sag_address, sag_email,
                                      sag_phone, sag_bankname, sag_bik, sag_inn, sag_kpp, sag_acc, sag_treasure_acc,
                                      sag_payee_name, sag_kbk, sag_oktmo)
    values (:sag_kind, :sag_code, :sag_name, :sag_address, :sag_email, :sag_phone, :sag_bankname,
            :sag_bik, :sag_inn, :sag_kpp, :sag_acc, :sag_treasure_acc, :sag_payee_name, :sag_kbk, :sag_oktmo)
    matching (sag_kind, sag_code)
    returning sag_id
    into sag_id;

    if (target = 'COURT') then
      update sud_loanclient
      set slc_auto_worldcourt = :sag_id
      where slc_id = :slc_id;
    else
      update sud_loanclient
      set slc_auto_federalcourt = :sag_id
      where slc_id = :slc_id;

  end 
end
