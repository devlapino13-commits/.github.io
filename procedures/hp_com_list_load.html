CREATE PROCEDURE HP_COM_LIST_LOAD(
    CML_ID D_ID,
    CMC_ID D_INTEGER)
RETURNS (
    CMCP_PARAM_NAME D_STR_CODE,
    CMCP_CAPTION D_STR_255,
    CMCP_DESCRIPTION D_STR_255,
    CMCP_EDIT_KIND D_STR_CODE,
    CMCP_MAX_LENGTH D_INTEGER,
    CMCP_DEFAULT_VALUE D_STR_CODE,
    CMCP_DEFAULT_TEXT D_STR_CODE,
    CMCP_READ_ONLY D_SMALLINT,
    CMCP_IS_HIDDEN D_SMALLINT,
    CMCP_COMBO_LIST D_STR_8100,
    PARAM_VALUE D_STR_CODE)
AS
declare variable cml_add_params d_str_long;
declare variable cml_date d_date;
declare variable is_read_only d_smallint;
declare variable cml_status d_integer;
declare variable cml_status_str d_str_code;
begin
  is_read_only = 0;

  cml_add_params = '';
  if (coalesce(cml_id, 0) <> 0) then
  begin
    select cml_from_cmc_id, cml_date, cml_add_params, cml_last_status
    from com_list
    where cml_id = :cml_id
    into cmc_id, cml_date, cml_add_params, cml_status;

    cml_status_str = f_enum('com_action_status', cml_status, 'code');
    if (cml_status_str in ('signing', 'done')) then
      is_read_only = 1;
  end else
    cml_date = current_date;

  if (coalesce(cmc_id, 0) = 0) then
    exception e_default 'CMC_ID is null';

  cmcp_param_name = 'com_list_kind';
  cmcp_caption = 'Вид реструктуризации';
  cmcp_description = 'Вид реструктуризации';
  cmcp_edit_kind = 'combo';
  cmcp_max_length = 0;
  cmcp_default_value = cmc_id;
  cmcp_default_text = '';
  cmcp_read_only = 1;
  cmcp_is_hidden = 0;
  cmcp_combo_list = '';
  param_value = cmcp_default_value;

  select list(f_format('{"id":%1, "name":"%2"}', cmc_id, cmc_name))
  from com_cfg
  where (cmc_id = :cmc_id or
        (cmc_state <> 0 and cmc_object_kind = f_objkind('kontract', 'id')))
  into cmcp_combo_list;
  if (cmcp_combo_list <> '') then
    cmcp_combo_list = '{"list":[' || cmcp_combo_list || ']}';

  cmcp_read_only = maxvalue(cmcp_read_only, is_read_only);
  suspend;

  cmcp_param_name = 'com_list_date';
  cmcp_caption = 'Дата реструктуризации';
  cmcp_description = 'Дата проведения реструктуризации';
  cmcp_edit_kind = 'date';
  cmcp_max_length = 0;
  cmcp_default_value = f_datetostr(cml_date);
  cmcp_default_text = '';
  cmcp_read_only = 0;
  cmcp_is_hidden = 0;
  cmcp_combo_list = '';
  param_value = cmcp_default_value;
  cmcp_read_only = maxvalue(cmcp_read_only, is_read_only);
  suspend;

  for select cmcp_param_name, cmcp_caption, cmcp_description, cmcp_edit_kind,
             cmcp_max_length, cmcp_default_value, cmcp_default_text, cmcp_read_only, cmcp_is_hidden, cmcp_combo_list
  from com_cfg_params
  where cmcp_from_cmc_id = :cmc_id
  order by cmcp_order
  into :cmcp_param_name, :cmcp_caption, :cmcp_description, :cmcp_edit_kind,
       :cmcp_max_length, :cmcp_default_value, :cmcp_default_text, :cmcp_read_only, :cmcp_is_hidden, :cmcp_combo_list
  do
  begin
    if (cmcp_edit_kind = 'refdata') then
    begin
      if (coalesce(cmcp_default_value, '') <> '') then
      begin
        select rd_name
        from refdata
        where rd_id = :cmcp_default_value
        into cmcp_default_text;
      end

      cmcp_edit_kind = 'combo';
      select combo_list
      from hp_com_list_refdata_json(:cmcp_combo_list)
      into :cmcp_combo_list;
    end

    if (cml_add_params <> '')
    then param_value = f_json_str_get_value(cml_add_params, cmcp_param_name, cmcp_default_value);
    else param_value = cmcp_default_value;

    if (cmcp_read_only = 1 and
        param_value = cmcp_default_value and
        cmcp_default_text <> '') then
      param_value = cmcp_default_text;

    cmcp_read_only = maxvalue(cmcp_read_only, is_read_only);
    suspend;
  end
end
