CREATE PROCEDURE FIND_DELINQ_CREDITS(
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    C_INN D_STR_20,
    REP_DATE D_DATE = current_date)
RETURNS (
    CLIENT_INFO D_STR_255,
    KONTRACT_NO D_STR_100,
    OKS_DELINQ_DAY D_INTEGER)
AS
declare variable c_id d_id;
begin
  select c_id,
         c_name_fio || ', ' || f_datetostr(c_birth_date)
  from clients
  where ((c_name_fio = :c_name_fio and
        c_birth_date = :c_birth_date) or
        c_inn = :c_inn) and
        c_id > 0 rows 1
  into c_id,
       client_info;

  if (c_id is not null) then
  begin
    select list(k_kontract_no, ', '),
           max(:rep_date - oks_delinq_date)
    from kontract
    inner join knclient on kc_from_k_id = k_id
    inner join oksummary on oks_from_k_id = k_id
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          oks_close_state = 0 and
          :rep_date between oks_date_begin and oks_date_end and
          :rep_date - oks_delinq_date > 0
    into kontract_no, -- список контрактов с активной просрочкой
         oks_delinq_day; -- кол-во дней максимальной активной просрочки

    if (oks_delinq_day > 0) then
      suspend;
  end
end
