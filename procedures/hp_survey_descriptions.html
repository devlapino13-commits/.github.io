CREATE PROCEDURE HP_SURVEY_DESCRIPTIONS(
    ID INTEGER)
RETURNS (
    ANSWER_ID INTEGER,
    QUES_CAPTION VARCHAR(255),
    QUES_MEMO VARCHAR(255),
    ANSWER_CODES VARCHAR(255),
    ANSWER_CAPTION VARCHAR(255),
    ANSWER_MEMO VARCHAR(255),
    GLOBAL_MEMO VARCHAR(1024))
AS
begin
  global_memo = 'Отправьте СМС на номер +7 903 7676000 с вариантом ответа или голосуйте через мобильное приложение.';

  select svq_caption, svq_memo
  from survey_ques
  where svq_id = :id
  into :ques_caption, ques_memo;

  for select sva_id, sva_codes, sva_caption, sva_memo
      from survey_answer
      where sva_from_svq_id = :id
      order by sva_codes
      into :answer_id, answer_codes, answer_caption, answer_memo
  do
  begin
    answer_codes = substring(answer_codes from 2 for char_length(answer_codes) - 2);
    answer_codes = replace(answer_codes, '~', ' / ');
    suspend;
  end
end
