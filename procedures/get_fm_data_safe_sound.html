CREATE PROCEDURE GET_FM_DATA_SAFE_SOUND(
    CATEGORY D_STR_20,
    BLS_ID D_ID,
    BASE_PATH D_STR_100)
RETURNS (
    MAX_PAST_DUE D_INTEGER,
    TOTAL_OUTSTANDING_NOT_OUR D_CURRENCY,
    TOTAL_AMOUNT_NOT_OUR D_CURRENCY,
    GUARANTOR D_INTEGER)
AS
declare variable path d_str_255;
declare variable subcategory d_str_20;
declare variable max_past_due_days d_integer;
declare variable arr_length d_integer;
declare variable i d_integer;
declare variable json_blb d_blob_text;
declare variable provider d_str_255;
declare variable loan_status_code d_str_10;
declare variable role_code d_str_10;
begin
  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into :json_blb;
  
  max_past_due = 0;
  subcategory = null;
  total_outstanding_not_our = 0;
  total_amount_not_our = 0;
  guarantor = 0;

  for select trim('granted_contract') as subcategory
    from rdb$database
    union all
    select trim('not_granted_contract') from rdb$database
    into :subcategory
  do
  begin
    path = base_path || '.' || category || '.' || subcategory;
    arr_length = f_json_blob_get_count(:json_blb, :path);
    i = 0;
    max_past_due_days = 0;
    while (i < arr_length)
    do
    begin
      provider = f_json_blob_get_value(:json_blb, :path || f_format('[%1].%2.provider_code_desc', i, :subcategory), '');
      if (provider <> 'MKK M Bulak') then
      begin
        max_past_due_days = cast(f_json_blob_get_value(json_blb, path || f_format('[%1].maximum.max_days_past_due', i), 0) as d_integer);
        if (max_past_due_days > max_past_due) then
          max_past_due = max_past_due_days;
        total_outstanding_not_our = total_outstanding_not_our + cast(f_json_blob_get_value(json_blb, path || f_format('[%1].granted_installment.principal_outstanding_amount', i), 0) as d_currency);
        loan_status_code = f_json_blob_get_value(json_blb, path || f_format('[%1].granted_contract.contract_phase', i), '');
        if (loan_status_code = 'AC') then
          total_amount_not_our = total_amount_not_our + 1;
          
        role_code = f_json_blob_get_value(json_blb, path || f_format('[%1].granted_contract.role', i), '');
        if (role_code = 'G') then
          guarantor = guarantor + 1;
      end
      i = i + 1;
      suspend;
    end
  end
end
