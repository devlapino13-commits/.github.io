CREATE PROCEDURE HP_FINPOLIS_INSERT(
    IN_NUMBER VARCHAR(20),
    IN_NAMEU VARCHAR(255),
    IN_GR DATE,
    IN_TERRTYPE VARCHAR(255))
AS
declare variable v_terr_categ_id integer;
declare variable fp_id integer;
begin
  in_terrtype = trim(in_terrtype);

  select rd.rd_id
  from refdata rd
  where rd.rd_main_code = 'FionpolisDirectory' and
        rd.rd_parent_id is not null and
        rd.rd_name containing (:in_terrtype) rows 1
  into :v_terr_categ_id;

  in_number = replace(in_number, ',', '.');
  in_number = substring(in_number from 1 for position('.', in_number) - 1);

  insert into finpolis (number, nameu, gr, terr_categ_id)
  values (:in_number, :in_nameu, :in_gr, :v_terr_categ_id)
  returning fp_id into :fp_id;

-- добавлено в триггер FINPOLIS_AI0
--execute procedure hp_finpolis_parcer_fuzzy (:fp_id);
end
