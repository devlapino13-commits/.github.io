CREATE PROCEDURE HP_DOCRECOGREQUEST_NOTIFY(
    DRR_ID INTEGER,
    KIND VARCHAR(50))
AS
declare variable msg_subject d_str_code;
declare variable msg_body d_str_long;
declare variable reason_str d_str_255;
declare variable c_id d_id;
begin
  -- Процедура рассылки уведомлений (PUSH, SMS) для запросов на закрепление документов

  if (kind in ('on_create', 'on_rework')) then
  begin
    select drr_from_c_id, rd_name
    from docrecogrequest
      left join refdata on rd_id = drr_reason_rd_id
    where drr_id = :drr_id
    into c_id, reason_str;

    if (kind = 'on_create') then
    begin
      msg_subject = 'Прикрепите документ';
      msg_body = 'Прикрепите в приложении М Булак запрошенный документ.';
    end

    if (kind = 'on_rework') then
    begin
      msg_subject = 'Прикрепите документ повторно';
      msg_body = f_format('Документ, который вы прикрепили, не прошел проверку качества. Пожалуйста, исправьте. Причина: %1', reason_str);
    end

    if (c_id is null or msg_body is null) then
      exit;


    execute procedure mc_send_push('clients', c_id, msg_subject, msg_body,  5);
  end

  if (kind = 'on_attach') then
  begin
    msg_subject = 'Верифицируйте документ';
    msg_body = 'Клиент прикрепил запрошенный документ.';

    select un_from_c_id
    from docrecogrequest
      left join username on un_id = drr_from_un_id
    where drr_id = :drr_id
    into c_id;

    if (c_id is null or msg_body is null) then
      exit;

    execute procedure hp_send_push_to_staff (c_id, msg_subject, msg_body, 'hp_docrecogrequest_notify.' || kind, 5);
  end


end
