CREATE PROCEDURE FM_HTML_EXTINFO_INCOME(
    EIL_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable LANG D_STR_LANG;
declare variable C_ID D_ID;
declare variable EIL_DATE D_DATE;
declare variable C_NAME_FIO D_STR_255;
declare variable K_ID D_ID;
declare variable CALC_STATUS D_STR_2048;
declare variable JSON_DOCUMENT D_BLOB_TEXT;
declare variable EIC_CODE D_STR_CODE;
declare variable ARR_LENGTH D_INTEGER;
declare variable I D_INTEGER;
declare variable BIN D_STR_CODE;
declare variable PAYMENT_DATE D_STR_CODE;
declare variable PAYMENT_NAME D_STR_255;
declare variable PAYMENT_SUM D_CURRENCY;
declare variable TOTAL_PAYMENT_SUM D_CURRENCY;
begin
  -- RFCKG-4255 Справки о доходах (ЗП, пенсия)
  -- отображение полученных справок от сервиса 
  lang = f_get_lang_code(un_id, null);

  for select eil_date, res_blob, eic_code, eil_from_c_id
  from extinfolist eil, extinfocfg eic, fm_blob_get(eil_from_bls_id)
  where eil_id = :eil_id and
        eil_from_eic_id = eic_id and
        eil_is_json = 1 and
        eil_status = 2
  rows 1
  into eil_date, :json_document, eic_code, c_id
  do
  begin
    c_name_fio = f_lang(lang, 'Клиент не найден');

    if (row_count = 0) then
      calc_status = f_lang(lang, 'Информация о пенсии не получена');
    else
    begin
      if (coalesce(:json_document, 0) = 0) then
        calc_status = f_lang(lang, 'Информация о доходах по клиенту отсутствует.');
    end
  
    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;
    res_html = (select res_style
                from fm_html_styles) || f_format('<body><a name="top"></a><h2>%1</h2>%2<br>', c_name_fio, calc_status);

    i = 0;
  
    arr_length = f_json_blob_get_count(:json_document, 'baseResponse.deductionDetailed');
  
    res_html = BLOB_APPEND(res_html, f_format('<h1>%1</h1>', iif(eic_code = 'salary_check', 'Информация о заплате', 'Информация о пенсии')));
  
    while(i < arr_length) do
    begin
      bin = f_json_blob_get_value(:json_document, f_format('baseResponse.deductionDetailed[%1].bin', i), '');
      payment_date = f_datetostr(cast(replace(f_json_blob_get_value(:json_document, f_format('baseResponse.deductionDetailed[%1].date', i), ''), 'T', ' ') as d_datetime));
      payment_name = f_json_blob_get_value(:json_document, f_format('baseResponse.deductionDetailed[%1].name', i), '');
      payment_sum = cast(f_json_blob_get_value(:json_document, f_format('baseResponse.deductionDetailed[%1].paymentSum', i), '') as d_currency);
      total_payment_sum = cast(f_json_blob_get_value(:json_document, f_format('baseResponse.deductionDetailed[%1].TotalpaymentSum', i), '') as d_currency);
  
      res_html = BLOB_APPEND(res_html, '<table><table width="100%"><col width="40%"><col width="60%">');
      res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'БИН работодателя') || '</b>', :bin, '', '', -1, -1, -1, -1, 2, ''));
      res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Дата получения зарплаты') || '</b>', :payment_date, '', '', -1, -1, -1, -1, 2, ''));
      res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Наименование работодателя') || '</b>', :payment_name, '', '', -1, -1, -1, -1, 2, ''));
      res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Сумма доходов') || '</b>', :payment_sum, '', '', -1, -1, -1, -1, 2, ''));
      res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Сумма доходов до налогообложения') || '</b>', :total_payment_sum, '', '', -1, -1, -1, -1, 2, ''));
      res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Дата и время получения ответа Компанией') || '</b>', f_datetostr(eil_date), '', '', -1, -1, -1, -1, 2, ''));
      res_html = BLOB_APPEND(res_html, '</table>');
      res_html = BLOB_APPEND(res_html, '<br>');
  
      i = i + 1;
    end
  end

  res_html = BLOB_APPEND(res_html, '</body>');
  suspend;
end
