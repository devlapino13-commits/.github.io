CREATE PROCEDURE MC_PHONE_CHANGE_CLEANUP
AS
begin
    /*
        RFCRU-5165: Процедура очищает зависшие заявки на смену номера телефона
        Зависшими считаем заявки, которые были созданы 24 часа назад и более,
        и у которых статус: "документ еще не создан" или "ошибка создания документа"
        (см. enum_list, el_kind: mc_phone_change)
    */

    delete from mc_phone_change mpc
    where datediff(minute, mpc.mpc_datetime, current_timestamp) >= 24 * 60 and -- 24 часа
          mpc.mpc_status in (f_enum_id('mc_phone_change', 'not_yet_created'), f_enum_id('mc_phone_change', 'doc_creation_error'));

    when gdscode lock_conflict,
         gdscode lock_timeout,
         gdscode update_conflict,
         gdscode deadlock
    do
    begin
        -- Если случается ошибка блокировки, ничего не делаем, для данной таблицы это не критично
    end
end
