CREATE PROCEDURE HP_KONTUR_IP_GET_BASE_INFO(
    OBJECT_KIND D_STR_CODE,
    OBJECT_ID D_ID)
RETURNS (
    REQ_STATUS D_SMALLINT,
    REQ_DATE D_DATE,
    ACTUALITY_DATE D_DATE,
    OGRN D_STR_CODE,
    IP_STATUS_STR D_STR_255,
    IP_STATUS D_SMALLINT,
    REG_DATE D_DATE)
AS
declare variable jso d_integer;
declare variable status_jso d_integer;
begin
  -- RFCRU-5304 получение ответа из Контур (Базовые реквизиты)
  -- входной object_kind может быть miniapp, kontract, knclient, clients.
/*
  req_status -- eil_status
  req_date -- Дата запроса
  actuality_date -- Дата актуальности (Дата запроса + eic_info_actual_days
*/
  select jso, req_status, req_date, actuality_date
  from hp_extinfo_get_jso('kontur_ip_base', :object_kind, :object_id)
  into jso, req_status, req_date, actuality_date;

/*
  ogrn -- Регистрационный номер индивидуального предпринимателя (ОГРНИП)  Api3ReqResponse[0].ogrn
  ip_status_str -- Статус текст  Api3ReqResponse[0].IP.status.statusString
  ip_status -- Статус   iif(Api3ReqResponse[0].IP.status.dissolving = false and Api3ReqResponse[0].UL.status.dissolved = false, 1, 0)
  reg_date -- Дата регистрации Api3ReqResponse[0].IP.registrationDate
*/

  if (coalesce(req_status, 0) < 2) then   -- запрос не найден
  begin
    suspend;
    exit;
  end 

  if (jso is null) then
    exception e_default 'json пустой. ' || iif(exists(select 1 from hp_db_is_actual), '', 'Необходимо синхронизировать blobfiles');

  for select json.obj_get_val(jso, 'ogrn'), json.obj_get_val(jso, 'IP.registrationDate'), json.obj_get_obj(jso, 'IP.status')
  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'data'), 1)
  where json.contains(jso, 'IP') = 1
  order by cast(json.obj_get_val(jso, 'IP.registrationDate', '01.01.1900') as date) desc
  rows 1
  into ogrn, reg_date, status_jso
  do
  begin
    if (status_jso is not null) then
    begin
      ip_status_str = json.obj_get_val(status_jso, 'statusString'); -- Статус текст

      -- Статус
      ip_status = iif(cast(json.obj_get_val(status_jso, 'dissolving', false) as d_boolean) or
                          cast(json.obj_get_val(status_jso, 'dissolved', false) as d_boolean),
                      0, 1);
    end
  end

  suspend;
end
