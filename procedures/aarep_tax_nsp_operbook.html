CREATE PROCEDURE AAREP_TAX_NSP_OPERBOOK(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE,
    REPORT_KIND VARCHAR(10))
RETURNS (
    TAX_AMOUNT DECIMAL(18,4),
    TAX_REGION VARCHAR(100),
    TAX_REGION_ID INTEGER,
    TAX_BRANCH_ID INTEGER,
    TAX_AMOUNT_BALANCE DECIMAL(18,4),
    TAX_AMOUNT_DIFF DECIMAL(18,4),
    ACC_NSP VARCHAR(20),
    ACC_EXP VARCHAR(20),
    LLC_MEMO VARCHAR(50),
    IS_IPF INTEGER,
    LLC_IS_SPECIAL D_INTEGER)
AS
declare variable REG_CODE varchar(20);
declare variable REPORT_NSP_BALANCE varchar(20);
declare variable REPORT_NSP_ACC_NO integer;
declare variable REPORT_EXP_BALANCE varchar(20);
declare variable REPORT_EXP_ACC_NO integer;
declare variable DB_ID integer;
declare variable SQL_TEXT varchar(2048);
begin
  db_id = f_get_db_id();
  if (upper(report_kind) = 'LLC') then
  begin
    if (db_id in (39801, 41701)) then
    begin
      sql_text = 'With LLC_LIST as
        (select sum(lkc_llc_sum) as TAX_AMOUNT,
         lm_from_reg_id as TAX_REGION_ID,
         iif(llc_reserv_kind=''Общий резерв'',0,1) as LLC_IS_SPECIAL,
         offce_name as TAX_REGION,
         is_ipf
         from aarep_llc_clients(' || :report_region_id || ',''' || :report_date_end || ''')
         group by lm_from_reg_id, llc_reserv_kind, offce_name, is_ipf
         union select 0 as Tax_Amount, reg_unicode as TAX_REGION_ID, 0 as llc_is_special, reg_code||'' ''||reg_name as TAX_REGION, 0
         from region where reg_status=3
         union select 0 as Tax_Amount, reg_unicode as TAX_REGION_ID, 1 as llc_is_special, reg_code||'' ''||reg_name as TAX_REGION, 0
         from region where reg_status=3
         union select 0 as Tax_Amount, reg_unicode as TAX_REGION_ID, 0 as llc_is_special, reg_code||'' ''||reg_name as TAX_REGION, 1
         from region where reg_status=3
         union select 0 as Tax_Amount, reg_unicode as TAX_REGION_ID, 1 as llc_is_special, reg_code||'' ''||reg_name as TAX_REGION, 1
         from region where reg_status=3
         )
         select sum(Tax_Amount), TAX_REGION_ID, llc_is_special, TAX_REGION, is_ipf
         from LLC_LIST
         group by TAX_REGION_ID, llc_is_special, TAX_REGION, is_ipf';
    end

    if (db_id = 81001) then
    begin
      sql_text = 'With LLC_LIST as
        (select sum(llc_rppu) as Tax_Amount, llc_Office_id as TAX_REGION_ID, llc_is_special, llc_office_name as TAX_REGION
        From AAREP_LLC_REPORT_TEMP(' || :report_region_id || ',''' || :report_date_end || ''')
        group by llc_Office_id, llc_is_special, llc_office_name
        union select 0 as Tax_Amount, reg_unicode as TAX_REGION_ID, 0 as llc_is_special, reg_code||'' ''||reg_name as TAX_REGION
        from region where reg_status=3
        union select 0 as Tax_Amount, reg_unicode as TAX_REGION_ID, 1 as llc_is_special, reg_code||'' ''||reg_name as TAX_REGION
        from region where reg_status=3)
        select sum(Tax_Amount), TAX_REGION_ID, llc_is_special, TAX_REGION, 0 as is_ipf
        from LLC_LIST
        group by TAX_REGION_ID, llc_is_special, TAX_REGION';
    end

    if (db_id = 94401) then
    begin
      if (report_date_end >= '31.01.2025') then
        sql_text = 'With LLC_LIST as
          (select sum(llc_rppu) as Tax_Amount, llc_Office_id as TAX_REGION_ID, -1 as llc_is_special, llc_office_name as TAX_REGION
          From AAREP_LLC_REPORT_TEMP(' || :report_region_id || ',''' || :report_date_end || ''')
          group by llc_Office_id, llc_is_special, llc_office_name)
          select sum(Tax_Amount), TAX_REGION_ID, llc_is_special, TAX_REGION, 0 as is_ipf
          from LLC_LIST
          group by TAX_REGION_ID, llc_is_special, TAX_REGION';
       else
         sql_text = 'With LLC_LIST as
          (select sum(llc_rppu) as Tax_Amount, llc_Office_id as TAX_REGION_ID, llc_is_special, llc_office_name as TAX_REGION
          From AAREP_LLC_REPORT_TEMP(' || :report_region_id || ',''' || :report_date_end || ''')
          where llc_is_special > 0
          group by llc_Office_id, llc_is_special, llc_office_name)
          select sum(Tax_Amount), TAX_REGION_ID, llc_is_special, TAX_REGION, 0 as is_ipf
          from LLC_LIST
          group by TAX_REGION_ID, llc_is_special, TAX_REGION';
    end

    for execute statement :sql_text
            into :tax_amount, :tax_region_id, :llc_is_special, :tax_region, :is_ipf
    do
    begin
      tax_branch_id = :tax_region_id;
      if (llc_is_special = 0) then
        llc_memo = 'Общий РППУ';
      else
      if (llc_is_special = 1) then
        llc_memo = 'Спец.РППУ';
      else
        llc_memo = 'Формирование РППУ на основную сумму за ' || f_monthname(report_date_end);

      if (llc_is_special = 0) then
      begin
        if (db_id not in (39801, 41701)) then
        begin
          report_nsp_balance = '11002';
          report_exp_balance = '80431';
        end
        else
        begin
          if (is_ipf = 0) then
          begin
            report_nsp_balance = '11002';
            report_exp_balance = '80421';
          end
          else
          begin
            report_nsp_balance = '11032';
            report_exp_balance = '80433';
          end
        end

        report_nsp_acc_no = 1;
        report_exp_acc_no = 1;
      end
      else
      begin
        if (db_id = 94401) then
        begin
          report_nsp_balance = '21910';
          report_exp_balance = '89150';
        end
        if (db_id = 81001) then
        begin
          report_nsp_balance = '11001';
          report_exp_balance = '80431';
        end
        if (db_id in (39801, 41701)) then
        begin
          if (is_ipf = 0) then
          begin
            report_nsp_balance = '11001';
            report_exp_balance = '80421';
          end
          else
          begin
            report_nsp_balance = '11031';
            report_exp_balance = '80433';
          end
        end
        report_nsp_acc_no = 1;
        report_exp_acc_no = 1;
      end

      if (db_id = 94401 and report_date_end >= '31.01.2025') then
      begin
        report_nsp_balance = '21910';
        report_exp_balance = '89150';
      end

      select reg_code
      from region
      where reg_unicode = :tax_branch_id
      into :reg_code;
      acc_nsp = :reg_code || '417' || :report_nsp_balance || right(cast(1000000 + report_nsp_acc_no as varchar(7)), 6);
      acc_exp = :reg_code || '417' || :report_exp_balance || right(cast(1000000 + report_exp_acc_no as varchar(7)), 6);

    if (db_id = 94401) then
    begin
        acc_nsp = :reg_code || '944' || :report_nsp_balance || right(cast(1000000 + report_nsp_acc_no as varchar(7)), 6);
        acc_exp = :reg_code || '944' || :report_exp_balance || right(cast(1000000 + report_exp_acc_no as varchar(7)), 6);
    end

      tax_amount_balance = f_al_bal(acc_nsp, report_date_end);
      tax_amount_balance = -tax_amount_balance;

      tax_amount_diff = tax_amount - tax_amount_balance;
      if ((tax_amount_balance <> 0) or (tax_amount <> 0)) then
        suspend;
    end
  end
end
