CREATE PROCEDURE BMG_NEW_REQUEST(
    C_ID D_ID,
    UN_ID D_ID,
    BY_VERIFY_PHONE D_BOOL)
AS
declare variable EIC_ID D_ID;
declare variable INN D_STR_20;
declare variable PHONE_NUMBER D_STR_20;
declare variable EIL_ID D_ID;
declare variable REQUEST_JSON D_STR_100;
begin
  eic_id = (select t.eic_id from extinfocfg t where t.eic_code = 'bmg');

  if (eic_id is null) then
    exception e_message 'конфигурация extinfocfg = bmg не найдена';

  inn = (select cl.c_inn from clients cl where cl.c_id = :c_id);

  if (by_verify_phone = 1) then
    phone_number = f_cont_info('clients', :c_id, 'cell_main', 'ok');
  else
    phone_number = f_cont_info('clients', :c_id, 'cell_main', 'last');

  if (coalesce(phone_number, '') = '') then
    exception e_message 'телефонный номер не найден';

  insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_object_kind, eil_object_id, eil_is_json, eil_from_un_id)
  values (:c_id, :eic_id, 0, :c_id, 1, :un_id)
  returning eil_id
  into :eil_id;

  request_json = f_format('{"inn":"%1", "phone_number":"%2"}', inn, phone_number);

  insert into bmg_clients (bc_from_c_id, bc_from_eil_id, bc_request)
  values (:c_id, :eil_id, :request_json);

  post_event 'bmg';
end
