    K_KONTRACT_NO VARCHAR(20),
    CLIENT_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_NAME_FIO VARCHAR(100),

    RES_CLIENT_IS_RESIDENT INTEGER,

    CLI_PERC_SALDO CURR_TYPE,

    CLI_RPPU_PERC CURR_TYPE,

    CLI_PO_SUMM CURR_TYPE,
    TERM_IN_DAYS INTEGER,
    PERCENT_YEAR PERC_TYPE,

    OKS_DIFF_ESP_NORM CURR_TYPE,

    Z_COST_LIKVID CURR_TYPE,

    ID_RECOMMEND STR_20)
AS
declare variable SH_DATE date;
declare variable K_AMOUNT_APPROVED CURR_TYPE;
declare variable DOLYA double precision;
declare variable C_ID ID_TYPE;
declare variable CITIZEN_RU_ID integer;
declare variable DOC_VNJ_ID integer;
declare variable PR_CODE varchar(100);
declare variable PR_LOAN_TYPE integer;
declare variable K_KOLICH_CLIENT_APPROVED integer;
declare variable K_ID_PRIOR ID_TYPE;
declare variable PR_PRODUCT_KIND integer;
declare variable PR_ID integer;
declare variable OKS_BALANCE CURR_TYPE;
declare variable REP_PERIOD integer;
declare variable KCP_PERC_SALDO CURR_TYPE;
declare variable LK_BALANCE CURR_TYPE;
declare variable LK_CALCPERC CURR_TYPE;
declare variable PO_SUMM CURR_TYPE;
declare variable LK_DELIQ_DAYS integer;
declare variable MAX_SH_DATE date;
declare variable PERCENT_SCHEDULE decimal(18,4);
declare variable K_IS_DEVIDED integer;
declare variable K_BASE_K_ID ID_TYPE;
declare variable K_BASE_K_ID_OLD ID_TYPE;
declare variable LK_DELIQ_DAYS_TMP integer;
declare variable DELINQ_DAY_TMP integer;
declare variable LK_PERCENT_TMP PERC_TYPE;
declare variable PR_ID_OLD integer;
declare variable LK_DELIQ_DAYS_OLD integer;
declare variable PERIOD_DAY integer;
declare variable OKS_BALANCE_TMP CURR_TYPE;
declare variable KCP_PERC_SALDO_TMP CURR_TYPE;
declare variable K integer;
declare variable REP_DATE_BEGIN date;
declare variable OKS_CLOSE_STATE smallint;
declare variable OKS_BAL_LAW_TMP CURR_TYPE;
declare variable SAB_ID D_ID;
declare variable KNR_NON_RESIDENT smallint;
declare variable ACC_ID D_ID;
declare variable OKS_DIFF_ESP_BAL_TMP CURR_TYPE;
declare variable FIND_DRL_ID ID_TYPE;
declare variable DRV_FINAL_VALUE STR_255;
declare variable DRF_DATA_TYPE smallint;
declare variable DRF_CODE D_STR_CODE;
declare variable K_ID decimal(18,0);
declare variable AG_RD_ID integer;
declare variable K_FACT_DISB_OL_ID ID_TYPE;
declare variable SAB_ID_ZALOG_COST D_ID;
begin
  rep_period = extract(year from cast(:rep_date_end as date)) || right('0' || extract(month from cast(:rep_date_end as date)), 2);
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;

  citizen_ru_id = null;
  select rd_id
  from refdata
  where rd_code = 'Citizen.RU'
  into :citizen_ru_id;

  ag_rd_id = null;
  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.loyal_agreement'
  into ag_rd_id;

  doc_vnj_id = null;
  select pk_id
  from passkind, passdoctype
  where pdt_id = pk_from_pdt_id and
        pk_is_delete = 0 and
        pdt_code = 'registration_vnj' rows 1
  into :doc_vnj_id;

  k_id_prior = 0;
  k_base_k_id_old = 0;
  pr_id_old = 0;
  lk_deliq_days_old = 0;

  select sab_id
  from sys_accbalance
  --  where sab_full_code = 'Микрозаймы.Внебаланс.Залог'
  where sab_full_code = 'Микрозаймы.Внебаланс.Поручительство'
  into sab_id;        

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Залог'
  into :sab_id_zalog_cost;

  for select oks_from_k_id, oks_balance, :rep_date_end - oks_delinq_date, k_kontract_no, k_amount_approved, k_disb_date,
             k_kolich_client_approved, coalesce(k_is_devided, 0), k_base_k_id, k_from_pr_id, k_stop_perc, k_fact_disb_ol_id
      from oksummary oks, kontract kont
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks.oks_from_k_id = kont.k_id
            and oks_close_state = 0
            --            and k_kontract_no = '3801-089755-08'
      order by k_base_k_id, k_id
      into k_id, oks_balance, delinq_day_tmp, k_kontract_no, k_amount_approved, k_disb_date, k_kolich_client_approved,
           k_is_devided, k_base_k_id, pr_id, k_stop_perc, k_fact_disb_ol_id
  do
  begin

    u_cli = '';
    select sbl_name from sb_akt, sb_list
    where sba_from_k_id = :k_id
    and :rep_date_end between sba_date_begin and sba_date_end
    and sba_status = 2
    and sba_is_delete = 0
    and sba_from_sbl_id = sbl_id
    into u_cli;
    if (u_cli <> 'ук') then u_cli = '';

    kl_date = null;
    select max(kl_datetime) from kontlog
    where kl_from_k_id = :k_id
    and kl_state = 1
    into kl_date;
                      
    z_cost_market = 0;
    z_cost_likvid = 0;

    select sum(coalesce(zal.z_cost_market, 0))--, sum(coalesce(zal.z_cost_likvid, 0))
    from zalog zal
    where zal.z_from_k_id = :k_id
    into :z_cost_market;--, :z_cost_likvid;

    z_cost_market = coalesce(z_cost_market, 0);
--    z_cost_likvid = coalesce(z_cost_likvid, 0);

    lk_balance = null;
    lk_calcperc = null;
    lk_percent = null;

    lk_percent_tmp = null;
    lk_deliq_days_tmp = null;

    if (pr_id <> pr_id_old) then
    begin
      select pr_code, pr_loan_type, pr_product_kind, pr_name
      from product
      where pr_id = :pr_id
      into pr_code, pr_loan_type, pr_product_kind, pr_name;

      pr_id_old = pr_id;
    end

    if (delinq_day_tmp < 0) then
      delinq_day_tmp = 0;

    if (k_is_devided = 1) then
    begin
      if (k_base_k_id <> k_base_k_id_old) then
      begin
        delinq_day = 0;
        is_restruct = 0;
        restruct_date_last = null;
        delinq_day_rest_date = null;
        is_secured = 0;

        lk_deliq_days = null;

        select res_deliq_days, res_is_restruct, res_sh_close_date, restruct_date_last, is_secured, delinq_day_rest_date,
               lk_deliq_days
        from hp_devided_kont_result(:rep_date_end, :k_base_k_id)
        into :delinq_day, :is_restruct, max_sh_date, restruct_date_last, is_secured, delinq_day_rest_date,
             lk_deliq_days;

        k_base_k_id_old = k_base_k_id;
      end
      if (k_id <> k_base_k_id) then
      begin
        select kont.k_kontract_no
        from kontract kont
        where kont.k_id = :k_base_k_id
        into :k_kontract_no;
      end
    end
    else
    begin
      delinq_day = delinq_day_tmp;
      lk_deliq_days = null;

      select is_restruct, restruct_date_last
      from aarep_get_restructuring(:k_id, :rep_date_end, 81001)
      into :is_restruct, :restruct_date_last;

      max_sh_date = null;

      select max(sh_date)
      from schedule
      where sh_from_k_id = :k_id
      into max_sh_date;

      is_secured = null;
      select is_secured
      from aarep_get_secured_loan(:pr_code, :pr_loan_type, :pr_product_kind, :pr_id, :k_disb_date, :k_kolich_client_approved, :k_id)
      into is_secured;

      delinq_day_rest_date = null;
      if (is_restruct = 1) then
      begin
        restruct_date_last = restruct_date_last - 1;
        select :restruct_date_last - oks_delinq_date
        from oksummary
        where oks_from_k_id = :k_id and
              :restruct_date_last between oks_date_begin and oks_date_end
              and oks_close_state = 0
        into delinq_day_rest_date;

        if (delinq_day_rest_date < 0) then
          delinq_day_rest_date = null;
        if (delinq_day_rest_date > 0) then
          delinq_day_rest_date = delinq_day_rest_date + 1;
        restruct_date_last = restruct_date_last + 1;
      end
    end

    if (pr_product_kind = 4) then
      loan_emp = 1;
    else
      loan_emp = 0;

    percent_year = null;
    select kd_percent_per_day * 365
    from kontdata
    where :k_disb_date between kd_ol_date and kd_date_end and
          kd_ol_is_delete = 0 and
          kd_from_k_id = :k_id
    into percent_year;

    select sum(sh_perc)
    from schedule
    where sh_from_k_id = :k_id
    into percent_schedule;

    if (max_sh_date is not null) then
      term_in_days = max_sh_date - k_disb_date;
    else
      term_in_days = null;

    kcp_perc_saldo = null;

    select lk_balance, lk_calcperc, lk_percent, lk_deliq_days
    from llc_main, llc_kont
    where lm_is_signed = 1 and
          lm_id = lk_from_lm_id and
          :rep_date_end between lk_date and lk_date_end and
          lk_from_k_id = :k_id
    into lk_balance, lk_calcperc, lk_percent_tmp, lk_deliq_days_tmp;

    if (k_disb_date < '28.02.2012') then
    begin
      -- из kontcalcperc данные тянуть не могу, т.к. для амнистированных займов в kontcalcperc нет записей.
      select sum(sh_calc_perc)
      from aarep_make_zad_list(:k_id, :rep_date_end, 1)
      where sh_date is not null
      into :kcp_perc_saldo;

      kcp_perc_saldo = coalesce(kcp_perc_saldo, 0);
    end
    else
      kcp_perc_saldo = coalesce(lk_calcperc, 0);

    lk_balance = coalesce(lk_balance, 0);
    lk_calcperc = coalesce(lk_calcperc, 0);
    lk_percent_tmp = coalesce(lk_percent_tmp, 0);
    lk_deliq_days_tmp = coalesce(lk_deliq_days_tmp, 0);

    if (k_is_devided = 0) then
    begin
      lk_deliq_days = lk_deliq_days_tmp;
      lk_percent = lk_percent_tmp;
    end
    if (k_disb_date < '28.02.2012') then
    begin
      --          rppu_perc = 0;
      if (lk_balance = 0 and
          delinq_day = 0 and
          kcp_perc_saldo > 0 and
          rep_date_end > '30.09.2014') then
      begin
        sh_date = null;
        select sh_date
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        where sh_calc_perc > 0 and
              zad_perc = 0 and
              zad_os = 0 and
              sh_repay_type = 1 and
              sh_date is not null
        order by sh_date
        rows 1
        into sh_date;

        if (sh_date is not null) then
        begin
          delinq_day = :rep_date_end - :sh_date;
        end
      end
    end

    if (delinq_day < lk_deliq_days) then
      delinq_day = lk_deliq_days;

    --      rppu_os = cast(lk_balance as double precision); -- * lk_percent / 100;
    --      rppu_perc = cast(lk_calcperc as double precision); --* lk_percent / 100;

    --      if (k_disb_date < '28.02.2012') then rppu_perc = 0;

    select po_percent
    from get_percent_po(:rep_date_end, :delinq_day)
    into po_percent;

    po_summ = po_percent / 100 * (oks_balance + kcp_perc_saldo);

    oks_balance_tmp = oks_balance;
    kcp_perc_saldo_tmp = kcp_perc_saldo;
    k = 0;
    --    kol_cli = k_kolich_client_approved;
    for select kc_id, kc_from_c_id, kc_amount_approved, 1 - coalesce(kc_non_resident, 0)
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into kc_id, c_id, kc_amount, cli_is_resident_on_disb
    do
    begin
      c_name_fio = 'n' || :c_id;
      dolya = cast(kc_amount as double precision) / k_amount_approved;

      select 1 - res_non_resident
      from hp_get_client_type(:c_id, :citizen_ru_id, :doc_vnj_id, :rep_date_end)
      into :res_client_is_resident;

      period_day = max_sh_date - k_disb_date;

      if (k_is_devided = 1) then
      begin
        select percent_rvpz
        from hp_get_rvpz(:kc_amount, :period_day, :is_secured, :is_restruct, :lk_deliq_days)
        into lk_percent;
      end
      /*
        cli_balance = dolya * oks_balance;
        cli_perc_saldo = dolya * kcp_perc_saldo;

        k = k + 1;
        if (k < kol_cli) then
        begin
          cli_balance_round = round(cli_balance, 0);
          cli_perc_saldo_round = round(cli_perc_saldo, 0);

          oks_balance_tmp = oks_balance_tmp - cli_balance_round;
          kcp_perc_saldo_tmp = kcp_perc_saldo_tmp - cli_perc_saldo_round;
        end else
        begin
          cli_balance_round = oks_balance_tmp;
          cli_perc_saldo_round = kcp_perc_saldo_tmp;
        end
        */
      --        cli_rppu_os = dolya * rppu_os;
      --        cli_rppu_perc = dolya * rppu_perc;

      --        cli_rppu_os = cast(cli_rppu_os as double precision) * lk_percent / 100;
      --        cli_rppu_perc = cast(cli_rppu_perc as double precision) * lk_percent / 100;

      cli_po_summ = dolya * po_summ;

      cli_percent_schedule = dolya * percent_schedule;

      cli_balance = null;
      cli_perc_saldo = null;
      select oks_balance, oks_bal_perc, oks_kont_last_date, oks_last_sh_date, oks_restruct_date, oks_kommis_amount, -- доп %% график
             iif(oks_delinq_days > 0, oks_bal_zad_os, 0), iif(oks_delinq_days > 0, oks_bal_zad_perc, 0), oks_bal_kommis,
             oks_ecl_percent, oks_llr_os, -- iif(oks_disb_date >= '28.02.2018', oks_llr_os, 0),
             iif(oks_close_state = 0, oks_diff_ecl_llr, 0), iif(oks_close_state = 0, oks_ecl_vbs, 0), oks_ecl_acc_kind,
             oks_llr_kind,
             --             iif(oks_close_state = 0, oks_diff_esp_bal, 0),
             oks_llr_perc, oks_percent_esp, oks_bal_law, --as "Сальдо.госпош.",
             oks_reserv_law, --as "Резерв.госпош."
             oks_ead_amount -- Значение EAD для расчета РО
      from kn_oksummary
      where oks_from_kc_id = :kc_id and
            :rep_date_end between oks_date_begin and oks_date_end and
            (oks_close_state in (0, 1) or
            oks_bal_law <> 0 or
            oks_reserv_law <> 0)
      into cli_balance, cli_perc_saldo, oks_kont_last_date, -- дата погаш по первонач графику
           oks_last_sh_date, -- дата погаш по актуальному графику
           oks_restruct_date, -- дата 1ой реструктуризации
           oks_kommis_amount, -- доп %% график
           delinq_os, --  просроч од
           delinq_perc, -- просроч осн %%
           oks_bal_kommis, -- доп %% (4хх09)
           oks_ecl_percent, --ставка ро
           oks_llr_os, --рвпз од
           oks_diff_ecl_llr, -- кор-ка до ро
           oks_ecl_vbs, -- ро
           oks_ecl_acc_kind, --корзина ро
           oks_llr_kind, -- шкала рвпз
           --           oks_diff_esp_bal, -- накопит кор-ка до вбс епс (4хх07 или 4хх08)
           cli_rppu_perc, -- рвпз %%
           percent_effect, -- oks_percent_esp
           oks_bal_law, oks_reserv_law, oks_ead_amount;
      if (cli_balance is null) then
        cli_balance = 0;
      if (cli_perc_saldo is null) then
        cli_perc_saldo = 0;
      --        if (k_disb_date < '28.02.2012') then  oks_llr_os = cli_balance * oks_ecl_percent / 100;

      oks_diff_esp_norm = 0;
      select sum(oks_diff_esp_norm)
      from kn_oksummary
      where oks_from_kc_id = :kc_id and
            oks_date_begin between :rep_date_begin and :rep_date_end
      into oks_diff_esp_norm; -- в т.ч. Кор-ка от изм графика
      if (oks_diff_esp_norm is null) then
        oks_diff_esp_norm = 0;

      --      в т.ч. Кор-ка от разницы ставок
      oks_diff_esp_razn = oks_diff_esp_bal - oks_diff_esp_norm;

      select oks_close_state, oks_bal_law
      from kn_oksummary
      where oks_from_kc_id = :kc_id and
            oks_date_begin <= :rep_date_end and
            (oks_is_pmt = 1 or
            oks_is_month = 1)
      order by oks_date_begin desc
      rows 1
      into oks_close_state, oks_bal_law_tmp;

      if (not(oks_close_state = 2 and
          oks_bal_law_tmp = 0)) then
      begin

        knr_non_resident = 1 - cli_is_resident_on_disb;
        select knr_non_resident
        from kn_resident
        where knr_from_kc_id = :kc_id and
              :rep_date_end between knr_date_begin and knr_date_end
        into knr_non_resident;

        acc_no = null;
        if (knr_non_resident = 1) then
          acc_no = '492';
        else
        if (knr_non_resident = 0) then
          acc_no = '488';

        saldo_eps91414 = f_get_acc_amount(:kc_id, :sab_id, :rep_date_end, :rep_date_end, 'bal');

        kol_cli = iif(oks_close_state = 0, 1, 0);

        --      Накопительную корректировку вычисляем через обороты т.е. 48807 - 48808 + 49207 - 49208
        oks_diff_esp_bal = 0;
        for select acc_id
            from ref_account
            where acc_object_kind = 8 and
                  acc_object_id = :kc_id and
                  :rep_date_end >= acc_date_open and
                  acc_ab_code in (49207, 48807, 49208, 48808)
            into acc_id
        do
        begin
          oks_diff_esp_bal_tmp = 0;
          select kobs_bal_dt_sum - kobs_bal_kt_sum
          from kn_obsummary
          where kobs_acc_id = :acc_id and
                :rep_date_end between kobs_date and kobs_date_end
          into oks_diff_esp_bal_tmp;

          oks_diff_esp_bal = oks_diff_esp_bal + oks_diff_esp_bal_tmp;
        end

        client_name_fio = null;
        c_birth_date = null;
        select c_name_fio, c_birth_date
        from clients c
        where c_id = :c_id
        into client_name_fio, c_birth_date;

        pts_vin = null;
        pts_engine_no = null;
        pts_chassis_no = null;
        pts_body_no = null;

        if (pr_code containing '.zalog_car.' or pr_code containing '.zalog.') then
        begin
          find_drl_id = null;
          select cd.cds_from_drl_id
          from clientdocs cd
          where cd.cds_from_c_id = :c_id and
                cd.cds_is_delete = 0 and
                cds_from_drl_id > 0 and
                cd.cds_from_pk_id = 315 --Паспорт трансп. средства (ПТС)
          order by cds_doc_date_begin desc, cd.cds_datetime desc
          rows 1
          into find_drl_id;
          /*
          for
          select drv_final_value, drf_data_type, drf_code
          from docrecogvalue, docrecogfield
          where drv_from_drl_id = :find_drl_id and
                drv_from_drf_id = drf_id and
                drf_inactive = 0 and
                drf_code in ('vin', 'engine_no', 'chassis_no', 'body_no')
          into drv_final_value, drf_data_type, drf_code
          do
          begin
            if (drv_final_value = '' and drf_data_type <> 0) then drv_final_value = null;
            if (drf_code = 'vin') then pts_vin = drv_final_value;
            if (drf_code = 'engine_no') then pts_engine_no = drv_final_value;
            if (drf_code = 'chassis_no') then pts_chassis_no = drv_final_value;
            if (drf_code = 'body_no') then pts_body_no = drv_final_value;
          end */
          pts_vin = f_docrecog_value(:find_drl_id, 'vin'); --Идентификационный номер (VIN)
          pts_engine_no = f_docrecog_value(:find_drl_id, 'engine_no'); --Модель, № двигателя
          pts_chassis_no = f_docrecog_value(:find_drl_id, 'chassis_no'); --Шасси (рама) №
          pts_body_no = f_docrecog_value(:find_drl_id, 'body_no'); --Кузов (кабина, прицеп) №
                              
          z_cost_likvid = -f_get_acc_amount(:kc_id, :sab_id_zalog_cost, :rep_date_end, :rep_date_end, 'bal');  -- Залоговая стоимоть

        end

        final_pdn = null;
        select kc_pdn_koef from knclient_pdn
        where kc_id = :kc_id
          and kcp_kind = 0
        into final_pdn;

--      ID клиента в программе лояльности
        agent_id = null;
        select ag_id from agreements
        where ag_object_kind = 0
        and ag_object_id = :c_id
        and ag_from_rd_id = :ag_rd_id
        into agent_id;

--      ID рекомендателя
        id_recommend = null;
        select ogr_from_ag_id from operagr
        where ogr_from_ol_id = :k_fact_disb_ol_id
        and ogr_from_rd_id = :ag_rd_id
        and ogr_ol_is_active = 1
        into id_recommend;

        suspend;
      end
    end
