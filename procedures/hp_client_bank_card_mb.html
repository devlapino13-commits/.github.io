CREATE PROCEDURE HP_CLIENT_BANK_CARD_MB(
    C_ID ID_TYPE,
    ON_DATE D_DATE = current_date,
    HARD_CHECK D_SMALLINT = 1)
RETURNS (
    CARD_ID D_INTEGER,
    CCA_ID D_INTEGER,
    CCA_CODE D_STR_CODE,
    K_ID D_ID,
    K_DISB_DATE D_DATE)
AS
declare variable cal_status1 d_integer;
declare variable cal_status2 d_integer;
declare variable cal_status3 d_integer;
begin
  cal_status1 = f_enum('cardlist_status', 'transfer_to_client', 'id');
  if (hard_check = 1) then
  begin
    cal_status2 = cal_status1;
    cal_status3 = cal_status1;
  end else
  begin
    cal_status2 = f_enum('cardlist_status', 'stoplist_in', 'id');
    cal_status3 = f_enum('cardlist_status', 'hard_stoplist_in', 'id');
  end

  if (c_id > 0)
  then
    select first 1
      cal_card_id,
      cca_id,
      cca_code,
      k_id,
      k_disb_date
    from cardlistlnk
      inner join cardlist
        on cal_card_id = cll_card_id
          and cal_status in (:cal_status1, :cal_status2, :cal_status3)
      inner join cardclientacc
        on cca_id = cal_from_cca_id
      left join kontract
        on k_id = cll_from_k_id
    where (cll_from_c_id = :c_id)
      and (:hard_check = 0 or :on_date between cll_date_begin and cll_date_end)
    order by cal_status, cll_date_begin desc
    into
      :card_id,
      :cca_id,
      :cca_code,
      :k_id,
      :k_disb_date;
  else
    select first 1
      cal_card_id,
      cca_id,
      cca_code,
      k_id,
      k_disb_date
    from cardlistloan
      inner join cardlist
        on cal_card_id = clo_card_id
          and cal_status in (:cal_status1, :cal_status2, :cal_status3)
      inner join cardclientacc
        on cca_id = cal_from_cca_id
      left join kontract
        on k_id = clo_from_k_id
    where (clo_from_k_id = -:c_id)
      and (:hard_check = 0 or k_disb_date between clo_date and current_date)
    order by cal_status, clo_date desc
    into
      :card_id,
      :cca_id,
      :cca_code,
      :k_id,
      :k_disb_date;

  if (card_id > 0) then
    suspend;
end
