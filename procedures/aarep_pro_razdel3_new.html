CREATE PROCEDURE AAREP_PRO_RAZDEL3_NEW(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_REVERSE SMALLINT,
    REP_CURRENCY VARCHAR(3),
    REPORT_ID INTEGER)
RETURNS (
    BSL_NAME STR_255,
    RESTRUCT_BALANCE CURR_TYPE,
    DELINQ_BALANCE CURR_TYPE,
    LLC_STANDART_BALANCE CURR_TYPE,
    LLC_POD_NABLUD_BALANCE CURR_TYPE,
    LLC_SUBSTANDART_BALANCE CURR_TYPE,
    LLC_SOMNITEL_BALANCE CURR_TYPE,
    LLC_POTERY_BALANCE CURR_TYPE,
    NO_ACCRUAL_STATUS_BALANCE CURR_TYPE,
    LLC_OBSHY_REZERV CURR_TYPE,
    LLC_SPEC_REZERV CURR_TYPE,
    DELINQ_KOL INTEGER,
    LLC_STANDART_KOL INTEGER,
    LLC_POD_NABLUD_KOL INTEGER,
    LLC_SUBSTANDART_KOL INTEGER,
    LLC_SOMNITEL_KOL INTEGER,
    LLC_POTERY_KOL INTEGER)
AS
declare variable ed_izmer curr_type;
declare variable restruct_balancej curr_type;
declare variable delinq_balancej curr_type;
declare variable llc_standart_balancej curr_type;
declare variable llc_pod_nablud_balancej curr_type;
declare variable llc_substandart_balancej curr_type;
declare variable llc_somnitel_balancej curr_type;
declare variable llc_potery_balancej curr_type;
declare variable llc_obshy_rezervj curr_type;
declare variable llc_spec_rezervj curr_type;
declare variable no_accrual_status_balancej curr_type;

declare variable restruct_balanced curr_type;
declare variable delinq_balanced curr_type;
declare variable llc_standart_balanced curr_type;
declare variable llc_pod_nablud_balanced curr_type;
declare variable llc_substandart_balanced curr_type;
declare variable llc_somnitel_balanced curr_type;
declare variable llc_potery_balanced curr_type;
declare variable llc_obshy_rezervd curr_type;
declare variable llc_spec_rezervd curr_type;
declare variable no_accrual_status_balanced curr_type;
begin
  ed_izmer = 1000;
  delete from temp_nbkr_pro;
  -- < строка количества
  select sum(iif(is_delinq = 1, 1, 0)), --просроченные
         sum(iif(lkc_status in (1, 3), 1, 0)), -- стандартные
         sum(iif(lkc_status = 4, 1, 0)), -- активы под наблюдением
         sum(iif(lkc_status = 5, 1, 0)), -- субстандартные
         sum(iif(lkc_status = 6, 1, 0)), -- сомнительные
         sum(iif(lkc_status = 7, 1, 0)) -- потери
  from aarep_restrukt_dt_client(:rep_region_id, :rep_date_end) rep

  into delinq_kol, llc_standart_kol, llc_pod_nablud_kol, llc_substandart_kol, llc_somnitel_kol, llc_potery_kol;

  -- < столбец 4
  for select bt_name, sum(iif(dlq.restruct_date is not null, dlq.rep_active_balance, 0)) / :ed_izmer -- реструктуризированные
      from aarep_restrukt_dt_client(:rep_region_id, :rep_date_end) dlq
      group by bt_name
      into bsl_name, restruct_balance

  do
  begin
    insert into temp_nbkr_pro (bsl_name, restruct_balance)
    values (:bsl_name, :restruct_balance);
  end -->

  -- < столбей 6,6-1,7,8,9,11,12
  for select bt_name,

             sum(iif(stan.lkc_status in (1, 3), stan.lkc_balance, 0)) / :ed_izmer, -- стандартные
             sum(iif(stan.lkc_status = 4, stan.lkc_balance, 0)) / :ed_izmer, -- активы под наблюдением
             sum(iif(lkc_status = 5, stan.lkc_balance, 0)) / :ed_izmer, -- субстандартные
             sum(iif(lkc_status = 6, stan.lkc_balance, 0)) / :ed_izmer, -- сомнительные
             sum(iif(lkc_status = 7, stan.lkc_balance, 0)) / :ed_izmer, -- потери
             sum(iif(stan.llc_reserv_kind = 'общий резерв', stan.lkc_llc_sum, 0)) / :ed_izmer, -- общий резерв
             sum(iif(stan.llc_reserv_kind = 'специальный резерв', stan.lkc_llc_sum, 0)) / :ed_izmer -- специальный резерв

      from aarep_llc_clients(:rep_region_id, :rep_date_end) stan
      group by bt_name
      into bsl_name, llc_standart_balance, llc_pod_nablud_balance, llc_substandart_balance, llc_somnitel_balance,
           llc_potery_balance, llc_obshy_rezerv, llc_spec_rezerv
  do
  begin
    if (exists(select *
               from temp_nbkr_pro ntp
               where ntp.bsl_name = :bsl_name)) then
    begin
      update temp_nbkr_pro
      set llc_standart_balance = :llc_standart_balance,
          llc_pod_nablud_balance = :llc_pod_nablud_balance,
          llc_substandart_balance = :llc_substandart_balance,
          llc_somnitel_balance = :llc_somnitel_balance,
          llc_potery_balance = :llc_potery_balance,
          llc_obshy_rezerv = :llc_obshy_rezerv,
          llc_spec_rezerv = :llc_spec_rezerv
      where bsl_name = :bsl_name;
    end
    else
    begin
      insert into temp_nbkr_pro (bsl_name, llc_standart_balance, llc_pod_nablud_balance, llc_substandart_balance,
                                 llc_somnitel_balance, llc_potery_balance, llc_obshy_rezerv, llc_spec_rezerv)
      values (:bsl_name, :llc_standart_balance, :llc_pod_nablud_balance, :llc_substandart_balance,
              :llc_somnitel_balance, :llc_potery_balance, :llc_obshy_rezerv, :llc_spec_rezerv);
    end
  end -->
  --< столбей 10
  for select knc_business_type, sum(iif(no_accrual_status = 1, knc_rep_active_balance, 0)) / :ed_izmer -- статус не начисления
      from aarep_cli_stopperc(:rep_region_id, :rep_date_end) stan
      group by knc_business_type
      into bsl_name, no_accrual_status_balance

  do
  begin
    if (exists(select *
               from temp_nbkr_pro ntp
               where ntp.bsl_name = :bsl_name)) then
    begin
      update temp_nbkr_pro
      set no_accrual_status_balance = :no_accrual_status_balance
      where bsl_name = :bsl_name;
    end
    else
    begin
      insert into temp_nbkr_pro (bsl_name, no_accrual_status_balance)
      values (:bsl_name, :no_accrual_status_balance);
    end
  end -->

  -- < столбей 5
  for select knc_business_type, sum(rep.knc_rep_active_balance) / :ed_izmer -- ----
      from aarep_kont_clients_list(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_reverse, :rep_currency, :report_id) rep
      group by knc_business_type
      into bsl_name, delinq_balance

  do
  begin
    if (exists(select *
               from temp_nbkr_pro ntp
               where ntp.bsl_name = :bsl_name)) then
    begin
      update temp_nbkr_pro
      set delinq_balance = :delinq_balance
      where bsl_name = :bsl_name;
    end
    else
    begin
      insert into temp_nbkr_pro (bsl_name, delinq_balance)
      values (:bsl_name, :delinq_balance);
    end
  end -->

  for select bsl_name, restruct_balance, delinq_balance, llc_standart_balance, llc_pod_nablud_balance,
             llc_substandart_balance, llc_somnitel_balance, llc_potery_balance, llc_obshy_rezerv, llc_spec_rezerv,
             no_accrual_status_balance
      from temp_nbkr_pro
      into :bsl_name, :restruct_balance, :delinq_balance, :llc_standart_balance, :llc_pod_nablud_balance,
           :llc_substandart_balance, :llc_somnitel_balance, :llc_potery_balance, :llc_obshy_rezerv, :llc_spec_rezerv,
           :no_accrual_status_balance
  do
  begin

    select restruct_balance, delinq_balance, llc_standart_balance, llc_pod_nablud_balance, llc_substandart_balance,
           llc_somnitel_balance, llc_potery_balance, llc_obshy_rezerv, llc_spec_rezerv, no_accrual_status_balance
    from temp_nbkr_pro
    where bsl_name = 'Животноводство'
    into :restruct_balancej, :delinq_balancej, :llc_standart_balancej, :llc_pod_nablud_balancej,
         :llc_substandart_balancej, :llc_somnitel_balancej, :llc_potery_balancej, :llc_obshy_rezervj, :llc_spec_rezervj,
         :no_accrual_status_balancej;

    if (bsl_name = 'Сельское хозяйство') then
    begin
      restruct_balance = restruct_balance + restruct_balancej;
      delinq_balance = delinq_balance + delinq_balancej;
      llc_standart_balance = llc_standart_balance + llc_standart_balancej;
      llc_pod_nablud_balance = llc_pod_nablud_balance + llc_pod_nablud_balancej;
      llc_substandart_balance = llc_substandart_balance + llc_substandart_balancej;
      llc_somnitel_balance = llc_somnitel_balance + llc_somnitel_balancej;
      llc_potery_balance = llc_potery_balance + llc_potery_balancej;
      no_accrual_status_balance = no_accrual_status_balance + no_accrual_status_balancej;
      llc_obshy_rezerv = llc_obshy_rezerv + llc_obshy_rezervj;
      llc_spec_rezerv = llc_spec_rezerv + llc_spec_rezervj;
    end

    select restruct_balance, delinq_balance, llc_standart_balance, llc_pod_nablud_balance, llc_substandart_balance,
           llc_somnitel_balance, llc_potery_balance, llc_obshy_rezerv, llc_spec_rezerv, no_accrual_status_balance
    from temp_nbkr_pro
    where bsl_name = 'Другое'
    into :restruct_balanced, :delinq_balanced, :llc_standart_balanced, :llc_pod_nablud_balanced,
         :llc_substandart_balanced, :llc_somnitel_balanced, :llc_potery_balanced, :llc_obshy_rezervd, :llc_spec_rezervd,
         :no_accrual_status_balanced;

    if (bsl_name = 'Физические лица зарплата') then
    begin
      restruct_balance = restruct_balance + restruct_balanced;
      delinq_balance = delinq_balance + delinq_balanced;
      llc_standart_balance = llc_standart_balance + llc_standart_balanced;
      llc_pod_nablud_balance = llc_pod_nablud_balance + llc_pod_nablud_balanced;
      llc_substandart_balance = llc_substandart_balance + llc_substandart_balanced;
      llc_somnitel_balance = llc_somnitel_balance + llc_somnitel_balanced;
      llc_potery_balance = llc_potery_balance + llc_potery_balanced;
      no_accrual_status_balance = no_accrual_status_balance + no_accrual_status_balanced;
      llc_obshy_rezerv = llc_obshy_rezerv + llc_obshy_rezervd;
      llc_spec_rezerv = llc_spec_rezerv + llc_spec_rezervd;
    end

    if (bsl_name <> 'Животноводство' and
        bsl_name <> 'другое') then
    begin
      if (bsl_name = 'Производство') then
        bsl_name = 'a) промышленность';

      if (bsl_name = 'Сельское хозяйство') then
        bsl_name = 'б) сельское хозяйство';

      if (bsl_name = 'Торговля') then
        bsl_name = 'г) торговые и коммерческие операции';

      if (bsl_name = 'Услуги') then
        bsl_name = 'д) услуги';

      if (bsl_name = 'Физические лица зарплата') then
        bsl_name = 'и) физические лица (потребит. кредиты)';
      suspend;
    end
  end
end
