CREATE PROCEDURE SB_AKT_UPDATE(
    K_ID DECIMAL(18,0))
AS
declare variable k_no integer;
declare variable d date;
declare variable d_prior date;
begin
  /* Procedure Text */
  d_prior = cast('31.12.9999' as date);

  update sb_akt
  set sba_date_end = sba_date_begin - 1
  where sba_from_k_id = :k_id and
        (sba_is_delete = 1 or
        sba_status < 0);/*IGUSEV закоментировал 17.04.13 and (sba_is_delete=1 or sba_status<=0);*/

  for select sba_number, sba_date_begin
      from sb_akt
      where sba_from_k_id = :k_id and
            sba_is_delete = 0 and
            sba_status >= 0 /*IGUSEV закоментировал 17.04.13   and sba_status>=1*/
      order by sba_date_begin desc, sba_number desc
      into :k_no, :d
  do
  begin
    update sb_akt
    set sba_date_end = :d_prior
    where sba_from_k_id = :k_id and
          sba_number = :k_no and
          (sba_date_end <> :d_prior or
          sba_date_end is null);
    d_prior = d - 1;
  end
  update kontract
  set k_is_problem = 1
  where k_id = :k_id and
        k_is_problem <> 1;
end
