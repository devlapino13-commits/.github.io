CREATE PROCEDURE HP_TRANSFERTMC_VIEWD(
    TRT_ID D_INTEGER)
RETURNS (
    BILL_ID D_ID,
    BILL_LINE_NO D_INTEGER,
    WPS_ID D_ID,
    WPS_NAME D_STR_100,
    WPR_ID D_ID,
    WPR_NAME D_STR_100,
    MOLS_ID D_ID,
    MOLS_NAME D_STR_100,
    MOLR_ID D_ID,
    MOLR_NAME D_STR_100,
    INVENT_NO D_INTEGER,
    INVENT_STR D_STR_CODE,
    MC_ID D_ID,
    ACC_BALANCE D_STR_20,
    ACC_AMORT D_STR_20,
    IS_CALC_AMORT D_SMALLINT,
    ML_ID D_ID,
    ML_NAME D_STR_255,
    ML_MODEL D_STR_255,
    YL_CODE D_STR_20,
    QTY D_CURRENCY,
    SUMMA D_CURRENCY,
    ACC D_STR_20,
    STATE_RD_ID D_INTEGER,
    STATE_NAME D_STR_255,
    STATE_NOTE D_STR_100)
AS
declare variable status d_smallint;
declare variable jsdata d_blob_text;
begin
  select
    trt_status,
    trt_json
  from transfertmc
  where trt_id = :trt_id
  into
    :status,
    :jsdata;

  if (status < 2) then
  begin
    select
      offices_id,
      c1.c_name_fio,
      officer_id,
      c2.c_name_fio
    from hp_transfertmc_jshead(:jsdata)
      inner join clients as c1
        on c1.c_id = offices_id
      inner join clients as c2
        on c2.c_id = officer_id
    into
      :mols_id,
      :mols_name,
      :molr_id,
      :molr_name;

    for
      select
        a.bill_id,
        a.bill_no,
        a.wps_id,
        a.wpr_id,
        a.qty,
        a.summa,
        a.acc,
        a.state_id,
        a.state_note,
        a.invent_no,
        a.invent_str,
        a.mc_id,
        mc.mc_acc_balance || '/' || mc.mc_acc_balance_no,
        mc.mc_acc_amort || '/' || mc.mc_acc_amort_no,
        mc.mc_is_calc_amort,
        a.ml_id,
        ml.ml_name,
        iif(ml.ml_model = ml.ml_name, ' -//-', ml.ml_model),
        yl_code
      from hp_transfertmc_jsdetail(:jsdata) as a
        inner join matcateg as mc
          on mc.mc_id = a.mc_id
      inner join matlist as ml
        on ml.ml_id = a.ml_id
      inner join yedizlist
        on yl_id = ml.ml_from_yl_id
    into
      :bill_id,
      :bill_line_no,
      :wps_id,
      :wpr_id,
      :qty,
      :summa,
      :acc,
      :state_rd_id,
      :state_note,
      :invent_no,
      :invent_str,
      :mc_id,
      :acc_balance,
      :acc_amort,
      :is_calc_amort,
      :ml_id,
      :ml_name,
      :ml_model,
      :yl_code
    do
    begin
      wps_name = (select wp_name from workplace where wp_id = :wps_id);
      wpr_name = (select wp_name from workplace where wp_id = :wpr_id);
      state_name = (select rd_name from refdata where rd_id = :state_rd_id);
      suspend;
    end
  end

  if (status = 2) then
  for
    select
      om_bill_id,
      om_bill_line_no,
      om_workplace_id,
      om_workplace_info,
      om_matotvet_id,
      om_matotvet_info,
      om_kolich_minus,
      om_summa_minus,
      om_account,
      om_state_rd_id,
      om_state_note,
      ob_invent_no,
      ob_invent_str,
      mc_id,
      mc_acc_balance || '/' || mc_acc_balance_no,
      mc_acc_amort || '/' || mc_acc_amort_no,
      mc_is_calc_amort,
      ml_id,
      ml_name,
      iif(ml_model = ml_name, ' -//-', ml_model),
      yl_code
    from transfertmc
      inner join opermat
        on om_from_ol_id = trt_save_ol_id
          and om_main_line = 1
      inner join operbill
        on ob_bill_id = om_bill_id
          and ob_bill_line_no = om_bill_line_no
          and trt_date between ob_date_begin and ob_date_end
          and ob_is_delete = 0
      inner join matcateg
        on mc_id = ob_from_mc_id
      inner join matlist
        on ml_id = ob_from_ml_id
      inner join yedizlist
        on yl_id = ml_from_yl_id
    where trt_id = :trt_id
  into
    :bill_id,
    :bill_line_no,
    :wps_id,
    :wpr_id,
    :mols_id,
    :molr_id,
    :qty,
    :summa,
    :acc,
    :state_rd_id,
    :state_note,
    :invent_no,
    :invent_str,
    :mc_id,
    :acc_balance,
    :acc_amort,
    :is_calc_amort,
    :ml_id,
    :ml_name,
    :ml_model,
    :yl_code
  do
  begin
    wps_name = (select wp_name from workplace where wp_id = :wps_id);
    wpr_name = (select wp_name from workplace where wp_id = :wpr_id);
    mols_name = (select c_name_fio from clients where c_id = :mols_id);
    molr_name = (select c_name_fio from clients where c_id = :molr_id);
    state_name = (select rd_name from refdata where rd_id = :state_rd_id);
    suspend;
  end
end
