CREATE PROCEDURE HP_EXTINFO_TO_SEND_INN_GNIVC(
    C_ID ID_TYPE,
    EIL_ID D_ID = 0)
RETURNS (
    JSON_VALUE D_STR_8100)
AS
declare variable BIRTH_DATE D_STR_CODE;
declare variable doc_no d_str_20;
declare variable PASS_SERIA D_STR_10;
declare variable PASS_NUMBER D_STR_CODE;
declare variable DOC_DATE D_STR_CODE;
declare variable DOC_TYPE D_SMALLINT;
declare variable FIRST_NAME D_STR_CODE;
declare variable LAST_NAME D_STR_CODE;
declare variable MIDDLE_NAME D_STR_CODE;
declare variable DOC_LIST D_STR_4096;
begin
  doc_list = '';

  for select t.icq_fname,
             t.icq_sname,
             t.icq_patronimic,
             f_datetostr(t.icq_datebirth),
             t.icq_doc_type,
             t.icq_doc_no,
             t.icq_pass_seria,
             t.icq_pass_number,
             f_datetostr(t.icq_doc_date)
      from inn_check_queries t
      where t.icq_from_c_id = :c_id
      and t.icq_from_eil_id = :eil_id
      into :first_name,
           :last_name,
           :middle_name,
           :birth_date,
           :doc_type,
           :doc_no,
           :pass_seria,
           :pass_number,
           :doc_date
  do
  begin
      doc_list = doc_list || iif(doc_list = '', '', ',') ||
        '{' ||
        f_json_make_pair('doctype', doc_type, 'str', 1) ||
        f_json_make_pair('docno', doc_no, 'str', 1) ||
        f_json_make_pair('docseries', pass_seria, 'str', 1) ||
        f_json_make_pair('docnumber', pass_number, 'str', 1) ||
        f_json_make_pair('docdt', doc_date, 'str') ||
        '}';

      doc_list = '[' || doc_list || ']';

      json_value = '{' ||
        f_json_make_pair('fam', last_name, 'str', 1) ||
        f_json_make_pair('nam', first_name, 'str', 1) ||
        iif(middle_name = '', '"opt_otch":"1",', f_json_make_pair('otch', middle_name, 'str', 1)) ||
        f_json_make_pair('bdate', birth_date, 'str', 1) ||
        f_json_make_pair('docs', doc_list, 'array') ||
        '}';

      suspend;
  end
end
