CREATE PROCEDURE HP_KONTMAT_ADD(
    K_ID ID_TYPE,
    AG_ID D_INTEGER,
    FEG_ID D_INTEGER,
    ML_NAME D_STR_255,
    ML_MODEL D_STR_255,
    ML_GUAR D_INTEGER,
    ML_COUNT CURR_TYPE,
    ML_COST CURR_TYPE,
    ML_UNIT D_INTEGER = 1,
    IS_RELATIVE D_INTEGER = 0,
    KML_IS_USED D_SMALLINT = 0,
    IMEI D_ID = null)
RETURNS (
    KML_ID D_INTEGER)
AS
declare variable ml_id id_type;
begin
  ml_id = (select ml_id
           from hp_matlist_insert(:ml_name, :ml_model, :ml_unit));
  insert into kontmat (kml_object_kind, kml_object_id, kml_from_ag_id, kml_from_feg_id, kml_from_ml_id,
                       kml_guarantee_month, kml_count, kml_cost, kml_is_relative, kml_is_used)
  values (9, :k_id, :ag_id, :feg_id, :ml_id, :ml_guar, :ml_count, :ml_cost, :is_relative, :kml_is_used)
  returning kml_id
  into :kml_id;
  if (imei is not null) then
    insert into kontmat_imei(kmi_kml_id, kmi_imei)
    values(:kml_id, :imei);
  suspend;
end
