CREATE PROCEDURE GET_KONT_SPS(
    K_ID ID_TYPE,
    REP_DATE DATE,
    IS_PERC_VALUE INTEGER = 1)
RETURNS (
    REP_SPS_SIMPLE PERC_TYPE,
    REP_SPS_NBKR PERC_TYPE,
    REP_SPS_SIMPLE_MULT_BALANCE PERC_TYPE,
    REP_SPS_NBKR_MULT_BALANCE PERC_TYPE)
AS
declare variable perc integer;
declare variable kolich_repay integer;
declare variable monthly_pay curr_type;
declare variable ipf_markup curr_type;
declare variable k_amount_approved curr_type;
declare variable pr_code str_100;
declare variable rep_percent_sum curr_type;
declare variable rep_balance_all curr_type;
begin
-- рассчитываю СПС по кредиту

-- REP_SPS_SIMPLE простой метод рассчета
-- REP_SPS_NBKR рассчет по формуле НБКР

-- Внимание! Полученное значение СПС (REP_SPS_SIMPLE, REP_SPS_NBKR) еще не является верным.
-- Чтобы довести рассчет до конца, нужно данное значение умножить на баланс кредита ( готовые значения в REP_SPS_SIMPLE_MULT_BALANCE, REP_SPS_NBKR_MULT_BALANCE)
-- и разделить на баланс всех кредитов в разрезе вычиcления (н-р: по компании/по продукту/ офису ...)

  perc = 1;
  if (is_perc_value = 1) then
    perc = 100;

  rep_sps_simple = 0;
  rep_sps_nbkr = 0;
  kolich_repay = 0;
  monthly_pay = 0;
  ipf_markup = 0;

  rep_percent_sum = 0;
  rep_balance_all = 0;

  select oks_balance as rep_balance_all,
         iif(pr.pr_code containing '.IPF.', kd_percent_per_day * iif(pr.pr_percent_for_days = 30, 3.6000, 3.6500), kd_percent_per_day * 3.6500) as rep_percent_sum,
         k_amount_approved, pr.pr_code
  from oksummary oks, kontdata kd, product pr, kontract kont
  where :rep_date between oks_date_begin and oks_date_end and
        oks.oks_from_k_id = :k_id and
        oks_balance > 0 and
        oks.oks_from_k_id = kd_from_k_id and
        kd_ol_is_delete = 0 and
        :rep_date between kd_ol_date and kd_date_end and
        oks_close_state = 0 and
        kont.k_id = oks.oks_from_k_id and
        pr.pr_id = kont.k_from_pr_id
  into :rep_balance_all, :rep_percent_sum, :k_amount_approved, :pr_code;

  if (rep_balance_all <> 0) then
  begin
    rep_sps_simple = rep_percent_sum * perc;
  end

  /*
    if (pr_code containing '.MURABAHA.') then
    begin
-- получаю срок займа в месяцах, с учетом реструктуризаций
      select  round((max(ok.ok_sh_date) - min(ok.ok_sh_date))/30)
      from operkred ok
      where ok.ok_from_k_id = :K_ID
      and ok.ok_ol_is_active = 1
      and ok.ok_ol_date <= :REP_DATE
      and ok_type in (1, 2, 3)
      into :kolich_repay;
      
-- сумма наценки, указанная в договоре
      select sum(sh.sh_perc), sum(sh.sh_os)/:kolich_repay
      from schedule sh
      where sh.sh_from_k_id = :K_ID
      into :ipf_markup, :monthly_pay;

      rep_sps_nbkr = (ipf_markup * kolich_repay)/(k_amount_approved * kolich_repay - monthly_pay *((kolich_repay - 1) * kolich_repay/2))*12/kolich_repay * perc;
    end
    else
    begin
        rep_sps_nbkr =  rep_sps_simple;
    end
    */
  rep_sps_nbkr = rep_sps_simple;

  rep_sps_simple_mult_balance = rep_sps_simple * rep_balance_all;
  rep_sps_nbkr_mult_balance = rep_sps_nbkr * rep_balance_all;

  suspend;
end
