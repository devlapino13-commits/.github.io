CREATE PROCEDURE FM_DOCRECOG_ADD_CLIENT(
    DRL_ID D_ID,
    IGNORE_DUPLICATE D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    C_ID ID_TYPE,
    RES_TEXT D_STR_255,
    IS_CRITICAL INTEGER)
AS
declare variable TMP_C_ID ID_TYPE;
declare variable MCL_ID ID_TYPE;
declare variable ACL_ID ID_TYPE;
declare variable ACL_STATUS integer;
declare variable PK_ID integer;
declare variable CITIZEN_ID integer;
declare variable C_NAME_FIO D_STR_255;
declare variable C_NAME_ENG D_STR_255;
declare variable C_NAME_ALT D_STR_255;
declare variable C_BIRTH_DATE date;
declare variable C_BIRTH_COUNTRY D_STR_255;
declare variable C_BIRTH_CITY D_STR_255;
declare variable C_SEX integer;
declare variable C_INN D_STR_CODE;
declare variable CDS_ID D_ID;
declare variable DB_ID integer;
declare variable BLS_1 D_ID;
declare variable BLS_2 D_ID;
declare variable SURNAME D_STR_100;
declare variable NAME D_STR_100;
declare variable PATRONYMIC D_STR_100;
declare variable SURNAME_ENG D_STR_100;
declare variable NAME_ENG D_STR_100;
declare variable PATRONYMIC_ENG D_STR_100;
declare variable RD_ID D_INTEGER;
begin
  is_critical = null;
  res_text = '';
  c_id = 0;
  db_id = f_get_db_id();
  uxf$set_user(un_id);

  -- если клиент успешно создан, с_id <> 0
  -- если найдены некритичные дубликаты, то c_id = 0, is_critical = 0
  -- если найдены критичные дубликаты, то c_id = 0, is_critical = 1
  -- если ошибка при создании, c_id = 0, res_text <> ''

  select drl_from_c_id, drl_from_mcl_id, drl_from_pk_id, drl_from_bls_1, drl_from_bls_2
  from docrecoglist
  where drl_id = :drl_id
  into :tmp_c_id, mcl_id, pk_id, bls_1, bls_2;

  if (tmp_c_id is not null) then
    res_text = iif(db_id <> 94401, 'По данному документу клиент уже создан', 'Bu sənəd üzrə müştəri artıq əlavə olunub');

  if (res_text = '' and
      pk_id is null) then
    res_text = iif(db_id <> 94401, 'Тип документа не определен', 'Sənədin növü müəyyən edilmədi');

  if (res_text = '' and
      mcl_id <> 0) then
  begin
    select acl_id, acl_from_rd_id, acl_status from actionlist
    where acl_object_kind = f_objkind('DocRecogList', 'id') and
      acl_object_id = :drl_id
    into acl_id, rd_id, acl_status;

    if (acl_id is null) then
    begin
      res_text = iif(db_id <> 94401, 'Документ находится в процессе обработки', 'Sənəd emal prossesindədir');
    end
    else
    begin
      select acl_status
      from actionlist
      where acl_id = :acl_id
      into acl_status;

      if (acl_status <> 1) then
      begin
        if (db_id <> 94401) then
          res_text = case acl_status
                       when -3 then 'Запрос отменен пользователем'
                       when -2 then 'Отказ от комитета в создании клиента'
                       when -1 then 'Документ отклонен комитетом, отправлен на доработку'
                       when 0 then 'Документ в процессе проверки'
--                       when 1 then 'Документ в процессе проверки'
                       when 2 then 'Документ уже подтвержден'
                     end;
        else
          res_text = case acl_status
                       when -3 then 'Sorğu istifadəçi tərəfindən imtina olunub'
                       when -2 then 'Müştərinin yaradılması Komitə tərəfindən imtina olunub'
                       when -1 then 'Sənəd Komitə tərəfindən imtina olunub, tamamlanmaya göndərilib'
                       when 0 then 'Sənəd yoxlama mərhələsindədir'
--                       when 1 then 'Sənəd yoxlama mərhələsindədir'
                       when 2 then 'Документ уже подтвержден'
                     end;
        res_text = trim(res_text);
      end
    end
  end

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  select pk_country_id
  from passkind
  where pk_id = :pk_id
  into :citizen_id;

  if (ignore_duplicate <> 2) then
  begin
    select is_critical
    from fm_find_dbl_client('', null, null, null, null, :un_id, :drl_id)
    where (is_critical = 1 or
          (is_critical = 0 and :ignore_duplicate = 0))
    order by is_critical desc
    rows 1
    into :is_critical;
  end

  if (db_id = 81001) then  -- RFCRU-5800
    rdb$set_context('USER_TRANSACTION', 'tmp_mcl_id', :mcl_id);

  if (is_critical is null) then
  begin
    c_name_fio = f_docrecog_value(:drl_id, 'calc_name_full', '');
    c_name_eng = f_docrecog_value(:drl_id, 'calc_name_full_eng', '');
    c_name_alt = f_docrecog_value(:drl_id, 'calc_name_full_alt', '');
    c_birth_date = f_docrecog_value(:drl_id, 'birth_date', null);
    c_birth_city = f_docrecog_value(:drl_id, 'birth_place', '');
    c_birth_country = f_docrecog_value(:drl_id, 'birth_country', '');
    c_sex = f_docrecog_value(:drl_id, 'calc_gender', null);
    c_inn = f_docrecog_value(:drl_id, 'inn', '');
    surname = f_docrecog_value(:drl_id, 'surname', '');
    name = f_docrecog_value(:drl_id, 'name', '');
    patronymic = f_docrecog_value(:drl_id, 'patronymic', '');
    surname_eng = f_docrecog_value(:drl_id, 'name_eng', '');
    name_eng = f_docrecog_value(:drl_id, 'surname_eng', '');
    patronymic_eng = f_docrecog_value(:drl_id, 'patronymic_eng', '');
    if (res_text = '' and
        c_name_fio = '') then
      res_text = iif(db_id <> 94401, 'ФИО не заполнено', 'SAA qeyd olunmayıblar');
    if (res_text = '' and
        c_birth_date is null) then
      res_text = iif(db_id <> 94401, 'Дата рождения не заполнена', 'Doğum tarixi qeyd olunmayıb');
    if (res_text = '' and
        c_sex is null) then
      res_text = iif(db_id <> 94401, 'Пол не указан', 'Cinsi qeyd olunmayıb');

    -- RFCRU-5644
    if (res_text = '' and
      db_id = 81001) then
    res_text = (
      select
        err_txt
      from hp_separate_fio_checkvalid(0, :name, :patronymic, :surname, :name_eng, :patronymic_eng, :surname_eng, null));

    if (res_text <> '') then
    begin
      suspend;
      exit;
    end

    -- создаем клиента
    select c_id, err_text
    from fm_add_client(:c_name_fio, :c_sex, :c_birth_date, :c_birth_country, :c_birth_city, 0,
    :citizen_id, :pk_id, '', null, :c_inn, :un_id, :surname, :name, :patronymic, :surname_eng, :name_eng, :patronymic_eng)
    into c_id, res_text;

    if (coalesce(c_id, 0) <> 0) then
    begin
      update clients
      set c_name_eng = :c_name_eng,
          c_alternat_fio = :c_name_alt,
          c_blob_pass1_id = :bls_1,
          c_blob_pass2_id = :bls_2
      where c_id = :c_id;

      update docrecoglist
      set drl_from_c_id = :c_id,
          drl_client_created = 1
      where drl_id = :drl_id;

      -- Здесь, по хорошему, нужно вернуть и текст ошибки,
      -- Но если вернем, то пользователь опять попытается вызвать эту процедуру,
      -- и создаст дубликат клиента, которого создали чуть выше.

      select cds_id
      from fm_docrecog_add_clientdoc(:drl_id, :un_id)
      into :cds_id;

      -- RFCRU-5246
      if (db_id = 81001 and rd_id = f_get_rd_id('ActionGroup.DocRecog_Client_OiVO')) then
        update actionlist set
          acl_from_rd_id = f_get_rd_id('ActionGroup.Passport_Verify_OiVO')
        where acl_object_kind = 0
          and acl_object_id = :c_id
          and acl_from_rd_id = f_get_rd_id('ActionGroup.Passport_Verify');
    end
  end
  suspend;
end
