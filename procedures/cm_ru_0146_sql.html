CREATE PROCEDURE CM_RU_0146_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable reg_id integer;
declare variable pr_code str_100;
declare variable reg_name  str_100;
declare variable pr_name str_100;
begin
  res_error_is_critical = 1;
  for select reg.reg_unicode, pr.pr_code, reg.reg_name, pr.pr_name
  from kontract kont
  inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
  inner join region reg on  reg.reg_unicode = kal.kal_new_reg_id
  inner join product pr on pr.pr_id = kont.k_from_pr_id
  where kont.k_id = :k_id
  and kont.k_disb_date between kal.kal_date and kal.kal_date_end
  and kal.is_delete = 0
  and reg.reg_logic_label like '%.mini_office.%'
  into :reg_id, :pr_code, :reg_name, :pr_name
  do
  begin
   /*
    1120300 -- Благородный
    1120400 -- Радушный
    1130200 -- 13-02 г. Омск
    1280300 -- Ак Барс
    1380200 -- Салават (Уфа)
    1400200 -- Ленинский (Воронеж)
    1460100 -- Владимир
    1490200 -- Советский (Самара)
    1490300 -- Тольятти
    1500100 -- Ульяновск
    1510100 -- Рязань
    1520100 -- Кострома
   */
    if (reg_id in (1120300, -- Благородный
                   1120400, -- Радушный
                   1130200, -- 13-02 г. Омск
                   1280300, -- Ак Барс
                   1380200, -- Салават (Уфа)
                   1400200, -- Ленинский (Воронеж)
                   1460100, -- Владимир
                   1490200, -- Советский (Самара)
                   1490300  -- Тольятти
    )) then
    begin
        if (pr_code not like '%.shans.%') then
        begin
            res_error_text = 'Для офиса '||:reg_name||', выдачи по продукту '||:pr_name||' - запрещены';
            suspend;    
        end
    end
    if (reg_id in (1490400 -- Содружество (Самара)
    )) then
    begin
        if (pr_code not like '%.status.%') then
        begin
            res_error_text = 'Для офиса '||:reg_name||', выдачи по продукту '||:pr_name||' - запрещены';
            suspend;    
        end
    end
    if (reg_id in (1500100 -- Ульяновск
    )) then
    begin
        if ((pr_code not like '%.status.%') and
            (pr_code not like '%.shans.%')) then
        begin
            res_error_text = 'Для офиса '||:reg_name||', выдачи по продукту '||:pr_name||' - запрещены';
            suspend;    
        end
    end
    if (reg_id in (1510100 -- Рязань
    )) then
    begin
        if ((pr_code not like '%.status.%') and
            (pr_code not like '%.shans.%') and
            (pr_code not like '%.express.%')) then
        begin
            res_error_text = 'Для офиса '||:reg_name||', выдачи по продукту '||:pr_name||' - запрещены';
            suspend;    
        end
    end
    if (reg_id in (1520100 -- Кострома
    )) then
    begin
        if ((pr_code not like '%.shans.%') and
            (pr_code not like '%.express.%')) then
        begin
            res_error_text = 'Для офиса '||:reg_name||', выдачи по продукту '||:pr_name||' - запрещены';
            suspend;    
        end
    end
  end
end
