CREATE PROCEDURE HP_TRANSFERTMC_VIEW(
    JS_PARAM D_STR_255)
RETURNS (
    TRT_ID D_INTEGER,
    TRT_SAVE_OL_ID D_ID,
    TRT_SIGN_OL_ID D_ID,
    TRT_NUMBER D_INTEGER,
    TRT_DATE D_DATE,
    TRT_SIGN_DATE D_DATE,
    TRT_OFFICES_ID D_INTEGER,
    OFFICES_NAME D_STR_CODE,
    TRT_OFFICER_ID D_INTEGER,
    OFFICER_NAME D_STR_CODE,
    TRT_MOLS_ID D_ID,
    MOLS_NAME D_STR_100,
    TRT_MOLR_ID D_ID,
    MOLR_NAME D_STR_100,
    TRT_MOLS_SIGN_DATE D_DATETIME,
    TRT_MOLR_SIGN_DATE D_DATETIME,
    TRT_MOLS_EDOC_ID D_ID,
    TRT_MOLR_EDOC_ID D_ID,
    TRT_WPR_ID D_ID,
    WPR_NAME D_STR_100,
    TRT_STATUS D_SMALLINT,
    STATUS D_STR_CODE,
    IS_DELETE D_SMALLINT)
AS
declare variable id d_integer;
declare variable c_id d_id;
declare variable use_sign d_bool;
declare variable only_self d_bool;
declare variable reg_id d_integer;
declare variable dt_beg d_date;
declare variable dt_end d_date;
declare variable reg_lst d_str_8100;
begin
  id = f_json_str_get_value(js_param, 'trt_id', 0);
  c_id = f_json_str_get_value(js_param, 'c_id', null);
  use_sign = f_json_str_get_value(js_param, 'use_sign', 0);
  only_self = f_json_str_get_value(js_param, 'only_self', 1);
  reg_id = f_json_str_get_value(js_param, 'reg_id', null);
  dt_beg = f_json_str_get_value(js_param, 'dt_beg', null);
  dt_end = f_json_str_get_value(js_param, 'dt_end', null);

  if (id = 0 and use_sign = 0) then
    reg_lst = coalesce((
      select list(id_list, ';')
      from region_list(:reg_id,1,3,1)), '');

  for
    select
      trt_id,
      trt_save_ol_id,
      trt_sign_ol_id,
      trt_number,
      trt_date,
      trt_sign_date,
      trt_offices_id,
      trt_officer_id,
      trt_mols_id,
      trt_molr_id,
      trt_mols_sign_date,
      trt_molr_sign_date,
      trt_mols_edoc_id,
      trt_molr_edoc_id,
      trt_wpr_id,
      trt_status,
      coalesce(is_delete, 0)
    from transfertmc
      left join operlist
        on ol_id = trt_save_ol_id
    where :id > 0
      and trt_id = :id
    union
    select
      trt_id,
      trt_save_ol_id,
      trt_sign_ol_id,
      trt_number,
      trt_date,
      trt_sign_date,
      trt_offices_id,
      trt_officer_id,
      trt_mols_id,
      trt_molr_id,
      trt_mols_sign_date,
      trt_molr_sign_date,
      trt_mols_edoc_id,
      trt_molr_edoc_id,
      trt_wpr_id,
      trt_status,
      coalesce(is_delete, 0)
    from transfertmc
      left join operlist
        on ol_id = trt_save_ol_id
    where :id = 0 and :use_sign = 1 and coalesce(is_delete, 0) = 0
      and ((trt_mols_id = :c_id and trt_status = 0) or
           (trt_molr_id = :c_id and trt_status = 1))
    union
    select
      trt_id,
      trt_save_ol_id,
      trt_sign_ol_id,
      trt_number,
      trt_date,
      trt_sign_date,
      trt_offices_id,
      trt_officer_id,
      trt_mols_id,
      trt_molr_id,
      trt_mols_sign_date,
      trt_molr_sign_date,
      trt_mols_edoc_id,
      trt_molr_edoc_id,
      trt_wpr_id,
      trt_status,
      coalesce(is_delete, 0)
    from transfertmc
      left join operlist
        on ol_id = trt_save_ol_id
    where :id = 0 and :use_sign = 0
      and (trt_date between :dt_beg and :dt_end)
      and ((:only_self = 0) or ((:only_self = 1) and (trt_mols_id = :c_id or trt_molr_id = :c_id)))
  into
    :trt_id,
    :trt_save_ol_id,
    :trt_sign_ol_id,
    :trt_number,
    :trt_date,
    :trt_sign_date,
    :trt_offices_id,
    :trt_officer_id,
    :trt_mols_id,
    :trt_molr_id,
    :trt_mols_sign_date,
    :trt_molr_sign_date,
    :trt_mols_edoc_id,
    :trt_molr_edoc_id,
    :trt_wpr_id,
    :trt_status,
    :is_delete
  do
  begin
    if (is_delete = 0 and trt_status = -1) then
      is_delete = 1;

    if ((id = 0) and (use_sign = 0)
      and not (reg_lst containing trt_offices_id)
      and not (reg_lst containing trt_officer_id))
    then continue;

    offices_name = (select reg_name from region where reg_unicode = :trt_offices_id);
    officer_name = (select reg_name from region where reg_unicode = :trt_officer_id);
    mols_name = (select c_name_fio from clients where c_id = :trt_mols_id);
    molr_name = (select c_name_fio from clients where c_id = :trt_molr_id);
    wpr_name = (select wp_name from workplace where wp_id = :trt_wpr_id);
    status =
      case
        when is_delete = 1  then 'Аннулирован'
        when trt_status = 0 then 'Создан'
        when trt_status = 1 then 'Передан'
        when trt_status = 2 then 'Принят'
      end;
    suspend;
  end
end
