CREATE PROCEDURE FM_HTML_REFUND_KONTRACT(
    MA_ID D_INTEGER,
    OBJ_KIND D_STR_100,
    DATE_END D_DATE,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable enabled_to_refund_table d_blob_text;
declare variable refund_table d_blob_text;
declare variable thead d_str_long;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable pr_name d_str_code;
declare variable zad_os d_currency;
declare variable zad_perc d_currency;
declare variable zad_total d_currency;
declare variable zad_str d_str_255;
declare variable res_link d_str_long;
declare variable row_counter d_integer;
declare variable row_counter_internal d_integer;
declare variable row_counter_external d_integer;
declare variable bki_status_id d_smallint;
declare variable bki_status_text d_str_255;
declare variable ma_date_app d_date;
declare variable ma_date_oper d_date;
declare variable uuid d_str_code;
declare variable cred_partner_type_str d_str_code;
declare variable cred_debt_sum d_currency;
declare variable ssp_summ d_currency;
declare variable ssp_total d_currency;
declare variable params d_str_long;
declare variable is_error d_smallint;
declare variable rez_text d_str_100;
declare variable kontract_name d_str_100;
declare variable cred_sum_overdue d_currency;
declare variable cred_sum d_currency;
begin
  -- RFCRU-3486 Изменение порядка рефинансирования займов
  -- процедура отвечает за отображение таблицы всех активных займов клиента, подпадающих под рефинансирование на новый продукт
  -- вызывается через f_uri_html() см FM_REPORT_HTML_MAIN

  thead = '<table width="100%">
                  <col width="30%">
                  <col width="35%">
                  <col width="30%">
                  <col width="5%">
                  <thead>
                    <tr>
                      <th scope="col">Договор</th>
                      <th scope="col">Продукт</th>
                      <th scope="col">Остаток Долга</th>
                      <th scope="col">Действие</th>
                    </tr>
                  </thead>';

  select mac.mac_from_c_id
  from miniappclient mac
  where mac.mac_from_ma_id = :ma_id
    and mac.mac_amount > 0
  into :c_id;

  select ma.ma_date_app, ma.ma_date_oper
  from miniapp ma
  where ma.ma_id = :ma_id
  rows 1
  into ma_date_app, ma_date_oper;

  res_html = (select res_style from fm_html_styles) || '<body>'; -- добавление стилей
  res_html = blob_append(res_html, '<style>.nowrap{text-wrap: nowrap;}</style>');

  res_html = blob_append(res_html, '<h4>Укажите займы, которые будут рефинансированы</h4>');

  -- Таблица ВОЗМОЖНО РЕФИНАНСИРОВАТЬ
  enabled_to_refund_table = '<p>Возможно рефинансировать:</p>';
  enabled_to_refund_table = blob_append(enabled_to_refund_table, thead);
  row_counter_internal = 0;
  for select
          k.k_id,
          k.k_kontract_no,
          k.k_disb_date,
          pr.pr_name,
          f_uuid_cbr(f_uuid('knclient', kc_id))
  from kontstate ks
  inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
  inner join kontract k on k.k_id = ks.ks_from_k_id
  inner join product pr on pr.pr_id = k.k_from_pr_id
  where ks.ks_ol_is_delete = 0
    and ks.ks_state = 1
    and :date_end between ks.ks_date_begin and ks.ks_date_end
    and kc.kc_from_c_id = :c_id
    and kc.kc_amount_applicate <> 0
  into k_id,
       k_kontract_no,
       k_disb_date,
       pr_name,
       uuid
  do
  begin
    if (exists (select * from kontrefinance kr where kr.kr_from_ma_id = :ma_id and kr.kr_from_k_id = :k_id)) then
      continue;

    if (exists (select *
                from kontrefinance kr
                inner join kontract k on k.k_from_ma_app_id = kr.kr_from_ma_id
                inner join kontstate ks on ks.ks_from_k_id = k.k_id
                where kr.kr_loan_uuid = :uuid
                and ks.ks_ol_is_delete = 0
                and ks.ks_state = 1)) then
    continue;   --RFCRU-4492

    -- основная сумма и проценты
    select sum(zad_os), sum(sh_calc_perc),
      sum(iif(coalesce(sh_date, cast('01.01.1900' as date)) < :date_end, zad_os + sh_calc_perc, 0))
    from aarep_make_zad_list(:k_id, :date_end, 1)
    into zad_os, zad_perc, cred_sum_overdue;

    -- среднемесячная сумма платежа (RFCRU-4021)
    ssp_summ = 0;
    select calc_month_repay_final
    from hp_refund_kontract_ssp(:k_id, :date_end)
    into ssp_summ;

    if (ssp_summ is null) then
      ssp_summ = 0;

    zad_os = round_currency_up(zad_os, 'int');
    zad_perc = round_currency_up(zad_perc, 'int');
    ssp_summ = round_currency_up(ssp_summ, 'int');
    cred_sum_overdue = round_currency_up(cred_sum_overdue, 'int');

    cred_debt_sum = zad_os + zad_perc;

    zad_str = f_format('<div class="nowrap" style="color:%4">%1 ОС<br>%2 %%<br>%3 СП</div>',
          f_numtostr(zad_os, 0),
          f_numtostr(zad_perc, 0),
          f_numtostr(ssp_summ, 0),
          iif(cred_sum_overdue > 0, 'red', 'black'));

    params = f_format('%1|%2|%3|', ma_id, ma_date_app, 'insert');
    params = params || f_format('%1|%2|%3|%4|%5|', uuid, cred_debt_sum, zad_perc, ssp_summ, '-');  -- RFCRU-4456 То что показано, то и должно быть записано.
    params = params || f_format('%1|', cred_sum_overdue);  -- RFCRU-4646

    kontract_name = f_format('<div class="nowrap">%1</div> от %2', k_kontract_no, f_datetostr(k_disb_date));
    res_link = f_uri_sql('Добавить', 'refund_kontract', 'kontract', k_id, 'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', params);
    enabled_to_refund_table = blob_append(:enabled_to_refund_table, f_html_table_row(kontract_name, pr_name, zad_str, res_link, 0, 0, 1, 0, 4, ''));
    row_counter_internal = row_counter_internal + 1;
  end

  select status_id, status_text
  from hp_bki_get_status(:c_id)
  into bki_status_id, bki_status_text;

  if (coalesce(bki_status_id, 0) <> 2) then
  begin
    res_html = bki_status_text; -- нет запроса в БКИ или другая причина отсутствия БКИ (RFCRU-4132)
    suspend;
    exit;
  end

  select is_error, rez_text
  from hp_get_info_extinfocfg(:c_id, current_date, 'bki_ssp_40')
  into :is_error, rez_text;

  if (is_error = 1) then
  begin
    res_html = :rez_text;
    suspend;
    exit;
  end

  bki_status_text = '';
  row_counter_external = 0;

  if (
    (current_date between '10.06.2026' and '30.06.2026') or
    (current_date between '06.08.2026' and '31.08.2026')
    ) then  -- ограничения по показу внешних займов для рефинансирования указывать тут15
  begin
    -- внешние займы клиента (RFCRU-4021)
    for select uuid, cred_partner_type_str, cred_debt_sum, ssp_summ, error_text, cred_sum_overdue, cred_date, cred_sum
    from hp_bki_get_list_for_refinace(:c_id, null, 0, 1)
    into uuid, cred_partner_type_str, cred_debt_sum, ssp_summ, bki_status_text, cred_sum_overdue, k_disb_date, cred_sum
    do
    begin
      if (exists (select * from kontrefinance where kr_from_ma_id = :ma_id and kr_loan_uuid = :uuid)) then
        continue;
  
      if (exists (select *
                  from kontrefinance kr
                  inner join kontract k on k.k_from_ma_app_id = kr.kr_from_ma_id
                  inner join kontstate ks on ks.ks_from_k_id = k.k_id
                  where kr.kr_loan_uuid = :uuid
                  and ks.ks_ol_is_delete = 0
                  and ks.ks_state = 1)) then
      continue;  --RFCRU-4492
  
      if (coalesce(bki_status_text, '') <> '') then
      begin
        res_html = bki_status_text;
        suspend;
        exit;
      end
  
      cred_debt_sum = round_currency_up(cred_debt_sum, 'int');
      ssp_summ = round_currency_up(ssp_summ, 'int');
  
      params = f_format('%1|%2|%3|', ma_id, ma_date_oper, 'insert');
      params = params || f_format('%1|%2|%3|%4|%5|', uuid, cred_debt_sum, 0, ssp_summ, cred_partner_type_str);
      params = params || f_format('%1|%2|%3|',cred_sum_overdue, cred_sum, k_disb_date);  -- RFCRU-4646
  
      kontract_name = f_format('<div class="nowrap">%1</div> от %2 ****%3', f_numtostr(cred_sum, 0), f_datetostr(k_disb_date), right(uuid, 6));
      res_link = f_uri_sql('Добавить', 'refund_kontract', 'kontract', 0, 'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', :params);
      zad_str = f_format('<div class="nowrap" style="color:%3">%1 ВД<br>%2 СП</div>',
          f_numtostr(round_currency_up(cred_debt_sum, 'int'), 0),
          f_numtostr(round_currency_up(ssp_summ, 'int'), 0),
          iif(cred_sum_overdue > 0, 'red', 'black'));
  
      enabled_to_refund_table = blob_append(enabled_to_refund_table, f_html_table_row(kontract_name, cred_partner_type_str, zad_str, res_link, 0, 0, 1, 0, 4));
      row_counter_external = row_counter_external + 1;
    end
  end

  enabled_to_refund_table = blob_append(:enabled_to_refund_table, '</table>');

  if (row_counter = 0) then
    enabled_to_refund_table = '';

  -- Таблица БУДУТ РЕФИНАНСИРОВАНЫ
  refund_table = '<p>Будут рефинансированы:</p>';
  refund_table = blob_append(refund_table, thead);
  row_counter = 0;
  zad_total = 0;
  ssp_total = 0;

  for select kr_loan_uuid, kr_balance, kr_perc_saldo, kr_avg_repay, kr_mfo_type, kr_from_k_id, kr_debt_overdue, kr_cred_sum, kr_cred_date
  from kontrefinance
  where kr_from_ma_id = :ma_id
  order by kr_id
  into uuid, cred_debt_sum, zad_perc, ssp_summ, cred_partner_type_str, k_id, cred_sum_overdue, cred_sum, k_disb_date
  do
  begin
    if (k_id is not null) then
    begin
      select k_kontract_no, k_disb_date, pr_name
      from kontract, product
      where pr_id = k_from_pr_id
        and k_id = :k_id
      into k_kontract_no, k_disb_date, cred_partner_type_str;

      kontract_name = f_format('<div class="nowrap">%1</div> от %2', k_kontract_no, f_datetostr(k_disb_date));
      zad_str = f_format('<div class="nowrap" style="color:%4">%1 ОС<br>%2 %%<br>%3 СП</div>',
          f_numtostr(cred_debt_sum - zad_perc, 0),
          f_numtostr(zad_perc, 0),
          f_numtostr(ssp_summ, 0),
          iif(cred_sum_overdue > 0, 'red', 'black'));
    end
    else
    begin
      zad_str = f_format('<div class="nowrap" style="color:%3">%1 ВД<br>%2 СП</div>',
          f_numtostr(cred_debt_sum, 0),
          f_numtostr(ssp_summ, 0),
          iif(cred_sum_overdue > 0, 'red', 'black'));

      kontract_name = f_format('<div class="nowrap">%1</div> от %2 ****%3', f_numtostr(cred_sum, 0), f_datetostr(k_disb_date), right(uuid, 6));
    end

    params = f_format('%1|%2|%3|%4|', ma_id, ma_date_oper, 'delete', uuid);
    res_link = f_uri_sql('Удалить', 'refund_kontract', 'kontract', 0, 'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', params);
    refund_table = blob_append(refund_table, f_html_table_row(kontract_name, cred_partner_type_str, zad_str, res_link, 0, 0, 1, 0, 4));

    zad_total = zad_total + cred_debt_sum;
    ssp_total = ssp_total + ssp_summ;
    row_counter = row_counter + 1;
  end

  zad_str = '<b>' || f_format('<div class="nowrap">%1 ВД</div>%2 СП', f_numtostr(zad_total, 0), f_numtostr(ssp_total, 0)) || '</b>';
  refund_table = blob_append(refund_table, f_html_table_row('<b>Итого</b>', '<b>' || row_counter || '</b>', zad_str, '', 0, 0, 1, 0, 4));
  refund_table = blob_append(:refund_table, '</table>');

  if (row_counter = 0) then
    refund_table = '';

  if (row_counter = 0 and row_counter_internal = 0 and row_counter_external = 0) then
    res_html = 'Нет займов, подходящих под рефинансирование'; -- RFCRU-4132
  else
    res_html = blob_append(res_html, coalesce(enabled_to_refund_table || refund_table, '') || '</body>');

  suspend;
end
