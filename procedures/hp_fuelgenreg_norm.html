CREATE PROCEDURE HP_FUELGENREG_NORM(
    PET_ID D_INTEGER,
    DT_OPER D_DATE)
RETURNS (
    NORM_LST D_STR_255)
AS
declare variable norm d_integer;
declare variable ped_id d_integer;
begin
  norm_lst = '';

  norm = (
    select pet_norma_km
    from pettrans
    where pet_id = :pet_id);

  ped_id = (
    select first 1 ped_id
    from petdata
    where ped_from_pet_id = :pet_id
      and ped_date <= :dt_oper
      order by ped_date desc);
  for
    with tmp as (
      select * from petdata
      where ped_from_pet_id = :pet_id
        and ped_id = :ped_id)
    select
      list(a.val / :norm, ascii_char(13) || ascii_char(10))
    from (
      select ped_expense_summer as val from tmp
      union
      select ped_expense_winter as val from tmp
      union
      select ped_expense_summer_mount as val from tmp
      union
      select ped_expense_winter_mount as val from tmp ) as a
  into
    :norm_lst
  do
    norm_lst = replace(norm_lst, '.', ',');

  suspend;
end
