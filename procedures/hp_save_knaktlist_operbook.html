CREATE PROCEDURE HP_SAVE_KNAKTLIST_OPERBOOK(
    KA_ID D_ID)
AS
declare variable main_ol_id d_id;
declare variable ss_memo d_str_long;
declare variable un_id d_id;
declare variable ka_date d_date;
declare variable ol_id d_id;
declare variable pat_id d_integer;
declare variable main_acc d_str_code;
declare variable acc_no d_integer;
declare variable acc_1 d_str_code;
declare variable acc_2 d_str_code;
declare variable reg_code d_str_code;
declare variable cm_default_currency d_str_code;
declare variable new_reg_code d_str_code;
declare variable res_reg_id d_integer;
declare variable res_acc_dt d_str_code;
declare variable res_acc_kt d_str_code;
declare variable reg_id d_integer;
declare variable kf_id d_id;
declare variable k_id d_id;
declare variable oks_balance d_currency;
begin
  select cm_default_currency
  from company
  into cm_default_currency;

  select distinct kal_prior_reg_id, reg_code
  from knaktlist, region
  where kal_from_ka_id = :ka_id and
        kal_prior_reg_id = reg_unicode
  rows 1
  into reg_id, reg_code;

  select ka_from_ol_id, ka_date, ka_from_un_id
  from knakt
  where ka_id = :ka_id
  into main_ol_id, ka_date, un_id;

  if (main_ol_id is not null) then
  begin
    if (exists(select * from operlist where ol_id = :main_ol_id and is_delete = 0)) then
      exception e_message 'Бух.проводки уже построены';
    else
      main_ol_id = null;
  end

  ss_memo = 'Прием-передача кредитной папки между офисами';

  insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                        ol_status, is_delete, ol_user_id, ol_document_client, ol_from_mo_id,
                        ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                        ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month)
  values (:reg_id, 2, :ka_date, null, '', 9, :ss_memo,
          0, 0, :un_id, null, null,
          0, null, null, null, 22, :ka_date, null, null, 0, null, null)
  returning ol_id
  into main_ol_id;

  for select kal_from_kf_id, reg_code
  from knaktlist, region
  where kal_from_ka_id = :ka_id and
        kal_new_reg_id = reg_unicode
  into kf_id, new_reg_code
  do
  begin
    for select k_id, pr_from_pat_id
    from kontract, product
    where k_from_kf_id = :kf_id and
          k_from_pr_id = pr_id
    into k_id, pat_id
    do
    begin
      oks_balance = 0;
      select oks_balance
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :ka_date and
            oks_date_end >= :ka_date and
            oks_close_state = 0
      into oks_balance;

      if (oks_balance > 0) then
      begin
        main_acc = '';
        acc_no = 0;
        select pal_main_acc, pal_main_acc_no
        from prodacclist
        where pal_from_pat_id = :pat_id and
              pal_from_at_id = 1 -- кредитный портфель
        into main_acc, acc_no;

        if (coalesce(main_acc, '') = '' or coalesce(acc_no, 0) = 0) then
          exception e_default 'Лиц.счет портфеля не указан';

        acc_1 = reg_code || cm_default_currency || main_acc || lpad(acc_no, 6, '0');
        acc_2 = new_reg_code || cm_default_currency || main_acc || lpad(acc_no, 6, '0');

        for select res_reg_id, res_acc_dt, res_acc_kt
        from aarep_loronostro_link(:reg_id, :acc_1, :acc_2)
        into :res_reg_id, :res_acc_dt, :res_acc_kt
        do
        begin
          ol_id = null;

          if (res_reg_id = reg_id) then
            ol_id = main_ol_id;
          else
          begin
            select ol_id
            from operlist
            where ol_parent_id = :main_ol_id and
                  ol_from_reg_id = :res_reg_id
            into ol_id;

            if (ol_id is null) then
            begin
              insert into operlist (ol_parent_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                                    ol_status, is_delete, ol_user_id, ol_document_client, ol_from_mo_id,
                                    ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                                    ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month)
              values (:main_ol_id, :res_reg_id, 2, :ka_date, null, '', 9, :ss_memo,
                      0, 0, :un_id, null, null,
                      0, null, null, null, 22, :ka_date, null, null, 0, null, null)
              returning ol_id
              into ol_id;
            end
          end

          insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_summa_val,
                                ob_summa, ob_account_kt, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
          values (:ol_id, :cm_default_currency, 1, :res_acc_dt, :oks_balance,
                  :oks_balance, :res_acc_kt, 'Прием передача КП', :ka_date, 0, :res_reg_id);
        end
      end
    end
  end

  update knakt
  set ka_from_ol_id = :main_ol_id
  where ka_id = :ka_id;
end
