CREATE PROCEDURE HP_GET_RVPZ(
    REP_AMOUNT_CLI CURR_TYPE,
    REP_PERIOD_DAY INTEGER,
    REP_IS_SECURED SMALLINT,
    REP_IS_RESTRUCT SMALLINT,
    REP_DELINQ_DAY INTEGER)
RETURNS (
    GROUP_NUMBER INTEGER,
    PERCENT_RVPZ PERC_TYPE)
AS
begin

  if (rep_amount_cli <= 30000.0 and
      rep_period_day <= 30) then
  begin
    if (rep_is_restruct = 0) then
    begin
      group_number = 1;
    end
    else
    begin
      group_number = 2;
    end
  end
  else
  begin
    if (rep_is_restruct = 0) then
    begin
      if (rep_is_secured = 1) then
      begin
        group_number = 3;
      end
      else
      begin
        group_number = 4;
      end
    end
    else
    begin
      if (rep_is_secured = 1) then
      begin
        group_number = 5;
      end
      else
      begin
        group_number = 6;
      end
    end
  end

  percent_rvpz = case group_number
                   when 1 then case
                                 when rep_delinq_day <= 0 then 0
                                 when rep_delinq_day <= 30 then 50
                                 when rep_delinq_day <= 60 then 80
                                 when rep_delinq_day <= 90 then 90
                                 when rep_delinq_day <= 120 then 100
                                 when rep_delinq_day <= 180 then 100
                                 when rep_delinq_day <= 270 then 100
                                 when rep_delinq_day <= 360 then 100
                                 else 100
                               end
                   when 2 then case
                                 when rep_delinq_day <= 0 then 0
                                 when rep_delinq_day <= 30 then 50
                                 when rep_delinq_day <= 60 then 80
                                 when rep_delinq_day <= 90 then 90
                                 when rep_delinq_day <= 120 then 100
                                 when rep_delinq_day <= 180 then 100
                                 when rep_delinq_day <= 270 then 100
                                 when rep_delinq_day <= 360 then 100
                                 else 100
                               end
                   when 3 then case
                                 when rep_delinq_day <= 0 then 0
                                 when rep_delinq_day <= 30 then 1.5
                                 when rep_delinq_day <= 60 then 7
                                 when rep_delinq_day <= 90 then 15
                                 when rep_delinq_day <= 120 then 30
                                 when rep_delinq_day <= 180 then 45
                                 when rep_delinq_day <= 270 then 60
                                 when rep_delinq_day <= 360 then 75
                                 else 100
                               end
                   when 4 then case
                                 when rep_delinq_day <= 0 then 0
                                 when rep_delinq_day <= 30 then 3
                                 when rep_delinq_day <= 60 then 10
                                 when rep_delinq_day <= 90 then 20
                                 when rep_delinq_day <= 120 then 40
                                 when rep_delinq_day <= 180 then 50
                                 when rep_delinq_day <= 270 then 65
                                 when rep_delinq_day <= 360 then 80
                                 else 100
                               end
                   when 5 then case
                                 when rep_delinq_day <= 0 then 2
                                 when rep_delinq_day <= 30 then 5
                                 when rep_delinq_day <= 60 then 20
                                 when rep_delinq_day <= 90 then 40
                                 when rep_delinq_day <= 120 then 60
                                 when rep_delinq_day <= 180 then 80
                                 when rep_delinq_day <= 270 then 95
                                 when rep_delinq_day <= 360 then 100
                                 else 100
                               end
                   when 6 then case
                                 when rep_delinq_day <= 0 then 5
                                 when rep_delinq_day <= 30 then 10
                                 when rep_delinq_day <= 60 then 30
                                 when rep_delinq_day <= 90 then 50
                                 when rep_delinq_day <= 120 then 70
                                 when rep_delinq_day <= 180 then 90
                                 when rep_delinq_day <= 270 then 100
                                 when rep_delinq_day <= 360 then 100
                                 else 100
                               end
                 end;

  suspend;
end
