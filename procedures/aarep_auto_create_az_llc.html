CREATE PROCEDURE AAREP_AUTO_CREATE_AZ_LLC(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
AS
declare variable aa_llc_percent d_percent;
declare variable aa_llc_id d_integer;
declare variable lm_no d_integer;
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate d_currency;
declare variable cr_rate d_currency;
declare variable cr_curr_unit d_integer;
declare variable k_disb_date d_date;
declare variable k_percent d_percent;
declare variable lir_rate d_percent;
declare variable k_by_rating d_smallint;
declare variable is_secured boolean;
declare variable is_business boolean;

declare variable reg_id d_integer;
declare variable reg_code d_str_20;
declare variable reg_name d_str_100;
declare variable lm_id d_id;
declare variable aa_k_id d_id;
declare variable aa_amount_active_end d_currency;
declare variable aa_deliq_days d_integer;
declare variable aa_pr_is_indiv d_integer;
declare variable sh_calc_perc d_currency;
declare variable rep_period d_integer;
begin
  rep_period = extract(year from cast(:rep_date_end as date)) || right('0' ||
    extract(month from cast(:rep_date_end as date)), 2);

  select
    cr_rate,
    cr_curr_unit
  from opercurr
  where cr_from_cu_id = '840'
    and :rep_date_end between cr_date and cr_date_end
  into
    :cr_rate,
    :cr_curr_unit;
  cr_rate = coalesce(cr_rate, 1);
  cr_curr_unit = coalesce(cr_curr_unit, 1);

  delete from llc_main
  where lm_date = :rep_date_end;

  for
    select
      reg_unicode,
      reg_code,
      reg_name
    from region
    where reg_status = 3
    order by reg_code
    into
      :reg_id,
      :reg_code,
      :reg_name
  do
  begin
    select
      result
    from get_kv_id(10)
    into
      :lm_id;
    lm_no = right(:lm_id, 4);

    insert into llc_main (
      lm_id,
      lm_date,
      lm_date_status,
      lm_from_reg_id,
      lm_no,
      lm_memo,
      lm_no_str,
      lm_is_signed)
    values (
      :lm_id,
      :rep_date_end,
      :rep_date_end,
      :reg_id,
      :lm_no,
      '',
      :lm_no,
      1);

    insert into llc_sign (
      ls_from_lm_id,
      ls_number,
      ls_sign_type,
      ls_fio)
    values (
      :lm_id,
      1,
      '1',
      '');

    for
      select
        ks_from_k_id as aa_k_id,
        0 as aa_amount_active_end,
        0 as aa_deliq_days
      from kontstate
      where ks_ol_is_delete = 0
        and ks_state = 2
        and ks_region_id = :reg_id
        and ks_date_begin between :rep_date_begin and :rep_date_end
      union
      select
        oks_from_k_id,
        oks_balance,
        :rep_date_end - oks_delinq_date
      from oksummary
      where oks_region_id = :reg_id
        and :rep_date_end between oks_date_begin and oks_date_end
        and oks_close_state = 0
      into
        :aa_k_id,
        :aa_amount_active_end,
        :aa_deliq_days
    do
    begin
      select
        pr_loan_type,
        k_index_curr_code,
        k_index_curr_rate,
        k_disb_date,
        k_percent,
        coalesce(k_by_rating, 0),
        pr_name containing 'zaminlə',
        pr_name containing 'Biznes'
      from kontract
        inner join product
          on pr_id = k_from_pr_id
      where k_id = :aa_k_id
      into
        :aa_pr_is_indiv,
        :k_index_curr_code,
        :k_index_curr_rate,
        :k_disb_date,
        :k_percent,
        :k_by_rating,
        :is_secured,
        :is_business;

      if (aa_deliq_days < 0) then
        aa_deliq_days = 0;
      if (aa_amount_active_end <> 0 and
          k_index_curr_code = '840') then
      begin
        if (k_index_curr_rate > cr_rate) then
          aa_amount_active_end = cast(aa_amount_active_end * k_index_curr_rate / cr_curr_unit as decimal(18,2));
        else
          aa_amount_active_end = cast(aa_amount_active_end * cr_rate / cr_curr_unit as decimal(18,2));
      end

      if (is_secured) then
      begin
        -- обеспеченные поручительством
        if (aa_deliq_days between 0 and 30) then
        begin
          aa_llc_id = 3; -- удовлетворительные
          aa_llc_percent = 1;
        end else
        if (aa_deliq_days between 31 and 90) then
        begin
          aa_llc_id = 4; -- под наблюдением
          aa_llc_percent = iif(is_business, 2, 5);
        end else
        if (aa_deliq_days between 91 and 240) then
        begin
          aa_llc_id = 5; -- неудовлетворительные
          aa_llc_percent = 25;
        end else
        if (aa_deliq_days between 241 and 360) then
        begin
          aa_llc_id = 6; -- опасные
          aa_llc_percent = 50;
        end else
        begin
          aa_llc_id = 7; -- безнадежные
          aa_llc_percent = 100;
        end
      end else
      begin
        -- не обеспеченные поручительством
        if (aa_deliq_days between 0 and 30) then
        begin
          aa_llc_id = 3;
          aa_llc_percent = 1;
        end else
        if (aa_deliq_days between 31 and 90) then
        begin
          aa_llc_id = 4;
          aa_llc_percent = iif(is_business, 2, 5);
        end else
        if (aa_deliq_days between 91 and 180) then
        begin
          aa_llc_id = 5;
          aa_llc_percent = 25;
        end else
        if (aa_deliq_days between 181 and 270) then
        begin
          aa_llc_id = 6;
          aa_llc_percent = 50;
        end else
        begin
          aa_llc_id = 7;
          aa_llc_percent = 100;
        end
      end

      sh_calc_perc = coalesce((
        select kcp_perc_saldo
        from kontcalcperc
        where kcp_from_k_id = :aa_k_id
          and kcp_period = :rep_period), 0);
      if (sh_calc_perc <= 0) then
        sh_calc_perc = null;

      insert into llc_kont (
        lk_from_lm_id,
        lk_from_k_id,
        lk_deliq_days,
        lk_from_reg_id,
        lk_balance,
        lk_status,
        lk_percent,
        lk_date,
        lk_date_end,
        lk_memo,
        lk_is_indiv,
        lk_is_signed,
        lk_is_changed_mannual,
        lk_percent_orig,
        lk_calcperc)
      values (
        :lm_id,
        :aa_k_id,
        :aa_deliq_days,
        :reg_id,
        :aa_amount_active_end,
        :aa_llc_id,
        :aa_llc_percent,
        :rep_date_end,
        null,
        '',
        :aa_pr_is_indiv,
        1,
        0,
        :aa_llc_percent,
        :sh_calc_perc);
    end
  end
end
