CREATE PROCEDURE KIBSEND_CLOSED_FILL_KREDIT(
    REP_DATE DATE,
    REP_DELIQ_DAYS INTEGER)
AS
declare variable pr_categ integer;
declare variable kc_amount decimal(18,4);
declare variable kkc_status integer;
declare variable bank_code integer;
declare variable fil_code integer;
declare variable uniq_cred_num decimal(18,0);
declare variable credtype_code varchar(20);
declare variable credstatex_code varchar(20);
declare variable uniq_k_id decimal(18,0);
declare variable pocket_id integer;
declare variable c_id decimal(18,0);
declare variable repaystatus integer;
declare variable guarantee varchar(255);
declare variable countofdelinquency integer;
declare variable unpaidamount decimal(18,4);
declare variable unpaidpercentamount decimal(18,4);
declare variable need_check integer;
declare variable pocket_status integer;
declare variable paymethod smallint;
declare variable unpaidfine decimal(18,4);
declare variable fact_amount decimal(18,4);
declare variable plan_amount decimal(18,4);
declare variable deliq_days integer;
declare variable pay_frequency integer;
declare variable overdueamount decimal(18,4);
declare variable percentoverdueamount decimal(18,4);
declare variable fine decimal(18,4);
declare variable payoutday smallint;
declare variable fact_repayment_date date;
declare variable overduepayoutnumber integer;
begin
  /* CHANGED BY IGUSEV 14.06.2012
     changes:
     Добавил выборку по полю pr_interest_method из таблицы Product с целью дальнейшей обратотки и записи этого значение в поле KKR_PAYMETHOD таблицы kibkredit
  */

  /*
    CHANGED BY IGUSEV 27.07.2012
    changes:  дОбавлена выборка по следующим полям: UNPAIDFINE, FACT_AMOUNT, PLAN_AMOUNT, DELIQ_DAYS, PAY_FREQUENCY, PAYMETHOD, kpk.deliq_os, kpk.deliq_perc, fine, PayoutDay

  */

  for select bank_code, fil_code, uniq_cred_num, credtype_code, credstatex_code, uniq_k_id, pocket_id, repaystatus,
             guarantee, countofdelinquency, unpaidamount, unpaidpercentamount, stat, unpaidfine, fact_amount,
             plan_amount, deliq_days, pay_frequency, paymethod, kpk.deliq_os, kpk.deliq_perc, fine, payoutday,
             fact_repayment_date, overduepayoutnumber
      from kibsend_closed_contracts(:rep_date, :rep_deliq_days) kpk
      into :bank_code, :fil_code, :uniq_cred_num, :credtype_code, :credstatex_code, :uniq_k_id, :pocket_id,
           :repaystatus, :guarantee, :countofdelinquency, :unpaidamount, :unpaidpercentamount, :pocket_status,
           :unpaidfine, :fact_amount, :plan_amount, :deliq_days, :pay_frequency, :paymethod, :overdueamount,
           :percentoverdueamount, :fine, :payoutday, :fact_repayment_date, :overduepayoutnumber
  do
  begin
    need_check = 0;
    if (pocket_status = 0) then
      need_check = 1;

    if (exists(select *
               from kib_ignore ki
               where ki.ki_from_k_id = :uniq_cred_num)) then
    begin
      /* Если в кред. договоре есть клиенты которых нет в списке "Запрета на отправку в КИБ", тогда создаю запись в таблице kibkredit*/
      kkc_status = -1;

      for select c_id, kc_amount_approved, pr_category
          from clients, knclient, product pr, kontract
          where kc_from_c_id = c_id and
                kc_from_k_id = :uniq_k_id and
                c_type = 0 and
                k_id = :uniq_k_id and
                pr_id = k_from_pr_id and
                kc_from_c_id not in (select ki.ki_from_c_id
                                     from kib_ignore ki
                                     where ki.ki_from_k_id = :uniq_cred_num)
          order by c_id
          into :c_id, :kc_amount, :pr_categ
      do
      begin
        if (((kc_amount is null) or (kc_amount = 0)) and
            (pr_categ = 1)) then
        begin
          kkc_status = 1;
        end
        if ((kc_amount is not null) and
            (kc_amount <> 0)) then
        begin
          kkc_status = 0;
        end
      end
      if (kkc_status <> -1) then
      begin
        /* в КИБ-е
                     PayMethod.ID
                     (1 аннуитет
                      2 дифференцированный/стандарт
                      3 индивидуальный/гибкий  график
                      4 другое)
                */

        /**/
        insert into kibkredit (kkr_from_kp_id, kkr_uniq_cred_num, kkr_bank_code, kkr_fil_code, kkr_credtype_code,
                               kkr_credstatex_code, kkr_send_status, kkr_send_result, kkr_transid, kkr_from_k_id,
                               kkr_send_time, kkr_repaystatus, kkr_guarantee, kkr_countofdelinquency, kkr_unpaidamount,
                               kkr_unpaidpercentamount, kkr_paymethod, kkr_unpaidfine, kkr_fact_amount, kkr_plan_amount,
                               kkr_deliq_days, kkr_pay_frequency, kkr_overdueamount, kkr_percentoverdueamount, kkr_fine,
                               kkr_payoutday, kkr_factrepaymentdate, kkr_overduepayoutnumber)
        values (:pocket_id, :uniq_cred_num, :bank_code, :fil_code, :credtype_code, :credstatex_code, :need_check, '', 0,
                :uniq_k_id, null, :repaystatus, :guarantee, :countofdelinquency, :unpaidamount, :unpaidpercentamount,
                :paymethod, :unpaidfine, :fact_amount, :plan_amount, :deliq_days, :pay_frequency, :overdueamount,
                :percentoverdueamount, :fine, :payoutday, :fact_repayment_date, :overduepayoutnumber);
      end
    end
    else
    begin
      insert into kibkredit (kkr_from_kp_id, kkr_uniq_cred_num, kkr_bank_code, kkr_fil_code, kkr_credtype_code,
                             kkr_credstatex_code, kkr_send_status, kkr_send_result, kkr_transid, kkr_from_k_id,
                             kkr_send_time, kkr_repaystatus, kkr_guarantee, kkr_countofdelinquency, kkr_unpaidamount,
                             kkr_unpaidpercentamount, kkr_paymethod, kkr_unpaidfine, kkr_fact_amount, kkr_plan_amount,
                             kkr_deliq_days, kkr_pay_frequency, kkr_overdueamount, kkr_percentoverdueamount, kkr_fine,
                             kkr_payoutday, kkr_factrepaymentdate, kkr_overduepayoutnumber)
      values (:pocket_id, :uniq_cred_num, :bank_code, :fil_code, :credtype_code, :credstatex_code, :need_check, '', 0,
              :uniq_k_id, null, :repaystatus, :guarantee, :countofdelinquency, :unpaidamount, :unpaidpercentamount,
              :paymethod, :unpaidfine, :fact_amount, :plan_amount, :deliq_days, :pay_frequency, :overdueamount,
              :percentoverdueamount, :fine, :payoutday, :fact_repayment_date, :overduepayoutnumber);
    end
    /*********************************************************/

    for select c_id, kc_amount_approved, pr_category
        from clients, knclient, product, kontract
        where kc_from_c_id = c_id and
              kc_from_k_id = :uniq_k_id and
              c_type = 0 and
              k_id = :uniq_k_id and
              pr_id = k_from_pr_id
              /*IGUSEV 06.11.12 добавил group by т.к. возникала ситуация когда в договор забивают одного и того-же человека несколько раз с суммой = 0*/
        group by c_id, kc_amount_approved, pr_category
        order by c_id
        into :c_id, :kc_amount, :pr_categ
    do
    begin
      kkc_status = -1;
      if (((kc_amount is null) or (kc_amount = 0)) and
          (pr_categ = 1)) then
      begin
        kkc_status = 1;
      end
      if ((kc_amount is not null) and
          (kc_amount <> 0)) then
      begin
        kkc_status = 0;
      end
      if (kkc_status <> -1) then
      begin
        if (not exists(select *
                       from kib_ignore ki
                       where ki.ki_from_k_id = :uniq_cred_num and
                             ki.ki_from_c_id = :c_id)) then
        begin
          insert into kibclients (kkc_from_kp_id, kkc_from_k_id, kkc_from_c_id, kkc_status)
          values (:pocket_id, :uniq_cred_num, :c_id, :kkc_status);
        end
      end
    end
  end
end
