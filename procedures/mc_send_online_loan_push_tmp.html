CREATE PROCEDURE MC_SEND_ONLINE_LOAN_PUSH_TMP
RETURNS (
    MCL_ID D_ID,
    C_ID D_ID)
AS
declare variable PREVIOUS_C_ID D_ID;
declare variable AMOUNT D_CURRENCY;
begin
    previous_c_id = 0;

    for select
        mcl.mcl_id,
        mcl.mcl_from_c_id
    from mc_login mcl
    where mcl.mcl_is_delete = 0
        and mcl.mcl_from_c_id is not null
    order by mcl.mcl_from_c_id
    into
        mcl_id,
        c_id
    do
    begin
        if(c_id <> previous_c_id) then
        begin
            previous_c_id = c_id;

            amount = 0;
            select amount from mc_get_loan_limit(:mcl_id) into amount;
        
            if(exists
            (
                select 1
                from mc_online_loan_visibility(:mcl_id, :amount, 1)
                where is_visible = 1
        
            )) then
                suspend;

        end
    end
end
