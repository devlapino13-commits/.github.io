CREATE PROCEDURE AAREP_ANKETA_ANSWERS_KG(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    A_NATION D_STR_255,
    A_STATUS_PASS D_STR_255,
    A_STATUS_FACT D_STR_255,
    A_CNT D_STR_255,
    A_MEMBERS D_STR_255,
    A_EDU D_STR_255,
    A_WITH D_STR_255,
    A_OWN D_STR_255,
    A_HOUSE_RATE D_STR_255,
    A_HOUSE_TYPE D_STR_255,
    A_HOUSE_TIME D_STR_255,
    A_SOCIAL D_STR_255,
    A_DRIVE D_STR_255,
    A_CARD_OTHER D_STR_255,
    A_WORK_PLACE D_STR_255,
    A_WORK_ADDR D_STR_255,
    A_WORK_STATUS D_STR_255,
    A_WORK_KIND D_STR_255,
    A_WORK_JOB D_STR_255,
    A_WORK_STAJ D_STR_255,
    A_KNOW D_STR_255,
    A_VISIT_HOUSE D_STR_255,
    A_VISIT_WORK D_STR_255,
    A_IND D_STR_255,
    A_ORG D_STR_255,
    A_INC_BASE D_STR_255,
    A_INC_OTHER D_STR_255,
    A_OUT_BASE D_STR_255,
    A_OUT_HOUSE D_STR_255,
    A_OUT_DOCS D_STR_255,
    A_OUT_OTHER D_STR_255,
    A_OUT_LOANS D_STR_255,
    A_PROFIT D_STR_255,
    A_PAY D_STR_255,
    A_CARD_BANK D_STR_255,
    A_CARD_TYPE D_STR_255,
    A_REMOTE_PAY D_STR_255,
    A_REMOTE_GET D_STR_255,
    A_GOAL D_STR_255)
AS
declare variable reg_id d_integer;
declare variable c_id id_type;
declare variable rd_nation d_integer;
declare variable rd_status_pass d_integer;
declare variable rd_status_fact d_integer;
declare variable rd_cnt d_integer;
declare variable rd_members d_integer;
declare variable rd_edu d_integer;
declare variable rd_with d_integer;
declare variable rd_own d_integer;
declare variable rd_house_rate d_integer;
declare variable rd_house_type d_integer;
declare variable rd_house_time d_integer;
declare variable rd_social d_integer;
declare variable rd_drive d_integer;
declare variable rd_card_other d_integer;
declare variable rd_work_place d_integer;
declare variable rd_work_addr d_integer;
declare variable rd_work_status d_integer;
declare variable rd_work_kind d_integer;
declare variable rd_work_job d_integer;
declare variable rd_work_staj d_integer;
declare variable rd_know d_integer;
declare variable rd_visit_house d_integer;
declare variable rd_visit_work d_integer;
declare variable rd_ind d_integer;
declare variable rd_org d_integer;
declare variable rd_inc_base d_integer;
declare variable rd_inc_other d_integer;
declare variable rd_out_base d_integer;
declare variable rd_out_house d_integer;
declare variable rd_out_docs d_integer;
declare variable rd_out_other d_integer;
declare variable rd_out_loans d_integer;
declare variable rd_profit d_integer;
declare variable rd_pay d_integer;
declare variable rd_card_bank d_integer;
declare variable rd_card_type d_integer;
declare variable rd_remote_pay d_integer;
declare variable rd_remote_get d_integer;
declare variable rd_goal d_integer;
declare variable opr_subject_id d_integer;
declare variable ovr_value d_str_255;
begin

  -- отчет "Анкетные данные по активным займам" (RFCKG-1957)

  -- вопросы анкеты в KG:
  rd_nation = f_get_rd_id('anketaquestions.nation'); -- национальность
  rd_status_pass = f_get_rd_id('anketaquestions.family'); -- семейное положение по паспорту
  rd_status_fact = f_get_rd_id('anketaquestions.family_fact'); -- семейное положение по факту
  rd_cnt = f_get_rd_id('anketaquestions.1332'); -- количество членов семьи
  rd_members = f_get_rd_id('anketaquestions.1333'); -- в состав членов семьи входят
  rd_edu = f_get_rd_id('anketaquestions.education'); -- образование
  rd_with = f_get_rd_id('anketaquestions.1335'); -- с кем проживает
  rd_own = f_get_rd_id('anketaquestions.1348'); -- что есть в собственности у клиента
  rd_house_rate = f_get_rd_id('anketaquestions.1336'); -- оценка жилья
  rd_house_type = f_get_rd_id('anketaquestions.1337'); -- вид жилья
  rd_house_time = f_get_rd_id('anketaquestions.1338'); -- срок проживания
  rd_social = f_get_rd_id('anketaquestions.1339'); -- профиль в соц.сети
  rd_drive = f_get_rd_id('anketaquestions.1340'); -- водительские права
  rd_card_other = f_get_rd_id('anketaquestions.use_bankcard'); -- наличие банковской карты другого ФКУ
  rd_work_place = f_get_rd_id('work_place_name'); -- описание доходоприносящей деятельности
  rd_work_addr = f_get_rd_id('work_address_orientir'); -- ориентир места работы-бизнеса
  rd_work_status = f_get_rd_id('work_status'); -- вид занятости клиента
  rd_work_kind = f_get_rd_id('work_business_kind'); -- сфера деятельности клиента
  rd_work_job = f_get_rd_id('work_job_name'); -- должность клиента
  rd_work_staj = f_get_rd_id('anketaquestions.work_staj'); -- стаж работы в этой сфере
  rd_know = f_get_rd_id('anketaquestions.infosource'); -- как узнали о компании
  rd_visit_house = f_get_rd_id('anketaquestions.visithomeneeded'); -- требуется ли визит осмотр места жительства
  rd_visit_work = f_get_rd_id('anketaquestions.visitbusinessneeded'); -- требуется ли визит осмотр места работы
  rd_ind = f_get_rd_id('ind_entrepreneur'); -- индивидуальный предприниматель(да/нет)
  rd_org = f_get_rd_id('organization_type'); -- вид организации
  rd_inc_base = f_get_rd_id('monthly_income'); -- ежемесячный основной доход (бизнес/зарплата)
  rd_inc_other = f_get_rd_id('extra_amount'); -- сумма ежемесячных дополнительных доходов
  rd_out_base = f_get_rd_id('living_expenses'); -- семейные расходы в месяц (еда, одежда, дети, моб. связь, транспорт и прочее)
  rd_out_house = f_get_rd_id('anketaquestions.communal_expenses'); -- ежемесячные расходы на жилье (аренда, коммунальные услуги и др.)
  rd_out_docs = f_get_rd_id('anketaquestions.legal_docs_expenses'); -- расходы в месяц на патент и соц.полис (ип)
  rd_out_other = f_get_rd_id('anketaquestions.other_business_expenses'); -- другие ежемесячные расходы по бизнесу (аренда, сборы, транспорт и прочее)
  rd_out_loans = f_get_rd_id('anketaquestions.repayment_obligations'); -- погашение других долговых обязательств
  rd_profit = f_get_rd_id('net_profit'); -- чистая прибыль
  rd_pay = f_get_rd_id('amount_monthly_payments'); -- сумма ежемесячной выплаты
  rd_card_bank = f_get_rd_id('anketaquestions.bankcardsystem'); -- вид банковской карты
  rd_card_type = f_get_rd_id('anketaquestions.typeofcard'); -- тип банковской карты
  rd_remote_pay = f_get_rd_id('anketaquestions.repayloancard'); -- хотели ли бы вы погашать свой кредит в м булак удаленно через карту?
  rd_remote_get = f_get_rd_id('anketaquestions.getloancard'); -- хотели ли бы впредь получать от м булак кредит удаленно на карту?
  rd_goal = f_get_rd_id('anketaquestions.descr_of_loan'); -- подробное описание по цели

  -- перебираем офисы заданного региона
  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into reg_id
  do
  begin
    -- перебираем клиентов активных займов по одному офису
    for select kc_from_c_id
        from knclient
        inner join kontstate on ks_from_k_id = kc_from_k_id
        where kc_amount_approved > 0 and
              ks_region_id = :reg_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and -- активные займы
              :rep_date_end between ks_date_begin and ks_date_end -- фильтр на дату
        group by kc_from_c_id
        into c_id
    do
    begin
      c_name_fio = '';
      c_inn = '';
      a_nation = '';
      a_status_pass = '';
      a_status_fact = '';
      a_cnt = '';
      a_members = '';
      a_edu = '';
      a_with = '';
      a_own = '';
      a_house_rate = '';
      a_house_type = '';
      a_house_time = '';
      a_social = '';
      a_drive = '';
      a_card_other = '';
      a_work_place = '';
      a_work_addr = '';
      a_work_status = '';
      a_work_kind = '';
      a_work_job = '';
      a_work_staj = '';
      a_know = '';
      a_visit_house = '';
      a_visit_work = '';
      a_ind = '';
      a_org = '';
      a_inc_base = '';
      a_inc_other = '';
      a_out_base = '';
      a_out_house = '';
      a_out_docs = '';
      a_out_other = '';
      a_out_loans = '';
      a_profit = '';
      a_pay = '';
      a_card_bank = '';
      a_card_type = '';
      a_remote_pay = '';
      a_remote_get = '';
      a_goal = '';

      select c_name_fio,
             c_inn
      from clients
      where c_id = :c_id
      into c_name_fio,
           c_inn;

      opr_subject_id = 0;
      ovr_value = '';

      -- перебираем вопросы и ответы анкеты 'Информация о клиенте'
      for select opr_subject_id,
                 iif(coalesce(min(ovr_value), '') = '', list(rd_name, ', '), min(ovr_value))
          from objectprops
          inner join objectvalues on ovs_from_opr_id = opr_id
          inner join objectvalres on ovr_from_opr_id = opr_id and ovr_from_ovs_id = ovs_id
          inner join refdata on rd_id = ovr_result_rd_id
          where opr_object_kind = 0 and
                opr_subject_kind = -1 and
                opr_object_id = :c_id and -- фильтр по клиенту
                :rep_date_end between ovs_date_begin and ovs_date_end -- фильтр на дату
          group by opr_subject_id -- группировка по вопросу, т.к. ответ может состоять из нескольких значений
          into opr_subject_id, -- id вопроса
               ovr_value -- ответ
      do
      begin
        -- в ответе заменяем перенос строки на пробел
        ovr_value = trim(ovr_value);
        if (char_length(ovr_value) <= 253) then
          ovr_value = replace(replace(ovr_value, ascii_char(10), ' '), ascii_char(13), ' ');
        else
          ovr_value = replace(replace(ovr_value, ascii_char(10), ''), ascii_char(13), '');

        -- в ответе заменяем прямые парные кавычки на одинарные, т.к. в эксель парные не видны (если не экранированы)
        -- кроме того, текст между ними считается одной строкой, что мешает при экспорте в эксель через csv-файл
        ovr_value = replace(ovr_value, ascii_char(34), ascii_char(39));

        -- в ответе заменяем точки с запятой на запятые, т.к. в csv-файле они будут разделителями полей
        ovr_value = replace(ovr_value, ascii_char(27), ascii_char(12));

        -- ответ пишем в соответствующий выходной параметр
        if (opr_subject_id = rd_nation) then a_nation = ovr_value;
        else if (opr_subject_id = rd_status_pass) then a_status_pass = ovr_value;
        else if (opr_subject_id = rd_status_fact) then a_status_fact = ovr_value;
        else if (opr_subject_id = rd_cnt) then a_cnt = ovr_value;
        else if (opr_subject_id = rd_members) then a_members = ovr_value;
        else if (opr_subject_id = rd_edu) then a_edu = ovr_value;
        else if (opr_subject_id = rd_with) then a_with = ovr_value;
        else if (opr_subject_id = rd_own) then a_own = ovr_value;
        else if (opr_subject_id = rd_house_rate) then a_house_rate = ovr_value;
        else if (opr_subject_id = rd_house_type) then a_house_type = ovr_value;
        else if (opr_subject_id = rd_house_time) then a_house_time = ovr_value;
        else if (opr_subject_id = rd_social) then a_social = ovr_value;
        else if (opr_subject_id = rd_drive) then a_drive = ovr_value;
        else if (opr_subject_id = rd_card_other) then a_card_other = ovr_value;
        else if (opr_subject_id = rd_work_place) then a_work_place = ovr_value;
        else if (opr_subject_id = rd_work_addr) then a_work_addr = ovr_value;
        else if (opr_subject_id = rd_work_status) then a_work_status = ovr_value;
        else if (opr_subject_id = rd_work_kind) then a_work_kind = ovr_value;
        else if (opr_subject_id = rd_work_job) then a_work_job = ovr_value;
        else if (opr_subject_id = rd_work_staj) then a_work_staj = ovr_value;
        else if (opr_subject_id = rd_know) then a_know = ovr_value;
        else if (opr_subject_id = rd_visit_house) then a_visit_house = ovr_value;
        else if (opr_subject_id = rd_visit_work) then a_visit_work = ovr_value;
        else if (opr_subject_id = rd_ind) then a_ind = ovr_value;
        else if (opr_subject_id = rd_org) then a_org = ovr_value;
        else if (opr_subject_id = rd_inc_base) then a_inc_base = ovr_value;
        else if (opr_subject_id = rd_inc_other) then a_inc_other = ovr_value;
        else if (opr_subject_id = rd_out_base) then a_out_base = ovr_value;
        else if (opr_subject_id = rd_out_house) then a_out_house = ovr_value;
        else if (opr_subject_id = rd_out_docs) then a_out_docs = ovr_value;
        else if (opr_subject_id = rd_out_other) then a_out_other = ovr_value;
        else if (opr_subject_id = rd_out_loans) then a_out_loans = ovr_value;
        else if (opr_subject_id = rd_profit) then a_profit = ovr_value;
        else if (opr_subject_id = rd_pay) then a_pay = ovr_value;
        else if (opr_subject_id = rd_card_bank) then a_card_bank = ovr_value;
        else if (opr_subject_id = rd_card_type) then a_card_type = ovr_value;
        else if (opr_subject_id = rd_remote_pay) then a_remote_pay = ovr_value;
        else if (opr_subject_id = rd_remote_get) then a_remote_get = ovr_value;
        else if (opr_subject_id = rd_goal) then a_goal = ovr_value;
      end

      suspend;
    end
  end
end
