CREATE PROCEDURE AAREP_ACCOUNT_ONEREP(
    REPORT_ACCOUNT_ID VARCHAR(20),
    REPORT_DATE DATE)
RETURNS (
    LINE_NAME VARCHAR(40),
    LINE_DT DECIMAL(18,4),
    LINE_KT DECIMAL(18,4))
AS
declare variable acc_summa1 decimal(18,4);
declare variable acc_summadt decimal(18,4);
declare variable acc_summakt decimal(18,4);
begin
  acc_summa1 = 0;
  acc_summadt = 0;
  acc_summakt = 0;

  acc_summa1 = f_al_bal(report_account_id, report_date, 'val');

  line_name = 'Сальдо на начало';
  if (acc_summa1 >= 0) then
  begin
    line_dt = acc_summa1;
    line_kt = 0;
  end
  else
  begin
    line_dt = 0;
    line_kt = -1 * acc_summa1;
  end
  suspend;

  select sum(obs_dt_val), sum(obs_kt_val)
  from obsummary
  where obs_from_al_id = :report_account_id and
        obs_date = :report_date
  into :acc_summadt, :acc_summakt;

  if (acc_summadt is null) then
    acc_summadt = 0;
  if (acc_summakt is null) then
    acc_summakt = 0;

  line_name = 'Оборот за день';
  line_dt = acc_summadt;
  line_kt = acc_summakt;
  suspend;
  acc_summa1 = acc_summa1 + acc_summadt - acc_summakt;
  line_name = 'Сальдо на конец';
  if (acc_summa1 >= 0) then
  begin
    line_dt = acc_summa1;
    line_kt = 0;
  end
  else
  begin
    line_dt = 0;
    line_kt = -1 * acc_summa1;
  end
  suspend;

end
