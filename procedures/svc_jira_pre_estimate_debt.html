CREATE PROCEDURE SVC_JIRA_PRE_ESTIMATE_DEBT(
    ADD_PROJ_WITH_TO D_BOOL = 0,
    ADD_PROJ_WITHOUT_TO D_BOOL = 0)
RETURNS (
    JPP_HAS_TO D_BOOL,
    JRI_ISSUE_ID D_INTEGER,
    JRI_PARENT_ID D_INTEGER,
    JRI_DEV_ID D_ID,
    JRI_TYPE_ID D_INTEGER,
    JRI_JIRA_CODE D_STR_CODE,
    JRI_TITLE D_STR_LONG,
    JRI_PROJECT D_STR_CODE,
    JRI_LEVEL D_INTEGER,
    JRI_PRIORITY D_SMALLINT,
    JRI_KPI_TIMEZONE D_SMALLINT,
    JRI_ESTIMATE_SECS D_ID,
    JRI_ESTIMATE_PERIOD D_STR_CODE,
    JRI_WORKED_SECS D_ID,
    JRI_CREATED D_DATETIME,
    JRI_FINISHED D_DATETIME,
    JRI_CLOSED D_DATETIME,
    JRI_STATE_ID D_SMALLINT,
    JRI_STATE_DATETIME D_DATETIME,
    JRI_RECALC D_SMALLINT,
    JRI_KPI_PERIOD D_INTEGER,
    JRI_CHECK_STATUS D_SMALLINT,
    JRI_CHECK_MEMO D_STR_255,
    JRI_CATEGORY_ID D_INTEGER,
    JRI_SOFTWARE_ID D_INTEGER,
    JRI_JOB_KIND D_SMALLINT,
    JRD_LOGIN D_STR_CODE,
    JRD_FIO D_STR_100,
    JRT_TEXT D_STR_CODE,
    TASK_LAST_STATE D_STR_CODE,
    TASK_STACK D_STR_CODE)
AS
declare variable JPP_CODE D_STR_CODE;
declare variable JPP_COUNTRY D_INTEGER;
declare variable JPI_QTY D_INTEGER;
declare variable JIRA_CODE_TMP D_STR_10;
declare variable I D_INTEGER;
declare variable JPI_ID D_INTEGER;
begin
  if ((add_proj_with_to = 0) and (add_proj_without_to = 0)) then
    exit;

  for
      select jp.jpp_code,
             jp.jpp_description,
             jp.jpp_country,
             jp.jpp_has_to,
             ji.jpi_level,
             ji.jpi_qty,
             js.jts_description,
             ji.jpi_id
        from jira_pre_estimate_proj jp
        inner join jira_pre_estimate_issues ji on jp.jpp_id = ji.jpi_from_jpp_id
        inner join jira_tech_stack js on js.jts_id = ji.jpi_stack
      into :jpp_code,
           :jri_title,
           :jpp_country,
           :jpp_has_to,
           :jri_level,
           :jpi_qty,
           :task_stack,
           :jpi_id
  do
  begin
    if ((add_proj_with_to = 1) and (add_proj_without_to = 0) and (jpp_has_to = 0)) then
      continue;

    if ((add_proj_with_to = 0) and (add_proj_without_to = 1) and (jpp_has_to = 1)) then
      continue;

    jira_code_tmp =
      case
        when jpp_country = 41701 then 'RFCKG'
        when jpp_country = 81001 then 'RFCRU'
        when jpp_country = 94401 then 'RFCAZ'
        when jpp_country = 39801 then 'RFCKZ'
        else 'RFCKG'
      end;

    i = 1;
    while (i <= coalesce(jpi_qty, 0)) do
    begin
      jri_jira_code = jira_code_tmp || '-' || jpp_code || '-' || cast(jpi_id as d_str_20);

      jri_issue_id = -1 * jpi_id;
      jri_parent_id = null;
      jri_dev_id = null;
      jri_type_id = null;
      jri_project = '';
      jri_priority = null;
      jri_kpi_timezone = null;
      jri_estimate_secs = null;
      jri_estimate_period = null;
      jri_worked_secs = null;
      jri_created = current_timestamp;
      jri_finished = null;
      jri_closed = null;
      jri_state_id = 1;
      jri_state_datetime = current_timestamp;
      jri_recalc = null;
      jri_kpi_period = null;
      jri_check_status = null;
      jri_check_memo = null;
      jri_category_id = null;
      jri_software_id = null;
      jri_job_kind = null;
      jrd_login = null;
      jrd_fio = null;
      jrt_text = 'Задача';
      task_last_state = 'Создана';

      suspend;

      i = i + 1;
    end
  end
end
