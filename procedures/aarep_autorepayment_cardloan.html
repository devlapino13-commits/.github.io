CREATE PROCEDURE AAREP_AUTOREPAYMENT_CARDLOAN(
    REP_REGION_ID D_INTEGER = null,
    REP_DATE_BEGIN D_DATE = null,
    REP_DATE_END D_DATE = null)
RETURNS (
    REGNAME D_STR_CODE,
    OFFNAME D_STR_CODE,
    KFNAME D_STR_100,
    KONTNO D_STR_20,
    DTPAY D_DATE,
    AMOUNTC D_CURRENCY,
    AMOUNT D_CURRENCY)
AS
declare variable offid d_integer;
declare variable reg_lst d_str_8100;
begin
  rep_region_id = coalesce(rep_region_id, 1000000);
  rep_date_begin = coalesce(rep_date_begin, current_date);
  rep_date_end = coalesce(rep_date_end, current_date);

  reg_lst = '';
  select
    list(id_list, ';')
  from region_list(:rep_region_id,1,3,1)
  into
    :reg_lst;

  for
    select
      kal_new_kf_name,
      k_kontract_no,
      r1.reg_unicode, 
      r1.reg_name,
      r2.reg_name,
      arc_ol_date,
      arc_cs_amount,
      arc_amount
    from autorepayment_cardloan
      inner join kontract
        on k_id = arc_from_k_id
      inner join knaktlist
        on kal_from_kf_id = k_from_kf_id
          and arc_ol_date between kal_date and kal_date_end
          and is_delete = 0
      inner join region as r1
        on r1.reg_unicode = arc_from_reg_id
      left join region as r2
        on r2.reg_unicode = r1.parent
    where arc_ol_date between :rep_date_begin and :rep_date_end
      and arc_from_ol_id > 0
  into
    :kfname,
    :kontno,
    :offid,
    :offname,
    :regname,
    :dtpay,
    :amountc,
    :amount
  do
  begin
    if (reg_lst containing offid)
      then suspend;
  end
end
