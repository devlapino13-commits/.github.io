CREATE PROCEDURE HP_BLOBFILES_SYNC
AS
declare variable blf_id integer;
declare variable blf_filename d_str_255;
declare variable blf_first_id id_type;
declare variable blf_last_id id_type;
declare variable blf_filename2 d_str_255;
declare variable blf_used d_smallint;

declare variable db_name d_str_64;
declare variable sql_text d_str_2048;
begin
  if (exists(select * from hp_db_is_actual)) then exit;

  db_name = rdb$get_context('SYSTEM', 'DB_NAME');

  if (not db_name containing '_last' or (exists(select * from hp_db_is_actual)) ) then
    exception e_default 'Необходимо выполнять на ласт базе';
  db_name = replace(db_name, '_last', '');

  sql_text = 'select blf_id, blf_filename, blf_first_id, blf_last_id, blf_filename2,
             blf_used
      from blobfiles
      where --(blf_id = 0) or
            (blf_id > (select max(blf_id) - 2
                       from blobfiles))
      order by blf_id';
  for execute statement :sql_text
          on external '127.0.0.1/3050:' || db_name
          as user 'SYSDBA' password 'F!ora80X'
          into :blf_id, :blf_filename, :blf_first_id, :blf_last_id,
          :blf_filename2, :blf_used
  do
  begin
    update or insert into blobfiles (blf_id, blf_filename, blf_first_id,
                                     blf_last_id, blf_filename2, blf_used)
    values (:blf_id, :blf_filename, :blf_first_id, :blf_last_id, :blf_filename2,
            :blf_used)
    matching (blf_id);
  end
end
