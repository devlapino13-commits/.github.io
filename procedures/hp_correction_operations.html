CREATE PROCEDURE HP_CORRECTION_OPERATIONS(
    K_ID D_ID)
AS
declare variable k_last_reg_id integer;
declare variable ks_from_ol_id d_id;
declare variable ks_date_begin date;
declare variable sh_id d_integer;
declare variable sh_date date;
declare variable zad_os d_currency;
declare variable zad_perc d_currency;
begin
  select kon.k_last_reg_id, ks.ks_from_ol_id, ks.ks_date_begin
  from kontract kon
  inner join kontstate ks on ks.ks_from_k_id = kon.k_id
  where kon.k_id = :k_id and
        ks.ks_ol_is_delete = 0 and
        ks.ks_state = 2
  into k_last_reg_id, ks_from_ol_id, ks_date_begin;

  sh_id = null;
  sh_date = null;
  zad_os = null;
  zad_perc = null;
  for select sh_id, sh_date, sum(zad_os), sum(zad_perc)
      from aarep_make_zad_list(:k_id, current_date, 1)
      group by sh_id, sh_date
      into sh_id, sh_date, zad_os, zad_perc
  do
  begin
    -- создаем запись ok_type 3
    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                          ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                          ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
    values (:ks_from_ol_id, -1, :k_id, :sh_id, :sh_date, 3, 0, 0, 0, 3, 0, 0, -:zad_perc, 0, 0, :ks_date_begin,
            :k_last_reg_id, 1, 0);

  end
end
