CREATE PROCEDURE MC_CLIENT_PURCHASE_JOURNAL(
    MCL_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    MP_NAME_MAGAZIN D_STR_255,
    MP_ID D_ID,
    MP_DATETIME D_DATETIME,
    MP_AMOUNT DECIMAL(18,2),
    MP_STATUS D_INTEGER,
    JSON_RES D_STR_LONG,
    CATEGORY D_STR_100)
AS
declare variable gb_status_int d_integer;
declare variable gb_date d_date;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable kagent_fio d_str_100;
declare variable mp_date d_date;
declare variable mp_status_str d_str_100;
begin
  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into c_id;

  -- процедура отображает покупки клиента по товарам в рассрочку за указанный период
  for select cs.cs_name, mcr.mp_id, mcr.mp_datetime, mcr.mp_amount, mcr.mp_status, c.c_name_fio, mcr.mp_date,
             feg.feg_name,
             case mcr.mp_status
               when -1 then 'Ошибка'
               when 0 then 'Отменено'
               when 1 then 'В обработке'
               when 2 then 'Успешно'
             end
      from mc_purchase mcr
      inner join mc_login mcl on mcl.mcl_id = mcr.mp_from_mcl_id
      inner join clientshops cs on cs.cs_id = mcr.mp_from_cs_id
      inner join clients c on cs.cs_from_c_id = c.c_id
      inner join foreign_econom_goods_ref feg on feg.feg_id = mcr.mp_from_feg_id
      where mcl.mcl_from_c_id = :c_id and
            mcr.mp_date between :rep_date_begin and :rep_date_end and
            mcr.mp_from_mcl_id is not null and
            mcr.mp_is_finished = 1
      order by mcr.mp_datetime desc
      into :mp_name_magazin, :mp_id, :mp_datetime, :mp_amount, :mp_status, :kagent_fio, :mp_date, :category,
           :mp_status_str
  do
  begin
    c_name_fio = null;

    select c.c_name_fio
    from clients c
    where c_id = :c_id
    into c_name_fio;

    json_res = '{
                         "operation_type": "Покупка в рассрочку",
                         "operation_no": "' || mp_id || '",
                         "client_name": "' || c_name_fio || '",
                         "seller_name": "' || replace(kagent_fio, '"', '\"') || '",
                         "amount_total": "' || mp_amount || '",
                         "amount": "' || mp_amount || '",
                         "operation_status": "' || :mp_status_str || '",
                         "operation_date": "' || f_datetostr(mp_date) || '"
                        }';

    suspend;
  end
end
