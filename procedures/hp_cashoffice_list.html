CREATE PROCEDURE HP_CASHOFFICE_LIST(
    SHOW_CLOSED D_BOOL)
RETURNS (
    CSO_OFFICE_ID D_INTEGER,
    CSO_IS_ONLINE D_BOOL,
    CSO_ON_PAUSE D_BOOL,
    CSO_FROM_UN_ID D_ID,
    CSO_IS_ACTIVE D_BOOL,
    CSO_ACTIVATE_DATE D_DATE,
    CSO_DATETIME D_DATETIME,
    REG_NAME D_STR_100,
    REG_ADDRESS D_STR_255,
    UN_NAME D_STR_CODE,
    CASH_COUNT D_INTEGER,
    REG_DATE_CLOSE D_DATE,
    REG_STATE D_STR_CODE)
AS
begin
  for select cso_office_id,
             cso_is_online,
             cso_on_pause,
             cso_from_un_id,
             cso_is_active,
             cso_activate_date,
             un_name,
             cso_datetime,
             reg_code || ' ' || reg_name,
             adl_address_str,
             reg_date_close
      from cashoffice
           inner join region on (cso_office_id = reg_unicode)
           inner join username on (cso_from_un_id = un_id)
           left join addrlist on (reg_address_fact = adl_id)
      where :show_closed = 1 or coalesce(reg_date_close, current_date) >= current_date
      into :cso_office_id,
           :cso_is_online,
           :cso_on_pause,
           :cso_from_un_id,
           :cso_is_active,
           :cso_activate_date,
           :un_name,
           :cso_datetime,
           :reg_name,
           :reg_address,
           :reg_date_close
  do
  begin
    select count(*)
    from cashplace
    where csp_office_id = :cso_office_id and
          coalesce(csp_date_open, current_date) <= current_date and
          coalesce(csp_date_close, current_date) >= current_date
    into cash_count;

    reg_state = 'Действующие офисы';

    if (reg_date_close <= current_date) then
      reg_state = 'Закрытые офисы';

    suspend;
  end
end
