CREATE PROCEDURE AAREP_PETROL_OBOROT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PEO_PROBEG_FACT_LITER DECIMAL(18,4),
    PEO_PROBEG_FACT_KM DECIMAL(18,4),
    PEO_OFFICE_NAME VARCHAR(100),
    PEO_BRANCH_NAME VARCHAR(100),
    PEO_OL_DATE DATE,
    PEO_OL_MEMO VARCHAR(1024),
    PEO_C_NAME_FIO VARCHAR(100),
    PEO_TRANS_NAME VARCHAR(255),
    PEO_TRANS_LABEL VARCHAR(50),
    PEO_TRANS_LITER_PLUS DECIMAL(18,4),
    PEO_TRANS_SUMMA_PLUS DECIMAL(18,4),
    PEO_TRANS_LITER_MINUS DECIMAL(18,4),
    PEO_TRANS_SUMMA_MINUS DECIMAL(18,4),
    PEO_OPER_TYPE_STR VARCHAR(50),
    PEO_PETROL_NAME VARCHAR(100))
AS
declare variable peo_trans_liter decimal(18,4);
declare variable peo_trans_summa decimal(18,4);
declare variable peo_oper_type integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select peo_oper_type, sum(peo_trans_liter), sum(peo_trans_summa), sum(peo_probeg_fact_liter),
             sum(peo_probeg_fact_km), r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, ol_date,
             ol_memo, c_name_fio, pet_name, pet_label, pen_name
      from petopers, operlist, clients, pettrans, region r1, region r2, petname, region_list(:rep_region_id, 1, -1, 1)
      where peo_from_ol_id = ol_id and
            peo_from_c_id = c_id and
            peo_from_pet_id = pet_id and
            peo_from_reg_id = r1.reg_unicode and
            ol_from_reg_id = r2.reg_unicode and
            (peo_from_reg_id = id_list) and
            ol_date_oper_day between :rep_date_begin and :rep_date_end and
            operlist.is_delete = 0 and
            peo_from_pen_id = pen_id
      group by peo_oper_type, peo_probeg_fact_liter, peo_probeg_fact_km, r1.reg_code, r1.reg_name, r2.reg_code, r2.reg_name, ol_date, ol_memo, c_name_fio, pet_name, pet_label, pen_name
      having sum(peo_trans_liter) <> 0 or sum(peo_trans_summa) <> 0 or sum(peo_probeg_fact_liter) <> 0 or sum(peo_probeg_fact_km) <> 0

      into :peo_oper_type, :peo_trans_liter, :peo_trans_summa, :peo_probeg_fact_liter, :peo_probeg_fact_km,
           :peo_office_name, :peo_branch_name, :peo_ol_date, :peo_ol_memo, :peo_c_name_fio, :peo_trans_name,
           :peo_trans_label, :peo_petrol_name
  do
  begin
    peo_trans_liter_plus = 0;
    peo_trans_summa_plus = 0;
    peo_trans_liter_minus = 0;
    peo_trans_summa_minus = 0;

    if (peo_trans_liter >= 0) then
      peo_trans_liter_plus = :peo_trans_liter;
    else
      peo_trans_liter_minus = -1 * :peo_trans_liter;

    if (peo_trans_summa >= 0) then
      peo_trans_summa_plus = :peo_trans_summa;
    else
      peo_trans_summa_minus = -1 * :peo_trans_summa;

    if (peo_oper_type = 0) then
      peo_oper_type_str = 'Первоначальные остатки';
    if (peo_oper_type = 1) then
      peo_oper_type_str = 'Приход на склад';
    if (peo_oper_type = 2) then
      peo_oper_type_str = 'Залито через Чек';
    if (peo_oper_type = 3) then
      peo_oper_type_str = 'Передан в другой офис';
    if (peo_oper_type = 4) then
      peo_oper_type_str = 'Принят из другого офиса';
    if (peo_oper_type = 5) then
      peo_oper_type_str = 'Залито в бак со склада';
    if (peo_oper_type = 6) then
      peo_oper_type_str = 'Возврашен из бака';
    if (peo_oper_type = 7) then
      peo_oper_type_str = 'Списан из бака';
    if (peo_oper_type = 8) then
      peo_oper_type_str = 'Заменен мат.ответ.';
    if (peo_oper_type = 9) then
      peo_oper_type_str = 'Передача между объектами';

    suspend;
  end
end
