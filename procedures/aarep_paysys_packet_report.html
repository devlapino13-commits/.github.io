CREATE PROCEDURE AAREP_PAYSYS_PACKET_REPORT(
    REP_REPORT_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    FIRST_PSI_CODE VARCHAR(20),
    LAST_PSI_CODE VARCHAR(20),
    PSI_COUNT INTEGER,
    PSP_NAME VARCHAR(255))
AS
declare variable psi_id decimal(18,0);
declare variable psi_code varchar(20);
declare variable id decimal(18,0);
declare variable first_psi_id decimal(18,0);
declare variable last_psi_id decimal(18,0);
declare variable last_psp_id varchar(255);
declare variable psp_id integer;
declare variable psp_name_tmp varchar(255);
begin
-- Выдает список пачек бланков по проекту, по типу и ID источника
-- kind = 1 поиск в офисе
-- kind = 2 поиск у сотрудника
  for select reg_unicode, reg_code || ' ' || reg_name
      from region, region_list(:rep_report_id, 1, 3, 1)
      where reg_unicode = id_list
      into :id, :reg_name
  do
  begin
    last_psi_id = null;
    for select psl_from_psi_id
        from paysys_list, paysys_main
        where psl_object_kind = 1 and
              psl_object_id = :id and
              psl_is_delete = 0 and
              psl_from_psm_id = psm_id and
              psm_is_delete = 0 and
              psm_date <= :rep_date_end
        group by psl_from_psi_id
        having sum(psl_sign) = 1
        into :psi_id
    do
    begin
      psi_code = null;
      select psi_code, psp_id, psp_name
      from paysys_item, paysys_project
      where psi_id = :psi_id and
            psi_from_psp_id = psp_id
      into :psi_code, :psp_id, :psp_name_tmp;
      if (psi_code is not null) then
      begin
        if (last_psi_id is null) then
        begin
          first_psi_id = psi_id;
          first_psi_code = psi_code;
          last_psp_id = psp_id;
          psp_name = psp_name_tmp;
        end
        else
        if (not(psi_id = last_psi_id + 1 and
            cast(psi_code as decimal(18,0)) = cast(last_psi_code as decimal(18,0)) + 1 and
            last_psp_id = psp_id)) then
        begin
-- Встретили разрыв, пачка кончилась
          psi_count = last_psi_id - first_psi_id + 1;
          suspend;
-- Начинаем новую
          first_psi_id = psi_id;
          first_psi_code = psi_code;
          psp_name = psp_name_tmp;
        end
      end
      last_psi_id = psi_id;
      last_psi_code = psi_code;
      last_psp_id = psp_id;
    end
    if (last_psi_id is not null) then
    begin
      psi_count = last_psi_id - first_psi_id + 1;
      suspend;
    end
  end
end
