CREATE PROCEDURE UXP$UXFRAME_TREE(
    UFM_UUID D_UUID,
    MASTER_FRAME D_UUID = null,
    UFM_UUID_CODE D_STR_CODE = null)
RETURNS (
    FR_UUID D_UUID,
    MASTER_FR_UUID D_UUID,
    FR_CODE D_STR_CODE,
    FR_MASTER_FIELD D_STR_CODE,
    FR_DETAIL_FIELD D_STR_CODE,
    FR_IS_GRID D_BOOL,
    FR_OWNER D_UUID,
    RES_CAPTION D_STR_100,
    RES_IMAGE D_INTEGER,
    FR_SOURCE D_STR_CODE,
    FR_US_IS_TABLE D_BOOL,
    FR_SOURCE_UUID D_UUID,
    UFR_TREE_PARENT_FIELD D_STR_CODE,
    UFR_TREE_IMAGE_FIELD D_STR_CODE,
    UFR_REFRESH_PARAMS D_STR_LONG,
    UFR_TREE_POSITION_FIELD D_STR_CODE,
    UFR_TREE_CAN_DRAG_FIELD D_STR_CODE,
    UFR_TREE_CAN_DROP_FIELD D_STR_CODE)
AS
declare variable uff_is_mainframe d_bool;
declare variable form_source d_str_code;
declare variable tmp_master d_uuid;
declare variable fr_count d_integer;
begin
  if (coalesce(ufm_uuid, '') = '') then
    ufm_uuid = uxf$uniq(ufm_uuid_code);

  if (trim(coalesce(master_frame, '')) = '') then
    master_frame = null;

  select count(*)
  from ux$frame
  where ufr_owner = :ufm_uuid
  into fr_count;

  select us_code
  from ux$form, ux$code, ux$source
  where ufm_uuid = :ufm_uuid and
        ufm_ux_uuid = ux_uuid and
        ux_us_uuid = us_uuid
  into form_source;

  for select ufr_uuid, ufr_name,
             ufr_code, ufr_master_ufr_uuid, ufr_master_field, ufr_detail_field,
             ufr_is_grid, ufr_tree_parent_field, ufr_tree_image_field, ufr_is_mainframe, ufr_owner,
             us_code, us_uuid, us_is_table, ufr_refresh_params,
             ufr_tree_position_field, ufr_tree_can_drag_field, ufr_tree_can_drop_field
  from ux$frame, ux$source
  where ufr_owner = :ufm_uuid and
        ufr_master_ufr_uuid is not distinct from :master_frame and
        ufr_us_uuid = us_uuid
  order by ufr_is_mainframe desc
  into fr_uuid, res_caption, fr_code, master_fr_uuid, fr_master_field, fr_detail_field,
       fr_is_grid, ufr_tree_parent_field, ufr_tree_image_field, uff_is_mainframe, fr_owner,
       fr_source, fr_source_uuid, fr_us_is_table, ufr_refresh_params,
       ufr_tree_position_field, ufr_tree_can_drag_field, ufr_tree_can_drop_field
  do
  begin
    res_image = 17;

    if (fr_code <> '' and res_caption not containing fr_code) then
      res_caption = res_caption || ' (' || fr_code || ')';

    if (fr_master_field is null) then
      fr_master_field = '';

    if (fr_detail_field is null) then
      fr_detail_field = '';

    if (uff_is_mainframe = 1) then
    begin
      if (fr_count > 1) then
        res_caption = res_caption || '  (основной)';

      res_image = 47;
    end else
      if (fr_source <> form_source) then
        res_image = 18;

    suspend;

    tmp_master = fr_uuid;

    for select fr_uuid, master_fr_uuid, fr_code, fr_master_field, fr_detail_field,
               fr_is_grid, ufr_tree_parent_field, ufr_tree_image_field, fr_owner, res_caption, res_image,
               fr_source, fr_source_uuid, fr_us_is_table, ufr_refresh_params,
               ufr_tree_position_field, ufr_tree_can_drag_field, ufr_tree_can_drop_field
    from uxp$uxframe_tree(:ufm_uuid, :tmp_master)
    into :fr_uuid, :master_fr_uuid, :fr_code, :fr_master_field, :fr_detail_field,
         :fr_is_grid, :ufr_tree_parent_field, :ufr_tree_image_field, :fr_owner, :res_caption, :res_image,
         fr_source, fr_source_uuid, fr_us_is_table, ufr_refresh_params,
         ufr_tree_position_field, ufr_tree_can_drag_field, ufr_tree_can_drop_field
    do
      suspend;
  end
end
