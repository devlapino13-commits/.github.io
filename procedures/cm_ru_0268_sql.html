CREATE PROCEDURE CM_RU_0268_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
begin
  for select kc_from_c_id
      from kontract
      inner join knclient on kc_from_k_id = k_id
      inner join product on pr_id = k_from_pr_id
      where kc_amount_approved > 0 and
            kc_from_k_id = :k_id and
            (pr_code containing '.universal.' or
            pr_code containing '.oivo_pack.')

      into :c_id
  do
  begin
    if ((select iif(res_date > current_date - 90, res_text, '-')
         from hp_anketa_print(0, :c_id, 'Anketa.Main', current_date)
         where ques_code in ('AnketaQuestions.LoanPurpose', 'AnketaQuestions.InfoSource', 'work_place_name', 'work_address_orientir', 'work_address_metro', 'work_staj_last_place', 'work_status', 'work_monthly_income', 'work_income_period', 'work_income_days',
                             'work_business_kind', 'work_job_name', 'work_boss_phone_no', 'work_extra_income', 'work_extra_amount', 'AnketaQuestions.MonetaryObligations', 'AnketaQuestions.DefendantInCourt', 'AnketaQuestions.previous_fio')
         order by 1 rows 1) = '-') then
    begin
      res_error_is_critical = 2;
      res_error_text = 'В Анкете клиента имеются вопросы без ответов';
      suspend;
    end

    if ((select res_rd_code
    from hp_objectprops_get(0, :c_id, -1, 'AnketaQuestions.Nation', current_date)) <> 'Nation.RU')  then
    begin
      res_error_is_critical = 2;
      res_error_text = 'В Анкете клиента ответ на вопрос "Язык общения с клиентом" должен быть "Русский"';
      suspend;
    end 

    if ((select res_rd_code
    from hp_objectprops_get(0, :c_id, -1, 'AnketaQuestions.CanBeSurety', current_date)) <> 'AnketaAnswers.6719')  then
    begin
      res_error_is_critical = 2;
      res_error_text = 'В Анкете клиента ответ на вопрос "Могу предоставить обеспечение поручительством" должен быть "Не требуется"';
      suspend;
    end
  end
end
