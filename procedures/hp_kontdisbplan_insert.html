CREATE PROCEDURE HP_KONTDISBPLAN_INSERT(
    MGR_NAME D_STR_100,
    PLAN_DATE D_DATE,
    PLAN_AMOUNT D_CURRENCY)
AS
declare variable mgr_id d_integer;
declare variable kdp_id d_integer;
declare variable kdp_summa_plan d_currency;
begin
  if (coalesce(mgr_name, '') = '') then
    exit;

  if (plan_date is null) then
    exit;

  select lml_id
  from loanmgrlist
  where lml_name = :mgr_name
  into mgr_id;

  if (mgr_id is null) then
    exception e_message 'Не найден офис/менеджер: ' || mgr_name;

  select kdp_id, kdp_summa_plan
  from kontdisbplan
  where kdp_mgr_id = :mgr_id and
        kdp_date_begin = :plan_date
  into kdp_id, kdp_summa_plan;

  if (kdp_id is not null) then
  begin
    if (kdp_summa_plan <> plan_amount) then
    begin
      update kontdisbplan
      set kdp_summa_plan = :plan_amount
      where kdp_id = :kdp_id;
    end
  end else
  begin
    insert into kontdisbplan (kdp_mgr_id, kdp_date_begin, kdp_summa_plan)
    values (:mgr_id, :plan_date, :plan_amount);
  end
end
