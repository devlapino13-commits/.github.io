CREATE PROCEDURE FSROBOT_OKTELL_MAILING(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0),
    CL_NAME STR_100)
AS
declare variable object_kind integer;
declare variable object_id id_type;
declare variable res_has_code integer;
declare variable res_str_values str_255;
--declare variable reg_email_manager str_255;
declare variable acl_id id_type;
declare variable anketa_code str_255;
declare variable anketa_name str_255;
declare variable question str_255;
declare variable answer str_255;
declare variable str varchar(1000);
declare variable res_date date;
declare variable res_html_tmp blob sub_type 1 segment size 80;
declare variable res_html2 blob sub_type 1 segment size 80;
declare variable res_date_tmp date;
declare variable questionmo str_255;
declare variable answermo str_255;
begin
  acl_id = rep_params;

  res_maillist = '';

  select al.acl_object_kind, al.acl_object_id, rd.rd_code, rd.rd_name
  from actionlist al
  inner join refdata rd on al.acl_from_rd_id = rd.rd_id
  where al.acl_id = :acl_id
  into :object_kind, :object_id, anketa_code, anketa_name;

  res_has_code = null;

  if (anketa_code in ('OKTELL.Projects.WebSite', 'OKTELL.Projects.InCall.Miniapp')) then
  begin
    questionmo = 'AnketaQuestions.OKTELL.WebSite.MOMailSend';
    answermo = 'Results.Yes';

    question = 'AnketaQuestions.OKTELL.WebSite.OfficePlan';
    answer = 'Results.StrValue';
  end

  res_has_code = null;

  select res_has_code
  from hp_objectprops_anketa_get(:object_kind, :object_id, -1, :questionmo, current_date, :answermo)
  into res_has_code;

  if (res_has_code <> 1) then
    exit;

  res_has_code = null;
  res_has_code = null;

  select res_has_code, res_str_values
  from hp_objectprops_anketa_get(:object_kind, :object_id, -1, :question, current_date, :answer)
  into res_has_code, res_str_values;

  res_has_code = coalesce(res_has_code, 0);
  res_str_values = trim(coalesce(res_str_values, ''));

  if ((res_has_code = 1) and
      (res_str_values <> '')) then
  begin
    /*
        select reg.reg_email_manager
        from region reg where reg.reg_name = :RES_STR_VALUES
        and reg.reg_status = 3
        and reg.reg_date_close is null
        into :RES_MAILLIST;
         */
    res_maillist = 'i.gusev@mbfplanet.com'; -- На время тест ОКТЕЛЛ будет моя почта

    res_html2 = '';
    res_html_tmp = '';

    if (res_maillist <> '') then
    begin
      res_subject = anketa_name;

      for select ques_text || ': <b>' || res_text || '</b>', res_date
          from hp_anketa_print(:object_kind, :object_id, :anketa_code, null)
          into :str, res_date_tmp
      do
      begin
        if (res_date is null) then
          res_date = res_date_tmp;
        if (res_date_tmp is not null) then
          res_date = minvalue(res_date, res_date_tmp);
        res_html_tmp = blob_append(res_html_tmp, str, '<br>');
      end
      if (res_html_tmp <> '') then
        res_html2 = blob_append(res_html2, '<font color="#02B316"><b>Инфо от ' || coalesce(right('0' || extract(day from res_date), 2) || '.' || right('0' || extract(month from res_date), 2) || '.' || extract(year from res_date), 'хх.хх.хххх'), '</b></font><br>', res_html_tmp);

      res_body = res_html2;

      suspend;
    end
  end
end
