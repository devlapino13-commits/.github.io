CREATE PROCEDURE UXP$UI_USERDENY(
    UFM_UUID D_UUID,
    UN_ID D_ID)
RETURNS (
    DENIED_ACTIONS D_STR_2048)
AS
declare variable uxa_uuid d_uuid;
declare variable ua_uuid d_uuid;
begin
  denied_actions = '';

  -- для формы "Просмотр бланка" не проверяем права
  if (not exists(select *
  from ux$code, ux$form
  where ufm_uuid = :ufm_uuid and
        ux_uuid = ufm_ux_uuid and
        ux_fcode = 'fm_ux$preblanks')) then
  begin
    for select ua_uuid, ua_uxa_uuid
    from ux$action
    where ua_owner = :ufm_uuid and
          ua_usr_id_field = ''
    union
    select ua_uuid, ua_uxa_uuid
    from ux$frame, ux$action
    where ufr_owner = :ufm_uuid and
          ufr_uuid = ua_owner and
          ua_usr_id_field = ''
    into ua_uuid, uxa_uuid
    do
    begin
      if (uxf$access(un_id, uxa_uuid) <> 1) then
        denied_actions = denied_actions || '[' || uuid_to_char(ua_uuid) || ']';
    end
  end
 
  suspend;
end
