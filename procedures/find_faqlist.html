CREATE PROCEDURE FIND_FAQLIST(
    FM_ID INTEGER,
    TEXT1 VARCHAR(255),
    TEXT2 VARCHAR(255),
    HAS_ANSWER INTEGER,
    LAST_DAYS INTEGER)
RETURNS (
    FL_ID INTEGER,
    FL_FROM_FM_ID INTEGER,
    FL_QUESTION VARCHAR(1024),
    FL_ANSWER VARCHAR(4096),
    FL_AUTHOR VARCHAR(255),
    FL_HAS_ANSWER INTEGER,
    FL_QUESTION_USER_ID DECIMAL(18,0),
    FL_QUESTION_TIME TIMESTAMP,
    FL_ANSWER_USER_ID DECIMAL(18,0),
    FL_ANSWER_TIME TIMESTAMP,
    FL_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80)
AS
begin
  for select fl_id, fl_from_fm_id, fl_question, fl_answer, fl_author, fl_has_answer, fl_question_user_id,
             fl_question_time, fl_answer_user_id, fl_answer_time, fl_blob
      from faqlist
      where ((:fm_id = -1) or
            (fl_from_fm_id = :fm_id)) and
            ((:has_answer = -1) or
            (fl_has_answer = :has_answer)) and
            ((:last_days = -1) or
            ((cast('NOW' as timestamp) - fl_question_time) <= :last_days) or
            ((cast('NOW' as timestamp) - fl_answer_time) <= :last_days)) and
            ((fl_question like '%' || :text1 || '%') or
            (fl_answer like '%' || :text2 || '%'))
      into :fl_id, :fl_from_fm_id, :fl_question, :fl_answer, :fl_author, :fl_has_answer, :fl_question_user_id,
           :fl_question_time, :fl_answer_user_id, :fl_answer_time, :fl_blob
  do
  begin
    suspend;
  end
end
