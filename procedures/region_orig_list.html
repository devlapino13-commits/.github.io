CREATE PROCEDURE REGION_ORIG_LIST(
    ID INTEGER,
    SELF_INCLUDE INTEGER,
    KIND INTEGER,
    IS_DOWN INTEGER)
RETURNS (
    ID_LIST INTEGER,
    ID_PARENT INTEGER,
    PARENT_NAME VARCHAR(50))
AS
begin
  if (is_down = 1) then
  begin
    for with recursive reg_list
        as (select reg_unicode, parent, reg_status
            from region
            where reg_unicode = :id
            union all
            select r.reg_unicode, r.parent, r.reg_status
            from region r
            join reg_list on (reg_list.reg_unicode = r.parent))
        select reg_unicode, parent
        from reg_list
        where ((:self_include = 0 and
              reg_unicode <> :id) or
              (:self_include = 1)) and
              ((:kind = -1) or
              (:kind <> -1 and
              reg_status = :kind))
        into :id_list, :id_parent
    do
    begin
      if (id_parent is not null) then
      begin
        select distinct reg_name
        from region
        where reg_unicode = :id_parent rows 1
        into :parent_name;
      end
      else
      begin
        id_parent = 1000000;
        parent_name = '';
      end
      suspend;
    end
  end
  else
  begin
    for with recursive reg_list
        as (select reg_unicode, parent, reg_status
            from region
            where reg_unicode = :id
            union all
            select r.reg_unicode, r.parent, r.reg_status
            from region r
            join reg_list on (reg_list.parent = r.reg_unicode))
        select reg_unicode, parent
        from reg_list
        where ((:self_include = 0 and
              reg_unicode <> :id) or
              (:self_include = 1)) and
              ((:kind = -1) or
              (:kind <> -1 and
              reg_status = :kind))
        into :id_list, :id_parent
    do
    begin
      if (id_parent is not null) then
      begin
        select distinct reg_name
        from region
        where reg_unicode = :id_parent rows 1
        into :parent_name;
      end
      else
      begin
        id_parent = 1000000;
        parent_name = '';
      end
      suspend;
    end
  end
end
