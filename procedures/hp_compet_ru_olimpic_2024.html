CREATE PROCEDURE HP_COMPET_RU_OLIMPIC_2024(
    CPP_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable res_date_begin date;
declare variable res_date_end date;
declare variable stage integer;
declare variable rd_id integer;
declare variable res_period_id integer;
declare variable res_team_id integer;
declare variable points_1 decimal(18,4) = 0;
declare variable points_2 decimal(18,4) = 0;
declare variable cpr_id integer;
declare variable cpr_group_name varchar(255);
declare variable cpd_comment d_str_4096;
declare variable cpp_orderno integer;
declare variable kind integer;
declare variable cpr_final_points decimal(18,4) = 0;
declare variable points_3 decimal(18,4) = 0;
declare variable points_4 decimal(18,4) = 0;
declare variable kol integer;
declare variable points_5 decimal(18,4) = 0;
declare variable ad_delinq_balance_30 curr_type;
declare variable date_begin date;
declare variable is_month_end smallint;
declare variable ad_delinq_balance_diff_30 curr_type;
declare variable avg_disb_amount curr_type;
declare variable avg_points curr_type;
declare variable lo_panelty_count curr_type;
declare variable total_points curr_type;
declare variable add_points curr_type;
declare variable ad_disb_new_amount curr_type;
declare variable avg_delinq_balance_diff_30 curr_type;
declare variable disb_sum_15 curr_type;
declare variable bal_no_del_diff curr_type;
declare variable bal_no_del_end curr_type;
declare variable bal_no_del_begin curr_type;
declare variable prinyat_bal0 curr_type;
declare variable super_pokazatel curr_type;
declare variable disb_sum_new_cl curr_type;
declare variable kol_lo_end decimal(4,1);
declare variable par30 d_currency;
declare variable rank_par30 d_smallint;
declare variable rank_active_balance d_smallint;
declare variable rank_disb_amount d_smallint;
declare variable date_begin_etap d_date;
declare variable ad_begin_active_balance d_currency;
declare variable ad_active_balance decimal(18,4) = 0;
declare variable ad_delinq_balance_begin1 d_currency;
declare variable ad_delinq_balance_end1 d_currency;
declare variable ad_delinq_balance_begin3 d_currency;
declare variable ad_delinq_balance_end3 d_currency;
declare variable ad_delinq_par_end3 d_currency;
declare variable cpd_add_data_12 d_integer;
declare variable avg_disb_new_amount d_currency;
declare variable cpd_id d_integer;
declare variable dif_portfel_par_0 d_currency;
declare variable dif_portfel_par_30 d_currency;
declare variable avg_portfel_par_0 d_currency;
declare variable avg_portfel_par_30 d_currency;
declare variable cps_from_lml_id d_integer;
declare variable lml_from_reg_id d_integer;
declare variable lml_from_c_id d_id;
declare variable plan_portfel d_currency;
declare variable plan_portfel_all d_currency;
declare variable perc_plan_kp d_currency;
declare variable rep_period d_integer;
declare variable rank_superpokaz d_currency;
declare variable kol_lo_end_all decimal(4,1);
declare variable ad_disb_new_amount_15 d_currency;
declare variable ad_delinq_balance_diff3 d_currency;
declare variable ks_from_k_id d_id;
declare variable ks_from_ol_id d_id;
declare variable in_agent d_currency;
declare variable in_agent_all d_currency;
declare variable dif_active_balance d_currency;
declare variable ad_disb_amount d_currency;
declare variable n d_smallint;
declare variable zalog_qnt d_integer;
declare variable zalog_all d_integer;
declare variable dif_ad_disb_amount d_currency;
begin
  -- Расчет итогов олимпиады РФ
  -- Stage:
  -- 0 - период (декада)
  -- 1 - уровень месяц
  -- 2 - уровень этапа
  -- 3 - Финальные результаты

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end, cpp_orderno
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end, cpp_orderno;

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end

  if (extract(day from res_date_end) not in (10, 20)) then
    is_month_end = 1;
  else
    is_month_end = 0;
  date_begin = res_date_end - extract(day from res_date_end) + 1;

  if (cpp_orderno = 0 and
      res_date_begin = cast('31.01.2024' as date)) then
  begin
    res_date_begin = '01.02.2024';
    res_date_end = '01.02.2024';

    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    -- Формируем в памяти данные по РФ
    -- Тип отчета передаем "olimpic", чтобы система учитывала особенности расчета для олимпиады
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'olimpic')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    -- Группируем данные из памяти по ID команды и ID СВЗ

    for select res_field_1, sum(ad_active_balance)
        from hp_analitica_report('team_id', 'tree_group', 8, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
        where exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        group by 1
            --        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода  -- убрали от 16.3.2023 Альбина попросила
        into :res_team_id, :ad_active_balance
    do
    begin
      if (exists(select *
                 from refdata
                 where rd_id = :res_team_id and
                       rd_is_active = 1)) then
      begin
        insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                  cpr_from_un_id, cpr_comment)
        values (:cpp_id, '', -1, :res_team_id, 0, :un_id, '')
        returning cpr_id
        into :cpr_id;
      end
    end
    suspend;
    exit;
  end

  if (stage = 0) then -- Если уровень Декады
  begin
    -- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);
    -- Формируем в памяти данные по РФ
    -- Тип отчета передаем "olimpic", чтобы система учитывала особенности расчета для олимпиады
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'olimpic')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    -- Группируем данные из памяти по ID команды и ID СВЗ
    for select res_field_1, ad_disb_new_amount /*, --Сумма выдачи новым
               ad_begin_active_balance, --Баланс (КП) на начало
               ad_active_balance, --Баланс (КП)  на конец
               ad_delinq_balance_begin1, --Баланс просроч. на начало  par>=0
               ad_delinq_balance_end1, --Баланс просроч. на конец par>=0
               ad_delinq_balance_begin3, --PAR 30 на начало:
               ad_delinq_balance_end3, --PAR 30 на конец:
               ad_delinq_par_end3 --PAR 30 (%) на конец:*/
        from hp_analitica_report('team_id,', 'tree_group', 8, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
        where exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id) /*and
              res_field_1 = 9933*/
        into :res_team_id, ad_disb_new_amount /*, :ad_begin_active_balance, --ad_begin_active_balance
             ad_active_balance, --ad_active_balance
             ad_delinq_balance_begin1, --ad_delinq_balance_begin1
             ad_delinq_balance_end1, --ad_delinq_balance_end1
             ad_delinq_balance_begin3, --ad_delinq_balance_begin3
             ad_delinq_balance_end3, --ad_delinq_balance_end3
             ad_delinq_par_end3 --ad_delinq_par_end3*/

    do
    begin
      cpd_comment = '';
      kol_lo_end_all = 0;

      -- считаем количество сотрудников внутри команды
      for select cps.cps_from_lml_id
          from competstructure cps
          where cps.cps_from_rd_id = f_get_rd_id('Competitions.Olimpic_2024') and
                cps.cps_team_id = :res_team_id and
                :res_date_end between cps.cps_date_begin and cps.cps_date_end
          into :cps_from_lml_id
      do
      begin
        --суммуриуем планы по менеджерам офисов команды
        lml_from_reg_id = null;
        lml_from_c_id = null;

        select lml.lml_from_reg_id, lml.lml_from_c_id
        from loanmgrlist lml
        where lml.lml_id = :cps_from_lml_id
        into :lml_from_reg_id, lml_from_c_id;

        --  фактическое кол-во Випс на дату
        kol_lo_end = 0;

        select sum(iif(jk_name = 'СВЗ 0,5', 0.5, 1))
        from hr_pstatus, joblist, jobkind
        where :res_date_end between ps_date_begin and ps_date_end and
              ps_from_reg_id = :lml_from_reg_id and
              ps_is_working = 1 and
              ps_status = 1 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code = 'JobSystem.LO_Standart' and
              exists(select *
                     from hp_employer_in_dekret(ps_from_c_id, :res_date_end)
                     where is_decret = 0) and
              :cps_from_lml_id = f_lo_mgr_id(null, :res_date_end, ps_from_c_id, ps_from_reg_id)
        into kol_lo_end;

        kol_lo_end_all = kol_lo_end_all + kol_lo_end;
      end

      cpd_comment = cpd_comment || 'Кол-во ВИПС на отчетную дату: <b>' || round(kol_lo_end_all, 2) || '</b><br>';
      cpd_comment = cpd_comment || 'Сумма выдач новым клиентам команды за декаду: <b>' || round(ad_disb_new_amount, 0) || ' р.</b><br>';

      -- Находим нужную строку команды за этот период
      cpr_id = null;
      select cpr_id
      from competresult
      where cpr_from_cpp_id = :cpp_id and
            cpr_object_kind = -1 and
            cpr_object_id = :res_team_id
      into :cpr_id;

      if (cpr_id is null) then -- если такой строки нет, создаем новую
      begin
        select cpr_group_name
        from competresult, competperiod
        where cpr_from_cpp_id = cpp_id and
              cpp_from_rd_id = :rd_id and
              cpr_object_kind = -1 and
              cpr_object_id = :res_team_id and
              cpr_group_name <> '' and
              cpp_orderno < :cpp_orderno
        order by cpp_orderno desc
        rows 1
        into :cpr_group_name;

        if (coalesce(cpr_group_name, '') = '') then
        begin
          select rd_memo
          from refdata
          where rd_id = :res_team_id
          into :cpr_group_name;
        end

        insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                  cpr_from_un_id, cpr_comment)
        values (:cpp_id, :cpr_group_name, -1, :res_team_id, 0, :un_id, '')
        returning cpr_id
        into :cpr_id;
      end

      insert into competdetails (cpd_from_cpr_id, cpd_object_kind, cpd_object_id, cpd_add_data_5, cpd_add_data_12,
                                 cpd_comment)
      values (:cpr_id, -1, :res_team_id, :ad_disb_new_amount, :kol_lo_end_all, left(:cpd_comment, 4032));

    end
    ----------------------------------------------------------------------------------------------------------------------отсюда нужно продолжать ----------------Улан
    if (is_month_end = 1) then
    begin
      n = 1;
      -- Формируем в памяти данные по РФ
      -- Тип отчета передаем "olimpic", чтобы система учитывала особенности расчета для олимпиады
      execute procedure hp_analitica_create_period(1000000, :date_begin, :res_date_end, 1, null, null, 30, 0,
          'standart')
          returning_values :res_period_id;

      -- Группируем данные из памяти по ID команды и ID СВЗ
      for select res_field_1, ad_disb_new_amount, --Сумма выдачи новым
                 ad_disb_amount, --Сумма выдачи
                 coalesce(ad_begin_active_balance, 0), --Баланс (КП) на начало
                 coalesce(ad_active_balance, 0), --Баланс (КП)  на конец
                 coalesce(ad_delinq_balance_begin1, 0), --Баланс просроч. на начало  par>=0
                 coalesce(ad_delinq_balance_end1, 0), --Баланс просроч. на конец par>=0
                 coalesce(ad_delinq_balance_begin3, 0), --PAR 30 на начало:
                 coalesce(ad_delinq_balance_end3, 0), --PAR 30 на конец:
                 coalesce(ad_delinq_par_end3, 0), --PAR 30 (%) на конец:
                 coalesce(ad_delinq_balance_diff3, 0)
          from hp_analitica_report('team_id,', 'tree_group', 8, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
          where exists(select *
                       from refdata
                       where rd_main_code = 'StaffTeamList' and
                             rd_id = res_field_1 and
                             rd_tag = :rd_id) /*and
                res_field_1 = 9933*/
          into :res_team_id, ad_disb_new_amount, :ad_disb_amount, :ad_begin_active_balance, --ad_begin_active_balance
               ad_active_balance, --ad_active_balance
               ad_delinq_balance_begin1, --ad_delinq_balance_begin1
               ad_delinq_balance_end1, --ad_delinq_balance_end1
               ad_delinq_balance_begin3, --ad_delinq_balance_begin3
               ad_delinq_balance_end3, --ad_delinq_balance_end3
               ad_delinq_par_end3, --ad_delinq_par_end3
               ad_delinq_balance_diff3
      do
      begin
        cpd_comment = '';
        cpd_id = null;
        dif_portfel_par_0 = 0;
        dif_portfel_par_30 = 0;

        --изменение портфеля без просрочки
        dif_portfel_par_0 = (ad_active_balance - ad_delinq_balance_end1) - (ad_begin_active_balance - ad_delinq_balance_begin1);

        --Изменение баланса PAR30:
        dif_portfel_par_30 = ad_delinq_balance_end3 - ad_delinq_balance_begin3;

        --выолнение плана по КП
        plan_portfel_all = 0;
        rep_period = extract(year from :res_date_end) || right((extract(month from :res_date_end) + 100), 2);
        for select cps.cps_from_lml_id
            from competstructure cps
            where cps.cps_from_rd_id = f_get_rd_id('Competitions.Olimpic_2024') and
                  cps.cps_team_id = :res_team_id and
                  :res_date_end between cps.cps_date_begin and cps.cps_date_end
            into :cps_from_lml_id
        do
        begin
          --суммуриуем планы по менеджерам офисов команды
          lml_from_reg_id = null;
          lml_from_c_id = null;
          plan_portfel = 0;

          select lml.lml_from_reg_id, lml.lml_from_c_id
          from loanmgrlist lml
          where lml.lml_id = :cps_from_lml_id
          into :lml_from_reg_id, lml_from_c_id;

          --если lml_from_c_id пустое значит офис с одним менеджером, передаем любое число кроме 0 (стоит блок на 0)
          lml_from_c_id = coalesce(lml_from_c_id, 999);

          select round(sum(po_amount), 2)
          from hp_get_planoper_mgr_info(:lml_from_reg_id, :lml_from_c_id, 'PlanIndicators.Portfolio_amount', :res_date_end, :rep_period)
          into :plan_portfel;

          plan_portfel_all = plan_portfel_all + plan_portfel;
        end
        perc_plan_kp = 0;
        if (plan_portfel_all > 0) then
          perc_plan_kp = (ad_active_balance / plan_portfel_all) * 100;

        cpd_comment = cpd_comment || 'КП без просрочки на начало: <b>' || round(ad_delinq_balance_begin1, 2) || ' р.</b><br>';
        cpd_comment = cpd_comment || 'КП без просрочки на конец : <b>' || round(ad_delinq_balance_end1, 2) || ' р.</b><br>';
        cpd_comment = cpd_comment || 'Значение PAR30 в % на конец периода: <b>' || round(ad_delinq_par_end3, 2) || ' %.</b><br>';
        cpd_comment = cpd_comment || 'Баланс PAR30 на начало: <b>' || round(ad_delinq_balance_begin3, 2) || ' р.</b><br>';
        cpd_comment = cpd_comment || 'Баланс PAR30 на конец: <b>' || round(ad_delinq_balance_end3, 2) || ' р.</b><br>';
        cpd_comment = cpd_comment || 'План КП : <b>' || round(plan_portfel_all, 2) || ' р.</b><br>';
        cpd_comment = cpd_comment || 'Процент выполнения плана по КП : <b>' || round(perc_plan_kp, 2) || ' %.</b><br>';

        if (res_date_end < '1.5.24') then
        begin
          delete from temp_objlist t
          where t.tol_kind = 'super_pokazatel_in_agent_042024';

          for select ks_from_k_id, ks_from_ol_id
              from kontstate, kontract
              where k_id = ks_from_k_id and
                    ks_date_begin between :date_begin and :res_date_end and
                    ks_state = 1 and
                    ks_ol_is_delete = 0
              into :ks_from_k_id, ks_from_ol_id
          do
          begin
            in_agent = 0;
            if (exists(select *
                       from operagr
                       where ogr_from_ol_id = :ks_from_ol_id and
                             ogr_object_id is not null and
                             ogr_ol_is_active = 1)) then
            begin
              in_agent = 1;
            end

            insert into temp_objlist (tol_kind, tol_object_kind, tol_object_id, tol_status)
            values ('super_pokazatel_in_agent_042024', 9, :ks_from_k_id, :in_agent);
          end

          in_agent_all = 0;

          for select cps.cps_from_lml_id
              from competstructure cps
              where cps.cps_from_rd_id = f_get_rd_id('Competitions.Olimpic_2024') and
                    cps.cps_team_id = :res_team_id and
                    :res_date_end between cps.cps_date_begin and cps.cps_date_end
              into :cps_from_lml_id
          do
          begin
            in_agent = 0;
            select sum(t.tol_status)
            from temp_objlist t
            where t.tol_kind = 'super_pokazatel_in_agent_042024' and
                  t.tol_object_kind = 9 and
                  f_lo_mgr_id(t.tol_object_id, :res_date_end, null, null) = :cps_from_lml_id
            into :in_agent;

            in_agent_all = in_agent_all + in_agent;
          end
        end

        if (res_date_end < '1.10.24') then
        begin
          if (n = 1) then
          begin
            n = n + 1;
            delete from temp_objlist t
            where t.tol_kind = 'super_pokazatel_zalog_092024';

            for select ks_from_k_id, ks_from_ol_id
                from kontstate, kontract k, product pr
                where k_id = ks_from_k_id and
                      pr.pr_id = k.k_from_pr_id and
                      pr.pr_code containing '.zalog.' and
                      ks_date_begin between :date_begin and :res_date_end and
                      ks_state = 1 and
                      ks_ol_is_delete = 0
                into :ks_from_k_id, ks_from_ol_id
            do
            begin
              insert into temp_objlist (tol_kind, tol_object_kind, tol_object_id, tol_status)
              values ('super_pokazatel_zalog_092024', 9, :ks_from_k_id, 1);
            end
          end
          zalog_all = 0;

          for select cps.cps_from_lml_id
              from competstructure cps
              where cps.cps_from_rd_id = f_get_rd_id('Competitions.Olimpic_2024') and
                    cps.cps_team_id = :res_team_id and
                    :res_date_end between cps.cps_date_begin and cps.cps_date_end
              into :cps_from_lml_id
          do
          begin
            zalog_qnt = 0;
            select coalesce(sum(t.tol_status), 0)
            from temp_objlist t
            where t.tol_kind = 'super_pokazatel_zalog_092024' and
                  t.tol_object_kind = 9 and
                  f_lo_mgr_id(t.tol_object_id, :res_date_end, null, null) = :cps_from_lml_id
            into :zalog_qnt;

            zalog_all = zalog_all + zalog_qnt;
          end
        end

        super_pokazatel = 0;
        if (res_date_end < '1.11.23') then
        begin
          super_pokazatel = bal_no_del_diff;
          cpd_comment = cpd_comment || 'Рост портфеля без просрочек: <b>' || round(bal_no_del_diff, 2) || '</b><br>';
        end
        else
        if (res_date_end < '1.3.24') then
        begin
