CREATE PROCEDURE AAREP_CLIENTS_PARALLEL_LOAN(
    REP_DATE D_DATE = current_date)
RETURNS (
    OFFICE_ID D_INTEGER,
    OFFICE_NAME D_STR_CODE,
    OFFICE_PHONE D_STR_CODE,
    K_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    OKS_BALANCE D_CURRENCY,
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    C_PHONE D_STR_100,
    SCORE D_INTEGER,
    MAX_SUM D_CURRENCY,
    AVAILABLE_SUM D_CURRENCY)
AS
declare variable cnt d_integer;
declare variable k_id_tmp d_id;
declare variable sh_date d_date;
declare variable sh_sum d_currency;
declare variable ok_date d_date;
declare variable ok_sum d_currency;
declare variable oks_is_delinq d_bool;
declare variable oks_delinq_day d_integer;
declare variable last_year_begin d_date;
begin
  -- RFCAZ: 284, 350
  -- процедура возвращает список активных клиентов и номера их телефонов для СМС-рассылки
  -- клиенты должны подходить под условия для получения параллельного кредита:
    -- 1) У клиента нет активной просрочки по текущему кредиту
    -- 2) По текущему кредиту полностью оплачено не менее 4 платежей
    -- 3) Клиент не допускал просрочек по кредитам в Компании за последний год, при этом допускается не более одной просрочки до 3-х дней включительно
    -- 4) На одного клиента не может быть более двух кредитов в Компании (основной и параллельный)
    -- 5) У клиента 8 и более баллов по программе лояльности
    -- 6) Общий баланс активного кредита и получаемого параллельного кредита не должен превышать максимально допустимый долг согласно накопленным баллам


  if (f_get_db_id() <> 94401) then
    exit;

  -- Перебираем активные контракты
  for select ks_from_k_id,
             kc_from_c_id,
             ks_region_id
      from kontstate
      inner join knclient on kc_from_k_id = ks_from_k_id
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date between ks_date_begin and ks_date_end and
            kc_amount_approved > 0
      into k_id,
           c_id,
           office_id
  do
  begin
    c_phone = f_cont_info('clients', :c_id, 'cell_main', 'ok');
    if (coalesce(c_phone, '') = '') then
      continue; -- у клиента нет верифицированного основного номера телефона, пропускаем

    --***-- проверка условия: 1) У клиента нет активной просрочки по текущему кредиту
    oks_is_delinq = 0;
    oks_balance = 0;
    select oks_balance,
           oks_is_delinq
    from oksummary
    where :rep_date between oks_date_begin and oks_date_end and
          oks_from_k_id = :k_id
    into oks_balance, -- баланс активного кредита на текущий день
         oks_is_delinq; -- признак просрочки: кредит просрочен, если oks_is_delinq = 1

    if (oks_balance is null) then
      oks_balance = 0;

    if (oks_is_delinq = 1) then
      continue; -- текущий кредит с активной просрочкой

    --***-- проверка условия: 2) По текущему кредиту полностью оплачено не менее 4 платежей
    cnt = 0;
    select count(*)
    from schedule -- график погашений из деталей контракта
    where sh_date < :rep_date and
          (sh_os + sh_perc) > 0 and
          sh_from_k_id = :k_id
    into cnt;

    if (coalesce(cnt, 0) < 4) then
      continue; -- по графику 4 погашения еще не наступили

    cnt = 0;
    for select ok_sh_date,
               sum_of_ok_os + sum_of_ok_perc
        from kontract_full_oper_list_detail(:k_id, :rep_date) -- график погашений из отчета "Детальное распределение погашений"
        where ok_sh_date < :rep_date and
              ok_type = 4 and
              (sum_of_ok_os + sum_of_ok_perc) < 0
        into sh_date,
             sh_sum
    do
    begin
      ok_date = null;
      ok_sum = 0;

      select max(ok_ol_date),
             sum(ok_os + ok_perc)
      from operkred
      where ok_ol_is_active = 1 and
            ok_type = 4 and
            ok_from_k_id = :k_id and
            ok_sh_date = :sh_date
      into ok_date, -- дата последнего фактического платежа по плановому погашению
           ok_sum; -- сумма платежа (ОС + %)

      if (sh_date - ok_date <= 10 and -- платеж не ранее чем за 10 дней до даты платежа по графику
          abs(sh_sum) + ok_sum <= 5) then -- платеж полностью закрыл сумму погашения по графику (ok_sum отр. число, а недоплата до 5 манат включительно легитимно, RFCAZ-286)
        cnt = cnt + 1; -- "правильные" платежи
    end

    if (cnt < 4) then
      continue; -- по текущему кредиту еще не произведены 4 погашения

    --***-- проверка условия: 3) Клиент не допускал просрочек по кредитам в Компании за последний год, при этом допускается не более одной просрочки до 3-х дней включительно
    last_year_begin = dateadd(year, -1, :rep_date); -- начало последнего года от текущей даты
    cnt = 0;
    k_id_tmp = null;
    for select ks_from_k_id
        from kontstate
        inner join knclient on kc_from_k_id = ks_from_k_id
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_date_end >= :last_year_begin and -- активные кредиты за последний год
              kc_amount_approved > 0 and
              kc_from_c_id = :c_id -- фильтр по клиенту
        into k_id_tmp
    do
    begin
      oks_delinq_day = 0;
      for select oks_delinq_day
          from oksummary
          where oks_delinq_day > 0 and -- просроченные платежи
                oks_delinq_date between :last_year_begin and :rep_date and -- за последний год
                oks_from_k_id = :k_id_tmp -- фильтр по контракту клиента
          into oks_delinq_day -- кол-во просроченных дней
      do
      begin
        cnt = cnt + 1; -- кол-во просрочек
        if (oks_delinq_day > 3) then
          cnt = cnt + 1; -- просрочка более 3-х дней считается как 2 просрочки до 3-х дней
      end
    end

    if (cnt > 1) then
      continue; -- проверка на просрочки не пройдена, параллельный кредит не доступен

    --***-- проверка условия: 4) На одного клиента не может быть более двух кредитов в Компании (основной и параллельный)
    cnt = 0;
    select count(*)
    from kontstate
    inner join knclient on kc_from_k_id = ks_from_k_id
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date between ks_date_begin and ks_date_end and
          kc_amount_approved > 0 and
          kc_from_c_id = :c_id
    into cnt;

    if (coalesce(cnt, 0) >= 2) then
      continue; -- у клиента уже есть 2 кредита в Компании, более быть не может

    --***-- проверка условия: 5) У клиента 8 и более баллов по программе лояльности
    score = f_point(c_id, rep_date);

    if (score < 8) then
      continue; -- у клиента менее 8 баллов, следовательно параллельный кредит не доступен

    --***-- сумма максимально допустимого долга клиента перед компанией (общий долг по основному и параллельному кредиту)
    max_sum = case
                when score between 0 and 2 then 1800
                when score between 3 and 5 then 2200
                when score between 6 and 11 then 3000
                when score between 12 and 17 then 3500
                when score between 18 and 23 then 4000
                when score between 24 and 29 then 4500
                when score between 30 and 35 then 5000
                when score between 36 and 41 then 5500
                when score between 42 and 47 then 6000
                when score >= 48 then 6500
                else 0
              end;

    --***-- обеспечение условия: 6) Общий баланс активного кредита и получаемого параллельного кредита не должен превышать максимально допустимый долг согласно накопленным баллам
    if (oks_balance = 0) then
      available_sum = max_sum; -- если нет активного кредита, то теоретически доступная сумма параллельного кредита = сумме максимально допустимого долга
    else
      available_sum = trunc(max_sum - oks_balance, -2); -- округленная до тысяч в меньшую сторону доступная сумма параллельного кредита

    if (available_sum < 1000) then
      continue; -- если доступная сумма меньше тысячи, то СМС не отправляем

    --***-- Заполняем оставшиеся выходные параметры
    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;

    select k_kontract_no
    from kontract
    where k_id = :k_id
    into k_kontract_no;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :office_id
    into office_name;

    office_phone = case
                     when office_name = '0203 Nəsimi' then '050-246-61-32, 050-246-61-55'
                     when office_name = '0204 Sumqayıt' then '050-246-60-74, 050-246-61-23'
                     when office_name = '0301 Ağstafa' then '050-246-62-80, 050-246-62-78'
                     when office_name = '0302 Tovuz' then '055-205-02-77, 050-205-49-69'
                     when office_name = '0303 Qazax' then '055-220-29-11, 050-246-62-75'
                     when office_name = '0401 Mingəçevir' then '050-246-61-30, 050-246-60-65, 050-246-61-08'
                     when office_name = '0402 Bərdə' then '050-246-61-34, 050-246-61-57'
                     when office_name = '0403 Tərtər' then '055-220-22-16, 051-201-62-13'
                     when office_name = '0404 Ağdaş' then '050-246-62-41, 050-246-60-72'
                     when office_name = '0405 Goranboy' then '050-246-61-44, 050-246-61-39'
                     when office_name = '0501 Lənkəran' then '055-207-78-55, 055-255-16-26'
                     when office_name = '0502 Masallı' then '050-246-61-43, 055-207-70-38'
                     when office_name = '0503 Astara' then '050-246-60-98, 050-246-60-28'
                     when office_name = '0504 Cəlilabad' then '055-201-29-12, 050-246-60-84'
                     when office_name = '0506 Biləsuvar' then '050-246-61-72'
                     when office_name = '0601 Zaqatala' then '050-246-62-13, 050-246-60-67'
                     when office_name = '0602 Balakən' then '050-246-61-09, 055-220-97-96'
                     when office_name = '0603 Qax' then '050-246-61-69, 055-220-59-20'
                     when office_name = '0701 İsmayıllı' then '050 246 60 40, 050-246-60-83'
                     when office_name = '0702 Qəbələ' then '050 246 62 98, 050 246 60 85'
                     when office_name = '0704 Ağsu' then '050 246 62 46, 050 246 62 06'
                     when office_name = '0801 Quba' then '050-246-61-89, 050-246-61-14'
                     when office_name = '0803 Xaҫmaz' then '055-214-34-67, 051-206-18-84'
                     when office_name = '0804 Xudat' then '050-246-61-28, 055-207-75-83'
                     else ''
                   end;

    suspend;
  end
end
