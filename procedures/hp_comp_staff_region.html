CREATE PROCEDURE HP_COMP_STAFF_REGION(
    C_ID D_ID)
RETURNS (
    REGION_NAME VARCHAR(60),
    DATE_BEGIN DATE,
    DATE_END DATE,
    CHL_ID D_INTEGER)
AS
declare variable reg_id d_integer;
begin
  for select reg.reg_unicode, reg.reg_code || ' ' || reg.reg_name, ps.ps_date_begin, ps.ps_date_end
      from hr_pstatus ps
      inner join region reg on ps.ps_from_reg_id = reg.reg_unicode
      where ps.ps_from_c_id = :c_id and
            current_date between ps.ps_date_begin and ps.ps_date_end
      into :reg_id, :region_name, date_begin, date_end
  do
  begin
    if (not exists(select *
                   from compet_hr_list chl
                   where chl.chl_c_id = :c_id and
                         chl.chl_reg_id = :reg_id)) then
      suspend;
  end

  for select reg.reg_code || ' ' || reg.reg_name, chl.chl_date_begin, chl.chl_date_end, chl.chl_id
      from compet_hr_list chl
      inner join region reg on chl.chl_reg_id = reg.reg_unicode
      where chl.chl_c_id = :c_id
      into :region_name, :date_begin, :date_end, :chl_id
  do
  begin
    suspend;
  end

end
