CREATE PROCEDURE CM_KG_0143_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id id_type;
declare variable oks_balance d_currency;
declare variable total_debt d_currency;
declare variable k_disb_date d_date;
declare variable kc_amount_approved d_currency;
declare variable pr_code d_str_100;
begin
  -- задача 143-блока - не пропустить клиента с общим совокупным балансом более 500 тыс. сом (RFCKG: 3578, 3748)

  res_error_is_critical = 2;
  res_error_text = '';

  for select k_disb_date,
             kc_from_c_id,
             kc_amount_approved,
             pr_code
      from kontract
      inner join knclient on kc_from_k_id = k_id
      inner join product on pr_id = k_from_pr_id
      where k_id = :k_id and
            kc_amount_approved > 0
      into k_disb_date,
           c_id,
           kc_amount_approved,
           pr_code
  do
  begin
    if (pr_code containing '.kardhasan.' or
        pr_code containing '.bittamlik.' or
        (pr_code containing '.point_ignore.' and
         pr_code containing '.insurance.')) then
      exit;

    oks_balance = 0;
    select sum(oks_balance * (kc_amount_approved / k_amount_approved))
    from oksummary
    inner join knclient on kc_from_k_id = oks_from_k_id
    inner join kontract on k_id = oks_from_k_id
    where k_id <> :k_id and -- RFCKG-3616
          k_disb_date <= :k_disb_date and
          kc_amount_approved > 0 and
          kc_from_c_id = :c_id and
          oks_close_state = 0 and
          :k_disb_date between oks_date_begin and oks_date_end
    into oks_balance; -- общий активный долг

    total_debt = coalesce(oks_balance, 0) + kc_amount_approved; -- общий совокупный долг: активный долг + сумма заявки

    if (total_debt > 500000) then
    begin
      select c_name_fio
      from clients
      where c_id = :c_id
      into res_info_text;

      res_error_text = 'Общий совокупный баланс (' || round(total_debt) || ') превышает максимально допустимое значение (500 тыс.)';
      suspend;
    end
  end
end
