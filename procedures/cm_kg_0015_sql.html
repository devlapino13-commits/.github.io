CREATE PROCEDURE CM_KG_0015_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID ID_TYPE;
declare variable MA_PERCENT PERC_TYPE;
declare variable KF_ID ID_TYPE;
declare variable K_KREDLINE_END date;
declare variable K_ID ID_TYPE;
declare variable REP_DATE date;
declare variable C_CURR_BONUS_STATUS integer;
declare variable MAX_DELINQ_DAY integer;
declare variable K_KONTRACT_NO STR_100;
declare variable K_BONUS_TEXT STR_100;
declare variable DELINQ_DAY_REPAYMENT integer;
declare variable OKS_DELINQ_DAY integer;
declare variable MA_LIMIT integer;
begin
    if (current_date between '17.08.2015' and '31.08.2015') then -- период действия акции  по приказу №66-08-15
    begin
        for select maa.ma_percent,
                   maa.ma_from_kf_id,
                   maa.ma_date_app
        from miniapp maa
        inner join product pr on pr.pr_id =  ma_from_pr_id
        where maa.ma_id = :ma_id
        and pr.pr_loan_type = 0 -- только групповой
        and maa.ma_date_end='31.12.9999'
        and ma_percent = 27 -- по приказу % = 27
        and maa.ma_is_active = 1
        into :ma_percent,
             :kf_id,
             :rep_date
        do
        begin
            if (:rep_date not between '17.08.2015' and '31.08.2015') then
            begin
                res_error_is_critical=1;
                res_error_text = 'Срок действия акции с 17.08.2015 по 31.08.2015';
                Suspend;     
            end
            else
            begin
                --  если уже имется 500 активных заявок по данной акции, тогда, выдаю ошибку
                ma_limit = null;
                select count(maa.ma_id)
                from miniapp maa
                where  maa.ma_date_end='31.12.9999'
                and ma_percent = 27 -- по приказу % = 27
                and maa.ma_is_active = 1
                and maa.ma_id <> :ma_id
                and maa.ma_date_app between cast('17.08.2015' as date) and cast('31.08.2015' as date)
                into :ma_limit;
    
                ma_limit = coalesce(ma_limit,0);
                if (ma_limit >= 500) then
                begin
                    res_error_is_critical=1;
                    res_error_text = 'По пакции уже зарегистрировано 500 групп';
                    Suspend;                
                end
                else
                begin
                    k_kredline_end = null;
                    k_id = null;
        
                    /*
                    -- проверяю, есть ли активный займ
                    select ks.ks_from_k_id, kont.k_kredline_end, kont.k_kontract_no
                    from kontstate ks
                    inner join kontract kont on kont.k_id = ks.ks_from_k_id
                    where ks.ks_state = 1
                    and kont.k_from_kf_id =:kf_id
                    and :rep_date between  ks.ks_date_begin and ks.ks_date_end
                    and ks.ks_ol_is_delete = 0
                    order by kont.k_kredline_end
                    rows 1
                    into :k_id, :k_kredline_end, :k_kontract_no;
                    */
--                  Предыдущий исправлено на следующее т.е. нужно брать только основные групповые кредиты
--                  Изменен по письму Алтын от 17.08.2015 тема письма "RE: приказ - Акция ко Дню независимости"
                    select k_id, k_kredline_end, k_kontract_no
                    from kontstate, kontract, product
                    where ks_ol_is_delete = 0
                    and ks_state = 1
                    and :rep_date between ks_date_begin and ks_date_end
                    and ks_from_k_id = k_id
                    and k_from_kf_id = :kf_id
                    and k_kredline_end is not null
                    and k_from_pr_id = pr_id
                    and pr_product_kind = 1 -- Брать только групповые кредиты
                    order by k_kredline_end
                    rows 1
                    into k_id, k_kredline_end, k_kontract_no;
        
                    if (k_id is not null) then
                    begin  -- если есть активный займ
                        if (k_kredline_end < '31.10.2015') then
                        begin
                            -- >> определяю статус группы
                            select C_CURR_BONUS_STATUS
                            from HP_KONTSTATE_LOAN_STATUS(0, 0, :ma_id, :rep_date, null)
                            into :C_CURR_BONUS_STATUS;   
        
                            if (C_CURR_BONUS_STATUS = -1)  then k_bonus_text='Не определен';
                            if (C_CURR_BONUS_STATUS = 0)   then k_bonus_text='Без статуса';
                            if (C_CURR_BONUS_STATUS = 1)   then k_bonus_text='Бронзовый';
                            if (C_CURR_BONUS_STATUS = 2)   then k_bonus_text='Серебрянный';
                            -- <<
        
                            if (C_CURR_BONUS_STATUS >= 3) then
                            begin
                                max_delinq_day = 0; -- Максимальный срок задержки кредита
                                delinq_day_repayment = 0;
                                oks_delinq_day = 0;
                        
                                select max(ok.ok_ol_date - ok_sh_date) from operkred ok
                                where ok.ok_ol_is_active = 1
                                and ok.ok_ol_date  <= :REP_DATE
                                and ok.ok_ol_date  > ok_sh_date
                                and ok_type < 7
                                and ok_from_k_id = :k_id
                                and ok_os + ok_perc < 0
                                and ok_from_c_id=0
                                into delinq_day_repayment;
                        
                                select (cast (:rep_date as date) - oks_delinq_date) oks_delinq_day
                                from oksummary oks
                                where oks.oks_from_k_id = :k_id
                                and oks_close_state = 0
                                and :rep_date between oks_date_begin and oks_date_end
                                and cast(:rep_date as date) - oks_delinq_date >= 0
                                into oks_delinq_day;
                        
                                if (delinq_day_repayment is null) then delinq_day_repayment = 0;
                                if (oks_delinq_day is null) then oks_delinq_day= 0;
                        
                                if (oks_delinq_day > delinq_day_repayment) then
                                     MAX_DELINQ_DAY = oks_delinq_day;
                                else
                                     MAX_DELINQ_DAY = delinq_day_repayment;
                    
                                if (max_delinq_day > 0) then
                                begin
                                    -- Ошибка Не должно быть просрочек
                                    res_error_is_critical=1;
                                    res_error_text='У группы в активной кредите имеются просрочки. Договор № '||:k_kontract_no;
                                    Suspend;
                                end
                             end
                             else
                             begin
                                -- Ошибка Акция только для групп со статусом Золотой , Брил
                                res_error_is_critical=1;
                                res_error_text='У группы статус "'||:k_bonus_text||'". Акции подлежат группы со статусом "Золотой", "Бриллиантовый"';
                                Suspend;
                             end
                        end
                        else
                        begin
                            -- Ошибка Дата последнего погашения должна выпадать до 31 октября 2015
                            res_error_is_critical=1;
                            res_error_text='Дата последнего платежа должна выпадать до 31.10.2015. Договор № '||:k_kontract_no;
                            Suspend;
                        end
                    end
                    else  --  если нет активного займа, проверяю закрытый заем, сортирую по дате выдачи
                    begin
                        k_id = null;
        
                        select kont.k_id, kont.k_kontract_no
                        from kontstate ks
                        inner join kontract kont on kont.k_id = ks.ks_from_k_id
                        where ks.ks_ol_is_delete = 0
                        and ks.ks_state = 2
                        and ks.ks_date_begin <= :rep_date
                        and kont.k_from_kf_id = :kf_id
                        order by kont.k_disb_date desc
                        rows 1
                        into :k_id, :k_kontract_no;
        
                        if (k_id is null) then
                        begin
                            -- Предыдущий займ группы не определен
                            res_error_is_critical=1;
                            res_error_text='Предыдущий займ группы не определен';
                            Suspend;
                        end
                        else
                        begin
                            max_delinq_day =0;
        
                            select max(ok.ok_ol_date - ok_sh_date)
                            from operkred ok
                            where ok.ok_ol_is_active = 1
                            and ok.ok_ol_date  <= :REP_DATE
                            and ok.ok_ol_date  > ok_sh_date
                            and ok_type < 7
                            and ok_from_k_id = :k_id
                            and ok_os + ok_perc < 0
                            and ok_from_c_id=0
                            into MAX_DELINQ_DAY;
            
                            if (MAX_DELINQ_DAY is null) then MAX_DELINQ_DAY = 0;
            
                            if (max_delinq_day <= 0) then
                            begin
                                /* >> определяю статус группы                           */
                                select C_CURR_BONUS_STATUS
                                from HP_KONTSTATE_LOAN_STATUS(0, 0, :ma_id, :rep_date, null)
                                into :C_CURR_BONUS_STATUS;
                            
                                if (C_CURR_BONUS_STATUS = -1)  then k_bonus_text='Не определен';
                                if (C_CURR_BONUS_STATUS = 0)   then k_bonus_text='Без статуса';
                                if (C_CURR_BONUS_STATUS = 1)   then k_bonus_text='Бронзовый';
                                if (C_CURR_BONUS_STATUS = 2)   then k_bonus_text='Серебрянный';
                                /* <<  */
                                if (C_CURR_BONUS_STATUS < 3) then
                                begin
                                    res_error_is_critical=1;
                                    res_error_text='У группы статус "'||:k_bonus_text||'". Акции подлежат группы со статусом "Золотой", "Бриллиантовый"';
                                    Suspend;
                                end
                            end
                            else
                            begin
                                -- Ошибка Не должно быть просрочек
                                res_error_is_critical=1;
                                res_error_text='У группы в последнем цикле были просрочки. (Договора № '||:k_kontract_no||')';
                                Suspend;
                            end
                        end
                    end
                end
            end
        end
    end
end
