CREATE PROCEDURE HP_GET_PLACES(
    PL_PARENT_ID DECIMAL(18,0),
    FIND_TXT VARCHAR(100))
RETURNS (
    PL_ID DECIMAL(18,0),
    PL_NAME VARCHAR(100),
    PL_NAME_KOMITET VARCHAR(100),
    PL_KIND VARCHAR(100),
    PL_CAN_SELECT INTEGER)
AS
declare variable pl_id_tmp decimal(18,0);
declare variable plt_show_in_address integer;
declare variable need_suspend integer;
declare variable rec_count integer;
declare variable find_str1 varchar(255);
declare variable find_str2 varchar(255);
begin
  find_txt = coalesce(find_txt, '');

  if (pl_parent_id is null) then
  begin
    select count(*), min(pl_id)
    from places
    where parent is null
    into :rec_count, pl_parent_id;

    if (rec_count is null) then
      rec_count = 0;
    if (rec_count > 1) then
    begin
      for select pl_id, pl_name, pl_name_komitet, plt_name, plt_can_select, plt_show_in_address
          from places, placetype
          where parent is null and
                pl_status = plt_id
          order by pl_name
          into :pl_id, pl_name, pl_name_komitet, pl_kind, pl_can_select, plt_show_in_address
      do
      begin
        suspend;
      end
      exit;
    end
  end

  for select pl_id, pl_name, pl_name_komitet, plt_name, plt_can_select, plt_show_in_address
      from places, placetype
      where parent = :pl_parent_id and
            pl_status = plt_id
      into :pl_id_tmp, pl_name, pl_name_komitet, pl_kind, pl_can_select, plt_show_in_address
  do
  begin
    need_suspend = 0;
    if (plt_show_in_address = 0) then
    begin
      for select pl_id, pl_name, pl_name_komitet, plt_name, plt_can_select
          from places, placetype
          where parent = :pl_id_tmp and
                pl_status = plt_id
          into :pl_id, pl_name, pl_name_komitet, pl_kind, pl_can_select
      do
      begin
        need_suspend = 0;
        if (find_txt = '') then
          need_suspend = 1;
        else
        if (position(pl_name, find_txt) > 0 or position(find_txt, pl_name) > 0) then
          need_suspend = 1;
        else
        if (f_levenstain_is_same(:pl_name, :find_txt) = 1) then
          need_suspend = 1;
        else
          need_suspend = f_levenstain_is_same(:pl_name_komitet, :find_txt);

        if (need_suspend = 0) then
        begin
          find_str1 = (select res_text
                       from hp_txt_clear(:find_txt));
          find_str1 = f_metaphone(:find_str1);

          find_str2 = (select res_text
                       from hp_txt_clear(:pl_name));
          find_str2 = f_metaphone(:find_str2);

          if (position(pl_name, find_txt) > 0 or position(find_txt, pl_name) > 0) then
            need_suspend = 1;
        end
        if (need_suspend = 1) then
          suspend;
      end
    end
    else
    begin
      pl_id = pl_id_tmp;
      if (find_txt = '') then
        need_suspend = 1;
      else
      if (position(pl_name, find_txt) > 0 or position(find_txt, pl_name) > 0) then
        need_suspend = 1;
      else
      if (f_levenstain_is_same(:pl_name, :find_txt) = 1) then
        need_suspend = 1;
      else
        need_suspend = f_levenstain_is_same(:pl_name_komitet, :find_txt);

      if (need_suspend = 0) then
      begin
        find_str1 = (select res_text
                     from hp_txt_clear(:find_txt));
        find_str1 = f_metaphone(:find_str1);

        find_str2 = (select res_text
                     from hp_txt_clear(:pl_name));
        find_str2 = f_metaphone(:find_str2);

        if (position(pl_name, find_txt) > 0 or position(find_txt, pl_name) > 0) then
          need_suspend = 1;
      end

      if (need_suspend = 1) then
        suspend;
    end
  end
end
