CREATE PROCEDURE AAREP_DELINQ_PERSONNEL(
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(80),
    K_DISB_DATE DATE,
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    PR_LOAN_TYPE_STR CHAR(24),
    PR_CODE VARCHAR(100),
    PR_NAME VARCHAR(200),
    REG_NAME_KONT VARCHAR(244),
    C_NAME_FIO VARCHAR(400),
    JL_NAME VARCHAR(400),
    REG_NAME_HR VARCHAR(244),
    REP_DELIQ_BEGIN_DATE DATE,
    REP_ACTIVE_BALANCE DECIMAL(18,4),
    KNC_REP_ACTIVE_BALANCE DECIMAL(18,4),
    REP_DELIQ_DAYS_KOLICH INTEGER,
    PERSON_TYPE_STR VARCHAR(20),
    REP_DELIQ_OS DECIMAL(18,4),
    REP_DELIQ_PERC DECIMAL(18,4),
    REP_DELIQ_FINE DECIMAL(18,4),
    KNC_REP_DELIQ_OS DECIMAL(18,4),
    KNC_REP_DELIQ_PERC DECIMAL(18,4),
    KNC_REP_DELIQ_FINE DECIMAL(18,4))
AS
declare variable k_amount_approved decimal(18,4);
declare variable k_id decimal(18,0);
declare variable ps_from_c_id decimal(18,0);
declare variable ps_from_jl_id integer;
declare variable ps_from_reg_id decimal(15,2);
declare variable person_type integer;
declare variable kc_dolya double precision;
begin
  for select ps_from_c_id, ps_from_jl_id, ps_from_reg_id
      from hr_pstatus
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_is_working = 1
      into :ps_from_c_id, :ps_from_jl_id, ps_from_reg_id
  do
  begin
    for select ks_from_k_id, kc_amount_approved, 1 as person_type
        from kontstate, knclient
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end between ks_date_begin and ks_date_end and
              kc_from_k_id = ks_from_k_id and
              kc_amount_approved <> 0 and
              kc_from_c_id = :ps_from_c_id
        union
        select ks_from_k_id, kc_amount_approved, 2 as person_type
        from kontstate, zalog, knclient
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end between ks_date_begin and ks_date_end and
              ks_from_k_id = z_from_k_id and
              z_type = 0 and
              z_from_c_id = :ps_from_c_id and
              kc_from_k_id = ks_from_k_id and
              kc_amount_approved <> 0
        into :k_id, :kc_amount_approved, :person_type
    do
    begin
      rep_deliq_begin_date = null;
      rep_deliq_days_kolich = null;
      rep_active_balance = 0;
      knc_rep_active_balance = 0;
      jl_name = null;
      c_name_fio = null;
      reg_name_hr = null;
      k_kontract_no = null;
      k_disb_date = null;
      k_amount_approved = null;
      pr_loan_type_str = null;
      pr_code = null;
      pr_name = null;
      reg_name_kont = null;
      -- Проверка на просрочку кредита
      -- Баланс контракта
      rep_active_balance = 0;
      select oks_balance, oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_close_state = 0
      into :rep_active_balance, :rep_deliq_begin_date;
      if (rep_deliq_begin_date <= rep_date_end) then
      begin
        -- Должность сотрудника
        select jl_name
        from joblist
        where jl_id = :ps_from_jl_id
        into :jl_name;
        -- ФИО Сотрудника
        select c_name_fio
        from clients
        where c_id = :ps_from_c_id
        into :c_name_fio;
        -- Офис регистрации сотрудника
        select reg_code || ' ' || reg_name
        from region
        where reg_unicode = :ps_from_reg_id
        into reg_name_hr;
        -- Параметры контракта
        select k_kontract_no, k_disb_date, k_amount_approved,
               case pr_loan_type
                 when 0 then 'групп.'
                 when 1 then 'индив.'
               end as pr_loan_type_str,
               pr_code, pr_name, reg_code || ' ' || reg_name as reg_name_kont
        from kontract, product, knaktlist, region
        where k_id = :k_id and
              k_from_pr_id = pr_id and
              knaktlist.is_delete = 0 and
              k_from_kf_id = kal_from_kf_id and
              :rep_date_end between kal_date and kal_date_end and
              kal_new_reg_id = reg_unicode
        into k_kontract_no, k_disb_date, k_amount_approved, pr_loan_type_str, pr_code, pr_name, reg_name_kont;

        -- Баланс клиента
        kc_dolya = :kc_amount_approved / :k_amount_approved;
        knc_rep_active_balance = :kc_amount_approved / :k_amount_approved * :rep_active_balance;
        rep_deliq_days_kolich = :rep_date_end - rep_deliq_begin_date;
        -- Задолженности контракта по ОС, Процент и Пеня
        select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine)
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        where sh_date is not null and
              sh_repay_type = 1
        into :rep_deliq_os, :rep_deliq_perc, :rep_deliq_fine;
        -- Задолженности клиента по ОС, Процент и Пеня
        knc_rep_deliq_os = round(:rep_deliq_os * kc_dolya, 0);
        knc_rep_deliq_perc = round(:rep_deliq_perc * kc_dolya, 0);
        knc_rep_deliq_fine = round(:rep_deliq_fine * kc_dolya, 0);
        -- Тип сотруника, 1 - участник в группе или 2 - поручитель
        person_type_str = case :person_type
                            when 1 then 'Заемщик'
                            when 2 then 'Поручитель'
                          end;
        suspend;
      end
    end
  end
end
