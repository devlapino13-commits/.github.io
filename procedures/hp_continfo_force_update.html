CREATE PROCEDURE HP_CONTINFO_FORCE_UPDATE(
    C_ID D_ID,
    CONT_KIND D_STR_CODE,
    CONT_VALUE D_STR_100,
    CONT_STATUS D_SMALLINT,
    UN_ID D_ID,
    NEW_DATETIME D_DATETIME = current_timestamp)
AS
declare variable UNIQ_SUBJECT_ID decimal(18,0);
declare variable UNIQ_OBJECT_KIND integer;
declare variable UNIQ_SUBJECT_KIND integer;
declare variable OVS_FROM_OPR_ID decimal(18,0);
declare variable DATE_BEGIN D_DATE;
declare variable IS_NEW_CONTDATA D_SMALLINT;
declare variable CONT_IS_MAIN D_SMALLINT;
declare variable CDA_ID D_ID;
declare variable IS_NEW D_SMALLINT;
declare variable IS_EMPOYEE D_SMALLINT;
declare variable DB_ID D_INTEGER;
declare variable MA_ID D_ID;
declare variable K_ID D_ID;
begin
  db_id = f_get_db_id();
  -- Заменяет контактные данные клиента
  -- Из других систем вызывается через процедуру HP_CONTINFO_UPDATE
  -- Через FinStart может вызываться напрямую, если пользователь выбрал игнорирование ошибки
  is_new = 0;
  if (new_datetime is null) then
    new_datetime = current_timestamp;
  date_begin = cast(new_datetime as date);

  if (cont_kind is null) then
    exception e_default 'Cont_kind is null';

  if (coalesce(c_id, 0) = 0) then
    exception e_default 'C_ID is null';

  -- Процедура устанавливает новое значение для пары Object/Subject
  uniq_object_kind = f_objkind('clients', 'id');
  uniq_subject_kind = f_objkind('refdata', 'id');

  if (:db_id = 41701) then  -- RFCKG-4030
  begin
    select ma.ma_id
    from miniapp ma
    inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
    where mac.mac_from_c_id = :c_id
          and ma.ma_is_active = 1
          and ma.ma_date_end >= '31.12.9999'
          and mac.mac_amount > 0
    order by ma.ma_date_app desc
    rows 1
    into :ma_id;

    if(ma_id is not null) then
    begin
      if (f_edoc_status('loan.application', :ma_id, current_date) = 'signed') then
      begin
        -- Если заявка подписана, проверяем наличие контракта
        select kn.k_id
        from kontract kn
        inner join knclient kc on kc.kc_from_k_id = kn.k_id
        where kc.kc_from_c_id = :c_id
            and kn.k_from_ma_app_id = :ma_id
            and kc.kc_amount_approved > 0
        into :k_id;
        
        if (k_id is null or (f_edoc_status('loan.agreement', :k_id, current_date) <> 'signed')) then
          -- Аннулируем заявку, так как контракт не подписан
          execute procedure hp_edoc_cancel('loan.application', 'miniapp', :ma_id);
      end
    end
  end

  select rd_id, cont_kind_code, cont_is_main
  from hp_contkind_info(:cont_kind)
  into :uniq_subject_id, :cont_kind, :cont_is_main;

  if (cont_status is null) then
    cont_status = 0;

  is_new_contdata = f_get_const('new_contdata');

  is_new_contdata = 2;

  cont_value = f_cont_trim(cont_value, cont_kind);

  if (cont_kind = 'cell' and
      cont_is_main = 1) then
  begin
    -- Если основной номер является корпоративным, подтверждаем автоматически
    if (exists(select *
               from cellaktlist
               where cal_from_c_id = :c_id and
                     cal_cell_no = :cont_value and
                     cal_date_begin <= current_date and
                     cal_date_end >= current_date and
                     cal_state = 1 and
                     cal_is_corporate = 1)) then
    begin
      -- Проверяем, действующий ли сотрудник
      if (exists(select *
                 from hr_pstatus
                 where ps_from_c_id = :c_id and
                       ps_date_begin <= current_date and
                       ps_date_end >= current_date and
                       ps_is_working = 1)) then
        cont_status = 2;
    end
  end

  if (is_new_contdata in (0, 2)) then
  begin
    select opr_id
    from objectprops
    where opr_object_kind = :uniq_object_kind and
          opr_object_id = :c_id and
          opr_subject_kind = :uniq_subject_kind and
          opr_subject_id = :uniq_subject_id
    into :ovs_from_opr_id;

    if (ovs_from_opr_id is null) then
    begin
      insert into objectprops (opr_object_kind, opr_object_id, opr_subject_kind, opr_subject_id)
      values (:uniq_object_kind, :c_id, :uniq_subject_kind, :uniq_subject_id)
      returning (opr_id)
      into :ovs_from_opr_id;

      is_new = 1;
    end

    insert into objectvalues (ovs_from_opr_id, ovs_date_begin, ovs_value, ovs_add_value, ovs_from_un_id)
    values (:ovs_from_opr_id, :date_begin, :cont_value, :cont_status, :un_id)
    returning ovs_add_value
    into cont_status;
  end

  if (is_new_contdata in (1, 2)) then
  begin
    if (cont_value is null) then
      cont_value = '';

    select cda_id
    from hp_contdata_update(:c_id, :cont_kind, :cont_value, :cont_is_main, :un_id, :new_datetime)
    into cda_id;

    if (cont_status > 0 and
        cda_id is not null) then
      execute procedure hp_contveri_set_ok(:cda_id, :un_id, :new_datetime);
  end

  /*Отправка на проверку основного телефонного номера клиента и аннулирование ЭЦП*/
  is_empoyee = 0;
  if (cont_kind = 'cell' and
      cont_is_main = 1 and
      cont_status <> 2 and
      is_new <> 1 and
      f_get_const('check_contact_number') = 1) then
  begin
    if (exists(select *
               from hr_pstatus
               where ps_from_c_id = :c_id and
                     ps_date_begin <= current_date and
                     ps_date_end >= current_date and
                     ps_is_working = 1)) then
      is_empoyee = 1;
    execute procedure hp_empl_main_cell_drop_verify(:c_id, :cont_value, :un_id);

    --if (db_id <> 41701 or f_get_const('antifrod_phone_active') = 0) then
    --  execute procedure hp_actlist_ins_by_change_phone(:c_id, :un_id, :is_empoyee);
  end
end
