CREATE PROCEDURE FIND_PAYSYS_REPAYS_DOUBLICATE(
    REPAY_PSI_ID DECIMAL(18,0),
    REPAY_DOC_NO INTEGER,
    REPAY_DATE DATE,
    REPAY_AMOUNT DECIMAL(18,4) = 0)
RETURNS (
    ALREADY_SAVED INTEGER)
AS
declare variable psr_from_ol_id decimal(18,0);
begin
  already_saved = 0;
  for select psr_from_ol_id
      from paysys_repays
      where psr_from_psi_id = :repay_psi_id and
            psr_date = :repay_date and
            psr_doc_no = :repay_doc_no and
            psr_amount = :repay_amount
      into :psr_from_ol_id
  do
  begin
    if (exists(select *
               from operlist
               where ol_id = :psr_from_ol_id and
                     is_delete = 0)) then
      already_saved = 1;
  end
  suspend;
end
