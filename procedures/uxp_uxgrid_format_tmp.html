CREATE PROCEDURE UXP$UXGRID_FORMAT_TMP
RETURNS (
    RES_TEXT_COLOR D_STR_CODE,
    RES_BRUSH_COLOR D_STR_CODE,
    RES_FONT_BOLD BOOLEAN,
    RES_FONT_ITALIC BOOLEAN,
    RES_FONT_UNDERLINE BOOLEAN,
    RES_FONT_STRIKEOUT BOOLEAN)
AS
declare variable ugr_formatting d_str_8100;
declare variable jsa d_integer;
declare variable jso d_integer;
declare variable jsm d_integer;
declare variable arr_count d_integer;
declare variable arr_index d_integer;
begin
  for select ugr_formatting
  from ux$grid
  where trim(coalesce(ugr_formatting, '')) <> ''
  into ugr_formatting
  do
  begin
    ugr_formatting = '{"arr":' || ugr_formatting || '}';

    -- Парсинг массива
    jsm = json.parse(ugr_formatting);
    jsa = json.obj_get_arr(jsm, 'arr');
    arr_count = json.count_get(jsa);

    arr_index = 0;
    while (arr_index < arr_count) do
    begin
      res_text_color = 'clWindowText';
      res_brush_color = 'clWindow';
      res_font_bold = false;
      res_font_italic = false;
      res_font_underline = false;
      res_font_strikeout = false;
  
      jso = json.arr_get_obj(jsa, arr_index);

      -- Получение значения из объекта
      res_text_color = json.obj_get_val(jso, 'text_color', res_text_color);
      res_brush_color = json.obj_get_val(jso, 'brush_color', res_brush_color);
      res_font_bold = json.obj_get_val(jso, 'font.b', res_font_bold);
      res_font_italic = json.obj_get_val(jso, 'font.i', res_font_italic);
      res_font_underline = json.obj_get_val(jso, 'font.u', res_font_underline);
      res_font_strikeout = json.obj_get_val(jso, 'font.s', res_font_strikeout);

      if (res_text_color <> 'clWindowText' or
          res_brush_color <> 'clWindow' or
          res_font_bold or
          res_font_italic or
          res_font_underline or
          res_font_strikeout) then
        suspend;

      arr_index = arr_index + 1;
    end

    json.destroy(jsm);
  end
end
