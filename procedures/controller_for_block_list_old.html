CREATE PROCEDURE CONTROLLER_FOR_BLOCK_LIST_OLD(
    REP_DATE DATE,
    DAY_COUNT_PRIOR INTEGER,
    REP_REGION_ID ID_TYPE)
RETURNS (
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    K_DISB_DATE DATE,
    COUNTER INTEGER,
    STATE VARCHAR(10),
    CONTROL_FIO STR_100,
    KONT_IS_CHECKED SMALLINT,
    CL_IS_CHECKED SMALLINT,
    IS_CONTROL_BLOCK SMALLINT,
    CHECK_OK SMALLINT,
    CHECK_NOT_OK SMALLINT,
    KF_COUNTER INTEGER,
    UN_ID ID_TYPE,
    CHECK_RESULT STR_255,
    KF_ID ID_TYPE,
    MA_APP_ID ID_TYPE)
AS
declare variable reg_id integer;
declare variable dw smallint;
declare variable cl_checked_id id_type;
declare variable k_fact_disb_date date;
declare variable k_fact_closed_date date;
declare variable is_ok smallint;
declare variable rd_controller_id integer;
begin
  counter = 0;
  kf_counter = 0;
  dw = extract(weekday from rep_date);

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ControllerType.Controller'
  into :rd_controller_id;

  for select ks_from_k_id, kont.k_from_kf_id, k_last_reg_id, k_disb_date,
             (
             case ks_state
               when 0 then 'В процессе'
               else 'Выдан'
             end),
             kont.k_fact_disb_date, kont.k_fact_closed_date, kont.k_from_ma_app_id
      from kontstate ks
      inner join kontract kont on kont.k_id = ks.ks_from_k_id
      inner join region_list(:rep_region_id, 1, -1, 1) on cast(kont.k_last_reg_id as id_type) = id_list
      where ks_ol_is_delete = 0 and
            ((ks_state = 0) and
            (:dw not in (0, 6)) or
            ((ks_state in (0, 1)) and
            (:dw in (0, 6)))) and
            :rep_date between ks_date_begin and ks_date_end and
            ks.ks_date_begin >= :rep_date - :day_count_prior and
            kont.k_status > -1
      order by kont.k_disb_date, :reg_id, :kf_id
      into :k_id, :kf_id, :reg_id, :k_disb_date, :state, :k_fact_disb_date, :k_fact_closed_date, :ma_app_id
  do
  begin
    is_ok = 0;
    if ((k_fact_disb_date is null) and
        (:dw not in (0, 6))) then
      is_ok = 1; -- в будние только договора в процессе

    if (:dw in (0, 6)) then
    begin
      is_ok = 1;

      if (k_fact_disb_date is not null) then
      begin
        k_disb_date = k_fact_disb_date;
      end
    end

    if (k_fact_closed_date is not null) then
      is_ok = 0;

    if (is_ok = 1) then
    begin
      cl_checked_id = null;
      control_fio = '';
      kont_is_checked = 0;

      check_ok = 0;
      check_not_ok = 0;
      check_result = null;

      is_control_block = 0;
      un_id = null;

      select cl_cont.c_name_fio,
             (
             case
               when cont.contr_checked_date is not null then 1
               else 0
             end),
             cont.contr_from_c_id, rd_res.rd_code, cont.contr_from_un_id
      from controller cont
      inner join username un on un.un_id = cont.contr_from_un_id
      inner join clients cl_cont on cl_cont.c_id = un.un_from_c_id
      inner join refdata rd_res on rd_res.rd_id = cont.contr_check_result
      where cont.contr_from_k_id = :k_id and
            cont.contr_is_delete = 0 and
            cont.contr_rd_type = :rd_controller_id rows 1
      into :control_fio, :kont_is_checked, :cl_checked_id, :check_result, :un_id;

      check_result = coalesce(check_result, 'ControllerCheckStatus.NotChecked');
      if (control_fio <> '') then
        is_control_block = 9;

      if (check_result = 'ControllerCheckStatus.Confirmed') then
      begin
        check_ok = 1;
        is_control_block = 1;
      end

      if (check_result = 'ControllerCheckStatus.NotConfirmed') then
      begin
        check_not_ok = 1;
        is_control_block = -1;
      end

      kf_counter = kf_counter + 1;

      for select kc.kc_from_c_id
          from knclient kc
          where kc_from_k_id = :k_id and
                kc_amount_approved > 0
          into :c_id
      do
      begin
        cl_is_checked = 0;
        if (cl_checked_id = c_id) then
          cl_is_checked = 1;
        counter = counter + 1;

        suspend;
      end
    end
  end
end
