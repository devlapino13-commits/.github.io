CREATE PROCEDURE HP_FT_INFOFIELD_VALUE(
    FIC_ID D_ID,
    FIL_ID D_ID,
    FIC_DATE D_DATE,
    FLD_NAME D_STR_CODE = null)
RETURNS (
    FIF_ID D_ID,
    FIF_CAPTION D_STR_100,
    FIF_BLANK_CAPTION D_STR_2048,
    FIF_ORDER_NO D_INTEGER,
    FIF_IS_DELETE D_SMALLINT,
    FIV_ID D_ID,
    FIV_VALUE D_STR_LONG,
    FIV_DATE_BEGIN D_DATE,
    IS_CHANGED D_SMALLINT,
    IS_MANNUAL D_SMALLINT,
    FIV_ORIG_VALUE D_STR_LONG)
AS
declare variable fib_id d_id;
declare variable c_id d_id;
declare variable fil_date_close d_date;
declare variable fif_fieldname d_str_100;
declare variable fil_date_open d_date;
declare variable fic_date_tmp date;
declare variable fib_kind d_smallint;
declare variable base_eil_id d_id;
declare variable lic_eil_id d_id;
declare variable brt_date_end d_date;
begin
  if (fic_id is null) then fic_id = 0;
  if (fic_id <> 0) then
  begin
    select fic_date, fil_from_fib_id, fil_from_c_id, fil_date_close, fil_date_open
    from ft_infocheck, ft_infolist
    where fic_id = :fic_id and
          fic_from_fil_id = fil_id
    into fic_date, fib_id, c_id, fil_date_close, fil_date_open;
  end else
  begin
    select fil_from_fib_id, fil_from_c_id, fil_date_close, fil_date_open
    from ft_infolist
    where fil_id = :fil_id
    into fib_id, c_id, fil_date_close, fil_date_open;
  end

  if (fib_id is null) then Exit;

  select fib.fib_kind
  from ft_infoblank fib
  where fib.fib_id = :fib_id
  into fib_kind;

  select eil.eil_id
  from extinfolist eil
  inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
  where eil.eil_from_c_id = :c_id and
        eil.eil_date <= :fic_date and
        eic.eic_code = 'kontur_ip_base' and
        eil.eil_status = 2
  order by eil.eil_date desc
  rows 1
  into base_eil_id;

  select eil.eil_id
  from extinfolist eil
  inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
  where eil.eil_from_c_id = :c_id and
        eil.eil_date <= :fic_date and
        eic.eic_code = 'kontur_ip_lic' and
        eil.eil_status = 2
  order by eil.eil_date desc
  rows 1
  into lic_eil_id;

  for select fif_id,
             fif_caption,
             fif_blank_caption,
             fif_order_no,
             fif_is_delete,
             ff.fif_fieldname
      from ft_infofield ff
      where fif_from_fib_id = :fib_id and
            fif_date <= :fic_date and
            fif_is_hidden = 0 and
            fif_fieldname = coalesce(:fld_name, fif_fieldname)
      order by fif_order_no, fif_is_delete, fif_id
      into :fif_id,
           :fif_caption,
           :fif_blank_caption,
           :fif_order_no,
           :fif_is_delete,
           :fif_fieldname
  do
  begin
    fiv_id = null;
    fiv_value = '';
    fiv_orig_value = ''; -- если значение изменено вручную, в этом поле будем отображать значение до изменения
    is_changed = 0;
    is_mannual = 0;

    if (fif_fieldname in ('client_fio', 'client_fio (blank:1-3)')) then
    begin
        select res_name_fio
        from ft_fio_change_hist(:c_id, 1)
        where cast(res_datetime as date) <=  :fil_date_open
        order by res_datetime desc
        rows 1
        into :fiv_value;
    end
    else
    if (fif_fieldname = 'citizen') then
    begin
        select rd_name
        from clientdocs, passkind, refdata
        where cds_from_c_id = :c_id and
            cds_create_datetime < (:fil_date_open + 1) and
            cds_doc_date_begin <= :fil_date_open and
            --coalesce(cds_doc_date_end, :fil_date_open) >= :fil_date_open and --RFCRU-4076
            cds_is_main_doc = 1 and
            cds_from_pk_id = pk_id and
            pk_country_id = rd_id --and
            --cds_is_delete = 0
        order by cds_create_datetime desc
        rows 1
        into fiv_value;
    end
    else if (fif_fieldname in ('pass_props', 'pass_props (blank:1-3)')) then
    begin
        select 'Серия, номер: ' || coalesce(cds_doc_no, '') || '; Дата выдачи: ' || f_datetostr(cds_doc_date_begin) || '; Орган, выдавший документ: ' ||
            coalesce(cds_doc_issued_by, '') || iif(:fic_date < cast('24.07.2025' as date) or cds_from_pk_id not in (12,13), '', coalesce('; Код подразделения: ' || cds_doc_issued_code, ''))
        from clientdocs
        where cds_from_c_id = :c_id and
                cds_create_datetime < (:fic_date + 1) and
                cds_is_main_doc = 1 --and
                --cds_is_delete = 0  -- igusev 11.10.2019 убрал данное условие т.к. в триггере CLIENTS_AU0 есть скрипт, который аннулирует основной документ если прикрепляется новый
        order by cds_create_datetime desc
        rows 1
        into fiv_value;
    end
    else if (fif_fieldname = 'relation_finished') then
    begin
        if (not exists (
            select *
            from FT_INFOLIST fil
            where fil.fil_from_c_id = :c_id
            and fil.fil_date_open = :fil_date_close + 1))
        then
            fiv_value = f_datetostr(:fil_date_close); 
        else
        begin
           if (exists (select * from kontstate ks
                       inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
                       where ks.ks_state = 2
                       and ks.ks_ol_is_delete = 0
                       and ks.ks_date_begin = :fil_date_close
                       and kncl.kc_from_c_id = :c_id
                       and kncl.kc_amount_approved > 0)) then
              fiv_value = f_datetostr(:fil_date_close);
           else
              fiv_value = '';
        end
    end
    else if (fif_fieldname = 'start_relation') then
    begin
        select f_datetostr(start_relation_date)
        from ft_start_relation_calc(:fil_date_open, :c_id)
        into :fiv_value;
    end
    else
    begin
        select fiv_id, fiv_value, fiv_date_begin
        from ft_infovalue
        where fiv_from_c_id = :c_id and
              fiv_from_fif_id = :fif_id and
              fiv_date_begin <= :fic_date and
              fiv_date_end >= :fic_date
        into :fiv_id, :fiv_value, fiv_date_begin;
    end

    if ((fif_fieldname = 'mcard_props') and (fiv_value = '---')) then
    begin
        select res_text
        from ft_mcard_props_get(:c_id, :fil_date_open)
        into :fiv_value;

        if (fiv_value containing '---') then
        begin
            select fic_date
            from ft_infocheck fic, ft_infolist fil
            where fic_from_fil_id = fil_id
            and fic.fic_date < :fil_date_open
            and fil.fil_from_c_id = :c_id
            order by fic_date desc
            rows 1
            into :fic_date_tmp;
    
            if (fic_date_tmp is not null) then
            begin
                select res_text
                from ft_mcard_props_get(:c_id, :fic_date_tmp)
                into :fiv_value;
            end
        end
    end

    if (fif_fieldname = 'acard_props') then
    begin
        select res_text
        from ft_acard_props_get(:c_id, :fic_date)
        into :fiv_value;

        if (fiv_value containing '---') then
        begin
            select fic_date
            from ft_infocheck fic, ft_infolist fil
            where fic_from_fil_id = fil_id
            and fic.fic_date < :fic_date
            and fil.fil_from_c_id = :c_id
            order by fic_date desc
            rows 1
            into :fic_date_tmp;
    
            if (fic_date_tmp is not null) then
            begin
                select res_text
                from ft_acard_props_get(:c_id, :fic_date_tmp)
                into :fiv_value;
            end    
        end
    end

    if (fif_fieldname = 'ogrnip_no' and
        fib_kind = 2) then
    begin
      select ogrn
      from hp_kontur_ip_get_base_info('extinfolist', :base_eil_id)
      into :fiv_value;

      fiv_value = coalesce(fiv_value, '');
    end

    if (fif_fieldname = 'ip_reg_date' and
        fib_kind = 2) then
    begin

      select reg_date
      from hp_kontur_ip_get_base_info('extinfolist', :base_eil_id)
      into :fiv_value;

      fiv_value = f_datetostr(fiv_value);
    end

    if (fif_fieldname = 'business_licence' and
        fib_kind = 2) then
    begin
      select license_txt
      from hp_kontur_ip_get_lic_info('extinfolist', :lic_eil_id)
      into :fiv_value;

      if (coalesce(fiv_value, '') = '') then
        fiv_value = '«Отсутствует»';
    end

    if (fif_fieldname = 'address_legal' and
        fib_kind = 2) then
    begin
      select res_value
      from hp_history_client(:c_id, 1, 'address')
      where res_date <= :fic_date
      order by res_date desc
      rows 1
      into fiv_value;

      if (coalesce(fiv_value, '') = '') then
      begin
        select adl.adl_address_str
        from addrclient adcl
        inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
        where adcl.adc_from_c_id = :c_id and
              adcl.adc_address_type = 1
        into :fiv_value;
      end
      if (coalesce(fiv_value, '') = '') then
        fiv_value = '---';
    end

    if (fif_fieldname = 'ogrnip_address' and
        fib_kind = 2) then
    begin
      select res_value
      from hp_history_client(:c_id, 1, 'address')
      where res_date <= :fic_date
      order by res_date desc
      rows 1
      into fiv_value;

      if (coalesce(fiv_value, '') = '') then
      begin
        select adl.adl_address_str
        from addrclient adcl
        inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
        where adcl.adc_from_c_id = :c_id and
              adcl.adc_address_type = 1
        into :fiv_value;
      end
      if (coalesce(fiv_value, '') = '') then
        fiv_value = '---';
    end

    if (fif_fieldname = 'financial_position' and
        fib_kind = 2) then
    begin
      select max(brt_date_end)
      from bankrupt
      where brt_from_c_id = :c_id
        and brt_date_fact <= :fic_date
      into brt_date_end;

      if (brt_date_end is not null) then
      begin
        select delo_no || ' ' || status || ' ' || status_date
        from hp_get_kontur_ext_info(:c_id, :fic_date, :brt_date_end)
        into :fiv_value;
      end
      else
        fiv_value = '«Отсутствуют производства о несостоятельности (банкротстве)»';
    end

    if (fif_fieldname = 'business_reputation' and
        fib_kind = 2) then
    begin
      if (exists (select 1
                  from clientdocs cds
                  inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
                  inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
                  where pdt.pdt_code = 'fssp' and
                        cds.cds_verify_status = 2 and
                        cds.cds_from_c_id = :c_id and
                        :fic_date between cds.cds_doc_date_begin and cds.cds_doc_date_end)) then
         fiv_value = '«Присутствуют cведения о наличии исполнительных производств»';
       else 
         fiv_value = '«Отсутствуют cведения о наличии исполнительных производств»';

    end

    if (fiv_date_begin = fic_date) then is_changed = 1;

    if (exists(select * from ft_infoeditvalue
               where fie_from_c_id = :c_id and
                     fie_from_fif_id = :fif_id and
                     fie_date = :fic_date)) then is_mannual = 1;

    suspend;
  end
end
