CREATE PROCEDURE HP_GET_ADDRLIST(
    ADL_ID D_ID,
    REP_DATETIME D_DATETIME)
RETURNS (
    ADL_FROM_PL_ID D_ID,
    ADL_ADDRESS_STR D_STR_255)
AS
declare variable adl_datetime timestamp;
begin
    select adl_from_pl_id, adl_address_str, adl_datetime
    from addrlist
    where adl_id = :adl_id
    into adl_from_pl_id, adl_address_str, adl_datetime;

    if (adl_datetime > rep_datetime) then
    begin
--      adl_from_pl_id = null;
--      adl_address_str = null;


      select adl_from_pl_id, adl_address_str
      from addrloglist
      where adl_id = :adl_id and
            adl_datetime <= :rep_datetime
      order by adl_datetime desc
      rows 1
      into adl_from_pl_id, adl_address_str;
    end
    -- Если ничего не будет найдено, все равно возвращаем одну строку
  suspend;
end
