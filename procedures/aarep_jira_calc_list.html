CREATE PROCEDURE AAREP_JIRA_CALC_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_LINE_ID D_INTEGER,
    JRD_ID D_ID,
    JRD_FIO D_STR_100,
    JRD_COUNTRY D_STR_10,
    JRD_LEVEL D_SMALLINT,
    JRD_IS_MANAGER D_BOOL,
    JRC_PERIOD D_INTEGER,
    JRC_PERIOD_STR D_STR_20,
    JRC_DONE_IN_TIME D_INTEGER,
    JRC_DONE_TASKS D_INTEGER,
    JRC_KPI_1 D_CURRENCY,
    JRC_KPI_PERC_1 D_CURRENCY,
    JRC_WORK_HOURS D_CURRENCY,
    JRC_NORM_HOURS D_INTEGER,
    JRC_KPI_2 D_CURRENCY,
    JRC_KPI_PERC_2 D_CURRENCY,
    JRC_TASKS D_INTEGER,
    JRC_TOTAL_DIFFICULT D_INTEGER,
    JRC_DISTINCT_DIFFICULT D_INTEGER,
    JRC_AVG_DIFFICULT D_CURRENCY,
    JRC_KPI_3 D_CURRENCY,
    JRC_KPI_PERC_3 D_CURRENCY,
    JRC_FACT_WORK_HOURS D_INTEGER,
    JRC_AVG_WORK_EXPECT D_CURRENCY,
    JRC_KPI_4 D_CURRENCY,
    JRC_KPI_PERC_4 D_CURRENCY,
    JRC_REWORKED D_INTEGER,
    JRC_KPI_5 D_CURRENCY,
    JRC_KPI_PERC_5 D_CURRENCY,
    JRC_KPI_FINAL D_CURRENCY,
    JRC_ADD_PERC D_CURRENCY,
    JRC_KPI_RESULT D_CURRENCY,
    JRC_ADD_COMMENT D_STR_255)
AS
declare variable rep_period_begin d_integer;
declare variable rep_period_end d_integer;
begin
  rep_period_begin = extract(year from rep_date_begin) * 100 + extract(month from rep_date_begin);
  rep_period_end = extract(year from rep_date_end) * 100 + extract(month from rep_date_end);
  res_line_id = 0;

  for select jrd_id, jrd_fio, jrd_country, jrd_level, jrd_is_manager, jrc_period,
             jrc_done_in_time, jrc_done_tasks, jrc_kpi_1, jrc_kpi_perc_1, -- kpi-1
             jrc_work_hours, jrc_norm_hours, jrc_kpi_2, jrc_kpi_perc_2, -- kpi-2
             jrc_tasks, jrc_total_difficult, jrc_distinct_difficult, jrc_avg_difficult, jrc_kpi_3, jrc_kpi_perc_3, -- kpi-3
             jrc_fact_work_hours,  jrc_avg_work_expect, jrc_kpi_4, jrc_kpi_perc_4, -- kpi-4
             jrc_reworked, jrc_kpi_5, jrc_kpi_perc_5, -- kpi-4
             jrc_kpi_final, jrc_add_perc, jrc_kpi_result, jrc_add_comment -- Результат
  from jira_calc, jira_dev
  where jrc_dev_id = jrd_id and
        jrc_period between :rep_period_begin and :rep_period_end
  order by jrd_fio, jrc_period
  into jrd_id, jrd_fio, jrd_country, jrd_level, jrd_is_manager, jrc_period,
       jrc_done_in_time, jrc_done_tasks, jrc_kpi_1, jrc_kpi_perc_1, -- kpi-1
       jrc_work_hours, jrc_norm_hours, jrc_kpi_2, jrc_kpi_perc_2, -- kpi-2
       jrc_tasks, jrc_total_difficult, jrc_distinct_difficult, jrc_avg_difficult, jrc_kpi_3, jrc_kpi_perc_3, -- kpi-3
       jrc_fact_work_hours,  jrc_avg_work_expect, jrc_kpi_4, jrc_kpi_perc_4, -- kpi-4
       jrc_reworked, jrc_kpi_5, jrc_kpi_perc_5, -- kpi-4
       jrc_kpi_final, jrc_add_perc, jrc_kpi_result, jrc_add_comment -- Результат
  do
  begin
    res_line_id = res_line_id + 1;
    jrc_period_str = left(jrc_period, 4) || '-' || right(jrc_period, 2);
    suspend;
  end
end
