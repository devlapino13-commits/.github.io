CREATE PROCEDURE AAREP_GET_REGION_MANAGER(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    C_ID_MANAGER ID_TYPE,
    PS_DATE_BEGIN DATE,
    REG_ID_THE_DATE INTEGER,
    REG_ID_FROM_BONUS INTEGER,
    AFTER_15 SMALLINT,
    DATE_ACTIVE DATE)
AS
declare variable date_c date;
begin
--  дата на 15е число
  date_c = cast('15.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date);
  for select ps1.ps_from_c_id, ps_date_begin, ps_from_reg_id, iif(ps_date_begin <= :date_c, ps_from_reg_id, -- если дата началы КМ <= 15 числа, то офис КМ на конец будет брать
             iif((select ps_from_reg_id -- если на 15 число не было КМом то офис КМа будет брать на конец
                  from hr_pstatus, joblist, jobkind
                  where :date_c between ps_date_begin and ps_date_end and
                        ps_from_jl_id = jl_id and
                        jl_kind = jk_id and
                        ps_is_working = 1 and
                        jk_system_code = 'JobSystem.Region_KM' and
                        ps_status <> -1 and
                        ps_from_c_id = ps1.ps_from_c_id) is null, ps_from_reg_id,
             (select ps_from_reg_id -- если на 15 число был КМом то офис КМа берет на 15е число
              from hr_pstatus, joblist, jobkind
              where :date_c between ps_date_begin and ps_date_end and
                    ps_from_jl_id = jl_id and
                    jl_kind = jk_id and
                    ps_is_working = 1 and
                    jk_system_code = 'JobSystem.Region_KM' and
                    ps_status <> -1 and
                    ps_from_c_id = ps1.ps_from_c_id))), -------------------------
             iif(ps_date_begin <= :date_c, 0, iif((select ps_from_reg_id
                                                   from hr_pstatus, joblist, jobkind
                                                   where :date_c between ps_date_begin and ps_date_end and
                                                         ps_from_jl_id = jl_id and
                                                         jl_kind = jk_id and
                                                         ps_is_working = 1 and
                                                         jk_system_code = 'JobSystem.Region_KM' and
                                                         ps_status <> -1 and
                                                         ps_from_c_id = ps1.ps_from_c_id) is null, 0, 1))
      from hr_pstatus ps1, joblist, jobkind
      where :rep_date_end between ps_date_begin and ps_date_end
--    and ps_from_reg_id=:reg_id
            and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            ps_is_working = 1 -- только активных сотрудников
            and
            jk_system_code = 'JobSystem.Region_KM' -- выборка по кредитным менеджерам
            and
            ps_status <> -1 and
            (iif(ps_date_begin <= :date_c, ps_from_reg_id, -- если дата началы КМ <= 15 числа, то офис КМ на конец будет брать
            iif((select ps_from_reg_id -- если на 15 число не было КМом то офис КМа будет брать на конец
                 from hr_pstatus, joblist, jobkind
                 where :date_c between ps_date_begin and ps_date_end and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       ps_is_working = 1 and
                       jk_system_code = 'JobSystem.Region_KM' and
                       ps_status <> -1 and
                       ps_from_c_id = ps1.ps_from_c_id) is null, ps_from_reg_id, (select ps_from_reg_id -- если на 15 число был КМом то офис КМа берет на 15е число
                                                                                  from hr_pstatus, joblist, jobkind
                                                                                  where :date_c between ps_date_begin and ps_date_end and
                                                                                        ps_from_jl_id = jl_id and
                                                                                        jl_kind = jk_id and
                                                                                        ps_is_working = 1 and
                                                                                        jk_system_code = 'JobSystem.Region_KM' and
                                                                                        ps_status <> -1 and
                                                                                        ps_from_c_id = ps1.ps_from_c_id)))) = :rep_region_id
      into c_id_manager, -- id КМ
           ps_date_begin, -- дата началы работы или дата перевода
           reg_id_the_date, -- ID региона на текущую дату
           reg_id_from_bonus, -- ID региона для расчета бонуса КМ
           after_15 -- если after_15 = 1 то переведет  после 15 число
  do
  begin
    if (after_15 = 1) then
      date_active = date_c;
    else
      date_active = :rep_date_end;
    suspend;
  end
end
