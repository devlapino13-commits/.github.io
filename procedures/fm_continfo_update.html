CREATE PROCEDURE FM_CONTINFO_UPDATE(
    OBJECT_KIND VARCHAR(100),
    OBJECT_ID VARCHAR(255),
    SUBJECT_ID VARCHAR(255),
    SET_VALUE VARCHAR(255),
    UN_ID DECIMAL(18,0))
RETURNS (
    CONT_STATUS VARCHAR(100),
    CONT_STATUS_ID INTEGER)
AS
begin
  if (f_objkind(object_kind, 'name') <> 'clients') then
    exception e_default 'Incorrect object_kind, expected (0 or CLIENTS)';

  uxf$set_user(un_id);
  execute procedure hp_continfo_update(:object_id, :subject_id, :set_value, 0, :un_id);

  -- Получаем обновленный статус
  select res_value_status, res_verify_status
  from aarep_object_one_continfo(:object_kind, :object_id, :subject_id, 0, :un_id)
  into cont_status, cont_status_id;
  suspend;
end
