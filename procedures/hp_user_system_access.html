CREATE PROCEDURE HP_USER_SYSTEM_ACCESS(
    UN_ID DECIMAL(18,0))
RETURNS (
    PA_ID INTEGER,
    PA_NAME VARCHAR(100),
    PA_COMMAND_NAME VARCHAR(100),
    PA_ACCESS_KIND INTEGER)
AS
begin
  for select pa_id, pa_name, pa_command_name
      from permactions
      where pa_parent_id is null
      into :pa_id, pa_name, pa_command_name
  do
  begin
    select access_kind
    from hp_permission_access_check(:un_id, 'PermActions', 0, :pa_command_name)
    into :pa_access_kind;
    suspend;
  end
end
