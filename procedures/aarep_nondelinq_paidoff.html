CREATE PROCEDURE AAREP_NONDELINQ_PAIDOFF(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    CL_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_TYPE D_STR_10,
    PR_NAME D_STR_255,
    K_DISB_DATE D_DATE,
    LO_FIO D_STR_100,
    MO_FIO D_STR_100,
    PAY_SUM D_CURRENCY,
    DELINQ_DAYS D_SMALLINT,
    OL_DOC_KIND_TXT D_STR_100,
    OL_DOC_CLIENT_NAME D_STR_255,
    OL_DATE D_DATE,
    KF_ID D_ID,
    K_ID D_ID,
    C_ID D_ID,
    OL_ID D_ID,
    SH_DATE D_DATE,
    SH_OS D_CURRENCY,
    SH_PERC D_CURRENCY,
    C_TEL D_STR_20)
AS
begin
  /* report on clients who have paid off active delinquencies - отчет по клиентам погасившим активные просрочки*/
  for select ok_from_k_id, ok_from_ol_id, ok_ol_date, ok_sh_date,
             -sum(ok.ok_os + ok.ok_perc + ok.ok_fine + ok.ok_kommis + ok.ok_kommis_adv + ok.ok_tax)
      from oksummary oks
      inner join region_list(:rep_region_id, 1, -1, 1) on oks.oks_region_id = id_list
      inner join operkred ok on oks_from_k_id = ok_from_k_id and
            ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_ol_date > ok_sh_date and
            --ok_from_k_id = 101020000069901201 and
            ok_type = 4
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks.oks_close_state = 0
      group by ok_from_k_id, ok_from_ol_id, ok_ol_date, ok_sh_date
      having sum(ok.ok_os + ok.ok_perc + ok.ok_fine + ok.ok_kommis + ok.ok_kommis_adv + ok.ok_tax) < 0
      into k_id, ol_id, ol_date, sh_date, pay_sum
  do
  begin
    sh_os = 0;
    sh_perc = 0;
    select sum(ok_os + ok_sh_os) sh_os, sum(ok_perc + ok_sh_perc) sh_perc
    from operkred ok
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_type < 7 and
          ok_ol_date <= :ol_date and
          ok_from_ol_id <= :ol_id and
          ok_sh_date = :sh_date
    group by ok_sh_date
    having sum(ok_os + ok_sh_os) > 0 or sum(ok_perc + ok_sh_perc) > 0
    into sh_os, sh_perc;
    sh_os = coalesce(sh_os, 0);
    sh_perc = coalesce(sh_perc, 0);

    if (sh_os = 0 and
        sh_perc = 0) then
    begin
      delinq_days = ol_date - sh_date;
      if (coalesce(delinq_days, -1) < 0) then
        delinq_days = 0;

      c_id = null;
      ol_doc_kind_txt = null;
      ol_doc_client_name = null;
      select ol.ol_doc_client_name, ol.ol_document_client,
             case
               when ol.ol_doc_kind = 0 then 'Кассовый расходный'
               when ol.ol_doc_kind = 1 then 'Кассовый приходный'
               when ol.ol_doc_kind = 2 then 'Счет фактура'
               when ol.ol_doc_kind = 3 then 'Акты'
               when ol.ol_doc_kind = 4 then 'Распоряжение на списание кредита'
               when ol.ol_doc_kind = 5 then 'Распоряжение на реструктуризацию'
               when ol.ol_doc_kind = 6 then 'Платежное поручение'
               when ol.ol_doc_kind = 7 then 'Безналичный Приходный'
               when ol.ol_doc_kind = 8 then 'Начисление'
               when em.esm_project = 'Companion' then 'Компаньон'
               when em.esm_project = 'doscredo' then 'Доскредо'
               when em.esm_project = 'DosCredoAcq' then 'Доскредо'
               when em.esm_project = 'elcard_disb' then 'Элкард'
               when em.esm_project = 'elsom' then 'Элсом'
               when em.esm_project = 'FINCA' then 'Финка'
               when em.esm_project = 'Grintelecom' then 'ГринТелеком'
               when em.esm_project = 'Grintelecom' then 'ГринТелеком'
               when em.esm_project = 'IPC' then 'IPC'
               when em.esm_project = 'keremet' then 'Керемет'
               when em.esm_project = 'kicb' then 'KICB'
               when em.esm_project = 'kicb_paysoft' then 'KICB'
               when em.esm_project = 'mobilnik' then 'Мобильник'
               when em.esm_project = 'Pay24' then 'Pay24'
               when em.esm_project = 'PayBoxAcq' then 'PayBoxAcq'
               when em.esm_project = 'Quickpay' then 'Quickpay'
               when em.esm_project = 'svetofor' then 'svetofor'
               when em.esm_project = 'umai' then 'umai'
               when em.esm_project = 'Onoi' then 'Onoi'
               when em.esm_project = 'website' then 'website'
               else 'Кассовый приходный'
             end
      from operlist ol
      left join extsysoper eo on eo.eso_from_ol_id in (ol_id, ol.ol_parent_id)
      left join extsysmain em on eo.eso_from_esm_id = em.esm_id
      where ol.ol_id = :ol_id
      into ol_doc_client_name, c_id, ol_doc_kind_txt;

      ol_doc_kind_txt = coalesce(ol_doc_kind_txt, '');
      ol_doc_client_name = coalesce(ol_doc_client_name, '');

      cl_fio = '';
      select c_name_fio
      from clients c
      where c_id = :c_id
      into cl_fio;

      k_kontract_no = null;
      k_amount_approved = null;
      k_type = null;
      pr_name = null;
      mo_fio = null;
      select k_from_kf_id, k_kontract_no, k_amount_approved, k_disb_date,
             iif(pr_code containing '.ipf.', 'ИПФ', 'Классика') as k_type, --
             pr_name, --
             c_name_fio as mo_fio
      from kontract k
      inner join product pr on k_from_pr_id = pr_id
      left join clients c on c.c_id = k.k_approved_manager
      where k_id = :k_id
      into :kf_id, :k_kontract_no, :k_amount_approved, :k_disb_date, :k_type, :pr_name, :mo_fio;

      lo_fio = null;
      select c_name_fio
      from knaktlist kal
      inner join clients c on kal.kal_new_kf_officer = c_id
      where kal.kal_from_kf_id = :kf_id and
            kal.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end
      into :lo_fio;

      c_tel = null;
      select cont_value
      from hp_contdata_get_cell_valid(0, :c_id)
      into :c_tel;
      c_tel = coalesce(c_tel, 'Не указан');

      suspend;
    end
  end
end
