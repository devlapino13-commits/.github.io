CREATE PROCEDURE HR_PSTATUS_UPDATE(
    C_ID DECIMAL(18,0))
AS
declare variable cm_database_id integer;
declare variable min_d date;
declare variable max_d date;
declare variable d date;
declare variable d_prior date;
declare variable chet integer;
declare variable k_no decimal(18,0);
declare variable k_num integer;
declare variable reg integer;
declare variable reg_prior integer;
begin
  /* Procedure Text */

  :cm_database_id = f_get_db_id();
  select min(ps_date_begin), max(ps_date_begin)
  from hr_pstatus, hr_orders, sa_worktype
  where ps_from_c_id = :c_id and
        ps_from_ho_id = ho_id and
        ho_order_type = sawt_id and
        sawt_base_type = 1 and
        ps_is_on_staff = 1 and
        sawt_id = 1
  into :min_d, :max_d;

  d = iif(cm_database_id in (41701, 94401), max_d, min_d);

  if (d is null) then
  begin
    update clients
    set c_hired_date = :d
    where c_id = :c_id and
          c_hired_date is not null;
  end
  else
  begin
    update clients
    set c_hired_date = :d
    where c_id = :c_id and
          (c_hired_date is null or
          c_hired_date <> :d);
  end

  reg_prior = 0;
  chet = 0;
  for select ps_date_begin, ps_from_ho_id, ps_id
      from hr_pstatus
      where ps_from_c_id = :c_id and
            ps_status = 1 -- Только по утвержденным приказам
      order by ps_date_begin desc, ps_is_working desc, ps_from_ho_id desc, ps_id
      into :d, :k_no, :k_num
  do
  begin
    chet = :chet + 1;
    if (chet > 1) then
    begin
      update hr_pstatus
      set ps_date_end = :d_prior - 1
      where ps_from_c_id = :c_id and
            ps_id = :k_num and
            ps_from_ho_id = :k_no and
            ps_date_end <> :d_prior - 1;
    end
    else
    begin
      update hr_pstatus
      set ps_date_end = '31.12.9999'
      where ps_from_c_id = :c_id and
            ps_id = :k_num and
            ps_from_ho_id = :k_no and
            ps_date_end <> '31.12.9999';
    end
    d_prior = d;
  end

end
