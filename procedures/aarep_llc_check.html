CREATE PROCEDURE AAREP_LLC_CHECK(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    LM_ID DECIMAL(18,0),
    LM_DATE DATE,
    LM_DATE_STATUS DATE,
    LM_NO INTEGER,
    LM_MEMO VARCHAR(2048),
    LM_NO_STR VARCHAR(50),
    REG_NAME VARCHAR(100),
    REG_ID INTEGER,
    LM_IS_SIGNED_STR VARCHAR(30),
    LM_KONT_KOLICH INTEGER,
    LM_BALANCE_CHECK VARCHAR(30),
    LM_LLC_BALANCE DECIMAL(15,2),
    BALANCE_OFFICE DECIMAL(18,4),
    LM_DATE_CHECK VARCHAR(30))
AS
declare variable lm_is_signed smallint;
begin
  if (rep_region_id = 0) then
    rep_region_id = 1000000;

  for select reg_unicode, reg_code || ' ' || reg_name
      from region, region_orig_list(:rep_region_id, 1, 3, 1)
      where reg_unicode = id_list
      order by reg_unicode
      into :reg_id, :reg_name
  do
  begin
    balance_office = 0;
    lm_llc_balance = 0;
    lm_id = null;
    lm_date = null;
    lm_date_status = null;
    lm_no = null;
    lm_memo = '';
    lm_no_str = '';
    lm_is_signed = null;
    lm_kont_kolich = 0;
    select lm_id, lm_date, lm_date_status, lm_no, lm_memo, lm_no_str, lm_is_signed
    from llc_main
    where lm_from_reg_id = :reg_id and
          lm_date_status <= :rep_date
    order by lm_date_status desc
    rows 1
    into :lm_id, :lm_date, :lm_date_status, :lm_no, :lm_memo, :lm_no_str, :lm_is_signed;

    if (lm_is_signed is null) then
      lm_is_signed_str = 'Не создан';
    if (lm_is_signed = 0) then
      lm_is_signed_str = 'В процессе';
    if (lm_is_signed = 1) then
      lm_is_signed_str = 'Утвержден';

    if (lm_id is not null) then
    begin
      select count(lk_from_k_id), sum(lk_balance)
      from llc_kont
      where lk_from_lm_id = :lm_id
      into :lm_kont_kolich, :lm_llc_balance;
    end

    select sum(ac_k_balance)
    from aarep_kontract_active_temp(:reg_id, :lm_date_status, '000', 0)
    into :balance_office;
    if (balance_office is null) then
      balance_office = 0;

    if (((balance_office - lm_llc_balance) >= 1) or ((balance_office - lm_llc_balance) <= -1)) then
      lm_balance_check = 'Портфель не совпадает';
    else
      lm_balance_check = 'Совпадает';

    if (lm_kont_kolich is null) then
      lm_kont_kolich = 0;
    if (rep_date <> lm_date_status) then
      lm_date_check = 'Неверная дата протокола';
    else
      lm_date_check = 'Совпадает';

    suspend;
  end

end
