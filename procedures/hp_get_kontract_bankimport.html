CREATE PROCEDURE HP_GET_KONTRACT_BANKIMPORT(
    PURPOSE_PAYMENT VARCHAR(1024),
    DT_PAYMENT DATE NOT NULL = current_date)
RETURNS (
    K_ID ID_TYPE,
    K_NO STR_20,
    C_ID ID_TYPE,
    C_NAME STR_100,
    P_AMOUNT CURR_TYPE,
    P_COMMISSION CURR_TYPE,
    KS_STATE INTEGER,
    NOT_EXIST_DEBT INTEGER)
AS
declare variable pos1 integer;
declare variable pos2 integer;
declare variable pos3 integer;
declare variable numb integer;
declare variable strg str_100;
declare variable c_id_tmp id_type;
declare variable c_name_tmp str_100;
begin
  k_id = null;
  k_no = '';
  c_id = null;
  c_name = '';
  p_amount = 0;
  p_commission = 0;
  ks_state = 0;

  pos1 = position(' от ', purpose_payment);

  -- выделяем номер договора
  if (pos1 > 19) then
  begin
    -- выделяем номер договора
    strg = substring(purpose_payment from pos1 - 19 for 19);
    pos2 = maxvalue(position(' ', strg), position('№', strg), position('.', strg));
    if (pos2 > 0) then
    begin
      strg = substring(strg from pos2 + 1 for 19);
      if ((position('-', strg) = 0) and
          (char_length(strg) >= 12)) then
      begin
        strg = overlay(strg placing '-' from 5 for 0);
        strg = overlay(strg placing '-' from 12 for 0);
      end
      if (char_length(strg) >= 14) then
        k_no = trim(strg);
    end
  end

  -- Если не нашли договор выше, ищем через регулярное выражение
  if (k_no = '') then
  begin
    k_no = substring(purpose_payment similar '%#"[[:digit:]]{4}#-[[:digit:]]{6,7}#-[[:digit:]]{2}#"%' escape '#');
    k_no = coalesce(k_no, '');
  end

  -- выделяем ФИО контрагента
  if (pos1 > 0) then
  begin
    pos3 = position(' от ', purpose_payment, pos1 + 1);
    if (pos3 > 0) then
      pos1 = pos3;
    pos2 = position(';', purpose_payment);
    if ((pos2 > 0) and
        (pos2 > pos1 + 4)) then
      c_name = substring(purpose_payment from pos1 + 4 for pos2 - pos1 - 4);
  end

  -- выделяем сумму
  pos1 = position('СУМ:', purpose_payment);
  if (pos1 > 0) then
  begin
    pos2 = position(';', purpose_payment, pos1 + 1);
    if (pos2 > pos1 + 4) then
      p_amount = cast(substring(purpose_payment from pos1 + 4 for pos2 - pos1 - 4) as curr_type);
  end

  -- выделяем комиссию
  pos1 = position('КОМ:', purpose_payment);
  if (pos1 > 0) then
  begin
    pos2 = position(';', purpose_payment, pos1 + 1) - 1;
    if (pos2 < 0) then
      pos2 = char_length(purpose_payment);
    p_commission = cast(substring(purpose_payment from pos1 + 4 for pos2 - pos1 - 3) as curr_type);
  end

  -- поиск k_id, c_id ...
  if (k_no <> '') then
    for select k_id, c_id, replace(trim(c_name_fio), '  ', ' '), 2 - ks_state
        from kontract
        inner join kontstate on ks_from_k_id = k_id and
              ks_ol_is_delete = 0 and
              ks_state > 0 and
              current_date between cast(ks_date_begin as date) and cast(ks_date_end as date)
        inner join knclient on kc_from_k_id = k_id and
              kc_amount_approved > 0
        inner join clients on c_id = kc_from_c_id
        where k_kontract_no = :k_no
        into :k_id, :c_id_tmp, :c_name_tmp, :ks_state
    do
    begin
      if ((c_name_tmp containing c_name) or (c_name containing c_name_tmp)) then
      begin
        c_id = c_id_tmp;
        c_name = c_name_tmp;
      end
    end

  if ((k_id is null) and
      (c_name <> '')) then
  begin
    numb = 0;
    for select k_id, k_kontract_no, c_id, f_replace_all(trim(c_name_fio), '  ', ' '), 2 - ks_state
        from clients
        inner join knclient on kc_from_c_id = c_id and
              kc_amount_approved > 0
        inner join kontract on k_id = kc_from_k_id
        inner join kontstate on ks_from_k_id = k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              current_date between cast(ks_date_begin as date) and cast(ks_date_end as date)
        where c_name_fio = :c_name
        into :k_id, :k_no, :c_id_tmp, :c_name_tmp, :ks_state
    do
    begin
      numb = numb + 1;
      if (numb > 1) then
      begin
        k_id = null;
        c_id = null;
        leave;
      end
      else
      begin
        c_id = c_id_tmp;
        c_name = c_name_tmp;
      end
    end
  end

  -- Наличие задолженности
  if (k_id is null) then
    not_exist_debt = 1;
  else
    not_exist_debt = 1 - (select result
                          from hp_debt_check(:k_id, null, :dt_payment));

  suspend;
end
