CREATE PROCEDURE AAREP_SA_TABEL_PRINT_DETAIL2(
    REPORT_REGION INTEGER,
    REPORT_YEAR INTEGER,
    REPORT_MONTH INTEGER)
RETURNS (
    SAT_SYMBOL VARCHAR(10),
    SAT_NAME VARCHAR(100),
    SAT_NORMA DECIMAL(18,4),
    SAT_HOURS DECIMAL(18,4),
    SAT_DAYS_NORMA DECIMAL(18,4),
    SAT_DAYS_TOTAL DECIMAL(18,4))
AS
declare variable tnorm integer;
declare variable tnorm_days decimal(18,4) = 0;
begin
  select sum(scd_work_hours), sum(iif(scd_work_hours <> 0, 1, 0))
  from sa_calendays
  where scd_year_month = (:report_year * 100 + :report_month)
  into :tnorm, :tnorm_days;
  if (tnorm is null) then
    tnorm = 0;
  if (tnorm_days is null) then
    tnorm_days = 0;

  for select sawt_symbol, sawt_name, sum(iif(sat_from_sawt_id = 0, :tnorm, 0)), sum(sat_total_hours),
             sum(iif(sat_from_sawt_id = 0, :tnorm_days, 0)), sum(sat_days_total)
      from sa_tabel, region_list(:report_region, 1, -1, 1), region, sa_worktype
      where sat_year_month = (:report_year * 100 + :report_month) and
            (sat_office_id = id_list) and
            sat_is_ready = 1 and
            sawt_id = sat_from_sawt_id and
            sat_office_id = reg_unicode
      group by sawt_name, sawt_symbol
      into :sat_symbol, :sat_name, :sat_norma, :sat_hours, :sat_days_norma, :sat_days_total
  do
  begin
    suspend;
  end
end
