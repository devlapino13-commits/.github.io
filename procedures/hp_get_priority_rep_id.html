CREATE PROCEDURE HP_GET_PRIORITY_REP_ID(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID)
RETURNS (
    PRIORITY_ID D_SMALLINT)
AS
declare variable JSON_BLOB D_BLOB_TEXT;
declare variable REPORT_COUNT D_INTEGER;
declare variable IDX D_INTEGER;
declare variable PRIOR_LIST D_STR_20;
declare variable prior D_STR_10;
declare variable POS D_INTEGER;
declare variable SELECTED_IMPORTCODE D_STR_10;
declare variable BLS_ID D_ID;
declare variable C_ID D_ID;
begin

  if (:obj_kind = f_objkind('clients', 'id')) then
    c_id = :obj_id;
  else
  if (:obj_kind = f_objkind('miniapp', 'id')) then
    select mac.mac_from_c_id
    from miniappclient mac
    where mac.mac_from_ma_id = :obj_id and
          mac.mac_amount > 0
    into :c_id;
  else
  if (:obj_kind = f_objkind('kontract', 'id')) then
    select kc.kc_from_c_id
    from knclient kc
    where kc.kc_from_k_id = :obj_id
    into :c_id;
  else
    exit;

  select first 1 eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eic.eic_code = 'pkb_available_reports' and
        eil.eil_object_kind = f_objkind('clients', 'id') and
        eil.eil_object_id = :c_id and
        eil.eil_status = 2
  order by eil.eil_status_datetime desc
  into :bls_id;

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into :json_blob;

  report_count = f_json_str_get_count(:json_blob, 'Report');

  -- список приоритетов
  prior_list = '6,4,2,7';
  --prior_list = '0,9,17,0'; - для теста

  while (prior_list is not null) do
  begin
    pos = position(',' in prior_list);
    if (pos > 0) then
    begin
      prior = substring(prior_list from 1 for pos-1);
      prior_list = substring(prior_list from pos+1);
    end
    else
    begin
      prior = prior_list;
      prior_list = null;
    end

    idx = 0;
    while (idx < report_count) do
    begin
      if (f_json_str_get_value(
            :json_blob,
            'Report[' || idx || '].importcode',
            ''
           ) = prior) then
      begin
        selected_importcode = prior;
        priority_id = cast(selected_importcode as d_smallint);
        suspend;
        exit;
      end
      idx = idx + 1;
    end
  end
  priority_id = -1; -- нужно для проверки до (отправить на проверку в пкб)
  suspend;
end
