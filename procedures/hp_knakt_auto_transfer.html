CREATE PROCEDURE HP_KNAKT_AUTO_TRANSFER(
    REP_LO_ID DECIMAL(18,0),
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    RES_TEXT VARCHAR(255),
    IS_ERROR SMALLINT)
AS
declare variable db_id integer;
begin

  db_id = f_get_db_id();
  res_text = '';
  is_error = 1;
  if (db_id = 81001) then
  begin
    execute procedure hp_knakt_auto_transfer_ru(:rep_lo_id, :rep_region_id, :rep_date)
        returning_values :res_text, :is_error;
  end
  if (db_id in (39801, 41701)) then
  begin
--    res_text='Авто-распределение не работает для этой страны';
    execute procedure hp_knakt_auto_transfer_ru(:rep_lo_id, :rep_region_id, :rep_date)
        returning_values :res_text, :is_error;
  end
  if (db_id = 94401) then
  begin
    res_text = 'Авто-распределение не работает для этой страны';
  end
  suspend;
end
