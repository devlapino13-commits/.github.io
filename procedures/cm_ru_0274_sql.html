CREATE PROCEDURE CM_RU_0274_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable cds_id d_id;
declare variable ma_office_id d_id;
declare variable c_citizen d_integer;
declare variable c_citizen_str d_str_255;
declare variable status_code d_integer;
declare variable block_date d_date;
declare variable lo_id d_id;
declare variable subject d_str_255;
declare variable body1 d_str_long;
declare variable c_name_fio d_str_100;
declare variable error_text d_str_255;
declare variable c_birth_date d_date;
declare variable msg_body d_str_long;
declare variable res_txt d_str_long;
begin
  for select mac_from_c_id, c_citizen, ma_expert_id, c_name_fio, c_birth_date, ma_office_id
  from miniapp, miniappclient, clients
  where ma_id = :ma_id
    and ma_is_active = 1
    and ma_date_end = '31.12.9999'
    and ma_id = mac_from_ma_id
    and mac_amount > 0
    and mac_from_c_id = c_id
  into c_id, c_citizen, lo_id, c_name_fio, c_birth_date, ma_office_id
  do
  begin
    c_citizen_str = '';

    select coalesce(rd_code, '')
    from refdata
    where rd_id = :c_citizen
    into c_citizen_str;

    if (c_citizen_str <> 'Citizen.RU') then
      continue;

    status_code = 0;
    error_text = '';

    select coalesce(status_code, 0), error_text, cds_id,
      coalesce(status_desc || ' ', '') || coalesce(status_code || ' ', '') || coalesce(invalidity_reason_desc || ' ', '') ||
      coalesce(invalidity_reason_code || ' ', '') || f_datetostr(invalidity_since) || ' ' || coalesce(error_text, '')
    from hp_smevpass_get_result(:c_id, :ma_id, current_date)
    into status_code, error_text, cds_id, res_txt;

    if (error_text <> '') then
    begin
      res_error_is_critical = 2;
      res_error_text = 'СМЭВ. Проверка паспорта: ' || error_text;
      suspend;
      exit;
    end

    if (status_code <> 300) then
    begin
      res_error_is_critical = 2;

      /*1.4. При переводе заявки в договор, если срабатывает блок 2, но клиент не блокирован (основания для блока отсутствуют),
             то в заявке нужно отказать по причине «Сомнения в подлинности документов клиента» и отправить ПУШ специалисту
             «Паспорт клиента недействителен. В займе отказано.» */

      block_date = null;

      select cast(sbc_block_datetime as date)
      from smevpass_blockclients
      where current_date between cast(sbc_block_datetime as date) and cast(coalesce(sbc_unblock_datetime, '31.12.9999') as date)
        and sbc_from_c_id = :c_id
      into block_date;

      if (block_date is null) then
      begin
        res_error_text = 'Сомнения в подлинности документов клиента.';

        insert into kontreslist (krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id, krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
        values (1, 1000000000000, null, 0, :ma_id, current_timestamp, current_date, current_date, 0);

        execute procedure hp_miniapp_status_change(:ma_id, 1000000000000, 0);

        -- отправка ПУШ специалисту
        subject = 'Паспорт клиента недействителен. В займе отказано.';
        body1 = subject||' Килент:'||c_name_fio;

        execute procedure hp_send_push_to_staff(:lo_id, :subject, :body1, :un_id, '', 7);

        --RFCRU-4804
        msg_body = f_format('Клиент: %1, дата рождения: %2, паспорт: %3, офис: %4, результат проверки: %5',
                          c_name_fio,
                          f_datetostr(c_birth_date),
                          (select cds_doc_no from clientdocs where cds_id = :cds_id),
                          (select reg_name from region r where reg_unicode = :ma_office_id),
                          res_txt);

        insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id, msg_command_kind)
        values (0, 'ru.monitoring@molbulak.com;zara@molbulak.com', 'СМЭВ: Недействительный паспорт', :msg_body, 7, 1000000000001, '');
      end
      else if (status_code = 301) then
      begin
        res_error_text = 'Паспорт клиента недействителен. В займе отказано.';
        execute procedure hp_smevpass_ma_reject(:ma_id);
      end 
      else
        res_error_text = 'Действительность паспорта не подтверждена. Откажите в заявке';

      suspend;
    end
  end
end
