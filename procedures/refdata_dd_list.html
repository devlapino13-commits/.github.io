CREATE PROCEDURE REFDATA_DD_LIST(
    CODE_FILTER VARCHAR(255),
    WITH_ZERO INTEGER)
RETURNS (
    DD_ID INTEGER,
    RD_PARENT_ID INTEGER,
    RD_CODE VARCHAR(255),
    RD_MAIN_CODE VARCHAR(255),
    DD_NAME VARCHAR(255),
    RD_MEMO VARCHAR(255),
    RD_STR_ID VARCHAR(100),
    DD_ORDER INTEGER,
    DD_IS_VISIBLE INTEGER,
    RD_IS_LOCKED INTEGER,
    RD_TAG INTEGER,
    IS_DELETE INTEGER,
    DD_CATEGORY INTEGER)
AS
begin
  if (code_filter is null) then
    code_filter = '';
  if (code_filter <> '') then
  begin
    for select rd_id, rd_parent_id, rd_code, rd_main_code, rd_name, rd_memo, rd_str_id, rd_orderno, rd_is_active,
               rd_is_locked, rd_tag
        from refdata
        where rd_main_code = :code_filter and
              rd_parent_id is not null
        into :dd_id, :rd_parent_id, :rd_code, :rd_main_code, :dd_name, :rd_memo, :rd_str_id, :dd_order, :dd_is_visible,
             :rd_is_locked, :rd_tag
    do
    begin
      is_delete = 1 - dd_is_visible;
      dd_category = -rd_parent_id;
      suspend;
    end
    if (with_zero = 1) then
    begin
      for select rd_id, rd_parent_id, rd_code, rd_main_code, rd_name, rd_memo, rd_str_id, rd_orderno, rd_is_active,
                 rd_is_locked, rd_tag
          from refdata
          where rd_id = 0
          into :dd_id, :rd_parent_id, :rd_code, :rd_main_code, :dd_name, :rd_memo, :rd_str_id, :dd_order,
               :dd_is_visible, :rd_is_locked, rd_tag
      do
      begin
        dd_category = -rd_parent_id;
        is_delete = 1 - dd_is_visible;
        suspend;
      end
    end
  end
end
