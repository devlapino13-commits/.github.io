CREATE PROCEDURE IPC_TOKEN_ACTION(
    C_ID D_ID,
    ACTION_CODE D_STR_CODE,
    REQUEST_TXT D_STR_8100,
    ANSWER_TXT D_STR_8100)
AS
declare variable TOKEN D_STR_CODE;
declare variable TOKEN_ID D_STR_CODE;
declare variable CARD_NAME STR_100;
declare variable CARD_LAST4 D_STR_10;
declare variable ET_ID D_ID;
declare variable ACTION_ID D_INTEGER;
declare variable ERR_CODE D_INTEGER;
declare variable CARD_EXPIRE D_STR_10;
declare variable BANK_ID D_INTEGER;
declare variable TOKEN_STATE D_INTEGER;
begin
  --процедура для реализации проекта Токенизация карт в КГ

  -- сам токен хранится в т. extsys_token
  -- действия с токеном в т. extsys_token_actions (хвост для хранения запросов в МПЦ и ответов из МПЦ)
  action_id = f_enum_id('ipc_token_actions', :action_code);
  if (action_id is null) then
    exception e_default 'Передан не существующий action_code!';

  err_code = f_json_str_get_value(:answer_txt, 'code', 0);
  token_id = f_json_str_get_value(:request_txt, 'token_id', '');

  if (action_code <> 'new_token') then
  begin
      select et_id
      from extsys_token et
      inner join extsys_token_add eta on eta.eta_from_et_id = et.et_id
      where et.et_from_c_id = :c_id and
            eta.eta_token_id = :token_id
      into et_id;
  end

  if (err_code = 0) then
  begin
       -- Создание нового токена
      if (action_code = 'new_token') then
      begin
        token = f_json_str_get_value(:answer_txt, 'token', '');
        card_last4 = f_json_str_get_value(:answer_txt, 'last4', '');
        card_name = f_json_str_get_value(:answer_txt, 'card_name', '');
        card_expire = left(f_json_str_get_value(:answer_txt, 'expiry', ''), 10);
        bank_id = f_json_str_get_value(:answer_txt, 'bank_id', 0);
        token_state = f_json_str_get_value(:answer_txt, 'state', 0);
        
        et_id = null;

        insert into extsys_token (et_from_c_id, et_token, et_is_actual, et_cardnumber)
        values (:c_id, :token, 0, :card_last4)
        returning et_id
        into :et_id;
        
        insert into extsys_token_add (eta_from_et_id, eta_token_id, eta_name_on_card, eta_expiry, eta_bank_id, eta_state)
        values (:et_id, :token_id, :card_name, :card_expire, :bank_id, :token_state);
      end
      -- Подтверждение токена
      else if (action_code = 'confirmation_save') then
      begin
        update extsys_token et
        set et.et_code = f_json_str_get_value(:request_txt, 'verification_value', ''),
            et.et_is_actual = 1
        where et.et_id = :et_id;

        update extsys_token_add eta
        set eta.eta_state = f_enum_id('card_token_state', 'confirmed')
        where eta.eta_from_et_id = :et_id;
      end
      -- Удаление токена
      else if (action_code = 'token_delete') then
      begin
        update extsys_token et
        set et.et_is_actual = -1
        where et.et_id = :et_id;

        update extsys_token_add eta
        set eta.eta_state = f_enum_id('card_token_state', 'deleted')
        where eta.eta_from_et_id = :et_id;
      end
  end

  -- Сохранение действия в таблицу
  insert into extsys_token_actions (eta_from_c_id, eta_from_et_id, eta_action_id, eta_request_txt, eta_answer_txt)
  values (:c_id, :et_id, :action_id, :request_txt, :answer_txt);
end
