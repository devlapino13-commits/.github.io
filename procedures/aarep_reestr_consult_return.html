CREATE PROCEDURE AAREP_REESTR_CONSULT_RETURN(
    REP_REG D_INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    C_FIO D_STR_100,
    REG_NAME VARCHAR(50),
    K_NO D_STR_20,
    CONSULT_APP_DATE DATE,
    AMOUNT D_CURRENCY,
    CONSULT_PAYMENT_DATE DATE,
    DATE_APP DATE,
    SIGN_TYPE D_STR_20,
    RETURNED_DATE DATE)
AS
declare variable reg_id d_integer;
declare variable rd_id d_integer;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable consult_office_id d_integer;
declare variable consult_office_id_tmp d_integer;
begin

  rd_id = f_get_rd_id('CashTransactionIncom.Consultation');
  consult_office_id = 0;
  reg_id = 0;

  for select id_list
      from region_list(:rep_reg, 1, 3, 1)
      into :reg_id

  do
  begin

    for select c.c_name_fio, arci.arci_consult_office_id, k.k_kontract_no, arci.arci_amount, arc.arc_date,
               case arci.arci_sign_type
                 when 0 then 'Электронная подпись'
                 when 1 then 'Физическая подпись'
               end,
               cast(arc.arc_status_datetime as date), c.c_id, k.k_id
        from app_return_consult arc
        inner join app_return_consult_info arci on arc.arc_id = arci.arci_id and
              arci.arci_app_office_id = :reg_id
        inner join clients c on arc.arc_from_c_id = c.c_id
        inner join kontract k on k.k_id = arc.arc_from_k_id
        where arc.arc_status = f_enum('app_return_consult', 'done', 'id') and
              arc.arc_date between :rep_date_begin and :rep_date_end
        order by 2
        into :c_fio, :consult_office_id_tmp, :k_no, :amount, :date_app, :sign_type, :returned_date, :c_id, :k_id

    do
    begin

      if (:consult_office_id_tmp <> :consult_office_id) then
      begin
        consult_office_id = :consult_office_id_tmp;
        select reg.reg_name
        from region reg
        where reg.reg_unicode = :reg_id
        into :reg_name;
      end

      select ol.ol_date
      from operkontract ok
      inner join operlist ol on ok.ok_from_ol_id = ol.ol_id and
            ol.is_delete = 0
      inner join knclient kc on kc.kc_from_k_id = ok.ok_from_k_id and
            kc.kc_from_c_id = :c_id
      where ok.ok_operation_type = :rd_id and
            ok.ok_amount > 0 and
            ok.ok_from_k_id = :k_id
      rows 1
      into :consult_payment_date;

      select cast(edl.edl_create_datetime as date)
      from edoc_list edl
      inner join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id and
            edc.edc_code = 'consulting.agreement'
      inner join knclient kc on kc.kc_from_c_id = :c_id
      inner join kontract k on k.k_id = kc.kc_from_k_id and
            k.k_id = :k_id
      inner join product pr on pr.pr_id = k.k_from_pr_id and
            pr.pr_code containing '.consultation.'
      where edl.edl_object_id = kc.kc_id and  -- RFCKG-4363
            edl.edl_status = 1 and
            edl.edl_object_kind = 0 and
            cast(edl.edl_create_datetime as date) = k.k_kontract_date
      into :consult_app_date;

      if (consult_app_date is null) then
          select cast(edl.edl_create_datetime as date)
          from edoc_list edl
          inner join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id and
                edc.edc_code = 'consulting.agreement'
          inner join knclient kc on kc.kc_from_c_id = :c_id
          inner join kontract k on k.k_id = kc.kc_from_k_id and
                k.k_id = :k_id
          inner join product pr on pr.pr_id = k.k_from_pr_id and
                pr.pr_code containing '.consultation.'
          where edl.edl_object_id = kc.kc_id and -- RFCKG-4363
                edl.edl_status = 1 and
                edl.edl_object_kind = 0
          into :consult_app_date;

      suspend;
    end

  end
end
