CREATE PROCEDURE PROC_CALC_AMORT(
    REG_ID INTEGER,
    YEAR_NO INTEGER,
    MONTH_NO INTEGER)
RETURNS (
    PCA_BILL_ID DECIMAL(18,0),
    PCA_BILL_LINE_NO INTEGER,
    PCA_START_YEAR INTEGER,
    PCA_START_MONTH INTEGER,
    PCA_YEAR_NO INTEGER,
    PCA_MONTH_NO INTEGER,
    PCA_NEW_AMORT DECIMAL(18,4),
    PCA_KOEF INTEGER,
    PCA_AGE INTEGER,
    PCA_TOTAL_OS DECIMAL(18,4),
    PCA_TOTAL_AMORT DECIMAL(18,4),
    PCA_LIFETIME INTEGER,
    PCA_CALC_TYPE SMALLINT)
AS
declare variable all_period integer;
declare variable need_period integer;
declare variable dont_calc smallint;
declare variable last_calc_period integer;
begin
  /* Procedure Text */
  for select oam_bill_id, oam_bill_line_no, oam_calc_type, oam_koef, oam_start_year, oam_start_month,
             sum(oal_lifetime_month), sum(oal_summa_os), sum(oal_summa_amort), sum(oal_dont_calc),
             max(oal_year * 100 + oal_month)
      from operamormain, operamorlist
      where oam_is_delete = 0 and
            oal_is_delete = 0 and
            oal_bill_id = oam_bill_id and
            oal_bill_line_no = oam_bill_line_no
            /* and ol_from_reg_id=:reg_id */
            and
            (oal_year * 12 + oal_month) <= (:year_no * 12 + :month_no) and
            (oam_start_year * 12 + oam_start_month) <= (:year_no * 12 + :month_no)
      group by oam_bill_id, oam_bill_line_no, oam_calc_type, oam_koef, oam_start_year, oam_start_month
      having sum(oal_summa_os) > sum(oal_summa_amort)
      into :pca_bill_id, :pca_bill_line_no, :pca_calc_type, :pca_koef, :pca_start_year, :pca_start_month, :pca_lifetime,
           :pca_total_os, :pca_total_amort, :dont_calc, :last_calc_period
  do
  begin
    pca_year_no = year_no;
    pca_month_no = month_no;
    pca_age = (year_no * 12 + month_no) - (pca_start_year * 12 + pca_start_month) + 1;
    pca_age = pca_age - dont_calc;
    if (pca_age < 0) then
      pca_age = 0;
    pca_new_amort = 0;

--    if (last_Calc_Period<(Year_No*100+Month_No)) then
--    begin
    /* МБП, 100% сразу */
    if (pca_calc_type = 1) then
      pca_new_amort = (pca_total_os - pca_total_amort);

    /* простой, линейный метод */
    if (pca_calc_type = 2) then
    begin
      /* Старый метод: Общую сумму делим на общий срок службы */
      /*
          if (pca_lifetime<>0)
          then pca_new_amort=pca_koef*pca_total_os/pca_lifetime;
          pca_new_amort=pca_new_amort*pca_age;
          pca_new_amort=cast(pca_new_amort as decimal(18,2));
          pca_new_amort=pca_new_amort-pca_total_amort;
          */
      /* Новый метод: Остаточную сумму (Первонач-Накоп.износ) делим на остаточный срок */
      if ((pca_lifetime - pca_age + 1) > 0) then
        pca_new_amort = pca_koef * (pca_total_os - pca_total_amort) / (pca_lifetime - pca_age + 1);
      else
        pca_new_amort = (pca_total_os - pca_total_amort);
      if (pca_new_amort < 0) then
        pca_new_amort = 0;
      pca_new_amort = cast(pca_new_amort as decimal(18,2));
    end

    /* ускоренний, сумма чисел */
    if (pca_calc_type = 3) then
    begin
      all_period = cast(((pca_lifetime * (pca_lifetime + 1)) / 2) as integer);
      need_period = (pca_lifetime - pca_age);
      need_period = cast(((need_period * (need_period + 1)) / 2) as integer);
      need_period = all_period - need_period;
      if (all_period <> 0) then
        pca_new_amort = pca_koef * pca_total_os / all_period;
      pca_new_amort = pca_new_amort * need_period;
      pca_new_amort = cast(pca_new_amort as decimal(18,2));
      pca_new_amort = pca_new_amort - pca_total_amort;
    end

    /* ускоренний, уменьшающий */
    if (pca_calc_type = 4) then
    begin
      if (pca_lifetime <> 0) then
        pca_new_amort = (pca_total_os - pca_total_amort) / pca_lifetime;
      else
        pca_new_amort = 0;
      pca_new_amort = cast(pca_new_amort as decimal(18,2));
    end

    /* контроль на минусовую амортизацию */
    if (pca_new_amort < 0) then
      pca_new_amort = 0;

    if ((pca_new_amort + pca_total_amort) > pca_total_os) then
      pca_new_amort = pca_total_os - pca_total_amort;

    if (pca_new_amort < 0) then
      pca_new_amort = 0;
    if (pca_new_amort > 0) then
      suspend;
  end
--  end
end
