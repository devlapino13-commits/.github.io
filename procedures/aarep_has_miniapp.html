CREATE PROCEDURE AAREP_HAS_MINIAPP(
    KF_ID DECIMAL(18,0),
    MA_OFFICE INTEGER)
RETURNS (
    MA_REG INTEGER,
    MA_ID INTEGER,
    MA_NUMBER INTEGER,
    MA_NO INTEGER,
    MA_DATE DATE,
    MA_KONT_NO VARCHAR(20),
    MA_STATE INTEGER,
    K_ID DECIMAL(18,0),
    MA_MEMO VARCHAR(255))
AS
declare variable ma_reg_prior integer;
declare variable ma_id_prior integer;
declare variable k_status integer;
begin
  for select ma_reg, ma_id, ma_number, ma_no, ma_date, ma_kont_no, ma_state, k_id
      from aarep_has_miniapp_tmp(:kf_id, :ma_office)
      order by ma_date desc
      rows 1
      into :ma_reg, :ma_id, :ma_number, :ma_no, :ma_date, :ma_kont_no, :ma_state, :k_id
  do
  begin
    if (ma_state = -1) then
      ma_memo = 'Существует действующая заявка в кредитной папке';
    if (ma_state = 0) then
      ma_memo = 'Существует действующий договор в процессе в этой кредитной папке';
    suspend;
  end
end
