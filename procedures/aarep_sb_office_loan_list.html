CREATE PROCEDURE AAREP_SB_OFFICE_LOAN_LIST(
    REP_DATE_END DATE)
RETURNS (
    OKS_REGION_ID INTEGER,
    OKS_FROM_K_ID DECIMAL(18,0),
    OKS_BALANCE DECIMAL(18,4),
    OKS_KC_ID DECIMAL(18,0),
    OKS_DELINQ_DATE DATE,
    OKS_SORT INTEGER,
    OKS_CATEGORY_ID INTEGER,
    OKS_DELINQ_DAYS INTEGER,
    OKS_LAST_OL_DATE DATE,
    OKS_LAST_OL_DAYS INTEGER,
    OKS_SBL_DATE DATE,
    OKS_SBL_C_ID ID_TYPE,
    OKS_SBL_FIO VARCHAR(100),
    OKS_SBL_ID INTEGER,
    OKS_ZAD_OS DECIMAL(18,4),
    OKS_CL_COUNT INTEGER,
    OKS_KF_NAME VARCHAR(100),
    OKS_LO_FIO VARCHAR(100),
    OKS_KONTRACT_NO VARCHAR(20),
    OKS_SBL_TYPE VARCHAR(20))
AS
declare variable def_curr varchar(3);
declare variable kf_id decimal(18,0);
declare variable sbl_type integer;
begin
-- необходимо отсортировать по oks_sort и сумма баланса

  select cm_default_currency
  from company
  into :def_curr;

  for select oks_region_id, k_kontract_no, oks_from_k_id, oks_balance, oks_kc_id, oks_delinq_date,
             k_kolich_client_approved, k_from_kf_id
      from oksummary, kontract
      where oks_delinq_date < :rep_date_end and
            oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_balance > 0 and
            oks_close_state = 0 and
            oks_from_k_id = k_id
      into :oks_region_id, :oks_kontract_no, :oks_from_k_id, :oks_balance, :oks_kc_id, :oks_delinq_date, :oks_cl_count,
           :kf_id
  do
  begin
    oks_kf_name = '';
    select kal_new_kf_name
    from knaktlist
    where kal_date <= :rep_date_end and
          kal_date_end >= :rep_date_end and
          kal_from_kf_id = :kf_id and
          knaktlist.is_delete = 0
    into :oks_kf_name;

    oks_lo_fio = '';
    select c_name_fio
    from clients
    where c_id = :oks_kc_id
    into :oks_lo_fio;

    oks_sort = 0;
    oks_delinq_days = datediff(day, :oks_delinq_date, :rep_date_end);
    oks_category_id = 1;

    if ((rep_date_end >= '01.12.2012') and
        (def_curr = '810')) then
    begin
      select aa_category_id
      from aarep_get_category_id(:oks_from_k_id, :rep_date_end, :def_curr, :oks_kc_id, null, null, null, :oks_delinq_days)
      into :oks_category_id;
    end
    if (oks_category_id in (1, 2)) then
      oks_sort = oks_sort + 1;

    if (oks_delinq_days > 30) then
      oks_sort = oks_sort + 1;

    oks_last_ol_date = null;
    select max(ok_ol_date)
    from operkred
    where ok_from_k_id = :oks_from_k_id and
          (ok_os + ok_perc + ok_fine) < 0 and
          ok_type = 4 and
          ok_from_c_id = 0
    into :oks_last_ol_date;

    oks_last_ol_days = datediff(day, oks_last_ol_date, rep_date_end);
    if (oks_last_ol_days > 10) then
      oks_sort = oks_sort + 1;

    oks_sbl_date = null;
    oks_sbl_c_id = null;
    oks_sbl_fio = '';
    oks_sbl_id = -1;
    oks_sbl_type = '';

    select sba_from_sbl_id, sba_date_begin, sbl_from_c_id, c_name_fio, sbl_type
    from sb_akt, sb_list, clients
    where sba_is_delete = 0 and
          sba_status = 2 and
          sba_from_k_id = :oks_from_k_id and
          sba_date_begin <= :rep_date_end and
          sba_date_end >= :rep_date_end and
          sba_from_sbl_id = sbl_id and
          sba_from_sbl_id <> 0 and
          sbl_id <> 0 and
          sbl_from_c_id = c_id
    into :oks_sbl_id, :oks_sbl_date, :oks_sbl_c_id, :oks_sbl_fio, :sbl_type;

    if ((oks_sbl_id is null) or (oks_sbl_id = -1)) then
      oks_sort = oks_sort + 1;
    if (oks_sbl_id > 0) then
    begin
      if (sbl_type = 0) then
        oks_sbl_type = 'СБ';
      else
        oks_sbl_type = 'Коллектор';
    end

    oks_zad_os = 0;
    select sum(ok_os + ok_sh_os)
    from operkred
    where ok_sh_date < :rep_date_end and
          ok_from_k_id = :oks_from_k_id and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date_end and
          ok_type < 8 and
          ok_from_c_id = 0
    into :oks_zad_os;
    if (oks_zad_os is null) then
      oks_zad_os = 0;

    if (oks_zad_os >= 10000) then
      oks_sort = oks_sort + 1;
    suspend;
  end
end
