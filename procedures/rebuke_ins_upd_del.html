CREATE PROCEDURE REBUKE_INS_UPD_DEL(
    RB_ID INTEGER,
    RL_TYPE INTEGER,
    RB_TYPE SMALLINT,
    RB_FROM_REG_ID INTEGER,
    USER_ID DECIMAL(18,0),
    RB_NUMBER VARCHAR(20),
    RB_DATE DATE,
    RB_FROM_C_ID DECIMAL(18,0),
    RB_BONUS_CHANGE SMALLINT,
    RB_BONUS_KOEF DECIMAL(18,2),
    RB_PERIOD_START DATE,
    RB_PERIOD_END DATE,
    RB_BODY VARCHAR(255),
    RB_REASON VARCHAR(255),
    RB_IS_DELETE SMALLINT,
    RB_NUMBER_N INTEGER,
    RB_SUB_NUMBER SMALLINT,
    RB_INITIATOR_ID DECIMAL(18,0),
    RB_NO_NUMBER SMALLINT,
    RS_SCORES D_SMALLINT = 0,
    RS_PROTOCOL_NO D_INTEGER = null,
    RS_PROTOCOL_DATE D_DATE = null,
    RS_EXPLANATORY_DATE D_DATE = null)
RETURNS (
    ERR_TEXT VARCHAR(755))
AS
declare variable buf_id integer;
declare variable buf_number varchar(20);
declare variable buf_date date;
declare variable buf_from_c_id decimal(18,0);
declare variable buf_from_reg_id integer;
declare variable buf_from_jl_id integer;
declare variable buf_type smallint;
declare variable buf_body varchar(500);
declare variable buf_reason varchar(500);
declare variable buf_bonus_change smallint;
declare variable buf_period_start date;
declare variable buf_period_end date;
declare variable buf_is_delete smallint;
declare variable buf_bonus_koef decimal(15,2);
declare variable memo varchar(2000);
declare variable buf_name_fio varchar(255);
declare variable buf_reg_name varchar(255);
declare variable buf_jl_name varchar(255);
declare variable user_c_id decimal(18,0);
declare variable buf_initiator_id decimal(18,0);
declare variable buf_no_number smallint;
declare variable db_id d_integer;
declare variable rs_scores_old d_smallint;
declare variable rs_protocol_no_old d_str_20;
declare variable rs_protocol_date_old d_date;
declare variable rs_explanatory_date_old d_date;
begin
  /*
    CREATED BY IGUSEV 02.08.2012
    PURPOSE : Модуль - "Регистрация выговоров и замечаний"
              Производит добавление, Редактирование, Удаление (is_delete = 1) документов.
  */

  /* Проверка корректности входящих параметров --------------------------------------------*/
  err_text = 'Ошибки:';
  if ((rb_id is null) or (rb_id = 0)) then
    err_text = err_text || ' ID документа не определен;';

  if (rb_number is null) then
    rb_number = '';
  else
    rb_number = trim(rb_number);

  if (rb_from_reg_id is null) then
    rb_from_reg_id = 0;
  if (user_id is null) then
    user_id = 0;
  if (rl_type is null) then
    rl_type = 0;
  if (rb_type is null) then
    rb_type = 0;
  if (rb_from_c_id is null) then
    rb_from_c_id = 0;
  if (rs_scores is null) then
    rs_scores = 0;

  if (user_id = 0) then
    err_text = err_text || ' Пользователь не определен;';
  if (rl_type = 0) then
    err_text = err_text || ' Тип операции не определен;';

  db_id = f_get_db_id();

  if (db_id in (39801, 41701)) then
    rb_bonus_change = 1;

  if (rl_type in (1, 2)) then
  /* если это не удаление*/
  begin
    /*
      if (rb_no_number  = 0) then
      begin
          if (exists(select rb_number
                     from rebuke rb
                     where rb.rb_number = :RB_NUMBER
                     and RB_ID <> :RB_ID
                     and rb.rb_is_delete = 0)) then
          err_text = err_text||' Документ с таким номером уже существует;';
      end
      */

    if (rb_from_c_id = 0) then
      err_text = err_text || ' Сотрудник не определен;';
    if (rb_from_reg_id = 0) then
      err_text = err_text || ' Офис документа не определен;';
    if (rb_type = 0) then
      err_text = err_text || ' Тип документа не определен';

    if (rb_date is null) then
      rb_date = '30.12.1899';
    if (rb_bonus_change = 1 and
        db_id not in (39801, 41701)) then
    /* влияет на бонусы */
    begin
      if (rb_period_start is null) then
        rb_period_start = '30.12.1899';
      if (rb_period_end is null) then
        rb_period_end = '30.12.1899';
      if (rb_bonus_koef is null) then
        rb_bonus_koef = 0;

      if (rb_bonus_koef < 0) then
        err_text = err_text || ' Коэффицент влияния на бонус не может быть меньше 0';

    end
    if (rb_bonus_change = 0) then
    /* не влияет на бонусы */
    begin
      if (rb_period_start is not null) then
        rb_period_start = null;
      if (rb_period_end is not null) then
        rb_period_end = null;
      if (rb_bonus_koef is not null) then
        rb_bonus_koef = 0;
    end

    /* Для устных замечаний номер докмента - не обязательное поле */
    if ((rb_no_number = 0) and
        (rb_number = '')) then
      err_text = err_text || ' Не указан номер документа;';

    if (rb_no_number = 1) then
    begin
      rb_number_n = 0;
      rb_sub_number = 0;
    end

    if (rb_date = '30.12.1899') then
      err_text = err_text || ' Не указана дата документа;';
    /* else
      if (RB_DATE < '01.01.2012')  then
        err_text = err_text||' Не корректная дата документа';
      */
    if (rb_bonus_change = 1 and
        db_id not in (39801, 41701)) then
    begin
      if ((rb_period_start = '30.12.1899') or (rb_period_end = '30.12.1899') or (rb_period_start > rb_period_end)) then
        err_text = err_text || ' Не корректный период влияния на бонус';

      if (rb_bonus_koef = 0) then
        err_text = err_text || ' Не корректный коэфицент влияния на бонус. Должен отличаться от 0';
    end
  end
  if (((rl_type = 3) or (rl_type = 2)) and
      rb_id = 0) then
    err_text = err_text || ' Документ не определен!';
  if (((rl_type = 3) or (rl_type = 2)) and
      rb_id <> 0) then
  begin
    if (not exists(select rb.rb_id
                   from rebuke rb
                   where rb.rb_id = :rb_id rows 1)) then
      err_text = err_text || ' Документа с id = ' || cast(rb_id as varchar(20)) || ' не существует.';
  end

  if (rb_initiator_id = 0) then
    rb_initiator_id = null;
  /*----------------------------------------------------------------------------------------------------*/

  /*
  if (rl_type in (1,2)) then
  begin
     select cl.c_id
     from username un
     inner join clients cl on cl.c_id = un.un_from_c_id
     where un.un_id = :USER_ID
     into :user_c_id;
  end
  */

  if (err_text = 'Ошибки:') then
  begin
    if (rl_type = 3) then
    /* DELETE*/
    begin
      update rebuke
      set rb_is_delete = 1
      where (rb_id = :rb_id);

      insert into rebuke_log (rl_from_rb_id, rl_from_un_id, rl_type, rl_memo)
      values (:rb_id, :user_id, :rl_type, 'Документ удален');

      if (db_id in (39801, 41701)) then
        update rebuke_scores
        set rs_from_un_id = :user_id,
            rs_is_delete = 1
        where (rs_from_rb_id = :rb_id);
    end

    if (rl_type = 1) then
    /* INSERT */
    begin

      insert into rebuke (rb_id, rb_number, rb_date, rb_from_c_id, rb_from_reg_id, rb_type, rb_body, rb_reason,
                          rb_bonus_change, rb_period_start, rb_period_end, rb_is_delete, rb_bonus_koef, rb_number_n,
                          rb_sub_number, rb_initiator_id, rb_no_number)
      values (:rb_id, :rb_number, :rb_date, :rb_from_c_id, :rb_from_reg_id, :rb_type, :rb_body, :rb_reason,
              :rb_bonus_change, :rb_period_start, :rb_period_end, :rb_is_delete, :rb_bonus_koef, :rb_number_n,
              :rb_sub_number, :rb_initiator_id, :rb_no_number);

      insert into rebuke_log (rl_from_rb_id, rl_from_un_id, rl_type, rl_memo)
      values (:rb_id, :user_id, :rl_type, 'Документ создан');

      if (db_id in (39801, 41701)) then
        insert into rebuke_scores (rs_from_rb_id, rs_from_c_id, rs_scores, rs_from_un_id, rs_is_delete, rs_date, rs_protocol_no, rs_protocol_date, rs_explanatory_date)
        values (:rb_id, :rb_from_c_id, :rs_scores, :user_id, :rb_is_delete, :rb_date, :rs_protocol_no, :rs_protocol_date, :rs_explanatory_date);
    end
    if (rl_type = 2) then
    /* UPDATE*/
    begin
      /* Проверяю изменения ------------------------------------------------------------------------------------- >>> */

      select rb_number, rb_date, rb_from_c_id, rb_from_reg_id, rb_type, rb_body, rb_reason, rb_bonus_change,
             rb_period_start, rb_period_end, rb_is_delete, rb_bonus_koef, rb_initiator_id, rb.rb_no_number
      from rebuke rb
      where rb.rb_id = :rb_id
      into :buf_number, :buf_date, :buf_from_c_id, :buf_from_reg_id, :buf_type, :buf_body, :buf_reason,
           :buf_bonus_change, :buf_period_start, :buf_period_end, :buf_is_delete, :buf_bonus_koef, :buf_initiator_id,
           buf_no_number;

      memo = 'Изменения:';

      if (buf_number <> rb_number) then
        memo = memo || ' номер документа (был:' || buf_number || ');';
      if (buf_date <> rb_date) then
        memo = memo || ' дата документа (была:' || cast(buf_date as varchar(20)) || ');';
      if (buf_no_number <> rb_no_number) then
        memo = memo || ' Без номера (был:' || cast(buf_no_number as varchar(20)) || ');';

      if (buf_from_c_id <> rb_from_c_id) then
      begin
        select cl.c_name_fio
        from clients cl
        where cl.c_id = :buf_from_c_id
        into :buf_name_fio;
        memo = memo || ' сотрудник (был:' || buf_name_fio || ');';
      end

      if (buf_from_reg_id <> rb_from_reg_id) then
      begin
        select reg.reg_name
        from region reg
        where reg.reg_unicode = :buf_from_reg_id
        into :buf_reg_name;
        memo = memo || ' офис документа (был:' || buf_reg_name || ');';
      end

      if (buf_type <> rb_type) then
      begin
        if (buf_type = 1) then
          memo = memo || ' тип документа (был: выговор);';
        if (buf_type = 2) then
          memo = memo || ' тип документа (был: замечание);';
        if (buf_type = 3) then
          memo = memo || ' тип документа (был: устное замечание);';
      end

      if (buf_body <> rb_body) then
      begin
        memo = memo || ' Содержание (было:' || buf_body || ');';
      end

      if (buf_reason <> rb_reason) then
        memo = memo || ' Основание (было:' || buf_reason || ');';
      if (buf_bonus_change <> rb_bonus_change) then
      begin
        if (buf_bonus_change = 0) then
          memo = memo || ' Влияние на бонус (было: не влияет);';
        if (buf_bonus_change = 1) then
          memo = memo || ' Влияние на бонус (было: влияет);';
      end
      if (buf_period_start <> rb_period_start) then
        memo = memo || ' Начало периода (было: ' || cast(buf_period_start as varchar(255)) || ');';
      if (buf_period_end <> rb_period_end) then
        memo = memo || ' Конец периода (был: ' || cast(buf_period_end as varchar(255)) || ');';
      if ((buf_period_start is null) and
          (rb_period_start is not null)) then
        memo = memo || ' Начало периода (не было указано);';
      if ((buf_period_end is null) and
          (rb_period_end is not null)) then
        memo = memo || ' Конец периода (не был указан);';
      if ((buf_period_start is not null) and
          (rb_period_start is null)) then
        memo = memo || ' Начало периода (было: ' || cast(buf_period_start as varchar(255)) || ');';
      if ((buf_period_end is not null) and
          (rb_period_end is null)) then
        memo = memo || ' Конец периода (был: ' || cast(buf_period_start as varchar(255)) || ');';

      if (buf_bonus_koef <> rb_bonus_koef) then
        memo = memo || ' Коэффицент (было: ' || buf_bonus_koef || ');';

      if (buf_initiator_id is null) then
        buf_initiator_id = 0;
      if (buf_initiator_id <> rb_initiator_id) then
      begin
        if (buf_initiator_id = 0) then
          memo = memo || ' Инициатор (раньше не был указан);';
        else
        begin
          select cl.c_name_fio
          from clients cl
          where cl.c_id = :buf_initiator_id
          into :buf_name_fio;
          memo = memo || ' Инициатор (был: ' || buf_name_fio || ');';
        end
      end

      if (buf_is_delete <> rb_is_delete) then
      begin
        if (buf_is_delete = 0) then
          memo = memo || ' Удаление (было: активный);';
        if (buf_is_delete = 1) then
          memo = memo || ' Активизация (было: удаленный);';
      end

      if (db_id in (39801, 41701)) then
      begin
        select rs_scores, rs_protocol_no, rs_protocol_date, rs_explanatory_date
        from rebuke_scores rs
        where rs.rs_from_rb_id = :rb_id
        into :rs_scores_old, :rs_protocol_no_old, :rs_protocol_date_old, :rs_explanatory_date_old;

        if (:rs_scores_old is null) then
          :rs_scores_old = -1;

        if (:rs_scores_old <> :rs_scores) then
          memo = memo || ' Изменение баллов (было: ' || iif(:rs_scores_old = -1, 'null', :rs_scores_old) || ');';
        if (:rs_protocol_no_old <> :rs_protocol_no) then
          memo = memo || ' Изменение баллов (было: ' || iif(:rs_protocol_no_old is null, 'null', :rs_protocol_no_old) || ');';
        if (:rs_protocol_date_old = :rs_protocol_date) then
          memo = memo || ' Изменение баллов (было: ' || iif(:rs_protocol_date_old is null, 'null', :rs_protocol_date_old) || ');';
        if (:rs_explanatory_date_old = :rs_explanatory_date) then
          memo = memo || ' Изменение баллов (было: ' || iif(:rs_explanatory_date_old is null, 'null', :rs_explanatory_date_old) || ');';
      end

      /*<<< ---------------------------------------------------------------------------------------------*/
      if (memo <> 'Изменения:') then
      /* Если изменения были зафиксированы*/
      begin
        update rebuke rb
        set rb_number = :rb_number,
            rb_date = :rb_date,
            rb_from_c_id = :rb_from_c_id,
            rb_from_reg_id = :rb_from_reg_id,
            rb_type = :rb_type,
            rb_body = :rb_body,
            rb_reason = :rb_reason,
            rb_bonus_change = :rb_bonus_change,
            rb_period_start = :rb_period_start,
            rb_period_end = :rb_period_end,
            rb_is_delete = :rb_is_delete,
            rb_bonus_koef = :rb_bonus_koef,
            rb_number_n = :rb_number_n,
            rb_sub_number = :rb_sub_number,
            rb_initiator_id = :rb_initiator_id,
            rb_no_number = :rb_no_number
        where (rb_id = :rb_id);

        insert into rebuke_log (rl_from_rb_id, rl_from_un_id, rl_type, rl_memo)
        values (:rb_id, :user_id, :rl_type, :memo);

        if (db_id in (39801, 41701)) then
          update rebuke_scores
          set rs_from_c_id = :rb_from_c_id,
              rs_scores = :rs_scores,
              rs_from_un_id = :user_id,
              rs_is_delete = :rb_is_delete,
              rs_date = :rb_date,
              rs_protocol_no = :rs_protocol_no,
              rs_protocol_date = :rs_protocol_date,
              rs_explanatory_date = :rs_explanatory_date
          where (rs_from_rb_id = :rb_id);
      end
    end
  end
  /***********/
  if (err_text = 'Ошибки:') then
    err_text = '';
  suspend;
end
