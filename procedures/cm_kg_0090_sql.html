CREATE PROCEDURE CM_KG_0090_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable k_from_kf_id id_type;
declare variable c_id id_type;
declare variable last_par_k_id id_type;
declare variable last_k_id id_type;
declare variable last_k_id_1 id_type;
declare variable last_k_id_2 id_type;
declare variable last_k_id_3 id_type;
declare variable last_k_id_4 id_type;
declare variable last_k_kontract_no varchar(50);
declare variable last_pr_name varchar(50);
declare variable pr_code varchar(100);
declare variable last_pr_code varchar(100);
declare variable k_kontract_no varchar(100);
declare variable last_ks_date_end date;
declare variable all_delinq_days integer;
declare variable curr_delinq_days integer;
declare variable max_delinq_days integer;
declare variable k_kolich_client_approved integer;
declare variable kol_cl integer;
declare variable kol_cl_2 integer;
declare variable kol_cl_3 integer;
declare variable kol_cl_4 integer;
declare variable k_refund integer;
begin
  if (K_ID = 109010093512800201) then
  exit;
  if (exists(select *
             from kontstate ks
             where ks_from_k_id = :k_id and
                   ks_ol_is_delete = 0 and
                   current_date between ks_date_begin and ks_date_end and
                   ks_state in (1, 2))) then
    exit;
  if (K_ID = 0) then
    exit;
  if (K_ID = 109010002378601701) then
    exit;
  /*
     
Разработать БЛОК на выдачу кредитов по продуктам «ГРУППОВОЙ», «ГРУППОВОЙ-МУГАЛИМ», «БИРГЕ», «БИРГЕ-МУГАЛИМ», «БИРГЕ-СОЦИАЛЬНЫЙ» «БИРГЕ-ЭКСПРЕСС»
- если нет соответствий следующим условиям:
- ГСО/клиента осталось 2 последних погашения по текущему кредиту по вышеуказанным продуктам;
- общее количество просроченных дней по текущему циклу по вышеуказанным продуктам не составляет 7 и более дней. 
         
Если у клиент Бирге стандарт (текущий) то он может взять только по Бирге, как следующий цикл.
Если у ГСО Груповой стандарт (текущий) то ГСО может взять только всей группой Групповой стандарт, как следующий цикл. 
Исключение:
  для клиентов по продукту «БИРГЕ-ЭКСПРЕСС»;
  ГСО/клиентам по текущему основному кредиту которых осталось 2 (два) последних погашения и на момент подачи Заявки у ГСО/клиента количество просроченных дней по текущему основному кредиту не превышает 7 дней (общее количество просроченных дней за текущий цикл). При этом ГСО/ клиент может получить кредит только по продукту текущего основного кредита.
*/
  res_error_is_critical = 1;
  select K_FROM_KF_ID, K.K_AMOUNT_APPROVED, K_DISB_DATE, K.K_KOLICH_CLIENT_APPROVED, PR_CODE, k_refund
  from KONTRACT K
  inner join PRODUCT PR on (K_FROM_PR_ID = PR_ID and
        PR_CODE not containing '.PAR.')
  where K_ID = :K_ID
  into K_FROM_KF_ID, KC_AMOUNT, KF_DISB_DATE, K_KOLICH_CLIENT_APPROVED, PR_CODE, k_refund;

  if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
   exit;

  if (K_FROM_KF_ID is null) then
    exit;
  for select KC_FROM_C_ID
      from KNCLIENT KC
      where KC_FROM_K_ID = :K_ID and
            KC_AMOUNT_APPROVED > 0
      into C_ID
  do
  begin
    select C_NAME_FIO
    from CLIENTS
    where C_ID = :C_ID
    into :res_info_text;
    --Исключение:   для клиентов по продукту «БИРГЕ-ЭКСПРЕСС»;
    if (not(PR_CODE containing '.BIRGE.' and
        PR_CODE containing '.EXPRESS.')) then
    begin
      if (PR_CODE similar to '%.GR.%|%.BIRGE.%') then
      begin
        LAST_K_ID = null;
        LAST_K_KONTRACT_NO = null;
        LAST_PR_CODE = null;
        LAST_PR_NAME = null;
        LAST_KS_DATE_END = null;
        select K_ID, K_KONTRACT_NO, PR_CODE, PR_NAME, KS_DATE_END
        from KNCLIENT KC
        inner join KONTSTATE KS on (KC_FROM_K_ID = KS_FROM_K_ID and
              KS_STATE = 1 and
              :KF_DISB_DATE between KS_DATE_BEGIN and KS_DATE_END)
        inner join KONTRACT K on (KS_FROM_K_ID = K_ID)
        inner join PRODUCT PR on (K_FROM_PR_ID = PR_ID and
              PR_CODE not containing '.PAR.' and
              PR_CODE not containing '.IPF.' and
              (not(PR_CODE containing '.BIRGE.' and
              PR_CODE containing '.EXPRESS.')))
        where KC_FROM_C_ID = :C_ID and
              KC_AMOUNT_APPROVED > 0
        order by KS_DATE_BEGIN desc rows 1
        into LAST_K_ID, LAST_K_KONTRACT_NO, LAST_PR_CODE, LAST_PR_NAME, LAST_KS_DATE_END;
        if (K_KOLICH_CLIENT_APPROVED > 1) then
        begin
          if (0 = coalesce(LAST_K_ID_1, 0)) then
            LAST_K_ID_1 = LAST_K_ID;
          if (LAST_K_ID = LAST_K_ID_1) then
            KOL_CL = coalesce(KOL_CL, 0) + 1;
          else
          begin
            if (0 = coalesce(LAST_K_ID_2, 0)) then
              LAST_K_ID_2 = LAST_K_ID;
            if (LAST_K_ID = LAST_K_ID_2) then
              KOL_CL_2 = coalesce(KOL_CL_2, 0) + 1;
            else
            begin
              if (0 = coalesce(LAST_K_ID_3, 0)) then
                LAST_K_ID_3 = LAST_K_ID;
              if (LAST_K_ID = LAST_K_ID_3) then
                KOL_CL_3 = coalesce(KOL_CL_3, 0) + 1;
              else
              begin
                if (0 = coalesce(LAST_K_ID_4, 0)) then
                  LAST_K_ID_4 = LAST_K_ID;
                if (LAST_K_ID = LAST_K_ID_4) then
                  KOL_CL_4 = coalesce(KOL_CL_4, 0) + 1;
              end
            end
          end
        end
        LAST_PAR_K_ID = null;
        if (LAST_KS_DATE_END>kf_disb_date) then
        select K_ID
        from KNCLIENT KC
        inner join KONTSTATE KS on (KC_FROM_K_ID = KS_FROM_K_ID and
              KS_STATE = 1 and
              :KF_DISB_DATE between KS_DATE_BEGIN and KS_DATE_END)
        inner join KONTRACT K on (KS_FROM_K_ID = K_ID)
        inner join PRODUCT PR on (K_FROM_PR_ID = PR_ID and
              PR_CODE containing '.PAR.')
        where KC_FROM_C_ID = :C_ID and
              KC_AMOUNT_APPROVED > 0
        order by KS_DATE_BEGIN desc rows 1
        into LAST_PAR_K_ID;
        if (((PR_CODE containing '.BIRGE.' and
            LAST_PR_CODE not containing '.BIRGE.') or --
            (PR_CODE containing '.GR.' and
            LAST_PR_CODE not containing '.GR.')) and (k_refund = 0)) then
        begin
          --Если у клиент Бирге стандарт (текущий) то он может взять только по Бирге, как следующий цикл.
          --Если у ГСО Груповой стандарт (текущий) то ГСО может взять только всей группой Групповой стандарт, как следующий цикл.
          res_error_text = 'Для клиента выбран неправильный продукт. Выберите продукт:' || LAST_PR_NAME;
          --          res_error_is_critical = 1;
          suspend;
        end
        CURR_DELINQ_DAYS = null;
        MAX_DELINQ_DAYS = null;
        ALL_DELINQ_DAYS = null;
        select sum(:KF_DISB_DATE - OKS_DELINQ_DATE)
        from OKSUMMARY
        where :KF_DISB_DATE between OKS_DATE_BEGIN and OKS_DATE_END and
              :KF_DISB_DATE > OKS_DELINQ_DATE and
              oks_close_state = 0 and
              OKS_FROM_K_ID in (:LAST_K_ID, :LAST_PAR_K_ID)
        into CURR_DELINQ_DAYS;
        if (coalesce(CURR_DELINQ_DAYS, -1) < 0) then
          CURR_DELINQ_DAYS = 0;
        select sum(OK_OL_DATE - OK_SH_DATE)
        from OPERKRED
        where OK_OL_IS_ACTIVE = 1 and
              OK_FROM_K_ID in (:LAST_K_ID, :LAST_PAR_K_ID) and
              OK_OL_DATE <= :KF_DISB_DATE and
              OK_OL_DATE > OK_SH_DATE and
              OK_TYPE in (4, 5) and
              OK_OS + OK_PERC < 0
        into MAX_DELINQ_DAYS;
        if (coalesce(MAX_DELINQ_DAYS, -1) < 0) then
          MAX_DELINQ_DAYS = 0;
        ALL_DELINQ_DAYS = CURR_DELINQ_DAYS + MAX_DELINQ_DAYS;
        if (k_refund = 0 and ALL_DELINQ_DAYS > 7) then
        begin
          -- общее количество просроченных дней по текущему циклу по вышеуказанным продуктам не составляет 7 и более дней.
          res_error_text = ALL_DELINQ_DAYS || '-дн. суммарной просрочки в предыдущем цикле. Выдача запрещается';
          --            res_error_is_critical = 1;
          suspend;
        end
        if (k_refund = 0 and exists(select count(SH_DATE)
                   from AAREP_MAKE_ZAD_LIST(:LAST_K_ID, :KF_DISB_DATE, 0)
                   having count(SH_DATE) > 2)) then
        begin
          -- ГСО/клиента осталось 2 последних погашения по текущему кредиту по вышеуказанным продуктам;
          res_error_text = 'Имеется более двух план.платежей. Выдача запрещается';
          --            res_error_is_critical = 1;
          suspend;
        end
      end
    end
  end
  if (PR_CODE containing '.GR.') then
  begin
    KOL_CL = iif(KOL_CL > KOL_CL_2, KOL_CL, KOL_CL_2);
    KOL_CL = iif(KOL_CL > KOL_CL_3, KOL_CL, KOL_CL_3);
    KOL_CL = iif(KOL_CL > KOL_CL_4, KOL_CL, KOL_CL_4);
    if (K_KOLICH_CLIENT_APPROVED / 2 > KOL_CL) then
    begin
      res_error_text = 'Количество новых участников превышает старых. Выдача запрещается';
      --            res_error_is_critical = 1;
      suspend;
    end
  end
end
