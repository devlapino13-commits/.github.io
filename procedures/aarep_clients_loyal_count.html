CREATE PROCEDURE AAREP_CLIENTS_LOYAL_COUNT(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    COUNT_ALL_MINIAPP D_INTEGER,
    COUNT_UNBLOCKED_VIPS D_INTEGER,
    COUNT_BLOCKED_VIPS_ST0 D_INTEGER,
    COUNT_BLOCK_VIPS D_INTEGER,
    COUNT_BLOCKED_VIPS_ST1 D_INTEGER,
    COUNT_ISSUED D_INTEGER,
    COUNT_BLOCKED_VIPS_REFUSAL D_INTEGER,
    COUNT_MINIAPP_CREDIT D_INTEGER,
    COUNT_MINIAPP_OTK D_INTEGER,
    COUNT_MINIAPP_PRO D_INTEGER,
    COUNT_MINIAPP D_INTEGER,
    COUNT_PROSECC_KR D_INTEGER,
    KONSEVR D_CURRENCY,
    SHARE_UNBLOCKED_VIPS D_CURRENCY,
    SHARE_VIPS_REFUSALS_ZAV D_CURRENCY,
    RD_NAME D_STR_255,
    COUNT_REFUSAL D_INTEGER)
AS
begin
  ---кол-во заявок
  select count(*)
  from extminiapp
  left join mc_login on mcl_id = ema_from_mcl_id
  left join cc_website_miniapp on ema_from_tkt_id = wma_from_tkt_id
  where ema_date between :rep_date_begin and :rep_date_end and
        ema_from_c_id is not null and
        exists(select *
               from clientlink
               where (clk_link_cell_no = mcl_login or clk_link_cell_no = wma_phone_no))
  into count_all_miniapp;

  --не заблокировано випсом
  count_unblocked_vips = null;
  select count(*)
  from extminiapp
  left join mc_login on mcl_id = ema_from_mcl_id
  left join cc_website_miniapp on ema_from_tkt_id = wma_from_tkt_id
  where ema_from_c_id is not null and
        ema_date between :rep_date_begin and :rep_date_end and
        ema_block_datetime is null and
        exists(select *
               from clientlink
               where (clk_link_cell_no = mcl_login or clk_link_cell_no = wma_phone_no))
  into count_unblocked_vips;

  -----Заблокиров Випсом со статусом процессе
  count_blocked_vips_st0 = null;
  select count(*)
  from extminiapp
  left join mc_login on mcl_id = ema_from_mcl_id
  left join cc_website_miniapp on ema_from_tkt_id = wma_from_tkt_id
  where ema_from_c_id is not null and
        ema_date between :rep_date_begin and :rep_date_end and
        ema_block_datetime is not null and
        ema_status in (0) and
        exists(select *
               from clientlink
               where (clk_link_cell_no = mcl_login or clk_link_cell_no = wma_phone_no))
  into count_blocked_vips_st0;

  -----Заблокиров Випсом со статусом процессе
  count_blocked_vips_st1 = null;
  select count(*)
  from extminiapp
  left join mc_login on mcl_id = ema_from_mcl_id
  left join cc_website_miniapp on ema_from_tkt_id = wma_from_tkt_id
  where ema_from_c_id is not null and
        ema_date between :rep_date_begin and :rep_date_end and
        ema_block_datetime is not null and
        ema_status in (1) and
        exists(select *
               from clientlink
               where (clk_link_cell_no = mcl_login or clk_link_cell_no = wma_phone_no))
  into count_blocked_vips_st1;

  ----Количество отказов
  count_blocked_vips_refusal = null;
  select count(*)
  from extminiapp
  left join mc_login on mcl_id = ema_from_mcl_id
  left join cc_website_miniapp on ema_from_tkt_id = wma_from_tkt_id
  where ema_from_c_id is not null and
        ema_date between :rep_date_begin and :rep_date_end and
        ema_block_datetime is not null and
        ema_status = -1 and
        exists(select *
               from clientlink
               where (clk_link_cell_no = mcl_login or clk_link_cell_no = wma_phone_no))
  into count_blocked_vips_refusal;

  count_block_vips = 0;
  count_block_vips = count_blocked_vips_st0 + count_blocked_vips_st1 + count_blocked_vips_refusal;

  count_miniapp_credit = null;
  select count(*)
  from miniapp
  inner join extminiapp on ema_from_tkt_id = ma_from_tkt_id
  left join mc_login on mcl_id = ema_from_mcl_id
  left join cc_website_miniapp on ema_from_tkt_id = wma_from_tkt_id
  where ma_date_app between :rep_date_begin and :rep_date_end and
        ma_date_end = '31.12.9999' and
        exists(select *
               from clientlink
               where (clk_link_cell_no = mcl_login or clk_link_cell_no = wma_phone_no))
  into count_miniapp_credit;

  count_miniapp_otk = null;
  select count(*)
  from miniapp
  inner join extminiapp on ema_from_tkt_id = ma_from_tkt_id
  left join mc_login on mcl_id = ema_from_mcl_id
  left join cc_website_miniapp on ema_from_tkt_id = wma_from_tkt_id
  where ma_date_app between :rep_date_begin and :rep_date_end and
        ma_date_end = '31.12.9999' and
        ma_is_active = -1 and
        exists(select *
               from clientlink
               where (clk_link_cell_no = mcl_login or clk_link_cell_no = wma_phone_no))
  into count_miniapp_otk;

  count_miniapp = null;
  select count(*)
  from miniapp
  inner join extminiapp on ema_from_tkt_id = ma_from_tkt_id
  left join mc_login on mcl_id = ema_from_mcl_id
  left join cc_website_miniapp on ema_from_tkt_id = wma_from_tkt_id
  where ma_date_app between :rep_date_begin and :rep_date_end and
        ma_date_end = '31.12.9999' and
        ma_is_active = 1 and
        exists(select *
               from clientlink
               where (clk_link_cell_no = mcl_login or clk_link_cell_no = wma_phone_no))
  into count_miniapp;

  --- кол-во выданных
  select count(*)
  from kontstate
  inner join kontract on k_id = ks_from_k_id
  inner join extminiapp on ema_from_tkt_id = k_from_tkt_id
  left join mc_login on mcl_id = ema_from_mcl_id
  left join cc_website_miniapp on ema_from_tkt_id = wma_from_tkt_id
  where ks_ol_is_delete = 0 and
        ks_date_begin between :rep_date_begin and :rep_date_end and
        ks_state = 1 and
        exists(select *
               from clientlink
               where (clk_link_cell_no = mcl_login or clk_link_cell_no = wma_phone_no))
  into count_issued;

  count_miniapp_pro = 0;
  count_prosecc_kr = 0;
  count_miniapp_pro = count_issued - count_miniapp;
  count_prosecc_kr = count_blocked_vips_st1 - count_miniapp_credit;

  ------------------------------------------------------------------------------------------

  if (count_all_miniapp = 0) then
    konsevr = 0;
  else
    konsevr = (cast(count_issued as d_currency) / cast(count_all_miniapp as d_currency)) * 100;

  -------------------------------------------------------------------------------------------

  if (count_all_miniapp = 0) then
    share_unblocked_vips = 0;
  else
    share_unblocked_vips = (cast(count_unblocked_vips as d_currency) / cast(count_all_miniapp as d_currency)) * 100;

  -----------------------------------------------------------------------------------------------
  if (count_all_miniapp = 0) then
    share_vips_refusals_zav = 0;
  else
    share_vips_refusals_zav = (cast(count_blocked_vips_refusal as d_currency) / cast(count_all_miniapp as d_currency)) * 100;

  for select count(*), iif(rd_name is null, 'Не заполнено', rd_name)
      from extminiapp
      left join refdata on rd_id = ema_result_rd_id
      left join mc_login on mcl_id = ema_from_mcl_id
      left join cc_website_miniapp on ema_from_tkt_id = wma_from_tkt_id
      where ema_from_c_id is not null and
            ema_date between :rep_date_begin and :rep_date_end and
            ema_status = -1 and
            ema_lo_id is not null and
            exists(select *
                   from clientlink
                   where (clk_link_cell_no = mcl_login or clk_link_cell_no = wma_phone_no))
      group by ema_result_rd_id, rd_name
      into count_refusal, rd_name
  do
  begin
    suspend;
  end
end
