CREATE PROCEDURE HP_FSDATALOG_UPDATE(
    UN_ID DECIMAL(18,0),
    REP_ID VARCHAR(255),
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    DATE_BEGIN DATE,
    DATE_END DATE,
    UN_IP VARCHAR(20),
    SOURCE_SYSTEM INTEGER,
    INTERFACE_KIND INTEGER)
AS
declare variable fdk_id integer;
declare variable obj_kind_id integer;
declare variable rep_caption varchar(255);
declare variable id integer;
declare variable dot_pos integer;
declare variable un_ip_address bigint;
declare variable ip_1 integer;
declare variable ip_2 integer;
declare variable ip_3 integer;
declare variable ip_4 integer;
begin
  if (f_get_db_id() in (39801, 41701)) then  -- т.к. коннекты зависают
    exit;

  if (obj_id is null) then
    exit;

  ip_1 = 0;
  ip_2 = 0;
  ip_3 = 0;
  ip_4 = 0;
  un_ip_address = 0;

  dot_pos = position('.', un_ip, 1);
  if (dot_pos > 0) then
  begin
    ip_1 = left(un_ip, dot_pos - 1);
    un_ip = substring(un_ip from dot_pos + 1);
  end
  un_ip_address = un_ip_address + ip_1;

  dot_pos = position('.', un_ip, 1);
  if (dot_pos > 0) then
  begin
    ip_2 = left(un_ip, dot_pos - 1);
    un_ip = substring(un_ip from dot_pos + 1);
  end
  un_ip_address = un_ip_address * 256 + ip_2;

  dot_pos = position('.', un_ip, 1);
  if (dot_pos > 0) then
  begin
    ip_3 = left(un_ip, dot_pos - 1);
    un_ip = substring(un_ip from dot_pos + 1);
  end
  un_ip_address = un_ip_address * 256 + ip_3;

  if (un_ip <> '') then
    ip_4 = un_ip;
  un_ip_address = un_ip_address * 256 + ip_4;

  if (un_ip_address >= 2147483648) then
    un_ip_address = 2147483647 - un_ip_address;

  select fdk_id
  from fsdatakind
  where fdk_code = :rep_id
  into :fdk_id;

  if (fdk_id is null) then
  begin
    rep_caption = '';
    if (rep_id starting '[frx_blank_') then --  Значит это служебные бланки
    begin
      id = substring(rep_id from 12 for (char_length(rep_id) - 12));
      select bln_name
      from blankname
      where bln_id = :id
      into :rep_caption;
      if (coalesce(rep_caption, '') <> '') then
        rep_caption = 'Бланк: ' || rep_caption;
    end
    if (rep_id starting '[blank_') then --  Значит это служебные бланки
    begin
      id = substring(rep_id from 8 for (char_length(rep_id) - 8));
      select bln_name
      from blankname
      where bln_id = :id
      into :rep_caption;
      if (coalesce(rep_caption, '') <> '') then
        rep_caption = 'Шаблон: ' || rep_caption;
    end

    if (rep_id starting '[Report_Grid_') then --  Значит это служебные бланки
    begin
      id = substring(rep_id from 14 for (char_length(rep_id) - 14));
      select rl_name_rus
      from reportlist
      where rl_id = :id
      into :rep_caption;
      if (coalesce(rep_caption, '') <> '') then
        rep_caption = 'Отчет-таблица: ' || rep_caption;
    end
    if (rep_id starting '[Report_Frx_') then --  Значит это служебные бланки
    begin
      id = substring(rep_id from 13 for (char_length(rep_id) - 13));
      select rl_name_rus
      from reportlist
      where rl_id = :id
      into :rep_caption;
      if (coalesce(rep_caption, '') <> '') then
        rep_caption = 'Отчет-бланк: ' || rep_caption;
    end

    if (rep_id starting '[Report_Form_') then --  Значит это служебные бланки
    begin
      id = substring(rep_id from 14 for (char_length(rep_id) - 14));
      select rl_name_rus
      from reportlist
      where rl_id = :id
      into :rep_caption;
      if (coalesce(rep_caption, '') <> '') then
        rep_caption = 'Отчет-форма: ' || rep_caption;
    end

    if (coalesce(rep_caption, '') = '') then
      rep_caption = rep_id;

    insert into fsdatakind (fdk_code, fdk_caption)
    values (:rep_id, :rep_caption)
    returning fdk_id
    into :fdk_id;
  end

  begin
    if (obj_kind is not null) then
      obj_kind_id = f_objkind(obj_kind, 'id');
    when any do
      obj_kind_id = null;
  end

  if (not(exists(select *
                 from fsdatalog
                 where fl_from_un_id = :un_id and
                       fl_datakind_id = :fdk_id and
                       fl_object_kind = :obj_kind_id and
                       fl_object_id = :obj_id and
                       fl_datetime >= dateadd(minute, -5, current_timestamp)))) then
  begin
    insert into fsdatalog (fl_from_un_id, fl_datakind_id, fl_object_kind, fl_object_id, fl_date_begin, fl_date_end,
                           fl_ip, fl_source_system, fl_interface_kind)
    values (:un_id, :fdk_id, :obj_kind_id, :obj_id, :date_begin, :date_end, :un_ip_address, :source_system,
            :interface_kind);
  end
  -- По первым числам месяца удаляем старые записи за последние 2 месяца.
  -- При необходимости можно поднять данные из бэкапа нужный период.
  if (extract(day from current_date) = 1) then
    delete from fsdatalog
    where fl_datetime < dateadd(month, -2, current_date);
end
