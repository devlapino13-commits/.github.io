CREATE PROCEDURE CM_KG_0212_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable pr_code_k d_str_100;
declare variable c_id d_id;
declare variable mac_amount d_integer;
declare variable ma_date_oper d_date;
declare variable pr_code_m d_str_100;
declare variable cnt_classic d_smallint;
declare variable cnt_ipf d_smallint;
begin
  if (:ma_id = 3615875) then
    exit; --Ticket#2023071888000934 — договор не оформляется
  if (:ma_id = 3613836) then
    exit; --RFCKG-708
  if (:ma_id = 3616589) then
    exit; --RFCKG-714

  for select mc.mac_from_c_id, ma.ma_summa_minus, ma.ma_date_oper, mc.mac_from_ma_id, pr.pr_code
      from miniappclient mc
      inner join miniapp ma on ma.ma_id = mc.mac_from_ma_id
      inner join product pr on pr.pr_id = ma.ma_from_pr_id
      where mc.mac_from_ma_id = :ma_id and
            mc.mac_is_poruch = 0 and
            ma.ma_is_active = 1 and
            ma.ma_date_end >= '31.12.2999' and
            pr.pr_code not containing '.sugarmill.' and
            pr.pr_code not containing '.installment_goods.'
      into :c_id, :mac_amount, :ma_date_oper, :ma_id, :pr_code_m
  do
  begin

    cnt_classic = 0;
    cnt_ipf = 0;

    for select pt.pr_code
        from kontstate ks
        inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
        inner join kontract kt on kt.k_id = ks.ks_from_k_id
        inner join product pt on pt.pr_id = kt.k_from_pr_id
        where kc.kc_from_c_id = :c_id and
              :ma_date_oper between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_state = 1 and
              ks.ks_ol_is_delete = 0 and
              kc.kc_amount_approved > 0 and
              kt.k_disb_date = :ma_date_oper
        into :pr_code_k
    do
    begin
      if (pr_code_k containing '.IPF.') then
        cnt_ipf = cnt_ipf + 1;
      else
        cnt_classic = cnt_classic + 1;
    end

    if (pr_code_m containing '.IPF.' and cnt_ipf >= 2) then
    begin
      res_error_text = 'По продукту ИПФ допускается не более двух выдач в день';
      res_error_is_critical = 1;
      suspend;
      exit;
    end
    else if (pr_code_m not containing '.IPF.' and cnt_classic >= 1) then
    begin
      res_error_text = 'По продукту Классика допускается не более одной выдачи в день';
      res_error_is_critical = 1;
      suspend;
      exit;
    end
  end
end
