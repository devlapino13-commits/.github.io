CREATE PROCEDURE HP_USER_CASH(
    REG_ID INTEGER,
    ACC_TYPE INTEGER,
    C_ID ID_TYPE = null)
RETURNS (
    UC_FROM_REG_ID INTEGER,
    UC_FROM_C_ID ID_TYPE,
    UC_NUM_CASH INTEGER,
    UC_ACCTYPE INTEGER,
    UC_DATE_BEG TIMESTAMP,
    UC_DATE_END TIMESTAMP,
    REG_NAME VARCHAR(50),
    C_NAME_FIO STR_100)
AS
begin
  if (c_id > 0) then begin
    for
      select
        uc_from_reg_id,
        uc_from_c_id,
        uc_num_cash,
        uc_acctype,
        uc_date_beg,
        uc_date_end,
        reg_name,
        c_name_fio
      from User_Cash
        inner join Clients
          on c_id = uc_from_c_id
        inner join Region
          on reg_unicode = uc_from_reg_id
      where uc_from_c_id = :c_id
      into
        :uc_from_reg_id,
        :uc_from_c_id,
        :uc_num_cash,
        :uc_acctype,
        :uc_date_beg,
        :uc_date_end,
        :reg_name,
        :c_name_fio
    do suspend;
  end else begin
    for
      select
        uc_from_reg_id,
        uc_from_c_id,
        uc_num_cash,
        uc_acctype,
        uc_date_beg,
        uc_date_end,
        reg_name,
        c_name_fio
      from User_Cash
        inner join Clients
          on c_id = uc_from_c_id
        inner join Region
          on reg_unicode = uc_from_reg_id
      where uc_from_reg_id = :reg_id
        and uc_acctype = :acc_type
    into
        :uc_from_reg_id,
        :uc_from_c_id,
        :uc_num_cash,
        :uc_acctype,
        :uc_date_beg,
        :uc_date_end,
        :reg_name,
        :c_name_fio
    do suspend;
  end
end
