CREATE PROCEDURE UXP$UXFRAME_CLONE(
    UX_UUID D_UUID,
    NEW_OWNER D_UUID,
    IS_IMPORT D_BOOL = 0,
    UX_UUID_CODE D_STR_CODE = null,
    NEW_OWNER_CODE D_STR_CODE = null)
RETURNS (
    FR_UUID D_UUID)
AS
declare variable ufi_uuid d_uuid;
declare variable new_ufi_uuid d_uuid;
declare variable ufr_new_code d_str_code;
declare variable ufp_uuid d_uuid;
declare variable new_ufp_uuid d_uuid;
declare variable tmp_uuid d_uuid;
declare variable ufb_img_uuid d_uuid;
declare variable ufb_hide_menu d_bool;
declare variable grid_uuid d_uuid;
declare variable ufm_uuid d_uuid;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  if (coalesce(new_owner, '') = '') then
    new_owner = uxf$uniq(new_owner_code);

  if (is_import is null) then
    is_import = 0;

  if (is_import = 1) then
  begin
    delete from ux$pairuuid
    where 1 = 1;
  end

  for select ufr_uuid, ufr_owner, ufr_code, ufr_name, ufr_us_uuid, ufr_is_grid,
             ufr_can_update, ufr_can_insert, ufr_can_delete, ufr_tree_parent_field, ufr_tree_image_field,
             ufr_master_ufr_uuid, ufr_is_mainframe, ufr_master_field, ufr_detail_field,
             ufr_hide_on_group, ufr_group_panel, ufr_find_panel, ufr_hide_summary,
             ufr_autowidth, ufr_grid_disable_edit, ufr_sql_where, ufr_sql_orderby, ufr_refresh_params,
             ufr_tree_position_field, ufr_tree_can_drag_field, ufr_tree_can_drop_field, ufr_add_table_fields
  from ux$frame
  where ufr_uuid = :ux_uuid
  as cursor cur
  do
  begin
    ufm_uuid = cur.ufr_owner;

    ufr_new_code = cur.ufr_code;

    if (is_import = 1) then
    begin
      if (exists(select * from ux$frame
      where ufr_code = :ufr_new_code and
            ufr_owner = :new_owner)) then
        ufr_new_code = ufr_new_code || '_copy';
    end

    insert into ux$frame (ufr_owner, ufr_code, ufr_name, ufr_us_uuid, ufr_is_grid,
                                    ufr_can_update, ufr_can_insert, ufr_can_delete, ufr_tree_parent_field, ufr_tree_image_field,
                                    ufr_hide_on_group, ufr_group_panel, ufr_find_panel, ufr_hide_summary,
                                    ufr_autowidth, ufr_grid_disable_edit, ufr_sql_where, ufr_sql_orderby, ufr_refresh_params,
                                    ufr_tree_position_field, ufr_tree_can_drag_field, ufr_tree_can_drop_field, ufr_add_table_fields,
                                    ufr_is_mainframe,
                                    ufr_master_field,
                                    ufr_detail_field)
    values (:new_owner, :ufr_new_code, cur.ufr_name, cur.ufr_us_uuid, cur.ufr_is_grid,
             cur.ufr_can_update, cur.ufr_can_insert, cur.ufr_can_delete, cur.ufr_tree_parent_field, cur.ufr_tree_image_field,
             cur.ufr_hide_on_group, cur.ufr_group_panel, cur.ufr_find_panel, cur.ufr_hide_summary,
             cur.ufr_autowidth, cur.ufr_grid_disable_edit, cur.ufr_sql_where, cur.ufr_sql_orderby, cur.ufr_refresh_params,
             cur.ufr_tree_position_field, cur.ufr_tree_can_drag_field, cur.ufr_tree_can_drop_field, cur.ufr_add_table_fields,
             iif(:is_import = 0, cur.ufr_is_mainframe, 0),
             iif(:is_import = 0, cur.ufr_master_field, ''),
             iif(:is_import = 0, cur.ufr_detail_field, ''))
    returning ufr_uuid
    into fr_uuid;

    execute procedure uxp$uxpairuuid_add(:ux_uuid, :fr_uuid);
  end

  for select xls_ufr_uuid, xls_cfg_json, xls_memo
  from ux$excel
  where xls_ufr_uuid = :ux_uuid
  as cursor cur
  do
  begin
    update or insert into ux$excel (xls_ufr_uuid, xls_cfg_json, xls_memo)
    values (:fr_uuid, cur.xls_cfg_json, cur.xls_memo)
    matching (xls_ufr_uuid);
  end

  for select ufd_kind, ufd_us_uuid
  from ux$frdata
  where ufd_ufr_uuid = :ux_uuid
  as cursor cur
  do
  begin
    update or insert into ux$frdata (ufd_ufr_uuid, ufd_kind, ufd_us_uuid)
    values (:fr_uuid, cur.ufd_kind, cur.ufd_us_uuid)
    matching (ufd_ufr_uuid, ufd_kind);
  end

  for select ufp_uuid, ufp_uf_uuid, ufp_parent, ufp_editor, ufp_refresh, ufp_not_null
  from ux$frparam
  where ufp_ufr_uuid = :ux_uuid and
        ufp_parent is null
  as cursor cur
  do
  begin
    update or insert into ux$frparam (ufp_ufr_uuid, ufp_uf_uuid, ufp_parent, ufp_editor, ufp_refresh, ufp_not_null)
    values (:fr_uuid, cur.ufp_uf_uuid, cur.ufp_parent, cur.ufp_editor, cur.ufp_refresh, cur.ufp_not_null)
    matching (ufp_ufr_uuid, ufp_uf_uuid, ufp_parent)
    returning (ufp_uuid)
    into tmp_uuid;

    execute procedure uxp$uxpairuuid_add(cur.ufp_uuid, :tmp_uuid);
  end

  for select ufp_uuid, ufp_uf_uuid, ufp_parent, ufp_editor, ufp_refresh, ufp_not_null
  from ux$frparam
  where ufp_ufr_uuid = :ux_uuid and
        ufp_parent is not null
  as cursor cur
  do
  begin
    update or insert into ux$frparam (ufp_ufr_uuid, ufp_uf_uuid, ufp_parent, ufp_editor, ufp_refresh, ufp_not_null)
    values (:fr_uuid, cur.ufp_uf_uuid, uxf$pair(cur.ufp_parent), cur.ufp_editor, cur.ufp_refresh, cur.ufp_not_null)
    matching (ufp_ufr_uuid, ufp_uf_uuid, ufp_parent)
    returning ufp_uuid
    into tmp_uuid;

    execute procedure uxp$uxpairuuid_add(cur.ufp_uuid, :tmp_uuid);
  end

  execute procedure uxp$uxexpression_clone(:ux_uuid, :fr_uuid);

  for select ufs_ufr_uuid, ufs_sql_kind, ufs_uf_uuid, ufs_ufp_uuid, ufs_default
  from ux$framesql
  where ufs_ufr_uuid = :ux_uuid
  as cursor cur
  do
  begin
    new_ufp_uuid = null;
    if (cur.ufs_ufp_uuid is not null) then
      new_ufp_uuid = uxf$pair(cur.ufs_ufp_uuid);

    update or insert into ux$framesql (ufs_ufr_uuid, ufs_sql_kind, ufs_uf_uuid, ufs_ufp_uuid, ufs_default)
    values (:fr_uuid, cur.ufs_sql_kind, cur.ufs_uf_uuid, :new_ufp_uuid, cur.ufs_default)
    matching (ufs_ufr_uuid, ufs_sql_kind, ufs_uf_uuid);
  end

  for select ufp_uuid
  from ux$frparam
  where ufp_ufr_uuid = :ux_uuid
  into ufp_uuid
  do
  begin
    new_ufp_uuid = uxf$pair(ufp_uuid);

    for select ufc_combo_source, ufc_combo_field, ufc_hide_text, ufc_kind
    from ux$frcombo
    where ufc_ufp_uuid = :ufp_uuid
    as cursor cur
    do
    begin
      update or insert into ux$frcombo (ufc_ufp_uuid, ufc_combo_source, ufc_combo_field, ufc_hide_text, ufc_kind)
      values (:new_ufp_uuid, cur.ufc_combo_source, cur.ufc_combo_field, cur.ufc_hide_text, cur.ufc_kind)
      matching (ufc_ufp_uuid);
    end

    for select uma_ux_uuid, uma_mask, uma_message
    from ux$frmask
    where uma_ux_uuid = :ufp_uuid
    as cursor cur
    do
    begin
      update or insert into ux$frmask (uma_ux_uuid, uma_mask, uma_message)
      values (:new_ufp_uuid, cur.uma_mask, cur.uma_message)
      matching (uma_ux_uuid);
    end

    for select ufl_us_uuid, ufl_list_kind, ufl_lookup_id, ufl_lookup_text
    from ux$frlookup
    where ufl_ufp_uuid = :ufp_uuid
    as cursor cur
    do
    begin
      update or insert into ux$frlookup (ufl_ufp_uuid, ufl_us_uuid, ufl_list_kind, ufl_lookup_id, ufl_lookup_text)
      values (:new_ufp_uuid, cur.ufl_us_uuid, cur.ufl_list_kind, cur.ufl_lookup_id, cur.ufl_lookup_text)
      matching (ufl_ufp_uuid);
    end

    for select upd_ufp_uuid, upd_uxa_uuid, upd_def_value, upd_locked
    from ux$paramdef
    where upd_ufp_uuid = :ufp_uuid and
          upd_uxa_uuid is null
    as cursor cur
    do
    begin
      update or insert into ux$paramdef (upd_ufp_uuid, upd_def_value, upd_locked)
      values (:new_ufp_uuid, cur.upd_def_value, cur.upd_locked)
      matching (upd_ufp_uuid);
    end

    for select uma_mask, uma_message
    from ux$frmask
    where uma_ux_uuid = :ufp_uuid
    as cursor cur
    do
    begin
      update or insert into ux$frmask (uma_ux_uuid, uma_mask, uma_message)
      values (:new_ufp_uuid, cur.uma_mask, cur.uma_message)
      matching (uma_ux_uuid);
    end

    for select fer_ufp_uuid, fer_erl_uuid
    from ux$frereflist
    where fer_ufp_uuid = :ufp_uuid
    as cursor cur
    do
    begin
      update or insert into ux$frereflist (fer_ufp_uuid, fer_erl_uuid)
      values (:new_ufp_uuid, cur.fer_erl_uuid)
      matching (fer_ufp_uuid);
    end

    execute procedure uxp$uxexpression_clone(:ufp_uuid, :new_ufp_uuid);
  end

  execute procedure uxp$uxaction_clone(:ux_uuid, :fr_uuid);

  for select ugr_uuid, ugr_ufr_uuid, ugr_name, ugr_memo, ugr_is_private, ugr_default, ugr_hide_on_group,
             ugr_group_panel, ugr_find_panel, ugr_hide_summary, ugr_autowidth, ugr_header_autoheight,
             ugr_show_info_panel, ugr_hide_inplace_editor, ugr_show_excel_import,
             ugr_disable_edit, ugr_inactive, ugr_owner_user, ugr_excel, ugr_ds_code, ugr_group_field, ugr_formatting,
             ugr_tree_can_move, ugr_allow_append, ugr_ui_kind, ugr_calendar_fields
  from ux$grid
  where ugr_ufr_uuid = :ux_uuid
  as cursor cur
  do
  begin
    insert into ux$grid (ugr_ufr_uuid, ugr_name, ugr_memo, ugr_is_private, ugr_default,
                         ugr_hide_on_group, ugr_group_panel, ugr_find_panel, ugr_hide_summary, ugr_autowidth,
                         ugr_header_autoheight, ugr_show_info_panel, ugr_hide_inplace_editor, ugr_show_excel_import,
                         ugr_disable_edit, ugr_inactive, ugr_owner_user, ugr_excel, ugr_ds_code, ugr_group_field, ugr_formatting,
                         ugr_tree_can_move, ugr_allow_append, ugr_ui_kind, ugr_calendar_fields)
    values (:fr_uuid, cur.ugr_name, cur.ugr_memo, cur.ugr_is_private, cur.ugr_default, cur.ugr_hide_on_group,
            cur.ugr_group_panel, cur.ugr_find_panel, cur.ugr_hide_summary, cur.ugr_autowidth,
            cur.ugr_header_autoheight, cur.ugr_show_info_panel, cur.ugr_hide_inplace_editor, cur.ugr_show_excel_import,
            cur.ugr_disable_edit, cur.ugr_inactive, cur.ugr_owner_user, cur.ugr_excel, cur.ugr_ds_code, cur.ugr_group_field, cur.ugr_formatting,
            cur.ugr_tree_can_move, cur.ugr_allow_append, cur.ugr_ui_kind, cur.ugr_calendar_fields)
    returning ugr_uuid
    into tmp_uuid;

    if (grid_uuid is null) then
      grid_uuid = tmp_uuid;

    execute procedure uxp$uxpairuuid_add(cur.ugr_uuid, :tmp_uuid);

    insert into ux$gritem (ugi_ugr_uuid, ugi_uf_uuid, ugi_position, ugi_width, ugi_sizable,
                           ugi_order_index, ugi_group_index, ugi_fixed_kind, ugi_summary_kind,
                           ugi_caption, ugi_hint, ugi_hidden, ugi_readonly,
                           ugi_header_align, ugi_value_align)
    select :tmp_uuid, ugi_uf_uuid, ugi_position, ugi_width, ugi_sizable, ugi_order_index,
           ugi_group_index, ugi_fixed_kind, ugi_summary_kind,
           ugi_caption, ugi_hint, ugi_hidden, ugi_readonly,
           ugi_header_align, ugi_value_align
    from ux$gritem
    where ugi_ugr_uuid = :cur.ugr_uuid
    order by ugi_position;
  end

  for select ufi_uuid, ufi_parent_uuid, ufi_owner, ufi_ufr_uuid, ufi_uf_uuid, ufi_kind, ufi_control, ufi_position,
             ufi_name, ufi_caption_align, ufi_hint, ufi_hidden, ufi_readonly,
             ufi_group_direction, ufi_h_align, ufi_v_align, ufi_width, ufi_height,
             ufi_sizable, ufi_order_index, ufi_group_index, ufi_fixed_kind, ufi_summary_kind, ufi_font_size, ufi_value_align
  from ux$fritem
  where ufi_owner = :ux_uuid and
        ufi_parent_uuid is null
  as cursor cur
  do
  begin
    ufi_uuid = cur.ufi_uuid;

    update or insert into ux$fritem (ufi_parent_uuid, ufi_owner, ufi_ufr_uuid, ufi_uf_uuid, ufi_kind, ufi_control,
                                     ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_hidden, ufi_readonly,
                                     ufi_group_direction, ufi_h_align, ufi_v_align, ufi_width, ufi_height,
                                     ufi_sizable, ufi_order_index, ufi_group_index, ufi_fixed_kind, ufi_summary_kind,
                                     ufi_font_size, ufi_value_align)
    values (null, :fr_uuid, :fr_uuid, cur.ufi_uf_uuid, cur.ufi_kind, cur.ufi_control,
            cur.ufi_position, cur.ufi_name, cur.ufi_caption_align, cur.ufi_hint, cur.ufi_hidden, cur.ufi_readonly,
            cur.ufi_group_direction, cur.ufi_h_align, cur.ufi_v_align, cur.ufi_width, cur.ufi_height,
            cur.ufi_sizable, cur.ufi_order_index, cur.ufi_group_index, cur.ufi_fixed_kind, cur.ufi_summary_kind,
            cur.ufi_font_size, cur.ufi_value_align)
    matching (ufi_owner, ufi_parent_uuid)
    returning (ufi_uuid)
    into new_ufi_uuid;

    execute procedure uxp$uxpairuuid_add(:ufi_uuid, :new_ufi_uuid);

    delete from ux$fritem
    where ufi_parent_uuid = :new_ufi_uuid;

    execute procedure uxp$uxfritem_clone_tree(:fr_uuid, :new_ufi_uuid, :ufi_uuid);
    execute procedure uxp$uxexpression_clone(:ufi_uuid, :new_ufi_uuid);
  end

  for select ufb_ufi_uuid, ufb_ua_uuid, ufb_img_uuid, ufb_hide_menu
  from ux$fritem, ux$frbutton
  where ufi_owner = :ux_uuid and
        ufi_uuid = ufb_ufi_uuid
  into ufi_uuid, tmp_uuid, ufb_img_uuid, ufb_hide_menu
  do
  begin
    update or insert into ux$frbutton (ufb_ufi_uuid, ufb_ua_uuid, ufb_img_uuid, ufb_hide_menu)
    values (uxf$pair(:ufi_uuid), uxf$pair(:tmp_uuid), :ufb_img_uuid, :ufb_hide_menu)
    matching (ufb_ufi_uuid);
  end

  for select ufr_uuid
  from ux$frame
  where ufr_uuid = :ux_uuid
  into tmp_uuid
  do
    execute procedure uxp$uxevent_clone(:tmp_uuid);

  for select ufp_uuid
  from ux$frparam
  where ufp_ufr_uuid = :ux_uuid
  into tmp_uuid
  do
    execute procedure uxp$uxevent_clone(:tmp_uuid);

  for select ufi_uuid
  from ux$fritem
  where ufi_owner = :ux_uuid
  into tmp_uuid
  do
    execute procedure uxp$uxevent_clone(:tmp_uuid);

  if (grid_uuid is not null) then
  begin
    update ux$fritem
    set ufi_uuid = :grid_uuid
    where ufi_ufr_uuid = :fr_uuid and
          ufi_control in ('grid', 'tree')
    rows 1;
  end

  if (is_import = 1 and ufm_uuid is not null) then
  begin
    if (not exists(select * from ux$fritem
    where ufi_owner = :new_owner and
          ufi_ufr_uuid = :fr_uuid and
          ufi_control = 'frame')) then
    begin
      insert into ux$fritem (ufi_parent_uuid, ufi_owner, ufi_ufr_uuid, ufi_uf_uuid, ufi_kind, ufi_control,
                             ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_hidden, ufi_readonly,
                             ufi_group_direction, ufi_h_align, ufi_v_align, ufi_width, ufi_height, ufi_sizable,
                             ufi_order_index, ufi_group_index, ufi_fixed_kind, ufi_summary_kind, ufi_bnd_id, ufi_font_size,
                             ufi_value_align)
      select :new_owner, :new_owner, :fr_uuid, ufi_uf_uuid, ufi_kind, ufi_control, ufi_position, ufi_name,
             ufi_caption_align, ufi_hint, ufi_hidden, ufi_readonly, ufi_group_direction, ufi_h_align, ufi_v_align, ufi_width,
             ufi_height, ufi_sizable, ufi_order_index, ufi_group_index, ufi_fixed_kind, ufi_summary_kind, ufi_bnd_id,
             ufi_font_size, ufi_value_align
      from ux$fritem
      where ufi_owner = :ufm_uuid and
            ufi_ufr_uuid = :ux_uuid and
            ufi_control = 'frame'
      rows 1;
    end
  end

  suspend;
end
