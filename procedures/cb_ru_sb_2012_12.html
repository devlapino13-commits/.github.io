CREATE PROCEDURE CB_RU_SB_2012_12(
    REP_REGION_ID INTEGER,
    REP_BONUS_ID INTEGER,
    REP_USER_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    ID_LIST INTEGER,
    OKRUG_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(50),
    REG_NAME VARCHAR(50),
    REG_BALANCE_LAST CURR_TYPE,
    REG_BALANCE_CURR CURR_TYPE,
    REG_BALANCE_DELINQ CURR_TYPE,
    PAR7 DECIMAL(15,1),
    CB_BONUS_SB_TOTAL CURR_TYPE,
    PERSONAL_NAME VARCHAR(100),
    JL_NAME VARCHAR(100),
    REG_KOLICH_SB INTEGER,
    CB_PERSONAL_STAGE DATE)
AS
declare variable rep_date_begin date;
declare variable rep_date_end_2 date;
declare variable cb_personal_id curr_type;
begin
  rep_date_begin = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date); -- Начало периода
  rep_date_end_2 = rep_date_begin - 1; -- Конец предыдущего месяца
  if (:rep_date_end between '02.01.2013' and '31.01.2013') then -- Для того чтобы если за январь2013 считать то за пред-й портфель брал 25.12.2012
  begin
    rep_date_end_2 = '25.12.2012';
    rep_date_begin = rep_date_end_2 + 1;
  end
  for select id_list, r3.reg_code || ' ' || r3.reg_name, r2.reg_code || ' ' || r2.reg_name,
             r1.reg_code || ' ' || r1.reg_name
      from region_list(:rep_region_id, 1, 3, 1), region r1, region r2, region r3
      where r1.reg_unicode = id_list and
            r2.reg_unicode = r1.parent and
            r3.reg_unicode = r2.parent
      order by r3.reg_code, r2.reg_code, r1.reg_code
      into id_list, okrug_name, branch_name, reg_name
  do
  begin
--  Портфель всех займов, выданных СВЗ данного офиса с 1 июня 2012
--  Портфель, выданных СВЗ до 1 июня 2012, просрочка на 01 декабря отсутствует
--  Портфель, выданных СВЗ до 1 июня 2012, просрочка на 01 декабря от 1 до 180 дней.

--  Расчет баланса региона за предыдущий месяц
    reg_balance_last = null;
    with tabl1
    as (select
               case
                 when ks_date_begin >= '01.06.2012' then k_id
                 else case
                        when ks_date_begin < '01.06.2012' and coalesce((select rep_delinq_days
                                                                        from aarep_kont_delinq_status_new(k_id, '01.12.2012')), 0) < 180 then k_id
                        else 0
                      end
               end as k_id
        from kontstate, kontract, knaktlist
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end_2 between ks_date_begin and ks_date_end and
              ks_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              knaktlist.is_delete = 0 and
              kal_new_reg_id = :id_list and
              :rep_date_end_2 between kal_date and kal_date_end)
    select sum(ok_os)
    from tabl1, operkred
    where ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date_end_2 and
          ok_from_c_id = 0 and
          ok_type in (4, 5, 6, 8) and
          ok_from_k_id = k_id
    into reg_balance_last;
    if (reg_balance_last is null) then
      reg_balance_last = 0;
--  Расчет баланса региона за текущий месяц
    reg_balance_curr = null;
    with tabl1
    as (select
               case
                 when ks_date_begin >= '01.06.2012' then k_id
                 else case
                        when ks_date_begin < '01.06.2012' and coalesce((select rep_delinq_days
                                                                        from aarep_kont_delinq_status_new(k_id, '01.12.2012')), 0) < 180 then k_id
                        else 0
                      end
               end as k_id
        from kontstate, kontract, knaktlist
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end between ks_date_begin and ks_date_end and
              ks_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              knaktlist.is_delete = 0 and
              kal_new_reg_id = :id_list and
              :rep_date_end between kal_date and kal_date_end)
    select sum(ok_os)
    from tabl1, operkred
    where ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date_end and
          ok_from_c_id = 0 and
          ok_type in (4, 5, 6, 8) and
          ok_from_k_id = k_id
    into reg_balance_curr;
    if (reg_balance_curr is null) then
      reg_balance_curr = 0;

--  Расчет просроченного баланса текущего месяца
    reg_balance_delinq = null;
    with tabl1
    as (select
               case
                 when coalesce((select rep_delinq_days
                                from aarep_kont_delinq_status_new(k_id, :rep_date_end)), 0) between 8 and 180 then k_id
                 else 0
               end as k_id
        from kontstate, kontract, knaktlist
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end between ks_date_begin and ks_date_end and
              ks_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              knaktlist.is_delete = 0 and
              kal_new_reg_id = :id_list and
              :rep_date_end_2 between kal_date and kal_date_end)
    select sum(ok_os)
    from tabl1, operkred
    where ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date_end and
          ok_from_c_id = 0 and
          ok_type in (4, 5, 6, 8) and
          ok_from_k_id = k_id
    into reg_balance_delinq;
    if (reg_balance_delinq is null) then
      reg_balance_delinq = 0;
--  Расчет PAR,
    if (reg_balance_last > 0) then
      par7 = reg_balance_delinq / reg_balance_last * 100;
    else
      par7 = 100;
--  Расчет суммы бонусов
    cb_bonus_sb_total = 0;
    cb_bonus_sb_total = case
                          when par7 <= 2.0 then :reg_balance_curr * 0.20 / 100
                          when par7 between 2.1 and 4 then :reg_balance_curr * 0.15 / 100
                          when par7 between 4.1 and 6 then :reg_balance_curr * 0.10 / 100
                          else 0
                        end;
--  Расчет кол-во СБ данного офиса
    select count(ps_from_c_id)
    from hr_pstatus, joblist
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_reg_id = :id_list and
          ps_status > -1 -- не аннулированные приказы
          and
          ps_is_working = 1 -- все активные сотрдуники
          and
          ps_from_jl_id = jl_id and
          jl_kind = 8
    into reg_kolich_sb;
--  Если кол-во ССВЗ > 1 данного офиса то все суммы расчетов делит на кол-во ССВЗ
    if (reg_kolich_sb > 1) then
    begin
      reg_balance_last = reg_balance_last / reg_kolich_sb;
      reg_balance_curr = reg_balance_curr / reg_kolich_sb;
      reg_balance_delinq = reg_balance_delinq / reg_kolich_sb;
      cb_bonus_sb_total = cb_bonus_sb_total / reg_kolich_sb;
    end
    else
    if (reg_kolich_sb < 1) then
      cb_bonus_sb_total = 0;
    if (cb_bonus_sb_total > 6000.0) then
      cb_bonus_sb_total = 6000.0;
    cb_personal_stage = null;
--  Вытаскивает ФИО и должность СБ
    personal_name = null;
    jl_name = null;
    for select c_name_fio, jl_name, c_id
        from hr_pstatus, joblist, clients
        where :rep_date_end between ps_date_begin and ps_date_end and
              ps_from_reg_id = :id_list and
              ps_status > -1 -- не аннулированные приказы
              and
              ps_is_working = 1 -- все активные сотрдуники
              and
              ps_from_jl_id = jl_id and
              jl_kind = 8 -- Выборка только по ССВЗ
              and
              ps_from_c_id = c_id
        into personal_name, jl_name, cb_personal_id
    do
    begin
--    Дата поступления на работу
      cb_personal_stage = null;
      select ps_date_begin
      from hr_pstatus --, hr_orders , joblist
      where ps_from_c_id = :cb_personal_id and
            ps_date_begin is not null and
            ps_status = 1
      order by ps_date_begin
      rows 1
      into :cb_personal_stage;

      if (cb_personal_stage is not null) then
      begin
        if (cb_personal_stage > dateadd(month, -1, :rep_date_end)) then
          cb_bonus_sb_total = 0;
      end

      suspend;
    end
    if (jl_name is null) then
      suspend; -- если в данном офисе нету ССВЗ то возвращает данные

--    suspend;
  end
end
