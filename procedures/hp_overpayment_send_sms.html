CREATE PROCEDURE HP_OVERPAYMENT_SEND_SMS
AS
declare variable sms_text d_str_long;
declare variable phone d_str_20;
declare variable c_id d_id;
declare variable lifetime_days d_integer;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable overpay_sum decimal(15,2);
declare variable ovm_id d_id;
declare variable msg_id d_integer;
declare variable k_id d_id;
begin
  lifetime_days = -1;

  for select ovm_id, kc_from_c_id, ok_cell_main, k_kontract_no, k_disb_date, k_id
  from overpayment_msg ovm, knclient kc, continfo cci, kontract k
  where ovm_status = 999        -- готовы к отправке
    and ovm_from_msg_id is null -- еще не отправлено
    and ovm_is_delete = 0
    and kc_id = ovm_from_kc_id
    and cci.cci_object_kind = 0
    and cci_object_id = kc_from_c_id
    and k_id = kc_from_k_id
  rows 200                  -- максимальное количество СМС отправляемых за раз
  into ovm_id, c_id, phone, k_kontract_no, k_disb_date, k_id
  do
  begin
    if (coalesce(phone, '') = '') then
    begin
      update overpayment_msg ovm
      set ovm.ovm_status = -2     -- нет основного номера телефона
      where ovm_id = :ovm_id;

      continue;
    end

    if (f_clientcom_allowed(c_id, 0, 1, 0) = 0) then
      continue;

    overpay_sum = (select -sum(ok_os)
                  from operkred
                  where ok_from_k_id = :k_id
                    and ok_ol_is_active = 1
                    and ok_type = 9);

    if (coalesce(overpay_sum, 0) < 1.00) then
    begin
      update overpayment_msg ovm
      set ovm.ovm_status = -1     -- сумма меньше 1р
      where ovm_id = :ovm_id;

      continue;
    end

    sms_text = f_format('Pri pogashenii vashego zaima po dogovoru №%1 ot %2 obrazovalas pereplata %3 rublei. Za vozvratom deneg obratites'' s pasportom v lyuboi ofis OOO "MKK M Bulak".',
                        k_kontract_no,
                        f_datetostr(k_disb_date),
                        overpay_sum);

    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                msg_command_kind, msg_object_kind, msg_object_id)
    values (1, :phone, '', :sms_text, :lifetime_days, 1000000000000, 'sms', 0, :c_id)
    returning msg_id
    into msg_id;

    update overpayment_msg
    set ovm_from_msg_id = :msg_id
    where (ovm_id = :ovm_id);
  end
end
