CREATE PROCEDURE HP_ETNDOC_LOAD(
    TXT D_BLOB_TEXT)
RETURNS (
    PRODUCTID D_STR_CODE,
    UNITCODE D_STR_CODE,
    AMOUNT D_CURRENCY,
    PRICE D_CURRENCY)
AS
declare variable res_str d_str_8100;
declare variable pb d_integer;
declare variable pe d_integer;
begin
  for
    select
      res_str
    from hp_splitb_str(:txt, ascii_char(13)||ascii_char(10))
  into
    :res_str
  do
  begin
    pb = position('<productid>', res_str);
    if (pb > 0) then
      pe = position('</productid>', res_str);
    if (pb > 0 and pe > 0) then
      productid = substring(res_str from pb + 11 for pe - pb - 11);

    pb = position('<unitcode>', res_str);
    if (pb > 0) then
      pe = position('</unitcode>', res_str);
    if (pb > 0 and pe > 0) then
      unitcode = substring(res_str from pb + 10 for pe - pb - 10);

    pb = position('<amount>', res_str);
    if (pb > 0) then
      pe = position('</amount>', res_str);
    if (pb > 0 and pe > 0) then
      amount = substring(res_str from pb + 8 for pe - pb - 8);

    pb = position('<price>', res_str);
    if (pb > 0) then
      pe = position('</price>', res_str);
    if (pb > 0 and pe > 0) then
      price = substring(res_str from pb + 7 for pe - pb - 7);

    if (productid is not null and unitcode is not null
      and amount is not null and price is not null) then
    begin
      suspend;
      productid = null;
      unitcode = null;
      amount = null;
      price = null;
    end
  end
end
