CREATE PROCEDURE HP_EDOC_WEB_BANKCARD_INS_CUST(
    UUID D_STR_CODE,
    CODE D_STR_CODE,
    CARD_NUMBER D_STR_CODE,
    EXP_YEAR D_SMALLINT,
    EXP_MONTH D_SMALLINT,
    CARDHOLDER D_STR_CODE)
RETURNS (
    RES_TEXT D_STR_255,
    RES_BUTTON_STATE D_STR_CODE)
AS
declare variable obj_id d_id;
declare variable edl_status d_integer;
declare variable obj_code d_str_code;
declare variable c_id d_id;
declare variable edc_code d_str_code;
declare variable t d_integer;
declare variable exp_per d_integer;
declare variable card_number_masked d_str_code;
begin
  -- Сохранение данных банковской карты для проекта bankcard.elcart
  -- Сохранение в extsys_token, epay_acc/card на триггере.
  res_text = '';
  res_button_state = 'disabled';

  if (res_text = '' and char_length(card_number) <> 16) then
  begin
    res_text = 'Неправильно введен номер карты';
    res_button_state = 'enabled';
  end

  t = extract(year from current_date) * 100 + extract(month from current_date);
  if (exp_year < 100) then
    exp_year = exp_year + 2000;
  exp_per = exp_year * 100 + exp_month;

  if (res_text = '' and exp_per < t) then
  begin
    res_text = 'Срок действия карты истек';
    res_button_state = 'enabled';
  end

  if (res_text = '' and code is null) then
    res_text = 'Код подтверждения не передан';

  obj_code = '';
  if (res_text = '') then
  begin
    select obj_kind, obj_id
    from hp_uuid(:uuid)
    into obj_code, :obj_id;
  
    if (obj_id is null) then
      res_text = 'Неверная ссылка';
    else
      obj_code = f_objkind(obj_code, 'name');

    if (obj_code <> 'edoc_sign') then
      res_text = 'Неверная ссылка';
  end


  if (res_text = '') then
  begin
    select iif(coalesce(edl_date_end, current_date) < current_date, -1, edl_status),
      edc_code, eds_from_c_id
    from edoc_sign, edoc_list, edoc_cfg
    where eds_id = :obj_id
      and edl_from_edc_id = edc_id
      and eds_from_edl_id = edl_id
    into edl_status, edc_code, c_id;
  end

  if (res_text = '' and edc_code <> 'bankcard.elcart') then
    res_text = 'Неверный проект';

  if (res_text = '' and edl_status = -1) then
    res_text = 'Данные устарели';

  if (res_text = '' and edl_status = 1) then
    res_text = 'Данные карты уже успешно привязаны';


  if (res_text = '' and c_id is null) then
    res_text = 'Отсутствует идентификатор клиента';

  if (res_text = '' and exists(
      select *
      from extsys_token
      where et_token = :card_number
        and et_from_c_id <> :c_id)) then
    res_text = 'Карта ранее использовалась другим клиентом. Обратитесь в офис Компании';


  if (res_text = '') then
  begin
    -- Маскируем часть номера карты. Пример: 521178******2582
    card_number_masked = left(card_number, 6) || '******' || right(card_number, 4);

    insert into extsys_token (et_from_esm_id, et_from_c_id, et_code,
      et_cardnumber, et_token, et_token_datetime, et_token_esm_id,
      et_is_actual)
    values (null, :c_id, :code, :card_number_masked, :card_number,
      current_timestamp, null, 1);

    update epay_card
    set
      epc_expire_period = :exp_per,
      epc_cardholder = :cardholder
    where epc_card_id = :card_number;

    update edoc_msg
    set edm_status = 1
    where edm_from_eds_id = :obj_id and
          edm_status = 0;

    update edoc_sign
    set eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'online', 'id')
    where eds_id = :obj_id and
          eds_enum_edoc_sign_kind <> f_enum('edoc_sign_kind', 'online', 'id');

    res_text = 'Данные карты успешно сохранены';

  end

  Suspend;
end
