CREATE PROCEDURE AAREP_SIRENA_BLOCK_DATA(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OPER_CALLED STR_100,
    BLOCK_DATE DATE,
    KF_NAME STR_100,
    PROJECT_TYPE STR_100,
    CC_PROJECT STR_100)
AS
declare variable k_id id_type;
begin
  for select pantk.pant_from_k_id, pantk.pant_oper_called, pantk.pant_date_begin, rd.rd_name
      from pantera_kontracts pantk
      inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
      where pantk.pant_date_begin between :rep_date_begin and :rep_date_end -- блокировки за период
--and pantk.pant_project_type in (4, 5, 6) -- Сирена, Сирена1, Сирена2
            and
            rd.rd_code in ('OKTELL.Projects.Sirena', 'OKTELL.Projects.Sirena1', 'OKTELL.Projects.Sirena2') and
            pantk.pant_date_end >= pantk.pant_date_begin
      order by pant_oper_called
      into :k_id, :oper_called, :block_date, cc_project
  do
  begin
    select kal.kal_new_kf_name, iif(kal.kal_new_kf_officer = 1000000000004, 'КЦ Сирена', 'Просрочка')
    from kontract kont
    inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
    where kont.k_id = :k_id and
          kal.is_delete = 0 and
          :block_date between kal.kal_date and kal.kal_date_end
    into :kf_name, project_type;

    suspend;
  end
end
