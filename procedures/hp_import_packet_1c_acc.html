CREATE PROCEDURE HP_IMPORT_PACKET_1C_ACC
RETURNS (
    NEW_ACC_COUNT D_INTEGER,
    IMPORTED_COUNT D_INTEGER)
AS
declare variable pca_id d_id;
declare variable pca_code d_str_code;
declare variable pca_name d_str_255;
declare variable pca_date_open d_date;
declare variable pca_bank_account d_str_code;
declare variable pca_bank_code d_str_code;
declare variable pca_from_ab_id d_id;
declare variable pca_from_cu_id d_str_lang;
declare variable pca_from_reg_id d_id;
declare variable pca_from_inx_id d_id;
declare variable pca_number d_id;
declare variable err_text d_str_long;
declare variable pca_object_kind d_integer;
declare variable pca_object_id d_id;
declare variable pca_date_close d_date;
declare variable pca_non_resident d_integer;
declare variable acc_id d_id;
declare variable acc_from_sab_id d_id;
declare variable sab_invert_code d_str_code;
declare variable al_id d_str_code;
declare variable pca_client_guid d_str_code;
begin
-- Импорт лицевых счетов из таблицы packet_1c_acc в Ref_Account;
  new_acc_count = 0;
  imported_count = 0;

  update packet_1c_acc
  set pca_links_filled = 0
  where pca_is_exist = 0 and
        pca_links_filled = -1;

  update packet_1c_acc
  set pca_links_filled = 0
  where pca_imported = -1 and
        pca_links_filled = 1;

  for select pca_id, pca_is_exist, pca_imported, pca_links_filled, pca_code, pca_name, pca_office_name,
             pca_currency_name, pca_date_open, pca_date_close, pca_bank_account, pca_bank_code, pca_client_type,
             pca_non_resident, pca_client_name, pca_client_birth_date, pca_client_inn, pca_client_pass_no,
             pca_client_guid, pca_from_ab_id, pca_from_sab_id, pca_from_cu_id, pca_from_reg_id, pca_from_inx_id,
             pca_object_kind, pca_object_id, pca_number, pca_message
      from packet_1c_acc
      where pca_is_exist = 0 and
            pca_imported <> 1 as cursor cur
  do
  begin
    err_text = '';
    new_acc_count = new_acc_count + 1;

    if (cur.pca_from_ab_id is null) then
      err_text = err_text || iif(err_text = '', '', ', ') || 'Балансовый счет';

    if (cur.pca_from_reg_id is null) then
      err_text = err_text || iif(err_text = '', '', ', ') || 'Код офиса';

    if (left(cur.pca_code, 1) = '7' and
        cur.pca_from_inx_id is null) then
      err_text = err_text || iif(err_text = '', '', ', ') || 'Код ОФР';

    if (left(cur.pca_code, 5) in ('20501', '20803') and
        coalesce(cur.pca_bank_account, '') = '') then
      err_text = err_text || iif(err_text = '', '', ', ') || 'Расчетный счет';

    if (cur.pca_object_id is null and
        coalesce(cur.pca_client_name, '') <> '') then
      err_text = err_text || iif(err_text = '', '', ', ') || 'Контрагент';

    if (err_text <> '') then
    begin
      update packet_1c_acc
      set pca_imported = -1,
          pca_message = :err_text
      where pca_id = :pca_id;
    end
    else
    begin
      update packet_1c_acc
      set pca_imported = 0,
          pca_message = ''
      where pca_id = :pca_id and
            pca_imported <> 0;
    end
  end

  for select pca_id, trim(pca_code), trim(pca_name), pca_date_open, pca_date_close, trim(pca_bank_account),
             trim(pca_bank_code), pca_from_ab_id, pca_from_cu_id, pca_from_reg_id, pca_object_kind, pca_object_id,
             pca_from_inx_id, pca_number, pca_non_resident, trim(pca_client_guid)
      from packet_1c_acc
      where pca_is_exist = 1 and
            pca_client_name <> '' and
            pca_object_id is not null
      into :pca_id, :pca_code, :pca_name, :pca_date_open, :pca_date_close, :pca_bank_account, :pca_bank_code,
           :pca_from_ab_id, :pca_from_cu_id, :pca_from_reg_id, :pca_object_kind, :pca_object_id, :pca_from_inx_id,
           :pca_number, :pca_non_resident, :pca_client_guid
  do
  begin
    if (exists(select *
               from ref_account
               where acc_code = :pca_code and
                     acc_object_id is null)) then
    begin
      update ref_account
      set acc_object_kind = :pca_object_kind,
          acc_object_id = :pca_object_id
      where acc_code = :pca_code;

      if (pca_object_id is not null and
          coalesce(pca_client_guid, '') <> '') then
      begin
        update or insert into packet_1c_guid (pcg_object_kind, pcg_object_id, pcg_guid)
        values (:pca_object_kind, :pca_object_id, :pca_client_guid)
        matching (pcg_object_kind, pcg_object_id);
      end
    end
  end

  for select pca_id, trim(pca_code), trim(pca_name), pca_date_open, pca_date_close, trim(pca_bank_account),
             trim(pca_bank_code), pca_from_ab_id, pca_from_cu_id, pca_from_reg_id, pca_object_kind, pca_object_id,
             pca_from_inx_id, pca_number, pca_non_resident, trim(pca_client_guid)
      from packet_1c_acc
      where pca_imported = 0 and
            pca_is_exist = 0
      into :pca_id, :pca_code, :pca_name, :pca_date_open, :pca_date_close, :pca_bank_account, :pca_bank_code,
           :pca_from_ab_id, :pca_from_cu_id, :pca_from_reg_id, :pca_object_kind, :pca_object_id, :pca_from_inx_id,
           :pca_number, :pca_non_resident, :pca_client_guid
  do
  begin
    if (not exists(select *
                   from ref_account
                   where acc_code = :pca_code)) then
    begin
      acc_id = null;
      acc_from_sab_id = null;

      insert into ref_account (acc_from_ab_id, acc_from_reg_id, acc_from_cu_id, acc_from_inx_id, acc_object_kind,
                               acc_object_id, acc_code, acc_number, acc_name, acc_date_open, acc_date_close,
                               acc_non_resident)
      values (:pca_from_ab_id, :pca_from_reg_id, :pca_from_cu_id, :pca_from_inx_id, :pca_object_kind, :pca_object_id,
              :pca_code, :pca_number, :pca_name, :pca_date_open, :pca_date_close, :pca_non_resident)
      returning acc_id, acc_from_sab_id
      into :acc_id, :acc_from_sab_id;

      select sab_invert_code
      from sys_accbalance
      where sab_id = :acc_from_sab_id
      into :sab_invert_code;

      if (sab_invert_code starting 'РасчСчет.Банк' or sab_invert_code starting 'СпецСчет.Банк') then
      begin
        for select br_name
            from bankregion
            where br_account_no = :pca_bank_account and
                  br_from_b_id = :pca_bank_code
            into :al_id
        do
        begin
          update acc_list
          set al_from_acc_id = :acc_id
          where al_id = :al_id and
                al_from_acc_id is null;
        end
      end

      update packet_1c_acc
      set pca_imported = 1,
          pca_is_exist = 1
      where pca_id = :pca_id;

      imported_count = imported_count + 1;
    end
    else
    begin
      update packet_1c_acc
      set pca_is_exist = 1
      where pca_id = :pca_id;
    end

    if (pca_object_id is not null and
        coalesce(pca_client_guid, '') <> '') then
    begin
      update or insert into packet_1c_guid (pcg_object_kind, pcg_object_id, pcg_guid)
      values (:pca_object_kind, :pca_object_id, :pca_client_guid)
      matching (pcg_object_kind, pcg_object_id);
    end
  end
  suspend;
end
