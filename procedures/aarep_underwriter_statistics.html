CREATE PROCEDURE AAREP_UNDERWRITER_STATISTICS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    IS_UNDERWRITER SMALLINT)
RETURNS (
    RES_REG_NAMEP STR_100,
    RES_REG_NAME STR_100,
    SERTIFIKATOR STR_100,
    KOL_PROCESSED INTEGER,
    KOL_REFUSED INTEGER,
    KOL_REFUSED_NEW INTEGER,
    KOL_REJECTED INTEGER)
AS
begin
  if (is_underwriter is null) then
    is_underwriter = 0;

  for select reg_namep, reg_name, iif(:is_underwriter = 1, first_user_fio, officer_name), -- 1 - на уровне андерайтеров; 0 - на уровне СВЗ
             count(*), -- общее кол-во обработанных заявок
             sum(iif(first_status = 'Доработка', 1, 0)), -- кол-во отправленных на доработку
             sum(iif(first_status = 'Доработка' and first_status_reason not in ('Доработка: Доработки вне учета','Доработка: Технические доработки'), 1, 0)), -- кол-во отправленных на доработку (по 7-ми станд.причинам)
             sum(iif(first_status = 'Отказан', 1, 0)) -- кол-во отказанных
      from aarep_underwriter_consolid_new(:rep_region_id, :rep_date_begin, :rep_date_end)
      group by 1, 2, 3
      into :res_reg_namep, :res_reg_name, :sertifikator, :kol_processed, :kol_refused, :kol_refused_new, :kol_rejected
  do
  begin
    suspend;
  end
end
