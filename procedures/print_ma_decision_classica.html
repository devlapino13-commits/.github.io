CREATE PROCEDURE PRINT_MA_DECISION_CLASSICA(
    K_ID D_ID)
RETURNS (
    MA_NO INTEGER,
    K_KONTRACT_NO D_STR_100,
    APPROVED_MANAGER D_STR_100,
    K_PERIOD D_INTEGER,
    K_PERCENT D_PERCENT,
    APPROVED_DATE D_DATE,
    FOLDER_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    KC_STATUS D_STR_100,
    CLIENT_FIO D_STR_100,
    LOAN_TARGET D_STR_100,
    KOMMIS_PERC D_PERCENT,
    LOAN_AMOUNT D_CURRENCY,
    PR_NAME D_STR_255,
    IS_CASH D_SMALLINT,
    POINT D_INTEGER,
    RISK_RATE_DESCRIPTION D_STR_CODE,
    BY_RATING D_SMALLINT)
AS
declare variable kf_id d_id;
declare variable pr_code d_str_100;
declare variable ma_manager_id d_id;
declare variable k_disb_date d_date;
declare variable kbm_score_rate d_integer;
declare variable risk_rate char(2);
declare variable ma_id d_id;
declare variable c_id d_id;
begin
  select maa.ma_no, kont.k_kontract_no, kont.k_period, kont.k_percent,
         kont.k_from_kf_id, kont.k_kommis_perc, pr.pr_name, iif(kont.k_disb_kind = 1, 1, 0),
         pr.pr_code, maa.ma_manager_id, coalesce(k_by_rating, 0), k_disb_date, ma_id
  from kontract kont
  inner join miniapp maa on maa.ma_id = kont.k_from_ma_app_id
  inner join product pr on pr.pr_id = kont.k_from_pr_id
  where kont.k_id = :k_id
  and maa.ma_date_end = '31.12.9999'
  into :ma_no, :k_kontract_no, :k_period, k_percent,
       :kf_id, :kommis_perc, :pr_name, is_cash, pr_code, ma_manager_id, by_rating, k_disb_date, ma_id;

  select max(kl.kl_datetime)
  from kontlog kl
  where kl.kl_from_k_id = :k_id and
        kl.kl_state = 1
  into :approved_date;

  select cl.c_name_fio
  from kontlog kl
  inner join username un on kl.kl_user_id = un.un_id
  inner join clients cl on un.un_from_c_id = cl.c_id
  where kl.kl_from_k_id = :k_id
  order by kl.kl_datetime desc
  rows 1
  into :approved_manager;

  if (pr_code containing '.sugarmill.' or
      pr_code containing '.installment_goods.') then
  begin
    select cl.c_name_fio
    from clients cl
    where cl.c_id = :ma_manager_id
    into :approved_manager;
  end 

  select kal.kal_new_kf_name,
         reg.reg_name
  from knaktlist kal
  inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
  where kal.is_delete = 0
  and kal.kal_from_kf_id = :kf_id
  and :approved_date between kal.kal_date and kal.kal_date_end
  into :folder_name,
       :office_name;

  for select trim(iif(kncl.kc_status = 2, 'Казначей', 'Клиент')),
             cl.c_name_fio,
             rd_name as loan_target,
             kncl.kc_amount_approved,
             kc_from_c_id,
             f_point(kc_from_c_id, :k_disb_date, 0)
  from knclient kncl
  inner join clients cl on cl.c_id = kncl.kc_from_c_id
  inner join refdata rd on rd.rd_id = kncl.kc_loan_target
  where kncl.kc_from_k_id = :k_id
  and kncl.kc_amount_approved > 0
  into :kc_status, :client_fio, :loan_target, :loan_amount, :c_id, :point
  do
  begin
    kbm_score_rate = 0;
    select kbm_score_rate
    from hp_kibminiapp_last(:c_id, :ma_id)
    into kbm_score_rate;

    risk_rate = (case
                   when kbm_score_rate between 250 and 250 then 'E3'
                   when kbm_score_rate between 251 and 529 then 'E2'
                   when kbm_score_rate between 530 and 573 then 'E1'
                   when kbm_score_rate between 574 and 603 then 'D3'
                   when kbm_score_rate between 604 and 619 then 'D2'
                   when kbm_score_rate between 620 and 640 then 'D1'
                   when kbm_score_rate between 641 and 652 then 'C3'
                   when kbm_score_rate between 653 and 664 then 'C2'
                   when kbm_score_rate between 665 and 676 then 'C1'
                   when kbm_score_rate between 677 and 688 then 'B3'
                   when kbm_score_rate between 689 and 700 then 'B2'
                   when kbm_score_rate between 701 and 712 then 'B1'
                   when kbm_score_rate between 713 and 730 then 'A3'
                   when kbm_score_rate between 731 and 742 then 'A2'
                   when kbm_score_rate between 743 and 900 then 'A1'
                   else '--'
                 end);

    risk_rate_description = (case
                             when risk_rate in ('E3', 'E2', 'E1') then 'Очень высокий риск'
                             when risk_rate in ('D3', 'D2', 'D1') then 'Высокий риск'
                             when risk_rate in ('C3', 'C2', 'C1') then 'Средний риск'
                             when risk_rate in ('B3', 'B2', 'B1') then 'Низкий риск'
                             when risk_rate in ('A3', 'A2', 'A1') then 'Очень низкий риск'
                             else '--'
                           end);
    suspend;
  end
end
