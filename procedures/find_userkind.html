CREATE PROCEDURE FIND_USERKIND(
    FIND_KIND VARCHAR(100))
RETURNS (
    RES_ID DECIMAL(18,0),
    RES_NAME VARCHAR(255),
    RES_MEMO VARCHAR(255))
AS
declare variable c_id decimal(18,0);
declare variable rn_id decimal(18,0);
declare variable role_tmp varchar(255);
declare variable kol integer;
declare variable un_id integer;
declare variable uniqtab_username integer;
declare variable uniqtab_rolename integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;

  if (find_kind = 'User') then
  begin
    for select un_id, un_from_c_id
        from username
        into :res_id, c_id
    do
    begin
      if (exists(select un_is_active
                 from hp_un_is_active(:res_id)
                 where un_is_active = 1)) then
      begin
        res_name = '';
        if (c_id is not null) then
        begin
          select c_name_fio
          from clients
          where c_id = :c_id
          into :res_name;
        end

        res_memo = '';
        for select distinct pm_subject_id
            from permissions
            where pm_object_kind = :uniqtab_username and
                  pm_object_id = :res_id and
                  pm_subject_kind = :uniqtab_rolename and
                  pm_is_delete = 0 and
                  (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                  (coalesce(pm_date_end, '31.12.2999') >= current_date)
            into :rn_id
        do
        begin
          select rn_name
          from rolename
          where rn_id = :rn_id
          into :role_tmp;
          if (res_memo = '') then
            res_memo = role_tmp;
          else
            res_memo = ', ' || role_tmp;
        end
        suspend;
      end
    end
  end
  if (find_kind = 'Role') then
  begin
    for select rn_id, rn_name
        from rolename
        where rn_is_active = 1
        into :res_id, res_name
    do
    begin
      res_memo = '';
      kol = 0;
      for select pm_object_id
          from permissions
          where pm_object_kind = :uniqtab_username and
                pm_subject_kind = :uniqtab_rolename and
                pm_subject_id = :res_id and
                pm_is_delete = 0 and
                (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                (coalesce(pm_date_end, '31.12.2999') >= current_date)
          into :un_id
      do
      begin
        if (exists(select *
                   from hp_un_is_active(:un_id)
                   where un_is_active = 1)) then
          kol = kol + 1;
      end
      if (kol = 0) then
        res_memo = 'Нет действующих пользователей';
      else
        res_memo = 'Пользователей: ' || kol;
      if (kol > 0) then
      begin
        res_name = lower(res_name);
        suspend;
      end
    end
  end
end
