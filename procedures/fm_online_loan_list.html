CREATE PROCEDURE FM_ONLINE_LOAN_LIST(
    UN_ID D_ID,
    NEED_FULL_OFFICE D_SMALLINT = 0)
RETURNS (
    C_NAME_FIO D_STR_100,
    K_AMOUNT_APPROVED DECIMAL(18,0),
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    K_ID D_ID)
AS
declare variable c_id d_id;
declare variable reg_id d_id;
declare variable jk_system_code d_str_code;
begin
  select un_from_c_id
  from username un
  where un.un_id = :un_id
  into c_id;

  select ps.ps_from_reg_id, jk.jk_system_code
  from hr_pstatus ps
  inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
  inner join jobkind jk on jk.jk_id = jl.jl_kind
  where ps.ps_from_c_id = :c_id and
        current_date between ps.ps_date_begin and ps.ps_date_end and
        ps.ps_status = 1 and
        ps.ps_is_working = 1
  into reg_id, jk_system_code;

  if (jk_system_code not in ('jobsystem.region_km', 'jobsystem.lo_standart', 'jobsystem.lo_senior',
                             'jobsystem.finstart_programmer')) then
    exit;

  for select cl.c_name_fio, kt.k_amount_approved, kt.k_kontract_no, kt.k_disb_date, kt.k_id
      from kontstate ks
      inner join expressapp exa on exa.exa_from_k_id = ks.ks_from_k_id
      inner join online_app oa on oa.oa_from_exa_id = exa.exa_id and
            iif(:need_full_office = 1, oa.oa_is_fired_lo = 1, oa.oa_from_c_id = :c_id) and
            oa.oa_lo_denied_datetime is null
      inner join kontract kt on kt.k_id = ks.ks_from_k_id
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id and
            kc.kc_amount_approved > 0
      inner join clients cl on cl.c_id = kc.kc_from_c_id
      inner join knaktlist kal on kal.kal_from_kf_id = kt.k_from_kf_id and
            kal.is_delete = 0 and
            current_date between kal.kal_date and kal.kal_date_end and
            kal.kal_new_kf_officer = 1000000000102
      where ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_date_begin <= current_date and
            ks.ks_region_id = :reg_id
      into c_name_fio, k_amount_approved, k_kontract_no, k_disb_date, k_id
  do
  begin
    suspend;
  end 
end
