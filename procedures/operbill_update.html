CREATE PROCEDURE OPERBILL_UPDATE(
    BILL_ID DECIMAL(18,0),
    BILL_LINE_NO INTEGER,
    OL_ID DECIMAL(18,0))
AS
declare variable d date;
declare variable d_prior date;
declare variable k_no decimal(18,0);
declare variable chet integer;
begin
  /* Procedure Text */
  chet = 0;
  for select ob_date_begin, ob_from_ol_id
      from operbill
      where ob_bill_id = :bill_id and
            ob_bill_line_no = :bill_line_no and
            ob_is_delete = 0
      order by ob_date_begin desc, ob_from_ol_id desc
      into :d, :k_no
  do
  begin
    chet = :chet + 1;
    if (chet > 1) then
    begin
      update operbill
      set ob_date_end = :d_prior - 1
      where ob_bill_id = :bill_id and
            ob_bill_line_no = :bill_line_no and
            ob_from_ol_id = :k_no and
            ob_date_end <> :d_prior - 1;
    end
    else
    begin
      update operbill
      set ob_date_end = '31.12.9999'
      where ob_bill_id = :bill_id and
            ob_bill_line_no = :bill_line_no and
            ob_from_ol_id = :k_no and
            ob_date_end <> '31.12.9999';
    end
    d_prior = d;
  end
end
