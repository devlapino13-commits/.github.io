CREATE PROCEDURE HP_OBSUMMARY_CREATE_SALDOS
RETURNS (
    OL_DATE D_DATE)
AS
declare variable ob_account d_str_code;
declare variable ob_dt_summa d_currency;
declare variable ob_dt_summa_val d_currency;
declare variable ob_kt_summa d_currency;
declare variable ob_kt_summa_val d_currency;
begin
  delete from obsummary
  where 1 = 1
  order by obs_from_al_id desc, obs_date desc;

  for select al_id
  from acc_list
  order by al_id
  into ob_account
  do
  begin
    for with tab as (
        select ob_ol_date as ol_date,
               sum(ob_summa) as ob_dt_summa,
               sum(ob_summa_val) as ob_dt_summa_val,
               sum(0.0) as ob_kt_summa,
               sum(0.0) as ob_kt_summa_val
        from operbook
        where ob_account_dt = :ob_account and
              ob_ol_is_active = 1
        group by 1
        union all
        select ob_ol_date as ol_date,
               sum(0.0) as ob_dt_summa,
               sum(0.0) as ob_dt_summa_val,
               sum(ob_summa) as ob_kt_summa,
               sum(ob_summa_val) as ob_kt_summa_val
        from operbook
        where ob_account_kt = :ob_account and
              ob_ol_is_active = 1
        group by 1)
        select ol_date, sum(ob_dt_summa), sum(ob_dt_summa_val), sum(ob_kt_summa), sum(ob_kt_summa_val)
        from tab
        group by 1
        into :ol_date, :ob_dt_summa, :ob_dt_summa_val, :ob_kt_summa, :ob_kt_summa_val
    do
    begin
      insert into obsummary (obs_from_al_id, obs_date, obs_dt_ekv, obs_dt_val, obs_kt_ekv, obs_kt_val)
      values (:ob_account, :ol_date, :ob_dt_summa, :ob_dt_summa_val, :ob_kt_summa, :ob_kt_summa_val);
    end

    suspend;
  end
end
