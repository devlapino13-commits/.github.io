CREATE PROCEDURE HP_CANCEL_TERMINATION(
    K_ID D_ID,
    TO_DATE D_DATE)
RETURNS (
    RES_TXT D_STR_255)
AS
declare variable ol_id d_id;
declare variable ol_datetime d_datetime;
declare variable k_stop_perc d_date;
begin
  res_txt = '';

  select k_stop_perc, ol_id, ol_datetime
  from kontract
  left join operlist on ol_id = k_stopped_perc_ol_id and
        is_delete = 0 and
        ol_date = :to_date
  where k_id = :k_id
  into k_stop_perc, ol_id, ol_datetime;

  if (ol_id is not null) then
  begin
    -- Есть операция сворачивания графика
    if (not exists(select *
                   from operkred
                   inner join operlist on ol_id = ok_from_ol_id and
                         is_delete = 0 and
                         ol_id > :ol_id and
                         ol_datetime > :ol_datetime
                   where ok_from_k_id = :k_id)) then
    begin
      update operlist
      set is_delete = 1
      where ol_id = :ol_id;
    end
    else
      res_txt = 'Есть активные операции после расторжения.';
  end
  else
  if (k_stop_perc is not null) then
  begin
    -- График не сворачивался, operlist'a нет
    -- Проверяем только операции датой приостановки и позже
    if (exists(select *
               from operkred
               inner join operlist on ol_id = ok_from_ol_id and
                     is_delete = 0 and
                     ol_datetime >= :k_stop_perc
               where ok_from_k_id = :k_id)) then
      res_txt = 'Есть активные операции в дату расторжения или позже.';
  end
  else
    res_txt = 'На дату не найдено расторжение договора.';

  if (res_txt = '') then
    update kontract
    set k_stop_perc = null,
        k_stop_fine = null
    where k_id = :k_id;

  suspend;
end
