CREATE PROCEDURE HP_TRAVEL_EMP(
    REP_REGION_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    C_ID ID_TYPE,
    REG_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    JOB_NAME VARCHAR(255),
    HRV_DATE_BEGIN DATE,
    HRV_DATE_END DATE,
    HRV_DIRECTION VARCHAR(100),
    HO_ORDER_NO VARCHAR(100),
    HO_REASON VARCHAR(255),
    HO_DATE DATE)
AS
begin
  for select hrv.hrv_date_begin, hrv.hrv_date_end_fact, hrv_direction, ho_order_no, ho_date, ho_reason, hrv_from_c_id
      from hr_orders hro, hr_voctravel hrv
      where ho_order_type = 12 and
            hro.ho_id = hrv.hrv_from_ho_id and
            hrv_date_begin between :rep_date_begin and :rep_date_end and
            hro.ho_status = 1 and
            hrv.hrv_status = 1
      into hrv_date_begin, hrv_date_end, hrv_direction, ho_order_no, ho_date, ho_reason, c_id
  do
  begin
    for select jl_name, reg_name, c_name_fio
        from clients, hr_pstatus, hr_orders, sa_worktype, joblist, region, region_list(:rep_region_id, 1, -1, 1)
        where c_id = ps_from_c_id and
              ps_status = 1 and
              ps_from_c_id = :c_id and
              :hrv_date_end between ps_date_begin and ps_date_end and
              ps_from_ho_id = ho_id and
              ho_order_type = sawt_id and
              ps_from_jl_id = jl_id and
              ps_from_reg_id = reg_unicode and
              (reg_unicode = id_list)
        into job_name, reg_name, c_name_fio
    do
    begin
      suspend;
    end
  end
end
