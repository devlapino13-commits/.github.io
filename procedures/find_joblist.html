CREATE PROCEDURE FIND_JOBLIST(
    TEXT_STR VARCHAR(100),
    IS_JOBLIST INTEGER)
RETURNS (
    LIST_JL_ID INTEGER,
    LIST_JL_NAME VARCHAR(100),
    LIST_JK_NAME VARCHAR(100),
    LIST_JOB_TITLE VARCHAR(255))
AS
begin
  if (text_str is null) then
    text_str = '';
  text_str = upper(:text_str);
  if (is_joblist = 1) then
  begin
    for select distinct jl_id, jl_name, jk_name, jtl_name
        from joblist, jobkind, jobtitle
        where jl_kind = jk_id and
              jl_title = jtl_id and
              ((jl_name like '%' || :text_str || '%') or
              (jk_name like '%' || :text_str || '%') or
              (jk_memo like '%' || :text_str || '%') or
              (jtl_name like '%' || :text_str || '%') or
              (jtl_memo like '%' || :text_str || '%'))
        into :list_jl_id, :list_jl_name, :list_jk_name, :list_job_title
    do
      suspend;
  end
  else
  begin
    list_jl_name = '';
    for select distinct jk_id, jk_name, jk_memo
        from joblist, jobkind, jobtitle
        where jl_title = jtl_id and
              jl_kind = jk_id and
              ((jk_name like '%' || :text_str || '%') or
              (jk_memo like '%' || :text_str || '%') or
              (jtl_name like '%' || :text_str || '%') or
              (jtl_memo like '%' || :text_str || '%'))
        union
        select distinct jk_id, jk_name, jk_memo
        from jobkind
        where ((jk_name like '%' || :text_str || '%') or
              (jk_memo like '%' || :text_str || '%'))
        into :list_jl_id, :list_jk_name, :list_job_title
    do
      suspend;
  end
end
