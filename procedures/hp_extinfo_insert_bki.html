CREATE PROCEDURE HP_EXTINFO_INSERT_BKI(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    FORCE_INSERT D_SMALLINT,
    UN_ID D_ID)
AS
declare variable main_config d_str_code;
declare variable add_config d_str_code;
declare variable obj_kind_str d_str_code;
declare variable pr_code d_str_100;
declare variable c_id d_id;
declare variable ma_id d_id;
declare variable ma_kind_id d_integer;
declare variable k_refund d_integer;
declare variable z_from_c_id d_id;
declare variable k_id d_id;
begin
  main_config = f_get_const('main_bki_config');

  execute procedure hp_extinfo_insert(main_config, obj_kind, obj_id, force_insert, un_id);

  -- RFCRU-3290 Проверка на Самозапрет осуществлять отдельно по каждой заявке на заем
  obj_kind_str = f_objkind(obj_kind, 'name');

  if (obj_kind_str = 'Kontract') then
  begin
    select pr_code, kc_from_c_id, k_from_ma_app_id, k_refund, k_id
    from kontract, product, knclient
    where k_id = :obj_id
      and k_from_pr_id = pr_id
      and kc_from_k_id = k_id
      and kc_amount_approved > 0
    into pr_code, c_id, ma_id, k_refund, k_id;
  end
  else if (obj_kind_str = 'Miniapp') then
  begin
    ma_id = obj_id;

    select pr_code, mac_from_c_id
    from miniapp, product, miniappclient
    where ma_id = :obj_id
      and ma_from_pr_id = pr_id
      and mac_from_ma_id = ma_id
    order by ma_number desc
    rows 1
    into pr_code, c_id;

    select k_id, k_refund
    from kontract
    where k_from_ma_app_id = :ma_id
    order by k_status desc
    rows 1
    into k_id, k_refund;
  end

  ma_kind_id = f_objkind('Miniapp', 'id');

  execute procedure hp_extinfo_insert_bki_cfg('bki_ssp_40', :c_id, :ma_kind_id, :ma_id, :force_insert, :un_id);

  if (obj_kind_str = 'Kontract' and
      pr_code containing '.zalog') then
  begin
    select z_from_c_id
    from zalog
    where z_from_k_id = :obj_id and
          z_subtype = 4 and
          z_type = 1 and
          z_from_c_id is not null and
          current_date between coalesce(z_date_begin, '01.01.1990') and coalesce(z_date_end, '31.12.9999')
    order by coalesce(z_date_begin, '01.01.1990') rows 1
    into z_from_c_id;

    if (c_id <> z_from_c_id) then
    begin
      execute procedure hp_extinfo_insert_bki_cfg('equifax_40', :z_from_c_id, :ma_kind_id, :ma_id, :force_insert, :un_id);
      execute procedure hp_extinfo_insert_bki_cfg('nbki_40', :z_from_c_id, :ma_kind_id, :ma_id, :force_insert, :un_id);
    end
  end

  -- дополнительная проверка
  add_config = 'nbki_40';

  -- если основная и дополнительная конфигурации совпадают, то ничего не делаем
  if (add_config = main_config) then
    exit;

  if (pr_code containing '.zalog' or
      pr_code containing '.reliable_emp.' or
      pr_code containing '.refinance.' or -- RFCRU-4976
      pr_code containing '.refinance_oivo.' or
      coalesce(k_refund, 0) = 1 or
      pr_code containing '.education.')
  then
  begin
    -- RFCRU-4976 НБКИ нужен только один на контракт/заявку, т.к. в ПДН не участвует
    -- исключение повторый выбор рефинансов, но там решается ручным запросом
    -- поэтому если есть по заявке/договору уже запрос повторно автоматически не запрашиваем
    if (exists(select eil_id
              from extinfolist, extinfocfg
              where eic_code = :add_config
                and eil_from_eic_id = eic_id
                and eil_from_c_id = :c_id
                and eil_status >= 0
                and ((eil_object_kind = 11 and eil_object_id = :ma_id)
                  or (eil_object_kind = 9 and eil_object_id = coalesce(:k_id, 0)))))
    then
      exit;

    execute procedure hp_extinfo_insert(add_config, obj_kind, obj_id, 0, un_id);
  end
end
