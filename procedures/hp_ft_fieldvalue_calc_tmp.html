CREATE PROCEDURE HP_FT_FIELDVALUE_CALC_TMP(
    FIL_ID D_ID,
    C_ID D_ID,
    FIELD_NAME D_STR_CODE,
    REP_DATE D_DATE,
    FROM_HISTORY D_SMALLINT)
RETURNS (
    RES_TEXT D_STR_LONG)
AS
declare variable region_id integer;
declare variable fio str_100;
declare variable info_date d_date;
declare variable tmp_value d_str_long;
declare variable crlf d_str_10;
declare variable crlf2 d_str_10;
declare variable fib_kind d_smallint;
begin
  crlf = _ascii x'0D0A'; -- Перевод строки
  crlf2 = crlf || crlf; -- Два перевода строки

  if (from_history is null) then
    from_history = 0;
  info_date = current_date;
  if (from_history = 1) then
    info_date = rep_date;

  res_text = '';

  select fib.fib_kind
  from ft_infolist fil
  inner join ft_infoblank fib on fil.fil_from_fib_id = fib.fib_id
  where fil_id = :fil_id
  into fib_kind;

  if (field_name in ('client_fio', 'client_fio (blank:1-3)')) then
  begin
    select res_name_fio
    from ft_fio_change_hist(:c_id, 1)
    where cast(res_datetime as date) <= :info_date
    order by res_datetime desc
    rows 1
    into :res_text;
  end
  else
  if (field_name = 'city') then
  begin
    -- Извлекаю Город офиса, последнего займа, полученного клиентом, по отношению к текущей дате.
    select fil_from_reg_id
    from ft_infolist
    where fil_id = :fil_id
    into region_id;

    select pl.pl_name
    from region reg
    inner join addrlist adl on adl.adl_id = reg.reg_address_legal
    inner join places pl on pl.pl_id = adl.adl_from_pl_id
    where reg.reg_unicode = :region_id
    into :res_text;

    if (res_text is null) then
      res_text = '';
  end
  else
  if (field_name in ('header',
                     'header (blank:1-3)')) then
  begin
     if (field_name = 'header (blank:1-3)') then
        res_text = 'СВЕДЕНИЯ, ПОЛУЧАЕМЫЕ В ЦЕЛЯХ ИДЕНТИФИКАЦИИ КЛИЕНТОВ – ФИЗИЧЕСКИХ ЛИЦ, ПРЕДСТАВИТЕЛЕЙ КЛИЕНТА, ВЫГОДОПРИОБРЕТАТЕЛЕЙ – ФИЗИЧЕСКИХ ЛИЦ И БЕНЕФИЦИАРНЫХ ВЛАДЕЛЬЦЕВ (НУЖНОЕ ПОДЧЕРКНУТЬ)';
     else
        res_text = 'СВЕДЕНИЯ, ПОЛУЧАЕМЫЕ В ЦЕЛЯХ ИДЕНТИФИКАЦИИ КЛИЕНТОВ – ФИЗИЧЕСКИХ ЛИЦ, ИДЕНТИФИКАЦИИ ПРЕДСТАВИТЕЛЕЙ КЛИЕНТА, ВЫГОДОПРИОБРЕТАТЕЛЕЙ – ФИЗИЧЕСКИХ ЛИЦ И БЕНЕФИЦИАРНЫХ ВЛАДЕЛЬЦЕВ КЛИЕНТОВ';
  end
  else
  if (field_name = 'citizen') then
  begin
    if (from_history = 1) then
    begin
      select rd_name
      from clientdocs, passkind, refdata
      where cds_from_c_id = :c_id and
            cds_create_datetime < (:info_date + 1) and
            cds_doc_date_begin <= :info_date and
            coalesce(cds_doc_date_end, :info_date) >= :info_date and
            cds_is_main_doc = 1 and
            cds_from_pk_id = pk_id and
            pk_country_id = rd_id --and
            --cds_is_delete = 0
      order by cds_create_datetime desc
      rows 1
      into res_text;
    end
    if (from_history = 0 or
        coalesce(res_text, '') = '') then
    begin
      res_text = (select rd.rd_name
                  from clients cl
                  inner join refdata rd on rd.rd_id = cl.c_citizen
                  where c_id = :c_id);
    end
  end
  else
  if (field_name = 'date_place_birth') then
  begin
    select iif(cl.c_citizen = 13, f_datetostr(c_birth_date) || ', ' || coalesce(c_birth_country, '') || ', ' || coalesce(c_birth_city, ''), f_datetostr(c_birth_date) || ', ' || coalesce(c_birth_country, ''))
    from clients cl
    where c_id = :c_id
    into :res_text;
  end
  else
  if (field_name in ('pass_props', 'pass_props (blank:1-3)')) then
  begin
    if (from_history = 1) then
    begin
      select 'Серия, номер: ' || coalesce(cds_doc_no, '') || '; Дата выдачи: ' || f_datetostr(cds_doc_date_begin) || '; Орган, выдавший документ: ' ||
          coalesce(cds_doc_issued_by, '') || iif(:rep_date < cast('24.07.2025' as date) or cds_from_pk_id not in (12,13), '', coalesce('; Код подразделения: ' || cds_doc_issued_code, ''))
      from clientdocs
      where cds_from_c_id = :c_id and
            cds_create_datetime < (:info_date + 1) and
            cds_is_main_doc = 1 --and
            --cds_is_delete = 0  -- igusev 11.10.2019 убрал данное условие т.к. в триггере CLIENTS_AU0 есть скрипт, который аннулирует основной документ если прикрепляется новый
      order by cds_create_datetime desc
      rows 1
      into res_text;
    end
    if (from_history = 0 or
        coalesce(res_text, '') = '') then
    begin
      select 'Серия, номер: ' || cl.c_pass_no || '; Дата выдачи: ' || f_datetostr(cl.c_pass_date) || '; Орган, выдавший документ: ' || cl.c_pass_who ||
         iif(:rep_date < cast('24.07.2025' as date) or c_citizen <> 13, '', coalesce('; Код подразделения: ' || c_pass_who_code, ''))
      from clients cl
      where c_id = :c_id
      into :res_text;
    end
  end
  else
  if (field_name = 'registration_fiz') then
  begin
     select 'Номер: ' || coalesce(cds_doc_no, '') || '; Дата выдачи: ' || f_datetostr(cds_doc_date_begin) || '; Дата окончания: ' || f_datetostr(cd.cds_doc_date_end) || ';'
        from clientdocs cd, passkind, passdoctype
        where cds_from_c_id = :c_id and
              cds_from_pk_id = pk_id and
              pk_from_pdt_id = pdt_id and
              pdt_code = 'registration_fiz' and
              cds_is_delete = 0 and
              cds_create_datetime < (:info_date + 1) and
              :info_date between cd.cds_doc_date_begin and cd.cds_doc_date_end
        order by cds_create_datetime desc
        rows 1
        into :res_text;
        if (coalesce(res_text, '') = '') then
          res_text = '---';
  end
  else
  if (field_name = 'mcard_props') then
  begin
    select res_text
    from ft_mcard_props_get(:c_id, :info_date)
    into :res_text;
  end
  else
  if (field_name = 'acard_props') then
  begin
    select res_text
    from ft_acard_props_get(:c_id, :info_date)
    into :res_text;
  end
  else
  if (field_name = 'address_legal') then
  begin

    select adl_address_str
    from addrclog, addrlist
    where adcl_from_c_id = :c_id and
          adcl_address_type = 1 and
          adcl_from_adl_id = adl_id and
          adcl_datetime >= :info_date
    order by adcl_datetime
    rows 1
    into res_text;

    if (coalesce(res_text, '') = '') then
    begin
      select adl.adl_address_str
      from addrclient adcl
      inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
      where adcl.adc_from_c_id = :c_id and
            adcl.adc_address_type = 1
      into :res_text;
    end
    if (coalesce(res_text, '') = '') then
      res_text = '---';
  end
  else
  if (field_name in ('inn', 'inn (blank:1-3)')) then
  begin
    res_text = (select coalesce(cl.c_inn, '')
                from clients cl
                where c_id = :c_id);
  end
  else
  if (field_name = 'snils') then
  begin
    if ((select cl.c_citizen
         from clients cl
         where cl.c_id = :c_id) = 13) then
    begin
      select cds_doc_no
      from clientdocs, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = pdt_id and
            pdt_code = 'snils' and
            cds_is_delete = 0 and
            cds_create_datetime < (:info_date + 1)
      order by cds_create_datetime desc, cds_doc_date_begin desc
      rows 1
      into :res_text;
    end

    if (coalesce(res_text, '') = '') then
      res_text = '---';
  end
  else
  if (field_name in ('cell_no', 'cell_no (blank:1-3)')) then
  begin
    res_text = '';

    if (f_get_const('new_contdata') = 0) then
    begin
      select cff.cv_contact_value
      from contverify  cff
      where cff.cv_object_kind = 0
      and cff.cv_object_id = :c_id
      and cff.cv_from_rd_id = f_get_rd_id('ContactInfo.Cell_Main')
      and cff.cv_status >= 1
      and cff.cv_datetime < (:info_date + 1)
      order by cv_id desc
      rows 1
      into :res_text;
  
      res_text = coalesce(res_text, '');
    end else
    begin
      select f_cont_code(cda_from_clt_id)
      from contdata, contveri
      where cda_from_c_id = :c_id and
            cda_cont_kind = f_enum('cont_kind', 'cell', 'id') and
            cda_is_main = 1 and
            cda_id = cvr_from_cda_id and
            cvr_status = 1 and
            cvr_datetime < (:info_date + 1)
      order by cvr_id desc
      rows 1
      into res_text;
    end

    if (res_text = '') then
      res_text = '---';
  end
  else
  if (field_name in ('email', 'email (blank:1-3)')) then
  begin
    res_text = '';
    if (f_get_const('new_contdata') = 0) then
    begin
      select cff.cv_contact_value
      from CONTVERIFY  cff
      where cff.cv_object_kind = 0
      and cff.cv_object_id = :c_id
      and cff.cv_from_rd_id = f_get_rd_id('ContactInfo.Email')
      and cff.cv_status >= 1
      and cff.cv_datetime < (:info_date + 1)
      order by cv_id desc
      rows 1
      into :res_text;
  
      res_text = coalesce(res_text, '');
    end else
    begin
      select f_cont_code(cda_from_clt_id)
      from contdata, contveri
      where cda_from_c_id = :c_id and
            cda_cont_kind = f_enum('cont_kind', 'email', 'id') and
            cda_id = cvr_from_cda_id and
            cvr_status = 1 and
            cvr_datetime < (:info_date + 1)
      order by cvr_id desc
      rows 1
      into res_text;
    end
    if (res_text = '') then
      res_text = '---';
  end
  else
  if (field_name in ('job_position', 'job_position (blank:1-3)')) then
  begin
    res_text = 'Не является';
  end
  else
  if (field_name = 'relative_status') then
  begin
    res_text = 'Не является';
  end
  else
  if (field_name in ('representative_doc', 'representative_doc (blank:1-3)')) then
  begin
    res_text = 'Не имеет';
  end
  else
  if (field_name = 'beneficiary') then
  begin
    res_text = 'Выгодоприобретатели отсутствуют';
  end
  else
  if (field_name in  ('relation_purpose',
                      'relation_purpose (blank: 2)',
                      'relation_purpose (blank: 3)',
                      'relation_purpose (blank: 4)',
                      'relation_purpose (blank: 5-7)'
                      )) then
  begin
    if (field_name in  ('relation_purpose (blank: 2)',
                        'relation_purpose (blank: 3)')) then
    begin
        if ((select res_risk_categ
        from hp_ft_risk_categ(:c_id, :info_date)) = 0) then
            res_text = '---';
        else
            res_text = 'Получение микрозайма';
    end
    else
    begin
        if ((select res_risk_categ
        from hp_ft_risk_categ(:c_id, :info_date)) = 0) then
        begin
          res_text =
            '---' || crlf2 ||
            '---' || crlf2 ||
            '---';
        end
        else
        begin
          res_text =
            'Получение микрозайма' || crlf2 ||
            'Долгосрочное сотрудничество для периодческого получения заемных средств' || crlf2 ||
            'Отсутствуют';
        end
    end
  end
  else
  if (field_name in  ('business_relationship (blank: 2)',
                      'business_relationship (blank: 3)')) then
  begin
    if ((select res_risk_categ
        from hp_ft_risk_categ(:c_id, :info_date)) = 0) then
        res_text = '---';
    else
        res_text = 'Долгосрочное сотрудничество для периодческого получения заемных средств';
  end
  else
  if (field_name in  ('activity_purpose')) then
  begin
    if ((select res_risk_categ
        from hp_ft_risk_categ(:c_id, :info_date)) = 0) then
        res_text = '---';
    else
        res_text = 'Отсутствуют';
  end
  else
  if (field_name in ('finance_state', 'finance_state (blank: 2,3)')) then
  begin
    res_text = '---';
    if ((select res_risk_categ
         from hp_ft_risk_categ(:c_id, :info_date)) = 1) then
      res_text = 'Устойчивое';
  end
  else
  if (field_name in ('reputation', 'reputation (blank: 2,3)')) then
  begin
    res_text = '---';
    if ((select res_risk_categ
         from hp_ft_risk_categ(:c_id, :info_date)) = 1) then
      res_text = 'Положительная';
  end
  else
  if (field_name in ('finance_source', 'finance_source (blank: 2,3)')) then
  begin
    res_text = 'Заработная плата';
  end
  else
  if (field_name in ('beneficial_owner',
                     'beneficial_owner (blank:4)',
                     'beneficial_owner (blank:5)',
                     'beneficial_owner (blank:6,7)')) then
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into :fio;

    res_text =
      'Бенефициарный владелец - ' || :fio || crlf2 ||
      'Бенефициарный владелец - ' || :fio || ' в связи с отсутствием оснований полагать, что бенефициарным владельцем является иное физическое лицо';
  end
  else
  if (field_name = 'beneficial_client_status') then
  begin
    res_text = 'Отсутствует';
  end
  else
  if (field_name in ('risk_info',
                     'risk_info (blank:1)',
                     'risk_info (blank:2,3)')) then
  begin
    if (field_name = 'risk_info (blank:1)') then
    begin
        res_text = 'Риск по типу клиента:' || crlf;
        res_text = res_text|| 'Высокий' || crlf;
        res_text = res_text|| 'В соответствии с иными факторами, самостоятельно определенными компанией в соответствии с правилами внутреннего контроля ООО «МОЛ.БУЛАК.РУ» в целях противодействия легализации (отмыванию) доходов, полученных преступным путем, и финансированию терроризма';
    end
    else
    begin
        select (case found_source
                  when '639_list'  then 'Группа риска клиента: "высокий", в связи с получением информации (сведений), в соответствии с 639-П'
                  when 'citizen'   then 'Группа риска клиента: "высокий" (гражданин республики таджикистан)'
                  when 'blacklist' then 'Группа риска клиента: «повышенный» (совпадение с Перечнем экстремистов/ Решением о применении мер по замораживанию)'
                  when '639_list_0102_update' then 'Степень (уровень) риска «стандартный» в связи с отсутствием факторов «критичного» и «высокого» уровней риска'
                  else 'ГРУППА РИСКА КЛИЕНТА: «НИЗКИЙ» В СВЯЗИ С ОТСУТСТВИЕМ ФАКТОРОВ «ПОВЫШЕННОГО» И «ВЫСОКОГО» УРОВНЕЙ РИСКА'
                end)
        from hp_ft_risk_categ(:c_id, :info_date)
        into :res_text;
    end
  end
  if (field_name = 'exist_in_fp_list') then
  begin
    res_text = 'Отсутствует в перечне на дату '||f_datetostr(:info_date);
  end
  else
  if (field_name = 'solution_in_fp') then
  begin
    res_text = 'Решение отсутствует на дату '||f_datetostr(:info_date);
  end
  else
  if (field_name = 'other_country_pers') then
  begin
    res_text = 'Не является';
  end
  else
  if (field_name in ('other_info',
                     'other_info (blank:1)',
                     'other_info (blank:2,3)')) then
  begin
    if (field_name = 'other_info (blank:1)') then
         res_text = 'Представители отсутствуют, бенефициарные владельцы отсутствуют, клиент не действует к выгоде третьего лица';
    else if (field_name = 'other_info (blank:2,3)') then
    begin
        select cl.c_name_fio
        from clients cl
        where cl.c_id = :c_id
        into :fio;

        res_text = 'Представители отсутствуют, клиент не действует к выгоде третьего лица. Бенефициарный владелец – '||fio||' в связи с отсутствием оснований полагать, что бенефициарным владельцем является иное физическое лицо';
    end
    else
    begin
        res_text = 'Представители отсутствуют, клиент не действует к выгоде третьего лица.';

        if (exists(select *
             from finpolis_639_cli_link fcl -- если найдено совпадение по перечню 639, тогда уровень риска - Высокий
             where fcl.fcl_from_c_id = :c_id
             and fcl.fcl_date <= :info_date
             and :rep_date >= '01.02.2021'
             and fcl.fcl_date >= '01.02.2021')) then
       begin
            res_text = res_text || 'В отношении клиента получена информация (сведения) в соответствии с 764-П';
       end
    end
  end
  else
  if (field_name = 'start_relation') then
  begin
    /*
    select f_datetostr(ks_date_begin)
    from kontstate ks
    inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
    where ks.ks_state = 1 and
          ks.ks_date_begin <= :info_date and
          ks.ks_ol_is_delete = 0 and
          kncl.kc_amount_approved > 0 and
          kncl.kc_from_c_id = :c_id
    order by ks.ks_date_begin desc
    rows 1
    into :res_text;  */

    --select f_datetostr(start_relation_date)
    --from ft_start_relation_calc(:info_date, :c_id)
    --into :res_text;
  end
  else
  if (field_name in ('ank_create_date',
                     'ank_create_date (blank:1-3)')) then
  begin
    select f_datetostr(fil_date_open)
    from ft_infolist
    where fil_id = :fil_id
    into :res_text;
  end
  else
  if (field_name in ('resp_fio',
                     'resp_fio (blank:1-3)',
                     'resp_fio (blank:4)',
                     'resp_fio (blank:5-7)')) then
  begin
    select res_text
    from ft_resp_fio_get(:c_id, :info_date)
    into :res_text;
  end
  else
  if (field_name = 'sign_pers') then
  begin
    select 'Подпись:_________________'|| :crlf || res_text
    from ft_resp_fio_get(:c_id, :info_date)
    into :res_text;
  end
  else
  if (field_name = 'relation_finished') then
  begin
    res_text = '';
  end
  else
  if (field_name in ('footer', 'footer (blank:1-3)')) then
  begin
