CREATE PROCEDURE OKT_SITE_MINIAPP_LIST(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ESM_ID ID_TYPE,
    ESM_DATE DATE,
    ESM_DATETIME TIMESTAMP,
    CITY STR_100,
    CL_NAME STR_100,
    CL_BIRTHDAY DATE,
    CL_CITIZENSHIP STR_100,
    CL_GENDER STR_100,
    CL_LOAN_TERM STR_100,
    PHONE_NO STR_100,
    CK_BLOCKED_OPER STR_100,
    CK_MEMO VARCHAR(1000),
    CK_STATUS INTEGER,
    CK_CALL_QTY INTEGER,
    IS_BLOCK SMALLINT,
    CK_NEXT_CALL_DATETIME TIMESTAMP,
    CL_LOAN_AMOUNT CURR_TYPE,
    LAST_CHECKED_DATETIME TIMESTAMP,
    LAST_MEMO VARCHAR(1000),
    CK_CITY ID_TYPE,
    CL_NEXT_CALL_VALUE TIMESTAMP,
    CL_NEXT_CALL_TYPE STR_100,
    CL_KOLICH_REPAY INTEGER,
    CL_LANGUAGE VARCHAR(20),
    ACTION_CALL_DATETIME TIMESTAMP,
    MEETING_POINT STR_255,
    MEETING_ADDRESS STR_255,
    DATA_SOURCE STR_100)
AS
declare variable old_phone_no str_100;
declare variable ck_blocked_datetime timestamp;
declare variable ck_checked_date date;
declare variable ck_fio str_100;
declare variable old_esm_date date;
begin
  old_phone_no = '';
  old_esm_date = cast('01.01.1900' as date);
-- вытаскиваю заявки с сайта за период
  for select esm_id, esm_date, cl_name, cl_birthday, cl_citizenship, cl_gender, cl_loan_term, phone_no, esm_datetime,
             city, cl_loan_amount, cl_next_call_value, cl_next_call_type, cl_kolich_repay, cl_language, meeting_point,
             meeting_address, data_source
      from aarep_website_miniapp_analiz(:rep_date_begin, :rep_date_end, 1, 1, 1)
      where esm_date < '11.06.2019'
      and (trim(coalesce(phone_no, '')) <> '')
      order by phone_no, esm_date, esm_id
      into :esm_id, :esm_date, :cl_name, :cl_birthday, :cl_citizenship, :cl_gender, :cl_loan_term, :phone_no,
           :esm_datetime, :city, :cl_loan_amount, :cl_next_call_value, :cl_next_call_type, :cl_kolich_repay,
           :cl_language, :meeting_point, :meeting_address, :data_source
  do
  begin
    phone_no = iif(phone_no starting(7), phone_no, '7' || phone_no);

    if ((phone_no <> old_phone_no) or (esm_date <> old_esm_date)) then -- убираю дубли, когда клиент дважды в течении дня отправил заявку
    begin
      action_call_datetime = iif(cl_next_call_value is null, esm_datetime, cl_next_call_value);

      old_esm_date = esm_date;
      old_phone_no = phone_no;
      ck_blocked_datetime = null;
      ck_checked_date = null;
      ck_blocked_oper = null;
      ck_memo = '';
      ck_status = null;
      ck_call_qty = 0;
      ck_next_call_datetime = null;
      last_checked_datetime = null;
      last_memo = '';
      ck_city = null;

-- определею общее число звонков и дату/время последнего звонка
      for select chk.ck_checked_datetime, chk.ck_memo
          from cc_check_knclient chk
          where chk.ck_from_esm_id = :esm_id and
                chk.ck_type = 6 and
                chk.ck_checked_date is not null
          order by chk.ck_checked_datetime
          into :last_checked_datetime, :last_memo
      do
      begin
        ck_call_qty = ck_call_qty + 1;
      end

      ck_fio = null;

-- вытаскиваю последний удавшийся звонок
      select chk.ck_blocked_datetime, chk.ck_blocked_oper, chk.ck_memo, chk.ck_status, chk.ck_checked_date,
             chk.ck_next_call_datetime, chk.ck_fio, chk.ck_city
      from cc_check_knclient chk
      where chk.ck_from_esm_id = :esm_id and
            chk.ck_type = 6 and
            chk.ck_status <> -1 -- не дозвон не считаю
      order by chk.ck_id desc
      rows 1
      into :ck_blocked_datetime, :ck_blocked_oper, :ck_memo, :ck_status, :ck_checked_date, :ck_next_call_datetime,
           :ck_fio, :ck_city;
      if (coalesce(ck_fio, '') <> '') then
        cl_name = ck_fio;

-- если статус не конечный и число звонков меньше 6 тогда ...
      if (((ck_checked_date is null) or (ck_status in (0, 1))) and
          (ck_call_qty < 6)) then
      begin
        is_block = 0;
        if (ck_blocked_datetime is not null) then
          is_block = 1;
        if (ck_next_call_datetime is null) then
        begin
          suspend;
        end
        else
        begin
--if (ck_next_call_datetime  <= dateadd(hour, 3, current_timestamp)) then
--begin
          suspend;
-- end
        end
        /*
                else if (rep_date_end = current_date) then
                begin
                    if (ck_next_call_datetime  <= dateadd(hour, 3, current_timestamp)) then
                    begin
                        suspend;
                    end    
                end
                else if (rep_date_end <> current_date) then
                begin
                    if (cast(ck_next_call_datetime as date) <= :rep_date_end) then
                    begin
                        suspend;
                    end     
                end  */
      end
    end
  end
end
