CREATE PROCEDURE HP_LAW_GET_RESOLUTION(
    K_ID ID_TYPE,
    C_ID ID_TYPE)
RETURNS (
    RESOLUTION_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable k_kontract_no str_100;
declare variable k_disb_date date;
declare variable c_name_fio str_100;
declare variable c_birth_date date;
declare variable c_pass_no str_100;
declare variable k_disb_date_str str_100;
declare variable c_birth_date_str str_100;
declare variable address_legal str_255;
begin
  select kont.k_kontract_no, kont.k_disb_date
  from kontract kont
  where kont.k_id = :k_id
  into :k_kontract_no, :k_disb_date;

  k_disb_date_str = replace(iif(extract(day from k_disb_date) < 10, '0', '') || extract(day from k_disb_date) || '.' || iif(extract(month from k_disb_date) < 10, '0', '') || extract(month from k_disb_date) || '.' || extract(year from k_disb_date), ' ', '');
  select cl.c_name_fio, cl.c_birth_date, cl.c_pass_no
  from clients cl
  where cl.c_id = :c_id
  into :c_name_fio, :c_birth_date, :c_pass_no;

  c_birth_date_str = replace(iif(extract(day from c_birth_date) < 10, '0', '') || extract(day from c_birth_date) || '.' || iif(extract(month from c_birth_date) < 10, '0', '') || extract(month from c_birth_date) || '.' || extract(year from c_birth_date), ' ', '');
  select adl_address_str
  from addrlist adl
  inner join addrclient adcl on adcl.adc_from_adl_id = adl.adl_id
  where adcl.adc_from_c_id = :c_id and
        adcl.adc_address_type = 1 -- Юридический (Адрес регистрации)
  into :address_legal;

  --resolution_text = 'Взыскать с ' || :c_name_fio || ' ' || :c_birth_date_str || ' г. р. паспорт ' || :c_pass_no || ' адрес ' || coalesce(:address_legal, 'XXX') || ', задолженность по основной сумме займа и процентам в сумме XXX рублей по договору займа № ' || :k_kontract_no || ' от ' || :k_disb_date_str || ' года, расходы по оплате государственной пошлины в размере XXX рублей.';
  resolution_text = 'Взыскать с ' || :c_name_fio || ' ' || :c_birth_date_str || ' г.р. пасп. ' || :c_pass_no || ' адр. ' || coalesce(:address_legal, 'XXX') || ', задолж. по ОС и %% в сумме XXXр. по дог. №' || :k_kontract_no || ' от ' || :k_disb_date_str || 'г. и расходы по оплате ГП - XXXр.';
  suspend;
end
