CREATE PROCEDURE UXP$UXACTION_LIFETIME_SET(
    UA_UUID D_UUID,
    KEY_VALUE D_ID,
    LIFETIME_IN_SECS D_INTEGER)
AS
declare variable ux_fcode d_str_100;
declare variable ux_uuid  d_uuid;
declare variable id       d_id;
begin
  if (coalesce(key_value, 0) = 0) then
    exit;

  select ux_fcode, uxa_owner
  from ux$action, ux$alias, ux$code
  where ua_uuid = :ua_uuid and
        ua_uxa_uuid = uxa_uuid and
        uxa_owner = ux_uuid
  into ux_fcode, ux_uuid;

  if (ux_fcode = 'fm_perstemp.kk') then
  begin
    select pte_acl_id
    from perstemp
    where pte_from_c_id = :key_value
    into id;

    if (coalesce(id, 0) = 0) then
    begin
      select acl_id
      from actionlist
      where acl_object_kind = 0 and
            acl_object_id = :key_value and
            acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO'))
      order by acl_id desc
      rows 1
      into id;
    end

    if (coalesce(id, 0) <> 0) then
    begin
      update actionlist
      set acl_lifetime_in_second = :lifetime_in_secs
      where acl_id = :id;
    end
  end else
  begin
    update or insert into ux$lifetime (ult_key_value, ult_ux_uuid, ult_lifetime_in_secs)
    values (:key_value, :ux_uuid, :lifetime_in_secs)
    matching (ult_key_value, ult_ux_uuid);
  end
end
