CREATE PROCEDURE HP_CLIENT_RELATION_TMP(
    REP_C_ID DECIMAL(18,0),
    MAX_LEVEL INTEGER)
RETURNS (
    REL_OTHER_C_ID DECIMAL(18,0),
    REL_CODE VARCHAR(50),
    REL_LEVEL INTEGER,
    REL_IS_ACTIVE INTEGER,
    REL_MEMO VARCHAR(255),
    REL_LINK_C_ID DECIMAL(18,0),
    REL_LINK_OTHER_C_ID DECIMAL(18,0),
    REL_LINK_REL_ID VARCHAR(50),
    C_SEX_ORIG INTEGER,
    CREL_FULL_CODE VARCHAR(50),
    MARK D_BOOL)
AS
declare variable prev_id decimal(18,0);
declare variable db_id   d_integer;
declare variable use_mark boolean;
begin
  db_id = f_get_db_id();
  use_mark = (db_id = 41701)
    and exists(
      select *
      from  hr_pstatus
      where ps_from_c_id = :rep_c_id
        and current_date between ps_date_begin and ps_date_end
        and ps_status = 1
        and ps_is_working = 1);

  select c_sex
  from clients
  where c_id = :rep_c_id
  into :c_sex_orig;

  prev_id = 0;
  rel_is_active = 1;
-- rel_path - id уже пройденых родственников, нужен чтобы не повторяться
-- rel_depth - расстояние до родственника, нужно чтобы выбрать минимальное
  for with recursive
-- первая итерация
      rel1
      as (-- родственники, направление вперед
      select crel_other_c_id as rel_id, crel_from_rel_id as rel_type,
             cast(:rep_c_id as varchar(255)) || '.' || crel_other_c_id as rel_path, 0 as rel_depth,
             crel_from_c_id as crel_id1, crel_other_c_id as crel_id2
      from clientrelation
      where crel_from_c_id = :rep_c_id and
            (:use_mark or crel_is_active = 1)
      union
-- родственники, направление назад
      select crel_from_c_id,
             (select rel_code
              from hp_get_relation_reverse(crel_from_rel_id, c_sex)),
             cast(:rep_c_id as varchar(255)) || '.' || crel_from_c_id, 1, crel_from_c_id, crel_other_c_id
      from clientrelation, clients
      where crel_other_c_id = :rep_c_id and
            c_id = crel_from_c_id and
            (:use_mark or crel_is_active = 1)),

      rel2
      as (-- берем первую итерацию
      select rel_id, rel_type, rel_path, rel_depth, crel_id1, crel_id2
      from rel1
      union all
-- здесь начинается цикл
      select -- собираем родственников по направлению вперед
crel_other_c_id, rel_type || crel_from_rel_id, rel_path || '.' || crel_other_c_id, rel_depth + 1, crel_from_c_id,
crel_other_c_id
      from clientrelation, rel2
      where crel_from_c_id = rel_id and
            rel_depth <= :max_level -- макс. глубина вложенности
            and (:use_mark or crel_is_active = 1) and
            position(crel_other_c_id, rel_path) = 0 -- запрет повторений
      union all
-- и по направлению назад
      select crel_from_c_id, rel_type ||(select rel_code
                                         from hp_get_relation_reverse(crel_from_rel_id, c_sex)),
             rel_path || '.' || crel_from_c_id, rel_depth + 1, crel_from_c_id, crel_other_c_id
      from clientrelation, clients, rel2
      where crel_other_c_id = rel_id and
            c_id = crel_from_c_id
--    and crel_from_rel_id = rel_id
            and
            (:use_mark or crel_is_active = 1) and
            rel_depth <= :max_level -- макс. глубина вложенности
            and
            position(crel_from_c_id, rel_path) = 0 -- запрет повторений
      ),
      fltr
      as (-- находим длину самого короткого пути до каждого родственника
      select rel_id, min(rel_depth) as min_rel_depth
      from rel2
      group by rel_id)
-- возвращаем родственника и самый короткий путь до него, с остальными полями
      select rel2.rel_id, rel_type, rel_depth, crel_id1, crel_id2
      from rel2, fltr
      where rel2.rel_id = fltr.rel_id and
            rel_depth = min_rel_depth
      order by rel2.rel_id, rel_type
      into rel_other_c_id, rel_code, rel_level, rel_link_c_id, rel_link_other_c_id
  do
  begin
    mark = 0;
    if (rel_other_c_id <> prev_id) then
    begin
      select crel_from_rel_id, crel_memo, mark
      from clientrelation
      where crel_from_c_id = :rel_link_c_id and
            crel_other_c_id = :rel_link_other_c_id and
            crel_is_active = 1
      into rel_link_rel_id, rel_memo, :mark;
      prev_id = rel_other_c_id;

      crel_full_code = rel_code;
      rel_code = (select rel_code
                  from hp_get_relation_short_id(:rel_code, :c_sex_orig));

      -- RFCKG-3841
      if (db_id <> 81001) then
      begin
          if(rel_code in ('1', '2') and rel_level > 2) then continue; -- Супруг/Супруга
          if(rel_code in ('13', '14', '23', '24') and rel_level > 2) then continue; -- Свекр/Свекровь Тесть/Теща
      end

      if (not use_mark or mark = 1) then
      suspend;
    end
  end
end
