CREATE PROCEDURE HP_DOCRECOGREQUEST_CREATE(
    C_ID D_ID,
    DP_ID D_INTEGER,
    DRM_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable drr_id d_integer;
begin
  res_is_error = 0;
  res_text = '';

  if (drm_id = -1) then
    drm_id = null;

  insert into docrecogrequest (drr_from_dp_id, drr_from_drm_id, drr_from_c_id, drr_from_un_id)
  values (:dp_id, :drm_id, :c_id, :un_id)
  returning drr_id
  into :drr_id;

  execute procedure hp_docrecogrequest_notify(drr_id, 'on_create');
  res_text = 'Запрос успешно создан';
  suspend;
end
