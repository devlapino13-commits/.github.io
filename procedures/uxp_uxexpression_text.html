CREATE PROCEDURE UXP$UXEXPRESSION_TEXT(
    UX_UUID D_UUID)
RETURNS (
    EXP_EXPRESSION_ALL D_STR_4096,
    EXP_EXPRESSION_0 D_STR_LONG,
    EXP_EXPRESSION_1 D_STR_LONG,
    EXP_EXPRESSION_2 D_STR_LONG,
    EXP_MESSAGE D_STR_LONG,
    EXP_KINDS D_SMALLINT)
AS
declare variable tmp_expression d_str_long;
declare variable tmp_message d_str_long;
declare variable exp_kind d_smallint;
declare variable exp_action d_smallint;
begin
  exp_expression_0 = '';
  exp_expression_1 = '';
  exp_expression_2 = '';
  exp_message = '';
  exp_expression_all = '';

  exp_kinds = -1;
  for select exp_expression, exp_message, exp_kind, exp_action
  from ux$expression
  where exp_ux_uuid = :ux_uuid
  into tmp_expression, tmp_message, exp_kind, exp_action
  do
  begin
    exp_kinds = exp_kind;

    if (exp_kind = 0) then
    begin
      exp_expression_0 = tmp_expression;
      exp_message = tmp_message;

      if (exp_expression_all <> '') then
        exp_expression_all = exp_expression_all || ascii_char(13) || ascii_char(10);

      exp_expression_all = exp_expression_all || 'Органичение: ' || tmp_expression || ascii_char(13) || ascii_char(10);
      exp_expression_all = exp_expression_all || 'Сообщение: "' || tmp_message || '"';
    end

    if (exp_kind = 1) then
    begin
      exp_expression_1 = tmp_expression;
      exp_expression_all = exp_expression_all || 'Вычисление: ' || tmp_expression;
    end

    if (exp_kind in (2, 3)) then
    begin
      exp_expression_2 = tmp_expression;

      if (exp_action = 2) then
        exp_expression_all = exp_expression_all || 'Видимый если: ' || tmp_expression;
      else
      if (exp_action = 3) then
        exp_expression_all = exp_expression_all || 'Доступен если: ' || tmp_expression;
      else
      if (exp_action = 4) then
        exp_expression_all = exp_expression_all || 'Переоткрыть если: ' || tmp_expression;
      else
        exp_expression_all = exp_expression_all || 'Выражение: ' || tmp_expression;
    end
  end

  if (row_count > 1) then
    exp_kinds = -2;

  suspend;
end
