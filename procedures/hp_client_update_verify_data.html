CREATE PROCEDURE HP_CLIENT_UPDATE_VERIFY_DATA(
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    C_PASS_NO D_STR_CODE,
    C_PASS_DATE D_DATE,
    C_DOC_TYPE D_INTEGER,
    C_CITIZEN D_INTEGER,
    C_PASS_DATE_EXPIRE D_DATE,
    C_INN D_STR_20,
    C_PINCODE D_STR_20,
    C_ALTERNAT_FIO D_STR_100,
    UN_ID D_ID,
    C_PASS_WHO D_STR_255,
    C_PASS_WHO_CODE D_STR_20,
    C_BIRTH_COUNTRY D_STR_100,
    C_BIRTH_CITY D_STR_100,
    C_NAME_ENG D_STR_100)
AS
begin
  -- Процедура обновления данных клиента из окна верификации паспорта
  if (f_get_db_id() = 81001 and c_citizen = 8) then
    update clients
      set
          c_name_fio = :c_name_fio,
          c_birth_date = :c_birth_date,
          c_pass_no = :c_pass_no,
          c_pass_date = :c_pass_date,
          c_doc_type = :c_doc_type,
          c_citizen = :c_citizen,
          c_pass_date_expire = :c_pass_date_expire,
          c_ext_inn = :c_inn,
          c_pincode = :c_pincode,
          c_alternat_fio = :c_alternat_fio,
          c_from_un_id = :un_id,
          c_pass_who = :c_pass_who,
          c_pass_who_code = :c_pass_who_code,
          c_birth_country = :c_birth_country,
          c_birth_city = :c_birth_city,
          c_name_eng = :c_name_eng
      where
          c_id = :c_id;
  else
    update clients
    set
        c_name_fio = :c_name_fio,
        c_birth_date = :c_birth_date,
        c_pass_no = :c_pass_no,
        c_pass_date = :c_pass_date,
        c_doc_type = :c_doc_type,
        c_citizen = :c_citizen,
        c_pass_date_expire = :c_pass_date_expire,
        c_inn = :c_inn,
        c_pincode = :c_pincode,
        c_alternat_fio = :c_alternat_fio,
        c_from_un_id = :un_id,
        c_pass_who = :c_pass_who,
        c_pass_who_code = :c_pass_who_code,
        c_birth_country = :c_birth_country,
        c_birth_city = :c_birth_city,
        c_name_eng = :c_name_eng
    where
        c_id = :c_id;
end
