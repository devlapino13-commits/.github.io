CREATE PROCEDURE ACT_CL_CLOS_CARDS(
    REP_DATE D_DATE,
    REP_REG_ID D_ID)
RETURNS (
    REGION_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    KC_FIO D_STR_100,
    CL_FIO D_STR_100,
    LAST_CARD_TYPE D_STR_100,
    LAST_CARD_NUM D_STR_100,
    K_KONTRACT_NO D_STR_20,
    DISB_AMOUNT CURR_TYPE,
    ACT_KRED_PROD D_STR_100,
    LAST_CARD_OFFICE D_STR_100,
    POINT D_INTEGER,
    CL_TEL D_STR_100)
AS
declare variable k_id d_id;
declare variable c_id d_id;
declare variable kc_id d_id;
declare variable ks_date_begin d_date;
declare variable kf_id d_id;
declare variable parent_reg_id_old d_id;
declare variable parent_reg_id d_id;
declare variable child_reg_id_old d_id;
declare variable pr_id_list d_str_255;
declare variable id_list d_id;
declare variable pr_id d_id;
begin
  parent_reg_id_old = 0;
  child_reg_id_old = 0;

  select '.' || list(pr.pr_id, '.') || '.' --Список всех pr_id содержащие .mbulak_cards. для того чтобы не читать каждый раз таблицу product
  from product pr
  where pr.pr_id > 0 and
        pr.pr_code containing '.mbulak_cards.'
  into pr_id_list;

  for select id_list,
             id_parent -- Данный селект для того чтобы уменьшить чтения в таблице region
      from region_list(:rep_reg_id, 1, 3, 1) t
      order by t.id_list
      into id_list, parent_reg_id
  do
  begin
    for select ks.ks_from_k_id,
               kc.kc_from_c_id,
               ks.ks_date_begin -- Выводим все активные кредиты на отчетную дату
        from kontstate ks
        inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
        where :rep_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_state = 1 and
              ks.ks_ol_is_delete = 0 and
              ks.ks_region_id = :id_list and
              kc.kc_amount_approved > 0
        into k_id, c_id, ks_date_begin
    do
    begin
      if (exists(select * -- Условие проводит проверку на наличие когда либо полученных карточных кредитов
                 from kontstate ks
                 inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
                 inner join kontract k on k.k_id = ks.ks_from_k_id
                 where kc.kc_from_c_id = :c_id and
                       ks.ks_state = 1 and
                       ks.ks_ol_is_delete = 0 and
                       :rep_date between ks.ks_date_begin and ks.ks_date_end and
                       kc.kc_amount_approved > 0 and
                       :pr_id_list containing '.' || k.k_from_pr_id || '.')) then
        continue;

      if (not exists(select * -- Условие проводит проверку на отсутствие дополнительного активного карточного кредита
                     from kontstate ks
                     inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
                     inner join kontract k on k.k_id = ks.ks_from_k_id
                     where kc.kc_from_c_id = :c_id and
                           kc.kc_amount_approved > 0 and
                           ks.ks_from_k_id <> :k_id and
                           ks.ks_state = 1 and
                           ks.ks_ol_is_delete = 0 and
                           :pr_id_list containing '.' || k.k_from_pr_id || '.' and
                           ks.ks_date_end < :ks_date_begin)) then
        continue;

      if (parent_reg_id <> parent_reg_id_old) then -- Данное условие проверяет на повторяемость региона активного кредита
      begin
        select reg.reg_name
        from region reg
        where reg.reg_unicode = :parent_reg_id
        into region_name;

        parent_reg_id_old = parent_reg_id;
      end

      if (id_list <> child_reg_id_old) then
      begin
         select reg.reg_name
         from region reg
         where reg.reg_unicode = :id_list
         into office_name;

         child_reg_id_old = id_list;
      end

      k_kontract_no = '';
      kf_id = 0;
      disb_amount = 0;
      pr_id = 0;
      select k.k_kontract_no,
             k.k_from_kf_id,
             k.k_amount_approved,
             k.k_from_pr_id
      from kontract k
      where k.k_id = :k_id
      into k_kontract_no, kf_id, disb_amount,pr_id;

      act_kred_prod = '';
      select pr.pr_name
      from product pr
      where pr.pr_id = :pr_id
      into act_kred_prod;

      cl_tel = f_cont_info('clients', :c_id, 'cell_main', 'ok');

      cl_fio = '';
      select c.c_name_fio
      from clients c
      where c.c_id = :c_id
      into :cl_fio;

      kc_fio = '';
      select c.c_name_fio
      from knaktlist kal
      inner join clients c on c.c_id = kal.kal_new_kf_officer
      where kal.kal_from_kf_id = :kf_id and
            :rep_date between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into kc_fio;

      point = f_point(:c_id, :rep_date);

      last_card_num = '';
      last_card_office = '';
      last_card_type = '';
      select k.k_kontract_no, -- Номер последнего закрытого карточного кредита
             reg.reg_name, -- Офис в котором выдали последний карточный кредит
             pr.pr_name
      from kontract k
      inner join kontstate ks on ks.ks_from_k_id = k.k_id
      inner join knclient kc on kc.kc_from_k_id = k.k_id
      inner join region reg on reg.reg_unicode = ks.ks_region_id
      inner join product pr on pr.pr_id = k.k_from_pr_id
      where kc.kc_from_c_id = :c_id and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            pr.pr_code containing '.mbulak_cards.' and
            ks.ks_date_begin < :rep_date
      order by ks.ks_date_begin desc
      rows 1
      into last_card_num, last_card_office,last_card_type;

      suspend;
    end
  end
end
