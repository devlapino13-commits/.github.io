CREATE PROCEDURE AAREP_SITE_INTERESTED_DELINQ(
    REP_DATE_END DATE)
RETURNS (
    CK_CHECKED_DATE DATE,
    CK_CHECKED_OPER STR_100,
    C_PHONE VARCHAR(20),
    STATUS_DATE DATE,
    STATUS VARCHAR(50),
    MANAGER_FIO STR_100,
    REG_NAME STR_100,
    K_KONTRACT_NO VARCHAR(50),
    MAX_DEL_DAY INTEGER,
    CUR_DEL_DAY INTEGER,
    KC_AMOUNT_APPROVED CURR_TYPE,
    C_NAME_FIO STR_100,
    STATUS_REG STR_100,
    PASS_NO_OPER VARCHAR(30),
    CELL_NO_OPER VARCHAR(20),
    OS_BALANCE CURR_TYPE)
AS
declare variable k_id id_type;
declare variable c_id id_type;
declare variable k_id_old id_type;
begin
  k_id_old = 0;

  for select k_id, c_id, ck_checked_date, ck_checked_oper, c_phone, status_date, status, manager_fio, reg_name,
             kc_amount_approved, status_reg, pass_no_oper, cell_no_oper
      from aarep_site_interested_cli('21.05.2015', :rep_date_end)
      where status_id = 1
      order by status_date, k_id
      into :k_id, :c_id, :ck_checked_date, :ck_checked_oper, :c_phone, :status_date, :status, :manager_fio, :reg_name,
           :kc_amount_approved, :status_reg, :pass_no_oper, :cell_no_oper
  do
  begin
    if (k_id <> k_id_old) then
    begin
      os_balance = 0;
      k_id_old = k_id;

      select kont.k_kontract_no
      from kontract kont
      where kont.k_id = :k_id
      into :k_kontract_no;

      max_del_day = null;
      cur_del_day = null;

      select max(iif(oks_date_end > cast(:rep_date_end as date), cast(:rep_date_end as date) - oks_delinq_date, oks_date_end + 1 - oks_delinq_date)) max_del_day,
             sum(iif(cast(:rep_date_end as date) between oks_date_begin and oks_date_end and oks_delinq_date < cast(:rep_date_end as date), cast(:rep_date_end as date) - oks_delinq_date, 0)) cur_del_day
      from oksummary oks
      where oks_close_state = 0 and
            oks_from_k_id = :k_id and
            oks_delinq_date <= :rep_date_end and
            oks_date_begin <= :rep_date_end
      into :max_del_day, :cur_del_day;

      select sum(oks.oks_balance)
      from oksummary oks
      where oks.oks_from_k_id = :k_id and
            :rep_date_end between oks.oks_date_begin and oks.oks_date_end
      into :os_balance;

      os_balance = coalesce(os_balance, 0);
      max_del_day = coalesce(max_del_day, 0);
      cur_del_day = coalesce(cur_del_day, 0);

      if (max_del_day < 0) then
        max_del_day = 0;
      if (cur_del_day < 0) then
        cur_del_day = 0;
    end

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into :c_name_fio;

    suspend;
  end
end
