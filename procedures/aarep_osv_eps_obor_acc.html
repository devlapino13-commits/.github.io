CREATE PROCEDURE AAREP_OSV_EPS_OBOR_ACC(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_DT_AB_ID ID_TYPE,
    REP_KT_AB_ID ID_TYPE,
    REP_FOLD_ACCOUNT INTEGER,
    REP_DT_AB_CODE D_STR_CODE = null,
    REP_KT_AB_CODE D_STR_CODE = null)
RETURNS (
    DT_ACC_ID ID_TYPE,
    KT_ACC_ID ID_TYPE,
    OB_SUMMA CURR_TYPE,
    OB_SUMMA_VAL CURR_TYPE,
    OB_SUMMA_TAX CURR_TYPE,
    OB_SUMMA_TAX_DT CURR_TYPE,
    OB_SUMMA_TAX_KT CURR_TYPE,
    DT_ACC_CODE D_STR_CODE,
    KT_ACC_CODE D_STR_CODE,
    DT_ACC_NAME D_STR_255,
    KT_ACC_NAME D_STR_255)
AS
declare variable acc_mask_dt d_str_code;
declare variable acc_mask_kt d_str_code;
declare variable n_start integer;
begin
  if (rep_dt_ab_code is null) then
    rep_dt_ab_code = '';
  if (rep_kt_ab_code is null) then
    rep_kt_ab_code = '';
  acc_mask_dt = '';
  acc_mask_kt = '';
  n_start = position('(', rep_dt_ab_code);
  if (n_start > 0) then
  begin
    acc_mask_dt = left(rep_dt_ab_code, 5);
    rep_dt_ab_code = substring(rep_dt_ab_code from n_start + 1);
    rep_dt_ab_code = left(rep_dt_ab_code, char_length(rep_dt_ab_code) - 1);
    n_start = char_length(rep_dt_ab_code);
    acc_mask_dt = acc_mask_dt || '_____' || rep_dt_ab_code || left('__________', 10 - n_start);
  end

  n_start = position('(', rep_kt_ab_code);
  if (n_start > 0) then
  begin
    acc_mask_kt = left(rep_kt_ab_code, 5);
    rep_kt_ab_code = substring(rep_kt_ab_code from n_start + 1);
    rep_kt_ab_code = left(rep_kt_ab_code, char_length(rep_kt_ab_code) - 1);
    n_start = char_length(rep_kt_ab_code);
    acc_mask_kt = acc_mask_kt || '_____' || rep_kt_ab_code || left('__________', 10 - n_start);
  end

  if (rep_dt_ab_id is null) then
  begin
    for select dt_acc_id, kt_acc_id, a_dt.acc_name, a_kt.acc_name, ob_summa, ob_summa_val, ob_summa_tax_dt,
               ob_summa_tax_kt
        from aarep_osv_eps_obor_tmp(:rep_date_begin, :rep_date_end, :rep_dt_ab_id, :rep_kt_ab_id, :rep_fold_account), ref_account a_dt, ref_account a_kt
        where dt_acc_id = a_dt.acc_id and
              kt_acc_id = a_kt.acc_id
        into :dt_acc_id, :kt_acc_id, :dt_acc_name, :kt_acc_name, :ob_summa, :ob_summa_val, :ob_summa_tax_dt,
             :ob_summa_tax_kt
    do
    begin
      dt_acc_code = f_fmt_acc(dt_acc_id);
      kt_acc_code = f_fmt_acc(kt_acc_id);
      ob_summa_tax = ob_summa_tax_dt;
      suspend;
    end
  end
  else
  begin
    for select dt_acc_id, kt_acc_id, a_dt.acc_name, a_kt.acc_name, ob_summa, ob_summa_val, ob_summa_tax_dt,
               ob_summa_tax_kt
        from aarep_osv_eps_obor_tmp(:rep_date_begin, :rep_date_end, :rep_dt_ab_id, :rep_kt_ab_id, :rep_fold_account), ref_account a_dt, ref_account a_kt
        where dt_acc_id = a_dt.acc_id and
              kt_acc_id = a_kt.acc_id and
              a_dt.acc_from_ab_id = :rep_dt_ab_id and
              a_kt.acc_from_ab_id = :rep_kt_ab_id and
              (:acc_mask_dt = '' or
              a_dt.acc_code like :acc_mask_dt) and
              (:acc_mask_kt = '' or
              a_kt.acc_code like :acc_mask_kt)
        into :dt_acc_id, :kt_acc_id, :dt_acc_name, :kt_acc_name, :ob_summa, :ob_summa_val, :ob_summa_tax_dt,
             :ob_summa_tax_kt
    do
    begin
      dt_acc_code = f_fmt_acc(dt_acc_id);
      kt_acc_code = f_fmt_acc(kt_acc_id);
      ob_summa_tax = ob_summa_tax_dt;
      suspend;
    end
  end
end
