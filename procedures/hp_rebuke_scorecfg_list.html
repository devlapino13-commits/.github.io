CREATE PROCEDURE HP_REBUKE_SCORECFG_LIST(
    IN_RBK_ID D_INTEGER)
RETURNS (
    RSC_ID D_INTEGER,
    RSC_FROM_RBK_ID D_INTEGER,
    RSC_REBUKE_TYPE D_INTEGER,
    RSC_SCORE_MIN D_INTEGER,
    RSC_SCORE_MAX D_INTEGER,
    RES_CHECKED D_BOOL,
    RES_REBUKE_NAME D_STR_100)
AS
begin
  rsc_from_rbk_id = in_rbk_id;

  for select id, text
  from hp_rebuke_type_lookup
  order by text
  into :rsc_rebuke_type, :res_rebuke_name
  do
  begin
    rsc_id = null;
    rsc_score_min = null;
    rsc_score_max = null;
    res_checked = 0;

    select rsc_id, rsc_score_min, rsc_score_max, 1
    from rebuke_scorecfg
    where rsc_from_rbk_id = :rsc_from_rbk_id and
          rsc_rebuke_type = :rsc_rebuke_type
    into :rsc_id, :rsc_score_min, :rsc_score_max, res_checked;

    suspend;
  end
end
