CREATE PROCEDURE PAYSYS_AKT_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PSM_AKT_NO DECIMAL(18,0),
    REG_NAME VARCHAR(50),
    C_NAME_FIO_SRC VARCHAR(100),
    C_NAME_FIO_DST VARCHAR(100),
    JL_NAME_SRC VARCHAR(100),
    JL_NAME_DST VARCHAR(100),
    PSM_DATE DATE,
    PSL_COUNT INTEGER,
    OBJ_NAME_SRC VARCHAR(100),
    OBJ_KIND_SRC INTEGER,
    OBJ_KIND_TXT_SRC VARCHAR(100),
    OBJ_NAME_DST VARCHAR(100),
    OBJ_KIND_TXT_DST VARCHAR(100),
    OBJ_KIND_DST INTEGER,
    PSM_ID DECIMAL(18,0),
    PSM_CAN_EDIT INTEGER,
    PSM_IS_DELETE INTEGER,
    PSM_AKT_NO_MIN VARCHAR(20),
    PSM_AKT_NO_MAX VARCHAR(20))
AS
declare variable c_id_src decimal(18,0);
declare variable c_id_dst decimal(18,0);
declare variable obj_kind integer;
declare variable obj_id decimal(18,0);
begin
  for select psm_id, reg_name, psm_date, psm_from_c_source, psm_from_c_destin, psm_is_delete
      from paysys_main, region, region_list(:rep_region_id, 1, 3, 1)
      where psm_date between :rep_date_begin and :rep_date_end and
            psm_from_reg_id = reg_unicode and
            reg_unicode = id_list
      order by psm_date, psm_id
      into :psm_id, reg_name, psm_date, c_id_src, c_id_dst, psm_is_delete
  do
  begin
    psm_can_edit = 1;
    if (exists(select *
               from paysys_list
               where psl_from_psm_id = :psm_id and
                     psl_object_kind = 3)) then
    begin
-- В акте участвует кредиты и клиенты, запрешаем редактирование или аннулирование
      psm_can_edit = 0;
    end
    psm_akt_no_min = '';
    psm_akt_no_max = '';
    select min(psi_code), max(psi_code)
    from paysys_item, paysys_list, paysys_project
    where psi_id = psl_from_psi_id and
          psi_from_psp_id = psp_id and
          psp_is_card = 0 and
          psl_from_psm_id = :psm_id
    into :psm_akt_no_min, psm_akt_no_max;
    if (psm_akt_no_min is null) then
      psm_akt_no_min = '';
    if (psm_akt_no_max is null) then
      psm_akt_no_max = '';

    select psm_akt_no, reg_name, c_name_fio_src, c_name_fio_dst, jl_name_src, jl_name_dst, psm_date, psl_count,
           obj_name_src, obj_kind_src, obj_kind_txt_src, obj_name_dst, obj_kind_txt_dst, obj_kind_dst
    from paysys_print_akt_common(:psm_id)
    into :psm_akt_no, :reg_name, :c_name_fio_src, :c_name_fio_dst, :jl_name_src, :jl_name_dst, :psm_date, :psl_count,
         :obj_name_src, :obj_kind_src, :obj_kind_txt_src, :obj_name_dst, :obj_kind_txt_dst, :obj_kind_dst;
    suspend;
  end
end
