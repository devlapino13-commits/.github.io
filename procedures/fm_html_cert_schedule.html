CREATE PROCEDURE FM_HTML_CERT_SCHEDULE(
    UN_ID DECIMAL(18,0),
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable ma_id integer;
declare variable lo_id decimal(18,0);
declare variable reg_id integer;
declare variable is_readonly integer;
declare variable res_time timestamp;
declare variable gr_name varchar(100);
declare variable res_status varchar(100);
declare variable res_can_reserv integer;
declare variable res_obj_id integer;
declare variable res_link varchar(1024);
declare variable c_id decimal(18,0);
declare variable reg_name varchar(100);
declare variable curr_kf_name varchar(100);
declare variable view_mode varchar(100);
declare variable ma_plan_disb_time timestamp;
declare variable db_id integer;
declare variable k_id decimal(18,0);
declare variable kk_approved integer;
declare variable lang d_str_lang;
begin

  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);

  res_text = (select res_style from fm_html_styles) || '<body>';

  is_readonly = 1;

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  if (obj_kind = 'miniapp') then
  begin
    ma_id = obj_id;
    select k_id
    from kontract
    where k_from_ma_app_id = :obj_id
    order by k_disb_date desc
    rows 1
    into :k_id;
  end
  else
  begin
    select k_from_ma_app_id
    from kontract
    where k_id = :obj_id
    into ma_id;

    k_id = obj_id;
  end
  kk_approved = 1;

  if (db_id = 81001) then
  begin
    if (k_id is not null) then
    begin
      -- Если нет одобрения КК, пасписание сертификаций открываем в режиме чтения
      if (not(exists(select *
                     from hp_actionlist_kontract_get(:k_id)
                     where acl_status = 2))) then
        kk_approved = 0;
    end
    else
      kk_approved = 0;
  end

  ma_plan_disb_time = null;
  view_mode = iif(db_id <> 94401, 'Режим просмотра', 'Baxış rejimi');
  if (coalesce(ma_id, 0) <> 0) then
  begin
    select kal_new_kf_officer, kal_new_reg_id, reg_code || ' ' || reg_name, kal_new_kf_name, ma_plan_disb_time
    from miniapp, knaktlist, region
    where ma_id = :ma_id and
          ma_date_end >= '31.12.2999' and
          ma_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          kal_date <= :rep_date and
          kal_date_end >= :rep_date and
          kal_new_reg_id = reg_unicode
    into :lo_id, reg_id, reg_name, curr_kf_name, ma_plan_disb_time;

    if (lo_id is not null) then
    begin
      if (lo_id = c_id) then
      begin
        is_readonly = 0;
        view_mode = f_format('%1: <b>%2</b>', iif(db_id <> 94401, 'Группа', 'Qrup'), curr_kf_name);
      end
    end
  end
  else
  begin
    select ps_from_reg_id, reg_code || ' ' || reg_name
    from hr_pstatus, region
    where ps_from_c_id = :c_id and
          ps_status = 1 and
          ps_date_begin <= :rep_date and
          ps_date_end >= :rep_date and
          ps_is_working = 1 and
          ps_from_reg_id = reg_unicode
    into :reg_id, :reg_name;
  end

  if (is_readonly = 1) then
    ma_plan_disb_time = null;

  if (reg_id is null) then
  begin
    res_text = BLOB_APPEND(res_text, f_format('%1,<br> %2.', iif(db_id <> 94401, 'Офис пользователя не определен', 'İstifadəçinin filialı təyin olunmadı'),
                                                   iif(db_id <> 94401, 'нет действующего приказа', 'aktiv əmr mövcüd deyil')));
    suspend;
    exit;
  end


  res_text = BLOB_APPEND(res_text, f_format('<h1 class="green_text">%1</h1>', iif(db_id <> 94401, 'Расписание сертификаций', 'Sertifikasiyanın cədvəli')));
  res_text = BLOB_APPEND(res_text, f_format('%1: <b>%2</b><br>', iif(db_id <> 94401, 'Офис', 'Filial'), reg_name));
  res_text = BLOB_APPEND(res_text, f_format('%1: <b>', iif(db_id <> 94401, 'Дата', 'Tarix')) || right('00' || extract(day from rep_date), 2) || '.' || right('00' || extract(month from rep_date), 2) || '.' || extract(year from rep_date) || '</b><br>');
  res_text = BLOB_APPEND(res_text, view_mode || '<br><br>');

  res_link = '';
  if (ma_plan_disb_time is not null and
      is_readonly = 0) then
  begin
    res_text = BLOB_APPEND(res_text, f_format('%1: <b>', f_lang(lang, 'Время сертификации')) || right('0' || extract(hour from ma_plan_disb_time), 2) || ':' || right('0' || extract(minute from ma_plan_disb_time), 2) || '</b>');
    res_link = f_uri_sql(f_lang(lang, 'Отменить сертификацию'), 'set_cert_time',
      'miniapp', :ma_id, 'none',
      f_lang(lang, 'Хотите отменить назначенное время сертификации?'), '', 4,
      355, '', '', 'dialog', 'null');

    res_text = BLOB_APPEND(res_text, '<br>' || res_link || '<br><br>');
  end

  if (kk_approved = 0) then
    is_readonly = 1;

  if (is_readonly = 0) then
  begin
    if (res_link = '') then
      res_text = BLOB_APPEND(res_text, f_format('%1.<br>', iif(db_id <> 94401, 'Если хотите назначить время сертификации - выберите необходимое время в таблице нажатием на ссылку',
                                                                     'Sertifikasiya üçün yeni vaxt təyin etmək istəyirsinizsə cədvəldə göstərilən vaxtı seçmək məqsədə uyğundur')));
    else
      res_text = BLOB_APPEND(res_text, f_format('%1.<br>', iif(db_id <> 94401, 'Если хотите назначить другое время сертификации - выберите необходимое время в таблице нажатием на ссылку',
                                                                     'Əgər sertifikasiya üçün təyin olunan vaxtı dəyişmək istəyirsiniz, cədvəldə qeyd olunan vaxtı seçmək məqsədə uyğundur')));
  end
  else
  begin
-- текст для тех, кто не получил добро КК
    if (kk_approved = 0) then
    begin
      res_text = BLOB_APPEND(res_text, f_format('%1.<br>', iif(db_id <> 94401, 'Пока нет положительного решения КК, нельзя устанавливать время сертификации',
                                                                     'Пока нет положительного решения КК, нельзя устанавливать время сертификации')));
    end
  end

  res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="20%">
       <col width="40%">
       <col width="40%">
       <tr>' || f_format('<th>%1</th><th>%2</th><th>%3</th>', iif(db_id <> 94401, 'Время', 'Vaxt'),
                                                              iif(db_id <> 94401, 'Группа', 'Qrup'),
                                                              iif(db_id <> 94401, 'Статус', 'Status')) || '</tr>');


  for select res_time, res_text, res_status, res_can_reserv, res_obj_id
      from hp_certificate_schedule(:un_id, :reg_id, :rep_date, :is_readonly)
      into res_time, gr_name, res_status, res_can_reserv, res_obj_id
  do
  begin
    res_link = '';
    if (res_time is not null) then
      res_link = right('0' || extract(hour from :res_time), 2) || ':' || right('0' || extract(minute from :res_time), 2);

    if (res_can_reserv = 1) then
      res_link = f_uri_sql(res_link, 'set_cert_time', 'miniapp', ma_id,
        'none', f_lang(lang, 'Хотите назначить время сертификации?'), '', 4,
        355, '', '', 'dialog', :res_time);

    if (res_obj_id = ma_id and
        ma_id <> 0) then
    begin
      res_text = BLOB_APPEND(res_text, '<tr><td align="center" class="filled">' || res_link || '</td>' || '<td align="center" class="filled">' || gr_name || '</td>' || '<td align="center" class="filled">' || res_status || '</td></tr>');
    end
    else
    begin
      res_text = BLOB_APPEND(res_text, '<tr><td align="center">' || res_link || '</td>' || '<td align="center">' || gr_name || '</td>' || '<td align="center">' || res_status || '</td></tr>');
    end
  end
  res_text = BLOB_APPEND(res_text, '</table></body>');
  suspend;
end
