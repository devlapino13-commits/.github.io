CREATE PROCEDURE HP_CARD_STATEMENT_VIEW(
    ID D_STR_20,
    KIND D_SMALLINT = 0,
    DTBEG D_DATE = null,
    DTEND D_DATE = null)
RETURNS (
    OL_ID D_ID,
    DT_OPER D_DATETIME,
    K_ID D_ID,
    AMOUNT_OPER D_CURRENCY,
    AMOUNT_ADD D_CURRENCY,
    AMOUNT_DEL D_CURRENCY,
    DETAIL D_STR_255)
AS
declare variable card_id d_integer;
declare variable acc d_str_10;
declare variable status d_integer;
declare variable p d_integer;
declare variable specific_loan boolean;
begin
  dtbeg = coalesce(dtbeg, '01.01.1900');
  dtend = coalesce(dtend, '31.12.9999');
  status = f_enum_id('cardlist_status', 'transfer_to_client');

  p = position('-', id);
  specific_loan = false;
  if (p > 0) then
  begin
    if (p = 1) then
    begin
      id = -cast(id as d_id);
      specific_loan = true;
    end else
      id = (select k_id from kontract where k_kontract_no = :id);
    kind = 2;
  end

  -- № карты
  if (kind = 0) then
    card_id = (
      select
        cal_card_id
      from cardlist where cal_card_id = :id
        /*and cal_status = :status*/);
  -- № картсчета
  if (kind = 1) then
    card_id = (
      select
        cal_card_id
      from cardlist where cal_from_cca_id = :id
        /*and cal_status = :status*/);
  -- № контракта
  if (kind = 2) then
    card_id = (
      select
        cal_card_id
      from kontract
        inner join cardlistloan
          on clo_id = (
            select first 1 clo_id
            from cardlistloan
            where clo_from_k_id = k_id
            order by clo_date desc)
        inner join cardlist
          on cal_card_id = clo_card_id
            --and cal_status = :status
      where k_id = :id);

  if (card_id > 0) then
  begin
    --k_id = (select cll_from_k_id from cardlistlnk where cll_card_id = :card_id);

    acc = '20001/1';
    for
      with cards as
      (
        select distinct clo2.clo_card_id as card_id
        from cardlistloan as clo1
          inner join cardlistloan as clo2
            on clo2.clo_from_k_id = clo1.clo_from_k_id
        where clo1.clo_card_id = :card_id
      )
      select
        k_id,
        ol_id,
        ol_datetime,
        ob_summa,
        iif(ob_short_acc_kt = :acc, ob_summa, 0),
        iif(ob_short_acc_dt = :acc, ob_summa, 0),
        coalesce(ob_detail, ol_memo)
      from operbook
        inner join (
          select distinct ol_id as olo_id, ok_from_k_id as k_id
            from cardlistloan
              inner join operkred
                on ok_from_k_id = clo_from_k_id
              inner join operlist
                on ol_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = ok_from_ol_id)
            where clo_card_id = :card_id
          union
          select distinct ol_id as olo_id, oko_from_k_id as k_id
            from cardlistloan
              inner join operkroff
                on oko_from_k_id = clo_from_k_id
              inner join operlist
                on ol_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = oko_from_ol_id)
            where clo_card_id = :card_id
          union
          select ol_id as olo_id, po_card_k_id as k_id
            from pvnopers
              inner join cards
                on card_id = po_card_id
              inner join operlist
                on ol_parent_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = po_from_ol_id )
                  or ol_id = (select coalesce(ol_parent_id, ol_id) from operlist where ol_id = po_from_ol_id )
            where po_is_mb_card = 1
              and po_from_ol_id > 0
              and po_is_delete = 0
          union
          select ol_id as olo_id, ied_from_k_id as k_id
            from ipc_exd_detail
              inner join cards
                on card_id = ied_from_card_id
              inner join operlist
                on ied_from_ol_id in (ol_id, ol_parent_id)
            where ied_from_ol_id > 0
          union
          select
            ol_id as olo_id, cac_from_k_id as k_id
          from cardaktclient
            inner join refdata
              on rd_id = cac_from_rd_id
                and rd_code in ('ReceivCard_LossTheft', 'ReceivCard_Spoilage')
            inner join cards
              on card_id = cac_card_id
            inner join operlist
              on ol_id = cac_from_ol_id_com
          where cac_status >= 4
          ) as a
          on a.olo_id = ob_from_ol_id
        inner join operlist
          on ol_id = ob_from_ol_id
            and ol_date between :dtbeg and :dtend
            and is_delete = 0
      where ob_short_acc_dt = :acc or ob_short_acc_kt = :acc
      order by ol_datetime, ol_id
    into
      :k_id,
      :ol_id,
      :dt_oper,
      :amount_oper,
      :amount_add,
      :amount_del,
      :detail
    do
    begin
      if (card_id in (2028, 2030, 2043, 2015)) then
        amount_del = amount_add;
      if (not specific_loan or k_id = id) then
        suspend;
    end

    if (card_id in (618870, 647842, 669459)) then
    for
      select
        ol_id,
        ol_datetime,
        ob_summa,
        iif(ob_short_acc_kt = :acc, ob_summa, 0),
        iif(ob_short_acc_dt = :acc, ob_summa, 0),
        coalesce(ob_detail, ol_memo)
      from operbook
        inner join operlist
          on ol_id = ob_from_ol_id
      where ob_from_ol_id = iif(:card_id = 618870, 1090100070021223,
        iif(:card_id = 647842, 1090100069992968, 1090100070021033))
        and :acc in (ob_short_acc_dt, ob_short_acc_kt)
    into
      :ol_id,
      :dt_oper,
      :amount_oper,
      :amount_add,
      :amount_del,
      :detail
    do
    suspend;
  end
end
