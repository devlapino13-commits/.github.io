CREATE PROCEDURE CM_RU_0021_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
begin
    if (exists(select * from miniapp, knaktlist
               where ma_id=:ma_id
               and ma_from_kf_id = kal_from_kf_id
               and ma_date_end='31.12.9999'
               and knaktlist.is_delete = 0
               and current_date between kal_date and kal_date_end
               and kal_new_kf_officer <= 1000000000003
               and iif(kal_new_kf_officer = 1000000000002
                    and ma_from_un_id in(1010100000148, 1010100000136, 1010100000001), 1, 0) = 0 --Добавил Музаффар 13.09.16
               )) then
    begin
      res_error_is_critical=1;
      res_error_text='Данная папка находится у виртуального сотрудника, выдача запрещена!';
      suspend;
    end
end
