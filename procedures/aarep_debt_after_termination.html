CREATE PROCEDURE AAREP_DEBT_AFTER_TERMINATION(
    REG_ID D_ID,
    K_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    OK_FROM_K_ID D_ID,
    K_KONTRACT_NO D_STR_100,
    TERM_SH_ID D_SMALLINT,
    TERM_SH_DATE D_DATE,
    TERM_SH_OS D_CURRENCY,
    PAY_FINE D_CURRENCY,
    CURR_SH_OS D_CURRENCY,
    TERM_SH_PERC D_CURRENCY,
    PAY_PERC D_CURRENCY,
    CURR_SH_PERC D_CURRENCY,
    PAY_OS D_CURRENCY,
    PAY_KOMMIS D_CURRENCY,
    PRV_PAY_OS D_CURRENCY,
    PRV_PAY_PERC D_CURRENCY,
    PRV_SH_OS D_CURRENCY,
    PRV_SH_PERC D_CURRENCY)
AS
declare variable rep_date_begin d_date;
declare variable prv_date_begin d_date;
declare variable prv_date_end d_date;
begin
  /* Процедура для получения долга на указанный день из контрактов по
  которым было расторжения договора в предыдущем месяце */

  prv_date_end = rep_date_end - extract(day from rep_date_end);
  rep_date_begin = prv_date_end + 1;
  prv_date_begin = dateadd(month, -1, rep_date_begin);

  for select ok_from_k_id, k_kontract_no, ok_sh_os, ok_sh_perc, ok_sh_id, ok_sh_date
      from operkred ok
      inner join kontract k on ok_from_k_id = k_id
            --      and k.k_last_reg_id = coalesce(:reg_id, k_last_reg_id) --пока необходимости нет
      inner join operlist ol on k.k_stopped_perc_ol_id = ol_id and
            ol_id = ok_from_ol_id and
            ol_date = ok_ol_date and
            ol.is_delete = 0 and
            ol_date between :prv_date_begin and :prv_date_end
      where --
            --ok_from_k_id = coalesce(:k_id, ok_from_k_id) and --пока необходимости нет
            ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_type = 3 and
            ok_sh_os > 0
      into ok_from_k_id, k_kontract_no, --
           term_sh_os, --Долг по ОС при расторжении договора
           term_sh_perc, --Долг по %% при расторжении договора
           term_sh_id, --план.номер при расторжении
           term_sh_date --Дата расторжения
  do
  begin
    prv_pay_os = null;
    prv_pay_perc = null;
    select -sum(ok_os), -sum(ok_perc)
    from operkred ok
    where ok_from_k_id = :ok_from_k_id and
          ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_ol_date between :prv_date_begin and :prv_date_end and
          ok_sh_id = :term_sh_id and
          ok_sh_date = :term_sh_date
    into prv_pay_os, prv_pay_perc;
    prv_pay_os = coalesce(prv_pay_os, 0); --Оплата по ОС за пред.месяц
    prv_pay_perc = coalesce(prv_pay_perc, 0); --Оплата по %% за пред.месяц
    prv_sh_os = term_sh_os - prv_pay_os; --Долг на начало отчета по ОС
    prv_sh_perc = term_sh_perc - prv_pay_perc; --Долг на начало отчета по ОС

    pay_os = null;
    pay_perc = null;
    pay_kommis = null;
    pay_fine = null;
    select -sum(ok_os), -sum(ok_perc), sum(ok_kommis), sum(ok_fine)
    from operkred ok
    where ok_from_k_id = :ok_from_k_id and
          ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_ol_date between :rep_date_begin and :rep_date_end and
          ok_sh_id = :term_sh_id and
          ok_sh_date = :term_sh_date
    into pay_os, pay_perc, pay_kommis, pay_fine;
    pay_os = coalesce(pay_os, 0); --Оплата по ОС в текущем периоде
    pay_perc = coalesce(pay_perc, 0); --Оплата по %% в текущем периоде
    pay_kommis = coalesce(pay_kommis, 0); --Оплата комиссии в текущем периоде
    pay_fine = coalesce(pay_fine, 0); --Оплаченные пении в тек.периоде

    curr_sh_os = prv_sh_os - pay_os; --Долг на конец отчетного периода по ОС
    curr_sh_perc = prv_sh_perc - pay_perc; --Долг на конец отчетного периода по %%

    suspend;
  end
end
