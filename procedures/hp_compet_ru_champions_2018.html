CREATE PROCEDURE HP_COMPET_RU_CHAMPIONS_2018(
    CPP_ID INTEGER,
    UN_ID DECIMAL(18,0))

    RES_TEXT VARCHAR(255))
AS
declare variable res_date_begin date;
declare variable res_date_end date;
declare variable stage integer;
declare variable rd_id integer;
declare variable res_period_id integer;
declare variable res_team_id integer;
declare variable res_lo_id decimal(18,0);
declare variable ad_active_client integer;
declare variable ad_active_balance decimal(18,4) = 0;
declare variable ad_diff_client integer;
declare variable ad_delinq_balance_30 decimal(18,4) = 0;
declare variable ad_delinq_balance_diff_30 decimal(18,4) = 0;
declare variable points_1 decimal(18,4) = 0;
declare variable points_2 decimal(18,4) = 0;
declare variable points_3 decimal(18,4) = 0;
declare variable points_all decimal(18,4) = 0;
declare variable res_team_id_prior integer;
declare variable cpr_id integer;
declare variable cpr_group_name varchar(255);
declare variable cpd_comment d_str_4096;
declare variable lo_panelty_count integer;
declare variable avg_points decimal(18,4) = 0;
declare variable total_points decimal(18,4) = 0;
declare variable add_points decimal(18,4) = 0;
declare variable max_points decimal(18,4) = 0;
declare variable group_name varchar(255);
declare variable cpp_orderno integer;
declare variable kind integer;
declare variable cpr_final_points decimal(18,4) = 0;
declare variable stage_cpp_id integer;
declare variable ad_disb_amount decimal(18,4) = 0;
declare variable is_otpusk varchar(20);
declare variable par1 decimal(18,4) = 0;
declare variable avg_amount decimal(18,4) = 0;
declare variable sum_ad_diff_client integer;
declare variable kol_svz integer;
declare variable avg_ad_diff_client curr_type;
declare variable avg_rost_portfel curr_type;
declare variable points_4 curr_type;
declare variable kol_komand integer;
declare variable k integer;
declare variable ad_diff_balance curr_type;
declare variable otpusk_days integer;
declare variable n integer;
declare variable dd_tmp date;
declare variable ad_disb_amount_ipf curr_type;
declare variable avg_amount_ipf curr_type;
declare variable points_5 curr_type;
declare variable kol_kred_closed integer;
declare variable in_period_disb integer;
declare variable coeff_uderj curr_type;
declare variable points_6 curr_type;
declare variable group_name_prior varchar(255);
declare variable ad_disb_loan integer;
declare variable prinyat_portfel curr_type;
declare variable svz_portfel_begin curr_type;
declare variable svz_portfel_end curr_type;
declare variable par30 curr_type;
begin
-- Расчет итогов олимпиады КР 2017
-- Stage:
-- 0 - период (неделя)
-- 1 - Промежуточные итоги
-- 2 - Итоги
-- 3 - Финальные результаты

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end, cpp_orderno
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end, cpp_orderno;

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end

--  if (cpp_orderno=0 and res_date_begin=cast('01.10.2017' as date)) then
--  begin
--    res_date_begin='02.10.2017';
--    res_date_end='02.10.2017';

  if (cpp_orderno = 0 and
      res_date_begin = cast('14.01.2018' as date)) then
  begin
    res_date_begin = '15.01.2018';
    res_date_end = '15.01.2018';

    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

-- Формируем в памяти данные по КР
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'olimpic')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

-- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    for
    /*
    select res_field_1, sum(ad_active_balance)
    from hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group',
                             31, 999999,
                             0, 999999,
                             8, 999999,
                             31, 999999,
                             181, 999999,
                             1065, 0,
                             'standart')
    group by 1
    having sum(ad_active_balance)>0.0  -- Берем тех у кого имеется портфель на конец периода
    into :res_team_id, :ad_active_balance
    */
        select res_field_1 --, ad_active_balance
        from hp_analitica_report('kont_lo_teamname_id', 'tree_group', 31, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
        where ad_active_balance > 0 and
              exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        into :res_team_id
    do
    begin
      if (exists(select *
                 from refdata
                 where rd_id = :res_team_id and
                       rd_is_active = 1)) then
      begin
        insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                  cpr_from_un_id, cpr_comment)
        values (:cpp_id, '', -1, :res_team_id, 0, :un_id, '')
        returning cpr_id
        into :cpr_id;
      end
    end
    suspend;
    exit;
  end

-- Ищем итоги тура
  stage_cpp_id = null;
  select cpp_id
  from competperiod
  where cpp_from_rd_id = :rd_id and
        cpp_stage = 2 and
        cpp_status = 1 and
        cpp_date_end <= :res_date_end
  order by cpp_date_begin desc, cpp_id desc
  rows 1
  into :stage_cpp_id;

  if (stage = 0) then -- Если уровень недели
  begin
-- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

-- Формируем в памяти данные по КР
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'olimpic')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

-- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    kol_svz = 0;
    cpr_id = null;
    for select res_field_1, res_field_2, sum(ad_active_client), sum(ad_active_balance), sum(ad_diff_client),
               sum(ad_delinq_balance_end3), sum(ad_delinq_balance_diff3), sum(ad_disb_amount), sum(ad_disb_loan)
        from hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group', 31, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
        where /* exists(select * from hr_pstatus
                  where ps_from_c_id = res_field_2
                  and ps_date_end >= :res_date_end
                  and ps_date_begin <= :res_date_end
                  and ps_status=1
                  and ps_is_working=1)
    and*/ exists(select *
                 from refdata
                 where rd_main_code = 'StaffTeamList' and
                       rd_id = res_field_1 and
                       rd_tag = :rd_id)
        group by 1, 2
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :res_team_id, -- ID команды
             :res_lo_id, -- ID КСника
             :ad_active_client, -- Клиенты на конец периода
             :ad_active_balance, -- Баланс на конец периода
             :ad_diff_client, -- Прирост клиентов           -1-
             :ad_delinq_balance_30, -- Баланс просрочек
             :ad_delinq_balance_diff_30, -- 1.    Изменение баланса просрочек более 30 дней СВЗ (рублей) по итогам отчетной недели
             :ad_disb_amount, --  2.    Сумма всех выдач за неделю (рублей)  по итогам отчетной недели
             :ad_disb_loan -- Кол-во выдач
    do
    begin
-- Исключаем если нет действующего
      is_otpusk = '';
      if (res_team_id <> 0) then
      begin
-- Исключаем если в отпуске.
-- Проверяем статус отпуска на каждый день недели:
-- если все 5 дней был в отпуске, исключаем из списка
        otpusk_days = 0;
        n = 0;
        while (n <= 4) do
        begin
          dd_tmp = dateadd(day, n, res_date_begin);
          if (exists(select *
                     from hr_voctravel, hr_orders, sa_worktype
                     where hrv_from_ho_id = ho_id and
                           ho_order_type = sawt_id and
                           hrv_from_c_id = :res_lo_id and
                           sawt_base_type in (4, 5) -- отпуск, декретный отпуск
                           and
                           sawt_id <> 20 and
                           sawt_hours_real >= 0 and
                           hrv_date_begin <= :dd_tmp and
                           hrv_date_end_fact >= :dd_tmp and
                           hrv_status = 1)) then
            otpusk_days = otpusk_days + 1;
          n = n + 1;
        end

        if (otpusk_days = 5) then
        begin
          if (not(exists(select *
                         from hr_voctravel, hr_orders
                         where hrv_from_ho_id = ho_id and
                               ho_order_type = 20 and
                               hrv_from_c_id = :res_lo_id and
                               hrv_date_begin <= :res_date_end and
                               hrv_date_end_fact >= :res_date_begin and
                               hrv_status = 1))) then
          begin
--            res_team_id=0;
            is_otpusk = 'Отпуск.';
          end
        end
      end

      points_1 = 0.0; --1.    Изменение баланса просрочек более 30 дней СВЗ (рублей) по итогам отчетной недели
      points_2 = 0.0;
      points_3 = 0.0;

      /*
      if (ad_active_balance<>0)
      then par1=round(ad_delinq_balance_1*100.00/ad_active_balance,2);
      else par1=0;
--    Сумма выдачи по исламским продуктам                          -4-
      ad_disb_amount_ipf = null;
      select sum(k_amount_approved)
      from kontstate, kontract, product, knaktlist
      where ks_ol_is_delete = 0
      and ks_state = 1
      and ks_date_begin between :res_date_begin and :res_date_end
--      and ks_kc_id = :res_lo_id
      and ks_from_k_id = k_id
      and k_from_pr_id = pr_id
      and pr_code containing '.IPF.'
      and k_from_kf_id = kal_from_kf_id
      and knaktlist.is_delete = 0
      and :res_date_end between kal_date and kal_date_end
      and kal_new_kf_officer = :res_lo_id
      into ad_disb_amount_ipf;

      if (ad_disb_amount_ipf is null) then ad_disb_amount_ipf = 0;
      ad_disb_amount = ad_disb_amount - ad_disb_amount_ipf;     
      */

      par30 = cast(ad_delinq_balance_30 as double precision) / ad_active_balance * 100;
      cpd_comment = '';
--      cpd_comment=cpd_comment||'Актив.кл: <b>'||round(ad_active_client)||'</b><br>';
      cpd_comment = cpd_comment || 'PAR30: <b>' || round(par30, 4) || '%</b><br>';
      cpd_comment = cpd_comment || 'Изменение PAR30: <b>' || round(ad_delinq_balance_diff_30, 2) || ' р.</b><br>';
      cpd_comment = cpd_comment || 'Сумма выдач СВЗ за период: <b>' || round(ad_disb_amount, 0) || ' р.</b><br>';
--      cpd_comment=cpd_comment||'Сумма выдач СВЗ за период по ИПФ: <b>'||round(ad_disb_amount_ipf,0)||'</b><br>';
--      cpd_comment=cpd_comment||'Баланс просрочки PAR>=1: <b>'||round(ad_delinq_balance_1,2)||'%</b><br>';
      cpd_comment = cpd_comment || 'Кол-во выдач: <b>' || ad_disb_loan || '</b><br>';
      cpd_comment = cpd_comment || 'Прирост кл: <b>' || round(ad_diff_client) || '</b><br>';

      /*
--    Вычисляем баллы по приросту
      points_1 = case
                   when ad_diff_client >= 1 then 2
                   when ad_diff_client >= 0 then 1
                   when ad_diff_client >= -5 then -2
                   else -4
                 end;
      */

--    1.    Изменение баланса просрочек более 30 дней СВЗ (рублей) по итогам отчетной недели
      if (ad_delinq_balance_diff_30 < -20000.0 or par30 < 2.0) then
        points_1 = 5;
      else
      if (ad_delinq_balance_diff_30 < -10000.0) then
        points_1 = 3;
      else
      if (ad_delinq_balance_diff_30 <= 0) then
        points_1 = 1;
      else
      if (ad_delinq_balance_diff_30 <= 10000) then
        points_1 = -2;
      else
        points_1 = -5;

--    2.    Сумма всех выдач за неделю (рублей)  по итогам отчетной недели
      points_2 = case
                   when ad_disb_amount > 500000.0 then 7
                   when ad_disb_amount > 300000.0 then 5
                   when ad_disb_amount > 200000.0 then 0
                   else -5
                 end;

--  Е. Удержание повторных клиентов в процентном соотношении (доля клиентов, получивших кредит повторно из чискла погасивших кредит в отчетном периоде)
      kol_kred_closed = 0;
      in_period_disb = 0;
      coeff_uderj = 0;
      points_4 = 0;
      if (datediff(day, cast('14.01.2018' as date), res_date_end) = (datediff(day, cast('14.01.2018' as date), res_date_end) / 28) * 28) then
      begin
        /*
        with tabl1 as (
            select kc_from_c_id,
                   (select 1
                   from kontstate, knclient
                   where ks_ol_is_delete = 0
                   and ks_state = 1
                   and ks_date_begin between :res_date_end - 27 and :res_date_end
                   and ks_from_k_id = kc_from_k_id
                   and kc_amount_approved > 0
                   and kc_from_c_id = kc1.kc_from_c_id
                   and kc_when_prior_loan_closed = ks1.ks_date_begin
                   rows 1
                   ) as in_period_disb
            from kontstate ks1, knclient kc1
            where ks_ol_is_delete = 0
            and ks_state = 2
            and ks_date_begin between :res_date_end - 27 and :res_date_end
            and ks_kc_id = :res_lo_id
            and ks_from_k_id = kc_from_k_id
            and kc_amount_approved > 0)
        select count(*) kol_closed,
               sum(in_period_disb) in_period_disb
        from tabl1
        into kol_kred_closed, in_period_disb;
        */

        select res_closed, res_repeat, res_koef
        from aarep_lo_retention_koef(:res_lo_id, dateadd(day, -27, :res_date_end), :res_date_end)
        into kol_kred_closed, in_period_disb, coeff_uderj;

        if (kol_kred_closed is null) then
          kol_kred_closed = 0;
        if (in_period_disb is null) then
          in_period_disb = 0;
        cpd_comment = cpd_comment || 'Всего закрытых за период: <b>' || round(kol_kred_closed, 0) || '</b><br>';
        cpd_comment = cpd_comment || 'Повторно выданные за период: <b>' || round(in_period_disb, 0) || '</b><br>';
--        if (kol_kred_closed > 0) then coeff_uderj = cast(in_period_disb as double precision) / kol_kred_closed * 100;
        cpd_comment = cpd_comment || 'коэффициент удержание клиентов: <b>' || round(coeff_uderj, 2) || '</b><br>';

        points_4 = case
                     when coeff_uderj > 40.0 then 12
                     when coeff_uderj > 35.0 then 8
                     when coeff_uderj > 30.0 then 0
                     when coeff_uderj > 25.0 then -4
                     else -8
                   end;
      end

      if (res_team_id <> res_team_id_prior) then -- Если команда отличается
      begin
-- Находим нужную строку команды за этот период
        cpr_id = null;
        select cpr_id
        from competresult
        where cpr_from_cpp_id = :cpp_id and
              cpr_object_kind = -1 and
              cpr_object_id = :res_team_id
        into :cpr_id;

        if (cpr_id is null) then -- если такой строки нет, создаем новую
        begin
          cpr_group_name = '';
          stage_cpp_id = null;
-- Получаем название Группы где последний раз была команда
          if (stage_cpp_id is not null) then
          begin
-- Если это первая неделя второго этапа, название лиги берем из поля комментария предыдущего расчета
            select coalesce(cpr_comment, '')
            from competresult, competperiod
            where cpr_from_cpp_id = :stage_cpp_id and
                  cpr_object_kind = -1 and
                  cpr_object_id = :res_team_id rows 1
            into :cpr_group_name;
          end
          else
          begin -- Если это не первая неделя второго Этапа
            select cpr_group_name
            from competresult, competperiod
            where cpr_from_cpp_id = cpp_id and
                  cpp_from_rd_id = :rd_id and
                  cpr_object_kind = -1 and
                  cpr_object_id = :res_team_id and
                  cpr_group_name <> '' and
                  cpp_orderno < :cpp_orderno
            order by cpp_orderno desc
            rows 1
            into :cpr_group_name;
          end
          if (cpr_group_name is null) then
            cpr_group_name = '';

          insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                    cpr_from_un_id, cpr_comment)
          values (:cpp_id, :cpr_group_name, -1, :res_team_id, 0, :un_id, '')
          returning cpr_id
          into :cpr_id;
        end
        res_team_id_prior = res_team_id;
      end

-- Вставляем данные по сотруднику
--      if (is_otpusk='') then -- отпускников убираю из вычисления письмо от 25.04.2017
--      begin
      insert into competdetails (cpd_from_cpr_id, cpd_object_kind, cpd_object_id, cpd_points_all, cpd_points_1,
                                 cpd_points_2, cpd_comment, cpd_staff_kolich, -- 1 если не отпуск, иначе 0

                                 cpd_add_data_1, -- Актив. клиенты
                                 cpd_add_data_2, -- Прирост клиентов
                                 cpd_add_data_3, -- PAR1
                                 cpd_add_data_4, -- Изменение PAR30 ------
                                 cpd_add_data_5, -- Сумма выдачи СВЗ
                                 cpd_add_data_6, -- Кол-во выдач
                                 cpd_add_data_9, cpd_points_3, cpd_points_4, cpd_points_6) -- Сумма выдачи СВЗ по ИПФ
      values (:cpr_id, 0, :res_lo_id, 0, :points_1, :points_2, left(:cpd_comment, 4032), iif(:is_otpusk = '', 1, 0),
              :ad_active_client, :ad_diff_client, :par1, :ad_delinq_balance_diff_30, :ad_disb_amount, :ad_disb_loan,
              :ad_disb_amount_ipf, :points_3, :points_4, :points_6);
--      end
    end

--    Средний рост кр. портфеля на одного СпФ в команде, учитывается на 2й и на 4й неделе
    if (datediff(day, cast('14.01.2018' as date), res_date_end) = (datediff(day, cast('14.01.2018' as date), res_date_end) / 14) * 14) then
    begin
-- Формируем в памяти данные по КР
      execute procedure hp_analitica_create_period(1000000, dateadd(day, -13, :res_date_end), :res_date_end, 1, null,
          null, 30, 0, 'olimpic')
          returning_values :res_period_id;

      if (res_period_id is null) then
      begin
        res_text = 'Не удалось расчитать данные за период';
        suspend;
        exit;
      end

      for select res_field_1, res_field_2, sum(ad_diff_balance)
          from hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group', 31, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
          where /* exists(select * from hr_pstatus
                  where ps_from_c_id = res_field_2
                  and ps_date_end >= :res_date_end
                  and ps_date_begin <= :res_date_end
                  and ps_status=1
                  and ps_is_working=1)
      and */ exists(select *
                    from refdata
                    where rd_main_code = 'StaffTeamList' and
                          rd_id = res_field_1 and
                          rd_tag = :rd_id)
          group by 1, 2
          having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
          into :res_team_id, :res_lo_id, :ad_diff_balance
      do
      begin
--        Прирост не учитывается по портфелю, принятому от другого СВЗ в отчетном периоде
        svz_portfel_begin = null;
        select sum(oks_balance)
        from oksummary
        where :res_date_begin - 8 between oks_date_begin and oks_date_end and
              :res_date_begin - 8 - oks_delinq_date <= 360 and
              oks_kc_id = :res_lo_id and
              oks_close_state = 0
        into svz_portfel_begin;
        if (svz_portfel_begin is null) then
          svz_portfel_begin = 0;

        svz_portfel_end = null;
        select sum(oks_balance)
        from oksummary
        where :res_date_end between oks_date_begin and oks_date_end and
              :res_date_end - oks_delinq_date <= 360 and
              oks_kc_id = :res_lo_id and
              oks_close_state = 0
        into svz_portfel_end;
        if (svz_portfel_end is null) then
          svz_portfel_end = 0;

        prinyat_portfel = null;
        select sum(oks_balance)
        from knaktlist, kontract, oksummary
        where knaktlist.is_delete = 0 and
              kal_date between :res_date_begin - 7 and :res_date_end and
              kal_from_ka_id > 0 and
              kal_new_kf_officer <> kal_prior_lo_id and
              kal_new_kf_officer = :res_lo_id and
              kal_from_kf_id = k_from_kf_id and
              k_id = oks_from_k_id and
              :res_date_end between oks_date_begin and oks_date_end and
              oks_balance > 0 and
              oks_kc_id = :res_lo_id and
              oks_close_state = 0
        into prinyat_portfel;
        if (prinyat_portfel is null) then
          prinyat_portfel = 0;

        ad_diff_balance = svz_portfel_end - svz_portfel_begin - prinyat_portfel;

        cpr_id = null;
        select cpr_id
        from competresult
        where cpr_from_cpp_id = :cpp_id and
              cpr_object_kind = -1 and
              cpr_object_id = :res_team_id
        into :cpr_id;

        update competdetails
        set cpd_add_data_7 = :ad_diff_balance,
            cpd_comment = cpd_comment || 'Рост портфеля КС: <b>' || round(:ad_diff_balance, 0) || '</b><br>'
        where cpd_from_cpr_id = :cpr_id and
              cpd_object_kind = 0 and
              cpd_object_id = :res_lo_id;
      end

      kol_komand = 37;
      k = 0;
      for select cpr_id, cpr_group_name, cpd_object_id, cpd_add_data_7
          from competresult, competdetails
          where cpr_from_cpp_id = :cpp_id and
                cpr_group_name <> '' and
                cpr_object_kind = -1 -- Refdata, названия групп
                and
                cpd_from_cpr_id = cpr_id
--      and cpd_staff_kolich<>0   -- отпуск
          order by cpr_group_name, cpd_add_data_7 desc
          into :cpr_id, :cpr_group_name, :res_lo_id, :avg_rost_portfel

      do
      begin
--      пункт 3.1, Д. Рост кредитного портфеля.
        k = k + 1;
        points_3 = 0;

        if (k <= 10) then
          points_3 = 6;
        else
        if (k > kol_komand - 10) then
          points_3 = -6;

        update competdetails
        set cpd_add_data_8 = :avg_rost_portfel,
            cpd_points_3 = :points_3
        where cpd_from_cpr_id = :cpr_id and
              cpd_object_id = :res_lo_id
--        and cpd_staff_kolich<>0
        ;
      end

    end

-- Пересчитываем итоги по командам
    for select cpr_id
        from competresult
        where cpr_from_cpp_id = :cpp_id
        into :cpr_id
    do
    begin
      update competdetails
      set cpd_points_all = cpd_points_1 + cpd_points_2 + cpd_points_3 + cpd_points_4 + cpd_points_5 + cpd_points_6 + cpd_points_7 + cpd_points_8 + cpd_points_9
      where cpd_from_cpr_id = :cpr_id;

      avg_points = 0.0;
      lo_panelty_count = 0.0;
      select round(avg(cpd_points_all), 2), -- средний балл
             /*
      sum(iif(cpd_points_1       -- кол-во свз со штрафами
      +cpd_points_2
      +cpd_points_3
      +cpd_points_4
      +cpd_points_5
      +cpd_points_6
      +cpd_points_7
      +cpd_points_8
      +cpd_points_9<0, 1, 0))
      */
             sum(iif(cpd_points_1 < 0 or cpd_points_2 < 0 or cpd_points_3 < 0 or cpd_points_4 < 0, 1, 0))
      from competdetails
      where cpd_from_cpr_id = :cpr_id
      into :avg_points, :lo_panelty_count;

      if (avg_points is null) then
        avg_points = 0;
      if (lo_panelty_count is null) then
        lo_panelty_count = 0;

      total_points = avg_points; -- Средний балл комманды

      add_points = 0.0;
      if (lo_panelty_count = 0) -- Если никто не штрафован, плюс 2 балла
      then
        add_points = 2.0;

      if (lo_panelty_count >= 1) -- Если оштрафованы 2 и более СВЗ, минус 2 балла
      then
        add_points = -2.0;

      update competresult
      set cpr_calced_points = :total_points,
          cpr_additional_points = :add_points
      where (cpr_id = :cpr_id);
    end

-- Присвоение одного балла у кого самый больший балл в группе
    add_points = 1;
    /* for select cpr_group_name, max(cpr_final_points)
    from competresult
    where cpr_from_cpp_id=:cpp_id
    group by cpr_group_name
    into :group_name, max_points
    do
    begin
      update competresult
      set cpr_additional_points = cpr_additional_points + :add_points
      where cpr_from_cpp_id=:cpp_id
      and cpr_group_name=:group_name
      and cpr_final_points=:max_points;
    end
    */
    group_name_prior = '[{x}]';
    for select cpr_id, cpr_group_name
        from competresult, competdetails
        where cpr_from_cpp_id = :cpp_id and
              cpr_group_name <> '' and
              cpr_object_kind = -1 and
              cpd_from_cpr_id = cpr_id
        order by cpr_group_name, cpr_final_points desc, cpd_add_data_6 desc, cpd_add_data_2 desc
        into :cpr_id, :group_name
    do
    begin
      if (group_name <> group_name_prior) then
      begin
        group_name_prior = group_name;
        update competresult
        set cpr_additional_points = cpr_additional_points + :add_points
        where cpr_id = :cpr_id;
      end
    end

    execute procedure hp_competition_recalc_order(:cpp_id);
  end

  if (stage = 1) then -- если промежуточные итоги тура
  begin
-- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    for select cpr_object_kind, cpr_object_id, sum(cpr_final_points)
        from competresult, competperiod
        where cpp_from_rd_id = :rd_id and
              cpp_id = cpr_from_cpp_id and
              cpp_stage = (:stage - 1) and
              cpp_date_end between :res_date_begin and :res_date_end
        group by cpr_object_kind, cpr_object_id
        into :kind, :res_team_id, :cpr_final_points
    do
    begin
      cpr_group_name = '';

-- Получаем название Группы где последний раз была команда
      select cpr_group_name
      from competresult, competperiod
      where cpr_from_cpp_id = cpp_id and
            cpp_from_rd_id = :rd_id and
            cpr_object_kind = -1 and
            cpr_object_id = :res_team_id and
            cpr_group_name <> '' and
            cpp_orderno < :cpp_orderno
      order by cpp_orderno desc
      rows 1
      into :cpr_group_name;
      if (cpr_group_name is null) then
        cpr_group_name = '';

      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
                                cpr_result_kind, cpr_from_un_id)
      values (:cpp_id, :cpr_group_name, :kind, :res_team_id, :cpr_final_points, 0, :un_id)
      returning cpr_id
      into :cpr_id;
    end

    execute procedure hp_competition_recalc_order(:cpp_id);
  end

  if (stage = 2) then -- если промежуточные итоги этапа
  begin
-- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    for select cpr_object_kind, cpr_object_id, avg(cpr_final_points)
        from competresult, competperiod
        where cpp_from_rd_id = :rd_id and
              cpp_id = cpr_from_cpp_id and
              cpp_stage = (:stage - 1) and
              cpp_date_end between :res_date_begin and :res_date_end
        group by cpr_object_kind, cpr_object_id
        into :kind, :res_team_id, :cpr_final_points
    do
    begin
      cpr_group_name = '';

-- Получаем название Группы где последний раз была команда
      select cpr_group_name
      from competresult, competperiod
      where cpr_from_cpp_id = cpp_id and
            cpp_from_rd_id = :rd_id and
            cpr_object_kind = -1 and
            cpr_object_id = :res_team_id and
            cpr_group_name <> '' and
            cpp_orderno < :cpp_orderno
      order by cpp_orderno desc
      rows 1
      into :cpr_group_name;
      if (cpr_group_name is null) then
        cpr_group_name = '';

      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
                                cpr_result_kind, cpr_from_un_id)
      values (:cpp_id, :cpr_group_name, :kind, :res_team_id, :cpr_final_points, 0, :un_id)
      returning cpr_id
      into :cpr_id;
    end

    execute procedure hp_competition_recalc_order(:cpp_id);
  end

  if (stage = 3) then -- если финальные итоги
  begin
    res_text = 'Автоматизация итогов еще не готов';
  end
  suspend;
end
