CREATE PROCEDURE ISHENIM_GET_KONTRACT_DATA_TMP(
    POCKET_ID INTEGER)
RETURNS (
    K_ID ID_TYPE,
    ORDER_MARK SMALLINT,
    PHASEOFCONTRACT VARCHAR(10),
    IS_FIXED SMALLINT)
AS
declare variable k_id_old id_type;
declare variable rep_date_begin date;
declare variable rep_date_end date;
declare variable pocket_type varchar(1);
begin
  k_id_old = 0;

  select cp.cp_date_begin, cp.cp_date_end, cp.cp_pocket_type
  from cib_pocket cp
  where cp.cp_id = :pocket_id
  into :rep_date_begin, :rep_date_end, :pocket_type;

  if ((rep_date_end is null) or (pocket_type is null)) then
    exit;

  for select ks_from_k_id, order_mark, phaseofcontract, is_fixed
      from (
            select ks_from_k_id, order_mark, 'Closed' as phaseofcontract, 0 as is_fixed
            from (select ks.ks_from_k_id, 1 as order_mark
                  from kontstate ks
                  where ks.ks_state = 2 and
                        ks.ks_date_begin between :rep_date_begin and :rep_date_end and
                        ks.ks_ol_is_delete = 0
            ) closedKont
            union
            select ks_from_k_id, order_mark, 'Open' as phaseofcontract, 0 as is_fixed
            from (select ks.ks_from_k_id, 2 as order_mark
                  from kontstate ks
                  where ks.ks_state = 1 and
                        :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
                        ks.ks_ol_is_delete = 0
            ) activeKont
            union
            select ks_from_k_id, order_mark, phaseofcontract, 1 as is_fixed
            from (select ks.ks_from_k_id,
                  0 as order_mark,
                  iif(ks.ks_state = 1, 'Open', 'Closed') as phaseofcontract
                  from CIB_KONTRACT_ERRORS cke
                  inner join kontstate ks  on ks.ks_from_k_id = cke.cke_from_k_id
                  where :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
                        ks.ks_ol_is_delete = 0 and
                        cke.cke_checked_datetime  >= :rep_date_end - 1 and
                        cke.cke_checked_datetime < :rep_date_end
            ) CheckedKont
            ---------------- Актуализаци информации в Ишеним ----------------- >
            -- > Активные в КИБ, но закрытые в МБулак +
            -- > Активные в МБулак, которые в КИБ не были отпралены
            union
            select ks_from_k_id, order_mark, phaseofcontract, 1 as is_fixed
            from (select ks.ks_from_k_id,
                  0 as order_mark,
                  iif(ks.ks_state = 1, 'Open', 'Closed') as phaseofcontract
                  from cib_actuality cke
                  inner join kontstate ks  on ks.ks_from_k_id = cke.k_id
                  where :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
                        ks.ks_ol_is_delete = 0
            ) CheckedKont
            ------------------------------------------------------------------ <
      ) total
      order by ks_from_k_id, order_mark
      into :k_id, :order_mark, :phaseofcontract, :is_fixed
  do
  begin
    if (:k_id <> :k_id_old) then
    begin
      k_id_old = k_id;

      if (not exists(select *
                     from kib_ignore ci
                     where ci.ki_from_k_id = :k_id)) then
        suspend;
    end
  end
end
