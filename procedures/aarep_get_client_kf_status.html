CREATE PROCEDURE AAREP_GET_CLIENT_KF_STATUS(
    C_ID DECIMAL(18,0),
    REGION_ID INTEGER,
    REP_DATE DATE,
    KFOLDER_ID DECIMAL(18,0),
    C_FIO VARCHAR(100))
RETURNS (
    KS_DATE_BEGIN DATE,
    KF_ID DECIMAL(18,0),
    KF_NAME VARCHAR(255),
    KF_OTHER_OFFICE INTEGER,
    KF_OFFICER_ID DECIMAL(18,0),
    KF_OFFICER_FIO VARCHAR(100),
    KF_CODE VARCHAR(100))
AS
declare variable ks_state integer;
declare variable ma_id integer;
declare variable ma_number integer;
declare variable ma_date_app date;
declare variable fio varchar(100);
declare variable reg_name varchar(100);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  kf_name = '';
  kf_code = '';
  kf_other_office = 0;
  if ((c_id is not null) and
      (c_id <> 0)) then
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into :fio;
  end
  else
  begin
    fio = :c_fio;
  end

  /* Проверка в договорах в этом офисе */
  if ((c_id is not null) and
      (c_id <> 0)) then
  begin
    for select ks_date_begin, k_from_kf_id, kal_new_kf_name, ks_state, kal_new_kf_officer, c_name_fio
        from kontract, knclient, knaktlist, kontstate, clients
        where k_id = kc_from_k_id and
              kc_from_c_id = :c_id and
              k_from_kf_id = kal_from_kf_id and
              knaktlist.is_delete = 0 and
              kal_date <= :rep_date and
              kal_date_end >= :rep_date and
              kc_amount_approved is not null and
              kc_amount_approved <> 0 and
              k_id = ks_from_k_id and
              cast(ks_date_begin as date) <= :rep_date and
              ks_date_end >= :rep_date and
              ks_ol_is_delete = 0 and
              kal_new_reg_id = :region_id and
              k_from_kf_id <> :kfolder_id and
              k_status >= 0 and
              kal_new_kf_officer = c_id
        order by ks_date_begin desc
        rows 1
        into :ks_date_begin, :kf_id, :kf_name, :ks_state, :kf_officer_id, :kf_officer_fio
    do
    begin
      kf_code = kf_name;
      if (ks_state = 0) then
        kf_name = kf_name || ' в процессе с ' || ks_date_begin;
      if (ks_state = 1) then
        kf_name = kf_name || ' активный с ' || ks_date_begin;
      if (ks_state = 2) then
        kf_name = kf_name || ' закрыт с ' || ks_date_begin;
    end
    if (ks_date_begin is null) then
      ks_date_begin = cast('NOW' as date) - 60;
    /* Проверка в заявках в этом офисе */
    for select distinct ma_id, ma_number, ma_date_app, ma_from_kf_id, kal_new_kf_name, kal_new_kf_officer, c_name_fio
        from miniapp, miniappclient, knaktlist, clients
        where mac_from_ma_id = ma_id and
              mac_from_c_id = :c_id and
              ma_date_app >= :ks_date_begin and
              ma_date_end = '31.12.9999' and
              kal_date <= :rep_date and
              kal_date_end >= :rep_date and
              kal_from_kf_id <> :kfolder_id and
              kal_from_kf_id = ma_from_kf_id and
              ma_office_id = :region_id and
              ma_is_active = 1 and
              mac_amount <> 0 and
              kal_new_kf_officer = c_id and
              mac_is_poruch = 0 /* не поручитель*/
              rows 1
        into :ma_id, :ma_number, :ma_date_app, :kf_id, :kf_name, :kf_officer_id, :kf_officer_fio
    do
    begin
      kf_code = kf_name;
      kf_name = kf_name || ' заявка от ' || ma_date_app;
    end
    if (kf_name = '') then
    begin
      /* Проверка в других офисах */
      for select ks_date_begin, k_from_kf_id, kal_new_kf_name, ks_state, kal_new_kf_officer, c_name_fio, reg_name
          from kontract, knclient, knaktlist, kontstate, clients, region
          where k_id = kc_from_k_id and
                kc_from_c_id = :c_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                kal_date <= :rep_date and
                kal_date_end >= :rep_date and
                kc_amount_approved is not null and
                kc_amount_approved <> 0 and
                k_id = ks_from_k_id and
                cast(ks_date_begin as date) <= :rep_date and
                ks_date_end >= :rep_date and
                ks_ol_is_delete = 0 and
                ks_state <= 1 and
                kal_new_reg_id <> :region_id and
                k_status >= 0 and
                kal_new_kf_officer = c_id and
                kal_new_reg_id = reg_unicode
          order by ks_date_begin desc
          rows 1
          into :ks_date_begin, :kf_id, :kf_name, :ks_state, :kf_officer_id, :kf_officer_fio, :reg_name
      do
      begin
        kf_code = kf_name;
        kf_other_office = 1;
        if (ks_state = 0) then
          kf_name = kf_name || ' в процессе, офис: ' || :reg_name || ', от ' || ks_date_begin;
        if (ks_state = 1) then
          kf_name = kf_name || ' активный дог. офис: ' || :reg_name || ', от ' || ks_date_begin;
      end
      if (kf_name = '') then
      begin
        for select ma_id, ma_number, ma_date_app, ma_from_kf_id, kal_new_kf_name, kal_new_kf_officer, c_name_fio,
                   reg_name
            from miniapp, miniappclient, knaktlist, clients, region
            where mac_from_ma_id = ma_id and
                  mac_from_c_id = :c_id and
                  ma_date_app >= (:rep_date - 30) and
                  ma_date_end >= '31.12.9999' and
                  kal_date <= :rep_date and
                  kal_date_end >= :rep_date and
                  kal_from_kf_id = ma_from_kf_id and
                  ma_office_id <> :region_id and
                  ma_is_active = 1 and
                  mac_amount <> 0 and
                  kal_new_kf_officer = c_id and
                  kal_new_reg_id = reg_unicode and
                  mac_is_poruch = 0 /* не поручитель*/
                  and
                  knaktlist.is_delete = 0
            order by ma_id desc, ma_number desc
            rows 1
            into :ma_id, :ma_number, :ma_date_app, :kf_id, :kf_name, :kf_officer_id, :kf_officer_fio, :reg_name
        do
        begin
          kf_code = kf_name;
          kf_other_office = 1;
          kf_name = kf_name || ' есть заявка, офис: ' || :reg_name || ' от ' || ma_date_app;
        end
      end
    end
  end

  if (kf_name <> '') then
    suspend;
end
