CREATE PROCEDURE UXP$UXSOURCE_SQL_JOIN(
    INDEX_NAME D_STR_100,
    MAIN_PREFIX D_STR_10,
    FK_PREFIX D_STR_10)
RETURNS (
    JOIN_TEXT D_STR_LONG,
    ERROR_TEXT D_STR_255)
AS
declare variable field_source d_str_code;
declare variable fields d_str_100;
declare variable fk_index d_str_100;
declare variable fk_source d_str_code;
declare variable fk_fields d_str_100;
declare variable fields_count d_integer;
declare variable fk_fields_count d_integer;
declare variable fields_item d_str_code;
declare variable fk_fields_item d_str_code;
declare variable pos d_integer;
begin
  join_text = '';
  error_text = '';

  if (trim(coalesce(index_name, '')) = '') then
  begin
    error_text = 'empty index_name';
    suspend;
    exit;
  end

  select uix_source, uix_fields, uix_fk_index, uix_count
  from ux$index
  where uix_index = :index_name
  into :field_source, :fields, :fk_index, fields_count;

  if (row_count = 0) then
  begin
    error_text =  f_format('Index not found: %1', index_name);
    suspend;
    exit;
  end

  if (fk_index = '') then
  begin
    error_text = f_format('"%1" is not FK index', index_name);
    suspend;
    exit;
  end

  select uix_source, uix_fields, uix_count
  from ux$index
  where uix_index = :fk_index
  into :fk_source, :fk_fields, fk_fields_count;


  if (fields_count <> fk_fields_count) then
  begin
    error_text = f_format('Different field count. FK fields: %1, PK fields: %2', fields, fk_fields);
    suspend;
    exit;
  end

  if (coalesce(main_prefix, '') = '') then
  begin
    pos = position('_', fields);
    if (pos > 0) then
      main_prefix = left(fields, pos - 1);
    else
      main_prefix = left(field_source, 3);
  end

  if (coalesce(fk_prefix, '') = '') then
  begin
    pos = position('_', fk_fields);
    if (pos > 0) then
      fk_prefix = left(fk_fields, pos - 1);
    else
      fk_prefix = left(fk_source, 3);
  end

  while (fields <> '') do
  begin
    pos = position(';', fields);
    if (pos > 0) then
    begin
      fields_item = left(fields, pos - 1);
      fields = substring(fields from pos + 1);
    end else
    begin
      fields_item = fields;
      fields = '';
    end

    pos = position(';', fk_fields);
    if (pos > 0) then
    begin
      fk_fields_item = left(fk_fields, pos - 1);
      fk_fields = substring(fk_fields from pos + 1);
    end else
    begin
      fk_fields_item = fk_fields;
      fk_fields = '';
    end

    if (join_text <> '') then
      join_text = join_text || ' and';

    join_text = join_text || main_prefix || '.' || fields_item || ' = ' || fk_prefix || '.' || fk_fields_item;
  end

  join_text = ' inner join ' || fk_source || ' ' || fk_prefix || ' on (' || join_text || ')';
  suspend;
end
