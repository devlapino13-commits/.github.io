    if (char_length(res_pass_no_mask) >= 5) then
      res_pass_no_mask = left(res_pass_no_mask, char_length(res_pass_no_mask) - 5) || '***' || right(res_pass_no_mask, 2);

    if (char_length(res_inn_mask) >= 7) then
      res_inn_mask = left(res_inn_mask, char_length(res_inn_mask) - 7) || '***' || right(res_inn_mask, 4);

    suspend;
  end
