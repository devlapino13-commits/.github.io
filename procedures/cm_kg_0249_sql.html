CREATE PROCEDURE CM_KG_0249_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable C_NAME_FIO D_STR_100;
declare variable SELF_PROHIBITION_STATUS integer;
begin

  res_error_is_critical = 2;

  for select
      kc.kc_from_c_id,
      cl.c_name_fio
  from knclient kc
  inner join clients cl on cl.c_id = kc.kc_from_c_id
  where kc.kc_from_k_id = :k_id
        and kc.kc_amount_approved > 0
  into :c_id, :c_name_fio
  do
  begin
    select res
    from hp_self_prohibition_get_status(:c_id, 'kontract', :k_id)
    into :self_prohibition_status;

    if (self_prohibition_status is null) then
    begin
      res_error_text = 'Ответ от сервиса Самозапрет ещё не получен!';
      suspend;
    end
    else if (self_prohibition_status = 0) then
    begin
      res_error_text = 'Запрос в сервис Самозапрет обрабатывается, подождите!';
      suspend;
    end
    else if (self_prohibition_status = -1) then
    begin
      res_error_text = 'Ошибка при получении ответа от сервиса Самозапрет, переотправьте запрос!';
      suspend;
    end
    else if (self_prohibition_status = 1) then
    begin
      res_error_text = 'У клиента действует самозапрет — выдача невозможна!';
      suspend;
    end
    else if (self_prohibition_status = -2) then
    begin
      res_error_text = 'Нет информации по самозапрету. По клиенту "' || :c_name_fio || '" нужно прикрепить скриншот из Тундука!';
      suspend;
    end
    else if (self_prohibition_status = -3) then
    begin
      res_error_text = 'Запрос в сервис Самозапрет осуществлялся, отправьте запрос!';
      suspend;
    end
    else
      continue;
  end
end
