CREATE PROCEDURE HP_PDN_LOG(
    KC_ID D_ID)
RETURNS (
    KPLG_DATETIME D_DATETIME,
    UN_FIO D_STR_255,
    KPLG_TAG D_STR_CODE,
    PDN D_CURRENCY,
    SP_NEW D_CURRENCY,
    SP_OTHER D_CURRENCY,
    CALC_PERIOD_MONTH D_INTEGER,
    PERCENT_SUM D_CURRENCY,
    INCOME D_CURRENCY,
    CREDDATE D_DATE,
    CREDENDDATE D_DATE,
    CALC_DATE D_DATE)
AS
begin
  for select kplg_datetime, kplg_pdn_koef,
        (select un_fio from username where un_id = kplg_from_un_id),
        f_enum('pdn_calc_source', kplg_tag_id, 'name', 0),
        kplg_sp_new, kplg_sp_other, kplg_calc_period_month, kplg_percent_sum,
        kplg_income, kplg_creddate, kplg_credenddate, kplg_calc_date
  from knclient_pdn, knclient_pdn_log
  where kplg_from_kcp_id = kcp_id
    and kc_id = :kc_id
  order by 1
  into kplg_datetime, pdn, un_fio, kplg_tag,
        sp_new, sp_other, calc_period_month, percent_sum,
        income, creddate, credenddate, calc_date
  do
  begin
    if (un_fio = 'Робот FinStart') then
      un_fio = 'Не определено';
    suspend;
  end
end
