CREATE PROCEDURE AAREP_PETOPERS_PRINT(
    OL_ID DECIMAL(18,0))
RETURNS (
    PEO_FROM_OL_ID DECIMAL(18,0),
    PEO_ID INTEGER,
    PEO_BILL_ID DECIMAL(18,0),
    PEO_BILL_LINE_NO INTEGER,
    PEO_OPER_TYPE INTEGER,
    PEO_FROM_C_ID DECIMAL(18,0),
    PEO_COST DECIMAL(18,4),
    PEO_FROM_PEN_ID INTEGER,
    PEO_KOLICH DECIMAL(18,4),
    PEO_LITER DECIMAL(18,4),
    PEO_SUMMA DECIMAL(18,4),
    PEO_FROM_PET_ID INTEGER,
    PEO_TRANS_LITER DECIMAL(18,4),
    PEO_TRANS_SUMMA DECIMAL(18,4),
    PEO_PROBEG_FACT_LITER DECIMAL(18,4),
    PEO_PROBEG_FACT_KM DECIMAL(18,4),
    PEO_SPEEDOMETR INTEGER,
    PEO_OPER_DATE DATE,
    PEO_FROM_REG_ID INTEGER,
    PEO_TALON_LITER DECIMAL(18,4),
    PEO_CLIENT_FIO VARCHAR(100),
    PEO_OL_DATE DATE,
    PEO_OL_MEMO VARCHAR(1024),
    PEO_OPER_STR VARCHAR(50),
    PEO_TRANS_NAME D_STR_LONG,
    PEO_REG_NAME VARCHAR(100),
    PEO_PEN_NAME VARCHAR(100))
AS
declare variable peo_matotvet_id decimal(18,0);
declare variable peo_matotvet_fio varchar(100);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select peo_from_ol_id, peo_id, peo_bill_id, peo_bill_line_no, peo_oper_type, peo_from_c_id, peo_cost,
             peo_from_pen_id, peo_kolich, peo_liter, peo_summa, peo_from_pet_id, peo_trans_liter, peo_trans_summa,
             peo_probeg_fact_liter, peo_probeg_fact_km, peo_speedometr, peo_oper_date, peo_from_reg_id, peo_talon_liter,
             c_name_fio, ol_date, ol_memo, pet_name || ' ' || pet_label, reg_name, pen_name, ol_document_client
      from region, operlist, clients, petname, petopers
      left join pettrans on (peo_from_pet_id = pet_id)
      where ol_id = peo_from_ol_id and
            ol_id = :ol_id and
            peo_from_c_id = c_id and
            peo_from_reg_id = reg_unicode and
            pen_id = peo_from_pen_id
      order by peo_from_c_id, peo_from_pen_id
      into :peo_from_ol_id, :peo_id, :peo_bill_id, :peo_bill_line_no, :peo_oper_type, :peo_from_c_id, :peo_cost,
           :peo_from_pen_id, :peo_kolich, :peo_liter, :peo_summa, :peo_from_pet_id, :peo_trans_liter, :peo_trans_summa,
           :peo_probeg_fact_liter, :peo_probeg_fact_km, :peo_speedometr, :peo_oper_date, :peo_from_reg_id,
           :peo_talon_liter, :peo_client_fio, :peo_ol_date, :peo_ol_memo, :peo_trans_name, :peo_reg_name, :peo_pen_name,
           :peo_matotvet_id
  do
  begin
    peo_matotvet_fio = '';
    if ((peo_oper_type = 1) or (peo_oper_type = 2)) then
    begin
      select c_name_fio
      from clients
      where c_id = :peo_matotvet_id
      into :peo_matotvet_fio;
      if (peo_matotvet_fio is null) then
        peo_matotvet_fio = '';
    end
    if (peo_matotvet_fio <> '') then
      peo_ol_memo = peo_ol_memo || ' через ' || :peo_matotvet_fio;
    if (peo_oper_type = 0) then
      peo_oper_str = 'Первоначальные остатки';
    if (peo_oper_type = 1) then
      peo_oper_str = 'Приход талона на склад';
    if (peo_oper_type = 2) then
      peo_oper_str = 'Залит через Чек';
    if (peo_oper_type = 3) then
      peo_oper_str = 'Передан в другой офис';
    if (peo_oper_type = 4) then
      peo_oper_str = 'Принят из другого офиса';
    if (peo_oper_type = 5) then
      peo_oper_str = 'Талон передан на расход';
    if (peo_oper_type = 6) then
      peo_oper_str = 'Возврат из бака';
    if (peo_oper_type = 7) then
      peo_oper_str = 'Списан из бака';
    if (peo_oper_type = 8) then
      peo_oper_str = 'Заменена матответ.лица';
    if (peo_oper_type = 9) then
      peo_oper_str = 'Передача между объектами';
    suspend;
  end
end
