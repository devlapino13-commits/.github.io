CREATE PROCEDURE OKT_PANTERA_MAKE_CALL(
    K_ID DECIMAL(18,0),
    NUM_ID INTEGER,
    CALLED_OPER VARCHAR(100),
    STATUS SMALLINT,
    PANT_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    CC_ID INTEGER,
    MEMO_STR VARCHAR(1000),
    CALL_RESULT SMALLINT,
    CELLNO_TYPE STR_100,
    PROJECT_TYPE SMALLINT,
    PORUCH_ID ID_TYPE,
    NEXT_CALL_DATETIME TIMESTAMP,
    ASKED_CALLBACK SMALLINT,
    BLOCKED_IGNORE INTEGER = 0,
    IS_DISMISS SMALLINT = 0)
RETURNS (
    MESS_STR VARCHAR(255))
AS
declare variable do_block smallint;
declare variable oper_max_limit smallint;
declare variable db_id integer;
declare variable old_pant_id id_type;
declare variable new_pant_id id_type;
declare variable transfer_reason_code str_255;
declare variable subject_kind_id integer;
begin
  cc_id = coalesce(cc_id, 0);
  c_id = coalesce(c_id, 0);
  poruch_id = coalesce(poruch_id, 0);
  mess_str = '';
  is_dismiss = coalesce(is_dismiss, 0);

  oper_max_limit = 200;

  do_block = 1;

  :db_id = f_get_db_id();
  if (status = 0) then -- оператор бронирует за собой группу, выбирая ее для обзвона
  begin
    old_pant_id = null;

    select pk.pant_id, pk.pant_from_num_id
    from pantera_kontracts pk
    where pk.pant_from_k_id = :k_id
--and pk.pant_project_type = :PROJECT_TYPE
          and
          pk.pant_date_end = cast('30.12.9999' as date)
    into :old_pant_id, :num_id;

    if ((blocked_ignore = 0) and
        (old_pant_id is not null)) then
    begin
      do_block = 0;
      mess_str = 'Группа уже заблокирована';
    end
    /*
      else
      begin
        if ((PROJECT_TYPE in (4, 5, 6)) and (db_id = 81001)) then
        begin
            if ((select count(*)
                        from pantera_kontracts pantk
                        inner join kontstate ks on pantk.pant_from_k_id = ks.ks_from_k_id
                        where pantk.pant_oper_called = :CALLED_OPER
                        and current_date between pantk.pant_date_begin and pantk.pant_date_end
                        and pantk.pant_project_type = :PROJECT_TYPE
                        and current_date between ks.ks_date_begin and ks.ks_date_end
                        and ks.ks_ol_is_delete = 0
                        and ks.ks_state = 1) >= :oper_max_limit
                        )  then
             begin
                do_block = 0;
                MESS_STR = 'За оператором уже закреплено '||cast(oper_max_limit as str_100)||' групп. Блокировка запрещена.';
             end    
        end
      end
      */

    if (do_block = 1) then
    begin
      new_pant_id = null;

      insert into pantera_kontracts (pant_from_k_id, pant_from_num_id, pant_oper_called, pant_date_begin, pant_date_end,
                                     pant_project_type, pant_is_dismiss)
      values (:k_id, :num_id, :called_oper, current_date, cast('30.12.9999' as date), :project_type, :is_dismiss)
      returning pant_id
      into :new_pant_id;

      if (old_pant_id is null) then
        transfer_reason_code = 'SirenaTransferReason.OperatorBlocked';
      else
      begin
        if (is_dismiss = 1) then -- передача из-за увольнения Оператора
          transfer_reason_code = 'SirenaTransferReason.TransferForOperDismissal';
        else
          transfer_reason_code = 'SirenaTransferReason.OperTransfer';
      end
-- фиксирую факт передачи
      execute procedure hp_pantera_transfer_insert(current_date, :old_pant_id, :new_pant_id, :transfer_reason_code);
    end
  end
  if (status = 1) then -- оператор совершает звонок
  begin
    select rd.rd_id
    from refdata rd
    where rd.rd_code = :cellno_type
    into :subject_kind_id;

    if ((cc_id = 0) and
        (poruch_id = 0)) then --  если оператор позвонил участнику группы
    begin
      insert into pantera_clients (pcl_from_pan_id, pcl_from_c_id, pcl_res, pcl_memo, pcl_next_call, pcl_asked_callback,
                                   pcl_subject_kind)
      values (:pant_id, :c_id, :call_result, :memo_str, :next_call_datetime, :asked_callback, :subject_kind_id);
    end
    else
    if (cc_id <> 0) then --  если оператор позвонил контактному лицу
    begin
      insert into pantera_contacts (pcon_from_pan_id, pcon_from_c_id, pcon_from_cc_id, pcon_res, pcon_memo,
                                    pcon_next_call, pcon_asked_callback, pcon_subject_id)
      values (:pant_id, :c_id, :cc_id, :call_result, :memo_str, :next_call_datetime, :asked_callback, :subject_kind_id);
    end
    else
    if (poruch_id <> 0) then --  если оператор позвонил поручителю
    begin
      insert into pantera_contacts (pcon_from_pan_id, pcon_from_c_id, pcon_from_poruch_id, pcon_res, pcon_memo,
                                    pcon_next_call, pcon_asked_callback, pcon_subject_id)
      values (:pant_id, :c_id, :poruch_id, :call_result, :memo_str, :next_call_datetime, :asked_callback,
              :subject_kind_id);
    end
  end

  suspend;
end
