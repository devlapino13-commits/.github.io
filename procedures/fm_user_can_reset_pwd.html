CREATE PROCEDURE FM_USER_CAN_RESET_PWD(
    UN_NAME D_STR_CODE,
    CAN_USE_BIO D_BOOL = 0)
RETURNS (
    ERR_TEXT D_STR_255,
    PHONE_NO D_STR_CODE,
    CHECK_KIND D_BOOL,
    C_ID D_ID)
AS
declare variable db_id d_integer;
declare variable un_id d_id;
declare variable c_citizen d_integer;
begin
  -- Процедура проверяет возможность сброса пароля через смс
  -- Если пользователь найден и может изменить пароль, то в параметре
  --   phone_no возвращается номер телефона
  -- Иначе возвращается текст ошибки в параметре err_text

  db_id = f_get_db_id();
  can_use_bio = coalesce(can_use_bio, 0);
  check_kind = 0;

  c_id = null;
  err_text = '';
  phone_no = '';
  select un_id, un_from_c_id, c.c_citizen
  from username
  inner join clients c on c.c_id = un_from_c_id
  where un_name = :un_name rows 1
  into un_id, c_id, c_citizen;

  if (c_id is null) then
    err_text = iif(db_id <> 94401, 'Пользователь не найден.', 'Istifadechi tapilmadi.');
  else
  begin
    phone_no = '';

    if (not(exists(select *
                   from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.Login')
                   where has_access = 1))) then
    begin
      if (not(exists(select *
                     from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobileIOS.Login')
                     where has_access = 1))) then
      begin
        err_text = 'У вас нет доступа к данной системе.';
        suspend;
        exit;
      end
    end

    select cont_value
    from hp_contdata_get_cell_main(0, :c_id)
    into :phone_no;

    if (coalesce(phone_no, '') = '') then
      err_text = case db_id
                   when 81001 then 'Отсутствует подтвержденный № телефона, обратитесь в бухгалтерию.'
                   when 94401 then 'Tesdiqlenen mobil nomre yoxdur, texniki destek shobesine muraciyet edin'
                   else 'Отсутствует подтвержденный № телефона, обратитесь в тех. поддержку.'
                 end;
    else
    if (char_length(phone_no) >= 6) then
      phone_no = left(phone_no, char_length(phone_no) - 6) || '****' || right(phone_no, 2);

    if (db_id = 41701 and
        can_use_bio = 1 and
        c_citizen = f_get_rd_id('Citizen.KG')) then
    begin
      check_kind = 1;

      if (not exists(select 1
                     from extinfolist eil
                     inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                     inner join extinfolistbls eib on eib.eib_from_eil_id = eil.eil_id
                     where eil.eil_from_c_id = :c_id and
                           eic.eic_code = 'Tunduk' and
                           eil.eil_status = 2 and
                           eil.eil_is_json = 1 and
                           eil.eil_from_bls_id > 0 and
                           eib.eib_kind = f_enum_id('extinfo_bls_kind', 'tunduk_client_photo'))) then
        err_text = 'Необходимо добавить фото с ГРС, обратитесь в отдел кадров';
    end
    -- Сохраняем флаг использования биометрии для дальнейшей проверки в FM_USER_RESET_PWD
    update username
    set un_password_change_bio = :check_kind
    where un_id = :un_id and
          un_password_change_bio <> :check_kind;

  end
  suspend;
end
