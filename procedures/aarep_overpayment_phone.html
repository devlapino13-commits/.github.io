CREATE PROCEDURE AAREP_OVERPAYMENT_PHONE(
    OVP_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    NAME_FIO D_STR_100,
    PASS_NO D_STR_CODE,
    PASS_WHO D_STR_255,
    PASS_DATE D_DATE,
    KONTRACT_NO D_STR_20,
    PHONE D_STR_20,
    AMOUNT D_CURRENCY,
    SIGN_DATE D_DATE,
    QR_CODE D_STR_LONG)
AS
declare variable C_ID D_ID;
declare variable EDL_ID D_ID;
begin
  for
    select
      c_id,
      c_name_fio,
      c_pass_no,
      c_pass_who,
      c_pass_date,
      k_kontract_no,
      ovp_edl_id,
      ovp_phone,
      ovp_amount
    from overpayment
      inner join kontract
        on k_id = ovp_from_k_id
          and k_disb_date < '12.04.2021'
      inner join clients
        on c_id = ovp_from_c_id
    where ovp_id = :ovp_id
      and ovp_edl_id > 0
      and coalesce(ovp_phone, '') <> ''
  into
    :c_id,
    :name_fio,
    :pass_no,
    :pass_who,
    :pass_date,
    :kontract_no,
    :edl_id,
    :phone,
    :amount
  do
  begin
    sign_date = (
      select first 1 eds_status_datetime
      from edoc_sign
      where eds_from_edl_id = :edl_id
        and eds_status = 1
      order by eds_id desc);

    if (coalesce(add_params, '') <> '') then
      qr_code = f_edoc_qr_by_edl(f_json_str_get_value(add_params, 'edl_id', null), :c_id);

    suspend;
  end
end
