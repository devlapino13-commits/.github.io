CREATE PROCEDURE CM_KG_0171_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable point d_integer;
declare variable kc_amount_approved curr_type;
declare variable k_period d_integer;
declare variable k_disb_date d_date;
declare variable pr_code d_str_100;
begin
  select
    k_period,
    k_disb_date,
    pr_code
  from kontract k
  inner join product pr on pr.pr_id = k.k_from_pr_id
  where k.k_id = :k_id and
  pr.pr_code containing 'MBULAK_CARDS'
  into
    :k_period,
    :k_disb_date,
    :pr_code;

  if (row_count = 1) then
  begin
    res_error_is_critical = 1;
    for
      select
        f_point(kc_from_c_id, :k_disb_date),
        kc_amount_approved
      from knclient
      where kc_from_k_id = :k_id
        and kc_amount_approved > 0
      into
        :point,
        :kc_amount_approved
    do
    begin
      if (point <= 3) then
      begin
        if (kc_amount_approved > 80000) then
        begin
          res_error_text = 'У клиента балл меньше 3, поэтому он не может получить сумму больше 80 000 сом';
          suspend;
        end

        if (k_period > 18) then
        begin
          res_error_text = 'У клиента балл меньше 3, поэтому он не может получить заем на срок больше 18 месяцев';
          suspend;
        end
      end
      else
      if (point between 4 and 9) then
      begin
        if (kc_amount_approved > 100000) then
        begin
          res_error_text = 'По данному продукту максимально допустимая сумма 100 000 сом, поэтому клиент не может получить сумму больше 100 000 сом';
          suspend;
        end

        if (k_period > 18) then
        begin
          res_error_text = 'По данному продукту максимальный период 18 месяцев';
          suspend;
        end
      end
      if (point between 10 and 19) then
      begin
        if (kc_amount_approved > 150000) then
        begin
          res_error_text = 'По данному продукту максимально допустимая сумма 150 000 сом, поэтому клиент не может получить сумму больше 100 000 сом';
          suspend;
        end

        if (k_period > 18) then
        begin
          res_error_text = 'По данному продукту максимальный период 18 месяцев';
          suspend;
        end
      end
      else
      if (point >= 20) then
      begin
        if (kc_amount_approved > 200000) then
        begin
          res_error_text = 'Максимальная сумма займа 200 000 сом';
          suspend;
        end

        if (k_period > 24) then
        begin
          res_error_text = 'Срок займа не может превышать 24 месяцев';
          suspend;
        end
      end
    end
  end
end
