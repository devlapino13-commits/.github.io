CREATE PROCEDURE AAREP_REPORT_FOR_RKL(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    C_NAME_FIO STR_100,
    C_BIRTH_DATE D_DATE,
    CITIZEN STR_100,
    DOC_NO STR_100,
    DOC_DATE D_DATE,
    K_DISB_DATE D_DATE,
    K_KONTRACT_NO STR_100,
    REP_ACTIVE_BALANCE CURR_TYPE,
    AAREP_K_DELIQ_DAYS D_INTEGER,
    KF_REGION_NAME STR_100,
    KF_OFFICER_NAME STR_100,
    COUNTER D_INTEGER)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable kf_id d_id;
begin
  counter = 0;
  for select aarep_k_id, aarep_k_balance, aarep_k_deliq_days
      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 3)
      into k_id, rep_active_balance, aarep_k_deliq_days
  do
  begin
    select k_kontract_no, k_disb_date, k_from_kf_id
    from kontract
    where k_id = :k_id
    into k_kontract_no, k_disb_date, kf_id;

    select reg_name, c_name_fio
    from knaktlist kal
    inner join region on reg_unicode = kal_new_reg_id
    inner join clients on c_id = kal_new_kf_officer
    where kal_from_kf_id = :kf_id and
          kal_date_end = '31.12.9999' and
          kal.is_delete = 0
    into kf_region_name, kf_officer_name;

    doc_no = null;
    doc_date = null;
    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id
    do
    begin
      select c_name_fio, c_birth_date, rd_name
      from clients
      inner join refdata on rd_id = c_citizen
      where c_id = :c_id
      into c_name_fio, c_birth_date, citizen;

      for select cds_doc_no, cds_doc_date_begin
          from clientdocs
          where cds_from_c_id = :c_id and
                cds_is_main_doc = 1 and
                cds_is_delete = 0
          into doc_no, doc_date
      do
      begin
        if (aarep_k_deliq_days <= 120) then
        begin
          counter = counter + 1;
          suspend;
        end
      end
    end
  end
end
