CREATE PROCEDURE HP_ACTIONLIST_CREATE_CHECK_INFO(
    C_ID D_ID,
    REG_ID D_INTEGER,
    IS_CHECK_ERROR D_SMALLINT,
    UN_ID D_ID)
AS
declare variable rd_id d_integer;
declare variable rd_id_err d_integer;
declare variable acl_id d_id;
begin
  -- отправляет клиента в очередь, для проверки инфо от внешних источников сотрудниками КК


  rd_id = f_get_rd_id('ActionGroup.Check_Info');
  rd_id_err = f_get_rd_id('ActionGroup.Check_Info_error');

  if (rd_id is null) then exit;

  select aco_id
  from actionorder
  where aco_object_kind = f_objkind('clients', 'id') and
        aco_object_id = :c_id and
        aco_from_rd_id = :rd_id
  into acl_id;

  if (acl_id is null) then
  begin
    if (exists(select *
    from actionorder
    where aco_object_kind = f_objkind('clients', 'id') and
          aco_object_id = :c_id and
          aco_from_rd_id = :rd_id_err)) then exit;

    insert into actionlist (acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                            acl_respons_c_id, acl_from_un_id, acl_dont_show_in_order)
    values (current_date, :reg_id, f_objkind('clients', 'id'), :c_id,
            iif(:is_check_error = 0, :rd_id, :rd_id_err), 0, null, :un_id, 0);
  end else
  begin
    if (is_check_error = 1) then
    begin
      update actionlist
      set acl_from_rd_id = :rd_id_err
      where acl_id = :acl_id;

      update actionorder
      set aco_from_rd_id = :rd_id_err
      where aco_id = :acl_id;
    end
  end

end
