CREATE PROCEDURE HP_EFFECT_RATE_TO_DATE(
    K_ID D_ID,
    REP_DATE D_DATE,
    APPLY_NEW_METHOD D_SMALLINT = 0)
RETURNS (
    EFFECT_RATE DECIMAL(18,3),
    EFFECT_RATE_COMMON DECIMAL(18,3))
AS
declare variable db_id d_integer;
declare variable ol_id d_id = null;
declare variable new_float d_bool;
declare variable sh_id d_integer;
declare variable sh_date d_date;
declare variable sh_all d_currency;
declare variable sh_days d_integer;
declare variable sh_date_prev d_date;
begin
  -- Заполняем временную таблицу
  select ok_from_ol_id
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_ol_date <= :rep_date
  order by ok_ol_date desc, ok_from_ol_id desc
  rows 1
  into ol_id;

  db_id = f_get_db_id();
  new_float = 0;

  if (db_id = 81001 and rep_date >= '08.09.2021') then
  begin
    if (exists(select * from kontstate, kontract, product
    where ks_from_k_id = :k_id and
          ks_date_begin = :rep_date and
          ks_state = 1 and
          ks_from_ol_id = :ol_id and
          ks_ol_is_delete = 0 and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_percent_kind = 1)) then
    new_float = 1;
  end

  if (new_float = 1) then
  begin
    delete from temp_schedule
    where 1 = 1;

    for select 0, min(sh_date), -sum(sh_os - sh_kommis)
    from hp_get_base_schedule(:k_id)
    union
    select sh_id, sh_date, sh_os + sh_perc
    from hp_get_base_schedule(:k_id)
    where sh_id > 0
    into :sh_id, :sh_date, :sh_all
    do
    begin
      if (sh_date_prev is null) then
        sh_days = 0;
      else
        sh_days = sh_date - sh_date_prev;
  
      insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
      values (:sh_id, :sh_date, :sh_days, :sh_all);
      sh_date_prev = sh_date;
    end
  end else
    execute procedure hp_effect_rate_fill_tmp(:k_id, :ol_id, :apply_new_method);

  select calc_eff
  from hp_effect_rate_kg(:k_id)
  into effect_rate_common;
  if (db_id in (39801, 41701)) then
    effect_rate = effect_rate_common;

  if (db_id = 81001) then
    select calc_eff
    from hp_effect_rate_ru(:k_id)
    into effect_rate;

  if (db_id = 94401) then
    select calc_eff
    from hp_effect_rate_az(:k_id)
    into effect_rate;
  suspend;
end
