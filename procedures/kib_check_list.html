CREATE PROCEDURE KIB_CHECK_LIST(
    STEP_ D_SMALLINT,
    IS_EMPLOYEE D_BOOL = 0,
    IS_ONLINE D_BOOL = 0)
RETURNS (
    C_ID DECIMAL(18,0),
    NAME VARCHAR(100),
    FNAME VARCHAR(100),
    LNAME VARCHAR(100),
    PATRONYMIC VARCHAR(100),
    SEX_CODE VARCHAR(10),
    BIRTHD DATE,
    BIRTHD_STR VARCHAR(10),
    COUNTRY_CODE VARCHAR(10),
    PASSN VARCHAR(30),
    SOCC VARCHAR(50),
    OKPO VARCHAR(50),
    C_DOC_TYPE INTEGER,
    C_TYPE SMALLINT,
    KBM_ID DECIMAL(18,0),
    KBM_PR_ID VARCHAR(255),
    KBM_VER_INFO SMALLINT,
    KBM_IS_CHECKED SMALLINT,
    KBM_ERR_TEXT VARCHAR(255),
    KBM_PERSON_ID INTEGER,
    KBM_DATE_TIME TIMESTAMP)
AS
declare variable C_NAME_FIO varchar(255);
declare variable C_SEX integer;
declare variable KK integer;
declare variable ST_STAFF integer;
declare variable ST_CLIENT integer;
declare variable DO_SUSPEND smallint;
declare variable ERROR_TMP STR_255;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  is_employee = coalesce(is_employee, 0);

  country_code = 'kgz';
  if ((step_ is null) or (step_ = 0)) then
    step_ = 1;

  for select kbm_id
      from kib_check_list_tmp(:step_, :is_employee, :is_online)
      into :kbm_id
  do
  begin
    select kbm.kbm_from_c_id, kbm.kbm_pr_id, kbm.kbm_ver_info, kbm.kbm_is_checked, kbm.kbm_err_text, kbm.kbm_datetime
    from kibminiapp kbm
    where kbm.kbm_id = :kbm_id
    into :c_id, :kbm_pr_id, :kbm_ver_info, :kbm_is_checked, :kbm_err_text, :kbm_date_time;

    if (((kbm_date_time >= current_date - 10) and
        (coalesce(kbm_err_text, '') <> '')) or (coalesce(kbm_err_text, '') = '')) then
    begin
      select c.c_name_fio, c.c_sex, c.c_birth_date, c_inn, c_okpo, c_doc_type, c_pass_no, c.c_type,
             coalesce(c_kib_id, 0)
      from clients c
      where c.c_id = :c_id
      into :c_name_fio, :c_sex, :birthd, :socc, :okpo, :c_doc_type, :passn, :c_type, :kbm_person_id;

      birthd_str = f_datetostr(birthd);

      c_name_fio = replace(c_name_fio, 'ң', 'н');
      c_name_fio = replace(c_name_fio, '�©', 'о');
      c_name_fio = replace(c_name_fio, 'ү', 'у');

      name = c_name_fio;
      if (socc is null) then
        socc = '';
      fname = '';
      lname = '';
      patronymic = '';
      c_name_fio = (:c_name_fio);
      kk = position(' ', :c_name_fio) - 1;
      if (kk > 0) then
      begin
        lname = substring(:c_name_fio from 1 for kk);
        c_name_fio = substring(:c_name_fio from kk + 2 for 255);
        kk = position(' ', c_name_fio) - 1;
        if (kk > 0) then
        begin
          fname = substring(:c_name_fio from 1 for kk);
          c_name_fio = substring(:c_name_fio from kk + 2 for 255);
          patronymic = :c_name_fio;
        end
        else
          fname = :c_name_fio;
      end
      if (c_sex is null) then
        sex_code = 'Н';
      if (c_sex = 0) then
        sex_code = 'Ж';
      if (c_sex = 1) then
        sex_code = 'М';
      fname = trim(fname);
      lname = trim(lname);
      patronymic = trim(patronymic);

      if (((step_ = 1) and
          (coalesce(kbm_pr_id, '') = '') and
          (kbm_is_checked = 0)) or ((step_ = 2) and
          (kbm_is_checked = 0))) then
      begin
        do_suspend = 1;

        if (kbm_err_text in ('Дата рождения не соответствует ИНН', 'ИНН должен быть 14 символов', 'ИНН не указан')) then
        begin
          do_suspend = 0;

          select error_text
          from hp_inn_check(:socc, :birthd, :c_sex)
          into :error_tmp;

          if (error_tmp = '') then
            do_suspend = 1;
        end
        if (do_suspend = 1) then
          suspend;
      end
    end
  end
end
