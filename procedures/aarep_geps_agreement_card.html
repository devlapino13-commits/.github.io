CREATE PROCEDURE AAREP_GEPS_AGREEMENT_CARD(
    REP_REGION_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_100,
    K_DISB_DATE D_DATE,
    C_NAME_FIO D_STR_100,
    KONT_STATUS D_STR_100,
    CLIENT_STATUS D_STR_100,
    CATEG D_STR_100,
    GS_AMOUNT D_CURRENCY,
    PREDPISANIYE_STATUS D_STR_100,
    HAS_ASP SMALLINT,
    KS_FIO D_STR_100,
    KNC_FULL_ADDRESS D_STR_255)
AS
declare variable CONFIG D_STR_100;
declare variable EDC_ID D_INTEGER;
declare variable C_ID D_ID;
declare variable KC_ID D_ID;
declare variable KAL_NEW_KF_OFFICER D_ID;
declare variable EDOC_STATUS D_SMALLINT;
begin
  predpisaniye_status = null;
  edoc_status = null;
  for select reg.reg_name,
             office.reg_name,
             kl.kal_new_kf_name,
             kt.k_kontract_no,
             kt.k_disb_date,
             cl.c_name_fio,
             iif(ks.ks_state = 1, 'Активный', 'Закрытый'),
             gs.gs_amount,
             gs_from_edc_id,
             cl.c_id,
             kc_id,
             kal_new_kf_officer
      from geps_agreement_card gs
      inner join kontract kt on kt.k_id = gs.gs_k_id
      inner join knaktlist kl on kl.kal_from_kf_id = kt.k_from_kf_id
      inner join region_list(:rep_region_id, 1, 3, 1) rl on cast(rl.id_list as d_id) = kl.kal_new_reg_id
      inner join region office on office.reg_unicode = rl.id_list
      inner join region reg on reg.reg_unicode = office.parent
      inner join knclient kncl on kncl.kc_id = gs.gs_from_kc_id
      inner join clients cl on cl.c_id = kncl.kc_from_c_id
      inner join kontstate ks on ks.ks_from_k_id = kt.k_id
      where kl.is_delete = 0 and
            :rep_date_end between kl.kal_date and kl.kal_date_end and
            :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_ol_is_delete = 0
      into :reg_name,
           :office_name,
           :kf_name,
           :k_kontract_no,
           :k_disb_date,
           :c_name_fio,
           :kont_status,
           :gs_amount,
           :edc_id,
           :c_id,
           :kc_id,
           :kal_new_kf_officer
  do
  begin
    ks_fio = '';

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :kal_new_kf_officer
    into :ks_fio;

    has_asp = 0;
    if (exists(select *
               from hp_edoc_status('asp.agreement', 0, :c_id)
               where client_sign_status = 1)) then
      has_asp = 1;

    categ = 'не определено';
    config = '';

    if (edc_id = 34) then
    begin
      categ = 'Доп_согл_ГЭПС_Карта_Стандарт';
      config = 'geps_agreement_card_standart';
    end
    else
    if (edc_id = 35) then
    begin
      categ = 'Доп_согл_ГЭПС_Карта_Ишеним';
      config = 'geps_agreement_card_ishenim';
    end

    -- cds_date = null;
    if (edc_id is not null) then
    begin
      select edoc_status_str,
             edoc_status
      from hp_edoc_status(:config, 8, :kc_id)
      into :predpisaniye_status,
           :edoc_status;

      if (edoc_status = -2) then
      begin
        predpisaniye_status = 'не сформирован т.к номер не верифицирован';
      end

      if (:edoc_status <> 1) then
      begin
        if (exists(select *
                   from clientdocs cds, passkind, passdoctype
                   where cds_from_c_id = :c_id and
                         cds_from_pk_id = pk_id and
                         pk_from_pdt_id = pdt_id and
                         cds_is_delete = 0 and
                         pdt_code containing 'geps_sog_card')) then
        begin
          predpisaniye_status = 'Подписан вручную';

          /*  select cds.cds_create_datetime
            from clientdocs cds
            inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
            inner join passdoctype pdt on pk.pk_from_pdt_id = pdt_id
            where cds.cds_from_c_id = :c_id and
                  cds.cds_is_delete = 0 and
                  pdt_code containing 'geps_sog_card'
            order by cds.cds_create_datetime
            rows 1
            into :cds_date;  */
        end
      end
    end
    -- BOARD-4897
    client_status = '';
    if(exists(select cds.cds_doc_blob_1
               from clientdocs cds, passkind pk
               where cds.cds_from_c_id = :c_id and
                     cds.cds_from_pk_id = pk.pk_id and
                     cds.cds_is_delete = 0 and
                     pk.pk_id = 301 and
                     :rep_date_end between cds.cds_doc_date_begin and coalesce(cds.cds_doc_date_end, '31.12.9999'))) then
    client_status = 'смерть клиента/умер';

    knc_full_address = null;
    select adl.adl_address_str
    from addrclient adc
    inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
    where adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 1
    into knc_full_address;
    suspend;
  end
end
