CREATE PROCEDURE HP_REGISTER_LAST_CLIENT(
    C_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0),
    KIND INTEGER)
AS
declare variable last_c_id decimal(18,0);
declare variable last_id integer;
declare variable last_kind integer;
begin
  select cul_from_c_id, cul_id, cul_kind
  from clientlast
  where cul_from_un_id = :un_id
  order by cul_id desc
  rows 1
  into :last_c_id, :last_id, :last_kind;
  if (last_c_id is null) then
    last_c_id = 0;
  if (last_c_id <> c_id) then
  begin
    insert into clientlast (cul_from_c_id, cul_from_un_id, cul_id, cul_kind)
    values (:c_id, :un_id, -1, :kind);
  end
  else
  begin
    if (kind = 1) then
    begin
      update clientlast
      set cul_datetime = cast('NOW' as timestamp),
          cul_kind = 1
      where cul_from_un_id = :un_id and
            cul_id = :last_id;
    end
    else
    begin
      update clientlast
      set cul_datetime = cast('NOW' as timestamp)
      where cul_from_un_id = :un_id and
            cul_id = :last_id;
    end
  end
end
