declare function del_twice_space(str d_str_255)
returns d_str_255
as
begin
str = replace(str, ascii_char(9), ' ');
str = replace(str, '\', '');
str = replace(str, 'ʻ', '');
str = f_replace_all(str, ' -', '-');
str = f_replace_all(str, '- ', '-');
str = f_replace_all(str, '  ', ' ');

return trim(str);
end

begin

from miniappclient mac
where mac.mac_from_ma_id = :object_id
and mac.mac_amount > 0
into :c_id
do

end
else if (object_kind = 0) then

begin
--Процедура формирует событие 3.1. при изменении показателей
--групп [FL_1_4_Group – FL_9_ AddrFact] в связи с выявленной ошибкой RFCRU-4887

  for select c_id from myhp_get_c_id(:object_kind, :object_id)
  into c_id
  do
  begin
    select max(coalesce(bev_event_date, cast('01.01.1900' as d_date))), max(bev_event_id = 3), max(bev_event_id = 5)
    from bki_packet_type, bki_packet, bki_loans,bki_clients, bki_events
    where bkp_from_bkt_id = bkt_id
      and bkl_from_bkp_id = bkp_id
      and bkc_from_bkl_id = bkl_id
      and bev_from_bkc_id = bkc_id
      and bkt_format = 0
      and bkp_from_bkk_id = :bkk_id
      and bkc_from_c_id = :c_id
      and bkp_status > 0
      and bev_status > 0
      and coalesce(bev_event_id, 0) <> 0
      and bkp_date < :rep_date
    into last_event_date, sended_1_4, sended_1_9;
  
    if (coalesce(last_event_date, '01.01.1900') = '01.01.1900') then
      continue;
  
    event_id = f_enum('bki_event', '3.1_correction_title', 'id');

    if (object_kind = 9) then
      select new_bev_id
      from hp_bki_corr_add_loan(:object_id, :c_id, :event_id, :rep_date, :bkk_id, null, null)
      into :new_bev_id;
    else
      select new_bev_id
      from hp_bki_corr_add_loan(null, :c_id, :event_id, :rep_date, :bkk_id, null, :object_id)
      into :new_bev_id;

    for select bfv_id, xml_node
    from hp_bki_parameters_list(:new_bev_id, 0) p
    where p.parent_id is not null
    into :bfv_id, :xml_node
    do
    begin
      if (xml_node starting with 'FL_') then
      begin
        number = null;
        number = cast (substring (xml_node from 4 for position('_' in substring(xml_node from 4)) - 1) as integer);
        if (number between 1 and 9) then
          execute procedure hp_bki_corr_set_parametr(:new_bev_id, :bfv_id, 1);
      end
    end
  
    execute procedure hp_bki_corr_fill_values(:new_bev_id, 3, 0);
    execute procedure hp_bki_corr_fill_values(:new_bev_id, 4, 0);  
  
    for select bfv_id
    from hp_bki_parameters_list(:new_bev_id, 0)
    where db_field = 'hist_name_sign' or db_field = 'hist_doc_sign'
    into :bfv_id
    do
      execute procedure hp_bki_corr_set_value(:bfv_id, :new_bev_id, 0, 0);
  
    for select pif_field_name, pif_source, pif$id_val(pda_value_id), pva_datetime
    from perslist pls, persvalue pva, persdata pda, ux$persfield pif
    where pls_from_c_id = :c_id
      and pls_datetime between :rep_date and :rep_date + 1
      and pls_edit_kind = 0 -- исправление ошибок
      and pva_from_pls_id = pls_id
      and pda_id = pva_from_pda_id
      and pif_uuid = pda_from_pif_uuid
    into pif_field_name, pif_source, new_val, pva_datetime
    do
    begin
      old_val = null;

      select pva_text_value
      from hp_persinfo_loglist(:c_id, :pif_source, :pif_field_name)
      where pva_datetime < :pva_datetime
      order by pva_is_signed desc, pva_datetime desc
      rows 1
      into old_val;
  
      if (new_val is distinct from old_val) then
      begin
        if (:pif_field_name = 'c_citizen_str') then  --4.1
        begin
          select rd_memo
          from refdata
          where rd_main_code = 'CountryOKSM'
            and (rd_name =  case :old_val
                              when 'Кыргызстан' then 'Киргизия'
                              when 'Республика Молдова' then 'Молдова, Республика'
                              else :old_val
                            end)
          into doc_country;
  
          if (doc_country is null) then
            doc_country = '999';
          
          f_set_old_value(get_param_id(:new_bev_id, 'doc_country'), :new_bev_id, :doc_country);
        end
  
        c_doc_type = null;
        select c_doc_type, c.c_birth_date
        from clients c
        where c.c_id = :c_id
        into :c_doc_type, c_birth_date;
  
        select pk_country_id
        from passkind
        where pk_id = :c_doc_type
        into :c_citizen;
  
        if ((:pif_field_name = 'c_name_fio' and :c_citizen = 13) or (:pif_field_name = 'c_name_eng' and :c_citizen <> 13)) then
          begin
            select del_twice_space(name_last), del_twice_space(name_first), del_twice_space(name_middle)
            from hp_fio_clear_ru(:old_val)
            into name_last, name_first, name_middle;
  
            f_set_old_value(get_param_id(:new_bev_id, 'name_last'), :new_bev_id, :name_last);--1.1
            f_set_old_value(get_param_id(:new_bev_id, 'name_first'), :new_bev_id, :name_first);--1.2
            f_set_old_value(get_param_id(:new_bev_id, 'name_middle'), :new_bev_id, :name_middle);--1.3
          end
        
        if (:pif_field_name = 'c_doc_type') then
        begin
          f_set_old_value(get_param_id(:new_bev_id, 'doc_type'), :new_bev_id, :old_val);--4.3
          f_set_old_value(get_param_id(:new_bev_id, 'foreigner_code'), :new_bev_id, trim(iif(:old_val in ('21', '22.1'), 3, 1)));--4.11
          c_doc_type = :old_val;
        end
  
        if (:pif_field_name starting with 'c_pass_') then
        begin
          if (:pif_field_name = 'c_pass_no') then  --4.5
          begin
            doc_number = replace(old_val, ' ', '');
            doc_number = replace(doc_number, '-', '');
            doc_serial = null;
  
            serial_length = case :c_doc_type
                              when 12  then 4 -- Паспорт гражданина РФ
                              when 13  then 2 -- Загран. паспорт РФ
                              when 17  then 2 -- Паспорт УЗ (зелёный)
                              when 330 then 2 -- Биометрический паспорт УЗ (бордовый)
                              when 15  then 1 -- Паспорт Тадж.
                              else 0
                            end;
  
            if (char_length(doc_number) > serial_length and serial_length > 0) then
            begin
              doc_serial = upper(left(doc_number, serial_length));
              doc_number = upper(substring(doc_number from serial_length + 1));
            end
             
            f_set_old_value(get_param_id(:new_bev_id, 'doc_serial'), :new_bev_id, :doc_serial);--4.5
            f_set_old_value(get_param_id(:new_bev_id, 'doc_number'), :new_bev_id, :doc_number);--4.6
          end
  
          if (:pif_field_name = 'c_pass_date') then    --4.7
            f_set_old_value(get_param_id(:new_bev_id, 'doc_date'), :new_bev_id, f_date_format('yyyy-mm-dd', :old_val));
          if (:pif_field_name = 'c_pass_who') then    --4.8
          begin
            old_val = upper(trim(:old_val));
            select trim(out_string)
            from hp_check_characters_str(:old_val, '[a-zA-Zа-яА-ЯЁё0-9@- .,'';\/№():_+|"#?!;]*')
            into old_val;
            
            if (coalesce(old_val, '') = '') then
              old_val = '-';
            f_set_old_value(get_param_id(:new_bev_id, 'doc_who'), :new_bev_id, :old_val);
          end
          if (:pif_field_name = 'c_pass_who_code') then    --4.9
          begin
            if (:c_doc_type = 12) then
            begin
              old_val = replace(old_val, ' ', '');  -- Код подразделения
              
              if (old_val not similar to '[0-9]{3}\-[0-9]{3}' escape '\') then
              begin -- если Код подразделения не подходит по маске, то попытаемся его привести к нужному формату
                select out_string
                from hp_check_characters_str(:old_val, '[0-9]*')
                into old_val;
                
                if (char_length(old_val) = 6) then  -- если цифр достаточно
                  old_val = left(old_val, 3) || '-' || substring(old_val from 4);
                else
                  old_val = trim(iif(current_date < '18.02.2024', '999-999', null)); -- заглушка
              end
            end
            else
              old_val = null;
            f_set_old_value(get_param_id(:new_bev_id, 'doc_department_code'), :new_bev_id, :old_val);
          end
          if (:pif_field_name = 'c_pass_date_expire') then  --4.10
            f_set_old_value(get_param_id(:new_bev_id, 'doc_end_date'), :new_bev_id,
              iif(coalesce(trim(old_val), '') = '', null, f_date_format('yyyy-mm-dd', :old_val)));
        end
  
        if (:pif_field_name = 'c_birth_date') then    --3.1
          f_set_old_value(get_param_id(:new_bev_id, 'birth_date'), :new_bev_id, f_date_format('yyyy-mm-dd', :old_val));
--        if (:pif_field_name = 'c_birth_country') then    --3.2
--        begin
--          select rd_memo from refdata
--          where rd_main_code = 'CountryOKSM'
--            and (rd_name = case f_my_fix_wrong_cyr(:old_val)
--                              when 'Кыргызстан' then 'Киргизия'

--                              when 'Убекистан' then 'Узбекистан'
--                              when 'Республика Узбекистан' then 'Узбекистан'
--                              when 'Республика Таджикистан' then 'Таджикистан'
--                              else f_my_fix_wrong_cyr(:old_val)
--                           end)
--          into old_val;
--        
--          old_val = coalesce(old_val, '999');
--          f_set_old_value(get_param_id(:new_bev_id, 'birth_country'), :new_bev_id, :old_val);
--        end
  
        if (:pif_field_name = 'c_inn') then  --6.2
          f_set_old_value(get_param_id(:new_bev_id, 'inn_no'), :new_bev_id, :old_val);
        
        if (pif_field_name in ('address_legal', 'address_fact') and sended_1_4 and sended_1_9) then
        begin
          select full_text from hp_str_clear(:old_val)
          into :full_addr_text;
  
          if (:full_addr_text is null) then
          begin
            update bki_events
            set bev_status = -2
            where bev_id = :new_bev_id;
  
            error_text = 'Не найден адрес: ' || :old_val;
            insert into bki_errors (bke_from_bev_id, bke_date, bke_error_code, bke_error_id)
            values (:new_bev_id, current_date, 'DB', f_memo_id(:error_text));
            exit;
          end
  
          a_fias = null;       -- Номер адреса в ФИАС
          a_okato = null;      -- Код населенного пункта по ОКАТО
          a_other_statement = null; --  Иной населенный пункт
          a_index = null;      -- Почтовый индекс
          a_street = null;     -- Улица
          a_house = null;      -- Дом

          a_block = null;      -- Корпус

                 replace(upper(adl_section_no), ' ', ''), replace(upper(adl_building_no), ' ', ''), replace(upper(adl_vladen_no), ' ', ''),
                 adl_street_id, adl_from_pl_id, adl_post_index, substring(cast(adl_id as str_255) from 1 for 3),
                 add_registration_date, upper(add_authority_code), upper(add_authority_name)
          from addrlist adl
          left join addrdetails on add_adl_id = adl.adl_id
          where hash(adl.adl_indexed_address) = hash(:full_addr_text)
          order by adl.adl_id desc
          rows 1
          into a_apartment, a_house, a_block, a_build, a_domain, adl_street_id,
               adl_from_pl_id, a_index, republ_code,
               ar_reg_date, ar_reg_department_code, ar_reg_place;
  
          if (a_apartment = '0') then a_apartment = null;
          if (a_house = '0') then a_house = null;
          if (a_block = '0') then a_block = null;
          if (a_build = '0') then a_build = null;
          if (a_domain = '0') then a_domain = null;
  
          a_country = (select rd2.rd_memo
                from refdata rd1, refdata rd2
                where rd1.rd_str_id = :republ_code
                  and rd1.rd_main_code = 'Citizen'
                  and rd2.rd_main_code = 'CountryOKSM'
                  and rd2.rd_name = case rd1.rd_name
                                        when 'Кыргызстан' then 'Киргизия'
                                        when 'Республика Молдова' then 'Молдова, Республика'
                                        else rd1.rd_name
                                     end);
  
          a_country_text = '';
          if (coalesce(a_country, '') = '') then
          begin
            a_country = '999';
            a_country_text = (select upper(rd_name)
                              from refdata rd
                              where rd_str_id = :republ_code
                                and rd_main_code = 'Citizen');
        
            if (coalesce(a_country_text, '') = '') then
              a_country_text = '-';
           end
  
          select upper(pl.pl_name), trim(pl.pl_post_index), trim(pl.pl_add_code)
          from places pl
          where pl.pl_id = :adl_from_pl_id
          into a_other_statement, a_index, a_okato;
  
          if (a_index not similar to '[0-9]{6}') then
            a_index = null;
        
          if (coalesce(a_okato, '') = '') then
            a_okato = '99999999999';
        
          if (a_okato <> '99999999999') then
            a_other_statement = null;
      
          select del_twice_space(upper(pl.pl_name)),
            trim(iif(trim(coalesce(pl_post_index, '')) not similar to '[0-9]{6}', :a_index, pl_post_index))  -- по умолчанию наибольший приоритет отдаю почтовому индексу, указанному на уровне улицы
          from places pl
          inner join placetype pt on pl.pl_status = pt.plt_id
          where pl.pl_id = :adl_street_id
          into a_street, a_index;
        
          select del_twice_space(out_string)
          from hp_check_characters_str(:a_street, '[а-яА-ЯЁё0-9- ]*')
          into a_street;
  
          if (:pif_field_name = 'address_legal') then
          begin
            ar_reg_date = null;
            a_reg_code = 1;
            begin
              select cds_doc_date_begin, cds_id
              from clientdocs cds, passkind pk, passdoctype pdt
              where pk_id = cds_from_pk_id
                and pdt_id = pk_from_pdt_id
                and cds_from_c_id = :c_id
                and cds_doc_date_begin <= :rep_date
                and coalesce(cds_delete_datetime, current_timestamp) >= cast(:rep_date as date)  -- Не аннулирован
                and cds_verify_status = 2  -- подтвержденный
                and pdt.pdt_code in ('registration_permanent', 'registration_fiz', 'registration_yur')
              order by cds_create_datetime desc
              rows 1
              into ar_reg_date, cds_id;

              select rd_tag
              from addresstype_registration
              inner join refdata rd on rd.rd_id = atr_from_rd_id
              where atr_from_cds_id = :cds_id
              order by atr_datetime desc
              rows 1
              into a_reg_code;
            end
            
            f_set_old_value(get_param_id(:new_bev_id, 'ar_reg_code'), :new_bev_id, a_reg_code);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_index'), :new_bev_id, :a_index);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_country'), :new_bev_id, :a_country);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_country_text'), :new_bev_id, :a_country_text);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_fias'), :new_bev_id, :a_fias);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_okato'), :new_bev_id, :a_okato);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_other_statement'), :new_bev_id, :a_other_statement);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_street'), :new_bev_id, :a_street);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_house'), :new_bev_id, :a_house);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_domain'), :new_bev_id, :a_domain);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_block'), :new_bev_id, :a_block);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_build'), :new_bev_id, :a_build);
            f_set_old_value(get_param_id(:new_bev_id, 'ar_apartment'), :new_bev_id, :a_apartment); 
            --f_set_old_value(get_param_id(:new_bev_id, 'ar_reg_department_code'), :new_bev_id, :ar_reg_department_code);
            --f_set_old_value(get_param_id(:new_bev_id, 'ar_reg_place'), :new_bev_id, :ar_reg_place);
      
            if (ar_reg_date > coalesce(:c_birth_date, '01.01.1900')) then
              f_set_old_value(get_param_id(:new_bev_id, 'ar_reg_date'), :new_bev_id, :ar_reg_date);
          end
          else
          begin
            f_set_old_value(get_param_id(:new_bev_id, 'af_sign'), :new_bev_id, 1);
            f_set_old_value(get_param_id(:new_bev_id, 'af_index'), :new_bev_id, :a_index);
            f_set_old_value(get_param_id(:new_bev_id, 'af_country'), :new_bev_id, :a_country);
            f_set_old_value(get_param_id(:new_bev_id, 'af_country_text'), :new_bev_id, :a_country_text);
            f_set_old_value(get_param_id(:new_bev_id, 'af_fias'), :new_bev_id, :a_fias);
            f_set_old_value(get_param_id(:new_bev_id, 'af_okato'), :new_bev_id, :a_okato);
            f_set_old_value(get_param_id(:new_bev_id, 'af_other_statement'), :new_bev_id, :a_other_statement);
            f_set_old_value(get_param_id(:new_bev_id, 'af_street'), :new_bev_id, :a_street);
            f_set_old_value(get_param_id(:new_bev_id, 'af_house'), :new_bev_id, :a_house);
            f_set_old_value(get_param_id(:new_bev_id, 'af_domain'), :new_bev_id, :a_domain);
            f_set_old_value(get_param_id(:new_bev_id, 'af_block'), :new_bev_id, :a_block);
            f_set_old_value(get_param_id(:new_bev_id, 'af_build'), :new_bev_id, :a_build);
            f_set_old_value(get_param_id(:new_bev_id, 'af_appartment'), :new_bev_id, :a_apartment);
          end
        end
      end
    end

    -- если изменений нет, то воизбежиние отправки пустого пакета, удалим созданную запись
    if (not exists(select 1 from bki_send_data_corr
                  where bsd_from_bev_id = :new_bev_id
                    and bsd_from_bkd_id is distinct from bsd_new_value_id))
    then
    begin
      bkc_id = null;
      bkl_id = null;
      bkp_id = null;

      select bkc_id, bkl_id, bkl_from_bkp_id
      from bki_events, bki_clients, bki_loans
      where bkc_id = bev_from_bkc_id
        and bkl_id = bkc_from_bkl_id
        and bev_id = :new_bev_id
      into bkc_id, bkl_id, bkp_id;

      delete from bki_send_data_corr
      where bsd_from_bev_id = :new_bev_id;

      delete from bki_events
      where bev_id = :new_bev_id;

      -- удалим вышестоящие записи, если пустые
      if ((select count(1) = 0 from bki_events where bev_from_bkc_id = :bkc_id)) then
      begin
        delete from bki_clients
        where bkc_id = :bkc_id;

        if ((select count(1) = 0 from bki_clients where bkc_from_bkl_id = :bkl_id)) then
        begin
          delete from bki_loans
          where bkl_id = :bkl_id;
  
          if ((select count(1) = 0 from bki_loans where bkl_from_bkp_id = :bkp_id)) then
          begin
            delete from bki_packet
            where bkp_id = :bkp_id;
          end
        end 
      end 
    end
  end
