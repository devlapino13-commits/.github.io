CREATE PROCEDURE FM_HTML_VERIFY_DETAILS(
    C_ID DECIMAL(18,0),
    UN_ID D_ID = null)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable c_verify_status           integer;
declare variable color_black               varchar(100);
declare variable color_yellow              varchar(100);
declare variable color_green               varchar(100);
declare variable color_red                 varchar(100);
declare variable res_color                 varchar(100);
declare variable str_tmp                   varchar(512);
declare variable db_id                     integer;
declare variable acl_id                    decimal(18,0);
declare variable acm_memo                  varchar(4000);
declare variable str_memo                  varchar(512);
declare variable json_key                  varchar(255);
declare variable json_value                varchar(255);
declare variable need_passport_second_page integer;
declare variable c_doc_type                integer;
declare variable c_citizen                 integer;
declare variable str_comment               varchar(255);
declare variable str_tmp_1                 d_str_100;
declare variable str_tmp_2                 d_str_100;
declare variable acl_status                d_smallint;
declare variable gc_status                 d_integer;
declare variable acl_status_photo          d_smallint;
declare variable gc_from_eil_id            d_id;
declare variable lang                      d_str_code;
declare variable cds_create_datetime       d_datetime;
declare variable grs_active                d_smallint;
declare variable c_pers_status             d_integer;
declare variable perstemp_active           d_bool;
declare variable c_blob_pass1_id           d_id;
declare variable c_blob_pass2_id           d_id;
declare variable c_pass_no                 d_str_code;
declare variable c_pass_date               d_date;
declare variable c_pass_who                d_str_255;
begin
  db_id = f_get_db_id();
  lang = f_get_lang_code(:un_id, null);
  grs_active = f_grs_active(:un_id);
  res_text = '';
  select c_verify_status, c_doc_type, c_citizen,
    c_blob_pass1_id, c_blob_pass2_id,
    c_pass_no, c_pass_date, c_pass_who
  from clients
  where c_id = :c_id
  into :c_verify_status, c_doc_type, c_citizen,
        c_blob_pass1_id, c_blob_pass2_id,
        c_pass_no, c_pass_date, c_pass_who;
  if (c_verify_status is null) then
    c_verify_status = 0;

  perstemp_active = f_tmp_const('use_perstemp');
  c_pers_status = c_verify_status;

  --RFCRU-4383
  if (db_id = 81001 and
        (((coalesce(c_blob_pass1_id, 0) = 0) and (coalesce(c_blob_pass2_id, 0) = 0)) or
         c_pass_no is null or
         c_pass_date is null or
         c_pass_who is null)) then
    c_verify_status = 0;

  -- если включен режим перс данных, получаем расширенный статус
  if (perstemp_active = 1) then
    c_pers_status = f_pers_status(c_id, c_verify_status);

  color_black = 'black';
  color_yellow = '#F08205';
  color_green = '#02B316';
  color_red = 'red';

  res_color = color_black;
  res_color = (case c_verify_status
                 when -2 then color_red
                 when -1 then color_black
                 when 0 then color_yellow
                 when 1 then color_yellow
                 when 2 then color_green
               end);

  if (db_id <> 94401) then
  begin
    str_tmp = (case c_verify_status
                 when -2 then 'Данные не верные'
                 when -1 then 'Верификация не требуется'
                 when 0 then 'Данные не полные'
                 when 1 then 'В очереди'
                 when 2 then 'Подтверждена'
               end);

    if (perstemp_active = 1) then
    begin
      -- если включен режим перс данных, получаем текст расширеннего статуса
      select res_caption
      from hp_perstemp_verify_info(:c_pers_status)
      into :str_tmp;
    end

    if (c_verify_status = 1 and
        (f_get_const('send_query_verify') = 1) and :grs_active <> 1) then
    begin
      select coalesce(acl_status, -9)
      from actionlist
      where acl_object_kind = 0 and
            acl_object_id = :c_id and
            acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO'))
      order by acl_datetime desc
      rows 1
      into acl_status;

      if (acl_status not in (0, 1)) then
        str_tmp = 'Запрос еще не отправлен';

    end

  end
  else
  begin
    str_tmp = (case c_verify_status
                 when -2 then 'Məlumatlar səhvdir'
                 when -1 then 'Verifikasiyaya lazım deyil'
                 when 0 then 'Məlumatlar tam deyil'
                 when 1 then 'Verifikasiya növbəsindədir'
                 when 2 then 'Təsdiqlənib'
               end);
  end

  res_text = BLOB_APPEND(res_text, f_format('<br>%1:<br>', iif(db_id <> 94401, 'Верификация персональных данных', 'Pasport məlumatların verifikasiyası')));
  str_tmp = '<font color="' || res_color || '">' || trim(:str_tmp) || '</font><br>';
  res_text = BLOB_APPEND(res_text, str_tmp);
  if (:db_id <> 81001 and
      coalesce(:un_id, 0) > 0 and
      :grs_active = 1) then
  begin
    str_tmp_1 = '';
    str_tmp_2 = '';
    select gc.gc_status, gc.gc_from_eil_id
    from grs_clients gc
    where gc.gc_from_c_id = :c_id and
          gc.gc_is_last = 1
          rows 1
    into gc_status, gc_from_eil_id;
    if(coalesce(gc_status, -3) <> -3 )then
    begin
    str_tmp_1 = case gc_status
                  when -2 then ''
                  when 0 then f_lang(lang, 'Паспорт проверяется в ГРС')
                  when 1 then f_lang(lang, 'Требуется проверка, изменены паспортные данные')
                  when 2 then f_lang(lang, 'Верификация паспорта в ГРС прошла успешно')
                  when 3 then f_lang(lang, 'Верификация паспорта в ГРС прошла успешно, но не получен юр.адрес, прошу заполнить юр.адрес')
                  else f_lang(lang, 'Требуется проверка, изменены паспортные данные')
                end;
    if (gc_status = 3 and exists(select * from addrclient adc
                                 where adc.adc_address_type = 1 and
                                 adc.adc_from_c_id = :c_id and
                                 adc.adc_from_adl_id <> 41701000000000)) then
     str_tmp_1 = f_lang(lang,'Верификация паспорта в ГРС прошла успешно, юр. адрес заполнен со стороны СпФ');
    if (gc_status = -2 and
        exists (select *
                from extinfoerr eir
                where eir.eir_from_eil_id = :gc_from_eil_id and
                      eir.eir_answer not starting 404)) then
      str_tmp_1 = f_lang(lang,'В ходе верификации возникла ошибка, отправьте клиента верификаторам');

    if (gc_status in (2, 3)) then
    begin
      select cds.cds_create_datetime
      from clientdocs cds
      where cds.cds_from_c_id = :c_id and
            cds.cds_is_delete = 0 and
            cds.cds_is_main_doc = 1
      order by cds.cds_id desc
      rows 1
      into cds_create_datetime;

      select acl.acl_status
      from actionlist acl
      where acl.acl_object_kind = f_objkind('clients', 'id') and
            acl.acl_object_id = :c_id and
            acl.acl_from_rd_id = f_get_rd_id('ActionGroup.Photo_Verify') and
            acl.acl_last_datetime > :cds_create_datetime
      order by acl.acl_datetime desc
      rows 1
      into acl_status_photo;

      if (coalesce(:acl_status_photo, 3) = 3 and :c_verify_status <> 2) then
        str_tmp_2 = f_lang(lang, 'Нужно загрузить фото клиента');
      else
        str_tmp_2 = case acl_status_photo
                      when 0 then f_lang(lang, 'Фото клиента в ожидании верификации')
                      when 1 then f_lang(lang, 'Фото клиента на проверке')
                      when 2 then f_lang(lang, 'Фото клиента успешно верифицирован')
                      when -2 then f_lang(lang, 'Фото не прошла верификацию')
                      else ''
                    end;

    end
    res_text = BLOB_APPEND(res_text, str_tmp_1 || '<br>' || str_tmp_2 || '<br>');
    suspend;
    exit;
    end
  end
  if (not(c_verify_status in (-2, 0))) then
  begin
    suspend;
    exit;
  end

  if (c_verify_status = -2) then
  begin
    str_memo = '';
    select acl_id
    from actionlist
    where acl_object_kind = 0 and
          acl_object_id = :c_id and
          acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO'))
    order by acl_id desc
    rows 1
    into :acl_id;

    select acm_memo
    from actionmemo
    where acm_from_acl_id = :acl_id
    into :acm_memo;

    str_comment = '';

    if (coalesce(acm_memo, '') <> '') then
    begin
      for select json_key, json_value
          from fm_json_parse(:acm_memo, 0, 'verify')
          into :json_key, :json_value
      do
      begin
        str_tmp = '';
        if ((json_value = '1') and
            (json_key <> 'comment')) then
        begin
          str_tmp = trim(case trim(json_key)
                           when 'fio' then iif(:db_id <> 94401, 'ФИО', 'ASA')
                           when 'sex' then iif(:db_id <> 94401, 'Пол', 'Cinsi')
                           when 'birth_date' then iif(:db_id <> 94401, 'ДР.', 'D.Tarixi')
                           when 'passno' then iif(:db_id <> 94401, 'Пасп.№', 'Passport №')
                           when 'passdate_disb' then iif(:db_id <> 94401, 'Дата выдачи', 'Verilmə tarixi')
                           when 'passkind' then iif(:db_id <> 94401, 'Тип документа и скан', 'Passport növü və skanı')
                           when 'passdate_expire' then iif(:db_id <> 94401, 'Срок действия', 'Bitmə tarixi')
                           when 'inn' then iif(:db_id <> 94401, 'ИНН', 'VOEN')
                           when 'photo' then iif(:db_id <> 94401, 'Фото', 'Şəkil')
                           when 'passport' then iif(:db_id <> 94401, 'Скан паспорта', 'Pasport skanı')
                           when 'passport_who' then iif(:db_id <> 94401, 'Место выдачи', 'Verən yer')
                           when 'birth_place' then iif(:db_id <> 94401, 'Место рождения', 'Dogum yeri')
                         end);
          if (str_memo <> '') then
            str_memo = str_memo || ', ' || str_tmp;
          else
            str_memo = str_tmp;
        end

        if (json_key = 'comment') then
        begin
          if (coalesce(json_value, '') <> '') then
            str_comment = iif(:db_id <> 94401, 'Комментарий', 'Şərh') || ': ' || :json_value || ';';
        end
      end
    end
    res_text = BLOB_APPEND(res_text, f_format('%1: <font color="', iif(:db_id <> 94401, 'Ошибки', 'Ошибки')) || color_red || '">' || trim(:str_memo) || '</font><br>' || :str_comment || '<br>');
  end

  if (c_verify_status = 0) then
  begin
    str_memo = '';
    str_tmp = '';

    select acl_id
    from actionlist
    where acl_object_kind = 0 and
          acl_object_id = :c_id and
          acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO')) and
          acl_status = -1
    order by acl_id desc
    rows 1
    into :acl_id;

    select acm_memo
    from actionmemo
    where acm_from_acl_id = :acl_id
    into :acm_memo;

    if (coalesce(acm_memo, '') <> '') then
    begin
      for select json_key, json_value
          from fm_json_parse(:acm_memo, 0, 'verify')
          into :json_key, :json_value
      do
      begin
        str_tmp = '';
        if (json_key = 'comment') then
          str_tmp = json_value;
      end
    end

    if (str_tmp <> '') then
      res_text = BLOB_APPEND(res_text, f_format('%1: <font color="', iif(:db_id <> 94401, 'Комментарий', 'Комментарий')) || color_red || '">' || trim(:str_tmp) || '</font><br>');

    need_passport_second_page = 1;
    if (exists(select *
               from passdoctype, passkind, refdata
               where pk_id = :c_doc_type and
                     pk_from_pdt_id = pdt_id and
                     rd_id = :c_citizen and
                     pdt_code = 'passport_trans' -- Загран.паспорт
                     and
                     (rd_code = 'Citizen.KG' -- Кыргызстан
                     or
                     rd_code = 'Citizen.TJ'))) -- Таджикистан
    then
      need_passport_second_page = 0; -- для этих паспортов вторая часть необязательна

    select trim(iif(coalesce(c_birth_date, '') = '', iif(:db_id <> 94401, 'ДР.,', 'D.Tarixi,'), '')) || trim(iif(coalesce(c_pass_no, '') = '', iif(:db_id <> 94401, 'Пасп.№,', 'Passport №,'), '')) || trim(iif(coalesce(c_pass_date, '') = '', iif(:db_id <> 94401, 'Дата выдачи,', 'Verilmə tarixi,'), '')) || trim(iif(coalesce(c_pass_date_expire, '') = '', iif(:db_id <> 94401, 'Срок действия,', 'Bitmə tarixi,'), '')) || trim(iif(coalesce(c_blob_photo_id, 0) <= 0, iif(:db_id <> 94401, 'Фото,', 'Şəkil,'), '')) || trim(iif(coalesce(c_blob_pass1_id, 0) <= 0, iif(:db_id <> 94401, 'Скан паспорта,', 'Pasport skanı,'), '')) || trim(iif(:need_passport_second_page = 1, iif(coalesce(c_blob_pass1_id, 0) <= 0, iif(:db_id <> 94401, 'Скан паспорта (обрат.),', 'Pasport skanı 2,'), ''), '')) || trim(iif(coalesce(c_pass_who, '') = '', iif(:db_id <> 94401, 'Место выдачи,', 'Verən yer,'), '')) || trim(iif(coalesce(c_birth_city, '') = '', iif(:db_id <> 94401, 'Место рождения,', 'Doğum yeri,'), ''))
    from clients
    where c_id = :c_id
    into :str_tmp;

    str_tmp_1 = '';
    str_tmp_2 = '';
    if (db_id in (39801, 41701)) then
    begin
      if (exists(select *
                 from clients cl
                 where cl.c_id = :c_id and
                       cl.c_doc_type = 296)) then
      begin
        if (not exists(select *
                       from clientdocs cd
                       inner join passkind pk on pk.pk_id = cd.cds_from_pk_id
                       inner join passdoctype pdt on pdt_id = pk_from_pdt_id
                       where cd.cds_from_c_id = :c_id and
                             pdt.pdt_code = 'info_with_chip' and
                             current_date between cd.cds_doc_date_begin and coalesce(cd.cds_doc_date_end, '31.12.2999') and
                             cd.cds_is_delete = 0)) then
          str_tmp_1 = 'информация с чипа биом. данных,';

      end

      if (not exists(select *
                     from addrclient adc
                     inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
                     where adc.adc_from_c_id = :c_id and
                           adc_address_type = 1 and
                           adl.adl_id <> 41701000000000)) then
        str_tmp_2 = 'юр. адрес,';
    end

    str_tmp = str_tmp || str_tmp_1 || str_tmp_2;
    if (str_tmp <> '') then
    begin
      str_tmp = left(str_tmp, char_length(str_tmp) - 1);
      str_tmp = replace(str_tmp, ',', ', ');
      f_format('%1: <font color="%2">%3</font><br>', iif(db_id <> 94401, 'Не хватает', 'Qeyd olunmayıb'), color_yellow, trim(:str_tmp));
      res_text = BLOB_APPEND(res_text, str_tmp);
    end
  end
  res_text = BLOB_APPEND(res_text, '<br>');

  suspend;
end
