CREATE PROCEDURE UXP$UXEXPORT_PREPARE(
    UFM_UUID D_UUID,
    UA_LIST VARCHAR(8000) CHARACTER SET WIN1251,
    INCLUDE_SOURCE D_BOOL,
    EXISTING_EXP_ID D_INTEGER)
RETURNS (
    EXP_ID D_INTEGER)
AS
declare variable exi_uuid d_uuid;
declare variable tmp_uuid d_uuid;
declare variable last_index d_integer;
declare variable exp_index d_integer;
declare variable use_own_uuid d_bool;
declare variable ugr_uuid d_uuid;
declare variable is_search d_bool;
begin
  if (include_source is null) then
    include_source = 0;

  if (coalesce(existing_exp_id, 0) = 0) then
  begin
    insert into ux$export (exp_ufm_uuid)
    values (:ufm_uuid)
    returning exp_id, exp_index
    into :exp_id, :exp_index;
  end else
  begin
    insert into ux$export (exp_id, exp_ufm_uuid)
    values (:existing_exp_id, :ufm_uuid)
    returning exp_index
    into :exp_index;

    exp_id = existing_exp_id;
  end

  if (include_source = 1) then
    execute procedure uxp$uxexport_src_prepare(:ufm_uuid, :exp_id, :exp_index);

  execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :ufm_uuid);

  use_own_uuid = 0;
  if (include_source = 0) then
  begin
    if (exists(select * from ux$expoform
    where exf_ufm_uuid = :ufm_uuid and
          exf_new_kind = 1 and
          exf_parent_id is null)) then
      use_own_uuid = 1;
  end

  for select uxa_uuid
  from ux$form, ux$alias
  where ufm_uuid = :ufm_uuid and
        uxa_owner = ufm_ux_uuid
  into exi_uuid
  do
    execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :exi_uuid, :use_own_uuid);

  for select ua_uuid
  from ux$action
  where ua_owner = :ufm_uuid
  into exi_uuid
  do
  begin
    if (ua_list = 'all' or position(uuid_to_char(exi_uuid), ua_list) > 0) then
      execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :exi_uuid);
  end

  for select fr_uuid
  from uxp$uxframe_tree(:ufm_uuid)
  into :exi_uuid
  do
    execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :exi_uuid);

  for select ufi_uuid
  from uxp$uxfritem_export_tree(:ufm_uuid, null)
  into :exi_uuid
  do
    execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :exi_uuid);

  for select fr_uuid
  from uxp$uxframe_tree(:ufm_uuid)
  into :tmp_uuid
  do
  begin
    for select ufp_uuid
    from ux$frparam
    where ufp_ufr_uuid = :tmp_uuid and
          ufp_uuid is distinct from ufp_uf_uuid and
          ufp_parent is null
    into exi_uuid
    do
      execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :exi_uuid);

    for select ufp_uuid
    from ux$frparam
    where ufp_ufr_uuid = :tmp_uuid and
          ufp_uuid is distinct from ufp_uf_uuid and
          ufp_parent is not null
    into exi_uuid
    do
      execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :exi_uuid);

    for select ugr_uuid
    from ux$grid
    where ugr_ufr_uuid = :tmp_uuid
    into ugr_uuid
    do
    begin
      execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :ugr_uuid);

      for select ugi_uuid
      from ux$gritem
      where ugi_ugr_uuid = :ugr_uuid
      into :exi_uuid
      do
        execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :exi_uuid);
    end

    for select ufi_uuid
    from uxp$uxfritem_export_tree(:tmp_uuid, null)
    into :exi_uuid
    do
      execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :exi_uuid);

    for select ua_uuid, iif(coalesce(ua_usr_id_field, '') <> '', 1, 0)
    from ux$action
    where ua_owner = :tmp_uuid
    into exi_uuid, is_search
    do
    begin
      if (ua_list = 'all' or is_search = 1 or position(uuid_to_char(exi_uuid), ua_list) > 0) then
        execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :exi_uuid);
    end
  end

  select max(exi_index)
  from ux$expoitem
  where exi_exp_id = :exp_id
  into last_index;

  if (last_index > 0) then
  begin
    for select exi_ux_uuid
    from ux$expoitem
    where exi_exp_id = :exp_id and
          exi_index <= :last_index
    into :tmp_uuid
    do
    begin
      for select evt_uuid
      from ux$event
      where evt_sender_uuid = :tmp_uuid
      into :exi_uuid
      do
        execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :exi_uuid);

      for select evt_uuid
      from ux$event
      where evt_target_uuid = :tmp_uuid
      into :exi_uuid
      do
        execute procedure uxp$uxexpoitem_add(:exp_id, :exp_index, :exi_uuid);
    end
  end

  suspend;
end
