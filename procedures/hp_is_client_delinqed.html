CREATE PROCEDURE HP_IS_CLIENT_DELINQED(
    C_ID D_ID,
    K_ID D_ID = null)
RETURNS (
    IS_DELINQ D_BOOL)
AS
declare variable delinq_amount d_currency;
begin
  is_delinq = 0;

  if (k_id is not null) then
  begin
    -- если передан K_ID, проверяем только этот займ
    if (exists(select * from oksummary
               where oks_from_k_id = :k_id and
                     oks_delinq_date < current_date and
                     oks_date_end >= '31.12.2999')) then
      is_delinq = 1;
    else
    begin
      -- Проверяем среди ранее закрытых займов на наличие внесистемных займах
      delinq_amount = 0;
      select sum(oko_os + oko_perc + oko_sh_os + oko_sh_perc)
      from operkroff o
       where oko_from_k_id = :k_id and
             oko_ol_is_active = 1 and
             oko_ol_date < current_date
      into delinq_amount;
  
      if (delinq_amount > 0) then
        is_delinq = 1;
    end
  end else
  begin
    -- иначе проверяем все активные займы клиента
    for select ks_from_k_id
    from knclient, kontstate
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = ks_from_k_id and
          ks_state = 1 and
          ks_ol_is_delete = 0 and
          ks_date_end >= current_date
    into k_id
    do
    begin
      if (exists(select * from oksummary
                 where oks_from_k_id = :k_id and
                       oks_delinq_date < current_date and
                       oks_date_end >= '31.12.2999')) then
      begin
        is_delinq = 1;
        break;
      end
    end

    if (is_delinq = 0) then
    begin
      -- Проверяем среди ранее закрытых займов на наличие внесистемных займах
      delinq_amount = 0;
      select sum(oko_os + oko_perc + oko_sh_os + oko_sh_perc)
      from operkroff o
       where oko_from_c_id = :c_id and
             oko_ol_is_active = 1 and
             oko_ol_date < current_date
      into delinq_amount;
  
      if (delinq_amount > 0) then
        is_delinq = 1;
    end
  end

  suspend;
end
