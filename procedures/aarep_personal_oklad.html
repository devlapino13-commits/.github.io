CREATE PROCEDURE AAREP_PERSONAL_OKLAD(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    SAO_DATE_BEGIN DATE,
    SAO_HO_DATE DATE,
    SAO_HO_TEXT VARCHAR(1024),
    SAO_HO_ORDER_NO VARCHAR(20),
    SAO_JL_NAME VARCHAR(100),
    SAO_JL_ORDER_NO INTEGER,
    SAO_REG_CODE_NAME D_STR_100,
    SAO_IS_WORKING_TEXT D_STR_100,
    SAO_JL_CATEGORY D_STR_100,
    SAO_FIO VARCHAR(100),
    SAO_FROM_C_ID DECIMAL(18,0),
    SAO_JL_ID INTEGER,
    SAO_OKLAD_SUMMA DECIMAL(18,4),
    SAO_LAST_DATE DATE,
    SAO_IS_NET INTEGER,
    SAO_CHILD_KOLICH INTEGER,
    SAO_IS_LGOTA INTEGER,
    SAO_MOUNTAIN_PERC DECIMAL(18,7),
    SAO_IS_HOUR INTEGER,
    SAO_SF_LGOTA DECIMAL(18,7),
    SAO_PDNAL_LGOTA DECIMAL(18,4),
    SAO_DONATION DECIMAL(18,4),
    PS_CURR_STATUS VARCHAR(255),
    SAO_JK_NAME VARCHAR(255))
AS
declare variable ps_is_working integer;
begin
  for select ps_date_begin, ho_date, ho_text, ho_order_no, jl_id, jl_name, jl_order_no, reg_code || ' ' || reg_name,
             ps_is_working, jl_category, c_name_fio, ps_from_c_id, jk_name
      from hr_orders, hr_pstatus, joblist, jobkind, region, clients, region_list(:report_region_id, 1, -1, 1)
      where ps_from_ho_id = ho_id and
            ps_date_begin <= :report_date_end and
            ps_date_end >= :report_date_end and
            ps_status = 1 and
            ps_from_jl_id = jl_id and
            ps_from_reg_id = reg_unicode and
            ps_from_c_id = c_id and
            reg_unicode = id_list and
            jl_kind = jk_id
      order by jl_order_no, jl_id
      into :sao_date_begin, :sao_ho_date, :sao_ho_text, :sao_ho_order_no, :sao_jl_id, :sao_jl_name, :sao_jl_order_no,
           :sao_reg_code_name, :ps_is_working, :sao_jl_category, :sao_fio, :sao_from_c_id, :sao_jk_name
  do
  begin
    if (ps_is_working = 1) then
      sao_is_working_text = 'Работает';
    else
      sao_is_working_text = 'Уволен';

    ps_curr_status = '';
    select sawt_name
    from hr_voctravel, hr_orders, sa_worktype
    where hrv_from_ho_id = ho_id and
          ho_order_type = sawt_id and
          hrv_from_c_id = :sao_from_c_id and
          hrv_status = 1 and
          hrv_date_begin <= :report_date_end and
          hrv_date_end_fact >= :report_date_end
    order by hrv_date_begin desc
    rows 1
    into :ps_curr_status;

    sao_oklad_summa = 0;
    sao_is_net = 0;
    sao_child_kolich = 0;
    sao_is_lgota = 0;
    select sum(sao_summa), sum(sao_is_net), sum(sao_child_kolich), sum(sao_is_lgota), max(sao_date),
           sum(sao_mountain_perc), sum(sao_is_hour), sum(sao_sf_lgota), sum(sao_pdnal_lgota), sum(sao_donation)
    from sa_oklad
    where sao_from_c_id = :sao_from_c_id and
          sao_date <= :report_date_end
    into :sao_oklad_summa, :sao_is_net, :sao_child_kolich, :sao_is_lgota, :sao_last_date, :sao_mountain_perc,
         :sao_is_hour, :sao_sf_lgota, :sao_pdnal_lgota, sao_donation;
    if (sao_oklad_summa is null) then
    begin
      sao_oklad_summa = 0;
    end
    if (sao_is_net is null) then
      sao_is_net = 0;
    if (sao_child_kolich is null) then
      sao_child_kolich = 0;
    if (sao_is_lgota is null) then
      sao_is_lgota = 0;
    if (sao_mountain_perc is null) then
      sao_mountain_perc = 0;
    if (sao_is_hour is null) then
      sao_is_hour = 0;
    if (sao_sf_lgota is null) then
      sao_sf_lgota = 0;
    if (sao_pdnal_lgota is null) then
      sao_pdnal_lgota = 0;
    if (sao_donation is null) then
      sao_donation = 0;
    suspend;
  end
end
