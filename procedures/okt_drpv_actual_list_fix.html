CREATE PROCEDURE OKT_DRPV_ACTUAL_LIST_FIX
AS
declare variable k_id id_type;
declare variable rd_id integer;
declare variable pant_id id_type;
declare variable mess_str str_255;
begin
--  шаг 1 выгоняю группы из ignor-листа
  for select t.k_id
      from drvp_actual_list t
      into :k_id
  do
  begin
    update pantera_ignore p
    set p.pi_date_end = current_date - 1
    where p.pi_object_id = :k_id and
          p.pi_object_kind = 9 and
          p.pi_is_active = 1;
  end

--  шаг 2 открепляю лишние группы от операторов
  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'OKTELL.Projects.Investigation_And_Recovery'
  into :rd_id;

  for select pantk.pant_from_k_id, pantk.pant_id
      from pantera_kontracts pantk
      inner join kontstate ks on ks.ks_from_k_id = pantk.pant_from_k_id
      where ks.ks_state = 1 and
            current_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_ol_is_delete = 0 and
            pantk.pant_date_end = '30.12.9999' and
            pantk.pant_rd_project_type = :rd_id
      into :k_id, :pant_id
  do
  begin
    if (not exists(select *
                   from drvp_actual_list t
                   where t.k_id = :k_id)) then
    begin
      execute procedure okt_sirena_make_call(1, current_date - 1, 'UNBLOCK', 'SirenaTransferReason.TechnicalUnBlock',
          'OKTELL.Projects.Investigation_And_Recovery', :pant_id, null, 0, null, null, null, null, null, null, null,
          null, 0)
          returning_values :mess_str;
    end
  end
end
