CREATE PROCEDURE AAREP_CALC_BONUS_RU_KS_20121008(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_KC_ID ID_TYPE,
    AA_CURR_KC_FIO VARCHAR(100),
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_AMOUNT_ACTIVE_END CURR_TYPE,
    AA_DELIQ_BALANCE7 CURR_TYPE,
    PS_IS_ON_STAFF_STR VARCHAR(100),
    AA_REPAY_ALL_CATEG_B CURR_TYPE,
    PS_IS_WORKING_STR VARCHAR(100),
    PAR CURR_TYPE,
    AA_AMOUNT_DISB CURR_TYPE,
    AA_KOLICH_CL_DISB INTEGER,
    DISB_AVG CURR_TYPE,
    CLI_CLOSED INTEGER,
    BONUS_TOTAL_CATEG_A CURR_TYPE,
    BONUS_TOTAL_CATEG_B CURR_TYPE,
    BONUS_TOTAL CURR_TYPE,
    NEW_BONUS_TOTAL CURR_TYPE,
    AA_KOLICH_DISB INTEGER,
    AA_LO_STAGE_DATE DATE,
    AA_LO_STAGE CURR_TYPE,
    JL_NAME VARCHAR(100))
AS
declare variable bn_disb integer;
declare variable bn_close integer;
declare variable ps_is_on_staff integer;
declare variable ps_is_working integer;
declare variable ps_from_reg_id integer;
declare variable cli_closed_bn integer;
declare variable kc_prior_id decimal(18,0);
declare variable aa_curr_region_id integer;
declare variable cli_clos_woff integer;
declare variable cli_clos_sb integer;
begin
  kc_prior_id = 0;
  for select aa_curr_kc_id, aa_curr_kc_fio, aa_curr_region_id, aa_curr_office_name, aa_curr_branch_name,
             aa_curr_level_name, sum(
             case aa_loan_kind_new
               when 'Категория A' then aa_amount_active_end
               else 0
             end),
-- Портфель по категории А
             sum(
             case aa_loan_kind_new
               when 'Категория A' then aa_deliq_balance7
               else 0
             end),
-- Просроченный портфель по категории А
             sum(
             case aa_loan_kind_new
               when 'Категория A' then aa_amount_disb
               else 0
             end),
-- Выданная сумма
             sum(
             case aa_loan_kind_new
               when 'Категория A' then aa_kolich_cl_disb
               else 0
             end),
-- Количество колиентов выданным
             sum(
             case aa_loan_kind_new
               when 'Категория A' then aa_kolich_disb
               else 0
             end),
-- Количество групп выданных
             sum(
             case aa_loan_kind_new
               when 'Категория B' then aa_repay_all_os + aa_repay_all_perc + aa_repay_all_fine
               else 0
             end) -- Погашенная сумма по категории B

      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, null, null, 0, 1, 0, 0, 0, 0)
      where aa_loan_kind_new in ('Категория A', 'Категория B')

      group by aa_curr_kc_id, aa_curr_kc_fio, aa_curr_region_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name
      order by aa_curr_level_name, aa_curr_branch_name, aa_curr_office_name, aa_curr_kc_fio
      into :aa_curr_kc_id, :aa_curr_kc_fio, :aa_curr_region_id, :aa_curr_office_name, :aa_curr_branch_name,
           :aa_curr_level_name,

           :aa_amount_active_end, :aa_deliq_balance7, :aa_amount_disb, :aa_kolich_cl_disb, :aa_kolich_disb, :aa_repay_all_categ_b

  do
  begin
    par = 0;
    if (:aa_amount_active_end <> 0) then
      par = cast(:aa_deliq_balance7 / :aa_amount_active_end * 100 as decimal(15,1));

-- Вычисляет среднию сумму выдачи
    disb_avg = 0;
    if (aa_kolich_cl_disb <> 0 and
        aa_amount_disb is not null) then
      disb_avg = cast(:aa_amount_disb / aa_kolich_cl_disb as decimal(15,0));

    /*Вычисляет суммы выдачи за каждого выданного и за каждого закрытого*/
    bn_disb = 0;
    bn_close = 0;
    if (disb_avg between 10000 and 29999) then
    begin
      bn_disb = 350;
      bn_close = 700;
    end
    else
    if (disb_avg >= 30000) then
    begin
      bn_disb = 500;
      bn_close = 1000;
    end

--    if (:AA_CURR_KC_ID <> kc_prior_id) then
--    begin
    /*Вычисляет кол-во закрытых клиентов*/
    select count(*)
    from knaktlist, kontract, kontstate, knclient
    where ks_date_begin between kal_date and kal_date_end and
          knaktlist.is_delete = 0 and
          kal_from_kf_id = k_from_kf_id and
          ks_from_k_id = k_id and
          ks_ol_is_delete = 0 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_state = 2 and
          k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          kal_new_kf_officer = :aa_curr_kc_id and
          kal_new_reg_id = :aa_curr_region_id -- Выборка по текущему региону
          and
          not(k_from_pr_id between 18 and 30) -- Не включает категорию B
          and
          datediff(month, k_disb_date, ks_date_begin) >= 3 -- выборка по тем контрактам которые закрылись отработав 3 и более месяцев
    into :cli_closed;

    if (:cli_closed is null) then
      cli_closed = 0;

    /*Вычисляет кол-во закрытых со списаниями*/
    with tab1
    as (select distinct k1.k_id
        from operkred, kontstate, kontract k1, knaktlist
        where ok_from_ol_id = ks_from_ol_id and
              ks_ol_is_delete = 0 and
              ks_date_begin between :rep_date_begin and :rep_date_end and
              ks_state = 2 and
              k1.k_id = ks_from_k_id and
              ok_type = 5 and
              ok_os <> 0 and
              knaktlist.is_delete = 0 and
              kal_from_kf_id = k1.k_from_kf_id and
              ks_date_begin between kal_date and kal_date_end and
              kal_new_kf_officer = :aa_curr_kc_id and
              kal_new_reg_id = :aa_curr_region_id -- Выборка по текущему региону
              and
              not(k_from_pr_id between 18 and 30) -- Не включает категорию B
              and
              ok_from_c_id = 0 and
              datediff(month, k1.k_disb_date, ks_date_begin) >= 3 -- закрывшие отработав 3 и более месяцев
              and
              (not exists(select *
                          from sb_akt, sb_list
                          where sba_from_k_id = k1.k_id /*Не включает переданные в СБ*/
                                and
                                sba_date_begin <= :rep_date_end and
                                sba_date_end >= :rep_date_end and
                                sba_is_delete = 0 and
                                sba_from_sbl_id <> 0 and
                                sba_status = 2 and
                                sba_from_sbl_id = sbl_id and
                                sbl_type = 1)))
    select count(*)
    from knclient, tab1
    where tab1.k_id = kc_from_k_id and
          kc_amount_approved > 0
    into :cli_clos_woff;
    if (:cli_clos_woff is null) then
      cli_clos_woff = 0;

    /*Вычисляет кол-во закрытых клиентов переданные в СБ*/
    select count(*)
    from knaktlist, kontract k2, kontstate, knclient
    where ks_date_begin between kal_date and kal_date_end and
          knaktlist.is_delete = 0 and
          kal_from_kf_id = k2.k_from_kf_id and
          ks_from_k_id = k2.k_id and
          ks_ol_is_delete = 0 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_state = 2 and
          k2.k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          kal_new_kf_officer = :aa_curr_kc_id and
          kal_new_reg_id = :aa_curr_region_id -- Выборка по текущему региону
--      and k2.K_Is_Problem=1
          and
          not(k_from_pr_id between 18 and 30) -- Не включает категорию B
          and
          datediff(month, k2.k_disb_date, ks_date_begin) >= 3 -- закрывшие отработав 3 и более месяцев
          and
          exists(select *
                 from sb_akt, sb_list
                 where sba_from_k_id = k2.k_id /*Выборка только по СБ*/
                       and
                       sba_date_begin <= :rep_date_end and
                       sba_date_end >= :rep_date_end and
                       sba_is_delete = 0 and
                       sba_from_sbl_id <> 0 and
                       sba_status = 2 and
                       sba_from_sbl_id = sbl_id and
                       sbl_type = 1)
    into :cli_clos_sb;
    if (:cli_clos_sb is null) then
      cli_clos_sb = 0;
--    Не включает закрытые со списанием и закрытые по СБ
    if (cli_closed > cli_clos_sb + cli_clos_woff) then
      cli_closed_bn = cli_closed - (cli_clos_sb + cli_clos_woff);
    else
      cli_closed_bn = 0;
--    Рачет бонуса
    if (par > 7) then
      bonus_total_categ_a = 0;
    else
      bonus_total_categ_a = bn_disb * aa_kolich_cl_disb + bn_close * cli_closed_bn;

-- По пункту  4.2 не доконца понял, т.е. нужно от пагашенной суммы 8% считать или после того как контракт закроется ?
    bonus_total_categ_b = :aa_repay_all_categ_b * 8 / 100;

-- Параметры кредитного служащего
    select jl_name, ps_is_on_staff, ps_is_working, ps_from_reg_id
    from hr_pstatus, joblist
    where ps_from_c_id = :aa_curr_kc_id and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_jl_id = jl_id
    order by ps_is_working desc
    rows 1
    into :jl_name, :ps_is_on_staff, :ps_is_working, :ps_from_reg_id;

    if (ps_is_working is null or ps_is_working = 0) then
    begin
      bonus_total_categ_a = 0;
      bonus_total_categ_b = 0;
    end
    bonus_total = bonus_total_categ_a + bonus_total_categ_b;

    if (ps_is_on_staff is not null) then
    begin
      if (ps_is_on_staff = 1) then
        ps_is_on_staff_str = 'Штатный';
      else
        ps_is_on_staff_str = 'Внештатный';
      if (ps_is_working = 0) then
      begin
        ps_is_on_staff_str = ps_is_on_staff_str || ', ' || 'уволен';
        ps_is_working_str = 'Уволен';
      end
      else
      if (ps_is_working = 1) then
        ps_is_working_str = 'Работает';
    end
    else
      ps_is_on_staff_str = 'Приказ не оформлен';

    /*    if (ps_from_reg_id is not null) then
    begin
      select AA_CURR_OFFICE_NAME, AA_CURR_BRANCH_NAME
      from AAREP_GET_OFFICE_CATEG(:ps_from_reg_id, :REP_DATE_END)
      into :AA_CURR_OFFICE_NAME, :AA_CURR_BRANCH_NAME;
    end */

-- Вычисляет стаж СВЗ
    /*   select min(ks_date_begin) from kontstate, kontract, knaktlist
     where ks_from_k_id = k_id
     and ks_ol_is_delete = 0
     and ks_state = 1
     and k_from_kf_id = kal_from_kf_id
     and knaktlist.is_delete = 0
     and kal_new_kf_officer = :AA_CURR_KC_ID
     and cast(ks_date_begin as date) between kal_date and kal_date_end
     into :AA_LO_STAGE_DATE;  */

    aa_lo_stage_date = null;

    select ps_date_begin
    from hr_pstatus --, hr_orders , joblist
    where ps_from_c_id = :aa_curr_kc_id
--      and ps_from_ho_id=ho_id
          and
          ps_date_begin is not null
--      and jl_id=ps_from_jl_id
--      and jl_kind=5
    order by ps_date_begin
    rows 1
    into :aa_lo_stage_date;

    if (aa_lo_stage_date is null) then
      aa_lo_stage_date = :rep_date_end;
    aa_lo_stage = datediff(day from aa_lo_stage_date to :rep_date_end) / 30;

    new_bonus_total = 0;
    if (aa_kolich_disb >= 2 and
        aa_lo_stage >= 1) then
    begin
      new_bonus_total = bonus_total;
    end
    else
    begin
      new_bonus_total = 0;
    end

    suspend;
  end

end
