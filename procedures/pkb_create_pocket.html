CREATE PROCEDURE PKB_CREATE_POCKET(
    POCKET_DATE D_DATETIME,
    KIB_NAME D_STR_100,
    POCKET_TYPE VARCHAR(1) = null)
RETURNS (
    POCKET_ID D_INTEGER,
    POCKET_NAME VARCHAR(50))
AS
declare variable pocket_date_begin d_date;
declare variable pocket_date_end d_date;
declare variable date_str d_str_100;
declare variable cp_counter d_integer;
declare variable cib_id d_integer;
begin
  pocket_type = coalesce(pocket_type, '');
  cib_id = case kib_name
             when 'GKB' then f_get_rd_id('CIB.GKB')
             when 'PKB' then f_get_rd_id('CIB.PKB')
             else f_get_rd_id('CIB.PKB')
           end;
  if (pocket_type = '') then
  begin
    /*--last_month_date = (datediff(day from pocket_date to dateadd(1 month to pocket_date)));
-- проверяю последний ли это день месяца или нет. если последний  и тип пакета не указан, что  дневной, тогда формирую месячный пакет

    if ((extract(month from pocket_date) <> (extract(month from dateadd(1 day to pocket_date))) and
        (pocket_type <> 'd'))) then
      pocket_type = 'm';
    else */
    pocket_type = 'd';
  end
  pocket_date_end = cast(pocket_date as date);
  date_str = cast(cast(pocket_date as date) as varchar(10));
  if (pocket_type = 'm') then
    pocket_date_begin = '01.' || substring(date_str from 6 for 2) || '.' || substring(date_str from 1 for 4);
  else
    pocket_date_begin = cast(pocket_date as date);

  select max(cp.cp_counter)
  from cib_pocket cp
  where cp.cp_pocket_type = :pocket_type and
        cp.cp_date_end = :pocket_date_end and
        cp.cp_cib_id = :cib_id
  into :cp_counter;

  cp_counter = coalesce(cp_counter, 0) + 1;
  date_str = left(replace(replace(replace(cast(:pocket_date as varchar(24)), '-', ''), ' ', ''), ':', ''), 14);

  --date_str = substring(date_str from 1 for 4) || substring(date_str from 6 for 2);
  --pocket_id = cast((date_str||'99'||id_finish_side) as integer);
  pocket_name = 'mbulak_' || date_str;
  pocket_name = upper(pocket_name);

  insert into cib_pocket (cp_counter, cp_date_begin, cp_date_end, cp_pocket_name, cp_pocket_type, cp_pocket_status,
                          cp_cib_id)
  values (:cp_counter, :pocket_date_begin, :pocket_date_end, :pocket_name, :pocket_type, 0, :cib_id)
  returning cp_id
  into :pocket_id;

  suspend;
end
