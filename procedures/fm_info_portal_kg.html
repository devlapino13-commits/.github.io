CREATE PROCEDURE FM_INFO_PORTAL_KG(
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable went_to_portal d_str_10;
declare variable c_id d_id;
declare variable region_manager_id d_id;
declare variable c_name_fio d_str_100;
declare variable reg_name d_str_100;
declare variable date_begin d_date;
declare variable date_end d_date;
begin
  --Данная процедура отображает ссылку на сайт информационного портала для ВИПСов и МО
  --В данной процедуре нет проверки, так как проверка уже в процедуре fm_report_list
  date_begin = '01.' || extract(month from current_date) || '.' || extract(year from current_date);
  date_end = dateadd(month, 1, date_begin) - 1;

  res_text = (select res_style
              from fm_html_styles) || '<body>';

  res_text = BLOB_APPEND(res_text, '<a href="https://academy.mbulak.kg/?uid=' || :un_id || '"> Цифровая академия AkylDo</a><br></br>');
  res_text = BLOB_APPEND(res_text, 'Академия Акылдуу – обучающая платформа для сотрудников М Булак. 
                          Единый ресурс совмещает всё, что необходимо для продуктивной работы в нашей Компании.
                          В разделе "Корпоративное обучение" собраны материалы по актуальным инструкциям, процедурам и правилам.
                          Есть и полезные советы,  которые пригодятся не только новичкам!
                          "Общеобразовательные курсы" - это материалы о менеджменте, маркетинге, психологии и финансах полезные для каждого.
                          "Библиотека" - супер подборка книг для развития, в текстовом и аудио формате.
                          Материалы всегда остаются открыты для повторного изучения.');

  select un.un_from_c_id
  from username un
  where un.un_id = :un_id
  into region_manager_id;

  if (exists(select *
             from hr_pstatus ps
             inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
             inner join jobkind jk on jk.jk_id = jl.jl_kind
             where ps.ps_from_c_id = :region_manager_id and
                   current_date between ps.ps_date_begin and ps.ps_date_end and
                   ps.ps_is_working = 1 and
                   ps.ps_status = 1 and
                   jk.jk_system_code in ('JobSystem.Region_Director', 'JobSystem.FinStart_Programmer'))) then

  begin
    res_text = BLOB_APPEND(res_text, '<h2>Список стажёров</h2>');
    res_text = BLOB_APPEND(res_text, '<table width="100%"><col width="30%"><col width="60%"><col width="10%">');
    res_text = BLOB_APPEND(res_text, f_html_table_row('<b>Офис</b>', '<b>Стажёр</b>', '<b>Заходил/Не заходил</b>', '', -1, -1, -1, -1, 3, ''));
    for select t.name_fio, t.reg_name, t.c_id
        from hp_interns_view(:un_id, 'reg_manager', 1000000, :date_begin, :date_end, 1, 0, 1) t
        order by t.name_fio, t.reg_name
        into c_name_fio, reg_name, c_id
    do
    begin
      went_to_portal = 'Нет';

      if (exists(select *
                 from un_portal_visit_svod upvs
                 inner join username un on un.un_id = upvs.upvs_from_un_id
                 where un.un_from_c_id = :c_id)) then
        went_to_portal = 'Да';

      res_text = BLOB_APPEND(res_text, f_html_table_row(:reg_name, :c_name_fio, :went_to_portal, '', -1, -1, -1, -1, 3, ''));
    end
    res_text = BLOB_APPEND(res_text, '</table></body>');
  end

  suspend;
end
