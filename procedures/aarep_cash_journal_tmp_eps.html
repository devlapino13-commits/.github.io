CREATE PROCEDURE AAREP_CASH_JOURNAL_TMP_EPS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OL_ID DECIMAL(18,0),
    OL_DATE DATE,
    OL_DOC_NO INTEGER,
    OL_CLIENT_NAME VARCHAR(255),
    OL_DOC_NO_STR VARCHAR(50),
    OL_SUMMA_PLUS DECIMAL(18,4),
    OL_SUMMA_MINUS DECIMAL(18,4),
    OL_ACCOUNT VARCHAR(255),
    OL_KOL_PLUS INTEGER,
    OL_KOL_MINUS INTEGER,
    OL_SALDO_BEGIN_DT DECIMAL(18,4),
    OL_SALDO_BEGIN_KT DECIMAL(18,4),
    OL_SALDO_END_DT DECIMAL(18,4),
    OL_SALDO_END_KT DECIMAL(18,4),
    OL_KIND INTEGER,
    CORR_ACC VARCHAR(20),
    OL_BRANCH_NAME VARCHAR(255),
    KOB_SUMMA D_CURRENCY,
    SUB_NO D_INTEGER,
    REG_CODE D_STR_20,
    KC_ID DECIMAL(18,0))
AS
declare variable ol_document_client decimal(18,0);
declare variable ol_doc_client_name varchar(255);
declare variable reg_id integer;
declare variable acc_code_kt d_str_code;
declare variable acc_code_dt d_str_code;
declare variable acc_ab_code_kt d_str_code;
declare variable acc_ab_code_dt d_str_code;
declare variable ol_doc_no_old d_integer;
declare variable kc_id_old id_type;
declare variable sab_id d_id;
declare variable saldo_begin d_currency;
declare variable saldo_end d_currency;
declare variable acc_id d_id;
begin
  ol_doc_no = null;

  ol_kol_plus = 0;
  ol_kol_minus = 0;
  ol_kind = 0;
  ol_summa_minus = 0;
  ol_summa_plus = 0;
  ol_saldo_begin_dt = 0;
  ol_saldo_begin_kt = 0;
  ol_saldo_end_dt = 0;
  ol_saldo_end_kt = 0;
  ol_doc_no_str = '';

  select reg_kont_company
  from region
  where reg_unicode = :rep_region_id
  into :ol_branch_name;

  saldo_begin = 0;
  saldo_end = 0;

  for select sab_id
  from sys_accbalance
  where sab_full_code starting 'Деньги.Нал.Касса'
  into sab_id
  do
  begin
    for select acc_id
    from ref_account, region_list(:rep_region_id, 1, 3, 1)
    where acc_from_sab_id = :sab_id and
          acc_from_reg_id = id_list
    into acc_id
    do
    begin
      saldo_begin = saldo_begin + f_acc_bal(acc_id, rep_date_begin - 1);
      saldo_end = saldo_end + f_acc_bal(acc_id, rep_date_end);
    end
  end

  if (saldo_begin >= 0) then
  begin
    ol_saldo_begin_dt = saldo_begin;
    ol_saldo_begin_kt = 0;
  end
  else
  begin
    ol_saldo_begin_dt = 0;
    ol_saldo_begin_kt = abs(saldo_begin);
  end

  if (saldo_end >= 0) then
  begin
    ol_saldo_end_dt = saldo_end;
    ol_saldo_end_kt = 0;
  end
  else
  begin
    ol_saldo_end_dt = 0;
    ol_saldo_end_kt = abs(saldo_end);
  end


  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into :reg_id
  do
  begin
    ol_kind = 1;
    ol_summa_minus = 0;
    sub_no = 1;
    for select ol.ol_id, ol_date, ol_doc_no, ol_document_client, ol_doc_client_name, kt.acc_code, kob_summa,
               reg_code, dt.acc_code,
               kob.kob_from_kc_id, sdt.sab_full_code, skt.sab_full_code
        from kn_operbook kob
        inner join operlist ol on ol.ol_id = kob.kob_from_ol_id
        inner join region on reg_unicode = kob.kob_ol_reg_id
        inner join ref_account kt on kt.acc_id = kob.kob_acc_kt
        inner join ref_account dt on dt.acc_id = kob.kob_acc_dt
        inner join sys_accbalance sdt on sdt.sab_id = dt.acc_from_sab_id
        inner join sys_accbalance skt on skt.sab_id = kt.acc_from_sab_id
        where kob_ol_date between :rep_date_begin and :rep_date_end and
              kob.kob_ol_is_active = 1 and
              kob.kob_ol_reg_id = :reg_id and
              (sdt.sab_full_code starting 'Деньги.Нал.Касса' or
               skt.sab_full_code starting 'Деньги.Нал.Касса')
        order by ol.ol_doc_no, kob.kob_from_kc_id
        into :ol_id, :ol_date, :ol_doc_no, :ol_document_client, :ol_doc_client_name, :acc_code_kt, :kob_summa,
             :reg_code, acc_code_dt,
             :kc_id, acc_ab_code_dt, acc_ab_code_kt
    do
    begin
      if (ol_doc_no = :ol_doc_no_old and :kc_id <> :kc_id_old) then
        sub_no = sub_no + 1;

      ol_doc_no_str = reg_code || '-' || right('000000' || ol_doc_no, 6);

      if (kc_id is not null) then
        ol_doc_no_str = ol_doc_no_str || '/' || sub_no;

      if (ol_doc_client_name is null) then
        ol_doc_client_name = '';

      if (ol_document_client is not null) then
      begin
        select c_name_fio
        from clients cl
        where c_id = :ol_document_client
        into :ol_client_name;
      end
      else
      begin
        if (kc_id is not null) then
        begin
          select c_name_fio
          from knclient, clients
          where kc_id = :kc_id and
                kc_from_c_id = c_id
          into :ol_client_name;
        end
        else
          ol_client_name = :ol_doc_client_name;
      end

      ol_summa_minus = 0;
      ol_summa_plus = 0;

      if (acc_ab_code_kt starting 'Деньги.Нал.Касса') then
      begin
        ol_account = acc_code_dt;
        corr_acc = acc_code_kt;
        ol_summa_minus = kob_summa;
      end
      else
      if (acc_ab_code_dt starting 'Деньги.Нал.Касса') then
      begin
        ol_account = acc_code_kt;
        corr_acc = acc_code_dt;
        ol_summa_plus = kob_summa;
      end

      suspend;

      if (ol_doc_no is null and kc_id_old is null) then
        sub_no = 1;
      if (:ol_doc_no <> :ol_doc_no_old) then
        sub_no = 1;

      ol_doc_no_old = ol_doc_no;
      kc_id_old = kc_id;
    end
  end
end
