CREATE PROCEDURE UXP$UXFRPARAM_CFG_UPDATE(
    UFP_UFR_UUID D_UUID,
    UFP_UF_UUID D_UUID,
    UFP_IN_USE D_BOOL,
    UFP_REFRESH D_BOOL,
    UFP_NOT_NULL D_BOOL,
    UFP_DEFAULT D_STR_100,
    UFP_LOCKED D_BOOL,
    UFP_EDITOR D_STR_CODE,
    UFP_PUBLIC D_BOOL,
    UF_NAME D_STR_100,
    UFP_PARENT D_UUID)
AS
declare variable ufp_uuid d_uuid;
begin
  if (ufp_in_use = 1) then
  begin
    update ux$field
    set uf_name = :uf_name
    where uf_uuid = :ufp_uf_uuid and
          uf_name is distinct from :uf_name;

    update or insert into ux$frparam
           (ufp_ufr_uuid, ufp_uf_uuid, ufp_parent,  ufp_refresh, ufp_not_null, ufp_editor, ufp_public)
    values (:ufp_ufr_uuid, :ufp_uf_uuid, :ufp_parent, :ufp_refresh, :ufp_not_null, :ufp_editor, :ufp_public)
    matching (ufp_ufr_uuid, ufp_uf_uuid, ufp_parent)
    returning ufp_uuid
    into ufp_uuid;

    execute procedure uxp$uxparamdef_update(:ufp_uuid, null, :ufp_default, :ufp_locked);
  end else
  begin
    delete from ux$frparam
    where ufp_ufr_uuid = :ufp_ufr_uuid and
          ufp_uf_uuid = :ufp_uf_uuid and
          ufp_parent is not distinct from :ufp_parent;
  end
end
