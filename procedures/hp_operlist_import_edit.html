CREATE PROCEDURE HP_OPERLIST_IMPORT_EDIT(
    DT_OPER D_DATE,
    OFFICE_NAME D_STR_CODE,
    OFFICE_ID D_INTEGER,
    ACC_DT D_STR_20,
    ACC_DT_ID D_STR_20,
    ACC_KT D_STR_20,
    ACC_KT_ID D_STR_20,
    CLIENT_DT D_STR_100,
    CLIENT_DT_ID D_ID,
    CLIENT_DT_INN D_STR_20,
    CLIENT_KT D_STR_100,
    CLIENT_KT_ID D_ID,
    CLIENT_KT_INN D_STR_20,
    AMOUNT D_CURRENCY,
    MEMO D_STR_LONG,
    OPER_NO D_STR_20,
    DT_SUPPLY D_DATE,
    PAY_CODE D_STR_20,
    UN_ID D_ID,
    STATUS D_BOOL)
AS
declare variable ol_id d_id;
declare variable is_porder boolean;
begin
  if (status = 1) then
  begin
    is_porder = pay_code is not null;
    insert into operlist (
      ol_from_reg_id,
      ol_doc_no,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_kind,
      ol_document_client,
      ol_acc_no_client,
      ol_document_office,
      ol_acc_no_office,
      ol_memo,
      ol_is_client,
      ol_from_rd_id,
      ol_user_id)
    values (
      :office_id,
      gen_id(gen_operlist_import_id, 1),
      2,
      iif(:is_porder, 7, 11),
      :dt_oper,
      iif(:is_porder, 6, 9),
      iif(:is_porder and :client_dt_id > 0, :client_dt_id, null),
      iif(:client_dt_id > 0, (select first 1 bc_number from bankclient where bc_from_c_id = :client_dt_id and bc_is_active = 1), null),
      iif(:is_porder and :client_dt_id < 0, -:client_dt_id, null),
      iif(:client_dt_id < 0, (select first 1 br_number from bankregion where br_from_reg_id = -:client_dt_id and br_is_active = 1), null),
      iif(:is_porder, :memo, null),
      iif(:is_porder, 1, 0),
      f_get_rd_id('Import_Operlist'),
      :un_id)
    returning ol_id
    into :ol_id;

    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_client_dt,
      ob_client_kt,
      ob_from_pc_id,
      ob_summa,
      ob_summa_val,
      ob_detail)
    values (
      :ol_id,
      '417',
      1,
      :acc_dt_id,
      :acc_kt_id,
      iif(:client_dt_id > 0, :client_dt_id, null),
      :client_kt_id,
      iif(:is_porder, :pay_code, null),
      :amount,
      :amount,
      :memo);

    if (oper_no is not null and not is_porder) then
      insert into operlist_add (
        ola_from_ol_id,
        ola_oper_no,
        ola_dtsupply)
      values (
        :ol_id,
        :oper_no,
        :dt_supply);
  end
end
