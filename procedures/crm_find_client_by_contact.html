CREATE PROCEDURE CRM_FIND_CLIENT_BY_CONTACT(
    PHONE_NUMBER STR_255)
RETURNS (
    FIO STR_100,
    STATUS STR_100)
AS
declare variable c_id d_id;
begin

    for select cc_from_c_id, cc_status
    from contacts
    where cc_cell_no = :phone_number
      and cc_is_old_contact = 0
    into c_id, status
    do
    begin
        for select c_name_fio
        from clients c
        inner join knclient kc on kc.kc_from_c_id = c.c_id
        inner join oksummary oks on kc.kc_from_k_id = oks.oks_from_k_id
        where :c_id = c.c_id
        and current_date between oks.oks_date_begin and oks.oks_date_end
        and oks_close_state = 0  
        and current_date - oks.oks_delinq_date > 0
        into fio
        do
            suspend;
    end

end
