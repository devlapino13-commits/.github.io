CREATE PROCEDURE HP_EDOC_INS_ASP_AGREEMENT(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable IS_ACTIVE_FM_USER D_BOOL;
declare variable CLIENT_UN_ID D_ID;
declare variable DB_ID D_ID;
declare variable RD_ID D_ID;
declare variable VER_STATUS D_SMALLINT;
declare variable IS_CHECK_NUMBER D_SMALLINT;
declare variable ADD_TIME D_INTEGER;
declare variable ERROR_TEXT D_STR_255;
declare variable PERSTEMP_EXISTS D_BOOL;
declare variable IS_OIVO D_SMALLINT;
declare variable IS_ASSIGNMENT D_SMALLINT;
begin
  -- Процедура создает договор на оказание информационных услуг (Договор АСП)
  -- Готовит бланки и отправляет СМС.
  res_text = '';
  res_is_error = 0;
  is_oivo = 0;
  db_id = f_get_db_id();
  is_check_number = f_get_const('check_contact_number');

  for select c.c_name_fio, c.c_birth_date, c.c_pass_no, c.c_pass_who, c.c_pass_date
      from clients c
      where c.c_id = :c_id as cursor cur_c
  do
  begin
    if (coalesce(cur_c.c_name_fio, '') = '') then
      res_text = left(res_text || 'Не заполнено поле ФИО клиента. ', 255);
    if (cur_c.c_birth_date is null) then
      res_text = left(res_text || 'Не заполнено поле "Дата рождения". ', 255);
    if (db_id in (39801, 41701) and
        cur_c.c_birth_date > dateadd(year, -18, current_date)) then
      res_text = left(res_text || 'Клиенту еще не исполнилось 18 лет. ', 255);
    if (coalesce(cur_c.c_pass_no, '') = '') then
      res_text = left(res_text || 'Не заполнено поле "Номер документа удостоверяющего личность". ', 255);
    if (coalesce(cur_c.c_pass_who, '') = '') then
      res_text = left(res_text || 'Не заполнено поле "Орган выдавший документ". ', 255);
    if (cur_c.c_pass_date is null) then
      res_text = left(res_text || 'Не заполнено поле "Дата выдачи документа". ', 255);
  end

  if (db_id = 81001 and res_text = '') then
  begin
    for select 'Совпадение с лицом по ' || c_comment || ' ' || c_name_fio
    from hp_check_client_fio_ft(:c_id, 1)
    into res_text
    do
      res_text = res_text;
  end

  if (db_id = 81001 and res_text = '' and f_tmp_const('use_perstemp') = '1') then
  begin
    error_text = '';
    perstemp_exists = 0;

    select res_error_text, res_is_exists
    from hp_check_exists_perstemp(:c_id, 'name_date_place_birth')
    into error_text, perstemp_exists;

    if (error_text = '' and perstemp_exists = 1) then
      res_text = 'Были изменены ФИО или дата/место рождения, нельзя отправить договор АСП'; -- RFCRU-2205
  end

  if (db_id = 81001 and res_text = '') then
  begin
    select is_oivo, res_id
    from hp_check_oivo_assignment(:c_id, :un_id)
    into :is_oivo, is_assignment;

    if (is_oivo = 1) then
    begin
      if(is_assignment <> 1) then
        res_text = 'Верификация не прошла, так как клиент закреплен не за Вами!';

      if(res_text = '') then
        select error_text
        from hp_smevpass_res_before_miniapp(:c_id)
        into :res_text;

      if (res_text = '' and
        exists(
          select 1 from addrclient
          where adc_from_c_id = :c_id and
                adc_address_type = 1 and
                mod(adc_from_adl_id,1000000000) = 0
        ) or
        not exists(
            select 1
            from clientdocs cld
            join passkind pk on pk.pk_id = cld.cds_from_pk_id
            join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
            where cld.cds_from_c_id = :c_id and
                  cld.cds_is_delete = 0 and
                  pdt.pdt_code in ('registration_permanent', 'registration_fiz')
            )
        ) then
            res_text = 'Не указан юридический адрес или не приложена регистрация в РФ';
    end
  end

  if (res_text = '' and is_oivo <> 1) then
  begin
    add_time = null;
    select add_time
    from hp_qr_additional_time(:c_id)
    into add_time;
    add_time = coalesce(add_time, 0);

    -- QR верификация требуется для всех, кроме сотрудников и действующих пользователей МП сотрудников
    if (f_is_employee(c_id, current_date) = 0 and
        f_meet_valid(c_id, 120 + add_time) = 0) then
    begin
      is_active_fm_user = 0;

      select un_id
      from username
      where un_from_c_id = :c_id
      into client_un_id;

      if (client_un_id is not null) then
      begin
        if (exists(select *
                   from hp_permission_access_check(:client_un_id, 'PermActions', 0, 'FsMobile.Login')
                   where has_access = 1)) then
          is_active_fm_user = 1;
      end

      if (is_active_fm_user = 0) then
        res_text = f_lang(f_get_lang_code(un_id, null), 'Требуется верификация клиента по QR-коду.');
    end
  end

  if (db_id in (39801, 41701) and
      :is_check_number = 1) then
  begin
    select rd.rd_id
    from refdata rd
    where rd.rd_code = 'ActionGroup.SimpleVerify_Phone_Number_Verify'
    into :rd_id;

    select acl.acl_status
    from actionlist acl
    where acl.acl_object_kind = f_objkind('clients', 'id') and
          acl.acl_object_id = :c_id and
          acl.acl_from_rd_id = :rd_id
    order by acl.acl_datetime desc
    rows 1
    into ver_status;

    if (ver_status <> 2) then
      res_text = 'Номер ещё не проверен, отправка ЭЦП невозможна';
  end

  if (res_text = '') then
  begin
    select edl_id, res_text
    from hp_edoc_insert('asp.agreement', f_objkind('clients', 'id'), :c_id, :un_id)
    into edl_id, res_text;
  end

  if (res_text <> '') then
    res_is_error = 1;

  suspend;
end
