CREATE PROCEDURE AAREP_INSTALLED_POS(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    TRM_TYPE D_STR_10,
    TRM_IPC_ID D_STR_20,
    OFFICE_INIATATOR VARCHAR(50),
    MCC_CODE D_STR_100,
    C_NAME_FIO D_STR_100,
    PLACE_NAME D_STR_255,
    ADL_ADDRESS_STR D_STR_255,
    CS_QR_DATA D_STR_10,
    TRM_INVENT_STR VARCHAR(50),
    TRM_SERIAL_NO D_STR_255,
    ITO_DATE_BEGIN D_DATE,
    CS_PHONE D_STR_20,
    TRM_SIMCARD_NO D_STR_20,
    ITO_STATUS D_STR_100,
    ITA_DOC_DATE D_DATE)
AS
declare variable ito_object_id d_id;
declare variable ito_object_kind d_integer;
declare variable cs_from_c_id d_id;
declare variable obj_kind_shop_id d_integer;
declare variable trm_id d_id;
begin
   obj_kind_shop_id = f_objkind('clientshops', 'id');

   for select trm_type,
              trm_ipc_id,
              trm_invent_str,
              trm_serial_no,
              trm_simcard_no,
              ito_object_kind,
              ito_object_id,
              ito_date_begin,
              ito_trm_id,
              (case apt.ito_status  when 1 then 'В процессе регистрации'
                                    when 3 then 'Активный'
                                    when -1 then 'Закрытый'
                                    when 2 then 'На конфигурации у IT'
                                    when 0 then 'Установленный'
                                    end)
   from aarep_pos_tmp('installed', :rep_date_begin, :rep_date_end) apt
   into :trm_type,
        :trm_ipc_id,
        :trm_invent_str,
        :trm_serial_no,
        :trm_simcard_no,
        :ito_object_kind,
        :ito_object_id,
        :ito_date_begin,
        :trm_id,
        :ito_status
   do
   begin
       cs_from_c_id = null;
       c_name_fio = null;
       adl_address_str= null;
       cs_qr_data = null;
       cs_phone = null;

       select cls.cs_from_c_id,
              cl.c_name_fio,
              adl.adl_address_str,
              iif (cls.cs_qr_data <> '' , 'Есть' , 'Нет'),
              cls.cs_phone

       from clientshops cls
       inner join clients cl on cl.c_id = cls.cs_from_c_id
       inner join addrlist adl on adl.adl_id = cls.cs_from_adl_id
       where cls.cs_id = :ito_object_id
       into :cs_from_c_id,
            :c_name_fio,
            :adl_address_str,
            :cs_qr_data,
            :cs_phone;

       place_name = null;
       if (ito_object_kind = obj_kind_shop_id) then
            place_name = (select cs_name
                          from clientshops
                          where cs_id = :ito_object_id);
       else
            place_name = (select rg.reg_name
                          from region rg
                          where rg.reg_unicode = :ito_object_id);
       mcc_code = null;

       select first 1 rd.rd_str_id
       from agreements ag
       inner join agreement_feac afe on afe.afe_from_ag_id = ag.ag_id
       inner join refdata rd on rd.rd_id = afe.afe_from_rd_id
       inner join refdata rd2 on rd2.rd_id = ag.ag_from_rd_id
       where ag.ag_object_kind = 0
       and ag.ag_object_id = :cs_from_c_id
       and ag.ag_date_begin between :rep_date_begin and :rep_date_end
       and rd2.rd_code in ('agreementkinds.tsp', 'agreementkinds.tsp_and_ipf')
       --order by ag.ag_id desc,
       --         afe.afe_from_rd_id
       into :mcc_code;

       office_iniatator = null;
       ita_doc_date = null;

       select reg.reg_name, ita.ita_doc_date
       from ipc_terminal_app ita
       inner join region reg on reg.reg_unicode = ita.ita_from_reg_id
       where ita.ita_from_trm_id = :trm_id
       order by ita.ita_create_datetime desc
       rows 1
       into office_iniatator, ita_doc_date;
       suspend;
   end
end
