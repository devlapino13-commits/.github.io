CREATE PROCEDURE UXP$PREBLANKS_PRESAVE(
    PREBLANK$ACTION_ID D_STR_CODE,
    PREBLANK$KIND D_STR_CODE,
    PREBLANK$CODE D_STR_CODE,
    IN_OBJ_KIND D_STR_CODE,
    IN_OBJ_ID D_STR_CODE,
    REP_REGION_ID D_INTEGER = null)
RETURNS (
    OML_ID D_INTEGER)
AS
declare variable kind_id d_integer;
declare variable save_kind d_str_code;
declare variable ua_preblank_save d_integer;
declare variable obj_kind d_str_code;
declare variable oml_regist_no d_str_code;
declare variable oml_number d_integer;
declare variable oml_categ_id d_integer;
begin
  delete from outmailnum
  where 1 = 1;

  preblank$code = trim(coalesce(preblank$code, ''));

  kind_id = f_enum('uxaction_preblank_kind', preblank$kind, 'id');
  if (kind_id is null) then
    kind_id = 0;

  if (preblank$action_id is null) then
    preblank$action_id = '';

  select ua_preblank_save
  from ux$action
  where ua_uuid = char_to_uuid(:preblank$action_id)
  into ua_preblank_save;

  save_kind = f_enum('uxaction_preblank_save', ua_preblank_save, 'code');
  if (save_kind <> 'outmailing') then
    exit;

  obj_kind = f_objkind(in_obj_kind, 'name');

  if (obj_kind <> 'kontract') then
    exit;

  select iif(:rep_region_id is null, oks_region_id, :rep_region_id)
  from oksummary
  where oks_from_k_id = :in_obj_id and
        oks_date_begin <= current_date and
        oks_date_end >= current_date
  into rep_region_id;

  select oct_id
  from outcateg
  where oct_name = 'Уведомление'
  into oml_categ_id; -- RFCRU-4114, пункт ТЗ "3b"

  if (oml_categ_id is null) then
    oml_categ_id = 0;

  insert into outmailing (oml_is_out, oml_date, oml_group_kind, oml_from_reg_id, oml_subject, oml_memo, oml_is_delinq, oml_from_oct_id)
  values (1, current_date, f_enum('outmailing_group_kind', 'office', 'id'), :rep_region_id, '', 'Предварительная запись', 1, :oml_categ_id)
  returning oml_id, oml_number, oml_regist_no
  into :oml_id, :oml_number, :oml_regist_no;

  insert into outmailnum (oml_number, oml_regist_no)
  values (:oml_number, :oml_regist_no);

  suspend;
end
