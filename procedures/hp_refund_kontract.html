CREATE PROCEDURE HP_REFUND_KONTRACT(
    K_ID D_ID,
    PARAMS D_STR_LONG)
RETURNS (
    RES_TEXT D_BLOB_TEXT,
    SHOW_RES D_STR_10)
AS
declare variable ma_id d_integer;
declare variable ma_date_app d_date;
declare variable cred_date d_date;
declare variable action_code d_str_20;
declare variable uuid d_str_code;
declare variable cred_debt_sum d_currency;
declare variable perc_sum d_currency;
declare variable ssp_summ d_currency;
declare variable debt_overdue_sum d_currency;
declare variable cred_sum d_currency;
declare variable cred_partner_type_str d_str_code;
begin
  -- RFCRU-3486 / RFCRU-4021

  show_res = 'dialog';

  -- PARAMS = "ma_id|ma_date_app|action_code|uuid|cred_debt_sum|perc_sum|ssp_summ|cred_partner_type_str|debt_overdue_sum|cred_sum|cred_date|"
  ma_id = cast(substring(params from 1 for position('|' in params) - 1) as d_integer);
  params = substring(params from position('|' in params) + 1);
  ma_date_app = cast(substring(params from 1 for position('|' in params) - 1) as d_date);
  params = substring(params from position('|' in params) + 1);
  action_code = cast(substring(params from 1 for position('|' in params) - 1) as d_str_20);
  params = substring(params from position('|' in params) + 1);
  uuid = cast(substring(params from 1 for position('|' in params) - 1) as d_str_code);
  params = substring(params from position('|' in params) + 1);

  if (char_length(params) > 0) then
  begin
    cred_debt_sum = cast(substring(params from 1 for position('|' in params) - 1) as d_currency);
    params = substring(params from position('|' in params) + 1);
    perc_sum = cast(substring(params from 1 for position('|' in params) - 1) as d_currency);
    params = substring(params from position('|' in params) + 1);
    ssp_summ = cast(substring(params from 1 for position('|' in params) - 1) as d_currency);
    params = substring(params from position('|' in params) + 1);
    cred_partner_type_str = cast(substring(params from 1 for position('|' in params) - 1) as d_str_code);
    params = substring(params from position('|' in params) + 1);
    debt_overdue_sum = cast(substring(params from 1 for position('|' in params) - 1) as d_currency);
    params = substring(params from position('|' in params) + 1);
  end

  if (char_length(params) > 0) then
  begin
    cred_sum = cast(substring(params from 1 for position('|' in params) - 1) as d_currency);
    params = substring(params from position('|' in params) + 1);
    cred_date = cast(substring(params from 1 for position('|' in params) - 1) as d_date);
  end

  if (k_id = 0) then
    k_id = null;

  if (cred_partner_type_str = '-') then
    cred_partner_type_str = null;

  if (action_code = 'insert') then
  begin
    insert into kontrefinance (kr_from_ma_id, kr_from_k_id, kr_date, kr_loan_uuid, kr_balance, kr_perc_saldo, kr_avg_repay, kr_mfo_type,
        kr_debt_overdue, kr_cred_sum, kr_cred_date)
    values (:ma_id, :k_id, :ma_date_app, :uuid, :cred_debt_sum, :perc_sum, :ssp_summ, :cred_partner_type_str,
        :debt_overdue_sum, :cred_sum, :cred_date);

    res_text = '<span>Контракт добавлен для рефинансирования</span>';
  end

  if (action_code = 'delete') then
  begin
    delete from kontrefinance
    where kr_from_ma_id = :ma_id and
          kr_loan_uuid = :uuid;

    res_text = '<span>Контракт удален c рефинансирования</span>';
  end

  suspend;
end
