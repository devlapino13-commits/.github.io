CREATE PROCEDURE AAREP_ACTIVE_CLI_INFO(
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    C_NAME_FIO STR_100,
    C_PASS_NO VARCHAR(50),
    C_PINCODE STR_20,
    C_STATUS STR_20,
    KC_ID_STR STR_20,
    KOD_FOR_MKR STR_20,
    KC_AMOUNT CURR_TYPE,
    K_PERC_YEAR PERC_TYPE,
    SUMM_KOMMIS CURR_TYPE,
    K_DISB_DATE DATE,
    K_KONT_LAST_DATE DATE,
    K_PERIOD DECIMAL(15,2),
    K_KOLICH_LGOTA INTEGER,
    K_KONTRACT_NO D_STR_255,
    SUM_MONTH_KC CURR_TYPE,
    BT_NAME VARCHAR(50),
    LOAN_TYPE VARCHAR(20),
    BS_NAME VARCHAR(50),
    RESTRUCT VARCHAR(10),
    RESTRUCT_KOL INTEGER,
    RESTRUCT_DATE_LAST DATE,
    DELINQ_DAY_RESTRUCT INTEGER,
    REPAY_OS_LAST_KC CURR_TYPE,
    DATE_REPAY_OS DATE,
    REPAY_PERC_LAST_KC CURR_TYPE,
    DATE_REPAY_PERC DATE,
    ZAD_OS_KC CURR_TYPE,
    ZAD_OS_NO_DELINQ_KC CURR_TYPE,
    ZAD_OS_DELINQ_KC CURR_TYPE,
    ZAD_OS_OPERKROFF_KC CURR_TYPE,
    ZAD_PERC_KC CURR_TYPE,
    ZAD_PERC_NO_DELINQ_KC CURR_TYPE,
    ZAD_PERC_DELINQ_KC CURR_TYPE,
    ZAD_PERC_OPERKROFF_KC CURR_TYPE,
    DELINQ_DAY_OS INTEGER,
    DELINQ_DAY_PERC INTEGER,
    MAX_DELINQ_DAY INTEGER,
    LK_PERCENT PERC_TYPE,
    RPPU_BALANCE_KC CURR_TYPE,
    RPPU_CALCPERC_KC CURR_TYPE,
    IS_SECURED VARCHAR(50),
    Z_NAME D_STR_255,
    Z_COST_MARKET CURR_TYPE,
    DATE_RATING DATE,
    DISB_REG_NAME STR_100,
    IS_AZREPAYKONT SMALLINT,
    LAF_NAME D_STR_255,
    LAL_NAME D_STR_255,
    LAM_MEMO D_STR_255)
AS
declare variable K_ID ID_TYPE;
declare variable OKS_BALANCE CURR_TYPE;
declare variable REG_ID integer;
declare variable REG_ID_PRIOR integer;
declare variable C_ID ID_TYPE;
declare variable KC_NUMBER integer;
declare variable K_BASE_K_ID ID_TYPE;
declare variable K_KOMMIS_PERC PERC_TYPE;
declare variable K_KONT_LAST_DATE2 date;
declare variable K_AMOUNT_APPROVED CURR_TYPE;
declare variable DOLYA double precision;
declare variable SUM_MONTH decimal(18,4);
declare variable KC_BUSUNESS_SUBTYPE integer;
declare variable PR_ID integer;
declare variable OL_DATE date;
declare variable REPAY_OS_LAST CURR_TYPE;
declare variable REPAY_PERC_LAST CURR_TYPE;
declare variable ZAD_OS CURR_TYPE;
declare variable ZAD_OS_NO_DELINQ CURR_TYPE;
declare variable ZAD_OS_DELINQ CURR_TYPE;
declare variable ZAD_OS_OPERKROFF CURR_TYPE;
declare variable ZAD_PERC CURR_TYPE;
declare variable ZAD_PERC_NO_DELINQ CURR_TYPE;
declare variable ZAD_PERC_DELINQ CURR_TYPE;
declare variable ZAD_PERC_OPERKROFF CURR_TYPE;
declare variable DELINQ_DATE_OS date;
declare variable DELINQ_DATE_PERC date;
declare variable LK_BALANCE CURR_TYPE;
declare variable LK_CALCPERC CURR_TYPE;
declare variable RPPU_BALANCE decimal(18,4);
declare variable RPPU_CALCPERC decimal(18,4);
declare variable DISB_REG_ID integer;
declare variable DISB_REG_ID_PRIOR integer;
declare variable LAM_ID D_ID;
declare variable PERIOD D_INTEGER;
declare variable KCP_PERC_SALDO D_CURRENCY;
begin
  reg_id_prior = 0;
  disb_reg_id_prior = 0;
  period = extract(year from rep_date_end) * 100 + extract(month from rep_date_end);
  for select oks_from_k_id, oks_balance, oks_region_id
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_close_state = 0
            --  and oks_from_k_id = 101010005479700101
            --  rows 100
      into k_id, oks_balance, reg_id
  do
  begin
    if (k_id in (101010001501000102, 101010003262100101)) then
      continue;

    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;
    end

    select k_base_k_id, k_percent_per_day * 360, k_kommis_perc, k_disb_date, k_kont_last_date, k_period, k_kolich_lgota,
           k_amount_approved, k_from_pr_id, k_fact_disb_reg_id, k_kontract_no
    from kontract
    where k_id = :k_id
    into k_base_k_id, k_perc_year, k_kommis_perc, k_disb_date, k_kont_last_date, k_period, k_kolich_lgota,
         k_amount_approved, pr_id, disb_reg_id, k_kontract_no;

    if (disb_reg_id_prior <> disb_reg_id) then
    begin
      disb_reg_id_prior = disb_reg_id;
      select reg_name
      from region
      where reg_unicode = :disb_reg_id
      into disb_reg_name;
    end

    select iif(pr_loan_type = 0, 'qrup', 'fərdi'), iif(pr_code containing '.micro_biznes.', 'biznes', 'istehlak')
    from product
    where pr_id = :pr_id
    into loan_type, bt_name;

    k_base_k_id = coalesce(k_base_k_id, k_id);

    select max(sh_date), max(iif(:k_kolich_lgota + 1 = sh_id, sh_os + sh_perc, 0))
    from schedule
    where sh_from_k_id = :k_id
    into k_kont_last_date2, sum_month;
    k_kont_last_date = coalesce(k_kont_last_date, k_kont_last_date2);

    select iif(is_restruct = 1, 'hə', 'yox'), restruct_kol, restruct_date_last
    from aarep_get_restructuring(:k_id, :rep_date_end, 94401)
    into restruct, restruct_kol, restruct_date_last;
    delinq_day_restruct = null;
    if (restruct_date_last is not null) then
    begin
      select :restruct_date_last - 1 - oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            :restruct_date_last - 1 between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into delinq_day_restruct;
      if (delinq_day_restruct < 0) then
        delinq_day_restruct = null;
    end

    ol_date = null;
    repay_os_last = null;
    repay_perc_last = null;
    select ok_ol_date, -sum(ok_os), -sum(ok_perc)
    from operkred
    where ok_ol_is_active = 1 and
          ok_from_k_id = :k_id and
          ok_type = 4 and
          ok_ol_date <= :rep_date_end
    group by ok_ol_date
    having sum(ok_os) + sum(ok_perc) < 0
    order by ok_ol_date desc
    rows 1
    into ol_date, repay_os_last, repay_perc_last;
    date_repay_os = ol_date;
    date_repay_perc = ol_date;

    if (ol_date is not null and
        (coalesce(repay_os_last, 0) = 0 or coalesce(repay_perc_last, 0) = 0)) then
    begin
      if (coalesce(repay_perc_last, 0) = 0) then
      begin
        select ok_ol_date, -sum(ok_perc)
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :k_id and
              ok_type = 4 and
              ok_ol_date <= :rep_date_end
        group by ok_ol_date
        having sum(ok_os) + sum(ok_perc) < 0 and sum(ok_perc) < 0
        order by ok_ol_date desc
        rows 1
        into date_repay_perc, repay_perc_last;
      end
      else
      if (coalesce(repay_os_last, 0) = 0) then
      begin
        select ok_ol_date, -sum(ok_os)
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :k_id and
              ok_type = 4 and
              ok_ol_date <= :rep_date_end
        group by ok_ol_date
        having sum(ok_os) + sum(ok_perc) < 0 and sum(ok_os) < 0
        order by ok_ol_date desc
        rows 1
        into date_repay_os, repay_os_last;
      end
    end

    select sum(zad_os),
           sum(iif(sh_date >= :rep_date_end, zad_os, 0)),
           sum(iif(sh_date < :rep_date_end, zad_os, 0)),
           sum(iif(sh_date is null, zad_os, 0)),
           sum(sh_calc_perc),
           sum(iif(sh_date >= :rep_date_end, sh_calc_perc, 0)),
           sum(iif(sh_date < :rep_date_end, sh_calc_perc, 0)),
           sum(iif(sh_date is null, sh_calc_perc, 0)),
           min(iif(zad_os > 0, sh_date, null)) delinq_date_os,
           min(iif(zad_perc > 0, sh_date, null)) delinq_date_perc

    from aarep_make_zad_list(:k_id, :rep_date_end, 1)
    into zad_os, zad_os_no_delinq, zad_os_delinq, zad_os_operkroff, zad_perc, zad_perc_no_delinq, zad_perc_delinq,
         zad_perc_operkroff, delinq_date_os, delinq_date_perc;

    kcp_perc_saldo = null;
    select kcp_perc_saldo from kontcalcperc
    where kcp_period = :period
    and kcp_from_k_id = :k_id 
    into kcp_perc_saldo;

    kcp_perc_saldo = coalesce(kcp_perc_saldo, 0);

    if (kcp_perc_saldo is not null) then
    begin
      zad_perc = kcp_perc_saldo;
      zad_perc_no_delinq = zad_perc - zad_perc_delinq;
    end 

    delinq_day_os = null;
    if (delinq_date_os <= :rep_date_end) then
      delinq_day_os = rep_date_end - delinq_date_os;
    delinq_day_perc = null;
    if (delinq_date_perc <= :rep_date_end) then
      delinq_day_perc = rep_date_end - delinq_date_perc;
    max_delinq_day = maxvalue(coalesce(delinq_day_os, 0), coalesce(delinq_day_perc, 0));

    select lk_balance, lk_calcperc, lk_percent
    from llc_main, llc_kont
    where lm_is_signed = 1 and
          lm_id = lk_from_lm_id and
          :rep_date_end between lk_date and lk_date_end and
          lk_from_k_id = :k_id
    into lk_balance, lk_calcperc, lk_percent;

    rppu_balance = cast(lk_balance as double precision) * lk_percent / 100;
    rppu_calcperc = cast(lk_calcperc as double precision) * lk_percent / 100;

    z_name = null;
    z_cost_market = 0;
    select list(z_name), sum(z_cost_market)
    from zalog
    where z_from_k_id = :k_id and
          z_type = 1
    into z_name, z_cost_market;

    if (z_name is not null) then
    begin
      is_secured = 'təminatlı';
      date_rating = k_disb_date;
    end
    else
    begin
      is_secured = 'təminatsız';
      z_name = 'yox';
      date_rating = null;
    end

    if (exists(select *
               from azrepaykont
               where ark_from_k_id = :k_id)) then
      is_azrepaykont = 1;
    else
      is_azrepaykont = 0;

    for select kc_from_c_id, kc_from_c_id, -- kc_id,
               kc_number, kc_amount_approved, kc_busuness_subtype
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id, kc_id_str, kc_number, kc_amount, kc_busuness_subtype
    do
    begin
      dolya = cast(kc_amount as double precision) / k_amount_approved;
      select c_name_fio, c_pass_no, c_pincode,
             case c_type
               when 0 then 'fiziki səxs'
               when 1 then 'huquqi səxs'
             end
      from clients
      where c_id = :c_id
      into c_name_fio, c_pass_no, c_pincode, c_status;

      kod_for_mkr = k_base_k_id || '-' || kc_number;
      summ_kommis = k_kommis_perc * kc_amount / 100;
      sum_month_kc = dolya * sum_month;

      select --bt_name,
bs_name
      from bustype, bussubtype
      where bt_id = bs_from_bt_id and
            bs_id = :kc_busuness_subtype
      into --bt_name,
           bs_name;
      laf_name = '';
      lam_memo = '';
      lal_name = '';
      lam_id = 0;

      select law.lam_id, law.lam_memo, lal.lal_name
      from lawkredit lak
      inner join lawmain law on law.lam_id = lak.lak_from_lam_id
      inner join lawlist lal on lal.lal_id = law.lam_from_lal_id
      where lak.lak_from_c_id = :c_id and
            lak.lak_from_k_id = :k_id
      order by law.lam_datetime desc
      rows 1
      into lam_id, lam_memo, lal_name;

      select distinct laf.laf_name
      from lawdata lad
      inner join lawfields laf on lad.lad_from_laf_id = laf.laf_id and
            laf.laf_is_active = 1
      where lad.lad_from_lam_id = :lam_id and
            lad.lad_is_done = 1
      order by lad.lad_date desc
      rows 1
      into laf_name;

      repay_os_last_kc = dolya * repay_os_last;
      repay_perc_last_kc = dolya * repay_perc_last;

      zad_os_kc = dolya * zad_os;
      zad_os_no_delinq_kc = dolya * zad_os_no_delinq;
      zad_os_delinq_kc = dolya * zad_os_delinq;
      zad_os_operkroff_kc = dolya * zad_os_operkroff;
      zad_perc_kc = dolya * zad_perc;
      zad_perc_no_delinq_kc = dolya * zad_perc_no_delinq;
      zad_perc_delinq_kc = dolya * zad_perc_delinq;
      zad_perc_operkroff_kc = dolya * zad_perc_operkroff;

      rppu_balance_kc = dolya * rppu_balance;
      rppu_calcperc_kc = dolya * rppu_calcperc;

      suspend;
    end

  end
end
