CREATE PROCEDURE AAREP_SVZ_DELINQ_REPORT(
    REP_C_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    KF_NAME VARCHAR(20),
    C_NAME_FIO VARCHAR(100),
    DELINQ_DAY INTEGER,
    DELINQ_OS CURR_TYPE,
    DELINQ_PERC CURR_TYPE,
    DELINQ_FINE CURR_TYPE,
    DELINQ_ZADOL CURR_TYPE,
    PAR7 DECIMAL(5,5),
    C_PLASTCARD_NO VARCHAR(30))
AS
declare variable oks_from_k_id decimal(18,0);
declare variable k_clients_cashier_id decimal(18,0);
begin
  par7 = null;
  for select oks_from_k_id, :rep_date_end - oks_delinq_date
      from oksummary
      where oks_kc_id = :rep_c_id and
            :rep_date_end between oks_date_begin and oks_date_end and
            :rep_date_end - oks_delinq_date between 1 and 180 and
            oks_close_state = 0 and
            not exists(select *
                       from sb_akt, sb_list -- Пункт 2.3
                       where sba_is_delete = 0 and
                             sba_status = 2 and
                             sba_from_sbl_id <> 0 and
                             :rep_date_end between sba_date_begin and sba_date_end and
                             sba_from_k_id = oks_from_k_id and
                             sba_from_sbl_id = sbl_id and
                             sbl_type = 1)
      order by 2
      into oks_from_k_id, delinq_day
  do
  begin
    if (par7 is null) then
    begin
      select cast(delinq_balance_7 / (balance_b - balance_c) * 100 as curr_type)
      from (select oks_kc_id,
                   sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 7, oks_balance, 0)) as delinq_balance_7,
                   sum(oks_balance) as balance_b, sum(iif(cast(:rep_date_end as date) - (select k_disb_date
                                                                                         from kontract
                                                                                         where k_id = oks_from_k_id) < 30, oks_balance, 0)) as balance_c
            from oksummary
            where :rep_date_end between oks_date_begin and oks_date_end and
                  oks_balance > 0 and
                  oks_kc_id = :rep_c_id
                  and oks_close_state = 0
            group by oks_kc_id
            having sum(oks_balance) - sum(iif(cast(:rep_date_end as date) - (select k_disb_date
                                                                             from kontract
                                                                             where k_id = oks_from_k_id) < 30, oks_balance, 0)) > 0) a
      into :par7;
    end

    kf_name = null;
    k_clients_cashier_id = null;
    select kal_new_kf_name, k_clients_cashier_id
    from kontract, knaktlist
    where k_id = :oks_from_k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end
    into kf_name, k_clients_cashier_id;

    c_name_fio = null;
    c_plastcard_no = null;

    select c_name_fio,
           (select cont_value
            from hp_contdata_get_cell_valid(0, c_id))
    from clients
    where c_id = :k_clients_cashier_id
    into c_name_fio, c_plastcard_no;
    if (c_plastcard_no is null) then
      c_plastcard_no = '';
    if (c_plastcard_no <> '') then
      c_plastcard_no = '+' || c_plastcard_no;

    delinq_zadol = null;

    delinq_os = null;
    delinq_perc = null;
    delinq_fine = null;

    select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine)
    from aarep_make_zad_list(:oks_from_k_id, :rep_date_end, 1)
    where sh_date <= :rep_date_end
    into delinq_os, delinq_perc, delinq_fine;

    delinq_zadol = delinq_os + delinq_perc + delinq_fine;

    suspend;
  end

end
