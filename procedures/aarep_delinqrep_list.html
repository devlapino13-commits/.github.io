      if (del_m = 8) then
        aarep_k_delinq_month = del_y || ', ' || '08 Aug';
      else
      if (del_m = 9) then
        aarep_k_delinq_month = del_y || ', ' || '09 Sep';
      else
      if (del_m = 10) then
        aarep_k_delinq_month = del_y || ', ' || '10 Oct';
      else
      if (del_m = 11) then
        aarep_k_delinq_month = del_y || ', ' || '11 Nov';

        if (sbl_name = 'Не передан в СБ') then

        sba_status_txt = 'Не передан в СБ';

      del_y = extract(year from k_disb_date);

        aarep_k_disb_month = del_y || ', ' || '01 Jan';
      else
      if (del_m = 2) then
        aarep_k_disb_month = del_y || ', ' || '02 Feb';
      else
      if (del_m = 3) then
        aarep_k_disb_month = del_y || ', ' || '03 Mar';
      else
      if (del_m = 4) then
        aarep_k_disb_month = del_y || ', ' || '04 Apr';
      else
      if (del_m = 5) then
        aarep_k_disb_month = del_y || ', ' || '05 May';
      else
      if (del_m = 6) then
        aarep_k_disb_month = del_y || ', ' || '06 Jun';
      else
      if (del_m = 7) then
        aarep_k_disb_month = del_y || ', ' || '07 Jul';
      else
      if (del_m = 8) then
        aarep_k_disb_month = del_y || ', ' || '08 Aug';
      else
      if (del_m = 9) then
        aarep_k_disb_month = del_y || ', ' || '09 Sep';
      else
      if (del_m = 10) then
        aarep_k_disb_month = del_y || ', ' || '10 Oct';
      else
      if (del_m = 11) then
        aarep_k_disb_month = del_y || ', ' || '11 Nov';
      else
      if (del_m = 12) then
        aarep_k_disb_month = del_y || ', ' || '12 Dec';

      select c_id, c_name_fio, kal_date, reg_code || ' ' || reg_name, reg_unicode, kal_new_kf_name
      from knaktlist, kontract, clients, region
      where kal_from_kf_id = :aarep_kf_id and
            knaktlist.is_delete = 0 and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            k_id = :aarep_k_id and
            kal_new_kf_officer = c_id and
            kal_new_reg_id = reg_unicode
      into :k_curr_lo_id, :k_curr_lo_fio, :k_curr_akt_date, :aa_curr_office_name, :k_curr_reg_id, :kf_name;

      if (k_disb_lo_id = k_curr_lo_id) then
        k_is_own_str = 'Своя кр.папка';
