CREATE PROCEDURE HP_EDOC_INS_LOYAL(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable ag_id d_integer;
declare variable c_comment d_str_100;
declare variable c_name_fio d_str_100;
begin
  -- Процедура создает соглашение по программе лояльность.
  -- Готовит бланки и отправляет СМС.
  res_text = '';
  if (exists(select *
             from moratory m
             where m.m_from_c_id = :c_id and
                   m.m_is_black = 2 and
                   m.m_is_active = 1)) then
  begin
    res_text = 'black_list';
    suspend;
    exit;
  end

  -- Проверка ПОД ФТ
  for
  select c_name_fio, c_comment
  from hp_check_client_fio_ft(:c_id, 1)
  into :c_name_fio, :c_comment
  do
  begin
    res_text = 'Результат ПОД ФТ: Совпадение с лицом по ' || c_comment || ' ' || c_name_fio;
  end

  if (f_clientcom_allowed(c_id, 1, 1, 1) = 0) then
    res_text = 'Отправка сообщений данному клиенту запрещена по причине отсутствия согласий';

  if (res_text = '') then
  begin
    select ag_id
    from hp_agreements_ins(:c_id, 'AgreementKinds.loyal_agreement', null, current_date, :un_id)
    into :ag_id;
  
    select edl_id, res_text
    from hp_edoc_insert('loyal.agreement', f_objkind('agreements', 'id'), :ag_id, :un_id)
    into :edl_id, :res_text;
  end

  suspend;
end
