CREATE PROCEDURE PRINT_PORUCH(
    OBJ_ID DECIMAL(18,0),
    OBJ_KIND VARCHAR(100) = 'Kontract',
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_TYPE INTEGER,
    CA_ADDRESS VARCHAR(255),
    C_FULL_ADDRESS VARCHAR(255),
    C_BIRTH_DATE DATE,
    C_EMAIL VARCHAR(100),
    C_CELL_NO VARCHAR(20),
    C_PASS_WHO_CODE VARCHAR(20),
    PHONE_HOME STR_100,
    FACT_ADDRESS STR_255,
    QR_CODE D_STR_LONG)
AS
declare variable DB_ID integer;
declare variable PR_LOAN_TYPE integer;
declare variable PR_CODE STR_100;
declare variable C_ID ID_TYPE;
declare variable EDL_ID D_ID;
declare variable ROWC D_INTEGER;
begin
  db_id = f_get_db_id();
  obj_kind = f_objkind(obj_kind, 'id');

  rowc = 0;
  if (obj_kind = 9) then -- если передан кредит
  begin
    select pr.pr_loan_type, pr.pr_code
    from kontract kont
    inner join product pr on kont.k_from_pr_id = pr.pr_id
    where kont.k_id = :obj_id
    into :pr_loan_type, pr_code;

    if (((pr_loan_type <> 0) and
        (db_id in (39801, 41701))) or (db_id not in (39801, 41701)) or (pr_code containing ('.PAR.'))) then --
    begin
      for select z_name_fio as c_name_fio, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire, c_type,
                 adl_address_str as ca_address, cl.c_birth_date,
                 (select cont_value
                  from hp_contdata_get_email_main(0, c_id)),
                 (select cont_value
                  from hp_contdata_get_cell_main(0, c_id)), cl.c_pass_who_code, cl.c_id ------------added by Aleksei on 18.02.2019
          from clients cl, zalog, addrclient, addrlist
          where z_from_k_id = :obj_id and
                z_type = 0 and
                z_from_c_id = c_id and
                adc_from_c_id = c_id and
                adc_address_type = 1 and
                adc_from_adl_id = adl_id
          order by z_id
          into :c_name_fio, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_type, :ca_address,
               :c_birth_date, :c_email, :c_cell_no, :c_pass_who_code, :c_id ------------added by Aleksei on 18.02.2019
      do
      begin
        rowc = rowc + 1;
        c_full_address = ca_address;
        phone_home = f_cont_info('clients', c_id, 'Phone_Home', 'last');

        select adl.adl_address_str
        from clients c
        inner join addrclient adc on adc.adc_from_c_id = c.c_id
        inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
        where c.c_id = :c_id and
              adc.adc_address_type = 2
        into fact_address;
         if (coalesce(add_params, '') <> '') then
        begin
          edl_id = f_json_str_get_value(add_params, 'edl_id', null);
          qr_code = f_edoc_qr_by_edl(:edl_id, :c_id);
        end



        suspend;
      end
      if (rowc = 0) then suspend;
    end
  end

  if (obj_kind = 11) then -- если передана заявка
  begin
    select pr.pr_loan_type, pr.pr_code
    from miniapp maa
    inner join product pr on maa.ma_from_pr_id = pr.pr_id
    where maa.ma_date_end >= '31.12.2999' and
          maa.ma_is_active = 1 and
          maa.ma_id = :obj_id
    into :pr_loan_type, :pr_code;

    if (((pr_loan_type <> 0) and
        (db_id in (39801, 41701))) or (db_id not in (39801, 41701)) or (pr_code containing ('.PAR.'))) then --
    begin
      for select cl.c_name_fio, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire, c_type,
                 adl_address_str as ca_address, cl.c_birth_date,
                 (select cont_value
                  from hp_contdata_get_email_main(0, c_id)),
                 (select cont_value
                  from hp_contdata_get_cell_main(0, c_id)), cl.c_pass_who_code, cl.c_id ------------added by Aleksei on 18.02.2019
          from clients cl, miniappclient macl, addrclient, addrlist
          where macl.mac_from_ma_id = :obj_id and
                macl.mac_is_poruch = 1 and
                macl.mac_from_c_id = c_id and
                adc_from_c_id = c_id and
                adc_address_type = 1 and
                adc_from_adl_id = adl_id
--order by  Z_ID
          into :c_name_fio, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_type, :ca_address,
               :c_birth_date, :c_email, :c_cell_no, :c_pass_who_code, :c_id ------------added by Aleksei on 18.02.2019
      do
      begin
        c_full_address = ca_address;

        phone_home = '';
        fact_address = '';

        phone_home = f_cont_info('clients', c_id, 'Phone_Home', 'last');

        select adl.adl_address_str
        from addrclient adc
        inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
        where adc.adc_from_c_id = :c_id and
              adc.adc_address_type = 2
        into fact_address;


        suspend;
      end
    end
  end
end
