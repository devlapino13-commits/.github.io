CREATE PROCEDURE HP_SUD_LOAD_LOANS(
    MODE D_STR_CODE,
    TAG D_SMALLINT,
    SMD_RESPONS_SAG_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable k_id d_id;
declare variable slt_id d_id;
declare variable smr_id d_id;
declare variable sag_id d_id;
declare variable kontract_no d_str_code;
declare variable status d_smallint;
declare variable err_txt d_str_255;
declare variable sud_name d_str_255;
declare variable proc_is_isk d_smallint;
declare variable sud_mainreq_obj_kind d_integer;
begin
  res_text = '';
  -- mode: clear - очистка, check - проверка, load - загрузка
  -- tag: 1 - первичная, только договор. 2 - второй этап, суды, тип производства и ответственный

  if (mode is null or coalesce(tag, 0) not in (1,2)) then
    exit;

  if (mode = 'load' and tag = 2 and smd_respons_sag_id is null) then
  begin
    res_text = 'Не выбран ответственный!';
    suspend;
    exit;
  end

  sud_mainreq_obj_kind = f_objkind('sud_mainreq', 'id');

  if (mode in ('check', 'load')) then
  begin
    for select slt_id, slt_kontract_no, slt_sud_name, iif(slt_proc_is_isk in (0,1), slt_proc_is_isk, null), slt_sud_sag_id
    from sud_loans_tmp
    where :mode = 'check'
            or (:mode = 'load' and slt_status = 1)
    into slt_id, kontract_no, sud_name, proc_is_isk, sag_id
    do
    begin
      status = 1;
      err_txt = '';

      k_id = (select k_id from kontract where k_kontract_no = :kontract_no);
      
      if (k_id is null) then
      begin
        status = -1;
        err_txt = 'Договор не найден';
      end
      else if (exists(select 1 from sud_loans_tmp
                  where slt_kontract_no = :kontract_no
                    and slt_id < :slt_id))
      then
      begin
        status = -1;
        err_txt = 'Договор дублируется';
      end

      if (status = 1 and tag = 1) then
      begin
        if (exists(select 1 from sud_mainreq, sud_mainlog
                      where sul_id = smr_last_sul_id
                        and smr_from_k_id = :k_id
                        and sul_status not in (-2, 2))) -- завершено или отказано
        then
        begin
          status = -1;
          err_txt = 'Договор есть в модуле ДРПВ в незавершенной стадии!!!';
        end
      end

      if (status = 1 and tag = 2) then
      begin
        if (proc_is_isk is null) then
        begin
          status = -1;
          err_txt = 'Не проставлено поле "Это исполнительное производство"';
        end

        if (status = 1 and sag_id is null) then
        begin
          select sag_id
          from sud_agents
          where replace(sag_name, ' ', '') = replace(:sud_name, ' ', '')
            and sag_kind = iif(:proc_is_isk = 0, 2, 3)
          rows 1
          into sag_id;

          if (sag_id is null) then
          begin
            status = -1;
            err_txt = 'Суд не найден';
          end
        end

        smr_id = null;

        select smr_id
        from sud_mainreq, sud_maindocs
        where smd_id = smr_id
          and smr_from_k_id = :k_id
          and smd_status = 0
        into smr_id;

        if (status = 1 and smr_id is null) then
        begin
          status = -1;
          err_txt = 'Договор не найден в модуле ДРПВ на стадии D1';
        end
      end

      update sud_loans_tmp
      set slt_status = :status,
          slt_error_text = :err_txt
      where slt_id = :slt_id;

      if (status = 1 and mode = 'load') then
      begin
        if (tag = 1) then
        begin
          insert into sud_mainreq (smr_from_k_id, smr_from_un_id)
          values (:k_id, :un_id)
          returning smr_id
          into smr_id;
        
          --записать клиентов
          insert into sud_loanclient (slc_sum_os, slc_sum_perc, slc_sum_fine, slc_sum_other, slc_sum_all, slc_from_smr_id, slc_sum_date, slc_address, slc_from_kc_id)
          select cur_cl_os, cur_cl_perc, cur_cl_fine, 0, cur_cl_all, :smr_id, current_date, c_address_yur_disb,
                 (select kc_id from knclient where kc_from_k_id = :k_id and kc_from_c_id = cz_from_c_id)
          from print_client_zadol(:k_id);

          update sud_mainreq
          set smr_status = 2,
              smr_from_un_id = :un_id
          where smr_id = :smr_id
            and smr_status <> 2;

          -- запись во временную таблицу для отправки досудебной претензии. Отправка в процедуре hp_sud_cession_pretrial_claim_send
          insert into temp_objlist (tol_kind, tol_object_kind, tol_object_id, tol_status, tol_date)
          values ('hp_sud_load_loans', :sud_mainreq_obj_kind, :smr_id, 0, current_date);
        end
        else if (tag = 2) then
        begin
          update sud_maindocs
          set smd_proc_type = :proc_is_isk,
              smd_sud_sag_id = :sag_id,
              smd_respons_sag_id = :smd_respons_sag_id,
              smd_status = 1,
              smd_from_un_id = :un_id
          where smd_id = :smr_id;
        end
         
        delete from sud_loans_tmp where slt_id = :slt_id;
      end
    end

    if (mode = 'load' and slt_id is null) then
    begin
      res_text = 'Нет ни одной проверенной строчки';
      suspend;
      exit;
    end
  end
  else if (mode = 'clear') then
    delete from sud_loans_tmp where 1 = 1;
end
