CREATE PROCEDURE FM_BLOBPDFDOC_SET_PDF(
    BPD_ID D_ID,
    BPD_BLOB D_BLOB_BINARY)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable save_kind d_str_code;
declare variable src_kind d_smallint;
declare variable un_id d_id;
declare variable res_is_error smallint;
begin
  res_is_error = 0;
  res_text = '';

  select bpd_object_kind, bpd_object_id, bpd_save_kind, bpd_src_kind, bpd_from_un_id
  from blobpdfdoc
  where bpd_id = :bpd_id
  into obj_kind, obj_id, save_kind, src_kind, un_id;

  if (row_count = 0) then
    exception e_default f_format('Document with bpd_id: %1 not found', bpd_id);

  if (save_kind = 'edoc_blank_scan') then
  begin
    if (f_objkind(obj_kind, 'code') <> 'edoc_blank') then
      exception e_default f_format('incorrect obj_kind for save_kind = edoc_blank_scan: %1', obj_kind);

    select res_is_error, res_text from
    hp_edoc_blank_set_scan(:obj_id, :bpd_blob, :un_id, :src_kind, 1)
    into res_is_error, res_text;
    if (res_is_error = 0) then
    begin
      -- Удаляем скан и страницы
      res_text = '';
      delete from blobpdfdoc
      where bpd_id = :bpd_id;
    end
  end
  else
    exception e_default f_format('bpd_save_kind is not implemented:%1', save_kind);

  suspend;
end
