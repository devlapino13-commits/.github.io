CREATE PROCEDURE UXP$UXSOURCE_CREATE(
    UX_CODE D_STR_CODE,
    UX_NAME D_STR_100,
    US_IS_TABLE D_BOOL,
    US_KIND D_SMALLINT,
    US_MEMO D_STR_8100,
    DONT_SYNC_IF_EXIST D_BOOL = 0)
AS
declare variable us_uuid d_uuid;
declare variable is_view d_bool;
declare variable rel_name d_str_code;
begin
  select us_uuid, us_kind
  from ux$source
  where us_code = :ux_code
  into us_uuid, us_kind;

  if (us_uuid is null) then
  begin
    is_view = 0;

    if (us_is_table is null) then
    begin
      us_is_table = 0;
      if (exists(select * from rdb$relations r
      where r.rdb$relation_name = upper(:ux_code))) then
        us_is_table = 1;
    end

    select iif(r.rdb$relation_type = 1, 1, 0)
    from rdb$relations r
    where r.rdb$relation_name = upper(:ux_code)
    into is_view;

    if (is_view = 1) then
    begin
      for select lower(trim(rdb$relation_name))
      from rdb$view_relations
      where rdb$view_name = upper(:ux_code) and
            rdb$context_type = 0
      into rel_name
      do
        execute procedure uxp$uxsource_create(:rel_name, :rel_name, 1, 0, '', 1);

      for select lower(trim(rdb$relation_name))
      from rdb$view_relations
      where rdb$view_name = upper(:ux_code) and
            rdb$context_type = 1
      into rel_name
      do
        execute procedure uxp$uxsource_create(:rel_name, :rel_name, 1, 0, '', 1);
    end

    insert into ux$source (us_code, us_name, us_is_table, us_is_view, us_kind, us_memo)
    values (:ux_code, :ux_name, :us_is_table, :is_view, :us_kind, :us_memo)
    returning us_uuid, us_kind
    into us_uuid, us_kind;
  end else
    if (dont_sync_if_exist = 1) then
      us_kind = null;

  if (us_kind = 0) then
    execute procedure uxp$uxsource_sync(:us_uuid, 1);
end
