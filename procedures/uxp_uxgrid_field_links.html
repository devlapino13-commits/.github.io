CREATE PROCEDURE UXP$UXGRID_FIELD_LINKS(
    GRID_UUID D_UUID)
RETURNS (
    CAL_FIELD_NAME D_STR_CODE,
    CAL_DATA_TYPE D_STR_CODE,
    CAL_MEMO D_STR_LONG,
    CAL_NOT_NULL D_BOOL,
    CAL_GRID_FIELD D_STR_CODE,
    CAL_COMPONENT D_STR_CODE)
AS
declare variable grid_fields d_str_4096;
begin
  select ugr_calendar_fields
  from ux$grid
  where ugr_uuid = :grid_uuid
  into grid_fields;

  if (grid_fields is null) then
    grid_fields = '';

  cal_component = 'gantt';
  for select 'g.' || cal_field_name, cal_data_type, cal_memo, cal_not_null, res_str
  from uxp$gantt_fields
       left join uxp$split_str(:grid_fields, ';') on (res_str starting 'g.' || cal_field_name || '=')
  into :cal_field_name, :cal_data_type, :cal_memo, :cal_not_null, cal_grid_field
  do
  begin
    if (cal_grid_field is null) then
      cal_grid_field = '';

    if (cal_grid_field <> '') then
      cal_grid_field = replace(cal_grid_field, cal_field_name || '=', '');

    suspend;
  end

  cal_component = 'scheduler';
  for select 's.' || cal_field_name, cal_data_type, cal_memo, cal_not_null, res_str
  from uxp$calendar_fields
       left join uxp$split_str(:grid_fields, ';') on (res_str starting 's.' || cal_field_name || '=')
  into :cal_field_name, :cal_data_type, :cal_memo, :cal_not_null, cal_grid_field
  do
  begin
    if (cal_grid_field is null) then
      cal_grid_field = '';

    if (cal_grid_field <> '') then
      cal_grid_field = replace(cal_grid_field, cal_field_name || '=', '');

    suspend;
  end
end
