Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_PASSPORT_SERIAL_NUM(
    DOC_FULL_NO D_STR_CODE,
    PK_ID D_INTEGER)
RETURNS (
    RES_DOC_SERIA D_STR_CODE,
    RES_DOC_NUMBER D_STR_CODE)
AS
declare variable serial_length d_integer;
begin
  doc_full_no = trim(coalesce(doc_full_no, ''));

  res_doc_seria = '';
  res_doc_number = '';
  doc_full_no = replace(doc_full_no, ' ', '');
  doc_full_no = replace(doc_full_no, '-', '');

  serial_length = case pk_id
                    when 12  then 4 -- Паспорт гражданина РФ
                    when 13  then 2 -- Загран. паспорт РФ
                    when 17  then 2 -- Паспорт УЗ (зелёный)
                    when 330 then 2 -- Биометрический паспорт УЗ (бордовый)
                    when 15  then 1 -- Паспорт Тадж.
                    else 0
                  end;

  if (char_length(doc_full_no) > serial_length and serial_length > 0) then
  begin
    res_doc_seria = left(doc_full_no, serial_length);
    res_doc_number = substring(doc_full_no from serial_length + 1);
  end else
    res_doc_number = doc_full_no;

  suspend;
end
