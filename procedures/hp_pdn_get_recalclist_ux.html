CREATE PROCEDURE HP_PDN_GET_RECALCLIST_UX(
    PDR_KIND D_SMALLINT,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    PDR_ID D_ID,
    PDR_DATE D_DATE,
    K_KONTRACT_NO D_STR_20,
    STATUS D_STR_CODE,
    K_DISB_DATE D_DATE,
    REPEAT_CALC_DATE D_DATE,
    REPEAT_DAYS_CNT D_INTEGER,
    REFINANCE_LIST D_STR_8100,
    CLOSE_DATE D_DATE)
AS
declare variable pdr_status d_id;
  declare variable kc_id d_id;
  declare variable ma_id d_id;
  declare variable k_id d_id;
begin
  for select pdr_id, pdr_create_date, pdr_kontract_no, pdr_status, k_disb_date, k_from_ma_app_id, kc_id, k_id
  from pdn_manual_recalclist
  left join kontract on k_id = pdr_from_k_id
  left join knclient on kc_from_k_id = k_id
  where pdr_kind = :pdr_kind
    and pdr_create_date between :rep_date_begin and :rep_date_end
  into pdr_id, pdr_date, k_kontract_no, pdr_status, k_disb_date, ma_id, kc_id, k_id
  do
  begin
    status = '';
    repeat_calc_date = null;
    repeat_days_cnt = null;
    close_date = null;

    if (pdr_status = 1) then
    begin
      select calc_status
      from hp_extinfolist_pdn_get_eil(:kc_id, current_date, null, 'equifax_40', 3)
      into status;

      if (status = '') then
        status = 'Кредитная история не найдена!!!';
    end
    else
      status = case pdr_status
                 when -3 then 'Закрыт'
                 when -2 then 'Дубль'
                 when -1 then 'Договор не найден'
                 when 0 then 'Новый'
                 when 2 then 'ПДН рассчитан'
                 else 'Неизвестно'
               end;

    select iif(oks_date_end < current_date, oks_date_end + 1, null)
    from oksummary oks
    where oks_from_k_id = :k_id
    order by oks_from_k_id, oks_date_end desc
    rows 1
    into close_date;

    select list(k_kontract_no)
    from kontrefinance kef, kontract
    where kr_from_ma_id = :ma_id
      and k_id = kr_from_k_id
    into refinance_list;

    if (pdr_status = 2) then
      select kc_date, kc_date - :k_disb_date
      from knclient_pdn
      where kc_id = :kc_id
        and kcp_kind = :pdr_kind
      into repeat_calc_date, repeat_days_cnt;

    suspend;
  end 
end
