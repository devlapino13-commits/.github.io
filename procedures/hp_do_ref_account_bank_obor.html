CREATE PROCEDURE HP_DO_REF_ACCOUNT_BANK_OBOR(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_FILL_OPERMONEY INTEGER,
    REP_OL_ID D_ID = 0)
RETURNS (
    RES_TOTAL D_INTEGER,
    RES_IMPORTED D_INTEGER,
    RES_CREATED D_INTEGER,
    RES_LINKED D_INTEGER)
AS
declare variable ol_id id_type;
declare variable ob_currency d_str_lang;
declare variable ob_curr_rate curr_type;
declare variable ob_client_dt id_type;
declare variable ob_account_dt d_str_code;
declare variable ob_summa_val curr_type;
declare variable ob_summa curr_type;
declare variable ob_from_pc_id d_str_code;
declare variable ob_from_kp_id d_str_20;
declare variable ob_account_kt d_str_code;
declare variable ob_client_kt id_type;
declare variable ob_detail d_str_long;
declare variable ob_ol_date date;
declare variable ob_ol_is_active smallint;
declare variable ob_ol_reg_id integer;
declare variable acc_main d_str_code;
declare variable acc_corr d_str_code;
declare variable is_ready integer;
declare variable main_is_dt integer;
declare variable acc_main_id id_type;
declare variable acc_corr_id id_type;
declare variable acc_dt id_type;
declare variable acc_kt id_type;
declare variable client_id id_type;
declare variable bg_corr d_str_code;
declare variable sys_full_code d_str_code;
declare variable child_ol_id d_id;
declare variable main_reg_id d_integer;
declare variable main_acc_is_branch d_smallint;
begin
  if (rep_fill_opermoney is null) then
    rep_fill_opermoney = 0;
  res_total = 0;
  res_imported = 0;
  res_created = 0;
  res_linked = 0;

  -- вставляет все операции связанные с кассой отсутствующие в KN_OperMoney
  if (abs(rep_fill_opermoney) = 1) then
  begin
    insert into kn_opermoney (kom_from_ol_id, kom_ol_date)
    select distinct ob.ob_from_ol_id, ob.ob_ol_date
    from operbook ob
    left join (kn_operbook inner join ref_account on (acc_id = kob_acc_dt and
          acc_from_sab_id in (5, 6))) on (ob.ob_from_ol_id = kob_from_ol_id)
    left join kn_opermoney on (ob.ob_from_ol_id = kom_from_ol_id)
    where ob.ob_ol_date between :rep_date_begin and :rep_date_end and
          ob.ob_ol_is_active = 1 and
          (ob_short_acc_dt starting '51-01' or
          ob_short_acc_dt starting '52-01' or
          ob_short_acc_dt starting '55-04') and
          kob_id is null and
          kom_from_ol_id is null
    union
    select distinct ob.ob_from_ol_id, ob.ob_ol_date
    from operbook ob
    left join (kn_operbook inner join ref_account on (acc_id = kob_acc_kt and
          acc_from_sab_id in (5, 6))) on (ob.ob_from_ol_id = kob_from_ol_id)
    left join kn_opermoney on (ob.ob_from_ol_id = kom_from_ol_id)
    where ob.ob_ol_date between :rep_date_begin and :rep_date_end and
          ob.ob_ol_is_active = 1 and
          (ob_short_acc_kt starting '51-01' or
          ob_short_acc_kt starting '52-01' or
          ob_short_acc_kt starting '55-04') and
          kob_id is null and
          kom_from_ol_id is null;

    res_total = res_total + row_count;

    if (rep_fill_opermoney = -1) then
    begin
      suspend;
      exit;
    end
  end

  if (rep_ol_id is null) then
    rep_ol_id = 0;

  for select kom_from_ol_id
      from kn_opermoney, operlist
      where kom_ol_date between :rep_date_begin and :rep_date_end and
            kom_is_done = 0 and
            kom_from_ol_id = ol_id and
            cast(ol_parent_id as d_id) is null and
            (cast(:rep_ol_id as d_id) = 0 or
            kom_from_ol_id = :rep_ol_id)
      into ol_id
  do
  begin
    for select ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_from_pc_id,
               ob_from_kp_id, ob_account_kt, ob_client_kt, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id
        from operbook
        where ob_from_ol_id = :ol_id and
              (((ob_short_acc_dt starting '51-01' or
              ob_short_acc_dt starting '52-01' or
              ob_short_acc_dt starting '55-04') and
              ob_short_acc_kt starting '57-04') or
              ((ob_short_acc_kt starting '51-01' or
              ob_short_acc_kt starting '52-01' or
              ob_short_acc_kt starting '55-04') and
              ob_short_acc_dt starting '57-04'))
        into ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_from_pc_id,
             ob_from_kp_id, ob_account_kt, ob_client_kt, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id
    do
    begin
      res_total = res_total + 1;
      acc_main = '';
      acc_corr = '';
      acc_main_id = null;
      acc_corr_id = null;
      client_id = null;
      is_ready = 0;
      main_is_dt = -1;

      child_ol_id = null;
      select ol_id
      from operlist
      where ol_parent_id = :ol_id rows 1
      into :child_ol_id;
      if (child_ol_id is null) then
        continue;

      if (substring(ob_account_dt from 8 for 5) in ('51-01', '55-04', '52-01')) then
      begin
        acc_main = ob_account_dt;
        acc_corr = ob_account_kt;
        client_id = ob_client_kt;
        main_is_dt = 1;
      end
      else
      begin
        acc_main = ob_account_kt;
        acc_corr = ob_account_dt;
        client_id = ob_client_dt;
        main_is_dt = 0;
      end

      select al_from_acc_id
      from acc_list
      where al_id = :acc_main
      into :acc_main_id;

      if (main_is_dt = 1) then
      begin
        ob_account_kt = null;
        acc_corr = null;

        select ob_account_kt
        from operbook
        where ob_from_ol_id = :child_ol_id and
              (ob_short_acc_kt starting '51-01' or
              ob_short_acc_kt starting '52-01' or
              ob_short_acc_kt starting '55-04') and
              ob_short_acc_dt starting '57-04' and
              ob_summa = :ob_summa
        into ob_account_kt;
        acc_corr = ob_account_kt;
      end
      else
      begin
        ob_account_dt = null;
        acc_corr = null;

        select ob_account_dt
        from operbook
        where ob_from_ol_id = :child_ol_id and
              (ob_short_acc_dt starting '51-01' or
              ob_short_acc_dt starting '52-01' or
              ob_short_acc_dt starting '55-04') and
              ob_short_acc_kt starting '57-04' and
              ob_summa = :ob_summa
        into ob_account_dt;
        acc_corr = ob_account_dt;
      end

      bg_corr = substring(acc_corr from 8 for 5);

      -- корреспондеция с денежными счетами
      select al_from_acc_id
      from acc_list
      where al_id = :acc_corr
      into :acc_corr_id;

      if (acc_corr_id is not null) then
        is_ready = 1;

      if (is_ready = 1) then
      begin
        if (acc_main_id is null) then
          exception e_default 'Main account not linked: ' || coalesce(acc_main, '<null>');

        if (acc_corr_id is null) then
          exception e_default 'Corr account not linked: ' || coalesce(acc_corr, '<null>');

        if (main_is_dt = 1) then
        begin
          acc_dt = acc_main_id;
          acc_kt = acc_corr_id;
        end
        if (main_is_dt = 0) then
        begin
          acc_dt = acc_corr_id;
          acc_kt = acc_main_id;
        end

        if (ob_currency = '810') then
          ob_summa_val = 0;
        if (char_length(ob_detail) > 255) then
          ob_detail = left(ob_detail, 255);

        execute procedure hp_do_kn_operbook_ins(:ol_id, null, :acc_dt, :acc_kt, null, null, :ob_summa, ob_summa_val,
            :ob_summa, :ob_ol_date, :ob_ol_reg_id, :ob_ol_is_active, :ob_detail, 0, 1, :ob_client_dt, :ob_client_kt,
            ob_currency, ob_curr_rate);

        update kn_opermoney
        set kom_is_done = 1
        where kom_from_ol_id = :ol_id;

        update kn_opermoney
        set kom_is_done = 1
        where kom_from_ol_id = :child_ol_id;

        res_linked = res_linked + 1;
      end
    end
  end

  for select kom_from_ol_id
      from kn_opermoney
      where kom_ol_date between :rep_date_begin and :rep_date_end and
            kom_is_done = 0 and
            (cast(:rep_ol_id as d_id) = 0 or
            kom_from_ol_id = :rep_ol_id)
      into ol_id
  do
  begin
    for select ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_from_pc_id,
               ob_from_kp_id, ob_account_kt, ob_client_kt, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id
        from operbook
        where ob_from_ol_id = :ol_id and
              (ob_short_acc_dt starting '51-01' or
              ob_short_acc_dt starting '52-01' or
              ob_short_acc_dt starting '55-04' or
              ob_short_acc_kt starting '51-01' or
              ob_short_acc_kt starting '52-01' or
              ob_short_acc_kt starting '55-04')
        into ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_from_pc_id,
             ob_from_kp_id, ob_account_kt, ob_client_kt, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id
    do
    begin
      res_total = res_total + 1;
      acc_main = '';
      acc_corr = '';
      acc_main_id = null;
      acc_corr_id = null;
      client_id = null;
      is_ready = 0;
      main_is_dt = -1;

      if (substring(ob_account_dt from 8 for 5) in ('51-01', '55-04', '52-01')) then
      begin
        acc_main = ob_account_dt;
        acc_corr = ob_account_kt;
        client_id = ob_client_kt;
        main_is_dt = 1;
      end
      else
      begin
        acc_main = ob_account_kt;
        acc_corr = ob_account_dt;
        client_id = ob_client_dt;
        main_is_dt = 0;
      end

      main_acc_is_branch = 0;
      select al_from_acc_id, al_from_reg_id, iif(acc_from_reg_id = 1010100, 0, 1)
      from acc_list, ref_account
      where al_id = :acc_main and
            al_from_acc_id = acc_id
      into :acc_main_id, :main_reg_id, main_acc_is_branch;

      bg_corr = substring(acc_corr from 8 for 5);

      if (main_acc_is_branch = 1 and
          acc_corr = '010181057-01000001') then
        acc_corr = left(acc_main, 4) || '81057-01000001';
      else
      if (bg_corr = '57-01' and
          main_reg_id = 1010100) then
        continue;

      -- корреспондеция с денежными счетами
      select al_from_acc_id
      from acc_list
      where al_id = :acc_corr
      into :acc_corr_id;

      if (acc_corr_id is not null) then
        is_ready = 1;
      else
      begin
        sys_full_code = '';
        if (bg_corr = '70-01') then
          sys_full_code = 'Сотрудники.Зарплата'; -- 60305
        if (bg_corr = '71-01') then
          sys_full_code = 'Сотрудники.АвансПодотчет'; -- 60307
        if (bg_corr = '73-02') then
          sys_full_code = 'Сотрудники.Недостача'; -- 60307
        if (bg_corr = '94-01' and
            right(acc_corr, 3) = '001') then
          sys_full_code = 'Сотрудники.Недостача';
        if (bg_corr starting '60-' or
            bg_corr starting '62-' or
            bg_corr starting '76-' or
            bg_corr starting '68-' or
            bg_corr starting '69-') then
          sys_full_code = 'Расчеты.Контрагенты';

        if (sys_full_code <> '' and
            client_id is not null) then
        begin
          is_ready = 1;

          select acc_id
          from get_ref_account_clients(:sys_full_code, '', :client_id, 0, 3, :ob_ol_date)
          into :acc_corr_id;

          if (acc_corr_id is null) then
          begin
            select acc_id
            from get_ref_account_clients(:sys_full_code, '', :client_id, 1, 3, :ob_ol_date)
            into :acc_corr_id;
          end

          if (acc_corr_id is null and
              bg_corr = '68-01' and
              right(acc_corr, 3) = '001') then
          begin
            select acc_id
            from ref_account
            where acc_code = '60301810000000000001'
            into acc_corr_id;
          end

          if (acc_corr_id is null) then
            is_ready = 0;
        end

        if (sys_full_code = 'Сотрудники.Зарплата' and
            acc_corr_id is null) then
        begin
          is_ready = 1;

          select acc_id
          from get_ref_account_clients(:sys_full_code, '', 0, 0, 3, :ob_ol_date)
          into :acc_corr_id;

          if (acc_corr_id is null) then
          begin
            select acc_id
            from get_ref_account_clients(:sys_full_code, '', 0, 1, 3, :ob_ol_date)
            into :acc_corr_id;
          end

          if (acc_corr_id is null) then
            is_ready = 0;
        end
      end

      if (is_ready = 1) then
      begin
        if (acc_main_id is null) then
          exception e_default 'Main account not linked: ' || coalesce(acc_main, '<null>');

        if (acc_corr_id is null) then
          exception e_default 'Corr account not linked: ' || coalesce(acc_corr, '<null>');

        if (main_is_dt = 1) then
        begin
          acc_dt = acc_main_id;
          acc_kt = acc_corr_id;
        end
        if (main_is_dt = 0) then
        begin
          acc_dt = acc_corr_id;
          acc_kt = acc_main_id;
        end

        if (ob_currency = '810') then
          ob_summa_val = 0;
        if (char_length(ob_detail) > 255) then
          ob_detail = left(ob_detail, 255);

        execute procedure hp_do_kn_operbook_ins(:ol_id, null, :acc_dt, :acc_kt, null, null, :ob_summa, ob_summa_val,
            :ob_summa, :ob_ol_date, :ob_ol_reg_id, :ob_ol_is_active, :ob_detail, 0, 1, :ob_client_dt, :ob_client_kt,
            ob_currency, ob_curr_rate);

        update kn_opermoney
        set kom_is_done = 1
        where kom_from_ol_id = :ol_id;

        res_linked = res_linked + 1;
      end
    end
  end

  suspend;
end
