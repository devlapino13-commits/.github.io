CREATE PROCEDURE HP_TEST_ANNU(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_K_ID D_ID = 0)
RETURNS (
    K_ID D_ID,
    CURR_ANNU D_CURRENCY,
    CURR_LAST_PAY D_CURRENCY,
    NEW_ANNU D_CURRENCY,
    NEW_LAST_PAY D_CURRENCY,
    K_AMOUNT D_CURRENCY,
    K_PERCENT_DAY D_PERCENT,
    DAYS_LIST D_STR_LONG)
AS
declare variable sh_date d_date;
declare variable sh_grace d_smallint;
declare variable sh_date_prv d_date;
declare variable days d_integer;
begin
  if (rep_k_id is null) then
    rep_k_id = 0;

  for select ks_from_k_id, k_amount_approved, k_percent_per_day
  from kontstate, kontract
  where k_id = ks_from_k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        ((:rep_k_id = 0 and ks_date_begin between :rep_date_begin and :rep_date_end) or
          ks_from_k_id = :rep_k_id)
  into k_id, k_amount, k_percent_day
  do
  begin
    curr_annu = 0;
    curr_last_pay = 0;

    select sh_os + sh_perc
    from schedule
    where sh_from_k_id = :k_id and
          sh_grace = 0 and
          sh_id > 0
    group by sh_os + sh_perc
    order by count(*) desc
    rows 1
    into curr_annu;

    if (curr_annu is null) then
      curr_annu = 0;

    select sh_os + sh_perc
    from schedule
    where sh_from_k_id = :k_id and
          sh_id > 0
    order by sh_id desc
    rows 1
    into curr_last_pay;

    if (curr_last_pay is null) then
      curr_last_pay = 0;

    days_list = '';
    sh_date_prv = null;
    for select sh_date, sh_grace
    from schedule
    where sh_from_k_id = :k_id
    order by sh_id
    into sh_date, sh_grace
    do
    begin
      if (sh_date_prv is not null) then
      begin
        days = sh_date - sh_date_prv;
        if (sh_grace = 1) then
          days = -days;

        if (days_list <> '') then
          days_list = days_list || ',';

        days_list = days_list || days;
      end

      sh_date_prv = sh_date;
    end

    new_annu = f_annu(k_amount, 0, 0, k_percent_day, days_list, 0, 0, 0, 0, 0);
    new_last_pay = f_annu(k_amount, 0, 0, k_percent_day, days_list, 0, 0, 0, 0, 1);

    suspend;
  end
end
