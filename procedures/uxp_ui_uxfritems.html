CREATE PROCEDURE UXP$UI_UXFRITEMS(
    UA_UUID D_UUID,
    RES_KIND D_STR_CODE = null,
    VER_KIND D_STR_CODE = null,
    UX_UUID D_UUID = null)
RETURNS (
    UFI_UUID D_UUID,
    UFI_CODE D_STR_CODE,
    UFI_PARENT_UUID D_UUID,
    UFI_PARENT_CODE D_STR_CODE,
    UFI_KIND D_INTEGER,
    UFI_CONTROL D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_NAME D_STR_100,
    UFI_CAPTION_ALIGN D_INTEGER,
    UFI_HINT D_STR_LONG,
    UFI_HIDDEN D_SMALLINT,
    UFP_REQUIRED D_BOOL,
    UFI_REFRESH D_BOOL,
    UFI_READONLY D_BOOL,
    UFI_GROUP_DIRECTION D_INTEGER,
    UFI_H_ALIGN D_INTEGER,
    UFI_V_ALIGN D_INTEGER,
    UFI_WIDTH D_INTEGER,
    UFI_HEIGHT D_INTEGER,
    UFI_SIZABLE D_BOOL,
    UFI_ORDER_INDEX D_SMALLINT,
    UFI_GROUP_INDEX D_SMALLINT,
    UFI_FIXED_KIND D_SMALLINT,
    UFI_SUMMARY_KIND D_SMALLINT,
    UFI_DATASET_FIELD D_STR_100,
    UFI_LOOKUP_SQL D_STR_LONG,
    UFI_UFR_CODE D_STR_100,
    UFI_UFR_UUID D_UUID,
    UFI_IS_PARAM D_SMALLINT,
    UFI_COMBO_UF_UUID D_UUID,
    FR_CODE D_STR_100,
    FR_UUID D_UUID,
    UFI_UA_UUID D_UUID,
    USR_ID_FIELD D_STR_CODE,
    USR_TEXT_FIELD D_STR_CODE,
    UA_USR_ID_FIELD D_STR_CODE,
    UA_USR_TEXT_FIELD D_STR_CODE,
    USR_EDITOR_PARAM_FIELD D_STR_CODE,
    USR_EDITOR_READONLY D_BOOL,
    UFP_EDIT_MASK D_STR_LONG,
    UFP_MASK_MESSAGE D_STR_255,
    UFI_LIST_KIND D_SMALLINT,
    UFP_PUBLIC D_BOOL,
    UFI_GLOBAL_PARAM D_STR_CODE,
    UFI_IMG_UUID_CODE D_STR_CODE,
    UFI_UF_UUID D_UUID,
    UFI_EVENT_CONTROL D_STR_CODE,
    EDI_NAME D_STR_CODE,
    USR_MULTISELECT_DELIMITER D_STR_CODE,
    UFI_BAND_INDEX D_INTEGER,
    UFI_HEADER_ALIGN D_SMALLINT,
    UFI_VALUE_ALIGN D_SMALLINT,
    UFP_UUID D_UUID,
    USR_IS_FORM_EDITOR D_BOOL,
    UFI_FONT_SIZE D_SMALLINT,
    UFI_ADD_EXPR_BUTTON D_BOOL,
    UFI_ADD_ACTION_BUTTON D_BOOL,
    UFI_ACTION_NAME D_STR_255)
AS
declare variable is_btn_edit d_bool;
declare variable parent_is_readonly d_bool;
declare variable ua_uxa_uuid d_uuid;
declare variable is_image d_bool;
declare variable blob_edit d_bool;
declare variable ufi_code_msg d_str_100;
declare variable us_uuid d_uuid;
declare variable error_text d_str_255;
declare variable uf_field_name d_str_code;
declare variable uxa_uuid d_uuid;
declare variable owner_ufp_code d_str_100;
declare variable param_locked d_bool;
declare variable item_prefix d_str_code;
declare variable ufm_uuid d_uuid;
declare variable has_modify_link d_bool;
declare variable tmp_ufr_uuid d_uuid;
declare variable ufc_hide_text d_bool;
declare variable ufi_code_owner d_str_100;
declare variable ux_kind d_str_code;
declare variable use_frame_uuid d_uuid;
declare variable ua_owner d_uuid;
declare variable ua_owner_kind d_str_code;
declare variable uf_type d_smallint;
declare variable edi_uuid d_uuid;
declare variable ua_auto_open d_bool;
declare variable uxa_fcode d_str_100;
declare variable uf_data_type d_str_code;
declare variable exp_calc_kind d_smallint;
declare variable ufc_kind d_integer;
declare variable main_us_uuid d_uuid;
declare variable erl_code d_str_code;
declare variable use_enum_code d_bool;
declare variable tmp_text d_str_255;
begin
  if (coalesce(res_kind, '') = '') then
    res_kind = trim(coalesce(res_kind, ''));

  item_prefix = 'ufi__';

  if (res_kind = 'split') then
    item_prefix = 'par__';

  if (res_kind = 'uxgrid') then
    item_prefix = 'col__';

  uxa_uuid = null;

  has_modify_link = 0;
  ufi_band_index = -1;
  ufi_header_align = 0;
  ufi_value_align = 0;
  ufi_font_size = 0;
  ufi_add_expr_button = 0;
  ufi_add_action_button = 0;
  ufi_action_name = '';

  ux_kind = uxf$code(ua_uuid, 'kind');

  if (ux_kind = 'uxaction') then
  begin
    select uxa_uuid, uxf$ver(ua_uxa_owner, :ver_kind)
    from uxp$uxalias_info(:ua_uuid)
    into :uxa_uuid, :ufm_uuid;
  end
  else
  begin
    if (res_kind = 'design') then
    begin
      ux_uuid = ua_uuid;
      res_kind = '';
    end
    else
    if (res_kind = 'unit') then
      ux_uuid = ua_uuid;
    else
    if (ux_uuid is null) then
      exception e_default 'Empty UX_UUID param';

    ux_kind = uxf$code(ux_uuid, 'kind');
    if (ux_kind = 'uxframe') then
    begin
      select ufr_uuid, uxa_uuid, ufr_owner
      from ux$frame, ux$form, ux$alias
      where ufr_uuid = :ux_uuid and
            ufr_owner = ufm_uuid and
            ufm_ux_uuid = uxa_owner rows 1
      into :use_frame_uuid, :uxa_uuid, :ufm_uuid;
    end
    else
    if (ux_kind = 'uxfritem') then
    begin
      select ufr_uuid, uxa_uuid, ufr_owner
      from ux$fritem, ux$frame, ux$form, ux$alias
      where ufi_uuid = :ux_uuid and
            ufr_uuid = ufi_ufr_uuid and
            ufr_owner = ufm_uuid and
            ufm_ux_uuid = uxa_owner rows 1
      into :use_frame_uuid, :uxa_uuid, :ufm_uuid;
    end
    else
    if (ux_kind = 'uxgrid') then
    begin
      select ufr_uuid, uxa_uuid, ufr_owner
      from ux$grid, ux$frame, ux$form, ux$alias
      where ugr_uuid = :ux_uuid and
            ufr_uuid = ugr_ufr_uuid and
            ufr_owner = ufm_uuid and
            ufm_ux_uuid = uxa_owner rows 1
      into :use_frame_uuid, :uxa_uuid, :ufm_uuid;
    end
    else
    if (ux_kind = 'uxunit') then
    begin
      uxa_uuid = ux_uuid;
      ufm_uuid = ux_uuid;
    end
    else
    begin
      select uxa_uuid, ufm_uuid
      from ux$form, ux$alias
      where ufm_uuid = :ux_uuid and
            ufm_ux_uuid = uxa_owner rows 1
      into :uxa_uuid, :ufm_uuid;
    end

    if (row_count = 0 and
        ux_kind <> 'uxunit') then
      exception e_default f_format('Unknown ux_kind: %1', ux_kind);
  end

  if (ux_uuid is null) then
    ux_uuid = ufm_uuid;

  ufi_code_owner = uxf$code(ux_uuid, 'fcode');

  if (ufm_uuid is not null and
      use_frame_uuid is null) then
  begin
    select ufr_uuid
    from ux$frame
    where ufr_owner = :ufm_uuid and
          ufr_is_mainframe = 1
    into use_frame_uuid;
  end

  if (res_kind = 'split') then
  begin
    ufi_position = 0;
    ufi_parent_uuid = null;

    ufi_kind = -1; -- Группа
    ufi_hint = uxf$lang('Параметры приготовления');
    ufi_hidden = 0;
    ufi_group_direction = 1; -- Вертикально
    ufi_h_align = 3; -- На всю ширину
    ufi_v_align = 4; -- По умолчанию
    ufi_width = 0;
    ufi_height = 0;
    ufi_caption_align = 1; -- Сверху группы
    ufi_readonly = 0;

    ufi_ufr_uuid = null;
    ufi_uf_uuid = null;
    ufi_ufr_code = '';
    uf_field_name = '';
    ufi_name = uxf$lang('Параметры');
    ufi_control = 'group';
    ufi_refresh = 0;
    ufp_public = 0;

    ufi_sizable = 1;
    ufi_order_index = 0;
    ufi_group_index = 0;
    ufi_fixed_kind = 0;
    ufi_summary_kind = 0;

    suspend;
  end

  parent_is_readonly = 0;
  if (res_kind <> 'uxgrid') then
  begin
    select ufi_readonly
    from ux$fritem
    where ufi_owner = :ux_uuid and
          ufi_parent_uuid is null
    into parent_is_readonly;
  end
  else
  begin
    select ufi_readonly
    from ux$fritem
    where ufi_uuid = :ux_uuid
    into parent_is_readonly;
  end

  ufi_position = -1;
  for select ufi_uuid, ufi_parent_uuid, ufi_kind, ufi_control, ufi_ufr_code, ufi_ufr_uuid, ufi_name, ufi_caption_align,
             ufi_hint, ufi_hidden, iif(:parent_is_readonly = 1, 1, ufi_readonly), ufi_group_direction, ufi_h_align,
             ufi_v_align, ufi_width, ufi_height, ufi_uf_uuid, uf_field_name, uf_is_param, uf_not_null, ufi_refresh,
             ufp_public, ufi_sizable, ufi_order_index, ufi_group_index, ufi_fixed_kind, ufi_summary_kind,
             ufi_event_control, ufi_band_index, ufi_header_align, ufi_value_align, ufi_font_size
      from uxp$uxfritem_temp(:ux_uuid, 1, :ufm_uuid, :res_kind)
      into :ufi_uuid, :ufi_parent_uuid, :ufi_kind, :ufi_control, ufi_ufr_code, ufi_ufr_uuid, :ufi_name,
           :ufi_caption_align, :ufi_hint, :ufi_hidden, :ufi_readonly, :ufi_group_direction, :ufi_h_align, :ufi_v_align,
           :ufi_width, :ufi_height, :ufi_uf_uuid, :ufi_dataset_field, :ufi_is_param, :ufp_required, :ufi_refresh,
           :ufp_public, :ufi_sizable, :ufi_order_index, :ufi_group_index, :ufi_fixed_kind, :ufi_summary_kind,
           :ufi_event_control, :ufi_band_index, :ufi_header_align, :ufi_value_align, :ufi_font_size
  do
  begin
    ufp_uuid = null;
    edi_uuid = null;
    edi_name = '';
    uf_type = 0;
    if (ufi_header_align = 0) then
      ufi_header_align = 3;

    if (ufp_required is null) then
      ufp_required = 0;

    if (coalesce(ufi_hint, '') = '' and
        ufi_caption_align = -1) then
      ufi_hint = ufi_name;

    ufi_name = uxf$lang(ufi_name);
    ufi_hint = uxf$lang(ufi_hint);

    if (ufi_uf_uuid is not null) then
    begin
      select ufp_uuid, uf_type, uf_edi_uuid, ufp_refresh, ufp_not_null, uf_data_type
      from ux$frparam, ux$field
      where ufp_ufr_uuid = :ufi_ufr_uuid and
            ufp_uf_uuid = :ufi_uf_uuid and
            uf_uuid = ufp_uf_uuid
      into ufp_uuid, uf_type, edi_uuid, ufi_refresh, ufp_required, uf_data_type;

      if (edi_uuid is not null) then
      begin
        select ufp_editor, uf_field_name, uf_data_type
        from ux$frparam, ux$field
        where ufp_uuid = :edi_uuid and
              ufp_uf_uuid = uf_uuid
        into ufi_control, edi_name, uf_data_type;
      end

      if (ufi_summary_kind <> 4 and
          (uf_data_type in ('string', 'blob', 'uuid'))) then
        ufi_summary_kind = 0;

      if (edi_uuid is null) then
        edi_uuid = ufp_uuid;

      if (tmp_ufr_uuid is distinct from ufi_ufr_uuid and
          ufi_ufr_uuid is not null) then
      begin
        has_modify_link = 0;

        main_us_uuid = null;

        select us_uuid
        from ux$frame, ux$source
        where ufr_uuid = :ufi_ufr_uuid and
              ufr_add_table_fields = us_code
        into main_us_uuid;

        if (exists(select *
                   from ux$frame, ux$frdata
                   where ufr_uuid = :ufi_ufr_uuid and
                         ufd_ufr_uuid = ufr_uuid and
                         ufr_us_uuid <> ufd_us_uuid and
                         (:main_us_uuid is null or
                         :main_us_uuid <> ufd_us_uuid) and
                         ufd_kind in (1, 2))) then
        begin
          if (exists(select *
                     from ux$framesql
                     where ufs_ufr_uuid = :ufi_ufr_uuid and
                           ufs_sql_kind in (1, 2))) then
            has_modify_link = 1;
        end

        tmp_ufr_uuid = ufi_ufr_uuid;
      end
    end

    ufi_global_param = '';
    if (ufi_refresh is null) then
      ufi_refresh = 0;

    if (edi_name starting 'current$' and
        edi_name <> 'current$user') then
    begin
      ufi_control = edi_name;
      ufi_global_param = ufi_control;
    end

    if (edi_name = 'current$user') then
      ufi_readonly = 1;

    -- Если параметры отображаем в отдельном окне и формируем динамически,
    -- не показываем если связан с полем или забокирован
    if (ufi_uf_uuid is not null) then
    begin
      owner_ufp_code = '';
      param_locked = 0;

      select res_owner_ufp_code, res_locked
      from uxp$uxfrparam_value(:ufp_uuid, :uxa_uuid, :ua_uuid)
      into :owner_ufp_code, param_locked;

      if (res_kind = 'split') then
      begin
        if (owner_ufp_code <> '' or
            param_locked = 1) then
          continue;
      end
      else
      if (param_locked = 1) then
        ufi_readonly = 1;
    end

    ufi_code = uuid_to_char(ufi_uuid);
    ufi_code = item_prefix || replace(ufi_code, '-', '_');

    ufi_parent_code = null;
    if (ufi_parent_uuid is not null) then
    begin
      ufi_parent_code = uuid_to_char(ufi_parent_uuid);
      ufi_parent_code = item_prefix || replace(ufi_parent_code, '-', '_');
    end

    if (uf_type = -1) then
    begin
      ufi_control = 'text';
      ufi_readonly = 1;
    end

    -- Если фрейм редактрируемый но нет связок полей для обновления поля,
    -- то редактор становится не редактируемый
    if (has_modify_link = 1 and
        ufi_is_param = 0 and
        ufi_readonly = 0 and
        uf_type = 0 and
        ufp_uuid is not null) then
    begin
      if (not exists(select *
                     from ux$framesql
                     where ufs_ufr_uuid = :ufi_ufr_uuid and
                           ufs_ufp_uuid = :ufp_uuid and
                           ufs_sql_kind in (1, 2))) then
        ufi_readonly = 1;
    end

    ufi_add_expr_button = 0;
    ufi_add_action_button = 0;

    if (ufi_readonly = 0 and
        edi_uuid is not null) then
    begin
      exp_calc_kind = null;
      select exp_calc_kind
      from ux$expression
      where exp_ux_uuid = :edi_uuid and
            exp_kind = 1
      into exp_calc_kind;

      if (exp_calc_kind = 0) then
        ufi_readonly = 1;
      else
      if (exp_calc_kind in (1, 2)) then
        ufi_add_expr_button = 1;
    end

    if (ux_kind = 'uxunit') then
      uf_field_name = ufi_dataset_field;
    else
      uf_field_name = uxf$code(:ufi_uf_uuid, 'code');

    ufi_code_msg = ufi_name;
    if (ufi_kind = 0) then
      ufi_code_msg = ufi_code_owner || ', ' || uf_field_name;

    ufi_lookup_sql = '';
    ufi_ua_uuid = null;
    usr_id_field = '';
    usr_text_field = '';
    ua_usr_id_field = '';
    ua_usr_text_field = '';
    usr_editor_readonly = 1;
    usr_editor_param_field = '';
    ufp_edit_mask = '';
    ufp_mask_message = '';

    is_btn_edit = 0;
    ufi_combo_uf_uuid = null;

    if (ufi_parent_uuid is null and
        ufi_control = 'form') then
    begin
      -- Если это форма, растягиваем на всю свободную область
      ufi_h_align = 3; -- ahClient
      ufi_v_align = 3; -- avClient
      ufi_caption_align = -1;
      ufi_name = uxf$lang('Форма');
    end

    fr_code = ufi_ufr_code;
    fr_uuid = ufi_ufr_uuid;

    if (ufi_dataset_field is null) then
      ufi_dataset_field = '';

    ufi_lookup_sql = '';
    ufi_ua_uuid = null;
    usr_id_field = '';
    usr_text_field = '';
    ua_usr_id_field = '';
    ua_usr_text_field = '';
    is_btn_edit = 0;
    ufi_combo_uf_uuid = null;
    usr_editor_readonly = 1;
    usr_editor_param_field = '';
    usr_is_form_editor = 0;
    ufi_img_uuid_code = '';

    if (ufi_control = 'combobox') then
    begin
      ufc_hide_text = 0;
      ufc_kind = 0;
      select uf_uuid, ufc_hide_text, ufc_kind
      from ux$frcombo, ux$source, ux$field
      where ufc_ufp_uuid = :edi_uuid and
            ufc_combo_source = us_code and
            uf_owner = us_uuid and
            uf_field_name = ufc_combo_field
      into ufi_combo_uf_uuid, ufc_hide_text, ufc_kind;

      if (ufc_kind <> 0) then
        ufi_control = 'combobox_radio';
      else
      if (ufc_hide_text = 1) then
        ufi_control = 'combobox_hide';
    end

    if (ufi_control = 'enum') then
    begin
      ufc_hide_text = 0;
      ufc_kind = 0;
      for select ufp_uf_uuid, erl_code, fer_use_enum_code, fer_hide_text, fer_kind
          from ux$frereflist, ux$ereflist, ux$frparam
          where ufp_uuid = :edi_uuid and
                fer_ufp_uuid = ufp_uuid and
                fer_erl_uuid = erl_uuid and
                erl_kind = 0 -- enum
          into ufi_combo_uf_uuid, erl_code, use_enum_code, ufc_hide_text, ufc_kind
      do
      begin
        if (ufc_kind = 0 and ufc_hide_text = 0 and use_enum_code = 0) then
        begin
          if (not exists(select *
                         from enum_list
                         where el_kind = :erl_code and
                               el_img_uuid is not null)) then
            continue;
        end

        ufi_control = 'combobox';

        if (ufc_kind <> 0) then
          ufi_control = 'combobox_radio';
        else
        if (ufc_hide_text = 1) then
          ufi_control = 'combobox_hide';
      end
    end

    if ((ufi_width = 0 or
        ufi_height = 0) and
        ufi_control <> 'combobox_radio') then
    begin
      select iif(:ufi_width > 0, :ufi_width, default_width), iif(:ufi_height > 0, :ufi_height, default_height)
      from uxp$control_kind
      where editor_id = :ufi_control
      into ufi_width, ufi_height;
    end

    us_uuid = null;
    error_text = '';
    ufi_list_kind = 0;

    if (ufi_control = 'lookup') then
    begin
      select ufl_us_uuid, ufl_list_kind
      from ux$frlookup, ux$source
      where ufl_ufp_uuid = :edi_uuid and
            ufl_us_uuid = us_uuid
      into us_uuid, ufi_list_kind;

      if (us_uuid is not null) then
      begin
        select lookup_sql, error_text
        from uxp$uxsource_sql_lookup(:edi_uuid)
        into :ufi_lookup_sql, error_text;

        if (error_text <> '') then
          exception e_default error_text;
      end
    end

    if (ufi_control in ('enum', 'reflist')) then
    begin
      ufi_lookup_sql = '';

      select uuid_to_char(erl_uuid)
      from ux$frereflist, ux$ereflist
      where fer_ufp_uuid = :edi_uuid and
            fer_erl_uuid = erl_uuid
      into ufi_lookup_sql;

      if (row_count = 0) then
      begin
        ufi_control = 'text';
        ufi_readonly = 1;
      end
    end

    if (ufi_control = 'mask') then
    begin
      select uma_mask, uma_message
      from ux$frmask
      where uma_ux_uuid = :edi_uuid
      into ufp_edit_mask, ufp_mask_message;

      ufp_mask_message = uxf$lang(ufp_mask_message);
    end

    if (ufi_control = 'mask' and
        ufp_edit_mask = '') then
    begin
      ufi_control = 'text';
      ufp_mask_message = '';
    end

    ufi_ua_uuid = null;
    ua_uxa_uuid = null;
    ufi_img_uuid_code = '';
    ua_owner = null;
    ufi_action_name = '';

    if (ufi_kind in (0, 2)) then
    begin
      select ufb_ua_uuid, ua_name, uuid_to_char(coalesce(ufb_img_uuid, ua_img_uuid)), ua_owner,
             uxf$code(ua_owner, 'kind')
      from ux$frbutton, ux$action
      where ufb_ufi_uuid = :ufi_uuid and
            ufb_ua_uuid = ua_uuid
      into ufi_ua_uuid, ufi_action_name, ufi_img_uuid_code, ua_owner, ua_owner_kind;

      ufi_action_name = uxf$lang(ufi_action_name);

      if (row_count = 1) then
        ufi_add_action_button = 1;
    end

    if (ufi_control in ('button', 'expressbtn', 'mediabutton', 'blob', 'blob_json', 'blob_bls', 'blob_edoc')) then
    begin
      select uuid_to_char(ufb_img_uuid)
      from ux$frbutton
      where ufb_ufi_uuid = :ufi_uuid and
            ufb_img_uuid is not null
      into ufi_img_uuid_code;
    end

    if (ufi_control = 'button') then
    begin
      ufi_add_action_button = 0;
      ufi_action_name = '';
    end

    if (ufi_ua_uuid is not null and
        res_kind <> 'unit') then
    begin
      if (ufi_ufr_uuid is null) then
      begin
        if (ua_owner_kind <> 'uxframe') then
          ufi_ufr_uuid = use_frame_uuid;
        else
          ufi_ufr_uuid = ua_owner;

        ufi_ufr_code = uxf$code(ufi_ufr_uuid, 'fcode');

        fr_uuid = ufi_ufr_uuid;
        fr_code = ufi_ufr_code;
      end

      if (not exists(select *
                     from ux$aliasaction
                     where uaa_uxa_uuid = :uxa_uuid and
                           uaa_ua_uuid = :ufi_ua_uuid)) then
        ufi_readonly = 1;
    end
    else
    if (ufi_control = 'button') then
      exception e_default f_format('Button action not found, ufi_code: %1', ufi_code_msg);

    if (ufi_control = 'button') then
    begin
      if (coalesce(ufi_name, '') = '' or
          ufi_caption_align = -1) then
      begin
        if (ufi_width = 0) then
          ufi_width = 30;
        if (ufi_h_align = 3 or
            ufi_h_align = 4) then
          ufi_h_align = 2; -- Справа
      end
    end

    -- DialogEdit - это тоже самое ButtonEdit,
    -- отличие в том что процедура UXP$UI_UXACTION будет обрабатывать его по особому
    -- Поисковик привязанный как DialogEdit будет открываться до появления формы для выбора значения

    if (ufi_control = 'dialogedit') then
      ufi_control = 'buttonedit';

    if (ufi_control = 'buttonedit') then
    begin
      ufi_control = 'text';

      if (ufi_hidden = 0 and
          res_kind <> 'unit') then
      begin
        ufi_hidden = 1;
        is_btn_edit = 1;

        tmp_text = ufi_name;
        if (coalesce(ufi_name, '') <> '') then
          ufi_name = ufi_name || ', Id';
        else
          ufi_name = 'Id';

        suspend;

        ufi_name = tmp_text;
      end
    end

    is_image = 0;
    blob_edit = 0;
    if (ufi_control in ('blob_json', 'blob_bls', 'blob_edoc', 'blob')) then
    begin
      if (ufi_hidden = 0) then
      begin
        ufi_hidden = 1;
        is_image = 1;
        if (ufi_control = 'blob') then
          blob_edit = 1;
      end
      ufi_control = 'text';
    end

    if (is_btn_edit = 0 and
        is_image = 0) then
    begin
      ufi_position = ufi_position + 1;
      suspend;
    end

    usr_multiselect_delimiter = '';
    if (is_btn_edit = 1) then
    begin
      ua_uxa_uuid = null;
      usr_id_field = '';
      usr_text_field = '';
      usr_editor_readonly = 0;
      usr_editor_param_field = '';
      usr_is_form_editor = 0;

      select ua_uuid, ua_uxa_uuid, ua_usr_id_field, ua_usr_text_field, uxa_fcode
      from ux$frparam, ux$action, ux$field, ux$alias
      where ufp_uuid = :edi_uuid and
            ufp_uf_uuid = uf_uuid and
            ufp_ufr_uuid = ua_owner and
            uf_field_name = ua_usr_id_field and
            ua_uxa_uuid = uxa_uuid
      into ufi_ua_uuid, ua_uxa_uuid, ua_usr_id_field, ua_usr_text_field, uxa_fcode;

      if (ua_uxa_uuid is not null) then
      begin
        select usr_id_field, usr_text_field, usr_editor_readonly, usr_editor_param_field
        from ux$search
        where usr_uuid = :ua_uxa_uuid
        into usr_id_field, usr_text_field, usr_editor_readonly, usr_editor_param_field;

        if (row_count = 0) then
          exception e_default f_format('Поисковик не настроен, алиас: %1', uxa_fcode);

        if (not exists(select *
                       from ux$alias, ux$code, ux$frame
                       where uxa_uuid = :ua_uxa_uuid and
                             ux_uuid = uxa_owner and
                             ufr_owner = coalesce(ux_draft_uuid, ux_release_uuid) and
                             ufr_is_grid = 1)) then
          usr_is_form_editor = 1;
      end

      if (ua_usr_text_field containing '$') then
      begin
        usr_multiselect_delimiter = ua_usr_text_field;
        usr_editor_readonly = 1;
        ua_usr_text_field = '';
      end

      if (edi_name <> '') then
      begin
        ua_usr_id_field = ufi_dataset_field;
        ua_usr_text_field = '';
      end

      if (usr_editor_readonly = 1 and
          coalesce(usr_editor_param_field, '') = '') then
      begin
        -- если в окне поисковика один фрейм и отсутствуют параметры (или они заблокированы),
        -- то в редакторе разрешаем ввод текста и передаем его в поле поиска таблицы
        ua_auto_open = 0;
        select res_id
        from uxp$uxalias_auto_open(:ufi_ua_uuid)
        into :ua_auto_open;

        if (ua_auto_open = 1) then
        begin
          usr_editor_readonly = 0;
          usr_editor_param_field = 'use$find_panel';
        end
      end

      if (coalesce(ufi_ua_uuid, '') = '') then
        exception e_default f_format('ButtonEdit action not found, ufi_code: %1', ufi_code_msg);

      if (ua_usr_text_field = '') then
      begin
        -- если поле нужно формировать динамически, получаем текст SQL
        select ux_us_uuid
        from ux$alias, ux$code
        where uxa_uuid = :ua_uxa_uuid and
              uxa_owner = ux_uuid
        into us_uuid;

        if (us_uuid is not null) then
        begin
          select keyname_sql, error_text
          from uxp$uxsource_sql_prepare(:us_uuid, :ufi_dataset_field, uuid_to_char(:ua_uxa_uuid))
          into :ufi_lookup_sql, error_text;

          if (error_text <> '') then
            exception e_default error_text;
        end

        if (ufi_lookup_sql is null) then
          ufi_lookup_sql = '';

        if (ufi_lookup_sql = '') then
          exception e_default f_format('Keyname lookup sql not found, ufi_code: %1', ufi_code_msg);

        ua_usr_text_field = ua_usr_id_field || '$text';
      end

      ufi_code = ufi_code || '_text';
      ufi_position = ufi_position + 1;
      ufi_dataset_field = ua_usr_text_field;
      ufi_hidden = 0;
      ufi_control = 'buttonedit';

      suspend;

      usr_multiselect_delimiter = '';
      ufi_ua_uuid = null;
    end

    if (is_image = 1) then
    begin
      ufi_dataset_field = ufi_dataset_field || '$text';
      ufi_code = ufi_code || '_text';
      ufi_hidden = 0;
      ufi_position = ufi_position + 1;

      if (blob_edit = 0) then
        ufi_control = 'blob_json';
      else
        ufi_control = 'blob';

      suspend;
    end
  end
end
