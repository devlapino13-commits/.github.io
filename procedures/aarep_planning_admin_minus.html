CREATE PROCEDURE AAREP_PLANNING_ADMIN_MINUS(
    REP_REGION_ID INTEGER,
    REP_APLIC_DATE DATE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    MA_DATE_OPER DATE,
    MA_SUMMA_PLUS_OS DECIMAL(18,4),
    MA_SUMMA_PLUS_PERC DECIMAL(18,4),
    MA_SUMMA_PLUS_ADMIN DECIMAL(18,4),
    MA_ADMIN_PLUS DECIMAL(18,4),
    MA_ADMIN_MINUS DECIMAL(18,4),
    MA_TRANSFER_PLUS DECIMAL(18,4),
    MA_TRANSFER_MINUS DECIMAL(18,4),
    MA_SUMMA_MINUS_REPEAT DECIMAL(18,4),
    MA_SUMMA_MINUS_NEW DECIMAL(18,4),
    MA_SUMMA_MINIAPPNEW DECIMAL(18,4),
    MA_ADMIN_MINUS_PROCESS DECIMAL(18,4),
    MA_FROM_RM_ID DECIMAL(18,0),
    MA_FROM_RL_NUMBER INTEGER,
    MA_RL_TEXT VARCHAR(255),
    MA_RL_KOLICH DECIMAL(18,4),
    MA_RL_CORR_OFFICE INTEGER,
    MA_OFFICE INTEGER,
    RM_FROM_C_ID DECIMAL(18,0),
    MA_IS_MINI_APP INTEGER,
    MA_IS_DELIQ VARCHAR(20),
    MA_DATE_ORIG DATE,
    MA_RM_NO INTEGER,
    MA_OFFICE_NAME VARCHAR(100))
AS
declare variable reg_id_prior integer;
declare variable ma_id_prior integer;
declare variable ma_is_new integer;
declare variable rm_status integer;
declare variable rs timestamp;
declare variable rm_id decimal(18,0);
begin
  rs = rep_date_begin + 1;
  reg_id_prior = -1;
  ma_id_prior = -1;
  ma_summa_minus_repeat = 0;
  ma_summa_minus_new = 0;
  ma_summa_plus_os = 0;
  ma_summa_plus_perc = 0;
  ma_summa_plus_admin = 0;
  ma_transfer_plus = 0;
  ma_transfer_minus = 0;
  ma_admin_plus = 0;
  ma_admin_minus = 0;
  ma_is_new = 0;
  ma_summa_miniappnew = 0;
  ma_admin_minus_process = 0;

  /* Админ расходы, трансферы */
  for select rm_id, rm_from_c_id, rm_from_reg_id, rl_from_rm_id, rl_number, rl_text, rl_kolich, rl_corr_office, rl_date,
             rl_summa_plus, rl_summa_minus, rm_from_rk_id, rm_status, rm_no, reg_code || ' ' || reg_name
      from reqlist, reqmain, region_list(:rep_region_id, 1, -1, 1), region
      where rl_from_rm_id = rm_id and
            rm_from_reg_id = reg_unicode and
            cast(rm_date as date) <= :rep_aplic_date and
            rl_date between :rep_date_begin and :rep_date_end and
            (rm_from_reg_id = id_list)
      into :rm_id, :rm_from_c_id, :ma_office, :ma_from_rm_id, :ma_from_rl_number, :ma_rl_text, :ma_rl_kolich,
           :ma_rl_corr_office, :ma_date_oper, :ma_admin_plus, :ma_admin_minus, :ma_is_mini_app, :rm_status, :ma_rm_no,
           :ma_office_name
  do
  begin

    rm_status = -2;
    select distinct rs_status
    from reqsign
    where rs_from_rm_id = :rm_id and
          rs_datetime < :rs
    order by rs_number desc
    rows 1
    into :rm_status;
    if (rm_status is null) then
      rm_status = -2;

    ma_is_deliq = '';
    ma_date_orig = :ma_date_oper;
    ma_admin_minus_process = 0;
    ma_transfer_plus = 0;
    ma_transfer_minus = 0;
    if (ma_is_mini_app = 0) then
    begin
      ma_transfer_plus = :ma_admin_plus;
      ma_transfer_minus = :ma_admin_minus;
      ma_admin_plus = 0;
      ma_admin_minus = 0;
    end
    if ((rm_status = 0) or (rm_status = 4) or (rm_status = 5)) then
    begin
      ma_admin_minus_process = ma_admin_minus;
      ma_admin_minus = 0;
    end
    if (ma_date_oper < :rep_aplic_date) then
    begin
      ma_is_deliq = 'Просрочена';
      ma_date_orig = :rep_aplic_date;
      ma_rl_text = 'Просроч.' || cast(ma_date_oper as varchar(10)) || ' ' || ma_rl_text;
      ma_date_oper = :rep_date_begin;
    end
    if (rm_status >= 0) then
      suspend;
  end
  ma_admin_minus_process = 0;
  /* Корреспондирующая часть трансферов */

  for select rm_id, rm_from_c_id, rm_from_reg_id, rl_from_rm_id, rl_number, rl_text, rl_kolich, rl_corr_office, rl_date,
             rl_summa_plus, rl_summa_minus, rm_from_rk_id, rm_no, reg_code || ' ' || reg_name
      from reqlist, reqmain, region_list(:rep_region_id, 1, -1, 1), region
      where rl_from_rm_id = rm_id and
            rl_corr_office = reg_unicode and
            cast(rm_date as date) <= :rep_aplic_date and
            rl_date between :rep_date_begin and :rep_date_end and
            rm_from_rk_id = 0 and
            (rl_corr_office = id_list)
      into :rm_id, :rm_from_c_id, :ma_rl_corr_office, :ma_from_rm_id, :ma_from_rl_number, :ma_rl_text, :ma_rl_kolich,
           :ma_office, :ma_date_oper, :ma_admin_minus, :ma_admin_plus, :ma_is_mini_app, :ma_rm_no, :ma_office_name
  do
  begin

    rm_status = -2;
    select distinct rs_status
    from reqsign
    where rs_from_rm_id = :rm_id and
          rs_datetime < :rs
    order by rs_number desc
    rows 1
    into :rm_status;
    if (rm_status is null) then
      rm_status = -2;

    ma_is_deliq = '';
    ma_date_orig = :ma_date_oper;
    ma_transfer_plus = 0;
    ma_transfer_minus = 0;
    if (ma_is_mini_app = 0) then
    begin
      ma_transfer_plus = :ma_admin_plus;
      ma_transfer_minus = :ma_admin_minus;
      ma_admin_plus = 0;
      ma_admin_minus = 0;
    end
    ma_rl_text = 'Передача денежных средств';
    if (ma_date_oper < :rep_aplic_date) then
    begin
      ma_is_deliq = 'Просрочена';
      ma_date_orig = :rep_aplic_date;
      ma_rl_text = 'Просроч.' || cast(ma_date_oper as varchar(10)) || ' ' || ma_rl_text;
      ma_date_oper = :rep_date_begin;
    end
    if (rm_status >= 0) then
      suspend;
  end
end
