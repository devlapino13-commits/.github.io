CREATE PROCEDURE HP_GET_KPD(
    KC_ID D_ID,
    REP_DATE D_DATE = null)
RETURNS (
    MONTHLY_INCOME DECIMAL(15,2),
    MONTHLY_INCOME_ERR D_STR_255,
    EXTRA_AMOUNT DECIMAL(15,2),
    EXTRA_AMOUNT_ERR D_STR_255,
    COMMUNAL_EXPENSES DECIMAL(15,2),
    COMMUNAL_EXPENSES_ERR D_STR_255,
    LIVING_EXPENSES DECIMAL(15,2),
    LIVING_EXPENSES_ERR D_STR_255,
    LEGAL_DOCS_EXPENSES DECIMAL(15,2),
    LEGAL_DOCS_EXPENSES_ERR D_STR_255,
    OTHER_BUSINESS_EXPENSES DECIMAL(15,2),
    OTHER_BUSINESS_EXPENSES_ERR D_STR_255,
    REPAYMENT_OBLIGATIONS DECIMAL(15,2),
    REPAYMENT_OBLIGATIONS_ERR D_STR_255,
    NET_PROFIT DECIMAL(15,2),
    NET_PROFIT_ERR D_STR_255,
    AMOUNT_MONTHLY_PAYMENTS DECIMAL(15,2),
    AMOUNT_MONTHLY_PAYMENTS_ERR D_STR_255,
    LASTCH_DATETIME D_DATETIME,
    KPD DECIMAL(15,2),
    KPD_ERR D_STR_255)
AS
declare variable c_id id_type;
declare variable kc_amount_approved curr_type;
declare variable ovs_value str_100;
declare variable kf_client_fio d_str_100;
declare variable ma_id d_integer;
declare variable mac_id d_integer;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
begin
  lastch_datetime = null;

  for select knc.kc_from_c_id, knc.kc_amount_approved, cl.c_name_fio, kont.k_from_ma_app_id
      from knclient knc
      inner join kontract kont on kont.k_id = knc.kc_from_k_id
      inner join clients cl on cl.c_id = knc.kc_from_c_id
      where knc.kc_id = :kc_id
      into :c_id, kc_amount_approved, kf_client_fio, ma_id
  do
  begin
    kpd = null;
    mac_id = (select mac_id from miniappclient
      where mac_from_ma_id = :ma_id and mac_from_c_id = :c_id);

    obj_kind = f_objkind('MiniAppClient', 'id');
    obj_id = mac_id;
    if (not exists(
      select *
      from actionlist
      where acl_object_kind = :obj_kind
        and acl_object_id = :obj_id
        and acl_from_rd_id = f_get_rd_id('Anketa.Solvency'))) then
    begin
      obj_kind = f_objkind('Clients', 'id');
      obj_id = c_id;
    end

    --------------------------------------------------------------------------------------------------
    ovs_value = null;
    monthly_income = null;
    monthly_income_err = '';

    select ovs_value, iif(:lastch_datetime > ovs_datetime, :lastch_datetime, ovs_datetime)
    from hp_objectprops_get(:obj_kind, :obj_id, -1, 'monthly_income', :rep_date)
    into :ovs_value, :lastch_datetime; -- ежемесячный основной доход № 1

    ovs_value = replace(ovs_value, 'значение: ', '');
    ovs_value = replace(ovs_value, 'значение', '');

    if (ovs_value <> '') then
    begin
      monthly_income = cast(ovs_value as decimal);
      when any do
      begin
        monthly_income_err = '"Ежемесячный основной доход" заполенно не корректно [ ' || kf_client_fio || ' значение: ' || ovs_value || ']';
      end
    end
    else
    begin
      monthly_income_err = 'Не заполнено "Ежемесячный основной доход" [ФИО: ' || kf_client_fio || ']';
    end

    --------------------------------------------------------------------------------------------------
    ovs_value = null;
    extra_amount = null;
    extra_amount_err = '';

    select ovs_value, iif(:lastch_datetime > ovs_datetime, :lastch_datetime, ovs_datetime)
    from hp_objectprops_get(:obj_kind, :obj_id, -1, 'extra_amount', :rep_date)
    into :ovs_value, :lastch_datetime; -- сумма ежемесячных дополнительных доходов № 2

    ovs_value = replace(ovs_value, 'Значение: ', '');
    ovs_value = replace(ovs_value, 'Значение', '');

    if (ovs_value <> '') then
    begin
      extra_amount = cast(ovs_value as decimal);
      when any do
      begin
        extra_amount_err = '"Сумма ежемесячных дополнительных доходов" заполенно не корректно [ ' || KF_CLIENT_FIO || ' значение: ' || OVS_VALUE || ']';
      end
    end
    else
    begin
      extra_amount_err = 'Не заполнено "Сумма ежемесячных дополнительных доходов" [ФИО: ' || KF_CLIENT_FIO || ']';
    end
    --------------------------------------------------------------------------------------------------
    ovs_value = null;
    communal_expenses = null;
    communal_expenses_err = '';

    select ovs_value, iif(:lastch_datetime > ovs_datetime, :lastch_datetime, ovs_datetime)
    from hp_objectprops_get(:obj_kind, :obj_id, -1, 'AnketaQuestions.communal_expenses', :rep_date)
    into :ovs_value, :lastch_datetime; -- Ежемесячные расходы на жилье   № 3

    ovs_value = replace(ovs_value, 'Значение: ', '');
    ovs_value = replace(ovs_value, 'Значение', '');

    if (ovs_value <> '') then
    begin
      communal_expenses = cast(ovs_value as decimal);
      when any do
      begin
        communal_expenses_err = '"Ежемесячные расходы на жилье" заполенно не корректно [ФИО: ' || KF_CLIENT_FIO || ' значение: ' || OVS_VALUE || ']';
      end
    end
    else
    begin
      communal_expenses_err = 'Не заполнено "Ежемесячные расходы на жилье" [ФИО: ' || KF_CLIENT_FIO || ']';
    end
    --------------------------------------------------------------------------------------------------
    ovs_value = null;
    living_expenses = null;
    living_expenses_err = '';

    select ovs_value, iif(:lastch_datetime > ovs_datetime, :lastch_datetime, ovs_datetime)
    from hp_objectprops_get(:obj_kind, :obj_id, -1, 'living_expenses', :rep_date)
    into :ovs_value, :lastch_datetime; -- ежемесячные семейные расходы  № 4

    ovs_value = replace(ovs_value, 'Значение: ', '');
    ovs_value = replace(ovs_value, 'Значение', '');

    if (ovs_value <> '') then
    begin
      living_expenses = cast(ovs_value as decimal);
      when any do
      begin
        living_expenses_err = '"Ежемесячные семейные расходы" заполенно не корректно [ФИО: ' || KF_CLIENT_FIO || ' значение: ' || OVS_VALUE || ']';
      end
    end
    else
    begin
      living_expenses_err = 'Не заполнено "Ежемесячные семейные расходы" [ФИО: ' || KF_CLIENT_FIO || ']';
    end
    --------------------------------------------------------------------------------------------------
    OVS_VALUE = null;
    LEGAL_DOCS_EXPENSES = null;
    LEGAL_DOCS_EXPENSES_ERR = '';

    select OVS_VALUE, iif(:lastch_datetime > ovs_datetime, :lastch_datetime, ovs_datetime)
    from HP_OBJECTPROPS_GET(:obj_kind, :obj_id, -1, 'AnketaQuestions.legal_docs_expenses', :rep_date)
    into :OVS_VALUE, :lastch_datetime; -- Расходы на патент и соц. полис (в месяц)  № 6

    OVS_VALUE = replace(OVS_VALUE, 'Значение: ', '');
    OVS_VALUE = replace(OVS_VALUE, 'Значение', '');

    if (OVS_VALUE <> '') then
    begin
      LEGAL_DOCS_EXPENSES = cast(OVS_VALUE as decimal);
      when any do
      begin
        LEGAL_DOCS_EXPENSES_ERR = '"Расходы на патент и соц. полис (в месяц)" заполенно не корректно [ФИО: ' || KF_CLIENT_FIO || ' значение: ' || OVS_VALUE || ']';
      end
    end
    else
    begin
      LEGAL_DOCS_EXPENSES_ERR = 'Не заполнено "Расходы на патент и соц. полис (в месяц)" [ФИО: ' || KF_CLIENT_FIO || ']';
    end
    --------------------------------------------------------------------------------------------------
    OVS_VALUE = null;
    OTHER_BUSINESS_EXPENSES = null;
    OTHER_BUSINESS_EXPENSES_ERR = '';

    select OVS_VALUE, iif(:lastch_datetime > ovs_datetime, :lastch_datetime, ovs_datetime)
    from HP_OBJECTPROPS_GET(:obj_kind, :obj_id, -1, 'AnketaQuestions.Other_business_expenses', :rep_date)
    into :OVS_VALUE, :lastch_datetime; -- Другие ежемесячные расходы по бизнесу   № 7

    OVS_VALUE = replace(OVS_VALUE, 'Значение: ', '');
    OVS_VALUE = replace(OVS_VALUE, 'Значение', '');

    if (OVS_VALUE <> '') then
    begin
      OTHER_BUSINESS_EXPENSES = cast(OVS_VALUE as decimal);
      when any do
      begin
        OTHER_BUSINESS_EXPENSES_ERR = '"Другие ежемесячные расходы по бизнесу" заполенно не корректно [ФИО: ' || KF_CLIENT_FIO || ' значение: ' || OVS_VALUE || ']';
      end
    end
    else
    begin
      OTHER_BUSINESS_EXPENSES_ERR = 'Не заполнено "Другие ежемесячные расходы по бизнесу" [ФИО: ' || KF_CLIENT_FIO || ']';
    end
    --------------------------------------------------------------------------------------------------
    ovs_value = null;
    repayment_obligations = null;
    repayment_obligations_err = '';

    select ovs_value, iif(:lastch_datetime > ovs_datetime, :lastch_datetime, ovs_datetime)
    from hp_objectprops_get(:obj_kind, :obj_id, -1, 'AnketaQuestions.repayment_obligations', :rep_date)
    into :ovs_value, :lastch_datetime; -- погашение других долговых обязательств  № 8

    ovs_value = replace(ovs_value, 'Значение: ', '');
    ovs_value = replace(ovs_value, 'Значение', '');

    if (ovs_value <> '') then
    begin
      repayment_obligations = cast(ovs_value as decimal);
      when any do
      begin
        repayment_obligations_err = '"Погашение других долговых обязательств" заполенно не корректно [ФИО: ' || KF_CLIENT_FIO || ' значение: ' || OVS_VALUE || ']';
      end
    end
    else
    begin
      repayment_obligations_err = 'Не заполнено "Погашение других долговых обязательств" [ФИО: ' || KF_CLIENT_FIO || ']';
    end
    --------------------------------------------------------------------------------------------------
    ovs_value = null;
    net_profit = null;
    net_profit_err = '';

    select ovs_value, iif(:lastch_datetime > ovs_datetime, :lastch_datetime, ovs_datetime)
    from hp_objectprops_get(:obj_kind, :obj_id, -1, 'net_profit', :rep_date)
    into :ovs_value, :lastch_datetime; -- чистая прибыль № 9

    ovs_value = replace(ovs_value, 'Значение: ', '');
    ovs_value = replace(ovs_value, 'Значение', '');

    if ((ovs_value <> '') or (ovs_value ='0')) then
    begin
      net_profit = cast(ovs_value as decimal);
      when any do
      begin
        net_profit_err = '"чистая прибыль" заполенно не корректно [ФИО: ' || KF_CLIENT_FIO || ' значение: ' || OVS_VALUE || ']';
      end
    end
    else
    begin
      net_profit_err = 'Не заполнено "чистая прибыль" [ФИО: ' || KF_CLIENT_FIO || ']';
    end
    --------------------------------------------------------------------------------------------------
    ovs_value = null;
    amount_monthly_payments = null;
    amount_monthly_payments_err = '';

    select ovs_value, iif(:lastch_datetime > ovs_datetime, :lastch_datetime, ovs_datetime)
    from hp_objectprops_get(:obj_kind, :obj_id, -1, 'amount_monthly_payments', :rep_date)
    into :ovs_value, :lastch_datetime; -- сумма ежемесчной выплаты № 10

    ovs_value = replace(ovs_value, 'значение: ', '');
    ovs_value = replace(ovs_value, 'значение', '');

    if (ovs_value <> '') then
    begin
      amount_monthly_payments = cast(ovs_value as decimal);
      when any do
      begin
        amount_monthly_payments_err = '"сумма ежемесячной выплаты" заполенно не корректно [ФИО: ' || KF_CLIENT_FIO || ' значение: ' || OVS_VALUE || ']';
      end
    end
    else
    begin
      amount_monthly_payments_err = 'Не заполнено "сумма ежемесчной выплаты" [ФИО: ' || KF_CLIENT_FIO || ']';
    end
    ----------------------------------------------------------------------------
    kpd_err = '';

    if ((net_profit is not null) and
        (amount_monthly_payments is not null) and (amount_monthly_payments > 0)) then
    begin
      :kpd = (net_profit / amount_monthly_payments) * 100;

      if (:kpd < 130) then
      begin
        kpd_err = 'КПД должно быть > 130% ' || kf_client_fio; -- Для всех продуктов
        suspend;
        exit;
      end

      if ((kc_amount_approved > 200000) and
          (:kpd < 140)) then
      begin
        kpd_err = 'КПД должно быть не менее 140% , т.к. сумма займа более 200 000' || kf_client_fio; -- для всех продуктов
        suspend;
        exit;
      end
    end
    else if (amount_monthly_payments = 0) then
    begin
        amount_monthly_payments_err = '"сумма ежемесячной выплаты" не может быть ровно нулю [ФИО: ' || kf_client_fio || ']';
    end
    if ((monthly_income is not null) and
        (extra_amount is not null) and
        (communal_expenses is not null) and
        (living_expenses is not null) and
        --(unforeseen_expenses is not null) and      board -1263
        (legal_docs_expenses is not null) and
        (other_business_expenses is not null) and
        (repayment_obligations is not null) and
        (net_profit is not null)) then
    begin
      if (net_profit <> monthly_income + extra_amount - communal_expenses - living_expenses - /*unforeseen_expenses - */legal_docs_expenses - other_business_expenses - repayment_obligations) then
      begin
        kpd_err = 'чистая прибыль указана не корректно ФИО: ' || KF_CLIENT_FIO || '  ' || NET_PROFIT || ' !<> ' || MONTHLY_INCOME || ' + ' || EXTRA_AMOUNT || ' - ' || COMMUNAL_EXPENSES || ' - ' || LIVING_EXPENSES || ' - ' ||/* UNFORESEEN_EXPENSES ||*/ ' - ' || LEGAL_DOCS_EXPENSES || ' - ' || OTHER_BUSINESS_EXPENSES || ' - ' || REPAYMENT_OBLIGATIONS; -- Для всех продуктов
      end
    end

    suspend;
  end
end
