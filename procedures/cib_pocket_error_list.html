CREATE PROCEDURE CIB_POCKET_ERROR_LIST
RETURNS (
    CP_ID D_INTEGER,
    CP_COUNTER D_INTEGER,
    CP_CIB_NAME STR_100,
    CP_DATE_BEGIN D_STR_100,
    CP_DATE_END D_STR_100,
    CP_POCKET_NAME D_STR_100,
    CP_POCKET_TYPE STR_100,
    CP_CREATE_DATE D_STR_100,
    CP_STATUS_DATETIME D_STR_100,
    CP_ERROR_CHECKED D_SMALLINT,
    CP_PERIOD STR_100)
AS
declare variable DB_ID D_INTEGER;
begin
    db_id = f_get_db_id();

    for select cp_id,
               cp_counter,
               f_datetostr(cp_date_begin),
               f_datetostr(cp_date_end),
               cp_pocket_name,
               cp_pocket_type,
               f_datetimetostr(cp_datetime),
               f_datetimetostr(cp_status_datetime),
               cp_error_checked,
               rd.rd_name
        from cib_pocket cp
        left join refdata rd on rd.rd_id = cp.cp_cib_id
        where ((cp.cp_pocket_type like '%.kont.%') or (:db_id in (39801, 41701)))
        and cp.cp_pocket_kib_status = iif(:db_id in (39801, 41701), 'FinishedWithErrors', 'Processed')
        order by cp.cp_id desc
        into :cp_id,
             :cp_counter,
             :cp_date_begin,
             :cp_date_end,
             :cp_pocket_name,
             :cp_pocket_type,
             :cp_create_date,
             :cp_status_datetime,
             :cp_error_checked,
             :cp_cib_name
    do
    begin
        if ((db_id = 81001) and (cp_id < 8958)) then
            continue;

        cp_date_begin = iif(cp_date_begin = '', '[не указан]', cp_date_begin);
        cp_date_end = iif(cp_date_end = '', '[не указан]', cp_date_end);
        cp_period = 'c '||cp_date_begin||' по '||cp_date_end;

        if ((cp_pocket_type like '%.d.%') and (cp_pocket_type like '%.kont.%')) then
            cp_pocket_type = 'Ежедневный, кредитный пакет';

        if (db_id in (39801, 41701)) then
        begin
            if (cp_id >= 499) then
                suspend;
        end
        else
            suspend;
    end
end
