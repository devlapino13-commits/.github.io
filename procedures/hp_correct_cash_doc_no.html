CREATE PROCEDURE HP_CORRECT_CASH_DOC_NO(
    OL_YEAR INTEGER)
RETURNS (
    OL_ID DECIMAL(18,0),
    OL_DATE DATE,
    MAX_NO INTEGER)
AS
declare variable ol_date_prior date;
declare variable ol_id_prior decimal(18,0);
declare variable ol_doc_no_prior integer;
begin
  ol_date_prior = cast('NOW' as date);
  ol_id_prior = 0;
  ol_doc_no_prior = 0;
  max_no = 0;
  for select distinct ol_id, ol_date
      from operlist, operbook
      where ol_id = ob_from_ol_id and
            ob_account_dt like '%81050-01%' and
            extract(year from ol_date) = :ol_year and
            operlist.is_delete = 0 and
            ob_summa <> 0
      order by ol_date, ol_id
      into :ol_id, :ol_date
  do
  begin
    max_no = max_no + 1;
    update operlist
    set ol_doc_no = :max_no,
        ol_doc_no_str = :max_no
    where ol_id = :ol_id;
  end

  max_no = 0;
  for select distinct ol_id, ol_date
      from operlist, operbook
      where ol_id = ob_from_ol_id and
            ob_account_kt like '%81050-01%' and
            extract(year from ol_date) = :ol_year and
            operlist.is_delete = 0 and
            ob_summa <> 0
      order by ol_date, ol_id
      into :ol_id, :ol_date
  do
  begin
    max_no = max_no + 1;
    update operlist
    set ol_doc_no = :max_no,
        ol_doc_no_str = :max_no
    where ol_id = :ol_id;
  end

end
