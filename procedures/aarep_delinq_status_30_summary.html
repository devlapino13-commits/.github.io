CREATE PROCEDURE AAREP_DELINQ_STATUS_30_SUMMARY(
    REP_DATE_END DATE,
    IS_CATEGORY_3 INTEGER)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(60),
    AA_AMOUNT_ACTIVE_END CURR_TYPE,
    AA_JUST_CL_END INTEGER,
    AA_KOLICH_ACTIVE_END INTEGER,
    AA_DELIQ_KOLICH30 INTEGER,
    PAR7 CURR_TYPE,
    PAR30 CURR_TYPE,
    ZAD_OS CURR_TYPE,
    SH_CALC_PERC CURR_TYPE,
    SH_CALC_FINE CURR_TYPE,
    IN_OFFICE INTEGER,
    IN_COLLECTOR_RU INTEGER,
    IN_COLLECTOR_HOME INTEGER,
    DD_NAME VARCHAR(30),
    AA_DELIQ_BALANCE7 DECIMAL(18,4),
    AA_DELIQ_BALANCE30 DECIMAL(18,4))
AS
declare variable aa_curr_region_id integer;
begin
  for select aa_curr_region_id, aa_curr_office_name, sum(aa_amount_active_end), sum(aa_just_cl_end),
             sum(aa_kolich_active_end), sum(aa_deliq_kolich30), sum(aa_deliq_balance7), sum(aa_deliq_balance30)
      from aarep_analitica_port_temp(1000000, :rep_date_end, :rep_date_end, '', 0, 1, 0, 1, 0, 0, 0)
      where ((:is_category_3 = 1 and
            aa_category_id = 3) or
            (:is_category_3 = 0 and
            aa_category_id in (1, 2)))
      group by aa_curr_region_id, aa_curr_office_name
      into aa_curr_region_id, aa_curr_office_name, aa_amount_active_end, aa_just_cl_end, aa_kolich_active_end,
           aa_deliq_kolich30, aa_deliq_balance7, aa_deliq_balance30
  do
  begin
    if (aa_amount_active_end > 0) then
    begin
      par7 = aa_deliq_balance7 / aa_amount_active_end * 100;
      par30 = aa_deliq_balance30 / aa_amount_active_end * 100;
    end
    else
    begin
      par7 = 0;
      par30 = 0;
    end
    zad_os = 0;
    sh_calc_perc = 0;
    sh_calc_fine = 0;
    in_office = 0;
    in_collector_ru = 0;
    in_collector_home = 0;
    for select dd_name, sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine), sum(in_office), sum(in_collector_ru),
               sum(in_collector_home)
        from aarep_delinq_status_30_v_2(:aa_curr_region_id, :rep_date_end, :is_category_3)
        group by dd_name
        into dd_name, zad_os, sh_calc_perc, sh_calc_fine, in_office, in_collector_ru, in_collector_home
    do
    begin
      if (zad_os is null) then
        zad_os = 0;
      if (sh_calc_perc is null) then
        sh_calc_perc = 0;
      if (sh_calc_fine is null) then
        sh_calc_fine = 0;
      if (in_office is null) then
        in_office = 0;
      if (in_collector_ru is null) then
        in_collector_ru = 0;
      if (in_collector_home is null) then
        in_collector_home = 0;
      suspend;
      aa_amount_active_end = null;
      aa_just_cl_end = null;
      aa_kolich_active_end = null;
      aa_deliq_kolich30 = null;
      par7 = null;
      par30 = null;
      aa_deliq_balance7 = null;
      aa_deliq_balance30 = null;
    end
  end

end
