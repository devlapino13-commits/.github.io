CREATE PROCEDURE FM_HTML_LOAD_KMP_REPORT_TEST(
    UN_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable lo_id decimal(18,0);
declare variable c_id d_id;
declare variable c_name_fio varchar(100);
declare variable reg_id d_id;
declare variable dowland_clients decimal(18,2);
declare variable dowland_clients_pr decimal(18,2);
declare variable dowland_clients_30 decimal(18,2);
declare variable dowland_clients_30_pr decimal(18,2);
declare variable i decimal(18,2);
declare variable i_30 decimal(18,2);
declare variable version_kmp d_smallint;
declare variable n d_integer;
begin
  n = 1;
  select ps_from_reg_id
  from hr_pstatus hps
  inner join username u on u.un_from_c_id = hps.ps_from_c_id
  inner join joblist jl on jl.jl_id = ps_from_jl_id
  inner join jobkind jk on jk.jk_id = jl.jl_kind
  where u.un_id = :un_id and
        :rep_date between ps_date_begin and ps_date_end and
        ps_is_working = 1 and
        ps_status = 1 and
        jk_system_code containing 'JobSystem.Region_KM'
  into :reg_id;

  res_text = (select res_style
              from fm_html_styles) || '<body>';
  res_text = BLOB_APPEND(res_text, f_format('<h1 class="green_text">%1</h1><br><p>на дату: %2</p>', 'Сводный отчет по загрузке КМП', f_datetostr(:rep_date)));
  res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="3%">
       <col width="50%">
       <col width="12%">
       <col width="12%">
       <col width="12%">
       <col width="12%">
       <tr><th>№</th>' || f_format('<th >%1</th><th>%2</th><th align="center">%3</th><th>%4</th><th>%5</th></tr>', 'ВИПС', 'Процент активных клиентов загрузивших КМП (до 30 дн. просрочка, включительно)', 'Процент активных клиентов загрузивших КМП', 'Количество активных клиентов загрузивших КМП (до 30 дн. просрочка, включительно)', 'Количество активных клиентов загрузивших КМП'));

  for select distinct hps.ps_from_c_id
      from hr_pstatus hps
      inner join joblist jl on jl.jl_id = ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where hps.ps_from_reg_id = :reg_id and
            :rep_date between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            ps_status = 1 and
            jk_system_code containing 'JobSystem.LO_Standart' and
            hps.ps_from_c_id = 1010100065937
      into :lo_id
  do
  begin
    dowland_clients = 0;
    dowland_clients_30 = 0;
    dowland_clients_pr = 0;
    dowland_clients_30_pr = 0;
    i = 0;
    i_30 = 0;
    select c_name_fio
    from clients cl
    where cl.c_id = :lo_id
    into :c_name_fio;

    for select distinct knc.kc_from_c_id
        from oksummary oks
        inner join knclient knc on knc.kc_from_k_id = oks.oks_from_k_id and
              knc.kc_amount_approved > 0
        where oks_kc_id = :lo_id and --id sotrudnika
              :rep_date between oks_date_begin and oks_date_end and
              oks_close_state = 0
        into :c_id
    do
    begin
      i = i + 1; --общее число клиентов
      version_kmp = 0;
      version_kmp = f_fsm_client_status(:c_id);

      if (version_kmp > 0) then
        dowland_clients = dowland_clients + 1; --активный клиент загрузивших КМП

      if (not exists(select oks_from_k_id
                     from oksummary oks
                     inner join knclient knc on knc.kc_from_k_id = oks.oks_from_k_id
                     where knc.kc_from_c_id = :c_id and
                           :rep_date between oks_date_begin and oks_date_end and
                           cast(:rep_date as date) - oks_delinq_date > 30 and
                           oks_close_state = 0)) then
      begin
        i_30 = i_30 + 1;
        if (version_kmp > 0) then
          dowland_clients_30 = dowland_clients_30 + 1; --активный клиент загрузивших КМП  до 30 дн. просрочка, включительно
      end
    end
    if (i > 0) then
      dowland_clients_pr = (dowland_clients / i) * 100;

    if (i_30 > 0) then
      dowland_clients_30_pr = (dowland_clients_30 / i_30) * 100;

    dowland_clients_30_pr = coalesce(dowland_clients_30_pr, 0);
    dowland_clients_pr = coalesce(dowland_clients_pr, 0);
    dowland_clients_30 = coalesce(dowland_clients_30, 0);
    dowland_clients = coalesce(dowland_clients, 0);

    res_text = BLOB_APPEND(res_text, '<tr><td align="center">' || n || '</td><td>' || c_name_fio || '</td><td>' || dowland_clients_30_pr || '</td><td>' || dowland_clients_pr || '</td><td>' || round(dowland_clients_30) || '</td><td>' || round(dowland_clients) || '</td></tr>');
    n = n + 1;
  end
  res_text = BLOB_APPEND(res_text, '</table></body>');
  suspend;
end
