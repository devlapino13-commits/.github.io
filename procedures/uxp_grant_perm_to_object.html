CREATE PROCEDURE UXP$GRANT_PERM_TO_OBJECT(
    OBJ_TYPE D_STR_CODE,
    OBJ_NAME D_STR_CODE)
AS
declare variable role_name d_str_code;
declare variable access_kind d_str_code;
declare variable res_text d_str_long;
begin
  for select r.rdb$role_name
  from rdb$roles r
  where coalesce(r.rdb$system_flag, 0) = 0
  into role_name
  do
  begin
    role_name = trim(role_name);
    access_kind = 'ALL';

    if (obj_type in ('table', 'view')) then
    begin
      if (role_name in ('SEL', 'APR') and not(obj_name starting 'TEMP_')) then
      begin
        if (not exists(select * from ux$readonly
                       where uro_object_name = :obj_name)) then
          access_kind = 'select';
      end
    end else
    if (obj_type in ('sequence', 'exception')) then
      access_kind = 'usage';
    else
      access_kind = 'execute';


    if (obj_type in ('table', 'view')) then
      res_text = 'grant ' || access_kind || ' on ' || obj_name || ' to ' || role_name || ';';
    else
      res_text = 'grant ' || access_kind || ' on ' || obj_type || ' ' || obj_name || ' to ' || role_name || ';';

    execute statement (res_text);
  end
end
