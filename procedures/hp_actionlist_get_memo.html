CREATE PROCEDURE HP_ACTIONLIST_GET_MEMO(
    ACL_ID D_ID,
    OPR_ID D_ID = null)
RETURNS (
    RES_MEMO D_BLOB_TEXT)
AS
declare variable acl_object_kind d_integer;
declare variable acl_object_id d_id;
declare variable acl_from_rd_id d_integer;
declare variable acl_respons_c_id d_id;
declare variable un_id d_id;
declare variable obm_memo d_str_8100;
declare variable obm_datetime d_datetime;
declare variable curr_c_id d_id;
declare variable obm_datetime_str d_str_code;
declare variable un_id_prior d_id;
declare variable is_kk d_str_code;
begin
  res_memo = '';

  select acl_object_kind, acl_object_id, acl_from_rd_id, acl_respons_c_id
  from actionlist
  where acl_id = :acl_id
  into acl_object_kind, acl_object_id, acl_from_rd_id, acl_respons_c_id;

  if (opr_id is null) then
  begin
    select ovs_from_opr_id
    from hp_objectprops_get(:acl_object_kind, :acl_object_id, -1, :acl_from_rd_id, null)
    into :opr_id;
  end

  -- переписки по объекту
  res_memo = '';
  un_id_prior = 0;
  if (opr_id is not null) then
  begin
    for select f_get_verify_memo(left(obm_memo, 2048), 'reason_memo'), obm_datetime, obm_from_un_id
        from objectmemo
        where obm_from_opr_id = :opr_id
        order by obm_from_opr_id desc, obm_datetime desc
        into :obm_memo, :obm_datetime, :un_id
    do
    begin
      if (un_id <> un_id_prior) then
      begin
        select un_from_c_id
        from username
        where un_id = :un_id
        into :curr_c_id;
        un_id_prior = un_id;
      end
      obm_datetime_str = f_datetimetostr(obm_datetime);

      if (coalesce(obm_memo, '') <> '') then
      begin
        if (res_memo <> '') then
          res_memo = BLOB_APPEND(res_memo, '<br>');
        is_kk = '';
        if (curr_c_id = acl_respons_c_id) then
          is_kk = ', комментарий от КК.';
        res_memo = BLOB_APPEND(res_memo, obm_datetime_str || is_kk || '<br>');
        res_memo = BLOB_APPEND(res_memo, '<b>' || obm_memo || '</b><br>');
      end
    end
  end

  suspend;
end
