CREATE PROCEDURE HP_EXTSYS_PROCESS_ONE(
    REP_ESM_ID INTEGER = 0,
    OPERDELLIST_ID D_INTEGER = 0)
RETURNS (
    ERR_TEXT D_STR_8100)
AS
declare variable esm_date date;
declare variable esm_account varchar(20);
declare variable esm_object_id decimal(18,0);
declare variable esm_k_id decimal(18,0);
declare variable esm_amount d_currency;
declare variable esm_amount_val d_currency;
declare variable esm_doc_no_str varchar(20);
declare variable esm_id integer;
declare variable c_name_fio varchar(100);
declare variable ol_id decimal(18,0);
declare variable loan_ol_id d_id;
declare variable head_office_id integer;
declare variable repay_dt_account_no d_str_20;
declare variable repay_dt_accountv_no d_str_20;
declare variable rep_rus_name varchar(100);
declare variable db_id integer;
declare variable esm_project d_str_20;
declare variable pr_code varchar(100);
declare variable kommis_ol_id d_id;
declare variable kommis_amount d_currency;
declare variable moneta_acc_id d_id;
declare variable kommis_acc_id d_id;
declare variable kommis_al_id d_str_code;
declare variable ol_memo d_str_long;
declare variable ipf_acc str_100;
declare variable classic_acc str_100;
declare variable esm_action d_str_20;
declare variable esm_status d_integer;
declare variable esm_full_source blob sub_type 1 segment size 80;
declare variable field_value d_str_100;
declare variable tm_date_end d_date;
declare variable acc_no d_str_20;
declare variable acck_no d_str_20;
declare variable accv_no d_str_20;
declare variable curr_rate d_currency;
declare variable acq_amount d_currency;
declare variable insurance_pay_amount d_currency;
declare variable insur_amount_due d_currency;
declare variable ol_id_insurance d_id;
declare variable ipf_insurance_only d_smallint;
declare variable is_consulting_only d_bool;
declare variable cons_kc_id d_id;
declare variable cons_balance d_currency;
declare variable cons_pay_amount d_currency;
declare variable ol_id_consulting d_id;
declare variable obj_id d_id;
begin
  -- обработка платежных операций
  -- поступающие через платежные терминалы и банковские карточки
  if (coalesce(rep_esm_id, 0) = 0) then
    exception e_default 'edm_id is empty';

  kommis_acc_id = null;
  kommis_al_id = '';
  moneta_acc_id = null;

  db_id = f_get_db_id();

  -- общие настройки, зависящие от страны
  head_office_id = f_hq_id(db_id);

  if (operdellist_id is null) then
    operdellist_id = 0;

  if (operdellist_id = 0) then
    operdellist_id = 1;

  err_text = '';
  ipf_acc = null;
  classic_acc = null;
  -- получаем параметры платежа через переданный rep_esm_id
  for select esm_id, esm_date, esm_trans_id, esm_account, esm_object_id, esm_amount, esm_from_k_id, esm_project,
             esm_action, esm_status, esm_object_id, esm_full_source
      from extsysmain
      where esm_id = :rep_esm_id
      into :esm_id, :esm_date, :esm_doc_no_str, :esm_account, :esm_object_id, esm_amount, esm_k_id, :esm_project,
           esm_action, esm_status, esm_object_id, :esm_full_source
  do
  begin
    if (esm_action <> 'pay' or esm_status <> 0 or esm_object_id is null) then
      exit;

    rep_rus_name = '';
    repay_dt_account_no = '';
    esm_amount_val = esm_amount;
    select esp_name, esp_account_no, esp_acc_classic, esp_acc_ipf
    from extsyspro
    where esp_project_kind = 'payment' and -- только платежные сстемы
          esp_project = :esm_project and
          esp_status = 1 -- только действующие
    into :rep_rus_name, -- местное название проекта, будет использовано в сообщения и чеках
         :repay_dt_account_no, -- бухгалтерский счет, куда будет посажено погашение
         classic_acc, -- счет по классику
         ipf_acc; -- счет по ИПФ

    -- если проект отсутствует или недействующий, завершаем работу
    if (rep_rus_name = '') then
      exit;

    -- id транзакции
    if (esm_doc_no_str is null) then
      esm_doc_no_str = '';

    if (db_id = 81001) then
    begin
      if (esm_project in ('moneta', 'qiwi', 'tinkoff', 'wirebank', 'CaRuSell')) then
      begin
        kommis_al_id = '010181091-02000003';

        select al_from_acc_id
        from acc_list
        where al_id = :repay_dt_account_no
        into moneta_acc_id;

        kommis_acc_id = case esm_project
                          when 'moneta' then f_get_acc_id('71702810005310310101')
                          when 'wirebank' then f_get_acc_id('71702810005310370101')
                          when 'tinkoff' then f_get_acc_id('71702810005310350101')
                          when 'qiwi' then f_get_acc_id('71702810005310340101')
                          when 'CaRuSell' then f_get_acc_id('71702810005310360101')
                        end;
      end
    end

    -- находим активный кредит с ближайшим плановым погашением
    if (esm_k_id is null and
        esm_project <> 'prepayment_qr') then
    begin
      select oks_from_k_id
      from oksummary, knclient, kontract
      where oks_from_k_id = kc_from_k_id and
            kc_from_c_id = :esm_object_id and
            kc_amount_approved > 0 and
            oks_date_end >= '31.12.2999' and
            oks_close_state = 0 and
            oks_from_k_id = k_id and
            k_disb_date <= :esm_date
      order by oks_delinq_date, oks_balance
      rows 1
      into :esm_k_id;

      if (esm_k_id is null) then
      begin
        select k_id
        from (select okf.oko_from_k_id as k_id, max(okf.oko_ol_date) as ol_date
              from operkroff okf
              inner join kontstate ks on ks.ks_from_k_id = okf.oko_from_k_id
              inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
              where okf.oko_ol_is_active = 1 and
                    ks.ks_state = 2 and
                    ks.ks_ol_is_delete = 0 and
                    current_date between ks.ks_date_begin and ks.ks_date_end and
                    kncl.kc_amount_approved > 0 and
                    kncl.kc_from_c_id = :esm_object_id -- <<
              group by okf.oko_from_k_id
              having sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc) > 0) a
        order by ol_date
        rows 1
        into :esm_k_id;
      end

      if (esm_k_id is null) then
      begin
        select okf.ox_object_id as k_id
        from operextra okf
        inner join kontstate ks on ks.ks_from_k_id = okf.ox_object_kind
        inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
        where okf.ox_ol_is_active = 1 and
              okf.ox_object_kind = f_objkind('kontract', 'id') and
              ks.ks_state = 2 and
              ks.ks_ol_is_delete = 0 and
              current_date between ks.ks_date_begin and ks.ks_date_end and
              kncl.kc_amount_approved > 0 and
              kncl.kc_from_c_id = :esm_object_id
        group by okf.ox_object_id
        having sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff) > 0
        order by max(okf.ox_ol_date)
        rows 1
        into :esm_k_id;
      end
      if (db_id in (39801, 41701) and
          esm_k_id is null) then
      begin
        select ks.ks_from_k_id
        from kontstate ks
        inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id and
              kc.kc_from_c_id = :esm_object_id and
              kc.kc_amount_approved > 0
        where ks.ks_state = 2 and
              ks.ks_ol_is_delete = 0 and
              current_date between ks.ks_date_begin and ks.ks_date_end
        order by ks.ks_date_begin desc
        rows 1
        into :esm_k_id;
      end
    end

    if (db_id in (39801, 41701) and
        esm_k_id is not null) then
    begin
      select pr.pr_code
      from kontract kont
      inner join product pr on pr.pr_id = kont.k_from_pr_id
      where kont.k_id = :esm_k_id
      into :pr_code;

      if (esm_project = 'doscredo') then
      begin
        ipf_acc = '10003/102';
        classic_acc = '10003/101';

        field_value = ''; -- реализация по задаче на jira 3097 от 12,02,2021 (разделение терминалов)

        select field_value
        from hp_extsys_parcer(:esm_full_source, 'terminal_number')
        into :field_value;

        field_value = coalesce(field_value, '');

        tm_date_end = null;
        select tm_date_end
        from terminals
        where tm_terminal_id = :field_value
        order by tm_date_begin desc
        rows 1
        into tm_date_end;

        if (field_value <> '' and
            exists(select *
                   from terminals
                   where tm_terminal_id = :field_value) and
            tm_date_end is null or (tm_date_end > current_date)) then --добавлено условие по задаче на jira 3274 (закрытие терминалов)
        begin
          ipf_acc = '10003/138';
          classic_acc = '10003/137';
        end
      end
      if (esm_project = 'umai') then
      begin
        ipf_acc = '10003/128';
        classic_acc = '10003/127';

        field_value = '';

        select field_value
        from hp_extsys_parcer(:esm_full_source, 'terminal_number')
        into :field_value;

        field_value = coalesce(field_value, '');

        tm_date_end = null;
        select tm_date_end
        from terminals
        where tm_terminal_id = :field_value
        order by tm_date_begin desc
        rows 1
        into tm_date_end;

        if (field_value <> '' and
            exists(select *
                   from terminals
                   where tm_terminal_id = :field_value) and
            tm_date_end is null or (tm_date_end > current_date)) then
        begin
          ipf_acc = '10003/138';
          classic_acc = '10003/137';
        end
      end
      if (db_id in (39801, 41701) and
          esm_project = 'Moneta') then
      begin
        repay_dt_account_no = (select al_id
                               from acc_list
                               where al_from_reg_id = 1090100 and
                                     al_account = iif(:pr_code containing ('.ipf.'), :ipf_acc, :classic_acc) and
                                     al_currency = '417');
        acc_no = (select al_id
                  from acc_list
                  where al_from_reg_id = 1090100 and
                        al_account = '11520/1' and
                        al_currency = '417');
        acck_no = (select al_id
                   from acc_list
                   where al_from_reg_id = 1090100 and
                         al_account = iif(:pr_code containing ('.ipf.'), '80509/116', '80509/16') and
                         al_currency = '417');

        repay_dt_accountv_no = (select al_id
                                from acc_list
                                where al_from_reg_id = 1090100 and
                                      al_account = iif(:pr_code containing ('.ipf.'), :ipf_acc, :classic_acc) and
                                      al_currency = '810');
        accv_no = (select al_id
                   from acc_list
                   where al_from_reg_id = 1090100 and
                         al_account = '11520/1' and
                         al_currency = '810');

        curr_rate = (select curr_rate
                     from get_curr_rate('810', :esm_date));
        esm_amount = (select field_value
                      from hp_extsys_parcer(:esm_full_source, 'kgs_amount'));
        kommis_amount = abs((select field_value
                             from hp_extsys_parcer(:esm_full_source, 'mnt_fee')));
        --kommis_amount = esm_amount_val - round(esm_amount/curr_rate, 2);
      end
      else
      begin
        select al_id
        from find_acc_list(iif(:pr_code containing ('.ipf.'), :ipf_acc, :classic_acc))
        where al_from_reg_id = 1090100
        into :repay_dt_account_no;
      end
    end

    if (db_id = 81001 and
        esm_k_id is not null) then
    begin
      select pr.pr_code
      from kontract kont
      inner join product pr on pr.pr_id = kont.k_from_pr_id
      where kont.k_id = :esm_k_id
      into :pr_code;

      if (:pr_code containing '.oivo_pack.') then
      begin
        repay_dt_account_no = (select al_id
                               from acc_list
                               where al_from_reg_id = 1010100 and
                                     al_account = iif(:esm_project = 'Wirebank', '51-01/26', '51-01/24') and
                                     al_currency = '810');

        select al_from_acc_id
        from acc_list
        where al_id = :repay_dt_account_no
        into moneta_acc_id;
      end
    end

    if (db_id in (39801, 41701) and
        esm_project = 'prepayment_qr') then
    begin
      if (coalesce(esm_k_id, 0) = 0) then
        esm_k_id = (select field_value
                    from hp_extsys_parcer(:esm_full_source, 'k_id'));

      select ol_id
      from fm_operkontract_ins(:esm_k_id, :esm_amount)
      into :ol_id;

      insert into extsysoper (eso_from_esm_id, eso_from_ol_id)
      values (:esm_id, :ol_id);

      update extsysmain
      set esm_from_k_id = :esm_k_id
      where esm_id = :esm_id and
            (coalesce(esm_from_k_id, 0) <> :esm_k_id);

    end

    is_consulting_only = 0;
    ol_id_consulting = 0;
    if (db_id = 41701 and
        err_text = '' and
        exists(select field_value
               from hp_extsys_parcer(:esm_full_source, '&obj_kind')
               where field_value = 'knclient')) then
    begin
      -- клиент погашает консультационные услуги, RFCKG-4903
      obj_id = (select field_value
                from hp_extsys_parcer(:esm_full_source, '&obj_id'));

      select kc_id,
             amount_balance
      from hp_consult_installment_amount(:esm_k_id)
      into cons_kc_id,
           cons_balance;

      if (obj_id <> cons_kc_id) then
        err_text = 'Ошибка: не найден клиент в кредите';
      else if (cons_balance <= 0) then
        err_text = 'Ошибка: не найден договор консультационных услуг';
      else
      begin
        cons_pay_amount = iif(esm_amount > cons_balance, cons_balance, esm_amount);

        -- обработка погашения конс. услуг
        select res_text,
               parent_ol_id
        from hp_consult_installment_repay(:esm_id, :esm_k_id, :cons_kc_id, :cons_pay_amount)
        into err_text,
             ol_id_consulting;

        -- остаток идет на кредит
        esm_amount = esm_amount - cons_pay_amount;

        -- если вся сумма ушла на конс.услуги, дальше не обрабатываем
        if (esm_amount = 0) then
          is_consulting_only = 1;
      end
    end

    ipf_insurance_only = 0;
    ol_id_insurance = 0;
    if (db_id = 41701 and -- RFCKG-5391
        err_text = '' and
        exists(select 1
               from hp_ipf_check_insurance_pay(:esm_k_id)
               where res_text <> '') and
        exists(select 1
               from kontstate ks
               where ks.ks_from_k_id = :esm_k_id and
                     ks.ks_ol_is_delete = 0 and
                     ks.ks_state = 1 and
                     current_date between ks.ks_date_begin and ks.ks_date_end)) then
    begin
      -- сумма задолженности по страхованию
      select res_amount
      from hp_opercash_loan_other_amount(:esm_k_id, 'CashTransactionIncom.Insurance')
      into :insur_amount_due;

      insurance_pay_amount = iif(esm_amount > insur_amount_due, insur_amount_due, esm_amount);

      -- сначала списываем оплату на страхование
      select res_text, parent_ol_id
      from hp_ipf_insurance_operbook_our_true(:esm_id, :esm_k_id, :insurance_pay_amount)
      into :err_text, ol_id_insurance;

      -- остаток идет на кредит
      esm_amount = esm_amount - insurance_pay_amount;

      -- если вся сумма ушла на страхование, дальше не обрабатываем
      if (esm_amount = 0) then
        ipf_insurance_only = 1;
    end

    if (esm_k_id is not null and
        esm_project <> 'prepayment_qr' and
        (err_text = '' and
        ipf_insurance_only = 0 and
        is_consulting_only = 0)
    ) then
    begin
      -- сохраняем операцию
      ol_id = null;
      select ol_id, loan_ol_id
      from hp_do_repayment(:esm_k_id, 0, :esm_date, :esm_amount, 1, -- :repay_to_os,
      1, -- :repay_to_perc,
      1, -- :repay_to_fine,
      :operdellist_id, -- :repay_to_kommis или Id заявки на удаление и перепринятие
      1, -- :need_save,
      0, -- :is_full_repay,
      0, -- :is_writeoff,
      :head_office_id, -- :repay_fact_office_id, -- головной офис в рф
      iif(:kommis_al_id = '', 3, 2), -- :repay_dt_account_kind, -- внутренний бух.счет европлата
      :repay_dt_account_no, -- :repay_dt_account_no,
      1, -- :repay_curr_rate,
      1000000000000, -- :repay_user_id,
      :esm_object_id, -- :repay_document_client,
      iif(:db_id <> 94401, 'Погашение через ' || :esm_project || '.', :esm_project || ' vasitəsi ilə ödəniş.'), -- :repay_doc_client_name_reason,
      null, -- :repay_manager_c_id,
      0, -- :repay_from_rd_id,
      0) -- контрольная сумма)
      into :ol_id, :loan_ol_id;

      if (ol_id > 0) then
      begin
        if (db_id = 41701 and
            exists(select oko_from_k_id
                   from operkroff
                   where oko_from_k_id = :esm_k_id and
                         oko_ol_is_active = 1
                   group by oko_from_k_id
                   having sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc) <> 0)) then
          execute procedure hp_opercash_do_operkroff(:loan_ol_id, :esm_amount, true);

        if (db_id = 39801 and
            esm_project = 'Processing_KZ') then
        begin
          acq_amount = round(0.0139 * esm_amount, 2);
          if (acq_amount > 0) then
          begin
            insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_detail,
                                  ob_summa, ob_summa_val)
            values (:ol_id, '398', 1, (select al_id
                                     from acc_list
                                     where al_from_reg_id = :head_office_id and
                                           al_currency = '398' and
                                           al_account = '7470.21/1'), (select al_id
                                                                       from acc_list
                                                                       where al_from_reg_id = :head_office_id and
                                                                             al_currency = '398' and
                                                                             al_account = '1021/1'),
                  'Комиссия эквайринга', :acq_amount, :acq_amount);
          end
        end

        if (db_id in (39801, 41701) and
            esm_project = 'Moneta') then
        begin
          -- Для Монеты в KG
          update operbook
          set ob_account_dt = :acc_no
          where ob_from_ol_id = :ol_id;

          insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                                ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt,
                                ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date, ob_ol_reg_id)
          select first 1 ob_from_ol_id, '810', :curr_rate, ob_client_dt, :repay_dt_accountv_no,
                         round(:esm_amount / :curr_rate, 2), :esm_amount, ob_from_pc_id, ob_from_kp_id, ob_oper_mask,
                         :accv_no, ob_client_kt, ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date,
                         ob_ol_reg_id
          from operbook
          where ob_from_ol_id = :ol_id;

          if (kommis_amount <> 0) then
          begin
            insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                                  ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt,
                                  ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date, ob_ol_reg_id)
            select first 1 ob_from_ol_id, '810', :curr_rate, ob_client_dt, :accv_no, :kommis_amount,
                           round(:kommis_amount * :curr_rate, 2), ob_from_pc_id, ob_from_kp_id, ob_oper_mask,
                           :repay_dt_accountv_no, ob_client_kt, 'Комиссия пл.системы', ob_subkonto_dt,
                           ob_subkonto_kt, ob_from_sac_id, ob_ol_date, ob_ol_reg_id
            from operbook
            where ob_from_ol_id = :ol_id;

            insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                                  ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt,
                                  ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date, ob_ol_reg_id)
            select first 1 ob_from_ol_id, '417', 1, ob_client_dt, :acck_no, round(:kommis_amount * :curr_rate, 2),
                           round(:kommis_amount * :curr_rate, 2), ob_from_pc_id, ob_from_kp_id, ob_oper_mask, :acc_no,
                           ob_client_kt, 'Комиссия пл.системы', ob_subkonto_dt, ob_subkonto_kt,
                           ob_from_sac_id, ob_ol_date, ob_ol_reg_id
            from operbook
            where ob_from_ol_id = :ol_id;
          end
        end

        -- после успешной обработки заносим в таблицу
        insert into extsysoper (eso_from_esm_id, eso_from_ol_id)
        values (:esm_id, :ol_id);

        -- номер транзакции закидываем в operlist
        update operlist
        set ol_doc_no_str = :esm_doc_no_str
        where (ol_id = :ol_id or
              ol_parent_id = :ol_id);

        update extsysmain
        set esm_from_k_id = :esm_k_id
        where esm_id = :esm_id and
              (coalesce(esm_from_k_id, 0) <> :esm_k_id);

        -- если перешли на прием напрямую на расчетный счет,
        -- то сразу строим проводку по комиссии
        if (kommis_al_id <> '') then
        begin
          if (esm_project = 'qiwi') then
          begin
            kommis_amount = round(esm_amount * 0.010 / 100, 2);
          end
          else
          begin
            kommis_amount = 0;
            if (esm_project in ('moneta', 'tinkoff', 'wirebank', 'CaRuSell')) then
            begin
              select iif(field_value = '', 0, field_value)
              from hp_extsys_parcer(:esm_full_source, 'MNT_FEE')
              into kommis_amount;
              kommis_amount = -kommis_amount;
            end
          end

          if (kommis_amount <> 0) then
          begin
            ol_memo = '';
            select iif(:db_id <> 94401, 'договор займа ' || k_kontract_no || ' от ' || f_datetostr(k_kontract_date), k_kontract_no || ' nomreli ' || f_datetostr(k_kontract_date) || ' tarixli muqavile')
            from kontract
            where k_id = :esm_k_id
            into ol_memo;
            if (ol_memo is null) then
              ol_memo = '';
            ol_memo = iif(:db_id <> 94401, 'Комиссия за погашение через ' || :esm_project || '. ' || ol_memo, :esm_project || ' vasitəsi ilə ödəniş haqqı. ' || ol_memo);

            insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no_str, ol_doc_kind, ol_memo,
                                  ol_user_id, ol_document_client, ol_oper_subtype, ol_parent_id)
            values (:head_office_id, 2, :esm_date, :esm_doc_no_str, 9, :ol_memo, 1000000000001, :esm_object_id, 36,
                    :ol_id)
            returning ol_id
            into kommis_ol_id;

            update or insert into operloan (opl_from_ol_id, opl_acc_kind, opl_from_al_id, opl_from_acc_id)
            values (:kommis_ol_id, 2, :kommis_al_id, :kommis_acc_id)
            matching (opl_from_ol_id);

            insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                                  ob_summa, ob_account_kt, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
            values (:kommis_ol_id, '810', 1, :esm_object_id, :kommis_al_id, :kommis_amount, :kommis_amount,
                    :repay_dt_account_no, 'Комиссия за погашение через ' || :esm_project,
                    :esm_date, 1, :head_office_id);

            execute procedure hp_do_kn_operbook_ins_simple(null, :kommis_ol_id, null, :kommis_acc_id, :moneta_acc_id,
                :kommis_amount, 0, :esm_date, :head_office_id, 1,
                'Комиссия за погашение через ' || :esm_project, :esm_object_id, null, '810', 1);
          end
        end
      end
      else
      begin
        if (:ol_id_insurance > 0) then
          update operlist ol -- аннулируем операцию по оплате страховой премии, если она успешно сложилась, а оплата по кредиту прошла неуспешно
          set ol.is_delete = 1
          where ol.ol_id = :ol_id_insurance or ol.ol_parent_id = :ol_id_insurance;

        if (:ol_id_consulting > 0) then
          update operlist ol -- аннулируем операцию по оплате конс. услуг, если она прошла успешно, а оплата по кредиту прошла неуспешно
          set ol.is_delete = 1
          where ol.ol_id = :ol_id_consulting or ol.ol_parent_id = :ol_id_consulting;
      end
    end

    if ((:ipf_insurance_only = 1 and :ol_id_insurance > 0) or -- если оплатили успешно только за страхование
        (:ol_id_insurance > 0 and ipf_insurance_only = 0 and ol_id > 0)) then -- или оплатили успешно за страхование и кредит
    begin
      -- после успешной обработки заносим в таблицу операцию по страхованию
      insert into extsysoper (eso_from_esm_id, eso_from_ol_id)
      values (:esm_id, :ol_id_insurance);

      -- номер транзакции закидываем в operlist для страховки и её дочерних записей
      update operlist
      set ol_doc_no_str = :esm_doc_no_str
      where (ol_id = :ol_id_insurance or ol_parent_id = :ol_id_insurance);
    end

    if ((is_consulting_only = 1 and ol_id_consulting > 0) or -- если оплатили успешно только за конс. услуги
        (is_consulting_only = 0 and ol_id_consulting > 0 and ol_id > 0)) then -- или оплатили успешно за конс. услуги и кредит
    begin
      insert into extsysoper (eso_from_esm_id, eso_from_ol_id)
      values (:esm_id, :ol_id_consulting);

      update operlist
      set ol_doc_no_str = :esm_doc_no_str -- номер транзакции (esm_trans_id)
      where (ol_id = :ol_id_consulting or ol_parent_id = :ol_id_consulting);
    end

    if (coalesce(esm_k_id, 0) = 0 or
        (coalesce(ol_id, 0) = 0 and ipf_insurance_only = 0 and is_consulting_only = 0) or
        (ipf_insurance_only = 1 and coalesce(ol_id_insurance, 0) = 0) or
        (is_consulting_only = 1 and coalesce(ol_id_consulting, 0) = 0)
    ) then
    begin
      -- если у клиента нет активных кредитов,
      -- возвращаем ошибку и уведомляем бухгалтерам и программистам
      c_name_fio = '';
      select c_name_fio
      from clients
      where c_id = :esm_object_id
      into :c_name_fio;

      err_text = err_text || '<br>';
      if (coalesce(esm_k_id, 0) = 0) then
        err_text = err_text || f_format('<b>%1 %2 %3.</b><br>', 'платеж', :rep_rus_name, 'не обработан так как не найден активный кредит');
      else
        err_text = err_text || f_format('<b>%1 %2 %3. ol_id<=0. </b><br>', 'платеж', :rep_rus_name, 'не обработан');
      err_text = err_text || 'фио: ' || c_name_fio || '<br>';
      err_text = err_text || 'дата рождения и лиц.счет: ' || esm_account || '<br>';

      err_text = err_text || '№ квитанции ' || :rep_rus_name || ': ' || esm_doc_no_str || '<br>';
      err_text = err_text || 'сумма погашения за вычетом комисии: ' || esm_amount || '<br>';

      update extsysmain
      set esm_status = -3,
          esm_from_k_id = null
      where esm_id = :esm_id;
    end
  end -- конец обработки транзакций по проекту

  suspend;
end
