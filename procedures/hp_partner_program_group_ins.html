CREATE PROCEDURE HP_PARTNER_PROGRAM_GROUP_INS(
    C_ID D_ID,
    PPG_BONUS D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable ppg_id d_id;
declare variable grp d_integer;
begin
  res_is_error = 1;
  res_text = '';

  if (not exists(select *
                 from username
                 where un_id = :un_id)) then
    res_text = 'Пользователь не найден';

  if (not exists(select *
                 from clients
                 where c_id = :c_id)) then
    res_text = 'Клиент не найден';

  select ppg_bonus
  from partner_program_group
  where ppg_from_c_id = :c_id
  into grp;

  if (coalesce(grp, 0) > 0) then
    res_text = 'Клиент уже включен в группу N' || grp;

  if (coalesce(ppg_bonus, 0) = 0) then
    res_text = 'Бонусная группа не указана';

  if (res_text = '') then
    insert into partner_program_group (ppg_from_c_id, ppg_bonus, ppg_create_user)
    values (:c_id, :ppg_bonus, :un_id)
    returning ppg_id
    into ppg_id;

  if (coalesce(ppg_id, 0) > 0) then
    res_is_error = 0;

  suspend;
end
