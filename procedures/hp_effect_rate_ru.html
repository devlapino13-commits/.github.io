CREATE PROCEDURE HP_EFFECT_RATE_RU(
    K_ID D_ID)
RETURNS (
    CALC_EFF DECIMAL(18,3))
AS
declare variable max_iter integer = 100;
declare variable calc_error double precision = 0.000001;
declare variable iter integer;
declare variable sh_date date;
declare variable first_date date;
declare variable prc double precision;
declare variable l_prc double precision;
declare variable h_prc double precision;
declare variable q integer;
declare variable e double precision;
declare variable a_sum double precision;
declare variable bp integer;
declare variable sh_amount decimal(18,4) = 0;
declare variable rep_count integer;
begin
  -- Расчет полной стоимости кредита (Россия)

  calc_eff = 0.0;
  if (not(exists(select *
                 from temp_schedule))) then
  begin
    suspend;
    exit;
  end

  -- Расчет базового периода для массива платежей
  -- Ищем минимальный интервал, встречающийся с наибольшей частотой
  select sh_days, count(*)
  from temp_schedule
  where sh_id > 0
  group by sh_days
  order by count(*) desc, sh_days asc
  rows 1
  into bp, rep_count;

  if (rep_count = 1) then
  begin
    -- Если нет повторяющихся периодов, берем среднее арифметическое
    select ceil(sum(sh_days * 1.0) / count(*))
    from temp_schedule
    where sh_id > 0
    into bp;
  end

  -- Расчет ПСК
  l_prc = 0; -- 0%
  h_prc = cast(10.0 as double precision) / 365.0 * bp; -- предполагаем что не больше 1000% в год
  iter = 0;
  select sh_date
  from temp_schedule
  where sh_id = 0
  into first_date;
  while (iter <= max_iter) do
  begin
    a_sum = 0;
    prc = (l_prc + h_prc) / 2;

    for select sh_date, sh_os
        from temp_schedule
        order by sh_date
        into sh_date, sh_amount
    do
    begin
      q = (sh_date - first_date) / bp;
      e = mod(sh_date - first_date, bp) * 1.000000 / bp;
      a_sum = a_sum + sh_amount / ((1 + e * prc) * power(1 + prc, q));
    end

    if (abs(a_sum) < calc_error) then
      break;
    if (a_sum > 0) then
      l_prc = prc;
    else
      h_prc = prc;
    iter = iter + 1;
  end
  if (iter >= max_iter) then
    exception e_default 'Cannot calc effective rate for contract: ' || k_id;
  calc_eff = round(prc * 365 / bp, 5) * 100;
  suspend;
end
