CREATE PROCEDURE PRINT_FP_ANK(
    C_ID ID_TYPE)
RETURNS (
    FPA_ID ID_TYPE,
    FPA_CLIENT_FIO STR_255,
    FPA_FROM_UN_ID ID_TYPE,
    FPA_CITY STR_255,
    FPA_CITIZEN STR_255,
    FPA_DP_BIRTH STR_255,
    FPA_PASS_KIND STR_255,
    FPA_PASS_PROPS STR_255,
    FPA_MCARD_PROPS STR_255,
    FPA_ACARD_PROPS STR_255,
    FPA_INN STR_255,
    FPA_PBUL_PROPS STR_255,
    FPA_ADDRESS_LEGAL STR_255,
    FPA_ADDRESS_FACT STR_255,
    FPA_REL_START DATE,
    FPA_FT_MEMO STR_255,
    FPA_RESP_FIO STR_255,
    FPA_CELL_NO STR_20,
    FPA_EMAIL STR_100,
    EXIST_IN_FP_LIST STR_255,
    SOLUTION_IN_FP STR_255,
    RISK_LEVEL STR_255,
    FPA_SNILS STR_100,
    REP_EXTRA_1 STR_100,
    REP_EXTRA_2 STR_100,
    REP_EXTRA_3 STR_100,
    REP_EXTRA_4 STR_100,
    REP_EXTRA_5 STR_100,
    FPA_DATE_STOP_REL_STR STR_100,
    CHECKED_DAYS BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    CHECK_DESCRIPTION VARCHAR(2000))
AS
declare variable fpa_date_check date; /* FPA_REL_START */
declare variable ma_id d_integer;
declare variable fpa_date_check_str str_100;
declare variable fpa_citizen_id d_integer;
declare variable risk smallint;
declare variable last_ks_state d_integer;
declare variable ks_date_begin date;
declare variable i d_integer;
declare variable fd_date date;
declare variable fd_date_str str_100;
declare variable fpd_date_begin date;
declare variable fpd_date_end date;
declare variable checked_days_in_fp_list blob sub_type 1 segment size 80;
declare variable j d_integer;
declare variable fpa_city_id id_type;
declare variable fpa_city_1 id_type;
declare variable fpa_city_2 id_type;
declare variable fpa_ank_resp id_type;
declare variable fpa_ank_resp_1 id_type;
declare variable fpa_ank_resp_2 id_type;
declare variable fpa_rel_start_1 date;
declare variable fpa_rel_start_2 date;
declare variable ma_id_1 d_integer;
declare variable ma_id_2 d_integer;
declare variable last_ks_state_1 d_integer;
declare variable last_ks_state_2 d_integer;
declare variable ks_date_begin_1 d_date;
declare variable ks_date_begin_2 d_date;
declare variable v_onn d_str_20;
begin
  for select fpa_id, fpa.fpa_client_fio, ddr.rd_name as fpa_citizen, fpa_from_un_id, fpa_date_place_birth,

             (
             case
               when pk.pk_country = 'rus' and lower(pk.pk_name) = 'паспорт' then 'Паспорт гражданина РФ'
               when pk.pk_country = 'rus' and lower(pk.pk_name) = 'внутрен. паспорт' then 'Паспорт гражданина РФ'
               when pk.pk_country = 'kgz' and lower(pk.pk_name) = 'паспорт' then 'Паспорт гражданина Республики Кыргызстан'
               when lower(pk.pk_name) = 'загран. паспорт' then 'Заграничный паспорт'
               when lower(pk.pk_name) = 'паспорт пластик' then 'ID-карта'
               when pk.pk_country = 'tjk' and lower(pk.pk_name) = 'паспорт' then 'Паспорт гражданина Республики Таджикистан'
               when pk.pk_country = 'uzb' and lower(pk.pk_name) = 'паспорт' then 'Паспорт гражданина Республики Узбекистан'
               when pk.pk_country = 'kaz' and lower(pk.pk_name) = 'паспорт' then 'Паспорт гражданина Республики Казахстан'
               else pk.pk_name
             end),

             coalesce(fpa_pass_props, ''), coalesce(fpa_mcard_props, ''), coalesce(fpa_acard_props, ''),
             coalesce(fpa_inn, ''), coalesce(fpa_pbul_props, ''), coalesce(fpa_address_legal, ''),
             coalesce(fpa_address_fact, ''), fpa_rel_start, coalesce(fpa_ft_memo, ''), coalesce(fpa.fpa_cell_no, ''),
             coalesce(fpa.fpa_email, ''), fpa_date_check, fpa.fpa_city, fpa.fpa_citizen, coalesce(fpa.fpa_snils, '')
      from fp_anketa_cl fpa
      inner join passkind pk on pk.pk_id = fpa_pass_kind_id
      inner join refdata ddr on fpa.fpa_citizen = ddr.rd_id
      where fpa.fpa_c_id = :c_id and
            fpa.fpa_id = 1 -- << это важно!
            rows 1 -- << это важно!
      into :fpa_id, :fpa_client_fio, :fpa_citizen, :fpa_from_un_id, :fpa_dp_birth, :fpa_pass_kind, :fpa_pass_props,
           :fpa_mcard_props, :fpa_acard_props, :fpa_inn, :fpa_pbul_props, :fpa_address_legal, :fpa_address_fact,
           :fpa_rel_start, :fpa_ft_memo, :fpa_cell_no, :fpa_email, :fpa_date_check, :fpa_city_id, :fpa_citizen_id,
           :fpa_snils
  do
  begin
    if (fpa_inn = '') then
      fpa_inn = '---';
    if (fpa_pass_props = '') then
      fpa_pass_props = '---';
    if (fpa_mcard_props = '') then
      fpa_mcard_props = '---';
    if (fpa_acard_props = '') then
      fpa_acard_props = '---';
    if (fpa_inn = '') then
      fpa_inn = '---';
    if (fpa_pbul_props = '') then
      fpa_pbul_props = '---';
    if (fpa_address_legal = '') then
      fpa_address_legal = '---';
    if (fpa_address_fact = '') then
      fpa_address_fact = '---';
    if (fpa_cell_no = '') then
      fpa_cell_no = '---';
    if (fpa_email = '') then
      fpa_email = '---';
    if (fpa_snils = '') then
      fpa_snils = '---';

    fpa_pass_kind = trim(fpa_pass_kind);
    fpa_date_check_str = '';

    -- город в шапке и ответственное лицо беру из адреса офиса первой выдачи займа клиента где лицо является клиентом или залогодателем.
    -- проверяю среди клиентов
    select reg.reg_from_pl_id, un_from_c_id, ks.ks_date_begin, kont.k_from_ma_app_id
    from knclient kncl
    inner join kontract kont on kont.k_id = kncl.kc_from_k_id
    inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id
    inner join region reg on reg.reg_unicode = ks.ks_region_id
    inner join operlist ol on ol.ol_id = ks.ks_from_ol_id
    inner join username un on un.un_id = ol.ol_user_id
    where ks.ks_ol_is_delete = 0 and
          ks.ks_state = 1 and
          kncl.kc_amount_approved > 0 and
          kont.k_status <> -1 and
          kncl.kc_from_c_id = :c_id
    order by ks.ks_date_begin
    rows 1
    into :fpa_city_1, :fpa_ank_resp_1, :fpa_rel_start_1, :ma_id_1;

    -- проверяю среди залогодателей
    select reg.reg_from_pl_id, cl.c_id, ks.ks_date_begin, kont.k_from_ma_app_id
    from zalog zal
    inner join kontract kont on kont.k_id = zal.z_from_k_id
    inner join kontstate ks on ks.ks_from_k_id = k_id
    inner join region reg on reg.reg_unicode = ks.ks_region_id
    inner join operlist ol on ol.ol_id = ks.ks_from_ol_id
    inner join username un on un.un_id = ol.ol_user_id
    inner join clients cl on cl.c_id = un.un_from_c_id
    where ks.ks_ol_is_delete = 0 and
        ks.ks_state = 1 and
        zal.z_from_c_id is not null  and
        kont.k_status <> -1 and
        zal.z_from_c_id = :c_id and
        zal.z_is_ready = 1
    order by ks.ks_date_begin
    rows 1
    into :fpa_city_2, :fpa_ank_resp_2, :fpa_rel_start_2, :ma_id_2;

     -- определяю кем лицо стал реньше клиенто или залогодателем, исходя из этого заполняю шапку
    if (coalesce(fpa_rel_start_1, '01.01.1900') > coalesce(fpa_rel_start_2, '01.01.1900')) then
    begin
        fpa_city_id = fpa_city_1 ;
        fpa_ank_resp = fpa_ank_resp_1;
        fpa_rel_start = fpa_rel_start_1;
        ma_id = ma_id_1;
    end
    else
    begin
        fpa_city_id = fpa_city_2;
        fpa_ank_resp = fpa_ank_resp_2;
        fpa_rel_start = fpa_rel_start_2;
        ma_id = ma_id_2;
    end

    select maa.ma_date_app
    from miniapp maa
    where maa.ma_id = :ma_id and
          maa.ma_date_end = '31.12.9999' rows 1
    into :fpa_date_check;

    fpa_date_check_str = replace(iif(extract(day from fpa_date_check) < 10, '0', '') || extract(day from fpa_date_check) || '.' || iif(extract(month from fpa_date_check) < 10, '0', '') || extract(month from fpa_date_check) || '.' || extract(year from fpa_date_check), ' ', '');

    exist_in_fp_list = 'Отсутствует в перечне на дату ' || fpa_date_check_str;
    solution_in_fp = 'Решение отсутствует на дату ' || fpa_date_check_str;

    checked_days = 'Дата проверки : ';

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :fpa_ank_resp
    into :fpa_resp_fio;

    select pl.pl_name
    from places pl
    where pl.pl_id = :fpa_city_id
    into :fpa_city;

    risk = 0;
    if (fpa_citizen_id = 12) then
      risk = 1;

    if (exists(select *
               from finpolis_639_cli_link fcl
               where fcl.fcl_from_c_id = :c_id)) then
      risk = 2;

    rep_extra_1 = '---';
    rep_extra_2 = '---';
    rep_extra_3 = '---';
    rep_extra_4 = '---';
    rep_extra_5 = '---';

    if (risk = 0) then
      risk_level = 'Степень (уровень) риска клиента: «стандартный» в связи с отсутствием факторов «критичного» и «высокого» уровней риска';
    else
    if (risk = 1) then
    begin
      risk_level = 'Степень (уровень) риска клиента: «высокий» (гражданин республики таджикистан)';

      rep_extra_1 = '- Получение микрозайма;';
      rep_extra_2 = '- Долгосрочное сотрудничество для периодического получения заемных средств;';
      rep_extra_3 = '- Отсутствуют;';
      rep_extra_4 = '- Устойчивое;';
      rep_extra_5 = '- Положительная;';
    end
    else
    if (risk = 2) then
    begin
      risk_level = 'Степень уровня риска клиента: «высокая», в связи с получением информации (сведений), в соответствии с 639-П';
      risk = 1;
    end

    ------ >
    select ks.ks_state, ks.ks_date_begin
    from kontstate ks
    inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
    where ks.ks_state in (1, 2) and
          ks.ks_ol_is_delete = 0 and
          kncl.kc_from_c_id = :c_id and
          kncl.kc_amount_approved > 0
    order by ks_date_begin desc
    rows 1
    into :last_ks_state_1, :ks_date_begin_1;

    select ks.ks_state, ks.ks_date_begin
    from kontstate ks
    inner join zalog zal on zal.z_from_k_id = ks.ks_from_k_id
    where ks.ks_state in (1, 2) and
          ks.ks_ol_is_delete = 0 and
          zal.z_from_c_id = :c_id
    order by ks_date_begin desc
    rows 1
    into :last_ks_state_2, :ks_date_begin_2;

    if (coalesce(ks_date_begin_1, '01.01.1900') > coalesce(ks_date_begin_2, '01.01.1900')) then
    begin
        last_ks_state = last_ks_state_1;
        ks_date_begin = ks_date_begin_1;
    end
    else
    begin
        last_ks_state = last_ks_state_2;
        ks_date_begin = ks_date_begin_2;
    end
    ------ <

    if (last_ks_state = 2) then
      fpa_date_stop_rel_str = f_datetostr(ks_date_begin);

    check_description = null;

    fpd_date_begin = null;
    fpd_date_end = null;

    select fpd.fpd_description,
           fpd.fpd_date_begin,
           fpd.fpd_date_end
    from fp_anketa_descript fpd
    where fpd.fpd_anketa_no = 0 and
          fpd.fpd_from_c_id = :c_id
    into :check_description, :fpd_date_begin, :fpd_date_end;

    fpd_date_begin = coalesce(fpd_date_begin, '31.12.9999');
    fpd_date_end = coalesce(fpd_date_end, '01.01.1999');

    check_description = coalesce(check_description, '');

    if (f_get_db_id() in (39801, 41701)) then
    v_onn = 'ФРОМУ';
    if (f_get_db_id() = 94401) then
    v_onn = 'ОНН';
    if (f_get_db_id() = 81001) then
    v_onn = 'ОНН';

    if (check_description = '') then
      check_description = 'КЛИЕНТ ОТСУТСТВУЕТ В ПЕРЕЧНЕ РОСФИНМОНИТОРИНГА ТЕРРОРИСТОВ/ЭКСТРЕМИСТОВ И В ПЕРЕЧНЕ ЛИЦ, В ОТНОШЕНИИ КОТОРЫХ ПРИНЯТЫ РЕШЕНИЯ О ПРИМЕНЕНИИ МЕР ПО ЗАМОРАЖИВАНИЮ/БЛОКИРОВАНИЮ ДЕНЕЖНЫХ СРЕДСТВ И ИНОГО ИМУ�©ЕСТВА И ПЕРЕЧНЕ ' || v_onn;

    i = 0;
    j = 0;
    checked_days_in_fp_list = '';

    for select fpd.fd_date
        from finpolis_dates fpd
        where fpd.fd_date >= :fpa_rel_start
        order by fpd.fd_date desc
        into :fd_date
    do
    begin
      fd_date_str = f_datetostr(fd_date);
      if ((fd_date < :fpd_date_begin) or (fd_date > :fpd_date_end)) then
      begin
        if (i = 0) then
          checked_days = BLOB_APPEND(checked_days, '' || fd_date_str);
        else
          checked_days = BLOB_APPEND(checked_days, ', ' || fd_date_str);
        i = i + 1;
      end
      else
      begin
        if (j = 0) then
          checked_days_in_fp_list = BLOB_APPEND(checked_days_in_fp_list, '' || fd_date_str);
        else
          checked_days_in_fp_list = BLOB_APPEND(checked_days_in_fp_list, ', ' || fd_date_str);
        j = j + 1;
      end
    end

    check_description = replace(check_description, '[CHECKED_DAYS_IN_FP_LIST]', checked_days_in_fp_list);

--checked_days_in_fp_list
    suspend;
  end
end
