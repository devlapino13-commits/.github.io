CREATE PROCEDURE HP_UXPARAM_FINAL_VALUE(
    UFP_CODE D_STR_100,
    UXA_CODE D_STR_CODE,
    UA_CODE D_STR_100)
RETURNS (
    RES_FR_CODE D_STR_CODE,
    RES_FIELD_NAME D_STR_CODE,
    RES_IS_PARAM D_BOOL,
    RES_DEF_VALUE D_STR_100,
    RES_SOURCE_CODE D_STR_CODE,
    RES_SOURCE_FIELD D_STR_CODE,
    RES_SOURCE_IS_PARAM D_BOOL,
    RES_SOURCE_UFP_CODE D_STR_100,
    RES_USE_DEFAULT D_SMALLINT,
    RES_STATE D_SMALLINT,
    RES_LOCKED D_BOOL,
    RES_READONLY D_BOOL,
    RES_LOOKUP_CODE D_STR_100)
AS
declare variable ufp_parent_code d_str_code;
declare variable tmp_ufp_field d_str_code;
declare variable ufp_editor d_str_code;
begin
  if (uxa_code is null) then
    uxa_code = '';

  if (ua_code is null) then
    ua_code = '';

  res_def_value = '';
  res_source_ufp_code = '';
  res_source_code = '';
  res_source_field = '';
  res_source_is_param = 0;
  res_use_default = -1;
  res_state = 0;
  res_locked = 0;
  res_readonly = 0;

  res_lookup_code = '';
  res_fr_code = '';
  res_field_name = '';
  res_is_param = 0;

  select ufp_ufr_code, ufp_origin_name, ufp_parent_code, ufp_editor,
         iif(ufp_is_param = 1 or (ufd_is_table = 1 and ufp_pk_field <> 0), 1, 0)
  from uxfrparam, uxfrdata
  where ufp_code = :ufp_code and
        ufd_ufr_code = ufp_ufr_code and
        ufd_kind = 0
  into res_fr_code, res_field_name, ufp_parent_code, ufp_editor, res_is_param;

  tmp_ufp_field = '';
  if (ufp_parent_code is not null) then
  begin
    select ufp_origin_name
    from uxfrparam
    where ufp_code = :ufp_parent_code
    into tmp_ufp_field;

    res_lookup_code = res_fr_code || '$' || tmp_ufp_field;
    res_field_name = tmp_ufp_field || '$' || res_field_name; -- Название параметра Lookup
  end

  if (ufp_editor = 'current_user') then
  begin
    res_state = 2;
    res_use_default = 1;
    res_locked = 1;

    res_def_value = uxf$user();
  end else
  begin
    select upd_def_value, upd_locked
    from uxparamdef
    where upd_ufp_code = :ufp_code and
          upd_uxa_code is null
    into res_def_value, res_locked;
  
    if (row_count = 1) then
    begin
      res_use_default = 1;
      res_state = 2;
    end
  end

  if (res_locked = 0 and uxa_code <> '') then
  begin
    select upd_def_value, upd_locked
    from uxparamdef
    where upd_ufp_code = :ufp_code and
          upd_uxa_code = :uxa_code
    into res_def_value, res_locked;

    if (row_count = 1) then
    begin
      res_use_default = 1;
      res_state = 3;
    end
  end

  if (res_locked = 0 and ua_code <> '') then
  begin
    select uap_from_ufp_code, uap_default, uap_use_default, uap_locked
    from uxactionparam
    where uap_ua_code = :ua_code and
          uap_link_ufp_code = :ufp_code
    into res_source_ufp_code, res_def_value, res_use_default, res_locked;

    if (row_count = 1) then
      res_state = 4;

    if (res_source_ufp_code <> '' and res_use_default = 0) then
    begin
      select ufp_ufr_code, ufp_origin_name, ufp_is_param
      from uxfrparam
      where ufp_code = :res_source_ufp_code
      into res_source_code, res_source_field, res_source_is_param;
    end
  end

  if ((res_state = 2 and uxa_code = '' and ua_code = '') or
      (res_state = 3 and ua_code = '') or
       res_state = 4) then
    res_state = 1;

  if (res_locked = 1) then
  begin
    if (res_state <> 1) then
      res_readonly = 1;

    res_state = -res_state;
  end

  if (res_state not in (-1, 1)) then
    res_use_default = -1;

  suspend;
end
