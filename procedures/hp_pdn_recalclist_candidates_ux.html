CREATE PROCEDURE HP_PDN_RECALCLIST_CANDIDATES_UX(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    DAYS_CNT D_INTEGER,
    REFINANCE_LIST D_STR_8100,
    ACTIVE_LIST D_STR_8100)
AS
declare variable ma_id d_id;
declare variable ref_date_end d_date;
begin
  for select distinct k_kontract_no, k_disb_date, k_from_ma_app_id
  from kontract, kontrefinance, kontstate
  where kr_from_ma_id = k_from_ma_app_id
    and ks_from_k_id = k_id
    and minvalue(current_date, k_disb_date + 60) between ks_date_begin and ks_date_end
    and ks_ol_is_delete = 0
    and ks_state > 0
    and k_disb_date between :rep_date_begin and :rep_date_end
    and kr_from_k_id is not null
    and not exists(select 1 from pdn_manual_recalclist where pdr_from_k_id = k_id and pdr_status >=0)
  order by 2
  into k_kontract_no, k_disb_date, ma_id
  do
  begin
    ref_date_end = minvalue(current_date, k_disb_date + 60);
    days_cnt = current_date - k_disb_date;

    select list(k_kontract_no)
    from kontrefinance kef, kontract
    where kr_from_ma_id = :ma_id
      and k_id = kr_from_k_id
    into refinance_list;

    select list(k_kontract_no)
    from kontrefinance kef, kontract, oksummary
    where kr_from_ma_id = :ma_id
      and k_id = kr_from_k_id
      and oks_from_k_id = k_id
      and :ref_date_end between oks_date_begin and oks_date_end
      and (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc + oks_out_other) > 0
    into active_list;

    if (active_list is not null) then
      suspend;
  end
end
