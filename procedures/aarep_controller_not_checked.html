CREATE PROCEDURE AAREP_CONTROLLER_NOT_CHECKED(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    KF_NAME STR_100,
    K_KONTRACT_NO VARCHAR(30),
    CHECK_STATUS STR_100,
    CL_STATUS STR_100,
    C_NAME_FIO STR_100,
    K_CIRCLE INTEGER,
    K_DISB_DATE DATE)
AS
declare variable k_id id_type;
declare variable c_id id_type;
declare variable is_supsend integer;
declare variable cdv_value str_100;
declare variable check_result str_255;
begin
-- REP_REGION_ID =  1010000;

  /* 24.06.2015 На данный момент процедура работает только по специалистам ВО.
       НА данный момент СВЗ не проводят визит-осмотр.

       1)  Вытаскиваю все займы выданные за период
           которые на день выдачи числились за КЦ Сирена или СВЗ-партнере.
           Получаю клиентов этих кредитов.

       2)  Проверяю статус клиента (работает/безработный). Если он работает, проверяю, проведен ли визит - осмотр
       3)  Если ВО проведен, проверяю был ли установлен статус "Отказ от визит-осмотра", "Место работы не подтверждено"
       4) В качестве результат отчета попадают клиенты для которых Визит-осмотр не был произведен, или статус визит осмотра:   "Отказ от визит-осмотра", "Место работы не подтверждено"

    */

  for select reg.reg_name, kal.kal_new_kf_name, kont.k_kontract_no, kont.k_id, kncl.kc_from_c_id, ks.ks_date_begin,
             kont.k_circle
      from kontract kont
      inner join kontstate ks on ks.ks_from_k_id = kont.k_id
      inner join region_list(:rep_region_id, 1, -1, 1) on id_list = ks.ks_region_id
      inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
      inner join region reg on reg.reg_unicode = ks.ks_region_id
      where ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            kncl.kc_amount_approved > 0 and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.kal_new_kf_officer in (-- Сирена
      select 1000000000004
      from rdb$database
-- СВЗ - партнеры
      union all
      select hps.ps_from_c_id
      from hr_pstatus hps
      inner join joblist jl on jl.jl_id = ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where ks.ks_date_begin between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            jk.jk_system_code = 'JobSystem.LO_Partner')
      order by reg_name, ks.ks_date_begin, kal.kal_new_kf_name, kont.k_kontract_no
      into :reg_name, :kf_name, :k_kontract_no, :k_id, :c_id, :k_disb_date, :k_circle
  do
  begin
    is_supsend = 0;
    cdv_value = null;
--        select CDV_VALUE from HP_GET_CUSTOMDATAVALUE(0, :C_ID, 'work_status', :k_disb_date)
--        into  CDV_VALUE;

    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'work_status', :k_disb_date)
    into cdv_value;

    cdv_value = coalesce(cdv_value, '');

    if (cdv_value <> 'Не работает') then
    begin
      check_result = null;

      select rd_res.rd_code
      from controller cont
      inner join refdata rd_res on rd_res.rd_id = cont.contr_check_result
      inner join refdata rd_type on rd_type.rd_id = cont.contr_rd_type
      where cont.contr_from_k_id = :k_id and
            cont.contr_from_c_id = :c_id and
            cont.contr_checked_date <= :k_disb_date and
            rd_type.rd_code in ('ControllerType.Specialist_Visitor') and
            cont.contr_is_delete = 0
      into :check_result;

      check_result = coalesce(check_result, '');

      if (check_result = 'ControllerCheckStatus.Refused') then
      begin
        check_status = 'Отказ от визит-осмотра';
        is_supsend = 1;
      end
      else
      if (check_result = 'ControllerCheckStatus.NotConfirmed') then
      begin
        check_status = 'Место работы не подтверждено';
        is_supsend = 1;
      end
      else
      if (check_result = '') then
      begin
        check_status = 'Визит-осмотр не проведен';
        is_supsend = 1;
      end
    end

    if (is_supsend = 1) then
    begin
      if (exists(select *
                 from knclient kncl
                 inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id
                 where kncl.kc_from_c_id = :c_id and
                       kncl.kc_amount_approved > 0 and
                       ks.ks_state = 1 and
                       ks.ks_ol_is_delete = 0 and
                       ks.ks_date_begin < :k_disb_date -- дата выдачи кредита меньше даты выдачи текущего кредита
                       and
                       ks.ks_from_k_id <> :k_id)) then
        cl_status = 'Повторник';
      else
        cl_status = 'Новый клиент';

      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into :c_name_fio;

      suspend;
    end
  end
end
