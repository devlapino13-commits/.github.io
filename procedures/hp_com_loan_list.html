CREATE PROCEDURE HP_COM_LOAN_LIST(
    CMP_ID D_INTEGER,
    LIST_KIND D_STR_CODE)
RETURNS (
    CML_ID D_ID,
    CML_FROM_CMC_ID D_INTEGER,
    CML_DATE D_DATE,
    CML_FROM_REG_ID D_INTEGER,
    CML_INIT_C_ID D_ID,
    CML_OBJECT_KIND D_INTEGER,
    CML_OBJECT_ID D_ID,
    CML_ADD_PARAMS D_STR_LONG,
    CML_LAST_CMA_ID D_ID,
    CML_LAST_CMP_ID D_INTEGER,
    CML_LAST_STATUS D_SMALLINT,
    CML_FROM_UN_ID D_ID,
    CMC_NAME D_STR_100,
    REG_NAME D_STR_100,
    INIT_C_NAME D_STR_100,
    OBJECT_NAME D_STR_255,
    LAST_STATUS_STR D_STR_100,
    INIT_UN_NAME D_STR_CODE)
AS
declare variable status_approved d_smallint;
declare variable status_done d_smallint;
declare variable status_signing d_smallint;
declare variable status_sent d_smallint;
begin
  -- если cmp_id = 0, то показываем новые запросы, иначе то что принято на этот КК
  if (cmp_id is null) then
    cmp_id = 0;

  status_approved =  f_enum('com_action_status', 'approved', 'id');
  status_done = f_enum('com_action_status', 'done', 'id');
  status_signing = f_enum('com_action_status', 'signing', 'id');
  status_sent = f_enum('com_action_status', 'sent', 'id');

  if (trim(coalesce(list_kind, '')) = '') then
    list_kind = 'full';

  for select cml_id, cml_from_cmc_id, cml_date, cml_from_reg_id, cml_init_c_id, cml_object_kind, cml_object_id,
             cml_add_params, cml_from_un_id,
             cml_last_cma_id, cml_last_cmp_id, cml_last_status,
             cmc_name
  from com_list, com_cfg
  where cml_from_cmc_id = cmc_id and
        ((cml_last_cmp_id = :cmp_id and
         (:list_kind = 'full' or (:list_kind = 'execute' and cml_last_status in (:status_approved, :status_done, :status_signing)))) or
        (:list_kind = 'full' and
          :cmp_id = 0 and
          cml_last_cmp_id is null and
          cml_last_status = :status_sent))
  into cml_id, cml_from_cmc_id, cml_date, cml_from_reg_id, cml_init_c_id, cml_object_kind, cml_object_id,
       cml_add_params, cml_from_un_id,
       cml_last_cma_id, cml_last_cmp_id, cml_last_status,
       cmc_name
  do
  begin
    reg_name = '';
    init_c_name = '';
    object_name = '';
    last_status_str = '';
    init_un_name = '';

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :cml_from_reg_id
    into reg_name;

    select un_name
    from username
    where un_id = :cml_from_un_id
    into init_un_name;

    select c_name_fio
    from clients
    where c_id = :cml_init_c_id
    into init_c_name;

    object_name = f_objname(cml_object_kind, cml_object_id);
    last_status_str = f_enum('com_action_status', cml_last_status, 'name');

    suspend;
  end
end
