CREATE PROCEDURE AAREP_CASHCHECK_LIST_FORM(
    REP_ID INTEGER)
RETURNS (
    CCL_BANKNOTE TYPE OF COLUMN CASHCHECKLIST.CCL_BANKNOTE,
    CCL_BANKNOTE_NUM INTEGER,
    CCL_BANKNOTE_TXT VARCHAR(20),
    CCL_COUNT TYPE OF COLUMN CASHCHECKLIST.CCL_COUNT,
    CCL_SUMMA DECIMAL(18,4))
AS
declare variable curr_name varchar(20);
declare variable curr_cent varchar(20);
begin
  select cm_currency_name, cm_currency_cent
  from company
  into :curr_name, curr_cent;

  for with tab
      as (select ccl_banknote, ccl_count
          from cashchecklist
          where ccl_from_ccm_id = :rep_id
          union
          select bnl_id as ccl_banknote, 0 as cl_count
          from bondlist)
      select ccl_banknote, sum(ccl_count)
      from tab
      group by ccl_banknote
      order by ccl_banknote desc
      into :ccl_banknote, :ccl_count
  do
  begin
    if (ccl_count is null) then
      ccl_count = 0;
    ccl_summa = ccl_banknote * ccl_count;
    if (ccl_banknote >= 1) then
      ccl_banknote_num = ccl_banknote;
    else
      ccl_banknote_num = mod(ccl_banknote * 100, 100);
    if (ccl_banknote >= 1) then
      ccl_banknote_txt = curr_name;
    else
      ccl_banknote_txt = curr_cent;
    suspend;
  end
end
