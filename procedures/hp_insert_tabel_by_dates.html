CREATE PROCEDURE HP_INSERT_TABEL_BY_DATES(
    SAT_YEAR_MONTH D_STR_10,
    C_ID D_ID,
    SAWT_ID D_INTEGER,
    REG_ID D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    WEEKEND_LIST D_STR_100,
    IS_CUSTOM_WEEKEND D_BOOL)
AS
declare variable hours d_integer;
declare variable scd_work_hours d_smallint;
declare variable scd_date d_date;
declare variable work_days d_integer;
declare variable day_of_week d_integer;
declare variable scal_year d_smallint;
declare variable scal_name d_str_100;
declare variable s_where d_str_255;
declare variable i d_integer;
begin
  hours = 0;
  work_days = 0;
  weekend_list = '.' || weekend_list;

  -- создаем запись с полученным sawt_id, если ее нету
  s_where = ' where sat_year_month = ' || sat_year_month || ' and sat_from_c_id = ' || c_id || ' and sat_office_id = ' || reg_id || ' and sat_from_sawt_id = ';
  execute statement 'select count(*) from sa_tabelo' || s_where || :sawt_id into i;
  if (i = 0) then
    insert into sa_tabelo (sat_year_month, sat_from_c_id, sat_from_sawt_id, sat_office_id, sat_day_1, sat_day_2, sat_day_3, sat_day_4, sat_day_5, sat_day_6, sat_day_7, sat_day_8, sat_day_9, sat_day_10, sat_day_11, sat_day_12, sat_day_13, sat_day_14,
                           sat_day_15, sat_day_16, sat_day_17, sat_day_18, sat_day_19, sat_day_20, sat_day_21, sat_day_22, sat_day_23, sat_day_24, sat_day_25, sat_day_26, sat_day_27, sat_day_28, sat_day_29, sat_day_30, sat_day_31)
    values (:sat_year_month, :c_id, :sawt_id, :reg_id, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours, :hours,
            :hours, :hours, :hours, :hours, :hours, :hours, :hours);

  select sowd_days
  from sa_office_working_days
  where sowd_from_reg_id = :reg_id and
        :date_begin between sowd_date_begin and sowd_date_end
  into work_days; -- количество рабочих дней офиса в неделю (5, 6 или 7 дней)

  if (work_days = 0) then
    select
           case left(rd.rd_code, 14)
             when 'office_hours_6' then 6
             when 'office_hours_7' then 7
             else 5
           end
    from refdata rd
    inner join region reg on reg.reg_office_hours_id = rd.rd_id
    where reg.reg_unicode = :reg_id
    into work_days; -- количество рабочих дней офиса в неделю из ref_data

  -- перебираем дни отчетного месяца из sa_calendays (должен быть заполнен до этого)
  -- sa_calendays заполняется кадровиками в начале года и с расчетом 5 рабочих дней в неделю

  scal_year = extract(year from date_begin);

  for with t as (select cast('' || scal_day || '.' || scal_month || '.' || :scal_year as date) holiday,
                        scal_name
                 from sa_calendar
                 where scal_active = 1)
      select scd_date,
             scd_work_hours,
             scal_name
      from sa_calendays
      left join t on scd_date = holiday
      where scd_year_month = :sat_year_month and
            scd_date between :date_begin and :date_end
      into scd_date, -- календарный день отчетного месяца
           scd_work_hours, -- кол-во рабочих часов
           scal_name -- название праздника
  do
  begin
    i = extract(day from scd_date);
    day_of_week = extract(weekday from scd_date);
    hours = scd_work_hours;

    -- офис работает больше 5 дней в неделю и текущий день суббота
    if (work_days > 5 and
        day_of_week = 6) then
      hours = 8;

    -- офис работает 7 дней в неделю и текущий день воскресенье
    if (work_days = 7 and
        day_of_week = 0) then
      hours = 8;

    if (is_custom_weekend = 1 and
        weekend_list containing ('.' || i || '.')) then
      hours = 0; -- выходные дни после индивидуального редактирования
    else if (work_days > 5 and
        (scal_name = 'Перенос' or
        scal_name = 'Курман Айт')) then
      hours = 8; -- офисы с графиком 6-7 дней в дни переноса работают
    else if (weekend_list containing ('.' || i || '.')) then
      hours = 0; -- день праздничный по календарю, все отдыхают
    else if (scd_work_hours = 7) then
      hours = 7; -- день предпраздничный

    if (sawt_id = 0 and
        is_custom_weekend = 1 and
        weekend_list not containing ('.' || i || '.')) then
    begin
      if (scd_work_hours = 7) then
        hours = 7; -- RFCKG-3882
      else
        hours = 8; -- RFCKG-3400
    end

    -- вид отпуска: трудовой; неоплачиваемый; по беременности и родам; по уходу за ребенком без содержания; декретный
    if (sawt_id in (7, 8, 17, 18, 24)) then
      hours = 8;

    -- обновляем часы работы в графике
    execute statement 'update sa_tabelo set sat_day_' || i || ' = ' || hours || s_where || sawt_id;

    -- если sawt_id отпуск, то обнуляем часы работы в строке "Отработано" за этот день
    if (sawt_id > 0) then
      execute statement 'update sa_tabelo set sat_day_' || i || ' = 0' || s_where || '0';
  end

  -- подсчет итоговых дней и часов
  execute procedure hp_tabel_calc_days_hours(:sat_year_month, :c_id, :reg_id, :sawt_id);
  if (sawt_id > 0) then
    execute procedure hp_tabel_calc_days_hours(:sat_year_month, :c_id, :reg_id, 0);
end
