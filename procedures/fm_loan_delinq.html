CREATE PROCEDURE FM_LOAN_DELINQ(
    UN_ID DECIMAL(18,0),
    REP_DATE DATE,
    REG_ID INTEGER,
    FIND_TEXT VARCHAR(100) = '')
RETURNS (
    RES_TEXT VARCHAR(255),
    RES_DELINQ_DAY INTEGER,
    RES_BALANCE DECIMAL(18,4),
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0))
AS
declare variable k_id decimal(18,0);
declare variable kf_id decimal(18,0);
declare variable k_kontract_no varchar(20);
declare variable kf_name varchar(100);
declare variable c_name_fio varchar(100);
declare variable lo_id decimal(18,0);
declare variable need_suspend integer;
declare variable db_id integer;
declare variable max_delinq integer;
declare variable regions_str d_str_4096;
begin
    db_id = f_get_db_id();

    max_delinq = 9999;
    if (db_id = 81001) then
        max_delinq = 360;

    select un_from_c_id
    from username
    where un_id = :un_id
    into :lo_id;

    if (lo_id is null) then
        exit;

    if (not exists(select ps_from_reg_id
                   from hr_pstatus, joblist, jobkind
                   where ps_from_c_id = :lo_id and
                         ps_date_begin <= current_date and
                         ps_date_end >= current_date and
                         ps_status = 1 and
                         ps_from_jl_id = jl_id and
                         jl_kind = jk_id and
                         jk_system_code in ('JobSystem.Region_KM') and
                         ps_is_working = 1)) then
        reg_id = -1;
    
    if (reg_id <> -1) then
    begin
        regions_str = '.';
    
        for select id_list
            from region_list(:reg_id, 1, 3, 1)
            into :reg_id
    do
        regions_str = regions_str || :reg_id || '.' ;
    end
    else
        regions_str = '.-1.';

    for select value_out
        from str_with_separator_parser(:regions_str, '.')
        into :reg_id
    do
    begin
        -- Нужно для того, чтобы у заменяющих сотрудников отображались данные того сотрудника, которого они заменяют
        for select res_c_id
            from hp_staff_changer_list(:un_id, 1)
            where :reg_id = -1
            union
            select res_c_id
            from hp_staff_changer_list(:un_id, -1)
            where :reg_id <> -1
            into :lo_id
        do
        begin
            for select k_id, kf_id, k_kontract_no, res_delinq_day
                from fm_loan_delinq_tmp(:rep_date, :reg_id, :lo_id, :max_delinq)
                into :k_id, :kf_id, :k_kontract_no, :res_delinq_day
            do
            begin
                need_suspend = iif(find_text = '', 1, 0);
        
                obj_kind = 'kontract';
                obj_id = :k_id;
        
                -- Отображаем только задолженность по ОС
                select sum(ok.ok_sh_os + ok.ok_os)
                from operkred ok
                where ok.ok_from_k_id = :k_id and
                      ok.ok_sh_date <= :rep_date and
                      ok.ok_ol_is_active = 1 and
                      ok.ok_type < 7
                into :res_balance;
        
                select kal_new_kf_name
                from knaktlist
                where kal_from_kf_id = :kf_id and
                      knaktlist.is_delete = 0 and
                      kal_date <= :rep_date and
                      kal_date_end >= :rep_date
                into :kf_name;
        
                select c_name_fio
                from knclient, clients
                where kc_from_k_id = :k_id and
                      kc_amount_approved > 0 and
                      kc_from_c_id = c_id
                into :c_name_fio;
        
                res_text = kf_name || ' | ' || k_kontract_no || ' | ' || c_name_fio;
        
                if (find_text <> '') then
                begin
                    if (res_text like '%' || :find_text || '%') then
                        need_suspend = 1;
                    else
                    if (exists(select 1
                               from knclient, clients
                               where kc_from_k_id = :k_id and
                                     kc_amount_approved > 0 and
                                     kc_from_c_id = c_id and
                                     c_name_fio like '%' || :find_text || '%')) then
                        need_suspend = 1;
                end
        
                if (need_suspend = 1) then
                    suspend;
            end
        end
    end
end
