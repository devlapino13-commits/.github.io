CREATE PROCEDURE AAREP_ACCOUNT_JOURNAL(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AJ_DATE DATE,
    AJ_DOC_NO INTEGER,
    AJ_CURRENCY VARCHAR(3),
    AJ_ACCOUNT_DT VARCHAR(20),
    AJ_ACCOUNT_KT VARCHAR(20),
    AJ_SUMMA DECIMAL(18,4),
    AJ_USER_NAME VARCHAR(30),
    AJ_SAVE_TIME TIMESTAMP,
    AJ_SUMMA_EKV DECIMAL(18,4),
    AJ_CURR_RATE DECIMAL(18,4),
    AJ_REGION_NAME VARCHAR(100),
    AJ_ACC_SHORT_DT VARCHAR(20),
    AJ_ACC_SHORT_KT VARCHAR(20),
    AJ_CLIENT_NAME VARCHAR(200),
    AJ_ACC_NAME_DT VARCHAR(200),
    AJ_ACC_NAME_KT VARCHAR(200),
    OL_ID DECIMAL(18,0),
    AJ_MEMO VARCHAR(250))
AS
declare variable ob_mem1 varchar(1024);
declare variable ob_mem varchar(250);
declare variable curr_currency varchar(3);
declare variable client_dt decimal(18,0);
declare variable client_kt decimal(18,0);
declare variable c_name varchar(100);
declare variable aj_memo1 varchar(1024);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select cm_default_currency
  from company
  into :curr_currency;

  for select ol_id, ol_date, ol_doc_no, ol_memo, ob_currency, ob_account_dt, ob_account_kt, ob_summa_val, ob_summa,
             un_name, ol_datetime, ob_detail, ob_curr_rate, reg_code || ' ' || reg_name, dt.al_account, kt.al_account,
             ob_client_dt, ob_client_kt, dt.al_name, kt.al_name
      from username, operbook, operlist, region_list(:report_region_id, 1, -1, 1), region, acc_list dt, acc_list kt
      where ob_from_ol_id = ol_id and
            ((ob_currency = :report_currency) or
            (:report_currency = '000') or
            (:report_currency = '---')) and
            ol_date_oper_day between :report_date_begin and :report_date_end and
            operlist.is_delete = 0 and
            (ol_from_reg_id = id_list) and
            un_id = ol_user_id and
            reg_unicode = ol_from_reg_id and
            ob_account_dt = dt.al_id and
            ob_account_kt = kt.al_id

      order by ol_date_oper_day, ol_id
      into :ol_id, :aj_date, :aj_doc_no, :aj_memo1, :aj_currency, :aj_account_dt, :aj_account_kt, :aj_summa,
           :aj_summa_ekv, :aj_user_name, :aj_save_time, :ob_mem1, :aj_curr_rate, aj_region_name, :aj_acc_short_dt,
           :aj_acc_short_kt, :client_dt, :client_kt, :aj_acc_name_dt, :aj_acc_name_kt
  do
  begin
    aj_memo = null;
    ob_mem = null;
    aj_client_name = '';
    if (aj_memo1 is not null) then
      aj_memo = left(aj_memo1, 250);
    if (ob_mem1 is not null) then
      ob_mem = left(ob_mem1, 250);

    if (ob_mem is not null) then
    begin
      if (aj_memo is null) then
        aj_memo = ob_mem;
      else
        aj_memo = left(aj_memo || ' / ' || ob_mem, 250);
    end

    if (curr_currency = aj_currency) then
      aj_summa = 0;

    if (client_dt is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :client_dt
      into :aj_client_name;

      if (client_kt is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :client_kt
        into :c_name;
        aj_client_name = 'Дт: ' || :aj_client_name || ' / Кт: ' || :c_name;
      end
    end
    else
    begin
      if (client_kt is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :client_kt
        into :aj_client_name;
      end
    end

    suspend;
  end
end
