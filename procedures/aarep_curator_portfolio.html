CREATE PROCEDURE AAREP_CURATOR_PORTFOLIO(
    REP_C_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    OFFICER_NAME VARCHAR(100),
    REG_NAME VARCHAR(100),
    BALANCE_BEGIN CURR_TYPE,
    KOL_CLI_BEGIN CURR_TYPE,
    DELINQ_BAL_7_BEGIN CURR_TYPE,
    DELINQ_BAL_30_BEGIN CURR_TYPE,
    BALANCE_END CURR_TYPE,
    KOL_CLI_END CURR_TYPE,
    DELINQ_BAL_7_END CURR_TYPE,
    DELINQ_BAL_30_END CURR_TYPE,
    BALANCE_BEGIN_MONTH CURR_TYPE,
    KOL_CLI_BEGIN_MONTH CURR_TYPE,
    DELINQ_BAL_7_BEGIN_MONTH CURR_TYPE,
    DELINQ_BAL_30_BEGIN_MONTH CURR_TYPE,
    BALANCE_BEGIN_DAY CURR_TYPE,
    KOL_CLI_BEGIN_DAY CURR_TYPE,
    DELINQ_BAL_7_BEGIN_DAY CURR_TYPE,
    DELINQ_BAL_30_BEGIN_DAY CURR_TYPE)
AS
declare variable REG_ID integer;
declare variable C_ID CURR_TYPE;
declare variable C_ID_PRIOR CURR_TYPE;
declare variable HALF_YEAR date;
declare variable REP_DATE_BEGIN date;
declare variable REP_DATE_BEGIN_MONTH date;
declare variable REP_DATE_BEGIN_DAY date;
begin
  c_id_prior = 0;
  for select co_officer_id, co_region_id
      from curatoroffice
      where (co_officer_id = :rep_c_id or
            :rep_c_id = 0) and
            :rep_date_end between co_date_begin and co_date_end
      into c_id, reg_id
  do
  begin
    if (c_id_prior <> c_id) then
    begin
      c_id_prior = c_id;
      select c_name_fio
      from clients
      where c_id = :c_id
      into officer_name;
    end

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;
--  дата начало отчета, начало полугодие
    half_year = cast('01.08.' || extract(year from :rep_date_end) as date) - 1;
    if (rep_date_end <= half_year) then
      rep_date_begin = cast('01.01.' || extract(year from :rep_date_end) as date) - 1;
    else
      rep_date_begin = half_year;

--  КП на начало периода
    balance_begin = 0;
    kol_cli_begin = 0;
    delinq_bal_7_begin = 0;
    delinq_bal_30_begin = 0;
    select sum(oks_balance), sum((select res_just_client_count
                                  from hp_get_just_client_in_loan(:rep_date_begin, k_id, k_disb_date))),
           sum(iif(:rep_date_begin - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(:rep_date_begin - oks_delinq_date > 30, oks_balance, 0))
    from oksummary, kontract
    where :rep_date_begin between oks_date_begin and oks_date_end and
          oks_region_id = :reg_id and
          oks_balance > 0 and
          oks_from_k_id = k_id
          and oks_close_state = 0 
    into balance_begin, kol_cli_begin, delinq_bal_7_begin, delinq_bal_30_begin;
    balance_begin = coalesce(balance_begin, 0);
    kol_cli_begin = coalesce(kol_cli_begin, 0);
    delinq_bal_7_begin = coalesce(delinq_bal_7_begin, 0);
    delinq_bal_30_begin = coalesce(delinq_bal_30_begin, 0);

--  КП на конец
    balance_end = 0;
    kol_cli_end = 0;
    delinq_bal_7_end = 0;
    delinq_bal_30_end = 0;
    select sum(oks_balance), sum((select res_just_client_count
                                  from hp_get_just_client_in_loan(:rep_date_end, k_id, k_disb_date))),
           sum(iif(:rep_date_end - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(:rep_date_end - oks_delinq_date > 30, oks_balance, 0))
    from oksummary, kontract
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_region_id = :reg_id and
          oks_balance > 0 and
          oks_from_k_id = k_id
          and oks_close_state = 0 
    into balance_end, kol_cli_end, delinq_bal_7_end, delinq_bal_30_end;
    balance_end = coalesce(balance_end, 0);
    kol_cli_end = coalesce(kol_cli_end, 0);
    delinq_bal_7_end = coalesce(delinq_bal_7_end, 0);
    delinq_bal_30_end = coalesce(delinq_bal_30_end, 0);

--  Новые добавленные разработки по письму Отабека от 13.09.2016
--  КП на начало месяца
    rep_date_begin_month = rep_date_end - extract(day from :rep_date_end);

    balance_begin_month = 0;
    kol_cli_begin_month = 0;
    delinq_bal_7_begin_month = 0;
    delinq_bal_30_begin_month = 0;
    select sum(oks_balance), sum((select res_just_client_count
                                  from hp_get_just_client_in_loan(:rep_date_begin_month, k_id, k_disb_date))),
           sum(iif(:rep_date_begin_month - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(:rep_date_begin_month - oks_delinq_date > 30, oks_balance, 0))
    from oksummary, kontract
    where :rep_date_begin_month between oks_date_begin and oks_date_end and
          oks_region_id = :reg_id and
          oks_balance > 0 and
          oks_from_k_id = k_id
          and oks_close_state = 0 
    into balance_begin_month, kol_cli_begin_month, delinq_bal_7_begin_month, delinq_bal_30_begin_month;
    balance_begin_month = coalesce(balance_begin_month, 0);
    kol_cli_begin_month = coalesce(kol_cli_begin_month, 0);
    delinq_bal_7_begin_month = coalesce(delinq_bal_7_begin_month, 0);
    delinq_bal_30_begin_month = coalesce(delinq_bal_30_begin_month, 0);

--  КП на начало дня
    rep_date_begin_day = :rep_date_end - 1;
    balance_begin_day = 0;
    kol_cli_begin_day = 0;
    delinq_bal_7_begin_day = 0;
    delinq_bal_30_begin_day = 0;
    select sum(oks_balance), sum((select res_just_client_count
                                  from hp_get_just_client_in_loan(:rep_date_begin_day, k_id, k_disb_date))),
           sum(iif(:rep_date_begin_day - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(:rep_date_begin_day - oks_delinq_date > 30, oks_balance, 0))
    from oksummary, kontract
    where :rep_date_begin_day between oks_date_begin and oks_date_end and
          oks_region_id = :reg_id and
          oks_balance > 0 and
          oks_from_k_id = k_id
          and oks_close_state = 0 
    into balance_begin_day, kol_cli_begin_day, delinq_bal_7_begin_day, delinq_bal_30_begin_day;
    balance_begin_day = coalesce(balance_begin_day, 0);
    kol_cli_begin_day = coalesce(kol_cli_begin_day, 0);
    delinq_bal_7_begin_day = coalesce(delinq_bal_7_begin_day, 0);
    delinq_bal_30_begin_day = coalesce(delinq_bal_30_begin_day, 0);

    suspend;
  end
end
