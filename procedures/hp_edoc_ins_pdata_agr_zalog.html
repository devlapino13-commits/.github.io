CREATE PROCEDURE HP_EDOC_INS_PDATA_AGR_ZALOG(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable count_zalog d_smallint;
begin
  -- Процедура создает Согласие на обработку персональных данных залогодателя
  -- Готовит бланки и отправляет СМС.
  res_text = '';
  res_is_error = 0;

  count_zalog = 0;

  select count(z.z_from_k_id)
  from zalog z
  where z.z_from_k_id = :k_id
  into :count_zalog;

  if (count_zalog = 1) then
  begin
    select edl_id, res_text
    from hp_edoc_insert('pdata.agreement.zalog', f_objkind('kontract', 'id'), :k_id, :un_id)
    into :edl_id, :res_text;
  end
  else
  begin
    res_text = 'Залогодателей более одного человека';
  end

  if (res_text <> '') then
    res_is_error = 1;

  suspend;
end
