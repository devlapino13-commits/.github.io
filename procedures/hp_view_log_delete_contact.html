CREATE PROCEDURE HP_VIEW_LOG_DELETE_CONTACT(
    DATE_BEGIN DATE,
    DATE_END DATE)
RETURNS (
    C_NAME_FIO D_STR_255,
    OBJECT D_STR_255,
    SUBJECT D_STR_255,
    TYPE_SUBJECT D_STR_255,
    MEMO D_STR_255,
    PCL_SOURCE_SYSTEM D_STR_20,
    RD_NAME D_STR_255,
    PCL_DATETIME D_DATETIME,
    UN_FIO D_STR_255,
    PCL_DECLARER D_STR_20)
AS
declare variable pcl_un_id d_id;
declare variable pcl_reason d_integer;
declare variable pcl_obj_id d_id;
declare variable pcl_cleared_data d_blob_text;
declare variable pcl_obj_kind d_integer;
begin
  for select
             case
               when pcl_source_system = 0 then 'Finstart'
               when pcl_source_system = 1 then 'FsMobile'
             end,
             pcl_un_id, pcl_reason, pcl_datetime, pcl_cleared_data, pcl_obj_id, pcl_obj_kind, r.rd_name
      from person_clear_log pcl
      inner join refdata r on r.rd_id = pcl.pcl_declarer
      where cast(pcl_datetime as date) between :date_begin and :date_end
      into :pcl_source_system, :pcl_un_id, :pcl_reason, :pcl_datetime, :pcl_cleared_data, pcl_obj_id, pcl_obj_kind, pcl_declarer
  do
  begin
    un_fio = '';
    rd_name = '';
    object = '';
    subject = '';
    type_subject = 'иные';
    memo = '';
    select u.un_fio
    from username u
    where u.un_id = :pcl_un_id
    into :un_fio;

    select r.rd_name
    from refdata r
    where r.rd_id = :pcl_reason
    into :rd_name;

    if (pcl_obj_kind = f_objkind('contacts', 'id')) then
    begin
      select cl.c_name_fio, cs.cc_name
      from clients cl
      inner join contacts cs on cs.cc_from_c_id = cl.c_id
      where cs.cc_id = :pcl_obj_id
      into :c_name_fio, :object;

      subject = f_json_str_get_value(:pcl_cleared_data, 'cc_cell_no', '');

      type_subject = 'иные';
      memo = 'номер телефона и иные контактные данные';
    end
    suspend;
  end
end
