CREATE PROCEDURE AAREP_KONTRACT_OUT_BALANCE(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REP_C_NAME VARCHAR(100),
    REP_OFF_END_OS DECIMAL(18,4),
    REP_OFF_END_PERC DECIMAL(18,4),
    REP_OFF_END_OTHER DECIMAL(18,4),
    REP_WRITEOFF_DATE DATE,
    K_DISB_DATE DATE,
    K_KONTRACT_NO VARCHAR(20),
    DISB_REG_NAME VARCHAR(100),
    AA_PRODUCT_NAME VARCHAR(100),
    AAREP_DISB_LO_FIO VARCHAR(100),
    RD_NAME VARCHAR(255),
    REST_MEMO D_STR_255,
    STATE_STR D_STR_100,
    SBA_DATE_BEGIN D_DATE,
    SBL_NAME D_STR_100,
    OL_DOC_CLIENT_NAME D_STR_255,
    FACT_ADR D_STR_255,
    INDEXED_ADR D_STR_255,
    CELL_NO_CLIENTS D_STR_100,
    CONT_CLIENTS_REL D_STR_100)
AS
declare variable RD_ID ID_TYPE;
declare variable K_FACT_DISB_LO_ID ID_TYPE;
declare variable K_ID D_ID;
declare variable C_ID D_ID;
declare variable REP_OFF_BEGIN_OS decimal(18,4);
declare variable REP_OFF_BEGIN_PERC decimal(18,4);
declare variable REP_OFF_BEGIN_OTHER decimal(18,4);
declare variable REP_OFF_SH_OS decimal(18,4);
declare variable REP_OFF_SH_PERC decimal(18,4);
declare variable REP_OFF_SH_OTHER decimal(18,4);
declare variable REP_OFF_PAY_OS decimal(18,4);
declare variable REP_OFF_PAY_PERC decimal(18,4);
declare variable REP_OFF_PAY_OTHER decimal(18,4);
declare variable REP_OFF_DEAD_OS decimal(18,4);
declare variable REP_OFF_DEAD_PERC decimal(18,4);
declare variable REP_OFF_DEAD_OTHER decimal(18,4);
declare variable REP_LASTPAY_DATE D_DATE;
declare variable REP_FACT_WRITEOFF_DATE D_DATE;
declare variable CREL_OTHER_C_ID D_ID;
declare variable REL_NAME D_STR_100;
declare variable CREL_FROM_REL_ID D_ID;
declare variable C_NAME_FIO D_STR_100;
declare variable C_PLASTCARD_NO D_STR_100;
begin
  for select distinct oko_from_k_id,
                      oko_from_c_id,
                      pr_name,
                      kont.k_fact_disb_lo_id,
                      r.reg_code || ' ' || r.reg_name,
                      kont.k_disb_date,
                      kont.k_kontract_no
      from operkroff, kontract kont, product,  region r
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(r.reg_unicode as d_integer)
      where k_id = oko_from_k_id and
            k_from_pr_id = pr_id and
            kont.k_fact_disb_reg_id = r.reg_unicode
      into :k_id, :c_id, :aa_product_name, k_fact_disb_lo_id,
            disb_reg_name, k_disb_date, k_kontract_no
  do
  begin

    aarep_disb_lo_fio = null;

    select c_name_fio
    from clients
    where c_id = :k_fact_disb_lo_id
    into :aarep_disb_lo_fio;
    if (aarep_disb_lo_fio is null) then
      aarep_disb_lo_fio = 'не найден';


    select c_name_fio
    from clients cl
      where c_id = :c_id
      into :rep_c_name;



    select sum(oko_sh_os + oko_os), sum(oko_sh_perc + oko_perc), sum(oko_sh_other + oko_other)
    from operkroff, operlist
    where oko_from_ol_id = ol_id and
          oko_from_k_id = :k_id and
          oko_from_c_id = :c_id and
          ol_date < :rep_date_begin and
          operlist.is_delete = 0
    into :rep_off_begin_os, :rep_off_begin_perc, :rep_off_begin_other;
    if (rep_off_begin_os is null) then
      rep_off_begin_os = 0;
    if (rep_off_begin_perc is null) then
      rep_off_begin_perc = 0;
    if (rep_off_begin_other is null) then
      rep_off_begin_other = 0;

    select sum(oko_sh_os), sum(oko_sh_perc), sum(oko_sh_other)
    from operkroff, operlist
    where oko_from_ol_id = ol_id and
          oko_from_k_id = :k_id and
          oko_from_c_id = :c_id and
          ol_date_oper_day between :rep_date_begin and :rep_date_end and
          oko_type = 1 and
          operlist.is_delete = 0
    into :rep_off_sh_os, :rep_off_sh_perc, :rep_off_sh_other;
    if (rep_off_sh_os is null) then
      rep_off_sh_os = 0;
    if (rep_off_sh_perc is null) then
      rep_off_sh_perc = 0;
    if (rep_off_sh_other is null) then
      rep_off_sh_other = 0;

    select sum(oko_os), sum(oko_perc), sum(oko_other)
    from operkroff, operlist
    where oko_from_ol_id = ol_id and
          oko_from_k_id = :k_id and
          oko_from_c_id = :c_id and
          ol_date_oper_day between :rep_date_begin and :rep_date_end and
          oko_type in (4, 7) and
          operlist.is_delete = 0
    into :rep_off_pay_os, :rep_off_pay_perc, :rep_off_pay_other;
    if (rep_off_pay_os is null) then
      rep_off_pay_os = 0;
    if (rep_off_pay_perc is null) then
      rep_off_pay_perc = 0;
    if (rep_off_pay_other is null) then
      rep_off_pay_other = 0;

    select sum(oko_os), sum(oko_perc), sum(oko_other)
    from operkroff, operlist
    where oko_from_ol_id = ol_id and
          oko_from_k_id = :k_id and
          oko_from_c_id = :c_id and
          ol_date_oper_day between :rep_date_begin and :rep_date_end and
          oko_type = 5 and
          operlist.is_delete = 0
    into :rep_off_dead_os, :rep_off_dead_perc, :rep_off_dead_other;
    if (rep_off_dead_os is null) then
      rep_off_dead_os = 0;
    if (rep_off_dead_perc is null) then
      rep_off_dead_perc = 0;
    if (rep_off_dead_other is null) then
      rep_off_dead_other = 0;

    rep_off_end_os = rep_off_begin_os + rep_off_sh_os + rep_off_pay_os;
    rep_off_end_perc = rep_off_begin_perc + rep_off_sh_perc + rep_off_pay_perc ;
    rep_off_end_other = rep_off_begin_other + rep_off_sh_other + rep_off_pay_other + rep_off_dead_other;


    if ((rep_off_begin_os <> 0) or (rep_off_begin_perc <> 0) or (rep_off_begin_other <> 0) or (rep_off_end_os <> 0) or (rep_off_end_perc <> 0) or (rep_off_end_other <> 0) or (rep_off_sh_os <> 0) or (rep_off_sh_perc <> 0) or (rep_off_sh_other <> 0) or (rep_off_pay_os <> 0) or (rep_off_pay_perc <> 0) or (rep_off_pay_other <> 0) or (rep_off_dead_os <> 0) or (rep_off_dead_perc <> 0) or (rep_off_dead_other <> 0)) then
    begin
      select min(ol_date_oper_day), max(ol_date_oper_day)
      from operkroff, operlist
      where oko_from_k_id = :k_id and
            oko_from_ol_id = ol_id and
            oko_from_c_id = :c_id and
            operlist.is_delete = 0
      into :rep_writeoff_date, :rep_lastpay_date;

      --Добавил новое поле. Дата перехода на внесистемный учет 09.04.15 Музаффар >>>>>
      ol_doc_client_name = null;
      rd_id = null;
      rep_fact_writeoff_date = null;
      select distinct ok_ol_date, ol.ol_doc_client_name, --Dobavlen 090316
                      ol.ol_from_rd_id --Dobavlen 090316
      from operkred ok, operkroff, operlist ol
      where ok_ol_is_active = 1 and
            ok_from_c_id = 0 and
            ok_type = 5 and
            ok_from_k_id = :k_id and
            ok_from_ol_id = oko_from_ol_id and
            oko_from_c_id = :c_id and
            (ok.ok_os + ok.ok_perc) < 0 and
            ok_from_ol_id = ol_id and
            ol.is_delete = 0
      order by ol_date desc
      rows 1
      into :rep_fact_writeoff_date, ol_doc_client_name, rd_id; --Конец <<<<<
      if (ol_doc_client_name is null) then
        ol_doc_client_name = '';

      rd_name = null;
      select rd_name
      from refdata rd
      where rd.rd_is_active = 1 and
            rd.rd_id = :rd_id
      into rd_name;
      if (rd_name is null) then
        rd_name = 'Не заполнен';

      -----------------------------------------------------------------

      rest_memo = null;
      select ol_memo
      from operrest opr, operlist, region_list(:rep_region_id, 1, 3, 1)
      where opr.or_ol_is_active = 1 and
            opr.or_ol_date between :rep_date_begin and :rep_date_end and
            or_from_ol_id = ol_id and
            opr.or_from_reg_id = id_list and
            opr.or_from_k_id = :k_id rows 1
      into rest_memo;
      ----------------------------------------------------------------------------------------


      state_str ='закрытый';

     if (exists (select oks.oks_close_state
      from oksummary oks
      where oks.oks_from_k_id = :k_id and
            :rep_date_end between oks.oks_date_begin and oks.oks_date_end
             and  oks_close_state = 1)) then
             state_str = 'Спец.учет';


     if (exists(select  ks.ks_state
      from kontstate ks
      where ks.ks_from_k_id=:k_id
      and :rep_date_end between ks.ks_date_begin and ks.ks_date_end
      and ks.ks_ol_is_delete=0
      and ks.ks_state in (1))) then
             state_str= 'Активный';



   indexed_adr = null;

    select adl.adl_address_str
    from addrclient adc
    inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
    where adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 1
          rows 1
    into indexed_adr;

    fact_adr = null;

    select adl.adl_address_str
    from addrclient adc
    inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
    where adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 2
          rows 1
    into fact_adr;


    cell_no_clients = f_cont_info('clients', :c_id, 'cell_main', 'ok');
    
    crel_other_c_id =null;

     select  ct.crel_other_c_id ,ct.crel_from_rel_id
     from clientrelation ct
     where ct.crel_from_c_id =:c_id
      and crel_from_rel_id in ('1', '2') and
          crel_is_active = 1
     rows 1
     into  :crel_other_c_id ,crel_from_rel_id;

     c_name_fio=null;
     c_plastcard_no =null;
     rel_name =null;

     select  tt.c_name_fio,tt.c_plastcard_no,tt.rel_name
     from  HP_CLIENT_RELATION_LIST (:c_id, 0) tt
     rows 1
     into  c_name_fio, c_plastcard_no,rel_name;


      cont_clients_rel = c_name_fio || ', ' || rel_name || ', ' || c_plastcard_no;

      sba_date_begin = null;
      sbl_name = null;


      select sba.sba_date_begin, sbl.sbl_name
      from sb_akt sba, sb_list sbl
      where sba.sba_is_delete = 0 and
            sba.sba_from_k_id = :k_id and
            sba_from_sbl_id <> 0 and
            sba.sba_from_sbl_id = sbl.sbl_id and
            :rep_date_end between sba_date_begin and sba_date_end and
            sba.sba_status = 2 rows 1
      into sba_date_begin, sbl_name;
      if (sbl_name is null) then
      begin
        sbl_name = '';
      end

      suspend;
    end
  end
end
