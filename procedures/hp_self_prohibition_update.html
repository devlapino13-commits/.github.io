CREATE PROCEDURE HP_SELF_PROHIBITION_UPDATE(
    EIL_ID D_ID)
AS
declare variable JSON_STR D_STR_8100;
declare variable IS_CR_LOCKED D_SMALLINT;
declare variable EIL_STATUS_DATETIME D_DATETIME;
declare variable EIC_CODE D_STR_CODE;
declare variable EIL_STATUS D_SMALLINT;
declare variable SPD_ID D_ID;
declare variable SPD_EIL_ISHENIM D_ID;
declare variable SPD_EIL_SSOUND D_ID;
declare variable TOTAL_RESULT D_INTEGER;
declare variable IS_FOUND_CLIENT D_SMALLINT;
declare variable BLS_ID D_ID;
declare variable SELF_BAN_ENABLED D_SMALLINT;
declare variable FREEZED D_SMALLINT;
declare variable COOLING_PERIOD D_SMALLINT;
declare variable IS_LOCKED_ISHENIM D_SMALLINT;
begin

  select eic.eic_code, eil.eil_status, eil.eil_from_bls_id
  from extinfolist eil, extinfocfg eic
  where eic.eic_code in ('stop_cr_kib', 'stop_cr_crif') and
        eic.eic_id = eil.eil_from_eic_id and
        eil.eil_id = :eil_id
  into :eic_code, eil_status, bls_id;

  select cast(blx_blob as d_blob_text)
  from hp_get_blobex_blob(:bls_id)
  into :json_str;

  -- запрос от Ишеним
  if (eic_code = 'stop_cr_kib') then
  begin
    select spd.spd_id, spd.spd_eil_ishenim, spd.spd_eil_ssound
    from self_prohibition_detail spd
    where spd.spd_eil_ishenim = :eil_id
    into: spd_id, spd_eil_ishenim, spd_eil_ssound;

    if (eil_status = 2 and coalesce(json_str, '') <> '') then
    begin
      is_found_client = iif(f_json_str_get_value(:json_str, 'data.creditLockId', 'not') = '', 0, 1);

      if (is_found_client = 1) then
      begin
        is_locked_ishenim = iif(f_json_str_get_value(:json_str, 'data.isCreditLocked', '') = 'true', 1, 2);
        cooling_period = iif(f_json_str_get_value(:json_str, 'data.coolingOffPeriod.isCoolingOff', '') = 'true', 1, 2);

        if ( is_locked_ishenim = 1 or cooling_period = 1) then
          is_cr_locked = 1;
        else
          is_cr_locked = 2;
      end
      else
        is_cr_locked = -2; -- клиент не найден
    end
    else if (eil_status = -3) then
      is_cr_locked = -3; -- задача отменена
    else
      is_cr_locked = -1; -- ошибка (если eil_status -2 или json пустой)

    -- заливаем ответ от Ишеним
    update self_prohibition_detail spd
    set spd.spd_ishenim_result = :is_cr_locked
    where spd.spd_eil_ishenim = :eil_id;
  end
  else
  begin
    select spd.spd_id, spd.spd_eil_ishenim, spd.spd_eil_ssound
    from self_prohibition_detail spd
    where spd.spd_eil_ssound = :eil_id
    into: spd_id, spd_eil_ishenim, spd_eil_ssound;

    -- запрос от Сэйф энд Саунд
    if (eil_status = 2 and coalesce(json_str, '') <> '') then
    begin
      is_found_client = iif(f_json_str_get_value(:json_str, 'result', '') = 'Клиент не найден', 0, 1);

      if (is_found_client = 1) then
      begin
        self_ban_enabled = iif(f_json_str_get_value(:json_str, 'self_ban_enabled', '') = 'TRUE', 1, 0);
        freezed = iif(f_json_str_get_value(:json_str, 'freezed', '') = 'TRUE', 1, 0);

        if (self_ban_enabled = 1 or freezed = 1) then
          is_cr_locked = 1;
        else
          is_cr_locked = 2;
      end
      else
        is_cr_locked = -2; -- клиент не найден
    end
    else if (eil_status = -3) then
      is_cr_locked = -3; -- задача отменена
    else
      is_cr_locked = -1; -- ошибка (если eil_status -2 или json пустой)

    -- заливаем ответ от т Сэйф энд Саунд
    update self_prohibition_detail spd
    set spd.spd_ssound_result = :is_cr_locked
    where (spd.spd_eil_ssound = :eil_id);
  end

  -- вычисляем итоговый результат
  total_result = f_self_prohibition_total_result(:spd_eil_ishenim, :spd_eil_ssound);

  update self_prohibition_detail spd
  set spd.spd_total_result = :total_result
  where (spd.spd_id = :spd_id);
end
