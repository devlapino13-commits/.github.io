CREATE PROCEDURE CM_RU_0005_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable c_id id_type;
declare variable kc_last_k_id id_type;
declare variable sum_delinq_day integer;
begin
  for
  select c_name_fio, c_pass_no, c_id, k_disb_date
  from knclient, clients, kontract
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  and kc_from_c_id = c_id
  and c_citizen = 8
  and kc_from_k_id = k_id
  and k_disb_date <= '31.10.17'
  into res_info_text, kf_client_pass_no, c_id, kf_disb_date
  do
  begin
       /*       igusev 12.12.2014
                Согласно письма Зары от 11.12.2014
                от граждан КР можно брать вместо ID-карты (в качестве дополнительного документа) другие документы гос.образца: свидетельство о рождении клиента, водительское удостоверение и диплом о высшем/средне-специальном образовании.
      */
      if (not exists(
          select *
          from passdoctype, CLIENTDOCS, passkind
          where cds_from_c_id = :c_id
--          and cds_id > 0 -- кроме основного документа
--          and CDS_Is_Main_Doc = 0 -- добоавлено от 30.11.2015 после обновления, письмо от Санжара
          and cds_from_pk_id = pk_id
          and pk_from_pdt_id = pdt_id
          --and pdt_code = 'passport_card')) then   -- igusev 12.12.2014 закоментировал эту строку и добавил строку ниже
          and pdt_code in ('passport_card',
                           'certificate_birth',
                           'certificate_driver',
                           'diplom_vuz',
                           'diplom_suz',
                           'diplom_school',
                           'military_id',
                           'certificate_marriage',            --   certificate_marriage,  certificate_annulment добавил igusev 16.01.2015
                           'certificate_annulment',
                           'certificate_family'))) then
       begin
         kc_last_k_id = null;
         select ks_from_k_id from kontstate, knclient
         where ks_ol_is_delete = 0
         and ks_state = 1
         and ks_date_begin < :kf_disb_date
         and ks_from_k_id = kc_from_k_id
         and kc_amount_approved > 0
         and kc_from_c_id = :c_id
         order by ks_date_begin desc
         rows 1
         into kc_last_k_id;
         if (kc_last_k_id is null) then kc_last_k_id = 0;
    
         if (kc_last_k_id <> 0) then
         begin
    --                 Суммарно просроченные дни
           sum_delinq_day = null;
           select sum(ok_ol_date - ok_sh_date) from operkred
           where ok_ol_is_active = 1
           and ok_from_k_id = :kc_last_k_id
           and ok_type in (4,5)
           and ok_os + ok_perc < 0
           and ok_ol_date > ok_sh_date
           and ok_from_c_id=0
           into sum_delinq_day;
           if (sum_delinq_day is null) then sum_delinq_day = 0;
           if (sum_delinq_day > 14) then
           begin
             res_error_is_critical=2;
             res_error_text='Необходим хотя бы один документ гос. образца (паспорт пластик, водит. права, свидет. о рожден., диплом о высш/сред. образов), в предыдушем цикле просроч. дни в сумме>14';
             Suspend;
           end
         end else
         begin
           res_error_is_critical=2;
           res_error_text='Нет ни одного документа гос. образца (паспорт пластик, водит. права, свидет. о рожден., диплом о высш/сред. образов)';
           Suspend;
         end 
       end
  end
end
