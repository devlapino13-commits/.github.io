CREATE PROCEDURE AAREP_NBKR_REPORT(
    REP_REGION_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    C_NAME D_STR_100,
    K_NO D_STR_20,
    DISB_DATE D_DATE,
    KREDLINE_END D_DATE,
    CURRENCY VARCHAR(3),
    AMOUNT_APPROVED CURR_TYPE,
    INTEREST_RATE DECIMAL(18,2),
    LOAN_PURPOSE_CODE VARCHAR(3),
    BT_NAME D_STR_100,
    REGION_CODE D_STR_10,
    LKC_BALANCE D_CURRENCY,
    SH_CALC_PERC_CL D_CURRENCY,
    NAC_ZAOS D_CURRENCY,
    DELINQ_OS D_INTEGER,
    ZAD_OS_DELINQ_KC D_CURRENCY,
    DELINQ_PERC D_INTEGER,
    ZAD_PERC_DELINQ_KC D_CURRENCY,
    LKC_STATUS_STR D_STR_100,
    LKC_LLC_SUM D_CURRENCY,
    LLC_SUM_PERC D_CURRENCY,
    PERCENT_CLASSIF D_STR_20,
    RESTRUCT_DATE_LAST D_DATE,
    RESTRUCT_COUNT D_INTEGER,
    PR_IS_SECURED_STR D_STR_20,
    ACCRUAL_STATUS D_STR_100,
    C_TYPE_STR D_STR_20,
    PARALLELISM_SIGN D_STR_20,
    CONNECTION_TYPE D_STR_20,
    PR_NAME D_STR_100,
    REG_STAT_REP_REGION VARCHAR(30),
    PR_LOAN_TYPE_STR D_STR_10,
    REG_NAME VARCHAR(50),
    KF_BRANCH_NAME VARCHAR(50),
    CLIENT_INN D_STR_20,
    C_ID_TXT D_STR_20,
    K_DISB_KINDD D_STR_100,
    K_BY_RATING D_STR_CODE)
AS
declare variable k_id id_type;
declare variable c_id id_type;
declare variable lm_id id_type;
declare variable reg_id id_type;
declare variable k_stop_perc d_date;
declare variable pr_loan_type d_integer;
declare variable pr_is_secured d_integer;
declare variable rep_date_last d_date;
declare variable k decimal(18,7);
declare variable dd d_date;
declare variable kc_amount_approved_fact d_currency;
declare variable sh_calc_perc d_currency;
declare variable zad_perc_delinq d_currency;
declare variable zad_os_delinq d_currency;
declare variable aarep_perc_end d_currency;
declare variable rep_currency varchar(3);
declare variable def_curr varchar(3);
declare variable k_index_curr_rate curr_type;
declare variable cr_rate curr_type;
declare variable rep_date_beg d_date;
declare variable rep_period_begin d_integer;
declare variable rep_period_end d_integer;
declare variable pr_code d_str_100;
declare variable reg_id_old id_type;
declare variable pr_id d_id;
declare variable amount_approved_fact d_currency;
declare variable llc_sum_percl d_currency;
declare variable k_disb_kind d_integer;
begin
  k_disb_kindd = '';
  reg_id_old = 0;
  rep_date_last = :rep_date - extract(day from :rep_date);
  dd = :rep_date_last + 33;
  dd = :dd - extract(day from :dd);
  if (:rep_date < :dd) then
    rep_date = rep_date_last;
  rep_date_beg = dateadd(day, -extract(day from rep_date) + 1, rep_date);

  rep_period_begin = 0;
  rep_period_end = 0;
  if (extract(day from :rep_date_beg) = 1) then
    rep_period_begin = extract(year from :rep_date_beg) * 100 + extract(month from :rep_date_beg);
  if (extract(day from (:rep_date + 1)) = 1) then
    rep_period_end = extract(year from :rep_date) * 100 + extract(month from :rep_date);
  if ((rep_period_begin = 0) or (rep_period_end = 0)) then
    exit;

  for select kon.k_id, kon.k_from_pr_id, kon.k_disb_date, kon.k_kredline_end, kon.k_kontract_no, kon.k_stop_perc,
             round(kon.k_percent, 2), kon.k_index_curr_rate, kon.k_currency, kal.kal_new_reg_id, kon.k_disb_kind,
             iif(kon.k_by_rating = 1, 'По кредитному рейтингу', 'Стандартная выдача')
      from kontstate ks
      inner join kontract kon on kon.k_id = ks.ks_from_k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kon.k_from_kf_id
      inner join region_list(:rep_region_id, 1, 3, 1) on cast(kal.kal_new_reg_id as integer) = id_list
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            :rep_date between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      order by kal.kal_new_reg_id
      into :k_id, :pr_id, :disb_date, :kredline_end, :k_no, :k_stop_perc, :interest_rate, :k_index_curr_rate,
           :rep_currency, :reg_id, k_disb_kind, k_by_rating
  do
  begin
    k_disb_kindd = case :k_disb_kind
                     when 6 then 'На карту Мбулак'
                     when 5 then 'На карту клиента'
                     else 'Касса'
                   end;
    select pr.pr_loan_type, pr.pr_name, pr.pr_is_secured, pr.pr_code
    from product pr
    where pr.pr_id = :pr_id
    into :pr_loan_type, :pr_name, :pr_is_secured, :pr_code;

    if (reg_id_old <> reg_id) then
    begin
      reg_id_old = reg_id;

      select reg.reg_name, r2.reg_name, reg.reg_stat_rep_region
      from region reg
      inner join region r2 on reg.parent = r2.reg_unicode
      where reg.reg_unicode = :reg_id
      into reg_name, kf_branch_name, :reg_stat_rep_region;
    end

    delinq_perc = 0;
    delinq_os = 0;

    select iif((cast(:rep_date as date) - oks.oks_delinq_date < 0), 0, cast(:rep_date as date) - oks.oks_delinq_date)
    from oksummary oks
    where :rep_date between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_from_k_id = :k_id and
          oks_close_state = 0
    into :delinq_os;
    delinq_perc = delinq_os;

    select iif(kcp_period = :rep_period_end, kcp_perc_saldo, 0) as aarep_perc_end
    from kontcalcperc kcp
    where kcp.kcp_from_k_id = :k_id and
          kcp_period between :rep_period_begin and :rep_period_end
    into :aarep_perc_end;

    llc_sum_percl = 0;
    if (delinq_os >= 90 or coalesce(k_stop_perc, '31.12.9999') <= :rep_date) then
    begin
      llc_sum_percl = aarep_perc_end;
    end

    select restruct_kol, restruct_date_last -- < получаем количество реструктуризаций
    from aarep_get_restructuring(:k_id, :rep_date, 41701)
    into restruct_count, restruct_date_last;

    -- Тип залога
    pr_is_secured_str = (case pr_is_secured
                           when 0 then 'без залога'
                           when 1 then 'залог'
                           when 2 then 'поручители'
                           when 3 then 'залог,поручит.'
                           else ''
                         end);

    pr_loan_type_str = '';
    pr_loan_type_str = (iif(pr_loan_type = 0, 'группа', 'индив.'));

    -- Статус начисления\неначисления
    accrual_status = 'статус начисления';
    if (k_stop_perc is not null and
        k_stop_perc <= rep_date) then
      accrual_status = 'статус неначисления';

    if (pr_code containing '.KARDHASAN.') then
      accrual_status = '';

    -- Код региона
    region_code = (case reg_stat_rep_region
                     when 'ысыкульская область' then '02'
                     when 'жалалабадская область' then '03'
                     when 'нарынская область' then '04'
                     when 'баткенская область' then '05'
                     when 'ошская область' then '06'
                     when 'г.ош' then '06'
                     when 'таласская область' then '07'
                     when 'чуйская область' then '08'
                     when 'г.бишкек' then '08'
                   end);

    zad_perc_delinq_kc = 0;
    zad_os_delinq_kc = 0;

    select sum(iif(sh_date < :rep_date, sh_calc_perc, 0)), sum(iif(sh_date < :rep_date, zad_os, 0)), sum(sh_calc_perc)
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    into :zad_perc_delinq, :zad_os_delinq, :sh_calc_perc;

    if (pr_code containing '.IPF.') then
      percent_classif = (iif(delinq_perc >= 90, 'Потери', 'Стандартные'));
    else
      percent_classif = (iif(accrual_status = 'Статус начисления', 'Стандартные', 'Потери'));

    if (pr_code containing '.KARDHASAN.') then
      percent_classif = '';

    connection_type = '';
    cr_rate = null;

    select cm_default_currency
    from company cm
    into def_curr;

    select cr_rate
    from opercurr
    where :rep_date between cr_date and cr_date_end and
          cr_from_cu_id = 417
    into :cr_rate;

    if (cr_rate is null) then
      cr_rate = 1;

    if (rep_currency <> def_curr) then
    begin
      if (k_index_curr_rate > cr_rate) then
      begin
        llc_sum_percl = llc_sum_percl * k_index_curr_rate;
        amount_approved = amount_approved * k_index_curr_rate;
        lkc_balance = lkc_balance * k_index_curr_rate;
        sh_calc_perc_cl = sh_calc_perc_cl * k_index_curr_rate;
      end
      else
      begin
        llc_sum_percl = llc_sum_percl * cr_rate;
        amount_approved = amount_approved * cr_rate;
        lkc_balance = lkc_balance * cr_rate;
        sh_calc_perc_cl = sh_calc_perc_cl * cr_rate;
      end
    end
    select cl.cu_symbol
    from currlist cl
    where cl.cu_id = :def_curr
    into :currency;

    for select c.c_id, c.c_name_fio, c.c_inn, kc.kc_amount_approved,
               iif(c_type = 1, 'юр. лицо', 'физ. лицо')
        from knclient kc
        inner join clients c on c.c_id = kc.kc_from_c_id
        where kc.kc_from_k_id = :k_id and
              kc.kc_amount_approved > 0
        into :c_id, :c_name, :client_inn, :amount_approved, :c_type_str
    do
    begin
      parallelism_sign = 'непараллельный';
      if (exists(select k.ks_from_k_id
                 from kontstate k
                 inner join knclient kc on kc.kc_from_k_id = k.ks_from_k_id
                 where kc.kc_from_c_id = :c_id and
                       k.ks_ol_is_delete = 0 and
                       k.ks_state = 1 and
                       k.ks_from_k_id <> :k_id and
                       kc.kc_amount_approved > 0 and
                       :rep_date between k.ks_date_begin and k.ks_date_end)) then
        parallelism_sign = 'параллельный';

      select bt_name
      from knclient kc
      inner join bustype bt on kc.kc_business_type = bt.bt_id
      where kc_from_c_id = :c_id and
            kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into bt_name;
      c_id_txt = cast(c_id as d_str_20);
      loan_purpose_code = (case bt_name
                             when 'производство' then '01'
                             when 'сельское хозяйство' then '02'
                             when 'животноводство' then '02'
                             when 'торговля' then '05'
                             when 'другое' then '08'
                             when 'физические лица зарплата' then '08'
                             when 'услуги' then '11'
                           end);

      nac_zaos = round(0, 0);

      for select lm_id
          from llc_main
          where lm_is_signed = 1 and
                lm_date_status between :rep_date - 4 and :rep_date + 4 and
                lm_from_reg_id = :reg_id
          into :lm_id
      do
      begin
        lkc_balance = 0;
        lkc_llc_sum = 0;
        select lkc.lkc_balance, cs.cs_name as lkc_status_str,
               cast(lkc.lkc_balance as double precision) * lkc.lkc_percent / 100 as lkc_llc_sum
        from llc_kont_cli lkc
        inner join classific cs on cs.cs_id = lkc_status
        where lkc.lkc_from_lm_id = :lm_id and
              lkc.lkc_from_k_id = :k_id and
              lkc.lkc_from_c_id = :c_id
        into :lkc_balance, :lkc_status_str, :lkc_llc_sum;

        kc_amount_approved_fact = 0;
        k = 1;

        select sum(iif(ol_date >= :rep_date or ol_date is null, kc_amount_approved, 0))
        from knclient kncl
        left join operlist ol on ol.ol_id = kncl.kc_closed_ol_id and
              ol.is_delete = 0
        where kncl.kc_from_k_id = :k_id
        into :amount_approved_fact;

        select sum(iif(ol_date >= :rep_date or ol_date is null, kc_amount_approved, 0))
        from knclient kncl
        left join operlist ol on ol.ol_id = kncl.kc_closed_ol_id and
              ol.is_delete = 0
        where kncl.kc_from_k_id = :k_id and
              kncl.kc_from_c_id = :c_id
        into :kc_amount_approved_fact;
        if (amount_approved_fact <> 0) then
          k = kc_amount_approved_fact / amount_approved_fact;
        else
          k = 0;

        llc_sum_perc = 0;
        sh_calc_perc_cl = coalesce(sh_calc_perc * k, 0);
        zad_os_delinq_kc = coalesce(zad_os_delinq * k, 0);
        zad_perc_delinq_kc = coalesce(zad_perc_delinq * k, 0);
        llc_sum_perc = coalesce(llc_sum_percl * k, 0);
      end

      suspend;
    end
  end
end
