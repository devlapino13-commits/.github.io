CREATE PROCEDURE HP_INS_FACERECOGLIST(
    C_ID D_ID,
    IS_PHOTO_DISB D_SMALLINT = 0)
AS
declare variable eil_id d_id;
declare variable grs_photo_id d_id;
declare variable c_photo_id d_id;
declare variable reg_id d_id;
declare variable photo_disb d_id;
declare variable db_id d_id;
begin
  db_id = f_get_db_id();
  if (db_id in (39801, 41701)) then
  begin
    select eil.eil_id
    from extinfolist eil
    where eil.eil_is_json = 1 and
          eil.eil_status = 2 and
          eil.eil_from_eic_id = 2 and -- tunduk
          eil.eil_from_c_id = :c_id
    order by eil.eil_id desc
    rows 1
    into eil_id;

    select eib.eib_from_bls_id
    from extinfolistbls eib
    where eib.eib_kind = f_enum_id('extinfo_bls_kind', 'tunduk_client_photo') and
          eib.eib_from_eil_id = :eil_id
    into grs_photo_id;

    select c.c_blob_photo_id, c.c_from_reg_id
    from clients c
    where c.c_id = :c_id
    into c_photo_id, reg_id;

    if (:is_photo_disb = 1) then
    begin
      select cds.cds_doc_blob_1
      from clientdocs cds
      where cds.cds_from_c_id = :c_id and
            cds.cds_from_pk_id = 312
      order by cds.cds_id desc
      rows 1
      into photo_disb;
    end
  end
  if (db_id = 94401) then
  begin
    select eil.eil_id
    from extinfolist eil
    where eil.eil_object_kind = 0 and
          eil.eil_is_json = 1 and
          eil.eil_status = 2 and
          eil.eil_from_eic_id = 4 and -- tunduk
          eil.eil_object_id = :c_id
    order by eil.eil_id desc
    rows 1
    into eil_id;

    select eib.eib_from_bls_id
    from extinfolistbls eib
    where eib.eib_kind = f_enum_id('extinfo_bls_kind', 'asan_client_photo') and
          eib.eib_from_eil_id = :eil_id
    into grs_photo_id;

    select c.c_blob_photo_id, c.c_from_reg_id
    from clients c
    where c.c_id = :c_id
    into c_photo_id, reg_id;
  end
  insert into facerecoglist (frl_from_c_id, frl_photo_bls_id, frl_doc_bls_id, frl_state)
  values (:c_id, iif(:is_photo_disb = 1 and coalesce(:photo_disb, 0) <> 0, :photo_disb, :c_photo_id), :grs_photo_id, 0);

  insert into actionlist (acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                          acl_respons_c_id, acl_from_un_id, acl_dont_show_in_order, acl_quota_used)
  values (current_date, :reg_id, 0, :c_id, f_get_rd_id('ActionGroup.Photo_Verify'), 0, null, 1000000000000, 0, 0);

end
