CREATE PROCEDURE HP_CHECK_EDF(
    EDL_OBJECT_KIND D_INTEGER,
    EDL_OBJECT_ID D_ID,
    EDF_ADD_CODE D_STR_CODE,
    EDOC_KIND D_STR_CODE)
RETURNS (
    NEED_TO_CONTINUE D_SMALLINT)
AS
declare variable DB_ID D_INTEGER;
declare variable PR_CODE D_STR_100;
declare variable OPER_SIGN D_SMALLINT;
begin
  --Процедура нужна для сравнения продукта и edoc_frx
  --По некоторым продуктам должны создаваться шаблоны в одном пакете с другим шаблоном
  --Пример: По продукту Иджара товары вместе с кредитным договором должен создаваться шаблон акта приёма передачи,
  --        но если продукт не Иджара товары то шаблон акта приёма передачи не должен создаваться

  --need_to_continue = 0 - указанный edoc_frx создаётся
  --need_to_continue = 1 - указанный edoc_frx не создается
  need_to_continue = 0;

  db_id = f_get_db_id();

  -- RFCKG-4753
  if (db_id = 41701 and edoc_kind = 'loan.agreement') then
  begin
    if (edf_add_code = 'income_statement') then
    begin
      if (((select k.k_by_rating
            from kontract k
            where k.k_id = :edl_object_id) <> 0)) then
        need_to_continue = 1;
    
      if (not exists(select 1
                     from kontract kt
                     inner join miniappclient mac on mac.mac_from_ma_id = kt.k_from_ma_app_id
                     where kt.k_id = :edl_object_id
                       and mac.mac_loan_target = f_get_rd_id('Loan_Target.Consumer'))) then
        need_to_continue = 1;
    end
  end

  if (db_id in (39801, 41701)) then
  begin
    if (edoc_kind = 'loan.agreement') then
    begin
      select pr.pr_code
      from kontract kt
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      where kt.k_id = :edl_object_id
      into pr_code;

      if (pr_code not containing '.idjara_goods.' and
          edf_add_code = 'idjara_goods.client_act') then
      begin
        need_to_continue = 1;
      end

      if (not(pr_code containing '.online_loan.' and pr_code containing '.elcard_disb.') and
          edf_add_code = 'online_loan.elcard_order') then --Поручение по переводу денег на стороннюю карту вида элькарт
      begin
        need_to_continue = 1;
      end
    end
  end

  if (edoc_kind = 'opercash.book') then
  begin
    if (edf_add_code = 'plus') then
      oper_sign = 1;
    else
    if (edf_add_code = 'minus') then
      oper_sign = -1;
    else
    if (edf_add_code = 'closeday') then
      oper_sign = 0;

    if (oper_sign is not null) then
    begin
      if (not exists(select * from opercash
                     where opc_id = :edl_object_id and
                           opc_sign = :oper_sign)) then
      begin
        if (not exists(select * from opercash
                       where opc_parent_id = :edl_object_id and
                             opc_sign = :oper_sign)) then
          need_to_continue = 1;
      end
    end

    if (edf_add_code = 'akt') then
    begin
      if (not exists(select * from opercash
                     where opc_id = :edl_object_id and
                           opc_from_cil_id is not null)) then
      begin
        if (not exists(select * from opercash
                       where opc_parent_id = :edl_object_id and
                             opc_from_cil_id is not null)) then
          need_to_continue = 1;
      end
    end
  end

  if (edoc_kind = 'asp.agreement') then
  begin
    if (edf_add_code = 'perstemp.change' and edl_object_kind = 0) then
    begin
      if (not exists(select * from perstemp
                     where pte_from_c_id = :edl_object_id and
                           pte_edit_kind = 1)) then
        need_to_continue = 1;
    end else
    if (edf_add_code = 'perstemp.fix' and edl_object_kind = 0) then
    begin
      if (not exists(select * from perstemp
                     where pte_from_c_id = :edl_object_id and
                           pte_edit_kind = 0)) then
        need_to_continue = 1;
    end
  end


  if (db_id = 41701) then
  begin
    if (edoc_kind = 'purchase_act.agreement' and edf_add_code = 'purchase_act.agreement.car.') then
    begin
        if (not exists(
                       select 1
                       from kontmatag kma
                       join kontract kn on kn.k_id = kma.kma_from_k_id
                       join product pr on pr.pr_id = kn.k_from_pr_id
                       where kma.kma_id = :edl_object_id
                       and pr.pr_code containing '.car.'))
        then
            need_to_continue = 1;
        else
            need_to_continue = 0;
    end

    if (edoc_kind = 'loan.application' and edf_add_code = 'necessary_conditions') then
    begin
      if (f_get_const('new_loan_conditions ') = '0') then
        need_to_continue = 1;

      else
      begin
        if (not exists
          (
            select *
            from miniappclient mac
            where mac.mac_from_ma_id = :edl_object_id
                and mac.mac_loan_target = f_get_rd_id('Loan_Target.Consumer')
          )
        ) then
          need_to_continue = 1;
            
        else
          need_to_continue = 0; -- Формируем бланк только для потребительских целей

      end
    end

    if (edoc_kind = 'loan.application' and edf_add_code = 'loan.application_npdl') then  -- RFCKG-4801
    begin
      if ( not exists
        (
          select 1
          from miniappclient mac
          inner join clients cl on cl.c_id = mac.mac_from_c_id
          where mac.mac_from_ma_id = :edl_object_id and
                mac.mac_amount > 0 and
                coalesce(cl.c_is_public, 0) = 1
        )
      ) then
          need_to_continue = 1;
        else
          need_to_continue = 0; -- Формируем бланк если клиент является НПДЛ

    end

    if (edoc_kind = 'loan.agreement' and edf_add_code = 'purchase_act.Spytnik_Inshur') then  -- RFCKG-4831 Данный документ формируется только для  .insur_kontrakt.
    begin
      select pr.pr_code
      from kontract kt
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      where kt.k_id = :edl_object_id
      into pr_code;

      if (pr_code containing '.insur_kontrakt.') then
        need_to_continue = 0;
      else
        need_to_continue = 1;
    end
  end

  if (db_id = 81001 and
      edoc_kind = 'loan.agreement') then
  begin
    if (edf_add_code = 'notify_loan_date') then
    begin
      if (coalesce((select freeze_time_h
                    from hp_get_freeze_data(:edl_object_id)), 0) = 0) then
        need_to_continue = 1;
    end
  end

  suspend;
end
