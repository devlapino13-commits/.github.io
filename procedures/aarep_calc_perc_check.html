CREATE PROCEDURE AAREP_CALC_PERC_CHECK
RETURNS (
    REC_NO D_INTEGER,
    K_KONTRACT_NO D_STR_CODE,
    K_DISB_DATE D_DATE,
    CHECK_DATE D_DATE,
    K_ID D_ID,
    DIFF_KIND D_STR_CODE,
    DIFF_PERC_SALDO D_CURRENCY,
    KCP_PERC_SALDO D_CURRENCY,
    NEW_PERC_SALDO D_CURRENCY,
    CLOSED_PERC_SALDO D_CURRENCY,
    K_STOP_PERC D_DATE,
    K_KREDLINE_END D_DATE,
    CLOSED_DATE D_DATE,
    DIFF_SIGN D_STR_CODE,
    CLOSE_STATE D_STR_CODE,
    CPF_STATUS_STR D_STR_CODE)
AS
declare variable total_diff_saldo d_currency;
declare variable cpf_status d_integer;
begin
  rec_no = 0;

  for select cpf_id, cpf_from_k_id, k_kontract_no, k_stop_perc, k_kredline_end,
             cpf_disb_date, cpf_close_date, cpf_check_date, cpf_orig_bal_perc,
             cpf_new_bal_perc, cpf_closed_bal_perc, cpf_status
  from temp_calcperc_fix, kontract
  where cpf_from_k_id = k_id
  into :rec_no, :k_id, :k_kontract_no, k_stop_perc, k_kredline_end,
       :k_disb_date, :closed_date, :check_date, :kcp_perc_saldo,
       :new_perc_saldo, :closed_perc_saldo, cpf_status
  do
  begin
    diff_perc_saldo = new_perc_saldo - kcp_perc_saldo;

    if (abs(diff_perc_saldo) > abs(closed_perc_saldo))
    then total_diff_saldo = diff_perc_saldo;
    else total_diff_saldo = closed_perc_saldo;

    cpf_status_str = '';
    cpf_status_str = case cpf_status
    when -5 then 'Отмена, без движений'
    when -4 then 'Отмена, разница отсутствует'
    when -3 then 'Отмена, сальдо больше 0'
    when -2 then 'Отмена, закрыт через списание'
    when -1 then 'Отмена, разница меньше 5 руб.'
    when 0 then 'Отмена, не обработан'
    else 'Обработан'
    end;

    if (cpf_status > 0) then
      cpf_status_str = cpf_status_str || ', схема ' || (cpf_status / 10) || '.' || (mod(cpf_status, 10));

    close_state = 'Активный';
    if (closed_date is not null) then
      close_state = 'Закрытый';

    diff_sign = 'Новый меньше';
    if (total_diff_saldo > 0) then
      diff_sign = 'Новый больше';

    if (abs(total_diff_saldo) <= 1) then
      diff_kind = 'A. <= 1';
    else
    if (abs(total_diff_saldo) <= 5) then
      diff_kind = 'B. <= 5';
    else
    if (abs(total_diff_saldo) <= 10) then
      diff_kind = 'C. <= 10';
    else
    if (abs(total_diff_saldo) <= 50) then
      diff_kind = 'D. <= 50';
    else
    if (abs(total_diff_saldo) <= 500) then
      diff_kind = 'E. <= 500';
    else
    if (abs(total_diff_saldo) <= 1000) then
      diff_kind = 'F. <= 1000';
    else
      diff_kind = 'G. > 1000';

    suspend;
  end
end
