CREATE PROCEDURE HP_E_HR_REBUKE_UPDATE(
    RB_ID D_INTEGER,
    RB_NUMBER D_STR_20,
    RB_DATE D_DATE,
    RB_FROM_REG_ID D_INTEGER,
    RB_FROM_C_ID D_ID,
    RB_TYPE D_INTEGER,
    RB_BODY D_STR_255,
    RB_REASON D_STR_255,
    RB_BONUS_CHANGE D_INTEGER,
    RB_PERIOD_START D_DATE,
    RB_PERIOD_END D_DATE,
    RB_BONUS_KOEF D_CURRENCY,
    RB_SUB_NUMBER D_INTEGER,
    RB_NUMBER_N D_INTEGER,
    RB_INITIATOR_ID D_ID,
    RB_NO_NUMBER D_SMALLINT,
    RB_SIGNER_ID D_INTEGER,
    RB_NOTE_DATE D_DATE,
    RS_PROTOCOL_NO D_INTEGER,
    RS_PROTOCOL_DATE D_DATE,
    RS_EXPLANATORY_DATE D_DATE,
    RS_FROM_RBK_ID D_INTEGER,
    RS_SCORES D_INTEGER,
    UN_ID D_ID)
AS
declare variable db_id d_integer;
begin 
  db_id = f_get_db_id();

  if (rs_from_rbk_id = 0 and rb_type = f_enum_id('rebuke_type', 'remark_light_2')) then
    rs_from_rbk_id = null; -- RFCKG-4466

  if (db_id = 41701 and rb_from_reg_id in(1090100, 1060100)) then
  begin
    -- в КР для офисов ГО и БО взыскания идут без учета баллов (rebuke_scores не используется), некоторые поля не заполняются (RFCKG-4362)

    if (coalesce(rb_id, 0) > 0 and exists(select * from rebuke where rb_id = :rb_id)) then
      update rebuke
      set rb_number = :rb_number,
          rb_date = :rb_date,
          rb_from_reg_id = :rb_from_reg_id,
          rb_from_c_id = :rb_from_c_id,
          rb_type = :rb_type,
          rb_body = :rb_body,
          rb_reason = :rb_reason,
          rb_bonus_change = :rb_bonus_change,
          rb_period_start = :rb_period_start,
          rb_period_end = :rb_period_end,
          rb_bonus_koef = :rb_bonus_koef,
          rb_sub_number = :rb_sub_number,
          rb_number_n = :rb_number_n,
          rb_initiator_id = :rb_initiator_id,
          rb_no_number = :rb_no_number,
          rb_signer_id = :rb_signer_id,
          rb_note_date = :rb_note_date,
          rb_explanatory_date = :rs_explanatory_date
      where rb_id = :rb_id;
    else
      insert into rebuke (rb_id, rb_number, rb_date, rb_from_reg_id, rb_from_c_id, rb_type, rb_body, rb_reason, rb_bonus_change, rb_period_start,
        rb_period_end, rb_bonus_koef, rb_sub_number, rb_number_n, rb_initiator_id, rb_no_number, rb_signer_id, rb_note_date, rb_explanatory_date, rb_from_un_id)
      values (:rb_id, :rb_number, :rb_date, :rb_from_reg_id, :rb_from_c_id, :rb_type, :rb_body, :rb_reason, :rb_bonus_change, :rb_period_start,
        :rb_period_end, :rb_bonus_koef, :rb_sub_number, :rb_number_n, :rb_initiator_id, :rb_no_number, :rb_signer_id, :rb_note_date, :rs_explanatory_date, :un_id);

    exit;
  end

  if (coalesce(rb_id, 0) > 0 and exists(select * from rebuke where rb_id = :rb_id)) then
  begin
    update rebuke
    set rb_number = :rb_number,
        rb_date = :rb_date,
        rb_from_reg_id = :rb_from_reg_id,
        rb_from_c_id = :rb_from_c_id,
        rb_type = :rb_type,
        rb_body = :rb_body,
        rb_reason = :rb_reason,
        rb_bonus_change = :rb_bonus_change,
        rb_period_start = :rb_period_start,
        rb_period_end = :rb_period_end,
        rb_bonus_koef = :rb_bonus_koef,
        rb_sub_number = :rb_sub_number,
        rb_number_n = :rb_number_n,
        rb_initiator_id = :rb_initiator_id,
        rb_no_number = :rb_no_number,
        rb_signer_id = :rb_signer_id,
        rb_note_date = :rb_note_date
    where rb_id = :rb_id;

    update rebuke_scores
    set rs_protocol_no = :rs_protocol_no,
        rs_protocol_date = :rs_protocol_date,
        rs_explanatory_date = :rs_explanatory_date
    where rs_from_rb_id = :rb_id;
  end
  else
  begin
    insert into rebuke (
        rb_id,
        rb_number,
        rb_date,
        rb_from_reg_id,
        rb_from_c_id,
        rb_type,
        rb_body,
        rb_reason,
        rb_bonus_change,
        rb_period_start,
        rb_period_end,
        rb_bonus_koef,
        rb_sub_number,
        rb_number_n,
        rb_initiator_id,
        rb_no_number,
        rb_signer_id,
        rb_note_date)
    values (
        :rb_id,
        :rb_number,
        :rb_date,
        :rb_from_reg_id,
        :rb_from_c_id,
        :rb_type,
        :rb_body,
        :rb_reason,
        :rb_bonus_change,
        :rb_period_start,
        :rb_period_end,
        :rb_bonus_koef,
        :rb_sub_number,
        :rb_number_n,
        :rb_initiator_id,
        :rb_no_number,
        :rb_signer_id,
        :rb_note_date)
    returning (rb_id)
    into rb_id;

    insert into rebuke_scores (rs_from_rb_id, rs_from_c_id, rs_from_rbk_id, rs_rebuke_type, rs_scores, rs_from_un_id,
                               rs_date, rs_protocol_no, rs_protocol_date, rs_explanatory_date)
    values (:rb_id, :rb_from_c_id, :rs_from_rbk_id, :rb_type, :rs_scores, :un_id,
            :rb_date, :rs_protocol_no, :rs_protocol_date, :rs_explanatory_date);
  end
end
