CREATE PROCEDURE FM_DOCRECOGREQUEST_LIST(
    UN_ID D_ID)
RETURNS (
    DRR_ID D_INTEGER,
    C_NAME_FIO D_STR_100,
    OBJ_NAME D_STR_100,
    DRR_STATUS D_INTEGER,
    DRR_STATUS_STR D_STR_CODE,
    DRR_CREATE_DATE D_DATE)
AS
begin
  for select drr_id, c_name_fio, iif(drm_name = '', dp_name, drm_name),
             drr_create_datetime, drr_status
      from hp_docrecogrequest_list(:un_id)
      order by iif(drr_status = 2, -3, drr_status) desc
      into drr_id, c_name_fio, obj_name, drr_create_date, drr_status
  do
  begin
    -- Тут все возможные статусы, на случай изменения условий выборки выше
    -- Используем emoji-символы, которые лучше выглядят в МП
    drr_status_str = case drr_status
                       when -2 then f_emoji('cross mark') -- Аннулирован
                       when -1 then f_emoji('counterclockwise arrows button') -- Переснять
                       when 0 then f_emoji('hourglass not done') -- Новый
                       when 1 then f_emoji('check mark button') -- Прикреплен
                       when 2 then f_emoji('page facing up') -- Готов
                     end;
    suspend;
  end
end
