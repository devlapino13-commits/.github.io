CREATE PROCEDURE HP_EXTINFO_TO_SEND_MONETA_DOC(
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    JSON_DATA D_STR_8100)
AS
declare variable operation_id d_str_255;
declare variable esm_id d_id;
begin
  -- для запроса платежного поручения. Монета.
  if (coalesce(object_kind, 0) <> f_objkind('kontract', 'id')) then
    exception e_default 'object_kind must be "kontract"';


  select esm_id, f_json_blob_get_value(replace(esa_message, 'answer:', ''), 'id', '')
  from extsysmain em, extsysanswer ea
  where esa_from_esm_id = esm_id
    and esm_from_k_id = :object_id
    and esm_project = 'moneta_disb'
    and esm_action in ('CardDisb', 'PhoneDisb', 'AccDisb')
    and esm_status = 1
    and esa_message starting with 'answer:'
    -- Убрал строку ниже из-за договоров: 101010061267900301, 101010068038700401
    -- Подозрение что допроводили вручную
    -- and f_json_blob_get_value(replace(ea.esa_message, 'answer:', ''), 'attrs.statusid', '') = 'SUCCEED'
  into esm_id, operation_id;

  if (operation_id is null) then
    exception e_default 'operation_id not found for k_id' || object_id;

  json_data =
    '{' ||
    f_json_make_pair('operation_id', operation_id , 'num', 1) ||
    f_json_make_pair('transaction_id', esm_id , 'num') ||
    '}';

  suspend;
end
