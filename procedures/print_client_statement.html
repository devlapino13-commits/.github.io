CREATE PROCEDURE PRINT_CLIENT_STATEMENT(
    RI_ID D_ID)
RETURNS (
    STA_LINE_NO INTEGER,
    STA_ACCOUNT_DT VARCHAR(20),
    STA_ACCOUNT_KT VARCHAR(20),
    STA_DATE DATE,
    STA_DOC_NO INTEGER,
    STA_MEMO VARCHAR(1024),
    STA_OB_DT DECIMAL(18,4),
    STA_OB_KT DECIMAL(18,4),
    STA_CURRENCY VARCHAR(3),
    STA_ACC_NAME VARCHAR(255),
    STA_SALDO_DT DECIMAL(18,4),
    STA_SALDO_KT DECIMAL(18,4),
    STA_BEGIN_DT DECIMAL(18,4),
    STA_BEGIN_KT DECIMAL(18,4))
AS
declare variable sta_ob_saldo decimal(18,4);
declare variable beg_saldo_kt decimal(18,4);
declare variable ob_detail varchar(1024);
declare variable doc_kind varchar(255);
declare variable ob_from_ol_id decimal(18,0);
declare variable ob_number integer;
declare variable ri_from_c_id d_id;
declare variable ri_from_reg_id d_integer;
declare variable ri_balance_id d_str_20;
declare variable ri_date_begin date;
declare variable ri_date_end date;
declare variable ri_account_no d_integer;
declare variable ri_saldo_begin_dt decimal(18,2);
declare variable ri_saldo_begin_kt decimal(18,2);
declare variable ri_saldo_end_dt decimal(18,2);
declare variable ri_saldo_end_kt decimal(18,2);
begin

  select ri_from_c_id, ri_from_reg_id, ri_date_begin, ri_date_end, ri_balance_id, ri_account_no, ri_saldo_begin_dt,
         ri_saldo_begin_kt, ri_saldo_end_dt, ri_saldo_end_kt
  from reconciliation_info ri
  where ri.ri_id = :ri_id
  into :ri_from_c_id, :ri_from_reg_id, :ri_date_begin, :ri_date_end, :ri_balance_id, :ri_account_no, :ri_saldo_begin_dt,
       :ri_saldo_begin_kt, :ri_saldo_end_dt, :ri_saldo_end_kt;

  for select sta_line_no, sta_account_dt, sta_account_kt, sta_date, sta_doc_no, sta_memo, sta_ob_dt, sta_ob_kt,
             sta_currency, sta_acc_name, sta_saldo_dt, sta_saldo_kt, sta_begin_dt, sta_begin_kt
      from aarep_client_statement(:ri_from_reg_id, :ri_date_begin, :ri_date_end, :ri_balance_id, :ri_account_no, :ri_from_c_id)
      into :sta_line_no, :sta_account_dt, :sta_account_kt, :sta_date, :sta_doc_no, :sta_memo, :sta_ob_dt, :sta_ob_kt,
           :sta_currency, :sta_acc_name, :sta_saldo_dt, :sta_saldo_kt, :sta_begin_dt, :sta_begin_kt
  do
  begin
    suspend;
  end
end
