CREATE PROCEDURE HP_GET_JUST_CLIENT_IN_LOAN(
    REP_DATE DATE,
    K_ID DECIMAL(18,0),
    K_DISB_DATE DATE)
RETURNS (
    RES_JUST_CLIENT_COUNT INTEGER)
AS
declare variable C_ID_TMP decimal(18,0);
declare variable CHET_KOL_END integer;
declare variable DD_CLOSE_TMP date;
begin
  res_just_client_count = 0;
  for select kc_from_c_id
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into :c_id_tmp
  do
  begin
    chet_kol_end = 1;

    for select kc_date_close_os
        from knclient
        where ((kc_date_disb between '01.01.1900' and (:k_disb_date - 1)) or
              (kc_date_disb = :k_disb_date and
              kc_from_k_id < :k_id)) and
              kc_date_close_os between :k_disb_date and '31.12.9999' and
              kc_from_c_id = :c_id_tmp
        into :dd_close_tmp
    do
    begin
      if (dd_close_tmp >= :rep_date) then
      begin
        if (dd_close_tmp = :k_disb_date) then
        begin
          chet_kol_end = 1;
        end
        else
        begin
          chet_kol_end = 0;
        end
      end
    end
    res_just_client_count = res_just_client_count + chet_kol_end;
  end
  if (res_just_client_count is null) then
    res_just_client_count = 0;
  suspend;
end
