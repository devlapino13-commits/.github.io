CREATE PROCEDURE FSROBOT_MAILING_HR_ORDERS_7(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable c_name_fio varchar(100);
declare variable reg_name varchar(100);
declare variable jl_name varchar(100);
declare variable ps_date date;
declare variable c_login varchar(100);
declare variable res_tmp_bode blob sub_type 1 segment size 80;
declare variable c_email varchar(100);
declare variable c_cell_no varchar(100);
declare variable c_id decimal(18,0);
declare variable kol integer;
declare variable hr_order_no varchar(20);
declare variable ps_date_end date;
declare variable date_format_str varchar(20);
declare variable date_end_format_str varchar(20);
declare variable reg_id integer;
declare variable db_id integer;
declare variable jk_code varchar(100);
declare variable txt_kind varchar(100);
begin
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID - пользователь

  :db_id = f_get_db_id();

  rep_params = (trim(rep_params));
  rep_email = (trim(rep_email));
  res_maillist = :rep_email;
  res_c_id = 0;

-- HR_Orders_7 - Трудовой отпуск
  res_body = '<b>Внимание !</b><br>';
  res_body = BLOB_APPEND(res_body, '<br>');

  res_tmp_bode = '';
  kol = 0;
  for select c_id, c_name_fio, hrv_date_begin, hrv_date_end,
             (select cont_value
              from hp_contdata_get_email_main(0, c_id)),
             (select cont_value
              from hp_contdata_get_cell_main(0, c_id)), ho_order_no, sawt_name
      from hr_voctravel, clients, hr_orders, sa_worktype
      where hrv_from_ho_id = :rep_params and
            hrv_status = 1 and
            hrv_from_c_id = c_id and
            hrv_from_ho_id = ho_id and
            ho_order_type = sawt_id
      into c_id, c_name_fio, ps_date, ps_date_end, c_email, c_cell_no, hr_order_no, txt_kind
  do
  begin
    if (c_email = '') then
      c_email = 'Отсутствует';
    if (c_cell_no = '') then
      c_cell_no = 'Отсутствует';

    reg_name = 'Не определен';
    jl_name = 'Не определен';

    date_format_str = right('0' || extract(day from ps_date), 2) || '.' || right('0' || extract(month from ps_date), 2) || '.' || extract(year from ps_date);
    date_end_format_str = right('0' || extract(day from ps_date_end), 2) || '.' || right('0' || extract(month from ps_date_end), 2) || '.' || extract(year from ps_date_end);

    select reg_code || ' ' || reg_name, jl_name, ps_from_reg_id, jk_system_code
    from hr_pstatus, region, joblist, jobkind
    where ps_date_begin <= :ps_date and
          ps_date_end >= :ps_date and
          ps_status = 1 and
          ps_from_c_id = :c_id and
          ps_from_reg_id = reg_unicode and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id
    into reg_name, jl_name, reg_id, jk_code;

    jk_code = coalesce(jk_code, '');
    reg_id = coalesce(reg_id, 0);

    if (db_id in (39801, 41701) and
        txt_kind in ('Перерыв на кормление ребенка', 'Материальная помощь')) then
      exit;

    if (not((db_id in (39801, 41701)) and
        (reg_id = 1090100 or jk_code = 'JobSystem.LO_Standart'))) then
    begin
      kol = kol + 1;
      c_login = '';
      select iif(un_is_active = 1, un_name || ', доступ актуальный', un_name || ', доступ закрыт')
      from hp_client_un_is_active(:c_id)
      into :c_login;
      if (c_login = '') then
        c_login = 'Отсутствует';

      res_tmp_bode = BLOB_APPEND(res_tmp_bode, '<br>ФИО: <B>' || c_name_fio || '</B><br>');
      res_tmp_bode = BLOB_APPEND(res_tmp_bode, 'ОП: <B>' || reg_name || '</B><br>');
      res_tmp_bode = BLOB_APPEND(res_tmp_bode, 'Должность: <B>' || jl_name || '</B><br>');
      res_tmp_bode = BLOB_APPEND(res_tmp_bode, 'Сотовый номер в БД: <B>' || c_cell_no || '</B><br>');
      res_tmp_bode = BLOB_APPEND(res_tmp_bode, 'Логин в FinStart: <B>' || c_login || '</B><br>');
      res_tmp_bode = BLOB_APPEND(res_tmp_bode, 'Email в БД: <B>' || c_email || '</B><br>');
      res_tmp_bode = BLOB_APPEND(res_tmp_bode, 'Период отпуска: <B>' || date_format_str || ' - ' || date_end_format_str || '</B>');
      res_tmp_bode = BLOB_APPEND(res_tmp_bode, '<br>');
    end
  end

  if (kol = 0) then
    exit;

  res_body = '<b>Внимание !</b><br>';
  res_body = BLOB_APPEND(res_body, '<br>');

  if (kol = 1) then
  begin
    res_subject = txt_kind || ', приказ № ' || coalesce(hr_order_no, 'б/н');
    res_body = BLOB_APPEND(res_body, '<b>Оформлен ' || :txt_kind || ' сотруднику:</b>');
  end
  else
  begin
    res_subject = txt_kind || ', приказ № ' || coalesce(hr_order_no, '(не указан)');
    res_body = BLOB_APPEND(res_body, '<b>Оформлен ' || :txt_kind || ' сотрудникам:</b>');
  end
  res_body = BLOB_APPEND(res_body, '<br>');
  res_body = BLOB_APPEND(res_body, res_tmp_bode);
  res_body = BLOB_APPEND(res_body, '<br>');
  if (user_fio <> '') then
  begin
    res_body = BLOB_APPEND(res_body, 'Автор изменений: ' || user_fio);
    res_body = BLOB_APPEND(res_body, '<br>');
    res_body = BLOB_APPEND(res_body, '<br>');
  end
  res_body = BLOB_APPEND(res_body, 'Всем службам произвести необходимые действия!');
  res_body = BLOB_APPEND(res_body, '<br>');

  suspend;

end
