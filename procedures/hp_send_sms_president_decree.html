CREATE PROCEDURE HP_SEND_SMS_PRESIDENT_DECREE(
    REP_KIND SMALLINT)
AS
declare variable lss_cell_no str_100;
declare variable lss_sms_text1 varchar(512);
declare variable lss_sms_text2 varchar(512);
begin
  for
  select lss_cell_no, lss_sms_text1, lss_sms_text2
  from listsmssend
  where lss_send_status <> 1
  and lss_cell_no <> ''
  and lss_kind = :rep_kind -- 5 июля нужно отправить по след. 1,3 и 4
--  and lss_id = 9047
  into lss_cell_no, lss_sms_text1, lss_sms_text2
  do
  begin
    if (lss_sms_text1 <> '') then execute procedure hp_send_message_staff(1,:lss_cell_no, :lss_sms_text1, '', 1000000000000);
    if (lss_sms_text2 <> '') then execute procedure hp_send_message_staff(1,:lss_cell_no, :lss_sms_text2, '', 1000000000000);
  end
end
