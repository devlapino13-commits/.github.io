CREATE PROCEDURE HP_CHANGE_GNI_INTERN(
    C_ID D_ID,
    UN_ID D_ID)
AS
declare variable rd_id d_integer;
declare variable code d_str_20;
declare variable cgc_id d_id;
declare variable c_sex d_smallint;
declare variable c_birth_date d_date;
begin
  select c.c_sex, c.c_birth_date
  from clients c
  where c.c_id = :c_id
  into :c_sex, :c_birth_date;

  if (((:c_birth_date < '01.01.1964') and
      (:c_sex = 1)) or ((:c_birth_date < '01.01.1969') and
      (:c_sex = 0))) then
    code = 'GNI.106-64-69';
  else
    code = 'GNI.4328';

  rd_id = f_get_rd_id(:code);
  cgc_id = -1;
  select t.cgc_id
  from client_gni_category t
  where t.cgc_from_c_id = :c_id and
        t.cgc_is_delete = 0 and
        current_date between t.cgc_date_begin and t.cgc_date_end
  into :cgc_id;
  if (cgc_id <> -1) then
    update client_gni_category
    set cgc_date_end = current_date - 1,
        cgc_from_un_id_update = :un_id
    where (cgc_id = :cgc_id);

  insert into client_gni_category (cgc_from_c_id, cgc_from_category, cgc_from_un_id_add, cgc_date_begin)
  values (:c_id, :rd_id, :un_id, current_date);
end
