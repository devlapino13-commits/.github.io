CREATE PROCEDURE HP_FINCERT_INCIDENTS_SEND(
    FHL_ID D_ID)
RETURNS (
    HEAD_SCHEMA_TYPE D_STR_20,
    HEAD_SCHEMA_VERSION D_STR_10,
    HEAD_SOURCE_ID D_ID,
    HEAD_VERSION D_INTEGER,
    I_FIXATION_DATE D_DATE,
    I_ASSISTANCE D_STR_10,
    I_SENT_TO_GOS_SOPKA D_BOOLEAN,
    I_DESCRIPTION D_STR_64,
    I_VECTOR_CODE D_STR_10,
    I_TYPE_OF_ATTACK D_STR_10,
    FD_FEDERAL_DISTRICT_ID D_STR_64,
    FD_ID D_STR_255,
    FD_NAME D_STR_255,
    SOF_CODE D_INTEGER,
    SOF_FIAS_ID D_STR_64,
    SOF_FEDERAL_DISTRICT_NAME D_STR_100,
    SOF_ID D_STR_100,
    SOF_NAME D_STR_255,
    LOCALITY D_STR_255,
    REG_DEPARTMENT D_STR_64,
    REG_TECHNICAL_DEVICE D_STR_64,
    FINAL_REP_CLOSE_DATE_AT D_DATE,
    DMG_TOTAL_DAMAGE D_STR_64,
    IMP_ATTACHMENT D_STR_64,
    TRANS_SOURCE_ID D_STR_255,
    TRANS_DETECTED_BY D_STR_64,
    PAYER_ORG_TYPE D_STR_20,
    PAYER_HASH D_STR_255,
    PAYER_INN D_STR_20,
    TR_PAYER_TYPE D_STR_20,
    TR_PAYER_NUMBER D_STR_100,
    TR_PAYER_COUNTRY_CODE D_STR_10,
    TR_PAYER_SUM D_CURRENCY,
    TR_PAYER_CURRENCY D_STR_10,
    TR_PAYER_DATETIME_AT D_DATETIME,
    TR_PAYEE_TYPE D_STR_20,
    TR_PAYEE_ORG_TYPE D_STR_10,
    TR_PAYEE_NUMBER D_STR_100,
    TR_PAYEE_COUNTRY_CODE D_STR_10,
    TR_PAYEE_FAST_PAY_MID D_STR_100,
    TR_PAYEE_FAST_PAY_TRID D_STR_100,
    TR_PAYER_BANK_ACCOUNT_NUMBER D_STR_100,
    LAW_REQUEST D_STR_CODE,
    LAW_NUMBER D_STR_64,
    LAW_NUMBER_TICKET D_STR_64,
    LAW_DATETIME_AT D_STR_64,
    ST_TYPE D_STR_10,
    ST_DESCRIPTION D_STR_20,
    CSF_TYPE_OF_INCIDENT D_STR_10,
    CSF_EVENTS D_STR_10,
    CSF_TA_MFUNDS D_STR_10,
    E_COMMERCE_PAYMENT_SYS D_STR_100)
AS
declare variable C_ID D_ID;
declare variable OBJ_KIND D_INTEGER;
declare variable OBJ_ID D_ID;
declare variable MA_ID D_ID;
declare variable MAC_ID D_ID;
declare variable K_ID D_ID;
declare variable FHL_INCIDENT_DATE_TIME D_DATETIME;
declare variable MA_OFFICE_ID D_ID;
declare variable FHL_RES_INCIDENT D_STR_255;
declare variable INN D_STR_20;
declare variable C_PASS_NO D_STR_64;
declare variable CONTRACT_ISSUED D_SMALLINT;
declare variable K_AMOUNT_APPROVED D_CURRENCY;
declare variable CONTACT_NO D_STR_100;
declare variable FHL_LAW_REQ_REQUEST D_STR_CODE;
declare variable FHL_LAW_REQ_NUMBER D_STR_64;
declare variable FHL_LAW_REQ_TICKET D_STR_64;
declare variable FHL_LAW_REQ_DATETIME_AT D_DATETIME;
declare variable CLT_ID D_ID;
declare variable FAST_PAY_MID_TMP D_STR_100;
declare variable JSON_TEXT_ESM D_BLOB_TEXT;
declare variable JSON_TEXT_ESA D_BLOB_TEXT;
declare variable TARGETTRANSACTION D_STR_100;
begin
    if (:fhl_id is null) then
      exception e_message('Input parameter: FHL_ID is missing');

    select fhl.fhl_from_c_id, fhl.fhl_obj_kind, fhl.fhl_obj_id, fhl.fhl_res_incident, fhl.fhl_incident_date_time,
               fhl.fhl_law_req_request, fhl.fhl_law_req_number, fhl.fhl_law_req_ticket, fhl.fhl_law_req_datetime_at
    from fincert_history_log fhl
    where fhl.fhl_id = :fhl_id
    into :c_id, obj_kind, :obj_id, :fhl_res_incident, :fhl_incident_date_time,
         :fhl_law_req_request, :fhl_law_req_number, :fhl_law_req_ticket, :fhl_law_req_datetime_at;

    if (row_count = 0) then
      exit;

    contract_issued = 0;
    contact_no = '';

    if (:obj_kind = f_objkind('kontract', 'id')) then
    begin
      if (exists(select 1
                 from kontstate ks
                 where ks.ks_from_k_id = :obj_id and
                       ks.ks_ol_is_delete = 0 and
                       ks.ks_state in (1, 2) and
                       current_date between ks.ks_date_begin and ks.ks_date_end)) then
        contract_issued = 1;

      select kn.k_from_ma_app_id, kn.k_amount_approved, kn.k_id
      from kontract kn
      where kn.k_id = :obj_id
      into :ma_id, :k_amount_approved, :k_id;
    end
    else
      ma_id = :obj_id;

    -- блок header
    head_schema_type = 'incident';
    head_schema_version = '3.0.14';
    head_source_id = null;
    head_version = 1;

    -- блок incident
    i_fixation_date = :fhl_incident_date_time;
    i_assistance = 'NND';
    i_sent_to_gos_sopka = false;
    i_description = 'Операция без согласия';
    i_vector_code = 'EXT';
    i_type_of_attack = 'other';

    select ma.ma_office_id, pl.pl_name, mac.mac_id
    from miniapp ma
    inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id and
          mac.mac_from_c_id = :c_id
    inner join region reg on reg.reg_unicode = ma.ma_office_id
    inner join places pl on pl.pl_id = reg.reg_from_pl_id
    where ma.ma_id = :ma_id and
          ma.ma_date_end = '31.12.9999'
    rows 1
    into :ma_office_id, :locality, :mac_id;

    select fl.fl_district_code,
           fl.fl_district_id,
           fl.fl_district_name,
           sof.sof_code,
           sof.sof_fias_id,
           sof.sof_federal_district_name,
           sof.sof_sof_id,
           sof.sof_name
    from region_location_link rll
    inner join subject_of_federation sof on sof.sof_id = rll.rll_subject_of_federation
    inner join federation_location fl on fl.fl_id = sof.sof_from_fl_id
    where rll.rll_reg_id = :ma_office_id
    into :fd_federal_district_id,  -- блок federalDistrict --
         :fd_id,
         :fd_name,
         :sof_code,  -- блок subjectOfFederation --
         :sof_fias_id,
         :sof_federal_district_name,
         :sof_id,
         :sof_name;

    -- блок Registration --
    reg_department = 'Клиент';
    reg_technical_device = 'Техническое средство клиента';

    -- блок serviceType --
    st_type = 'OTH';
    st_description = 'Иная система';

    -- блок classification --
    csf_type_of_incident = 'MTR';
    csf_events = 'OTH';
    csf_ta_mfunds = 'OTH';

    -- блок attachments (просто ключ "attachments" с пустым массивом)

    -- блок finalReport
    final_rep_close_date_at = null;

    -- блок damage
    dmg_total_damage = null;

    -- блок impacts
    imp_attachment = null;

    -- блок transfers
    trans_source_id = null; -- Реализацией займется Антон со своей стороны
    trans_detected_by = iif(fhl_res_incident containing 'Совпадение со списком Финцерт', 'AFR', 'CLNT');

    -- блок payer
    select cl.c_inn, cl.c_pass_no
    from clients cl
    where cl.c_id = :c_id
    into :inn, :c_pass_no;

    payer_org_type = 'individual';
    payer_hash = f_fincert_passport_hash(:c_pass_no);
    payer_inn = iif(contract_issued = 1, :inn, null);

    contact_no = f_cont(:c_id, 'cell', 'ok', 'main');

    if (contract_issued = 0) then
    begin
      clt_id = f_cont_id(:contact_no, 'add');

      select fb.fpb_code
      from fps_application_banks fab
      inner join fps_client_banks fcb on fab.fab_from_fcb_id = fcb.fcb_id and
            fcb.fcb_from_c_id = :c_id and
            fcb.fcb_from_clt_id = :clt_id
      inner join fps_banks fb on fcb.fcb_from_fpb_id = fb.fpb_id
      where fab.fab_from_mac_id = :mac_id
      into :fast_pay_mid_tmp;
    end
    else
    begin
      select esm.esm_full_source, esa.esa_message
      from extsysmain esm
      inner join extsysanswer esa on esa.esa_from_esm_id = esm.esm_id
      where esm.esm_from_k_id = :k_id and
            esa.esa_message containing 'answer:'
      into json_text_esm, :json_text_esa;

      json_text_esa = substring(:json_text_esa from position('{' in json_text_esa));

      fast_pay_mid_tmp = f_json_blob_get_value(:json_text_esm, 'bankId', '');
      targettransaction = f_json_blob_get_value(:json_text_esa, 'attrs.targettransaction', '');

      select ol.ol_datetime
      from operlist ol
      inner join operkred ok on ok.ok_from_ol_id = ol.ol_id
      where ok.ok_from_k_id = :k_id and
            ok.ok_type = 8 and
            ol.is_delete = 0
      into :fhl_incident_date_time;
    end

    -- блок transferInfo
    if (contract_issued = 0) then
    begin
      tr_payer_type = 'eWallet';
      tr_payer_number = '7728744752';
      e_commerce_payment_sys = '[' || :ma_id || ']:' || '[б/н]:' || '[:' || f_datetostr(fhl_incident_date_time) || ']';
      tr_payer_sum = 0;
      tr_payer_currency = '643';
      tr_payer_datetime_at = :fhl_incident_date_time;

      tr_payee_type = 'phoneNumber';
      tr_payee_number = '+' || :contact_no;
      tr_payee_country_code = 'RU';
    end
    else
    begin
      tr_payer_type = 'fastPay';
      tr_payer_number = '+' || :contact_no;
      tr_payer_bank_account_number = '40701810100065562035'; -- счет НКО «Монета»
      tr_payer_country_code = 'RU';
      tr_payer_sum = :k_amount_approved;
      tr_payer_currency = '643';
      tr_payer_datetime_at = :fhl_incident_date_time;

      tr_payee_org_type = 'individual';
      tr_payee_type = 'fastPay';
      tr_payee_number = '+' || :contact_no;
      tr_payee_country_code = 'RU';
      tr_payee_fast_pay_mid = :fast_pay_mid_tmp;
      tr_payee_fast_pay_trid = :targettransaction;
    end 

    if (fhl_law_req_request is null) then
      fhl_law_req_request = 'UNKW';

    -- блок lawEnforcementRequest
    law_request = upper(:fhl_law_req_request);
    law_number = :fhl_law_req_number;
    law_number_ticket = :fhl_law_req_ticket;
    law_datetime_at = :fhl_law_req_datetime_at;
    /*
    update fincert_history_log
    set fhl_status = f_enum_id('fincert_status', 'sent')
    where (fhl_id = :fhl_id);
    */
    suspend;
end
