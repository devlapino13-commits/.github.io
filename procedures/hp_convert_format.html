CREATE PROCEDURE HP_CONVERT_FORMAT(
    ORIG_TEXT VARCHAR(255),
    CONVERT_KIND VARCHAR(10))
RETURNS (
    RESULT_TXT VARCHAR(255))
AS
declare variable i integer;
declare variable w varchar(255);
declare variable len integer;
declare variable sym_old varchar(1);
declare variable sym_new varchar(10);
begin
-- конвертирует первоначальный текст на нужный (rus/lat)
  result_txt = '';
  orig_text = coalesce(orig_text, '');
  if (orig_text = '') then
  begin
    suspend;
    exit;
  end
  w = orig_text;
  if (convert_kind = 'rus') then
  begin
    w = replace(w, 'ya', 'я');
    w = replace(w, 'yu', 'ю');
    w = replace(w, 'yo', 'ё');
    w = replace(w, 'zh', 'ж');
    w = replace(w, 'sh', 'ш');
    w = replace(w, 'ch', 'ч');
    w = replace(w, 'ck', 'к');
    w = replace(w, 'kh', 'х');
    w = replace(w, 'ph', 'ф');

    len = char_length(w);
    result_txt = '';
    i = 1;
    while (i <= len) do
    begin
      sym_old = substring(w from i for 1);
      sym_new = (case sym_old
                   when 'Ң' then 'нг'
                   when 'ң' then 'нг'
                   when 'a' then 'а'
                   when 'b' then 'б'
                   when 'c' then 'ж'
                   when 'd' then 'д'
                   when 'e' then 'е'
                   when 'f' then 'ф'
                   when 'g' then 'г'
                   when 'h' then 'х'
                   when 'i' then 'и'
                   when 'j' then 'ж'
                   when 'k' then 'к'
                   when 'l' then 'л'
                   when 'm' then 'м'
                   when 'n' then 'н'
                   when 'o' then 'о'
                   when 'p' then 'п'
                   when 'q' then 'к'
                   when 'r' then 'р'
                   when 's' then 'с'
                   when 't' then 'т'
                   when 'u' then 'у'
                   when 'v' then 'в'
                   when 'w' then 'в'
                   when 'x' then 'х'
                   when 'y' then 'и'
                   when 'z' then 'з'
                   when 'Ç' then 'ч'
                   when 'ç' then 'ч'
                   when 'ҫ' then 'ч'
                   when 'Ə' then 'э'
                   when 'ә' then 'э'
                   when 'Ğ' then 'г'
                   when 'ğ' then 'г'
                   when 'İ' then 'и'
                   when 'i' then 'и'
                   when 'ı' then 'и'
                   when 'Ö' then 'о'
                   when 'ö' then 'о'
                   when 'Ş' then 'ш'
                   when 'ş' then 'ш'
                   when 'Ü' then 'у'
                   when 'ü' then 'у'
                   when 'Ө' then 'о'
                   when '�©' then 'о'
                   when 'Ү' then 'у'
                   when 'ү' then 'у'
                   else :sym_old
                 end);

      result_txt = result_txt || sym_new;
      i = i + 1;
    end
  end

  if (convert_kind = 'lat') then
  begin
    len = char_length(w);
    result_txt = '';
    i = 1;
    while (i <= len) do
    begin
      sym_old = substring(w from i for 1);
      sym_new = (case sym_old
                   when 'а' then 'a'
                   when 'б' then 'b'
                   when 'в' then 'v'
                   when 'г' then 'g'
                   when 'д' then 'd'
                   when 'е' then 'e'
                   when 'ё' then 'e'
                   when 'ж' then 'zh'
                   when 'з' then 'z'
                   when 'и' then 'i'
                   when 'й' then 'y'
                   when 'к' then 'k'
                   when 'л' then 'l'
                   when 'м' then 'm'
                   when 'н' then 'n'
                   when 'о' then 'o'
                   when 'п' then 'p'
                   when 'р' then 'r'
                   when 'с' then 's'
                   when 'т' then 't'
                   when 'у' then 'u'
                   when 'ф' then 'ph'
                   when 'х' then 'kh'
                   when 'ц' then 'tch'
                   when 'ч' then 'ch'
                   when 'ш' then 'sh'
                   when 'щ' then 'sh'
                   when 'ъ' then ''
                   when 'ь' then ''
                   when 'ы' then 'i'
                   when 'э' then 'e'
                   when 'ю' then 'yu'
                   when 'я' then 'ya'
                   else :sym_old
                 end);

      result_txt = result_txt || sym_new;
      i = i + 1;
    end
  end

  suspend;
end
