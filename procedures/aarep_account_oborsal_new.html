CREATE PROCEDURE AAREP_ACCOUNT_OBORSAL_NEW(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_WITH_ACCOUNT INTEGER,
    REPORT_WITH_CLIENT INTEGER,
    REPORT_WITH_SUBKONTO INTEGER,
    REPORT_WITH_CURRENCY INTEGER)
RETURNS (
    OB_BALANCE VARCHAR(20),
    OB_BALANCE_NAME VARCHAR(255),
    OB_ACCOUNT VARCHAR(20),
    OB_CURRENCY VARCHAR(3),
    OB_CLIENT_ID DECIMAL(18,0),
    OB_SUBKONTO_ID INTEGER,
    OB_OBOROT_DT DECIMAL(18,4),
    OB_OBOROT_KT DECIMAL(18,4),
    OB_OBOROT_DT_EKV DECIMAL(18,4),
    OB_OBOROT_KT_EKV DECIMAL(18,4),
    OB_ACCOUNT_NAME VARCHAR(255),
    OB_END DECIMAL(18,4),
    OB_END_EKV DECIMAL(18,4),
    OB_BEGIN DECIMAL(18,4),
    OB_BEGIN_EKV DECIMAL(18,4),
    OB_CLIENT_NAME VARCHAR(100),
    OB_SUBKONTO_NAME VARCHAR(255),
    OB_GROUP_NAME VARCHAR(255),
    OB_CLASS_NAME VARCHAR(255),
    ACC_KIND_NAME VARCHAR(255))
AS
declare variable cl_prior decimal(18,0);
declare variable bg_kind integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable class_id varchar(20);
declare variable class_id_prior varchar(20);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  cl_prior = 0;
  report_with_subkonto = 0;
  ob_subkonto_id = 0;
  ob_subkonto_name = 'Не указан';
  if (report_with_client = 0) then
  begin
    ob_client_id = null;
    ob_client_name = 'Без контрагента';
    class_id_prior = '';

    for select account, account_name, acc_balance_no, acc_balance, acc_group, acc_class, curr_code, acc_kind_name,
               acc_kind, sum(end_dt_curr - end_kt_curr), sum(ob_dt_curr), sum(ob_kt_curr), sum(end_dt - end_kt),
               sum(ob_dt), sum(ob_kt)
        from aarep_account_oborsaldo_temp(:report_region_id, :report_date_begin, :report_date_end, 0, '000')
        group by account, account_name, acc_balance_no, acc_balance, acc_group, acc_class, curr_code, acc_kind_name, acc_kind_name, acc_kind
        order by curr_code, acc_balance_no, account
        into :ob_account, :ob_account_name, :ob_balance, :ob_balance_name, :ob_group_name, :class_id, :ob_currency,
             :acc_kind_name, :bg_kind, :ob_end, :ob_oborot_dt, :ob_oborot_kt, :ob_end_ekv, :ob_oborot_dt_ekv,
             :ob_oborot_kt_ekv
    do
    begin
      if (class_id <> class_id_prior) then
      begin
        ob_class_name = '';
        select bg_name
        from balgroup
        where bg_id = :class_id
        into :ob_class_name;
      end
      ob_begin = ob_end - (ob_oborot_dt - ob_oborot_kt);
      ob_begin_ekv = ob_end_ekv - (ob_oborot_dt_ekv - ob_oborot_kt_ekv);

      if ((ob_begin <> 0) or (ob_oborot_dt <> 0) or (ob_oborot_kt <> 0) or (ob_end <> 0) or (ob_begin_ekv <> 0) or (ob_oborot_dt_ekv <> 0) or (ob_oborot_kt_ekv <> 0) or (ob_end_ekv <> 0)) then
      begin
        acc_kind_name = 'Не указан';
        if (bg_kind = 1) then
        begin
          acc_kind_name = '1 Активы';
        end
        if ((bg_kind = 2) or (bg_kind = 3)) then
        begin
          acc_kind_name = '2 Обязательства и капитал';
        end
        if ((bg_kind = 4) or (bg_kind = 5)) then
        begin
          acc_kind_name = '3 Доходы и расходы';
        end
        suspend;
      end
    end
  end
  else
  begin
    for select ob_balance, ob_balance_name, ob_account, ob_currency, ob_client_id, sum(ob_end_dt - ob_end_kt),
               sum(ob_oborot_dt), sum(ob_oborot_kt), sum(ob_end_dt_ekv - ob_end_kt_ekv), sum(ob_oborot_dt_ekv),
               sum(ob_oborot_kt_ekv), ob_account_name, b2.bg_id || ' ' || b2.bg_name, b3.bg_name, b1.bg_kind
        from aarep_account_oborsal_temp(:report_region_id, :report_date_begin, :report_date_end, :report_with_account, :report_with_client, :report_with_subkonto, :report_with_currency), balgroup b1, balgroup b2, balgroup b3
        where b1.bg_id = ob_balance and
              b2.bg_id = b1.parent and
              b3.bg_id = b2.parent
        group by ob_balance, ob_balance_name, ob_account, ob_currency, ob_client_id, ob_account_name, b2.bg_id, b2.bg_name, b3.bg_name, b1.bg_kind
        order by ob_client_id
        into :ob_balance, :ob_balance_name, :ob_account, :ob_currency, :ob_client_id, :ob_end, :ob_oborot_dt,
             :ob_oborot_kt, :ob_end_ekv, :ob_oborot_dt_ekv, :ob_oborot_kt_ekv, :ob_account_name, :ob_group_name,
             :ob_class_name, :bg_kind
    do
    begin
      ob_begin = ob_end - (ob_oborot_dt - ob_oborot_kt);
      ob_begin_ekv = ob_end_ekv - (ob_oborot_dt_ekv - ob_oborot_kt_ekv);

      if ((ob_begin <> 0) or (ob_oborot_dt <> 0) or (ob_oborot_kt <> 0) or (ob_end <> 0) or (ob_begin_ekv <> 0) or (ob_oborot_dt_ekv <> 0) or (ob_oborot_kt_ekv <> 0) or (ob_end_ekv <> 0)) then
      begin
        if (ob_client_id is not null) then
        begin
          if ((cl_prior is null) or (cl_prior <> ob_client_id)) then
          begin
            select c_name_fio
            from clients
            where c_id = :ob_client_id
            into :ob_client_name;
          end
        end
        else
          ob_client_name = 'Без контрагента';

        acc_kind_name = 'Не указан';
        if (bg_kind = 1) then
        begin
          acc_kind_name = '1 Активы';
        end
        if ((bg_kind = 2) or (bg_kind = 3)) then
        begin
          acc_kind_name = '2 Обязательства и капитал';
        end
        if ((bg_kind = 4) or (bg_kind = 5)) then
        begin
          acc_kind_name = '3 Доходы и расходы';
        end

        suspend;
        cl_prior = ob_client_id;
      end
    end
  end
end
