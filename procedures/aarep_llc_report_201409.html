CREATE PROCEDURE AAREP_LLC_REPORT_201409(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_IS_PRIOR INTEGER)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    C_NAME_FIO VARCHAR(100),
    K_DISB_DATE DATE,
    LK_IS_SECURED_STR VARCHAR(5),
    LK_IS_RESTRUCT_STR VARCHAR(5),
    LK_BALANCE CURR_TYPE,
    LK_CALCPERC CURR_TYPE,
    LK_DELIQ_DAYS INTEGER,
    LK_PERCENT PERC_TYPE,
    RPPU_BALANCE CURR_TYPE,
    RPPU_CALCPERC CURR_TYPE,
    REG_ID INTEGER)
AS
declare variable k_id decimal(18,0);
declare variable lk_is_secured integer;
declare variable lk_is_restruct integer;
declare variable k_kolich_client_approved integer;
declare variable pr_id integer;
declare variable pr_code varchar(100);
declare variable pr_loan_type integer;
declare variable pr_product_kind integer;
begin
  if (rep_date_end >= '31.08.2017') then -- после этой даты выданным до 28.02.2012 обеспеченный или нет будет расчитываться
  begin
    for select ks_from_k_id, ks_date_begin, lk_is_secured, lk_is_restruct, lk_balance, lk_calcperc, lk_deliq_days,
               lk_percent, id_list
        from llc_main, llc_kont, kontstate, region_list(:rep_region_id, 1, 3, 1)
        where lm_is_signed = 1 and
              lm_id = lk_from_lm_id and
              :rep_date_end between lk_date and lk_date_end and
              lk_from_k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end between ks_date_begin and ks_date_end and
              (:rep_is_prior = 1 and
              ks_date_begin < '28.02.2012' or
              :rep_is_prior = 0 and
              ks_date_begin >= '28.02.2012') and
              lk_from_reg_id = id_list
        into k_id, k_disb_date, lk_is_secured, lk_is_restruct, lk_balance, lk_calcperc, lk_deliq_days, lk_percent,
             reg_id
    do
    begin
      select k_kontract_no, c_name_fio
      from kontract
      left join clients on (k_clients_cashier_id = c_id)
      where k_id = :k_id
      into k_kontract_no, c_name_fio;

      rppu_balance = cast(lk_balance as double precision) * lk_percent / 100;
      rppu_calcperc = cast(lk_calcperc as double precision) * lk_percent / 100;

      if (lk_is_secured is null) then
        lk_is_secured = 0;
      if (lk_is_restruct is null) then
        lk_is_restruct = 0;

      if (lk_is_secured = 1) then
        lk_is_secured_str = 'Да';
      else
        lk_is_secured_str = 'Нет';

      if (lk_is_restruct = 1) then
        lk_is_restruct_str = 'Да';
      else
        lk_is_restruct_str = 'Нет';

      suspend;
    end
  end
  else
  begin -- если дата отчета до 31.08.2017 то обеспеченный или нет выданным до 28.02.2012 будет рассчитываться в отчете
    for select ks_from_k_id, ks_date_begin, lk_is_secured, lk_is_restruct, lk_balance, lk_calcperc, lk_deliq_days,
               lk_percent, id_list
        from llc_main, llc_kont, kontstate, region_list(:rep_region_id, 1, 3, 1)
        where lm_is_signed = 1 and
              lm_id = lk_from_lm_id and
              :rep_date_end between lk_date and lk_date_end and
              lk_from_k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end between ks_date_begin and ks_date_end and
              (:rep_is_prior = 1 and
              ks_date_begin < '28.02.2012' or
              :rep_is_prior = 0 and
              ks_date_begin >= '28.02.2012') and
              lk_from_reg_id = id_list
        into k_id, k_disb_date, lk_is_secured, lk_is_restruct, lk_balance, lk_calcperc, lk_deliq_days, lk_percent,
             reg_id
    do
    begin
      select k_kontract_no, c_name_fio, k_kolich_client_approved, k_from_pr_id
      from kontract
      left join clients on (k_clients_cashier_id = c_id)
      where k_id = :k_id
      into k_kontract_no, c_name_fio, k_kolich_client_approved, pr_id;

      rppu_balance = cast(lk_balance as double precision) * lk_percent / 100;
      rppu_calcperc = cast(lk_calcperc as double precision) * lk_percent / 100;

      if (:rep_is_prior = 1) then
      begin
        select pr_code, pr_loan_type, pr_product_kind
        from product
        where pr_id = :pr_id
        into pr_code, pr_loan_type, pr_product_kind;

        select is_secured
        from aarep_get_secured_loan(:pr_code, :pr_loan_type, :pr_product_kind, :pr_id, :k_disb_date, :k_kolich_client_approved, :k_id)
        into lk_is_secured;
      end

      if (lk_is_secured is null) then
        lk_is_secured = 0;
      if (lk_is_restruct is null) then
        lk_is_restruct = 0;

      if (lk_is_secured = 1) then
        lk_is_secured_str = 'Да';
      else
        lk_is_secured_str = 'Нет';

      if (lk_is_restruct = 1) then
        lk_is_restruct_str = 'Да';
      else
        lk_is_restruct_str = 'Нет';

      suspend;
    end
  end
end
