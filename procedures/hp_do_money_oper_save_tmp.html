CREATE PROCEDURE HP_DO_MONEY_OPER_SAVE_TMP(
    OL_ID ID_TYPE,
    REG_ID INTEGER,
    DOC_NO INTEGER,
    OPER_DATE DATE,
    DOC_NO_STR D_STR_CODE,
    DOC_KIND INTEGER,
    MEMO D_STR_2048,
    UN_ID ID_TYPE,
    CLIENT_ID ID_TYPE,
    BANK_ACC_NO_CLIENT D_STR_CODE,
    OFFICE_ID ID_TYPE,
    BANK_ACC_NO D_STR_CODE,
    OPER_SUBTYPE INTEGER,
    KONTRAGENT_TYPE INTEGER,
    CURR_ID D_STR_10,
    CURR_RATE D_CURRENCY,
    OB_CLIENT_DT ID_TYPE,
    DT_ACC D_STR_CODE,
    AMOUNT_VAL D_CURRENCY,
    AMOUNT D_CURRENCY,
    PC_CODE D_STR_CODE,
    KP_CODE D_STR_CODE,
    KT_ACC D_STR_CODE,
    OB_CLIENT_KT ID_TYPE,
    OB_DETAIL D_STR_LONG,
    SAC_ID D_INTEGER)
AS
begin
  insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                        ol_status, is_delete, ol_user_id, ol_document_client, ol_acc_no_client, ol_is_mo_modified,
                        ol_document_office, ol_acc_no_office, ol_oper_subtype, ol_is_client)
  values (:ol_id, :reg_id, 2, -- денежная операция
          :oper_date, :doc_no, :doc_no_str, :doc_kind, :memo, 0, -- ol_status
          0, -- is_delete
          :un_id, :client_id, iif(:client_id is not null, :bank_acc_no_client, null), 0, -- ol_is_mo_modified
          :office_id, iif(:office_id is not null, :bank_acc_no, null), :oper_subtype, iif(:kontragent_type = 1, 1, 0));

  insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                        ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                        ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
  values (:ol_id, -1, :curr_id, :curr_rate, :ob_client_dt, :dt_acc, :amount_val, :amount, :pc_code, :kp_code, null,
          :kt_acc, :ob_client_kt, :ob_detail, :sac_id, :oper_date, 1, :reg_id);
end
