CREATE PROCEDURE UXP$DB_SOURCE_FIELDS(
    SOURCE_NAME D_STR_CODE,
    SOURCE_FIELD D_STR_CODE = null)
RETURNS (
    RES_INDEX D_INTEGER,
    RES_IS_PARAM D_SMALLINT,
    RES_NAME D_STR_CODE,
    RES_DOMAIN D_STR_CODE,
    RES_DEFAULT D_STR_100,
    RES_ORIG_DEFAULT D_STR_100,
    RES_DATA_TYPE D_STR_CODE,
    RES_REQUIRED D_SMALLINT,
    RES_BASE_FIELD D_STR_CODE,
    RES_BASE_TABLE D_STR_CODE,
    RES_READONLY D_BOOLEAN,
    US_IS_TABLE D_BOOL)
AS
declare variable v_context_id d_integer;
declare variable v_context_name d_str_code;
declare variable v_context_type d_smallint;
begin
  source_name = upper(source_name);
  source_field = upper(coalesce(source_field, ''));

  us_is_table = 0;

  if (exists(select rdb$relation_type
             from rdb$relations
             where rdb$relation_name = :source_name)) then
    us_is_table = 1;

  res_base_table = '';
  res_base_field = '';

  if (us_is_table = 1) then
  begin
    for select lower(trim(r.rdb$field_name)), r.rdb$field_position, coalesce(r.rdb$null_flag, 0),
               (f.rdb$computed_source is not null), lower(trim(r.rdb$base_field)), r.rdb$view_context
    from rdb$fields f, rdb$relation_fields r
    where f.rdb$field_name = r.rdb$field_source and
          r.rdb$system_flag = 0 and
          r.rdb$relation_name = :source_name and
          (r.rdb$field_name = :source_field or :source_field = '')
    order by r.rdb$relation_name, r.rdb$field_position
    into :res_name, :res_index, :res_required, :res_readonly, res_base_field, v_context_id
    do
    begin
      res_base_table = '';
      if (res_base_field is null) then
        res_base_field = '';

      if (v_context_id is not null) then
      begin
        v_context_type = 1;
        v_context_name = source_name;

        while (v_context_type = 1) do
        begin
          v_context_type = 0;

          select lower(trim(rdb$relation_name)), r.rdb$context_name, r.rdb$context_type
          from rdb$view_relations r
          where rdb$view_name = :v_context_name and
                rdb$view_context = :v_context_id
          rows 1
          into :res_base_table, v_context_name, v_context_type;

          if (v_context_type = 1) then
          begin
            select lower(trim(r.rdb$base_field)), (f.rdb$computed_source is not null), r.rdb$view_context
            from rdb$fields f, rdb$relation_fields r
            where f.rdb$field_name = r.rdb$field_source and
                  r.rdb$system_flag = 0 and
                  r.rdb$relation_name = upper(:res_base_table) and
                  r.rdb$field_name = upper(:res_base_field)
            into res_base_field, res_readonly, v_context_id;
          end
        end
      end

      res_is_param = 0;
      for select field_default, field_orig_default, field_domain, field_type, (:res_readonly or field_readonly)
      from uxp$uxfield_db_info(iif(:res_base_table = '', :source_name, :res_base_table),
                               iif(:res_base_field = '', :res_name, :res_base_field), :us_is_table)
      into res_default, res_orig_default, res_domain, res_data_type, res_readonly
      do
      begin
        if (res_domain starting 'rdb$') then
          exception e_default 'Field domain not defined: ' || source_name || '.' || res_name || ' -> ' || res_domain;
  
        suspend;
      end
    end
  end else
  begin
    res_readonly = false;

    for select lower(trim(rdb$parameter_name)), rdb$parameter_number, iif(rdb$parameter_type = 0, 1, 0),
               coalesce(rdb$null_flag, 0)
    from rdb$procedure_parameters p
    where rdb$procedure_name = :source_name and
          rdb$package_name is null and
          (rdb$parameter_name = :source_field or :source_field = '')
    order by 3, 2
    into :res_name, :res_index, res_is_param, res_required
    do
    begin
      res_name = lower(res_name);

      for select field_default, field_orig_default, field_domain, field_type
      from uxp$uxfield_db_info(:source_name, :res_name, :us_is_table)
      into res_default, res_orig_default, res_domain, res_data_type
      do
      begin
        if (res_domain starting 'rdb$') then
          exception e_default 'Field domain not defined: ' || source_name || '.' || res_name || ' -> ' || res_domain;

        suspend;
      end
    end
  end
end
