CREATE PROCEDURE AAREP_NUMBERS_VERIFICATION(
    REP_REG_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REP_OFFICE D_STR_100,
    REP_REGION D_STR_100,
    NAME_FIO D_STR_100,
    OLD_PHONE D_STR_100,
    CURR_PHONE D_STR_100,
    NAME_FIO_RESP D_STR_100,
    MEMO D_STR_255,
    STATUS_STR D_STR_CODE,
    ACL_DATE D_DATE,
    START_DATETIME D_DATETIME,
    LAST_DATETIME D_DATETIME,
    LIFE_TIME D_INTEGER,
    TOTAL_TIME D_INTEGER,
    EVENT_MAKER D_STR_100)
AS
declare variable c_id d_id;
declare variable reg_id d_integer;
declare variable rd_id d_integer;
declare variable rd_id_cont d_integer;
declare variable resp_c_id d_id;
declare variable status d_smallint;
declare variable object_kind d_integer;
declare variable un_id d_id;
declare variable un_id_tmp d_id;
begin
  rd_id = null;
  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ActionGroup.SimpleVerify_Phone_Number_Verify'
  into :rd_id;

  rd_id_cont = null;
  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ContactInfo.Cell_Main'
  into :rd_id_cont;

  object_kind = f_objkind('clients', 'id');
  for select reglist.id_list, reg.reg_name, reglist.parent_name
      from region_list(:rep_reg_id, 1, 3, 1) as reglist
      inner join region reg on reg.reg_unicode = reglist.id_list
      into :reg_id, :rep_office, :rep_region
  do
  begin
    for select acl.acl_object_id, acl.acl_respons_c_id, acl.acl_status, acl.acl_date, acl.acl_datetime, acl.acl_last_datetime,
               cast((acl.acl_lifetime_in_second / 60) as integer),
               datediff(minute, acl.acl_datetime, acl.acl_last_datetime)
        from actionlist acl
        where acl.acl_from_rd_id = :rd_id and
              acl.acl_object_kind = :object_kind and
              acl.acl_date between :rep_date_begin and :rep_date_end and
              acl.acl_from_reg_id = :reg_id and
              acl.acl_status in (-2, 2)
        into :c_id, :resp_c_id, :status, :acl_date, :start_datetime, :last_datetime, :life_time, :total_time
    do
    begin
      memo = '';
      if (status = 2) then
        status_str = 'Подтвержден';
      if (status = -2) then
      begin
        status_str = 'Отказан';
        select obm.obm_memo
        from objectprops opr
        inner join objectmemo obm on opr.opr_id = obm.obm_from_opr_id
        where opr.opr_object_kind = :object_kind and
              opr.opr_object_id = :c_id and
              opr.opr_subject_id = :rd_id and
              cast(obm.obm_datetime as date) = cast(:last_datetime as date)
        order by obm.obm_id desc
        rows 1
        into :memo;
      end
      name_fio = '';
      select c.c_name_fio
      from clients c
      where c.c_id = :c_id
      into :name_fio;

      name_fio_resp = '';
      select c.c_name_fio
      from clients c
      where c.c_id = :resp_c_id
      into :name_fio_resp;
          
      select ovs.ovs_from_un_id
      from objectprops opr
      inner join objectvalues ovs on opr.opr_id = ovs.ovs_from_opr_id
      where opr.opr_object_id = :c_id
            and opr.opr_object_kind = :object_kind
            and opr.opr_subject_id = :rd_id
            and cast(ovs.ovs_datetime as date) = cast(:last_datetime as date)
      rows 1
      into :un_id_tmp;

      un_id = 0;
      if (:un_id <> :un_id_tmp) then
      begin
        un_id = :un_id_tmp;
        event_maker = '';
        select un.un_fio
        from username un
        where un.un_id = :un_id
        into :event_maker;
      end

      curr_phone = '';
      curr_phone = f_cont_info('clients', :c_id, 'cell_main', 'ok');

      old_phone = '';
      select ovs.ovs_value
      from objectvalues ovs
      inner join objectprops opr on ovs.ovs_from_opr_id = opr.opr_id
      where opr.opr_object_kind = :object_kind and
              opr.opr_object_id = :c_id and
              opr.opr_subject_id = :rd_id_cont and
              ovs.ovs_date_end between :rep_date_begin and :rep_date_end
      rows 1
      into :old_phone;

      if (old_phone = '') then
      begin
        old_phone = '';
        select ovs.ovs_value
        from objectvalues ovs
        inner join objectprops opr on ovs.ovs_from_opr_id = opr.opr_id
        where opr.opr_object_kind = :object_kind and
              opr.opr_object_id = :c_id and
              opr.opr_subject_id = :rd_id_cont and
              ovs.ovs_value <> '' and
              ovs.ovs_date_begin between :rep_date_begin and :rep_date_end
        rows 1
        into :old_phone;
      end
      suspend;
    end
  end
end
