CREATE PROCEDURE PRINT_EQUIFAX_PMT_LOAD(
    BLS_ID D_ID,
    CRED_ID D_STR_CODE)
RETURNS (
    RES_YEAR D_STR_10,
    RES_JAN D_STR_20,
    RES_FEB D_STR_20,
    RES_MAR D_STR_20,
    RES_APR D_STR_20,
    RES_MAY D_STR_20,
    RES_JUN D_STR_20,
    RES_JUL D_STR_20,
    RES_AUG D_STR_20,
    RES_SEP D_STR_20,
    RES_OCT D_STR_20,
    RES_NOV D_STR_20,
    RES_DEC D_STR_20)
AS
declare variable rep_date d_date;
declare variable path_base d_str_100;
declare variable month_cnt d_smallint;
declare variable cred_cnt d_smallint;
declare variable json_full d_blob_text;
declare variable i d_smallint;
declare variable j d_smallint;
declare variable month_no d_smallint;
declare variable month_delta d_smallint;
declare variable month_val d_str_20;
declare variable month_val_cnt d_smallint;
declare variable cred_id_tmp d_str_code;
declare variable date_tmp d_date;
declare variable year_next d_smallint;
declare variable jso d_integer;
declare variable jsa d_integer;
begin
  -- Процедура возвращает "Платежную нагрузку по договору" отчета Эквифакс
  -- Переведена на пакет json, т.к. предыдущая версия ломалась при большом массиве
  -- Переписывание без json, привело к времени выполнения 3 минуты, для одного кредита

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into json_full;

  if (json_full is null) then
    exit;

  -- Базовая ссылка, на список месяцев
  path_base = 'response.addPart.attrs._package[1].section[0].period';
  jso = json.parse(json_full);

  --Перемещаемся на начало месяца, чтобы было удобно отнимать по одному месяцу
  rep_date = json.obj_get_val(jso, 'response.dateofreport', current_date);
  rep_date = rep_date - extract(day from rep_date) + 1;
  res_year = -1; -- Текущий год
  month_val_cnt = 0; -- Количество непустых значений (сумм) в текущем году

  -- Получаем размер массива месяцев
  month_cnt = json.count_get(json.obj_get_arr(jso, path_base));

  i = 0;
  while (i < month_cnt) do
  begin
    month_delta = json.obj_get_val(jso, path_base || f_format('[%1].month', i), 0);
    date_tmp = dateadd(month, - month_delta, rep_date);

    year_next = extract(year from date_tmp);
    if (year_next <> res_year and res_year <> -1) then
    begin
      -- При смене года делаем suspend и зачистку, при условии что были суммы
      if (month_val_cnt > 0) then
        suspend;

      res_jan = null;
      res_feb = null;
      res_mar = null;
      res_apr = null;
      res_may = null;
      res_jun = null;
      res_jul = null;
      res_aug = null;
      res_sep = null;
      res_oct = null;
      res_nov = null;
      res_dec = null;
      month_val_cnt = 0;
    end
    res_year = year_next;

    -- Получаем размер массива выплаченных сумм кредитов внутри месяца
    jsa = json.obj_get_arr(jso, path_base || f_format('[%1].attr', i));
    cred_cnt = json.count_get(jsa);
    j = 0;
    while (j < cred_cnt) do
    begin
      cred_id_tmp = json.obj_get_val(json.arr_get_obj(jsa, j), 'ci', '-');
      if (cred_id_tmp = cred_id) then
      begin
        month_val = json.obj_get_val(json.arr_get_obj(jsa, j), 'v', '-');
        month_no = extract(month from date_tmp);
        if (month_no = 1)  then res_jan = month_val;
        if (month_no = 2)  then res_feb = month_val;
        if (month_no = 3)  then res_mar = month_val;
        if (month_no = 4)  then res_apr = month_val;
        if (month_no = 5)  then res_may = month_val;
        if (month_no = 6)  then res_jun = month_val;
        if (month_no = 7)  then res_jul = month_val;
        if (month_no = 8)  then res_aug = month_val;
        if (month_no = 9)  then res_sep = month_val;
        if (month_no = 10) then res_oct = month_val;
        if (month_no = 11) then res_nov = month_val;
        if (month_no = 12) then res_dec = month_val;
        if (coalesce(month_val, '-') <> '-') then
          month_val_cnt = month_val_cnt + 1;

      end
      j = j + 1;
    end
    i = i + 1;
  end
  -- И возвращаем последний (старейший) год
  if (month_val_cnt > 0) then
    suspend;
end
