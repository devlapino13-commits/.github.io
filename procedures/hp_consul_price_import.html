CREATE PROCEDURE HP_CONSUL_PRICE_IMPORT(
    UN_ID D_ID)
AS
declare variable cpi_id d_id;
declare variable cpi_from_pr_id d_integer;
declare variable cpi_from_prl_id d_integer;
declare variable cpi_min_point d_integer;
declare variable cpi_max_point d_integer;
declare variable cpi_min_amount d_currency;
declare variable cpi_max_amount d_currency;
declare variable cpi_price d_currency;
declare variable cpi_level d_smallint;
declare variable cpi_period d_smallint;
begin
  for select cpi_id,
             cpi_from_pr_id,
             cpi_min_point,
             cpi_max_point,
             cpi_min_amount,
             cpi_max_amount,
             cpi_price,
             cpi_level,
             coalesce(cpi_period, 3)
  from consul_price_import
  where coalesce(cpi_is_imported,0) = 0
  into :cpi_id,
       :cpi_from_pr_id,
       :cpi_min_point,
       :cpi_max_point,
       :cpi_min_amount,
       :cpi_max_amount,
       :cpi_price,
       :cpi_level,
       :cpi_period
  do
  begin

    select first 1 prl.prl_id
    from prodlogic prl
    where prl.prl_from_pr_id = :cpi_from_pr_id
      and current_date between prl.prl_date_begin and prl.prl_date_end
    order by prl.prl_date_begin desc
    into :cpi_from_prl_id;

    insert into consul_price (
        cp_from_pr_id,
        cp_from_prl_id,
        cp_min_point,
        cp_max_point,
        cp_min_amount,
        cp_max_amount,
        cp_price,
        cp_level,
        cp_period,
        cp_from_un_id)
    values (
        :cpi_from_pr_id,
        :cpi_from_prl_id,
        :cpi_min_point,
        :cpi_max_point,
        :cpi_min_amount,
        :cpi_max_amount,
        :cpi_price,
        :cpi_level,
        :cpi_period,
        :un_id);

    update consul_price_import cpi
    set cpi.cpi_is_imported = 1
    where cpi.cpi_id = :cpi_id;

  end
end
