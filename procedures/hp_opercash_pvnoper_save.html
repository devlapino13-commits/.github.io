CREATE PROCEDURE HP_OPERCASH_PVNOPER_SAVE(
    IN_PO_ID D_INTEGER)
AS
declare variable po_date d_date;
declare variable po_from_ol_id d_id;
declare variable po_oper_type d_smallint;
declare variable po_card_id d_integer;
declare variable po_from_c_id d_id;
declare variable po_amount d_currency;
declare variable po_un_id d_id;
declare variable po_office_id d_integer;
declare variable po_status d_integer;
declare variable po_acc_kt d_str_20;
declare variable po_card_office_id d_integer;
declare variable ob_from_ol_id d_id;
declare variable ol_memo d_str_long;
declare variable opc_oper_kind d_integer;
declare variable opc_manager_id d_id;
declare variable is_ipf d_bool;
declare variable opc_card_no d_str_20;
declare variable cash_id d_integer;
declare variable acc_dt d_str_20;
declare variable ol_doc_no d_integer;
declare variable cso_is_active d_bool;
declare variable cso_activate_date d_date;
begin
  for select po_date, po_from_ol_id, po_oper_type, po_card_id, po_from_c_id, po_amount,
             po_un_id, po_office_id, po_status, po_acc_kt, po_card_office_id
  from pvnopers
  where po_id = :in_po_id
  into :po_date, :po_from_ol_id, :po_oper_type, :po_card_id, :po_from_c_id, :po_amount,
       :po_un_id, :po_office_id, :po_status, :po_acc_kt, :po_card_office_id
  do
  begin
    if (po_oper_type <> 0) then
      exit;

    if (po_from_ol_id is null) then
      exit;

    if (po_status not in (3, 4)) then
      exit;

    if (po_amount = 0) then
      exit;

    select cso_is_active, cso_activate_date
    from cashoffice
    where cso_office_id = :po_office_id
    into cso_is_active, cso_activate_date;

    if (row_count = 0) then
      exit;

    if (cso_is_active = 0 or cso_activate_date is null) then
      exit;

    if (cso_activate_date > po_date) then
      exit;

    select csp_id, iif(csp_cls_al_id = :po_acc_kt, 0, 1)
    from cashplace
    where csp_office_id = :po_office_id and
          (csp_cls_al_id = :po_acc_kt or csp_ipf_al_id = :po_acc_kt)
    into cash_id, is_ipf;

    select coalesce(ol_parent_id, ol_id), ol_doc_no, ol_memo
    from operlist
    where ol_id = :po_from_ol_id
    into ob_from_ol_id, ol_doc_no, ol_memo;

    select ob_account_dt
    from operbook
    where ob_from_ol_id = :ob_from_ol_id and
          ob_summa = :po_amount
    into acc_dt;

    opc_oper_kind = f_enum_id('cashoperkind', 'minus.mcard');

    select opc_manager_id
    from opercash
    where opc_from_csp_id = :cash_id and
          opc_manager_id is not null
    order by opc_id desc
    rows 1
    into opc_manager_id;

    if (po_card_id is not null) then
    begin
      select cal_pan_short
      from cardlist
      where cal_card_id = :po_card_id
      into opc_card_no;

      if (coalesce(opc_card_no, '') <> '') then
        opc_card_no = left(opc_card_no, 6) || '******' || right(opc_card_no, 4);
    end

    insert into opercash (opc_date, opc_doc_no, opc_office_id, opc_from_csp_id, opc_oper_kind, opc_sign,
                          opc_amount, opc_manager_id, opc_from_c_id, opc_is_ipf, opc_corr_al_id,
                          opc_corr_reg_id, opc_card_no, opc_memo, opc_status,
                          opc_from_ol_id, opc_from_un_id)
    values (:po_date, :ol_doc_no, :po_office_id, :cash_id, :opc_oper_kind, -1,
            :po_amount, :opc_manager_id, :po_from_c_id, :is_ipf, :acc_dt,
            :po_card_office_id, :opc_card_no, :ol_memo, 1,
            :po_from_ol_id, :po_un_id);
  end
end
