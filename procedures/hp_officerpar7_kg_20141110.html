CREATE PROCEDURE HP_OFFICERPAR7_KG_20141110(
    REP_DATE_END DATE)
RETURNS (
    OKS_KC_ID ID_TYPE,
    OKS_DELINQ_BALANCE CURR_TYPE,
    OKS_BALANCE_BEGIN CURR_TYPE,
    PAR DECIMAL(5,5))
AS
declare variable rep_date_begin date;
begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end);
-- Баланс просрчоки на каждого КСа
  for select oks_kc_id,
             sum(iif(k_disb_date >= cast('01.01.2014' as date) and :rep_date_end >= oks_delinq_date, oks_balance, 0))
      from oksummary, kontract
      where :rep_date_end between oks_date_begin and oks_date_end and
            :rep_date_end >= oks_delinq_date and
            oks_from_k_id = k_id and
            oks_close_state = 0
      group by oks_kc_id
      into oks_kc_id, oks_delinq_balance
  do
  begin
--  Баланс на начало
    oks_balance_begin = null;
    select sum(oks_balance)
    from oksummary
    where :rep_date_begin between oks_date_begin and oks_date_end and
          oks_kc_id = :oks_kc_id and
          oks_close_state = 0
    into oks_balance_begin;
    if (oks_balance_begin is null) then
      oks_balance_begin = 0;
--  Расчет ПАР
    if (oks_delinq_balance = 0) then
    begin
      par = 0;
    end
    else
    if (oks_balance_begin = 0) then
    begin
      par = 100;
    end
    else
    begin
      par = cast(oks_delinq_balance as double precision) / oks_balance_begin * 100;
    end

    insert into officerpar7 (op_kc_id, op_number, op_date_begin, op_date_end, op_balance, op_delinq_bal8_180,
                             op_par8_180)
    values (:oks_kc_id, null, :rep_date_end, '31.12.9999', :oks_balance_begin, :oks_delinq_balance, :par);

--    suspend;
  end
end
