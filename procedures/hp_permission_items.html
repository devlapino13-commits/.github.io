CREATE PROCEDURE HP_PERMISSION_ITEMS
RETURNS (
    ML_ID D_INTEGER,
    ML_PARENT_ID D_INTEGER,
    ML_CAPTION D_STR_255,
    ML_DESCRIPTION D_STR_LONG,
    ML_IS_REPORT D_SMALLINT,
    ML_SECURE_KIND D_SMALLINT,
    PA_COMMAND_NAME D_STR_100)
AS
declare variable ux_uuid d_uuid;
declare variable ux_kind d_str_code;
declare variable tmp_name d_str_long;
declare variable ua_uuid d_uuid;
begin
  pa_command_name = '';
  ml_id = 0;
  ml_parent_id = null;
  ml_caption = 'Пункты меню FinStart';
  ml_description = 'Выбранные пункты меню будут доступны в системе FinStart';
  ml_is_report = 0;
  ml_secure_kind = -1;
  suspend;

  for select ml_id, coalesce(ml_parent_id, 0), ml_caption, ml_description, ml_is_report, ml_secure_kind,
             uml_ua_uuid
  from menulist left join ux$menulist on (uml_ml_id = ml_id)
  into ml_id, ml_parent_id, ml_caption, ml_description, ml_is_report, ml_secure_kind, ua_uuid
  do
  begin
    if (ml_is_report is null) then
    begin
      if (ml_id > 0) then
        ml_is_report = 1;
      else
        ml_is_report = 0;
    end

    if (ml_description is null) then
      ml_description = '';

    if (ua_uuid is not null) then
    begin
      ux_uuid = null;
      tmp_name = '';

      select uxa_name || '   / ' || uxa_fcode, ux_uuid
      from ux$action, ux$alias, ux$code, ux$uniq
      where ua_uuid = :ua_uuid and
            ua_uxa_uuid = uxa_uuid and
            uxa_owner = ux_uuid and
            uxa_uuid = uu_id
      order by uu_datetime
      into tmp_name, ux_uuid;

      if (ux_uuid is not null) then
      begin
        ux_kind = uxf$kind(uxf$code(ux_uuid, 'kind'), 'caption');
        tmp_name = ux_kind || ', ' || tmp_name;
        ml_description = ml_description || ',  FsConfig: ' || tmp_name;
      end
    end

    suspend;
  end

  ml_id = 1000000;
  ml_parent_id = null;
  ml_caption = 'Действия и разрешения';
  ml_description = 'Действия и разрешения во всех системах Компании';
  ml_is_report = 2;
  ml_secure_kind = -1;
  suspend;

  for select pa_id + 2000000, coalesce(pa_parent_id + 2000000, 1000000),
             pa_name, pa_memo, pa_secure_kind, pa_command_name
  from permactions
  into ml_id, ml_parent_id, ml_caption, ml_description, ml_secure_kind, pa_command_name
  do
  begin
    if (ml_description is null) then
      ml_description = '';

    if (pa_command_name is null) then
      pa_command_name = '';

    if (pa_command_name <> '') then
    begin
      if (ml_description <> '') then
      begin
        if ((position(pa_command_name, ml_description) = 0) and
            (position(ml_description, pa_command_name) = 0)) then
          ml_description = ml_description || '   (' || lower(pa_command_name) || ')';
      end else
        ml_description = pa_command_name;
    end

    suspend;
  end

  pa_command_name = '';

  ml_id = 5000000;
  ml_parent_id = null;
  ml_caption = 'Динамические формы';
  ml_description = 'Алиасы динамических форм из конфигуратора FsConfig без привязки к главному меню';
  ml_is_report = 5;
  ml_secure_kind = -1;
  suspend;

  ml_secure_kind = 0;
  ml_parent_id = ml_id;

  for select uxa_id + 5000000,
             uxa_name || '   / ' || uxa_fcode,
             ux_name || '   (' || lower(uuid_to_char(ux_uuid)) || ',  ' || lower(uuid_to_char(uxa_uuid)),
             ux_uuid
  from ux$alias, ux$code, ux$uniq
  where uxa_owner = ux_uuid and
        uxa_in_mainmenu = 0 and
        uxa_uuid = uu_id and
        uxa_need_access = 1
  order by uu_datetime
  into ml_id, ml_caption, ml_description, ux_uuid
  do
  begin
    ux_kind = uxf$kind(uxf$code(ux_uuid, 'kind'), 'caption');
    ml_description = 'FsConfig: ' || ux_kind || ', ' || ml_description;

    suspend;
  end

end
