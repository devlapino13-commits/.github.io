CREATE PROCEDURE AAREP_AVTO_LOAN_TMP(
    REP_DATE DATE,
    REP_MODE INTEGER = 0)
RETURNS (
    KAL_NEW_KF_NAME STR_100,
    K_KONTRACT_NO STR_20,
    K_DISB_DATE DATE,
    KS_STATE STR_100,
    KS_DATE_BEGIN DATE,
    CAR_OWNER STR_20)
AS
declare variable k_id id_type;
declare variable c_id id_type;
begin
  -- При выдаче / закрытии автозайма
  if (rep_mode = 0) then
  for
    select
      kal_new_kf_name,
      k_kontract_no,
      k_disb_date,
      case ks_state
        when 1 then 'Выдан'
        when 2 then 'Закрыт/Списан'
      end,
      ks_date_begin,
      k_id
    from kontract
      inner join product
        on k_from_pr_id = pr_id
          and (pr_code containing '.ZALOG_CAR.' or pr_code containing '.zalog.')
      inner join knclient
        on kc_from_k_id = k_id
      inner join kontstate
        on ks_from_k_id = k_id
          and ks_ol_is_delete = 0
          and ks_state in (1, 2)
          and ks_date_begin = :rep_date
      inner join knaktlist
        on kal_from_kf_id = k_from_kf_id
          and :rep_date between kal_date and kal_date_end
          and knaktlist.is_delete = 0
  into
    :kal_new_kf_name,
    :k_kontract_no,
    :k_disb_date,
    :ks_state,
    :ks_date_begin,
    :k_id
  do
  begin
    if (ks_state = 'Выдан') then begin
      if (exists(
            select *
            from aarep_autoloan_somename(:k_id,null) as ow
              inner join aarep_autoloan_somename(:k_id,null,0) as cl
                on cl.res_name = ow.res_name))
      then car_owner = 'да';
        else car_owner = 'нет';
    end else car_owner = '';

    if (ks_state = 'Закрыт/Списан') then  begin
    if (exists( select oks_close_state
                from oksummary
                where oks_from_k_id = :k_id
                and :rep_date between oks_date_begin and oks_date_end
                and oks_close_state = 1 ))then ks_state = 'Спец.учет';
    else ks_state = 'Закрыт';
    end
    suspend;
  end

  -- При прикреплении документа ПТС / ПТС(галерея) после выдачи автозайма,
  -- для тех займов, у которых на момент выдачи собственник авто не совпадал с залогодателем
  if (rep_mode = 1) then begin
    ks_date_begin = rep_date;
    ks_state = 'Прикреплен ПТС';
    for
      select distinct
        cds_from_c_id
      from clientdocs
      where datediff(day, cds_create_datetime, :rep_date) = 0
        and cds_is_delete = 0
        and cds_from_pk_id in (315,322)
    into
      :c_id
    do
    begin
      for
        select
          k_id,
          k_kontract_no,
          k_disb_date,
          kal_new_kf_name
        from kontract
          inner join (
              select z_from_k_id as kn_id
              from zalog
              where z_from_c_id = :c_id and z_type = 1
              union
              select kc_from_k_id as kn_id
              from knclient
              where kc_from_c_id = :c_id) as k
            on k.kn_id = k_id
              and datediff(day, k_disb_date, :rep_date) >= 0
          inner join kontstate
            on ks_from_k_id = k_id
              and ks_ol_is_delete = 0
              and ks_state = 1
              and :rep_date between ks_date_begin and ks_date_end
          inner join knaktlist
            on kal_from_kf_id = k_from_kf_id
              and knaktlist.is_delete = 0
              and :rep_date between kal_date and kal_date_end
          inner join product
            on pr_id = k_from_pr_id
              and (pr_code containing '.ZALOG_CAR.' or pr_code containing '.zalog.')
         where not exists(
                select *
                from aarep_autoloan_somename(k_id,null) as ow
                  inner join aarep_autoloan_somename(k_id, null, 0) as cl
                    on cl.res_name = ow.res_name)
      into
        :k_id,
        :k_kontract_no,
        :k_disb_date,
        :kal_new_kf_name
      do
      begin
        if (exists(
              select *
              from aarep_autoloan_somename(:k_id, :rep_date) as ow
                inner join aarep_autoloan_somename(:k_id, :rep_date, 0) as cl
                  on cl.res_name = ow.res_name))
        then car_owner = 'да';
          else car_owner = 'нет';
        suspend;
      end
    end
  end

  -- При неприкреплении в течении 3-х дней, хотя бы одного документа ПТС / ПТС (галерея)
  -- после выдачи автозайма, для тех займов, у которых на момент выдачи
  -- собственник авто не совпадал с залогодателем
  if (rep_mode = 2) then begin
    ks_date_begin = rep_date;
    ks_state = 'Не прикреплен ПТС';
    for
      select
        k_id,
        k_kontract_no,
        k_disb_date,
        kal_new_kf_name
      from kontract
        inner join product
          on k_from_pr_id = pr_id
            and (pr_code containing '.ZALOG_CAR.' or pr_code containing '.zalog.')
        inner join kontstate
          on ks_from_k_id = k_id
            and ks_ol_is_delete = 0
            and ks_state = 1
            and ks_date_begin = :rep_date - 3
        inner join knaktlist
          on kal_from_kf_id = k_from_kf_id
            and knaktlist.is_delete = 0
            and :rep_date between kal_date and kal_date_end
      WHERE not exists(
                select *
                from aarep_autoloan_somename(k_id, null) as ow
                  inner join aarep_autoloan_somename(k_id, null, 0) as cl
                    on cl.res_name = ow.res_name)
    into
      :k_id,
      :k_kontract_no,
      :k_disb_date,
      :kal_new_kf_name
    do
    begin
      if (not exists(
          select *
          from clientdocs
            inner join (
              select z_from_c_id as c_id from zalog where z_from_k_id = :k_id and z_type = 1
              union
              select kc_from_c_id as c_id from knclient where kc_from_k_id = :k_id) as cl
            on cl.c_id = cds_from_c_id
          where datediff(day, :k_disb_date, cds_create_datetime) > 0
            and datediff(day, cds_create_datetime, :rep_date) > 0
            and cds_is_delete = 0
            and cds_from_pk_id in (315,322)))
      then begin
          if (exists(
                select *
                from aarep_autoloan_somename(:k_id, :rep_date) as ow
                  inner join aarep_autoloan_somename(:k_id, :rep_date, 0) as cl
                    on cl.res_name = ow.res_name))
          then car_owner = 'да';
            else car_owner = 'нет';
          suspend;
      end
    end
  end

  if (rep_mode = 3) then
  begin
    ks_date_begin = rep_date;
    ks_state = 'Прикреплен "СТС в случае ЭПТС (выписка)"';
    for select k.k_id, k_kontract_no, k.k_disb_date, kal_new_kf_name
    from clientdocs cds, knclient kc, kontstate ks, kontract k, product pr, knaktlist kal
    where cds.cds_from_pk_id = 343
      and cast (cds.cds_create_datetime as date) = :rep_date
      and kc.kc_from_c_id = cds.cds_from_c_id
      and ks_from_k_id = kc.kc_from_k_id
      and ks_ol_is_delete = 0
      and ks_state = 1
      and :rep_date between ks.ks_date_begin and ks.ks_date_end
      and k_id = kc_from_k_id
      and k_from_pr_id = pr_id
      and (pr_code containing '.ZALOG_CAR.' or pr_code containing '.zalog.')
      and kal_from_kf_id = k_from_kf_id
      and kal.is_delete = 0
      and :rep_date between kal_date and kal_date_end
     into k_id, k_kontract_no, k_disb_date, kal_new_kf_name
    do
      suspend;
  end
end
