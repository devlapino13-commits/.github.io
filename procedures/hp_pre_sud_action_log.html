CREATE PROCEDURE HP_PRE_SUD_ACTION_LOG(
    REP_DATE_START D_DATE,
    REP_DATE_STOP D_DATE,
    REP_K_ID D_ID)
RETURNS (
    LOG_ID D_ID,
    ACT_ID D_ID,
    ACT_K_ID D_ID,
    ACT_K_KONTRACT_NO D_STR_20,
    ACT_DATETIME D_DATETIME,
    ACT_TYPE D_STR_CODE,
    ACT_UN_FIO D_STR_100)
AS
begin
  -- Журнал действий в рамках досудебных взысканий (RFCRU-4792)
  -- Финстарт: Документы -> Модуль досудебного взыскания

  if (rep_k_id is null) then
  begin
    if (rep_date_start > rep_date_stop) then
      exception e_message 'Указан некорректный период';
    else if (rep_date_stop - rep_date_start > 30) then
      exception e_message 'Период не может быть больше 30 дней';

    for select psal_id,
               psal_action_datetime,
               f_enum_name('pre_sud_actions', psal_action_type),
               psal_action_id,
               psal_action_k_id,
               psal_action_un_fio
        from pre_sud_action_log
        where cast(psal_action_datetime as date) between :rep_date_start and :rep_date_stop
        order by psal_id desc
        into log_id,
             act_datetime,
             act_type,
             act_id,
             act_k_id,
             act_un_fio
    do
    begin
      act_k_kontract_no = null;
      select k_kontract_no
      from kontract
      where k_id = :act_k_id
      into act_k_kontract_no;

      suspend;
    end
  end
  else
    for select psal_id,
               psal_action_datetime,
               f_enum_name('pre_sud_actions', psal_action_type),
               psal_action_id,
               psal_action_k_id,
               psal_action_un_fio
        from pre_sud_action_log
        where psal_action_k_id = :rep_k_id
        order by psal_id desc
        into log_id,
             act_datetime,
             act_type,
             act_id,
             act_k_id,
             act_un_fio
    do
    begin
      act_k_kontract_no = null;
      select k_kontract_no
      from kontract
      where k_id = :act_k_id
      into act_k_kontract_no;

      suspend;
    end
end
