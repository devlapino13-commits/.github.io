CREATE PROCEDURE AAREP_CASH_JOURNAL_EPS(
    REP_REGION_ID INTEGER,
    REP_DATE DATE,
    REP_DATE_END DATE = null)
RETURNS (
    OL_ID DECIMAL(18,0),
    OL_DOC_NO INTEGER,
    OL_CLIENT_NAME VARCHAR(255),
    OL_SUMMA_PLUS DECIMAL(18,4),
    OL_SUMMA_MINUS DECIMAL(18,4),
    OL_ACCOUNT VARCHAR(255),
    OL_KOL_PLUS INTEGER,
    OL_KOL_MINUS INTEGER,
    OL_SALDO_BEGIN_DT DECIMAL(18,4),
    OL_SALDO_BEGIN_KT DECIMAL(18,4),
    OL_SALDO_END_DT DECIMAL(18,4),
    OL_SALDO_END_KT DECIMAL(18,4),
    OL_KIND INTEGER,
    CORR_ACC VARCHAR(20),
    OL_BRANCH_NAME VARCHAR(255),
    OL_DOC_NO_STR VARCHAR(50))
AS
declare variable ol_id_prior decimal(18,0);
declare variable bl_doc_no_str_prior varchar(50);
begin
  ol_kol_plus = 0;
  ol_kol_minus = 0;
  ol_id_prior = 0;
  bl_doc_no_str_prior = '';
  if (rep_date_end is null) then
    rep_date_end = rep_date;

  for select ol_id, ol_doc_no, ol_doc_no_str, ol_client_name, ol_summa_plus, ol_summa_minus, ol_account,
             ol_saldo_begin_dt, ol_saldo_begin_kt, ol_saldo_end_dt, ol_saldo_end_kt, ol_kind, corr_acc, ol_branch_name
      from aarep_cash_journal_tmp_eps(:rep_region_id, :rep_date, :rep_date_end)
      where (ol_summa_plus <> 0 or ol_summa_minus <> 0)
      order by ol_doc_no_str, ol_id
      into :ol_id, :ol_doc_no, :ol_doc_no_str, :ol_client_name, :ol_summa_plus, :ol_summa_minus, :ol_account,
           :ol_saldo_begin_dt, :ol_saldo_begin_kt, :ol_saldo_end_dt, :ol_saldo_end_kt, :ol_kind, :corr_acc,
           :ol_branch_name
  do
  begin
    if (ol_doc_no_str is null) then
      ol_doc_no_str = ol_doc_no;

    if (ol_summa_plus > 0 and (ol_id_prior <> ol_id or bl_doc_no_str_prior <> ol_doc_no_str)) then
      ol_kol_plus = ol_kol_plus + 1;

    if (ol_summa_minus > 0 and (ol_id_prior <> ol_id or bl_doc_no_str_prior <> ol_doc_no_str)) then
      ol_kol_minus = ol_kol_minus + 1;


    if (ol_id_prior = ol_id and bl_doc_no_str_prior = ol_doc_no_str) then
      ol_client_name = '';

    suspend;

    bl_doc_no_str_prior = ol_doc_no_str;
    ol_id_prior = ol_id;
  end
end
