CREATE PROCEDURE KANSLIST_LIST_TMP(
    KSO_ID INTEGER)
RETURNS (
    KSL_ID INTEGER,
    KSL_FROM_KSM_ID INTEGER,
    KSL_NAME VARCHAR(100),
    KSL_MEMO VARCHAR(255),
    KSL_IS_ACTIVE INTEGER,
    KSL_YEDIZ VARCHAR(20),
    KSL_KOLICH DECIMAL(18,4),
    KSM_NAME VARCHAR(255))
AS
begin
  ksl_kolich = 0;
  for select ksl_id, ksl_from_ksm_id, ksl_name, ksl_memo, ksl_is_active, ksl_yediz, ksm_name
      from kanslist, kansmain
      where ksl_from_ksm_id = ksm_id and
            ksl_is_active = 1
      into :ksl_id, :ksl_from_ksm_id, :ksl_name, :ksl_memo, :ksl_is_active, :ksl_yediz, :ksm_name
  do
  begin
    suspend;
  end
  if (kso_id > 0) then
  begin
    for select kol_from_ksl_id, kol_kolich, ksl_from_ksm_id, ksl_name, ksl_memo, ksl_is_active, ksl_yediz, ksm_name
        from kanslist, kansmain, kansorlist
        where ksl_from_ksm_id = ksm_id and
              kol_from_ksl_id = ksl_id and
              kol_kolich <> 0 and
              kol_from_kso_id = :kso_id
        into :ksl_id, :ksl_kolich, :ksl_from_ksm_id, :ksl_name, :ksl_memo, :ksl_is_active, :ksl_yediz, :ksm_name
    do
    begin
      suspend;
    end
  end
end
