CREATE PROCEDURE HP_CALC_DATETIME(
    DT D_DATE,
    TM D_STR_10 = '')
RETURNS (
    RES D_DATETIME)
AS
declare variable tm0 d_str_10;
begin
  res = null;
  tm0 = '00:00:00';
  if (dt is not null) then
  begin
    if (coalesce(tm, '') = '') then
      tm = tm0;
    res = dateadd(datediff(second, cast(tm0 as d_time), cast(tm as d_time)) second to cast(dt as d_datetime));
  end
  suspend;
end
