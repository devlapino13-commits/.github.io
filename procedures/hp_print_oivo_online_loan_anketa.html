CREATE PROCEDURE HP_PRINT_OIVO_ONLINE_LOAN_ANKETA(
    C_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    QUESTION_CODE D_STR_255,
    QUESTION_NAME D_STR_255,
    IS_FILLED D_SMALLINT,
    STR_VALUE D_STR_2048)
AS
declare variable question_id d_integer;
begin
    rep_date = coalesce(:rep_date, current_date);

     for
      select q.question_code, rd.rd_name, rd.rd_id
      from (
          select 'AnketaQuestions.RegistrationAddress ' question_code from rdb$database
          union all
          select 'AnketaQuestions.FactAddress' from rdb$database
          union all
          select 'work_boss_phone_no' from rdb$database
          union all
          select 'work_monthly_income' from rdb$database
          union all
          select 'AnketaQuestions.CommitmentsSumMonth' from rdb$database
          union all
          select 'AnketaQuestions.WorkPlaceAddress' from rdb$database
          union all
          select 'work_job_name' from rdb$database
          union all
          select 'work_place_name' from rdb$database
    ) q
      inner join refdata rd on rd.rd_code = q.question_code
    into
      :question_code, :question_name, :question_id
    do
    begin
        is_filled = 0;
        str_value = '';
        
        select first 1
          ovr.ovr_value
        from objectprops opr
          inner join objectvalues ovs on ovs.ovs_from_opr_id = opr.opr_id
          inner join objectvalres ovr on ovr.ovr_from_opr_id = opr.opr_id and
                                         ovr.ovr_from_ovs_id = ovs.ovs_id
          inner join refdata rd_res on rd_res.rd_id = ovr.ovr_result_rd_id
        where opr.opr_object_kind = 0 and
              opr.opr_object_id = :c_id and
              opr.opr_subject_kind = -1 and
              opr.opr_subject_id = :question_id and
              ovs.ovs_date_begin <= :rep_date and
              ovs.ovs_date_end >= :rep_date and
              rd_res.rd_code = 'Results.StrValue' and
              coalesce(trim(ovr.ovr_value), '') <> ''
        order by ovs.ovs_id desc
        into :str_value;
        
        if (coalesce(trim(:str_value), '') <> '') then
          is_filled = 1;
        
        suspend;
    end
end
