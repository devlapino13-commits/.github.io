Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_OPERDELKONT_DELETE(
    IN_ODK_ID D_INTEGER)
AS
declare variable is_done d_bool;
declare variable is_main d_bool;
begin
  select iif(m.odm_state = 2, 1, 0), iif(k.odk_loan_ol_id is null, 0, 1)
  from operdelkont k, operdelmain m
  where k.odk_id = :in_odk_id and
        k.odk_from_odm_id = m.odm_id
  into is_done, is_main;

  if (is_done = 1) then
    exception e_message 'Заявка уже обработана';

  if (is_main = 1) then
    exception e_message 'Основную операцию нельзя удалять.' || ascii_char(13) ||
                        'Вместо этого можете аннулировать заявку';

  delete from operdelkont
  where odk_id = :in_odk_id;
end
