CREATE PROCEDURE FIND_US_CLIENTS(
    FIND_ID D_STR_CODE,
    FIND_TEXT D_STR_255,
    ADD_PARAMS D_STR_8100,
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_ID TYPE OF D_ID,
    RES_CODE TYPE OF D_STR_255,
    RES_TEXT TYPE OF D_STR_255,
    RES_INFO D_STR_100,
    C_STATUS D_STR_100)
AS
declare variable rows_count_max integer;
declare variable isfound integer;
begin
  rows_count_max = 100; -- Максимальное количество возвращаемых строк

  res_code = '';
  res_text = '';
  find_text = coalesce(find_text, '');
  if (add_params is null) then add_params = '';

  if (coalesce(find_id, '') <> '' and find_id <> '0') then
  begin
    for select c_id, iif(c_type = 0, f_datetostr(c_birth_date), coalesce(c_inn, ''))
        from clients
        where c_id = :find_id
        into res_id, res_info
    do
    begin
      select c_status
      from hp_find_client_status(:res_id)
      into :c_status;
      if (c_status = 'Кредит') then c_status = 'Клиент';

      res_code = f_objname('clients', find_id);

      res_text = res_code;
      if (res_info <> '') then
        res_text = res_text || ' (' || res_info || ')';
      suspend;
    end

    Exit;
  end

  isfound = 0;
  for select distinct fc_id
      from fm_find_client_tmp(:find_text)
      rows :rows_count_max
      into :res_id
  do
  begin
    isfound = isfound + 1;
    for select iif(c_type = 0, f_datetostr(c_birth_date), coalesce(c_inn, ''))
        from clients
        where c_id = :res_id
        into res_info
    do
    begin
      select c_status
      from hp_find_client_status(:res_id)
      into :c_status;
      if (c_status = 'Кредит') then c_status = 'Клиент';

      res_code = f_objname('clients', res_id);

      res_text = res_code;
      if (res_info <> '') then
        res_text = res_text || ' (' || res_info || ')';
      suspend;
    end
  end
  if (isfound > 0) then
    exit;

  if (find_text <> '') then
  begin
    -- Поиск клиентов через fiolist
    for select distinct c_id, iif(c_type = 0, f_datetostr(c_birth_date), coalesce(c_inn, ''))
        from clients, hp_find_from_namelist(0, 0, :find_text)
        where c_id = object_id
--              c_type = 0 -- Берем только физ. лиц
        order by res_priority, c_name_fio
        rows :rows_count_max
        into res_id, res_info
    do
    begin
      select c_status
      from hp_find_client_status(:res_id)
      into :c_status;
      if (c_status = 'Кредит') then c_status = 'Клиент';

      res_code = f_objname('clients', res_id);

      res_text = res_code;
      if (res_info <> '') then
        res_text = res_text || ' (' || res_info || ')';
      suspend;
    end
  end
  else
  begin
    -- Если строка поиска пустая, выдаем список клиентов офиса/ов пользователя
    -- у которых есть заявка, и нет фото
    for with last_clients
        as (select cul_from_c_id, max(cul_datetime) as cul_datetime
            from clientlast
            where cul_from_un_id = :un_id
            group by cul_from_c_id
            order by 2 desc)
        select c_id, iif(c_type = 0, f_datetostr(c_birth_date), coalesce(c_inn, ''))
        from clients, last_clients
        where c_id = cul_from_c_id
        order by cul_datetime desc
        rows :rows_count_max
        into res_id, res_info
    do
    begin
      select c_status
      from hp_find_client_status(:res_id)
      into :c_status;
      if (c_status = 'Кредит') then c_status = 'Клиент';

      res_code = f_objname('clients', res_id);

      res_text = res_code;
      if (res_info <> '') then
        res_text = res_text || ' (' || res_info || ')';
      suspend;
    end
  end
end
