CREATE PROCEDURE GET_MAX_OFFICER_CODE_NO(
    CODE VARCHAR(20))
RETURNS (
    MAX_NO INTEGER)
AS
begin
  /* Procedure Text */
  select kal_new_no
  from knaktlist
  where kal_new_officer_code = :code and
        is_delete = 0
  order by kal_new_officer_code desc, kal_new_no desc
  rows 1
  into :max_no;
  if (max_no is null) then
    max_no = 0;
  suspend;
end
