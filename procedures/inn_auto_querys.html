CREATE PROCEDURE INN_AUTO_QUERYS(
    C_ID D_ID,
    UN_ID D_ID,
    CREATION_STAGE D_STR_CODE)
AS
declare variable fname d_str_code;
declare variable sname d_str_code;
declare variable patronimic d_str_code;
declare variable doc_no d_str_20;
declare variable pass_number d_str_code;
declare variable pass_seria d_str_10;
declare variable doc_date d_date;
declare variable birth_date d_date;
declare variable doc_type d_smallint;
declare variable eic_id d_id;
declare variable eil_id d_id;
declare variable counter d_integer;
declare variable inn_ru_source d_integer;
declare variable cfg d_str_20;
begin
  counter = 0;
  inn_ru_source = f_get_const('inn_ru_source');

  -- таблица с вариативным перебором
  for select result_fname,
             result_sname,
             result_patronimic,
             result_pass_number,
             result_pass_seria,
             result_doc_no,
             result_doc_date,
             result_birth_date,
             result_doc_type
      from inn_check_queries_populate(:c_id)
      into fname,
           sname,
           patronimic,
           pass_number,
           pass_seria,
           doc_no,
           doc_date,
           birth_date,
           doc_type
  do
  begin
    -- осуществлялся ли уже запрос инн с данными клиентскими данными сегодня
    if (exists(select *
               from inn_check_queries icq
               inner join extinfolist eil on eil.eil_id = icq.icq_from_eil_id
               where icq.icq_from_c_id = :c_id and
                     icq.icq_fname = :fname and
                     icq.icq_sname = :sname and
                     icq.icq_patronimic = :patronimic and
                     icq.icq_doc_no = :doc_no and
                     eil.eil_date = current_date and
                     eil_status >= 0))
    then
      continue;

    -- если это гражданин рф то пропускаем
    if (doc_type <> 10) then
      continue;

    if (inn_ru_source > 0) then
    begin
      for select eic_code, eic_id
          from extinfocfg
          where (:inn_ru_source in (1,3) and eic_code = 'inn_smev')
            or (:inn_ru_source in (2,3) and eic_code = 'inn_gnivc')
          into cfg, eic_id
      do
      begin
        -- создается запрос на получение инн
        execute procedure hp_extinfo_insert(:cfg, 'clients', :c_id, 0, :un_id);

        select eil.eil_id
        from extinfolist eil
        where eil.eil_from_c_id = :c_id and
              eil.eil_from_eic_id = :eic_id
        order by eil.eil_id desc rows 1
        into :eil_id;

        -- создаю дочернюю запись
        insert into inn_check_queries (icq_from_c_id, icq_fname, icq_sname, icq_patronimic, icq_datebirth, icq_doc_type,
                                       icq_doc_no, icq_doc_date, icq_creation_stage, icq_status, icq_inn_received,
                                       icq_from_eil_id, icq_pass_seria, icq_pass_number,
                                       icq_info_source)
        values (:c_id, :fname, :sname, :patronimic, :birth_date, :doc_type, :doc_no, :doc_date,
                f_enum_id('inn_request_source', :creation_stage), 0, 0, :eil_id, :pass_seria,
                :pass_number, iif(:cfg = 'inn_gnivc', 0, 1));

        counter = counter + 1;
      end
    end
  end

  if (counter > 0) then
  begin
    for select eic_code
    from extinfocfg
    where (:inn_ru_source in (1,3) and eic_code = 'inn_smev')
      or (:inn_ru_source in (2,3) and eic_code = 'inn_gnivc')
    into cfg
    do
      post_event :cfg;
  end
end
