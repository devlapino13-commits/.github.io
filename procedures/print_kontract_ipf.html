CREATE PROCEDURE PRINT_KONTRACT_IPF(
    PRINT_K_ID DECIMAL(18,0))
RETURNS (
    PREPAYMENT_SUMMA DECIMAL(18,4),
    PROJECT_DESCRIPTION VARCHAR(2000),
    OBJECT_QTY INTEGER,
    PROJECT_FINISH_DATE DATE,
    PR_IPF_COMMISSION CURR_TYPE,
    K_SUM_DISB_PREPAY CURR_TYPE,
    NO_FEES D_INTEGER,
    TOTAL_SUMM CURR_TYPE,
    TOTAL_PROFIT CURR_TYPE)
AS
declare variable c_id decimal(15,2);
begin
  total_summ = 0;
  for select coalesce(k.k_prepayment_summa, 0), coalesce(mif.mai_project_description, ''),
             coalesce(mif.mai_object_qty, 0), mif.mai_project_finish_date, coalesce(pr.pr_ipf_commission, 0),
             k.k_amount_approved
      from kontract k
      inner join knclient kc on kc.kc_from_k_id = k.k_id
      inner join product pr on pr.pr_id = k.k_from_pr_id
      left join miniapp_ipf mif on mif.mai_from_ma_id = k.k_from_ma_app_id
      where k.k_id = :print_k_id
      into :prepayment_summa, :project_description, :object_qty, :project_finish_date, :pr_ipf_commission,
           :k_sum_disb_prepay
  do
  begin

    select round((sum(kml_total) + sum(kml_sell_nsp_all) + sum(kml_sell_profit_all)), 0),
           round((sum(kml_total) + sum(kml_sell_profit_all)), 0)
    from print_kontmat(:print_k_id, null)
    into :total_summ, total_profit;

    /*
    Выращивание кормовых культур и их семян
    Оптовая торговля прочим топливом, смазочными материалами, техническими маслами и т.п.
    Оптовая торговля химическими продуктами
    Оптовая торговля химическими продуктами (Удобрения)
    Оптовая торговля химическими продуктами (Убодрения1)
    */
    if (exists(select *
               from kontmat
               inner join foreign_econom_goods_ref on feg_id = kml_from_feg_id
               inner join refdata on rd_id = feg_from_rd_id and
                     (rd_code starting '46.75.0' or rd_code starting '01.19.1' or rd_code starting '46.71.9')
               where kml_object_kind = 9 and
                     kml_object_id = :print_k_id)) then
      no_fees = 1;
    else
      no_fees = 0;

    suspend;
  end
end
