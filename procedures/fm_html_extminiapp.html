    OBJ_KIND VARCHAR(100),

    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)

  into lo_id;

         f_source_str(tkt.tkt_source), coalesce(tkt.tkt_is_closed, 0), ema.ema_status, ema.ema_office_id,
         iif(reg.reg_logic_label containing '.mini_office.', 1, 0)

  inner join region reg on reg.reg_unicode = ema.ema_office_id
  where ema.ema_id = :obj_id
  into :app_date, :cl_fio, :loan_amount, :loan_period, :tkt_id, :c_id, :ema_lo_id, :tkt_source, :tkt_is_closed,
       ema_status, ema_office_id, is_mini_office;

  if (:ema_office_id not in (select reg_id
                             from fm_user_regions(:un_id))) then
    
  begin

    res_text = blob_append(res_text, '</body>');
    suspend;
    exit;
  end

  if (tkt_id is null) then -- если заявка не найдена, тогда выход.

Свяжитесь с технической поддердкой', 'не найдена. Свяжитесь с технической поддердкой')));

    res_text = blob_append(res_text, '</body>');
    suspend;
    exit;
  end

  if ((:jk_system_code <> 'JobSystem.Region_KM') or (is_mini_office = 1)) then
  begin
    if (lo_id <> ema_lo_id) then
    begin
      if (db_id = 81001 and has_access = 1) then
        begin
          res_text = BLOB_APPEND(res_text, f_uri_sql(f_lang(lang, 'Вернуть заявку в общую очередь'),
          'return_appl_public', 'extminiapp', :obj_id, '',
          f_lang(lang, 'Вернуть заявку в общую очередь?'), '', 0, 0, '', '',
          'dialog', '')|| '<br><br>');
        end

      res_text = blob_append(res_text, iif(:db_id <> 94401, 'Заявка закреплена за другим пользователем', 'Заявка закреплена за другим пользователем'));
      res_text = blob_append(res_text, '</body>');
      suspend;
      exit;
    end
  end

  if (ema_status <> 0) then
  begin
    if (ema_status = 1) then
    begin
      res_text = blob_append(res_text, iif(:db_id <> 94401, 'Заявка успешно обработана', 'Заявка успешно обработана'));
      res_text = blob_append(res_text, '</body>');
      suspend;
      exit;
    end
    else
    if (ema_status = -1) then
    begin
      res_text = blob_append(res_text, iif(:db_id <> 94401, 'Отказ успешно сохранен', 'Отказ успешно сохранен'));
      res_text = blob_append(res_text, '</body>');
      suspend;
      exit;
    end
  end

  if (tkt_is_closed = 1) then
  begin
    res_text = blob_append(res_text, iif(:db_id <> 94401, 'Работа по данной заявке завершена', 'Работа по данной заявке завершена'));
    res_text = blob_append(res_text, '</body>');
    suspend;
    exit;
  end

  res_text = blob_append(res_text, '<h1 class="green_text">' || :cl_fio || '</h1>');

  if ((:jk_system_code <> 'JobSystem.Region_KM') or (is_mini_office = 1)) then  -- команды для ВИПС
  begin
    if (ema_lo_id is null) then -- если заявка в общей очереди
    begin
      res_text = blob_append(res_text, f_uri_sql(f_lang(lang, 'Забрать заявку'),
        'extminiapp_take', 'extminiapp', :obj_id, 'none',
        f_lang(lang, 'Вы действительно хотите забрать заявку?'), '', 0, 0,
        '', '', 'dialog', '') || '<br><br>');
    end
    else
    begin
      if (db_id = 81001) then
      begin
        if (has_access = 1) then
        begin

          res_text = blob_append(res_text,
          f_uri_open(f_lang(lang, 'Открыть окно клиента'),
          'client', 'Clients', c_id, left(cl_fio, 50), '', un_id) || '<br><br>' ||
          f_uri_open(f_lang(lang, 'Зарегистрировать заявку на кредит'), 'loanapp',
          'extminiapp', obj_id, '', '', un_id) || '<br><br>' || 'Отказать:' || '<br>' ||
          '<table>' ||
              '<tbody>' ||
                  '<tr>' ||
                      '<td>' ||
                          f_uri_sql('Отказать - Недостоверная информация', 'extminiapp_refuse_1', 'extminiapp',

                          '', 'dialog', '') ||
                      '</td>' ||
                  '</tr>' ||
                  '<tr>' ||
                      '<td>' ||
                          f_uri_sql('Отказать - Сомнения в платежеспособности клиента', 'extminiapp_refuse_2', 'extminiapp',
                          obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                          '', 'dialog', '') ||
                      '</td>' ||
                  '</tr>' ||
                  '<tr>' ||
                      '<td>' ||
                          f_uri_sql('Отказать - Кредитная история клиента', 'extminiapp_refuse_3', 'extminiapp',
                          obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                          '', 'dialog', '') ||
                      '</td>' ||
                  '</tr>' ||
                  '<tr>' ||
                      '<td>' ||
                          f_uri_sql('Отказать - Негативная информация о клиенте', 'extminiapp_refuse_4', 'extminiapp',
                          obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                          '', 'dialog', '') ||
                      '</td>' ||

                  '<tr>' ||
                      '<td>' ||
                          f_uri_sql('Отказать - Агрессивное поведение клиента', 'extminiapp_refuse_5', 'extminiapp',
                          obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                          '', 'dialog', '') ||

                  '</tr>' ||
                  '<tr>' ||
                      '<td>' ||
                          f_uri_sql('Отказать - Отказ по инициативе клиента', 'extminiapp_refuse_6', 'extminiapp',
                          obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                          '', 'dialog', '') ||
                      '</td>' ||
                  '</tr>' ||
                  '<tr>' ||
                      '<td>' ||
                          f_uri_sql('Отказать - Документы клиента не соответствуют требованиям', 'extminiapp_refuse_7', 'extminiapp',
                          obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                          '', 'dialog', '') ||
                      '</td>' ||
                  '</tr>' ||
              '</tbody>' ||
          '</table>' ||
          '<br><br>');
        end
        if (has_access <> 1) then
            res_text = blob_append(res_text,
            f_uri_open(f_lang(lang, 'Открыть окно клиента'),
            'client', 'Clients', c_id, left(cl_fio, 50), '', un_id) || '<br><br>' ||
            f_uri_open(f_lang(lang, 'Зарегистрировать заявку на кредит'), 'loanapp',
            'extminiapp', obj_id, '', '', un_id) || '<br><br>');
      end
      else
      begin
      res_text = blob_append(res_text,
        f_uri_open(f_lang(lang, 'Открыть окно клиента'),
        'client', 'Clients', c_id, left(cl_fio, 50), '', un_id) || '<br><br>' ||
        f_uri_open(f_lang(lang, 'Зарегистрировать заявку на кредит'), 'loanapp',
        'extminiapp', obj_id, '', '', un_id) || '<br><br>' ||
        f_uri_sql(f_lang(lang,'Отказать'), 'extminiapp_refuse', 'extminiapp',
        obj_id, 'combo', f_lang(lang, 'Выберите причину отказа'), '', 0, 0, '',
        '', 'dialog', '') || '<br><br>');
      end
    end
  end

  if (:jk_system_code = 'JobSystem.Region_KM') then                      -- команды для МО
  begin
    if (ema_lo_id is not null) then -- если заявка в общей очереди
    begin
          res_text = blob_append(res_text, f_uri_sql(f_lang(lang, 'Вернуть заявку в общую очередь'),
          'return_appl_public', 'extminiapp', :obj_id, '',
          f_lang(lang, 'Вернуть заявку в общую очередь?'), '', 0, 0, '', '',
          'dialog', '')|| '<br>');
    end
    else
    begin
      if (db_id = 81001) then   --кнопка "Отказать" добавлено по запросу от Зары. jira 4823
      begin
          res_text = blob_append(res_text, f_uri_sql(f_lang(lang, 'Передать в другой офис'),
          'transfer_office', 'extminiapp', :obj_id, 'combo',
          f_lang(lang, 'Выберите офис'), '', 0, 0, '', '', 'dialog', '') ||
          '<br><br>' ||
          f_uri_sql(f_lang(lang, 'Закрепить заявку за ВИПСом'),
          'transfer_svz', 'extminiapp', :obj_id, 'combo',
          f_lang(lang, 'Выберите ВИПС'), '', 0, 0, '', '', 'dialog', '') ||

          '<br><br>' || 'Отказать:' || '<br>' ||
            '<table>' ||
                '<tbody>' ||
                    '<tr>' ||
                        '<td>' ||
                            f_uri_sql('Отказать - Недостоверная информация', 'extminiapp_refuse_1', 'extminiapp',
                            obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                            '', 'dialog', '') ||
                        '</td>' ||
                    '</tr>' ||
                    '<tr>' ||
                        '<td>' ||
                            f_uri_sql('Отказать - Сомнения в платежеспособности клиента', 'extminiapp_refuse_2', 'extminiapp',
                            obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                            '', 'dialog', '') ||
                        '</td>' ||
                    '</tr>' ||
                    '<tr>' ||
                        '<td>' ||
                            f_uri_sql('Отказать - Кредитная история клиента', 'extminiapp_refuse_3', 'extminiapp',
                            obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                            '', 'dialog', '') ||
                        '</td>' ||
                    '</tr>' ||
                    '<tr>' ||
                        '<td>' ||
                            f_uri_sql('Отказать - Негативная информация о клиенте', 'extminiapp_refuse_4', 'extminiapp',
                            obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                            '', 'dialog', '') ||
                        '</td>' ||
                    '</tr>' ||
                    '<tr>' ||
                        '<td>' ||
                            f_uri_sql('Отказать - Агрессивное поведение клиента', 'extminiapp_refuse_5', 'extminiapp',
                            obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                            '', 'dialog', '') ||
                        '</td>' ||
                    '</tr>' ||
                    '<tr>' ||
                        '<td>' ||
                            f_uri_sql('Отказать - Отказ по инициативе клиента', 'extminiapp_refuse_6', 'extminiapp',
                            obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                            '', 'dialog', '') ||
                        '</td>' ||
                    '</tr>' ||
                    '<tr>' ||
                        '<td>' ||
                            f_uri_sql('Отказать - Документы клиента не соответствуют требованиям', 'extminiapp_refuse_7', 'extminiapp',
                            obj_id, 'combo', 'Выберите причину отказа', '', 0, 0, '',
                            '', 'dialog', '') ||
                        '</td>' ||
                    '</tr>' ||
                '</tbody>' ||
            '</table>' ||
            '<br><br>');
      end
      else
      begin
        res_text = blob_append(res_text, f_uri_sql(f_lang(lang, 'Передать в другой офис'),
        'transfer_office', 'extminiapp', :obj_id, 'combo',
        f_lang(lang, 'Выберите офис'), '', 0, 0, '', '', 'dialog', '') ||
        '<br><br>'|| f_uri_sql(f_lang(lang,'Отказать'), 'extminiapp_refuse', 'extminiapp',
        obj_id, 'combo', f_lang(lang, 'Выберите причину отказа'), '', 0, 0, '',
        '', 'dialog', '') || '<br><br>');
      end
    end
  end
  res_text = blob_append(res_text,
    f_format('%1: <b>%2</b><br>', f_lang(lang, 'Источник заявки'), tkt_source) ||
    f_format('%1: <b>%2</b><br>', f_lang(lang, 'Дата заявки'), app_date) ||
    f_format('%1: <b>%2%3</b><br>', f_lang(lang, 'Сумма займа'), :loan_amount, :curr) ||
    f_format('%1: <b>%2 мес.</b><br>', f_lang(lang, 'Срок'), :loan_period));

  cell_no = '';  

  if (ema_lo_id is not null) then  
  begin  
    cell_no = f_cont_info('clients', c_id, 'cell_main', 'last');  

    -- Если не удалось получить номер клиента из БД, берем его номер телефона из mc_login
    if (db_id = 94401 and cell_no = '') then
    begin
      select first 1 mcl_login  
      from mc_login  
      where mcl_from_c_id = :c_id  
        and mcl_is_delete = 0
      order by mcl_datetime desc  
      into :cell_no;
    end
  end

  if (ema_lo_id is null) then
    cell_no = f_format('<b>%1</b>', f_lang(lang, 'Будет доступ после того, как заберете заявку'));
  else
    cell_no = iif(cell_no = '', f_format('<b>%1</b>', f_lang(lang, 'Не указан')),
      f_uri_phone(cell_no, 0, c_id));
  res_text = blob_append(res_text, f_format('%1: %2<br>', f_lang(lang, 'Телефонный номер'), cell_no));

  --------- Блок КЦ ----------->>
  cc_inform = 0;

  for select coalesce(client_in_company_status.rd_name, iif(:db_id <> 94401, 'Не заполнен', 'Не заполнен')),
             coalesce(language.rd_name, iif(:db_id <> 94401, 'Не заполнен', 'Не заполнен')),
             coalesce(wma.wma_phone_no, ''), coalesce(wmt.wmat_phone_add, ''),
             coalesce(citizenship.rd_name, iif(:db_id <> 94401, 'Не заполнен', 'Не заполнен')),
             coalesce(wmt.wmat_city, ''), coalesce(wmt.wmat_pass_no, ''), f_datetostr(wmt.wmat_birthday),
             coalesce(wmt.wmat_doclist, ''),
             coalesce(work_status.rd_name, iif(:db_id <> 94401, 'Не заполнен', 'Не заполнен')),
             (
             case wmt.wmat_current_loan_mbf
               when 0 then iif(:db_id <> 94401, 'Нет', 'Нет')
               when 1 then iif(:db_id <> 94401, 'Есть', 'Есть')
               else ''
             end),
             (
             case wmt.wmat_active_loan_othe_company
               when 0 then iif(:db_id <> 94401, 'Нет', 'Нет')
               when 1 then iif(:db_id <> 94401, 'Есть', 'Есть')
               else ''
             end),
             coalesce(loan_product.rd_name, iif(:db_id <> 94401, 'Не заполнен', 'Не заполнен')),
             coalesce(meeting_point.rd_name, iif(:db_id <> 94401, 'Не заполнен', 'Не заполнен')),
             coalesce(wmt.wmat_meeting_address, ''), f_datetostr(wmt.wmat_meeting_date),
             coalesce(wmt.wmat_anketa_memo, '')
      from cc_website_miniapp wma
      inner join cc_website_miniapp_tmp wmt on wmt.wmat_id = wma.wma_last_wmat_id
      left join refdata language on language.rd_id = wmt.wmat_language
      left join refdata client_in_company_status on client_in_company_status.rd_id = wmt.wmat_client_in_company_status
      left join refdata citizenship on citizenship.rd_id = wmt.wmat_citizenship
      left join refdata work_status on work_status.rd_id = wmt.wmat_work_status
      left join refdata loan_product on loan_product.rd_id = wmt.wmat_loan_product
      left join refdata meeting_point on meeting_point.rd_id = wmt.wmat_meeting_point
      where wma.wma_from_tkt_id = :tkt_id
      into :cc_client_in_company_status, :cc_language, :cc_phone_no, :cc_phone_add, :cc_citizenship, :cc_city,
           :cc_pass_no, :cc_birthday, :cc_doclist, :cc_work_status, :cc_mbf_active_loan, :cc_other_active_loan,
           :cc_loan_product, :cc_meeting_point, :cc_meeting_address, :cc_meeting_date, :cc_anketa_memo
  do
  begin
    if (cc_inform = 0) then
    begin
      cc_inform = 1;
      res_text = blob_append(res_text, f_format('<br><h2>%1</h2>', iif(:db_id <> 94401, 'Информация от колл-центра', 'Информация от колл-центра')));
    end

    if (cc_client_in_company_status <> iif(:db_id <> 94401, 'Не заполнен', 'Не заполнен')) then
      res_text = blob_append(res_text, f_format('%1: <b>%2</b><br>', iif(:db_id <> 94401, 'Клиент компании', 'Клиент компании'), :cc_client_in_company_status));

    if (cc_language <> iif(:db_id <> 94401, 'Не заполнен', 'Не заполнен')) then
      res_text = blob_append(res_text, f_format('%1: <b>%2</b><br>', iif(:db_id <> 94401, 'Язык общения', 'Язык общения'), :cc_language));

    if (cc_phone_no <> '') then
    begin
      if (ema_lo_id is not null) then
