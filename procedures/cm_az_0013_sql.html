CREATE PROCEDURE CM_AZ_0013_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id ID_TYPE;
declare variable C_NAME_FIO str_100;
declare variable C_COMMENT str_255;
declare variable cl_fio str_100;
begin
    -- проверка под ФТ на уровне зявки
   if(ma_id in(113305, 96915, 73682)) then exit;
    for select macl.mac_from_c_id
    from miniappclient macl
    where macl.mac_from_ma_id = :ma_id
    and ((macl.mac_amount>0) or (macl.mac_is_poruch = 1))
    into :c_id
    do
    begin
       for select C_NAME_FIO, C_COMMENT
       from AAREP_CHECK_CLIENT_FIO(:C_ID, '', '', '', 0, null, '', 1)
       where C_COMMENT_ID = 1
       into :C_NAME_FIO, :C_COMMENT
       do
       begin
            select cl.c_name_fio
            from clients cl
            where cl.c_id = :c_id
            into :res_info_text;
            if (c_comment='РФМ (СФР): ФИО') then
                res_error_is_critical = 1;
            else
                res_error_is_critical = 1;
            res_error_text='Совпадение с лицом по '||C_COMMENT||' '||C_NAME_FIO;
            suspend;
       end
    end
end
