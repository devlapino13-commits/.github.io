CREATE PROCEDURE MC_CLEAR_BLOCKS(
    PHONE D_STR_CODE,
    IP_ADDR VARCHAR(50) = null)
RETURNS (
    RES_TEXT D_STR_LONG)
AS
declare variable mcc_id d_id;
begin
  res_text = '';
  if (left(phone, 1)= '+') then
    phone = substring(phone from 2);
  if (left(phone, 1) = '8') then
    phone = overlay(phone placing '7' from 1);
  phone = replace(phone, ' ', '');
  phone = replace(phone, '-', '');
  phone = replace(phone, '(', '');
  phone = replace(phone, ')', '');
  if (char_length(phone) <> 11) then
    res_text = 'Некорректный формат номера телефона';
  if (res_text = '') then
  begin
    delete from msgquotalist
    where mql_cont_code = :phone
      and mql_from_mqc_id in (6, 7);
    if (row_count <> 0) then
      res_text = 'квота номера';

    if (ip_addr is not null) then
      delete from msgquotalist
      where mql_cont_code = :ip_addr
        and mql_from_mqc_id in (6, 7);
      if (row_count <> 0) then
        res_text = res_text || iif(res_text <> '', ',', '') || 'квота ip (вход)';

    ip_addr = null;

    for select mcc_id, mcc_last_ip
    from mc_check
    where mcc_login = :phone
    into mcc_id, ip_addr
    do
    begin
      if (mcc_id is not null) then
      begin
        update mc_check
        set
          mcc_login_try_cnt = 0,
          mcc_send_code_cnt = 0,
          mcc_check_code_cnt = 0
        where mcc_id = :mcc_id
          and mcc_login_try_cnt + mcc_send_code_cnt + mcc_check_code_cnt > 0;
        if (row_count <> 0) then
          res_text = res_text || iif(res_text <> '', ',', '') || 'счетчики проверок';
      end
  
      if (ip_addr is not null) then
      begin
        delete from msgquotalist
        where mql_cont_code = :ip_addr
          and mql_from_mqc_id in (6, 7);
        if (row_count <> 0) then
          res_text = res_text || iif(res_text <> '', ',', '') || 'квота ip (mcc)';
      end
      if (res_text = '') then
        res_text = f_format('Номер %1. Не найден в блоках. Ничего не сделано.', phone);
      else
        res_text = f_format('Номер %1. Сброшено: ', phone) ||res_text;
    end
  end
  suspend;
end
