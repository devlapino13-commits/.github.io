CREATE PROCEDURE FSROBOT_OPERCOMPENS(
    REP_DATE D_DATE,
    REP_K_ID D_ID = null)
AS
declare variable kcc_id d_integer;
declare variable k_id d_id;
declare variable region_id d_integer;
declare variable dt_acc_base d_str_100;
declare variable perc_and_nsp d_currency;
declare variable c_id d_id;
declare variable al_id d_str_100;
declare variable ol_id d_id;
begin
    for select kcc_id,
               k_id,
               c_id,
               kal_new_reg_id,
               iif(pr_code like '%.ipf.%', '11531/202', '11531/201'),
               cli_perc_dolia_corrected
    from fsrobot_opercompens_tmp(:rep_date, 2, :rep_k_id)
    where ((k_id = :rep_k_id) or (:rep_k_id is null))
    into :kcc_id,
         :k_id,
         :c_id,
         :region_id,
         :dt_acc_base,
         :perc_and_nsp
    do
    begin
      -- если операция ранее уже была создана, тогда считаю что робот отработал и новую не создаю.
      if (exists(select *
                 from compens_robot_log t
                 inner join operlist ol on t.crl_from_ol_id = ol.ol_id
                 where ol.ol_date = :rep_date
                 and ol.is_delete = 0
                 and t.crl_from_kcc_id = :kcc_id
             )) then
                continue;

      al_id = null;

      select al_id
      from find_acc_list(:dt_acc_base)
      where al_from_reg_id = :region_id
      into :al_id;

      if (al_id is null) then
        exception e_default 'kcc_id = '||kcc_id||'; region_id = '||:region_id||'; al_id is null';

      ol_id = null;

      select ol_id
      from  hp_do_repayment (
          :k_id,     -- k_id
          0,                      -- c_id
          :rep_date,           -- rep_date
          :perc_and_nsp,                   -- repay_amount
          0,                      -- repay_to_os
          1,                      -- repay_to_perc
          0,                      -- repay_to_fine
          0,                      -- repay_to_kommis
          1,                      -- need_save
          0,                      -- is_full_repay
          0,                      -- is_writeoff
          :region_id,                -- repay_fact_office_id
          3,                      -- repay_dt_account_kind
          :al_id,   -- repay_dt_account_no
          0,                      -- repay_curr_rate
          1000000000000,          -- repay_user_id
          :c_id,          -- repay_document_client
          'погашение процентов государством',       -- repay_doc_client_name_reason
          0,                      -- repay_manager_c_id
          0,                      -- repay_from_rd_id
          0,                      -- repay_control_amount
          0                       -- need_writeoff_operextra
      )
      into :ol_id;    

      if (coalesce(ol_id, 0) > 0) then
      begin
          insert into compens_robot_log (crl_from_kcc_id, crl_from_ol_id)
          values (:kcc_id, :ol_id);
      end
   end

   if (rep_k_id is null) then
   begin
      delete from fsrobot_eventlist fse
      where fel_command_kind in ('Loan_LO_Repay_OK', 'Loan_Repay_OK', 'Loan_Repay_X', 'Push_Loan_LO_Repay');
   end
end
