CREATE PROCEDURE AAREP_ACTIONLIST_GET_NEXT(
    UN_ID DECIMAL(18,0),
    REP_ACL_ID DECIMAL(18,0),
    REP_STATUS INTEGER,
    REP_IS_OWN INTEGER,
    REP_RD_CODE VARCHAR(255))
RETURNS (
    ACL_ID DECIMAL(18,0),
    ACL_OBJECT_KIND INTEGER,
    ACL_OBJECT_ID DECIMAL(18,0),
    RD_STR_ID VARCHAR(100),
    RD_CODE VARCHAR(255),
    ACL_FROM_REG_ID INTEGER)
AS
declare variable c_id      decimal(18,0);
declare variable rd_filter varchar(255);
begin
  if (rep_rd_code = 'double_clients') then
  begin
    acl_object_kind = f_objkind('clients', 'id');
    acl_id = 0;
    rd_str_id = rep_rd_code;
    rd_code = rep_rd_code;
    acl_from_reg_id = f_hq_id();
    select dc_from_c_id_1
    from doubleclients
    where dc_state_un_id = :un_id and
          dc_state > 0
    order by dc_state
    rows 1
    into acl_object_id;

    if (acl_object_id is null) then
    begin
      select dc_from_c_id_1
      from doubleclients
      where dc_state_un_id is null and
            dc_state > 0
      order by dc_state
      rows 1
      into acl_object_id;

      if (acl_object_id is not null) then
      begin
        update doubleclients
        set dc_state_un_id = :un_id
        where dc_from_c_id_1 = :acl_object_id and
              dc_state_un_id is null and
              dc_state > 0;

        update doubleclients
        set dc_state_un_id = :un_id
        where dc_from_c_id_2 = :acl_object_id and
              dc_state_un_id is null and
              dc_state > 0;
      end
    end

    if (acl_object_id is not null) then
      suspend;

    exit;
  end

  if (rep_is_own = -2 and
      rep_acl_id > 0) then
  begin
    select acl_object_kind, acl_object_id, rd_str_id, rd_code, acl_from_reg_id
    from actionlist, refdata
    where acl_id = :rep_acl_id and
          acl_from_rd_id = rd_id
    into :acl_object_kind, acl_object_id, rd_str_id, rd_code, acl_from_reg_id;

    suspend;
    exit;
  end
  if (rep_rd_code is null) then
    rep_rd_code = '';

  -- Процедура возвращает очередной объект для проверки
  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  rd_filter = '';
  for select distinct rd_str_id
      from refdata
      where rd_main_code = 'ActionGroup' and
            rd_main_code <> rd_code and
            (:rep_rd_code = '' or
            rd_code = :rep_rd_code)
      into :rd_str_id
  do
  begin
    if (exists(select has_access
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'ActionGroup.' || :rd_str_id)
               where has_access = 1)) then
    begin
      rd_filter = rd_filter || '[' || rd_str_id || ']';
    end
  end

  if (rep_acl_id is null) then
    rep_acl_id = 0;
  if (rep_acl_id <> 0) then
  begin
    select aco_id
    from actionorder
    where aco_id = :rep_acl_id
    order by aco_id
    rows 1
    into :acl_id;

    if (acl_id is not null) then
    begin

      if (rep_rd_code starting 'ActionGroup.Passport_Verify') then
      begin
        update actionlist
        set acl_status = 1,
            acl_respons_c_id = :c_id
            --            acl_from_un_id = :un_id
        where acl_id = :acl_id;
      end
      else
      begin
        update actionlist
        set acl_status = 1,
            acl_respons_c_id = :c_id,
            acl_from_un_id = :un_id
        where acl_id = :acl_id;
      end

      select acl_object_kind, acl_object_id, rd_str_id, rd_code, acl_from_reg_id
      from actionlist, refdata
      where acl_id = :acl_id and
            acl_from_rd_id = rd_id
      into :acl_object_kind, acl_object_id, rd_str_id, rd_code, acl_from_reg_id;

      suspend;
    end
    exit;
  end
  if (rep_status is null) then
    rep_status = -1;
  if (rep_is_own is null) then
    rep_is_own = -1;

  if (rep_is_own = -1 or rep_is_own = 1) then
  begin
    if (rep_status = -1 or rep_status = 1) then
    begin
      -- Ищем ранее закрепленный объект данному сотруднику со статусом "В обработке"
      select aco_id
      from actionorder, refdata
      where aco_status = 1 and
            aco_respons_c_id = :c_id and
            aco_from_rd_id = rd_id and
            rd_main_code = 'ActionGroup' and
            :rd_filter like '%[' || rd_str_id || ']%'
            --      order by aco_date, aco_id
      order by aco_orderno, aco_id
      rows 1
      into :acl_id;

      if (acl_id is not null) then
      begin
        select acl_object_kind, acl_object_id, rd_str_id, rd_code, acl_from_reg_id
        from actionlist, refdata
        where acl_id = :acl_id and
              acl_from_rd_id = rd_id
        into :acl_object_kind, acl_object_id, rd_str_id, rd_code, acl_from_reg_id;

        suspend;
        exit;
      end
    end

    if (rep_status = -1 or rep_status = 0) then
    begin
      -- Ищем ранее закрепленный объект со статусом "В ожидании",
      -- такое может быть после отправки в доработку, после доработки
      select aco_id
      from actionorder, refdata
      where aco_status = 0 and
            aco_respons_c_id = :c_id and
            aco_from_rd_id = rd_id and
            rd_main_code = 'ActionGroup' and
            :rd_filter like '%[' || rd_str_id || ']%'
            --      order by aco_date, aco_id
      order by aco_orderno, aco_id
      rows 1
      into :acl_id;

      if (acl_id is not null) then
      begin
        select acl_object_kind, acl_object_id, rd_str_id, rd_code, acl_from_reg_id
        from actionlist, refdata
        where acl_id = :acl_id and
              acl_from_rd_id = rd_id
        into :acl_object_kind, acl_object_id, rd_str_id, rd_code, acl_from_reg_id;

        if (rep_rd_code starting 'ActionGroup.Passport_Verify') then
        begin
          update actionlist
          set acl_status = 1
              --              acl_from_un_id = :un_id
          where acl_id = :acl_id;
        end
        else
        begin
          update actionlist
          set acl_status = 1,
              acl_from_un_id = :un_id
          where acl_id = :acl_id;
        end
        suspend;
        exit;
      end
    end
  end
  if (rep_is_own = -1 or rep_is_own = 0) then
  begin
    -- Ищем объект "В ожидании" без закрепленного исполнителя
    if (rep_rd_code <> 'ActionGroup.Check_Loan_Data_PostCheck') then
      select aco_id
      from actionorder, refdata, user_reg_list(:un_id)
      where aco_respons_c_id is null and
            aco_from_rd_id = rd_id and
            rd_main_code = 'ActionGroup' and
            :rd_filter like '%[' || rd_str_id || ']%' and
            aco_from_reg_id = r_list
            --    order by aco_date, aco_id
      order by aco_orderno, aco_id
      rows 1
      into :acl_id;
    else
      select max(aco_id)
      from actionorder, refdata, user_reg_list(:un_id)
      where aco_respons_c_id is null and
            aco_from_rd_id = rd_id and
            rd_main_code = 'ActionGroup' and
            :rd_filter like '%[' || rd_str_id || ']%' and
            aco_from_reg_id = r_list
      into :acl_id;

    if (acl_id is not null) then
    begin
      select acl_object_kind, acl_object_id, rd_str_id, rd_code, acl_from_reg_id
      from actionlist, refdata
      where acl_id = :acl_id and
            acl_from_rd_id = rd_id
      into :acl_object_kind, acl_object_id, rd_str_id, rd_code, acl_from_reg_id;

      if (rep_rd_code starting 'ActionGroup.Passport_Verify') then
      begin
        update actionlist
        set acl_respons_c_id = :c_id,
            acl_status = 1
            --            acl_from_un_id = :un_id
        where acl_id = :acl_id;
      end
      else
      begin
        update actionlist
        set acl_respons_c_id = :c_id,
            acl_status = 1,
            acl_from_un_id = :un_id
        where acl_id = :acl_id;
      end

      suspend;
      exit;
    end
  end

end
