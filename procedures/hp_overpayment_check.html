CREATE PROCEDURE HP_OVERPAYMENT_CHECK
AS
declare variable rd_id d_integer;
declare variable over_sum d_currency;
declare variable ovm_id d_integer;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable status d_integer;
declare variable memo d_str_255;
declare variable kc_id d_id;
declare variable phone d_str_100;
declare variable start_time timestamp;
begin
  -- процедура производит фильтрацию зафиксированных переплат, для отправки СМС

  start_time = cast('now' as timestamp);
  rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  for select ovm_id, kc_from_c_id, kc_from_k_id, kc_id
  from overpayment_msg ovm, knclient kc
  where ovm_status = 0       -- новые
    and kc_id = ovm_from_kc_id
  into ovm_id, c_id, k_id, kc_id
  do
  begin
    status = 999;  -- конечный статус, если все проверки пройдены
    memo = null;

    -- проверка "не была начислена судебная госпошлина"
    if (status > 0) then
      if (exists(select * from operextra ox
                    where ox.ox_object_kind = 9
                      and ox.ox_object_id = :k_id
                      and ox.ox_ol_is_active = 1
                      and ox.ox_amount_plus > 0
                      and ox.ox_from_rd_id =  :rd_id)
        or
          exists(select * from opersud
                  where osd_from_k_id = :k_id and
                        osd_ol_is_delete = 0))
      then
          status = - 3;

    -- сумма переплаты должна быть не меньше рубля
    over_sum = (select -sum(ok_os)
                from operkred
                where ok_from_k_id = :k_id
                  and ok_ol_is_active = 1
                  and ok_type = 9);

    if (coalesce(over_sum, 0) < 1) then
      status = - 1;

    -- проверка наличия основного номера телефона
    if (status > 0) then
    begin
      phone = (select cci.ok_cell_main from continfo cci
                  where cci.cci_object_id = :c_id
                    and cci.cci_object_kind = 0);

      if (coalesce(phone, '') = '') then
        status = - 2;
    end 

    -- проверка ПОДФТ
    if (status > 0) then
    begin
      memo = (select 'ПОДФТ: Совпадение с лицом по ' || c_comment || ' ' || c_name_fio
              from hp_check_client_fio_ft(:c_id, 1));

      if (memo is not null) then
        status = - 4;
    end 

    -- проверка нет долга
    if(status > 0 and (select coalesce(sum(zad_os + zad_perc), 0) > 0
          from aarep_make_zad_list(:k_id, current_date, 1)))
    then
      status = - 5;


    update overpayment_msg
    set ovm_status = :status
    where (ovm_id = :ovm_id);

    if (memo is not null) then
      update or insert into overpayment_msg_memo (omm_id, omm_memo)
      values (:ovm_id, :memo)
      matching (omm_id);

    -- Работаем не больше 25 минут
    if (datediff(minute, start_time, cast('now' as timestamp)) > 25) then
      break;
  end
end
