CREATE PROCEDURE CM_RU_0095_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_client_pass_no varchar(30);
declare variable ma_refund integer;
declare variable ma_kommis_perc perc_type;
declare variable pr_code varchar(100);
declare variable c_id id_type;
declare variable is_pilot_group integer;
declare variable is_pilot_cli integer;
begin
 for
  select k_refund, k_kommis_perc, pr_code
  from product, kontract
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_code containing '.PILOT.'
  into ma_refund, ma_kommis_perc, pr_code
  do
  begin
    if (ma_refund = 1 and ma_kommis_perc > 0) then
    begin
      res_error_text = 'На рефинансирование доп.% должно быть 0 (ноль)';
      res_error_is_critical = 2;
      suspend;
      exit;
    end else
    if (pr_code not containing '.FAMILY.'
        and ma_refund = 0
        and ma_kommis_perc = 0) then
    begin
      res_error_text = 'У данного продукта доп.% не должно быть 0 (ноль)';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
    is_pilot_group = 0;
    for
    select kc_from_c_id,
           kc_amount_approved
    from knclient
    where kc_from_k_id = :k_id
    and kc_amount_approved > 60000
    into c_id,
         kc_amount
    do
    begin
      is_pilot_cli = 0;
      if (exists(select * from clientspilot
          where cp_from_c_id = :c_id)) then
      begin
        is_pilot_group = 1;
        is_pilot_cli = 1;
      end
      if (kc_amount > 60000 and is_pilot_cli = 0) then
      begin
        select c_name_fio, c_pass_no
        from clients
        where c_id = :c_id
        into res_info_text, kf_client_pass_no;

        res_error_text = 'Клиент не может брать по пилотному проекту более 60 тыс.';
        res_error_is_critical = 2;
        suspend;
      end
       
    end
    if (is_pilot_group = 0) then
    begin
        res_error_text = 'Данная группа не может брать по пилотному проекту, выберите другой кр.продукт.';
        res_error_is_critical = 2;
        suspend;
    end
  end
end
