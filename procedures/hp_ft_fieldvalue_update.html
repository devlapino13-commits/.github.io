CREATE PROCEDURE HP_FT_FIELDVALUE_UPDATE(
    FIL_ID D_ID,
    C_ID D_ID,
    FIF_ID D_ID,
    REP_DATE D_DATE,
    NEW_VALUE D_STR_LONG,
    UN_ID D_ID,
    IS_EDITED D_SMALLINT,
    ORIG_TEXT D_STR_LONG)
AS
declare variable old_value d_str_long;
declare variable fif_field_kind d_str_code;
declare variable fic_id d_id;
declare variable fiv_date_end d_date;
begin
  select fif_field_kind
  from ft_infofield
  where fif_id = :fif_id
  into fif_field_kind;

  select fiv_date_end, fiv_value
  from ft_infovalue
  where fiv_from_c_id = :c_id and
        fiv_from_fif_id = :fif_id and
        fiv_date_begin < :rep_date
  order by fiv_date_begin desc
  rows 1
  into fiv_date_end, old_value;

  if (row_count = 0 or
      old_value <> new_value) then
  begin
    update or insert into ft_infovalue (fiv_from_c_id, fiv_from_fif_id, fiv_date_begin, fiv_value)
    values (:c_id, :fif_id, :rep_date, :new_value)
    matching (fiv_from_c_id, fiv_from_fif_id, fiv_date_begin);

    if (fil_id is not null) then
    begin
      select fic_id
      from ft_infocheck
      where fic_from_fil_id = :fil_id and
            fic_date = :rep_date
      into fic_id;

      if (fic_id is not null) then
      begin
        update ft_infocheck
        set fic_data_changed = iif(:fif_field_kind = '', 1, fic_data_changed),
            fic_found_in_bl = iif(:fif_field_kind = 'stop_list', 1, fic_found_in_bl),
            fic_from_un_id = :un_id
        where fic_id = :fic_id;
      end
      else
      begin
        insert into ft_infocheck (fic_from_fil_id, fic_date, fic_data_changed, fic_found_in_bl, fic_from_un_id)
        values (:fil_id, :rep_date, iif(:fif_field_kind = '', 1, 0), iif(:fif_field_kind = 'stop_list', 1, 0), :un_id);
      end
    end
  end
  else
  begin
    if (fiv_date_end < cast('31.12.9999' as date) and fiv_date_end + 1 = rep_date) then
    begin
      delete from ft_infovalue
      where fiv_from_c_id = :c_id and
            fiv_from_fif_id = :fif_id and
            fiv_date_begin = :rep_date;

      if (fil_id is not null) then
      begin
        if (not exists(select fil_date_open
                       from ft_infolist
                       where fil_id = :fil_id and
                             fil_date_open = :rep_date)) then
        begin
          if (not exists(select *
                         from ft_infovalue
                         where fiv_from_c_id = :c_id and
                               fiv_date_begin = :rep_date)) then
          begin
            delete from ft_infocheck
            where fic_from_fil_id = :fil_id and
                  fic_date = :rep_date;
          end
        end
      end
    end
  end

  if (coalesce(is_edited, 0) = 1) then
  begin
    if (orig_text is null) then orig_text = '';

    if (not exists(select * from ft_infoeditvalue
                   where fie_from_c_id = :c_id and
                         fie_from_fif_id = :fif_id and
                         fie_date = :rep_date)) then
    begin
      insert into ft_infoeditvalue (fie_from_c_id, fie_from_fif_id, fie_date, fie_edit_value, fie_from_un_id)
      values (:c_id, :fif_id, :rep_date, :orig_text, :un_id);
    end

    insert into ft_infoeditvalue (fie_from_c_id, fie_from_fif_id, fie_date, fie_edit_value, fie_from_un_id)
    values (:c_id, :fif_id, :rep_date, :new_value, :un_id);
  end
end
