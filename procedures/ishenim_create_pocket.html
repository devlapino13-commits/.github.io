CREATE PROCEDURE ISHENIM_CREATE_POCKET(
    POCKET_DATE DATE,
    POCKET_TYPE VARCHAR(1) = null)
RETURNS (
    POCKET_ID INTEGER,
    POCKET_NAME VARCHAR(50))
AS
declare variable LAST_MONTH_DATE integer;
declare variable POCKET_DATE_BEGIN date;
declare variable POCKET_DATE_END date;
declare variable DATE_STR varchar(10);
declare variable CP_COUNTER integer;
declare variable ID_FINISH_SIDE varchar(10);
begin
  pocket_type = coalesce(pocket_type, '');
  pocket_date = pocket_date;

  if (pocket_type = '') then
  begin
--last_month_date = (datediff(day from POCKET_date to dateadd(1 month to POCKET_date)));
-- проверяю последний ли это день месяца или нет. если последний  и тип пакета не указан, что  дневной, тогда формирую месячный пакет

    if ((extract(month from pocket_date) <> (extract(month from dateadd(1 day to pocket_date))) and
        (pocket_type <> 'd'))) then
      pocket_type = 'm';
    else
      pocket_type = 'd';
  end

  pocket_date_begin = pocket_date;
  pocket_date_end = pocket_date;

  if (pocket_type = 'm') then
  begin
    date_str = cast(pocket_date as varchar(10));
    -- если пакет - месячный, получаю первый день месяца
    pocket_date_begin = '01.' || substring(date_str from 6 for 2) || '.' || substring(date_str from 1 for 4);
  end

  select max(cp.cp_counter)
  from cib_pocket cp
  where cp.cp_pocket_type = :pocket_type and
        cp.cp_date_end = :pocket_date_end and
        coalesce(cp.cp_cib_id, 0) <> f_get_rd_id('CIB.CRIF')
  into :cp_counter;

  cp_counter = coalesce(cp_counter, 0) + 1;

  id_finish_side = trim(iif(cp_counter < 10, '0' || cast(cp_counter as varchar(10)), cast(cp_counter as varchar(10))));

  date_str = cast(pocket_date_end as varchar(10));

  if (pocket_type = 'd') then
  begin
    date_str = substring(date_str from 1 for 4) || substring(date_str from 6 for 2) || substring(date_str from 9 for 2);
--POCKET_id = cast((date_str||id_finish_side) as integer);
    pocket_name = 'molbulak_' || date_str || '_' || pocket_type || '_def_' || cast(cp_counter as varchar(10)) || '_xml';
  end
  else
  if (pocket_type = 'm') then
  begin
    date_str = substring(date_str from 1 for 4) || substring(date_str from 6 for 2);
--POCKET_id = cast((date_str||'99'||id_finish_side) as integer);
    pocket_name = 'molbulak_' || date_str || '_' || pocket_type || '_def_' || cast(cp_counter as varchar(10)) || '_xml';
  end

  pocket_name = upper(pocket_name);
  
  insert into cib_pocket (cp_counter, cp_date_begin, cp_date_end, cp_pocket_name, cp_pocket_type, cp_pocket_status, cp_cib_id)
  values (:cp_counter, :pocket_date_begin, :pocket_date_end, :pocket_name, :pocket_type, 0, f_get_rd_id('CIB.ISHENIM'))
  returning cp_id
  into :pocket_id;

  suspend;
end
