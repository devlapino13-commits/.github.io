CREATE PROCEDURE HP_CLIENT_RELATION_LIST(
    C_ID DECIMAL(18,0),
    WITH_EMPTY INTEGER)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_SEX INTEGER,
    C_PLASTCARD_NO VARCHAR(30),
    C_CITIZEN INTEGER,
    DD_NAME VARCHAR(100),
    CREL_FROM_REL_ID VARCHAR(50),
    REL_NAME VARCHAR(100),
    C_FROM_CNT_ID VARCHAR(3),
    CNT_COUNTRY_NAME VARCHAR(100),
    CREL_IS_ACTIVE INTEGER,
    CREL_FROM_C_ID DECIMAL(18,0),
    CREL_OTHER_C_ID DECIMAL(18,0),
    CREL_FROM_UN_ID DECIMAL(18,0),
    CREL_MEMO VARCHAR(255),
    CREL_CAN_EDIT INTEGER,
    CREL_LEVEL INTEGER,
    CREL_LINK_C_ID DECIMAL(18,0),
    CREL_LINK_OTHER_C_ID DECIMAL(18,0),
    CREL_LINK_REL_ID VARCHAR(50),
    CREL_LINK_MEMO VARCHAR(255),
    CREL_FULL_CODE VARCHAR(50),
    IS_ACTIVE D_SMALLINT)
AS
declare variable c_sex_orig integer;
declare variable c_tmp decimal(18,0);
declare variable tmp_fio varchar(100);
declare variable tmp_sex integer;
declare variable tmp_rel_name varchar(100);
declare variable tmp_ss varchar(10);
declare variable cra_family integer;
declare variable cra_3 integer;
declare variable cra_4 integer;
declare variable cra_5 integer;
declare variable cra_6 integer;
declare variable cra_7 integer;
declare variable cra_8 integer;
declare variable cra integer;
begin
  crel_from_c_id = c_id;
  select c_sex
  from clients
  where c_id = :c_id
  into :c_sex_orig;

  c_from_cnt_id = '000';
  cnt_country_name = '';

-- tmp_ss - кодовое обозначение всех близких родственников
  if (c_sex_orig = 0) then
    tmp_ss = '1';
  else
    tmp_ss = '2';
  tmp_ss = tmp_ss || '345678';

  for select rel_other_c_id, rel_code, rel_level, rel_is_active, rel_memo, rel_link_c_id, rel_link_other_c_id,
             rel_link_rel_id, crel_full_code
      from hp_client_relation_tmp(:c_id, 4)
      order by rel_level, rel_code
      into :crel_other_c_id, :crel_from_rel_id, :crel_level, crel_is_active, crel_memo, :crel_link_c_id,
           :crel_link_other_c_id, :crel_link_rel_id, crel_full_code
  do
  begin
    is_active = :crel_is_active;

    crel_can_edit = 0;
    if (crel_level = 0) then
      crel_can_edit = 1;

    if (char_length(crel_from_rel_id) = 1) then
    begin
      tmp_ss = replace(tmp_ss, crel_from_rel_id, '');
    end

    rel_name = '';
    select rel_name
    from relationtype
    where rel_id = :crel_from_rel_id
    into :rel_name;
    if (rel_name = '') then
      rel_name = 'Дальн.родств.';

    tmp_rel_name = '';
    select rel_name
    from relationtype
    where rel_id = :crel_link_rel_id
    into :tmp_rel_name;
    if (tmp_rel_name = '') then
      tmp_rel_name = 'Дальн.родств.';

    select c_name_fio, c_birth_date, c_sex,
           (select cont_value
            from hp_contdata_get_cell_valid(0, :crel_other_c_id)), c_citizen, rd_name
    from clients, refdata
    where c_id = :crel_other_c_id and
          c_citizen = rd_id
    into :c_name_fio, :c_birth_date, :c_sex, :c_plastcard_no, c_citizen, dd_name;

    if (crel_other_c_id = crel_link_other_c_id) then
      c_tmp = :crel_link_c_id;
    else
      c_tmp = :crel_link_other_c_id;

    select c_name_fio, c_sex
    from clients
    where c_id = :c_tmp
    into :tmp_fio, :tmp_sex;

    if (crel_other_c_id = crel_link_other_c_id) then
      crel_link_memo = tmp_fio || ',' || iif(tmp_sex = 0, ' её ', ' его ') || lower(trim(tmp_rel_name)) || ': ' || c_name_fio;
    else
      crel_link_memo = c_name_fio || ',' || iif(c_sex = 0, ' её ', ' его ') || lower(trim(tmp_rel_name)) || ': ' || tmp_fio;

    suspend;
  end

  if (with_empty = 1) then
  begin
    crel_link_c_id = :c_id;
    crel_other_c_id = null;
    crel_link_other_c_id = null;
    is_active = null;

    crel_level = 1;
    crel_is_active = 1;
    crel_memo = '';
    crel_link_memo = '';

    crel_from_rel_id = '';
    crel_link_rel_id = '';
    crel_full_code = '';
    rel_name = '';

    c_name_fio = '';
    c_birth_date = null;
    c_sex = 0;
    c_plastcard_no = '';
    c_citizen = 0;
    dd_name = '';
    c_from_cnt_id = '000';
    cnt_country_name = '';

    select cra_family, cra_3, cra_4, cra_5, cra_6, cra_7, cra_8
    from clientrelabsent
    where cra_from_c_id = :c_id rows 1
    into :cra_family, :cra_3, :cra_4, :cra_5, :cra_6, :cra_7, :cra_8;

    while (tmp_ss <> '') do
    begin
      c_name_fio = 'Не заполнено';
      crel_memo = 'Необходимо заполнить данные';

      crel_from_rel_id = substring(tmp_ss from 1 for 1);
      crel_link_rel_id = crel_from_rel_id;
      crel_full_code = crel_from_rel_id;
      tmp_ss = substring(tmp_ss from 2 for 10);

      select rel_name
      from relationtype
      where rel_id = :crel_from_rel_id
      into :rel_name;

      cra = 0;
      cra = (case
               when (crel_from_rel_id = '1') then :cra_family
               when (crel_from_rel_id = '2') then :cra_family
               when (crel_from_rel_id = '3') then :cra_3
               when (crel_from_rel_id = '4') then :cra_4
               when (crel_from_rel_id = '5') then :cra_5
               when (crel_from_rel_id = '6') then :cra_6
               when (crel_from_rel_id = '7') then :cra_7
               when (crel_from_rel_id = '8') then :cra_8
             end);
      if (cra = 1) then
      begin
        c_name_fio = 'Отсутствует';
        crel_memo = 'Поле заполнено - такого родственника нет';
      end
      crel_link_memo = crel_memo;
      suspend;
    end
  end

end
