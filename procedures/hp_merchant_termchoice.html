CREATE PROCEDURE HP_MERCHANT_TERMCHOICE(
    TTYPE D_SMALLINT = 1)
RETURNS (
    TRM_ID D_INTEGER,
    TRM_IPC_ID D_STR_20,
    TRM_INVENT_STR D_STR_CODE,
    TRM_SERIAL_NO D_STR_255,
    TRM_SIMCARD_NO D_STR_20,
    ITO_DATE_END D_DATE,
    TRM_IS_NEW D_BOOL)
AS
begin
  for
    -- новые
    select
      trm_id,
      trm_ipc_id,
      trm_invent_str,
      trm_serial_no,
      trm_simcard_no,
      null,
      1
    from ipc_terminals
    where trm_type = :ttype
      and trm_actual = 1
      and not exists(select * from ipc_terminal_ownership where ito_trm_id = trm_id) and trm_id not in (305,306,308,309,310,311)
    union
    -- выведенные из эксплуатации но не распределённые
    select
      trm_id,
      trm_ipc_id,
      trm_invent_str,
      trm_serial_no,
      trm_simcard_no,
      max(ito_date_end),
      0
    from ipc_terminal_ownership as i
      inner join ipc_terminals
        on trm_id = i.ito_trm_id
          and trm_type = :ttype
          and trm_actual = 1
    where i.ito_status = -1 and
      not exists(select * from ipc_terminal_ownership where ito_trm_id = i.ito_trm_id and ito_status > -1)
    group by
      trm_id,
      trm_ipc_id,
      trm_invent_str,
      trm_serial_no,
      trm_simcard_no
  into
    :trm_id,
    :trm_ipc_id,
    :trm_invent_str,
    :trm_serial_no,
    :trm_simcard_no,
    :ito_date_end,
    :trm_is_new
  do
  begin
    if (ito_date_end is null or ito_date_end = '31.12.9999') then
      ito_date_end = current_date - 1;
    suspend;
  end
end
