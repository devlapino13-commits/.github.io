CREATE PROCEDURE HP_KKT_GET_DEVICE_ACTIVE(
    REG_ID D_INTEGER,
    OPER_DATE D_DATE)
RETURNS (
    MODE D_SMALLINT,
    REP_PRINT_MODE D_BOOL,
    DEVICE_LIST D_STR_255)
AS
declare variable test_device d_str_20;
begin
  -- (вызывается в Финстарте в окне погашения)
  -- процедура возвращает режим работы и список заводских номеров касс прикрепленных к офису
  -- mode:
  --    0 - работать без ККТ (по старому)
  --    1 - если ККТ подключена и есть в списке, то работать с ККТ, иначе работать без ККТ (по старому)
  --    2 - работать только с ККТ из списка. Если ККТ не подключена или ККТ в списке не найдена, то запрещено работать
  -- rep_print_mode: 0 - не печатать бумажные отчеты, 1 - печатать

  test_device = '00109324659314'; -- тестовая касса, доступна везде

  if (f_get_db_id() <> 81001
    or f_get_const('kkt') <> 1)
  then
  begin
    mode = 0;
    device_list = '';

    suspend;
    exit;
  end

  rep_print_mode = f_get_const('kkt_print_mode');

  -- если тестовая база или тестовый период, то работать только с тестовой ККТ или без ККТ (по старому)
  if (not exists(select * from hp_db_is_actual)
    or oper_date < '04.09.2023')
  then
  begin
    mode = 1;
    device_list = test_device;

    suspend;
    exit;
  end 

  -- в боевой базе в боевом периоде можно работать только с боевыми ККТ
  mode = 2;

  select coalesce(list(kkd_code), '')
  from kkt_devices
  where kkd_from_reg_id = :reg_id
    and :oper_date between kkd_date_begin and kkd_date_end
    and kkd_code <> :test_device
  into device_list;

  suspend;
end
