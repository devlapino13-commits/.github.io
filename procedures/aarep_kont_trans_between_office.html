CREATE PROCEDURE AAREP_KONT_TRANS_BETWEEN_OFFICE(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_CURRENCY VARCHAR(3))
RETURNS (
    KA_AKT_NO VARCHAR(20),
    KA_DATE DATE,
    PARENT_NAME VARCHAR(50),
    REG_NAME_OLD VARCHAR(50),
    KF_NAME_OLD VARCHAR(100),
    OFFICER_NAME_OLD VARCHAR(100),
    ZNAK VARCHAR(2),
    REG_NAME_NEW VARCHAR(50),
    KF_NAME_NEW VARCHAR(50),
    OFFICER_NAME_NEW VARCHAR(100),
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    REP_ACTIVE_BALANCE DECIMAL(18,4))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ka_akt_no, b.kal_date, parent_name, c.reg_code || ' ' || c.reg_name, a.kal_new_kf_name, cl1.c_name_fio,
             '->', d.reg_code || ' ' || d.reg_name, b.kal_new_kf_name, cl2.c_name_fio, k_id, k_kontract_no,
             k_kolich_client_approved,

             sum(
             case
               when :rep_currency = '000' then (case
                                                  when cr_rate < k_index_curr_rate then ok_os * k_index_curr_rate
                                                  else ok_os * cr_rate
                                                end)
               when :rep_currency = '---' then ok_os
               when :rep_currency = k_index_curr_code then ok_os
             end)

      from knaktlist a, knaktlist b, region c, region d, operkred, kontract, kontstate, knakt, region_list(:rep_region_id, 1, 3, 1), clients cl1, clients cl2, opercurr

      where a.kal_from_kf_id = b.kal_from_kf_id and
            b.kal_date between :rep_date_begin and :rep_date_end and
            b.is_delete = 0 and
            a.kal_new_reg_id <> b.kal_new_reg_id and
            a.kal_date_end = b.kal_date - 1 and
            a.kal_new_reg_id = c.reg_unicode and
            b.kal_new_reg_id = d.reg_unicode and
            ok_ol_is_active = 1 and
            ok_ol_date < b.kal_date and
            ok_from_k_id = k_id and
            ok_type not in (1, 2, 3, 9, 10) and
            k_from_kf_id = b.kal_from_kf_id and
            ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            cast(ks_date_begin as date) <= :rep_date_end and
            ks_date_end >= :rep_date_begin - 1 and
            (c.reg_unicode = id_list) and
            c.reg_status = 3 and
            b.kal_from_ka_id = ka_id and
            knakt.is_delete = 0 and
            cl1.c_id = a.kal_new_kf_officer and
            cl2.c_id = b.kal_new_kf_officer and
            k_index_curr_code = cr_from_cu_id and
            b.kal_date between cr_date and cr_date_end and
            ok_from_c_id = 0 and
            ((:rep_currency = '---') or
            (:rep_currency = '000') or
            (:rep_currency = k_index_curr_code))
      group by ka_akt_no, b.kal_date, parent_name, c.reg_code, c.reg_name, a.kal_new_kf_name, cl1.c_name_fio, d.reg_code, d.reg_name, b.kal_new_kf_name, cl2.c_name_fio, k_id, k_kontract_no, k_kolich_client_approved

      into :ka_akt_no, :ka_date, :parent_name, :reg_name_old, :kf_name_old, :officer_name_old, :znak, :reg_name_new,
           :kf_name_new, :officer_name_new, :k_id, :k_kontract_no, :k_kolich_client_approved, :rep_active_balance

  do
  begin
    suspend;
  end
end
