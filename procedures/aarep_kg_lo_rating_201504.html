CREATE PROCEDURE AAREP_KG_LO_RATING_201504(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    RES_WITH_LO INTEGER,
    RES_PAR_LIMIT DECIMAL(18,7))
RETURNS (
    RES_REGION_ID INTEGER,
    RES_LO_ID DECIMAL(18,0),
    RES_REGION_NAME VARCHAR(100),
    RES_LO_FIO VARCHAR(100),
    RES_CLIENT_END DECIMAL(18,4),
    RES_BALANCE_END DECIMAL(18,4),
    RES_DELINQ_BALANCE7_END DECIMAL(18,4),
    RES_CLIENT_BEGIN DECIMAL(18,4),
    RES_BALANCE_BEGIN DECIMAL(18,4),
    RES_DELINQ_BALANCE7_BEGIN DECIMAL(18,4),
    RES_CLIENT_DIFF DECIMAL(18,4),
    RES_DELINQ_BALANCE7_DIFF DECIMAL(18,4),
    RES_PAR7_END DECIMAL(18,7),
    RES_DELINQ_BALANCE30_BEGIN DECIMAL(18,4),
    RES_DELINQ_BALANCE30_END DECIMAL(18,4),
    RES_DELINQ_BALANCE30_DIFF DECIMAL(18,4),
    RES_PAR30_END DECIMAL(18,7),
    RES_LO_COUNT DECIMAL(18,4),
    RES_DISB_NEW DECIMAL(18,4),
    RES_DISB_REPEAT DECIMAL(18,4),
    RES_DELINQ_BALANCE1_BEGIN DECIMAL(18,4),
    RES_DELINQ_BALANCE1_END DECIMAL(18,4),
    RES_DELINQ_BALANCE1_DIFF DECIMAL(18,4),
    RES_PAR1_END DECIMAL(18,4),
    RES_PLAN_OS DECIMAL(18,4),
    RES_FACT_OS DECIMAL(18,4),
    RES_RETURN_KOEF DECIMAL(18,7),
    RES_CLOSE DECIMAL(18,4),
    RES_UDERJ DECIMAL(15,2),
    RES_PLAN_KOL_REPAY INTEGER,
    RES_BRANCH_NAME VARCHAR(100),
    RES_OKRUG_NAME VARCHAR(100),
    RES_WOFF_OS DECIMAL(18,4))
AS
declare variable lo_id_tmp decimal(18,0);
declare variable res_cl integer;
declare variable res_region_id_prior integer;
begin
  res_lo_fio = '';
  res_region_id_prior = 0;
  for select res_region_id, res_lo_id, sum(res_just_cl_end), sum(res_balance_end), sum(res_delinq_balance7_end),
             sum(res_just_cl_begin), sum(res_balance_begin), sum(res_delinq_balance7_begin),
             sum(res_delinq_balance30_begin), sum(res_delinq_balance30_end), sum(res_delinq_balance1_begin),
             sum(res_delinq_balance1_end), sum(res_disb_new), sum(res_disb_repeat), sum(res_fact_os), sum(res_plan_os),
             sum(res_close), sum(res_plan_kol_repay), sum(res_woff_os)
      from aarep_kg_lo_rating_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, :res_with_lo)
      group by res_region_id, res_lo_id
      into :res_region_id, :res_lo_id, :res_client_end, :res_balance_end, :res_delinq_balance7_end, :res_client_begin,
           :res_balance_begin, :res_delinq_balance7_begin, res_delinq_balance30_begin, res_delinq_balance30_end,
           :res_delinq_balance1_begin, :res_delinq_balance1_end, :res_disb_new, :res_disb_repeat, :res_fact_os,
           :res_plan_os, :res_close, :res_plan_kol_repay, :res_woff_os
  do
  begin
-- Получаем список СВЗ у кого есть КП на конец периода
    res_lo_count = 1;
    if (res_balance_end <> 0) then
    begin
      if (res_lo_id <> 0) then
      begin
        select c_name_fio
        from clients
        where c_id = :res_lo_id
        into :res_lo_fio;
      end
      if (res_with_lo = 0) then
      begin
        res_lo_count = 0;
        for select distinct oks_kc_id
            from oksummary
            where oks_date_begin <= :rep_date_end and
                  oks_date_end >= :rep_date_end and
                  oks_kc_id > 1000000000003 and
                  oks_region_id = :res_region_id and
                  oks_close_state = 0 and
                  (:rep_date_end - oks_delinq_date) <= 180 and
                  oks_kc_id <> 1010100001411 -- исключил из КСников Санжара
                  and
                  oks_balance > 0
            into :lo_id_tmp
        do
        begin
          res_lo_count = res_lo_count + 1;
        end
      end

      if (res_region_id_prior <> res_region_id) then
      begin
        res_region_id_prior = res_region_id;
        select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
        from region r1, region r2, region r3
        where r1.reg_unicode = :res_region_id and
              r1.parent = r2.reg_unicode and
              r2.parent = r3.reg_unicode
        into :res_region_name, :res_branch_name, :res_okrug_name;
      end

      if (res_lo_count = 0) then
        res_lo_count = 1;
      res_cl = res_client_end;
      res_client_end = round(res_client_end / res_lo_count, 2);
      res_client_begin = round(res_client_begin / res_lo_count, 2);
      res_disb_new = round(res_disb_new / res_lo_count, 2);
      res_disb_repeat = round(res_disb_repeat / res_lo_count, 2);
      res_close = round(res_close / res_lo_count, 2);

      res_client_diff = res_client_end - res_client_begin;
--      res_delinq_balance7_diff=res_delinq_balance7_end-res_delinq_balance7_begin;
--      res_delinq_balance30_diff=res_delinq_balance30_end-res_delinq_balance30_begin;
--      res_delinq_balance1_diff=res_delinq_balance1_end-res_delinq_balance1_begin;
--
      res_delinq_balance7_diff = round((res_delinq_balance7_end - res_delinq_balance7_begin - res_woff_os) / res_lo_count, 2);
      res_delinq_balance30_diff = round((res_delinq_balance30_end - res_delinq_balance30_begin) / res_lo_count, 2);
      res_delinq_balance1_diff = round((res_delinq_balance1_end - res_delinq_balance1_begin) / res_lo_count, 2);

      res_par7_end = res_delinq_balance7_end / res_balance_end;
      res_par7_end = res_par7_end * 100.0;

      res_par30_end = res_delinq_balance30_end / res_balance_end;
      res_par30_end = res_par30_end * 100.0;

      res_par1_end = res_delinq_balance1_end / res_balance_end;
--      res_par1_end=res_par1_end*100.0;

      res_return_koef = 0;
--      if (res_plan_os<>0) then res_return_koef = round((res_fact_os/res_plan_os)*100.0, 1);
--      if (res_return_koef > 100.0) then res_return_koef=100.0;
      if (res_plan_os <> 0) then
        res_return_koef = res_fact_os / res_plan_os;
      if (res_return_koef > 1.0) then
        res_return_koef = 1.0;

      if (res_close > 0) then
      begin
        res_uderj = cast(res_disb_repeat as curr_type) / res_close;
      end
      else
        res_uderj = 0;

      if (res_par7_end <= res_par_limit) then
      begin
        if ((res_with_lo = 1) or ((res_with_lo = 0) and
            (res_lo_count >= 2))) then
          suspend;
      end
--      then
--      begin
--        if ((RES_WITH_LO=1) or ((RES_WITH_LO=0) and (res_cl>500) and (RES_LO_COUNT>2))) then Suspend;
--      end

    end
  end
end
