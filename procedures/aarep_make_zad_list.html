CREATE PROCEDURE AAREP_MAKE_ZAD_LIST(
    KONT_ID DECIMAL(18,0),
    DD DATE,
    IS_NEED_PERC SMALLINT,
    SHOW_ZERO_LINE D_SMALLINT = 0)
RETURNS (
    SH_ID INTEGER,
    SH_DATE DATE,
    SH_CALC_FINE DECIMAL(18,4),
    SH_CALC_PERC DECIMAL(18,4),
    ZAD_OS DECIMAL(18,4),
    ZAD_PERC DECIMAL(18,4),
    SH_REPAY_TYPE INTEGER,
    S_OS DECIMAL(18,4),
    S_PERC DECIMAL(18,4),
    S_ALL DECIMAL(18,4),
    V_PAY_OS DECIMAL(18,4),
    V_PAY_PERC DECIMAL(18,4),
    IS_GRACE INTEGER,
    PAC_TYPE INTEGER,
    PAC_PERCENT DECIMAL(18,7),
    PAC_INC_INT SMALLINT,
    K_STOP_FINE DATE,
    K_INDEX_CURR_RATE DECIMAL(18,4),
    K_CURRENCY_RATE DECIMAL(18,4),
    SH_FINE_BASE DECIMAL(18,4),
    PR_MIN_FINE_BASE DECIMAL(18,4),
    SH_DATE_SH_ORIG DATE,
    SUM_FINE_BASE_OS DECIMAL(18,4),
    SUM_FINE_BASE_PERC DECIMAL(18,4),
    SH_FINE_KOL_DAY INTEGER,
    K_STOP_PERC DATE,
    DD_CHANGE_BEGIN DATE,
    DD_CHANGE_END DATE,
    DD_CHANGE_DAYS INTEGER,
    ZAD_KOMMIS DECIMAL(18,4),
    SH_PRIOR_PERCENT_PER_DAY D_PERCENT,
    IS_EXTRA D_SMALLINT,
    IS_HIDDEN_LINE D_SMALLINT,
    SH_ADD_PERC D_CURRENCY)
AS
declare variable min_base decimal(18,4);
declare variable sum_fine_base decimal(18,4);
declare variable db_id integer;
declare variable need_suspend d_integer;
begin
  min_base = 0;
  sum_fine_base = 0;
  select pr_min_fine_base
  from product, kontract
  where k_from_pr_id = pr_id and
        k_id = :kont_id
  into :min_base;

  db_id = f_get_db_id();

  if (min_base is null) then
    min_base = 0;
  if (min_base > 0) then
  begin
    select sum(sh_fine_base)
    from aarep_make_zad_list_temp(:kont_id, :dd, 0, :show_zero_line)
    into :sum_fine_base;
  end
  if (sum_fine_base is null) then
    sum_fine_base = 0;

  for select sh_id, sh_date, sh_calc_fine, sh_calc_perc, zad_os, zad_perc, sh_repay_type, s_os, s_perc, s_all, v_pay_os,
             v_pay_perc, is_grace, pac_type, pac_percent, pac_inc_int, k_stop_fine, k_index_curr_rate, k_currency_rate,
             sh_fine_base, pr_min_fine_base, sh_date_sh_orig, sh_fine_base_os, sh_fine_base_perc, sh_fine_kol_day,
             k_stop_perc, dd_change_begin, dd_change_end, dd_change_days, zad_kommis, sh_prior_percent_per_day,
             is_extra, is_hidden_line, sh_add_perc
      from aarep_make_zad_list_temp(:kont_id, :dd, :is_need_perc, :show_zero_line)
      into :sh_id, :sh_date, :sh_calc_fine, :sh_calc_perc, :zad_os, :zad_perc, :sh_repay_type, :s_os, :s_perc, :s_all,
           :v_pay_os, :v_pay_perc, :is_grace, :pac_type, :pac_percent, :pac_inc_int, :k_stop_fine, :k_index_curr_rate,
           :k_currency_rate, :sh_fine_base, :pr_min_fine_base, :sh_date_sh_orig, :sum_fine_base_os, :sum_fine_base_perc,
           :sh_fine_kol_day, :k_stop_perc, :dd_change_begin, :dd_change_end, :dd_change_days, :zad_kommis, sh_prior_percent_per_day,
           is_extra, is_hidden_line, sh_add_perc
  do
  begin
    is_hidden_line = 0;
    if (sh_repay_type = 0 and
        sh_date < dd and
        dd >= '30.09.2018') then
      sh_repay_type = 1;
    if (sum_fine_base < min_base) then
      sh_calc_fine = 0;

    need_suspend = 0;
    if ((cast(sh_calc_fine as decimal(18,2)) <> 0) or
        (cast(sh_calc_perc as decimal(18,2)) <> 0) or
        (cast(zad_kommis as decimal(18,2)) <> 0) or
        (cast(zad_perc as decimal(18,2)) <> 0) or
        (cast(sh_add_perc as decimal(18,2)) <> 0) or
        (cast(zad_os as decimal(18,2)) <> 0)) then
    begin
      -- если имеется задолженность, возвращаем строки
      need_suspend = 1;

      -- если будущим периодом встретили строку с задоженностью, отключаем последующий показ нулевых строк
      if (sh_date >= dd) then
        show_zero_line = 0;
    end

    -- если нет задолженности в будущих платежах, но входящий параметр требует нулевых, отображаем
    if (need_suspend = 0 and
        sh_date >= dd and
        show_zero_line = 1)
    then
    begin
      is_hidden_line = 1;
      need_suspend = 1;
    end

    if (need_suspend = 1) then
      suspend;
  end

  if (db_id not in (81001, 94401)) then  -- добавил igusev 22.09.2018 #operkroff_pay
    Exit; -- пока завершаем работу здесь. Sanjari, 09.08.2018

-- после реализации приема погашений забалансовых долгов, нужно убрать Exit

  if (sh_id is null) then
    sh_id = 0;
  sh_id = sh_id + 1;

  is_extra = 0;
  sh_date = null; -- это признак того, что задолженность по внесистемному учету
  sh_calc_fine = 0;
  sh_calc_perc = 0;
  zad_os = 0;
  zad_perc = 0;
  sh_repay_type = 1;
  s_os = 0;
  s_perc = 0;
  v_pay_os = 0;
  v_pay_perc = 0;
  is_grace = 0;
  zad_kommis = 0;
  is_hidden_line = 0;
  sh_add_perc = 0;

  select sum(oko_os + oko_sh_os), sum(oko_perc + oko_sh_perc)
  from operkroff, operlist ol
  where oko_from_k_id = :kont_id and
        oko_from_ol_id = ol_id and
        ol.is_delete = 0 and
        ol.ol_date <= :dd
  into :zad_os, :zad_perc;

  if (zad_os is null) then
    zad_os = 0;
  if (zad_perc is null) then
    zad_perc = 0;
  if (zad_os = 0 and
      zad_perc = 0) then
    exit;

  sh_calc_perc = zad_perc;

  s_all = zad_os + zad_perc;
  suspend;
end
