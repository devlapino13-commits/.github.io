CREATE PROCEDURE HP_BLACKLIST_MATCH(
    INPUT_C_ID D_ID)
RETURNS (
    MATCHED_C_ID D_ID,
    MATCHED_FIO D_STR_100,
    MATCHED_BIRTH_DATE D_DATE,
    MATCHED_PASS_NO VARCHAR(30),
    MATCHED_INN D_STR_20,
    MATCHED_CELL_NO D_STR_20,
    MATCH_TYPE D_STR_ACC,
    MATCH_COMMENT D_STR_255,
    IS_CRITICAL D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE)
AS
declare variable MATCHED_M_ID D_ID;
declare variable C_FIO D_STR_100;
declare variable C_BIRTH D_DATE;
declare variable C_PASS varchar(30);
declare variable C_INN varchar(30);
declare variable C_CELL D_STR_20;
declare variable TMP_COMMENT_ID D_INTEGER;
declare variable TMP_IS_BLACK D_INTEGER;
begin
  -- получаем данные клиента по input_c_id
  select c_name_fio,
         c_birth_date,
         coalesce(c_pass_no, ''),
         coalesce(c_inn, ''),
         coalesce(c_plastcard_no, '')
  from clients
  where c_id = :input_c_id
  into c_fio,
       c_birth,
       c_pass,
       c_inn,
       c_cell;

  -- ищем совпадения
  select m_id,
         c_id,
         c_name_fio,
         c_birth_date,
         c_pass_no,
         c_inn,
         c_plastcard_no,
         c_comment_id,
         c_is_black,
         max(c_comment),
         m_date_begin,
         m_date_end
  from hp_blacklist_search(:input_c_id, :c_fio, :c_birth, :c_pass, :c_inn, :c_cell,
                           1, -- проверка черного списка
                           1, -- проверка отказов
                           0  -- не проверяем мораторий
                           )
  where c_comment_id = 1  -- только критичные
  group by m_id,
           c_id,
           c_name_fio,
           c_birth_date,
           c_pass_no,
           c_inn,
           c_plastcard_no,
           c_comment_id,
           c_is_black,
           m_date_begin,
           m_date_end
  into matched_m_id,
       matched_c_id,
       matched_fio,
       matched_birth_date,
       matched_pass_no,
       matched_inn,
       matched_cell_no,
       tmp_comment_id,
       tmp_is_black,
       match_comment,
       date_begin,
       date_end;

  -- Если что-то найдено
  if (:matched_c_id is not null
    or :matched_m_id is not null)
  then
  begin
    -- Определяем тип совпадения
    if (tmp_is_black = 2) then
      match_type = 'Черный список';
    else if (tmp_is_black = 1) then
      match_type = 'Ранее отказан';
    else
      match_type = 'Совпадение';

    is_critical = 1;
    suspend;
  end
end
