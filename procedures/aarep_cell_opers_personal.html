CREATE PROCEDURE AAREP_CELL_OPERS_PERSONAL(
    REPORT_YEAR INTEGER,
    REPORT_MONTH INTEGER,
    REPORT_DATE DATE,
    REPORT_PLACE INTEGER,
    SALARY_YEAR INTEGER,
    SALARY_MONTH INTEGER,
    SALARY_TAX_KOEF DECIMAL(18,7),
    LIMIT_DATE DATE)
RETURNS (
    CB_AMOUNT DECIMAL(18,4),
    CAL_FROM_C_ID DECIMAL(18,0),
    CAL_HAS_LIMIT INTEGER,
    CAL_FIO VARCHAR(100),
    CAL_REGION_ID INTEGER,
    CAL_BRANCH_ID INTEGER,
    CAL_FROM_JL_ID INTEGER,
    CAL_FROM_JL_NAME VARCHAR(100),
    CAL_REGION_NAME VARCHAR(100),
    CAL_LIMIT_AMOUNT DECIMAL(18,4),
    CB_AMOUNT_WITH_LIMIT DECIMAL(18,4),
    CAL_IS_WORKING INTEGER,
    CAL_AMOUNT_TO_SALARY DECIMAL(18,4),
    CAL_AMOUNT_TAX_TO_SALARY DECIMAL(18,4),
    CAL_FROM_SAC_ID INTEGER,
    CAL_SALARY_SAVED DECIMAL(18,4),
    CAL_SALARY_TAX_SAVED DECIMAL(18,4),
    CAL_HEAD_OFFICE INTEGER,
    CAL_MONTH_MINUS_ID INTEGER,
    CAL_MONTH_SALDO_ID INTEGER)
AS
declare variable jl_kind integer;
begin
  if (salary_tax_koef is null) then
    salary_tax_koef = 0;
  select sac_id
  from sa_config
  where sac_is_credit = 2 rows 1
  into :cal_from_sac_id;
  if (cal_from_sac_id is null) then
    cal_from_sac_id = -1;

  select sac_id
  from sa_config
  where sac_is_credit = 9 rows 1
  into :cal_month_minus_id;
  if (cal_month_minus_id is null) then
    cal_month_minus_id = -1;

  select sac_id
  from sa_config
  where sac_is_credit = 10 rows 1
  into :cal_month_saldo_id;
  if (cal_month_saldo_id is null) then
    cal_month_saldo_id = -1;

  for
    select
      sum(cb_amount),
      cal_from_c_id,
      max(cal_has_limit),
      cal_fio,
      cal_region_id,
      cal_branch_id,
      cal_from_jl_id,
      cal_from_jl_name,
      cal_region_name,
      sum(cal_limit_amount),
      sum(cb_amount_with_limit),
      cal_is_working,
      cal_head_office
    from aarep_cell_opers(:report_year, :report_month, :report_date, :report_place)
      left join joblist
        on jl_id = cal_from_jl_id
    group by
      cal_from_c_id,
      cal_fio,
      cal_region_id,
      cal_branch_id,
      cal_from_jl_id,
      cal_from_jl_name,
      cal_region_name,
      cal_is_working,
      cal_head_office
  into
    :cb_amount,
    :cal_from_c_id,
    :cal_has_limit,
    :cal_fio,
    :cal_region_id,
    :cal_branch_id,
    :cal_from_jl_id,
    :cal_from_jl_name,
    :cal_region_name,
    :cal_limit_amount,
    :cb_amount_with_limit,
    :cal_is_working,
    :cal_head_office
  do
  begin
    if ((cal_fio is null) or (cal_fio = '')) then
      cal_fio = 'VIP номер или владелец не установлен';

    cal_salary_saved = 0;
    if (cal_salary_saved is null) then
      cal_salary_saved = 0;
    cal_salary_tax_saved = 0;
    if (cal_salary_saved <> 0) then
    begin
      if (salary_tax_koef <> 0) then
      begin
        cal_salary_tax_saved = cal_salary_saved / (100 + :salary_tax_koef);
        cal_salary_tax_saved = cal_salary_tax_saved * :salary_tax_koef;
        cal_salary_tax_saved = cast(cal_salary_tax_saved as decimal(15,2));
        cal_salary_saved = cal_salary_saved - :cal_salary_tax_saved;
      end
    end
    if (cal_limit_amount > cb_amount_with_limit) then
      cal_limit_amount = cb_amount_with_limit;
    cal_amount_to_salary = (:cb_amount_with_limit - :cal_limit_amount);
    cal_amount_to_salary = :cal_amount_to_salary + :cb_amount;
    cal_amount_tax_to_salary = cal_amount_to_salary / 100;
    cal_amount_tax_to_salary = cast((cal_amount_tax_to_salary * salary_tax_koef) as decimal(15,2));

    suspend;
  end
end
