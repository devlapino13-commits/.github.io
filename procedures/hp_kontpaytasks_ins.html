CREATE PROCEDURE HP_KONTPAYTASKS_INS(
    ACL_ID ID_TYPE,
    USER_ID ID_TYPE)
AS
declare variable k_id id_type;
declare variable seller_id d_id;
declare variable ag_id d_integer;
declare variable project d_str_100;
declare variable pr_ipf boolean;
declare variable pr_goods boolean;
declare variable pr_sugar boolean;
declare variable kpt_id d_integer;
declare variable kpt_status d_smallint;
declare variable kpt_error_text d_str_255;
declare variable kpt_account d_str_20;
declare variable kpt_bank_id d_integer;
declare variable kpt_amount d_currency;
declare variable err_txt d_str_255;
declare variable qr_account d_str_8100;
declare variable apq_id d_integer;
declare variable kpt_project d_str_20;
begin
  err_txt = '';

  select
    kpt_id,
    kpt_status,
    coalesce(lower(kpt_error_text), ''),
    kpt.kpt_project
  from kontpaytasks kpt
  where kpr_from_acl_id = :acl_id
  and kpt.kpt_status >= -1
  into
    :kpt_id,
    :kpt_status,
    :kpt_error_text,
    :kpt_project;

  if (kpt_status > 0) then
    err_txt = 'Платеж уже осуществлен';
  else if (kpt_status = 0) then
    err_txt = 'Оплата в процессе';

  if (kpt_project = 'qr' and kpt_status <> -1) then
    err_txt = 'Платеж уже обрабатывался. Повторная отправка в обработку чз QR - невозможна';


  if (err_txt = '') then
  begin
    select
      acl_object_id,
      aco_object_id,
      pr_code containing '.ipf.',
      pr_code containing '.installment_goods.',
      pr_code containing '.sugarmill.',
      rd_str_id
    from actionlist
      inner join actionorder
        on aco_id = acl_parent_id
          and aco_object_kind = 9
      inner join kontract
        on k_id = aco_object_id
      inner join product
        on pr_id = k_from_pr_id
      inner join refdata
        on rd_id = acl_from_rd_id
    where acl_id = :acl_id
      and acl_object_kind = 0
    into
      :seller_id,
      :k_id,
      :pr_ipf,
      :pr_goods,
      :pr_sugar,
      :project;
    if (seller_id is null or k_id is null) then
      err_txt = 'Продавец или контрагент не определен';
  end

  if (err_txt = '') then
  begin
    if (pr_goods or pr_sugar) then
    begin
      -- товары в рассрочку
      if (pr_goods) then
        kpt_amount = (
          select
            mp_amount * (1 - cs_discount / 100)
          from kontract
            inner join mc_purchase
              on mp_from_ma_id = k_from_ma_app_id
            inner join clientshops
              on cs_id = mp_from_cs_id
                and cs_from_rd_id = f_get_rd_id('ClientShops.outlets')
                and cs_is_delete = 0
          where k_id = :k_id);
      else
      -- сах.завод
      if (pr_sugar) then
        kpt_amount = (
          select
            spt_amount
          from sugarpaymenttask
          where spt_k_id = :k_id);

      ag_id = (
        select first 1
          ag_id
        from agreements
        where ag_object_kind = 0
          and ag_object_id = :seller_id
          and current_date between
            coalesce(ag_date_begin, '01.01.1900') and
            coalesce(ag_date_end, '31.12.9999')
          and ag_from_rd_id = f_get_rd_id('AgreementKinds.installment_goods')
          and ag_status = 1
        order by ag_id desc);
    end else
    -- ИПФ
    if (pr_ipf) then
      select
        max(ag_id),
        sum(kml_total)-coalesce(sum(kml_commis),0)
      from kontmat
        inner join agreements
          on ag_id = kml_from_ag_id
            and ag_object_id = :seller_id
            and ag_object_kind = 0
            and ag_from_rd_id in (
              f_get_rd_id('AgreementKinds.IPF_Sellers'),
              f_get_rd_id('AgreementKinds.Tsp_and_Ipf'),
              f_get_rd_id('AgreementKinds.IPF_Idjara'),
              f_get_rd_id('AgreementKinds.IPF_WithCommission'))
            and ag_status = 1
       where kml_object_kind = 9
         and kml_object_id = :k_id
         and kml_status = 1
       into
         :ag_id,
         :kpt_amount;
    kpt_amount = coalesce(kpt_amount, 0);

    -- счет продавца
    select
      case :project
        when 'elsom' then ag_elsom_account
        when 'elcard' then ag_elcard_account
        when 'mbank' then ag_mbank_account
        when 'qr' then 'оплата ч/з QR-код'
      else '' end,
      ag_elsom_bank_id
    from agreements
    where ag_id = :ag_id
    into
      :kpt_account,
      :kpt_bank_id;

    kpt_account = coalesce(trim(kpt_account), '');
    if (kpt_account = '') then
      err_txt = 'счат для оплаты "' || project || '" не определен';
    else
    if (project = 'elsom' and kpt_bank_id is null) then
      err_txt = '"' || project || '", Банк не определен';
    else
    if (kpt_amount = 0) then
      err_txt = 'Amount = 0';

    if (project = 'qr') then
    begin
        select apq_id,
               url_data
        from agreement_payment_qr_json(:ag_id)
        into :apq_id,
             :qr_account;

        qr_account = coalesce(qr_account, '');
        if (qr_account = '') then
            err_txt = 'Отсутствует информация по QR-коду';
    end
  end

  if (err_txt = '' and kpt_error_text <> '') then
  begin
    if (project = 'elsom' and
      kpt_error_text not in (
        'код ошибки не определен',
        'ошибка! код: 142 ; описание: exceeds max limit amount',
        'ошибка! код: -1 ; описание: unknown error',
        'ошибка! код: 116 ; описание: destination account not found',
        'ошибка! код: -1 ; описание: invalid credentials',
        'ошибка! код: -1 ; описание: user hash is locked'))
    then
      err_txt = kpt_error_text;
    else
    if (project = 'mbank' and
      kpt_error_text not containing 'недокументированная ошибка')
    then
      err_txt = kpt_error_text;
  end

  if (err_txt = '') then
  begin
    update actionlist set
      acl_status = 2,
      acl_result_rd_id = f_get_rd_id('EPayTaskResults.TransferredToPayment')
    where acl_id = :acl_id
      and acl_status in (0, 1);

    update or insert into kontpaytasks (
      kpt_id, kpt_project, kpr_from_acl_id, kpr_from_k_id, kpt_seller_id, kpt_account,
      kpt_amount, kpt_from_esm_id, kpt_error_text, kpt_from_un_id, kpt_status, kpt_bank_id)
    values (
      :kpt_id, :project, :acl_id, :k_id, :seller_id, :kpt_account,
      :kpt_amount, null, '', :user_id, 0, :kpt_bank_id)
    matching (kpt_id)
    returning kpt_id
    into :kpt_id;

    if (project = 'qr') then
        update or insert into kontpaytasks_qr (kptq_from_apq_id, kpt_qr_json, kptq_from_kpt_id)
        values (:apq_id, :qr_account, :kpt_id)
        matching (kptq_from_kpt_id);

    if (project = 'mbank') then
      post_event('new_mbank_transfer');
    else if (project = 'qr') then
      post_event('new_qr_to_contractor_transfer');
  end else
    exception e_message err_txt;
end
