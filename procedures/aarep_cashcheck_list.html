CREATE PROCEDURE AAREP_CASHCHECK_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    USER_C_ID D_ID = null)
RETURNS (
    CCM_ID TYPE OF COLUMN CASHCHECKMAIN.CCM_ID,
    CCM_DATE TYPE OF COLUMN CASHCHECKMAIN.CCM_DATE,
    CCM_FROM_REG_ID TYPE OF COLUMN CASHCHECKMAIN.CCM_FROM_REG_ID,
    CCM_ORDER_KIND TYPE OF COLUMN CASHCHECKMAIN.CCM_ORDER_KIND,
    CCM_ORDER_NO TYPE OF COLUMN CASHCHECKMAIN.CCM_ORDER_NO,
    CCM_ORDER_DATE TYPE OF COLUMN CASHCHECKMAIN.CCM_ORDER_DATE,
    CCM_IS_DELETE TYPE OF COLUMN CASHCHECKMAIN.CCM_IS_DELETE,
    CCM_CASHIER_CURR TYPE OF COLUMN CASHCHECKMAIN.CCM_CASHIER_CURR,
    CCM_CASHIER_CURR_DATESIGN D_DATETIME,
    CCM_CASHIER_NEW TYPE OF COLUMN CASHCHECKMAIN.CCM_CASHIER_NEW,
    CCM_CASHIER_NEW_DATESIGN D_DATETIME,
    CCM_AUDITOR TYPE OF COLUMN CASHCHECKMAIN.CCM_AUDITOR,
    CCM_AUDITOR_DATESIGN D_DATETIME,
    CCM_CASH_SALDO_AUTO TYPE OF COLUMN CASHCHECKMAIN.CCM_CASH_SALDO_AUTO,
    CCM_CASH_SALDO_MANNUAL TYPE OF COLUMN CASHCHECKMAIN.CCM_CASH_SALDO_MANNUAL,
    CCM_CASH_SALDO_FACT TYPE OF COLUMN CASHCHECKMAIN.CCM_CASH_SALDO_FACT,
    CCM_LAST_DOC_NO_INCOME TYPE OF COLUMN CASHCHECKMAIN.CCM_LAST_DOC_NO_INCOME,
    CCM_LAST_DOC_NO_OUTCOME TYPE OF COLUMN CASHCHECKMAIN.CCM_LAST_DOC_NO_OUTCOME,
    CCM_IS_VIDEO_WORKING TYPE OF COLUMN CASHCHECKMAIN.CCM_IS_VIDEO_WORKING,
    CCM_RESULT_MEMO TYPE OF COLUMN CASHCHECKMAIN.CCM_RESULT_MEMO,
    CCM_FROM_BLS_ID TYPE OF COLUMN CASHCHECKMAIN.CCM_FROM_BLS_ID,
    CCM_FROM_UN_ID TYPE OF COLUMN CASHCHECKMAIN.CCM_FROM_UN_ID,
    CCM_DATETIME TYPE OF COLUMN CASHCHECKMAIN.CCM_DATETIME,
    CCM_DELETE_USER_ID TYPE OF COLUMN CASHCHECKMAIN.CCM_DELETE_USER_ID,
    CCM_DELETE_DATETIME TYPE OF COLUMN CASHCHECKMAIN.CCM_DELETE_DATETIME,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    CASHIER_CURR_FIO VARCHAR(100),
    CASHIER_NEW_FIO VARCHAR(100),
    AUDITOR_FIO VARCHAR(100),
    HAS_SCAN INTEGER,
    ORDER_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    UN_NAME VARCHAR(30),
    CCM_RESULT_STR VARCHAR(50),
    CCM_IS_VIDEO_STR VARCHAR(50),
    CCM_OL_ID D_ID,
    OL_DOC_KIND D_INTEGER,
    CCM_EDL_ID D_ID,
    CAN_SIGN D_BOOL,
    CAN_SMS D_BOOL,
    CCM_STATUS D_SMALLINT,
    STATUS_STR D_STR_CODE,
    CAN_ATTACH_FILE D_BOOL)
AS
declare variable diff_amount decimal(18,4);
declare variable ccm_mbcard_fact d_integer;
declare variable ccm_mbcard_calc d_integer;
declare variable ccm_mbpinc_fact d_integer;
declare variable ccm_mbpinc_calc d_integer;
declare variable ccm_cash_saldo_delta d_currency;
declare variable ccm_attach_file_count d_integer;
declare variable const_attach_try d_integer;
declare variable db_id d_integer;
begin
  diff_amount = 100;
  const_attach_try = 1;
  db_id = f_get_db_id();

  for select ccm_id, ccm_date, ccm_from_reg_id, ccm_order_kind, ccm_order_no, ccm_order_date, ccm_is_delete,
             ccm_cashier_curr, ccm_cashier_curr_datesign, ccm_cashier_new, ccm_cashier_new_datesign, ccm_auditor, ccm_cash_saldo_auto, ccm_cash_saldo_mannual,
             ccm_cash_saldo_fact, ccm_last_doc_no_income, ccm_last_doc_no_outcome, ccm_is_video_working,
             ccm_result_memo, ccm_from_bls_id, ccm_from_un_id, ccm_datetime, ccm_delete_user_id, ccm_delete_datetime,
             r1.reg_code, r1.reg_name, r2.reg_name, un_name, ccm_mbcard_fact, ccm_mbcard_calc, ccm_mbpinc_fact, ccm_mbpinc_calc,
             ccm_auditor_datesign, ccm_ol_id, ccm_edl_id, ccm_status, ccm_attach_file_count
      from cashcheckmain, region_list(:rep_region_id, 1, -1, 1), region r1, region r2, username
      where r1.reg_unicode = id_list and
            ccm_from_reg_id = r1.reg_unicode and
            r1.parent = r2.reg_unicode and
            ccm_from_un_id = un_id and
            ccm_date between :rep_date_begin and :rep_date_end
      into :ccm_id, :ccm_date, :ccm_from_reg_id, :ccm_order_kind, :ccm_order_no, :ccm_order_date, :ccm_is_delete,
           :ccm_cashier_curr, :ccm_cashier_curr_datesign, :ccm_cashier_new, :ccm_cashier_new_datesign, :ccm_auditor, :ccm_cash_saldo_auto, :ccm_cash_saldo_mannual,
           :ccm_cash_saldo_fact, :ccm_last_doc_no_income, :ccm_last_doc_no_outcome, :ccm_is_video_working,
           :ccm_result_memo, :ccm_from_bls_id, :ccm_from_un_id, :ccm_datetime, :ccm_delete_user_id,
           :ccm_delete_datetime, :reg_code, :reg_name, :branch_name, :un_name, :ccm_mbcard_fact, :ccm_mbcard_calc, :ccm_mbpinc_fact, :ccm_mbpinc_calc,
           :ccm_auditor_datesign, :ccm_ol_id, :ccm_edl_id, :ccm_status, :ccm_attach_file_count
  do
  begin
    cashier_curr_fio = '';
    cashier_new_fio = '';
    auditor_fio = '';
    can_attach_file = 0;

    if (ccm_attach_file_count < const_attach_try and db_id = 41701) then    -- RFCKG-4550
      can_attach_file = 1;

    ccm_cash_saldo_delta = ccm_cash_saldo_mannual - ccm_cash_saldo_fact;
    if (db_id = 41701 and ccm_cash_saldo_delta > 0 and ccm_cash_saldo_delta < 0.01) then
      ccm_cash_saldo_mannual = ccm_cash_saldo_mannual - ccm_cash_saldo_delta; -- RFCKG-4558

    if (ccm_cashier_curr is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :ccm_cashier_curr
      into :cashier_curr_fio;
    end
    if (ccm_cashier_new is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :ccm_cashier_new
      into :cashier_new_fio;
    end
    if (ccm_auditor is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :ccm_auditor
      into :auditor_fio;
    end
    if (ccm_from_bls_id > 0) then
      has_scan = 1;
    else
      has_scan = 0;

    if (ccm_order_kind = 0) then
      order_name = 'Приказ';
    if (ccm_order_kind = 1) then
      order_name = 'Распоряжение';
    if (ccm_order_kind = 2) then
      order_name = 'Внезапная проверка';
    if (ccm_order_kind = 3) then
      order_name = 'Прием-передача';
    if (ccm_order_kind = 4) then
      order_name = 'Закрытия опердня';
    if (ccm_order_kind = 5) then
      order_name = 'Превышение установленного лимита';

    ccm_result_str = '';
    ccm_cash_saldo_mannual = coalesce(ccm_cash_saldo_mannual, 0);
    ccm_cash_saldo_fact = coalesce(ccm_cash_saldo_fact, 0);
    if (ccm_cash_saldo_mannual - ccm_cash_saldo_fact > diff_amount) then
      ccm_result_str = 'Недосдача';
    if ((ccm_cash_saldo_mannual - ccm_cash_saldo_fact > 0) and
        (ccm_cash_saldo_mannual - ccm_cash_saldo_fact <= diff_amount)) then
      ccm_result_str = 'Небольшая недосдача';
    if (ccm_cash_saldo_mannual - ccm_cash_saldo_fact = 0) then
      ccm_result_str = 'Нормально';
    if ((ccm_cash_saldo_fact - ccm_cash_saldo_mannual > 0) and
        (ccm_cash_saldo_fact - ccm_cash_saldo_mannual <= diff_amount)) then
      ccm_result_str = 'Небольшой излишек';
    if (ccm_cash_saldo_fact - ccm_cash_saldo_mannual > diff_amount) then
      ccm_result_str = 'Излишек';

    if (coalesce(ccm_mbcard_fact, 0) - coalesce(ccm_mbcard_calc, 0) <> 0 or
        coalesce(ccm_mbpinc_fact, 0) - coalesce(ccm_mbpinc_calc, 0) <> 0) then
      ccm_result_str = ccm_result_str || iif(ccm_result_str = '', '', '; ') || 'Расхождение(карты/пин)';

    ccm_is_video_str = '';
    if (ccm_is_video_working = 0) then
      ccm_is_video_str = 'Работает';
    if (ccm_is_video_working = 1) then
      ccm_is_video_str = 'Выключен';
    if (ccm_is_video_working = 2) then
      ccm_is_video_str = 'Неисправен';
    if (ccm_is_video_working = 3) then
      ccm_is_video_str = 'Отсутствует';

    can_sign = 0;
    can_sms = 0;
    if (ccm_is_delete = 0) then
    begin
      ol_doc_kind = (select ol_doc_kind from operlist where ol_id = :ccm_ol_id);
      if ((ccm_auditor = user_c_id and ccm_auditor_datesign is null) or
          (ccm_cashier_curr = user_c_id and ccm_cashier_curr_datesign is null) or
          (ccm_cashier_new = user_c_id and ccm_cashier_new_datesign is null) or
        exists(
          select *
          from cashcheckpart
          where ccp_from_ccm_id = :ccm_id
            and ccp_from_c_id = :user_c_id
            and ccp_datesign is null))
      then
        can_sign = 1;
      if (ccm_edl_id is null)
        then
          can_sms = 1;
    end

    if (ccm_is_delete = 0 and ccm_edl_id > 0) then
    begin
      if (ccm_status = 0 and exists(select * from edoc_sign where eds_from_edl_id = :ccm_edl_id and eds_status = 1))
        then ccm_status = 1;
      else
      if (ccm_status = 1 and (select signed_all from hp_cashcheck_signedall(:ccm_id)))
        then ccm_status = 2;
    end

    status_str =
      case
        when ccm_is_delete = 1
          then 'аннулирован'
        when ccm_status = 0
          then 'создан'
        when ccm_status = 1
          then 'частично подписан'
        when ccm_status = 2
          then 'подписан'
      end;

    suspend;
  end
end
