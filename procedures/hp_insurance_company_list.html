CREATE PROCEDURE HP_INSURANCE_COMPANY_LIST(
    IA_ID D_INTEGER)
RETURNS (
    O_IA_ID D_INTEGER,
    O_AG_ID D_INTEGER,
    O_AG_NAME VARCHAR(100),
    O_PERCENT D_SMALLINT)
AS
begin
    for select 
        ic.ic_from_ia_id,
        ic.ic_from_ag_id,
        ag.ag_caption,
        ic.ic_percent
    from insurance_company ic
    inner join agreements ag on ag.ag_id = ic.ic_from_ag_id
    where ic.ic_from_ia_id = :ia_id
    order by ic.ic_percent desc
    into 
        :o_ia_id,
        :o_ag_id,
        :o_ag_name,
        :o_percent
    do
        suspend;

end
