CREATE PROCEDURE AAREP_KONTRACT_LIST_BUX_SORT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    K_ID VARCHAR(18),
    KF_REGION_NAME D_STR_100,
    K_KONTRACT_NO VARCHAR(20),
    CL_CASHIER_NAME VARCHAR(100),
    K_DISB_DATE DATE,
    KC_ID ID_TYPE,
    K_AMOUNT_APPROVED CURR_TYPE,
    PR_NAME VARCHAR(50),
    K_KREDLINE_END D_DATE,
    TERM_IN_DAYS D_INTEGER,
    PERCENT_YEAR PERC_TYPE,
    PERCENT_EFFECT PERC_TYPE,
    CRED_FULL_COST DECIMAL(15,3),
    LOAN_EMP INTEGER,
    OKS_IS_REFUND_LLR_STR D_STR_10,
    ACC_NO D_STR_10,
    FINAL_PDN CURR_TYPE,
    STR_ID D_STR_100,
    K_FACT_DISB_ONLINE D_INTEGER,
    C_BIRTH_DATE D_DATE,
    OL_DATETIME D_TIME,
    KC_CIRCLE D_INTEGER,
    C_ID_N D_STR_100,
    UID D_STR_CODE,
    CLI_IS_RESIDENT_ON_DISB D_INTEGER,
    CLIENT_TYPE D_STR_20)
AS
declare variable max_sh_date d_date;
declare variable oks_is_refund_llr d_integer;
declare variable knr_non_resident d_smallint;
declare variable kc_date d_date;
declare variable pr_code d_str_100;
begin

  for select k_id, kc_id, k_kontract_no, k_disb_date, k_amount_approved, pr_name, k_kredline_end, cred_full_cost,
             loan_emp, k_fact_disb_online, cl_cashier_name, c_birth_date, kf_region_name, term_in_days, percent_year,
             percent_effect, oks_is_refund_llr_str, acc_no, final_pdn, str_id, ol_datetime, kc_circle, c_id_n, uid,
             cli_is_resident_on_disb, pr_code

      from aarep_kontract_list_bux(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency)
      order by k_disb_date, ol_datetime
      into :k_id, :kc_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :pr_name, :k_kredline_end, :cred_full_cost,
           :loan_emp, :k_fact_disb_online, :cl_cashier_name, :c_birth_date, :kf_region_name, :term_in_days,
           :percent_year, :percent_effect, :oks_is_refund_llr_str, :acc_no, :final_pdn, :str_id, :ol_datetime,
           :kc_circle, :c_id_n, :uid, :cli_is_resident_on_disb, pr_code
  do
  begin
    client_type = 'ФЛ';
    if (pr_code containing '.selfemployed.') then
      client_type = 'СЗ';

    suspend;
  end
end
