CREATE PROCEDURE HP_INVOICE_STARTUP_CREATE(
    REP_PERIOD INTEGER)
RETURNS (
    INVOICE_NO_MIN INTEGER,
    INVOICE_NO_MAX INTEGER,
    INVOICE_COUNT INTEGER,
    K_ID DECIMAL(18,0))
AS
declare variable k_year integer;
declare variable im_id decimal(18,0);
declare variable reg_id integer;
declare variable k_id_old decimal(18,0);
declare variable k_id_new decimal(18,0);
declare variable kcp_perc_calced decimal(18,4) = 0;
declare variable kcp_kommis_calced decimal(18,4) = 0;
declare variable need_add integer;
begin
  k_year = rep_period / 100;
  invoice_no_min = 0;
  invoice_no_max = 0;
  invoice_count = 0;
  select im_invoice_no
  from invoicemain
  where im_period starting :k_year
  order by im_period desc, im_invoice_no desc
  rows 1
  into :invoice_no_min;
  if (invoice_no_min is null) then
    invoice_no_min = 0;
  invoice_no_max = invoice_no_min;

  for select kcp_from_reg_id, kcp_from_k_id, kcp_perc_calced, kcp_kommis_calced
      from kontcalcperc
      where kcp_period = :rep_period
      order by kcp_from_reg_id, kcp_from_k_id
      into :reg_id, :k_id, :kcp_perc_calced, :kcp_kommis_calced
  do
  begin
    k_id_old = null;
    select kd_k_orig
    from kontdevides
    where kd_k_new = :k_id and
          (extract(year from kd_date) * 100 + extract(month from kd_date)) > :rep_period
    into :k_id_old;

    if (k_id_old is null) then
    begin
      need_add = 0;
      if (kcp_perc_calced <> 0 or kcp_kommis_calced <> 0) then
        need_add = 1;
      if (need_add = 0) then
      begin
        if (exists(select *
                   from kontcalcperc, kontdevides
                   where kcp_period = :rep_period and
                         (kcp_perc_calced <> 0 or
                         kcp_kommis_calced <> 0) and
                         kd_k_new = kcp_from_k_id and
                         kd_k_orig = :k_id and
                         (extract(year from kd_date) * 100 + extract(month from kd_date)) > :rep_period)) then
          need_add = 1;
      end
      if (need_add = 1) then
      begin
        invoice_count = invoice_count + 1;
        invoice_no_max = invoice_no_max + 1;
        insert into invoicemain (im_id, im_invoice_no, im_period, im_from_reg_id)
        values (-1, :invoice_no_max, :rep_period, :reg_id)
        returning (im_id)
        into :im_id;

        insert into invoicekont (ik_from_im_id, ik_from_k_id)
        values (:im_id, :k_id);
      end
    end
  end

  insert into invoicekont (ik_from_im_id, ik_from_k_id)
  select im_id, kd_k_new
  from kontcalcperc, kontdevides, invoicekont, invoicemain
  where kcp_period = :rep_period and
        (kcp_perc_calced <> 0 or
        kcp_kommis_calced <> 0) and
        kd_k_new = kcp_from_k_id and
        kd_k_orig = ik_from_k_id and
        ik_from_im_id = im_id and
        im_period = :rep_period and
        (extract(year from kd_date) * 100 + extract(month from kd_date)) > :rep_period;

  if (invoice_count > 0) then
  begin
    invoice_no_min = invoice_no_min + 1;
  end
  else
  begin
    invoice_no_min = 0;
    invoice_no_max = 0;
  end
  suspend;

end
