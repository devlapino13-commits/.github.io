CREATE PROCEDURE HP_GET_OPERSAL_SACVALUE(
    REP_C_ID DECIMAL(18,0),
    REP_START_YEAR_MONTH INTEGER,
    REP_FINISH_YEAR_MONTH INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    SAC_CONDITION VARCHAR(255))
RETURNS (
    RES_VALUE DECIMAL(18,4))
AS
declare variable sac_id integer;
declare variable sac_list varchar(255);
declare variable sql_script varchar(1024);
begin
  select osd_value
  from opersaldetail
  where osd_from_osm_id = -1
  into :res_value;
  select osm_oborot
  from opersalmain
  where osm_id = -1
  into :res_value;
  res_value = 0;

  sql_script = 'select list(sac_id) from sa_config ' || sac_condition;
  execute statement sql_script
      into :sac_list;

  if (sac_list is not null) then
  begin
    sac_list = trim(sac_list);
    sql_script = 'select sum(osd_value) from opersalmain, opersaldetail';
    sql_script = sql_script || ' where osm_ol_is_active=1';
    sql_script = sql_script || ' and osm_id=osd_from_osm_id';
    sql_script = sql_script || ' and osm_from_c_id=' || rep_c_id;
    sql_script = sql_script || ' and osd_from_sac_id in (' || :sac_list || ')';
    if (rep_date_end is null) then
    begin
      if (rep_start_year_month = rep_finish_year_month) then
        sql_script = sql_script || ' and osm_year_month=' || rep_start_year_month;
      else
        sql_script = sql_script || ' and osm_year_month between ' || rep_start_year_month || ' and ' || rep_finish_year_month;
    end
    else
    begin
      if (rep_date_begin = rep_date_end) then
        sql_script = sql_script || ' and osm_ol_date=''' || :rep_date_end || '';
      else
        sql_script = sql_script || ' and osm_ol_date between ''' || :rep_date_begin || ''' and ''' || :rep_date_end || '';
    end

    execute statement sql_script
        into :res_value;

    if (res_value is null) then
      res_value = 0;
  end
  suspend;
end
