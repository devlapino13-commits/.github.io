CREATE PROCEDURE AAREP_CALC_AMORT_KOMMIS_TOTAL(
    REPORT_REGION_ID DECIMAL(18,0),
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_NEED_PERC INTEGER)
RETURNS (
    RES_AMOUNT_BEGIN DECIMAL(18,4),
    RES_AMOUNT_END DECIMAL(18,4),
    RES_AMOUNT_PERIOD DECIMAL(18,4),
    RES_KOMMIS_AMOUNT DECIMAL(18,4),
    AAREP_CURRENCY VARCHAR(3),
    AAREP_REGION_NAME VARCHAR(100),
    AAREP_PR_ID INTEGER,
    AAREP_REGION_ID INTEGER,
    AAREP_K_FOND VARCHAR(20),
    RES_KOMMIS_PAYED DECIMAL(18,4),
    AAREP_ACCOUNT_PROFIT VARCHAR(20),
    AAREP_ACCOUNT_PAYED VARCHAR(20),
    AAREP_PRODUCT_NAME VARCHAR(50),
    RES_KRED_AMOUNT DECIMAL(18,4),
    RES_NSP_PERIOD DECIMAL(18,2),
    K_KONTRACT_NO D_STR_100,
    K_DISB_DATE DATE)
AS
begin
  for select res_amount_begin,
             res_amount_end,
             res_amount_period,
             res_kommis_amount,
             aarep_currency,
             aarep_region_name,
             aarep_pr_id,
             aarep_region_id,
             aarep_k_fond,
             res_kommis_payed,
             aarep_account_profit,
             aarep_account_payed,
             aarep_product_name,
             res_kred_amount,
             res_nsp_period,
             k_kontract_no,
             k_disb_date
  from aarep_calc_amort_kommis(:report_region_id, :report_date_begin, :report_date_end, :report_need_perc)
  into
    :res_amount_begin,
    :res_amount_end,
    :res_amount_period,
    :res_kommis_amount,
    :aarep_currency,
    :aarep_region_name,
    :aarep_pr_id,
    :aarep_region_id,
    :aarep_k_fond,
    :res_kommis_payed,
    :aarep_account_profit,
    :aarep_account_payed,
    :aarep_product_name,
    :res_kred_amount,
    :res_nsp_period,
    :k_kontract_no,
    :k_disb_date
  do
  begin
    if (res_amount_begin = 0 and res_amount_end = 0 and res_amount_period = 0) then
      continue;
    suspend;
  end
end
