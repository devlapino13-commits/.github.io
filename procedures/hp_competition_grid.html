CREATE PROCEDURE HP_COMPETITION_GRID(
    CPP_ID INTEGER)
RETURNS (
    CPG_ID INTEGER,
    CPG_FROM_RD_ID INTEGER,
    CPG_DATE_BEGIN DATE,
    CPG_TXT_POINTS_1 VARCHAR(100),
    CPG_TXT_POINTS_2 VARCHAR(100),
    CPG_TXT_POINTS_3 VARCHAR(100),
    CPG_TXT_POINTS_4 VARCHAR(100),
    CPG_TXT_POINTS_5 VARCHAR(100),
    CPG_TXT_POINTS_6 VARCHAR(100),
    CPG_TXT_POINTS_7 VARCHAR(100),
    CPG_TXT_POINTS_8 VARCHAR(100),
    CPG_TXT_POINTS_9 VARCHAR(100),
    CPG_TXT_DATA_1 VARCHAR(100),
    CPG_TXT_DATA_2 VARCHAR(100),
    CPG_TXT_DATA_3 VARCHAR(100),
    CPG_TXT_DATA_4 VARCHAR(100),
    CPG_TXT_DATA_5 VARCHAR(100),
    CPG_TXT_DATA_6 VARCHAR(100),
    CPG_TXT_DATA_7 VARCHAR(100),
    CPG_TXT_DATA_8 VARCHAR(100),
    CPG_TXT_DATA_9 VARCHAR(100))
AS
declare variable rd_id integer;
declare variable cpp_date_end date;
begin
  select cpp_from_rd_id, cpp_date_end
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :cpp_date_end;

  for select cpg_id, cpg_from_rd_id, cpg_date_begin, cpg_txt_points_1, cpg_txt_points_2, cpg_txt_points_3,
             cpg_txt_points_4, cpg_txt_points_5, cpg_txt_points_6, cpg_txt_points_7, cpg_txt_points_8, cpg_txt_points_9,
             cpg_txt_data_1, cpg_txt_data_2, cpg_txt_data_3, cpg_txt_data_4, cpg_txt_data_5, cpg_txt_data_6,
             cpg_txt_data_7, cpg_txt_data_8, cpg_txt_data_9
      from competgrid
      where cpg_from_rd_id = :rd_id and
            cpg_date_begin <= :cpp_date_end
      order by cpg_date_begin desc
      rows 1
      into :cpg_id, :cpg_from_rd_id, :cpg_date_begin, :cpg_txt_points_1, :cpg_txt_points_2, :cpg_txt_points_3,
           :cpg_txt_points_4, :cpg_txt_points_5, :cpg_txt_points_6, :cpg_txt_points_7, :cpg_txt_points_8,
           :cpg_txt_points_9, :cpg_txt_data_1, :cpg_txt_data_2, :cpg_txt_data_3, :cpg_txt_data_4, :cpg_txt_data_5,
           :cpg_txt_data_6, :cpg_txt_data_7, :cpg_txt_data_8, :cpg_txt_data_9
  do
  begin
    suspend;
  end
end
