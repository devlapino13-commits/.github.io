CREATE PROCEDURE ACREPLINE_LIST(
    REP_ID INTEGER,
    ID INTEGER,
    IS_DOWN INTEGER)
RETURNS (
    ID_LIST INTEGER)
AS
begin
  if (is_down = 1) then
  begin
    for with recursive arc_list
        as (select ar1.arl_number
            from acrepline ar1
            where ar1.arl_number = :id and
                  ar1.arl_from_arm_id = :rep_id
            union all
            select ar2.arl_number
            from acrepline ar2
            join arc_list on (arc_list.arl_number = ar2.parent) and
                  ar2.arl_from_arm_id = :rep_id)
        select arl_number
        from arc_list
        into :id_list
    do
      suspend;
  end
  else
  begin
    id_list = :id;
    suspend;
  end
end
