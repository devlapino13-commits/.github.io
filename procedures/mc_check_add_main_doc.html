CREATE PROCEDURE MC_CHECK_ADD_MAIN_DOC(
    MCL_ID D_ID)
RETURNS (
    RES_TEXT VARCHAR(250),
    RES_CODE D_SMALLINT)
AS
declare variable c_pass_date_expire d_date;
declare variable c_id d_id;
declare variable lang_code varchar(10);
declare variable c_pass_no varchar(30);
begin
  res_code = 0;
  --процедура вызывается при добавлении нового паспорта, проверяет актуальность имеющегося паспорта.
  select mcl_from_c_id, mcl_lang_code
  from mc_login
  where mcl_id = :mcl_id
  into :c_id, :lang_code;

  if (c_id = 0) then
  begin
    exit;
  end
  else
  begin
    select c_pass_date_expire, c_pass_no
    from clients
    where c_id = :c_id
    into :c_pass_date_expire, :c_pass_no;

    if (not exists(select *
                   from clientdocs cd
                   where cd.cds_from_c_id = :c_id and
                         cd.cds_doc_no = :c_pass_no)) then
      res_code = 0;
    else
    begin

      if (:c_pass_date_expire > current_date) then
      begin
        if (lang_code = 'kg') then
        begin
          res_text = 'Урматтуу кардар! Сиз берген паспорттун м�©�©н�©тү бүт�© элек. Эгерде бул паспорт жарактуу болбой калса, компаниянын кеңсесине же адисиңизге кайрылыңыз.';
        end
        if (lang_code = 'ru') then
        begin
          res_text = 'Уважаемый клиент! Сроки ранее предоставленного вами паспорта еще не истекли. Если данный паспорт уже недействителен, обратитесь в офис Компании или к своему специалисту.';
        end
        if (lang_code = 'az') then
        begin
          res_text = 'Hörmətli müştəri! Əvvəl təqdim etdiyiniz şəxsiyyət vəsiqəsinin müddəti hələ bitməyib. Əgər bu vəsiqə artıq etibarlı deyilsə, təşkilatın ofisinə və ya öz mütəxəssisinizə müraciət edin..';
        end
        res_code = 1;
        suspend;
        exit;
      end
      if (c_pass_date_expire is null and
          c_pass_no is not null) then
      begin
        res_text = 'Уважаемый клиент! Сроки ранее предоставленного вами паспорта еще не истекли. Если данный паспорт уже недействителен, обратитесь в офис Компании или к своему специалисту.';
        res_code = 1;
        suspend;
        exit;
      end
    end
  end
  suspend;
end
