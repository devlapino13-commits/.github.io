CREATE PROCEDURE HP_INFO_PORTAL_ACCESS_LIST
RETURNS (
    IPA_ID D_INTEGER,
    UN_NAME D_STR_CODE,
    REG_NAME D_STR_CODE,
    JL_NAME D_STR_100,
    ACCESS_DATE_BEGIN D_DATE,
    ACCESS_DATE_END D_DATE)
AS
begin
  for select ipa.ipa_id,
             un.un_name,
             rg.reg_name,
             jl.jl_name,
             cast(ipa.ipa_datetime as date),
             ipa.ipa_access_date_end
      from info_portal_access ipa
      inner join username un on un.un_id = ipa.ipa_from_un_id
      inner join region rg on rg.reg_unicode = ipa.ipa_from_reg_id
      inner join joblist jl on jl.jl_id = ipa.ipa_from_jl_id
      where ipa.ipa_access = 1 and
            current_date <= ipa.ipa_access_date_end
      into ipa_id,
           un_name,
           reg_name,
           jl_name,
           access_date_begin,
           access_date_end
  do
    suspend;
end
