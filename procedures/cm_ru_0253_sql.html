CREATE PROCEDURE CM_RU_0253_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID ID_TYPE;
declare variable PR_CODE D_STR_100;
declare variable Q_INN STR_20;
declare variable IS_FINISHED D_SMALLINT;
declare variable INN_CLI_IS_EQUAL D_BOOL;
begin
  if (f_get_const('inn_ru_source') = 0) then
    exit;

  res_error_text = '';

  for select pr_code, mac_from_c_id
      from miniapp, product, miniappclient
      where ma_id = :ma_id and
            ma_id = mac_from_ma_id and
            ma_from_pr_id = pr_id and
            ma_date_end = '31.12.9999' and
            mac_amount > 0
      into :pr_code, :c_id
  do
  begin
    if (not(pr_code containing '.zalog.' or pr_code containing '.ZALOG_CAR.')) then
    begin
      res_error_is_critical = 2;

      select inn, is_finished, inn_cli_is_equal
      from inn_query_status(:c_id)
      into q_inn, is_finished, inn_cli_is_equal;

      if (coalesce(q_inn, '') <> '' and inn_cli_is_equal = 0) then
        res_error_text = 'ИНН в карточке физ лица отличается от ИНН полученного от ФНС';

      if (coalesce(q_inn, '') = '' and is_finished = 0) then
        res_error_text = 'У клиента нет ИНН. Задачи по поиску ИНН в процессе обработки. Работа с заявкой невозможна до получения ИНН';

      if (coalesce(q_inn, '') = '' and coalesce(is_finished, 1) = 1) then
        res_error_text = 'У клиента нет ИНН, работа с заявкой невозможна до получения ИНН';

      if (res_error_text <> '') then
        suspend;
    end
  end
end
