CREATE PROCEDURE HP_NEW_CLISTATUS(
    REP_C_ID ID_TYPE,
    REP_CURR_STATUS SMALLINT,
    REP_NEW_STATUS SMALLINT,
    REP_DATE DATE,
    UN_ID ID_TYPE,
    REP_CLI_FIO VARCHAR(100),
    POLEZ_PLATEJ SMALLINT,
    REP_DETAIL VARCHAR(1024))
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable last_date_status date;
declare variable k_last_reg_id integer;
declare variable new_ol_id id_type;
declare variable ol_memo varchar(1024);
declare variable k_id id_type;
begin
-- присвоение клиенту новый статус
  res_text = '';
  if (rep_curr_status >= rep_new_status) then
  begin
    res_text = 'Заданный статус не должно быть меньше или равно чем текущий статус!';
    suspend;
    exit;
  end
  rep_detail = coalesce(rep_detail, '');
  if (rep_detail = '') then
  begin
    res_text = 'Укажите основание!';
    suspend;
    exit;
  end

  last_date_status = null;
  select max(cs_date_begin)
  from clientstatus
  where cs_from_c_id = :rep_c_id and
        cs_ol_is_delete = 0
  into last_date_status;

  if (last_date_status is not null and
      last_date_status > rep_date) then
  begin
    res_text = 'Имеется присвоенный статус более поздней датой, дата:' || last_date_status;
    suspend;
    exit;
  end

  k_last_reg_id = null;
  select k_last_reg_id, k_id
  from knclient, kontract, kontstate
  where kc_from_c_id = :rep_c_id and
        kc_amount_approved > 0 and
        kc_from_k_id = k_id and
        k_id = ks_from_k_id and
        ks_ol_is_delete = 0 and
        ks_state = 1
  order by ks_date_begin desc
  rows 1
  into k_last_reg_id, k_id;

  if (k_last_reg_id is null) then
  begin
    res_text = 'У клиента не имеется кредитная история!';
    suspend;
    exit;
  end

  select kv
  from get_kv(:k_last_reg_id, 5)
  into :new_ol_id;
  if (rep_cli_fio is null) then
  begin
    select c_name_fio
    from clients
    where c_id = :rep_c_id
    into rep_cli_fio;
  end

  ol_memo = 'Изменение статуса клиента:' || rep_cli_fio;

  insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                        ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                        ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                        ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month, ol_parent_id,
                        ol_manager_c_id, ol_from_rd_id)
  values (:new_ol_id, :k_last_reg_id, 1, :rep_date, null, null, null, :ol_memo, 0, 0, current_timestamp, :un_id,
          :rep_c_id, null, null, null, null, null, null, :rep_date, null, null, null, null, null, null, null, null);

  update or insert into clientstatus (cs_from_c_id, cs_from_k_id, cs_from_ol_id, cs_ol_is_delete, cs_date_begin,
                                      cs_date_end, cs_star_total, cs_polez_platej)
  values (:rep_c_id, :k_id, :new_ol_id, 0, :rep_date, '31.12.9999', :rep_new_status, :polez_platej)
  matching (cs_from_c_id, cs_from_k_id, cs_from_ol_id);

  update or insert into clientnewstatus (cns_from_ol_id, cns_from_c_id, cns_ol_is_delete, cns_date, cns_old_status,
                                         cns_new_status, cns_polez_platej, cns_detail, cns_user_id)
  values (:new_ol_id, :rep_c_id, 0, :rep_date, :rep_curr_status, :rep_new_status, :polez_platej, :rep_detail, :un_id)
  matching (cns_from_ol_id);

  suspend;
end
