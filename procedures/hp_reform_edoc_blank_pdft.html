CREATE PROCEDURE HP_REFORM_EDOC_BLANK_PDFT(
    C_ID D_ID)
AS
declare variable pp_date_begin d_date;
declare variable pp_date_end d_date;
declare variable edb_id d_id;
begin
  --Процедура обновляет Анкету ПДФТ/ЛДП в электронной заявке
  --Клиент может занимать несколько должностей одновременно, поэтому используется цикл for
  --Нужно обновить только те анкеты ПДФТ/ЛДП которые были созданы в период pp_date_begin и pp_date_end
  for select pp.pp_date_begin, pp.pp_date_end
      from public_persons pp
      where pp.pp_from_c_id = :c_id and
            current_date between pp.pp_date_begin and pp.pp_date_end and
            pp.pp_is_delete = 0
      into pp_date_begin, pp_date_end
  do
  begin
    for select edb.edb_id
        from edoc_sign eds
        inner join edoc_list edl on edl.edl_id = eds.eds_from_edl_id
        inner join edoc_blank edb on edb.edb_from_edl_id = edl.edl_id
        where eds.eds_from_c_id = :c_id and
              eds.eds_status = 1 and
              edl.edl_object_kind = f_objkind('miniapp', 'id') and
              edl.edl_blank_date between :pp_date_begin and :pp_date_end and
              edb.edb_from_bln_id = 1585 -- Анкета ПОФТЭ
        order by edl.edl_id desc
        into edb_id
    do
    begin
      --Проверка на то что бланк не переформировывается
      if (not(exists(select *
                     from edoc_blank edb
                     where edb.edb_id = :edb_id and
                           edb.edb_status = 0))) then
      begin
        update edoc_blank edb
        set edb.edb_status = 0
        where edb.edb_id = :edb_id;
      end
    end
  end
end
