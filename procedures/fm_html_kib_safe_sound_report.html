CREATE PROCEDURE FM_HTML_KIB_SAFE_SOUND_REPORT(
    MA_ID D_ID,
    C_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable EIL_DATE D_DATE;
declare variable UN_FIO D_STR_100;
declare variable C_NAME_FIO D_STR_100;
declare variable BIRTH_DATE D_DATE;
declare variable PASS_NO D_STR_20;
declare variable INN D_STR_20;
declare variable MAC_AMOUNT D_CURRENCY;
declare variable MA_KOLICH_REPAY D_INTEGER;
declare variable TOTAL_OUTSTANDING D_CURRENCY;
declare variable AMOUNT D_INTEGER;
declare variable TOTAL_HISTORY D_CURRENCY;
declare variable SCORE D_INTEGER;
declare variable RISK_LEVEL D_STR_10;
declare variable RISK_STR D_STR_20;
declare variable IPF_POINTS D_INTEGER;
declare variable COMMON_POINTS D_INTEGER;
declare variable ROWS_NO D_INTEGER;
declare variable PROVIDER D_STR_CODE;
declare variable CURRENCY D_STR_20;
declare variable ROLE_STR D_STR_20;
declare variable LOAN_STATUS D_STR_100;
declare variable LOAN_TYPE D_STR_20;
declare variable START_DATE D_DATE;
declare variable END_DATE D_DATE;
declare variable LOAN_AMOUNT D_CURRENCY;
declare variable DURATION D_INTEGER;
declare variable OUTSTANDING D_CURRENCY;
declare variable INTEREST_OUTSTANDING D_CURRENCY;
declare variable PAYMENT_STATUS D_STR_100;
declare variable MAX_PAST_DUE_DAYS D_STR_100;
declare variable BORROWERS D_STR_2048;
declare variable GUARANTORS D_STR_2048;
declare variable INTEREST_RATE D_STR_10;
declare variable PR_CODE D_STR_100;
declare variable BASE_K_ID D_ID;
declare variable BASE_MA_ID D_INTEGER;
begin
  select pr.pr_code
  from miniapp ma
  inner join product pr on pr.pr_id = ma.ma_from_pr_id
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.2999'
  into pr_code;

  if (pr_code containing '.insur_kontrakt.') then  -- RFCKG-5051
  begin
    select lir.lir_kontract_base
    from kontract k
    inner join loan_insurance_relation lir on k.k_id = lir.lir_kontract_insurance
    where k.k_from_ma_app_id = :ma_id
    into base_k_id;

    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :base_k_id
    into base_ma_id;

    if (coalesce(base_ma_id, 0) <> 0) then
      ma_id = base_ma_id;
  end

  res_html = (select res_style from fm_html_styles) || '<body>';
  res_html = blob_append(res_html, '<b>Отчет по проверке в ЗАО КБ "Сэйф энд Саунд"</b><br>');

  select cast(t.eil_datetime as date), t.un_fio, t.c_name_fio, t.birth_date, t.pass_no, t.inn,
         t.mac_amount, t.ma_kolich_repay, t.total_outstanding, t.amount,
         t.total_history, t.score, t.risk_level, t.risk_str, t.ipf_points, t.common_points
  from print_report_safe_and_sound(:ma_id, :c_id) t
  rows 1
  into :eil_date, un_fio, c_name_fio, birth_date, pass_no, inn,
        mac_amount, ma_kolich_repay, total_outstanding, amount,
        total_history, score, risk_level, risk_str, ipf_points, common_points;

  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Дата проверки', :eil_date, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Пользователь', :un_fio, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  res_html = blob_append(res_html, '<b>Клиент ' || :c_name_fio || '<b><br>');
  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Дата рождения', f_datetimetostr(:birth_date), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Документ', :pass_no, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Код соц. страхов.', :inn, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  res_html = blob_append(res_html, '<b>Предполагаемые суммы по всем активным кредитам<b><br>');
  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Сумма в заявке', round(:mac_amount), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Срок кредита в заявке', :ma_kolich_repay, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Основная сумма', round(:total_outstanding), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Кол-во', :amount, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Всего кредитных историй', round(:total_history), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Отображено', round(:total_history), '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  res_html = blob_append(res_html, '<b>Скоринг<b><br>');
  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Скоринг балл', :score, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Степень риска', :risk_level, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Пояснение', :risk_str, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  res_html = blob_append(res_html, '<b>Внутренний скоринг Компании<b><br>');
  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Баллы по классике', :common_points, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Баллы по ИПФ', :ipf_points, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  for select t.rows_no, iif(t.is_our_inst = 1, 'В М Булак', 'В другом ФКУ') as provider,
    t.currency_inst, t.role_str_inst, t.loan_status_inst, t.own_type_inst,
    t.start_date_inst, t.end_date_inst, t.loan_amount_inst, t.duration_inst, t.outstanding_inst, t.interest_outstanding_inst,
    t.payment_status_inst, t.max_past_due_inst, t.borrowers_inst, t.guarantors_inst, t.interest_rate
  from print_report_safe_and_sound(:ma_id, :c_id) t
  into :rows_no, provider, currency, role_str, loan_status, loan_type,
        start_date, end_date, loan_amount, duration, outstanding, interest_outstanding,
        payment_status, max_past_due_days, borrowers, guarantors, interest_rate
  do
  begin
    res_html = blob_append(res_html, '<b>Кредит № '|| :rows_no || '<b><br>');

    res_html = blob_append(res_html, '<table width="100%"><col width="25%"><col width="25%"><col width="25%"><col width="25%">' ||
               '<tr><th>ФКУ</th><th>Валюта</th><th>Участник</th><th>Статус</th></tr>' ||
               f_html_table_row(:provider, :currency, :role_str, :loan_status, -1, -1, -1, -1, 4, '') ||
               '<br></table>');

    res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Тип', :loan_type, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Дата выдачи', f_datetostr(:start_date), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Дата окончания', f_datetostr(:end_date), '', '', -1, -1, -1, -1, 2, '') ||
         f_html_table_row('Процентная ставка', :interest_rate, '', '', -1, -1, -1, -1, 2, '') ||

               f_html_table_row('Общая сумма', round(:loan_amount, 2), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Срок кредита', round(:duration), '', '', -1, -1, -1, -1, 2, '') ||
            
               f_html_table_row('Остаток ОС', round(:outstanding, 2), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Остаток %', round(:interest_outstanding, 2), '', '', -1, -1, -1, -1, 2, '') ||

               f_html_table_row('Статус кредита', :loan_status, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Статус погашения', :payment_status, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Макс. дн. просроч.', :max_past_due_days, '', '', -1, -1, -1, -1, 2, '') ||

               f_html_table_row('Заемщики', :borrowers, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Поручители', :guarantors, '', '', -1, -1, -1, -1, 2, '')|| '</table><br>');
  end

  res_html = blob_append(res_html, '</body>');
  suspend;
end
