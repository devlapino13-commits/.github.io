CREATE PROCEDURE HP_KONTRACT_SET_STATUS(
    K_ID D_ID,
    K_STATUS D_INTEGER,
    IS_FINSTART D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_LONG,
    RES_NEED_ROLLBACK D_SMALLINT)
AS
declare variable db_id d_id;
declare variable mac_id d_integer;
declare variable fpb_id d_integer;
begin
  db_id = f_get_db_id();
  res_text = '';
  res_need_rollback = 0;

  if (k_status = 0) then  -- RFCRU-4491
  begin
    if (exists(select *
               from kontstate
               where ks_from_k_id = :k_id and
                     ks_state > 0 and
                     ks_date_end >= '31.12.2999' and
                     ks_ol_is_delete = 0)) then
    begin
      res_need_rollback = 1;
      res_text = 'Займ уже выдан, команда отменяется';
      suspend;
      exit;
    end
  end

  if (k_status = -2) then
  begin
    if (not exists(select *
                   from kontstate
                   where ks_from_k_id = :k_id and
                         ks_state = 0 and
                         ks_date_end >= '31.12.2999' and
                         ks_ol_is_delete = 0)) then
    begin
      res_need_rollback = 1;
      res_text = 'Договор ранее активирован, команда отменяется';
      suspend;
      exit;
    end

    if (exists (select *
    from kontract
    where k_id = :k_id and k_status = -1)) then
    begin
      res_need_rollback = 1;
      res_text = 'По договору уже был отказ, команда отменяется';
      suspend;
      exit;
    end

    if (exists(select 1      -- RFCRU-4491
               from kontract
               where k_id = :k_id and
                     k_status = 1)) then
    begin
      res_need_rollback = 1;
      res_text = 'Договор утвержден, команда отменяется';
      suspend;
      exit;
    end

    if (exists(select acl_status
      from hp_actionlist_kontract_get(:k_id)
      where acl_status = 1)) then
    begin
      res_need_rollback = 1;
      res_text = 'Договор принят на рассмотрение КК. ождитесь завершения процесса.';
      suspend;
      exit;
    end
  end
  if (k_status = 1) then
  begin
    select res_text, res_need_rollback
    from hp_approve_loan('kontract', :k_id, :un_id, :is_finstart)
    into :res_text, :res_need_rollback;
  end
  else
  begin
    update kontract
    set k_status = :k_status,
        k_user_id = :un_id
    where k_id = :k_id and
          k_status <> :k_status;
  end

  if (db_id = 81001 and k_status = -1) then
  begin
    -- договор нельзя отказ делать после утверждения не прошли 5 рабочих дней
    select res_text
    from hp_cancel_loan_status(:k_id, :un_id, 1)
    into res_text;
    if (res_text <> '') then res_need_rollback = 1;
  end

  if (db_id = 41701 and k_status = -1) then  -- RFCKG-5090
  begin
    select res_text
    from hp_annual_insurance_loan(:k_id, :un_id)
    into res_text;
    if (res_text <> '') then res_need_rollback = 1;
  end 

  -- RFCRU-5012
  if (db_id = 81001 and res_text = '') then
  begin
    for select mac.mac_id, fcb.fcb_from_fpb_id
    from kontract k
    inner join knclient kc on kc.kc_from_k_id = k.k_id
      and kc.kc_amount_approved > 0
    inner join miniappclient mac on mac.mac_from_ma_id = k.k_from_ma_app_id
      and mac.mac_from_c_id = kc.kc_from_c_id
    left join fps_loan_banks flb on flb.flb_from_kc_id = kc.kc_id
    left join fps_client_banks fcb on flb.fbl_from_fcb_id = fcb.fcb_id
    where k.k_id = :k_id
      and k.k_disb_kind = 7
      and k.k_from_ma_app_id = mac.mac_from_ma_id
    into :mac_id, :fpb_id
    do
    begin
      select res_is_error, res_text
      from fm_set_fps_loan_banks('miniappclient', :mac_id, :fpb_id, :un_id)
      into res_need_rollback, res_text;

      if (res_need_rollback = 1) then
        leave;

      res_text = '';
    end
  end

  suspend;
end
