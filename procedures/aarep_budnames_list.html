CREATE PROCEDURE AAREP_BUDNAMES_LIST(
    REP_REGION_ID INTEGER,
    REP_YEAR INTEGER,
    REP_WITH_RESERV INTEGER)
RETURNS (
    BN_ID INTEGER,
    BN_NAME VARCHAR(255),
    BN_MEMO VARCHAR(1024),
    BN_FOR_REGION INTEGER,
    BN_YEAR INTEGER,
    BN_IS_RESERV INTEGER,
    BN_REGION_NAME VARCHAR(100),
    BN_STATUS_STR VARCHAR(20),
    BN_STATUS INTEGER,
    BN_LAST_EDIT_DATE TIMESTAMP,
    BN_LAST_USER_FIO VARCHAR(100),
    BN_IS_RESERV_STR VARCHAR(20),
    BN_FROM_CC_ID INTEGER,
    CC_NAME VARCHAR(100))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Санжар Исламов */
  /* Для всех стран*/
  /* Модуль: Бюджетирование, FinStart */
  /* Список названий бюджетов для дальнейшего редактирования */
  /* Создан 01.06.2012*/

  for select bn_id, bn_name, bn_memo, bn_for_region, bn_year, bn_is_reserv, bn_status, reg_code || ' ' || reg_name,
             bn_from_cc_id
      from budnames, region_list(:rep_region_id, 1, -1, 1), region
      where (bn_year = :rep_year or
            :rep_year = 0) and
            bn_for_region = id_list and
            bn_is_reserv <= :rep_with_reserv and
            reg_unicode = bn_for_region
      into :bn_id, :bn_name, :bn_memo, :bn_for_region, :bn_year, :bn_is_reserv, :bn_status, :bn_region_name,
           :bn_from_cc_id
  do
  begin
    if (bn_status = -1) then
      bn_status_str = 'Аннулирован';
    if (bn_status = 0) then
      bn_status_str = 'Действующий';
    if (bn_status = 1) then
      bn_status_str = 'Закрыт';

    if (bn_is_reserv = 1) then
      bn_is_reserv_str = 'Резерв';
    if (bn_is_reserv = 0) then
      bn_is_reserv_str = 'Основной';

    cc_name = null;
    if (bn_from_cc_id is not null) then
    begin
      select cc_name
      from costcenter
      where cc_id = :bn_from_cc_id
      into :cc_name;
    end

    select bom_datetime, c_name_fio
    from budopermain, username, clients
    where bom_from_bn_id = :bn_id and
          bom_from_un_id = un_id and
          un_from_c_id = c_id
    order by bom_datetime desc
    rows 1
    into :bn_last_edit_date, bn_last_user_fio;

    suspend;
  end
end
