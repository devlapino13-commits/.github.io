CREATE PROCEDURE FM_KONTMAT_UPDATE(
    KML_ID D_ID,
    KML_OBJECT_KIND D_INTEGER,
    KML_OBJECT_ID D_ID,
    AG_ID D_INTEGER,
    ML_NAME D_STR_255,
    ML_MODEL D_STR_255,
    KML_GUARANTEE_MONTH D_INTEGER,
    KML_COUNT D_CURRENCY,
    KML_COST D_CURRENCY,
    KML_TOTAL D_CURRENCY,
    ML_FROM_YL_ID D_INTEGER,
    FEG_ID D_INTEGER,
    KML_IS_RELATIVE D_SMALLINT,
    UN_ID D_ID,
    KML_IS_USED D_SMALLINT,
    KMI_IMEI D_ID = null)
RETURNS (
    ERR_TEXT VARCHAR(255))
AS
declare variable ml_id decimal(18,0);
declare variable kol d_currency;
declare variable kmi_id d_id;
begin
  err_text = '';
  if (coalesce(kml_object_kind, 0) = 0) then
    err_text = 'Не указан тип объекта';
  if (coalesce(kml_object_id, 0) = 0) then
    err_text = 'Не указан идентификатор объекта';
  if (coalesce(ag_id, 0) = 0) then
    err_text = 'Не указан продавец(договор)';
  if (coalesce(ml_name, '') = '') then
    err_text = 'Не указано название товара';
  if (coalesce(ml_model, '') = '' or (coalesce(feg_id, 0) = 0)) then
    err_text = 'Не указана модель товара';
  -- Гарантию не проверяем (если не дадут)
  kml_guarantee_month = coalesce(kml_guarantee_month, 0);
  if (coalesce(kml_total, 0) = 0) then
    err_text = 'Не указана сумма';
  if (coalesce(kml_count, 0) = 0) then
    err_text = 'Не указано количество товара';
  if (coalesce(kml_cost, 0) = 0) then
    err_text = 'Не указана цена товара';
  if (coalesce(ml_from_yl_id, 0) = 0) then
    err_text = 'Не указана единица измерения';
  kml_is_relative = coalesce(kml_is_relative, 0);
  kml_is_used = coalesce(kml_is_used, 0);

  kol = 0;
  if (ag_id > 0 and
      feg_id > 0) then
  begin
    select coalesce(feg_max_number, 0)
    from hp_foreign_econom_goods_ref(:ag_id)
    where feg_id = :feg_id
    into :kol;

    if (kml_count > kol) then
      err_text = 'Для данной номенклатуры, количество превышает максимально допустимое';
  end
  if (err_text <> '') then
  begin
    suspend;
    exit;
  end

  -- Вставка/обновление товаров (kontmat)
  -- Генерируем ml_id в любом случае
  select ml_id
  from hp_matlist_insert(:ml_name, :ml_model, :ml_from_yl_id)
  into ml_id;

  if (kml_id is null) then -- Вставка
  begin
    insert into kontmat (kml_object_kind, kml_object_id, kml_from_ml_id, kml_guarantee_month, kml_count, kml_cost,
                         kml_total, kml_status, kml_from_ag_id, kml_from_feg_id, kml_is_relative, kml_is_used)
    values (:kml_object_kind, :kml_object_id, :ml_id, :kml_guarantee_month, :kml_count, :kml_cost, :kml_total, 1,
            :ag_id, :feg_id, :kml_is_relative, :kml_is_used)
    returning kml_id
    into kml_id;

    if (kmi_imei is not null) then
      insert into kontmat_imei (kmi_kml_id, kmi_imei)
      values (:kml_id, :kmi_imei);
  end
  else -- Обновление
  begin
    if (exists(select *
               from kontmat, operlist
               where ol_id = kml_bill_id and
                     is_delete = 0 and
                     kml_id = :kml_id)) then
      err_text = 'Операция запрещена. Сначала удалите бухгалтерскую проводку.';
    else
    begin
      update or insert into kontmat (kml_id, kml_object_kind, kml_object_id, kml_from_ml_id, kml_guarantee_month,
                                     kml_count, kml_cost, kml_total, kml_status, kml_from_ag_id, kml_from_feg_id,
                                     kml_is_relative, kml_is_used)
      values (:kml_id, :kml_object_kind, :kml_object_id, :ml_id, :kml_guarantee_month, :kml_count, :kml_cost,
              :kml_total, 1, :ag_id, :feg_id, :kml_is_relative, :kml_is_used);

      select kmi.kmi_id
      from kontmat_imei kmi
      where kmi.kmi_kml_id = :kml_id
      into kmi_id;
      if (kmi_imei is not null) then
        update or insert into kontmat_imei (kmi_id, kmi_kml_id, kmi_imei)
        values (:kmi_id, :kml_id, :kmi_imei);
    end
  end
  suspend;
end
