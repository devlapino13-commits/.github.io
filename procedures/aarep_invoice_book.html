CREATE PROCEDURE AAREP_INVOICE_BOOK(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_INV_DATE DATE,
    AAREP_INV_NO VARCHAR(15),
    AAREP_INV_CUSTOMER VARCHAR(100),
    AAREP_INV_AMOUNT DECIMAL(18,4))
AS
declare variable rep_period_begin integer;
declare variable rep_period_end integer;
declare variable reg_code varchar(10);
declare variable k_id decimal(18,0);
declare variable c_id decimal(18,0);
begin
-- Книга продаж (счета-фактуры)
  rep_period_begin = 0;
  rep_period_end = 0;
  if (extract(day from :report_date_begin) = 1) then
    rep_period_begin = extract(year from :report_date_begin) * 100 + extract(month from :report_date_begin);
  if (extract(day from (:report_date_end + 1)) = 1) then
    rep_period_end = extract(year from :report_date_end) * 100 + extract(month from :report_date_end);
-- По ТЗ печатается на всю РФ, только за один период - целый месяц
  if ((rep_period_begin = 0) or (rep_period_end = 0) or (rep_period_begin <> rep_period_end) or (report_region_id <> 1000000)) then
    exit;

  aarep_inv_date = report_date_end;

  for select im_invoice_no, reg_code, sum(kcp_perc_calced - kcp_kommis_calced), kcp_from_k_id
      from kontcalcperc, invoicemain, invoicekont, region
      where im_id = ik_from_im_id and
            im_period = :rep_period_end and
            kcp_period = :rep_period_end and
            ik_from_k_id = kcp_from_k_id and
            kcp_from_reg_id = reg_unicode
      group by im_invoice_no, reg_code, kcp_from_k_id
      into aarep_inv_no, reg_code, aarep_inv_amount, k_id
  do
  begin
    aarep_inv_customer = null;

    select c_name_fio
    from kontract, clients
    where k_clients_cashier_id = c_id and
          k_id = :k_id
    into aarep_inv_customer;

    if (aarep_inv_customer is null) then
      select c_name_fio
      from knclient, clients
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      order by kc_status desc
      rows 1
      into aarep_inv_customer;

    aarep_inv_no = right('000000000' || aarep_inv_no, 9) || '/' || left(reg_code, 2) || '-' || substring(reg_code from 3 for 2);
    suspend;
  end
end
