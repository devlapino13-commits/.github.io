CREATE PROCEDURE OKT_CC_PERSON_ANSWER_SET(
    ANSWER_TEXT VARCHAR(1000),
    OPER_ID ID_TYPE,
    CP_ID INTEGER,
    CPC_NO INTEGER,
    EDIT_FORBID INTEGER,
    MEMO STR_255,
    CATEG_STR STR_100)
AS
begin
  update cc_person_calls cpc
  set cpc.cpc_answer_text = :answer_text,
      cpc.cpc_answer_date = current_date,
      cpc.cpc_answer_datetime = current_timestamp,
      cpc.cpc_answer_oper_id = :oper_id,
      cpc.cpc_blocked_oper_id = null,
      cpc.cpc_lawyer_id = :oper_id,
      cpc.cpc_edit_forbid = :edit_forbid,
      cpc.cpc_memo = :memo,
      cpc.cpc_categ_str = :categ_str
  where cpc.cpc_from_cp_id = :cp_id and
        cpc.cpc_no = :cpc_no;
end
