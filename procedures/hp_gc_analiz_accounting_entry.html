CREATE PROCEDURE HP_GC_ANALIZ_ACCOUNTING_ENTRY(
    IFA_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable iga_id d_integer;
declare variable ifa_file_name d_str_100;
declare variable ifa_date d_date;
declare variable id d_id;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable ag_id d_integer;
declare variable terminal_id d_integer;
declare variable trm_ipc_id d_str_20;
declare variable object_id d_id;
declare variable type_op d_smallint;
declare variable ib_id d_integer;
declare variable bank_name d_str_255;
declare variable subaccount d_integer;
declare variable card_no d_str_20;
declare variable trans_no d_str_10;
declare variable trans_dt d_date;
declare variable inet_acquiring d_smallint;
declare variable currency_str d_str_lang;
declare variable amount d_currency;
declare variable commis_amount d_currency;
declare variable commis_amount_mb d_currency;
declare variable commis_amount_other d_currency;
declare variable reg_id d_integer;
declare variable acc_dt d_str_20;
declare variable acc1_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable acc_kt1 d_str_20;
declare variable acc_kt2 d_str_20;
declare variable acc1_kt d_str_20;
declare variable acc2_kt d_str_20;
declare variable acc3_kt d_str_20;
declare variable acc_com_dt d_str_20;
declare variable acc_com_dt1 d_str_20;
declare variable acc_com_kt d_str_20;
declare variable acc1_com_kt d_str_20;
declare variable acc_issuer d_str_20;
declare variable ol_id d_id;
declare variable ol_id1 d_id;
declare variable olcm_id d_id;
declare variable ob_detail d_str_long;
declare variable curr_code d_str_lang;
declare variable res_reg_id d_integer;
declare variable res_acc_dt d_str_20;
declare variable res_acc_kt d_str_20;
declare variable total_line d_integer;
declare variable res_line_no d_integer;
declare variable num d_integer;
begin
  err_text = '';
  curr_code = '417';
  num = 0;

  select
    ifa_file_name,
    ifa_date,
    iga_id
  from ipc_file_analiz
    inner join ipc_gc_analiz
      on iga_from_ifa_id = ifa_id
  where ifa_id = :ifa_id
    and ifa_status = 0
  into
    :ifa_file_name,
    :ifa_date,
    :iga_id;
  if (row_count = 0) then
  begin
    err_text = 'Данные для построения бухгалтерских проводок не определены.';
    suspend;
    exit;
  end

  reg_id = 1090100;
  acc_dt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '10150/101');
  acc_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '10024/1');
  acc_kt1 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '10024/3');
  acc_kt2 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '10024/6');
  acc1_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '21122/1');
  acc2_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '21114/11');
  acc_com_dt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '80509/13');
  acc_com_dt1 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '80509/17');
  acc_com_kt = acc_dt;
  acc1_com_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '21114/12');

  insert into operlist (
    ol_from_reg_id,
    ol_oper_type, 
    ol_oper_subtype,
    ol_date,
    ol_doc_no,
    ol_doc_kind, 
    ol_memo,
    ol_user_id
   )
  values (
    :reg_id,
    2,
    11,
    :ifa_date,
    :ifa_id,
    9,
    f_format('GC-файл "%1" от %2 г.', :ifa_file_name, f_datetostr(:ifa_date)),
    :un_id
  )
  returning ol_id
  into :ol_id;

  -- POS
  for
    select
      igdp_id,
      igdp_from_c_id,
      igdp_from_ag_id,
      igdp_terminal_id,
      trm_ipc_id,
      ito_object_id,
      igdp_type,
      igdp_from_ib_id,
      ib_ipc_bank_name,
      ib_subaccount,
      igdp_card_no,
      igdp_trans_no,
      igdp_trans_date,
      igdp_inet_acquiring,
      igdp_currency,
      igdp_amount,
      igdp_ipc_emitent_commis_amount
    from ipc_gc_analiz_detail_pos
      inner join ipc_banks
        on ib_id = igdp_from_ib_id
      left join ipc_terminals
        on trm_id = igdp_terminal_id
      left join ipc_terminal_ownership
        on ito_trm_id = trm_id
          and :ifa_date between ito_date_begin and ito_date_end
    where igdp_from_iga_id = :iga_id
      and igdp_is_our_card = 0
      and igdp_is_error = 0
  into
    :id,
    :c_id,
    :ag_id,
    :terminal_id,
    :trm_ipc_id,
    :object_id,
    :type_op,
    :ib_id,
    :bank_name,
    :subaccount,
    :card_no,
    :trans_no,
    :trans_dt,
    :inet_acquiring,
    :currency_str,
    :amount,
    :commis_amount
  do
  begin
    num = num + 1;
    acc_issuer = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '10004/' || :subaccount);
    ob_detail = f_format('%1-операция; терминал: %2; транзакция: %3; карта: %4; банк-эмитент: %5',
      trim(iif(inet_acquiring = 0, iif(type_op = 1, 'ТСП', 'ПВН'), 'Эквайринг')),
      case inet_acquiring
        when 0 then trm_ipc_id
        when 1 then 'TS210001'
        when 2 then 'IA210001'
      end, 
      trans_no || ' от ' || f_datetostr(trans_dt),
      card_no,
      bank_name
    );
    if (type_op = 1) then
    begin
      if (c_id > 0) then
      begin
        c_name_fio = (select c_name_fio from clients where c_id = :c_id);
        ob_detail = ob_detail || f_format(';контрагент: %1', c_name_fio);
      end
    end

    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_client_kt,
      ob_detail,
      ob_summa,
      ob_summa_val
    )
    values (
      :ol_id,
      :curr_code,
      1,
      :acc_dt,
      :acc_issuer,
      :c_id,
      :ob_detail,
      :amount,
      :amount
    );

    if (inet_acquiring > 0) then
    begin
      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_client_kt,
        ob_detail,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id,
        :curr_code,
        1,
        :acc_issuer,
        iif(:inet_acquiring = 1,:acc_kt1, :acc_kt2),
        :c_id,
        :ob_detail,
        :amount,
        :amount
      );
    end else
    begin
      if (type_op = 1) then
      begin
        acc3_kt = (select ag_from_al_id from agreements where ag_id = :ag_id);
        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_client_kt,
          ob_detail,
          ob_summa,
          ob_summa_val
        )
        values (
          :ol_id,
          :curr_code,
          1,
          :acc_issuer,
          :acc3_kt,
          :c_id,
          :ob_detail,
          :amount,
          :amount
        );
      end else begin
        acc3_kt = (select al_id from acc_list where al_from_reg_id = :object_id and al_currency = :curr_code and al_account = '10024/2');
        for
          select
            res_reg_id,
            res_acc_dt,
            res_acc_kt,
            total_line,
            res_line_no
          from aarep_loronostro_link(:reg_id, :acc_issuer, :acc3_kt)
        into
          :res_reg_id,
          :res_acc_dt,
          :res_acc_kt,
          :total_line,
          :res_line_no
        do
        begin
          if (res_line_no = 1)
          then
            ol_id1 = ol_id;
          else
            insert into operlist (
              ol_parent_id,
              ol_from_reg_id,
              ol_oper_type,
              ol_oper_subtype,
              ol_date,
              ol_doc_no,
              ol_doc_kind,
              ol_memo,
              ol_user_id
            )
            values (
              :ol_id,
              :res_reg_id,
              2,
              11,
              :ifa_date,
              :ifa_id,
              9,
              f_format('GC-файл "%1" от %2 г.', :ifa_file_name, f_datetostr(:ifa_date)),
              :un_id
            )
            returning ol_id
            into :ol_id1;

            insert into operbook (
              ob_from_ol_id,
              ob_currency,
              ob_curr_rate,
              ob_account_dt,
              ob_account_kt,
              ob_detail,
              ob_summa,
              ob_summa_val
            )
            values (
              :ol_id1,
              :curr_code,
              1,
              :res_acc_dt,
              :res_acc_kt,
              :ob_detail,
              :amount,
              :amount
            );
        end
      end
    end

    if (commis_amount > 0) then
    begin
      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_detail,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id,
        :curr_code,
        1,
        iif(:inet_acquiring = 0, iif(:type_op = 1, :acc_com_dt, :acc_dt), :acc_com_dt1),
        :acc_issuer,
        'Комиссия ' || :ob_detail,
        :commis_amount,
        :commis_amount
      );
      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_detail,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id,
        :curr_code,
        1,
        :acc_issuer,
        iif(:inet_acquiring = 0, iif(:type_op = 1, :acc_com_kt, :acc1_com_kt), :acc_com_kt),
        'Комиссия ' || :ob_detail,
        :commis_amount,
        :commis_amount
      );
    end
  end

  -- QR
  for
    select
      igdq_id,
      igdq_from_ib_id,
      ib_ipc_bank_name,
      ib_subaccount,
      igdq_card_no,
      igdq_trans_no,
      igdq_trans_date,
      igdq_currency,
      igdq_amount,
      igdq_ipc_emitent_commis_amount
    from ipc_gc_analiz_detail_qr
      inner join ipc_banks
        on ib_id = igdq_from_ib_id
    where igdq_from_iga_id = :iga_id
      and igdq_is_our_card = 0
      and igdq_is_error = 0
  into
    :id,
    :ib_id,
    :bank_name,
    :subaccount,
    :card_no,
    :trans_no,
    :trans_dt,
    :currency_str,
    :amount,
    :commis_amount
  do
  begin
    num = num + 1;
    acc_issuer = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '10004/' || :subaccount);
    ob_detail = f_format('QR-операция; транзакция: %1; карта: %2; банк-эмитент: %3',
      trans_no || ' от ' || f_datetostr(trans_dt),
      card_no,
      bank_name
    );
    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_detail,
      ob_summa,
      ob_summa_val
    )
    values (
      :ol_id,
      :curr_code,
      1,
      :acc_dt,
      :acc_issuer,
      :ob_detail,
      :amount,
      :amount
    );
    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_detail,
      ob_summa,
      ob_summa_val
    )
    values (
      :ol_id,
      :curr_code,
      1,
      :acc_issuer,
      :acc_kt,
      :ob_detail,
      :amount,
      :amount
    );
    if (commis_amount > 0) then
    begin
      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_detail,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id,
        :curr_code,
        1,
        :acc_com_dt,
        :acc_issuer,
        'Комиссия ' || :ob_detail,
        :commis_amount,
        :commis_amount
      );
      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_detail,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id,
        :curr_code,
        1,
        :acc_issuer,
        :acc_com_kt,
        'Комиссия ' || :ob_detail,
        :commis_amount,
        :commis_amount
      );
    end
  end

  -- COMMISSION
  for
    select
      igc_id,
      igc_from_c_id,
      c_name_fio,
      igc_from_ag_id,
      igc_amount_cards_mb,
      igc_amount_cards_other
    from ipc_gc_commission
      inner join clients
        on c_id = igc_from_c_id
    where igc_from_iga_id = :iga_id
      and (igc_amount_cards_mb > 0 or igc_amount_cards_other > 0)
  into
    :id,
    :c_id,
    :c_name_fio,
    :ag_id,
    :commis_amount_mb,
    :commis_amount_other
  do
  begin
    num = num + 1;
    acc1_dt = (select ag_from_al_id from agreements where ag_id = :ag_id);
    insert into operlist (
      ol_parent_id,
      ol_from_reg_id,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_document_client,
      ol_memo,
      ol_user_id
     )
    values (
      :ol_id,
      :reg_id,
      2,
      11,
      :ifa_date,
      :ifa_id,
      9,
      :c_id,
      f_format('GC-файл "%1" от %2 г. (комиссия)', :ifa_file_name, f_datetostr(:ifa_date)),
      :un_id
    )
    returning ol_id
    into :olcm_id;

    if (commis_amount_mb > 0) then
    begin
      ob_detail = f_format('Комиссия эквайера по картам МБулак; контрагент: %1', c_name_fio);
      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_detail,
        ob_summa,
        ob_summa_val
      )
      values (
        :olcm_id,
        :curr_code,
        1,
        :acc1_dt,
        :acc2_kt,
        :ob_detail,
        :commis_amount_mb,
        :commis_amount_mb
      );
    end
    if (commis_amount_other > 0) then
    begin
      ob_detail = f_format('Комиссия эквайера по чужим картам; контрагент: %1', c_name_fio);
      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_detail,
        ob_summa,
        ob_summa_val
      )
      values (
        :olcm_id,
        :curr_code,
        1,
        :acc1_dt,
        :acc2_kt,
        :ob_detail,
        :commis_amount_other,
        :commis_amount_other
      );
    end

    update ipc_gc_commission set
      igc_from_ol_id = :olcm_id
    where igc_id = :id;
  end

  if (num > 0) then
  begin
    update ipc_file_analiz set
      ifa_status = 1,
      ifa_from_un_id = :un_id,
      ifa_datetime = current_timestamp,
      ifa_from_ol_id = :ol_id
    where ifa_id = :ifa_id;
    suspend;
  end begin
    if (exists(
      select *
      from ipc_gc_analiz_detail_pos
      where igdp_from_iga_id = :iga_id
        and igdp_is_our_card = 1
        and igdp_is_error = 0))
    then
      update ipc_file_analiz set
        ifa_status = -1,
        ifa_from_un_id = :un_id,
        ifa_datetime = current_timestamp
      where ifa_id = :ifa_id;
    else
      exception e_message 'Данные для построения бухгалтерских проводок не определены.';
  end
end
