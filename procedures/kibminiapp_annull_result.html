CREATE PROCEDURE KIBMINIAPP_ANNULL_RESULT(
    C_ID ID_TYPE)
AS
declare variable kbm_id id_type;
begin
  for select kaa.kbm_id
      from kibminiapp kaa
      where kaa.kbm_from_c_id = :c_id
      order by kaa.kbm_id desc
      rows 1
      into :kbm_id
  do
  begin
    delete from kibhistory kh
    where kh.kh_from_kbm_id = :kbm_id;

    update kibminiapp kaa
    set kaa.kbm_pr_id = null,
        kaa.kbm_pr_id_datetime = null,
        kaa.kbm_is_checked = 0,
        kaa.kbm_checked_datetime = null,
        kaa.kbm_ver_info = 0
    where kbm_id = :kbm_id;
  end
end
