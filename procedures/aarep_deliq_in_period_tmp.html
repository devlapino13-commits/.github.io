CREATE PROCEDURE AAREP_DELIQ_IN_PERIOD_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OKS_FROM_K_ID ID_TYPE,
    OKS_DELINQ_DATE DATE,
    OKS_DATE_END DATE,
    DELINQ_DAY INTEGER,
    REGION_ID INTEGER,
    MAX_DELINQ_DAY INTEGER,
    OKS_BALANCE CURR_TYPE,
    IS_DELINQ SMALLINT)
AS
begin
  for select distinct oks_from_k_id, kal_new_reg_id
      from oksummary, kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1)
      where --   просроченные кредиты на дату конец
            ((:rep_date_end between oks_date_begin and oks_date_end and
            oks_delinq_date <= :rep_date_end)
--   просроченные кредиты на дату начало
            or
            (:rep_date_begin between oks_date_begin and oks_date_end and
            oks_delinq_date <= :rep_date_begin)
--   погашение просроченных кредитов за период
            or
            (oks_date_end >= oks_delinq_date and
            oks_date_end + 1 between :rep_date_begin and :rep_date_end))
--   ограничение по региону
            and
            oks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            kal_new_reg_id = id_list and
            oks_close_state = 0
      into oks_from_k_id, region_id
  do
  begin
--  Вычисления просрчоенного кредита на дату конец
    oks_delinq_date = null;
    delinq_day = null;
    oks_balance = null;
    select oks_delinq_date, :rep_date_end - oks_delinq_date, oks_balance
    from oksummary
    where oks_from_k_id = :oks_from_k_id and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_delinq_date <= :rep_date_end and
          oks_close_state = 0
    into oks_delinq_date, delinq_day, oks_balance;
    if (delinq_day is null) then
      delinq_day = 0;
    if (oks_balance is null) then
      oks_balance = 0;
--  Если контракт просрочен, то вычисляет только максимальный просроченный день
    is_delinq = 0;
    max_delinq_day = 0;
    oks_date_end = null; -- дата закрытия просрочки
    if (oks_delinq_date is not null) then
    begin
      is_delinq = 1;
      select max(oks_date_end - oks_delinq_date + 1)
      from oksummary
      where cast(oks_date_end as date) >= cast(oks_delinq_date as date) and
            cast(oks_date_end as date) < :rep_date_end and
            oks_from_k_id = :oks_from_k_id and
            oks_close_state = 0
      into max_delinq_day;
      if (max_delinq_day is null) then
        max_delinq_day = 0;
      if (delinq_day > max_delinq_day) then
        max_delinq_day = :delinq_day;
    end
    else
--  Если не просрочен то вытаскивает другие данные
    begin
      select oks_delinq_date, oks_date_end, oks_date_end - oks_delinq_date + 1
      from oksummary
      where cast(oks_date_end as date) >= cast(oks_delinq_date as date) and
            cast(oks_date_end as date) < :rep_date_end and
            oks_from_k_id = :oks_from_k_id and
            oks_close_state = 0
      order by 3 desc
      rows 1
      into oks_delinq_date, oks_date_end, max_delinq_day;
    end
    suspend;
  end
end
