CREATE PROCEDURE OKT_VERIFICATION_CANCEL(
    OBJECT_KIND STR_100,
    OBJECT_ID ID_TYPE)
AS
declare variable obj_kind integer;
declare variable acl_id id_type;
declare variable anketa_rd_id integer;
declare variable res smallint;
declare variable rd_cancel_memo str_255;
declare variable cc_id integer;
begin
  /*
      OKTELL
      Процедура отменяет задачи по "Верификации" контактных лиц
  */

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'OKTELL.Projects.Verification'
  into :anketa_rd_id;

  if (object_kind = 'Kontract') then
  begin
    obj_kind = f_objkind('Contacts', 'id');

    select rd.rd_name
    from refdata rd
    where rd.rd_code = 'ActionStatus.OKTELL.Canceled'
    into :rd_cancel_memo;

    for select ao.aco_id, cont.cc_id
        from knclient kncl
        inner join contacts cont on cont.cc_from_c_id = kncl.kc_from_c_id
        inner join actionorder ao on ao.aco_object_id = cont.cc_id and
              ao.aco_object_kind = :obj_kind and
              ao.aco_from_rd_id = :anketa_rd_id
        where kncl.kc_amount_approved > 0 and
              kncl.kc_from_k_id = :object_id
        into :acl_id, :cc_id
    do
    begin
      execute procedure okt_fill_status(:acl_id, 'ActionStatus.OKTELL.Canceled')
          returning_values :res;

      update contacts cont
      set cc_check_result = 1,
          cc_total_res = 1,
          cc_client_ok = 1,
          cc_name_ok = 1,
          cc_status_ok = 1,
          cc_address_ok = 1,
          cc_call_user = 'IGUSEV',
          cc_check_memo = 'Соответствует требованиям процедуры',
          cc_checked_datetime = current_timestamp,
          cc_checked_date = current_date
      where cont.cc_id = :cc_id;
    end
  end
end
