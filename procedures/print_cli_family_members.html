CREATE PROCEDURE PRINT_CLI_FAMILY_MEMBERS(
    C_ID D_ID,
    ADD_PARAMS D_ID)
RETURNS (
    EIL_DATE D_DATE,
    C_NAME_FIO D_STR_255,
    C_INN D_STR_20,
    REL_NAME D_STR_255,
    REL_ROLE D_STR_255,
    MESSAGE_TEXT D_STR_255,
    SHOW_DESCRIPTION D_STR_255,
    JSON_COUNT D_INTEGER)
AS
declare variable JSON D_BLOB_TEXT;
declare variable COUNTER D_INTEGER;
declare variable JSON_ONE D_BLOB_TEXT;
declare variable BLS_ID D_ID;
declare variable K_DISB_DATE D_DATE;
declare variable MSG D_STR_255;
begin
  message_text = '';

  select k.k_disb_date
  from kontract k
  where k.k_id = :add_params
  into k_disb_date;

  k_disb_date = coalesce(k_disb_date, current_date);

  select eil.eil_date, eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id and
        eic.eic_code = 'family_members'
  where eil.eil_from_c_id = :c_id and
        eil.eil_is_json = 1 and
        eil.eil_status = 2 and
        eil.eil_date <= :k_disb_date --RFCKG-3426
  order by eil.eil_id desc
  rows 1
  into eil_date, bls_id;

  if (bls_id is null) then -- Если не удалось найти бланк берем последний
  begin
    select eil.eil_date, eil.eil_from_bls_id
    from extinfolist eil
    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id and
        eic.eic_code = 'family_members'
    where eil.eil_from_c_id = :c_id and
          eil.eil_is_json = 1 and
          eil.eil_status = 2
    order by eil.eil_id desc
    rows 1
    into eil_date, bls_id;

    eil_date = k_disb_date;
  end

  select c_name_fio, c_inn
  from clients
  where c_id = :c_id
  into c_name_fio, c_inn;

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into :json;

  json_count = f_json_blob_get_count(:json, 'members');
  counter = 0;

  msg = f_json_str_get_value(:json, 'message', '');
  if (msg = 'person has no family') then
    begin
       show_description = 'По данному клиенту в информации от ГРС отсутствуют актуальные данные';
       suspend;
    end


  while (json_count > counter) do
  begin
    json_one = f_json_blob_get_value(:json, f_format('members[%1]', :counter), '');
    rel_name = f_json_str_get_value(:json_one, 'name', '');
    rel_role = f_json_str_get_value(:json_one, 'role', '');

    counter = counter + 1;
    suspend;
  end
end
