CREATE PROCEDURE HP_OPERKROFF_SPLIT_PAYMENT(
    K_ID ID_TYPE,
    OS_PAYMENT_AMOUNT CURR_TYPE,
    PERC_PAYMENT_AMOUNT CURR_TYPE,
    IS_RECURSIVE_EXEC SMALLINT = 0)
RETURNS (
    C_ID ID_TYPE,
    ZAD_OS_PRIMARY CURR_TYPE,
    ZAD_PERC_PRIMARY CURR_TYPE,
    PAY_OS_FACT_RESULT CURR_TYPE,
    PAY_PERC_FACT_RESULT CURR_TYPE,
    STORE_OS CURR_TYPE,
    STORE_PERC CURR_TYPE)
AS
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable kont_part_perc double precision;
declare variable pay_os curr_type;
declare variable pay_perc curr_type;
declare variable k_amount_approved curr_type;
declare variable kont_own_amount curr_type;
declare variable os_payment_amount_tmp curr_type;
declare variable perc_payment_amount_tmp curr_type;
declare variable diff_os curr_type;
declare variable diff_perc curr_type;
declare variable store_os_tmp curr_type;
declare variable store_perc_tmp curr_type;
declare variable zad_os_total curr_type;
declare variable zad_perc_total curr_type;
declare variable cli_qty integer;
declare variable alone_c_id id_type;
begin
--K_ID = 101010000054600101;

-- Процедура предназначена для распределения суммы оплаты по внесистемному учету между участниками группы

  if (is_recursive_exec = 0) then
  begin
    delete from temp_operkroff_repay;

    select kont.k_amount_approved
    from kontract kont
    where kont.k_id = :k_id
    into :k_amount_approved;

-- шаг 1. Получаю задолженность каждого клиента, долю каждого участника группы из общей суммы займа.
    zad_os_total = 0;
    zad_perc_total = 0;
    cli_qty = 0;

    for select okf.oko_from_c_id, coalesce(sum(okf.oko_os + okf.oko_sh_os), 0),
               coalesce(sum(okf.oko_perc + okf.oko_sh_perc), 0)
        from operkroff okf
        where okf.oko_from_k_id = :k_id and
              okf.oko_ol_is_active = 1
        group by okf.oko_from_c_id
        order by okf.oko_from_c_id
        into :c_id, :zad_os, :zad_perc
    do
    begin

-- MMusabaev 22.01.2019 если не погашаем проценты
      if (perc_payment_amount = 0) then
      begin
        zad_perc = 0;
      end
-- MMusabaev 22.01.2019

      if ((zad_os > 0) or (zad_perc > 0)) then
      begin
        alone_c_id = c_id;
        cli_qty = cli_qty + 1;
      end

      select sum(kncl.kc_amount_approved),
             cast(sum(kncl.kc_amount_approved) as double precision) * 100 / :k_amount_approved
      from knclient kncl
      where kncl.kc_from_k_id = :k_id and
            kncl.kc_from_c_id = :c_id
      into :kont_own_amount, :kont_part_perc;

      insert into temp_operkroff_repay (t_from_c_id, t_zad_os, t_zad_perc, t_kont_own_amount, t_kont_part_perc,
                                        t_zad_os_primary, t_zad_perc_primary)
      values (:c_id, :zad_os, :zad_perc, :kont_own_amount, :kont_part_perc, :zad_os, :zad_perc);

      zad_os_total = zad_os_total + zad_os;
      zad_perc_total = zad_perc_total + zad_perc;
    end

    if ((os_payment_amount > zad_os_total) and
        (os_payment_amount <> 0)) then
    begin
      exception e_default 'sp: HP_OPERKROFF_SPLIT_PAYMENT; Error: payment_OS > delinq_OS';
      exit;
    end

    if (perc_payment_amount > zad_perc_total) then
    begin
      exception e_default 'sp: HP_OPERKROFF_SPLIT_PAYMENT; Error: payment_PERC > delinq_PERC';
      exit;
    end

    if (cli_qty = 1) then -- если на внеситсемке только один участник группы, сажаю всю сумму на него
    begin
      update temp_operkroff_repay t
      set t.t_pay_os_fact_result = :os_payment_amount,
          t.t_pay_perc_fact_result = :perc_payment_amount
      where t.t_from_c_id = :alone_c_id;

      for select t_from_c_id, t_zad_os_primary, t_zad_perc_primary, t_pay_os_fact_result, t_pay_perc_fact_result
          from temp_operkroff_repay
          into :c_id, zad_os_primary, zad_perc_primary, pay_os_fact_result, pay_perc_fact_result
      do
      begin
        suspend;
      end

      exit; -- важно!
    end
  end

  update temp_operkroff_repay t
  set t.t_payment_part_os = 0,
      t.t_payment_part_perc = 0,
      t.t_pay_os = 0,
      t.t_pay_perc = 0,
      t.t_pay_os_fact = 0,
      t.t_pay_perc_fact = 0;

-- >> рассчитываю, какая доля в %-ом соотношении, кому полагается, исходя из его собственной суммы займа
  if (exists(select *
             from temp_operkroff_repay t
             where t.t_zad_os = 0 or
                   (t.t_zad_perc = 0))) then
  begin
    update temp_operkroff_repay t
    set t.t_payment_part_os = t.t_kont_part_perc * 100 / (select sum(t.t_kont_part_perc)
                                                          from temp_operkroff_repay t
                                                          where t.t_zad_os > 0)
    where t.t_zad_os > 0;

    update temp_operkroff_repay t
    set t.t_payment_part_perc = t.t_kont_part_perc * 100 / (select sum(t.t_kont_part_perc)
                                                            from temp_operkroff_repay t
                                                            where t.t_zad_perc > 0)
    where t.t_zad_perc > 0;
  end
  else
  begin
    update temp_operkroff_repay t
    set t.t_payment_part_os = t.t_kont_part_perc,
        t.t_payment_part_perc = t.t_kont_part_perc;
  end

-- << ------------------------------------------------------------------------
-- << определяю кому какая сумма полагается в рамках его доли
  update temp_operkroff_repay t
  set t.t_pay_os = round(cast(t.t_payment_part_os as double precision) * :os_payment_amount / 100, 2),
      t.t_pay_perc = round(cast(t.t_payment_part_perc as double precision) * :perc_payment_amount / 100, 2); -- << ------------------------------------------------------------------------

--<< если выявлена разница в итоговых суммах, т.е. например на платеж на 15 000 рублей, а распределилось 15 001 или 14 9999
  select sum(t.t_pay_os), sum(t.t_pay_perc)
  from temp_operkroff_repay t
  into :os_payment_amount_tmp, perc_payment_amount_tmp;

  diff_os = os_payment_amount_tmp - :os_payment_amount;
  diff_perc = perc_payment_amount_tmp - :perc_payment_amount;

  if (diff_os > 0) then
  begin
    for select t.t_from_c_id, t.t_pay_os
        from temp_operkroff_repay t
        where t.t_pay_os > 0
        into :c_id, :pay_os
    do
    begin
      if (diff_os > 0) then
      begin
        if (diff_os > pay_os) then
        begin
          update temp_operkroff_repay t
          set t.t_pay_os = 0
          where t.t_from_c_id = :c_id;

          diff_os = diff_os - pay_os;
        end
        else
        begin
          update temp_operkroff_repay t
          set t.t_pay_os = t_pay_os - :diff_os
          where t.t_from_c_id = :c_id;

          diff_os = 0;
        end
      end
    end
  end

  if (diff_os < 0) then
  begin
    c_id = null;

    select t.t_from_c_id
    from temp_operkroff_repay t
    where t.t_pay_os > 0 rows 1
    into :c_id;

    if (c_id is not null) then
    begin
      update temp_operkroff_repay t
      set t.t_pay_os = t_pay_os + (:diff_os * -1)
      where t.t_from_c_id = :c_id;
    end
  end

  if (diff_perc > 0) then
  begin
    for select t.t_from_c_id, t.t_pay_perc
        from temp_operkroff_repay t
        where t.t_pay_os > 0
        into :c_id, :pay_perc
    do
    begin
      if (diff_perc > 0) then
      begin
        if (diff_perc > pay_perc) then
        begin
          update temp_operkroff_repay t
          set t.t_pay_perc = 0
          where t.t_from_c_id = :c_id;

          diff_perc = diff_perc - pay_perc;
        end
        else
        begin
          update temp_operkroff_repay t
          set t.t_pay_perc = t_pay_perc - :diff_perc
          where t.t_from_c_id = :c_id;

          diff_perc = 0;
        end
      end
    end
  end
  if (diff_perc < 0) then
  begin
    c_id = null;

    select t.t_from_c_id
    from temp_operkroff_repay t
    where t.t_pay_perc > 0 rows 1
    into :c_id;

    if (c_id is not null) then
    begin
      update temp_operkroff_repay t
      set t.t_pay_perc = t_pay_perc + (:diff_perc * -1)
      where t.t_from_c_id = :c_id;
    end
  end
-- << --
-- определяю какая сумма фактически может быть принята клиентом
  update temp_operkroff_repay t
  set t.t_pay_os_fact = t.t_pay_os_fact + iif(t.t_zad_os >= t.t_pay_os, t.t_pay_os, t.t_zad_os),
      t.t_pay_perc_fact = t.t_pay_perc_fact + iif(t.t_zad_perc >= t.t_pay_perc, t.t_pay_perc, t.t_zad_perc);

--<< получаю оставшуюся сумму которую распределить не удалось, это сумма которую клинты принять не смогли т.к. их задолженности были закрыты
  store_os = 0;
  store_perc = 0;

  for select t.t_from_c_id, iif(t.t_pay_os > t.t_zad_os, t.t_pay_os - t.t_zad_os, 0),
             iif(t.t_pay_perc > t.t_zad_perc, t.t_pay_perc - t.t_zad_perc, 0)
      from temp_operkroff_repay t
      into :c_id, :store_os_tmp, :store_perc_tmp
  do
  begin
    store_os = store_os_tmp + store_os;
    store_perc = store_perc_tmp + store_perc;
  end
-- << --
-- << корректирую задолженность клиента с учетом суммы которая закрывает задолженность --
  update temp_operkroff_repay t
  set t.t_zad_os = t.t_zad_os - t.t_pay_os_fact,
      t.t_zad_perc = t.t_zad_perc - t.t_pay_perc_fact,
      t.t_pay_os_fact_result = t.t_pay_os_fact_result + t.t_pay_os_fact,
      t.t_pay_perc_fact_result = t.t_pay_perc_fact_result + t.t_pay_perc_fact;
-- << --

-- << -- если осталась сумма, которую распределить не удалось, тогда вызываю процедуру рекурсивно пока сумма вся не распределится между участниками группы, у которых есть задолженность
-- нераспределенная сумма может остаться в случае если в групповом займе участнику группы исходя из его доли полагается например 1000р. по ОС,
-- но, его долг составляет всего 200р. по ОС, таким образом 800 уйдут в нераспреденную сумму и будут распределены уже в будующих итерациях между участниками у которых осталась задолженность по ОС
  while ((store_os > 0) or (store_perc > 0)) do
  begin
    select store_os, store_perc
    from hp_operkroff_split_payment(:k_id, :store_os, :store_perc, 1)
    rows 1
    into store_os, store_perc;
  end

-- MMusabaev 22.01.2019 Если есть разница между суммой погашения и общей суммой распределения
-- то остаток закинем участнику с маскимальной задолженностью
  select :os_payment_amount - sum(t_pay_os_fact_result), :perc_payment_amount - sum(t_pay_perc_fact_result)
  from temp_operkroff_repay
  into :pay_os_fact_result, :pay_perc_fact_result;

  if (pay_os_fact_result > 0) then
  begin
    update temp_operkroff_repay
    set t_pay_os_fact_result = t_pay_os_fact_result + :pay_os_fact_result
    where t_zad_os_primary = (select max(tor2.t_zad_os_primary)
                              from temp_operkroff_repay tor2) rows 1;
  end
  if (pay_perc_fact_result <> 0) then
  begin
    update temp_operkroff_repay
    set t_pay_perc_fact_result = t_pay_perc_fact_result + :pay_perc_fact_result
    where t_zad_perc_primary = (select max(tor2.t_zad_perc_primary)
                                from temp_operkroff_repay tor2) rows 1;
  end
-- MMusabaev 22.01.2019

  for select t_from_c_id, t_zad_os_primary, t_zad_perc_primary, t_pay_os_fact_result, t_pay_perc_fact_result
      from temp_operkroff_repay
      into :c_id, zad_os_primary, zad_perc_primary, pay_os_fact_result, pay_perc_fact_result
  do
  begin
    suspend;
  end
end
