CREATE PROCEDURE HP_GET_KN_OKSUMMARY_ESP_ALL(
    OKS_DATE_BEGIN D_DATE)
RETURNS (
    OKS_FROM_KC_ID D_ID,
    RES_OKS_BAL_ESP_NORM_TMP D_CURRENCY,
    RES_OKS_BAL_ESP_NORM D_CURRENCY,
    RES_OKS_BAL_ESP_DELINQ D_CURRENCY,
    RES_SH_COUNT D_INTEGER,
    RES_DELINQ_DATE D_DATE,
    RES_STOP_PERC D_DATE,
    RES_FUTURE_PERC D_CURRENCY,
    RES_FUTURE_VBS_4 D_CURRENCY,
    RES_EAD D_CURRENCY,
    OKS_BAL_ESP_END D_CURRENCY,
    OKS_PMT_ALL_ADV D_CURRENCY,
    OKS_PMT_IS_ADV D_SMALLINT,
    K_KONTRACT_NO D_STR_100,
    K_DISB_DATE D_DATE)
AS
declare variable tmp_oks_date_prv d_date;
declare variable tmp_oks_close_state d_smallint;
declare variable tmp_oks_percent_esp d_percent;
declare variable tmp_oks_date_stop_perc d_date;
declare variable tmp_oks_bal_esp_tmp d_currency;
declare variable tmp_oks_vbs_end d_currency;
declare variable tmp_oks_pmt_all_adv d_currency;
declare variable tmp_oks_pmt_os_norm d_currency;
declare variable tmp_oks_pmt_perc_norm d_currency;
declare variable tmp_oks_bal_perc d_currency;
declare variable tmp_oks_pmt_perc_delinq d_currency;
declare variable tmp_oks_bal_zad_perc_prv d_currency;
declare variable tmp_oks_delinq_date d_date;
begin
  res_oks_bal_esp_norm_tmp = 0;
  res_oks_bal_esp_norm = 0;
  res_oks_bal_esp_delinq = 0;

  for select oks_from_kc_id, oks_date_prv, oks_close_state, oks_percent_esp, oks_date_stop_perc, oks_delinq_date,
             oks_bal_esp_tmp, oks_vbs_end, oks_pmt_all_adv, oks_pmt_os_norm, oks_pmt_perc_norm, oks_bal_perc,
             oks_pmt_perc_delinq, oks_bal_zad_perc_prv, oks_bal_esp_end, k_kontract_no, k_disb_date, oks_pmt_all_adv,
             iif((((oks_pmt_is_adv <> 0 or oks_date_begin = oks_date_stop_perc) and oks_date_begin <= coalesce(oks_date_stop_perc, oks_date_begin)) or oks_pmt_other_adv <> 0 or (oks_date_begin > oks_date_stop_perc and oks_date_begin <= oks_last_sh_date and oks_acr_perc <> 0)) and oks_is_flat = 0, 1, 0)
      from kn_oksummary, kontract
      where oks_date_stop_perc is null and
            oks_bal_esp_end <> 0 and
            oks_close_state = 0 and
            oks_date_begin = :oks_date_begin and
            k_id = oks_from_k_id
      into oks_from_kc_id, tmp_oks_date_prv, tmp_oks_close_state, tmp_oks_percent_esp, tmp_oks_date_stop_perc,
           tmp_oks_delinq_date, tmp_oks_bal_esp_tmp, tmp_oks_vbs_end, tmp_oks_pmt_all_adv, tmp_oks_pmt_os_norm,
           tmp_oks_pmt_perc_norm, tmp_oks_bal_perc, tmp_oks_pmt_perc_delinq, tmp_oks_bal_zad_perc_prv, oks_bal_esp_end,
           k_kontract_no, k_disb_date, oks_pmt_all_adv, oks_pmt_is_adv
  do
  begin
    select res_oks_bal_esp_norm_tmp, res_oks_bal_esp_norm, res_oks_bal_esp_delinq, res_delinq_date, res_stop_perc,
           res_future_perc, res_future_vbs_4, res_ead, res_sh_count
    from hp_get_kn_oksummary_esp_data(:oks_from_kc_id, :oks_date_begin, :tmp_oks_date_prv, :tmp_oks_delinq_date, :tmp_oks_close_state, :tmp_oks_percent_esp, :tmp_oks_date_stop_perc, :tmp_oks_bal_esp_tmp, :tmp_oks_vbs_end, :tmp_oks_pmt_all_adv, :tmp_oks_pmt_os_norm, :tmp_oks_pmt_perc_norm, :tmp_oks_bal_perc, :tmp_oks_pmt_perc_delinq, :tmp_oks_bal_zad_perc_prv, :oks_pmt_is_adv)
    into :res_oks_bal_esp_norm_tmp, :res_oks_bal_esp_norm, :res_oks_bal_esp_delinq, :res_delinq_date, :res_stop_perc,
         :res_future_perc, :res_future_vbs_4, :res_ead, res_sh_count;
    suspend;
  end
end
