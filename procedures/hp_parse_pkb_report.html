CREATE PROCEDURE HP_PARSE_PKB_REPORT(
    BLS_ID D_ID)
RETURNS (
    SURNAME D_STR_255,
    NAME D_STR_255,
    FATHERS_NAME D_STR_255,
    C_NAME_FIO D_STR_255,
    DOB D_DATE,
    PRINCIPAL_AMOUNT DECIMAL(18,2),
    DOC_NUMBER D_STR_100,
    IS_GAMBLING D_SMALLINT,
    IS_STOP_CREDIT D_SMALLINT,
    IS_BANKRUPTCY D_SMALLINT,
    IS_TAX_DEBT D_SMALLINT,
    IS_FAKE_BUSINESS D_SMALLINT,
    IS_BAD_SUPPLIER D_SMALLINT,
    IS_BLACK_LIST D_SMALLINT,
    IS_WANTED_PERSON D_SMALLINT,
    IS_WANTED_ALIMENT D_SMALLINT,
    ALL_CREDIT_HISTORY_COUNT D_INTEGER,
    DATE_OF_CREDIT_START D_DATE,
    DATE_OF_CREDIT_END D_DATE,
    LOAN_TERM D_INTEGER,
    TOTAL_AMOUNT DECIMAL(18,2),
    OUTSTANDING_AMOUNT_SUM DECIMAL(15,2),
    OUTSTANDING_AMOUNT DECIMAL(15,2),
    CONTRACT_STATUS D_SMALLINT,
    PAYMENT_STATUS D_STR_20,
    TYPE_CONTRACT D_STR_255,
    ACTIVE_KONT D_INTEGER,
    PORUCH_KONT_QTY D_INTEGER,
    ALL_K_OVERDUE_MAX D_INTEGER,
    EX_OVERDUE_MAX D_INTEGER,
    NUMBER_OF_CONTRACTS D_INTEGER,
    TE_CR_HISTORY_COUNT D_INTEGER,
    FINANCIAL_INSTITUTION D_STR_255,
    PORUCH_NAME D_STR_255,
    K_NO D_INTEGER,
    CURRENCY_CODE D_STR_10,
    TOTAL_MONTHLY_INS_AMOUNT DECIMAL(15,2),
    CNT_OVERDUE_10D D_INTEGER,
    CNT_OVERDUE_60D D_INTEGER)
AS
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable JSON_HEADER D_BLOB_TEXT;
declare variable JSON_SUM_INFO D_BLOB_TEXT;
declare variable JSON_PUBLIC_SOURCES D_BLOB_TEXT;
declare variable CNTKONTR D_INTEGER;
declare variable CNTSUMEXIST D_INTEGER;
declare variable SUS_CTN_COUNT D_INTEGER;
declare variable IDX D_INTEGER;
declare variable IDX2 D_INTEGER;
declare variable HAS_CONTRACTS D_SMALLINT;
declare variable BASEPATH D_STR_255;
declare variable SUBPATH D_STR_255;
declare variable TMPSTR D_STR_100;
declare variable TMPBLOB D_BLOB_TEXT;
declare variable OVD D_INTEGER;
declare variable ISTERMINATED D_SMALLINT;
declare variable POS D_INTEGER;
declare variable DOC_COUNT D_INTEGER;
declare variable DOC_PATH D_STR_255;
declare variable DOC_ITEM D_BLOB_TEXT;
declare variable TEMP_DOC_NAME D_STR_64;
declare variable TYPE_OF_GUARANTEE D_STR_255;
declare variable TYPE_OF_FOUNDING D_STR_255;
declare variable TEMP_OUT_AMOUNT decimal(15,2);
declare variable IS_ARRAY D_SMALLINT;
declare variable MONTHLY_INS_AMOUNT decimal(15,2);
declare variable FOUNDING_TYPE_STR D_STR_10;
begin

  active_kont = 0;
  outstanding_amount_sum = 0;
  poruch_kont_qty = 0;
  all_k_overdue_max = 0;
  te_cr_history_count = 0;
  poruch_name = '';
  k_no = 1;
  total_monthly_ins_amount = 0;
  cnt_overdue_10d = 0;
  cnt_overdue_60d = 0;

  select cast(blx_blob as d_blob_text)
  from hp_get_blobex_blob(:bls_id)
  into :json_text;
  if (json_text is null) then
    exit;

  -- заголовок
  json_header = f_json_blob_get_value(json_text, 'Report.Root.Header', '');
  surname = f_json_blob_get_value(json_header, 'Surname.value', '');
  name = f_json_blob_get_value(json_header, 'Name.value', '');
  fathers_name = f_json_blob_get_value(json_header, 'FathersName.value', '');
  dob = cast(nullif(trim(f_json_blob_get_value(json_header, 'DateOfBirth.value', '')), '') as d_date);
  c_name_fio = f_translit_fio_grs('orig','orig',surname || ' ' || name || ' ' || fathers_name,'kz');

  -- ищем и берем нужный документ (Name.id = 7)
  doc_count = f_json_blob_get_count(json_text, 'Report.Root.IdentificationDocuments.Document');
  idx = 0;
  while (idx < doc_count) do
  begin
    doc_path = 'Report.Root.IdentificationDocuments.Document[' || idx || ']';
    doc_item = f_json_blob_get_value(json_text, doc_path, '');
    if (doc_item is null or doc_item = '') then
    begin
      idx = idx + 1; continue;
    end
    temp_doc_name = f_json_str_get_value(doc_item, 'Name.id', '');
    if (temp_doc_name = '7') then
    begin
      doc_number = f_json_str_get_value(doc_item, 'Number.value', '');
      leave;
    end
    idx = idx + 1;
  end
  ----------------------------summaryinformation--------------------------------
  is_array = 1;
  cntsumexist = f_json_blob_get_count(json_text, 'Report.Root.SummaryInformation.ExistingContracts.SubjectRole');
  json_sum_info = f_json_blob_get_value(json_text, 'Report.Root.SummaryInformation.ExistingContracts', '');
  if (cntsumexist = -1) then
  begin
    tmpstr = trim(f_json_blob_get_value(json_text, 'Report.Root.SummaryInformation.ExistingContracts.SubjectRole.id', ''));
    if (tmpstr <> '') then
    begin
      is_array = 0;
      cntsumexist = 1; -- отметка: это не массив, а единичный объект
    end
    else
      cntsumexist = 0; -- ничего нет — не заходить в цикл
  end
  idx = 0;
  while (idx < cntsumexist) do
  begin
    if (is_array = 1) then
      basepath = 'SubjectRole[' || idx || ']';
    else
      basepath = 'SubjectRole';

    tmpstr = f_json_str_get_value(json_sum_info, basepath || '.id', '');
    if (tmpstr = '1') then
    begin
      tmpstr = f_clean_value(f_json_str_get_value(json_sum_info, basepath || '.TotalOutstandingDebt.value', ''));
      principal_amount = case when tmpstr <> '' then cast(tmpstr as decimal(18,2)) else 0 end;

      tmpstr = f_clean_value(f_json_str_get_value(json_sum_info, basepath || '.NumberOfContracts.value', ''));
      number_of_contracts = case when tmpstr <> '' then cast(tmpstr as d_integer) else 0 end;

      leave;
    end
    idx = idx + 1;
  end
  ------------------------------------------------------------------------------
  all_credit_history_count = number_of_contracts;
  is_array = 1;
  cntsumexist = f_json_blob_get_count(json_text, 'Report.Root.SummaryInformation.TerminatedContracts.SubjectRole');
  json_sum_info = f_json_blob_get_value(json_text, 'Report.Root.SummaryInformation.TerminatedContracts', '');
  if (cntsumexist = -1) then
  begin
    -- пробуем прочитать как одиночный объект
    tmpstr = trim(f_json_blob_get_value(json_text, 'Report.Root.SummaryInformation.TerminatedContracts.SubjectRole.id', ''));
    if (tmpstr <> '') then
    begin
      is_array = 0;
      cntsumexist = 1; -- отметка: это не массив, а единичный объект
    end
    else
      cntsumexist = 0; -- ничего нет — не заходить в цикл
  end
  idx = 0;
  while (idx < cntsumexist) do
  begin
    if (is_array = 1) then
      basepath = 'SubjectRole[' || idx || ']';
    else
      basepath = 'SubjectRole';

    tmpstr = f_json_str_get_value(json_sum_info, basepath || '.id', '');
    if (tmpstr = '1') then
    begin
      tmpstr = f_clean_value(f_json_str_get_value(json_sum_info, basepath || '.NumberOfContracts.value', ''));
      te_cr_history_count = case when tmpstr <> '' then cast(tmpstr as d_integer) else 0 end;
      all_credit_history_count = number_of_contracts + te_cr_history_count;
      leave;
    end
    idx = idx + 1;
  end
  ------------------------------------------------------------------------------
  json_public_sources = f_json_blob_get_value(json_text, 'Report.Root.PublicSources', '');
  -- информация из доп источников
  is_gambling = iif(f_json_blob_get_value(json_text, 'Report.Root.Gamblers.GamblerFeature', '') = 'true', 1, 0);
  is_stop_credit = iif(f_json_blob_get_value(json_text, 'Report.Root.StopCredit', '') = 'true', 1, 0);
  is_bankruptcy = iif(f_json_blob_get_count(json_text, 'Report.Root.NegativeData.NegativeStatus') >= 0
    and f_json_blob_get_value(json_text,'Report.Root.NegativeData.NegativeStatus[0].NegativeStatusOfClient.NegativeStatusOfClient.id', '')in ('17','18'), 1, 0);
  is_tax_debt = iif(f_json_blob_get_value(json_public_sources, 'Areears.Status.id', '') = '1', 1, 0);
  is_fake_business = iif(f_json_blob_get_value(json_public_sources, 'FalseBusi.Status.id', '') = '1', 1, 0);
  is_bad_supplier = iif(f_json_blob_get_value(json_public_sources, 'RNUGosZAKUP.Status', '') = '1', 1, 0);
  is_black_list = iif(f_json_blob_get_value(json_public_sources, 'L150o10.Status.id', '') = '1', 1, 0);
  is_wanted_person = iif(f_json_blob_get_value(json_public_sources, 'QamqorList.Status.id', '') = '1', 1, 0);
  is_wanted_aliment = iif(f_json_blob_get_value(json_public_sources, 'QamqorAlimony.Status.id', '') = '1', 1, 0);
  ------------------------------------------------------------------------------
  cntkontr = f_json_blob_get_count(json_text, 'Report.Root.ExistingContracts.Contract');
  has_contracts = 0;
  sus_ctn_count = 0;

  idx = 0;
  idx2 = 0;
  is_array = 1;
  if (cntkontr = -1) then
  begin
    -- пробуем прочитать как одиночный объект
    tmpblob = trim(f_json_blob_get_value(json_text, 'Report.Root.ExistingContracts.Contract.ContractType', ''));
    if (tmpblob <> '') then
    begin
      is_array = 0;
      cntkontr = 1; -- отметка: это не массив, а единичный объект
    end
    else
      cntkontr = 0; -- ничего нет — не заходить в цикл
  end
  active_kont = cntkontr;
  while (idx < cntkontr) do
  begin
    has_contracts = 1;
    isterminated = 0;
    pos = idx;
    if (is_array = 1) then
      basepath = 'Report.Root.ExistingContracts.Contract[' || pos || ']';
    else
      basepath = 'Report.Root.ExistingContracts.Contract';

    financial_institution = f_json_blob_get_value(json_text, basepath || '.FinancialInstitution.value', '');

    -- вычисление ex_overdue_max для текущего контракта
    ex_overdue_max = cast(f_json_blob_get_value(json_text, basepath || '.NumberOfOverdueInstalmentsMax.value', '0') as d_integer);   ---

    if (f_json_blob_get_count(json_text, basepath || '.Collateral') >= 0) then
      type_of_guarantee = f_json_blob_get_value(json_text, basepath || '.Collateral[0].TypeOfGuarantee.value','');
    else
      type_of_guarantee = f_json_blob_get_value(json_text, basepath || '.Collateral.TypeOfGuarantee.value','');

    type_of_founding = f_json_blob_get_value(json_text, basepath || '.TypeOfFounding.value','');
    if (type_of_guarantee = 'Бланковые') then
      type_contract = type_of_founding || ' беззалоговый';
    else
      type_contract = type_of_founding || ' залоговый';

    founding_type_str = f_json_blob_get_value(json_text, basepath || '.TypeOfFounding.id', '');

    if (founding_type_str = '9') then
      tmpstr = f_clean_value(f_json_blob_get_value(json_text, basepath || '.ResidualAmount.value', ''));
    else
      tmpstr = f_clean_value(f_json_blob_get_value(json_text, basepath || '.OutstandingAmount.value', ''));
    outstanding_amount = case when tmpstr <> '' then cast(tmpstr as decimal(15,2)) else 0 end;

    if (founding_type_str = '9') then
      tmpstr = f_clean_value(f_json_blob_get_value(json_text, basepath || '.CreditLimit.value', ''));
    else
      tmpstr = f_clean_value(f_json_blob_get_value(json_text, basepath || '.TotalAmount.value', ''));
    total_amount = case when tmpstr <> '' then cast(tmpstr as decimal(15,2)) else 0 end;

    while (idx2 < cntkontr) do
    begin
      if (is_array = 1) then
        subpath = 'Report.Root.ExistingContracts.Contract[' || idx2 || ']';
      else
        subpath = 'Report.Root.ExistingContracts.Contract';

      -- обновление all_k_overdue_max (макс по всем)
      ovd = cast(f_json_blob_get_value(json_text, subpath || '.NumberOfOverdueInstalmentsMax.value', '0') as d_integer);
      all_k_overdue_max = maxvalue(ovd, all_k_overdue_max);

      if (ovd >= 10) then     -- RFCKZ-129
        cnt_overdue_10d = cnt_overdue_10d + 1;
      if (ovd >= 60) then
        cnt_overdue_60d = cnt_overdue_60d + 1;

      if (f_json_blob_get_value(json_text, subpath || '.TypeOfFounding.id', '') = '9') then
        tmpstr = f_clean_value(f_json_blob_get_value(json_text, subpath || '.ResidualAmount.value', ''));
      else
        tmpstr = f_clean_value(f_json_blob_get_value(json_text, subpath || '.OutstandingAmount.value', ''));

      temp_out_amount = case when tmpstr <> '' then cast(tmpstr as decimal(15,2)) else 0 end;
      outstanding_amount_sum = outstanding_amount_sum + temp_out_amount;

      tmpstr = f_clean_value(f_json_blob_get_value(json_text, subpath || '.MonthlyInstalmentAmount.value', ''));
      monthly_ins_amount = case when tmpstr <> '' then cast(tmpstr as decimal(15,2)) else 0 end;
      total_monthly_ins_amount = total_monthly_ins_amount + monthly_ins_amount;

      idx2 = idx2 + 1;
    end 

    -- даты и сроки
    tmpstr = f_json_blob_get_value(json_text, basepath || '.DateOfCreditStart.value', '');
    date_of_credit_start = case when tmpstr <> '' then cast(tmpstr as d_date) else null end;
    tmpstr = f_json_blob_get_value(json_text, basepath || '.DateOfCreditEnd.value', '');
    date_of_credit_end = case when tmpstr <> '' then cast(tmpstr as d_date) else null end;
    if (date_of_credit_start is not null and date_of_credit_end is not null) then
      loan_term = datediff(month from date_of_credit_start to date_of_credit_end);

    payment_status = f_json_blob_get_value(json_text, basepath || '.NumberOfOverdueInstalments.value', '0');
    contract_status = iif(isterminated = 0, 1, 0);

    -- поручители
    if (f_json_blob_get_value(json_text, basepath || '.InterconnectedSubjects.TypeOfLink.value', '') containing 'Поручитель') then
    begin
      poruch_name = f_json_blob_get_value(json_text, basepath || '.InterconnectedSubjects.InterconnectedSubject.SubjectName.value', '');
      poruch_kont_qty = poruch_kont_qty + 1;
    end

    -- валюта
    currency_code = f_json_blob_get_value(json_text, basepath || '.CurrencyCode.value', '');

    sus_ctn_count = sus_ctn_count + 1;
    suspend;

    idx = idx + 1;
    k_no = k_no + 1;
  end
  if (has_contracts = 0) then
    suspend;
end
