CREATE PROCEDURE MC_VISIT_USER_CREATE(
    PHONE_NO STR_100,
    C_ID ID_TYPE)
AS
declare variable mcl_id id_type;
declare variable mcl_id_old id_type;
declare variable login_cnt integer;
begin
  select mcl.mcl_id
  from mc_login mcl
  where mcl.mcl_login = :phone_no
  into :mcl_id_old;

  if (mcl_id_old is not null) then
  begin
      select count(*)
      from mc_login
      where mcl_login starting :phone_no
      into login_cnt;

      update mc_login
      set mcl_login = :phone_no || '_' || lpad(:login_cnt, 3, '0')
      where mcl_id = :mcl_id_old;
  end

  insert into mc_login (mcl_login, mcl_from_c_id)
  values (:phone_no, :c_id)
  returning mcl_id
  into :mcl_id;

  insert into mc_visit_login (mvl_mcl_id)
  values (:mcl_id);
end
