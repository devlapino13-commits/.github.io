CREATE PROCEDURE LLC_ONE_INFO(
    KONT_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    DELIQ_DAY_KOL INTEGER,
    BALANCE DECIMAL(18,4),
    LOAN_TYPE SMALLINT,
    PERCENT_KOEF DECIMAL(18,7),
    STATUS_ID INTEGER,
    REG_ID INTEGER)
AS
declare variable pros_date date;
declare variable perc_ind decimal(18,7);
declare variable perc_gr decimal(18,7);
begin
  /* Procedure Text */
  select min(sh_date)
  from deliq_temp_one(:kont_id, :rep_date)
  into :pros_date;

  if (pros_date is null) then
    deliq_day_kol = 0;
  else
    deliq_day_kol = cast((rep_date - pros_date) as integer);

  select min(pr_loan_type), sum(ok_os), min(kal_new_reg_id)
  from operkred, kontract, product, knaktlist
  where k_from_pr_id = pr_id and
        kal_from_kf_id = k_from_kf_id and
        kal_date <= :rep_date and
        kal_date_end >= :rep_date and
        knaktlist.is_delete = 0 and
        ok_ol_is_active = 1 and
        ok_from_k_id = k_id and
        ok_from_k_id = :kont_id and
        ok_ol_date <= :rep_date and
        ok_type > 3 and
        ok_type <= 8 and
        ok_from_c_id = 0
  into :loan_type, :balance, :reg_id;

  if (balance is null) then
    balance = 0;
  if (loan_type is null) then
    loan_type = 0;
  if (reg_id is null) then
    reg_id = 1000000;

  select cs_is_for_group, cs_is_for_indiv, cs_id
  from classific
  where cs_day_min <= :deliq_day_kol and
        cs_day_max >= :deliq_day_kol
  order by cs_id desc
  rows 1
  into :perc_gr, :perc_ind, :status_id;

  if (loan_type = 0) then
    percent_koef = :perc_gr;
  else
    percent_koef = :perc_ind;
  if (percent_koef is null) then
    percent_koef = 0;
  if (status_id is null) then
    status_id = 1;
  suspend;
end
