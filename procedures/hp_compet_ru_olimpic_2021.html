begin
-- Расчет итогов олимпиады РФ 2017
-- Stage:
-- 0 - период (неделя)
-- 1 - Промежуточные итоги
-- 2 - Итоги
-- 3 - Финальные результаты

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end, cpp_orderno
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end, cpp_orderno;

  kol_day_decade = res_date_end - res_date_begin + 1;

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end

  if (extract(day from res_date_end) not in (10, 20)) then
    is_month_end = 1;
  else
    is_month_end = 0;
  date_begin = res_date_end - extract(day from res_date_end) + 1;

  if (cpp_orderno = 0 and
      res_date_begin = cast('31.01.2021' as date)) then
  begin
    res_date_begin = '01.02.2021';
    res_date_end = '01.02.2021';

    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

-- Формируем в памяти данные по РФ
--    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0, 'standart')
--    returning_values :res_period_id;

-- Формируем в памяти данные по РФ
-- Тип отчета передаем "olimpic", чтобы система учитывала особенности расчета для олимпиады
-- Sanjari, 04.06.2017
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0, 'olimpic')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

-- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    for select res_field_1, sum(ad_active_balance)
        from hp_analitica_report('kont_lo_teamname_id', 'tree_group', 8, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
        where exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        group by 1
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :res_team_id, :ad_active_balance
    do
    begin
      if (exists(select *
                 from refdata
                 where rd_id = :res_team_id and
                       rd_is_active = 1)) then
      begin
        insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                  cpr_from_un_id, cpr_comment)
        values (:cpp_id, '', -1, :res_team_id, 0, :un_id, '')
        returning cpr_id
        into :cpr_id;
      end
    end
    suspend;
    exit;
  end

-- Ищем итоги тура
  stage_cpp_id = null;
  select cpp_id
  from competperiod
  where cpp_from_rd_id = :rd_id and
        cpp_stage = 2 and
        cpp_status = 1 and
        cpp_date_end <= :res_date_end
  order by cpp_date_begin desc, cpp_id desc
  rows 1
  into :stage_cpp_id;

  if (stage = 0) then -- Если уровень Декады
  begin
-- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

-- Формируем в памяти данные по РФ
--    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0, 'standart')
--    returning_values :res_period_id;

-- Формируем в памяти данные по РФ
-- Тип отчета передаем "olimpic", чтобы система учитывала особенности расчета для олимпиады
-- Sanjari, 04.06.2017
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0, 'olimpic')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

-- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    for select res_field_1, res_field_2,
               sum(ad_disb_new_amount),
               sum(ad_active_balance),
               sum(ad_delinq_balance_end2)
        from hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group', 8, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
        where exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        and res_field_2 <> 1010100656389 -- как временное решение, сотрудница ушла в декрет
--        and res_field_2 <> 1010100012700 -- Жолдошова Аида Дооронбековна с июля нужно включить обратно т.е. убрать данну строку
        group by 1, 2
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :res_team_id, :res_lo_id,
             :ad_disb_new_amount, -- Сумма выдачи новым клиентам
             :ad_active_balance,
             :ad_delinq_balance_7
    do
    begin
      cpd_comment = '';
      if (ad_disb_new_amount is null) then ad_disb_new_amount = 0;

-- Исключаем если нет действующего
      if (not(exists(select *
                     from hr_pstatus
                     where ps_from_c_id = :res_lo_id and
                           ps_date_end >= :res_date_end and
                           ps_date_begin <= :res_date_end and
                           ps_status = 1 and
                           ps_is_working = 1))) then
        res_team_id = 0;

      if (res_team_id <> 0) then
      begin
-- Исключаем если не СВЗ/КС
        if (exists(select *
                   from hr_pstatus, joblist, jobkind
                   where ps_from_c_id = :res_lo_id and
                         ps_date_end >= :res_date_end and
                         ps_date_begin <= :res_date_end and
                         ps_status = 1 and
                         ps_from_jl_id = jl_id and
                         jl_kind = jk_id and
                         not(jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner')) and
                         ps_is_working = 1)) then
          res_team_id = 0;
      end

      is_otpusk = '';
      is_otpusk_month = '';
      is_otpusk_1_15 = '';
      if (res_team_id <> 0) then
      begin
-- Исключаем если в отпуске.
-- Проверяем статус отпуска на каждый день декады:
-- если из декады проработал менее 5 дней то будем считать что в отпуске
        otpusk_days = 0;
        n = 0;
        while (n <= 9 and dateadd(day, n, res_date_begin) < res_date_end) do
        begin
          dd_tmp = dateadd(day, n, res_date_begin);
          if (exists(select *
                     from hr_voctravel, hr_orders, sa_worktype
                     where hrv_from_ho_id = ho_id and
                           ho_order_type = sawt_id and
                           hrv_from_c_id = :res_lo_id and
                           sawt_base_type in (4, 5) -- отпуск, декретный отпуск
                           and
                           sawt_id <> 20 and
                           sawt_hours_real >= 0 and
                           hrv_date_begin <= :dd_tmp and
                           hrv_date_end_fact >= :dd_tmp and
                           hrv_status = 1)) then
            otpusk_days = otpusk_days + 1;
          n = n + 1;
        end
        kol_day_worked = kol_day_decade - otpusk_days;

        if (kol_day_worked < 5) then
        begin
          if (not(exists(select *
                         from hr_voctravel, hr_orders
                         where hrv_from_ho_id = ho_id and
                               ho_order_type = 20 and
                               hrv_from_c_id = :res_lo_id and
                               hrv_date_begin <= :res_date_end and
                               hrv_date_end_fact >= :res_date_begin and
                               hrv_status = 1))) then
          begin
--            res_team_id=0;
            is_otpusk = 'Отпуск декады.';
          end
        end
      end

      points_1 = 0.0;
      points_2 = 0.0;
      points_3 = 0.0;
      points_4 = 0.0;

--    проверяем, если конец месяца т.е. не 10е и не 20е то включаем дополнительные расчеты
      ad_diff_balance = null;
      disb_sum = 0;
      disb_sum_15 = 0;
      disb_kol = 0;
      super_pokazatel = 0;
      if (is_month_end = 1) then
      begin
        svz_portfel_begin = null;
        bal_no_del_begin = null;
        select sum(oks_balance),
               sum(iif(:date_begin - 1 - oks_delinq_date < 0, oks_balance,0))
        from oksummary
        where :date_begin - 1 between oks_date_begin and oks_date_end and
              :date_begin - 1 - oks_delinq_date <= 30 and   -- было  :date_begin - 1 - oks_delinq_date <= 360
              oks_kc_id = :res_lo_id and
              oks_close_state = 0
        into svz_portfel_begin, bal_no_del_begin;
        svz_portfel_begin = coalesce(svz_portfel_begin, 0);
        bal_no_del_begin = coalesce(bal_no_del_begin, 0);

        svz_portfel_end = null;
        bal_no_del_end = null;
        select sum(oks_balance),
               sum(iif(:res_date_end - oks_delinq_date < 0, oks_balance,0))
        from oksummary
        where :res_date_end between oks_date_begin and oks_date_end and
              :res_date_end - oks_delinq_date <= 30 and -- было  :date_begin - 1 - oks_delinq_date <= 360
              oks_kc_id = :res_lo_id and
              oks_close_state = 0
        into svz_portfel_end, bal_no_del_end;
        svz_portfel_end = coalesce(svz_portfel_end, 0);
        bal_no_del_end = coalesce(bal_no_del_end, 0);

        prinyat_portfel = null;
        prinyat_bal0 = null;
        select sum(oks_balance),
               sum(iif(:res_date_end - oks_delinq_date < 0, oks_balance,0))
        from knaktlist, kontract, oksummary
        where knaktlist.is_delete = 0 and
              kal_date between :date_begin and :res_date_end and
              kal_from_ka_id > 0 and
              kal_new_kf_officer <> kal_prior_lo_id and
              kal_new_kf_officer = :res_lo_id and
              kal_from_kf_id = k_from_kf_id and
              k_id = oks_from_k_id and
              :res_date_end between oks_date_begin and oks_date_end and
              :res_date_end - oks_delinq_date <= 30 and -- мунусуем принятые без просрочки и с просрочкой 0-30 дней
              oks_balance > 0 and
              oks_kc_id = :res_lo_id and
              oks_close_state = 0 and
              k_fact_disb_lo_id <> oks_kc_id -- если принемает то что сам выдавал то их из портфеля не минусуют
-- если передача была внутри одной команды то не минусуют, Приказ № 235.2-09-18
              and
              (coalesce((select res_rd_id
                         from hp_objectprops_get(0, kal_new_kf_officer, -1, 'objectprops.staffteam', :res_date_end)), -999) -- вместо :res_date_end было kal_date т.е. нужно проверять на дату конец
              <> coalesce((select res_rd_id
                           from hp_objectprops_get(0, kal_prior_lo_id, -1, 'objectprops.staffteam', :res_date_end)), -777) or
--   если сотрудник был уволен и была передача то такой портфел будет минусоваться
              not exists(select *
                         from hr_pstatus
                         where ps_from_c_id = kal_prior_lo_id and
                               :res_date_end between ps_date_begin and ps_date_end and
                               ps_status = 1 and
                               ps_is_working = 1))
        into prinyat_portfel, prinyat_bal0;
        prinyat_portfel = coalesce(prinyat_portfel, 0);
        prinyat_bal0 = coalesce(prinyat_bal0, 0);

--      Рост портфеля без просрочек и с просрочкой 0-30 дней за отчетный месяц ----------------
        ad_diff_balance = svz_portfel_end - svz_portfel_begin - prinyat_portfel;
--      Рост портфеля без просрчки для супер показателя
        bal_no_del_diff = bal_no_del_end - bal_no_del_begin - prinyat_bal0;

--      обьем выдач
--      кол-во выдач для супер показателя
        disb_sum = null;
        disb_kol = null;
        disb_new_client = null;
        disb_sum_new_cl = null;
        select sum(k_amount_approved),
               count(*),
               sum((select 1 from knclient
               where kc_from_k_id = k_id
               and kc_amount_approved > 0
               and kc_circle <=1)),
               sum((select kc_amount_approved from knclient
               where kc_from_k_id = k_id
               and kc_amount_approved > 0
               and kc_circle <=1))
        from kontstate, kontract
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_date_begin between :date_begin and :res_date_end and
              ks_kc_id = :res_lo_id and
              ks_from_k_id = k_id
        into disb_sum, disb_kol, disb_new_client, disb_sum_new_cl;
        disb_sum = coalesce(disb_sum,0);
        disb_kol = coalesce(disb_kol,0);
        disb_new_client = coalesce(disb_new_client,0);
        disb_sum_new_cl = coalesce(disb_sum_new_cl,0);

--      Суммы выдачи с 1 по 15 числа
        disb_sum_15 = 0;
        select sum(k_amount_approved)
        from kontstate, kontract
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_date_begin between :date_begin and :date_begin + 14 and
              ks_kc_id = :res_lo_id and
              ks_from_k_id = k_id
        into disb_sum_15;
        if (disb_sum_15 is null) then disb_sum_15 = 0;

--        if (plan_sum > 0) then plan_perc = (disb_sum - dosroch_sum) / plan_sum * 100;
--        else plan_perc = 100;

        is_otpusk_month = '';
        is_otpusk_1_15 = '';
        if (res_team_id <> 0) then
        begin
-- Исключаем если в отпуске.
-- Проверяем статус отпуска на каждый день декады:
-- если из декады проработал менее 5 дней то будем считать что в отпуске
          otpusk_days = 0;
          otpusk_days_1_15 = 0;
          n = 0;
          while (dateadd(day, n, date_begin) <= res_date_end) do
          begin
            dd_tmp = dateadd(day, n, date_begin);
            if (exists(select *
                       from hr_voctravel, hr_orders, sa_worktype
                       where hrv_from_ho_id = ho_id and
                             ho_order_type = sawt_id and
                             hrv_from_c_id = :res_lo_id and
                             sawt_base_type in (4, 5) -- отпуск, декретный отпуск
                             and
                             sawt_id <> 20 and
                             sawt_hours_real >= 0 and
                             hrv_date_begin <= :dd_tmp and
                             hrv_date_end_fact >= :dd_tmp and
                             hrv_status = 1)) then
              begin
                otpusk_days = otpusk_days + 1;
                if (n < 15) then otpusk_days_1_15 = otpusk_days_1_15 + 1;
              end
              n = n + 1;
          end
          kol_day_month = extract(day from res_date_end);
          kol_day_worked = kol_day_month - otpusk_days;

--          if (kol_day_worked < 15) then
          if (otpusk_days > 15) then -- изменен по письму от Альбины 11.03.2021г. ВИПС который находился в отпуске более 15 календарных дней
          begin
            if (not(exists(select *
                           from hr_voctravel, hr_orders
                           where hrv_from_ho_id = ho_id and
                                 ho_order_type = 20 and
                                 hrv_from_c_id = :res_lo_id and
                                 hrv_date_begin <= :res_date_end and
                                 hrv_date_end_fact >= :res_date_begin and
                                 hrv_status = 1))) then
            begin
--            res_team_id=0;
              is_otpusk_month = 'Отпуск месяца.';
            end
          end
          if (otpusk_days_1_15 > 7) then
          begin
            if (not(exists(select *
                           from hr_voctravel, hr_orders
                           where hrv_from_ho_id = ho_id and
                                 ho_order_type = 20 and
                                 hrv_from_c_id = :res_lo_id and
                                 hrv_date_begin <= :res_date_end and
                                 hrv_date_end_fact >= :res_date_begin and
                                 hrv_status = 1))) then
            begin
--            res_team_id=0;
              is_otpusk_1_15 = 'Отпуск в первой половине месяца';
            end
          end 


        end
      end

      cpd_comment = is_otpusk || '<br>';
