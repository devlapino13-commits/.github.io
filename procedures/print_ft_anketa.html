CREATE PROCEDURE PRINT_FT_ANKETA(
    OBJ_ID D_ID,
    ADD_PARAMS D_STR_LONG)
RETURNS (
    FAL_DATE D_DATE,
    FAK_NAME D_STR_100,
    C_NAME_FIO D_STR_100,
    C_NAME_ENG D_STR_100,
    C_BIRTH_DATE D_DATE,
    COMPANY_NAME D_STR_255,
    QR_CODE D_STR_LONG,
    C_INN D_STR_20)
AS
declare variable EDL_ID D_ID;
declare variable C_ID D_ID;
begin
  -- Печать заголовка/примечания анкеты ПОД/ФТ

  company_name = f_company_name('short', 'local');

  select fal_date, c_id, c_name_fio, c_name_eng, c_birth_date, fak.fak_name, cl.c_inn
  from ft_anketa_list fal
  inner join clients cl on cl.c_id = fal.fal_from_c_id
  left join ft_anketa_kind fak on fak.fak_id = fal.fal_from_fak_id
  where fal_id = :obj_id
  into fal_date, c_id, c_name_fio, c_name_eng, c_birth_date, fak_name,  c_inn;

  if (f_get_const('selfemployed_active') <> 1) then
    c_inn = '';

  -- QR-код электронной подписи клиента. Текст пустой если не подписан
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);
    qr_code = f_edoc_qr_by_edl(edl_id, c_id);
  end

  suspend;
end
