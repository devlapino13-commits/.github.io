CREATE PROCEDURE HP_BKI_GET_INFO4REQ(
    C_ID ID_TYPE,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID ID_TYPE)
RETURNS (
    BIRTHDAY D_STR_20,
    BIRTHPLACE D_STR_100,
    CONSENTDATE D_STR_20,
    ADMCODEINFORM D_STR_10,
    REQUEST_REASON_CODE D_STR_10,
    CONSENT_REASON_CODE D_STR_10,
    APP_SUM DECIMAL(15,2),
    CONSENTHASH D_STR_64,
    CONSENTPERIOD D_STR_10,
    APP_CURRENCY D_STR_10,
    CONTRACTDATE D_STR_20,
    C_INN D_STR_20,
    INN_CHECKED D_BOOL,
    REQ_TYPE D_STR_CODE,
    OGRN D_STR_CODE)
AS
declare variable c_citizen d_integer;
declare variable c_birth_city str_100;
declare variable c_birth_country str_100;
declare variable ma_id d_id;
declare variable k_id d_id;
declare variable edb_id d_id;
declare variable edc_id d_integer;
declare variable ma_kind d_integer;
declare variable inn_received d_bool;
declare variable inn d_str_20;
declare variable pr_code d_str_100;
declare variable mac_from_c_id d_id;
begin
  if (coalesce(object_kind, 0) not in (9,11)) then
    exception e_default 'object_kind must be 11 or 9';

  if (object_kind = 11) then
  begin
    ma_id = object_id;
  
    select k_id from kontract
    where k_from_ma_app_id = :ma_id
      and k_status >= 0
    order by k_status desc
    rows 1
    into k_id;
  end
  else
  begin
    k_id = object_id;

    select k_from_ma_app_id
    from kontract
    where k_id = :k_id
    into ma_id;
  end

  select ma_summa_minus, pr_code
  from miniapp, product
  where ma_id = :ma_id
    and pr_id = ma_from_pr_id
  order by ma_number desc
  rows 1
  into app_sum, pr_code;

  select f_datetostr(k_disb_date)
  from kontract, kontstate
  where k_id = :k_id
    and ks_from_k_id = k_id
    and current_date between ks_date_begin and ks_date_end
    and ks_ol_is_delete = 0
    and ks_state = 1
  into contractdate;

  edc_id = iif(f_get_const('doc_optimize_pack2') = 1, f_edoc_kind('loan.application','id'), f_edoc_kind('bki.agreement','id'));  -- RFCRU-2603
  ma_kind = f_objkind('miniapp', 'id');

  select mac_from_c_id
  from miniappclient
  where mac_from_ma_id = :ma_id
  into mac_from_c_id;

  if (c_id = mac_from_c_id) then
  begin
    select edb_id, f_datetostr(eds_status_datetime)
    from miniappclient, edoc_sign, edoc_list, edoc_blank, edoc_frx
    where eds_from_c_id = mac_from_c_id and
          edl_object_id = mac_from_ma_id and
          eds_from_edl_id = edl_id and
          edb_from_edl_id = edl_id and
          edb_from_edf_id = edf_id and
          edf_code = 'bki.agreement' and
          edl_object_kind = :ma_kind and
          edl_object_id = :ma_id and
          edl_from_edc_id = :edc_id and
          eds_status = 1 and
          edl_status = 1
    order by eds_status_datetime desc rows 1

    into edb_id, consentdate;

    -- если не найдено разрешение в объединенном документе, то поищем в отдельном
    if (edb_id is null) then
      select edb_id, f_datetostr(eds_status_datetime)
      from miniappclient, edoc_sign, edoc_list, edoc_blank
      where eds_from_c_id = mac_from_c_id and
            edl_object_id = mac_from_ma_id and
            eds_from_edl_id = edl_id and
            edb_from_edl_id = edl_id and
            edl_object_kind = f_objkind('miniapp', 'id') and
            edl_object_id = :ma_id and
            edl_from_edc_id = f_edoc_kind('bki.agreement', 'id') and
            eds_status = 1 and
            edl_status = 1
      order by eds_status_datetime desc rows 1

      into edb_id, consentdate;
  end
  else if (k_id is not null) then
  begin
    select edb_id, f_datetostr(eds_status_datetime)
    from edoc_sign, edoc_list, edoc_blank
    where eds_from_c_id = :c_id and
          eds_from_edl_id = edl_id and
          eds_from_edl_id = edl_id and
          edb_from_edl_id = edl_id and
          edl_object_kind = f_objkind('kontract', 'id') and
          edl_object_id = :k_id and
          edl_from_edc_id = f_edoc_kind('bki.agreement.zalog', 'id') and
          eds_status = 1 and
          edl_status = 1
    order by eds_status_datetime desc rows 1
    into edb_id, consentdate;
  end

  consenthash = (select lower(ebh_hash) from edoc_blank_hash where ebh_from_edb_id = :edb_id);

  admcodeinform = 1;
  consentperiod = iif(contractdate is null, 1, 3);           -- Срок действия согласия
  request_reason_code = 2;     -- Код цели запроса   (2 - Потребительский микрозаем)
  consent_reason_code = 2;     -- Код цели согласия  (2 - Потребительский микрозаем)
  app_currency = 'RUB';

  for select f_datetostr(c_birth_date), trim(coalesce(c_birth_city, '')), trim(coalesce(c_birth_country, '')), c_citizen, c_inn
  from clients
  where c_id = :c_id
  into birthday, c_birth_city, c_birth_country, c_citizen, c_inn
  do
  begin
    birthplace = c_birth_city;
    if (birthplace = '') then
      birthplace = c_birth_country;

    if (birthplace = '') then
    begin
      select ddr.rd_name
      from refdata ddr
      where ddr.rd_id = :c_citizen
      into :birthplace; --  место роджения беру исходя из гражданства
    end

    -- RFCRU-3296
    inn_checked = 0;

    select inn_received, inn
    from inn_query_status(:c_id)
    into inn_received, inn;

    if (inn_received = 1) then
    begin
      inn_checked = 1;
      c_inn = inn;
    end

    req_type = '1234577353'; -- стандартный
    if (current_date < '01.08.2025') then -- до этой даты отправляем на все запрос со скорингом/ После надо удалить
      req_type = '1234579669'; -- 1234579669       Кредитный отчёт со скорингом. Версия 4.0. Installment + PDL
    else
    begin
      if (pr_code containing ('.universal.')
        or pr_code containing ('.oivo_pack.')
        or c_citizen = f_get_rd_id('Citizen.RU')    -- RFCRU-3998
        or f_point(c_id, current_date) = 0)         -- RFCRU-3998
      then
        req_type = '1234579668';  -- 1234579668   Кредитный отчёт со скорингом. Версия 4.0. Installment
    end

    select ogrn
    from hp_kontur_ip_get_base_info('clients', :c_id)
    into ogrn;

    suspend;
  end
end
