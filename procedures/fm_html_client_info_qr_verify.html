CREATE PROCEDURE FM_HTML_CLIENT_INFO_QR_VERIFY(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_8100)
AS
declare variable lang d_str_lang;
declare variable tmp_datetime d_datetime;
declare variable qrl_id_last d_id;
declare variable qrl_state d_smallint;
declare variable qrl_state_str d_str_code;
declare variable qrl_state_color d_str_code;
declare variable color_yellow d_str_code;
declare variable color_red d_str_code;
declare variable acl_memo d_str_8100;
declare variable acl_id d_id;
declare variable verify_datetime d_datetime;
begin
  -- Формирует блок информации о QR-верификации

  res_text = '';
  color_yellow = '#F08205';
  color_red = 'red';

  if (f_get_const('qr_verify_enabled') = 0) then
  begin
    suspend;
    exit;
  end

  lang = f_get_lang_code(un_id, null);

  res_text = f_lang(lang,  'Последняя верификация встречи') || ': ';

  tmp_datetime = null;
  verify_datetime = null;

  -- Выясняем дату/время последней успешной верификации
  select qrl_datetime
  from qr_list
  where qrl_from_c_id = :c_id and
        qrl_is_delete = 0 and
        qrl_state = 2
  order by qrl_datetime desc
  rows 1
  into verify_datetime;

  if (row_count = 0) then
    res_text = res_text || f_lang(lang, 'Отсутствует');
  else
    res_text = res_text || f_datetimetostr(verify_datetime);

  res_text = res_text || '<br>';

  -- Проверяем статус верификации на стороне Андеррайтеров
  qrl_id_last = null;
  select qrl_id, qrl_datetime, qrl_state
  from qr_list
  where qrl_from_c_id = :c_id and
        qrl_is_delete = 0 and
        qrl_state in (0, 1, -2)
  order by qrl_datetime desc
  rows 1
  into qrl_id_last, tmp_datetime, qrl_state;

  if (tmp_datetime is not null) then
  begin
    if (verify_datetime is null or tmp_datetime > verify_datetime) then
    begin
      qrl_state_str = '';

      if (qrl_state = 0) then
      begin
        qrl_state_str = f_lang(lang, 'Сгенерирован QR');
        qrl_state_color = color_yellow;
      end
      else if (qrl_state = 0) then
      begin
        qrl_state_str = f_lang(lang, 'На проверке');
        qrl_state_color = color_yellow;
      end
      else if (qrl_state = -2) then
      begin
        qrl_state_str = f_lang(lang, 'Отклонена');
        qrl_state_color = color_red;
      end

      res_text = res_text || f_format('%2: <font color="%1">%3</font><br>',
        qrl_state_color,
        f_datetimetostr(tmp_datetime),
        qrl_state_str);

      -- Для отказов и в обработке возвращаем комментарий Андеррайтеров
      if (qrl_state in (1, -2)) then
      begin
        acl_id = null;
        select acl_id from actionlist
        where acl_object_kind = f_objkind('qr_photo', 'id')
        and acl_object_id = :qrl_id_last
        into acl_id;
        if (acl_id is not null) then
        begin
          select left(res_memo, 255)
          from hp_actionlist_get_memo(:acl_id)
          into acl_memo;
  
          if (coalesce(acl_memo, '') <> '') then
            res_text = res_text || acl_memo;
        end
      end
    end
  end

  res_text = res_text || '<br>';
  suspend;
end
