CREATE PROCEDURE HP_REPAYMENT_DETAIL_FACT_KG(
    K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    FACT_DATE D_DATE,
    PAID_OS D_CURRENCY,
    PAID_PERC D_CURRENCY,
    PAID_ALL D_CURRENCY,
    DISB_AMOUNT D_INTEGER,
    OKS_BAL_TO_PERC D_INTEGER,
    ACCR_PERC D_CURRENCY,
    OK_DAYS D_INTEGER,
    OKS_BALANCE D_INTEGER,
    CALC_PERC D_CURRENCY,
    NSP D_CURRENCY,
    PERCENT_NORM D_CURRENCY)
AS
declare variable k_stop_perc d_date;
declare variable pr_code d_str_100;
declare variable disb_date d_date;
declare variable ok_month d_integer;
declare variable ok_year d_integer;
declare variable ok_month_prv d_integer;
declare variable ok_year_prv d_integer;
declare variable prv_date d_date;
declare variable counter d_integer;
declare variable sh_date d_date;
declare variable sh_last_date d_date;
declare variable ol_id d_id;
begin
  select k.k_stop_perc, k.k_amount_approved, k.k_disb_date, pr.pr_code
  from kontract k
  inner join product pr on k.k_from_pr_id = pr.pr_id
  where k.k_id = :k_id
  into k_stop_perc, disb_amount, disb_date, pr_code;

  if (counter is null) then
    begin
      fact_date = disb_date;
      paid_os = 0;
      paid_perc = 0;
      paid_all = 0;
      oks_balance = disb_amount;
      oks_bal_to_perc = disb_amount;
    suspend;
    end

  counter = 0;

  for select ok.ok_from_ol_id, -sum(ok.ok_os + ok.ok_sh_os), -sum(ok.ok_perc + ok.ok_sh_perc),
                      -sum(ok.ok_os + ok.ok_perc + ok.ok_tax)
      from operkred ok
      where ok.ok_ol_date <= :rep_date and
            ok.ok_from_k_id = :k_id and
            ok.ok_type = 4 and
            ok.ok_ol_is_active = 1
      group by ok.ok_from_ol_id
      into :ol_id, :paid_os, :paid_perc, :paid_all
  do
  begin
    counter = counter + 1;

    fact_date = null;
    select ok.ok_ol_date
    from operkred ok
    where ok.ok_ol_date <= :rep_date and
          ok.ok_from_ol_id = :ol_id and
          ok.ok_from_k_id = :k_id and
          ok.ok_type = 4 and
          ok.ok_ol_is_active = 1
    rows 1
    into :fact_date;

    sh_last_date = null;
    select first 1 sh_date
    from schedule sh
    where sh.sh_from_k_id = :k_id
    order by sh.sh_date desc
    into :sh_last_date;

    ok_month = extract(month from :fact_date);
    ok_year = extract(year from :fact_date);
    ok_month_prv = extract(month from :fact_date) - 1;

    sh_date = null;
    select sh.sh_date
    from schedule sh
    where sh.sh_date <= :fact_date
          and sh.sh_from_k_id = :k_id
    order by sh.sh_date desc
    rows 1
    into :sh_date;

    if (ok_month_prv = 0) then
    begin
      ok_month_prv = 12;
      ok_year_prv = ok_year - 1;
    end

    if (counter > 1 and
        paid_perc <= 0) then
      continue;

    if (counter = 2) then
      oks_balance = disb_amount;

    if (counter >= 1) then
      disb_amount = 0;

    oks_balance = null;
    select oks.oks_balance
    from oksummary oks
    where oks.oks_from_k_id = :k_id and
          extract(month from oks_date_begin) = :ok_month and
          extract(year from oks_date_begin) = :ok_year
    order by oks.oks_date_end desc rows 1
    into :oks_balance;

    oks_balance = coalesce(oks_balance, 0);

    oks_bal_to_perc = null;
    select oks.oks_balance
    from oksummary oks
    where oks.oks_from_k_id = :k_id and
          extract(month from oks_date_begin) = :ok_month_prv and
          extract(year from oks_date_begin) = iif(:ok_month_prv = 12, :ok_year_prv, :ok_year)
    order by oks.oks_date_end desc rows 1
    into :oks_bal_to_perc;

    if (oks_bal_to_perc is null) then
    begin
      oks_bal_to_perc = null;
      select oks.oks_balance
      from oksummary oks
      where oks.oks_from_k_id = :k_id and
            extract(month from oks_date_begin) = :ok_month and
            extract(year from oks_date_begin) = :ok_year rows 1
      into :oks_bal_to_perc;
    end

    prv_date = null;
    select ok.ok_ol_date
    from operkred ok
    where ok.ok_from_k_id = :k_id and
          extract(month from ok.ok_ol_date) = :ok_month_prv and
          extract(year from ok.ok_ol_date) = iif(:ok_month_prv = 12, :ok_year_prv, :ok_year) and
          ok.ok_ol_is_active = 1 and
          ok.ok_type = 4
    order by ok.ok_ol_date desc rows 1
    into :prv_date;

    if (prv_date is null) then
      prv_date = fact_date;

    for select sum(zad.sh_calc_perc)
        from aarep_make_zad_list(:k_id, :sh_date, 1, 0) zad
        where zad.sh_calc_perc <> 0
        into :calc_perc
    do
    begin
        calc_perc = coalesce(calc_perc, 0);
    end

    if (k_stop_perc is null or sh_date <= k_stop_perc) then
    begin
      select kd.kd_percent
      from kontdata kd
      where kd.kd_from_k_id = :k_id and
            kd.kd_ol_is_delete = 0 and
            kd.kd_ol_date <= :prv_date rows 1
      into :percent_norm;
    end

    ok_days = fact_date - prv_date;
    ok_days = coalesce(ok_days, 0);

    for select kfop.calced_perc
        from kontract_full_oper_perc(:k_id, :fact_date) kfop
        into :accr_perc
    do
    begin
      accr_perc = coalesce(accr_perc, 0);
    end
    suspend;
  end
end
