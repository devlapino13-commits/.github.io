CREATE PROCEDURE FM_SET_BLOB_CLIENT_PHOTO(
    BLS_BLOB TYPE OF D_BLOB_BINARY,
    BLS_FILENAME TYPE OF D_STR_100,
    BLS_CAPTION TYPE OF D_STR_100,
    BLS_KIND D_INTEGER,
    UN_ID TYPE OF D_ID,
    C_ID TYPE OF D_ID,
    NEED_CHECK D_BOOL = 0)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable bls_id type of d_id;
declare variable blf_id type of d_integer;
declare variable db_id d_integer;
declare variable c_citizen d_integer;
declare variable lang d_str_lang;
declare variable pk_id d_integer;
declare variable need_photo2 d_bool;
declare variable grs_active d_smallint;
begin
  uxf$set_user(un_id);

  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);
  grs_active = f_grs_active(un_id);
  err_text = '';
  bls_id = null;
  err_text = '';

  select c_blob_photo_id, c_citizen, c_doc_type
  from clients
  where c_id = :c_id
  into bls_id, c_citizen, pk_id;

  if (:grs_active = 1) then
  begin
     if (((db_id in (39801, 41701) and c_citizen = f_get_rd_id('Citizen.KG')) or
        (db_id = 94401 and c_citizen = f_get_rd_id('Citizen.AZ'))) and
        not exists(select *
                   from grs_clients gc
                   inner join enum_list el on el.el_id = gc.gc_status and el.el_kind = 'grs_status'
                   where gc.gc_from_c_id = :c_id and
                         gc.gc_is_last = 1 and
                         el.el_code in ('successful', 'successful_adr_empty', 'manual_check'))) then
        begin
            err_text = f_lang(lang, 'Сначала надо отправить запрос в ГРС, а после сфотографировать клиента');
            suspend;
            exit;
        end
  end

  blf_id = -1;
  select blf_id
  from blobfiles
  where :bls_id between blf_first_id and blf_last_id
  into blf_id;

  if (bls_id is null or (blf_id <> 0)) then
  begin
    -- Вставка если фото еще не было закреплено или если блоб сохранен в архивном blx
    execute procedure hp_insert_into_blobstore (:bls_filename, :bls_caption, :bls_blob, :un_id, :bls_kind, 1)
    returning_values :bls_id;
    update clients
    set c_blob_photo_id = :bls_id
    where c_id = :c_id;
  end
  else
  begin
    select drm_has_page2
    from hp_get_drm_has_pages(:pk_id)
    into :need_photo2;

    -- Обновление существующего фото (для экономии места в blx)
    execute procedure hp_update_blobstore (:bls_id, :bls_filename, :bls_caption,
                                           :bls_blob, :un_id, bls_kind, 1);
    if (exists(select *
               from clients
               where c_id = :c_id and
                     coalesce(c_name_fio, '') <> '' and
                     c_birth_date is not null and
                     coalesce(c_pass_no, '') <> '' and
                     coalesce(c_pass_date, '') <> '' and
                     coalesce(c_blob_photo_id, 0) > 0 and
                     coalesce(c_blob_pass1_id, 0) > 0 and
                     (:need_photo2 = 0 or coalesce(c_blob_pass2_id, 0) > 0) and
                     c_verify_status <> 1)) then
    begin
      if (db_id in (81001, 94401) or
        --Доп прооверки для КГ
        exists(select *
               from hp_status_param_verif(:c_id)
               where result = 0))
      then
        update clients
        set c_verify_status = 1
        where c_id = :c_id;
    end
  end

  if (:grs_active = 1) then
  begin
    if (exists(
          select 1
          from grs_clients gc
          where gc.gc_from_c_id = :c_id and
                gc.gc_is_last = 1 and
                f_enum_code('grs_status', gc.gc_status) <> 'manual_check'
        )) then
      execute procedure hp_ins_facerecoglist(c_id);
  end


  if (db_id in (39801, 41701)) then
    execute procedure hp_photo_on_date_disb_auto_add(:c_id, :bls_id, :un_id);

  if (need_check = 1) then
    update clientstate
    set cs_photo_manual_check = 1
    where cs_from_c_id = :c_id;

  suspend;
end
