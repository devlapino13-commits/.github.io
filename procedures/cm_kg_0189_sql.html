CREATE PROCEDURE CM_KG_0189_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable k_disb_date d_date;
declare variable pr_code d_str_100;
declare variable compare_pr d_integer;
declare variable compare_pr_card d_integer;
begin
  for select kc.kc_from_c_id, k.k_disb_date
      from kontract k
      inner join knclient kc on kc.kc_from_k_id = k.k_id and
            kc.kc_from_k_id = :k_id and
            kc.kc_amount_approved > 0
      inner join product pr on k.k_from_pr_id = pr.pr_id
            and pr.pr_code containing '.IND.' and
            pr.pr_code containing '.PAR.' and
            pr.pr_code not containing '.IPF.' and
            pr.pr_code not containing '.BIRGE.'
      into :c_id, :k_disb_date
  do
  begin
    compare_pr_card = 0;
    compare_pr = 0;
    for select pr.pr_code
        from kontract k
        inner join knclient kc on k.k_id = kc.kc_from_k_id and
              kc.kc_amount_approved > 0 and
              kc.kc_from_c_id = :c_id
        inner join product pr on k.k_from_pr_id = pr_id
        inner join kontstate ks on k.k_id = ks.ks_from_k_id
              and ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1 and
              :k_disb_date between ks.ks_date_begin and ks.ks_date_end
        into :pr_code
    do
    begin
      if (pr_code containing '.MBULAK_CARDS.') then
      begin
        compare_pr_card = compare_pr_card + 1;
      end

      compare_pr = compare_pr + 1;
    end
    if (compare_pr_card = compare_pr) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Клиент, имеющий только карточные продукты, не может получить Индивидуальный Ишеним Плюс и Индивидуальный Стандарт Плюс';
    suspend;
    end
  end
end
