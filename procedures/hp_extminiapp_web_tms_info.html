CREATE PROCEDURE HP_EXTMINIAPP_WEB_TMS_INFO(
    IN_UUID D_STR_CODE)
RETURNS (
    RES_JSON D_STR_8100)
AS
declare variable ema_uuid d_uuid;
declare variable ema_id d_id;
declare variable res_text d_str_255;
declare variable obj_kind d_integer;
declare variable client_fio d_str_100;
declare variable client_gender d_str_code;
declare variable agent_fio d_str_code;
begin
  res_text = '';
  res_json = '{"result":{"fio":"%1", "gender":"%2", "agent":"%3", "error_text":"%4"}}';

  begin
    ema_uuid = char_to_uuid(in_uuid);
    when any do
      res_text = 'Неверная ссылка';
  end

  client_fio = '';
  client_gender = '';
  agent_fio = '';

  if (res_text = '') then
  begin
    obj_kind = f_objkind('extminiapp', 'id');

    select udl_object_id
    from uuidlist
    where udl_uuid = :ema_uuid and
          udl_object_kind = :obj_kind
    into ema_id;

    if (ema_id is not null) then
    begin
      select c.c_name_fio, trim(iif(c.c_sex = 1, 'male', 'female')), s.c_name_fio
      from extminiapp exa, clients c, clients s
      where exa.ema_id = :ema_id and
            exa.ema_from_c_id = c.c_id and
            exa.ema_lo_id = s.c_id
      into client_fio, client_gender, agent_fio;
    end

    if (client_fio = '') then
      res_text = 'Приглашение на найдено';
  end

  res_json = f_format(res_json, client_fio, client_gender, agent_fio, res_text);

  suspend;
end
