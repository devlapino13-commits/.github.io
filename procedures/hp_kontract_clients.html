CREATE PROCEDURE HP_KONTRACT_CLIENTS(
    C_ID DECIMAL(18,0))
RETURNS (
    REP_C_ID DECIMAL(18,0),
    REP_K_ID DECIMAL(18,0),
    REP_DATE DATE,
    REP_COUNT INTEGER)
AS
declare variable rep_status integer;
begin
  c_id = 1010100000050;
  rep_count = 0;
  rep_c_id = c_id;
  for with tab
      as (select kc_from_k_id, ks_date_begin as rep_date, 1 as rep_status
          from knclient, kontstate
          where kc_from_c_id = :c_id and
                kc_amount_approved > 0 and
                kc_from_k_id = ks_from_k_id and
                ks_ol_is_delete = 0 and
                ks_state = 1
          union
          select kc_from_k_id, ks_date_end as rep_date, -1 as rep_status
          from knclient, kontstate
          where kc_from_c_id = :c_id and
                kc_amount_approved > 0 and
                kc_from_k_id = ks_from_k_id and
                ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_end < '31.12.2999')
      select kc_from_k_id, rep_date, rep_status
      from tab
      order by rep_date, rep_status
      into :rep_k_id, rep_date, rep_status
  do
  begin
    rep_count = rep_count + rep_status;
    suspend;
  end
end
