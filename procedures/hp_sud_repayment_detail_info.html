CREATE PROCEDURE HP_SUD_REPAYMENT_DETAIL_INFO(
    SMR_ID D_ID)
RETURNS (
    RES_HEADER D_STR_255,
    K_KONTRACT_NO D_STR_CODE,
    K_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_KOLICH_REPAY D_INTEGER,
    K_PERIOD D_FLOAT,
    K_STOP_PERC D_DATE,
    K_DELINQ_DATE D_DATE,
    K_PERCENT_YEAR D_PERCENT,
    K_USED_DAYS D_INTEGER,
    K_PERCENT_METHOD D_STR_255,
    K_ANNU_FORMULA D_STR_LONG,
    K_ANNU_CALC_STR D_STR_LONG,
    DATE_STOP_KONTRACT D_STR_20)
AS
declare variable k_id d_id;
begin
  select smr_from_k_id
  from sud_mainreq
  where smr_id = :smr_id
  into k_id;

  for select res_header, k_kontract_no, k_disb_date, k_amount_approved, k_kolich_repay, k_period, k_stop_perc, k_delinq_date,
         k_percent_year, k_used_days, k_percent_method, k_annu_formula, k_annu_calc_str, date_stop_kontract
  from hp_repayment_detail_info(:k_id, current_date)
  into :res_header, :k_kontract_no, :k_disb_date, :k_amount_approved, :k_kolich_repay, :k_period, :k_stop_perc,
       :k_delinq_date, :k_percent_year, :k_used_days, :k_percent_method, :k_annu_formula, :k_annu_calc_str, :date_stop_kontract
  do
    suspend;
end
