CREATE PROCEDURE HP_EXTINFO_SET_BLOB(
    ID D_ID NOT NULL,
    KIND D_STR_CODE NOT NULL,
    BLOB_FILE D_BLOB_BINARY NOT NULL)
AS
declare variable bls_id d_id;
declare variable file_name d_str_255;
declare variable caption d_str_255;
declare variable kind_id d_id;
declare variable db_id d_id;
begin
  kind_id = f_enum_id('extinfo_bls_kind', kind);
  db_id = f_get_db_id();

  -- Имя файла и заголовок
  file_name = '';
  select eic_code, f_objname(eil_object_kind, eil_object_id)
  from extinfocfg, extinfolist
  where eil_id = :id and
        eil_from_eic_id = eic_id
  into file_name, caption;

  if (kind = 'tunduk_client_photo' or kind = 'asan_client_photo' or (kind = 'bioguard_client_photo' and db_id = 39801)) then
  begin
    file_name = file_name || '.jpg';

    execute procedure hp_insert_into_blobstore (:file_name, :caption, :blob_file, 1000000000000, 2, 3)
    returning_values :bls_id;
  end

  if (kind in ('moneta_doc_payment_order', 'moneta_doc_bank_order', 'moneta_doc_receipt')) then
  begin
    file_name = file_name || '.pdf';

    execute procedure hp_insert_into_blobstore (:file_name, :caption, :blob_file, 1000000000000, 2, 3)
    returning_values :bls_id;
  end


  if (kind = 'policy_sertificate') then
  begin
    file_name = file_name || '.pdf';

    execute procedure hp_insert_into_blobstore (:file_name, :caption, :blob_file, 1000000000000, 2, 3)
    returning_values :bls_id;
  end

  if (kind = 'equifax_xml_request') then
  begin
    file_name = file_name || '_req_.xml';

    execute procedure hp_insert_into_blobstore (:file_name, :caption, :blob_file,1000000000000, 2, 3)
    returning_values :bls_id;
  end

  if (kind = 'equifax_xml_response') then
  begin
    file_name = file_name || '_res_.xml';

    execute procedure hp_insert_into_blobstore (:file_name, :caption, :blob_file, 1000000000000, 2, 3)
    returning_values :bls_id;

  end

  if (bls_id is not null) then
  begin
    update or insert into extinfolistbls(eib_from_eil_id, eib_kind, eib_from_bls_id)
    values (:id, :kind_id, :bls_id)
    matching (eib_from_eil_id, eib_kind);
  end
end
