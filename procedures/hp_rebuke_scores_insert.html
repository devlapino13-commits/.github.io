CREATE PROCEDURE HP_REBUKE_SCORES_INSERT(
    UN_ID D_ID,
    C_ID D_ID,
    SCORE D_SMALLINT)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable points d_integer;
declare variable points_new d_integer;
declare variable act d_str_20;
begin
  -- процедура добавляет корректирующий бонусный или штрафной балл сотруднику (RFCKG-2189)

  res_info_text = '';
  res_error_text = '';

  if (not exists(select *
                 from username
                 where un_from_c_id = :c_id)) then
    res_error_text = 'Сотрудник не найден';

  if (score is null or score = 0) then
    res_error_text = 'Корректирующий балл не может равняться нулю';
  else if (score < 0) then
    act = 'Штрафной балл';
  else
    act = 'Бонусный балл';

  select sum(rs_scores)
  from rebuke_scores
  where rs_from_c_id = :c_id and
        rs_is_delete = 0
  into points; -- текущий итоговый балл

  if ((points + score) < 0) then
    res_error_text = 'Штрафный балл не может превышать текущий итоговый балл';

  if (res_error_text = '') then
  begin
    insert into rebuke_scores (rs_from_c_id, rs_scores, rs_date, rs_protocol_no, rs_from_un_id, rs_datetime)
    values (:c_id, :score, current_date, -1, :un_id, cast('now' as timestamp));

    if (row_count = 1) then
    begin
      select sum(rs_scores)
      from rebuke_scores
      where rs_from_c_id = :c_id and
            rs_is_delete = 0
      into points_new; -- итоговый балл после обновления

      res_info_text = act || ' успешно добавлен.' || ascii_char(13) || ascii_char(10);
      res_info_text = res_info_text || 'Изменение итогового балла: ' || points || ' -> ' || points_new;
    end
    else
      res_error_text = act || ' добавить не удалось, возникла ошибка';
  end

  suspend;
end
