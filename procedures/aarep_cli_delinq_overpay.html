CREATE PROCEDURE AAREP_CLI_DELINQ_OVERPAY(
    REP_DATE_END D_DATE)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    C_INN D_STR_20,
    K_KONTRACT_NO D_STR_20,
    K_KONTRACT_DATE D_DATE,
    OVERPAY_SUM DECIMAL(18,2),
    PREPAYMENT_SUM DECIMAL(18,2),
    DELINQ_DAYS D_INTEGER,
    DELINQ_ZAD DECIMAL(18,2),
    ZAD_OS DECIMAL(18,2),
    ZAD_PERC DECIMAL(18,2),
    ZAD_FINE DECIMAL(18,2),
    ZAD_ALL DECIMAL(18,2),
    K_STATE D_STR_255,
    OK_OL_DATE D_DATE)
AS
declare variable k_id d_id;
begin
  for select distinct c.c_id,
                      c.c_name_fio,
                      c.c_inn
  from clients c
  join knclient kc on kc.kc_from_c_id = c.c_id
  join oksummary oks on oks.oks_from_k_id = kc.kc_from_k_id
  where kc.kc_amount_approved > 0
    and oks.oks_delinq_date < :rep_date_end
    and oks.oks_close_state = 0
    and :rep_date_end between oks.oks_date_begin and oks.oks_date_end
    and (exists(select 1
                from knclient kc1
                join operkred ok1 on ok1.ok_from_k_id = kc1.kc_from_k_id
                join oksummary oks1 on oks1.oks_from_k_id = ok1.ok_from_k_id
                where oks1.oks_close_state = 0
                  and :rep_date_end between oks1.oks_date_begin and oks1.oks_date_end
                  and kc1.kc_from_c_id = c.c_id
                  and kc1.kc_amount_approved > 0
                  and ok1.ok_type = 4
                  and ok1.ok_ol_is_active = 1
                  and ok1.ok_sh_date > :rep_date_end + 27
                having sum(ok1.ok_os) < 0) or
         exists(select sum(ok2.ok_os)
                from knclient kc2
                join operkred ok2 on ok2.ok_from_k_id = kc2.kc_from_k_id
                where kc2.kc_from_c_id = c.c_id
                  and kc2.kc_amount_approved > 0
                  and ok2.ok_type = 9
                  and ok2.ok_ol_is_active = 1
                having sum(ok2.ok_os) < 0))
  into :c_id,
       :c_name_fio,
       :c_inn
  do
  begin
    for select k.k_id,
               k.k_kontract_no,
               k.k_kontract_date,
               :rep_date_end - oks.oks_delinq_date
    from knclient kc
    join kontract k on k.k_id = kc.kc_from_k_id
    join oksummary oks on oks.oks_from_k_id = k.k_id
    where kc.kc_from_c_id = :c_id
      and kc.kc_amount_approved > 0
      and oks.oks_delinq_date < :rep_date_end
      and oks.oks_close_state = 0
      and :rep_date_end between oks.oks_date_begin and oks.oks_date_end
    into :k_id,
         :k_kontract_no,
         :k_kontract_date,
         :delinq_days
    do
    begin
      select sum(iif(sh_date < :rep_date_end, zad_os, 0)) + sum(iif(sh_date < :rep_date_end, sh_calc_perc, 0)),
             sum(iif(sh_date < :rep_date_end, zad_os, 0)),
             sum(iif(sh_date < :rep_date_end, sh_calc_perc, 0)),
             sum(sh_calc_fine),
             sum(iif(sh_date < :rep_date_end, zad_os, 0)) + sum(iif(sh_date < :rep_date_end, sh_calc_perc, 0)) + sum(iif(sh_date < :rep_date_end, sh_calc_fine, 0))
      from aarep_make_zad_list(:k_id, :rep_date_end, 1)
      where sh_date is not null
      into :delinq_zad,
           :zad_os,
           :zad_perc,
           :zad_fine,
           :zad_all;

      select max(ok.ok_ol_date)
      from operkred ok
      where ok.ok_from_k_id = :k_id
        and ok.ok_type = 4
        and ok.ok_ol_is_active = 1
      into :ok_ol_date;

      prepayment_sum = null;
      overpay_sum = null;
      k_state = 'Активный контракт с просрочкой';
      suspend;
    end

    for select k.k_kontract_no,
               k.k_kontract_date,
               sum(ok.ok_os),
               max(ok.ok_ol_date)
    from knclient kc
    join kontract k on k.k_id = kc.kc_from_k_id
    join operkred ok on ok.ok_from_k_id = k.k_id
    join oksummary oks on oks.oks_from_k_id = ok.ok_from_k_id
    where oks.oks_close_state = 0
      and :rep_date_end between oks.oks_date_begin and oks.oks_date_end
      and kc.kc_from_c_id = :c_id
      and kc.kc_amount_approved > 0
      and ok.ok_type = 4
      and ok.ok_ol_is_active = 1
      and ok.ok_sh_date > :rep_date_end + 27
    group by k.k_kontract_no, k.k_kontract_date
    having sum(ok.ok_os) < 0
    into :k_kontract_no,
         :k_kontract_date,
         :prepayment_sum,
         :ok_ol_date
    do
    begin
      delinq_days = null;
      delinq_zad = null;
      zad_os = null;
      zad_perc = null;
      zad_fine = null;
      zad_all = null;
      overpay_sum = null;
      k_state = 'Активный контракт с досрочным погашением';
      suspend;
    end

    for select k.k_kontract_no,
               k.k_kontract_date,
               sum(ok.ok_os),
               max(ok.ok_ol_date)
    from knclient kc
    join kontract k on k.k_id = kc.kc_from_k_id
    join operkred ok on ok.ok_from_k_id = k.k_id
    where kc.kc_from_c_id = :c_id
      and kc.kc_amount_approved > 0
      and ok.ok_type = 9
      and ok.ok_ol_is_active = 1
    group by k.k_kontract_no, k.k_kontract_date
    having sum(ok.ok_os) < 0
    into :k_kontract_no,
         :k_kontract_date,
         :overpay_sum,
         :ok_ol_date
    do
    begin
      delinq_days = null;
      delinq_zad = null;
      zad_os = null;
      zad_perc = null;
      zad_fine = null;
      zad_all = null;
      prepayment_sum = null;
      k_state = 'Закрытый контракт с переплатой';
      suspend;
    end
  end
end
