CREATE PROCEDURE HP_TRANSFERTMC_PRINT_QR(
    TRT_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    TRT_NUMBER D_INTEGER,
    TRT_DATE D_DATE,
    TRT_SIGN_DATE D_DATE,
    OFFICES_NAME D_STR_CODE,
    OFFICER_NAME D_STR_CODE,
    MOLS_NAME D_STR_100,
    MOLR_NAME D_STR_CODE,
    MOLS_JT D_STR_100,
    MOLR_JT D_STR_100,
    TRT_MOLS_SIGN_DATE D_DATETIME,
    TRT_MOLR_SIGN_DATE D_DATETIME,
    TRT_STATUS D_SMALLINT,
    STATUS D_STR_CODE,
    WPS_NAME D_STR_100,
    WPR_NAME D_STR_100,
    INVENT_NO D_INTEGER,
    INVENT_STR D_STR_CODE,
    ML_NAME D_STR_255,
    ML_MODEL D_STR_255,
    YL_CODE D_STR_20,
    QTY D_CURRENCY,
    SUMMA D_CURRENCY,
    STATE_NAME D_STR_255,
    STATE_NOTE D_STR_100,
    COMPANY_NAME D_STR_255,
    MOLS_LOGIN D_STR_CODE,
    MOLR_LOGIN D_STR_CODE,
    MOLS_QR D_STR_LONG,
    MOLR_QR D_STR_LONG)
AS
declare variable TRT_MOLS_ID D_ID;
declare variable TRT_MOLR_ID D_ID;
declare variable TRT_MOLS_EDOC_ID D_ID;
declare variable TRT_MOLR_EDOC_ID D_ID;
declare variable EDL_ID D_ID;
declare variable STATE_RD_ID D_INTEGER;
begin
  company_name = f_company_name('short', 'rus');
  for
    select
      trt_id,
      trt_number,
      trt_date,
      trt_sign_date,
      offices_name,
      officer_name,
      trt_mols_id,
      mols_name,
      trt_molr_id,
      molr_name,
      trt_mols_sign_date,
      trt_molr_sign_date,
      trt_mols_edoc_id,
      trt_molr_edoc_id,
      trt_status,
      status
    from hp_transfertmc_view(f_format('{"trt_id":%1}', :trt_id))
  into
    :trt_id,
    :trt_number,
    :trt_date,
    :trt_sign_date,
    :offices_name,
    :officer_name,
    :trt_mols_id,
    :mols_name,
    :trt_molr_id,
    :molr_name,
    :trt_mols_sign_date,
    :trt_molr_sign_date,
    :trt_mols_edoc_id,
    :trt_molr_edoc_id,
    :trt_status,
    :status
  do
  begin
    if (trt_mols_sign_date is not null) then
      mols_login = (
        select first 1 un_name
        from username
        where un_from_c_id = :trt_mols_id
        order by un_id desc);
    if (trt_molr_sign_date is not null) then
      molr_login = (
        select first 1 un_name
        from username
        where un_from_c_id = :trt_molr_id
        order by un_id desc);

    mols_jt = (
      select jl_name
      from hr_pstatus
        inner join joblist
          on jl_id = ps_from_jl_id
      where ps_from_c_id = :trt_mols_id
        and :trt_date between ps_date_begin and ps_date_end
        and ps_is_working = 1);
    molr_jt = (
      select jl_name
      from hr_pstatus
        inner join joblist
          on jl_id = ps_from_jl_id
      where ps_from_c_id = :trt_molr_id
        and :trt_date between ps_date_begin and ps_date_end
        and ps_is_working = 1);

    mols_qr = f_edoc_qr_by_edl(:trt_mols_edoc_id, :trt_mols_id);
    molr_qr = f_edoc_qr_by_edl(:trt_molr_edoc_id, :trt_molr_id);
    /*
    if (coalesce(add_params, '') <> '') then
    begin
      edl_id = f_json_str_get_value(:add_params, 'edl_id', null);
      if (edl_id = trt_mols_edoc_id) then
        mols_qr = f_edoc_qr_by_edl(:edl_id, :trt_mols_id);
      if (edl_id = trt_molr_edoc_id) then
        molr_qr = f_edoc_qr_by_edl(:edl_id, :trt_molr_id);
    end
    */
    for
      select
        wps_name,
        wpr_name,
        invent_no,
        invent_str,
        ml_name,
        ml_model,
        yl_code,
        qty,
        summa,
        state_rd_id,
        state_note
      from hp_transfertmc_viewd(:trt_id)
    into
      :wps_name,
      :wpr_name,
      :invent_no,
      :invent_str,
      :ml_name,
      :ml_model,
      :yl_code,
      :qty,
      :summa,
      :state_rd_id,
      :state_note
    do
    begin
      state_name = (select rd_name from refdata where rd_id = :state_rd_id);
      suspend;
    end
  end
end
