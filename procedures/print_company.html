CREATE PROCEDURE PRINT_COMPANY(
    REG_ID INTEGER,
    ONLY_ACTIVE INTEGER,
    REP_DATE DATE,
    MANAGER_ID INTEGER = 0)
RETURNS (
    KSR_KONT_TITUL VARCHAR(1024),
    KSR_KONT_POSITION VARCHAR(255),
    KSR_OTHER_1 VARCHAR(255),
    KSR_OTHER_2 VARCHAR(255),
    KSR_OTHER_3 VARCHAR(255),
    KSR_IS_ACTIVE INTEGER,
    KSR_FIO VARCHAR(100),
    HRS_JOBMEMO VARCHAR(255),
    KSR_FROM_C_ID DECIMAL(18,0),
    KSR_FROM_REG_ID INTEGER,
    KSR_ID INTEGER,
    KSR_REGION_NAME VARCHAR(100),
    KSR_REGION_ADDRESS VARCHAR(255),
    KSR_REGION_COMPANY VARCHAR(255),
    KSR_REGIST_NO VARCHAR(255),
    KSR_PASS_NO VARCHAR(30),
    KSR_PASS_DATE DATE,
    KSR_PASS_WHO VARCHAR(255),
    KSR_FULL_ADRESS VARCHAR(255),
    KSR_REG_ADDRESS VARCHAR(255),
    KSR_REG_CITY VARCHAR(255),
    KSR_CAN_SIGN_LOANS INTEGER,
    KSR_REG_COMPANY_NAME STR_255,
    KSR_REG_COMPANY_FULL_NAME STR_255,
    KSR_INN VARCHAR(20),
    C_ALTERNAT_FIO STR_100,
    KSR_QR_CODE D_STR_8100,
    KSR_REG_COMPANY_NAME_SHORT D_STR_255,
    REG_NAME_ALTERNAT D_STR_100,
    KSR_REGION_NAME_SHORT D_STR_100,
    KSR_REGIST_NUM D_STR_CODE,
    KSR_REGIST_DATE D_DATE)
AS
declare variable db_id d_integer;
declare variable ksm_is_active d_integer;
declare variable dl_from_reg_id d_integer;
declare variable crlf d_str_lang;
begin
  db_id = f_get_db_id();
  manager_id = coalesce(manager_id, 0);
  crlf = ascii_char(13) || ascii_char(10);

  if (datediff(day from rep_date to current_date) > 45) then
    rep_date = current_date;

  ksr_reg_company_name_short = f_company_name('short', iif(db_id = 94401, 'local', 'rus'), rep_date);
  ksr_reg_company_full_name = f_company_name('full', iif(db_id = 94401, 'local', 'rus'), rep_date);

  if (db_id = 81001) then
    ksr_qr_code = ksr_reg_company_full_name || ', № в гос. реестре МФО 2120177001168, тел. +7 (495) 504-37-04' || crlf ||
    'Правила предоставления микрозаймов: https://www.molbulak.ru/loans/about/docs' || crlf ||
    'Сайт: https://www.molbulak.ru' || crlf ||
    'Сайт СРО: https://npmir.ru' || crlf ||
    'Форма для подачи жалоб и обращений в СРО: https://npmir.ru/' || crlf ||
    'Сайт Банка России: https://cbr.ru/' || crlf ||
    'Государственный реестр МФО: https://cbr.ru/microfinance/registry/' || crlf ||
    'Интернет приемная Банка России: https://www.cbr.ru/reception' || crlf ||
    'Интернет приемная ФССП: http://fssprus.ru/form' || crlf ||
    'Сайт финансового уполномоченного: https://finombudsman.ru/';
  else
    ksr_qr_code = '';

  select oml_regist_no
  from outmailnum
  rows 1
  into :ksr_regist_num;

  if (coalesce(ksr_regist_num, '') <> '') then
    ksr_regist_date = current_date;

  select
         case plt_short_name_side
           when 0 then plt_short_name
           when 1 then pl_name || ' ' || plt_short_name
           else plt_short_name || ' ' || pl_name
         end,
         reg_name
  from region, places, addrlist, placetype
  where reg_address_fact = adl_id and
        adl_from_pl_id = pl_id and
        plt_id = pl_status and
        reg_unicode = :reg_id
  into ksr_reg_city, ksr_region_name_short;

  -- Анатолий �©егольский, логика после поломки бланков эл. документов, когда отключили Бактыгуль
  -- Если передан manager_id, мы не будем смотреть действителен ли он (only_active, ksr_is_active, ksm_is_active)
  -- Если менеджер не передан, возвращаем всех активных (only_active=1), или всех, включая неактивных (only_active=0)

  for select distinct ksr_id, ksr_kont_titul, ksr_kont_postion, ksr_other_1, ksr_other_2, ksr_other_3, ksr_is_active,
                      ksm_is_active, c_name_fio, ksm_from_c_id, ksr_from_reg_id, reg_code || ' ' || reg_name,
                      al.adl_address_str, reg_kont_company, reg_regist_no, c_pass_no, c_pass_date, c_pass_who,
                      ac.adl_address_str, af.adl_address_str, reg.reg_company_name, coalesce(c_name_eng, ''), c_inn,
                      reg.reg_name_alternat
      from kontsignreg, clients, kontsignmen, region_list(:reg_id, 1, -1, 0),
           region reg, addrlist al, addrlist af, addrclient, addrlist ac
      where id_list = ksr_from_reg_id and
            ksm_id = ksr_from_ksm_id and
            ksm_from_c_id = c_id and
            reg_unicode = :reg_id and
            reg_address_legal = al.adl_id and
            reg_address_fact = af.adl_id and
            ac.adl_id = adc_from_adl_id and
            adc_from_c_id = c_id and
            adc_address_type = 1 and
            (
              -- Менеджер передан, возвращаем только его
              (:manager_id <> 0 and ksr_id = :manager_id) or
              (
                -- Менеджер не передан, возвращаем с учетом дат
                :manager_id = 0 and
                coalesce(ksr_date_begin, '01.01.1990') <= current_date and
                coalesce(ksr_date_end, '31.12.9999') >= current_date and
                (
                  -- Всех активных, когда only_active = 1
                  (:only_active = 1 and ksr_is_active = 1 and ksm_is_active = 1) or
                  -- И активных и неактивных, когда only_active = 0
                  (:only_active = 0)
                )
              )
            )
      order by c_name_fio
      into :ksr_id, :ksr_kont_titul, :ksr_kont_position, :ksr_other_1,
           :ksr_other_2, :ksr_other_3, :ksr_is_active, :ksm_is_active, :ksr_fio,
           :ksr_from_c_id, :ksr_from_reg_id, :ksr_region_name,
           :ksr_region_address, :ksr_region_company, :ksr_regist_no,
           :ksr_pass_no, :ksr_pass_date, :ksr_pass_who, :ksr_full_adress,
           :ksr_reg_address, :ksr_reg_company_name, c_alternat_fio, :ksr_inn,
           :reg_name_alternat
  do
  begin
    ksr_is_active = :ksr_is_active * :ksm_is_active;
    ksr_can_sign_loans = 0;
    if (rep_date is not null) then
    begin
      for select distinct dvc_from_reg_id, jl_memo
          from dovlist, dovcateg, dovclients, hr_pstatus, joblist
          where dl_from_dc_id = dc_id and
                dl_id = dvc_from_dl_id and
                dvc_from_c_id = :ksr_from_c_id and
                dc_sign_loans = 1 and
                dl_status = 1 and
                dl_date_begin <= :rep_date and
                dl_date_end >= :rep_date and
                ((dl_date_end_fact is null) or
                (dl_date_end_fact >= :rep_date))
                and
                ps_from_c_id = dl_manager_id and
                ps_from_jl_id = jl_id and
                ps_date_begin <= dl_date_begin and
                ps_date_end >= dl_date_begin and
                ps_is_working = 1
          into :dl_from_reg_id, :hrs_jobmemo
      do
      begin
        if (dl_from_reg_id is not null) then
        begin
          if (exists(select *
                     from region_list(:reg_id, 1, -1, 0)
                     where id_list = :dl_from_reg_id)) then
          begin
            ksr_can_sign_loans = 1;
          end
        end
        else
          ksr_can_sign_loans = 1;
      end
    end

    if (db_id = 81001 and
        ksr_can_sign_loans = 0 and
        exists(select *
               from hr_pstatus
               inner join joblist on jl_id = ps_from_jl_id
               inner join jobkind on jk_id = jl_kind and
                     jk_system_code = 'JobSystem.LO_Standart'
               where ps_from_c_id = :ksr_from_c_id and
                     :rep_date between ps_date_begin and ps_date_end and
                     ps_from_reg_id = :reg_id and
                     ps_is_working = 1 and
                     ps_status = 1)) then
      ksr_can_sign_loans = 1;

    suspend;
  end
end
