CREATE PROCEDURE HP_HR_PSTATUS_ORDER_LAST(
    EMP_WORK_PLACE_ID D_INTEGER,
    EMP_ID D_ID)
RETURNS (
    PS_ID D_INTEGER,
    HO_ID ID_TYPE,
    RS_TEXT D_STR_LONG)
AS
declare variable ps_from_dep_id d_integer;
declare variable ps_from_jl_id d_integer;
declare variable office_name d_str_100;
declare variable dep_name d_str_100;
declare variable jl_name d_str_100;
declare variable c_name_fio d_str_100;
declare variable c_inn d_str_100;
declare variable sawt_name d_str_100;
declare variable ho_order_type d_integer;
declare variable ho_text d_str_255;
begin
  -- Возвращает последний приказ о найме (ротации, изменении условий) по заданному действующему сотруднику (RFCAZ-372)
  -- emp_work_place_id - место работы (офис) сотрудника

  select ps_id,
         ps_from_ho_id,
         ps_from_dep_id,
         ps_from_jl_id,
         ho_order_type,
         left(ho_order_no || ' от ' || f_datetostr(ho_date) || '. ' || ho_text, 255)
  from hr_pstatus
  inner join hr_orders on ho_id = ps_from_ho_id
  where ho_status = 1 and -- утвержденные
        ho_order_type between 1 and 3 and -- прием на работу, ротация, изменение условий договора
        ps_from_reg_id = :emp_work_place_id and
        ps_from_c_id = :emp_id and
        ps_status = 1 and
        ps_is_working = 1 and
        current_date between ps_date_begin and ps_date_end
  into ps_id,
       ho_id,
       ps_from_dep_id,
       ps_from_jl_id,
       ho_order_type,
       ho_text;

  if (ps_id is null or
      ho_id is null) then
    exit;

  select c_inn,
         c_name_fio
  from clients
  where c_id = :emp_id
  into c_inn,
       c_name_fio;

  select reg_code || ' ' || reg_name
  from region
  where reg_unicode = :emp_work_place_id
  into office_name;

  select dep_name
  from depart
  where dep_id = :ps_from_dep_id
  into dep_name;

  select jl_name
  from joblist
  where jl_id = :ps_from_jl_id
  into jl_name;

  select sawt_name
  from sa_worktype
  where sawt_id = :ho_order_type
  into sawt_name;

  rs_text = '<body><h3> ' || c_name_fio || '</h3>';
  rs_text = rs_text || '<p>ИНН (ПИН):<b> ' || c_inn || '</b></p>';
  rs_text = rs_text || '<p>Офис:<b> ' || office_name || '</b></p>';
  rs_text = rs_text || '<p>Отдел:<b> ' || dep_name || '</b></p>';
  rs_text = rs_text || '<p>Должность:<b> ' || jl_name || '</b></p>';
  rs_text = rs_text || '<p>Тип приказа:<b> ' || sawt_name || '</b></p>';
  rs_text = rs_text || '<p>Приказ:<b> ' || ho_text || '</b></p>';

  suspend;
end
