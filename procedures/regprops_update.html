CREATE PROCEDURE REGPROPS_UPDATE(
    REG_ID INTEGER)
AS
declare variable d_prior date;
declare variable rp_id integer;
declare variable rp_date_begin date;
declare variable rp_date_end date;
begin
  /* Procedure Text */
  d_prior = '31.12.9999';
  for select rp_id, rp_date_begin, rp_date_end
      from regprops
      where rp_from_reg_id = :reg_id
      order by rp_date_begin desc, rp_id desc
      into :rp_id, :rp_date_begin, :rp_date_end
--  as cursor cur
  do
  begin
    if ((rp_date_end <> d_prior) or (rp_date_end is null)) then
    begin
      update regprops
      set rp_date_end = :d_prior
      where rp_id = :rp_id;
--        Where current of cur;
    end
    d_prior = rp_date_begin - 1;
  end
end
