CREATE PROCEDURE HP_CFP_POSITION_QGROUPS_SET(
    CP_ID D_INTEGER,
    GR_LST D_STR_8100,
    UN_ID D_ID)
AS
declare variable cg_id d_integer;
begin
  update cfp_position_qgroups set
    cpg_is_delete = 1,
    cpg_from_un_id = :un_id
  where cpg_from_cp_id = :cp_id
    and not (:gr_lst containing cpg_from_cp_id)
    and cpg_is_delete = 0;

  for
    select
      res_str
    from hp_split_str(:gr_lst, ';')
  into
    :cg_id
  do
  begin
    if (not exists(
      select *
      from cfp_position_qgroups
      where cpg_from_cp_id = :cp_id
        and cpg_from_cg_id = :cg_id
        and cpg_is_delete = 0)) then
    update or insert into cfp_position_qgroups (
      cpg_from_cp_id,
      cpg_from_cg_id,
      cpg_is_delete,
      cpg_from_un_id
    ) values (
      :cp_id,
      :cg_id,
      0,
      :un_id
    ) matching (
      cpg_from_cp_id,
      cpg_from_cg_id);
  end
end
