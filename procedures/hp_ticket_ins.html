CREATE PROCEDURE HP_TICKET_INS(
    TKT_GROUP D_ID,
    TKT_KIND D_INTEGER,
    TKT_SOURCE D_INTEGER,
    TKT_OBJECT_KIND D_INTEGER,
    TKT_OBJECT_ID D_ID,
    TKT_DATETIME D_DATETIME = null)
RETURNS (
    TKT_ID D_ID)
AS
begin
  insert into ticket (
      tkt_group,
      tkt_kind,
      tkt_source)
  values (
      :tkt_group,
      :tkt_kind,
      :tkt_source)
  returning tkt_id
  into tkt_id;

  insert into ticketlog (tkl_from_tkt_id, tkl_version, tkl_object_kind, tkl_object_id, tkl_datetime)
  values (:tkt_id, 0, :tkt_object_kind, :tkt_object_id, :tkt_datetime);

  Suspend;
end
