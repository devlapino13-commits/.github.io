CREATE PROCEDURE HP_DOCRECOG_CHANGE_TYPE(
    CDS_ID D_INTEGER,
    NEW_PK_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable old_drm_id d_integer;
declare variable new_drm_id d_integer;
declare variable drl_id d_id;
declare variable old_pk_id d_integer;
declare variable drv_id d_id;
declare variable drf_code d_str_255;
declare variable drf_id d_integer;
declare variable new_is_pass d_smallint;
declare variable old_is_main_doc d_smallint;
begin
   -- Процедура заменяет тип документа (pk_id, drm_id) для ClientDocs и DocRecogList
   -- Для DocRecogList алгоритм следующий:
   -- Значения полей из DocRecogValue совпадающие по drf_code,
   -- переезжают со старого drm_id на новый drm_id, по сути обновляется drv_from_drf_id
   -- Если в новом документа нет поля с таким кодом, мы его не обновляем и не удаляем.
   -- Сделано для того, чтобы если тип документа поменяли по ошибке,
   -- можно было вернуть старый тип, и при этом поля сохранились.
   -- Т.е. паспорт превратили в справку, у которой есть только № документа и дата,
   -- чтобы остальные поля не исчезли, когда тип обратно сделают паспортом.
   -- Минус что docrecogvalue одного документа будет имет поля от двух типов, старого и нового.
   -- Это нужно учитывать при запросе - drf_from_drm_id = drl_from_drm_id
  res_text = '';

  if (coalesce(cds_id, 0) = 0) then
    res_text = 'Не указан документ для изменения типа';

  if (coalesce(new_pk_id, 0) = 0) then
    res_text = 'Не указан новый тип документа';

  if (res_text = '') then
  begin
    select pk_from_drm_id, pdt_is_main_doc
    from passkind, passdoctype
    where pk_id = :new_pk_id
      and pk_from_pdt_id = pdt_id
    into new_drm_id, new_is_pass;

    select cds_from_drl_id, cds_from_pk_id, cds_is_main_doc
    from clientdocs
    where cds_id = :cds_id
    into drl_id, old_pk_id, old_is_main_doc;

    if (drl_id is not null) then
    begin
      select drl_from_drm_id
      from docrecoglist
      where drl_id = :drl_id
      into old_drm_id;
    end

    if (old_drm_id is not distinct from new_drm_id) then
      res_text = 'Старый и новый тип документа одинаковые. Изменений не производилось.';

    if (old_is_main_doc = 1 and new_is_pass <> 1) then
      res_text = 'Основной документ можно поменять только на другой паспорт.';
  end

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  -- Перенос полей Docrecog
  if (new_drm_id is not null) then
  begin
    -- Выбираем все поля, для конвертации, кроме f_полей
    for select drv_id, drf_code
    from docrecogvalue left join docrecogfield on drv_from_drf_id  = drf_id
    where drv_from_drl_id = :drl_id
      and drf_code not starting 'f_'
    into drv_id, drf_code do
    begin
      -- Ищем в новом документе поле с таким же кодом
      drf_id = null;
      select drf_id
      from docrecogfield
      where drf_code = :drf_code
        and drf_from_drm_id = :new_drm_id
      into drf_id;

      -- Если нашли, заменяем
      if (drf_id is not null) then
        update docrecogvalue
        set drv_from_drf_id = :drf_id
        where drv_id = :drv_id;
    end

    -- Досоздаем требуемые поля.
    -- Т.е. когда поле не пришло с клиента, но нам нужно чтобы оно было создано
    for select drf_id
    from docrecogfield
    left join docrecogvalue on drv_from_drf_id = drf_id and
          drv_from_drl_id = :drl_id
    where drf_inactive = 0 and
          drf_from_drm_id = :new_drm_id and
          drv_id is null
    into drf_id
    do
    begin
      insert into docrecogvalue (drv_from_drl_id, drv_from_drf_id, drv_is_accepted, drv_recog_value, drv_final_value)
      values (:drl_id, :drf_id, 0, '', '');
    end

    -- Обновляем сам документ
    update docrecoglist
    set drl_from_drm_id = :new_drm_id,
        drl_from_pk_id = :new_pk_id
    where drl_id = :drl_id;
  end

  if (old_pk_id is distinct from new_pk_id) then
    update clientdocs
    set cds_from_pk_id = :new_pk_id,
        cds_from_un_id = :un_id
    where cds_id = :cds_id;

  suspend;
end
