CREATE PROCEDURE HP_FPS_LOCK_KONRACT(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    K_AMOUNT_APPROVED D_CURRENCY,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    C_ALTERNAT_FIO D_STR_100,
    C_NAME_ENG D_STR_100,
    CELL_NO D_STR_20,
    FCB_ID D_INTEGER,
    FPB_CODE D_STR_CODE,
    FPB_NAME D_STR_255,
    MNT_DESCRIPTION D_STR_255,
    ERROR_TEXT D_STR_255,
    MNT_ID D_STR_10)
AS
declare variable k_status d_integer;
declare variable k_disb_date d_date;
declare variable k_disb_kind d_integer;
declare variable check_count d_integer;
declare variable cell_no_main d_str_code;
declare variable pr_code d_str_100;
declare variable edoc_pmt_project_disb d_str_255;
begin
  -- Проверки перед блокировкой
  error_text = '';

  -- Клиент в договоре должен быть только один
  if (error_text = '' and
      (select count(*)
       from knclient
       where kc_from_k_id = :k_id and
             kc_amount_approved > 0) <> 1) then
    error_text = 'В договоре должен быть только один клиент, выдача невозможна.';

  -- Проверяем состояние, возможно займ уже выдан
  if (error_text = '' and
      exists(select ks_state
             from kontstate
             where ks_from_k_id = :k_id and
                   ks_state = 1 and
                   ks_ol_is_delete = 0)) then
    error_text = 'Договор уже активный, займ ранее выдан.';

  -- Проверяем, что займ в процессе и дату выдачи
  if (error_text = '') then
  begin
    edoc_pmt_project_disb = f_get_const('edoc_pmt_project_disb');

    select k_status,
           k_disb_date,
           k_disb_kind,
           pr.pr_code
    from kontract
    inner join product pr on pr.pr_id = k_from_pr_id
    where k_id = :k_id
    into k_status,
         k_disb_date,
         k_disb_kind,
         pr_code;

    mnt_id = iif(:pr_code containing '.oivo_pack.', f_get_const(f_format('%1_oivo_acc', :edoc_pmt_project_disb)), f_get_const(f_format('%1_acc', :edoc_pmt_project_disb)));

    if (k_disb_kind <> 7) then
      error_text = 'Тип выдачи должен быть СБП.';

    if (k_status <> 1) then
      error_text = 'Договор должен быть утвержден.';

    if (error_text = '' and
        k_disb_date <> current_date) then
      error_text = 'Дата выдачи отличается, выдача невозможна.';
  end

  -- Снимаем старые блокировки
  delete from fps_locks
  where datediff(minute, fpl_datetime, current_timestamp) >= 30;

  -- Проверяем блокировку
  if (error_text = '' and
      exists(select *
             from fps_locks
             where fpl_from_k_id = :k_id)) then
    error_text = 'Договор уже в процессе выдачи (возможно другим пользователем). Выберите следующий договор.';

  if (error_text = '') then
  begin
    -- Собираем информацию для оплаты
    select k_kontract_no, k_amount_approved, c_id, c_name_fio, c_alternat_fio,
           c_name_eng
    from kontract, knclient, clients
    where k_id = :k_id and
          kc_from_k_id = k_id and
          kc_from_c_id = c_id and
          kc_amount_approved > 0
    into k_kontract_no, k_amount_approved, c_id, c_name_fio, c_alternat_fio,
         c_name_eng;

    -- Получаем банк для выплаты, и номер телефона
    select fcb_id, fpb_code, fpb_name, f_cont_code(fcb_from_clt_id)
    from knclient, fps_loan_banks, fps_client_banks, fps_banks
    where kc_from_k_id = :k_id and
          kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          flb_from_kc_id = kc_id and
          fbl_from_fcb_id = fcb_id and
          fcb_from_fpb_id = fpb_id and
          fpb_status = 1 -- Активен
    into fcb_id, fpb_code, fpb_name, cell_no;

    if (error_text = '' and
        coalesce(fpb_code, '') = '') then
      error_text = 'Банк клиента по СБП не указан';

    if (error_text = '' and
        coalesce(cell_no, '') = '') then
      error_text = 'Номер телефона клиента по СБП не указан';

    -- Проверяем, что номер телефона не изменился
    if (error_text = '') then
    begin
      cell_no_main = f_cont(c_id, 'cell', 'ok', 'main');
      if (cell_no_main is distinct from cell_no) then
        error_text = 'Номер телефона изменился. Верните договор в процесс и заново выберите банк по СБП';
    end
  end

  -- Проверяем превышение лимита проверок в сутки
  -- За последние сутки не должно быть более 5 проверок
  if (error_text = '') then
  begin
    check_count = 0;
    select count(*)
    from fps_client_banks_checks
    where fcc_from_fcb_id = :fcb_id and
          datediff(hour, fcc_datetime, current_timestamp) <= 24
    into check_count;

    if (check_count >= 5) then
      error_text = 'Превышен лимит проверок клиента в СБП. Выдача через СБП невозможна. Используйте другие способы выдачи займа.';
  end

  if (error_text = '') then
  begin
    mnt_description = 'Выдача займа по договору №' || k_kontract_no || ', ' || c_name_fio || ', без НДС';

    insert into fps_locks (fpl_from_k_id, fpl_from_un_id)
    values (:k_id, :un_id);
    when gdscode unique_key_violation do
      -- Не смогли вставить из-за того что запись уже есть
      error_text = 'Договор уже в процессе выдачи (возможно другим пользователем). Выберите следующий договор.';
  end

  suspend;
end
