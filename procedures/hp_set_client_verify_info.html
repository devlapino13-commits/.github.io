CREATE PROCEDURE HP_SET_CLIENT_VERIFY_INFO(
    ACL_ID ID_TYPE NOT NULL,
    SUM_PAYMENT CURR_TYPE,
    SUM_DEBT CURR_TYPE,
    DAYS_DELAY STR_255,
    FSSP_DATA STR_255,
    EXTB_DATA STR_255,
    FMS_DATA STR_255,
    EFRSB_DATA STR_255,
    COMMENT STR_255,
    CR_LEVEL INTEGER,
    UN_ID ID_TYPE NOT NULL)
AS
declare variable acm_memo d_blob_text;
begin
  sum_payment = coalesce(sum_payment, 0);
  sum_debt = coalesce(sum_debt, 0);
  days_delay = coalesce(days_delay, 'нет данных');
  fssp_data = replace(coalesce(fssp_data, ''), '"', '''');
  extb_data = replace(coalesce(extb_data, ''), '"', '''');
  fms_data = replace(coalesce(fms_data, ''), '"', '''');
  efrsb_data = replace(coalesce(efrsb_data, ''), '"', '''');
  comment = replace(coalesce(comment, ''), '"', '''');
  cr_level = coalesce(cr_level, -1);
  if (exists(select *
             from actionlist
             where acl_id = :acl_id and
                   acl_object_kind = 0 and
                   acl_status <> 2)) then
  begin
    update actionlist
    set acl_status = 2,
        acl_from_un_id = :un_id
    where acl_id = :acl_id;
    acm_memo = '{"verify":{"dtver":"' || cast(current_timestamp as timestamp) || '","sumpayment":"' || sum_payment || '","sumdebt":"' || sum_debt || '","daysdelay":"' || days_delay || '","fsspdata":"' || fssp_data || '","extbdata":"' || extb_data || '","fmsdata":"' || fms_data || '","efrsbdata":"' || efrsb_data || '","comment":"' || comment || '","level":"' || cr_level || '"}}';
    update or insert into actionmemo (acm_from_acl_id, acm_memo)
    values (:acl_id, :acm_memo)
    matching (acm_from_acl_id);
  end
end
