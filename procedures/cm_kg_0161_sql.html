CREATE PROCEDURE CM_KG_0161_SQL(
    K_ID ID_TYPE NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable c_id d_id;
declare variable k_last_id d_id;
declare variable pr_code d_str_255;
declare variable pr_code_last d_str_255;
declare variable kol_repay d_integer;
declare variable kol_pros d_integer;
declare variable pr_name d_str_255;
declare variable pr_last_name d_str_255;
begin
  res_error_text = '';
  for select kr.k_disb_date, kc.kc_amount_approved, cl.c_name_fio, cl.c_id, pr.pr_code, pr.pr_name
      from kontract kr
      inner join product pr on pr.pr_id = kr.k_from_pr_id and
            pr.pr_code not containing '.ipf.'
      inner join knclient kc on kr.k_id = kc.kc_from_k_id and
            kc.kc_amount_approved > 0
      inner join clients cl on cl.c_id = kc.kc_from_c_id
      where kr.k_id = :k_id
      into kf_disb_date, kc_amount, res_info_text, :c_id, :pr_code, :pr_name
  do
  begin
    if (:pr_code containing '.express_online.') then exit;

    if (exists(select *
               from kontract kont
               inner join knclient kc on kc.kc_from_k_id = kont.k_id and
                     kc.kc_amount_approved > 0
               inner join kontstate ks on ks.ks_from_k_id = kont.k_id
               where kc.kc_from_c_id = :c_id and
                     ks.ks_ol_is_delete = 0 and
                     ks.ks_state = 1 and
                     :kf_disb_date between ks.ks_date_begin and ks.ks_date_end)) then
    begin
      for select pr.pr_code, pr.pr_name, k.k_id
          from kontract k
          inner join knclient kc on kc.kc_from_k_id = k.k_id and
                kc.kc_amount_approved > 0
          inner join product pr on pr.pr_id = k.k_from_pr_id and
                pr.pr_code not containing '.ipf.'
          inner join kontstate ks on ks.ks_from_k_id = k.k_id
          where kc.kc_from_c_id = :c_id and
                ks.ks_ol_is_delete = 0 and
                ks.ks_state = 1 and
                :kf_disb_date between ks.ks_date_begin and ks.ks_date_end

          into pr_code_last, pr_last_name, k_last_id
      do
      begin
        if (:pr_name = :pr_last_name) then
        begin
          res_error_text = f_format('У клиента есть активный кредит: %1', :pr_name);
          res_error_is_critical = 2;
        end
        if (:pr_code containing '.gr.' and
            :pr_code_last containing '.ind.') then
        begin
          res_error_text = 'У клиента есть активный индивидуальный кредит, выдать групповой невозможно';
          res_error_is_critical = 2;
        end
        if (:pr_code containing '.ind.' and
            :pr_code_last containing '.gr.' and
            :pr_code_last containing '.standart.' and
            :pr_code_last not containing '.consultation.' and
            :pr_code not containing '.consultation.' and
            :pr_code containing '.standart.' and
            :pr_code not containing '.par.') then
        begin
          res_error_text = f_format('У клиента есть активный групповой кредит, выдать %1 невозможно', :pr_name);
          res_error_is_critical = 2;
        end
        if (:pr_code containing '.ind.' and
            :pr_code_last containing '.gr.' and
            :pr_code_last containing '.consultation.' and
            :pr_code containing '.consultation.' and
            :pr_code not containing '.par.') then
        begin
          res_error_text = f_format('У клиента есть активный групповой кредит, выдать %1 невозможно', :pr_name);
          res_error_is_critical = 2;
        end
      end
      with tabl1
      as (select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os) zad_os, sum(ok_perc + ok_sh_perc) zad_perc
          from operkred
          where ok_from_k_id = :k_last_id and
                ok_ol_is_active = 1 and
                ok_type < 7
          group by 1, 2
          having sum(ok_os + ok_sh_os) > 0 or sum(ok_perc + ok_sh_perc) > 0)
      select count(*)
      from tabl1
      into kol_repay;
      :kol_repay = coalesce(:kol_repay, 0);
      select coalesce(sum(oks.oks_delinq_day_prv) - sum(oks.oks_delinq_day), 0)
      from oksummary oks
      where oks.oks_from_k_id = :k_last_id
      into kol_pros;
      if (kol_repay <= 2 and
          :kol_pros <= 7) then
        exit;
      if (:res_error_text = '') then
        exit;
      suspend;
    end
  end
end
