CREATE PROCEDURE HP_ANNUL_IPF_INSURANCE_OPERBOOK(
    EDL_ID D_ID)
AS
declare variable ol_id d_id;
begin
  -- аннулирование бух. проводок по страхованию ИПФ займа (RFCKG-5390)
  -- проводки создаются при edl_status = 1 и аннулируются при аннулировании edoc

  for select iio_ol_id
      from ipf_insurance_operbook
      where iio_edl_id = :edl_id
      into ol_id
  do
  begin
    update operlist
    set is_delete = 1
    where ol_id = :ol_id and
          is_delete = 0;

    delete from ipf_insurance_operbook
    where iio_edl_id = :edl_id and
          iio_ol_id = :ol_id;
  end
end
