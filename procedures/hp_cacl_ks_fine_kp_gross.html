CREATE PROCEDURE HP_CACL_KS_FINE_KP_GROSS(
    EXP D_INTEGER,
    WORKED_DAYS_FACT DECIMAL(18,4),
    WORKING_DAYS_IN_MONTH DECIMAL(18,4),
    PAR30 DECIMAL(18,4),
    KONTR_COUNT DECIMAL(18,4),
    REP_DATE_END D_DATE)
RETURNS (
    RESULT_DEDUCTION D_CURRENCY)
AS
declare variable first_cat d_currency;
declare variable second_cat d_currency;
declare variable third_cat d_currency;
declare variable first_cat_par30 d_currency;
declare variable second_cat_par30 d_currency;
declare variable worked_days_koef decimal(18,4);
declare variable rep_date_month d_integer;
begin
  rep_date_month = extract(month from :rep_date_end);

  first_cat = 4938.2700;
  second_cat = 2469.1400;
  third_cat = 1234.5700;

  first_cat_par30 = 2469.1400;
  second_cat_par30 = 1234.5700;

  result_deduction = 0;

  worked_days_koef = worked_days_fact / working_days_in_month;

  if (:exp >= 1) then
  begin
    if (:par30 > 1 and
        :par30 <= 2) then
      result_deduction = result_deduction + second_cat_par30;
    if (:par30 > 2) then
      result_deduction = result_deduction + first_cat_par30;

    if (worked_days_fact > 0) then
    begin
      kontr_count = round(kontr_count / worked_days_koef);

      -- RFCKG-809 Для сотрудников со стажем < 7 с августа по октябрь 2023 года отдельные условия по выдачам
      -- Для других сотрудников используется старая логика
      if (:rep_date_end between '01.08.2023' and '31.10.2023' and
          :exp < 7) then
      begin
        if (:exp = 3) then
          if (:kontr_count < 10) then
            result_deduction = result_deduction + first_cat;

        if (:exp between 4 and 5) then
        begin
          if (:kontr_count < 10) then
            result_deduction = result_deduction + first_cat;
          if (:kontr_count between 10 and 17) then
            result_deduction = result_deduction + second_cat;
        end

        if (:exp = 6) then
        begin
          if (:kontr_count < 10) then
            result_deduction = result_deduction + first_cat;
          if (:kontr_count between 10 and 17) then
            result_deduction = result_deduction + second_cat;
          if (:kontr_count between 18 and 24) then
            result_deduction = result_deduction + third_cat;
        end
      end
      -- RFCKG-809 END
      else
      begin
      if (:exp = 3) then
      begin
        -- Условия по выдачам на январь месяц отличаются от других месяцев (mantis-413)
        if (:rep_date_month = 1) then
        begin
          if (:kontr_count < 12) then
            result_deduction = result_deduction + first_cat;
        end
        else
        begin
          if (:kontr_count < 15) then
            result_deduction = result_deduction + first_cat;
        end
      end

      if (:exp between 4 and 5) then
      begin
        if (:rep_date_month = 1) then
        begin
          if (:kontr_count < 12) then
            result_deduction = result_deduction + first_cat;
          if (:kontr_count between 12 and 18) then
            result_deduction = result_deduction + second_cat;
        end
        else
        begin
          if (:kontr_count < 15) then
            result_deduction = result_deduction + first_cat;
          if (:kontr_count between 15 and 24) then
            result_deduction = result_deduction + second_cat;
        end
      end
      if (:exp > 5) then
      begin
        if (:rep_date_month = 1) then
        begin
          if (:kontr_count < 12) then
            result_deduction = result_deduction + first_cat;
          if (:kontr_count between 12 and 18) then
            result_deduction = result_deduction + second_cat;
          if (:kontr_count between 19 and 27) then
            result_deduction = result_deduction + third_cat;
        end
        else
        begin
          if (:kontr_count < 15) then
            result_deduction = result_deduction + first_cat;
          if (:kontr_count between 15 and 24) then
            result_deduction = result_deduction + second_cat;
          if (:kontr_count between 25 and 34) then
            result_deduction = result_deduction + third_cat;
        end
      end
      end
    end
  end
  --result_deduction = result_deduction * worked_days_koef;
  suspend;
end
