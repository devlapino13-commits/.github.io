CREATE PROCEDURE AAREP_ADDRLIST(
    ADDRESS_ID DECIMAL(18,0))
RETURNS (
    ADL_ID TYPE OF COLUMN ADDRLIST.ADL_ID,
    ADL_FROM_PL_ID TYPE OF COLUMN ADDRLIST.ADL_FROM_PL_ID,
    ADL_DEPART_NO D_STR_20,
    ADL_DEPART_PREF TYPE OF COLUMN ADDRLIST.ADL_DEPART_PREF,
    ADL_FLAT_NO D_STR_20,
    ADL_BUILDING_NO D_STR_20,
    ADL_SECTION_NO D_STR_20,
    ADL_VLADEN_NO D_STR_20,
    ADL_CROSS_STREET_ID TYPE OF COLUMN ADDRLIST.ADL_CROSS_STREET_ID,
    ADL_ADDRESS_STR TYPE OF COLUMN ADDRLIST.ADL_ADDRESS_STR,
    ADL_COMMENT TYPE OF COLUMN ADDRLIST.ADL_COMMENT,
    ADL_POST_INDEX TYPE OF COLUMN ADDRLIST.ADL_POST_INDEX,
    ADL_IS_EDITED TYPE OF COLUMN ADDRLIST.ADL_IS_EDITED,
    ADL_STREET_ID TYPE OF COLUMN ADDRLIST.ADL_STREET_ID,
    PLACE_FULL VARCHAR(255),
    PLACE_FULL_REGION VARCHAR(255),
    PL_PATH VARCHAR(255),
    PL_NAME VARCHAR(100),
    ADL_IS_WITHOUT_STREET INTEGER,
    ADL_LANG_KIND INTEGER)
AS
declare variable txt varchar(100);
begin
  for select adl_id, adl_from_pl_id, adl_depart_no, adl_depart_pref, adl_flat_no, adl_building_no, adl_section_no,
             adl_cross_street_id, adl_address_str, adl_comment, adl_post_index, adl_is_edited, adl_street_id,
             adl_is_without_street, adl_lang_kind, adl_vladen_no
      from addrlist
      where adl_id = :address_id
      into :adl_id, :adl_from_pl_id, :adl_depart_no, :adl_depart_pref, :adl_flat_no, :adl_building_no, :adl_section_no,
           :adl_cross_street_id, :adl_address_str, :adl_comment, :adl_post_index, :adl_is_edited, :adl_street_id,
           :adl_is_without_street, adl_lang_kind, adl_vladen_no
  do
  begin
    if (adl_vladen_no is null) then
      adl_vladen_no = '';

    place_full_region = '';
    place_full = '';
    pl_name = '';
    if (adl_lang_kind is null) then
      adl_lang_kind = 0;

    if (adl_from_pl_id is not null) then
    begin
      select place_full, place_full_region
      from aarep_places_lang_link(:adl_from_pl_id, :adl_lang_kind)
      into :place_full, :place_full_region;

      select iif(:adl_lang_kind = 0 or coalesce(pl_name_komitet, '') = '', pl_name, pl_name_komitet)
      from places
      where pl_id = :adl_from_pl_id
      into :pl_name;
    end
    pl_path = '';
    for select iif(:adl_lang_kind = 0 or coalesce(pl_name_komitet, '') = '', pl_name, pl_name_komitet)
        from places_list(:adl_from_pl_id, 0, -1, 0), places
        where id_list = pl_id
        into :txt
    do
    begin
      pl_path = :txt || ' ' || :pl_path;
    end

    suspend;
  end
end
