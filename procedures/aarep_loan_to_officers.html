CREATE PROCEDURE AAREP_LOAN_TO_OFFICERS(
    REG_ID D_INTEGER,
    REP_DATE D_DATE,
    UN_ID D_ID)
RETURNS (
    KE_FROM_P_ID D_ID,
    C_NAME_FIO D_STR_100,
    KE_CODE VARCHAR(20),
    KE_FROM_REG_ID D_INTEGER,
    KE_OPER_DAYS D_INTEGER)
AS
declare variable db d_integer;
begin
  db = f_get_db_id();
  for select ke_from_p_id, c_name_fio, ke_code, ke_from_reg_id, ke_oper_days
      from aarep_loan_officers(:reg_id, :rep_date)
      into :ke_from_p_id, :c_name_fio, :ke_code, :ke_from_reg_id, :ke_oper_days

  do
  begin
    if (ke_from_p_id = 1000000000005 and
        db in (39801, 41701) and
        not exists(select *
                   from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.CanSendFolderORIV')
                   where has_access = 1)) then
      continue;

    suspend;
  end
end
