CREATE PROCEDURE HP_DAILY_CHECK_BALANCE_CLOSING
AS
declare variable k_id d_id;
declare variable cs_id d_integer;
declare variable csc_id d_integer;
declare variable dt_curr d_date;
declare variable dt_begin d_date;
declare variable dt_end d_date;
begin
  /*
    Используется в роботе (fsm_id = 114) для генерации запросов баланса
    в МПЦ по закрытым, в течении последних двух дней, займам
   ( для определения существования пере-платы ).
  */
  dt_curr = current_date;
  dt_begin = dt_curr - 2;
  dt_end = dt_curr - 1;
  csc_id = f_get_cardsynccfg_id('check_balance_on_closing');

  for
    select
      k_id
    from kontstate
      inner join kontract
        on k_id = ks_from_k_id
          and k_disb_kind = 6
    where ks_date_begin between :dt_begin and :dt_end
      and ks_state = 2
      and ks_ol_is_delete = 0
  into
    :k_id
  do
  begin
    -- пропускаем если баланс нынче уже проверялся
    if (exists(
      select *
      from cardsync
      where cs_from_k_id = :k_id
        and cs_from_csc_id = :csc_id
        and cs_date = :dt_curr))
    then continue;

    execute procedure card_check_balance_new('kontract', :k_id, true)
    returning_values :cs_id;
  end
end
