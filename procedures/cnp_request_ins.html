CREATE PROCEDURE CNP_REQUEST_INS(
    K_ID D_ID,
    UN_ID D_ID)
AS
declare variable c_id d_id;
declare variable kc_amount_approved d_currency;
declare variable freeze_datetime_end d_datetime;
declare variable loan_sign_datetime d_datetime;
declare variable freeze_time_h d_integer;
begin
  select freeze_datetime_end, freeze_time_h
  from hp_get_freeze_data(:k_id)
  into freeze_datetime_end, freeze_time_h;

  if (freeze_time_h = 0 or freeze_datetime_end < current_timestamp) then
  begin
    delete from freeze_main
    where fzm_from_k_id = :k_id;

    for select kc.kc_from_c_id, kc.kc_amount_approved
        from knclient kc
        where kc.kc_from_k_id = :k_id and
              kc.kc_amount_approved > 0
        into c_id, kc_amount_approved
    do
    begin
  
      if (not exists(select *
                     from cnp_payment_requests cpr
                     where cpr.cpr_from_k_id = :k_id and
                           cpr.cpr_status <> -2)) then
        insert into cnp_payment_requests (cpr_from_c_id, cpr_from_k_id, cpr_amount, cpr_action, cpr_from_un_id)
        values (:c_id, :k_id, :kc_amount_approved, 'pay', :un_id);
    end
  end
  else
  begin
    select coalesce(min(eds_status_datetime), current_timestamp)
    from edoc_sign, edoc_list
    where eds_from_edl_id = edl_id
      and edl_from_edc_id = f_edoc_kind('loan.agreement', 'id')
      and edl_object_kind = 9
      and edl_object_id = :k_id
      and eds_status = 1
      and edl_status = 1
    into loan_sign_datetime;

    update or insert into freeze_main (fzm_from_k_id, fzm_datetime_begin, fzm_datetime_end)
    values (:k_id, :loan_sign_datetime, :freeze_datetime_end)
    matching(fzm_from_k_id);
  end 
end
