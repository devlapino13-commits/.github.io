CREATE PROCEDURE AAREP_MONEY_SALDO_LIST_DAILY(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    RES_REGION_NAME VARCHAR(100),
    RES_ACCOUNT VARCHAR(20),
    RES_ACCOUNT_NAME VARCHAR(200),
    RES_CURRENCY VARCHAR(3),
    RES_SALDO DECIMAL(18,4),
    RES_SALDO_EKV DECIMAL(18,4),
    RES_ACC_KIND INTEGER,
    RES_REGION_ID INTEGER)
AS
declare variable al_id varchar(20);
begin
  for select al_id, res_region_name, res_account, res_account_name, res_currency, res_acc_kind, res_region_id,
             res_saldo, res_saldo_ekv
      from aarep_money_saldo_list_dtmp(:report_region_id, :report_date_end)
      order by res_region_name, res_account
      into :al_id, :res_region_name, :res_account, :res_account_name, :res_currency, :res_acc_kind, :res_region_id,
           :res_saldo, :res_saldo_ekv
  do
  begin
    if (res_saldo is null) then
      res_saldo = 0;
    if (res_saldo_ekv is null) then
      res_saldo_ekv = 0;
    if ((res_saldo <> 0) or (res_saldo_ekv <> 0)) then
      suspend;
  end
end
