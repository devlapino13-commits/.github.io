CREATE PROCEDURE HP_KONTDISBPLAN_CHECK(
    K_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable start_date d_date;
declare variable finish_date d_date;
declare variable lo_id d_id;
declare variable mgr_id d_integer;
declare variable limit_saldo d_currency;
begin
  start_date = '04.03.2024';
  finish_date = '31.12.2999'; -- Здесь указать последний день актуальности контроля над лимитом выдачи
  res_text = '';

  for select k_from_kf_id, k_disb_date, k_last_reg_id, k_status, k_amount_approved
  from kontract
  where k_id = :k_id
  as cursor cur
  do
  begin
    if (cur.k_status <> 1 and cur.k_disb_date between start_date and :finish_date) then
    begin
      select k.kal_new_kf_officer
      from knaktlist k
      where k.kal_from_kf_id = cur.k_from_kf_id and
            k.kal_date <= cur.k_disb_date and
            k.kal_date_end >= cur.k_disb_date and
            k.is_delete = 0
      into lo_id;
  
      mgr_id = f_lo_mgr_id(null, cur.k_disb_date, lo_id, cur.k_last_reg_id);

      limit_saldo = 0;
      select kdp_summa_saldo
      from kontdisbplan p
      where p.kdp_mgr_id = :mgr_id and
            p.kdp_date_begin = cur.k_disb_date
      into limit_saldo;

      if (limit_saldo < cur.k_amount_approved) then
        res_text = 'Исчерпан лимит на выдачу на: ' || f_datetostr(cur.k_disb_date);
    end
  end

  suspend;
end
