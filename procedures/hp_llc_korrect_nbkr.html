CREATE PROCEDURE HP_LLC_KORRECT_NBKR(
    REP_DATE_END DATE)
RETURNS (
    LK_FROM_LM_ID DECIMAL(18,0),
    LK_FROM_K_ID DECIMAL(18,0),
    LK_PERCENT DECIMAL(18,7),
    LK_BALANCE DECIMAL(18,4),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    KC_FROM_C_ID DECIMAL(18,0),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    LK_PERCENT_CLI DECIMAL(18,7),
    KC_DOLYA_PERC DECIMAL(18,7),
    LK_PERC_CLI_NEW DECIMAL(18,7),
    LK_PERCENT_NEW DECIMAL(18,7),
    K INTEGER)
AS
declare variable is_parallel integer;
declare variable lk_percent_max perc_type;
declare variable is_parallel_1 integer;
declare variable lk_is_indiv smallint;
begin
  k = 1;
  /*Делает выборку по РППУ из таблицы llc_kont выданные с 21.05.2012 и % РППУ<25*/
  for select lk_from_lm_id, ks_from_k_id, lk_percent, lk_balance, k_amount_approved, lk_is_indiv
      from kontstate, llc_kont, kontract
      where ks_ol_is_delete = 0 and
            ks_state = 1
            /*and ks_date_begin>='21.05.2012'*/
            and
            :rep_date_end between ks_date_begin and ks_date_end and
            lk_from_k_id = ks_from_k_id and
            lk_date between :rep_date_end - 2 and :rep_date_end + 2
-- and lk_percent < 10
            and
            k_id = ks_from_k_id
      into :lk_from_lm_id, :lk_from_k_id, :lk_percent, :lk_balance, :k_amount_approved, :lk_is_indiv
  do
  begin
-- Проверка на параллельные кредиты в нашей компании
    is_parallel = 0;
    if (exists(select 1
               from kontstate ks1, kontstate ks2, knclient kc1, knclient kc2
-- Выборка только активных для первого контракта
               where ks1.ks_ol_is_delete = 0 and
                     ks1.ks_state = 1 and
                     :rep_date_end between ks1.ks_date_begin and ks1.ks_date_end and
                     ks1.ks_from_k_id = kc1.kc_from_k_id
-- Выборка только активных для второго
                     and
                     ks2.ks_ol_is_delete = 0 and
                     ks2.ks_state = 1 and
                     :rep_date_end between ks2.ks_date_begin and ks2.ks_date_end and
                     ks2.ks_from_k_id = kc2.kc_from_k_id and
                     kc1.kc_from_c_id = kc2.kc_from_c_id -- Один и тот же клиент, первого и второго контракта
-- and kc1.kc_from_c_id = :c_id
                     and
                     kc1.kc_from_k_id <> kc2.kc_from_k_id -- есть активные кредиты в двух разных контрактах
                     and
                     kc1.kc_from_k_id = :lk_from_k_id and
                     kc1.kc_amount_approved > 0 and
                     kc2.kc_amount_approved > 0 and
                     kc1.kc_has_loans = 0 -- Выборка по тем у кого галочки не проставлены
--      and kc1.kc_has_family_loans = 0
--      and cast(ks1.ks_date_begin as date) > cast(ks2.ks_date_begin as date)
        )) then
      is_parallel = 1;

    if (exists(select kc_from_c_id
               from knclient /*Проверка клиента на то, что брал ли кредит из других ФКУ или у родственника есть кредит*/
               where kc_from_k_id = :lk_from_k_id and
                     kc_amount_approved > 0 and
                     (kc_has_loans <> 0 or
                     kc_has_family_loans <> 0)) or (:is_parallel = 1)) then
    begin

      lk_perc_cli_new = 0;
      lk_percent_new = 0;
      /*Вытаскивает клиентов данного контракта с вычислением % РППУ*/
      for select kc3.kc_from_c_id, kc3.kc_amount_approved,
                 case
                   when (kc3.kc_has_loans <> 0 -- or kc_has_family_loans<>0
                 or (exists(select 1
                            from kontstate ks1, kontstate ks2, knclient kc1, knclient kc2
                            where ks1.ks_ol_is_delete = 0 and
                                  ks1.ks_state = 1 and
                                  :rep_date_end between ks1.ks_date_begin and ks1.ks_date_end and
                                  ks1.ks_from_k_id = kc1.kc_from_k_id and
                                  ks2.ks_ol_is_delete = 0 and
                                  ks2.ks_state = 1 and
                                  :rep_date_end between ks2.ks_date_begin and ks2.ks_date_end and
                                  ks2.ks_from_k_id = kc2.kc_from_k_id and
                                  kc1.kc_from_c_id = kc2.kc_from_c_id and
                                  kc1.kc_from_c_id = kc3.kc_from_c_id -- Привязка с внешней таблицей
                                  and
                                  kc1.kc_from_k_id <> kc2.kc_from_k_id and
                                  kc1.kc_amount_approved > 0 and
                                  kc2.kc_amount_approved > 0 and
                                  kc1.kc_has_loans = 0))) then 1 -- 1 это есть параллельный кредит
                   else 0
                 end,
-- 0 нет параллельный кредит
                 kc_amount_approved / :k_amount_approved * 100
          from knclient kc3
          where kc3.kc_from_k_id = :lk_from_k_id and
                kc3.kc_amount_approved > 0
          into :kc_from_c_id, :kc_amount_approved, :is_parallel_1, :kc_dolya_perc
      do
      begin
        lk_percent_cli = :lk_percent;
        if (is_parallel_1 = 1) then
        begin
          select max(lk_percent)
          from kontstate, knclient, llc_kont
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                :rep_date_end between ks_date_begin and ks_date_end and
                lk_from_k_id = ks_from_k_id and
                lk_date between :rep_date_end - 2 and :rep_date_end + 2 and
                ks_from_k_id = kc_from_k_id and
                kc_amount_approved > 0 and
                kc_from_c_id = :kc_from_c_id
          into lk_percent_max;
          if (lk_percent_max > 10) then
            lk_percent_cli = lk_percent_max;
          else
            lk_percent_cli = 10;
        end
        else
        begin
          if (lk_is_indiv = 1 and
              lk_percent_cli < 10 and
              exists(select 1
                     from knclient
                     where kc_from_c_id = :kc_from_c_id and
                           kc_from_k_id = :lk_from_k_id and
                           kc_amount_approved > 0 and
                           kc_has_family_loans <> 0 -- нужно добавить просрочки у родственников клиента
              )) then
            lk_percent_cli = 10;
        end

        lk_perc_cli_new = :lk_percent_cli * :kc_dolya_perc / 100;
        lk_percent_new = lk_percent_new + lk_perc_cli_new;
        suspend;
      end

      /*Исправляет % РППУ в таблице llc_kont*/
      update llc_kont lk_upd
      set lk_upd.lk_percent = :lk_percent_new,
          lk_upd.lk_percent_orig = :lk_percent_new,
          lk_upd.lk_memo = cast(coalesce(lk_upd.lk_memo || '; ', '') as varchar(1024)) || 'Корректирован % РППУ по требованию НБКР',
          lk_upd.lk_status = 99
      where lk_upd.lk_from_lm_id = :lk_from_lm_id and
            lk_upd.lk_from_k_id = :lk_from_k_id;

      k = k + 1;
      /*suspend;*/
    end

-- Меняет классификацию на 1 если меньше или равно 4
-- update llc_kont set lk_status=1
-- where lk_from_lm_id=:lk_from_lm_id
-- and lk_from_k_id=:lk_from_k_id
-- and lk_status<=4;

  end
end
