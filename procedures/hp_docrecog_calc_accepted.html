CREATE PROCEDURE HP_DOCRECOG_CALC_ACCEPTED(
    DRL_ID ID_TYPE)
AS
declare variable drm_id d_integer;
declare variable drf_id d_integer;
declare variable drf_tag d_smallint;
declare variable drf_mrz_id d_integer;
declare variable drf_mrz_value d_str_255;
declare variable mrz_is_accepted d_bool;
begin
  -- Расчет drv_is_accepted_calc для документа.
  -- Базовая проверка: поля с одинаковым drf_tag должны иметь одинаковые значения
  for select drf_from_drm_id, drf_tag
      from docrecogvalue, docrecogfield
      where drv_from_drf_id = drf_id and
            drv_from_drl_id = :drl_id and
            coalesce(drf_tag, 0) <> 0 and
            drf_inactive = 0
      group by drf_from_drm_id, drf_tag
      having min(drv_recog_value) <> max(drv_recog_value)
      into drm_id, drf_tag
  do
  begin
    -- если значения полей с одинаковым TAG отличаются, за выставляем drv_is_accepted_calc = 0
    -- и если одно из этих полей является MRZ полем, то его значение берем за базу и копируем в
    -- остальные обычные поля с идентичным TAG.
    drf_mrz_id = null;
    mrz_is_accepted = 0;
    drf_mrz_value = '';
    select drv_from_drf_id, drv_is_accepted, drv_final_value
    from docrecogvalue, docrecogfield
    where drv_from_drl_id = :drl_id and
          drv_from_drf_id = drf_id and
          drf_inactive = 0 and
          drf_tag = :drf_tag and
          drf_api_code like '%_mrz' or drf_code like '%_mrz' -- Нежелательно. Лучше завести поле, на которое будет приоритет, при одинаковости tag'а
    into :drf_mrz_id, mrz_is_accepted, drf_mrz_value;

    for select drf_id
        from docrecogfield
        where drf_from_drm_id = :drm_id and
              drf_tag = :drf_tag and
              drf_id <> coalesce(:drf_mrz_id, 0)
        into drf_id
    do
    begin
      update docrecogvalue
      set drv_is_accepted_calc = iif(:drf_mrz_id is not null, :mrz_is_accepted, 0),
          drv_final_value = iif(:drf_mrz_id is not null, :drf_mrz_value, drv_final_value)
      where drv_from_drl_id = :drl_id and
            drv_from_drf_id = :drf_id;
    end
  end
  -- Дополнительные проверки можно делать ниже
end
