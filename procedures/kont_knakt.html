CREATE PROCEDURE KONT_KNAKT(
    PAR1 DATE,
    PAR2 DATE,
    PAR_REG INTEGER)
RETURNS (
    K_ID1 DECIMAL(18,0),
    K_CURRENCY1 VARCHAR(3),
    K_REGION1 INTEGER,
    K_KC_ID DECIMAL(18,0),
    KAL_DATE1 DATE)
AS
begin
  for select kontract.k_id, kontract.k_currency, knaktlist.kal_new_reg_id, knaktlist.kal_new_kf_officer,
             knaktlist.kal_date
      from knaktlist, kontract, kontstate, region_list(:par_reg, 1, -1, 1)
      where knaktlist.is_delete = 0 and
            kal_date >= :par1 and
            kal_date <= :par2 and
            kal_from_ka_id <> 0 and
            kontract.k_from_kf_id = kal_from_kf_id and
            ks_from_k_id = kontract.k_id and
            ks_ol_is_delete = 0 and
            kontstate.ks_state = 1 and
            kal_date >= cast(ks_date_begin as date) and
            kal_date <= ks_date_end and
            (knaktlist.kal_new_reg_id = region_list.id_list)
      into :k_id1, :k_currency1, :k_region1, :k_kc_id, :kal_date1
  do
  begin
    suspend;
  end
end
