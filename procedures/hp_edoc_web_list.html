CREATE PROCEDURE HP_EDOC_WEB_LIST(
    UUID D_STR_CODE)
RETURNS (
    RES_JSON D_STR_8100)
AS
declare variable str_list d_str_8100;
declare variable edl_id d_id;
declare variable eds_status d_integer;
declare variable msg_text d_str_255;
declare variable show_sign_button d_smallint;
declare variable show_cancel_button d_smallint;
declare variable sign_button_caption d_str_255;
declare variable cancel_button_caption d_str_255;
declare variable show_bankcard d_smallint;
declare variable edl_status d_smallint;
declare variable obj_id d_id;
declare variable c_id d_id;
declare variable c_name d_str_255;
declare variable obj_code d_str_code;
declare variable edf_print_name d_str_255;
declare variable edb_id d_id;
declare variable is_bold d_smallint;
declare variable obj_kind d_integer;
declare variable cv_status d_integer;
declare variable edc_code d_str_code;
declare variable edf_is_online d_bool;
declare variable show_documents d_smallint;
declare variable edl_blank_date d_date;
declare variable edc_process_kind d_smallint;
declare variable def_checked d_bool;
declare variable eds_id d_id;
declare variable edf_optional_sign d_bool;
declare variable edb_checked d_bool;
declare variable c_pin d_str_20;
begin
  res_json = '';
  msg_text = '';
  c_name = '';
  show_sign_button = 0;
  show_cancel_button = 0;
  show_bankcard = 0;
  show_documents = 0;
  sign_button_caption = 'Подписать документы';
  cancel_button_caption = 'Отказаться от подписания';
  def_checked = 1; -- По-умолчанию для каждого бланка выставлена галочка для подписания
  str_list = '';
  obj_code = '';
  c_pin = '';

  select obj_kind, obj_id
  from hp_uuid(:uuid)
  into obj_kind, obj_id;

  if (obj_id is null)
  then msg_text = 'Неверная ссылка';
  else obj_code = f_objkind(obj_kind, 'name');

  if (obj_code = 'edoc_sign') then
  begin
    eds_id = obj_id;
    select eds_from_edl_id, eds_status, eds_from_c_id, edc_code, edl_blank_date,
           iif(coalesce(edl_date_end, current_date) < current_date, -1, edl_status),
           edc_process_kind
    from edoc_sign, edoc_list, edoc_cfg
    where eds_id = :eds_id and
          eds_from_edl_id = edl_id and
          edl_from_edc_id = edc_id
    into edl_id, eds_status, c_id, edc_code, edl_blank_date, edl_status,
         edc_process_kind;
  
    -- доп.соглашене по реструктуризации действует только 2 дня
    if (msg_text = '' and edl_status = 0 and edc_code = 'loan.reschedule') then
    begin
      if (edl_blank_date < (current_date - 1)) then
      begin
        edl_status = -1;

        update edoc_list
        set edl_status = :edl_status
        where edl_id = :edl_id;
      end
    end

    if (msg_text = '' and edl_status = -1) then msg_text = 'Данные устарели';
  
    if (not exists(
                   select * from edoc_msg
                   where edm_from_eds_id = :eds_id)) then msg_text = 'Неверная ссылка';

    if (msg_text = '' and edc_code in ('bankcard.moneta', 'bankcard.elcart')) then
    begin
      show_bankcard = 1;
      show_sign_button = 1;
      show_cancel_button = 1;
      sign_button_caption = 'Продолжить';
      cancel_button_caption = 'Отказаться';
      msg_text = 'Ввод данных банковской карты';
    end

    if (msg_text = '') then
    begin
      for select edb_id, edf_print_name, edf_is_online, edf_optional_sign,
        iif(eds_status = 1, iif(eds_from_c_id is null, eds_status, edb_sign_status), :def_checked)
      from edoc_sign, edoc_blank, edoc_frx
      where eds_id = :eds_id and
            edb_from_edl_id = eds_from_edl_id and
            edb_from_edf_id = edf_id and
            (edb_from_c_id is null or edb_from_c_id = :c_id) -- выдаем общие документы или конкретного участника
      into edb_id, edf_print_name, edf_is_online, edf_optional_sign, edb_checked
      do
      begin
        is_bold = 1;
        if (exists(
        select * from edoc_open
        where edo_from_eds_id = :eds_id and
              edo_from_edb_id = :edb_id and
              (edo_before_sign_datetime is not null or
               edo_after_sign_datetime is not null))) then is_bold = 0;

        if (str_list <> '') then str_list = str_list || ',';
        str_list = str_list || '{' || f_json_make_pair('name', edf_print_name, 'str') || ',' ||
                                      f_json_make_pair('bold', is_bold, 'num') || ',' ||
                                      f_json_make_pair('id', edb_id, 'num') || ',' ||
                                      f_json_make_pair('is_online', edf_is_online, 'num') || ',' ||
                                      f_json_make_pair('checked', edb_checked, 'num') || ',' ||
                                      f_json_make_pair('opt', edf_optional_sign, 'num') || '}';
      end
      if (str_list = '') then
        msg_text = 'Нет документов для отображения';

    end
  
    if (msg_text = '') then
    begin
      if (eds_status = 1)
      then msg_text = 'Документы уже подписаны';
      else
      begin
        msg_text = 'Документы для подписания';
        show_sign_button = 1;
        show_cancel_button = 1;
      end
    end
  
    if (c_id is not null) then
    begin
      select c_name_fio, c_pincode
      from clients
      where c_id = :c_id
      into c_name, c_pin;
      c_name = coalesce(c_name, '');
      c_pin = coalesce(c_pin, '');
    end
    else
      c_name = 'Контролер';
  end

  if (obj_code = 'contverify') then
  begin
    str_list = '';
    cancel_button_caption = 'Отказаться';
    sign_button_caption = 'Подтвердить';

    select cv_status, c_name_fio
    from contverify, clients
    where cv_id = :obj_id and
          cv_object_kind = 0 and
          cv_object_id = c_id
    into cv_status, c_name;
  
    if (msg_text = '') then
    begin
      if (cv_status >= 1) then
        msg_text = 'Данные уже подтверждены';
      else
      begin
        msg_text = 'Подтверждение данных';
        show_sign_button = 1;
        show_cancel_button = 1;
      end
    end
  end

  if (obj_code = 'contveri') then
  begin
    str_list = '';
    sign_button_caption = 'Подтвердить';
    cancel_button_caption = 'Отказаться';

    select cvr_status, c_name_fio
    from contveri, contdata, clients
    where cvr_id = :obj_id and
          cvr_from_cda_id = cda_id and
          cda_from_c_id = c_id
    into cv_status, c_name;
  
    if (msg_text = '') then
    begin
      if (cv_status = 1) then
        msg_text = 'Данные уже подтверждены';
      else
      begin
        msg_text = 'Подтверждение данных';
        show_sign_button = 1;
        show_cancel_button = 1;
      end
    end
  end

  -- Когда подписание не требуется, контрольно убираем кнопку
  if (edc_process_kind = 0) then
  begin
    show_sign_button = 0;
    show_cancel_button = 0;
  end

  show_cancel_button = 0;
  if (edc_code = 'loan.express') then
    show_cancel_button = 1;

  if (str_list is null) then str_list = '';
  res_json = '{' ||
             f_json_make_pair('message', msg_text, 'str') || ',' ||
             f_json_make_pair('add_message', c_name, 'str') || ',' ||
             f_json_make_pair('show_sign_button', show_sign_button, 'num') || ',' ||
             f_json_make_pair('show_cancel_button', show_cancel_button, 'num') || ',' ||
             f_json_make_pair('show_documents', show_documents, 'num') || ',' ||
             f_json_make_pair('show_bankcard', show_bankcard, 'num') || ',' ||
             f_json_make_pair('ds_provider', f_get_const('edoc_ds_provider'), 'str') || ',' ||
             f_json_make_pair('client_pin', c_pin, 'str') || ',' ||
             iif(show_bankcard = 1,
               f_json_make_pair('bankcard_code', edc_code, 'str') || ',' ||
               f_json_make_pair('client_id', c_id,  'num') ||  ',', '') ||
             f_json_make_pair('sign_button_caption', sign_button_caption, 'str') || ',' ||
             f_json_make_pair('cancel_button_caption', cancel_button_caption, 'str') || ',' ||
             '"list":['||str_list||']}';
  
  Suspend;
end
