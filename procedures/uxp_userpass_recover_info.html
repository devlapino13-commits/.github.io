CREATE PROCEDURE UXP$USERPASS_RECOVER_INFO(
    USER_NAME D_STR_CODE)
RETURNS (
    C_EMAIL D_STR_100,
    C_CELL D_STR_CODE,
    RECOVERY_BY_EMAIL D_INTEGER,
    C_EMAIL_MASKED D_STR_100,
    C_CELL_MASKED D_STR_CODE,
    CAN_RECOVER D_BOOL)
AS
declare variable c_id d_id;
declare variable un_id d_id;
begin
  can_recover = 1;
  if (not exists(select * from hp_db_is_actual)) then
  begin
    can_recover = 0;
    suspend;
    exit;
  end

  select un_id, c_id
  from clients, username
  where un_from_c_id = c_id and
        un_name = :user_name
  into un_id, c_id;

  c_email = f_cont_info(0, c_id, 'email', 'ok');
  c_cell = f_cont_info(0, c_id, 'cell', 'ok');

  if (c_email <> '') then
    c_email_masked = uxf$mask(c_email, 'email');
  else
    c_email_masked = 'Нет подтвержденного эл.адреса';

  if (c_cell <> '') then
    c_cell_masked = uxf$mask(c_cell, 'cell');
  else
    c_cell_masked = 'Нет подтвержденного номера';

  recovery_by_email = 1;
  if (f_get_db_id() in (39801, 41701)) then
  begin
    if (not exists(select * from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.PassRecoveryEMail')
                   where has_access = 1)) then
      recovery_by_email = 0;
  end

  suspend;
end
