CREATE PROCEDURE HP_RELATIVES_ACTIVE_EMPLOYEES
AS
declare variable lo_id d_id;
declare variable rel_other_c_id d_id;
declare variable crel_full_code varchar(50);
begin
  --процедура вызывается роботом, заполняет родственников активных сотрудников
  delete from relatives_active_employees;

  execute statement('SET GENERATOR GEN_REA_ID TO 0;');


  for select ps_from_c_id
      from hr_pstatus h
      where current_date between ps_date_begin and ps_date_end and
            h.ps_is_working = 1 and
            h.ps_status = 1
      into :lo_id
  do
  begin
    for select rel_other_c_id, crel_full_code
        from hp_client_relation_tmp(:lo_id, 4)
        into :rel_other_c_id, crel_full_code
    do
    begin
      insert into relatives_active_employees (rae_from_lo_id, rae_from_c_id, rea_rel_id, rea_is_delete, rea_date)
      values (:lo_id, :rel_other_c_id, :crel_full_code, 0, current_date);
    end
  end
end
